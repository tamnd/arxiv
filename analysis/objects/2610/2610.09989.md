<!-- lemma-1: body_md -->
There exists a family of smooth maps $F_\epsilon:\operatorname{Herm}(n,\mathbb{C})\rightarrow \mathbb{R}$ satisfying the following properties: 
\begin{enumerate}
\item $F_\epsilon$ are concave, i.e., for all $A,A_0\in \operatorname{Herm}(n,\mathbb{C})$ we have $$F_\epsilon(A_0) \leq F_\epsilon(A) + DF_\epsilon(A)(A_0-A).$$
\item (Uniform ellipticity) For every $P\geq 0$, $$\frac{\epsilon}{n}\operatorname{Tr}(P) \leq F_\epsilon(A+P) - F_\epsilon(A) \leq \frac{1}{n\epsilon}\operatorname{Tr}(P).$$
\item (Uniform approximation) There exists a constant $C = C(n)$ such that for all compact sets $K\subset \operatorname{Herm}^+(n,\mathbb{C})$ there exists a constant $\epsilon_K$ with the following property: for all $\epsilon < \epsilon_K$ we have $$\Vert F_\epsilon - F\Vert_{L^\infty(K)}< C\epsilon^2.$$
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $0 < \epsilon \le 1$, we consider the truncated Bellman family: \begin{align*}
\mathcal{B}_\epsilon&:= \{B\in \operatorname{Herm}^+(n,\mathbb{C})~|~ \det B = 1,~\epsilon I\leq B\leq \epsilon^{-1}I\}\\
\widehat{F}_\epsilon(A) &:= \frac{1}{n}\min_{B\in \mathcal{B}_\epsilon}\operatorname{Tr}(AB).
\end{align*}
Then $\widehat{F}_\epsilon$ is concave and uniformly elliptic with the constants $\epsilon/n$ and $1/n\epsilon$. To obtain $F_\epsilon$ we use the standard method of mollifying with an approximation to identity. Since the uniform estimate is crucial to the final step of the proof, we include the details for the convenience of the reader. Identifying $\operatorname{Herm}(n,\mathbb{C})$ with $\mathbb{R}^{n^2}$ we let $\eta$ be the standard radial mollifier satisfying $$\eta \in C^\infty_c(\operatorname{Herm}(n,\mathbb{C})), \quad \eta \ge 0, \quad \mathrm{Supp} \ \eta \subset \{\vert Z\vert<1\},\quad \int_{\operatorname{Herm}(n,\mathbb{C})}\eta(Z)\,dZ = 1,$$ and $$\eta_\sigma(Z) = \sigma^{-n^2}\eta\Big(\frac{Z}{\sigma}\Big).$$ We then let $$F_{\epsilon,\sigma}(A) = \widehat F_\epsilon \star\eta_\sigma(A):=\int_{\operatorname{Herm}(n,\mathbb{C})}\eta(Z)\widehat F_\epsilon(A-\sigma Z)\,dZ.$$ It is easy to see that $F_{\epsilon,\sigma}$ is concave and uniformly elliptic with the same constants $\epsilon/n$ and $1/n\epsilon$. The ellipticity of $\widehat F_\epsilon$ also implies that $\operatorname{Lip}(\widehat F_\epsilon) \leq C_n/\epsilon$ for some dimensional constant $C_n$ and it then follows immediately that for any $A\in \operatorname{Herm}(n,\mathbb{C})$, we have $$|F_{\epsilon, \sigma}(A) - \widehat F_\epsilon(A)| \leq \frac{C_n\sigma}{\epsilon}.$$ We now pick $\sigma = \epsilon^3$ and set $F_\epsilon = F_{\epsilon,\sigma}.$ Then 
\begin{equation}\label{eq:bellman-smoothening-uniform-est} 
|F_\epsilon(A) - \widehat F_\epsilon(A)| \leq C_n \epsilon^2\quad\text{ for all $A\in \operatorname{Herm}(n,\mathbb{C})$.}
\end{equation}
Finally observe that given a compact set $K\subset \operatorname{Herm}^+(n,\mathbb{C})$, then there exists an $\epsilon_K>0$ such that for all $\epsilon< \epsilon_K$ we have $\widehat F_\epsilon(A) = F(A).$ To see this, note that for any $A\in \operatorname{Herm}^+(n,\mathbb{C})$ the Bellman minimiser is completely explicit. Indeed, one can take $$B_A = (\det A)^{1/n}A^{-1}.$$ So if $K\subset \operatorname{Herm}^+(n,\mathbb{C})$ is compact, then there exist $\epsilon_K$ such that $$\epsilon_K \leq B_A \leq \epsilon_K^{-1}\quad \text{ for all $A\in K$},$$ and so $\widehat F_\epsilon(A) = F(A)$ for all $A\in K$ and $\epsilon< \epsilon_K$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\begin{enumerate}
\item For every $A\in \operatorname{Herm}(n,\mathbb{C})$, $\operatorname{Tr}(DF_\epsilon(A)) \geq 1.$
\item For $n\geq 2$, if $A_\epsilon\rightarrow A$ as $\epsilon\rightarrow 0^+$, and $\lambda_{min}(A) <0 $, then $$F_\epsilon(A_\epsilon) \rightarrow -\infty.$$
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The first part follows from the following finite difference inequality and the smoothness of $F_\epsilon$: 
\begin{equation}\label{eq:finite-diff-F-ep}
F_\epsilon(A + tI) - F_\epsilon(A) \geq t,
\end{equation} for $t \ge 0$ and the reversed inequality for $t \le 0$. It is enough to prove this finite difference inequality with $F_\epsilon$ replaced by the truncated Bellman operators $\widehat F_\epsilon$ used above. For $\widehat F_\epsilon$ we have 
\begin{align*}
\widehat F_\epsilon(A+tI) &= \frac{1}{n}\inf_{B\in \mathcal{B}_\epsilon} \operatorname{Tr}(AB + tB)\\
&= \frac{1}{n}\inf_{B\in \mathcal{B}_\epsilon} (\operatorname{Tr}(AB) + t\operatorname{Tr}(B)).
\end{align*}
By the AM-GM inequality, since $\det B = 1$ we have $$\operatorname{tr}(B) \geq n,$$ and so $$\widehat F_\epsilon(A+tI) \geq \widehat F_\epsilon(A) + t.$$

For the second part, we claim that  if $A_\epsilon\rightarrow A$ as $\epsilon\rightarrow 0^+$, and $\lambda_{min}(A) <0 $, then 
\begin{equation}\label{eq:bellman-neg-ev}
\widehat F_\epsilon(A_\epsilon) \rightarrow -\infty.
\end{equation} The required result then follows from combining this with the estimate \eqref{eq:bellman-smoothening-uniform-est}  above. To prove \eqref{eq:bellman-neg-ev} suppose there is a unit eigenvector $e_1$ such that $$\langle Ae_1,e_1\rangle = -2a<0.$$ Then for $\epsilon<<1$, $\langle A_\epsilon e_1,e_1\rangle < -a.$ Also $|A_\epsilon| \leq M$ for some $M>0$. Assume $n \ge 2$ and extend $e_1$ to an orthonormal basis $\{e_1,\cdots,e_n\}$ and let $$B_\epsilon = \mathrm{diag}(\epsilon^{-1},\epsilon^{1/(n-1)},\cdots, \epsilon^{1/(n-1)}).$$ Then for small $\epsilon$, $B_\epsilon \in \mathcal{B}_\epsilon$ and so we have 
\begin{align*}
\widehat F_\epsilon(A_\epsilon) &\leq \frac{1}{n}\operatorname{Tr}(A_\epsilon B_\epsilon)\\
&\leq \frac{1}{n}\Big(-\frac{a}{\epsilon} + \epsilon^{1/(n-1)}\sum_{i=2}^nA_{i\bar i}\Big)\\
&\leq -\frac{a}{n\epsilon} + \sqrt{n}M\epsilon^{1/(n-1)}.
\end{align*}
Taking $\epsilon\rightarrow 0$ we prove \eqref{eq:bellman-neg-ev}.

<!-- lemma-3: body_md -->
\label{lem:cutoffs}
Let $(X,\omega)$ be a complete noncompact K\"ahler manifold with $BK>0$. 
\begin{enumerate}
\item There exists a smooth exhaustion function $\rho$ and constant $C_\rho>0$ such that  $$\vert\nabla\rho\vert+|H(\rho)|\le 1.$$
\item There exists a $R_0>0$ such that for any $R>R_0$ there is a function $\chi_R\in C^\infty$ satisfying \begin{equation*}\label{eq:cutoffs}
 \chi_R=1\text{ on }B(o,R),\ \ \ 
 0 \le  \chi_R \le 1, \ \ \ 
 \operatorname{supp}\chi_R\subset B(o,2R),\ \  \
 |\nabla\chi_R|+|dd^c\chi_R|\le C/R.
\end{equation*}
In particular, $\chi_R\to1$ locally and for every fixed integer $b\ge2$, the exists $C_b >0$ such that 
\begin{equation*}\label{eq:cutoff-powers}
 -\frac{C_b}{R}\chi_R^{b-2}\omega
 \ledd^c(\chi_R^b)
 \le\frac{C_b}{R}\chi_R^{b-2}\omega.
\end{equation*}
Here $B$ is the $\rho$-ball.
\item There is a smooth function $u_0: X  \rightarrow {\mathbb R}$ and a constant $C_0$ (depending on $u_0$)   such that
$$\vert \nabla u_0\vert\leq 1 \text{ and }0 < H(u_0) < C_0I.$$
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
(1) is just Proposition 3.1 of \cite{NT3}: time 1 heat-flow of a distance function gives a smooth exhaustion function $\rho$ with $$|\nabla \rho|, | H(\rho)| \leq C .$$ For the second part, one simply takes $\chi = \eta(\rho/R)$ where $\eta$ is a standard one-variable cut-off function. For the third part, instead of the distance function, we take the heat flow at unit time of a Buseman function given by $$b(x) = \lim_{t\rightarrow \infty}(t - d(x,\gamma(t)),$$ where $\gamma$ is a ray. It is well known \cite{Wu, GW, NT3} that if $BK>0$, $H(P_1b) >0$. Then $u_0 = P_1b$, where $P_1b$ is the time 1 heat-flow of $b$, satisfies the required estimates once again by the aforementioned work of Ni and Tam \cite{NT, NT2}.

<!-- lemma-4: body_md -->
\label{lem:betterbarrier}
For each $a>0$ there exists a smooth function $\zeta_a$ solving $$\Delta\zeta_a = a(\zeta_a - u_0).$$ Moreover, there exists a dimensional constant $C_n$ such that $$|\zeta_a - u_0| \leq \frac{C_n}{\sqrt{a}}.$$
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Let $P_t$ be the heat semi-group as above. We then let $$\zeta_a(x) = a\int_0^\infty e^{-at}P_tu_0(x)\,dt.$$ It is easy to check that this function solves the equation. Next, we note that since $\operatorname{Ric} \geq 0$ and $|\nabla u_0|\leq 1$, we have the standard estimate (cf. \cite[Lemma 6]{DPS1}) $$\Vert P_tu_0 - u_0\Vert_{L^\infty(X)} \leq C\sqrt{t}. $$ We then estimate 
\begin{align*}
|\zeta_a(x) - u_0(x)|  &\leq a\int_0^\infty e^{-at}|P_tu_0(x) - u_0(x)|\,dt\\
&\leq Ca\int_0^\infty e^{-at}\sqrt{t}\,dt\\
&= \frac{C'}{\sqrt{a}},
\end{align*}
for some $C'>0$.
<!-- end proof-4 -->
<!-- lemma-5: body_md -->
Let $(X,\omega)$ be a complete K\"ahler manifold with $BK>0$. Then there exists an exhaustion function $f\in C^\infty(X,\mathbb{R})$ satisfying the following properties:
\begin{enumerate}
\item $\sup_{X}e^{-f/n} < \infty.$
\item $e^{-f/n}\vert \nabla f\vert  \leq \min(nB/4,1).$
\item $\int_X e^{-f}\omega^n < \infty.$
\item $0<h(x)<\frac{1}{2}m(x)$ where $m(x)=(\det(H(u_0))(x))^{1/n}$
\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
We reproduce the proof from \cite{DPS1}. Without loss of generality we may assume that $\inf_X \rho = 0$. Fix a point $o\in \rho^{-1}(0)$. We let $B_r = \{\rho < r\}$ and $S_r = \{\rho = r\}$. Let $g:[0,\infty)\rightarrow (0,\infty)$ be a smooth strictly decreasing function such that $$g(r) \leq \inf_{x\in S_r}e^{-d(o,x)},$$ and $$g(r)<\frac{1}{2} \inf_{\rho\leq r} m.$$ Note that since $\rho$ is an exhaustion, the infimum on the right is positive for each $r>0$. Note also that $\lim_{r\rightarrow \infty}g(r) = 0.$ We now set $$\xi(\rho) = -n\ln\Big(\int_{\rho}^\infty G(r)\,dr\Big), $$ where we choose $G$ to be a smooth function satisfying $$0< G(r) < \min\Big(-g'(r), \frac{1}{4}\inf_{x\in S_r}B(x), \frac{1}{n}\Big).$$ 
Then $f(x) = \xi(\rho(x))$ is the required function.

<!-- lemma-6: body_md -->
\label{lem:comparison}
There exists a constant $K = K(\Vert h\Vert_{L^\infty(X)},\epsilon)$ (but independent of $\delta$) such that $$\sup_X\vert u_{\epsilon,\delta} - u_0\vert < \frac{K}{\delta}.$$
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
It suffices to prove that $$\sup_X\vert u_{\epsilon,\delta,j} - u_0\vert < \frac{K}{\delta}$$ on $\overline{\Omega_j}$ as long as $K$ is independent of $j$. Let $p$ be a minimum of $v = u_{\epsilon,\delta,j} - u_0$. If $p$ is on the boundary, then $v(p) = 0$. Else $p$ is in the interior, and so by the maximum principle $H(u_{\epsilon,\delta,j}) \geq H(u_0)$  at $p$. Hence by ellipticity of $F$, $$h + \delta v(p) = F_\epsilon(H(u_{\epsilon,\delta,j}))(p) \geq  F_\epsilon(H(u_0))(p) \geq -C,$$ since $H(u_0)$ lies in a compact set and $F_\epsilon$ is smooth. In any case, we then have that $$\delta \inf v \geq -C - \Vert h\Vert_{L^\infty(X)}.$$ The upper bound is established in the same way.

<!-- prop-2: body_md -->
$\sup_{X}\vert\nabla u_{\epsilon,\delta}\vert \leq 1.$
<!-- end prop-2 -->
<!-- proof-7: body_md -->
Set $w = \vert\nabla u_{\epsilon,\delta}\vert^2$, $q = \delta(u-u_0) + h$, and let $L$ denote the linearised operator $$L = F_\epsilon^{i\bar j}\nabla_i\nabla_{\bar j}\text{ where }F_\epsilon^{i\bar j} = \frac{\partial F_\epsilon}{\partial_{\lambda_{i\bar j}}}(H(u)).$$ We first claim that 
\begin{equation}\label{eq:claim-bounded-grad}
\sup_{X}w < \infty.
\end{equation}Computing at a point in coordinates which are normal for $\omega$ and in which the Hessian of $u$ is diagonal (with say eigenvalues $\mu_1,\cdots,\mu_n$), we have 
\begin{align*}
Lw &= F_\epsilon^{i\bar j}(g^{a\bar b}u_au_{\bar b})_{i\bar j}\\
&= F_\epsilon^{i\bar j}\Big(u_{ai\bar j}u_{\bar a} + u_{a}u_{i\bar j \bar a} + u_{ia}u_{\bar j\bar a}+u_{a\bar j}u_{i\bar a} + R_{i\bar j a\bar b}u_a u_{\bar b}\Big)\\
&\geq F_\epsilon^{i\bar j}\Big(u_{ai\bar j}u_{\bar a} + u_{a}u_{i\bar j \bar a} + u_{ia}u_{\bar j\bar a}+u_{a\bar j}u_{i\bar a}\Big) + B |\nabla u|^2,
\end{align*}
where as usual we sum over repeated indices and we used the observation that $\sum_i F^{i\bar i} \geq 1.$ Crucially, the positive holomorphic bisectional curvature was used to get the last inequality.  Next, differentiating the equation we have that $F_\epsilon^{i\bar j}u_{i\bar j a} = q_a,$ and so setting $\lambda_\epsilon = \epsilon/n$ we have 
\begin{align*}
Lw &\geq q_au_{\bar a}+ q_{\bar a}u_a + F_\epsilon^{i\bar j}\Big(u_{ia}u_{\bar j\bar a}+u_{a\bar j}u_{i\bar a}\Big) + B|\nabla u|^2\\
&\geq q_au_{\bar a}+ q_{\bar a}u_a + \lambda_\epsilon\sum_{i,a}|u_{ia}|^2 +\lambda_\epsilon\sum_{k}\mu_k^2 + Bw\\
&=  \lambda_\epsilon\sum_{i,a}|u_{ia}|^2 +\lambda_\epsilon\sum_{k}\mu_k^2 + (B + 2\delta)w + 2\Re(h_a u_{\bar a}) - 2\delta\Re((u_0)_au_{\bar a})
\end{align*} Naively, if $w$ were to take a maximum value at some point, the above inequality would immediately imply an upper bound on $w$. Since in principle the maximum of $w$ could drift off to infinity, we instead consider $U = \eta_Rw$, where $\eta_R = \chi_R^2$ and $\chi_R$ is the cut-off function from Lemma \ref{lem:cutoffs}. Note that $$\frac{|\nabla \eta|^2}{\eta} \leq \frac{C}{R^2} \text{ and } |dd^c\eta| \leq \frac{C}{R}.$$Then (suppressing the subscript $R$)
\begin{align*}
LU &\geq \eta\lambda_\epsilon\sum_{i,a}|u_{ia}|^2 +\eta\lambda_\epsilon\sum_{k}\mu_k^2 + (B + 2\delta)U + 2\eta\Re(h_a u_{\bar a}) - 2\eta\delta\Re((u_0)_au_{\bar a})\\
&+ 2F^{i\bar j}\Re(\eta_{i}w_{\bar j}) + F_\epsilon^{i\bar j}w\eta_{i\bar j}\\
&\geq \eta\lambda_\epsilon\sum_{i,a}|u_{ia}|^2 +\eta\lambda_\epsilon\sum_{k}\mu_k^2 + (B + 2\delta)U  - 2C \eta\sqrt{w} +2F_\epsilon^{i\bar j}\Re(\eta_{i}w_{\bar j})  - \frac{C}{R\lambda_\epsilon}w.
\end{align*}
Denoting by $Q = \sum_{i,a}|u_{ia}|^2 + \sum_{k}\mu_k^2,$ we control the second-last term by 
\begin{align*}
|2F_\epsilon^{i\bar j}\Re(\eta_{i}w_{\bar j})| \leq C\lambda_\epsilon^{-1}|\nabla \eta| \sqrt{w}\sqrt{Q} \leq \frac{\eta\lambda_\epsilon}{2}Q + \frac{C|\nabla\eta|^2}{2\lambda_\epsilon^3\eta}w \leq \frac{\eta\lambda_\epsilon}{2}Q + \frac{C}{2\lambda_\epsilon^3 R^2}w,
\end{align*} and so $$LU \geq  - 2C \eta\sqrt{w}  +(B + 2\delta)U - \frac{C}{R}w,$$ where in the last term in the last line, the constant is of course allowed to depend on $\epsilon$ but not $R$. At this stage, if the final negative term had a $\eta w$ instead of just a $w$ we could have absorbed it in the second term by choosing $R$ large and an application of the maximum principle would have shown that $w$ is bounded. Instead, we use $B>0$ to simply drop the second term and write $$LU \geq - 2C \eta\sqrt{w} - \frac{C}{R}w.$$ Now we need some more positivity to absorb the terms with $-w$. So consider the function $$V = U + \psi(v),$$ where we set $v = u - u_0$ and $\psi$ will chosen shortly and satisfies $$\psi ' < 0 \text{ and }\psi'' \geq c>0$$ on the range of $v$. We now compute:  
\begin{align*}
LV &= LU + L\psi(v)\\
 &= LU + \psi'(v) Lv + \psi''(v)F_\epsilon^{i\bar j}v_iv_{\bar j}\\
 &\geq LU +  \psi'(v) Lv + c\epsilon |\nabla v|^2.
\end{align*} 
The key point now is to use concavity of $F_\epsilon$. Indeed, we have $$F_\epsilon(H(u_0)) \leq F_\epsilon(H(u)) + F_\epsilon^{i\bar j} ((u_0)_{i\bar j} - u_{i\bar j}),$$ and so $$Lv \leq F_\epsilon(H(u)) - F_\epsilon(H(u_0)) = h + \delta v - F_\epsilon(H(u_0)):= G.$$ Since $\psi'<0$ we then have that $$\psi'(v)Lv \geq \psi'(v)G.$$For the third term in the lower bound on $LV$, by AM-GM inequality we have $$|\nabla v|^2 \geq \frac{1}{2}w -C $$ using the fact that $|\nabla u_0|$ is uniformly bounded. Combined with the earlier estimate on $LU$ we have $$LV \geq - 2C \eta\sqrt{w} - \frac{C}{R}w +\psi'(v)G +  \frac{c\epsilon}{2}w - C.$$ Now we choose $R>>1$ such that $$LV \geq - 2C \eta\sqrt{w}  + \frac{c\epsilon}{4}w + \psi'(v)G - C.$$ At this point, we now pick $$\psi(x) = (M-x)^2,$$ where $M = \sup v + 1$. Then $\psi$ satisfies both the required properties (with $c  =2$). Note that $\psi'(x) = -2(M-x)$ and so $\psi'(v)<0$ and $\psi''(v) \geq 2$. Also $G$ has a uniform upper bound, and so in the end we have $$LV \geq - 2C \eta\sqrt{w}  + \frac{c\epsilon}{4}w - C.$$

We now consider the maximum of $V$ on $B(0,2R)$. If the maximum occurs on the boundary, $\eta_R$ is zero there and so, $$V =  \psi(v) \leq \sup_{X}\psi(v) \leq C_{\epsilon,\delta}.$$ Crucially the constant does not depend on $R$. On the other hand, if the maximum is attained on the interior of $B(o,2R)$, say at a point $p$, then by the maximum principle $$0 \geq - 2C \eta\sqrt{w}  + \frac{c\epsilon}{4}w - C,$$ and so $w(p) \leq C$ for some constant $C>0$ independent of $R$. But then in particular this implies that $$\sup_{B(o,R)}w \leq C$$ for some constant independent of $R$. This proves the claim \eqref{eq:claim-bounded-grad}. 

We now prove that in fact $w\leq 1$. In fact we claim that there exists a constant $C_\epsilon>0$ such that for any $\alpha>0$ we have 
\begin{equation}\label{eq:w-al}
w(x) \leq 1 + \alpha\Big(\rho(x) + \frac{C_\epsilon}{\delta}\Big).
\end{equation} We then let $\alpha\rightarrow 0$. To prove the claim, we consider the function $W = w - \alpha\rho$, where note that we have normalized so that $\rho\geq 0$. Since $\rho$ is an exhaustion, this certainly takes the maximum at some point, say at $p$. Then at $p$, $$Lw(p) \leq \alpha L\rho(p) \leq \alpha C_\epsilon$$If $w(p) \geq 1$, by our calculations earlier, and using the fact that $|\nabla h| \leq B?2$ and $|\nabla u_0| \leq 1$ we have 
\begin{align*}
Lw&\geq (B+2\delta)(w - \sqrt{w})(p)\\
&\geq \delta(w(p)-1),
\end{align*}
and so $$w(p) \leq 1 + \alpha\frac{C_\epsilon}{\delta},$$ and so this proves \eqref{eq:w-al}. If $w(p) \leq 1$ the same upper bound is automatic.

<!-- lemma-7: body_md -->
\label{lem:betterestimate}
There exists a constant $C$ that is independent of $\delta,\epsilon$ such that $$\delta \Vert u_{\epsilon,\delta} - u_0 \Vert_{L^{\infty}}\leq C\sqrt{\delta}+C\epsilon^2.$$
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
It suffices to prove that 
$$\delta \Vert u_{\epsilon,\delta,j} - u_0 \Vert_{L^{\infty}}\leq C\sqrt{\delta}+C\epsilon^2$$
on $\bar{\Omega}_j$ for $C$ independent of $j,\delta,\epsilon$. Note that by choosing $B= I$, $$ F_{\epsilon}(A)\leq \hat{F}_{\epsilon}(A)\leq \frac{1}{n}\operatorname{Tr}(A). $$
Thus the better barrier $\zeta_{n\delta}$ produced by Lemma \ref{lem:betterbarrier} satisfies
$$ F_{\epsilon}(H(\zeta_{n\delta}))-\delta(\zeta_{n\delta}-u_0)\leq 0<F_{\epsilon}(H(u_{\epsilon,\delta,j}))-\delta(u_{\epsilon,\delta,j}-u_0).$$
If the maximum of $v=u_{\epsilon,\delta,j}-\zeta_{n\delta}$ occurs at an interior point, then $v\leq 0$ throughout. If it is attained on the boundary of $\Omega_j$, then $$v\leq \Vert \zeta_{n\delta}- u_0\Vert_{L^{\infty}}\leq \frac{C}{\sqrt{\delta}} .$$ Thus $$u_{\epsilon,\delta}-\zeta_{n\delta}\leq\frac{C}{\sqrt{\delta}}.$$
Now we note that $$F_{\epsilon}(H(u_0))\geq m-C\epsilon^2\geq h-C\epsilon^2.$$
Thus, setting $\tilde{u}_0=u_0-\frac{C\epsilon^2}{\delta}$, and comparing with $u_{\epsilon,\delta,j}$ as above, we get the desired lower bound.

<!-- lemma-8: body_md -->
$\phi$ is psh and solves $$(dd^c\phi)^n = e^{-f}\omega^n$$ in the Bedford-Taylor sense.
<!-- end lemma-8 -->
<!-- proof-9: body_md -->
In the proof, we suppress the subscript $k$ on $\epsilon_k$. So for instance when we say let $\epsilon\rightarrow 0$ we actually mean that $\epsilon_k\rightarrow 0$. Since the claim is local, we fix a coordinate unit ball $B$. Then $\varphi_k\rightarrow \phi$ uniformly on $\overline{B}$. Choose smooth boundary data $\phi_m$ on $\partial B$ which converge uniformly to $\phi$ on $\partial B$. Let $v_m$ be a soluion to the Dirichlet problem $$\begin{cases}(dd^cv_m)^n = e^{-f}\omega^n\quad\text{ on $B$}\\ v_m = \phi_m\quad\text{ on $\partial B$}.  \end{cases}$$ We claim that for each fixed $m$,
\begin{equation}\label{eq:final-est}
\Vert \phi - v_m\Vert_{L^\infty(B)} \leq \Vert \phi - \phi_m\Vert_{L^\infty(\partial B)}.
\end{equation} Assuming this for now we immediately get that $v_m\rightarrow \phi$ uniformly on $B$. In particular $\phi$ is psh and also solves $$(dd^c\phi)^n = e^{-f}\omega^n.$$ In order to prove the claim, we first note that since $v_m$ is strictly psh and smooth on $\overline{B}$, $$K_m:= \{H(v_m)(x)~|~x\in \overline{B}\}$$ is compact in $\operatorname{Herm}^+(n,\mathbb{C}).$ For a fixed $m$, there exists $\epsilon_m>0$ such that for all $\epsilon<\epsilon_m$ we have $$\sup_{A\in K_m}|F_\epsilon(A) - F(A)| \leq C\epsilon^2,$$ where recall that $C$ is independent of $K_m$ and hence $m$. Now, $F(H(v_m)) = h$ and so $$|F_\epsilon(H(v_m)) - h| \leq r_\epsilon,\quad r_{\epsilon}:= C\epsilon^2.$$ Recall that $\varphi_\epsilon$ solves $F(H(\varphi)) = h + c_\epsilon.$ Consider the ``test function" $q(z) = |z|^2 - 1$. This satisfies $$-1\leq q\leq 0 \text{ on $ B$},~q\equiv 0 \text{ on $\partial B$ and }H(q) \geq  \mu_BI,$$ where $\mu_B>0$ is some fixed constant depending only on $\omega\Big|_{\overline{B}}$. We set $$a_\epsilon:= \frac{r_\epsilon + |c_\epsilon|}{\mu_B} \text{ and }b_{m,\epsilon}:= \Vert \varphi_\epsilon - \phi_m\Vert_{L^\infty(\partial B)},$$ and consider the lower and upper barriers defined by $$\underline{v}_{\epsilon,m} := v_m + a_\epsilon q - b_{m,\epsilon}\quad\underline{v}_{\epsilon,m} := v_m - a_\epsilon q + b_{m,\epsilon}, \text{ where }.$$ Since $H(q) \geq 0$. it follows from \eqref{eq:finite-diff-F-ep} that $$F_\epsilon(H(\underline{v}_{m,\epsilon})) \geq  F_\epsilon(\varphi_\epsilon)\geq F_\epsilon(H(\overline{v}_{m,\epsilon})).$$ Indeed, by \eqref{eq:finite-diff-F-ep} 
\begin{align*}
F_\epsilon(H(\underline{v}_{m,\epsilon})) &\geq F_\epsilon(H(v_m)) + a_\epsilon\mu_B \\
&\geq h - r_\epsilon + a_\epsilon\mu_B\\
&\geq  h + c_\epsilon = F_\epsilon(\varphi_\epsilon).
\end{align*}
The other estimate is proven similarly. On the boundary $\partial B$ we have $$\underline{v}_{m,\epsilon} \leq \phi_\epsilon \leq \overline{v}_{m,\epsilon}.$$  Then by the standard comparison principle (see for instance the argument in the proof of Lemma \ref{lem:comparison}) we have that $$\underline{v}_{m,\epsilon} \leq \phi_\epsilon \leq \overline{v}_{m,\epsilon}$$  on all of $B$. Since $|q| \leq 1$ this implies that $$\Vert\varphi_\epsilon - v_m\Vert_{L^\infty(B)} \leq a_\epsilon + b_{m,\epsilon}.$$ Fixing $m$ and letting $\epsilon \rightarrow 0$ proves the claim \eqref{eq:final-est}, since $a_\epsilon\rightarrow 0$.
