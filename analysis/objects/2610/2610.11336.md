\label{l3}\cite{kittaneh1993norm}
Let \(A,B,X \in B(H)\) be such that \(A\) and \(B\) are positive and
\(X \in B_p(H)\), where \(1 \le p \le \infty\), and let
\(0 \le t \le 1\). Then
\[
\|A^t X B^{1-t}\|_p
\le
\|AX\|_p^t \|XB\|_p^{1-t}.
\]

\cite{kittaneh1997operator}\label{lt1}
Let $T,S\in B_p(H)$ be positive, and let $p\ge1$. Then
\[
\|T+S\|_p
\le
\left(\|T\|_p^p+\|S\|_p^p\right)^{\frac1p}
+2^{\frac1p}\bigl\|T^{\frac12}S^{\frac12}\bigr\|_p .
\]

<!-- proof-1: body_md -->
Let $T=U|T|$ be the polar decomposition of $T$. 
        
                        
                                                                                                 
    
         
Consider,
\begin{align}
\|\Re_t(e^{i\theta}T)\|_p
\nonumber &=\left\|
te^{i\theta}T + (1-t)e^{-i\theta}T^*
\right\|_p\\
\nonumber &=\left\|
te^{i\theta}T +te^{-i\theta}T^*+ (1-2t)e^{-i\theta}T^*
\right\|_p\\
\nonumber &=\left\|
te^{i\theta}T +te^{-i\theta}T^*\|_p+ \|(1-2t)e^{-i\theta}T^*
\right\|_p\\
\nonumber &=
t\left\|e^{i\theta}U|T|+e^{-i\theta}|T|U^*\right\|_p + |1-2t|\|e^{-i\theta}|T|U^*\|\\
    
                                                               
\nonumber &=
t\left\|
\begin{bmatrix}
e^{i\theta}Ug(|T|) & f(|T|)\\
0 & 0
\end{bmatrix}
\begin{bmatrix}
f(|T|) & 0\\
e^{-i\theta}g(|T|)U^* & 0
\end{bmatrix}
\right\|_p+|1-2t|\||T|\|_p
\end{align}
Here it is easy to see that the product $\begin{bmatrix}
e^{i\theta}Ug(|T|) & f(|T|)\\
0 & 0
\end{bmatrix}
\begin{bmatrix}
f(|T|) & 0\\
e^{-i\theta}g(|T|)U^* & 0
\end{bmatrix}$ is self adjoint. Since $T \in B_p(H)$, applying the Lemma \ref{lem1}, we have
\begin{align}\label{k2}
\nonumber \|\Re_t(e^{i\theta}T)\|_p&\le t
\left\|
\Re\left(
\begin{bmatrix}
f(|T|) & 0\\
e^{-i\theta}g(|T|)U^* & 0
\end{bmatrix}
\begin{bmatrix}
e^{i\theta}Ug(|T|) & f(|T|)\\
0 & 0
\end{bmatrix}
\right)
\right\|_p+|1-2t|\|T\|_p\\
                                                       
\nonumber &=
t\left\|\Re\left(
\begin{bmatrix}
e^{i\theta}f(|T|)Ug(|T|)
&
f^2(|T|)
\\[2mm]
g^2(|T|)
&
e^{-i\theta}g(|T|)U^*f(|T|)
\end{bmatrix}\right)
\right\|_p+|1-2t|\|T\|_p\\
\nonumber &=t
\left\|
\begin{bmatrix}
\Re(e^{i\theta}\widetilde{T}_{f,g})
&
\dfrac12\bigl(f^2(|T|)+g^2(|T|)\bigr)
\\[2mm]
\dfrac12\bigl(f^2(|T|)+g^2(|T|)\bigr)
&
\Re\left(e^{-i\theta}(\widetilde{T}_{f,g})^*\right)
\end{bmatrix}
\right\|_p+|1-2t|\|T\|_p\\
\nonumber &\le t
\left\|
\begin{bmatrix}
\Re(e^{i\theta}\widetilde{T}_{f,g}) & 0\\
0 & \Re\left(e^{-i\theta}(\widetilde{T}_{f,g})^*\right)
\end{bmatrix}
\right\|_p\\
\nonumber &\hspace{1cm}+\frac{1}{2}t
\left\|
\begin{bmatrix}
0 &
\bigl(f^2(|T|)+g^2(|T|)\bigr)
\\[2mm]
\bigl(f^2(|T|)+g^2(|T|)\bigr)
& 0
\end{bmatrix}
\right\|_p+|1-2t|\|T\|_p\\
\nonumber &=
2^{\frac1p}t
\left\|
\Re(e^{i\theta}\widetilde{T}_{f,g})
\right\|_p
+
2^{\frac1p-1}t
\left\|
f^2(|T|)+g^2(|T|)
\right\|_p+|1-2t|\|T\|_p\\
\nonumber &\hspace{5cm}(\text{Using} ~\eqref{qe1})\\
 &=
2^{\frac1p}t
\left[\left\|
\Re(e^{i\theta}\widetilde{T}_{f,g})
\right\|_p
+
\frac{1}{2}
\left\|
f^2(|T|)+g^2(|T|)
\right\|_p\right]+|1-2t|\|T\|_p.
\end{align}
By taking the supremum on both sides over $\theta \in \mathbb{R}$, we get 
\begin{align*}
w_{t,p}(T)\leq2^{\frac1p}t\left[
w_p(\widetilde{T}_{f,g})
+
\frac{1}{2}
\left\|
f^2(|T|)+g^2(|T|)
\right\|_p\right]+|1-2t|\|T\|_p.
\end{align*}
