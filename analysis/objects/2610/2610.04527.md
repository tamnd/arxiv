<!-- conj-1: body_md -->
\label{conj:god}
For any convex body $K\subset \mathbb{R}^n$ and any $1\le k\le n-1$,  
\begin{equation*}
V_n(K[k], -K[n-k])\le \binom{n}{k} \operatorname{vol}_n(K),\end{equation*}
with equality if and only if $K$ is an $n$-simplex.

<!-- thm-1: body_md -->
\label{thm:km_ineq}
For any convex body $K\subset \mathbb{R}^n$ and any $1\le k\le n-1$,  
\begin{equation}\label{eq:Godbersen-ineq}
V_n(K[k], -K[n-k])\le \binom{n}{k} \operatorname{vol}_n(K).\end{equation}
If $K$ is an $n$-polytope and equality holds in \eqref{eq:Godbersen-ineq}, then $K$ is an $n$-simplex.
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\label{thm:godbersen_equality} If $K \subset \mathbb{R}^n$ is a convex body satisfying $V_n(K[k], -K[n - k]) = \binom{n}{k} \operatorname{vol}_n(K)$ for some $k\in \{1,\dots,n-1\}$, then $K$ is an $n$-simplex.

\label{conj:higher_godbersen}
For any convex body $K\subset \mathbb{R}^n$, any integer $p \ge 1$, and any $a_0, \dots, a_p \in \{1, \dots, n\}$ with $a_0 + \cdots + a_p = pn$, one has
\begin{equation}\label{eq:higher_godbersen}
V_{pn}(-\Delta_p K[a_0], \imath_1 K[a_1], \dots, \imath_p K[a_p]) \le \binom{n}{n - a_0, \dots, n - a_p} \frac{(n!)^p}{(pn)!} \operatorname{vol}_n(K)^p,\end{equation}
with equality if and only if $K$ is an $n$-simplex.

<!-- lem-1: body_md -->
\label{lem:exact}
Let $1\leq d\leq N-1$, let $E\subset\mathbb{R}^N$ be a $d$-dimensional linear subspace, let $P_{E^\perp}$ denote the orthogonal projection onto $E^{\perp}$, and let $L_1,\dots,L_d,M_1,\dots,M_{N-d}\in \mathcal K^N$ with $L_i\subset E$ for each $i=1,\dots,d$. Then
\begin{equation}
\label{eq:projectionidentity}
\binom{N}{d} V_N(L_1,\dots,L_d,M_1,\dots,M_{N-d}) = V_E(L_1,\dots,L_d) V_{E^\perp}(P_{E^\perp} M_1,\dots,P_{E^\perp} M_{N-d}),
\end{equation}
where $V_E$ and $V_{E^\perp}$ denote the mixed volumes in $E$, $E^\perp$, respectively.

<!-- thm-5: body_md -->
\label{thm:higher_godbersen} For any $K \in \mathcal K^n$, one has \begin{equation}\label{eq:higher_godbersen_redux}
V_{pn}(-\Delta_p K[a_0], \imath_1 K[a_1], \dots, \imath_p K[a_p]) \le \binom{n}{n - a_0, \dots, n - a_p}  \frac{(n!)^p}{(pn)!} \operatorname{vol}_n(K)^p.
\end{equation}

<!-- proof-1: body_md -->
First of all, if $\operatorname{vol}_n(K) = 0$, the affine span of $K$ is not all of $\mathbb{R}^n$. In this case, after translation, all the bodies $-\Delta_p K, \imath_1 K, \dots, \imath_p K$ will lie in a proper affine subspace of $\mathbb{R}^{pn}$, so the left-hand side of \eqref{eq:higher_godbersen_redux} is $0$ as well. Thus we may assume that $\operatorname{vol}_n(K) > 0$.

As in \cite{KM2026}, we lift to a space with one more coordinate block. Let \[\tilde\imath_0, \dots, \tilde\imath_{p}: \mathbb{R}^n \to \mathbb{R}^{(p + 1) n}\]
be the $p + 1$ block coordinate embeddings of $\mathbb{R}^n$ into $\mathbb{R}^{(p + 1) n}$. (Note that in $\mathbb R^{(p + 1) n}$, we choose to index the coordinate blocks starting from $0$.) Also let $\Delta_{p + 1}: \mathbb{R}^n \to \mathbb{R}^{(p + 1) n}$ be the diagonal embedding. We consider the mixed volumes
\begin{equation}\label{eq:lifted_mv}
    \mathcal V_{a_0, \dots, a_p} := V_{(p + 1)n}(\Delta_{p + 1} K[n], \tilde\imath_0 K[a_0], \dots, \tilde\imath_p K[a_p]).
\end{equation}

Let $A \in \mathbb{R}^{(p + 1)n \times (p + 1) n}$ be the block matrix
\[A = \begin{pmatrix}
-I_n   & \\ 
-I_n   & I_n & \\
\vdots &     & \ddots & \\
-I_n   &     &        &  I_n
\end{pmatrix},\]
which has determinant $(-1)^n$ and satisfies
\begin{align*}
    A\Delta_{p + 1} &= -\tilde\imath_0 \\
    A\tilde\imath_0 &= -\Delta_{p + 1} \\
    A\tilde\imath_k &= \tilde\imath_k, \qquad k = 1, \dots, p.
\end{align*}

Then by linear invariance of mixed volumes (viii), we have
\begin{align*}
\mathcal V_{a_0, \dots, a_p} &= V_{(p + 1)n}(A\Delta_{p + 1} K[n], A\tilde\imath_0 K[a_0], A\tilde\imath_1 K[a_1], \dots, A\tilde\imath_p K[a_p]) \\
&= V_{(p + 1)n}(-\tilde\imath_0 K[n], -\Delta_{p + 1} K[a_0], \tilde\imath_1 K[a_1], \dots, \tilde\imath_p K[a_p]).
\end{align*}

Let $E = \operatorname{Im} \tilde\imath_0$ be the span of the first coordinate block in $\mathbb{R}^{(p + 1)n}$; then $-\tilde\imath_0 K \subset E$ and $\dim(E)=n$, so by Lemma \ref{lem:exact}, we have
\begin{equation}\label{eq:proa_pplus1}
\mathcal V_{a_0, \dots, a_p} = \binom{(p + 1) n}{n}^{-1} \operatorname{vol}_n(K) \cdot V_{E^\perp}(P_{E^\perp}(-\Delta_{p + 1} K)[a_0], P_{E^\perp} \tilde\imath_1 K[a_1], \dots, P_{E^\perp} \tilde\imath_p K[a_p]).
\end{equation}

Note that $E^{\perp}=\{0\} \times \mathbb{R}^{pn}$. Therefore $E^\perp$ can be identified with $\mathbb{R}^{pn}$ by simply forgetting the first $n$ coordinates. Under this identification, it is easy to see that 
\begin{align*}
    P_{E^\perp} \circ \Delta_{p + 1} &= \Delta_p \\
    P_{E^\perp} \circ \tilde\imath_k &= \imath_k, \quad k = 1, \dots, p.
\end{align*}
Hence, we obtain from \eqref{eq:proa_pplus1} that
\begin{equation}\label{eq:diag_mv}
\mathcal V_{a_0, \dots, a_p} = \binom{(p + 1) n}{n}^{-1} \operatorname{vol}_n(K) \cdot V_{pn}(-\Delta_p K[a_0], \imath_1 K[a_1], \dots, \imath_p K[a_p]),
\end{equation}
which, up to a binomial coefficient and the factor of $\operatorname{vol}_n(K)$, is precisely the left-hand-side of \eqref{eq:higher_godbersen_redux}.

Thus, it remains to bound $\mathcal V_{a_0, \dots, a_p}$ from above. Let 
\[K^{p + 1} = \overbrace{K \times K \times \cdots \times K}^{\text{$p + 1$ times}} = \tilde\imath_0 K + \cdots + \tilde \imath_p K.\]
Then clearly we have $\Delta_{p + 1} K \subset K^{p + 1}$, and so, returning to the original definition \eqref{eq:lifted_mv} of $\mathcal V_{a_0, \dots, a_p}$ and using monotonicity of mixed volumes, we have
\begin{align*}
    \mathcal V_{a_0, \dots, a_p} &= V_{(p + 1)n}(\Delta_{p + 1} K[n], \tilde\imath_0 K[a_0], \dots, \tilde\imath_p K[a_p]) \\
    &\le V_{(p + 1)n}(K^{p + 1}[n], \tilde\imath_0 K[a_0], \dots, \tilde\imath_p K[a_p]).
\end{align*}

We expand the right-hand side by multilinearity:
\begin{equation}\label{eq:mv_prod_expansion}
\mathcal V_{a_0, \dots, a_p} \le \underset{\substack{s_0, s_1, \dots, s_p \ge 0 \\ s_0 + \cdots + s_p = n}}\sum \binom{n}{s_0, \dots, s_p} V_{(p + 1)n}( \tilde\imath_0 K[a_0 + s_0], \tilde\imath_1 K[a_1 + s_1], \dots, \tilde\imath_p K[a_p + s_p]).
\end{equation}
For each $i = 0, \dots, p$, $\tilde\imath_j K$ is a convex set whose affine span has dimension $n$, so the only terms in the sum on the right-hand side of \eqref{eq:mv_prod_expansion} which can be non-zero are those for which $a_i + s_i \le n$ for all $i$, and hence satisfy 
$$\sum_{i = 0}^p s_i \leq \sum_{i=0}^p (n-a_i)=(p+1)n-pn=n.$$ 
Since $s_0 + \cdots + s_p =n$, we see that the only non-vanishing term corresponds to the case $s_i=n-a_i$ for all $i$. Thus we obtain

\begin{equation}\label{eq:mv_prod_reduction}
\mathcal V_{a_0, \dots, a_p} \le \binom{n}{n - a_0, \dots, n - a_p} V_{(p + 1)n}( \tilde\imath_0 K[n], \tilde\imath_1 K[n], \dots, \tilde\imath_p K[n]).
\end{equation}

On the right-hand side of \eqref{eq:mv_prod_reduction}, we have $p + 1$ bodies of dimension $n$ lying in mutually orthogonal subspaces, and so iteratively applying the projection formula for mixed volumes (Lemma \ref{lem:exact}) to each coordinate block, we see that the right-hand side equals
\[\binom{n}{n - a_0, \dots, n - a_p} \frac{(n!)^{p + 1}}{((p + 1)n)!} \operatorname{vol}_n(K)^{p + 1}.\]
Finally, combining the inequality
\[\mathcal V_{a_0, \dots, a_p} \le \binom{n}{n - a_0, \dots, n - a_p} \frac{(n!)^{p + 1}}{((p + 1)n)!} \operatorname{vol}_n(K)^{p + 1}\]
with the identity \eqref{eq:diag_mv}, we obtain
\begin{multline*}
\binom{n}{n - a_0, \dots, n - a_p}^{-1}  V_{pn}(-\Delta_p K[a_0], \imath_1 K[a_1], \dots, \imath_p K[a_p]) \\
\le \binom{(p + 1) n}{n} \operatorname{vol}_n(K)^{-1} \cdot \frac{(n!)^{p + 1}}{((p + 1)n)!} \operatorname{vol}_n(K)^{p + 1} = \frac{(n!)^p}{(pn)!} \operatorname{vol}_n(K)^p,
\end{multline*}
which is precisely \eqref{eq:higher_godbersen_redux}. The proof is complete.

<!-- lem-2: body_md -->
For any integers $a,b,c\geq 0$ summing to $2n$, one has $H_{a,b,c}=H_{b,a,c}=H_{a,c,b}$. In other words, $H_{a,b,c}$ only depends on $K$ and on $\{a,b,c\}$.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Consider the linear map $\pi_{1,2}$ given by $\pi_{1,2}(x,y)=(y,x)$. One has $\det \pi_{1, 2} = (-1)^n$, $\pi_{1, 2} \circ i_1 = i_2$, $\pi_{1, 2} \circ i_2 = i_1$ and $\pi_{1,2} \circ \Delta=\Delta$. So for any $a,b,c\ge 0$,
\begin{align*}
    V_{2n}(i_1 K[a],i_2 K[b],-\Delta K[c]) &=V_{2n}(\pi_{1,2}( i_1 K)[a],\pi_{1,2}(i_2 K)[b],\pi_{1,2}(-\Delta K)[c]) \\
    &=
    V_{2n}(i_2 K[a],i_1 K[b],-\Delta K[c]),
\end{align*}
i.e., $H_{a,b,c}=H_{b,a,c}$.

Similarly, let $T$ be the transformation with block matrix $\left(\begin{smallmatrix} I_n & -I_n \\ 0 & -I_n\end{smallmatrix}\right)$. We have $\det T = (-1)^n$, $T\circ i_1=i_1$, $T\circ i_2=-\Delta$ and $T\circ (-\Delta)=i_2$.
Therefore
\begin{align*}
    V_{2n}(i_1 K[a],i_2 K[b],-\Delta K[c]) &=V_{2n}(T(i_1 K)[a],T(i_2 K)[b],T(-\Delta K)[c]) \\
    &=V_{2n}(i_1K[a],-\Delta K[b],i_2 K[c]),
\end{align*}
and so $H_{a,b,c}=H_{a,c,b}$.

<!-- prop-2: body_md -->
\label{prop:moreequalities-AFargument}
Fix $1\leq k\leq n-1$. Assume $K\subset \mathbb{R}^n$ is a convex body satisfying 
\begin{equation}\label{eq:godbersen_equality_lifted}
V_{2n}(K^2[n],i_1 K[k],i_2 K[n-k])=V_{2n}(\Delta K[n],i_1 K[k],i_2 K[n-k]).
\end{equation}
Then for any $s \in \{1, \dots, n\}$ and any compact convex sets $L_1, \dots, L_{n-s} \in \mathcal K^{2n}$, one has
\begin{multline}\label{eq:godbersen_free}
V_{2n}(\Delta K[s], L_1,\dots, L_{n-s},i_1 K[k],i_2 K[n-k]) \\ 
= V_{2n}(K^2[s], L_1,\dots, L_{n-s},i_1 K[k],i_2 K[n-k]).
\end{multline}

<!-- proof-3: body_md -->
To simplify the notation, let $\mathcal{C} = (i_1 K[k], i_2 K[n-k])$. We will prove by induction on $j \in \{0, \dots, n-1\}$ that for any compact convex sets $L_1, \dots, L_j \in \mathcal{K}^{2n}$, the mixed volume
\[M_r^{(j)}(L_1,\dots,L_j) := V_{2n}(L_1, \dots, L_j, \Delta K[n-j-r], K^2[r], \mathcal{C})\]
is independent of $r\in \{0,1, \dots, n-j\}$. In the proof, we may assume that $L_1, \ldots, L_j$ are full-dimensional, which ensures that the $M_r^{(j)}$ are strictly positive; the general case follows by approximation and by continuity of mixed volumes.

When $j=0$, we simply write $M_r^{(0)} = M_r$. Since $\Delta K \subset K^2$, monotonicity of mixed volumes yields
\[M_0 \leq M_1 \leq \dots \leq M_{n-1} \leq M_n.\]
Our assumption is that $M_0 = M_n$, which implies that $M_0 = M_1 = \dots = M_n$. This is the case $j=0$ of our claim.

Assume now that the claim holds for some integer $0 \leq j \leq n-2$. Let $L_1, \dots, L_j \in \mathcal{K}^{2n}$ be fixed, and write $M_r^{(j)}=V_{2n}(L_1, \dots,L_j,\Delta K[n-j-r], K^2[r],\mathcal{C})$. By the induction hypothesis, $M_0^{(j)} = \dots = M_{n-j}^{(j)}$. In particular, for any $1 \leq r \leq n-j-1$, we have equality in the Alexandrov-Fenchel inequality
$(M_r^{(j)})^2 \geq M_{r-1}^{(j)} M_{r+1}^{(j)}.$
By Theorem \ref{thm:af_equality}, this equality implies that there exists a constant $\alpha > 0$ such that for any convex body $M \in \mathcal{K}^{2n}$,
\begin{multline*}
    V_{2n}(M, L_1, \dots, L_j, \Delta K[n-j-r], K^2[r-1], \mathcal{C}) \\ 
    = \alpha V_{2n}(M, L_1, \dots, L_j, \Delta K[n-j-r-1], K^2[r], \mathcal{C}).
\end{multline*}
Choosing $M = K^2$ gives $M_r^{(j)} = \alpha M_{r+1}^{(j)}$. Since $M_r^{(j)} = M_{r+1}^{(j)}$, we deduce that $\alpha = 1$. Consequently, for any arbitrary compact convex set $L_{j+1} \in \mathcal{K}^{2n}$, taking $M = L_{j+1}$ in the above equation yields:
\[M_{r-1}^{(j+1)} (L_1,\dots,L_{j+1}) = M_r^{(j+1)}(L_1,\dots,L_{j+1})\]
for all $1 \leq r \leq n-j-1$. This proves that the sequence $(M_r^{(j+1)})_{0 \leq r \leq n-j-1}$ is constant, completing the induction step.

<!-- cor-1: body_md -->
Let $1\leq k\leq n-1$. Assume $K\subset \mathbb{R}^n$ is a convex body satisfying 
    \[\alpha_k =V_n(K[n-k],-K[k])=\binom{n}{k} \operatorname{vol}_n(K).\]
    Then for any $s\in \{1,\dots,n-1\}$, one has
\begin{equation}
\label{eqn:akasidentity}
\alpha_s \alpha_k=\sum_{i=0}^s \binom{s}{i}  H_{k+i, n-k+s-i,n-s}.
\end{equation}
<!-- end cor-1 -->
<!-- proof-4: body_md -->
Applying Proposition \ref{prop:moreequalities-AFargument} with $L_1=\dots=L_{n-s}=-\Delta K$ yields
 \begin{multline}
 \label{eqn:equalitywiths}
 \binom{2n}{n} V_{2n}(\Delta K[s], -\Delta K[n-s],i_1 K[k],i_2 K[n-k]) \\ 
= \binom{2n}{n} V_{2n}(K^2[s], -\Delta K[n-s],i_1 K[k],i_2 K[n-k]).
 \end{multline}
 
Let $E = \Delta \mathbb R^n$, and denote $\tilde{\Delta}(x)=(x,-x)$ so $\tilde{\Delta}(\mathbb{R}^n)=E^{\perp}$. By Lemma \ref{lem:exact}, the LHS rewrites
 \begin{align}\label{eq:equalitywiths_lhs}
 & \binom{2n}{n} V_{2n}(\Delta K[s], -\Delta K[n-s],i_1 K[k],i_2 K[n-k]) \nonumber \\
 &\qquad = V_E(\Delta K[s],-\Delta K[n-s]) V_{E^{\perp}}((P_{E^\perp}\circ i_1)(K)[k],(P_{E^\perp}\circ i_2)(K)[n-k]) \nonumber \\
 &\qquad =2^{-n}V_E(\Delta K[s],-\Delta K[n-s])V_{E^{\perp}}(\tilde{\Delta}(K)[k],\tilde{\Delta}(-K)[n-k]) \nonumber  \\
 &\qquad = \alpha_s \alpha_k,
 \end{align}
 where we used that $P_{E^\perp} \circ i_1=\frac{1}{2}\tilde{\Delta}$, $P_{E^\perp} \circ i_2=-\frac{1}{2}\tilde{\Delta}$ along with homogeneity of mixed volumes for the second equality, and that $|\det(\Delta)|=|\det(\tilde{\Delta})|=2^{n/2}$ for the last equality.
By multilinearity, since $K^2=i_1K+i_2 K$, the RHS of \eqref{eqn:equalitywiths} rewrites as
 \begin{equation} 
 \label{eq:equalitywiths_rhs}
 \binom{2n}{n}V_{2n}(K^2[s],-\Delta K[n-s],i_1K[k],i_2K[n-k])=\sum_{i=0}^s \binom{s}{i} H_{k+i,n-k+s-i, n-s}
 \end{equation}
 where we recall the notation $H_{a,b,c}=\binom{2n}{n}V_{2n}(i_1 K[a],i_2 K[b], -\Delta K[c])$. Combining \eqref{eqn:equalitywiths}, \eqref{eq:equalitywiths_lhs}, and \eqref{eq:equalitywiths_rhs} gives \eqref{eqn:akasidentity}.

Let $1\leq k\leq n-1$, and assume that $K\in \mathcal{K}^n$ is a convex body such that $V_n(K[n-k],-K[k])=  \binom{n}{k} \operatorname{vol}_n(K)$, i.e., $\alpha_k=\binom{n}{k} v$. To show that $K$ must be a simplex, we will show that $\alpha_1=n v$.

Taking $s=k$ in \eqref{eqn:akasidentity}, and using \eqref{eqn:generalisedgodbersen} with $(a,b,c)=(k+i,n-i,n-k)$, $i\in \{0,\dots, k\}$, one finds
\begin{align*} 
\alpha_k^2=\sum_{i=0}^k  \binom{k}{i} H_{k+i,n-i,n-k} &\leq \sum_{i=0}^k \binom{k}{k - i} \binom{n}{n-k-i,i,k} v^2 \\ 
&=\binom{n}{k} \sum_{i=0}^k \binom{k}{k-i}\binom{n-k}{i} v = \binom{n}{k}^2 v^2= \alpha_k^2,
\end{align*}
where the last equality is due to our assumption $\alpha_k = \binom{n}{k} v$. Therefore we must have $H_{a,b,c}= \binom{n}{n-a,n-b,n-c} v^2$ for any $\{a,b,c\}\in \{\{k+i,n-i,n-k\}; 0\leq i\leq k\}$. In particular, taking $i=1$ yields 
\begin{equation}
    \label{eqn:eli}
    H_{k+1,n-k,n-1}=\binom{n}{n-k-1,k,1} v^2 =(n-k)\binom{n}{k} v^2.
\end{equation}

Similarly, applying \eqref{eqn:akasidentity} with $k$ replaced by $n - k$ (which is valid since $\alpha_k = \alpha_{n - k}$), taking $s = n - k$, and using \eqref{eqn:generalisedgodbersen}, one finds
\begin{align*}\alpha_{n-k}^2 = \sum_{i=0}^{n-k} \binom{n-k}{i} H_{n-k+i,n-i,k} &\leq \sum_{i=0}^{n-k} \binom{n-k}{i} \binom{n}{k-i,i,n-k} v^2 \\
&= \binom{n}{k} \sum_{i=0}^{n-k} \binom{n-k}{i}\binom{k}{k-i} v^2=\binom{n}{k}^2 v^2 = \alpha_{n-k}^2.
\end{align*}
Thus the inequalities $H_{a,b,c}\leq \binom{n}{n-a,n-b,n-c} v^2$ must be equalities for 
\[\{a,b,c\}\in \{\{n-k+i,k,n-i\} ; 0\leq i\leq n-k\}.\]
In particular, choosing $i=1$ we get
\begin{equation}
\label{eqn:maud}
H_{k,n-k+1,n-1} = \binom{n}{n-k, k-1, 1} v^2 = k\binom{n}{k} v^2.
\end{equation}

Now, taking $s=1$ in \eqref{eqn:akasidentity} yields
\begin{align*}
    \alpha_1 \cdot \alpha_k  &=H_{k,n-k+1,n-1}+H_{k+1,n-k,n-1} \\
    &=k \binom{n}{k}v^2 +(n-k)\binom{n}{k} v^2 \\
    &= n \binom{n}{k} v^2 = n v\cdot \alpha_k,
\end{align*}
where we used \eqref{eqn:eli} and \eqref{eqn:maud} for the second equality, and the assumption that $\alpha_k=\binom{n}{k} v$ for the last equality. Dividing through by $\alpha_k>0$, we obtain $\alpha_1 = n v$, i.e., $V_n(K[n - 1], -K) = n \operatorname{vol}_n(K)$, from which we deduce by Proposition \ref{prop:godbersen_j=1_equality} that $K$ is a simplex.
