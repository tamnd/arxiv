<!-- theorem-1: body_md -->
\label{thm:intro-all-points}
For every $z\in\mathbb R$,
\begin{equation}\label{eq:intro-main}
   \inf_{b\in\mathcal F}F_z(b)
   =
   \log\Delta(S_a-z)+\frac12
   \qquad\text{in }[-\infty,\infty).
\end{equation}
Both sides equal $-\infty$ if and only if $z$ is an atom of $\mu_{S_a}$.
The regularized scales form a minimizing sequence:
$F_z(b_\varepsilon)\to\log\Delta(S_a-z)+\frac12$ as
$\varepsilon\downarrow0$.
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
\label{def:nondegenerate-regular}
A point $z\in\mathbb R$ is called \textbf{nondegenerate regular} if the
boundary value
$
   G(z+i0)
$
exists, and either
$
   -\operatorname{Im}G(z+i0)\succ0,
$
or $z$ belongs to an open component of
$
   \mathbb R\setminus\operatorname{supp}\mu_{S_a}.
$
<!-- end definition-1 -->
<!-- theorem-2: body_md -->
\label{thm:intro-attainment}
Let $z$ be a nondegenerate regular point.  Then the regularized scales
converge, $b_z:=\lim_{\varepsilon\downarrow0}b_\varepsilon$ exists, and
the infimum in \eqref{eq:intro-main} is attained at $b_z$.  If $-\operatorname{Im}G(z+i0)\succ0$, then $b_z\in\partial\mathcal F$.
If $z$ lies in the resolvent set, then
$b_z\in\operatorname{int}\mathcal F$; if moreover $z$ lies in an
unbounded component of the resolvent set, then
$
   b_z=|G(z)|^{-1}.
$
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:intro-capacity}
For $a=0$,
\begin{equation}\label{eq:intro-capacity}
   \inf_{b\in\mathcal F}\Delta(b)
   =
   \operatorname{Cap}(\eta)^{1/(2m)}.
\end{equation}
Consequently,
\[
   \log\Delta(S_0)
   =
   -\frac12
   +
   \frac1{2m}\log\operatorname{Cap}(\eta),
\]
which is the Mai--Speicher formula.  Moreover, the infimum in
\eqref{eq:intro-capacity} is attained if and only if $\eta$ is
symmetrically doubly-stochastic scalable.

<!-- corollary-1: body_md -->
\label{cor:intro-brown}
Let $a_0,a_1,\dots,a_r\in M_m(\mathbb C)$ be arbitrary and
$S=a_0\otimes1+\sum_ia_i\otimes s_i$.  For $z\in\mathbb C$ let
$\mathcal F^{\mathrm h}\subset M_{2m}(\mathbb C)$ and
$F^{\mathrm h}_z$ be the feasible set and the functional
\eqref{eq:intro-F} of the self-adjoint matrix semicircle
\[
   H_z
   =
   \begin{pmatrix}0&S-z\\ S^*-\overline{z}&0\end{pmatrix}
   \in M_{2m}(\mathbb C)\otimes\mathcal A,
\]
at the spectral parameter $0$.  Then, for every $z\in\mathbb C$,
\[
   \log\Delta(S-z)
   =
   -\frac12+\inf_{b\in\mathcal F^{\mathrm h}}F^{\mathrm h}_z(b)
   \qquad\text{in }[-\infty,\infty),
\]
with both sides equal to $-\infty$ if and only if $S-z$ has a nontrivial
kernel, and with the infimum attained whenever $S-z$ is invertible.  In
particular, the Brown measure of $S$ is $\frac1{2\pi}\nabla_z^2$ of the
right-hand side.

<!-- theorem-4: body_md -->
\label{thm:closed}
For $w\in\mathbb C^+$ define
\begin{equation}\label{eq:H-def}
   H(w)
   :=
   -\frac1m\log\det G(w)
   +
   \frac12
   \operatorname{tr}_m\bigl(
      G(w)\eta(G(w))
   \bigr),
\end{equation}
where any holomorphic branch of $\log\det G$ on $\mathbb C^+$ may be
chosen.  Then
\begin{equation}\label{eq:H-logdet}
   \boxed{
   \operatorname{Re}H(w)
   =
   \log\Delta(S_a-w)
   }
   \qquad
   (w\in\mathbb C^+).
\end{equation}

Consequently, if $x\in\mathbb R$ is such that the boundary value
$
   G:=G(x+i0)
$
exists and is invertible, then
\begin{equation}\label{eq:closed-boundary}
   \boxed{
   \log\Delta(S_a-x)
   =
   -\frac1m\log|\det G|
   +
   \frac12
   \operatorname{Re}
   \operatorname{tr}_m\bigl(G\eta(G)\bigr).
   }
\end{equation}
<!-- end theorem-4 -->
<!-- proof-1: body_md -->
The key observation is that $H$ is a holomorphic primitive of the scalar
Cauchy transform.

Differentiate \eqref{eq:H-def}.  Since
\[
   \frac{d}{dw}\log\det G(w)
   =
   \operatorname{Tr}\bigl(G(w)^{-1}G'(w)\bigr),
\]
we obtain
\[
\begin{aligned}
   H'(w)
   &=
   -\operatorname{tr}_m\bigl(G^{-1}G'\bigr)
   \\
   &\quad
   +
   \frac12
   \operatorname{tr}_m\bigl(
      G'\eta(G)+G\eta(G')
   \bigr).
\end{aligned}
\]
For the covariance map
\[
   \eta(X)=\sum_j A_jXA_j,
   \qquad A_j=A_j^*,
\]
cyclicity of the trace gives
\[
   \operatorname{tr}_m\bigl(G'\eta(G)\bigr)
   =
   \operatorname{tr}_m\bigl(G\eta(G')\bigr).
\]
Hence
\begin{equation}\label{eq:H-prime-1}
   H'(w)
   =
   -\operatorname{tr}_m(G^{-1}G')
   +
   \operatorname{tr}_m\bigl(G\eta(G')\bigr).
\end{equation}

Differentiating the Dyson equation \eqref{eq:closed-dyson} gives
$
   -G^{-1}G'G^{-1}
   +
   \eta(G')
   =
   I,
$
and therefore
\[
   \eta(G')
   =
   I+G^{-1}G'G^{-1}.
\]
Substituting this into \eqref{eq:H-prime-1},
\begin{align*}
   \operatorname{tr}_m\bigl(G\eta(G')\bigr)
   &=
   \operatorname{tr}_mG
   +
   \operatorname{tr}_m(G'G^{-1}),
\end{align*}
so the logarithmic-derivative terms cancel and
\begin{equation}\label{eq:H-prime}
   \boxed{
   H'(w)=\operatorname{tr}_mG(w).
   }
\end{equation}

Now define the holomorphic logarithmic potential
\[
   \Lambda(w)
   :=
   \int_{\mathbb R}
   \log(w-t)\,d\mu_{S_a}(t),
   \qquad
   w\in\mathbb C^+,
\]
using the standard branch of the logarithm on the upper half-plane.
Then
\[
   \Lambda'(w)
   =
   \int_{\mathbb R}
   \frac{1}{w-t}\,d\mu_{S_a}(t)
   =
   \operatorname{tr}_mG(w).
\]
Thus
$
   H-\Lambda
$
is constant on $\mathbb C^+$.

As $|w|\to\infty$ non-tangentially,
\[
   G(w)
   =
   w^{-1}I+O(w^{-2}),
\]
and therefore
\[
   H(w)
   =
   \log w+o(1)
\]
up to the purely imaginary constant coming from the chosen branch of
$\log\det G$.  Similarly,
\[
   \Lambda(w)=\log w+o(1).
\]
Hence $H-\Lambda$ is purely imaginary, and consequently
\[
   \operatorname{Re}H(w)
   =
   \operatorname{Re}\Lambda(w)
   =
   \int_{\mathbb R}
   \log|w-t|\,d\mu_{S_a}(t).
\]
This proves \eqref{eq:H-logdet}.

For the boundary formula, set $w=x+i\varepsilon$.  Then
\[
   \operatorname{Re}H(x+i\varepsilon)
   =
   \int_{\mathbb R}
   \log\sqrt{(t-x)^2+\varepsilon^2}\,
   d\mu_{S_a}(t).
\]
As $\varepsilon\downarrow0$, the right-hand side decreases to
\[
   \int_{\mathbb R}\log|t-x|\,d\mu_{S_a}(t)
   =
   \log\Delta(S_a-x),
\]
possibly with value $-\infty$.

If $G(x+i0)$ exists and is invertible, then
\[
   -\frac1m\log|\det G(x+i\varepsilon)|
   +
   \frac12
   \operatorname{Re}
   \operatorname{tr}_m\bigl(
      G(x+i\varepsilon)\eta(G(x+i\varepsilon))
   \bigr)
\]
converges to the right-hand side of

<!-- remark-1: body_md -->
\label{rem:extended-action}

 We record three observations.

\smallskip
\noindent\textbf{No minimax.}

Taking $b=tI$ with $t\downarrow0$
shows that $\inf_{b\succ0}\mathcal{K}_z(b,u,\gamma)=-\infty$ for every
$(u,\gamma)$.  Hence
\[
   \sup_{\substack{u=u^*\\ \gamma\succeq0}}\ \inf_{b\succ0}
   \mathcal{K}_z(b,u,\gamma)
   =-\infty ,
\]
and the order of the extrema in \eqref{eq:intro-F0} and
\eqref{eq:intro-main-formula} cannot be interchanged.

\smallskip
\noindent\textbf{Common critical point and critical value.}
Consider the regularized functional
$\mathcal{K}_{z,\varepsilon}=\mathcal{K}_z+\varepsilon\operatorname{tr}_m\gamma$ of Section~\ref{sec:regularized}.
The computation in the proof of
Proposition~\ref{prop:action-identity} gives
$\mathcal{K}_{z,\varepsilon}(b,u,\gamma)=\operatorname{Re}\mathcal A_{z+i\varepsilon}(M)+\frac12$ on normalized
triples.  The stationarity conditions of $\mathcal{K}_{z,\varepsilon}$ in $u$, $\gamma$
and $b$ are
\[
   L_b^+u=a_z,
   \qquad
   L_b^-\gamma=\varepsilon I,
   \qquad
   b^{-1}=ubu+\gamma b\gamma ,
\]
which are the balanced Dyson equations \eqref{eq:balanced}.  

Eliminating a group of variables through its own stationarity condition,
when that condition determines them uniquely and smoothly, changes neither
the critical points nor the critical values of a function.  For
$b\in\mathcal F^{\circ}$, eliminating $(u,\gamma)$ produces the reduced objective
\[
   F_{z,\varepsilon}(b)=\max_{u,\gamma}\mathcal{K}_{z,\varepsilon}(b,u,\gamma).
\]
For $\gamma\succ0$, the function $b\mapsto\mathcal{K}_{z,\varepsilon}(b,u,\gamma)$ is
strictly geodesically concave: $\operatorname{tr}_m\log b$ is affine along geodesics,
and by Lemma~\ref{lem:C-geodesic} the second derivative of
$\langle\gamma,\mathcal C_{b_t}\gamma\rangle$ at $t=0$ is
$\|Xw+wX\|_2^2$ with $w=b^{1/2}\gamma b^{1/2}\succ0$, which is positive
for $X\neq0$.  Its unique critical point is the unitary normalizer of
$-u-i\gamma$, so by Proposition~\ref{prop:action-identity}
\[
   \max_{b\succ0}\mathcal{K}_{z,\varepsilon}(b,u,\gamma)
   =\operatorname{Re}\mathcal A_{z+i\varepsilon}(-u-i\gamma)+\frac12 .
\]
Thus $F_{z,\varepsilon}$ and $\operatorname{Re}\mathcal A_{z+i\varepsilon}+\frac12$ are two reductions of
the same function, and the equality of their critical values is automatic.

\smallskip
\noindent\textbf{Type of the critical point.}
What is not automatic is the nature of the critical point.  Since
$\mathcal A_w$ is holomorphic in $M$, a nondegenerate critical point of
$\operatorname{Re}\mathcal A_w$ has as many negative as positive directions, so the Dyson
action itself admits no extremal characterization of the Dyson solution.
In the extended functional the negative directions are carried by the
auxiliary variables $(u,\gamma)$ and are removed by the inner
maximization.  The remaining problem in $b$ is geodesically convex
(Corollary~\ref{cor:Freg-convex}), and its critical point is a global
minimizer.  This is the reason for introducing the scale $b$ as an
independent variable.

<!-- lemma-1: body_md -->
\label{lem:feasible-pf}
For every $b\succ0$, the normalized covariance map
$\widehat\eta_b$ has a nonzero positive semidefinite eigenmatrix
$P_b$ at its spectral radius:
\begin{equation}\label{eq:positive-Perron-eigenmatrix}
   \widehat\eta_b(P_b)=r_bP_b,
   \qquad
   P_b\succeq0,\quad P_b\ne0,
   \qquad
   r_b
   =
   \|\widehat\eta_b\|_{2\to2}
   =
   \lambda_{\max}(\widehat\eta_b).
\end{equation}
For $b\succ0$, the following are equivalent:
\begin{enumerate}
\item[(i)] $b\in\mathcal{F}$;
\item[(ii)] $\lambda_{\max}(\widehat\eta_b)\leq1$;
\item[(iii)] $\|\widehat\eta_b\|\leq1$;
\item[(iv)]
$L_b^- \succeq0$ 
   and
 $  L_b^+ \succeq0$ as Hilbert-Shmidt operators. 
\end{enumerate}
Moreover,
\begin{equation}\label{eq:interior-feasible}
   b\in\operatorname{int}\mathcal{F}
   \quad\Longleftrightarrow\quad
   \|\widehat\eta_b\|<1.
\end{equation}
In particular,
$L_b^- \succ 0$ and $L_b^+\succ0$ for $b\in\mathcal F^\circ$.

Let
$

$

$

   T_b(\mathcal C_b-\eta)T_b

   I-\widehat\eta_b.
\end{equation}

   \quad\Longleftrightarrow\quad
   I-\widehat\eta_b\succeq0.
\]
Because $\widehat\eta_b$ is Hilbert--Schmidt self-adjoint, the latter is

   \lambda_{\max}(\widehat\eta_b)\leq1.

We next compare the largest eigenvalue with the operator norm.

\]

   =
   \lambda^{-2}>1.
\]

boundary point.  This proves \eqref{eq:interior-feasible}.

\label{prop:indicator}
For every $b\succ0$,
\begin{equation}\label{eq:indicator}
 \sup_{\gamma\succeq0}
 \left\{-\frac12\left\langle \gamma,L_b^-\gamma\right\rangle\right\}
 =\begin{cases}
   0,&b\in\mathcal{F},\\
   +\infty,&b\notin\mathcal{F}.
  \end{cases}
\end{equation}
Consequently,
\begin{equation}\label{eq:three-variable-real}
 \inf_{b\succ0}\sup_{\substack{u=u^*\\\gamma\succeq0}}
    \mathscr K_z(b,u,\gamma)
 =\inf_{b\in\mathcal{F}}F_z(b).
\end{equation}

<!-- proof-4: body_md -->
If $b\in\mathcal{F}$, then $L_b^-=\mathcal C_b-\eta\succeq0$ on the whole
Hilbert--Schmidt space, so every quantity in the supremum is
nonpositive, and $\gamma=0$ gives the value $0$.

Let $b\notin\mathcal{F}$.  By Lemma~\ref{lem:feasible-pf},
$\|\widehat\eta_b\|>1$.  The map $\widehat\eta_b$ is positive, so by the
Perron--Frobenius theorem for positive maps \cite{EvansHoeghKrohn1978}
its spectral radius $r$ is an eigenvalue with a positive semidefinite
eigenvector: there is $Q\succeq0$, $Q\neq0$, with
$\widehat\eta_b(Q)=rQ$.  Since $\widehat\eta_b$ is Hilbert--Schmidt
self-adjoint, $r=\|\widehat\eta_b\|>1$.

Put $\gamma_0:=b^{-1/2}Qb^{-1/2}\succeq0$.  Then
$\mathcal C_b(\gamma_0)=b^{1/2}Qb^{1/2}$, and by the definition
\eqref{eq:eta-hat} of $\widehat\eta_b$,
\[
 \left\langle \gamma_0,\mathcal C_b\gamma_0\right\rangle=\left\langle Q,Q\right\rangle,
 \qquad
 \left\langle \gamma_0,\eta(\gamma_0)\right\rangle
 =\left\langle Q,\widehat\eta_b(Q)\right\rangle
 =r\left\langle Q,Q\right\rangle.
\]
Hence $\left\langle \gamma_0,L_b^-\gamma_0\right\rangle=(1-r)\left\langle Q,Q\right\rangle<0$.  For $t>0$ we
have $t\gamma_0\succeq0$ and
\[
 -\frac12\left\langle t\gamma_0,L_b^-(t\gamma_0)\right\rangle
 =\frac{t^2}{2}(r-1)\left\langle Q,Q\right\rangle\longrightarrow+\infty
 \qquad(t\to\infty).
\]
This proves \eqref{eq:indicator}.

For fixed $b\succ0$, $\mathscr K_z(b,u,\gamma)$ is the sum of $\operatorname{tr}_m\log b$,
a function of $u$ alone, and a function of $\gamma$ alone.  Therefore
\[
 \sup_{\substack{u=u^*\\\gamma\succeq0}}\mathscr K_z(b,u,\gamma)
 =\operatorname{tr}_m\log b
  +\sup_{u=u^*}\Bigl\{\left\langle a_z,u\right\rangle-\frac12\left\langle u,L_b^+u\right\rangle\Bigr\}
  +\sup_{\gamma\succeq0}\Bigl\{-\frac12\left\langle \gamma,L_b^-\gamma\right\rangle\Bigr\}.
\]
The supremum over $u$ is at least $0$ (take $u=0$), so the right-hand
side is well defined in $(-\infty,+\infty]$.  By \eqref{eq:indicator}
and \eqref{eq:intro-F}, it equals $F_z(b)$ for $b\in\mathcal{F}$ and $+\infty$
for $b\notin\mathcal{F}$.  Taking the infimum over $b\succ0$ gives
\eqref{eq:three-variable-real}; the order of the extrema is unchanged.

$

\label{lem:resolvent-geodesic}
Let $t\mapsto\mathcal C_t$ be a geodesic in $\operatorname{PD}(B)$,
and let $\Theta=\Theta^*$ be a fixed operator on $B$. Suppose
$L_t:=\mathcal C_t+\Theta\succ0$ throughout the interval under
consideration. Define
\begin{equation*}
   \widehat\Theta_t
   :=\mathcal C_t^{-1/2}\Theta\mathcal C_t^{-1/2},
   \qquad
   K_t:=\mathcal C_t^{-1/2}\mathcal C_t'L_t^{-1}.
\end{equation*}
Then
\begin{equation}\label{eq:resolvent-hessian}
   \frac{d^2}{dt^2}L_t^{-1}=K_t^*\Gamma_tK_t,
\end{equation}
where
\begin{equation}\label{eq:Cayley-factor}
   \Gamma_t
   :=(I-\widehat\Theta_t)(I+\widehat\Theta_t)^{-1}.
\end{equation}
In particular, at every $t$ for which $\Theta\preceq\mathcal C_t$,
\begin{equation*}
   \frac{d^2}{dt^2}L_t^{-1}\succeq0.
\end{equation*}
No positivity assumption on the fixed operator $\Theta$ is required.

Put $R_t:=L_t^{-1}$. Differentiating the inverse gives
\begin{equation*}
   R_t'=-R_t\mathcal C_t'R_t,
   \qquad
   R_t''=2R_t\mathcal C_t'R_t\mathcal C_t'R_t
            -R_t\mathcal C_t''R_t.
\end{equation*}
The geodesic equation
$\mathcal C_t''=\mathcal C_t'\mathcal C_t^{-1}\mathcal C_t'$
therefore yields
\begin{equation}\label{eq:resolvent-second-factor}
   R_t''
   =R_t\mathcal C_t'
      \bigl(2R_t-\mathcal C_t^{-1}\bigr)
      \mathcal C_t'R_t.
\end{equation}
Since
\begin{equation*}
   L_t=\mathcal C_t^{1/2}
           (I+\widehat\Theta_t)\mathcal C_t^{1/2},
\end{equation*}
we have
\begin{align}
   2L_t^{-1}-\mathcal C_t^{-1}
   &=\mathcal C_t^{-1/2}
       \bigl[2(I+\widehat\Theta_t)^{-1}-I\bigr]
       \mathcal C_t^{-1/2}\notag\\
   &=\mathcal C_t^{-1/2}
       (I-\widehat\Theta_t)(I+\widehat\Theta_t)^{-1}
       \mathcal C_t^{-1/2}.
   \label{eq:Cayley-factorization}
\end{align}
Substitution into \eqref{eq:resolvent-second-factor} proves

Hence $\Gamma_t\succeq0$ by functional calculus, and

\label{prop:geodesic-hessian}

\end{equation*}

      (X\widetilde u+\widetilde uX)
   \right\rangle.
   }
\end{equation}

<!-- proof-10: body_md -->
On $\operatorname{int}\mathcal F$,
\begin{equation*}

\end{equation*}
The quadratic term is geodesically convex by
Corollary~\ref{cor:two-inverse-geodesic}. The logarithmic term is

   \operatorname{tr}_m\log b_t
   =\operatorname{tr}_m\log b
      +t\,\operatorname{tr}_mX.
\end{equation*}
This proves geodesic convexity of $F_z$.

\end{equation*}

\label{lem:pup}
Let

   \qquad
   X=X^*,
   \qquad

\label{rem:two-stability-operators}
The balanced Dyson equations explain the appearance of the two operators
\[
   I+\widehat\eta_b
   \qquad\text{and}\qquad
   I-\widehat\eta_b
\]
in the variational problem.

The real part of the Dyson solution is determined through
$I+\widehat\eta_b$, while its imaginary part is controlled by
$I-\widehat\eta_b$.  In particular, at a real boundary point
$\varepsilon=0$,
\[
   (I-\widehat\eta_b)Q=0.
\]
Thus a nonzero imaginary part of the boundary Dyson solution produces a
nontrivial kernel of the stability operator
$I-\widehat\eta_b$, and hence an active feasibility constraint for the limiting scale $b_z$.

\label{prop:elimination}
Fix $z\in\mathbb R$ and $\varepsilon>0$. For $b\in\mathcal F^{\circ}$, the unique
maximizers in \eqref{eq:regularized-F-definition} are
\begin{equation}\label{eq:maximizers}
 u_b=(L_b^+)^{-1}a_z,
 \qquad
 \gamma_b=\varepsilon(L_b^-)^{-1}I\succ0.
\end{equation}
Their value is
\begin{equation}\label{eq:partial-sup}
\boxed{\begin{aligned}
 F_{z,\varepsilon}(b)
 ={}&\operatorname{tr}_m\log b
 +\frac12\left\langle a_z,(L_b^+)^{-1}a_z\right\rangle\\
 &+\frac{\varepsilon^2}{2}\left\langle I,(L_b^-)^{-1}I\right\rangle.
\end{aligned}}
\end{equation}
For $b\notin\mathcal F^{\circ}$, the supremum is $+\infty$. Thus $\mathcal F^{\circ}$ is

<!-- end proposition-4 -->
<!-- proof-14: body_md -->
For $b\in\mathcal F^{\circ}$, both $L_b^+$ and $L_b^-$ are positive definite
as Hilbert--Schmidt operators. Completing the square in the two
Hermitian variables gives the unconstrained maximizers
\eqref{eq:maximizers} and the value \eqref{eq:partial-sup}.
It remains to check that the second maximizer is positive definite.

Write
$\widehat\eta_b=\mathcal C_{b^{-1/2}}\eta\mathcal C_{b^{-1/2}}$.
The congruence identities give
\begin{equation}\label{eq:positive-inverse}
 (L_b^-)^{-1}
 =\mathcal C_{b^{-1/2}}
   (\mathrm{id}-\widehat\eta_b)^{-1}
   \mathcal C_{b^{-1/2}},
 \qquad
 (\mathrm{id}-\widehat\eta_b)^{-1}
 =\sum_{k=0}^{\infty}\widehat\eta_b^{\,k}.
\end{equation}
The series converges because $\|\widehat\eta_b\|_{2\to2}<1$.
Each summand is a positive map. On applying the full expression to
$I$, its zeroth summand is $b^{-2}$. Consequently,
\begin{equation*}
 (L_b^-)^{-1}I\succeq b^{-2}\succ0.
\end{equation*}
Hence $\gamma_b\succ0$: the positivity restriction on $\gamma$
does not change the maximizer or its value.

Now suppose that $b\notin\mathcal F^{\circ}$. By
Lemma~\ref{lem:feasible-pf}, the map $\widehat\eta_b$ has a
nonzero positive semidefinite eigenmatrix $P$ with
$\widehat\eta_b(P)=rP$, where
$r=\|\widehat\eta_b\|_{2\to2}\geq1$.
Put $\gamma_0=b^{-1/2}Pb^{-1/2}\succeq0$. Then
\begin{equation*}
 \left\langle \gamma_0,L_b^-\gamma_0\right\rangle
 =(1-r)\|P\|_2^2,
 \qquad \operatorname{tr}_m\gamma_0>0.
\end{equation*}
Taking $u=0$ and $\gamma=t\gamma_0$ gives
\begin{equation*}
 \mathcal{K}_{z,\varepsilon}(b,0,t\gamma_0)
 =\operatorname{tr}_m\log b
   +\frac{t^2}{2}(r-1)\|P\|_2^2
   +\varepsilon t\operatorname{tr}_m\gamma_0.
\end{equation*}
This tends to $+\infty$ as $t\to\infty$, whether $r>1$ or $r=1$.

<!-- proof-15: body_md -->
By Lemma~\ref{lem:pup}, applied with $X=-u_\varepsilon$ and $Y=\gamma_\varepsilon$,
the normalizer $b_\varepsilon$ exists, and by \eqref{eq:normalizer-inverse},
$G_\varepsilon^{-1}=b_\varepsilon G_\varepsilon^*b_\varepsilon$.  The Dyson equation and this identity
give two expressions for the inverse: 
\begin{equation*}
\begin{aligned}
 G_\varepsilon^{-1}
 &=-a_z+\eta(u_\varepsilon)
        +i\bigl(\varepsilonI+\eta(\gamma_\varepsilon)\bigr),\\
 G_\varepsilon^{-1}
 &=-b_\varepsilon u_\varepsilon b_\varepsilon+i b_\varepsilon\gamma_\varepsilon b_\varepsilon.
\end{aligned}
\end{equation*}

\eqref{eq:balanced}. Adding
$G_\varepsilon^*b_\varepsilon G_\varepsilon=b_\varepsilon^{-1}$ and
$G_\varepsilon b_\varepsilon G_\varepsilon^*=b_\varepsilon^{-1}$ cancels the mixed terms
and gives the third equation.

\begin{equation*}
 Q-\widehat\eta_{b_\varepsilon}(Q)=\varepsilon b_\varepsilon^{-1}.
\end{equation*}

 &=\left\langle P_0,Q\right\rangle-\left\langle P_0,\widehat\eta_{b_\varepsilon}(Q)\right\rangle\\
 &=\left\langle P_0,Q\right\rangle-\left\langle \widehat\eta_{b_\varepsilon}(P_0),Q\right\rangle\\

\end{aligned}
\end{equation*}

<!-- theorem-5: body_md -->
\label{thm:regularized}
For every $z\in\mathbb R$ and $\varepsilon>0$,
\begin{equation}\label{eq:regularized-theorem}
 \boxed{
 \log\Delta(S_a-z-i\varepsilon)
 =-\frac12+\min_{b\in\mathcal F^{\circ}}F_{z,\varepsilon}(b).
 }
\end{equation}
The minimum is attained at the scale $b_\varepsilon$ from
Lemma~\ref{lem:Dyson-scale}. Equivalently,
\begin{equation}\label{eq:full-regularized}
 \log\Delta(S_a-z-i\varepsilon)
 =-\frac12+\min_{b\succ0}
    \sup_{\substack{u=u^*\\\gamma\succeq0}}
       \mathcal{K}_{z,\varepsilon}(b,u,\gamma).
\end{equation}
<!-- end theorem-5 -->
<!-- proof-18: body_md -->
\textbf{Identification of the maximizing variables and stationarity.}
At the Dyson scale, the first two balanced equations give explicitly
\begin{equation*}
 u_{b_\varepsilon}=(L_{b_\varepsilon}^+)^{-1}a_z=u_\varepsilon,
 \qquad
 \gamma_{b_\varepsilon}=\varepsilon(L_{b_\varepsilon}^-)^{-1}I=\gamma_\varepsilon.
\end{equation*}
Substituting these identities into the gradient formula yields
\begin{equation*}
\begin{aligned}
 \nabla F_{z,\varepsilon}(b_\varepsilon)
 &=b_\varepsilon^{-1}-u_\varepsilon b_\varepsilon u_\varepsilon
                    -\gamma_\varepsilon b_\varepsilon\gamma_\varepsilon\\
 &=\underbrace{\bigl(b_\varepsilon^{-1}-u_\varepsilon b_\varepsilon u_\varepsilon\bigr)}
      _{\displaystyle=\,\gamma_\varepsilon b_\varepsilon\gamma_\varepsilon}
      -\gamma_\varepsilon b_\varepsilon\gamma_\varepsilon
 =0.
\end{aligned}
\end{equation*}
Here the underbraced equality is exactly the third balanced equation.
Thus the first two equations identify the inner maximizers, while the
third makes their reduced objective stationary in $b$.

\textbf{Global minimality.}
Fix $c\in\mathcal F^{\circ}$ and take the geodesic
\begin{equation*}
 b_t=b_\varepsilon^{1/2}e^{tX}b_\varepsilon^{1/2},
 \qquad
 X=\log(b_\varepsilon^{-1/2}cb_\varepsilon^{-1/2}),\quad 0\leq t\leq1.
\end{equation*}
It lies in $\mathcal F^{\circ}$. The function $f(t)=F_{z,\varepsilon}(b_t)$ is convex
by Corollary~\ref{cor:Freg-convex}, and $f'(0)=0$ by stationarity.
Therefore $F_{z,\varepsilon}(c)=f(1)\geq f(0)=F_{z,\varepsilon}(b_\varepsilon)$.
This proves global minimality without requiring compactness of a
minimizing sequence.

\textbf{Value at the minimum.}
Write $G=G(z+i\varepsilon)=-u_\varepsilon-i\gamma_\varepsilon$.
At the identified maximizers, the explicit formula for the objective
becomes
\begin{equation*}
 F_{z,\varepsilon}(b_\varepsilon)
 =\operatorname{tr}_m\log b_\varepsilon
   +\frac12\left\langle a_z,u_\varepsilon\right\rangle
   +\frac{\varepsilon}{2}\operatorname{tr}_m\gamma_\varepsilon.
\end{equation*}
Multiplying the Dyson equation by $G$, taking the normalized trace,
and then taking real parts gives
\begin{equation*}
 \left\langle a_z,u_\varepsilon\right\rangle+\varepsilon\operatorname{tr}_m\gamma_\varepsilon
 =1+\operatorname{Re}\operatorname{tr}_m\bigl(G\eta(G)\bigr).
\end{equation*}
Also, unitarity of $b_\varepsilon^{1/2}Gb_\varepsilon^{1/2}$ implies
$\operatorname{tr}_m\log b_\varepsilon=-m^{-1}\log|\det G|$. Consequently,
\begin{equation*}
\begin{aligned}
 F_{z,\varepsilon}(b_\varepsilon)-\frac12
 &=-\frac1m\log|\det G|
    +\frac12\operatorname{Re}\operatorname{tr}_m\bigl(G\eta(G)\bigr)\\
 &=\log\Delta(S_a-z-i\varepsilon),
\end{aligned}
\end{equation*}
where the last equality is the closed determinant identity of
Theorem~\ref{thm:closed}. This proves
\eqref{eq:regularized-theorem}.
Finally, $F_{z,\varepsilon}=\sup_{u,\gamma}\mathcal{K}_{z,\varepsilon}$ by definition,
and Proposition~\ref{prop:elimination} makes this supremum infinite
off $\mathcal F^{\circ}$. This proves \eqref{eq:full-regularized}.

<!-- proof-19: body_md -->
For each fixed $b\in\mathcal F^{\circ}$, positivity of $(L_b^-)^{-1}$ as a
Hilbert--Schmidt operator gives
\begin{equation}\label{eq:pointwise-limit}
 F_{z,\varepsilon}(b)
 =F_z(b)+\frac{\varepsilon^2}{2}\left\langle I,(L_b^-)^{-1}I\right\rangle
 \downarrow F_z(b)

\end{equation}

$m_0=\inf_{\mathcal F^{\circ}}F_z$. These infima are bounded above by the value

$\varepsilon\downarrow0$ and satisfy $m_\varepsilon\geq m_0$. Conversely, for

 \leq\lim_{\varepsilon\downarrow0}F_{z,\varepsilon}(b)=F_z(b).

<!-- proof-21: body_md -->
We first recall why the regularized scales form a minimizing
family for the unregularized objective. Put
\begin{equation*}
   m_\varepsilon
   :=
   \min_{b\in\operatorname{int}\mathcal F}
      F_{z,\varepsilon}(b).
\end{equation*}
The real-axis limiting argument gives
$m_\varepsilon\to m_z$. Since the regularization term is
nonnegative and $b_\varepsilon$ minimizes
$F_{z,\varepsilon}$,
\begin{equation*}
   m_z
   \leq F_z(b_\varepsilon)
   \leq F_{z,\varepsilon}(b_\varepsilon)
   =m_\varepsilon.
\end{equation*}
It follows that
\begin{equation}\label{eq:attainment-minimizing-family}
   F_z(b_\varepsilon)\longrightarrow m_z
   \qquad(\varepsilon\downarrow0),
\end{equation}
also when $m_z=-\infty$.

Next, $b_{\varepsilon_k}\in\mathcal F$ and continuity of
$b\mapsto\mathcal C_b$ give
\begin{equation*}
   \mathcal C_{b_*}-\eta
   =
   \lim_{k\to\infty}
   \bigl(\mathcal C_{b_{\varepsilon_k}}-\eta\bigr)
   \succeq0.
\end{equation*}
Because $b_*\succ0$, this proves $b_*\in\mathcal F$.

We use the supremum definition to pass to the limit in the
objective, without assuming that $L_{b_*}^+$ is invertible.
For every fixed Hermitian matrix $u$, set
\begin{equation*}
   \ell_z(b,u)
   :=
   \operatorname{tr}_m\log b
   +\langle a_z,u\rangle
   -\frac12\langle u,L_b^+u\rangle.
\end{equation*}
The function $b\mapsto\ell_z(b,u)$ is continuous on the
positive definite cone, and
\begin{equation*}
   F_z(b)=\sup_{u=u^*}\ell_z(b,u),
   \qquad b\in\mathcal F.
\end{equation*}
Therefore, for every fixed $u=u^*$,
\begin{equation*}
   \ell_z(b_*,u)
   =
   \lim_{k\to\infty}\ell_z(b_{\varepsilon_k},u)
   \leq
   \liminf_{k\to\infty}F_z(b_{\varepsilon_k}).
\end{equation*}
Taking the supremum over $u$ proves lower semicontinuity
along this sequence:
\begin{equation*}
   F_z(b_*)
   \leq
   \liminf_{k\to\infty}F_z(b_{\varepsilon_k})
   =m_z.
\end{equation*}

Taking $u=0$ in the supremum definition gives
\begin{equation*}
   F_z(b_*)\geq\operatorname{tr}_m\log b_*>-\infty.
\end{equation*}
Hence $m_z=-\infty$ is impossible under the cluster-point
hypothesis. Moreover, $m_z<+\infty$, since $F_z$ is finite
at any strictly feasible matrix. Thus $m_z$ is finite.
Finally, feasibility of $b_*$ gives the reverse inequality
$m_z\leq F_z(b_*)$, and therefore
$
   F_z(b_*)=m_z.
$

<!-- remark-6: body_md -->
\label{rem:attainment}
Corollary~\ref{cor:regularized-value-limit} proves attainment
at nondegenerate regular points. More generally,
Proposition~\ref{prop:attainment-cluster-point} shows that
any positive definite cluster point of the regularized
scales is a minimizer. We explain what this criterion gives
in the different boundary regimes.

\smallskip
\noindent\textbf{Finite boundary values.}
Suppose that the finite limit $G:=G(x+i0)$ exists.
By Lemma~\ref{lem:boundary-invertible}, $G$ is invertible.
The unitary factorization
\begin{equation*}
   G(x+i\varepsilon)
   =
   b_\varepsilon^{-1/2}
   U_\varepsilon
   b_\varepsilon^{-1/2}
\end{equation*}
therefore gives
\begin{equation*}
   \det b_\varepsilon
   =
   |\det G(x+i\varepsilon)|^{-1}
   \longrightarrow
   |\det G|^{-1}\in(0,\infty).
\end{equation*}

Consequently, an upper bound on the scales along a sequence
also gives a uniform positive lower bound on their eigenvalues.
Indeed, suppose that
$\|b_{\varepsilon_k}\|\leq M$ for some
$\varepsilon_k\downarrow0$. For all sufficiently large $k$,
we have $\det b_{\varepsilon_k}\geq d$ for some $d>0$, and hence
\begin{equation*}
   \lambda_{\min}(b_{\varepsilon_k})
   \geq
   \frac{\det b_{\varepsilon_k}}
        {\|b_{\varepsilon_k}\|^{m-1}}
   \geq
   \frac{d}{M^{m-1}}>0.
\end{equation*}
Thus these scales lie in a compact subset of the positive
definite cone. A further subsequence converges to
$b_*\succ0$, and
Proposition~\ref{prop:attainment-cluster-point} gives
\begin{equation*}
   b_*\in\mathcal F,
   \qquad
   F_x(b_*)
   =
   \inf_{b\in\mathcal F}F_x(b)
   =
   \log\Delta(S_a-x)+\frac12.
\end{equation*}

Therefore, whenever $G(x+i0)$ exists as a finite matrix,
boundedness of the regularized scales along just one sequence
is sufficient for attainment. Convergence of the whole family
is not required. We do not know whether such a bounded sequence
always exists, in particular at degenerate regular points
and spectral edges. In Example~\ref{exa:coupled}, numerical
computations suggest convergence of the normalizers.

\smallskip
\noindent\textbf{Singular boundary values.}
If $G(x+i0)$ does not exist as a finite matrix, the Puiseux
expansion in Lemma~\ref{lem:local-puiseux-control} gives
$
   \|G(x+i\varepsilon)\|\longrightarrow\infty.
$
The unitary factorization implies
\begin{equation*}
   \|G(x+i\varepsilon)\|
   \leq
   \|b_\varepsilon^{-1}\|,
\end{equation*}
and therefore
\begin{equation*}
   \lambda_{\min}(b_\varepsilon)
   =
   \|b_\varepsilon^{-1}\|^{-1}
   \leq
   \|G(x+i\varepsilon)\|^{-1}
   \longrightarrow0.
\end{equation*}
Thus the regularized scales leave every compact subset of the
positive definite cone, and the cluster-point criterion does
not apply. This escape does not by itself rule out attainment
by some other feasible matrix: the criterion is sufficient,
not an established characterization of attainment.

\smallskip
\noindent\textbf{Non-attainment.}
At an atom, the infimum is $-\infty$ and cannot be attained,
because every $b\in\mathcal F$ satisfies
\begin{equation*}
   F_x(b)
   \geq
   \operatorname{tr}_m\log b
   >
   -\infty.
\end{equation*}
At general non-atomic singular points, the preceding
compactness argument leaves attainment unresolved.

The centered case provides a more precise conclusion.
For $a=0$ and $x=0$,
Theorem~\ref{thm:centered-capacity} states that the infimum
is attained if and only if $\eta$ is symmetrically
doubly-stochastic scalable. In particular, when
$\operatorname{Cap}(\eta)>0$ but $\eta$ is not symmetrically
doubly-stochastic scalable, the infimum is finite but is
not attained.

$\Delta(H_z)=\Delta(S-z)^{1/2}\Delta((S-z)^*)^{1/2}=\Delta(S-z)$; see
\cite[Lemma~3.5(ii)]{MaiSpeicher2024}.  This proves
\eqref{eq:brown-variational}.

By Corollary~\ref{cor:minus-infinity-atoms}, both sides are $-\infty$ exactly when
$\mu_{H_z}$ has an atom at $0$, that is, when $H_z$ has a nontrivial
kernel.  Since $\ker H_z=\ker(S-z)\oplus\ker((S-z)^*)$ and, in a finite
von Neumann algebra, $\ker T$ and $\ker T^*$ have the same trace, this
happens if and only if $\ker(S-z)\neq\{0\}$.

If $S-z$ is invertible, then $H_z$ is invertible, so $0$ lies in an open
component of $\mathbb R\setminus\operatorname{supp}\mu_{H_z}$ and is a nondegenerate
regular point of $H_z$; the infimum is attained by Corollary~\ref{cor:regularized-value-limit}.
Finally, $z\mapsto\log\Delta(S-z)$ is the logarithmic potential of the
Brown measure of $S$, which is recovered from it as
$\frac1{2\pi}\nabla_z^2\log\Delta(S-z)$ in the sense of distributions
\cite{Brown1986}.

For every self-adjoint completely positive covariance map $\eta$,
\begin{equation}\label{eq:centered-capacity}
   \boxed{
   \inf_{b\in\mathcal{F}}\Delta(b)
   =
   \operatorname{Cap}(\eta)^{1/(2m)}.
   }
\end{equation}

Moreover, the infimum in \eqref{eq:centered-capacity} is attained if and
only if $\eta$ is symmetrically doubly-stochastic scalable; equivalently,
if and only if there exists $C\succ0$ such that

\end{equation}

We first prove the lower bound.

Let $b\in\mathcal{F}$.  Recall
$
   \widehat\eta_b
   =
   \mathcal C_{b^{-1/2}}
   \eta
   \mathcal C_{b^{-1/2}}
   =
   \mathcal S_{b^{-1/2},b^{-1/2}}(\eta).
$
By the scaling rule \eqref{eq:capacity-scaling},
\begin{equation}\label{eq:capacity-scaled-b}
   \operatorname{Cap}(\widehat\eta_b)
   =
   \det(b)^{-2}
   \operatorname{Cap}(\eta).
\end{equation}

Taking $X=I$ in the definition of capacity gives
$
   \operatorname{Cap}(\widehat\eta_b)
   \leq
   \det\widehat\eta_b(I).
$
Since $b\in\mathcal{F}$, Lemma~\ref{lem:feasible-pf} gives
$
   I-\widehat\eta_b\succeq0
$
as an operator on Hilbert--Schmidt space.  Pairing this inequality with
$I$ yields
\[
   \operatorname{tr}_m\widehat\eta_b(I)
   =
   \left\langle I,\widehat\eta_b(I)\right\rangle
   \leq
   \left\langle I,I\right\rangle
   =
   1.
\]
Moreover, $\widehat\eta_b(I)\succeq0$ as a matrix.  Hence the
arithmetic--geometric mean inequality gives
\[
   \det\widehat\eta_b(I)
   \leq
   \bigl(
      \operatorname{tr}_m\widehat\eta_b(I)
   \bigr)^m
   \leq1.
\]
Therefore
$
   \operatorname{Cap}(\widehat\eta_b)\leq1.
$
Using \eqref{eq:capacity-scaled-b},
$
   \operatorname{Cap}(\eta)
   \leq
   \det(b)^2.
$
Since
$
   \Delta(b)=\det(b)^{1/m},
$
we obtain
\[
   \Delta(b)
   \geq
   \operatorname{Cap}(\eta)^{1/(2m)}.
\]
Taking the infimum over $b\in\mathcal{F}$ gives
\begin{equation}\label{eq:capacity-lower}
   \inf_{b\in\mathcal{F}}\Delta(b)
   \geq
   \operatorname{Cap}(\eta)^{1/(2m)}.
\end{equation}

We next prove the reverse inequality.

Suppose first that $\eta$ is symmetrically doubly-stochastic scalable.
Choose $C\succ0$ satisfying
$
   \eta(C)=C^{-1}
$
and put
$
   b:=C^{-1}.
$
Then
$
   \widehat\eta_b
   =
   \mathcal S_{C^{1/2},C^{1/2}}(\eta)
$
is unital:
\[
   \widehat\eta_b(I)
   =
   C^{1/2}\eta(C)C^{1/2}
   =
   I.
\]
The map $\widehat\eta_b$ is Hilbert--Schmidt self-adjoint, so unitality
also implies trace preservation:
\[
\begin{aligned}
   \operatorname{tr}_m\widehat\eta_b(X)
   &=
   \left\langle I,\widehat\eta_b(X)\right\rangle\\
   &=
   \left\langle \widehat\eta_b(I),X\right\rangle\\
   &=
   \operatorname{tr}_m X.
\end{aligned}
\]
Thus $\widehat\eta_b$ is doubly stochastic.

By the Kadison--Schwarz inequality,
\[
   \widehat\eta_b(X)^*
   \widehat\eta_b(X)
   \preceq
   \widehat\eta_b(X^*X).
\]
Taking normalized traces and using trace preservation,
$
   \|\widehat\eta_b(X)\|_2^2
   \leq
   \|X\|_2^2.
$
Hence
$
   \|\widehat\eta_b\|_{\mathrm{HS}\to\mathrm{HS}}\leq1,
$
and therefore $b\in\mathcal{F}$.

By Lemma~\ref{lem:ds-capacity},
$
   \operatorname{Cap}(\widehat\eta_b)=1.
$
Using \eqref{eq:capacity-scaled-b},
$
   1
   =
   \det(b)^{-2}\operatorname{Cap}(\eta),
$
so
$
   \det(b)^2
   =
   \operatorname{Cap}(\eta).
$
Consequently,
\[
   \Delta(b)
   =
   \operatorname{Cap}(\eta)^{1/(2m)}.
\]
Thus equality holds in \eqref{eq:capacity-lower} whenever an exact
symmetric doubly-stochastic scaling exists.

For a general $\eta$, introduce the strictly positive regularization
\begin{equation}\label{eq:capacity-regularization}
   \eta_\varepsilon(X)
   :=
   \eta(X)
   +
   \varepsilon\,\operatorname{tr}_m(X)I,
   \qquad
   \varepsilon>0.
\end{equation}
Each $\eta_\varepsilon$ is self-adjoint, completely positive, and strictly
positive; hence it is symmetrically doubly-stochastic scalable, \cite{KarginNonSingular2026}.  Denote its
feasible set by
\[
   \mathcal{F}_\varepsilon
   :=
   \left\{
      b\succ0:
      \mathcal C_b-\eta_\varepsilon\succeq0
   \right\}.
\]
Since
$
   \eta_\varepsilon\succeq\eta,
$
we have
$
   \mathcal{F}_\varepsilon\subseteq\mathcal{F}.
$
By the already proved scalable case,
\[
   \inf_{b\in\mathcal{F}_\varepsilon}\Delta(b)
   =
   \operatorname{Cap}(\eta_\varepsilon)^{1/(2m)}.
\]
Therefore
\begin{equation}\label{eq:capacity-upper-eps}
   \inf_{b\in\mathcal{F}}\Delta(b)
   \leq
   \operatorname{Cap}(\eta_\varepsilon)^{1/(2m)}.
\end{equation}

It remains to let $\varepsilon\downarrow0$.  We claim that
\begin{equation}\label{eq:capacity-continuity}
   \operatorname{Cap}(\eta_\varepsilon)
   \longrightarrow
   \operatorname{Cap}(\eta).
\end{equation}
Since
$
   \eta_\varepsilon(X)\succeq\eta(X),$ with$X\succ0$,
monotonicity of the determinant on the positive cone gives
$
   \operatorname{Cap}(\eta_\varepsilon)
   \geq
   \operatorname{Cap}(\eta).
$

Conversely, fix $\delta>0$ and choose $X_\delta\succ0$ with
$
   \det X_\delta=1
$
and
\[
   \det\eta(X_\delta)
   \leq
   \operatorname{Cap}(\eta)+\delta.
\]
For this fixed $X_\delta$,
$
   \eta_\varepsilon(X_\delta)
   \longrightarrow
   \eta(X_\delta),
$
so
$
   \det\eta_\varepsilon(X_\delta)
   \longrightarrow
   \det\eta(X_\delta).
$
Since
\[
   \operatorname{Cap}(\eta_\varepsilon)
   \leq
   \det\eta_\varepsilon(X_\delta),
\]
we obtain
\[
   \limsup_{\varepsilon\downarrow0}
   \operatorname{Cap}(\eta_\varepsilon)
   \leq
   \operatorname{Cap}(\eta)+\delta.
\]
Letting $\delta\downarrow0$ proves
\eqref{eq:capacity-continuity}.

Passing to the limit in
\eqref{eq:capacity-upper-eps} gives
\[
   \inf_{b\in\mathcal{F}}\Delta(b)
   \leq
   \operatorname{Cap}(\eta)^{1/(2m)}.
\]
Together with \eqref{eq:capacity-lower}, this proves
\eqref{eq:centered-capacity}.

It remains to characterize attainment.  Suppose that
$b\in\mathcal{F}$ attains the infimum.  Then
$
   \det(b)^2
   =
   \operatorname{Cap}(\eta),
$
and therefore, by \eqref{eq:capacity-scaled-b},
$
   \operatorname{Cap}(\widehat\eta_b)=1.
$
In the chain
\[
   \operatorname{Cap}(\widehat\eta_b)
   \leq
   \det\widehat\eta_b(I)
   \leq
   \bigl(
      \operatorname{tr}_m\widehat\eta_b(I)
   \bigr)^m
   \leq
   1,
\]
all inequalities must be equalities.  In particular,
$
   \operatorname{tr}_m\widehat\eta_b(I)=1
$
and
$
   \det\widehat\eta_b(I)=1.
$
Equality in the arithmetic--geometric mean inequality forces
$
   \widehat\eta_b(I)=I.
$
Since $\widehat\eta_b$ is self-adjoint, it is therefore doubly stochastic.

Writing
$
   C:=b^{-1},
$
the identity
$
   \widehat\eta_b(I)=I
$
is exactly
$
   C^{1/2}\eta(C)C^{1/2}=I,
$
or
$
   \eta(C)=C^{-1}.
$
Thus $\eta$ is symmetrically doubly-stochastic scalable.

The converse was proved above: every symmetric doubly-stochastic scaler
produces an optimizer.

<!-- remark-7: body_md -->
\label{rem:capacity-attainment}
The theorem separates two qualitatively different centered regimes.

If $\eta$ is symmetrically doubly-stochastic scalable, the capacity
minimum is realized by a finite positive scale.

If $\operatorname{Cap}(\eta)>0$ but $\eta$ is not symmetrically
doubly-stochastic scalable, then
\[
   \inf_{b\in\mathcal{F}}\Delta(b)>0
\]
but the infimum is not attained.  Any minimizing sequence must therefore
leave every compact subset of the positive definite cone; otherwise a
convergent subsequence would produce an optimizer.

Thus non-attainment of the centered variational problem is precisely the
failure of exact symmetric doubly-stochastic scaling.

\end{equation}

$

$
   0\leq\theta(x)\leq 1
 $
and, after decreasing $r$ if necessary,

   \|G(x+\zeta)\|
   \leq
   C|\zeta|^{-\theta(x)}
\end{equation}

By entrywise algebraicity, the classical Newton--Puiseux theorem \cite{Walker1950} gives a
convergent Puiseux expansion for every entry $G_{pq}$ on the local branch
selected from $\mathbb C^+$.  Taking a common multiple of the finitely
many ramification indices gives \eqref{eq:local-puiseux}.

Factoring out the lowest nonzero power gives

   G(x+\zeta)
   =
   \zeta^{-\theta(x)}

      C_{k_0}
      +
      \sum_{k>k_0}
      C_k\zeta^{(k-k_0)/N}
   \right).
\]
The series in parentheses is uniformly bounded for sufficiently small
$\zeta$ in the upper half-disc.  This gives
\eqref{eq:local-G-bound}.

$
Comparison with the leading Puiseux behavior shows that

Its right-hand side remains bounded as $w\to x$, whereas

The matrix $P_x$ is positive semidefinite, so

\label{thm:nonatomic-local-regularity}
Let $x\in\mathbb R$ satisfy
$
   \mu_{S_a}(\{x\})=0.
$
Then there exist an open interval $I\ni x$, a number $p>1$, and a
nonnegative function
$
   \rho\in L^p(I)
$
such that
\begin{equation}\label{eq:local-ac}
   d\mu_{S_a}(t)\big|_I
   =
   \rho(t)\,dt.
\end{equation}

More precisely, if $\theta=\theta(x)$ is the exponent from
Lemma~\ref{lem:local-puiseux-control}, then $\theta<1$ and one may choose
$I$ so that
\begin{equation}\label{eq:rho-pointwise-bound}
   0\leq
   \rho(t)
   \leq
   C|t-x|^{-\theta}
   \qquad
   \text{for a.e. }t\in I.
\end{equation}
Consequently,
\[
   \rho\in L^p(I)
   \qquad
   \text{for every }
   1<p<\frac1\theta
\]
when $\theta>0$, while for $\theta=0$ one has
$
   \rho\in L^\infty(I).
$

<!-- proof-32: body_md -->
By Lemma~\ref{lem:theta-one-atom}, non-atomicity implies
$
   \theta:=\theta(x)<1.
$
Choose $I\ni x$ sufficiently small that the estimate
\[
   \|G(t+i\varepsilon)\|
   \leq
   C
   \bigl(
      (t-x)^2+\varepsilon^2
   \bigr)^{-\theta/2}
\]
holds for $t\in I$ and all sufficiently small $\varepsilon>0$.

Define the Poisson regularization of the spectral measure by
\[
   \rho_\varepsilon(t)
   :=
   -\frac1\pi
   \operatorname{Im}g(t+i\varepsilon).
\]
Since $g$ is the scalar Cauchy transform of $\mu_{S_a}$,
\begin{equation}\label{eq:poisson-density}
   \rho_\varepsilon(t)
   =
   \int_{\mathbb R}
   P_\varepsilon(t-s)\,d\mu_{S_a}(s),
\end{equation}
where
\[
   P_\varepsilon(s)
   :=
   \frac1\pi
   \frac{\varepsilon}{s^2+\varepsilon^2}.
\]
Moreover,
\[
   0\leq
   \rho_\varepsilon(t)
   \leq
   \frac1\pi\|G(t+i\varepsilon)\|.
\]
Hence
\begin{equation}\label{eq:poisson-domination}
   0\leq
   \rho_\varepsilon(t)
   \leq
   C'
   \bigl(
      (t-x)^2+\varepsilon^2
   \bigr)^{-\theta/2}
   \leq
   C'|t-x|^{-\theta}
\end{equation}
for $t\neq x$.

Let $\varphi\in C_c(I)$ be nonnegative.  From
\eqref{eq:poisson-density} and Fubini's theorem,
\[
\begin{aligned}
   \int_I
   \varphi(t)\rho_\varepsilon(t)\,dt
   &=
   \int_{\mathbb R}
   (P_\varepsilon*\varphi)(s)\,
   d\mu_{S_a}(s).
\end{aligned}
\]
Since
$
   P_\varepsilon*\varphi
   \longrightarrow
   \varphi
$
uniformly as $\varepsilon\downarrow0$, we obtain
\[
   \int_I\varphi(t)\,d\mu_{S_a}(t)
   =
   \lim_{\varepsilon\downarrow0}
   \int_I
   \varphi(t)\rho_\varepsilon(t)\,dt.
\]
Using \eqref{eq:poisson-domination},
\[
   \int_I\varphi(t)\,d\mu_{S_a}(t)
   \leq
   C'
   \int_I
   \varphi(t)|t-x|^{-\theta}\,dt.
\]
Therefore, as measures on $I$,
\begin{equation}\label{eq:measure-domination}
   \mu_{S_a}|_I
   \leq
   C'|t-x|^{-\theta}\,dt.
\end{equation}
It follows that
\[
   \mu_{S_a}|_I=\rho(t)\,dt
\]
for some measurable $\rho\geq0$ satisfying
\eqref{eq:rho-pointwise-bound}.

If $\theta=0$, this immediately gives
$
   \rho\in L^\infty(I).
$
If $0<\theta<1$, then
\[
   |t-x|^{-p\theta}\in L^1_{\mathrm{loc}}
\]
whenever
\[
   p\theta<1.
\]
Thus
\[
   \rho\in L^p(I)
   \qquad
   \text{for every }
   1<p<\frac1\theta.
\]
Since $\theta<1$, such a $p>1$ always exists.
