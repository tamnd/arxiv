<!-- theorem--1: body_md -->
Let $1\leq n\leq 6$, and let the initial datum $E_0 \subset \mathbb R^{n+1}$ be any bounded set of finite perimeter with $\mathcal{L}^{n+1}(\partial E_0)=0$. Then, for every flat flow $E_t$ obtained from the
minimizing-movements scheme and varifolds  $V_t$ obtained as above, the couple $(\{E_t\},\{V_t\})$ is a generalized BV flow. Specifically,  $\{V_t\}_{t\geq0}$ is an integral Brakke flow with
\begin{align}
|D\chi_{E_0}| = \|V_0\| &= \lim_{t \to 0^+} \|V_t\|\,, \label{intro:initial} \\
 |D\chi_{E_t}| &\le \|V_t\| \quad\qquad\text{for a.e. }t>0\,, \label{intro:domination}\\
 \partial\llbracket E_t \rrbracket &= [V_t] \quad\quad\quad \,\text{in the sense of currents $\mathrm{mod}(2)$, for a.e. $t > 0$}\,, \label{intro:comp}
\end{align}      
and the sets $E_t$ satisfy the following evolution law:  for every $0\leq s<t$ and every smooth compactly supported
test function $\phi=\phi(x,\tau)$,
\begin{equation}\label{intro:gen-BV}
    \int_{E_t}\phi(x,t)\,dx
    -
    \int_{E_s}\phi(x,s)\,dx
    =
    \int_s^t\int_{E_\tau}
    \partial_\tau\phi(x,\tau)\,dx\,d\tau
    +
    \int_s^t\int_{\partial^*E_\tau}
    \phi(x,\tau)\,
    \mathbf{H}_{V_\tau}\cdot\nu_{E_\tau}\,
    d\mathcal{H}^n\,d\tau \,,
\end{equation}
where $\mathbf H_{V_t}$ is generalized mean curvature vector of the varifold $V_t$.

<!-- lemma-1: body_md -->
\label{lem:prel-est}
Let $E_0$ be a bounded set of finite perimeter. There exist dimensional constants $C>c>0, \gamma>1$ such that the following estimates
hold for $k\ge 1$:
\begin{align}
 P(E^h_{k})+\frac1hD(E^h_{k},E^h_{k-1})&\leq P(E^h_{k-1})\,,                                     
\nonumber\\
 \sup_{\partial E^h_{k}}|d_{E^h_{k-1}}|&\leq C\sqrt h\,,                              \nonumber\\
 \min\bigl\{|E^h_{k}\cap B_r(x)|\,,\ |B_r(x)\setminus E^h_{k}|\bigr\}
 &\geq cr^{n+1}\,,
 \qquad \, x\in\partial E^h_{k},\quad 0<r\leq\gamma\sqrt h\,,                   
 \label{density}\\
 cr^n\leq P(E^h_{k};B_r(x))&\leq Cr^n,
 \qquad \quad x\in\partial E^h_{k},\quad 0<r\leq\gamma\sqrt h\,,                   
 \label{area}\\
 |E^h_{k+1}\triangle E^h_{k}|
 &\leq C\left(
 \ell P(E^h_{k+1})+\frac1\ell D(E^h_{k+1},E^h_{k})
 \right),
 \qquad 0<\ell\leq\gamma\sqrt h\,,                         
\nonumber\\
 h\int_{\partial E^h_{k}}|H_{E^h_{k}}|^2\,d\mathcal H^n
 &\leq \frac ChD(E^h_{k},E^h_{k-1})\,.                                              
\nonumber
\end{align}
In particular, if $M:=P(E_0)$ and $Kh\leq T$, then
\begin{equation}\label{sums}
 \sup_{0\leq k\leq K}P(E^h_{k})\leq M\,,
 \quad
 \sum_{k=1}^KD(E^h_{k},E^h_{k-1})\leq hM\,,
 \quad
 \sum_{k=1}^Kh
 \int_{\partial E^h_{k}}|H_{E^h_{k}}|^2\,d\mathcal H^n
 \leq CM\,,
\end{equation}
and it holds 
\begin{equation}\label{volsum}
 \sum_{k=1}^K|E^h_k\triangle E^h_{k-1}|
 = |E_1^h \triangle E_0| + \sum_{k=2}^K|E^h_k\triangle E^h_{k-1}| \leq |E_1^h \triangle E_0|+ C(T+1)M\,.
\end{equation}
Moreover, the comparison principle holds and, in particular,  if $E_0\subseteq B_R$ for $R>0$ then $E^h_k\subseteq B_R$ for every $k\ge 1$.

\label{rmk:c0}

    \[
    |E^h_1\triangle E_0|\to 0,\quad \text{as }h\to0\,.

        &\le |(E^h_1\triangle E_0)\cap\{|d_{E_0}|\le r\}| + \frac 1r \int_{(E^h_1\triangle E_0)\cap\{|d_{E_0}|>r\} } |d_{E_0}|\\

<!-- proposition-1: body_md -->
\label{prop:AM}
    Let $F_j$ be $(\Lambda_j, R_j)$-minimizers in expanding domains $U_j \uparrow \mathbb R^{n+1}$ with
    \begin{equation} \label{hp:AM}
        \Lambda_j \to 0\,, \qquad R_j \to \infty \,,
    \end{equation}
    and assume uniform local perimeter bounds and $0 \in \partial^*F_j$. Then, up to a subsequence
    \begin{equation} \label{th:AM}
        \chi_{F_j} \to \chi_{F_\infty} \quad \mbox{in $L^1_{loc}$}\,,
    \end{equation}
    where $F_\infty$ is locally perimeter minimizing in $\mathbb R^{n+1}$. Furthermore, the perimeter measures converge locally, and, upon choosing representatives of $F_j$ and $F_\infty$ so that $\partial F_j=\operatorname{spt}|D\chi_{F_j}|$ and $\partial F_\infty=\operatorname{spt}|D\chi_{F_\infty}|$, the boundaries $\partial F_j$ converge locally in Hausdorff distance to $\partial F_\infty$, and $0 \in \partial F_\infty$. Finally, if $n \leq 6$, the boundaries $\partial F_j$ converge locally as $C^{1,\alpha}$ graphs to $\partial F_\infty$ near every point of $\partial F_\infty$.

<!-- proposition-2: body_md -->
\label{prop:curvature}
For $1\le n\le6$, every  iterate $E^h_k$ for $k\ge 1$ of \eqref{scheme} satisfies
\begin{equation*}
 \|A_{E^h_k}\|_{L^\infty(\partial E^h_k)}\le C_n/ \sqrt h\,.

<!-- proof-1: body_md -->
We drop the superscript $h$ for simplicity. Suppose the estimate fails. Since the boundaries are (uniformly) compact and $C^2$, choose
\[
 \lambda_j=\max_{\partial E_j}|A_{E_j}|=|A_{E_j}|(p_j)\,,
\qquad  \lambda_j\sqrt{h_j}\longrightarrow\infty\,.
\]
Define $\widetilde E_j=\lambda_j(E_j-p_j)$. Then
\begin{equation}\label{normalize}
 |A_{\widetilde E_j}|\le1,\qquad |A_{\widetilde E_j}|(0)=1\,.
\end{equation}
The rescaled sets $\widetilde E_j$ are $\left(C (\lambda_j \sqrt{h_j})^{-1}, \gamma\lambda_j\sqrt{h_j} \right)$-minimizers of the perimeter in $\mathbb R^{n+1}$, and $0 \in \partial \widetilde E_j$. Moreover, rescaling \eqref{area} yields that for every fixed $R$
\[
P(\widetilde E_j;B_R)\le CR^n \qquad \mbox{for all sufficiently large $j$}\,.
\]
We can then apply Proposition \ref{prop:AM}, and conclude the existence of a locally perimeter minimizing set $E_\infty$ with $0 \in \partial E_\infty$ and Euclidean perimeter  growth $P(E_\infty;B_R)\le C R^n$. Its boundary is smooth, with local  $C^{1,\alpha}$ graphical convergence of $\partial \widetilde E_j$ to $\partial E_\infty$. 



Next,  rescale \eqref{EL} to see that the  sets $\widetilde E_j$ satisfy 
\[
 H_{\widetilde E_j}(y)=-\frac{d_{F_j}(p_j+y/\lambda_j)}{h_j\lambda_j},
 \quad
 |H_{\widetilde E_j}(0)|\le\frac{C}{\lambda_j\sqrt{h_j}},\quad
 \operatorname{Lip}(H_{\widetilde E_j})\le\frac1{h_j\lambda_j^2}\,,
\]
where $F_j$ is the predecessor of $E_j$, and where we have also used Lemma \ref{lem:prel-est} and the fact that $d_F$ is $1$-Lipschitz. Thus $H_{\widetilde E_j}\to0$ locally uniformly, with vanishing Lipschitz seminorm. The $C^{1,\alpha}$ convergence of the boundaries implies local graphicality over a common plane $\pi_0$, and assume without loss of generality that $\pi_0=\mathbb R^n\times\{0\}$. If the coordinates are labeled $y=(y',y_{n+1})$, we can then let $y_{n+1}=v_j(y')$ denote the function locally representing $\partial \widetilde E_j$, and therefore satisfying
\[
-\operatorname{div}\left(\frac{\nabla v_j}{\sqrt{1+|\nabla v_j|^2}} \right)(y') = H_{\widetilde E_j}(y',v_j(y'))
\]
The $C^{1,\alpha}$ bounds for the graphs and the uniform Lipschitz bounds for $H_{\widetilde E_j}$ give, by Schauder estimates, uniform local $C^{2,\alpha}$ estimates, which in turn imply that $v_j$ converge to a limit $v_\infty$ representing $\partial E_\infty$ in $C^{2,\beta}$ for all $0 < \beta < \alpha$. Hence, we deduce that
\begin{equation}\label{limitnorm}
 H_{ E_\infty}=0\,,\qquad |A_{ E_\infty}|(0)=1\,.
\end{equation}
Since $E_\infty$ is locally perimeter minimizing, $\partial E_\infty$ is stable. The component $M_\infty$ through $0$ of $\partial E_\infty$ is then two-sided and stable; it is also complete and properly embedded, since it is a
connected component of the closed smooth hypersurface
$\partial E_\infty$, and it has Euclidean area growth. The stable Bernstein theorem (see, for instance, \cite{Bel}) implies then that $M_\infty$ is a hyperplane when $2\le n\le6$.  This contradicts \eqref{limitnorm}. The case $n=1$ follows straightforwardly from the Euler-Lagrange equation.

<!-- lemma-4: body_md -->
\label{lem:reach}
Let $1\le n\le6$. Assume that $\partial E$ is a compact $C^2$ boundary such that there exist a radius $r_0 > 0$ and positive numbers $K,L$ so that for every $x\in\partial E$ the following holds in $B_{r_0}(x)$:
\begin{enumerate}
\item $|A_E|\le K/r_0$;
\item \eqref{density}--\eqref{area} hold for all $r\le r_0$, with fixed constants;
\item  the set $E$ is a $(L/r_0,  r_0)$-minimizer of the perimeter in $B_{r_0}(x)$.
\end{enumerate}
Then, there exists $c>0$ depending only on the constants above  such that
\[
 \operatorname{reach}(\partial E)\ge c r_0\,.
\]
<!-- end lemma-4 -->
<!-- proof-2: body_md -->
Assume the contradiction: there exist $E_j$ satisfying the assumptions with $r_0=r_{0,j}$ and  $\rho_j=\operatorname{reach}(\partial E_j)$ with $\rho_j/r_{0,j}\to0$ as $j\to +\infty$. By (1), the focal radius is at least $r_{0,j}/K$, so that, since $\rho_j = o(r_{0,j})$, for all sufficiently large $j$ $\rho_j$ is smaller than the focal radius. By definition of reach, then, there exists a point $m_j$ with $\text{dist}(m_j,\partial E_j)=\rho_j$ and  with two distinct nearest points $p_j,q_j\in\partial E_j$. Recenter and rescale   defining $\widetilde E_j=(E_j-p_j)/\rho_j$ and $\tilde m_j,\tilde p_j,\tilde q_j$ accordingly. For such sets $\operatorname{reach}(\partial \widetilde E_j)=1$, and the assumptions are satisfied with $r_0$ replaced by $r_{0,j}/\rho_j \to \infty$. In particular, the curvature bound becomes $|A_{\widetilde E_j}|\le K\rho_j/r_{0,j}\to 0$,  and 
\[
 \tilde p_j=0,\qquad |\tilde m_j|=1,\qquad |\tilde q_j-\tilde m_j|=1,\qquad |\tilde q_j|\le2.
\]
By assumption $(3)$, the sets are uniformly $(L\rho_j/r_{0,j},  r_{0,j}/\rho_j)$-minimizers, and thus Proposition~\ref{prop:AM}
produces, up to extraction of a (not relabeled) subsequence,  a complete smooth locally minimizing boundary $\partial E_\infty$ with local  graphical convergence in $C^{1,\alpha}$. The vanishing curvature bound implies that each limiting component is a hyperplane. 

Note that there can be only one hyperplane. Indeed, local perimeter finiteness makes any family of parallel components of $\partial E_\infty$ locally finite. Intersecting planes are also excluded by the regularity of $E_\infty$.   Consider two adjacent planes  bounding a slab occupied by $E_\infty$. Consider a cylinder of radius $R$ and height $L$, with base  parallel to the planes. It can be chosen to intersect only one slab. 
Removing the cylinder from $E_\infty$  decreases the perimeter by  $  cR^n$ 
and adds only the lateral perimeter $cLR^{n-1}$. For $R$ large enough this contradicts  local minimality. Hence the limit is a single plane. 


Note that, in particular, the graphical convergence holds in every neighborhood of $0$. Therefore, for $j$ large enough, both nearest points $\tilde p_j, \tilde q_j$ lie  on one graph over this plane in a fixed cylinder. Assuming for simplicity that the limiting plane is $\{x_{n+1}=0\}$, we can parametrize the rescaled sets $\widetilde E_j$ in a fixed neighborhood $B_3\subseteq \{x_{n+1}=0\}$ as graphs $v_j: B_3\to \mathbb R$. By the convergence in $C^{1,\alpha}$ we get $\|v_j\|_\infty+\|\nabla v_j\|_\infty \to 0$, and in fact also 
\[
\|\nabla^2 v_j\|_\infty \leq C (1 + \|\nabla v_j\|_\infty^2)^{3/2} \|A_{\widetilde E_j}\|_\infty \to 0\,.
\]
As a consequence, on $B_3$ the squared distance from $\partial \widetilde E_j$ to $\tilde m_j$ has positive definite Hessian: writing $x=(x',x_{n+1})$ it holds 
\[
 D^2\bigl|(x',v_j(x'))-\tilde m_j\bigr|^2
 =2I+2\nabla v_j\otimes \nabla v_j+2(v_j-\tilde m_{j,n+1})\nabla^2v_j>0\,,
\]
for all $j$ sufficiently large, where we used $|\tilde m_j|\le1$.
Therefore, such function  cannot have two distinct interior critical points. Recalling that $\tilde p_j=0$ and $|\tilde q_j|\le 2$,  there are two nearest points to $\tilde m_j$ in $B_3$. Nearest points are critical for the squared distance function, a contradiction.

\label{lem:ROF}

\label{lem:allmin}
Let $u,z$ be as above.  If $E$ is any minimizer of \eqref{scheme} with $F$ replacing $E^h_{k-1}$, then

<!-- proof-4: body_md -->
For a bounded set of finite perimeter  $A\subset \mathbb T^{n+1}$, note that  
\[
 P(A)-\int_A\operatorname{div} z
 =\int_{\partial^*A}(1-[z,\nu_A])d\mathcal H^n\ge0\,,
\]
where the the first identity follows from \eqref{eq:pairing-gauss-green} and  last inequality follows from \eqref{eq:nor-tr-est}.
By the coarea formula \eqref{eq:pairing-coarea}
it holds 
\[
(z,Du)=\int_\mathbb R (z,D\chi_{\{u>\lambda\}})d \lambda, \qquad |Du|=\int_\mathbb R|D\chi_{\{u>\lambda\}}|d \lambda\,,
\]
in the sense of measures. Since in general $|(z,D\chi_A)|\le |D\chi_A|$ as $|z|\le 1$, by the pairing \eqref{zconditions}
we get 
\[
0=|Du|-(z,Du)=\int_\mathbb R \Big( |D\chi_{\{u>\lambda\}}|- (z,D\chi_{\{u>\lambda\}}) \Big) d\lambda \ge 0,
\]
and thus the integrand is zero for a.e. $\lambda$. Since $|\{u=\lambda\}|=0 $ for a.e. $\lambda$, $D \chi_{\{u<\lambda\}} = - D \chi_{\{u>\lambda\}}$ and therefore 
\[
|D\chi_{\{u<\lambda\}}|=- (z,D\chi_{\{u<\lambda\}})\quad \text{for a.e. }\lambda.
\]
Thus, by Gauss-Green \eqref{eq:pairing-gauss-green} we deduce for a.e. $\lambda$
\[
\int_{\{ u<\lambda \}} \operatorname{div} z=-(z,D\chi_{\{u<\lambda \}}) (\mathbb T^{n+1})=|D\chi_{\{u<\lambda\}}|(\mathbb T^{n+1})=P(\{u<\lambda \})\,.
\]
Choose levels $\lambda_j\nearrow0$ satisfying the previous. The characteristic functions $\chi_{\{u<\lambda_j\}}$ converge in $L^1(\mathbb T^{n+1})$ to that of $E_-:=\{u<0\}$. Since $\operatorname{div} z\in L^\infty(\mathbb T^{n+1})$, lower semicontinuity gives
\[
 P(E_-)\le\liminf_jP(\{u<\lambda_j\})
 =\int_{E_-}\operatorname{div} z\le P(E_-).
\]
Thus $P(E_-)=\int_{E_-}\operatorname{div} z$. For $\mathcal J_F(G)=P(G)+h^{-1}\int_G d_F$, using the previous identity one gets

 \mathcal J_F(G)-\mathcal J_F(E_-)
 =P(G)-\int_G \operatorname{div} z+\int_{G\cap\{u> 0\}}\tfrac uh
                  +\int_{G^c\cap\{u<0\}}(-\tfrac uh).

The term $P(G)-\int_G \operatorname{div} z$ and the other two terms on the right are nonnegative. Consider another minimizer $E$ of \eqref{scheme}. Minimality of $E$ makes the left hand side of \eqref{defect}, with $G=E$, non positive. Hence  every term vanishes. The last two give \eqref{sign}; the first gives \eqref{trace}.

<!-- lemma-9: body_md -->
\label{lem:diss-bd}
For $h$ sufficiently small, depending on $b$, it holds 
\begin{equation}\label{endpointineq}
 -\frac{D_\varphi(E,F)}h
 \le h\delta P(E)[\varphi b]

Since $d_F=0$ on $\partial F$, Gauss--Green and the Euler-Lagrange equation \eqref{EL} give
\begin{equation}\label{gaussd}
 -h\delta P(E)[\varphi b] = \int_{\partial^* E} d_F \varphi \nu\cdot b - \int_{\partial^* F} d_F \varphi \nu\cdot b 
 =\int(\chi_E-\chi_F)
 \left(\varphi\nabla d_F\cdot b+d_F\operatorname{div}(\varphi b)\right)\,.
\end{equation}
Write $\sigma=\operatorname{sgn}(d_F)$, so $\chi_E-\chi_F=\sigma|\chi_E-\chi_F|$ a.e.. Bounding  $\operatorname{div} (\varphi b)\le C$ and $|\chi_E-\chi_F|\le 1$ we get 
\begin{equation}\label{signedbound}
 -h\delta P(E)[\varphi b]- \frac{D_\varphi(E,F)}h
 \le
 \int\varphi\left(\sigma\nabla d_F\cdot b-|d_F|/h\right)_+
       +C D(E,F)\,.
\end{equation}
The positive part lies in $\{|d_F|\le h\|b\|_\infty\}$. Since  $h\|b\|_\infty \ll\sqrt h$, for  $h$ small enough we can use normal coordinates $\Psi$ from Lemma \ref{lem:tubecalc} and \eqref{hessd} to write

= 
\int_{\partial F}\int_0^{h\|b\|_\infty} \varphi\circ \Psi \left( ( \nabla d_F\cdot b)\circ\Psi - \frac sh \right)_+ J_\Psi
\\
 +\int_{\partial F}\int^0_{-h\|b\|_\infty} \varphi\circ \Psi \left( -(\nabla d_F\cdot b)\circ\Psi + \frac sh \right)_+ J_\Psi \,.
\end{multline*}
Using Lemma \ref{lem:tubecalc}, we see that freezing the argument of    $b,\varphi$ contribute $O(h^2P(F))$, whereas the Jacobian \eqref{jac} contributes $O(h^{3/2}P(F))$.

\begin{multline*}

 + \int_0^\infty\bigl(-b(p)\cdot\nu_F(p)-s/h\bigr)_+d s= \frac h2 |b(p)\cdot \nu_F(p)|^2,

we conclude  \eqref{endpointineq}.

\label{thm:brakke}
Assume $1\le n\le6$. Let  $E_0\subset\mathbb R^{n+1}$ be a bounded set 
with finite perimeter with $|\partial E_0|=0$, and let

  V_{E_t^{h_j}}dt
 &\rightharpoonup V_tdt\quad \text{as varifolds in space-time}\,, \\
 \|V_{E_t^{h_j}}\| &\rightharpoonup \|V_t\|  \quad \text{for every $t \geq 0$, as measures in space}\,,\\
 \{V_t\}_{t \geq 0} &\mbox{ is a Brakke flow with } V_0=V_{E_0}\,.
 \end{split}

    \end{split}
\end{equation}

Space-time convergence of $\|V^h_t\|dt=\mu^h_tdt\rightharpoonup \|V_t\|dt $  and dominated convergence shows that 
\begin{equation}\label{eq:conv-weight2}
     \|V_t\|=\mu_t
 \quad\text{for a.e. }t\,.
\end{equation}
Indeed, for every $f\in C^0_c((0,T)),\varphi\in C^0_c(\mathbb R^{n+1})$ it holds 
\[
\int f(t) \varphi(x) d\|V_t\|(x)dt = \lim_{h\to 0}
\int f \varphi\, d\|V^h_t\|dt = \lim_{h\to 0}
\int f\int \varphi \,d\mu^h_tdt =
\int f\int \varphi \,d\mu_t dt   \,.
\]
We need to show that the varifolds $V_t$ are integral with curvatures  in $L^2$, for a.e. $t\ge 0$.
Set
\[
 e_h(t):=\int|\mathbf H_t^{h}|^2\,d\mu_t^h,\qquad t\ge 0\,.
\]
By \eqref{sums}, $e_h\in L^1_{loc}([0,+\infty))$ and $\int_0^T e_h(t)\,dt\le CM$. 
For every $b\in C^1_c(\mathbb R^{n+1}\times[0,T);\mathbb R^{n+1})$  it holds 
\begin{equation}\label{eq:fvar-curv}
     \int_0^T\delta V_t^h(b)\,dt
 =
 -\int_0^T\int\mathbf H_t^{h}\cdot b\,d\mu_t^h\,dt + \int_0^h\delta V_t^h(b)\,dt\,,
\end{equation}
where we recall that $\mathbf H^h_\tau:=0$  for $\tau\in [0,h)$. The last integral in \eqref{eq:fvar-curv}  vanishes as $h\to0$. Indeed, by the first variation formula, for every $\tau\in [0,h)$ it holds $ \delta V_\tau ^h(b(\cdot,\tau))=\int_{\partial^* E_0} \operatorname{div}_\tau b(\cdot,\tau)$. This term is uniformly bounded, thus integrating and sending $h\to 0$ yields the previous claim.

Since $  V^h_tdt \rightharpoonup   V_tdt $ we can pass to the limit on the \textit{lhs} of \eqref{eq:fvar-curv}. Applying Cauchy--Schwarz
on the right gives
\[
 \left|\int_0^T\delta V_t(b)\,dt\right|
 \le
 \left(\liminf_{h\to0}\int_0^T e_h(t)\,dt\right)^{1/2}
 \left(\int_0^T\int|b|^2\,d\mu_t\,dt\right)^{1/2}\,.
\]
Riesz representation  and disintegration therefore give $\mathbf H\in L^2(d\mu_t dt)$ such that 
\[
 \delta V_t=-\mathbf H_t\mu_t
 \quad\text{for a.e. }t,
 \qquad
 \int_0^T\int|\mathbf H_t|^2\,d\mu_t\,dt\le CM\,.
\]
Therefore, for a.e.  $t$, the varifold  $V_t$ has locally bounded
first variation. 

To  show that $V_t$ is integral for a.e. $t$, we first note that by Fatou's lemma $\liminf_{h\to0} e_h(t)<\infty$ for
a.e. $t$. For each such $t$, choose a subsequence (depending on $t$)
with bounded $e_h(t)$. Since 
\begin{equation}\label{eq:hp-Allard}
    \|\delta V_t^h\|(\mathbb R^{n+1})\le  M^{1/2}e_h(t)^{1/2}\,,
\end{equation}
which is bounded along the subsequence,  
Allard's integral compactness theorem \cite{Allard} provides a further subsequence and an integral varifolds  $W_t$ with $V^h_t\rightharpoonup W_t$. By the convergence \eqref{eq:conv-weight2} it holds $\|W_t\|=\|V_t\|$ for a.e. $t$.  Since  the varifold $V_t$ has bounded first variation  and its weight $\|W_t\|$ has positive
integer density almost everywhere, Allard's rectifiability
theorem implies that $V_t$ is rectifiable.
Since a rectifiable varifold is determined by its weight,
\begin{equation}\label{eq:V=W}
 V_t=W_t
 \quad\text{for a.e.  }t\,.   
\end{equation}
Thus the spacetime limit itself is integral for a.e. $t$  and has no singular
first variation. Its mean curvature $\mathbf H_t$ is normal
to its approximate tangent plane by Brakke's perpendicularity theorem \cite[Theorem 5.8]{Bra}.

\smallskip

\noindent\textbf{Step 2:} \textit{$\mathrm{mod}(2)$ compatibility}.
Consider the full measure set of times  $t$ as in Step 1 such that \eqref{eq:hp-Allard} holds. Let $T_t^h := \llbracket E_t^h \rrbracket$, so that, as observed above, $\|V_t^h\| = \| \partial T_t^h \|$. Notice also that $[V_t^h] = \partial T_t^h \quad \mathrm{mod}(2)$, namely $V_t^h$ and $\partial T_t^h$ are compatible, in the language of \cite{Whi09}. Let also $T_t := \llbracket E_t \rrbracket$, and let $\mathcal F(T_1,T_2)$ denote the integral flat distance between flat chains $T_1, T_2$. Since
\[
\mathcal F(T_t^h, T_t) \leq |E_t^h \triangle E_t|\,,
\]
as a consequence of \eqref{flat-convergence} we have that along a suitable subsequence $h_j$ (depending on $t$) we have, as $j \to \infty$ that both hold:
\[
\begin{split}
   & V_t^{h_j} \mbox{ satisfy the assumptions of Allard's compactness, and } V_t^{h_j} \rightharpoonup V_t\,, \\
    & \mathcal F(\partial T_t^{h_j}, \partial T_t) \to 0 \,.
\end{split}
\]
By \cite[Theorem 1.2]{Whi09}, $V_t$ and $\partial T_t$ are compatible, that is \eqref{mod2-equiv} holds. 

\smallskip

\noindent\textbf{Step 3:} \textit{Brakke's inequality}.
Fix tests in space-time $  \varphi\in C_c^2(\mathbb R^{n+1}\times[0,\infty);\mathbb R_+)$ and  $b\in C_c^2(\mathbb R^{n+1}\times[0,\infty);
          \mathbb R^{n+1})$. 
Let $0<s<t$ and take  $h$ sufficiently small so that  $2\le [s/h]<[t/h]$.  
The convergence established in Step~1 and 
  the 
uniform mass bound imply 
\[
\mu_{s}^h(\varphi_{[s/h]h})\to \mu_s(\varphi_s), \quad \mu_{t}^h(\varphi_{[t/h]h})\to \mu_t(\varphi_t)\,.
\]
Sum \eqref{dualineq} for $k=[s/h]+1,\ldots, [t/h] $, using
the spatial tests $\varphi_{kh}$ and $b_{kh}$, and add the
temporal increments
\[
P_{\varphi_{kh}}(E^h_{k-1})
-
P_{\varphi_{(k-1)h}}(E^h_{k-1})\,.
\]
The left-hand side is a telescopic sum, equal to 
\[
\mu_{[t/h]h}^h(\varphi_{[t/h]h})
-
\mu_{[s/h]h}^h(\varphi_{[s/h]h})=\mu_{t}^h(\varphi_{[t/h]h})
-
\mu_{s}^h(\varphi_{[s/h]h})\,,
\]
while the temporal increments satisfy 
\[ 
\sum_{k=[s/h]+1}^{[t/h]}
\left[
P_{\varphi_{kh}}(E^h_{k-1})
-
P_{\varphi_{(k-1)h}}(E^h_{k-1})
\right]
=
\int_{[s/h]h}^{[t/h]h}
\int \partial_\tau\varphi\,d\mu_\tau^h\,d\tau
\underset{h\to 0}{\longrightarrow}
\int_s^t
\int \partial_\tau\varphi\,d\mu_\tau\,d\tau\,,
\]
where $\partial_\tau$ denotes derivative in the time variable. 
For fixed smooth tests, the  time integrals of the expressions in  the \textit{rhs} of 
\eqref{dualineq} is continuous with respect to  varifold convergence in space-time. Note also that the   uniform mass bound implies  
\[
\lim_{h\to 0} \int_s^t \delta V^h_\tau(\varphi_\tau b_\tau) \,d\tau=\lim_{h\to 0}  \int_s^t \delta V^h_{\tau+h}(\varphi_\tau b_\tau)\,d\tau  =  \int_s^t\delta V_\tau(\varphi_\tau b_\tau)\,d\tau\,.
\]
Using again the   uniform mass bound and the error estimate \eqref{rho}, we can pass to the limit for every $0<s<t$
\begin{equation}\label{eq:dual-limit}
\mu_t(\varphi_t)-\mu_s(\varphi_s)
\le \int_s^t \Bigg[
\int \partial_\tau\varphi\,d\mu_\tau
+2\delta V_\tau(\varphi_\tau b_\tau)
-\delta V_\tau(\nabla\varphi_\tau)
+\int \varphi_\tau
  |P_{T_xV_\tau}^{\perp}b_\tau|^2\,d\mu_\tau
\Bigg]\,d\tau\,.
\end{equation}
The equation \eqref{eq:dual-limit} also holds for $s=0$. Indeed, apply \eqref{local} at $k=1$ with $G=E_0$ and spatial test $\varphi_h$ to get 
\[
\mu_h^h(\varphi_h)-\mu_0(\varphi_0)
\le
\|\nabla\varphi_h\|_\infty
|E_h^h\triangle E_0| + M \|\varphi_h-\varphi_0\|_\infty\,.
\]
On the other hand, $|E_h^h\triangle E_0| = |E^h_1 \triangle E_0|\to0$ as $h\to 0$ as a consequence of the assumption $|\partial E_0|=0$ by the argument in Remark \ref{rmk:c0}, and, of course, $\|\varphi_h-\varphi_0\|_\infty \to 0$: hence, this time increment vanishes and the discussion above can be repeated to include $s=0$. 

By Step~1, $\delta V_\tau=-\mathbf H_\tau\mu_\tau$ and
$\mathbf H_\tau$ is normal to $T_xV_\tau$ for almost every
$\tau$. Hence the right-hand side of \eqref{eq:dual-limit}
equals
\[
\int_s^t\int
\left(
\partial_\tau\varphi
+\nabla\varphi\cdot\mathbf H_\tau
-\varphi|\mathbf H_\tau|^2
+\varphi
 |P_{T_xV_\tau}^{\perp}b_\tau-\mathbf H_\tau|^2
\right)\,d\mu_\tau\,d\tau\,.
\]
Smooth compactly supported vector fields are dense in $L^2(\varphi\mu_\tau d\tau;\mathbb R^{n+1})\ni \mathbf{H}$. Hence, taking the infimum over all $b$ of the previous display proves, for every $0 \leq s < t$
\[
\mu_t(\varphi_t)-\mu_s(\varphi_s)
\le \int_s^t\int
\left(
\partial_\tau\varphi
+\nabla\varphi\cdot\mathbf H_\tau
-\varphi|\mathbf H_\tau|^2
\right)\,d\mu_\tau\,d\tau\,.
\]
By Step 1 then $\|V_t\|=\mu_t$ for a.e. $t\ge 0$. 
Defining $V_t$ on the remaining exceptional times so that
$\|V_t\|=\mu_t$ (for instance, setting $V_0=V_{E_0}$ and $V_t := \mu_t \otimes \delta_{S}$ for an arbitrary $n$-plane $S$ at any exceptional bad time $t > 0$) completes the proof of Brakke's inequality for nonnegative tests $\varphi$ of class $C^2$. For a nonnegative $C_c^1$ test, choose nonnegative smooth approximations converging in $C^1$, with supports in a common compact set. The mass bound, the $L^2(d\mu_t\,dt)$ bound for
$\mathbf H$, and Cauchy--Schwarz allow passage to the limit in every term, so that the asserted $C_c^1$ formulation follows. 

\smallskip

\noindent \textbf{Step 4}: \textit{Dissipation, domination, and continuity at the initial time}

The dissipation inequality \eqref{eq:dissipation} is now an immediate consequence of Brakke's inequality after testing with $\varphi \equiv 1$ (and a suitable cut-off outside of the large ball containing the evolution). The initial datum satisfies 
$\mu_0=|D\chi_{E_0}|$, so that $\|V_0\|(\mathbb R^{n+1}) = \mu_0(\mathbb R^{n+1})=P(E_0)$.

For every $t\ge0$, the convergence
$E_t^h\to E_t$ in $L^1$, lower semicontinuity of perimeter,
and $\mu_t^h\rightharpoonup\mu_t$ imply
\[
|D\chi_{E_t}|(\psi)
\le
\liminf_{h\to0}|D\chi_{E_t^h}|(\psi)
=
\mu_t(\psi)
\qquad
\text{for every }0\le\psi\in C_c^0(\mathbb R^{n+1})\,.
\]
Thus $|D\chi_{E_t}|\le\mu_t$ for every $t\ge0$, thus proving domination.

As for continuity at $t=0$, by $L^1$ continuity of $E_t$, lower semicontinuity of perimeter, domination, and the global mass bound,
\[
    P(E_0)\le\liminf_{t\downarrow0}P(E_t)
     \le\liminf_{t\downarrow0}\|V_t\|(\mathbb R^{n+1})\le P(E_0)\,.
\]
Since both $P(E_t)$ and $\|V_t\|(\mathbb R^{n+1})$ are bounded above by $P(E_0)$, they both converge to $P(E_0)$.
For any sequence $t_j\downarrow0$, the uniform confinement and mass bound
permit extraction of a weak limit $\|V_{t_j}\|\rightharpoonup\nu$.
For every nonnegative $\psi\in C_c^0$, again by lower semicontinuity of perimeter and domination
\[
    |D\chi_{E_0}|(\psi)
    \le\liminf_j|D\chi_{E_{t_j}}|(\psi)
    \le\nu(\psi).
\]
The mass bound forces $\nu=|D\chi_{E_0}|$, proving weak continuity. Finally, by domination the measure $\|V_t\|-|D\chi_{E_t}|$ is nonnegative, and thus its total variation is simply $\|V_t\|(\mathbb R^{n+1}) - P(E_t)$, which tends to zero. This completes the proof.

The proof  essentially follows from Theorem \ref{thm:brakke} and the arguments of \cite{StuTon}.
    Note first that $V_t$ is a Brakke flow, thus  by the proof of \cite[Thm 4.3]{StuTon} it is also an $L^2$-flow with velocity $v(\cdot,t)=\mathbf H_{t}$.  By the integrality of $V_t$, item \textit{i)} in Theorem \ref{thm:L2-to-BV} follows. 
   
    We then show item \textit{ii)}. Set $M=P(E_0)<+\infty$, and recall the notation $\mathcal E$ for the space-time tube of the flat flow. Note first that $D_x\chi_{\mathcal E}=D\chi_{E_t}\,dt$. Indeed, for every $b\in C_c^1(\mathbb R^{n+1}\times(0,T);\mathbb R^{n+1})$,  
    Fubini's theorem  and Gauss-Green (which can be applied since $E_t$ is of finite perimeter for every $t$) give
    \[
     \langle D_x\chi_{\mathcal E}, b\rangle
     =-\iint_{\mathcal E}\operatorname{div}_x b\,dx\,dt
     =\iint b(\cdot,t)\cdot dD\chi_{E_t}\,dt\,.
    \]
   Then, by \eqref{domin} it holds 
\[
|D_x \chi_{\mathcal E}|=|D\chi_{E_t}| dt =\|V_{E_t}\|dt \le \|V_t\|dt,
\]
thus we only need to show the absolute continuity of $D_t\chi_{\mathcal E}$ with respect to $\|V_t\|dt$. 
We follow the argument of
\cite[Prop~4.5 and Cor~4.6]{StuTon}.
The result can be shown by using the estimates  of
\cite[Proposition 4.0.1]{MSS} \footnote{We explicitly note that this result does not require the convergence of the perimeter energies: in \cite{MSS} the convergence of the perimeters is only needed to characterize the limiting velocity as the mean curvature of the evolving sets.}.
In order to make the proof self-contained, we present a different proof based on the localized estimates of Lemma \ref{lem:tubecalc}. 

Fix a test function $\varphi\in C^2_c(\mathbb R^{n+1} \times (0,T))$, and set $\varphi_{kh}(x) :=\varphi(x,kh)$. For $k \geq 2$, also denote $\sigma=\operatorname{sgn}(d_{E^h_{k-1}})$.
Then
\[ 
\begin{aligned} \int(\chi_{E^h_k}-\chi_{E^h_{k-1}})\varphi_{kh} -\frac{D({E^h_{k}},{E^h_{k-1}})}h &= \int|\chi_{E^h_{k}}-\chi_{E^h_{k-1}}|\left(\sigma \varphi_{kh}-\frac{|d_{E^h_{k-1}}|}{h}\right)\\ &\le \int\left(\sigma \varphi_{kh}-\frac{|d_{E^h_{k-1}}|}{h}\right)_+. 
\end{aligned} 
\]
The positive part is supported in $|d_{E^h_{k-1}}|\le h\|\varphi\|_\infty$. For sufficiently small $h$ (depending on $\|\varphi\|_\infty$), it is then supported in the $O(\sqrt{h})$-neighborhood of $E^h_{k-1}$ where we can use normal coordinates by Lemma \ref{lem:tubecalc}. The same computations from Lemma \ref{lem:diss-bd} yield
\begin{equation}\label{eq:incr-dh}
    \int(\chi_{E^h_{k}}-\chi_{E^h_{k-1}})\varphi_{kh} \le \frac{D({E^h_{k}},{E^h_{k-1}})}h +\frac h2\int_{\partial {E^h_{k-1}}}\varphi_{kh}^2\,d\mathcal H^n +C_{\varphi} h^{3/2}M\,,
\end{equation}
for all $k \geq 2$. For $k=1$, the left-hand side vanishes for $h$ sufficiently small, because $\varphi$ vanishes in a neighborhood of $t=0$. Now, sum the discrete increments \eqref{eq:incr-dh} with $2 \leq k \leq [T/h]$ and then pass to the limit $h\to 0$. By discrete integration by parts, and using that $\varphi$ vanishes both near $t=0$ and $t=T$, we have 
\[
\sum_k \int (\chi_{E_k^h}-\chi_{E^h_{k-1}}) \varphi_{kh} = - \int_0^T \int_{E^h_t} \partial_t \varphi\,dxdt \xrightarrow{h \to 0} -\int_0^T\int_{E_t} \partial_t \varphi\, dxdt\,.
\]
The quadratic term converges to $\frac12 \int \varphi^2\,d\mu_t\,dt$ by measure convergence and uniform continuity of the test, whereas the sum of the remainders tends to zero. By  the summability of the dissipations \eqref{sums} we then get  
\[ 
    D_t\chi_{ \mathcal E}(\varphi) \le M+\frac12\int \varphi^2\,d\mu_tdt. 
\]
Apply this inequality replacing $\varphi$ with $\lambda \varphi$ and $-\lambda \varphi$, for each
fixed $\lambda>0$. Optimizing \emph{after} the limit $h\to0$ yields
\[
    | D_t\chi_{\mathcal E}(\varphi)|
    \le\inf_{\lambda>0}
       \left(\frac M\lambda+\frac\lambda2\|\varphi\|_{L^2(\mu)}^2\right)
    =\sqrt{2M}\,\|\varphi\|_{L^2(\mu)}.
\]
Therefore,
\[
D_t\chi_{\mathcal E}=w\,d\mu_tdt, \qquad \int_0^T\int w^2\,d\mu_tdt\le CM.   \]
In particular,
\[
\begin{split}
    |D_{x,t}\chi_{\mathcal E}|
    &\le (1+|w|)\,d\mu,\\
    |D_{x,t}\chi_{\mathcal E}|
       (\mathbb R^{n+1}\times(0,T))
    &\le TM+\sqrt{TM}\sqrt{2M}
     =M\bigl(T+\sqrt{2T}\bigr).
\end{split}
\]
This proves the required absolute continuity and space-time BV regularity,  whereas \eqref{eq:12Holder} gives $\chi_{E_t}\in C^{1/2}_{\mathrm{loc}}([0,\infty);L^1)$. All assumptions of Theorem \ref{thm:L2-to-BV} are now satisfied. Applying with vector velocity $\mathbf H$ completes the proof.

\begin{equation} \label{en-conv}
    \lim_{j \to \infty}\int_0^T P(E_t^{h_j})\,dt
    =\int_0^T P(E_t)\,dt.
\end{equation}
Moreover, $\{V_t\}$ is a unit density Brakke flow, and therefore for a.e. $t > 0$ there exists $B_t$ with $\mathcal H^n(B_t)=0$ such that for every $x \in \mathrm{spt}\,\|V_t\| \setminus B_t$ $(x,t)$ is a regular point of the flow.

Test \eqref{eq:BV-gener} with $\phi(x,\tau) = \eta(\tau)\psi(x)$, where $\eta \in C^1_c([0,\infty))$ with $\eta(\tau) \equiv 1$ on $[s,t]$ and $\psi \in C^1_c(\mathbb R^{n+1})$ with $\|\psi\|_\infty \leq 1$. Then by Cauchy-Schwarz  
\[
\left| \int_{\mathbb R^{n+1}} (\chi_{E_t}-\chi_{E_s})\,\psi \,dx\right|\leq \int_s^t \sqrt{a(\tau) \,e(\tau)} \, d\tau\,,
\]
where $e(t) = \int |\mathbf H_t|^2\, d\mu_t$. Taking the supremum over $\psi$ on the left then yields
\begin{equation}
    |E_t \triangle E_s| \leq \int_s^t \sqrt{a(\tau) \,e(\tau)} \, d\tau\,.
\end{equation}
The nonincreasing function $a$ satisfies $e(t)\le-a'(t)$ almost everywhere thanks to \eqref{eq:def-Brakke}. Thus we may continue
\[
 \int_s^t \sqrt{a(\tau) \,e(\tau)} \, d\tau \leq \int_s^t \sqrt{-\frac{(a^2(\tau))'}{2}}\,.
\]
We deduce \eqref{eq:holder-sharp} from Cauchy-Schwarz.

For the second part of the statement, observe first that \eqref{eq:holder-sharp} gives
\[
|E_t| \geq |E_0| - \sqrt{t/2} \,P(E_0)\,,
\]
and thus $|E_t| > 0$ for all 
\[
t < 2 \left( \frac{|E_0|}{P(E_0)}\right)^2\,.
\]
For all such $t$, $P(E_t)>0$ by the isoperimetric inequality, and thus $\|V_t\|(\mathbb R^{n+1})\geq P(E_t)>0$ by domination. This implies \eqref{eq:extinction} assuming $|E_0|>0$.
