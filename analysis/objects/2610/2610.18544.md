\label{definition_bv}
Let $\Omega\subset\mathbb{R}^d$ be open.
We denote by $\mathrm{BV}(\Omega)$ the set of all $f\in L^1_\mathrm{loc}(\Omega)$ whose gradient can be represented as a Radon measure, in the sense that there exists an $\mathbb{R}^d$-valued Radon measure $\mathrm D f$ on $\Omega$ such that for all $\bm{\varphi}\in C^1_\mathrm{c}(\Omega,\mathbb{R}^d)$ we have
\[
\int_{\Omega}
f \operatorname{div} \bm{\varphi} 
=
\int_{\Omega}
 \bm{\varphi} \cdot \mathop{}\!\mathrm{d} \mathrm D f
.
\]
Moreover, for $f\in L^1_\mathrm{loc}(\Omega)$ denote
\begin{equation}\label{20261008_21:50}
\operatorname{Var}_\Omega(f)
=
\sup\Bigl\{
\int_{\Omega}
f \operatorname{div} \bm{\varphi} 
:
\bm{\varphi} \in C^1_\mathrm{c}(\Omega,\mathbb{R}^d)
,\ 
\|\bm{\varphi}\|_{L^\infty(\Omega)}\leq 1
\Bigr\}
.
\end{equation}
Then $f\in\mathrm{BV}(\Omega)$ if and only if $\operatorname{Var}_\Omega(f)<\infty$, in which case $|\mathrm D f|(\Omega)=\|\mathrm D f\|=\operatorname{Var}_\Omega(f)$.

We denote by $\mathrm{BV}_\mathrm{loc}(\Omega)$ the set of all $f\in L^1_\mathrm{loc}(\Omega)$ such that, for all open $\Omega'  \subset  \subset \Omega$, we have $f\in\mathrm{BV}(\Omega')$.

\begin{equation}
\label{weak_definition_weighted}
\int_{(a,b)}
r^{d-1}
\mathop{}\!\mathrm{d}|\widetilde f'|(r)
=
\sup\left\{
\int
\varphi'
\widetilde f
:
\varphi\in C^1_\mathrm{c}((a,b)),\ |\varphi(r)|\leq r^{d-1}
\right\}
.
\end{equation}
First consider $0<a < b<\infty$. Then $\varphi\in C^1_\mathrm{c}((a,b))$ if and only if $r\mapsto\varphi(r)/r^{d-1}\in C^1_\mathrm{c}((a,b))$, and
\[
\int(\varphi(r)/r^{d-1})r^{d-1}\mathop{}\!\mathrm{d}{\widetilde f'}(r)
=
\int\varphi'\widetilde f
.
\]
Moreover, $r^{d-1}\widetilde f'(r)\chi_{(a,b)}(r)$ represents a Radon measure, call it $\nu$.

This implies
\(
\int
\psi_k
\mathop{}\!\mathrm{d} \nu
\rightarrow
\|\nu\|
.

Moreover, since $|\alpha(x)|=1$ for $\nu$-almost every $x$, the sequence $\{\psi_n\}_{n\geq 1}$ can be taken with \(\|\psi_n\|_{L^\infty((0,\infty))}\leq1\). Thus, taking suprema over $\varphi$ in the previous equality we obtain \zcref{weak_definition_weighted} which finishes the proof. \label{radialcharfrepresentative}
Let $f\in\mathrm{BV}(\mathbb{R}^d)$ be a radial characteristic function with $\operatorname{Var}(f)<\infty$.
Then exist $N\in\mathbb{N}\cup\{\infty\}$ and sequences $\{a_n\}_{-N-1<n\leq0}$ and $\{b_n\}_{-N-1<n\leq0}$, with $0< a_n<b_n<a_{n+1}$ for each $n$, such that $g:\mathbb{R}^d\rightarrow\mathbb{R}$ given by
\[
\widetilde g = \sum_{n=-N}^0 \chi_{(a_n, b_n)}
\]
satisfies \(\operatorname{Var}(g)=\operatorname{Var}(f)\) and $\mathrm M f=\mathrm M g$.

\label{Prop11_contbv}
Let $\varphi\in L^1(\mathbb{R})$ be non-negative and let $f\in\mathrm{BV}(\mathbb{R})$.
Then, for every $z\in\mathbb{R}$, we have
\[
\mathrm M^s_\varphi f(z)
=
\liminf_{z\neq x\rightarrow z}
\mathrm M^s_\varphi f(x)
.
\]

<!-- proof-17: body_md -->
It suffices to consider $z=0$.
In general, maximal functions of convolution type are lower semicontinuous, so we have
\[\mathrm M^s_{\varphi}f(0) \leq \liminf_{0\neq x \to 0} \mathrm M^s_{\varphi}f(x).\]
Then it suffices to show that 
\begin{equation}\label{20260508_15:08}
\liminf_{0\neq x \to 0} \mathrm M^s_{\varphi}f(x) \leq \mathrm M^s_{\varphi}f(0).
\end{equation}

Since $f\in\mathrm{BV}(\mathbb{R})$ we have $C:=\|f\|_{L^\infty(\mathbb{R}^d)}<\infty$.
Thus, for any $r>0$ and $x,y\in\mathbb{R}$ we have
\begin{equation}
\label{bigr_converge}
\begin{split}
|
f*\varphi_r(x)
-
f*\varphi_r(y)
|
&=
\left|
\int
f(-z)[
\varphi_r(z+x)
-
\varphi_r(z+y)
]
\mathop{}\!\mathrm{d} z
\right|
\\
&\leq
C
\int
|
\varphi_r(z+x)
-
\varphi_r(z+y)
|
\mathop{}\!\mathrm{d} z
\\
&=
C
\int
|
\varphi(z)
-
\varphi(z+(y-x)/r)
|
\mathop{}\!\mathrm{d} y
\\
&\rightarrow0
\qquad
\text{as }
(y-x)/r
\rightarrow0
.
\end{split}
\end{equation}

Let $x_n \rightarrow0$ for which $\mathrm M^{\mathrm s}_\varphi f(x_n)$ converges.
Then there exist $r_n>0$ with
\begin{equation}
\label{good_rn}
\lim_{n\rightarrow\infty} f*\varphi_{r_n}(x_n)
=
\lim_{n\rightarrow \infty} \mathrm M^s_{\varphi}f(x_n)
.
\end{equation}
We consider the following dichotomy:
\begin{enumerate}
\item
\label{largeradius}
There exist $R>0$ and $x_n\rightarrow0$ for which $\mathrm M^{\mathrm s}_\varphi f(x_n)$ converges, and $r_n\geq R$ such that \zcref{good_rn} holds.
\item
\label{smallradius}
For all $x_n\rightarrow0$ for which $\mathrm M^{\mathrm s}_\varphi f(x_n)$ converges, if $r_n>0$ satisfies \zcref{good_rn} then $r_n\rightarrow0$.
\end{enumerate}
Note that both statements also quantify over the choice $x_n=0$.

Assume \zcref{largeradius} holds.
Then for any other sequence $y_n\rightarrow0$ by \zcref{bigr_converge} we can conclude
\[
\limsup_{n\rightarrow\infty}
\mathrm M^s_{\varphi}f(y_n)
\geq
\limsup_{n\rightarrow\infty}
f*\varphi_{r_n}(y_n)
=
\limsup_{n\rightarrow\infty}
f*\varphi_{r_n}(x_n)
=
\lim_{n\rightarrow \infty} \mathrm M^{\mathrm s}_{\varphi}f(x_n)
.
\]
Taking $y_n\neq0$ realizing $\liminf_{0\neq x \to 0} \mathrm M^s_{\varphi}f(x)$, i.e.
\[
\lim_{n\rightarrow\infty} \mathrm M^s_{\varphi}f(y_n)
=
\liminf_{0\neq x \to 0} \mathrm M^s_{\varphi}f(x)
,
\]
this implies that also $x_n$ realizes it.
Then taking $y_n=0$ this concludes the proof of \zcref{20260508_15:08} in this case.

It remains to consider \zcref{smallradius}.
Since $f\in\mathrm{BV}(\mathbb{R})$ there exists $C\in\mathbb{R}$ such that $f=C\chi_{[0,\infty)}+g$ almost everywhere, where $g$ is continuous in $0$.
By symmetry it suffices to consider $C\geq0$.
Let $x_n\rightarrow0$ and $r_n\rightarrow0$.
Then
\begin{align*}
C\chi_{[0,\infty)}*\varphi_{r_n}(x_n)
&=
C\int_{-\infty}^{x_n/r_n}
\varphi
,&
g*\varphi_{r_n}(x_n)
&\rightarrow
g(0)
.
\end{align*}
In particular taking $x_n=0$ this means
\[
\mathrm M^{\mathrm s}_\varphi f(0)
\geq
g(0)
+
C\int_{-\infty}^0\varphi
.
\]
There exists a sequence $0>x_n\rightarrow0$ for which $\mathrm M^{\mathrm s}_\varphi f(x_n)$ converges and
$0<r_n\rightarrow0$ satisfying \zcref{good_rn}.
Since $\varphi\geq0$, we can conclude
\[
\liminf_{0\neq x\rightarrow0}
\mathrm M^{\mathrm s}_\varphi f(x)
\leq
\lim_{n\rightarrow\infty}
\mathrm M^{\mathrm s}_\varphi f(x_n)
=
\lim_{n\rightarrow\infty}
f*\varphi_{r_n}(x_n)
\leq
g(0)
+
C\int_{-\infty}^0\varphi
\leq
\mathrm M^{\mathrm s}_\varphi f(0)
,
\]
which finishes the proof.
