			\item $F=\mathbb{Q}(\sqrt{2q_1q_2})$ with $q_1\equiv3\pmod8$ or $q_2\equiv3\pmod8.$
			\item $F=\mathbb{Q}(\sqrt{pq_1q_2})$ with $\left(\frac{p}{q_1}\right)=-1$ or $\left(\frac{p}{q_2}\right)=-1.$

		\end{enumerate}

\label{classpd}
Let  $p\equiv1\pmod4$ be a prime number and   $d$ a  positive square-free integer co-prime with $p$. Put $K=\mathbb{Q}(\sqrt{p},\sqrt{d})$.	Then $h_2(K)=2$ if and only if, $d$ is in one of the following forms: 
\begin{enumerate}[$A)$]
	\item $d=2$, with $p\equiv1\pmod8$, $\left(\frac{2}{p}\right)_4=(-1)^{\frac{p-1}{8}}$ and $h_2(2p)=4.$
	\item $d=p_1,$ with $p_1\equiv1\pmod4$ a prime number such that $h_2(pp_1)=4.$
	
	\item $d=\eta q,$ where $\eta\in \{1,2\}$ and $q\equiv3\pmod4$ is a prime number such that $p\equiv1\pmod8$,  $\left(\frac{p}{q}\right)=1$ and  $h_2(\eta pq)=4$.
	
	\item $d=q_1q_2$, where $q_1\equiv q_2\equiv
	3\pmod4$ are distinct prime numbers  such that  $\left(\frac{p}{q_1}\right)=\left(\frac{p}{q_2}\right)=1$ and $h_2(pq_1q_2)=4.$
	\item $d=p_1q$, where $p_1\equiv-q\equiv1\pmod4$ are distinct prime numbers such that $p_1\equiv5\pmod8$ and one of the following conditions holds: 
	\begin{enumerate}[$1)$]
		\item $p\equiv1\pmod8$ and $\left(\frac{p}{p_1}\right)=\left(\frac{p}{q}\right)=-\left(\frac{p_1}{q}\right)=-1,$
		                                                                                                                               
		\item $p\equiv1\pmod8, q\equiv3\pmod8$ and $\left(\frac{p}{p_1}\right)=\left(\frac{p}{q}\right)=\left(\frac{p_1}{q}\right)=-1,$

<!-- lemma-7: body_md -->
\label{ZekTAo}
Let \( d \equiv 1 \pmod{4} \) be a positive square free integer and \( \varepsilon_d = x + y\sqrt{d} \) be the fundamental unit of \( \mathbb{Q}(\sqrt{d}) \). Assume \( N(\varepsilon_d) = 1 \), then
\begin{enumerate}[$1)$]
	\item \( x + 1 \) and \( x - 1 \) are not squares in \( \mathbb{Q} \).
	\item For all integer \(\delta \) dividing \( d \), \( \delta(x + 1) \) and \( \delta(x - 1) \) are not squares in \( \mathbb{Q} \).
\end{enumerate}
<!-- end lemma-7 -->
<!-- proof-11: body_md -->
As $d\equiv1\pmod4$, then $\varepsilon_{d}=x+y\sqrt{d}$, where $x$ and $y$ are integers or semi integers :\\
\noindent\ding{229} If $x$ and $y$ are integers then $x+1$ and $x-1$ are not squares in $\mathbb{N}$ and  $x^2-1=dy^2$ (cf.  \cite[First item in Lemma 2]{AzAekhTouCor}). Assume that there is a divisor $\delta$ of $d$ such that:  \( \delta(x+1) \) or \( \delta(x-1) \) is a square in \( \mathbb{N} \). Then   we have  

$$\left\{ \begin{array}{ll}
	x\pm 1=\delta y_1^2\\
	x\mp 1=d' y_2^2,
\end{array}\right.$$
where $y_1, y_2$ and $d'$ are integers such that $y=y_1y_2$ and $d=\delta d'$. Then 

$$\delta d'y_2^2=\delta(x\mp1)=\delta(x\pm1\mp2)=\delta (x\pm1)\mp2\delta=\delta^2y_1^2\mp2\delta,$$ then $$dy_2^2=\delta^2y_1^2\mp2\delta.$$
As $d\equiv 1\pmod4$ and $\delta$ divides $d$,  is follows that  $\delta\equiv \pm1\pmod4$. Reducing modulo 4 we get $$  \mp 2 \equiv y_1^2 - y_2^2 \pmod 4 $$
which is an absurd.\\

\noindent\ding{229} If $x$ and $y$ are semi integers : $x=a/2$ and $y=b/2$,  for some odd integers $a$ and $b$ such that $a^2-4=db^2$. First of all if $a-2$ or $a+2$ is a square in $\mathbb{N}$, then we have:

$$\left\{ \begin{array}{ll}
	a\pm 2=b_1^2\\
	a\mp 2=d b_2^2,
\end{array}\right.$$
where $b_1$ and $b_2$ are two integers such that $b=b_1b_2$. Then $\varepsilon_d=\frac{1}{4}(b_1^2+b_2^2 d+2b\sqrt{d})$, thus $\sqrt{\varepsilon_d}\in \mathbb{Q}(\sqrt{d})$, which is absurd. Therefore, \(a+2\) and \(a-2\) are not squares in \(\mathbb{N}\); hence, a fortiori, \(x-1\) and \(x+1\) are not squares in \(\mathbb{Q}\).


\medskip

Now  if there is some divisor  $\delta$ of $d$ such that  $\delta(x+1)$ or $\delta(x-1)$ is a square in $\mathbb{Q}$. Then as $a$ is odd we have $\delta(x\pm1)=\frac{\delta(a\pm2)}{2}$ which can not be a square in $\mathbb{Q}$ because $\delta(a\pm2)$ and $2$ are coprime and both square free (A rational number is a square in $\mathbb{Q}$ if and only if both its numerator and its denominator are squares in $\mathbb{N}$.).
