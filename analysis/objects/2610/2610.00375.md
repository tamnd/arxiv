<!-- theorem-1: body_md -->
\label{thm:triangles}
Every set \(X=\{p_1,\ldots,p_n\}\subset\mathbb{R}^2\) has a coloring \(\chi\in\{-1,1\}^n\) such that:
\[
|\chi(X\cap T)|\leqslant C\,n^{1/4}\qquad\text{for all triangles }T,
\]
where \(C>0\) is an absolute constant.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:prefix1}
Every ordered \(X=\{p_1,\ldots,p_n\}\subset\mathbb{R}^2\) has a coloring \(\chi\in\{-1,1\}^n\) such that:
\[
|\chi(F\cap X_k)|\leqslant C\,n^{1/4}\qquad\forall F\in\mathcal H(X),\ \ 0\leqslant k\leqslant n,
\]
where \(C>0\) is an absolute constant.
<!-- end theorem-2 -->
<!-- proposition-1: body_md -->
\label{slabs}
Let \(v_1,\ldots,v_M\in\mathbb{R}^n\) and \(a_1,\ldots,a_M>0\). Set:
\[
K:=\bigl\{z\in\mathbb{R}^n:\ |\langle v_j,z\rangle|\leqslant a_j\quad \forall j\bigr\},
\]
and assume \(K\) is bounded. Define:
\[
\psi(z):=Z^{-1/2}\,e^{-\|z\|_2^2/4}\prod_{j=1}^M
g\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr),
\]
where \(Z\) is chosen so that \(\|\psi\|_{L^2(\mathbb{R}^n)}=1\).
Then:
\begin{equation}\label{eq:slab-fisher}
4\int_{\mathbb{R}^n}\|\nabla\psi\|_2^2
\leqslant
n+2\,C\sum_{j=1}^M\frac{\|v_j\|_2^2}{a_j^2}
\exp\Bigl(-\frac{a_j^2}{8\|v_j\|_2^2}\Bigr).
\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Since \(0 \leqslant g \leqslant 1\), 
\[
Z\psi(z)^2\leqslant e^{-\|z\|_2^2/2},
\]
with equality near \(z=0\). Hence, \(0<Z<\infty\). Clearly, \(\psi\in C^\infty(\mathbb{R}^n)\) and \(\operatorname{supp}\psi\subset K\). Since \(K\) is bounded, \(\psi\in C_c^\infty(\mathbb{R}^n)\).
\smallskip 

Let \(\Omega:=\{\psi>0\}\); this is the interior of \(K\). Write:
\[
\psi^2=Z^{-1}e^{-V},\quad
\text{where }\ V(z)=\frac{\|z\|_2^2}{2}+\sum_{j=1}^M U\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr).
\]
Then:
\begin{equation}\label{DeltaV}
\Delta V=n+\sum_{j=1}^M\frac{\|v_j\|_2^2}{a_j^2}\,\ddot U\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr)\quad\text{on }\Omega.
\end{equation}
Also, \(V=-2\log\psi-\log Z\) on \(\Omega\). Hence:
\[
\psi^2\Delta V=2\|\nabla\psi\|_2^2-2\psi\Delta\psi\quad\text{on }\Omega.
\]
Consequently,
\[
\int_\Omega\psi^2\Delta V=
\int_\Omega\bigl(2\|\nabla\psi\|_2^2-2\psi\Delta\psi\bigr).
\]
\(\psi\) vanishes outside \(\Omega\), and so does 
\(\nabla\psi\). Therefore:
\[
\int_\Omega\psi^2\Delta V=
\int_{\mathbb{R}^n}\bigl(2\|\nabla\psi\|_2^2-2\psi\Delta\psi\bigr).
\]
Since \(\psi\in C_c^\infty(\mathbb{R}^n)\), integration by parts gives:
\[
-\int_{\mathbb{R}^n}\psi\Delta\psi=\int_{\mathbb{R}^n}\|\nabla\psi\|_2^2.
\]
Consequently,
\[
\int_\Omega\psi^2\Delta V=4\int_{\mathbb{R}^n}\|\nabla\psi\|_2^2.
\]
Let us substitute \eqref{DeltaV} for \(\Delta V\). 
By \eqref{eq:gU}, each \(\ddot U(\langle v_j,z\rangle/a_j)\,\psi^2\) is at most \(C/Z\); and \(\Omega\) is bounded. So each term is integrable, and we obtain:
\begin{equation}\label{eq:energy-sum}
4\int_{\mathbb{R}^n}\|\nabla\psi\|_2^2
=n+\sum_{j=1}^M\frac{\|v_j\|_2^2}{a_j^2}
\int_\Omega\ddot U\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr)\psi(z)^2\,dz.
\end{equation}
Let us bound each integral on the right. Fix \(j\). For each \(z\), use the orthogonal decomposition:
\[
z=y\,\frac{v_j}{\|v_j\|_2}+w,\quad
y=\frac{\langle z,v_j\rangle}{\|v_j\|_2},\quad w\perp v_j.
\]
Then:
\[
\psi(z)^2=
\frac1Z\,e^{-y^2/2}\,
g\Bigl(\frac{\|v_j\|_2\,y}{a_j}\Bigr)^2
\cdot
e^{-\|w\|_2^2/2}
\prod_{k\ne j}
g\!\left(
\frac{y\left\langle v_k,\frac{v_j}{\|v_j\|_2}\right\rangle+\langle v_k,w\rangle}{a_k}
\right)^2.
\]
Let \(z\sim\psi^2\). Integrating over \(w\in v_j^\perp\), we find the density of \(y\):
\[
p_j(y)
=
\frac1Z\,e^{-y^2/2}\,
g\Bigl(\frac{\|v_j\|_2\,y}{a_j}\Bigr)^2
\int_{v_j^\perp}
e^{-\|w\|_2^2/2}
\prod_{k\ne j}
g\!\left(
\frac{y\left\langle v_k,\frac{v_j}{\|v_j\|_2}\right\rangle+\langle v_k,w\rangle}{a_k}
\right)^2
dw.
\]
Set:
\[
H(y,w):=e^{-\|w\|_2^2/2}
\prod_{k\ne j}
g\!\left(
\frac{y\left\langle v_k,\frac{v_j}{\|v_j\|_2}\right\rangle+\langle v_k,w\rangle}{a_k}
\right)^2,
\qquad
h(y):=\int_{v_j^\perp}H(y,w)\,dw.
\]
Then:
\[
p_j(y)=\frac1Z\,e^{-y^2/2}\,g\Bigl(\frac{\|v_j\|_2\,y}{a_j}\Bigr)^2h(y).
\]
\(g\) is log-concave on \(\mathbb{R}\); hence, each factor of \(H\) is log-concave in the joint variable \((y,w)\).
Thus, \(H(y,w)\) is log-concave. By Pr\'ekopa's theorem (eg, \cite{SaumWell}), \(h(y)\) is also log-concave. 
\smallskip 
 
\(g\) is even; hence, \(H(y,w)=H(-y,-w)\). Thus:
\[
h(y)=\int_{v_j^\perp}H(y,w)\,dw=
\int_{v_j^\perp}H(-y,-w)\,dw=
\int_{v_j^\perp}H(-y,w')\,dw'=h(-y),
\]
and \(h(y)\) is also even.
\smallskip 
 
An even log-concave function is non-increasing on \([0,\infty)\). Therefore:
\begin{equation}\label{h-monotone}
\forall s,t\geqslant0\qquad h(s+t)\leqslant h(s).
\end{equation}
Let us bound the \(j\)-th integral in \eqref{eq:energy-sum}. We compute:
\[
\int_\Omega\ddot U\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr)\psi^2
=\operatorname{\mathbb{E}}_y\,\ddot U\Bigl(\frac{\|v_j\|_2\,y}{a_j}\Bigr)
=\frac1Z\int_{\mathbb{R}}\bigl(g^2\ddot U\bigr)\Bigl(\frac{\|v_j\|_2\,y}{a_j}\Bigr)e^{-y^2/2}h(y)\,dy,
\]
where the integrand is set to zero for \(|y|\geqslant a_j/\|v_j\|_2\). By \eqref{eq:gU}, \(g^2\ddot U\leqslant C\). Also, \(\ddot U=0\) on \((-1/2,1/2)\). Hence:
\[
\int_\Omega\ddot U\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr)\psi^2
\leqslant C\int_{t<|y|<2t}q(y)\,dy,
\qquad\text{where }\ t:=\frac{a_j}{2\|v_j\|_2},\quad q(y):=\frac1Z\,e^{-y^2/2}h(y).
\]
From \eqref{h-monotone}:
\[
\forall s \geqslant 0\quad 
q(s+t) = \frac1Z\,e^{-(s+t)^2/2}h(s+t) \leqslant 
\frac1Z\, e^{-s^2/2} e^{-t^2/2} h(s) = 
e^{-t^2/2} q(s).
\]
Hence:
\[
\int_t^{2t}q(y)\,dy\leqslant e^{-t^2/2}\int_0^{t}q(y)\,dy.
\]
\(q\) is even; hence:
\[
\int_{t<|y|<2t}q(y)\,dy = 2 \int_{t<y<2t}q(y)\,dy.
\]
Therefore:
\[
\int_\Omega\ddot U\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr)\psi^2 \leqslant 2 C e^{-t^2/2}\int_0^{t}q(y)\,dy.
\]
For \(|y|\leqslant t\), \(g(\|v_j\|_2\,y/a_j)=1\); hence: \(q(y)=p_j(y)\). Thus:
\[
\int_{0}^{t} q(y)\,dy=
\int_{0}^{t} p_j(y)\,dy = \mathbb P\bigl(0 \leqslant y \leqslant t\bigr)\leqslant1.
\]
Consequently,
\[
\int_\Omega\ddot U\Bigl(\frac{\langle v_j,z\rangle}{a_j}\Bigr)\psi^2
\leqslant 2C e^{-t^2/2}
=2C \exp\Bigl(-\frac{a_j^2}{8\|v_j\|_2^2}\Bigr),
\]
and the proof follows. \qed

<!-- proposition-2: body_md -->
\label{prop:halfplanes}
For each \(D > 0\), there 
exists \(\psi_D\in C_c^\infty(\mathbb{R}^n)\) such that:
\[
\|\psi_D\|_{L^2(\mathbb{R}^n)}=1,\quad
\operatorname{supp}\psi_D\subset K_{D}(X),
\]
and such that:
\begin{equation}\label{eq:halfplane-energy}
\int_{\mathbb{R}^n}\|\nabla\psi_D\|_2^2\leqslant C_1\,\frac{n^{3/2}}{D^2},
\end{equation}
where \(C_1 > 0\) is an absolute constant.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Choose \(k_0\) so that:
\[
\sqrt n\leqslant2^{k_0}<2\sqrt n.
\]
One may take \(k_0:=\lceil\frac12\log_2n\rceil\). Set:
\[
b_i:=n^{1/4}\,2^{-|i-k_0|/4},
\]
and set:
\[
K':=\bigl\{z\in\mathbb{R}^n:\ |\langle v,z\rangle|\leqslant b_i\ \text{for all }v\in\mathcal V_i\text{ and all }i\bigr\}.
\]
By \eqref{eq:telescope}, for \(z\in K'\) and \(H\in\mathcal H(X)\):
\[
|\langle\mathbf{1}_H,z\rangle| \leqslant 
\sum_{i=1}^L
|\langle\mathbf{1}_{F_i}-\mathbf{1}_{F_{i-1}},z\rangle| \leqslant\sum_{i=1}^Lb_i,
\]
since each \(\mathbf{1}_{F_i}-\mathbf{1}_{F_{i-1}}\) is either 0 or in \(\mathcal V_i\). Consequently,
\[
|\langle\mathbf{1}_H,z\rangle| \leqslant Bn^{1/4},\qquad
B:=\sum_{i\in\mathbb Z}2^{-|i|/4}.
\]
Hence:
\begin{equation}\label{eq:chain-support}
K'\subset K_{Bn^{1/4}}(X).
\end{equation}
In particular, \(K'\) is bounded.
\smallskip 

\emph{Fisher cost.} Apply Proposition~\ref{slabs} to \(K'\): the vectors are \(v\in\mathcal V_i\), and the width of \(v\in\mathcal V_i\) is \(b_i\). Let \(\psi\) be the resulting function. By \eqref{eq:slab-fisher},
\[
\int_{\mathbb{R}^n}\|\nabla\psi\|_2^2\leqslant
n+O\Bigl(\sum_{i=1}^L\sum_{v\in\mathcal V_i}f_i\bigl(\|v\|_2^2\bigr)\Bigr),
\qquad\text{where }\ f_i(x):=\frac{x}{b_i^2}\exp\Bigl(-\frac{b_i^2}{8x}\Bigr).
\]
Each \(f_i\) is increasing on \((0,\infty)\). By \eqref{eq:increments}, \(\|v\|_2^2\leqslant s_i\) for \(v\in\mathcal V_i\); hence, \(f_i(\|v\|_2^2)\leqslant f_i(s_i)\). Also by \eqref{eq:increments}, the inner sum has at most \(C_0\,4^i\) terms. Therefore:
\[
\int_{\mathbb{R}^n}\|\nabla\psi\|_2^2\leqslant
n+O\Bigl(\sum_{i=1}^L4^i\,\frac{s_i}{b_i^2}\exp\Bigl(-\frac{b_i^2}{8s_i}\Bigr)\Bigr).
\]
Let us bound 
the \(i\)-th term. Write: 
\[
j:=i-k_0.
\] 
Since \(2^{k_0}<2\sqrt n\), 
\[
4^{i} = 4^{j} 4^{k_0} = O(n\,4^j).
\]
Since \(2^{k_0}\geqslant\sqrt n\), 
\[
s_i= n 2^{1-i} = 2n2^{-k_0}\,2^{-j}\leqslant2\sqrt n\,2^{-j}.
\]
Since \(b_i^2=\sqrt n\,2^{-|j|/2}\),
\[
4^i\,\frac{s_i}{b_i^2}=O\bigl(n\,2^{j+|j|/2}\bigr),\qquad
\frac{b_i^2}{8s_i}\geqslant\frac{2^{j-|j|/2}}{16}.
\]
\emph{Case \(j \leqslant 0\).} In this case, 
\[
4^i\,\frac{s_i}{b_i^2} = 
O\bigl(n\,2^{j+|j|/2}\bigr) = 
O\bigl(n\,2^{-|j|/2}\bigr).
\]
We bound the \(i\)-th term as follows:
\[
4^i\,\frac{s_i}{b_i^2}\exp\Bigl(-\frac{b_i^2}{8s_i}\Bigr) \leqslant 
4^i\,\frac{s_i}{b_i^2} = O\bigl(n\,2^{-|j|/2}\bigr).
\]
\emph{Case \(j > 0\).} In this case, 
\[
4^i\,\frac{s_i}{b_i^2} = 
O\bigl(n\,2^{j+|j|/2}\bigr) = 
O\bigl(n\,2^{3j/2}\bigr),\qquad 
\frac{b_i^2}{8s_i}\geqslant\frac{2^{j-|j|/2}}{16} = \frac{2^{j/2}}{16}.
\]
We bound the \(i\)-th term as follows:
\[
4^i\,\frac{s_i}{b_i^2}\exp\Bigl(-\frac{b_i^2}{8s_i}\Bigr) \leqslant 
O\bigl(n\,2^{3j/2}\bigr) \cdot 
\exp\bigl(-\frac{2^{j/2}}{16}\bigr) = O\bigl(n\,2^{3j/2}\bigr) \cdot O(2^{-2j}) = O\bigl(n\,2^{-j/2}\bigr).
\]
In both cases, the \(i\)-th term is \(O(n\,2^{-|j|/2})\). Therefore:
\[
\sum_{i=1}^L4^i\,\frac{s_i}{b_i^2}\exp\Bigl(-\frac{b_i^2}{8s_i}\Bigr) \leqslant 
O\bigl(
\sum_{j\in\mathbb Z} n\,2^{-|j|/2}
\bigr) = O(n).
\]
Consequently,
\[
\int_{\mathbb{R}^n}\|\nabla\psi\|_2^2\leqslant
n+O\Bigl(\sum_{i=1}^L4^i\,\frac{s_i}{b_i^2}\exp\Bigl(-\frac{b_i^2}{8s_i}\Bigr)\Bigr) = n + O(n) = O(n).
\]
\emph{Scaling.} Now, we rescale \(\psi\). Set:
\[
r:=D/(Bn^{1/4}),\quad 
\psi_D(z):=r^{-n/2}\psi(z/r).
\]
Then \(\|\psi_D\|_{L^2(\mathbb{R}^n)}=1\). Also, \(\operatorname{supp}\psi_D=r\operatorname{supp}\psi\). By \eqref{eq:chain-support},
\[
\operatorname{supp}\psi_D\subset rK_{Bn^{1/4}}(X)=K_{D}(X).
\]
Finally, 
\[
\int\|\nabla\psi_D\|_2^2=r^{-2}\int\|\nabla\psi\|_2^2,\quad 
\int_{\mathbb{R}^n}\|\nabla\psi_D\|_2^2=O\bigl(r^{-2}n\bigr)=O\Bigl(\frac{n^{3/2}}{D^2}\Bigr),
\]
and the proof follows. \qed

<!-- theorem-5: body_md -->
\label{thm:SV}
Let \(K\subset\mathbb{R}^n\) be convex and centrally symmetric. Let \(\psi\in C_c^\infty(\mathbb{R}^n)\) be such that:
\[
\|\psi\|_{L^2(\mathbb{R}^n)}=1,\qquad 
\operatorname{supp}\psi\subset K.
\]
Let \(v_1,\ldots,v_m\in\mathbb{R}^n\) be any vectors, and let \(\varepsilon_1,\ldots,\varepsilon_m\) be iid fair signs. Then there is an online algorithm that discards some terms \(\varepsilon_iv_i\) so that, for each \(k\):
\[
\sum_{i\leqslant k\ \mathrm{accepted}}\varepsilon_iv_i\in2K
\quad\text{deterministically},
\]
and:
\[
\operatorname{\mathbb{E}}\,\#\{\text{\rm discarded}\}\leqslant\sum_{i=1}^m\bigl\|\langle\nabla\psi,v_i\rangle\bigr\|_{L^2(\mathbb{R}^n)}.
\]

<!-- lemma-1: body_md -->
\label{lem:partial}
Every \(X=\{p_1,\ldots,p_n\}\) has a partial coloring \(\chi\in\{-1,0,1\}^n\) with at most \(n/4\) zeros and with:
\[
|\chi(F\cap X_k)|=O(n^{1/4})\qquad\forall F\in\mathcal H(X),\ \ 0\leqslant k\leqslant n.
\]
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
Fix \(D>0\), and choose 
it later. Apply Theorem~\ref{thm:SV} to \(K=K_D(X)\): choose \(\psi=\psi_D\) from Proposition~\ref{prop:halfplanes}, and \(v_i:=e_i\), \(i=1,\ldots,n\). Set:
\[
\chi_i:=\begin{cases}
\varepsilon_i, & \text{if \(\varepsilon_ie_i\) is accepted},\\
0, & \text{if \(\varepsilon_ie_i\) is discarded}.
\end{cases}
\]
Then:
\[
\forall k\quad \sum_{i\leqslant k}\chi_ie_i\in2K_D(X).
\]
Consequently,
\begin{equation}\label{eq:partial-prefix}
|\chi(F\cap X_k)|\leqslant2D\qquad\forall F\in\mathcal H(X),\ \ 0\leqslant k\leqslant n.
\end{equation}
By Theorem~\ref{thm:SV},
\[
\operatorname{\mathbb{E}}\,\#\{\chi_i=0\}
\leqslant\sum_{i=1}^n\|\partial_i\psi_D\|_{L^2(\mathbb{R}^n)}.
\]
By Cauchy-Schwarz,
\[
\operatorname{\mathbb{E}}\,\#\{\chi_i=0\}
\leqslant \sqrt n\,\Bigl(\int_{\mathbb{R}^n}\|\nabla\psi_D\|_2^2\Bigr)^{1/2}.
\]
From \eqref{eq:halfplane-energy}:
\[
\operatorname{\mathbb{E}}\,\#\{\chi_i=0\} =O\Bigl(\frac{n^{5/4}}{D}\Bigr).
\]
Choose \(D=O(n^{1/4})\) so that 
\(\operatorname{\mathbb{E}}\,\#\{\chi_i=0\}\leqslant n/4\). Then there exists some \(\chi\) with at most \(n/4\) zeros, and the proof follows. \qed

\smallskip 

We iterate Lemma~\ref{lem:partial}, as in 
\cite[\S\,2]{Versh-Sm-4}. First, apply it to \(X\).

\[

\]
and the proof follows. \qed
