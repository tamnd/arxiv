<!-- defn-1: body_md -->
\label{def:global solution}
 (i) $\mathcal{G}_c:=\big\{$ $u$: $u$ satisfies the obstacle problem (\ref{E1.1}), and $\Omega^c(u)$ is compact.$\big\}$

 (ii) $\mathcal{G}_{\infty}:=\big\{$ $u$: $u$ satisfies the obstacle problem (\ref{E1.1}) in $\mathbb{R}^n$, where $n\geq 2$. For such $u$,
 there exists a unique unit vector $e\in\partial B_1$, such that $u(\cdot+te)\leq u(\cdot)$ for $t\in\mathbb{R}_+$.$\big\}$

(iii) $\mathcal{Q}_c:=\big\{q(x)$ : $q$ is a function satisfies (\ref{E1.23})-(\ref{E1.24}). $\big\}$

\begin{equation}\label{E1.23}
q(x)=\left\{\begin{split}
&\frac{1}{2}(x-x_0)^TA(x-x_0)+D,\qquad n\geq 3,\\
&\frac{1}{2}(x-x_0)^TA(x-x_0)+D\log|P(x-x_0)|,\qquad n=2,
\end{split}\right.
\end{equation}
where $A\in\mathcal{S}_{n\times n}$ is a positive definite matrix, $x_0\in \mathbb{R}^n$, and $D\leq 0$. Furthermore, we have $F(A)=1$.

The positive definite matrix $P$ is determined as follows:
\begin{equation}\label{E1.24}
P=\left(\sqrt{[F_{ij}(A)]_{2\times 2}}\right)^{-1},
\end{equation}
where $F_{ij}(\cdot)$ denotes the derivatives of the operator $F$.

<!-- thm-1: body_md -->
\label{thm:1}
 Let $F$ be a convex operator satisfying assumptions (H1)-(H3), and let $u$ be a global solution to the fully nonlinear obstacle problem (\ref{E1.1}). Suppose that the free boundary $\Gamma(u)\neq\emptyset$.

 Then $u$ is convex. Furthermore, one of the following cases holds:

 (i) $\Omega^{c}(u)$ is a single point in $\mathbb{R}^n$, and $u$ is a quadratic polynomial of the form
\begin{equation*}
u(x)=\frac{1}{2}(x-x_0)^TA(x-x_0),
\end{equation*}
where $A\in\mathcal{S}_{n\times n}$ is a positive definite matrix.

(ii) $u\in\mathcal{G}_c$ with $\mathrm{Int}\,\Omega^{c}(u)\neq\emptyset$. In addition, $\Omega^{c}(u)$ is a $C^{1,\alpha}$ domain for some exponent $\alpha\in(0,1)$.

(iii) $n=1$, and $\Omega^{c}(u)$ is a ray in $\mathbb{R}^1$. After a suitable translation and rotation of the coordinate system, $u$ can be rewritten as
\begin{equation}\label{E1.5}
u(x)=\frac{\gamma}{2}x_{+}^{2},
\end{equation}
where the constant \(\gamma\in [1/\Lambda,1/\lambda]\) satisfies \(F(\gamma)=1\).

(iv) $u\in\mathcal{G}_{\infty}$. After a suitable translation and rotation of the coordinate system, $u$ satisfies the following properties (a)-(c).

  $\quad$(a) $\partial_n u>0$ in $\Omega(u)$, and \(0\in\Omega^{c}(u)\subseteq\{x\in\mathbb{R}^{n}: x_n\leq 0\}.\)

\(\quad\)(b) There exists a unique positive definite matrix \(A_{n-1}\in\mathcal{S}_{(n-1)\times (n-1)}\) such that
 \begin{equation}\label{E1.13}
 u_{r}(x',x_n)\rightarrow\frac{1}{2}x'^TA_{n-1}x'\quad\text{in}\quad C^{1,\alpha}_{loc}(\mathbb{R}^{n}),\quad\text{as}\quad r\rightarrow +\infty,
 \end{equation}
for all $\alpha\in(0,1)$ (see Notations 9). Furthermore,
 \begin{equation}\label{E1.25}
F\left(
\begin{gathered}
\begin{matrix}
A_{n-1} & 0 \\ 0 & 0
\end{matrix}
\end{gathered}\right)=1.
 \end{equation}

  $\quad$(c) For any \(\rho>0\), define the sliced non-contact set
  \begin{equation}\label{E1.19}
  \Omega^{c}_{\rho}(u):=\Omega^{c}(u)\cap\{x_n=-\rho\}.
  \end{equation}
The following relations hold:
\begin{equation}\label{E1.18}
\left\{\begin{split}
&\lim\limits_{\rho\rightarrow+\infty}\mbox{diam}(\Omega^{c}_{\rho}(u))=+\infty,\\
&\lim\limits_{\rho\rightarrow+\infty}\mbox{diam}(\Omega^{c}_{\rho}(u))/\rho=0,\\
&\lim\limits_{\rho\rightarrow+\infty}\mbox{diam}\big(\Omega^{c}_{\rho}(u)\big)/\mathrm{MD}\big(\Omega^{c}_{\rho}(u)\big)= C,
\end{split}\right.
\end{equation}
where the constant $C$ depends only on the operator $F$ and the matrix \(A_{n-1}\).

(v) $u$ is a lower dimensional global solution, that is, there exists a unit vector $e\in\partial B_1$, such that $u(\cdot+te)\equiv u(\cdot)$ for all $t\in\mathbb{R}$.

<!-- thm-2: body_md -->
\label{thm:5}
  Assume that the spatial dimension $n\geq 2$. Let $F$ be a convex operator satisfying assumptions (H1)-(H3). Then there exists a bijection $\Phi:\mathcal{G}_c\rightarrow \mathcal{Q}_c$
 such that
\begin{equation}\label{E1.12}
u(x)=\left\{\begin{split}
\Phi(u)&+O(|x|^{2-n}),\qquad n\geq 3,\\
\Phi(u)&+C+O(|x|^{-1}),\quad n=2,
\end{split}\right.
\quad\mbox{as}\quad |x|\rightarrow +\infty,
\end{equation}
where $C$ is a unique constant depending only on $\Phi(u)$.

\label{pro:2.11}
 Let the operator $F$ be convex, and let $u$ be a viscosity solution to the fully nonlinear obstacle problem in $B_1$. Then the following results hold.

 (1) Nondegeneracy. If $x\in\overline{\Omega(u)}$ and $B_{r}(x)\subset\subset B_1$, then
\begin{equation}\label{E2.15}
\sup\limits_{y\in \partial B_{r}(x)}(u(y)-u(x))\geq C(n,\Lambda)r^2;
\end{equation}

(2) Interior $C^{1,1}$ estimates. If $u(0)=0$, then
\begin{equation}\label{E2.12}
\|D^{2}u\|_{L^{\infty}(B_{1/2})}\leq C(n,\lambda, \Lambda);
\end{equation}

(3) Local  $C^1$ regularity and local $C^{1,\alpha}$ regularity of $\Gamma(u)$. Suppose there exists a constant $\epsilon_0>0$ such that
\begin{equation}\label{E2.20}
\delta_r(u,y)\geq\epsilon_0\qquad \forall y\in B_{1/2}(0)\cap \Gamma(u),\qquad \forall r<1/4.
\end{equation}
Then $B_{1/4}(0)\cap\Gamma(u)$ is a $C^1$ manifold.
Furthermore, if the free boundary $\Gamma(u)\cap B_{1/2}$ is a $C^{1}$ manifold,
then $\Gamma(u)\cap B_{1/4}$ is a $C^{1,\alpha}$ manifold, where $\alpha\in(0,1)$ is a universal constant.

<!-- lemma-1: body_md -->
\label{lemma 3.2}
Let $F$ be a convex operator, and let $u, v\in\mathcal{G}_c$. Assume that $u>v>0$ on $\partial B_1$. Assume in addition that $\operatorname{Int}\Omega^{c}(u)\neq\emptyset$. Then
\begin{equation}\label{E3.64}
\Omega^{c}(u)\subseteq\operatorname{Int}\Omega^{c}(v).
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let us assume by contradiction that (\ref{E3.64}) fails. Using Proposition \ref{pro:2.16}, we deduce that $v\leq u$ in $B_1$. This implies that
$\Omega^{c}(u)\subseteq \Omega^{c}(v)\subset\subset B_1$. Then we may choose $y\in\mathbb{R}^{n}$, such that $y\in\Gamma(u)\cap\Gamma(v)$.
After a translation of coordinates, we set $0\in\Gamma(u)\cap\Gamma(v)$. Then we have
\begin{equation}\label{E3.44}
|\nabla u|(0)=|\nabla v|(0)=0.
\end{equation}
Let $w:=u-v$. Since $F(D^2u)=F(D^2v)$ in $B_1\cap\Omega(v),$ we have
\begin{equation*}
\left\{\begin{split}
\mathcal{M}^-(\lambda/n, \Lambda, D^2w)\leq 0\qquad &\mbox{in}\qquad B_1\cap\Omega(v),\\
w>0\qquad &\mbox{on}\qquad \partial B_1,\\
w\geq 0\qquad &\mbox{on}\qquad \Gamma(v).
\end{split}\right.
\end{equation*}
The strong maximum principle implies
\begin{equation*}
v<u\qquad\mbox{in}\quad B_1\cap\Omega(v).
\end{equation*}

Applying Proposition \ref{pro:2.12}, we find that $u$ and $v$ are convex functions. By the convexity of $\Omega^{c}(v)$, there exists $z\in\Omega(v)$ and $\rho>0$, such that
\begin{equation}\label{E3.47}
B_\rho(z)\subset\Omega(v),\qquad\mbox{and}\quad \partial B_\rho(z)\cap\Omega^{c}(v)=\{0\}.
\end{equation}
Then we have
\begin{equation*}
\left\{\begin{split}
w \in S(\lambda/n, \Lambda, 0)\quad &\mbox{in}\qquad B_\rho(z),\\
w >0\qquad &\mbox{in}\qquad B_\rho(z),\\
w(0) =0.&
\end{split}\right.
\end{equation*}
The Hopf lemma gives
\begin{equation*}
\frac{\partial}{\partial \nu}w(0)>0,
\end{equation*}
where $\nu$ denotes the unit outward normal vector to $\Omega^c(v)$ at the origin (recalling the $C^1$ regularity of the free boundary). However, this contradicts (\ref{E3.44}).

<!-- lemma-3: body_md -->
\label{lemma 3.1}
 Suppose all hypotheses of Theorem \ref{thm:1} hold. Suppose further that the spatial dimension $n=2$. If $u\in\mathcal{G}_c$, then there exists a unique function $q\in\mathcal{Q}_c$ such that
\begin{equation}\label{E3.41}
u(x)=q(x)+C+O(|x|^{-1})\qquad\text{as}\quad |x|\rightarrow +\infty,
\end{equation}
where $C$ is a constant.
Moreover, if $q$ is a quadratic polynomial, then $C\le 0$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
We divide the proof into two steps.

\indent\textit{Step 1. We define $q$.}

Since $\Omega^{c}(u)$ is compact, we may assume $0\in\Omega^{c}(u)\subseteq B_1$. Then
\begin{equation*}
F(D^2u)-1=0\qquad\mbox{in}\quad\mathbb{R}^{2}\backslash B_1.
\end{equation*}
In view of \cite[Theorem 1.1]{LL}, we deduce that
\begin{equation}\label{E4.6}
u(x)=\frac{1}{2}x^TAx+Bx+C+D\log|P x|+O(|x|^{-1}), \qquad\text{as}\quad |x|\rightarrow +\infty,
\end{equation}
where $A,P\in\mathcal{S}_{2\times 2}$. Recall that $P$ is defined in (\ref{E1.24}).
By the interior $C^{1,1}$ estimate, (\ref{E4.6}) implies that
\begin{equation*}
D^2u(x)\rightarrow A,\qquad\text{as}\quad |x|\rightarrow +\infty.
\end{equation*}
It follows that $F(A)=1$.

Since $u\geq 0$, we see that $A\geq 0$. After a suitable rotation of coordinates, we have
\begin{equation*}
u(x)=(a_1x_1^2+a_2x_2^2)+(b_1x_1+b_2x_2)+C+D\log|P x|+O(|x|^{-1}),\qquad\text{as}\quad|x|\rightarrow +\infty.
\end{equation*}
It is readily seen that $a_i\geq 0$ for $i=1,2$. In fact, combining $u\geq 0$ and (\ref{E4.6}), we see that if $a_i=0$, then $b_i=0$. In summary, we obtain
\begin{equation}\label{E3.42}
u(x)=\frac{1}{2}(x-x_0)^TA(x-x_0)+D\log|P (x-x_0)|+C+O(|x|^{-1}),\qquad\text{as}\quad|x|\rightarrow +\infty.
\end{equation}
We define
\begin{equation}\label{E4.55}
q(x):=\frac{1}{2}(x-x_0)^TA(x-x_0)+D\log|P (x-x_0)|.
\end{equation}
This yields (\ref{E3.41}). Uniqueness of $q$ is clear.

\indent\textit{Step 2. $q(x)\in\mathcal{Q}_c$.}

We prove $D\leq 0$ in (\ref{E4.55}). Suppose for contradiction that $D>0$. For $\epsilon>0$, consider the auxiliary function $v(x)$ defined by
\begin{equation*}
v(x):=\frac{1}{2}(x-x_0)^TA(x-x_0)+\epsilon.
\end{equation*}
For sufficiently large $R\gg1$ and small $\epsilon>0$, the condition $D>0$ implies $u\geq v$ on $\partial B_R(x_0)$. Using
\begin{equation*}
F(D^2u)\leq 1=F(D^2v),\qquad \mbox{in}\qquad B_R(x_0),
\end{equation*}
Proposition \ref{pro:2.16} then gives $u\geq v>0$ in $B_R(x_0)$. Hence $\Omega^c(u)=\emptyset$, which is a contradiction.

Before establishing positivity of $A$, we first show that if $D=0$ in (\ref{E3.42}), then $C\leq 0$. Suppose for contradiction that $D=0$ and $C>0$.
Choose $R\gg1$ and small $\epsilon>0$. Then $u\geq v>0$ on $\partial B_R(x_0)$. Using Proposition \ref{pro:2.16} again, then $u\geq v>0$ in $B_R(x_0)$, yielding a contradiction.

We now verify $A>0$. If $D^2+C^2>0$, then nonnegativity $u\ge 0$ implies
\begin{equation*}
\frac{1}{2}(x-x_0)^TA(x-x_0)>0,\qquad\text{as}\quad|x|\rightarrow +\infty.
\end{equation*}
In this case, all eigenvalues of $A$ are positive.
If $C=D=0$, then for $\epsilon>0$, we have
\begin{equation*}
u>\frac{1}{2}(x-x_0)^TA(x-x_0)-\epsilon,\qquad\text{as}\quad|x|\rightarrow +\infty.
\end{equation*}
Applying the comparison principle and letting $\epsilon\rightarrow 0^+$, we obtain $u\geq\frac{1}{2}(x-x_0)^TA(x-x_0)$.
Since $\Omega^{c}(u)\neq\emptyset$, we deduce that $u(x)=\frac{1}{2}(x-x_0)^TA(x-x_0).$ Since $\Omega^{c}(u)$ is bounded, it follows that $A>0$ in this case.

<!-- lemma-4: body_md -->
\label{lemma 3.3}
 Suppose that all hypotheses of Lemma \ref{lemma 3.1} hold. Suppose further that $q\in\mathcal{Q}_c$ is given by (\ref{E3.41}). Then

(i) $x_0\in \mathrm{Int}\,\Omega^c(u)$ provided $D<0$ (see (\ref{E1.23}));

(ii) $u(x)=q(x)=\frac{1}{2}(x-x_0)^TA(x-x_0)$ provided $D=0$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
We show that (i) holds. Without loss of generality, we assume that $x_0=0$. Suppose on the contrary that $D<0$, and the origin $0\notin\mathrm{Int}\,\Omega^{c}(u)$. We prove
 \begin{equation}\label{E3.62}
 \mathrm{Int}\,\Omega^{c}(u)\neq\emptyset.
 \end{equation}

 In fact, if (\ref{E3.62}) is fails, then we have
 \begin{equation*}
 F(D^2 u)=1,\qquad\mbox{a.e. in}\quad\mathbb{R}^n.
\end{equation*}
 By the interior $C^{1,1}$ estimate, we observe that $u$ is a strong $L^n$ solution in $\mathbb{R}^n$. Under assumptions about the operator $F$, strong $L^n$ solutions
 are also viscosity solutions (cf.\cite{Ca02.5}). Then the Evans-Krylov theorem implies that $u$ is a quadratic polynomial. Together with (\ref{E3.41}) and (\ref{E4.55}),
 we deduce that
 \begin{equation*}
 u(x)=q(x)=\frac{1}{2}x^TAx.
 \end{equation*}
 This contradicts $D<0.$ The claim follows.

By $D<0$, we have $u<\frac{1}{2}x^TAx$ as $|x|\rightarrow +\infty$. Using Proposition \ref{pro:2.16} again, we have
\begin{equation*}

\mbox{dist}(te,\mathcal{K})>\frac{r}{2}\sin(\pi/4-\theta/2)\geq\delta/2\quad\mbox{for}\quad t\in[r/2,r).
\end{equation*}
Then we obtain that $w(te)>0$ for $t\in[r/2,r)$.

Next we show that $w\geq-\alpha$ in $\mathcal{C}_{1/2}$. Picking $x\in\mathcal{C}_{1-3\delta}\cap\Big(\Omega_{\delta/2}(\mathcal{K})\backslash\Omega_{\delta}(\mathcal{K})\Big)$,
let $x_0$ be defined as in the previous step. Let $z:=x+\delta\nu(x_0)$. Then $z\in\mathcal{C}_{1-2\delta}$. By
\begin{equation*}
B_{\delta}(z)\cap\mathcal{K}=\emptyset,
\end{equation*}
we have
\begin{equation*}
\inf_{y\in B_{\delta/2}(z)}\mbox{dist}(y,\mathcal{K})\geq\inf_{y\in B_{\delta/2}(z)}\mbox{dist}(y,L(x_0))>\delta/2.
\end{equation*}
Using (\ref{E4.71}), we have $w>0$ in $B_{\delta/2}(z).$ Then $w^{-}$ satisfies
\begin{equation*}
\left\{\begin{split}
\mathcal{L}w^{-}&\geq 0,\quad\mbox{in}\quad B_{2\delta}(z)\subset\mathcal{C}_1,\\
|\{B_{2\delta}(z)&:w^{-}=0\}|\geq|B_{\delta/2}(z)|.
\end{split}\right.
\end{equation*}

Applying the weak Harnack inequality for subsolution (cf. \cite{KS}), we deduce that
\begin{equation*}
w^{-}(x)\leq\sup_{B_{\delta}(z)}w^{-}\leq(1-\beta)\sup_{B_{2\delta}(z)}w^{-}\leq 1-\beta,
\end{equation*}
where $\beta>0$ is a constant depending only on $n,\lambda,\Lambda$, in other words, we have
\begin{equation*}
w^{-}\leq 1-\beta\qquad\mbox{in}\quad\mathcal{C}_{1-3\delta}.
\end{equation*}

Iterating this result, we find that
\begin{equation*}
w\geq-(1-\beta)^{k}\qquad\mbox{in}\quad\mathcal{C}_{1-3k\delta}.
\end{equation*}
Then (\ref{E4.72}) follows by choosing
\begin{equation*}
k=\left[\frac{\log\alpha}{\log(1-\beta)}\right]+1,\quad\quad\delta=\min\big((6k)^{-1}, \sin(\pi/4-\theta/2), 1/32\big).
\end{equation*}

\indent\textit{Step 3. } After a dilation, we repeat the process again. Consider
\begin{equation*}
w_{j}(x):=\alpha^{-j}w(x/2^j),\qquad \mathcal{K}_{j}:=\{x:x/2^j\in\mathcal{K}\},\qquad j=1,2,\cdots
\end{equation*}
By iterating, we conclude that
\begin{equation*}
 w\geq-\alpha^{-j}\quad\mbox{in}\quad\mathcal{C}_{2^{-j}},\qquad\mbox{and}\qquad w(te)>0\quad\mbox{for}\quad t\geq 2^{-j}r.
 \end{equation*}
Hence we obtain (\ref{E4.70}).

<!-- lemma-7: body_md -->
\label{le:2.4}
Let $u(x)$ be a nonnegative convex function in $\mathbb{R}^{n}(n\geq 2).$ If $e\in V(u)$, then $u(\cdot+te)\leq u(\cdot)$ for $t\in\mathbb{R}_+.$ Clearly, we have
$\partial_e u\leq 0$ if $u\in C^1(\mathbb{R}^{n})$.
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Without loss of generality, we set $e=e_{n}$. In other words, there exists a sequence $P_k(P_k',(P_k)_n)$ such that $(P_k)_n\rightarrow+\infty,P_k'/(P_k)_n\rightarrow 0,$
and $u(P_k)=0.$

For any $t\in\mathbb{R}_+$, choosing $\varepsilon_k=t/(P_k)_n\rightarrow 0^+$, we have $|\varepsilon_k P_k'|=o(1)$. By convexity of $u$, it is easily seen that
\begin{equation*}
 \begin{split}
u[(1-\varepsilon_k)x'+\varepsilon_k P_k', (1-\varepsilon_k)x_n+t]&=u[(1-\varepsilon_k)x+\varepsilon_k P_k]\\
&\leq(1-\varepsilon_k)u(x',x_n)+\varepsilon_k u(P_k).
 \end{split}
\end{equation*}
Hence the result follows by taking $k\rightarrow+\infty$.

<!-- lemma-8: body_md -->
\label{le:2.6}
Let $u(x)$ be a nonnegative convex function in $\mathbb{R}^{n}(n\geq 2)$. Suppose that $\mathrm{Int}\,\Omega^{c}(u)\neq\emptyset$. Suppose further that
$V(u)\neq\emptyset$. Then $\mathbb{R}_+V(u)$ is a closed convex cone in $\mathbb{R}^{n}$.
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
Without loss of generality, we assume in addition that the origin $0\in\Omega^{c}(u)$. For any unit vector $\tau\in V(u)$, Lemma \ref{le:2.4} implies that the ray
$\{\mathbb{R}_+\tau\}\subseteq\Omega^{c}(u).$ Hence $\mathbb{R}_+V(u)$ is a cone.

We verify that the cone is convex. Choosing $x_1,x_2\in\mathbb{R}_+V(u)$, let
$$
x_{\theta}:=\theta x_1+(1-\theta)x_2,\qquad \forall\theta\in (0,1).
$$
We want to show $x_{\theta}\in\mathbb{R}_+V(u)$. Once we verify that $sx_{\theta}\in\Omega^{c}(u)$ for any $s\geq 0$, this result follows. In fact, Lemma \ref{le:2.4} implies that
$sx_{1},sx_{2}\in\Omega^{c}(u)$. Since $u\geq 0$ in $\mathbb{R}^{n}$, $u(sx_{\theta})=0$ follows from the convexity of $u$.

Assume now that we have unit vectors $\{\nu_k\}_{k=1}^{\infty}\subset V(u)$ with $\nu_k\rightarrow\nu$. Then there exists
$\{P^{k}_{j}\}_{j=1}^{\infty}\subset\Omega^{c}(u)$ such that
$$
P^{k}_{j}/|P^{k}_{j}|\rightarrow \nu_k \qquad\mbox{as}\quad j\rightarrow +\infty.
$$
Taking a subsequence if necessary, we have
$$
P^{k}_{j}/|P^{k}_{j}|\rightarrow\nu\qquad \mbox{as}\quad j, k\rightarrow +\infty.
$$
Then we deduce that $\nu\in V(u)$, that is, the cone $\mathbb{R}_+V(u)$ is closed.

\{-\mathbb{R}_+e_n\}\subseteq\Omega^{c}(u_R).

 This contradicts the fact that $\Omega^{c}(u_R)$ lies on the same side of $L_{x_0}$. If $\angle\vartheta(\nu(x_0),e_n)\in(\pi/2,\pi)$, then $\{-\mathbb{R}_+e_n\}\cap\Omega^{c}(u_R)=\emptyset$, a contradiction. In summary, we deduce that

\nu(x_0)=a\nu'(x_0)+\sqrt{1-a^2}e_n,
\end{equation*}

\label{thm:thm4.1}
 Suppose that all hypotheses of Corollary \ref{cor: 4.1} hold. Then $V(u)=\{\tau\}$, where $\tau$ is a unit vector.

We divide the proof into four steps.

\indent\textit{Step 1. }
Suppose on the contrary that $\tau_1,\tau_2\in V(u)\cap\partial B_1$, where $\tau_1\neq\tau_2$. Applying Corollary \ref{cor: 4.1}, we have
 \begin{equation*}
0<\sup_{e,\tau\in V(u)}\angle\vartheta(e,\tau)=2\theta_0<\pi.
\end{equation*}
We now define
\begin{equation*}
\tau(\alpha):=\Big(\cos(\pi/2+\alpha),0,\cdots,0,\sin(\pi/2+\alpha)\Big).
\end{equation*}

 Without loss of generality, we can assume that $-\tau(-\theta_0),-\tau(\theta_0)\in V(u)$. Notice that $\tau(0)=e_n$. Since $u$ is a convex function,
by Lemma \ref{le:2.6}, we obtain
 \begin{equation*}
 -\tau(\alpha)\in V(u),\qquad\mbox{for all}\quad\alpha\in[-\theta_0,+\theta_0].
 \end{equation*}
Using Lemma \ref{le:2.4}, we observe that
\begin{equation*}
 \partial_{\tau(\alpha)}u\geq 0,\qquad\mbox{for all}\quad\alpha\in[-\theta_0,+\theta_0].
 \end{equation*}
Fixing $\alpha\in[-\theta_0,+\theta_0]$, then $\partial_{\tau(\alpha)}u$ satisfies
\begin{equation}\label{E6.10}
\left\{\begin{split}
a^{ij}(x)\partial_{ij}w=0 &\quad\mbox{in}\quad\Omega(u),\\
w=0 &\quad\mbox{on}\quad \Gamma(u),
\end{split}\right.
\end{equation}
where $a^{ij}(x)=F_{ij}(D^{2}u(x))$. Then we observe that
\begin{equation*}
\partial_{\tau(\alpha)}u>0\quad\mbox{in}\quad\Omega(u),\qquad\mbox{for}\quad\alpha\in[-\theta_0,+\theta_0].
\end{equation*}
Others, the strong maximum principle implies $\partial_{\tau(\alpha)}u\equiv 0$ in $\Omega(u)$, a contradiction.

By Remark \ref{re:4.1}, we may assume that (\ref{E6.3})-(\ref{E4.11}) holds. Let $T:=\{x_1,0,\cdots,0,x_n\}$. We denote by $D_{T}u_R(e_n)$ the projection of $Du_R(e_n)$ onto
the plane $T$. Let
\begin{equation*}
 \tau\big(\beta(e_n)\big):=D_{T}u_R(e_n)/|D_{T}u_R(e_n)|.
\end{equation*}
Since $\partial_{n}u_R(e_n)>0$, it is clear that $\angle\vartheta\Big(\tau\big(\beta(e_n)\big),e_n\Big)<\pi/2$, i.e.,
\begin{equation*}
\beta(e_n)\in(-\pi/2,\pi/2).
\end{equation*}

\indent\textit{Step 2. }
 We show that there exists a universal constant
$N>1$, depending only on $n,\lambda,\Lambda,\theta_0$ such that either
\begin{equation}\label{E4.80}
\partial_{n}u_R\leq N\partial_{\tau(-\theta_0)}u_R\quad\mbox{in}\quad B_1,
\end{equation}
or
\begin{equation}\label{E4.81}
\partial_{n}u_R\leq N\partial_{\tau(\theta_0)}u_R\quad\mbox{in}\quad B_1,
\end{equation}
for all $R>R_0$, where $R_0$ is given by (\ref{E4.11}).

\indent\textit{Step 2.1}
We verify that (\ref{E4.80}) holds provided $\beta(e_n)\in(-\pi/2,0]$.

 Let $\epsilon_R:=\partial_{n}u_R(e_n)>0$. Choosing $l=\tan\theta_0+1$, the interior Harnack inequality implies that
 \begin{equation*}
\partial_{n}u_R<C(n,\lambda,\Lambda,\theta_0)\epsilon_R\quad\mbox{on}\quad B'_{6l}(2e_n).
\end{equation*}
 Using the convexity of $u_R$, we find that
 \begin{equation}\label{E4.83}
\partial_{n}u_R<C_1(n,\lambda,\Lambda,\theta_0)\epsilon_R\quad\mbox{in}\quad\mathcal{C}_{6l, 2}(0).
\end{equation}
By $\beta(e_n)\in(-\pi/2,0]$, we observe that
 \begin{equation*}
\angle\vartheta\Big(\tau\big(\beta(e_n)\big),\tau(-\theta_0)\Big)\leq\max(\theta_0,\pi/2-\theta_0).
\end{equation*}
Denoting $\gamma:=\max(\theta_0,\pi/2-\theta_0)$, then we have
\begin{equation}\label{E4.85}
\partial_{\tau(-\theta_0)}u_R(e_n)\geq|D_{T}u_R(e_n)|\cos\gamma\geq\epsilon_R\cos\gamma.
\end{equation}

To prove (\ref{E4.80}),  we claim that
\begin{equation}\label{E4.84}
\partial_{n}u_R\leq N\partial_{\tau(-\theta_0)}u_R\quad\mbox{on}\quad\partial\mathcal{C}_{3l,1}(0).
\end{equation}
This claim can be seen as follows. Picking $y\in\partial\mathcal{C}_{3l,1}(0)$, since $|Du_R|=0$ on $\Omega^{c}(u_R)$, we may assume that $y\in\Omega(u_R)$.

For the case $y\in B'_{3l}(e_n)\cup(\partial B'_{3l}(0)\times(-1,1)),$ using (\ref{E3.32}), it is easily seen that $\Omega^{c}_2(u_R)\subseteq B'_{2l}(-2e_n)$ for all $R>R_0.$
Using $\partial_n u_R\geq 0$, we have
 \begin{equation*}
\Omega^{c}(u_R)\cap\{x_n>-2\}\subseteq B'_{2l}(0)\times[-2,0].
\end{equation*}
Hence dist$(y,\Omega^{c}(u_R))\geq 1$. Since $\partial_n u_R$ and $\partial_{\tau(-\theta_0)}u_R$ are solutions to (\ref{E6.10}), together with (\ref{E4.83})-(\ref{E4.85}), we obtain $\partial_{n}u_R(y)\leq N\partial_{\tau(-\theta_0)}u_R(y)$ provided $N>1$ sufficiently large, where we are using the Harnack inequality.

For the case $y(y',-1)\in B'_{3l}(-e_n)\cap\Omega(u_R)$, we can select $y_0(y'_0,-1)\in\partial\Omega^{c}_1(u_R)$ such that $|y-y_0|=\mbox{dist}(y,\Omega^{c}_1(u_R))$.

\end{equation*}

\end{equation*}
Applying Lemma \ref{lemma 4.0}, we conclude that
\begin{equation*}
w(t\nu'(y_0)+y_0)\geq 0,\quad\mbox{for}\quad t\in(0,3l).
\end{equation*}
Hence
\begin{equation*}
\frac{M+2}{C_2}\partial_{\tau(-\theta_0)}(u_R)(y)\geq\frac{1}{C_1}\partial_{n}(u_R)(y).
\end{equation*}
Taking $N(n,\lambda,\Lambda,\theta_0)=C_1(M+2)/C_2$, we obtain (\ref{E4.84}).

 (\ref{E4.80}) follows from the comparison principle.

 Repeating the previous argument, we get (\ref{E4.81}) provided $\beta(e_n)\in(0,\pi/2).$

 Applying \cite[Proposition 2.9]{Ca02} again, it follows that $v_{\infty}$ is a global solution to the fully nonlinear obstacle problem (\ref{E1.1}).

We claim that
 \begin{equation}\label{E4.100}
\partial_n v_{\infty}=0.
 \end{equation}
 This claim is justified as follows. Fixing $r>0$, by $\partial_{nn}u\geq 0$, we deduce that
 \begin{equation*}
\begin{split}
\partial_{n}u(\rho_k x',-k/2)\frac{k}{2}&\leq u(\rho_kx',0)-u(\rho_kx',-k/2)\\
&\leq u(\rho_kx',0)\leq C(n,\lambda,\Lambda)(r\rho_k)^2,\qquad\mbox{in}\quad B_{r},
\end{split}
\end{equation*}
 where we are use the interior $C^{1,1}$ estimate. Since
 \begin{equation*}
\rho_kx_n-k\leq -k/2,\quad\mbox{for}\quad k\gg 1,
 \end{equation*}
 choosing $k$ sufficiently large, we find that
\begin{equation*}
\begin{split}
\partial_{n}v_{k}(x)&=\rho_k^{-1}\partial_{n}u(\rho_kx',\rho_kx_n-k)\\
&\leq \rho_k^{-1}\partial_{n}u(\rho_kx',-k/2)\leq  C(n,\lambda,\Lambda)r^2(\rho_k/k),\qquad\mbox{in}\quad B_{r}.
\end{split}
 \end{equation*}
Letting $k\rightarrow+\infty$, then we obtain (\ref{E4.100}).

\indent\textit{Step 2.2 }
Let $v(x'):=v_{\infty}(x',0)$. We claim that $\Omega^{c}(v)\subseteq B'_1$.

This claim is justified as follows. By (\ref{E4.100}), we deduce that $v$ is a global solution to (\ref{EE4.21}).
Since $v$ is convex, as already noted in Corollary \ref{cor:4.3}, $\Omega^{c}(v)\subseteq B'_1$ is a direct consequence of
\begin{equation}\label{E5.62}
\mathrm{Int}\,\Omega^{c}(v)\subseteq B'_1.
\end{equation}

On the contrary, we suppose that (\ref{E5.62}) fails. Then there exists $P'\in\mathbb{R}^{n-1}$ such that
\begin{equation}\label{E5.63}
B'_{2\delta}(P')\subseteq\Omega^{c}(v)\backslash B'_1,
\end{equation}
where $\delta>0$ is a small constant. Picking $P_k(\rho_k P',-k)\in\mathbb{R}^{n}$, we have
\begin{equation*}
\mbox{dist}(P_k,\mathbb{R}e_n)\geq(1+2\delta)\rho_k>\mbox{diam}(\Omega^{c}_{k}(u)),\quad k=1,2,\cdots.
\end{equation*}
It is readily seen that $P_k\in\Omega(u)$. Let
\begin{equation*}
u(Q_k):=\sup_{B_{\delta\rho_k}(P_k)}u.
\end{equation*}
Using nondegeneracy, we have
\begin{equation*}
v_{k}\big(Q'_k/\rho_k,(Q_{k,n}+k)/\rho_k\big)=\rho_k^{-2}u(Q_k)\geq C(n,\lambda,\Lambda)\delta^2.
\end{equation*}

Notice that  $\big(Q'_k/\rho_k,(Q_{k,n}+k)/\rho_k\big)\in\overline{B_{\delta}}(P',0)$. Passing to a subsequence, we assume that
$Q'_k/\rho_k\rightarrow Q'\in\overline{B'_{\delta}}(P')$. By (\ref{E4.100}), we have
\begin{equation*}
v(Q')=\lim_{k\rightarrow+\infty}v_{k}(Q'_k/\rho_k,0)\geq C\delta^2.
\end{equation*}
This contradicts (\ref{E5.63}), as we wanted.

\indent\textit{Step 2.3 }
We verify that $\mathrm{Int}\,\Omega^{c}(v)\neq\emptyset$. Let us assume by contradiction that $\mathrm{Int}\,\Omega^{c}(v)=\emptyset$. Then $v$ is a quadratic polynomial.
 By $\Omega^{c}(v)\subseteq B'_1$, we observe that $\Omega^{c}(v)$ is a single point. Since the origin $0\in\Omega^{c}(u)$, using the monotonicity of $u$,
we deduce that $v(0)=0$, i.e., $\Omega^{c}(v)=\{0\}$.

On the other hand, we can select a sequence $M_k(M'_k,-k)\in\Omega^{c}_{k}(u)$ such that
\begin{equation*}
|M'_k|=\mbox{diam}\Omega^{c}_{k}(u)=\rho_k.
\end{equation*}
Up to a subsequence, we see that $(M'_k/\rho_k)\rightarrow M'$ with $|M'|=1$. It is not hard to see that $v(M')=0,$ i.e., $M'\in\Omega^{c}(v)$, which yields the desired contradiction.

\indent\textit{Step 2.4 }
Using $\Omega^{c}(v)\subseteq B'_1$, we deduce that
\begin{equation*}
v_r(x')\rightarrow\frac{1}{2}x'^TA^*_{n-1}x',\quad\mbox{as}\quad r\rightarrow +\infty,
\end{equation*}
where the matrix $A^*_{n-1}>0$. It follows that
\begin{equation}\label{E5.66}
F_{n-1}(A^*_{n-1})=F_{n-1}(A_{n-1})=1.
\end{equation}

Since $v(x')\in\mathcal{G}_c$, for $\epsilon>0$, we may choose $R(\epsilon)>1$ large such that
\begin{equation*}
v(x')\geq\frac{1}{2}x'^{T}\left(A^*_{n-1}-\frac{\epsilon}{2}I_{n-1}\right)x',\qquad\mbox{for}\quad |x'|>R(\epsilon),
\end{equation*}
where we use Theorem \ref{thm:5} (or (\ref{E1.6})). Fix $R>R(\epsilon)$. By (\ref{E5.60}), we deduce that
\begin{equation*}
\rho_k^{-2}u(\rho_k R e-ke_n)=v_k(Re)\geq\frac{1}{2}R^2e^T(A^*_{n-1}-\epsilon I)e,\quad\forall e\in\partial B'_1,
\end{equation*}
as $k\rightarrow+\infty.$ By the monotonicity of $u$, we have
\begin{equation*}
\rho_k^{-2}u(\rho_kR e-ke_n)\leq\rho_k^{-2}u(\rho_kR e)\leq\frac{1}{2}R^2e^T(A_{n-1}+\epsilon I)e,\quad\forall e\in\partial B'_1,
\end{equation*}
as $k\rightarrow+\infty.$ Then we conclude that
\begin{equation}\label{E4.107}
A^*_{n-1}\leq A_{n-1},
\end{equation}
by taking $\epsilon>0$ sufficiently small.

Together with (\ref{E5.66}) and (\ref{E4.107}), we obtain $A_{n-1}=A^*_{n-1}>0.$
