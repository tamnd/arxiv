<!-- conj-1: body_md -->
\label{conj:miller_vissuet}
All dihedral groups $D_n$ for $n \geq 3$ are product-dominant.
<!-- end conj-1 -->
<!-- defn-1: body_md -->
The $r$-th \textbf{raw moment} of $d(A)$ is defined by
\begin{align}
   \mu_r(G)\ :=\ \frac{1}{2^{|G|}}\sum_{A \subseteq G}(d(A))^r. 
\end{align}
For ease of notation, we write
\begin{align}
    M_r(G)\ :=\ \sum_{A \subseteq G}(d(A))^r\ =\ 2^{|G|}\mu_r(G).
\end{align}
Another useful statistic is the number of subsets which are \textbf{balanced}, meaning $d(A)=0$. Let
\begin{align}
    B(G)\ :=\ \# \{A \subseteq G: d(A)=0\}.
\end{align}
We occasionally write $M_r:=M_r(G)$ and $B:=B(G)$.
<!-- end defn-1 -->
<!-- thm-1: body_md -->
\label{thm:dom_ineq}
Let $G$ be a finite group, and suppose
\begin{align}
    \frac{M_1^2}{(2^{|G|}-B)M_2}\ >\ \frac 12. \label{eq:dom_ineq}
\end{align}
Then
\begin{enumerate}[(i)]
    \item $M_1>0 \ \implies\ G$ is product-dominant,
    \item $M_1<0\ \implies\ G$ is quotient-dominant.
\end{enumerate}
<!-- end thm-1 -->
<!-- defn-2: body_md -->
If $\{a_k\},\{b_k\}$ are two sequences, we say $a_k = o(b_k)$ if $\lim_{k \to \infty}{a_k}/{b_k}=0$.
<!-- end defn-2 -->
<!-- thm-2: body_md -->
\label{thm:mptqresult}
    Let $\{G_k\}$ be a sequence of finite groups with $|G_k| \to \infty$. Let $Z(G_k)$ denote the center of $G_k$, and define
    \begin{align}
        T_k\ &:=\ \# \{g \in G_k : \operatorname{ord}(g)=2 \}, \\
        Z_k\ &:=\ |Z(G_k)|, \\
        R_k\ &:=\ \#\{ g \in G_k : g^2 \in Z(G_k)\}.
    \end{align}
    Then, we have the following properties.
    \begin{enumerate}[(i)]
        \item If $Z_k = o(T_k)$, then $G_k$ is product-dominant for sufficiently large $k$.
        \item If $T_k\cdot \sqrt 3^{R_k/Z_k} = o(Z_k)$, then $G_k$ is quotient-dominant for sufficiently large $k$.
    \end{enumerate}
<!-- end thm-2 -->
<!-- proof-1: body_md -->
We prove statement (i), and the proof (ii) is nearly identical, hence omitted. To prove $G$ is product-dominant, we must show
\begin{align} \label{eq:mptq_prob_cond}
    P(d(A)>0)\ >\ P(d(A)<0),
\end{align}
where $P$ denotes the probability over a uniformly chosen random subset $A \subseteq G$.  We use the shorthand $P(d<0)$ and $P(d>0)$.
Let $\mu_1$ be the expected value of $d(A)$ and $\sigma^2$ be the variance. Since $M_1>0$ implies $\mu_1>0$, Cantelli's inequality \cite{cantelli} gives 
\begin{align}
    P(d\leq 0)\ =\ P(d - \mu_1\leq -\mu_1)\ \leq\ \frac{\sigma^2}{\sigma^2 +\mu_1^2}.
\end{align}
Hence
\begin{align}
    P(d>0)\ =\ 1-P(d\leq 0)\ \geq\ 1-\frac{\sigma^2}{\sigma^2 + \mu_1^2}.
\end{align}
On the other hand, let $\beta := P(d=0)=B/2^{|G|}$ be the proportion of balanced sets. Then,
\begin{align}
    P(d<0)\ &=\ 1-P(d\geq 0)\ =\ 1-P(d=0)-P(d>0) \notag
    \\ &=\ 1-\beta-P(d>0)\ \leq\ 1- \beta -\left ( 1-\frac{\sigma^2}{\sigma^2 + \mu_1^2} \right )  \notag
    \\ &=\ \frac{\sigma^2}{\sigma^2 + \mu_1^2}-\beta.
\end{align}
Therefore, from \eqref{eq:mptq_prob_cond} we see 
\begin{align} \label{eq:probv1}
    1-\frac{\sigma^2}{\sigma^2 + \mu_1^2} \ >\  \frac{\sigma^2}{\sigma^2  +\mu_1^2}-\beta \ \implies\ G \text{ is product-dominant}. 
\end{align}

Now, we may simplify the expression in \eqref{eq:probv1} by using the fact that $\sigma^2 + \mu_1^2 = \mu_2$ (see \cite{statproofbookSecondMoment}, for example). This gives
\begin{align}
1-\frac{\sigma^2}{\sigma^2 + \mu_1^2}\ >\  \frac{\sigma^2}{\sigma^2  +\mu_1^2}-\beta \ &\iff\ 1+\beta\ >\  \frac{2 \sigma^2}{\sigma^2 + \mu_1^2} \notag
    \\ &\iff 1 +\beta \ >\   \frac{2\mu_2 - 2\mu_1^2}{\mu_2} \notag
    \\ &\iff \frac{2\mu_1^2}{\mu_2}\ >\ 1 -\beta \notag
    \\ &\iff \frac{2M_1^2}{M_2} \ >\ 2^{|G|} -B \notag
    \\ &\iff \frac{M_1^2}{(2^{|G|}-B)M_2}\ >\ \frac 12,
\end{align}
as desired.

<!-- defn-3: body_md -->
\label{def:forb_1}
Let $G$ be a group. A \textbf{forbiddance graph} is a graph whose vertices are the elements of $G$, with an edge $x \to y$ whenever $x$ and $y$ satisfy a predetermined product or quotient relationship with respect to some fixed element(s) of $G$. In particular, for fixed $g \in G$, we define the forbiddance graphs $\mathcal F_P(g)$ and $\mathcal F_Q(g)$ by the following rules.
\begin{align}
    &\mathcal F_P(g) \text{ has } x \to y \text{ whenever } xy=g ,
    \\ &\mathcal F_Q(g) \text{ has } x \to y \text{ whenever } xy^{-1}=g .
\end{align}
<!-- end defn-3 -->
<!-- defn-4: body_md -->
For a directed or undirected graph $\Gamma=(V,E)$, a set $A\subseteq V$ is said to be an \textbf{independent set} if there are no $x,y \in A$ (not necessarily distinct) joined by an edge. The number of independent sets of $\Gamma$ is denoted $i(\Gamma)$.
<!-- end defn-4 -->
<!-- ex-1: body_md -->
Let $G = S_4$ and $g = (1\:2\:3)$. \Cref{fig:FP_S4} depicts $\mathcal F_P(g)$, with an example independent set boxed and colored in red.
<!-- end ex-1 -->
<!-- rmk-1: body_md -->
For a finite group $G$, and $g \in G$,
    \begin{align}
        i(\mathcal F_P(g))\ &=\ \# \{A \subseteq G: g \not \in AA\}, \label{eq:prod_indep_id}
    \\ i(\mathcal F_Q(g))\ &=\ \# \{A \subseteq G: g \not \in AA^{-1}\} .\label{eq:quot_indep_id}
    \end{align}
<!-- end rmk-1 -->
<!-- lem-1: body_md -->
\label{lem:M1_sum}
    For a finite group $G$, we have
    \begin{align}
        M_1(G)\ =\ \sum_{g \in G}\left ( i(\mathcal F_Q(g)) - i(\mathcal F_P(g)) \right ) . \label{eq:M1_sum}
    \end{align}
<!-- end lem-1 -->
<!-- proof-2: body_md -->
This follows from \eqref{eq:exp_breakdown}, \eqref{eq:prod_indep_id}, and \eqref{eq:quot_indep_id}.
<!-- end proof-2 -->
<!-- lem-2: body_md -->
\label{lem:disjoint_cycles}
    For any $g \in G$, both $\mathcal F_Q(g)$ and $\mathcal F_P(g)$ are disjoint unions of cycle graphs.
<!-- end lem-2 -->
<!-- proof-3: body_md -->
For $x \in G$, define
    \begin{align}
        q_g(x)\ :=\ q_g^{(1)}(x)\ :=\ g^{-1}x, \quad \quad &q_g^{(n)}(x)\ :=\ q_g(q_g^{(n-1)}(x)),
        \\ p_g(x)\ :=\ p_g^{(1)}(x)\ :=\ x^{-1}g, \quad \quad &p_g^{(n)}(x)\ :=\ p_g(p_g^{(n-1)}(x)).
    \end{align}
    Then $y=q_g(x)$ is the unique element satisfying $xy^{-1}=g$, and similarly $y=p_g(x)$ is the unique element satisfying $xy=g$. Thus, the forbiddance graph $\mathcal F_Q(g)$ locally around $x$ looks like
    \begin{align}
        x \ \to\ q_g^{(1)}(x)\ \to\ q_g^{(2)}(x)\ \to\ \cdots, \label{eq:FQ_path}
    \end{align}
    and this path must eventually loop back around to $x$, since $q_g$ is a bijection and $G$ is finite. The same holds for $p_g$. It follows that both forbiddance graphs are composed of disjoint cycles.
<!-- end proof-3 -->
<!-- lem-3: body_md -->
\label{lem:pgqg_forms}
    For any $n \geq 1$,
    \begin{align}
        q_g^{(n)}(x)\ =\ g^{-n}x. \label{eq:qg_form}
    \end{align}
    Moreover, 
    \begin{align}
        p_g^{(2n-1)}(x)\ &=\ g^{-n+1}x^{-1}g^{n}, \label{eq:pg_odd_form}
        \\ p_g^{(2n)}(x)\ &= \  g^{-n}xg^n .\label{eq:pg_even_form}
    \end{align}
<!-- end lem-3 -->
<!-- proof-4: body_md -->
We proceed by induction. The base case for $q_g^{(1)}$ is by definition, and then
\begin{align}
    q_g^{(n)}(x)\ =\ q_g^{(1)}(q_g^{(n-1)}(x))\ =\ g^{-1}(g^{-(n-1)}x)\ =\ g^{-n}x.
\end{align}
For $p_g$, the base case $p_g^{(1)}$ is by definition. Then
\begin{align}
    p_g^{(2n)}(x)\ &=\ p_g(p_g^{(2n-1)}(x))\ =\ p_g(g^{-n+1}x^{-1}g^n) \notag
    \\ &=\ (g^{-n+1}x^{-1}g^n)^{-1}g\ =\ g^{-n}xg^n,
\end{align}
and
\begin{align}
p^{(2n+1)}_g(x)\ &=\ p_g(p_g^{(2n)}(x))\ =\ p_g(g^{-n}xg^n) \notag
\\ &=\ (g^{-n}xg^n)^{-1}g\ =\ g^{-n}x^{-1}g^{n+1},
\end{align}
as desired.
<!-- end proof-4 -->
<!-- rmk-2: body_md -->
\label{rmk:lucas_forbiddance}
It is known that the number of independent sets of a cycle graph of length $n$ is the $n^{\text{th}}$ Lucas number\footnote{$L_1=1$, $L_2=3$, and $L_{n+2}=L_{n+1}+L_n$ for $n \geq 1$.} $L_n$, where a cycle of length $1$ is a single vertex with a loop and a cycle of length $2$ is a single edge. Moreover, if $\Gamma := \Gamma_1 \sqcup \Gamma_2$ is the disjoint union of two component graphs, then 
\begin{align} \label{eq:indeps_in_disj_graphs}
    i(\Gamma)\ =\ i(\Gamma_1)\cdot i(\Gamma_2).
\end{align}
See \cite{ZhaoGroups} for proofs. Thus, to find $i(\mathcal F_P(g))$ and $i(\mathcal F_Q(g))$, we must compute the lengths of their component cycles.
<!-- end rmk-2 -->
<!-- lem-4: body_md -->
\label{lem:FQ_form}
    For a finite group $G$, and $g \in G$,
    \begin{align}
        i(\mathcal F_{Q}(g))\ =\ L_{\operatorname{ord}(g)}^{|G|/\operatorname{ord}(g)}.
    \end{align}
<!-- end lem-4 -->
<!-- proof-5: body_md -->
For any element $x \in G$, the length of the cycle in $\mathcal F_Q(g)$ containing $x$ is precisely the minimal $\ell \geq 1$ such that $q_g^{(\ell)}(x) = x$. By \Cref{lem:pgqg_forms}, this equates to the minimal $\ell$ for which $g^{-\ell}x=x$, which is $\operatorname{ord} (g)$. Since every vertex in $\mathcal F_Q(g)$ is therefore contained in a cycle of length $\operatorname{ord}(g)$, it follows that $\mathcal F_Q(g)$ is precisely composed of $|G|/\operatorname{ord}(g)$ cycles of length $\operatorname{ord} (g)$, so the claim follows by \Cref{rmk:lucas_forbiddance} and \eqref{eq:indeps_in_disj_graphs}.
<!-- end proof-5 -->
<!-- lem-5: body_md -->
\label{lem:lucas_asym}
    Let $N$ be a positive integer, and let $a_1,\dots,a_j$ be positive integers with $a_i\neq 2$ for all $i$, and $a_1+\cdots + a_j = N$. Then
    \begin{align}
        L_{a_1}\cdot L_{a_2}\cdots L_{a_j}\ \leq \ 7^{N/4}.
    \end{align}
<!-- end lem-5 -->
<!-- proof-6: body_md -->
Let $\phi = (1+\sqrt 5)/2$ be the golden ratio. It is well known that
\begin{align}
    L_n\ =\ \phi^n + (1-\phi)^n,
\end{align}
where notably $|1-\phi|<1$, so $L_n \sim \phi^n$. Moreover, since $1-\phi <0$, one sees that $L_n > \phi^n \iff n$ is even. Thus, as $L_4=7$, and $(1-\phi)^n < (1-\phi)^4$ for all $n>4$, we get that
\begin{align}
    L_n\ \leq\ \left ( 7^{1/4} \right ) ^n
\end{align}
for all $n \geq 4$. This inequality also holds for $L_1$ and $L_3$. As $a_i \neq 2$ for all $i$, we have
\begin{align}
    \prod_{i=1}^jL_{a_i}\ \leq \ \prod_{i=1}^j 7^{a_i/4}\ =\ 7^{N/4}.
\end{align}
<!-- end proof-6 -->
<!-- cor-1: body_md -->
\label{cor:FQ_asym}
    Let $G$ be a finite group of size $N$, and let $T$ be the number of elements of $G$ with order $2$. Then
    \begin{align}
        T \cdot \sqrt 3^{N}\ \leq\ \sum_{g \in G}i(\mathcal F_Q(g))\ \leq\ T \cdot \sqrt 3^{N} + (N-T)\cdot 7^{N/4}. \label{eq:FQ_asym}
    \end{align}

By \Cref{lem:FQ_form},
    \begin{align}
        \sum_{g \in G}i(\mathcal F_Q(g))\ &=\ \sum_{g \in G}L_{\operatorname{ord}(g)}^{N/\operatorname{ord}(g)}\ \notag 
        \\ &=\ \, \sum_{\mathclap{\substack{g \in G \\ \operatorname{ord}(g)=2}}} \:L_2^{N/2} + \: \: \sum_{\mathclap{\substack{g \in G \\ \operatorname{ord} (g) \neq 2}}} \: L_{\operatorname{ord}(g)}^{N/\operatorname{ord}(g)} \notag

    \end{align}

<!-- lem-6: body_md -->
\label{lem:FP_asym}
    Let $G$ be a finite group of size $N$ with center $Z(G)$. Let $Z:=|Z(G)|$, and let
    \begin{align}
        R\ :=\ \#\{x \in G: x^2 \in Z(G)\}.
    \end{align}
    Then
    \begin{align}
       Z\cdot \sqrt 3^{N-R/Z} \ \leq \ \sum _{g \in G}i(\mathcal F_P(g)) \ \leq\ Z\cdot \sqrt 3^N + (N-Z)\cdot 1.68^N. \label{eq:FP_asym}
    \end{align}
    Moreover, for every $g \not \in Z(G)$,
    \begin{align}
        i(\mathcal F_P(g))\ \leq\ \left ( 3^{1/4}\cdot 7^{1/8} \right ) ^N\ <\ 1.68^N. \label{eq:noncentral_P}
    \end{align}
<!-- end lem-6 -->
<!-- proof-8: body_md -->
Suppose $g \in Z(G)$. By \eqref{eq:pg_even_form}, $p_g^{(2)}(x) = g^{-1}xg$. Thus, as $gx=xg$, we know every $x\in G$ is contained in either a $1$-cycle or a $2$-cycle in $\mathcal F_P(g)$ by the proof of \Cref{lem:disjoint_cycles}. In particular, $x$ is contained in a $2$-cycle if and only if $p_g^{(1)}(x)\neq x$, which is equivalent to $x^2\neq g$. Let 
\begin{align}
    r(g)\ :=\ \# \{x \in G: x^2=g\}.
\end{align}
Then, the forbiddance graph $\mathcal F_P(g)$ is composed of $r(g)$ $1$-cycles and $(N-r(g))/2$ two-cycles. Therefore, for any $g \in Z(G)$,
\begin{align}
    i(\mathcal F_P(g))\ =\ L_1^{r(g)}\cdot L_2^{\frac{N-r(g)}{2}}\ =\ \sqrt 3^{N-r(g)}.
\end{align}
On the other hand, suppose $g \not \in Z(G)$. By \eqref{eq:pg_even_form}, $p_g^{(2)}(x)=x$ if and only if $xg=gx$, so every vertex lying in a $2$-cycle of $\mathcal F_P(g)$ belongs to the centralizer $C_G(g)$. Let $k$ be the number of such vertices. As $g$ is non-central, $C_G(g)$ is a proper subgroup of $G$, hence $k \leq |C_G(g)|\leq N/2$. The $2$-cycles contribute $L_2^{k/2}=\sqrt 3^{\,k}$ independent sets, while the remaining cycles have lengths different from $2$ summing to $N-k$, so by \Cref{lem:lucas_asym} they contribute at most $7^{(N-k)/4}$. Since $\sqrt 3 > 7^{1/4}$, the resulting bound increases with $k$, and therefore
\begin{align}
    i(\mathcal F_P(g))\ \leq\ \sqrt 3^{\,k}\cdot 7^{(N-k)/4}\ \leq\ \sqrt 3^{\,N/2}\cdot 7^{N/8}\ =\ \left ( 3^{1/4}\cdot 7^{1/8} \right ) ^N\ <\ 1.68^N,
\end{align}
which proves \eqref{eq:noncentral_P}. Splitting the sum over the central and non-central elements of $G$ now gives
\begin{align}
    \sum_{g \in G}i(\mathcal F_P(g))\ &=\ \sum_{g \in Z(G)}\sqrt 3^{N-r(g)} + \sum_{g \not \in Z(G)}i(\mathcal F_P(g)). \label{eq:FP_true_sum}
\end{align}
Since $\sqrt 3^{N-r(g)}\leq \sqrt 3^N$, and since each of the $N-Z$ non-central elements contributes at most $1.68^N$ by \eqref{eq:noncentral_P}, we have
\begin{align}
    \sum_{g \in G}i(\mathcal F_P(g))\ \leq\ Z\cdot \sqrt 3^N + (N-Z)\cdot 1.68^N,
\end{align}
which is the second inequality in \eqref{eq:FP_asym}. To obtain the lower bound, we discard the non-negative second sum in \eqref{eq:FP_true_sum} and observe that $f(x):=\sqrt 3^{N-x}$ is a convex function. Therefore, by Jensen's inequality \cite{jensen1906, stein2005real},
\begin{align}
    \frac{1}{Z}\sum_{g \in Z(G)}\sqrt 3^{N-r(g)}\ \geq\ \sqrt 3^{ N-(1/Z)\sum_{g \in Z(G)}r(g)}\ =\ \sqrt 3^{N-R/Z}, \label{eq:jensen}
\end{align}
since $\sum_{g \in Z(G)}r(g) = R$. Thus, by \eqref{eq:FP_true_sum},
\begin{align}
    \sum_{g \in G}i(\mathcal F_P(g))\ \geq\ Z\cdot \sqrt 3^{N-R/Z}
\end{align}
as desired.

<!-- lem-7: body_md -->
\label{lem:2nd_mom}
    Recall $M_2(G):= \sum_{A \subseteq G}(d(A))^2$. For any finite group $G$,
    \begin{align}
        M_2(G)\ =\ \sum_{g,h \in G}\left [ i(\mathcal F_{PP}(g,h))+ i(\mathcal F_{QQ}(g,h))- 2 i(\mathcal F_{PQ}(g,h)) \right ] .
    \end{align}

<!-- proof-9: body_md -->
We proceed similarly to the first moment. Let $p(A) := |AA|$ and $q(A):= |AA^{-1}|$. Then
\begin{align}
    M_2(G)\ :=&\ \sum_{A \subseteq G}(p(A) - q(A))^2\ =\ \sum_{A \subseteq G}\left [ p(A)^2+q(A)^2 - 2p(A)q(A) \right ] . \label{eq:binomvar}
\end{align}
Now, note that
\begin{align}
    p(A)^2\ =&\ \ \left ( \sum_{x}\mathbb 1_{AA}(x) \right ) ^2 = \sum_{g,h}\mathbb 1_{AA}(g)\mathbb 1_{AA}(h),
    \\ q(A)^2\ =&\ \ \left ( \sum_{x}\mathbb 1_{AA^{-1}}(x) \right ) ^2\ =\ \sum_{g,h}\mathbb 1_{AA^{-1}}(g)\mathbb 1_{AA^{-1}}(h),
    \\ p(A)q(A)\ =&\ \ \left ( \sum_{g}\mathbb 1_{AA}(g) \right ) \left ( \sum_{h}\mathbb 1_{AA^{-1}}(h) \right ) \ =\ \sum_{g,h}\mathbb 1_{AA}(g) \mathbb 1_{AA^{-1}}(h).
\end{align}
Thus, we rewrite \eqref{eq:binomvar} as 
\begin{align}
    &\ \ \sum_{A \subseteq G} \sum_{g,h}\left [ \mathbb 1_{AA}(g)\mathbb 1_{AA}(h) + \mathbb 1_{AA^{-1}}(g)\mathbb 1_{AA^{-1}}(h) - 2\cdot \mathbb 1_{AA}(g) \mathbb 1_{AA^{-1}}(h) \right ]  \notag
    \\ =&\ \ \sum_{g,h}\sum_{A \subseteq G}\left [ \mathbb 1_{AA}(g)\mathbb 1_{AA}(h) + \mathbb 1_{AA^{-1}}(g)\mathbb 1_{AA^{-1}}(h) - 2\cdot \mathbb 1_{AA}(g) \mathbb 1_{AA^{-1}}(h) \right ]  \notag
    \\ =&\ \ \sum_{g,h}\left [  \# \{ A: g,h \in AA\} + \#\{A: g, h \in AA^{-1}\} -2 \# \{ A: g \in AA, h \in AA^{-1}\}  \right ] . \label{eq:varandsets}
\end{align}

Now, to obtain the expressions in \Cref{rmk:two_elm_indep}, we use the inclusion--exclusion principle to obtain
\begin{align}
    \# \{ A: g,h \in AA\}\ \ =&\ \ \ 2^{|G|}-\# \{ A: g\not \in AA \text{ or } h \not \in AA\}  \notag
    \\ =&\ \ \ 2^{|G|}-\#\{A: g \not \in AA\} - \#\{A: h \not \in AA\} \notag
    \\ &\ \ +\ \#\{ A: g \not \in AA \text{ and } h \not \in AA\} \notag
    \\ =&\ \ \ 2^{|G|}-i(\mathcal F_{P}(g)) - i(\mathcal F_{P}(h))  + i(\mathcal F_{PP}(g,h)), \label{eq:varprod}
\end{align}
and similarly
\begin{align}
    \# \{ A:g,h\in AA^{-1}\}\ =&\ \ 2^{|G|}-i(\mathcal F_Q(g)) - i(\mathcal F_Q(h))  + i(\mathcal F_{QQ}(g,h)), \label{eq:varquot}
    \\ \# \{A: g \in AA,h \in AA^{-1}\}\ =&\ \ 2^{|G|} - i(\mathcal F_P(g)) - i(\mathcal F_Q(h)) + i(\mathcal F_{PQ}(g,h)). \label{eq:varboth}
\end{align}
Plugging these into \eqref{eq:varandsets} yields (after rearranging and canceling)
\begin{align}
    M_2(G)\ =\ \ &\sum_{g,h}\left [ i(\mathcal F_P(g))- i(\mathcal F_P(h))+ i(\mathcal F_Q(h))-i(\mathcal F_Q(g)) \right ]  \label{eq:varsingles} 
    \\ &+\ \sum_{g,h}\left [ i(\mathcal F_{PP}(g,h))+ i(\mathcal F_{QQ}(g,h))- 2 i(\mathcal F_{PQ}(g,h)) \right ] . \label{eq:vardoubles}
\end{align}
Notice that the entire sum in \eqref{eq:varsingles} cancels, and so we are left with
\begin{align}
    M_2(G)\ =\ \sum_{g,h}\left [ i(\mathcal F_{PP}(g,h))+ i(\mathcal F_{QQ}(g,h))- 2 i(\mathcal F_{PQ}(g,h)) \right ] , \label{eq:M2_final}
\end{align}
as desired.

To find an upper bound for the expression in \eqref{eq:zhao_strong_indep}, note that
    \begin{align}
        \log \left ( \prod_{uv \in E} \left ( 2^{d_u}+2^{d_v}-1 \right ) ^{\frac{1}{d_ud_v}}  \right ) \ =\ \sum_{uv \in E}\frac{\log(2^{d_u}+2^{d_v}-1)}{d_ud_v}.
    \end{align}
    We wish to use the graph-theoretic identity
    \begin{align}
        \sum_{uv \in E}\left ( \frac{1}{d_u}+\frac{1}{d_v} \right ) \ =\ |V|,
    \end{align}
    so we want the minimal constant $c$ with
    \begin{align}
        \frac{\log(2^{d_u}+2^{d_v}-1)}{d_ud_v} \ \leq \ c\left ( \frac{1}{d_u}+\frac{1}{d_v} \right ) ,
    \end{align}
    which is equivalent to
    \begin{align}
        \frac{\log(2^{d_u}+2^{d_v}-1)}{d_u+d_v}\ \leq\ c. \label{eq:logfraclessc}
    \end{align}
    Since $d_u,d_v \in \{2,3,4\}$, one can compute that the maximum of the left-hand side of \eqref{eq:logfraclessc} is attained at $\{d_u,d_v\}=\{2,4\}$, so we set $c=\log(19)/6$. Then
    \begin{align}
        \sum_{uv \in E}\frac{\log(2^{d_u}+2^{d_v}-1)}{d_ud_v}\ \leq\ \frac{\log(19)}{6}\sum_{uv \in E}\left ( \frac{1}{d_u}+\frac{1}{d_v} \right ) \ =\ \frac{\log (19)}{6}|V|,

    \begin{align}
        i(\Gamma)\ \leq\ \exp\left (  \sum_{uv \in E}\frac{\log (2^{d_u}+2^{d_v}-1)}{d_ud_v} \right ) \ \leq\  \exp\left ( \frac{\log (19)}{6}|V| \right ) \ =\ 19^{|V|/6},
    \end{align}
    as desired.

\label{lem:double_bound}

We first remark on three facts about independent sets. First, a self-adjacent vertex lies in no independent set, so deleting all self-adjacent vertices (and the edges attached to them) does not change the number of independent sets. Second, adding edges to a graph on a fixed vertex set can only decrease the number of independent sets. In particular, since $\mathcal F_P(g)$ and $\mathcal F_P(h)$ are subgraphs of $\mathcal F_{PP}(g,h)$ on the same vertex set,
\begin{align}
    i(\mathcal F_{PP}(g,h))\ \leq\ \min\left \{ i(\mathcal F_P(g)),\, i(\mathcal F_P(h)) \right \} . \label{eq:PP_le_P}
\end{align}
Third, for any directed graph $\mathcal F$, we define the underlying undirected graph $\Gamma$ to have the same vertex set as $\mathcal F$, where each directed edge of $\mathcal F$ becomes an undirected edge, and two or more directed edges between the same pair of vertices are identified as a single undirected edge; see \Cref{fig:directed_undirected}.

Then it is clear that $i(\mathcal F)=i(\Gamma)$. Moreover, if $\mathcal F$ has no self-adjacent vertices, then $\Gamma$ is a simple, undirected graph.


We now prove \eqref{eq:FQQ_bound}. If $g=e$ or $h=e$, then every $x \in G$ is self-adjacent in $\mathcal F_{QQ}(g,h)$, since $xx^{-1}=e$, so $i(\mathcal F_{QQ}(g,h))=1$ and the claim follows trivially. Otherwise if $g,h \neq e$, then no vertex is self-adjacent. Let $\Gamma_{QQ}$ denote the simple, undirected graph corresponding to $\mathcal F_{QQ}(g,h)$. The neighbors of $x$ in $\Gamma_{QQ}$ are the elements $\{g^{-1}x,\ h^{-1}x,\ gx,\ hx\}$. These are all different from $x$, and $g^{-1}x \neq h^{-1}x$, so every vertex has degree $2$, $3$, or $4$. Thus \eqref{eq:FQQ_bound} follows from \Cref{cor:Zhao_strong_indep}.

To prove \eqref{eq:FPP_bound}, we first assume $g \not \in Z(G)$ (the case $h \not \in Z(G)$ is identical). Then $i(\mathcal F_P(g)) \leq \left ( 3^{1/4}\cdot 7^{1/8} \right ) ^N$ by \eqref{eq:noncentral_P}, and so \eqref{eq:FPP_bound} follows from \eqref{eq:PP_le_P}.

Finally, suppose $g,h \in Z(G)$. We then get $xy=g$ if and only if $yx=g$, and the unique $y$ with $xy=g$ is $p_g(x) := x^{-1}g$. The map $p_g$ is an involution, since $p_g(p_g(x)) = g^{-1}xg = x$. Hence, in the underlying undirected graph $\Gamma_{PP}$ of $\mathcal F_{PP}(g,h)$, the neighbors of $x$ are exactly $p_g(x)$ and $p_h(x)$, and these are distinct because $g \neq h$. The vertex $x$ is self-adjacent if and only if 
$x \in S := \{x \in G : x^2 \in \{g,h\}\}$. Let $s := |S|$, and let $\Gamma'$ be the simple graph obtained from $\Gamma_{PP}$ by deleting $S$, so that $\Gamma'$ is simple, undirected, and $i(\mathcal F_{PP}(g,h)) = i(\Gamma')$.



Every vertex of $\Gamma_{PP}$ outside $S$ has exactly two neighbors, both different from itself, while every vertex in $S$ has exactly one neighbor different from itself. In particular, $\Gamma'$ has maximum degree at most $2$, so it is a disjoint union of cycles of lengths $c_1,\dots,c_a \geq 3$ and paths with $n_1,\dots,n_b \geq 1$ vertices, where
\begin{equation}
    \sum_i c_i + \sum_j n_j = N-s.
\end{equation}
Consider a path component $\Pi$ with $n$ vertices. If $n>1$, the endpoints of $\Pi$ have degree $1$ as vertices in $\Gamma'$, but degree $2$ as vertices in $\Gamma_{PP}$, so the neighbor each endpoint loses lies in $S$. If $n=1$, then $\Pi$ is a single vertex $x \not \in S$, which has two distinct neighbors in $\Gamma_{PP}$, and both lie in $S$. In either case, each path component of $\Gamma'$ accounts for two elements of $S$, and these elements are distinct for distinct path components, since every element of $S$ has exactly one neighbor other than itself. Therefore $2b \leq s$. Therefore, by \Cref{lem:lucas_asym}, \Cref{rmk:lucas_forbiddance}, and \Cref{lem:path_bound},
\begin{align}
    i(\mathcal F_{PP}(g,h))\ =\ \prod_{i=1}^a L_{c_i}\cdot \prod_{j=1}^b F_{n_j+2} \ &\leq\ 7^{\sum_i c_i/4}\cdot \prod_{j=1}^b \left ( 2\cdot 7^{(n_j-1)/4} \right )  \notag
    \\ &=\ 7^{(N-s)/4}\cdot \left ( 2\cdot 7^{-1/4} \right ) ^b \notag
    \\ &\leq\ 7^{(N-s)/4}\cdot \left ( 2\cdot 7^{-1/4} \right ) ^{s/2} \notag
    \\ &=\ 7^{N/4}\cdot\left ( \sqrt 2\cdot 7^{-3/8} \right ) ^s\ \leq\ 7^{N/4}, \label{eq:central_PP_chain}
\end{align}
where we used $2 \cdot 7^{-1/4} > 1$ and $\sqrt 2 < 7^{3/8}$. This proves \eqref{eq:FPP_central}. Since $7^{1/4} < 3^{1/4}\cdot 7^{1/8}$, this also completes the proof of \eqref{eq:FPP_bound}.

Let $N_k := |G_k|$. By \Cref{thm:dom_ineq}, to prove that $G_k$ is product- or quotient-dominant for sufficiently large $k$, it suffices to show that
\begin{align}
    \frac{1}{2}\ <\ \frac{(M_1(G_k))^2}{(2^{N_k}-B(G_k))\cdot (M_2(G_k))} \label{eq:dom_goal}

for all sufficiently large $k$, and to determine the sign of $M_1(G_k)$.

\begin{lem} \label{lem:PQPPQQ}
    Let
    \begin{align}
    P_k\ :=\ \sum_{g \in G_k}i(\mathcal F_{P}(g)), \quad &\quad \quad Q_k \ :=\ \sum_{g \in G_k}i(\mathcal F_Q(g)),
    \\ PP_k\ :=\ \sum_{g\neq h \in G_k}i(\mathcal F_{PP}(g,h)), \quad &\quad \quad QQ_k\ :=\ \sum_{g\neq h \in G_k}i(\mathcal F_{QQ}(g,h)).
\end{align}
Then
\begin{align}
    M_1(G_k)\ &=\ Q_k-P_k, \label{eq:M1_in_terms}
    \\ M_2(G_k)\ &\leq\ Q_k+P_k+QQ_k+PP_k, \label{eq:M2_in_terms}
    \\ 2^{N_k}-B(G_k)\ &\leq\ Q_k+P_k. \label{eq:B_in_terms}
\end{align}
\end{lem}

\begin{proof}
    We immediately obtain \eqref{eq:M1_in_terms} from \Cref{lem:M1_sum}, and \eqref{eq:M2_in_terms} was shown in \eqref{eq:M2_less_singles} and \eqref{eq:M2_less_doubles}. Finally, \eqref{eq:B_in_terms} holds for any finite group $G$, as
    \begin{align}
    2^{|G|}-B(G)\ &=\ \ 2^{|G|} - \# \{ A \subseteq G:d(A)=0\} \notag
    \\ & \leq\ \ 2^{|G|} -\# \{ A: AA=AA^{-1}=G\} \notag
    \\ &=\ \ \# \{ A: \exists g \not \in AA \text{ or } \exists h \not \in AA^{-1}\} \notag
    \\ &\leq\ \ \sum_{g \in G}\# \{ A: g \not \in AA \text{ or } g \not \in AA^{-1}\} \notag
    \\ &=\ \ \sum _{g \in G}\left ( i(\mathcal F_{P}(g)) + i(\mathcal F_{Q}(g)) - i (\mathcal F_{PQ}(g,g)) \right )  \notag
    \\ &\leq\ \ \sum_{g \in G}\left ( i(\mathcal F_P(g)) + i (\mathcal F_{Q}(g)) \right ) .
\end{align}
\end{proof}

We reformulate the desired inequality in \eqref{eq:dom_goal} as a statement about $P_k,Q_k,PP_k$, and $QQ_k$.

\begin{lem} \label{lem:uv_criterion}
    Let $G$ be a finite group, and let $P,Q,PP,QQ$ be as in \Cref{lem:PQPPQQ}, formed with $G$ in place of $G_k$. Suppose $P \neq Q$, and set
    \begin{align}
        X\ :=\ \max\{P,Q\},\quad u\ :=\ \frac{\min\{P,Q\}}{X}, \quad v\ :=\ \frac{PP+QQ}{X}.
    \end{align}
    If
    \begin{align}
        f(u,v)\ :=\ u^2-6u+1-v(1+u)\ >\ 0, \label{eq:f_criterion}
    \end{align}
    then $G$ is product-dominant when $Q>P$, and quotient-dominant when $P>Q$. Moreover, the following hold.
    \begin{enumerate}[(i)]
        \item The condition \eqref{eq:f_criterion} is equivalent to
        \begin{align}
            2(Q-P)^2\ >\ (Q+P)\left ( Q+P+PP+QQ \right ) . \label{eq:f_product_form}
        \end{align}
        \item The function $f$ is strictly decreasing in $u$ on $[0,3)$ and in $v$ on $[0,\infty)$. Hence, if $u \leq u^+ < 3$ and $v \leq v^+$, then $f(u,v) \geq f(u^+,v^+)$, so it suffices to verify \eqref{eq:f_criterion} for any such upper bounds $u^+$ and $v^+$.
        \item The condition \eqref{eq:f_criterion} holds whenever $u \leq 1$ and $6u+2v<1$.
    \end{enumerate}
\end{lem}

\begin{proof}
    By \eqref{eq:M1_in_terms}, we have 
    \begin{align}
        |M_1(G)| &= |Q-P| = X(1-u),
        \\ 2^{|G|}-B(G)\ &\leq\ Q+P\ =\ X(1+u),
        \\ M_2(G)\ &\leq\ Q+P+QQ+PP\ =\ X(1+u+v).
    \end{align}
Thus,
    \begin{align}
        \frac{(M_1(G))^2}{(2^{|G|}-B(G))\cdot M_2(G)}\ \geq\ \frac{X^2(1-u)^2}{X(1+u)\cdot X(1+u+v)}\ =\ \frac{(1-u)^2}{(1+u)(1+u+v)}. \label{eq:uv_ratio}
    \end{align}
    Since $(1+u)(1+u+v)>0$, we have
    \begin{align}
        \frac{(1-u)^2}{(1+u)(1+u+v)}\ >\ \frac 12\ &\iff\ 2(1-u)^2\ >\ (1+u)(1+u+v) \notag
        \\ &\iff\ u^2-6u+1-v(1+u)\ >\ 0,
    \end{align}
    so \eqref{eq:f_criterion} gives \eqref{eq:dom_ineq}. The conclusion now follows from \Cref{thm:dom_ineq}, as $M_1(G)=Q-P$ is positive exactly when $Q>P$.

    For (i), multiply \eqref{eq:f_criterion} through by $X^2>0$ and use $X(1-u) = |Q-P|$, $X(1+u) = Q+P$, and $Xv = PP+QQ$, which turns $2(1-u)^2 > (1+u)(1+u+v)$ into \eqref{eq:f_product_form}. For (ii), let $u \in [0,3)$ and $v \in [0,\infty)$, so
    \begin{align}
        \frac{\partial f}{\partial u}\ =\ 2u-6-v\ <\ 0, \qquad  \frac{\partial f}{\partial v}\ =\ -(1+u)\ <\ 0.
    \end{align}
    For (iii), note that $u^2 \geq 0$ and $1+u \leq 2$, so $f(u,v) \geq 1-6u-2v$.
\end{proof}

We next collect the bounds from \Cref{sec:M1,sec:M2} in a common form. Let $G$ be a finite group of order $N$, let $T$, $Z$, and $R$ be as in \Cref{thm:mptqresult} (formed with $G$ in place of $G_k$), and let $P,Q,PP,QQ$ be as in \Cref{lem:PQPPQQ}. Since $7^{1/4}<1.68$, \Cref{cor:FQ_asym} and \Cref{lem:FP_asym} give
\begin{align}
    T\sqrt 3^{N}\ &\leq\ Q\ \leq\ T\sqrt 3^N + N\cdot 1.68^N, \label{eq:Q_explicit}
    \\ Z \sqrt 3^{N-R/Z}\ &\leq\ P\ \leq\ Z\sqrt 3^N + N\cdot 1.68^N. \label{eq:P_explicit}
\end{align}

\begin{align}
    PP+QQ\ \leq\ 2N(N-1)\cdot 1.68^N\ \leq\ 2N^2\cdot 1.68^N. \label{eq:double_explicit}
\end{align}
Finally, as $1.68<\sqrt 3$, we remark that the quantity
\begin{align}
    \varepsilon(N)\ :=\ N^3\left ( \frac{1.68}{\sqrt 3} \right ) ^N \label{eq:eps_def}
\end{align}
satisfies $\varepsilon(N)\to 0$ as $N \to \infty$. The error terms obtained below are controlled by this $\varepsilon(N)$.

To prove statement (i) of the theorem, suppose $Z_k = o(T_k)$. As $Z_k \geq 1$ for every $k$, this forces $T_k \to \infty$; in particular $T_k \geq 1$ for all large $k$, so $Q_k \geq T_k\sqrt 3^{N_k}>0$ by \eqref{eq:Q_explicit}. Dividing the upper bound in \eqref{eq:P_explicit} by $T_k \sqrt 3^{N_k}$ gives
\begin{align}
    \frac{P_k}{Q_k}\ \leq\ \frac{Z_k\sqrt 3^{N_k}+N_k\cdot 1.68^{N_k}}{T_k\sqrt 3^{N_k}}\ \leq\ \frac{Z_k}{T_k}+N_k\left ( \frac{1.68}{\sqrt 3} \right ) ^{N_k}\ \leq\ \frac{Z_k}{T_k}+\varepsilon(N_k), \label{eq:case1_u}
\end{align}
and \eqref{eq:double_explicit} gives
\begin{align}
    \frac{PP_k+QQ_k}{Q_k}\ \leq\ \frac{2N_k^2\cdot 1.68^{N_k}}{T_k\sqrt 3^{N_k}}\ \leq\ 2\varepsilon(N_k). \label{eq:case1_v}
\end{align}
The right-hand sides of \eqref{eq:case1_u} and \eqref{eq:case1_v} tend to $0$, since $Z_k/T_k \to 0$ by assumption and $\varepsilon(N_k)\to 0$. Hence, for all sufficiently large $k$, we have $P_k<Q_k$ and
\begin{align}
    6\cdot \frac{P_k}{Q_k} + 2\cdot \frac{PP_k+QQ_k}{Q_k}\ <\ 1 .
\end{align}
Applying \Cref{lem:uv_criterion} with $X=Q_k$, $u = P_k/Q_k$, and $v = (PP_k+QQ_k)/Q_k$, we conclude that $G_k$ is product-dominant for all sufficiently large $k$.

Now, for case (ii) of the theorem, suppose $T_k\cdot \sqrt 3^{R_k/Z_k} = o(Z_k)$, and set
\begin{align}

\end{align}

\begin{align}
    \sqrt 3^{R_k/Z_k}\ \leq\ N_k \label{eq:sqrt3_le_N}
\end{align}
for all sufficiently large $k$. If $T_k \geq 1$, then 
\begin{align}
    \sqrt 3^{R_k/Z_k}\  \leq\  T_k \sqrt 3^{R_k/Z_k} \ <\  Z_k \ \leq\ N_k
\end{align}
for large $k$, by the assumption $T_k\sqrt 3^{R_k/Z_k}=o(Z_k)$. If instead $T_k=0$, then $G_k$ has no element of order $2$, and therefore has odd order by Cauchy's theorem. In that case, if $x^2 \in Z(G_k)$, then $\overline x^2 = \overline e$ in the quotient group $G_k / Z(G_k)$, which also has odd order, so $\overline x = \overline e$ and $x \in Z(G_k)$. Thus $R_k = Z_k$ and $\sqrt 3^{R_k/Z_k} = \sqrt 3 \leq N_k$. This proves \eqref{eq:sqrt3_le_N}, and combining it with \eqref{eq:Wk_def} yields
\begin{align}
    W_k\ \geq\ \frac{Z_k\sqrt 3^{N_k}}{N_k}\ \geq\ \frac{\sqrt 3^{N_k}}{N_k}. \label{eq:Wk_lower}
\end{align}
Using \eqref{eq:Q_explicit}, \eqref{eq:Wk_def}, and \eqref{eq:Wk_lower}, we obtain
\begin{align}
    \frac{Q_k}{P_k}\ \leq\ \frac{T_k\sqrt 3^{N_k}+N_k\cdot 1.68^{N_k}}{W_k}\ \leq\ \frac{T_k\cdot \sqrt 3^{R_k/Z_k}}{Z_k} + N_k^2\left ( \frac{1.68}{\sqrt 3} \right ) ^{N_k}\ \leq\ \frac{T_k\cdot \sqrt 3^{R_k/Z_k}}{Z_k}+\varepsilon(N_k), \label{eq:case2_u}
\end{align}
where the middle inequality uses $T_k\sqrt 3^{N_k}/W_k = T_k\sqrt 3^{R_k/Z_k}/Z_k$. Similarly, \eqref{eq:double_explicit} and \eqref{eq:Wk_lower} give
\begin{align}
    \frac{PP_k+QQ_k}{P_k}\ \leq\ \frac{2N_k^2\cdot 1.68^{N_k}}{W_k}\ \leq\ 2N_k^3\left ( \frac{1.68}{\sqrt 3} \right ) ^{N_k}\ =\ 2\varepsilon(N_k). \label{eq:case2_v}
\end{align}
The right-hand sides of \eqref{eq:case2_u} and \eqref{eq:case2_v} tend to $0$, since $T_k \sqrt 3^{R_k/Z_k}/Z_k \to 0$ by assumption and $\varepsilon(N_k) \to 0$. Hence, for all sufficiently large $k$, we have $Q_k<P_k$ and
\begin{align}
    6\cdot \frac{Q_k}{P_k} + 2\cdot \frac{PP_k+QQ_k}{P_k}\ <\ 1.
\end{align}
Applying \Cref{lem:uv_criterion} with $X=P_k$, $u = Q_k/P_k$, and $v = (PP_k+QQ_k)/P_k$, we conclude that $G_k$ is quotient-dominant for all sufficiently large $k$.

\label{lem:dihedral_PP}
    Let $D_k$ be the dihedral group of order $N=2k$, with $k \geq 3$. Then for all $g \neq h \in D_k$,
    \begin{align}
        i(\mathcal F_{PP}(g,h))\ \leq\ \frac 97\cdot 19^{N/6}. \label{eq:dihedral_PP}
    \end{align}
<!-- end lem-12 -->
<!-- proof-16: body_md -->
Write $\rho$ for a rotation of order $k$, so the rotations form the cyclic subgroup $\left \langle \rho \right \rangle$ and every other element is a reflection. First suppose $g$ or $h$ is a reflection, say $g$. The centralizer of a reflection has order at most $4$, so, as in the proof of \Cref{lem:FP_asym}, at most $4$ vertices lie in $2$-cycles of $\mathcal F_P(g)$ and
    \begin{align}
        i(\mathcal F_{PP}(g,h))\ \leq\ i(\mathcal F_P(g))\ \leq\ \sqrt 3^{\,4}\cdot 7^{(N-4)/4}\ =\ \frac 97 \cdot 7^{N/4}\ \leq\ \frac 97\cdot 19^{N/6}. \label{eq:PP97dihedral}
    \end{align}
    Now suppose $g = \rho^a$ and $h = \rho^b$ are both rotations, with $a \neq b$ modulo $k$. A product $xy$ is a rotation exactly when $x$ and $y$ are both rotations or both reflections, so $\mathcal F_{PP}(g,h)$ has no edges between rotations and reflections, and $i(\mathcal F_{PP}(g,h))$ is the product of the numbers of independent sets of the two induced subgraphs. The subgroup of rotations is abelian, so we are in the situation of the last part of the proof of \Cref{lem:double_bound} with $G$ replaced by $\left \langle \rho \right \rangle$, hence the rotations contribute at most $7^{k/4}$.
    
    On the reflections, a reflection $x$ is joined to $xg,\ gx,\ xh,\ hx$, which are the reflections obtained from $x$ by rotating through $\pm a$ and $\pm b$. If $a = 0$ or $b=0$, every reflection is self-adjacent and contributes a factor of $1$. Otherwise, the reflections form a simple graph with every degree in $\{2,3,4\}$, so they contribute at most $19^{k/6}$ by \Cref{cor:Zhao_strong_indep}. Since $7^{1/4} < 19^{1/6}$, we conclude that
    \begin{align}
        i(\mathcal F_{PP}(g,h))\ \leq\ 7^{k/4}\cdot 19^{k/6}\ \leq\ 19^{k/3}\ =\ 19^{N/6}.
    \end{align}
    As $19^{N/6} \leq \frac 97 \cdot 19^{N/6}$, the bound in \eqref{eq:PP97dihedral} is the larger of the two, so \eqref{eq:dihedral_PP} holds in both cases.

<!-- thm-6: body_md -->
\label{thm:simplegroups}
    All sufficiently large non-abelian finite simple groups are product-dominant.
<!-- end thm-6 -->
<!-- proof-19: body_md -->
Let $G$ be a non-abelian finite simple group. Then $Z = |Z(G)| = 1$, since $Z(G)$ is a normal subgroup of $G$ and is proper because $G$ is non-abelian. Moreover, by the Feit--Thompson theorem \cite{feit-thompson}, $G$ has even order, hence has an element $g$ of order $2$. Let $\mathcal C_g$ be the conjugacy class of $g$. Then, the action of $G$ on $\mathcal C_g$ by conjugation induces a homomorphism $\varphi :G \to S(\mathcal C_g)$, where $S(\mathcal C_g)$ denotes the symmetric group on $\mathcal C_g$. If $\ker \varphi = G$, then $xgx^{-1}=g$ for all $x \in G$, hence $g \in Z(G)$ which cannot be the case. Therefore, $\ker\varphi$ is a proper normal subgroup of $G$, thus is trivial. So, $G$ embeds in $S(\mathcal C_g)$, and $|G| \leq |\mathcal C_g|!$. Therefore, if $\{G_k\}$ is a sequence of non-abelian finite simple groups with $|G_k| \to \infty$, we must have an involution $g_k \in G_k$ for all $k$, and since $|G_k| \leq |\mathcal C_{g_k}|!$, we must have $|\mathcal C_{g_k}| \to \infty$. Finally, every element in the conjugacy class $\mathcal C_{g_k}$ must also have order $2$, thus $T_k \geq |\mathcal C_{g_k}| \to \infty$, so it follows that $Z_k = 1 = o(T_k)$, and case (i) of \Cref{thm:mptqresult} applies.

Let $N := |G|$, and let $T$ be the number of elements of order $2$. The kernel of the homomorphism $x\mapsto 2x$ consists of the identity and the $T$ elements of order $2$, so the first isomorphism theorem gives $N = m\cdot (T+1)$.

    Let $P$ and $Q$ be as in \Cref{lem:PQPPQQ} (formed with $G$ instead of $G_k$). Since $G$ is abelian, $Z(G)=G$, and the number $r(g)$ of square roots of $g$ is $T+1$ if $g \in 2G$ and $0$ otherwise. Hence, by \eqref{eq:FP_true_sum} and \Cref{cor:FQ_asym},
    \begin{align}
        P \ &=\ (N-m)\sqrt{3}^N + m\sqrt{3}^{N-(T+1)}  \ =\ mT\sqrt{3}^N + m\sqrt{3}^{N-(T+1)},  \label{eq:abelianP} \\
        Q \ &\leq\ T\sqrt{3}^N + (N-T)7^{N/4}. \label{eq:abelianQ}
    \end{align}
    Let $Q^+ := T\sqrt{3}^N + (N-T)7^{N/4}$, so $Q \leq Q^+$. Since $m\geq 6$ and $5T + 6\cdot 3^{-(T+1)/2} \geq 3$ for every $T \geq 0$, \eqref{eq:abelianP} and \eqref{eq:abelianQ} give
    \begin{align}
        M_1(G)\ \overset{\eqref{eq:M1_sum}}{=}\ Q - P \ \leq\ Q^+-P \ \leq\ \sqrt 3^N\left ( N(7/9)^{N/4} - 3 \right ) \ <\ 0,
    \end{align}
    where the last inequality holds as $N \geq 171$. Hence case (ii) of \Cref{thm:dom_ineq} applies. Since $G$ is abelian, every pair $g,h$ is central, so \eqref{eq:FPP_central} gives $PP \leq N(N-1)\cdot 7^{N/4}$, while \eqref{eq:FQQ_bound} gives $QQ \leq N(N-1)\cdot 19^{N/6}$ (here $PP$ and $QQ$ are as in \Cref{lem:PQPPQQ}, not to be confused with the products $P\cdot P$ and $Q\cdot Q$). Write $W := N(N-1)\left ( 19^{N/6}+7^{N/4} \right ) $, so that $PP+QQ \leq W$.

    As $Q<P$, we may apply \Cref{lem:uv_criterion} with $X=P$, and the quantities $u,v$ of that lemma satisfy
    \begin{align}
        \frac QP\ \leq\ \frac{Q^+}{P}\ =:\ u \quad \quad \text{and} \quad \quad \frac{PP+QQ}{P}\ \leq\ \frac WP\ =:\ v.
    \end{align}
    By \Cref{lem:uv_criterion}(ii), it therefore suffices to prove that
    \begin{align}
        f(u,v)\ =\ u^2-6u+1-v(1+u)\ >\ 0 \label{eq:abelian_f_goal}
    \end{align}
    for these $u$ and $v$. We find explicit upper bounds for $u$ and $v$ which depend only on $N$. If $T=0$, then $m=N$ and $P = N\sqrt 3^{N-1}$, so $u = \sqrt 3\,(7/9)^{N/4}$ and $v = \sqrt 3\,(N-1)\left ( (19/27)^{N/6}+(7/9)^{N/4} \right ) $, and both bounds below hold. If $T \geq 1$, we have
    \begin{align}
        u \ &=\ \frac{T\sqrt{3}^N+(N-T)7^{N/4}}{(N-m)\sqrt{3}^N + m\sqrt{3}^{N-(T+1)}} \notag \\
        &\leq\ \frac{T\sqrt{3}^N+(N-T)7^{N/4}}{mT\sqrt{3}^N} \notag \\
        &=\ \frac{1}{m} + \frac{N-T}{mT} (7/9)^{N/4} \notag \\
        &\leq\ \frac{1}{6} + 2\left(\frac{7}{9}\right)^{N/4}
    \end{align}
    as $(N-T)/(mT) < N/(mT) = (T+1)/T \leq 2$. Moreover,
    \begin{align}
        v\ &=\ \frac{N(N-1)\left ( 19^{N/6}+7^{N/4} \right ) }{(N-m)\sqrt{3}^N+m\sqrt{3}^{N-(T+1)}} \notag \\
        &\leq\ \frac{N(N-1)\left ( 19^{N/6}+7^{N/4} \right ) }{mT\sqrt{3}^N} \notag \\
        &\leq\ 2(N-1)\left ( \left(\frac{19}{27}\right)^{N/6}+\left(\frac{7}{9}\right)^{N/4} \right ) , 
    \end{align}
    since $N/(mT) = (T+1)/T \leq 2$. Thus, 
    \begin{align}
        f(u,v)\ &\geq\ f\left(\frac{1}{6}+2\left(\frac{7}{9}\right)^{N/4},\ 2(N-1)\left ( \left(\frac{19}{27}\right)^{N/6}+\left(\frac{7}{9}\right)^{N/4} \right ) \right) \ =:\ h(N).
    \end{align}
    Finally, it is not hard to show that $h(N)$ is increasing for every $N\geq 12$, and numerical analysis shows that $h(N) > 0$ for all $N\geq 171$.

Recall from \Cref{sec:M1} the definition of $p_g^{(n)}(x)$. For $g,x \in G$, let $\ell_{g,x}$ denote the minimal positive integer $\ell$ such that $p_g^{(\ell)}(x)=x$. Thus, the cycle containing $x$ in $\mathcal F_P(g)$ has length $\ell_{g,x}$.

<!-- lem-13: body_md -->
\label{lem:divisibility}
    If $p^{(d)}_g(x)=x$, then $\ell_{g,x}\mid d$.
<!-- end lem-13 -->
<!-- proof-22: body_md -->
Write $\ell := \ell_{g,x}$ and $d = q\ell + s$ with $q \geq 0$ and $0 \leq s < \ell$. Since $p_g^{(\ell)}(x)=x$, applying this $q$ times gives $p_g^{(q\ell)}(x) = x$, hence
\begin{align}
    x\ =\ p_g^{(d)}(x)\ =\ p_g^{(s)}\left ( p_g^{(q\ell)}(x) \right ) \ =\ p_g^{(s)}(x), \label{eq:div_alg}
\end{align}
where $p_g^{(0)}$ denotes the identity map. By the minimality of $\ell$, we must have $s=0$, so $\ell \mid d$.

    so by \Cref{lem:divisibility} we have $\ell_{g,x} \mid 2\operatorname{ord}(g)$.

<!-- lem-15: body_md -->
\label{lem:Ldg_formula}
    For any $g \in G$, $d \in \mathbb{Z}^+$, let
    \begin{align}
        L(d,g)\ :=\ \#\left \{  x\in G: p_g^{(d)}(x)=x \right \} .
    \end{align}
    Let $\chi_1,\dots, \chi_n$ be the irreducible characters of $G$, and let $\nu$ be the Frobenius--Schur indicator
    \begin{align}
        \nu(\chi)\ :=\ \frac{1}{|G|}\sum_{h \in G}\chi(h^2).
    \end{align}
    If $d=2k$ is even, then
    \begin{align}
        L(2k,g)\ =\ \left | C_G(g^k) \right | \ =\ \sum_{i=1}^n\left | \chi_i(g^k) \right | ^2. \label{eq:L_even_d}
    \end{align}
    If $d=2k+1$ is odd, then
    \begin{align} \label{eq:L_odd_d}
        L(2k+1,g)\ =\ \sum_{i=1}^n \nu(\chi_i)\cdot \chi_i(g^{-2k-1}).
    \end{align}
<!-- end lem-15 -->
<!-- proof-24: body_md -->
Recall that $p_g^{(2k)}(x) = x \iff xg^k=g^kx$. Thus, $L(2k,g) = |C_G(g^k)|$. The second equality in \eqref{eq:L_even_d} follows from the second orthogonality relation for characters.

    When $d=2k+1$ is odd, \Cref{lem:pgqg_forms} gives
    \begin{align}
        p_g^{(2k+1)}(x)\ =\ x \ \iff\ g^{-k}x^{-1}g^{k+1}\ =\ x.
    \end{align}
    Substituting $x=g^{k+1}y$ yields
    \begin{align}
        p_g^{(d)}(x)=x\ &\iff\  g^{-k}y^{-1}g^{-(k+1)}g^{k+1}=g^{k+1}y \notag
        \\ &\iff\ g^{-d}=y^2.
    \end{align}
Our substitution induces a bijection on $G$, hence the number of $y \in G$ satisfying $g^{-d}=y^2$ is precisely $L(d,g)$. The number of square roots of $g^{-d}$ is known in general to be
\begin{align}
    \sum_{i=1}^n\nu(\chi_i) \cdot \chi_i(g^{-d})
\end{align}
(see \cite{isaacs2006character} or \cite{rowe2022squareroots}, for example). This proves \eqref{eq:L_odd_d}.

<!-- thm-9: body_md -->
\label{thm:num_of_cycles_prod}
    Fix $g\in G$. Let $c(m,g)$ be the number of $m$-length cycles in $\mathcal F_P(g)$. Then
    \begin{align}
        c(m,g)\ =\ \frac{1}{m}\sum_{d\mid m}\mu\left(\frac{m}{d}\right)L(d,g).
    \end{align}
<!-- end thm-9 -->
<!-- proof-25: body_md -->
Define
    \begin{align}
    f(d) \ :=\ \#\{x\in G:\ell_{g,x}=d\}.
    \end{align} 
   Then, by \Cref{lem:divisibility}, we see that
    \begin{align}
        L(m,g)\ =\ \sum_{d\mid m}f(d).
    \end{align}
    Thus, by the Möbius inversion formula, we have 
    \begin{align}
        f(m)\ =\ \sum_{d\mid m}\mu(m/d)\cdot L(d,g), \label{eq:mobiususe}
    \end{align}
    where $\mu$ is the Möbius function.
    This gives us the number of $x\in G$ contained in $m$-cycles in $\mathcal F_P(g)$. We divide by $m$ to obtain the total number of $m$-length cycles.

<!-- thm-10: body_md -->
\label{thm:expthm}
    Let $G$ be a finite group, let $\mathcal C_1,\dots,\mathcal C_n$ be the distinct conjugacy classes of $G$, and let $g_i$ be a representative for $\mathcal C_i$. Then
    \begin{align} \label{eq:expecG}
        M_1(G)\ =\ \sum_{i=1}^n|\mathcal C_i|\left (  L_{\operatorname{ord}{g_i}}^{|G|/\operatorname{ord}{g_i}} - \prod_{m \mid \left ( 2\cdot \operatorname{ord}{g_i} \right ) }L_m^{c(m,g_i)}  \right ) .
    \end{align}
<!-- end thm-10 -->
<!-- proof-26: body_md -->
By \Cref{lem:M1_sum},
    \begin{align}
        M_1(G)\ =\ \sum_{g \in G}\left ( i(\mathcal F_Q(g)) - i(\mathcal F_P(g)) \right ) . \label{eq:M1_mention2}
    \end{align}
By the work done in \Cref{sec:M1} (namely \Cref{lem:disjoint_cycles}, \Cref{rmk:lucas_forbiddance},  and \Cref{lem:FQ_form}), as well as the definition of $c(m,g)$ and \Cref{lem:poss_lengths}, we see that
\begin{align}
    i(\mathcal F_Q(g)) - i(\mathcal F_P(g))\ = \ L_{\operatorname{ord}(g)}^{|G|/\operatorname{ord}(g)}-\prod_{m \mid (2 \cdot \operatorname{ord} g)}L_m^{c(m,g)} .\label{eq:idiffinlucas}
\end{align}
Moreover, from the computation leading to \eqref{eq:exp_breakdown}, recall that
\begin{align}
    i(\mathcal F_Q(g)) - i(\mathcal F_P(g)) \ = \ \#\{A \subseteq G: g\in AA\} - \# \{ A \subseteq G: g \in AA^{-1}\}. \label{eq:subsetcontainment}
\end{align}
Now, for any $g \in G$, $A \subseteq G$, and any automorphism $\sigma$ of $G$, note that
\begin{align}
    g \in AA \ &\iff\ g = xy \text{ for } x,y\in A  \notag
    \\ &\iff\  \sigma(g)=\sigma(x)\sigma(y)  \text{ for } x,y \in A \notag
    \\ &\iff\  \sigma(g) \in \sigma(A)\sigma(A).
\end{align}
We obtain a similar equivalence for $g \in AA^{-1}$. Since $A \mapsto \sigma(A)$ is a bijection on the subsets of $G$, the expression in \eqref{eq:subsetcontainment} takes the same value at $g$ and at $\sigma(g)$. In particular, they are invariant under the action of conjugation, so it follows from \eqref{eq:idiffinlucas} that
\begin{align}
    \sum_{g \in G}\left ( i(\mathcal F_Q(g)) - i(\mathcal F_P(g)) \right ) \ &=\ \ \sum _{i=1}^n|\mathcal C_i|\left ( i(\mathcal F_Q(g_i)) - i(\mathcal F_P(g_i)) \right )  \notag
    \\ &=\ \  \sum_{i=1}^n|\mathcal C_i|\left (  L_{\operatorname{ord}{g_i}}^{|G|/\operatorname{ord}{g_i}} - \prod_{m \mid \left ( 2\cdot \operatorname{ord}{g_i} \right ) }L_m^{c(m,g_i)}  \right ) .
\end{align}
