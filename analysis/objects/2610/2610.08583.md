<!-- definition-1: body_md -->
A link is called \emph{positive} if it admits a positive diagram.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm: maintheorem1}
Let $L$ be a non-split positive link, with a base point $p$. Then
\begin{equation*}
Kh_{e,\operatorname{red}}^{1,j}(L,p) \cong Kh_{o,\operatorname{red}}^{1,j}(L) =
\begin{cases}
\mathbb{Z}^{p_1(L)} & \text{if } j = 3 - \chi(L),\\
0 & \text{otherwise,}
\end{cases}
\end{equation*}
where $\chi(L)$ denotes the Euler characteristic of a genus-minimizing Seifert surface of $L$, and $p_1(L)$ is the first cyclomatic number of the reduced Seifert graph of any positive diagram of $L$ (see Section~\ref{sec: Khovanov homology} for a precise definition).
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm: maintheorem2}
For a non-split positive link $L$, the unreduced odd Khovanov homology satisfies
\begin{equation*}
\begin{aligned}
Kh_{o}^{1,j}(L) &=
\begin{cases}
\mathbb{Z}^{p_1(L)} & \text{if } j = 2-\chi(L) \text{ or } 4-\chi(L),\\
0 & \text{otherwise.}
\end{cases}
\end{aligned}
\end{equation*}
where $\chi(L)$ and $p_1(L)$ are defined as before.
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor: maincorollary3}
Let $L$ be a non-split positive link with basepoint $p$. Then the following are equivalent: $L$ is fibered; $Kh_{e,\operatorname{red}}^{1,*}(L,p) = 0$; $Kh_{o,\operatorname{red}}^{1,*}(L) = 0$; $Kh_{e}^{1,*}(L) = 0$ and $Kh_{o}^{1,*}(L) = 0$.
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
The results for reduced and odd homology follow immediately from Theorems~\ref{thm: maintheorem1} and \ref{thm: maintheorem2}, together with A.~Stoimenov's observation \cite{Stoimenow_PolynomailValues_Unknottingnumbers} that a positive link diagram $D$ has $p_1(D) = 0$ if and only if the link it represents is fibered. The result for even Khovanov homology follows from \cite[Theorem~1.2]{KMMS_Khovanov_pos_fib}.
<!-- end proof-1 -->
<!-- theorem-3: body_md -->
Let $L$ be a non-split positive link with basepoint $p$. Then, for every $j$,
\[
\operatorname{Tor}(Kh_{e,\operatorname{red}}^{2,j}(L,p)) = \operatorname{Tor}(Kh_{o,\operatorname{red}}^{2,j}(L)) = \operatorname{Tor}(Kh_{o}^{2,j}(L)) = 0,
\]
where $\operatorname{Tor}(\cdot)$ denotes the torsion subgroup.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm: maintheorem4}
    If $K$ is a positive knot, then for every $q\geq p\geq2$ the Khovanov homology of its $(p,q)$-cable $K_{p,q}$ fulfills 
    \begin{align*}
    Kh_{*,\operatorname{red}}^{i,j}(K_{p,q})&=0 \,\textrm{ if } \,i<0,\\
    Kh_{*,\operatorname{red}}^{0,j}(K_{p,q})&=\begin{cases}
        \mathbb{Z} &\textrm{ if } \,j=1-\chi(K_{p,q}),\\
    0  &\textrm{ otherwise, } \\
    \end{cases}\\
    Kh_{*,\operatorname{red}}^{1,j}(K_{p,q})&=\begin{cases}
        \mathbb{Z}^{p_1(K)} &\textrm{ if } \,j=3-\chi(K_{p,q}) ,\\
    0  &\textrm{ otherwise, } 
    \end{cases}\\
    \operatorname{Tor}(Kh^{2,j}_{*,\operatorname{red}}(K_{p,q})) &=0 \quad \text{for every} \quad j,
\end{align*}
where $*\in\{e,o\}$, $\chi(L)$, $p_1(L)$ and $\operatorname{Tor}(\cdot)$ are defined as before. Here, we suppress the base point from the notation of reduced Khovanov homology to avoid unnecessary clutter, since the result is independent of its choice.

<!-- definition-2: body_md -->
\label{def: even module}
	Let $I_e \subset T(S)$ be the two-sided ideal generated by
	\[
	x \otimes x \quad \text{and} \quad x \otimes y - y \otimes x,
	\qquad x,y \in S.
	\]
	The \emph{even algebra} associated to $S$ is defined by
	\[
	\Lambda_e(S) := T(S)/I_e.
	\]
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def: odd module}
	Let $I_o \subset T(S)$ be the two-sided ideal generated by
	\[
	x \otimes x \quad \text{and} \quad x \otimes y + y \otimes x,
	\qquad x,y \in S.
	\]
	The \emph{odd algebra} associated to $S$ is defined by
	\[
	\Lambda_o(S) := T(S)/I_o.
	\]
<!-- end definition-3 -->
<!-- remark-1: body_md -->
\begin{enumerate}
	\item The algebra $\Lambda_e(S)$ is the symmetric algebra on $S$ modulo the ideal generated by the squares of the generators. The algebra $\Lambda_o(S)$ is the exterior algebra generated by $S$. 
	
	\item If $S$ is a singleton, then $\Lambda_e(S) \cong \mathbb{Z}[x]/(x^2)$. More generally, for a finite set $S$, we have
	\[
	\Lambda_e(S) \cong \left(\mathbb{Z}[x]/(x^2)\right)^{\otimes |S|}.
	\]
	
	\end{enumerate}
<!-- end remark-1 -->
<!-- definition-4: body_md -->
Let $D$ be an oriented link diagram with $n$ ordered crossings $\operatorname{cr}(D) = \{c_1, \dots, c_n\}$. A \emph{state} of a diagram \(D\) is a function  $s \colon \operatorname{cr}(D) \to \{0,1\},$ assigning to each crossing of \(D\) a value \(0\) or \(1\), called a \emph{marker}.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
The \emph{Seifert state} is the state in which each positive crossing is assigned the marker $0$ and each negative crossing is assigned the marker $1$.
	The \emph{Seifert resolution} is the resolution of a diagram corresponding to the Seifert state.
<!-- end definition-5 -->
<!-- definition-6: body_md -->
Given a state $s\in \mathcal{S}(D)$, the \textit{state graph} $G_s(D)$ is a planar graph obtained by assigning a vertex to each circle in $sD$ and adding an edge between two vertices for each crossing connecting the corresponding circles in the original diagram.
<!-- end definition-6 -->
<!-- definition-7: body_md -->
Given a graph $G$, its associated reduced graph $G^{\text{red}}$ is the graph obtained from $G$ by identifying parallel edges.
<!-- end definition-7 -->
<!-- definition-8: body_md -->
A link diagram $D$ is called \textit{$0$-adequate} if the state graph $G_0(D)$ contain no loops,  \textit{$1$-adequate} if the state graph $G_1(D)$ contain no loops and \textit{adequate} if both $G_0(D)$ and $G_1(D)$ contains no loops.
<!-- end definition-8 -->
<!-- example-1: body_md -->
Every positive diagram is \(0\)-adequate.
<!-- end example-1 -->
<!-- definition-9: body_md -->
\label{def: cyclomatic number}
	The \textit{cyclomatic number }$p_1(G)$ of a graph $G$ is the minimal number of edges that one must delete to transform $G$ into a collection of disjoint tree graphs. It can be computed as $p_1(G) = e-v+k$, where $v, e$ and $k$ denote the number of vertices, edges and connected components of $G$, respectively. 
	
	Given a link diagram $D$, we define its cyclomatic number $p_1(D)$ as the cyclomatic number of the reduced $G_0(D)^{\operatorname{red}}$ graph associated with $D$.
<!-- end definition-9 -->

We will not distinguish, in our notation, between states or their resolutions for decorated and undecorated diagrams. When discussing even Khovanov homology, all diagrams are understood to be undecorated, whereas in the odd Khovanov homology setting, diagrams are assumed to be decorated.
<!-- end remark-2 -->

A categorical structure induced on $\mathbf{2}^n$ by the partial order, where $\operatorname{Hom}_{\underline{2}^n}(s,s')$ has a single element if $s \leq s'$ and is empty otherwise.  Write $\phi_{s,s'}$ for the unique morphism $s \to s'$ if it exists. We will denote this by $\underline{2}^n$ when we regard cube as a catgeory, rather than as a poset or simply as a set.

<!-- definition-10: body_md -->
\label{def:edge-assignment}
	A \emph{type-X} (respectively, \emph{type-Y}) \emph{edge assignment} for a diagram $D$ with oriented crossings is a cochain

<!-- proposition-2: body_md -->
\label{prop: even technical}
The reduced even Khovanov homology of the diagram $(D', p)$ constructed above satisfies
\[
Kh^{1}_{e, \mathrm{red}}(D, p) = Kh^{1}_{e, \mathrm{red}}(D', p).
\]
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
We suppress quantum grading shifts throughout, as neither the statement nor the argument depends on them. Since the diagram \( D \) contains no negative crossings, the homological grading \( i \) equals \( |s| \).
	
	As recalled in Section~\ref{sec: reduced Khovanov homology}, there is a natural inclusion of chain complexes
	\[
	CKh^{i}_{e,\mathrm{red}}(D,p) \hookrightarrow CKh^{i}_e(D).
	\]
	We therefore describe cycles in \( CKh^{i}_e(D) \); the same description applies to the reduced complex.
	Let \( G = G_0(D) \) be the state graph associated to the state \( s_0 \), which assigns the \(0\)-marker to every crossing. Let \( E = \{ e_1, \ldots, e_n \} \) be the edge set of \( G \), where each edge \( e_j \) corresponds to a crossing \( c_j \). Let \( l = |s_0 D| \) denote the number of circles in the resolution \( s_0 D \).
	
	The group \( CKh^1_e(D) \) is generated by states with exactly one \(1\)-marker, and decomposes as
	\[
	CKh^1_e(D) = \bigoplus_{i=1}^n A_i,
	\]
	where \( A_i =\Lambda_e(s^iD)\) and the state \( s^i \) assigning a \(1\)-marker at \( c_i \) and \(0\)-markers elsewhere. The proof of \cite[Theorem~4.1]{KMMS_Khovanov_pos_fib} establishes the following fact. If
	\[
	z = (z_i)_{i=1}^n \in CKh^1_e(D)
	\]
	satisfies \( \partial z = 0 \), then \( z_i = z_j \) whenever the edges \( e_i \) and \( e_j \) are parallel in \( G \).
	
	Therefore, for the purpose of computing \( Kh^1_{e,\operatorname{red}}(D,p) \), we may replace \( D \) by a diagram \( D' \) obtained by retaining exactly one crossing between each pair of Seifert circles and resolving all remaining crossings using the \(0\)-marker. In particular,
	\[
	Kh^1_{e,\operatorname{red}}(D,p) \cong Kh^1_{e,\operatorname{red}}(D',p).
	\]

<!-- lemma-1: body_md -->
\label{lem: extreme even KH}
If $D$ is $0$-adequate link diagram with $n_+$ positive and $n_-$ negative crossings and a basepoint $p$, then its reduced even Khovanov homology satisfies the following.
\[
Kh_{e,\operatorname{red}}^{i,n_+ - 2n_- - |s_0 D| + 1}(D,p) =
\begin{cases}
	\mathbb{Z} & \text{if } i = -n_- \\
	0 & \text{otherwise.}
\end{cases}
\]
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
Consider the generator
	\[
	x = a_1 \otimes a_2 \otimes \cdots \otimes a_{|s_0 D|}
	\]
	in $\Lambda_e(s_0 D)$, where each $a_i$ corresponds to a circle of $s_0 D$. Observe that this generator lies in the minimal possible quantum grading.
	
	Since $D$ is $0$-adequate, performing a $1$-smoothing at any crossing reduces the number of circles in the smoothing of that state compared to $s_0 D$. Consequently, $\partial x=0$. Furthermore, as there are no chain groups in lower homological degrees whose differential maps to $x$, the element $x$ represents a nontrivial homology class.
	The same argument shows that, for every state $s_0\leq s$, the quantum grading of any generator in $\Lambda_e(sD)$ is strictly greater than the minimal quantum grading by $2$. Hence, no additional generators in the required quantum grading arise from states obtained by $1$-smoothings.
	
	Hence, the chain complex in this minimal quantum grading is generated uniquely by $x$, and so is its homology.
<!-- end proof-3 -->
<!-- lemma-2: body_md -->
\label{lem: almost extreme even KH}
If $D$ is a positive diagram of $L$ with $n$ crossings, then its reduced even Khovanov homology satisfies the following.
\[
Kh_{e,\operatorname{red}}^{0,n - |s_0 D| + 3}(D,p) =0.
\]
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
By \cite[Remark~6.4]{almost_extremeKH}, the unreduced even Khovanov complex admits a unique generating cycle in bidegree $(i,j) = (0, n - |s_0 D| + 3)$, explicitly given by
    \[
    \sum_{v \in V_{G_0(D)}} \sigma(v)\, a_1 \otimes \cdots \widehat{a_v} \cdots \otimes a_{|s_0D|},
    \]
    where $\sigma \colon V_{G_0(D)} \to \{\pm 1\}$ assigns opposite signs to adjacent vertices, and $\widehat{a_v}$ indicates omission of the corresponding tensor factor. For details regarding the derivation of this generator, see the proof of Theorem~\ref{thm: maintheorem5} in Section~\ref{sec: Almost extreme reduced odd Khovanov homology}.

    This element does not lie in the subcomplex
    \[
    CKh_{e,\operatorname{red}}^{0,\, n - |s_0 D| + 3}(D,p) \hookrightarrow CKh_{e}^{0,\, n- |s_0 D| + 2}(D),
    \]
    and therefore does not contribute to the reduced even Khovanov homology.
<!-- end proof-4 -->
<!-- lemma-3: body_md -->
\label{lem: reduced even KH last_group}
 The reduced even Khovanov homology of the diagram $(D',p)$ above satisfies 
	\begin{align*}
		Kh^{1, j}_{e,\operatorname{red}}(D',p)=\begin{cases}
			\mathbb{Z}^{p_1(D')} &\textrm{ if } j=n'-|s_0D|+3,\\
			0 &\textrm{ otherwise,}
		\end{cases}
	\end{align*}
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Recall that, by construction, $D'$ a positive, alternating diagram representing a non-split link. Without loss of generality, we may also assume that $D'$ is reduced. By \cite{Lee_KhovanovHomology}, the reduced Khovanov homology of a non-split alternating link is thin, i.e.,
	\[
	Kh^{i,j}_{e,\operatorname{red}}(L') \neq 0 \quad \text{only if} \quad j - 2i = \sigma(L').
	\]
	In particular, for $i = 1$, the reduced Khovanov homology can be nonzero only in the quantum grading
	\[
	j = \sigma(L') + 2.
	\]
	By Equation~\eqref{eqn:signature_cyclomaticnumber_equality} and Definition~\ref{def: cyclomatic number}, we obtain	\[

		\]
		where $w$ and $w_1$ denote the writhes of $D'$ and $D_1$, respectively.

		\[
		Kh_{e,\mathrm{red}}^{0, p_1(D_0) + 2}(D_0,p) = 0.

	0 \longrightarrow \mathbb{Z} \longrightarrow Kh_{e,\mathrm{red}}^{1, p_1(D') + 2}(D',p) \longrightarrow \mathbb{Z}^{\,p_1(D') - 1} \longrightarrow 0.
	\]
	
	This completes the induction step and proves the desired result for all $p_1(D')$.

<!-- proposition-5: body_md -->
\label{prop: general coefficent}
Let $L$ be a non-split positive link with base point $p$, and let $R$ be a commutative ring with unit. Then the reduced even and odd Khovanov homologies over $R$ satisfy
\[
\begin{aligned}
Kh^{1,j}_{*,\operatorname{red}}(L,p;R) &\cong
\begin{cases}
R^{p_1(L)} & \text{if } j = 3 - \chi(L),\\
0 & \text{otherwise,}
\end{cases} \\
Kh^{i,\,3-\chi(L)}_{*,\operatorname{red}}(L,p;R) &\cong
\begin{cases}
R^{p_1(L)} & \text{if } i = 1,\\
0 & \text{otherwise.}
\end{cases}
\end{aligned}
\]
where $* \in \{e,o\}$, and $\chi(L)$ and $p_1(L)$ are as previously defined.
<!-- end proposition-5 -->
<!-- proof-17: body_md -->
The proof proceeds by verifying that the arguments of the previous sections extend to this general setting.	

	\begin{enumerate}
	\item In Section~\ref{sec: Reduced even Khovanov homology of positive links}, Proposition~\ref{prop: even technical} and Lemma~\ref{lem: extreme even KH} are independent of the choice of coefficient ring; their statements and proofs remain valid without modification. For Lemma~\ref{lem: almost extreme even KH}, the argument carries over directly and shows that the relevant cycle does not lie in the reduced complex. Furthermore, Lemma~\ref{lem: reduced even KH last_group} relies only on exact sequences and Lee's theorem for alternating links, both of which hold over arbitrary coefficient rings. 
	Thus, the arguments in this section extend without essential modification to give the first statement of the theorem in the even case. 
	
	Similarly, the arguments of Section~\ref{sec: Reduced odd Khovanov homology of positive links} extend to arbitrary coefficient rings by the same reasoning as above, yielding the first statement in the odd case.

	\item The second statement follows from an application of the universal coefficient theorem to Proposition~\ref{thm: mainproposition7}.
\end{enumerate}

<!-- corollary-3: body_md -->
\label{cor: torsion}
Let $L$ be a non-split positive link with base point $p$. Then
\[
\operatorname{Tor}\bigl(Kh^{2,j}_{*,\operatorname{red}}(L,p)\bigr) = 0 \quad \text{for all } j,
\]
where $*\in \{e,o\}$ and $\operatorname{Tor}$ denotes the torsion subgroup.
<!-- end corollary-3 -->
<!-- proof-18: body_md -->
The universal coefficient theorem gives a short exact sequence
\[
0 \to Kh^{1,j}_{*,\operatorname{red}}(L,p)\otimes R
\to Kh^{1,j}_{*,\operatorname{red}}(L,p;R)
\to \operatorname{Tor}^{\mathbb{Z}}_{1}\bigl(Kh^{2,j}_{*,\operatorname{red}}(L,p),R\bigr)
\to 0.
\]

\begin{enumerate}
	\item For $j = 1-\chi(L)$, the result follows from Lemmas~\ref{lem: extreme even KH} and \ref{lem: extreme odd KH}.
	\item For $j = 3-\chi(L)$, the result follows from Proposition~\ref{thm: mainproposition7}. 
	\item For $j > 3-\chi(L)$, both $Kh^{1,j}_{*,\operatorname{red}}(L,p;R)$ and $Kh^{1,j}_{*,\operatorname{red}}(L,p)\otimes R$ vanish by Proposition~\ref{prop: general coefficent} and Theorem~\ref{thm: maintheorem1}. Hence
	\[ \operatorname{Tor}^{\mathbb{Z}}_{1}\bigl(Kh^{2,j}_{*,\operatorname{red}}(L,p),R\bigr)=0 \]
	for all $R$, and therefore $\operatorname{Tor}\bigl(Kh^{2,j}_{*,\operatorname{red}}(L,p)\bigr)=0.$
\end{enumerate}

<!-- proposition-6: body_md -->
\label{prop:twisted_cable}
	Let $K$ be a positive knot. Then, for $2\leq p \leq q-1$ and $1\leq m \leq p-1$ we have 
	\begin{align*}
		Kh^{i,j}_{*,\operatorname{red}}(K_{p,q;m})&=0 \,\textrm{ for every }\,  i<0,\\
		Kh^{0,j}_{*,\operatorname{red}}(K_{p,q;m})&=\begin{cases}
			\mathbb{Z}\,&\textrm{ if }\,  j=1-p\chi(K)+q(p-1)-m, \\
			0 \,&\textrm{else,}
		\end{cases}\\
		Kh^{1,j}_{*,\operatorname{red}}(K_{p,q;m})&=\begin{cases}
			\mathbb{Z}^{p_1(K)}\,&\textrm{ if }\,  j=3-p\chi(K)+q(p-1)-m,\\
            0 \,&\textrm{else}
		\end{cases}\\
		\operatorname{Tor}(Kh^{2,j}_{*,\operatorname{red}}(K_{p,q;m})) &=0 \quad \text{for every} \quad j.
	\end{align*}
<!-- end proposition-6 -->
<!-- proof-21: body_md -->
The proof of this theorem proceeds as in \cite[Theorem~5.5]{KMMS_Khovanov_pos_fib}, since the underlying argument relies entirely on exact sequences that are equally valid in the reduced setting.
<!-- end proof-21 -->
<!-- proof-22: body_md -->
The proof naturally splits into two cases.
For $q \geq pw$, the cable $K_{p,q}$ is positive, the result follows by combining Theorem~\ref{thm: maintheorem1} with the observation that $p_1(K_{p,q}) = p_1(K)$ for $q \geq pw$, together with Schubert’s cabling formula~\cite{Schubert}
\[
-\chi(K_{p,q}) = -p\chi(K) + q(p - 1).
\] 
For $p+1 \leq q < pw$, the result follows from Proposition~\ref{prop:twisted_cable}. For the boundary case $p=q$, we use the identification $K_{p,q}=K_{p,q+1;p-1}$. Since $(q+1)-1 \geq p$, Proposition~\ref{prop:twisted_cable} applies to $K_{p,q+1;p-1}$. Furthermore,
$$1-p\chi(K)+(q+1)(p-1)-(p-1)=1-\chi(K_{p,q}).$$
Hence, the quantum gradings given by Proposition~\ref{prop:twisted_cable} agree with those in the statement of Theorem~\ref{thm: maintheorem5}.
