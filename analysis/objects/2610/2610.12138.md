<!-- head-1: body_md -->
A couple of days prior to the submission of this manuscript, Nguyen~\cite{Nguyen} proved a shape theorem for the centrally excited random walk, albeit with a different excitation formula that yields an $\ell^1$ ball (instead of the Euclidean $\ell^2 $ ball here) as the limiting shape. Yet, for the radial model considered in this paper, we obtain here a stronger conclusion, with almost sure convergence and an explicit uniform conical local-time profile. Furthermore, the proof presented here is simpler and much shorter than in \cite{Nguyen}, so the author believes the paper has its own merits and is worth presenting to the mathematical community.
<!-- end head-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Let \(v_d\) denote the volume of the Euclidean unit ball and define
\begin{equation}\label{eq:constants}
 C_* :=\frac{2d\beta v_d}{d+1} \qquad\hbox{and}\qquad
 r_n :=(n/C_*)^{1/(d+1)}.
\end{equation}
It holds that, almost surely,
\begin{enumerate}[label=(\roman*)]
 \item For every \(\varepsilon\in(0,1)\), 
\begin{equation}\label{eq:main-shape}
 D_{(1-\varepsilon)r_n}\subset A_n\subset D_{(1+\varepsilon)r_n}
 \qquad\hbox{for all $n$ large enough,}
 \end{equation} 
 where $D_r:=\{x\in\mathbb Z^d:|x|<r\}$. In particular, the volume and outer radius satisfy
 \begin{equation}\label{eq:main-clock}
 \sharp A_n \sim v_dr_n^d,\qquad
 \max_{x\in A_n}|x|\sim r_n.
 \end{equation}
 \item The local-time process is ``cone-shaped'' and the following uniform approximation holds:
 \begin{equation}\label{eq:main-local}
 \lim_{n\to\infty}\frac1{r_n}
 \sup_{x\in\mathbb Z^d}
 \left|T_n(x)-2d\beta(r_n-|x|)_+\right|=0.
 \end{equation}
 In particular, every fixed site is visited infinitely often, so the walk is recurrent and we have
\begin{equation}\label{eq:main-pointwise}
 \frac{T_n(x)}{r_n}\longrightarrow2d\beta
 \qquad\text{for every fixed }x\in\mathbb Z^d.
\end{equation}
\end{enumerate}

<!-- proposition-1: body_md -->
\label{prop:noise}
Fix $R\geq 2$. We have $\tau_R<\infty$ a.s. Define the ``good event''
$$
 \mathcal G_R:=
 \left\{\max_{z\in D_{2R}}\max_{0\le n\le\tau_R}|M_n(z)|<R^{3/4}\right\}.
$$
There exist constants $C,C_{\rm time}$ such that
$$
 \mathbb P(\mathcal G_R^c)\le CR^{-3}.
$$
Furthermore, on $\mathcal G_R$, it holds that
\begin{equation}\label{eq:good-clock}
 \max_x T_{\tau_R}(x)\le CR
 \qquad \hbox{and}\qquad \tau_R\le C_{\rm time}R^{d+1}.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
The fact that $\tau_R < \infty$ a.s. is obvious since there is a single cookie per site and the process behaves like a simple random walk otherwise, so it cannot remain contained in any bounded region.

We first bound the two terms other than $M_n(z)$ in \eqref{eq:green-identity}. For $z\in D_{2R}$ and $n\le\tau_R$, estimate \eqref{eq:green-growth} gives the crude upper bound
\begin{equation}\label{eq:green-endpoint}
 |G(-z)-G(X_n-z)|\le C\log R
\end{equation}
for some $C>0$. Furthermore, \eqref{eq:green-increment-bound} and the definition of $J$ in \eqref{eq:cookie-correction} imply that $|J(x,z)|\le C(1+|x-z|)^{1-d}$.  Thus, for $z\in D_R$ and $n\le\tau_R$ we find that
$$
 \left|\sum_{x\in A_n}J(x,z)\right|
 \leq C\sum_{x\in D_R}(1+|x-z|)^{1-d}  \leq  C\sum_{x\in D_{2R}}(1+|x|)^{1-d} = O(R).
$$
Combining this bound with \eqref{eq:green-endpoint} in \eqref{eq:green-identity} gives
\begin{equation}\label{eq:pathwise-occupation}
 T_n(z)\le M_n(z)+CR,
 \qquad \hbox{for $z\in D_R$ and $n\le\tau_R$.}
\end{equation}
Now introduce
$$
 \gamma_R:=\inf\left\{n\ge0:\max_{z\in D_{2R}}|M_n(z)|\ge R^{3/4}\right\},
 \qquad \sigma:=\tau_R\wedge\gamma_R.
$$
Equation \eqref{eq:pathwise-occupation} gives
$$
\max_x T_\sigma(x)\le\max_x T_{\sigma-1}(x)+1
 \le R^{3/4}+CR+1\le C'R.
$$
For $z\in D_{2R}$, define the stopped martingale $N_n(z):=M_{n\wedge\sigma}(z)$. By \eqref{eq:green-increment-bound}, its increments satisfy
$$
 |N_{k+1}(z)-N_k(z)|
 \le a_k(z):=C(1+|X_k-z|)^{1-d}\mathbf 1_{\{k<\sigma\}},
$$
where $a_k(z)$ is $\mathcal F_k$-measurable. Grouping the visits by site and using the bound on $T_\sigma$, we find that, for $z \in D_{2R}$, 
$$
 \sum_{k\ge0}a_k(z)^2 \le C\sum_{x\in D_R}T_\sigma(x)(1+|x-z|)^{2-2d}
 \le CR\sum_{y\in D_{3R}}(1+|y|)^{2-2d} \le CR\log R.
$$
Applying the Azuma--Hoeffding inequality with predictable increment bounds \cite[Corollary~1(a)]{HRMS} gives
$$
 \mathbb P\left(\sup_n|N_n(z)|\ge R^{3/4}\right)
 \le 2\exp\left\{-\frac{R^{3/2}}{CR\log R}\right\}
 \le 2e^{-c\sqrt R/\log R}.
$$
Finally, observe that on $\mathcal G_R^c$, at least one of these stopped martingales must reach the threshold, so a union bound over the $O(R^d)$ sites of $D_{2R}$ gives
$$
 \mathbb P(\mathcal G_R^c)\le CR^d e^{-c\sqrt R/\log R},
$$
which is $O(R^{-3})$ as claimed. Finally, on $\mathcal G_R$, \eqref{eq:pathwise-occupation} gives $\max_xT_{\tau_R}(x)\le CR$ and 
$$
\tau_R=\sum_{x\in D_R}T_{\tau_R}(x) \le CR \;\sharp D_R \le C_{\rm time} R^{d+1}.
$$

<!-- proposition-2: body_md -->
\label{prop:deterministic-input}
Let $R\ge2$ and fix $A\subset D_R$. Choose $z_0\notin A$ with minimal norm and set $r=|z_0|$. There is a constant $C$ depending only on $d$ such that
\begin{equation}\label{eq:field-rigidity}
 \sup_{z\in\mathbb Z^d}
 \left|W_{R,A}(z)-\left(\frac{r-|z|}{R}\right)_+\right|
 \le C\left(|W_{R,A}(z_0)|+R^{-1/d}\right)^{1/d}.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Set $H=A\setminus D_{r}$. By the choice of $z_0$, we have $D_{r} \subset A$ so $A=D_{r}\sqcup H$. Let us for the time being admit that, for the ball $D_{r}$, it holds that
\begin{equation}\label{eq:ball-field}
 \sup_{z\in\mathbb Z^d}
 \left|W_{R,D_{r}}(z)-\left(\frac{r-|z|}{R}\right)_+\right|
 \le CR^{-1/d}.
\end{equation}
The proof of this estimate follows from a comparison with the continuous setting (where the identity is exact) and is postponed to the end of the proof.

\noindent We first upper bound the number of sites in $H$. For every $x\in H$, we have $|x|\ge|z_0|$; hence
$$
 x\cdot(x-z_0)
 =\frac{|x-z_0|^2+|x|^2-|z_0|^2}{2}
 \ge\frac{|x-z_0|^2}{2},
$$
and, since $|x-z_0|\le2|x|$ and $|x|\le R$, it follows that
$$
 \mathbf n(x)\cdot K(x-z_0)
 \ge\frac{1}{2|x|\,|x-z_0|^{d-2}}
 \ge\frac{1}{(2R)^{d-1}}.
$$
This shows that all sites of $H$ contribute positively to the field at $z_0$, with a uniform lower bound. Consequently, we find that
$$
 W_{R,H}(z_0)\ge c\frac{\sharp H}{R^d}.
$$
At $z_0$, the cone in \eqref{eq:ball-field} vanishes. Hence \eqref{eq:ball-field} and the identity $W_{R,H}=W_{R,A}-W_{R,D_{r}}$ give
\begin{equation}\label{eq:tail-mass}
 \frac{\sharp H}{R^d}
 \le C\left(|W_{R,A}(z_0)|+R^{-1/d}\right).
\end{equation}
We now control the field $W_{R,H}$ uniformly in $z$. We use that $|\mathbf n(x)\cdot K(x-z)|\le|x-z|^{1-d}$ for $x\ne z$ (and the diagonal term vanishes because $K(0)=0$). Let $m:=\sharp H>0$ and split the sum in $W_{R,H}$ between sites at distance at most  $m^{1/d}$ from $z$ and those farther away. For any $z\in\mathbb Z^d$, we find that
$$
\begin{aligned}
 d v_d R|W_{R,H}(z)|
 &\le \sum_{\substack{x\in H\\0<|x-z|\le m^{1/d}}}|x-z|^{1-d}
 +\sum_{\substack{x\in H\\|x-z|>m^{1/d}}}|x-z|^{1-d}\\
 &\le \sum_{0<|y|\le m^{1/d}}|y|^{1-d}
 +m\,m^{(1-d)/d}
 \le Cm^{1/d}.
\end{aligned}
$$
Finally, since $W_{R,A}=W_{R,D_{r}}+W_{R,H}$, combining this with \eqref{eq:tail-mass} and \eqref{eq:ball-field} proves \eqref{eq:field-rigidity}.

\medskip

It remains to prove \eqref{eq:ball-field}.
 Fix $z\in\mathbb Z^d$ and set $F_z(y)=\mathbf n(y)\cdot K(y-z)$ for $y\in\mathbb R^d$. The divergence theorem gives
$$
 \int_{|y|=t}F_z(y)\,\mathrm d S(y)
 =d v_d\,\mathbf 1_{\{t > |z|\}},
 \qquad t>0,\quad t\ne|z|,
$$
where $\,\mathrm d S(y)$ denotes surface measure on the sphere $\{|y|=t\}$. Indeed, $K(\cdot-z)$ has zero divergence away from $z$ and flux $d v_d$ around that point. Integrating over $t$ yields
\begin{equation}\label{eq:continuous-ball}
 \int_{|y|<r}F_z(y)\,\mathrm d y=d v_d(r-|z|)_+.
\end{equation}
This integral is absolutely convergent because $|F_z(y)|\le|y-z|^{1-d}$. This identity explains the conical profile: the flux through a sphere is constant once its radius exceeds $|z|$.

We now compare this integral with the lattice sum using the unit cubes $Q_x=x+[-1/2,1/2)^d$.
Set aside the cubes meeting the sphere $\{|y|=r\}$ or either of the balls of radius $\sqrt R$ centered at $0$ and $z$.
Since $r\le R+1$ and $d\ge2$, their number, and hence their total volume, is at most $C R^{d-1}$.
Their total contribution to the sum and the integral is therefore at most $CR^{1-1/d}$, by the bound for finite sums above and its integral analogue
$$
 \int_E|y-z|^{1-d}\,\mathrm d y\le C\operatorname{vol}(E)^{1/d},
$$
obtained by splitting at distance $\operatorname{vol}(E)^{1/d}$ from $z$. Every remaining cube lies entirely inside or outside the ball. On those inside, both singularities are at distance at least $\sqrt R$, so the mean value theorem gives
$$
 |F_z(x)-F_z(y)|\le CR^{-1/2}|x-z|^{1-d},\qquad \hbox{for } y\in Q_x.
$$
Summing over these cubes costs at most $C\sqrt R$, since the bound for finite sums above gives $\sum_{x\in D_r\setminus\{z\}}|x-z|^{1-d}\le CR$.
Consequently, uniformly in $z$,
$$
 \left|\sum_{x\in D_r}F_z(x)-\int_{|y|<r}F_z(y)\,\mathrm d y\right|
 \le C(R^{1-1/d}+\sqrt R)\le CR^{1-1/d}.
$$
Together with \eqref{eq:continuous-ball}, division by $d v_dR$ proves \eqref{eq:ball-field}.

\label{lem:annulus}
There exist constants $c,C>0$ such that, for all $1/2<\eta<1$, $R\ge2$ and positive integers $L,M$, we have
\begin{equation}\label{eq:annulus-bound}
 \mathbb P\left(\tau_R\le M,\ \sup_{|x|>\eta R}T_{\tau_R}(x)<L\right)
 \le CM\exp\left\{-\frac{c(1-\eta)R}{L}\right\}.
\end{equation}

<!-- proof-3: body_md -->
Choose $c>0$ sufficiently small (whose exact value will be specified later). It suffices to consider $L\le cR$, since otherwise the bound is trivial after increasing $C$. In particular, we may assume $R\ge4$.
Set $\theta:=c/L$ and define $f(x):=e^{\theta|x|}$. Just as in Step 1, we consider again a Doob martingale but associated this time with $f(X_n)$ and previsible increments 
$$
 \mu_k:=\mathbb E[f(X_{k+1})-f(X_k)\mid\mathcal F_k].
$$
We want to control $\mu_k$. To this end, assume $|x|>R/2$ and $e\in\mathcal E$; then Taylor's formula gives
$$
f(x+e)-f(x) \le\theta f(x)\left[\mathbf n(x)\cdot e+C_0\left(R^{-1}+\theta\right)\right],
$$
where the constant $C_0$ depends only on $d$, and the bound is uniform in $R,\theta,x,e$. By assumption \eqref{eq:radial-mean}, the radial component of the next step has conditional mean $-\beta$ at the first departure from $x$ and zero afterward. Thus, taking the conditional expectations yields
$$
 \mu_k\le\theta f(X_k)\left[C_0\left(R^{-1}+\theta\right)
 -\beta\mathbf 1_{\{X_k\notin A_k\}}\right]\quad\hbox{when $|X_k|>\eta R$.}
$$
Consider now the stopping time
$$
 \sigma:=\tau_R\wedge
 \inf\left\{n:\sup_{|x|>\eta R}T_n(x)\ge L\right\} \wedge M.
$$
Now, each site contributes exactly one cookie. Thus, using that $T_\sigma(x)\le L$, we find that, for any site $x$ with $|x|>\eta R$ visited, its contribution is bounded by 
$$
 \begin{aligned}
 \sum_{\substack{k<\sigma\\X_k=x}}\mu_k
 &\le\theta f(x)\left[C_0\left(R^{-1}+\theta\right)T_\sigma(x)-\beta\right]\\
 &\le\theta f(x)\left[C_0\left(R^{-1}+\theta\right)L - \beta\right]
 \le\theta f(x)(2C_0c-\beta) \le 0,
 \end{aligned}
$$
where we used $L/R\le c$ and $L\theta=c$.  The last inequality holds as soon as $c\le\min\{1/4,\beta/(2C_0)\}$ (which depends only on $d$ and $\beta$) so we can now fix $c$ equal to this minimum.

For the time indices such that $|X_k| \leq \eta R$, we just use the crude bound $\mu_k\le e^{\theta(\eta R+1)}$.
Putting things together and applying the optional stopping theorem to the martingale $f(X_n)-\sum_{k<n}\mu_k$ with the (bounded by $M$) stopping time $\sigma$, we find that
$$
 \mathbb E(f(X_\sigma))=1+\mathbb E\big(\sum_{k<\sigma}\mu_k\big)
 \le1+M e^{\theta(\eta R+1)}.
$$
On the event in \eqref{eq:annulus-bound}, we have $\sigma=\tau_R$ and $f(X_\sigma)\ge e^{\theta R}$. Consequently,
$$
 \mathbb P\left(\tau_R\le M,\ \sup_{|x|>\eta R}T_{\tau_R}(x)<L\right) \le e^{-\theta R}\bigl(1+M e^{\theta(\eta R+1)}\bigr) \le CM e^{-c(1-\eta)R/L},
$$
as claimed.

We first identify the exact rate of growth of $\tau_R$. By definition, $\tau_R$ is the sum of the local times over all sites in $D_R$, which has $O(R^d)$ sites, so using \eqref{eq:exit-profile-final} gives, a.s. 
$$
 \tau_R=\sum_{z\in D_R} T_{\tau_R}(z) = 2d \beta\sum_z(R-|z|)_++o(R^{d+1}).
$$
A Riemann-sum approximation yields
$$
 \frac1{R^{d+1}}\sum_z(R-|z|)_+
 \longrightarrow\int_{\mathbb R^d}(1-|y|)_+\,\mathrm d y
 =d v_d\int_0^1(1-r)r^{d-1}\,\mathrm d r
 =\frac{v_d}{d+1}.
$$
Therefore, we conclude that
\begin{equation}\label{eq:clock-final}
 \lim_{R\to\infty} \frac{\tau_R}{R^{d+1}} =  \frac{2d\beta v_d}{d+1} =: C_* \quad\hbox{a.s.}
\end{equation}
We now transfer the result from hitting times to deterministic times using a standard ``sandwich argument'' based on monotonicity of the local times. For $n\ge1$, define $j_n := \sup(j\in \mathbb N :\, \tau_{j}\le n) $. We have
\begin{equation}\label{eq:time-squeeze}
 \tau_{j_n}\le n<\tau_{j_n+1}.
\end{equation}
On  the one hand, using \eqref{eq:clock-final} and recalling that $r_n :=(n/C_*)^{1/(d+1)}$, it follows that 
$$
\frac{j_n}{r_n}\longrightarrow 1.
$$
On the other hand, monotonicity of local times combined with \eqref{eq:time-squeeze} implies, simultaneously for every $z$, that
$$
 T_{\tau_{j_n}}(z)\le T_n(z)\le T_{\tau_{j_n+1}}(z).
$$
Since $j_n/r_n\to 1$ and $r\mapsto(r-|z|)_+$ is Lipschitz, \eqref{eq:exit-profile-final} shows that both the lower and upper bounds are equal to $2d\beta(r_n-|z|)_++o(r_n)$, uniformly in $z$. The same therefore holds for $T_n(z)$, proving 

\begin{equation}\label{eq:deterministic-profile}
\lim_{n\to\infty}\frac1{r_n}
\sup_{z\in\mathbb Z^d}\left|T_n(z)-2d\beta(r_n-|z|)_+\right|=0
\qquad\text{a.s.},
\end{equation}
which is exactly statement \eqref{eq:main-local} of Theorem \ref{thm:main}. In particular, for a fixed site $z$, this gives 
$$
 \frac{T_n(z)}{r_n}
 \longrightarrow2d\beta>0,
$$
which is the limit stated in \eqref{eq:main-pointwise} of Theorem \ref{thm:main} and implies the recurrence of the walk.

We now turn to the range $A_n$. Fix $\varepsilon\in(0,1)$. The approximation \eqref{eq:deterministic-profile} just established gives
$$
 \min_{z\in D_{(1-\varepsilon)r_n}}T_n(z)\ge2d\beta\varepsilon r_n-o(r_n)>0
$$
for all sufficiently large $n$. Thus $D_{(1-\varepsilon)r_n}\subset A_n$. On the other hand, $n<\tau_{j_n+1}$ implies $A_n\subset D_{j_n+1} \subset D_{(1+\varepsilon)r_n}$ for $n$ large since $(j_n+1)/r_n\to 1$. Thus, we have proved that 
$$
D_{(1-\varepsilon)r_n}\subset A_n \subset D_{(1+\varepsilon)r_n} \quad \hbox{for all $n$ large enough a.s.}
$$
Finally, the asymptotics $\sharp A_n \sim v_dr_n^d$ and $\max_{x\in A_n}|x|\sim r_n$ follow trivially from this shape theorem  and the estimate $\sharp D_r\sim v_dr^d$ so the proof of the theorem is complete.
