<!-- definition-1: body_md -->
The \emph{partial free difference quotients} are defined as the derivations
\[
\partial_i:M_0\to L^2(M)\bar\otimes L^2(M^{op}),
\]
which are determined on the generators by
\(
\partial_i(X_j)=\delta_{ij}(1\otimes1).
\)
\newline
We recall that the free difference quotients are \emph{real} derivations: for every $x\in M_0$,
\[
\partial_i(x^*)=(\partial_i(x))^\dagger,
\]
where $(a\otimes b)^\dagger:=b^*\otimes a^*$.
<!-- end definition-1 -->
<!-- remark-1: body_md -->
\label{real}
Suppose that each $\partial_i$ is closable, viewed as an unbounded operator
\(
L^2(M)\to L^2(M)\bar\otimes L^2(M^{op}),
\)
and denote by $\bar\partial_i$ its closure. The row operator $\partial=(\partial_1,\ldots,\partial_n)$ on $M_0$ is also closable; throughout the paper, $\bar\partial$ denotes its closure. Thus $M_0$ is a core for $\bar\partial$, and
\[
\mathrm{Dom}(\bar\partial)\subseteq\bigcap_{i=1}^n\mathrm{Dom}(\bar\partial_i),
\qquad
\bar\partial x=(\bar\partial_1x,\ldots,\bar\partial_nx),
\quad x\in\mathrm{Dom}(\bar\partial).
\]
The real property of the free difference quotients then gives, by the usual double-adjoint argument, that $\mathrm{Dom}(\bar\partial)$ is $^*$--stable and
\[\label{adjunc}
\bar\partial_i(x^*)=(\bar\partial_i(x))^\dagger,
\qquad x\in\mathrm{Dom}(\bar\partial).
\]
We refer to Dabrowski~\cite{Dab10} for the complete argument.
<!-- end remark-1 -->
<!-- definition-2: body_md -->
We define the \emph{free Fisher information} of $(X_1,\ldots,X_n)$ by
\[
\Phi^*(X_1,\ldots,X_n)
:=\sum_{i=1}^n\left\lVert \partial_i^*(1\otimes1)\right\rVert_2^2,
\]
whenever each $\partial_i^*(1\otimes1)$ exists in $L^2(M)$; otherwise, we set $\Phi^*(X)=+\infty$.
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
Whenever $\Phi^*(X_1,\ldots,X_n)<\infty$, each $\partial_i$ is closable as an unbounded operator
\[
L^2(M)\to L^2(M)\bar\otimes L^2(M^{op}).
\]
We denote again by $\bar\partial_i$ its closure, and by $\bar\partial$ the closure of the row operator $\partial=(\partial_1,\ldots,\partial_n)$ on $M_0$, as in Remark~\ref{real}.
<!-- end proposition-1 -->
<!-- definition-3: body_md -->
\label{lip}
Let $M=(W^*(X_1,\ldots,X_n),\tau)$ be a tracial von Neumann algebra. We say that $M$ satisfies the \emph{Lipschitz conjugate variable condition} whenever:
\begin{enumerate}
\item the partial free difference quotients $\partial_i$ are densely defined and closable,
\item the conjugate variables $\xi_i:=\partial_i^*(1\otimes1)$ exist in $L^2(M)$ for every $i=1,\ldots,n$, belong to $\mathrm{Dom}(\bar\partial)$, and satisfy
\[
\bar\partial_j\partial_i^*(1\otimes1)
\in M\bar\otimes M^{op}
\subset L^2(M\bar\otimes M^{op})
\cong L^2(M)\bar\otimes L^2(M^{op})
\cong L^2(M)\bar\otimes L^2(M)
\]
for every $i,j=1,\ldots,n$.
\end{enumerate}

<!-- proposition-2: body_md -->
\label{prop5}
Assume that $M=(W^*(X_1,\ldots,X_n),\tau)$ satisfies the Lipschitz conjugate variable condition. We then have:
\begin{enumerate}
\item $\xi_i:=\partial_i^*(1\otimes1)\in M$ for every $i$,
\item $\xi_i^{(2)}:=\partial_i^*(1\otimes\partial_i^*(1\otimes1))=\partial_i^*(1\otimes\xi_i)=\partial_i^*(\xi_i\otimes1)\in M$ for every $i$,
\item $(\partial_i\otimes\mathrm{id})\circ\partial_j:M_0=\mathbb C\langle X_1,\ldots,X_n\rangle\to L^2(M)^{\bar\otimes3}$ is densely defined and closable for every $i,j=1,\ldots,n$.
\end{enumerate}

<!-- proposition-3: body_md -->
\label{resolvent}
Let $M=W^*(X_1,\ldots,X_n)$ be the tracial von Neumann algebra generated by self-adjoint variables $(X_1,\ldots,X_n)$ in a tracial $W^*$--probability space $(\mathcal M,\tau)$, and suppose that the partial free difference quotients are closable. Then the following assertions hold:
\begin{enumerate}
\item The operator
\(
\Delta:=\partial^*\bar\partial,
\)
where $\bar\partial=(\bar\partial_1,\ldots,\bar\partial_n)$, is the generator of a completely Dirichlet form on $L^2(M)$; see~\cite{Sauvage}. In particular, all its matrix amplifications are again Dirichlet.

\item The resolvent maps
\(
\rho_\alpha:=\alpha(\alpha+\Delta)^{-1},
\)
$\alpha>0$, are normal, unital, tracial $(\tau\circ\rho_\alpha=\tau)$, completely positive, and contractive on both $L^2(M)$ and $M$; see, for instance,~\cite{Ma}. We also have
\[
\left\lVert x-\rho_\alpha(x)\right\rVert\leq2\left\lVert x\right\rVert,
\qquad
\left\lVert x-\rho_\alpha(x)\right\rVert_2\to0
\quad(\alpha\to\infty)
\]
for every $x\in M$, as follows from~\cite[Proposition~2.5]{CiprianiSauvageot2003}.

\item The semigroup $\varphi_t=e^{-t\Delta}$ is strongly continuous with generator $-\Delta$, and
\[
\rho_\alpha
=\alpha\int_0^\infty e^{-\alpha t}\varphi_t\,dt,
\]
where the integral is understood with respect to the strong operator topology on $L^2(M)$.

\item The square-root resolvent
\[
\zeta_\alpha:=\rho_\alpha^{1/2}
=\pi^{-1}\int_0^\infty\frac{t^{-1/2}}{1+t}\,
\rho_{\alpha(1+t)/t}\,dt
\]
satisfies
\[
\mathrm{Ran}(\rho_\alpha)=\mathrm{Dom}(\Delta)\subset\mathrm{Dom}(\bar\partial),
\qquad
\mathrm{Ran}(\zeta_\alpha)=\mathrm{Dom}(\Delta^{1/2})=\mathrm{Dom}(\bar\partial),
\]
and therefore $\bar\partial\circ\zeta_\alpha$ is bounded (see~\cite{Peterson}).
\end{enumerate}

<!-- lemma-1: body_md -->
\label{voipub}
Let $Y\in\mathrm{Dom}(\bar\partial)$ be self-adjoint or non-self-adjoint. Then there exists a constant $C>0$, which depends on the $X_i$ but not on $Y$, such that

\leq
C\sum_{j=1}^n\left\lVert \bar\partial_jY\right\rVert_2^2.

\[
C_P\leq4n\max_{i=1,\ldots,n}\left\lVert X_i\right\rVert^2.

<!-- proposition-5: body_md -->
\label{tensor}
The positive self-adjoint operators $\Delta\otimes \mathrm{id}$ and $\mathrm{id}\otimes\Delta$ on $L^2(M)\bar\otimes L^2(M^{op})$ strongly commute. If $E(\cdot)$ denotes the spectral measure of $\Delta$, with our preceding convention on the $L^2(M^{op})$ leg, their joint spectral measure $\mathbb E$ is characterized by
\[
\mathbb E(B\times C)=E(B)\otimes E(C),
\qquad
B,C\subset[0,\infty)\ \text{Borel}.
\]
The operator
\begin{equation}\label{delta2}
\Delta^\otimes
:=\int_{[0,\infty)^2}(\lambda+\mu)\,d\mathbb E(\lambda,\mu)
\end{equation}
is the closure of $\Delta\otimes \mathrm{id}+\mathrm{id}\otimes\Delta$ on $\mathrm{Dom}(\Delta)\otimes\mathrm{Dom}(\Delta)$ (where $\otimes$ denotes here the algebraic tensor product). Moreover, for every $s\geq0$, we have
\begin{equation}\label{tensorproj}
\mathds1_{[0,s]}(\Delta^\otimes)
\leq
\mathds1_{[0,s]}(\Delta)\otimes\mathds1_{[0,s]}(\Delta).
\end{equation}
Consequently, if $N_\Delta(s):=\operatorname{rank}\mathds1_{[0,s]}(\Delta)<\infty$, we get
\[
\operatorname{rank}\mathds1_{[0,s]}(\Delta^\otimes)
\leq
N_\Delta(s)^2.
\]
Finally,
\begin{equation}\label{tensorker}
\mathrm{Ker}(\Delta^\otimes)
=\mathrm{Ker}(\Delta)\bar\otimes\mathrm{Ker}(\Delta)
=\mathbb C.(1\otimes 1).
\end{equation}
<!-- end proposition-5 -->
<!-- proof-1: body_md -->
The spectral projections of $\Delta\otimes \mathrm{id}$ and $\mathrm{id}\otimes\Delta$ are $E(B)\otimes \mathrm{id}$ and $\mathrm{id}\otimes E(C)$, and they therefore commute strongly. The joint spectral theorem gives the preceding identity together with the operator in~\eqref{delta2}. It remains first to verify the assertion concerning the core. For $n\geq 0$, put 
\(
Q_n:=E([0,n])\otimes E([0,n]).
\)
For every $x\in\mathrm{Dom}(\Delta^\otimes)$, the joint spectral calculus gives $Q_nx\to x$ in the graph norm of $\Delta^\otimes$. On $\mathrm{Ran}(Q_n)$, this graph norm is equivalent to the Hilbert-space norm, and 
\[
E([0,n])L^2(M)\otimes E([0,n])L^2(M^{op})
\]
is dense in $\mathrm{Ran}(Q_n)$ and contained in $\mathrm{Dom}(\Delta)\otimes\mathrm{Dom}(\Delta)$. Hence, the algebraic tensor product is indeed a core.
\newline
We now verify the projection estimate and set
\[
T_s:=\{(\lambda,\mu)\in[0,\infty)^2:\lambda+\mu\leq s\}.
\]
Then
\[
\mathds1_{[0,s]}(\Delta^\otimes)=\mathbb E(T_s),
\qquad
\mathds1_{[0,s]}(\Delta)\otimes\mathds1_{[0,s]}(\Delta)
=\mathbb E([0,s]^2).
\]
Since $0\leq \lambda+\mu\leq s$ implies $0\leq \lambda,\mu\leq s$, we have the immediate inclusion $T_s\subset[0,s]^2$, and the monotonicity of projection-valued measures yields~\eqref{tensorproj}. The range of the projection in the right-hand side of~\eqref{tensorproj} has dimension $N_\Delta(s)^2$, proving the desired rank estimate. Finally, since $\lambda,\mu\geq0$, the equality $\lambda+\mu=0$ is equivalent to $\lambda=\mu=0$, and~\eqref{tensorker} follows immediately from~\eqref{kernel}. This concludes the proof.

<!-- definition-4: body_md -->
\label{rightaction}
Let $(M,\tau)$ be a tracial $W^*$--probability space. We first define on the algebraic tensor product $L^2(M)\otimes L^2(M^{op})$, for $x\otimes y\in M\otimes M^{op}$ and $a\otimes b\in L^2(M)\otimes L^2(M^{op})$,
\[
(a\otimes b)\sharp(x\otimes y):=ax\otimes by.
\]
Here the product on the second tensor leg is taken in $M^{op}$. By bilinearity and continuity, this extends to a bounded right action of $M\bar\otimes M^{op}$ on $L^2(M)\bar\otimes L^2(M^{op})$.

For $n\in\mathbb N$ and $T=(T_{ij})_{i,j=1}^n\in M_n(M\bar\otimes M^{op})$, we then define the \emph{right-leg} multiplication
\[
\mathcal R_T:
(L^2(M)\bar\otimes L^2(M^{op}))^{\oplus n}
\to
(L^2(M)\bar\otimes L^2(M^{op}))^{\oplus n}
\]
by
\[
(\mathcal R_T\eta)_i:=\sum_{j=1}^n\eta_j\sharp T_{ji},
\qquad i=1,\ldots,n,
\]
for $\eta=(\eta_1,\ldots,\eta_n)\in(L^2(M)\bar\otimes L^2(M^{op}))^{\oplus n}$.
<!-- end definition-4 -->
<!-- proposition-6: body_md -->
\label{rightprop}
For each $n\geq1$ and $T,S\in M_n(M\bar\otimes M^{op})$, the operators introduced in Definition~\ref{rightaction} enjoy the following properties:
\begin{enumerate}
  \item \emph{Linearity:} $\mathcal R_{T+S}=\mathcal R_T+\mathcal R_S$.
  \item \emph{Boundedness:}
  \[
  \left\lVert \mathcal R_T\right\rVert\leq \left\lVert T\right\rVert_{M_n(M\bar\otimes M^{op})}.
  \]
  Hence
  \[
  \mathcal R_T\in B\!\left((L^2(M)\bar\otimes L^2(M^{op}))^{\oplus n}\right)
  \cong M_n\!\left(B(L^2(M)\bar\otimes L^2(M^{op}))\right).
  \]
  \item \emph{$^*$--structure:} $\mathcal R_{T^*}=(\mathcal R_T)^*$, where $(T^*)_{ij}=T_{ji}^*$ for the usual $^*$--involution on $M\bar\otimes M^{op}$.


\item $\mathcal R_{(1\otimes 1)\otimes I_n}=\mathrm{id}_{(L^2(M){\bar\otimes}\,L^2(M^{op}))^{\oplus n}}$.
  \item \emph{Multiplication rule:} Writing $TS$ for the matrix product formed using the multiplication $\sharp$ in $M\bar\otimes M^{op}$, we have
  

\[
  \mathcal R_S\circ\mathcal R_T=\mathcal R_{T S}=\mathcal R_{T\sharp S}
  \qquad\text{for all }T,S\in M_n(M\bar\otimes M^{op}).
  \]
In particular, whenever $T$ is invertible in $M_n(M\bar\otimes M^{op})$, the operator $\mathcal R_T$ is invertible and $(\mathcal R_T)^{-1}=\mathcal R_{T^{-1}}$.


    \item \emph{Positivity:}
  if $T\ge 0$ in $M_n(M\bar\otimes M^{op})$ (resp.\ $T>0$, i.e.\ $T\ge c.(1\otimes 1)\otimes I_n$ for some $c>0$), then 
  $\mathcal R_T\ge 0$ (resp.\ $\mathcal R_T\ge c\,\mathrm{id}$) as an operator on 
  $(L^2(M)\bar\otimes L^2(M^{op}))^{\oplus n}$.
\end{enumerate}
Thus, $T\mapsto\mathcal R_T$ is a $*$--anti-representation of $M_n(M\bar\otimes M^{op})$, or equivalently a $*$--representation of the opposite algebra $M_n(M\bar\otimes M^{op})^{op}$.

<!-- proposition-7: body_md -->
\label{proplip}
Under Definition~\ref{lip}, the matrix
\[
\mathscr J\xi
=(\bar\partial_j\xi_i)_{i,j=1}^n
\in M_n(M\bar\otimes M^{op})
\]
is self-adjoint.
<!-- end proposition-7 -->
<!-- proof-2: body_md -->
This was proved in~\cite[Lemma~2.19]{DiezObata}. We nevertheless recall the short argument for the convenience of the reader.
\newline
We denote by $\dagger$ the isometric Hilbert--Schmidt involution on $M\otimes M^{op}$, given on simple tensors by
\(
(a\otimes b^{op})^\dagger=b^*\otimes(a^*)^{op},
\)
and let
\(
\sigma(a\otimes b^{op})=b\otimes a^{op}
\)
denote the flip. The real property of the free difference quotients first gives
\(
(\bar\partial_j\xi_i)^\dagger=\bar\partial_j\xi_i,
\)
whereas the Schwarz symmetry of the conjugate system gives
\(
\bar\partial_j\xi_i=\sigma(\bar\partial_i\xi_j)
\). Since $\sigma$ is an involutive $*$--anti-automorphism, we obtain
    

\[
      (\bar\partial_j\xi_i)^\ast=\sigma\big((\bar\partial_j\xi_i)^{\dagger}\big)
      =\sigma(\bar\partial_j\xi_i)=\bar\partial_i\xi_j.
    \]
This proves the self-adjointness of $\mathscr J\xi$.

<!-- lemma-2: body_md -->
\label{dab}
Under the assumptions in Definition~\ref{lip}, we have,                     
\begin{enumerate}
\item
                                           
      
Set $\Delta_j:=\partial_j^*\bar\partial_j$. For any $x\in M_0$, we have

<!-- theorem-1: body_md -->
\label{main}
Let $1\leq n<\infty$, and let $X=(X_1,\ldots,X_n)$ satisfy $CD(c,\infty)$ for some $c>0$. Then $\Delta=\partial^*\bar\partial$ has compact resolvent. More exactly, we have
\begin{equation}\label{gap}
N_\Delta(t)=1,
\qquad
0\leq t<c,
\end{equation}
and
\begin{equation}\label{count}
N_\Delta(t)-1
\leq
nN_\Delta(t-c)^2,
\qquad
t\geq c.
\end{equation}

<!-- proof-3: body_md -->
For $t\geq0$, we denote by
\[
E_t:=\mathds1_{(0,t]}(\Delta)L^2(M)
\]
the positive spectral subspace of $\Delta$ below $t$. Fix $f\in E_t$, and denote by $\nu_f$ its scalar spectral measure with respect to $\Delta$:
\[
\nu_f(B):=\left\langle \mathds1_B(\Delta)f,f\right\rangle_2,
\qquad
B\subset[0,\infty)\ \text{Borel}.
\]
By definition, the measure $\nu_f$ is supported in $(0,t]$. Thus, for every $r>0$, we get
\[
\int_{[0,\infty)}\lambda^{2r}\,d\nu_f(\lambda)
=
\int_{(0,t]}\lambda^{2r}\,d\nu_f(\lambda)
\leq
t^{2r}\left\lVert f\right\rVert_2^2
<\infty.
\]
The spectral characterization of the domains of the powers of a positive self-adjoint operator therefore shows that $f\in\mathrm{Dom}(\Delta^r)$ for every $r\geq0$. In particular,
\(
f\in\mathrm{Dom}(\Delta^{3/2})
\)
and
\(
\Delta f\in\mathrm{Dom}(\Delta^{1/2})=\mathrm{Dom}(\bar\partial).
\)
\newline
We may now apply Dabrowski's almost-commutation formula from Lemma~\ref{dab}. It gives $\bar\partial_i f\in\mathrm{Dom}(\Delta^\otimes)$ for every $i=1,\ldots,n$, together with the fundamental identity
\[
\bar\partial\Delta f
=
(\Delta^\otimes\otimes I_n+\mathcal R_{\mathscr J\xi})\bar\partial f.
\]
Taking the scalar product of this identity with $\bar\partial f$, we obtain
\begin{align*}
\sum_{i=1}^n\left\langle \Delta^\otimes\bar\partial_i f,\bar\partial_i f\right\rangle_2
&=
\sum_{i=1}^n\left\langle \bar\partial_i\Delta f,\bar\partial_i f\right\rangle_2
-
\left\langle \mathcal R_{\mathscr J\xi}\bar\partial f,\bar\partial f\right\rangle_2.
\end{align*}
But $\Delta f\in\mathrm{Dom}(\bar\partial)$ and $f\in\mathrm{Dom}(\Delta)$. Therefore, the duality for the closed Dirichlet form yields
\[
\sum_{i=1}^n\left\langle \bar\partial_i\Delta f,\bar\partial_i f\right\rangle_2
=
\left\langle \bar\partial\Delta f,\bar\partial f\right\rangle_2
=
\left\langle \Delta f,\Delta f\right\rangle_2
=
\left\lVert \Delta f\right\rVert_2^2.
\]
We therefore have the identity
\[
\sum_{i=1}^n\left\langle \Delta^\otimes\bar\partial_i f,\bar\partial_i f\right\rangle_2
=
\left\lVert \Delta f\right\rVert_2^2
-
\left\langle \mathcal R_{\mathscr J\xi}\bar\partial f,\bar\partial f\right\rangle_2.
\]
We now use in an essential way the fact that $\nu_f$ is supported in $(0,t]$. Indeed,
\begin{align*}
\left\lVert \Delta f\right\rVert_2^2
&=
\int_{(0,t]}\lambda^2\,d\nu_f(\lambda)\\
&\leq
t\int_{(0,t]}\lambda\,d\nu_f(\lambda)\\
&=
t\left\langle \Delta f,f\right\rangle_2
=
t\left\lVert \bar\partial f\right\rVert_2^2.
\end{align*}
On the other hand, the condition $CD(c,\infty)$ combined with Proposition~\ref{rightprop} gives
\[
\left\langle \mathcal R_{\mathscr J\xi}\bar\partial f,\bar\partial f\right\rangle_2
\geq
c\left\lVert \bar\partial f\right\rVert_2^2.
\]
Combining all the preceding identities and estimates, we arrive at the following spectral-lowering estimate
\begin{align}
\left\langle (\Delta^\otimes\otimes I_n)\bar\partial f,\bar\partial f\right\rangle_2
&=
\sum_{i=1}^n\left\langle \Delta^\otimes\bar\partial_i f,\bar\partial_i f\right\rangle_2\nonumber\\
&\leq
(t-c)\left\lVert \bar\partial f\right\rVert_2^2.
\label{lower}
\end{align}
We first consider the interval $0\leq t<c$. The left-hand side of~\eqref{lower} is non-negative since $\Delta^\otimes\geq0$, and its right-hand side is strictly negative unless $\bar\partial f=0$. We thus necessarily have $\bar\partial f=0$, and~\eqref{kernel} shows that $f$ is scalar. But also
\[
E_t\perp\mathrm{Ker}(\Delta)=\mathbb C\,1,
\]
because $E_t$ and $\mathrm{Ker}(\Delta)$ are the ranges of the spectral projections corresponding to the disjoint Borel sets $(0,t]$ and $\{0\}$. We obtain $f=0$, hence $E_t=\{0\}$, and consequently
\[
N_\Delta(t)=\dim\mathrm{Ker}(\Delta)=1,
\qquad
0\leq t<c.
\]
This proves~\eqref{gap}, and in particular already proves that the spectral gap is forced by the convexity criterion.
\newline
We turn to the more interesting case $t\geq c$. Set $s=t-c$, and assume for the moment that
\[
N_\Delta(s)<\infty.
\]
We consider the following natural lowering map:
\begin{align}
E_t
&\to
\left(
\mathds1_{[0,s]}(\Delta^\otimes)
\big(L^2(M)\bar\otimes L^2(M^{op})\big)
\right)^{\oplus n},
\nonumber\\
f
&\mapsto
\big(
\mathds1_{[0,s]}(\Delta^\otimes)\bar\partial_1f,
\ldots,
\mathds1_{[0,s]}(\Delta^\otimes)\bar\partial_nf
\big).
\label{lowmap}
\end{align}
The main point is that this map is injective. To see it, let $f\in E_t$ and suppose that
\[
\mathds1_{[0,s]}(\Delta^\otimes)\bar\partial_i f=0,
\qquad
i=1,\ldots,n.
\]
For every $i=1,\ldots,n$, denote by
\[
\nu_{i,f}(B)
:=
\left\langle \mathds1_B(\Delta^\otimes)\bar\partial_i f,\bar\partial_i f\right\rangle_2,
\qquad
B\subset[0,\infty)\ \text{Borel},
\]
the scalar spectral measure of $\bar\partial_i f$ with respect to $\Delta^\otimes$. Our assumption gives $\nu_{i,f}([0,s])=0$. Moreover, since $\bar\partial_i f\in\mathrm{Dom}(\Delta^\otimes)$, the first moment of $\nu_{i,f}$ is finite, and we have
\begin{align*}
\sum_{i=1}^n
\left(
\left\langle \Delta^\otimes\bar\partial_i f,\bar\partial_i f\right\rangle_2
-
s\left\lVert \bar\partial_i f\right\rVert_2^2
\right)
=
\sum_{i=1}^n
\int_{(s,\infty)}(\lambda-s)\,d\nu_{i,f}(\lambda).
\end{align*}
Assume by contradiction that $\bar\partial f\neq0$. Then $\bar\partial_i f\neq0$ for at least one $i$, and consequently
\[
\nu_{i,f}((s,\infty))
=
\left\lVert \bar\partial_i f\right\rVert_2^2
>0.
\]
Since
\(
(s,\infty)=\bigcup_{m\geq1}[s+m^{-1},\infty),
\)
we can find $m\geq1$ such that $\nu_{i,f}([s+m^{-1},\infty))>0$. It follows at once that
\[
\int_{(s,\infty)}(\lambda-s)\,d\nu_{i,f}(\lambda)
\geq
\frac1m\nu_{i,f}([s+m^{-1},\infty))
>0.
\]
Thus, the right-hand side above is strictly positive, and we obtain
\[
\sum_{i=1}^n\left\langle \Delta^\otimes\bar\partial_i f,\bar\partial_i f\right\rangle_2
>
s\left\lVert \bar\partial f\right\rVert_2^2.
\]
But $s=t-c$, and this is in direct contradiction with~\eqref{lower}. Hence $\bar\partial f=0$. Using again~\eqref{kernel}, together with $f\perp1$, we get $f=0$. This proves that the map~\eqref{lowmap} is injective.
\newline
The target space in~\eqref{lowmap} is the $n$--fold direct sum of the range of $\mathds1_{[0,s]}(\Delta^\otimes)$. By Proposition~\ref{tensor} and our assumption $N_\Delta(s)<\infty$, this space is finite-dimensional. The injectivity proved above now gives
\begin{align*}
\dim E_t
&\leq
n\,\operatorname{rank}\mathds1_{[0,s]}(\Delta^\otimes)\\
&\leq
nN_\Delta(s)^2
=
nN_\Delta(t-c)^2.
\end{align*}
Moreover, the spectral calculus gives the orthogonal decomposition
\[
\mathds1_{[0,t]}(\Delta)L^2(M)
=
\mathrm{Ker}(\Delta)\oplus E_t,
\]
and we already know that $\dim\mathrm{Ker}(\Delta)=1$. We have thus established
\begin{equation}\label{count2}
N_\Delta(t)-1
\leq
nN_\Delta(t-c)^2
\end{equation}
under the only condition that $t\geq c$ and $N_\Delta(t-c)<\infty$.
\newline
It only remains to prove that $N_\Delta(t)<\infty$ for every finite $t$, and thus to remove this (apparently circular) finiteness assumption. We do this by a very simple induction. For $k\geq0$, consider the assertion
\[
\mathcal P(k):
\qquad
N_\Delta(t)<\infty
\quad\text{for every }0\leq t\leq kc.
\]
The case $k=0$ follows from $N_\Delta(0)=\dim\mathrm{Ker}(\Delta)=1$. Suppose that $\mathcal P(k)$ holds for some $k\geq0$, and take $0\leq t\leq(k+1)c$. If $0\leq t<c$, then~\eqref{gap} gives $N_\Delta(t)=1$. If instead $c\leq t\leq(k+1)c$, then $0\leq t-c\leq kc$, and the induction hypothesis gives $N_\Delta(t-c)<\infty$. We can therefore apply the conditional estimate~\eqref{count2}, which yields
\[
N_\Delta(t)
\leq
1+nN_\Delta(t-c)^2
<\infty.
\]
This proves $\mathcal P(k+1)$. By induction, $N_\Delta(t)<\infty$ for every finite $t$. In particular, the finiteness assumption in~\eqref{count2} is now automatically satisfied for every $t\geq c$, and~\eqref{count} follows.
\newline
We finally deduce compactness of the resolvent. For $R>0$, the spectral calculus gives
\[
(1+\Delta)^{-1}
=
\mathds1_{[0,R]}(\Delta)(1+\Delta)^{-1}
+
\mathds1_{(R,\infty)}(\Delta)(1+\Delta)^{-1}.
\]
The first operator has finite rank, since $N_\Delta(R)<\infty$, while
\[
\left\lVert \mathds1_{(R,\infty)}(\Delta)(1+\Delta)^{-1}\right\rVert
\leq
\frac1{1+R}.
\]
We have proved that $(1+\Delta)^{-1}$ is a norm limit of finite-rank operators. It is therefore compact, and the proof is complete.

\label{lmacumul}
Let \(\alpha\) and \(\beta\) be locally finite positive Borel measures on \([0,\infty)\), and let \(s\geq0\). Assume that \[ \alpha([0,u]) \leq \beta([0,u]), \qquad \text{for all}\quad 0\leq u\leq s. \]
Then \begin{equation}\label{convdom} (\alpha*\alpha)([0,s]) \leq (\beta*\beta)([0,s]). \end{equation}

Indeed, we have
\begin{align*}
(\alpha*\alpha)([0,s])
&=
\int_{[0,s]}
\alpha([0,s-u])\,d\alpha(u)\\
&\leq
\int_{[0,s]}
\beta([0,s-u])\,d\alpha(u)\\
&=
(\alpha*\beta)([0,s])\\
&=
\int_{[0,s]}
\alpha([0,s-u])\,d\beta(u)\\
&\leq
\int_{[0,s]}
\beta([0,s-u])\,d\beta(u)\\
&=
(\beta*\beta)([0,s]).
\end{align*}
Here both inequalities follow directly from the cumulative domination

identity follows from the
commutativity of classical convolution and Tonelli's theorem.

By Theorem~\ref{main}, \(\Delta\) has compact resolvent. We can
therefore consider its spectral counting measure (namely, the trace of its spectral measure),
\(
\mu_\Delta
:=
\sum_{\lambda\in\sigma(\Delta)}
\dim\mathrm{Ker}(\Delta-\lambda\:\mathrm{id})\,\delta_\lambda,
\)
so that
\(
N_\Delta(t)=\mu_\Delta([0,t]).
\)
The only change in the preceding proof occurs when estimating the
dimension of the target of~\eqref{lowmap}. Indeed, the joint
spectral calculus recalled in Proposition~\ref{tensor} gives the precise identity
\[
\operatorname{rank}\mathds1_{[0,s]}(\Delta^\otimes)
=\sum_{\substack{\lambda,\mu\in\sigma(\Delta)\\
                \lambda+\mu\leq s}}
\dim\mathrm{Ker}(\Delta-\lambda\:\mathrm{id})\,
\dim\mathrm{Ker}(\Delta-\mu\:\mathrm{id})=
(\mu_\Delta*\mu_\Delta)([0,s]).
\]
The injectivity of~\eqref{lowmap} therefore yields
\begin{equation}\label{conv}
N_\Delta(t)-1
\leq
n(\mu_\Delta*\mu_\Delta)([0,t-c]),
\qquad
t\geq c.
\end{equation}
\newline
The form of inequality~\eqref{conv} suggests looking for a
discrete measure which satisfies the same relation with equality.
Since each successive application of the descent procedure decreases the
spectral level by \(c\), it is natural to look for such a measure supported
on the non-negative lattice \(c\mathbb Z_{\geq 0}=\{kc:k\geq0\}\).
\newline
Let us write
\(
\nu
:=
\sum_{k=0}^{\infty}a_k\,\delta_{kc},
\)
where \(a_k\geq0\). We impose the fixed-point convolution equation
\(
\nu
=
\delta_0+n\,\delta_c*(\nu*\nu).
\)
Comparing first the masses at \(0\), and then at \((k+1)c\), we obtain
\[
a_0=1,
\qquad
a_{k+1}
=
n\sum_{j=0}^{k}a_ja_{k-j},
\qquad
k\geq0.
\]
Since a locally finite measure supported on \(c\mathbb Z_{\geq0}\) is uniquely determined by the sequence of its point masses, this recursion determines \(\nu\) uniquely. The Catalan recursion therefore gives
\(
a_k=C_kn^k
\)
for every \(k\geq0\).
\bigbreak
Consequently, the unique locally finite positive measure supported on \(c\mathbb Z_{\geq0}\) which satisfies the preceding fixed-point relation is
\(
\nu
:=
\sum_{k=0}^{\infty}C_kn^k\,\delta_{kc}.
\)
It is locally finite since every bounded interval contains only finitely many points of \(c\mathbb Z_{\geq0}\). The Catalan recursion
\(
C_{k+1}=\sum_{j=0}^{k}C_jC_{k-j}
\)
is equivalent to the convolution identity
\(
\nu
=
\delta_0+n\,\delta_c*(\nu*\nu).
\)
Thus, setting \(G(t):=\nu([0,t])\), we have
\(
G(t)-1
=
n(\nu*\nu)([0,t-c]),
\) for
\(t\geq c\).
\newline
We claim that \(N_\Delta(t)\leq G(t)\) for every \(t\geq0\).
By~\eqref{gap}, this holds for \(0\leq t<c\), while
\eqref{conv} at \(t=c\) gives
\[
N_\Delta(c)\leq1+n=G(c).
\]
Suppose that the claim holds on \([0,kc]\) for some \(k\geq1\), and
take \(kc\leq t\leq(k+1)c\). Set \(s=t-c\), so that \(s\leq kc\).
For every \(0\leq r\leq s\), the induction hypothesis gives
\[
\mu_\Delta([0,r])
=
N_\Delta(r)
\leq
G(r)
=
\nu([0,r]).
\]
Now using Lemma~\ref{lmacumul}, applied with
\(\alpha=\mu_\Delta\) and \(\beta=\nu\), gives
\[
(\mu_\Delta*\mu_\Delta)([0,s])
\leq
(\nu*\nu)([0,s]).
\]
Combining this with~\eqref{conv}, we obtain
\[
N_\Delta(t)-1
\leq
n(\nu*\nu)([0,t-c])
=
G(t)-1.
\]
The assertion thus follows by induction.
\newline
Using the elementary bound
\(C_k\leq4^k\), we conclude that
\[
N_\Delta(t)
\leq
\sum_{k=0}^{\lfloor t/c\rfloor}C_kn^k
\leq
\sum_{k=0}^{\lfloor t/c\rfloor}(4n)^k
=
\frac{(4n)^{\lfloor t/c\rfloor+1}-1}{4n-1},
\]
which proves the first estimate.
\newline
It remains to establish the trace-class estimate of the heat semigroup. Fix \(t>0\) (at this stage, all
the quantities below initially are understood as elements of \([0,\infty]\)). For every
\(\lambda\geq0\), one has
\[
e^{-t\lambda}
=
t\int_0^\infty
e^{-tr}\mathds1_{[0,r]}(\lambda)\,dr.
\]
Since \(N_\Delta(r)\leq G(r)\) for every \(r\geq0\), Tonelli's theorem
therefore gives
\begin{align*}
\operatorname{Tr}_{L^2(M)}(e^{-t\Delta})
&=
\int_{[0,\infty)}e^{-t\lambda}\,d\mu_\Delta(\lambda)\\
&=
t\int_0^\infty e^{-tr}
\left(
\int_{[0,\infty)}
\mathds1_{[0,r]}(\lambda)\,d\mu_\Delta(\lambda)
\right)dr\\
&=
t\int_0^\infty e^{-tr}\mu_\Delta([0,r])\,dr\\
&\leq
t\int_0^\infty e^{-tr}\nu([0,r])\,dr\\
&=
\int_{[0,\infty)}e^{-t\lambda}\,d\nu(\lambda)\\
&=
\sum_{k=0}^{\infty}C_k(ne^{-tc})^k.
\end{align*}
If \(C\) denotes the generating function of the Catalan numbers, then, for every \(0<x\leq1/4\),
\newline
\(
C(x):=\sum_{k=0}^{\infty}C_kx^k=\frac{1-\sqrt{1-4x}}{2x}.
\)
Taking \(x=ne^{-t c}\), we finally obtain
\[
\operatorname{Tr}_{L^2(M)}(e^{-t\Delta})
\leq
\frac{1-\sqrt{1-4ne^{-tc}}}{2ne^{-tc}}=\frac{2}{1+\sqrt{1-4ne^{-t c}}}
\]
when
\(
t\geq \frac{\log(4n)}{c}.
\)

\label{gibbs}
Let
\(
\mathscr P:=\mathbb C\langle t_1,\ldots,t_n\rangle
\)
denote the free algebra in $n$ self-adjoint non-commuting indeterminates, and let $V=V^*\in\mathscr P$. We say that a bounded self-adjoint tuple $X=(X_1,\ldots,X_n)$ is a free Gibbs realization for $V$ when its tracial law $\tau_V$ satisfies the Schwinger--Dyson equations
\begin{equation}\label{schwin}
\tau_V(\mathscr D_iV(X)P(X))
=
(\tau_V\otimes\tau_V)(\partial_iP(X)),
\qquad
P\in\mathscr P,\quad i=1,\ldots,n.
\end{equation}
These equations immediately identify the conjugate variables of $X$ as
\[
\xi_i=\mathscr D_iV(X).
\]
Here $\mathscr DV=(\mathscr D_1V,\ldots,\mathscr D_nV)$ is the cyclic gradient, and $\mathscr D_iV=m\circ\sigma(\partial_iV)$ is its $i$--th component.
In particular, Dabrowski~\cite[Lemma~37]{Dab14} shows that $X$ is algebraically free, so evaluation identifies $\mathscr P$ with
\(
M_0=\mathbb C\langle X_1,\ldots,X_n\rangle.
\)
Moreover, since $V$ is polynomial and $X$ is bounded, we have
\[
\bar\partial_j\xi_i
=
\partial_j\mathscr D_iV(X)
\in
M_0\otimes M_0^{op}
\subset
M\bar\otimes M^{op}.
\]
Thus, in this important situation, the conjugate variables are automatically Lipschitz, and
\[
\mathscr J\xi
=
\mathscr J\mathscr DV(X)
\in
M_n(M_0\otimes M_0^{op})
\subset
M_n(M\bar\otimes M^{op}).
\]

Let $(S_t^{(i)})_{t\geq0}$, $1\leq i\leq n$, be free (standard) Brownian motions,
free from $M$. Set
\[
 N_t=W^*(S_s^{(i)}:0\leq s\leq t,\ 1\leq i\leq n),
 \qquad N=W^*(N_t:\ t\geq0),
\]
and $\widetilde M=M*N$, $\widetilde M_t=M*N_t$.
The trace on $\widetilde M$ is again denoted without confusion by $\tau$, and we have
$N\cong L(\mathbb F_\infty)$.
Following~\cite[Section~2.1]{Dab14}, extend the polynomial free difference
quotient by zero on the free Brownian variables:
\[
 \mathrm{Dom}(\delta)=M_0*
 \mathbb C\langle S_t^{(i)}:t\geq0,\ 1\leq i\leq n\rangle,
 \qquad \delta_i|_{M_0}=\partial_i,
 \quad \delta_i(S_t^{(j)})=0,
\]
where the free product is algebraic. This derivation is closable.
Denote its joint closure by $\bar\delta$ and put, for $\alpha>0$,
\[
 \widetilde\Delta=\delta^*\bar\delta,\qquad
 \widetilde\varphi_t=e^{-t\widetilde\Delta},\qquad
 \widetilde\rho_\alpha=\alpha(\alpha+\widetilde\Delta)^{-1}.
\]
The subspace $L^2(M)$ reduces $\widetilde\Delta$, whose restriction there
is $\Delta$. We also write
\[
 \widetilde\Delta^\otimes
 =\overline{\widetilde\Delta\otimes\mathrm{id}
                 +\mathrm{id}\otimes\widetilde\Delta},\qquad
 \widetilde\varphi_t^\otimes=e^{-t\widetilde\Delta^\otimes},\qquad
 \widetilde\rho_\alpha^\otimes
       =\alpha(\alpha+\widetilde\Delta^\otimes)^{-1}.
\]
Tensor operators act componentwise on the $n$-fold direct sum.
As usual, we use the notation
$(a\otimes b^{op})\#dS_t^{(i)}=a\,dS_t^{(i)}b$ for stochastic integration.
The filtration properties of these operators are those established
in~\cite[Section~2.1, Lemma~16]{Dab14}.

The Lipschitz conjugate variable hypothesis satisfies~\cite[Assumption~1]{Dab14}, as explained in the proof of~\cite[Corollary~25]{Dab14} there.
The results of~\cite[Theorems~17 and~11(ii)]{Dab14}, together with~\cite[Propositions~23--24]{Dab14}, therefore
provide normal unital trace-preserving embeddings
$\Phi_t:M\to\widetilde M_t$ on this free product.
For $x\in M_0$, the process $Y_t=\Phi_t(x)$ is the mild solution
\begin{equation}\label{eq:embedding-mild}
 Y_t=\widetilde\varphi_{t/2}(x)+\gamma(Y)_t,\qquad
 \gamma(Y)_t=\sum_{i=1}^n\int_0^t
   \widetilde\varphi_{(t-s)/2}^\otimes
                  (\bar\delta_iY_s)\#dS_s^{(i)}.
\end{equation}
We use its continuous $L^2$ representative. In particular,
$Y\in\mathcal B_{2,\bar\delta}^{a}([0,T])$ for every $T<\infty$,
where the adapted graph spaces are those of~\cite[Section~1.2]{Dab14} (the factors $1/2$ in~\eqref{eq:embedding-mild} come from Dabrowski's
normalization~\cite{Dab14}, which uses the generator $-\widetilde\Delta/2$).
\newline
For $\epsilon\in(0,1)$, let $Y^\epsilon$ be the regularized solution
from~\cite[Theorem~11(ii)]{Dab14}:
\begin{equation}\label{eq:embedding-epsilon}
 Y_t^\epsilon=\widetilde\varphi_{t/2}(x)
                  +(1-\epsilon)\gamma(Y^\epsilon)_t.
\end{equation}
Since the two derivations in that theorem coincide here, $Y^\epsilon$
is a strong solution and belongs to
$\mathcal B_{2,\widetilde\Delta}^{a}([0,T])$ for every $0<T<\infty$.
The spaces $\mathcal B_{2,\bar\delta}^{a}([0,T])$ and
$\mathcal B_{2,\widetilde\Delta}^{a}([0,T])$ are the completions
of the corresponding simple adapted processes for the respective
norms
\[
 \|X\|_{\mathcal B_{2,\bar\delta}^{a}([0,T])}
 :=
 \left(
 \int_0^T
 \left[
 \|\bar\delta X_s\|_2^2
 +\|X_s\|_2^2
 \right]ds
 \right)^{1/2}
\]
and
\[
 \|X\|_{\mathcal B_{2,\widetilde\Delta}^{a}([0,T])}
 :=
 \left(
 \int_0^T\sum_{b=0}^{2}
 \|\widetilde\Delta^{b/2}X_s\|_2^2\,ds
 \right)^{1/2}.
\]
These correspond to the choices $\beta=0$ and $\beta=1$,
respectively, in the definitions of~\cite[Section~1.2]{Dab14}.
In particular, $Y_s^\epsilon\in\mathrm{Dom}(\widetilde\Delta)$
for almost every $s\in[0,T]$, and
\[
 \int_0^T\|\widetilde\Delta Y_s^\epsilon\|_2^2\,ds<\infty.
\]
Moreover, as $\epsilon\downarrow0$, $Y^\epsilon$ converges to $Y$
weakly in $\mathcal B_{2,\bar\delta}^{a}([0,T])$
and uniformly on $[0,T]$ in $L^2(\widetilde M)$.
\newline
We first record the coercivity given by positive curvature, noting that the conjugate variables of the zero extension remain $\xi_i$.
By using again Lemma~\ref{dab}, in its zero-extension form from~\cite[Lemma~19(iii)]{Dab14},
\[
 \bar\delta\widetilde\Delta u
 =(\widetilde\Delta^\otimes\otimes I_n
                 +\mathcal R_{\mathscr J\xi})\bar\delta u,
 \qquad u\in\mathrm{Dom}(\widetilde\Delta^{3/2}),
\]
including the corresponding domain assertion.
The spatial inclusion
$M\bar\otimes M^{op}\subset\widetilde M\bar\otimes\widetilde M^{op}$
preserves positivity. By the representation properties recalled
above in Proposition~\ref{rightprop}, we also have
$\mathcal R_{\mathscr J\xi}\geq c\,\mathrm{id}$ on the enlarged
gradient space. The free Bochner formula (taking the inner product with $\bar\delta u$)
therefore gives
\[
 \|\widetilde\Delta u\|_2^2
 =\|(\widetilde\Delta^\otimes\otimes I_n)^{1/2}\bar\delta u\|_2^2
   +\langle\mathcal R_{\mathscr J\xi}\bar\delta u,\bar\delta u\rangle_2
 \geq
 \|(\widetilde\Delta^\otimes\otimes I_n)^{1/2}\bar\delta u\|_2^2
   +c\|\bar\delta u\|_2^2.
\]
For $u\in\mathrm{Dom}(\widetilde\Delta)$, apply this inequality to
$\widetilde\rho_\alpha u\in\mathrm{Dom}(\widetilde\Delta^{3/2})$.
We have convergence to $u$ in the graph norm of $\widetilde\Delta$,
and the free difference quotients converge to $\bar\delta u$ in $L^2$.
Applying the same inequality to
$\widetilde\rho_\alpha u-\widetilde\rho_\beta u$, we see that
$(\widetilde\Delta^\otimes\otimes I_n)^{1/2}\bar\delta\widetilde\rho_\alpha u$
is Cauchy in $L^2$. By closedness, we therefore obtain
$\bar\delta u\in\mathrm{Dom}((\widetilde\Delta^\otimes\otimes I_n)^{1/2})$
and can pass to the limit in the inequality
(see also~\cite[proof of Theorem~17]{Dab14}, explained in greater detail
in~\cite[Lemma~3.6]{DiezObata}).
Thus, writing
\[
 \widetilde{\mathcal E}_2(u)
 :=\|(\widetilde\Delta^\otimes\otimes I_n)^{1/2}\bar\delta u\|_2^2,
\]
we have
\begin{equation}\label{eq:embedding-coercivity}
 \widetilde{\mathcal E}_2(u)+c\|\bar\delta u\|_2^2
 \leq\|\widetilde\Delta u\|_2^2,
 \qquad u\in\mathrm{Dom}(\widetilde\Delta).
\end{equation}
We fix $\epsilon\in(0,1)$ and $T>0$.
We apply~\cite[Proposition~5, equation~(6)]{Dab14} with
$B=\widetilde\Delta^{1/2}$ and $\alpha'=\alpha$, as in ~\cite[Step~1
of the proof of Theorem~11 and
Lemma~3(ii)--(iii)]{Dab14}. Together with the free It\^o isometry of Biane--Speicher~\cite{BS}, this gives the following identity
(the real-part of the inner product in~\cite[Proposition~5, equation~(6)]{Dab14}
can be omitted here, since the corresponding inner product equals a squared norm by spectral calculus):
\[
 \|\widetilde\Delta^{1/2}\widetilde\rho_\alpha
                         \gamma(Y^\epsilon)_T\|_2^2
   +\int_0^T\|\widetilde\Delta\widetilde\rho_\alpha
                         \gamma(Y^\epsilon)_s\|_2^2\,ds=
   \int_0^T
   \|(\widetilde\Delta^\otimes\otimes I_n)^{1/2}
       \widetilde\rho_\alpha^\otimes\bar\delta Y_s^\epsilon\|_2^2\,ds.
\]
By~\eqref{eq:embedding-coercivity}, the right-hand side is bounded by
\[
 \int_0^T\widetilde{\mathcal E}_2(Y_s^\epsilon)\,ds
 \leq\int_0^T\|\widetilde\Delta Y_s^\epsilon\|_2^2\,ds<\infty.
\]
We can let $\alpha\to\infty$ by using monotone convergence.
The (deterministic) semigroup satisfies
\[
 \|\widetilde\Delta^{1/2}\widetilde\varphi_{T/2}(x)\|_2^2
 +\int_0^T\|\widetilde\Delta\widetilde\varphi_{s/2}(x)\|_2^2\,ds
 =\|\bar\partial x\|_2^2.
\]
Moreover, $\gamma(Y^\epsilon)_t\perp L^2(M)$, since its conditional
expectation onto $M$ is zero, while the deterministic
term belongs to $L^2(M)$. Since this subspace reduces
$\widetilde\Delta$, these terms remain orthogonal after applying the
spectral powers above. Adding the two energy identities and
using~\eqref{eq:embedding-epsilon}, we obtain
\begin{equation}\label{eq:embedding-regularized-energy}
 \begin{split}
 \|\bar\delta Y_T^\epsilon\|_2^2
 +\int_0^T\|\widetilde\Delta Y_s^\epsilon\|_2^2\,ds
 =\|\bar\partial x\|_2^2
 +(1-\epsilon)^2\int_0^T
                 \widetilde{\mathcal E}_2(Y_s^\epsilon)\,ds.
 \end{split}
\end{equation}
Since $1-(1-\epsilon)^2\geq0$, the coercivity
estimate~\eqref{eq:embedding-coercivity} yields
\[
 \|\bar\delta Y_T^\epsilon\|_2^2
 +c\int_0^T\|\bar\delta Y_s^\epsilon\|_2^2\,ds
 \leq\|\bar\partial x\|_2^2.
\]
Omitting the first term and using weak lower semicontinuity in
$\mathcal B_{2,\bar\delta}^{a}([0,T])$, we can reach the limit
$\epsilon\to0$. Letting then $T\to\infty$, we obtain
\begin{equation}\label{eq:embedding-integrated-gradient}
 \int_0^\infty\|\bar\delta\Phi_s(x)\|_2^2\,ds
 \leq\frac1c\|\bar\partial x\|_2^2.
\end{equation}
We next identify the strong limit of $\widetilde\varphi_r$. Since $\delta$ vanishes on the polynomials in the free Brownian
variables, this is clear by density that $L^2(N)\subset \mathrm{Ker}(\widetilde\Delta)$. We now prove the reverse inclusion using the word-space
decomposition introduced before assumption $\Gamma_{0u}(\omega)$
in~\cite[p.~1409]{Dab14} and the tensor representation of the zero
extension in~\cite[Section~2.1.3]{Dab14}.
We choose an orthonormal basis $(e_j)_{j\geq1}$ of
$L^2(N)\ominus\mathbb C.1$ obtained by applying Gram--Schmidt to a countable $L^2$-dense family of trace-zero non-commutative polynomials in the free Brownian variables at non-negative rational times.
\newline
For $k\geq1$, set
\[
 \mathcal V_k
 =L^2(M)\bar\otimes
   (L^2(M)\ominus\mathbb C.1)^{\bar\otimes(k-1)}
   \bar\otimes L^2(M),
\]
with the middle factor omitted when $k=1$.
On this space, $\Delta^{\otimes(k+1)}$ denotes the positive
self-adjoint closure of
\[
 \Delta^{\otimes(k+1)}(a_0\otimes\cdots\otimes a_k)
 =\sum_{j=0}^k
   a_0\otimes\cdots\otimes\Delta(a_j)\otimes\cdots\otimes a_k
\]
on $\mathrm{Dom}(\Delta)^{\otimes (k+1)}\cap \mathcal V_k$ (the algebraic tensor product).
For each $i=(i_1,\ldots,i_k)\in\mathbb N^k$, write
$e_i=e_{i_1}\otimes\cdots\otimes e_{i_k}$.
We also use the multiplication map
\(
 (a_0\otimes\cdots\otimes a_k)\#e_i
 =a_0e_{i_1}a_1\cdots e_{i_k}a_k,
\)
initially defined for $a_0,\ldots,a_k\in M$ with
$\tau(a_j)=0$ for $1\leq j\leq k-1$.
By freeness, this map extends to an isometry on $\mathcal V_k$, we also denote its (closed) range by setting
\(
 \mathcal V_k\#e_i
 :=\{U\#e_i:U\in\mathcal V_k\}
\). By the standard Hilbert space decomposition of a tracial free product (see~\cite[Section~5.3.2 and Remark~5.3.5]{AnanPopa})
\[
 L^2(\widetilde M)
 =L^2(M)\oplus
   \bigoplus_{k\geq1}\ \bigoplus_{i\in\mathbb N^k}
   (\mathcal V_k\#e_i).
\]
The last assertion of~\cite[Corollary~15]{Dab14} verifies his
condition $\Gamma_{0u}(\omega)\:e.2)$ for these tensor extensions of $\Delta$: for
$U\in\mathrm{Dom}(\Delta^{\otimes(k+1)})$, one has
$U\#e_i\in\mathrm{Dom}(\widetilde\Delta)$ and
\[
 \widetilde\Delta(U\#e_i)
 =\Delta^{\otimes(k+1)}(U)\#e_i.
\]
Applying this to $U=\bigl(\alpha+\Delta^{\otimes(k+1)}\bigr)^{-1}(W)$ with $\alpha>0$ gives
\[
 (\alpha+\widetilde\Delta)^{-1}(W\#e_i)
 =\bigl((\alpha+\Delta^{\otimes(k+1)}\bigr)^{-1}(W))\#e_i,
 \qquad W\in\mathcal V_k.
\]
Each closed word space is therefore invariant under the self-adjoint
resolvent, and hence reduces $\widetilde\Delta$, whose restriction
there is unitarily equivalent to $\Delta^{\otimes(k+1)}$.
Since $\ker\Delta=\mathbb C.1$, the kernel of $\Delta^{\otimes 2}$ on
$\mathcal V_1$ is $\mathbb C.(1\otimes1)$, whose inserted copies are
the $e_j$'s. For $k\geq2$, a centered internal factor makes the
kernel on $\mathcal V_k$ zero. Consequently, we deduce that
\(
 \ker\widetilde\Delta=L^2(N)
\)
and denoting by $E_N$ the trace-preserving conditional expectation onto $N$, also viewed as the orthogonal projection onto $L^2(N)$,
the spectral theorem gives
\begin{equation}\label{eq:embedding-heat-limit}
 \widetilde\varphi_r\longrightarrow E_N
 \quad\text{strongly on }L^2(\widetilde M)
 \quad\text{as }r\to\infty.
\end{equation}
We now use~\eqref{eq:embedding-integrated-gradient} to pass to the
limit in the mild equation. Fix $T\geq0$. We split the stochastic
integral in~\eqref{eq:embedding-mild} at $T$ and, using the semigroup
identity of~\cite[Lemma~3(ii)]{Dab14}, we obtain, for $t\geq T$,
\[
 Y_t=\widetilde\varphi_{(t-T)/2}(Y_T)
      +\sum_{i=1}^n\int_T^t
        \widetilde\varphi_{(t-s)/2}^\otimes
                       (\bar\delta_iY_s)\#dS_s^{(i)}.
\]
The free It\^o isometry and $L^2$-contractivity of the semigroup associated to $\widetilde\Delta^{\otimes}$ give
\begin{align*}
 \|Y_t-\widetilde\varphi_{(t-T)/2}(Y_T)\|_2^2
 &=\int_T^t
   \|\widetilde\varphi_{(t-s)/2}^\otimes\bar\delta Y_s\|_2^2\,ds\\
 &\leq\int_T^\infty\|\bar\delta Y_s\|_2^2\,ds.
\end{align*}
For fixed $T$, the deterministic term converges to $E_NY_T$
by~\eqref{eq:embedding-heat-limit}. Hence
\begin{equation}\label{eq:embedding-tail}
 \limsup_{t\to\infty}\|Y_t-E_NY_T\|_2
 \leq
 \left(\int_T^\infty\|\bar\delta Y_s\|_2^2\,ds\right)^{1/2}.
\end{equation}
The triangle inequality gives
\(
 \underset{t,t'\to\infty}{\limsup}\|Y_t-Y_{t'}\|_2
 \leq2\left(\int_T^\infty\|\bar\delta Y_s\|_2^2\,ds\right)^{1/2}.
\)
The right-hand side tends to zero as $T\to\infty$, so $Y_t$ is
Cauchy in $L^2(\widetilde M)$. Its limit then belongs to $L^2(N)$,
since~\eqref{eq:embedding-tail} approximates it by $E_NY_T\in L^2(N)$
with an error tending to zero. We thus have proved that
\[
 \Phi_t(x)\longrightarrow\Phi_\infty(x)\in L^2(N),
 \qquad x\in M_0.
\]
The $L^2$ isometry of the maps $\Phi_t$ and density of $M_0$
extend this convergence to every $x\in M$.
Moreover, we see that $E_N\Phi_t(x)\in N$ converges to $\Phi_\infty(x)$ in $L^2$
and has operator norm at most $\|x\|$. Since the 
operator-norm balls in $N$ are closed in the $L^2$ topology, we get
$\Phi_\infty(x)\in N$ and $\|\Phi_\infty(x)\|\leq\|x\|$.
For $x,y\in M$,
\[
 \|\Phi_t(x)\Phi_t(y)-\Phi_\infty(x)\Phi_\infty(y)\|_2\leq
 \|x\|\,\|\Phi_t(y)-\Phi_\infty(y)\|_2
 +\|y\|\,\|\Phi_t(x)-\Phi_\infty(x)\|_2\longrightarrow0.
\]
Since $\Phi_t(xy)=\Phi_t(x)\Phi_t(y)$ and
$\Phi_t(xy)\to\Phi_\infty(xy)$ in $L^2$, uniqueness of the limit gives
\[
 \Phi_\infty(xy)=\Phi_\infty(x)\Phi_\infty(y).
\]
Therefore, we see that multiplicativity is preserved at the limit.
The same is true for linearity, the involution, the unit and the trace.
Injectivity is ensured by preservation of the faithful trace.
The map is also normal: indeed, if $0\leq x_\alpha\uparrow x$, then
the positive element
$\Phi_\infty(x)-\underset{\alpha}{\sup}\:\Phi_\infty(x_\alpha)$
has trace zero by trace preservation and normality of the traces,
hence vanishes by faithfulness.
Finally, the subgroup generated freely by
$\{b^kab^{-k}:k\in\mathbb Z\}\subset\mathbb F_2=\langle a,b\rangle$
gives a trace-preserving inclusion
$L(\mathbb F_\infty)\subset L(\mathbb F_2)$.
Composing the two embeddings gives the conclusion.

<!-- definition-9: body_md -->
Let $P$ be a finite von Neumann algebra and let
$\sigma:\Gamma\curvearrowright P$ be a $\tau$-preserving action. The action is called \emph{weakly compact} if there exists a net of unit vectors
\(
\eta_n\in L^2(P\bar\otimes\bar P)_+
\)
such that
\begin{enumerate}
\item $\lVert\eta_n-(v\otimes\bar{v})\eta_n\rVert_2\to0$ for every $v\in\mathcal U(P)$.
\item $\lVert\eta_n-(\sigma_g\otimes\bar{\sigma}_g)(\eta_n)\rVert_2\to0$ for every $g\in\Gamma$.
\item $\langle (x\otimes1)\eta_n,\eta_n\rangle_2=\tau(x)=\langle \eta_n,(1\otimes\bar{x})\eta_n\rangle_2$
for every $x\in P$ and every $n$.
\end{enumerate}
A von Neumann subalgebra $P\subset M$ is called \emph{weakly compact
in $M$} if the conjugation action $\mathcal N_M(P)\curvearrowright P$
is weakly compact. These conditions force $P$ to be amenable.

<!-- theorem-3: body_md -->
\label{solid}
Let $X=(X_1,\ldots,X_n)$ be a finite self-adjoint tuple whose free difference quotients are closable, and assume that the free Laplacian  
\(
\Delta=\partial^*\bar\partial
\)
is \emph{proper}, that is, has compact resolvent. Let $P\subset M=W^*(X_1,\ldots,X_n)$ be a diffuse amenable von Neumann subalgebra, and let $\mathcal G\subset\mathcal N_M(P)$ be a subgroup whose conjugation action on $P$ is weakly compact. Then $\mathcal G''$ is amenable. In particular, $M$ is solid.
<!-- end theorem-3 -->
<!-- proof-9: body_md -->
The algebraic free difference quotient
\(
\partial:M_0\to(L^2(M)\bar\otimes L^2(M^{op}))^{\oplus n}
\)
is a real closable derivation, and its closure is $\bar\partial$. Indeed, we have
\(
(\partial_i x)^\dagger=\partial_i(x^*)
\)
for every $x\in M_0$, and by Remark~\ref{real} this identity passes to the closure. The target is a finite direct sum of coarse $M$--$M$ correspondences (hence it is weakly contained in the coarse correspondence). Moreover, the positive operator associated with our derivation is exactly $\Delta=\partial^*\bar\partial$. The first assertion follows directly from Ozawa--Popa~\cite[Remark~5.4]{OP2}, since $\Delta$ has compact resolvent.
\newline
We now recall how solidity is obtained. Take a diffuse von Neumann subalgebra $Q\subset M$, choose a diffuse abelian von Neumann subalgebra $A\subset Q$, and set
\(
\mathcal G:=\mathcal U(Q'\cap M).
\)
Every element of $\mathcal G$ commutes with $A$, and thus the conjugation action $\mathcal G\curvearrowright A$ is trivial. The algebra $A$ is amenable, while the trivial action is compact and hence weakly compact by Ozawa--Popa~\cite[Proposition~3.2]{OP1}. Applying the first part of the
theorem with \(P=A\) shows that $\mathcal G''$ is amenable. But
\[
\mathcal G''=\mathcal U(Q'\cap M)''=Q'\cap M.
\]
Thus $Q'\cap M$ is amenable, which is exactly solidity of $M$.
<!-- end proof-9 -->
<!-- definition-10: body_md -->
A finite von Neumann algebra $M$ has the \emph{(weak$^*$) completely bounded approximation property}, denoted \emph{W$^*$CBAP}, if there exist a constant $C\geq1$ and a net $(\phi_n)_n$ of normal finite-rank maps $\phi_i:M\to M$ such that
\[
\limsup_m\left\lVert \phi_m\right\rVert_{cb}\leq C,
\qquad
\left\lVert x-\phi_m(x)\right\rVert_2\to0
\]
for every $x\in M$. The \emph{Cowling--Haagerup constant} $\Lambda_{cb}(M)$ is defined as the infimum of the constants $C$ for which such a net exists (the infimum is always achieved; see~\cite{haagcowl} for details). We say that $M$ has the \emph{(weak$^*$) complete metric approximation property}, usually abbreviated CMAP, if
\(
\Lambda_{cb}(M)=1.
\)
By Connes' theorem~\cite{Connes}, amenability implies the \emph{CMAP}.

=

+

By Theorem~\ref{main}, the operator \(\Delta\) has compact resolvent and
\(
\ker(\Delta)=\mathbb C.1.
\)
Consequently, its minimal non-zero spectral value is an eigenvalue. We denote it by
\(
\lambda
:=
\min(\sigma(\Delta)\setminus\{0\}).
\)
The free Poincaré constant is therefore given by
\(
C_P(X)=\lambda^{-1}.
\)
\newline
We choose a normalized centered eigenfunction \(f\in L^2(M)\):  
\[
\Delta f=\lambda f,
\qquad
\tau(f)=0,
\qquad
\lVert f\rVert_2=1.
\]
Note that $\tau(f)=0$ is immediate as $f$ belongs to the orthogonal complement of $\mathrm{Ker}(\Delta)=\mathbb C.1$. Moreover, since \(f\) belongs to an eigenspace of \(\Delta\), it belongs to
\(\operatorname{Dom}(\Delta^r)\) for every \(r>0\). In particular,
\(
f\in\operatorname{Dom}(\Delta^{3/2}).
\)
Dabrowski's almost-commutation formula~\ref{dab} therefore applies to \(f\), and
\(
\bar\partial_i f\in\operatorname{Dom}(\Delta^\otimes),
\)  for all \(i=1,\ldots,n.
\)
\newline
Let \(P_0\) denote the orthogonal projection of \(L^2(M)\) onto
\(\mathbb C.1\), and use the same notation for the corresponding
projection of \(L^2(M^{op})\) onto \(\mathbb C.1\). By a slight abuse of notation, the identity operators on \(L^2(M)\), on \(L^2(M^{op})\),
and on
\(
L^2(M)\bar\otimes L^2(M^{op})
\)
are all denoted by the same symbol \(\mathrm{id}\). By the definition of \(\lambda\)
and the spectral theorem, we have
\[
\Delta\geq\lambda(\mathrm{id}-P_0)
\]
in the sense of quadratic forms. Recalling that
\[
\Delta^\otimes
=
\overline{
\Delta\otimes\mathrm{id}
+
\mathrm{id}\otimes\Delta
},
\]
it follows that
\begin{align}
\Delta^\otimes
&\geq
\lambda(
(\mathrm{id}-P_0)\otimes\mathrm{id}
+
\mathrm{id}\otimes(\mathrm{id}-P_0)
)\nonumber\\
&\geq
\lambda(
\mathrm{id}-(P_0\otimes P_0)
).
\end{align}
Indeed, one can check on the core $\mathrm{Dom}(\Delta)\otimes \mathrm{Dom}(\Delta)$ (here again $\otimes$ denotes the algebraic tensor product) that 
\begin{align}
&
(
(\mathrm{id}-P_0)\otimes\mathrm{id}
+
\mathrm{id}\otimes(\mathrm{id}-P_0)
)
-
(
\mathrm{id}-(P_0\otimes P_0)
)
=
(\mathrm{id}-P_0)\otimes(\mathrm{id}-P_0)
\geq0.\nonumber
\end{align}
Since \(P_0\otimes P_0\), initially defined on simple tensors, extends to the orthogonal projection onto
\(\mathbb C.(1\otimes1)\), we conclude that, for every
\(\eta\in\operatorname{Dom}(\Delta^\otimes)\),
\begin{equation}\label{tensorP}
\langle
\Delta^\otimes\eta,\eta
\rangle_2
\geq
\lambda
(
\lVert\eta\rVert_2^2
-
\lvert
\langle
\eta,1\otimes1
\rangle_2
\rvert^2
).
\end{equation}
The same conclusion can also be seen directly by diagonalizing
\(\Delta\). Indeed, if
\(
0=\lambda_0<\lambda_1=\lambda\leq\lambda_2\leq\cdots
\)
are its eigenvalues, counted with multiplicity, then the eigenvalues
of the tensor extension \(\Delta^\otimes\) (which thus has discrete spectrum) are the sums
\(\lambda_i+\lambda_j\). The only zero sum is for
\(\lambda_0+\lambda_0\), and every other sum is at least \(\lambda\).
\bigbreak
For \(i=1,\ldots,n\), set
\(
m_i
:=
\langle
\bar\partial_i f,1\otimes1
\rangle_2
=:
(\tau\otimes\tau)(\bar\partial_i f)
\).
Applying the preceding estimate to each \(\bar\partial_i f\) and
summing over \(i\), we obtain
\begin{align}
\langle
(\Delta^\otimes\otimes I_n)\bar\partial f,
\bar\partial f
\rangle_2
&\geq
\lambda
\bigg(
\lVert\bar\partial f\rVert_2^2
-
\sum_{i=1}^n\lvert m_i\rvert^2
\bigg).
\end{align}
Since
\(
\lVert\bar\partial f\rVert_2^2
=
\langle\Delta f,f\rangle_2
=
\lambda
\),
this becomes
\[
\langle
(\Delta^\otimes\otimes I_n)\bar\partial f,
\bar\partial f
\rangle_2
\geq
\lambda^2
-
\lambda\sum_{i=1}^n\lvert m_i\rvert^2.
\]
We now apply Dabrowski's almost-commutation formula~\ref{dab}:
\[
\bar\partial\Delta f
=
(
\Delta^\otimes\otimes I_n
+
\mathcal R_{\mathscr J\xi}
)\bar\partial f.
\]
Taking the inner product with \(\bar\partial f\), and using
\(\Delta f=\lambda f\), we get
\[
\lambda^2
=
\langle
(\Delta^\otimes\otimes I_n)\bar\partial f,
\bar\partial f
\rangle_2
+
\langle
\mathcal R_{\mathscr J\xi}\bar\partial f,
\bar\partial f
\rangle_2.
\]
Combining the last two estimates gives
\[
\langle
\mathcal R_{\mathscr J\xi}\bar\partial f,
\bar\partial f
\rangle_2
\leq
\lambda\sum_{i=1}^n\lvert m_i\rvert^2.
\]
Equivalently,
\[
\sum_{i=1}^n\lvert m_i\rvert^2
\geq
\frac{1}{\lambda}
\langle
\mathcal R_{\mathscr J\xi}\bar\partial f,
\bar\partial f
\rangle_2.
\]
This is the free analogue of the ``\emph{preferred-direction}'' estimate in Klartag's terminology (see the discussion after~\cite[Proposition~2.3]{Klartaglog}).
\newline
For \(i=1,\ldots,n\), we denote
\(
X_i^\circ:=X_i-\tau(X_i).1.
\)
Since the free difference quotients are taken with respect to the
variables \(X_1,\ldots,X_n\), we have
\[
\bar\partial_jX_i^\circ
=
\delta_{ij}(1\otimes1).
\]
Integration by parts therefore gives
\begin{align}
m_i
&=
\langle
\bar\partial f,\bar\partial X_i^\circ
\rangle_2\nonumber\\
&=
\langle
\Delta f,X_i^\circ
\rangle_2\nonumber\\
&=
\lambda
\langle
f,X_i^\circ
\rangle_2.
\end{align}
Hence
\[
\sum_{i=1}^n\lvert m_i\rvert^2
=
\lambda^2
\sum_{i=1}^n
\lvert
\langle
f,X_i^\circ
\rangle_2
\rvert^2.
\]
Consider the operator
\[
T:\mathbb C^n\longrightarrow L^2(M),
\qquad
T(a)
:=
\sum_{i=1}^n a_iX_i^\circ,
\qquad
a=(a_1,\ldots,a_n).
\]
We can check for every \(a,b\in\mathbb C^n\), using the self-adjointness of the \(X_i^\circ\), that we have
\begin{align}
\langle T(a),T(b)\rangle_2
&=
\left\langle
\sum_{i=1}^n a_iX_i^\circ,
\sum_{j=1}^n b_jX_j^\circ
\right\rangle_2\nonumber\\
&=
\sum_{i,j=1}^n
a_i\overline{b_j}\,
\tau(X_j^\circ X_i^\circ)\nonumber\\
&=
\left\langle
\operatorname{Cov}(X)a,b
\right\rangle_{\mathbb C^n}.
\end{align}
Hence, the Gram matrix of \(T\) is \(\operatorname{Cov}(X)\). Therefore, we have
\(
T^*T=\operatorname{Cov}(X),
\) so that \(
\lVert T\rVert^2
=
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}
\).
Since \(\lVert f\rVert_2=1\), it follows that
\begin{align}\label{eq21}
\sum_{i=1}^n
\left\lvert
\left\langle
f,X_i^\circ
\right\rangle_2
\right\rvert^2
&=
\lVert T^*f\rVert_{\mathbb C^n}^2\nonumber\\
&\leq
\lVert T\rVert^2\lVert f\rVert_2^2\nonumber\\
&=
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}.
\end{align}
We then get
\(
\sum_{i=1}^n\lvert m_i\rvert^2
\leq
\lambda^2
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}.
\)
Combining this with the preferred-direction estimate gives
\[
\langle
\mathcal R_{\mathscr J\xi}\bar\partial f,
\bar\partial f
\rangle_2
\leq
\lambda^3
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}.
\]
Finally, the \(CD(c,\infty)\) condition implies
\(
\mathcal R_{\mathscr J\xi}
\geq
c\,\mathrm{id}.
\)
Consequently,
\begin{align}
c\lambda
&=
c\lVert\bar\partial f\rVert_2^2\nonumber\\
&\leq
\langle
\mathcal R_{\mathscr J\xi}\bar\partial f,
\bar\partial f
\rangle_2\nonumber\\
&\leq
\lambda^3
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}.
\end{align}
Since \(\lambda>0\), this yields
\[
\lambda^2
\geq
\frac{c}{\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}},
\]
and therefore we deduce
\[
C_P(X)
=
\frac{1}{\lambda}
\leq
\sqrt{
\frac{\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}}{c}
}.
\]
It remains to establish the lower bound. It is easy to see that testing the free Poincar\'e inequality on linear functions $
T(a):=a_1X_1^{\circ}+\ldots+a_nX_n^{\circ}$ and taking the supremum over all unit vectors \(a\in\mathbb C^n\) gives
\[
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}
\leq
C_P(X).
\]
Finally, the free Lichnerowicz inequality under \(CD(c,\infty)\)
gives
\[
C_P(X)\leq\frac{1}{c}.
\]
Together with
\(
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}\leq C_P(X),
\)
this implies
\[
\sqrt{
\frac{\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}}{c}
}
\leq
\frac{1}{c},
\]
and completes the proof.

Since \(\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}>0\), we may center the
tuple and apply a dilation to normalize its maximal variance to one.
Thus, without loss of generality, we assume that
\(
\tau(X_i)=0,
\) for all \(
i=1,\ldots,n,
\) and \(
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}=1.
\)
\newline
In the sequel, we keep the same notation for the tuple and all the
associated objects. Under this dilation, the free Laplacian and the
curvature constant are multiplied by the original value of
\(\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}\), while the Poincaré
constant is divided by the same quantity. We continue to denote the
(new) curvature constant by \(c\). Thus the normalized tuple
satisfies \(CD(c,\infty)\), and the equality assumption becomes
\[
C_P(X)=\frac{1}{\sqrt c}.
\]
Testing the free Poincaré inequality on a linear combination of the
variables in a direction of maximal variance gives
\(1\leq C_P(X)=c^{-1/2}\), and hence \(c\leq1\). The goal is to prove
that \(c=1\). This will show that equality in the improved free
Lichnerowicz inequality implies equality in the usual free Lichnerowicz
inequality, and the conclusion will then follow from the free Obata
rigidity theorem~\cite[Theorem~3.15]{DiezObata}.
\newline
By Theorem~\ref{main}, the spectral gap is attained. 


Since
\(\Delta\) commutes with the Tomita conjugation
(see the proof of~\cite[Corollary~3.4]{DiezObata}), we can choose a
self-adjoint eigenfunction \(f\in L^2(M)\) satisfying
\(
\Delta f=\sqrt c\,f,
\) $\tau(f)=0$,
and \(
\lVert f\rVert_2=1
\).
\newline
As
\(
f\in\ker(\Delta-\sqrt c\,\mathrm{id}),
\)
it belongs to \(\operatorname{Dom}(\Delta^r)\) for every \(r>0\). We
also have
\(
\lVert\bar\partial f\rVert_2^2
=
\langle\Delta f,f\rangle_2
=
\sqrt c,
\) and \(
\lVert\Delta f\rVert_2^2=c.
\)
For every \(i=1,\ldots,n\), integration by parts gives, using the self-adjointness of
\(\Delta f\) and \(X_i\),
\[
\left\langle
\bar\partial_i f,1\otimes1
\right\rangle_2
=
\left\langle
\bar\partial f,\bar\partial X_i
\right\rangle_2
=
\left\langle
\Delta f,X_i
\right\rangle_2
\in\mathbb R.
\]
If
\(
\left(
\langle
\bar\partial_i f,1\otimes1
\rangle_2
\right)_{i=1}^n
=0,
\)
then applying~\eqref{bochner} to $f$ would yield
\(c\geq c+c^{3/2}\), which is impossible. We may therefore define
\(
\theta
:=
\frac{
(
\langle
\bar\partial_i f,1\otimes1
\rangle_2
)_{i=1}^n
}{
\left(
\sum_{i=1}^n
\lvert
\langle
\bar\partial_i f,1\otimes1
\rangle_2
\rvert^2
\right)^{1/2}
}.
\)
Then \(\theta\in\mathbb R^n\) and \(\lvert\theta\rvert=1\). Moreover,
\begin{align}
\sum_{i=1}^n
\left\lvert
\left\langle
\bar\partial_i f,1\otimes1
\right\rangle_2
\right\rvert^2
&=
\left\lvert
\sum_{i=1}^n
\theta_i
\left\langle
\bar\partial_i f,1\otimes1
\right\rangle_2
\right\rvert^2\nonumber\\
&=
\left\lvert
\left\langle
\bar\partial f,
\bar\partial
\left(
\sum_{i=1}^n\theta_iX_i
\right)
\right\rangle_2
\right\rvert^2\nonumber\\
&=
\left\lvert
\left\langle
\Delta f,
\sum_{i=1}^n\theta_iX_i
\right\rangle_2
\right\rvert^2 \nonumber\\
&\leq
\lVert\Delta f\rVert_2^2
\left\lVert
\sum_{i=1}^n\theta_iX_i
\right\rVert_2^2
\label{cauchy}\\
&\leq
\lVert\Delta f\rVert_2^2
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}
\lvert\theta\rvert^2
=
\lVert\Delta f\rVert_2^2
=
c.
\label{direction}
\end{align}
Combining this estimate with~\eqref{bochner}, applied to
\(f\), gives
\begin{align}
c
&\geq
c
-\sqrt c \sum_{i=1}^n \lvert \langle \bar\partial_i f,1\otimes1\rangle_2
\rvert^2
+
c^{3/2}\nonumber\\
&\geq
c-c^{3/2}+c^{3/2}
=
c.
\end{align}
Hence, equality holds at every step, and in particular
in~\eqref{cauchy}
and~\eqref{direction}.
\newline
Equality in~\eqref{direction} means that
\(\theta\) is a direction of unit variance, and hence a direction of
maximal variance under the normalization under consideration.
Equivalently,
\[
\operatorname{Cov}(X)\theta=\theta.
\]
Equality in~\eqref{cauchy} implies that
\(
\Delta f
=
\lambda
\sum_{i=1}^n\theta_iX_i
\)
for some \(\lambda\in\mathbb R\). Taking the \(L^2\)-norms of both
sides then gives \(\lvert\lambda\rvert=\sqrt c\). Since
\(\Delta f=\sqrt c\,f\), we deduce that
\(
f
=
\pm
\sum_{i=1}^n\theta_iX_i.
\)
Therefore,
\(
\bar\partial_jf
=
\pm\theta_j(1\otimes1),
\)
for all \(j=1,\ldots,n\), and hence
\[
\sqrt c
=
\lVert\bar\partial f\rVert_2^2
=
\sum_{j=1}^n\theta_j^2
=
1.
\]
Thus \(c=1\), and consequently \(C_P(X)=1\) in the normalized setting.
\newline
The normalized tuple therefore saturates the usual free Lichnerowicz
inequality. The conclusion follows from its equality case, that is, the
free Obata rigidity theorem~\cite[Theorem~3.15]{DiezObata}. Replacing
\(\theta\) by \(-\theta\), if necessary, and removing the normalization,
we obtain
\[
\lVert\operatorname{Cov}(X)\rVert_{\mathrm{op}}
=
C_P(X)
=
\frac1c,
\]
where \(c\) again denotes the curvature constant of the original tuple,
and \(\theta\) is a direction of maximal variance. After an orthogonal
change of variables whose first direction is \(\theta\), the first
centered variable is semicircular with variance \(1/c\), it is free
from the remaining variables, and, when $n\geq2$, the remaining tuple satisfies
\(CD(c,\infty)\) by the conditional-expectation argument in the proof of Corollary~\ref{obata}.

<!-- proof-19: body_md -->
Voiculescu's free Poincar\'e inequality~\ref{voipub} gives a positive
spectral gap on
\(
L^2_0(M):=L^2(M)\ominus\mathbb C.1.
\)
Writing $\lambda:=\inf\sigma(\Delta|_{L^2_0(M)})$, the spectral theorem gives
\begin{equation}\label{eq:kls-inverse}
\lambda=C_P(X)^{-1}>0,
\qquad
0\leq\Delta^{-1}\leq\frac1\lambda\mathrm{id},
\qquad
\|\Delta^{-1}\|_{\mathbb B(L^2_0(M))}=\frac1\lambda.
\end{equation}
Here and below, $\Delta^{-1}$ acts on $L^2_0(M)$.
We do not assume that $\lambda$ is an eigenvalue.
Choose a centered self-adjoint $f\in\mathrm{Dom}(\Delta^{3/2})$.
Recall that the free Poincar\'e inequality is equivalently written as
$\|\bar\partial f\|_2^2\leq C_P(X)\|\Delta f\|_2^2$.
Our aim is to prove the following estimate:
\begin{equation}\label{eq:kls-self-improvement}
\|\bar\partial f\|_2^2
\leq
\left(\frac{7}{8\lambda}
+\frac54\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)
\|\Delta f\|_2^2.
\end{equation}
Since $C_P(X)=1/\lambda$, the coefficient $7/8<1$ will allow us to
absorb the term involving $C_P(X)$ and obtain the upper bound in the theorem.
We divide the proof into five steps.

\medskip
\noindent\emph{Step 1:}
Let $P_0$ be the orthogonal projection of $L^2(M)$ onto $\mathbb C.1$.
The spectral theorem gives
\(
\Delta\geq\lambda(\mathrm{id}-P_0),
\) so that
\[
\Delta\otimes\mathrm{id}
\geq\lambda((\mathrm{id}-P_0)\otimes\mathrm{id})
\]
as quadratic forms. If $\eta\in\mathrm{Dom}(\Delta^\otimes)$, the joint
spectral calculus gives $\eta\in\mathrm{Dom}(\Delta\otimes\mathrm{id})$. Hence,
using the tensor Dirichlet form identity
(see, for instance,~\cite[Proposition~3.5]{DiezObata}),
\begin{align}
\lambda\|((\mathrm{id}-P_0)\otimes\mathrm{id})\eta\|_2^2
&\leq\langle(\Delta\otimes\mathrm{id})\eta,\eta\rangle_2\nonumber\\
&=\sum_{j=1}^n
\left\|\overline{\partial_j\otimes\mathrm{id}}\,\eta\right\|_2^2.
\label{eq:kls-first-leg-energy}
\end{align}
For a simple tensor,
\(
(P_0\otimes\mathrm{id})(x\otimes y)
=\tau(x).1\otimes y
=1\otimes(\tau\otimes\mathrm{id})(x\otimes y).
\)
Both sides extend continuously to the Hilbert tensor product.
Orthogonality therefore gives
\[
\|\eta\|_2^2
=\|(\tau\otimes\mathrm{id})\eta\|_2^2
+\|((\mathrm{id}-P_0)\otimes\mathrm{id})\eta\|_2^2.
\]
Together with~\eqref{eq:kls-first-leg-energy}, this proves
\begin{equation}\label{eq:kls-tensor-Poincare}
\|\eta\|_2^2
\leq\|(\tau\otimes\mathrm{id})\eta\|_2^2
+\frac1\lambda\sum_{j=1}^n
\left\|\overline{\partial_j\otimes\mathrm{id}}\,\eta\right\|_2^2.
\end{equation}
Dabrowski's almost commutation formula in Lemma~\ref{dab}
gives, for
$u\in\mathrm{Dom}(\Delta^{3/2})$,
\begin{equation}\label{eq:kls-commutation}
\bar\partial u\in\mathrm{Dom}(\Delta^\otimes\otimes I_n),
\qquad
\bar\partial\Delta u
=(\Delta^\otimes\otimes I_n
+\mathcal R_{\mathscr J\xi})\bar\partial u.
\end{equation}
Define the tensor Dirichlet energy by
\begin{align*}
\mathcal E_2(u)
&:=\langle(\Delta^\otimes\otimes I_n)\bar\partial u,
                   \bar\partial u\rangle_2\\
&=\sum_{i,j=1}^n
\left\|\overline{\partial_i\otimes\mathrm{id}}\,\bar\partial_j u\right\|_2^2
+\sum_{i,j=1}^n
\left\|\overline{\mathrm{id}\otimes\partial_j}\,\bar\partial_i u\right\|_2^2.
\end{align*}
Taking the inner product of~\eqref{eq:kls-commutation} with
$\bar\partial u$, and using $CD(0,\infty)$, yields
\begin{equation}\label{eq:kls-Bochner}
\mathcal E_2(u)
=\|\Delta u\|_2^2
-\langle\mathcal R_{\mathscr J\xi}\bar\partial u,
                         \bar\partial u\rangle_2
\leq\|\Delta u\|_2^2.
\end{equation}
Co-associativity also holds on this domain:
\begin{equation}\label{eq:kls-coass}
\overline{\partial_i\otimes\mathrm{id}}\,\bar\partial_j u
=\overline{\mathrm{id}\otimes\partial_j}\,\bar\partial_i u,
\qquad 1\leq i,j\leq n.
\end{equation}
Indeed, approximate $u$ by polynomials in the graph norm of
$\bar\partial$ and test the polynomial identity against
$V\in M_0^{\otimes3}$. Voiculescu's formula for the adjoint~\cite[Proposition~4.1]{Voic} and density of $M_0^{\otimes3}$ then
give~\eqref{eq:kls-coass}.
In particular, the two tensor energies are equal:
\begin{equation}\label{eq:kls-half}
\sum_{i,j=1}^n
\left\|\overline{\partial_i\otimes\mathrm{id}}\,\bar\partial_j u\right\|_2^2
=\sum_{i,j=1}^n
\left\|\overline{\mathrm{id}\otimes\partial_j}\,\bar\partial_i u\right\|_2^2
=\frac12\mathcal E_2(u).
\end{equation}

\medskip
\noindent\emph{Step 2: Reduction to the centered partial traces of the free gradients}
For the function $f$ fixed above, set
\[
m_i:=(\tau\otimes\tau)(\bar\partial_i f),
\qquad
a_i:=(\mathrm{id}\otimes\tau)(\bar\partial_i f)-m_i.1,
\qquad
A:=\sum_{i=1}^n\|a_i\|_2^2.
\]
Thus $\tau(a_i)=0$. Moreover, $a_i\in\mathrm{Dom}(\Delta)$.
This follows from Dabrowski's Lemma~\ref{dab}(ii).
One can also see it directly: $\mathrm{id}\otimes P_0$ commutes with
the spectral projections of
$\Delta^\otimes$ and its range is $L^2(M)\bar\otimes\mathbb C.1$.
The restriction of $\Delta^\otimes$ to this range is $\Delta\otimes\mathrm{id}$.
Applying this observation to $\bar\partial_i f\in\mathrm{Dom}(\Delta^\otimes)$
gives the conclusion.
\newline
The real property of free difference quotients gives
\(
\bar\partial_i(f^*)=(\bar\partial_i f)^\dagger.
\)
Since $f=f^*$, we have $m_i\in\mathbb R$. Using Mai's result~\cite[Lemma~4.6]{Mai} again, for any $u\in L^2(M)\bar\otimes L^2(M^{op})$ one has 
\(
(\mathrm{id}\otimes \tau)(u)^\ast = (\tau\otimes \mathrm{id})(u^\dagger)
\), so that
\begin{equation}\label{eq:kls-opposite-trace}
(\tau\otimes\mathrm{id})(\bar\partial_i f)=a_i^*+m_i\,1.
\end{equation}
In particular, by orthogonality, since $\tau(a_i)=0$, we have $a_i^*\perp1$ in $L^2(M)$.
Pythagoras' theorem and traciality therefore give
\[
\|(\tau\otimes\mathrm{id})(\bar\partial_i f)\|_2^2
=\|a_i\|_2^2+|m_i|^2.
\]
Apply~\eqref{eq:kls-tensor-Poincare} to $\bar\partial_i f$ and sum
over $i=1,\ldots,n$. By~\eqref{eq:kls-half},
\begin{equation}\label{eq:kls-reduction}
\|\bar\partial f\|_2^2
\leq\sum_{i=1}^n|m_i|^2+A+\frac1{2\lambda}\mathcal E_2(f).
\end{equation}
Writing $X_i^\circ:=X_i-\tau(X_i).1$, integration by parts gives
$m_i=\langle\Delta f,X_i^\circ\rangle_2$.
Since $\operatorname{Cov}(X)$ is the Gram matrix of the $X_i^\circ$,
Cauchy--Schwarz yields
\begin{equation}\label{eq:kls-covariance-means}
\sum_{i=1}^n|m_i|^2
\leq\|\operatorname{Cov}(X)\|_{\mathrm{op}}\|\Delta f\|_2^2.
\end{equation}
It remains to estimate $A$. Applying Poincar\'e directly to each
$a_i$ would give no improvement in~\eqref{eq:kls-reduction}.
We instead use the inverse Laplacian.

\medskip
\noindent\emph{Step 3: The Poisson equation and exterior Hessian traces.}
To express $\|a_i\|_2^2$ by integration by parts, we solve the Poisson
equation. For each $i=1,\ldots,n$, let
\[
h_i:=\Delta^{-1}a_i\in L^2_0(M),
\qquad \Delta h_i=a_i.
\]
Integration by parts and~\eqref{eq:kls-inverse} give
\begin{equation}\label{eq:kls-Hminus-bound}
\sum_{i=1}^n\|\bar\partial h_i\|_2^2
=\sum_{i=1}^n\langle a_i,\Delta^{-1}a_i\rangle_2
\leq\frac A\lambda.
\end{equation}
Each term $\langle a_i,\Delta^{-1}a_i\rangle_2$ is precisely the squared
$H^{-1}$ norm of $a_i$. Thus~\eqref{eq:kls-Hminus-bound} controls the
energy of the Poisson solutions, while $\Delta h_i=a_i$ gives
$\|a_i\|_2^2=\langle\bar\partial a_i,\bar\partial h_i\rangle_2$.
This is the identity we will use to pair the Hessian of $f$ with the
gradients of the $h_i$.
\newline
Since $a_i\in\mathrm{Dom}(\Delta)$, in particular
$a_i\in\mathrm{Dom}(\Delta^{1/2})$. On $L^2_0(M)$, spectral calculus gives
$\Delta^{3/2}\Delta^{-1}a_i=\Delta^{1/2}a_i$, so
\(
h_i\in\mathrm{Dom}(\Delta^{3/2})
\).
The free Bochner formula and co-associativity therefore apply to each $h_i$.
\newline
Define the Hessian
\begin{equation}\label{eq:kls-Hessian}
H_{ij}:=\overline{\partial_i\otimes\mathrm{id}}\,\bar\partial_j f
=\overline{\mathrm{id}\otimes\partial_j}\,\bar\partial_i f,\qquad 1\leq i,j\leq n.
\end{equation}
By Dabrowski~\cite[Lemma~18]{Dab14}, the ``sliced'' derivatives, i.e. composing the conditional expectation $\mathrm{id}\otimes\tau$ with the free difference quotients:
$(\mathrm{id}\otimes\tau)\circ \partial_j$, extend to bounded operators
on $L^2(M)$. The same holds for $(\tau\otimes\mathrm{id})\circ\partial_i$
by applying the involution.
As in~\cite[Lemma~19(ii)]{Dab14}, co-associativity on polynomials,
approximation in the graph norm of $\bar\partial$, and closedness
of $\bar\partial$ therefore give
\begin{align}
\bar\partial_i\bigl((\mathrm{id}\otimes\tau)\bar\partial_j f\bigr)
&=\bigl(\mathrm{id}\otimes
  [(\mathrm{id}\otimes\tau)\bar\partial_j]\bigr)\bar\partial_i f
\nonumber\\
&=(\mathrm{id}\otimes\mathrm{id}\otimes\tau)H_{ij},
\label{eq:kls-trace-derivative-right}\\
\bar\partial_j\bigl((\tau\otimes\mathrm{id})\bar\partial_i f\bigr)
&=\bigl([(\tau\otimes\mathrm{id})\bar\partial_i]
  \otimes\mathrm{id}\bigr)\bar\partial_j f
\nonumber\\
&=(\tau\otimes\mathrm{id}\otimes\mathrm{id})H_{ij}.
\label{eq:kls-trace-derivative-left}
\end{align}
Here the ``bracketed'' compositions of conditional expectations with
derivatives denote their bounded extensions on $L^2(M)$.
For the last equalities, the corresponding identities hold on tensors of non-commutative polynomials, which form a core for the closed operators $\overline{\mathrm{id}\otimes\partial_j}$ and
$\overline{\partial_i\otimes\mathrm{id}}$
in~\eqref{eq:kls-Hessian}. They therefore extend by approximation of
$\bar\partial_i f$ and $\bar\partial_j f$ in the respective graph norms,
using boundedness of these extensions and contractivity of
the $L^2$-extensions of the conditional expectations
$\mathrm{id}\otimes\mathrm{id}\otimes\tau$ and
$\tau\otimes\mathrm{id}\otimes\mathrm{id}$.
Since constants have zero derivative, the definition of $a_j$
and~\eqref{eq:kls-opposite-trace} yield
\begin{align}
(\mathrm{id}\otimes\mathrm{id}\otimes\tau)H_{ij}
&=\bar\partial_i a_j,
\nonumber\\
(\tau\otimes\mathrm{id}\otimes\mathrm{id})H_{ij}
&=\bar\partial_j(a_i^*).
\label{eq:kls-two-outer-traces}
\end{align}
The adjoints of these conditional expectation maps are respectively given by
$x\mapsto x\otimes1$ and $x\mapsto1\otimes x$.
Thus, by integration by parts, we see immediately that
\begin{align*}
A
&=\sum_{j=1}^n\langle a_j,\Delta h_j\rangle_2\\
&=\sum_{i,j=1}^n\langle\bar\partial_i a_j,
                              \bar\partial_i h_j\rangle_2\\
&=\sum_{i,j=1}^n\langle H_{ij},\bar\partial_i h_j\otimes1\rangle_2.
\end{align*}
Since $\Delta$
commutes with the Tomita involution, we have $\Delta(h_i^*)=a_i^*$. Consequently, using traciality,
\begin{align*}
A
&=\sum_{i=1}^n\langle a_i^*,\Delta(h_i^*)\rangle_2\\
&=\sum_{i,j=1}^n\langle\bar\partial_j(a_i^*),
                              \bar\partial_j(h_i^*)\rangle_2\\
&=\sum_{i,j=1}^n\langle H_{ij},1\otimes\bar\partial_j(h_i^*)\rangle_2.
\end{align*}
Adding the two representations gives
\begin{equation}\label{eq:kls-double-duality}
2A=\sum_{i,j=1}^n
\left\langle H_{ij},
\bar\partial_i h_j\otimes1+1\otimes\bar\partial_j(h_i^*)
\right\rangle_2.
\end{equation}
Each outer trace of the Hessian produces one copy of $A$.
Co-associativity allows us to combine them in the same scalar product.
Since $\sum_{i,j=1}^n\|H_{ij}\|_2^2=\mathcal E_2(f)/2$
by~\eqref{eq:kls-half}, Cauchy--Schwarz yields
\begin{equation}\label{eq:kls-double-CS}
4A^2\leq\frac12\mathcal E_2(f)
\sum_{i,j=1}^n
\left\|\bar\partial_i h_j\otimes1
+1\otimes\bar\partial_j(h_i^*)\right\|_2^2.
\end{equation}

\medskip
\noindent\emph{Step 4: Estimating the mixed term.}
A direct calculation on simple tensors, extended by linearity and
continuity, gives the following identity (recall that our inner products
are linear in the first variable): 
\(
\langle x\otimes1,1\otimes y\rangle_2
=\left\langle(\tau\otimes\mathrm{id})x,
                   (\mathrm{id}\otimes\tau)y\right\rangle_2.
\)

Set
\(
s_{ij}:=(\tau\otimes\mathrm{id})(\bar\partial_i h_j).
\)
Using Mai's result~\cite[Lemma~4.6]{Mai} once more gives
\(
(\mathrm{id}\otimes\tau)\bar\partial_j(h_i^*)=s_{ji}^*.
\)
Hence, Cauchy--Schwarz implies
\[
\left|
\sum_{i,j=1}^n
\left\langle\bar\partial_i h_j\otimes1,
                   1\otimes\bar\partial_j(h_i^*)\right\rangle_2
\right|
=\left|\sum_{i,j=1}^n\langle s_{ij},s_{ji}^*\rangle_2\right|
\leq\sum_{i,j=1}^n\|s_{ij}\|_2^2.
\]
Expanding the squared norms, and using
$\|\bar\partial(h_j^*)\|_2=\|\bar\partial h_j\|_2$ (since $^\dagger$ is an isometric involution), we obtain 
\begin{equation}\label{eq:kls-mixed-bound}
\sum_{i,j=1}^n
\left\|\bar\partial_i h_j\otimes1
+1\otimes\bar\partial_j(h_i^*)\right\|_2^2\leq2\sum_{j=1}^n\|\bar\partial h_j\|_2^2
+2\sum_{i,j=1}^n\|s_{ij}\|_2^2.
\end{equation}
The first sum is bounded by $A/\lambda$ by
\eqref{eq:kls-Hminus-bound}.

For the second sum, the domains in~\eqref{eq:kls-commutation} and
the trace identity~\eqref{eq:kls-trace-derivative-left}, applied to $h_j$, give
$s_{ij}\in\mathrm{Dom}(\bar\partial)$, with
\[
\bar\partial_k s_{ij}
=(\tau\otimes\mathrm{id}\otimes\mathrm{id})
\overline{\mathrm{id}\otimes\partial_k}\,\bar\partial_i h_j.
\]
The partial trace is contractive. Thus, using~\eqref{eq:kls-half},
\eqref{eq:kls-Bochner}, and $\Delta h_j=a_j$, we obtain
\begin{align}
\sum_{i,j,k=1}^n\|\bar\partial_k s_{ij}\|_2^2
&\leq\sum_{i,j,k=1}^n
\left\|\overline{\mathrm{id}\otimes\partial_k}\,\bar\partial_i h_j\right\|_2^2
\nonumber\\
&=\frac12\sum_{j=1}^n\mathcal E_2(h_j)
\leq\frac12\sum_{j=1}^n\|a_j\|_2^2
=\frac A2.
\label{eq:kls-s-derivative}
\end{align}
The $h_j$ need not be self-adjoint: the Bochner estimate
and~\eqref{eq:kls-half} hold for arbitrary complex $u$.
\newline
The means of the $s_{ij}$ are also controlled. Integration by parts gives
\begin{align*}
\tau(s_{ij})
&=(\tau\otimes\tau)(\bar\partial_i h_j)\\
&=\langle\bar\partial h_j,\bar\partial X_i^\circ\rangle_2\\
&=\langle\Delta h_j,X_i^\circ\rangle_2
=\langle a_j,X_i^\circ\rangle_2.
\end{align*}
As in~\eqref{eq:kls-covariance-means}, this yields
\begin{equation}\label{eq:kls-s-means}
\sum_{i,j=1}^n|\tau(s_{ij})|^2
\leq\|\operatorname{Cov}(X)\|_{\mathrm{op}}A.
\end{equation}
We now apply the free Poincar\'e inequality to the centered family $s_{ij}-\tau(s_{ij})\,1$.  Using
\eqref{eq:kls-s-derivative} and~\eqref{eq:kls-s-means} gives
\begin{align}
\sum_{i,j=1}^n\|s_{ij}\|_2^2
&\leq\frac1\lambda
\sum_{i,j,k=1}^n\|\bar\partial_k s_{ij}\|_2^2
+\sum_{i,j=1}^n|\tau(s_{ij})|^2\nonumber\\
&\leq\left(\frac1{2\lambda}+\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)A.
\label{eq:kls-s-bound}
\end{align}
Combining this with~\eqref{eq:kls-Hminus-bound} and
\eqref{eq:kls-mixed-bound} yields
\begin{equation}\label{eq:kls-second-factor}
\sum_{i,j=1}^n
\left\|\bar\partial_i h_j\otimes1
+1\otimes\bar\partial_j(h_i^*)\right\|_2^2
\leq\left(\frac3\lambda+2\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)A.
\end{equation}
Substituting into~\eqref{eq:kls-double-CS}, we obtain
\[
4A^2\leq\frac12\mathcal E_2(f)
\left(\frac3\lambda+2\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)A.
\]
Dividing by $4A$ when $A>0$ gives the following bound, which also
holds trivially when $A=0$:
\begin{equation}\label{eq:kls-A-bound}
A\leq\left(\frac{3}{8\lambda}
+\frac14\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)\mathcal E_2(f).
\end{equation}

\medskip
\noindent\emph{Step 5: Self-improvement of the Poincar\'e constant.}
Combining~\eqref{eq:kls-reduction},
\eqref{eq:kls-covariance-means}, and~\eqref{eq:kls-A-bound} gives
\[
\|\bar\partial f\|_2^2
\leq\|\operatorname{Cov}(X)\|_{\mathrm{op}}\|\Delta f\|_2^2+\left(\frac{3}{8\lambda}
+\frac14\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)\mathcal E_2(f)
+\frac1{2\lambda}\mathcal E_2(f).
\]
By~\eqref{eq:kls-Bochner}, we get
\[
\|\bar\partial f\|_2^2
\leq\left(\frac{7}{8\lambda}
+\frac54\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)\|\Delta f\|_2^2,
\]
which is~\eqref{eq:kls-self-improvement}.

We have proved this for centered self-adjoint
$f\in\mathrm{Dom}(\Delta^{3/2})$, and both sides are unchanged by adding a scalar.
Since $\Delta$ commutes with the Tomita involution $f^*:=Jf$
(see~\cite[Corollary~3.4]{DiezObata}), its spectral calculus does as well.
For $f=\Re f+i\Im f$, with $\Re f=(f+f^*)/2$ and
$\Im f=(f-f^*)/(2i)$, the real property of the free difference quotient, stability under adjunction~\ref{adjunc} and the reality of the corresponding inner products give
$\|\bar\partial f\|_2^2=\|\bar\partial\Re f\|_2^2
+\|\bar\partial\Im f\|_2^2$, and the same decomposition holds for
$\|\Delta f\|_2^2$. Thus~\eqref{eq:kls-self-improvement} holds for every
$f\in\mathrm{Dom}(\Delta^{3/2})$.
For $f\in\mathrm{Dom}(\Delta)$, consider the spectral truncations
\[
f_N:=\mathbf1_{[0,N]}(\Delta)f.
\]
Then $f_N\in\mathrm{Dom}(\Delta^r)$ for every $r>0$, and
$f_N\to f$, $\Delta f_N\to\Delta f$ in $L^2(M)$.
Moreover,
\[
\|\bar\partial(f_N-f)\|_2^2
=\langle\Delta(f_N-f),f_N-f\rangle_2\longrightarrow0.
\]
Thus~\eqref{eq:kls-self-improvement} extends to all
$f\in\mathrm{Dom}(\Delta)$.
\newline
Now take $g\in L^2_0(M)$ and apply this estimate to
$f=\Delta^{-1}g$. Integration by parts gives
\begin{align*}
\langle g,\Delta^{-1}g\rangle_2
&=\|\bar\partial\Delta^{-1}g\|_2^2\\
&\leq\left(\frac{7}{8\lambda}
+\frac54\|\operatorname{Cov}(X)\|_{\mathrm{op}}\right)\|g\|_2^2.
\end{align*}
Taking the supremum over unit vectors $g$ and using the positivity of
$\Delta^{-1}$ yields
\[
\frac1\lambda
=\|\Delta^{-1}\|
\leq\frac{7}{8\lambda}+\frac54\|\operatorname{Cov}(X)\|_{\mathrm{op}}.
\]
Rearranging gives
\(
\frac1{8\lambda}\leq\frac54\|\operatorname{Cov}(X)\|_{\mathrm{op}},
\)
and hence
\[
C_P(X)=\frac1\lambda\leq10\,\|\operatorname{Cov}(X)\|_{\mathrm{op}},
\]
which concludes the proof.

<!-- corollary-13: body_md -->
\label{cor:free-thin-shell}
Let $X=(X_1,\ldots,X_n)$ be a centered isotropic self-adjoint
tuple satisfying $CD(0,\infty)$.
Let $\mathbb{X}:=\sum_{i=1}^n X_i^2$. Then we have
\[
\tau\bigl((\mathbb{X}-n.1)^2\bigr)\leq2Cn,
\qquad
\tau\bigl((\mathbb{X}^{1/2}-\sqrt n.1)^2\bigr)\leq2C,
\]
where $C$ denotes the optimal universal constant in the free
KLS inequality.
<!-- end corollary-13 -->
<!-- proof-20: body_md -->
Since $X$ is centered and isotropic, $\tau(\mathbb{X})=n$ and
\[
\partial_j\mathbb{X}=X_j\otimes1+1\otimes X_j,
\qquad
\|\partial_j\mathbb{X}\|_2^2=2.
\]
The free Poincar\'e inequality and the definition of $C$ give
\[
\tau\bigl((\mathbb{X}-n.1)^2\bigr)
\leq C_P(X)\sum_{j=1}^n\|\partial_j\mathbb{X}\|_2^2
=2nC_P(X)\leq2Cn.
\]
Moreover, functional calculus yields
\[
\begin{aligned}
(\mathbb{X}^{1/2}-\sqrt n.1)^2
&=(\mathbb{X}-n.1)^2
  (\mathbb{X}^{1/2}+\sqrt n.1)^{-2}\\
&\leq\frac1n(\mathbb{X}-n.1)^2.
\end{aligned}
\]
Taking the trace proves the second assertion and concludes the proof.

<!-- proposition-11: body_md -->
\label{milman1}
Let \(\nu_u\) be the free Gibbs measure associated with a smooth
potential \(u:\mathbb R\to\mathbb R\) satisfying
\(
u''\geq c>0
\).
Assume that the optimal transport map \(T\) sending the standard
semicircular distribution \(\eta\) onto \(\nu_u\) is of class
\(\mathcal C^3\). Then
\[
\lambda_k(\Delta_{\nu_u})
\geq
ck,
\qquad
k\geq0.
\]
<!-- end proposition-11 -->
<!-- proof-21: body_md -->
If $X$ has law $\nu_u$, then its conjugate variable is $\xi=u'(X)$ and
\[
\partial u'(x,y)
=
\frac{u'(x)-u'(y)}{x-y}
\geq c,
\]
and on the diagonal $\partial u'(x,x)=u''(x)$. Thus $X$ satisfies $CD(c,\infty)$, and Theorem~\ref{main} gives compact resolvent. Moreover, the free Caffarelli contraction theorem~\cite[Theorem~7]{Di4} gives $\operatorname{Lip}(T)\leq c^{-1/2}$. The map $f\mapsto f\circ T$ is an isometry from $L^2(\nu_u)$ into $L^2(\eta)$, and the chain rule for the free difference quotient yields, first for a polynomial $f$, the following comparison of the Dirichlet forms $\mathcal E_\eta$ and $\mathcal E_{\nu_u}$:
\[
\mathcal E_\eta(f\circ T)
\leq
\frac1c\mathcal E_{\nu_u}(f).
\]
Standard approximation for $f$ and closedness then extend this inequality to the full form domain.
The min--max principle therefore yields the following comparison of eigenvalues: 
\(
\lambda_k(\Delta_{\nu_u})\geq c\lambda_k(\Delta_\eta)=ck
\)
for every $k\geq0$. The reader can consult similar details in E. Milman~\cite[Section~3]{MilmanSpectral} and \cite{Di4} for precise explicit computations of the free Laplacian in this one-dimensional case.
