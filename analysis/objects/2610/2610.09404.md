<!-- theorem-1: body_md -->
\label{ns_max}
Fix an integer $d\ge1$.
Let $Z_1,\ldots,Z_n$ be jointly Gaussian with standard normal
marginals, and suppose that
\begin{align}\label{ns_rows}
 \max_{1\le i\le n}
 \sum_{j=1}^n|\mathbb{E}[Z_iZ_j]|^d\le K<\infty.
\end{align}
For every $f\in L^2(\gamma)$ with Hermite rank at least $d$,
\begin{align}\label{ns_l2}
 \mathbb{E}\left[\max_{0\le m\le n}
 \left|\sum_{k=1}^m f(Z_k)\right|^2\right]
 \le C_{d,K}n\|f\|_2^2.
\end{align}
More generally, for every $2<p<\infty$ and every
$f\in L^p(\gamma)$ with Hermite rank at least $d$,
\begin{align}\label{ns_bdd}
 \left\|\max_{0\le m\le n}
 \left|\sum_{k=1}^m f(Z_k)\right|\right\|_p
 \le C_{p,d,K}\sqrt n\,\|f\|_p.
\end{align}

<!-- remark-1: body_md -->
\rm
(i) The constant $C_{d, K}$ in  the endpoint estimate
\eqref{ns_l2} is uniform in $f\in L^2(\gamma)$ and
in the   covariance matrix, subject to the Hermite-rank
condition and row-sum bound \eqref{ns_rows}.
The exponent $p=2$ is critical for the usual dyadic maximal
argument.  A block estimate of order $\sqrt{|I|}$ in $L^p$
produces the factors $2^{-j(\frac{1}{2}-\frac{1}{p})}$ at level
$j$.  These form a summable geometric series for $p>2$;
at $p=2$ the decay disappears, so summation over the dyadic
levels produces a logarithmic loss.\footnote{The geometric
series is evaluated in \eqref{lp_geometric} and used in
\eqref{lp_max_proof}.  At $p=2$, its finite counterpart has
$L+1$ terms, where $L$ is of order $\log n$.}
Indeed, from the orthogonality relation \eqref{OL_pq},
we can deduce that
$\|\sum_{i\in I}f(Z_i)\|^2_2
 \le\|f\|_2^2\sum_{i,j\in I}|\mathbb{E}[Z_iZ_j]|^d
 \le K|I|\,\|f\|_2^2$,
and direct dyadic summation gives only
$C\sqrt{Kn}\log(2n)\|f\|_2$ for the maximal $L^2$ norm.
The main estimate \eqref{ns_l2} removes this logarithmic loss
under the sole finite-variance assumption.
It is the key input used for
the BMD principles in Section~\ref{bm_intro}.

\smallskip

(ii) For independent Gaussian coordinates, \eqref{ns_l2} follows
from Doob's inequality.  In the dependent setting, the partial
sums of $f(Z_k)$ are generally not martingales, so neither
Doob's inequality nor the Burkholder--Davis--Gundy inequalities
\cite{BDG72} apply directly; see
\cite[Chapters~II and~IV]{RY99} for standard treatments.
The right-hand side of \eqref{ns_l2} is $n\|f\|_2^2$, not the terminal variance.
For $G\sim\mathcal{N}(0,1)$, the choice $Z_1=G$, $Z_2=-G$, and
$f(x)=x$ satisfies \eqref{ns_rows} with $d=1$, $K=2$, but the
terminal sum is zero and the maximal partial sum is $|G|$.
Gaussian entropy and concentration inequalities
  give the bound when $f(x)=ax$ and
$d=1$; see, e.g.,  \cite[Chapters~1--2]{AT07}.
For nonlinear $f$, however, the partial-sum process need not
be Gaussian, and square integrability alone does not supply
sub-Gaussian increment control.  For a fixed finite Hermite
expansion, Nelson's hypercontractivity, followed
by a higher-moment maximal argument, gives a bound with a
constant depending on the Hermite cutoff; see
\cite[Section~2.8]{NP12}.  The endpoint \eqref{ns_l2} controls
the full $L^2(\gamma)$ class of the prescribed  Hermite rank 
uniformly in that cutoff.
 


\smallskip

(iii) The resulting $\sqrt n$ scale agrees with that for independent
sums and square-integrable martingales, and is optimal: for
independent standard Gaussian coordinates and $f=h_d$, the
terminal sum already has $L^2$ norm $\sqrt n$.

\label{bm_thm}
Assume that $\varphi\in L^2(\gamma)$ has Hermite rank $d\ge1$ as in
\eqref{d_phi} and $\rho_G\in\ell^d(\mathbb{Z})$.  Then the
finite-dimensional distributions of $V_n$ converge to those of
$\sigma_G\mathcal{W}$, where
\begin{align}\label{d_var}
 \sigma_G^2
 =\sum_{q=d}^{\infty}q!c_q^2
   \sum_{k\in\mathbb{Z}}\rho_G(k)^q
 \in[0,\infty).
\end{align}

<!-- proposition-1: body_md -->
\label{path_cor}
If $f,g\in L^2(\gamma)$ and $f-g$ has Hermite rank at least $d$,
then, under \eqref{ns_rows},
\begin{align}\label{path_l2}
 \mathbb{E} \big[   \|\mathcal S_n^f-\mathcal S_n^g\|_\infty^2 \big]
 \le C_{d,K}\|f-g\|_2^2.
\end{align}
The constant is uniform over all $n$ and all covariance matrices
satisfying \eqref{ns_rows}.  In particular, for
$\varphi_M=\sum_{q=d}^M c_qH_q$,
\begin{align}\label{path_tail}
 \sup_n\mathbb{E}  \big[   \|\mathcal S_n^\varphi
                  -\mathcal S_n^{\varphi_M}\|_\infty^2 \big]
 \le C_{d,K}\sum_{q>M}q!c_q^2\xrightarrow{M\to+\infty}0.
\end{align}

\noindent
The same estimates hold for the polygonal paths
$\widehat{\mathcal S}_n^f$, obtained by linear interpolation of
$\mathcal S_n^f$ at $0, 1/n, 2/n,\ldots,1$.

<!-- proposition-2: body_md -->
\label{ns_int}
Fix an integer $d\ge1$.  Let $Y=(Y(s))_{s\ge0}$ be a jointly
measurable Gaussian process with standard normal marginals.
Suppose that there is some $K<\infty$ such that
\begin{align}\label{ns_fib}
 \sup_{u\in[0,1]}\sup_{i\in\{0,1,\ldots\}}
 \sum_{j=0}^{\infty}
 \big|\mathbb{E}[Y(1+u+i)Y(1+u+j)]\big|^d\le K.
\end{align}
For $g\in L^2(\gamma)$ with Hermite rank at least $d$, define
\begin{align}\label{int_proc}
 F_\varepsilon^g(t)=\sqrt\varepsilon\int_0^{t/\varepsilon}g(Y(s))\,ds,
 \quad t\ge0.
\end{align}
These processes have almost surely continuous paths and satisfy
\begin{align}\label{ns_intbd}
 \sup_{0<\varepsilon\le1}
 \mathbb{E}\left[\sup_{t\in[0,1]}|F_\varepsilon^g(t)|^2\right]
 \le C_{d,K}\|g\|_2^2.
\end{align}
For $f,g\in L^2(\gamma)$ such that $f-g$ has Hermite rank at least
$d$, the same estimate holds for $F_\varepsilon^f-F_\varepsilon^g$, with
$\|f-g\|_2^2$ on the right-hand side.

The same conclusion holds for $F_\varepsilon^\varphi$.

<!-- lemma-2: body_md -->
\label{diag_lem}
Let $\mathcal{I}$ be a countable index set,
and let
 $(Z_j)_{j\in\mathcal I}$ be a   centered Gaussian
family with standard normal marginals.
Define
$r(i,j)=\mathbb{E}[Z_iZ_j]$ for $i,j\in\mathcal{I}$.
 Suppose that there is some $\varepsilon\in(0,\frac{1}{3})$ such that
\begin{align}\label{small_c1}
 \sup_{i\ne j}|r(i,j)|\le\varepsilon ,
\end{align}
and
\begin{align}\label{small_c2}
   K_d:=\sup_{i\in\mathcal I}
       \sum_{j\in\mathcal I}|r(i,j)|^d<\infty.
\end{align}
Given $A\subset\mathcal I$  with cardinality
$|A|  \leq m$,
the following statements hold, with
a constant $C$ depending only on $d$, $\varepsilon$, and $K_d$,
where
a prime on a multiple sum means that all
displayed indices are pairwise distinct.

\smallskip
\textup{(i)} If $f_1, f_2, f_3, f_4\in L^2(\gamma)$ have Hermite rank at
least
$d$, then
\begin{align}\label{diag4}
  \sum_{t_1,\ldots,t_4\in A}'
  \left|\mathbb{E}\left[\prod_{i=1}^4f_i(Z_{t_i})\right]\right|
  \le Cm^2\prod_{i=1}^4\|f_i\|_2.
\end{align}

\textup{(ii)} If $f_0,f_1,f_2\in L^2(\gamma)$ all have Hermite rank at least
$d$, then
\begin{align}\label{diag3}
 \sum_{x,y,z\in A}'
 |\mathbb{E}[f_0(Z_x)f_1(Z_y)f_2(Z_z)]|
 \le Cm^{\frac{3}{2}}\prod_{i=0}^2\|f_i\|_2.
\end{align}

\textup{(iii)}    Let $1\le s<d$ and $B=bh_s$ for some $b\in\mathbb{R}$,
where $h_s$ is given as in \eqref{def_hq}.
Let $f_1,f_2\in L^2(\gamma)$ have
Hermite rank at least $d$.  Then
\begin{align}\label{diag_mix}
 \sum_{x,y,z\in A}'
 |\mathbb{E}[B(Z_x)f_1(Z_y)f_2(Z_z)]|
 \le C |b|    m^{2-\frac{s}{2d}}\|f_1\|_2\|f_2\|_2.
\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
For parts~\textup{(i)} and \textup{(ii)}, we can apply
\cite[Lemma~1]{BS13a} with scalar coordinates, common rank threshold $d$,
and all $p$ functions subject to that threshold, where $p=4$ and $p=3$,
respectively.  That result applies to nonstationary Gaussian arrays.  In
its notation, $\nu=1$, $\alpha=p$, and
\begin{align*}
 Q_A:=\max_{t\in A}\sum_{s\in A\setminus\{t\}}|r(t,s)|^d\le K_d
\end{align*}
with $K_d$ as in \eqref{small_c2}.
Its small-correlation hypothesis is $\varepsilon<\frac{1}{p-1}$, which is
satisfied by
$\varepsilon\in(0,\frac{1}{3})$.
As a consequence, we have
\begin{align*}
 \sum_{t_1,\ldots,t_p\in A}'
 \left|\mathbb{E}\prod_{i=1}^pf_i(Z_{t_i})\right|
 \le C|A|^{\frac{p}{2}}Q_A^{\frac{p}{2}}\prod_{i=1}^p\|f_i\|_2.
\end{align*}
This gives \eqref{diag4} and \eqref{diag3}.

\medskip

We next prove part~\textup{(iii)}.  We first assume that $f_1$ and
$f_2$ have finite Hermite expansions:
\begin{align*}
 f_1=\sum_{q\ge d}u_qh_q,\quad
 f_2=\sum_{\ell\ge d}v_\ell h_\ell .
\end{align*}

\noindent
Recall that
\begin{align*}
 \|B\|_2=|b|,\quad
 \|f_1\|_2^2=\sum_{q\ge d}u_q^2,
 \quad{\rm and}\quad
 \|f_2\|_2^2=\sum_{\ell\ge d}v_\ell^2.
\end{align*}

Let us first represent the Gaussian family as
$
 Z_j= \mathcal{X} (e_j)
 $,
 where $\mathcal{X} $ is an isonormal Gaussian process over a real separable Hilbert
 space
$\mathfrak H$ and
the 
$\{e_j\}$ are unit vectors such that
 $\langle e_i,e_j\rangle_{\mathfrak H}=r(i,j)$;
see \cite[Proposition 7.2.3]{NP12}.  Since
$
 h_q(Z_j)=\frac{1}{\sqrt{q!}}I_q(e_j^{\otimes q}),
$
we deduce from the product formula \cite[Theorem 2.7.10]{NP12} that

\noindent
\begin{align*}
 h_q(Z_y)h_\ell(Z_z)
 &=
 \frac{1}{\sqrt{q!\ell!}}
 \sum_{c=0}^{q\wedge\ell}
 c!\binom qc\binom\ell c
 r(y,z)^c
 I_{q+\ell-2c}\big( e_y^{\otimes(q-c)}  \widetilde{\otimes}
 e_z^{\otimes(\ell-c)}  \big),
\end{align*}

\noindent
where  $e_y^{\otimes(q-c)}  \widetilde{\otimes}  e_z^{\otimes(\ell-c)} $
is the symmetrization of $e_y^{\otimes(q-c)}  \otimes
e_z^{\otimes(\ell-c)}$.
Then, we deduce from  the orthogonality relation
\cite[Proposition 2.7.5]{NP12} that
 only the value of $c$
satisfying
$q+\ell-2c=s$
can contribute to $\mathbb{E}[ h_s(Z_x) h_q(Z_y)h_\ell(Z_z) ]$.
Put
 $\alpha=q-c$ and $\beta=\ell-c.$
Then,
$q+\ell-2c=s$ is equivalent to
\begin{align}\label{deg_rel}
 \alpha+\beta=s,\quad
 q=\alpha+c,\quad
 \ell=\beta+c.
\end{align}
In particular, since $q,\ell\ge d$,
we have
\begin{align}\label{c_low}
 c\ge d-\min(\alpha,\beta).
 \end{align}
 Since
$ \langle
 e_y^{\otimes\alpha}\widetilde{\otimes}
 e_z^{\otimes\beta},
 e_x^{\otimes s}
  \rangle_{\mathfrak H^{\otimes s}}
 = \langle
 e_y^{\otimes\alpha} \otimes
 e_z^{\otimes\beta},
 e_x^{\otimes s}
  \rangle_{\mathfrak H^{\otimes s}}
 =
 r(x,y)^\alpha r(x,z)^\beta$ for $\alpha + \beta =s$,
 we have
\begin{align}\label{3diag_exact}
\begin{aligned}
 \mathbb{E}[h_s(Z_x)h_{\alpha+c}(Z_y)h_{\beta+c}(Z_z)]
& = \frac{c! \binom{\alpha+c}{c} \binom{\beta+c}{c} s!}{ \sqrt{s! (\alpha+c)!
(\beta+c)! }}
 r(x,y)^\alpha r(x,z)^\beta r(y,z)^c
\\
&= \frac{\sqrt{s!(\alpha+c)!(\beta+c)!}}{\alpha!\beta!c!}\,
 r(x,y)^\alpha r(x,z)^\beta r(y,z)^c.
\end{aligned}
\end{align}
Again by  $\alpha+\beta=s$, the factorial coefficient in
\eqref{3diag_exact} satisfies
\begin{align*}
 \frac{\sqrt{s!(\alpha+c)!(\beta+c)!}}{\alpha!\beta!c!}
 =  \frac{\sqrt{s!}}{\alpha!\beta!}
 \left(    \frac{(\alpha+c)!}{c!}   \frac{(\beta+c)!}{c!}
 \right)^{\frac{1}{2}}
 \leq C_s(1+c)^{\frac{s}{2}}.
\end{align*}
Therefore,
\begin{align}\label{wick3}
 \big|   \mathbb{E}[h_s(Z_x)h_{\alpha+c}(Z_y)h_{\beta+c}(Z_z)]  \big|
\leq   C_s(1+c)^{s/2}
 |r(x,y)|^\alpha
 |r(x,z)|^\beta
 |r(y,z)|^c.
\end{align}

We next sum the correlation factors over the distinct indices.
Since $\alpha+\beta=s$, at least one of $\alpha,\beta$ is at least $s/2$.
Without loss of generality, let us assume
$
 \beta\geq  s/2,
$
while the case $\alpha\geq  s/2$ can be dealt with in the same way.
With $\beta\geq s/2$,
we have
 $\beta\ge\alpha$, and thus
$c\ge d-\alpha$ by \eqref{c_low}.
 For fixed distinct $x,z$,
 it follows from
\eqref{small_c1}
that

\noindent
\begin{align*}
\sum_{y\in A\setminus\{x,z\}}
 |r(x,y)|^\alpha |r(y,z)|^c
\leq
 \varepsilon^{\alpha+c-d}
 \sum_{y\in A\setminus\{x,z\}}
 |r(x,y)|^\alpha |r(y,z)|^{d-\alpha}.
\end{align*}
If $1\le\alpha<d$, we can deduce from H\"older's inequality
that

\noindent
\begin{align*}
 &\sum_y
 |r(x,y)|^\alpha |r(y,z)|^{d-\alpha}
\leq
 \left(\sum_y|r(x,y)|^d\right)^{\frac{\alpha}{d}}
 \left(\sum_y|r(y,z)|^d\right)^{\frac{d-\alpha}{d}}\le K_d,
\end{align*}
with $K_d$ as in \eqref{small_c2}.
The same bound is also immediate for $\alpha=0$.  Thus,
\begin{align}\label{conv_bd}
 \sup_{x\ne z}
 \sum_{y\in A\setminus\{x,z\}}
 |r(x,y)|^\alpha |r(y,z)|^c
 \le K_d\varepsilon^{\alpha+c-d}.
\end{align}

We now sum the remaining factor $|r(x,z)|^\beta$.  By H\"older's inequality,
we have
\begin{align}
 \sum_{x,z\in A}
 |r(x,z)|^\beta \mathbf{1}_{\{x\ne z\}}
& \le
 |A|^{2-2\frac{\beta}{d}}
 \Bigg(
   \sum_{x,z\in A }
   |r(x,z)|^d\mathbf{1}_{\{x\ne z\}}
 \Bigg)^{\frac{\beta}{d}} \notag \\
& \le
 |A|^{2-2\frac{\beta}{d}}(|A|K_d)^{\frac{\beta}{d}}  \notag  \\
& \leq
 C m^{2-\frac{\beta}{d}} \leq   C m^{2-\frac{s}{2d}}.
 \label{edge_sum}
\end{align}
Here, $\beta\le s<d$ by \eqref{deg_rel},
so that the factor $K_d^{\beta/d}$, bounded by $\max\{1, K_d\}$,
 can be absorbed into
the constant $C$. The last inequality in \eqref{edge_sum}
is due to   $\beta\ge s/2$.

Combining \eqref{conv_bd} and \eqref{edge_sum}, and interchanging
$y,z$ when $\alpha>\beta$, yields
\begin{align}\label{fixed_diagram_sum}
 &\sum_{x,y,z\in A}'
 |r(x,y)|^\alpha
 |r(x,z)|^\beta
 |r(y,z)|^c
 \leq
 C\varepsilon^{\min\{\alpha,\beta\}+c-d}
 m^{2-\frac{s}{2d}}.
\end{align}

\smallskip

We finally sum over the Hermite orders.
In view of \eqref{wick3} and \eqref{fixed_diagram_sum},
each fixed pair
$(\alpha,\beta)$ with $\alpha+\beta=s$ contributes to the left side of
\eqref{diag_mix},
and its contribution is bounded by

\noindent
\begin{align*}
 &C|b|m^{2-\frac{s}{2d}}
 \sum_{c\ge d-\min(\alpha,\beta)}
 |u_{\alpha+c}v_{\beta+c}|
 (1+c)^{\frac{s}{2}}\varepsilon^{\min\{\alpha,\beta\}+c-d}.
\end{align*}

\noindent
Since $\varepsilon\in(0,\frac{1}{3})$, we have

\noindent
\begin{align*}
\sup_{\substack{\alpha,\beta\ge0\\ \alpha+\beta=s}} \sup_{c\ge d- \min\{\alpha,\beta\}}
 (1+c)^{\frac{s}{2}}\varepsilon^{\min\{\alpha,\beta\}+c-d}
 <\infty.
\end{align*}
Therefore, we can apply the above uniform bound
and  Cauchy--Schwarz to get
\begin{align*}
 \sum_{c\ge d-\min\{\alpha,\beta\}}
 |u_{\alpha+c}v_{\beta+c}|
 (1+c)^{\frac{s}{2}}\varepsilon^{\min\{\alpha,\beta\}+c-d}
&\leq
 C_{d,s,\varepsilon}
 \bigg(\sum_{q\ge d}u_q^2\bigg)^{\frac{1}{2}}
 \bigg(\sum_{\ell\ge d}v_\ell^2\bigg)^{\frac{1}{2}} \\
&=
 C_{d,s,\varepsilon}\|f_1\|_2\|f_2\|_2.
\end{align*}
There are only finitely many pairs $(\alpha,\beta)$ with
$\alpha+\beta=s$.
As a result, we get

\noindent
\begin{align}\label{diag_mix_finite}
 \sum_{x,y,z\in A}'
 |\mathbb{E}[B(Z_x)f_1(Z_y)f_2(Z_z)]|
 \le
 C|b|m^{2- \frac{s}{2d}}
 \|f_1\|_2\|f_2\|_2,
\end{align}
when $f_1, f_2$ have finite Hermite expansions, or equivalently,
when they are polynomials.
Note that the constant $C$ in \eqref{diag_mix_finite}
does not depend on the degrees of the polynomials $f_1, f_2$.

It remains to remove the finite-expansion assumption.  Let
$f_{1,N} = \sum_{q=d}^N u_q h_q$
and $f_{2,N} = \sum_{\ell=d}^N v_\ell h_\ell$.
 Then,
 $f_{i,N} \to f_i$
in  $L^2(\gamma)$
and
$ \|f_{i,N}\|_2\le\|f_i\|_2$ for $i=1,2$.

For distinct $x,y,z$, let $\Sigma$ be the covariance matrix of
$(Z_x,Z_y,Z_z)$.   By \eqref{small_c1} and Gershgorin's circle theorem
(see, e.g., \cite[Theorem~1.1]{Var04}),
the spectrum of the covariance matrix $\Sigma$
is contained in $ [1-2\varepsilon,1+2\varepsilon]\subset(0,2)$.
In particular, $\Sigma$ is positive definite, so the law of
$(Z_x,Z_y,Z_z)$ is nondegenerate and admits a density $L$ with
respect to $\gamma^{\otimes3}$.  Moreover, the operator norm $\|\Sigma\|<2$ implies that
$2\Sigma^{-1}-I$ is positive definite, and hence
$L\in L^2(\gamma^{\otimes3})$.
Consequently,

\noindent
\begin{align*}
\Big\vert  \mathbb{E}\big[
& B(Z_x)(f_{1,N}-f_1)(Z_y)f_{2,N}(Z_z)  \big] \Big\vert  \\
 &= \left\vert  \int_{\mathbb{R}^3 } B(w_1)(f_{1,N}-f_1)(w_2)f_{2,N}(w_3) L(w_1,
 w_2, w_3) \gamma^{\otimes 3}(dw_1, dw_2, dw_3) \right\vert\\
 & \le
  \| L\|_2  |b|
 \|f_{1,N}-f_1\|_2
 \|f_{2,N}\|_2
 \xrightarrow{N\to+\infty}0,
\end{align*}
and the analogous estimate holds for the second factor.  Thus,
for every distinct triple $(x,y,z)$,
\begin{align*}
 \mathbb{E}[B(Z_x)f_{1,N}(Z_y)f_{2,N}(Z_z)]
 \longrightarrow
 \mathbb{E}[B(Z_x)f_1(Z_y)f_2(Z_z)].
\end{align*}
Since the primed sum over $A$ is finite, we may pass to the limit
in \eqref{diag_mix_finite}.  This proves \eqref{diag_mix} and
hence  completes the proof of Lemma  \ref{diag_lem}.
\qedhere

\rm
\label{rem_mor}
It is useful to compare  Lemma \ref{max_lem}
with the classical
moment-to-maximum inequality of M\'oricz
\cite[Theorem~1]{Mor76}.  In a simplified form, M\'oricz's result
starts from a one-block moment estimate
$
 \mathbb{E}|S_I|^p\le C|I|^\beta
 $
 with 
 $\beta>1$,
and yields a maximal estimate of the same power order.
In the present setting, the natural one-block estimate is only
$
 \mathbb{E}[S_I^2]\le C|I|.
$
Thus the one-block second moment lies at the borderline $\beta=1$,
and the elementary dyadic argument based on this estimate gives
$\mathbb{E}\big[ \max_{m\le N}|S_{(0,m]}|^2 \big] \leq C N(\log(2N))^2.$
Uniform finite-variance approximation therefore requires a
sharper maximal bound.


 

The additional information available here has a different form.
For adjacent disjoint blocks $I,J$, we prove the bound \eqref{abs_adj},
in which the strict power gain is  sufficient
to recover the diffusive maximal bound \eqref{abs_max}
through a dyadic argument.
In this sense,  Lemma \ref{max_lem} is complementary to M\'oricz's principle:
M\'oricz exploits a superlinear moment bound for a single block,
whereas here the single-block second moment remains linear and the
strict gain appears only after coupling two adjacent blocks.

The proof is divided into three steps.  Throughout, $(\ell,r]$
denotes the integer interval $(\ell,r]\cap\mathbb N$.

\smallskip
\noindent
{\bf Step 1: Reduction to a two-sided cut maximal function.}
Choose the smallest dyadic integer $M=2^L$ such that $M\ge N$,
so that $N\le M<2N$, and set
$X_k=0$ for $N<k\le M$.  This allows us to work on a complete
dyadic tree down to singleton intervals, while changing the relevant
scales only by an absolute constant.
For an integer interval $D=(\ell,r]\subset(0,M]$, define
\begin{align}\label{cut_def}
 R(D)
 :=
 \max_{\ell\le m\le r}
 \min\big\{
 |S_{(\ell,m]}|,
 |S_{(m,r]}|
 \big\}.
\end{align}
Thus $R(D)$ measures, over all cuts of $D$, the largest possible
value of the smaller of the two partial sums created by the cut.

Define  $P_m=\sum_{k=1}^mX_k$.
Then, we have
$S_{(0,m]}=P_m$ and
$S_{(m,M]}=P_N-P_m.$
Therefore,
$\min\{|P_m|,|P_N-P_m|\}\le R((0,M])$.
Moreover,
\begin{align}\label{prefix_bd}
 |P_m|
 \le
 |P_N|
 +
 \min\{|P_m|,|P_N-P_m|\}
 \le
 |P_N|+R((0,M]).
\end{align}

\noindent
Hence,  due to Assumption \eqref{abs_var}, it remains to prove
$
 \|R((0,M])\|_2 \leq C\sqrt N.
$

\smallskip
\noindent
{\bf Step 2: Dyadic recursion for the cut maximal function.}
Let $D=(\ell,r]$ be a dyadic interval,
 $a=\frac{\ell+r}{2}$,
and write $D^-=(\ell,a]$ and $D^+=(a,r]$.
Set $z_D= \min\{|S_{D^-}|,|S_{D^+}|\}.$
We claim that
\begin{align}\label{dy_rec}
 R(D)
 \le
 z_D+\max\{R(D^-),R(D^+)\}.
\end{align}

To prove this, first note the elementary inequality\footnote{To
verify \eqref{min_ineq}, if $|a|\le |b|$, then it is immediate.
  If $|b|<|a|$, then
$
 \min\{|a|,|b+c|\}-|b|
 =\min\{|a|-|b|,\ |b+c|-|b|\}
 \le\min\{|a+b|,\ |c|\},
$
which implies \eqref{min_ineq}.}
\begin{align}\label{min_ineq}
 \min\{|a|,|b+c|\}
 \le
 \min\{|a|,|b|\}
 +
 \min\{|a+b|,|c|\}.
\end{align}

Consider  $m\in D^-$.  Then,
we apply   \eqref{min_ineq}
with $a_1:=S_{(\ell,m]}$, $b_1:=S_{(m,a]}$, and
$c_1:=S_{D^+}$
to get

\noindent
\begin{align*}
 \min\big\{  |S_{(\ell,m]}|,   |S_{(m,r]}|  \big\}
 =  \min\big\{  |a_1 |,   |b_1 + c_1 |  \big\}
 &\leq
 \min\{|a_1|,|b_1|\} +  \min\{|a_1+b_1|,|c_1|\} \\
 & \leq R(D^-)+z_D.
\end{align*}

\noindent
For $m\in D^+$, the same argument  yields
$\min\big\{ |S_{(\ell,m]}|,   |S_{(m,r]}| \big\} \leq  z_D+R(D^+).$
Taking the maximum over all  $m\in D$ proves \eqref{dy_rec}.

At dyadic level $j$, the interval $(0,M]$ is partitioned into
$2^j$ dyadic subintervals
\begin{align*}
 D_{j,k}
 =
 \big(kM2^{-j},(k+1)M2^{-j}\big],
 \quad k=0,\ldots,2^j-1,
\end{align*}
each of length
$ \ell_j=M2^{-j}$,
  $0\le j<L$.
Define
\begin{align}\label{def_ZJ}
 Z_j :=  \max\{  z_D:  D\text{ is a dyadic interval   at level $j$} \}.
\end{align}
Iterating \eqref{dy_rec} from the root down to singleton sets gives
\begin{align}\label{cut_sum}
 R((0,M])
 \leq
 \sum_{j=0}^{L-1}Z_j.
\end{align}
Indeed, at each generation one pays at most the largest $z_D$ on that
level, while at the final level the cut maximal function of a singleton
is zero.

\smallskip
\noindent
{\bf Step 3: Estimate each dyadic scale and sum over the scales.}
Fix a level $j$.  There are
$M/\ell_j$
dyadic intervals of length $\ell_j$.  For each such interval  $D$,

\noindent
\begin{align}\label{ZD4}
 z_D^4
 =
 \min\{|S_{D^-}|,|S_{D^+}|\}^4
 \le
 S_{D^-}^2S_{D^+}^2.
\end{align}
After intersecting $D^-$ and $D^+$ with $\{1,\ldots,N\}$, the two
sets remain adjacent integer intervals.  If one of them is empty, then
$z_D=0$.  Otherwise, by \eqref{abs_adj} and \eqref{ZD4},
we have

\noindent
\begin{align*}
 \mathbb{E}[z_D^4]
 \le
 C_1N^{1-\alpha}\ell_j^{1+\alpha}.
\end{align*}
By the definition \eqref{def_ZJ} of $Z_j$,
we can write, with $M<2N$,

\noindent
\begin{align}
 \mathbb{E}[Z_j^4]
 \le
 \sum_{|D|=\ell_j}\mathbb{E}[z_D^4]
 \le
 \frac{M}{\ell_j}
 C_1N^{1-\alpha}\ell_j^{1+\alpha}
 =
 C_1MN^{1-\alpha}\ell_j^\alpha 
 \le
 2C_1N^{2-\alpha}\ell_j^\alpha .
 \label{Z4_bd}
\end{align}
 

\noindent
Therefore,
 $\|Z_j\|_2 \le  \|Z_j\|_4  \le C_1'  N^{\frac{1}{2}- \frac{\alpha}{4}}
 \ell_j^{\alpha/4}.$
Combining \eqref{cut_sum} with Minkowski's inequality yields,
with $M < 2N$,
\begin{align*}
 \|R((0,M])\|_2
 \le
 \sum_{j=0}^{L-1}\|Z_j\|_2
 &\le
 C_1' N^{\frac{1}{2}- \frac{\alpha}{4}}
 \sum_{j=0}^{L-1}
 (M2^{-j})^{\alpha/4}\\
 &\leq
2^{\frac{\alpha}{4}} C_1'  N^{\frac{1}{2} }
 \sum_{j=0}^{\infty}
 2^{-j\alpha/4}.
\end{align*}

\noindent
Since $\alpha>0$, we get
$ \|R((0,M])\|_2 \le C_{\alpha,C_1}\sqrt N$.\footnote{This is precisely where
the
 condition $\alpha>0$ is used; if
$\alpha=0$, the sum over the dyadic scales would be of order
$\log N$ rather than uniformly bounded.}
Finally,   from \eqref{abs_var}  and  \eqref{prefix_bd},
we get
\[
 \left\|
 \max_{1\le m\le N}|P_m|
 \right\|_2
 \le
 \|P_N\|_2+\|R((0,M])\|_2
 \le
 C_{\alpha,C_0,C_1}\sqrt N.
\]
Hence, the proof is completed.
\qedhere

\end{align*}
Since $L=1+\lfloor K8^d\rfloor$ depends only on $d$ and $K$,
