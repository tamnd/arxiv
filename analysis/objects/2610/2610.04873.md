<!-- theorem-1: body_md -->
\label{HJBreformulation.theorem}
Suppose that ${\bf A}$, ${\bf B}$, ${\bf P}$, $\mathbf{Q}$, and $\mathbf{\Pi}$ are as in \eqref{A.def},
\eqref{B.def}, \eqref{P.def}, \eqref{Q.def}, and \eqref{Pi.def}. Under the ansatz \eqref{P.def}, the
lifted HJB equation \eqref{HJB.linearization} is equivalent to
\begin{align}
    \Psi^\top
    \Big( \mathbf{\Pi}\bar{\bf A}+\bar{\bf A}^\top \mathbf{\Pi}^\top + {\bf Q}
    - \frac{1}{r}\mathbf{\Pi}\mathbf{g}\mathbf{g}^\top\mathbf{\Pi}^\top\Big)\Psi=0,
    \label{HJB.linearization.invariant}
\end{align}
and the control law \eqref{eqn:bilinearHJB} becomes
\begin{equation} \label{eqn:bilinearHJB.invariant}
u^*=-\frac{1}{r}\, \Psi^\top \mathbf{\Pi}\, {\bf g} .
\end{equation}

<!-- corollary-1: body_md -->
\label{U.cor}
Let $\mathbf{\Pi}$ be a lower-triangular Toeplitz matrix of the form
\eqref{Pi.def} that satisfies \eqref{HJB.linearization.invariant}, and set
\begin{equation}\label{U.def}
U(x) := 2\,\Psi^\top\mathbf{\Pi}\,\mathbf{e}_1
= 2\sum_{m=2}^\infty \pi_m(\mathbf{P})\,x^{m-1} .
\end{equation}
Then, $U(x)=2x^{-1}\Psi^\top{\bf P}\Psi$ and the function $U$ satisfies the HJB
equation
\begin{equation}
\label{U.HJB}
q(x)+U(x) f(x)-\frac{1}{4r} \big(g(x) U(x)\big)^2=0,
\end{equation}
and the control \eqref{eqn:bilinearHJB.invariant} equals
$u^*=-\frac{1}{2r}g(x)U(x)$, as in \eqref{eqn:uStar}. Conversely, let $U$ be
analytic around the origin with $U(0)=0$ and satisfy \eqref{U.HJB}. Then, the
Toeplitz matrix $\mathbf{\Pi}$ whose entries $\pi_m$ are one half of the
coefficient of $x^{m-1}$ in $U$ satisfies \eqref{HJB.linearization.invariant}.

<!-- remark-1: body_md -->
By \eqref{P.def} and the chain rule,
\begin{align}
\label{VxU.eqn}
    V_x=\frac{dV}{d\Psi}\frac{d\Psi}{dx}= x^{-1} V_\Psi^\top\mathbf{D} \Psi= 2x^{-1} \Psi^\top {\bf P} \Psi =U(x).
\end{align}
Thus, the function $U$ in \eqref{U.def} is the derivative of the lifted value
function along $\Psi(x)$. Along the closed loop, \eqref{U.HJB} and
\eqref{eqn:bilinearHJB.invariant} give
\begin{align}
    \dot{V}&= V_x \big(f(x)+g(x) u^*\big)
    = -\Psi^\top\mathbf{Q}\Psi-\frac{1}{r}\,\Psi^\top\mathbf{\Pi}\mathbf{g}\mathbf{g}^\top\mathbf{\Pi}^\top\Psi
    \nonumber\\
    &=-q(x)-r\,|u^*(x)|^2 \le 0 .
    \nonumber
\end{align}
This is the Lyapunov inequality one expects from an HJB equation. Moreover, the solution constructed in Theorem~\ref{HJBcoefficient.thm} has
$\pi_2({\bf P})>0$ by \eqref{HJBcoefficient.thm.eq1}, and $U$ is analytic
with $U'(0)=2\pi_2({\bf P})>0$. By Section~\ref{sec:problem}, there is only
one such solution of the HJB equation. Hence, $U$ is the closed-form $V_x$ in
\eqref{eqn:V(x)}, the closed-loop system is $\dot x=-x\sqrt{l(x)}$ as in
\eqref{closedloop.explicit}, the origin is asymptotically stable in the sense of Lyapunov, and $V$ is
the value function of the problem. Thus, the positive-diagonal solution of the lifted Riccati-type equation gives the optimal cost while serving as a Lyapunov function.

<!-- theorem-2: body_md -->
\label{HJBcoefficient.thm}
Suppose that the system \eqref{eqn:GeneralNonlinear} with $f$, $g$ and $q$
satisfying \eqref{eqn:taylorAssumption}, \eqref{costfunction.expansion} and
\eqref{g0.requirement}, and let $\Delta>0$ be as in \eqref{g0f1.requirement}.
Then, the Riccati-type equation \eqref{HJB.linearization.invariant} has
exactly one lower-triangular Toeplitz solution $\mathbf{\Pi}$ with positive
diagonal entries, where its diagonal entry is
\begin{equation}
 \label{HJBcoefficient.thm.eq1}
\pi_2({\bf P})=
r g_0^{-2} (f_1+\sqrt{\Delta}),
\end{equation}
and its remaining entries $\pi_{m+2}({\bf P})$, for $m\ge 1$, are given
recursively by
\begin{align}   \label{HJBcoefficient.thm.eq2}
 & \hspace{-0.2cm} \pi_{m+2}({\bf P}) =   (2\sqrt{\Delta})^{-1}\Big( q_{m+2}\nonumber \\
& \hspace{-0.2cm} \ \ +
2\sum_{k=0}^{m-1} \big(f_{m-k+1}- g_0 g_{m-k}\pi_2({\bf P})/r\big) \pi_{k+2}({\bf P})\nonumber\\
& \hspace{-0.2cm} \ \ -
\frac{1}{r} \sum_{n=1}^{m-1}
       \sum_{j=0}^n \sum_{j'=0}^{m-n} g_{n-j} g_{m-n-j'}
\pi_{j+2}({\bf P}) \pi_{j'+2}({\bf P})\Big).
\end{align}
Moreover, $2\pi_m({\bf P})=v_{m-1}$ for all $m\ge2$, where $v_k$ are the
Maclaurin coefficients of $V_x$ in \eqref{Vx.expansion}.

<!-- theorem-3: body_md -->
\label{finitesection.thm}
Suppose that the system \eqref{eqn:GeneralNonlinear} with $f$, $g$ and $q$
satisfying \eqref{eqn:taylorAssumption}, \eqref{costfunction.expansion} and
\eqref{g0.requirement}, and let $V_x$ be the solution \eqref{eqn:V(x)} of the
HJB equation \eqref{eqn:nonlinearHJB}. Let $\pi_m({\bf P})$, for $m\ge 2$, be as
in \eqref{HJBcoefficient.thm.eq1} and \eqref{HJBcoefficient.thm.eq2}, and
let $\mathbf{\Pi}_N$ and $U_N$ be as in \eqref{PiN.def} and \eqref{UN.def}.
Then, 
\begin{align}\label{finitesection.thm.eq1}
|U_N(x)-V_x|\le M (1-|x|/\rho)^{-1} |x/\rho|^{N} 
\end{align}
for all $x\in (-\rho, \rho)$ and $N\ge 1$, where $M$ and $\rho$ are the
constants in \eqref{vk.def}.

<!-- remark-2: body_md -->
For an infinite matrix ${\bf C}=[c(i,j)]_{i,j=1}^\infty$, define its Schur
norm by
$$\|{\bf C}\|_{\mathcal S}=\max \Bigg\{\sup_{i\ge 1}\sum_{j=1}^\infty |c(i,j)|,\ \sup_{j\ge 1}\sum_{i=1}^\infty |c(i,j)|\Bigg\}. $$
Suppose that the radius of convergence of $V_x$ in \eqref{Vx.expansion}
exceeds one, and choose $\rho>1$ in \eqref{vk.def}. Since $\mathbf{\Pi}$ is
lower-triangular Toeplitz, every column sum equals
$\sum_{m=2}^\infty|\pi_m({\bf P})|$, and by $2\pi_m({\bf P})=v_{m-1}$ and
\eqref{vk.def},
\begin{align}
    \|{\bf\Pi}\|_{\mathcal S}=\sum_{m=2}^\infty |\pi_m({\bf P})|\le \frac{M}{2}
    \sum_{m=2}^\infty \rho^{-(m-1)} = \frac{M}{2(\rho-1)} . \nonumber
\end{align}
By the Schur test, $\mathbf{\Pi}$ is then a bounded operator on $\ell^p$ for
every $1\le p\le \infty$, where $\ell^p$ is the Banach space of $p$-summable
sequences $\{c(k)\}_{k=1}^\infty$ with its standard norm. The threshold one
reflects the choice of unit weights. 
On the weighted space with norm
$\|c\|_{p, \sigma}:=\big(\sum_{k=1}^\infty |c(k)|^p\sigma^{kp}\big)^{1/p}$ for $1\le p<\infty$ and $\|c\|_{\infty, \sigma}=\sup_{k\ge 1} |c(k)|\sigma^k$, the same computation shows that $\mathbf{\Pi}$ is
bounded for every $0<\sigma<\rho$.
