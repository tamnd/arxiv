\label{lattice_theorem} Let $d \ge 2$, $m \ge 1$ and assume the system $\alpha_1, \alpha_2, \ldots, \alpha_m \in \mathbb{R}^d$ satisfies \eqref{diophantinelattice} with a function $\psi: [1, \infty) \to [1, \infty)$ such that $x^{-d} \psi (x)$ is nondecreasing and there exists a constant $c>0$ for which $\psi(xy) \le c e^{x^2} \psi(y)$ holds for all $x,y \ge 1$. \[ W_{\infty} (\mu_N, \mathrm{Vol}) \ll \frac{(\log N)^{1/2}}{F_{\log N}^{-1} \left( \frac{(\log N)^{1/2} N}{\min \left\{ \log N, \log \left( \frac{N}{\psi^{-1}(N)^d}+2 \right) \right\}} \right)} \]
with implied constants that depend only on $I$, $c$, $d$ and $m$.

\label{lipschitz_lemma} Let $(X, \rho_X)$ and $(Y, \rho_Y)$ be compact metric spaces, and let $f: X \to Y$ be Lipschitz with Lipschitz constant $K \ge 0$, that is, $\rho_Y (f(x),f(y)) \le K \rho_X (x,y)$ for all $x,y \in X$. Let $\mu, \nu \in \mathcal{P}(X)$, and let $f_* \mu, f_* \nu \in \mathcal{P}(Y)$ denote the pushforward measures of $\mu$ and $\nu$. Then $W_p (f_* \mu, f_* \nu) \le K W_p (\mu, \nu)$ for all $1 \le p \le \infty$.

This is clear from the definition of the Wasserstein metric, see \cite[p.\ 110]{VI2}.

<!-- lem-2: body_md -->
\label{WqWplemma} Let $M=\mathbb{R}^d/\mathbb{Z}^d$ or $\mathbb{S}^d$, let $a_1, a_2, \ldots, a_N \in M$ be arbitrary points and let $\mu_N=N^{-1} \sum_{n=1}^N \delta_{a_n}$. For any $1 \le p \le q \le \infty$, we have
\[ N^{1/q} W_q (\mu_N, \mathrm{Vol}) \le 2 N^{1/p} W_p (\mu_N, \mathrm{Vol}) . \]
In particular,
\[ W_{\infty} (\mu_N, \mathrm{Vol}) \le 2e W_{\log N} (\mu_N, \mathrm{Vol}) . \]
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Let $1 \le p \le q < \infty$. The Wasserstein distance from $\mu_N$ to $\nu_N=N^{-1} \sum_{n=1}^N \delta_{b_n}$ with arbitrary points $b_1, b_2, \ldots, b_N \in M$ can be expressed as
\[ W_p (\mu_N, \nu_N) = \min_{\sigma \in S_N} \left( \frac{1}{N} \sum_{n=1}^N \rho_M (a_n, b_{\sigma(n)})^p \right)^{1/p} , \]
where the minimum is over all bijections $\sigma: \{ 1, 2, \ldots, N \} \to \{ 1, 2, \ldots, N \}$, see \cite[p.\ 5]{VI1}. Let $\sigma$ be the bijection for which the minimum is attained in the previous formula. An application of the general inequality $(\sum_{n=1}^N x_n)^{p/q} \le \sum_{n=1}^N x_n^{p/q}$, $x_n \ge 0$, with the exponent $0<p/q \le 1$ leads to
\begin{equation}\label{WqWpinequality}
N^{1/q} W_q(\mu_N, \nu_N) \le \left( \sum_{n=1}^N \rho_M (a_n, b_{\sigma(n)})^q \right)^{1/q} \le \left( \sum_{n=1}^N \rho_M (a_n, b_{\sigma(n)})^p \right)^{1/p} = N^{1/p} W_p (\mu_N, \nu_N) .
\end{equation}

If $M=\mathbb{R}^d / \mathbb{Z}^d$, then let $U$ be a uniformly distributed random variable in $\mathbb{R}^d/\mathbb{Z}^d$, and let $b_n=a_n+U$. If $M=\mathbb{S}^d$, then let $U$ be a uniformly distributed random variable in $\mathrm{SO}(d+1)$, and let $b_n=Ua_n$. In both cases $b_n$ is a uniformly distributed random variable in $M$ for each $1 \le n \le N$, hence $\mathbb{E} \nu_N = \mathrm{Vol}$. As the map $x \mapsto x+U$ on $\mathbb{R}^d/\mathbb{Z}^d$ resp.\ $x \mapsto Ux$ on $\mathbb{S}^d$ is an isometry that preserves the uniform measure $\mathrm{Vol}$, Lemma \ref{lipschitz_lemma} yields $W_p (\nu_N, \mathrm{Vol})=W_p(\mu_N, \mathrm{Vol})$. Using the convexity of the optimal cost \cite[Theorem 4.8]{VI2}, inequality \eqref{WqWpinequality} and the triangle inequality for the Wasserstein metric we thus deduce
\[ \begin{split} N^{1/q} W_q (\mu_N, \mathrm{Vol}) &= N^{1/q} W_q (\mu_N, \mathbb{E}\nu_N) \le \mathbb{E} (N^{1/q} W_q(\mu_N, \nu_N)) \le \mathbb{E} (N^{1/p} W_p (\mu_N, \nu_N)) \\ &\le \mathbb{E} (N^{1/p} W_p (\mu_N, \mathrm{Vol}) + N^{1/p} W_p(\nu_N, \mathrm{Vol})) = 2 N^{1/p} W_p (\mu_N, \mathrm{Vol}) . \end{split} \]
This finishes the proof in the case $1 \le p \le q < \infty$. Taking the limit as $q \to \infty$ establishes the claim for $q=\infty$. In particular, choosing $p=\log N$ and $q=\infty$ leads to $W_{\infty}(\mu_N, \mathrm{Vol}) \le 2 N^{1/\log N}W_{\log N}(\mu_N, \mathrm{Vol}) = 2 e W_{\log N}(\mu_N, \mathrm{Vol})$, as claimed.
<!-- end proof-2 -->

\[ \sqrt{2-\frac{e^c-1}{c}} \left( \int_M f (\Delta^{-1} f) \, \mathrm{d}\mathrm{Vol} \right)^{1/2} \le W_2 (\mu, \mathrm{Vol}) \le \frac{2(1-\sqrt{1-c})}{c} \left( \int_M f (\Delta^{-1} f) \, \mathrm{d}\mathrm{Vol} \right)^{1/2}, \]
where by Green's formula,
\[ \begin{split} \left( \int_M f (\Delta^{-1} f) \, \mathrm{d}\mathrm{Vol} \right)^{1/2} &= \left( \int_M |\nabla (\Delta^{-1} f)|^2 \, \mathrm{d}\mathrm{Vol} \right)^{1/2} \\ &= \sup \left\{ \left| \int_M \langle \nabla (\Delta^{-1} f), \nabla g \rangle \, \mathrm{d}\mathrm{Vol} \right| : \int_M |\nabla g|^2 \, \mathrm{d}\mathrm{Vol} \le 1 \right\} \\ &= \sup \left\{ \left| \int_M f g \, \mathrm{d} \mathrm{Vol} \right| : \int_M |\nabla g|^2 \, \mathrm{d}\mathrm{Vol} \le 1 \right\} \\ &= \sup \left\{ \left| \int_M g \, \mathrm{d}(\mu-\mathrm{Vol}) \right| : \int_M |\nabla g|^2 \, \mathrm{d}\mathrm{Vol} \le 1 \right\} \\ &= \| \mu - \mathrm{Vol} \|_{\dot{H}_{-1}} \end{split} \]
is in fact a certain dual Sobolev norm of the signed measure $\mu - \mathrm{Vol}$. The same upper bound in a more general form appeared also in \cite[Lemma 13]{BC}, whereas the lower bound with a different constant
\[ \frac{\log (1+c)}{2 (\sqrt{1+c}-1)} \int_M f (\Delta^{-1} f) \, \mathrm{d}\mathrm{Vol} \le W_2 (\mu, \mathrm{Vol}) \]
was proved earlier in \cite[Theorem 2.5]{PE}. All three constants
\[ \sqrt{2-\frac{e^c-1}{c}}, \quad \frac{\log (1+c)}{2 (\sqrt{1+c}-1)}, \quad \frac{2(1-\sqrt{1-c})}{c} \]
are $1+O(c)$ for $0<c \le 1$, which proves the claim.

\label{D2torus_lemma} Let $\mu \in \mathcal{P}(\mathbb{R}^d / \mathbb{Z}^d)$. For any $0<r \le 1/2$, we have
\[ D_2(\mu, \mathrm{Vol}, r) = \bigg( \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} b_k(r) |\widehat{\mu}(k)|^2 \bigg)^{1/2} \quad \textrm{and} \quad D_2(\mu, \mathrm{Vol}) = \bigg( \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} b_k |\widehat{\mu}(k)|^2 \bigg)^{1/2} \]
with coefficients
\[ \begin{split} b_k(r) &=\frac{r^d}{|k|^d} J_{d/2}(2 \pi r |k|)^2 \ll \min \left\{ r^{2d}, \frac{r^{d-1}}{|k|^{d+1}} \right\}, \\ b_k &= 2 \int_0^{1/2} b_k(r) \, \mathrm{d}r = \frac{1}{d 2^d \pi^2 |k|^{d+1}} + O \left( \frac{1}{|k|^{d+2}} \right) . \end{split} \]
The total sums of the coefficients are
\[ \begin{split} \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} b_k(r) &= \frac{\pi^{d/2}r^d}{\Gamma \left( \frac{d}{2}+1 \right)} \left( 1- \frac{\pi^{d/2}r^d}{\Gamma \left( \frac{d}{2}+1 \right)} \right), \\ \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} b_k &= \frac{\pi^{d/2}}{(d+1)2^d \Gamma \left( \frac{d}{2}+1 \right)} - \frac{\pi^d}{(2d+1)2^{2d} \Gamma \left( \frac{d}{2} +1 \right)^2} . \end{split} \]

Let $V(r)=\mathrm{Vol}(B(0,r))=(\pi^{d/2}/\Gamma (d/2+1))r^d$. Observe that for all $y \in \mathbb{R}^d / \mathbb{Z}^d$, $k \in \mathbb{Z}^d$ and $0<r \le 1/2$, we have
\begin{equation}\label{ballfouriertransform}
\begin{split} \int_{\mathbb{R}^d / \mathbb{Z}^d} &\left( \mathds{1}_{B(x,r)}(y) - V(r) \right) e^{- 2 \pi i \langle k, x \rangle} \, \mathrm{d}\mathrm{Vol}(x) \\ &= \int_{\mathbb{R}^d / \mathbb{Z}^d} \left( \mathds{1}_{B(0,r)}(x-y) - V(r) \right) e^{- 2 \pi i \langle k, x \rangle} \, \mathrm{d}\mathrm{Vol}(x) \\ &= \left\{ \begin{array}{ll} 0 & \textrm{if } k=0, \\ e^{- 2 \pi i \langle k, y \rangle} r^{d/2} |k|^{-d/2} J_{d/2}(2 \pi r |k|) & \textrm{if } k \neq 0. \end{array} \right. \end{split}
\end{equation}
Therefore the Fourier coefficients of the function $\mu (B(x,r)) - \mathrm{Vol}(B(x,r))$ are
\[ \begin{split} \int_{\mathbb{R}^d / \mathbb{Z}^d} &\left( \mu (B(x,r)) - \mathrm{Vol}(B(x,r)) \right) e^{-2 \pi i \langle k, x \rangle} \, \mathrm{d}\mathrm{Vol}(x) \\ &= \int_{\mathbb{R}^d / \mathbb{Z}^d} \int_{\mathbb{R}^d / \mathbb{Z}^d} \left( \mathds{1}_{B(x,r)}(y) - V(r) \right) e^{- 2 \pi i \langle k, x \rangle} \, \mathrm{d}\mathrm{Vol}(x) \, \mathrm{d}\mu (y) \\ &= \left\{ \begin{array}{ll} 0 & \textrm{if } k=0, \\ \widehat{\mu}(k) r^{d/2} |k|^{-d/2} J_{d/2}(2 \pi r |k|) & \textrm{if } k \neq 0. \end{array} \right. \end{split} \]
An application of the Parseval formula thus yields
\[ D_2(\mu, \mathrm{Vol},r)^2 = \int_{\mathbb{R}^d / \mathbb{Z}^d} \left( \mu (B(x,r)) - \mathrm{Vol}(B(x,r)) \right)^2 \, \mathrm{d}\mathrm{Vol}(x) = \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} b_k(r) |\widehat{\mu}(k)|^2 \]
with coefficients $b_k(r) = r^d |k|^{-d} J_{d/2}(2 \pi r |k|)^2$, as claimed.

The asymptotic bounds $J_{d/2}(t) \ll t^{d/2}$ as $t \to 0^+$ and $J_{d/2}(t) \ll t^{-1/2}$ as $t \to \infty$ \cite[p.\ 209 and 218]{AAR} imply $J_{d/2}(2 \pi r |k|)^2 \ll \min \{ r^d |k|^d , r^{-1} |k|^{-1} \}$ and the desired upper estimate for $b_k(r)$ follows. Setting $y=0$ in \eqref{ballfouriertransform} and applying the Parseval formula leads to the total sum of the coefficients
\[ \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} b_k(r) = \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{r^{d}}{|k|^d} J_{d/2}(2 \pi r |k|)^2 = \int_{\mathbb{R}^d / \mathbb{Z}^d} \left( \mathds{1}_{B(0,r)}(x) - V(r) \right)^2 \, \mathrm{d}\mathrm{Vol}(x) = V(r)(1-V(r)). \]

A further averaging over $0<r \le 1/2$ yields $D_2(\mu, \mathrm{Vol})=(\sum_{k \neq 0} b_k |\widehat{\mu}(k)|^2)^{1/2}$ with coefficients
\[ b_k = 2 \int_0^{1/2} b_k(r) \, \mathrm{d}r = 2 \int_0^{1/2} \frac{r^d}{|k|^d} J_{d/2}(2 \pi r |k|)^2 \, \mathrm{d}r , \]
as claimed. In particular, the total sum of the coefficients $b_k$ is
\[ \begin{split} \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} b_k &= 2 \int_0^{1/2} \frac{\pi^{d/2}r^d}{\Gamma \left( \frac{d}{2}+1 \right)} \left( 1- \frac{\pi^{d/2}r^d}{\Gamma \left( \frac{d}{2}+1 \right)} \right) \, \mathrm{d}r \\ &=\frac{\pi^{d/2}}{(d+1)2^d \Gamma \left( \frac{d}{2}+1 \right)} - \frac{\pi^d}{(2d+1)2^{2d} \Gamma \left( \frac{d}{2} +1 \right)^2}. \end{split} \]
Finally, the asymptotics of the Bessel function \cite[p.\ 209]{AAR}
\[ J_{d/2}(t)= \sqrt{\frac{2}{\pi t}} \left( \cos \left( t-\frac{(d+1)\pi}{4} \right) + O \left( \frac{1}{t} \right) \right) \quad \textrm{as } t \to \infty \]
shows that
\[ \begin{split} b_k &= 2 \int_0^{1/2} \frac{r^d}{|k|^d} \cdot \frac{1}{\pi^2 r |k|} \cos^2 \left( 2 \pi r |k| - \frac{(d+1)\pi}{4} \right) \, \mathrm{d}r + O \left( \int_0^{1/2} \frac{r^d}{|k|^d} \cdot \frac{1}{r^2 |k|^2} \, \mathrm{d} r \right) \\ &= \frac{1}{d 2^d \pi^2 |k|^{d+1}} + O \left( \frac{1}{|k|^{d+2}} \right) . \end{split} \]

Let us first prove the theorem for $M=\mathbb{R}^d / \mathbb{Z}^d$. Lemma \ref{D2torus_lemma} expresses $D_2(\mu, \mathrm{Vol})$ in terms of the Fourier transform as
\[ D_2 (\mu, \mathrm{Vol}) \asymp \bigg( \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{|\widehat{\mu}(k)|^2}{|k|^{d+1}} \bigg)^{1/2} . \]
An application of Theorem \ref{torussmoothing_theorem1} with $p=2$ shows that for any integer $H>2$,
\[ \begin{split} W_2 (\mu, \mathrm{Vol}) &\ll \frac{1}{H} + \bigg( \sum_{\substack{k \in [-H,H]^d \\ k \neq 0}} \frac{|\widehat{\mu}(k)|^2}{|k|^2} \bigg)^{1/2} \ll \frac{1}{H} + \bigg( \sum_{\substack{k \in [-H,H]^d \\ k \neq 0}} \frac{|\widehat{\mu}(k)|^2}{|k|^{d+1}} H^{d-1} \bigg)^{1/2} \\ &\ll \frac{1}{H} + D_2 (\mu, \mathrm{Vol}) H^{(d-1)/2} . \end{split} \]
The optimal choice $H \asymp D_2(\mu, \mathrm{Vol})^{-2/(d+1)}$ leads to $W_2(\mu, \mathrm{Vol}) \ll D_2(\mu, \mathrm{Vol})^{2/(d+1)}$, as claimed.

We now prove the theorem for $M=\mathbb{S}^d$. Since $V(r)=\mathrm{Vol}(B(x,r))$ does not depend on $x$, we have
\[ \begin{split} D_2(\mu, \mathrm{Vol})^2 &= \frac{1}{2} \int_0^{\pi} \int_{\mathbb{S}^d} |\mu (B(x,r)) - V(r)|^2 \sin r \, \mathrm{d} \mathrm{Vol}(x) \, \mathrm{d} r \\ &= \frac{1}{2} \int_0^{\pi} \int_{\mathbb{S}^d} \left| \int_{\mathbb{S}^d} \left( \mathds{1}_{B(x,r)}(y) - V(r) \right) \mathrm{d}\mathrm{\mu}(y) \right|^2 \sin r \, \mathrm{d}\mathrm{Vol}(x) \, \mathrm{d}r \\ &= \frac{1}{2} \int_{\mathbb{S}^d} \int_{\mathbb{S}^d} \int_0^{\pi} \int_{\mathbb{S}^d} \left( \mathds{1}_{B(y_1, r)}(x) - V(r) \right) \left( \mathds{1}_{B(y_2, r)}(x) -V(r) \right)  \sin r \, \mathrm{d} \mathrm{Vol}(x) \, \mathrm{d} r \, \mathrm{d}\mu(y_1) \mathrm{d}\mu (y_2) \\ &= \frac{1}{2} \int_{\mathbb{S}^d} \int_{\mathbb{S}^d} \int_0^{\pi} \left( \mathrm{Vol}(B(y_1, r) \cap B(y_2, r)) - V(r)^2 \right) \sin r \, \mathrm{d} r \, \mathrm{d}\mu(y_1) \, \mathrm{d}\mu(y_2) . \end{split} \]
Let $P_{\ell}^{(\alpha, \beta)}(t)$ denote the Jacobi polynomials. Skriganov \cite[Theorem 4.1 (ii)]{SK} showed the expansion
\[ \int_0^{\pi} \left( \mathrm{Vol}(B(y_1, r) \cap B(y_2, r)) - V(r)^2 \right) \sin r \, \mathrm{d}r = \sum_{\ell=1}^{\infty} b_{\ell} P_{\ell}^{(d/2-1, d/2-1)}(\langle y_1, y_2 \rangle) \]
with positive real coefficients
\[ b_{\ell} = \frac{\Gamma (d) (2\ell+d-1) \Gamma (\ell)^2 \Gamma (\ell+d-1)}{2^{2d} \Gamma \left( \frac{d}{2} \right) \Gamma \left( \ell + \frac{d}{2} \right)^3} \int_0^{\pi} (\sin r)^{2d+1} P_{\ell-1}^{(d/2, d/2)} (\cos r)^2 \, \mathrm{d}r \asymp \frac{1}{\ell^{d/2+1}} . \]
The order of magnitude of $b_{\ell}$ follows from the asymptotic formula of Darboux for Jacobi polynomials, see Szeg\H{o} \cite[p.\ 173 and p.\ 196]{SZ}. In particular,
\begin{equation}\label{D2expansion}
D_2 (\mu, \mathrm{Vol})^2 \asymp \sum_{\ell=1}^{\infty} \frac{1}{\ell^{d/2+1}} \int_{\mathbb{S}^d} \int_{\mathbb{S}^d} P_{\ell}^{(d/2-1, d/2-1)} (\langle y_1, y_2 \rangle) \, \mathrm{d}\mu(y_1) \, \mathrm{d}\mu (y_2) .
\end{equation}

The addition formula for spherical harmonics
\[ \sum_{m=1}^{d_{\ell}} Y_{\ell}^m (y_1) \overline{Y_{\ell}^m(y_2)} = \frac{(2\ell+d-1)\Gamma \left( \frac{d}{2} \right) \Gamma (\ell+d-1)}{\Gamma(d) \Gamma \left( \ell + \frac{d}{2} \right)} P_{\ell}^{(d/2-1, d/2-1)} (\langle y_1, y_2 \rangle), \qquad y_1, y_2 \in \mathbb{S}^d, \]
shows that
\[ \begin{split} \sum_{m=1}^{d_{\ell}} |\widehat{\mu}(\ell, m)|^2 &= \int_{\mathbb{S}^d} \int_{\mathbb{S}^d} \sum_{m=1}^{d_{\ell}} Y_{\ell}^m (y_1) \overline{Y_{\ell}^m (y_2)} \, \mathrm{d}\mu(y_1) \, \mathrm{d} \mu(y_2) \\ &= \frac{(2\ell+d-1)\Gamma \left( \frac{d}{2} \right) \Gamma (\ell+d-1)}{\Gamma(d) \Gamma \left( \ell + \frac{d}{2} \right)} \int_{\mathbb{S}^d} \int_{\mathbb{S}^d} P_{\ell}^{(d/2-1, d/2-1)}(\langle y_1, y_2 \rangle) \, \mathrm{d}\mu(y_1) \, \mathrm{d}\mu(y_2) \\ &\asymp \ell^{d/2} \int_{\mathbb{S}^d} \int_{\mathbb{S}^d} P_{\ell}^{(d/2-1, d/2-1)}(\langle y_1, y_2 \rangle) \, \mathrm{d}\mu(y_1) \, \mathrm{d}\mu(y_2) . \end{split} \]
An application of Theorem \ref{spheresmoothing_theorem} with $p=2$, \eqref{D2expansion} and the previous formula thus show that for any positive integer $L$,
\[ \begin{split} W_2 (\mu, \mathrm{Vol}) &\ll \frac{1}{L} + \bigg( \sum_{\ell=1}^L \frac{1}{\ell^2} \sum_{m=1}^{d_{\ell}} |\widehat{\mu}(\ell,m)|^2 \bigg)^{1/2} \\ &\ll \frac{1}{L} + \left( \sum_{\ell=1}^L \frac{L^{d-1}}{\ell^{d/2+1}} \int_{\mathbb{S}^d} \int_{\mathbb{S}^d} P_{\ell}^{(d/2-1, d/2-1)}(\langle y_1, y_2 \rangle) \, \mathrm{d}\mu(y_1) \, \mathrm{d}\mu(y_2) \right)^{1/2} \\ &\ll \frac{1}{L} + D_2(\mu, \mathrm{Vol}) L^{(d-1)/2} . \end{split} \]
The optimal choice $L \asymp D_2(\mu, \mathrm{Vol})^{-2/(d+1)}$ leads to $W_2 (\mu, \mathrm{Vol}) \ll D_2 (\mu, \mathrm{Vol})^{2/(d+1)}$, as claimed.

\label{cassels_lemma} Let $\alpha_1, \alpha_2, \ldots, \alpha_m \in \mathbb{R}^d$, let $N_1, N_2, \ldots, N_m \ge 2$ be integers and let $N=N_1 N_2 \cdots N_m$. Under the assumption \eqref{diophantinelattice}, every axis-parallel cube in $\mathbb{R}^d / \mathbb{Z}^d$ of sidelength
\[ \frac{((d+m)!)^2}{2^{d+m-1} \psi^{-1} \left( 2^{(d+m-2)m} ((d+m)!)^{-2m} I N \right)} \]
contains at least one point from the set
\begin{equation}\label{latticeset}
\left\{ n_1 \alpha_1 + n_2 \alpha_2 + \cdots + n_m \alpha_m + \mathbb{Z}^d : n_1 \in [1, N_1], \,\, n_2 \in [1, N_2], \,\, \ldots, \,\, n_m \in [1,N_m] \right\} .
\end{equation}

Let $\beta \in \mathbb{R}^d$ be arbitrary and let
\[ C = \frac{((d+m)!)^2}{2^{d+m} \psi^{-1} \left( 2^{(d+m-2)m} ((d+m)!)^{-2m} I N \right)} \]
be half the desired sidelength. Let $\alpha_j=(\alpha_{j1}, \alpha_{j2}, \ldots, \alpha_{jd})$, $1 \le j \le m$. Consider the linear forms
\[ \begin{split} f_i (n_1, \ldots, n_m, y_1, \ldots, y_d) &= \left\{ \begin{array}{ll} C^{-1} \left( \sum_{j=1}^m n_j \alpha_{ji} + y_i \right) & \textrm{if } 1 \le i \le d, \\ \left( \frac{N_{i-d}-1}{2} \right)^{-1} n_{i-d} & \textrm{if } d< i \le d+m, \end{array} \right. \\ g_i (v_1, \ldots, v_m, k_1, \ldots, k_d) &= \left\{ \begin{array}{ll} C k_i & \textrm{if } 1 \le i \le d, \\ \left( \frac{N_{i-d}-1}{2} \right) \left( v_{i-d} - \langle k, \alpha_{i-d} \rangle \right) & \textrm{if } d< i \le d+m, \end{array} \right. \end{split} \]
and note that $\sum_{i=1}^{d+m} f_i g_i = \sum_{i=1}^m n_i v_i + \sum_{i=1}^d y_i k_i$. We claim that for all integers $v_1, \ldots, v_m$, $k_1, \ldots, k_d$, we have
\begin{equation}\label{cassels}
\left\| \sum_{i=1}^d k_i \beta_i \right\|_{\mathbb{R}/\mathbb{Z}} \le \frac{2^{d+m-1}}{((d+m)!)^2} \max \left\{ C \| k \|_{\infty}, \max_{1 \le j \le m} \frac{N_j-1}{2} \left| v_j - \langle k, \alpha_j \rangle \right| \right\} .
\end{equation}
Indeed, if $k=0$, then the left-hand side vanishes, consequently \eqref{cassels} holds. Otherwise the left-hand side is at most $1/2$, and by assumption \eqref{diophantinelattice},
\[ \max_{1 \le j \le m} \frac{N_j-1}{2} \left| v_j - \langle k, \alpha_j \rangle \right| \ge \left( \prod_{1 \le j \le m} \frac{N_j}{4} \| \langle k, \alpha_j \rangle \|_{\mathbb{R}/\mathbb{Z}} \right)^{1/m} \ge \frac{I^{1/m} N^{1/m}}{4 \psi (\| k \|_{\infty})^{1/m}} . \]
The choice of $C$ ensures that for any $k \in \mathbb{Z}^d$, $k \neq 0$, either
\[ \frac{1}{2} \le \frac{2^{d+m-1}}{((d+m)!)^2} C \| k \|_{\infty}  \]
or
\[ \frac{1}{2} \le \frac{2^{d+m-1}}{((d+m)!)^2} \cdot \frac{I^{1/m} N^{1/m}}{4 \psi (\| k \|_{\infty})^{1/m}} \]
holds, and \eqref{cassels} follows. An application of \cite[Theorem XVI, p.\ 97]{CA} then shows that there exist integers $n_1, \ldots, n_m, y_1, \ldots, y_d$ such that
\[ \begin{aligned} \bigg| \sum_{j=1}^m n_j \alpha_{ji} + y_i - \beta_i \bigg| &\le C, & &1 \le i \le d, \\ |n_j| &\le \frac{N_j-1}{2}, & &1 \le j \le m . \end{aligned} \]
In particular, the point $\sum_{j=1}^m n_j \alpha_j + \mathbb{Z}^d$ lies in the axis-parallel cube centered at $\beta+\mathbb{Z}^d$ with sidelength $2C$. Since a translated copy of the set
\[ \left\{ n_1 \alpha_1 + n_2 \alpha_2 + \cdots + n_m \alpha_m + \mathbb{Z}^d : |n_1| \le \frac{N_1-1}{2}, \,\, |n_2| \le \frac{N_2-1}{2}, \,\, \ldots, |n_m| \le \frac{N_m-1}{2} \right\} \]
is contained in the set \eqref{latticeset}, the latter also intersects all axis-parallel cubes of sidelength $2C$, as claimed.

Throughout the proof, implied constants depend only on $I$, $c$, $d$ and $m$. The claim for $W_{\infty}$ follows from substituting $p=\log N$ in the claim for $W_p$ and using Lemma \ref{WqWplemma}.

Let $2 \le p < \infty$ and $1 < q \le 2$ be such that $1/p+1/q=1$, and let $b \ge 0$, $r>0$ and $t>0$ to be chosen. The Fourier transform of
\[ \mu_N = \frac{1}{N} \sum_{n_1=1}^{N_1} \sum_{n_2=1}^{N_2} \cdots \sum_{n_m=1}^{N_m} \delta_{n_1 \alpha_1 + n_2 \alpha_2 + \cdots + n_m \alpha_m + \mathbb{Z}^d} \]
can be estimated as
\[ |\widehat{\mu_N}(k)| = \left| \frac{1}{N} \prod_{j=1}^m \sum_{n_j=1}^{N_j} e^{- 2 \pi i n_j \langle k, \alpha_j \rangle}  \right| \le \frac{1}{N} \prod_{j=1}^m \frac{2}{|1-e^{-2 \pi i \langle k, \alpha_j \rangle}|} \ll \frac{1}{N \prod_{j=1}^m \| \langle k, \alpha_j \rangle \|_{\mathbb{R}/\mathbb{Z}}}. \]
An application of Theorem \ref{torussmoothing_theorem2} thus yields
\begin{equation}\label{Wpkroneckerestimate}
W_p (\mu_N, \mathrm{Vol}) \ll p^{1/2} t^{1/2} + \frac{p(1-\delta^{1/p})}{(1-\delta)N} \bigg( \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{e^{-4 \pi^2 |k|^2 q t}}{|k|^q \prod_{j=1}^m \| \langle k, \alpha_j \rangle \|^q_{\mathbb{R}/\mathbb{Z}}} \bigg)^{1/q} ,
\end{equation}
where
\[ \delta = \min \left\{ b \frac{e^{-r^2/(4t)}}{(4 \pi t)^{d/2}}, 1 \right\} . \]

Lemma \ref{cassels_lemma} states that any axis-parallel cube in $\mathbb{R}^d / \mathbb{Z}^d$ of sidelength $\asymp 1/\psi^{-1}(N)$ contains at least one point $n_1 \alpha_1+n_2 \alpha_2 + \cdots + n_m \alpha_m+\mathbb{Z}^d$ with $1 \le n_j \le N_j$, $1 \le j \le m$. Any ball $B(x,r) \subseteq \mathbb{R}^d / \mathbb{Z}^d$ contains $\gg (r \psi^{-1}(N))^d$ disjoint axis-parallel cubes of sidelength $\asymp 1/\psi^{-1}(N)$, therefore $\mu_N(B(x,r)) \gg N^{-1} (r \psi^{-1}(N))^d$. We can thus choose $b \asymp N^{-1} (r \psi^{-1}(N))^d$, hence
\[ \delta \gg \frac{1}{N} \left( r \psi^{-1}(N) \right)^d \frac{e^{-r^2/(4t)}}{(4 \pi t)^{d/2}} .  \]
The optimal choice is $r=t^{1/2}$, in which case $\delta \gg (\psi^{-1}(N))^d/N$. Observe
\[ \frac{p(1-\delta^{1/p})}{1-\delta} \le p, \]
and using $e^x \ge 1+x$, $x \in \mathbb{R}$, also
\[ \frac{p(1-\delta^{1/p})}{1-\delta} = \frac{p (1-e^{-(\log (1/\delta))/p})}{1-\delta} \le \frac{\log (1/\delta)}{1-\delta} . \]
Therefore
\begin{equation}\label{deltaestimate}
\frac{p (1-\delta^{1/p})}{1-\delta} \le \min \left\{ p, \frac{\log (1/\delta)}{1-\delta} \right\} \ll \min \left\{ p, \log \left(\frac{N}{(\psi^{-1}(N))^d} +2 \right) \right\} .
\end{equation}

In order to estimate the sum over $k$ in \eqref{Wpkroneckerestimate}, we use a lattice point counting estimate of Fregoli \cite[Proposition 1.2]{FR}, which states that under assumption \eqref{diophantinelattice}, for any $\varepsilon>0$ and $Q \ge 1$, we have
\[ \bigg| \bigg\{ k \in \mathbb{Z}^d : \prod_{j=1}^m \left\| \langle k, \alpha_j \rangle \right\|_{\mathbb{R} / \mathbb{Z}} < \varepsilon, \,\, 0<\| k \|_{\infty} \le Q \bigg\} \bigg| \ll \left( (\varepsilon \psi(Q))^{\frac{d+m-1}{d+m}} + \varepsilon Q^d \right) \left( \log \left( \frac{1}{\varepsilon} +2 \right) \right)^{m-1} . \]
In particular,
\[ \begin{split} \sum_{\substack{k \in \mathbb{Z}^d \\ 0<\| k \|_{\infty} \le Q}} &\frac{1}{\prod_{j=1}^m \| \langle k, \alpha_j \rangle \|_{\mathbb{R}/\mathbb{Z}}^q} \\ &\ll \sum_{1 \le n \le \frac{\log (\psi(Q)/I)}{\log 2}} 2^{qn} \bigg| \bigg\{ k \in \mathbb{Z}^d : \prod_{j=1}^m \left\| \langle k, \alpha_j \rangle \right\|_{\mathbb{R} / \mathbb{Z}} \in (2^{-n-1}, 2^{-n}], \,\, 0< \| k \|_{\infty} \le Q \bigg\} \bigg| \\ &\ll \sum_{1 \le n \le \frac{\log (\psi(Q)/I)}{\log 2}} \left( 2^{\left( q-\frac{d+m-1}{d+m} \right) n} \psi(Q)^{\frac{d+m-1}{d+m}} + 2^{(q-1)n} Q^d \right) n^{m-1} \\ &\ll \left( \psi(Q)^q + \frac{Q^d \psi(Q)^{q-1}}{2^{q-1}-1} \right) \left( \log (\psi(Q)+2) \right)^{m-1} . \end{split} \]
Here $1/(2^{q-1}-1) \asymp 1/(q-1) \asymp p$. For any integer $\ell \ge 0$, we thus have
\[ \begin{split} \sum_{\substack{k \in \mathbb{Z}^d \\ 2^{\ell} \le |k| < 2^{\ell+1}}} &\frac{e^{-4 \pi^2 |k|^2 q t}}{|k|^q \prod_{j=1}^m \| \langle k, \alpha_j \rangle \|^q_{\mathbb{R}/\mathbb{Z}}} \\ &\le \frac{e^{-4 \pi^2 2^{2\ell}qt}}{2^{q\ell}} \sum_{\substack{k \in \mathbb{Z}^d \\ 0< \|k \|_{\infty} < 2^{\ell+1}}} \frac{1}{\prod_{j=1}^m \| \langle k, \alpha \rangle \|^q_{\mathbb{R}/\mathbb{Z}}} \\ &\ll e^{-4 \pi^2 2^{2\ell}qt} \left( \frac{\psi (2^{\ell+1})^q}{2^{q \ell}}+p 2^{(d-q) \ell} \psi (2^{\ell+1})^{q-1} \right) (\log (\psi (2^{\ell+1})+2))^{m-1} . \end{split} \]
Summing the previous estimate over all $\ell \ge 0$ leads to
\begin{equation}\label{sumoverell1}
\begin{split} \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} &\frac{e^{-4 \pi^2 |k|^2 q t}}{|k|^q \prod_{j=1}^m \| \langle k, \alpha_j \rangle \|^q_{\mathbb{R}/\mathbb{Z}}} \\ &\ll \sum_{\ell=0}^{\infty} e^{-4 \pi^2 2^{2\ell}qt} \left( \frac{\psi (2^{\ell+1})^q}{2^{q \ell}}+p 2^{(d-q) \ell} \psi (2^{\ell+1})^{q-1} \right) (\log (\psi (2^{\ell+1})+2))^{m-1}. \end{split}
\end{equation}
The assumption that $x^{-d}\psi(x)$ is nondecreasing shows that the terms with $2^{\ell+1} \le t^{-1/2}$ satisfy
\[ \psi(2^{\ell+1}) \le 2^{d(\ell+1)} t^{d/2} \psi (t^{-1/2}), \]
hence
\begin{equation}\label{sumoverell2}
\begin{split} \sum_{2^{\ell+1} \le t^{-1/2}} &e^{-4 \pi^2 2^{2\ell}qt} \left( \frac{\psi (2^{\ell+1})^q}{2^{q \ell}}+p 2^{(d-q) \ell} \psi (2^{\ell+1})^{q-1} \right) (\log (\psi (2^{\ell+1})+2))^{m-1} \\ &\ll \left( t^{dq/2} \psi (t^{-1/2})^q + p t^{d(q-1)/2} \psi(t^{-1/2})^{q-1} \right) (\log (\psi (t^{-1/2})+2))^{m-1} \sum_{2^{\ell+1} \le t^{-1/2}} 2^{(d-1)q \ell} \\ &\ll \left( t^{q/2} \psi (t^{-1/2})^q + p t^{(q-d)/2} \psi (t^{-1/2})^{q-1} \right) (\log (\psi (t^{-1/2})+2))^{m-1} . \end{split}
\end{equation}
The assumption $\psi(xy) \le c e^{x^2} \psi (y)$ shows that the terms with $2^{\ell+1} > t^{-1/2}$ satisfy
\[ \psi (2^{\ell+1}) \le c e^{2^{2\ell+2}t} \psi (t^{-1/2}), \]
hence
\begin{equation}\label{sumoverell3}
\begin{split} \sum_{2^{\ell+1}>t^{-1/2}} & e^{-4 \pi^2 2^{2\ell}qt} \left( \frac{\psi (2^{\ell+1})^q}{2^{q \ell}}+p 2^{(d-q) \ell} \psi (2^{\ell+1})^{q-1} \right) (\log (\psi (2^{\ell+1})+2))^{m-1} \\ &\ll  \sum_{2^{\ell+1}>t^{-1/2}} e^{-4 (\pi^2-1) 2^{2\ell}qt} \left( \frac{\psi(t^{-1/2})^q}{2^{q \ell}} + p 2^{(d-q)\ell} \psi(t^{-1/2})^{q-1} \right) \\ &\hspace{20mm} \times \left( \log (\psi (t^{-1/2})+2) + 2^{2\ell}t \right)^{m-1} \\ &\ll \left( t^{q/2} \psi (t^{-1/2})^q + p t^{(q-d)/2} \psi (t^{-1/2})^{q-1} \right) (\log (\psi (t^{-1/2})+2))^{m-1} . \end{split}
\end{equation}
The estimates \eqref{sumoverell1}--\eqref{sumoverell3} yield
\[ \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{e^{-4 \pi^2 |k|^2 q t}}{|k|^q \prod_{j=1}^m \| \langle k, \alpha_j \rangle \|^q_{\mathbb{R}/\mathbb{Z}}} \ll t^{q/2} \psi (t^{-1/2})^q \left( 1+p \frac{t^{-d/2}}{\psi (t^{-1/2})} \right) (\log (\psi (t^{-1/2})+2))^{m-1} . \]

Substituting the previous formula and \eqref{deltaestimate} in \eqref{Wpkroneckerestimate} leads to
\[ \begin{split} W_p (\mu_N, \mathrm{Vol}) &\ll p^{1/2} t^{1/2} + \min \left\{ p, \log \left( \frac{N}{(\psi^{-1}(N))^d}+2 \right) \right\} \\ &\hspace{24mm} \times \frac{t^{1/2} \psi (t^{-1/2})}{N} \left( 1+p \frac{t^{-d/2}}{\psi(t^{-1/2})} \right) (\log (\psi (t^{-1/2})+2))^{(m-1)/q} \\ &= p^{1/2} t^{1/2} + \min \left\{ p, \log \left( \frac{N}{(\psi^{-1}(N))^d}+2 \right) \right\} \frac{t^{1/2}}{N} F_p (t^{-1/2}) \end{split} \]
in terms of the function
\[ F_p(x) = \left( \psi(x)+px^d \right) \left( \log (\psi(x)+2) \right)^{(m-1)(p-1)/p} . \]
The optimal choice for $t$ is
\[ t \asymp \left( F_p^{-1} \left( \frac{p^{1/2} N}{\min \left\{ p, \log \left( \frac{N}{(\psi^{-1}(N))^d}+2 \right) \right\}} \right) \right)^{-2}, \]
which proves the desired estimate for $W_p (\mu_N, \mathrm{Vol})$.

The Diophantine condition \eqref{diophantinelattice} depends only on the fractional parts of the coordinates of the vectors $\alpha_1, \alpha_2, \ldots, \alpha_m$. In particular, it will be enough to prove that $\mathrm{Vol}$-a.e.\ $(\alpha_1, \alpha_2, \ldots, \alpha_m) \in \mathbb{R}^{dm}/\mathbb{Z}^{dm}$ satisfies \eqref{diophantinelattice}.

Let $k \in \mathbb{Z}^d$, $k \neq 0$. Observe that the map $\mathbb{R}^{dm} / \mathbb{Z}^{dm} \to \mathbb{R}^m / \mathbb{Z}^m$, $(\alpha_1, \alpha_2, \ldots, \alpha_m) \mapsto (\langle k, \alpha_1 \rangle, \langle k, \alpha_2 \rangle, \ldots, \langle k, \alpha_m \rangle)$ is a continuous surjective homomorphism. In particular, the map preserves the normalized Haar measure $\mathrm{Vol}$, hence for any $t>0$,
\[ \begin{split} \mathrm{Vol} \bigg( \bigg\{ (\alpha_1, \alpha_2, \ldots, \alpha_m) &\in \mathbb{R}^{dm} / \mathbb{Z}^{dm} : \prod_{j=1}^m \left\| \langle k, \alpha_j \rangle \right\|_{\mathbb{R}/\mathbb{Z}} \le t \bigg\} \bigg) \\ &= \mathrm{Vol} \bigg( \bigg\{ (x_1, x_2, \ldots, x_m) \in \mathbb{R}^m / \mathbb{Z}^m : \prod_{j=1}^m \| x_j \|_{\mathbb{R}/\mathbb{Z}} \le t \bigg\} \bigg) . \end{split} \]
Identifying $\mathbb{R}^m/\mathbb{Z}^m$ by the unit cube $(-1/2,1/2]^m$, one readily checks by induction on $m$ that
\[ \mathrm{Vol} \bigg( \bigg\{ (x_1, x_2, \ldots, x_m) \in \left( -\frac{1}{2}, \frac{1}{2} \right]^m : \prod_{j=1}^m |x_j| \le t \bigg\} \bigg) \ll t \left( \log \left( \frac{1}{t} +2 \right) \right)^{m-1} \]
with an implied constant that depends only on $m$. An application of the previous two formulas with $t=1/\psi (\| k \|_{\infty})$ yields
\[ \mathrm{Vol} \bigg( \bigg\{ (\alpha_1, \alpha_2, \ldots, \alpha_m) \in \mathbb{R}^{dm} / \mathbb{Z}^{dm} : \psi (\| k \|_{\infty}) \prod_{j=1}^m \left\| \langle k, \alpha_j \rangle \right\|_{\mathbb{R}/\mathbb{Z}} \le 1 \bigg\} \bigg) \ll \frac{(\log (\psi (\| k \|_{\infty})+2))^{m-1}}{\psi(\| k \|_{\infty})} . \]
The number of lattice points $k \in \mathbb{Z}^d$ such that $\| k \|_{\infty}=n$ is $\ll n^{d-1}$, therefore the sum of volumes converges by assumption:
\[ \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{( \log (\psi(\| k \|_{\infty})+2))^{m-1}}{\psi(\| k \|_{\infty})} \ll \sum_{n=1}^{\infty} \frac{(\log (\psi(n)+2))^{m-1}}{\psi(n)} n^{d-1} < \infty . \]
The Borel--Cantelli lemma thus shows that for $\mathrm{Vol}$-a.e.\ $(\alpha_1, \alpha_2, \ldots, \alpha_m) \in \mathbb{R}^{dm} / \mathbb{Z}^{dm}$, the inequality
\[ \psi (\| k \|_{\infty}) \prod_{j=1}^m \| \langle k, \alpha_j \rangle \|_{\mathbb{R}/\mathbb{Z}} > 1 \]
holds for all but finitely many $k \in \mathbb{Z}^d$. In particular, \eqref{diophantinelattice} is satisfied.

Let $q$ be a prime and let $a \in (\mathbb{Z}/q\mathbb{Z})^{\times}$. The Fourier transform of the empirical measure $\mu_{a,q,d}$ is
\[ \widehat{\mu_{a,q,d}}(k) = \frac{1}{(q-1)^{d-1}} \sum_{\substack{x_1, x_2, \ldots, x_d \in (\mathbb{Z}/q\mathbb{Z})^{\times} \\ x_1 x_2 \cdots x_d \equiv a \pmod{q}}} e^{-2 \pi i (k_1 x_1 + k_2 x_2 + \cdots + k_d x_d)/q} , \]
which is clearly invariant under permutations of the coordinates of $k=(k_1, k_2, \ldots, k_d) \in \mathbb{Z}^d$.

If none of the integers $k_1, k_2, \ldots, k_d$ are divisible by $q$, then a reindexing of the sum leads to $\widehat{\mu_{a,q,d}}(k) = (q-1)^{-(d-1)} K_d((-1)^d a k_1 k_2 \cdots k_d;q)$, where
\[ K_d(n;q) = \sum_{\substack{x_1, x_2, \ldots, x_d \in (\mathbb{Z}/q\mathbb{Z})^{\times} \\ x_1 x_2 \cdots x_d \equiv n \pmod{q}}} e^{2 \pi i (x_1+x_2+\cdots+x_d)/q} \]
is the generalized Kloosterman sum. The Weil bound due to Deligne \cite[Section 6.7]{DE} states that $|K_d(n;q)| \le d q^{(d-1)/2}$ for all $n \in (\mathbb{Z}/q\mathbb{Z})^{\times}$, hence $|\widehat{\mu_{a,q,d}}(k)| \ll q^{-(d-1)/2}$.

If $j \ge 1$ many of the integers $k_1, k_2, \ldots, k_d$ are divisible by $q$, then
\[ \begin{split} \widehat{\mu_{a,q,d}}(k) &= \frac{1}{(q-1)^{d-1}} \sum_{x_{j+1}, x_{j+2}, \ldots, x_d \in (\mathbb{Z}/q\mathbb{Z})^{\times}} e^{-2 \pi i (k_{j+1} x_{j+1} + k_{j+2} x_{j+2} + \cdots + k_d x_d)/q} \sum_{\substack{x_1, x_2, \ldots, x_j \in (\mathbb{Z}/q \mathbb{Z})^{\times} \\ x_1 x_2 \cdots x_d \equiv a \pmod{q}}} 1 \\ &= \frac{1}{(q-1)^{d-j}} \left( \sum_{x_{j+1} \in (\mathbb{Z}/q\mathbb{Z})^{\times}} e^{-2 \pi i k_{j+1} x_{j+1}/q} \right) \cdots \left( \sum_{x_d \in (\mathbb{Z} /q \mathbb{Z})^{\times}} e^{-2 \pi i k_d x_d/q} \right) \\ &= \frac{(-1)^{d-j}}{(q-1)^{d-j}} . \end{split} \]

Let $1 \le H<q$ be an integer to be chosen. An application of Theorem \ref{torussmoothing_theorem1} with $p=2$ yields
\[ W_2 (\mu_{a,q,d}, \mathrm{Vol}) \ll \frac{1}{H} + \bigg( \sum_{\substack{k \in [-H,H]^d \\ k \neq 0}} \frac{|\widehat{\mu_{a,q,d}}(k)|^2}{|k|^2} \bigg)^{1/2} . \]
The contribution of all $k \in [-H,H]^d$ none of whose coordinates are zero is
\[ \sum_{k \in ([-H,H]\backslash \{ 0 \})^d} \frac{|\widehat{\mu_{a,q,d}}(k)|^2}{|k|^2} \ll \frac{1}{q^{d-1}} \sum_{\substack{k \in [-H,H]^d \\ k \neq 0}} \frac{1}{|k|^2} \ll \left\{ \begin{array}{ll} q^{-1} \log H & \textrm{if } d=2, \\ q^{-(d-1)} H^{d-2} & \textrm{if } d \ge 3. \end{array} \right. \]
The contribution of all $k \in [-H,H]^d$ with $1 \le j \le d-1$ many zero coordinates is
\[ \sum_{\substack{k \in [-H,H]^d \\ |\{ 1 \le i \le d \, : \, k_i=0 \}|=j}} \frac{|\widehat{\mu_{a,q,d}}(k)|^2}{|k|^2} \ll \frac{1}{q^{2(d-j)}} \sum_{\substack{\ell \in [-H,H]^{d-j} \\ \ell \neq 0}} \frac{1}{|\ell|^2} \ll \left\{ \begin{array}{ll} q^{-2(d-j)} H^{d-j-2} & \textrm{if } 1 \le j \le d-3, \\ q^{-2(d-j)} \log H & \textrm{if } j=d-2, \\ q^{-2(d-j)} & \textrm{if } j=d-1. \end{array} \right. \]
The assumption $H<q$ implies that the case $j=d-1$ yields the largest error term. Adding the previous two estimates leads to
\[ \sum_{\substack{k \in [-H,H]^d \\ k \neq 0}} \frac{|\widehat{\mu_{a,q,d}}(k)|^2}{|k|^2} \ll \left\{ \begin{array}{ll} q^{-1} \log H & \textrm{if } d=2, \\ q^{-(d-1)} H^{d-2} + q^{-2} & \textrm{if } d \ge 3, \end{array} \right. \]
consequently
\[ W_2 (\mu_{a,q,d}, \mathrm{Vol}) \ll \frac{1}{H} + \left\{ \begin{array}{ll} q^{-1/2} (\log H)^{1/2} & \textrm{if } d=2, \\ q^{-(d-1)/2} H^{(d-2)/2} + q^{-1} & \textrm{if } d \ge 3.  \end{array} \right. \]
The optimal choice $H \asymp q^{(d-1)/d}$ proves the desired estimate for the quadratic Wasserstein metric
\[ W_2 (\mu_{a,q,d}, \mathrm{Vol}) \ll \left\{ \begin{array}{ll} (\log q)^{1/2} q^{-1/2} & \textrm{if } d=2, \\ q^{-(d-1)/d} & \textrm{if } d \ge 3. \end{array} \right. \]
Note this matches the best possible rate \eqref{Wpbestrate} in dimension $d \ge 3$.

We use Lemma \ref{D2torus_lemma} in order to estimate $D_2(\mu_{a,q,d}, \mathrm{Vol})$. If all coordinates of $k \in \mathbb{Z}^d$ are divisible by $q$, then $|\widehat{\mu_{a,q,d}}(k)|=1$. The contribution of all such terms is
\[ \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0 \\ q \mid k_1, k_2, \ldots, k_d}} \frac{|\widehat{\mu_{a,q,d}}(k)|^2}{|k|^{d+1}} = \sum_{\substack{\ell \in \mathbb{Z}^d \\ \ell \neq 0}} \frac{1}{q^{d+1} |\ell|^{d+1}} \ll \frac{1}{q^{d+1}} . \]
If at least one coordinate of $k$ is not divisible by $q$, then $|\widehat{\mu_{a,q,d}}(k)| \ll \max \{ q^{-1}, q^{-(d-1)/2} \}$. The contribution of all such terms is
\[ \sum_{\substack{k \in \mathbb{Z}^d \\ q \nmid k_i \textrm{ for some } i}} \frac{|\widehat{\mu_{a,q,d}}(k)|^2}{|k|^{d+1}} \ll \max \{ q^{-2}, q^{-(d-1)} \} \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{1}{|k|^{d+1}} \ll \left\{ \begin{array}{ll} q^{-1} & \textrm{if } d=2, \\ q^{-2} & \textrm{if } d \ge 3. \end{array} \right. \]
This proves the upper bound
\[ D_2 (\mu_{a,q,d}, \mathrm{Vol}) \asymp \bigg( \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{|\widehat{\mu_{a,q,d}}|^2}{|k|^{d+1}} \bigg)^{1/2} \ll \left\{ \begin{array}{ll} q^{-1/2} & \textrm{if } d=2, \\ q^{-1} & \textrm{if } d \ge 3. \end{array} \right. \]
Since $|\widehat{\mu_{a,q,d}}(k)| = 1/(q-1)$ whenever $k$ has exactly one coordinate that is not divisible by $q$, we have
\[ \sum_{\substack{k \in \mathbb{Z}^d \\ k \neq 0}} \frac{|\widehat{\mu_{a,q,d}}(k)|^2}{|k|^{d+1}} \ge \frac{1}{(q-1)^2}\sum_{\substack{\ell \in \mathbb{Z} \\ q \nmid \ell}} \frac{1}{|\ell|^{d+1}} \gg \frac{1}{q^2} . \]
This proves the matching lower bound $D_2 (\mu_{a,q,d}, \mathrm{Vol}) \gg q^{-1}$ in dimension $d \ge 3$.

Let $q$ be a prime and let $a \in (\mathbb{Z}/q\mathbb{Z})^{\times}$. We first apply a smoothing procedure to $\mu_{a,q,2}$ using the heat kernel
\[ P_t (x) = \sum_{k \in \mathbb{Z}^2} e^{-4 \pi^2 |k|^2 t} e^{2 \pi i \langle k, x \rangle}, \qquad t>0, \,\, x \in \mathbb{R}^2 / \mathbb{Z}^2 . \]
The convolution $P_t * \mu_{a,q,2}$ is an absolutely continuous probability measure on $\mathbb{R}^2 / \mathbb{Z}^2$ and has density
\[ \frac{\mathrm{d}(P_t * \mu_{a,q,d})}{\mathrm{d}\mathrm{Vol}}(x) = \int_{\mathbb{R}^2 / \mathbb{Z}^2} P_t (x-y) \, \mathrm{d} \mu_{a,q,2}(y) = \sum_{k \in \mathbb{Z}^2} e^{-4 \pi^2 |k|^2 t} \widehat{\mu_{a,q,2}}(k) e^{2 \pi i \langle k, x \rangle} = 1+f(x) \]
with the smooth function
\[ f(x) = \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0}} e^{-4 \pi^2 |k|^2 t} \widehat{\mu_{a,q,2}}(k) e^{2 \pi i \langle k, x \rangle} . \]
The heat kernel is contractive with respect to the metric $W_2$, that is, $W_2 (P_t * \mu, P_t * \nu) \le W_2 (\mu, \nu)$ for any $\mu, \nu \in \mathcal{P}(\mathbb{R}^2 / \mathbb{Z}^2)$, see \cite{AST}. Since $P_t*\mathrm{Vol}=\mathrm{Vol}$, we have
\begin{equation}\label{W2heatkernellowerbound1}
W_2 (P_t * \mu_{a,q,2}, \mathrm{Vol}) \le W_2 (\mu_{a,q,2}, \mathrm{Vol}).
\end{equation}

As observed in the proof of Theorem \ref{modularhyperbola_theorem}, we have
\begin{equation}\label{muaq2fouriercoefficients}
\widehat{\mu_{a,q,2}}(k) = \left\{ \begin{array}{ll} 1 & \textrm{if } q \mid k_1, k_2, \\ -1/(q-1) & \textrm{if } q \textrm{ divides exactly one of } k_1, k_2, \\ K_2 (ak_1 k_2;q)/(q-1) & \textrm{if } q \nmid k_1, k_2,  \end{array} \right.
\end{equation}
where
\[ K_2 (n;q) = \sum_{x \in (\mathbb{Z}/q\mathbb{Z})^{\times}} e^{2 \pi i (nx+\overline{x})/q} \]
is the Kloosterman sum. The contribution of the terms $q \mid k_1, k_2$ in $f(x)$ is
\[ \Bigg| \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0 \\ q \mid k_1, k_2}} e^{-4 \pi^2 |k|^2 t} \widehat{\mu_{a,q,2}}(k) e^{2 \pi i \langle k, x \rangle} \Bigg| \le \sum_{\substack{\ell \in \mathbb{Z}^2 \\ \ell \neq 0}} e^{-4 \pi^2 q^2 |\ell|^2 t} \ll \frac{1}{q^2 t} . \]
Using the Weil bound $|K_2 (n;q)| \le 2 q^{1/2}$, $n \in (\mathbb{Z}/q\mathbb{Z})^{\times}$, the contribution of all other terms is
\[ \Bigg |\sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0 \\ q \nmid k_1 \textrm{ or } q \nmid k_2}} e^{-4 \pi^2 |k|^2 t} \widehat{\mu_{a,q,2}}(k) e^{2 \pi i \langle k, x \rangle} \Bigg| \ll \frac{1}{q^{1/2}} \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0}} e^{-4 \pi^2 |k|^2 t} \ll \frac{1}{q^{1/2} t} . \]
The previous two estimates show that $|f(x)| \ll q^{-1/2} t^{-1}$. Choosing, say, $t=q^{-1/4}$, we thus have $|f(x)| \ll q^{-1/4}$, and an application of Lemma \ref{ledoux_lemma} leads to
\begin{equation}\label{W2heatkernellowerbound2}
\left( \int_{\mathbb{R}^2 / \mathbb{Z}^2} f (\Delta^{-1} f) \, \mathrm{d} \mathrm{Vol} \right)^{1/2} \ll W_2 (P_t * \mu_{a,q,2}, \mathrm{Vol}) .
\end{equation}

Applying the inverse of the Laplace--Beltrami operator to $f(x)$ gives the absolutely convergent Fourier series
\[ (\Delta^{-1} f)(x) = \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0}} e^{-4 \pi^2 |k|^2 t} \frac{\widehat{\mu_{a,q,2}}(k)}{4 \pi^2 |k|^2} e^{2 \pi i \langle k, x \rangle}, \]
therefore the Parseval formula leads to
\[ \int_{\mathbb{R}^2 / \mathbb{Z}^2} f (\Delta^{-1} f) \, \mathrm{d} \mathrm{Vol} = \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0}} e^{-8 \pi^2 |k|^2 t} \frac{|\widehat{\mu_{a,q,2}}(k)|^2}{4 \pi^2 |k|^2} . \]
Estimates \eqref{W2heatkernellowerbound1}, \eqref{W2heatkernellowerbound2} and the previous formula yield
\[ W_2 (\mu_{a,q,2}, \mathrm{Vol})^2 \gg \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0}} e^{-8 \pi^2 |k|^2 t} \frac{|\widehat{\mu_{a,q,2}}(k)|^2}{|k|^2} \gg \frac{1}{q^2} \sum_{\substack{k \in \mathbb{Z}^2 \\ q \nmid k_1, k_2}} e^{-8 \pi^2 |k|^2 t} \frac{K_2(ak_1 k_2; q)^2}{|k|^2} . \]

Let us now take the average over $a \in (\mathbb{Z}/q\mathbb{Z})^{\times}$. Observe that for any given $k_1, k_2$ not divisible by $q$, the values $ak_1k_2$ attain each residue class in $(\mathbb{Z}/q\mathbb{Z})^{\times}$ exactly once. Therefore
\[ \frac{1}{q-1} \sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} W_2 (\mu_{a,q,2}, \mathrm{Vol})^2 \gg \frac{1}{q^3} \sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} K_2 (a;q)^2 \sum_{\substack{k \in \mathbb{Z}^2 \\ q \nmid k_1, k_2}} \frac{e^{-8 \pi^2 |k|^2 t}}{|k|^2} . \]
The sum of squares of Kloosterman sums is evaluated as
\begin{equation}\label{kloostermansecondmoment}
\begin{split} \sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} K_2(a;q)^2 &= \sum_{x \in (\mathbb{Z}/q\mathbb{Z})^{\times}} \sum_{y \in (\mathbb{Z}/q\mathbb{Z})^{\times}} \sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} e^{2 \pi i (a(x-y) +\overline{x}-\overline{y})/q} \\ &= \sum_{x \in (\mathbb{Z}/q\mathbb{Z})^{\times}} \sum_{y \in (\mathbb{Z}/q\mathbb{Z})^{\times}} \left( (q-1) \mathds{1}_{\{ x=y \}} - e^{2 \pi i (\overline{x}-\overline{y})/q} \mathds{1}_{\{ x \neq y \}} \right) \\ &= \sum_{x \in (\mathbb{Z}/q\mathbb{Z})^{\times}} \left( q+e^{2 \pi i \overline{x}/q} \right) = q^2-q-1. \end{split}
\end{equation}
Finally, by the choice $t=q^{-1/4}$,
\[ \sum_{\substack{k \in \mathbb{Z}^2 \\ q \nmid k_1, k_2}} \frac{e^{-8 \pi^2 |k|^2 t}}{4 \pi^2 |k|^2} \gg \sum_{\substack{k \in \mathbb{Z}^2 \\ |k| \le q^{1/8} \\ q \nmid k_1, k_2}} \frac{1}{|k|^2} \gg \log q . \]
The previous three estimates prove the desired relation
\[ \frac{1}{q-1} \sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} W_2 (\mu_{a,q,2}, \mathrm{Vol})^2 \gg \frac{\log q}{q}. \]

Let $q$ be a prime, let $a \in (\mathbb{Z}/q\mathbb{Z})^{\times}$ and let $0<r \le 1/2$. Lemma \ref{D2torus_lemma} states that
\[ D_2(\mu_{a,q,2}, \mathrm{Vol},r)^2 = \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0}} b_k(r) |\widehat{\mu_{a,q,2}}(k)|^2 \]
with coefficients $b_k(r)=r^2 |k|^{-2} J_1 (2 \pi r |k|)^2 \ll \min \{ r^4, r |k|^{-3} \}$. The total sum of all coefficients is $\sum_{k \neq 0} b_k(r) = \pi r^2 (1-\pi r^2)$.

Recall \eqref{muaq2fouriercoefficients} and note that both $b_k(r)$ and $\widehat{\mu_{a,q,2}}(k)$ are invariant under switching the coordinates of $k$. Hence
\begin{equation}\label{D2muaq2explicit}
D_2 (\mu_{a,q,2}, \mathrm{Vol}, r)^2 = \frac{1}{(q-1)^2} \sum_{\substack{k \in \mathbb{Z}^2 \\ q \nmid k_1, k_2}} b_k(r) K_2(ak_1k_2;q)^2 + \frac{2}{(q-1)^2} \sum_{\substack{k \in \mathbb{Z}^2 \\ q \mid k_1, \,\, q \nmid k_2}} b_k(r) + \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0 \\ q \mid k_1, k_2}} b_k(r) .
\end{equation}
Taking the average over $a \in (\mathbb{Z}/q\mathbb{Z})^{\times}$ and using formula \eqref{kloostermansecondmoment} for the sum of squares of Kloosterman sums leads to
\begin{equation}\label{D2muaq2}
\begin{split} \frac{1}{q-1} \sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} &D_2 (\mu_{a,q,2}, \mathrm{Vol}, r)^2 \\ &= \frac{q^2-q-1}{(q-1)^3} \sum_{\substack{k \in \mathbb{Z}^2 \\ q \nmid k_1, k_2}} b_k(r) + \frac{2}{(q-1)^2} \sum_{\substack{k \in \mathbb{Z}^2 \\ q \mid k_1, \,\, q \nmid k_2}} b_k(r) + \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0 \\ q \mid k_1, k_2}} b_k(r) . \end{split}
\end{equation}

It remains to estimate the sums of coefficients $b_k(r)$ in \eqref{D2muaq2}. Consider the last sum in \eqref{D2muaq2}. If $r \le 1/q$, then
\[ \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0 \\ q \mid k_1, k_2}} b_k(r) \ll \sum_{\substack{\ell \in \mathbb{Z}^2 \\ \ell \neq 0}} \min \left\{ r^4, \frac{r}{q^3 |\ell|^3} \right\} \ll \sum_{\substack{\ell \in \mathbb{Z}^2 \\ 0<|\ell| \le 1/(rq)}} r^4 + \sum_{\substack{\ell \in \mathbb{Z}^2 \\ |\ell| > 1/(rq)}} \frac{r}{q^3 |\ell|^3} \ll \frac{r^2}{q^2}. \]
If $r>1/q$, then
\[ \sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0 \\ q \mid k_1, k_2}} b_k(r) \ll \sum_{\substack{\ell \in \mathbb{Z}^2 \\ \ell \neq 0}} \frac{r}{q^3 |\ell|^3} \ll \frac{r}{q^3}. \]
Altogether,
\begin{equation}\label{qmidk1k2}
\sum_{\substack{k \in \mathbb{Z}^2 \\ k \neq 0 \\ q \mid k_1, k_2}} b_k(r) \ll \min \left\{ \frac{r^2}{q^2}, \frac{r}{q^3} \right\} .
\end{equation}

Consider now the second sum in \eqref{D2muaq2} with the extra condition $k_1 \neq 0$. If $r \le 1/q$, then
\[ \begin{split} \sum_{\substack{k \in \mathbb{Z}^2 \\ q \mid k_1, \,\, q \nmid k_2 \\ k_1 \neq 0}} b_k(r) &\ll \sum_{\substack{\ell \in \mathbb{Z}^2 \\ \ell \neq 0}} \min \left\{ r^4, \frac{r}{(q^2 \ell_1^2 + \ell_2^2)^{3/2}} \right\} \\ &\ll \sum_{|\ell_1| \le 1/(rq)} \sum_{|\ell_2| \le 1/r} r^4 + \sum_{|\ell_1| \le 1/(rq)} \sum_{|\ell_2| > 1/r} \frac{r}{|\ell_2|^3} \\ &{\phantom{\ll}} + \sum_{|\ell_1| > 1/(rq)} \sum_{|\ell_2| \le q|\ell_1|} \frac{r}{q^3 |\ell_1|^3} + \sum_{|\ell_1| > 1/(rq)} \sum_{|\ell_2|> q |\ell_1|} \frac{r}{|\ell_2|^3} \\ &\ll  \frac{r^2}{q} . \end{split} \]
If $r>1/q$, then
\[ \sum_{\substack{k \in \mathbb{Z}^2 \\ q \mid k_1, \,\, q \nmid k_2 \\ k_1 \neq 0}} b_k(r) \ll \sum_{\substack{\ell \in \mathbb{Z}^2 \\ \ell_1, \ell_2 \neq 0}} \frac{r}{(q^2 \ell_1^2 + \ell_2^2)^{3/2}} \ll \sum_{\ell_1 \neq 0} \sum_{|\ell_2| \le q |\ell_1|} \frac{r}{q^3 |\ell_1|^3} + \sum_{\ell_1 \neq 0} \sum_{|\ell_2 | > q|\ell_1|} \frac{r}{|\ell_2|^3} \ll \frac{r}{q^2} . \]
Altogether,
\begin{equation}\label{qmidk1nmidk2}
\sum_{\substack{k \in \mathbb{Z}^2 \\ q \mid k_1, \,\, q \nmid k_2 \\ k_1 \neq 0}} b_k(r) \ll \min \left\{ \frac{r^2}{q}, \frac{r}{q^2} \right\} .
\end{equation}
Combining \eqref{D2muaq2}--\eqref{qmidk1nmidk2} leads to
\begin{equation}\label{D2muaq2second}
\begin{split} \frac{1}{q-1} \sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} &D_2 (\mu_{a,q,2}, \mathrm{Vol}, r)^2 \\ &= \frac{q^2-q-1}{(q-1)^3} \sum_{\substack{k \in \mathbb{Z}^2 \\ q \nmid k_1, k_2}} b_k(r) + \frac{2}{(q-1)^2} \sum_{\substack{k \in \mathbb{Z}^2 \\ k_1=0, \,\, q \nmid k_2}} b_k(r)+ O \left( \min \left\{ \frac{r^2}{q^2}, \frac{r}{q^3} \right\} \right) . \end{split}
\end{equation}

The reason for the deviation of the constants from the i.i.d.\ case is that the sum of the coefficients $b_k(r)$ over $k_1=0$, $k_2 \neq 0$ is not negligible. Indeed,
\[ \sum_{\substack{k \in \mathbb{Z}^2 \\ k_1=0, \,\, q \nmid k_2}} b_k(r) = \sum_{\substack{\ell \in \mathbb{Z} \\ q \nmid \ell}} \frac{r^2}{\ell^2} J_1 (2 \pi r |\ell|)^2 = \sum_{\substack{\ell \in \mathbb{Z} \\ \ell \neq 0}} \frac{r^2}{\ell^2} J_1 (2 \pi r |\ell|)^2 + O \left( \frac{r^2}{q^2} \right) . \]
To evaluate the infinite series in the previous formula, observe that the Fourier coefficients of the even function $f(x)=\mathds{1}_{\{ |x| \le r \}} \sqrt{1-x^2/r^2}$, $x \in [-1/2, 1/2]$, are
\[ \int_{-1/2}^{1/2} f(x) e^{-2 \pi i \ell x} \, \mathrm{d}x = r \int_{-1}^1 \sqrt{1-x^2} e^{2 \pi i r |\ell| x} \, \mathrm{d}x = \frac{J_1(2 \pi r |\ell|)}{2 |\ell|} , \quad \ell \in \mathbb{Z} \backslash \{ 0 \} . \]
An application of the Parseval formula thus yields
\[ \sum_{\substack{\ell \in \mathbb{Z} \\ \ell \neq 0}} \frac{J_1 (2 \pi r |\ell|)^2}{4 \ell^2} = \int_{-r}^r \left( 1-\frac{x^2}{r^2} \right) \, \mathrm{d}x - \left( \int_{-r}^r \sqrt{1-\frac{x^2}{r^2}} \, \mathrm{d}x \right)^2 = \frac{4}{3} r - \frac{\pi^2}{4} r^2, \]
hence
\[ \sum_{\substack{k \in \mathbb{Z}^2 \\ k_1=0, \,\, q \nmid k_2}} b_k(r) = \frac{16}{3} r^3 - \pi^2 r^4 + O \left( \frac{r^2}{q^2} \right) . \]
The fact that $\sum_{k \neq 0} b_k(r)=\pi r^2 (1-\pi r^2)$, \eqref{qmidk1k2}, \eqref{qmidk1nmidk2} and the previous formula lead to
\[ \sum_{\substack{k \in \mathbb{Z}^2 \\ q \nmid k_1, k_2}} b_k(r) = \pi r^2 (1-\pi r^2) - \frac{32}{3} r^3 + 2 \pi^2 r^4 +O \left( \min \left\{ \frac{r^2}{q}, \frac{r}{q^2} \right\} \right) . \]
The previous two formulas simplify \eqref{D2muaq2second} to
\[ \begin{split} \frac{1}{q-1} &\sum_{a \in (\mathbb{Z}/q\mathbb{Z})^{\times}} D_2 (\mu_{a,q,2}, \mathrm{Vol}, r)^2 \\ &= \frac{q^2-q-1}{(q-1)^3} \left( \pi r^2 (1-\pi r^2) - \frac{32}{3} r^3 + 2 \pi^2 r^4 \right) + \frac{(32/3)r^3 - 2 \pi^2 r^4}{(q-1)^2} + O \left( \min \left\{ \frac{r^2}{q^2}, \frac{r}{q^3} \right\} \right) . \end{split} \]
Here $(q^2-q-1)/(q-1)^3 = 1/(q-1) + 1/(q-1)^2+O(q^{-3})$, and the desired formula for the quadratic average of $D_2 (\mu_{a,q,2}, \mathrm{Vol},r)$ follows. Further averaging over $0<r \le 1/2$ leads to the claim about the quadratic average of $D_2(\mu_{a,q,2}, \mathrm{Vol})$.

Let $q$ be a prime and let $a \in (\mathbb{Z}/q\mathbb{Z})^{\times}$, $b \in \mathbb{Z}/q\mathbb{Z}$. The Fourier transform of the empirical measure $\mu_{a,b,q}$ is
\[ \widehat{\mu_{a,b,q}}(k) = \frac{1}{q} \sum_{x \in \mathbb{Z}/q\mathbb{Z}} e^{-2 \pi i (k_1 x + k_2 (ax^2+bx))/q} . \]
If $q \nmid k_2$, then this is a quadratic Gauss sum, therefore $|\widehat{\mu_{a,b,q}}(k)|=1/\sqrt{q}$, see \cite[p.\ 50]{IK}. If $q \mid k_2$, then the sum can be evaluated as a geometric series. Hence

|\widehat{\mu_{a,b,q}}(k)| = \left\{ \begin{array}{ll} 1 & \textrm{if } q \mid k_1, k_2, \\ 0 & \textrm{if } q \nmid k_1, \,\, q \mid k_2, \\ 1/\sqrt{q} & \textrm{if } q \nmid k_2 . \end{array} \right.

\label{fibonacci_lemma} For any $2 \le p < \infty$, we have
\[ W_p (\nu_N, \mathrm{Vol}) \ll p^2 N^{-1/2} \]
with an implied constant that depends only on $c$.

<!-- proof-13: body_md -->
We may assume $p+2 \le \lfloor \sqrt{cN} \rfloor$ by Lemma \ref{WqWplemma}. Let $1<q \le 2$ be such that $1/p+1/q=1$, and let us apply Theorem \ref{torussmoothing_theorem1} with $H=\lfloor \sqrt{cN} \rfloor$ to deduce
\[ W_p (\nu_N, \mathrm{Vol}) \ll \frac{p}{\sqrt{N}} + p \bigg( \sum_{\substack{k \in [-H,H]^2 \\ k \neq 0}} \frac{|\widehat{\nu_N(k)}|^q}{|k|^q} \bigg)^{1/q} . \]
The Fourier transform of $\nu_N(k)$ for $k=(k_1, k_2)$, $k_1 \neq 0$, can be estimated as
\[ \begin{split} |\widehat{\nu_N}(k)| &= \frac{1}{2N} \bigg| \sum_{n=0}^{N-1} \left( e^{-2 \pi i (k_1 \alpha + k_2/(2N))n} + e^{-2 \pi i (k_1 \alpha - k_2 /(2N))n} \right) \bigg| \\ &\le \frac{1}{N} \left( \frac{1}{|1-e^{-2 \pi i (k_1 \alpha + k_2 /(2N))}|} +\frac{1}{|1-e^{-2 \pi i (k_1 \alpha - k_2 /(2N))}|} \right) \\ &\ll \frac{1}{N} \left( \frac{1}{\| k_1 \alpha + k_2/(2N) \|_{\mathbb{R}/\mathbb{Z}}} + \frac{1}{\| k_1 \alpha - k_2/(2N) \|_{\mathbb{R}/\mathbb{Z}}} \right) . \end{split} \]
For all $k \in [-H,H]^2$, $k_1 \neq 0$, we have $|k_1|, |k_2| \le H \le \sqrt{cN}$, therefore $\| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}} \ge c/|k_1| \ge |k_2|/N$ and
\[ \| k_1 \alpha \pm k_2/(2N) \|_{\mathbb{R}/\mathbb{Z}} \ge \| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}} - \frac{|k_2|}{2N} \ge \frac{1}{2} \| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}} . \]
Consequently, $|\widehat{\nu_N}(k)| \ll 1/(N \| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}})$ whenever $k_1 \neq 0$. If $k \in [-H,H]^2$, $k_1 =0$, then
\[ \begin{split} \widehat{\nu_N}(k) &= \frac{1}{2N} \sum_{n=0}^{N-1} \left( e^{-2 \pi i k_2 n/(2N)} + e^{2 \pi i k_2 n /(2N)} \right) = \frac{1}{2N} \left( \frac{1-(-1)^{k_2}}{1-e^{- 2 \pi i k_2 /(2N)}} + \frac{1-(-1)^{k_2}}{1-e^{2 \pi i k_2/(2N)}} \right) \\ &= \left\{ \begin{array}{ll} 0 & \textrm{if } k_2 \textrm{ is even}, \\ 1/N & \textrm{if } k_2 \textrm{ is odd}. \end{array} \right. \end{split} \]
Hence
\[ W_p (\nu_N, \mathrm{Vol}) \ll \frac{p}{\sqrt{N}} + p \bigg( \sum_{\substack{k \in [-H,H]^2 \\ k_1 \neq 0}} \frac{1}{N^q |k|^q \| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}}^q} + \sum_{\substack{k_2 \in [-H,H] \\ k_2 \neq 0}} \frac{1}{N^q |k_2|^q} \bigg)^{1/q} . \]
The second sum in the previous formula can be estimated as
\[ \sum_{\substack{k_2 \in [-H,H] \\ k_2 \neq 0}} \frac{1}{N^q |k_2|^q} \le \frac{2}{N^q} \sum_{n=1}^{\infty} \frac{1}{n^q} \ll \frac{1}{(q-1)N^q} \ll \frac{p}{N^q}. \]
Let $1 \le Q \le H$. In particular, $2Q \le 2\sqrt{cN} \le \sqrt{N}$. For any $k_1, k_1' \in [-Q,Q]$, $k_1 \neq k_1'$, we have $\| k_1 \alpha - k_1' \alpha \|_{\mathbb{R}/\mathbb{Z}} \ge c/|k_1-k_1'| \gg 1/Q$. By the pigeonhole principle, we thus have
\[ \sum_{0<|k_1| \le Q} \frac{1}{\| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}}^q} \ll \sum_{n=1}^{\infty} \frac{1}{(n/Q)^q} \ll p Q^q. \]
Therefore
\[ \sum_{Q/2 < \| k \|_{\infty} \le Q} \frac{1}{|k|^q \| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}}^q} \ll \frac{1}{Q^q} \sum_{|k_2| \le Q} \sum_{0<|k_1| \le Q} \frac{1}{\| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}}^q} \ll p Q, \]
and a dyadic decomposition of the interval $[1,H]$ leads to
\[ \sum_{\substack{k \in [-H,H]^2 \\ k_1 \neq 0}} \frac{1}{|k|^q \| k_1 \alpha \|_{\mathbb{R}/\mathbb{Z}}^q} \ll p H \ll p \sqrt{N} . \]
Hence
\[ W_p (\nu_N, \mathrm{Vol}) \ll \frac{p}{\sqrt{N}} + p \bigg( \frac{p \sqrt{N}}{N^q} \bigg)^{1/q} . \]
The exponent of $N$ in the second term is $1/(2q) -1 < -1/2$, and the claim follows.

<!-- lem-8: body_md -->
\label{phipushforward_lemma} Let $\mu \in \mathcal{P}((\mathbb{R}/\mathbb{Z}) \times [0,1])$ and $1 \le p < 2$. For any $p'>2p/(2-p)$, we have
\[ W_p (\phi_* \mu, \mathrm{Vol}) \ll \left( 1-\frac{pp'}{2(p'-p)} \right)^{1/p'-1/p} W_{p'}(\mu, \mathrm{Vol}) \]
with a universal implied constant.
<!-- end lem-8 -->
<!-- proof-14: body_md -->
For the sake of readability, let $M=(\mathbb{R}/\mathbb{Z}) \times [0,1]$ denote the cylinder. Let $\pi \in \mathrm{Coupling}(\mu, \mathrm{Vol})$ for which the minimum in the definition of the Wasserstein metric $W_{p'}(\mu, \mathrm{Vol})$ is attained:
\[ W_{p'} (\mu, \mathrm{Vol}) = \left( \int_{M \times M} \rho_M (x,y)^{p'} \, \mathrm{d}\pi (x,y) \right)^{1/p'} . \]
Let $F: M \times M \to \mathbb{S}^2 \times \mathbb{S}^2$, $F(x,y)=(\phi(x), \phi(y))$. One readily checks that the pushforward $F_* \pi \in \mathcal{P}(\mathbb{S}^2 \times \mathbb{S}^2)$ is a coupling between $\phi_* \mu$ and $\phi_* \mathrm{Vol}=\mathrm{Vol}$. By the definition of the Wasserstein metric,
\[ \begin{split} W_p (\phi_* \mu, \mathrm{Vol}) &\le \left( \int_{\mathbb{S}^2 \times \mathbb{S}^2} \rho_{\mathbb{S}^2} (u,v)^p \, \mathrm{d} (F_* \pi)(u,v) \right)^{1/p} \\ &= \left( \int_{M \times M} \rho_{\mathbb{S}^2} (\phi(x), \phi(y))^p \, \mathrm{d}\pi (x,y) \right)^{1/p} . \end{split} \]
The partial derivatives of the Lambert map are
\[ \begin{split} \frac{\partial \phi}{\partial x_1} &= \left( -4 \pi \sqrt{x_2-x_2^2} \sin (2 \pi x_1), 4 \pi \sqrt{x_2-x_2^2} \cos (2 \pi x_1), 0 \right), \\ \frac{\partial \phi}{\partial x_2} &= \left( \frac{1-2x_2}{\sqrt{x_2-x_2^2}} \cos (2 \pi x_1), \frac{1-2x_2}{\sqrt{x_2-x_2^2}} \sin (2 \pi x_1), -2 \right) . \end{split} \]
In particular,
\[ \left| \frac{\partial \phi}{\partial x_1} \right| \ll 1 \qquad \textrm{and} \qquad \left| \frac{\partial \phi}{\partial x_2} \right| \ll \frac{1}{\sqrt{x_2-x_2^2}} \ll \frac{1}{\sqrt{x_2}} + \frac{1}{\sqrt{1-x_2}} . \]
For all $x=(x_1, x_2), y=(y_1, y_2) \in M$, we thus deduce
\[ \begin{split} |\phi (x) - \phi(y)| &\le |\phi (x_1, x_2) - \phi (y_1, x_2)| + |\phi (y_1, x_2) - \phi (y_1, y_2)| \\ &\ll \| x_1-y_1 \|_{\mathbb{R}/\mathbb{Z}} + \left| \int_{x_2}^{y_2} \left( \frac{1}{\sqrt{t}} + \frac{1}{\sqrt{1-t}} \right) \, \mathrm{d}t \right| \\ &\ll \| x_1-y_1 \|_{\mathbb{R}/\mathbb{Z}} + \frac{|x_2-y_2|}{\sqrt{x_2} + \sqrt{y_2}} + \frac{|x_2-y_2|}{\sqrt{1-x_2} + \sqrt{1-y_2}} \\ &\ll \rho_M (x,y) \left( \frac{1}{\sqrt{y_2}} + \frac{1}{\sqrt{1-y_2}} \right) . \end{split} \]
Since the geodesic metric $\rho_{\mathbb{S}^2}$ is bounded above by a constant times the Euclidean metric, we have
\[ \rho_{\mathbb{S}^2} (\phi(x), \phi(y))^p \ll \rho_M (x,y)^p \left( \frac{1}{y_2^{p/2}} + \frac{1}{(1-y_2)^{p/2}} \right) . \]
An application of the H\"older inequality with exponents $p'/p>1$ and $p'/(p'-p)$ leads to
\[ \begin{split} W_p (\phi_* \mu, \mathrm{Vol}) &\ll \left( \int_{M \times M} \rho_M (x,y)^p \left( \frac{1}{y_2^{p/2}} + \frac{1}{(1-y_2)^{p/2}} \right) \, \mathrm{d}\pi (x,y) \right)^{1/p} \\ &\ll \left( \int_{M \times M} \rho_M (x,y)^{p'} \, \mathrm{d}\pi (x,y) \right)^{1/p'} \\ &{\phantom{\ll}} \times \left( \int_{M \times M} \left( \frac{1}{y_2^{pp'/(2(p'-p))}} + \frac{1}{(1-y_2)^{pp'/(2(p'-p))}} \right) \, \mathrm{d}\pi (x,y) \right)^{(p'-p)/(pp')} . \end{split} \]
The first integral in the previous formula is $W_{p'}(\mu, \mathrm{Vol})$ because $\pi$ was chosen to be an optimal coupling. Since the second integral is a function of $y$, $\pi$ is a coupling between $\mu$ and $\mathrm{Vol}$, and $pp'/(2(p'-p))<1$ by assumption, we obtain the desired estimate
\[ \begin{split} W_p (\phi_* \mu, \mathrm{Vol}) &\ll W_{p'}(\mu, \mathrm{Vol}) \left( \int_M \left( \frac{1}{y_2^{pp'/(2(p'-p))}} + \frac{1}{(1-y_2)^{pp'/(2(p'-p))}} \right) \, \mathrm{d}\mathrm{Vol}(y) \right)^{(p'-p)/(pp')} \\ &\ll W_{p'}(\mu, \mathrm{Vol}) \left( \int_0^{1/2} \frac{1}{t^{pp'/(2(p'-p))}} \, \mathrm{d}t \right)^{1/p-1/p'} \\ &\ll W_{p'}(\mu, \mathrm{Vol}) \left( 1-\frac{pp'}{2(p'-p)} \right)^{1/p'-1/p} . \end{split} \]

Fix $1 \le p < 2$, and let $p'>2p/(2-p)$ to be chosen. An application of Lemmas \ref{fibonacci_lemma} and \ref{phipushforward_lemma} and formula \eqref{fpushforward} lead to
\[ \begin{split} W_p (\mu_N, \mathrm{Vol}) &= W_p (\phi_* f_* \nu_N, \mathrm{Vol}) \ll \left( 1-\frac{pp'}{2(p'-p)} \right)^{1/p'-1/p} W_{p'}(f_* \nu_N, \mathrm{Vol}) \\ &\ll \left( 1-\frac{pp'}{2(p'-p)} \right)^{1/p'-1/p} W_{p'}(\nu_N, \mathrm{Vol}) \ll \left( 1-\frac{pp'}{2(p'-p)} \right)^{1/p'-1/p} (p')^2 N^{-1/2} . \end{split} \]
Choosing $p'=4p/(2-p)$ proves the claim $W_p (\mu_N, \mathrm{Vol}) \ll (2-p)^{-5/2} N^{-1/2}$.
