\label{thm:separation}
For every $G,r,B$ as above, the matrix
$H_BH_B^{\mathsf T}-J_B$ is positive semidefinite. Consequently,
\begin{equation}\label{eq:ancestor}
 \sum_{x,y\in V}f(x)f(y)a_B(x,y)
 \le \left(\sum_{x\in V}f(x)H_B(x)\right)^2
\end{equation}
for every nonnegative $f:V\to\mathbb R$.
For all $x,y\in V$, one also has
\begin{equation}\label{eq:pointwise}
 a_B(x,y)\le 2H_B(x)H_B(y).
\end{equation}
In particular, if $f$ is a probability distribution on $V$ and
$X,Y$ are independent with common law $f$,
independently of $T$, then \eqref{eq:ancestor} becomes
$\mathbb P(\operatorname{LCA}_T(X,Y)\in B)
\le\mathbb P(\gamma_X\cap B\ne\varnothing)^2$.

\label{thm:bk}
Let $G$ be a finite connected transitive weighted graph with $n$ vertices.
Let $T$ have law \eqref{eq:tree-law}, and let $X,Y$ be independent uniform vertices, independent of $T$.
Write $\gamma=[X,Y]_T$.
For every deterministic $B\subseteq V$,
\begin{equation}\label{eq:set-hit}
 \mathbb P(\gamma\cap B\ne\varnothing)\ge\sqrt{\frac{|B|}{n}}.
\end{equation}
For a deterministic partition $\mathcal P=\{B_1,\ldots,B_k\}$ of $V$ into nonempty sets,
let $N_{\mathcal P}$ count the blocks of $\mathcal P$ visited by $\gamma$. Then
\begin{equation}\label{eq:partition-hit}
 \mathbb E N_{\mathcal P}\ge\sum_{i=1}^k\sqrt{\frac{|B_i|}{n}}.
\end{equation}
In particular, let $d_T(X,Y)$ be the number of edges
in the unique path from $X$ to $Y$ in $T$. Then
\begin{equation}\label{eq:bk}
 \mathbb E d_T(X,Y)\ge\sqrt n-1.
\end{equation}

\label{thm:matroid-input}
Let $\mathsf M$ be a matroid on a finite ground set $F$, with
independent sets $\mathcal I(\mathsf M)$.
Then
\begin{equation}\label{eq:matroid-input}
 \sum_{I\in\mathcal I(\mathsf M)}
 s^{|F|-|I|}\prod_{e\in I}z_e
 \quad\text{and}\quad
 \sum_{\substack{A\subseteq F\\ A\text{ a basis of }\mathsf M}}
 \prod_{e\in A}z_e
\end{equation}
are homogeneous completely log-concave polynomials.
Moreover, the class of homogeneous completely log-concave
polynomials is closed under products, partial derivatives, and
nonnegative linear substitutions.

The complete log-concavity of the first polynomial is proved in
\cite[Theorem~4.1]{ALOV}; the same property for both polynomials
follows from \cite[Proposition~4.25 and Theorem~2.30]{BH}. The closure properties follow from \cite[Theorem~2.10 and Corollary~2.32]{BH}.

<!-- proposition-1: body_md -->
\label{prop:Q}
The polynomial $Q_B$ is completely log-concave.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Put $U=V\setminus B$, $k=|V|-1$, and $E_U=E(G[U])$.
Add a new vertex $*$ and the star edges $M=\{s_u=\{*,u\}:u\in U\}$.
On $E\sqcup M$, take the direct sum $\mathsf N$ of the graphic matroid of $G[U]+M$ and the free matroid on $E\setminus E_U$.
Set $N=|E|+|U|$.

Introduce variables $t_e$ for $e\in E$, variables $z_u$ for $u\in U$, and a homogenizing variable $s$.
The polynomial
\begin{equation}\label{eq:I}
 I(s,t,z)=\sum_{J\in\mathcal I(\mathsf N)}s^{N-|J|}
       \prod_{e\in J\cap E}(c_et_e)\prod_{s_u\in J\cap M}z_u
\end{equation}
is homogeneous completely log-concave by Theorem~\ref{thm:matroid-input} after the change of variables \(t_e\mapsto c_et_e\).
So is the basis generating polynomial of the dual graphic matroid
\begin{equation}\label{eq:B-star}
 B^*(t)=\sum_{T\text{ spanning tree of }G}\prod_{e\in E\setminus T}t_e.
\end{equation}

Extracting the coefficient of \(\prod_{e\in E}t_e\) gives
\begin{equation}\label{eq:coefficient}
 F(s,z)=\left[\prod_{e\in E}t_e\right]\bigl(I(s,t,z)B^*(t)\bigr)
 =\left.\left(\prod_{e\in E}\partial_{t_e}\right)(IB^*)\right|_{t=0}.
\end{equation}
The closure properties above show that $F$ is also homogeneous completely log-concave, of degree $D=N-k$.
For each spanning tree $T$, a subset $S\subseteq M$
can be added to $T\cap E_U$ without creating a cycle
precisely when $S$ contains at most one edge $s_u$
with $u$ in each component of $T\setminus B$.
Consequently,
\begin{equation}\label{eq:F}
 F(s,z)=\sum_T w(T)s^{D-|\mathcal C_B(T)|}
                       \prod_{C\in\mathcal C_B(T)}\left(s+\sum_{u\in C}z_u\right).
\end{equation}
All exponents in~\eqref{eq:F} are nonnegative, since $D\ge |U|\ge|\mathcal C_B(T)|$.

Differentiate $F(s,z)$ with respect to \(z_r\) and then set \(s=1\).
Since only the factor belonging to $C_r(T)$ is differentiated, we have
\begin{equation}\label{eq:Q-extract}
 Q_B(z)=\left.Z_G^{-1}\partial_{z_r}F(s,z)\right|_{s=1}.
\end{equation}
Note that partial differentiation preserves homogeneous complete log-concavity, and specialization at \(s=1\) preserves complete log-concavity in the remaining variables. We conclude.

We work with continuous-time simple random walk with unit jump rate.
We first obtain a local connectivity estimate at killing rate $R^{-2}$, and then extend it to all \(0<m\le R^{-2}\) by monotonicity in the killing rate.

\emph{Local connectivity estimate.}
We first prove a second-moment estimate on any transitive weighted undirected graph with
$\sum_y c_{xy}=1$ for every $x\in V$, which will also be used in Section~\ref{sec:killed}.
Let $p_t(x,y)$ be its transition kernel, fix an origin $o$, and put
\begin{equation}\label{eq:massive-green}
 g_m(x,y)=\int_0^\infty e^{-mt}p_t(x,y)\,\mathrm d t,
 \qquad \mathcal B(m)=\sum_zg_m(o,z)^2.
\end{equation}
Note that
\begin{equation}\label{eq:bubble-identity}
 \sum_zg_m(x,z)=m^{-1},\qquad
 \mathcal B(m)=\int_0^\infty t e^{-mt}p_t(o,o)\,\mathrm d t.
\end{equation}
For independent walks $X,Y$ started at $x,y$, with independent exponential lifetimes $\zeta_X,\zeta_Y$ of rate $m$, define
\begin{equation}\label{eq:I-occupation}
 I_{x,y}=\int_0^{\zeta_X}\int_0^{\zeta_Y}
                 \mathbf 1_{\{X_s=Y_t\}}\,\mathrm d t\,\mathrm d s.
\end{equation}
Writing $g=g_m$, we have
\begin{align}
 \mathbb E I_{x,y}
 &=\sum_u g(x,u)g(y,u),\label{eq:I-first}\\
 \mathbb E I_{x,y}^2
 &=2\sum_{u,v}g(x,u)g(y,u)g(u,v)^2
   +2\sum_{u,v}g(x,u)g(y,v)g(u,v)^2\le4\mathcal B(m)^2.\label{eq:I-second}
\end{align}
The last inequality follows from Cauchy--Schwarz inequality and that $ab\le(a^2+b^2)/2$.
Wilson's algorithm and the inequality of~\cite[Lemma~1.2 and Remark~1.3]{LPS} imply
\begin{equation}\label{eq:LPS}
 \mathbb P_m(y\in C_m(x))
 =\mathbb P\bigl(\operatorname{LE}(X[0,\zeta_X))\cap Y[0,\zeta_Y)\ne\varnothing\bigr)
 \ge2^{-8}\mathbb P(I_{x,y}>0).
\end{equation}

Now on $\mathbb Z^3$, set $\mu=R^{-2}$, and take $x,y\in B_R$.
The heat-kernel bounds give
\begin{equation}\label{eq:3d-green}
 \inf_{x,u\in B_R}g_\mu(x,u)\ge cR^{-1},
 \qquad \mathcal B(\mu)\le CR.
\end{equation}
Here, the first estimate follows by integrating over $t\in[R^2,2R^2]$, and the second follows from
$p_t(0,0)\le C(1+t)^{-3/2}$ and \eqref{eq:bubble-identity}; see~\cite[Theorem~2.1.3]{LL}.
Combining~\eqref{eq:I-first} and \eqref{eq:I-second} gives
$\mathbb E I_{x,y}\ge cR$ and $\mathbb E I_{x,y}^2\le CR^2$.
Hence the second-moment inequality gives $\mathbb P(I_{x,y}>0)\ge c$, and \eqref{eq:LPS} gives $\inf_{x,y\in B_R}\mathbb P_\mu(y\in C_\mu(x))
\ge a_0>0$, where $a_0$ is independent of $R$.

Write \(x\longleftrightarrow_D y\) if \(x\) and \(y\) are connected in \(\mathfrak F_m\) and their unique path in \(\mathfrak F_m\) is contained in \(D\).
Doob's inequality applied up to the exponential lifetime $\zeta_X$ gives
\[
 \sup_{x\in B_R}\mathbb P_x\left(\sup_{t<\zeta_X}|X_t|\ge KR\right)
 \le \frac{C}{(K-1)^2}.
\]
If both $X[0,\zeta_X)$ and $Y[0,\zeta_Y)$
stay in $B_{KR}$, and $Y$ hits
$\operatorname{LE}(X[0,\zeta_X))$ before $\zeta_Y$,
then Wilson's algorithm gives
$x\longleftrightarrow_{B_{KR}}y$.
Choose $K>1$ sufficiently large such that
$2C/(K-1)^2\le a_0/2$.
Then with $a=a_0/2$, we obtain
\begin{equation}\label{eq:local-hyp}
 \inf_{x,y\in B_R}
 \mathbb P_\mu(x\longleftrightarrow_{B_{KR}}y)\ge a.
\end{equation}

\emph{Removing the killing.}
For $0<m\le\mu$, the massive spanning forests
can be coupled so that $\mathfrak F_\mu\subseteq\mathfrak F_m$.
To see this, orient the edges arbitrarily and define
$\nabla:\ell^2(E)\to\ell^2(V)$ by $\nabla\delta_e=\sqrt{c_e}(\delta_{e^+}-\delta_{e^-})$.
By the transfer-current theorem
\cite[Section~4.2]{LP}, the edge set of $\mathfrak F_m$
is determinantal with kernel $ \mathcal K_m=\nabla^*(\nabla\nabla^*+mI)^{-1}\nabla$.
The same formula holds in infinite volume by exhaustion; see e.g.~\cite[Section~10.1]{LP}. Now monotonicity of the resolvent gives
$0\preceq \mathcal{K}_\mu\preceq \mathcal{K}_m\preceq I$, and the determinantal domination theorem (see~\cite[Theorem~2.9]{Lyons}) therefore couples the spatial edge sets with $\mathfrak F_\mu\subseteq\mathfrak F_m$. Consequently, \eqref{eq:local-hyp} holds for every $0<m\le\mu$ as the event $\{x\longleftrightarrow_{B_{KR}}y\}$ there is increasing.

On this event, $\operatorname{LCA}(x,y)$ lies on their connecting path, hence in $B_{KR}$.
Put
\[
 h_m(u)=\mathbb P_m(u\in\gamma_0^m),\qquad
 A_m(t)=\sum_{u\in B_t}h_m(u).
\]
We apply \eqref{eq:ancestor} with $f=\mathbf 1_{B_R}$ and $B=\{v\}$ to each $v\in B_{KR}$, and then sum over \(v\).
Translation invariance then gives
\begin{align}
 a|B_R|^2
 &\le\sum_{v\in B_{KR}}\sum_{x,y\in B_R}
              \mathbb P_m(\operatorname{LCA}(x,y)=v)\notag\\
 &\le\sum_{v\in B_{KR}}
       \left(\sum_{x\in B_R}h_m(v-x)\right)^2
 \le |B_{KR}|A_m((K+1)R)^2.\label{eq:local-bound}
\end{align}
Thus $A_m((K+1)R)\ge cR^{3/2}$, uniformly for $0<m\le R^{-2}$.
By transience, for every fixed \(u\), the indicator \(\mathbf1_{\{u\in\operatorname{LE}(S[0,t])\}}\) converges to  \(\mathbf1_{\{u\in\eta\}}\) as \(t\to\infty\).
Consequently, $h_m(u)\to\mathbb P(u\in\eta)$ as $m\downarrow0$.
Taking $m\downarrow0$ in the finite sum $A_m((K+1)R)$ and rescaling $R$ by the fixed factor $K+1$ proves \eqref{eq:occupation-main} after adjusting the constant \(c\).

\emph{The growth exponent.}
For independent simple random walks $S^1,S^2$ started at $0$, define
\begin{equation}\label{eq:Es}
 \operatorname{Es}(r)=\mathbb P\bigl(\operatorname{LE}(S^1[0,\tau_r^1])
                    \cap S^2[1,\tau_r^2]=\varnothing\bigr),
\end{equation}
where $\tau_r^i$ is the first exit time of $S^i$ from $B_r$.
By~\cite[Corollary~1.3]{LS}, we have
\begin{equation}\label{eq:Es-beta}
 \operatorname{Es}(r)\asymp r^{\beta-2},\qquad r\ge1.
\end{equation}
The finite-volume one-point estimate in~\cite[Proposition~8.2]{ShiG}, combined with transience and the Green-function bound $\sum_{j\ge0}\mathbb P_0(S_j=z)\le C/(1+|z|)$~\cite[Theorem~4.3.1]{LL}, implies
\begin{equation}\label{eq:one-point-infinite}
 \mathbb P(z\in\eta)\le\frac{C}{1+|z|}\operatorname{Es}(|z|),
 \qquad z\ne0.
\end{equation}
Summing over lattice thus yields $\mathbb E|\eta\cap B_R|\le CR^\beta$.
Combined with \eqref{eq:occupation-main} gives $\beta\ge3/2$.

\label{thm:independent-lerw}
Let $\eta_a$ and $\eta_b$ be two independent infinite LERWs on $\mathbb Z^3$, started at arbitrary $a,b\in\mathbb Z^3$. Then
\[
 \mathbb P\bigl(|\eta_a\cap\eta_b|=\infty\bigr)=1.
\]
<!-- end theorem-5 -->
<!-- proof-6: body_md -->
Put $q=3-\beta$ and $\kappa=3-2q=2\beta-3\ge0$.
By \eqref{eq:Es-beta} and \eqref{eq:one-point-infinite}, $\mathbb P(z\in\eta_x)\le C(1+|z-x|)^{-q}$.
For $n\ge1$, let
\[
 D_n(x)=\{z\in\mathbb Z^3:
 |z-x-(n/2)e_1|_\infty<n/1000\},
 \qquad e_1=(1,0,0),
\]
and let $\sigma_n^x$ be the first exit time of $\eta_x$
from $B(x,n):=x+B_n$.

Choose $n_0\ge10^5(1+|a|+|b|)$ and set $n_j=2^jn_0$.
The sets $Q_j:=D_{n_j}(a)\cap D_{n_j}(b)$ are pairwise disjoint, satisfy $|Q_j|\asymp n_j^3$,
and lie at distance comparable to $n_j$ from both
$a$ and $b$. Define
\[
 Z_j=\bigl|Q_j\cap\eta_a[0,\sigma_{n_j}^a]
                    \cap\eta_b[0,\sigma_{n_j}^b]\bigr|.
\]
By \cite[Proposition~3.16]{ACHS}, applied with
$K=\varnothing$, for $z\ne w$ in $D_n(x)$,
\[
 \mathbb P(z\in\eta_x[0,\sigma_n^x])\ge cn^{-q},
 \qquad
 \mathbb P(z,w\in\eta_x[0,\sigma_n^x])
 \le Cn^{-q}|z-w|^{-q}.
\]
Independence therefore gives $\mathbb E Z_j\asymp n_j^\kappa$ and $\mathbb E Z_j^2
 \le Cn_j^\kappa
 \left(1+\sum_{r=1}^{n_j}r^{2-2q}\right)
 \le Cn_j^{2\kappa}(1+\log n_j)$,
where the last inequality uses $2-2q=\kappa-1$
and $\kappa\ge0$.

For $i<j$, $z\in Q_i$, $w\in Q_j$, and
$x\in\{a,b\}$, we have
\[
 |z-x|\asymp n_i,\qquad
 |w-x|\asymp|z-w|\asymp n_j.
\]
Applying \cite[Proposition~8.2, case~(I), $k=2$]{ShiG}
in $B(x,N)$ and letting $N\to\infty$ yield
\[
 \mathbb P(z,w\in\eta_x)
 \le C\bigl(n_i^{-q}n_j^{-q}+n_j^{-2q}\bigr)
 \le Cn_i^{-q}n_j^{-q}.
\]
Consequently, $\mathbb E(Z_iZ_j)\le Cn_i^\kappa n_j^\kappa$.

Set $Y_j=n_j^{-\kappa}Z_j$ and $W_K=\sum_{j=1}^K Y_j$.
The preceding bounds imply
\[
 \mathbb E Y_j\asymp1,\qquad
 \mathbb E Y_j^2\le C(1+\log n_j),\qquad
 \mathbb E(Y_iY_j)\le C\quad(i<j).
\]
Since $n_j=2^jn_0$, we have
\[
 \mathbb E W_K\ge cK,\qquad
 \mathbb E W_K^2\le C(K^2+K\log n_0).
\]
Thus, for $K\ge1+\log n_0$, the second-moment
inequality gives $\mathbb P(W_K\ge cK)\ge c_0>0$,
with constants independent of $a,b$.
Since $\kappa\ge0$ and the sets $Q_j$ are disjoint, we have $W_K\le\sum_{j=1}^K Z_j
 \le|\eta_a\cap\eta_b|$.
Letting $K\to\infty$ yields
\begin{equation}\label{eq:infinite-overlap-positive}
 \mathbb P\bigl(|\eta_a\cap\eta_b|=\infty\bigr)\ge c_0.
\end{equation}

Finally, note that three-dimensional simple random walk has
infinitely many cut times a.s.~\cite[Theorem~1.2]{LawlerCut}.
Consequently, removing any finite initial segment
of an infinite random walk changes its loop-erasure by only
finitely many vertices.
Let $A=\{|\eta_a\cap\eta_b|=\infty\}$, and let
$\mathcal F_n=\sigma(S_j^a,S_j^b:0\le j\le n)$,
where $S^a,S^b$ are the underlying independent random walks.
The Markov property and \eqref{eq:infinite-overlap-positive} imply $\mathbb P(A\mid\mathcal F_n)
 =\mathbb P\bigl(
   |\eta_{S_n^a}\cap\eta_{S_n^b}|=\infty
   \bigr)
 \ge c_0$.
By martingale convergence,
$\mathbb P(A\mid\mathcal F_n)\to\mathbf 1_A$ a.s.
Therefore, $\mathbb P(A)=1$.

<!-- proposition-2: body_md -->
\label{thm:return}
For every $m>0$, we have
\begin{equation}\label{eq:return-bound}
 \frac{1}{2^{10}m^2\mathcal B(m)}
 \le\chi(m)\le L(m)^2,
 \qquad
 \frac{1}{32m\sqrt{\mathcal B(m)}}\le L(m)\le1+m^{-1}.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-7: body_md -->
We first prove $\chi(m)\le L(m)^2$.
For each $v\in V$ and each finite set $F\subseteq V$, apply
\eqref{eq:ancestor} with $B=\{v\}$ and $f=\mathbf 1_F$.
Letting $F\uparrow V$ and using monotone convergence gives
\[
 A(v):=\sum_{x,y\in V}\mathbb P_m(\operatorname{LCA}(x,y)=v)
 \le
 \left(\sum_{x\in V}\mathbb P_m(v\in\gamma_x^m)\right)^2.
\]
Applying \eqref{eq:MT} to $F(x,v)=\mathbb P_m(v\in\gamma_x^m)$ and to
$F(x,v)=\sum_y\mathbb P_m(\operatorname{LCA}(x,y)=v)$ gives
\[
 \sum_x\mathbb P_m(o\in\gamma_x^m)=L(m),\qquad
 A(o)=\sum_{v,y}\mathbb P_m(\operatorname{LCA}(o,y)=v)=\chi(m).
\]
By Tonelli's theorem, both identities hold in \([0,\infty]\).
It follows that $\chi(m)\le L(m)^2$.
The expected number of jumps before killing is $m^{-1}$, so $L(m)\le1+m^{-1}$.

For the lower bound, write $I_y=I_{o,y}$ in \eqref{eq:I-occupation}.
Summing \eqref{eq:I-first}--\eqref{eq:I-second} over $y$, using \eqref{eq:bubble-identity} and transitivity implies $\sum_y\mathbb E I_y=m^{-2}$ and $\sum_y\mathbb E I_y^2=4m^{-2}\mathcal B(m)$.
Cauchy--Schwarz inequality, first on finite sets and then by exhaustion, gives
\[
 \sum_y\mathbb P(I_y>0)
 \ge
 \frac{(\sum_y\mathbb E I_y)^2}{\sum_y\mathbb E I_y^2}
 =\frac{1}{4m^2\mathcal B(m)}.
\]
Summing \eqref{eq:LPS} yields the lower bound on $\chi(m)$.
The lower bound on $L(m)$ follows from $\chi(m)\le L(m)^2$.

<!-- proposition-3: body_md -->
\label{prop:long-range}
For $3/2<\alpha<2$, let $S$ be a symmetric random walk on $\mathbb Z^3$
started at $0$, with jump distribution $q(z)\asymp |z|^{-3-\alpha}$. Let $\tau_R=\inf\{n\ge0:S_n\notin B_R\}$.
Then
\begin{equation}\label{eq:long-range-growth}
 \beta_\alpha:=\liminf_{R\to\infty}
 \frac{\log\mathbb E|\operatorname{LE}(S[0,\tau_R])|}{\log R}\in\left[\frac32,\alpha\right].
\end{equation}
In particular, $\beta_\alpha\to3/2$ as $\alpha\downarrow3/2$. The same results hold with $\limsup$ in place of $\liminf$.

<!-- proof-8: body_md -->
For $r>0$, let $X^{(r)}$ be the continuous-time random walk started
at $0$ with jump rates
$q_r(z)=q(z)\mathbf1_{\{|z|\le r\}}$, killed at an
independent exponential time $\zeta$ of rate
$m_r=\sum_{|z|>r}q(z)\asymp r^{-\alpha}$.
Write $L_r=|\operatorname{LE}(X^{(r)}[0,\zeta))|$.
Then the corresponding $\mathcal B(m_r)$ is bounded by a constant multiple
of that for $S$.
By Proposition~\ref{thm:return} and the heat-kernel estimates~\cite[Theorem~1.1]{BL},  $\mathbb E L_r$ is thus at least
$c r^{3/2}$ for some $c>0$.

We claim that restricting to random walk trajectories contained in $B_{Ar\log r}$ preserves this lower bound for sufficiently large $A$. Indeed, an exponential supermartingale argument
(cf.~\cite[Proposition~2.5]{BL}) gives
\[
 \mathbb P\left(\sup_{t<\zeta}|X_t^{(r)}|
       \ge Ar\log r\right)\le Cr^{-cA}.
\]
Since the number $N_\zeta$ of jumps before killing is geometric
with parameter $m_r$, we have
$\mathbb E L_r^2\le\mathbb E(N_\zeta+1)^2
\le Cr^{2\alpha}$. Cauchy--Schwarz inequality then
implies that trajectories leaving $B_{Ar\log r}$
contribute $o(r^{3/2})$ to $\mathbb E L_r$ for sufficiently
large $A$.

We view $S$ as a continuous-time random walk with unit jump rate.
Its jumps longer than $r$ occur at rate $m_r$,
independently of the shorter jumps, so its trajectory
before the first such jump has the law of
$X^{(r)}[0,\zeta)$.
This jump exceeds $2R$ with probability $m_{2R}/m_r$.
If the preceding trajectory stays in $B_{R/2}$,
the jump exits $B_R$ without erasing its loop-erasure. Hence, by taking $r=\lfloor R/(4A\log R)\rfloor$,
we obtain
\[
 \mathbb E|\operatorname{LE}(S[0,\tau_R])|
 \ge \frac{m_{2R}}{m_r}
 \mathbb E\left[L_r;\sup_{t<\zeta}|X_t^{(r)}|<R/2\right]
 \ge \frac{cR^{3/2}}{(\log R)^{\alpha+3/2}},
\]
where the last inequality follows from the previous paragraph. Finally, since $\mathbb E|\operatorname{LE}(S[0,\tau_R])|\le1+\mathbb E\tau_R\le CR^\alpha$, we conclude.
