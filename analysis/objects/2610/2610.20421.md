Let $(X,d,\mu)$ be a locally compact, complete, unbounded, geodesic metric measure space, and let $(\mathcal{E},\mathcal{F})$ be a strongly local regular Dirichlet form on $L^2(X,\mu)$ with generator $\Delta$ and semigroup $P_t=e^{t\Delta}$. Assume that $\mu$ is Ahlfors $d_h$-regular, so that $\mu(B(x,r))\simeq r^{d_h}$ for all $x\in X$ and $r>0$. We also assume that $P_t$  admits a jointly continuous heat kernel $p_t(x,y)$ satisfying two-sided sub-Gaussian bounds
\begin{equation}\label{eq:subGauss-intro}
p_t(x,y)\simeq t^{-\frac{d_h}{d_w}}\exp\biggl(-c\Bigl(\frac{d(x,y)^{d_w}}{t}\Bigr)^{\frac1{d_w-1}}\biggr),\qquad t>0,\ x,y\in X\,,
\end{equation}
for some $d_w\ge2$. The Hunt process $(B_t)_{t\ge0}$ associated with $(\mathcal{E},\mathcal{F})$ is a diffusion, which we call Brownian motion on $X$. We assume that it is defined from every starting point. The precise hypotheses appear in Assumption~\ref{hyp:dirichlet-form}. We refer to~\cite{GrigoryanTelcs2012,GrigoryanHuLau2014} for a general account on sub-Gaussian heat kernel estimates on metric measure spaces and to~\cite{Barlow1998,Kigami2001,BarlowBass1999} for the fractal examples which provide a motivation for this work.

\begin{mainthm}\label{thm:main-existence}
Suppose that Assumption~\ref{hyp:dirichlet-form} holds, $\alpha$ satisfies~\eqref{eq:alpha-range-intro}, $\beta>0$, and $u_0\in L^\infty(X,\mu)$. Then~\eqref{eq:pam-intro} has a unique mild solution $u$ in the sense of Definition~\ref{def:mild-PAM-solution}. It is given by its Wiener chaos expansion, and for every real $p\ge2$ and every $T>0$,
\begin{equation*}
\sup_{(t,x)\in(0,T]\times X}\mathbb{E}\bigl[|u(t,x)|^p\bigr]<\infty\,.
\end{equation*}
If $u_0\ge0$ $\mu$-a.e., then $u(t,x)\ge0$ a.s.\ for every $(t,x)\in\mathbb{R}_+\times X$. If, in addition, $u_0\ne0$ in $L^\infty(X,\mu)$, then $u(t,x)>0$ a.s.\ for every such $(t,x)$.
\end{mainthm}

\label{thm:main-holder}
Let Assumption~\ref{hyp:dirichlet-form} hold and suppose that, for some $\kappa\in(0,1]$, the semigroup satisfies the weak Bakry--\'Emery condition $\mathrm{wBE}(\kappa)$ of~\cite{Besov3}: there is $C>0$ such that
\begin{equation}\label{eq:wBE-intro}
|P_tg(x)-P_tg(y)|\le C\,\frac{d(x,y)^\kappa}{t^{\kappa/d_w}}\,\|g\|_{L^\infty},\qquad t>0,\ g\in L^\infty(X,\mu),\ x,y\in X\,.
\end{equation}
Let $\alpha$ satisfy~\eqref{eq:alpha-range-intro} and let $\beta>0$. For $u_0\in L^\infty(X,\mu)$, let $u$ be the solution of~\eqref{eq:pam-intro}. Then $u$ admits a jointly continuous modification on $(0,\infty)\times X$. On every set $[T,T']\times B(x_0,R)$ with $0<T<T'<\infty$, $x_0\in X$ and $R>0$, this modification is $\gamma_t$-H\"older continuous in time and $\gamma_x$-H\"older continuous in space for any
\begin{equation*}
0<\gamma_t<\frac{(1+2\alpha)d_w-d_h}{2d_w},\qquad 0<\gamma_x<\min\Bigl\{\kappa,\frac{(1+2\alpha)d_w-d_h}{2}\Bigr\}\,.
\end{equation*}

\label{thm:main-lyapunov}
Let Assumption~\ref{hyp:dirichlet-form} hold, let $\alpha$ satisfy~\eqref{eq:alpha-range-intro}, and let $\beta>0$. For every real $p\ge1$, every nonzero nonnegative $u_0\in L^\infty(X,\mu)$ and every $x\in X$, the finite limit

\lambda_p(\beta):=\lim_{t\to\infty}\frac1t\log\mathbb{E}\bigl[u(t,x)^p\bigr]
\end{equation}

<!-- mainthm-4: body_md -->
\label{thm:main-moments}
Let Assumption~\ref{hyp:dirichlet-form} hold, let $\alpha$ satisfy~\eqref{eq:alpha-range-intro} and let $\beta>0$. Let $\lambda_p(\beta)$ be the exponents~\eqref{eq:lyapunov-intro} associated with nonzero nonnegative bounded initial conditions.
\begin{enumerate}
\item For every real $p\ge2$,
\begin{equation}\label{eq:UB-intro}
\lambda_p(\beta)\le C\,p\,\bigl(\beta^2(p-1)\bigr)^{\frac{d_w}{(1+2\alpha)d_w-d_h}}\,.
\end{equation}
\item For every real $p>1$,
\begin{equation}\label{eq:LB-intro}
\lambda_p(\beta)\ge c\,p\,\bigl(\beta^2(p-1)\bigr)^{\frac{d_w}{(1+2\alpha)d_w-d_h}}\,.
\end{equation}
\end{enumerate}
Moreover, $\lambda_1(\beta)=0$, the function $p\mapsto\lambda_p(\beta)/p$ is strictly increasing on $(1,\infty)$, and
\begin{equation}\label{eq:intermittency-intro}
\lim_{p\to\infty}\frac{\lambda_p(\beta)}p=\infty\,.
\end{equation}

Existence, uniqueness and upper moment bounds follow from Wiener chaos expansions and heat kernel estimates, extending the approach of~\cite{BCHOTW24}. For H\"older continuity, we use the weak Bakry--\'Emery estimates developed in~\cite{Besov3}. To prove existence of the Lyapunov exponents, we first construct the fundamental solution and its stochastic flow, extending the framework of Alberts, Khanin and Quastel~\cite{AlbertsKhaninQuastel2014}. We then adapt the moment comparisons of Berger, Chong and Lacoin~\cite[Lemma~5.1 and proof of Proposition~2.8]{BergerChongLacoin2023}.The argument applies to every real $p\ge1$ without spatial translation invariance. Finally, the Gaussian exponential test of~\cite[Proposition~3.20]{ChenDeyaOuyangTindel2021}, combined with confinement of the diffusion to a ball, gives lower bounds for every real $p>1$.

\label{hyp:dirichlet-form}
Let $(X,d,\mu)$ be a locally compact, complete, unbounded, geodesic metric measure space, and let $(\mathcal{E},\mathcal{F})$ be a Dirichlet form on $L^2(X,\mu)$ with associated semigroup $\{P_t\}_{t\ge0}$ and nonpositive self-adjoint generator $\Delta$. We assume:
\begin{enumerate}
\item The measure $\mu$ is Ahlfors $d_h$-regular: there are $d_h>0$ and constants $c,C>0$ such that, for all $x\in X$ and $r>0$,
\begin{equation*}
cr^{d_h}\le\mu\bigl(B(x,r)\bigr)\le Cr^{d_h}\,.
\end{equation*}
\item $\mathcal{E}$ is regular and strongly local.                                                                                              
\item The semigroup is stochastically complete: $P_t1=1$ for every $t\ge0$.
\item $\{P_t\}_{t\ge0}$ admits a heat kernel $p_t(x,y)$, jointly continuous on $(0,\infty)\times X\times X$, satisfying two-sided sub-Gaussian estimates: there are $d_w\ge2$ and constants $c,C>0$ such that, for all $t>0$ and $x,y\in X$,
\begin{equation}\label{eq:subGauss}
ct^{-d_h/d_w}\exp\biggl(-C\Bigl(\frac{d(x,y)^{d_w}}{t}\Bigr)^{\frac{1}{d_w-1}}\biggr)
\le p_{t}(x,y)\le Ct^{-d_h/d_w}\exp\biggl(-c\Bigl(\frac{d(x,y)^{d_w}}{t}\Bigr)^{\frac{1}{d_w-1}}\biggr)\,.
\end{equation}
\item The Hunt process $(\Omega,(B_t)_{t\ge0},(\mathbb{P}_x)_{x\in X})$ associated with $(\mathcal{E},\mathcal{F})$ is defined for every starting point $x\in X$, has transition density $p_t$ with respect to $\mu$, i.e.\ $\mathbb{P}_x(B_t\in A)=\int_Ap_t(x,y)\,d\mu(y)$ for every Borel set $A$, and is a strong Markov process with continuous sample paths.
\end{enumerate}

<!-- example-1: body_md -->
Assumption~\ref{hyp:dirichlet-form} is satisfied by $\mathbb{R}^d$ with Lebesgue measure and the standard Dirichlet form, Carnot groups of homogeneous dimension $Q$, and complete $d$-dimensional Riemannian manifolds with nonnegative Ricci curvature and maximal volume growth. In these cases $d_w=2$ and $d_h=d,Q,d$, respectively. It also holds on unbounded nested fractals~\cite{Barlow1998,Kigami2001}, including the infinite Sierpi\'nski gasket~\cite{BarlowPerkins1988}, for which $d_h=\log3/\log2$ and $d_w=\log5/\log2$, and on unbounded generalized Sierpi\'nski carpets~\cite{BarlowBass1999}. In the last two families $d_w>2$, and both the strictly recurrent regime $d_h<d_w$ and the regime $d_h\ge d_w$ occur.

\label{not:parameters}
Throughout the paper, unless otherwise stated, we consider a parameter $\alpha$ which satisfies
\begin{equation}\label{eq:alpha-range}
0\le\alpha<\frac{d_h}{2d_w},\qquad \alpha>\frac{d_h}{2d_w}-\frac12\,,
\end{equation}
so that
\begin{equation}\label{eq:parameter-inequalities}
0<1+2\alpha-\frac{d_h}{d_w}<1,\qquad 0<d_h-2\alpha d_w<d_w\,.
\end{equation}
Thus $\alpha=0$ is admissible if and only if $d_h<d_w$.

\label{lem:tail-exit-UB}
There exist constants $C,c>0$ such that for all $z\in X$, $s>0$ and $R\ge s^{1/d_w}$,
\begin{equation}\label{eq:oneParticleTail}
\mathbb{P}_z\bigl(d(B_s,z)\ge R\bigr)\le C\exp\Bigl(-c\Bigl(\frac{R^{d_w}}{s}\Bigr)^{\frac1{d_w-1}}\Bigr)\,.
\end{equation}
<!-- end lem-1 -->
<!-- proof-1: body_md -->
Set $\lambda=(R^{d_w}/s)^{1/(d_w-1)}\ge1$ and $\gamma=d_w/(d_w-1)$. Decomposing $\{d(\cdot,z)\ge R\}$ into the annuli $A_n=B(z,2^{n+1}R)\setminus B(z,2^nR)$, $n\ge0$, and using the upper bound in~\eqref{eq:subGauss} together with the Ahlfors regularity, we get
\begin{align*}
\mathbb{P}_z\bigl(d(B_s,z)\ge R\bigr)&\le\sum_{n\ge0}Cs^{-d_h/d_w}e^{-c2^{n\gamma}\lambda}\mu\bigl(B(z,2^{n+1}R)\bigr)
\\
&\le C\Bigl(\frac{R^{d_w}}{s}\Bigr)^{d_h/d_w}\sum_{n\ge0}2^{nd_h}e^{-c2^{n}\lambda}\,.
\end{align*}
For $\lambda\ge1$, the series is bounded by $Ce^{-c\lambda}$. Since $(R^{d_w}/s)^{d_h/d_w}=\lambda^{d_h(d_w-1)/d_w}$, the polynomial factor can be absorbed by decreasing $c$. This proves~\eqref{eq:oneParticleTail}.

\label{lem:ball-exit-time-LB}
There are constants $\delta_0\in(0,1)$ and $C>0$, depending only on the geometric data, such that, for all $x\in X$, $r>0$ and $t>0$,
\begin{equation}\label{eq:exit-LB}
\mathbb{P}_x\bigl(\tau_{B(x,r)}>t\bigr)\ge\delta_0\exp\Bigl(-C\frac{t}{r^{d_w}}\Bigr)\,.
\end{equation}
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Fix $x\in X$ and $r>0$, and write $B=B(x,r)$ and $\tau=\tau_B$. Ahlfors regularity gives volume doubling and, together with completeness, implies that all balls are relatively compact. By~\cite[Theorem 3.1 and (3.10)]{BarlowGrigoryanKumagai2012}, applied with the rate function $s\mapsto s^{d_w}$, the estimates~\eqref{eq:subGauss} imply that there exist $\varepsilon\in(0,1)$ and $c>0$, depending only on the structural constants, such that the killed heat kernel satisfies
\begin{equation*}
p_{(\varepsilon r)^{d_w}}^B(y,z)\ge c(\varepsilon r)^{-d_h},
\qquad y,z\in B(x,\varepsilon^2r)\,.
\end{equation*}
The continuity assertion in that theorem extends the bound from almost every pair $(y,z)$ to every pair. Integrating over $B(x,\varepsilon^2r)$ and using Ahlfors regularity gives a constant $\delta_0\in(0,1)$, independent of $x$ and $r$, such that
\begin{equation}\label{eq:one-step}
\inf_{y\in B(x,\varepsilon^2r)}
\mathbb{P}_y\bigl(\tau>(\varepsilon r)^{d_w},\
B_{(\varepsilon r)^{d_w}}\in B(x,\varepsilon^2r)\bigr)
\ge\delta_0\,.
\end{equation}

Since $x\in B(x,\varepsilon^2r)$, iterating the Markov property at the times $k(\varepsilon r)^{d_w}$ and using~\eqref{eq:one-step} yields, for every integer $n\ge1$,
\begin{equation*}
\mathbb{P}_x\bigl(\tau>n(\varepsilon r)^{d_w},\
B_{n(\varepsilon r)^{d_w}}\in B(x,\varepsilon^2r)\bigr)
\ge\delta_0^n\,.
\end{equation*}
For $t>0$, choose $n=\lceil t/(\varepsilon r)^{d_w}\rceil
\le t/(\varepsilon r)^{d_w}+1$. Then
\begin{align*}
\mathbb{P}_x(\tau>t)
&\ge\mathbb{P}_x\bigl(\tau>n(\varepsilon r)^{d_w}\bigr)
\ge\delta_0^n\\
&\ge\delta_0\exp\Bigl(
-\varepsilon^{-d_w}\log(1/\delta_0)\frac{t}{r^{d_w}}
\Bigr)\,,
\end{align*}
This proves~\eqref{eq:exit-LB}.

\label{def:fractional-laplacian}
For $\gamma>0$, the operator $(-\Delta)^{-\gamma}$ is defined by the spectral theorem: if $-\Delta=\int_{[0,\infty)}\lambda\,dE_\lambda$, then
\begin{equation*}
(-\Delta)^{-\gamma}=\int_{(0,\infty)}\lambda^{-\gamma}\,dE_\lambda,\qquad
\mathcal{D}\bigl((-\Delta)^{-\gamma}\bigr)=\Bigl\{f\in L^2(X,\mu):\int_{(0,\infty)}\lambda^{-2\gamma}\,d\langle E_\lambda f,f\rangle<\infty\Bigr\}\,.
\end{equation*}

\label{lem:fraclap-integral}
Let $\gamma>0$ and $f\in L^2(X,\mu)$ satisfy $\int_0^\infty s^{\gamma-1}\|P_sf\|_{L^2}\,ds<\infty$. Then $f\in\mathcal{D}((-\Delta)^{-\gamma})$ and
\begin{equation}\label{eq:fraclap-integral}
(-\Delta)^{-\gamma}f=\frac1{\Gamma(\gamma)}\int_0^\infty s^{\gamma-1}P_sf\,ds.
\end{equation}

<!-- lem-4: body_md -->
\label{lem:Schwarz-properties}
Let $0<\gamma<\frac{d_h}{2d_w}$. Then $L^1(X,\mu)\cap L^2(X,\mu)\subset\mathcal{D}((-\Delta)^{-\gamma})$. Moreover, for every $f\in L^1(X,\mu)\cap L^2(X,\mu)$ and $t>0$,
\begin{equation*}
{\left\lVert{(-\Delta)^{-\gamma}P_tf}\right\rVert}_{L^2}\le C\,t^{\gamma-\frac{d_h}{2d_w}}{\left\lVert{f}\right\rVert}_{L^1}\,.
\end{equation*}
<!-- end lem-4 -->
<!-- proof-3: body_md -->
By~\eqref{eq:heatkernel-square-bound} and Minkowski's inequality, $\|P_sf\|_{L^2}\le Cs^{-d_h/(2d_w)}\|f\|_{L^1}$. Together with $L^2$ contractivity, this yields
\begin{equation*}
\int_0^\infty s^{\gamma-1}\|P_sf\|_{L^2}\,ds
\le\frac{\|f\|_{L^2}}{\gamma}
+C\|f\|_{L^1}\int_1^\infty s^{\gamma-1-\frac{d_h}{2d_w}}\,ds<\infty\,.
\end{equation*}
Lemma~\ref{lem:fraclap-integral} proves the domain assertion. Applying the same estimate to $P_tf$ gives
\begin{equation*}
\|(-\Delta)^{-\gamma}P_tf\|_{L^2}
\le\frac{C\|f\|_{L^1}}{\Gamma(\gamma)}
\int_0^\infty s^{\gamma-1}(s+t)^{-\frac{d_h}{2d_w}}\,ds
=C_\gamma\|f\|_{L^1}t^{\gamma-\frac{d_h}{2d_w}}\,.
\end{equation*}

\label{lem:fractional-laplacian-kernel-existence-and-bound}
Let $\gamma\in(0,\frac{d_h}{d_w})$. There exist constants $c,C>0$ such that for all $x\ne y$,
\begin{equation}\label{eq:Riesz-bounds}
c\,d(x,y)^{\gamma d_w-d_h}\le G_\gamma(x,y)\le C\,d(x,y)^{\gamma d_w-d_h}\,.
\end{equation}
In particular, $G_\gamma$ is symmetric, continuous and finite off the diagonal, and $G_\gamma(x,y)\to\infty$ as $y\to x$.

<!-- lem-6: body_md -->
\label{lem:fractional-laplacian-kernel-is-a-kernel}
Let $\gamma\in(0,\frac{d_h}{2d_w})$. For every $f\in L^1(X,\mu)\cap L^\infty(X,\mu)$, the function $y\mapsto G_\gamma(x,y)f(y)$ is $\mu$-integrable for every $x\in X$, uniformly in $x$, and
\begin{equation}\label{eq:kernel-rep}
(-\Delta)^{-\gamma}f(x)=\int_XG_\gamma(x,y)f(y)\,d\mu(y)\qquad\text{for $\mu$-a.e. }x\in X\,.
\end{equation}
<!-- end lem-6 -->
<!-- proof-5: body_md -->
For $f\in L^1\cap L^\infty$ and $0<\gamma<d_h/d_w$, Tonelli's theorem, contractivity and the upper heat kernel bound give uniformly in $x$, by splitting the time integral at $1$:
\begin{equation}\label{eq:G-integrable}
\begin{aligned}
\int_XG_\gamma(x,y)|f(y)|\,d\mu(y)
&=\frac1{\Gamma(\gamma)}\int_0^\infty t^{\gamma-1}P_t|f|(x)\,dt\\
&\le\frac1{\Gamma(\gamma)}\left(
\frac{\|f\|_{L^\infty}}{\gamma}
+\frac{C\|f\|_{L^1}}{d_h/d_w-\gamma}\right).
\end{aligned}
\end{equation}
Fubini's theorem therefore gives, for every $x\in X$,
\begin{equation*}
\int_XG_\gamma(x,y)f(y)\,d\mu(y)=\frac1{\Gamma(\gamma)}\int_0^\infty t^{\gamma-1}P_tf(x)\,dt\,,
\end{equation*}
where the right-hand side converges absolutely. On the other hand, $f\in L^1\cap L^2$, so Lemmas~\ref{lem:Schwarz-properties} and~\ref{lem:fraclap-integral} give $\frac1{\Gamma(\gamma)}\int_0^Rt^{\gamma-1}P_tf\,dt\to(-\Delta)^{-\gamma}f$ in $L^2$ as $R\to\infty$, while it converges pointwise to the right-hand side above. Identifying the limits along a subsequence proves~\eqref{eq:kernel-rep}.

<!-- lem-8: body_md -->
\label{lem:Kernel-2alpha-correspondence}
Let $\gamma\in(0,\frac{d_h}{2d_w})$. For all $x,y\in X$,
\begin{equation*}
G_{2\gamma}(x,y)=\int_XG_\gamma(x,z)G_\gamma(z,y)\,d\mu(z)
\end{equation*}
(both sides being $+\infty$ when $x=y$).
<!-- end lem-8 -->
<!-- proof-7: body_md -->
By Tonelli's theorem and~\eqref{eq:chapman},
\begin{equation*}
\int_XG_\gamma(x,z)G_\gamma(z,y)\,d\mu(z)=\frac1{\Gamma(\gamma)^2}\int_0^\infty\!\!\int_0^\infty(st)^{\gamma-1}p_{s+t}(x,y)\,ds\,dt\,.
\end{equation*}
The change of variables $r=s+t$, $t=ru$ gives
\begin{align*}
\int_0^\infty\!\!\int_0^\infty(st)^{\gamma-1}p_{s+t}(x,y)\,ds\,dt&=\int_0^\infty p_r(x,y)r^{2\gamma-1}dr\int_0^1(1-u)^{\gamma-1}u^{\gamma-1}du\\
&=\Gamma(2\gamma)\mathrm{B}(\gamma,\gamma)G_{2\gamma}(x,y)\,,
\end{align*}
and the claim follows from $\mathrm{B}(\gamma,\gamma)=\Gamma(\gamma)^2/\Gamma(2\gamma)$.

<!-- lem-9: body_md -->
\label{lem:sobolev-IP}
Let $\gamma\in(0,\frac{d_h}{2d_w})$ and let $f,g:X\to\mathbb{R}$ be measurable functions such that
\begin{equation}\label{eq:Lambda-cond}
\int_X\int_X|f(x)||f(y)|G_{2\gamma}(x,y)\,d\mu(x)d\mu(y)<\infty\,,
\end{equation}
and likewise for $g$. Then $G_\gamma f(x):=\int_XG_\gamma(x,y)f(y)\,d\mu(y)$ converges absolutely for $\mu$-a.e. $x$, $G_\gamma f\in L^2(X,\mu)$, and
\begin{equation}\label{eq:sobolev-IP}
\int_XG_\gamma f\,G_\gamma g\,d\mu=\int_X\int_Xf(x)g(y)G_{2\gamma}(x,y)\,d\mu(x)d\mu(y)\,,
\end{equation}
the right-hand side being absolutely convergent. In particular the right-hand side of~\eqref{eq:sobolev-IP} is a positive semidefinite bilinear form on the space of functions satisfying~\eqref{eq:Lambda-cond}. Moreover, every $f\in L^1(X,\mu)\cap L^\infty(X,\mu)$ satisfies~\eqref{eq:Lambda-cond}, and for $f,g$ in this space,
\begin{equation}\label{eq:sobolev-IP-S}
\int_X(-\Delta)^{-\gamma}f\,(-\Delta)^{-\gamma}g\,d\mu=\int_X\int_Xf(x)g(y)G_{2\gamma}(x,y)\,d\mu(x)d\mu(y)\,.
\end{equation}
<!-- end lem-9 -->
<!-- proof-8: body_md -->
By Lemma~\ref{lem:Kernel-2alpha-correspondence} and Tonelli's theorem,
\begin{equation*}
\int_X\Bigl(\int_XG_\gamma(x,y)|f(y)|\,d\mu(y)\Bigr)^2d\mu(x)=\int_X\int_X|f(y)||f(z)|G_{2\gamma}(y,z)\,d\mu(y)d\mu(z)<\infty\,,
\end{equation*}
which shows that $G_\gamma|f|\in L^2$; in particular $G_\gamma|f|<\infty$ a.e., so $G_\gamma f$ is a.e.\ absolutely convergent and $|G_\gamma f|\le G_\gamma|f|$. The same holds for $g$, and by Cauchy--Schwarz the function $(x,y,z)\mapsto f(y)g(z)G_\gamma(x,y)G_\gamma(x,z)$ is integrable on $X^3$. Fubini's theorem and Lemma~\ref{lem:Kernel-2alpha-correspondence} then give~\eqref{eq:sobolev-IP}, and taking $f=g$ shows the positive semidefiniteness.  If $f\in L^1\cap L^\infty$, then estimate~\eqref{eq:G-integrable},
which was proved for every exponent in $(0,d_h/d_w)$, applies
with $2\gamma$ in place of $\gamma$, since $2\gamma<d_h/d_w$. Thus, $\sup_y\int_X|f(x)|G_{2\gamma}(x,y)\,d\mu(x)<\infty$, so that $\iint|f(x)||f(y)|G_{2\gamma}(x,y)\le C\|f\|_{L^1}<\infty$. Finally, \eqref{eq:sobolev-IP-S} follows from~\eqref{eq:sobolev-IP} and Lemma~\ref{lem:fractional-laplacian-kernel-is-a-kernel}.

\label{def:Sobolev-space}
Let $\gamma\in[0,\frac{d_h}{2d_w})$. If $\gamma>0$, let $\Lambda_\gamma$ be the vector space of measurable functions $f:X\to\mathbb{R}$ satisfying~\eqref{eq:Lambda-cond}, endowed with the positive semidefinite bilinear form
\begin{equation*}
\langle f,g\rangle_{\mathcal{W}^{-\gamma}}=\int_X\int_Xf(x)g(y)G_{2\gamma}(x,y)\,d\mu(x)d\mu(y)\,.
\end{equation*}
The Sobolev space $\mathcal{W}^{-\gamma}$ is the Hilbert space obtained by completing $\Lambda_\gamma$ with respect to $\langle\cdot,\cdot\rangle_{\mathcal{W}^{-\gamma}}$. By Lemma~\ref{lem:sobolev-IP}, the map $f\mapsto G_\gamma f$ extends to a linear isometry from $\mathcal{W}^{-\gamma}$ into $L^2(X,\mu)$, which we also denote by $(-\Delta)^{-\gamma}$. For $f\in L^1(X,\mu)\cap L^\infty(X,\mu)\subset\Lambda_\gamma$, this notation agrees with the spectral definition.

If $\gamma=0$, we set
\begin{equation*}
\Lambda_0=\mathcal{W}^0=L^2(X,\mu),\qquad
\langle f,g\rangle_{\mathcal{W}^0}=\int_Xf(x)g(x)\,d\mu(x),\qquad
(-\Delta)^0=I\,.
\end{equation*}

<!-- lem-10: body_md -->
\label{lem:semigroup-negative-sobolev}
Let $\gamma\in[0,\frac{d_h}{2d_w})$. For every $f\in\Lambda_\gamma$ and $t>0$, the heat-kernel integral $P_tf$ is $\mu$-a.e.\ absolutely convergent and belongs to $\Lambda_\gamma$. These operators extend uniquely to a strongly continuous contraction semigroup on $\mathcal{W}^{-\gamma}$ and for every $f\in\mathcal{W}^{-\gamma}$ and $t>0$,
\begin{equation*}
(-\Delta)^{-\gamma}P_tf=P_t(-\Delta)^{-\gamma}f,
\qquad
{\left\lVert{P_tf}\right\rVert}_{\mathcal{W}^{-\gamma}}\le{\left\lVert{f}\right\rVert}_{\mathcal{W}^{-\gamma}}\,.
\end{equation*}
<!-- end lem-10 -->
<!-- proof-9: body_md -->
For $\gamma=0$, these are the usual $L^2$ properties of the heat semigroup. Suppose that $\gamma>0$ and let $f\in\Lambda_\gamma$ and $g=|f|$. Lemma~\ref{lem:sobolev-IP} gives $G_\gamma g\in L^2$. Tonelli's theorem, \eqref{eq:RieszKernel} and~\eqref{eq:chapman} yield
\begin{equation*}
G_\gamma P_tg=P_tG_\gamma g
\end{equation*}
as an identity of nonnegative functions, initially allowing infinite values. The right-hand side belongs to $L^2$. Since $G_\gamma$ is strictly positive off the diagonal, this also implies that $P_tg$ is finite $\mu$-a.e. By Lemma~\ref{lem:Kernel-2alpha-correspondence} and Tonelli's theorem,
\begin{equation*}
\begin{aligned}
&\int_X\int_X P_tg(x)P_tg(y)G_{2\gamma}(x,y)\,d\mu(x)d\mu(y)\\
&\qquad={\left\lVert{G_\gamma P_tg}\right\rVert}_{L^2}^2
={\left\lVert{P_tG_\gamma g}\right\rVert}_{L^2}^2
\le{\left\lVert{G_\gamma g}\right\rVert}_{L^2}^2<\infty\,.
\end{aligned}
\end{equation*}
Thus $P_tg\in\Lambda_\gamma$, and $|P_tf|\le P_tg$ gives $P_tf\in\Lambda_\gamma$. Fubini's theorem now gives $G_\gamma P_tf=P_tG_\gamma f$. The isometry in Definition~\ref{def:Sobolev-space} proves contractivity, so the operators extend to $\mathcal{W}^{-\gamma}$. The semigroup property and the intertwining identity pass to the completion. Finally, strong continuity on $L^2$ gives, for every $f\in\mathcal{W}^{-\gamma}$,
\begin{equation*}
{\left\lVert{P_tf-f}\right\rVert}_{\mathcal{W}^{-\gamma}}
={\left\lVert{P_t(-\Delta)^{-\gamma}f-(-\Delta)^{-\gamma}f}\right\rVert}_{L^2}
\longrightarrow0\qquad(t\downarrow0)\,.\qedhere
\end{equation*}

\label{def:H-alpha}
We let $\mathcal{H}_\alpha=L^2(\mathbb{R}_+;\mathcal{W}^{-\alpha})$, i.e.\ the Hilbert space of (classes of) measurable maps $\varphi:\mathbb{R}_+\to\mathcal{W}^{-\alpha}$ with
\begin{equation*}
\begin{aligned}
\langle\varphi,\psi\rangle_{\mathcal{H}_\alpha}
&=\int_0^\infty\langle\varphi(t),\psi(t)\rangle_{\mathcal{W}^{-\alpha}}dt\\
&=\int_0^\infty\langle(-\Delta)^{-\alpha}\varphi(t),(-\Delta)^{-\alpha}\psi(t)\rangle_{L^2(X,\mu)}dt\,.
\end{aligned}
\end{equation*}
If $\alpha>0$ and $\varphi,\psi:\mathbb{R}_+\times X\to\mathbb{R}$ are measurable with $\varphi(t,\cdot),\psi(t,\cdot)\in\Lambda_\alpha$ for a.e.\ $t$, then
\begin{equation}\label{eq:H-inner-product}

If $\alpha=0$, then $\mathcal{H}_0=L^2(\mathbb{R}_+\times X,dt\otimes\mu)$ and

\end{equation}

\label{def:fractional-noise}
On a complete probability space $(\Omega,\mathcal{F},\mathbb{P})$, let $\{W_\alpha(\varphi):\varphi\in\mathcal{H}_\alpha\}$ be an isonormal Gaussian process on $\mathcal{H}_\alpha$, i.e.\ a centered Gaussian family such that $\varphi\mapsto W_\alpha(\varphi)$ is linear and
\begin{equation*}
\mathbb{E}\bigl[W_\alpha(\varphi)W_\alpha(\psi)\bigr]=\langle\varphi,\psi\rangle_{\mathcal{H}_\alpha},\qquad\varphi,\psi\in\mathcal{H}_\alpha\,.
\end{equation*}
We write $W_\alpha(\varphi)=\int_0^\infty\!\int_X\varphi(s,y)W_\alpha(ds,dy)$. For $t\ge0$ we let $\mathcal{F}_t$ be the $\sigma$-field generated by $\{W_\alpha(\varphi):\varphi\in\mathcal{H}_\alpha,\ \varphi(s,\cdot)=0\text{ for }s>t\}$ and the $\mathbb{P}$-null sets.

\label{def:mild-PAM-solution}
Let $u_0\in L^\infty(X,\mu)$ and $\beta>0$. A random field $u=\{u(t,x):(t,x)\in\mathbb{R}_+\times X\}$ is a (mild) solution of~\eqref{eq:pam-D} if:

\begin{equation}\label{eq:mild}

<!-- prop-1: body_md -->
\label{prop:chaos-exp-summability-implies-solution}
Let $u_0\in L^\infty(X,\mu)$ and $\beta>0$.
\begin{enumerate}
\item If $u$ is a solution of~\eqref{eq:pam-D} in the sense of Definition~\ref{def:mild-PAM-solution}, then for every $(t,x)\in\mathbb{R}_+\times X$ one has $f_k(\cdot;t,x)\in\mathcal{H}_\alpha^{\otimes k}$ for all $k\ge1$ and
\begin{equation}\label{eq:chaos-expansion}
u(t,x)=P_tu_0(x)+\sum_{k=1}^\infty I_k\bigl(f_k(\cdot;t,x)\bigr)\quad\text{in }L^2(\Omega)\,,
\end{equation}
so that, by orthogonality and~\eqref{eq:Ik-isometry},
\begin{equation}\label{eq:second-moment-chaos}
\mathbb{E}[u(t,x)^2]=(P_tu_0(x))^2+\sum_{k=1}^\infty k!\,\|f_k(\cdot;t,x)\|^2_{\mathcal{H}_\alpha^{\otimes k}}\,.
\end{equation}
\item Conversely, if $f_k(\cdot;t,x)\in\mathcal{H}_\alpha^{\otimes k}$ and
\begin{equation}\label{eq:summability}
\sum_{k=1}^\infty k!\,\|f_k(\cdot;t,x)\|^2_{\mathcal{H}_\alpha^{\otimes k}}<\infty\qquad\text{for every }(t,x)\in\mathbb{R}_+\times X\,,
\end{equation}
then the random field defined by~\eqref{eq:chaos-expansion} is a solution of~\eqref{eq:pam-D}, and it is the unique solution.
\end{enumerate}

<!-- lem-11: body_md -->
\label{lem:expansion-l2-estimates}
There exists a constant $C_\alpha>0$, depending only on the structural constants and on $\alpha$, such that for all $k\ge1$, $\rho>0$, $t>0$ and $x\in X$, $f_k(\cdot;t,x)\in\mathcal{H}^{\otimes k}$ and

{\left\lVert{f_k(\cdot;t,x)}\right\rVert}^2_{\mathcal{H}^{\otimes k}}\le\frac{\beta^{2k}}{k!}\,{\left\lVert{u_0}\right\rVert}_{L^\infty}^2\,C_\alpha^k\,\rho^{-k(1+2\alpha-\frac{d_h}{d_w})}\,e^{\rho t}\,.
\end{equation}

<!-- proof-10: body_md -->
We use the exponentially weighted recursion of~\cite[Section~3.2]{BCO24} and~\cite[Sections~3--4]{FoondunKhoshnevisan2009}. Starting from $v_0(t,x)=P_tu_0(x)$, construct the chaos terms successively by
\begin{equation*}
v_{k+1}(t,x)=\beta\int_0^t\!\int_Xp_{t-s}(x,z)v_k(s,z)\,W_\alpha(ds,dz)\,.
\end{equation*}
The estimates below justify this construction at each step. Set
\begin{equation}\label{eq:Mk-def}
M_k(t):=\sup_{x\in X}\mathbb{E}[|v_k(t,x)|^2],\qquad k\ge0\,.
\end{equation}
In particular, $M_0(t)\le\|u_0\|_{L^\infty}^2$.

Suppose that $v_k$ has been constructed. For $\alpha>0$, the stochastic-integral isometry gives
\begin{align*}
\mathbb{E}[|v_{k+1}(t,x)|^2]
&=\beta^2\int_0^t\int_{X^2}p_{t-s}(x,z)p_{t-s}(x,z')G_{2\alpha}(z,z')\\
&\qquad\times\mathbb{E}[v_k(s,z)v_k(s,z')]\,d\mu(z)d\mu(z')\,ds\,.
\end{align*}
By Cauchy--Schwarz,
\begin{equation*}
|\mathbb{E}[v_k(s,z)v_k(s,z')]|\le
\mathbb{E}[|v_k(s,z)|^2]^{1/2}\mathbb{E}[|v_k(s,z')|^2]^{1/2}\le M_k(s)\,.
\end{equation*}
Since $G_{2\alpha}\ge0$, estimate~\eqref{eq:two-point-kernel} yields
\begin{equation}\label{eq:Mk-recursion}
M_{k+1}(t)\le C\beta^2\int_0^t(t-s)^{2\alpha-d_h/d_w}M_k(s)\,ds\,.
\end{equation}
For $\alpha=0$, the same conclusion follows from
\begin{align*}
\mathbb{E}[|v_{k+1}(t,x)|^2]
&=\beta^2\int_0^t\int_Xp_{t-s}(x,z)^2\mathbb{E}[|v_k(s,z)|^2]\,d\mu(z)\,ds\\
&\le\beta^2\int_0^t p_{2(t-s)}(x,x)M_k(s)\,ds
\le C\beta^2\int_0^t(t-s)^{-d_h/d_w}M_k(s)\,ds\,
\end{align*}
see also~\cite[Lemma~3.9]{BCHOTW24}. These calculations also bound the expected squared norm of the stochastic integrand. Thus finiteness of the right-hand side of~\eqref{eq:Mk-recursion} suffices to define $v_{k+1}$ and justify the isometry.

Fix $\rho>0$ and choose $C_\alpha=C\Gamma(1+2\alpha-d_h/d_w)$, where $C$ is the constant in~\eqref{eq:Mk-recursion}. We prove by induction that
\begin{equation}\label{eq:Mk-bound}
M_k(t)\le\|u_0\|_{L^\infty}^2e^{\rho t}
\left(C_\alpha\beta^2\rho^{d_h/d_w-1-2\alpha}\right)^k\,.
\end{equation}
The case $k=0$ follows from the $L^\infty$ contraction of $P_t$. If the bound holds for $k$, then~\eqref{eq:Mk-recursion} gives
\begin{align*}
M_{k+1}(t)
&\le\|u_0\|_{L^\infty}^2e^{\rho t}
\left(C_\alpha\beta^2\rho^{d_h/d_w-1-2\alpha}\right)^k
C\beta^2\int_0^t e^{-\rho r}r^{2\alpha-d_h/d_w}\,dr\\
&\le\|u_0\|_{L^\infty}^2e^{\rho t}
\left(C_\alpha\beta^2\rho^{d_h/d_w-1-2\alpha}\right)^{k+1}\,.
\end{align*}
Here we used
\begin{equation*}
\int_0^\infty e^{-\rho r}r^{2\alpha-d_h/d_w}\,dr
=\Gamma\left(1+2\alpha-\frac{d_h}{d_w}\right)\rho^{d_h/d_w-1-2\alpha}<\infty\,,
\end{equation*}
which is finite by~\eqref{eq:alpha-range}. This proves both the recursive construction and~\eqref{eq:Mk-bound}.

Iterating the stochastic integrals and symmetrizing their ordered kernels gives $v_k(t,x)=I_k(f_k(\cdot;t,x))$, with $f_k$ as in~\eqref{eq:fk}; see~\cite[Section~1.3.1]{Nualart2006}. In particular, $f_k(\cdot;t,x)\in\mathcal{H}^{\otimes k}$, and the isometry gives
\begin{equation}\label{eq:fk-vs-Mk}
k!\,{\left\lVert{f_k(\cdot;t,x)}\right\rVert}_{\mathcal{H}^{\otimes k}}^2
=\mathbb{E}[|v_k(t,x)|^2]\le M_k(t)\,.
\end{equation}
Combining~\eqref{eq:fk-vs-Mk} with~\eqref{eq:Mk-bound} proves~\eqref{eq:key-estimate}.

\label{thm:exist-unique-wholespace}
 Let $\alpha$ satisfy~\eqref{eq:alpha-range}, $\beta>0$ and $u_0\in L^\infty(X,\mu)$. Then the equation

We use Proposition  \ref{prop:chaos-exp-summability-implies-solution}. \begin{equation*}
u_\varepsilon(t,x)=\mathbb{E}_x\Bigl[u_0(B_t)\exp\Bigl(\beta W_\alpha(g_{t,B}^\varepsilon)-\frac{\beta^2}{2}\|g_{t,B}^\varepsilon\|_{\mathcal{H}}^2\Bigr)\Bigr]\ge0\,,

\end{equation*}

\label{lem:weighted-increments}
For every $\vartheta\in(0,1)$ there exist constants $C>0$ and $c_\vartheta\ge1$ such that:
\begin{enumerate}
\item for all $t>0$ and $x,y,z\in X$,

\end{equation}
\item for all $0<s<t\le2s$ and $y,z\in X$,
\begin{equation}\label{eq:weighted-time}
|p_t(y,z)-p_s(y,z)|\le C\Bigl(\frac{t-s}{s}\Bigr)^{\vartheta}\bigl[p_{c_\vartheta t}(y,z)+p_{c_\vartheta s}(y,z)\bigr]\,.
\end{equation}
\end{enumerate}

For every $c'>0$, the lower bound in~\eqref{eq:subGauss} gives constants $C>0$ and $c''\ge1$ such that, for all $t>0$ and $x,z\in X$,
\begin{equation}\label{eq:exp-to-kernel}
t^{-\frac{d_h}{d_w}}\exp\Bigl(-c'\Bigl(\frac{d(x,z)^{d_w}}{t}\Bigr)^{\frac1{d_w-1}}\Bigr)\le C p_{c''t}(x,z)\,.
\end{equation}
Indeed, a sufficiently large time dilation makes the exponential decay in the heat kernel lower bound slower than that on the left. Fix $\vartheta\in(0,1)$. Combining the upper bound in~\eqref{eq:subGauss} with~\eqref{eq:exp-to-kernel}, and choosing $c_\vartheta\ge1$ sufficiently large, gives, for all $r>0$ and $a,z\in X$,
\begin{equation*}
p_r(a,z)^{1-\vartheta}\le C r^{\vartheta d_h/d_w}p_{c_\vartheta r}(a,z)\,.
\end{equation*}
We use this comparison for each kernel separately, together with $(A+B)^{1-\vartheta}\le A^{1-\vartheta}+B^{1-\vartheta}$ for $A,B\ge0$.

&\le C\Bigl(\frac{d(x,y)}{t^{1/d_w}}\Bigr)^{\kappa\vartheta}t^{-\vartheta d_h/d_w}\bigl[p_t(x,z)^{1-\vartheta}+p_t(y,z)^{1-\vartheta}\bigr]\,.
\end{align*}
Applying the preceding comparison with $r=t$ proves~\eqref{eq:weighted-space}.

\end{align*}
Since $t/s\le2$, the factor $(t/s)^{\vartheta d_h/d_w}$ is bounded, and~\eqref{eq:weighted-time} follows.

A_3&=\int_s^t\!\!\int_Xp_{t-\tau}(y,z)u(\tau,z)\,W_\alpha(d\tau,dz)\,.
\end{align*}
Set
\begin{equation*}
K_{T'}:=\sup_{(\tau,z)\in(0,T']\times X}\|u(\tau,z)\|_{L^p(\Omega)}<\infty\,,
\end{equation*}
which is finite by Proposition~\ref{prop:Lp-estimates}. We first record the common estimate used for the three terms. Let $h_\tau(z)$ be a deterministic measurable function on $(0,T')\times X$ such that $\int_0^{T'}\||h_\tau|\|_{\mathcal{W}^{-\alpha}}^2\,d\tau<\infty$. If $\alpha>0$, the covariance formula~\eqref{eq:H-inner-product}, Minkowski's inequality and H\"older's inequality give
\begin{align*}
\Bigl\|\|h_\tau u(\tau,\cdot)\|^2_{\mathcal{W}^{-\alpha}}\Bigr\|_{L^{p/2}(\Omega)}
&\le\int_{X^2}|h_\tau(z)h_\tau(z')|G_{2\alpha}(z,z')\\
&\qquad\times\|u(\tau,z)\|_{L^p(\Omega)}\|u(\tau,z')\|_{L^p(\Omega)}\,d\mu(z)d\mu(z')\\
&\le K_{T'}^2\bigl\||h_\tau|\bigr\|^2_{\mathcal{W}^{-\alpha}}\,.
\end{align*}
For $\alpha=0$, the same bound follows from~\eqref{eq:H0-inner-product} and
\begin{equation*}
\Bigl\|\|h_\tau u(\tau,\cdot)\|_{L^2(X,\mu)}^2\Bigr\|_{L^{p/2}(\Omega)}
\le\int_X|h_\tau(z)|^2\|u(\tau,z)\|_{L^p(\Omega)}^2\,d\mu(z)
\le K_{T'}^2\|h_\tau\|_{L^2(X,\mu)}^2\,.
\end{equation*}
In particular, $h_\tau u(\tau,\cdot)\in\Lambda_\alpha$ almost surely for almost every $\tau$, and its expected squared norm is integrable in time. Since $u$ is adapted, the stochastic integral is therefore well defined. As in~\cite[Proposition~6.7]{HamblyYang2026}, the Burkholder--Davis--Gundy inequality~\eqref{eq:BDG} and Minkowski's inequality now yield
\begin{align}
\left\|\int_0^{T'}\!\!\int_Xh_\tau(z)u(\tau,z)\,W_\alpha(d\tau,dz)\right\|_{L^p(\Omega)}^2
&\le C_p\int_0^{T'}\Bigl\|\|h_\tau u(\tau,\cdot)\|^2_{\mathcal{W}^{-\alpha}}\Bigr\|_{L^{p/2}(\Omega)}\,d\tau\label{eq:holder-bdg-first}\\
&\le C_pK_{T'}^2\int_0^{T'}\bigl\||h_\tau|\bigr\|^2_{\mathcal{W}^{-\alpha}}\,d\tau\,.\label{eq:holder-bdg-common}
\end{align}
We apply~\eqref{eq:holder-bdg-common} to the integrands defining $A_1,A_2,A_3$, extended by zero outside their respective time intervals. The following estimates also verify the required time integrability.

\emph{The term $A_1$.} Choose $\vartheta\in(0,1)$ with $\kappa\vartheta=\vartheta_x$; this is possible since $\vartheta_x<\kappa$, and then $\vartheta<((1+2\alpha)d_w-d_h)/(2\kappa)$ since $\vartheta_x<((1+2\alpha)d_w-d_h)/2$. By~\eqref{eq:W-space},
\begin{align*}
\mathbb{E}\bigl[|A_1|^p\bigr]^{2/p}
&\le C\int_0^td(x,y)^{2\kappa\vartheta}(t-\tau)^{2\alpha-\frac{d_h}{d_w}-\frac{2\kappa\vartheta}{d_w}}d\tau\\
&=C\,d(x,y)^{2\vartheta_x}\int_0^tr^{2\alpha-\frac{d_h}{d_w}-\frac{2\vartheta_x}{d_w}}dr
\le C_{T'}\,d(x,y)^{2\vartheta_x}\,,
\end{align*}
the integral being finite because $1+2\alpha-\frac{d_h}{d_w}-2\vartheta_x/d_w>0$.

\emph{The term $A_2$.} By~\eqref{eq:W-time} with $\vartheta=\vartheta_t$,
\begin{align*}
\mathbb{E}\bigl[|A_2|^p\bigr]^{2/p}&\le C\int_0^s(t-s)^{2\vartheta_t}(s-\tau)^{2\alpha-\frac{d_h}{d_w}-2\vartheta_t}d\tau\\
&=C\,(t-s)^{2\vartheta_t}\int_0^sr^{2\alpha-\frac{d_h}{d_w}-2\vartheta_t}dr\le C_{T'}\,(t-s)^{2\vartheta_t}\,,
\end{align*}
the integral being finite because $2\vartheta_t<1+2\alpha-\frac{d_h}{d_w}$.

\emph{The term $A_3$.} By the bound $\|p_{t-\tau}(y,\cdot)\|^2_{\mathcal{W}^{-\alpha}}\le C(t-\tau)^{2\alpha-\frac{d_h}{d_w}}$ recalled in Corollary~\ref{cor:W-increments},
\begin{equation*}
\mathbb{E}\bigl[|A_3|^p\bigr]^{2/p}\le C\int_s^t(t-\tau)^{2\alpha-\frac{d_h}{d_w}}d\tau=\frac{C}{1+2\alpha-\frac{d_h}{d_w}}(t-s)^{1+2\alpha-\frac{d_h}{d_w}}\le C_{T'}\,(t-s)^{2\vartheta_t}\,,
\end{equation*}
since $2\vartheta_t<1+2\alpha-\frac{d_h}{d_w}$ and $t-s\le T'$.

Collecting the bounds, raising to the power $p/2$ and using $(v_1+v_2+v_3)^{p}\le3^{p-1}(v_1^p+v_2^p+v_3^p)$ gives~\eqref{eq:increment-moments}.

For the modification, fix a compact cylinder $[T,T']\times\overline B(o,R)$ and admissible $\vartheta_t,\vartheta_x$. The metric
\begin{equation*}
\varrho\bigl((t,x),(s,y)\bigr)=|t-s|^{\vartheta_t}+d(x,y)^{\vartheta_x}
\end{equation*}
induces the product topology, and Ahlfors regularity bounds its covering numbers by $C\varepsilon^{-(1/\vartheta_t+d_h/\vartheta_x)}$. Estimate~\eqref{eq:increment-moments} holds for arbitrarily large $p$. Hence the metric-space Kolmogorov--Chentsov theorem~\cite[Theorem~1.1]{KraetschmerUrusov2023} gives a modification that is $\varrho$-H\"older of every fixed order $r<1-(1/\vartheta_t+d_h/\vartheta_x)/p$. Choosing $p$ large and $r$ close to $1$ yields every pair of orders strictly below $\vartheta_t,\vartheta_x$.

We then apply this construction to the countable exhaustion $[m^{-1},m]\times\overline B(o,m)$, $m\ge2$, and to countable sequences of admissible exponents approaching the stated limits, with $r\uparrow1$. The continuous modifications agree almost surely on countable dense subsets of their overlaps, hence everywhere on those overlaps and hence define a single modification with all the asserted local H\"older exponents.

We first obtain a uniform estimate by induction on the chaos order. Put $g_0(\cdot;t,x,y)=p_t(x,y)$ and interpret $\mathcal{H}_\alpha^{\otimes0}$ as $\mathbb{R}$. For this proof, set
\begin{equation*}
M_k(t):=\sup_{x,y\in X}k!\,{\left\lVert{g_k(\cdot;t,x,y)}\right\rVert}_{\mathcal{H}_\alpha^{\otimes k}}^2,\qquad k\ge0\,.
\end{equation*}
The heat kernel upper bound gives $M_0(t)\le Ct^{-2d_h/d_w}$. We claim that
\begin{equation}\label{eq:flow-chaos-recursion}
M_{k+1}(t)\le C\beta^2\int_0^{t/2}r^{2\alpha-d_h/d_w}M_k(t-r)\,dr\,.
\end{equation}
For $\alpha>0$, symmetry in the time-space variables and the covariance formula give
\begin{align*}
\|g_{k+1}(\cdot;t,x,y)\|_{\mathcal{H}_\alpha^{\otimes(k+1)}}^2
&=(k+1)!\int_{0<a_1<\cdots<a_{k+1}<t}
\int_{X^{2(k+1)}}g_{k+1}(a,z;t,x,y)g_{k+1}(a,z';t,x,y)\\
&\qquad\times\prod_{i=1}^{k+1}G_{2\alpha}(z_i,z_i')
\,d\mu^{\otimes(k+1)}(z)\,d\mu^{\otimes(k+1)}(z')\,da\,.
\end{align*}
The integrand is nonnegative because the heat kernels and $G_{2\alpha}$ are nonnegative. Moreover, on the ordered time simplex,
\begin{equation*}
1\le
\mathbf{1}_{\{a_{k+1}\ge t/2\}}
+\mathbf{1}_{\{a_1\le t/2\}}\,.
\end{equation*}
Inserting this inequality into the preceding integral bounds the squared norm by the sum of the same integral restricted respectively to $\{a_{k+1}\ge t/2\}$ and $\{a_1\le t/2\}$. These regions may overlap; their intersection is then counted twice, which is harmless for an upper bound.


                                                                                                                                                                      

We now multiply the squared-norm identity by $(k+1)!$. On the first region, put $r=t-a_{k+1}$. Integrating over the remaining $k$ times and their spatial variables gives the following contribution to $(k+1)!\,\|g_{k+1}(\cdot;t,x,y)\|_{\mathcal{H}_\alpha^{\otimes(k+1)}}^2$:
\begin{align*}
&\beta^2\int_0^{t/2}\int_{X^2}p_r(x,z)p_r(x,z')G_{2\alpha}(z,z')\\
&\qquad\times k!\,\bigl\langle g_k(\cdot;t-r,z,y),g_k(\cdot;t-r,z',y)\bigr\rangle_{\mathcal{H}_\alpha^{\otimes k}}
\,d\mu(z)d\mu(z')\,dr\,.
\end{align*}
By Cauchy--Schwarz inequality,
\begin{equation*}
k!\,\bigl|\bigl\langle g_k(\cdot;t-r,z,y),g_k(\cdot;t-r,z',y)\bigr\rangle_{\mathcal{H}_\alpha^{\otimes k}}\bigr|
\le M_k(t-r)\,.
\end{equation*}
Thus~\eqref{eq:two-point-kernel} bounds this contribution by
\begin{align*}
&\beta^2\int_0^{t/2}M_k(t-r)
\int_{X^2}p_r(x,z)p_r(x,z')G_{2\alpha}(z,z')\,d\mu(z)d\mu(z')\,dr\\
&\qquad\le C\beta^2\int_0^{t/2}r^{2\alpha-d_h/d_w}M_k(t-r)\,dr\,.
\end{align*}
Reversing time and both chains of heat kernels, and interchanging $x,y$, transforms the second region into the first. It therefore satisfies the same bound, proving~\eqref{eq:flow-chaos-recursion}. When $\alpha=0$, each primed spatial variable coincides with its unprimed counterpart. A similar argument which uses
\begin{equation*}
\int_Xp_r(x,z)^2\,d\mu(z)=p_{2r}(x,x)\le Cr^{-d_h/d_w}
\end{equation*}
again gives~\eqref{eq:flow-chaos-recursion}.

Now, fix $\rho>0$. Multiplying~\eqref{eq:flow-chaos-recursion} by $t^{2d_h/d_w}e^{-\rho t}$ and using $t-r\ge t/2$, we obtain
\begin{align*}
\sup_{t>0}t^{2d_h/d_w}e^{-\rho t}M_{k+1}(t)
&\le C\beta^2\left(\int_0^\infty e^{-\rho r}r^{2\alpha-d_h/d_w}\,dr\right)
\sup_{s>0}s^{2d_h/d_w}e^{-\rho s}M_k(s)\\
&\le C\beta^2\rho^{d_h/d_w-1-2\alpha}
\sup_{s>0}s^{2d_h/d_w}e^{-\rho s}M_k(s)\,.
\end{align*}
 Starting from the bound for $M_0$, an induction yields
\begin{equation*}
M_k(t)\le Ct^{-2d_h/d_w}e^{\rho t}
\left(C\beta^2\rho^{d_h/d_w-1-2\alpha}\right)^k\,.
\end{equation*}

We next handle the off-diagonal  sub-Gaussian decay. By~\eqref{eq:subGauss}, we may fix $c>0$ sufficiently small and $\sigma\ge1$ sufficiently large that
\begin{equation}\label{harck:local}

\exp\left[-c\left(\frac{d(a,b)^{d_w}}r\right)^{1/(d_w-1)}\right]

\end{equation}

\label{lem:regularized-flow}
For every  $0\le s<t$, the solution operator $U_{s,t}$ has the following properties.
\begin{enumerate}
\item There is a nonnegative random kernel $\mathcal{Z}_{s,t}(x,y)$, jointly measurable in $(\omega,x,y)$ and $\mathcal G_{s,t}$-measurable in $\omega$, such that, for every bounded nonnegative measurable $f$ and every $x\in X$,
\begin{equation}\label{eq:flow-kernel}
U_{s,t}f(x)=\int_X\mathcal{Z}_{s,t}(x,y)f(y)\,d\mu(y)\quad\text{a.s.}
\end{equation}
The same formula defines $U_{s,t}f$ for every nonnegative measurable $f$, with values in $[0,\infty]$, and satisfies
\begin{equation}\label{eq:flow-mean}
\mathbb{E}[U_{s,t}f(x)]=P_{t-s}f(x)\,.
\end{equation}
\item For every finite family of pairwise disjoint deterministic intervals $(s_j,t_j)$, the kernels $\mathcal{Z}_{s_j,t_j}$ are independent. Moreover, $\mathcal{Z}_{s,t}$ has the law of $\mathcal{Z}_{0,t-s}$, and hence $U_{s,t}$ has the law of $U_{t-s}$.
\item For deterministic times $0\le r<s<t$, every nonnegative measurable $f$ and each fixed $x\in X$,
\begin{equation}\label{eq:flow-composition}
U_{r,t}f(x)=U_{s,t}(U_{r,s}f)(x)\quad\text{a.s.}
\end{equation}
For every fixed pair $(x,y)\in X^2$, almost surely,
\begin{equation}\label{eq:flow-kernel-composition}
\mathcal{Z}_{r,t}(x,y)=\int_X\mathcal{Z}_{s,t}(x,z)\mathcal{Z}_{r,s}(z,y)\,d\mu(z)\,.
\end{equation}
For each fixed $x$, this identity also holds almost surely for $\mu$-almost every $y$.
\item The transposed kernel $\mathcal{Z}_{s,t}^*(x,y)=\mathcal{Z}_{s,t}(y,x)$ satisfies
\begin{equation}\label{eq:flow-reversal}
\mathcal{Z}_{s,t}^*\stackrel{\mathrm{law}}=\mathcal{Z}_{s,t}
\end{equation}
as random kernels modulo $\mu\otimes\mu$-null sets. Moreover, for each fixed $x\in X$, the random functions $\mathcal Z_{s,t}(x,\cdot)$ and $\mathcal Z_{s,t}(\cdot,x)$ have the same law modulo $\mu$-null sets.
\item For every finite $q\ge1$, there is $C>0$, depending on $q$, $\beta$, $\alpha$ and the geometric data, such that, for every $x,y\in X$,
\begin{equation}\label{eq:flow-kernel-moment-bound}
{\left\lVert{\mathcal Z_{s,t}(x,y)}\right\rVert}_{L^q(\Omega)}\le Ce^{C(t-s)}p_{\sigma(t-s)}(x,y)
\end{equation}
Consequently, for every nonnegative measurable $f$ and every $x\in X$,
\begin{equation}\label{eq:heat-kernel-smoothing}
{\left\lVert{U_{s,t}f(x)}\right\rVert}_{L^q(\Omega)}\le Ce^{C(t-s)}P_{\sigma(t-s)}f(x)\,.
\end{equation}
\end{enumerate}

We first construct the kernel and establish its representation, time-reversal and composition identities. We then use positivity of the solution to choose a nonnegative version. 

\emph{Construction and representation.}
For $s=0$, define
\begin{equation}\label{eq:flow-kernel-chaos}
\mathcal Z_{0,t}(x,y)=p_t(x,y)+\sum_{k\ge1}I_k(g_k(\cdot;t,x,y))\,.
\end{equation}
For general $s<t$, use the same construction with $t-s$ and noise times shifted by $s$. The kernels depend measurably on $(x,y)$ as elements of the separable Hilbert space $\mathcal{H}_\alpha^{\otimes k}$. Their Wiener integrals therefore admit jointly measurable versions that are $\mathcal G_{s,t}$-measurable in $\omega$. Estimate~\eqref{eq:flow-chaos-summability} gives absolute convergence in every $L^q(\Omega)$ and almost surely for each fixed pair $(x,y)$. Taking the sum wherever it converges absolutely, and zero elsewhere, defines a jointly measurable random kernel.

For every bounded $f$,~\eqref{eq:flow-chaos-summability} and stochastic completeness give
\begin{equation*}
\int_X\left[p_t(x,y)+\sum_{k\ge1}{\left\lVert{I_k(g_k(\cdot;t,x,y))}\right\rVert}_{L^2(\Omega)}\right]|f(y)|\,d\mu(y)
\le Ce^{Ct}{\left\lVert{f}\right\rVert}_{L^\infty}\,.
\end{equation*}
Hence the integral in space and the chaos series may be interchanged in $L^2(\Omega)$. The Wiener isometry also allows spatial integration inside each multiple integral. Integrating~\eqref{eq:flow-chaos-kernel} against $f(y)$ replaces $p_{a_1}(z_1,y)$ by $P_{a_1}f(z_1)$, recovering the solution kernel~\eqref{eq:fk}. Comparison with~\eqref{eq:chaos-expansion} proves~\eqref{eq:flow-kernel}. Taking expectation in the series defining  $\mathcal Z_{s,t}$ gives
\begin{equation}\label{eq:flow-kernel-mean}
\mathbb{E}[\mathcal Z_{s,t}(x,y)]=p_{t-s}(x,y),\qquad x,y\in X\,.
\end{equation}

\emph{Time reversal.}
For functions supported in $(s,t)\times X$, put
\begin{equation*}
(R\varphi)(a,z)=\varphi(s+t-a,z),\qquad
W_\alpha^{\leftarrow}(\varphi)=W_\alpha(R\varphi)\,.
\end{equation*}
Time reflection is an isometry of $L^2((s,t);\mathcal{W}^{-\alpha})$, so $W_\alpha^{\leftarrow}$ has the same law as $W_\alpha$ restricted to $(s,t)$. Reversing the order of the variables in~\eqref{eq:flow-chaos-kernel} and using symmetry of $p_t$ exchanges $x$ and $y$. The identity $I_k^{W_\alpha^{\leftarrow}}(h)=I_k^{W_\alpha}(R^{\otimes k}h)$, first on elementary tensors and then by the Wiener isometry, therefore gives
\begin{equation*}
\mathcal Z_{s,t}^{\leftarrow}(x,y)=\mathcal Z_{s,t}(y,x)\quad\text{a.s. for each fixed }x,y\,.
\end{equation*}
By Fubini's theorem, the identity holds almost surely modulo $\mu\otimes\mu$-null sets, proving~\eqref{eq:flow-reversal}. For each fixed $x$, Fubini's theorem also gives the identity almost surely for $\mu$-almost every $y$, which proves the equality in law of the corresponding row and column. For the related convolution identity in one dimension, see~\cite[Proposition~1.6 and Section~4.3]{GhosalLin2023}.
\emph{Composition.}
We adapt the uniqueness argument of~\cite[proof of Theorem~3.1(vii)]{AlbertsKhaninQuastel2014}. Fix $0\le r<s$ and $y\in X$. For $t>s$, the fields
\begin{equation*}
\mathcal Z_{r,t}(x,y)
\qquad\text{and}\qquad
\int_X\mathcal Z_{s,t}(x,z)\mathcal Z_{r,s}(z,y)\,d\mu(z)
\end{equation*}
satisfy the same mild equation on $(s,\infty)$, with initial condition $\mathcal Z_{r,s}(\cdot,y)$ and the same noise. For the first field, this follows by splitting its stochastic convolution at $s$; for the second, by linearity and stochastic Fubini. These operations are justified by~\eqref{eq:flow-chaos-summability} and independence on disjoint time intervals. In particular,
\begin{equation*}
\int_X
{\left\lVert{\mathcal Z_{s,t}(x,z)}\right\rVert}_{L^2(\Omega)}
{\left\lVert{\mathcal Z_{r,s}(z,y)}\right\rVert}_{L^2(\Omega)}\,d\mu(z)
\le Ce^{C(t-r)}p_{\sigma(t-r)}(x,y)\,.
\end{equation*}
Since $s-r>0$, both fields have uniformly bounded second moments for $s<t\le T$, $x\in X$, for each $T>s$. The exponentially weighted estimate in the proof of Lemma~\ref{lem:expansion-l2-estimates}, applied to their difference with zero initial condition, gives uniqueness upon choosing $\rho$ so that $C\beta^2\rho^{d_h/d_w-1-2\alpha}<1$. This proves~\eqref{eq:flow-kernel-composition} in $L^2(\Omega)$ for each fixed pair $(x,y)$.

\emph{Positivity and the choice of version.}
For each fixed $x$, the kernel $\mathcal Z_{s,t}(x,\cdot)$ is integrable almost surely by~\eqref{eq:flow-chaos-summability}. Testing~\eqref{eq:flow-kernel} against indicators from a countable algebra generating $\mathcal B(X)$, Theorem~\ref{thm:exist-unique-wholespace} and the monotone class theorem give $\mathcal Z_{s,t}(x,y)\ge0$ for $\mu$-almost every $y$, almost surely. Time reversal gives the corresponding assertion for each fixed column, and composition at the midpoint yields $\mathcal Z_{s,t}(x,y)\ge0$ almost surely for every fixed pair $(x,y)$.

Replacing the kernel by its positive part therefore preserves every fixed-endpoint random variable. Fubini's theorem preserves the integral identities, with the exceptional sets specified in the statement. Formula~\eqref{eq:flow-kernel} now defines $U_{s,t}f$ for every nonnegative measurable $f$, and Tonelli's theorem gives~\eqref{eq:flow-mean} and~\eqref{eq:flow-composition}. Dependence only on the noise in $(s,t)$ gives independence on disjoint intervals, while time-translation invariance gives stationarity; see~\cite[proof of Theorem~3.1(ii), (vi)]{AlbertsKhaninQuastel2014}. Taking the positive part also preserves time reversal.

\emph{Moment estimates.}
The chaos expansion~\eqref{eq:flow-kernel-chaos}, the triangle inequality and~\eqref{eq:flow-chaos-summability} give~\eqref{eq:flow-kernel-moment-bound} when $s=0$; stationarity gives the general case. The kernel representation~\eqref{eq:flow-kernel} and Minkowski's integral inequality then prove~\eqref{eq:heat-kernel-smoothing} for bounded nonnegative $f$. Monotone approximation extends the estimate to every nonnegative measurable $f$, with infinite right-hand sides allowed.

Fix a real $p\ge1$ and an arbitrary reference point $o\in X$. Let $\sigma$ be as in~\eqref{eq:flow-kernel-moment-bound}, and set
\begin{equation*}
f(y)=p_\sigma(o,y),\qquad a(t)=\mathbb{E}[(U_tf(o))^p]\,.
\end{equation*}
The function $f$ is bounded and positive. 

\vspace{0.4cm}

\emph{Step 1: a pointwise supermultiplicative inequality.} By~\eqref{eq:flow-kernel-moment-bound}, fix $C_0>0$ such that $\|\mathcal Z_{0,1}(x,y)\|_{L^p(\Omega)}\le C_0p_\sigma(x,y)$ for every $x,y\in X$. Minkowski's inequality then gives
\begin{equation}\label{eq:profile-smoothing}
{\left\lVert{U_1g(o)}\right\rVert}_{L^p(\Omega)}\le C_0P_\sigma g(o)
=C_0\int_Xf(y)g(y)\,d\mu(y)\,,
\end{equation}
for every nonnegative measurable $g$. If $g$ is random and independent of the noise on $[0,1]$, the same estimate holds conditionally on $g$, first for bounded truncations and then by monotone convergence for arbitrary functions. In particular, independence and the composition $U_{1+s}=U_{s,1+s}U_s$ give
\begin{equation}\label{eq:profile-moment-smoothing}
a(s+1)\le C_0^p\mathbb{E}\left[\bigl(P_\sigma U_sf(o)\bigr)^p\right],\qquad s>0\,.
\end{equation}

We now observe that one can find $L>1$ and $c>0$ such that
\begin{equation}\label{eq:profile-minorization}
p_L(z,y)\ge c f(z)f(y),\qquad z,y\in X\,.
\end{equation}
Indeed, writing $\gamma=d_w/(d_w-1)$, the triangle inequality gives
\begin{equation*}
d(z,y)^\gamma\le2^{d_w/(d_w-1)-1}\bigl(d(z,o)^{d_w/(d_w-1)}+d(o,y)^{d_w/(d_w-1)}\bigr)\,,
\end{equation*}
so that  ~\eqref{eq:profile-minorization} follows from the heat kernel estimates.

Split the flow $U_{s+L+t}$ in the intervals $[0,s], [s,s+L],[s+L,s+L+t]$ of lengths $s,L,t$ and condition on $\mathcal A=\mathcal G_{0,s}\vee\mathcal G_{s+L,s+L+t}$. The middle operator $U_{s,s+L}$ is independent of $\mathcal A$ and has mean $P_L$. Conditional Tonelli's theorem and~\eqref{eq:profile-minorization} give
\begin{align*}
\mathbb{E}[U_{s+L+t}f(o)\mid\mathcal A]
&=\bigl(U_{s+L,s+L+t}P_LU_sf\bigr)(o)\\
&\ge c\,\bigl(U_{s+L,s+L+t}f\bigr)(o)\,P_\sigma U_sf(o)\,.
\end{align*}
The two factors on the right are independent. Conditional Jensen's inequality, stationarity and~\eqref{eq:profile-moment-smoothing} therefore yield
\begin{equation*}
a(s+L+t)\ge c^pa(t)\mathbb{E}\left[\bigl(P_\sigma U_sf(o)\bigr)^p\right]
\ge(c/C_0)^pa(t)a(s+1)\,.
\end{equation*}
Consequently,
\begin{equation}\label{eq:compact-supermultiplicativity}
a(s+t+L-1)\ge(c/C_0)^pa(s)a(t),\qquad s,t>1\,.
\end{equation}
On the other hand, Jensen's inequality, the mean identity~\eqref{eq:flow-mean} and~\eqref{eq:solution-moment-bound} give
\begin{equation*}
p_{t+\sigma}(o,o)^p\le a(t)\le Ce^{Ct}\,.
\end{equation*}
Thus $\log a$ is locally bounded below, and the diagonal heat kernel lower bound gives $a(t)\ge ct^{-pd_h/d_w}$ for $t\ge1$. Applying then the continuous-parameter Fekete argument~\cite[Lemma~2, p.~993]{Matkowski1993} to the supermultiplicative function $t\mapsto(c/C_0)^pa(t-L+1)$ on $(L,\infty)$ yields the existence of
\begin{equation}\label{eq:compact-rate}
\ell_p(o):=\lim_{t\to\infty}\frac{\log a(t)}t\in[0,\infty)\,.
\end{equation}

\emph{Step 2: comparison with the constant initial condition.} We first record an inequality for independent copies $F,F'$ of a nonnegative jointly measurable random function on a measurable set $D$ with $0<\mu(D)<\infty$. If $\int_DF\,d\mu\in L^p(\Omega)$, then
\begin{equation}\label{eq:independent-pair-inequality}
\Bigl\|\int_DFF'\,d\mu\Bigr\|_{L^p(\Omega)}
\ge\mu(D)^{-1}\Bigl\|\int_DF\,d\mu\Bigr\|_{L^p(\Omega)}^2\,.
\end{equation}
Indeed, set $M=\int_DF\,d\mu$ and $M'=\int_DF'\,d\mu$. Tonelli's theorem, independence and Cauchy--Schwarz give
\begin{align*}
\mathbb{E}\Bigl[\Bigl(\int_DFF'\,d\mu\Bigr)M^{p-1}(M')^{p-1}\Bigr]
&=\int_D\bigl(\mathbb{E}[F(y)M^{p-1}]\bigr)^2\,d\mu(y)\\
&\ge\mu(D)^{-1}\bigl(\mathbb{E}[M^p]\bigr)^2\,.
\end{align*}
For $p>1$, H\"older's inequality bounds the left-hand side by $\|\int_DFF'\,d\mu\|_{L^p(\Omega)}(\mathbb{E}[M^p])^{2(p-1)/p}$, proving~\eqref{eq:independent-pair-inequality} when $\mathbb{E}[M^p]>0$ and the left-hand side of~\eqref{eq:independent-pair-inequality} is finite; the other cases are immediate. For $p=1$, omit the factors $M^{p-1}(M')^{p-1}$ in the calculation.

We now come back to the proof. The composition identity~\eqref{eq:flow-kernel-composition} gives
\begin{equation*}
\mathcal Z_{0,2t}(o,o)=\int_X\mathcal Z_{t,2t}(o,z)\mathcal Z_{0,t}(z,o)\,d\mu(z)\,.
\end{equation*}
By independence, time reversal and stationarity, the two factors are independent random functions of $z$ with the same law as $\mathcal Z_{0,t}(o,z)$, modulo $\mu$-null sets. Restricting the integral to $D$ and applying~\eqref{eq:independent-pair-inequality} yields
\begin{equation*}
\mathbb{E}[(U_t\1_D(o))^p]
\le\mu(D)^{p/2}\mathbb{E}[\mathcal Z_{0,2t}(o,o)^p]^{1/2}\,.
\end{equation*}

We then use:
\begin{equation*}
\mathcal Z_{0,T+1}(o,o)=\int_X\mathcal Z_{1,T+1}(o,z)\mathcal Z_{0,1}(z,o)\,d\mu(z)\,.
\end{equation*}
Condition on $\mathcal G_{1,T+1}$ and apply Minkowski's inequality to the independent first interval $[0,1]$. The kernel moment bound~\eqref{eq:flow-kernel-moment-bound} gives $\|\mathcal Z_{0,1}(z,o)\|_{L^p(\Omega)}\le C_0f(z)$. Therefore
\begin{equation}\label{eq:diagonal-profile-comparison}
\mathbb{E}[\mathcal Z_{0,T+1}(o,o)^p]
\le C_0^p\mathbb{E}[(U_{1,T+1}f(o))^p]=C_0^pa(T)\,.
\end{equation}
Combining the last two estimates, for $t>1/2$ we obtain
\begin{equation}\label{eq:compact-unrestricted-comparison}
\mathbb{E}[(U_t\1_D(o))^p]\le C\mu(D)^{p/2}a(2t-1)^{1/2}\,.
\end{equation}

Let $D_t=B(o,rt)$, where $r$ is a sufficiently large fixed constant. The tail bound~\eqref{eq:oneParticleTail} at time $\sigma t$ and~\eqref{eq:heat-kernel-smoothing} give
\begin{equation*}
{\left\lVert{U_t\1_{X\setminus D_t}(o)}\right\rVert}_{L^p(\Omega)}
\le C\exp\bigl((C-cr^{d_w/(d_w-1)})t\bigr)\le Ce^{-t},\qquad t\ge1\,.
\end{equation*}
Since $a(t)\le{\left\lVert{f}\right\rVert}_{L^\infty}^p\mathbb{E}[(U_t\1(o))^p]$, Ahlfors regularity,~\eqref{eq:compact-unrestricted-comparison} and Minkowski's inequality give
\begin{equation}\label{eq:averaged-flat-rate}
{\left\lVert{f}\right\rVert}_{L^\infty}^{-p}a(t)
\le\mathbb{E}[(U_t\1(o))^p]
\le Ct^{pd_h/2}a(2t-1)^{1/2}+Ce^{-pt},\qquad t\ge2\,.
\end{equation}
Taking logarithms, dividing by $t$ and using~\eqref{eq:compact-rate} and $\ell_p(o)\ge0$, we obtain
\begin{equation*}
\lim_{t\to\infty}\frac1t\log\mathbb{E}[(U_t\1(o))^p]=\ell_p(o)\,.
\end{equation*}
The reference point $o$ was arbitrary, so this limit exists at every point of $X$.

\emph{Step 3: independence of $o$ and initial condition.} Fix $x\in X$. For a sufficiently large fixed $\eta$, the two-sided heat kernel bounds give
\begin{equation*}
p_\eta(x,y)\ge c_{x,o}p_\sigma(o,y),\qquad y\in X\,.
\end{equation*}
 We deduce:
\begin{align}\label{eq:solution-pointwise-average}
\mathbb{E}[(U_{t+\eta}\1(x))^p]
&\ge\mathbb{E}[(P_\eta U_t\1(x))^p]\notag\\
&\ge c_{x,o}^p\mathbb{E}[(P_\sigma U_t\1(o))^p]\notag\\
&\ge(c_{x,o}/C_0)^p\mathbb{E}[(U_{t+1}\1(o))^p]\,.
\end{align}
Indeed, conditioning on $\mathcal G_{0,t}$, the mean identity and Jensen's inequality give the first inequality. The last follows by applying~\eqref{eq:profile-smoothing} conditionally to $U_t\1$.
The limits established in Step~2 imply $\ell_p(x)\ge\ell_p(o)$. Interchanging $x$ and $o$ proves equality. Therefore $\ell_p(o)$ is independent of $o$.

Finally let $u_0$ be bounded, nonnegative and nonzero. Choose $b>0$ and a bounded measurable set $A$ of positive measure such that $u_0\ge b\1_A$ almost everywhere. For a sufficiently large fixed $\eta$, the same heat kernel comparison as before gives
\begin{equation*}
P_\eta u_0\ge c f\,.
\end{equation*}
To see this, choose $R<\infty$ with $A\subset B(o,R)$, use $d(y,z)\le d(y,o)+R$ for $z\in A$ in the lower bound for $p_\eta(y,z)$, and integrate over $A$. Conditioning on $\mathcal G_{\eta,t+\eta}$ replaces the initial operator $U_{0,\eta}$ by $P_\eta$. Jensen's inequality, stationarity and positivity therefore yield
\begin{equation}\label{eq:solution-outer-intervals}
\mathbb{E}[(U_{t+\eta}u_0(o))^p]
\ge\mathbb{E}[(U_t(P_\eta u_0)(o))^p]
\ge c^pa(t)\,.
\end{equation}
Together with
\begin{equation*}
\mathbb{E}[(U_tu_0(o))^p]\le{\left\lVert{u_0}\right\rVert}_{L^\infty}^p\mathbb{E}[(U_t\1(o))^p]\,,
\end{equation*}
this proves that the moment growth rate for $u_0$ at $o$ is $\ell_p(o)$. Since $o$ was arbitrary and $\ell_p(o)$ is independent of $o$,~\eqref{eq:mainLimit} follows with this common value as $\lambda_p(\beta)$.

\label{lem:Gaussian-test}
Let $\beta>0$ and let $u_0\in L^\infty(X,\mu)$ be nonnegative. Let $\psi\in\Lambda_\alpha$ be real Borel and suppose that $V_{|\psi|}$ is bounded on $X$. Then, for every real $p>1$, $t>0$ and $x\in X$,
\begin{equation}\label{eq:Gaussian-test}
{\left\lVert{u(t,x)}\right\rVert}_{L^p(\Omega)}
\ge \exp\Bigl(-\frac{t{\left\lVert{\psi}\right\rVert}_{\mathcal{W}^{-\alpha}}^2}{2(p-1)}\Bigr)
\mathbb{E}_x\Bigl[u_0(B_t)\exp\Bigl(\beta\int_0^tV_\psi(B_s)\,ds\Bigr)\Bigr]\,.
\end{equation}

We adapt the argument of~\cite[Proposition~3.20]{ChenDeyaOuyangTindel2021}. Set $h_t(s,y)=\1_{(0,t)}(s)\psi(y)\in\mathcal{H}_\alpha$ and
\begin{equation*}
Z_{h_t}:=\exp\Bigl(W_\alpha(h_t)-\frac12{\left\lVert{h_t}\right\rVert}_{\mathcal{H}_\alpha}^2\Bigr),
\qquad {\left\lVert{h_t}\right\rVert}_{\mathcal{H}_\alpha}^2=t{\left\lVert{\psi}\right\rVert}_{\mathcal{W}^{-\alpha}}^2\,.
\end{equation*}
The chaos expansion $Z_{h_t}=1+\sum_{k\ge1}I_k(h_t^{\otimes k})/k!$ converges in $L^2(\Omega)$. Pairing it with~\eqref{eq:chaos-expansion} gives
\begin{equation}\label{eq:Gaussian-pairing}
\mathbb{E}[u(t,x)Z_{h_t}]
=P_tu_0(x)+\sum_{k=1}^\infty
\langle f_k(\cdot;t,x),h_t^{\otimes k}\rangle_{\mathcal{H}_\alpha^{\otimes k}}\,.
\end{equation}
The covariance pairing in each spatial variable produces $V_\psi$, with $V_\psi=\psi$ when $\alpha=0$. Restricting the symmetric integrand to ordered times cancels the factor $1/k!$ in~\eqref{eq:fk}. The Markov property used at times $t-s_k,\ldots,t-s_1$ then gives
\begin{align*}
\langle f_k(\cdot;t,x),h_t^{\otimes k}\rangle_{\mathcal{H}_\alpha^{\otimes k}}
&=\beta^k\mathbb{E}_x\Bigl[u_0(B_t)
\int_{0<s_1<\cdots<s_k<t}\prod_{j=1}^kV_\psi(B_{t-s_j})\,ds\Bigr]\\
&=\frac{\beta^k}{k!}\mathbb{E}_x\Bigl[u_0(B_t)\Bigl(\int_0^tV_\psi(B_s)\,ds\Bigr)^k\Bigr]\,.
\end{align*}
The second equality follows from symmetry of the time integrand and the change of variables $r=t-s$. To justify these integrations, apply Tonelli's theorem with $|\psi|$ in place of $\psi$. Since $|V_\psi|\le V_{|\psi|}$, the semigroup property bounds the absolute integral by
\begin{equation*}
{\left\lVert{u_0}\right\rVert}_{L^\infty}
\frac{\bigl(\beta t{\left\lVert{V_{|\psi|}}\right\rVert}_{L^\infty}\bigr)^k}{k!}\,.
\end{equation*}
This bound justifies Fubini's theorem and is summable in $k$. Summing the exponential series in~\eqref{eq:Gaussian-pairing} therefore gives
\begin{equation}\label{eq:Gaussian-FK}
\mathbb{E}[u(t,x)Z_{h_t}]
=\mathbb{E}_x\Bigl[u_0(B_t)\exp\Bigl(\beta\int_0^tV_\psi(B_s)\,ds\Bigr)\Bigr]\,.
\end{equation}
For $1<p<2$, $u(t,x)\in L^p(\Omega)$ since $u(t,x)\in L^2(\Omega)$; for $p\ge2$, this follows from Proposition~\ref{prop:Lp-estimates}. For $q=p/(p-1)$, direct Gaussian integration gives
\begin{equation*}
{\left\lVert{Z_{h_t}}\right\rVert}_{L^q(\Omega)}
=\exp\Bigl(\frac{t{\left\lVert{\psi}\right\rVert}_{\mathcal{W}^{-\alpha}}^2}{2(p-1)}\Bigr)\,.
\end{equation*}
Since the right-hand side of~\eqref{eq:Gaussian-FK} is nonnegative, H\"older's inequality proves~\eqref{eq:Gaussian-test}.

\label{thm:moment-LB}
Let $\alpha$ satisfy~\eqref{eq:alpha-range}, let $\beta>0$, and let $\lambda_p(\beta)$ be the exponents of Theorem~\ref{thm:lyapunov-main}. There exists $c>0$, depending only on $\alpha$ and the structural constants, such that, for every real $p>1$,
\begin{equation}\label{eq:LB-global}
\lambda_p(\beta)\ge c\,p\,\bigl(\beta^2(p-1)\bigr)^{\frac{d_w}{(1+2\alpha)d_w-d_h}}\,.
\end{equation}

<!-- proof-21: body_md -->
By Theorem~\ref{thm:lyapunov-main}, it suffices to take $u_0\equiv1$. Fix $p>1$, $x\in X$ and $r>0$. We first estimate the potential and covariance norm of the indicator of $B(x,r)$. We claim there are constants $c,C>0$, depending only on $\alpha$ and the structural constants, such that
\begin{equation}\label{eq:ball-test-estimates}
\begin{aligned}
\inf_{z\in B(x,r)}V_{\1_{B(x,r)}}(z)&\ge c r^{2\alpha d_w},
\qquad \sup_{z\in X}V_{\1_{B(x,r)}}(z)\le C r^{2\alpha d_w},\\
{\left\lVert{\1_{B(x,r)}}\right\rVert}_{\mathcal{W}^{-\alpha}}^2&\le C r^{d_h+2\alpha d_w}\,.
\end{aligned}
\end{equation}
For $\alpha=0$, these follow from $V_{\1_{B(x,r)}}=\1_{B(x,r)}$ and Ahlfors regularity. Suppose $\alpha>0$. The lower bound in~\eqref{eq:Riesz-bounds}, the inequality $d(z,y)<2r$ for $z,y\in B(x,r)$, and $\mu(B(x,r))\ge cr^{d_h}$ give the first estimate. For the upper bound, split the integral at distance $r$ from $z$. On $B(x,r)\setminus B(z,r)$ it is at most $Cr^{2\alpha d_w-d_h}\mu(B(x,r))$. The integral over $B(z,r)$ is bounded by summing over dyadic annuli:
\begin{equation*}
\int_{B(z,r)}d(z,y)^{2\alpha d_w-d_h}\,d\mu(y)
\le C r^{2\alpha d_w}\sum_{j\ge0}2^{-2j\alpha d_w}
\le C_\alpha r^{2\alpha d_w}\,.
\end{equation*}
Finally, ${\left\lVert{\1_{B(x,r)}}\right\rVert}_{\mathcal{W}^{-\alpha}}^2=\int_{B(x,r)}V_{\1_{B(x,r)}}\,d\mu$ gives the last estimate in~\eqref{eq:ball-test-estimates}. In particular, $\1_{B(x,r)}\in\Lambda_\alpha$ and its potential is bounded.

Apply Lemma~\ref{lem:Gaussian-test} with $\psi=\eta\1_{B(x,r)}$, $\eta>0$, and restrict the diffusion expectation to $\{\tau_{B(x,r)}>t\}$. Estimates~\eqref{eq:ball-test-estimates} and~\eqref{eq:exit-LB} give
\begin{equation}\label{eq:white-noise-test-ball}
{\left\lVert{u(t,x)}\right\rVert}_{L^p(\Omega)}
\ge\delta_0\exp\left(
t\left[c\beta\eta r^{2\alpha d_w}-Cr^{-d_w}
-\frac{C\eta^2r^{d_h+2\alpha d_w}}{2(p-1)}\right]\right)\,.
\end{equation}
The quadratic expression in $\eta$ in the logarithmic rate is maximized at $\eta=(c/C)\beta(p-1)r^{-d_h}$. Set $A=(c^2/(2C))\beta^2(p-1)$. Taking logarithmic rates gives
\begin{equation}\label{eq:LB-eps}
\lambda_p(\beta)\ge p\bigl(A r^{2\alpha d_w-d_h}-Cr^{-d_w}\bigr)\,.
\end{equation}
By~\eqref{eq:parameter-inequalities}, $0<d_h-2\alpha d_w<d_w$. The expression in parentheses is maximized at
\begin{equation*}
r_*=
\Bigl(\frac{d_wC}{(d_h-2\alpha d_w)A}\Bigr)^{\frac1{(1+2\alpha)d_w-d_h}}\,,
\end{equation*}
and substitution gives~\eqref{eq:LB-global}, since every radius is admissible.

\label{cor:lyapunov-bounds}
For every real $p\ge2$,
\begin{equation}\label{eq:lyapunov-two-sided}
cp\bigl(\beta^2(p-1)\bigr)^{\frac{d_w}{(1+2\alpha)d_w-d_h}}
\le\lambda_p(\beta)
\le Cp\bigl(\beta^2(p-1)\bigr)^{\frac{d_w}{(1+2\alpha)d_w-d_h}}\,,
\end{equation}
where $c,C>0$ depend only on $\alpha$ and the geometric data. The lower bound also holds for every real $p>1$. For every nonzero nonnegative $u_0\in L^\infty(X,\mu)$ and every $x\in X$,
\begin{equation}\label{eq:first-moment-exponent}
\lambda_1(\beta):=\lim_{t\to\infty}\frac1t\log\mathbb{E}[u(t,x)]=0\,,
\end{equation}
the function $p\mapsto\lambda_p(\beta)/p$ is strictly increasing on $(1,\infty)$, and
\begin{equation}\label{eq:strict-intermittency}
\lambda_1(\beta)<\frac{\lambda_2(\beta)}2<\frac{\lambda_3(\beta)}3<\cdots,
\qquad
\lim_{p\to\infty}\frac{\lambda_p(\beta)}p=+\infty\,.
\end{equation}
Thus the solution is fully intermittent at integer moment orders in the sense of~\cite[Section~1.2]{TindelAarhus2026}.

By the independence of the initial condition in Theorem~\ref{thm:lyapunov-main}, we may take $u_0\equiv1$ when computing $\lambda_1(\beta)$. \le\mathbb{E}[u(t,x)]^{(r-p)/(r-1)}

$\lambda_p(\beta)\le (p-1)\lambda_r(\beta)/(r-1)$.

\end{equation*}

W_\alpha\bigl((s,y)\mapsto\varphi(5^{-n}s,2^{-n}y)\bigr)\,.
\end{equation*}
For $\alpha>0$, the change of variables in the covariance~\eqref{eq:H-inner-product} contributes a factor $5^n3^{2n}5^{2\alpha n}3^{-n}=5^{n(1+2\alpha)}3^n$. The prefactor therefore gives
\begin{equation*}
\mathbb{E}[\widetilde W_\alpha(\varphi)\widetilde W_\alpha(\psi)]
=\langle\varphi,\psi\rangle_{\mathcal{H}_\alpha}\,.
\end{equation*}
At $\alpha=0$, the same identity follows directly from~\eqref{eq:H0-inner-product}, since $ds\,d\mu(y)$ scales by $5^n3^n$. By completion, $\widetilde W_\alpha$ is an isonormal Gaussian process on $\mathcal{H}_\alpha$ with the same law as $W_\alpha$.

Set $v(t,x)=u^{\beta,u_0}(5^nt,2^nx)$. Applying~\eqref{eq:sg-heat-scaling} to the deterministic and stochastic terms of~\eqref{eq:mild} gives
\begin{equation*}
v(t,x)=P_t[u_0(2^n\cdot)](x)
+\beta\Bigl(\frac{5^{1+2\alpha}}3\Bigr)^{n/2}
\int_0^t\!\!\int_Xp_{t-s}(x,y)v(s,y)\,\widetilde W_\alpha(ds,dy)\,.
\end{equation*}
The stochastic change of variables holds first for elementary predictable integrands and then by the isometry. Uniqueness in Theorem~\ref{thm:exist-unique-wholespace}, or equivalently the chaos expansion, proves~\eqref{eq:sg-solution-scaling}. Taking $u_0\equiv1$, passing to $p$-th moments and dividing their logarithms by $t$ proves~\eqref{eq:sg-lyapunov-scaling} by Theorem~\ref{thm:lyapunov-main}.

\label{rem:lyapunov-scaling}
For every fixed $p\ge1$, the function
\begin{equation*}
s\longmapsto
\exp\biggl(-\frac{2d_w}{(1+2\alpha)d_w-d_h}\,s\biggr)\lambda_p(e^s)
\end{equation*}
is periodic with period $\frac12\bigl((1+2\alpha)\log5-\log3\bigr)$. This follows from~\eqref{eq:sg-lyapunov-scaling} with $n=1$, since $d_w=\log5/\log2$ and $d_h=\log3/\log2$. Thus the dependence on $\beta$ may include a factor periodic in $\log\beta$ multiplying the power $\beta^{2d_w/((1+2\alpha)d_w-d_h)}$.
