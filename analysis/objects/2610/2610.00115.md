<!-- theorem-1: body_md -->
\label{thm:classification}
For positive integers $m,n$, the following are equivalent.
\begin{enumerate}[label=\textup{(\roman*)}]
\item There is an integral domain $D$ such that
\[
                   \dim D[X]=m,
                   \qquad \dim\operatorname{Int}(D)=n.
\]
\item Either $(m,n)=(1,1)$, or $(m,n)=(2,2)$, or
$m\geq3$ and $n\geq m-1$.
\end{enumerate}
Moreover, for every finite field $k$ and every pair in \textup{(ii)} with
$m\geq3$, the realizing domain may be chosen to be a local domain
$(D_{m,n},\mathfrak m_{m,n})$ satisfying
\[
 D_{m,n}/\mathfrak m_{m,n}\cong k,
 \qquad \dim D_{m,n}=m-2.
\]

\label{thm:base-row}
For every finite field $k=\mathbb F_q$ and every integer $n\geq3$, there is
a one-dimensional local domain $(D_{3,n},\mathfrak m_{3,n})$ such that
\[
  D_{3,n}/\mathfrak m_{3,n}\cong k,\qquad
  \dim D_{3,n}[X]=3,\qquad
  \dim\operatorname{Int}(D_{3,n})=n.
\]

<!-- proposition-1: body_md -->
\label{prop:universal-pair-bounds}
Let $D$ be a domain and suppose that
\[
                         m=\dim D[X]
\]
is finite.  Put $n=\dim\operatorname{Int}(D)$, $d=\dim D$, and $v=\dim_v(D)$, allowing
$n$ and $v$ initially to be infinite.  The following inequalities are
interpreted in $\mathbb Z_{\geq0}\cup\{\infty\}$:
\[
 d+1\leq m,\qquad d+1\leq n,\qquad m-1\leq n\leq v+1.
\]
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For every finite prime chain
$\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_\ell$ in $D$, the chain
\[
 \mathfrak p_0[X]\subsetneq\cdots\subsetneq
 \mathfrak p_\ell[X]\subsetneq(\mathfrak p_\ell,X)
\]
has length $\ell+1$ in $D[X]$.  Since $m$ is finite, this proves
$d\leq m-1$; in particular $d$ is finite and $d+1\leq m$.

For any $a\in D$, evaluation is a surjection
\[
 \operatorname{ev}_a:\operatorname{Int}(D)\longrightarrow D,\qquad f\longmapsto f(a).
\]
Its kernel
\[
 \mathcal Q_a=(X-a)K[X]\cap\operatorname{Int}(D)
\]
is a nonzero prime, since it contains $X-a$.  Taking inverse images of a
length-$d$ prime chain in $D$ chosen to begin with $(0)$, and prepending
$(0)\subsetneq\mathcal Q_a$, gives $d+1\leq n$.

The inequality $m-1\leq n$ is Cahen's universal estimate
\cite[Corollaire~1.2]{Cahen1990Dimension}; see also
\cite[Corollary~1.11]{FontanaIzelgueKabbajTartarone1997}.  For the last
inequality, first note that
\[
              \operatorname{Frac}(\operatorname{Int}(D))=\operatorname{Frac}(D[X])=K(X).
\]
Thus every valuation overring of $\operatorname{Int}(D)$ is also a valuation overring of
$D[X]$.
Consequently
\[
 \dim\operatorname{Int}(D)\leq\dim_v\operatorname{Int}(D)
 \leq\dim_v(D[X])=\dim_v(D)+1;
\]
see also \cite{FontanaIzelgueKabbajTartarone1997,
Chabert2010Overview}.

<!-- proposition-2: body_md -->
\label{prop:small-first-coordinate}
If $\dim D[X]=1$, then $\dim\operatorname{Int}(D)=1$.  If $\dim D[X]=2$, then
$\dim\operatorname{Int}(D)=2$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
If $\dim D[X]=1$, \cref{prop:universal-pair-bounds} gives $\dim D=0$.
A zero-dimensional domain is a field, so $\operatorname{Int}(D)=D[X]$ and both dimensions
are $1$.

Suppose now that $\dim D[X]=2$.  The same proposition gives $\dim D\leq1$.
The dimension cannot be zero: a zero-dimensional domain is a field, and then
$\dim D[X]=1$.  Hence $\dim D=1$.  We claim that $\dim_v(D)=1$.  If not,
some valuation overring of $D$ has a chain of primes of length at least two.  Localizing that
valuation ring at the top of a length-two subchain gives a valuation
overring $W\subseteq K$ with
\begin{equation}\label{eq:valuation-three-primes}
                  (0)\subsetneq Q\subsetneq M,
\end{equation}
where $M$ is the maximal ideal of $W$.

Every nonzero prime $P$ of a valuation overring of $D$ contracts nontrivially
to $D$.  Indeed, if $P\cap D=(0)$, then $W_P$ contains the inverse of every
nonzero element of $D$.  Hence
\[
                     K=\operatorname{Frac}(D)\subseteq W_P\subseteq K,
\]
so $W_P=K$, contradicting $0\neq PW_P$.
It follows from \eqref{eq:valuation-three-primes} and $\dim D=1$ that
\[
                 Q\cap D=M\cap D=\mathfrak p
\]
for a maximal ideal $\mathfrak p$ of $D$.

Choose $\xi\in M\setminus Q$ and write $\xi=a/b$ with $a,b\in D$ and
$b\neq0$.  If $b\notin\mathfrak p$, then $b$ is a unit of $W$, whereas
\[
                a=b\xi\in M\cap D=\mathfrak p=Q\cap D.
\]
This would imply $\xi\in Q$, a contradiction.  Thus
$b\in\mathfrak p=Q\cap D$.  Since $Q$ is an ideal of $W$, it follows that
$a=b\xi\in Q\cap D=\mathfrak p$.  Hence
\[
                         a,b\in\mathfrak p,\qquad b\in Q.
\]
Consider the upper to zero
\[
 H=(X-\xi)K[X]\cap D[X]
   =\ker\bigl(D[X]\xrightarrow{\,X\mapsto\xi\,}K\bigr).
\]
It is nonzero because $bX-a\in H$.  We next prove
\begin{equation}\label{eq:H-below-pX}
                              H\subseteq\mathfrak p[X].
\end{equation}
Suppose that $f=\sum_i d_iX^i\in H$ has a coefficient outside
$\mathfrak p$, and let $j$ be the least index with
$d_j\notin\mathfrak p$.  Reduce the equation $f(\xi)=0$ in the valuation
domain $W$ modulo $Q$.  The terms with $i<j$ vanish.  Since
$\xi\notin Q$, cancellation of $\overline\xi^{\,j}$ in the domain $W/Q$
gives
\[
 \overline d_j+
 \sum_{i>j}\overline d_i\,\overline\xi^{\,i-j}=0.
\]
The first term is a unit of the valuation domain $W/Q$, because
$d_j\notin M\cap D$, while the sum belongs to its maximal ideal $M/Q$,
because $\xi\in M$.  This is impossible and proves
\eqref{eq:H-below-pX}.

All inclusions in
\[
 (0)\subsetneq H\subsetneq\mathfrak p[X]
       \subsetneq(\mathfrak p,X)
\]
are strict prime inclusions: a nonzero constant from $\mathfrak p$ is not
in $H$, and $X\notin\mathfrak p[X]$.  This length-three chain contradicts
$\dim D[X]=2$.  Hence $\dim_v(D)=1$.  Finally,
\cref{prop:universal-pair-bounds} yields
\[
                  2=\dim D+1\leq\dim\operatorname{Int}(D)
                  \leq\dim_v(D)+1=2.
\]
The conclusion follows.

<!-- lemma-1: body_md -->
\label{lem:two-point-polynomial}
Let $(A,M)$ be a local domain with quotient field $K$, residue field $k$,
and $\operatorname{Spec}(A)=\{(0),M\}$.  Then $\dim A[X]\leq3$.  Suppose in addition
that $A\subseteq V\subseteq K$ and that a homomorphism $\rho:V\to L$ to a
field $L$ satisfies
\[
              \ker(\rho|_A)=M,\qquad \rho(A)=k.
\]
Here $k=A/M$ is identified with the subfield $\rho(A)$ of $L$.
If there are $\theta\in V$ and $0\neq c\in M$ such that $c\theta\in A$ and
$\rho(\theta)$ is transcendental over $k$, then $\dim A[X]=3$.
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
In a prime chain of $A[X]$, the primes contracting to $(0)$ form an initial
block.  Localization at $A\setminus\{0\}$ identifies that block, with all
strict inclusions preserved, with a chain in the principal ideal domain
$K[X]$, so it has at most one edge.  The primes contracting to $M$ form the
terminal block.  Each contains $M[X]$, and quotienting by $M[X]$ identifies
that block, again with strict inclusions preserved, with a chain in $k[X]$;
it too has at most one edge.  There is at most one transition between the
blocks, proving $\dim A[X]\leq3$.

For the second assertion, put
\[
                 Q=(X-\theta)K[X]\cap A[X].
\]
This is prime and nonzero because $c(X-\theta)\in Q$.  If
$f=\sum_i a_iX^i\in Q$, then $\sum_i a_i\theta^i=0$.  Applying $\rho$ gives
\[
                 \sum_i\rho(a_i)\rho(\theta)^i=0.
\]
Since $\rho(a_i)\in k$ and $\rho(\theta)$ is transcendental over $k$, every
$a_i$ lies in $M$.  Hence $Q\subseteq M[X]$, and the inclusion is strict
because the nonzero constant $c$ is not in $Q$.  Therefore
\[
                  (0)\subsetneq Q\subsetneq M[X]\subsetneq(M,X)
\]
is a prime chain of length $3$.

<!-- lemma-2: body_md -->
\label{lem:expansion-residue}
Every $g\in S$ has a unique finite expansion
\[
                 g=\sum_{\gamma\geq0}a_\gamma t^\gamma,
                 \qquad a_\gamma\in R.
\]
Moreover, $E$ is a ring, $P$ is an ideal of $E$, and
\[
              \rho(E)=R,\qquad \ker(\rho|_E)=P.
\]
In particular, $E/P\cong R$, every nonzero element of $P$ has positive
$\nu$-value, and $D_{3,n}/P\cong k$.
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
Consider a finite $L$-linear relation among distinct rational powers of $t$.
Choose $\ell$ so that all exponents become integral multiples of $1/\ell!$.
The relation becomes a polynomial relation over $L$ for the transcendental
element $t^{1/\ell!}$; hence every coefficient is zero.  Thus the rational
powers of $t$ are $L$-linearly independent.  Combining equal exponents gives
the asserted unique expansion for elements of $S$.

The set $I$ is an ideal of $V$, and $S_+$ is an ideal of $S$.  Therefore
\[
 (S+I)(S+I)\subseteq S+I,
 \qquad
 (S+I)(S_++I)\subseteq S_++I.
\]
This proves that $E$ is a ring and that $P$ is an ideal of $E$.

For a finite expansion $g=\sum a_\gamma t^\gamma$, reduction modulo the
maximal ideal of $V$ gives $\rho(g)=a_0$.  Hence $\rho(E)=R$ and
$P\subseteq\ker(\rho|_E)$.  Conversely, suppose $g+z\in E$ has residue
zero, where $g\in S$ and $z\in I$.  Since $\rho(z)=0$, the constant
coefficient $a_0$ of $g$ is zero.  Thus $g\in S_+$ and $g+z\in P$.
This proves $\ker(\rho|_E)=P$.  The remaining assertions follow by taking
quotients and then restricting $\rho$ to $D_{3,n}=k+P$.  Notice here that
$k\cap P=(0)$, since nonzero elements of $k$ have value zero while nonzero
elements of $P$ have positive value.

<!-- lemma-3: body_md -->
\label{lem:local}
The ring $D_{3,n}$ is local with maximal ideal $\mathfrak m_{3,n}=P$.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Since $D_{3,n}/P\cong k$, the ideal $P$ is maximal.  Let
$a+p\in D_{3,n}\setminus P$,
where $a\in k^\times$ and $p\in P$.  If $p=0$, the element is plainly a
unit.  Otherwise put $\delta=\nu(p)>0$ and choose $h\geq0$ with
$(h+1)\delta\geq1$.  The finite geometric identity in $\Omega$ gives
\[
 \frac1{a+p}
  =\frac1a\sum_{j=0}^{h}\left(-\frac pa\right)^j
   +\frac{(-1)^{h+1}p^{h+1}}{a^{h+1}(a+p)}.
\]
The finite sum lies in $k+P=D_{3,n}$.  Since $\nu(a+p)=0$, the last summand
has value $(h+1)\delta\geq1$ and consequently lies in
$I\subseteq D_{3,n}$.  Thus $(a+p)^{-1}\in D_{3,n}$, so every element
outside $P$ is a unit.

<!-- lemma-4: body_md -->
\label{lem:spectrum}
One has
\[
          \operatorname{Frac}(D_{3,n})=\Omega,
          \qquad \operatorname{Spec}(D_{3,n})=\{(0),\mathfrak m_{3,n}\}.
\]
In particular, $\dim D_{3,n}=1$.
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
The element $t$ has value $1$, so $t\in I\subseteq D_{3,n}$.  For every
$x\in V$ one has $tx\in I\subseteq D_{3,n}$, and therefore
$x=(tx)/t\in\operatorname{Frac}(D_{3,n})$.  Hence
\[
  \Omega=\operatorname{Frac}(V)\subseteq\operatorname{Frac}(D_{3,n})\subseteq\Omega,
\]
which proves $\operatorname{Frac}(D_{3,n})=\Omega$.

Let $0\neq f\in\mathfrak m_{3,n}$ and let $g\in\mathfrak m_{3,n}$.  The case $g=0$ is
immediate.  If
$g\neq0$, then $\nu(f),\nu(g)>0$, so there is an integer $N\geq1$ such that
\[
                   N\nu(g)-\nu(f)\geq1.
\]
It follows that $g^N/f\in I\subseteq D_{3,n}$, and hence
$g^N\in fD_{3,n}$.  We have proved
$\mathfrak m_{3,n}\subseteq\sqrt{fD_{3,n}}$.  The reverse inclusion follows from
$fD_{3,n}\subseteq\mathfrak m_{3,n}$ and the primality of $\mathfrak m_{3,n}$.  Thus
\begin{equation}\label{eq:radical}
                         \sqrt{fD_{3,n}}=\mathfrak m_{3,n}
                         \qquad(0\neq f\in\mathfrak m_{3,n}).
\end{equation}
Every nonzero proper prime ideal of $D_{3,n}$ contains such an $f$, and
\eqref{eq:radical} forces that prime to equal $\mathfrak m_{3,n}$.  Since $D_{3,n}$
is a domain, its only remaining prime is $(0)$.

<!-- proposition-4: body_md -->
\label{prop:weak-localization}
Let $(T,\mathfrak n)$ be a local domain with quotient field $K_T$ and residue
field $T/\mathfrak n\cong\mathbb F_q$.  Define
\begin{equation}\label{eq:CT}
 \mathcal C(T)=\{c\in K_T:\text{there is }N\geq0\text{ such that }
                  cy^N\in T\text{ for every }y\in\mathfrak n\}.
\end{equation}
Then $\mathcal C(T)$ is a subring of $K_T$.  If $F=X^q-X$, then
\[
             \operatorname{Int}(T)\subseteq \mathcal C(T)[X],
             \qquad
             \operatorname{Int}(T)[1/F]=\mathcal C(T)[X,1/F].
\]
<!-- end proposition-4 -->
<!-- proof-8: body_md -->
Suppose $N_c$ and $N_d$ witness membership in $\mathcal C(T)$ for $c$ and
$d$, respectively.  For every $y\in\mathfrak n$,
\begin{align*}
 (c+d)y^{N_c+N_d}
   &=(cy^{N_c})y^{N_d}+(dy^{N_d})y^{N_c}\in T,\\
 cd\,y^{N_c+N_d}&=(cy^{N_c})(dy^{N_d})\in T.
\end{align*}
The same exponent that works for $c$ also works for $-c$, and
$0,1\in\mathcal C(T)$.  Thus $\mathcal C(T)$ is a ring.

Let $f=\sum_{j=0}^e c_jX^j\in\operatorname{Int}(T)$.  If $e=0$, then
$c_0=f(0)\in T\subseteq\mathcal C(T)$.  Assume $e>0$ and fix
$0\neq y\in\mathfrak n$.  Evaluate $f$ at
\[
                         y,y^2,\ldots,y^{e+1}.
\]
These points are distinct: if $1\leq i<j\leq e+1$, then
\[
               y^j-y^i=y^i(y^{j-i}-1)\neq0,
\]
because $y^{j-i}-1$ is a unit of the local domain $T$.  The determinant of
the associated Vandermonde matrix is
\begin{equation}\label{eq:vandermonde}
 \Delta_y
  =\prod_{1\leq i<j\leq e+1}(y^j-y^i)
  =y^{M_e}\prod_{1\leq i<j\leq e+1}(y^{j-i}-1),
\end{equation}
where
\[
 M_e=\sum_{i=1}^{e}i(e+1-i)
   =\frac{e(e+1)(e+2)}6.
\]
The values $f(y^i)$ lie in $T$.  Cramer's rule, applied over $K_T$, gives
$\Delta_yc_j\in T$ for every $j$.  The second product on the right of
\eqref{eq:vandermonde} is a unit of $T$, so
\[
                         c_jy^{M_e}\in T
                         \qquad(0\leq j\leq e).
\]
The same assertion is automatic when $y=0$.  The integer $M_e$ is
independent of $y$, hence every $c_j$ belongs to $\mathcal C(T)$.  This proves
$\operatorname{Int}(T)\subseteq\mathcal C(T)[X]$ and therefore
\[
                       \operatorname{Int}(T)[1/F]\subseteq\mathcal C(T)[X,1/F].
\]

Conversely, reduction modulo $\mathfrak n$ shows
$F(a)=a^q-a\in\mathfrak n$ for every $a\in T$.  If $c\in\mathcal C(T)$ and $N$
witnesses \eqref{eq:CT}, then
\[
                    cF(a)^N\in T\qquad(a\in T).
\]
Thus $cF^N\in\operatorname{Int}(T)$ and
$c=(cF^N)/F^N\in\operatorname{Int}(T)[1/F]$.  Since $X\in\operatorname{Int}(T)$ and $F$ is invertible
in the localization, this yields
$\mathcal C(T)[X,1/F]\subseteq\operatorname{Int}(T)[1/F]$ and proves equality.

<!-- proposition-5: body_md -->
\label{prop:C-chain}
One has $C_{3,n}\subseteq V$ and $\rho(C_{3,n})=R$.  Put
\[
 \mathfrak q_0=\ker(\rho|_{C_{3,n}}),\qquad
 \mathfrak q_j=(\rho|_{C_{3,n}})^{-1}((u_1,\ldots,u_j))\quad(1\leq j\leq n-2).
\]
These are prime ideals and form a strict chain
\begin{equation}\label{eq:C-chain}
 (0)\subsetneq\mathfrak q_0\subsetneq\mathfrak q_1\subsetneq\cdots
       \subsetneq\mathfrak q_{n-2}
\end{equation}
of length $n-1$, with $C_{3,n}/\mathfrak q_{n-2}\cong k$.
<!-- end proposition-5 -->
<!-- proof-9: body_md -->
Suppose $c\in\Omega$ has $\nu(c)<0$.  For an arbitrary fixed $N\geq0$,
choose $\varepsilon\in\mathbb{Q}_{>0}$ so small that
\[
                         \nu(c)+N\varepsilon<0;
\]
for $N=0$ any positive $\varepsilon$ works.  Then
$y=t^\varepsilon\in S_+\subseteq\mathfrak m_{3,n}$, but
$cy^N\notin V$ and therefore $cy^N\notin D_{3,n}$.  No exponent can witness
membership in $C_{3,n}$, so $C_{3,n}\subseteq V$.

Let $c\in C_{3,n}$.  The assertion $\rho(c)\in R$ is clear for $c=0$ or
$\nu(c)>0$.  Suppose $\nu(c)=0$, and choose an exponent $N$ witnessing
\eqref{eq:C}.  If $N=0$, then $c\in D_{3,n}$ and
$\rho(c)\in k\subseteq R$.  If $N>0$, choose
$\varepsilon\in\mathbb{Q}_{>0}$ with $N\varepsilon<1$ and put
\[
                    d=ct^{N\varepsilon}\in D_{3,n},
                    \qquad \alpha=N\varepsilon.
\]
Then $\nu(d)=\alpha\in(0,1)$, so $d\in P$.  Write
$d=g+z$ with $g\in S_+$ and $z\in I$.  Since
$\nu(z)\geq1>\alpha=\nu(d)$, one has $\nu(g)=\alpha$.  By
\cref{lem:expansion-residue}, write
\[
                     g=\sum_{\gamma>0}a_\gamma t^\gamma,
                     \qquad a_\gamma\in R.
\]
The least occurring exponent is $\alpha$.  Hence
\[
       \rho(t^{-\alpha}d)=\rho(t^{-\alpha}g)=a_\alpha\in R,
\]
because $\nu(t^{-\alpha}z)\geq1-\alpha>0$.  Since
$t^{-\alpha}d=c$, this proves $\rho(C_{3,n})\subseteq R$.

Conversely, take $a\in R$ and $y\in\mathfrak m_{3,n}=P$.  Since
$a\in S\subseteq E$ and $P$ is an ideal of $E$, one has
$ay\in P\subseteq D_{3,n}$.  Thus exponent $1$ witnesses
$a\in C_{3,n}$.  The map $\rho$ is the identity on $R$, so
$R\subseteq\rho(C_{3,n})$.

It follows that $C_{3,n}/\mathfrak q_0\cong R$.  Taking inverse images of
\[
 (0)\subsetneq(u_1)\subsetneq\cdots
       \subsetneq(u_1,\ldots,u_{n-2})
\]
gives all strict inclusions in \eqref{eq:C-chain} after $\mathfrak q_0$; when $n=3$,
the displayed chain means simply $(0)\subsetneq(u_1)$.  The remaining
inclusion $(0)\subsetneq\mathfrak q_0$ is strict because
$0\neq t\in\mathfrak q_0$.  The final quotient is
$R/(u_1,\ldots,u_{n-2})\cong k$.

<!-- lemma-5: body_md -->
\label{lem:trdeg-bound}
Let $k$ be a field and let $T$ be a domain containing $k$.  If
$M=\operatorname{Frac}(T)$ has finite transcendence degree over $k$, then
\begin{equation}\label{eq:trdeg-bound}
                         \dim T\leq\operatorname{trdeg}_k M.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-10: body_md -->
We recall the valuation argument.  Krull's valuation theorem says that every
finite chain of prime ideals of a domain is dominated by a chain of the same
length in a valuation overring of its quotient field; equivalently,
$\dim T\leq\dim_v T$ \cite[Chapter~VI]{BourbakiCA}
(see also \cite{Gilmer1972}).  One obtains the chain form by successively
choosing valuations that dominate the relevant local quotient domains and
composing them through their residue fields.  Let $W$ be such a valuation
overring, with
valuation
\[
                         \omega:M^\times\to\Gamma.
\]
Because $k\subseteq W$ and $k^\times\subseteq W^\times$, the valuation is
trivial on $k$.

Suppose $W$ contains a strict chain of $\ell+1$ primes.  Prime ideals of a
valuation ring correspond, in reverse order, to convex subgroups of its
value group.  Thus $\Gamma$ has a strict chain of $\ell+1$ convex subgroups.

of adjacent subgroups.  These elements are $\mathbb{Q}$-linearly independent.  To
see this, in a nontrivial integral relation take the term belonging to the
largest convex layer.  All lower-layer terms lie in the preceding convex
subgroup, whereas a nonzero multiple of the chosen largest-layer element
does not: quotients by convex subgroups are ordered and hence torsion-free.
This is a contradiction.

$x_1,\ldots,x_\ell$ have distinct values, by the $\mathbb{Q}$-linear independence of
the $\gamma_i$.  Since coefficients from $k^\times$ have value zero, a
nonzero polynomial in the $x_i$ has a unique term of least value and cannot
vanish.  Hence $x_1,\ldots,x_\ell$ are algebraically independent over $k$.
Therefore
\[
                         \ell\leq\operatorname{trdeg}_k M.
\]
This bounds the length of every prime chain in every valuation overring of
$T$, and \eqref{eq:trdeg-bound} follows.

\label{lem:pullback-suspension}
Let $A$ be a domain with quotient field $K$, let
$A[X]\subseteq B\subseteq K[X]$ be a domain of finite dimension $d$, and
let
\[
 W=K[Y]_{(Y)},\qquad M=YW.
\]
Write $\lambda:W[X]\to K[X]$ for coefficientwise reduction modulo $M$ and
put $B^\sharp=\lambda^{-1}(B)$.  Then
\[
                         \dim B^\sharp=d+1.
\]
<!-- end lemma-6 -->
<!-- proof-12: body_md -->
Put $J=M[X]$, an ideal of both $W[X]$ and $B^\sharp$.  We have
\[
 B^\sharp/J\cong B,
 \qquad
 B^\sharp[1/Y]=K(Y)[X].
\]
For the second equality, observe that $Yg\in J\subseteq B^\sharp$ for
every $g\in W[X]$, and that $W[1/Y]=K(Y)$.  Moreover, for a prime
$P$ of $B^\sharp$, one has
\[
                  J\subseteq P\quad\Longleftrightarrow\quad Y\in P.
\]
Only the reverse implication requires comment.  If $Y\in P$ and
$h=Yg\in J$, then
$h^2=Y(Yg^2)$ with $Yg^2\in J\subseteq B^\sharp$; hence $h^2\in P$ and
$h\in P$.

It follows that chains of primes not containing $J$---equivalently, chains
whose primes avoid the powers of $Y$---are carried, with all strict inclusions
preserved, to chains in the one-dimensional PID $K(Y)[X]$.
Thus such a chain has at most one edge.  Primes containing $J$ correspond,
in an order-preserving way, to primes of $B$.

We also need the following compatibility at the crossing.  No nonzero prime
$P$ of $B^\sharp$ that does not contain $J$ can be contained in $J$.  Indeed,
localize $P$ in $K(Y)[X]$ and choose a nonzero polynomial in the resulting
prime.  After multiplying by an element of $K(Y)^\times$, it may be taken
to be a polynomial $g\in W[X]$ having at least one unit coefficient.  Its
reduction $\overline g\in K[X]$ is therefore nonzero.  Since
$K=\operatorname{Frac}(A)$, there is $0\neq a\in A$ which clears the finitely many
denominators of the coefficients of $\overline g$.  Then
\[
 ag\in P B^\sharp[1/Y]\cap B^\sharp,
 \qquad \lambda(ag)=a\overline g\in A[X]\subseteq B,
 \qquad \lambda(ag)\neq0.
\]
Since $Y\notin P$, contraction from the localization gives
\[
             P B^\sharp[1/Y]\cap B^\sharp=P.
\]
Thus $ag\in P\setminus J$, as claimed.

Consider now an arbitrary chain in $\operatorname{Spec}(B^\sharp)$.  If none of its primes
contains $J$, it has length at most $1$.  If every prime contains $J$, its
length is at most $d$.  Otherwise let $P$ be the last prime not containing
$J$ and $P'$ the first prime containing $J$.  If the initial block has
no edge, the transition $P\subsetneq P'$ followed by a chain in $B$ gives
at most $1+d$ edges.  If that block has one edge, then $P\neq(0)$.  The
image $P'/J$ is nonzero, for otherwise $P'=J$, contradicting the preceding
paragraph.  A chain in the domain $B$ beginning at the nonzero prime
$P'/J$ has at most $d-1$ further edges, since it can be preceded by the
strict inclusion $(0)\subsetneq P'/J$.  The total is again at most
\[
                        1+1+(d-1)=d+1.
\]
Hence $\dim B^\sharp\leq d+1$.

Conversely, choose a chain
$(0)=\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_d$ of length $d$
in $B$.  Its inverse images form a chain beginning with $J$, and prepending
$(0)\subsetneq J$ gives a chain of length $d+1$ in $B^\sharp$.

<!-- proposition-6: body_md -->
\label{prop:one-step-suspension}
Let $(A,\mathfrak a)$ be a local domain with
$0\neq\mathfrak a$, quotient field $K$, and residue field $k=\mathbb F_q$.
Let $W=K[Y]_{(Y)}$, let $\pi:W\to K$ be reduction modulo $YW$, and define
\[
                       A^\sharp=\pi^{-1}(A).
\]
Then $A^\sharp$ is a local domain with residue field $k$ and quotient field
$K(Y)$, and
\begin{align}
 \dim A^\sharp&=\dim A+1,
       \label{eq:D-suspension}\\
 \dim A^\sharp[X]&=\dim A[X]+1,
       \label{eq:DX-suspension}\\
 \dim\operatorname{Int}(A^\sharp)&=\dim\operatorname{Int}(A)+1.
       \label{eq:Int-suspension}
\end{align}
Here the dimensions in the last two formulas are assumed finite.
<!-- end proposition-6 -->
<!-- proof-13: body_md -->
The ring $A^\sharp$ is local with maximal ideal
$\mathfrak a^\sharp=\pi^{-1}(\mathfrak a)$: an element outside this ideal
has residue a unit of $A$, and its inverse in $W$ has residue in $A$ as
well.  The residue field is therefore $k$.  Also
$YW\subseteq A^\sharp$, and multiplication by $Y$ shows
$W\subseteq\operatorname{Frac}(A^\sharp)$; hence
$\operatorname{Frac}(A^\sharp)=K(Y)$.

Every nonzero prime of $A^\sharp$ contains $M=YW$.  Indeed, let
$0\neq x$ belong to such a prime $P$.  If $x\notin M$, then $x$ is a unit
of $W$, and $y/x\in M\subseteq A^\sharp$ for every $y\in M$; hence
$y=x(y/x)\in P$.  If $x\in M$, write $e=\operatorname{ord}_Y(x)>0$.
For $0\neq y\in M$, choose $N$ with
$N\operatorname{ord}_Y(y)>e$.  Then $y^N/x\in M\subseteq A^\sharp$, so
$y^N\in P$ and therefore $y\in P$.  Thus $M\subseteq P$ in both cases.
Primes containing $M$ are precisely the inverse images of primes of $A$.
Consequently
\[
 \operatorname{Spec}(A^\sharp)=\{(0)\}\cup
 \{\pi^{-1}(\mathfrak p):\mathfrak p\in\operatorname{Spec}(A)\},
\]
and \eqref{eq:D-suspension} follows.

We shall need one coefficient consequence.  If
$c\in\mathcal C(A^\sharp)$ has witness $N$, choose
$0\neq a\in\mathfrak a$ and a lift $y\in A^\sharp$ of $a$.  Since
$\pi(y)=a\neq0$, the element $y$ is a unit of $W$.  The inclusion
$cy^N\in A^\sharp\subseteq W$ therefore implies $c\in W$.  Hence
\begin{equation}\label{eq:C-sharp-contained}
                         \mathcal C(A^\sharp)\subseteq W.
\end{equation}

Coefficientwise reduction identifies
\[
 A^\sharp[X]=\pi_X^{-1}(A[X])\subseteq W[X].
\]
Formula \eqref{eq:DX-suspension} is therefore
\cref{lem:pullback-suspension} applied to $B=A[X]$.

For the integer-valued polynomial ring we first claim the exact pullback
identity
\begin{equation}\label{eq:Int-pullback-identity}
 \operatorname{Int}(A^\sharp)=\pi_X^{-1}(\operatorname{Int}(A))\subseteq W[X].
\end{equation}
By the Vandermonde coefficient argument in
\cref{prop:weak-localization}, every coefficient of an element of
$\operatorname{Int}(A^\sharp)$ lies in $\mathcal C(A^\sharp)$, which is contained in $W$
by \eqref{eq:C-sharp-contained}.  Such a polynomial can therefore be reduced
coefficientwise.  Evaluating it at arbitrary lifts of elements of $A$
shows that its reduction is in $\operatorname{Int}(A)$.  This proves one inclusion in
\eqref{eq:Int-pullback-identity}.  Conversely, if $f\in W[X]$ and
$\pi_X(f)\in\operatorname{Int}(A)$, then, for every $x\in A^\sharp$,
\[
       \pi(f(x))=\pi_X(f)(\pi(x))\in A.
\]
Thus $f(x)\in\pi^{-1}(A)=A^\sharp$, proving the reverse inclusion.
Finally, $A[X]\subseteq\operatorname{Int}(A)\subseteq K[X]$ and
$\operatorname{Frac}(\operatorname{Int}(A))=K(X)$, so applying
\cref{lem:pullback-suspension} with $B=\operatorname{Int}(A)$ to
\eqref{eq:Int-pullback-identity} gives \eqref{eq:Int-suspension}.
