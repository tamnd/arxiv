The type $\mathrm{A}_n$ Hall--Littlewood polynomial indexed by a rectangular partition of length $n$ satisfies the following dual Jacobi--Trudi formula:
\label{thm-type-a}
\[
P_{(k^n)}(X;t) = \sum_{\substack{y\in\mathbb{Z}^k\\|y|=0}} \det_{1\le i,j\le k}(t^{k\binom{y_i}{2}+iy_i} e_{n-i+j-ky_i}(X))\,.
\]
<!-- end thm-4 -->
<!-- proof-1: body_md -->
Begin by expanding the Hall--Littlewood polynomial of type $\mathrm{A}_n$,
\[
P_{(k^n)}(X;t) = \frac{1}{[n]_t!}\sum_{\sigma\in S_n} \sigma.\!\left( \prod_{i=1}^n x_i^{\lambda_i}
 \prod_{i<j} \frac{x_i-t x_j}{x_i-x_j}
\right).
\]
When $\lambda = (k^n)$, then note that $\prod_{i=1}^nx_i^k$ is invariant under the action of the symmetric group. We get
\[
P_{(k^n)}(X;t) =\frac{\prod_{i=1}^n x_i^{k}}{[n]_t!}\sum_{\sigma\in S_n} \sigma.\!\left(
 \prod_{i<j} \frac{x_i-t x_j}{x_i-x_j}
\right)= \frac{\prod_{i=1}^nx_i^k}{[n]_t!} [n]_t!= \prod_{i=1}^n x_i^k = s_{(k^n)}(X)\,, 
\]
where we have used \cite[p. 207, (1.4)]{Macdonald1995}.

Now consider the right-hand side of the identity we aim to show, and factor out the power of $t$ to get
\begin{equation}
    \label{eq:power of t A}
\sum_{\substack{y\in\mathbb{Z}^k\\|y|=0}} \det_{1\le i,j\le k}(t^{k\binom{y_i}{2}+iy_i} e_{n-i+j-ky_i}(X))
= 
\sum_{\substack{y\in\mathbb{Z}^k\\|y|=0}} 
\Big( \prod_{i=1}^k t^{k\binom{y_i}{2}+iy_i} \Big)
\det_{1\le i,j\le k}(e_{n-i+j-ky_i}(X))\,.
\end{equation}
If $y\neq 0$ and $|y|=0$ then there exists an $y_i<0$. We then have that
$$n-i+r-ky_i\geq n-i+r+k>n,$$
for every column index $1\leq r\le k$. Thus the $i$-th row of the matrix in the determinant is zero and subsequently the determinant vanishes and only $y=0$ survives.
An application of the 
dual Jacobi--Trudi identity \eqref{eq: dual JT type A} gives that the right-hand side is $s_{(n^k)'}(X)$, as claimed.

Let $n,k\in\mathbb{Z}_{>0}$. The (multiparameter) Hall--Littlewood polynomial of type $\mathrm{BC}_n$ indexed by a rectangular partition satisfies
    \begin{multline*}
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,s_1,s_2)= \frac{\sqrt{t}^{\binom{n}{2}}}{2}
    \prod_{i=1}^nx_i^{n-1}\prod_{1\leq i<j\leq n}\frac{1}{(1-x_ix_j)(x_i-x_j)}\\\times\det_{1\le i,j\le n}\left(\frac{(1-s_1x_i)(1-s_2x_i)}{1-x_i^2}
    ((\sqrt{t})^{j-1}x_i^{j-1-k}+
    (\sqrt{t})^{-j+1}x_i^{-j+1-k})\right.\\\left.-\frac{(s_1-x_i)(s_2-x_i)}{1-x_i^2}
    ((\sqrt{t})^{j-1}x_i^{-j+1+k}
    +(\sqrt{t})^{-j+1}x_i^{j-1+k})\right).
\end{multline*}

The underlying set of $\mathbb{Z}_2\wr S_n$ is $\mathbb{Z}_2^n\times S_n $.
Split the sum in the expression \eqref{eq:intro-bc-definition}
over the wreath product accordingly,
\begin{multline*}
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,s_1,s_2) = \frac{1}{ v_{(k^n)}(t)} \\
    \times \sum_{\alpha \in \{\pm 1\}^n}\left[\sum_{\sigma\in S_n}\sigma.\!\left(\prod_{i=1}^n\bar{x}_i^{k}\frac{(1-s_1x_i)(1-s_2x_i)}{1-x_i^2}\prod_{1\leq i<j\leq n}\frac{(tx_i-x_j)(1-tx_ix_j)}{(x_i-x_j)(1-x_ix_j)}\right)\right]_{X=X^\alpha},
\end{multline*}
where we denote $X^\alpha=(x_1^{\alpha_1},\dots,x_n^{\alpha_n})$ for $\alpha\in \{\pm1\}^n$ and note that $v_{(k^n)}(t)= \prod_{i=1}^n \frac{1-t^i}{1-t}$.
We rewrite the sum over $S_n$ by factoring out terms that are invariant by permutations. The sum becomes
\begin{multline*}
    \prod_{i=1}^n\bar{x}_i^{k}\frac{(1-s_1x_i)(1-s_2x_i)}{1-x_i^2}\prod_{1\leq i<j\leq n}\frac{1-tx_ix_j}{1-x_ix_j}\sum_{\sigma\in S_n}\sigma.\!\left(\prod_{1\leq i<j\leq n}\frac{tx_i-x_j}{x_i-x_j}\right)\\
    =
    \prod_{i=1}^n\bar{x}_i^{k}\frac{(1-s_1x_i)(1-s_2x_i)}{1-x_i^2}\prod_{1\leq i<j\leq n}\frac{1-tx_ix_j}{1-x_ix_j}v_{(k^n)}(t) \,,
\end{multline*}
where we have used~\cite[p.~207, (1.4)]{Macdonald1995}. The right-most term cancels the prefactor $1/v_{(k^n)}(t)$ and we obtain the following formula for the Hall--Littlewood polynomial:
\[
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,s_1,s_2) = \sum_{\alpha \in \{\pm 1\}^n}\left[\prod_{i=1}^n\bar{x}_i^{k}\frac{(1-s_1x_i)(1-s_2x_i)}{1-x_i^2}\prod_{1\leq i<j\leq n}\frac{1-tx_ix_j}{1-x_ix_j} \right]_{X=X^\alpha}.
\]


Now observe that for any $\alpha\in \{\pm1\}^n$ we have
\begin{equation*}
    \left[\prod_{1\leq i<j\leq n}\frac{1}{(x_i-x_j)(1-x_ix_j)}\right]_{X=X^\alpha}=\frac{\prod_{i\in\alpha^-}x_i^{2(n-1)}}{\prod_{1\leq i<j\leq n}(x_i-x_j)(1-x_ix_j)}\,,
\end{equation*}
where $\alpha^-=\{i\in [n]\mid \alpha_i=-1\}$. Dividing and multiplying the Hall--Littlewood polynomial by this expression gives
\begin{align*}
    P_{k^n}^{\mathrm{BC}_n}(X;&t,s_1,s_2)=\frac{1}{\prod_{1\leq i<j\leq n}(x_i-x_j)(1-x_ix_j)}\\
    &\times \sum_{\alpha \in \{\pm 1\}^n}
    \prod_{i\in\alpha^-}x_i^{2(n-1)}

    \frac{(1-s_1x_i)(1-s_2x_i)}{1-x_i^2}
    \prod_{1\leq i<j\leq n}(1-tx_ix_j)(x_i-x_j)\right]_{X=X^\alpha}.
\end{align*}
In the last equality we have used the following simplification,
\begin{multline*}
    \prod_{i\in\alpha^-} x_i^{2(n-1)}
\left[
\prod_{i=1}^n\bar{x}_i^{k}
\right]_{X=X^\alpha}

\label{lem:powery}
Let $A = (a_{ij}(x))_{1\le i,j\le k}$ be a matrix of univariate functions.
Let $R\subseteq\mathbb{Z}^k$ be a finite $S_k$-invariant subset of $\mathbb{Z}^k$.
Then,
   \[
    \sum_{y \in R} \det_{1 \le i,j \le k}
    (
    t^{jy_i} a_{i,j}(y_i)
    ) =  \sum_{y \in R} t^{\sum_{j=1}^k j y_j} \det_{1 \le i,j \le k} (
    a_{i,j}(y_j)).
    \]
<!-- end lem-2 -->
<!-- proof-3: body_md -->
Expand the left-hand side according to the Leibniz formula of the determinant,
    \[
     \sum_{y \in R} \sum_{\sigma \in S_k}  \mathrm{sgn}(\sigma)  \left( \prod_{j=1}^k
    t^{jy_{\sigma(j)}} a_{\sigma(j),j}(y_{\sigma(j)})
    \right).
    \]
Let $\mathrm{sort}(R)=\{y\in R\mid y_1\geq \dots\geq y_k\}$ and define an action of $S_k$ on $R$ by $(\mu\cdot y)_i=y_{\mu(i)}$. Observe that any $y\in R$ can be written as $y=\mu\cdot x$ for some $x\in \mathrm{sort}(R) $ and some $\mu\in S_k$. Conversely from one $x\in \mathrm{sort}(R) $ we can obtain the same $y\in R$ multiple times by acting with different $\mu$'s. The number of times this can happen is given by the size of the stabilizer of $x$ under this action and denoted by $m(x)=\vert\mathbf{Stab}(x)\vert$. Using this observation, we can rewrite the above sum as
    \[
     \sum_{x \in \mathrm{sort}(R)}\frac{1}{m(x)}\sum_{\mu\in S_k} \sum_{\sigma \in S_k}  \mathrm{sgn}(\sigma)  \left( \prod_{j=1}^k
    t^{jx_{\mu(\sigma(j))}} a_{\sigma(j),j}(x_{\mu(\sigma(j))})
    \right).
    \]
Summing over all $\rho=\mu\circ \sigma$ instead of $\mu\in S_k$, we obtain
    \[
     \sum_{x \in \mathrm{sort}(R)}\frac{1}{m(x)}\sum_{\rho\in S_k} \sum_{\sigma \in S_k}  \mathrm{sgn}(\sigma)  \left( \prod_{j=1}^k
    t^{jx_{\rho(j)}} a_{\sigma(j),j}(x_{\rho(j)})
    \right).
    \]
The inner sum now clearly is a determinant and this is equal to
\begin{multline*}
     \sum_{x \in \mathrm{sort}(R)}\frac{1}{m(x)}\sum_{\rho\in S_k} \prod_{j=1}^k
    t^{jx_{\rho(j)}} \det_{1\leq i,j\leq k}\left( a_{i,j}(x_{\rho(j)})
    \right)
    =\sum_{y\in R}\prod_{j=1}^k
    t^{jy_j} \det_{1\leq i,j\leq k}\left( a_{i,j}(y_j)
    \right)\\=\sum_{y \in R} t^{\sum_{j=1}^k j y_j} \det_{1 \le i,j \le k} (
    a_{i,j}(y_j)),
\end{multline*}
    where the first equality again follows from the definition of the group action. This proves the lemma.

\label{lem: Noodle-Doodle expansion first form}
Let $M$ be an $n\times(n-m)$ matrix, let $f_1,\ldots,f_n$ be functions
on $V$, and let $p_s,p_t$ be functions on $E$. Then
\begin{multline}\label{eq: to show pure graph}
 \det_{\substack{1\leq i\leq n\\ j\in E}}
 \bigl(M ~\mathbin{\raisebox{-.18em}{$\vdots$}}~ p_t(j)f_i(t_{\mathcal N}(j))
                         +p_s(j)f_i(s_{\mathcal N}(j))\bigr)
 \\
 =\sum_{A\in\mathcal A(\Lambda_{m,k,h^+,h^-})}
   \prod_{j\in E\setminus E_A}p_t(j)
   \prod_{j\in E_A}p_s(j)
   \det_{\substack{1\leq i\leq n\\ j\in E}}
     \bigl(M ~\mathbin{\raisebox{-.18em}{$\vdots$}}~ f_i(t_A(j))\bigr).
\end{multline}

<!-- proof-4: body_md -->
Using multilinearity in columns $n-m+1,\ldots,n$, we expand the
left-hand side by choosing one endpoint of each edge. Orienting each edge
towards the chosen endpoint identifies the resulting terms with
orientations of the graph.

By the antisymmetry of the determinant, an orientation with a vertex of
indegree two gives a zero summand, since two columns are equal. In every
remaining orientation, a component with $r$ vertices has $r-1$ vertices
of indegree one and precisely one of indegree zero. Its edges are directed
away from the latter vertex. Hence it suffices to sum over
$\mathcal A(\Lambda_{m,k,h^+,h^-})$.

Changing the root from $c_j$ to $z_j$ reverses precisely the edges between
these two vertices. Thus the edges in $E_A$ contribute $p_s$, and all other
edges contribute $p_t$, proving~\eqref{eq: to show pure graph}.

\label{lem: column order}
For an orientation $A\in\mathcal A(\Lambda_{m,k,h^+,h^-})$ with roots $z_0,\ldots,z_{\bar k}$, set
\[
 b_j=\bigl|\{i:i>j\text{ and }z_i>z_j\}\bigr|.
\]
Then
\begin{equation}\label{eq: sign of column order}
 (-1)^{\operatorname{inv}(t_A(1),\ldots,t_A(m))}=(-1)^{\binom{\bar k+1}{2}
       +\sum_{j=0}^{\bar k}\bigl((K-1)y_j
            +2|h^++h^-|[y_j<0]+b_j\bigr)},
\end{equation}
where $\operatorname{inv}(a_1,\dots,a_n)\coloneqq|\{(i,j)\in[1,n]^2\mid i>j\textrm{ and }a_i<a_j\}|$ is the number of \emph{inversions} of the array $(a_1,\dots,a_n)$.
<!-- end lem-5 -->
<!-- proof-6: body_md -->
We first count inversions in a list containing all
vertices, and then subtract the inversions involving roots.

Consider $(z_0,\ldots,z_{\bar k}\mid t_A(1),\ldots,t_A(m))$.
The vertical bar merely separates the roots from the targets. The roots
remain in component order, and the targets remain in edge-label order.
Every vertex occurs exactly once: roots have indegree zero, and every
other vertex is the target of precisely one edge.

For the natural orientation, this list is $ (c_0,\ldots,c_{\bar k}\mid
 t_{\mathcal N}(1),\ldots,t_{\mathcal N}(m))$,
which is the increasing list of all vertices. Its inversion number is
therefore zero.

Now move a root along the path from its center to its prescribed position.
Suppose one step moves the current root from $u$ to an adjacent vertex $v$,
across edge $\ell$. Before this move, the edge points from $u$ to $v$, so
the root entry is $u$ and the entry corresponding to edge $\ell$ is $v$.
Afterwards, the root entry is $v$, while the reversed edge has target $u$.
Thus this move exchanges exactly two entries of the list.
All other entries remain unchanged.

Exchanging two entries reverses the parity of the inversion number, even
when their positions are not adjacent. Consequently, after moving all roots,
\begin{equation}\label{eq: sign of augmented list}
 \operatorname{inv}(z_0,\ldots,z_{\bar k},t_A(1),\ldots,t_A(m))
 \equiv |E_A|
 =\sum_{j=0}^{\bar k}|y_j|\pmod2.
\end{equation}
This is a parity statement: one move need not change the inversion number
by exactly one.

We now separate the inversions of this full list into three kinds:
\begin{itemize}
 \item inversions between two roots;
 \item inversions between a root and a target;
 \item inversions between two targets.
\end{itemize}
The last kind is precisely what the lemma seeks to count.

For the first kind, recall that $b_j=\bigl|\{i:i>j,\ z_i>z_j\}\bigr|$.
Thus $b_j$ counts the non-inversions whose first entry is $z_j$
in the root list. Since the roots are distinct, each of the
$\binom{\bar k+1}{2}$ pairs is either an inversion or a non-inversion.
Hence
\[
 \operatorname{inv}(z_0,\ldots,z_{\bar k})
 =\binom{\bar k+1}{2}-\sum_{j=0}^{\bar k}b_j.
\]

For the second kind, every root appears before every target. Therefore a
root $z_j$ and a target form an inversion exactly when the target is smaller
than $z_j$.
Since the vertices are consecutive with smallest vertex $|h^++h^-|$,
there are exactly $z_j-|h^++h^-|$
vertices smaller than $z_j$. Some are roots, so the number of targets
smaller than $z_j$ is $ z_j-|h^++h^-|-\bigl|\{i:z_i<z_j\}\bigr|$. Here, the last set includes all smaller roots, regardless of their
indices as these are never targets of an edge.
When we sum this expression over the components $j$, each unordered pair of roots contributes exactly once to the subtracted term: its smaller member is counted for its larger member.
Thus $ \sum_{j=0}^{\bar k}\bigl|\{i:z_i<z_j\}\bigr|
 =\binom{\bar k+1}{2}.$
The number of root--target inversions is consequently
\[
 \sum_{j=0}^{\bar k}(z_j-|h^++h^-|)-\binom{\bar k+1}{2}.
\]

Combining the three contributions gives the exact equality
\begin{equation*}
    \operatorname{inv}(z_0,\ldots,z_{\bar k},t_A(1),\ldots,t_A(m))=\operatorname{inv}(t_A(1),\ldots,t_A(m))
 +\sum_{j=0}^{\bar k}\bigl(z_j-|h^++h^-|-b_j\bigr).
\end{equation*}
The two binomial coefficients have cancelled. Reducing modulo $2$, and
using the root-move argument, yields
\[
 \operatorname{inv}(t_A(1),\ldots,t_A(m))
 \equiv\sum_{j=0}^{\bar k}
 \bigl(|y_j|+z_j-|h^++h^-|+b_j\bigr)\pmod2.
\]
The remaining step in the lemma is just substitution. From the definition
of $z_j$,
\[
 |y_j|+z_j-|h^++h^-|
 =\begin{cases}
 (K+1)y_j+j,&y_j\geq0,\\
 -(K+1)y_j-j-2|h^++h^-|,&y_j<0.
 \end{cases}
\]
Modulo $2$, this becomes
\[
 (K-1)y_j+j+2|h^++h^-|[y_j<0].
\]
Substituting into the above and using
$\sum_{j=0}^{\bar k}j=\binom{\bar k+1}{2}$ gives the asserted sign.

\end{align*}
which is the desired expression.

\begin{multline*}
    \det_{1\leq i,j\leq n}\left(
(\sqrt{t})^{-(j-1/2)}\eta_i(k+1/2+j-1/2)
 +(\sqrt{t})^{j-1/2}\eta_i(k+1/2-(j-1/2)\right)\\=\sum_{\substack{(y_0,\dots,y_{ k}) \in\mathbb{Z}^{k+1}\\ y_0\geq 0}}
        (-1)^{|y| + \sum_{j=0}^k [y_j<0] + \operatorname{inv}(\vec z_{0,k})} t^{-\frac{n^2}{4}+\sum_{i=0}^k\left(\frac{1}{2}Ky_i^2+y_ii\right)}
         \det_{\substack{1\le i\le n \\0 \le j\le k+n \\ j \not=z(y_r), r=0,\dots,k}}({\eta_i(j)}),
\end{multline*}
where $z(y_j)=\alpha_j(Ky_j+j)$ and we denote $\vec z_{0,k}\coloneqq(z(y_o),\dots,z(y_k))$. This is because as in the previous proofs we have that $\binom{k+1}{2} - \sum_{j=0}^{k} b_j = \operatorname{inv}(\vec z_{0,k})$ and furthermore 
\begin{equation*}
    \prod_{j=1}^{n}p_+(j) 
        \prod_{i=0}^{k}
        \prod_{\ell=0}^{\alpha_iy_i-1}
        \frac{p_-(K\ell+k+\alpha_i i+1)}{p_+(K\ell+k+\alpha_i i+1)}=t^{-\frac{n^2}{4}+\sum_{i=0}^k\left(\frac{1}{2}Ky_i^2+y_ii\right)},
\end{equation*}
as the following simple calculations shows:
\begin{multline*}
    -\sum_{j=1}^n\left(\frac{j}{2}-\frac{1}{4}\right)+\sum_{i=0}^k\sum_{\ell=0}^{\vert y_i\vert-1}\left(K\ell+k+\alpha_ii+1-1/2\right)\\=-\frac{1}{2}\binom{n+1}{2}+\frac{n}{4}+\sum_{i=0}^k\left(K\binom{\vert y_i\vert }{2}+\left(k+\frac{1}{2}\right)\vert y_i\vert +y_ii\right)=-\frac{n^2}{4}+\sum_{i=0}^k\left(\frac{1}{2}Ky_i^2+y_ii\right).
\end{multline*}
Inserting this into our expression from before we have that
\begin{multline*}
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,-\sqrt{t},0)\\=\sum_{\substack{(y_0,\dots,y_{ k}) \in\mathbb{Z}^{k+1}\\ y_0\geq 0}}
        (-1)^{\binom{n}{2}+|y| + \sum_{j=0}^k [y_j<0] + \operatorname{inv}(\vec z_{0,k})} t^{\sum_{i=0}^k\left(\frac{1}{2}Ky_i^2+y_ii\right)}
         \frac{\det\limits_{\substack{1\le i\le n \\0 \le j\le k+n \\ j \not=z(y_r), r=0,\dots,k}}({\eta_i(j)})}{\det_{1\leq i,j\leq n}\left(\eta_i(n-j+1)\right)}.
\end{multline*}
Now consider the determinant
$$\det\limits_{\substack{1\le i\le n \\0 \le j\le k+n \\ j \not=z(y_r), r=0,\dots,k}}({\eta_i(j)}),$$
from the numerator and note that if $y_0\ne0$, the determinant vanishes as one column is all zeros. Therefore, we may assume $y_0=0$ from now on. Note that then
\begin{equation*}
    \operatorname{inv}(\vec z_{0,k})=\operatorname{inv}(0,z(y_1),\dots,z(y_k))=\operatorname{inv}(\vec z_{1,k})+\vert\{j\in[1,k]\mid z(y_j)<0\}=\operatorname{inv}(\vec z_{1,k}).
\end{equation*}
We have arrived at the following expression
\begin{multline*}
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,-\sqrt{t},0)\\=\sum_{(y_1,\dots,y_{ k}) \in\mathbb{Z}^{k}}(-1)^{\binom{n}{2}+|y| + \sum_{j=1}^k [y_j<0] + \operatorname{inv}(\vec z_{1,k})}t^{\sum_{i=1}^k\left(\frac{1}{2}Ky_i^2+y_ii\right)}\frac{\det\limits_{\substack{1\le i\le n \\1 \le j\le k+n \\ j \not=z(y_r), r=1,\dots,k}}({\eta_i(j)})}{\det_{1\leq i,j\leq n}\left(\eta_i(n-j+1)\right)}.
\end{multline*}
To be able to use the Weyl character formula of type $\mathrm{C}_n$ \eqref{eq: Weyl character formula type C} and Corollary~\ref{cor:useful.complementary}, we define a vector $\nu\coloneqq (\nu_1,\dots,\nu_k)$ by
$$\nu_i\coloneqq z(y_i)-1=\alpha_i(Ky_i+i)-1=\vert Ky_i+i\vert-1$$
and observe that $\operatorname{inv}(\vec z)=\operatorname{inv}(\nu).$
Therefore, we can rewrite our expression as
\begin{multline*}
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,-\sqrt{t},0)\\=\sum_{(y_1,\dots,y_{ k}) \in\mathbb{Z}^{k}}(-1)^{\binom{n}{2}+|y| + \sum_{j=1}^k [y_j<0] + \operatorname{inv}(\nu)}t^{\sum_{i=1}^k\left(\frac{1}{2}Ky_i^2+y_ii\right)}\frac{\det\limits_{\substack{1\le i\le n \\1 \le j\le k+n \\ j \not=\nu_r+1, r=1,\dots,k}}({\eta_i(j)})}{\det_{1\leq i,j\leq n}\left(\eta_i(n-j+1)\right)}.
\end{multline*}
By the Weyl character formula of type $\mathrm{C}_n$ \eqref{eq: Weyl character formula type C} and Corollary~\ref{cor:useful.complementary} we have that
$$(-1)^{\binom{n}{2}}\frac{\det\limits_{\substack{1\le i\le n \\1 \le j\le k+n \\ j \not=\nu_r+1, r=1,\dots,k}}(\eta_i(j))}{\det\limits_{1 \le i,j \le n}(\eta_i(n-j+1))}=sp_{\lambda(\nu)}(X). $$
Note that the condition for the matrix in the numerator to be a square matrix is precisely the condition that the $y_a$'s corresponding to the $\nu_a$'s are inside $\{0,\dots,n+k-1\}$ and are all different. Inserting into our expression from above yields
\begin{equation*}
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,-\sqrt{t},0)=\sum_{(y_1,\dots,y_{ k}) \in\mathbb{Z}^{k}}(-1)^{|y| + \sum_{j=1}^k [y_j<0] + \operatorname{inv}(\nu)}t^{\sum_{i=1}^k\left(\frac{1}{2}Ky_i^2+y_ii\right)}sp_{\lambda(\nu)}(X)
\end{equation*}
where we sum over all $y\in\mathbb{Z}^k$ such that the corresponding $\nu_a$'s are inside $\{0,\dots,n+k-1\}$ and are all different, as desired.

Let $\nu_a = |Ky_a + a| - 1$ as in Theorem~\ref{thm-type-bc-0-char-all}. 
Unravelling the definitions preceding Lemma~\ref{lem:flip} we have
\begin{multline*}
\lambda(\nu)'_i = (\mathrm{sort}(\nu) - \rho_k)^{\square n^k}_i = 
n - (\mathrm{sort}(\nu) - \rho_k)_{k+1-i} =
n - \mathrm{sort}(\nu)_{k+1-i} + i -1\\
= n - |Ky_{\sigma(i)} + \sigma(i)| + i\,,
\end{multline*}
where $\sigma$ is the permutation that sorts $\nu$ into increasing order.
Apply the dual Jacobi--Trudi identity of type $\mathrm{C}_n$ \eqref{eq: dual JT type C} to the formula of~\cref{thm-type-bc-0-char} to get
\begin{multline*}
    P_{(k^n)}^{\mathrm{BC}_n}(X;t,-\sqrt{t},0)\\=\sum_{y \in\mathbb{Z}^{k}}(-1)^{|y| + \sum_{j=1}^k [y_j<0] + \operatorname{inv}(\nu)}t^{\sum_{j=1}^k\left(\frac{1}{2}Ky_j^2+y_jj\right)}\det_{1\leq i,j\leq k}(\dot{e}_{\lambda(\nu)'_i-i+j}
- 
\dot{e}_{\lambda(\nu)'_i-i-j})
\end{multline*}
By the assumption in Theorem \ref{thm-type-bc-0-char-all}, the sum ranges over all vectors $y\in\mathbb{Z}^k$ such that $\nu_1,\dots,\nu_k\in[0,n+k-1]$ are all different elements. By Lemma \ref{lem: handling the summation range} the determinant on the right hand side of this sum vanishes outside of this range of summation, hence we can extend the sum, to a sum over all $y\in\mathbb{Z}^k$. 

Plugging in the expression for $\lambda(\nu)'_i$, from before this determinant is equal to
\[
\det_{1\leq i,j\leq k}(
\dot{e}_{
n - |Ky_{\sigma(i)} + \sigma(i)|  + j
} -
\dot{e}_{
n - |Ky_{\sigma(i)} + \sigma(i)| - j
}).
\]
Now rearrange the rows of the matrix by the permutation $\sigma^{-1}$. This cancels the factor $\operatorname{sgn}(\sigma) = (-1)^{\mathrm{inv}(\nu)}$, as $\sigma$ is the permutation that sorts $\nu$ in increasing order, hence the sign is given by the number of inversions of $\nu$. The matrix becomes
\[
\det_{1\leq i,j\leq k}(
\dot{e}_{
n - |Ky_{i} + i| + j 
} -
\dot{e}_{
n - |Ky_{i} + i| - j 
}).
\]
We distinguish cases according to the sign of $y_i$ and fix some $1\leq i,j\leq k$.
If $y_i$ is non-negative, we use the formula $\dot{e}_{n+r}=\dot{e}_{n-r}$  to get
\begin{equation*}\dot{e}_{
n - |Ky_{i} + i| + j 
} -
\dot{e}_{
n - |Ky_{i} + i| - j 
}= 
\dot{e}_{
n - Ky_{i} - i + j
} -
\dot{e}_{
n - Ky_{i} - i - j
} =
\dot{e}_{
n + Ky_{i} + i - j
} -
\dot{e}_{
n + Ky_{i} + i + j 
}\,.
\end{equation*}
If $y_i$ is negative, one directly gets
\begin{equation*}
\dot{e}_{
n - |Ky_{i} + i| + j 
} -
\dot{e}_{
n - |Ky_{i} + i| - j 
}=
-(
\dot{e}_{
n + Ky_{i} + i - j
} -
\dot{e}_{
n + Ky_{i} + i + j 
})\,.
\end{equation*}
Pulling this sign out of the determinant using the multilinearity, we obtain an additional factor of $(-1)^{\sum_{j=1}^k [y_j<0]}$. Replacing $y_i$ by $-y_i$ and then transposing the matrix yields
\[
P_{(k^n)}^{\mathrm{BC}_n}(X;t,-\sqrt{t},0)=
\sum_{y \in \mathbb{Z}^k}(-1)^{\vert y\vert}
t^{\sum_{j=1}^k \left( \frac{1}{2}K y_j^2 - jy_j \right)}
\det_{1 \le i,j \le k}
\left(
\dot{e}_{\,n-i+j-K y_j}
-
\dot{e}_{\,n+i+j- Ky_j}
\right).\]
By Lemma \ref{lem: handling the summation range}, the support of this sum is finite, so extending the sum to a finite $S_k$ invariant subset containing the support, by Lemma~\ref{lem:powery}, we conclude the proof.
