<!-- assumption-1: body_md -->
\label{ass:mesh}
\leavevmode
\begin{enumerate}[label=(M\arabic*),nosep]
  \item\label{it:M1} \emph{matching}: the intersection of two distinct cells is
        either empty, or a common vertex, edge or face (no hanging node)\footnote{In the meshing community this is known as a manifold mesh.}.
  \item\label{it:M2} each cell $K$ is the image of a fixed reference cell
        $\widehat{K}$ (simplex, quadrilateral/hexahedron, prism, \dots) by a
        smooth, invertible map $F_K : \widehat{K} \to K$ with
        $\det(\mathbb{J}_K) > 0$ everywhere on $\widehat{K}$, where
        $\mathbb{J}_K := \mathrm{D}F_K$;
  \item\label{it:M3} $\mu$ is constant on each $K$, with value $\mu_K$.
\end{enumerate}
<!-- end assumption-1 -->
<!-- remark-1: body_md -->
Because $V_h \subset V$, no \emph{potential} reconstruction is needed and the
framework of \cite{vohralik_notes,ernvohralik2015} simplifies to a flux
reconstruction only. For nonconforming, DG or mixed discretizations, the
estimator gains a nonconformity term; see \cite{ernvohralik2015}.
<!-- end remark-1 -->
<!-- definition-1: body_md -->
\label{def:quality}
For a cell $K$ we consider
\begin{enumerate}[label=(Q\arabic*),nosep]
  \item\label{it:Q1} the \emph{shape-regularity ratio}
        $\gamma_K := h_K / \rho_K$, $\rho_K$ being the diameter of the largest
        ball inscribed in $K$;
  \item\label{it:Q2} the \emph{aspect ratio}
        $\alpha_K := \varsigma_1(K) / \varsigma_d(K)$, i.e.\ the condition number
        of $\mathbb{J}_K$ in the spectral norm (for triangles we take
        $\widehat{K}$ equilateral here, so that $\alpha_K = 1$ characterises
        the equilateral cell and the value does not depend on the numbering
        of the vertices);
  \item\label{it:Q3} the \emph{scaled Jacobian}
                                   
                                                     
                                                         
                                
        $s_K := \min_{\widehat\boldsymbol{x}}
        \Big[ \det(\mathbb{J}_K(\widehat\boldsymbol{x})) \big/ \prod_{i=1}^d
        \lVert (\mathbb{J}_K(\widehat\boldsymbol{x}))_{:,i} \rVert_2 \Big] \in [-1,1]$
        equal to $1$ when the columns of $\mathbb{J}_K$ are
        orthogonal and tending to $0$
        for a flat (sliver, cap, needle) element, and negative for an inverted
        one, i.e.\ one whose vertices are traversed in the wrong
        orientation (                        
                        
        $\det(\mathbb{J}_K) < 0$), which \ref{it:M2} excludes. For a
        triangle the bracket is the sine of the angle at the vertex onto which
        $F_K$ maps the origin of $\widehat{K}$; to make the measure
        independent of the numbering of the vertices we take, as mesh-quality
        software does, its minimum over the three corners,
        $s_K = \min_i \sin\theta_i(K)$, $\theta_i(K)$ being the angles of $K$.
        This equals $1/\sqrt2 \approx 0.71$ on the right triangles of a
        structured mesh (the value $s_{\min}$ reported for family \ref{it:F1}
        in Section~\ref{sec:numerics}), $\sqrt3/2$ on the equilateral
        triangle, and is of order $t$ for both the needle and the cap of
        family \ref{it:F3};                                                                                            
  \item\label{it:Q4} the \emph{Frobenius condition number}
        $\kappa_K := \tfrac{1}{d}\,\lVert \mathbb{J}_K \rVert_F
        \lVert \mathbb{J}_K^{-1} \rVert_F \geq 1$ and the associated \emph{mean ratio}
        $1/\kappa_K$ \cite{knupp2001};
  \item\label{it:Q5} the extremal angles $\theta_{\min}(K)$,
        $\theta_{\max}(K)$ (dihedral angles in 3D).
\end{enumerate}

<!-- definition-2: body_md -->
\label{def:admissible}
A vector field $\boldsymbol{p}$ is \emph{admissible} for \eqref{eq:strong} if
\begin{equation}\label{eq:AC}
  \boldsymbol{p} \in \boldsymbol{H}(\operatorname{div},\Omega) \quad \text{and} \quad \operatorname{div} \boldsymbol{p} = f \ \text{ in } \Omega .
  \tag{AC}
\end{equation}
<!-- end definition-2 -->
<!-- theorem-1: body_md -->
\label{thm:prager}
Let $u \in V$ solve \eqref{eq:weak}, let $\boldsymbol{p}$ satisfy \eqref{eq:AC} and let
$v \in V$ be arbitrary. Then
\begin{equation}\label{eq:prager}
  \lVert \mu^{-1/2}\left(\boldsymbol{p} + \mu\nabla v\right) \rVert^2
  =
  \lVert \mu^{-1/2}\left(\boldsymbol{p} - \boldsymbol{\sigma}\right) \rVert^2 + \vert\!\vert\!\vert u-v \vert\!\vert\!\vert^2 .
\end{equation}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Write $\boldsymbol{p} + \mu\nabla v = (\boldsymbol{p} - \boldsymbol{\sigma}) + \mu\nabla(v-u)$, using
$\boldsymbol{\sigma} = -\mu\nabla u$. Expanding the square,
\[
  \lVert \mu^{-1/2}(\boldsymbol{p} + \mu\nabla v) \rVert^2
  = \lVert \mu^{-1/2}(\boldsymbol{p}-\boldsymbol{\sigma}) \rVert^2 + \vert\!\vert\!\vert u-v \vert\!\vert\!\vert^2
    + 2 \left( \boldsymbol{p} - \boldsymbol{\sigma} , \nabla(v-u) \right) .
\]
Set $w := v-u \in V$. By \eqref{eq:green} and \eqref{eq:AC},
$\left( \boldsymbol{p} , \nabla w \right) = -\left( \operatorname{div}\boldsymbol{p} , w \right) = -\left( f , w \right)$, while by \eqref{eq:weak},
$\left( -\boldsymbol{\sigma} , \nabla w \right) = \left( \mu\nabla u , \nabla w \right) = a(u,w) = \left( f , w \right)$. The two
contributions cancel, which is the \emph{orthogonality property} at the heart of
the identity.
<!-- end proof-1 -->
<!-- corollary-1: body_md -->
\label{cor:hypercircle}
Taking $v = u_h$ in \eqref{eq:prager} and dropping the first term on the right,
\begin{equation}\label{eq:APEB}
  \vert\!\vert\!\vert u-u_h \vert\!\vert\!\vert \ \leq \
  \inf_{\boldsymbol{p} \text{ satisfying } \eqref{eq:AC}}
  \lVert \mu^{-1/2}\left(\boldsymbol{p} + \mu\nabla u_h\right) \rVert .
\end{equation}
<!-- end corollary-1 -->
<!-- proposition-1: body_md -->
\label{prop:gap}
Define the primal and dual (complementary) energies
\begin{equation}\label{eq:energies}
  \mathcal{E}(v) := \tfrac12 a(v,v) - \ell(v), \quad v \in V,
  \qquad
  \mathcal{E}^*(\boldsymbol{p}) := -\tfrac12\lVert \mu^{-1/2}\boldsymbol{p} \rVert^2, \quad
  \boldsymbol{p} \text{ satisfying } \eqref{eq:AC} .
\end{equation}
Then $\min_{v\in V}\mathcal{E}(v) = \mathcal{E}(u)
= \mathcal{E}^*(\boldsymbol{\sigma}) = \max_{\boldsymbol{p}}\mathcal{E}^*(\boldsymbol{p})$ --- there is no duality
gap --- and, for every $v\in V$ and every admissible $\boldsymbol{p}$,
\begin{equation}\label{eq:gap}
  \mathcal{E}(v) - \mathcal{E}^*(\boldsymbol{p})
  = \tfrac12\vert\!\vert\!\vert u-v \vert\!\vert\!\vert^2 + \tfrac12\lVert \mu^{-1/2}(\boldsymbol{p}-\boldsymbol{\sigma}) \rVert^2
  = \tfrac12\lVert \mu^{-1/2}(\boldsymbol{p}+\mu\nabla v) \rVert^2 .
\end{equation}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Since $a(u,v)=\ell(v)$, one has
$\mathcal{E}(v) = \tfrac12 a(v-u,v-u) - \tfrac12 a(u,u)
= \tfrac12\vert\!\vert\!\vert u-v \vert\!\vert\!\vert^2 - \tfrac12\lVert \mu^{1/2}\nabla u \rVert^2$. On the other hand,
for $\boldsymbol{p}$ admissible, \eqref{eq:green} and \eqref{eq:weak} give
$\left( \mu^{-1}\boldsymbol{p} , \boldsymbol{\sigma} \right) = -\left( \boldsymbol{p} , \nabla u \right) = \left( \operatorname{div}\boldsymbol{p} , u \right) = \left( f , u \right)
= \lVert \mu^{1/2}\nabla u \rVert^2$, whence
\[
  \tfrac12\lVert \mu^{-1/2}(\boldsymbol{p}-\boldsymbol{\sigma}) \rVert^2
  = \tfrac12\lVert \mu^{-1/2}\boldsymbol{p} \rVert^2 - \lVert \mu^{1/2}\nabla u \rVert^2
    + \tfrac12\lVert \mu^{-1/2}\boldsymbol{\sigma} \rVert^2
  = -\mathcal{E}^*(\boldsymbol{p}) - \tfrac12\lVert \mu^{1/2}\nabla u \rVert^2 ,
\]
using $\lVert \mu^{-1/2}\boldsymbol{\sigma} \rVert = \lVert \mu^{1/2}\nabla u \rVert$. Adding the two
displays gives the first equality in \eqref{eq:gap}; the second one is
Theorem~\ref{thm:prager}. Taking $v=u$ and $\boldsymbol{p}=\boldsymbol{\sigma}$ shows that the gap
vanishes, and non-negativity of \eqref{eq:gap} gives weak duality.
<!-- end proof-2 -->
<!-- remark-2: body_md -->
\label{rem:duality}
Proposition~\ref{prop:gap} says that the squared estimator
$\lVert \mu^{-1/2}(\boldsymbol{\sigma}_h+\mu\nabla u_h) \rVert^2$ is nothing but \emph{twice the
primal--dual gap} evaluated at the computed pair $(u_h,\boldsymbol{\sigma}_h)$: it is a
by-product of weak duality, which holds for any admissible pair and involves no
discretization whatsoever. This is why the constant is one, and this is the
viewpoint --- \emph{a posteriori estimation by duality} --- underlying the
functional error estimates of Repin \cite{repin2000,repin2008} and Han
\cite{han2005}; Bartels and Kaltenbach \cite{bartels2023} have recently
extended it to general convex minimisation, including nonsmooth and degenerate
problems, where the quadratic structure used above is unavailable.
<!-- end remark-2 -->
<!-- definition-3: body_md -->
\label{def:eqflux}
A field $\boldsymbol{\sigma}_h$ is an \emph{equilibrated flux reconstruction} if
\begin{equation}\label{eq:EF}
  \boldsymbol{\sigma}_h \in \boldsymbol{H}(\operatorname{div},\Omega)
  \quad \text{and} \quad
  \left( \operatorname{div} \boldsymbol{\sigma}_h - f , 1 \right)_K = 0 \qquad \forall K \in \mathcal{T}_h .
  \tag{EF}
\end{equation}
<!-- end definition-3 -->
<!-- lemma-1: body_md -->
\label{lem:poincare}
Let $K$ be a \emph{convex} cell and $\varphi \in H^1(K)$ with mean value
$\overline{\varphi}_K$. Then
\begin{equation}\label{eq:poincare}
  \lVert \varphi - \overline{\varphi}_K \rVert_K
  \leq \frac{h_K}{\pi} \lVert \nabla\varphi \rVert_K .
\end{equation}
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
This is the Payne--Weinberger inequality \cite{payneweinberger1960}; the
constant $h_K/\pi$ is optimal (attained in the limit of a thin
rectangle/cylinder) and depends on $K$ only through its diameter.
<!-- end proof-3 -->
<!-- remark-3: body_md -->
\label{rem:nonconvex}
Inequality \eqref{eq:poincare} may fail on a non-convex cell. Three remedies keep
the bound guaranteed: (i) split each non-convex cell into convex sub-cells for
the purpose of evaluating $\eta_K^{\mathrm{R}}$ only, and use $h_{K'}/\pi$ on
each sub-cell $K'$ --- this only \emph{improves} the estimate since
$h_{K'}\le h_K$; (ii) use the constant $C_{P,K}h_K$ with $C_{P,K}$ computed from
a local eigenvalue problem (a small, purely local computation); (iii) when
$f$ is a piecewise polynomial of degree at most $p$, make the
term vanish by using a reconstruction with $\operatorname{div}\boldsymbol{\sigma}_h = \Pi_p f$, see
Proposition~\ref{prop:oscillation}; for a general $f$ the data oscillation
\eqref{eq:osc} remains and still involves the Poincar\'e constant of the
cell. In our numerical experiments all cells are triangles, hence
convex, and Lemma~\ref{lem:poincare} applies directly; the reconstructions
used there satisfy $\operatorname{div}\boldsymbol{\sigma}_h = \Pi_p f$ exactly, so that
$\eta_K^{\mathrm{R}}$ reduces to the data oscillation \eqref{eq:osc}.
<!-- end remark-3 -->
<!-- theorem-2: body_md -->
\label{thm:upper}
Let $u$ solve \eqref{eq:weak}, $u_h$ solve \eqref{eq:fem}, and let $\boldsymbol{\sigma}_h$
satisfy \eqref{eq:EF}. Assume the cells are convex (see
Remark~\ref{rem:nonconvex} otherwise). Then
\begin{equation}\label{eq:upper}
  \vert\!\vert\!\vert u-u_h \vert\!\vert\!\vert^2
  \ \leq \
  \sum_{K\in\mathcal{T}_h}
  \left( \eta_K^{\mathrm{R}} + \eta_K^{\mathrm{F}} \right)^2
  =: \eta^2 .
\end{equation}
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
Since $u - u_h \in V$, and by definition of the energy norm,
\begin{equation}\label{eq:dualchar}
  \vert\!\vert\!\vert u-u_h \vert\!\vert\!\vert
  = \sup_{\varphi \in V,\ \vert\!\vert\!\vert \varphi \vert\!\vert\!\vert=1} a(u-u_h,\varphi) .
\end{equation}
Fix such a $\varphi$. Using \eqref{eq:weak},
\[
  a(u-u_h,\varphi) = \left( f , \varphi \right) - \left( \mu\nabla u_h , \nabla\varphi \right).
\]
Insert $\pm \operatorname{div}\boldsymbol{\sigma}_h$ and use the Green formula \eqref{eq:green} for
$\boldsymbol{\sigma}_h \in \boldsymbol{H}(\operatorname{div},\Omega)$ and $\varphi \in V$, i.e.\
$\left( \operatorname{div}\boldsymbol{\sigma}_h , \varphi \right) = -\left( \boldsymbol{\sigma}_h , \nabla\varphi \right)$:
\begin{equation}\label{eq:splitting}
  a(u-u_h,\varphi)
  = \underbrace{\left( f - \operatorname{div}\boldsymbol{\sigma}_h , \varphi \right)}_{=: T_{\mathrm{R}}}
    \ \underbrace{-\left( \boldsymbol{\sigma}_h + \mu\nabla u_h , \nabla\varphi \right)}_{=: T_{\mathrm{F}}} .
\end{equation}
For $T_{\mathrm{F}}$, a cellwise Cauchy--Schwarz inequality gives
\[
  T_{\mathrm{F}}
  \leq \sum_{K\in\mathcal{T}_h}
       \lVert \mu^{-1/2}(\boldsymbol{\sigma}_h + \mu\nabla u_h) \rVert_K \,
       \lVert \mu^{1/2}\nabla\varphi \rVert_K
  = \sum_{K\in\mathcal{T}_h} \eta_K^{\mathrm{F}}\, \vert\!\vert\!\vert \varphi \vert\!\vert\!\vert_K .
\]
For $T_{\mathrm{R}}$, split over the cells and use \eqref{eq:EF} to subtract the
cell mean of $\varphi$ for free:
\[
  T_{\mathrm{R}}
  = \sum_{K\in\mathcal{T}_h} \left( f - \operatorname{div}\boldsymbol{\sigma}_h , \varphi \right)_K
  = \sum_{K\in\mathcal{T}_h} \left( f - \operatorname{div}\boldsymbol{\sigma}_h , \varphi - \overline{\varphi}_K \right)_K .
\]
Cauchy--Schwarz and Lemma~\ref{lem:poincare} then yield
\[
  \begin{aligned}
  T_{\mathrm{R}}
  &\leq \sum_{K\in\mathcal{T}_h}
       \lVert f-\operatorname{div}\boldsymbol{\sigma}_h \rVert_K \, \frac{h_K}{\pi}\lVert \nabla\varphi \rVert_K \\
  &= \sum_{K\in\mathcal{T}_h}
       \Big[ \tfrac{h_K}{\pi}\mu_K^{-1/2}\lVert f-\operatorname{div}\boldsymbol{\sigma}_h \rVert_K \Big]
       \Big[ \mu_K^{1/2}\lVert \nabla\varphi \rVert_K \Big]
  = \sum_{K\in\mathcal{T}_h} \eta_K^{\mathrm{R}}\, \vert\!\vert\!\vert \varphi \vert\!\vert\!\vert_K ,
  \end{aligned}
\]
where we inserted $1 = \mu_K^{-1/2}\mu_K^{1/2}$, licit because $\mu$ is constant
on $K$ by \ref{it:M3}. Summing the two bounds and applying a last, discrete
Cauchy--Schwarz inequality,
\[
  a(u-u_h,\varphi)
  \leq \sum_{K\in\mathcal{T}_h}(\eta_K^{\mathrm{R}}+\eta_K^{\mathrm{F}})\vert\!\vert\!\vert \varphi \vert\!\vert\!\vert_K
  \leq \Big( \sum_{K\in\mathcal{T}_h}(\eta_K^{\mathrm{R}}+\eta_K^{\mathrm{F}})^2\Big)^{1/2}
       \Big( \sum_{K\in\mathcal{T}_h}\vert\!\vert\!\vert \varphi \vert\!\vert\!\vert_K^2\Big)^{1/2} ,
\]
and the last factor equals $\vert\!\vert\!\vert \varphi \vert\!\vert\!\vert=1$. Taking the supremum over
$\varphi$ in \eqref{eq:dualchar} concludes.
<!-- end proof-4 -->
<!-- remark-4: body_md -->
The proof used: the conformity $V_h\subset V$ (through $u-u_h \in V$), the
$H(\operatorname{div})$-conformity of $\boldsymbol{\sigma}_h$ (through the Green formula), the cellwise
equilibration \eqref{eq:EF}, and the Poincar\'e inequality on convex cells. No
shape regularity, no minimum/maximum angle, no quasi-uniformity, no bound on the
patch cardinality, no asymptotic regime in $h$. In particular
\eqref{eq:upper} is valid on the coarsest and most distorted mesh, and even for
$k=1$ on a single element.
<!-- end remark-4 -->
<!-- remark-5: body_md -->
\label{rem:nonhomog}
Suppose $u = g$ on $\Gamma$ with $g$ in the trace space of $H^1(\Omega)$, and let
$u_h \in V_h$ satisfy \eqref{eq:fem} for all $v_h \in V_h \cap H^1_0(\Omega)$ with
its boundary values fixed to some $g_h$. Theorem~\ref{thm:upper} goes through
verbatim \emph{provided} $u - u_h \in H^1_0(\Omega)$, i.e.\ provided
$g_h = g$ exactly on $\Gamma$ --- the only place where the boundary condition
entered the proof was through $u - u_h \in V$ and the vanishing of the boundary
term in \eqref{eq:green}. If, as is usual, $g_h$ is merely an interpolant of
$g$, then $u - u_h \notin H^1_0(\Omega)$ and an additional term appears, measuring
the distance from $g_h$ to $g$ in $H^{1/2}(\Gamma)$; bounding it computably
requires an argument of a different nature, and the constant it brings does
depend on the shape of the boundary cells. This is why the main experiments of
Section~\ref{sec:numerics} are built on test cases with homogeneous Dirichlet
data. The L-shaped test case \ref{it:T2}, whose exact solution
$r^{2/3}\sin(2\theta/3)$ has non-polynomial boundary values, therefore falls
outside the theorem; we report it separately in
Section~\ref{sec:numerics_lshape} as an experiment on what happens when a
hypothesis is dropped, not as evidence for the bound. This is a genuine gap in
the theory presented here, not an oversight; see \cite{ernvohralik2015} for the
treatment of general boundary data.
<!-- end remark-5 -->
<!-- proposition-2: body_md -->
\label{prop:oscillation}
Assume the reconstruction satisfies the stronger property
$\operatorname{div}\boldsymbol{\sigma}_h = \Pi_p f$, where $\Pi_p$ denotes the $L^2$-orthogonal projection
onto the space of piecewise polynomials of degree at most $p$ (broken, i.e.\
cellwise). Then $\eta_K^{\mathrm{R}}$ reduces to the \emph{data oscillation}
\begin{equation}\label{eq:osc}
  \operatorname{osc}_K(f) := \frac{h_K}{\pi}\,\mu_K^{-1/2}\lVert f - \Pi_p f \rVert_K ,
\end{equation}
which is exactly computable, involves neither $u_h$ nor $\boldsymbol{\sigma}_h$, and is of
order $h_K^{p+2}$ on each cell where $f \in H^{p+1}(K)$ --- at least
one order better than $\eta_K^{\mathrm{F}}$, which is of order $h^{k}$,
as soon as $p \geq k-1$. It vanishes identically when $f$ is itself a
piecewise polynomial of degree $\le p$.
<!-- end proposition-2 -->
<!-- remark-6: body_md -->
\label{rem:dropR}
Only under the hypothesis of Proposition~\ref{prop:oscillation}, and even then
it should be \emph{computed} rather than dropped, since it is cheap. For a
reconstruction that merely satisfies \eqref{eq:EF}, $\eta_K^{\mathrm{R}}$ is of
the same order as $\eta_K^{\mathrm{F}}$ and neglecting it destroys the
guarantee. Classical presentations are often silent on this distinction; the
FEniCS script of Section~\ref{sec:code} evaluates both terms.
<!-- end remark-6 -->
<!-- theorem-3: body_md -->
\label{thm:efficiency}
For the patchwise reconstruction of Section~\ref{sec:patchwise} on a simplicial
mesh, there holds
\begin{equation}\label{eq:efficiency}
  \eta_K^{\mathrm{F}}
  \ \leq \ C_{\mathrm{st}}\big(\gamma_{\omega_K}\big)\,
  \Big( \vert\!\vert\!\vert u-u_h \vert\!\vert\!\vert_{\omega_K} + \operatorname{osc}_{\omega_K}(f) \Big),
  \qquad
  \operatorname{osc}_{\omega_K}(f)^2 := \sum_{K'\subset\omega_K}\operatorname{osc}_{K'}(f)^2 ,
\end{equation}
where $C_{\mathrm{st}}$ depends on the shape-regularity
$\gamma_{\omega_K} := \max_{K'\subset\omega_K}\gamma_{K'}$ of the patch and on
its cardinality, but neither on the polynomial degree $k$, which is the
content of \cite{braess2009,ernvohralik2020}, nor
on the diffusion contrast, which for conforming elements is the result of
Cai and Zhang \cite{caizhang2012}, obtained without any quasi-monotonicity
assumption on $\mu$.
<!-- end theorem-3 -->
<!-- remark-7: body_md -->
Following \cite{duprez2019damage}, if only a small subset
$\mathcal{T}_h^{\mathrm{dam}} \subset \mathcal{T}_h$ of cells is distorted, one expects
$C_{\mathrm{st}}$ to degrade only on the patches touching
$\mathcal{T}_h^{\mathrm{dam}}$, so that the global effectivity index remains $O(1)$: the
few bad patches contribute a bounded amount to the sum. The relevant parameter
should then be the \emph{diameter of the damaged patch}, not the quality of an
isolated cell. This is a conjecture we probe numerically in
Section~\ref{sec:numerics_damaged}.
<!-- end remark-7 -->
<!-- remark-8: body_md -->
Pros: three lines of \texttt{FEniCS}/\texttt{FreeFEM}, no local solver to write,
and it produces an \emph{exactly} equilibrated flux. Cons: it requires the
solution of a \emph{global} saddle-point problem, typically more expensive than
the primal problem it is supposed to certify, which defeats the purpose in a
production run. It is however perfect as a reference implementation, and this is
how we use it in Section~\ref{sec:numerics}.
<!-- end remark-8 -->
<!-- remark-9: body_md -->
\label{rem:noproject}
A tempting shortcut is to take $\boldsymbol{\sigma}_h := \Pi_{\boldsymbol{V}_h}(-\mu\nabla u_h)$, the
$L^2$-projection of the discrete flux onto $\boldsymbol{RT}_p$. The result does belong to
$\boldsymbol{H}(\operatorname{div},\Omega)$, so the Green formula used in Theorem~\ref{thm:upper} is still valid,
and the bound \eqref{eq:upper} still \emph{holds}. But nothing forces
\eqref{eq:EF}, so $\eta_K^{\mathrm{R}}$ is no longer a data oscillation: it is a
genuine term of the same order as $\eta_K^{\mathrm{F}}$, and the bound, though
valid, becomes useless in practice. Such a projection is essentially a
Zienkiewicz--Zhu recovery in disguise (Section~\ref{sec:comparison}); the whole
point of an \emph{equilibrated} reconstruction is precisely to make the residual
term collapse to \eqref{eq:osc}.

<!-- proposition-3: body_md -->
\label{prop:patchwise}
Let $k\geq 1$, so that $\psi_\mathrm{a}\inV_h$ (see \cite{ernvohralik2015}
for the choice of $p$ as a function of $k$). Then:
\begin{enumerate}[label=(\roman*),nosep]
  \item For $\mathrm{a}\in\mathcal{V}_h^{i}$, problem \eqref{eq:localpb} is well
        posed. Indeed, the pure Neumann compatibility condition
        $\left( g_\mathrm{a} , 1 \right)_{\omega_\mathrm{a}} = 0$ required for
        $\Pi_p g_\mathrm{a}$ to belong to $\operatorname{div}\boldsymbol{V}_h^{\mathrm{a}}
        = Q_h^{\mathrm{a}}$ reads
        \[
          \left( g_\mathrm{a} , 1 \right)_{\omega_\mathrm{a}}
          = \left( f , \psi_\mathrm{a} \right)_\Omega
            - \left( \mu\nabla u_h , \nabla \psi_\mathrm{a} \right)_\Omega
          = \ell(\psi_\mathrm{a}) - a(u_h,\psi_\mathrm{a}) = 0 ,
        \]
        which is exactly the discrete problem \eqref{eq:fem} tested with the hat
        function $\psi_\mathrm{a}\inV_h$. This is where the \emph{minus} sign in
        \eqref{eq:ga} is essential: with a plus sign one would obtain
        $2\,\ell(\psi_\mathrm{a})\neq 0$ and the local problem would have no
        solution.
  \item Each $\boldsymbol{\sigma}_h^{\mathrm{a}}$ has zero normal trace on
        $\partial\omega_\mathrm{a}\setminus\Gamma$, so its extension by zero has
        no normal jump across $\partial\omega_\mathrm{a}$ and belongs to
        $\boldsymbol{H}(\operatorname{div},\Omega)$; hence so does the finite sum $\boldsymbol{\sigma}_h$.
  \item $\operatorname{div}\boldsymbol{\sigma}_h^{\mathrm{a}} = \Pi_p g_\mathrm{a}$ on $\omega_\mathrm{a}$.
        For a boundary vertex this is immediate since $Q_h^{\mathrm{a}}$ is the
        full broken polynomial space. For an interior vertex, the second
        equation of \eqref{eq:localpb} only gives that
        $\operatorname{div}\boldsymbol{\sigma}_h^{\mathrm{a}} - \Pi_p g_\mathrm{a}$ is $L^2$-orthogonal to
        all zero-mean $q_h$, i.e.\ is a constant $c$ on $\omega_\mathrm{a}$; but
        $\left( \operatorname{div}\boldsymbol{\sigma}_h^{\mathrm{a}} , 1 \right)_{\omega_\mathrm{a}}
        = \int_{\partial\omega_\mathrm{a}}\boldsymbol{\sigma}_h^{\mathrm{a}}\cdot\boldsymbol{n} = 0
        = \left( \Pi_p g_\mathrm{a} , 1 \right)_{\omega_\mathrm{a}}$ by (i), so $c=0$.
  \item Consequently, using $\sum_\mathrm{a}\psi_\mathrm{a}\equiv 1$ and hence
        $\sum_\mathrm{a}\nabla\psi_\mathrm{a}\equiv \boldsymbol{0}$,
        \[
          \operatorname{div}\boldsymbol{\sigma}_h = \sum_\mathrm{a} \Pi_p g_\mathrm{a}
          = \Pi_p\Big( f\sum_\mathrm{a}\psi_\mathrm{a}
            - \mu\nabla u_h\cdot\sum_\mathrm{a}\nabla\psi_\mathrm{a}\Big)
          = \Pi_p f .
        \]
        Therefore \eqref{eq:EF} holds and Proposition~\ref{prop:oscillation}
        applies: the residual indicator reduces to the data oscillation
        \eqref{eq:osc}.
\end{enumerate}

<!-- algorithm-1: body_md -->
\label{alg:main}
\leavevmode
\begin{enumerate}[label=\arabic*.,nosep,leftmargin=*]
  \item Solve the primal problem \eqref{eq:fem} for $u_h$.
  \item For each vertex $\mathrm{a}\in\mathcal{V}_h$ (in parallel): assemble
        $g_\mathrm{a}$ by \eqref{eq:ga} and solve the local mixed problem
        \eqref{eq:localpb} for $\boldsymbol{\sigma}_h^{\mathrm{a}}$.
  \item Assemble $\boldsymbol{\sigma}_h := \sum_\mathrm{a}\boldsymbol{\sigma}_h^{\mathrm{a}}$ by
        \eqref{eq:assembly}.
  \item \emph{Certificate check} (cheap, optional but recommended): verify that
        $\max_K |\left( \operatorname{div}\boldsymbol{\sigma}_h - f , 1 \right)_K|$ is at round-off level.
  \item For each cell $K$, evaluate $\eta_K^{\mathrm{F}}$ by \eqref{eq:etaF} and
        $\eta_K^{\mathrm{R}} = \operatorname{osc}_K(f)$ by \eqref{eq:osc}.
  \item Return $\eta = \big(\sum_K(\eta_K^{\mathrm{R}}+\eta_K^{\mathrm{F}})^2
        \big)^{1/2}$. By Theorem~\ref{thm:upper},
        $\vert\!\vert\!\vert u-u_h \vert\!\vert\!\vert\leq\eta$, whatever the mesh.
\end{enumerate}

<!-- remark-10: body_md -->
Step 2 requires $\#\mathcal{V}_h$ independent mixed solves, each of a size
depending only on $p$ and on the number of cells in the patch, and each
independent of the others: the step is embarrassingly parallel and its
complexity is linear in $\#\mathcal{V}_h$, against a global saddle-point solve
for the reconstruction of Section~\ref{sec:mixed}. On the meshes of
Section~\ref{sec:numerics} with $k=1$ and $p=1$, each local system has at most
$43$ unknowns, so the arithmetic cost of step 2 is a few hundred flops per
vertex. We report that algebraic size rather than a wall-clock comparison: our
driver loops over the patches in \texttt{Python}, and its running time measures
the interpreter, not the algorithm (Section~\ref{sec:numerics_pw}).

<!-- remark-11: body_md -->
\label{rem:ziguigui}
An alternative and historically earlier route (Destuynder--M\'etivet
\cite{destuynder1999}, Lad\`eveze \cite{ladeveze2004}) is to build a first field
$\widetilde{\boldsymbol{p}}_h := \sum_\mathrm{a}\boldsymbol{p}_h^{\mathrm{a}}$ that equilibrates only
approximately, and then to add a cellwise correction $\delta\boldsymbol{p}_K$, obtained by
solving a local Neumann problem on $K$, so that
$\boldsymbol{p}_h|_K := \widetilde{\boldsymbol{p}}_h|_K + \delta\boldsymbol{p}_K$ is exactly admissible. The
subtlety, easily overlooked, is that \emph{the correction must be computed
exactly} for the bound to remain guaranteed; if it is computed in a finite
dimensional space, it is not exactly admissible, and one is back to the relaxed
setting of Definition~\ref{def:eqflux}. Presentations that assert that the
correction is ``negligible'' implicitly assume a fine and shape-regular mesh
--- which is precisely the assumption we do not make. The formulation \eqref{eq:localpb}
sidesteps the issue altogether: the constraint $\operatorname{div}\boldsymbol{\tau}_h = \Pi_p
g_\mathrm{a}$ is imposed \emph{inside} the discrete space, and the residual
defect is exactly the computable oscillation of
Proposition~\ref{prop:oscillation}.
<!-- end remark-11 -->
<!-- remark-12: body_md -->
On strongly distorted or nearly degenerate cells the local matrices in
\eqref{eq:localpb} become badly conditioned. Practical remedies: (i) scale the
$\boldsymbol{RT}$ basis by the face measures; (ii) solve \eqref{eq:localmin} in the
constrained least-squares form with an orthonormalised basis of the divergence
constraint; (iii) use a direct solver with pivoting on each patch --- the
matrices are tiny. Note that a badly conditioned \emph{local} problem does not
compromise the guarantee: any computed $\boldsymbol{\sigma}_h$ that lies in $\boldsymbol{H}(\operatorname{div},\Omega)$ and passes
the check $\left( \operatorname{div}\boldsymbol{\sigma}_h - f , 1 \right)_K = 0$ (verifiable a posteriori, to round-off)
gives a valid bound, however suboptimal. Note also that the approximation
properties of the $\boldsymbol{RT}_p$ interpolant on strongly anisotropic tetrahedra are
known and better than the isotropic theory suggests \cite{acosta2011}, which is
encouraging for the anisotropic mesh family \ref{it:F5} below.

<!-- proposition-4: body_md -->
\label{prop:goal}
There holds
\begin{equation}\label{eq:goalbound}
  \left| J(u) - J(u_h) \right|
  = \left| a(u-u_h, z-z_h) \right|
  \leq \vert\!\vert\!\vert u-u_h \vert\!\vert\!\vert\,\vert\!\vert\!\vert z-z_h \vert\!\vert\!\vert
  \leq \eta^{\mathrm{prim}}\,\eta^{\mathrm{dual}},
\end{equation}
where $\eta^{\mathrm{prim}}$ and $\eta^{\mathrm{dual}}$ are the
equilibrated flux estimators \eqref{eq:upper} of the primal and of the dual
problem.

$J(u)-J(u_h) = a(u-u_h,z)$ by \eqref{eq:dual}, and $a(u-u_h,z_h)=0$ by
Galerkin orthogonality \eqref{eq:galerkin}, whence
$J(u)-J(u_h)=a(u-u_h,z-z_h)$. Conclude with the Cauchy--Schwarz inequality in
the energy norm, then Theorem~\ref{thm:upper} applied twice.
