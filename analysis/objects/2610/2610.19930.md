<!-- theorem-1: body_md -->
\label{thm:ratios}
Under \eqref{eq:recurrence} and the stated sign conditions, suppose there are functions
$a_k:(0,\infty)\to[0,\infty)$ and
$h_k:(0,\infty)\to\mathbb R$, $0\le k<p$, such that
\begin{equation}\label{eq:local}
 \begin{aligned}
 &c_{m,N}\to a_k(t),\quad b_{m,N}\to h_k(t)\\[7pt]
 &\text{whenever }N\to\infty,\ m/N\to t>0,\ m\equiv k\pmod p.
 \end{aligned}
\end{equation}
Condition~\eqref{eq:local} already implies continuity of these functions.
Here $h_k=0$ if $r\ge2$. As $N\to\infty$, whenever $n/N\to t>0$ and
$n\equiv k\pmod p$,
\begin{equation}\label{eq:ratios}
 \begin{aligned}
 \frac{q_{n,N}(z)}{q_{n+1,N}(z)}&\to\varphi_k(z,t),\\[7pt]
 \frac{q_{n,N}(z)}{q_{n+p,N}(z)}&\to
 \Phi(z,t)=\prod_{k=0}^{p-1}\varphi_k(z,t).
 \end{aligned}
\end{equation}
locally uniformly on $\mathbb C\setminus\Sigma$; the second limit
does not require a fixed phase. The limits are nonzero and satisfy
\begin{equation}\label{eq:system}
 \frac1{\varphi_k(z,t)}
 =z-h_k(t)-a_{k-r}(t)\prod_{j=1}^r\varphi_{k-j}(z,t),
 \quad z\varphi_k(z,t)\to1\quad(z\to\infty).
\end{equation}
This normalization selects a unique analytic branch near infinity.
Both convergences are uniform when $n/N$ ranges over a compact
subinterval of $(0,\infty)$, with the limits evaluated at $t=n/N$.
The same uniform convergence holds after any fixed number of derivatives in $z$.

<!-- theorem-2: body_md -->
\label{thm:zeros}
Under \eqref{eq:recurrence} and the stated sign conditions, let
$a_k:(0,\infty)\to[0,\infty)$ and $h_k:(0,\infty)\to\mathbb R$,
$0\le k<p$, be measurable and finite almost everywhere,
with $h_k=0$ if $r\ge2$. Suppose that, for every $T>0$ and every $k$,
\begin{equation}\label{eq:measure}
 A_{k,N}\longrightarrow a_k,\quad H_{k,N}\longrightarrow h_k
 \quad\text{in measure on }(0,T).
\end{equation}
As $N\to\infty$, whenever $n/N\to T>0$, the normalized zero
measures, counting multiplicity, converge weakly:
\begin{equation}\label{eq:zero-average}
 \mu_{n,N}:=\frac1n\sum_{q_{n,N}(\zeta)=0}\delta_\zeta
 \quad\longrightarrow\quad
 \mu_T=\frac1T\int_0^T\mu_{a(t),h(t)}\,d t.
\end{equation}
On $\mathbb C\setminus\Sigma$,
\begin{equation}\label{eq:G}
 \begin{aligned}
 \frac1n\frac{q'_{n,N}(z)}{q_{n,N}(z)}&\to
 G_T(z):=\int\frac{\,d\mu_T(\zeta)}{z-\zeta}\\[7pt]
 &=-\frac1{pT}\int_0^T
       \frac{\partial_z\Phi(z;a(t),h(t))}{\Phi(z;a(t),h(t))}\,d t
 \end{aligned}
\end{equation}
locally uniformly. No coefficient bounds, integrability of $a_k,h_k$,
or limits at $t=0$ are required. The limiting measure need not have
compact support.

<!-- proposition-1: body_md -->
\label{prop:compact-zeros}
Under \eqref{eq:recurrence}--\eqref{eq:bound}, let
$a_k:(0,\infty)\to[0,\infty)$ and $h_k:(0,\infty)\to\mathbb R$,
$0\le k<p$, be measurable, with $h_k=0$ if $r\ge2$. Suppose
\begin{equation}\label{eq:L1}
 \|A_{k,N}-a_k\|_{L^1(0,T)}+
 \|H_{k,N}-h_k\|_{L^1(0,T)}\to0,\quad T>0,\quad 0\le k<p.
\end{equation}
Then \eqref{eq:zero-average} holds whenever $n/N\to T>0$.
The measures $\mu_{n,N}$ for all sufficiently large $N$ and $\mu_T$
are supported on $\Sigma_B$, where one may take
$B=B_r(C_L)$ for any $L>T$, with
\begin{equation}\label{eq:root-bound}
 B_r(C)=\begin{cases}
 C+2\sqrt C,&r=1,\\[7pt]
 \dfrac{r+1}{r}(rC)^{1/(r+1)},&r\ge2.
 \end{cases}
\end{equation}
The convergence in \eqref{eq:G} is locally uniform on
$\mathbb C\setminus\Sigma_B$, using the analytic continuation of
$\Phi(z;a(t),h(t))$ to this set.
Under \eqref{eq:bound}, conditions~\eqref{eq:measure} and
\eqref{eq:L1} are equivalent.

<!-- corollary-1: body_md -->
\label{cor:original}
Let $\gamma>0$, $\alpha_1,\ldots,\alpha_r\ge0$, and $c_m\ge0$ satisfy
$$
 c_m/m^\gamma\to\alpha_{k+1}
 \quad(m\to\infty,\ m\equiv k\pmod r),\quad 0\le k<r.
$$
Let $P_j(z)=z^j$ for $0\le j\le r$,
$P_{n+1}=zP_n-c_{n-r}P_{n-r}$ for $n\ge r$, and
$$
 Q_{n,N}(z)=N^{-n\gamma/d}P_n(N^{\gamma/d}z).
$$
As $N\to\infty$, if $n/N\to t\ge0$, then, locally uniformly off $\Sigma$,
\begin{equation}\label{eq:original-ratio}
 \begin{aligned}
 \frac{Q_{n,N}(z)}{Q_{n+r,N}(z)}&\to\Phi(z,t),\\[7pt]
 1&=\Phi(z,t)\prod_{j=1}^r(z-\alpha_jt^\gamma\Phi(z,t)),
 \quad z^r\Phi(z,t)\to1.
 \end{aligned}
\end{equation}
The normalized zero measures of $Q_{N,N}$ converge to a probability
measure $\mu$ with Cauchy transform
\begin{equation}\label{eq:original-G}
 G(z)=-\frac1r\int_0^1\frac{\partial_z\Phi(z,t)}{\Phi(z,t)}\,d t.
\end{equation}
Put
$$
 A(u)=\prod_{j=1}^r(1-\alpha_j u),
 \quad 1=yF(y)A(F(y)),\quad yF(y)\to1.
$$
For $t>0$, $\Phi(z,t)=zt^{-\gamma}F(z^dt^{-\gamma})$;
$\Phi(z,0)=z^{-r}$. With $y=z^d$,
\begin{equation}\label{eq:original-explicit}
 G(z)=-\frac1{rz}\left(1-\frac d\gamma
 \frac{\displaystyle\int_0^{F(y)}(uA(u))^{1/\gamma}\,\frac{du}{u}}
      {(F(y)A(F(y)))^{1/\gamma}}\right).
\end{equation}
The right-hand side, taken as a whole, is continued from large positive
$z$; this continuation is single-valued because it equals $G$.
For positive $\alpha_j$ and $c_m$, these are the ratio limit and
both transform formulas in
\cite[Theorems~1.1--1.2]{LoureiroVanAssche2026}, for the canonical
initial conditions above.

<!-- proof-1: body_md -->
The scaled coefficients $c_{m,N}=c_m/N^\gamma$ satisfy
\eqref{eq:bound} and have limiting functions $a_k(t)=\alpha_{k+1}t^\gamma$.
Apply Theorem~\ref{thm:ratios} and Proposition~\ref{prop:compact-zeros}
with $p=r$ and $b=0$. Each product in
\eqref{eq:system} then contains all $r$ phases, giving
\eqref{eq:original-ratio} for $t>0$. If $n/N\to0$, the power
bound $c_m\le K(1+m^\gamma)$ makes all scaled coefficients
up to $n+r$ tend uniformly to zero. The zero bound and
\eqref{eq:ratio-measure} give $\Phi(z,0)=z^{-r}$.
Substitution gives the equation for $F$.
Using the continuation outside the compact support, for sufficiently
large positive $z$ set $u=F(yt^{-\gamma})$ in
\eqref{eq:original-G}; since $t=(yuA(u))^{1/\gamma}$,
$$
 \int_0^1\frac{yt^{-\gamma}F'(yt^{-\gamma})}
                  {F(yt^{-\gamma})}\,d t
 =-\frac{y^{1/\gamma}}\gamma
     \int_0^{F(y)}(uA(u))^{1/\gamma}\,\frac{du}{u}.
$$
This proves \eqref{eq:original-explicit} near infinity.
The full expression agrees there with the Cauchy transform $G$,
so its continuation off the support is single-valued.

\label{lem:counting}
Let $n\ge1$ and compare two recurrences satisfying
\eqref{eq:recurrence} and the stated sign conditions. Set
$$
 K=\#\{0\le j<n:b_j\ne\widetilde b_j\}
   +\#\{0\le j<n-r:c_j\ne\widetilde c_j\}.
$$
Let $\mu_n,\widetilde\mu_n$ be their normalized zero measures, and put
$\nu_n=\mu_n$, $\widetilde\nu_n=\widetilde\mu_n$ when $r=1$;
when $r\ge2$, take their images under $\zeta\mapsto\zeta^d$.
For a real probability measure $\nu$, write
$F_\nu(x)=\nu((-\infty,x])$. Then
\begin{equation}\label{eq:counting}
 \sup_{x\in\mathbb R}|F_{\nu_n}(x)-F_{\widetilde\nu_n}(x)|
 \le\begin{cases}K/n,&r=1,\\[7pt] dK/n,&r\ge2.\end{cases}
\end{equation}

First take all $c_j>0$. For $r=1$ the zeros are real
and simple. For $r\ge2$, write $q_n(z)=z^sP(z^d)$, $0\le s<d$;
the zeros of $P$ are positive and simple by
\cite[Theorem~2.2]{BenRomdhane2008}; see also
\cite[Section~4.2]{DelvauxLopez2013}.
Interpolate one coefficient between its two endpoint values.
The subdiagonal coefficients stay positive throughout this interval, so
the relevant real roots remain simple and can be followed continuously.
The recurrence determinant is affine in each coefficient, which
occupies one matrix entry. For fixed real $x$, the value $q_n(x)$
when $r=1$, or $P(x)$ otherwise, is affine in the parameter and has
at most one zero, unless it vanishes identically. Since the zeros are
simple, at most one zero can pass through $x$ as the coefficient varies.
If this value vanishes identically as a function of the parameter,
a zero remains at $x$ and no other zero can reach it.
Each positive root of $P$ has weight $d/n$ in the real image.
Successive changes give the stated bounds. For nonnegative coefficients,
replace all $c_j$ by $c_j+\varepsilon$ and pass to the limit at
continuity points, then use right continuity.

<!-- proposition-2: body_md -->
\label{prop:local-ratios}
Under \eqref{eq:recurrence} and the stated sign conditions, let
$n_N\to\infty$, $a\in[0,\infty)^p$ and $h\in\mathbb R^p$,
with $h=0$ if $r\ge2$. The phases here are relative to $n_N$;
no congruence condition on $n_N$ is imposed. Suppose that
$$
 \begin{aligned}
 c_{n_N+j,N}&\to a_{j\bmod p}\quad\text{for every fixed integer }j\le p-r-1,\\[7pt]
 b_{n_N+j,N}&\to h_{j\bmod p}\quad\text{for every fixed integer }j\le p-1.
 \end{aligned}
$$
Then, locally uniformly on $\mathbb C\setminus\Sigma$,
$$
 \frac{q_{n_N,N}(z)}{q_{n_N+1,N}(z)}\to\varphi_0(z;a,h),
 \quad
 \frac{q_{n_N,N}(z)}{q_{n_N+p,N}(z)}\to\Phi(z;a,h),
$$
also after any fixed number of derivatives in $z$.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
The hypotheses control each fixed finite backward window, without
imposing a global coefficient bound. For $0\le j<p$, the coefficient
calculation gives
$[v^\ell]S_{n_N+j,N}\to[v^\ell]f_{j\bmod p}(v;a,h)$;
the same moment argument proves convergence of these shifted ratios,
and multiplication gives the second limit. The highest coefficient indices
entering $q_{n_N+p,N}$ are $n_N+p-r-1$ for $c$ and $n_N+p-1$ for $b$.

<!-- proof-4: body_md -->
For $M>0$, truncate the coefficients by
$$
 c^M=\min\{c,M\},\quad
 b^M=\max\{-M,\min\{b,M\}\},
$$
and denote the corresponding truncations of $a,h$ by $a^M,h^M$.
The truncated step functions converge in measure and are uniformly
bounded, so they converge in $L^1(0,U)$ for every $U>0$.
Proposition~\ref{prop:compact-zeros} applies to the recurrences with truncated coefficients.
If $K_N(M)$ is the sum of the numbers of changed diagonal and
subdiagonal coefficients entering $q_{n,N}$, use
$$
 \{A_{k,N}>M\}\subset
 \{a_k>M/2\}\cup\{|A_{k,N}-a_k|>M/2\},
$$
and the analogous inclusion for $|H_{k,N}|$ and $|h_k|$.
Each phase coefficient accounts for an interval of length $p/N$;
the initial and final incomplete intervals contribute $O(1)$ coefficients.
Replacing $n/N$ by $T$ changes the normalized count by
$O(|n/N-T|+N^{-1})$. Convergence in measure therefore gives
$$
 \limsup_{N\to\infty}\frac{K_N(M)}n
 \le\frac1{pT}\sum_{k=0}^{p-1}
 \bigl(|\{a_k>M/2\}\cap(0,T)|
       +|\{|h_k|>M/2\}\cap(0,T)|\bigr).
$$
The right-hand side tends to zero as $M\to\infty$ because the
functions $a_k,h_k$ are finite almost everywhere.
Lemma~\ref{lem:counting} therefore makes the distance between the
original and truncated real measures arbitrarily small in the
L\'evy metric, using their real images when $r\ge2$.
For almost every $t$, sufficiently large $M$ leaves every component
of $a(t)$ and $h(t)$ unchanged. Dominated convergence for bounded
continuous test functions shows that the averages
$\frac1T\int_0^T\mu_{a^M(t),h^M(t)}\,d t$ converge weakly to $\mu_T$.
The triangle inequality in the L\'evy metric gives \eqref{eq:zero-average}.
For $r\ge2$, weak convergence of the real images gives weak
convergence on the star by averaging each test function over the
$d$ roots of $u\ge0$.

Finally, the functions $z\mapsto(z-\zeta)^{-1}$ are uniformly bounded
and equicontinuous on compact subsets of $\mathbb C\setminus\Sigma$,
uniformly for $\zeta\in\Sigma$. Thus weak convergence gives the stated
locally uniform convergence. For every pair of coefficient vectors $a,h$
with finite entries,
\eqref{eq:potential} gives
$-\partial_z\Phi/(p\Phi)=G_{\mu_{a,h}}$.
The bound $|G_{\mu_{a,h}}(z)|\le\operatorname{dist}(z,\Sigma)^{-1}$
justifies averaging this identity without any integrability
assumption on the functions $a_k,h_k$.
