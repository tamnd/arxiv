<!-- head-1: body_md -->
We use the notation $\|\cdot\|$ to denote both the Euclidean norm of a vector and the spectral norm of a matrix. Given a matrix ${\bm{B}}\succeq {\bm{O}}_d$, we use
$\|\mathbf{v}\|_{\bm{B}}^2=\mathbf{v}^{\top}{\bm{B}}\mathbf{v}$ to denote the matrix ${\bm{B}}$-norm. Logarithms are natural unless their base is specified. We adopt the standard big ${\mathcal{O}}(\,\cdot\,)$ and $\Omega(\,\cdot\,)$ notations, and use $\tilde {\mathcal{O}}(\,\cdot\,)$ or $\tilde \Theta(\,\cdot\,)$ to hide the logarithmic factors.
<!-- end head-1 -->
<!-- asm-1: body_md -->
\label{asm:convex}
We assume that $f$ is convex, \textit{i.e.},
\[
f({\bm{x}}') \ge f({\bm{x}}) + \langle \nabla f({\bm{x}}), {\bm{x}}' - {\bm{x}} \rangle , \quad \forall {\bm{x}}, {\bm{x}}' \in {\mathbb{R}}^d.
\]
<!-- end asm-1 -->
<!-- asm-2: body_md -->
\label{asm:Hessian-Lip}
    We assume that $\nabla^2 f({\bm{x}})$ exists and is $L$-Lipschitz continuous ($L>0$), \textit{i.e.},
    \begin{align*}
        \Vert \nabla^2 f({\bm{x}}) - \nabla^2 f({\bm{x}}') \Vert \le L \Vert {\bm{x}}-  {\bm{x}}' \Vert, \quad \forall {\bm{x}},{\bm{x}}' \in {\mathbb{R}}^d.
    \end{align*}
<!-- end asm-2 -->
<!-- asm-3: body_md -->
\label{asm:bounded-minimizer}
We assume $f$ has at least one solution ${\bm{x}}^* \in \arg \min_{{\bm{x}} \in {\mathbb{R}}^d} f({\bm{x}})$.
<!-- end asm-3 -->
<!-- dfn-1: body_md -->
\label{dfn:sub-opt-point}
We call $\hat {\bm{x}} \in {\mathbb{R}}^d$  an $\epsilon$-solution to Problem (\ref{prob:main}) if $f(\hat {\bm{x}}) - f^* \le \epsilon$.
<!-- end dfn-1 -->
<!-- dfn-2: body_md -->
\label{dfn:alg-class}
An $m$-lazy second-order algorithm \texttt{A} operating on $f : {\mathbb{R}}^d \rightarrow {\mathbb{R}}$ consists of a sequence of measurable mappings $\{ \texttt{A}_t \}_{t=1}^T$ of the form (\ref{eq:alg-class}), where
the oracle responses satisfy 
\begin{align*}
 {\mathbb{O}}_t({\bm{x}}) =
 \begin{cases}
 (\nabla f({\bm{x}}), \nabla^2 f({\bm{x}})), &   t = m, 2m , \cdots, m \lfloor T/m \rfloor;\\
 \nabla f({\bm{x}}), &  \text{otherwise}.
 \end{cases}
\end{align*}
<!-- end dfn-2 -->
<!-- lem-1: body_md -->
\label{lem:block-zero-chain}
For any $m$-lazy second-order algorithm \texttt{A} operating on the instance $f_{2T}:{\mathbb{R}}^d \rightarrow {\mathbb{R}}$ defined in Eq.~(\ref{eq:block-zc}) with no more than $T$ iterations, for $d \ge 3T$, there exists an adversarial strategy to choose the vector set $\{ {\bm{v}}_t \}_{t=1}^{2T}$, such that
    \begin{align} \label{eq:ortho-vectors}
    \langle {\bm{x}}_T,{\bm{v}}_i\rangle =0,
    \qquad
    \forall i=T+1,\ldots,2T .
\end{align}
<!-- end lem-1 -->
<!-- lem-2: body_md -->
\label{lem:subgap}
If Eq. (\ref{eq:ortho-vectors}) holds, we have
\begin{align} \label{eq:lb-suboptimality-gap}
    f_{2T}({\bm{x}}_T) \ge \frac{\lambda a^3}{m^{3/2} \left( \lfloor T/m \rfloor +1  \right)^2}.
\end{align}
<!-- end lem-2 -->
<!-- thm-1: body_md -->
\label{thm:lower-bound-convex}
Given integers $1\le m\le T$ and any parameters $L>0$ and $D >0$, setting the parameters 
$\lambda =L/96$ and $a = D / \sqrt{2 T}$ for the instance  $f_{2T}:{\mathbb{R}}^d \rightarrow {\mathbb{R}}$ defined in Eq.~(\ref{eq:block-zc}) gives a function that has $L$-Lipschitz continuous Hessians and satisfies $\Vert {\bm{x}}^* \Vert \le D$, such that,
any $m$-lazy second-order algorithm \texttt{A} with $T$ iterations,
operating on $f_{2T}$ can not find an $\epsilon$-solution unless $T = \Omega( m^{1/7} (L D^3  / \epsilon)^{2/7} )$.
<!-- end thm-1 -->
<!-- dfn-3: body_md -->
\label{dfn:det-alg-class}
An deterministic second-order algorithm with unbalanced Hessian and gradient evaluations \texttt{A} operating on $f : {\mathbb{R}}^d \rightarrow {\mathbb{R}}$ consists of a sequence of measurable mappings $\{ \texttt{A}_t \}_{t=1}^T$ such that $\texttt{A}_t$ takes in the first $t$ oracle responses to produce the next iterate ${\bm{x}}_t$,  \textit{i.e.}, \texttt{A} generates a sequence $\{{\bm{x}}_t\}_{t=1}^{T}$ such that
\begin{align} \label{eq:alg-class}
    {\bm{x}}_t = \texttt{A}_t ( {\mathbb{O}}_1({\bm{x}}_0), {\mathbb{O}}_2({\bm{x}}_1), \cdots, {\mathbb{O}}_{t}({\bm{x}}_{t-1}) ), \quad t \ge 1; \qquad {\bm{x}}_0 = {\bm{0}},
\end{align}
where the oracle responses depend on the type $\phi_t \in \{ 1,2\}$:
\[
{\mathbb{O}}_t({\bm{x}}) =
 \begin{cases}
\nabla^2 f({\bm{x}}) , & \text{if} \quad \phi_t = 1; \\
 \nabla f({\bm{x}}), & \text{if} \quad \phi_t = 2.
 \end{cases}
\]
We let $n_1$ and $n_2$ be the total number of $\phi_t = 1$ and $\phi_t=2$, respectively.
<!-- end dfn-3 -->
<!-- lem-3: body_md -->
\label{lem:upper-schedule-simulation}
For $n_2 \in [0,n_1]$, any general second-order algorithm $\texttt{A}$ using $n_1$ gradient queries and $n_2$ Hessian
queries can be simulated by an $m$-lazy algorithm $\texttt{A}'$ in
$n_1+m n_2$ iterations.
<!-- end lem-3 -->
<!-- proof-1: body_md -->
When a gradient is needed for $\texttt{A}$, query it at the next iteration and discard the extra Hessian information if that
iteration is a snapshot. When a Hessian at a chosen point is needed for $\texttt{A}$, repeat queries at that point
until the next snapshot and discard the intervening responses. The
waiting time is fewer than $m$ iterations. Therefore, the total number of iterations of $\texttt{A}'$ is at most $n_1+m n_2$.
<!-- end proof-1 -->
<!-- thm-2: body_md -->
\label{thm:ALEN}
Under Assumptions~\ref{asm:convex},~\ref{asm:Hessian-Lip}, and~\ref{asm:bounded-minimizer},
                                                                   
setting the parameters
\begin{align} \label{eq:param-alen}
    \gamma = L / m^3, \quad T = {\mathcal{O}}( ( \gamma D^3 / \epsilon )^{2/7} ), \quad S = {\mathcal{O}}(\log m), \quad N = \tilde {\mathcal{O}}(m),
\end{align}
then Algorithm \ref{alg:A-LEN}
returns an $\epsilon$-solution in total gradient evaluations of
\begin{equation}\label{eq:upper-main}
\mathcal{O}\!\left(
\left(m+m^{1/7}\left(\frac{L D^3}{\epsilon}\right)^{2/7}\right)
\log^{11/3} m \right),
\end{equation}
where $D = \Vert {\bm{x}}_0 - {\bm{x}}^* \Vert$ is the initial distance to an arbitrary minimizer ${\bm{x}}^*$.
<!-- end thm-2 -->
<!-- cond-1: body_md -->
\label{cond:cubic-proxi}
The cubic proximal oracle ${\bm{y}} = \texttt{Cubic-Proximal}(f,{\bm{x}},\gamma,S)$ always returns an output ${\bm{y}} \in {\mathbb{R}}^d$ satisfies the following Monteiro--Svaiter (MS) \citep{monteiro2012iteration,monteiro2013accelerated} condition
\begin{equation}\label{eq:upper-ms-condition}
\|\nabla f_{{\bm{x}},\gamma}({\bm{y}})\|
\le \frac{\gamma}{32} \|{\bm{y}}-{\bm{x}}\|^2.
\end{equation}
<!-- end cond-1 -->
<!-- lem-4: body_md -->
\label{lem:A-NPE}
Under Assumption \ref{asm:convex} and \ref{asm:bounded-minimizer}, if the \texttt{Cubic-Proximal} oracle always return an output satisfying Condition \ref{cond:cubic-proxi}, then running Algorithm \ref{alg:A-LEN} guarantees $f({\bm{x}}_T) - f^* \le \epsilon$ in 
\begin{equation}\label{eq:upper-outer-count}
T = \mathcal{O}\!\left(
\left(\frac{\gamma D^3}{\epsilon}\right)^{2/7}\right)
\end{equation}
iterations, where 
$D = \Vert {\bm{x}}_0 - {\bm{x}}^* \Vert$ is the initial distance to an arbitrary minimizer ${\bm{x}}^*$.
<!-- end lem-4 -->
<!-- lem-5: body_md -->
\label{lem:sub-prob-grad-dom}
Under Assumptions \ref{asm:convex} and~\ref{asm:Hessian-Lip}, the function $f_{{\bm{x}},\gamma}({\bm{y}})$ defined in Eq. (\ref{eq:upper-cubic-objective}) satisfies:
                                                                                                                                                                                                                                                                                                                 
\begin{enumerate}
    \item $f_{{\bm{x}},\gamma}({\bm{y}})$ has $(L+2 \gamma)$-Lipschitz continuous Hessians.
    \item For any ${\bm{y}} \in {\mathbb{R}}^d$,
    \begin{align}
        \Vert \nabla f_{{\bm{x}}, \gamma}({\bm{y}}) \Vert^{3/2} \ge \frac{3}{2} \sqrt{\frac{\gamma}{2}} (f_{{\bm{x}}, \gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^*)
        \geq \frac{1}{2}\left( \frac{\gamma}{2} \Vert {\bm{y}} - {\bm{y}}_{{\bm{x}},\gamma}^* \Vert^2 \right)^{3/2}.\label{eq:cubic-grad-uniform}
    \end{align}
\end{enumerate}
<!-- end lem-5 -->
<!-- lem-6: body_md -->
\label{lem:crn}
Under Assumptions~\ref{asm:convex} and~\ref{asm:Hessian-Lip}, set
$H=L+2\gamma$. For ${\bm{y}}=\mathcal{C}_H({\bm{z}})$, we have the following:
\begin{enumerate}
    \item $\|\nabla f_{{\bm{x}},\gamma}({\bm{y}})\|^{3/2}
    \le 3\sqrt H\bigl(f_{{\bm{x}},\gamma}({\bm{z}})-f_{{\bm{x}},\gamma}^*\bigr)$.
    \item $f_{{\bm{x}},\gamma}({\bm{y}})-f_{{\bm{x}},\gamma}^*
    \le H\|{\bm{z}}-{\bm{y}}_{{\bm{x}},\gamma}^*\|^3/3$.
\end{enumerate}
<!-- end lem-6 -->
<!-- cond-2: body_md -->
\label{cond:gapreduce}
Under Assumptions \ref{asm:convex} and \ref{asm:Hessian-Lip}, for any $\Delta_s>0$ and ${\bm{y}}_s$ satisfying
$f_{{\bm{x}},\gamma}({\bm{y}}_s)-f_{{\bm{x}},\gamma}^*\le\Delta_s$,
the subprocedure ${\bm{y}}_{s+1}=\texttt{PAGD-Seg}
(f_{{\bm{x}},\gamma},{\bm{y}}_s,\Delta_s,H,\gamma,N)$ returns a point satisfying
\[
f_{{\bm{x}},\gamma}({\bm{y}}_{s+1})-f_{{\bm{x}},\gamma}^*
\le\Delta_{s+1}:=\frac34\Delta_s.
\]
<!-- end cond-2 -->
<!-- lem-7: body_md -->
\label{lem:reduce-gap}
Under Assumptions~\ref{asm:convex} and~\ref{asm:Hessian-Lip}, let
$\gamma>0$, $H=L+2\gamma$, and $\kappa=H/\gamma$.
If the \texttt{PAGD-Seg} subprocedure always ensures Condition~\ref{cond:gapreduce}, then
Algorithm~\ref{alg:MS-solver}, with
\[
S=\left\lceil\frac{\log(50\cdot384^{3/2}\kappa^2)}{\log(4/3)}\right\rceil,
\]
returns $\hat{\bm{y}}$ satisfying Condition~\ref{cond:cubic-proxi}, \textit{i.e.}, $ \| \nabla f_{{\bm{x}},\gamma}(\hat {\bm{y}})\| \le \gamma\| \hat {\bm{y}} - {\bm{x}} \|^2/32 $.
<!-- end lem-7 -->
<!-- lem-8: body_md -->
\label{lem:upper-snapshot-distance}
Under Assumptions~\ref{asm:convex} and \ref{asm:Hessian-Lip}, for any ${\bm{y}} \in {\mathbb{R}}^d$,
\begin{equation}\label{eq:upper-snapshot-distance}
({\bm{y}} - {\bm{y}}_{{\bm{x}},\gamma}^*)^{\top} \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) ({\bm{y}} - {\bm{y}}_{{\bm{x}},\gamma}^*) 
\le 6(1+\kappa^{2/3})  (f_{{\bm{x}},\gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^*).
\end{equation}
<!-- end lem-8 -->
<!-- proof-2: body_md -->
Let ${\bm{d}}  = {\bm{y}} - {\bm{y}}_{{\bm{x}},\gamma}^*$. The claim is immediate when ${\bm{d}}=0$.
Otherwise, by stationarity at ${\bm{y}}_{{\bm{x}},\gamma}^*$ and Hessian $H$-Lipschitzness  we obtain
\begin{align*}
    f_{{\bm{x}},\gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^*  &=\int_0^1 t {\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}} - t {\bm{d}}) {\bm{d}} {\rm d} t\\
    &\geq \int_0^1 t \max\{0,{\bm{d}}^\top (\nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) - Ht\Vert{\bm{d}}\Vert{\bm{I}}_d) {\bm{d}}\} {\rm d} t\\
    &=
    \begin{cases} 
        \dfrac{1}{2}{\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}} - \dfrac{H}{3}\Vert{\bm{d}}\Vert^3, &{\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}}\geq H\Vert{\bm{d}}\Vert^3;\\[2mm]
        \dfrac{1}{6}({\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}})^3/(H\Vert{\bm{d}}\Vert^3)^2, &{\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}} < H\Vert{\bm{d}}\Vert^3.
    \end{cases}
\end{align*}
If ${\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}}\geq H\Vert{\bm{d}}\Vert^3$, we have
\begin{align*}
    f_{{\bm{x}},\gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^*  &\geq \frac{1}{2}{\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}} - \frac{H}{3}\Vert{\bm{d}}\Vert^3 \geq \frac{1}{6}{\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}}.
\end{align*}
If ${\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}} < H\Vert{\bm{d}}\Vert^3$, by item 2 of Lemma \ref{lem:sub-prob-grad-dom}, we have $ H \| {\bm{d}} \|^3 \le 6\kappa(f_{{\bm{x}},\gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^*) $. Therefore, 
\begin{align*}
    f_{{\bm{x}},\gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^*  &\geq \frac{1}{6}({\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}})^3/(H\Vert{\bm{d}}\Vert^3)^2\\
    &\geq \frac{1}{6}({\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}})^3/(6\kappa(f_{{\bm{x}},\gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^*))^2,
\end{align*}
so $f_{{\bm{x}},\gamma}({\bm{y}}) - f_{{\bm{x}},\gamma}^* \geq (1/{6\kappa^{2/3}}){\bm{d}}^\top \nabla^2 f_{{\bm{x}},\gamma}({\bm{y}}) {\bm{d}}$. In both cases, we have inequality (\ref{eq:upper-snapshot-distance}) holds.
<!-- end proof-2 -->
<!-- lem-9: body_md -->
\label{lem:upper-extrapolated-curvature}
Under Assumptions \ref{asm:convex} and \ref{asm:Hessian-Lip}, for $\Delta>0 $ satisfies $ f_{{\bm{x}},\gamma}({\bm{y}})-f_{{\bm{x}},\gamma}^* \le \Delta$, let
\begin{equation}\label{eq:upper-metric}
{\bm{B}}=2 (J+1) \nabla^2f_{{\bm{x}},\gamma}({\bm{y}})+100 (J+1)^{4/3}(H^2 \Delta)^{1/3} {\bm{I}}_d.
\end{equation}
Then
\begin{equation}\label{eq:upper-sublevel-smoothness}
\nabla^2 g({\bm{v}})\preceq {\bm{B}}, \qquad \forall {\bm{v}} \in \{ {\bm{v}}' \in {\mathbb{R}}^d: g({\bm{v}}') \le g({\bm{y}}) \}.
\end{equation}
<!-- end lem-9 -->
<!-- proof-3: body_md -->
Recalling that $\lambda =  1/64$, we define $W = \sum_{i=0}^J \lambda^i \le 64/63$. Suppose $g({\bm{v}})\le g({\bm{y}})$, so
\begin{align}\label{eq:upper-individual-gaps}
    0\le f_{{\bm{x}},\gamma}({\bm{v}}_i)-f_{{\bm{x}},\gamma}^*
\le\lambda^{-i}\bigl(g({\bm{v}})-Wf_{{\bm{x}},\gamma}^*\bigr)
\le\lambda^{-i}\bigl(g({\bm{y}})-Wf_{{\bm{x}},\gamma}^*\bigr)
\le W\Delta\lambda^{-i},
\end{align}
For $\Delta>0$ satisfying $f_{{\bm{x}},\gamma}({\bm{y}})-f_{{\bm{x}},\gamma}^*\le \Delta$, we know from Lemma \ref{lem:sub-prob-grad-dom}(2), $\|{\bm{y}}-{\bm{y}}_{{\bm{x}},\gamma}^*\|$ is bounded by $R\triangleq(6\Delta/\gamma)^{1/3}$. Then, applying the second inequality of Lemma \ref{lem:sub-prob-grad-dom}(2),
we have
\begin{equation}\label{eq:upper-individual-distance}
\|{\bm{v}}-{\bm{y}}\|\le\|{\bm{v}}-{\bm{y}}_{{\bm{x}},\gamma}^*\|+\|{\bm{y}}-{\bm{y}}_{{\bm{x}},\gamma}^*\|\le(1+W^{1/3})R\le3R.
\end{equation}
For $i<J$, the adjacent extrapolated points satisfy the relationship 
${\bm{v}}_i=  ({\bm{y}}+ J {\bm{v}}_{i+1}) / (J+1)$.
For every unit vector $\mathbf{e}$ and $t>0$, convexity gives
\[
f_{{\bm{x}},\gamma}({\bm{v}}_i\pm t\mathbf{e})
\le  \frac{1}{J+1} \left(f_{{\bm{x}},\gamma}({\bm{y}} \pm (J+1) t\mathbf{e})
+ J f_{{\bm{x}},\gamma}({\bm{v}}_{i+1}) \right).
\]
Adding these inequalities and applying the second-order Taylor bounds
\begin{align*}
    \left|f_{{\bm{x}},\gamma}({\bm{u}}+{\bm{h}}) + f_{{\bm{x}},\gamma}({\bm{u}}-{\bm{h}}) - 2f_{{\bm{x}},\gamma}({\bm{u}}) - {\bm{h}}^\top \nabla^2f_{{\bm{x}},\gamma} ({\bm{u}}){\bm{h}}\right|
    \leq  \frac{H}{3} \Vert{\bm{h}}\Vert^3
\end{align*}
at ${\bm{y}}$ and ${\bm{v}}_i$ yields
\begin{align*}
    &2f_{{\bm{x}},\gamma}({\bm{v}}_i) + t^2\mathbf{e}^{\top}\nabla^2f_{{\bm{x}},\gamma}({\bm{v}}_i)\mathbf{e}-\frac{H}{3}t^3
    \\
    \leq & \frac{2}{J+1}f_{{\bm{x}},\gamma}({\bm{y}}) + (J+1)t^2\mathbf{e}^{\top} \nabla^2f_{{\bm{x}},\gamma}({\bm{y}}) \mathbf{e} + \frac{H}{3}(J+1)^2t^3 + \frac{2J}{J+1}f_{{\bm{x}},\gamma}({\bm{v}}_{i+1}).
\end{align*}
Next, we can use inequality (\ref{eq:upper-individual-gaps}) to obtain that
\begin{align*}
&\mathbf{e}^{\top}\nabla^2f_{{\bm{x}},\gamma}({\bm{v}}_i)\mathbf{e} \\
&\le (J+1) \mathbf{e}^{\top}  \nabla^2f_{{\bm{x}},\gamma}({\bm{y}}) \mathbf{e}+\frac{H(1+(J+1)^2)}3t
+\frac{2}{ t^2 (J+1)}(f_{{\bm{x}},\gamma}({\bm{y}})+J f_{{\bm{x}},\gamma}({\bm{v}}_{i+1})-(J+1)f_{{\bm{x}},\gamma}({\bm{v}}_i))\\
&\le (J+1)\mathbf{e}^{\top}  \nabla^2f_{{\bm{x}},\gamma}({\bm{y}}) \mathbf{e}+\frac{H(1+(J+1)^2)}3t
+\frac{2}{t^2 (J+1)}(f_{{\bm{x}},\gamma}({\bm{y}})-f_{{\bm{x}},\gamma}^*+J(f_{{\bm{x}},\gamma}({\bm{v}}_{i+1})-f_{{\bm{x}},\gamma}^*))\\
&\le (J+1) \mathbf{e}^{\top}  \nabla^2f_{{\bm{x}},\gamma}({\bm{y}}) \mathbf{e}+\frac{H(1+(J+1)^2)}3t
+\frac{2}{t^2 (J+1)}(\Delta+ J W \Delta \lambda^{-(i+1)})\\
&\le (J+1)\mathbf{e}^{\top}  \nabla^2f_{{\bm{x}},\gamma}({\bm{y}}) \mathbf{e}+\frac{H(1+(J+1)^2)}3t+\frac{2W \Delta \lambda^{-(i+1)}}{t^2}.
\end{align*}
Choose $t=(\Delta \lambda^{-(i+1)}/(H (J+1)^2))^{1/3}$ and write
$\delta=(H^2 \Delta )^{1/3}$. Since $J \ge 1$,
\begin{align} \label{eq:upper-adjacent-curvature}
\begin{split}

\preceq & (J+1) \nabla^2f_{{\bm{x}},\gamma}({\bm{y}}) +\left(2W+\frac{1+(J+1)^2}{3 (J+1)^2}\right) (J+1)^{4/3}H^{2/3} \Delta^{1/3}\lambda^{-(i+1)/3} {\bm{I}}_d \\
\preceq & (J+1) \nabla^2f_{{\bm{x}},\gamma}({\bm{y}})+\frac52 (J+1)^{4/3}\delta\lambda^{-(i+1)/3} {\bm{I}}_d.
\end{split}    
\end{align}
                              
For $c = (1+1/J)$, multiplying by $\lambda^ic^{2i}$ and summing over $i<J$, using Hessian Lipschitzness in Assumption~\ref{asm:Hessian-Lip} and inequality~(\ref{eq:upper-individual-distance}), we can deduce that
\begin{align*}

    &\preceq \sum_{i=0}^{J-1} \lambda^i c^{2i} \left( (J+1) \nabla^2f_{{\bm{x}},\gamma}({\bm{y}})+\frac52 (J+1)^{4/3}\delta\lambda^{-(i+1)/3}{\bm{I}}_d\right)

We first need a descent property in the sublevel set.
For any ${\bm{y}}'$ with $g({\bm{y}}')\le g({\bm{y}})$, put ${\bm{h}}=\nabla g({\bm{y}}')$ and $\phi(t)=g({\bm{y}}'-t{\bm{B}}^{-1}{\bm{h}})$ for $t\in[0,1]$. Whenever $\phi(t_0)\le\phi(0)$, convexity places the whole segment up to $t_0$ in the initial sublevel set. Lemma~\ref{lem:upper-extrapolated-curvature} then gives
\begin{align*}
\phi(t_0)-\phi(0)
&=-t_0\|{\bm{h}}\|_{{\bm{B}}^{-1}}^2
+\int_0^{t_0}(t_0-t)({\bm{B}}^{-1}{\bm{h}})^\top
\nabla^2g({\bm{y}}'-t{\bm{B}}^{-1}{\bm{h}})({\bm{B}}^{-1}{\bm{h}})\,\mathrm{d}t\\
&\le-(t_0-t_0^2/2)\|{\bm{h}}\|_{{\bm{B}}^{-1}}^2.
\end{align*}
If $\phi(1)>\phi(0)$, then $\phi'(0)=-\|{\bm{h}}\|_{{\bm{B}}^{-1}}^2<0$ implies that some $t_0\in(0,1)$ satisfies $\phi(t_0)=\phi(0)$, contradicting this strict decrease. Thus $\phi(1)\le\phi(0)$, and taking $t_0=1$ yields
\begin{equation}\label{eq:upper-preconditioned-descent}
g({\bm{y}}'-{\bm{B}}^{-1}{\bm{h}})\le g({\bm{y}}')-\|{\bm{h}}\|_{{\bm{B}}^{-1}}^2/2, \quad \forall \{ {\bm{y}}' \in {\mathbb{R}}^d: g({\bm{y}}') \le g({\bm{y}}) \}.
\end{equation}
Starting from ${\bm{x}}_0={\bm{y}}$, Condition~\ref{cond:segment-search} and inequality (\ref{eq:upper-preconditioned-descent}) imply inductively that
\[
g({\bm{x}}_{k+1})\le g({\bm{y}}_k)\le g({\bm{x}}_k)\le g({\bm{y}}).
\]
For the approximate minimizer $\bar{{\bm{v}}}$ in Lemma~\ref{lem:comparator}, define the Lyapunov function of AGD as
\[
E_k=A_k\bigl(g({\bm{x}}_k)-g(\bar{{\bm{v}}})\bigr)
+\|{\bm{v}}_k-\bar{{\bm{v}}}\|_{{\bm{B}}}^2/2.
\]
Therefore, the decrement of the Lyapunov function can be decomposed into two parts
\begin{align*}
E_{k+1} - E_k =& \underbrace{A_{k+1} (g({\bm{x}}_{k+1}) - g(\bar {\bm{v}})) - A_k (g({\bm{x}}_k) - g(\bar {\bm{v}}))}_{\rm (I)} \\    
+& \underbrace{\|{\bm{v}}_{k+1}-\bar{{\bm{v}}}\|_{{\bm{B}}}^2/2 - \|{\bm{v}}_k-\bar{{\bm{v}}}\|_{{\bm{B}}}^2/2 }_{\rm (II)}.
\end{align*}
                                                                                                                                   
                
                                                                                                                                  
                                                  
              

                
                                                                                                                    
                                                                                                
              
For part (I), we can use inequality (\ref{eq:upper-preconditioned-descent}) and the fact that $A_{k+1} = A_k + a_{k+1}$ to obtain
\begin{align*}
{\rm (I)} \le & A_{k+1} (g({\bm{y}}_k) - g(\bar {\bm{v}})) - \frac{a_{k+1}^2}{2} \|{\bm{h}}_k \|_{{\bm{B}}^{-1}}^2  - A_k(g({\bm{x}}_k) -g(\bar {\bm{v}})) \\
=& A_k (g({\bm{y}}_k) - g({\bm{x}}_k)) + a_{k+1} (g({\bm{y}}_k) - g(\bar{{\bm{v}}})) - \frac{a_{k+1}^2}{2} \|{\bm{h}}_k \|_{{\bm{B}}^{-1}}^2.
\end{align*}
For part (II), substituting the update rule ${\bm{v}}_{k+1} = {\bm{v}}_k - a_{k+1} {\bm{B}}^{-1} {\bm{h}}_k$ into the quadratic term yields
\begin{align*}
{\rm (II)} =& \frac{1}{2} \|{\bm{v}}_k - a_{k+1} {\bm{B}}^{-1} {\bm{h}}_k - \bar{{\bm{v}}}\|_{{\bm{B}}}^2 - \frac{1}{2} \|{\bm{v}}_k - \bar{{\bm{v}}}\|_{{\bm{B}}}^2 = -a_{k+1} \langle {\bm{h}}_k, {\bm{v}}_k - \bar{{\bm{v}}} \rangle + \frac{a_{k+1}^2}{2} \|{\bm{h}}_k\|_{{\bm{B}}^{-1}}^2.
\end{align*}
Combining (I) and (II), the squared-gradient term ${a_{k+1}^2} \|{\bm{h}}_k\|_{{\bm{B}}^{-1}}^2/2$ cancels out exactly, giving
\begin{align*}
E_{k+1} - E_k \le A_k \bigl(g({\bm{y}}_k) - g({\bm{x}}_k)\bigr) + a_{k+1} \bigl(g({\bm{y}}_k) - g(\bar{{\bm{v}}}) - \langle {\bm{h}}_k, {\bm{v}}_k - \bar{{\bm{v}}} \rangle\bigr).
\end{align*}
Condition~\ref{cond:segment-search} gives $g({\bm{y}}_k)\le g({\bm{x}}_k)$, while convexity gives $g({\bm{y}}_k)-g(\bar{{\bm{v}}})\le\langle{\bm{h}}_k,{\bm{y}}_k-\bar{{\bm{v}}}\rangle$. Since $A_k\ge0$, substituting these bounds and using the second inequality in Condition~\ref{cond:segment-search} yields
\begin{align*}
E_{k+1}-E_k
&\le a_{k+1}\bigl(\langle{\bm{h}}_k,{\bm{y}}_k-\bar{{\bm{v}}}\rangle-\langle{\bm{h}}_k,{\bm{v}}_k-\bar{{\bm{v}}}\rangle\bigr) =a_{k+1}\langle{\bm{h}}_k,{\bm{y}}_k-{\bm{v}}_k\rangle
\le a_{k+1}\zeta.
\end{align*}
                                                                                                                                                
                    
                
             
                                       
                                                                                 
                                             
                   
              
Since $A_0=0$, we have \(E_0=\|{\bm{y}}-\bar{{\bm{v}}}\|_{{\bm{B}}}^2/2\). Therefore, telescoping the above inequality yields
\begin{equation}\label{eq:upper-potential-bound}
E_k\le \|{\bm{y}}-\bar{{\bm{v}}}\|_{{\bm{B}}}^2/2 +A_k\zeta, \quad \forall 0\le k\le N.
\end{equation}
Note that we can prove $A_k \ge (k+1)^2/4$ for all $ k\ge 1$ by induction. The base case $A_1 =1$ is trivial. Assume $A_k \ge (k+1)^2/4$, then we have
\[
\sqrt{A_{k+1}} = \frac{1 + \sqrt{1 + 4A_k}}{2} 
\ge \frac{1 + \sqrt{1 + (k+1)^2}}{2} 
> \frac{1 + (k+1)}{2} 
= \frac{k+2}{2}.
\]
Squaring both sides gives \(A_{k+1} \ge (k+2)^2/{4}\), which completes the induction step. Thus, $A_k \ge {(k+1)^2}/{4}$ holds for all $k \ge 1$. Then, using Lemma~\ref{lem:comparator}(2), we obtain
\begin{equation}\label{eq:upper-accelerated-error}
g({\bm{x}}_N)-g(\bar{{\bm{v}}})
\le\frac{\|{\bm{y}}-\bar{{\bm{v}}}\|_{{\bm{B}}}^2}{2A_N}+\zeta
\le\left(\frac{180}{4096}+\frac1{100}\right)\Delta<0.054\Delta.
\end{equation}
Let $W=\sum_{i=0}^J\lambda^i$. Since every summand of $g({\bm{x}}_N)-Wf_{{\bm{x}},\gamma}^*$ is nonnegative and its $i=0$ summand is $f_{{\bm{x}},\gamma}({\bm{x}}_N)-f_{{\bm{x}},\gamma}^*$, Lemma~\ref{lem:comparator}(1) gives
\[
f_{{\bm{x}},\gamma}({\bm{x}}_N)-f_{{\bm{x}},\gamma}^*
\le g({\bm{x}}_N)-Wf_{{\bm{x}},\gamma}^*

\]
Lemma~\ref{lem:finite-segment-search} proves that each search uses $\mathcal{O}(\log\kappa)$ such gradients, yielding the stated total count.

For the function $f_{2T}: {\mathbb{R}}^d \rightarrow {\mathbb{R}}$ defined in Eq. (\ref{eq:block-zc}), we define 
${\bm{y}} \in {\mathbb{R}}^{2T}$ be defined by the vector whose $i$th element is

\end{align*}
Next, we show that there exists an adversarial strategy of choosing the orthonormal vector set $\{{\bm{v}}_t \}_{t=1}^{2T}$ such that Eq. (\ref{eq:ortho-vectors}) holds. The key observation is that the chain-like structure of the function $f_{2T}$ allows the adversary to simulate the query of any $t$-iteration deterministic algorithm \texttt{A} of the form (\ref{eq:alg-class}) with the truncated function (using the same notation in rotated coordinates and for its pullback to ${\bm{x}}$) that only depends on ${\bm{v}}_1,\cdots,{\bm{v}}_t$:
\begin{align} \label{eq:hard-ins-truncated-y}
\begin{split}
    f_t({\bm{y}})
    \triangleq& f_{2T}({\bm{y}}) \mid \{ y_{[i]}  = 0, ~i  = t+1,\cdots,2T\} \\
    =& \lambda \left( (a- y_{[1]})^2 + \sum_{i=1}^{m-1} (y_{[i]} - y_{[i+1]})^2  \right)^{3/2} \\

\end{split}                                                                                                                                                                                                 
\end{align}
Here the displayed expansion applies to $m\le t<2T$. For $1\le t<m$, it is instead
\[
f_t({\bm{y}})=\lambda\left((a-y_{[1]})^2+\sum_{i=1}^{t-1}(y_{[i]}-y_{[i+1]})^2+y_{[t]}^2\right)^{3/2}.
\]
In either case, $f_t$ is independent of $y_{[t+1]},\ldots,y_{[2T]}$.
Specifically, the adversary can define the vectors ${\bm{v}}_1,\cdots,{\bm{v}}_{2T}$ as follows:
\begin{enumerate}
    \item At the initialization ${\bm{x}}_0 = {\bm{0}}$, pick any unit vector ${\bm{v}}_1 \in {\mathbb{R}}^{d}$, and return
    the oracle of $f_1$ at ${\bm{x}}_0$. Its gradient is $-3\lambda a^2{\bm{v}}_1$; when $m=1$, also return its Hessian $6\lambda a{\bm{v}}_1{\bm{v}}_1^\top$.
                                                                                                                                                                                  
    \item For any $t = 1,\cdots,T-1$, repeat the following process: The adversary
    simulates the current query~${\bm{x}}_t $  by executing the same update of algorithm $\texttt{A}_t$, and then picks a new unit ${\bm{v}}_{t+1} \in {\mathbb{R}}^{d}$ that is orthogonal to all previous basis ${\bm{v}}_{1},\cdots,{\bm{v}}_{t}$ and queries ${\bm{x}}_1,\cdots,{\bm{x}}_t$:

            \langle {\bm{v}}_{t+1}, {\bm{v}}_i \rangle = 0 \quad {\rm and} \quad \langle {\bm{v}}_{t+1}, {\bm{x}}_i \rangle =0, \quad \forall i = 1,\cdots,t.
        \end{align}
        Such a vector ${\bm{v}}_{t+1} \in {\mathbb{R}}^d$ always exists when $d \ge 2t+1$. After that, the adversary returns its oracle using that of the truncated function~$f_{t+1}$ rather than the full function $f_{2T}$. 
    \item Finally, for the algorithm output ${\bm{x}}_T$, the adversary picks an orthonormal vector set $\{{\bm{v}}_{T+1},\cdots, {\bm{v}}_{2T}\}$ that is orthogonal to the subspace ${\rm span}\{{\bm{x}}_1,\cdots, {\bm{x}}_T, {\bm{v}}_1,\cdots,{\bm{v}}_T\}$, which is possible when $d \ge 3T$.
\end{enumerate}
It remains to prove that, in the $t$-th iteration, the oracle of the truncated function $f_{t}$ returned by the above adversarial strategy is consistent with the oracle of the full function $f_{2T}$ at its query point ${\bm{x}}_{t-1}$. In the following, we verify this property for both gradient and Hessian queries:
                                                                                                                                                        
\paragraph{Consistency of Gradient queries.} For $t=1,\ldots,T$, the $t$th response is evaluated at ${\bm{x}}_{t-1}$. Write $(y_{t-1})_{[i]}=\langle{\bm{x}}_{t-1},{\bm{v}}_i\rangle$. The adversary ensures that $(y_{t-1})_{[i]}=0$ for all $i=t,\ldots,2T$. The restriction defining $f_t$ preserves all partial derivatives in coordinates $1,\ldots,t$. For $i>t$, every edge difference contributing to the $i$th gradient coordinate is zero. Consequently,
\begin{align*}
    \frac{\partial f_t({\bm{y}}_{t-1})}{\partial y_{[i]}}=\frac{\partial f_{2T}({\bm{y}}_{t-1})}{\partial y_{[i]}},\quad i=1,\ldots,2T.
\end{align*}
Thus the gradient responses are consistent.
\paragraph{Consistency of Hessian queries.} For $t=m,2m,\ldots,m\lfloor T/m\rfloor$, the algorithm queries ${\mathbb{O}}_t({\bm{x}}_{t-1})$. At these block boundaries, the truncated function is
\begin{align*}
f_t({\bm{y}})=&\lambda\left((a-y_{[1]})^2+\sum_{i=1}^{m-1}(y_{[i]}-y_{[i+1]})^2\right)^{3/2}+\lambda\sum_{s=1}^{t/m-1}\left(\sum_{i=sm}^{(s+1)m-1}(y_{[i]}-y_{[i+1]})^2\right)^{3/2}+\lambda|y_{[t]}|^3.
\end{align*}
The displayed complete blocks coincide with those of $f_{2T}$. All remaining blocks of $f_{2T}$ have zero edge differences at ${\bm{y}}_{t-1}$, and $y_{[t]}=0$ there. Since the gradient and Hessian of $\|\cdot\|^3$ vanish at zero, the terminal term of $f_t$ and these remaining blocks have zero gradients and Hessians. Therefore,
\begin{align*}
\frac{\partial^2 f_t({\bm{y}}_{t-1})}{\partial y_{[i]}\partial y_{[j]}}=\frac{\partial^2 f_{2T}({\bm{y}}_{t-1})}{\partial y_{[i]}\partial y_{[j]}},\quad i,j=1,\ldots,2T.
\end{align*}
Hence the full Hessian responses are consistent at every Hessian query point ${\bm{x}}_{t-1}$.

\begin{align*}

From the form of the function $f_{2T}({\bm{x}})$ defined in Eq. (\ref{eq:block-zc}), it is easy to see that its minimizer ${\bm{x}}^*$ is achieved under the following constraint:
\begin{align*}
    \langle {\bm{x}}^*, {\bm{v}}_i \rangle = a, \quad \forall i = 1,\cdots,2T.
\end{align*}
Taking ${\bm{x}}^*=a\sum_{i=1}^{2T}{\bm{v}}_i$, we have $\Vert {\bm{x}}^* \Vert = a \sqrt{2T}$, which means that $\Vert {\bm{x}}^* \Vert \le D$ if we set $a = D / \sqrt{2T}$. Next, we justify the choice of~$\lambda$ that can ensure the function to have $L$-Lipschitz continuous Hessians. Note that $f_{2T}({\bm{x}})$ is the summation of $\lfloor 2T/m \rfloor+1$ blocks, and each coordinate only appears at most in two adjacent blocks. Therefore, it suffices to upper-bound the Hessian Lipschitz constant of each block, and the final constant is at most multiplied by~2. Define the matrix ${\bm{A}} \in {\mathbb{R}}^{m \times (m+1)}$ as
\begin{align*}
{\bm{A}} = 
\begin{bmatrix}
        1 & -1 & 0 & \cdots & 0 \\
        0 & 1 & -1 & \cdots & 0 \\
        \vdots &  \ddots & \ddots & \vdots \\
        0 & \cdots & \cdots & 1 & -1 \\
\end{bmatrix}.
\end{align*}
Then each block has the form of 
\begin{align*}
    g_s({\bm{y}}) = \lambda \Vert {\bm{A}} {\bm{y}}^{(s)} \Vert^3, \quad {\bm{y}}^{(s)} = \left( \langle {\bm{x}}, {\bm{v}}_{sm} \rangle, \cdots, \langle {\bm{x}}, {\bm{v}}_{(s+1)m} \rangle \right)^\top.
\end{align*}
Since the function $\Vert \,\cdot\, \Vert^3$ has $6$-Lipschitz continuous Hessians \citep[Lemma 4.2.4]{nesterov2018lectures} and $\Vert {\bm{A}} \Vert \le 2$, it is easy to check that $g_s({\bm{y}})$ has $48 \lambda$-Lipschitz continuous Hessians. Hence, we can conclude that the final function $f_{2T}({\bm{x}})$ has $96\lambda$-Lipschitz continuous Hessians. In other words, we can let $\lambda = L / 96$ to let the Hessian Lipschitz constant be bounded by $L$. The first and final (possibly incomplete) blocks obey the same bound by restricting the difference matrix and allowing an affine shift. Finally, since $T\ge m$ gives $\lfloor T/m\rfloor+1\le2T/m$ and $f_{2T}({\bm{x}}^*) = 0$, substituting the parameters $a = D / \sqrt{2T}$ and $\lambda = L / 96$ into inequality (\ref{eq:lb-suboptimality-gap})  yields that
\begin{align*}
    f_{2T}({\bm{x}}_T) - f_{2T}({\bm{x}}^*) = \Omega \left( \frac{\sqrt{m} L D^3}{T^{7/2}} \right).
\end{align*}
 In other words, to find an $\epsilon$-solution the algorithm has to take at least $T = \Omega( m^{1/7} (L D^3 /\epsilon)^{2/7} )$ iterations.
