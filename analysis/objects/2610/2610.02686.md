<!-- remark-1: body_md -->
When the parameter $b$ is a vector of unknown quantities then the sensitivity will be  a matrix given by 
\begin{equation}\label{eq:sensitivities_vector}
    W(t;b) = D_bv(t;b),
\end{equation}
where $D_b$ is the Jacobian with respect to parameters.
Following this, we will refer to the $W$ as the sensitivity operator.
<!-- end remark-1 -->
<!-- assumption-1: body_md -->
\label{assumption:projection_obs}
    The observation operator $I_h$ is a linear projection operator (i.e. idempotent, hermitian) with orthogonal complement $J_h$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{assumption:observable_parameters}
The unidentified parameters $b$ are restricted to the observable portion of the dynamics, i.e.
\begin{align}
    J_h D_bF = 0
\end{align}
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{assumption:absorbing_ball}
    Solutions to the coupled system \eqref{eq:truth} and \eqref{eq:nudge} have a forward invariant global absorbing set $A_{b,\mu}$ in the cross product space for each value of $b\in B$ the admissible set of parameters,
    and there exists a ball $\mathcal{B}$ (centered at $0$) of radius $R$ such that
    \begin{equation}
        \bigcup_{b\in B, \mu \geq\mu_0} A_{b,\mu} \subset \mathcal{B},
    \end{equation}
    where $\mu_0$ is described in assumption \ref{assumption:jacobian_dissipation} below.
    We will refer to the projection onto the $v$-subspace of $\mathcal{B}$ as $\mathcal{B}_v$.
    We will assume initial conditions for $(u(0),v(0))$ are chosen from within $A_{b,\mu}$ and thus stay within $\mathcal{B}$.
<!-- end assumption-3 -->
<!-- assumption-4: body_md -->
\label{assumption:technical}
    Assume that $F(v;b)$ is continuously differentiable in $v$ for all $v$ inside the absorbing ball implying that $F$ is Lipschitz (uniformly over $b$) in $v$ with Lipschitz constant
\begin{align}
    L_F = \sup_{v \in \mathcal{B}_v,b \in B } \left\|
    D_vF(v;b)\right\|
    \label{eq:Lipschitz}
    .
\end{align}
We also require $F$ to be Lipschitz in the parameter $b$
\begin{align}
    \| F(v;b) - F(v;a) \|
    \leq C_F |b-a|.
\end{align}
Finally, we also require that the mixed second derivative of $F$ is also bounded, i.e.
\begin{equation}
    \left\|D_b D_vF(v;b)\right\| \leq L_{Fb}.
\end{equation}
$D_b D_v F(v;b)$ is a linear map from $\mathbb{R}^n\times \mathbb{R}^m \to \mathbb{R}^n $ and the norm used here is the operator norm.
All the constants above ($L_F,~C_F,~L_{Fb}$) are finite.
<!-- end assumption-4 -->
<!-- assumption-5: body_md -->
\label{assumption:jacobian_dissipation}
    We will assume that there exists a $\mu_0 > 0$, and  some symmetric positive definite matrix $P$.
    Such that for $\mu\geq\mu_0$ we have
    \begin{align}
        P (D_vF + \mu I_h)
        + (D_vF^T + \mu I_h) P
        \succeq \alpha I \label{eq:Dissipation},
    \end{align}
    for some $\alpha > 0$, where $I$ is the identity matrix, and for all $v \in \mathcal{B}_v$ and $b \in B$.
<!-- end assumption-5 -->
<!-- lemma-1: body_md -->
\label{lem:jacobian_dissipation}
    Given the fact that $F$ is Lipschitz, then if we can find a symmetric positive definite matrix $P_J$ defined with range equal to the range of $J_h$ so that
    \begin{align}
        P_J J_h \{D_vF\} J_h
        + J_h \{D_vF\}^T J_h P_J \succeq \gamma J_h,
    \end{align}
    for some $\gamma > 0$, then Assumption \ref{assumption:jacobian_dissipation} holds.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Without loss of generality, we assume that the space is partitioned so that we can define $P$ on all of $\mathbb{R}^n$ in block matrix form on the $I_h$ and $J_h$ subspaces as
    \begin{align}
        P = I_h + P_J.
    \end{align}
    Because the range of $P_J$ is the range of $J_h$, and $P$ is symmetric, we see that $P J_h = J_h P = P_J $.
    This shows us that $P I_h = I_h P = I_h$. 
    From this we see  
    \begin{align}
        \|I_h x\|_P &= \|I_h x\|,
        \notag\\
        \|J_h x\|_P &= \| J_h x\|_{P_J}.
    \end{align}
    Using this and the fact that $I= I_h + J_h$, letting $x\in\mathbb{R}^n$ be arbitrary, we can write
    \begin{align}
        \langle P \{D_vF\} x , x \rangle
        &= \langle P \{D_vF\} J_h x, J_hx \rangle
        + \langle P \{D_vF\} J_h x, I_h x \rangle
        + \langle P \{D_vF\} I_h x, J_hx \rangle\notag\\
        &+ \langle P\{D_vF\} I_h x, I_h x \rangle
        \notag\\
        &=
        \langle P_J J_h \{D_vF\} J_h x, J_hx \rangle
        + \langle \{D_vF\} J_h x, I_h x \rangle
        + \langle P \{D_vF\} I_h x, J_hx \rangle\notag\\
        &+ \langle \{D_vF\} I_h x, I_h x \rangle
    \end{align}
    By the assumptions of the lemma we have 
    \begin{align}
        \langle P_J J_h \{D_vF\} J_h x, J_hx \rangle
        \geq \frac{\gamma}{2} \| J_h x\|^2.
    \end{align}
    $F$ being Lipschitz gives us
    \begin{align}
        \langle \{D_vF\} I_h x , I_h x \rangle 
        \geq -L_F \|I_h x\|^2,
    \end{align}
    and using Cauchy Schwarz and Young's inequalities we can write
    \begin{align}
        \langle  P \{D_vF\} J_h x, I_h x \rangle
        &=
        \langle \{D_vF\} J_h x, I_h x \rangle
        \geq -L_F \|J_h x\| \| I_h x\|
        \notag\\
        &\geq - \frac{\gamma}{8} \|J_hx \|^2
        - \frac{4L_F^2}{ \gamma} \|I_h x\|^2 
        \\
        \langle P \{D_vF\} I_h x, J_h x \rangle
        &\geq -L_F \lambda_{\max}(P_J) \|J_h x\| \| I_h x\|
        \notag\\
        &\geq - \frac{\gamma}{8} \|J_hx \|^2
        - \frac{4L_F^2 \lambda_{\max}(P_J)^2}{ \gamma} \|I_h x \|^2.
    \end{align}
    Putting everything together gives us 
    \begin{align}
        \langle P \{D_vF\} x , x \rangle
        \geq \frac{\gamma}{4} \|J_h x\|^2
        - \left(
        L_F + \frac{4L_F^2 \lambda_{\max}(P_J)^2}{ \gamma}
        + \frac{4L_F^2}{ \gamma}
        \right) 
        \| I_h x\|^2.
    \end{align}
    Choosing
    \begin{align}
        \mu_0 =\left(
        L_F + \frac{4L_F^2 \lambda_{\max}(P_J)^2}{ \gamma}
        + \frac{4L_F^2}{ \gamma}
        \right) + \frac{\gamma}{4}, 
    \end{align}
    then for $\mu> \mu_0$ we have 
    \begin{align}
        \langle P  \{D_vF\} x, x \rangle
        + \mu \langle P I_h x,x \rangle
        &=
        \langle P \{D_vF\} x, x \rangle
        + \mu \| I_h x \|^2
        \notag\\
        &\geq
        \frac{\gamma}{4}\|J_h x\|^2
        + \frac{\gamma}{4}\|I_h x\|^2
        = \frac{\gamma}{4} \|x\|^2.
    \end{align}
    This concludes the proof with $\alpha = \gamma/2$.

<!-- lemma-2: body_md -->
\label{lem:bounded_error}
    For the coupled system
 \eqref{eq:truth}-\eqref{eq:nudge},    
with the nudging parameter chosen so that $\mu > \mu_0$ with $\mu_0$ chosen according to assumption \ref{assumption:jacobian_dissipation} then the state error $z=v-u$ is bounded for all time as
    \begin{equation}
        \|z(t)\|^2 \leq C_1 \|z(0)\|^2e^{-\gamma t} + \frac{C_2}{\gamma^2}|b-a|^2,
    \end{equation}
    for some positive constants $\gamma, C_1, C_2$.

<!-- proof-2: body_md -->
The evolution of $z = v-u$ is given by
    \begin{align}
        \frac{dz}{dt} &=    \left[F(u;a)-F(v;b)\right]  - \mu I_h z\\
        &= - \mu I_h z - \left[F(v;b)-F(u;b)\right] - \left[F(u;b)-F(u;a)\right]. \label{eq:error_evolve1}
    \end{align}
    By Assumption \ref{assumption:jacobian_dissipation} there exists some symmetric positive definite matrix $P$ such that
    \begin{align}
        \langle P \{D_vF(u;z)\} z ,z \rangle
        + \mu \langle PI_h z, z \rangle
        \geq 
        \frac{\alpha}{2}
        \|z\|^2.
    \end{align}

    The mean value theorem for vector valued functions states that
    \begin{equation}
        F(v;b) - F(u;b) = \int_0^1D_vF(u+\theta(v-u);b)(v-u)d\theta,
    \end{equation}
    so that
    \begin{equation}
        \langle F(v;b)-F(u;b),Pz\rangle = \left\langle\int_0^1D_vF(u+\theta z;b)z d\theta,Pz\right\rangle.
    \end{equation}
    Taking the inner product with $Pz$ of \eqref{eq:error_evolve1} we find
    \begin{align}
        \frac{1}{2}\frac{d}{dt}\|z\|^2_P &=  - \mu 
        \langle P I_h z,z \rangle - \langle F(v;b)-F(u;b), Pz\rangle
        \notag\\
        &- \langle F(u;b)-F(u;a), Pz\rangle 
        \notag\\ 
        &\leq 
        -\left\langle P \int_0^1D_vF(u+\theta z; b)\} zd\theta, z \right\rangle
        - \mu \langle PI_h z, z \rangle 
        \notag\\
        &+ \lambda_{\max}(P) C_F|b-a|\|z\|_P
        \notag\\
        &\leq
        - \frac{\alpha}{2} \| z\|^2
        + C_F \sqrt{\lambda_{\max}(P)} |b-a|\|z\|_P
        \\
        &\leq
        - \frac{\alpha}
        {2 \lambda_{\max}(P)}
        \|z\|^2_P
        + C_F
        \sqrt{\lambda_{\max}(P)}
        |b-a|\|z\|_P.
    \end{align}
    Applying Young's inequality we arrive at:
    \begin{align}
        \frac{1}{2}\frac{d}{dt}\|z\|^2_P 
        &\leq -\frac{\alpha}{4\lambda_{\max}(P)}
        \|z\|^2_P 
        +
        \frac{C_F^2 \lambda_{\max}^2(P)}
        {\alpha}
        |b-a|^2.
    \end{align}
    Letting $\gamma = \frac{\alpha}{2 \lambda_{\max}(P)}$ and $C_{b} = C_F^2\lambda_{\max}(P)$ we have 
    \begin{align}
        \frac{1}{2}\frac{d}{dt}\|z\|^2_P 
        &\leq -
        \frac{\gamma}{2}
        \|z\|^2_P 
        +
        \frac{C_b}{2\gamma}
        |b-a|^2.
    \end{align}
    Then Gronwall's inequality gives us 
    \begin{align}
        \|z(t)\|_P^2
        \leq \|z(0)\|_P^2 e^{-\gamma t}
        + \frac{C_b}{\gamma^2}
        (1 - e^{- \gamma t})|b-a|^2.
    \end{align}
    \enlargethispage{\baselineskip}                                        
    Equivalence of the Euclidean and $P$-norms gives us
    \begin{align}
        \|z(t)\|^2
        &\leq 
        \frac{1}{\lambda_{\min}(P)}
        \|z(t)\|^2_P
        \notag\\
        &\leq 
        \frac{1}{\lambda_{\min}(P)}
        \|z(0)\|_P^2 e^{-\gamma t}
        + 
        \frac{1}{\lambda_{\min}(P)}
        \frac{C_b}{\gamma^2}
        (1 - e^{- \gamma t})|b-a|^2
        \notag\\
        &\leq
        \frac{1}{\lambda_{\min}(P)}
        \|z(0)\|^2 e^{-\gamma t}
        +
        \frac{1}{\lambda_{\min}(P)}
        \frac{C_b}{\gamma^2}
        |b-a|^2
    \end{align}

<!-- theorem-1: body_md -->
For $\Phi(t,s)$ satisfying \eqref{eq:transitionMatrix}, with $\mu \geq \mu_0$ we have 
    \begin{align}
        \|\Phi(t,s)\| \leq C_\phi e^{-\gamma_\phi (t-s)},
    \end{align}
    for some positive, $\mu$-independent constants $C_\phi$ and $\gamma_\phi$  where the norm above is the operator norm on the matrix.
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
Because $\Phi(t,s)$ is the time evolution operator we have 
    \begin{align}
        x(t) = \Phi(t,s) x(s),
    \end{align}
    for $t > s$ and $x(t)$ as the solution of \eqref{eq:linearized}.
    We just need to show that 
    \begin{align}
        \|x(t)\| \leq C_\phi  e^{- \gamma_\phi(t-s)} \|x(s)\|. 
    \end{align}
    For solutions of \eqref{eq:linearized} with any possible initial condition $x(s)$ starting at $t=s$, and constants $C_\phi$ and $\gamma_\phi$ chosen so that the inequality holds globally for all $t$ and $s$.
    Using assumption \ref{assumption:jacobian_dissipation}, we have
    \begin{align}
        \frac{d}{dt} \| x\|_P^2
        &= - 2\langle D_vF(v;b) x, Px \rangle
        - 2 \mu \langle I_h P x,x \rangle
        \notag\\
        &\leq - \alpha\| x\|^2
        \leq - \frac{\alpha}{ \lambda_{\max}(P)} \|x\|_P^2.
    \end{align}
    Gronwall's inequality gives us 
    \begin{align}
        \| x(t) \|_P  \leq
        e^{-\frac{\alpha}{2\lambda_{\max}(P)}(t-s)}
        \|x(s)\|_P.
    \end{align}
    Equivalence of the Euclidean and $P$-norms then gives us 
    \begin{align}

\label{lem:dvdt_bounded}
Let $z=v-u$ and suppose that the assumptions of the previous section
hold. Define
\[
E_0:=\|I_hz(0)\|,
\qquad
C_G:=L_FC_z+C_F|b-a|,
\]
where $\|z(t)\|\leq C_z$ uniformly in time. Then, for every $t\geq0$,
\[
\|I_hz(t)\|
\leq
e^{-\mu t}E_0
+
\frac{C_G}{\mu}\left(1-e^{-\mu t}\right).
\]
Consequently,
\[
\left\|\frac{dv}{dt}(t)\right\|
\leq
C_v+\mu E_0e^{-\mu t},
\qquad
C_v:=M_F+C_G,
\]
where
\[
M_F:=\sup_{t\geq0}\|F(v(t);b)\|.
\]

\[
\int_0^t
e^{-\gamma(t-s)}
\left\|\frac{dv}{ds}(s)\right\|\,ds
\leq
\frac{C_v}{\gamma}+E_0.
\]
All constants on the right-hand side are independent of $\mu$.

<!-- proof-4: body_md -->
The observable error satisfies
    \begin{equation}
        \frac{d q}{d t}  + \mu q = -I_h(F(v;b)-F(u;a)).
    \end{equation}
    Using variation of constants this is rewritten as
    \begin{equation}
        q(t) = e^{-\mu t}q(0) - \int_0^te^{-\mu(t-s)}I_h(F(v(s);b)-F(u(s);a))ds.
    \end{equation}
    Since
    \begin{equation}
        \|F(v;b)-F(u;a)\| \leq L_F\|v-u\| + C_F|b-a| \leq C_G,
    \end{equation}
    then we find that
    \begin{align}
        \|q(t)\| &\leq e^{-\mu t}E_0 + C_G\int_0^te^{-\mu (t-s)}ds\\
        &= e^{-\mu t}E_0 + \frac{C_G}{\mu}(1-e^{-\mu t}).
    \end{align}

    Now, we rewrite the evolution of $v(t)$ as
    \begin{equation}
        \frac{dv}{d t} = -F(v;b) - \mu q.
    \end{equation}
    Looking at the $L^2$ norm of this
    \begin{align}
        \left\|\frac{dv}{dt}\right\| &\leq M_F + \mu \|q\|,\\
        &\leq M_F + \mu E_0e^{-\mu t} + C_G(1-e^{-\mu t}),\\
        &\leq C_v + \mu E_0e^{-\mu t}.
    \end{align}

    \begin{align}

        &\leq \frac{C_v}{\gamma} + E_0,
    \end{align}
    where we used the fact that $e^{-\gamma (t-s)} \leq 1$ for the second integral above.

    \begin{align}

        + D_vF(v;b) W + \mu I_h W

        \\

    \begin{align}
        W = \Phi(t,0) W_0
        - \int_0^t \Phi(t,s) D_bF(v(s);b)
        ds.
    \end{align}
   On the initial condition dependent term;
    we have
    \begin{align}
        \|\Phi(t,0) W_0\|
        \leq 
        \frac{1}{\mu}
        C_\phi e^{-\gamma_\phi t} C_0.

        \Phi(t,s)D_bF(v(s);b) ds
        =
        \int_0^t
        \Phi(t,s) 
        I_h
        D_bF(v(s);b) ds
        \notag\\
        &= 
        \frac{1}{\mu}
        \int_0^t
        \left(
        \frac{\partial}{\partial s}
        \Phi(t,s)
        - \Phi(t,s) D_vF(v(s);b)
        \right)
        D_bF(v(s);b)
        ds.

    \begin{align}
        &= \frac{1}{\mu}
        \left[D_bF(v(t);b) 
        - \Phi(t,0) D_bF(v(0);b)
        \right]     
        - \frac{1}{\mu}
        \int_0^t \Phi(t,s)
        \frac{\partial}{\partial s}
        D_bF(v(s);b)
        ds
        \notag\\
        &+ 
        \frac{1}{\mu}
        \int_0^t
        \Phi(t,s) D_vF(v(s);b)
        D_bF(v(s);b)
        ds.
    \end{align}

    \begin{align}
        \left\|\frac{1}{\mu}
        \left(
        D_bF(v(t);b)
        - \Phi(t,0) 
         D_bF(v(0);b)
        \right)
        \right\|
        \leq \frac{1}{\mu}

        1 + C_\phi e^{-\gamma_\phi t}
        \right) C_F.

        \int_0^t \Phi(t,s)
        \frac{\partial}{\partial s}
        D_bF(v(s);b)
        ds
        \right\|

        \frac{1}{\mu}

        \right\|ds.

    \end{align}

        \left\|
        D_b\{D_vF(v(s);b)\}

        L_{Fb}
        \left\|\frac{dv}{ds}\right\|.

        D_bF(v(s);b)
        ds

        D_bF(v(s);b)

        &\leq \frac{1}{\mu}
        \frac{1 - e^{-\gamma_\phi t}}{\gamma_\phi}
        C_\phi
        L_F C_F,
    \end{align}
    which completes the proof.

    \end{equation}
    The time derivative of $\zeta$ is then given by
    \begin{align}
        \frac{d\zeta}{dt} &= I_h\frac{dW}{dt} + \frac{1}{\mu}\frac{d}{dt}

    {dt}
    D_bF
    ,\zeta\right\rangle.
\end{equation}

\begin{equation}
    \frac{d}{dt}
    D_bF
    = D_b\{D_vF(v;b)\}\frac{dv}{dt}.
\end{equation}

    + L_FC_w + L_{Fb}\left( 
    C_v + E_0
    \right)
    \right).
\end{align}
