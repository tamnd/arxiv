\rm
\label{dfn-index}
Let $\alpha = (\alpha_1,\cdots, \alpha_n)$ be a set of nonnegative integers.
We call the index set $I_\alpha=\{i\in \{1,\cdots, n\}\mid \alpha_i > 0\}$ the set of {\em homogeneity indices associated with $\alpha = (\alpha_1,\cdots, \alpha_n)$}.
Let $U\subset \mathbb{R}^n$.
We say that the domain $U\subset \mathbb{R}^n$ is {\em admissible with respect to the sequence $\alpha$}
if
\begin{equation*}
U = \left\{{\bf u}=(u_1,\cdots, u_n)\in \mathbb{R}^n \mid u_i\in \mathbb{R}\text{ if }i\in I_\alpha,\ (u_{j_1},\cdots, u_{j_{n-l}}) \in \tilde U\right\},
\end{equation*} 
where $\{j_1, \cdots, j_{n-l}\} = \{1,\cdots, n\}\setminus I_\alpha$ and $\tilde U$ is an open set in $\mathbb{R}^{n-l}$ spanning $(u_{j_1},\cdots, u_{j_{n-l}})$ with $\{j_1, \cdots, j_{n-l}\} = \{1,\cdots, n\}\setminus I_\alpha$.

\rm
\label{dfn-AQH}
Let $U\subset \mathbb{R}^n$ be an admissible set with respect to $\alpha$. 
Also, let $f_0:U \to \mathbb{R}$ be a function.
Let $\alpha_1,\ldots, \alpha_n$ be nonnegative integers with $(\alpha_1,\ldots, \alpha_n) \not = (0,\ldots, 0)$ and $k > 0$.
We say that $f_0$ is a {\em quasi-homogeneous function\footnote{
          
In preceding studies, all $\alpha_i$'s and $k$ are typically assumed to be natural numbers.
In the present study, on the other hand, the above generalization is valid.
               
} of type $\alpha = (\alpha_1,\ldots, \alpha_n)$ and order $k$}, denoted by $f_0\in QH_{n,1}(\alpha, k)$, if
\begin{equation*}
f_0( s^{\Lambda_\alpha}{\bf x} ) = s^k f_0( {\bf x} )\quad \text{ for all } {\bf x} = (x_1,\ldots, x_n)^T \in U \text{ and } s>0,
\end{equation*}
where\footnote{
Throughout the rest of this paper, the power of real positive numbers or functions to matrices is described in a similar manner.
          
}
\begin{equation*}
\Lambda_\alpha =  {\rm diag}\left(\alpha_1,\ldots, \alpha_n\right),\quad s^{\Lambda_\alpha}{\bf x} = (s^{\alpha_1}x_1,\ldots, s^{\alpha_n}x_n)^T.
\end{equation*}
Next, let $X = \sum_{i=1}^n f_i({\bf x})\frac{\partial }{\partial x_i}$ be a continuous vector field defined on $U$.
We say that $X$, or simply $f = (f_1,\ldots, f_n)^T$ is a {\em quasi-homogeneous vector field of type $\alpha = (\alpha_1,\ldots, \alpha_n)$ and order $k+1$}, denoted by $f\in QH_n(\alpha, k)$, if $f_i\in QH_{n,1}(\alpha, k+\alpha_i)$ for all $i=1,\ldots, n$.
                                                                                                
\par
Finally, we say that $X = \sum_{i=1}^n f_i({\bf x})\frac{\partial }{\partial x_i}$, or simply $f: U\to \mathbb{R}^n$ is an {\em asymptotically quasi-homogeneous vector field of type $\alpha = (\alpha_1,\ldots, \alpha_n)$ and order $k+1$ (at infinity)}, denoted by $f\in AQH_n(\alpha, k)$, if there is
                                                                                                                
$f_{\alpha, k}\in QH_n(\alpha, k)$
such that
\begin{equation}
\label{residual}
f_i( s^{\Lambda_\alpha}{\bf x} ) - s^{k+\alpha_i} f_{i;\alpha,k}( {\bf x} ) = o(s^{k+\alpha_i}),\quad i\in \{1,\ldots, n\}
 \end{equation}
as $s\to +\infty$ uniformly on $\left\{{\bf x}\in U \mid \sum_{i\in I_\alpha} x_i^2 = 1, (x_{j_1},\cdots, x_{j_{n-l}}) \in \tilde K\right\}$ for any compact subset $\tilde K\subset \tilde U$.

\rm
\label{dfn-quasi-para}
Let $\{\beta_i\}_{i\in I_\alpha}$ be the collection of natural numbers so that 
\begin{equation}
\label{LCM}
\alpha_i \beta_i \equiv c \in \mathbb{N},\quad i\in I_\alpha
\end{equation}

In particular, $\{\beta_i\}_{i\in I_\alpha}$ is chosen to be the smallest among possible collections.
If $i\not \in I_\alpha$, we formally set $\beta_i = 0$.

\begin{equation}
\label{func-p}
p_\alpha({\bf y}) \equiv \left( \sum_{i \in I_\alpha} y_i^{2\beta_i} \right)^{1/2c}.
\end{equation}
Define the mapping $T_{{\rm para};\alpha}: \mathbb{R}^n \to \mathbb{R}^n$ as the inverse of
\begin{equation}
\label{parabolic-cpt}
S_{{\rm para};\alpha}({\bf x}) = {\bf y},\quad y_j = \kappa_\alpha^{\alpha_j} x_j,\quad j=1,\ldots, n,
\end{equation}
where 
\begin{equation*}
\kappa_\alpha = \kappa_\alpha({\bf x}) = (1- p_\alpha({\bf x})^{2c})^{-1} \equiv \left( 1 - \sum_{j\in I_\alpha} x_j^{2\beta_j}\right)^{-1}.

<!-- lem-3: body_md -->
\label{lem-r}
Assume that, for (\ref{X-linear}),
\begin{equation}
\label{integral-exp}
\int_0^{T_\tau} G({\bf x}_\ast (\tau)) d\tau \not = 0. 
\end{equation}
Then (\ref{X-linear}) admits a unique $T_\tau$-periodic solution $X(\tau)$, which is bounded away from $0$.
In particular, for any $\tau$, quantities $\int_0^{T_\tau} G({\bf x}_\ast (\tau)) d\tau$ and $X(\tau)$ have identical signs.
<!-- end lem-3 -->
<!-- proof-1: body_md -->
For simplicity, we shall write $T = T_\tau$.
Letting
\begin{equation*}
a_G(\tau) := k\int_0^{\tau} G({\bf x}_\ast(s))ds,\quad I(\tau) = \int_0^{\tau} e^{-a_G(s)}ds,
\end{equation*}
the general solution is written as
\begin{equation*}
X(\tau) = e^{a_G(\tau)} (C- I(\tau)).
\end{equation*}
Indeed, we have
\begin{align*}
\frac{dX}{d\tau}(\tau) &= \frac{da_G}{d\tau}(\tau) e^{a_G(\tau)} (C- I(\tau)) + e^{a_G(\tau)} \frac{dI}{d\tau}(\tau) 
   
	= kG({\bf x}_\ast(\tau)) X - 1.
\end{align*}
Using the $T_\tau$-periodicity of $X(\tau)$, we have
\begin{equation*}
e^{a_G(T)} (C- I(T)) = e^{a_G(0)} (C- I(0)) = C,
\end{equation*}
and hence
\begin{equation*}
C = e^{a_G(T)} \frac{I(T)}{e^{a_G(T)} - 1}.
\end{equation*}
By assumption, we have $a_G(T)\not = 0$, in particular $e^{a_G(T)} \not = 1$ holds, implying that the constant $C$ is well-defined.
Next assume that there is a time $\tau$ such that $X(\tau) = 0$, which means that
\begin{equation*}
C = e^{a_G(T)} \frac{I(T)}{e^{a_G(T)} - 1} = I(\tau).
\end{equation*}
Now $I(\tau)$ is strictly increasing because $ e^{-a_G(\tau)} > 0$, and we have
\begin{equation*}
0 < I(\tau) < I(T), \quad \text{ for all } \tau \in (0,T)
\end{equation*}
The sign of $X(\tau)$ is therefore determined as follows.
\begin{description}
\item[Case 1: $\int_0^T G({\bf x}_\ast (\tau)) d\tau > 0$.] 
\end{description}
Because $e^{a_G(T)} > 1$, we have
\begin{equation*}
C = e^{a_G(T)}\frac{I(T)}{e^{a_G(T)} - 1} > I(T) \geq I(\tau), \quad \text{ for all } \tau \in [0,T],
\end{equation*}
which indicates $X(\tau) > 0$.
\begin{description}
\item[Case 2: $\int_0^T G({\bf x}_\ast (\tau)) d\tau < 0$.] 
\end{description}
Because $e^{a_G(T)} < 1$, we have
\begin{equation*}
C = e^{a_G(T)}\frac{I(T)}{e^{a_G(T)} - 1} = -e^{a_G(T)}\frac{I(T)}{1 - e^{a_G(T)}} < 0 \leq I(\tau), \quad \text{ for all } \tau \in [0,T],
\end{equation*}
which indicates $X(\tau) < 0$ and the proof is completed.

Two quotient bundles

\mathcal{Q}_{\bf x} = T\mathcal{E}|_{\gamma_\ast} /T\gamma_\ast

are isomorphic via the following bundle isomorphism:
\begin{equation*}
\mathcal{S}_\eta: \mathcal{Q}_{\bf x}(\tau(\eta))\to \mathcal{Q}_{\bf Y}(\eta),
\quad
\mathcal{S}_\eta[\xi]:=[r(\tau(\eta))^{\Lambda_\alpha}\xi],
\end{equation*}
whose inverse is given by
\begin{equation*}
\mathcal{S}_\eta^{-1} = \mathcal{T}_\eta: \mathcal{Q}_{\bf Y}(\eta)\to \mathcal{Q}_{\bf x}(\tau(\eta)),
\quad
\mathcal{T}_\eta[\zeta] :=[ (I-P_\ast(\tau(\eta))) p_\alpha({\bf Y}_0(\eta))^{-\Lambda_\alpha}\zeta].
\end{equation*}
<!-- end thm-3 -->
<!-- proof-3: body_md -->
We have already seen that both $\mathcal{S}_\eta$ and $\mathcal{T}_\eta$ are well-defined.
Here we only verify that they are mutually inverted.
For $\xi \in T_{{\bf x}_\ast(\tau)}\mathcal{E}$, apply $\mathcal{T}_\eta$ to $\mathcal{S}_\eta [\xi] = [r^{\Lambda_\alpha}\xi]$. 
Then
\begin{equation*}
\mathcal{T}_\eta \mathcal{S}_\eta [\xi] = [ (I-P_\ast(\tau(\eta))) p_\alpha({\bf Y}_0(\eta))^{-\Lambda_\alpha} r(\tau(\eta))^{\Lambda_\alpha}\xi] =  [ (I-P_\ast(\tau(\eta))) \xi] = [\xi].
\end{equation*}
On the other hand, for $[\zeta]\in \mathcal{Q}_{\bf Y}$, we have
\begin{equation*}
\mathcal{S}_\eta \mathcal{T}_\eta [\zeta] = [r(\tau(\eta))^{\Lambda_\alpha} (I-P_\ast(\tau(\eta))) p_\alpha({\bf Y}_0(\eta))^{-\Lambda_\alpha} \zeta].
\end{equation*}
The difference of this element from $[\zeta]$ itself is
\begin{equation*}
\zeta - r(\tau(\eta))^{\Lambda_\alpha} (I-P_\ast(\tau(\eta))) p_\alpha({\bf Y}_0(\eta))^{-\Lambda_\alpha} \zeta = r(\tau(\eta))^{\Lambda_\alpha} P_\ast(\tau(\eta)) p_\alpha({\bf Y}_0(\eta))^{-\Lambda_\alpha} \zeta.
\end{equation*}
More precisely,
\begin{equation*}
r^{\Lambda_\alpha} P_\ast r^{-\Lambda_\alpha}\zeta = r^{\Lambda_\alpha} \Lambda_\alpha {\bf x}_\ast ({\bf n}^T r^{-\Lambda_\alpha} \zeta) = \Lambda_\alpha r^{\Lambda_\alpha} {\bf x}_\ast ({\bf n}^T r^{-\Lambda_\alpha} \zeta) = ({\bf n}^T r^{-\Lambda_\alpha} \zeta) \Lambda_\alpha {\bf Y}_0.
\end{equation*}
The difference belongs to the subbundle $R_{\bf Y}^{(2)}$ indicating that it vanishes on the quotient bundle $\mathcal{Q}_{\bf Y}$.
The proof of $\mathcal{T}_\eta = \mathcal{S}_\eta^{-1}$ is therefore completed.

\label{thm-stability}
Let $f \in AQH_{n}(\alpha, k)$ satisfying Assumption \ref{ass-diff-QH}.
Let $\gamma_\ast$ be a hyperbolic periodic orbit on the horizon for the desingularized vector field $g$ associated with $f$ such that $W^s_{\rm loc}(\gamma_\ast; g)\cap \mathcal{D}\not = \emptyset$, and ${\bf Y}_0(\eta)$ be the corresponding periodic solution of the balance law (\ref{dfn-balance}) with $p_\alpha({\bf Y}_0(\eta)) > 0$ for any $\eta$.
If
\begin{align*}
m &:= \dim W^s_{\rm loc}(\gamma_\ast; g),\quad m_A = \sharp \{ \mu\in {\rm Ch}({\bf Y}_0) \mid |\mu| \leq 1 \},
\end{align*}
then we have $m = m_A+1$, where characteristic multipliers are counted with algebraic multiplicity.
<!-- end thm-6 -->
<!-- rem-6: body_md -->
In the computations below, periodic solutions are numerically computed by a single-shooting Newton method supplemented with a phase condition. 
Characteristic multipliers are then computed in the standard manner by simultaneously integrating the variational equation along the periodic solution (see, e.g., \cite{K2004} for the standard Floquet-theoretic background).
More precisely, for a periodic solution ${\bf Y}_0(\eta)$ of ${\bf Y}'= F({\bf Y})$ with period $T_\eta$, we solve
\begin{equation*}
\frac{d\Phi}{d\eta} = D_{\bf Y}F({\bf Y}_0(\eta)) \Phi,\quad \Phi(0) = I.
\end{equation*}
The matrix $\Phi(T_\eta)$ is the monodromy matrix, and its eigenvalues are taken as the characteristic multipliers. 
Periodic orbits and associated monodromy matrices are numerically computed by the shooting procedure described above.
