<!-- definition-1: body_md -->
\cite{VakarelovSequentALg}\label{def: S-algebras} A sequent Boolean
algebra, or \textbf{$S$}-algebra, is a pair $\left<\mathbf{A},\vdash\right>$,
where $\mathbf{A}$ is a Boolean algebra and $\vdash$ is a subset of $\mathcal{P}_{f}(A)\times A$
satisfying the following conditions: 
\begin{enumerate}
\item $a\vdash a$, 
\item If $X\vdash a$, then $X,b\vdash a$, 
\item If $X\vdash a$ and $X,a\vdash b$, then $X\vdash b$, 
\item $a\vdash b$ if and only if $a\leq b$. 
\item $\emptyset\vdash a$ if and only if $a=1$, 
\item If $X,a\vdash c$ and $X,b\vdash c$, then $X,a\vee b\vdash c$. 
\end{enumerate}

<!-- lemma-1: body_md -->
\label{props SequentALg} \cite{VakarelovSequentALg}
Let $\left<\mathbf{A},\vdash\right>$ be an $S$-algebra. Then the following
conditions hold: 
\begin{enumerate}
\item If $X,a,b\vdash c$, then $X,a\wedge b\vdash c$.

<!-- proposition-1: body_md -->
\label{prop: charact closure}Let $\mathbf{A}$ be a Boolean
algebra. Let $\mathsf{C}:\mathcal{P}(A)\rightarrow\mathcal{P}(A)$
be a finitary closure operator on $A$ satisfying the following conditions
for each $X\cup\left\{ a,b\right\} \subseteq A$: 
\begin{enumerate}
\item[$\mathbf{Fi}$] $\mathrm{C}(X)\subseteq\mathrm{Fg}(X)$. 
\item[$\mathbf{One}$] $\mathrm{C}(\emptyset)=\{1\}$.  
\item[$\mathbf{Up}$] $\mathrm{C}(X)\in\mathrm{Up}(A)$. 
\item[$\mathbf{Jo}$] $\mathrm{C}(X,a)\cap\mathrm{C}(X,b)\subseteq\mathrm{C}(X,a\vee b)$. 
\end{enumerate}
Define the relation ${\vdash_{\mathrm{C}}}\subseteq\mathcal{P}_{f}(A)\times A$
given by 
\[
X\vdash_{\mathrm{C}}b\;\text{ iff }\;\exists Y\in\mathcal{P}_{f}(X) (b\in\mathrm{C}(Y)).
\]
Then $\left<A,\vdash_{\mathrm{C}}\right>$ is an $S$-algebra.

Conversely, if $\left<\mathbf{A},\vdash\right>$ is an $S$-algebra, then the
mapping $\mathrm{C}_{\vdash}:\mathcal{P}(A)\rightarrow\mathcal{P}(A)$
given by 
\[
\mathrm{C}_{\vdash}(X)\coloneqq\left\{ a\in A:\exists Y\in\mathcal{P}_{f}(X)(Y\vdash a)\right\} ,
\]
is a finitary closure operator satisfying the conditions $\mathbf{Fi}$,
$\mathbf{One}$, $\mathbf{Up}$, and $\mathbf{Jo}$. Moreover, $\mathrm{C}=\mathrm{C}_{\vdash_{\mathrm{C}}}$
and ${\vdash}={\vdash_{\mathrm{C}_{\vdash}}}$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
We prove that $\left<\mathbf{A},\vdash_{\mathrm{C}}\right>$ is an $S$-algebra. To do so, let $a,b\in A$ and let $X\in\mathcal{P}_{f}(A)$. 
\begin{itemize}
\item[(1)] Since $a\in \mathrm{C}(a)$, we have $a\vdash_{\mathrm{C}}a$.
\item[(2)] If $X\vdash_{\mathrm{C}}a$, then $a\in\mathrm{C}(X)\subseteq\mathrm{C}\left(X\cup\left\{ b\right\} \right)$,
i.e., $X,b\vdash_{\mathrm{C}}a$.
\item[(3)] If $X\vdash_{\mathrm{C}}a$ and $X,a\vdash_{\mathrm{C}}b$, then $a\in\mathrm{C}(X)$
and $b\in\mathrm{C}\left(X\cup\left\{ a\right\} \right)$. So \[b\in\mathrm{C}\left(X\cup\left\{ a\right\} \right)\subseteq\mathrm{C}\left(X\cup\mathrm{C}(X)\right)=\mathrm{C}\left(\mathrm{C}(X)\right)=\mathrm{C}(X).\] Therefore, $X\vdash_{\mathrm{C}}b$.
\item[(4)] Now we show 
\[
a\leq b\text{ iff }a\vdash_{C}b.
\]
Let $a\leq b$. As $a\in\mathrm{C(a)},$ by $\mathbf{\mathbf{Up}},$
$b\in\mathrm{C}(a)$, so $a\vdash_{C}b$. Conversely, if $a\vdash_{C}b$,
then $b\in\mathrm{C}(a)\subseteq\mathrm{Fg}(a)$ (by $\mathbf{Fi}$),
thus, $a\leq b$.
\item[(5)] Suppose that $\emptyset\vdash_{\mathrm{C}}a$. Then, $a\in\mathrm{C}(\emptyset)\subseteq\mathrm{Fg}(\emptyset)=\left\{ 1\right\} $,
hence $a=1$. If $a=1$, then $a\in\mathrm{C}(\emptyset)$ by $\mathbf{One}$,
therefore $\emptyset\vdash_{\mathrm{C}}a$. 
\item[(6)] By condition $\mathbf{Jo}$, we have that if $X,a\vdash_{\mathrm{C}}c$
and $X,b\vdash_{\mathrm{C}}c$, then $X,a\vee b\vdash_{\mathrm{C}}c$. 
\end{itemize}
Then $\left<\mathbf{A},\vdash_{\mathrm{C}}\right>$ is an $S$-algebra. Conversely, if $\langle\mathbf{A},\vdash\rangle$ is an $S$-algebra, then it is easy to see that $\mathrm{C}_{\vdash}$ is a finitary closure operator satisfying conditions $\mathbf{Fi}$, $\mathbf{One}$, $\mathbf{Up}$, and $\mathbf{Jo}$. Moreover, it readily follows that $\mathrm{C}=\mathrm{C}_{\vdash_{\mathrm{C}}}$ and ${\vdash} = {\vdash_{\mathrm{C}_{\vdash}}}$.

\label{additivity}
Let $\mathbf{A}$ be a Boolean algebra and let
$\mathrm{C}$ be a closure operator on $A$ satisfying the property
$\mathbf{Jo}$ of Proposition \ref{prop: charact closure}. Then 
\[
\mathrm{C}(X,a)\cap\mathrm{C}(X,b)=\mathrm{C}(X,a\vee b),
\]
for any $X\cup\left\{ a,b\right\} \subseteq A$. Indeed, since $a,b\leq a\vee b$,
we have 
\[
\mathrm{C}(a\vee b)\subseteq\mathrm{C}(a)\cap\mathrm{C}(b)\subseteq\mathrm{C}\left(X,a\right)\cap\mathrm{C}\left(X,b\right).
\]
Hence $a\vee b\in\mathrm{C}\left(X,a\right)\cap\mathrm{C}\left(X,b\right)$.
Since $X\subseteq\mathrm{C}(X)\subseteq\mathrm{C}\left(X,a\right)\cap\mathrm{C}\left(X,b\right)$,
it follows that $X\cup\left\{ a\vee b\right\} \subseteq\mathrm{C}\left(X,a\right)\cap\mathrm{C}\left(X,b\right)$. Therefore,  
\[\mathrm{C}(X\cup\left\{ a\vee b\right\})\subseteq \mathrm{C}(\mathrm{C}\left(X,a\right)\cap\mathrm{C}\left(X,b\right))\subseteq \mathrm{C}\left(X,a\right)\cap\mathrm{C}\left(X,b\right),\] 
which establishes the equality.

\label{def:hypercontact semilattice}Let $\mathbf{L}=\left<L,\vee,0\right>$
be a join-semilattice. A subfamily $\Delta$ of $\mathcal{P}_{f}(L)$
is a hypercontact family if it satisfies the following conditions: 
\begin{enumerate}
\item If $\left\{ a_{1},\ldots,a_{n}\right\} \in\Delta$, then $a_{i}\neq0$,
for all $i\in\left\{ 1,\ldots,n\right\} $. 
\item If $U\subseteq V$ and $V\in\Delta$, then $U\in\Delta$. 
\item If $\left\{ a_{1},\ldots,a_{i},\ldots,a_{n}\right\} \in\Delta$ and
$a_{i}\leq b$, then $\left\{ a_{1},\ldots,a_{i},\ldots,a_{n},b\right\} \in\Delta$. 
\item If $b\neq0$, then $\left\{ b\right\} \in\Delta$. 
\end{enumerate}
A hypercontact semilattice (lattice) is a pair $\left<\mathbf{L},\Delta\right>$
such that $\mathbf{L}$ is a join-semilattice (lattice) and $\Delta$ is a
hypercontact family on $L$.

A hypercontact semilattice $\left<\mathbf{L},\Delta\right>$ is \emph{additive} if
$\left\{ a\vee b,c_{1},\ldots,c_{n}\right\} \in\Delta$ implies that
$\left\{ a,c_{1},\ldots,c_{n}\right\} \in\Delta$ or $\left\{ b,c_{1},\ldots,c_{n}\right\} \in\Delta$.

<!-- lemma-2: body_md -->
Let $\left<\mathbf{A},\mathrm{C}\right>$ be an $S$-algebra.
Then the family 
\[
\Delta_{\mathrm{C}}=\left\{ \left\{ a_{1},\ldots,a_{n}\right\} : 0\notin\mathrm{C}(a_{1},\ldots,a_{n}) \right\} 
\]
is an additive hypercontact family.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
If $\{a_{1},\ldots,a_{n}\} \in \Delta_{\mathrm{C}}$, then $0 \notin \mathrm{C}(a_{1},\ldots,a_{n})$. Since $\{a_{1},\ldots,a_{n}\} \subseteq \mathrm{C}(a_{1},\ldots,a_{n})$, it follows that $a_i \neq 0$ for all $i \in \{1,\ldots,n\}$, so (1) holds. 

To prove (2), let $\{a_{1},\ldots,a_{n}\} \subseteq \{b_{1},\ldots,b_{m}\}$ and $\{b_{1},\ldots,b_{m}\} \in \Delta_{\mathrm{C}}$. Then $0 \notin \mathrm{C}(b_{1},\ldots,b_{m})$. Since $\mathrm{C}$ is a closure operator, monotonicity implies that $\mathrm{C}(a_{1},\ldots,a_{n}) \subseteq \mathrm{C}(b_{1},\ldots,b_{m})$, and thus $0 \notin \mathrm{C}(a_{1},\ldots,a_{n})$. Therefore, $\{a_{1},\ldots,a_{n}\} \in \Delta_{\mathrm{C}}$.

For (3), let $\{a_{1},\ldots,a_{i},\ldots,a_{n}\} \in \Delta_{\mathrm{C}}$ and $a_{i} \leq b$. Thus, $0 \notin \mathrm{C}(a_{1},\ldots,a_{i},\ldots,a_{n})$ and $a_{i} \vee b = b$. It follows that
\[
\mathrm{C}(a_{1},\ldots,a_{n},b) = \mathrm{C}(a_{1},\ldots,a_{n},a_{i} \vee b) = \mathrm{C}(a_{1},\ldots,a_{n}) \cap \mathrm{C}(a_{1},\ldots,a_{n},b).
\]
Consequently, $\mathrm{C}(a_{1},\ldots,a_{n},b) \subseteq \mathrm{C}(a_{1},\ldots,a_{n})$. The reverse inclusion follows immediately from the monotonicity of $\mathrm{C}$. Thus, $\mathrm{C}(a_{1},\ldots,a_{n},b) = \mathrm{C}(a_{1},\ldots,a_{n})$, which implies that $0 \notin \mathrm{C}(a_{1},\ldots,a_{n},b)$, i.e., $\{a_{1},\ldots,a_{n},b\} \in \Delta_{\mathrm{C}}$.

For (4), if $b \neq 0$, then $0 \notin \mathrm{Fg}(b) = \mathrm{C}(b)$. Hence, $\{b\} \in \Delta_{\mathrm{C}}$. This completes the proof that $\Delta_{\mathrm{C}}$ is a hypercontact family. Finally, the additivity of $\Delta_{\mathrm{C}}$ is a direct consequence of the definition of $\Delta_{\mathrm{C}}$ and Remark \ref{additivity}.

satisfies the property $\mathbf{Up}$ of Proposition \ref{prop: charact closure}.

<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We apply Zorn's lemma. Consider the family 
\[
\mathcal{F}=\left\{ G\in\mathcal{C}_{\mathrm{C}}:H\subseteq G\text{ and }G\cap I=\emptyset\right\} .
\]
Since $H\in\mathcal{F},$ we have that $\mathcal{F}\neq\emptyset$. Observe that if $\Gamma$ is a non-empty chain of elements
of $\mathcal{F}$, then it is a nonempty upward-directed subfamily of $\mathcal{C}_{\mathrm{C}}$. From this, it is readily seen that $\bigcup\Gamma\in\mathcal{F}$. Thus, by Zorn's

Then, $a\vee b\in F\cap I$, which is a contradiction. Therefore, $F$ must be irreducible.

\label{lem:charcat irreducible closed} Let $\left<\mathbf{A},\mathrm{C}\right>$
be a Boolean closure space satisfying the properties $\mathbf{Up}$
and $\mathbf{Jo}.$ Then 
\begin{enumerate}
\item A $\mathrm{C}$-closed subset $F$ is irreducible iff, for any $a,b\in A$, $a\vee b\in F$
implies $a\in F$ or $b\in F$. 
\item If $F$ is an irreducible $\mathrm{C}$-closed subset, then for each $a\in F$ there exists $U\in\operatorname{Ul}(\mathbf{A})$

\end{enumerate}

<!-- proof-4: body_md -->
(1) $\Leftarrow)$ Suppose first that $F$ satisfies the stated join-prime
condition. Suppose that $T_{1},T_{2}\in\mathcal{C}_{\mathrm{C}}$
and $T_{1}\cap T_{2}=F$ but $T_{1}\nsubseteq F$ and $T_{2}\nsubseteq F$.
Then there are $a,b\in A$ such that $a\in T_{1}-F$ and $b\in T_{2}-F$.
By property $\mathbf{Up}$, every closed set is an upset. So, $a\vee b\in T_{1}\cap T_{2}=F$.
But this implies that $a\in F$ or $b\in F$, which is a contradiction.
Thus, $T_{1}=F$ or $T_{2}=F$.

$\Rightarrow)$ Now suppose that $F$ is irreducible. Let $a,b\in A$
such that $a\vee b\in F$. Since $F$ is closed, by property $\mathbf{Jo}$
we have $F=\mathrm{C}\left(F,a\vee b\right)=\mathrm{C}(F,a)\cap\mathrm{C}(F,b)$.
So, $\mathrm{C}(F,a)=F$ or $\mathrm{C}(F,b)=F$. Then, $a\in F$
or $b\in F$. \\
 (2) Let $a\in F$. Then $a\notin F^{c}$. Notice that by (1) and
$\mathbf{Up}$, it easily follows that $F^{c}$ is an ideal of $\mathbf{A}$.
So $\mathrm{Fg}(a)\cap F^{c}=\emptyset$ and by the Ultrafilter Theorem,
there exists $U\in\mathrm{Ul}(\mathbf{A})$ such that $a\in U$ and $U\cap F^{c}=\emptyset$.
Thus, $a\in U\subseteq F$, as desired.

<!-- remark-3: body_md -->
\label{S-clans} 
Let $\langle\mathbf{A},\vdash\rangle$ be an $S$-algebra. Recall from \cite{VakarelovSequentALg} that a subset $F \subseteq A$ is called an \emph{$S$-filter} provided that if $X \vdash b$, then $b \in F$ for any finite subset $X \subseteq F$. Moreover, an $S$-filter $F$ is a \emph{clan} if $0 \notin F$ and satisfies that $a \vee b \in F$ implies $a \in F$ or $b \in F$ for all $a,b \in A$.

Let $\langle\mathbf{A},\mathrm{C}_{\vdash}\rangle$ be its associated algebra via Proposition~\ref{prop: charact closure}. Since $\mathrm{C}_{\vdash}$ is a finitary closure operator, a subset $F$ of $A$ is $\mathrm{C}_{\vdash}$-closed iff $\mathrm{C}_{\vdash}(X) \subseteq F$ for any finite subset $X$ of $F$. Note that the latter condition is equivalent to stating that $X \vdash a$ entails $a \in F$ for any finite subset $X$ of $F$. Consequently, the notion of a $\mathrm{C}_{\vdash}$-closed subset coincides with that of an $S$-filter. Additionally, since property $\mathbf{Up}$ ensures that every $\mathrm{C}_{\vdash}$-closed subset is an upset of $\mathbf{A}$, it follows that $F$ is proper iff $0 \notin F$. Thus, this fact and Lemma~\ref{lem:charcat irreducible closed} lead us to conclude that the notions of $S$-clan and irreducible $\mathrm{C}_{\vdash}$-closed subset coincide.

let $\Delta\subseteq\mathcal{B}_\mathbf{A}$ such that $\bigcap\Gamma\subseteq\bigcup\Delta$.

and $b\in A$ such that $a_{1},\ldots,a_{n}\vdash b$. Then, by  Lemma \ref{lem: maps alpa and beta} (6) we have
that $\beta_{\mathbf{A}}(a_{1})\cap\ldots\cap\beta_{\mathbf{A}}(a_{n})\subseteq\beta_{\mathbf{A}}(b)$. Since
$\beta_{\mathbf{A}}(a_{1}),\ldots,\beta_{\mathbf{A}}(a_{n})\in\Gamma$ and $\Gamma$ is a closed

$\beta_{\mathbf{A}}(b)\in\Gamma$, i.e., $b\in F_{A}.$ Thus, $F_{A}$ is closed.
It is easy to see that if $a\vee b\in F_{A}$, then $a\in F_{A}$
or $b\in F_{A}.$ So, $F_{A}\in X(\mathbf{A})$. As $F_{A}\in\bigcap\Gamma\subseteq\bigcup\Delta$,
then there exists $\beta_{\mathbf{A}}(c)\in\Delta$ such that $F_{A}\in\beta_{\mathbf{A}}(c).$
So, $c\in F_{A}$ and by the definition of $F_{A}$,
we have $\beta_{\mathbf{A}}(c)\in\Gamma$. Thus, $\beta_{\mathbf{A}}(c)\in\Gamma\cap\Delta$ so $\Gamma\cap\Delta\neq \emptyset$.

<!-- lemma-8: body_md -->
Let $\left<\mathbf{A},\mathrm{C}\right>$ be a sequent algebra.
Then the sequent space $\left<X(\mathbf{A}),\tau_{\beta_{\mathbf{A}}\left[A\right]}\right>$
is compact.

<!-- proof-8: body_md -->
Let $B\subseteq A$ be a non-empty subset such that $X(\mathbf{A})\subseteq\bigcup\left\{ \beta_\mathbf{A}(a)^{c}:a\in B\right\} $.
Suppose that for any finite subset $B_{0}$ of $B$ we have $X(\mathbf{A})\nsubseteq\bigcup\left\{ \beta_\mathbf{A}(a)^{c}:a\in B_{0}\right\} $.
Then the filter in the Boolean algebra $\mathcal{P}(X(\mathbf{A}))$ generated
by the family $\left\{ \beta_\mathbf{A}(a):a\in B\right\} $ is proper. Then
there exists an ultrafilter $P$ of $\mathcal{P}(X(\mathbf{A}))$ such that
$\left\{ \beta_\mathbf{A}(a):a\in B\right\} \subseteq P$. Consider the set $F_{P}=\left\{ a\in A:\beta_\mathbf{A}(a)\in P\right\} .$
We prove that $F_{P}\in X(A)$. Let $X=\left\{ a_{1},\ldots,a_{n}\right\} $
be a finite subset of $F_{P}$. Let $b\in\mathrm{C}(X)$. By Lemma
\ref{lem: maps alpa and beta}, $\beta_\mathbf{A}(a_{1})\cap\ldots\cap\beta_\mathbf{A}(a_{n})\subseteq\beta_\mathbf{A}(b)$.

i.e., $F_{p}\notin\bigcup\left\{ \beta_\mathbf{A}(a)^{c}:a\in B\right\} $, which
is a contradiction. Thus, $\left<X(A),\tau_{\beta_{\mathbf{A}}[A]}\right>$ is

<!-- lemma-9: body_md -->
\label{lem:chacact homo contact}Let $\left<\mathbf{A},\mathrm{C}\right>$
and $\left<\mathbf{B},\mathrm{D}\right>$ be sequent algebras. Let $g:A\rightarrow B$
be a monotonic map. Then the following conditions are equivalent: 
\begin{enumerate}
\item $g$ is a join-homomorphism such that $g(1)=1$, and $g[\mathrm{C}(X)]\subseteq\mathrm{D}(g[X]),$
for each $X\subseteq A$. 
\item $g^{-1}[F]\in X(\mathbf{A})$, for each $F\in X(\mathbf{B})$. 
\end{enumerate}
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
$(1)\Rightarrow(2):$ Let $F\in X(\mathbf{B})$. We start by showing that $g^{-1}[F]$ is a $\mathrm{C}$-closed subset of $A$. To this end, we will prove that $\mathrm{C}(X)\subseteq g^{-1}[F]$, for every $X\subseteq g^{-1}[F]$. Let $a\in\mathrm{C}(X)$ and $X\subseteq g^{-1}[F]$. Then, $g(a)\in g[\mathrm{C}(X)]$.
Since $g[X]\subseteq F$ and $F$ is $\mathrm{D}$-closed, by (1) we have $g[C(X)]\subseteq \mathrm{D}(g[X])\subseteq F$.
This yields $a\in g^{-1}[F]$, as desired. 

Since $0=g(0)\notin F$, then it follows $0\notin g^{-1}[F]$, so $g^{-1}[F]$ is proper. Finally, to prove that $g^{-1}[F]$ is irreducible, let $a\vee b\in g^{-1}[F]$. Then $g(a\vee b)=g(a)\vee g(b)\in F$. Since $F$ is irreducible, Lemma \ref{lem:charcat irreducible closed} (1) yields $g(a)\in F$ or $g(b)\in F$. Hence, $a\in g^{-1}[F]$ or $b\in g^{-1}[F]$. Therefore, by the same result as above, $g^{-1}[F]\in X(\mathbf{A})$.

$(2)\Rightarrow(1):$ As $g$ is monotone, we have $g(a),g(b)\leq g(a\vee b)$ for all $a,b\in A$. Suppose that there exist $a,b\in A$ such that $g(a\vee b)\nleq g(a)\vee g(b)$. Then, by the Ultrafilter Theorem, there exists an ultrafilter $P$ of $\mathbf{B}$ such that $g(a\vee b)\in P$ and $g(a)\vee g(b)\notin P$. Consequently, $a\vee b\in g^{-1}[P]$, whereas $a,b\notin g^{-1}[P]$. This contradicts the irreducibility of $g^{-1}[P]$, which follows from Lemma \ref{lem: X(A) and ultrafilters} (1) and (2). 

To show that \(g(0)=0\), suppose that \(g(0)\nleq 0\). By the Ultrafilter Theorem, there exists an ultrafilter \(Q\) of \(\mathbf{B}\) such that \(g(0)\in Q\) and \(0\notin Q\). Thus, \(0\in g^{-1}[Q]\). By Lemma \ref{lem: X(A) and ultrafilters} (1), every ultrafilter of \(\mathbf{B}\) is an irreducible \(\mathrm{C}\)-closed subset of \(B\). Hence, by (2), \(g^{-1}[Q]\) is irreducible and, in particular, proper. This yields a contradiction. Therefore, \(g(0)=0\), and consequently, \(g\) is a join-homomorphism.

Now we show that \(g(1)=1\). Suppose, towards a contradiction, that \(g(1)<1\). Then \(0<\neg g(1)\), so \(\neg g(1)\nleq 0\). By the Ultrafilter Theorem, there exists an ultrafilter \(P\) of \(\mathbf{B}\) containing \(\neg g(1)\). Since \(g\) is monotone, \(g(a)\leq g(1)\) for every \(a\in A\). Hence, \(g(a)\notin P\) for every \(a\in A\), and thus \(g^{-1}[P]=\emptyset\). This contradicts (2), since \(\emptyset\) is not \(\mathrm{C}\)-closed: indeed, \(\mathbf{One}\) implies that \(1\in\mathrm{C}_{A}(\emptyset)\). Therefore, \(g(1)=1\).

Finally, let $X\subseteq A$. We show that
$g[\mathrm{C}(X)]\subseteq\mathrm{D}(g[X])$ by contradiction.
Suppose that there exists $a\in\mathrm{C}(X)$ such that
$g(a)\notin\mathrm{D}(g[X])$. Since $\mathrm{D}(g[X])$ is
$\mathrm{D}$-closed, Corollary \ref{cor: closed is inter of irreducible}
yields an $F\in X(\mathbf{B})$ such that
$\mathrm{D}(g[X])\subseteq F$ and $g(a)\notin F$.
Since $\mathrm{D}$ is a closure operator, we have $g[X]\subseteq F$,
and hence $X\subseteq g^{-1}[F]$. Since $\mathrm{C}$ is a closure
operator, it follows that
$\mathrm{C}(X)\subseteq\mathrm{C}(g^{-1}[F])$.
By (2), $g^{-1}[F]$ is $\mathrm{C}$-closed, so
$\mathrm{C}(X)\subseteq g^{-1}[F]$. In particular,
$a\in g^{-1}[F]$, contradicting $g(a)\notin F$.

<!-- remark-5: body_md -->
\label{rem: SeqSpm is not full}
Altough every sequent morphism is a sequent semi-morphism, the converse does not hold in general. To see this, consider $\mathbb{R}$ with its usual topology and the
sequent space $\langle\mathbb{R},\mathbf{RC}(\mathbb{R})\rangle$.
Let $\theta:\mathbb{R}\to\mathbb{R}$ be the constant map defined by
$\theta(x)=0$ for every $x\in\mathbb{R}$. For every
$U\in\mathrm{RC}(\mathbb{R})$, we have
\[
\theta^{\ast}(U)=
\begin{cases}
\mathbb{R} & \text{if }0\in U,\\
\emptyset & \text{if }0\notin U.
\end{cases}
\]
Hence, $\theta$ is a sequent map. Moreover,
$\theta^{\ast}(\emptyset)=\emptyset$ and
$\theta^{\ast}(\mathbb{R})=\mathbb{R}$. Furthermore, for every
$U,V\in\mathrm{RC}(\mathbb{R})$,
\[
\theta^{\ast}(U)\cup\theta^{\ast}(V)=
\begin{cases}
\mathbb{R} & \text{if }0\in U\cup V,\\
\emptyset & \text{if }0\notin U\cup V.
\end{cases}
\]
Thus,
\[
\theta^{\ast}(U)\cup\theta^{\ast}(V)
=\theta^{\ast}(U\cup V),
\]
so $\theta^{\ast}$ is a join-homomorphism. Nevertheless,
as shown in \cite{Goldblatt Mereocompactness},
$\theta^{\ast}$ preserves neither complementation nor binary meets
in $\mathrm{RC}(\mathbb{R})$. Therefore, $\theta$ is a sequent
semi-morphism that is not a sequent morphism.

Then, it is clear that the existence of the inverse of $h$ in $\mathsf{SA}$,
say $g$, makes $h$ bijective. Now take $Z\in\mathcal{P}(A)$. Since
$h$ is a sequent semi-homomorphism by assumption, in order to prove our
claim, it only remains to prove that $\mathrm{D}(h[Z])\subseteq h[\mathrm{C}(Z)]$.
Indeed, since $h[Z]\in\mathcal{P}(B)$ and $g$ is a sequent semi-homomorphism,
we have 
\[
g[\mathrm{D}(h[Z])]\subseteq\mathrm{C}(g[h[Z]])=\mathrm{C}(Z).
\]
From this fact, it is readily seen that the claim holds.
On the other hand, let us assume that $h$ is bijective and $h[\mathrm{C}(Z)]=\mathrm{D}(h[Z])$.
We only need to check that the inverse map of $h$

\mathrm{D}(W)=\mathrm{D}(h[(h^{-1})[W]])\subseteq h[\mathrm{C}((h^{-1})[W])].

<!-- theorem-1: body_md -->
\label{dually category} The categories $\mathsf{SA}$
and $\mathsf{SeqSp}$ are dually equivalent. Moreover, this duality restricts to a dual equivalence between the categories $\mathsf{SAB}$ and $\mathsf{SeqSp}_{\mathrm{m}}$.
<!-- end theorem-1 -->
<!-- proof-16: body_md -->
For the moreover part, let $h:\langle\mathbf{A},\mathrm{C}\rangle\to\langle\mathbf{B},\mathrm{D}\rangle$
be a sequent semi-homomorphism and let $\widehat{h}^{\,*}:
\mathcal{B}_{\mathbf{A}}\longrightarrow
\mathcal{B}_{\mathbf{B}}$ be the induced map of the semi-morphism $\widehat{h}:X(\mathbf{B})\longrightarrow X(\mathbf{A})$ defined as $\widehat{h}(F)=h^{-1}[F]$. We show that, $h$ preserves negation if and
only if $\widehat{h}^{\,*}$ does. Consequently, $h$ is a Boolean homomorphism if and only if, $\widehat{h}^{\,*}$ does.

Indeed, suppose that for every $a\in A$, $h(\neg a)=\neg h(a)$. From the
commutativity of diagram (\ref{commutative diagram}), the hypothesis
and Lemma \ref{lem: maps alpa and beta} (3) it follows: 
\[
\begin{array}{rcl}
-\widehat{h}^{\,*}(\beta_{\mathbf{A}}(a)) & = & -\beta_{\mathbf{B}}(h(a))\\
 & = & \beta_{\mathbf{B}}(\neg h(a))\\
 & = & \beta_{\mathbf{B}}(h(\neg a))\\
 & = & \widehat{h}^{\,*}(\beta_{\mathbf{A}}(\neg a))\\
 & = & \widehat{h}^{\,*}(-\beta_{\mathbf{A}}(a)).
\end{array}
\]
Conversely, suppose that $-\widehat{h}^{\,*}(\beta_{\mathbf{A}}(a))=\widehat{h}^{\,*}(-\beta_{\mathbf{A}}(a))$,
for every $a\in A$. Then, again by the commutativity of diagram (\ref{commutative diagram})
and Lemma \ref{lem: maps alpa and beta} (1) it is immediate that $h(\neg a)=\neg h(a)$. 

The preceding argument shows that $F$ and $G$ restrict to functors between $\mathsf{SAB}^{\mathsf{op}}$ and $\mathsf{SeqSp}_{\mathrm{m}}$. These restricted functors are faithful, because $F$ and $G$ are, and they are full, since by the equivalence $h$ is a Boolean homomorphism if and only if $\widehat{h}^{\,*}$ is, so that a morphism of $\mathsf{SA}$ (resp.\ $\mathsf{SeqSp}$) lies in $\mathsf{SAB}$ (resp.\ $\mathsf{SeqSp}_{\mathrm{m}}$) exactly when its image does. Hence the restricted functors are essentially surjective, and so they establish a dual equivalence between $\mathsf{SAB}$ and $\mathsf{SeqSp}_{\mathrm{m}}$.

<!-- proposition-8: body_md -->
\label{closed-closed} Let $\langle\mathbf{A},\mathrm{C}\rangle$
be a sequent algebra. Then the following hold: 
\begin{enumerate}
\item If $Y$ is a closed set of $\langle X(\mathbf{A}),\tau_{\beta_{\mathbf{A}}[A]}\rangle$,
then \[F_{Y}=\{a\in A\colon Y\subseteq\beta_\mathbf{A}(a)\}\]
is a $\mathrm{C}$-closed set of $A$.
\item If $F$ is a $\mathrm{C}$-closed set of $A$, then 
\[
Y_{F}=\bigcap\{\beta_{\mathbf{A}}(a)\colon a\in F\}=\{G\in X(\mathbf{A})\colon F\subseteq G\}
\]
is a closed set of $\langle X(\mathbf{A}),\tau_{\beta_{\mathbf{A}}[A]}\rangle$. 
\end{enumerate}
Furthermore, the poset of closed sets of $\langle X(\mathbf{A}),\tau_{\beta_{\mathbf{A}}[A]}\rangle$
is dually isomorphic to the poset $\langle\mathcal{C}_{\mathrm{C}},\subseteq\rangle$.
<!-- end proposition-8 -->
<!-- proof-17: body_md -->
(1) We only prove that $\mathrm{C}(F_{Y})\subseteq F_{Y}$,
since the converse inclusion follows from the fact that $\mathrm{C}$
is a closure operator on $A$. To this end, let $x\in\mathrm{C}(F_{Y})$.
As $\mathrm{C}$ is finitary, there exist $a_{1},\ldots,a_{n}\in F_{Y}$
such that $x\in\mathrm{C}(a_{1},\ldots,a_{n})$. By Lemma
\ref{lem: maps alpa and beta} (6),
\[
\beta_{\mathbf{A}}(a_{1})\cap\ldots\cap\beta_{\mathbf{A}}(a_{n})\subseteq\beta_{\mathbf{A}}(x).
\]
By assumption, $Y\subseteq\beta_{\mathbf{A}}(a_{i})$ for every $1\leq i\leq n$,
and thus from the above we get $Y\subseteq\beta_{\mathbf{A}}(x)$. Hence, $x\in F_{Y}$,
as required.

(2) This follows immediately since, for every $a\in A$, $\beta_{\mathbf{A}}(a)$
is a closed set by Lemma \ref{lem: maps alpa and beta}, and arbitrary
intersections of closed sets are closed.

For the ``furthermore'' part, we first note that, since $\beta_{\mathbf{A}}$
preserves finite joins, $\beta_{\mathbf{A}}(0)=\emptyset$ and $\beta_{\mathbf{A}}(1)=X(\mathbf{A})$
(because $1\in\mathrm{C}(\emptyset)$ belongs to every
closed set), the family $\beta_{\mathbf{A}}[A]$ is closed under finite unions.
Hence $\beta_{\mathbf{A}}[A]$ is a base (not only a subbase) of closed sets, and
every closed set of $X(\mathbf{A})$ is an intersection of sets of
the form $\beta_{\mathbf{A}}(a)$. Let $Y$ be a closed set of the dual sequent
space of $\langle\mathbf{A},\mathrm{C}\rangle$. It is
clear that $Y\subseteq Y_{F_{Y}}$. For the converse inclusion, suppose
that there is $P\in Y_{F_{Y}}$ such that $P\notin Y$. By the preceding
observation, there exists $a\in A$ with $Y\subseteq\beta_{\mathbf{A}}(a)$ and
$P\notin\beta_{\mathbf{A}}(a)$. Thus $a\in F_{Y}$. Since, by assumption, $F_{Y}\subseteq P$,
we get $a\in P$, which is a contradiction. Therefore $Y_{F_{Y}}=Y$.

On the other hand, let $F$ be a closed set of $\mathrm{C}$.
Observe that $a\in F_{Y_{F}}$ if and only if $\bigcap\{\beta_{\mathbf{A}}(b)\colon b\in F\}\subseteq\beta_{\mathbf{A}}(a)$.
By Lemma \ref{lem: maps alpa and beta} (6), the latter is equivalent
to $a\in\mathrm{C}(F)=F$. Hence $F_{Y_{F}}=F$. Therefore
the assignments $Y\mapsto F_{Y}$ and $F\mapsto Y_{F}$ are inverses
of each other, and both are clearly antitone.

<!-- theorem-2: body_md -->
\label{postive to mdecreceasing} Let $\langle\mathbf{A},\mathrm{C}\rangle$
be a sequent algebra. Then the following hold: 
\begin{enumerate}
\item If $F$ is a positive filter of $\mathbf{A}$, then $Y_{F}$ is an
m-decreasing closed set of $\langle X(\mathbf{A}),\tau_{\mathcal{B}_{\mathbf{A}}}\rangle$. 
\item If $Y$ is an m-decreasing closed set of $\langle X(\mathbf{A}),\tau_{\mathcal{B}_{\mathbf{A}}}\rangle$,
then $F_{Y}$ is a positive filter of $\mathbf{A}$. 
\end{enumerate}
Furthermore, the posets $\langle\mathrm{MD}(X(\mathbf{A})),\subseteq\rangle$
and $\langle\mathrm{PF}(\mathbf{A}),\subseteq\rangle$ are dually
isomorphic.
<!-- end theorem-2 -->
<!-- proof-19: body_md -->
(1) Let $F$ be a positive filter. Since $F$ is closed, $Y_{F}$
is closed by Proposition \ref{closed-closed}. Let $G\in Y_{F}$ and
let $U\in\mathrm{min}X(\mathbf{A})\cap(G]$. By Proposition \ref{maximals are ultrafilters},
$U$ is an ultrafilter and $U\subseteq G$. We show that $F\subseteq U$.
Suppose, to the contrary, that $a\in F\setminus U$. Since $U$ is
an ultrafilter, $\neg a\in U\subseteq G$. But $F\subseteq G$, contradicting
the positivity of $F$. Hence $U\in Y_{F}$, and therefore $Y_{F}$
is m-decreasing.

(2) Let $Y$ be an m-decreasing closed subset of $\langle X(\mathbf{A}),\tau_{\mathcal{B}_{\mathbf{A}}}\rangle$.
By Proposition \ref{closed-closed}, $F_{Y}$ is closed with respect
to $\mathrm{C}$. Since a closed set need not be a filter,
we verify directly that $F_{Y}$ is a filter. Clearly $1\in F_{Y}$,
and $F_{Y}$ is an upset. Let $a,b\in F_{Y}$ and $P\in Y$. By Lemma
\ref{lem:charcat irreducible closed} (2), there exists an ultrafilter
$U$ of $\mathbf{A}$ such that $U\subseteq P$. By Proposition \ref{maximals are ultrafilters},
$U\in\mathrm{min}X(\mathbf{A})\cap(P]$, and since $Y$ is m-decreasing,
$U\in Y$. Thus $a,b\in U$, so $a\wedge b\in U\subseteq P$. As $P\in Y$
was arbitrary, $Y\subseteq\beta(a\wedge b)$, and hence $a\wedge b\in F_{Y}$.
Therefore $F_{Y}$ is a filter.

It remains to prove positivity. Suppose that $a\in F_{Y}$, $G\in X(\mathbf{A})$,
$F_{Y}\subseteq G$, and $\neg a\in G$. By Lemma~\ref{lem:charcat irreducible closed} (2),
there is an ultrafilter $U$ of $\mathbf{A}$ with $\neg a\in U\subseteq G$. Since
$G\in Y_{F_{Y}}=Y$ by Proposition~\ref{closed-closed}, and $Y$
is m-decreasing, we have $U\in Y$. But $a\in F_{Y}$ means $Y\subseteq\beta(a)$,
so $a\in U$, contradicting $\neg a\in U$. Hence $F_{Y}$ is positive.

The final statement follows from (1), (2), and Proposition~\ref{closed-closed},
since every positive filter is closed, so that $F_{Y_{F}}=F$ for
each $F\in\mathrm{PF}(\mathbf{A})$.

\label{thm: congruences regular}Let $\left<A,\vdash\right>$
be a sequent algebra. Then the following hold: 
\begin{enumerate}
\item If $F$ is a regular filter of $\mathbf{A}$, then 
\[\theta(F)=\left\{ \left(a,b\right)\in A^{2}:\exists f\in F\,\left(a\wedge f=b\wedge f\right)\right\} \]
is a sequent congruence. 
\item If $\theta$ is a sequent congruence, then $1_{\theta}=\left\{ a\in A:\left(a,1\right)\in\theta\right\} $
is a regular filter. 
\end{enumerate}
Furthermore, the poset of sequent congruences of $\mathbf{A}$ is
isomorphic to the poset of regular filters of $\mathbf{A}$.
<!-- end theorem-3 -->
<!-- proof-25: body_md -->
(1) Let $F$ be a regular filter. It is well known that $\theta(F)$
is a Boolean congruence. Let $(a,b)\in\theta(F)$, let $\left\{ c\right\} \cup B\subseteq_{f}A$
be fixed, and let $e\in I(\neg a,B,c)$, i.e., 
\[
e,\neg a,B\vdash c.
\]
We need to find $d\in A$ such that $d,\neg b,B\vdash c$ and $(e,d)\in\theta(F)$.
Since $(a,b)\in\theta(F)$, there exists $f\in F$ such that $a\wedge f=b\wedge f$.
As $F$ is regular, there exists $z\in F$ such that $z,\neg f\vdash0$;
by Lemma \ref{lem: prec simple}, $z,\neg f,B\vdash c$. By Lemma
\ref{props SequentALg} (3) we have $e\wedge z,\neg a,B\vdash c$ and
$e\wedge z,\neg f,B\vdash c$, so condition (6) gives 
\[
e\wedge z,\neg a\vee\neg f,B\vdash c.
\]
Since 
\[
\neg a\vee\neg f=\neg(a\wedge f)=\neg(b\wedge f)=\neg b\vee\neg f
\]
and $\neg b\leq\neg b\vee\neg f$, Lemma \ref{props SequentALg} (3)
yields 
\[
e\wedge z,\neg b,B\vdash c.
\]
Since $z\in F$ and $(e\wedge z)\wedge z=e\wedge z$, we get $\left(e,e\wedge z\right)\in\theta(F)$.
Take $d=e\wedge z$. Then $d,\neg b,B\vdash c$ and $(e,d)\in\theta(F)$.

(2) Let $\theta$ be a sequent congruence. It is well known that $1_{\theta}$
is a filter. Let $f\in1_{\theta}$, i.e., $(1,f)\in\theta$. We apply
$\mathrm{(SCon)}$ with $c=0$ and $B=\emptyset$. Since $\neg1=0$
and $0,1\vdash0$, we have $1\in I(\neg1,\emptyset,0)$. Hence there
exists $z\in I(\neg f,\emptyset,0)$ such that $(1,z)\in\theta$,
that is, $z,\neg f\vdash0$ and $z\in1_{\theta}$. By Lemma \ref{lem: prec simple},
$z\prec f$. Thus $1_{\theta}$ is a regular filter.

For the last part, recall that the assignments $\theta\mapsto1_{\theta}$
and $F\mapsto\theta(F)$ are mutually inverse order-preserving bijections
between Boolean congruences and filters of $A$, i.e., $\theta(1_{\theta})=\theta$
and $1_{\theta(F)}=F$. By (1) and (2), they restrict to mutually
inverse order-isomorphisms between sequent congruences and regular
filters.

<!-- lemma-14: body_md -->
\label{existence of liftings} Let $\mathbf{A}$ be
a Boolean algebra and let $S=\{f_{i}:\mathbf{A}\to U(\langle\mathbf{B}_{i},\mathrm{C}_i\rangle)\}_{i\in I}$
be a $U$-structured source. Then $S$ has a lifting.
<!-- end lemma-14 -->
<!-- proof-27: body_md -->
Set 
\[
\mathcal{F}_{S}=\{f^{-1}_{i}[F_{i}]\colon F_{i}\in X(\mathbf{B}_{i}),\;i\in I\}\cup\mathrm{Fi}(\mathbf{A}).
\]
For $Z\subseteq A$, define 
\[
\mathrm{D}_{S}(Z)=\bigcap\{H\in\mathcal{F}_{S}\colon Z\subseteq H\}
\]
and let 
\[
\mathrm{C}_{S}(Z)=\bigcup\{\mathrm{D}_{S}(W)\colon W\subseteq_{f}Z\}.
\]
The operator $\mathrm{D}_{S}$ is the closure operator generated by
the family $\mathcal{F}_{S}$, and $\mathrm{C}_{S}$ is its finitarization.
A standard argument shows that $\mathrm{C}_{S}$ is again a closure
operator: extensivity and monotonicity are immediate, and if $x\in\mathrm{C}_{S}(\mathrm{C}_{S}(Z))$,
only finitely many members of $\mathrm{C}_{S}(Z)$ are involved; collecting
finite witnesses for these members and using the idempotence of $\mathrm{D}_{S}$
gives $x\in\mathrm{C}_{S}(Z)$. Thus $\mathrm{C}_{S}$ is finitary
by construction.

We first show that, for every $i\in I$ and $Z\subseteq A$, 
\[
f_{i}[\mathrm{D}_{S}(Z)]\subseteq\mathrm{C}_i(f_{i}[Z]).
\]
Suppose that $x\in\mathrm{D}_{S}(Z)$ but $f_{i}(x)\notin\mathrm{C}_i(f_{i}[Z])$.
By Proposition \ref{clans for a basis}, there is $P\in X(\mathbf{B}_{i})$
such that $f_{i}[Z]\subseteq P$ and $f_{i}(x)\notin P$. Hence $Z\subseteq f^{-1}_{i}[P]\in\mathcal{F}_{S}$
while $x\notin f^{-1}_{i}[P]$, contradicting $x\in\mathrm{D}_{S}(Z)$.
Consequently, 
\[
f_{i}[\mathrm{C}_{S}(Z)]\subseteq\mathrm{C}_i(f_{i}[Z]),
\]
because every element of $\mathrm{C}_{S}(Z)$ already belongs to $\mathrm{D}_{S}(W)$
for some finite $W\subseteq Z$.

We verify the closure conditions. Since $\{1\}\in\mathrm{Fi}(\mathbf{A})$
and by \textbf{One} every member of $\mathcal{F}_{S}$ contains $1$, we have $\mathrm{C}_{S}(\emptyset)=\mathrm{D}_{S}(\emptyset)=\{1\}$,
so $\mathbf{One}$ holds. If $W\subseteq_{f}Z$, then $\mathrm{Fg}^{\mathbf{A}}(W)\in\mathcal{F}_{S}$, $\mathrm{D}_{S}(\mathrm{Fg}^{\mathbf{A}}(W))=\mathrm{Fg}^{\mathbf{A}}(W)$ and $W\subseteq \mathrm{Fg}^{\mathbf{A}}(W)$, whence $\mathrm{D}_{S}(W)\subseteq\mathrm{Fg}^{\mathbf{A}}(W)$.
Therefore 
\[
\mathrm{C}_{S}(Z)\subseteq\bigcup_{W\subseteq_{f}Z}\mathrm{Fg}^{\mathbf{A}}(W)\subseteq \bigvee_{W\subseteq_{f}Z}\mathrm{Fg}^{\mathbf{A}}(W)=\mathrm{Fg}^{\mathbf{A}}(Z),
\]
so $\mathbf{Fi}$ holds. Every member of $\mathcal{F}_{S}$ is an
upset; hence each $\mathrm{D}_{S}(W)$, and therefore $\mathrm{C}_{S}(Z)$,
is an upset. Thus $\mathbf{Up}$ holds.

It remains to verify $\mathbf{Jo}$. First note that $\mathrm{D}_{S}$
satisfies it. Suppose $x\in\mathrm{D}_{S}(Z,a)\cap\mathrm{D}_{S}(Z,b)$,
and let $H\in\mathcal{F}_{S}$ contain $Z\cup\{a\vee b\}$. If $H$
is a proper Boolean filter and $x\notin H$, an ultrafilter extending
$H$ and omitting $x$ would contain $a$ or $b$, contradicting the
assumed membership of $x$ in $\mathrm{D}_{S}(Z,a)\cap\mathrm{D}_{S}(Z,b)$.
(For $H=A$ there is nothing to prove.) If $H=f^{-1}_{i}[F_{i}]$,
then $f_{i}(a)\vee f_{i}(b)=f_{i}(a\vee b)\in F_{i}$; since the clan
$F_{i}$ is join-prime, either $a\in H$ or $b\in H$, and again $x\in H$.
Hence $x\in\mathrm{D}_{S}(Z,a\vee b)$.

Now let $x\in\mathrm{C}_{S}(Z,a)\cap\mathrm{C}_{S}(Z,b)$. There are
finite $W_{1},W_{2}\subseteq Z$ such that $x\in\mathrm{D}_{S}(W_{1},a)$
and $x\in\mathrm{D}_{S}(W_{2},b)$. With $W=W_{1}\cup W_{2}$, monotonicity
and the preceding paragraph yield 
\[
x\in\mathrm{D}_{S}(W,a)\cap\mathrm{D}_{S}(W,b)\subseteq\mathrm{D}_{S}(W,a\vee b)\subseteq\mathrm{C}_{S}(Z,a\vee b).
\]
Thus $\langle\mathbf{A},\mathrm{C}_{S}\rangle$ is a sequent algebra
and the maps $f_{i}$ are sequent homomorphisms. Hence $S$ has a
lifting.

\label{lifting is initial} Let $\mathbf{A}$ be a Boolean
algebra and let $S=\{f_{i}:\mathbf{A}\to U(\langle\mathbf{B}_{i},\mathrm{C}_i\rangle)\}_{i\in I}$

<!-- proof-28: body_md -->
Let $\langle\mathbf{E},\mathrm{D}\rangle$ be a sequent
algebra and let $g:\mathbf{E}\to\mathbf{A}$ be a Boolean homomorphism 
such that every composite $f_{i}g$ is a sequent homomorphism. We
prove that $g$ is a sequent homomorphism. Let $W\subseteq E$ and
$x\in\mathrm{D}(W)$. By finitarity there is a finite
$W_{0}\subseteq W$ such that $x\in\mathrm{D}(W_{0})$.
It suffices to show that $g(x)\in\mathrm{D}_{S}(g[W_{0}])$.

Let $H\in\mathcal{F}_{S}$ and suppose $g[W_{0}]\subseteq H$. If
$H\in\mathrm{Fi}(\mathbf{A})$, then $\mathbf{Fi}$ gives $x\in\mathrm{Fg}^{\mathbf{E}}(W_{0})$. Since $g$ is a Boolean homomorphism, $g(x)\in\mathrm{Fg}^{\mathbf{A}}(g[W_{0}])\subseteq H$.
If instead $H=f^{-1}_{i}[F_{i}]$ for a clan $F_{i}\in X(\mathbf{B}_{i})$,
then $(f_{i}g)[W_{0}]\subseteq F_{i}$. Since $f_{i}g$ is a sequent
homomorphism, 
\[
(f_{i}g)(x)\in\mathrm{C}_{\mathbf{B}_{i}}((f_{i}g)[W_{0}])\subseteq F_{i},
\]
so again $g(x)\in H$. Therefore $g(x)$ belongs to every member of
$\mathcal{F}_{S}$ that contains $g[W_{0}]$, and hence 
\[
g(x)\in\mathrm{D}_{S}(g[W_{0}])\subseteq\mathrm{C}_{S}(g[W]).
\]
Thus $g$ is a sequent homomorphism, so the lifting is initial.

For uniqueness, if $\mathrm{C}$ gives another initial lifting on
the same Boolean algebra $\mathbf{A}$, initiality applied to the
identity map in both directions shows that $\mathrm{C}(Z)\subseteq\mathrm{C}_{S}(Z)$
and $\mathrm{C}_{S}(Z)\subseteq\mathrm{C}(Z)$ for every $Z\subseteq A$.
Hence $\mathrm{C}=\mathrm{C}_{S}$.

<!-- proposition-14: body_md -->
\label{fibre-complete} The poset of all finitary
closure operators on a Boolean algebra $\mathbf{A}$ satisfying $\mathbf{Fi}$,
$\mathbf{One}$, $\mathbf{Up}$, and $\mathbf{Jo}$ is a complete
lattice under pointwise inclusion. Its greatest element is $\mathrm{Fg}^{\mathbf{A}}$,
and its least element is the operator 
\[
\mathrm{C}_{\min}(Z)=\begin{cases}
\{1\}, & Z=\emptyset,\\
\uparrow Z, & Z\neq\emptyset.
\end{cases}
\]
<!-- end proposition-14 -->
<!-- proof-30: body_md -->
Let $\{\mathrm{C}_{i}\}_{i\in I}$ be a family of such operators and
define first 
\[
\mathrm{D}(Z)=\bigcap_{i\in I}\mathrm{C}_{i}(Z),\qquad\mathrm{C}_{\wedge}(Z)=\bigcup_{W\subseteq_{f}Z}\mathrm{D}(W).
\]
The operator $\mathrm{D}$ is a closure operator satisfying $\mathbf{Fi}$,
$\mathbf{One}$, $\mathbf{Up}$, and $\mathbf{Jo}$, although it need
not be finitary. Its finitarization $\mathrm{C}_{\wedge}$ is a finitary
closure operator and retains all four properties; the verification
of $\mathbf{Jo}$ uses the union of the two finite witnesses. Moreover,
$\mathrm{C}_{\wedge}\leq\mathrm{C}_{i}$ for every $i$, and every
finitary closure operator below all the $\mathrm{C}_{i}$ is below
$\mathrm{C}_{\wedge}$. Thus arbitrary meets exist. Since $\mathrm{Fg}^{\mathbf{A}}$
is a greatest element, the poset is a complete lattice.

The maximality of $\mathrm{Fg}^{\mathbf{A}}$ follows immediately
from $\mathbf{Fi}$. For the least element, $\mathrm{C}_{\min}$ is
readily checked to be a finitary closure operator satisfying all four
conditions. If $Z\neq\emptyset$ and $y\in\uparrow Z$, choose $z\in Z$
with $z\leq y$. Extensivity gives $z\in\mathrm{C}(Z)$, and $\mathbf{Up}$
yields $y\in\mathrm{C}(Z)$. If $Z=\emptyset$, $\mathbf{One}$ gives
$\mathrm{C}_{\min}(\emptyset)=\{1\}=\mathrm{C}(\emptyset)$. Hence
$\mathrm{C}_{\min}\leq\mathrm{C}$ for every sequent closure operator
$\mathrm{C}$.

<!-- remark-8: body_md -->
We stress that Proposition \ref{fibre-complete} allows
us to identify the functors $L$ and $R$ explicitly. Let $g:\mathbf{A}\to\mathbf{B}$

called the \emph{discrete functor}) is defined by $L(\mathbf{A})=\langle\mathbf{A},\mathrm{C}_{\min}\rangle$

<!-- proposition-15: body_md -->
\label{prop: SA not topological over BASH}
$\mathsf{SA}$ not topological over $\mathsf{BASH}$.
<!-- end proposition-15 -->
<!-- proof-31: body_md -->
Let $\mathbf A=\{0,a,b,1\}$ be the four-element Boolean algebra and
$g\colon \mathbf A\to V\langle\mathbf A,\mathrm{Fg}^{\mathbf A}\rangle$ the map with
$g(0)=0$ and $g(x)=1$ for $x\neq 0$. Since $g(x\vee y)=0$ iff $x=y=0$,
$g$ preserves finite joins and $1$, so it is a $\mathsf{BASH}$-morphism.

Observe that 
\[
\mathcal{F}_{g}=\{g^{-1}[F]\colon F\in X(\mathbf{A})\}\cup\mathrm{Fi}(\mathbf{A})=\{A, \{a,b,1\},\{a,1\}, \{b,1\}, \{1\}\},
\]
and since $\mathbf{A}$ is finite, for every $Z\subseteq A$ we get 
\[
\mathrm{C}_{g}(Z)=\bigcap\{H\in\mathcal{F}_{g}\colon Z\subseteq H\}=
\begin{cases}
A, & \text{if } 0\in Z,\\[2pt]
\{a,b,1\}, & \text{if } 0\notin Z \text{ and } a,b\in Z,\\[2pt]
\{a,1\}, & \text{if } a\in Z \text{ and } b,0\notin Z,\\[2pt]
\{b,1\}, & \text{if } b\in Z \text{ and } a,0\notin Z,\\[2pt]
\{1\}, & \text{if } Z\subseteq\{1\}.
\end{cases}
\]
Replicating the proof of Lemma~\ref{existence of liftings} (which only uses that the maps $f_i$ preserve finite joins and $1$),
$\langle\mathbf A,\mathrm C_g\rangle$ is a sequent algebra and
$g\colon\langle\mathbf A,\mathrm C_g\rangle\to U\langle\mathbf A,\mathrm{Fg}^{\mathbf A}\rangle$ is a sequent semi-homomorphism.
We claim that $\mathrm C_g$ is the only possible initial lifting of $\{g\}$.
Let $\mathrm C$ be any closure operator on $A$ such that $\langle\mathbf A,\mathrm C\rangle$ is a sequent algebra and
$g\colon\langle\mathbf A,\mathrm C\rangle\to U\langle\mathbf A,\mathrm{Fg}^{\mathbf A}\rangle$ is a sequent semi-homomorphism.
If $0\notin Z$, then $g[Z]\subseteq\{1\}$, so $g[\mathrm C(Z)]\subseteq\mathrm{Fg}^{\mathbf A}(g[Z])=\{1\}$, whence $0\notin\mathrm C(Z)$.
Together with $\mathbf{Fi}$ this gives $\mathrm C(Z)\subseteq\mathrm{Fg}^{\mathbf A}(Z)\setminus\{0\}\subseteq\mathrm C_g(Z)$,
and if $0\in Z$ then $\mathrm C_g(Z)=A\supseteq\mathrm C(Z)$. Hence $\mathrm C\subseteq\mathrm C_g$.


Now suppose $\mathrm C$ is an initial lifting of $\{g\}$. Since $\mathrm{id}_A$ is a $\mathsf{BASH}$-morphism and
$g\circ\mathrm{id}_A=g$ is a sequent semi-homomorphism out of $\langle\mathbf A,\mathrm C_g\rangle$,
initiality gives $\mathrm C_g\subseteq\mathrm C$. Thus $\mathrm C=\mathrm C_g$.
It remains to show that $\mathrm C_g$ is not initial. To do so, let $P=\{p,q,r\}$ and $\mathbf E$ be the Boolean algebra whose universe is $\mathcal P(P)$, with $0=\varnothing$ and $1=P$. Now we define for every $W\subseteq E$:
\[
\mathrm{D}(W)=
\begin{cases}
E, & \text{if } \varnothing\in W,\\[2pt]
\uparrow (\bigcap W)\setminus\{\varnothing\}, & \text{if } \varnothing\notin W.
\end{cases}
\]
\emph{Claim 1.} $\langle\mathbf{E},\mathrm{D}\rangle$ is a sequent algebra. 
\medskip

\noindent
By Proposition~\ref{prop: charact closure} we check that $\mathrm D$ is a finitary closure operator
satisfying $\mathbf{Fi}$, $\mathbf{One}$, $\mathbf{Up}$, $\mathbf{Jo}$.
Note that $\mathrm{Fg}^{\mathbf E}(W)=\uparrow (\bigcap W)$, so
$\mathrm D(W)=\mathrm{Fg}^{\mathbf E}(W)$ if $\varnothing\in W$ and
$\mathrm D(W)=\mathrm{Fg}^{\mathbf E}(W)\setminus\{\varnothing\}$ otherwise. To prove that $\mathrm{D}$ is a closure operator, we only show idempotency since the remainig conditions are easy. Let $K=\mathrm D(W)$. If $\varnothing\in W$, then $K=E$ and $\mathrm D(K)=E=K$.
Suppose $\varnothing\notin W$. Since $W\subseteq K$, we have $\bigcap K\subseteq\bigcap W$.
Conversely, every element of $K$ contains $\bigcap W$ and $K\neq\varnothing$ (it contains $P$),
so $\bigcap W\subseteq\bigcap K$. Thus $\bigcap K=\bigcap W$, and since $\varnothing\notin K$,
\[
\mathrm D(K)=\uparrow\!\bigl(\textstyle\bigcap K\bigr)\setminus\{\varnothing\}
=\uparrow\!\bigl(\textstyle\bigcap W\bigr)\setminus\{\varnothing\}=\mathrm D(W)=K.
\]

Observe that $\mathbf{Fi}$ holds by the remark above;
$\mathrm D(\varnothing)=\{P\}$ gives $\mathbf{One}$; and $\mathbf{Up}$ holds because
$\varnothing$ is the least element, so removing it from an up-set leaves an up-set. Finally, we prove \emph{$\mathbf{Jo}$.} Let $y\in\mathrm D(X,c)\cap\mathrm D(X,d)$ and $m=\bigcap X$.
If $\varnothing\in X$, then $\mathrm D(X,c\cup d)=E$. If $c=\varnothing$ (resp.\ $d=\varnothing$),
then $c\cup d=d$ (resp.\ $c$) and there is nothing to prove.
Otherwise $\varnothing\notin X\cup\{c\cup d\}$, $y\neq\varnothing$, and $m\cap c\subseteq y$, $m\cap d\subseteq y$,
so $m\cap(c\cup d)=\bigcap(X\cup\{c\cup d\})\subseteq y$. Hence $y\in\mathrm D(X,c\cup d)$.
\medskip

Define $h\colon\mathbf E\to\mathbf A$ by
\[
h(X)=\bigvee_{t\in X}h(\{t\}),\qquad h(\{p\})=h(\{q\})=a,\quad h(\{r\})=b,\quad h(\varnothing)=0 .
\]  
Then $h$ preserves finite joins and $h(P)=a\vee b=1$, so $h$ is a $\mathsf{BASH}$-morphism, and
$h(X)=0$ iff $X=\varnothing$. Note that $h$ does not preserve meets:
$h(\{p\})\wedge h(\{q\})=a$ while $h(\{p\}\cap\{q\})=0$.
\medskip

\noindent
\emph{Claim 2.} $g h$ is an sequent semi-homomorphism $\langle\mathbf E,\mathrm{D}\rangle\to\langle\mathbf A,\mathrm{Fg}^{\mathbf A}\rangle$.
\medskip

\noindent
It is a $\mathsf{BASH}$-morphism as a composite of such, and $g h(X)=0$ iff $X=\varnothing$.
If $\varnothing\in W$, then $0\in gh[W]$, so $\mathrm{Fg}^{\mathbf A}(gh[W])=A$ and there is nothing to prove.
If $\varnothing\notin W$, then $\varnothing\notin\mathrm{D}(W)$, so $gh[\mathrm{D}(W)]\subseteq\{1\}\subseteq \mathrm{Fg}^{\mathbf A}(gh[W])$.
\medskip

\noindent
\emph{Claim 3}: $h\colon\langle\mathbf E,\mathrm{D}\rangle\to\langle\mathbf A,\mathrm{C}_g\rangle$ is not an sequent semi-homomorphism.
\medskip

\noindent
Let $W=\{\{p\},\{q\}\}$. Then $\varnothing\notin W$ and $\{p\}\cap\{q\}=\varnothing$, so
$\mathrm{Fg}^{\mathbf E}(W)=E$ and $\{r\}\in E\setminus\{\varnothing\}=\mathrm{D}(W)$.
On the other hand $h[W]=\{a\}$, so $\mathrm{C}_g(h[W])=\mathrm{Fg}^{\mathbf A}(\{a\})=\{a,1\}$,
whereas $h(\{r\})=b\notin\{a,1\}$. Hence $h[\mathrm{D}(W)]\not\subseteq \mathrm{C}_g(h[W])$.

\medskip
\noindent By Claims 2 and 3, $h$ is a $\mathsf{BASH}$-morphism such that $gh$ is a sequent semi-homomorphism,
but $h\colon\langle\mathbf E,\mathrm D\rangle\to\langle\mathbf A,\mathrm C_g\rangle$ is not.
So $\mathrm C_g$ is not initial, and since it is the only candidate, $\{g\}$ has no initial lifting.
Hence $V$ is not topological.
