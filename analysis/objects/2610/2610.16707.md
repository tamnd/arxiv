<!-- head-1: body_md -->
This model captures applications such as ISAC, where edge nodes (e.g., dual-functional terminals) observe signals $X_i^{(k)}$ from the base station, and the center node combines side information $S$ (e.g., ambient temperature) with $U$ to estimate a target function $\psi(Y)$ (e.g., target velocity) rather than the high-dimensional $Y$ itself. Transmitting block averages also meets the low-latency requirements of ISAC, which preclude buffering long sequences, and aligns with gradient aggregation in distributed learning.
<!-- end head-1 -->
<!-- lemma-1: body_md -->
\label{lemma:optimal-estimator-fixed}
For any given local features $f$, the posterior distribution of $Y$ at the center node is approximated by
\begin{equation}
p_{Y|U,S}(y\mathrel{\mkern-2mu\vert\mkern-2mu} u,s) = p_{Y|S}(y\mathrel{\mkern-2mu\vert\mkern-2mu} s)\bigl[1 + N\,u^{\mathsf T}\Lambda_f^{-1}\beta(y,s)\bigr] + O(\epsilon^2).
\label{eq:posterior-leading}
\end{equation}
Consequently, for any $\psi \in \mathcal F_{\mathcal Y}$, the conditional expectation of $\psi(Y)$ given $U$ and $S$ is approximated by
\begin{equation}
\mathbb E[\psi(Y) \mathrel{\mkern-2mu\vert\mkern-2mu} U, S] = \mathbb E[\psi(Y) \mathrel{\mkern-2mu\vert\mkern-2mu} S] + N \cdot c_{\psi}(S)^{\mathsf T} \Lambda_f^{-1} \, U + O(\epsilon^2).
\label{eq:cond-expectation-leading}
\end{equation}
<!-- end lemma-1 -->
<!-- definition-1: body_md -->
\label{def:components}
The \emph{Markov-additive}, \emph{conditional-additive}, and \emph{interaction components} of the joint CDK are its projections onto the three subspaces in \eqref{eq:three-subspaces}:
\begin{equation*}
\begin{aligned}
\pi_{\mathsf M} &\triangleq \Pi\!\left(\mathfrak i_{X^K; S, Y};\, \mathcal A \otimes \mathcal F_{\mathcal S}\right) = \Pi\!\left(\mathfrak i_{X^K; S};\, \mathcal A \otimes \mathcal F_{\mathcal S}\right), \\
\pi_{\mathsf C} &\triangleq \Pi\!\left(\mathfrak i_{X^K; S, Y};\, \mathcal A \otimes \mathcal F_{\mathcal Y|\mathcal S}\right) = \Pi\!\left(\mathfrak i_{X^K; Y|S};\, \mathcal A \otimes \mathcal F_{\mathcal Y|\mathcal S}\right), \\
\pi_{\mathsf I} &\triangleq \mathfrak i_{X^K; S, Y} - \pi_{\mathsf M} - \pi_{\mathsf C}.
\end{aligned}
                           
\end{equation*}
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{prop:modal-decomposition}
For $\bar{f},\bar{g}, \tilde{f}, \tilde{g}$ defined in \eqref{eq:def-fb-tilde} and $\bar{d} \geq \operatorname{rank}\!\left( \pi_{\mathsf M} \right)$, the score \eqref{eq:hscore-mc} is maximized if and only if
\begin{equation}
\bar{f} \otimes \bar{g} = \pi_{\mathsf M}, \quad
                                                      
 \tilde{f} \otimes \tilde{g} =  \begin{cases}
   \zeta_{\leq \tilde{d}} (\pi_{\mathsf C}), & \tilde{d}<\operatorname{rank}\!\left( \pi_{\mathsf C} \right),\\    \pi_{\mathsf C}, & \tilde{d} \geq \operatorname{rank}\!\left( \pi_{\mathsf C} \right).
   \end{cases}
\label{eq:modal-decomposition-condition}
\end{equation}
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{prop:distributed-posterior}
With the local feature extractors $\tilde f^{(k)*}$, the posterior distribution and the conditional expectation at the center node are
\begin{align}
p_{Y|V,S}(y\mathrel{\mkern-2mu\vert\mkern-2mu} v,s) &= p_{Y|S}(y\mathrel{\mkern-2mu\vert\mkern-2mu} s)\bigl[ 1 + N v^{\mathsf T} \tilde{g}^{*}(y,s) \bigr] + O(\epsilon^2),
\label{eq:distributed-posterior}\\
\mathbb E[\psi(Y) \mathrel{\mkern-2mu\vert\mkern-2mu} V, S] &= \mathbb E[\psi(Y) \mathrel{\mkern-2mu\vert\mkern-2mu} S] + N \cdot V^{\mathsf T} \tilde{c}_{\psi}(S) + O(\epsilon^2),
\label{eq:distributed-expectation}
\end{align}
where $\tilde{c}_{\psi}(S) \triangleq \mathbb E\bigl[\tilde{g}^{*}(Y,S)\,\psi(Y) \,\big|\, S\bigr]$. Hence the MMSE estimator of $\psi$ in the leading order is
\begin{equation}
\hat{\psi}^{*}(V,S) = \mathbb E[\psi(Y) \mathrel{\mkern-2mu\vert\mkern-2mu} S] + N \cdot V^{\mathsf T} \tilde{c}_{\psi}(S),
\label{eq:optimal-estimator-distributed}
\end{equation}
and its MSE for any task $\psi \in \mathcal F_{\mathcal Y}$ is
\begin{equation}
\begin{split}
\mathsf{MSE}_{\psi}^{\mathrm{univ}}
=\;& \mathbb E\bigl[\operatorname{Var}(\psi(Y)\mathrel{\mkern-2mu\vert\mkern-2mu} S) \bigr] \\
&- N \,\mathbb E\bigl[\tilde{c}_{\psi}(S)^{\mathsf T} \Lambda_{\tilde{f}^{*}}  \tilde{c}_{\psi}(S) \bigr] + o(\epsilon^2).
\end{split}
\label{eq:mse-univ-prop4}
\end{equation}
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
Substitute $\Lambda_{\tilde f^*}^{-1}\tilde\beta = \tilde g^*$ from \eqref{eq:lambda-beta} into \eqref{eq:posterior-leading} and \eqref{eq:cond-expectation-leading}, and substitute $c_{\psi} = \Lambda_{\tilde f^*}\tilde{c}_{\psi}$, which follows from \eqref{eq:lambda-beta}, into \eqref{eq:mse-fixed-features}.
<!-- end proof-1 -->
<!-- theorem-1: body_md -->
\label{thm:universality-gap}
For any budget $\tilde d$ and any task $\psi \in \mathcal F_{\mathcal Y}$, the universal features in \eqref{eq:modal-decomposition-pic} satisfy
\begin{equation}
\mathsf{MSE}_{\psi}^{\mathrm{univ}} - \mathsf{MSE}_{\psi}^{*} = N \sum_{i=\tilde{d}+1}^{r} \sigma_i^2\, \mathbb E\bigl[\rho_{\psi,i}(S)^2\bigr] + o(\epsilon^2),
\label{eq:mse-gap}
\end{equation}
where $\rho_{\psi,i}(s) \triangleq \mathbb E[g_i^*(Y,s)\psi(Y)\mathrel{\mkern-2mu\vert\mkern-2mu} S=s]$ is the conditional correlation of the task with the $i$-th right mode.
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
Write $\mathfrak i_{X^K;Y|S} = \pi_{\mathsf C} + \pi_{\mathsf{IC}}$ with $\pi_{\mathsf{IC}} \triangleq \Pi\!\left(\mathfrak i_{X^K;Y|S};\, (\mathcal F_{\mathcal X^{K}} \mathbin{\boxminus} \mathcal A) \otimes \mathcal F_{\mathcal Y|\mathcal S}\right)$. In \eqref{eq:bar-psi-def}, the contribution of $\pi_{\mathsf{IC}}$ lies in $(\mathcal F_{\mathcal X^{K}} \mathbin{\boxminus} \mathcal A) \otimes \mathcal F_{\mathcal S}$ and is annihilated by the projection onto $\mathcal A \otimes \mathcal F_{\mathcal S}$, whereas that of $\pi_{\mathsf C}$ already lies in $\mathcal A \otimes \mathcal F_{\mathcal S}$. Hence
\begin{align*}
\Pi\!\left(\bar{\psi};\, \mathcal A \otimes \mathcal F_{\mathcal S}\right)(x^K,s) &= \bigl\langle \pi_{\mathsf C}(x^K, s, \cdot), \psi \bigr\rangle_{P_{Y|S=s}}\\
&= \textstyle\sum_{i=1}^{r} \sigma_i\, \rho_{\psi,i}(s)\, f_i^*(x^K),
\end{align*}
whose squared norm is $\sum_{i=1}^{r} \sigma_i^2\, \mathbb E[\rho_{\psi,i}(S)^2]$ by the orthonormality of $\{f_i^*\}$. Similarly, the definition of $\tilde c_\psi$ and \eqref{eq:modal-decomposition-pic} give
\begin{align*}
\tilde f^{*\mathsf T}(x^K)\,\tilde c_\psi(s) &= \mathbb E\bigl[(\tilde f^*\otimes\tilde g^*)(x^K,s,Y)\,\psi(Y) \bigm| S=s\bigr]\\
&= \textstyle\sum_{i=1}^{\min\{\tilde d, r\}} \sigma_i\, \rho_{\psi,i}(s)\, f_i^*(x^K),
\end{align*}
so that
\[
\mathbb E\bigl[\tilde{c}_{\psi}(S)^{\mathsf T} \Lambda_{\tilde{f}^{*}} \tilde{c}_{\psi}(S)\bigr] = \sum_{i=1}^{\min\{\tilde d, r\}} \sigma_i^2\, \mathbb E[\rho_{\psi,i}(S)^2].
\]
Substituting into \eqref{eq:single-task-optimal-mse-proj} and \eqref{eq:mse-univ-prop4} and subtracting yields \eqref{eq:mse-gap}.
<!-- end proof-2 -->
<!-- corollary-1: body_md -->
\label{cor:universality}
If $\tilde{d} \geq r$, then for every task $\psi \in \mathcal F_{\mathcal Y}$,
\[
\mathsf{MSE}_{\psi}^{\mathrm{univ}} = \mathsf{MSE}_\psi^{*} + o(\epsilon^2),
\]
i.e., the features are simultaneously optimal for all tasks.
