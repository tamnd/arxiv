<!-- problem-1: body_md -->
Given system \eqref{eq:uncertain-system}, a compact set ${\mathcal P}\subset\mathbb R^p$ such that $\theta^\star\in{\mathcal P}$, a safe set $\mathcal C$, and a goal point $x_{\rm c}$,
     
    design a control policy $k:\mathbb R\times\mathcal D\times\mathbb R^p\to\mathcal U$ such that the closed-loop trajectories under $u=k(t,x,\hat\theta)$ satisfy
    \begin{enumerate}
        \item \textbf{Reach}: For all $x(0)\notin \mathcal C$, there exists $T<\infty$ such that $x(t)\in \mathcal C$ for all $t\geq T$, and for all $x(0)\in \mathcal C$, $x(t)\in\mathcal C$ for all $t\geq 0$; and
        \item \textbf{Stabilize}: For all $x(0)\in \mathcal D$, $\lim_{t\to \infty}x(t) = x_{\rm c}$. 
    \end{enumerate}
<!-- end problem-1 -->
<!-- assumption-1: body_md -->
\label{assum:initial-bound}
A constant $\beta_0>0$ is known such that
$\|\theta^\star-\hat\theta(0)\|\le\beta_0.$
<!-- end assumption-1 -->
<!-- proposition-1: body_md -->
\label{prop:error-bound}
Let Assumption \ref{assum:initial-bound} hold, and define
\begin{align}
    \beta_{\rm p}(t)
    \stackrel{\triangle}{=}
    \left(
    \beta_0^2-2\Gamma\int_{0}^{t}\|e(\tau)\|^2\,{\rm d}\tau
    \right)^{1/2}.
    \label{eq:beta-p}
\end{align}
 
Then, for all $t\ge0,$ $\beta_{\rm p}(t)$ is well defined and nonnegative,
$\beta_{\rm p}$ is nonincreasing, and $|\tilde\theta(t)|\le\beta_{\rm p}(t).$
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Since $\dot{\tilde\theta}=-\dot{\hat\theta},$ it follows from
\eqref{eq:prediction-error} and \eqref{eq:gradient-law} that
\begin{align}
    \frac{{\rm d}}{{\rm d} t}\frac{\|\tilde\theta\|^2}{2}
    =
    -\Gamma\tilde\theta^{\rm T}\Phi_{\rm f}^{\rm T}\Phi_{\rm f}\tilde\theta
    =
    -\Gamma\|e\|^2,
    \label{eq:lyapunov-identity}
\end{align}
and thus, for all $t\ge0,$
\begin{align}
    \|\tilde\theta(t)\|^2
    =
    \|\tilde\theta(0)\|^2-2\Gamma\int_{0}^{t}\|e(\tau)\|^2\,{\rm d}\tau.
    \label{eq:error-identity}
\end{align}
 
Since $\|\tilde\theta(t)\|^2\ge0,$ \eqref{eq:error-identity} implies
$2\Gamma\int_{0}^{t}\|e(\tau)\|^2\,{\rm d}\tau\le\|\tilde\theta(0)\|^2,$ which, with
Assumption \ref{assum:initial-bound}, implies that the argument of the square
root in \eqref{eq:beta-p} is nonnegative, and hence $\beta_{\rm p}$ is well
defined and nonnegative.
 
Furthermore, $\beta_{\rm p}$ is nonincreasing since the integral in
\eqref{eq:beta-p} is nondecreasing.
 
Finally, \eqref{eq:error-identity} and Assumption \ref{assum:initial-bound}
imply $\|\tilde\theta(t)\|^2\le\beta_{\rm p}^2(t).$

<!-- proposition-2: body_md -->
\label{prop:exact-error}
For all $t\ge0,$ $\overline\Phi(t)\ge0$ and
\begin{align}
    \Xi(t)=-\overline\Phi(t)\tilde\theta(t).
    \label{eq:xi-identity}
\end{align}
 
Furthermore, if $\overline\Phi(t)>0,$ then
$\tilde\theta(t)=-\overline\Phi^{-1}(t)\Xi(t).$
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
It follows from \eqref{eq:linear-regression}, \eqref{eq:acc-phi}, and
\eqref{eq:acc-x} that
\begin{align}
    \frac{{\rm d}}{{\rm d} t}
    \left(\overline X-\overline\Phi\theta^\star\right)
    =
    -\mu\left(\overline X-\overline\Phi\theta^\star\right),
    \label{eq:acc-error-dynamics}
\end{align}
which, since $\overline X(0)-\overline\Phi(0)\theta^\star=0,$ implies
$\overline X=\overline\Phi\theta^\star,$ and thus \eqref{eq:xi} implies
\eqref{eq:xi-identity}.
 
Furthermore, \eqref{eq:acc-phi} implies
\begin{align}
    \overline\Phi(t)
    =
    \int_0^te^{-\mu(t-\tau)}\|\Phi_{\rm f}(\tau)\|^2\,{\rm d}\tau,
    \label{eq:acc-phi-solution}
\end{align}
which is nonnegative.

<!-- corollary-1: body_md -->
\label{cor:beta}
For all $t\ge0,$ $\beta(t)\ge0,$ $\beta$ is nonincreasing, and
$|\tilde\theta(t)|\le\beta(t).$
<!-- end corollary-1 -->
<!-- proof-3: body_md -->
Proposition \ref{prop:error-bound} implies $\beta_{\rm p}\ge0,$ and Proposition
\ref{prop:exact-error} implies that each term in the infimum in
\eqref{eq:beta} is nonnegative, and hence $\beta\ge0.$
 
Furthermore, $\beta_{\rm p}$ is nonincreasing, and ${\mathcal T}_{t_1}\subseteq{\mathcal T}_{t_2}$
for all $t_2\ge t_1,$ which implies that the infimum in \eqref{eq:beta} is
nonincreasing, and hence $\beta$ is nonincreasing.
 
Finally, Proposition \ref{prop:exact-error} implies that, for all
$\tau\in{\mathcal T}_t,$ $|\tilde\theta(\tau)|=|\overline\Phi^{-1}(\tau)\Xi(\tau)|,$
which, since $|\tilde\theta|$ is nonincreasing by \eqref{eq:error-identity},
implies $|\tilde\theta(t)|\le|\overline\Phi^{-1}(\tau)\Xi(\tau)|,$ and
combining with Proposition \ref{prop:error-bound} yields
$|\tilde\theta(t)|\le\beta(t).$

<!-- theorem-1: body_md -->
\label{thm:safety with delta}
    Under the conditions of Corollary \ref{cor:beta}, let $\gamma>0$ and $\rho\in(0,1),$ and let
    \begin{align}\label{eq: delta CBF}
        \delta_h(t) = \|L_\phi h(x)\||\beta(t)|.
    \end{align}
     
    If there exists a control signal $u:\mathbb R\to \mathcal U$ such that
    \begin{align}\label{eq: cbf delta cond}
        L_fh(x(t)) + L_gh(x(t))u + L_\phi h(x)\hat \theta(t) & \nonumber\\
        +\gamma\operatorname{sgn}(h(x))|h(x)|^\rho & \geq \delta_h(t)
    \end{align}
    along the solution $x(\cdot)$ of \eqref{eq:uncertain-system} for all $t\geq 0$, then the set $\mathcal C$ is forward invariant for \eqref{eq:uncertain-system} for every $x(0)\in \mathcal C.$
     
    Furthermore, for every $x(0)\notin \mathcal C,$ the trajectory reaches $\mathcal C$ at a time
    \begin{align}\label{eq: recovery time}
        T\le\frac{|h(x(0))|^{1-\rho}}{\gamma(1-\rho)}.
    \end{align}
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
Under the conditions of Corollary \ref{cor:beta}, it holds that the parameter estimation error is bounded by $\beta$.
     
    For a control signal $u$, let $x$ be the solution of \eqref{eq:uncertain-system}.
     
    If the condition \eqref{eq: cbf delta cond} holds along the solution $x(\cdot)$, we have
    \begin{align*}
        L_fh(x(t)) + L_gh(x(t))u + & L_\phi h(x)\hat \theta(t) \nonumber\\
        +\gamma\operatorname{sgn}(h(x))|h(x)|^\rho & \geq \delta_h(t)\\
        \implies L_fh(x(t)) + L_gh(x(t))u + & L_\phi h(x)\theta^\star  \nonumber\\
        +\gamma\operatorname{sgn}(h(x))|h(x)|^\rho & \geq \delta_h(t) + L_\phi h(x)\tilde \theta(t).
    \end{align*}
    Now, it holds that
    \begin{align*}
        \delta_h(t) + L_\phi h(x)\tilde \theta(t) & \geq \delta_h(t) - \|L_\phi h(x)\tilde \theta(t)\|\\
        & \geq \delta_h(t) - \|L_\phi h(x)\||\beta(t)|\\
        & \geq 0,
    \end{align*}
    for all $t\geq 0$, where the second inequality follows from Corollary \ref{cor:beta} and the last inequality follows from \eqref{eq: delta CBF}.
     
    Consequently, we have that
    \begin{align*}
        \dot h(t) \geq -\gamma\operatorname{sgn}(h(x(t)))|h(x(t))|^\rho.
    \end{align*}
     
    Suppose $x(0)\in\mathcal C$ and $h(x(t_1))<0$ for some $t_1>0.$
     
    Let $t_0\in[0,t_1)$ be the last time at which $h(x(t_0))\geq0.$
     
    Then $h(x(t_0))=0$ and $h(x(t))<0$ on $(t_0,t_1],$ so $\dot h\geq\gamma|h|^\rho>0$ on $(t_0,t_1],$ which contradicts $h(x(t_1))<h(x(t_0)).$
     
    Hence $\mathcal C$ is forward invariant.
     
    Now suppose $h(x(0))<0$ and let $w=-h.$
     
    While $w>0,$ $\dot w\leq-\gamma w^\rho,$ which implies
    \begin{align*}
        \frac{{\rm d}}{{\rm d} t}w^{1-\rho}=(1-\rho)w^{-\rho}\dot w\leq-\gamma(1-\rho).
    \end{align*}
     
    Since $w^{1-\rho}\geq0,$ $w$ reaches zero no later than the time in \eqref{eq: recovery time}, and forward invariance then holds from that time on.

\label{thm:convergence with delta}
    Under the conditions of Corollary \ref{cor:beta}, with
    \begin{align}\label{eq: delta CLF}
        \delta_V(t) = \|L_\phi V(x)\||\beta(t)|,
    \end{align}
    if there exists a control signal $u:\mathbb R\to \mathcal U$ such that the following holds:
    \begin{align}\label{eq: clf delta cond}
         L_fV(x(t)) + L_gV(x(t))u + L_\phi V(x(t))\hat \theta(t) \nonumber \\
         \leq -\lambda V(x(t)) - \delta_V(t),
    \end{align}
    along the solution $x(\cdot)$ of \eqref{eq:uncertain-system} for every $x(0)\in \mathcal D$ and for all $t\geq 0$, then $V(x(t))$ converges to $0$ exponentially and $x(t)$ converges to $x_{\rm c}$ for every $x(0)\in \mathcal D$.

<!-- proof-5: body_md -->
Under the conditions of Corollary \ref{cor:beta}, it holds that the parameter estimation error is bounded by $\beta$.
     
    For a control signal $u$, let $x$ be the solution of \eqref{eq:uncertain-system}.
     
    If the condition \eqref{eq: clf delta cond} holds along the solution $x(\cdot)$, we have
    \begin{align*}
       L_fV(x(t)) + & L_gV(x(t))u + L_\phi V(x(t))\hat \theta(t) \\
       \leq &-\lambda V(x(t)) - \delta_V(t)\\
       \implies L_fV(x(t)) + & L_gV(x(t))u + L_\phi V(x(t))\theta^\star \\
       \leq &-\lambda V(x(t)) - \delta_V(t) + L_\phi V(x(t))\tilde \theta(t).
    \end{align*}
    Now, it holds that
    \begin{align*}
        L_\phi V(x(t))\tilde \theta(t) - \delta_V(t) & \leq \|L_\phi V(x(t))\tilde \theta(t)\| - \delta_V(t)\\
         & \leq \|L_\phi V(x(t))\||\beta(t)| - \delta_V(t)\\
         & \leq 0,
    \end{align*}
    where the second inequality follows from Corollary \ref{cor:beta} and the last inequality follows from \eqref{eq: delta CLF}.
     
    Consequently, we have that
    \begin{align*}
     L_fV(x(t)) + & L_gV(x(t))u + L_\phi V(x(t))\theta^\star \\
       \leq &-\lambda V(x(t)),
    \end{align*}
    which implies $\dot V(x(t))\leq-\lambda V(x(t)),$ and thus $V(x(t))\leq e^{-\lambda t}V(x(0))$ for all $t\geq0.$
     
    Since $\alpha(\|x(t)-x_{\rm c}\|)\leq V(x(t))$ and $\alpha$ is of class $\mathcal K,$ $x(t)$ converges to $x_{\rm c}.$

<!-- remark-1: body_md -->
\label{rem:feasibility}
The set $\mathcal U$ enters Theorems \ref{thm:safety with delta} and \ref{thm:convergence with delta} only through the existence of a feasible $u,$ and the proofs hold for $\mathcal U={\mathbb R}^m.$
 
The bound on $u$ is one cause of infeasibility, since the recovery term $\gamma|h|^\rho$ and the margins $\delta_h$ and $\delta_V,$ which are largest at initialization where $\beta=\beta_0,$ increase the required control authority.
 
Enlarging $\mathcal U$ thus enlarges the set of states at which the QP is feasible.
 
Without a bound on $u,$ the QP is infeasible at a state $x$ only if $L_gh(x)=0$ or $L_gV(x)=0$ and the drift terms do not satisfy the corresponding constraint, or if $L_gh(x)$ and $L_gV(x)$ are positively proportional and their right-hand sides are incompatible.
<!-- end remark-1 -->
<!-- corollary-2: body_md -->
Under the conditions of Corollary \ref{cor:beta}, if the QP \eqref{eq:delta QP} is feasible for all $x\in \mathcal D$ and the solution $u$ of the QP is continuous, then the following holds for the resulting closed-loop solution $x(\cdot)$ of \eqref{eq:uncertain-system}:
    \begin{enumerate}
        \item For all $x(0)\in \mathcal D,$ $V(x(t))$ converges to $0$ exponentially and $x(t)$ converges to $x_{\rm c};$
        \item For all $x(0)\in \mathcal C,$ $x(t)\in \mathcal C$ for all $t\geq 0;$ and
        \item For all $x(0)\notin \mathcal C,$ there exists $T\leq |h(x(0))|^{1-\rho}/(\gamma(1-\rho))$ such that $x(t)\in \mathcal C$ for all $t\geq T.$
    \end{enumerate}
