\label{thm:main}
Under \eqref{eq:assumptions},
\begin{equation}
\lambda_1^D(M)\ge\max\{nK,B_\eta\}.
\label{eq:main-bound}
\end{equation}
For $0<s<1$ one has the strict model estimate
\begin{equation}
B_\eta<\mu<\lambda_1^D(M).
\label{eq:strict-model}
\end{equation}
At $s=1$, the weighted Kasue model has first eigenvalue
$\mu=nK$ and $B_\eta<nK$.

<!-- corollary-1: body_md -->

\begin{equation}
\frac{X}{\Lambda_R}
=1+cs+\beta a_0s^2+\beta a_1s^3+O_n(s^4),

\label{eq:main-expansion}
\end{equation}

\mu-B_\eta=O_n(\Lambda_Rs^4),\qquad
\frac{\mu-B_\eta}{\mu}=O_n(s^4).

<!-- theorem-2: body_md -->
\label{thm:geometric-sharpness}
Fix $n\ge2$, $K>0$, and $0<R\le\pi/(2\sqrt K)$.
Let $\mathcal C(n,K,R)$ consist of the compact connected
$n$-dimensional Riemannian manifolds with smooth nonempty boundary
satisfying \eqref{eq:assumptions} and $\operatorname{InRad}(M)=R$. Then
\[
\inf_{M\in\mathcal C(n,K,R)}\lambda_1^D(M)=\mu(n,K,R).
\]
For $R<\pi/(2\sqrt K)$ the infimum is not attained, but is approached
by smooth rotationally symmetric balls with sectional curvature at
least $K$, totally geodesic boundary, and inradius exactly $R$.
At $R=\pi/(2\sqrt K)$ it is attained by the round hemisphere of
sectional curvature $K$.

<!-- lemma-1: body_md -->
\label{lem:reduction}
Let $\lambda_\beta^{ND}$ be the first eigenvalue of
\[
H_\beta=-\frac{d^2}{dr^2}+\beta W
\]
on $[0,R]$ with Neumann condition at $0$ and Dirichlet condition at
$R$. Then
\begin{equation}
\mu=\alpha+\lambda_\beta^{ND},
\qquad
\mu\le L+\beta KJ(s).
\label{eq:reduction}
\end{equation}

<!-- proof-1: body_md -->
Let $\phi>0$ on $(0,R]$ be the first eigenfunction of
\eqref{eq:kasue-model}. With
$p(r)=\cos^{2c}(kr)$, the equation is
$-(p\phi')'=\mu p\phi$. Its mixed Rayleigh quotient gives $\mu>0$.
Integrating from $r$ to $R$ therefore yields
\[
p(r)\phi'(r)=\mu\int_r^R p(t)\phi(t)\,dt>0
\qquad(0\le r<R),
\]
so $y:=\phi'$ is positive on $[0,R)$. Put
$F=2ck\tan(kr)$. Differentiating gives
\begin{equation}
-y''+Fy'+F'y=\mu y,
\qquad y'(0)=0,\quad y(R)=0.
\label{eq:derivative}
\end{equation}
Here $y'(0)=\phi''(0)=0$ follows directly from
\eqref{eq:kasue-model} at $r=0$.

The gauge transform $v=\cos^c(kr)y$ yields
\[
-v''+c(c+1)K\tan^2(kr)v=(\mu-cK)v.
\]
The boundary conditions are $v'(0)=v(R)=0$, and $v>0$ on $(0,R)$.
Sturm oscillation identifies $v$ as the ground state, so
$\mu=\alpha+\lambda_\beta^{ND}$.

Testing the Rayleigh quotient of $H_\beta$ with
$\psi_0(r)=\cos(\sqrt{\Lambda_R}r)$ gives
$\lambda_\beta^{ND}\le\Lambda_R+\beta\overline W$, and therefore the
upper bound in \eqref{eq:reduction}.

<!-- theorem-3: body_md -->
\label{thm:moment-optimality}
Let $0<m_1<1$ and $m_1^2<m_2<m_1$. Let $\mathcal P(m_1,m_2)$ be
the set of probability measures on $[0,1]$ with first and second
moments $m_1$ and $m_2$. Then
\begin{equation}
\min_{\nu\in\mathcal P(m_1,m_2)}J_\nu(s)
=\frac{m_1s}{1-(m_2/m_1)s}
\qquad(0<s\le1).
\label{eq:moment-envelope}
\end{equation}
For each $s$, the unique minimizing measure is
\begin{equation}
\nu_*=
\left(1-\frac{m_1^2}{m_2}\right)\delta_0
+\frac{m_1^2}{m_2}\delta_{m_2/m_1},
\label{eq:extremal-law}
\end{equation}
where $\delta_x$ is the Dirac measure at $x$.
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
Fix $\nu\in\mathcal P(m_1,m_2)$ and $0<s\le1$. If
$J_\nu(s)=+\infty$, the inequality is immediate. In the finite case,
$\nu(\{1\})=0$ when $s=1$. Applying Cauchy--Schwarz to the
factorization $y=\sqrt{y/(1-sy)}\sqrt{y(1-sy)}$ gives
\[
m_1^2
\le\left(\int\frac{y}{1-sy}\,d\nu\right)
   \left(\int y(1-sy)\,d\nu\right)
=\frac{J_\nu(s)}s(m_1-sm_2).
\]
Since $m_1-sm_2>0$, this gives the lower bound in
\eqref{eq:moment-envelope}.
Equality holds precisely when $y$ is constant on $\{y>0\}$,
$\nu$-almost surely. Its value must be $m_2/m_1$; the total mass
and first moment therefore determine \eqref{eq:extremal-law} uniquely.
Direct substitution shows that this law has the prescribed second
moment and attains the bound for every $s$.
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lem:moments}
There is a probability measure $\nu_0$ on $[0,1]$ with
$\nu_0((0,1))=1$ and nonconstant coordinate variable $Y$ such that
\begin{equation}
J(s)=J_{\nu_0}(s),\qquad
\mathbb E Y=a_0,\quad \mathbb E Y^2=a_1,
\qquad 0<s\le1.
\label{eq:model-moments}
\end{equation}
In particular, $0<a_0<1$ and $a_0^2<a_1<a_0$. Moreover,
$J(1)=1$ and
\begin{equation}
J(s)=a_0s+a_1s^2+O(s^3)
\qquad(s\downarrow0).
\label{eq:Jexp}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Logarithmic differentiation of the cosine product
\cite[Eq.~4.22.2]{DLMF} gives
\[
\frac{\tan z}{z}=\frac8{\pi^2}\sum_{j=0}^{\infty}
\frac1{(2j+1)^2-4z^2/\pi^2},\qquad |z|<\frac\pi2.
\]
On $[0,\pi/2]\times\mathbb N_0$ define
\[
d\mathbb P(x,j)=\frac{32x\sin(2x)}{\pi^3(2j+1)^2}\,dx,
\qquad
Y(x,j)=\frac{4x^2}{\pi^2(2j+1)^2}.
\]
The choice of $Y$ matches the partial-fraction denominators;
$\sin(2x)$ comes from integrating the weight $\cos^2x$ by parts.
Since $\sum_{j\ge0}(2j+1)^{-2}=\pi^2/8$ and
$\int_0^{\pi/2}x\sin(2x)\,dx=\pi/4$, this is a probability measure.
The law $\nu_0$ of $Y$ assigns full mass to $(0,1)$, and $Y$ is
nonconstant.
Tonelli's theorem and the partial-fraction identity give, for $0<s<1$,
\begin{align*}
\mathbb E\frac1{1-sY}
&=\frac{32}{\pi^3}\int_0^{\pi/2}x\sin(2x)
  \sum_{j=0}^{\infty}
  \frac{dx}{(2j+1)^2-4sx^2/\pi^2}\\
&=\frac4{\pi\sqrt s}\int_0^{\pi/2}
  \sin(2x)\tan(\sqrt s\,x)\,dx\\
&=\frac4\pi\int_0^{\pi/2}
  \cos^2x\,\sec^2(\sqrt s\,x)\,dx
=1+J(s).
\end{align*}
The integration by parts uses
$\sin(2x)=-(\cos^2x)'$ and vanishing boundary terms. Subtracting $1$
gives $J(s)=s\mathbb E[Y/(1-sY)]$. At $s=1$, direct integration gives
\[
J(1)=\frac4\pi\int_0^{\pi/2}\sin^2x\,dx=1,
\]
and monotone convergence extends the representation to this endpoint.

Expanding $(1-sY)^{-1}$ uniformly near $s=0$ and using the Taylor
series of $\tan^2(\sqrt s\,x)$ yields
\begin{align*}
\mathbb E Y
&=\frac4\pi\int_0^{\pi/2}x^2\cos^2x\,dx
=\frac{\pi^2-6}{12}=a_0,\\

&=\frac8{3\pi}\int_0^{\pi/2}x^4\cos^2x\,dx
=\frac{\pi^4-20\pi^2+120}{120}=a_1.
\end{align*}
The strict moment inequalities follow from $0<Y<1$ almost surely
and the nonconstancy of $Y$. The same expansion proves \eqref{eq:Jexp}.

<!-- lemma-4: body_md -->
\label{lem:corrector}
For $0<s<1$, define on $[0,R)$
\begin{equation}
f(r)=\psi_0(r)^{-2}
\int_0^r(\overline W-W(t))\psi_0(t)^2\,dt.
\label{eq:f}
\end{equation}
Then $f$ extends smoothly to $[0,R]$, vanishes at both endpoints, and
satisfies the sharp estimates
\begin{align}
0<f(r)&<\frac{\overline W}{k}\tan(kr)
\qquad(0<r<R),\label{eq:f-bound}\\
\sup_{0<r<R}\frac{f(r)^2}{W(r)}&=J(s)^2.
\label{eq:sharp-corrector}
\end{align}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Let $A(r)$ denote the integral in \eqref{eq:f}. Since $W$ is strictly
increasing, $\overline W-W$ changes sign once, from positive to negative.
Thus $A$ first increases and then decreases. Its endpoint values are
zero by the definition of $\overline W$, so $A>0$ on $(0,R)$.

Near $R$, the factor $\psi_0^2$ has a zero of order two, whereas
$\overline W-W$ is smooth for $s<1$. Consequently
$A(r)=-\int_r^R(\overline W-W)\psi_0^2\,dt$ has a zero of order three:
\[
A(r)=(R-r)^3a_R(r),\qquad
\psi_0(r)=(R-r)b_R(r),
\]
with smooth $a_R,b_R$ and
$b_R(R)=\sqrt{\Lambda_R}\ne0$. Hence
$f=(R-r)a_R/b_R^2$ extends smoothly and vanishes at $R$; smoothness and
$f(0)=0$ at the left endpoint are immediate from \eqref{eq:f}.

To prove the upper bound, put $a=\sqrt{\Lambda_R}>k$ and consider
the weighted difference
\[
Z=\psi_0^2\left[\frac{\overline W}{k}\tan(kr)-f\right].
\]
A differentiation gives
\[
Z'=\psi_0^2\tan^2(kr)
\left[\overline W+K
-2\overline W\frac ak\frac{\tan(ar)}{\tan(kr)}\right].
\]
The ratio $\tan(ar)/\tan(kr)$ is strictly increasing. Indeed,
\[
\frac{d}{dr}\log\frac{\tan(ar)}{\tan(kr)}
=2\left(\frac a{\sin(2ar)}-\frac k{\sin(2kr)}\right)>0,
\]
because $0<2kr<2ar<\pi$ and $t/\sin t$ is strictly increasing on
$(0,\pi)$. Hence the bracket in the formula for $Z'$ is strictly
decreasing.

Since $Z(0)=Z(R)=0$, one has $\int_0^R Z'\,dr=0$.
The prefactor $\psi_0^2\tan^2(kr)$ is positive on $(0,R)$, so the
strictly decreasing bracket takes both signs and vanishes exactly once.
Consequently $Z$ increases from zero and then decreases to zero,
remaining positive on $(0,R)$. This proves the upper bound in
\eqref{eq:f-bound}; the lower bound follows from $A>0$.
Since $\sqrt W=k\tan(kr)$ and $\overline W=KJ(s)$, it follows that
$f^2<J(s)^2W$ in $(0,R)$. Finally, \eqref{eq:f} gives
$f'(0)=\overline W$, whereas $W(r)/r^2\to K^2$ as $r\downarrow0$.
Therefore $f(r)^2/W(r)\to J(s)^2$, proving \eqref{eq:sharp-corrector}.

At $s=1$, let $R=\pi/(2\sqrt K)$ and
$p(r)=\cos^{n-1}(\sqrt K\,r)$. The divergence-form operator
$-p^{-1}(p\phi')'$ acts in $L^2((0,R),p(r)\,dr)$ with form domain
\begin{align*}
\mathcal V_p
&=\overline{\{\psi\in C^\infty([0,R]):\psi(0)=0\}}^{\,\|\cdot\|_{H^1_p}},\\
\|\psi\|_{H^1_p}^2
&=\int_0^R p(r)\bigl(|\psi'(r)|^2+|\psi(r)|^2\bigr)\,dr.
\end{align*}
Since $p$ is bounded away from zero near $0$, the trace there is
continuous, and every $\psi\in\mathcal V_p$ satisfies $\psi(0)=0$.
The model operator is the self-adjoint operator associated with the
closed quadratic form
$\mathfrak a_p[\psi]=\int_0^R p(r)|\psi'(r)|^2\,dr$ on
$\mathcal V_p$. Its first eigenvalue is characterized by
\[
\mu=\inf_{\psi\in\mathcal V_p\setminus\{0\}}\mathcal Q[\psi],
\qquad
\mathcal Q[\psi]
=\frac{\int_0^R p(r)|\psi'(r)|^2\,dr}
       {\int_0^R p(r)|\psi(r)|^2\,dr}.
\]
The form domain prescribes no boundary condition at $R$; the condition
there is natural for the degenerate weight. The positive function
$\phi(r)=\sin(\sqrt K\,r)$ satisfies
$-(p\phi')'=nKp\phi$. For a smooth test function
$\psi=\phi g$ with $\psi(0)=0$, the ground-state identity gives
\[
\int_0^R p\bigl(|\psi'|^2-nK|\psi|^2\bigr)\,dr
=\int_0^R p\phi^2|g'|^2\,dr\ge0;
\]
the boundary term vanishes because $\phi(0)=0$ and
$p(R)\phi'(R)=0$. By density, the same inequality holds on the form
domain. Since $\phi\in\mathcal V_p$ and $\mathcal Q[\phi]=nK$, the
minimum is attained at $\phi$ and the first weighted eigenvalue is
$\mu=nK$.

For the algebraic bound, $\Lambda_R=K$ and
$j_\eta(1)<J(1)=1$ by Lemma~\ref{lem:moment-minorant}. Hence
\[
B_\eta=L+\Phi(j_\eta(1))<L+\Phi(1)=nK,
\]
since $\sqrt{1+4\beta}=2c+1=n$.
Together with Reilly's estimate \eqref{eq:reilly}, this completes
Theorem~\ref{thm:main} at $s=1$.

For $0<s<1$, use $y=\phi'$ and
$F=2c\sqrt K\tan(\sqrt K r)$ in \eqref{eq:derivative}. Then
$y>0$ on $[0,R)$ and
$F'=2cK\sec^2(\sqrt K r)\ge2\alpha$. To balance the gradient and
curvature contributions, fix $p>1$ and put $z=y^{p/2}$. Since
$y'(R)=-\mu\phi(R)<0$, the zero of $y$ at $R$ is simple; hence
$z'(r)=O((R-r)^{p/2-1})$ and $z\in H^1(0,R)$ for every $p>1$.

$[0,R-\varepsilon]$. The terms at $0$ vanish because $y'(0)=F(0)=0$.

=\frac{4(p-1)}{p^2}\int_0^R(z')^2\,dr

\qquad t=\frac{p-1}{p}\in(0,1).

\end{equation}

<!-- proof-13: body_md -->
Assume first $0<R<\pi/(2\sqrt K)$ and put $k=\sqrt K$.
We construct a smooth rotationally symmetric ball whose radial volume
density agrees with the model density outside a shrinking cap.

\paragraph{The cap.}
Fix $0<\delta<R$ and set
\[
h(t)=\cos(k(R-t)),\qquad 0\le t\le R.
\]
Then $h>0$, $h'>0$ on $[0,R)$, $h'(R)=0$, and $h''+Kh=0$.
Choose a nonzero $\rho_\delta\in C_c^\infty((0,\delta))$ with
$\rho_\delta\ge0$, and let
\[
a_\delta=\frac{h(0)}{\int_0^\delta h(s)\rho_\delta(s)\,ds},
\qquad
F_\delta''+KF_\delta=-a_\delta\rho_\delta,
\qquad F_\delta(0)=0,\quad F_\delta'(0)=1.
\]
Here $\rho_\delta$ is extended by zero outside $(0,\delta)$.
Near $0$, one has $F_\delta(t)=\sin(kt)/k$, so the metric below
extends smoothly across its pole.

For $\mathcal W=hF_\delta'-h'F_\delta$, differentiation gives
$\mathcal W'=-a_\delta h\rho_\delta$ and $\mathcal W(0)=h(0)$.
Thus
\[
\mathcal W(t)=a_\delta\int_t^\delta h(s)\rho_\delta(s)\,ds\ge0
\quad(0\le t\le\delta),\qquad
\mathcal W(t)=0\quad(\delta\le t\le R).
\]
Writing $y=F_\delta/h$, we have $y'=\mathcal W/h^2\ge0$.
Since $y(0)=0$ and $y'(0)>0$, it follows that $F_\delta>0$ on
$(0,R]$ and $F_\delta'=h'y+hy'>0$ on $(0,R)$.
Moreover $y$ is a positive constant $c_\delta$ on $[\delta,R]$,
so
\[
F_\delta(t)=c_\delta h(t)\quad(\delta\le t\le R),
\qquad F_\delta'(R)=0,
\qquad -F_\delta''\ge KF_\delta.
\]

\paragraph{Geometry.}
On the $n$-ball with pole $t=0$, use the metric
\[
g_\delta=dt^2+F_\delta(t)^2g_{\mathbb S^{n-1}},
\qquad 0\le t\le R,
\]
where $g_{\mathbb S^{n-1}}$ is the unit round metric.
The radial sectional curvature is $-F_\delta''/F_\delta\ge K$.
For $n\ge3$, the tangential sectional curvature is
$(1-(F_\delta')^2)/F_\delta^2$. To estimate it, observe that
\[
\frac{d}{dt}\bigl((F_\delta')^2+KF_\delta^2\bigr)
=2F_\delta'(F_\delta''+KF_\delta)\le0.
\]
The value at $0$ is $1$, hence
$1-(F_\delta')^2\ge KF_\delta^2$.
The warped-product curvature formula expresses the sectional curvature
of an arbitrary plane as a convex combination of the radial and
tangential values. Thus every sectional curvature is at least $K$;
for $n=2$ the radial calculation alone suffices.
The estimate extends across the pole by smoothness and implies
$\operatorname{Ric}_{g_\delta}\ge(n-1)Kg_\delta$.

The outward normal at $t=R$ is $\partial_t$, and the second
fundamental form is
$(F_\delta'(R)/F_\delta(R))g_{\partial M_\delta}=0$.
Furthermore, any curve from radial level $t$ to the boundary has
length at least $R-t$, since its speed is at least the absolute
radial speed. A radial segment realizes this length. Therefore
\[
\operatorname{dist}(x,\partial M_\delta)=R-t(x),
\qquad \operatorname{InRad}(M_\delta)=R.
\]
Each $M_\delta$ consequently belongs to $\mathcal C(n,K,R)$.

\paragraph{The Rayleigh limit.}
Let $\phi$ be the positive first eigenfunction of
\eqref{eq:kasue-model}, and set $u(t)=\phi(R-t)$.
This defines a Lipschitz function on $M_\delta$ with zero boundary
trace, hence an element of $H_0^1(M_\delta)$.
The Rayleigh principle gives
\[
\lambda_1^D(M_\delta)
\le
\frac{\int_0^R |u'(t)|^2F_\delta(t)^{n-1}\,dt}
     {\int_0^R |u(t)|^2F_\delta(t)^{n-1}\,dt}.
\]
The common spherical volume factor cancels. Define
\[
\omega_\delta(t)=\left(\frac{F_\delta(t)}{c_\delta}\right)^{n-1},
\qquad \omega(t)=h(t)^{n-1}.
\]
The monotonicity of $F_\delta/h$ yields
$0\le\omega_\delta\le\omega\le1$, with equality
$\omega_\delta=\omega$ on $[\delta,R]$. In particular,
\[
\|\omega_\delta-\omega\|_{L^1(0,R)}\le\delta\longrightarrow0.
\]
Divide the Rayleigh numerator and denominator by
$c_\delta^{n-1}$. Since $u$ and $u'$ are bounded, the resulting
quotient converges to
\[
\frac{\int_0^R |u'|^2\omega\,dt}{\int_0^R |u|^2\omega\,dt}
=\frac{\int_0^R |\phi'(r)|^2\cos^{n-1}(kr)\,dr}
       {\int_0^R |\phi(r)|^2\cos^{n-1}(kr)\,dr}
=\mu(n,K,R).
\]
The limiting denominator is positive. Kasue's lower bound
\eqref{eq:kasue-comparison} now gives
\[
\mu(n,K,R)\le\lambda_1^D(M_\delta)\longrightarrow\mu(n,K,R),
\]
which proves the infimum identity in the regular range.
The balls collapse: $F_\delta''<0$ and $F_\delta'(0)=1$ imply
$F_\delta(\delta)\le\delta$, so
$c_\delta\le\delta/h(\delta)\to0$ and $F_\delta\le c_\delta$.

Finally, Kasue's equality classification
\cite[Theorem~2.1 and Definition~1.2]{Kasue} permits equality for
$K>0$ and boundary parameter $A=0$ only for the round hemisphere,
whose inradius is $\pi/(2\sqrt K)$. Thus the infimum is not attained
in the regular range. At $R=\pi/(2\sqrt K)$, the hemisphere has
first Dirichlet eigenvalue $nK$, equal to the weighted model eigenvalue
in Lemma~\ref{lem:endpoint}; the comparison bound is therefore attained.
