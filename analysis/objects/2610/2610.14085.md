<!-- definition-3: body_md -->
\label{def:DG}
    For $r > 1$, let $\mu_{r}$ denote the ``punctured'' discrete Gaussian supported on $\mathbb{Z} \setminus \{0\}$ having mean zero and variance $r^2$. For a uniquely determined $\sigma := \sigma(r)$ and $Z:=Z(r)$, its probability mass function is
    \begin{equation}
        \mu_{r}(x) := \frac{1}{Z}\exp\left(-\frac{x^2}{2\sigma^2}\right),
        \qquad x \in \mathbb{Z} \setminus \{0\}.
    \end{equation}
    Equivalently, $\mu_r$ is the unique maximum-entropy distribution on
    $\mathbb{Z}\setminus\{0\}$ with mean zero and variance $r^2$. We also set
    $\mu_1$ as the uniform distribution on $\{-1,1\}$.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
The \textit{annealed free energy} function is given by
\begin{equation}\label{eq:F1-def}
    F_1(\kappa,\alpha,r):=H(\mu_r)+\alpha\log p(\kappa/r).
\end{equation}
The \textit{first-moment prediction for the satisfiability threshold} is
\begin{equation}\label{eq:capacity definition}
        \alpha_c(\kappa)
        :=\sup_{r\ge1}\frac{H(\mu_r)}{-\log p(\kappa/r)}.
    \end{equation}
Given $\kappa > 0$ and $\alpha > 1$, the \emph{maximizing radius} is $$r_*(\kappa,\alpha)=\arg\max_{r\ge1}F_1(\kappa,\alpha,r).$$

In the    computer science literature, especially in lattice cryptography, the discrete Gaussian is often parameterized by $\sigma^2$, the parameter appearing in the exponential. <!-- remark-3: body_md -->
A reasonable, but absolutely wrong, low-dimensional intuition is that $N_n(r)$ scales proportionally to the Euclidean volume of the $n$-ball of radius $r\sqrt{n}$ \cite{mazo1990lattice}. The above computations establish that this is off by an exponential factor. The intuition is that the discrepancy between the volume and number of integer points in a body is largely due to boundary effects, whereas almost all of the volume of a high-dimensional sphere lies near its boundary. However, as $r$ diverges, the entropy of $\mu_r$ approaches $\frac12\log(2\pi e r^2)$, the differential entropy of $N(0,r^2)$. For fixed $r$, this also equals the limiting normalized log-volume of the ball, $n^{-1}\log \mathrm{Vol} \,B_n(0,r\sqrt{n})$. Hence, as $r \to 1$ and $r \to \infty$, the integer geometry of our model approaches the discrete cube and continuum respectively, and is non-trivial everywhere in between.

\[

    \qquad
    R(t)=-\frac{d}{dt}\log Z(t).
\]
Then $R(t)=r^2$, $R'(t)=-\mathrm{Var}(X^2)$, and the desired inequality is

    \qquad

\]

\]

$u_1x^2+u_2xy+u_3y^2$ would be constant there. Fixing $y=1$ and varying
$x$ gives $u_1=u_2=0$, after which varying $y$ gives $u_3=0$, a
contradiction. 

Since
\[
    \log\mu^{(2)}_{r,\hat r,\theta}(x,y)
      =\mathbf a(\theta)\cdot T(x,y)-\Psi(\mathbf a(\theta)),
\]
we have
\begin{equation}\label{eq:entropy-Legendre}
    h(\theta)
      =\Psi(\mathbf a(\theta))
         -\mathbf a(\theta)\cdot m(\theta)
      =\inf_{\mathbf u:\,\Psi(\mathbf u)<\infty}
         \{\Psi(\mathbf u)-\mathbf u\cdot m(\theta)\}.
\end{equation}
The infimum is uniquely attained at $\mathbf u=\mathbf a(\theta)$ since
the gradient of the expression being minimized is
$\nabla\Psi(\mathbf u)-m(\theta)$ and its Hessian is positive definite. Differentiating the attained infimum in \eqref{eq:entropy-Legendre} and using
$\nabla\Psi(\mathbf a)=m(\theta)$, we obtain
\[
\begin{aligned}
    h'(\theta)
      &=\nabla\Psi(\mathbf a)\cdot\mathbf a'
        -\mathbf a'\cdot m(\theta)
        -\mathbf a\cdot m'(\theta) =-\mathbf a\cdot m'(\theta)
       =-b(\theta)r\hat r,
\end{aligned}
\]
which proves \eqref{eq:H-prime}. Differentiating
$m(\theta)=\nabla\Psi(\mathbf a(\theta))$ gives
\[
    m'(\theta)=\Gamma_{r,\hat r,\theta}\mathbf a'(\theta).
\]
Since $m'(\theta)=(0,r\hat r,0)$ is constant,
\[
    h''(\theta)
      =-\mathbf a'(\theta)\cdot m'(\theta)
      =-m'(\theta)^\top
         \Gamma_{r,\hat r,\theta}^{-1}m'(\theta)<0.
\]
This proves that the entropy is strictly concave. Moreover, as $\mu_{r,\hat{r},0}^{(2)}= \mu_r\otimes\mu_{\hat r}$, independence yields \eqref{eq:H''}:
\[
    h''(0)
      =-(r\hat r)^2
        \bigl(\Gamma_{r,\hat r,0}^{-1}\bigr)_{22}
      = - (r\hat{r})^2 \, \mathbb{E}_{\unskip\space}\left[X^2Y^2\right]^{-1} 
      =-1. \qedhere
\]

<!-- lemma-9: body_md -->
\label{lem:low-rank-cycle-resampling}
Fix $d\le2$. Let $U\in\mathbb{R}^{n\times d}$ satisfy $U^\top U=I_d$ and
$\mu(U)\le n^{-1/2}$. Let $G_j^\perp$, $j\in[m]$, be iid
$N(0,I_n-UU^\top)$ vectors. Independently, let
$\xi_j^{(0)}\sim N(0,I_d)$ and let $\xi_j^{(1)}$ be iid centered random
vectors such that, for some $K<\infty$,
\[
    \sup_{\|v\|_2=1}\mathbb{E}_{\unskip\space}\left[|\langle v,\xi_j^{(1)}\rangle|^4\right]\le K.
\]
Set
\[
    M_j^{(b)}:=G_j^\perp+U\xi_j^{(b)},
    \qquad
    \Sigma:=\mathbb{E}_{\unskip\space}\left[\xi_j^{(1)}(\xi_j^{(1)})^\top\right],
    \qquad
    D:=\sqrt{m/n}\,(\Sigma-I_d).
\]
There is a constant $C$, depending only on $K$ and the compact range of
$m/n$, such that, uniformly for
$1\le k\le\lambda\log n$,
\begin{align}
    \mathbb{E}_{\unskip\space}\left[C_k(m,M^{(1)})-C_k(m,M^{(0)})\right]
    &=\mathrm{tr}(D^k)
      +O\left(k^2\left(\mu(U)+m^{-1}\right)\|D\|_{\mathrm{op}}^k\right),
        \label{eq:low rank mean}\\
    \mathrm{Var}\left(C_k(m,M^{(1)})-C_k(m,M^{(0)})\right)
    &\le C^k k^C\left(\mu(U)+m^{-1}\right)
      \mathrm{Var}(C_k(m,M^{(0)})).
        \label{eq:low rank variance}
\end{align}
<!-- end lemma-9 -->
<!-- proof-15: body_md -->
Write $u_i:=U^\top e_i$ and $Q:=\Sigma-I_d$. Since
$M^{(0)}$ is an iid standard Gaussian matrix, its cycle counts are
centered. For $k=1$, direct calculation gives
\[
    \mathbb{E}_{\unskip\space}\left[C_1(m,M^{(1)})-C_1(m,M^{(0)})\right]
    =\sqrt{m/n}\,\mathrm{tr} Q=\mathrm{tr} D.
\]
For $k\ge2$, the row indices in a cycle are distinct, and the two column
indices incident to each row are distinct. Row independence therefore
gives
\begin{align*}
    \mathbb{E}_{\unskip\space}\left[C_k(m,M^{(1)})\right]
    &=\frac{(m)_k}{(mn)^{k/2}}
      \sum_{i_1,\ldots,i_k\ \mathrm{distinct}}
      \prod_{\ell=1}^k u_{i_\ell}^\top Q u_{i_{\ell+1}}.
\end{align*}
Without the distinctness restriction, the sum is $\mathrm{tr}(Q^k)$. To see
the cost of the restriction, fix $1\le a<b\le k$ and impose
$i_a=i_b$. Summing successively over the remaining column indices along
the two arcs from $a$ to $b$ gives
\[
    \sum_{i=1}^n
       (u_i^\top Q^{b-a}u_i)
       (u_i^\top Q^{k-b+a}u_i).
\]
The absolute value of this expression is at most
\[
    \|Q\|_{\mathrm{op}}^k\sum_{i=1}^n\|u_i\|_2^4
    \le d\,\mu(U)\|Q\|_{\mathrm{op}}^k.
\]
There are $O(k^2)$ possible equalities among the column indices.
Excluding them and using $(m)_k/m^k=1+O(k^2/m)$ proves
\eqref{eq:low rank mean}.

We next prove \eqref{eq:low rank variance}. Suppose first that $k\ge2$.
For distinct tuples $\mathbf{i}=(i_1,\ldots,i_k)$ and
$\mathbf{j}=(j_1,\ldots,j_k)$, write
\[
    P_b(\mathbf{i},\mathbf{j})
    :=\prod_{\ell=1}^k
       M^{(b)}_{j_\ell,i_\ell}M^{(b)}_{j_\ell,i_{\ell+1}},
    \qquad i_{k+1}:=i_1,
\]
and let
\[
    \overline P(\mathbf{i},\mathbf{j})
    :=P_1(\mathbf{i},\mathbf{j})-P_0(\mathbf{i},\mathbf{j})
      -\mathbb{E}_{\unskip\space}\left[P_1(\mathbf{i},\mathbf{j})-P_0(\mathbf{i},\mathbf{j})\right].
\]
Then, writing $Z_k:=C_k(m,M^{(1)})-C_k(m,M^{(0)})$,
\begin{equation}\label{eq:low rank variance expansion}
    \mathrm{Var}(Z_k)
    =\frac1{(mn)^k}
      \sum_{\mathbf{i},\mathbf{j},\mathbf{i}',\mathbf{j}'}
      \mathbb{E}_{\unskip\space}\left[\overline P(\mathbf{i},\mathbf{j})\overline P(\mathbf{i}',\mathbf{j}')\right],
\end{equation}
where each of the four tuples is restricted to have distinct entries.

We estimate this sum by expanding every matrix entry as
\[
    M^{(b)}_{j,i}=G^\perp_{j,i}+u_i^\top\xi_j^{(b)}.
\]
The term using $G^\perp$ at all $2k$ edges is identical for $b=0$ and
$b=1$, so it cancels from $P_1-P_0$. Since the row indices within each
cycle are distinct, a fixed row occurs at most twice in the product of
the two cycle monomials and hence contributes at most four random
factors. The fourth-moment assumption therefore bounds every moment of
the $\xi$ variables which occurs here by a constant depending only on
$K$.

Only the following elementary Gaussian formulas are needed. If
$Z\sim N(0,P)$, then
\begin{align}
    \mathbb{E}_{\unskip\space}\left[Z_aZ_b\right]&=P_{ab},\label{eq:Gaussian second moment}\\
    \mathbb{E}_{\unskip\space}\left[Z_aZ_bZ_cZ_d\right]
      &=P_{ab}P_{cd}+P_{ac}P_{bd}+P_{ad}P_{bc},
        \label{eq:Gaussian fourth moment}
\end{align}
and a product of an odd number of coordinates has mean zero. In the
present case, independence across rows and
$P=I_n-UU^\top$ give
\begin{equation}\label{eq:projected Gaussian covariance}
    \mathbb{E}_{\unskip\space}\left[G^\perp_{j,i}G^\perp_{j',i'}\right]
    =\mathbbm{1}_{\{j=j'\}}\left(\mathbbm{1}_{\{i=i'\}}-u_i^\top u_{i'}\right).
\end{equation}
Thus \eqref{eq:Gaussian second moment}--\eqref{eq:Gaussian fourth moment},
followed by expanding each covariance in
\eqref{eq:projected Gaussian covariance}, writes every summand in
\eqref{eq:low rank variance expansion} as a sum of at most $C^k$ terms.
Each such term consists of bounded coefficients, factors involving the
vectors $u_i$, and explicit requirements that certain row indices and
column indices in the two cycles be equal.

We now bound one of these terms. Say that an edge of the first cycle is
matched to an edge of the second if the first term in
\eqref{eq:projected Gaussian covariance} is used for their two
$G^\perp$ factors. The indicator in that formula then requires both the
row and the column index of the two edges to be equal. Such a term
cannot match two $G^\perp$ factors within one cycle: their row indices
are either different, or they are the two edges incident to the same
row and their column indices are different. Hence all matched edges are
shared by the two cycles.

Unless all $2k$ edges are matched, the matched edges form disjoint
paths. Indeed, every vertex has degree two in either cycle, so a
connected set of shared edges is either a path or the entire cycle; the
latter is exactly the all-$G^\perp$ term already cancelled above. Let
$s$ be the number of these paths. A path of $r$ edges has $r+1$
vertices. Once the labels in the first cycle are chosen, the equality
requirements prescribe the corresponding $r+1$ alternating row and
column labels in the second cycle.

We now sum all terms having fixed locations and orientations for these
paths. Let $r$ be the number of matched edges, and let $a$ and $b$ be
the numbers of row vertices and column vertices, respectively, lying
in the paths. Since a path has one more vertex than edge,
\begin{equation}\label{eq:path vertices}
    a+b=r+s.
\end{equation}
The row labels have at most $m^{2k-a}$ possible values: without the
paths the two cycles have $2k$ row labels, while the value of each of
the $a$ rows in the second cycle is prescribed by the first.

It remains to sum the column labels. Every occurrence of a matrix entry
which is not part of a matched pair carries one factor $u_i$. After the
$b$ shared column vertices are merged, there are $2k-b$ column
positions. Exactly $r-b$ of them are internal column vertices of the
paths, incident only to matched edges, and hence carry no $u_i$ factor;
every other column position carries two. A position carrying no such
factor has at most $n$ possible labels. A position carrying two is
summed using
\begin{equation}\label{eq:orthonormal row sum}
    \sum_{i=1}^n u_i u_i^\top=I_d.
\end{equation}
The requirement that the columns within either cycle be distinct does
not enlarge this bound: after some labels have been fixed, the relevant
sum has the form
\[
    \sum_{i\notin S}u_i u_i^\top
    =I_d-\sum_{i\in S}u_i u_i^\top,
\]
whose operator norm is at most one. Since $d\le2$ and every row
contributes a moment of at most four bounded random variables, the
remaining finite-dimensional sums contribute at most $C^k$. The full
column sum is therefore at most $C^k n^{r-b}$.

Including the normalization in
\eqref{eq:low rank variance expansion}, the contribution of the fixed
paths is at most
\begin{align}
    \frac{C^k m^{2k-a}n^{r-b}}{(mn)^k}
    &=C^k\left(\frac mn\right)^{k-a}n^{r-a-b}
      \le C^k n^{-s},\label{eq:path contribution bound}
\end{align}
where we used \eqref{eq:path vertices} and the compact range of $m/n$.
This is the entire counting step: the exponent $s$ appears because each
path has one more vertex than edge.

There are at most $C^k$ choices made when expanding the matrix factors
and the Gaussian moments, and at most $k^{4s}$ choices for the endpoints
and orientations of $s$ paths. The total contribution from terms with
$s\ge1$ is consequently at most
\[
    C^k\sum_{s\ge1}
       \left(\frac{k^4}{n}\right)^s.
\]
If $s=0$, the two centered monomials in
\eqref{eq:low rank variance expansion} are independent unless their row
tuples share a value. For each of the $k^2$ possible positions of such
a shared value, choosing the first row tuple leaves at most $m^{k-1}$
choices for the second. The same column summation as above bounds these
terms by
\[
    \frac{C^k k^2m^{2k-1}}{(mn)^k}
    =\frac{C^k k^2}{m}\left(\frac mn\right)^k.
\]
Since $m/n$ remains in a compact subset of $(0,\infty)$ and
$k=O(\log n)$, we obtain the stronger bound
\begin{equation}\label{eq:absolute low rank variance}
    \mathrm{Var}(Z_k)\le \frac{C^k k^C}{m}.
\end{equation}
Finally, a direct calculation gives
\[
    \mathrm{Var}(C_k(m,M^{(0)}))
    =2k\frac{(m)_k(n)_k}{m^kn^k}\ge1

    v:=\mathrm{Var}(X+Y)=(r-\hat r)^2+2r\hat r\epsilon.

\[
    H(Z)\le\frac12\log\left(2\pi e\,\mathrm{Var}(Z)\right)\,.

    H(X,Y)
    &\le H(X)+H(\mathbbm{1}_{\{X+Y\ne0\}})
       +\delta\, H(X+Y\mid X+Y\ne0)\\
    &\le H(X)+H(\mathbbm{1}_{\{X+Y\ne0\}})
       +\frac\delta2\log\left(2\pi e
          \,\frac v\delta\right).
\end{align*}

    \le\min\{F_1(\kappa,\alpha,r),F_1(\kappa,\alpha,\hat r)\}.
\]

\eqref{eq:uniform endpoint probability estimate} proves the lemma.

where the second inequality follows by completing the square and using
$s^2/n=o(1)$. Thus the proof of \eqref{eq:F1 trunc mid} applies without
further change.

On any fixed compact set away from $r_c$, it holds uniformly that
$p(\kappa/r)^s/p_c^s= n^{O(1)}$, while $\mathbb{E}_{\unskip\space}\left[\mathcal{Z}_r\right]$ is already exponentially smaller than $\mathbb{E}_{\unskip\space}\left[\mathcal{Z}_I\right]$. Finally, for large $r$, we have $p(\kappa/r)^s\le (Kr)^{C\log n}$ for some $K<\infty$. Since $\alpha_c>1$, the
summation in \eqref{eq:F1 trunc extreme} still decays exponentially. Hence, arguing as in \cref{thm:truncated 1mm} yields
\eqref{eq:critical radius truncation}.

It therefore suffices to sum over radii satisfying
$\sqrt n\,|r-r_c|\le w_n$. Uniformly over these radii,
\cref{prop:1d-lattice-count} gives
\[
    \mathbb{E}_{\unskip\space}\left[\mathcal{Z}_r\right]
    =\frac{1+o(1)}
      {\sqrt{2\pi n\mathrm{Var}_{X\sim\mu_r}(X^2)}}
      \exp\{nF_1(\kappa,\alpha_c,r)\}
      p(\kappa/r)^s.
\]
Recalling that $F_1(\kappa,\alpha_c,r_c)=0$ and that $r_c$ is the
maximizer of $F_1$, the definition of $\gamma_c$ gives
\[
    F_1(\kappa,\alpha_c,r)
    =-\frac{\gamma_c}{2}(r^2-r_c^2)^2
      +O(|r^2-r_c^2|^3)
    =-\frac{\gamma_c}{2}(r^2-r_c^2)^2+o(n^{-1}).
\]
Moreover, first-order expansions give
\[
    p(\kappa/r)^s
    =p_c^s\exp\{O(|s||r-r_c|)\}
    =(1+o(1))p_c^s, \quad
    \mathrm{Var}_{X\sim\mu_r}(X^2)
    =(1+o(1))\mathrm{Var}_{X\sim\mu_{r_c}}(X^2).
\]
Consequently, 
\begin{align*} \mathbb{E}_{\unskip\space}\left[\mathcal{Z}\right] &=(1+o(1)) \frac{p_c^s} {\sqrt{2\pi n\mathrm{Var}_{X\sim\mu_{r_c}}(X^2)}} \sum_{\substack{r\in\mathcal{R}_n\\ \sqrt n\,|r-r_c|\le\log\log n}} \exp\left\{-\frac{\gamma_c n}{2}(r^2-r_c^2)^2\right\} +o(p_c^s). \\ &= (1+o(1)) \frac{p_c^s} {\sqrt{\gamma_c\mathrm{Var}_{X\sim\mu_{r_c}}(X^2)}} =(C_\kappa+o(1))p_c^s. \end{align*}
Between the first and second lines, we have used that the Gaussian sum converges to $\sqrt{2\pi n/\gamma_c}$.

We prove the claims in \cref{thmStructure} in reverse order, beginning with item \ref{item:lognormal}. We begin by giving the more precise statement of the log-normal limit.

\begin{lemma}\label{lem:lognormal}
Let $\kappa, \alpha, n, m$ be as in \cref{thmStructure}. Let
\[
    B_*:=\frac{\sqrt\alpha}{2}
       \left(\mathbb{E}_{\unskip\space}\left[W^2\mid |W|\le\kappa/r_*\right]-1\right)
\]
and
\[
    W_{\kappa,\alpha}\sim\mathcal N\left(
       \frac14\log(1-4B_*^2),
       -\frac12\log(1-4B_*^2)
       \right),
\]
then
\begin{equation}\label{eq:main lognormal limit}
    \sup_{s\in\mathbb{R}}
    \left|
\mathbb{P}_{}\left[\mathcal{Z}>e^s\mathbb{E}_{\unskip\space}\left[\mathcal{Z}\right]\right]-\mathbb{P}_{}\left[W_{\kappa,\alpha}>s\right]
    \right|=o(1).
\end{equation}
\end{lemma}
\begin{proof} 
Let $\alpha_n=m/n$. Note that \cref{hyp:maxima} is robust to sufficiently small perturbations in $\alpha$: item~(2) and the local expansion at
$\theta=0$ control small overlaps, \cref{lemma:F2-endpoint-uniform}
controls a neighborhood of $\theta=\pm1$, and continuity and compactness
control the remaining values of $\theta$.
Consequently, \cref{hyp:maxima} holds throughout some compact interval
$J\subset(1,\alpha_c(\kappa)]$ containing $\alpha$ in its interior.
For all sufficiently large $n$, we have $\alpha_n\in J$ and
$m\le\alpha_c(\kappa)n-\eta\log n$ for some fixed $\eta>0$, so the results of
\cref{sec:second moment method} apply. Let
$I=I_n(\kappa,\alpha_n)$ and write $\mathcal{Z}^{\text{\tiny{typ}}}=\mathcal{Z}_I^{\text{\tiny{typ}}}$. By
\cref{thm:truncated 1mm,lemma:conditional 2mm},
\begin{equation}\label{eq:full versus typical}
    \frac{\mathcal{Z}-\mathcal{Z}^{\text{\tiny{typ}}}}{\mathcal{Z}^{\text{\tiny{typ}}}}=o_{\mathbb P}(1),
    \qquad
    \frac{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}\right]}{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right]}=1+o(1).
\end{equation}
Indeed, the numerator in the first ratio is at most
$e^{-c(\log\log n)^2}\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right]$ with high probability, while the denominator is lower bounded via
\eqref{eq:conditional 2mm - tail goal}. Hence, with high probability,
\begin{align*}
    \log\frac{\mathcal{Z}}{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}\right]}
    &=
    \log\frac{\mathcal{Z}^{\text{\tiny{typ}}}}{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right]}
    +\log\frac{\mathcal{Z}}{\mathcal{Z}^{\text{\tiny{typ}}}}
    +\log\frac{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right]}{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}\right]} \stackrel{\eqref{eq:full versus typical}}=
    \log\frac{\mathcal{Z}^{\text{\tiny{typ}}}}{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right]}
    +o_{\mathbb P}(1),
\end{align*}
Since $\alpha_n\to\alpha$, continuity
gives $r_*(\kappa,\alpha_n)\to r_*$ and $B_{\kappa/r_*(\kappa,\alpha_n),\alpha_n}\to B_*$.
Thus, the mean and variance of the Gaussian $W_{\alpha_n}$ in
\eqref{eq:conditional 2mm - scaling limit goal} converge to those of
$W_{\kappa,\alpha}$. Since these Gaussian densities are uniformly
bounded, the $o_{\mathbb P}(1)$ perturbation above does not affect the
Kolmogorov distance. This establishes \eqref{eq:main lognormal limit}, and also item \ref{item:lognormal} of \cref{thmStructure}. \end{proof}

We next prove item \ref{item:typical empirical} of \cref{thmStructure}, the assertion about empirical distributions. For fixed
$r\in I$, consider an empirical distribution $\nu$ of a vector in
$\Lambda_n^{\text{\tiny{typ}}}(r)$. 
Writing $\mu_r(k)=e^{a_rk^2+b_r}$, we have
\[
    \sum_k k^2\nu(k)=r^2=\sum_k k^2\mu_r(k).
\]
Hence,
\begin{align*}
    D(\nu\|\mu_r)
    =-H(\nu)-\sum_k\nu(k)(a_rk^2+b_r)&=-H(\nu)-a_rr^2-b_r =H(\mu_r)-H(\nu).
\end{align*}
On the other hand, there are at most $(n+1)^{2\log n}$ possible empirical
distributions $\nu$ since typical vectors $x$ satisfy $\|x\|_\infty\le \log n$ by construction. Moreover, standard counting considerations give that each empirical distribution $\nu$ corresponds to at most $e^{nH(\nu)}$
vectors. Hence, for every $s>0$,
\begin{equation}\label{eq:empirical type bound}
    \left|\left\{x\in\Lambda_n^{\text{\tiny{typ}}}(r):
       D(\nu_x\|\mu_r)\ge s\right\}\right|
    \le \exp\{nH(\mu_r)-ns+O((\log n)^2)\}.
\end{equation}
We remark that this is the main computation in the proof of
Sanov's theorem; however, it is more convenient here to use the explicit bound above rather than quote the full theorem.

Let $\epsilon_n=n^{-1/2}(\log n)^{3}$ and define the set of solutions with ``atypical'' empirical distribution by
\[
    A_0:=\left\{x\in\mathbf{X}_{n,m,\kappa}^{\text{\tiny{typ}}}:
       \|\nu_x-\mu_{r_x}\|_{\mathrm{TV}}>\epsilon_n\right\},
    \qquad r_x:=\|x\|_2/\sqrt n.
\]
Combining this with \eqref{eq:empirical type bound} gives
\begin{align*}
    \mathbb{E}_{\unskip\space}\left[|A_0|\right]
    &\le
    e^{-2n\epsilon_n^2+O((\log n)^2)}
    \sum_{r\in I}e^{nH(\mu_r)}p(\kappa/r)^m.
\end{align*}
Applying the local central limit
theorem to \cref{prop:1d-lattice-count}, and then using \eqref{eq:E[Z] - typical} to restrict to typical vectors, it holds uniformly over $r\in I$, 
\[
    e^{nH(\mu_r)}
    \le C\sqrt n\,|\Lambda_n^{\text{\tiny{typ}}}(r)|.
\]
Hence, as $n\epsilon_n^2 = \omega((\log n)^2)$,
\[
    \mathbb{E}_{\unskip\space}\left[|A_0|\right]
    \le e^{-cn\epsilon_n^2}\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right].
\]
Consequently, Markov's inequality and
\eqref{eq:conditional 2mm - tail goal} give
\begin{align*}
    \mathbb{P}_{}\left[\frac{|A_0|}{\mathcal{Z}^{\text{\tiny{typ}}}}>
             e^{-cn\epsilon_n^2/2}\right]
    &\le
    \mathbb{P}_{}\left[\mathcal{Z}^{\text{\tiny{typ}}}<(\log n)^{-1/2}\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right]\right]
      +(\log n)^{1/2}e^{-cn\epsilon_n^2/2}=o(1).
\end{align*}
That is, with high probability, the proportion of solutions with atypical empirical distribution is vanishing:
\begin{equation}\label{eq:empirical good solutions}
    \frac{|A_0|}{\mathcal{Z}^{\text{\tiny{typ}}}}=o_{\mathbb P}(1).
\end{equation}
Of course, our definition of $A_0$ compares to $\mu_{r_x}$ rather than $\mu_{r_*}$; it remains to correct for this. 
Since $\alpha_n\to\alpha$ and $I$ is centered at
$r_*(\kappa,\alpha_n)$,
\[
    \sup_{r\in I}|r-r_*|
    \le\frac{w_n}{\sqrt n}
       +|r_*(\kappa,\alpha_n)-r_*|
    =o(1).
\]
Continuity of the discrete Gaussian in its variance therefore gives
\[
    \sup_{r\in I}\|\mu_r-\mu_{r_*}\|_{\mathrm{TV}}=o(1).
\]
Thus every $x\in\mathbf{X}_{n,m,\kappa}^{\text{\tiny{typ}}}\setminus A_0$ satisfies
\[
    \|\nu_x-\mu_{r_*}\|_{\mathrm{TV}}
    \le\epsilon_n+
       \sup_{r\in I}\|\mu_r-\mu_{r_*}\|_{\mathrm{TV}}
    =o(1).
\]
Combining the above estimates readily yields
item \ref{item:typical empirical} of \cref{thmStructure}; moreover, we remark that tracking the errors more carefully above leads to the quantitative rate $\|\nu_x - \mu_{r_*}\|_{\mathrm{TV}} \le n^{-1/2}(\log n)^{c}$. \\

We establish item \ref{item:typical orthogonal} next, which asserts the near-orthogonality of typical solutions. Let $P^{\text{\tiny{typ}}}$ be the number of ordered pairs $(x,y)$ of typical
solutions $x,y\in\mathbf{X}_{n,m,\kappa}^{\text{\tiny{typ}}}$ such that
$y\notin\{x,-x\}$ and
$|\langle x,y\rangle|>\sqrt n\log n$. The unweighted estimates in the proof of
\cref{lemma:conditional 2mm}, obtained by removing the weights in the
arguments for
\eqref{eq:2mm-near intermediate}, \eqref{eq:2mm-bulk}, and
\eqref{eq:2mm-extreme off diagonal}, give
\[
    \mathbb{E}_{\unskip\space}\left[P^{\text{\tiny{typ}}}\right]=o\bigl(\mathbb{E}_{\unskip\space}\left[\mathcal{Z}^{\text{\tiny{typ}}}\right]^2\bigr).
\]
It follows from \eqref{eq:conditional 2mm - scaling limit goal} and
Markov's inequality that $P^{\text{\tiny{typ}}}/(\mathcal{Z}^{\text{\tiny{typ}}})^2=o_{\mathbb P}(1)$.
Except for pairs $(x,y)$ with $y\in\{x, -x\}$ (which are negligible in number compared to $\mathcal{Z}^2$), all remaining ordered pairs have at least one endpoint outside the
typical set and hence number at most
$2\mathcal{Z}(\mathcal{Z}-\mathcal{Z}^{\text{\tiny{typ}}})$. Applying \eqref{eq:full versus typical}
yields item~(2).\\

It remains to prove item \ref{item:freezing}, namely freezing in Hamming distance. Let $\delta$ be the constant in \eqref{eq:strong freezing conclusion}; we will prove item \ref{item:freezing} with a modified constant $\delta'$. Since $I$
remains in a compact subset of $(1,\infty)$, there are $c,C>0$ such
that the sub-Gaussian estimate $\mu_r(k)\le Ce^{-ck^2}$ holds uniformly for $r\in I$. Thus, there exists
some fixed $M<\infty$ with
\[
    \sup_{r\in I}
    \mathbb{E}_{\mu_r\unskip\space}\left[X^2\mathbbm{1}_{\{|X|>M\}}\right]\le\frac{\delta^2}{32}.
\]
For all $n$ sufficiently large, we then have for every $x\in\mathbf{X}_{n,m,\kappa}^{\text{\tiny{typ}}}\setminus A_0$,
\begin{align*}
    \frac1n\sum_{i:\,|x_i|>M}x_i^2 &\le \mathbb{E}_{\mu_{r_x}\unskip\space}\left[X^2\mathbbm{1}_{\{|X|>M\}}\right]
      + 2(\log n)^2\epsilon_n \le\frac{\delta^2}{16}\,.
\end{align*}
That is, ``large'' coordinates do not contribute significantly to $\|x\|$. Next, choose $\delta'\in(0,1/2)$ so that
$2M^2\delta'\le\delta^2/16$. If
$x,x'\in\mathbf{X}_{n,m,\kappa}^{\text{\tiny{typ}}}\setminus A_0$ differ in at most
$2\delta' n$ coordinates, decomposing the set $\{i\,:\, x_i \neq x_i'\}$ based on whether $|x_i| > M$ or $|x_i| \le M$ (and similarly for $x_i'$) gives
\begin{align*}
    \|x-x'\|_2^2
    &\le2\sum_{i:x_i\ne x_i'}x_i^2
       +2\sum_{i:x_i\ne x_i'}(x_i')^2 \le4\left(\frac{\delta^2n}{16}+2M^2\delta'n\right)
    \le\frac{\delta^2n}{2}.
\end{align*}
This gives freezing with respect to Hamming distance for pairs of typical solution vectors. It remains to briefly handle freezing in the case that at least one of the solution vectors is atypical. Define
\[
    A:=(\mathbf{X}_{n,m,\kappa}\setminus\mathbf{X}_{n,m,\kappa}^{\text{\tiny{typ}}})\cup A_0,
\]
the set of solutions which are atypical in radius, infinity norm, or empirical distribution. Also define the set of typical solutions violating freezing as
\[
    B := \{x \in \mathbf{X}_{n,m,\kappa}\setminus A\,:\, \exists\,y\in (\mathbf{X}_{n,m,\kappa}\setminus A)\setminus\{x\},~\|x-y\|_2 \le \delta \sqrt{n}\}\,.
\]
By
\eqref{eq:full versus typical} and
\eqref{eq:empirical good solutions}, $|A|$ is negligible; by
\eqref{eq:strong freezing conclusion}, $|B|$ is also negligible. That is,
\[
    \frac{|A|}{\mathcal{Z}}=o_{\mathbb P}(1),
    \qquad
    \frac{|B|}{\mathcal{Z}}=o_{\mathbb P}(1).
\]

A fixed vector $y\in A$ can be within Hamming distance $\delta'n$ of at most one solution outside $A\cup B$.
Indeed, if two distinct solutions $x,x'\notin A\cup B$ were each within Hamming distance $\delta'n$ of $y$, then $x$ and $x'$ would differ in at most $2\delta'n$
coordinates. The preceding bound would give
$\|x-x'\|_2\le\delta\sqrt n$, so the definition of $B$ would imply $x,x'\in B$, which is a contradiction.

Therefore, the total
number of solutions $x \in \mathbf{X}_{n,m,\kappa}$ having at least one (different) solution within Hamming distance
$\delta' n$ is at most
\[
    2|A|+|B|=o_{\mathbb P}(\mathcal{Z}).
\]
This proves item \ref{item:freezing} and completes all parts of the proof of \cref{thmStructure}.

We directly adapt the stopping-time argument of Sah and Sawhney
\cite[Section~2]{ss}, mutatis mutandis. There are some minor running differences due to the present model allowing for slight variation in the norms of the solutions. These are handled using the same type of Lipschitz estimates that have been used throughout the rest of the manuscript. We present the proof for completeness.

By assumption, \cref{hyp:maxima} holds at $\alpha_c$. The same stability argument used in the previous proof gives that \cref{hyp:maxima} also holds
throughout some interval $J=[\alpha_0,\alpha_c]$. We use
\cref{lemma:conditional 2mm} on this interval. Choose a sufficiently small fixed $\eta>0$ and reveal the first
\[
    m_0:=\lfloor\alpha_c n-\eta\log n\rfloor
\]
rows. Define $\mathcal F_{m_0}:=\sigma(G_1,\ldots,G_{m_0})$, and let $s_0 :=m_0-\alpha_c n$,
$I_0 :=I_n(\kappa,m_0/n)$, and
\[
    \mathcal T_0:=\bigcup_{r\in I_0}\Lambda_n^{\text{\tiny{typ}}}(r),
    \qquad
    \mathcal{Z}_0:=|\mathbf{X}_{n,m_0,\kappa}\cap\mathcal T_0|.
\]
Note that $\mathcal{T}_0$, which should be thought of as the surviving typical solutions at the precritical time, is $\mathcal{F}_{m_0}$--measurable. By \cref{lem:critical first moment,thm:truncated 1mm},
\begin{equation}\label{eq:pretime first moment}
    \mathbb{E}_{\unskip\space}\left[\mathcal{Z}_0\right]=(C_\kappa+o(1))\,p_c^{s_0}.
\end{equation}
Choose $\eta$ small enough that
$\eta\log(1/p_c)<1/8$. Then
\cref{lemma:conditional 2mm} implies that with high probability, 
\begin{equation}\label{eq:pretime good event1}
    \frac{\mathcal{Z}_0}{\mathbb{E}_{\unskip\space}\left[\mathcal{Z}_0\right]}
       \in\bigl((\log n)^{-1/2},(\log n)^{1/2}\bigr),
    \qquad
    \mathcal{Z}_0\le n^{1/4},
\end{equation}
and also, simultaneously for all pairs $x,y\in\mathbf{X}_{n,m_0,\kappa}\cap\mathcal T_0$ with
$y\notin\{x,-x\}$,
\begin{equation}\label{eq:pretime good event2}
    |\langle x,y\rangle|\le\sqrt n\log n\,.
\end{equation}



Fix $T<\infty$, let $m=\alpha_c n+t\in\mathbb N$ with $|t|\le T$, and
let $\ell=m-m_0$. Conditioned on $\mathcal F_{m_0}$, choose one
representative from each antipodal pair in
$\mathbf{X}_{n,m_0,\kappa}\cap\mathcal T_0$, and call the resulting set $S$.
Thus $|S|=\mathcal{Z}_0/2$. For a new standard Gaussian row $G_j=\sqrt nA_j$,
the vector
\[
    \left(\langle G_j,x/\|x\|_2\rangle\right)_{x\in S}
\]
is centered Gaussian with covariance matrix $R$ given by
\[
    R_{x,y}=\frac{\langle x,y\rangle}{\|x\|_2\|y\|_2}.
\]
Within the intersection of the high-probability events in \eqref{eq:pretime good event1} and \eqref{eq:pretime good event2}, the covariance $R$ trivializes:
\[
    \|R-I\|_{\mathrm{F}}
    \le C|S|\frac{\log n}{\sqrt n}=o(1).
\]
Consequently, using the formula for the relative entropy of multidimensional Gaussians readily gives
\[
    D\left(
       N(0,R)^{\otimes\ell}\,\middle\|\,N(0,I)^{\otimes\ell}
    \right)
    \le C\ell\,\|R-I\|_{\mathrm F}^2=o(1).
\]
In particular, Pinsker's inequality gives that the joint law for the Gaussian processes corresponding to the inner products of $S$ with the next
$\ell=O(\log n)$ rows is vanishingly close in total variation from the idealized law in
which all these Gaussian variables are independent. This is
the decoupling step in
\cite[Section~2]{ss}.

For $x\in S$ with $\|x\|_2=r_x\sqrt n$, its survival probability under the addition of $\ell$ fresh constraints is
\[
    q_x:=p(\kappa/r_x)^\ell.
\]
Next, the dependence on $x$ is removed from $q$. We claim:
\begin{equation}\label{eq:critical maximizing radius comparison}
    r_*(\kappa,m_0/n)=r_c+O_{\kappa,\eta}((\log n)/n).
\end{equation}
Indeed, let $\Phi(v,\beta):=\partial_vF_1(\kappa,\beta,\sqrt v)$. Then
$\Phi(v_c,\alpha_c)=0$ and
$\partial_v\Phi(v_c,\alpha_c)=-\gamma_c<0$. The implicit function
theorem therefore gives a $C^1$ function
$v_*(\beta)=r_*(\kappa,\beta)^2$ near $\alpha_c$, and hence
\[
    |v_*(\beta)-v_c|\le C|\beta-\alpha_c|,
\]
where $C$ depends only on $\kappa$. Since $v_c>0$ and $\eta$ is fixed,
this yields the claim. Consequently, we obtain that uniformly over $x\in S$,
\begin{equation}\label{eq:varying radii survival}
    q_x=p_c^\ell
       \left(1+O\left(\frac{(\log n)(\log\log n)}{\sqrt n}\right)\right).
\end{equation}
Within the high probability events in \eqref{eq:pretime good event1} and \eqref{eq:pretime good event2},
$\mathcal{Z}_0p_c^\ell=O(\sqrt{\log n})$ almost surely by
\eqref{eq:conditional 2mm - tail goal} and
\eqref{eq:pretime first moment}, while
$p_c^\ell=n^{-\eta\log(1/p_c)+o(1)}=o(1)$. Hence
\eqref{eq:varying radii survival} and the expansion
$\log(1-z)=-z+O(z^2)$ let us estimate the probability that no solutions from $\mathcal{T}_0$ survive after the $m^\text{th}$ constraint:

\begin{equation}\label{eq:conditional extinction}
    \mathbb{P}_{}\left[\mathcal T_0 \cap \mathbf{X}_{n,m,\kappa} = \emptyset
          \mid \mathcal F_{m_0}\right]
    = \exp\left\{-\frac12p_c^\ell\mathcal{Z}_0\right\}+o(1).
\end{equation}
The factor $1/2$ appears because solutions occur in antipodal pairs.

    r^2-1
      \le\sum_{j\ge2}(j^2-1)e^{-(j^2-1)t(r)}
      <7.22\cdot10^{-4}.
\]
Here the $j=2$ term is retained and the remaining positive tail is
bounded geometrically.  In particular,
\begin{equation}\label{eq:near-binary-delta}
    \delta:=r^2-1<7.3\cdot10^{-4}.
\end{equation}
Writing $X=S|X|$, where the sign $S$ is uniform on $\{\pm1\}$ and independent of $|X|$, we have
$\mathbb{E}_{\unskip\space}\left[|X|-1\right]\le\delta/3$.  The geometric distribution maximizes entropy
among nonnegative integer-valued random variables with a fixed mean, and
therefore
\[
    H(\mu_r)
       \le\log2+g(\delta/3)
       <1.0033\log2,
    \qquad
    g(x):=(1+x)\log(1+x)-x\log x.
\]
The assumption $H(\mu_r) + \frac{q(r)}{d(K)} \log p(K) \geq 0$ implies
\begin{equation}\label{eq:near-binary-alpha-upper}
    \alpha\le1.0033\,\alpha_{\rm bin}(K).
\end{equation}

For $(X,Y)\sim\mu^{(2)}_{r,r,\theta}$, let $S=\mathrm{sign}(X)$ and
$T=\mathrm{sign}(Y)$.  Applying data processing to
\eqref{eq:Dr-relative-entropy}, and using
$\mathbb{E}_{\unskip\space}\left[|X||Y|-1\right]\le r^2-1$, gives
\begin{equation}\label{eq:sign-projection}
    D_r(\theta)
      \ge I_{\rm bin}\bigl([\theta-\delta(1-\theta)]_+\bigr).
\end{equation}
Indeed,
\[
    \mathbb{E}_{\unskip\space}\left[ST\right]
      =\mathbb{E}_{\unskip\space}\left[XY\right]-\mathbb{E}_{\unskip\space}\left[ST(|X||Y|-1)\right]
      \ge\theta r^2-\delta
      =\theta-\delta(1-\theta).
\]
For $\theta\ge1/4$,
\begin{equation}\label{eq:binary-information-loss}
    I_{\rm bin}(\theta-\delta(1-\theta))
       \ge.9955\,I_{\rm bin}(\theta).
\end{equation}
To see this, note that
\[
    I_{\rm bin}(\theta)
     -I_{\rm bin}(\theta-\delta(1-\theta))
     \le\delta(1-\theta)\mathrm{arctanh}\theta,
\]
while $I_{\rm bin}(\theta)\ge\theta^2/2$; the resulting ratio is at
most $.0045$.

For $\theta\le1/4$,
\eqref{eq:Gaussian-Hermite-refined},
\eqref{eq:near-binary-alpha-upper}, and
\eqref{eq:Gaussian-tail-two-sided} give
\begin{equation}\label{eq:near-binary-center}
    \alpha G_K(\theta)<.045\theta^2
       <D_r(\theta).
\end{equation}
Here we used
\[
    \alpha_{\rm bin}(K)\frac{1-p(K)}{p(K)}
       \le\frac{\log2}{p(K)}<.7,
    \qquad
    \alpha_{\rm bin}(K)d(K)^2<10^{-12}.
\]

Now, if $\alpha\le.995\alpha_{\rm bin}(K),$
then \eqref{eq:ALS-global}--\eqref{eq:sign-projection} prove the main claim
for $\theta\ge1/4$, and \eqref{eq:near-binary-center} covers the
remaining interval.  

It thus remains to consider $\alpha>.995\alpha_{\rm bin}(K).$
Using
\[
    -\log p(K)\le\frac{1-p(K)}{p(K)} = \frac{2\overline{\Phi}(K)}{p(K)}
       \le\frac{2\phi(K)}{Kp(K)},
\]
we obtain
\[
    \frac{d(K)}{-\log p(K)}\ge K^2.
\]
Recall that stationarity gives $\alpha = q(r)/d(K)$, which yields 
\[
    q(r)>.995(\log2)K^2,
    \qquad
    t(r)>.239K^2.
\]
The defining series therefore yields
\begin{equation}\label{eq:binary-exponential-radius}
    r^2-1\le4e^{-3t(r)}
       \le4e^{-.717K^2}.
\end{equation}

We first treat
\[
    \frac14\le\theta\le
    \theta_U:=1-\frac1{3K^2}.
\]
Let $P_K(\theta)=p^{(2)}(K,K,\theta)$. Integrating
\eqref{eq:energy_derivative}, using the two-sided Gaussian tail bound
\eqref{eq:Gaussian-tail-two-sided}, and evaluating the resulting
one-variable expressions on the intervals
\[
    [1/4,1/2],\quad[1/2,.8],\quad[.8,.95],\quad[.95,.98]
\]
gives
\[
    I_{\rm bin}(\theta)
     -1.0033\,\alpha_{\rm bin}(K)G_K(\theta)>.0315,
    \qquad \frac14\le\theta\le.98.
\]
For $.98\le\theta\le\theta_U$, let
$J_\pm=\mathbb{P}_{}\left[W>K,\ \pm W_\theta>K\right]$. Conditioning on
$(W+W_\theta,W-W_\theta)$ gives
\[
    J_+<.693\overline\Phi(K),\qquad J_-<10^{-3}\overline\Phi(K).
\]
Since
\[
    P_K(\theta)=p(K)^2+2(J_++J_-)-4\overline\Phi(K)^2,
\]
it follows that
$\alpha_{\rm bin}(K)G_K(\theta)<.482$, and hence
\begin{equation}\label{eq:binary-middle-gap}
    I_{\rm bin}(\theta)
     -1.0033\,\alpha_{\rm bin}(K)G_K(\theta)>.0315,
    \qquad \frac14\le\theta\le\theta_U.
\end{equation}
All constants in the above computations are direct
outward-rounded evaluations, uniform for $K\ge25/3$.

The loss from replacing $\theta$ by
$\theta-\delta(1-\theta)$ in \eqref{eq:sign-projection} is at most
\[
    \delta(1-\theta)\mathrm{arctanh}\theta<.0045\log2 < .0032.
\]
Thus \eqref{eq:near-binary-alpha-upper} and
\eqref{eq:binary-middle-gap} prove the profile inequality on
$[1/4,\theta_U]$.

Finally, let $\theta=1-\varepsilon\in(\theta_U,1)$.  By \eqref{eq:binary-exponential-radius},
\[
    r^2-1<10^{-21},
    \qquad
    \mathbb{E}_{\unskip\space}\left[(Y-X)^2\right]=2r^2\varepsilon.
\]
Since $H(X)\le H(\mu_r)$, and since an integer-valued random variable
of second moment $2r^2\epsilon$ satisfies
\[
    H(Y-X)\le H(|Y-X|)+\mathbb{P}_{}\left[Y\ne X\right]\log2
       \le g(2r^2\epsilon)+2r^2\epsilon\log2,
\]
we obtain $H(\mu^{(2)}_{r,r,\theta})-H(\mu_r)\le2.5\varepsilon\log(e/\varepsilon).$
Concavity of $H(\mu_{r, r, \theta}^{(2)})$ (\cref{prop:F2 entropy derivatives}) therefore yields
\begin{equation}\label{eq:near-binary-entropy-derivative}
    \partial_\theta H(\mu^{(2)}_{r,r,\theta})
       \ge-2.5\log\frac e\varepsilon.
\end{equation}
On the other hand, the derivative computation
\eqref{eq:energy_derivative} yields the following direct estimate.
Since $\alpha>.995\alpha_{\rm bin}(K)$,
$P_K(\theta)\le p(K)$, and
$-p(K)\log p(K)\le1-p(K)=2\overline\Phi(K)$,
\[
    \frac{\alpha}{P_K(\theta)}
       \ge\frac{.995\log2}{2\overline\Phi(K)}.
\]
The upper bound in \eqref{eq:Gaussian-tail-two-sided} therefore gives
\[
 \alpha\partial_\theta\log P_K(\theta)
 \ge
 \frac{.995\log2}{\sqrt{2\pi}}\,
 \frac{K}{\sqrt{\varepsilon(2-\varepsilon)}}\,
 \exp\left(-\frac{K^2\varepsilon}{2(2-\varepsilon)}\right)
 \left(1-\exp\left(-\frac{2K^2\theta}{1-\theta^2}\right)\right).
\]
Here $K^2\varepsilon\le1/3$, $K\ge25/3$, and $\theta \geq 0.98$, so $1-\exp(-2K^2\theta/(1-\theta^2)) \geq 1-e^{-100}$.  Hence
\begin{equation}\label{eq:near-binary-energy-derivative}
    \alpha\partial_\theta\log P_K(\theta)
       >.17K\varepsilon^{-1/2}
       >.16K\varepsilon^{-1/2}.
\end{equation}
At $\varepsilon=1/(3K^2)$ and $K=25/3$, the absolute value of the
right-hand side of \eqref{eq:near-binary-entropy-derivative} is less
than $15.85$, while the right-hand side of
\eqref{eq:near-binary-energy-derivative} is greater than $19.24$.
The comparison improves as $K$ grows or $\varepsilon$ decreases.
Thus $F_2'(\theta)>0$ on $(\theta_U,1)$.  Finally,
\[
    F_2(1)=H(\mu_r)+\alpha\log p(K)
       \le2(H(\mu_r)+\alpha\log p(K))=F_2(0),
\]
so the profile inequality $F_2(\theta) < F_2(0)$ follows on this last interval $\theta\in(\theta_U, 1)$.


Finally, we prove that $q(r)d(K) < 1$. If $q(r)\le8$, then by \eqref{eq:energy_derivative} and the computation directly following it, Taylor expansion at $\theta=0$ gives 
\[
    \alpha G_K(\theta)
       =\frac{q(r)d(K)}2\theta^2+O(\theta^4).
\]
Here, we have recalled $\alpha=q(r)/d(K)$, and then used symmetry of $G_K$ to eliminate the odd terms in the Taylor expansion. Hence, combining this expansion with \eqref{eq:big-q-bound} and taking $\theta \to 0$ gives $q(r)d(K)\le2\cdot0.116<1$. If $q(r)>8$, \eqref{eq:near-binary-center} similarly gives $q(r)d(K)\le2\cdot0.045<1$. This concludes the proof.
