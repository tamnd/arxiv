<!-- theorem-1: body_md -->
\label{thm:logbm}
Let $n\ge2$ and $K,L\subset\mathbb R^n$ be centered convex bodies. Then for each $t\in(0,1)$,
\begin{equation}\label{eq:main-bm}
 \big|W[h_K^{1-t}h_L^t]\big|\ge |K|^{1-t}|L|^t
\end{equation}
with equality if and only if there are nonzero complementary linear subspaces $E_1,\ldots,E_r$, centered convex bodies $K_i\subset E_i$ relative to these subspaces, and positive numbers $a_i$ such that
\begin{equation}\label{eq:main-split}
 \mathbb R^n=E_1\oplus\cdots\oplus E_r,\qquad
 K=K_1\oplus\cdots\oplus K_r,\qquad
 L=a_1K_1\oplus\cdots\oplus a_rK_r.
\end{equation}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:logm}
Let $n\ge2$ and let $K,L\subset\mathbb R^n$ be centered convex bodies. Then
\begin{equation}\label{eq:main-m}
 \int_{S^{n-1}}\log\frac{h_L}{h_K}\,dV_K
 \ge \frac{|K|}{n}\log\frac{|L|}{|K|}
\end{equation}
with equality if and only if $K,L$ admit the common direct-sum representation \eqref{eq:main-split}.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:uniqueness}
Let $n\ge2$ and let $K,L\subset\mathbb R^n$ be centered convex bodies. Then $
 \overline V_K=\overline V_L,
$
if and only if they admit \eqref{eq:main-split}.

<!-- theorem-4: body_md -->
\label{thm:lp-bm}
Let $n\ge2$ and $0<p<\infty$. For convex bodies $K,L$ containing the origin in their interiors and every $t\in[0,1]$,
\begin{equation}\label{eq:lp-weighted-BM}
 \Psi(C_{p,t}(K,L))^{p/n}
 \ge(1-t)\Psi(K)^{p/n}+t\Psi(L)^{p/n}.
\end{equation}
In particular, for centered convex bodies $K,L$,
\begin{equation}\label{eq:lp-centered-BM}
 |C_{p,t}(K,L)|^{p/n}
 \ge(1-t)|K|^{p/n}+t|L|^{p/n}.
\end{equation}
For each fixed $t\in(0,1)$, equality in either  inequality holds if and only if $L=cK$ for some $c>0$.

<!-- theorem-5: body_md -->
\label{thm:lp-minkowski}
Let $n\ge2$ and $0<p<\infty$. For convex bodies $K,L$ containing the origin in their interiors,
\begin{equation}\label{eq:lp-weighted-minkowski}
 \widetilde V_p(K,L)\ge\Psi(K)^{1-p/n}\Psi(L)^{p/n}.
\end{equation}
In particular, for centered convex bodies $K,L$,
\begin{equation}\label{eq:lp-centered-minkowski}
 V_p(K,L)\ge |K|^{1-p/n}|L|^{p/n}.
\end{equation}
Equality in either inequality holds if and only if $L=cK$ for some $c>0$. Consequently, for centered $K,L$,
\begin{equation}\label{eq:lp-surface-uniqueness}
 S_p(K,\cdot)=S_p(L,\cdot)
 \quad\Longleftrightarrow\quad
 \begin{cases}
 K=L,&p\ne n,\\
 L=cK\text{ for some }c>0,&p=n.
 \end{cases}
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:reconstruct}
If $Q\in\mathcal K^n_o$ and $S=\operatorname{supp} S_Q$, then
\begin{equation}\label{eq:reconstruct}
 Q=\bigcap_{u\in S}\{x:\langle x,u\rangle\le h_Q(u)\}.
\end{equation}
If $a\in C(S^{n-1})$ is positive and $Q=W[a]$, then
\begin{equation*}
 a=h_Q\quad\text{on }\operatorname{supp} S_Q.
\end{equation*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The equilibrium identity $\int u\,dS_Q(u)=0$ and the fact that $S_Q$ is not concentrated on a proper great subsphere imply that $S$ positively spans $\mathbb R^n$. The right side of \eqref{eq:reconstruct} is therefore a convex body $P\supset Q$. Its support function agrees with $h_Q$ on $S$, so
\[
 V_1(Q,P)=\frac1n\int h_P\,dS_Q=|Q|.
\]
Minkowski's first inequality gives $|Q|\ge |Q|^{(n-1)/n}|P|^{1/n}$. Thus $|P|\le|Q|$, and the inclusion implies $P=Q$.

Fix $x\in\partial Q$.  At least one defining inequality is active at $x$.
Indeed, otherwise the positive continuous function
$u\mapsto a(u)-\langle x,u\rangle$ would have a positive minimum on the
sphere, placing a neighborhood of $x$ inside $Q$.  Thus there is a unit
vector $u$ with $\langle x,u\rangle=a(u)$.  Since $x\in Q$ and $h_Q\leq a$,
this gives $h_Q(u)=a(u)$, and $u$ is an outer normal at $x$.

At $\mathcal H^{n-1}$-almost every boundary point the outer unit normal is
unique.  At such a point the active direction must therefore be
$u=\nu_Q(x)$.  By the definition of surface area measure, $a=h_Q$ holds
$S_Q$-almost everywhere.  The nonnegative difference $a-h_Q$ is
continuous, so it vanishes on $\operatorname{supp} S_Q$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:product}
Suppose $\mathbb R^n=E_1\oplus\cdots\oplus E_r$ is an orthogonal decomposition and $K_i\subset E_i$ contains the origin in its relative interior. Put $d_i=\dim E_i$, $v_i=|K_i|_{d_i}$, and $K=\bigoplus_i K_i$. Regard the intrinsic measure $V^{(d_i)}_{K_i}$ as a measure on $S^{n-1}$ supported on $S(E_i)=E_i\capS^{n-1}$. Then
\begin{align}
 V_K&=\sum_{i=1}^r\frac{d_i}{n}
 \left(\prod_{j\ne i}v_j\right)V^{(d_i)}_{K_i},\notag
\\
 \overline V_K&=\sum_{i=1}^r\frac{d_i}{n}\frac{V^{(d_i)}_{K_i}}{v_i}.
 \label{eq:product-normalized}
\end{align}
For positive numbers $a_i$ and $L=\bigoplus_i a_iK_i$,
\begin{equation*}
 V_L=\left(\prod_i a_i^{d_i}\right)V_K,
 \qquad \overline V_L=\overline V_K.
\end{equation*}
These scaling identities hold for arbitrary complementary subspaces. In that case the normal subspaces are
\begin{equation}\label{eq:normal-subspaces}
 F_i=\left(\bigoplus_{j\ne i}E_j\right)^\perp,

\label{lem:tilt}
The minimizer defining $\xi_C$ exists and is unique. It is characterized by
\begin{equation}\label{eq:tilt-centering}
 \int_C xe^{-\langle\xi_C,x\rangle}\,dx=0.
\end{equation}

\begin{equation}\label{eq:tilt-volume}
 \mathcal P(C)\le\log|C|,

\end{equation}

 \mathcal P(aC)=\mathcal P(C)+n\log a\qquad(a>0).
\end{equation*}

\[
 \nabla_\xi\log Z_C(\xi)=-\mathbb E_\xi x,
 \qquad D^2_\xi\log Z_C(\xi)=\operatorname{Cov}_\xi(x)>0.
\]
Strict positivity follows from the nonempty interior of $C$. If $rB\subset C$ and $e=\xi/|\xi|$, then on $B(-re/2,r/4)$ one has $-\langle\xi,x\rangle\ge r|\xi|/4$.  <\int_L e^{-\xi_L\cdot x}\,dx=\Psi(L).
\]
The strict inequality holds because $L\setminus K$ has positive Lebesgue measure and the weight is positive. Taking logarithms proves strict monotonicity of $\mathcal P$. A change of variables proves the scaling identity, equivalently $\Psi(aC)=a^n\Psi(C)$.

\label{lem:H-variation}
For every $C\in\mathcal K^n_o$ and $f\in C(S^{n-1})$, the two-sided directional derivative exists and satisfies
\begin{equation}\label{eq:H-variation}
 D\mathcal H(\log h_C)[f]=n\int_{S^{n-1}}f\,d\omega_C.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
For fixed $\xi$, the weighted Aleksandrov formula gives
\[
 \left.\frac{d}{dt}\right|_{t=0}
 \int_{W[h_Ce^{tf}]}e^{-\langle\xi,x\rangle}\,dx
 =\int_{\partial C}f(\nu_C)h_C(\nu_C)e^{-\langle\xi,x\rangle}
 \,d\mathcal H^{n-1}(x).
\]
The multiplicative perturbation differs from $h_C+th_Cf$ by $O(t^2)$ uniformly. A dilation sandwich changes the corresponding weighted volumes by $O(t^2)$, so the additive first-variation formula applies.

Put $C_t=W[h_Ce^{tf}]$, $\xi_t=\xi_{C_t}$, and $J(t,\xi)=\log\int_{C_t}e^{-\langle\xi,x\rangle}\,dx$. Lemma~\ref{lem:tilt} gives $\xi_t\to\xi_C$. \[
 \big|[I(t,\xi_t)-I(0,\xi_t)]-[I(t,\xi_C)-I(0,\xi_C)]\big|
 \le |C_t\mathbin\triangle C|\,
       \sup_x|e^{-\xi_t\cdot x}-e^{-\xi_C\cdot x}|=o(|t|),
\]
where the supremum is over that fixed ball. The increments are $O(|t|)$ and $I(0,\xi_t)\to I(0,\xi_C)>0$. Expanding the logarithm therefore shows that the logarithmic difference quotients also have the same limit for the two parameters. Minimality gives
\begin{align*}
 J(t,\xi_t)-J(0,\xi_t)
 &\le\mathcal H(\log h_C+tf)-\mathcal H(\log h_C)\\
 &\le J(t,\xi_C)-J(0,\xi_C).
\end{align*}
After division by $t$, with the inequalities reversed if $t<0$, the two bounds have the common limit in \eqref{eq:H-variation}.

<!-- lemma-5: body_md -->
\label{lem:moment}
Let $V\in C^\infty(\mathbb R^n)$ satisfy $D^2V\succeq\varepsilon I$ for some $\varepsilon>0$, and suppose
\[
 Z=\int_{\mathbb R^n}e^{-V(x)}\,dx,\qquad
 d\mu(x)=Z^{-1}e^{-V(x)}\,dx,\qquad \int x\,d\mu(x)=0.
\]
There is a strictly convex $\varphi\in C^\infty(\mathbb R^n)$ such that
\begin{equation*}
 \int e^{-\varphi(z)}\,dz=1,\qquad
 (\nabla\varphi)_\#(e^{-\varphi(z)}\,dz)=\mu,\qquad
 0\prec D^2\varphi\preceq\frac4\varepsilon I.
\end{equation*}
The gradient $\nabla\varphi:\mathbb R^n\to\mathbb R^n$ is a diffeomorphism, $\varphi(z)\ge c|z|-b$ for some $c>0$, and
\begin{equation*}
 \log\det D^2\varphi(z)=V(\nabla\varphi(z))+\log Z-\varphi(z).
\end{equation*}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Uniform convexity gives Gaussian decay after an affine term, so $Z$ and all polynomial moments are finite. Set
\[
 Z_R=\int_{B_R}e^{-V(x)}\,dx,\qquad
 m_R=Z_R^{-1}\int_{B_R}xe^{-V(x)}\,dx.
\]
Then $m_R\to0$. Translate the truncated probability by $-m_R$:
\[
 d\mu_R(x)=Z_R^{-1}\mathbf 1_{B_R-m_R}(x)e^{-V(x+m_R)}\,dx.
\]
Its centroid is zero. The compact-target theorem and the Hessian estimate give a potential $\varphi_R$ with
\[
 (\nabla\varphi_R)_\#(e^{-\varphi_R}\,dz)=\mu_R,\quad
 \int e^{-\varphi_R}=1,\quad
 \nabla\varphi_R(\mathbb R^n)=B_R-m_R,\quad
 0\prec D^2\varphi_R\preceq\Lambda I,
 \qquad\Lambda=4/\varepsilon.
\]
Translate the source so that the minimum is at zero, and put $a_R=\varphi_R(0)$ and $g_R=\varphi_R-a_R$. Supporting planes with slopes surrounding zero show that each $\varphi_R$ grows at least linearly. Thus integration of $\operatorname{div}(ze^{-\varphi_R})$, justified by expanding compact cutoffs, gives
\begin{equation*}
 \int z\cdot\nabla\varphi_R(z)e^{-\varphi_R(z)}\,dz=n.
\end{equation*}
For $x=\nabla\varphi_R(z)$, convexity and the minimum normalization imply $x\cdot z\ge g_R(z)\ge0$ and
\[
 g_R(y)\ge g_R(z)+x\cdot y-x\cdot z.
\]
Together with $g_R(y)\ge0$, this yields $g_R(y)\ge(x\cdot y)_+-x\cdot z$. Integrating and using the preceding radial identity, we obtain
\begin{equation}\label{eq:source-coercive}
 g_R(y)\ge\int(x\cdot y)_+\,d\mu_R(x)-n\ge c|y|-n.
\end{equation}
The constant $c>0$ is independent of large $R$: a fixed ball about zero lies in $B_R-m_R$, and the densities there have a common positive lower bound. The Hessian estimate gives $g_R(y)\le\Lambda|y|^2/2$ and $|\nabla\varphi_R(y)|\le\Lambda|y|$. Normalization now implies
\begin{equation*}
 (2\pi/\Lambda)^{n/2}\le e^{a_R}=\int e^{-g_R(y)}\,dy
 \le e^n\int e^{-c|y|}\,dy.
\end{equation*}
Hence $a_R$ is bounded and the source densities have a common integrable exponential majorant.

The equation for the compact targets is
\[
 \log\det D^2\varphi_R=V(\nabla\varphi_R+m_R)+\log Z_R-\varphi_R.
\]
On each fixed source ball its right side and first derivatives are uniformly bounded. The determinant is bounded below there, and the Hessian upper bound gives a positive lower bound for each eigenvalue. Interior Evans--Krylov and Schauder estimates therefore give bounds of every order on smaller balls. One may apply the uniformly elliptic form of these estimates by extending the logarithm from the fixed eigenvalue interval to a smooth concave uniformly elliptic spectral function. A diagonal subsequence converges locally smoothly to $\varphi$, with the asserted moment equation, the Hessian bounds, and the linear lower bound. Dominated convergence gives $\int e^{-\varphi}=1$. For every bounded continuous $F$,
\[
 \int F(\nabla\varphi)e^{-\varphi}
 =\lim_R\int F(\nabla\varphi_R)e^{-\varphi_R}
 =\lim_R\int F\,d\mu_R=\int F\,d\mu.
\]
This proves the pushforward assertion.

The positive Hessian makes the gradient locally invertible and injective. Its image is dense because $\mu$ has positive density everywhere. Fix $x_0\in\mathbb R^n$ and choose finitely many image points $y_j=\nabla\varphi(z_j)$ whose convex hull contains a ball $B_r(x_0)$. Their supporting planes give
\[
 \varphi(z)-x_0\cdot z
 \ge\max_j\{(y_j-x_0)\cdot z+\varphi(z_j)-y_j\cdot z_j\}
 \ge r|z|-b_{x_0}.
\]
The left side attains a minimum, where $\nabla\varphi=x_0$. Thus the gradient is onto, and its local inverses form a global smooth inverse.

<!-- lemma-6: body_md -->
\label{lem:density}
Suppose $d\mu=\rho\,dx$ has positive smooth density and finite second moment, $\tau$ is smooth with $0\prec\tau\preceq\Lambda I$, and $A=x\cdot\nabla-\tau:D^2$ satisfies compact weighted symmetry and \eqref{eq:hessian-identity}. Then
\begin{equation}\label{eq:dense-range}
 \overline{\{A(A-I)v:v\in C_c^\infty(\mathbb R^n)\}}^{\,L^2(\mu)}
 =\operatorname{span}\{1,x_1,\ldots,x_n\}^{\perp}.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Write $\Gamma(f,h)=\nabla f\cdot\tau\nabla h$ and $\Gamma(f)=\Gamma(f,f)$. Since $A1=0$ and $Ax_j=x_j$, compact symmetry makes every element of the displayed range orthogonal to affine functions. Conversely, suppose $g\in L^2(\mu)$ annihilates that range. With $\mathscr L=A(A-I)$, formal symmetry gives
\[
 \mathscr L^t(\rho g)=\rho\mathscr Lg=0
\]
in distributions. Its principal symbol is $(\zeta\cdot\tau(x)\zeta)^2$, which is locally uniformly positive. To apply the Lebesgue-self-adjoint regularity statement in Lemma~\ref{lem:finite-elliptic}, set
\[
 \widehat{\mathscr L}=\rho^{1/2}\mathscr L\rho^{-1/2},\qquad
 \widehat g=\rho^{1/2}g.
\]
Weighted symmetry makes $\widehat{\mathscr L}$ formally self-adjoint for Lebesgue measure. It has smooth coefficients and the same principal symbol, and $\widehat{\mathscr L}\widehat g=0$ distributionally. The lemma gives $\widehat g\in C^4_\mathrm{loc}$, hence $g\in C^4_\mathrm{loc}$ because $\rho$ is smooth and positive. Compact $C^4$ tests can be used by approximation.

Choose $0\le\eta_r\le1$, equal to one on $B_r$ and supported in $B_{2r}$, with $|\nabla\eta_r|\le C/r$ and $|D^2\eta_r|\le C/r^2$. For $r\ge1$, the Hessian cap and the linear drift give
\begin{equation*}
 \Gamma(\eta_r)^{1/2}\le B/r,\qquad |A\eta_r|\le K,
\end{equation*}
where $B,K$ do not depend on $r$. All derivatives vanish off $B_{2r}\setminus B_r$. Put $v=Ag$, initially as a local function, and set
\[
 G=\|g\|_2,\qquad X_r=\|\eta_r^2v\|_2,\qquad
 Y_r^2=\mathbb E[\eta_r^2\Gamma(g)].
\]
Testing $\mathscr Lg=0$ against $\eta_r^4g$ gives $\mathbb E[v(A-I)(\eta_r^4g)]=0$. The product rule $A(ab)=aAb+bAa-2\Gamma(a,b)$ yields
\begin{align*}
 (A-I)(\eta_r^4g)
 &=\eta_r^4v+g\{4\eta_r^3A\eta_r-12\eta_r^2\Gamma(\eta_r)-\eta_r^4\}
   -8\eta_r^3\Gamma(\eta_r,g).
\end{align*}
Cauchy--Schwarz and the cutoff bounds, together with integration by parts in $\mathbb E[\eta_r^2gAg]$, imply
\begin{equation*}
 X_r\le aG+\frac{8B}{r}Y_r,\qquad
 Y_r^2\le GX_r+\frac{2B}{r}GY_r,
 \qquad a=1+4K+12B^2.
\end{equation*}
The first estimate is immediate when $X_r=0$; otherwise divide by $X_r$. These estimates give $Y_r^2\le aG^2+10BGY_r$, hence uniform bounds on $X_r,Y_r$. Exhaustion by balls proves $v\in L^2(\mu)$. Now put $w=g-v$. The equation gives
\[
 Aw=0,\qquad Av=v,\qquad w,v\in L^2(\mu).
\]
Testing $Aw=0$ with $\eta_r^2w$ gives
\[
 \big(\mathbb E[\eta_r^2\Gamma(w)]\big)^{1/2}\le\frac{2B}{r}\|w\|_2.
\]
Let $r\to\infty$ on any fixed ball. Positivity of $\tau$ shows that $w$ is constant. Similarly, with $Z_r^2=\mathbb E[\eta_r^2\Gamma(v)]$, testing $Av=v$ gives
\[
 Z_r^2\le\|v\|_2^2+\frac{2B}{r}\|v\|_2Z_r.
\]
Thus $\mathbb E\Gamma(v)<\infty$. For $f_r=\eta_rv$, the product rule gives
\[
 Af_r=\eta_rv+vA\eta_r-2\Gamma(\eta_r,v),\qquad
 f_r\to v,\quad Af_r\to v\quad\hbox{in }L^2(\mu).
\]
The second term has norm at most $K\|v\mathbf 1_{B_{2r}\setminus B_r}\|_2\to0$, and the last has norm at most $2Br^{-1}(\mathbb E\Gamma(v))^{1/2}$. Apply \eqref{eq:hessian-identity}, valid for compact $C^2$ tests by approximation:
\[
 \mathbb E\big\|\tau^{1/2}D^2f_r\tau^{1/2}\big\|_\mathrm{HS}^2
 =\|Af_r\|_2^2-\mathbb E[f_rAf_r]\longrightarrow0.
\]
On each fixed ball $f_r=v$ for large $r$, so $D^2v=0$. Hence $v$ is affine, and $Av=v$ eliminates its constant term. Therefore $g=w+v$ is affine. Taking orthogonal complements proves \eqref{eq:dense-range}.

\label{lem:finite-elliptic}
Let $\Omega\subset\mathbb R^n$ be open and let $\mathscr L$ be a scalar fourth-order operator, formally self-adjoint for Lebesgue measure, with coefficients in $C^{2n+14}_\mathrm{loc}(\Omega)$. Suppose its principal symbol is positive and locally uniformly elliptic. If $w\in L^2_\mathrm{loc}(\Omega)$ and $\mathscr Lw=0$ distributionally, then
\[
 w\in H^{n+8}_\mathrm{loc}(\Omega)\subset C^4_\mathrm{loc}(\Omega).
\]
<!-- end lemma-7 -->

Localize in a coordinate patch and place it inside a flat torus. With a smooth cutoff $\theta$ supported in the patch, extend the operator by
\[
 \widetilde{\mathscr L}=\theta\mathscr L\theta+(1-\theta)\Delta^2(1-\theta).
\]
The first term is extended by zero outside the patch. Both terms are formally self-adjoint. The principal symbol is $\theta^2p_4+(1-\theta)^2|\zeta|^4$, hence uniformly positive. The coefficients remain $C^N$, where $N=2n+14$, and the extension agrees with $\mathscr L$ where $\theta=1$.

On the torus of period $2\pi$, write $p_4$ for the extended principal symbol and put
\[
 b(x,k)=p_4(x,k)^{-1}\ (k\ne0),\quad b(x,0)=0,\qquad
 Qv(x)=\sum_{k\in\mathbb Z^n}b(x,k)\widehat v(k)e^{ik\cdot x}.
\]
Uniform ellipticity gives $|\partial_x^\alpha b(x,k)|\le C_\alpha\langle k\rangle^{-4}$ for $|\alpha|\le N$, where $\langle k\rangle=(1+|k|^2)^{1/2}$. The finite product rule gives
\[
 \widetilde{\mathscr L}Q=I+R,
 \qquad Rv=\sum_k r(x,k)\widehat v(k)e^{ik\cdot x},
\]
with $|\partial_x^\alpha r(x,k)|\le C_\alpha\langle k\rangle^{-1}$ for $|\alpha|\le N-4$; at $k=0$ set $r(x,0)=-1$. Let $d=N-4=2n+10$. Taking Fourier coefficients in $x$ and integrating by parts with $(1-\Delta_x)^{d/2}$ yields
\[
 |\widehat b(m,k)|\le C\langle m\rangle^{-d}\langle k\rangle^{-4},\qquad
 |\widehat r(m,k)|\le C\langle m\rangle^{-d}\langle k\rangle^{-1}.
\]
The Fourier matrix of $R^*$ therefore satisfies
\[
 |(R^*)_{\ell k}|\le C\langle\ell-k\rangle^{-d}\langle\ell\rangle^{-1}.
\]
Using $\langle\ell\rangle^s/\langle k\rangle^s\le C_s\langle\ell-k\rangle^{|s|}$ and Young's convolution inequality, we obtain
\begin{equation*}
 R^*:H^s\to H^{s+1}\quad(d>n+|s|),\qquad
 Q^*:H^t\to H^{t+4}\quad(d>n+|t|).
\end{equation*}
This also holds for negative exponents. Formal self-adjointness gives
\begin{equation*}
 Q^*\widetilde{\mathscr L}=I+R^*.
\end{equation*}
The identity extends by density to $H^s$ for integer $0\le s\le n+7$. Indeed, the same Fourier-matrix estimate makes $\widetilde{\mathscr L}:H^s\to H^{s-4}$ continuous; the coefficient condition $N>n+|s-4|$ holds throughout this range.

Suppose inductively that $w\in H^s_\mathrm{loc}$. Choose nested compact cutoffs $\chi,\chi_1$ where $\theta=1$, with $\chi_1=1$ near $\operatorname{supp}\chi$. Then
\[
 \widetilde{\mathscr L}(\chi w)=[\mathscr L,\chi](\chi_1w)\in H^{s-3}.
\]
The commutator has order at most three. Applying the adjoint parametrix identity and the preceding Sobolev mapping bounds gives
\[
 \chi w=Q^*[\mathscr L,\chi](\chi_1w)-R^*(\chi w)\in H^{s+1}.
\]
Both mapping conditions hold for $0\le s\le n+7$. Starting from $s=0$ and using successively nested cutoffs proves the claimed Sobolev regularity. Since $n+8>n/2+4$, Sobolev embedding gives $C^4_\mathrm{loc}$.

<!-- lemma-8: body_md -->
\label{lem:stability}
Let $C\in\mathcal K^n_o$, $\xi\in\mathbb R^n$, and
\[
 Z=\int_C e^{-\xi\cdot x}\,dx,\qquad
 d\mu=Z^{-1}\mathbf 1_C e^{-\xi\cdot x}\,dx,\qquad \int x\,d\mu=0.
\]
Let $V_j$ be convex potentials with $D^2V_j\succeq\varepsilon_jI$ for some $\varepsilon_j>0$, and let
\[
 Z_j=\int e^{-V_j},\qquad d\mu_j=Z_j^{-1}e^{-V_j}\,dx.
\]
Assume that every $\mu_j$ is centered and
\begin{equation*}
 W_1(\mu_j,\mu)\to0,\qquad Z_j\to Z,\qquad
 V_j\to(x\mapsto\xi\cdot x)\quad\hbox{locally in every fixed }C^k(\operatorname{int} C).
\end{equation*}
For finite-regularity potentials, their differentiability orders are understood to tend to infinity. Assume that their probability-normalized moment potentials $\varphi_j$ have positive definite Hessians and gradient diffeomorphisms on $\mathbb R^n$. After source translations, for every fixed finite $k$,
\begin{equation*}
 \varphi_j\to\varphi\quad\hbox{in }C^k_\mathrm{loc}(\mathbb R^n),
\end{equation*}
where $\varphi$ is the smooth strictly convex moment potential of $\mu$ and
\begin{equation*}
 \det D^2\varphi=Z\exp(-\varphi+\xi\cdot\nabla\varphi),
 \qquad\nabla\varphi:\mathbb R^n\longrightarrow\operatorname{int} C
 \quad\hbox{is a diffeomorphism}.
\end{equation*}
Consequently,
\begin{equation*}
 D^2\varphi_j\circ(\nabla\varphi_j)^{-1}
 \longrightarrow D^2\varphi\circ(\nabla\varphi)^{-1}
 \quad\hbox{locally in every fixed }C^k(\operatorname{int} C).
\end{equation*}
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
The compact-target theorem \cite[Theorem 1.1]{BB13}, applied to the positive smooth density $Z^{-1}e^{-\xi\cdot x}$ on $C$, gives the stated moment equation and uniqueness up to source translation. The required weighted centroid condition is exactly the hypothesis. Integrating through the gradient map verifies $\int e^{-\varphi}=1$. In particular, $e^{-\varphi}$ is a continuous strictly positive log-concave density on all of $\mathbb R^n$, without a regularity assumption on $\partial C$.

The measures $\mu_j,\mu$ are centered probabilities, have finite first moments, and are not supported on hyperplanes. Their moment potentials are essentially continuous. The stability theorem of Bonnet--Rubinstein \cite[Theorem 1.3]{BR26}, with $\mu$ fixed, gives translations $v_j$ such that
\begin{equation*}
 \|e^{-\varphi_j(\,\cdot-v_j)}-e^{-\varphi}\|_{L^1(\mathbb R^n)}\longrightarrow0.
\end{equation*}
If a target equals $\mu$, the same conclusion follows from translation uniqueness. Replace the potentials by these translates. This does not change the target-coordinate moment Hessian.

Convergence of log-concave densities to a continuous positive density is uniform on compact sets \cite[Proposition 2]{CS10}. Taking logarithms after this locally uniform convergence therefore gives local uniform convergence of $\varphi_j$. Convexity upgrades this to local uniform convergence of the gradients. Explicitly, for a compact set $Q$ with a compact neighborhood $Q^+$ and sufficiently small $s>0$, directional difference quotients give
\[
 \sup_Q|\nabla\varphi_j-\nabla\varphi|
 \le C_n\left(\frac{2\|\varphi_j-\varphi\|_{C^0(Q^+)}}s
                 +\omega_{Q^+}(s)\right),
\]
where $\omega_{Q^+}$ is a modulus of continuity of $\nabla\varphi$. Letting first $j\to\infty$ and then $s\downarrow0$ proves
\begin{equation*}
 \varphi_j\to\varphi\quad\hbox{in }C^1_\mathrm{loc}(\mathbb R^n).
\end{equation*}

We obtain uniform interior estimates without using the possibly diverging cap $4/\varepsilon_j$. The moment equation is
\begin{equation}\label{eq:approx-moment-equation}
 \det D^2\varphi_j=Z_j\exp\big(-\varphi_j+V_j(\nabla\varphi_j)\big).
\end{equation}
Fix a source ball $B_{2R}(z_0)$. Its closure has gradient image under $\varphi$ compactly contained in $\operatorname{int} C$. By the local $C^1$ convergence, all its images under $\nabla\varphi_j$, for large $j$, lie in a common compact set $Q'\Subset\operatorname{int} C$. Thus the hypotheses and \eqref{eq:approx-moment-equation} give
\begin{equation*}
 0<a\le\det D^2\varphi_j\le b<\infty
 \quad\hbox{on }B_{2R}(z_0),
\end{equation*}
with uniform bounds on every fixed derivative of $V_j$ on $Q'$.

The strict-convexity moduli can also be chosen uniformly. On the closure of this source ball let $D^2\varphi\succeq mI$, $m>0$, and put $e_j=\|\nabla\varphi_j-\nabla\varphi\|_{C^0}$. For $z,w$ in the ball,
\begin{align*}
 B_j(z,w)&:=\varphi_j(w)-\varphi_j(z)-\nabla\varphi_j(z)\cdot(w-z)\\
 &\ge\frac m2|w-z|^2-2e_j|w-z|.
\end{align*}
At every fixed separation $|w-z|\ge\rho>0$, this is at least $m\rho^2/4$ for all sufficiently large $j$. The finitely many remaining indices have a positive minimum on the same compact set of pairs. Hence the strict-convexity modulus is uniform at each fixed scale.

To specify the section geometry, subtract the tangent plane at $z_0$ and write
\[
 u_j(z)=\varphi_j(z)-\varphi_j(z_0)-\nabla\varphi_j(z_0)\cdot(z-z_0),
 \qquad S_j(t)=\{u_j<t\}.
\]
Using the strictly convex limit, choose $h,r>0$ such that its corresponding function $u$ satisfies $\min_{\partial B_R(z_0)}u>8h$ and $\sup_{B_r(z_0)}u<h/8$. For large $j$, local uniform convergence and convexity give
\[
 B_r(z_0)\subset S_j(h/4)\subset S_j(h/2)\subset S_j(h)
 \subset S_j(2h)\Subset B_R(z_0).
\]
The gradients of $u_j$ are bounded by a common $L>0$, and therefore
\[
 \operatorname{dist}(S_j(s),\partial S_j(t))\ge(t-s)/L
 \qquad(0<s<t\le2h).
\]
Caffarelli's interior section estimates, in the framework recalled in \cite[Sections 2.9 and 2.12]{BB13}, give a uniform local $C^{1,\alpha}$ bound from the determinant bounds and this section geometry. The right side of \eqref{eq:approx-moment-equation} is then uniformly H\"older continuous. The interior estimate of Figalli--Jhaveri--Mooney \cite[Theorem 1.1]{FJM16} gives a uniform local $C^{2,\alpha}$ bound on smaller neighborhoods, with the exponent decreased if necessary. Its constants are controlled by the determinant bounds, the H\"older bound, and the strict-convexity modulus just obtained. A finite covering gives, on every source compact set $Q$,
\begin{equation*}
 \delta_Q I\preceq D^2\varphi_j\preceq M_Q I
\end{equation*}
for positive constants independent of large $j$.

Let $A_j=(D^2\varphi_j)^{-1}$ and $w_{j,k}=\partial_k\varphi_j$. Differentiating the logarithm of \eqref{eq:approx-moment-equation} gives
\[
 (A_j)_{ab}\partial_{ab}w_{j,k}

We prove the stronger assertion
\begin{equation}\label{eq:H-concavity}
 \mathcal H((1-t)a+tb)\ge(1-t)\mathcal H(a)+t\mathcal H(b)

We first justify the moment-coordinate calculations for the exact one-sided powers used to approximate support constraints. Fix $q\ge2n+20$ be even, $\varepsilon>0$, $c_i>0$, and $p_i\ne0$. Suppose the probability with potential

\end{equation*}

The target field $\tau$ belongs to $C^{q-1,\alpha}_\mathrm{loc}$ and satisfies the same upper bound $4\varepsilon^{-1}I$. The compact identities \eqref{eq:stein}--\eqref{eq:hessian-identity}, the dense-range assertion \eqref{eq:dense-range}, and the tensor identities used below hold for this exact potential.

To establish these assertions, convolve each one-dimensional function $(s_+)^q$ with a nonnegative compactly supported smooth probability kernel of width $\delta$. The resulting convex nonnegative summands converge in $C^{q-1,\alpha}$ on compact sets, since $(s_+)^q\in C^{q-1,1}_\mathrm{loc}$. Denote the smoothed untilted potential by $U_\delta$ and choose $\xi_\delta$ minimizing $\int e^{-U_\delta-\xi\cdot x}\,dx$. Existence, uniqueness, and centering follow as in Lemma~\ref{lem:tilt}: the partition function is strictly convex, the potentials are uniformly bounded on a fixed ball, and integration on a ball opposite the tilt direction gives a common coercive bound. Also $\int e^{-U_\delta}\le\int e^{-\varepsilon|x|^2/2}$. The minimizers are therefore bounded. Gaussian domination and uniqueness imply $\xi_\delta\to\xi$, local $C^{q-1,\alpha}$ convergence of the target densities, and convergence of all polynomial moments.

Apply Lemma~\ref{lem:moment} to the smoothed centered probabilities and translate the source variables so that the minima occur at the origin. The proof of \eqref{eq:source-coercive} gives common source coercivity, because the target densities have a common positive lower bound on a fixed ball. The global Hessian cap and the moment equation give local positive lower Hessian bounds. Interior regularity and Schauder bootstrap then give uniform local $C^{q+1,\alpha}$ estimates, first with a slightly larger H\"older exponent and then with that exponent decreased. Passing to a subsequence proves the equation, normalization, pushforward, and the asserted regularity. On the support of a fixed smooth compact test, all coefficients and derivatives occurring in the compact adjoint and tensor calculations converge. The identities therefore pass to the exact potential. In particular, homogeneity is exact after this limit.

For the dense-range assertion, let $\rho=d\mu/dx$ and $\mathscr L=A(A-I)$. Its coefficients use at most two derivatives of $\tau$ and belong to $C^{q-3,\alpha}_\mathrm{loc}$. The conjugate

 H=D_x^2V=\varepsilon I+\sum_iH_i,\qquad H_i=D_x^2\psi_i,\qquad

 \end{gathered}

Thus $U$ is the orthogonal projection of $F$ onto the complement of affine functions.

For the compact fields in \eqref{eq:test-fields}, set
\begin{equation*}
 \mathcal T(v)=\mathbb E\operatorname{tr}((D_xW)^2)-\mathbb E\operatorname{tr}(MBHB).
\end{equation*}
The first square is a matrix square, not the squared Hilbert--Schmidt norm. Define the invariant derivative norm
\begin{equation}\label{eq:D-norm}
 \mathcal D(P)=\sum_{a,b,k,r,\ell,s}
 M_{ab}\tau_{kr}M_{\ell s}

We claim that the exact remainder is the following sum of nonnegative terms. 
For every $v\in C_c^\infty(\mathbb R^n)$,
\begin{align}
 D+\|u-U\|_{L^2(\mu)}^2
 &=\mathcal T(v)+\varepsilon\mathbb E\{\operatorname{tr}(MB^2)+|W|^2\}\notag\\
 &\quad+\sum_i\mathbb E\operatorname{tr}\big(M(B-\gamma_i\tau)H_i(B-\gamma_i\tau)\big)\notag\\
 &\quad+\sum_i\mathbb E\big[(W-\gamma_ix)^TH_i(W-\gamma_ix)\big].
 \label{eq:exact-remainder}
\end{align}
Every term on the right is nonnegative. More precisely,
\begin{equation}\label{eq:derivative-remainder}
 \mathcal T(v)\ge\frac12\mathbb E\mathcal D(P).
\end{equation}

To prove the claim, first weighted integration by parts gives
\begin{equation}\label{eq:vector-bochner}
 \mathbb E u^2=\mathbb E[W^THW]+\mathbb E\operatorname{tr}((D_xW)^2).
\end{equation}
Indeed, $\partial_{x_i}u=H_{ij}W_j+(V_{x_j}-\partial_{x_j})\partial_{x_i}W_j$, and pairing with $W_i$ proves the identity.

We next compute $\mathcal T(v)$. The calculation is the moment-coordinate tensor argument of \cite[Section 6]{OpenAI26}, with its full derivative remainder retained. Put $C_{ijk}=\varphi_{z_iz_jz_k}$, $(C_i)_{k\ell}=C_{ik\ell}$, and $\zeta_i=\operatorname{tr}(MC_i)$. Differentiating the moment equation gives

 =\tau_{ij}+M_{k\ell}\varphi_{z_kz_\ell z_iz_j}

 \partial_{z_d}M_{dk}=-\zeta_bM_{bk},\qquad
 \partial_{z_d}M_{dk}-x_dM_{dk}=-V_{x_k}.

\begin{equation*}
 (\tau H\tau-\tau)_{ij}
 =(\partial_{z_d}-x_d)(M_{dk}C_{ijk})-\operatorname{tr}(MC_iMC_j).

\begin{align*}

The contracted scalars in these two expansions are invariant under a constant dual change of coordinates

The normalized potentials are $\varphi'(z')=\varphi(Tz')-\log|\det T|$ and $V'(x')=V(T^{-T}x')$, with $Z'=|\det T|Z$. In particular,

At a fixed point choose the constant matrix $T=\tau^{-1/2}$. Thus $\tau=I$ in the new coordinates at that point. This normalization is performed after the integrations by parts; no varying frame is differentiated.

 Q(a)=\{x:\langle u_i,x\rangle\le e^{a_i},\ 1\le i\le N\},

\begin{equation}\label{eq:soft-partition}
 \begin{split}
 Z_{q,\varepsilon}(a,\xi)
 &=\int_{\mathbb R^n}\exp\left(-\frac\varepsilon2|x|^2
                    -\sum_i\psi_{i,q,a}(x)-\xi\cdot x\right)\,dx,\\
 J_{q,\varepsilon}(a,\xi)&=\log Z_{q,\varepsilon}(a,\xi),\qquad
 H_{q,\varepsilon}(a)=\inf_\xi J_{q,\varepsilon}(a,\xi).
 \end{split}
\end{equation}
On a compact set of $a$, there are common $r,R>0$ such that
\begin{equation}\label{eq:polytope-balls}
 rB\subset Q(a)\subset RB,\qquad

\end{equation}
The integrand and all fixed parameter derivatives are integrable locally uniformly in $(a,\xi)$. The logarithmic partition function is strictly convex in $\xi$, with gradient $-\mathbb E x$ and Hessian $\operatorname{Cov}(x)\succ0$. For $e=\xi/|\xi|$, integration on $B(-re/4,r/8)$ gives
\begin{equation}\label{eq:soft-coercivity}
 Z_{q,\varepsilon}(a,\xi)
 \ge |(r/8)B|\exp\left(\frac r8|\xi|-N2^{-q}-\frac{\varepsilon r^2}{8}\right).
\end{equation}

 0=-\operatorname{Cov}_s\big(x,F_s+\xi'(s)\cdot x\big),\qquad \xi'(s)=-\beta_s.

 &=\operatorname{Var}_s(F_s)-\mathbb E_s[xF_s]^T\Sigma_s^{-1}\mathbb E_s[xF_s]
   -q^2\sum_i f_i^2\mathbb E_s\psi_{i,q,a(s)}.
\end{align*}
The estimate \eqref{eq:variance}, with $b_i=-qf_i$, makes the displayed second derivative nonpositive. Hence $H_{q,\varepsilon}$ is concave. The covariance subtraction is exactly the Schur complement produced by optimizing the tilt.

The tilt term vanishes because $\mathbb E_sx=0$. The tails in \eqref{eq:polytope-balls} justify the integration and imply $|\frac d{ds}H_{q,\varepsilon}(a(s))|\le n\|f\|_\infty$.

with a constant majorant inside. If $a_j\to a$ and $\xi_j\to\xi$, the same almost-everywhere limit holds off $\partial Q(a)$.  H_{q,\varepsilon}(a)\longrightarrow\mathcal P(Q(a))
 \quad\hbox{locally uniformly, first as }q\to\infty\hbox{ and then as }\varepsilon\downarrow0.

Thus $a\mapsto\mathcal P(Q(a))$ is concave.

Each function on the left is concave in $a$. Its limit is therefore concave on $C(S^{n-1})$.

This proves the inequality in Theorem~\ref{thm:logbm}. The intermediate Wulff body has not been assumed centered.

\[
 \int_{1/3}^{2/3}-G_j''(t)\,dt
 =G_j'(1/3)-G_j'(2/3)\le C_1\varepsilon_j.
\]
Choose $t_j\in[1/3,2/3]$ with $-G_j''(t_j)\le C_2\varepsilon_j$. The same secant bounds give $|G_j'(t_j)|\le C_3\varepsilon_j$. Passing to a subsequence, let $t_j\to t_*\in[1/3,2/3]$ and put $C=C_{t_*}$.

Write $D_{m_j}=\{p_{1,j},\ldots,p_{N_j,j}\}$ and, at $t_j$, define
\begin{equation}\label{eq:equality-regularization}
 \begin{gathered}
 \psi_{i,j}(x)=\left(\frac{\langle p_{i,j},x\rangle_+}{e^{a_{t_j}(p_{i,j})}}\right)^{q_j},\qquad
 V_j(x)=\frac{\varepsilon_j}2|x|^2+\sum_i\psi_{i,j}(x)+\xi_j\cdot x,\\
 Z_j=\int e^{-V_j}=e^{G_j(t_j)},\qquad d\mu_j=Z_j^{-1}e^{-V_j}\,dx,
 \end{gathered}
\end{equation}
where $\xi_j$ is the minimizing tilt. These probabilities are centered.

The common ball estimates, \eqref{eq:soft-coercivity}, and \eqref{eq:choice-q} bound the tilts uniformly. On every compact subset of $\operatorname{int} C$, the normalized support ratios in \eqref{eq:equality-regularization} are at most some fixed $\rho<1$ for large $j$. For each fixed $k$, the derivatives of their sum are bounded by $C_kN_jq_j^k\rho^{q_j-k}\to0$. Outside $C$, a fixed direction from the dense union strictly violates the limiting support constraint; its corresponding power tends to infinity. Common coordinate constraints give an integrable majorant of the form $\exp(M|x|-(|x|/R_0)^{q_0})$ off a fixed ball, with $q_0>1$. Dominated convergence, including every fixed polynomial weight, therefore shows that any subsequential tilt limit centers $\mathbf 1_Ce^{-\xi\cdot x}$. Lemma~\ref{lem:tilt} identifies it as $\xi_C$. Consequently,
\begin{equation}\label{eq:equality-target-limit}
 \xi_j\to\xi:=\xi_C,\qquad Z_j\to1,\qquad
 d\mu_j\to d\mu:=\mathbf 1_Ce^{-\xi\cdot x}\,dx.
\end{equation}
The same dominated convergence with weight $1+|x|$ gives
\[
 \int_{\mathbb R^n}(1+|x|)\big|Z_j^{-1}e^{-V_j(x)}-\mathbf 1_C(x)e^{-\xi\cdot x}\big|\,dx\longrightarrow0.
\]
In particular, the density convergence is in $L^1$ and $W_1(\mu_j,\mu)\to0$: couple the common part of the densities identically and bound the transport cost of the remaining equal masses through the origin by the $|x|$-weighted difference. All fixed polynomial moments converge, and $V_j\to\xi\cdot x$ locally in every fixed differentiability class on $\operatorname{int} C$. The limiting measure is a centered probability, although the uniform measure on $C$ need not yet be centered.

Set
\[
 F_j=-q_j\sum_i f(p_{i,j})\psi_{i,j},\quad
 \Sigma_j=\int xx^T\,d\mu_j,\quad
 \beta_j=\Sigma_j^{-1}\int xF_j\,d\mu_j,\quad
 U_j=F_j-\int F_j\,d\mu_j-\beta_j\cdot x.
\]
The Schur-complement formula and the parameter choice give
\begin{equation*}
 D_j:=q_j^2\sum_i f(p_{i,j})^2\int\psi_{i,j}\,d\mu_j
       -\int U_j^2\,d\mu_j
 =-G_j''(t_j)\le C_2\varepsilon_j,
\end{equation*}
and
\begin{equation}\label{eq:F-mean}
 \left|\int F_j\,d\mu_j\right|=|G_j'(t_j)|\le C_3\varepsilon_j.
\end{equation}
By Lemma~\ref{lem:stability}, the moment potentials may be translated so that $\varphi_j\to\varphi$ locally in every fixed $C^k$, where $\varphi$ is the moment potential of $\mu$. Put $x_j=\nabla\varphi_j$ and $\tau_j=D^2\varphi_j$. On each source compact set, $\tau_j$ and $\tau_j^{-1}$ are uniformly bounded, and $e^{-\varphi_j}$ is bounded above and below by positive constants.

The approximation statement \eqref{eq:retained-fields} gives tests $v_j\in C_c^\infty$ whose fields satisfy
\[
 \|u_j-U_j\|_{L^2(\mu_j)}\le\varepsilon_j,\qquad
 \mathcal T(v_j)+\varepsilon_j\int|W_j|^2\,d\mu_j
 \le D_j+\varepsilon_j^2\le C_4\varepsilon_j.
\]
Thus
\begin{equation}\label{eq:uniform-flux}
 \mathcal T(v_j)\to0,\qquad \sup_j\int|W_j|^2\,d\mu_j<\infty.
\end{equation}
The retained quadratic regularizer is essential for the second bound. In source coordinates, \eqref{eq:derivative-remainder} and \eqref{eq:derivative-comparison} imply
\begin{equation*}
 \int_{B_R}|D_zP_j|^2\,dz\longrightarrow0\qquad(R<\infty).
\end{equation*}
Let $A_j$ be the average of $P_j$ over $B_1$. Poincar\'e's inequality and the identity
\begin{equation*}
 W_j=P_jx_j-\operatorname{div}_zP_j
\end{equation*}
give $P_j-A_j\to0$ and $W_j-A_jx_j\to0$ in $L^2(B_1)$. The global bound in \eqref{eq:uniform-flux} bounds $W_j$ there, since the source density is bounded below. Also
\[
 \int_{B_1}x_jx_j^T\,dz\longrightarrow
 \int_{B_1}(\nabla\varphi)(\nabla\varphi)^T\,dz\succ0.
\]
The last matrix is positive definite because the gradient image of $B_1$ is open. Therefore $A_j$ is bounded. After a subsequence, $A_j\to A$. Applying Poincar\'e's inequality on larger balls and comparing their averages on $B_1$, we obtain
\begin{equation*}
 P_j\to A,\qquad W_j\to A\nabla\varphi
 \quad\hbox{in }L^2_\mathrm{loc}(\mathbb R^n_z).
\end{equation*}
The convergent moment maps and their inverses give $W_j(x)\to Ax$ in $L^2_\mathrm{loc}(\operatorname{int} C)$. Since $B_j=P_j\tau_j$ is symmetric, passage to the limit gives
\begin{equation}\label{eq:matrix-commutation}
 AD^2\varphi(z)=D^2\varphi(z)A^T\qquad(z\in\mathbb R^n).
\end{equation}

Define a positive measure on positions and directions by
\begin{equation}\label{eq:joint-layer}
 d\Lambda_j(x,u)=\sum_i q_j\psi_{i,j}(x)\,d\mu_j(x)\,\delta_{p_{i,j}}(du).
\end{equation}
The direction is $p_{i,j}$ itself, because the powers are one-sided. Radial integration by parts gives
\begin{align*}
 \Lambda_j(\mathbb R^n\timesS^{n-1})&=n-\varepsilon_j\int|x|^2\,d\mu_j,
 \\
 \int|x|^2\,d\Lambda_j
 &=(n+2)\int|x|^2\,d\mu_j-\varepsilon_j\int|x|^4\,d\mu_j
   -\int|x|^2\xi_j\cdot x\,d\mu_j.
\end{align*}
The common moment bounds imply bounded mass and second moment, hence tightness and uniform integrability of the first moments. For a compact smooth function $g$,
\begin{align*}
 \int g(x)\,d\Lambda_j(x,u)
 &=\int\big(ng+x\cdot\nabla g-\varepsilon_jg|x|^2-g\xi_j\cdot x\big)\,d\mu_j\\
 &\longrightarrow\int_C(ng+x\cdot\nabla g-g\xi\cdot x)e^{-\xi\cdot x}\,dx\\
 &=\int_{\partial C}g(x)h_C(\nu_C)e^{-\xi\cdot x}\,d\mathcal H^{n-1}(x).
\end{align*}
Thus the limiting position marginal lies on $\partial C$.

Every joint limit is supported on contact pairs
\begin{equation*}
 x\in\partial C,\qquad \langle x,u\rangle=h_C(u)=e^{a_{t_*}(u)}.
\end{equation*}
Indeed, on a compact neighborhood where $\langle x,u\rangle\le(1-\delta)e^{a_{t_*}(u)}$, the mass in \eqref{eq:joint-layer} is at most $N_jq_j(1-\delta/2)^{q_j}$ for large $j$, and tends to zero by \eqref{eq:choice-q}. On the limiting boundary one always has $\langle x,u\rangle\le h_C(u)\le e^{a_{t_*}(u)}$. This proves the asserted restriction to contact pairs. The outer normal is unique at almost every boundary point, and the position marginal gives no mass to the exceptional set. Hence the joint limit is unique:
\begin{equation*}
 d\Lambda_j(x,u)\rightharpoonup
 h_C(\nu_C(x))e^{-\xi\cdot x}\,d\mathcal H^{n-1}(x)\,
 \delta_{\nu_C(x)}(du).
\end{equation*}
Since $F_j\,d\mu_j=-\int f(u)\,d\Lambda_j(x,u)$ and $f$ is continuous,
\[
 F_j\,d\mu_j\rightharpoonup
 -f(\nu_C)h_C(\nu_C)e^{-\xi\cdot x}\,d\mathcal H^{n-1}|_{\partial C}.
\]
The first-moment control permits linear-growth tests, so $\int xF_j\,d\mu_j$ converges. Because $\Sigma_j\to\Sigma=\int_Cxx^T\,d\mu\succ0$, the vectors $\beta_j$ converge to some $\beta\in\mathbb R^n$. The approximation of $U_j$ and \eqref{eq:F-mean} now give
\begin{equation}\label{eq:u-limit}
 u_j\,d\mu_j\rightharpoonup
 -f(\nu_C)h_C(\nu_C)e^{-\xi\cdot x}\,d\mathcal H^{n-1}|_{\partial C}
 -\beta\cdot x\,d\mu(x).
\end{equation}
The interior affine term is retained; it need not vanish separately.

For $g\in C_c^\infty(\mathbb R^n)$, adjunction gives
\[
 \int u_jg\,d\mu_j=\int W_j\cdot\nabla g\,d\mu_j.
\]
The right side tends to $\int_CAx\cdot\nabla g\,d\mu$. To justify passage across the boundary, first restrict to $C_\delta=\{x\in C:\operatorname{dist}(x,\partial C)\ge\delta\}$, where local convergence applies. The absolute value of the remaining integral is at most
\[
 \|\nabla g\|_\infty
 \left(\int|W_j|^2\,d\mu_j\right)^{1/2}
 \mu_j(\mathbb R^n\setminus C_\delta)^{1/2}.
\]
Its limit superior tends to zero as $\delta\downarrow0$, by \eqref{eq:uniform-flux} and the $L^1$ convergence in \eqref{eq:equality-target-limit}. Hence \eqref{eq:u-limit} implies
\begin{equation*}
 \int_C Ax\cdot\nabla g\,d\mu
 =-\int_{\partial C}gf(\nu_C)h_C(\nu_C)e^{-\xi\cdot x}\,d\mathcal H^{n-1}
   -\int_Cg\beta\cdot x\,d\mu.
\end{equation*}
On the other hand, weighted integration by parts in $C$ gives
\[
 \int_C Ax\cdot\nabla g\,d\mu
 =\int_{\partial C}gAx\cdot\nu_Ce^{-\xi\cdot x}\,d\mathcal H^{n-1}
  +\int_Cg\big((A^T\xi)\cdot x-\operatorname{tr} A\big)\,d\mu.
\]
Testing first inside $C$ yields an affine identity on an open set, and therefore
\begin{equation}\label{eq:affine-flux}
 \operatorname{tr} A=0,\qquad \beta=-A^T\xi.
\end{equation}
Subtracting the interior terms proves
\begin{equation}\label{eq:boundary-flux}
 Ax\cdot\nu_C(x)=-f(\nu_C(x))h_C(\nu_C(x))
 \quad\hbox{for }\mathcal H^{n-1}\hbox{-almost every }x\in\partial C.
\end{equation}
Here restrictions of compact smooth tests are dense in $C(\partial C)$, so the weak identity is an identity of finite boundary measures. No conclusion that $\xi=0$ has been used.

Equation \eqref{eq:matrix-commutation} makes $A$ self-adjoint for a positive definite inner product. For example, if $\tau_0=D^2\varphi(0)$, then $\tau_0^{-1/2}A\tau_0^{1/2}$ is symmetric. Use the constant dual change
\[
 z=Tz',\quad x'=T^Tx,\qquad
 T=(\det\tau_0)^{1/(2n)}\tau_0^{-1/2},\qquad\det T=1.
\]
It makes the transformed $A$ symmetric. The common normal support and the flux identity are preserved: the flux and the supporting distance transform with the same normalizing factor, as in \eqref{eq:covariance}. We henceforth work in these coordinates and omit primes.

Let $\lambda_1,\ldots,\lambda_r$ be the distinct eigenvalues of $A$, with orthogonal eigenspaces $E_i$. Equation \eqref{eq:matrix-commutation} says that $D^2\varphi$ commutes with $A$. All mixed Hessian blocks between distinct eigenspaces vanish. Since $\varphi$ is defined on the whole product space, integration gives
\[
 \varphi(z_1+\cdots+z_r)=c+\sum_i\varphi_i(z_i),\qquad z_i\in E_i.
\]
Its gradient image is a Cartesian product. Taking closures, we obtain
\[
 C=C_1\oplus\cdots\oplus C_r,\qquad C_i\subset E_i.
\]
Each factor has nonempty relative interior and contains zero in that interior. The boundary of a product, or Lemma~\ref{lem:product}, gives
\[
 \operatorname{supp} S_C\subset\bigcup_i S(E_i).
\]
Let $S$ be the common support of $S_C,S_K,S_L$ in \eqref{eq:common-measure}, and put $S_i=S\cap E_i$. Reconstruction in Lemma~\ref{lem:reconstruct} separates the constraints and yields
\[
 K=\bigoplus_iK_i,\qquad L=\bigoplus_iL_i,
\]
where
\[
 K_i=\{y\in E_i:\langle y,u\rangle\le h_K(u),\ u\in S_i\},\qquad
 L_i=\{y\in E_i:\langle y,u\rangle\le h_L(u),\ u\in S_i\}.
\]
These factors are bounded and full-dimensional relative to $E_i$, since their products are the compact full-dimensional endpoint bodies. Fubini's theorem gives $g(K_i)=g(L_i)=0$, because the endpoint centroids vanish.

At almost every boundary point of $C$ with normal $u\in E_i$, one has $Ax\cdot u=\lambda_i h_C(u)$. Thus \eqref{eq:boundary-flux} gives $f(u)=-\lambda_i$ for $S_C$-almost every such direction. Continuity extends this equality to $S_i$. It follows that
\[
 h_L(u)=e^{-\lambda_i}h_K(u)\quad(u\in S_i),\qquad
 L_i=e^{-\lambda_i}K_i.
\]
Set $a_i=e^{-\lambda_i}$. By \eqref{eq:affine-flux},
\[
 \prod_i a_i^{\dim E_i}=\exp\left(-\sum_i\lambda_i\dim E_i\right)
 =e^{-\operatorname{tr} A}=1.
\]
Undoing the common linear transformation gives complementary subspaces without an orthogonality requirement. Thus \eqref{eq:affine-hypothesis} implies the claimed common direct-sum representation. In particular, it proves necessity for the normalized equality pair in Theorem~\ref{thm:logbm}. Restoring the original endpoint volumes gives arbitrary positive dilation factors.

Conversely, assume \eqref{eq:main-split}. By \eqref{eq:covariance} we may make the subspaces orthogonal. For $u=\sum_i u_i$, $u_i\in E_i$, put $b_i=h_{K_i}(u_i)\ge0$. H\"older's inequality gives
\[
 \sum_i a_i^s b_i\le\left(\sum_i b_i\right)^{1-s}
                        \left(\sum_i a_i b_i\right)^s,
 \qquad 0<s<1.
\]
Hence $\bigoplus_i a_i^sK_i\subset W[h_K^{1-s}h_L^s]$. The defining inequalities in directions confined to $E_i$ give the reverse inclusion. Taking volumes,
\[
 \big|W[h_K^{1-s}h_L^s]\big|=|K|\prod_i a_i^{s\dim E_i}
 =|K|^{1-s}|L|^s.
\]
Thus the direct-sum condition is sufficient, and the proof is complete.
