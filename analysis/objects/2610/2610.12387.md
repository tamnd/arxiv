<!-- definition-1: body_md -->
\label{def:renormalized-solution}
Let $T>0$, 
and assume \eqref{eq:viscosity} and \eqref{eq:data}.
A pair $(\rho,u)$, with $\rho\ge0$ and $m=\rho u$, is a
\emph{finite-energy renormalized weak solution} of the
isothermal system \eqref{eq:mass}--\eqref{eq:stress} on $Q_T$ with
initial data $(\rho_0,m_0)$ if the following conditions hold.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=*,itemsep=\smallskipamount]
\item \emph{Regularity and initial traces.} We have
\begin{align}
 \rho&\in L^\infty(0,T;L\log L(\Omega))
       \capC_{\mathrm{weak}}([0,T];L^1(\Omega)),\label{eq:rho-class}\\
 u&\in L^2(0,T;W^{1,2}_0(\Omega;\mathbb R^2)),\qquad
 \sqrt\rho\,u\in L^\infty(0,T;L^2(\Omega;\mathbb R^2)),\label{eq:u-class}\\
 m&\inC_{\mathrm{weak}}([0,T];L^1(\Omega;\mathbb R^2)),\qquad
 \rho(0)=\rho_0,\quad m(0)=m_0.\label{eq:m-class}
\end{align}

\item \emph{Weak continuity and momentum equations.} For every smooth
scalar test $\psi$ on $[0,T)\times\overline\Omega$,
\begin{equation}\label{weak:mass}
 \int_0^T\!\int_\Omega(\rho\partial_t\psi+m\cdot\nabla\psi)
 +\int_\Omega\rho_0\psi(0)=0.
\end{equation}
For every smooth vector test $\varphi$ on the same set with zero
boundary trace,
\begin{equation}\label{weak:mom}
 \int_0^T\!\int_\Omega
 [m\cdot\partial_t\varphi+(\rho u\otimes u):\nabla\varphi
       +\rho\operatorname{div}\varphi-\mathbb S(u):\nabla\varphi]
 +\int_\Omega m_0\cdot\varphi(0)=0.
\end{equation}

\item \emph{Renormalized continuity.} For every
$b\in C^1([0,\infty))$ whose derivative is compactly supported,
\begin{equation}\label{eq:renormalized}
 \partial_t b(\rho)+\operatorname{div}(b(\rho)u)
 +(\rho b'(\rho)-b(\rho))\operatorname{div}u=0
\end{equation}
with initial value $b(\rho_0)$. Precisely, for every scalar test $\psi$
as in \textup{(ii)},
\begin{equation}\label{weak:renorm}
 \int_0^T\!\int_\Omega
 [b(\rho)\partial_t\psi+b(\rho)u\cdot\nabla\psi
  -(\rho b'(\rho)-b(\rho))\operatorname{div}u\,\psi]
 +\int_\Omega b(\rho_0)\psi(0)=0.
\end{equation}

\item \emph{Initial-time energy inequality.} For almost every
$t\in(0,T)$,
\begin{equation}\label{eq:energy}
 \int_\Omega\left[\frac12\rho|u|^2+H(\rho)\right](t)\,dx
 +\int_0^t\!\int_\Omega\mathbb S(u):\nabla u\,dx\,ds
 \le E_0.
\end{equation}
\end{enumerate}
A solution on $[0,\infty)$ is \emph{global} if these conditions hold
on every finite time interval with the same initial data.

\label{thm:stability}
Assume \eqref{eq:viscosity}. Let $\beta\ge6$ be fixed and let
$\delta_n\downarrow0$. Let $(\rho_n,u_n)$ be renormalized weak solutions
with no-slip velocity and pressure $\rho_n+\delta_n\rho_n^\beta$,
satisfying the fixed-parameter regularity and renormalization properties
stated above. Write $m_n=\rho_nu_n$.

Suppose the initial density and momentum converge in $L^1$ to
$(\rho_0,m_0)$ satisfying \eqref{eq:data}, their natural energy integrals
converge to $E_0$, and $\delta_n\int_\Omega\rho_{0,n}^\beta\to0$.
Assume that, on each finite interval $(0,T)$ on which the sequence is
defined, the following energy inequality holds for almost every
$t\in(0,T)$, with $E_{0,n}$ denoting the corresponding initial energy:
\begin{equation}\label{exist:approxenergy}
 E_n(t)+\int_0^t\!\int_\Omega\mathbb S(u_n):\nabla u_n\le E_{0,n},
 \quad E_n(t)=\int_\Omega\left[
 \frac12\rho_n|u_n|^2+H(\rho_n)+\frac{\delta_n}{\beta-1}\rho_n^\beta
 \right](t),
\end{equation}

On each such interval, assume in addition that
\[
 \underset{0<t<T}{\operatorname{ess\,sup}}
 \int_\Omega\left(\rho_n|u_n|^2+\rho_n\log(1+\rho_n)
                    +\delta_n\rho_n^\beta\right)dx
 +\int_{Q_T}|\nabla u_n|^2\,dx\,dt\le C
\]
and
\[
 \int_{Q_T}\zeta^2\bigl(\rho_n+\delta_n\rho_n^\beta\bigr)
                  \rho_n^{1/8}\,dx\,dt\le C_{\zeta,T}
 \qquad\text{for every }0\le\zeta\in C_c^\infty(\Omega),
\]
with constants independent of $n$.
Then, on every finite time interval $(0,T)$ on which the sequence is defined, a
subsequence satisfies
\begin{align*}
 \rho_n&\longrightarrow\rho&&\text{strongly in }L^1(Q_T),\\
 u_n&\rightharpoonup u&&\text{weakly in }L^2_tW^{1,2}_0,\\
 m_n&\rightharpoonup\rho u&&\text{weakly in }L^1(Q_T),\\
 \chi m_n&\longrightarrow\chi\rho u
       &&\text{strongly in }L^2_tW^{-1,192/95}_x
               \quad(\chi\in C_c^\infty(\Omega)),\\
 \rho_nu_n\otimes u_n&\rightharpoonup\rho u\otimes u
       &&\text{locally in }L^{6/5}_tL^{192/191}_x.
\end{align*}
The limit is a finite-energy renormalized weak solution in the sense of
Definition~\ref{def:renormalized-solution} on this interval. Artificial
pressure converges to zero in $L^1(Q_D)$ for every $D\Subset\Omega$.

The reflexive bounds in \eqref{mom:mixed} pass to $\rho$ and $m$.

matrix tests, and choose $\chi=1$ near their spatial supports, with
$\operatorname{supp}\chi\Subset B$. In time-integrated $H^{-1}(B)$--$H^1_0(B)$

\begin{equation}\label{convergence of mass in dis}
 \begin{aligned}
 \int_{Q_T}(\rho_nu_n-\rho u)\cdot\varphi
 &=\langle\chi(\rho_n-\rho),u_n\cdot\varphi\rangle
   +\langle\chi\rho,(u_n-u)\cdot\varphi\rangle,\\
 \int_{Q_T}(F_n-m\otimes u):\Phi
 &=\langle\chi(m_n-m),\Phi u_n\rangle
   +\langle\chi m,\Phi(u_n-u)\rangle.
 \end{aligned}
\end{equation}
The first terms vanish by Proposition~\ref{mom:compact}; the second
by weak Sobolev convergence of $u_n$. All pairings agree with ordinary
integrals: density--velocity products lie in $L^{4/3}_tL^1_x$, and
momentum--velocity products in $L^{6/5}_tL^1_x$, by
\eqref{mom:mixed} and finite planar Sobolev embeddings.
The first identity identifies $m=\rho u$ by comparison with
\eqref{mom:weak}. The second identifies the distributional limit of
$F_n$ as $m\otimes u=\rho u\otimes u$. Its reflexive mixed bound
then gives \eqref{mom:convection}.

\noindent\textbf{Admissibility of the momentum test.}
We first justify the approximate and limiting tests by mollification;
the inertial and stress calculations will then give the full identity.
For a smooth compactly supported vector test $\Psi$, the approximate
momentum equation corresponding to \eqref{app:fixedsystem}, with
$(r,v,\delta)=(\rho_n,u_n,\delta_n)$, reads
\begin{equation}\label{flux:approximate-momentum}
 0=\int_Q \left[ m_{n,i}\partial_t\Psi_i
    +F_{n,ij}\partial_j\Psi_i+p_n\partial_i\Psi_i
    -\mathbb S_{n,ij}\partial_j\Psi_i \right].
\end{equation}

Let $\Psi_{n,i}=\phi A_i[\chi T_k(\rho_n)]=\phi A_{n,i}$ and
$A_n^\varepsilon=A[b_n^\varepsilon]$, where the mollification is
specified below. Testing \eqref{flux:approximate-momentum} with
$\Psi_n^\varepsilon=\phi A_n^\varepsilon$ gives
\begin{equation}\label{flux:regularized-momentum-test}
 \begin{aligned}
 0=\int_Q \Big[&m_{n,i}\partial_t(\phi A_{n,i}^\varepsilon)
   +F_{n,ij}\partial_j(\phi A_{n,i}^\varepsilon)
   +p_n\partial_i(\phi A_{n,i}^\varepsilon)-\mathbb S_{n,ij}\partial_j(\phi A_{n,i}^\varepsilon)\Big].
 \end{aligned}
\end{equation}
Here, for fixed $n$ and $k$, we let $v^\varepsilon$ denote
standard mollification of $v$ in space and time with nonnegative
unit-mass kernels. The localized quantities $b_n$, $b_nu_n$, and $g_n$
are extended by zero in space before convolution; $\varepsilon$ is small
enough that time convolution near $\operatorname{supp}\phi$ stays inside
$(0,T)$. Mollifying \eqref{flux:localized-cap} and using
\eqref{flux:operators} gives
\begin{equation}\label{flux:regularized-cap}
 \begin{aligned}
 \partial_tb_n^\varepsilon
 +\operatorname{div}((b_nu_n)^\varepsilon)&=g_n^\varepsilon,\\
 \partial_tA_{n,i}^\varepsilon
 &=-R_{ij}[(b_nu_{n,j})^\varepsilon]+A_i[g_n^\varepsilon].
 \end{aligned}
\end{equation}
Thus $\Psi_n^\varepsilon$ is a smooth, compactly supported momentum
test whose time derivative is determined by
\eqref{flux:regularized-cap}.

To remove the smoothing at fixed $n$ and $k$, we use the following
pairings for the time, convective, and viscous terms, respectively.
On an interior cylinder $U=I\times D\Subset Q$ containing
$\operatorname{supp}\phi$, they follow from \eqref{mom:mixed} and
\eqref{flux:potential-bounds}:
\begin{align*}
 &m_n\in L^3_tL^{96/95}_x,
       &&\partial_tA_n\in L^2_tL^{96}_x,\\
 &F_n\in L^{6/5}_tL^{192/191}_x,
       &&\nabla A_n\in L^6_tL^{192}_x,\\
 &\mathbb S(u_n)\in L^2,
       &&\nabla\Psi_n\in L^2.
\end{align*}
All spaces in this display are restricted to $U$.
For fixed $n$ and $k$, as $\varepsilon\to0$, we have
\[
 \partial_tA_n^\varepsilon\longrightarrow\partial_tA_n
    \quad\text{in }L^2(I;L^{96}(D)),\qquad
 \nabla A_n^\varepsilon\longrightarrow\nabla A_n
    \quad\text{in }L^6(I;L^{192}(D)),
\]
and
\[
 \nabla\Psi_n^\varepsilon\longrightarrow\nabla\Psi_n
    \quad\text{in }L^2(U).
\]
H\"older's inequality and the convergences above pass the principal
time and convective contributions
\[
 \int_Q \phi m_{n,i}\partial_tA_{n,i}^\varepsilon,
 \qquad
 \int_Q \phi F_{n,ij}\partial_jA_{n,i}^\varepsilon,
\]
as $\varepsilon\to0$, while
$\nabla\Psi_n^\varepsilon\to\nabla\Psi_n$ in $L^2(U)$ passes the full
viscous term
\[
 -\int_Q\mathbb S_{n,ij}\partial_j(\phi A_{n,i}^\varepsilon).
\]
The cutoff contributions from the time and convective terms,
\[
 \int_Q(\partial_t\phi)m_n\cdot A_n^\varepsilon,
 \qquad
 \int_Q(\partial_j\phi)F_{n,ij}A_{n,i}^\varepsilon,
\]
converge by the uniform convergence
$A_n^\varepsilon\to A_n$ on $\overline U$.

For the pressure term, \eqref{flux:operators} gives the decomposition
\[
 \int_Q p_n\operatorname{div}\Psi_n^\varepsilon
 =\int_Q p_n\phi b_n^\varepsilon
  +\int_Q p_n\nabla\phi\cdot A_n^\varepsilon.
\]
At fixed $n$, $p_n\in L^1(Q)$, while
$0\le b_n^\varepsilon\le2k$ and $b_n^\varepsilon\to b_n$
almost everywhere. Dominated convergence therefore passes the
principal pressure contribution
$\int_Q p_n\phi b_n^\varepsilon$. The cutoff pressure contribution
$\int_Q p_n\nabla\phi\cdot A_n^\varepsilon$ converges because
$A_n^\varepsilon\to A_n$ uniformly on $\overline U$.
This justifies removing the smoothing in the approximate momentum
identity at fixed $n$ and $k$.

The limiting test is justified by the same procedure, independently
of the sequence limit. For $\Psi=\phi Ab$, we use the interior momentum
identity \eqref{weak:mom}, established in
Section~\ref{sec:momentum}, and mollify the limiting truncation
equation \eqref{flux:limiting-cap}.
The identities \eqref{flux:regularized-cap} then hold with
$(b_n,b_nu_n,g_n)$ replaced by $(b,bu,g)$. Writing
$A^\varepsilon=A[b^\varepsilon]$ and
$\Psi^\varepsilon=\phi A^\varepsilon$, the same finite-exponent
estimates pass the principal time and convective contributions
\[
 \int_Q\phi m_i\partial_tA_i^\varepsilon,
 \qquad
 \int_Q\phi F_{ij}\partial_jA_i^\varepsilon,
\]
and the $L^2$ convergence of $\nabla\Psi^\varepsilon$ passes the full
viscous term
\[
 -\int_Q\mathbb S_{ij}(u)\partial_j\Psi_i^\varepsilon.
\]
The corresponding time and convective cutoff contributions, containing
$(\partial_t\phi)A^\varepsilon$ and
$(\nabla\phi)A^\varepsilon$, pass by uniform local convergence of
$A^\varepsilon$ to $Ab$.

For the principal pressure contribution, since $\rho\in L^1(U)$,
$0\le b^\varepsilon\le2k$, and $b^\varepsilon\to b$ almost everywhere,
dominated convergence gives
\[
 \int_Q\rho\phi b^\varepsilon\longrightarrow\int_Q\rho\phi b.
\]
The cutoff pressure contribution
$\int_Q\rho\nabla\phi\cdot A^\varepsilon$ converges by the same
uniform local convergence and the local $L^1$ integrability of
$\rho$. This completes the justification of both momentum tests;
we now compute the approximate identity before passing $n\to\infty$.

\noindent\textbf{The inertial terms.}
Let $\mathbb S_n=\mathbb S(u_n)$ and
\[
 \mathcal C_{n,i}=[u_{n,j},R_{ij}]b_n
       :=u_{n,j}R_{ij}b_n-R_{ij}(u_{n,j}b_n).
\]
Using \eqref{flux:potential-bounds}, the principal time and convective
contributions
$\int_Q\phi m_{n,i}\partial_tA_{n,i}$ and
$\int_Q\phi F_{n,ij}\partial_jA_{n,i}$ combine exactly into
\begin{equation}\label{flux:inertia}
 \int_Q\phi m_{n,i}
       \{-R_{ij}(b_nu_{n,j})+A_ig_n+u_{n,j}R_{ij}b_n\}
 =\int_Q\phi m_n\cdot\mathcal C_n+
       \int_Q\phi m_n\cdot Ag_n.
\end{equation}
This grouping is performed before taking $n\to\infty$.

\noindent\textbf{The stress contribution.}
For the stress, symmetry of $R_{ij}$ and integration by parts give
\begin{equation}\label{flux:stress}
 \int_Q\phi\mathbb S_{n,ij}R_{ij}b_n
 =\nu\int_Q\phi d_nb_n
   +2\mu\int_Q u_{n,i}
       \bigl((\partial_i\phi)b_n-(\partial_j\phi)R_{ij}b_n\bigr).
\end{equation}
Indeed, each symmetric-gradient term equals
$\int_Q\phi d_nb_n+\int_Q u_{n,i}(\partial_i\phi)b_n
-\int_Q u_{n,i}(\partial_j\phi)R_{ij}b_n$;
the bulk-viscosity term is $\lambda\int_Q\phi d_nb_n$.

\noindent\textbf{The full identity.}
Substituting the grouped inertia and stress formula into the momentum
identity yields the formula to be passed to the limit:
\begin{align}
 \int_Q\phi(p_n-\nu d_n)b_n
 ={}&-\int_Q\phi m_n\cdot\mathcal C_n
      -\int_Q\phi m_n\cdot Ag_n
      -\int_Q(\partial_t\phi)m_n\cdot A_n \notag\\
    &-\int_Q(\partial_j\phi)F_{n,ij}A_{n,i}
      -\int_Qp_n\nabla\phi\cdot A_n
      +\int_Q\mathbb S_{n,ij}(\partial_j\phi)A_{n,i}\notag\\
    &+2\mu\int_Q u_{n,i}
       \bigl((\partial_i\phi)b_n-(\partial_j\phi)R_{ij}b_n\bigr).
       \label{flux:full}
\end{align}
The derivatives of $\chi$ enter through the source $g_n$ in
\eqref{flux:localized-cap}, whose potential is retained in the first
line. Since $\chi=1$ on $\operatorname{supp}\phi$, the left side contains
the original, unlocalized cap there. This proves the stated identity.

Both passages use the same negative-Sobolev duality: strong compactness
of the localized momentum is paired with weak convergence of the
commutator or source potential.

\noindent\textbf{The commutator term.}
We use the following first-derivative commutator estimate of
Coifman--Meyer type \cite{CM,Lions}:
\begin{equation}\label{flux:commutator-estimate}
 \|\nabla[v,R_{ij}]w\|_{L^r(\mathbb R^2)}
 \le C_{p,q,r}\|\nabla v\|_{L^p(\mathbb R^2)}
                   \|w\|_{L^q(\mathbb R^2)},
 \quad \frac1r=\frac1p+\frac1q,
 \quad 1<p,q,r<\infty.
\end{equation}
Taking $p=2$, $q=192$, and hence $r=192/97$ in
\eqref{flux:commutator-estimate}, the undifferentiated norm is bounded
with the same exponents:
\[
 \|u_{n,j}R_{ij}b_n\|_{L^{\frac{192}{97}}}
  +\|R_{ij}(u_{n,j}b_n)\|_{L^{\frac{192}{97}}}
 \le C\|u_n\|_{L^2}\|b_n\|_{L^{192}}.
\]
Poincar\'e's inequality and \eqref{flux:commutator-estimate} therefore give
\begin{equation}\label{flux:commutator-bound}
 \|\mathcal C_n\|_{L^2_tW^{1,\frac{192}{97}}(\mathbb R^2)}
       \le C_{k,\chi,\Omega}\|\nabla u_n\|_{L^2(Q)}.
\end{equation}

To identify the weak limit of the commutator, we use the cap
compactness, without strong $L^1$ density convergence. For any scalar
$\eta\in C_c^\infty((0,T)\times\mathbb R^2)$,
self-adjointness gives, with the time integrations understood,
\[
 \int\eta u_{n,j}R_{ij}b_n
       =\langle b_n,R_{ij}(\eta u_{n,j})\rangle,
 \qquad
 \int\eta R_{ij}(b_nu_{n,j})
       =\langle b_n,u_{n,j}R_{ij}\eta\rangle.
\]

Multiplication by the fixed smooth function \(\eta\) is bounded on
\(H^1(\mathbb R^2)\), and \(R_{ij}\) is bounded on the same space.
The zero extensions satisfy \(u_n\rightharpoonup u\) in
\(L^2(0,T;H^1(\mathbb R^2))\), and hence
\[
 R_{ij}(\eta u_{n,j})
 \rightharpoonup
 R_{ij}(\eta u_j)
 \qquad\text{in }L^2(0,T;H^1(\mathbb R^2)).
\]
Moreover, \(R_{ij}\eta\) and its first spatial derivatives are bounded
on \(\mathbb R^2\), uniformly in time, and hence
\[
 u_{n,j}R_{ij}\eta
 \rightharpoonup
 u_jR_{ij}\eta
 \qquad\text{in }L^2(0,T;H^1(\mathbb R^2)).
\]
On the other hand, \eqref{flux:cap-compactness} gives
\[
 b_n\longrightarrow b
 \qquad\text{strongly in }L^2(0,T;H^{-1}(\mathbb R^2)).
\]
The \(H^{-1}(\mathbb R^2)\)--\(H^1(\mathbb R^2)\) duality pairings above therefore pass
to the limit, giving
\[
 \mathcal C_n
 \longrightarrow
 [u_j,R_{ij}]b
 \qquad\text{in }\mathcal D'((0,T)\times\mathbb R^2).
\]

Combining this identification with \eqref{flux:commutator-bound} gives
\begin{equation}\label{flux:commutator-limit}
 \mathcal C_n\rightharpoonup
   \mathcal C:=[u_j,R_{ij}]b
       \quad\text{in }L^2_tW^{1,192/97}_x.
\end{equation}

We now pair this weak limit with the strongly convergent momentum.
Let $\zeta\in C_c^\infty(\Omega)$ equal one near the spatial support of
$\phi$, and choose a ball $B$ such that $\operatorname{supp}\zeta\Subset B$.
By Proposition~\ref{mom:compact}, in particular \eqref{mom:exact},
\[
 \zeta m_n\to\zeta m
       \quad\text{strongly in }L^2_tW^{-1,192/95}_x.
\]

Since \(\phi\) is compactly supported in \(B\), multiplication by
\(\phi\) maps \(W^{1,192/97}(B)\) continuously into
\(W^{1,192/97}_0(B)\). Hence \eqref{flux:commutator-limit} gives
\[
 \phi\mathcal C_n\rightharpoonup\phi\mathcal C
 \qquad\text{in }L^2(0,T;W^{1,192/97}_0(B)).
\]
The strong--weak pairing can therefore be written as
\begin{align*}
 \int_Q\phi m_n\cdot\mathcal C_n-
      \int_Q\phi m\cdot\mathcal C
 ={}&\langle\zeta(m_n-m),\phi\mathcal C_n\rangle\\
    &+\langle\zeta m,\phi(\mathcal C_n-\mathcal C)\rangle
       \longrightarrow0.
\end{align*}
This passes the principal commutator term of \eqref{flux:full}, with
dual exponents \(192/95\) and \(192/97\).

\noindent\textbf{The source-potential term.}
The same pairing applies to the source potential. By
\eqref{flux:limiting-cap} and the local potential estimate,
\[
 Ag_n\rightharpoonup Ag
 \qquad\text{in }L^2(0,T;H^1(B)),
\]
and therefore also
\[
 Ag_n\rightharpoonup Ag
 \qquad\text{in }L^2(0,T;W^{1,192/97}(B)).
\]
Since \(\phi\) is compactly supported in \(B\), multiplication by \(\phi\) yields
\[
 \phi Ag_n\rightharpoonup\phi Ag
 \qquad\text{in }L^2(0,T;W^{1,192/97}_0(B)).
\]
With the same cutoff \(\zeta\) and momentum convergence
\eqref{mom:exact}, we obtain
\[
\begin{aligned}
 \int_Q\phi m_n\cdot Ag_n-\int_Q\phi m\cdot Ag
 ={}&\langle\zeta(m_n-m),\phi Ag_n\rangle\\
    &+\langle\zeta m,\phi(Ag_n-Ag)\rangle
       \longrightarrow0.
\end{aligned}
\]
Hence
\begin{equation}\label{flux:source-passage}
 \int_Q\phi m_n\cdot Ag_n\longrightarrow
       \int_Q\phi m\cdot Ag .
\end{equation}
The source-potential term of \eqref{flux:full} thus passes by the same
duality as the commutator term. Throughout this passage,
\[
 g=-\chi S_k+t_ku\cdot\nabla\chi,
\]
and the source \(S_k\) has not been replaced by
\(a_k(\rho)\operatorname{div}u\).

\noindent\textbf{Remaining cutoff and stress terms.}
The terms containing the
undifferentiated potential follow from its uniform convergence. By
\eqref{flux:potential-compactness}, $A_n\to A$ uniformly on every
compact subset of $Q$. Together with the already identified weak limits
of $m_n$, $F_n$, and $\mathbb S_n$, this gives
\[
 \int_Q(\partial_t\phi)m_n\cdot A_n
      \longrightarrow \int_Q(\partial_t\phi)m\cdot A,
\]
\[
 \int_Q(\partial_j\phi)F_{n,ij}A_{n,i}
      \longrightarrow \int_Q(\partial_j\phi)F_{ij}A_i,
 \qquad
 \int_Q\mathbb S_{n,ij}(\partial_j\phi)A_{n,i}
      \longrightarrow \int_Q\mathbb S_{ij}(u)(\partial_j\phi)A_i.
\]
The two cutoff terms from the stress formula, in the last line of
\eqref{flux:full}, use cap compactness. For the first one, \eqref{flux:cap-compactness} and
$u_n\rightharpoonup u$ in $L^2_tH^1_x$ give
\[
 \int_Q u_{n,i}(\partial_i\phi)b_n
 =\int_0^T\langle b_n,u_{n,i}\partial_i\phi\rangle\,dt
 \longrightarrow
 \int_0^T\langle b,u_i\partial_i\phi\rangle\,dt.
\]
For the Riesz-transform term, self-adjointness gives
\[
 \int_Q u_{n,i}(\partial_j\phi)R_{ij}b_n
 =\int_0^T\langle b_n,R_{ij}(u_{n,i}\partial_j\phi)\rangle\,dt.
\]
Since
\[
 R_{ij}(u_{n,i}\partial_j\phi)
 \rightharpoonup R_{ij}(u_i\partial_j\phi)
 \qquad\text{in }L^2_tH^1_x,
\]
the strong $L^2_tH^{-1}_x$ convergence in
\eqref{flux:cap-compactness} yields
\[
 \int_Q u_{n,i}(\partial_j\phi)R_{ij}b_n
 \longrightarrow
 \int_Q u_i(\partial_j\phi)R_{ij}b.
\]

\noindent\textbf{Pressure and artificial pressure.}
For the pressure terms, we use the local artificial-pressure
convergence from Lemma~\ref{mom:artificial}, in
particular \eqref{mom:AP},
\[
 \delta_n\rho_n^\beta\longrightarrow0
 \qquad\text{strongly in }L^1((0,T)\times D)
\]
for every $D\Subset\Omega$. Hence, since $|b_n|\le2k$,
\[
 \delta_n\rho_n^\beta b_n\longrightarrow0
 \qquad\text{in }L^1((0,T)\times D).
\]
For the pressure cutoff term in \eqref{flux:full}, write
\[
 \int_Q p_n\nabla\phi\cdot A_n
 =\int_Q \rho_n\nabla\phi\cdot A_n
  +\int_Q\delta_n\rho_n^\beta\nabla\phi\cdot A_n.
\]
The first term converges to
$\int_Q\rho\nabla\phi\cdot A$ by the weak $L^1$ convergence of
$\rho_n$ and the uniform convergence of $A_n$ on
$\operatorname{supp}\nabla\phi$; the second tends to zero by
\eqref{mom:AP} and the same local uniform bound. Thus
\[
 \int_Q p_n\nabla\phi\cdot A_n
 \longrightarrow \int_Q\rho\nabla\phi\cdot A.
\]
For the principal pressure--viscous term on the left-hand side of
\eqref{flux:full}, the relations in \eqref{flux:weak-caps}, the fact
that $b_n=q_n$ on $\operatorname{supp}\phi$, and the local artificial-
pressure convergence above give
\begin{equation}\label{flux:left-passage}
 \int_Q\phi(p_n-\nu d_n)b_n
 \longrightarrow
 \int_Q\phi(P_k-\nu h_k).
\end{equation}




\noindent\textbf{Comparison with the limiting momentum equation.}
The preceding passages and Lemma~\ref{flux:nonlocal-lemma}
give the limit of \eqref{flux:full}. We compare it with the identity
obtained by testing the limiting momentum equation with $\phi A$.

The mollification argument in Lemma~\ref{flux:testing-lemma}
justifies this test. We compute $\partial_t A$ from
\eqref{flux:limiting-cap} and repeat the expansions in
\eqref{flux:inertia}--\eqref{flux:stress}, with $p=\rho$ and
$\mathbb S=\mathbb S(u)$. The resulting right-hand side is
exactly the limit obtained above. In particular, both identities
contain the same source-potential term, with
$g=-\chi S_k+t_ku\cdot\nabla\chi$. This term therefore cancels
in the comparison, without any identification of $S_k$.

Since $b=\chi t_k$ and $\chi=1$ on $\operatorname{supp}\phi$,
the principal pressure--viscous term of the limiting identity is
\[
 \int_Q\phi\bigl(\rho-\nu\operatorname{div}u\bigr)b
 =
 \int_Q\phi\bigl(\rho-\nu\operatorname{div}u\bigr)t_k.
\]
Comparison with \eqref{flux:left-passage} thus gives
\[
 \int_Q\phi(P_k-\nu h_k)
 =
 \int_Q\phi\bigl(\rho-\nu\operatorname{div}u\bigr)t_k
 \qquad\text{for every }\phi\in C_c^\infty(Q).
\]
This is \eqref{flux:identity} in $\mathcal D'(Q)$.
Since $P_k,\rho t_k\in L^1(Q)$ and
$h_k,t_k\operatorname{div}u\in L^2(Q)\subset L^1(Q)$,
the identity holds almost everywhere in $Q$.
This proves Proposition~\ref{flux:proposition}.

Let \(Y\) be an independent copy of \(X\). Expanding the products and using independence and identical distribution, we obtain
\begin{equation}
\label{name of Ck}
\begin{aligned}
\frac12\mathbb E\bigl[(X-Y)(T_k(X)-T_k(Y))\bigr]
&=\mathbb E[XT_k(X)]-\mathbb E X\,\mathbb E T_k(X)\\
&=\operatorname{Cov}(X,T_k(X)),
\end{aligned}

\[
\frac12\mathbb E|T_k(X)-T_k(Y)|^2
=\mathbb E|T_k(X)|^2-\bigl|\mathbb E T_k(X)\bigr|^2
=\operatorname{Var}(T_k(X)).
\]

\begin{align}
 C_k&=\frac12\mathbb E[(X-Y)(T_k(X)-T_k(Y))],\label{var:copies}\\
 V_k&=\frac12\mathbb E|T_k(X)-T_k(Y)|^2.
\end{align}
The integrand on the right-hand side of \eqref{var:copies} is integrable because \(T_k\) is bounded and \(\mathbb E X<\infty\).
 We denote that 
\[
 a=X-Y,\qquad b=T_k(X)-T_k(Y).
\]
Since $T_k$ is increasing, $ab\ge0$, while the one-Lipschitz property gives
$|b|\le|a|$. Hence, pointwise,
\(
 0\le b^2\le ab,
\)
and taking expectations yields $0\le V_k\le C_k$.

For $x>y$,
\[
 T_k(x)-T_k(y)=\int_y^x T'(s/k)\,ds\uparrow x-y.
\]
Indeed, concavity of $T$ makes $T'$ nonincreasing, so for each fixed
$s\ge0$ the quantity $T'(s/k)$ increases with $k$ and eventually equals
$1$. By symmetry, for every $x,y\ge0$,
\[
 (x-y)(T_k(x)-T_k(y))\uparrow |x-y|^2.
\]
Thus monotone convergence in \eqref{var:copies} gives
\[
 C_k\uparrow\frac12\mathbb E|X-Y|^2.
\]
If $\mathbb E X^2<\infty$, independence gives
\[
 \frac12\mathbb E|X-Y|^2=\mathbb E X^2-(\mathbb E X)^2=V.
\]

                               
If \(\mathbb E X^2=\infty\), then we have 
\(
\mathbb E|X-Y|^2=\infty.
\)
In fact, 
since \(X\) and \(Y\) are independent copies and \(\mathbb E X=\mathbb E Y=r<\infty\), one obtains that 
\[
\mathbb E(XY)=\mathbb E X\,\mathbb E Y=r^2<\infty.
\]
If instead \(\mathbb E|X-Y|^2<\infty\), then
we would obtain
\[
\mathbb E(X^2+Y^2)
=
\mathbb E|X-Y|^2+2\mathbb E(XY)
<\infty.
\]
But \(X\) and \(Y\) are identically distributed, so
\(
\mathbb E(X^2+Y^2)=2\mathbb E X^2=\infty.
\) Thus, this is
a contradiction to \(\mathbb E X^2=\infty\). Hence we have
\(
\mathbb E|X-Y|^2=\infty.
\)

We next show that \(V=\infty\). Indeed, if
\(
V=\mathbb E|X-r|^2<\infty,
\)
then
\[\mathbb E
X^2=\mathbb E( |(X-r)+r|^2)
\le \mathbb E ( 2|X-r|^2+2r^2)\leq 2V+2r^2<+\infty,
\]
again a contradiction. Thus \(V=\infty\). Consequently,
\[
\frac12\mathbb E|X-Y|^2=V=\infty,
\]
and \eqref{var:monotone} holds also in the extended sense.


                          






Finally, concavity of \(T_k\) and Jensen's inequality give
\[
t_k=\mathbb E T_k(X)\le T_k(\mathbb E X)=T_k(r),
\]
and hence
\[
T_k(r)-t_k
=\mathbb E\bigl[T_k(r)-T_k(X)\bigr]\ge0.
\]
Applying Jensen's inequality to the convex function \(z\mapsto |z|^2\), and then using the one-Lipschitz property of \(T_k\), we obtain
\[
|T_k(r)-t_k|^2
\le \mathbb E|T_k(r)-T_k(X)|^2
\le \mathbb E|r-X|^2=V.
\]

<!-- lemma-8: body_md -->
\label{close:globalcap}
The zero extensions of $t_K,u,S_K$ satisfy
\begin{equation}\label{close:capequation}
 \partial_t t_K+\operatorname{div}(t_Ku)=-S_K,
 \qquad t_K|_{t=0}=T_K(\rho_0).
\end{equation}
For every smooth scalar test $\psi$ up to the spatial boundary,
vanishing near $t=T$,
\begin{equation}\label{close:capweak}
 \int_Q(t_K\partial_t\psi+t_Ku\cdot\nabla\psi-S_K\psi)
 +\int_\Omega T_K(\rho_0)\psi(0)=0.
\end{equation}
On any ball containing $\overline\Omega$, the zero-extended caps
$q_{K,n}:=T_K(\rho_n)$ converge to $t_K$ in $C([0,T];H^{-1})$.

<!-- proof-15: body_md -->
We first globalize the approximate cap equation. Let
$d_\partial(x)=\operatorname{dist}(x,\partial\Omega)$ and choose
$\chi_a=0$ for $d_\partial<a$, $\chi_a=1$ for $d_\partial>2a$,
with $|\nabla\chi_a|\le C/d_\partial$. Hardy's inequality gives,
for bounded $w$ and $v\in L^2_tW^{1,2}_0(\Omega)$,
\begin{equation}\label{close:hardy}
 \int_Q|wv\cdot\nabla\chi_a|
 \le C\|w\|_\infty
 |(0,T)\times\{d_\partial<2a\}|^{1/2}
 \|v/d_\partial\|_{L^2(Q)}\longrightarrow0,
\end{equation}
as $a\to 0$. At fixed $K$, this convergence is uniform for
$(w,v)=(q_{K,n},u_n)$. Recall that
\[
 \partial_t q_{K,n}+\operatorname{div}(q_{K,n}u_n)
 =-a_K(\rho_n)\operatorname{div}u_n.
\]
Testing this equation with $\chi_a\psi$, the only boundary-layer term is
\[
 \int_Q q_{K,n}u_n\cdot\nabla\chi_a\,\psi,
\]
which tends to zero uniformly in $n$ by \eqref{close:hardy}. All
remaining terms pass to the limit as $a\to 0$, yielding the global
approximate cap equation with its initial term.

We next establish compactness of the caps. After zero extension to a
fixed ball containing $\overline\Omega$, the sequence $(q_{K,n})_n$ is
bounded in $L^\infty(0,T;L^2)$ and has fixed bounded support. Moreover,
both $q_{K,n}u_n$ and
$a_K(\rho_n)\operatorname{div}u_n$ are uniformly bounded in $L^2(Q)$,
so
\[
 \|\partial_t q_{K,n}\|_{L^2(0,T;H^{-1})}\le C_K.
\]
Consequently, for $0\le s<t\le T$,
\[
 \|q_{K,n}(t)-q_{K,n}(s)\|_{H^{-1}}
 \le C_K|t-s|^{1/2}.
\]
Since $L^2$ is compactly embedded into $H^{-1}$ on the auxiliary ball,
Arzel\`a--Ascoli yields relative compactness in
$C([0,T];H^{-1})$. The weak-$*$ limit of $q_{K,n}=T_K(\rho_n)$ is
$t_K=\mathbb E T_K(X)$, hence
\[
 q_{K,n}\longrightarrow t_K
 \qquad\text{in }C([0,T];H^{-1}),\text{ as } n\to \infty.
\]
Pairing this strong cap convergence with the weak
$L^2_tH^1_x$ convergence of $u_n$ gives
$q_{K,n}u_n\to t_Ku$ in distributions as $n\to\infty.$

It remains to identify the initial cap trace. Since
$q_{K,n}(0)=T_K(\rho_{0,n})$, the strong convergence of $q_{K,n}$ in
$C([0,T];H^{-1})$ gives, as $n\to\infty$, we have 
\[
 T_K(\rho_{0,n})\longrightarrow t_K(0)
 \qquad\text{in }H^{-1}.
\]
On the other hand,
\[
 \|T_K(\rho_{0,n})-T_K(\rho_0)\|_{L^p}^p
 \le(4K)^{p-1}\|\rho_{0,n}-\rho_0\|_{L^1}\longrightarrow0
 \quad(1\le p<\infty),
\]
so $t_K(0)=T_K(\rho_0)$ in $H^{-1}$. Passing $n\to\infty$ in the
global approximate cap equation therefore gives \eqref{close:capweak}
and hence \eqref{close:capequation}. The initial trace is thus obtained
without invoking any nonlinear trace for the density Young measure.

 \qquad\text{for }s\ge3k.

representatives on $[0,T]$, with the stated initial values.

Fix an integer $k\ge1$. By the linear growth of $L_k$ and the uniform
mass bound,
\[
 \sup_n\|\ell_{k,n}\|_{L^\infty(0,T)}\le C_k.
\]
Moreover, \eqref{close:entropyapprox} and $0\le T_k\le2k$ give

 |\ell_{k,n}'(t)|

\(
 H^1(0,T)\Subset C([0,T]),

for the  fixed $k$ there exists a subsequence $n_j=n_j(k)$ and a

 \ell_{k,n_j}\longrightarrow\overline\ell_k

Let

 D_k(t):=\overline\ell_k(t)-\ell_k(t).

By Lemma~\ref{close:initialdefect}, the definition of $\ell_k$, and

 D_k(t)
 =\int_\Omega\bigl(\mathbb E L_k(X)-L_k(\rho)\bigr)\,dx
 \ge0
 \qquad\text{for a.e. }t,
\]
since $\mathbb E X=\rho$. Both $\overline\ell_k$ and $\ell_k$ belong to
$H^1(0,T)$, hence $D_k$ is continuous on $[0,T]$. Its a.e.
nonnegativity therefore implies $D_k(t)\ge0$ for every $t\in[0,T]$,
and Lemma~\ref{close:initialdefect} gives
\(
 D_k(0)=0.
\)

We next derive the evolution of the defect. By
\eqref{close:barentropy} and \eqref{close:entropylimit}, for almost every
$t\in(0,T)$,
\[
\begin{aligned}
 D_k'(t)
 &=\overline\ell_k'(t)-\ell_k'(t)\\
 &=-\int_\Omega h_k\,dx
   +\int_\Omega T_k(\rho)\operatorname{div}u\,dx.
\end{aligned}
\]
The fixed-cap flux identity \eqref{flux:identity} reads
\[
 C_k=\nu\bigl(h_k-t_k\operatorname{div}u\bigr),
\]
and therefore
\[
 h_k=t_k\operatorname{div}u+\frac1\nu C_k.
\]
Substituting this expression for $h_k$ gives
\[
\begin{aligned}
 D_k'(t)
 &=-\int_\Omega
    \left(t_k\operatorname{div}u+\frac1\nu C_k\right)dx
   +\int_\Omega T_k(\rho)\operatorname{div}u\,dx\\
 &=-\frac1\nu\int_\Omega C_k\,dx
   +\int_\Omega\bigl(T_k(\rho)-t_k\bigr)
                  \operatorname{div}u\,dx.
\end{aligned}
\]
Thus, with
\(
 e_k:=T_k(\rho)-t_k,
\)
we have
\begin{equation}\label{close:defect}
 D_k'(t)
 =-\frac1\nu\int_\Omega C_k\,dx
  +\int_\Omega e_k\operatorname{div}u\,dx
\end{equation}
for almost every $t\in(0,T)$.

Integrating \eqref{close:defect} from $0$ to $t$ and using $D_k(0)=0$
yields
\[
 D_k(t)+\frac1\nu\int_0^t\!\int_\Omega C_k
 =\int_0^t\!\int_\Omega e_k\operatorname{div}u.
\]
Since $D_k(t)\ge0$ and $C_k\ge0$ by
Lemma~\ref{var:abstract}, the left-hand side is nonnegative. Hence,

\begin{equation}\label{close:defectintegrated}

 0
 &\le D_k(t)+\frac1\nu\int_0^t\!\int_\Omega C_k=\int_0^t\!\int_\Omega e_k\operatorname{div}u\le

\end{aligned}

\[

 \longrightarrow0,
\]

\[

 \qquad\text{in measure on }Q.

\[

 \int_\Omega \rho_n\log(1+\rho_n)\,dx\le C.

\begin{equation}\label{close:UI}
\begin{aligned}
 \sup_n\int_{\{(t,x)\in Q:\,\rho_n>R\}}\rho_n\,dx\,dt

 \frac{1}{\log(1+R)}
 \sup_n\int_Q\rho_n\log(1+\rho_n)\,dx\,dt\\
 &\le \frac{CT}{\log(1+R)}
 \longrightarrow0
 \qquad(R\to\infty).
\end{aligned}

 \rho_n\longrightarrow\rho
 \qquad\text{strongly in }L^1(Q),
\]
which is \eqref{close:strongLone}.

We use the approximation and effective-flux framework of

planar exponents specified below. The source treats three-dimensional
$C^{2+\eta}$ domains; the planar exponents, logarithmic potential,
$C^2$ parabolic input and initial-energy passages are supplied here. At fixed
$\delta$, the pressure is increasing and
$\lim_{z\to\infty}p_\delta(z)z^{-\beta}=\delta>0$; thus the general-pressure
extension in \cite[Section~7.5, p.~205, and Example~7.1, p.~207]{Lions} also corroborates the

\emph{Galerkin approximation and energy.}
Choose $0<c_*\le\inf r_0$ and $\chi_\varepsilon\in C_c^\infty(\Omega)$,
$0\le\chi_\varepsilon\le1$, tending pointwise to one. The data
$r_{0,\varepsilon}=c_*+\chi_\varepsilon(r_0-c_*)$ are constant near the
boundary and converge to $r_0$ in every finite $L^q$ and in energy.
Let $X_N$ be nested finite-dimensional spaces of compactly supported
smooth vector fields, containing $v_0$ and dense in every finite
$W^{1,q}_0$. Solve
\begin{align}
 \partial_t r_N+\operatorname{div}(r_Nv_N)
 &=\varepsilon\Delta r_N,\qquad
 \partial_{\boldsymbol n}r_N=0,\quad r_N(0)=r_{0,\varepsilon},
 \label{app:parabolic}\\
 \frac{d}{dt}\int_\Omega r_Nv_N\cdot w
 &=\int_\Omega\left(r_Nv_N\otimes v_N:\nabla w
                  +p_\delta(r_N)\operatorname{div}w
                  -\mathbb S(v_N):\nabla w\right)\notag\\
 &\quad-\varepsilon\int_\Omega(\nabla r_N\cdot\nabla)v_N\cdot w,
 \qquad w\in X_N,\quad v_N(0)=v_0.\label{app:Galerkin}
\end{align}
Local solvability uses finite-exponent Neumann regularity;
for the scalar Laplacian on $C^2$ domains this follows from
Denk--Hieber--Pr\"uss, Theorem~2.1.\footnote{R. Denk, M. Hieber and
J. Pr\"uss, \emph{Optimal $L_p$--$L_q$-regularity for parabolic problems
with inhomogeneous boundary data}, Konstanzer Schriften in Mathematik
und Informatik, no.~205 (2005), Theorem~2.1, p.~7, and Proposition~3.3,
p.~9:
\href{https://d-nb.info/1093431571/34}{original preprint}.}
The scalar order-two operator is normally elliptic, the normal Neumann
operator satisfies the complementing condition, and the smooth
boundary-compatible data satisfy its trace conditions.
Comparison keeps $r_N$ positive. With
$\widehat H_\delta(r)=r\log r+\delta r^\beta/(\beta-1)$,
$r\widehat H_\delta'-\widehat H_\delta=p_\delta$ and
$\widehat H_\delta''=r^{-1}+\delta\beta r^{\beta-2}$.
The Galerkin diffusion correction cancels the kinetic cross term,
giving
\begin{equation}\label{app:exactenergy}
 \frac{d}{dt}\widehat E_N
 +\int_\Omega\mathbb S(v_N):\nabla v_N
 +\varepsilon\int_\Omega
       (r_N^{-1}+\delta\beta r_N^{\beta-2})|\nabla r_N|^2=0,
\end{equation}
where $\widehat E_N=\int(r_N|v_N|^2/2+\widehat H_\delta(r_N))$.
Coercivity \eqref{app:coercivity} and finite-dimensional norm
 equivalence control $\int\|\operatorname{div}v_N\|_\infty$ on each finite
time interval; comparison keeps the mass matrix positive and energy controls
its velocity coefficients. The approximate solution therefore continues.

\emph{The limit $N\to\infty$ at fixed $\varepsilon,\delta$.}
Energy gives $r_N\in L^\infty_tL^\beta_x$, $v_N\in L^2_tH^1_0$,
$\nabla r_N\in L^2(Q_T)$ and
$r_N^{\beta/2}\in L^\infty_tL^2_x\cap L^2_tH^1_x$, hence
$r_N\in L^{2\beta}(Q_T)$ by the planar space-time estimate.
For $a=1/2-1/\beta$, spatial H\"older yields
\[
 \|r_Nv_N\|_2
 \le\|r_N|v_N|^2\|_1^a\|r_N\|_\beta^{1-a}
       \|v_N\|_{4\beta/(\beta-2)}^{2/\beta}.
\]
Thus $r_Nv_N\in L^\beta_tL^2_x$. The continuity equation gives
$\partial_t r_N\in L^2_tH^{-1}_x$, and compactness implies
\begin{equation}\label{app:firstdensity}
 r_N\to r_\varepsilon\text{ in }L^q(Q_T),\ q<2\beta,
 \qquad p_\delta(r_N)\to p_\delta(r_\varepsilon)\text{ in }L^1(Q_T).
\end{equation}
For the negative-space heat estimate, put $X=(H^1(\Omega))^{\prime}$
and define $A_\varepsilon:H^1(\Omega)\to X$ by
\[
 \langle A_\varepsilon w,\phi\rangle
 =\int_\Omega(w\phi+\varepsilon\nabla w\cdot\nabla\phi),
 \qquad \phi\in H^1(\Omega).
\]
As an operator in $X$, $A_\varepsilon=I-\varepsilon\Delta_N$ has domain
$H^1(\Omega)$ and is positive self-adjoint in the equivalent Hilbert
norm $\|f\|_X=\|(I-\Delta_N)^{-1/2}f\|_2$, defined spectrally by the
Neumann Laplacian. It is therefore invertible and $R$-sectorial of
angle zero. Regard \eqref{app:parabolic} as
\[
 \partial_t r_N+A_\varepsilon r_N
 =r_N-\operatorname{div}(r_Nv_N)\in L^\beta(0,T;X).
\]
Here divergence is defined by pairing with $-\nabla\phi$; the forcing
bound follows from $r_Nv_N\in L^\beta_tL^2_x$ and
$r_N\in L^\infty_tL^\beta_x$. The smooth datum satisfies
$r_{0,\varepsilon}\in H^1(\Omega)\subset
(X,H^1(\Omega))_{1-1/\beta,\beta}$.
Denk--Hieber--Pr\"uss, Proposition~3.3, p.~9 of the same preprint,
applied with time exponent $\beta$ (extending the forcing by zero
past $T$), yields
\[
 \begin{aligned}
 &\|\partial_t r_N\|_{L^\beta(0,T;X)}
   +\|r_N\|_{L^\beta(0,T;H^1)}\\
 &\quad\le C_{\varepsilon,\beta,T,\Omega}\bigl(
 \|r_Nv_N\|_{L^\beta_tL^2_x}+\|r_N\|_{L^\beta_tX}
 +\|r_{0,\varepsilon}\|_{(X,H^1)_{1-1/\beta,\beta}}\bigr).
 \end{aligned}
\]
Thus $\nabla r_N\in L^\beta_tL^2_x$, with a bound depending on
$\varepsilon,\delta,\beta,T,\Omega$ and the prescribed data,
independently of $N$. Consequently,
$-v_N\cdot\nabla r_N-r_N\operatorname{div}v_N\in L^{4/3}(Q_T)$;
here use $v_N\in L^2_tL^4_x$, $r_N\in L^{2\beta}(Q_T)$ and
$\beta\ge6$. The preceding scalar $C^2$ theorem at exponent $4/3$
then gives $\partial_t r_N,\Delta r_N\in L^{4/3}(Q_T)$.

Momentum ranges are uniformly bounded in
$L^{2\beta/(\beta+1)}\Subset H^{-1}$. For each fixed Galerkin test,
\eqref{app:Galerkin} bounds its scalar time derivative in
$L^{2\beta/(\beta+2)}$, whose exponent exceeds one.
Coordinate Ascoli and compactness of the ranges therefore give
momentum convergence in $C_tH^{-1}$; no norm bound for Galerkin
projections is required. Strong--weak pairing identifies convection.
The density-square balance
\[
 \frac12\frac{d}{dt}\|r_N\|_2^2+
 \varepsilon\|\nabla r_N\|_2^2
 =-\frac12\int r_N^2\operatorname{div}v_N
\]
and $r_N^2\to r_\varepsilon^2$ in $L^2(Q_T)$ give convergence of the
weighted gradient norms. Hence $\nabla r_N\to\nabla r_\varepsilon$
strongly on $Q_{T'}$, $T'<T$, identifying the diffusion correction.
The construction on a larger time interval covers the prescribed $Q_T$.

To retain the initial energy, integrate \eqref{app:exactenergy} with
$\theta\ge0$ and $\Theta(s)=\int_s^T\theta(t)dt$:
\begin{equation}\label{app:weightedinner}
 \int_0^T\theta\widehat E_N
 +\int_0^T\Theta(D_N+\varepsilon A_N)
 =\widehat E_{0,\varepsilon}\int_0^T\theta,
\end{equation}

$A_N=4\int|\nabla\sqrt{r_N}|^2+(4\delta/\beta)\int|\nabla r_N^{\beta/2}|^2$.
Convex and quadratic lower semicontinuity gives the parabolic energy
inequality with the same initial right-hand side.

\emph{Removal of density diffusion at fixed $\delta$.}
The Bogovski\u\i\ test with
$\mathcal B[r_\varepsilon-\langle r_\varepsilon\rangle]$,
as in \cite[Lemma~3.1]{FNP}, gives
\begin{equation}\label{app:fixedgain}
 \int_{Q_T}p_\delta(r_\varepsilon)r_\varepsilon\le C_\delta,
 \qquad\sqrt\varepsilon\|\nabla r_\varepsilon\|_2\le C_\delta.
\end{equation}
The required planar bounds are $\mathcal B[r_\varepsilon-
\langle r_\varepsilon\rangle]\in L^\infty$ since $\beta>2$, and
$r_\varepsilon v_\varepsilon\in L^2(Q_T)$; inertia is controlled by
$\int_0^T\|r_\varepsilon\|_3^2\|v_\varepsilon\|_6^2dt$.
The two diffusion terms are $O_\delta(\sqrt\varepsilon)$.
Time cutoffs of bounded total variation cover $(0,T)$, and
\[
 \varepsilon\nabla r_\varepsilon\to0\text{ in }L^2,
 \qquad\varepsilon(\nabla r_\varepsilon\cdot\nabla)v_\varepsilon
       \to0\text{ in }L^1.
\]

Write $\overline p$ for the weak $L^{(\beta+1)/\beta}$ pressure limit.
The product $p_\delta(r_\varepsilon)r_\varepsilon$ initially has a
locally weak-* measure limit. The full-density effective-flux argument

\begin{equation}\label{app:innerflux}
 \overline{p_\delta(r)r}-\nu\,\overline{r\operatorname{div}v}
 =\overline p\,r-\nu r\operatorname{div}v,
 \qquad\nu=2\mu+\lambda.
\end{equation}
Its div--curl exponent $2\beta/(\beta+3)\ge4/3$ is compactly embedded

compact because $\beta>2$. Both diffusion errors in the potential
test vanish by \eqref{app:fixedgain} and the $L^2$ boundedness of

The limiting continuity equation is renormalized since $r\in L^2$
and $\nabla v\in L^2$. Its initial value is retained by the extension

\[
 \int_\Omega r_\varepsilon\log r_\varepsilon(\tau)
 +\int_0^\tau\!\int r_\varepsilon\operatorname{div}v_\varepsilon
 \le\int_\Omega r_{0,\varepsilon}\log r_{0,\varepsilon}.
\]
Convex terminal lower semicontinuity and the limiting logarithmic
identity therefore give
\[
 \limsup_{\varepsilon\downarrow0}\int_0^\tau\!\int
 r_\varepsilon\operatorname{div}v_\varepsilon
 \le\int_0^\tau\!\int r\operatorname{div}v
\]
at admissible terminal times. Monotonicity makes the pressure-product
defect nonnegative. Increasing interior cutoffs in
\eqref{app:innerflux}, with the uniform
$L^{2\beta/(\beta+2)}$ bound on
$r_\varepsilon\operatorname{div}v_\varepsilon$, and this limsup
inequality yield the Minty identification
$\overline p=p_\delta(r)$; see \cite[Section~3.5]{FNP}.
This exhaustion uses positivity rather than uniform integrability

$\int r_\varepsilon^\beta\to\int r^\beta$. Uniform convexity then yields
\[
 r_\varepsilon\to r\text{ in }L^\beta(Q_T),\qquad
 p_\delta(r_\varepsilon)\to p_\delta(r)\text{ in }L^1(Q_T).

The weak equations retain $r_0,r_0v_0$. The quadratic renormalization

\[

 \le R^{4-\beta}\|r\|_{L^\beta(Q_T)}^\beta\to0

for the source paired with $\operatorname{div}v\in L^2$.
For transport, $r^2\mathbf1_{\{r>R\}}\to0$ in
$L^2_tL^{\beta/2}_x$ by time dominated convergence, paired with
$v\in L^2_tL^{\beta/(\beta-2)}_x$. The smooth bounded initial density
makes the initial tails vanish. The logarithm is regularized at zero;
the stated $C^2([0,\infty))$ class is already regular there.

Finally pass the parabolic energy inequality with the weights in
\eqref{app:weightedinner}, discard the nonnegative diffusion term,

representatives follow from their fixed high-power bounds and weak
equations with initial terms. Nested extractions on the intervals
$(0,L)$, $L=1,2,\ldots$, give the global solution, with all the stated properties.
