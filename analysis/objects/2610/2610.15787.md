<!-- th-1: body_md -->
\label{thm:main}
The optimal constants satisfy
\begin{align}
 C_s
 &\leq (18\pi)^{1/3}
 =\left(\frac{8}{3\pi}\right)^{1/3}
   \frac32\lvert \mathbb{S}^3\rvert^{1/3}
 <\frac32\lvert \mathbb{S}^3\rvert^{1/3},
 \label{eq:main-Cs}\\
 C_v
 &\leq C_v^{\mathcal{T}}
 \leq 2\left(\frac{80\pi^2}{49}\right)^{1/3}
 =2\left(\frac{40}{49}\right)^{1/3}\lvert \mathbb{S}^3\rvert^{1/3}
 <2\lvert \mathbb{S}^3\rvert^{1/3}.
 \label{eq:main-Cv}
\end{align}
Equivalently,
\begin{equation}\label{eq:main-power}
 S_{3/2,\mathrm{curl}}
 \leq S_{3/2,\mathrm{curl}}^{\mathcal{T}}
 \leq \frac{8\sqrt{10}}7\pi
 <4\pi.
\end{equation}
Consequently, neither $\psi_{\mathrm{LY}}$ nor $A_{\mathrm{LY}}$ is a global
optimizer of its respective inequality.

\label{lem:gauge}
Let $n\geq2$, $1<q<\infty$, and $A\in L^q(\mathbb{R}^n;\mathbb{R}^n)$. If
\begin{equation}\label{eq:nonlinear-gauge-q}
 \nabla\cdot(\lvert A\rvert^{q-2}A)=0
\end{equation}

For a real-valued weakly differentiable function $v$ and a constant unit

\[
 \lvert D_d(v\eta)\rvert^2
 =\langle \eta,(\boldsymbol{\gamma}\cdot\nabla v)^2\eta\rangle

 u_d^{q_d}=(1+r^{q_d})^{-d},

 \lvert u_d'(r)\rvert

\begin{equation}\label{eq:radial-integration-d}
 \int_{\mathbb{R}^d}F(\lvert x\rvert)\,\mathrm{d} x
 =\lvert \mathbb{S}^{d-1}\rvert\int_0^\infty F(r)r^{d-1}\,\mathrm{d} r.
\end{equation}
In the first integral below, set $t=r^{q_d}$. Since
$r^{d-1}\,\mathrm{d} r=q_d^{-1}t^{d/q_d-1}\,\mathrm{d} t$ and

 \int_{\mathbb{R}^d}u_d^{q_d}\,\mathrm{d} x
 &=\frac{\lvert \mathbb{S}^{d-1}\rvert}{q_d}

   \mathrm B\left(\frac{d-1}{2},\frac{d+1}{2}\right)
   =I_d^{\mathrm{AT}},\\

   \int_0^\infty

For the second computation we used
$p_d(q_d-1)=q_d$ and $p_d(d+1)/2=d$.

\eqref{eq:AT-quotient-d}.

   \left(\frac{\Gamma((d-1)/2)}{\Gamma(d/2)}\right)^2.

spaces in \eqref{eq:FL-sharp-spinor-general}.

\label{lem:vector-conformal-invariance}
Let $d\geq3$ and let $F$ be a M\"obius transformation of
$\mathbb{R}^d\cup\{\infty\}$. Identifying vector fields with one-forms, define
\[
 (F^*A)(x):=DF(x)^T A(F(x))
\]
away from the possible pole of $F$. Then $F^*$ maps $Y_d$ bijectively
onto itself and preserves both $\lVert \nabla\wedge A\rVert_{L^{d/2}}$ and
$\VERT A\VERT_{d}$. In particular, it preserves $R_d^v$ and $\mathcal Q_d^v$
whenever $\nabla\wedge A\neq0$.
<!-- end lem-3 -->
<!-- proof-4: body_md -->
Write $DF=\lambda O$, with $O$ orthogonal and $\lambda>0$.
Pullback multiplies the pointwise norm of a $k$-form by $\lambda^k$,
whereas $\lvert \det DF\rvert=\lambda^d$. Change of variables therefore
preserves the $L^d$ norm of one-forms and the $L^{d/2}$ norm of two-forms.
Exterior differentiation commutes with pullback away from the pole.
The identity extends distributionally across the pole: with balls
centred there and a cutoff supported in $B_{2\varepsilon}$, equal to
one on $B_\varepsilon$, and satisfying
$\lvert \nabla\chi_\varepsilon\rvert\leq C/\varepsilon$, the additional term
against any fixed smooth test form is bounded by
\[
 C\lVert F^*A\rVert_{L^d(B_{2\varepsilon})}
   \lVert \nabla\chi_\varepsilon\rVert_{L^{d/(d-1)}}
 \leq C\varepsilon^{d-2}
   \lVert F^*A\rVert_{L^d(B_{2\varepsilon})}\longrightarrow0.
\]
Thus a pulled-back $L^d$ gradient remains distributionally closed on
$\mathbb{R}^d$. The distributional Poincar\'e lemma makes it a gradient of a
function in $\dot W^{1,d}(\mathbb{R}^d)$. Applying the same argument to $F^{-1}$
shows that pullback maps the gradient subspace onto itself.
Its $L^d$ isometry consequently preserves distance to that subspace,
which is precisely $\VERT A\VERT_{d}$.

 G_d:=\frac{\Gamma(1+d/2)\Gamma((d+1)/2)}{\Gamma(d+1/2)}

 b_d:=\frac{2(d-1)}{3d-2}.

form considered by Frank and Loss, so the theorem gives an explicit upper
bound for $S_d^v$ rather than the disproof of an optimizer conjecture.
Wang and Zhang had already established nonoptimality of the conformal
Loss--Yau--Dunne--Min family in every odd dimension by their spherical
second-variation argument \cite[Theorem~1.3]{WangZhangConstants}.
On the nonlinear-gauge slice
$\nabla\cdot(\lvert A\rvert^{d-2}A)=0$, their functional is related to our two
normalizations by
\[
 J_{d/2,1}(A)
 =\frac{\left(\int_{\mathbb{R}^d}\lvert \nabla\wedge A\rvert^{d/2}\,\mathrm{d} x\right)^2}
        {\int_{\mathbb{R}^d}\lvert A\rvert^d\,\mathrm{d} x}
 =(R_d^v(A))^d=(\mathcal Q_d^v(A))^2.
\]
Here Lemma~\ref{lem:gauge} identifies the denominator with
$\VERT A\VERT_{d}^d$; without the nonlinear-gauge condition, this identity
need not hold. The present theorem supplies a particularly simple
explicit competitor whose quotient is evaluated in closed form.
