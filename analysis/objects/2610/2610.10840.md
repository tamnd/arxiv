<!-- definition-1: body_md -->
\label{def:local_metric_entropy}
    For a set $K$, the $\delta$-packing number is the maximum cardinality $\mathcal{M}(\delta,K)$ of a set $\{\Sigma_1,...,\Sigma_M\} \subseteq K$ such that $\|\Sigma_i - \Sigma_j\|_F \geq \delta$ $\forall i\neq j$. Then for a fixed constant $c>0$ \textbf{the local metric entropy} for the set $K$ is $\mathcal{M}_K^{\operatorname{loc}}(\delta,c) = \sup_{\Sigma \in K} \mathcal{M}(\delta/c, B(\Sigma,\delta) \cap K)$. We refer to $\delta$ as the \textbf{local radius} and c as the \textbf{radius factor}. When the set $K$ is unambiguous, we may drop the $K$ and write $\mathcal{M}^{\operatorname{loc}}(\delta,c)$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:star_shaped}
    A set $K$ is called \textbf{star-shaped with center $k^\star$} if $\exists k^\star \in K$ such that $\forall k \in K$ and $\alpha \in [0,1]$, $\alpha k + (1-\alpha)k^\star \in K$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
The minimax estimation under adversarial corruption is given as \[\mathfrak{M}(X) = \inf_{\hat\Sigma} \sup_{\Sigma} \sup_{\mathcal{C}_\varepsilon} \mathbb{E} \| \hat\Sigma(\mathcal{C}_\varepsilon(\tilde{X})) - \Sigma \|_F^2 \]
    where $\mathcal{C}_\varepsilon$ gives the adversarial corruption scheme, $\Sigma$ is the true covariance of $X$ before corruption, and $\hat\Sigma$ is the estimator of $\Sigma$ given the corrupted data $\mathcal{C}_\varepsilon(\tilde{X})$. When the corruption rate is clear we drop the $\varepsilon$ and simply write $\mathcal{C}(\tilde{X})$.

Consider the tournament $ T(\delta ,\Sigma ,S) $ from \Cref{definition:tournament_T}. Assume $ \Sigma \in K' $ and denote $ i^*\in \mathop{ \arg\min }\limits_{i\in [M]} T(\delta ,\Sigma _i,S)  $. Also assume $ \frac{\delta^2}{\Upsilon^2} \geq N^{-1}\vee C_1^2\varepsilon^2\log^2(1/\varepsilon)  $, then
    \begin{align*}
        \mathbb{P}\left( \left\Vert \Sigma _{i^*} - \Sigma  \right\Vert \geq (C+1)\delta  \right) \leq M\exp\left\{  -C_5N\min\left(\frac{\delta^2}{\Upsilon^2}, \frac{\delta}{\Upsilon}\right) \right\} 
    \end{align*}

<!-- proof-2: body_md -->
Since we have that $ S=\{\Sigma _i\}_{i=1}^M $ is a $\delta$-covering of $ K'\ni \Sigma  $, we can without loss of generality assume $ \Sigma _1 $ be the point in the covering closest to $ \Sigma  $ and thus $ \Sigma \in B(\Sigma _1, \delta ) $. Denote $ T_i:= T(\delta ,\Sigma _i,S) $. We have by definition of $ T(\, \cdot \, ) $ that 
    \begin{align*}
        1\kern-4pt 1(\left\Vert \Sigma _{i^*} -\Sigma _1 \right\Vert \geq C\delta ) \leq 1\kern-4pt 1(\max\{ T_{i^*},T_1 \} \geq C\delta ) \leq 1\kern-4pt 1(T_1 \geq C\delta ) .
    \end{align*}
    Therefore, if $T_1 \geq C\delta$ then $\exists\Sigma_j$ such that $\Sigma_j \succ \Sigma_1$. Because $\frac{\delta^2}{\Upsilon^2} \geq N^{-1} \vee C_1^2\varepsilon^2\log^2(1/\varepsilon)$, we can apply \Cref{theorem:two_point_prob_bound}. Then, applying the union bound to the $M-1$ possible $\Sigma_j$
    \begin{align*}
        \mathbb{P}_\Sigma(\|\Sigma_{i^\star} - \Sigma_1\| \geq C\delta) &\leq \mathbb{P}_\Sigma(T_1 \geq C \delta) \leq (M-1) \exp\left\{  -C_5 N \min\left(\frac{\delta^2}{\Upsilon^2}, \frac{\delta}{\Upsilon}\right) \right\},
    \end{align*}
    and by the triangle inequality 
    \begin{align*}
        \mathbb{P}_\Sigma(\|\Sigma_{i^\star} - \Sigma\| \geq (C+1) \delta) \leq M\exp\left\{  -C_5 N \min\left(\frac{\delta^2}{\Upsilon^2}, \frac{\delta}{\Upsilon}\right) \right\}.
    \end{align*}

<!-- theorem-4: body_md -->
\label{theorem:minimax_rate}
    Let $\varepsilon \in (0,\frac{1}{32})$ and $N\gtrsim \sup_\delta \log \mathcal{M}_K^{\textrm{loc}}(2\delta,c)$.  Define
    \begin{align*}
        \delta^\star=\sup\left\{ \delta\geq 0:\, N \frac{\delta^2}{\Upsilon^2} \leq \log \mathcal{M}_K^{\textrm{loc}}(\delta,c) \right\}
    \end{align*}
    Then for sufficiently large $c$, the minimax error is bounded as
    \begin{align}
        \left(\varepsilon^2 \vee \delta^{\star2} \right)\wedge d^2 \lesssim \mathfrak{M}(X) \lesssim \left(\varepsilon^2\log^2(1/\varepsilon) \vee \delta^{\star2} \right)\wedge d^2
    \end{align}
<!-- end theorem-4 -->
<!-- remark-2: body_md -->
A single minimax rate is not achieved for this setting - the left and right bounds differ by a factor of $\log^2(1/\varepsilon)$. However, this is a small price to pay. For example, when $\varepsilon = 0$, $\varepsilon^2 = \epsilon^2\log^2(1/\varepsilon)=0$. Indeed, whenever $\varepsilon$ is a constant this difference is of course negligible as it is just a constant. If $\varepsilon \asymp 1/n$ then the rate only differs by a factor of $\log^2 n$ which is of course dominated by the $1/n^2$ in the rate. This will be explored further in \Cref{sec:examples}.

Therefore, $N \gtrsim n^2$ is sufficient for the sample size condition $N \gtrsim \sup_\delta \log \mathcal{M}_K^{\textrm{loc}}(\delta,2c)$.

\label[corollary]{cor:lq_ball}
    Consider the uniformity class of matrices with the $\ell_q$ ball with bounded eigenvalues, $\mathcal{U}_q$. In this class each column of the covariance matrix $\Sigma$ is in the $\ell_q$ ball for $q \in [0,1]$:

        \mathcal{U}_q(s_0(n)) \coloneqq \left\{ \Sigma: \max_i \sum_{j=1}^n |\sigma_{ij}|^q \leq s_0(n) , 0 < c_0 \leq \lambda_{\min}(\Sigma) \leq \lambda_{\max}(\Sigma) \leq c_1 \right\} \hspace{2mm}
    \end{align*} for $s_0(n) \geq c_0^q$, $N \gtrsim n^2$, $n \geq 2$, and $X \sim N(0,\Sigma)$ such that $\Sigma \in U_q(s_0(n))$.

    \begin{align*}
        \frac{1}{n} \mathfrak{M}(X) \lesssim \frac{1}{n} \varepsilon^2 \log^2(1/\varepsilon)\bigvee \left(s_0(n) \left(\frac{\log n}{N}\right)^{1-q/2} \bigwedge \frac{n}{N}\right).
    \end{align*}

We have that $\mathcal{U}_q$ is star-shaped around the $c_0 \mathbb{I}$ matrix by \Cref{lemma:Uq_star_shaped}

By \Cref{theorem:minimax_rate}, the sampling component of the rate for estimating $\Sigma$ is $\delta^{\star 2} \wedge d^2$, for \[\delta^\star = \sup\left\{ \delta\geq 0:\, \frac{N}{\Upsilon^2} \delta^2\leq \log M^{\operatorname{loc}}_{\mathcal{U}_q}(\delta,c) \right\}.\]
Now note that 
\begin{align*}
    \mathcal{U}_q(s_0(n)) &\subseteq \left\{ \|\Sigma\|_q^q \leq n s_0(n),  0 < c_0 \leq \lambda_{\min}(\Sigma) \leq \lambda_{\max}(\Sigma) \leq c_1 \right\} 
    \\ &\subseteq \left\{ \|\Sigma\|_q^q \leq n s_0(n) \right\}  \eqqcolon U^{+}_q(ns_0(n)) 
\end{align*} and thus 
\[\delta^\star \lesssim \sup\bigg\{\delta \geq 0 : N \delta^2 \leq \log M_{\mathcal{U}_q^+}^{\operatorname{loc}}(\delta,c)\bigg\}.\]
Further note that $U^{+}$ is the $\ell_q$ ball with dimension $n^2$ and radius $(ns_0(n))^{1/q}$. 

We now split into cases based on the value of $q$.

\medskip
\noindent
\textbf{Case 1: $q \in (0,1]$.}
We split this case further into two cases based on the relative magnitude of $s_0(n) \big(\frac{\log n}{N}\big)^{1-q/2} \lessgtr \frac{n}{N}$.
\\\textbf{Case 1A: $s_0(n) \big(\frac{\log n}{N}\big)^{1-q/2} \leq \frac{n}{N}$.}
By the argument of \textcite{neykov_minimax_2023} (see also \textcite{guedon_euclidean_2000}) we see that the log cardinality of a maximal $\delta-$packing set of $U_q^+(R)$ is upper-bounded up to constants depending on $B_1$ and $q$ by $(R^{-1}\delta)^{-\frac{2q}{2-q}}\log\left(n^2(R^{-1}\delta)^{\frac{2q}{2-q}}\right)$ for $R^{-1}\delta \gtrsim (n^2)^{1/2-1/q}$ and $n^2(R^{-1}\delta)^{2q/(2-q)} > B_1 > 1$. 
Define $\tilde\delta^2 = n s_0(n) \big(\frac{\log n}{N}\big)^{1-q/2}$. Note that 
\begin{align*}
    (R^{-1}\tilde\delta)^{-\frac{2q}{2-q}} &= \bigg(n^{1/2-1/q}s_0(n)^{1/2-1/q} \Big(\frac{\log n}{N}\Big)^{1/2-q/4} \bigg)^{-\frac{2q}{2-q}}
    \\ &= ns_0(n)\Big(\frac{\log n}{N}\Big)^{-q/2}.
\end{align*}
Let $z \coloneqq n^2\big(\tilde\delta (cR)^{-1}\big)^{\frac{2q}{2-q}}$ and let $B_1 > 1$ be a positive constant. Split further into cases based on $z \lessgtr B_1$.

\medskip
\noindent
\textbf{Case 1A.1: $z > B_1$.}
First see that \begin{align*}
\log \mathcal{M}_{\mathcal{U}_q}^{\operatorname{loc}}(\tilde\delta,c) &\lesssim (R^{-1}\tilde\delta/c)^{-\frac{2q}{2-q}}\log\left(n^2(R^{-1}\tilde\delta/c)^{\frac{2q}{2-q}}\right).
\end{align*}
Recall that we assume $s_0(n) \big(\frac{\log n}{N}\big)^{1-q/2} \leq \frac{n}{N}$ and $N \gtrsim n^2$ and apply \Cref{lemma:delta_properties} to see that $R^{-1}\tilde\delta \gtrsim n^{1-2/q}$ and
\begin{align*}
\log \mathcal{M}_{\mathcal{U}_q}^{\operatorname{loc}}(\tilde\delta,c) &\lesssim (R^{-1}\tilde\delta/c)^{-\frac{2q}{2-q}}\log\left(n^2(R^{-1}\tilde\delta/c)^{\frac{2q}{2-q}}\right) 
\\ &\lesssim (R^{-1}\tilde\delta)^{-\frac{2q}{2-q}}\log n.
\\ &= ns_0(n)\Big(\frac{\log n}{N}\Big)^{-q/2} \log n
\\ &= N\tilde\delta^2.
\end{align*} 
\medskip
\noindent
\textbf{Case 1A.2: $z \leq B_1$.}
Note that $(R^{-1}\tilde\delta\big)^{-2q/(2-q)} = \frac{n^2}{zc^{2q/(2-q)}}$ and as in the previous case we still have
\begin{align*}
(R^{-1}\tilde\delta)^{-\frac{2q}{2-q}}\log n=  N\tilde\delta^2.
\end{align*} 
So we have 
\begin{align*}
    N\tilde\delta^2 &= (R^{-1}\tilde\delta)^{-\frac{2q}{2-q}}\log n
    \\ &= \frac{n^2}{zc^{2q/(2-q)}}\log n
    \shortintertext{or}
    n^2 &= \frac{N\tilde\delta^2}{\log n}zc^{2q/(2-q)}
\end{align*}
Now, from  \textcite[Example 5.8]{wainwright_highdimensional_2019} we have that, for $z$ bounded by constant $B_1$,
\begin{align*}
    \log \mathcal{M}_{\mathcal{U}_q}^{\operatorname{loc}}(\tilde\delta,c) \leq n^2 \log(1 + 2c) = \frac{N\tilde\delta^2}{\log n}zc^{2q/(2-q)}\log(1+2c)\lesssim N\tilde{\delta}^2
\end{align*}


Now, for both cases define $B_2$ such that $\log \mathcal{M}_{\mathcal{U}_q}^{\operatorname{loc}}(\tilde\delta,c) \leq B_2N\tilde\delta^2$. Then for $A > \sqrt{B_2}\Upsilon \vee 1$
\begin{align*}

\end{align*}
Therefore, $\delta^\star \leq A\tilde\delta$ and $\delta^{\star 2} \lesssim n s_0(n)\big(\frac{\log n}{N}\big)^{1-q/2}$.
\\\textbf{Case 1B: $s_0(n) \big(\frac{\log n}{N}\big)^{1-q/2} \geq \frac{n}{N}$.}
Recall that $N\frac{\delta^2}{\Upsilon^2} \leq \log \mathcal{M}_K^{\textrm{loc}}(\delta,c) \lesssim n^2$ and thus we immediately get $N\delta^2 \lesssim n^2 \implies \delta^2 \lesssim \frac{n^2}{N}$.

\medskip
\noindent
\textbf{Case 2: $q=0$.}
Set $s = \min\{n,\lfloor s_0(n)\rfloor\}$ and our set of matrices becomes:
\begin{align*}
    \mathcal{U}_0(s) \coloneqq \bigg\{\Sigma: \max_i \sum_j1\kern-4pt 1(|\sigma_{ij}| > 0)\leq s,\lambda(\Sigma) \in [c_0,c_1] \bigg\}. 
\end{align*}
Using the eigenvalue upper bound, discarding the lower bound, and loosening the row-wise sparsity to overall matrix sparsity we get that 
\begin{align*}
    \mathcal{U}_0(s) \subseteq \big\{\Sigma: \|\Sigma\|_0 \leq ns, \|\Sigma\|_F \leq \sqrt{n}c_1 \big\}
\end{align*}
Notice that 
\begin{align*}
    \mathcal{U}_0(s) \subseteq \bigg\{\Sigma: \|\Sigma\|_0 \leq ns, \|\Sigma\|_F \leq \sqrt{n}c_1 \bigg\} = \bigcup_{|S|=ns}\mathbb{B}_2^{S}(\sqrt{n}c_1).
\end{align*}
That is, this set is a union of $\binom{n^2}{ns}$ $\ell_2^{n^2}$ balls of radius $\sqrt{n}c_1$. 
Now for a local center $\Sigma_0$ we consider the local metric entropy $B_F(\Sigma_0,\delta) \cap \mathcal{U}_0(s)$. Fix the support $S$. From \textcite[Example 5.8]{wainwright_highdimensional_2019} we have that the local metric entropy of this set is 
\begin{align*}
    \log \mathcal{M}_K^{\textrm{loc}}(\delta,c) &\leq ns\log\frac{en}{s} + ns\log(1 + 2c)
    \\ &\lesssim ns \log \frac{en}{s}
\end{align*}
due to the fact that
\begin{align*}
    \binom{n^2}{ns} &= \frac{n^2!}{(ns)!(n^2-ns)!} 
    \\ &= \frac{n^2 \times ... \times (n^2 - ns + 1)}{(ns)!}
    \\ &\leq \frac{n^{2ns}}{(ns/e)^{ns}} = \bigg(\frac{en^2}{ns}\bigg)^{ns} = \bigg(\frac{en}{s}\bigg)^{ns}
\end{align*}
with the inequality due to Stirling's approximation. Combining with the bound \\$N\frac{\delta^2}{\Upsilon^2}\leq \log\mathcal{M}_K^{\textrm{loc}}(\delta,c) \lesssim n^2$ gives the required bound.

\label[lemma]{lemma:kl_bound}
    Let $\Sigma_i$ and $\Sigma_j$ be covariance matrices in $K$.  Then there exists a sufficiently large constant $C_0 > 1$ depending only on $c_0$ and $c_1$ such that we can bound the KL divergence of normal random variables with mean 0 and these covariances as follows:

Define $A \coloneqq \Sigma_j^{-1} \Sigma_i$ with eigenvalues $\lambda_{\min} \leq \dots \leq \lambda_{\max}$. Note that $\Sigma_j^{1/2} (\Sigma_j^{-1}\Sigma_i) \Sigma_j^{-1/2} = \Sigma_{j}^{-1/2}\Sigma_i\Sigma_j^{-1/2}$ so $\Sigma_j^{-1}\Sigma_i$ and $\Sigma_j^{-1/2}\Sigma_i\Sigma_{j}^{-1/2}$ are similar. Thus the eigenvalues $\lambda_i$ of $\Sigma_j^{-1}\Sigma_i$ are equal to those of the symmetric $\Sigma_j^{-1/2}\Sigma_i\Sigma_j^{-1/2}$.
    We use \Cref{lemma:min_max_eigen} and Condition \ref{cond:K_set} to bound $\lambda_{\min}(A)$  as follows
    \begin{align*}
        \lambda_{\max}((\Sigma_j^{-1/2}\Sigma_i\Sigma_j^{-1/2})^{-1})  &= \lambda_{\max}(\Sigma_i^{-1/2}\Sigma_j\Sigma_i^{-1/2}) 
        \\ &\leq \lambda_{\max}(\Sigma_{i}^{-1/2})\lambda_{\max}(\Sigma_j)\lambda_{\max}(\Sigma_{i}^{-1/2})
        \\ &\leq \frac{c_1}{c_0}
    \end{align*} 
    We apply \Cref{lemma:min_max_eigen} to this inequality and get 
    \begin{align}
        \frac{1}{\lambda_{\min}(A)} \leq \frac{c_1}{c_0} \implies \lambda_{\min}(A) \geq \frac{c_0}{c_1}, \label{eq:eigen_bound_sigsig}
    \end{align}
    On the other hand, 
    \begin{align*}
        \lambda_{\max}(\Sigma_j^{-1/2}\Sigma_i\Sigma_j^{-1/2}) &\leq \|\Sigma_j^{-1/2}\|_{\operatorname{op}}^2\|\Sigma_i\|_{\operatorname{op}} \leq \frac{c_1}{c_0}.
    \end{align*}
    Now we show \Cref{eq:kl_f_ineq}:
    \begin{align}
        KL(N(0,\Sigma_i) \| N(0, \Sigma_j)) &= \frac{1}{2} \left(tr(\Sigma_j^{-1}\Sigma_i) - n + \log \frac{|\Sigma_j|}{|\Sigma_i|} \right) \notag
       \\ &= \frac{1}{2} \sum_{i=1}^{n} (\lambda_i - 1 + \log \lambda_i^{-1}) \label{eq:kl_equality}
    \end{align}
    where $\lambda_i$ denotes the $i$th eigenvalue of $\Sigma_j^{-1/2}\Sigma_i\Sigma_j^{-1/2}$.  Next, let $\mu_i \coloneqq \lambda_i - 1$ and see that
    \begin{align*}
        \lambda_i - 1 + \log(\lambda_i)^{-1} &= \mu_i - \log(\mu_i + 1) 
    \end{align*}
    We claim 
    \begin{align}
        \mu_i - \log(\mu_i + 1) \leq C_0\mu_i^2 \hspace{3mm} \text{on } \mu_i \in \bigg[\frac{c_0}{c_1}-1,\frac{c_1}{c_0}-1\bigg]\label{eq:log_sq_ineq}
    \end{align} for a sufficiently large $C_0$.
    To see this, first let $h(x)= \begin{cases} \frac{x - \log(x + 1)}{x^2} & x \neq 0 \\ 1/2 & x = 0\end{cases}$. This function is continuous on all $\Big[\frac{c_0}{c_1}-1,\frac{c_1}{c_0}-1\Big]$. This interval is closed and bounded it is compact, so $h(x)$ is guaranteed to attain a maximum somewhere on $\Big[\frac{c_0}{c_1}-1,\frac{c_1}{c_0}-1\Big]$, call it $C_0^\star \coloneqq \max h(x)$. Then let $C_0 = C_0^\star \vee 1$. Thus, $h(u_i) = \frac{u_i - \log(u_i+1)}{u_i^2} \leq C_0$ and the property holds.
    Continuing on from \Cref{eq:kl_equality}, we have
    \begin{align*}
        KL(N(0,\Sigma_i) \| N(0, \Sigma_j)) &\leq C_0\sum_{i=1}^n \mu_i^2
        \\ &= C_0\sum_{i=1}^n (\lambda_i - 1)^2
        \\ &\leq C_0\|\Sigma_j^{-1}\Sigma_i - \mathbb{I}\|_{F}^2
        \\ &\leq C_0\|\Sigma_j^{-1}\|_2^2 \|\Sigma_i - \Sigma_j\|_F^2
        \\ &\leq \frac{C_0}{c_0^2}\|\Sigma_i - \Sigma_j\|_F^2,
    \end{align*}
    where the last inequality holds due to the bound on $\lambda_{\min}$.

\end{align*}
Hoeffding's inequality gives $\mathbb{P}(\operatorname{Bin}(N,\varepsilon^{\prime\prime}) - N\varepsilon^{\prime\prime} \geq t) \leq \exp(-2t^2/N)$ and $\mathbb{P}(\operatorname{Bin}(N,\varepsilon^{\prime\prime}) - N\varepsilon^{\prime\prime} \leq t) \geq 1- \exp(-2t^2/N)$.

\end{align*}
Now, we can see that the left-hand side is the TV distance between two conditional mixtures. For $J\in\{1,2\}$ and given $N$ observations of $R_J$, let $W_J$ be the number of times $Q_J$ is drawn in place of $P_J$.     \mathcal{M}_{K^0}^{\operatorname{loc}}
    \left(
        \frac{\delta}{K_{\max}},
        6c
    \right).
    \label{eq:small_spike_case}
\end{equation}

\medskip
\noindent
\textbf{Case 2: $a^2>2\delta$.}
In this case \eqref{eq:spike_norm_control} gives
\begin{equation}
    \frac{a}{\sqrt{2}}
    \leq b_i
    \leq
    \sqrt{\frac32}\,a
    \qquad
    \text{for every }i.
    \label{eq:bi_comparable_a}
\end{equation}
For each $i$, choose $s_i\in\{-1,1\}$ so that
$s_i\langle x_i,x_0\rangle\geq 0$.  We claim that
\begin{equation}
    \|x_i-s_ix_0\|_2
    \leq
    \frac{2\delta}{a}.
    \label{eq:local_inverse_rank_one}
\end{equation}
To see this, set
\[
    t_i
    \coloneqq
    \frac{|\langle x_i,x_0\rangle|}{ab_i}
    \in[0,1].
\]
The identity
\begin{align*}
    \|x_ix_i^\top-x_0x_0^\top\|_F^2
    &=
    b_i^4+a^4-2\langle x_i,x_0\rangle^2 \\
    &=
    (b_i^2-a^2)^2
    +
    2a^2b_i^2(1-t_i^2)
\end{align*}
and the fact that the left-hand side is at most $\delta^2$ implies
\[
    2a^2b_i^2(1-t_i^2)\leq \delta^2.
\]
Moreover,
\[
    |b_i-a|
    =
    \frac{|b_i^2-a^2|}{a+b_i}
    \leq
    \frac{\delta}{a}.
\]
Therefore, using $1-t_i\leq 1-t_i^2$ and
$b_i\geq a/\sqrt{2}$,
\begin{align*}
    \|x_i-s_ix_0\|_2^2
    &=
    (b_i-a)^2+2ab_i(1-t_i) \\
    &\leq
    \frac{\delta^2}{a^2}
    +
    2ab_i(1-t_i^2) \\
    &\leq
    \frac{\delta^2}{a^2}
    +
    \frac{\delta^2}{ab_i} \\
    &\leq
    (1+\sqrt{2})\frac{\delta^2}{a^2}
    <
    \frac{4\delta^2}{a^2},
\end{align*}
which proves \eqref{eq:local_inverse_rank_one}.

Partition the packing indices according to the sign $s_i$:
\[
    I_+\coloneqq\{i:s_i=1\},
    \qquad
    I_-\coloneqq\{i:s_i=-1\}.
\]
By \eqref{eq:local_inverse_rank_one},
\[
    \{x_i:i\in I_+\}
    \subseteq
    B_2\left(x_0,\frac{2\delta}{a}\right)\cap K^0.
\]
If $I_-\neq\varnothing$, choose any $i_0\in I_-$ and put
$z=x_{i_0}\in K^0$.  Then, again by
\eqref{eq:local_inverse_rank_one},
\[
    \{x_i:i\in I_-\}
    \subseteq
    B_2\left(z,\frac{4\delta}{a}\right)\cap K^0.
\]
Thus both sign classes are contained in a Euclidean ball, centered
at a point of $K^0$, of radius at most $4\delta/a$.

It remains to control their packing separation.  For $i\neq j$,
\begin{align*}
    \frac{\delta}{c}
    &\leq
    \|x_ix_i^\top-x_jx_j^\top\|_F \\
    &\leq
    (b_i+b_j)\|x_i-x_j\|_2 \\
    &\leq
    \sqrt{6}\,a\,\|x_i-x_j\|_2,
\end{align*}
where the last inequality follows from
\eqref{eq:bi_comparable_a}.  Hence
\[
    \|x_i-x_j\|_2
    \geq
    \frac{\delta}{\sqrt{6}\,ca}
    =
    \frac{4\delta/a}{4\sqrt{6}\,c}.
\]
It follows that each sign class has cardinality at most
\[
    \mathcal{M}_{K^0}^{\operatorname{loc}}
    \left(
        \frac{4\delta}{a},
        4\sqrt{6}\,c
    \right).
\]
Since $a\leq K_{\max}$,
\[
    \frac{4\delta}{a}
    \geq
    \frac{\delta}{K_{\max}},
\]
so the monotonicity of local metric entropy again yields
\begin{equation}
    M
    \leq
    2\,
    \mathcal{M}_{K^0}^{\operatorname{loc}}
    \left(
        \frac{\delta}{K_{\max}},
        4\sqrt{6}\,c
    \right).
    \label{eq:large_spike_case}
\end{equation}

Combining \eqref{eq:small_spike_case} and
\eqref{eq:large_spike_case}, and taking for example $C=10$,
gives \eqref{eq:spiked_local_entropy}.  Since the bound is uniform
over $\Sigma_0\in K^+$, taking the supremum over the local center
completes the proof.
