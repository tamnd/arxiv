\label{thm:ImZ-limit-kernel}
Let \(T>e\), put \(L=\log T\), and let \(t\asymp T\).  Define
\[
A_\infty(z)
:=
2\int_0^1(1-u)\sin(uz)\,du
=
\frac{2(z-\sin z)}{z^2},
\qquad A_\infty(0)=0.
\]
Then
\[
\boxed{
\Im Z\!\left(\frac12+it\right)
=
\Re\sum_{\rho}
A_\infty\!\left(
-iL\left(\rho-\frac12-it\right)
\right)
+
O\!\left(\frac1L\right),
}
\]
where the sum is over the non-trivial zeros of \(\zeta\), counted with
multiplicity and taken symmetrically.

In particular, under the Riemann Hypothesis this reduces to
\[
\Im Z\!\left(\frac12+it\right)
=
\sum_{\gamma}
A_\infty\!\bigl((\gamma-t)L\bigr)
+
O\!\left(\frac1L\right).
\]

This is a higher-order extension of the unconditional Fej\'er-kernel
identity proved by the author in \cite{Ary22}, corresponding to the case
\(k=1\).  Under the Riemann Hypothesis, the \(k=1\) identity was proved
earlier by Gonek \cite{Gonek}. \\

We briefly describe the architecture of the argument.  We begin with a
contour on the right of the critical strip, where the logarithmic derivative
of the zeta function can be expanded as an absolutely convergent Dirichlet
series.  We then move the contour to the left of the critical strip.  The
difference between the two contour integrals is given by the residues of the
non-trivial zeros, together with the contribution of the pole of the zeta
function.  On the left-hand contour we use the functional equation for the
logarithmic derivative to reflect the zeta term back to the right-hand
half-plane, where it can once again be expanded into a Dirichlet series.
The remaining term arising from the functional equation is the Gamma-factor
contribution.  This is treated separately by Stirling's formula, exactly as
in the \(k=1\) argument.  In this way no assumption on the location of the
non-trivial zeros is required.\\

Put
\[
s_0:=\frac12+it
\]
and, for \(k\geq1\), define
\begin{equation}
\label{eq:Gk-def}
G_k(w)
:=
L\int_{-1}^{1}(1-|u|)^k e^{Luw}\,du.
\end{equation}
The function \(G_k\) is entire and even.  Moreover,
\begin{equation}
\label{eq:Gk-Sk-complex}
\frac1L G_k(w)
=
\widetilde S_k\!\left(-\frac{iLw}{2}\right).
\end{equation}
Indeed,
\[
\frac1L G_k(w)
=
2\int_0^1(1-u)^k\cosh(Luw)\,du
=
2\int_0^1(1-u)^k
\cos\!\left(-iLuw\right)\,du.
\]

Let \(c=1+\delta\), where \(0<\delta<1\), and consider
\[
I_k^{+}(t)
:=
\frac{1}{2\pi i}
\int_{(c)}
\frac{\zeta'}{\zeta}(s)
G_k(s-s_0)\,ds.
\]
Since \(c>1\), we may use the absolutely convergent expansion
\[
\frac{\zeta'}{\zeta}(s)
=
-\sum_{n=2}^{\infty}\frac{\Lambda(n)}{n^s}.
\]
For \(n>1\), Perron's formula gives
\begin{equation}
\label{eq:Gk-Perron}
\frac{1}{2\pi i}
\int_{(c)}
n^{-s}G_k(s-s_0)\,ds
=
\begin{cases}
\displaystyle
n^{-s_0}
\left(1-\frac{\log n}{L}\right)^k,
& n<T,\\[3mm]
0,&n>T.
\end{cases}
\end{equation}
Consequently,
\begin{equation}
\label{eq:Iplus}
I_k^{+}(t)
=
-\sum_{n<T}\frac{\Lambda(n)}{n^{s_0}}
\left(1-\frac{\log n}{L}\right)^k.
\end{equation}
If we write
\[
P_k(t)
:=
\sum_{n<T}\frac{\Lambda(n)}{n^{s_0}}
\left(1-\frac{\log n}{L}\right)^k,
\]
then
\begin{equation}
\label{eq:Ck-P}
\mathcal C_k(t)
=
-\frac{2}{L}\Re P_k(t),
\qquad
I_k^{+}(t)=-P_k(t).
\end{equation}

We next introduce the corresponding integral on the line
\(\Re s=1-c=-\delta\):
\[
I_k^{-}(t)
:=
\frac{1}{2\pi i}
\int_{(1-c)}
\frac{\zeta'}{\zeta}(s)
G_k(s-s_0)\,ds.
\]
By the residue theorem, the difference between the two contour
integrals is
\begin{equation}
\label{eq:residue-strip}
I_k^{+}(t)-I_k^{-}(t)
=
\sum_{\rho}G_k(\rho-s_0)
-
G_k(1-s_0).
\end{equation}
Here the sum is over all non-trivial zeros of \(\zeta\), counted with
multiplicity.  The second term comes from the pole of \(\zeta\) at
\(s=1\), whose residue in \(\zeta'/\zeta\) is \(-1\).
Since \(0<\delta<1\), no trivial zero lies between the two vertical
lines. \\

We now treat \(I_k^{-}(t)\) by the functional equation.  Using
\begin{equation}
\label{eq:FE-logder}
\frac{\zeta'}{\zeta}(s)
=
\frac{\chi'}{\chi}(s)
-
\frac{\zeta'}{\zeta}(1-s),
\end{equation}
we obtain
\begin{equation}
\label{eq:Iminus-FE}
I_k^{-}(t)
=
\Gamma_k(t)-J_k(t),
\end{equation}
where
\[
\Gamma_k(t)
:=
\frac{1}{2\pi i}
\int_{(1-c)}
\frac{\chi'}{\chi}(s)
G_k(s-s_0)\,ds
\]
and
\[
J_k(t)
:=
\frac{1}{2\pi i}
\int_{(1-c)}
\frac{\zeta'}{\zeta}(1-s)
G_k(s-s_0)\,ds.
\]

In \(J_k(t)\), make the change of variables \(z=1-s\).  Since \(G_k\)
is even and
\[
1-s_0=\frac12-it=\overline{s_0},
\]
we obtain
\[
J_k(t)
=
\frac{1}{2\pi i}
\int_{(c)}
\frac{\zeta'}{\zeta}(z)
G_k(z-\overline{s_0})\,dz.
\]
We may again expand \(\zeta'/\zeta\) into its Dirichlet series, and the
same calculation as in \eqref{eq:Gk-Perron} gives
\[
J_k(t)
=
-\sum_{n<T}\frac{\Lambda(n)}{n^{\overline{s_0}}}
\left(1-\frac{\log n}{L}\right)^k
=
-\overline{P_k(t)}.
\]
Thus
\begin{equation}
\label{eq:Iminus-final}
I_k^{-}(t)
=
\Gamma_k(t)+\overline{P_k(t)}.
\end{equation}

Substituting \eqref{eq:Iplus} and \eqref{eq:Iminus-final} into
\eqref{eq:residue-strip}, we find
\[
-P_k(t)-\overline{P_k(t)}-\Gamma_k(t)
=
\sum_{\rho}G_k(\rho-s_0)-G_k(1-s_0).
\]
Taking real parts,

-2\Re P_k(t)
=
\Re\sum_{\rho}G_k(\rho-s_0)

\Re G_k(1-s_0)
+
\Re\Gamma_k(t).

\frac1L
\Re\sum_{\rho}G_k(\rho-s_0)
-
\frac1L\Re G_k(1-s_0)
+
\frac1L\Re\Gamma_k(t).
\end{equation}

\frac{1}{2\pi}
\int_{-\infty}^{\infty}
\frac{\chi'}{\chi}\left(\frac12+i\tau\right)
G_k\bigl(i(\tau-t)\bigr)\,d\tau
+
O_k(T^{-1}).
\]

\[
G_k(iv)
=
L\widetilde S_k\left(\frac{Lv}{2}\right)

\[

O_k(T^{-1}).

On the critical line, For \(|y|\geq2\), Stirling's formula gives

\[

:=
1-\frac1L\Re G_k(1-s_0)
+\frac1L\Re\Gamma_k(t).
\]
Since
\[
\Gamma_k(t)
=
-\log\frac{t}{2\pi}+O(1)
\]
uniformly in this range, and \(t\asymp T\), we have
\[
E_k(t)\ll\frac1L,
\qquad
E_k'(t)\ll\frac1{TL}.
\]
Thus \eqref{eq:Ck-pre-gamma} becomes
\[
\mathcal C_k(t)
=
-1+
\frac1L\Re\sum_{\rho}G_k(\rho-s_0)
+
E_k(t).
\]

\frac1L G_k(\rho-s_0)
=
\widetilde S_k\left(
-\frac{iL}{2}(\rho-s_0)
\right),

\[
\mathcal C_k(t)
=
-1+
\Re\sum_{\rho}
\widetilde S_k\left(
-\frac{iL}{2}
\left(\rho-\frac12-it\right)
\right)
+
E_k(t),
\]
where
\[
E_k(t)\ll\frac1L,
\qquad
E_k'(t)\ll\frac1{TL}.
\]
