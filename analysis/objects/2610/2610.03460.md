<!-- definition-1: body_md -->
Let $g=(g_1,\ldots,g_n)\in\mathbb{F}_{q^m}^n$ have $\mathbb{F}_q$-linearly independent entries, and let $1\le\kappa\le n$. The \textbf{Gabidulin code} of dimension $\kappa$ evaluated at $g$ is $\mathbf{Gab}(g,\kappa):=\{(f(g_1),\ldots,f(g_n)): \deg_q(f)<\kappa\}$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Let $\mathcal{C}\le\mathbb{F}_q^{m\times n}$ be a matrix code of dimension $k$. Let $\mathcal{C}_s\le\mathcal{C}$ have codimension $\ell_s$, and let $\mathcal{A}\le\mathbb{F}_q^{m\times n}$ have dimension $\ell_a$ and satisfy $\mathcal{A}\cap\mathcal{C}=\{0\}$. The corresponding \textbf{Add-and-Remove code} is $\mathcal{D}:=\mathcal{C}_s\oplus\mathcal{A}$.
It has dimension $\dim_{\mathbb{F}_q}(\mathcal{D})=k-\ell_s+\ell_a$.
We write $\mathcal{D}\in\mathsf{AddRemove}(\mathcal{C},\ell_a,\ell_s)$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
Let $\mathcal{C}\le\mathbb{F}_q^{m\times n}$ have dimension $k$. For $1\le i\le k$, the $i$-th \textbf{subcode distance} of $\mathcal{C}$ is $\alpha_i(\mathcal{C}):=\max\{\operatorname{d}(\mathcal{E}):\mathcal{E}\le\mathcal{C},\ \dim_{\mathbb{F}_q}(\mathcal{E})=i\}$.
<!-- end definition-3 -->
<!-- proposition-1: body_md -->
\label{prop:gab-alpha}
Let $m\ge n$ and $\mathcal{C}=\Psi_{\gamma}(\mathbf{Gab}(g,\kappa))\le\mathbb{F}_q^{m\times n}$. Then, for every $1\le i\le m\kappa$,
\begin{equation}\label{eq:gab-alpha}
 \alpha_i(\mathcal{C})=n-\left\lceil\frac{i}{m}\right\rceil+1.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Fix $1\le i\le m\kappa$ and set $a:=\lceil i/m\rceil\le\kappa$. Then $(a-1)m<i\le am$. By \eqref{eq:gab-subcode}, the Gabidulin subcode $\mathcal{C}_a\le\mathcal{C}$ has dimension $am$ and distance $n-a+1$. Any $i$-dimensional subcode $\mathcal{E}\le\mathcal{C}_a$ satisfies $\operatorname{d}(\mathcal{E})\ge \operatorname{d}(\mathcal{C}_a)=n-a+1$. Hence $\alpha_i(\mathcal{C})\ge n-a+1$. The opposite inequality is~\eqref{eq:alpha-sing}.
<!-- end proof-1 -->
<!-- proposition-2: body_md -->
\label{prop:general}
Let $\mathcal{C}$ be a matrix code of dimension $k$ and let $\mathcal{D}=\mathcal{C}_s\oplus\mathcal{A}\in\mathsf{AddRemove}(\mathcal{C},\ell_a,\ell_s)$. Then
\begin{align}
 \alpha_i(\mathcal{C}_s)&\ge \alpha_{i+\ell_s}(\mathcal{C}),
 &&1\le i\le k-\ell_s,\label{eq:remove-bound}\\
 \alpha_i(\mathcal{D})&\ge \alpha_i(\mathcal{C}_s),
 &&1\le i\le k-\ell_s,\label{eq:add-lower}\\
 \alpha_i(\mathcal{D})&\le \alpha_{i-\ell_a}(\mathcal{C}_s),
 &&\ell_a<i\le k-\ell_s+\ell_a.\label{eq:add-upper}
\end{align}
Consequently,
\begin{align}
 \alpha_i(\mathcal{D})&\ge \alpha_{i+\ell_s}(\mathcal{C}),
 &&1\le i\le k-\ell_s,\label{eq:general-lower}\\
 \alpha_i(\mathcal{D})&\le \alpha_{i-\ell_a}(\mathcal{C}),
 &&\ell_a<i\le k-\ell_s+\ell_a.\label{eq:general-upper}
\end{align}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
For \eqref{eq:remove-bound}, let $\mathcal{E}\le\mathcal{C}$ have dimension $i+\ell_s$ and satisfy $\operatorname{d}(\mathcal{E})=\alpha_{i+\ell_s}(\mathcal{C})$. Since $\mathcal{C}_s$ has codimension $\ell_s$ in $\mathcal{C}$, the intersection $\mathcal{E}\cap\mathcal{C}_s$ has dimension at least $i$. Choosing an $i$-dimensional subcode $\mathcal{F}\le\mathcal{E}\cap\mathcal{C}_s$ gives $\alpha_i(\mathcal{C}_s)\ge \operatorname{d}(\mathcal{F})\ge \operatorname{d}(\mathcal{E})=\alpha_{i+\ell_s}(\mathcal{C})$, proving \eqref{eq:remove-bound}. The inclusion $\mathcal{C}_s\le\mathcal{D}$ immediately gives \eqref{eq:add-lower}.

For \eqref{eq:add-upper}, choose an $i$-dimensional subcode $\mathcal{E}\le\mathcal{D}$ with $\operatorname{d}(\mathcal{E})=\alpha_i(\mathcal{D})$. Since $\mathcal{C}_s$ has codimension $\ell_a$ in $\mathcal{D}$, the intersection $\mathcal{E}\cap\mathcal{C}_s$ has dimension $r\ge i-\ell_a>0$. Hence $\alpha_r(\mathcal{C}_s)\ge \operatorname{d}(\mathcal{E}\cap\mathcal{C}_s)\ge \operatorname{d}(\mathcal{E})=\alpha_i(\mathcal{D})$. Monotonicity gives $\alpha_{i-\ell_a}(\mathcal{C}_s)\ge\alpha_r(\mathcal{C}_s)$, proving \eqref{eq:add-upper}. The bounds \eqref{eq:general-lower} and \eqref{eq:general-upper} then follow from \eqref{eq:remove-bound}--\eqref{eq:add-upper} and the inclusion $\mathcal{C}_s\le\mathcal{C}$.
<!-- end proof-2 -->
<!-- remark-1: body_md -->
\label{prop:dual-addremove}
If $\mathcal{D}\in\mathsf{AddRemove}(\mathcal{C},\ell_a,\ell_s)$, then $\mathcal{D}^\perp\in\mathsf{AddRemove}(\mathcal{C}^\perp,\ell_s,\ell_a)$. Indeed, $\mathcal{C}^\perp\cap\mathcal{D}^\perp=(\mathcal{C}+\mathcal{D})^\perp$, and $\dim_{\mathbb{F}_q}(\mathcal{C}+\mathcal{D})=\dim_{\mathbb{F}_q}(\mathcal{C})+\ell_a$ because $\mathcal{C}\cap\mathcal{D}=\mathcal{C}_s$. Hence $\mathcal{C}^\perp\cap\mathcal{D}^\perp$ has codimension $\ell_a$ in $\mathcal{C}^\perp$ and codimension $\ell_s$ in $\mathcal{D}^\perp$. Moreover, $\mathcal{D}^\perp=\mathcal{E}\oplus\mathcal{R}$, where $\mathcal{E}:=\mathcal{C}^\perp\cap\mathcal{A}^\perp=\mathcal{C}^\perp\cap\mathcal{D}^\perp$ (as $\mathcal{C}^\perp\le\mathcal{C}_s^\perp$) and $\mathcal{R}$ is any complement of $\mathcal{E}$ in $\mathcal{D}^\perp$; then $\dim_{\mathbb{F}_q}(\mathcal{R})=\ell_s$ and $\mathcal{R}\cap\mathcal{C}^\perp=\mathcal{R}\cap\mathcal{E}=\{0\}$.
<!-- end remark-1 -->
<!-- corollary-1: body_md -->
\label{cor:comparison}
Let $\mathcal{D}=\mathcal{C}_s\oplus\mathcal{A}\in\mathsf{AddRemove}(\mathcal{C},\ell_a,\ell_s)$ and let $\ell_a < i
\le k-\ell_s$. If the subcode distances of $\mathcal{C}$ with indices in $[i-\ell_a,\, i+\ell_s]$ are all equal, that is,
\[
   \alpha_{i-\ell_a}(\mathcal{C}) = \alpha_{i-\ell_a+1}(\mathcal{C})
   = \cdots = \alpha_{i+\ell_s}(\mathcal{C}) =: v,
\]
then $\alpha_i(\mathcal{D}) = v$. By Remark~\ref{prop:dual-addremove}, the same statement holds for $\mathcal{D}^\perp$ and $\mathcal{C}^\perp$ with $\ell_s$ and $\ell_a$ interchanged.
<!-- end corollary-1 -->
<!-- proof-3: body_md -->
Inequalities \eqref{eq:general-lower} and \eqref{eq:general-upper} give
$v = \alpha_{i+\ell_s}(\mathcal{C}) \le \alpha_i(\mathcal{D}) \le \alpha_{i-\ell_a}(\mathcal{C}) = v$.
<!-- end proof-3 -->
<!-- theorem-1: body_md -->
\label{thm:gab-general}
For every $1\le i\le m\kappa-\ell_s$, we have
\begin{equation}\label{eq:gab-bounds}
 n-\left\lceil\frac{i+\ell_s}{m}\right\rceil+1
 \le \alpha_i(\mathcal{D})\le
 n-\left\lceil\frac{i}{m}\right\rceil+1.
\end{equation}
Consequently, if
\begin{equation}\label{eq:same-ceil}
 \left\lceil\frac{i+\ell_s}{m}\right\rceil
 =\left\lceil\frac{i}{m}\right\rceil,
\end{equation}
then $\alpha_i(\mathcal{D})=n-\left\lceil i/m\right\rceil+1$.
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
By \eqref{eq:general-lower} and \eqref{eq:gab-alpha}, $\alpha_i(\mathcal{D})\ge\alpha_{i+\ell_s}(\mathcal{C})=n-\left\lceil(i+\ell_s)/m\right\rceil+1$. The upper bound is the Singleton estimate \eqref{eq:alpha-sing} applied to the $i$-dimensional subcodes of $\mathcal{D}$. If \eqref{eq:same-ceil} holds, the two bounds coincide.
<!-- end proof-4 -->
<!-- corollary-2: body_md -->
\label{cor:blocks}
Assume $0\le\ell_s<m$, and let $i=(a-1)m+t\le m\kappa-\ell_s$ with $1\le t\le m$. Then
\begin{enumerate}
 \item if $1\le t\le m-\ell_s$, then $ \alpha_i(\mathcal{D})=n-a+1$;
 \item if $m-\ell_s<t\le m$, then
 $\alpha_i(\mathcal{D})\in\{n-a+1,n-a\}$.
\end{enumerate}
In particular, if $\ell_s=0$, then $\alpha_i(\mathcal{D})=n-\lceil i/m\rceil+1$ for all $1\le i\le m\kappa$.
<!-- end corollary-2 -->
<!-- proposition-3: body_md -->
\label{prop:min}
Assume $0\le\ell_s<m$. Then
\begin{equation}\label{eq:min}
 \operatorname{d}(\mathcal{D})=\min\{n-\kappa+1,\Delta(\mathcal{A},\mathcal{C}_s)\}.
\end{equation}
If $\ell_a=\ell_s$, then $\mathcal{D}$ has dimension $m\kappa$ and is MRD if and only if
\begin{equation}\label{eq:mrd}
 \Delta(\mathcal{A},\mathcal{C}_s)\ge n-\kappa+1.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Every nonzero element of $\mathcal{D}=\mathcal{C}_s\oplus\mathcal{A}$ belongs either to $\mathcal{C}_s\setminus\{0\}$ or to a coset $X+\mathcal{C}_s$ with $0\ne X\in\mathcal{A}$. Consequently,
\begin{equation}\label{eq:min-decomposition}
 \operatorname{d}(\mathcal{D})=\min\{\operatorname{d}(\mathcal{C}_s),\Delta(\mathcal{A},\mathcal{C}_s)\}.
\end{equation}
Since $\mathcal{C}_s\le\mathcal{C}$ and $\operatorname{d}(\mathcal{C})=n-\kappa+1$, one has $\operatorname{d}(\mathcal{C}_s)\ge n-\kappa+1$. On the other hand, $\dim_{\mathbb{F}_q}(\mathcal{C}_s)=m\kappa-\ell_s$ and $0\le\ell_s<m$, so $\lceil(m\kappa-\ell_s)/m\rceil=\kappa$. The Singleton bound gives the reverse inequality, hence $\operatorname{d}(\mathcal{C}_s)=n-\kappa+1$, and \eqref{eq:min} follows from \eqref{eq:min-decomposition}. If $\ell_a=\ell_s$, then $\dim_{\mathbb{F}_q}(\mathcal{D})=m\kappa$. Such a code is MRD exactly when $\operatorname{d}(\mathcal{D})=n-\kappa+1$, which by \eqref{eq:min} is equivalent to \eqref{eq:mrd}.
<!-- end proof-5 -->
<!-- proposition-4: body_md -->
\label{prop:fqm-removal}
Assume $2\le\kappa<n$, let $\mathcal{C}_0=\Psi_{\gamma}(\mathbf{Gab}(g,\kappa-1))\le\mathcal{C}$, and let $\mathcal{A}\le\mathbb{F}_q^{m\times n}$ have dimension $m$ with $\mathcal{A}\cap\mathcal{C}=\{0\}$. Set $\mathcal{D}=\mathcal{C}_0\oplus\mathcal{A}$. Then, for every $1\le i\le m(\kappa-1)$,
\begin{equation}\label{eq:fqm-case}
 \alpha_i(\mathcal{D})=n-\left\lceil\frac{i}{m}\right\rceil+1.
\end{equation}
Moreover,
\begin{equation}\label{eq:fqm-d}
 \operatorname{d}(\mathcal{D})=\Delta(\mathcal{A},\mathcal{C}_0)\le n-\kappa+1.
\end{equation}
In particular, $\mathcal{D}$ is MRD if and only if $\Delta(\mathcal{A},\mathcal{C}_0)=n-\kappa+1$.
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
For $1\le i\le m(\kappa-1)$, the code $\mathcal{C}_0$ contains the nested Gabidulin subcodes $\mathcal{C}_a$ for $a\le\kappa-1$. The proof of Proposition~\ref{prop:gab-alpha} therefore gives a subcode of $\mathcal{C}_0\le\mathcal{D}$ attaining the bound \eqref{eq:alpha-sing} at every such index, proving \eqref{eq:fqm-case}.
The code $\mathcal{C}_0$ has minimum distance $n-\kappa+2$, while $\mathcal{D}$ has dimension $m\kappa$. As in \eqref{eq:min-decomposition}, $\operatorname{d}(\mathcal{D})=\min\{n-\kappa+2,\Delta(\mathcal{A},\mathcal{C}_0)\}$. The Singleton bound for $\mathcal{D}$ gives $\operatorname{d}(\mathcal{D})\le n-\kappa+1$. Therefore $\Delta(\mathcal{A},\mathcal{C}_0)\le n-\kappa+1$, so the minimum is attained by $\Delta(\mathcal{A},\mathcal{C}_0)$, proving \eqref{eq:fqm-d}. The MRD characterization follows immediately.
<!-- end proof-6 -->
<!-- proposition-5: body_md -->
\label{prop:any-code}
Let $m\ge n$ and let $\mathcal{U}\le\mathbb{F}_q^{m\times n}$ be any code of codimension $c$. Then, for $1\le i\le mn-c$,
\[
 n-\left\lceil\frac{i+c}{m}\right\rceil+1\le\alpha_i(\mathcal{U})\le n-\left\lceil\frac{i}{m}\right\rceil+1.
\]
<!-- end proposition-5 -->
<!-- proof-7: body_md -->
For every $g\in\mathbb{F}_{q^m}^n$ with $\mathbb{F}_q$-linearly independent entries, $\Psi_{\gamma}(\mathbf{Gab}(g,n))=\mathbb{F}_q^{m\times n}$, and $\mathcal{U}\in\mathsf{AddRemove}(\mathbb{F}_q^{m\times n},0,c)$. Apply Theorem~\ref{thm:gab-general} with $\kappa=n$, $\ell_s=c$ and $\ell_a=0$.
<!-- end proof-7 -->
<!-- example-1: body_md -->
\label{ex:f8}
Let $\mathbb{F}_8=\mathbb{F}_2[\theta]/(\theta^3+\theta+1)$, take $\gamma=(1,\theta,\theta^2)$ and $g=(1,\theta,\theta^2)$, and set $\mathcal{C}:=\Psi_{\gamma}(\mathbf{Gab}(g,2))$. Then $\mathcal{C}\le\mathbb{F}_2^{3\times3}$ has dimension $6$ and minimum distance $2$, and Proposition~\ref{prop:gab-alpha} gives $(\alpha_1(\mathcal{C}),\ldots,\alpha_6(\mathcal{C}))=(3,3,3,2,2,2)$.

For a linearized polynomial $f$, write $\operatorname{ev}_g(f):=(f(1),f(\theta),f(\theta^2))$. Let $\mathcal L_0:=\{a_0x+a_1x^2:\ a_0\in\mathbb{F}_8,\ a_1\in\langle1,\theta\rangle_{\mathbb{F}_2}\}$ and $\mathcal{C}_0:=\Psi_{\gamma}(\operatorname{ev}_g(\mathcal L_0))$. Then $\mathcal{C}_0$ is a $5$-dimensional $\mathbb{F}_2$-hyperplane of $\mathcal{C}$ and contains the $3$-dimensional matrix Gabidulin subcode $\Psi_{\gamma}(\mathbf{Gab}(g,1))$, whose minimum distance is $3$. Let $A=E_{11}$, whose inverse image under $\Psi_{\gamma}$ is $(1,0,0)$. If $A\in\mathcal{C}$, some $f(x)=a_0x+a_1x^2$ with $a_0,a_1\in\mathbb{F}_8$ would satisfy $f(\theta)=f(\theta^2)=0$ and $f(1)=1$. This is impossible, because such an $f$ is nonzero, and a nonzero linearized polynomial of $q$-degree at most one (here $q=2$) has $\mathbb{F}_2$-kernel of dimension at most one, while $\theta,\theta^2$ are $\mathbb{F}_2$-linearly independent. Thus $A\notin\mathcal{C}$.
Set $\mathcal{D}=\mathcal{C}_0\oplus\langle A\rangle_{\mathbb{F}_2}\in\mathsf{AddRemove}(\mathcal{C},1,1)$. Since $A$ has rank one, $\operatorname{d}(\mathcal{D})=1$ by Proposition~\ref{prop:min}. Corollary~\ref{cor:blocks} with $\ell_s=1$ gives $\alpha_1=\alpha_2=3$ and $\alpha_4=\alpha_5=2$, while the $3$-dimensional MRD subcode in $\mathcal{C}_0$ gives $\alpha_3=3$. Hence $(\alpha_1(\mathcal{D}),\ldots,\alpha_6(\mathcal{D}))=(3,3,3,2,2,1)$. Here the index $i=3$, the only index $i<6$ at which Corollary~\ref{cor:blocks} allows a decrease, keeps the value of $\mathcal{C}$.

This depends on the hyperplane. Write $b_0$ for the coefficient of $1$ of $b\in\mathbb{F}_8$ in the basis $\gamma$, and let $\mathcal{C}_0'$ be the image of $\{a_0x+a_1x^2:\ a_0,a_1\in\mathbb{F}_8,\ (a_0)_0=(a_1)_0\}$. An exhaustive computer search shows that $\mathcal{C}_0'$ contains no $3$-dimensional subcode of minimum distance $3$, and that $\alpha_3(\mathcal{C}_0'\oplus\langle A'\rangle_{\mathbb{F}_2})=2$ for every $A'\in\mathbb{F}_2^{3\times3}\setminus\mathcal{C}$. For $A'=E_{11}$, $\mathcal{D}'=\mathcal{C}_0'\oplus\langle E_{11}\rangle_{\mathbb{F}_2}\in\mathsf{AddRemove}(\mathcal{C},1,1)$ has $(\alpha_1(\mathcal{D}'),\ldots,\alpha_6(\mathcal{D}'))=(3,3,2,2,2,1)$, so the smaller value in case 2) of Corollary~\ref{cor:blocks} is attained.
