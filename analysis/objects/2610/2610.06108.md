<!-- lemma-2: body_md -->
\label{lem:Goursat}
    Let $G$ be a subgroup of $G_1 \times G_2$, such that the projections $p_1 \colon G \rightarrow G_1, p_2 \colon G \rightarrow G_2$ are surjective.
    Denote
    \[
    N_1 \coloneqq p_1(\ker p_2) = \{g_1 \mid (g_1, \operatorname{id}) \in G \} \unlhd G_1, \;\; \text{ and } \;\; N_2 \coloneqq p_2(\ker p_1) = \{g_2 \mid (\operatorname{id}, g_2) \in G \} \unlhd G_2.
    \]
    Then there is an isomorphism $\varphi \colon G_1/N_1 \xrightarrow{\sim} G_2/N_2$, and $G = \{(g_1, g_2) \mid \varphi(g_1 N_1) = g_2 N_2\}$.
<!-- end lemma-2 -->
<!-- example-1: body_md -->
Let $G$ be a subgroup of $\mathbb{Z}/4\mathbb{Z} \times \mathbb{Z}/6\mathbb{Z}$, whose projections on $\mathbb{Z}/4\mathbb{Z}$ and $\mathbb{Z}/6\mathbb{Z}$ are surjective.
    We use Goursat's lemma to find all such $G$.
    Since $N_1 \unlhd \mathbb{Z}/4\mathbb{Z}$, we have $(\mathbb{Z}/4\mathbb{Z})/N_1 = \mathbb{Z}/4\mathbb{Z}, \mathbb{Z}/2\mathbb{Z}$, or $\{\operatorname{id}\}$.
    Similarly for $N_2 \unlhd \mathbb{Z}/6\mathbb{Z}$, we have $(\mathbb{Z}/6\mathbb{Z})/N_2 = \mathbb{Z}/6\mathbb{Z}, \mathbb{Z}/3\mathbb{Z}, \mathbb{Z}/2\mathbb{Z}$, or $\{\operatorname{id}\}$.
    By Goursat's Lemma. we have $(\mathbb{Z}/4\mathbb{Z})/N_1 \cong (\mathbb{Z}/6\mathbb{Z})/N_2$.
    Therefore $(\mathbb{Z}/4\mathbb{Z})/N_1$ and $(\mathbb{Z}/6\mathbb{Z})/N_2$ can only be $\mathbb{Z}/2\mathbb{Z}$ or $\{\operatorname{id}\}$.
    \begin{enumerate}[nosep, label = (\roman*)]
        \item If $(\mathbb{Z}/4\mathbb{Z})/N_1 \cong (\mathbb{Z}/6\mathbb{Z})/N_2 \cong \mathbb{Z}/2\mathbb{Z}$. Then $N_1 = 2(\mathbb{Z}/4\mathbb{Z})$ and $N_2 = 2(\mathbb{Z}/6\mathbb{Z})$.
        In this case, $G = \{(g_1, g_2) \mid (g_1 \bmod 2) = (g_2 \bmod 2)\}$. That is, elements of $G$ are pairs $(a,b)$ of the same parity.
        \item If $(\mathbb{Z}/4\mathbb{Z})/N_1 \cong (\mathbb{Z}/6\mathbb{Z})/N_2 \cong \{\operatorname{id}\}$. Then $N_1 = \mathbb{Z}/4\mathbb{Z}$ and $N_2 = \mathbb{Z}/6\mathbb{Z}$. In this case, $G$ is the full group $\mathbb{Z}/4\mathbb{Z} \times \mathbb{Z}/6\mathbb{Z}$.
    \end{enumerate}

<!-- lemma-3: body_md -->
\label{lem:submonoid_to_subgroup}
    In $\operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$, Submonoid Membership reduces to Subgroup Membership.
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
Let $\sigma$ be the shift map $x \mapsto x+1$ and $\pi \colon \operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z} \rightarrow \mathbb{Z}$ be the projection.
    Let $g_1, \ldots, g_n \in \operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$ and let $M$ be the submonoid they generate.
    \begin{itemize}[leftmargin=5mm]
        \item If there exist $i,j$, such that $\pi(g_i) < 0 < \pi(g_j)$, then $M$ is a subgroup of $\operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$.
        Indeed, for any $h \in M$, there exists $k \geq 1, m,n \geq 0$ such that $k \pi(h) + m \pi(g_i) + n\pi(g_j)= 0$, so $h^k g_i^m g_j^n \in \operatorname{FSym}(\mathbb{Z})$ and $(h^k g_i^mg_j^n)^t = \operatorname{id}$ for some $t \geq 1$, which yields
        $$h^{-1} = h^{k-1} g_i^mg_j^n (h^k g_i^mg_j^n)^{t-1} \in M.$$
        Therefore $M$ is a group and membership in $M$ reduces to Subgroup Membership.
        \item Otherwise $\pi(g_i)$ are all non-negative or all non-positive.
        Given $g \in \operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$, any product of elements $g_i$ evaluating to $g$ may only contain at most $|\pi(g)|$ occurrence of $g_i$ with $\pi(g_i) \neq 0$.
        Therefore it suffices to verify whether $g$ can be written as a product $h_0 g_{i_1} h_1 g_{i_2} h_2 \cdots g_{i_m} h_m$ with $m \leq |\pi(g)|$ and $h_0, \ldots, h_m$ in the finite group generated by $\{g_j \mid \pi(g_j) = 0\}$. \qedhere
    \end{itemize}

\label{example:invariant}
    We illustrate the groups $\left\langle {\tau_1, \ldots, \tau_s, \rho} \right\rangle$ and $\left\langle {\tau_1, \ldots, \tau_s} \right\rangle_{\rho}$ with three examples.
    \begin{enumerate}[nosep, leftmargin=8mm, label = (\arabic*)]
        \item Consider the group $\left\langle {\tau, \rho} \right\rangle \leq \operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$, generated by the 2-cycle $\tau \coloneq (01) \in \operatorname{FSym}(\mathbb{Z})$ and the one-step shift $\rho \coloneqq \sigma$.
        See Figure~\ref{fig:exampleone} for an illustration.
        The intersection $\left\langle {\tau, \rho} \right\rangle \cap \operatorname{FSym}(\mathbb{Z})$ is equal to $\left\langle {\tau} \right\rangle_{\rho}$, the group generated by the (infinitely many) elements $\rho^n\tau\rho^{-n}, n \in \mathbb{Z}$.
        For example, the element $\rho^2 \tau \rho^3 \tau \rho^{-5}$ is in the intersection $\left\langle {\tau, \rho} \right\rangle \cap \operatorname{FSym}(\mathbb{Z})$ because $2 + 3 + (-5) = 0$; and we can indeed write $\rho^2 \tau \rho^3 \tau \rho^{-5}$ as $(\rho^2\tau\rho^{-2}) (\rho^5\tau\rho^{-5}) \in \left\langle {\tau} \right\rangle_{\rho}$.
        Since $\left\langle {\tau} \right\rangle_{\rho}$ is generated by the 2-cycles $\tau = (01), \rho\tau\rho^{-1} = (12), \rho^2\tau\rho^{-2} = (23), \ldots$, we have $\left\langle {\tau} \right\rangle_{\rho} = \operatorname{FSym}(\mathbb{Z})$.
                                                                             
        \item Consider the group $\left\langle {\tau, \rho} \right\rangle \leq \operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$, generated by the 2-cycle $\tau \coloneq (01) \in \operatorname{FSym}(\mathbb{Z})$ and the two-step shift $\rho \coloneqq \sigma^2$.
        See Figure~\ref{fig:exampletwo} for an illustration.
                                                                                      
        In this case, the intersection $\left\langle {\tau, \rho} \right\rangle \cap \operatorname{FSym}(\mathbb{Z}) = \left\langle {\tau} \right\rangle_{\rho}$ is generated by the (infinitely many) 2-cycles $\tau = (01), \rho\tau\rho^{-1} = (23), \rho^2\tau\rho^{-2} = (45), \ldots$.
        Therefore $\left\langle {\tau} \right\rangle_{\rho}$ is a direct sum $\bigoplus_{\mathbb{Z}} C_2$, where $C_2$ is the cyclic group of two elements.
                                                                                                                                                 
        \item Consider the group $\left\langle {\tau, \tau', \rho} \right\rangle \leq \operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$, generated by $\tau \coloneq (01)$, $\tau' \coloneqq (02)(13)$, and the two-step shift $\rho \coloneqq \sigma^2$.
        See Figure~\ref{fig:examplethree} for an illustration.
        The intersection $\left\langle {\tau, \tau', \rho} \right\rangle \cap \operatorname{FSym}(\mathbb{Z}) = \left\langle {\tau, \tau'} \right\rangle_{\rho}$ is generated by the 2-cycles $\rho^n\tau\rho^{-n} = (2n \; 2n+1), n \in \mathbb{Z}$, and the double cycles $\rho^n\tau'\rho^{-n} = (2n \; 2n+2)(2n+1 \; 2n+3), n \in \mathbb{Z}$.
        In this case, $\left\langle {\tau, \tau'} \right\rangle_{\rho}$ preserves the partition $\bigsqcup_{n \in \mathbb{Z}}\{2n, 2n+1\}$.
        Here, the generator $\rho^n\tau\rho^{-n}$ swaps the two elements within the individual block $\{2n, 2n+1\}$, whereas the generator $\rho^n\tau'\rho^{-n}$ swaps the block $\{2n, 2n+1\}$ with its neighbouring block $\{2n+2, 2n+3\}$.
        Therefore, $\left\langle {\tau, \tau'} \right\rangle_{\rho}$ is a wreath product $C_2 \wr_{\mathbb{Z}} \operatorname{FSym}(\mathbb{Z})$.
                                                                                                                              
    \end{enumerate}

\label{example:product}
    Consider the group $\left\langle {\tau, \tau', \rho} \right\rangle \leq \operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$, generated by $\tau \coloneq (01)(23)$, $\tau' \coloneqq (26)(37)$, and the four-step shift $\rho \coloneqq \sigma^4$.
    See Figure~\ref{fig:examplefour} for an illustration.
    The intersection $\left\langle {\tau, \tau', \rho} \right\rangle \cap \operatorname{FSym}(\mathbb{Z}) = \left\langle {\tau, \tau'} \right\rangle_{\rho}$ is generated by the double cycles $\rho^n\tau\rho^{-n} = (4n \; 4n+1)(4n+2 \; 4n+3), n \in \mathbb{Z}$, and $\rho^n\tau'\rho^{-n} = (4n+2 \; 4n+6)(4n+3 \; 4n+7), n \in \mathbb{Z}$.
    The finite orbits of $\left\langle {\tau, \tau'} \right\rangle_{\rho}$ are $\{0, 1\}, \{4, 5\}, \{8, 9\}, \ldots$;
    there is an infinite orbit $\{2, 3, 6, 7, 10, 11, \ldots\}$.
    
    In this case, the group $G = \left\langle {\tau, \tau'} \right\rangle_{\rho}$ acts on the finite orbits as $G_{\operatorname{fin}} = \bigoplus_{\mathbb{Z}} C_2$, similar to Example~\ref{example:invariant}(2).
    Meanwhile, $G$ acts on the infinite orbit as $G_{\infty} = C_2 \wr_{\mathbb{Z}} \operatorname{FSym}(\mathbb{Z})$, similar to Example~\ref{example:invariant}(3).
    Thus, $G$ is a subgroup of $G_{\operatorname{fin}} \times G_{\infty}$.
    
    However, $G$ is not equal to $G_{\operatorname{fin}} \times G_{\infty}$.
    To see this, note that each $\rho^n\tau\rho^{-n}$ simultaneously inverts the block $\{4n, 4n+1\}$ in the finite orbit and the block $\{4n+2, 4n+3\}$ in the infinite orbit.
    Therefore, for every element in $G$, the total number of inverted blocks must be even.
    Due to this parity constraint, the element $(0 1) \in G_{\operatorname{fin}} \times G_{\infty}$ cannot appear in $G$.
    Nevertheless, we can show (c.f.\ Proposition~\ref{prop:structure_invariant_subgroup}) that $G$ contains a ``big'' subgroup of $G_{\infty}$.

<!-- proposition-1: body_md -->
\label{prop:structure_invariant_subgroup}
    Let $G \leq \operatorname{FSym}(\{1, \ldots, d\} \times \mathbb{Z})$ be finitely generated as a $\rho$-invariant subgroup.
    Then $G$ is a subgroup of $G_{\operatorname{fin}} \times G_{\infty}$ whose projections on $G_{\operatorname{fin}}$ and $G_{\infty}$ are surjective, and
    \begin{enumerate}[itemsep=1pt, topsep=3pt, label = {\normalfont(\roman*)}]
        \item $G_{\operatorname{fin}}$ is a $\rho$-invariant subgroup of $\bigoplus_{\mathbb{Z}} H$ for some finite $H$, where $\rho$ acts on $\bigoplus_{\mathbb{Z}} H$ as the shift $(h_k)_{k \in \mathbb{Z}} \mapsto (h_{k-1})_{k \in \mathbb{Z}}$.
        \item $G_{\infty}$ is either trivial, or it can be effectively written as a subgroup of $W = W_1 \times \cdots \times W_k$, where $W_i = H_i \wr_{\Delta_i} \operatorname{FSym}(\Delta_i)$ for some finite $H_i$ and countably infinite sets $\Delta_1,\ldots,\Delta_k$.
    \end{enumerate}
    Furthermore, $G \geq \{\operatorname{id}\} \times W'$, where $W' = W_1' \times \cdots \times W_k'$ is the commutator subgroup of $W$.

<!-- lemma-4: body_md -->
\label{lem:compute_orbits}
    If $G\le \operatorname{FSym}(\mathbb{Z})\rtimes\mathbb{Z}$ is finitely generated and $\pi(G)\ne\{0\}$, then the action of $G$ on $\mathbb{Z}$ has finitely many orbits. Moreover one can effectively compute a decomposition
    \[ \mathbb{Z} = (\mathcal{F}_1\sqcup\ldots\sqcup \mathcal{F}_c) \sqcup (\mathcal{I}_1\sqcup \ldots \sqcup \mathcal{I}_d) \]
    from the generators of $G$, where $\mathcal{F}_i$ are the finite orbits and $\mathcal{I}_i$ are the infinite orbits.
<!-- end lemma-4 -->
<!-- proof-2: body_md -->
Suppose that $G=\left\langle g_1,g_2,\ldots,g_r\right\rangle$ where $g_i = \tau_i\sigma^{\pi(g_i)}$ with $\tau_i\in\operatorname{FSym}(\mathbb{Z})$ and $\pi(g_i)\in \mathbb{Z}$ (not all zero). Let $m=\max\left\lvert \pi(g_i) \right\rvert > 0$. We can compute an interval $[a,b]$ such that $\operatorname{supp}(\tau_i)\subset [a,b]$ for $i=1,\ldots,r$. 
                                                                                                   
We claim that for all $x>b$ and $y \in \mathbb{Z}$, $x$ and $y$ are in the same orbit if and only if and $x + m$ and $y$ are in the same orbit.
Indeed, without loss of generality supposing that $\pi(g_1) = m$, we have $g(x+m) = y \iff gg_1(x) = y$.
Similarly, for all $x < a$ and $y \in \mathbb{Z}$, $x$ and $y$ are in the same orbit if and only if and $x - m$ and $y$ are in the same orbit.
Therefore, to compute all the orbits, we can draw the finite graph with
\begin{itemize}[leftmargin=8mm]
    \item vertex set $\mathbb{Z}/\!\!\sim$, where the congruence relation $\sim$ is defined as $x\sim y$ if either ($x,y<a-m$ and $x\equiv y\pmod m$), or ($x,y>b+m$ and $x\equiv y\pmod m$), or $x=y$;
    \item edges given by $([x],[g_i(x)])$ for $x\in\mathbb{Z}$ and $i=1,\ldots,r$, where $[x]$ denotes the equivalence class of $x$ under $\sim$.
\end{itemize}
The congruence classes of each connected graph component corresponds to a $G$-orbit.
This orbit is infinite if it contains a vertex whose congruence class is infinite.

<!-- lemma-5: body_md -->
\label{lem:rho_inf_orbit}
    Let $G$ be a finitely generated subgroup of $\operatorname{FSym}(\mathbb{Z}) \rtimes \mathbb{Z}$.
    Suppose $G$ is infinite and let $\mathcal{F} = \mathcal{F}_1 \sqcup \cdots \sqcup \mathcal{F}_c$ be the union of finite orbits of $G$.
    Then, there exists an effectively computable element $\rho \in G$, such that 
    \begin{enumerate}[nosep, label={\normalfont(\roman*)}]
        \item $\pi(\rho) > 0$,
        \item the union of finite orbits of $\left\langle {\rho} \right\rangle$ is equal to $\mathcal{F}$,
        \item $\rho(z) = z$ for all $z \in \mathcal{F}$.
    \end{enumerate}

In particular, there are at most $\left\lvert F \right\rvert$ indices $i\in\{0,1,\ldots,t-1\}$ such that $h^i(x)\in F$, in which case $h^{i+1}(x)\ge h^i(x)-m$. <!-- lemma-8: body_md -->
\label{lem:ker_equals_inv}
    Consider a morphism $\pi\colon H\rightarrow \mathbb{Z}$ and a finitely generated subgroup $G=\left\langle {\tau_1,\ldots,\tau_s,\rho} \right\rangle \leq H$, with $\pi(\tau_1)= \cdots = \pi(\tau_s) = 0$ and $\pi(\rho) \neq 0$. <!-- end lemma-8 -->
<!-- proof-6: body_md -->
By restricting the domain of $\pi$ to $G$, it suffices to consider the case $H = G$ and prove $\ker(\pi) = \left\langle {\tau_1,\ldots,\tau_s} \right\rangle_\rho$.
    
    Recall that the group $\left\langle {\tau_1, \ldots, \tau_s} \right\rangle_{\rho}$ is generated by elements of the form $\rho^\ell \tau_i \rho^{-\ell}, \ell \in \mathbb{Z}, 1 \leq i \leq n$.
    Each of these generators satisfies $\pi(\rho^n \tau_i \rho^{-n}) = n \pi(\rho) + \pi(\tau_i) - n \pi(\rho) = 0$, and therefore belongs to $\ker(\pi)$.
    We thus have $\left\langle {\tau_1,\ldots,\tau_s} \right\rangle_\rho \leq \ker(\pi)$.
    
    For the other inclusion, consider any element of $\ker(\pi)$: it is of the form
    \begin{equation}\label{eq:element_ker}
    \rho^{k_1}\tau_{i_1}^{\varepsilon_1} \rho^{k_2}\tau_{i_2}^{\varepsilon_2} \cdots \rho^{k_m}\tau_{i_m}^{\varepsilon_m} \rho^{k_{m+1}},
    \end{equation}

    It can be rewritten as
    \[
    \rho^{\ell_1}\tau_{i_1}^{\varepsilon_1}\rho^{-\ell_1} \cdot \rho^{\ell_2}\tau_{i_2}^{\varepsilon_2} \rho^{-\ell_2} \cdot \ldots \cdot \rho^{\ell_m}\tau_{i_m}^{\varepsilon_m} \rho^{-\ell_m},
    \]
    where $\ell_i \coloneqq k_1 + \cdots + k_i, 1 \leq i \leq m$.
    Therefore the element~\eqref{eq:element_ker} is in $\left\langle {\tau_1, \ldots, \tau_s} \right\rangle_{\rho}$.

    Let $G$ be a $\rho$-invariant subgroup of $\operatorname{FSym}(\{1, \ldots, d\} \times \mathbb{Z})$, generated by finitely many elements.

<!-- proof-8: body_md -->
Since $G$ is $\rho$-invariant, the partition of $\{1, \ldots, d\} \times \mathbb{Z}$ into $G$-orbits is also preserved $\rho$.
    Therefore if $\{(j, z_j + \ell x) \mid j \in J, x \in \mathbb{Z}\}$ is a $G$-orbit then $\{(j, z_j + \ell x + 1) \mid j \in J, x \in \mathbb{Z}\}$ is also a $G$-orbit.
    The statement of the proposition follows thus directly from Lemma~\ref{lem:single_orbit}, subject to effectivity.
    We now show that the partition into $G$-orbits can be effectively computed.

    Given $(j, y) \in \{1, \ldots, d\} \times \mathbb{Z}$, we will compute its $G$-orbit using a saturation process.
    Let $\tau_1, \ldots, \tau_s$ be the generators of $G$ as a a $\rho$-invariant subgroup.
    For a set $\mathcal{O}$ of the form~\eqref{eq:orbit_finite}, we can check whether it is invariant under the action of $G$: it is invariant if and only if for each $z \in \mathbb{Z}, 1 \leq i \leq n$, such that $\operatorname{supp}(\rho^{z} \tau_i \rho^{-z}) \cap \mathcal{O} \neq \emptyset$, we have $\rho^{z} \tau_i \rho^{-z}(\mathcal{O}) \subseteq \mathcal{O}$.
    This can be checked effectively since there are only finitely many $z \in \mathbb{Z}, 1 \leq i \leq n$ such that $\rho^{z}(\operatorname{supp}(\tau_i)) \cap \mathcal{O} \neq \emptyset$.
    For a set $\mathcal{O}$ of the form~\eqref{eq:orbit_infinite}, we can also check whether it is invariant under the action of $G$: it is invariant if and only if for each $0 \leq z \leq \ell-1, 1 \leq i \leq n$, we have $\rho^{z} \tau_i \rho^{-z}(\mathcal{O}) \subseteq \mathcal{O}$.
    Again this can be checked effectively since $\mathcal{O}$ is periodic.
    Note that in both cases, if $\mathcal{O}$ is not invariant under the action of $G$, then we can produce an element $(i, x) \in G \mathcal{O} \setminus \mathcal{O}$.

    We repeat this step until $\mathcal{O}$ is invariant under the action of $G$.
    It is easy to see that an ascending chain of sets of the forms~\eqref{eq:orbit_finite} and \eqref{eq:orbit_infinite} must stabilize, therefore the above procedure must terminate.

\label{lem:effective_partition}

<!-- proof-9: body_md -->
We first prove that blocks in non-trivial $G$-invariant partitions cannot be too large. 
    By re-indexing we can without loss of generality suppose
    \[
    \mathcal{O} = \{(j, \ell x) \mid j \in J, x \in \mathbb{Z}\}
    \]
    for some finite index set $J$.
    Take any $\left\langle\rho^\ell\right\rangle$-invariant metric on $\mathcal{O}$.
    For example one can take the metric defined by the distance 
    \begin{equation}\label{eq:distance}
        \operatorname{dist}((i, z_i), (j, z_j)) \coloneqq |z_i - z_j| + \frac{|i - j|}{L},
    \end{equation}
    where $L > \max_{i,j \in J}\{|i-j|\}$.
    For a set $S \subset \mathcal{O}$, let $\operatorname{diam}(S)$ denote its diameter under this distance.

    Since $G$ is transitive, for each $j \in J$ we can effectively compute an element $h_j \in G$, such that $h_j(j, 0) = (j, \ell)$, and let 
    \[
    D=\max_{j\in J}\operatorname{diam}(\operatorname{supp}(h_j)).
    \]

\begin{adjustwidth}{2mm}{2mm}
    \textbf{Claim.} Let $S$ be a block of a non-trivial $G$-invariant partition. Then $S$ is finite and $\operatorname{diam} S\le D$.

    \medskip

    \noindent$\triangleright$ Elements of $G$ have finite support, hence cannot permute infinite blocks. If $S$ were infinite, it would therefore be fixed by $G$. Using that $G$ is transitive, this would imply $S=\mathcal{O}$, a contradiction.

    \medskip
    
    \noindent Fix $(j, z), (j', z') \in S$ with $\operatorname{dist}((j, z), (j', z'))$ maximal and $z\ge z'$. If $\operatorname{dist}((j, z), (j', z'))> D$, then the element $\rho^{z} h_j \rho^{-z} \in G$ maps $(j, z)$ to $(j, z+ \ell)$, but $\rho^{z} h_j \rho^{-z}(j', z') = (j', z')$ because the support of $\rho^{z} h_j \rho^{-z}$ has diameter at most $D$.
    Therefore the block $S$ must also contain $(j, z+ \ell)$, contradicting that $\operatorname{dist}((j,z),(j',z'))$ was maximal.
    \hfill$\triangleleft$
\end{adjustwidth}

\medskip

\noindent From the above claim, there is an upper bound $D$ on the size of blocks in non-trivial $G$-invariant partitions of $\mathcal{O}$. Hence, there exists a coarsest\footnote{Here, $\rho^{\ell}$-invariance is essential for our proof. \medskip

\begin{adjustwidth}{2mm}{2mm}
    \textbf{Claim.} The coarsest (proper) $G$-invariant partition is unique, and $\left\langle\rho^\ell\right\rangle$-invariant.

    \noindent $\triangleright$ Consider a coarsest partition $\mathcal{O}=\bigsqcup_{\delta\in \Delta} \Gamma_\delta$ and a block $S$ in another non-trivial $G$-invariant partition.
    There exist three blocks $\Gamma_i,\Gamma_j,\Gamma_k$ with 
    \[ 
    \Gamma_i\cap S\ne\emptyset \quad\text{and}\quad \Gamma_j\cap S=\Gamma_k\cap S=\emptyset. 
    \]
    Note that $G$ acts primitively on $\Delta$, therefore one may apply \cite[Corollary~31.2]{neumann2023infinite}
    (or Lemma \ref{lem:transitive_finite} below) to deduce that there exists $g \in G$ such that $g(\Gamma_i)=\Gamma_j$, $g(\Gamma_j)=\Gamma_k$, $g(\Gamma_k)=\Gamma_i$, and $g$ fixes all other blocks $\Gamma_\delta$ pointwise.
    Since $S$ comes from another $G$-invariant partition, we conclude that $S\subseteq\Gamma_i$, hence the partition $\mathcal{O}= \bigsqcup_{\delta\in\Delta} \Gamma_\delta$ is the \emph{unique} coarsest $G$-invariant partition. Since $\mathcal{O}= \bigsqcup_{\delta\in\Delta} \rho^\ell(\Gamma_\delta)$ is another coarsest $G$-invariant partition, we deduce that the coarsest proper partition is $\left\langle\rho^\ell\right\rangle$-invariant. 
    \hfill $\triangleleft$
\end{adjustwidth}

\medskip

\noindent Consider a block $\Gamma$ of the coarsest $G$-invariant partition $\mathcal{O}$. We observe that each $\left\langle\rho^\ell\right\rangle$-orbit intersects $\Gamma$ in at most one element. Indeed, if $o,\rho^{\ell d}(o)\in \Gamma$, then the pair $\rho^{n\ell d}(o),\rho^{(n+1)\ell d}(o)$ belongs to the same block of the partition for all $n\ge 0$, and this block is always $\Gamma$ by induction.

There are only finitely many such partitions, we can enumerate them, check for each one whether it is preserved by the generators of $G$, and select the coarsest one. This allows us to compute the partition $\mathcal{O} = \Gamma \times \Delta$.

\label{lem:transitive_embed}

We follow the proof of~\cite[Lemma~2.6]{neumann1976structure}, see Figure~\ref{fig:twist} for an illustration.
    Recall that the index set $\Delta$ can be decomposed as a product $S \times \ell\mathbb{Z}$ for some finite set $S = \{s_1, \ldots, s_t\}$.

    Then for $i = 1, \ldots, t$ and $x \geq 0$, the element
    \begin{equation}\label{eq:gix}
    g_{i, x} \coloneqq \rho^{x\ell}g_i\rho^{-x\ell} \cdot \rho^{(x-1)\ell}h\rho^{-(x-1)\ell} \cdot \cdots \cdot \rho^{\ell}h\rho^{-\ell} \cdot h = \rho^{x\ell}g_i \cdot (\rho^{-\ell} h)^x
    \end{equation}
    sends the block $(s_1, 0)$ to $(s_i, x\ell)$.
    A similar element can be defined for $x < 0$.
    Then, the set-wise stabilizer $G_{\{\Gamma \times \{s_i, x \ell\}\}}$ is conjugate to $G_{\{\Gamma \times \{s_1, 0\}\}}$ via $g_{i, x}$.

    We then identify the copy $\Gamma \times \{(s_i, x\ell)\}$ with $\Gamma$ through the map 
    \[
    g_{i, x}^{-1} \colon \Gamma \times \{(s_i, x\ell)\} \rightarrow \Gamma \times \{(s_1, 0)\} = \Gamma, \quad \gamma \mapsto g_{i, x}^{-1}(\gamma).
    \]

    The quotient $W/W'$ is isomorphic to $C_2 \times C_2$, where the projection map $W \rightarrow W/W' = C_2 \times C_2$ is $(f, b) \mapsto (\prod_{\delta \in \Delta} f_{\delta},\; \operatorname{sgn}(b))$.
    If $G \leq W$ satisfies the two conditions in Lemma~\ref{lem:transitive_finite}, then $G = AW'$ where $A$ is a subgroup of $C_2 \times C_2$.

<!-- proposition-5: body_md -->
\label{cor:N_contains_Wprime}
    Let $W' \leq G \leq W$ and $N$ be a normal subgroup of $G$. Then either $\pi_{\Delta}(N) = \{\operatorname{id}\}$ or $N\ge W'$.
    In the first case $\left\lvert G/N \right\rvert$ is infinite, and in the second case $\left\lvert G/N \right\rvert < \infty$.

<!-- proof-11: body_md -->
The first statement of the proposition is a combination of the following two claims. The first claim is a variation on the ``double commutator lemma'', which goes back to Higman \cite{Higman}.
\medskip

\begin{adjustwidth}{2mm}{2mm}
    \textbf{Claim.} Let $W' \le G \le W$. Suppose that $N \unlhd G$, then either $\pi_{\Delta}(N) = \{\operatorname{id}\}$ or $N \ge G'$.
    \medskip
    
    \noindent$\triangleright$ We first observe that $\operatorname{FAlt}(\Delta)\le\pi_{\Delta}(G)\le\operatorname{FSym}(\Delta)$ since $W'\le G \le W$. Since $\pi_{\Delta}(N)$ is a normal subgroup of $\pi_{\Delta}(G)$, the intersection $\pi_\Delta(N)\cap\operatorname{FAlt}(\Delta)$ is either $\{\operatorname{id}\}$ or $\operatorname{FAlt}(\Delta)$. In the former case, we have $\pi_{\Delta}(N) = \{\operatorname{id}\}$. Suppose that we are in the later, i.e., $\pi_{\Delta}(N) \ge \operatorname{FAlt}(\Delta)$.
    \medskip
    
    \noindent Given $g_1,g_2 \in G$, we prove that $[g_1,g_2] \in N$. Since $\pi_{\Delta}(N) \ge \operatorname{FAlt}(\Delta)$, there is an element $n \in N$, such that the support $\operatorname{supp}(ng_1^{-1}n^{-1})=n(\operatorname{supp}(g_1^{-1}))$ is disjoint from $\operatorname{supp}(g_2)$ and hence $[ng_1^{-1}n^{-1},g_2]= \operatorname{id}$. Since $[g,n]=gng^{-1}\cdot n^{-1}\in N$ for all $g\in G$ and $n\in N$, we have
    \[ [g_1,g_2] = g_1[ng_1^{-1}n^{-1},g_2]g_2g_1^{-1}g_2^{-1}=\bigl[[g_1,n],g_2\bigr]\in N.\]
    Alternatively $[g_1, g_2]\equiv [ng_1n^{-1}, g_2]=\operatorname{id} \pmod N$.
    We conclude that $N \geq G'$. \hfill$\triangleleft$
\end{adjustwidth}
\medskip
\noindent Since $W' \le G$, the case $N \ge G'$ already yields $N \ge G' \ge W''$. It remains to prove $W''=W'$.
\medskip
\begin{adjustwidth}{2mm}{2mm}
    \textbf{Claim.} We have $W'' = W'$.
    \medskip
    
    \noindent$\triangleright$ We exhibit different elements that belong to $W''$. 
    For $h \in H$ and $\delta \in \Delta$, we let $h@\delta$ denote the element in $\bigoplus_{\Delta} H$ whose $\delta$-coordinate is $h$ and whose all other coordinates are $\operatorname{id}$.
    \begin{itemize}[leftmargin=6mm]
        \item First $W''\ge \operatorname{FAlt}(\Delta)''=\operatorname{FAlt}(\Delta)$ since $W\ge\operatorname{FAlt}(\Delta)$ and $\operatorname{FAlt}(\Delta)' = \operatorname{FAlt}(\Delta)$ (Lemma~\ref{lem:transitive_finite}).
        \item For all $h_1,h_2\in H$ and $\delta\in \Delta$, we have $[h_1,h_2]@\delta\in W''$. Indeed, by Lemma~\ref{lem:transitive_finite}, we have
        \[ [h_1,h_2]@\delta = \bigl[ (h_1@\delta)\cdot (h_1^{-1}@\delta_1),\; (h_2@\delta)\cdot (h_2^{-1}@\delta_2)\bigr] \in [W',W']=W'' \]
        for any $\delta_1,\delta_2\in\Delta$ distinct from each other and distinct from $\delta$.
        \item For all $h\in H$ and distinct $\delta_1,\delta_2\in \Delta$, we have $(h@\delta_1)\cdot (h^{-1}@\delta_2) \in W''$. Indeed, we have
        \[ (h@\delta_1)\cdot (h^{-1}@\delta_2) = \bigl[ (h@\delta_1)\cdot (h^{-1}@\delta_3),\, (\delta_1,\delta_2,\delta_4)\bigr]\in [W',\operatorname{FAlt}(\Delta)]\le W''\]
        for any $\delta_3,\delta_4\in\Delta$ distinct from each other and distinct from $\delta_1,\delta_2$.
    \end{itemize}
    Finally, it is easy to see that $W'$ is generated by these three types of elements. (The first two types of elements generate the subgroup $H'\wr_\Delta\operatorname{FAlt}(\Delta)\ge \bigoplus_\Delta H'$, then we can argue in the quotient $W/\bigoplus_\Delta H'\simeq (H/H')\wr_\Delta\operatorname{FAlt}(\Delta)$.) We conclude that $W''\ge W'$ hence $W''=W'$. \hfill$\triangleleft$
\end{adjustwidth}
\medskip
    
    \noindent 
    In case $\pi_{\Delta}(N) = \{\operatorname{id}\}$, the quotient $G/N$ induces a permutation group of $\Delta$ that contains $\operatorname{FAlt}(\Delta)$, therefore $\left\lvert G/N \right\rvert$ is infinite.
    In case $N \geq W'$, we have $\left\lvert G/N \right\rvert \leq \left\lvert W/W' \right\rvert < \infty$ by Lemma~\ref{lem:transitive_finite}.

<!-- lemma-13: body_md -->
\label{lem:no_outer}
    Let $\Delta, \Delta'$ be two countably infinite sets, and $\varphi \colon \operatorname{FAlt}(\Delta) \rightarrow \operatorname{FAlt}(\Delta')$ or $\varphi \colon \operatorname{FSym}(\Delta) \rightarrow \operatorname{FSym}(\Delta')$ be an isomorphism.
    Then there exists an isomorphism $\xi \colon \Delta \rightarrow \Delta'$ such that $\varphi(g) = \xi g \xi^{-1}$ for all $g$.\footnote{Lemma~\ref{lem:no_outer} also holds for finite sets $\Delta, \Delta'$ with $\left\lvert \Delta \right\rvert = \left\lvert \Delta' \right\rvert \neq 6$.

<!-- lemma-14: body_md -->
\label{lem:diagonal_or_fi_two}
    Let $G$ be a subgroup of $W_i \times W_j$, and let $G_i, G_j$ denote respectively the projection of $G$ on the components $W_i$ and $W_j$.
    Suppose $G_i \geq W'_i$ and $G_j \geq W'_j$.
    Then $G$ falls into one of the two following cases:
    \begin{enumerate}[nosep, label={\normalfont(\roman*)}]
        \item either $G \geq W'_i \times W'_j$, 
        \item or there exists a bijection $\xi \colon \Delta_i \rightarrow \Delta_j$, such that $G$ acts on the blocks $\Delta_i \sqcup \Delta_j$ as the group $\{(g, \xi g \xi^{-1}) \mid g \in \operatorname{FAlt}(\Delta_i)\}$ or $\{(g, \xi g \xi^{-1}) \mid g \in \operatorname{FSym}(\Delta_i)\}$.
    \end{enumerate}
    In the second case, under the bijection $\xi^{-1} \colon \Delta_j \rightarrow \Delta_i$, the set $(\Gamma_i \times \Delta_i) \sqcup (\Gamma_j \times \Delta_j)$ can be identified with $(\Gamma_i \sqcup \Gamma_j) \times \Delta_i$, on which $G$ acts faithfully as a subgroup of $\operatorname{Sym}(\Gamma_i \sqcup \Gamma_j) \wr_{\Delta_i} \operatorname{FSym}(\Delta_i)$.

\label{lem:diagonal_or_fi_k}
    Let $k \geq 1$, $G$ be a subgroup of $W_1 \times \cdots \times W_k$ with $W_i = H_i \wr_{\Delta_i} \operatorname{FSym}(\Delta_i), i = 1, \ldots, k$, and let $G_i$ denote the projection of $G$ on the component $W_i$.
    Suppose $G_i \geq W'_i$ for all $1 \leq i \leq k$.
    Then either there exist two indices $i, j$ such that $G$ acts diagonally on the pair $(\Delta_i,\Delta_j)$, or $G \geq W'_1 \times \cdots \times W'_k$.

<!-- proof-13: body_md -->
We prove the lemma by induction on $k$.
    For $k = 1$ the claim is obvious, and for $k = 2$ the claim is simply Lemma~\ref{lem:diagonal_or_fi_two}.
    For the induction step let $k \geq 3$.
    Take $G \leq W_1 \times \cdots \times W_k$ whose projections $G_i \leq W_i$ satisfy $G_i \geq W'_i$.
    Consider the projection maps 
    \[
    p \colon G \rightarrow W_1 \times \cdots \times W_{k-1}, \;\; \text{ and } \;\; p' \colon G \rightarrow G_k.
    \]
    Consider $p'(\ker(p)) \unlhd G_k$.
    By Goursat's Lemma, we have
    \[
    p(G)/p(\ker(p')) \cong G_k/p'(\ker(p)).
    \]
    Here, $p(G)$ is a subgroup of $W_1 \times \cdots \times W_{k-1}$, whose projection on $W_i$ is equal to $G_i$ for $i = 1, \ldots, k-1$.
    If $p(G)$ is diagonal on some pair $(\Delta_i, \Delta_j), 1 \leq i < j \leq k-1$, then $G$ is also diagonal on the same pair.
    Otherwise by the induction hypothesis we have 
    \[
    p(G) \geq W'_1 \times \cdots \times W'_{k-1},
    \]
    which yields finiteness of the index 
    \[
    [W_1 \times \cdots \times W_{k-1} \colon p(G)] \leq [W_1 \times \cdots \times W_{k-1} \colon W'_1 \times \cdots \times W'_{k-1}] < \infty.
    \]
    By Proposition~\ref{cor:N_contains_Wprime}, there are two cases: either $p'(\ker(p)) \geq W'_k$, or $\pi_{\Delta_k}\left(p'(\ker(p))\right) = \{\operatorname{id}\}$.

    \textbf{Case one.} If $p'(\ker(p)) \geq W'_k$ then the quotients
    $
    p(G)/p(\ker(p')) \cong G_k/p'(\ker(p))
    $
    are finite. 
    The projection of $p(\ker(p')) \unlhd p(G)$ on each component $W_i, 1 \leq i \leq k-1$, must be a finite index normal subgroup of $G_i$ and hence contains $W'_i$ (by Proposition~\ref{cor:N_contains_Wprime}).
    We can therefore apply the induction hypothesis on the subgroup $p(\ker(p'))$ of $W_1 \times \cdots \times W_{k-1}$ and obtain that either $p(\ker(p'))$ acts diagonally on some pair $(\Delta_i, \Delta_j), 1 \leq i < j \leq k-1$, or $p(\ker(p')) \geq W'_1 \times \cdots \times W'_{k-1}$.
    But since the index 
    \[
    [W_1 \times \cdots \times W_{k-1} \colon p(\ker(p'))] = [W_1 \times \cdots \times W_{k-1} \colon p(G)] [p(G) \colon p(\ker(p'))]
    \]
    is finite, the group $p(\ker(p'))$ cannot act diagonally on any pair $(\Delta_i, \Delta_j), 1 \leq i < j \leq k-1$.
    This shows that $p(\ker(p')) \geq W'_1 \times \cdots \times W'_{k-1}$.
    Therefore, $G$ contains $\ker(p) \geq \{\operatorname{id}\} \times  W'_k$ as well as $\ker(p') \geq W'_1 \times \cdots \times W'_{k-1} \times \{\operatorname{id}\}$, so $G$ contains $W'_1 \times \cdots \times W'_{k}$.

    \textbf{Case two.} If $\pi_{\Delta_k}\left(p'(\ker(p))\right) = \{\operatorname{id}\}$.
    Consider the normal subgroup $p(\ker(p')) \unlhd p(G)$.
    For $i = 1, \ldots, k-1$, the projection $\left(p(\ker(p'))\right)_{i}$ of $p(\ker(p'))$ on the component $G_i$ is a normal subgroup, so either $\pi_{\Delta_i}\left(p(\ker(p'))\right) = \{\operatorname{id}\}$ or $\left(p(\ker(p'))\right)_{i} \geq W'_i$ (by Proposition~\ref{cor:N_contains_Wprime}).
    If $\pi_{\Delta_i}\left(p(\ker(p'))\right) = \{\operatorname{id}\}$ for some $1 \leq i \leq k-1$, then $G$ acts diagonally on the pair $(\Delta_i, \Delta_k)$.
    Otherwise $\left(p(\ker(p'))\right)_{i} \geq W'_i$ for all $1 \leq i \leq k-1$ and we can apply the induction hypothesis on the subgroup $p(\ker(p'))$ of $W_1 \times \cdots \times W_{k-1}$.
    Since the quotient $p(G)/p(\ker(p')) \cong G_k/p'(\ker(p)$ is infinite, we cannot have $p(\ker(p') \geq W'_1 \times \cdots \times W'_{k-1}$.
    Therefore $p(\ker(p'))$ must act diagonally on the pair $(\Delta_i, \Delta_j)$ for some $1 \leq i < j \leq k-1$.
    We will show that this is impossible since $p(\ker(p'))$ is normal in $p(G)$.

                                                                                                                                                                            
    Take any $a \in p(\ker(p'))$ such that $\pi_{\Delta_i}(a) \neq \operatorname{id}$.
                                                     
    Since $p(G) \geq W'_1 \times \cdots \times W'_{k-1}$, we can pick an element $h \in p(G)$ such that $\pi_{\Delta_i}(h)$ does not commute with $\pi_{\Delta_i}(a)$ but $\pi_{\Delta_j}(h) = \operatorname{id}$.
    Since $p(\ker(p'))$ is normal in $p(G)$, we have $hah^{-1} \in p(\ker(p'))$, and $hah^{-1}$ acts on $\Delta_i$ as $\pi_{\Delta_i}(h)\pi_{\Delta_i}(a)\pi_{\Delta_i}(h)^{-1} \neq \pi_{\Delta_i}(a)$, and acts on $\Delta_j$ as $\pi_{\Delta_j}(a)$.
    This means that $a, hah^{-1} \in p(\ker(p'))$ act differently on $\Delta_i$ but the same on $\Delta_j$, contradicting the claim that $p(\ker(p'))$ acts diagonally on the pair $(\Delta_i, \Delta_j)$.

    We have thus showed that in all cases, either there exist two indices $i, j$ such that $G$ acts diagonally on the pair $(\Delta_i,\Delta_j)$, or $G \geq W'_1 \times \cdots \times W'_k$.
    This concludes the induction step.

\label{prop:general_infinite_orbits}
    We can effectively compute a decomposition $J_{\infty} \times \mathbb{Z} = \bigsqcup_{i = 1}^k (\Gamma_i \times \Delta_i)$ where each $\Gamma_i$ is finite and $\Delta_i$ is infinite, as well as $H_i \leq \operatorname{Sym}(\Gamma_i)$,
    such that $W'_1 \times \cdots \times W'_k \leq G_{\infty} \leq W_1 \times \cdots \times W_k$, where $W_i = H_i \wr_{\Delta_i} \operatorname{FSym}(\Delta_i)$.

Recall that $G_{\infty}$ acts on the set $J_{\infty} \times \mathbb{Z}$ in such a way that all the $G_{\infty}$-orbits $\mathcal{O}_1, \ldots, \mathcal{O}_m$ are infinite.
                                                                                               
                                                            
                               
   
                                                         
   
                
By Lemma~\ref{lem:effective_partition}, the orbit $\mathcal{O}_i$ can be effectively written as a product $\Gamma_i \times \Delta_i$ for finite $\Gamma_i$ and infinite $\Delta_i$, such that $G$ acts primitively (and hence transitively) on the set of blocks $\Delta_i$.
Under this identification, the action of $G_{\infty}$ on $\mathcal{O}_i$ induces a group $G_i \leq W_i \coloneq H_i \wr_{\Delta_i} \operatorname{FSym}(\Delta_i)$, so $G_{\infty} \leq W_1 \times \cdots \times W_m$.
By Lemma~\ref{lem:transitive_finite}, we have $G_i \geq W'_i$.

We first give a procedure to decide whether $G_{\infty}$ acts diagonally on a pair $(\Delta_i, \Delta_j)$.
Suppose $G_{\infty}$ acts diagonally on the pair $(\Delta_i, \Delta_j)$.
By Lemma~\ref{lem:diagonal_or_fi_two} we can identify the projection $G_{\{i, j\}}$ of $G_{\infty}$ on the component $W_i \times W_j$ as a subgroup of $\operatorname{Sym}(\Gamma_i \sqcup \Gamma_j) \wr_{\Delta_i} \operatorname{FSym}(\Delta_i)$, under some bijection $\xi \colon \Delta_i \rightarrow \Delta_j$. 
Let $G_i, G_j$ be respectively the projection of $G_{\{i, j\}}$ on $W_i, W_j$.
Recall that $G_i, G_j$ are respectively $\rho^{\ell_i}$ and $\rho^{\ell_j}$-invariant.
Let $\ell = \ell_i \ell_j$, so $G_{\infty}, \Delta_i, \Delta_j$ are $\rho^{\ell}$-invariant.

Since $G_{\infty}$ acts on the pair $(\Delta_i, \Delta_j)$ as $(g, \xi g \xi^{-1}), g \in \operatorname{FAlt}(\Delta_i)$ or $\operatorname{FSym}(\Delta_i)$, by $\rho^{\ell}$-invariance it also acts as $(\rho^{\ell}g\rho^{-\ell}, \rho^{\ell} \xi g \xi^{-1} \rho^{-\ell})$.
Therefore $(\rho^{\ell}g\rho^{-\ell}, \rho^{\ell} \xi g \xi^{-1} \rho^{-\ell}) = (\rho^{\ell}g\rho^{-\ell}, \xi \rho^{\ell} g \rho^{-\ell} \xi^{-1})$ for all $g \in \operatorname{FAlt}(\Delta_i)$ or $\operatorname{FSym}(\Delta_i)$.
Consequently, the identification map $\xi \colon \Delta_i \rightarrow \Delta_j$ is $\rho^{\ell}$-invariant (i.e., $\rho^{\ell} \xi = \xi \rho^{\ell}$).
Given any $\rho^{\ell}$-invariant metric of $\bigsqcup_{i=1}^k \Delta_i$ (such as the one defined by Equation~\eqref{eq:distance}), the distances $\operatorname{dist}(\delta_i, \xi(\delta_i)), \delta_i \in \Delta_i$, are bounded by the maximal diameter of the generators of $G_{\infty}$.
Note that there are only finitely many $\rho^{\ell}$-invariant bijections $\xi \colon \Delta_i \rightarrow \Delta_j$ with a given bound on $\sup_{\delta_i \in \Delta_i}\operatorname{dist}(\delta_i, \xi(\delta_i))$.
