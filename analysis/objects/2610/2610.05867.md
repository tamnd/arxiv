<!-- definition-1: body_md -->
A randomized algorithm $\mathcal{M}$ that maps $\mathcal{Z}^n$ to some space $\mathcal{R}$ is $(\epsilon,\delta)$-differentially private (DP) for $\epsilon\geq0$ and $\delta\in[0,1)$ if, for every pair of neighboring datasets ${\mathbf Z},{\mathbf Z}^\prime\in \mathcal{Z}^n$ that differ by one individual datum and every measurable $U\subset \mathcal{R}$,
    \[
\mathbb{P}(\mathcal{M}({\mathbf Z})\in U ) \leq e^{\epsilon} \cdot \mathbb{P}(\mathcal{M}({\mathbf Z}^\prime)\in U) + \delta,
    \]
    where the probability measure $\mathbb{P}$ is induced by the randomness of $\mathcal{M}$ only.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def: sub-distance}
   A non-negative function $\gamma(\cdot) : \mathcal{Z}^n \mapsto [0,\infty)$ is called a sub-distance with respect to a selector $\widehat{S}: \mathcal{Z}^n \mapsto \mathcal{J}\cup\{\perp\}$, if
\begin{equation}
\{{\mathbf Z} : \gamma({\mathbf Z}) > 0\} \subset \mathrm{sub}_0(\widehat{S}),
\qquad \mbox{and} \qquad |\gamma({\mathbf Z}) - \gamma({\mathbf Z}')| \leq D_H({\mathbf Z},{\mathbf Z}')
, \label{eq: sub-distance definition}
\end{equation}
where the region $\mathrm{sub}_0(\widehat{S})$ is defined in (\ref{eq: sub 0 set}).
<!-- end definition-2 -->
<!-- theorem-1: body_md -->
\label{thm:saturated-ptr-privacy}
Let $\epsilon>0$ and $0<\delta<1$. If $\gamma(\cdot)$ is a sub-distance for $\widehat S$, then Algorithm~\ref{alg:saturated-ptr} is $(\epsilon,\delta)$-DP for all ${\mathbf Z} \in \mathcal{Z}^n$.
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
\label{prop:finite-saturation}
Let $\epsilon>0$, $0<\delta<1$, and let $\{q_k\}_{k\geq0}$ be defined by \eqref{eq:saturated-ptr-recurrence} with $q_0=\delta$. Among all nondecreasing sequences $\{u_k\}_{k\geq0}\subset[0,1]$ with $u_0\leq\delta$ satisfying
\[
u_{k+1}\leq e^\epsilon u_k+\delta,
\qquad
1-u_k\leq e^\epsilon(1-u_{k+1})+\delta,
\qquad k\geq0,
\]
the sequence $\{q_k\}_{k\geq 0}$ is pointwise largest: $u_k\leq q_k$ for every $k\geq0$. Moreover, the saturation index
$K_{\epsilon,\delta}:=\min\{k\geq0:q_k=1\}$ is finite, and $q_k=1$ for every
$k\geq K_{\epsilon,\delta}$. If $0 < \epsilon \leq 1 $ and $0< \delta \leq \epsilon/16$, then
\[
K_{\epsilon,\delta}\leq\frac{4}{\epsilon}\log\!\left(\frac{\epsilon}{\delta}\right).
\]
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
Recurrence~\eqref{eq:saturated-ptr-recurrence} also appears in private binomial testing
\citep{Awan-NeurIPS-2018} and partition selection
\citep{Desfontaines-POPETS-2022}.
Both the count-based statistic and the stability score $\gamma$ change by at most one under data replacement.  Saturated PTR therefore transfers this construction from private testing to the PTR framework: rather than applying a generic randomized test to the stability score, it directly chooses the largest
release probability permitted by the adjacent-layer privacy constraints. To our knowledge, this is the first explicit PTR formulation of this
optimal release rule for general discrete selectors equipped with
sub-distance scores.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
The output $\perp$ represents abstention and is distinct from the empty
support $\emptyset$. It indicates that the observed stability
certificate is insufficient to guarantee release at the prescribed privacy level, rather than that the true support is empty. Throughout, a no-release output $\perp$ is counted as a failure of exact support recovery.
Operationally, this reject option can be preferable to always releasing
a potentially inaccurate support estimate, as in noisy selection. We will discuss the implications of the non-release further in the case study presented in Section \ref{sec: real-data}.
<!-- end remark-2 -->
<!-- lemma-1: body_md -->
\label{lem: PDW}
Fix $A\subset[d]$ and $q_A\in\{\pm1\}^{|A|}$, and assume $\widehat\Sigma_{AA}({\mathbf Z})$ is invertible. Define the active oracle
\begin{equation}
\vartheta_A({\mathbf Z};q_A):=\widehat\Sigma_{AA}({\mathbf Z})^{-1}\{\widehat a_A({\mathbf Z})-\lambda q_A\}.
\label{eq:active-oracle}
\end{equation}
Further define the active and inactive KKT margins
\[
m_A^{\rm primal}({\mathbf Z}):=\min_{j\in A}q_{A,j}\vartheta_{A,j}({\mathbf Z}),
\qquad
m_A^{\rm dual}({\mathbf Z}):=\lambda-\max_{j\notin A}|\widehat a_j({\mathbf Z})-\widehat\Sigma_{jA}({\mathbf Z})\vartheta_A({\mathbf Z})|.
\]
If both margins are positive, the vector $\beta\in\mathbb{R}^d$ defined by $\beta_A=\vartheta_A({\mathbf Z})$ and $\beta_{A^c}=\mathbf0$ is the unique minimizer of \eqref{eq:lasso}. In particular, \(\widehat S_\lambda({\mathbf Z})=A\) and \(\mathrm{sign}(\widehat\boldsymbol{\theta}_A^\lambda({\mathbf Z}))=q_A\).

<!-- theorem-2: body_md -->
\label{thm: radius stability}
For integers $r>0$ and $s>0$, let $A=\widehat S_\lambda({\mathbf Z})$. Suppose: (i) $|A|\leq s$; (ii) $\Psi_A^{\rm Inv}({\mathbf Z})>(\psi^{\rm Inv})^{-1}+rs\eta_\Sigma$; (iii) $\Psi_A^{\rm Param}({\mathbf Z})<\psi^{\rm Param}$; (iv) $\Psi_A^{\rm Cross}({\mathbf Z})<\psi^{\rm Cross}-rs\eta_\Sigma$; (v) $m_A^{\rm primal}({\mathbf Z})>r\tau_{\rm primal}$; and (vi) $m_A^{\rm dual}({\mathbf Z})>r\tau_{\rm dual}$. Then, for every ${\mathbf W}$ satisfying $D_H({\mathbf W},{\mathbf Z})\leq r$, \(\widehat S_\lambda({\mathbf W})=\widehat S_\lambda({\mathbf Z})\).

\label{prop:maximum-of-sub-distances}

<!-- theorem-6: body_md -->
\label{thm: deterministic beta-min}
Let $S$ be the true support of $\boldsymbol{\theta}^*$, $s = |S|$ and $q_S = \mathrm{sign}(\boldsymbol{\theta}_S^*)$. Suppose there exists a positive integer $R$ such that
  \[
  \min\left\{ \frac{\Psi_S^{\rm Inv}({\mathbf Z}) - (\psi^{\rm Inv})^{-1}}{s\eta_{\Sigma}}, \frac{\psi^{\rm Cross} - \Psi_S^{\rm Cross}({\mathbf Z})}{s\eta_{\Sigma}}, \frac{\psi^{\rm Param} - \Psi_S^{\rm Param}({\mathbf Z})}{s\tau_{\rm primal}}\right\} > R.
  \]
  Also suppose that there exists a constant $\kappa \in (0,1)$ such that $\|\widehat{\Sigma}_{S^c S}\widehat{\Sigma}_{SS}^{-1}\|_\infty \leq 1 -\kappa$. Define $\boldsymbol{\xi} = \widetilde{{\mathbf X}}^\top (\widetilde{{{\mathbf Y}}} - \widetilde{{\mathbf X}}\boldsymbol{\theta}^*) / n$. If
\[
\lambda > \frac{(2-\kappa)\|\boldsymbol{\xi}\|_\infty + R \tau_{\rm dual}}{\kappa} \quad \mbox{ and } \quad \theta_{\min} > \psi^{\rm Inv} (\|\boldsymbol{\xi}\|_\infty+\lambda) + R\tau_{\rm primal},
\]
then we have $\widehat{S}_\lambda({\mathbf Z}) = S$ and $\gamma({\mathbf Z}) \geq R$, where $\gamma(\cdot)$ is defined in (\ref{eq: sub-distance}).

\begin{enumerate}[label=(\roman*),topsep=0pt]

\label{thm: sufficient beta-min}
Fix $\alpha\in(0,1/4)$, $0<\epsilon\leq1$, and
$0<\delta\leq\epsilon/16$. Under Assumption~\ref{assumption: regularity},
take the model and regularity bounds as fixed public constants.
Fix a public constant $B_X\geq B_x$ and set
$B_Y=\tau_0\sqrt{2\log(12n/\alpha)}$, where
$\tau_0^2=\sigma_x^2a_0^2+\sigma^2$, and
$\psi^{\rm Param}=4a_0$, $\psi^{\rm Inv}=4a_1$, and
$\psi^{\rm Cross}=3a_2$.  Then there exist positive constants $c_0,C_0,C_1$,
depending only on $(a_0,a_1,a_2,a_3,B_x,\sigma_x,\sigma)$, such that, whenever
\[
n\geq
\max\left\{
\frac{s^2\log(12ds/\alpha)}{c_0^2},
\frac{(B_XB_Y+B_X^2)sK_{\epsilon,\delta}}{c_0}
\right\} \quad \mbox{ and }
\]
\[
\lambda
=
C_0\left\{
\sqrt{\frac{\log(12d/\alpha)}{n}}
+
\frac{(B_XB_Y+B_X^2)K_{\epsilon,\delta}}{n}
\right\},
\]
both the fixed-sparsity
and adaptive Saturated LASSO procedures
correctly recover $S$  with probability at least $1-\alpha$ as long as
 $\theta_{\min}\geq C_1\lambda$, where the probability includes the
sample and mechanism randomness. The constants are uniform over the
distributions satisfying the stated bounds and do not depend on
$(n,d,s,\alpha,\epsilon,\delta)$ .

\label{cor: bounded noise rate}
Under Assumption~\ref{assumption: regularity}, use the privacy and
sample-size conditions of Proposition~\ref{thm: sufficient beta-min} and
its public buffers and exact penalty. For fixed $c',c''>0$, take
$\alpha=d^{-c'}+n^{-c''}<1/4$ and the response radius specified in that
proposition. Both procedures recover $S$ with probability at least
$1-d^{-c'}-n^{-c''}$ if
\begin{equation}
\theta_{\min}\geq C\left\{
\sqrt{\frac{\log d}{n}}+
\frac{\sqrt{\log n}\log(\epsilon/\delta)}{n\epsilon}
\right\}.
\label{eq: sufficient beta-min}
\end{equation}
If additionally $|\varepsilon|\leq B_\varepsilon$ almost surely for a
public constant $B_\varepsilon$, replace $B_Y$ by
$B_xa_0+B_\varepsilon$ in the same penalty and sample-size conditions.
The finite-sample guarantee then holds for every $\alpha\in(0,1/4)$
under $\theta_{\min}\geq C_1\lambda$. In particular, taking
$\alpha=d^{-c'}<1/4$ gives recovery probability at least $1-d^{-c'}$ if
\begin{equation}
\theta_{\min}\geq C\left\{
\sqrt{\frac{\log d}{n}}+
\frac{\log(\epsilon/\delta)}{n\epsilon}
\right\}.
\label{eq: bounded noise}
\end{equation}
Here $C>0$ is sufficiently large and may depend only on the fixed public
model and tuning bounds and the displayed confidence exponents.

\label{cor:public-buffer-aggregation}
For every $t\in\mathcal T$, $\gamma_{t,\mathcal G}$ is a sub-distance for the common LASSO selector $\widehat S_\lambda$, and so is $\gamma_{{\rm ad},\mathcal G}$. Let $A=\widehat S_\lambda({\mathbf Z})$ and $a=|A|$. If $A=\emptyset$, $a>n$, or $\widehat\Sigma_{AA}({\mathbf Z})$ is singular, then $\gamma_{{\rm ad},\mathcal G}({\mathbf Z})=0$; otherwise,
\[
\gamma_{{\rm ad},\mathcal G}({\mathbf Z})
=
\sup_{g\in\mathcal G}\gamma_{a,g}({\mathbf Z}).
\]
Consequently, Saturated PTR can be applied once to the aggregated score without privacy composition over $\mathcal G$, and the exact oracle-equivalence identity in \eqref{eq:adaptive-pointwise-oracle-equivalence} continues to hold with $\gamma_s$ and $\gamma_{\rm ad}$ replaced by $\gamma_{s,\mathcal G}$ and $\gamma_{{\rm ad},\mathcal G}$, respectively.

<!-- remark-4: body_md -->
\label{rem:existing-private-methods}
The sufficient beta-min condition for the $(\epsilon,\delta)$-DP perturbation-stability LASSO of \citet{Guha-COLT-2013} contains a privacy term of order \(s^{3/2}/(n\epsilon)\), up to logarithmic factors, and its statistical recovery guarantee is stated at constant success probability. The pure-DP best subset selection (BSS) \citep{Roy-2024} and top-$R$ \citep{Prastakos-NIPS-2025} algorithms provide high-probability exact support recovery under their respective design regularity conditions, with private beta-min terms of orders $\sqrt{s\log(d)/(n\epsilon)}$ and $s\sqrt{\log(d)/(n\epsilon)}$, respectively. Both subset-selection approaches use known sparsity; exact BSS uses the exponential mechanism over all size-$s$ supports, while top-$R$ uses mixed-integer programming. The analyses of two other methods do not provide the worst-case privacy guarantee: the privacy guarantee of the mistakes procedure of \citet{Prastakos-NIPS-2025} requires a data-dependent gap condition, while the computationally efficient MCMC of \citet{Roy-2024} relies on a model-dependent rapid-mixing bound.

Under Assumption~\ref{assumption: regularity}, Theorem~\ref{cor: bounded noise rate} gives high-probability exact recovery for the computationally efficient Saturated LASSO with no explicit dependence on $s$ or $d$ in its private beta-min term. Its adaptive version has the same recovery probability without knowing $s$, with otherwise identical public tuning parameters. Complementary comparisons with Noisy IHT are provided in Section~\ref{sec:estimation-versus-recovery}.

\label{thm: lower-bound}
Assume $d\ge 3s$, $s\ge 4$, $d-s+1\ge 16$, $0<\epsilon\le 1$, and $0\le\delta\le\epsilon/16$. If
\begin{equation}
 \theta_{\min}
 \le
 \sigma\left\{
 \frac{1}{4\sqrt2}
 \sqrt{\frac{\log(d-s+1)}{n}}
 +
 \frac{\sqrt2}{48n\epsilon}
 \max_{1\le r\le s}
 \frac{
 \min\{\log\left(\binom sr\binom{d-s}{r}\right),\log(\frac{\epsilon}{\delta})\}}{\sqrt r}
 \right\},
 \label{eq:complete-lower-bound}
\end{equation}
then $\mathcal R\ge\frac{1}{2}$.

<!-- corollary-2: body_md -->
\label{cor:minimax-optimality}
Let $\mathcal P_{\rm reg}(s,\theta_{\min})$ be the subclass of
$\mathcal P(s,\theta_{\min})$ satisfying
Assumption~\ref{assumption: regularity} with fixed public regularity
constants, where $B_x>1$ and $a_1,a_2\geq1$, and let
$\mathcal P_{{\rm reg},{\rm bd}}(s,\theta_{\min})$ be its subclass with
$|\varepsilon|\leq\sigma$ almost surely. Under the dimensional and
privacy conditions of Theorem~\ref{thm: lower-bound}, both classes are
nonempty exactly when $s\theta_{\min}<a_0$. On these nonempty classes,
the lower-bound rate in \eqref{eq:complete-lower-bound} also holds up to universal constants.
The corresponding recovery guarantees
\eqref{eq: sufficient beta-min} and \eqref{eq: bounded noise} hold
uniformly under the conditions of Theorem~\ref{cor: bounded noise rate},
with $B_\varepsilon=\sigma$ in the bounded-noise case.

Consequently, under these recovery conditions, whenever the
sufficient signal threshold is smaller than $a_0/s$, both fixed-sparsity
and adaptive Saturated LASSO are minimax optimal on these classes up to
logarithmic factors in $n$ and $1/\delta$, uniformly over
$0<\delta\leq\epsilon/16$.
When
$\epsilon/\{s(d-s)\}\leq\delta\leq\epsilon/16$, the rates match up to
the $\sqrt{\log n}$ response-clipping factor under sub-Gaussian noise
and up to constants under bounded noise.
