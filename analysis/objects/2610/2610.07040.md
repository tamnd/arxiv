\label{Ex: MV}
Consider $\beta=q(\theta)$ with
$\theta=(\ensuremath W,\ensuremath w)\in\ensuremath{\mathbb{R}}^{p\times m}\times\ensuremath{\mathbb{R}}^m$ and
$q(\theta)=\ensuremath W\ensuremath w$. Gradient flow on the reparameterized problem is
\begin{align}
\label{eq:multineuron}
\dot\ensuremath W(t)
&=-\nabla_\ensuremath W{\mathcal{L}}(\ensuremath W(t)\ensuremath w(t))
=-\nabla{\mathcal{L}}(\beta(t))\ensuremath w(t)^\top,\\
\dot\ensuremath w(t)
&=-\nabla_\ensuremath w{\mathcal{L}}(\ensuremath W(t)\ensuremath w(t))
=-\ensuremath W(t)^\top\nabla{\mathcal{L}}(\beta(t)).\nonumber
\end{align}
Assume, for simplicity, that the initialization satisfies
$\ensuremath W(0)^\top\ensuremath W(0)=\ensuremath w(0)\ensuremath w(0)^\top.$

By the Poincar\'e lemma (see Lemma \ref{poincare}), there is no function $F$
satisfying \eqref{eq:jac}. Nevertheless, as observed in
\cite{azulay2021implicit}, the induced dynamics in the original variable can
be written as
\begin{equation}
\label{eq: Uvanot0 Flow H1}
\left[\nabla^2F(\beta(t))\right]\dot\beta(t)
=-G(\beta(t))\nabla{\mathcal{L}}(\beta(t)),
\end{equation}
where
\begin{equation}
\label{eq:F}
F(\beta)=\frac{2}{3}\|\beta\|^{3/2},
\qquad
G(\beta)=\|\beta\|^{1/2}.
\end{equation}
See Section \ref{SS: AFCLN} for a proof.

<!-- remark-1: body_md -->
On any interval where $G(\beta(t))>0$, the time change
\(
    s(t)=\int_0^t G(\beta(u))\,du
\)
transforms \eqref{eq: x(t)flow} into the mirror flow
\[
    \frac{d\widetilde{\beta}}{ds}
    =-\left[\nabla^2F(\widetilde{\beta})\right]^{-1}
      \nabla\mathcal{L}(\widetilde{\beta}),
    \qquad \widetilde{\beta}(s(t))=\beta(t).
\]
This connection is established for conformal mirror descent
in \cite[Theorem~2]{kainth2022conformal}
under suitable assumptions on the relation between $F$ and $G$
that we do not require. In our case, this point of view does not seem advantageous with respect to a direct analysis of \eqref{eq: x(t)flow}.

\label{Assumption}\ \\
\ \\
Let $b\in\ensuremath{\mathbb{R}}^p$. 
\begin{enumerate}[label=A\arabic*]
\par\hangindent 40\p@\label{A: D1} The function $\mathcal{L}\colon \ \ensuremath{\mathbb{R}}^{p}\to\ensuremath{\mathbb{R}}$ is convex, differentiable and $\nabla \ensuremath{\mathcal L}$ is locally Lipschitz continuous. Moreover, $\mathcal{S} \neq \emptyset$ and $b \notin \mathcal{S}$.
\par\hangindent 40\p@\label{A: D3} The initialization satisfies $\mathcal{L}(\beta_0) < \mathcal{L}(b)$. 
\par\hangindent 40\p@\label{A: D2} The function $F\colon \ensuremath{\mathbb{R}}^p \to \ensuremath{\mathbb{R}}$ is strictly-convex and twice-differentiable on $\ensuremath{\mathbb{R}}^{p}\setminus\left\{b\right\}$. For every $\beta \in \ensuremath{\mathbb{R}}^{p}\setminus\left\{b\right\}$, $\nabla^{2}F(\beta)$ is invertible and the map $\beta\in\ensuremath{\mathbb{R}}^{p}\setminus\left\{b\right\} \mapsto (\nabla^{2}F(\beta))^{-1}$ is locally Lipschitz continuous. Moreover, the function $G: \ \ensuremath{\mathbb{R}}^{p}\setminus\left\{b\right\} \to \ensuremath{\mathbb{R}}_{+}$ is locally Lipschitz continuous. For every radius $R>0$ there exists a constant $C(R)>0$ such that, for all $\beta\not\in\mathbb{B}(b,R)$, $G(\beta)\geq C(R)$.
\par\hangindent 40\p@\label{A: D4} There exists $\bar{z}\in \mathcal{S}$ such that the function $D_{F}(\bar{z},\cdot)$ is coercive, where $D_{F}$ is the Bregman divergence associated with $F$ (see \cite{bregman1966relaxation}), defined as
\begin{align}
    D_{F}: \ \ensuremath{\mathbb{R}}^{p}\times\ensuremath{\mathbb{R}}^{p}\setminus\left\{b\right\} \to \ensuremath{\mathbb{R}}, \quad D_F(y,\beta)= F(y)-F(\beta)-{\left\langle{{\nabla F(\beta)}, \ {y-\beta}}\right\rangle}.
  \label{Def:BDA}
\end{align}
\end{enumerate}

\label{T: Existence}
Under Assumption \ref{Assumption}, the dynamical system \eqref{eq: x(t)flow} has a unique solution $\beta$ defined on $[0,+\infty)$.
Moreover, for every ${z}\in \mathcal{S}$, both the Bregman divergence $D_{F}(z,\beta(t))$ and $\mathcal{L}(\beta(t))$ are non-increasing functions of $t\geq 0$.

<!-- theorem-2: body_md -->
\label{T: Main result}
Assume that Assumption \ref{Assumption} holds and that $\beta$ is the unique solution of the dynamical system \eqref{eq: x(t)flow}. Then $\mathcal{L}(\beta(t))$ is non-increasing and $\mathcal{L}(\beta(t))\to \mathcal{L}^*$ for $t\to+\infty$. In addition, $\left[\mathcal{L}(\beta(t))-\mathcal{L}^{*}\right]\in L^{1}([0,+\infty))$, and, for every ${z}\in \mathcal{S}$ and $t > 0$,
\begin{align*}
  \ensuremath{\mathcal L}(\beta(t))-\ensuremath{\mathcal L}^{*} \quad \leq \quad \frac{D_{F}(z,\beta_0)}{C(r) t },
\end{align*}
where $C$ is the function introduced in Assumption \ref{A: D2} and $r:=\frac{\mathcal{L}(b)-\mathcal{L}(\beta_0)}{2 \|\nabla \mathcal{L}(b)\|}$.
Moreover, $\lim_{t\rightarrow +\infty}\beta(t)=\beta_{\infty}$ where
\begin{align}\label{minbreg1}
\left\{\beta_{\infty}\right\} \ = \ \operatorname*{\text{\rm argmin}}\limits_{z\in\mathcal{S}} \left\{ D_{F}(z,\beta_0)+\int_{0}^{+\infty}G(\beta(t))\nabla\mathcal{L}(\beta(t)), z \big\rangle dt\right \}.
\end{align}

\label{Remark: Saving energy} Suppose that Assumption \ref{Assumption} holds and let  $\beta$ be the unique solution of the dynamical system \eqref{eq: x(t)flow}. Let $X\in \ensuremath{\mathbb{R}}^{d\times p}$ and let $\ell\colon\ensuremath{\mathbb{R}}^d\to \ensuremath{\mathbb{R}}$ be differentiable. Assume that $\ensuremath{\mathcal L}=\ell \circ X$ and $\mathcal{S} = \left\{\beta\in\ensuremath{\mathbb{R}}^{p}\mid X\beta=y\right\}$ for some $y\in\ensuremath{\mathbb{R}}^{d}$. Let $P: \ \ensuremath{\mathbb{R}}^p \to \ensuremath{\mathbb{R}}^p$ be the projection operator onto the kernel of $X$. Then, for every $t>0$,
\begin{align}
\frac{d}{dt}\left(P(\nabla F\left(\beta(t)\right)\right)&=0.
\nonumber
\end{align}
Moreover, $\lim\limits_{t\rightarrow+\infty}\beta(t)=\beta_{\infty}$ with
\begin{align*} \{ \beta_{\infty} \} \ = \ \operatorname*{\text{\rm argmin}}\limits_{z\in\mathcal{S}} \ D_F(z,\beta_0). 
\end{align*}

<!-- proof-1: body_md -->
Using linearity of $P$, the dynamical system and the fact that $\ensuremath{\operatorname{ran}}(X^{\top})=\ker(X)^{\perp}$, we get
\begin{align*}
\frac{d}{dt}\left(P(\nabla F\left(\beta(t)\right)\right)& = P\left(\nabla^2 F\left(\beta(t)\right)\right)\dot{\beta}(t)\\
&=P\left(-G(\beta(t))\nabla \mathcal{L}\left(\beta(t)\right)\right)\dot{\beta}(t) \\
&= -G(\beta(t))P\left(X^{\top}\nabla \mathcal{\ell}\left(X\beta(t)\right)\right) \dot{\beta}(t)= 0.
\end{align*}
It follows from Theorem \ref{T: Main result} that $\lim\limits_{t\rightarrow+\infty} \beta(t)=\beta_{\infty}$ with 
\begin{align*}
\left\{ \beta_{\infty}\right\} = \operatorname*{\text{\rm argmin}}\limits_{z\in\mathcal{S}} \left\{D_F(z,\beta_0)+\int_{0}^{+\infty}G(\beta(t)){\left\langle{{\nabla\mathcal{L}(\beta(t))}, \ {z-\beta(t)}}\right\rangle}dt \right\}.
\end{align*}
To conclude the assertion, it is sufficient to prove that the integral is constant with respect to $z$. By the chain rule, for every $z\in\mathcal{S}$,
\begin{align*}
\int_{0}^{+\infty}G(\beta(t)){\left\langle{{\nabla\mathcal{L}(\beta(t))}, \ {z-\beta(t)}}\right\rangle}dt=&\int_{0}^{+\infty}G(\beta(t)){\left\langle{{X^{\top}\nabla \ell(X\beta(t))}, \ {z-\beta(t)}}\right\rangle}dt\nonumber\\=&\int_{0}^{+\infty}G(\beta(t)){\left\langle{{\nabla \ell(X\beta(t))}, \ {y-X\beta(t)}}\right\rangle}dt
\end{align*}
which is independent of $z\in\mathcal{S}$. Consequently, from \eqref{minbreg1}, we obtain that the limit of the trajectory satisfies
\begin{align*}
   \left\{ \beta_{\infty} \right\} = \operatorname*{\text{\rm argmin}}\limits_{z\in\mathcal{S}} \quad D_F(z,\beta_0).
\end{align*}\hfill

<!-- remark-4: body_md -->
Consider the case of $\ensuremath{\mathcal L}=\ell \circ X$ with $X$ linear operator. A sufficient condition for the set of minimizers of $\ensuremath{\mathcal L}$ to be of the form $\mathcal{S} = \left\{\beta\in\ensuremath{\mathbb{R}}^{p}\mid X\beta=y\right\}$ for some $y\in\ensuremath{\mathbb{R}}^{d}$ is to assume that $\ell$ is strictly-convex. Indeed, denote by $y$ the unique minimizer of the problem $$\min_{p\in \ensuremath{\operatorname{ran}}(X)} \ell(p).$$
If $X\bar{\beta}=y$ then $\bar{\beta} \in \operatorname*{\text{\rm argmin}} \ensuremath{\mathcal L}$. Now we want to show that if $\bar{\beta} \in \operatorname*{\text{\rm argmin}} \ensuremath{\mathcal L}$ then $X\bar{\beta}=y$. We have $$\ell(X\bar{\beta})=\ensuremath{\mathcal L}(\bar{\beta})=\min_{\beta\in\ensuremath \mathbb{R}^p} \ensuremath{\mathcal L}(\beta)=\min_{\beta \in \ensuremath \mathbb{R}^p} \ell(X\beta)=\min_{p\in \ensuremath{\operatorname{ran}}(X)} \ell(p)=\ell(y).$$
Since $y$ is the unique minimizer of $\ell$ on $\ensuremath{\operatorname{ran}}(X)$, $X\bar{\beta}\in\ensuremath{\operatorname{ran}}(X)$ and $\ell(X\bar{\beta})=\ell(y)$, then  $X\bar{\beta} = y$.

<!-- theorem-3: body_md -->
Let  Assumption~\ref{A: D1} and \ref{A: D3} hold with $b=0$ and  $(\ensuremath W(t),\ensuremath w(t))$ be the solution of \eqref{eq: UVector-Flow v-WNN} with
\begin{align}
\ensuremath W^{\top}_0\ensuremath W_0=\ensuremath w_0\ensuremath w^{\top}_0.
\label{Balance initialization}
\end{align}
Then $\beta(t)=\ensuremath W(t)\ensuremath w(t)$ satisfies \eqref{eq: x(t)flow} with $F(\beta)=(2/3)\|\beta\|^{3/2}$ and $G(\beta)=\|\beta\|^{1/2}$.

<!-- proof-2: body_md -->
If we multiply the expression of $\Dot{\ensuremath W}(t)$ on the left by $\ensuremath W^{\top}(t)$ and the one of $\Dot{\ensuremath w}(t)$ on the right by $\ensuremath w^{\top}(t)$, we get that
\begin{align}
      \ensuremath W^{\top}(t)\Dot{\ensuremath W}(t)=- \ensuremath W^{\top}(t)\nabla\mathcal{L}(\beta(t))\ensuremath w^{\top}(t)=\Dot{\ensuremath w}(t)\ensuremath w^{\top}(t).\nonumber
\end{align}
Adding to the previous equality the one obtained by transposing it, we have that,
\begin{align}
     \frac{d}{dt}\left(\ensuremath W^{\top}(t)\ensuremath W(t)\right)=\ensuremath W^{\top}(t)\Dot{\ensuremath W}(t)+\dot{\ensuremath W}^{\top}(t)\ensuremath W(t)=\Dot{\ensuremath w}(t)\ensuremath w^{\top}(t)+\ensuremath w(t)\Dot{\ensuremath w}^{\top}(t)=\frac{d}{dt}\left(\ensuremath w(t)\ensuremath w^{\top}(t)\right).\nonumber
\end{align}  
Then, from the latter equation and the initialization $\ensuremath W^{\top}(0)\ensuremath W(0)=\ensuremath w(0)\ensuremath w^{\top}(0)$, we deduce that, for every $t\geq 0$, 
\begin{align}
 \ensuremath W^{\top}(t)\ensuremath W(t)=\ensuremath w(t)\ensuremath w^{\top}(t).\label{eq: Matrix-eqq}\end{align}
Multiplying by $\ensuremath w(t)$ on the right, by $\ensuremath w^{\top}(t)$ on the left and recalling that $\beta(t)=\ensuremath W(t)w(t)$, we have that $\|\beta(t)\|^{2}=\|\ensuremath w(t)\|^{4}$ and so $\|\beta(t)\|=\|\ensuremath w(t)\|^{2}.$ 
\\ \\Analogously, multiplying \eqref{eq: Matrix-eqq} by $\ensuremath W(t)$ on the left and by $\ensuremath W^{\top}(t)$ on the right, we deduce that $(\ensuremath W(t)\ensuremath W^{\top}(t))^{2}=\beta(t)\beta^{\top}(t)$. Since both $\ensuremath W(t)\ensuremath W^{\top}(t)$ and $\beta(t)\beta^{\top}(t)$ are symmetric positive semi-definite, equality $(\ensuremath W(t)\ensuremath W^{\top}(t))^{2}=\beta(t)\beta^{\top}(t)$, implies
\begin{equation*}\ensuremath W(t)\ensuremath W^{\top}(t)=\frac{\beta(t)\beta^{\top}(t)}{\|\beta(t)\|}.\end{equation*}
Equation~\eqref{eq: UVector-Flow v-WNN} yields
\begin{align}
\dot{\beta}(t)=\dot{\ensuremath W}(t)\ensuremath w(t)+\ensuremath W(t)\dot{\ensuremath w}(t)=-\nabla \mathcal{L}(\beta(t)) \|\ensuremath w(t)\|^{2}- \ensuremath W(t)\ensuremath W^{\top}(t)\nabla\mathcal{L}(\beta(t)).\label{eq: Matrix-eq33}
     \end{align}
Plugging in \eqref{eq: Matrix-eq33} the expressions of $\|\ensuremath w(t)\|^{2}$
 and $\ensuremath W(t)\ensuremath W^{\top}(t)$ in terms of $\beta(t)$, we get
\begin{align}
\dot{\beta}(t)=&-\left(\|\beta(t)\|\ensuremath{\operatorname{Id}}\,+\frac{\beta(t)\beta^{\top}(t)}{\|\beta(t)\|} \right)\nabla\mathcal{L}(\beta(t)).
     \label{eq: Uvanot0 Flow11}
     \end{align}
Now notice that the Hessian of $F$ and its inverse are given by
\begin{equation*}
\nabla^2F(\beta)=\|\beta\|^{-1/2}\left(\ensuremath{\operatorname{Id}}\,-\frac{\beta\beta^{\top}}{2\|\beta\|^{2}}\right)\hspace{5mm}\text{ and }\hspace{5mm} \left[\nabla^2F(\beta)\right]^{-1}=\|\beta\|^{1/2}\left(\ensuremath{\operatorname{Id}}\,+\frac{\beta\beta^{\top}}{\|\beta\|^{2}}\right).
\end{equation*} 
Then, multiplying \eqref{eq: Uvanot0 Flow11} on the left by $\left[\nabla^2F(\beta)\right]$, we obtain the following time-warped dynamical system:
\begin{align} \label{eq: Uvanot0 Flow H} 
   \nabla^{2}\left(\frac{2}{3}{\|\beta(t)\|^{3/2}}\right)\dot{\beta}(t)=&-\|\beta(t)\|^{1/2}\nabla\mathcal{L}(\beta(t)),
\end{align}
which is indeed \eqref{eq: x(t)flow} with $F$ and $G$.\hfill

<!-- remark-5: body_md -->
The preceding proof illustrates the role of conservation laws
in obtaining intrinsic dynamics, as studied systematically in
\cite{marcotte2025intrinsic}.
The conserved quantity
\(
    \ensuremath W(t)^\top\ensuremath W(t)-\ensuremath w(t)\ensuremath w(t)^\top
\)
and the balanced initialization allow us to express the
induced dynamics solely in terms of $\beta=\ensuremath W\ensuremath w$:
\[
    \dot{\beta}=-K(\beta)\nabla\mathcal{L}(\beta),
    \qquad
    K(\beta)=\|\beta\|\ensuremath{\operatorname{Id}}\,+
           \frac{\beta\beta^\top}{\|\beta\|}.
\]
An intrinsic representation does not, by itself, guarantee
a mirror-flow representation.
Here, for $\beta\neq 0$, the additional identity
\(
    K(\beta)=G(\beta)\left[\nabla^2F(\beta)\right]^{-1}
\)
provides the time-warped mirror structure needed to apply
our well-posedness, convergence, and implicit-bias results.

For $b=0$, Assumption \ref{A: D2} is satisfied. In particular, it holds with $C(R)=\sqrt{R}$. Next we show that assumption \ref{A: D4} holds, namely that there exists $\bar{z}\in\mathcal{S}$ such that $D_{F}(\bar{z},\cdot)$ is coercive. Indeed, the Bregman divergence $D_{F}(z,\cdot)$ is coercive for every $z\in\ensuremath{\mathbb{R}}^p$: from the Cauchy-Schwarz inequality, 
\begin{align*}
D_{F}(z,\beta)=&\hspace{1mm}\frac{2}{3}\|z\|^{3/2}-\frac{2}{3}\|\beta\|^{3/2}-\|\beta\|^{-1/2}{\left\langle{{\beta}, \ {z-\beta}}\right\rangle}\nonumber\\

\end{align*}
Then, when the norm of $\beta$ tends to infinity, $D_{F}(z,\beta)$ also tends to infinity. The statement follows from Theorem \ref{T: Existence} and Theorem \ref{T: Main result}.\hfill

<!-- remark-11: body_md -->
By \eqref{eq: MF-Norm}, we get that

Consider the two layers vector-vector parameterization; i.e., $\beta=q(\theta)$ with $\theta=(\ensuremath \theta_1,\ensuremath \theta_2)\in \ensuremath{\mathbb{R}}^k \times \ensuremath{\mathbb{R}}^k$ and $q(\theta)=\ensuremath \theta_1\astrosun\ensuremath \theta_2$, which is indeed an overparameterization because there are more learnable parameters than variables to predict. \begin{align}

     \beta}\mathcal{L}(\beta(t))\astrosun \ensuremath \theta_1(t)\astrosun \ensuremath \theta_2(t)=\ensuremath \theta_2(t)\astrosun\Dot{\ensuremath \theta_2}(t).\nonumber

\label{L: Loss-Decreasing} Suppose that there exists a solution $\beta(t)$ of \ref{eq: x(t)flow} for $t \in \left[0,T\right)$. <!-- proof-6: body_md -->
The proof is the same in both cases. To prove that $t\mapsto D_{F}(z,\beta(t))$ is non-increasing, it is sufficient to note that, for every $t\in [0,T)$,
\begin{align}
\frac{d}{dt} D_{F}(z,\beta(t))=&\hspace{1mm}-{\left\langle{{\nabla F(\beta(t))}, \ {\dot{\beta}(t)}}\right\rangle} -{\left\langle{{\nabla^{2} F(\beta(t))\dot{\beta}(t)}, \ {z-\beta(t)}}\right\rangle}+{\left\langle{{\nabla F(\beta(t))}, \ {\dot{\beta}(t)}}\right\rangle}\nonumber\\
=&\hspace{1mm}-{\left\langle{{\nabla^{2} F(\beta(t))\dot{\beta}(t)}, \ {z-\beta(t)}}\right\rangle}\nonumber\\
=&\hspace{1mm}G(\beta(t)){\left\langle{{\nabla\mathcal{L}(\beta(t))}, \ {z-\beta(t)}}\right\rangle}\label{eq: derivative DF}\\ 
\leq& \ G(\beta(t))(\mathcal{L}(z)-\mathcal{L}(\beta(t)))\nonumber\\ \leq&\hspace{1mm} 0,\nonumber
\end{align}
where we used that $\beta$ is a solution of \eqref{eq: x(t)flow}, the gradient inequality for the convex function $\mathcal{L}$ and the fact that, for every $t\in [0,T)$,
 $\mathcal{L}(\beta(t))\geq \mathcal{L}(z)$ and $G(\beta(t))\geq 0$.
On the other hand, for $t\in [0,T)$, $\nabla^2 F(\beta(t))$ is invertible (and the inverse is positive definite). Then,
\begin{align}

\end{align}
which completes the proof.\hfill

    The proof of the fact $T_M >0$ is analogous to the one in Theorem~\ref{T: Existence}. Now we show that $T_{M}=+\infty$. Let $K$ be the closure of $\beta([0,T_M))$. Lemma \ref{L: Loss-Decreasingp} and Assumption~\ref{A: D4p} imply that $\beta([0,T_M))$ is bounded; therefore, $K$ is compact. Recall that the trajectory $\beta$ is continuous and that $\beta([0,T_M))$ is a subset of $\mathrm{int}(\operatorname*{\ensuremath{\text{\rm dom}}}(F))$. If $K\subset \mathrm{int}(\operatorname*{\ensuremath{\text{\rm dom}}}(F))$, by applying \cite[Proposition 1.57]{barreira2012ordinary}, we conclude that $T_{M}=+\infty$. Otherwise, there exists a point $\bar{\beta}\in \overline{\operatorname*{\ensuremath{\text{\rm dom}}}(F)} \setminus \mathrm{int}(\operatorname*{\ensuremath{\text{\rm dom}}}(F))$ for which there exists a sequence $(t_{j})_{j=1}^{+\infty}\subset \left[0,T_M\right)$ with $t_{j}\rightarrow T_{M}$ and $\beta(t_{j})\rightarrow \bar{\beta}$ for $j\to +\infty$. Integrating in time equation \eqref{eq: x(t)flow}, we get that, for every $j\in\ensuremath{\mathbb N}$,
    \begin{align}
      \|\nabla F(\beta(t_{j}))-\nabla F(\beta_{0})\|=\left\|\int^{t_{j}}_{0}G(\beta(t))
\nabla\mathcal{L}(\beta(t))dt\right\|&\leq \int^{t_{j}}_{0}G(\beta(t))\left\|
\nabla\mathcal{L}(\beta(t))\right\|dt\nonumber\\ & \leq T_{M}\max_{\beta\in K}G(\beta)\max_{\beta\in K}\|\nabla\mathcal{L}(\beta)\|.\label{Bounded derivative}
\end{align} 
The maxima are well-defined due to the fact that the functions $G$ and $\lVert\nabla \mathcal{L}\rVert$ are locally Lipschitz and so continuous, implying that they are bounded on the compact set $K$. For $j\to + \infty$, we have that $\beta(t_j) \to \bar{\beta} \in \overline{\operatorname*{\ensuremath{\text{\rm dom}}}(F)} \setminus \mathrm{int}(\operatorname*{\ensuremath{\text{\rm dom}}}(F))$ and therefore, since $F$ is a Legendre function, $\nabla F(\beta(t_{j}))\to+\infty$. Thus, the left-hand side of \eqref{Bounded derivative} tends to infinity as well, and this implies that $T_{M}=+\infty$.
 \ \\
\textbf{Uniqueness.} Since the right-hand side of \eqref{eq: x(t)flowI} is locally Lipschitz
at every point, local uniqueness holds for every initial state, and therefore globally.\hfill

Let $z\in \mathcal{S}$. By the Fundamental Theorem of Calculus and inequality \eqref{eq: derivative DF1}, we have that, for all $t\geq 0$,
\begin{align}\label{breg_div1}
   D_{F}(z,\beta(t))-D_{F}(z,\beta_0)=&\int_{0}^{t}\frac{d}{ds}D_{F}(z,\beta(s)) \ ds\\
   \leq& -\int_{0}^{t}G(\beta(s))(\ensuremath{\mathcal L}(\beta(s))-\ensuremath{\mathcal L}^{*})ds \nonumber \\
   \leq& -C\int_{0}^{t}(\ensuremath{\mathcal L}(\beta(s))-\ensuremath{\mathcal L}^{*})ds
   \leq0\nonumber,
   \end{align} 
where the last inequality holds due to Assumption \ref{A: D2p}.
Since $D_{F}(z,\beta(t))$ is non-negative, from the above inequality we derive that, for every $t\geq 0$,
   \begin{equation*}
   \int_{0}^{t}(\ensuremath{\mathcal L}(\beta(s))-\ensuremath{\mathcal L}^{*})ds\leq \frac{D_{F}(z,\beta_0)}{C}.
  \end{equation*}
In particular, $t\mapsto \left[\ensuremath{\mathcal L}(\beta(t))-\ensuremath{\mathcal L}^{*}\right]$ belongs to $L^1\left(\left[0,+\infty\right)\right)$. Moreover, since $t\mapsto \ensuremath{\mathcal L}(\beta(t))$ is non-increasing by Lemma \ref{L: Loss-Decreasingp}, we have that $\left(\ensuremath{\mathcal L}(\beta(t))-\ensuremath{\mathcal L}^{*}\right)\cdot t \leq \int_{0}^t (\ensuremath{\mathcal L}(\beta(s))-\ensuremath{\mathcal L}^{*})ds\leq D_{F}(z,\beta_0)/C$ for every $t\geq 0$. From the latter, we obtain the rate stated in the theorem.\\

Next, we prove the convergence of the trajectory $(\beta(t))_{t>0}$. Let $\bar{z}$ be the point in Assumption \ref{A: D4p}. By Lemma \ref{L: Loss-Decreasingp}, $D_{F}(\bar{z},\beta(t))$ is (non-negative and) non-increasing, and therefore there exists $\lim\limits_{t\rightarrow +\infty}D_{F}(\bar{z},\beta(t))\in\ensuremath{\mathbb{R}}$. Since $D_{F}(\bar{z},\cdot)$ is coercive by Assumption~\ref{A: D4p}, we have that $\beta(t)$ is bounded. Then, there exist a sequence $\left(t_{k}\right)_{k\geq0}$ and a point $\beta_*$ such that, for $k\to +\infty$, $t_k\to +\infty$ and $\beta(t_k)\to\beta_*$. Since $\beta(t_k)\in\mathrm{int}(\operatorname*{\ensuremath{\text{\rm dom}}}(F))$, and $\operatorname*{\ensuremath{\text{\rm dom}}}(F)$ is closed by assumption  \ref{A: D2p}, $\beta_*\in\operatorname*{\ensuremath{\text{\rm dom}}}(F)$. Moreover, since $\ensuremath{\mathcal L}$ is continuous and $\ensuremath{\mathcal L}(\beta(t))-\ensuremath{\mathcal L}^{*} \leq D_{F}(z,\beta_0)/(Ct)$, we deduce that 
\begin{equation*}\ensuremath{\mathcal L}(\beta_*)=\lim\limits_{k\to + \infty} \ensuremath{\mathcal L}(\beta(t_k))= \lim\limits_{t\to +\infty} \ensuremath{\mathcal L}(\beta(t))=\ensuremath{\mathcal L}^*;\end{equation*} 
   i.e., that $\beta_*\in\mathcal{S}$. We are going to show that indeed $\lim\limits_{t\rightarrow \infty}\beta(t)=\beta_*$. \\
Suppose that there exists another sequence $\left(t'_{n}\right)_{n=1}^{\infty}$ such that $t'_n\to +\infty$ and $\lim\limits_{n\to +\infty}\beta(t'_n) = \bar{\beta}$. With the same reasoning as above, we know that $\bar{\beta}$ belongs to $\mathcal{S}$. The existence of $\lim\limits_{t\to +\infty}D_{F}(\beta_*,\beta(t))$ yields
\begin{align}
 &\lim\limits_{n\to +\infty}D_{F}(\beta_*,\beta(t'_{n}))=\lim\limits_{t\to +\infty}D_{F}(\beta_*,\beta(t))=\lim\limits_{k\to +\infty}D_{F}(\beta_*,\beta(t_k))=0,\nonumber
\end{align}
where the last equality is given by Assumption \ref{A: D2p}. Again by Assumption \ref{A: D2p}, $\lim\limits_{n\to +\infty}D_{F}(\beta_*,\beta(t'_{n}))=0$ gives $\bar{\beta}=\beta_*$ and therefore $\lim\limits_{t\rightarrow +\infty}\beta(t)=\beta_*$.

\ \\
To conclude, note that, defining $\phi(z):=\lim\limits_{t\to +\infty} D_{F}(z,\beta(t))$, $\phi(z) \geq 0$ and $\phi(\beta_*)=0$. Then,
\begin{align}    \beta_*\in&\operatorname*{\text{\rm argmin}}\limits_{z\in\mathcal{S}} \ \phi(z). \nonumber
\end{align}
By \eqref{breg_div1} and \eqref{eq: derivative DF1}, 
\begin{align}
    \phi(z)\nonumber&=\lim\limits_{t\to +\infty} D_{F}(z,\beta(t))\nonumber= \lim\limits_{t\to +\infty} D_F(z,\beta_0)+\int_{0}^{t}G(\beta(s)){\left\langle{{\nabla\mathcal{L}(\beta(s))}, \ {z-\beta(s)}}\right\rangle}ds\nonumber \\
    & = F(z)-F(\beta_0)-{\left\langle{{\nabla F(\beta_{0})}, \ {z-\beta_0}}\right\rangle}+\int_{0}^{+\infty}G(\beta(s)){\left\langle{{\nabla\mathcal{L}(\beta(s))}, \ {z-\beta(s)}}\right\rangle}ds.\nonumber
\end{align}
Discarding the terms that are constant with respect to $z$, we obtain the claim.\hfill
