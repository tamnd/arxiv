<!-- proposition-1: body_md -->
\label{proposition:endpoint-conditions} Let the minimizer of Problem~\eqref{prob:main} be denoted by $[0,T]\ni t \mapsto (u_t^\star, p_t^\star)$.
   Also let $\gamma_0({\rm d}x, {\rm d}v) = \gamma_0(x,v)\, {\rm d}x \, {\rm d}v$ and
   \begin{align} \label{eq:quanti1}
     \xi_0(x,v) := 1 - \Lambda_0(x,v) + \log \frac{p_0^\star(x,v)}{\gamma_0(x,v)},
   \end{align} 
   for $p_0^\star(x,v) >0$.
Then, whenever the minimizer exists, 
   \begin{align} 
       \xi_0(x,v) &= \xi_0(x)  \label{eq:xi_0}
   \end{align} 
   for a.e. $(x,v) \in \mathbb R^{2d}$ at which $p_0^\star(x,v)>0$, and
   \begin{align}
       \Lambda_T(x,v) &= \Lambda_T(x), \label{eq:Lambda_T}
   \end{align} for a.e.  $(x,v) \in \mathbb R^{2d}$. That is, $\xi_0, \Lambda_T$ are independent of the velocity $v$ almost everywhere in their respective domains.

<!-- proof-1: body_md -->
The endpoint boundary conditions \eqref{eq:BC1} and \eqref{eq:BC2} define the surfaces $\mathcal S(q_0)$ and $\mathcal S(q_T)$, as in \eqref{eq:surface}. Thus, at the minimizer $t \mapsto (u_t^\star, p_t^\star)$,  it is necessary that $p_0^\star \in \mathcal S(q_0)$ and $p_T^\star \in \mathcal S(q_T)$.


Any admissible variations for the optimal value $p_0^\star$ and $p_T^\star$, namely $\delta p_0^\star$ and $\delta p_T^\star$, must be tangent to $\mathcal S(q_0)$ and $\mathcal S(q_T)$; cf.~\cite[Section 5.1]{kirk2004optimal}.  That is,
 \begin{align}\label{eq:admissible-variations}
     \delta p_0^\star \in T_{p_0^\star} (\mathcal S(q_0)), \quad   \delta p_T^\star \in T_{p_T^\star} (\mathcal S(q_T)).
 \end{align}
 This means
 \begin{align} \label{eq:zero-integral-variations}
     \int_{\mathbb R^{d}} \delta p_0^\star  \, {\rm d} v =   \int_{\mathbb R^{d}} \delta p_T^\star \, {\rm d} v =0.
 \end{align}
 
Let us define the projection mapping
 \begin{align}
    &\pi: L^1(\mathbb R^{2d}) \to L^1(\mathbb R^d), \nonumber \\
 \text{where }  \quad  &\pi (p)(x) = \int_{\mathbb R^{d}} p(x,v) \, {\rm d} v, \label{eq:pi-definition}
\end{align}
which is a bounded linear operator. 
By \eqref{eq:zero-integral-variations}, we have
\begin{align}
    \delta p_0^\star, \delta p_T^\star  \in \mathcal N (\pi),
\end{align}
where $\mathcal N(\pi)$ denotes the null space of $\pi$.  

 
The first-order necessary conditions of optimality for $p_0^\star,p_T^\star$ are defined by the vanishing variation of $\mathcal L$ in \eqref{eq:Lagrangian2} with respect to $p_0,p_T$. These are
\begin{subequations}\label{eq:orthogonality}
\begin{align}
    \int_{\mathbb R^{2d}} \xi_0(x,v) \delta p_0^\star(x,v) \, {\rm d}x \, {\rm d}v =0, \\
  \int_{\mathbb R^{2d}}  \Lambda_T(x,v)  \delta p_T^\star(x,v) \, {\rm d}x \, {\rm d }v =0,
\end{align}
\end{subequations}
where $\xi_0$ is given in \eqref{eq:quanti1} and $\delta p_0^\star$ and $\delta p_T^\star$ are as in \eqref{eq:admissible-variations}. Consequently, the quantities $\xi_0, \Lambda_T$ are \emph{orthogonal} to the admissible variations $ \delta p_0^\star,  \delta p_T^\star$ in the sense of \eqref{eq:orthogonality}.
                                                                                                                                                                                                                                
This means
\begin{align}
    \xi_0 , \Lambda_T \in \overline{\mathcal R(\pi^\dagger)},
\end{align}
where $\overline{\mathcal R(\pi^\dagger)}$ denotes the closure of the range of the adjoint of $\pi$. This adjoint is defined as 
                                                                     
\begin{align}
    \int_{\mathbb R^d} (\pi f)(x) g(x) \, {\rm d}x  = \int_{\mathbb R^{2d}} f (x,v) (\pi^\dagger g)(x,v) \, {\rm d}x \, {\rm d}v, 
\end{align}
for all $f \in L^1(\mathbb R^{2d})$ and $g \in (L^1(\mathbb R^{d}))^\dagger=L^\infty(\mathbb R^{d})$. Equivalently, by \eqref{eq:pi-definition}, 
    \begin{align*}
       \int_{\mathbb R^{2d}} f (x,v) &  \, g(x) \, {\rm d} v \, {\rm d} x \nonumber \\
       &=   \int_{\mathbb R^{2d}} f (x,v) (\pi ^\dagger g)(x,v) \, {\rm d} v  \,  \, {\rm d} x.
    \end{align*}
Since $f$ is arbitrary, the previous equation implies 
\begin{align}
    (\pi^\dagger g)(x,v) = g(x),
\end{align}
and $\mathcal R(\pi^\dagger)$ is the set of bounded measurable functions that are independent of the velocity coordinate.
Then, indeed, $\mathcal R(\pi^\dagger)= \overline{\mathcal R(\pi^\dagger)}$. This completes the proof.

<!-- proposition-2: body_md -->
\label{proposition:controller}
Any minimizer $[0,T] \ni t \mapsto (u_t^\star, p_t^\star)$ of Problem~\eqref{prob:main}, whenever it exists, must satisfy
   \begin{align}
   u_t^\star(x,v) &= \sigma^2 \nabla_v \Lambda_t (x,v),  \label{eq:u-interacting-Lambda}
   \end{align} 
   for a.e. $(t,x,v) \in [0,T] \times \mathbb R^{2d}$, at which $p_t^\star(x,v)>0$. Additionally,
   \begin{align} \label{eq:u_T}
     u_T^\star (x,v) =0, 
   \end{align}
   for a.e. $(x,v) \in \mathbb R^{2d}$ at which $p_T^\star(x,v)>0$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
The first variation of $\mathcal L$ in \eqref{eq:Lagrangian2} with respect to $u_\cdot$ is given by
\begin{align*}
 \!\!\! \!  \delta \mathcal L (u_\cdot, \delta u_\cdot) = \int_0^T \!\!\!  \int_{\mathbb R^{2d}}  p_t \Big \langle \Big(\frac{u_t}{\sigma^2} - \nabla_v \Lambda_t\Big) , \delta u_t  \Big \rangle  \, {\rm d} x \, {\rm d}  v  \, {\rm d}t.
\end{align*}
At the minimizer, $\delta \mathcal L (u_\cdot^\star, \delta u_\cdot^\star) =0$, which implies \eqref{eq:u-interacting-Lambda}.
Furthermore, by \eqref{eq:u-interacting-Lambda} and the terminal condition \eqref{eq:Lambda_T} in Proposition~\ref{proposition:endpoint-conditions}, we see that $\nabla_v \Lambda_T = 0$. Therefore, Eq.~\eqref{eq:u_T} follows.

<!-- theorem-1: body_md -->
\label{thm:main}
   Whenever a minimizer $[0,T] \ni t \mapsto (u_t^\star, p_t^\star)$ of Problem~\eqref{prob:main} exists, there exist $\varphi_t: \mathbb R^{2d} \to \mathbb R_{++}, \hat \varphi_t: \mathbb R^{2d} \to \mathbb R_{+}$ such that
    \begin{subequations}
    \begin{align}
    u_t^\star(x,v) &= \sigma^2 \, \nabla_v \log \varphi_t(x,v), \label{eq:u-interacting-phi} 
\end{align}
for a.e. $(t,x,v) \in [0,T] \times \mathbb R^{2d}$ at which $p_t^\star(x,v)>0$, and
  \begin{align}
    u_T^\star &=0, \label{eq:u-interacting-T}
    \end{align}
    almost everywhere at which $p_T^\star>0$.
Additionally, 
\begin{align}
    p_t^\star (x,v) &= \varphi_t(x,v) \hat \varphi_t(x,v),  \label{eq:pt-phi-hatphi}
\end{align}
for a.e. $(t,x,v) \in [0,T] \times \mathbb R^{2d}$.
\end{subequations}
The pair $(\varphi_t, \hat \varphi_t)$ satisfies  
    \begin{subequations}\label{eq:Schr-sys-main}
        \begin{align} 
    \partial_t \varphi_t + \big \langle v ,  \nabla_x \varphi_t \big \rangle   &+ \big\langle b_t[p_t^\star] , \nabla_v\varphi_t  \big \rangle   \nonumber \\
    &+\frac{\sigma^2}{2} \Delta_v \varphi_t = h_t[p_t^\star] \varphi_t,  \label{eq:varphit}  \\
     \partial_t \hat\varphi_t + \big \langle  v , \nabla_x \hat\varphi_t  \big \rangle  &+ \nabla_v  \cdot (\hat\varphi_t \, b_t[p_t^\star]) \nonumber \\
     &-\frac{\sigma^2}{2} \Delta_v \hat\varphi_t = -  h_t [p_t^\star] \hat\varphi_t, \label{eq:hatvarphit}
\end{align}
where
\begin{align*}
    \!\!\!\! h_t&[p_t^\star](x,v) = \nonumber \\
    & -\int_{\mathbb R^{2d}} \hat \varphi_t(\tilde{x},\tilde{v})  \Big \langle \Psi (\tilde{x}-x, \tilde{v} -v),   \nabla_{\tilde v} \varphi_t(\tilde{x},\tilde{v}) \Big \rangle \, {\rm d}\tilde{x}\: {\rm d}\tilde{v}.
\end{align*}
Eqs.~\eqref{eq:varphit} and \eqref{eq:hatvarphit} are subject to the following boundary conditions.
\begin{align} 
   &\hat \varphi_0(x,v) = \hat\eta_0(x) \gamma_0 (x,v),  \label{eq:BCs-partial-all-1} \\
&\hat \eta_0(x) \int_{\mathbb R^d} \gamma_0(x,v)\varphi_0(x,v)  \, {\rm d} v  = q_0(x),  \label{eq:BCs-partial-all-2} \\
   & \varphi_T(x,v) =\eta_T(x),  \label{eq:BCs-partial-all-3} \\
    &\eta_T(x) \int_{\mathbb R^d}  \hat \varphi_T(x,v) \, {\rm d} v = q_T(x),  \label{eq:BCs-partial-all-4}
\end{align}
for some measurable functions $\hat \eta_0 : \mathbb R^{d} \to \mathbb R_{+}, \eta_T: \mathbb R^{d} \to \mathbb R_{++}$, and a.e. $x,v \in \mathbb R^{d}$.
    \end{subequations}

<!-- proof-3: body_md -->
We addressed optimization over $p_0,p_T$ in Proposition~\ref{proposition:endpoint-conditions} and the optimization over $u_\cdot$ in Proposition~\ref{proposition:controller}. It remains to address optimization over $p_t$, where $t\in (0,T)$. To this end, we proceed by substituting \eqref{eq:u-interacting-Lambda} into \eqref{eq:Lagrangian2}. This yields
\begin{align*}
     \mathcal L &=   {\rm KL}(p_0,\gamma_0) + \int_0^T \!\!\!  \int_{\mathbb R^{2d}} p_t\Big\{ -\partial_t \Lambda_t - \big \langle v , \nabla_x \Lambda_t \big \rangle  \\
     &- \big \langle b_t[p_t], \nabla_v \Lambda_t \big \rangle    -   \frac{\sigma^2}{2} \Vert \nabla_v\Lambda_t  \Vert^2-  \frac{\sigma^2}{2}\Delta_v \Lambda_t \Big\}\, {\rm d} x\: {\rm d} v\: {\rm d}t \nonumber \\
     &+  \int_{\mathbb R^{2d}} \Big(p_T \, \Lambda_T - p_0 \, \Lambda_0 \Big) \,  {\rm d} x \, {\rm d} v.  
\end{align*}
Letting
\begin{align}
     \Upsilon :=  -  \int_0^T \!\!\!\int_{\mathbb R^{2d}} p_t  \big \langle  b_t[p_t], \nabla_v \Lambda_t \big\rangle \, {\rm d} x\: {\rm d} v\: {\rm d}t, \label{eq:upsilon-cs}
\end{align}
the first variation with respect to $p_t$ for $t \in (0,T)$ reads
\begin{align} \label{eq:dLdmu}
\!\!\!    \delta \mathcal L (p_\cdot, &\delta p_\cdot) = \int_0^T \!\!\!  \int_{\mathbb R^{2d}}   \Big\{ -\partial_t \Lambda_t - \big \langle v , \nabla_x \Lambda_t \big \rangle \nonumber \\
     & \qquad -  \frac{\sigma^2}{2} \Vert \nabla_v \Lambda_t  \Vert^2-  \frac{\sigma^2}{2}\Delta_v \Lambda_t \Big\} \delta p_t \, {\rm d} x\: {\rm d} v\: {\rm d}t  \nonumber \\
     & \qquad+ \delta  \Upsilon(p_\cdot, \delta p_\cdot).
\end{align}

Direct computation using \eqref{eq:convolution-expanded} yields 
    \begin{align}\label{eq:intermediate-step-in-optimality}
\!\!\! \delta  \Upsilon(p_\cdot, \delta p_\cdot)  = & \int_0^T \!\!\!  \int_{\mathbb R^{2d}}  \bigg [-\big\langle b_t[p_t]  , \nabla_v \Lambda_t \big \rangle  \nonumber\\
&\qquad\quad+ h_t[p_t](x,v) \bigg]  \delta p_t \, {\rm d} x \:{\rm d} v \, {\rm d} t,
\end{align}
where
\begin{align}\label{eq:qt}
 \!\!\!   h_t&[p_t](x,v) : = \nonumber \\
    & -\int_{\mathbb R^{2d}} p_t(\tilde{x},\tilde{v})  \Big \langle \Psi (\tilde{x}-x, \tilde{v}-v),   \nabla_{\tilde v} \Lambda_t(\tilde{x},\tilde{v}) \Big \rangle \, {\rm d}\tilde{x}\: {\rm d}\tilde{v}.
\end{align}
By substituting \eqref{eq:intermediate-step-in-optimality} into \eqref{eq:dLdmu}, the vanishing first variation of the Lagrangian $\mathcal L$ with respect to $p_t$ with $t\in (0,T)$ yields the Hamilton--Jacobi--Bellman equation
\begin{align}\label{eq:HJB-psi} 
\partial_t \Lambda_t + \big \langle v &, \nabla_x \Lambda_t \big \rangle + \big\langle b_t[p_t^\star] ,\nabla_v \Lambda_t  \big \rangle  \nonumber \\
 &+ \frac{\sigma^2}{2 }\Vert \nabla_v\Lambda_t  \Vert^2+ \frac{\sigma^2}{2}\Delta_v \Lambda_t =   h_t [p_t^\star].
\end{align}


Now, we define the change of variables 
\begin{align}\label{eq:log-transform}
    \Lambda_t(x,v) = \log \varphi_t(x,v),
\end{align}
where $\varphi_t$ is positive. This transformation is known as the \emph{Cole--Hopf} \cite{cole1951quasi,hopf1950partial} or \emph{Fleming logarithmic transformation} \cite{fleming2005logarithmic}. 
Using \eqref{eq:HJB-psi} and the identity
\begin{align}
    \Vert \nabla_v \log \varphi_t \Vert^2+ \Delta_v \log \varphi_t = \frac{\Delta_v \varphi_t}{\varphi_t},
\end{align}
we deduce that $\varphi_t$ in \eqref{eq:log-transform} satisfies \eqref{eq:varphit}.
Further, using \eqref{eq:u-interacting-Lambda} and \eqref{eq:log-transform}, the optimal control can be expressed as \eqref{eq:u-interacting-phi}.





Let us introduce
\begin{align} \label{eq:phi-hatphi}
    \hat \varphi_t(x,v) := \frac{p_t^\star(x,v)}{\varphi_t(x,v)}, \qquad t\in [0,T],
\end{align}
which is equivalent to \eqref{eq:pt-phi-hatphi}.
Then, by Proposition~\ref{proposition:controller}, \eqref{eq:VFP-controlled}, and \eqref{eq:varphit}, Eq.~\eqref{eq:hatvarphit} follows.

    
By Eqs.~\eqref{eq:quanti1}, \eqref{eq:xi_0} in Proposition~\ref{proposition:endpoint-conditions} together with \eqref{eq:BC1}, \eqref{eq:log-transform}, and \eqref{eq:pt-phi-hatphi}, 
we have
\begin{align*}
    \varphi_0 (x,v) \hat \varphi_0(x,v) = e^{\Lambda_0(x,v)} e^{-1+\xi_0(x)} \gamma_0(x,v).
\end{align*}
Let $\hat \eta_0 (x):= e^{-1+\xi_0(x)}$, whenever $q_0>0$ and zero otherwise. Then, using the transformation in \eqref{eq:log-transform} yields \eqref{eq:BCs-partial-all-1}. Then, by \eqref{eq:phi-hatphi} and \eqref{eq:BC1}, we obtain \eqref{eq:BCs-partial-all-2}. 
Analogously, by Eq.~\eqref{eq:Lambda_T} in Proposition~\ref{proposition:endpoint-conditions} and \eqref{eq:log-transform}, we have
\begin{align} \label{eq:varphi_T}
     \varphi_T (x,v) = e^{\Lambda_T(x)}.
\end{align}
Letting $\eta_T (x):= e^{\Lambda_T(x)}$ yields \eqref{eq:BCs-partial-all-3}. By \eqref{eq:phi-hatphi} and \eqref{eq:BC2}, we recover \eqref{eq:BCs-partial-all-4}.  Lastly, \eqref{eq:u-interacting-phi} and \eqref{eq:varphi_T} imply \eqref{eq:u-interacting-T}, which is consistent with Proposition~\ref{proposition:controller}.  This concludes the proof.
