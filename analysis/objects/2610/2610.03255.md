<!-- introtheorem-1: body_md -->
\label{ThmA}
Let $X$ be a hyperelliptic curve over $\mathbb{Q}$ of genus $g \geq 2$ defined by the equation $y^2=f(x)$, where $f\in \mathbb{Q}[x]$ is a separable polynomial of odd degree. Let $S$ be a finite set of primes containing $\{2,\infty\}$ and the primes of bad reduction for $X$. 
\begin{itemize}
    \item[(1)] For notational simplicity, denote
    \begin{align}\label{cubicintroineq}
    d_{X,S}\coloneqq \dim_{\mathbb{F}_2} H^1(G_S,(J[2] \otimes (\wedge^2J[2])/\mathbb{F}_2)/{\wedge^3J[2]}),
    \end{align}
    where $J[2]$ denotes the $2$-torsion subgroup of the Jacobian of $X$, and $G_S$ denotes the Galois group of the maximal extension of $\mathbb{Q}$ that is unramified outside of $S$. Then $d_{X,S}$ is effectively computable and $\dim_{\mathbb{Q}_2}H^1_f(G_S,U[3])<d_{X,S}$. 
    \item[(2)] If, in addition, the defining polynomial $f$ is irreducible and condition \hypersetup{linkcolor=black}\eqref{starassumption}\hypersetup{linkcolor=blue} holds, then
    \begin{align*}
    d_{X,S}=d_3-(d_2+d_1)+\#S.
    \end{align*}
    \item[(3)] Suppose that the following inequality holds:
    \begin{align*}
    d_{X,S}+\dim_{\mathbb{Q}_2}H^1_f(G_S,\wedge^2 V_2J)<\frac{1}{6}(g+1)(2g+1)(7g-6)-\mathrm{rk}_{\mathbb{Z}}J(\mathbb{Q}).
    \end{align*}
    Then the depth 3 Chabauty--Kim set $X(\mathbb{Q}_2)_3$ is finite. 
\end{itemize}

\label{ThmB}
Let $X$ be the genus $2$, Mordell--Weil rank $4$ hyperelliptic curve over $\mathbb{Q}$ defined as the smooth projective completion of the affine curve

y^2=4x^5 + 8x^4 + 4x^2 + 8x + 1.

\dim\mathrm{Sel}(U_n)<\dim H^1_f(\mathbb{Q}_p,U_n).
\end{align}
Then $X(\mathbb{Q}_p)_n$ is finite.

<!-- lemma-1: body_md -->
\label{netancor3}
Suppose that the following inequality holds:
\begin{align*}
\dim_{\mathbb{Q}_p} H^1_f(\mathbb{Q},U[2])+\dim_{\mathbb{Q}_p}H^1_f(\mathbb{Q},U[3]) < \frac{1}{6}(g+1)(2g+1)(7g-6) -\mathrm{rk}_\mathbb{Z}J(\mathbb{Q}).
\end{align*}
Then $X(\mathbb{Q}_p)_3$ is finite.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
By the inequality \eqref{SelDimIneq}, it suffices to compute the $\mathbb{Q}_p$-dimensions of the local Bloch--Kato Selmer groups $H^1_f(\mathbb{Q}_p,U[i])$ for $i\leq 3$. To that end, since each $U[i]$ is pure of negative weight, we may exploit the isomorphism
\begin{align*}
H^1_f(\mathbb{Q}_p,U[i])\simeq D_{\mathrm{dR}}(U[i])/\mathrm{Fil}^0D_{\mathrm{dR}}(U[i]),
\end{align*}
given that computation of the $\mathbb{Q}_p$-dimension of the right-hand side follows straightforwardly from the descriptions of $U[i]$ above. For $i\leq 2$, this calculation is essentially the proof of \cite[Lemma 25]{bk1}, which asserts that
\begin{align*}
\dim_{\mathbb{Q}_p}H^1_f(\mathbb{Q}_p,U[1])+\dim_{\mathbb{Q}_p}H^1_f(\mathbb{Q}_p,U[2])=\frac{1}{2}(3g-2)(g+1),
\end{align*}
and relies on the fact that $\mathrm{Fil}^0D_{\mathrm{dR}}(V_pJ)$ is $g$-dimensional. For the $3^{\mathrm{rd}}$ graded piece, the sequence \eqref{cubicDesc} implies that
\begin{align*}
\dim_{\mathbb{Q}_p}D_{\mathrm{dR}}(U[3])&=2g\bigg(\binom{2g}{2}-1\bigg)-\binom{2g}{3} \\
\dim_{\mathbb{Q}_p}\mathrm{Fil}^0D_{\mathrm{dR}}(U[3])&= g\binom{g}{2}-\binom{g}{3},
\end{align*}
from which the result follows.

<!-- lemma-2: body_md -->
\label{CondDep3}
Assume that Conjecture \ref{BKConjectures} holds. Then the depth $3$ set $X(\mathbb{Q}_p)_3$ is finite whenever 
\begin{align}\label{dep3ineq}
\mathrm{rk}_{\mathbb{Z}} J(\mathbb{Q})<\frac{1}{3}(4g^3+3g^2-4g)+\rho-1.
\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
If $W$ is a geometric $\mathbb{Q}_p$-representation of $G_K$ that is pure of negative weight, then \cite[Remark II.2.2.2]{FPR} implies that
\begin{align}\label{FPRSES}
\begin{split}
&\dim_{\mathbb{Q}_p}H^1_f(\mathbb{Q}_p,W)-\dim_{\mathbb{Q}_p}H^1_f(\mathbb{Q},W) \\ &= \dim_{\mathbb{Q}_p}H^0(\mathbb{R},W)-\dim_{\mathbb{Q}_p}H^1_f(\mathbb{Q},W^*(1))+\dim_{\mathbb{Q}_p}H^0(\mathbb{Q},W^*(1)).
\end{split}
\end{align}
Conjecture \ref{BKConjectures} implies that $H^1_f(\mathbb{Q},W^*(1))$ is trivial for both $W=U[2]$ and $W=U[3]$. Additionally, $H^0(\mathbb{Q},U[3]^*(1))$ is trivial for weight reasons, while 
\begin{align*}
\dim_{\mathbb{Q}_p}H^0(\mathbb{Q},U[2]^*(1))=\rho -1,
\end{align*}
by the proof of \cite[Lemma 2.5]{QC2}. Finally, the group $H^0(\mathbb{R},W)$ is the subspace of $W$ invariant under the action of complex conjugation, which, together with the isomorphism $V_pJ\simeq \mathbb{Q}_p^g\oplus \mathbb{Q}_p(1)^g$ of $G_{\mathbb{R}}$-modules, yields the following dimensions through simple calculation:
\begin{align*}
\dim_{\mathbb{Q}_p}H^0(\mathbb{R},U[2])=g(g-1), \hspace*{0.5cm} \dim_{\mathbb{Q}_p}H^0(\mathbb{R},U[3])=\frac{4}{3}g(g^2-1). 
\end{align*}
Substituting these simplifications into \eqref{FPRSES}, while assuming \eqref{dep3ineq}, we have
\begin{align*}
\dim H^1_f(\mathbb{Q}_p,U_3)-\dim \mathrm{Sel}(U_3) \geq g-\mathrm{rk}_{\mathbb{Z}}J(\mathbb{Q})+g(g-1)+\rho-1+\frac{4}{3}g(g^2-1)>0,
\end{align*}
which implies finiteness of $X(\mathbb{Q}_p)_3$ by Theorem \ref{KimThm}.

<!-- definition-1: body_md -->
\label{h1fdef}
Let $K$ be a number field, let $V$ be a finite-dimensional $\mathbb{Q}_p$-vector space with a continuous action of $G_K$, and let $T\subseteq V$ be a $G_K$-stable $\mathbb{Z}_p$-lattice. For a prime $v$ of $K$, define $H^1_f(K_v,T)$ as the preimage of $H^1_f(K_v,V)$ under the natural map in cohomology induced by the inclusion $T\subseteq V$. We define the corresponding global analogue by 
\begin{align}\label{IntegralBK}
H^1_f(K,T)\coloneqq \bigcap_{v} \mathrm{ker}\big(H^1(K,T)\rightarrow H^1(K_v,T)/H^1_f(K_v,T)\big),
\end{align}
where the intersection varies over all primes $v$ of $K$. For a prime $v$ in $K$, define $H^1_f(K_v,T\otimes \mathbb{F}_p)$ as the image of $H^1_f(K_v,T)$ under the injection 
\begin{align*}
H^1(K_v,T)\otimes \mathbb{F}_p\hookrightarrow H^1(K_v,T\otimes \mathbb{F}_p),
\end{align*}
induced by multiplication by $p$ on $T$. The global analogue $H^1_f(K,T\otimes \mathbb{F}_p)$ is then defined analogously to \eqref{IntegralBK}.

<!-- lemma-3: body_md -->
\label{wedgenlemma}
Let $X$ be a hyperelliptic curve over $K$ defined by the equation $y^2=f(x)$, where the degree of $f\in K[x]$ is odd. Then, for each integer $n>1$, there is a split short exact sequence 
\begin{align}\label{wedgenSES}
0\rightarrow {\wedge^n J[2]} \rightarrow {\wedge^n \mathbb{F}_2[\mathcal{R}_f]}&\rightarrow {\wedge^{n-1} J[2]} \rightarrow 0,
\end{align}
induced by the map 
\begin{align}\label{explicitwedgenmap}
\begin{split}
 {\wedge^n \mathbb{F}_2[\mathcal{R}_f]}&\rightarrow {\wedge^{n-1} J[2]} \\
\alpha_1\wedge \cdots \wedge \alpha_n &\mapsto \sum_{i=1}^n \alpha_1 \wedge\cdots \wedge \widehat{\alpha}_i \wedge \cdots \wedge \alpha_n,
\end{split}
\end{align}
where $\widehat{\alpha}_i$ denotes omission of the $i^{\mathrm{th}}$ factor.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The existence of the sequence follows from the well-known split short exact sequence\footnote{Throughout this paper, $J[2]$ will be understood as the submodule of $\mathbb{F}_2[\mathcal{R}_f]$ defined by this exact sequence, and we will rarely require its description in terms of, say, divisors on $X$.}
\begin{align}\label{DefJ2}
0\rightarrow J[2] \rightarrow \mathbb{F}_2[\mathcal{R}_f] \xrightarrow{\mathrm{deg}}\mathbb{F}_2 \rightarrow 0,
\end{align}
which is contained in \cite[Eq. (5)]{bk1} and the discussion thereafter, for example. To see that it splits, note that if 
\begin{align*}
\varphi:\mathbb{F}_2[\mathcal{R}_f]\rightarrow J[2]
\end{align*}
defines a retraction of \eqref{DefJ2}, then $\wedge^n \varphi$ defines a retraction of \eqref{wedgenSES}.

<!-- proposition-1: body_md -->
\label{MainApprox}
Let $X$ be a hyperelliptic curve over $\mathbb{Q}$ defined by the equation $y^2=f(x)$, where the degree of $f\in \mathbb{Q}[x]$ is odd, and let $S$ be a finite set of primes that contains $\{2,\infty\}$ and the primes of bad reduction for $X$. Then
\begin{align}\label{approxinclusion}
\dim_{\mathbb{Q}_2}H^1_f\big(\mathbb{Q}, U[3]\big)< \dim_{\mathbb{F}_2} H^1(G_S, J[2]\otimes \overline{\wedge^2 J[2]}/{\wedge^3J[2]}).
\end{align}
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
To simplify notation, we denote the $\mathbb{Q}_2$-rational and $\mathbb{Z}_2$-integral Tate modules of $J$ by $V\coloneqq V_2J$ and $T \coloneqq T_2J$, respectively. The result follows straightforwardly once we define a Galois stable $\mathbb{Z}_2$-lattice within $U[3]$. 

To that end, first define $\overline{\wedge^2 T}$ as the cokernel of the map dual to the cup product 
\begin{align}\label{integralcup}
\cup:\wedge^2 H^1_{\textnormal{\'{e}t}}(X_{\overline{\mathbb{Q}}},\mathbb{Z}_2)\rightarrow H^2_{\textnormal{\'{e}t}}(X_{\overline{\mathbb{Q}}},\mathbb{Z}_2)\simeq \mathbb{Z}_2(-1).
\end{align}
To see that $\overline{\wedge^2 T}$ itself defines a $\mathbb{Z}_2$-lattice within $\overline{\wedge^2 V}=U[2]$, we note that $\overline{\wedge^2 T}$ admits the following explicit description: if $\{e_i,e_{g+i}\}_{i=1}^g$ is a basis of $T$ that is dual to the symplectic basis of $H^1_{\textnormal{\'{e}t}}(X_{\overline{\mathbb{Q}}},\mathbb{Z}_2)$ (with respect to the cup-product pairing), then $\overline{\wedge^2 T}$ is the cokernel of the map 
\begin{align*}
\mathbb{Z}_2(1) &\rightarrow \wedge^2T \\ 
1 &\mapsto \sum_{i=1}^ge_i\wedge e_{g+i},
\end{align*}
which follows immediately from the definition of a symplectic pairing. In particular, $\overline{\wedge^2 T}$ is torsion-free and the set 
\begin{align}\label{IntermediateBasis}
\{e_i\wedge e_j:1\leq i<j\leq 2g, \ (i,j)\neq (g,2g)\}
\end{align}
constitutes a $\mathbb{Z}_2$-basis. 

Consequently, a canonical candidate for a $G_K$-stable lattice within $U[3]$ is the cokernel $\mathcal{T}\coloneqq T\otimes\overline{\wedge^2 T}/{\wedge^3T}$ of the integral analogue
\begin{align}\label{Z2mapcube}
\begin{split}
{\wedge^3 T}&\rightarrow T\otimes{\wedge^2 T} \\
e_i\wedge e_j \wedge e_k &\mapsto e_i\otimes (e_j \wedge e_k)+ e_j \otimes (e_k\wedge e_i) + e_k \otimes (e_i\wedge e_j).
\end{split}
\end{align}
to the injection in \eqref{cubicDesc}. We need only prove that this cokernel is torsion-free. To this end, given that ${\wedge^3 T}\rightarrow T \otimes \overline{\wedge^2 T}$ is injective, it suffices to show that the $(\mathrm{rk}_{\mathbb{Z}_2}T\otimes \overline{\wedge^2T})\times(\mathrm{rk}_{\mathbb{Z}_2}{\wedge^3 T})$-matrix representing this map has a $(\mathrm{rk}_{\mathbb{Z}_2}{\wedge^3 T}) \times (\mathrm{rk}_{\mathbb{Z}_2}{\wedge^3 T}) $-minor of determinant $\pm1$. If this holds, then, by the theory of Smith Normal Forms, the product of all elementary divisors of the matrix is $1$, from which we conclude that its cokernel is torsion-free. This is straightforward to prove for this particular map, given that each column of the matrix contains a $1$ for which it is the only non-zero entry in its row. 

To this end, via the basis \eqref{IntermediateBasis} of $\overline{\wedge^2 T}$, we fix the following basis of $T\otimes \overline{\wedge^2T}$:
\begin{align*}
\{e_i\otimes \overline{e_j\wedge e_k}:1 \leq i \leq 2g,\ 1\leq j<k\leq 2g \text{ with } (j,k)\neq (g,2g)\}.
\end{align*}
Then the image of $e_i\wedge e_j \wedge e_k\in {\wedge^3 T}$, for $\{g,2g\}\not\subseteq\{i,j,k\}$, under the map \eqref{Z2mapcube} is 
\begin{align*}
e_i\otimes \overline{e_j\wedge e_k}+e_j \otimes \overline{e_k\wedge e_i}+e_k\otimes \overline{e_i\wedge e_j},
\end{align*}
while the image of $e_i\wedge e_g\wedge e_{2g}$ is 
\begin{align*}
e_g\otimes \overline{e_{2g}\wedge e_i}+e_{2g}\otimes \overline{e_i\wedge e_g}-\sum_{\ell =1}^{g-1}e_i\otimes \overline{e_\ell \wedge e_{\ell+g}}.
\end{align*}
Accordingly, with respect to these bases, the matrix representing \eqref{Z2mapcube} is of the desired form. Indeed, the summand $e_j\otimes \overline{e_{j+g}\wedge e_i}$ in the image of $e_i\wedge e_j \wedge e_{j+g}$ does not appear as the summand in the image of another basis element; for the remaining $e_i\wedge e_j \wedge e_k$ not of this form, every summand appears uniquely in its image. We conclude that $\mathcal{T}$ is torsion-free. 

To identify a description of the mod-2 quotient of this lattice, note that the mod-2 analogue of \eqref{integralcup} may be identified with the Weil pairing $\wedge^2 J[2] \rightarrow \mathbb{F}_2$ (see, for instance, \cite[Remark V.2.4(f)]{MilneOGEC}). Consequently, upon dualising we obtain that the mod-2 quotient of $\overline{\wedge^2 T}$ is isomorphic to $\overline{\wedge^2 J[2]}$. Finally, this implies that $J[2]\otimes {\overline{\wedge^2 J[2]}}/{\wedge^3 J[2]}$ is isomorphic to the mod-2 quotient of $T\otimes \overline{\wedge^2T}/{\wedge^3 T}$. 

In summary, the $\mathbb{Q}_2$-dimension of $H^1_f(\mathbb{Q},U[3])$ is bounded by the $\mathbb{F}_2$-dimension of the image of $H^1_f(\mathbb{Q},\mathcal{T})$ in $H^1(G_S,J[2]\otimes {\overline{\wedge^2 J[2]}}/{\wedge^3 J[2]})$. In particular, the cruder bound in \eqref{approxinclusion} holds.

<!-- remark-2: body_md -->
\label{GSBK3}
The inequality \eqref{approxinclusion} from Proposition \ref{MainApprox} is strict, and Remark \ref{BK3remark} can be used to deduce the optimal bound. Specifically, \cite[Proposition 3.8]{bloch1990functions} states that $H^1_f(\mathbb{Q}_2,U[3])$ and $H^1_f(\mathbb{Q}_2,U[3]^*(1))$ are exact annihilators under the cup product pairing
\begin{align*}
H^1(\mathbb{Q}_2,U[3])\times H^1(\mathbb{Q}_2,U[3]^*(1))\xrightarrow{\cup} H^2(\mathbb{Q}_2,\mathbb{Q}_2(1))\simeq \mathbb{Q}_2, 
\end{align*}
which yields the following unconditional difference in dimensions:
\begin{align*}
\dim_{\mathbb{Q}_2}H^1(\mathbb{Q}_2,U[3])-\dim _{\mathbb{Q}_2}H^1_f(\mathbb{Q}_2,U[3])=\dim_{\mathbb{Q}_2}H^1_f(\mathbb{Q}_2,U[3]^*(1)).
\end{align*} 
Since $U[3]^*(1)$ is pure of non-zero weight, it follows that 
\begin{align*}
\dim_{\mathbb{Q}_2}H^1_f(\mathbb{Q}_2,U[3]^*(1))=\dim_{\mathbb{Q}_2}D_{\mathrm{dR}}(U[3]^*(1))/\mathrm{Fil}^0D_{\mathrm{dR}}(U[3]^*(1))=\dim_{\mathbb{Q}_2}\mathrm{Fil}^0D_{\mathrm{dR}}(U[3]),
\end{align*}
and this final dimension is computed in the proof of Lemma \ref{netancor3}. By the proof of \cite[Proposition II.2.2.1]{FPR}, Conjecture \ref{BKConjectures} implies that this local disparity lifts globally. Indeed, we obtain a surjection 
\begin{align*}
H^1(G_S,U[3]) \rightarrow \bigoplus_{p \in S}H^1(\mathbb{Q}_p,U[3])/H^1_f(\mathbb{Q}_p,U[3]),
\end{align*}
given that $H^1_f(\mathbb{Q},U[3]^*(1))=0$ under the assumption of the conjecture, where $S$ denotes a finite set of primes containing $\{2,\infty\}$ and the primes of bad reduction for $X$. Consequently, the sharpest bound for $\dim_{\mathbb{Q}_2}H^1_f(\mathbb{Q},U[3])$ obtainable by the methods of this paper is 
\begin{align*}
g(g^2-1)+g\binom{g}{2}-\binom{g}{3} = \frac{4}{3}g(g^2-1). 
\end{align*}
In particular, the sharpest bound we expect for a genus 2 curve is 8.

<!-- lemma-5: body_md -->
\label{wedge3split}
The short exact sequence
\begin{align}\label{FinallySplit}
0\rightarrow \wedge^3J[2] \rightarrow J[2]\otimes \overline{\wedge^2 J[2]}\rightarrow J[2]\otimes \overline{\wedge^2 J[2]}/{\wedge^3 J[2]} \rightarrow 0,
\end{align}
induced by \eqref{wedge3map}, is split.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
To show that the sequence is split, it suffices to construct a map
\begin{align*}
\mathbb{F}_2[\mathcal{R}_f]\otimes \wedge^2 \mathbb{F}_2[\mathcal{R}_f]\rightarrow {\wedge^3 \mathbb{F}_2[\mathcal{R}_f]}
\end{align*}
that both defines a retraction of 
\begin{align}\label{retractionmap}
{\wedge^3J[2]}\rightarrow J[2]\otimes {\wedge^2 J[2]}
\end{align}
and vanishes on the image of $J[2]$ under the map induced by \eqref{nonsplitmapf2}. To that end, for distinct $i,j,k$, consider the element
\begin{align*}
D_{i,j,k}\coloneqq \sum_{m,n}\alpha_i\wedge \alpha_m\wedge \alpha_n + \sum_{r,s,t \notin \{i,j,k\}}\alpha_r\wedge \alpha_s\wedge\alpha_t \in {\wedge^3 \mathbb{F}_2[\mathcal{R}_f]},
\end{align*}
from which we define the map 
\begin{align*}
\varphi: \mathbb{F}_2[\mathcal{R}_f]\otimes {\wedge^2 \mathbb{F}_2[\mathcal{R}_f]}&\rightarrow {\wedge^3\mathbb{F}_2[\mathcal{R}_f]}\\
\alpha_i\otimes (\alpha_j \wedge \alpha_k) &\mapsto \begin{cases}
D_{i,j,k} &\text{if $i,j,k$ are distinct;}\\
0 &\text{otherwise.}
\end{cases}
\end{align*}
Without loss of generality, to describe the composition of \eqref{reflate} with $\varphi$, we consider the image of the single triple $\alpha_1\wedge \alpha_2 \wedge \alpha_3 \in {\wedge^3 \mathbb{F}_2[\mathcal{R}_f]}$; rather than working directly with the resulting sum, it is enough to count the number of times a given $\alpha_i\wedge \alpha_j \wedge \alpha_k$ appears as a summand: 
\begin{itemize}
    \item If $\{i,j,k\}\cap \{1,2,3\}=\emptyset$, then $\alpha_i\wedge \alpha_j\wedge \alpha_k$ appears three times. 
    \item If $\# (\{i,j,k\}\cap \{1,2,3\})=1$, then $\alpha_i\wedge \alpha_j\wedge \alpha_k$ appears once. 
    \item If $\# (\{i,j,k\}\cap \{1,2,3\})=2$, then $\alpha_i\wedge \alpha_j\wedge \alpha_k$ appears twice.
    \item The triple $\alpha_1\wedge \alpha_2 \wedge \alpha_3$ appears 3 times.
\end{itemize}
In summary, we obtain that the above composition defines the map 
\begin{align}\label{anotheralmostretract}
\begin{split}
{\wedge^3\mathbb{F}_2[\mathcal{R}_f]}&\rightarrow {\wedge^3\mathbb{F}_2[\mathcal{R}_f]}\\
\alpha_1\wedge \alpha_2 \wedge \alpha_3 &\mapsto \alpha_1\wedge \alpha_2 \wedge \alpha_3 +\sum_{\substack{i<j<k \\ |\{i,j,k\}\cap \{1,2,3\}|\leq 1}} \alpha_{i}\wedge \alpha_j\wedge \alpha_k. 
\end{split}
\end{align}
To show that this restricts to identity on ${\wedge^3 J[2]}$, first note that the $n=3$ case of \eqref{wedgenSES} implies that ${\wedge^3 J[2]}$ is the $\mathbb{F}_2$-span of elements of the form 
\begin{align*}
\alpha_{i_1}\wedge \alpha_{i_2} \wedge \alpha_{i_3} +\alpha_{i_1}\wedge \alpha_{i_2} \wedge \alpha_{i_4} +\alpha_{i_1} \wedge \alpha_{i_3} \wedge \alpha_{i_4} +\alpha_{i_2} \wedge \alpha_{i_3}\wedge\alpha_{i_4}.
\end{align*}
Through another count of the frequency a given $\alpha_i\wedge \alpha_j \wedge \alpha_k$ appears in the image of this $\mathbb{F}_2$-basis element under \eqref{anotheralmostretract}, we obtain that this composition is indeed identity on ${\wedge^3 J[2]}$. We conclude that $\varphi$ defines the desired retraction of \eqref{retractionmap}. 

It remains to show that $\varphi$ vanishes on the image of $J[2]$ under the map 
\begin{align}\label{imageJ2map}
\begin{split}
\mathbb{F}_2[\mathcal{R}_f]&\rightarrow \mathbb{F}_2[\mathcal{R}_f] \otimes {\wedge^2 J[2]}\\
\alpha &\mapsto \sum_{i<j}\alpha\otimes (\alpha_i\wedge \alpha_j).
\end{split}
\end{align}
First note that the element $\alpha_1\in \mathbb{F}_2[\mathcal{R}_f]$ is mapped under the composition of \eqref{imageJ2map} with $\varphi$ to the following sum:
\begin{align}\label{awfsum}
\binom{\deg f -1}{2}\bigg(\sum_{m<n}\alpha_1\wedge\alpha_m\wedge \alpha_n\bigg)+\sum_{\substack{i<j\\i,j\neq 1}}\sum_{r,s,t \notin \{1,i,j\} } \alpha_r\wedge\alpha_s\wedge \alpha_t.
\end{align}
A given $\alpha_r\wedge \alpha_s\wedge \alpha_t$ (for $r,s,t \neq 1$) appears in this right-hand sum for any $i,j \notin \{1,r,s,t\}$, and hence \eqref{awfsum} simplifies to
\begin{align*}
&\binom{\deg f -1}{2}\bigg(\sum_{m<n}\alpha_1\wedge\alpha_m\wedge \alpha_n\bigg)+ \binom{\deg f -4}{2}\bigg(\sum_{r,s,t \neq 1}\alpha_r\wedge \alpha_s \wedge \alpha_t\bigg)\\
&=\binom{\deg f -4}{2} \bigg(\sum_{r,s,t}\alpha_r\wedge \alpha_s\wedge \alpha_t\bigg),
\end{align*}
where this final equality follows as the two binomial coefficients have the same parity whenever $\deg f$ is odd. Consequently, $\varphi$ vanishes on the image of even degree sums of roots of $f$ under \eqref{imageJ2map}, as required.

<!-- lemma-6: body_md -->
\label{cubInd}
For $1\leq n \leq \deg f$, we have the isomorphism 
\begin{align*}
H^1(K,\wedge^n \mathbb{F}_2[\mathcal{R}_f])\simeq K_f^{(n),\times}\otimes\mathbb{F}_2.
\end{align*}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Suppose $\mathcal{R}\simeq \mathcal{R}_f$ is the $G_K$-set $\mathrm{Hom}_K(K_f,K^\mathrm{sep})$, and denote by $\mathcal{R}^{(n)}$ the set of unordered $n$-tuples of distinct elements in $\mathcal{R}$. Then we have the following isomorphism of $G_K$-modules:
\begin{align*}
\wedge^n \mathbb{F}_2[\mathcal{R}]&\xrightarrow{\sim}\mathbb{F}_2[\mathcal{R}^{(n)}]\\
s_1\wedge \dots \wedge s_n &\mapsto \{s_1,\dots,s_n\}.
\end{align*}
Consequently, by \cite[Lemma 3]{bk1}, the result follows if we establish the isomorphism of $G_K$-sets
\begin{align}\label{s(3)isom}
{\mathcal{R}^{(n)}}\simeq \mathrm{Hom}_K(K_f^{(n)},K^{\mathrm{sep}}).
\end{align}
This can be shown by an identical argument to the proof of \cite[Lemma 2]{bk1}: we include it here for the sake of completeness. 

If we denote by $\mathcal{R}_{(n)}$ the ordered $n$-tuples of distinct elements of $\mathcal{R}$, then the injection of $K_{f,n}$ into $K_f^{{\otimes n}}$ yields the following isomorphism of $G_K$-sets:
\begin{align}\label{s3equiv}
\begin{split}
\mathcal{R}_{(n)}&\xrightarrow{\sim} \mathrm{Hom}_K(K_{f,n},K^{\mathrm{sep}})\\
(s_1,\dots,s_n)&\mapsto s_1\otimes \dots \otimes s_n.
\end{split}
\end{align}
Clearly, there is a canonical action of $S_n$ on both sides of \eqref{s3equiv}, under which this isomorphism is equivariant. Since the quotient of $\mathcal{R}_{(n)}$ by this action is isomorphic to $\mathcal{R}^{(n)}$, we establish \eqref{s(3)isom} if the map 
\begin{align}\label{pullbackmap}
\mathrm{Hom}_K(K_{f,n},K^{\mathrm{sep}})/S_n\rightarrow \mathrm{Hom}_K(K_f^{(n)},K^{\mathrm{sep}}),
\end{align}
induced by pulling back the inclusion $K_f^{(n)}\hookrightarrow K_{f,n}$, is an isomorphism. Finally, this follows as the pullback is surjective and factors through the quotient by $S_n$, and the orders of the domain and codomain of \eqref{pullbackmap} coincide.

<!-- lemma-7: body_md -->
\label{simple21isom}
We have the isomorphism 
\begin{align*}
H^1(K,\mathbb{F}_2[\mathcal{R}_f]\otimes {\wedge^2 J[2]})\simeq K_f^{(1,2),\times}\otimes \mathbb{F}_2.
\end{align*}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
This is a consequence of the following basic result: if $G$ is a group, $H\subseteq G$ is a subgroup, $M$ is an $H$-module and $N$ is a $G$-module, then $(\mathrm{Ind}^G_HM)\otimes N \simeq \mathrm{Ind}^G_H(M\otimes \mathrm{Res}^G_HN)$ as $G$-modules. Applying this to the case of interest yields
\begin{align*}
\mathbb{F}_2[\mathcal{R}_f]\otimes {\wedge^2 J[2]}\simeq \mathrm{Ind}_K^{K_f}\mathbb{F}_2 \otimes \wedge^2 J[2] \simeq \mathrm{Ind}_K^{K_f}\mathrm{Res}_K^{K_f}{\wedge^2 J[2]}\simeq \mathrm{Ind}_K^{K_f}( \wedge^2 \mathrm{Ind}_{K_f}^{K_{f,2}}\mathbb{F}_2).
\end{align*}
The lemma then follows from the isomorphism 
\begin{align*}
\wedge^2 \mathrm{Ind}_{K_f}^{K_{f,2}}\mathbb{F}_2\simeq \mathrm{Ind}_{K_f}^{K_f^{(1,2)}}\mathbb{F}_2,
\end{align*}
which is implied by \cite[Lemmas 1 and 2]{bk1} and the definition of $K_f^{(1,2)}$.

<!-- lemma-8: body_md -->
\label{splitlemmaJ2}
The short exact sequence 
\begin{align}\label{biggernonsplit}
0\rightarrow J[2] \rightarrow J[2] \otimes \wedge^2 J[2] \rightarrow J[2] \otimes \overline{\wedge^2 J[2]} \rightarrow 0,  
\end{align}
induced by \eqref{nonsplitmapf2}, is split.
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
If $\deg f\equiv 3 \textrm{ (mod $4$)}$, then the map 
\begin{align*}
\mathrm{deg}:{\wedge^2 \mathbb{F}_2}[\mathcal{R}_f]\rightarrow \mathbb{F}_2,\hspace*{0.5cm} \alpha_i\wedge \alpha_j &\mapsto 1
\end{align*}
induces a retraction of \eqref{nonsplitmapf2}, given that $\binom{\deg f}{2}$ is odd. Consequently, \eqref{biggernonsplit} is split in this case.

Hence, assume that $\deg f \equiv 1 \textrm{ (mod $4$)}$. A retraction of \eqref{biggernonsplit} is induced by the following map:
\begin{align}\label{retractionJ2}
\begin{split}
\mathbb{F}_2[\mathcal{R}_f] \otimes {\wedge^2 \mathbb{F}_2[\mathcal{R}_f]}&\rightarrow \mathbb{F}_2[\mathcal{R}_f]\\
\alpha_i\otimes (\alpha_j\wedge \alpha_k)&\mapsto \begin{cases}
\alpha_i+\alpha_j+\alpha_k &\text{if $i,j,k$ are distinct;}\\
0 & \text{otherwise.}
\end{cases}
\end{split}
\end{align}
The image of $\alpha_1\in \mathbb{F}_2[\mathcal{R}_f]$ under the composition of \eqref{imageJ2map} with \eqref{retractionJ2} is the sum 
\begin{align*}
\sum_{\substack{i<j\\i,j\neq 1}}\alpha_1+\alpha_i+\alpha_j = \alpha_1+\sum_i\alpha_i,
\end{align*}
given that $\binom{\deg f-1}{2}$ is even in this case. Consequently, this composition is identity on the even degree sums of roots of $f$, as desired.

<!-- remark-3: body_md -->
Although the $\mathbb{F}_2$-dimensions of $H^1(G_S,{\wedge^3J[2]})$ and $H^1(G_S,J[2]\otimes\overline{\wedge^2 J[2]})$ alone suffice for the purpose of computing the dimension of $H^1(G_S,J[2]\otimes \overline{\wedge^2 J[2]}/{\wedge^3J[2]})$, each of these cohomology groups admit a field-theoretic description in terms of the finite \'{e}tale algebras above; we collect them in this remark. 
\begin{enumerate}
    \item Consider the following diagram: 
    \begin{center}
    \begin{tikzcd}
    H^1(K,\wedge^3 \mathbb{F}_2[\mathcal{R}_f])\arrow{r}\arrow{d} & H^1(K,\mathbb{F}_2[\mathcal{R}_f]\otimes \wedge^2 \mathbb{F}_2[\mathcal{R}_f])\arrow{r} \arrow{d}& H^1(K,\wedge^2 \mathbb{F}_2[\mathcal{R}_f])\arrow{d}\\
    K_f^{(3),\times}\otimes \mathbb{F}_2 \arrow{r} & K_{f,2}^\times \otimes \mathbb{F}_2 \oplus K_f^{(1,2),\times}\otimes \mathbb{F}_2 \arrow{r} & K_f^{(2),\times} \otimes \mathbb{F}_2,
    \end{tikzcd}
    \end{center}
    where each of the maps are as follows:
    \begin{itemize}
        \item the vertical maps are the isomorphisms in Lemma \ref{cubInd}, Lemma \ref{simple21isom} and \eqref{WedgeSqDesc}, respectively;
        \item the first map in the bottom row is induced by the inclusion $K_f^{(3)}\hookrightarrow K_f^{(1,2)}$, and the second is the norm map; and
        \item the first map in the top row is induced by \eqref{reflate}, while the second is obtained from the map 
        \begin{align*}
        \mathbb{F}_2[\mathcal{R}_f] \otimes\wedge^2  \mathbb{F}_2[\mathcal{R}_f]
        &\longrightarrow \wedge^2 \mathbb{F}_2[\mathcal{R}_f] \\
        \alpha_1 \otimes  (\alpha_2 \wedge \alpha_3) 
        &\longmapsto  \alpha_2\wedge \alpha_3.
        \end{align*} 
    \end{itemize}
    Analogous to the proof of \cite[Lemma 6]{bk1}, commutativity of the left-hand (resp. right-hand) square above follows from Lemma 4 (resp. Lemma 5) from \cite{bk1}. This commutative diagram, coupled with the $n=3$ case of Lemma \ref{wedgenlemma}, implies that
    \begin{align*}
    H^1(K,\wedge^3J[2])\simeq \mathrm{ker}\big(K_f^{(3),\times}\otimes \mathbb{F}_2 \rightarrow K_f^{(2),\times}\otimes \mathbb{F}_2\big),
    \end{align*}
    where the map on the right-hand side is the composition 
    \begin{align*}
    K_f^{(3),\times}\otimes \mathbb{F}_2 \rightarrow K_f^{(1,2),\times}\otimes \mathbb{F}_2 \xrightarrow{\mathrm{Nm}}  K_f^{(2),\times}\otimes \mathbb{F}_2,
    \end{align*}
    induced by the inclusion of $K_f^{(3)}$ into $K_f^{(1,2)}$.

    A suitable modification of this argument implies the more general statement that $H^1(K,{\wedge^n J[2]})$ is isomorphic to the kernel of the composite map
    \begin{align*}
    K_f^{(n),\times}\otimes \mathbb{F}_2\rightarrow K_f^{(1,n-1),\times}\otimes \mathbb{F}_2\xrightarrow{\mathrm{Nm}} K_f^{(n-1),\times}\otimes \mathbb{F}_2,
    \end{align*}
    for $n>3$. This could be advantageous, for instance, in putative methods to develop analogous explicit descent techniques for higher graded pieces of the filtration on the fundamental group, as dictated by their descriptions in \cite[Corollary 8.3]{Hain1995InfinitesimalPO}.
    \item Exploiting the same lemmas \cite[Lemmas 4 and 5]{bk1} yields a field-theoretic description of $H^1(K,J[2]\otimes\overline{\wedge^2 J[2]})$. Indeed, the sequence \eqref{J2otimeswedgej2SES} implies that we have the isomorphism 
    \begin{align*}
    H^1(K,J[2]\otimes \wedge^2J[2])\simeq \mathrm{ker}\big(K_f^{(1,2),\times}\otimes \mathbb{F}_2 \xrightarrow{\mathrm{Nm}}K_f^{(2),\times}\otimes \mathbb{F}_2\big),
    \end{align*}
    while the cokernel of the map 
    \begin{align*}
    H^1(K,\mathbb{F}_2[\mathcal{R}_f])\rightarrow H^1(K,\mathbb{F}_2[\mathcal{R}_f]\otimes {\wedge^2 J[2]}),
    \end{align*}
    associated to \eqref{imageJ2map}, is induced by the inclusion $K_f\subseteq K_f^{(1,2)}$ of \'{e}tale algebras. 
\end{enumerate}

<!-- theorem-2: body_md -->
\label{mainthm}
Let $X$ be a hyperelliptic curve over $\mathbb{Q}$ defined by the equation $y^2=f(x)$, where the degree of $f\in \mathbb{Q}[x]$ is odd. Let $S$ be a finite set of primes containing $\{2,\infty\}$ and the primes of bad reduction for $X$. Additionally, suppose that the following inequality holds:
\begin{align*}
\dim_{\mathbb{Q}_2}H^1_f(\mathbb{Q},U[2])+d_{X,S}<\frac{1}{6}(g+1)(2g+1)(7g-6)-\mathrm{rk}_{\mathbb{Z}}J(\mathbb{Q}).
\end{align*}
Then the depth $3$ Chabauty--Kim set $X(\mathbb{Q}_2)_3$ is finite.
<!-- end theorem-2 -->
<!-- proof-9: body_md -->
By Lemma \ref{wedge3split}, Lemma \ref{splitlemmaJ2} and the dimension calculation \eqref{dimforthm}, it follows that $d_{X,S}$ is precisely the $\mathbb{F}_2$-dimension of $H^1(G_S,J[2]\otimes \overline{\wedge^2 J[2]}/{\wedge^3 J[2]})$, which, given the choice of $S$, is a bound for the $\mathbb{Q}_2$-dimension of $H^1_f(\mathbb{Q},U[3])$ by Proposition \ref{MainApprox}. The result then follows from Lemma \ref{netancor3}.

<!-- lemma-9: body_md -->
\label{d1lemma}
Let $S$ be a finite set of primes in $K$ containing primes above $\{2,\infty\}$. Then we have the following inequality:
\begin{align}\label{lem21ineq}
\dim_{\mathbb{F}_2}H^1(G_{K,S},\mathbb{F}_2[\mathcal{R}_f]\otimes \wedge^2 J[2])\leq 2\big(\mathrm{rk}_{\mathbb{Z}}\mathcal{O}_{K_{f,h},S}^\times + \dim_{\mathbb{F}_2}\mathrm{Cl}_S(K_{f,h})[2]\big)+1.
\end{align}
<!-- end lemma-9 -->
<!-- proof-10: body_md -->
This is a consequence of the following short exact sequence of $G_{K_{f,h}}$-modules: 
\begin{align}\label{quadboundary}
0\rightarrow \mathbb{F}_2 \rightarrow \mathrm{Ind}_{K_{f,h}}^{K_f^{(1,2)}}\mathbb{F}_2 \rightarrow \mathbb{F}_2 \rightarrow 0.
\end{align}
Specifically, by Lemma \ref{simple21isom} the dimension on the left-hand side of \eqref{lem21ineq} coincides with the dimension of the middle term in the exact sequence
\begin{align}\label{cokerkerSES}
0\rightarrow \mathrm{coker}(\delta_0)\rightarrow H^1(G_{K_{f,h},S},\mathrm{Ind}_{K_{f,h}}^{K_f^{(1,2)}}\mathbb{F}_2) \rightarrow \mathrm{ker}(\delta_1) \rightarrow 0
\end{align}
where $\delta_0$ and $\delta_1$ are the boundary maps
\begin{align*}
\delta_i&:H^i(G_{K_{f,h},S},\mathbb{F}_2)\rightarrow H^{i+1}(G_{K_{f,h},S},\mathbb{F}_2),
\end{align*}
associated to \eqref{quadboundary}. The boundary map $\delta_0$ is clearly injective as it sends $1\in \mathbb{F}_2$ to the class of $c\in K_{f,h}^\times \otimes \mathbb{F}_2$, so the dimension of its cokernel is $\dim_{\mathbb{F}_2}H^1(G_{K,S},\mathrm{Ind}_K^{K_{f,h}}\mathbb{F}_2)-1$. The result then follows from \eqref{cokerkerSES} and Lemma \ref{Scohomolem}.

<!-- theorem-3: body_md -->
\label{rank4thm}
Suppose that $X$ is the genus $2$ hyperelliptic curve over $\mathbb{Q}$, defined by the equation
\begin{align*}
y^2=4x^5 + 8x^4 + 4x^2 + 8x + 1,
\end{align*}
whose Jacobian has Mordell--Weil rank $4$. Then the depth $3$ Chabauty--Kim set $X(\mathbb{Q}_2)_3$ is finite.
<!-- end theorem-3 -->
<!-- proof-11: body_md -->
The curve has bad reduction only at $1115467$ and so we set $S=\{2,1115467,\infty\}$. To compute (a bound for) $d_X$, we use the splittings from Lemma \ref{wedgenlemma} and the fact that $J[2] \simeq \wedge^3 J[2]$ in the genus $2$ case to obtain the simplification
\begin{align}\label{dxsimp}
\begin{split}
d_{X} = \dim_{\mathbb{F}_2}H^1(G_S,\mathbb{F}_2[\mathcal{R}_f] \otimes {\wedge^2 J[2]})-\dim_{\mathbb{F}_2}H^1(G_S,\wedge^2 \mathbb{F}_2[\mathcal{R}_f])\\-\dim_{\mathbb{F}_2}H^1(G_S,\mathbb{F}_2[\mathcal{R}_f])+\dim_{\mathbb{F}_2}H^1(G_S,\mathbb{F}_2). 
\end{split}
\end{align}
To compute each of these we use Lemma \ref{Scohomolem}, while we additionally use Lemma \ref{d1lemma} to compute the first term. These computations are simplified by the fact that, through computing dimensions, we obtain the isomorphisms $\mathbb{Q}_f^{(1,2)}\simeq \mathbb{Q}(\alpha,\beta+\gamma)$, $\mathbb{Q}_f^{(2)}\simeq \mathbb{Q}(\alpha+\beta)$ and $\mathbb{Q}_f\simeq \mathbb{Q}(\alpha)$ for distinct roots $\alpha,\beta,\gamma \in \overline{\mathbb{Q}}$ of $f$. In particular, we need the following basic computations:
\begin{itemize}
    \item The number of finite primes above $S$ in $\mathbb{Q}_{f,h}$, $\mathbb{Q}_f^{(2)}$ and $\mathbb{Q}_f$ is $5$, $4$ and $3$, respectively. 
    \item The unit rank of $\mathbb{Q}_{f,h}$, $\mathbb{Q}_f^{(2)}$ and $\mathbb{Q}_f$ is $8$, $6$ and $3$, respectively. 
    \item The relevant $S$-class groups coincide:
    \begin{align*}
    \mathrm{Cl}_S(\mathbb{Q}_{f,h})=\mathrm{Cl}_S(\mathbb{Q}_f^{(2)})=\mathrm{Cl}_S(\mathbb{Q}_f)\simeq \mathbb{Z}/2\mathbb{Z}.
    \end{align*}
\end{itemize}
Accordingly, we obtain from Lemma \ref{d1lemma} that the $\mathbb{F}_2$-dimension of $H^1(G_S,\mathbb{F}_2 [\mathcal{R}_f]\otimes \wedge^2 J[2])$ is at most $29$. Substituting the remaining quantities into \eqref{dxsimp}, we obtain that $d_X\leq 12$. To bound the dimension of $H^1_f(\mathbb{Q},U[2])$, we use the boundary map techniques of \cite{bk2} to compute that this dimension is at most 3. The result then follows from Theorem \ref{mainthm}.
