<!-- theorem-1: body_md -->
\label{thm:intro-informal}
Let $\phi^N$ have finite moments of all orders. Assume the spectral scaling and higher-cumulant conditions of Assumptions \ref{ass:spectral}  and \ref{ass:cumulants}, and denote the limiting covariance profile by $\gamma$. For the rescaled covariance profile, we obtain
\[
\mathcal X_N^P\overset{d}\longrightarrow\mathcal X^P
\qquad \text{ in} \text{  } H^{-s}(\mathbb T^d)
\text{ for } s\text{ sufficiently large},
\]
where $\mathcal X^P$ is a centered Gaussian random distribution with covariance 
\[
\mathbb E[\mathcal X^P(f)\mathcal X^P(g)]
= \displaystyle\sum_{k\in\mathbb Z^d}
\overline{\widehat f(k)}\widehat g(k)|p(k)|^2\gamma(k),
\]
and $f,g$ are suitable test functions.
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
Suppose the fields satisfy the tree--graph bound (see \eqref{def:treegraph}). Then, for every $r\geq 3$ and fixed smooth test functions $f_1,\ldots, f_r$,
\[
\bigl|\operatorname{cum}(\mathcal X_N^P(f_1),\ldots,\mathcal X_N^P(f_r))\bigr|
\lesssim N^{d-r(d/2+m)}\longrightarrow0 \qquad \text{ as }N\to\infty.
\]
In particular, the higher-cumulant assumption of Theorem 1.1 is satisfied.
<!-- end proposition-1 -->
<!-- corollary-1: body_md -->
Let $(\sigma_x)_x$ denote a spin configuration and let $F^1,\ldots,F^n$ be bounded single-site observables. For real $f,g\in C_c^\infty(\mathbb R^d)$, call the centered observable 
\[
\phi_x^a=F^a(\sigma_x)-\mathbb E_\mu[F^a(\sigma_x)],
\qquad
\mathcal W_N^a(f)=N^{-d/2}\sum_{x\in\mathbb Z^d}\phi_x^a f(x/N).
\]
Then the Dobrushin condition for the spin model implies that
\[
(\mathcal W_N^1,\ldots,\mathcal W_N^n)
\xrightarrow{\mathrm d}\mathcal W_\chi
\quad\text{in }H^{-s}_{\mathrm{loc}}(\mathbb R^d;\mathbb R^n),\quad s>d/2,
\]
where $\mu$ is the unique Gibbs state and $\mathcal W_\chi$ is centered Gaussian white noise with
\[
\mathbb E[\mathcal W_\chi^a(f)\mathcal W_\chi^b(g)]
=\chi_{ab}\int_{\mathbb R^d}f(x)g(x)\,dx,
\qquad
\chi_{ab}=\sum_{z\in\mathbb Z^d}\operatorname{Cov}_\mu(\phi_0^a,\phi_z^b).
\]
The susceptibility matrix is finite and positive semidefinite; the limit may be degenerate.
<!-- end corollary-1 -->
<!-- assumption-1: body_md -->
\label{ass:spectral}
There exist $\gamma:\mathbb Z^d\setminus\{0\}\to[0,\infty)$ and constants $C,M\ge0$
such that, for every $k\in\mathbb Z^d\setminus\{0\}$,
\begin{equation}\label{eq:spectral-scaling}
\lim_{N\to\infty}N^{-2m}\,\widehat\Gamma_N(k/N)=\gamma(k),
\end{equation}
and, for all $N\ge1$ and all $k\in\widehat\mathbb T^d_N$ with $1\le|k|$,
\begin{equation}\label{eq:domination}
N^{-2m}\,\widehat\Gamma_N(k/N)\le C\,(1+|k|)^{M}.
\end{equation}
Here $m$ is the order of the operator $P$ fixed in \eqref{eq:def:PN}.
At the zero mode we assume: if $m=0$, the limit \eqref{eq:spectral-scaling} holds also at
$k=0$ with $\gamma(0)<\infty$ (finite susceptibility); if $m\ge1$,
\begin{equation}\label{eq:zero-mode}
\widehat\Gamma_N(0)\le C\,N^{2m+M},
\end{equation}
and we set $\gamma(0):=0$ by convention, so that sums over all $k\in\mathbb Z^d$ in the limit
formulas below are well defined.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:cumulants}
For every $r\ge3$,
\begin{equation}\label{eq:higher-cumulants}
\lim_{N\to\infty}N^{-r(d/2+m)}\sum_{x_1,\dots,x_r\in\Lambda_N}\big|\kappa_r^{(N)}(x_1,\dots,x_r)\big|=0.
\end{equation}
<!-- end assumption-2 -->
<!-- theorem-2: body_md -->
\label{thm:main}
Let $P$ be a scalar constant-coefficient differential operator of exact order $m\ge0$
with symbol $p$, and let $(\phi^N_x)$ be centered and stationary, with finite moments of
all orders, satisfying Assumptions~\ref{ass:spectral} and~\ref{ass:cumulants}.  Then, for every
$s>\tfrac d2+m+\tfrac M2$, the random distributions $\mathcal X_N^{P}$ defined in \eqref{eq:def:PN}
converge in law in $H^{-s}(\mathbb T^d)$ to the centered Gaussian random distribution
$\mathcal X^P$ characterized by
\begin{equation}\label{eq:limit-cov}
\mathbb E\big[\langle\mathcal X^P,f\rangle\,\overline{\langle\mathcal X^P,g\rangle}\big]
=\sum_{k\in\mathbb Z^d}\overline{\widehat f(k)}\,\widehat g(k)\,|p(k)|^2\,\gamma(k),
\qquad f,g\in C^\infty(\mathbb T^d).
\end{equation}
<!-- end theorem-2 -->
<!-- assumption-3: body_md -->
\label{ass:wholespace}
There exist a measurable $\gamma:\mathbb R^d\setminus\{0\}\to[0,\infty)$ and constants
$C,M\ge0$ such that:
\begin{enumerate}[label=\textup{(W\arabic*)}]
\item\label{it:ws-scaling} $G_N(\xi)\to\gamma(\xi)$ for almost every $\xi\in\mathbb R^d$;
\item\label{it:ws-domination} $0\le G_N\le\Phi$ almost everywhere, for all $N$, where
$\Phi$ is locally integrable and $\Phi(\xi)\le C(1+|\xi|)^{M}$ for $|\xi|\ge1$;
\item\label{it:ws-cumulants} for every compact $K\subset\mathbb R^d$ and every $r\ge3$,
\begin{equation}\label{eq:ws-cum}
N^{-r(d/2+m)}\sum_{x_1,\dots,x_r\in NK\cap\mathbb Z^d}
\big|\kappa_r^{(N)}(x_1,\dots,x_r)\big|\longrightarrow0 .
\end{equation}
\end{enumerate}
<!-- end assumption-3 -->
<!-- theorem-3: body_md -->
\label{thm:wholespace}
Let $P$ be as in Theorem~\ref{thm:main} and let $(\phi^N_x)_{x\in\mathbb Z^d}$ be as above,
satisfying Assumption~\ref{ass:wholespace}.  Then, for all $f,g\in C_c^\infty(\mathbb R^d)$,
\begin{equation}\label{eq:ws-cov-limit}
\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)\longrightarrow
\int_{\mathbb R^d}\overline{\widehat f(\xi)}\,\widehat g(\xi)\,|p(\xi)|^2\gamma(\xi)\,\mathrm d\xi,
\end{equation}
and, for every $s>d/2+m+M/2$, the fields $\mathcal X_N^{P}$ of \eqref{eq:def:PN-ws} converge in law
in $H^{-s}_{\mathrm{loc}}(\mathbb R^d)$ (i.e.\ $\psi\mathcal X_N^{P}\to\psi\mathcal X^P$ in law in
$H^{-s}(\mathbb R^d)$ for every $\psi\in C_c^\infty(\mathbb R^d)$) to the centered Gaussian random
distribution $\mathcal X^P$ with covariance \eqref{eq:ws-cov-limit}.
<!-- end theorem-3 -->
<!-- remark-1: body_md -->
\label{rem:ws-usage}
For a fixed field with spectral density $S$, \ref{it:ws-scaling} reads
$N^{-2m}S(\xi/N)\to\gamma(\xi)$ a.e., and \ref{it:ws-domination} is a global majorant
for the rescaled density. Condition \eqref{eq:ws-cum} follows from the $\Lambda_N$-version
\eqref{eq:higher-cumulants} by stationarity, the sum over $NK\cap\mathbb Z^d$ being bounded by
the sum over a box of side $\lceil CN\rceil$ containing $NK$, and from the tree--graph criterion in the form of
Remark~\ref{rem:treegraph-Zd}.  It is in this form that the theory is applied in
\S\ref{sec:applications}.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
\label{rem:vector}
If $P=(P^{(1)},\dots,P^{(n)})$ has scalar components $P^{(i)}$ of common order $m$ and symbols $p^{(i)}$, then $\mathcal X_N^{P}$ is $\mathbb R^n$-valued and Theorem~\ref{thm:main} holds verbatim by the Cram\'er--Wold device \cite{cramer_theorems_1936} applied to arbitrary linear combinations $\sum_i a_i\,\langle\mathcal X_N^{P},f\rangle^{(i)}$, the limiting covariance being the matrix multiplier with entries
$\overline{p^{(i)}(k)}\,p^{(j)}(k)\,\gamma(k)$.  In particular $P=\nabla$ ($p^{(i)}(k)=2\pi i k_i$) and $P=\nabla^{\otimes m}$ (tensor-valued, with component symbols $p^{(\alpha)}(k)=(2\pi i k)^{\alpha}$ for multi-indices $|\alpha|=m$) are covered; when $\gamma(k)\sim c_*\,|k|^{-2m}$ the $m$-th order symbols cancel the infrared singularity and $|p^{(\alpha)}(k)|^2\gamma(k)$ stays bounded.  The limit of the $m$-th gradient observable is then \emph{projected} tensor white noise, with covariance $c_*(2\pi)^{2m}\sum_{k\ne0}\overline{(ik)^{\alpha}}(ik)^{\alpha'}|k|^{-2m}\,
\overline{\widehat f(k)}\,\widehat g(k)$: for $m=1$ the matrix multiplier is proportional to $k\otimes k/|k|^2$ ($k\ne0$), so the limit is the gradient of the continuum mean-zero Gaussian free field and \emph{not} an ordinary $d$-component white noise (it is isonormal white noise only after restricting the test space to gradient vector fields).

<!-- corollary-2: body_md -->
\label{cor:white-noise}
Under the hypotheses of Theorem~\ref{thm:main}, suppose there is $c_*>0$ with
\[
|p(k)|^2\,\gamma(k)=c_*\qquad\text{for all }k\in\mathbb Z^d\setminus Z(p).
\]
When $P=\mathrm{Id}$, constancy forces $\gamma(k)=c_*$ for every $k\in\mathbb Z^d$, and $c_*=\lim_N\widehat\Gamma_N(0)$
is the limiting susceptibility of the fields.  Then
\[
\mathbb E\big[\langle\mathcal X^P,f\rangle\,\overline{\langle\mathcal X^P,g\rangle}\big]
=c_*\sum_{k\in\mathbb Z^d\setminus Z(p)}\overline{\widehat f(k)}\,\widehat g(k).
\]
In particular, if $P=\mathrm{Id}$ then $\mathcal X_N^{P}\xrightarrow{d}\sqrt{c_*}\,W$ in $H^{-s}(\mathbb T^d)$
for every $s>d/2+M/2$, with $W$ standard white noise on $\mathbb T^d$; and if $Z(p)=\{0\}$ (e.g.
$P=\partial_v$ in $d=1$, or any elliptic homogeneous $p$ vanishing only at $0$), the same
holds on the closed subspace of mean-zero test functions, in $H^{-s}(\mathbb T^d)$ for every
$s>d/2+m+M/2$.

<!-- proof-1: body_md -->
Insert $|p(k)|^2\gamma(k)=c_*$ for
$k\notin Z(p)$ into \eqref{eq:limit-cov}; the terms with $k\in Z(p)$ carry the factor
$|p(k)|^2=0$ and drop out.  When $P=\mathrm{Id}$, $Z(p)=\emptyset$ and the covariance is

The topology is the one supplied by Theorem~\ref{thm:main}: exact cancellation
constrains only the limiting fixed frequencies and gives no ultraviolet control on the
finite-$N$ modes, so it cannot improve the Sobolev index of the tightness estimate.  For
instance, Gaussian fields with all fixed low-frequency spectral weights equal to $1$ and
one symmetric pair of modes $\pm k_N$, $|k_N|\asymp N$, of weight $N^{M}$ satisfy

cancellation for $P=\mathrm{Id}$, yet the $H^{-s}$ mass of that pair is of order
$N^{M-2s}$, so the sequence is not tight in $H^{-s}$ when $M>2s$.  When the spectral
densities are uniformly bounded one may take $M=0$, and for $P=\mathrm{Id}$ the
convergence then holds for every $s>d/2$.

\label{ex:gff-gradient}
Let $\phi^N$ be the lattice GFF on $\mathbb T^d_N$ at inverse temperature
$\beta\in(0,\infty)$: the centered Gaussian field with
$\widehat\Gamma_N(\theta)=(\beta\lambda_N(\theta))^{-1}$ for $\theta\ne0$ and
$\widehat\Gamma_N(0):=0$, where $\lambda_N(\theta):=\sum_{j=1}^d4\sin^2(\pi\theta_j)$ is
the symbol of the discrete Laplacian (zero mode removed: the field is defined modulo
constants).  The spectral density blows up near the zero mode,
$\widehat\Gamma_N(k/N)\sim N^2/(\beta(2\pi)^2|k|^2)$ for fixed $k\ne0$, so the $m=0$ case of
Assumption~\ref{ass:spectral} fails---there is no finite limiting susceptibility---and it can be shown (by Equation \eqref{eq:cov-exact} below) that the variance of the order-zero average $\mathcal X_N^{\mathrm{Id}}(f)$ diverges like $N^{2}$ for every non-constant $f$.  Assumption~\ref{ass:spectral} instead holds with $m=1$: for the scaling \eqref{eq:spectral-scaling},
\[
N^{-2}\widehat\Gamma_N(k/N)=\frac{N^{-2}}{\beta\lambda_N(k/N)}
\longrightarrow\gamma(k)=\frac{1}{\beta\,(2\pi)^2|k|^2}\qquad(k\ne0),
\]
since $\lambda_N(k/N)=\sum_j4\sin^2(\pi k_j/N)\sim(2\pi)^2|k|^2/N^2$, confirming that
$N^{2}$ (i.e.\ the normalization $N^{-d/2-1}$ in \eqref{eq:def:PN}) is the correct
scaling; the domination \eqref{eq:domination} holds with $M=0$, since
$\sin^2(\pi t)\ge4t^2$ for $|t|\le\tfrac12$ gives
$N^{-2}\widehat\Gamma_N(k/N)\le(16\,\beta|k|^2)^{-1}\le C$ for all $k\in\widehat\mathbb T^d_N$
with $1\le|k|$; and the zero-mode bound \eqref{eq:zero-mode} is trivial,
$\widehat\Gamma_N(0)=0$.  The field is Gaussian, so Assumption~\ref{ass:cumulants} is
immediate.  The profile $\gamma(k)=c_*|k|^{-2}$ with $c_*=1/(\beta(2\pi)^2)$ is exactly the
cancellation regime of Remark~\ref{rem:vector}: for $P=\nabla$, with symbols
$p^{(i)}(k)=2\pi ik_i$,
\[
\sum_{i=1}^d|p^{(i)}(k)|^2\,\gamma(k)
=(2\pi)^2|k|^2\cdot\frac{1}{\beta\,(2\pi)^2|k|^2}=\frac1\beta\qquad(k\ne0),
\]
and the gradient observable converges on the mean-zero subspace to the projected white noise of Remark~\ref{rem:vector},
\[
\mathbb E\big[\langle\mathcal X^{\nabla,i},f\rangle\,\overline{\langle\mathcal X^{\nabla,j},g\rangle}\big]
=\frac1\beta\sum_{k\ne0}\frac{k_ik_j}{|k|^2}\,\overline{\widehat f(k)}\,\widehat g(k).
\]
The limit is $\beta^{-1/2}$ times the gradient of the
continuum mean-zero Gaussian free field, acting as white noise of intensity $1/\beta$ on
gradient test fields.
This is the exactly Gaussian, torus analogue of the low-temperature gradient white-noise limits of Newman and Wu \cite{newman_gaussian_2018}.

<!-- lemma-1: body_md -->
\label{lem:alias}
For $f,g\in C^\infty(\mathbb T^d)$,
\begin{equation}\label{eq:cov-exact}
\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)
=\sum_{k\in\widehat\mathbb T^d_N}\overline{\widehat{Pf}^{(N)}(k)}\,\widehat{Pg}^{(N)}(k)\;
N^{-2m}\,\widehat\Gamma_N\!\big(k/N\big),
\end{equation}
a \emph{finite} sum over the dual group $\widehat\mathbb T^d_N=(-N/2,N/2]^d\cap\mathbb Z^d$, where
\[
\widehat{Ph}^{(N)}(k):=N^{-d}\sum_{x\in\Lambda_N}(Ph)(x/N)\,e^{-2\pi i k\cdot x/N}
\]
is the discrete Fourier coefficient.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
By definition \eqref{eq:def:PN} and stationarity,
\[
\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)
=N^{-d-2m}\sum_{x,y\in\Lambda_N}\Gamma_N(x-y)\,\overline{(Pf)(x/N)}\,(Pg)(y/N).
\]
Insert the spectral representation \eqref{eq:disc-bochner} for $\Gamma_N(x-y)$ and the
discrete Fourier inversions
$(Ph)(x/N)=\sum_{k'\in\widehat\mathbb T^d_N}\widehat{Ph}^{(N)}(k')\,e^{2\pi i k'\cdot x/N}$ for
$h=f,g$.  The orthogonality relation
\begin{equation}\label{eq:disc-orth}
N^{-d}\sum_{x\in\Lambda_N}e^{2\pi i(k-k')\cdot x/N}=\mathbf 1_{\{k\equiv k'\ (\mathrm{mod}\ N)\}},
\qquad k,k'\in\widehat\mathbb T^d_N ,
\end{equation}
collapses the sums over $x$ and over $y$. Since we work in$\widehat\mathbb T^d_N$, only the diagonal terms survive, and collecting them gives
\eqref{eq:cov-exact}.

<!-- proposition-2: body_md -->
\label{prop:cov-limit}
Under Assumption~\ref{ass:spectral},
\begin{equation}\label{eq:cov-converges}
\lim_{N\to\infty}\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)

\end{equation}

<!-- proof-3: body_md -->
\emph{Step 1: aliasing.}  We first show that for every $h\in C^\infty(\mathbb T^d)$ the discrete
coefficients of Lemma~\ref{lem:alias} are the exact aliased sums
\begin{equation}\label{eq:alias}
\widehat{Ph}^{(N)}(k)=\sum_{\ell\in\mathbb Z^d}\widehat{Ph}(k+N\ell),\qquad
\widehat{Ph}(\zeta)=p(\zeta)\,\widehat h(\zeta),
\end{equation}
i.e., an equivalent statement to Poisson summation on $\Lambda_N$.  Since $Ph\in C^\infty(\mathbb T^d)$, the rapid decay
\eqref{eq:rapid-decay} applied to $h$ gives, for every $M''>0$,
\begin{equation}\label{eq:Ph-decay}
|\widehat{Ph}(\zeta)|=|p(\zeta)|\,|\widehat h(\zeta)|
\le C_p|\zeta|^m\,|\widehat h(\zeta)|
\le C_{h,M''}\,(1+|\zeta|)^{-M''},
\end{equation}
so the Fourier expansion $(Ph)(\theta)=\sum_{n\in\mathbb Z^d}\widehat{Ph}(n)\,e^{2\pi in\cdot\theta}$
converges absolutely.  Substituting it into the definition of $\widehat{Ph}^{(N)}(k)$ and
interchanging the two (absolutely convergent) sums,
\[
\widehat{Ph}^{(N)}(k)
=\sum_{n\in\mathbb Z^d}\widehat{Ph}(n)\,N^{-d}\sum_{x\in\Lambda_N}e^{2\pi i(n-k)\cdot x/N}
=\sum_{n\in\mathbb Z^d}\widehat{Ph}(n)\,\mathbf 1_{\{n\equiv k\ (\mathrm{mod}\ N)\}},
\]
by the orthogonality relation \eqref{eq:disc-orth}, which holds for arbitrary $n\in\mathbb Z^d$
because both sides are $N$-periodic in $n$; writing the surviving indices as $n=k+N\ell$,
$\ell\in\mathbb Z^d$, gives \eqref{eq:alias}.

\emph{Step 2: uniform decay.}  We next claim that on the dual box $\widehat\mathbb T^d_N$,
\begin{equation}\label{eq:disc-decay}
|\widehat{Ph}^{(N)}(k)|\le C_{h,M'}(1+|k|)^{-M'}\quad\text{uniformly in }N,\ \text{for every }M',
\end{equation}
with a constant depending also on $d$.
Fix $M'>0$ and use \eqref{eq:Ph-decay} with $M''=M'+d+1$.  The $\ell=0$ term of
\eqref{eq:alias} is bounded by $C_{h,M''}(1+|k|)^{-M'-d-1}\le C_{h,M''}(1+|k|)^{-M'}$.  For
the aliasing tail, every $k\in\widehat\mathbb T^d_N$ satisfies $|k_j|\le N/2$ coordinatewise, so for
$\ell\ne0$,
\[
|k_j+N\ell_j|\ge N|\ell_j|-N/2\ge\frac N2|\ell_j|\quad(\ell_j\ne0),
\]
hence
\[
|k+N\ell|^2\ge\sum_{j:\ell_j\ne0}\Big(\frac N2\Big)^2\ell_j^2=\Big(\frac N2\Big)^2|\ell|^2,
\]
so that
\[
\sum_{\ell\ne0}\big|\widehat{Ph}(k+N\ell)\big|
\le C_{h,M''}\sum_{\ell\ne0}\Big(\frac{N|\ell|}{2}\Big)^{-M'-d-1}
=C_{h,M''}\Big(\frac2N\Big)^{M'+d+1}\sum_{\ell\ne0}|\ell|^{-M'-d-1}
\le C'\,N^{-M'},
\]
the last series converging because $M'+d+1>d$.  Since $1+|k|\le1+\sqrt d\,N/2\le\sqrt d\,N$
on $\widehat\mathbb T^d_N$ (for $N\ge2$), the tail is also bounded by $C'_d(1+|k|)^{-M'}$, and
\eqref{eq:disc-decay} follows.  Moreover, for fixed $k$ the tail is $O(N^{-M'})$, so
\begin{equation}\label{eq:disc-conv}
\widehat{Ph}^{(N)}(k)\longrightarrow\widehat{Ph}(k)=p(k)\,\widehat h(k)
\qquad\text{as }N\to\infty,\ k\ \text{fixed}.
\end{equation}

\emph{Step 3: passage to the limit.}  The sum \eqref{eq:cov-exact} runs over the dual box
$\widehat\mathbb T^d_N$; write it as $\sum_{k\in\mathbb Z^d}\mathbf 1_{\widehat\mathbb T^d_N}(k)\,T_N(k)$ with
$T_N(k):=\overline{\widehat{Pf}^{(N)}(k)}\,\widehat{Pg}^{(N)}(k)\,N^{-2m}\widehat\Gamma_N(k/N)$.
Fix $k\in\mathbb Z^d\setminus\{0\}$.  For $N>2|k|$, $k$ lies in $\widehat\mathbb T^d_N$ and, by \eqref{eq:disc-conv},
$\widehat{Pf}^{(N)}(k)\to p(k)\widehat f(k)$ and $\widehat{Pg}^{(N)}(k)\to p(k)\widehat g(k)$,
while \eqref{eq:spectral-scaling} gives $N^{-2m}\widehat\Gamma_N(k/N)\to\gamma(k)$; thus
$\mathbf 1_{\widehat\mathbb T^d_N}(k)T_N(k)\to|p(k)|^2\gamma(k)\,\overline{\widehat f(k)}\,\widehat g(k)$
for every $k\ne0$.

The zero mode requires its own estimate when $m\ge1$: the discrete coefficient at $k=0$
is \emph{not} $p(0)\widehat f(0)$ but the alias sum \eqref{eq:alias}, in which only the
$\ell=0$ term vanishes.  By \eqref{eq:alias} and $p(0)=0$,
$\widehat{Ph}^{(N)}(0)=\sum_{\ell\ne0}\widehat{Ph}(N\ell)$, so \eqref{eq:Ph-decay} with
an exponent $M''>\max(d,M/2)$ gives
$|\widehat{Ph}^{(N)}(0)|\le C_{h,M''}\sum_{\ell\ne0}(N|\ell|)^{-M''}\le C\,N^{-M''}$
for $h=f,g$; together with the zero-mode bound \eqref{eq:zero-mode},
\[
|T_N(0)|\le C\,N^{-2M''}\cdot N^{-2m}\widehat\Gamma_N(0)\le C\,N^{M-2M''}\longrightarrow0 ,
\]
so $T_N(0)\to0=|p(0)|^2\gamma(0)$, consistently with the convention $\gamma(0)=0$ of
Assumption~\ref{ass:spectral}.  For $m=0$,
$T_N(0)\to|p(0)|^2\gamma(0)\,\overline{\widehat f(0)}\,\widehat g(0)$ directly, by
\eqref{eq:disc-conv} and the zero-mode part of Assumption~\ref{ass:spectral}.

For the interchange of limit and summation over $k\ne0$ we dominate uniformly in $N$: on
$\widehat\mathbb T^d_N$, \eqref{eq:disc-decay} gives
$|\widehat{Pf}^{(N)}(k)|,\,|\widehat{Pg}^{(N)}(k)|\le C(1+|k|)^{-(M+d+1)/2-m}$ (choosing $M'$
large), and \eqref{eq:domination} gives $N^{-2m}\widehat\Gamma_N(k/N)\le C(1+|k|)^M$ for
$k\in\widehat\mathbb T^d_N$ with $|k|\ge1$.  Hence $|\mathbf 1_{\widehat\mathbb T^d_N}(k)T_N(k)|\le C'(1+|k|)^{-(d+1)}$ for $k\ne0$, summable over $\mathbb Z^d$, and dominated convergence for series yields \eqref{eq:cov-converges}.

<!-- lemma-2: body_md -->
\label{lem:fdd}
Under Assumption~\ref{ass:cumulants}, for every $f_1,\dots,f_r\in C^\infty(\mathbb T^d)$ with
$r\ge3$, we have that
\begin{equation*}
    \operatorname{cum}(\mathcal X_N^{P}(f_1),\dots,\mathcal X_N^{P}(f_r))\to0.
\end{equation*}
Consequently the finite-dimensional
distributions of $\mathcal X_N^{P}$ converge to those of the centered Gaussian field with covariance
\eqref{eq:limit-cov}.
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
By multilinearity,
\[
\operatorname{cum}(\mathcal X_N^{P}(f_1),\dots,\mathcal X_N^{P}(f_r))
=N^{-r(d/2+m)}\sum_{x_1,\dots,x_r\in\Lambda_N}\kappa_r^{(N)}(x_1,\dots,x_r)\prod_{j=1}^r(Pf_j)(x_j/N).
\]
Bounding each $Pf_j$ by $\|Pf_j\|_\infty$ and using \eqref{eq:higher-cumulants} gives the
limit $0$.

It remains to upgrade the convergence of cumulants to convergence of laws; convergence
of all cumulants to the Gaussian ones does not, by itself, prove convergence of a
sequence of laws, and the missing step is the method of moments.  Let
$f_1,\dots,f_\ell\in C^\infty(\mathbb T^d)$ be real-valued, $a_1,\dots,a_\ell\in\mathbb R$, and set
$Z_N:=\sum_{j=1}^\ell a_j\mathcal X_N^{P}(f_j)$.  By multilinearity, the above, Proposition~\ref{prop:cov-limit}
($r=2$) and centering ($r=1$), every cumulant of $Z_N$ converges to the corresponding
cumulant of the centered Gaussian law with variance
$\sigma^2:=\sum_{i,j}a_ia_j\sum_k\overline{\widehat f_i(k)}\widehat f_j(k)|p(k)|^2\gamma(k)$.
It is well known that moments can be expressed as cumulants polynomials, i.e., by the moment--cumulant formula,
\[
\mathbb E[Z_N^r]=\sum_{\pi\in\mathcal P_r}\prod_{B\in\pi}\operatorname{cum}_{|B|}(Z_N),
\]
so every moment of $Z_N$ converges to the corresponding moment of the Gaussian law
\cite[Cor.~3.2.2]{peccati_wiener_2011}: this ensures that $Z_N\Rightarrow\mathcal N(0,\sigma^2)$. Finally, the Cram\'er--Wold device \cite{cramer_theorems_1936} converts the convergence of all real linear combinations into convergence of the finite-dimensional distributions; complex test functions are recovered by splitting into real and imaginary parts.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
By Lemma~\ref{lem:fdd} the finite-dimensional distributions converge to the claimed
Gaussian law.  It remains to prove tightness in $H^{-s}(\mathbb T^d)$ for $s>d/2+m+M/2$, after
which Prokhorov's theorem (tightness gives a relatively compact family on the separable
Hilbert space $H^{-s}(\mathbb T^d)$) together with the convergence of the finite-dimensional
distributions identifies the unique limit \cite[\S5]{billingsley_convergence_1999}.

Using $Pe_k=p(k)e_k$ and the exact covariance \eqref{eq:cov-exact} with $f=g=e_k$, by the
orthogonality relation \eqref{eq:disc-orth} the only principal frequency at which the
discrete coefficient of $Pe_k$ is nonzero is $\langle k\rangle_N$, where it equals $p(k)$;
hence
$\operatorname{Var}(\langle\mathcal X_N^{P},e_k\rangle)=|p(k)|^2\,N^{-2m}\widehat\Gamma_N(\langle k\rangle_N/N)$.  Using
$|p(k)|\le C_p(1+|k|)^m$, $|\langle k\rangle_N|\le|k|$, the domination \eqref{eq:domination},
and, for the residual modes $k\equiv0\ (\mathrm{mod}\,N)$, $k\ne0$ (which satisfy
$|k|\ge N$), the zero-mode part of Assumption~\ref{ass:spectral}, we obtain
\begin{equation}\label{eq:var-bound}
\operatorname{Var}\big(\langle\mathcal X_N^{P},e_k\rangle\big)\le C_*\,(1+|k|)^{2m+M},
\end{equation}
for a constant $C_*$ independent of $N,k$. We have used that for $m\ge1$, \eqref{eq:zero-mode} gives $N^{-2m}\widehat\Gamma_N(0)\le CN^{M}\le C(1+|k|)^{M}$, and for $m=0$ the convergence $\widehat\Gamma_N(0)\to\gamma(0)<\infty$ gives a uniform bound (when the zero mode is removed, $\widehat\Gamma_N(0)=0$, these terms are absent). 
Therefore, by Parseval in $H^{-s'}$, for every
$s'\in\mathbb R$,
\[
\sup_N\mathbb E\big[\|\mathcal X_N^{P}\|_{H^{-s'}}^2\big]
=\sup_N\sum_{k\in\mathbb Z^d}(1+|k|^2)^{-s'}\,\operatorname{Var}\big(\langle\mathcal X_N^{P},e_k\rangle\big)
\le C_*\sum_{k\in\mathbb Z^d}(1+|k|^2)^{-s'}(1+|k|)^{2m+M},
\]
which is finite as soon as $2s'-(2m+M)>d$, i.e. $s'>d/2+m+M/2$, by comparison of the series
with the integral $\int_{\mathbb R^d}(1+|\xi|)^{2m+M-2s'}\,\mathrm d\xi$ (the summand and the integrand
differ by a bounded factor on each unit cube $k+[0,1)^d$), which converges precisely when
$2s'-(2m+M)>d$.
Given $s>d/2+m+M/2$, pick $s'\in(d/2+m+M/2,\,s)$: the uniform bound holds in $H^{-s'}$,
and the embedding $H^{-s'}\hookrightarrow H^{-s}$ is compact.                                                                                                                           
Tightness follows: by Markov's inequality the closed $H^{-s'}$-balls
$K_R:=\{\|u\|_{H^{-s'}}\le R\}$, which are compact in $H^{-s}$, satisfy
$\sup_N\mathbb P(\mathcal X_N^{P}\notin K_R)\le R^{-2}\sup_N\mathbb E\|\mathcal X_N^{P}\|_{H^{-s'}}^2\to0$ as $R\to\infty$
\cite[\S5]{billingsley_convergence_1999}.  This completes the proof.

<!-- lemma-3: body_md -->
\label{lem:ws-cov}
For all $f,g\in C_c^\infty(\mathbb R^d)$,
\begin{equation}\label{eq:ws-exact-cov}
\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)
=\int_{Q_N}G_N(\xi)\,\overline{A_{N,Pf}(\xi)}\,A_{N,Pg}(\xi)\,\mathrm d\xi .
\end{equation}
Moreover, for every $h\in C_c^\infty(\mathbb R^d)$:
\begin{enumerate}[label=\textup{(\alph*)}]
\item\label{it:ws-unif} $\sup_{N\ge1}\sup_{\xi\in\mathbb R^d}|A_{N,h}(\xi)|<\infty$;
\item\label{it:ws-decay} for every $M'>0$ there is $C=C(h,M',d)$ with
$|A_{N,h}(\xi)|\le C(1+|\xi|)^{-M'}$ for all $N\ge1$ and $\xi\in Q_N$;
\item\label{it:ws-conv} $\sup_{\xi\in Q_N}|A_{N,h}(\xi)-\widehat h(\xi)|=O(N^{-M'})$
for every $M'>0$.
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
By \eqref{eq:def:PN-ws}, stationarity, and the convention
$\operatorname{Cov}(A,B)=\mathbb E[\overline{(A-\mathbb E A)}(B-\mathbb E B)]$,
\[
\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)
=N^{-d-2m}\sum_{x,y\in\mathbb Z^d}\Gamma_N(x-y)\,\overline{(Pf)(x/N)}\,(Pg)(y/N),
\]
a finite double sum.  Inserting the spectral representation of $\Gamma_N$, interchanging
the (finite) sums with the integral, and substituting $\theta=\xi/N$
($\,\mathrm d\theta=N^{-d}\,\mathrm d\xi$, $\theta\in\mathbb T^d\leftrightarrow\xi\in Q_N$) gives
\begin{equation*}
\begin{split}
    &\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)=\\
    &=\int_{Q_N}N^{-2m}S_N(\xi/N) \Big(\overline{N^{-d}\!\sum_x(Pf)(x/N)e^{-2\pi i\xi x/N}}\Big) \Big(N^{-d}\!\sum_y(Pg)(y/N)e^{-2\pi i\xi y/N}\Big)\,\mathrm d\xi,
\end{split}
\end{equation*}
which is \eqref{eq:ws-exact-cov}.  

Turning to the other statements: \ref{it:ws-unif} holds since $|A_{N,h}(\xi)|\le N^{-d}\sum_x|h(x/N)|$, a Riemann sum of $\int|h|$.  For \ref{it:ws-decay}, fix $M'$ and use the fact that $h$ is notably Schwartz function and thus $|\widehat h(\zeta)|\le C_{h,M''}(1+|\zeta|)^{-M''}$ with $M''=M'+d+1$. Putting this in \eqref{eq:ws-alias}: the $\ell=0$ term is $\le C(1+|\xi|)^{-M'}$, and for $\ell\ne0$, coordinatewise $|\xi_j+N\ell_j|\ge N|\ell_j|-N/2\ge(N/2)|\ell_j|$ on $Q_N$, hence $|\xi+N\ell|\ge(N/2)|\ell|$, so
\[
\sum_{\ell\ne0}|\widehat h(\xi+N\ell)|
\le C_{h,M''}\Big(\frac2N\Big)^{M''}\sum_{\ell\ne0}|\ell|^{-M''}\le C'N^{-M'-d-1},
\]
and $1+|\xi|\le C_dN$ on $Q_N$ makes the tail $\le C''(1+|\xi|)^{-M'}$ as well.  The
same tail estimate gives \ref{it:ws-conv}.

<!-- proof-7: body_md -->
\emph{Step 1: covariance.}  By Lemma~\ref{lem:ws-cov}\,\ref{it:ws-conv} and
\ref{it:ws-scaling}, the integrand of \eqref{eq:ws-exact-cov} converges almost
everywhere to
$\overline{\widehat{Pf}(\xi)}\,\widehat{Pg}(\xi)\gamma(\xi)
=\overline{\widehat f(\xi)}\,\widehat g(\xi)\,|p(\xi)|^2\gamma(\xi)$.  For a dominant,
split $\mathbb R^d$: on $|\xi|\le1$, Lemma~\ref{lem:ws-cov}\,\ref{it:ws-unif} bounds
$|\overline{A_{N,Pf}}A_{N,Pg}|$ by a constant and $G_N\le\Phi\in L^1(\{|\xi|\le1\})$; on
$|\xi|\ge1$, Lemma~\ref{lem:ws-cov}\,\ref{it:ws-decay} with $M'=(M+d+1)$ and
$G_N(\xi)\le C(1+|\xi|)^M$ give the integrable dominant $C(1+|\xi|)^{-d-1}$.  Dominated
convergence yields \eqref{eq:ws-cov-limit}, the limit being finite since
$\gamma\le\Phi$.

\emph{Step 2: cumulants and finite-dimensional distributions.}  Let
$f_1,\dots,f_r\in C_c^\infty(\mathbb R^d)$ be real, $r\ge3$, and let $K$ be a compact set
containing $\bigcup_j\operatorname{supp} f_j$, hence $\bigcup_j\operatorname{supp}(Pf_j)$.  By multilinearity,
\[
\big|\operatorname{cum}(\mathcal X_N^{P}(f_1),\dots,\mathcal X_N^{P}(f_r))\big|
\le\prod_{j=1}^r\|Pf_j\|_\infty\;
N^{-r(d/2+m)}\!\!\sum_{x_1,\dots,x_r\in NK\cap\mathbb Z^d}\!\!\big|\kappa_r^{(N)}(x_1,\dots,x_r)\big|
\longrightarrow0
\]
by \eqref{eq:ws-cum}.  Together with Step~1 ($r=2$) and centering ($r=1$), all joint cumulants converge to those of the Gaussian field with covariance \eqref{eq:ws-cov-limit}; the moment-method argument in the proof of Lemma~\ref{lem:fdd} applies verbatim and gives convergence of the finite-dimensional distributions of $\mathcal X_N^{P}$ to those of $\mathcal X^P$.

\emph{Step 3: tightness in $H^{-s}_{\mathrm{loc}}$.}  Fix $\psi\in C_c^\infty(\mathbb R^d)$
and $s'\in(d/2+m+M/2,\,s)$.  The random distribution $\psi\mathcal X_N^{P}$ is supported in the
fixed compact $\operatorname{supp}\psi$ and its Fourier transform is the function
$\widehat{\psi\mathcal X_N^{P}}(\xi)=\mathcal X_N^{P}(\psi e_{-\xi})$, $e_{-\xi}(t):=e^{-2\pi i\xi\cdot t}$.
Since $h_\xi:=\psi e_{-\xi}\in C_c^\infty(\mathbb R^d)$ with
$\widehat{Ph_\xi}(\zeta)=p(\zeta)\,\widehat\psi(\zeta+\xi)$, formula
\eqref{eq:ws-exact-cov} (with $f=g=h_\xi$) and \eqref{eq:ws-alias} give
\[
\mathbb E\big|\widehat{\psi\mathcal X_N^{P}}(\xi)\big|^2
=\int_{Q_N}G_N(\eta)\,\big|A_N(\eta,\xi)\big|^2\,\mathrm d\eta,
\qquad
A_N(\eta,\xi):=\sum_{\ell\in\mathbb Z^d}p(\eta+N\ell)\,\widehat\psi(\eta+N\ell+\xi).
\]
By Cauchy--Schwarz on the $\ell$-sum,
\[
|A_N(\eta,\xi)|^2
\le\Big(\sum_{\ell}|p(\eta+N\ell)|^2\,|\widehat\psi(\eta+N\ell+\xi)|\Big)
\Big(\sum_{\ell}|\widehat\psi(\eta+N\ell+\xi)|\Big),
\]
and the second factor is bounded by a constant $C_\psi$ uniformly in $N\ge1$, $\eta$,
$\xi$: the points $\{\eta+N\ell+\xi\}_\ell$ are $N$-separated and $\widehat\psi$ is
Schwartz, so $\sum_\ell(1+|\eta+N\ell+\xi|)^{-(d+1)}\le C_d$.  Hence, by Tonelli,
\[
\mathbb E\big[\|\psi\mathcal X_N^{P}\|_{H^{-s'}}^2\big]
=\int_{\mathbb R^d}(1+|\xi|^2)^{-s'}\,\mathbb E\big|\widehat{\psi\mathcal X_N^{P}}(\xi)\big|^2\,\mathrm d\xi
\le C_\psi\int_{Q_N}\!G_N(\eta)\sum_\ell|p(\eta+N\ell)|^2\,I_\ell(\eta)\,\mathrm d\eta,
\]
with $I_\ell(\eta):=\int(1+|\xi|^2)^{-s'}|\widehat\psi(\eta+N\ell+\xi)|\,\mathrm d\xi$.  By
Peetre's inequality
$(1+|\xi|^2)^{-s'}\le2^{s'}(1+|\eta+N\ell|^2)^{-s'}(1+|\xi+\eta+N\ell|^2)^{s'}$, which implies that
\[
I_\ell(\eta)\le C(1+|\eta+N\ell|^2)^{-s'}\int(1+|\zeta|^2)^{s'}|\widehat\psi(\zeta)|\,\mathrm d\zeta
=C'_\psi\,(1+|\eta+N\ell|^2)^{-s'} .
\]
With $|p(\zeta)|^2\le C(1+|\zeta|)^{2m}$ this yields, for $\eta\in Q_N$,
\begin{equation*}
\begin{split}
&\sum_\ell|p(\eta+N\ell)|^2 I_\ell(\eta)
\le C\sum_\ell(1+|\eta+N\ell|)^{2m-2s'}\\
&\le C\big[(1+|\eta|)^{2m-2s'}+N^{2m-2s'}\textstyle\sum_{\ell\ne0}|\ell|^{2m-2s'}\big]
\le C''(1+|\eta|)^{2m-2s'},
\end{split}
\end{equation*}
using $|\eta+N\ell|\ge(N/2)|\ell|$, the convergence of
$\sum_{\ell\ne0}|\ell|^{2m-2s'}$ (as $2s'-2m>d$), and $1+|\eta|\le C_dN$ with
$2m-2s'<0$.  Therefore
\[
\sup_N\mathbb E\big[\|\psi\mathcal X_N^{P}\|_{H^{-s'}}^2\big]
\le C\int_{|\eta|\le1}\Phi(\eta)\,\mathrm d\eta
+C\int_{|\eta|\ge1}(1+|\eta|)^{M+2m-2s'}\,\mathrm d\eta<\infty,
\]
since $\Phi$ is locally integrable and $2s'>d+2m+M$.

Next, the embedding of $\{u\in H^{-s'}(\mathbb R^d):\operatorname{supp} u\subset K_0\}$ into $H^{-s}(\mathbb R^d)$
is compact for every compact $K_0$ and $s>s'$: if $(u_n)$ is bounded in $H^{-s'}$ with
supports in $K_0$, pick $\varphi\in C_c^\infty$ equal to $1$ on $K_0$; then
$\widehat u_n(\xi)=u_n(\varphi e_{-\xi})$ and its derivatives in $\xi$ are bounded on
compacts (by $\|u_n\|_{H^{-s'}}\|\partial^\alpha_\xi(\varphi e_{-\xi})\|_{H^{s'}}
\le C_\alpha(1+|\xi|)^{s'+|\alpha|}$), so by Arzel\`a--Ascoli a subsequence of
$(\widehat u_n)$ converges locally uniformly, and the splitting
$\|u_n-u_{n'}\|^2_{H^{-s}}\le\int_{|\xi|\le R}+\;2\sup_n\|u_n\|^2_{H^{-s'}}
(1+R^2)^{-(s-s')}$ shows it is Cauchy in $H^{-s}$.  By Markov's inequality the sets
$\{\|\psi\mathcal X_N^{P}\|_{H^{-s'}}\le R\}$ then provide tightness of $(\psi\mathcal X_N^{P})_N$ in
$H^{-s}(\mathbb R^d)$.

Finally, let $(\psi_j)_{j\ge1}\subset C_c^\infty$ with $\psi_j\equiv1$ on the ball
$B_j$.  Given $\varepsilon>0$ choose $R_j$ with
$\sup_N\mathbb P(\|\psi_j\mathcal X_N^{P}\|_{H^{-s'}}>R_j)\le\varepsilon2^{-j}$.  The set
$K_\varepsilon:=\{u:\|\psi_ju\|_{H^{-s'}}\le R_j\ \forall j\}$ is relatively compact in
$H^{-s}_{\mathrm{loc}}(\mathbb R^d)$: by the previous paragraph and a diagonal extraction,
every sequence in $K_\varepsilon$ has a subsequence such that $\psi_ju_n$ converges in
$H^{-s}$ for every $j$, and for any $h\in C_c^\infty$ one has $h=h\psi_j$ for
$j$ large, with multiplication by $h$ bounded on $H^{-s}$.  Since
$\mathbb P(\mathcal X_N^{P}\notin K_\varepsilon)\le\varepsilon$ uniformly in $N$, the family $(\mathcal X_N^{P})$ is
tight in the Polish space $H^{-s}_{\mathrm{loc}}(\mathbb R^d)$.  By Prokhorov's theorem and
Step~2---the pairings $u\mapsto\langle u,f\rangle$, $f\in C_c^\infty$, being continuous on
$H^{-s}_{\mathrm{loc}}$ and law-determining---every subsequential limit is the
Gaussian field $\mathcal X^P$ with covariance \eqref{eq:ws-cov-limit}, which proves
convergence in law in $H^{-s}_{\mathrm{loc}}(\mathbb R^d)$
\cite{billingsley_convergence_1999,mitoma_tightness_1983}.

\label{prop:duality}
Let $P_N$ be the discrete operator \eqref{eq:def:PNdisc} on $\mathbb Z^d$, and let $P_N^*$ be its
adjoint with respect to the counting measure $\sum_{x\in\mathbb Z^d}\delta_x$, i.e. the operator
determined by $\sum_x(P_Nu)_x\,v_x=\sum_x u_x\,(P_N^*v)_x$ for finitely supported $u,v$.
Since $\sum_x(\nabla_{N,j}u)_x v_x=-\sum_x u_x(\nabla_{N,j}^*v)_x$ with the backward
difference $(\nabla_{N,j}^*v)_x:=v_x-v_{x-e_j}$, one has
$P_N^*=(-1)^m\sum_{|\alpha|=m}c_\alpha\,\nabla_N^{*\alpha}$ with
$\nabla_N^{*\alpha}:=\prod_j(\nabla_{N,j}^*)^{\alpha_j}$, and discrete symbol
\begin{equation}\label{eq:adjoint-symbol}
\pi_N^*(\theta)=(-1)^m\sum_{|\alpha|=m}c_\alpha\prod_j(1-e^{-2\pi i\theta_j})^{\alpha_j},
\end{equation}
to be distinguished from the symbol
$\pi_N(\theta)=\sum_{|\alpha|=m}c_\alpha\prod_j(e^{2\pi i\theta_j}-1)^{\alpha_j}$ of $P_N$
itself.  Both satisfy, uniformly on compacts in $\theta$,
\begin{equation}\label{eq:symbol-expansion}
\pi_N(\theta/N)=N^{-m}\,p(\theta)+O(N^{-m-1}),\qquad
\pi_N^*(\theta/N)=(-1)^m N^{-m}\,p(\theta)+O(N^{-m-1}),\qquad N\to\infty.
\end{equation}
Write $\mathcal W_N(g):=N^{-d/2}\sum_{x\in\mathbb Z^d}\phi_x\,g(x/N)$ for the order-zero fluctuation
observable.  Then, for every $f\in C_c^\infty(\mathbb R^d)$ and every centered stationary field
$(\phi_x)$ with $\sum_z|\Gamma(z)|<\infty$,
\begin{equation}\label{eq:duality-L2}
\mathcal Y_N^{P}(f):=N^{m-d/2}\sum_{x\in\mathbb Z^d}(P_N\phi)_x\,f(x/N)
=(-1)^m\,\mathcal W_N(Pf)+o_{L^2}(1).
\end{equation}

Each first difference has symbol $e^{2\pi i\theta_j}-1=2\pi i\theta_j+O(\theta_j^2)$; at
$\theta/N$ this is $2\pi i\theta_j/N+O(N^{-2})$, so a product of $m$ of them is
$N^{-m}\prod_j(2\pi i\theta_j)^{\alpha_j}+O(N^{-m-1})$; summing over $\alpha$ gives the
expansion of $\pi_N$ in \eqref{eq:symbol-expansion}, and the same computation for the
backward differences $1-e^{-2\pi i\theta_j}=2\pi i\theta_j+O(\theta_j^2)$ gives that of
$\pi_N^*$.  By exact summation by parts on $\mathbb Z^d$ (no boundary terms, since $f$ has compact
support),
\[
N^{m-d/2}\sum_x(P_N\phi)_x f(x/N)=N^{m-d/2}\sum_x\phi_x\,(P_N^*f)(x/N),
\]
with
\[
(P_N^*f)(x/N)=(-1)^m\sum_{|\alpha|=m}c_\alpha(\nabla_N^{*\alpha}f)(x/N).
\]
The backward differences sample the nearby points $(x-j)/N$, $0\le j_i\le m$, so
$(P_N^*f)(x/N)$ may be nonzero slightly \emph{outside} $N\operatorname{supp} f$; fix the compact
neighbourhood $K:=\{t\in\mathbb R^d:\operatorname{dist}(t,\operatorname{supp} f)\le1\}$, which contains all
sample points for $N\ge m$.  Taylor expansion of the smooth $f$ gives
$(P_N^*f)(x/N)=(-1)^m N^{-m}(Pf)(x/N)+r_N(x)$ with $r_N(x)=O(N^{-m-1})$ uniformly in
$x$, and $r_N(x)=0$ for $x\notin NK$; hence, using $N^{m-d/2}\cdot N^{-m}=N^{-d/2}$,
\begin{equation*}
\begin{split}
\mathcal Y_N^{P}(f)& =N^{m-d/2}\sum_x\phi_x\,(P_N^*f)(x/N)\\
& =(-1)^m N^{-d/2}\sum_x\phi_x\,(Pf)(x/N)+R_N =(-1)^m\,\mathcal W_N(Pf)+R_N,
\end{split}
\end{equation*}
with $R_N=N^{m-d/2}\sum_{x\in NK\cap\mathbb Z^d}\phi_x\,r_N(x)$.  Since $r_N=O(N^{-m-1})$,
\begin{equation*}
\begin{split}
\mathbb E[R_N^2]&=N^{2m-d}\!\!\sum_{x,y\in NK\cap\mathbb Z^d}\!\!\Gamma(x-y)\,r_N(x)r_N(y)\\
&\le C\,N^{2m-d}\,N^{-2m-2}\!\!\sum_{x,y\in NK\cap\mathbb Z^d}\!\!|\Gamma(x-y)|\\
&\le C'N^{-2}\sum_z|\Gamma(z)|\to0,
\end{split}
\end{equation*}
since $\#(NK\cap\mathbb Z^d)\asymp N^d$.  This proves \eqref{eq:duality-L2}.

<!-- remark-3: body_md -->
\label{rem:variable}
The duality with a microscopic difference operator relies on $P$ having constant coefficients.  For fixed variable (macroscopic) coefficients $P=\sum_{|\alpha|=m}c_\alpha(x)\partial^\alpha$
the symbol depends on $x$ and the covariance no longer factorizes through a single Fourier multiplier; one then expects a variable-coefficient limiting covariance. A constant effective operator arises only in a genuinely different, homogenization regime, driven by rapidly oscillating or random microscopic coefficients, as in the $\nabla\varphi$ literature \cite{naddaf_homogenization_1997,armstrong_c2_2022,armstrong_quantitative_2024}.  We do not pursue either direction here.

\label{def:treegraph}
Let $(X_i)_{i\in I}$ be centered with finite moments and let $w:I\times I\to[0,\infty)$ be
symmetric.  We say $(X_i)$ satisfies the \emph{tree--graph bound with weight $w$} if for
every $r\ge2$ there is $D_r<\infty$ such that, for all $i_1,\dots,i_r$ (not necessarily
distinct),
\begin{equation}\label{eq:tree-graph}
\big|\operatorname{cum}(X_{i_1},\dots,X_{i_r})\big|\le D_r\sum_{T\in\mathcal T_r}\prod_{(u,v)\in E(T)}w(i_u,i_v),
\end{equation}
$\mathcal T_r$ being the set of spanning trees on $\{1,\dots,r\}$.

\label{prop:tree-graph}
Let $(\phi^N_x)_{x\in\mathbb T^d_N}$ be centered and stationary, satisfying \eqref{eq:tree-graph}
with translation-invariant weight $w(x,y)=w_N(x-y)$, where $w_N:\mathbb T^d_N\to[0,\infty)$ (an
$N$-periodic function), $\|w_N\|_{\ell^1}:=\sum_{z\in\mathbb T^d_N}w_N(z)$, and $D_r$ is independent of $N$.
Then, for every $r\ge2$,
\begin{equation}\label{eq:cumulant-bound}
\sum_{x_1,\dots,x_r\in\Lambda_N}\big|\kappa_r^{(N)}(x_1,\dots,x_r)\big|\le D_r\,r^{r-2}\,|\Lambda_N|\,\|w_N\|_{\ell^1}^{\,r-1}.
\end{equation}
Consequently, for every scalar constant-coefficient operator $P$ of order $m\ge0$ and all
$f_1,\dots,f_r\in C^\infty(\mathbb T^d)$,
\begin{equation}\label{eq:cum-linear}
\big|\operatorname{cum}(\mathcal X_N^{P}(f_1),\dots,\mathcal X_N^{P}(f_r))\big|
\le D_r\,r^{r-2}\,N^{-r(d/2+m)}\,|\Lambda_N|\,\|w_N\|_{\ell^1}^{\,r-1}\prod_{j=1}^r\|Pf_j\|_\infty.
\end{equation}
In particular, if $\sup_N\|w_N\|_{\ell^1}<\infty$ then Assumption~\ref{ass:cumulants} holds for
every $m\ge0$ and every $d\ge1$.

<!-- proof-9: body_md -->
Fix $r\ge2$.  By \eqref{eq:tree-graph} and translation invariance,
\[
\sum_{x_1,\dots,x_r\in\Lambda_N}\big|\kappa_r^{(N)}\big|
\le D_r\sum_{T\in\mathcal T_r}\sum_{x_1,\dots,x_r\in\Lambda_N}\prod_{(u,v)\in E(T)}w_N(x_u-x_v).
\]
Fix a tree $T$, pick a leaf $v$ with neighbour $u$; summing first over $x_v$ (the map
$x_v\mapsto x_u-x_v$ being a bijection of $\Lambda_N$ onto $\mathbb T^d_N$),
$\sum_{x_v\in\Lambda_N}w_N(x_u-x_v)=\sum_{z\in\mathbb T^d_N}w_N(z)=\|w_N\|_{\ell^1}$.  Removing $v$ leaves a
tree on $r-1$ vertices; iterating $r-1$ times,
$\sum_{x_\bullet}\prod_{(u,v)}w_N(x_u-x_v)\le|\Lambda_N|\,\|w_N\|_{\ell^1}^{r-1}$.  Cayley's formula
$|\mathcal T_r|=r^{r-2}$ gives \eqref{eq:cumulant-bound}.  Then \eqref{eq:cum-linear}
follows by multilinearity and bounding $|Pf_j|\le\|Pf_j\|_\infty$.  Finally, with
$|\Lambda_N|=N^d$ and $\sup_N\|w_N\|_{\ell^1}<\infty$,
\[
\big|\operatorname{cum}(\mathcal X_N^{P}(f_1),\dots,\mathcal X_N^{P}(f_r))\big|\lesssim N^{d-r(d/2+m)}\prod_j\|Pf_j\|_\infty\to0
\quad(r\ge3),
\]
since for $r\ge3$ and $m\ge0$ we have $r(d/2+m)\ge 3d/2>d$, so the exponent
$d-r(d/2+m)<0$.  This is Assumption~\ref{ass:cumulants}.

\label{rem:nonlinear-gaussian}
The criterion is sufficient but not necessary for Assumption~\ref{ass:cumulants}, and
non-linear functionals of Gaussian fields are the instructive counterexample.  Let
$(\phi_x)_{x\in\mathbb Z^d}$ be a centered stationary Gaussian field with $\rho(x):=\mathbb E[\phi_0\phi_x]$,
$\rho(0)=1$.  Recall that $F\in L^2(\mathcal N(0,1))$ has \emph{Hermite rank} $q$ if
$F=\sum_{\ell\ge q}c_\ell\,\mathrm{He}_\ell$ with $c_q\ne0$, $(\mathrm{He}_\ell)$ being the Hermite
polynomials, and that $F$ is a \emph{Hermite polynomial functional} if this sum is finite.  Let
$Y_x:=\phi_x^2-1=\mathrm{He}_2(\phi_x)$, the Hermite polynomial functional of rank $2$.  The triangle
diagram gives $\operatorname{cum}(Y_0,Y_x,Y_y)=8\,\rho(x)\rho(y)\rho(x-y)$, which no constant $D_3$
dominates pointwise by $\sum_T\prod_{(i,j)\in E(T)}|\rho|^2$ (with all three correlations
equal to $t\to0$ the diagram is $\sim t^3$ while each spanning tree is $\sim t^4$), so the
pointwise bound \eqref{eq:tree-graph} with weight $|\rho|^2$ fails.  The summed form
\eqref{eq:cumulant-bound} fails as well, even in the summable regime: for
$\rho(z)\asymp|z|^{-3/5}$ in $d=1$ one has $\sum_z|\rho(z)|^2<\infty$, yet the summed
triangle contribution is $\sum_{x,y,z\in\Lambda_N}|\rho(x-y)\rho(y-z)\rho(x-z)|\asymp N^{6/5}$,
which exceeds $|\Lambda_N|\,\|\rho^2\|_{\ell^1}^2\asymp N$.  Nevertheless
$N^{-3/2}\sum_{x,y,z\in\Lambda_N}|\operatorname{cum}(Y_x,Y_y,Y_z)|\asymp N^{-3/10}\to0$, and in general
the cumulant condition \eqref{eq:ws-cum} holds (with $m=0$) for $Y_x=F(\phi_x)$ with $F$ any Hermite
polynomial functional of rank $q$ (so that $Y_0$ has finite moments of all orders and the joint
cumulants are finite sums of diagrams) as soon as $\sum_z|\rho(z)|^q<\infty$: this is the connected-diagram estimate at the core
of the Breuer--Major theorem \cite{breuer_central_1983}, see \cite[Ch.~7]{nourdin_normal_2012}
and, for the field-level form on $\mathbb Z^d$ and its multivariate version,
\cite{coppini_central_2025}. Theorem~\ref{thm:wholespace} therefore applies to such
functionals---for instance to the centered squared gradient of the Gaussian free field on $\mathbb Z^d$, $d\ge3$,
\cite{cipriani_properties_2023}, the multivariate analogue of a rank-two Hermite polynomial
functional (a quadratic form in the correlated Gaussian vector $\nabla\phi_x$; see
\cite{coppini_central_2025} for the multivariate form) with absolutely summable covariance,
whose volume average converges to white noise---but through the diagram estimate rather
than through the criterion of this section; we do not develop this direction here.

\label{rem:berry-esseen}
The bound \eqref{eq:cum-linear} is quantitative.  For the order-zero observable
$\mathcal X_N^{\mathrm{Id}}(g)=N^{-d/2}\sum_{x\in\Lambda_N}\phi^N_x\,g(x/N)$ it reads

the standardized variable with parameter of order $N^{d/2}$, and the Saulis--Statulevi\v{c}ius

$\mathcal X_N^{\mathrm{Id}}(g)/\sqrt{\operatorname{Var}(\mathcal X_N^{\mathrm{Id}}(g))}$ and $\mathcal N(0,1)$ by $C\,N^{-d/(2(1+2\delta))}$; for
$\delta=0$ (e.g.\ uniformly bounded finite-range fields with the standard dependency-graph
cumulant bound) this is the optimal $N^{-d/2}=|\Lambda_N|^{-1/2}$.  The constants obtained from the Dobrushin argument are not optimized. Tracking their dependence on the cumulant order gives factorial-power bounds, which can be used to obtain quantitative estimates when the limiting variance is positive. To the authors' knowledge, a rate for the random field, uniform over a metric on $H^{-s}$, is open.

 \mathbf D:=\sum_{j=0}^{\infty}\mathbf C^j=(I-\mathbf C)^{-1}.
\]
For a bounded local function $U$ on $\mathsf S^{\mathbb Z^d}$, let
\[
 \delta_x(U):=
 \sup_{\substack{\eta,\eta'\\
                  \eta_z=\eta'_z\ \text{for }z\neq x}}
 |U(\eta)-U(\eta')|.
\]
The Dobrushin--F\"ollmer covariance estimate gives
\begin{equation}
 |\operatorname{Cov}_\mu(U,V)|
 \leq
 \frac14\sum_{x,y\in\mathbb Z^d}
 \delta_x(U)\,\mathbf D_{xy}\,\delta_y(V);
 \label{eq:follmer-covariance-new}
\end{equation}
see \cite{Follmer1982} and
\cite[Theorem~2.2]{SalasSokal1997}.

Since $\mathbf C$ has range $R$, $(\mathbf C^j)_{xy}=0$ when
$j<|x-y|_1/R$.  Moreover,
\[
 \sup_x\sum_y(\mathbf C^j)_{xy}\leq c(\mathcal G)^j.
\]
For $0<c(\mathcal G)<1$ it follows that
\begin{equation}
 \mathbf D_{xy}
 \leq
 \sum_{j\geq\lceil |x-y|_1/R\rceil}c(\mathcal G)^j
 \leq
 \frac{1}{1-c(\mathcal G)}\,e^{-\tau|x-y|_1},
 \qquad
 \tau:=\frac1R\log\frac1{c(\mathcal G)}.
 \label{eq:dobrushin-resolvent-decay-new}
\end{equation}
The case $c(\mathcal G)=0$ is independent and is immediate.  Therefore, if
$U$ and $V$ are bounded local functions supported in finite sets $A$ and
$A'$, then
\begin{equation}
 |\operatorname{Cov}_\mu(U,V)|
 \leq C_{A,A'}\|U\|_\infty\|V\|_\infty
 e^{-\tau\operatorname{dist}(A,A')},
 \label{eq:dobrushin-two-block-new}
\end{equation}
and when $|A|,|A'|\leq r$ the prefactor is bounded by a constant depending only
on $r$ and the Dobrushin matrix.

Set $\bar F_i:=F_i-\mathbb E_\mu[F_i(\sigma_0)]$; by translation invariance of $\mu$,
$\bar F_i(\sigma_{x})$ is centered for every $x$, and $\|\bar F_i\|_\infty\le2$.
Cumulants of order at least two are invariant under adding constants, so it
is enough to bound $\operatorname{cum}_\mu(\bar F_1(\sigma_{x_1}),\ldots,\bar F_r(\sigma_{x_r}))$.  Fix a
nontrivial partition $I\sqcup J=\{1,\ldots,r\}$ and put
\[
 \operatorname{dist}(x_I,x_J):=\min_{i\in I,\,j\in J}|x_i-x_j|_1.
\]
For $B\subset\{1,\ldots,r\}$, write
\[
 \mathfrak m_B:=\mathbb E_\mu\Big[\prod_{i\in B}\bar F_i(\sigma_{x_i})\Big],
 \qquad
 \widetilde{\mathfrak m}_B:=\mathfrak m_{B\cap I}\,\mathfrak m_{B\cap J},
 \qquad \mathfrak m_\varnothing:=1.
\]
If $B$ meets both $I$ and $J$, then \eqref{eq:dobrushin-two-block-new},
applied to the two products, gives
\begin{equation}
 |\mathfrak m_B-\widetilde{\mathfrak m}_B|
 \leq C_r\,e^{-\tau\operatorname{dist}(x_I,x_J)}.
 \label{eq:moment-splitting-new}
\end{equation}
If $B$ lies in one side of the partition, the difference is zero.

The moment--cumulant formula is
\begin{equation}
 \operatorname{cum}_\mu\big(\bar F_1(\sigma_{x_1}),\ldots,\bar F_r(\sigma_{x_r})\big)
 =
 \sum_{\pi\in\mathcal P_r}
 (|\pi|-1)!(-1)^{|\pi|-1}
 \prod_{B\in\pi}\mathfrak m_B.
 \label{eq:moment-cumulant-new}
\end{equation}
The same polynomial evaluated at the moments $\widetilde{\mathfrak m}_B$ vanishes,
because these are the joint moments obtained by replacing the two subvectors
indexed by $I$ and $J$ by independent copies.  Subtracting the two
polynomials, telescoping each product, and using
\eqref{eq:moment-splitting-new} and boundedness of all moments yields
\begin{equation}
 \big|\operatorname{cum}_\mu\big(\bar F_1(\sigma_{x_1}),\ldots,\bar F_r(\sigma_{x_r})\big)\big|
 \leq C_r\,e^{-\tau\operatorname{dist}(x_I,x_J)}
 \label{eq:cumulant-cut-new}
\end{equation}
for every nontrivial partition $I\sqcup J$, with a new constant $C_r$.

Let
\[
 \operatorname{mst}(x_1,\ldots,x_r)
 :=\min_{T\in\mathcal T_r}
 \sum_{\{i,j\}\in E(T)}|x_i-x_j|_1
\]
be the minimum spanning-tree length, and let $T_*$ attain this minimum.
Remove a longest edge of $T_*$, of length $L_*$.  The resulting components
define a partition $I\sqcup J$.  By the cut property of minimum spanning
trees, every edge crossing this cut has length at least $L_*$; hence
$\operatorname{dist}(x_I,x_J)\geq L_*$.  Since $T_*$ has $r-1$ edges,
\[
 L_*\geq\frac{\operatorname{mst}(x_1,\ldots,x_r)}{r-1}.
\]
Using \eqref{eq:cumulant-cut-new},
\begin{equation}
 \big|\operatorname{cum}_\mu\big(\bar F_1(\sigma_{x_1}),\ldots,\bar F_r(\sigma_{x_r})\big)\big|
 \leq
 C_r
 \exp\!\left\{-\frac{\tau}{r-1}\operatorname{mst}(x_1,\ldots,x_r)\right\}.
 \label{eq:dobrushin-MST-exponential-new}
\end{equation}

The decay rate in \eqref{eq:dobrushin-MST-exponential-new} depends on $r$.
This does not prevent a tree--graph estimate with one fixed summable weight.
Indeed, let
\[
 C'_r:=
 \sup_{t\geq0}e^{-\tau t/(r-1)}(1+t)^{d+1}<\infty.
\]
Applying
$e^{-\tau t/(r-1)}\leq C'_r(1+t)^{-(d+1)}$ to every edge of $T_*$ gives
\[
 \exp\!\left\{-\frac{\tau}{r-1}\operatorname{mst}(x_1,\ldots,x_r)\right\}
 \leq
 (C'_r)^{r-1}
 \prod_{\{i,j\}\in E(T_*)}w(x_i,x_j).
\]
The last product is bounded by the sum over all spanning trees.  This proves
\eqref{eq:dobrushin-tree-graph-bound}, with
$D_r:=C_r\,(C'_r)^{r-1}$.  Equation
\eqref{eq:dobrushin-volume-cumulant-bound} is then Proposition~\ref{prop:tree-graph}.  If
$|\Lambda_N|=O(N^d)$, its right-hand side is $O_r(N^d)$, and therefore
\[
 N^{-r(d/2+m)}O_r(N^d)
 =O_r\!\left(N^{d-r(d/2+m)}\right)\longrightarrow0,
 \qquad r\geq3,
\]
which is Assumption~\ref{ass:cumulants}.  The same argument applies to the sums over
$NK\cap\mathbb Z^d$, which gives \eqref{eq:ws-cum}.

\label{rem:dobrushin-strength-converse}
There is no converse from the tree--graph bound, or from Assumption~\ref{ass:cumulants},
to Dobrushin uniqueness.  These cumulant properties concern one selected
Gibbs state, whereas the Dobrushin condition is a uniform contraction
property of the specification and implies uniqueness.  At zero field and
sufficiently low Ising temperature, the plus and minus states are distinct,
so Dobrushin uniqueness fails; nevertheless, the low-temperature plus state
satisfies an exponential weighted-dependency-graph bound in the
contour-expansion regime \cite{dousse_weighted_2019}.

\label{cor:dobrushin-fluctuation-fields}
Assume \eqref{eq:dobrushin-condition-new}.  Let $F^1,\ldots,F^n:\mathsf S\to\mathbb R$ be bounded
and put
\[
 \phi_x^a:=F^a(\sigma_x)-\mathbb E_\mu[F^a(\sigma_x)],
 \qquad a=1,\ldots,n.
\]
Their covariances are exponentially summable by \eqref{eq:dobrushin-two-block-new}.  Hence
the susceptibility matrix
\begin{equation}
 \boldsymbol\chi_{ab}
 :=\sum_{z\in\mathbb Z^d}
 \operatorname{Cov}_\mu(\phi_0^a,\phi_z^b)
 \label{eq:dobrushin-susceptibility-matrix-new}
\end{equation}
is finite and positive semidefinite.  For
$f\in C_c^\infty(\mathbb R^d)$, the joint field
\[
 \mathcal W_N^a(f):=N^{-d/2}\sum_{x\in\mathbb Z^d}
 \phi_x^a f(x/N)
\]
converges, in the sense of Theorem~\ref{thm:wholespace}, to an
$\mathbb R^n$-valued white noise $\mathcal W$ with
\begin{equation}
 \mathbb E[\mathcal W^a(f)\mathcal W^b(g)]
 =\boldsymbol\chi_{ab}\langle f,g\rangle_{L^2(\mathbb R^d)}.
 \label{eq:dobrushin-vector-white-noise-new}
\end{equation}
The limit may be degenerate when $\boldsymbol\chi$ has a nontrivial kernel.

Let $P$ be a homogeneous constant-coefficient operator of order $m$, and let
$P_N$ be the associated lattice difference operator \eqref{eq:def:PNdisc}.  Then, jointly in
$a=1,\ldots,n$,
\begin{equation}
 N^{m-d/2}\sum_{x\in\mathbb Z^d}
 (P_N\phi^a)_x f(x/N)
 \xrightarrow[]{\ d\ }
 (-1)^m\mathcal W^a(Pf),
 \label{eq:dobrushin-differentiated-new}
\end{equation}
with limiting covariance
$\boldsymbol\chi_{ab}\langle Pf,Pg\rangle_{L^2}$.

<!-- proposition-6: body_md -->
\label{prop:trichotomy}
Let $(\phi_x)_{x\in\mathbb Z^d}$ be centered and stationary, with finite moments of all orders and
spectral density
\begin{equation}\label{eq:spectral-singularity}
\widehat\Gamma(\theta)=c_0\,|\theta|^{-\nu}\,(1+o(1)),\qquad\theta\to0,
\end{equation}
for some $\nu\in[0,d)$ and $c_0>0$, bounded outside every neighbourhood of the origin.  Let
$P$ have order $m$ with symbol $p$ and let $\mathcal X_N^{P}$ be the whole-space observable
\eqref{eq:def:PN-ws}.  Then, according to how $m$ compares with $\nu$:
\begin{enumerate}[label=\textup{(\roman*)}]

\item \textup{(over-differentiation, $\nu<2m$)} $\operatorname{Var}(\mathcal X_N^{P}(f))\to0$ for every $f$, so
$\mathcal X_N^{P}\to0$ in probability;
\item \textup{(exact match, $\nu=2m$)} the covariance converges,
\begin{equation}\label{eq:spectral-BM-cov}
\operatorname{Cov}\big(\mathcal X_N^{P}(f),\mathcal X_N^{P}(g)\big)\longrightarrow
c_0\int_{\mathbb R^d}\overline{\widehat f(\xi)}\,\widehat g(\xi)\,|p(\xi)|^2|\xi|^{-2m}\,\mathrm d\xi,
\end{equation}
a bounded multiplier homogeneous of degree zero.

\item \textup{(under-differentiation, $\nu>2m$)} $\operatorname{Var}(\mathcal X_N^{P}(f))\asymp N^{\nu-2m}\to\infty$ whenever $\int|p|^2|\widehat f|^2|\xi|^{-\nu}\,\mathrm d\xi\ne0$, and under the self-similar normalization $N^{-(d+\nu)/2}$  \cite{dobrushin_non-central_1979}, the covariance converges to
\[
c_0\int\overline{\widehat f}\,\widehat g\,|p(\xi)|^2|\xi|^{-\nu}\,\mathrm d\xi.
\]
For Gaussian $\phi$ the limit law follows: it is $P$ applied to a fractional Gaussian field
$\mathrm{FGF}_{\nu/2}(\mathbb R^d)$ in the parametrization of \cite{lodhia_fractional_2016}.
\end{enumerate}

By \eqref{eq:spectral-singularity}, $N^{-2m}\widehat\Gamma(\xi/N)=c_0\,N^{\nu-2m}|\xi|^{-\nu}(1+o(1))$
for fixed $\xi\ne0$, which gives the pointwise limit of $G_N$ in each of the three cases.
All three cases rest on the same domination: for $\varsigma\in\{\nu,2m\}$, on the box $Q_N$
one has $N^{-\varsigma}\widehat\Gamma(\xi/N)\le C|\xi|^{-\nu}\,N^{\nu-\varsigma}$ for
$|\xi|\le\delta N$ by \eqref{eq:spectral-singularity}, while for $|\xi|>\delta N$ the assumed
boundedness of $\widehat\Gamma$ outside the $\delta$-ball gives
$N^{-\varsigma}\widehat\Gamma(\xi/N)\le C_\delta N^{-\varsigma}\le C'_\delta|\xi|^{-\varsigma}$,
since $|\xi|\le N\sqrt d/2$ on $Q_N$.

In case (ii) ($\varsigma=2m=\nu<d$), we have that
$G_N\le\Phi:=C|\xi|^{-2m}$, which is locally integrable and bounded for $|\xi|\ge1$, so
Assumption~\ref{ass:wholespace}\,\ref{it:ws-scaling}--\ref{it:ws-domination} hold with
$M=0$ and $\gamma=c_0|\cdot|^{-2m}$: the covariance convergence
\eqref{eq:spectral-BM-cov} is Step~1 of the proof of Theorem~\ref{thm:wholespace}, and
under the cumulant condition the full theorem applies.

In case (i) ($\varsigma=2m>\nu$), $G_N\to0$ pointwise with the same locally integrable majorant $C|\xi|^{-\nu}$, so $\operatorname{Var}(\mathcal X_N^{P}(f))\to0$ and Chebyshev gives convergence to $0$ in probability.

In case (iii), the self-similarly rescaled weight $N^{-\nu}\widehat\Gamma(\xi/N)$ satisfies the hypotheses with $\gamma=c_0|\cdot|^{-\nu}$ and majorant $C|\xi|^{-\nu}$ ($\nu<d$), so Step~1 of the proof of Theorem~\ref{thm:wholespace} gives the stated covariance limit,
and hence $\operatorname{Var}(\mathcal X_N^{P}(f))=N^{\nu-2m}\big(c_0\int|p|^2|\widehat f|^2|\xi|^{-\nu}\,\mathrm d\xi+o(1)\big)$.
For Gaussian $\phi$ the variables $\mathcal X_N^{P}(f)$ are centered Gaussian, so diverging variance
forbids tightness of any subsequence, and the self-similar covariance limit determines
the Gaussian law stated.

<!-- remark-8: body_md -->
\label{rem:noncentral}
The proposition is a statement about the covariance and not about Gaussianity. For instance, let $\phi$ be Gaussian and $Y_x:=F(\phi_x)$ with $F$ a Hermite polynomial functional of rank $q\ge2$ (Remark~\ref{rem:nonlinear-gaussian}), in the long-memory regime where the spectral density of $Y$ has the singularity \eqref{eq:spectral-singularity} with $\nu\in(0,d)$.  On the line $\nu=2m$ the normalization $N^{-d/2-m}$ coincides with the non-central normalization of Dobrushin--Major and Taqqu \cite{dobrushin_non-central_1979,taqqu_convergence_1979}. For instance, applying $P$ changes the test function and the spectral multiplier but not the chaos order: the limit of the normalized observable, when nonzero, lies in the $q$th Wiener chaos and is not Gaussian
\cite{janson_gaussian_1997}.
                                                                                                                                                                                                                                                            
The conditional Gaussian statement in (ii) therefore has content where the cumulant condition is checkable, e.g.\ for Gaussian fields ($q=1$); for
$q\ge2$ the correct conclusion, on the line $\nu=2m$ as well as below it, is the non-central Hermite--Rosenblatt limit, outside the present framework.  Slowly varying corrections to the
correlation and the logarithmic boundary case (susceptibility diverging like $\log N$, normalization $(N^d\log N)^{-1/2}$) are not treated
here but represent an interesting direction of future research.
