<!-- theorem-1: body_md -->
\label{main theorem}
Let $\gamma >\frac{1}{2}$ and $\omega_0 \in \dot{H}^1 \cap \dot{H}^{-1}(\mathbb{R}^2)$. Then there exists a unique solution to \eqref{logVorticityEquation} satisfying
$$
\omega \in C^0([0,\infty); \dot{H}^{-1}\cap \dot{H}^1(\mathbb{R}^2)).
$$
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
We remark, as the authors did in \cite{DL15}, that the role of $\omega_0 \in \dot{H}^{-1}(\mathbb{R}^2)$ is not artificial. This is since the local well-posedness argument in \cite{DL15} needs $u \in L^{\infty}(\mathbb{R}^2)$ and $\omega_0 \in H^1(\mathbb{R}^2)$ is not sufficient to handle the low frequencies arising from the Biot-Savart law (however, note that $\left\lVert\omega_0\right\rVert_{L^2}^2 \leq \left\lVert\omega_0\right\rVert_{\dot{H}^{-1}}\left\lVert\omega_0\right\rVert_{\dot{H}^1}$).
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
(Bernstein's Lemma) Let $(r_1,r_2)$ be a pair of strictly positive real numbers such that $r_1<r_2$ and let $\lambda$ denote a positive real number. Then there exists a constant $C>0$, such that, for every integer $k$ with $k \geq 0$, every pair of real numbers $(a,b)$ satisfying $b \geq a \geq 1$, and every function $u \in L^a(\mathbb{R}^d)$,
\begin{enumerate}

\item 
\label{Bernstein in Ball}
If $\operatorname{supp}{\hat{u}} \subset B_{r_1\lambda}(0)$ then
\begin{equation*}
 \sup_{\mid\!\alpha\!\mid = k} \left\lVert\partial^{\alpha}u\right\rVert_{L^b} \leq C^k\lambda^{k+d(\frac{1}{a}-\frac{1}{b})}\left\lVertu\right\rVert_{L^a}.
\end{equation*}\\
\item 
\label{Bernstein in Annulus}
Furthermore, if $\operatorname{supp}{\hat{u}} \subset \mathcal{C}_{r_1\lambda, r_2\lambda}(0)$ then
\begin{equation*}
 C^{-k}\lambda^k\left\lVertu\right\rVert_{L^a} \leq  \sup_{\mid\!\alpha\!\mid = k} \left\lVert\partial^{\alpha}u\right\rVert_{L^a} \leq C^k\lambda^k\left\lVertu\right\rVert_{L^a}.
\end{equation*}
\end{enumerate}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{generalized bernstein}
Let $m \in C^{d+1}(\mathbb{R}^d \setminus \{0\})$ be such that for any dyadic $N>0$, there exists a constant $\tilde{m}(N)$ so that 
$$
\sup_{\frac{N}{8} \leq |\xi| \leq 8N} |\partial_{\xi}^{\alpha}m(\xi)| \leq C_d \frac{\tilde{m}(N)}{N^{|\alpha|}}, \quad \text{ for all } |\alpha| \leq d+1.
$$
Let $T_m$ be the associated operator defined by the Fourier multiplier
$$
\mathcal{F}(T_mf)(\xi) = m(\xi)\mathcal{F}(f)(\xi).
$$
Then for any integer $j \geq 0$ and $1 \leq p \leq \infty$ we have 
$$
\left\lVertT_m \Delta_jf\right\rVert_{L^p} \lesssim_{d} \tilde{m}(2^j)\left\lVert\Delta_jf\right\rVert_{L^p}.
$$
<!-- end lemma-2 -->
<!-- definition-1: body_md -->
\label{besov space}
Let $p,q \in [1,\infty]$ and $ s \in \mathbb{R}$. We define the Besov space $B^{s}_{p,q}(\mathbb{R}^d)$ as the set of $f \in \mathcal{S}'(\mathbb{R}^d)$ such that the following norm (with the obvious modification when $q=\infty$) is finite
\begin{equation}
\label{inhomogeneous Besov space}
\left\lVertf\right\rVert_{B^{s}_{p,q}} = \left(\sum_{j \geq -1} 2^{jsq}\left\lVert\Delta_jf\right\rVert_{L^p}^q\right)^{1/q}.\\
\end{equation}
<!-- end definition-1 -->
<!-- lemma-3: body_md -->
\label{velocity gradient 1}
Let $\gamma>\frac{1}{2}$ and $f \in L^2 \cap B^0_{\infty,2}(\mathbb{R}^2)$. 
Then
$$
\left\lVert\nabla \nabla^{\perp}(-\Delta)^{-1}T_{\gamma}f\right\rVert_{L^{\infty}} \leq C\left\lVertf\right\rVert_{L^2} + C_{\gamma}\left\lVertf\right\rVert_{B^0_{\infty,2}}.
$$
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
Using a Littlewood-Paley decomposition, we write 
\begin{equation}
\label{vg1 e1}
\left\lVert\nabla \nabla^{\perp}(-\Delta)^{-1}T_{\gamma}f\right\rVert_{L^{\infty}} \leq \left\lVert\Delta_{-1}\nabla \nabla^{\perp}(-\Delta)^{-1}T_{\gamma}f\right\rVert_{L^{\infty}} + \sum_{j \geq 0}\left\lVert\Delta_j\nabla \nabla^{\perp}(-\Delta)^{-1}T_{\gamma}f\right\rVert_{L^{\infty}}. 
\end{equation}
To estimate the low frequencies, we see that 
\begin{equation}
\label{vg1 e2}
\left\lVert\Delta_{-1}\nabla \nabla^{\perp}(-\Delta)^{-1}T_{\gamma}f\right\rVert_{L^{\infty}} \leq \int_{\mathbb{R}^2} \left|\hat{\chi}(\xi) \ln^{-\gamma}(e+|\xi|) \hat{\omega}(\xi)\right|\,d\xi \leq C\left\lVert\hat{\chi}\right\rVert_{L^2}\left\lVert\hat{f}\right\rVert_{L^2} \leq C\left\lVertf\right\rVert_{L^2},
\end{equation}
where in the second inequality we used that $\ln^{-\gamma}(e+|\xi|)$ is bounded near the origin. To estimate the high frequencies, we apply Lemma \ref{generalized bernstein} to obtain
\begin{equation}
\label{vg1 e3}
\sum_{j \geq 0} \left\lVert\Delta_j \nabla \nabla^{\perp}(-\Delta)^{-1}T_{\gamma}f\right\rVert_{L^{\infty}} \leq C\sum_{j \geq 0}(1+j)^{-\gamma}\left\lVert\Delta_jf\right\rVert_{L^\infty} \leq C\left(\sum_{j \geq 0} (1+j)^{-2\gamma}\right)^{\frac{1}{2}} \left\lVertf\right\rVert_{B^0_{\infty,2}}.
\end{equation}
The last sum is finite when $\gamma >\frac{1}{2}$. Combining \eqref{vg1 e1}, \eqref{vg1 e2}, and \eqref{vg1 e3} yields the result.
<!-- end proof-1 -->
<!-- theorem-2: body_md -->
\label{Key Besov estimate} Let $r \in [1,\infty]$, $a_0 \in L^2 \cap B^0_{\infty,r}(\mathbb{R}^d)$, and $a$ be a solution to
\begin{equation}
\label{transport}
\tag{$\text{Tr}$}
\begin{cases}
\partial_ta + (v\cdot \nabla)a= 0,\\
\text{div }v=0,\\
a(0,x)=a_0(x) ,
\end{cases}
\end{equation}
such that 
$
a \in C^0([0,T]; L^2 \cap B^0_{\infty,r}(\mathbb{R}^d))$ and $ 
 v \in L^1([0,T]; W^{1,\infty}(\mathbb{R}^d)).$
Then for any $t \leq T$ one has
\begin{equation}
\label{key besov estimate eqn}
\left\lVerta(t)\right\rVert_{B^0_{\infty,r}} \leq C\left\lVerta_0\right\rVert_{B^0_{\infty,r}} \left(1+\int_0^t \left\lVert\nabla v(\tau)\right\rVert_{L^{\infty}}\,d\tau\right).
\end{equation}
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
For a fixed solution $a\in C^0([0,T]; L^2 \cap B^0_{\infty,r}(\mathbb{R}^d))$, we consider the system
\begin{equation}
\label{Transport q}
\tag{$\text{Tr}_q$}
\begin{cases}
\partial_ta_q + (v\cdot \nabla)a_q= 0,\\
\text{div }v=0,\\
a_q(0,x)=\Delta_qa_0(x).
\end{cases}
\end{equation}
Using the method of characteristics, we see that 
$
a_q(t,x)=\Delta_qa_0(\phi^{-1}_t(x)),
$
where $\phi$ satisfies the ODE
$$
\frac{d}{dt} \phi_t(x) = v(t,\phi_t(x)), \quad \phi(0,x)=x.
$$
Furthermore, since $\text{div }v=0$, we see that $\phi_t$ is measure preserving and 
\begin{equation}
    \label{TRq L2 convergence}
    \begin{split}
    \left\lVerta(t)-\sum_{q=-1}^n a_q(t)\right\rVert_{L^2} &= \left\lVert\left(a_0-  \sum_{q=-1}^n \Delta_qa_0\right)(\phi^{-1}_t(x))\right\rVert_{L^2}\\
    &= \left\lVerta_0 - \sum_{q=-1}^n\Delta_qa_0\right\rVert_{L^2} \to 0 \quad \text{as }n \to \infty.
    \end{split}
\end{equation}
Furthermore, since $\text{div }v=0$, solutions of \eqref{Transport q} satisfy a maximum principle and
\begin{equation}
\label{TRq eq1}
\left\lVert\Delta_j a_q(t)\right\rVert_{L^\infty} \leq C\left\lVerta_q(t)\right\rVert_{L^{\infty}} \leq C\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{equation}
Next, we show 
\begin{equation}
\label{TRq eq 2}
\left\lVert\Delta_ja_q(t)\right\rVert_{L^\infty} \leq Ce^{CV(t)}2^{-|j-q|}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}},
\end{equation}
where 
$$
V(t) = \int_0^t \left\lVert\nabla v(\tau)\right\rVert_{L^{\infty}}\,d\tau.
$$
Since $v \in L^1([0,T];W^{1,\infty}(\mathbb{R}^d))$, from Gr\"{o}nwall's Lemma, we see that 
\begin{equation}
\label{flow bounds}
\left\lVertD\phi_t\right\rVert_{L^{\infty}} + \left\lVertD\phi_{t}^{-1}\right\rVert_{L^{\infty}} \leq Ce^{CV(t)},
\end{equation}
where $Df(x)=\left(\frac{\partial f^i}{\partial_{x_k}}(x)\right)_{i,k=1}^d$ is the Jacobian matrix. To establish \eqref{TRq eq 2} we consider three cases, when $j>q, j<q$, and $j=q$. \\
\\
\textbf{Case 1 ($j>q$):} Since $q \geq -1$, we see that $j \geq 0$ and 
$$
\int_{\mathbb{R}^d} \varphi_j(x)\,dx = \int_{\mathbb{R}^d} \varphi_j(x)e^{2\pi i x \cdot 0}\,dx = \hat{\varphi_j}(0)=0.
$$
From the mean value theorem 
\begin{equation}
\label{case 1 eq 1}
\begin{split}
|\Delta_ja_q(t,x)| &= \left| \int_{\mathbb{R}^d} \varphi_j(y) (a_q(t,x-y)-a_q(t,x))\,dy\right|\\
& \leq \left\lVert\nabla a_q(t)\right\rVert_{L^{\infty}} \int_{\mathbb{R}^d} |\varphi_j(y)y|\,dy\\
&\leq C2^{-j}\left\lVert\nabla a_q(t)\right\rVert_{L^{\infty}}.
\end{split}
\end{equation}
From the transport structure of \eqref{Transport q} and the chain rule, we see that 
$$
\nabla a_q(t,x)=\nabla \left((\Delta_qa_0)(\phi_t^{-1}(x))\right) =(D\phi_{t}^{-1}(x))^T(\nabla\Delta_qa_0)(\phi_t^{-1}(x)).
$$
Using \eqref{flow bounds} we see that 
\begin{equation}
\label{case 1 eq 2}
\left\lVert\nabla a_q(t)\right\rVert_{L^{\infty}} \leq \left\lVertD\phi_t^{-1}\right\rVert_{L^{\infty}}\left\lVert\nabla \Delta_qa_0(\phi_t^{-1})\right\rVert_{L^{\infty}}\leq Ce^{CV(t)}2^q\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{equation}
Combining \eqref{case 1 eq 1} and \eqref{case 1 eq 2} yields
\begin{equation}
\label{case 1}
\left\lVert\Delta_j a_q(t)\right\rVert_{L^{\infty}} \leq Ce^{CV(t)}2^{q-j}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{equation}
\textbf{Case 2 ($q>j$):} Note that 
\begin{equation}
\label{case 2 eq 0}
\Delta_qa_0(x)=\text{div}(-\nabla(-\Delta)^{-1}\Delta_qa_0)(x).
\end{equation}
We first consider when $j \geq 0$. Then, since the flow is measure-preserving, by a change of variables and \eqref{case 2 eq 0}
\begin{equation}
\label{case 2 eq 1}
\begin{split}
|\Delta_ja_q(t,x)|&= \left|\int_{\mathbb{R}^d} \varphi_j(x-y)(\Delta_qa_0)(\phi_t^{-1}(y))\,dy\right|\\
&=\left|\int_{\mathbb{R}^d}\varphi_j(x-\phi_t(z))\text{div}(-\nabla(-\Delta)^{-1}\Delta_qa_0)(z)\,dz\right|\\
&= \left|\int_{\mathbb{R}^d}\nabla_z(\varphi_j(x-\phi_t(z)))\cdot(\nabla(-\Delta)^{-1}\Delta_qa_0)(z)\,dz\right|\\
&\leq C2^{-q}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}} \int_{\mathbb{R}^d} |\nabla_z(\varphi_j(x-\phi_t(z)))|\,dz\\
&\leq C2^{-q}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}} \left\lVertD\phi_t\right\rVert_{L^{\infty}} \int_{\mathbb{R}^d} |(\nabla\varphi_j)(x-\phi_t(z))|\,dz\\
&\leq Ce^{CV(t)}2^{j-q}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{split}
\end{equation}
A similar argument for $j=-1$ gives
\begin{equation}
\label{case 2 eq 2}
\begin{split}
\left\lVert\Delta_{-1}a_q(t)\right\rVert_{L^{\infty}} & \leq 2^{-q}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}\left\lVertD\phi_t\right\rVert_{L^{\infty}}\left\lVert\nabla \chi\right\rVert_{L^1}\\
&\leq Ce^{CV(t)}2^{-q}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}\\
&\leq Ce^{CV(t)}2^{-|1-q|}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{split}
\end{equation}
\textbf{Case 3 ($j=q$):} For the final case, we simply use that 
\begin{equation}
\label{case 3}
\left\lVert\Delta_ja_q(t)\right\rVert_{L^{\infty}} \leq C\left\lVerta_q(t)\right\rVert_{L^{\infty}} \leq C\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{equation}
Combining \eqref{case 1},\eqref{case 2 eq 1}, \eqref{case 2 eq 2}, and \eqref{case 3} yield \eqref{TRq eq 2}. 
\\
\\
Now, from \eqref{TRq eq1} and \eqref{TRq eq 2} we see that 
\begin{equation}
\label{TRq eq3}
\left\lVert\Delta_ja_q(t)\right\rVert_{L^{\infty}} \leq C\min\left\{1,e^{CV(t)}2^{-|j-q|}\right\}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{equation}
Since $a(t)=\sum_{q \geq -1} a_q(t)$ in $L^2(\mathbb{R}^d)$, from Bernstein's Lemma one has $\Delta_ja(t)=\Delta_j\sum_{q \geq -1} a_q(t)$ in $L^{\infty}(\mathbb{R}^d)$. It follows that 
\begin{equation}
\label{TRq eq 4}
\left\lVert\Delta_ja(t)\right\rVert_{L^{\infty}} \leq C\sum_{q \geq -1} \min\left\{1,e^{CV(t)}2^{-|j-q|}\right\}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}.
\end{equation}
Computing the $B^0_{\infty,r}(\mathbb{R}^d)$-norm (with the obvious modification when $r=\infty$), from \eqref{TRq eq 4} we find that 
\begin{equation}
\label{TR eq1}
\begin{split}
\left\lVerta(t)\right\rVert_{B^0_{\infty,r}} &= \left(\sum_{j \geq -1} \left\lVert\Delta_ja(t)\right\rVert_{L^{\infty}}^r\right)^{\frac{1}{r}}\\
&\leq C\left(\sum_{j \geq -1}\left(\sum_{q \geq -1} \min\left\{1,e^{CV(t)}2^{-|j-q|}\right\}\left\lVert\Delta_qa_0\right\rVert_{L^{\infty}}\right)^r\right)^{\frac{1}{r}}\\
&\leq C\left\lVerta_0\right\rVert_{B^0_{\infty,r}} \sum_{q \in \mathbb{Z}} \min\left\{1,e^{CV(t)}2^{-|q|}\right\},
\end{split}
\end{equation}
where in the last inequality we used Young's convolution inequality for series. We now split the sum when $e^{CV(t)}2^{-|q|}= 1$ and estimate the two terms separately. Hence, we let $N=\left\lceil \frac{CV(t)}{\ln(2)}\right\rceil$ and observe that
\begin{equation}
\label{Tr eq2}
\begin{split}
\sum_{q \in \mathbb{Z}} \min\left\{1,e^{CV(t)}2^{-|q|}\right\}&= \sum_{|q| \leq N}\min\left\{1,e^{CV(t)}2^{-|q|}\right\} +   \sum_{|q| > N}\min\left\{1,e^{CV(t)}2^{-|q|}\right\}\\
&\leq \sum_{|q| \leq N} 1 +   e^{CV(t)}\sum_{|q| > N} 2^{-|q|}\\
&= 2N +1 +Ce^{CV(t)}2^{-N}\\
&\leq C(1+V(t)),
\end{split}
\end{equation}
which concludes the proof.
