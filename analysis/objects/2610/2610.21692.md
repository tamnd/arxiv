\label{lem:gate-properties}
The functions $\psi$ and $\phi$ defined in equation 
equation~(\ref{eq:gate-functions}) satisfy the following
properties. 
\begin{enumerate}[label=(\alph*),leftmargin=0.65cm]
    \item The function $\psi$ satisfies $\psi(t)=0$ for all $t \le 1/2$, and 
    $\psi(t)=1$ for all $t\ge1$.
    \item The function $\psi$ satisfies
    $0\le\psi(t)\le1$, $0\le\psi'(t)\le4$, and $|\psi''(t)|\le24$ for all $t\in{\mathbb{R}}$.
    \item The function $\phi$ satisfies
    $0<\phi(t)<2$, $\phi'(t)=\operatorname{sech}^2(t)$, and    $|\phi''(t)|\le2\phi'(t)$ for all $t\in{\mathbb{R}}$.
    \item The derivative $\phi'$ is even and holds
    $\phi'(t)>2/5$ for all $t\in[-1, 1]$.
\end{enumerate}

<!-- definition-1: body_md -->
\label{def:probabilistic-zero-chain}
A stochastic gradient ${\bf{g}}:{\mathbb{R}}^T\times{\mathcal{Z}}\to{\mathbb{R}}^T$ is a probability-$\theta$ zero-chain with $\theta\in[0,1]$ if it holds
\begin{equation}
\mathbb P
\Big(\forall\,{\bf{y}}\in{\mathbb{R}}^T:
    \operatorname{prog}_0
    \big({\bf{g}}({\bf{y}};Z)\big) \le \operatorname{prog}_{1/4}({\bf{y}}),
    {\bf{g}}({\bf{y}};Z) ={\bf{g}}\big(\Pi_{\operatorname{prog}_{1/4}({\bf{y}})}({\bf{y}}); Z\big)
\Big) \ge 1-\theta,
\label{eq:prob-zero-chain-no-progress}
\end{equation}
and
\begin{equation}
\!\!\!\mathbb P_Z
\Big(\forall\,{\bf{y}}\in{\mathbb{R}}^T\!:
    \operatorname{prog}_0
    \big({\bf{g}}({\bf{y}};Z)\big) \le 1+\operatorname{prog}_{1/4}({\bf{y}}),
    {\bf{g}}({\bf{y}};Z) = {\bf{g}}\big(\Pi_{1+\operatorname{prog}_{1/4}({\bf{y}})}({\bf{y}}); Z\big)
\Big)=1.
\label{eq:prob-zero-chain-one-progress}
\end{equation}
Here, we denote $\Pi_s({\bf{y}})=[y_1,\dots,y_s,0,\dots,0]^\top$ for given ${\bf{y}}=[y_1,\dots,y_d]^\top\in{\mathbb{R}}^d$, where we let $\Pi_s({\bf{y}})={\bf{y}}$ when $s>T$ for convention.

For any given $p\in(1,2]$,
there exists numerical constant $c>0$ such that for all $L_0,L_1,\Delta,\sigma>0$, and $\epsilon\leq  c  \min\{\sqrt{L_0\Delta}, L_1\Delta\}$, it holds
\begin{equation}
   m_{\epsilon}^{\mathrm{rand}}(L_0,L_1,\Delta,\sigma,p)
\ge \Omega\left(\bigg(\frac{\sigma^{\frac{p}{p-1}}}{\epsilon^{\frac{p}{p-1}}}+1\bigg)\bigg(\frac{L_0\Delta}{\epsilon^2}+\frac{L_1\Delta}{\epsilon}\bigg)\right).
\label{eq:joint-lower-bound}
\end{equation}

\label{thm:nsgd}
We run Algorithm \ref{alg:sn-mb-sgd} to solve problem (\ref{eq:problem}) with  $f\in{\mathcal{F}}(L_0, L_1, \Delta)$ by using stochastic first-order oracle ${\bf{g}}:{\mathbb{R}}^d\times{\mathcal{Z}}\to{\mathbb{R}}^d$ such that

\qquad\text{and} \qquad
{\mathbb{E}}_Z\left[\left\| {\bf{g}}({\bf{x}};Z)-\nabla f({\bf{x}}) \right\|^p\right] \leq \sigma^p
\end{align*}

b=\bigg\lceil
\bigg(\frac{32\sqrt2\,\sigma}{\epsilon}\bigg)^{\frac{p}{p-1}}+1
\bigg\rceil,
\quad \text{and}\quad
T=\left\lceil\frac{4\Delta}{\eta\epsilon}\right\rceil.
\end{align}
the output satisfies ${\mathbb{E}}[\left\| \nabla f(\hat{\bf{x}}) \right\|]\leq\epsilon$.

Recall that the functions $\psi$ and $\phi$ are defined as
\begin{equation}
    \psi(t):=S(2t-1) \qquad \text{and}   \qquad   \phi(t):=1+\tanh(t).
    \label{eq:gate-functions1}
\end{equation}
respectively, where
\begin{equation}
    S(u)=
    \begin{cases}
        0, & u\le 0,\\
        10u^3-15u^4+6u^5, & u\in(0,1),\\
        1, & u\ge 1.
    \end{cases}
    \label{eq:gate-S1}
\end{equation}
We verify the first-order and second-order derivatives of $S$ as
\begin{align}\label{eq:S-prime}
S'(u) = \begin{cases}
30u^2(u-1)^2,  & \text{if}~u\in(0,1), \\
0, & \text{otherwise},
\end{cases}
\end{align}
and
\begin{align}\label{eq:S-second}
    S''(u) =\begin{cases}
    60u(u-1)(2u-1),  & \text{if}~u\in(0,1) \\
    0, & \text{otherwise}.
    \end{cases}
\end{align}

We now prove each statement of the lemma.

\paragraph{The proof of statement (a).} The definitions of $\psi$ and $S$ in equations (\ref{eq:gate-S1}) and (\ref{eq:S-second}) directly shows that~$\psi(t)=0$ for all $t \le 1/2$, and $\psi(t)=1$ for all $t\ge1$.


\paragraph{The proof of statement (b).}
We first consider the value of $\psi(t)$. 
By equation equation~(\ref{eq:S-prime}), we know that $S'(t)\geq 0$ on ${\mathbb{R}}$, which implies the function $S$ is non-decreasing.
Combining the facts $S(0)=0$ and $S(1)=1$, we obtain
$0\le S(u)\le1$ for all $u\in[0,1]$.
Combing the definitions $S(u)=0$ for all~$u\leq 0$ and $S(u)=1$ for all $u\geq 1$, we know that $S(u)\in[0,1]$ on ${\mathbb{R}}$.
Hence, the definition of~$\psi(t)=S(2t-1)$ implies
\begin{equation*}                   
    0\le\psi(t)\le1.
\end{equation*}

We then consider the value of $\psi'(t)$. By the chain rule, we can write 

\begin{align}\label{eq:psi-dev}
\psi'(t) = \begin{cases}
 60(2t-1)^2(2t-2)^2,       &  t\in[ 1/2,1], \\
 0 & \text{otherwise}.
\end{cases}
\end{align}
Clearly, it holds $\psi'(t)\ge0$. 
For $t\in[1/2,1]$, we let
\[
u:=2t-1\in[0,1],
\]
which leads to
\begin{align*}
\psi'(t)= 60(2t-1)^2(2t-2)^2 = 60u^2(u-1)^2 .
\end{align*}
Since for all $u\in[0,1]$, it holds 
\[
    u(u-1) = \left(u-\frac12\right)^2 - \frac14
    \in \left[-\frac14, 0\right],
\]
which implies
\[
    \psi'(t)
    \le
    60\cdot\frac1{16}
    =
    \frac{15}{4}
    <4.
\]
Recall that $\psi'(t)=0$ for all $t\in[0, 1/2]$, 
we obtain $0\le\psi'(t)\le4$ for all $t\in{\mathbb{R}}$.

We now consider the value of $\psi''(t)$. By the chain rule, we can write 
\[
    \psi''(t)=4S''(2t-1).
\]
Equation (\ref{eq:S-second}) indicates $\psi''(t)=0$ for all $t\not\in(1/2,1)$.
Hence, we focus on the case of~$t\in(1/2,1)$, that is $u=2t-1\in(0,1)$. 
We write
\begin{equation*}
    \psi''(t)=240u(1-u)(1-2u).
                                     
\end{equation*}
and define
\[
    g(u):=u(1-u)(1-2u)
    =
    u-3u^2+2u^3.
\]
Then we have
\[
    g'(u)=1-6u+6u^2,
\]
which has critical points 
\[
    u_{+} = \frac{3+\sqrt3}{6}
    \qquad\text{and}\qquad
    u_{-} = \frac{3-\sqrt3}{6}.
\]
Since it holds $g(0)=g(1)=0$, the maximum of $|g(u)|$ on $[0,1]$ is attained at one of these critical points.
We can verify
\begin{align*}
    |g(u_+)| = |g(u_-)|=\frac{\sqrt3}{18},
\end{align*}
which implies
\[
    |\psi''(t)|
    \le
    240 \cdot \frac{\sqrt3}{18}
    <24.
\]

\paragraph{The proof of statement  (c).}
The fact $\tanh t\in(-1,1)$ directly implies 
\begin{align*}
\phi(t)=1+\tanh t\in(0,2).    
\end{align*}
We can also verify
$\phi'(t)=\operatorname{sech}^2t$ and $\phi''(t) = -2\operatorname{sech}^2(t)\tanh(t)$, which implies
\begin{align*}
    |\phi''(t)|
= 2\operatorname{sech}^2(t)|\tanh(t)|
< 2\operatorname{sech}^2(t)
= 2\phi'(t).
\end{align*}

\paragraph{The proof of statement  (d).}
The statement that $\phi'$ is even can be verifies as 
\begin{equation*}                               
    \phi'(-t)=\operatorname{sech}^2(-t)=\operatorname{sech}^2(t)=\phi'(t).
\end{equation*}
It remains to show that $\phi'(t)>2/5$ for $|t|\le1$.
For all $t\geq 0$, it holds
\[
    \phi''(t) =
    -2\operatorname{sech}^2(t)\,\tanh (t)    \le0,
\]
which implies $\phi'$ is non-increasing on $[0,\infty)$.
Therefore, for all $t\in[0,1]$, it holds
\begin{equation*}
    \phi'(t) =\operatorname{sech}^2(t)
    \ge \operatorname{sech}^2(1)
    > \frac{2}{5}.
                                 
\end{equation*}
Since $\phi'(t)$ is even, we have $|\phi'(t)|>2/5$ when $|t|\leq 1$.

\end{align}

\paragraph{The Case of $|t|\le1/2$.}

|\psi''(t)|+|\psi''(-t)|=0\le24.

\paragraph{The Case of $t>1/2$.}

Combining with Lemma~\ref{lem:gate-properties}(b), we obtain

|\psi''(t)|+|\psi''(-t)|=|\psi''(-t)|\le24.
\]

Hence, equations (\ref{eq:gate-pair-value}), (\ref{eq:gate-pair-first}), and (\ref{eq:gate-pair-second}) hold for all $t\in{\mathbb{R}}$.

Recall that the function $\bar f_T$ is defined as
\begin{equation}
\bar f_T({\bf{y}}) := \sum_{i=1}^{T}A_i({\bf{y}}),
~~~~\text{where}~~
A_i({\bf{y}}):=
\begin{cases}
-\phi(y_1), & \text{if}~i=1, \\[0.1cm]
\psi(-y_{i-1})\phi(-y_i)
-
\psi(y_{i-1})\phi(y_i), & \text{otherwise}.    
\end{cases}
\label{eq:hard-zero-chain1}
\end{equation}

\paragraph{The proof of statement (a).}
According to definition (\ref{eq:gate-functions1}) and Lemma~\ref{lem:gate-properties}(a), we have $\phi(0)=1$ and~$\psi(0)=0$.
Hence, definition (\ref{eq:hard-zero-chain1}) implies
\begin{equation}
    A_i({\bf{0}})=\begin{cases}
        -1, &  \text{if}~i=1, \\
        0, &  \text{otherwise}.
    \end{cases}
    \qquad\text{and}\qquad     
    \bar f_T({\bf{0}})=-1.
    \label{eq:hard-chain-origin}
\end{equation}
For all ${\bf{y}}\in{\mathbb{R}}^T$, Lemma~\ref{lem:gate-properties}(c) implies
\[
    A_1({\bf{y}})=-\phi(y_1)>-2.
\]
Moreover, for every $i\ge2$, Lemma~\ref{lem:gate-properties}(b) and (c) implies
\[
\psi(-y_{i-1})\phi(-y_i)\ge0
\qquad\text{and}\qquad
0\le\psi(y_{i-1})\phi(y_i)<2.
\]
It follows from equation~(\ref{eq:hard-zero-chain1})
that
\[
    A_i({\bf{y}})>-2
    \qquad\text{and}\qquad
    \bar f_T({\bf{y}}) = \sum_{i=1}^T A_i({\bf{y}})>-2T.
\]
Thus, it holds
\[
    \inf_{{\bf{y}}\in{\mathbb{R}}^T}\bar f_T({\bf{y}})\ge-2T.
\]
We then consider the vector
\[
    {\bf{y}}^{(r)}= r[1,\ldots,1]^\top,
    \qquad \text{where}~~r\ge1.
\]
Lemma~\ref{lem:gate-properties}(a) implies $\psi(r)=1$ and $\psi(-r)=0$.
Consequently, definition equation~(\ref{eq:hard-zero-chain1}) implies
\[
    A_i({\bf{y}}^{(r)})
    =
    -\phi(r).
\]
for all $i\in[T]$, which results
\begin{equation}
    \lim_{r\to+\infty} \bar f_T({\bf{y}}^{(r)}) = \lim_{r\to+\infty} -T\phi(r) = -2T.
    \label{eq:hard-chain-infimum}
\end{equation}
Combining equations (\ref{eq:hard-chain-origin})
and (\ref{eq:hard-chain-infimum}) yields
\begin{equation*}
    \bar f_T({\bf{0}})-\inf_{{\bf{y}}\in{\mathbb{R}}^T}\bar f_T({\bf{y}})=2T-1.
\end{equation*}

\paragraph{The proof of statement (b).}
We first establish the uniform bound on the gradient of $\bar f_T$. Taking the derivative on equation~(\ref{eq:hard-zero-chain1}) and using the evenness of $\phi'$ from Lemma~\ref{lem:gate-properties}(d), we obtain
\small\begin{equation}\label{eq:hard-chain-gradient-coordinate}
\begin{split}    
\frac{\partial\bar f_T({\bf{y}})}{\partial y_i} = 
\begin{cases}
    -\phi'(y_1) -\psi'(-y_1)\phi(-y_2) -\psi'(y_1)\phi(y_2), & i = 1, \\
    -\big(\psi(-y_{i-1})+\psi(y_{i-1})\big)\phi'(y_i) -\psi'(-y_i)\phi(-y_{i+1})-\psi'(y_i)\phi(y_{i+1}), & i=2,\dots,T-1, \\
    -\big(\psi(-y_{T-1})+\psi(y_{T-1})\big)\phi'(y_T), & i=T. \\
\end{cases}
\end{split}
\end{equation}

According to Lemma~\ref{lem:gate-properties}(b) and (c) and Lemma~\ref{lem:gate-paired-properties}, we have
\[
    0\le (\psi(-y_{i-1})+\psi(y_{i-1}))\phi'(y_i) \le1
~~~~~~\text{and}~~~~~~
    0\le
    \psi'(-y_i)\phi(-y_{i+1})
    +\psi'(y_i)\phi(y_{i+1})
    \le8.
\]
Combing with equation~(\ref{eq:hard-chain-gradient-coordinate}) implies
\begin{equation*}                                     
    \left\| \nabla\bar f_T({\bf{y}}) \right\| 
= \sqrt{T} \max_{i\in[T]}\left|\frac{\partial\bar f_T}{\partial y_i}\right|
\le9\sqrt T.
\end{equation*}
We now consider the bound of Hessian. 
Lemma~\ref{lem:gate-properties}(c) implies
\[
    \left|\frac{\partial^2 A_1({\bf{y}})}{\partial y_1^2}\right|
    =|\phi''(y_1)|\le2.
\]
For all $i\ge2$, Lemma~\ref{lem:gate-properties}(c) and \ref{lem:gate-paired-properties} imply

\begin{align}
& \left|\frac{\partial^2 A_i({\bf{y}})}{\partial y_{i-1}^2}\right|
= |\psi''(-y_{i-1})\phi(-y_i)     -\psi''(y_{i-1})\phi(y_i)|
\le2(|\psi''(-y_{i-1})|+|\psi''(y_{i-1})|)
\le48, \label{eq:hard-A-xx-bound} \\[0.15cm] 
& \left|\frac{\partial^2 A_i({\bf{y}})}{\partial y_i^2}\right|
= |-(\psi(-y_{i-1})+\psi(y_{i-1}))\phi''(y_i)| \le2, \label{eq:hard-A-zz-bound} \\[0.15cm]
& \left|\frac{\partial^2 A_i({\bf{y}})}{\partial y_{i-1}\partial y_i}\right|
 = |(\psi'(-y_{i-1})-\psi'(y_{i-1}))\phi'(y_i)|
\le4.    \label{eq:hard-A-xz-bound}
\end{align}
Note that each $A_i({\bf{y}})$ depends only on $y_{i-1}$ and $y_i$ for $i\ge2$, which implies the Hessian of $\bar f_T({\bf{y}})$ is tridiagonal matrix, i.e.,
\begin{equation*}
    \frac{\partial^2\bar f_T({\bf{y}})}{\partial y_i\partial y_j}=0
    ~~~\text{for all}~~~i,j\in[T]~~~\text{such that}~~~|i-j|\ge2.
\end{equation*}
According to equations (\ref{eq:hard-A-xx-bound}) and (\ref{eq:hard-A-zz-bound}), the absolute value of each diagonal entry of $\nabla^2\bar f_T({\bf{y}})$ is bounded by $48+2=50$. 
Moreover, equation~(\ref{eq:hard-A-xz-bound}) implies that each nonzero off-diagonal entry of $\nabla^2\bar f_T({\bf{y}})$ has absolute value at most $4$.
Hence, every row of $\nabla^2\bar f_T({\bf{y}})$ has absolute row sum at most $50+2\cdot4=58$. 
Since the Hessian is symmetric, its spectral norm is bounded by its maximum absolute row sum, which implies
\begin{equation}
    \left\| \nabla^2\bar f_T({\bf{y}}) \right\|
    \le\left\| \nabla^2\bar f_T({\bf{y}}) \right\|_\infty
    \le58     
    \label{eq:hard-chain-hessian-bound}
\end{equation}

\paragraph{The proof of statement (c).}
We let $i\in[T]$ be the smallest index such that $|y_i|\le1$. 
If $i=1$, then equation~(\ref{eq:hard-chain-gradient-coordinate}) and the nonnegativity of $\psi'$ and $\phi$ from Lemma~\ref{lem:gate-properties}(b) and (c) implies
\[
    \frac{\partial\bar f_T({\bf{y}})}{\partial y_1}
    \le-\phi'(y_1).
\]
Combining with $|y_1|\le1$ and Lemma~\ref{lem:gate-properties}(d), we achieve
\[
    \left|\frac{\partial\bar f_T({\bf{y}})}{\partial y_1}\right|
    \ge\phi'(y_1)>\frac25.
\]
If $i\ge2$, the definition of $i$ implies $|y_{i-1}|>1$. 
According to Lemma~\ref{lem:gate-properties}(a), we obtain
\begin{equation}
    \psi(-y_{i-1})+\psi(y_{i-1})=1.
    \label{eq:previous-gate-one}
\end{equation}
Substituting equation~(\ref{eq:previous-gate-one}) into equation~(\ref{eq:hard-chain-gradient-coordinate}) and using the non-negativity of $\psi'$ and $\phi$ from Lemma~\ref{lem:gate-properties}(b) and (c), we have
\[
    \frac{\partial\bar f_T({\bf{y}})}{\partial y_i}
    \le-\phi'(y_i).
\]
Since $|y_i|\le1$, Lemma~\ref{lem:gate-properties}(d) implies
\[
    \left|\frac{\partial\bar f_T({\bf{y}})}{\partial y_i}\right|
    \ge\phi'(y_i)>\frac25.
\]
Combing above two inequality, we obtain
\begin{equation}
    \left\| \nabla\bar f_T({\bf{y}}) \right\|
    \ge
    \left|\frac{\partial\bar f_T}{\partial y_i}\right|
    >
    \frac25.
    \label{eq:hard-chain-certificate}
\end{equation}


\paragraph{The proof of statement (d).}
We now establish the gradient-dependent upper bound for Hessian.
We first consider the case that there exists some $i\in[T]$ such that $|y_i|\le1$. 
By equation~(\ref{eq:hard-chain-hessian-bound}) and equation~(\ref{eq:hard-chain-certificate}), we obtain
\begin{equation}
    \left\| \nabla^2\bar f_T({\bf{y}}) \right\|
    \le 58
    \leq 160\cdot\frac25
    \le160\left\| \nabla\bar f_T({\bf{y}}) \right\|.
    \label{eq:hard-chain-curvature-case-one}
\end{equation}
We then consider the case that $|y_i|>1$ holds for all $i\in[T]$. 
According to Lemma~\ref{lem:gate-properties}(a), we have
\[
    \psi(-y_i)+\psi(y_i)=1.
\]
Moreover, since both $y_i$ and $-y_i$ do not lie in the interval $(1/2,1)$, equation~(\ref{eq:gate-pair-derivative-zero}) implies
\[
    \psi'(-y_i)=\psi'(y_i)=0.
\]
Substituting the above two identities into equation~(\ref{eq:hard-chain-gradient-coordinate}), we obtain
\begin{equation}
    \frac{\partial\bar f_T({\bf{y}})}{\partial y_i}
    =-\phi'(y_i)
    \quad \text{for all}~~i\in[T].
    \label{eq:hard-chain-saturated-gradient}
\end{equation}
Hence, the Hessian of $\bar f_T({\bf{y}})$ is diagonal and satisfies
\begin{equation*}
    \nabla^2\bar f_T({\bf{y}})=-\operatorname{diag}\big(\phi''(y_1),\ldots,\phi''(y_T)\big).
                                            
\end{equation*}
Consequently, Lemma~\ref{lem:gate-properties}(c) and equation~(\ref{eq:hard-chain-saturated-gradient}) imply
\begin{align}
    \left\| \nabla^2\bar f_T({\bf{y}}) \right\|=
    \max_{i\in[T]}|\phi''(y_i)|
    \le2\max_{i\in[T]}\phi'(y_i)
    \le2\left\| \nabla\bar f_T({\bf{y}}) \right\|
    \le160\left\| \nabla\bar f_T({\bf{y}}) \right\|.
    \label{eq:hard-chain-curvature-case-two}
\end{align}
Combining equations (\ref{eq:hard-chain-curvature-case-one}) and  (\ref{eq:hard-chain-curvature-case-two}), we conclude that
\begin{equation*}
    \left\| \nabla^2\bar f_T({\bf{y}}) \right\|
    \le160\left\| \nabla\bar f_T({\bf{y}}) \right\|
    \qquad
\end{equation*}
for all ${\bf{y}}\in{\mathbb{R}}^T$.



\paragraph{The proof of statement (e).}
Follow the notation defined in Section \ref{sec:Preliminaries}, we let
\begin{align*}
    \mathrm{prog}_{1/4}({\bf{y}}):=\max\{i\in{\mathbb{N}} : |y_i| > 1/4\}.
\end{align*}
for given ${\bf{y}}=[y_1, \dots, y_T]\in{\mathbb{R}}^T$.
For each $i>\operatorname{prog}_{1/4}({\bf{y}})+1$, we have 
\begin{align}\label{eq:y_i14}
|y_i| \le 1/4 \qquad\text{and}\qquad |y_{i-1}| \le 1/4.    
\end{align}
Combining Lemma~\ref{lem:gate-properties}(a) and equation equation~(\ref{eq:gate-pair-derivative-zero}), equation~(\ref{eq:y_i14}) implies
\[
    \psi(-y_{i-1})=\psi(y_{i-1})=0  \qquad\text{and}\qquad     \psi'(-y_i)=\psi'(y_i)=0.
\]
Substituting the above results into equation~(\ref{eq:hard-chain-gradient-coordinate}), we obtain
\[
    \frac{\partial\bar f_T({\bf{y}})}{\partial y_i}=0
\]
for all $i>\operatorname{prog}_{1/4}({\bf{y}})+1$.
Hence, it holds
\begin{equation*}
    \operatorname{prog}_0\big(\nabla\bar f_T({\bf{y}})\big)
    \le\operatorname{prog}_{1/4}({\bf{y}})+1.
\end{equation*}

Following \citet[Lemma 13 and its proof]{arjevani2023lower}, we directly obtain
\begin{equation}

    \qquad\text{and}\qquad

\begin{equation}
    \alpha({\bf{x}}):=
    \left(1+\frac{\left\| {\bf{x}} \right\|^2}{R^2}\right)^{-1/2}.
    \label{eq:radial-alpha}
\end{equation}
Then we can write $\rho_R({\bf{x}})=\alpha({\bf{x}}){\bf{x}}$ and it holds  \citep[Lemma~13]{arjevani2023lower} 
\begin{equation}
    D\rho_R({\bf{x}})
    = \alpha({\bf{x}}){\bf I}_d - \frac{(\alpha({\bf{x}}))^3}{R^2}\cdot{\bf{x}}{\bf{x}}^\top.
    \label{eq:radial-jacobian}
\end{equation}
According to \citet[Lemma~13]{arjevani2023lower}, it holds
\begin{equation}
    \left\| D\rho_R({\bf{x}})-D\rho_R({\bf{y}}) \right\|
    \le\frac3R\left\| {\bf{x}}-{\bf{y}} \right\|
    \qquad\text{for all }{\bf{x}},{\bf{y}}\in{\mathbb{R}}^d.
    \label{eq:radial-jacobian-lipschitz}
\end{equation}
Since the mapping $\rho_R$ is twice continuously differentiable, for any ${\bf{h}},{\bf{k}}\in{\mathbb{R}}^d$, we have
\begin{align*}
    \left\| D^2\rho_R({\bf{x}})[{\bf{h}},{\bf{k}}] \right\|
    =
    \lim_{t\to0}
    \left\|
    \frac{
    \big(D\rho_R({\bf{x}}+t{\bf{k}})-D\rho_R({\bf{x}})\big){\bf{h}}
    }{t}
    \right\|\le
    \frac3R\left\| {\bf{h}} \right\|\left\| {\bf{k}} \right\|,
                                               
\end{align*}
where the inequality follows from equation~(\ref{eq:radial-jacobian-lipschitz}). Taking the supremum over all unit vectors ${\bf{h}}$ and ${\bf{k}}$, we achieve
\begin{equation}
    \left\| D^2\rho_R({\bf{x}}) \right\|
    \le\frac3R.
    \label{eq:radial-second-bound}
\end{equation}
This completes the proof.

Recall that equation~(\ref{eq:global-hard-function}) has defined

Combining with equation~(\ref{eq:hard-chain-origin}), we obtain
\begin{align*}
H_{{\bf U},\mu}({\bf{0}})=\bar f_T({\bf{0}})=-1.     
\end{align*}
Moreover, the lower bound of $\bar f_T({\bf{y}})\ge-2T$ in equation~(\ref{eq:hard-chain-infimum}) implies

    H_{{\bf U},\mu}({\bf{x}})

Combining above two results, we have

    H_{{\bf U},\mu}({\bf{0}})
    -\inf_{{\bf{x}}\in{\mathbb{R}}^d}H_{{\bf U},\mu}({\bf{x}})
    \le -1+2T   \le2T.
\end{equation*}

\paragraph{The lower bound of the Hessian.}
According to the chain rule and equation~(\ref{eq:radial-jacobian}), the gradient of~$H_{{\bf U},\mu}$ has the form of
\begin{equation}
    \nabla H_{{\bf U},\mu}({\bf{x}})
    =
    D\rho_R({\bf{x}}){\bf U}\nabla\bar f_T({\bf{y}})
    +\mu{\bf{x}}.
    \label{eq:global-gradient-formula}
\end{equation}
Since we have assumed there exists some $i\in[T]$ such that

we denote
\begin{equation*}

\end{equation*}
                                                                                                        
If $i_\star=1$, then equation~(\ref{eq:hard-chain-gradient-coordinate}) and the non-negativity of $\psi'$ and $\phi$ from Lemma~\ref{lem:gate-properties}(b) and (c) imply
\begin{align}\label{eq:d-f-1-phi}    
    \frac{\partial\bar f_T({\bf{y}})}{\partial y_1}
    \le-\phi'(y_1).
\end{align}
If $i_\star\ge2$, the minimality of $i_\star$ gives $|y_{i_\star-1}|>1$. 
Hence, Lemma~\ref{lem:gate-properties}(a) implies
\begin{align*}
\psi(-y_{i_\star-1})+\psi(y_{i_\star-1})=1.
\end{align*}
Substituting the above result into equation~(\ref{eq:hard-chain-gradient-coordinate}) also yields equation~(\ref{eq:d-f-1-phi}).

\begin{equation}

    \le-\phi'(y_{i_\star})
    <-\frac25.
    \label{eq:global-unfinished-gradient}
\end{equation}
In the case of $\left\| {\bf{x}} \right\|\ge20\sqrt T/\mu$, we apply equation~(\ref{eq:global-gradient-formula}), (\ref{eq:radial-jacobian-bound}), and \ref{eq:global-chain-bounds} to achieve

    &\ge 20\sqrt{T}
    -\left\| \nabla\bar f_T({\bf{y}}) \right\| \\
    &\ge  20\sqrt{T} -9\sqrt T,    \\
    &\ge  11.   
\end{split}

In the case of $\left\| {\bf{x}} \right\|<20\sqrt T/\mu$. The setting $R=1000\sqrt T/\mu$ implies $\left\| {\bf{x}} \right\|/R<1/50$. Therefore, equation~(\ref{eq:radial-alpha}) implies

\end{align}
Note that the definition of $i_\star$ indicates  $y_{i_\star}=\alpha({\bf{x}})\langle{\bf{u}}_{i_\star},{\bf{x}}\rangle$ and $|y_{i_\star}|\le1$, which results
\begin{equation}
    |\langle{\bf{u}}_{i_\star},{\bf{x}}\rangle|
    \le\frac1{\alpha({\bf{x}})}.
    \label{eq:global-coordinate-bound}
\end{equation}
For the second term on the right-hand side of inequality (\ref{eq:global-alpha-near}), we have
\begin{align}\label{eq:global-compression-error}
\begin{split}

\left| \langle{\bf{u}}_{i_\star},{\bf{x}}\rangle
\langle{\bf{x}},{\bf U}\nabla\bar f_T({\bf{y}})\rangle \right|  
\le \frac{(\alpha({\bf{x}}))^3}{R^2}
\left| \langle{\bf{u}}_{i_\star},{\bf{x}}\rangle \right|
\left\| {\bf{x}}\right\| \left\|{\bf U}\nabla\bar f_T({\bf{y}}) \right\|  \\

\end{split}
\end{align}

the second inequality is based on equation~(\ref{eq:global-coordinate-bound}),

For the last term on the right-hand side of inequality (\ref{eq:global-alpha-near}), equation~(\ref{eq:global-alpha-near}) and the setting $\mu\le10^{-2}$ implies

\paragraph{The upper bound for the Hessian.}
We now establish the gradient-dependent upper bound for the Hessian.

&=  \left\langle
    \nabla^2\bar f_T({\bf{y}})
    {\bf U}^\top D\rho_R({\bf{x}})[{\bf{h}}],
    {\bf U}^\top D\rho_R({\bf{x}})[{\bf{k}}]
    \right\rangle \\
&~~\quad + \left\langle
    {\bf U}\nabla\bar f_T({\bf{y}}),
    D^2\rho_R({\bf{x}})[{\bf{h}},{\bf{k}}]
    \right\rangle + \mu\langle{\bf{h}},{\bf{k}}\rangle.
\end{split}
\end{align}
Taking the supremum on ${\bf{h}}$ and ${\bf{k}}$ over the unit ball and and using the fact $\left\| {\bf U} \right\|=1$, we obtain
\begin{align}\label{eq:global-uniform-hessian}
\begin{split}    
     \left\| \nabla^2H_{{\bf U},\mu}({\bf{x}}) \right\|
\le &{} \left\| \nabla^2\bar f_T({\bf{y}}) \right\|\left\| D\rho_R({\bf{x}}) \right\|^2+\left\| \nabla\bar f_T({\bf{y}}) \right\|\left\| D^2\rho_R({\bf{x}}) \right\|+\mu \\
\le &{}  58+\frac{27\sqrt T}{R}+\mu 
\leq 58+\frac{1027\mu}{1000}  < 59,
\end{split}
\end{align}
where the second inequality is based on equations (\ref{eq:global-chain-bounds}), (\ref{eq:radial-jacobian-bound}), and (\ref{eq:radial-second-bound}), the second last two inequalities are based on the settings $R=1000\sqrt T/\mu$ and $\mu\le10^{-2}$.

    \right).

\end{equation}

If $\left\| {\bf{x}} \right\|\ge20\sqrt T/\mu$, then equations~(\ref{eq:global-uniform-hessian}) and (\ref{eq:global-large-radius-lower}) imply
\begin{align}
    \left\| \nabla^2H_{{\bf U},\mu}({\bf{x}}) \right\|
    &<59\le 240\cdot 11
    \le 240 \left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|\leq 240\left(\left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|+\mu\log \frac1\mu
    \right). \label{eq:global-curvature-case-two}        
\end{align}
If $\left\| {\bf{x}} \right\|<20\sqrt T/\mu$, then equation~(\ref{eq:global-alpha-near}) gives
\begin{align*}                    
    \alpha({\bf{x}})>\frac{99}{100} \qquad\text{and}\qquad 
    \frac{1}{\alpha({\bf{x}})}<\frac{100}{99}.
\end{align*}
The condition $|y_i|>1$ holds for all $i\in[T]$ indicates that we can combine equation~(\ref{eq:hard-chain-saturated-gradient}) and  Lemma~\ref{lem:gate-properties}(c) to achive
\begin{equation}

    =
    \max_{i\in[T]}|\phi''(y_i)|
    \le2\max_{i\in[T]}\phi'(y_i).
    \label{eq:global-completed-hessian-bound}
\end{equation}
We denote the index $j\in[T]$ such that $\phi'(y_j)=\max_{i\in[T]}\phi'(y_i)$ and analyze the right-hand side of inequality (\ref{eq:global-completed-hessian-bound}) by the following two cases.
\begin{enumerate}
\item If $|y_j|\ge\log(1/\mu)$, then Lemma~\ref{lem:gate-properties}(c) implies
\begin{equation}
    \max_{i\in[T]}\phi'(y_i)
    =
    \operatorname{sech}^2(y_j)
    \le4\mathrm e^{-2|y_j|}
    \le4\mu^2
    \le\mu\log\frac1\mu,
    \label{eq:global-phi-large-coordinate}
\end{equation}
where the first inequality follows from $\cosh t\ge\mathrm e^{|t|}/2$, and the last inequality uses $\mu\le10^{-2}$.
\item If $|y_j|<\log(1/\mu)$, then the identity $y_j=\alpha({\bf{x}})\langle{\bf{u}}_j,{\bf{x}}\rangle$ implies
\begin{equation}
    |\langle{\bf{u}}_j,{\bf{x}}\rangle|
    <
    \frac{\log(1/\mu)}{\alpha({\bf{x}})}.
    \label{eq:global-log-coordinate}
\end{equation}
Following the derivation of equation~(\ref{eq:global-directional-gradient}) by replacing ${\bf{u}}_{i_\star}$ with ${\bf{u}}_{j}$, we obtain
\begin{align*}
    \langle {\bf{u}}_j ,\nabla H_{{\bf U},\mu}({\bf{x}})\rangle  
    &=\frac{\alpha({\bf{x}})\partial\bar f_T({\bf{y}})}{\partial y_{j}}-\frac{(\alpha({\bf{x}}))^3}{R^2}\langle{\bf{u}}_j,{\bf{x}}\rangle\langle{\bf{x}},{\bf U}\nabla\bar f_T({\bf{y}})\rangle
    +\mu\langle{\bf{u}}_j,{\bf{x}}\rangle \\
    &=-\alpha({\bf{x}})\phi'(y_j)-\frac{(\alpha({\bf{x}}))^3}{R^2}\langle{\bf{u}}_j,{\bf{x}}\rangle\langle{\bf{x}},{\bf U}\nabla\bar f_T({\bf{y}})\rangle
    +\mu\langle{\bf{u}}_j,{\bf{x}}\rangle,
\end{align*}
where the last step is based on equation~(\ref{eq:hard-chain-saturated-gradient}).
Since we have $\phi'(y_j)=\max_{i\in[T]}\phi'(y_i)$ and~$\left\| {\bf{u}}_j \right\|=1$, applying the triangle inequality on above equation implies
\begin{align}\label{eq:alpha-max-phi-yi}
    \alpha({\bf{x}})\max_{i\in[T]}\phi'(y_i)\le\left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|+\frac{(\alpha({\bf{x}}))^3}{R^2}|\langle{\bf{u}}_j,{\bf{x}}\rangle|
    \left|\langle{\bf{x}},{\bf U}\nabla\bar f_T({\bf{y}})\rangle\right| + \mu|\langle{\bf{u}}_j,{\bf{x}}\rangle|.
\end{align}
For the second term on the right-hand side of equation~(\ref{eq:alpha-max-phi-yi}),
applying Cauchy--Schwarz inequality, equation~(\ref{eq:global-chain-bounds}) with the fact ${\bf U}^\top{\bf U}={\bf I}_T$ imply
\begin{align*}

\le \left\| {\bf{x}} \right\|\left\| \nabla\bar f_T({\bf{y}}) \right\|\le9\sqrt T\left\| {\bf{x}} \right\|.    
\end{align*}
Combining with equation~(\ref{eq:global-log-coordinate}), we obtain
\begin{align}\label{eq:alpha-max-phi-yi-1}
\begin{split}

\le & \frac{9(\alpha({\bf{x}}))^2\sqrt T\left\| {\bf{x}} \right\|\log(1/\mu)}{R^2} \\
< & \frac{9\mu\log(1/\mu)}{50000}<\frac{\mu\log(1/\mu)}{5000},
\end{split}
\end{align}
where the second inequality follows from $\alpha({\bf{x}})\le1$, $\left\| {\bf{x}} \right\|<20\sqrt T/\mu$, and $R=1000\sqrt T/\mu$.
Substituting equations (\ref{eq:alpha-max-phi-yi-1}) and (\ref{eq:global-log-coordinate}) into equation~(\ref{eq:alpha-max-phi-yi}), we obtain
\begin{align}

    \le
                                    
                                                                      
                 
        
    \left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|
    +
    \left(\frac1{5000}+\frac1{\alpha({\bf{x}})}
    \right)\mu\log\frac1\mu.
    \label{eq:global-phi-start-bound}
\end{align}
Dividing equation~(\ref{eq:global-phi-start-bound}) by $\alpha({\bf{x}})>0$ and applying equation~(\ref{eq:global-alpha-near}), we obtain
\begin{align}\label{eq:global-phi-small-coordinate}
\begin{split}    
    \max_{i\in[T]}\phi'(y_i)
    &\le \frac1{\alpha({\bf{x}})}
    \left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|+
    \left(\frac1{5000\alpha({\bf{x}})} + \frac1{(\alpha({\bf{x}}))^2} \right)\mu\log\frac1\mu \\
    &\le 2\left(
    \left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|
    +\mu\log\frac1\mu
    \right).
\end{split}
\end{align}
\end{enumerate}
Combining equations (\ref{eq:global-phi-large-coordinate}) and (\ref{eq:global-phi-small-coordinate}), we conclude that
\begin{equation}
    \max_{i\in[T]}\phi'(y_i)
    \le
    2\left(
    \left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|
    +\mu\log\frac1\mu
    \right).
    \label{eq:global-max-phi-bound}
\end{equation}
We now return to the bound of Hessian. The first inequality in equation~(\ref{eq:global-uniform-hessian}) implies
\begin{align}
\begin{split}
    \left\| \nabla^2H_{{\bf U},\mu}({\bf{x}}) \right\|
    &\le
    \left\| \nabla^2\bar f_T({\bf{y}}) \right\|\left\| D\rho_R({\bf{x}}) \right\|^2+\left\| \nabla\bar f_T({\bf{y}}) \right\|\left\| D^2\rho_R({\bf{x}}) \right\|+\mu\\
    &\le2\max_{i\in[T]}\phi'(y_i)+\frac{27\sqrt T}{R}+\mu
    \\
    &\le4\left(\left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|+\mu\log\frac1\mu\right)+\frac{1027}{1000}\mu
    \\
    &\le\frac{5027}{1000}\left(\left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|+\mu\log\frac1\mu\right)
    \\
    &<240\left(\left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|+\mu\log\frac1\mu\right),
\end{split}
\label{eq:global-curvature-case-three}
\end{align}
where the second inequality is based on 
equations (\ref{eq:global-completed-hessian-bound}),  
(\ref{eq:radial-jacobian-bound}), (\ref{eq:global-chain-bounds}),and  (\ref{eq:radial-second-bound});
the third inequality inequality follows from equation~(\ref{eq:global-max-phi-bound}) and setting $R=1000\sqrt T/\mu$;
the inequality uses the fact $\mu\le10^{-2}$ which implies $\mu\le\mu\log(1/\mu)$.

Combining equations (\ref{eq:global-curvature-case-one}), (\ref{eq:global-curvature-case-two}), and (\ref{eq:global-curvature-case-three}), we conclude that for every ${\bf{x}}\in{\mathbb{R}}^d$, it holds
\begin{equation*}
    \left\| \nabla^2H_{{\bf U},\mu}({\bf{x}}) \right\|
    \le
    240\left(
    \left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|
    +\mu\log\frac1\mu
    \right).
\end{equation*}
This completes the proof.

Recall that the definition of $f_{{\bf U}}({\bf{x}})=aH_{{\bf U},\mu}({\bf{x}}/b)$ in equation~(\ref{eq:scaled-hard-function}) implies 
\begin{align}
    \nabla f_{{\bf U}}({\bf{x}})
    =\frac{a}{b}\nabla H_{{\bf U},\mu}\left(\frac{{\bf{x}}}{b}\right),
\qquad\text{and}\qquad
    \nabla^2 f_{{\bf U}}({\bf{x}}) = \frac{a}{b^2}\nabla^2 H_{{\bf U},\mu}\left(\frac{{\bf{x}}}{b}\right).
    \label{eq:scaled-hessian-formula}
\end{align}
We verify the properties of the function $f_{{\bf U}}({\bf{x}})$ as follows.

\paragraph{The proof of statement (a).}
Combining the gradient-dependent curvature bound equation~(\ref{eq:global-hard-curvature}) and equation~(\ref{eq:scaled-hessian-formula}), we have
\begin{align}\label{eq:scaled-hessian-intermediate}
\begin{split}    
& \left\| \nabla^2 f_{{\bf U}}({\bf{x}}) \right\| 
= \frac{a}{b^2}\left\| \nabla^2 H_{{\bf U},\mu}\left(\frac{{\bf{x}}}{b}\right) \right\| \\
\leq &  \frac{240a}{b^2}\left(
    \left\| \nabla H_{{\bf U},\mu}({\bf{x}}) \right\|
    +\mu\log\frac1\mu
    \right) \\
=&  \frac{240}{b}\left\| \nabla f_{{\bf U}}({\bf{x}}) \right\|
    +\frac{240a\mu}{b^2}\log\frac1\mu \\
=& \frac{L_1}{2}\left\| \nabla f_{{\bf U}}({\bf{x}}) \right\|
    +8L_1\epsilon\mu\log\frac1\mu,
\end{split}
\end{align}
where the last step follows from the setting $b=480/L_1$ and $a/b=16\epsilon$ in equation~(\ref{eq:hard-scaling-parameters}).
For the term $\mu\log(1/\mu)$, recall we set
\begin{align}
    \delta_0=\min\left\{10^{-2},\frac{L_0}{16L_1\epsilon}\right\}\in(0, 10^{-2}]
    \qquad \text{and} \qquad
    \mu=\frac{\delta_0}{4\log(\mathrm e/\delta_0)}
    \label{eq:hard-scaling-parameters1}

in equation~(\ref{eq:hard-scaling-parameters}), which implies

\begin{equation*}

    =
    \frac{a}{b}
    D\rho_R\left(\frac{{\bf{x}}}{b}\right)^\top

\end{equation*}

    {\bf U}\nabla Q_T({\bf{y}}).
    \label{eq:oracle-noise-exact}
\end{equation}

\paragraph{Unbiasedness.}
Since $Z\sim\operatorname{Bernoulli}(q)$ with $q\in(0,1]$, taking the expectation on equation~(\ref{eq:oracle-noise-exact}) achieves
\begin{equation}
    {\mathbb{E}}_Z[\mathsf O_{f_{{\bf U}}}({\bf{x}};Z)]
    =
    \nabla f_{{\bf U}}({\bf{x}}).
    \label{eq:oracle-unbiased-proof}
\end{equation}

\paragraph{Bound on the $p$th central moment.}
Combing equations (\ref{eq:oracle-noise-exact}), (\ref{eq:radial-jacobian-bound}), and (\ref{eq:Q-gradient-bound}), we have
\begin{align}\label{eq:oracle-noise-pointwise}
\begin{split}    
    \left\| \mathsf O_{f_{{\bf U}}}({\bf{x}};Z)-\nabla f_{{\bf U}}({\bf{x}}) \right\|
    &\le
    \frac{a}{b}
    \left|\frac Zq-1\right|
    \left\| D\rho_R({\bf{x}}/b) \right\|
    \left\| {\bf U} \right\|
    \left\| \nabla Q_T({\bf{y}}) \right\| \\
    &\le
    16\epsilon\cdot249
    \left|\frac Zq-1\right|
    =
    3984\epsilon
    \left|\frac Zq-1\right|,
\end{split}
\end{align}
where we use the setting $\left\| {\bf U} \right\|=1$ and $a/b=16\epsilon$ from equation~(\ref{eq:hard-scaling-parameters}).
Since $Z\sim\operatorname{Bernoulli}(q)$ and~$p\in(1,2]$, we have
\begin{align}
    {\mathbb{E}}_Z\left[\left|\frac Zq-1\right|^p\right]
    &=
    (1-q)^p q^{1-p}+1-q\le
    q^{1-p}+1
    \le2q^{1-p},
    \label{eq:oracle-bernoulli-upper}
\end{align}
where the inequalities follow from $0<q\le1$ and $p>1$. Consequently, combining equations (\ref{eq:oracle-noise-pointwise}) and (\ref{eq:oracle-bernoulli-upper}) imply
\begin{equation}
    {\mathbb{E}}_Z\left[
    \left\| \mathsf O_{f_{{\bf U}}}({\bf{x}};Z)-\nabla f_{{\bf U}}({\bf{x}}) \right\|^p
    \right]
    \le
    2(3984\epsilon)^p q^{1-p}.
    \label{eq:oracle-noise-before-q}
\end{equation}
If $q<1$, then $\sigma>0$ and the definition of $q$ gives $q^{p-1}=2(3984)^p\epsilon^p\sigma^{-p}$.
Substituting this identity into equation~(\ref{eq:oracle-noise-before-q}), we obtain
\begin{align}

    \left\| \mathsf O_{f_{{\bf U}}}({\bf{x}};Z)-\nabla f_{{\bf U}}({\bf{x}}) \right\|^p
    \right]
    \le 2(3984\epsilon)^p q^{1-p}=2(3984\epsilon)^p\cdot\frac{\sigma^p}{2(3984)^p\epsilon^p}=\sigma^p.
    \label{eq:oracle-moment-q-less-one}
\end{align}
If $q=1$, then $Z=1$ almost surely. Hence, equation~(\ref{eq:oracle-unbiased-proof}) implies
\begin{equation}
    {\mathbb{E}}_Z\left[
    \left\| \mathsf O_{f_{{\bf U}}}({\bf{x}};Z)-\nabla f_{{\bf U}}({\bf{x}}) \right\|^p
    \right]
    =0\le\sigma^p.
    \label{eq:oracle-moment-q-one}
\end{equation}
Combining equation~(\ref{eq:oracle-moment-q-less-one}) and equation~(\ref{eq:oracle-moment-q-one}), we conclude that for every ${\bf{x}}\in{\mathbb{R}}^d$,
\begin{equation}
    {\mathbb{E}}_Z\left[
    \left\| \mathsf O_{f_{{\bf U}}}({\bf{x}};Z)-\nabla f_{{\bf U}}({\bf{x}}) \right\|^p
    \right]
    \le\sigma^p.
    \label{eq:oracle-p-moment-proof}
\end{equation}

\label{lem:normalize}
For all $\gamma \geq 0$, ${\bf{g}},{\bf{h}}\in{\mathbb{R}}^d$, it holds
\begin{align*}
\langle {\bf{h}},\varphi_\gamma({\bf{g}})\rangle
\ge \|{\bf{h}}\|-2\|{\bf{g}}-{\bf{h}}\|-\gamma.

<!-- end lemma-14 -->
<!-- proof-15: body_md -->
If ${\bf{g}}={\bf{0}}$ and $\gamma=0$, the result is trivial. Hence, we focus on the case of $\|{\bf{g}}\|+\gamma>0$.
According to Cauchy–Schwarz inequality and triangle inequality, we achieve 
\[
\begin{aligned}
\langle {\bf{h}},\varphi_\gamma({\bf{g}})\rangle
&=\frac{\|{\bf{g}}\|^2+\langle {\bf{h}}-{\bf{g}},{\bf{g}}\rangle}{\|{\bf{g}}\|+\gamma}\\
&\ge \frac{\|{\bf{g}}\|^2}{\|{\bf{g}}\|+\gamma}
      -\frac{\|{\bf{h}}-{\bf{g}}\|\cdot\|{\bf{g}}\|}{\|{\bf{g}}\|+\gamma}\\
&\ge \|{\bf{g}}\|-\gamma-\|{\bf{h}}-{\bf{g}}\|\\
&\ge \|{\bf{h}}\|-\gamma-2\|{\bf{h}}-{\bf{g}}\|.
\end{aligned}
\]
This finishes the proof.

\label{lem:heavy-tailed} 
For Algorithm \ref{alg:sn-mb-sgd}, it holds that
\begin{align*}
    {\mathbb{E}}[\left\| {\bf{g}}_t-\nabla f({\bf{x}}_t) \right\|] \leq   \frac{2 \sqrt{2}\sigma}{b^{1-1 / p}}
\end{align*}
<!-- end lemma-15 -->
<!-- proof-16: body_md -->
Recall that the stochastic gradient evaluation in the algorithm is 
\begin{align*}
    {\bf{g}}_t = \frac{1}{b}\sum_{i=1}^b  {\bf{g}}({\bf{x}}_t;Z_{t,i}),
    \qquad\text{where}~~ Z_{t,1},\dots,Z_{t,b}\overset{\mathrm{i.i.d}}\sim {\mathcal{Z}}.
\end{align*}
Hence, it holds
\begin{align}\label{eq:exp-dk}
\begin{split}    
     &   \mathbb{E}\left[\left\| {\bf{g}}_t-\nabla f({\bf{x}}_t) \right\|\right] \\
     = & \mathbb{E}\left[\left\| \sum_{i=1}^{b} \frac{1}{b}\left({\bf{g}}({\bf{x}}_t;Z_{t,i})-\nabla f({\bf{x}}_t)\right) \right\|\right]  \\
    \leq & 2 \sqrt{2} \mathbb{E}\left [\left(\frac{1}{b^{p}} \sum_{i=1}^{b}\left\| {\bf{g}}({\bf{x}}_t;Z_{t,i})-\nabla f({\bf{x}}_i) \right\|^{p}\right)^{1 / p}\right] \\
    = &\frac{2 \sqrt{2}}{b} \mathbb{E}\left[\left(\sum_{i=1}^{b}\left\| {\bf{g}}({\bf{x}}_t;Z_{t,i})-\nabla f({\bf{x}}_t) \right\|^{p}\right)^{1 / p}\right]\\
\leq & \frac{2 \sqrt{2}}{b} \left(\mathbb{E}\left[\sum_{i=1}^{b}\left\| {\bf{g}}({\bf{x}}_t;Z_{t,i})-\nabla f({\bf{x}}_t) \right\|^{p}\right]\right)^{1 / p}\\
   \leq  & \frac{2 \sqrt{2}}{b} \cdot b^{1 / p} \sigma= \frac{2 \sqrt{2}\sigma}{b^{1-1 / p}},
\end{split}
\end{align}
where the first inequality is based on Lemma 4.3 of  \citet{liu2025nonconvex},
second inequality is based on Jensen's inequality, and the last inequality is based on $p$-BCM assumption.
