<!-- theorem-1: body_md -->
\label{thm:main-results}
Fix $p$ and $I=[c_0,c_1]\subset(0,\infty)$.
\begin{enumerate}
\item[\textup{(i)}] For every sufficiently large admissible prime $q$, uniformly for $c\in I$,
\begin{equation}
\label{eq:individual-return-bound}
 \frac1{N_q}\ll_{p,I}\Delta_q(\alpha_{q,c})
 \ll_{p,I}\frac{\log\log q}{N_q}.
\end{equation}
\item[\textup{(ii)}] Assume the GRH for $L(s,\chi_D)$ for every negative fundamental discriminant $D$, where $\chi_D(n)=\left(\frac{D}
  {n}\right)$.
    For every sufficiently large admissible prime $q$, uniformly for $c\in I$,
\begin{equation}
\label{eq:grh-return}
 \Delta_q(\alpha_{q,c})
 \asymp_{p,I}\frac1{N_q}.
\end{equation}
\item[\textup{(iii)}] For every fixed real number $a\ge1$, as $Q\to\infty$,
\begin{equation}
\label{eq:main-moments}
 \frac1{|\mathcal A_p(Q)|}\sum_{q\in\mathcal A_p(Q)}
 \left(N_q\sup_{c\in I}\Delta_q(\alpha_{q,c})\right)^a
 \ll_{p,I,a}1.
\end{equation}
\end{enumerate}

<!-- corollary-1: body_md -->
\label{cor:density-one}
Let $w:[3,\infty)\to[1,\infty)$ be any nondecreasing function with $w(t)\to\infty$.
There is a subset $\mathcal A_p(w)\subset\mathcal A_p$ of relative density one such that, for all sufficiently large $q\in\mathcal A_p(w)$,
\begin{equation}
\label{eq:density-one}
 0\le\Delta_q(\alpha_{q,c})
 \le\frac{w(q)}{N_q}
 \qquad(c\in I).
\end{equation}
More quantitatively, for every $a\ge1$ and $H\ge1$,
\begin{equation}
\label{eq:exceptional-levels}
 \#\left\{q\in\mathcal A_p(Q):
 N_q\sup_{c\in I}\Delta_q(\alpha_{q,c})>H\right\}
 \ll_{p,I,a}\frac{Q}{H^a\log Q}.
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:path-series}
For $0<\alpha<1$ and $u$ as in \eqref{eq:alpha-u},
\begin{equation}\label{eq:path-series}
 \pi_{\alpha,x}(y)=\mathcal M_p(\alpha)
       \sum_{n\ge0}b_n(x,y)u^n,
 \qquad
 \mathcal M_p(\alpha)
       =(1-\alpha)\frac{1+pu^2}{1-u^2}.
\end{equation}
On the infinite regular tree, the corresponding probability at distance $r$ from the source is $\mathcal M_p(\alpha)u^r$.

There is one path of length zero from a vertex to itself, so $B_0=I$, and $B_1=A_X$.
Among all paths of length two, the forbidden ones go from a vertex to a neighbor and immediately back.
There are $p+1$ of them at each vertex. Hence $B_2=A_X^2-(p+1)I$.

For $n\ge2$, the entry $(A_XB_n)(x,y)$ counts paths obtained by adding an edge $x\to z$ before a non-backtracking path from $z$ to $y$.
An immediate reversal can occur only at the new join.
Such a path starts $x\to z\to x$.
Deleting these two edges leaves a non-backtracking path of length
$n-1$ from $x$ to $y$.
For each remaining path, $z$ can be any neighbor of $x$ except its first neighbor.
There are exactly $p$ choices.
Subtracting the forbidden paths gives
$$
 B_{n+1}=A_XB_n-pB_{n-1}\qquad(n\ge2).
$$

Let $F(u)=\sum_{n\ge0}B_nu^n$.
In the product $D_X(u)F(u)$, the constant coefficient is $I$, the coefficient of $u$ is zero, and the coefficient of $u^2$ is $-I$.
The recurrence makes every later coefficient zero.
Therefore
\begin{equation}
\label{eq:nb-generating}
 \sum_{n\ge0}B_nu^n=(1-u^2)D_X(u)^{-1}.
\end{equation}
These manipulations are valid for $u<p^{-1}$.
Indeed, the number of non-backtracking paths of length $n\ge1$ from a fixed vertex is $(p+1)p^{n-1}$: there are $p+1$ choices for the first step and $p$ for every later step.
The resulting geometric series converges.
Combining \eqref{eq:nb-generating} with \eqref{eq:pagerank-resolvent} and \eqref{eq:alpha-u} proves the finite-graph formula.

On the tree, let $f(y)=\mathcal M_p(\alpha)u^{d(o,y)}$.
There are $(p+1)p^{r-1}$ vertices at distance $r\ge1$, so
$$
 \sum_yf(y)=\mathcal M_p(\alpha)
\left(1+\frac{(p+1)u}{1-pu}\right)
=1.
$$
For the last equality, \eqref{eq:alpha-u} gives $\mathcal M_p(\alpha)=(1-pu)/(1+u)$.
At positive distance, the neighbor counts and the identity $u^r=\alpha(u^{r-1}+pu^{r+1})/(p+1)$ show that $f=\alpha P f$.
At the root, the required equation is
$\mathcal M_p=(1-\alpha)+\alpha\mathcal M_p u$, which follows from the same formulas.
Hence $f=(1-\alpha)\delta_o+\alpha P f$ on the whole tree.
The stopped-walk distribution satisfies this equation as well.
Two probability solutions agree: a random walk step cannot increase the sum of the absolute values of their difference, and iterating the equation bounds that sum by $2\alpha^n$ for every $n$.
Thus $f$ is the stopped-walk distribution on the tree.
Its value at the root is $\mathcal M_p(\alpha)$.

<!-- lemma-2: body_md -->
\label{lem:quaternion-count}
For every admissible $q$ and integer $k\ge1$,
\begin{equation}
\label{eq:quaternion-count}
 \mathcal T_{2k}(x,x)
 =1+S(p^k,q),
 \qquad
 b_{2k}(x,x)
 =S(p^k,q)-S(p^{k-1},q).
\end{equation}
In particular, $0\le b_{2k}(x,x)\le S(p^k,q)$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
A word of length $n$ in the Lubotzky--Phillips--Sarnak generators describes a path of length $n$.
Quaternionic factorization identifies non-backtracking words with primitive
integral quaternions
$$
 a=a_0+a_1i+a_2j+a_3k,
 \qquad
 \operatorname{Nrd}(a)=p^n,
$$
up to an overall sign, see \cite[Lemma~2.3]{LPSHecke1987}.
Here $a_0$ is odd, the other three coordinates are even, and primitive
means that the four coordinates are not all divisible by $p$.
Since $a_0$ is odd, it is nonzero.
We choose the unique overall sign for which $a_0>0$.

Every integral quaternion $a$ of norm $p^{2k}$ with these parity conditions
has a unique expression $a=p^j a'$ when $0\le j\le k$, where $a'$ is primitive.
Indeed, $j$ is the largest exponent such that $p^j$ divides all four coordinates.
Dividing by $p^j$ preserves the parity
conditions and the positivity of the real coordinate, and gives
$\operatorname{Nrd}(a')=p^{2k-2j}$.
Thus $a'$ corresponds to a non-backtracking path of length $2k-2j$.
Conversely, multiplying the primitive quaternion of such a path by $p^j$
gives a quaternion of norm $p^{2k}$.
This explains the decomposition
$\mathcal T_{2k}=B_{2k}+B_{2k-2}+\cdots+B_0$ in \eqref{eq:complete-hecke}.

We first count returns at the identity vertex of $\operatorname{PGL}_2(\mathbb F_q)$.
Let $\rho_q(a)$ denote the matrix obtained by reducing the quaternion
coordinates modulo $q$ and applying the matrix map in the Lubotzky--Phillips--Sarnak construction.
The endpoint of the associated path is the projective class $[\rho_q(a)]$.
Since the identity class in $\operatorname{PGL}_2(\mathbb F_q)$ consists of the nonzero
scalar matrices, the path returns precisely when $\rho_q(a)=cI$ for some $c\in\mathbb F_q^\times$.
The matrix is invertible because its determinant is the quaternion norm,
which is a power of $p$, and $q\ne p$.
Multiplication or division by $p$ changes this matrix by a nonzero scalar
modulo $q$, so it preserves the return condition.

It follows that $\mathcal T_{2k}(x,x)$ counts all integral quaternions
of norm $p^{2k}$ with positive real coordinate, the stated parity
conditions, and scalar image modulo $q$.
The scalar quaternion $p^k$ is included, its primitive part is $1$,
which corresponds to the empty path.
Left translation is a graph automorphism, so the same count holds
at every vertex $x$.

Write $P=p^k$.
The quaternion matrix map shows that the scalar condition is equivalent to
$a_1\equiv a_2\equiv a_3\equiv0\pmod q$.
Since these coordinates are also even and $q$ is odd, we may write $a_i=2qb_i$ for $i=1,2,3$.
The norm equation becomes $a_0^2+4q^2(b_1^2+b_2^2+b_3^2)=P^2$,
here $a_0>0$ odd.
There is exactly one scalar quaternion in this set: $a_0=P$ and $b_1=b_2=b_3=0$.
It accounts for the term $1$ in the first identity.

For every remaining quaternion, $0<a_0<P$.
Let $d_-=\frac{P-a_0}{2}$, $d_+=\frac{P+a_0}{2}$.
These are positive integers satisfying
$d_-+d_+=P$, $d_-d_+=q^2(b_1^2+b_2^2+b_3^2)$.
Since $q\nmid P$, the prime $q$ cannot divide both $d_-$ and $d_+$.
Their product is divisible by $q^2$, so exactly one of them is divisible
by $q^2$.
Write that factor as $q^2h$.
The other factor is then $P-q^2h$, and their positivity gives $0<h<\frac{P}{q^2}$.
Moreover,
$$
 a_0=|d_+-d_-|=|P-2q^2h|,
 \qquad
 b_1^2+b_2^2+b_3^2=h(P-q^2h).
$$
The integer $h$ is uniquely determined, since exactly one of
$d_-$ and $d_+$ is divisible by $q^2$.

Conversely, take an integer $h$ with $0<h<P/q^2$ and an integer triple
satisfying $b_1^2+b_2^2+b_3^2 =h(P-q^2h)$.
Let $a_0=|P-2q^2h|$, $a_i=2qb_i$ with $i=1,2,3$.
The number $a_0$ is odd and nonzero, since $P$ is odd.
The bounds on $h$ give $a_0<P$, and direct substitution gives $a_0^2+a_1^2+a_2^2+a_3^2=P^2$.
Thus these coordinates define a quaternion with the required positivity, parity, norm, and scalar image modulo $q$.
The uniqueness of $h$ shows that each quaternion is counted once.
Consequently,
$$
 \mathcal T_{2k}(x,x)
 =1+\sum_{1\le h<P/q^2}r_3\bigl(h(P-q^2h)\bigr)
 =1+S(P,q).
$$
This proves the first identity.

Finally, \eqref{eq:complete-hecke} gives $B_{2k}=\mathcal T_{2k}-\mathcal T_{2k-2}$.
Taking diagonal entries and applying the first identity yields
$$
 b_{2k}(x,x)
 =\bigl(1+S(p^k,q)\bigr)
  -\bigl(1+S(p^{k-1},q)\bigr)
 =S(p^k,q)-S(p^{k-1},q).
$$
The lower bound follows because $b_{2k}(x,x)$ counts paths.
The upper bound follows from $S(p^{k-1},q)\ge0$.

\label{lem:return-reduction}
Uniformly for sufficiently large admissible $q$,
\begin{equation}
\label{eq:return-reduction}
 N_q\sup_{c\in I}\Delta_q(\alpha_{q,c})
 \ll_{p,I}1+\frac1{\log q}\sum_{k\le K_q}\mathcal B_q(k).
\end{equation}

The path recurrence expresses $B_n=Q_n(A_q)$, with $Q_0=1$, $Q_1(\lambda)=\lambda$, and, for $n\ge2$,
$$
 Q_n(\lambda)=p^{n/2}
 U_n\left(\frac{\lambda}{2\sqrt p}\right)
 -p^{(n-2)/2}U_{n-2}
 \left(\frac{\lambda}{2\sqrt p}\right).
$$
Here $U_n$ is the Chebyshev polynomial of the second kind, characterized by $U_0=1$, $U_1(v)=2v$, and $U_{n+1}(v)=2vU_n(v)-U_{n-1}(v)$.
It satisfies $|U_n(v)|\le n+1$ for $|v|\le1$.
Each of the two trivial spectral projections has diagonal entries $1/N_q$.
The other eigenvalues lie in $[-2\sqrt p,2\sqrt p]$.
The spectral theorem thus gives
\begin{equation}
\label{eq:spectral-return}
 b_{2k}(x,x)
 =\frac{2(p+1)p^{2k-1}}{N_q}
 +O_p((k+1)p^k).
\end{equation}
The sum of the nontrivial spectral projections has diagonal at most one.

For $\alpha=\alpha_{q,c}$, write $M=\mathcal M_p(\alpha)$ and $t=pu$.
Equations \eqref{eq:critical-relations} and
\eqref{eq:critical-t} give
\begin{align*}
 M\sum_{k>K_q}b_{2k}(x,x)u^{2k}
 &\ll_p\frac{M}{N_q}\sum_{k>K_q}t^{2k}
 +M\sum_{k>K_q}(k+1)(t^2/p)^k
 \\
 &\ll_{p,I}\frac1{N_q}+M(K_q+1)p^{-K_q}
 \ll_{p,I}q^{-3}.
\end{align*}
For $k\le K_q$, use $b_{2k}\le S(p^k,q)$ and $t\le1$ to obtain
$$
 M\sum_{k\le K_q}
 b_{2k}(x,x)u^{2k}
 \le q^{-3}M
 \sum_{k\le K_q}\mathcal B_q(k).
$$
Since $M\ll_{p,I}1/\log q$ and $N_q\asymp q^3$, the result follows.

<!-- lemma-8: body_md -->
\label{lem:prime-euler-moments}
Fix a positive integer $j$.
Let $P,h$ be positive integers, let $z\ge2$, and let $J$ be a bounded interval in $(z,\infty)$ on which $m_q=h(P-q^2h)$ is positive.
Then
\begin{equation}
\label{eq:prime-euler-moments}
 \sum_{\substack{q\in J\\q\text{ prime}}}E_z(m_q)^j
 \ll_j
 \left(\frac{hP}{\varphi(hP)}\right)^{j+1}
 \frac{|J|}{\log z}
 +z^2(\log(2z))^{3j}.
\end{equation}
Here $|J|$ is the length of $J$.
The constant is independent of $h$, $P$, $J$, and $z$.
<!-- end lemma-8 -->
<!-- proof-9: body_md -->
After normalization, the Euler products become probabilities of avoiding random residue classes.
Excluding multiples of small primes
gives a term $1/\ell$ in the local density.
This gives sieve dimension $\kappa=j+1$.

Let $\kappa=j+1$, $M=2hP$, $F(a)=h^2a^2-hP$, and $\mathfrak M(z)=\prod_{\ell\le z}(1-1/\ell)^{-1}$.
For each odd prime $\ell\nmid M$, define
\begin{equation*}
 \Omega_\ell=
 \left\{a\bmod\ell:
 \left(\frac{F(a)}\ell\right)=-1\right\}.
\end{equation*}
The elementary identity
$\sum_{a\bmod\ell}(\frac{a^2-b}{\ell})=-1$ for $b\ne0\bmod\ell$ follows by counting the solutions of $(a-v)(a+v)=b$.
It gives
\begin{equation}
\label{eq:moment-local-counts}
 |\Omega_\ell|
 =\frac{\ell-\left(\frac{hP}{\ell}\right)}2,
 \qquad
 \mathbf 1_{0\in\Omega_\ell}
 =\frac{1-\left(\frac{-hP}{\ell}\right)}2.
\end{equation}
At a prime dividing $h$, the polynomial $F$ vanishes identically.
At an odd prime dividing $P$ it is the square $h^2a^2$.
Thus the Euler factors at primes dividing $M$ all have the value $(1-1/\ell)^{-1}$.

For $\ell\nmid M$, choose independent random variables $\epsilon_\ell\in\{0,1\}$ with
\begin{equation}
\label{eq:moment-activation}
 \Pr(\epsilon_\ell=1)=a_{\ell,j}
 =1-\left(\frac{\ell-1}{\ell+1}\right)^j.
\end{equation}
When $\epsilon_\ell=1$, exclude the classes in $\Omega_\ell$.
Also exclude the zero class at every prime.
More precisely, let
\begin{equation*}
 \Gamma_\ell=
 \begin{cases}
  \{0\}\cup\Omega_\ell,&\ell\nmid M,\ \epsilon_\ell=1,\\
  \{0\},&\text{otherwise}.
 \end{cases}
\end{equation*}
For a prime $q>z$, the probability that $q$ avoids all the sets $\Gamma_\ell$ with $\ell\le z$ is $E_z(m_q)^j/\mathfrak M(z)^j$.
Indeed, the ratio between the factors with $\psi_m(\ell)=-1$ and $\psi_m(\ell)=1$ is $(\ell-1)/(\ell+1)$.
Consequently,
\begin{equation}
\label{eq:moment-survival-majorant}
 \sum_{\substack{q\in J\\q\text{ prime}}}E_z(m_q)^j
 \le \mathfrak M(z)^j
 \sum_{a\in J\cap\mathbb Z}
 \mathbb E\,\mathbf 1_{a\notin\Gamma_\ell\ (\ell\le z)}.
\end{equation}

Write $g(\ell)=\mathbb E|\Gamma_\ell|/\ell$.
For $\ell\nmid M$, equations
\eqref{eq:moment-local-counts} and \eqref{eq:moment-activation} give
\begin{equation}
\label{eq:moment-local-density}
 g(\ell)=\frac1\ell+
 a_{\ell,j}\frac{\ell-1+\delta_\ell}{2\ell},
 \qquad
 \delta_\ell
 =\left(\frac{-hP}\ell\right)
 -\left(\frac{hP}\ell\right)\in\{-2,0,2\}.
\end{equation}
For $\ell\mid M$, one has $g(\ell)=1/\ell$.
The inequality $a_{\ell,j}\le2j/(\ell+1)$ shows that $\ell g(\ell)\le\kappa$.
Moreover, $0<g(\ell)<1$ at every prime.
For a prime outside $M$, there are at most $\ell-1$ nonzero classes in $\Omega_\ell$, and hence
\begin{equation}
\label{eq:moment-small-primes}
 1-g(\ell)\ge
 \left(1-\frac1\ell\right)
 \left(\frac{\ell-1}{\ell+1}\right)^j>0.
\end{equation}
The same lower bound holds at primes dividing $M$.

Choose a fixed $s_j\ge\max\{2,\beta_\kappa+1\}$ and let $y=z^{2/s_j}$, $D=z^2=y^{s_j}$.
We first suppose that $y\ge2$.
The Euler product remains truncated at $z$, and we sieve with the primes $\ell<y$.

For an integer $a\in J$ and a choice of the random sets, let
$n(a)=\prod_{\substack{
\ell<y\\a\bmod\ell\in\Gamma_\ell}}\ell$, $A_n=\sum_{a\in J\cap\mathbb Z}\Pr(n(a)=n)$.
This defines a finite nonnegative sequence $\mathcal A=(A_n)$.
Its sifted sum is
$$
 S(\mathcal A,y)=
 \sum_{a\in J\cap\mathbb Z}
 \Pr(a\bmod\ell\notin\Gamma_\ell\text{ for every }\ell<y).
$$
Since $y\le z$, equation \eqref{eq:moment-survival-majorant} gives
\begin{equation}
\label{eq:moment-sieve-majorant}
 \sum_{\substack{q\in J\\q\text{ prime}}}
 E_z(m_q)^j
 \le
 \mathfrak M(z)^j S(\mathcal A,y).
\end{equation}

Extend $g$ multiplicatively to square-free integers.
For $d\mid\mathcal P(y)$, independence and the Chinese remainder
theorem give
\begin{equation}
\label{eq:moment-crt-average}
 |\mathcal A_d|
 =|J|g(d)+r_d,\qquad
 |r_d|\ll\kappa^{\omega(d)}.
\end{equation}
Indeed, each residue class modulo $d$ contains $|J|/d+O(1)$ integers in $J$, and the expected number of excluded
classes modulo $d$ is $\prod_{\ell\mid d}\ell g(\ell)\le\kappa^{\omega(d)}$.

We check the dimension condition uniformly in $h$ and $P$.
Expanding \eqref{eq:moment-local-density} gives
$g(\ell)=\frac{\kappa}{\ell}+O_j(\ell^{-2})$ for $\ell\nmid M$, and $g(\ell)=\frac1\ell$ for $\ell\mid M$.
Thus, for sufficiently large $\ell$ in terms of $j$, $-\log(1-g(\ell))\le \kappa/\ell+O_j(\ell^{-2})$.
Mertens' estimate for the sum of reciprocal primes verifies \eqref{eq:beta-dimension} with $L=L_j$.
The finitely many smaller primes are controlled uniformly by
\eqref{eq:moment-small-primes}.
At a prime outside $M$ we also have $\frac{1-g(\ell)}{(1-1/\ell)^\kappa}
 =1+O_j(\ell^{-2})$.
The product of these ratios is bounded above and below by positive constants depending only on $j$.
Since $M=2hP$, it follows that
\begin{equation}
\label{eq:moment-sifting-product}
 V(y)
 \asymp_j(\log y)^{-\kappa}
 \prod_{\substack{\ell<y\\\ell\mid M}}
(1-1/\ell)^{1-\kappa}
 \ll_j
 \left(\frac{hP}{\varphi(hP)}\right)^\kappa
 (\log y)^{-\kappa}.
\end{equation}

Apply Lemma \ref{lem:upper-bound-sieve} with $X=|J|$ and use
\eqref{eq:moment-divisor-sum}. Equations
\eqref{eq:moment-crt-average} and \eqref{eq:moment-sifting-product}
give
$$
 S(\mathcal A,y)\ll_j
 \left(\frac{hP}{\varphi(hP)}\right)^\kappa
 \frac{|J|}{(\log y)^\kappa}
 +D(\log(2D))^{\kappa-1}.
$$
Finally, $\mathfrak M(z)\ll\log(2z)$ by Mertens' product estimate
\cite{MontgomeryVaughan2006}, while
$\log y=(2/s_j)\log z$ and $\kappa=j+1$.
Substitution in \eqref{eq:moment-sieve-majorant} yields
$$
 \sum_{\substack{q\in J\\q\text{ prime}}}E_z(m_q)^j
 \ll_j
 \left(\frac{hP}{\varphi(hP)}\right)^{j+1}
 \frac{|J|}{\log z}
 +z^2(\log(2z))^{2j}.
$$
This proves \eqref{eq:prime-euler-moments}.
If $y<2$, then $z$ is bounded in terms of $j$.
The bound $E_z(m)^j\le\mathfrak M(z)^j\ll_j1$ and the estimate
$\#(J\cap\mathbb Z)\le|J|+1$ prove the same assertion.

\label{prop:prime-shell-moments}
Fix a prime $p$ and a positive integer $j$. For $Q\ge3$ and every positive integer $k$ with $p^k\le(2Q)^4$, one has

\end{equation}
Here $\mathcal B_q(k)$ is defined by \eqref{eq:arithmetic-sums}.

<!-- remark-1: body_md -->
\label{rem:local}
Fix a nonnegative integer $r$. For all sufficiently large $q$, all vertices $x,y$ with $d(x,y)=r$, and all $c\in I$, one has
\begin{equation}
\label{eq:local-comparison}
 0\le\pi_{\alpha_{q,c},x}(y)-\mathcal M_p(\alpha_{q,c})u_{q,c}^r
 \ll_{p,I,r}\Delta_q(\alpha_{q,c}).
\end{equation}
Here $d$ is graph distance and $u_{q,c}$ is defined by \eqref{eq:alpha-u}.
The case $r=0$ is the definition of $\Delta_q$.
Suppose $r\ge1$.
Let $s_0=1$, $s_j=(p+1)p^{j-1}$ for $j\ge1$, and average over each distance sphere about $x$.
As $2r<g_q$ for large $q$, these spheres up to radius $r$ have the tree cardinalities.
Let $z=(p+1)/\alpha$ and let $h_j$ be the spherical average of $\pi_{\alpha,x}-\mathcal M_p(\alpha)u^{d(x,\cdot)}$.
The resolvent equation gives
$zh_0=(p+1)h_1$, and $zh_j=h_{j-1}+ph_{j+1}$ for $1\le j<r$.
At the source, $h_0=\Delta_q(\alpha)$.
The path polynomials in the proof of Lemma~\ref{lem:return-reduction} therefore give $s_rh_r=Q_r(z) \Delta_q(\alpha)$.
Each individual excess is nonnegative by the covering path expansion:
the unique path to a chosen lift at distance $r$ contributes $\mathcal M_p(\alpha)u^r$.
It is consequently at most $s_rh_r$.
For fixed $r$, $Q_r((p+1)/\alpha_{q,c})$ is bounded uniformly for $c\in I$, proving \eqref{eq:local-comparison}.
Thus the upper bounds in Theorem~\ref{thm:main-results}\textup{(i)}
and~\textup{(ii)}, the moment estimate in part~\textup{(iii)}, and
Corollary~\ref{cor:density-one} also hold, with constants depending on $r$,
for the largest excess at distance $r$ from any source.
