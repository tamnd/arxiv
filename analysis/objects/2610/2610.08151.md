<!-- proposition-1: body_md -->
\label{prop:vc-forbidden-trace}
Let $H$ be a set system on an $h$-element ground set, where $h\ge1$.  If
$\mathcal F$ is trace-$H$-free, then
\[
  \operatorname{VCdim}(\mathcal F)\le h-1.
\]
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
If an $h$-element set $S$ were shattered, every subset of $S$ would
occur as a trace of $\mathcal F$.  In particular, the traces would
contain a copy of $H$, contrary to the assumption.
<!-- end proof-1 -->
<!-- lemma-1: body_md -->
\label{lem:sauer}
Let $\mathcal F$ be a set system on an $M$-element ground set.  If
$\operatorname{VCdim}(\mathcal F)\le d$, then
\[
  |\mathcal F|\le\sum_{i=0}^d\binom Mi.
\]
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:one-inclusion}
If $\operatorname{VCdim}(\mathcal F)\le d$, then the one-inclusion graph of
$\mathcal F$ has at most $d|\mathcal F|$ edges.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:random-restriction}
Let $\mathcal F$ be a set system on an $M$-element ground set with
$\operatorname{VCdim}(\mathcal F)\le d$.  If $S$ is chosen uniformly at random from
the $s$-subsets of the ground set, where $0\le d\le s\le M$, then
\[
  \mathbb E\!\left[\left|\mathcal F|_S\right|\right]
  \ge |\mathcal F|\frac{(s)_d}{(M)_d}.
\]
<!-- end lemma-3 -->
<!-- theorem-1: body_md -->
\label{thm:polynomial-uniformity}
Let $H$ be a set system on $h\ge1$ vertices, and put $d=h-1$.
Let $\alpha\ge0$.  Assume that for every fixed integer $j\ge1$ there
is a constant $D_j$ such that
\begin{equation}\label{eq:fixed-uniformity-hypothesis}
  \operatorname{ex}_j(s,\operatorname{Tr}(H))
  \le D_j s^\alpha\qquad\text{for all }s\ge1.
\end{equation}
Then for every $\varepsilon>0$ there is a constant $C_{H,\alpha,\varepsilon}$
such that, for all integers $m\ge r\ge2$,
\[
  \operatorname{ex}_r(m,\operatorname{Tr}(H))
  \le C_{H,\alpha,\varepsilon}
       r^{d-\alpha+\varepsilon}m^\alpha.
\]
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
If $\alpha\ge d$, the Sauer--Shelah bound
(Lemma~\ref{lem:sauer}; see \cite{Sauer,Shelah,VapnikChervonenkis})
gives $\operatorname{ex}_r(m,\operatorname{Tr}(H))=O_H(m^d)$.
Since $m\ge r\ge2$,
\[
  m^d\le r^{d-\alpha+\varepsilon}m^\alpha,
\]
so the conclusion follows.  We therefore assume $0\le\alpha<d$.
Set $\delta=\varepsilon/(d-\alpha)>0$ and choose an integer $L$ with
\begin{equation}\label{eq:large-trace-cutoff}
  \delta L>d(1+\delta).
\end{equation}
Let $\mathcal G\subseteq\binom{[m]}r$ be trace-$H$-free, and write
$N=|\mathcal G|$.  Proposition~\ref{prop:vc-forbidden-trace} gives
$\operatorname{VCdim}(\mathcal G)\le d$.

First suppose $m>4d\,r^{1+\delta}$.  Choose uniformly at random
$S\in\binom{[m]}s$, where
\[
  s=\left\lfloor\frac{m}{r^{1+\delta}}\right\rfloor.
\]
Then $s\ge2d$ and $s/m\ge(2r^{1+\delta})^{-1}$.  If
$Z=\left|\mathcal G|_S\right|$ is the number of distinct traces, Lemma
\ref{lem:random-restriction} and
\eqref{eq:restriction-ratio} give
\begin{equation}\label{eq:general-trace-lower}
  \mathbb E Z
  \ge N\frac{(s)_d}{(m)_d}
  \ge 4^{-d}N r^{-d(1+\delta)}.
\end{equation}

Split $Z=Z_{<L}+Z_{\ge L}$ according to trace size.  Every level
of $\mathcal G|_S$ remains trace-$H$-free.  By
\eqref{eq:fixed-uniformity-hypothesis}, the levels of sizes
$1,\ldots,L-1$ contain at most $\sum_{j=1}^{L-1}D_j s^\alpha$
sets, and level zero contains at most one set.  Thus, for a constant
$K_L$ depending on the finitely many $D_j$,
\begin{equation}\label{eq:general-small-traces}
  Z_{<L}\le K_Ls^\alpha.
\end{equation}
For each $G\in\mathcal G$, if $L\le\min\{r,s\}$, then
\[
  \Pr(|G\cap S|\ge L)
  \le\mathbb E\binom{|G\cap S|}{L}
  =\binom rL\frac{(s)_L}{(m)_L}
  \le\frac1{L!}\left(\frac{rs}{m}\right)^L
  \le\frac{r^{-\delta L}}{L!}.
\]
If $L>\min\{r,s\}$, this probability is zero, so the same upper
bound holds.  Every distinct trace of size at least $L$ has such a preimage, so
\begin{equation}\label{eq:general-large-traces}
  \mathbb E Z_{\ge L}\le\frac{N}{L!}r^{-\delta L}.
\end{equation}
Combining \eqref{eq:general-trace-lower}--
\eqref{eq:general-large-traces} yields
\[
  4^{-d}N r^{-d(1+\delta)}
  \le K_Ls^\alpha+\frac{N}{L!}r^{-\delta L}.
\]
By \eqref{eq:large-trace-cutoff}, the final term can be absorbed
for all $r\ge r_0(H,\alpha,\varepsilon)$.  Since
$s\le m/r^{1+\delta}$, this gives
\begin{align*}
  N&\le 2\cdot4^dK_Lr^{d(1+\delta)}s^\alpha\\
   &\le 2\cdot4^dK_L
       r^{(d-\alpha)(1+\delta)}m^\alpha
    =2\cdot4^dK_L r^{d-\alpha+\varepsilon}m^\alpha.
\end{align*}
For the finitely many $2\le r<r_0$, the hypothesis
\eqref{eq:fixed-uniformity-hypothesis} gives the same conclusion
after enlarging the constant.

Finally, if $m\le4d\,r^{1+\delta}$, Lemma~\ref{lem:sauer} gives
\[
  N\le\sum_{i=0}^{d}\binom mi
   =O_H(m^d)
   =O_{H,\alpha,\varepsilon}
       \bigl(r^{(1+\delta)(d-\alpha)}m^\alpha\bigr).
\]
Because $(1+\delta)(d-\alpha)=d-\alpha+\varepsilon$, this is the
claimed bound in the remaining range.

<!-- proof-3: body_md -->
We use the following facts:
\begin{enumerate}
  \item There is a $C_4$-free graph $G$ on $N$ vertices with
  $\left(\frac16+o(1)\right)N^{3/2}$ triangles as $N\to\infty$
  \cite[Theorem~4.9, with $t=2$]{AlonShikhelman}.
  \item On three disjoint ordered sets of size $r$, the family
  \[
    \bigl\{\{u_1,\ldots,u_a,v_1,\ldots,v_b,w_1,\ldots,w_c\}:
    1\le a,b,c\le r\bigr\}
  \]
  is trace-$C_4$-free.  Indeed, among any four vertices, two belong
  to the same ordered set, and their nested membership prevents a
  trace $C_4$.  This family is the product of three chains.
\end{enumerate}
Put $N=\lfloor m/r\rfloor$ and label $rN$ of the ambient vertices
by $(v,t)$, where $v\in[N]$ and $t\in[r]$.  The remaining vertices
belong to no hyperedge.  Choose $G$ as above with vertex set $[N]$,
and write $v_t=(v,t)$.
For each triangle $\{u,v,w\}$ of $G$, with $u<v<w$, include in
$\mathcal H$ all hyperedges of the form
$\{u_1,\dots,u_a,v_1,\dots,v_b,w_1,\dots,w_c\}$,
where $a,b,c\ge1$ and $a+b+c=r$.

To see that the construction is trace-$C_4$-free, suppose that four
vertices support a trace $C_4$.  No two of them can have the same
label $v$: if $i<j$, every hyperedge containing $(v,j)$ also contains
$(v,i)$, whereas a $C_4$ has an edge containing $(v,j)$ but not $(v,i)$.
Thus the four vertices have distinct labels, say $x_1,x_2,x_3,x_4$
in cyclic order.  Each consecutive pair of labels belongs to the
triangle defining a hyperedge that witnesses the corresponding
trace edge, so it is an edge of $G$.  These four pairs form a $C_4$
in $G$, a contradiction.

Finally, each triangle contributes exactly $\binom{r-1}{2}$
hyperedges, one for each positive integer solution of $a+b+c=r$.
Each hyperedge uniquely determines its supporting triangle and its
three prefix lengths, so no hyperedge is counted twice.  Writing $\tau(G)$
for the number of triangles in $G$, we obtain, for fixed $r$ as
$m\to\infty$,
\begin{align*}
  |\mathcal H|
  &=\binom{r-1}{2}\tau(G)\\
  &=\binom{r-1}{2}\left(\frac16+o(1)\right)
    \left\lfloor\frac mr\right\rfloor^{3/2}\\
  &=\left(\frac{(r-1)(r-2)}{12r^{3/2}}+o_r(1)\right)m^{3/2}.
\end{align*}
This proves the claimed lower bound.  Its coefficient is asymptotic
to $\sqrt r/12$ as $r\to\infty$, and hence is unbounded.

<!-- proof-4: body_md -->
We induct on the size $M$ of the ground set.  The assertion is
immediate when $M=0$ or $d=0$.  Fix a ground-set element $x$ and write
\[
  \mathcal F_0:=\{A\in\mathcal F:x\notin A\},
  \qquad
  \mathcal F_1:=\{A\setminus\{x\}:A\in\mathcal F,\ x\in A\}.
\]
Both families are regarded as set systems on the remaining $M-1$
elements.  Put
\[
  \mathcal U:=\mathcal F_0\cup\mathcal F_1,
  \qquad
  \mathcal I:=\mathcal F_0\cap\mathcal F_1.
\]
Then $|\mathcal F|=|\mathcal U|+|\mathcal I|$.  Restriction cannot
increase VC dimension, so $\operatorname{VCdim}(\mathcal U)\le d$.  Moreover,
$\operatorname{VCdim}(\mathcal I)\le d-1$: if $\mathcal I$ shattered a set $Y$ of
size $d$, then $\mathcal F$ would shatter $Y\cup\{x\}$.

There are exactly $|\mathcal I|$ one-inclusion edges in the $x$
direction.  For any other coordinate, an edge may occur in one or in
both of the two layers $\mathcal F_0$ and $\mathcal F_1$.  Its first
occurrence is represented by an edge of the one-inclusion graph of
$\mathcal U$; if it occurs in both layers, it is also an edge of the
one-inclusion graph of $\mathcal I$.  Consequently,
\[
  e(G_1(\mathcal F))
  \le e(G_1(\mathcal U))+e(G_1(\mathcal I))+|\mathcal I|.
\]
Applying the induction hypothesis to $\mathcal U$ and $\mathcal I$
gives
\[
  e(G_1(\mathcal F))
  \le d|\mathcal U|+(d-1)|\mathcal I|+|\mathcal I|
  =d|\mathcal F|,
\]
as required.

<!-- proof-5: body_md -->
Suppose first that a family $\mathcal G$ of VC dimension at most $d$
is supported on a $q$-element ground set $Q$, where $q\ge1$.
If a uniformly random ground-set element is
deleted, the number of distinct members of $\mathcal G$ that are lost
is the number of one-inclusion edges in the direction of that element.
Every one-inclusion edge has exactly one direction.  Lemma
\ref{lem:one-inclusion} therefore gives
\[
  \mathbb E\!\left[\left|\mathcal G|_{Q\setminus\{x\}}\right|\right]
  =|\mathcal G|-\frac{e(G_1(\mathcal G))}{q}
  \ge \left(1-\frac dq\right)|\mathcal G|,
\]
where $x$ is uniform in $Q$.

Starting with $\mathcal F$, delete elements successively and uniformly
until $s$ remain.  At every stage the restricted family still has VC
dimension at most $d$, and the final set is a uniformly random
$s$-subset.  Iterating the preceding inequality yields
\[
  \mathbb E\!\left[\left|\mathcal F|_S\right|\right]
  \ge |\mathcal F|\prod_{q=s+1}^{M}\left(1-\frac dq\right)
  =|\mathcal F|\frac{(s)_d}{(M)_d}.
\]
