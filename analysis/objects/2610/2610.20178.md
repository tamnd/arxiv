<!-- theorem-1: body_md -->
\label{thm:fractional-stam}
Let $X_1,\ldots,X_n$ be Boolean independent self-adjoint random variables with symmetric laws and let $[n]=\{1,\ldots,n\}.$ For any $A\subseteq[n]$, set $S_A=\sum_{i\in A}X_i.$ Let $\mathcal C$ be a collection of nonempty subsets of $[n]$, and let $(\beta_A)_{A\in\mathcal C}$ be a fractional partition, namely
$$
\beta_A\geq 0,
\qquad
\sum_{A\in\mathcal C:\,i\in A}\beta_A=1
\quad\text{for every }i=1,\ldots,n.
$$
Then
\begin{equation}\label{eq:fractional stam}
    \frac{1}{\Psi(S_{[n]})}\geq\sum_{A\in\mathcal C}\frac{\beta_A}{\Psi(S_A)}.
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:stieltjes}
For every $\rho\in\mathcal P(\mathbb{R}_+)$,
\begin{equation}\label{eq:stieltjes}
    \Psi^{(2)}(\rho)
 =\int_0^\infty h_\rho(s)^2\,ds
 =\int_0^\infty\frac{ds}{(s+r_\rho(s))^2}.
\end{equation}
Moreover, $r_\rho$ is nonnegative and nondecreasing on $(0,\infty)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $x,y>0$, the elementary identity
\[
 \frac{\log x-\log y}{x-y}
 =\int_0^\infty\frac{ds}{(x+s)(y+s)}
\]
allows us, by Fubini's theorem, to obtain the first equality in \eqref{eq:stieltjes} and hence the representation in terms of $r_\rho$. Since $h_\rho(s)\le s^{-1}$, we have $r_\rho(s)\ge0$. Finally, differentiation gives
\[
 r_\rho'(s)
 =\frac{\displaystyle\int (x+s)^{-2}\,d\rho(x)}{\displaystyle\left(\int (x+s)^{-1}\,d\rho(x)\right)^2}-1\ge0
\]
by Cauchy--Schwarz.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:boolean-power}
For any $\rho\in\mathcal P(\mathbb{R}_+)$ and $c\ge1$, we have
\begin{equation}
    \Psi^{(2)}(\rho^{\uplus c})\le\frac1c\Psi^{(2)}(\rho).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since $K_{\rho^{\uplus c}}=cK_\rho$, we have $r_{\rho^{\uplus c}}(s)=cr_\rho(s)$. Therefore, by Lemma \ref{lem:stieltjes} and the change of variables $s=ca$,
\[
 \Psi^{(2)}(\rho^{\uplus c})
 =\int_0^\infty\frac{ds}{(s+cr_\rho(s))^2}
 =\frac1c\int_0^\infty\frac{da}{(a+r_\rho(ca))^2}.
\]
The monotonicity of $r_\rho$ gives $r_\rho(ca)\ge r_\rho(a)$, proving the claim.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:representation relative entropy}
    For any $\mu\in\mathcal{P}(\mathbb{R})$ with $m_2(\mu)=1$ and $\Gamma(\mu)>-\infty$, we have
    \begin{equation}\label{eq:relative-entropy-integral}
        \Gamma(\mu\mid \mathbf{b}) = \frac{1}{2}\int_0^\infty \left(h_{\mu^{(2)}}(s)-\frac{1}{1+s}\right)\,ds
    \end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Recall that if $m_2(\mu)=1$, one has
$$
\Gamma(\mu\mid\mathbf b) =-\Gamma(\mu),
\qquad
\Gamma(\mu)=\int_{\mathbb R}\log|x|\,d\mu(x),
$$
For every $x\neq0$, note that the following identity holds
$$
\log|x|
=
\frac12\int_0^\infty
\left(\frac{1}{1+s}-\frac{1}{x^2+s}\right)\,ds,
$$
which follows from
$$
\int_0^\infty
\left(\frac{1}{s+a}-\frac{1}{s+b}\right)\,ds
=\log\frac{b}{a},
\qquad a,b>0.
$$
Therefore,
$$
\Gamma(\mu)
=
\frac12\int_{-\infty}^{+\infty}\int_0^\infty
\left(\frac{1}{1+s}-\frac{1}{x^2+s}\right)
\,ds\,d\mu(x).
$$
Since $\Gamma(\mu)>-\infty$, the interchange of the two
integrals is justified by splitting the integral into the regions
$|x|\leq1$ and $|x|>1$. Indeed,
$$
\int_0^\infty
\left|
\frac{1}{1+s}-\frac{1}{x^2+s}
\right|\,ds
=2|\log|x||,
$$
and $\log^+|x|$ is integrable whenever $\mu$ has finite second
moment, while $\Gamma(\mu)>-\infty$ guarantees the integrability
of $\log^-|x|$.
Hence
$$
\Gamma(\mu)
=
\frac12\int_0^\infty
\left(
\frac{1}{1+s}
-
\int_{\mathbb R}\frac{1}{x^2+s}\,d\mu(x)
\right)\,ds,
$$
which yields the desired formula.
<!-- end proof-3 -->
<!-- remark-1: body_md -->
The representation in Lemma \ref{lem:representation relative entropy} admits a useful interpretation in terms of the nonmicrostates Boolean Fisher information. Recall that for a probability measure $\mu$, the nonmicrostates Fisher information of $\mu$ is defined as
    $$
    \Psi^*(\mu)=\int_{\mathbb{R}} \frac{1}{x^2}\,d\mu(x).
    $$
    Now let $(\tilde{\mu}_t)_{t\geq0}$ be the flow defined by
    $$
    \tilde{\mu}_t=(T_t)_\#\mu,
    \quad T_t(x)=\operatorname{sgn}(x)\sqrt{x^2+t}.
    $$
    That is, $\tilde{\mu}_t^{(2)}=(x\mapsto x+t)_\#\mu^{(2)},$ and hence
    $$
    \Psi^*(\tilde{\mu}_t)=\int_{\mathbb R}\frac{1}{x^2}\,d\tilde{\mu}_t(x)=\int_0^\infty \frac{1}{x+t}\,d\mu^{(2)}(x)=h_{\mu^{(2)}}(t).
    $$
    Moreover,
    $$
    \frac{d}{dt}\Gamma(\tilde{\mu}_t)=\frac12\Psi^*(\tilde{\mu}_t).
    $$
    Indeed, $T_t'(x)=1/(2T_t(x))$, and therefore
    $$
    \frac{d}{dt}\log|T_t(x)|=\frac{1}{2T_t(x)^2}.
    $$
    This should be compared with the nonmicrostates de Bruijn identity along the Boolean heat flow established in~\cite{CebronPan2026}.
<!-- end remark-1 -->
<!-- proof-4: body_md -->
First assume that $\Psi(X_i)<\infty$ for all $i=1,\ldots,n.$ By the Stam inequality \eqref{eq:stam ineq}, we have $\Psi(S_A)<\infty$ for all $A\subset [n]$. Set $\rho_i=\mathcal L(X_i^2),$ $\rho_A=\mathcal L(S_A^2)$. By \eqref{eq:square-convolution},
\begin{equation}\label{eq:r-additive}
    K_{\rho_A}=\sum_{i\in A}K_{\rho_i},
 \qquad
 r_{\rho_A}(s)=\sum_{i\in A}r_{\rho_i}(s).
\end{equation}
Thus Lemma \ref{lem:stieltjes} gives
\[
 \Psi(S_A)=\int_0^\infty
 \frac{ds}{\left(s+\sum_{i\in A}r_{\rho_i}(s)\right)^2}.
\]
Since each $r_{\rho_i}$ is nonnegative,
\begin{equation}\label{eq:monotone-psi}
    A\subseteq B\quad\Longrightarrow\quad \Psi(S_B)\le\Psi(S_A).
\end{equation}
Set
\[
 a_A=\frac1{\Psi(S_A)}>0,
 \qquad
 T=\sum_{A\in\mathcal C}\beta_Aa_A.
\]
If $a_A\ge T$ for some $A$, then \eqref{eq:monotone-psi} immediately gives
\[
 \frac1{\Psi(S_{[n]})}\ge a_A\ge T.
\]
We may therefore assume $a_A<T$ for every $A$. Define
\[
 c_A=\frac{T}{a_A}>1,
 \qquad
 w_A=\frac{\beta_Aa_A}{T}.
\]
Then $\sum_Aw_A=1$ and $w_Ac_A=\beta_A$.
Let $\tilde\rho_A=\rho_A^{\uplus c_A}$. By the definition of fractional partition and \eqref{eq:r-additive},
\[
 \sum_Aw_AK_{\tilde\rho_A}
 =\sum_A\beta_AK_{\rho_A}
 =\sum_{i=1}^m\left(\sum_{A\ni i}\beta_A\right)K_{\rho_i}
 =K_{\rho_{[n]}}.
\]
Since $K_\rho(z)=z-F_\rho(z)$ and $\sum_A w_A=1$, we obtain
\[
F_{\rho_{[n]}}(z)=\sum_A w_A F_{\tilde\rho_A}(z).
\]
Evaluating at $z=-s$ for $s>0$, we have
$F_{\rho_{[n]}}(-s)<0$ and $F_{\tilde\rho_A}(-s)<0$ for every $A$. Since $x\mapsto x^{-2}$ is convex on $(-\infty,0)$, Jensen's inequality therefore gives
\[
 G_{\rho_{[n]}}(-s)^2
 \le\sum_Aw_AG_{\tilde\rho_A}(-s)^2.
\]
Integrating with respect to $s$ and applying Lemma \ref{lem:stieltjes} and Lemma \ref{lem:boolean-power},
\[
 \Psi(S_{[n]})
 \le\sum_Aw_A\Psi^{(2)}(\tilde\rho_A)
 \le\sum_A\frac{w_A}{c_A}\Psi^{(2)}(\rho_A)
 =\frac1T.
\]
Taking reciprocals yields the desired inequality.

If \(\Psi(X_i)=\infty\) for some \(i\in[n]\), we regularize the corresponding law by setting \(\rho_i(\varepsilon):=\rho_i\uplus\delta_\varepsilon\). Then \(r_{\rho_i(\varepsilon)}(s)=r_{\rho_i}(s+\varepsilon)\). Applying the previous argument to the perturbed laws and then by Lemma \ref{lem:stieltjes}, letting \(\varepsilon\downarrow0\), the monotone convergence theorem completes the proof.

For any $s>0$, set
\[
 h_t(s)=\int_{\mathbb R}\frac{d\mu_t(x)}{x^2+s}=-\mathrm{Im} \frac{1}{\sqrt{s}}G_{\mu_t}(i\sqrt{s}).
\]
Fix $s>0$. Differentiating with respect to $t$ and using the semigroup relation \eqref{eq:semigroup} gives
\begin{equation}\label{eq:ht-derivative}
    t\partial_th_t(s)
 =-\bigl(\mathrm{Re}\,G_{\mu_t}(i\sqrt s)\bigr)^2
 -s\left(\int_{\mathbb R}\frac{d\mu_t(x)}{(x^2+s)^2}-h_t(s)^2\right).
\end{equation}
The second term on the right-hand side is nonnegative, and we deduce that $t\partial_th_t(s)\le 0$. Moreover,
$$
\operatorname{Re}G_{\mu_t}(i\sqrt s)
=
-\int_{\mathbb R}\frac{x}{x^2+s}\,d\mu_t(x).
$$
Therefore, by Cauchy-Schwarz,
$$
\begin{aligned}
\bigl(\operatorname{Re}G_{\mu_t}(i\sqrt s)\bigr)^2
&\leq
\int_{\mathbb R}\frac{x^2}{(x^2+s)^2}\,d\mu_t(x)\\
&=
\int_{\mathbb R}
\left(
\frac{1}{x^2+s}
-\frac{s}{(x^2+s)^2}
\right)d\mu_t(x)\\
&=
h_t(s)-s\int_{\mathbb R}\frac{d\mu_t(x)}{(x^2+s)^2}.
\end{aligned}
$$
Substituting this inequality into \eqref{eq:ht-derivative} yields
$$
-h_t(s)\le t\partial_t h_t(s)\le0.
$$
The right-hand inequality implies that $t\mapsto h_t(s)$ is decreasing. For the left-hand inequality, we observe that
$
\partial_t\bigl(t h_t(s)\bigr)
=h_t(s)+t\partial_t h_t(s)\ge0.
$
Thus $t\mapsto t h_t(s)$ is increasing, which implies that
$t h_t(s)\ge h_1(s),$ and consequently,
\begin{equation}\label{eq:ht-bounds}
    \frac1t h_1(s)\le h_t(s)\le h_1(s),\quad t\ge1.
\end{equation}

For $\varepsilon>0$, introduce the regularized logarithmic integral
\[
\Gamma_\varepsilon(\nu)=\frac12\int_{\mathbb R}\log(x^2+\varepsilon)\,d\nu(x).
\]
Tonelli's theorem gives,
\begin{equation}\label{eq:regularized-entropy}
    \Gamma_\varepsilon(\mu_t)-\Gamma(\mu_t)
 =\frac12\int_0^\varepsilon h_t(s)\,ds.
\end{equation}

Assume first that $\Gamma(\mu)>-\infty$. Taking $t=1$ in \eqref{eq:regularized-entropy} shows that $h_1$ is integrable on every interval $(0,\varepsilon)$. For fixed $s>0$, weak convergence $\mu_t\Rightarrow \mathbf{b}$ gives
\[

\]
The upper bound in \eqref{eq:ht-bounds} and dominated convergence therefore yield
\begin{equation}\label{eq:regularized-limit}
    \Gamma_\varepsilon(\mu_t)-\Gamma(\mu_t)
 \longrightarrow
 \frac12\int_0^\varepsilon\frac{ds}{1+s}
 =\frac12\log(1+\varepsilon).
\end{equation}
On the other hand, for sufficiently large $M>0$, decompose 
$$
\Gamma_\varepsilon(\mu_t)=\frac{1}{2}\int \log (x^2+\varepsilon)\wedge M\,d\mu_t(x)+\frac{1}{2}\int (\log (x^2+\varepsilon)-\log (x^2+\varepsilon)\wedge M)\,d\mu_t(x).
$$
Since the family $\{\mu_t\}_{t\ge1}$ has uniformly bounded second moments, it is uniformly integrable. Thus, the second term can be made uniformly arbitrarily small. The first term tends to $\Gamma_\varepsilon(\mathbf{b})$ by the weak convergence $\mu_t\Rightarrow \mathbf{b}$. Therefore, letting $M\to \infty$ yields
\[
\Gamma_\varepsilon(\mu_t)\longrightarrow\Gamma_\varepsilon(\mathbf{b})=\frac12\log(1+\varepsilon).
\]
Combining this with \eqref{eq:regularized-limit} proves $\Gamma(\mu_t)\longrightarrow0$.

Now assume $\Gamma(\mu)=-\infty$. Then \eqref{eq:regularized-entropy} at $t=1$ implies
\[

The lower bound in \eqref{eq:ht-bounds} gives
\[

 \ge \frac1t\int_0^\varepsilon h_1(s)\,ds
 =\infty.
\]
Since $\Gamma_\varepsilon(\mu_t)$ is finite, \eqref{eq:regularized-entropy} forces $\Gamma(\mu_t)=-\infty$.

Let $\rho=\mu^{(2)}$. Then $\rho$ is supported on $\mathbb{R}_+$, $\int x\,d\rho(x)=1$, and
\[
 \int(x-1)^2\,d\rho(x)=m_4(\mu)-1.
\]
For any $s>0$, set
\[
 h(s)=\int_0^\infty\frac{d\rho(x)}{s+x},
 \qquad
 r(s)=\frac1{h(s)}-s,
 \qquad
 q(s)=1-r(s).
\]
Jensen's inequality gives
\[
 \frac1{s+1}\le h(s)<\frac1s,
\]
so $0<r(s)\le1$ and $0\le q(s)<1$. By \eqref{eq:square-convolution},
\[
 \mu_n^{(2)}=D_{1/n}(\rho^{\uplus n}),
 \qquad
 K_{\mu_n^{(2)}}(z)=K_\rho(nz).
\]
Hence, for any $s>0$,
\[
 -G_{\mu_n^{(2)}}(-s)=\frac1{s+r(ns)}.
\]
Applying the change of variables $u = ns$ to the identity $\int_0^\infty (s+1)^{-2} \, ds = 1$, and setting $b_n(u) \coloneqq 1 + u/n$, we have
\begin{equation*}
    1 = \frac{1}{n} \int_0^\infty \frac{du}{b_n(u)^2}.
\end{equation*}
Substituting the identity in the integral representation of $\Psi(\mu_n)$ in Lemma~\ref{lem:stieltjes} yields
\begin{equation}\label{eq:psi-integral}
    \Psi(\mu_n) - 1 = \frac{1}{n} \int_0^\infty H_n(u) \, du,
\end{equation}
where $H_n(u) \coloneqq (b_n(u) - q(u))^{-2} - b_n(u)^{-2}$. A straightforward calculation gives
\begin{align}\label{eq:Hn}
    H_n(u) &= \frac{b_n(u)^2 - (b_n(u) - q(u))^2}{b_n(u)^2 (b_n(u) - q(u))^2} \nonumber \\
    &= q(u) \frac{2b_n(u) - q(u)}{b_n(u)^2 (b_n(u) - q(u))^2},
\end{align}
Recall that $b_4=m_4(\mu)-1.$ We first establish the following asymptotic behavior of $q$:
\begin{equation}\label{eq:q-asymptotic}
    \lim_{u\to \infty}uq(u)=b_4.
\end{equation}
Indeed, a rearrangement gives
\[
 q(u)=\frac{(u+1)h(u)-1}{h(u)},
\]
and by $\int(x-1)d\rho(x)=0$, we have
\[
 uq(u)
 =\frac{\displaystyle\int_0^\infty\frac{u(x-1)^2}{u+x}\,d\rho(x)}{\displaystyle(u+1)\int_0^\infty\frac1{u+x}\,d\rho(x)}.
\]
The dominated convergence theorem yields \eqref{eq:q-asymptotic}. If $b_4=0$, then $\rho=\delta_1$ and symmetry forces $\mu=\mathbf{b}$, so the conclusion is immediate. Assume henceforth $b_4>0$, and it remains to prove
\begin{equation}\label{eq:Hn-asymptotic}
    \frac{1}{n}\int_0^\infty H_n(u)\,du
 =2b_4\frac{\log n}{n}+o\left(\frac{\log n}{n}\right).
\end{equation}

We split the integral into a bounded part, an intermediate bulk, and a tail.

\medskip
\noindent\textbf{Bounded region.}
From \eqref{eq:Hn}, $0\le H_n(u)\le r(u)^{-2}.$ Moreover, $r(s)/s=(sh(s))^{-1}-1$ is decreasing in $s$, since $sh(s)$ is increasing in $s$. Thus for any $0<u\le M$,
\[
 r(u)\ge\frac{r(M)}M u,
 \qquad
 u+r(u)\le\left(1+\frac M{r(M)}\right)r(u).
\]
Since
\[
 \Psi(\mu)=\int_0^\infty\frac{du}{(u+r(u))^2}<\infty,
\]
we obtain
\begin{equation}\label{eq:bounded-Hn}
    \sup_{n\ge1}\int_0^M H_n(u)\,du<\infty.
\end{equation}

\medskip
\noindent\textbf{Bulk region.}
Fix $\eta\in(0,b_4)$. By \eqref{eq:q-asymptotic}, there exists some sufficiently large $M$ such that
\[
 \frac{b_4-\eta}{u}\le q(u)\le\frac{b_4+\eta}{u}
\qquad (u\ge M).
\]
For $M\le u\le\delta n$, write
\[
 H_n(u)=q(u)\Phi(b_n(u),q(u)),
 \qquad
 \Phi(b,q)=\frac{2b-q}{b^2(b-q)^2}.
\]
Because $\Phi(1,0)=2$, one may first choose $M$ large and then $\delta>0$ small enough that
\[
 2-\eta\le \Phi(b_n(u),q(u))\le2+\eta
\]
uniformly on $[M,\delta n]$. Hence
\begin{equation}\label{eq:bulk-Hn}
    (2-\eta)(b_4-\eta)
 \log\frac{\delta n}{M}
 \le\int_M^{\delta n}H_n(u)\,du
 \le(2+\eta)(b_4+\eta)
 \log\frac{\delta n}{M}.
\end{equation}

\medskip
\noindent\textbf{Tail.}
Again by \eqref{eq:q-asymptotic}, there exists $C>0$ such that $q(u)\le C/u$ for $u\ge M$. For fixed $\delta>0$ and sufficiently large $n$, $u\ge\delta n$ implies $q(u)\le1/2$, and hence
\[
 H_n(u)\le\frac{8q(u)}{(1+u/n)^3}
 \le\frac{8C}{u(1+u/n)^3}.
\]
Therefore
\begin{equation}\label{eq:tail-Hn}
    \int_{\delta n}^\infty H_n(u)\,du
 \le 8C\int_\delta^\infty\frac{dv}{v(1+v)^3}
 =O_\delta(1).
\end{equation}
Dividing \eqref{eq:bounded-Hn}, \eqref{eq:bulk-Hn}, and \eqref{eq:tail-Hn} by $ n$, then letting $n\to\infty$ and $\eta\downarrow0$, yields \eqref{eq:Hn-asymptotic}. Substitution into \eqref{eq:psi-integral} gives
\[
 \Psi(\mu_n)-1
 =2b_4\frac{\log n}{n}+o\!\left(\frac{\log n}{n}\right),
\]
as claimed.

Under the setup and notation of Example~\ref{ex:four-point}, the Boolean Fisher information of $\nu_{d,n}$ satisfies
\[
\Psi(\nu_{d,n}) - 1 = 2d^2 \frac{\log n}{n} + O\!\left(\frac1n\right).
\]
Using the explicit discrete representation of $\nu_{d,n}$ from Example~\ref{ex:four-point}, the Boolean Fisher information evaluates directly to
\[
\Psi(\nu_{d,n}) = \omega_{+,n}^2 \frac{1}{u_{+,n}} + 2 \omega_{+,n} \omega_{-,n} \frac{\log u_{+,n} - \log u_{-,n}}{u_{+,n} - u_{-,n}} + \omega_{-,n}^2 \frac{1}{u_{-,n}}.
\]
Substituting the asymptotic expansions of $u_{\pm,n}$, $\omega_{+,n}$, and $\omega_{-,n}$ as $n \to \infty$, we expand each term separately:
\begin{align*}
   \omega_{+,n}^2 \frac{1}{u_{+,n}} = \left(1 - \frac{2d^2}{n} + O(n^{-2})\right) \left(1 - \frac{d^2}{n} + O(n^{-2})\right) = 1 - \frac{3d^2}{n} + O(n^{-2}).
\end{align*}
\begin{align*}
   2 \omega_{+,n} \omega_{-,n} \frac{\log u_{+,n} - \log u_{-,n}}{u_{+,n} - u_{-,n}} 
   &= \left(\frac{2d^2}{n} + O(n^{-2})\right) \frac{\log n - \log(1-d^2) + O(n^{-1})}{1 + O(n^{-1})} \\
   &= 2d^2 \frac{\log n}{n} - \frac{2d^2 \log(1-d^2)}{n} + O\!\left(\frac{\log n}{n^2}\right).
\end{align*}
\begin{align*}
    \omega_{-,n}^2 \frac{1}{u_{-,n}} 
   &= \left(\frac{d^4}{n^2} + O(n^{-3})\right) \left(\frac{n}{1-d^2} + O(1)\right) = \frac{d^4}{(1-d^2)n} + O(n^{-2}).
\end{align*}
Summing these three contributions gives
\[
\Psi(\nu_{d,n}) = 1 + 2d^2 \frac{\log n}{n} + O\!\left(\frac1n\right).
\]
Using the identity \eqref{eq:relative fisher} yields the desired result.
