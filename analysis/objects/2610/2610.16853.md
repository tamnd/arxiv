<!-- maintheorem-1: body_md -->
\hypertarget{thm:main}{}
For all integers $k\geq l\geq1$ with $kl>1$, the Lawson surface
$\xi_{k,l}$ is not algebraic. More precisely, if
$P\in\mathbb{R}[X_1,X_2,X_3,X_4]$ vanishes on $\xi_{k,l}$, then there
exists $R\in\mathbb{R}[X_1,X_2,X_3,X_4]$ such that
\[
 P=(X_1^2+X_2^2+X_3^2+X_4^2-1)R.
\]

<!-- head-1: body_md -->
The minimal immersion determines an associated family of flat
$\operatorname{SL}(2,\mathbb{C})$-connections $\nabla^\lambda$, with spectral parameter
$\lambda\in\mathbb{C}^*$~\cite{Hitchin}. For the Lawson surfaces, symmetry
allows us to work with this family on the four-punctured sphere
$\mathcal S_4$~\cite{HellerHeller,HHS,BHS}.
Along an edge $E$ of a fundamental quadrilateral, the functions
$C,N$ encode the position vector and unit normal. They extend
meromorphically to the universal cover of $\mathcal S_4$.
If the Lawson surfaces were algebraic, then there would be a polynomial relation
between these functions.

The reflection principle (Proposition~\ref{prop:reflection}) relates
their analytic continuation to the monodromy of $\nabla^\lambda$.
Let $M_\gamma^\lambda$ be its monodromy matrix along a based loop
$\gamma$. Let $\delta$ be the antiholomorphic involution of
$\mathcal S_4$ induced by Schwarz reflection across $f(E)$,
and let $\delta\gamma$ be the reflected loop. If
continuation along $\gamma$ leaves $C,N$ unchanged, then
$M_\gamma^\lambda=M_{\delta\gamma}^\lambda$.
At the puncture corresponding to the chosen umbilic, this identity
determines the exact local branching order of $(C,N)$
(Corollary~\ref{cor:local-order}). Together with the Hopf differential, it lets us
show that an algebraic relation between $C,N$ makes both functions
algebraic over $\mathbb{C}(z)$, where $z$ is a coordinate on $\mathcal S_4$.
Their joint analytic continuation would
therefore have only finitely many branches.


Finitely many branches of $(C,N)$ would give a finite unbranched
cover of the Lawson surface. The reflection
identity and the nonabelian monodromy of the associated family
on this cover~\cite{Heller} would then force the local monodromy
around $p_1$ to have scalar square, contradicting its prescribed
eigenvalues. 

{\bf The paper is organized as follows.}
Section~\ref{sec:preliminaries} fixes the algebraicity convention
and recalls the Lawson surfaces and their integrable systems construction.
Section~\ref{sec:functions} treats the meromorphic functions $C,N$
and their algebraic relations. Section~\ref{sec:monodromy} contains
the reflection principle, the passage from algebraic dependence
to finite monodromy, and the proof of the \hyperlink{thm:main}{Main Theorem}.

\label{thm:monodromy-conditions}
The monodromy matrices of the associated family $\nabla^\lambda$
of the Lawson surface $\xi_{k,l}$ on $\mathcal S_4$ satisfy the
following properties~\cite{Manca,HellerHeller,HHS,BHS}.
\begin{enumerate}
\renewcommand{\textup{(\roman{enumi})}}{\textup{(\arabic{enumi})}}
\item For every $\lambda\in S^1$, the monodromy is unitary:
\[
 M_\gamma^\lambda\in\operatorname{SU}(2)
 \qquad\text{for all }\gamma\in\pi_1(\mathcal S_4,z_*).
\]
\item With the local degrees $n_j$ defined
in~\eqref{eq:quotient-local-degrees}, the local monodromy has eigenvalues
\[
 \operatorname{spec}(M_{\gamma_j}^\lambda)
 =\{e^{\pi i/n_j},e^{-\pi i/n_j}\},
 \qquad \lambda\in\mathbb{C}^*,\quad j=1,\ldots,4.
\]
\item At each Sym point $\lambda\in\{1,-1\}$, the four matrices
\[
 M_{\gamma_1}^\lambda,\quad M_{\gamma_2}^\lambda,\quad
 M_{\gamma_3}^\lambda,\quad M_{\gamma_4}^\lambda
\]
are simultaneously diagonalizable.
\end{enumerate}

\textbf{Construction of $V$.}
We use the bundle construction of~\cite[Lemma~5.4 and
Proposition~5.5]{BHS}. On the double cover
$\varpi\colon\widehat\Sigma\to\Sigma$ branched over
$\pi^{-1}(\{p_1,\ldots,p_4\})$, the immersion and its lifted
symmetries $\rho$ satisfy
\[
 f(\rho\widehat p)=a_\rho f(\widehat p)b_\rho^{-1},
 \qquad a_\rho,b_\rho\in\operatorname{SU}(2).
\]
Over $p_j$, choose $w$ with $z-p_j=w^{2n_j}$, where $n_j$
is defined in~\eqref{eq:quotient-local-degrees}.
The symmetries fixing $w=0$ are generated by $\rho_j$, which,
in a unitary basis of $\mathbb{C}^2$, satisfies
\[
 \rho_j(w)=\zeta_jw,\qquad
 b_{\rho_j}=\operatorname{diag}(\zeta_j,\zeta_j^{-1}),
 \qquad \zeta_j=e^{\pi i/n_j}.
\]
Using the local holomorphic frame $(e_+,e_-)$ from
\cite[Lemma~5.1 and proof of Proposition~5.2]{BHS}, write
$e_\pm=(e_{\pm,1},e_{\pm,2})$ and define
$\widehat V\to\widehat\Sigma$ by the transition matrices
\[
 g_{0j}(w)=
 \begin{pmatrix}
 w e_{+,1}(w)&w^{-1}e_{-,1}(w)\\
 w e_{+,2}(w)&w^{-1}e_{-,2}(w)
 \end{pmatrix},\qquad w\ne0.
\]
They satisfy $g_{0j}(\zeta_jw)=b_{\rho_j}g_{0j}(w)$. Define
\[
 V=\widehat V/\!\sim\ \longrightarrow\mathbb{CP}^1,
 \qquad (\widehat p,\widehat v_0)\sim
 (\rho\widehat p,\widehat v'_0).
\]
Here $\widehat v_0=(\widehat v_{0,1},\widehat v_{0,2})$ and
\[
 \widehat v'_{0,i}=\sum_{a=1}^2(b_\rho)_{ia}\widehat v_{0,a},
 \qquad i=1,2.\]
In the local coordinates
\[
 \widehat v_{0,i}=\sum_{a=1}^2(g_{0j}(w))_{ia}\widehat v_{j,a},
 \qquad i=1,2,
\]
the stabilizer acts by
$(w,\widehat v_j)\mapsto(\zeta_jw,\widehat v_j)$ and hence
trivially on the fiber. Thus $V$ is a rank-two bundle over $\mathbb{CP}^1$.

\textbf{(1) Unitarity.}
The identity $\rho^*\nabla^\lambda
=b_\rho\nabla^\lambda b_\rho^{-1}$ defines the associated family
on $V$ over $\mathcal S_4$. Since the original connections are
unitary for $\lambda\in S^1$ and $b_\rho\in\operatorname{SU}(2)$ preserves
the Hermitian metric, their monodromy is unitary, proving~\textup{(1)}.

\textbf{(2) Local monodromy.}
The logarithmic extensions have residue eigenvalues
$\pm1/(2n_j)$ for small nonzero $\lambda$~\cite[Proposition~5.5]{BHS},
giving local monodromy eigenvalues $e^{\pm\pi i/n_j}$.
The characteristic polynomial identity extends to all
$\lambda\in\mathbb{C}^*$ by holomorphic dependence, proving~\textup{(2)}.

\textbf{(3) Monodromy at the Sym points.}
At each Sym point, the monodromy matrices are, up to conjugation
and signs, the components $a_\rho$ or $b_\rho$ of the
lifted rotations~\cite[proof of Theorem~6]{HHS}.
These rotations lie in a torus of $\operatorname{SO}(4)$, whose inverse image
in $\operatorname{SU}(2)\times\operatorname{SU}(2)$ is also a torus.
Thus their lifts, and hence the monodromy matrices, commute.
By~\textup{(1)}, the matrices are unitary and therefore
simultaneously diagonalizable, proving~\textup{(3)}.

The power $\gamma_j^{n_j}$ lifts to a loop around a point above
$p_j$, which contracts in $\Sigma$ after the removed point is
filled in. By Lemma~\ref{thm:monodromy-conditions}\textup{(2)},

 &=(-1)^{a+b}\mathrm{Id}

\]

to a closed loop $\beta$ in $\Sigma^\circ$ exactly when its
corresponding product in the symmetry group defining $\pi$
is the identity.
The pullback of the associated family to $\Sigma^\circ$ agrees,
up to gauge, with the original family tensored with a fixed flat line

 \qquad\beta\in\pi_1(\Sigma^\circ,z_*).
\end{equation}
This sign twist leaves commutators of monodromy matrices unchanged.

<!-- proposition-1: body_md -->
\label{prop:potential}
For some $r_0>0$, the Lawson surface $\xi_{k,l}$ admits a Fuchsian
potential for $0<|\lambda|<r_0$ of the form
\begin{equation}\label{eq:potential}
 \eta^\lambda=\sum_{j=1}^4 A_j(\lambda)
                 \frac{\mathrm{d} z}{z-p_j},\qquad
 \sum_{j=1}^4 A_j(\lambda)=0.
\end{equation}
Here $A_j(\lambda)\in\mathfrak{sl}(2,\mathbb{C})$ is the residue matrix
at $p_j$, for $j=1,\ldots,4$. The potential has the following
properties.
\begin{enumerate}
\renewcommand{\textup{(\roman{enumi})}}{\textup{(\roman{enumi})}}
\item Each $\lambda A_j(\lambda)$ is holomorphic for
$|\lambda|<r_0$. The eigenvalues of $A_j(\lambda)$ are
$\pm\frac1{2n_j}$, independent of $\lambda$, where $n_j$ is
defined in~\eqref{eq:quotient-local-degrees}.
\item The leading coefficient
$\eta_{-1}=\lim_{\lambda\to0}\lambda\eta^\lambda$ is a nowhere
vanishing nilpotent section of
$\operatorname{End}(\mathcal O^2)\otimes K_{\mathbb{CP}^1}(p_1+\cdots+p_4)$,
the bundle of endomorphism-valued meromorphic one-forms with at most
simple poles at the four punctures of $\mathcal S_4$.
\item There is a positive gauge $B^\lambda$, single-valued and smooth
on $\mathcal S_4$, and holomorphic and invertible for
$|\lambda|<r_0$, such that
\begin{equation}\label{eq:global-positive-gauge}
 \mathrm{d}+\eta^\lambda=(\nabla^\lambda).B^\lambda,
 \qquad 0<|\lambda|<r_0.
\end{equation}
For sufficiently small $r$ with $0<r<\min\{r_0,1\}$, the $r$-Iwasawa
reconstruction from $\eta^\lambda$, with suitable initial data,
gives the Lawson immersion.
\end{enumerate}

For $l=1$, the potential is given by
\cite[Theorem~3]{HellerHeller}. For $l>1$, existence on the
quotient in Section~\ref{sec:lawson} follows from
\cite[Theorem~1.3]{Manca}; see also~\cite[Theorem~5.7]{BHS}.
We explain how the conventions in~\cite{Manca} give the extension
used here.

For $l>1$, the underlying holomorphic bundle is
$\mathcal O(-2)\oplus\mathcal O(-2)$ near $\lambda=0$
\cite[Section~4 and proof of Theorem~1.3\textup{(ii)}]{Manca}.
Tensor it with the fixed logarithmic connection on $\mathcal O(2)$
with residue $-1/2$ at each $p_j$. This gives
\[
 \bigl(\mathcal O(-2)\oplus\mathcal O(-2)\bigr)\otimes
 \mathcal O(2)=\mathcal O^2,\qquad
 \left(\frac12\pm\frac1{2n_j}\right)-\frac12
 =\pm\frac1{2n_j},
\]
where $n_j$ is defined in~\eqref{eq:quotient-local-degrees}.
In a flat trivialization of the pulled-back twisting line,
Manca's local frame changes by
\[
 w^{-n_j}(w^{n_j-1}s,w^{n_j+1}t)=(w^{-1}s,wt),
 \qquad z-p_j=w^{2n_j},
\]
where $(s,t)$ is the local holomorphic frame on $\widehat\Sigma$.
Up to exchanging the frame elements, this is the extension in
\cite[Proposition~5.2]{BHS}. The twisting line has monodromy $-1$
around each puncture, so it also adjusts the central signs to
our chosen lifts. Thus the resulting extension is the bundle $V$
constructed above.

The twist is independent of $\lambda$ and leaves the Higgs field
unchanged as an endomorphism-valued differential.
That it is nilpotent and has nonzero residues follows from
\cite[Proposition~4.2]{Manca}; away from the punctures it is
nonvanishing because $f$ is an immersion.
For $l=1$, the corresponding properties follow from
\cite{HellerHeller} and~\cite[proofs of Lemma~5.6 and
Theorem~5.7]{BHS}. This proves~\textup{(ii)}.

After shrinking $r_0$, choose a trivialization with determinant
one of the resulting holomorphic bundle family, holomorphic in
$\lambda$ and invertible also at $\lambda=0$. Comparing this trivialization
with the fixed smooth frame on $\mathcal S_4$ gives the
single-valued positive gauge $B^\lambda$.
The local gauges are regular at $\lambda=0$ by
\cite[proof of Proposition~5.5, equations~(5.20)--(5.21)]{BHS}.
Together with the residue computation above, this proves
\textup{(i)} and~\textup{(iii)}.

For the reconstruction, set
$\Psi^\lambda=(B^\lambda)^{-1}F^\lambda$ on
$\widetilde{\mathcal S_4}$ for $0<|\lambda|<r_0$.
Since $B^\lambda$ is single-valued, $\Psi^\lambda$ has the same
monodromy as $F^\lambda$:
\[
 \gamma^*\Psi^\lambda=\Psi^\lambda M_\gamma^\lambda.
\]
With the corresponding initial data at $z_*$, the reconstruction
in Section~\ref{sec:local-dpw} gives the Lawson immersion.

\label{prop:continuation}
The germs $C$ and $N$ extend to nonconstant meromorphic functions
on $\widetilde{\mathcal S_4}$. With suitable orientations of
the two circle coordinates, they satisfy
\begin{equation}\label{eq:differential-relation}

 C^\#=C^{-1},\qquad N^\#=N^{-1}.

$\gamma_j^{n_j}$ denotes the loop $\gamma_j$ repeated $n_j$ times.

<!-- proof-3: body_md -->
We first construct the meromorphic extensions of $C$ and $N$.
Choose $0<r<\min\{r_0,1\}$ sufficiently small. On the simply
connected surface $\widetilde{\mathcal S_4}$, the $r$-Iwasawa
decomposition~\eqref{eq:iwasawa}, applied pointwise to $\Psi$,
gives smooth factors
\[
 \Psi=B^{-1}F.
\]
Set $B_0=B^{\lambda=0}$. We use a two-circle factorization.
For a matrix-valued function $A(z,\lambda)$, define
$A^\#(z,\lambda)=\overline{A(\delta z,1/\overline\lambda)}^{\,T}$.
For $z\in J\subset\mathbb{R}$, we have $\delta z=\overline z=z$.
Thus, when $|\lambda|=1$, the operation $\#$ gives
$F^\#=F^*=F^{-1}$ by unitarity.
Since both sides are holomorphic in $\lambda$, the identity
$F^\#=F^{-1}$ extends to $r<|\lambda|<r^{-1}$ for $z\in J$.
We therefore define along $J$
\begin{equation}\label{eq:factorization-on-edge}
 P_0=B_0^*B,\qquad
 P_\infty=B_0^*(B^\#)^{-1},\qquad
 \mathcal F=B_0^*F.
\end{equation}
The inverse pair $(\Psi^{-1},\Psi^\#)$ has the common annular
factor $\mathcal F^{-1}$ on the left and the disk factors
$P_0,P_\infty$ on the right, as in
\cite[Section~2, Theorems~1--2 and Lemma~1]{DMPW}.
Inverting gives
\begin{equation}\label{eq:two-circles}
 \begin{aligned}
 \Psi&=P_0^{-1}\mathcal F&&\text{on }|\lambda|=r,\\
 (\Psi^\#)^{-1}&=P_\infty^{-1}\mathcal F
                  &&\text{on }|\lambda|=r^{-1},
 \end{aligned}
 \qquad P_\infty^{\lambda=\infty}=\mathrm{Id}.
\end{equation}
Here $P_0$ is holomorphic and invertible for $|\lambda|<r$,
and $P_\infty$ for $|\lambda|>r^{-1}$, including $\lambda=\infty$;
$\mathcal F$ is holomorphic for $r<|\lambda|<r^{-1}$.
The condition at $\lambda=\infty$ follows from
$(B^\#)^{\lambda=\infty}=B_0^*$ and makes the two-circle
factorization unique.
Since $\eta^\lambda$ is holomorphic on $\mathcal S_4$, its parallel
frame $\Psi$ is holomorphic in $z$ on
$\widetilde{\mathcal S_4}$. The inverse pair
$(\Psi^{-1},\Psi^\#)$ is therefore holomorphic in $z$ and admits
the factorization~\eqref{eq:two-circles} at $z_*$. Normalizing by
its factors at $z_*$, we apply~\cite[Section~2, Theorem~3]{DMPW}.
Thus all three factors $P_0,P_\infty,\mathcal F$ extend
meromorphically in $z$ to this universal covering. In particular, $\mathcal F^{\lambda=1}$ and
$\mathcal F^{\lambda=-1}$ are meromorphic in $z$.

By Proposition~\ref{prop:potential}(ii), near each puncture we can
write $\eta_{-1}=A(u)\mathrm{d} u/u$, where $A$ is holomorphic,
nilpotent and nonzero also at $u=0$. Thus $A$ has constant rank
one, and $L=\ker\eta_{-1}$ extends across the punctures as a
holomorphic line subbundle of $\mathcal O^2$ on $\mathbb{CP}^1$.
Every line bundle on $\mathbb{CP}^1$ is isomorphic to $\mathcal O(d)$
for some integer $d$, and hence admits a nonzero meromorphic
section. Choose such a section $v=(v_1,v_2)$ of $L$.
Define the meromorphic functions
\begin{equation}\label{eq:section-components}
 a_i=(P_0^{\lambda=0})_{i1}v_1+(P_0^{\lambda=0})_{i2}v_2,
 \qquad i=1,2.
\end{equation}
Define the meromorphic matrix
\[
 \mathcal P=\frac{1}{v_1^\#a_1+v_2^\#a_2}
 \begin{pmatrix}
 a_1v_1^\#&a_1v_2^\#\\
 a_2v_1^\#&a_2v_2^\#
 \end{pmatrix}.
\]
On $J$, define the functions
\begin{equation}\label{eq:edge-components}
 k_i=(B_0)_{i1}v_1+(B_0)_{i2}v_2,\qquad i=1,2.
\end{equation}
Then
\begin{equation}\label{eq:kernel-projection}
 \mathcal P=B_0^*\Pi(B_0^*)^{-1},\qquad
 \Pi=\frac{1}{|k_1|^2+|k_2|^2}
 \begin{pmatrix}
 |k_1|^2&k_1\overline{k_2}\\
 k_2\overline{k_1}&|k_2|^2
 \end{pmatrix},
\end{equation}
where $\Pi$ is the orthogonal projection onto
$B_0\ker\eta_{-1}$.
Equation~\eqref{eq:sym-formula} and the corresponding normal formula
\cite[Theorem~14.1, equations~(14.10)--(14.11)]{Bobenko}, expressed in the
meromorphic frame $\mathcal F$, give
\begin{equation}\label{eq:meromorphic-reconstruction}
 f=(\mathcal F^{\lambda=1})^{-1}\mathcal F^{\lambda=-1},\qquad
 \nu=\pm i(\mathcal F^{\lambda=1})^{-1}(2\mathcal P-\mathrm{Id})\mathcal F^{\lambda=-1}
\end{equation}
along $J$, with the sign fixed by the normal orientation.
Here $i(2\mathcal P-\mathrm{Id})$ represents the normal matrix
$i\operatorname{diag}(1,-1)$ after changing from a unitary
frame adapted to $B_0L$ to $\mathcal F$.
Both Sym points $\pm1$ belong to the annulus. These expressions are
therefore meromorphic in $z$, and their denominator is positive
on $J$ where $v$ is finite and nonzero. Hence the linear coordinates
\[
 C=\langle f,e_0\rangle+i\langle f,e_1\rangle,\qquad
 N=\langle\nu,e_2\rangle+i\langle\nu,e_3\rangle
\]
extend meromorphically as well.

We next verify the differential and reflection identities.
Write $z=x+iy$ and orient $e_3$
so that $f_y=\ell_x(-\sin\beta\,e_2+\cos\beta\,e_3)$ on $J$.
Differentiating $\langle f_y,\nu\rangle=0$ gives
$\langle f_{xy},\nu\rangle=-\langle f_y,\nu_x\rangle
=-\ell_x\beta_x$. By the computation following~\eqref{eq:hopf}
and the definitions $C=e^{i\ell}$, $N=e^{i\beta}$, we obtain
\[
 \begin{aligned}
 Q&=-\frac{i}{2}\langle f_{xy},\nu\rangle\mathrm{d} z^2
   =\frac{i}{2}\ell_x\beta_x\,\mathrm{d} z^2,\\
 \mathrm{d}\log C&=i\ell_x\,\mathrm{d} z,\qquad
 \mathrm{d}\log N=i\beta_x\,\mathrm{d} z,\\
 \mathrm{d}\log C\,\mathrm{d}\log N&=-\ell_x\beta_x\,\mathrm{d} z^2=2iQ
 \end{aligned}
\]
along $J$, and hence everywhere by meromorphic continuation.
Since $Q\ne0$, both functions are nonconstant. The
identities~\eqref{eq:scalar-reflection} likewise follow from
$|C|=|N|=1$ on $J$.

Finally, we examine analytic continuation around the punctures.

<!-- proof-6: body_md -->
Let $S$ be an algebraic surface containing an open arc of a great
circle. Stereographic projection from a point of the great circle outside
the arc maps the circle to a straight line. After a Euclidean
change of coordinates and restriction to a smaller arc, the image
of $S$ is a real analytic graph
\[
 s=\chi(x,y),\qquad \chi(x,0)=0,
\]
with the arc on the $x$-axis. Since stereographic projection and
its inverse are rational, a polynomial as in
Definition~\ref{def:algebraic} gives, after clearing denominators,
a nonzero polynomial $R\in\mathbb{R}[x,y,s]$ satisfying
$R(x,y,\chi(x,y))=0$. It is nonzero because the original polynomial
does not vanish identically on $\mathbb{S}^3$.

Write $R=\sum_{j=m}^d R_j$, where $R_j$ is homogeneous of degree
$j$ in $(y,s)$ and $R_m\ne0$. Since $R(x,0,0)=0$, we have
$m\ge1$. Set $a(x)=\chi_y(x,0)$. Substituting
$\chi(x,y)=a(x)y+O(y^2)$ and comparing the coefficient of $y^m$ gives
\[
 R_m(x,1,a(x))=0.
\]
The polynomial $R_m(x,1,T)$ is nonzero, since $R_m$ is nonzero
and homogeneous in $(y,s)$. Hence $a$ is algebraic over $\mathbb{C}(x)$.
The Euclidean unit normal along the arc is
\[
 \nu_{\mathrm E}(x)=\pm\frac{(0,-a(x),1)}{\sqrt{1+a(x)^2}},
\]
so its components are algebraic over $\mathbb{C}(x)$ as well.

Let $\varphi$ be the inverse stereographic projection in these
coordinates, with $\varphi^*g_{\mathbb{S}^3}=\rho^2g_{\mathbb{R}^3}$.
Both $\varphi$ and $\rho>0$ are rational, and the spherical unit
normal is
\[
 \nu=\rho^{-1}\mathrm{d}\varphi(\nu_{\mathrm E}).
\]
Thus the components of $\nu$ along the arc, and in particular
$N=\langle\nu,e_2\rangle+i\langle\nu,e_3\rangle$, are algebraic
over $\mathbb{C}(x)$. On the great circle, stereographic projection
expresses $x$ as a nonconstant rational function of $C$, since
\[
 \cos\ell=\frac{C+C^{-1}}{2},\qquad
 \sin\ell=\frac{C-C^{-1}}{2i}.
\]
Substituting this expression for $x$ into a polynomial equation
for $N$ over $\mathbb{C}(x)$ and clearing denominators gives
$A(C,N)=0$ with $A\in\mathbb{C}[U,V]$. The polynomial $A$ is nonzero
because $x$ is a nonconstant rational function of $C$.
The identity extends along the arc by real analyticity and to
the holomorphic germs by the identity theorem.

 =M_{\delta\rho}^{-1}M_{\gamma_1}^{-m}M_{\delta\rho}.

At $\lambda=1$, both $M_\rho^{\lambda=1}$ and $M_{\delta\rho}^{\lambda=1}$
commute with $M_{\gamma_1}^{\lambda=1}$ by
Lemma~\ref{lem:potential-monodromy}. The displayed identity
therefore reduces to
\[
 (M_{\gamma_1}^{\lambda=1})^m=(M_{\gamma_1}^{\lambda=1})^{-m}.
\]
Since the eigenvalues of $M_{\gamma_1}^{\lambda=1}$ are $e^{\pm\pi i/n_1}$,
we obtain $e^{2\pi i m/n_1}=1$, so $n_1$ divides $m$.
Proposition~\ref{prop:continuation} shows that continuation along
$\gamma_1^{n_1}$ leaves every branch unchanged, proving the assertion.

\label{prop:algebraic-parameter}
Let $C$ and $N$ be the meromorphic functions associated with the Lawson
surface $\xi_{k,l}$ in Section~\ref{sec:continuation}. If $C$ and $N$
are algebraically dependent over $\mathbb{C}$, then each is algebraic over
$\mathbb{C}(z)$. Thus there exist nonzero polynomials
$A_C,A_N\in\mathbb{C}[z,T]$ such that
\[
 A_C(z,C)=0,\qquad A_N(z,N)=0.
\]
In particular, the analytic continuation of $(C,N)$ has only
finitely many branches.

\textbf{Construction of the algebraic curve.}
Choose an irreducible polynomial $A\in\mathbb{C}[U,V]$ satisfying
$A(C,N)=0$. Let $Y$ be the normalization of the projective closure
in $\mathbb{CP}^2$ of the affine curve $A(U,V)=0$. Let $C_Y,N_Y$
denote the meromorphic functions on $Y$ obtained by pulling back
the rational coordinate functions $U,V$. Choose a small disk
$D_*\subset\mathcal S_4$ around $z_*$ on which the initial
functions $C,N$ are holomorphic. The nonconstant map
$z\mapsto(C(z),N(z))$ on $D_*$ lifts to a holomorphic map
$\iota:D_*\to Y$, with
\[
 C_Y(\iota(z))=C(z),\qquad N_Y(\iota(z))=N(z).
\]
For the remainder of the proof, we abbreviate $C_Y,N_Y$ by $C,N$.

Let $\pi_X:X\to Y$ be a connected cover of degree at most two on
which there is a meromorphic one-form $\alpha$ satisfying
\begin{equation}\label{eq:algebraic-square-root}
 \alpha^2=\frac{1}{2iq_*}\,\mathrm{d}\log C\,\mathrm{d}\log N
 =\frac{Q}{q_*}.
\end{equation}
Here $C,N$ denote their pullbacks from $Y$ under $\pi_X$;
the equality with $Q/q_*$ holds on the initial branch
by~\eqref{eq:differential-relation}.
If the square root is already well-defined on $Y$, we take $X=Y$.
After shrinking $D_*$, choose a holomorphic square root
$\sqrt{p(z)}$ on $D_*$ and a local lift $\iota_X:D_*\to X$
of $\iota$, so $\pi_X\circ\iota_X=\iota$.
We choose the sign of $\alpha$ so that
\[
 \iota_X^*\alpha=\frac{\mathrm{d} z}{\sqrt{p(z)}}.
\]

The compactification of $y^2=p(z)$ is an elliptic curve
$\mathcal E$, and $\mathrm{d} z/y$ is a nowhere vanishing holomorphic
differential on it. Uniformize $\mathcal E$ by $\mathbb{C}/\Lambda$ so
that $\mathrm{d} z/y=\mathrm{d} u$, and write
$z=R(u)$. Then $R$ is a meromorphic function of degree two on
$\mathbb{C}/\Lambda$, and $(R')^2=p(R)$. Choose $u_1\in\mathbb{C}$ with
$R(u_1)=p_1$. The involution of the double cover is
$u\mapsto2u_1-u$, so its fixed points give
\begin{equation}\label{eq:elliptic-branch-points}
 R^{-1}(\{p_1,p_2,p_3,p_4\})=u_1+\tfrac12\Lambda,\qquad
 R^{-1}(p_1)=u_1+\Lambda
\end{equation}
on $\mathbb{C}$. At these points $R$ has local degree two; elsewhere
it is locally invertible as a map to $\mathbb{CP}^1$.

Remove the zeros and poles of $\alpha$ from $X$ and denote the
result by $X^\circ$. Since $\iota_X^*\alpha$ is holomorphic and
nonzero on $D_*$, the image of $\iota_X$ lies in $X^\circ$.
The disk $D_*$ is simply connected, so $\iota_X$ lifts to a map
$\widetilde\iota:D_*\to\widetilde{X^\circ}$.
On $\widetilde{X^\circ}$, we use $C,N$ for the
pullbacks of these functions on $Y$ under the composition
\[
 \widetilde{X^\circ}\longrightarrow X^\circ
 \xrightarrow{\pi_X}Y,
\]
and $\alpha$ for the pullback of the one-form on $X^\circ$
under the universal covering map. Integration gives a holomorphic
function $U$ with $\mathrm{d} U=\alpha$.
Choose its additive constant so that
\[
 R\bigl(U(\widetilde\iota(z))\bigr)=z\qquad(z\in D_*).
\]
Continuing this local identification along $I$, we have
$U\to u_1$ as $z\to p_1$. On $\Sigma$, this limit corresponds
to the endpoint $x_1$ of the chosen lifted edge, as in
Lemma~\ref{lem:vertex}. The set
\[
 \Omega=\widetilde{X^\circ}\setminus
 U^{-1}(u_1+\tfrac12\Lambda)
\]
is connected, since the removed set is closed and discrete.
The map $R\circ U\colon\Omega\to\mathcal S_4$ is holomorphic
and locally invertible. Every local branch of $C,N$ obtained from a point
of $\Omega$ is therefore an analytic continuation of the initial
branch.
Each such branch admits meromorphic continuation along paths in
$\mathcal S_4$. By Proposition~\ref{prop:continuation}, analytic
continuation along $\gamma_j^{n_j}$ leaves both germs unchanged.
Thus the local monodromy of $(C,N)$ around any $p_j$ has order
dividing $n_j$. By Corollary~\ref{cor:local-order}, its order
around $p_1$ is exactly $n_1$ on every branch.

\textbf{Restriction of the periods.}
We show that all periods of $\alpha$ belong to
$\tfrac12\Lambda$. Let $w$ be the coordinate centered at
$x_1\in\Sigma$ in Lemma~\ref{lem:vertex}. The holomorphic map
$w\mapsto(C(w),N(w))$ takes values in the algebraic curve defining
$Y$ and lifts uniquely to a holomorphic map into its normalization
$Y$. Denote the image of $w=0$ under this lift by $a_1\in Y$, and
put $C_0=C(a_1)=C(0)$. The lifted map is unramified at $w=0$: its
local degree divides the vanishing order of $C-C_0$, which
is one. Thus $w$ is a local coordinate on $Y$ at $a_1$, and the
quadratic differential in~\eqref{eq:algebraic-square-root} has
order $n_1-2$ there. Choose the point of $X$ above $a_1$ determined
by the initial branch. If $n_1$ is even, the cover $X\to Y$ is
unramified at this point and $\alpha$ has order $(n_1-2)/2$.
If $n_1$ is odd, a coordinate $\xi$ on $X$ satisfies $w=\xi^2$,
and $\alpha$ has order $n_1-1$. Consequently, in a local coordinate
$\xi$ on $X$, the local primitive $U$ of $\alpha$ and the function
$C$ have the expansions
\begin{equation}\label{eq:primitive-at-vertex}
 \begin{aligned}
 U-u_1&=a\xi^m+O(\xi^{m+1}),&
 C-C_0&=b\xi^d+O(\xi^{d+1}),\\
 (m,d)&=
 \begin{cases}
 (n_1/2,1),& n_1\text{ even},\\
 (n_1,2),& n_1\text{ odd},
 \end{cases}
 &a,b&\in\mathbb{C}\setminus\{0\}.
 \end{aligned}
\end{equation}
In either case $d/m=2/n_1$ is not an integer.

Let $\tau$ be a period of $\alpha$ along a closed curve in
$X^\circ$. The corresponding transformation of
$\widetilde{X^\circ}$ replaces $U$ by $U+\tau$ and leaves $C,N$
unchanged. Continuing the local primitive near the chosen point
above $a_1$, the expansions~\eqref{eq:primitive-at-vertex} hold
with $u_1$ replaced by $u_1+\tau$. Suppose that $u_1+\tau\notin
u_1+\tfrac12\Lambda$. Then $R$ is unramified there, and its value
is an ordinary point of $\mathcal S_4$. In a coordinate $s$
vanishing at this value, the function $s(R(U))$ has order $m$
in $\xi$, whereas $C-C_0$ has order $d$. By the continuation
argument above, this branch of $C$ must be meromorphic in $s$.
Its finite limit $C_0$ makes it holomorphic at $s=0$, so $d$ must
be an integer multiple of $m$, a contradiction. If the value of
$R$ is infinity, the same argument uses $s=1/z$. We have proved
\begin{equation}\label{eq:period-restriction}
 \int_\gamma\alpha\in\tfrac12\Lambda
 \qquad\text{for every closed curve }\gamma\text{ in }X^\circ.
\end{equation}

\textbf{Exclusion of poles.}
We show that $\alpha$ has no poles. Each logarithmic
differential in~\eqref{eq:algebraic-square-root} has at most
simple poles. Hence their product has poles of order at most two.
A pole of order two gives a simple pole of $\alpha$ on the
unramified cover; at a pole of order one, the substitution
$w=\xi^2$ makes the pulled-back quadratic differential regular.
Thus every pole of $\alpha$ is simple, and both $C$ and $N$ have
a zero or pole there.

Suppose such a pole occurs at $\xi=0$. Choose $\varepsilon>0$
so small that the punctured disk $0<|\xi|<\varepsilon$ lies in
$X^\circ$. Its universal covering is the half-plane
$\operatorname{Re}v<\log\varepsilon$, with covering map
$\xi=e^v$. Choose a lift of this map to $\widetilde{X^\circ}$.
The pullback of $U$, again denoted by $U$, has the form
\[
 U(v)=av+h_0(e^v),\qquad a\in\mathbb{C}\setminus\{0\},
\]
where $a$ is the residue of $\alpha$ at the pole and $h_0$ is
holomorphic on $|\xi|<\varepsilon$. Since $C$ has a zero or pole
at $0$, we may also assume $C'(\xi)\ne0$ on the punctured disk.
Choose $\tau\in\Lambda$ with $\operatorname{Re}(\tau/a)<0$;
such a period exists because $\Lambda$ spans $\mathbb{C}$ over $\mathbb{R}$.

For $n\in\mathbb N$, set $v_n^0=(u_1+n\tau-h_0(0))/a$. Then
$\operatorname{Re}v_n^0\to-\infty$ as $n\to\infty$, and
\[
 U(v)-u_1-n\tau
 =a(v-v_n^0)+h_0(e^v)-h_0(0).
\]
Since $h_0(e^v)-h_0(0)\to0$ uniformly on $|v-v_n^0|\leq1$,
Rouch\'e's theorem gives, for sufficiently large $n$, a zero
$v_n$ of $U(v)-u_1-n\tau$ with $|v_n-v_n^0|<1$. Thus
\[
 U(v_n)=u_1+n\tau,\qquad
 \operatorname{Re}v_n\to-\infty,\qquad
 \xi_n:=e^{v_n}\to0.
\]

Moreover, $U'(v_n)=a+\xi_n h_0'(\xi_n)\to a\ne0$. Since $R$
has local degree two at $u_1+n\tau$ over $p_1$, for
sufficiently large $n$ we obtain
\[
 R(U(v))-p_1
 =c_n(v-v_n)^2+O((v-v_n)^3),
 \qquad c_n\in\mathbb{C}\setminus\{0\}.
\]
Analytic continuation once around a small positively oriented
circle centered at $p_1$ exchanges the two inverse branches.
Since the derivative of $C(e^v)$ at $v_n$ is
$\xi_n C'(\xi_n)\ne0$, they give distinct branches of $C$,
whereas analytic continuation twice around $p_1$ returns both
$C$ and $N$ to their initial germs. The chosen lift maps a
sufficiently small punctured neighborhood of $v_n$ into $\Omega$.
By the continuation argument above, both branches are analytic
continuations of the initial pair. Their local monodromy has order
two, contradicting
Corollary~\ref{cor:local-order}, which gives order $n_1>2$ on every
branch. Thus $\alpha$ is holomorphic on $X$.

\textbf{Construction of the finite cover.}
Since $\alpha$ is holomorphic, its periods define, by
\eqref{eq:period-restriction}, a homomorphism
\[
 \pi_1(X)\longrightarrow(\tfrac12\Lambda)/\Lambda,
 \qquad [\gamma]\longmapsto\int_\gamma\alpha\pmod\Lambda.
\]
The target has four elements. The kernel therefore defines a
connected unbranched covering $\widehat X\to X$ of degree at most
four. On this covering, all periods of the pullback of $\alpha$
belong to $\Lambda$. Integration therefore defines a holomorphic
map $\widehat X\to\mathbb{C}/\Lambda$; at a zero of $\alpha$, it is given
by a local primitive. Thus $R\circ U$ is a well-defined nonconstant
meromorphic function on $\widehat X$, agreeing with $z$ on the
initial branch. The surface $\widehat X$ is compact, and the
pullbacks of $C,N$ are meromorphic on it. A nonconstant meromorphic
function on a compact Riemann surface defines a finite map to
$\mathbb{CP}^1$. The field of meromorphic functions on $\widehat X$ is
therefore a finite extension of $\mathbb{C}(z)$, so $C,N$ are algebraic
over $\mathbb{C}(z)$.
Clearing denominators gives the stated polynomial relations,
and hence only finitely many branches of the pair.

\label{prop:infinite-monodromy}
For $k\geq l\geq1$ with $kl>1$, the pair $(C,N)$ associated with the Lawson surface
$\xi_{k,l}$ does not have finite monodromy on $\mathcal S_4$.

Suppose that $(C,N)$ has only finitely many branches. Let
$K\subset\pi_1(\mathcal S_4,z_*)$ be the kernel of the permutation action on these
branches, and put $H=K\cap\delta(K)$. Then $H$ is a normal subgroup
of finite index, invariant under $\delta$. Every $h\in H$ fixes
the initial pair, so Proposition~\ref{prop:reflection} gives
\begin{equation}\label{eq:reflection-subgroup}
 M_h=M_{\delta h}\qquad(h\in H).
\end{equation}

Consider the connected finite covering
$\Sigma'{}^\circ\to\Sigma^\circ$ corresponding to
$H\cap\pi_*\pi_1(\Sigma^\circ)$ under the covering $\pi$.
By Proposition~\ref{prop:continuation}, $\gamma_j^{n_j}\in K$ for
every $j$. Since $\delta$ permutes the punctures, preserves their
local degrees, and reverses orientation,
$\delta(\gamma_j^{n_j})$ is conjugate to $\gamma_a^{-n_a}$
for some $a$. Thus $\gamma_j^{n_j}\in H$.
A based loop going once around a removed point of $\Sigma$
maps to a conjugate of $\gamma_j^{n_j}$. Since $H$ is a normal
subgroup, this conjugate also belongs to $H$, so the covering
has local degree one at every removed point. Filling in these points gives an unbranched covering
$\widehat\pi\colon\Sigma'\to\Sigma$. If its degree is $d$, then
\[
 \operatorname{genus}(\Sigma')=1+d(g-1)>1.
\]
In particular, $f\circ\widehat\pi$ is a conformal minimal immersion of
a compact Riemann surface of genus at least two.

By~\cite[Theorem~4]{Heller}, the associated family of
$f\circ\widehat\pi$ has nonabelian monodromy for generic
$\lambda\in\mathbb{C}^*$.
It is nonabelian for some $\lambda_0\in S^1$: otherwise every
commutator of monodromy matrices would equal the identity on
$S^1$, and hence identically by holomorphic dependence on
$\lambda$. By the comparison in Section~\ref{sec:global-dpw},
the pullback of this representation under the natural surjection
$\pi_1(\Sigma'{}^\circ)\to\pi_1(\Sigma')$ agrees, up to
conjugation and the fixed central twist, with the representation
$h\mapsto M_h^\lambda$ on $H\cap\pi_*\pi_1(\Sigma^\circ)$. Since the twist does not change
commutators, the matrices $M_h^{\lambda=\lambda_0}$, $h\in H$, form a
nonabelian subgroup of $\operatorname{SU}(2)$.

Since $H$ is a normal subgroup, $\gamma h\gamma^{-1}\in H$
for every $\gamma\in\pi_1(\mathcal S_4,z_*)$ and $h\in H$. Applying
\eqref{eq:reflection-subgroup} to this element, we obtain
\[
 \begin{aligned}
 M_\gamma M_hM_\gamma^{-1}
 &=M_{\gamma h\gamma^{-1}}
 =M_{\delta(\gamma h\gamma^{-1})}\\
 &=M_{\delta\gamma}M_{\delta h}M_{\delta\gamma}^{-1}\\
 &=M_{\delta\gamma}M_hM_{\delta\gamma}^{-1}.
 \end{aligned}
\]
Here the second line uses that $\delta$ is an automorphism of
$\pi_1(\mathcal S_4,z_*)$, and the last equality uses
\eqref{eq:reflection-subgroup} for $h$.
Consequently, $M_{\delta\gamma}^{-1}M_\gamma$ commutes with
every $M_h$, $h\in H$. At $\lambda_0$, their common centralizer
in $\operatorname{SL}(2,\mathbb{C})$ is $\{\mathrm{Id},-\mathrm{Id}\}$.
Indeed, a nonscalar matrix in the centralizer would have a
one-dimensional eigenspace invariant under all $M_h^{\lambda=\lambda_0}$.
Unitarity would make its orthogonal complement invariant as well,
forcing the group to be abelian.

Taking $\gamma=\gamma_1$ and using
$\delta\gamma_1=\gamma_1^{-1}$ from
Remark~\ref{rem:reflected-local-loop}, we obtain
\[
 (M_{\gamma_1}^{\lambda=\lambda_0})^2\in
 \{\mathrm{Id},-\mathrm{Id}\}.
\]
But Lemma~\ref{lem:potential-monodromy} gives the eigenvalues
$e^{\pm\pi i/n_1}$ for $M_{\gamma_1}^{\lambda=\lambda_0}$. Their squares
$e^{\pm2\pi i/n_1}$ are distinct for $n_1>2$, a contradiction.
