<!-- theorem-1: body_md -->
\label{thm:fixed}
Suppose that \(\mathrm H_{\theta_0}\) holds for some fixed
\(1/2\leq\theta_0<1\). Then, for every \(\varepsilon>0\),
\begin{equation}\label{eq:seven}
 E(X)\ll_{\theta_0,\varepsilon}X^{7/12+\varepsilon}.
\end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:seven}
On invoking the uniform Dirichlet \(L\)-function theorem
of OpenAI \cite[Theorem~1.1]{OAI}, one obtains
\[
 E(X)\ll_\varepsilon X^{7/12+\varepsilon}.
\]
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{thm:main}
Assume \(\mathrm H_\theta\) for a fixed \(1/2\leq\theta\leq2/3\).
For every \(\varepsilon>0\),
\begin{equation}\label{eq:main}
 E(X)\ll_{\theta,\varepsilon}X^{g(\theta)+\varepsilon},
 \qquad
 g(\theta)=2\theta-\frac32+\frac{3(1-\theta)}{2-\theta}
 =\frac{\theta(5-4\theta)}{2(2-\theta)}.
\end{equation}
<!-- end theorem-2 -->
<!-- lemma-1: body_md -->
\label{lem:local}
Assume \(\mathrm H_\theta\), with \(1/2\leq\theta\leq2/3\).
For every \(\eta>0\), uniformly when
\[
 (a,q)=1,\qquad 1\leq q\leq2\sqrt X,
 \qquad |\beta|\leq\frac{2}{q\sqrt X},
\]
one has, with \(z=X^{-1}-2\pi i\beta\),
\begin{equation}\label{eq:local}
 F_X(a/q+\beta)=\frac{\mu(q)}{\varphi(q)z}
 +O_{\theta,\eta}\bigl(X^{a(\theta)+\eta}\bigr).
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Put
\[
 T=1+X|\beta|,\qquad B=qT.
\]
Then \(B\leq4\sqrt X\), \(|z|\asymp T/X\), and
\(\pi/2-|\arg z|\gg1/T\).

\paragraph{Character expansion and the explicit formula.}
For \((m,q)=1\), character orthogonality gives
\[
 e(am/q)=\frac{1}{\varphi(q)}
 \sum_{\chi\bmod q}\tau(\overline\chi)\chi(a)\chi(m).
\]
Here \(|\tau(\chi)|\leq\sqrt q\), including for imprimitive
characters. Replacing \(\chi\) by \(\chi^*\) changes the
corresponding von Mangoldt sum only on powers of primes dividing
\(q\). Their contribution, and that of \((m,q)>1\), is absorbed
in \(O(\sqrt q\log^2(qX))\).

For a primitive character of conductor \(f\leq q\), Mellin
inversion and a contour shift give
\begin{equation}\label{eq:explicit-primitive}
 \sum_{m\geq1}\Lambda(m)\chi^*(m)e^{-mz}
 =\frac{\delta_{\chi^*}}{z}
 -\sum_{\rho_{\chi^*}}\Gamma(\rho_{\chi^*})z^{-\rho_{\chi^*}}
 +O_\theta\bigl(\log^2(qX)\bigr),
\end{equation}
where \(\delta_{\chi^*}=1\) for conductor one and is zero otherwise.
The power of \(z\) uses the principal logarithm.
For completeness, start with
\[
 -\frac{1}{2\pi i}\int_{(2)}
 \frac{L'}{L}(s,\chi^*)\Gamma(s)z^{-s}\,\mathrm d s
\]
and move the line to \(\operatorname{Re} s=-1/2\). The poles crossed are the
pole at one, the nontrivial zeros, and the possible simple or
double pole of the integrand at zero. Under \eqref{eq:strip},
the last residue is \(O_\theta(\log^2(qX))\): the regular parts
of the logarithmic derivative at zero are bounded using the
functional equation and its partial-fraction formula at one
and two. Indeed, the zeros stay a fixed distance from one and
their sum with weight \((1+|\gamma|^2)^{-1}\) is
\(O(\log(f+3))\).
On the new line the functional equation gives
\(L'/L(s,\chi^*)\ll\log(f(|\operatorname{Im} s|+2))\).
Stirling's formula bounds the remaining integral by
\[
 O\!\left(|z|^{1/2}\log^2\bigl(q(T+2)\bigr)\right),
\]
which is within the error in \eqref{eq:explicit-primitive}.
Horizontal contours can be taken away from zeros; their
integrals tend to zero because \(|\arg z|<\pi/2\).
The same exponential decay makes the displayed zero sum
absolutely convergent.

Consequently, with \(c_\chi=\tau(\overline\chi)\chi(a)\),
\begin{equation}\label{eq:explicit}
 F_X(a/q+\beta)=\frac{\mu(q)}{\varphi(q)z}
 -\frac{1}{\varphi(q)}\sum_{\chi\bmod q}c_\chi
   \sum_{\rho_{\chi^*}}\Gamma(\rho_{\chi^*})z^{-\rho_{\chi^*}}
 +O_\theta\bigl(\sqrt q\log^2(qX)\bigr).
\end{equation}
The principal character supplies the main term because its
Gauss sum is \(\mu(q)\).

\paragraph{Estimating the zeros.}
For \(\rho=u+i\gamma\), Stirling's formula and
\eqref{eq:strip} give
\begin{equation}\label{eq:gamma}
 |\Gamma(\rho)z^{-\rho}|
 \ll_\theta X^u T^{-u}(1+|\gamma|)^{u-1/2}
             e^{-c|\gamma|/T}
\end{equation}
for an absolute \(c>0\).
If \(u<1/2\), the right side is at most
\(O_\theta(X^{1/2}T^{-1/2}e^{-c|\gamma|/T})\), since
\(X/T\gg\sqrt X\).
After summing by \eqref{eq:total} and multiplying by
\(\sqrt q/\varphi(q)\), all these zeros contribute
\begin{equation}\label{eq:low}
 \ll_{\theta,\eta} X^\eta X^{1/2}B^{1/2}.
\end{equation}

For \(u\geq1/2\), the bound \eqref{eq:gamma} is at most
\[
 C_\theta X^uT^{-1/2}
 (1+|\gamma|/T)^{1/6}e^{-c|\gamma|/T}.
\]
One may discard \(|\gamma|>T(\log X)^2\) with an error smaller
than any fixed negative power of \(X\), by \eqref{eq:total}.
Partition \([1/2,\theta]\) into intervals of length at most
\(1/\log X\), treating the endpoint \(\theta=1/2\) as a single
class. Apply \eqref{eq:density} to the left endpoint of each
interval, with \(U\asymp T(\log X)^2\).
Logarithmic factors, the number of intervals, and
\(q/\varphi(q)\) may all be absorbed in \(X^\eta\).
Together with \eqref{eq:low}, this bounds the zero term in
\eqref{eq:explicit} by
\begin{equation}\label{eq:zerobound}
 \ll_{\theta,\eta}X^\eta\left(
 X^{1/2}B^{1/2}+
 \sup_{1/2\leq\sigma\leq\theta}
 X^\sigma B^{D(\sigma)-1/2}\right).
\end{equation}
Here the factor \(B^{-1/2}\) is precisely the product of the
Gauss-sum factor \(q^{-1/2}\), up to logarithms, and
\(T^{-1/2}\).

For \(1/2\leq\sigma\leq2/3\), one has
\(D(\sigma)-1/2\geq1/4\). Since \(B\ll\sqrt X\), the second
term in \eqref{eq:zerobound} is bounded by the maximum of
\(X^{h(\sigma)}\), where
\[
 h(\sigma)=\sigma-\frac14+\frac{D(\sigma)}2,
 \qquad
 h'(\sigma)=1-\frac{3}{2(2-\sigma)^2}\geq\frac5{32}>0.
\]
Thus the maximum is \(X^{h(\theta)}=X^{a(\theta)}\).
The first term is \(O(X^{3/4})\), and
\(a(\theta)\geq a(1/2)=3/4\).
The last error in \eqref{eq:explicit} is smaller still.
This proves \eqref{eq:local}.
