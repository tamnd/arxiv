<!-- definition-1: body_md -->
The operator $\mathcal{P}:\mathscr{H}\to 2^{\mathscr{H}}$ is called 
\emph{$\Pi_{\mathcal{P}}$-monotone} if $\exists$ a constant 
$\Pi_{\mathcal{P}}\in\mathbb{R}$ s. t.
\[
\langle w-y,\,u-v\rangle
\ge
\Pi_{\mathcal{P}}\|w-y\|^2,
\quad
\forall w,y\in \mathscr{H},\ 
u\in \mathcal{P}(w),\ 
v\in \mathcal{P}(y).
\]
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\cite{Minh}
A $\Pi_{\mathcal{P}}$-monotone operator $\mathcal{P}$ is maximal if no proper $\Pi_{\mathcal{P}}$-monotone extension of $\mathcal{P}$ exists.\\
<!-- end definition-2 -->
<!-- remark-1: body_md -->
The parameter $\Pi_{\mathcal{P}}$ may be negative. 
The case $\Pi_{\mathcal{P}}=0$ corresponds to classical monotonicity, 
$\Pi_{\mathcal{P}}>0$ corresponds to strong monotonicity, 
and $\Pi_{\mathcal{P}}<0$ describes weak monotonicity.
<!-- end remark-1 -->
<!-- definition-3: body_md -->
We say that $\mathcal{Q}:\mathscr{H}\to\mathscr{H}$ is
\begin{itemize}
    \item Lipschitz continuous with $ \xi\ge 0$ if
    \[
    \|\mathcal{Q}(w_1)-\mathcal{Q}(w_2)\|\le  \xi\|w_1-w_2\|,

Differentiating the function \eqref{THL1} along the model trajectories. Then by Assumption~\eqref{assINDIs} yields
\begin{equation}
    \begin{split}
        \dot{\mathcal{L}}(s,w(s))
&=
\left\langle w-{\overline{\mathfrak{w}}}^*,\dot w\right\rangle
\\
&=
-
\left(
\phi_1(s)\frac{1}{\|w-\Psi(w)\|^{1-\lambda_1}}
+
\phi_2(s)\frac{1}{\|w-\Psi(w)\|^{1-\lambda_2}}
\right)
\cdot
\langle w-{\overline{\mathfrak{w}}}^*, w-\Psi(w)\rangle
\\&+ \delta\|w-{\overline{\mathfrak{w}}}^*\|^2.
    \end{split}
\end{equation}
Using Lemma \ref{tr1}, we get 
\begin{equation}\label{DTHL2}
    \begin{split}
        \dot{\mathcal{L}}(s,w(s))
&\le
-
(1- \aleph) \phi_1(s)
\frac{\|w-{\overline{\mathfrak{w}}}^*\|^2}
{\mathcal (1+\aleph)^{1-\lambda_1}\|w-{\overline{\mathfrak{w}}}^*\|^{1-\lambda_1}}
-
(1- \aleph) \phi_2(s)
\frac{\|w-{\overline{\mathfrak{w}}}^*\|^2}
{(1- \aleph)^{1-\lambda_2}\|w-{\overline{\mathfrak{w}}}^*\|^{1-\lambda_2}}\\
&\qquad+ \delta\|w-{\overline{\mathfrak{w}}}^*\|^2.
\\
&=
-
\frac{(1- \aleph)}{\mathcal (1+\aleph)^{1-\lambda_1}}
\phi_1(s)
\|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_1}
-
(1- \aleph)^{\lambda_2} \phi_2(s)
\|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_2}\\
&\qquad+ \delta\|w-{\overline{\mathfrak{w}}}^*\|^2.
    \end{split}
\end{equation}
If we suppose
\(
\Xi := \{w \mid \|w-{\overline{\mathfrak{w}}}^*\|\le 1\}.
\) Then it is form two cases.\\

\medskip
\noindent
\textbf{Case 1:} If $w\in \Xi$. Since $0<\lambda_1<1$ with $\|w-{\overline{\mathfrak{w}}}^*\|\le1$, it concluded that
\[
\|w-{\overline{\mathfrak{w}}}^*\|^2 - \|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_1} \le 0.

Thus,
\begin{equation}\label{ineq13}
\dot{\mathcal{L}}(s,w(s))
\le
- N_1 \|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_1}
- N_2 \|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_2},

N_2=

\) \\

\noindent
\textbf{Case 2:} If $w \notin \Xi$. Since $\lambda_2>1$ with $\|w-{\overline{\mathfrak{w}}}^*\|>1$, it concluded that
\[
\|w-{\overline{\mathfrak{w}}}^*\|^2 - \|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_2} \le 0.
\]
Thus, inequality \eqref{ineq13} is valid for both cases. Consequently, we obtain
\begin{equation*}
\dot{\mathcal{L}}(s,w(s))
\le
-
\left(\frac{(1- \aleph)}{\mathcal (1+\aleph)^{1-\lambda_1}}
\phi_1(s)-\delta\right)
\|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_1}
-
\left((1- \aleph)^{\lambda_2}\phi_2(s)-\delta\right)
\|w-{\overline{\mathfrak{w}}}^*\|^{1+\lambda_2}.
\end{equation*}

${\overline{\mathfrak{w}}}^*$ within  $\mathrm{FxT}$.
