<!-- theorem-1: body_md -->
\label{thm:poisson}
Let the iid potential have a bounded nonconstant law. Suppose $E$
lies in an open interval on which $N$ is continuous and strictly
increasing. The following convergences hold in distribution for the
vague topology.
\begin{enumerate}
\renewcommand{(\arabic{enumi})}{(\arabic{enumi})}
\item\label{part:ids-unfolding} \textbf{IDS unfolding.}
Without a differentiability assumption on $N$,
\[
 \xi_{n,E}\ \Longrightarrow\ \operatorname{PPP}(dx).
\]
Here $\operatorname{PPP}(dx)$ is the unit-intensity Poisson point
process on $\mathbb R$: its counts in pairwise disjoint bounded
intervals are independent Poisson variables with means equal to
the interval lengths.
For a nondegenerate two-point law on $\{a,b\}$, $a<b$, this conclusion
covers every $E$ in the interior of $[a-2,a+2]\cup[b-2,b+2]$,
with no additional IDS regularity assumption.
\item\label{part:linear-scaling} \textbf{Linear scaling.}
If, in addition, $N$ is differentiable at the single point $E$, with
$0<\rho(E)=N'(E)<\infty$, then
\[
 \sum_{j=1}^n\delta_{\,n\rho(E)(\lambda_{j,n}-E)}
       \ \Longrightarrow\ \operatorname{PPP}(dx).
\]
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:sturm}
For every realization and energy $E'$,
\begin{equation}\label{eq:sturm}
 \#\{1\leq j\leq n:\lambda_{j,n}<E'\}
    =n-\left\lfloor\frac{\theta_n(E')+\pi/2}{\pi}\right\rfloor.
\end{equation}
For $E'<E''$,
\begin{equation}\label{eq:sturm-interval}
 \#\{1\leq j\leq n:\lambda_{j,n}\in(E',E'']\}
 =\left\lceil\frac{\theta_n(E')-\pi/2}{\pi}\right\rceil
  -\left\lceil\frac{\theta_n(E'')-\pi/2}{\pi}\right\rceil.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The condition $u_{n+1}(E')=0$ is the Dirichlet eigenvalue condition,
or $\theta_n(E')=\pi/2$ modulo $\pi$. To see that the phase is strictly decreasing
in $E'$, abbreviate $\theta_j=\theta_j(E')$ and write
$\dot\theta_j=d\theta_j(E')/dE'$. Since $\theta_0=0$,
\eqref{eq:derivatives} gives
\[
 \dot\theta_0=0,\qquad
 \dot\theta_1=-\frac{1}{(E'-v_0)^2+1}<0.
\]
At each subsequent step the chain rule gives
\[
 \begin{aligned}
 \dot\theta_{j+1}
  &=\partial_\theta F_{E'}(v_j,\theta_j)\,\dot\theta_j
       +\partial_{E'}F_{E'}(v_j,\theta_j)\\
  &=\frac{\dot\theta_j-\cos^2\theta_j}
          {\left\lVert A_{E'}(v_j)\mathbf u(\theta_j)\right\rVert^2}<0,
       \qquad j\geq1.
 \end{aligned}
\]
Here $\partial_{E'}F$ differentiates at fixed input phase; the first
term accounts for the energy dependence of that input. The last
inequality follows inductively from $\dot\theta_j<0$ and the positive
denominator. Thus $\dot\theta_n<0$ for every $n\geq1$. As $E'$ increases from $-\infty$
to $+\infty$, the phase decreases from $n\pi$ to zero. The $n$
simple Dirichlet eigenvalues are its successive half-integer
crossings. This gives \eqref{eq:sturm}, with the displayed strict
endpoint convention. The non-strict count is
$n-\lceil(\theta_n(E')-\pi/2)/\pi\rceil$; subtract its values at
$E''$ and $E'$ to obtain \eqref{eq:sturm-interval}.

<!-- proposition-1: body_md -->
\label{prop:furstenberg-used}
For every real $E'$, the forward and inverse projective actions have
unique stationary probabilities $\nu_{E'}^+$ and $\nu_{E'}^-$.
Both probabilities are non-atomic. More explicitly, every stationary
probability for either action is non-atomic and satisfies
\begin{equation}\label{eq:stationary-log-positive}
 (P_{E'}^\pm)^*\nu=\nu
       \quad\Longrightarrow\quad \lambda_{E'}^\pm(\nu)>0.
\end{equation}
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
The accompanying Lean~4 formalization also proves
Proposition~\ref{prop:furstenberg-used} for the actual transfer law.
It obtains positivity for every stationary probability before using
finite-block contraction to prove uniqueness, as outlined in
Section~\ref{sec:formalization}. The final formalized Poisson statements
therefore have no external Furstenberg hypothesis.

<!-- lemma-2: body_md -->
\label{lem:harmonic}
There is a mean one-step rotation increment $r_{\mathrm{rot}}(E')$,
independent of the initial angle, such that, for every $\theta$,
\begin{equation}\label{eq:exactone}
 \int[\mathcal H_{E'}(F_{E'}(v,\theta))-\mathcal H_{E'}(\theta)]\,d\mu(v)=r_{\mathrm{rot}}(E').
\end{equation}
For every $x\geq y$,
\begin{equation}\label{eq:exactgap}
 \int[\mathcal H_{E'}(F_{E'}(v,x))-\mathcal H_{E'}(F_{E'}(v,y))]\,d\mu(v)
      =\mathcal H_{E'}(x)-\mathcal H_{E'}(y).
\end{equation}
With the chosen phase convention,
\begin{equation}\label{eq:rotationids}
 r_{\mathrm{rot}}(E')=\pi(1-N(E')).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
First suppose $0\leq x-y<\pi$, and let $J$ be the oriented arc from
$y$ to $x$ in $\mathbb {RP}^{1}$, in the increasing angular direction.
Since $F_{E'}(v,\cdot)$ is an increasing degree-one lift,
\[
 \mathcal H_{E'}(F_{E'}(v,x))-\mathcal H_{E'}(F_{E'}(v,y))=\pi\nu_{E'}^-(T_{E',v}J).
\]
This equality also holds when $J$ or its image crosses the fixed cut:
the continuous lift counts the wrapped part of the arc. In particular,
no independent integer correction is omitted. Integrating and using
\eqref{eq:inverse},
\[
 \int\nu_{E'}^-(T_{E',v}J)\,d\mu(v)
  =\int (T_{E',v}^{-1})_*\nu_{E'}^-(J)\,d\mu(v)
  =\nu_{E'}^-(J).
\]
This proves \eqref{eq:exactgap} for such gaps. Equivariance adds the
appropriate integer multiple of $\pi$ for an arbitrary gap. Define the averaged increment
\[
 g_{E'}(\theta):=\int[\mathcal H_{E'}(F_{E'}(v,\theta))
                       -\mathcal H_{E'}(\theta)]\,d\mu(v).
\]
For any $x\geq y$, subtracting its values at $x$ and $y$ gives
\[
 \begin{aligned}
 g_{E'}(x)-g_{E'}(y)
  &=\int[\mathcal H_{E'}(F_{E'}(v,x))
                 -\mathcal H_{E'}(F_{E'}(v,y))]\,d\mu(v)\\
  &\qquad-[\mathcal H_{E'}(x)-\mathcal H_{E'}(y)]=0,
 \end{aligned}
\]
where the last equality is exactly \eqref{eq:exactgap}. Thus
$g_{E'}$ is constant on $\mathbb R$. Denote this common value by
$r_{\mathrm{rot}}(E')$; this proves \eqref{eq:exactone}.

We now identify the rotation increment in terms of the IDS.
Start the physical recursion with $(u_1,u_0)=(1,0)$, hence $\theta_0=0$.
By \eqref{eq:exactone},
\[
 \mathbb E \mathcal H_{E'}(\theta_n)=\mathcal H_{E'}(0)+nr_{\mathrm{rot}}(E').
\]
The periodic difference $\mathcal H_{E'}(x)-x$ is bounded, so
$\mathbb E\theta_n/n\to r_{\mathrm{rot}}(E')$. Lemma~\ref{lem:sturm} identifies the normalized expected eigenvalue
count with $1-r_{\mathrm{rot}}(E')/\pi$ in the limit.
This proves \eqref{eq:rotationids}.

<!-- lemma-3: body_md -->
\label{lem:continuity}
The map $E'\mapsto \mathcal H_{E'}$ is continuous for the uniform norm on periodic
lift differences: the supremum over $[0,\pi]$ equals the supremum
over $\mathbb R$ because these differences are $\pi$-periodic. On
a compact interval $I$ with $E$ in its interior, define, for an
energy gap $\delta\geq0$,
\[
 \omega_{\mathrm{en}}(\delta)
 =\sup_{\substack{E',E''\in I\\|E''-E'|\leq\delta}}
       \left\lVert \mathcal H_{E''}-\mathcal H_{E'}\right\rVert_\infty,
\]
and, for a phase gap $h\geq0$,
\[
 \omega_{\mathrm{ph}}(h)=\sup_{E'\in I}\sup_{|x-y|\leq h}|\mathcal H_{E'}(x)-\mathcal H_{E'}(y)|.
\]
Then $\omega_{\mathrm{en}}(\delta)\to0$ as $\delta\downarrow0$, and
$\omega_{\mathrm{ph}}(h)\to0$ as $h\downarrow0$. The IDS is continuous on $I$; by \eqref{eq:exactone} and
\eqref{eq:rotationids},
\[
 N(E')=1-\frac1\pi\int
       \mathcal H_{E'}(F_{E'}(v,0))\,d\mu(v).
\]
Both moduli are deterministic and depend only on $\mu$ and this
fixed interval.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
If $E'_m\to E'$, compactness and the continuous dependence of the
inverse transition kernel show that every subsequential limit of
$\nu_{E'_m}^-$ is stationary at $E'$. Uniqueness identifies it with
$\nu_{E'}^-$. Since the limiting law is non-atomic, weak convergence
gives uniform convergence of its distribution functions, hence of
the normalized lifts. On the compact interval $I$, this continuous
family is uniformly continuous in energy and equicontinuous in phase,
which gives both moduli. Finally, the integrand in the displayed formula for $N(E')$
is continuous in energy and lies in $[0,\pi]$.
Dominated convergence proves continuity of $N$, and hence of
$r_{\mathrm{rot}}=\pi(1-N)$.

\label{lem:overshoot}
Here $\delta>0$ is the energy gap between $E'$ and $E'+\delta$, and
$D_j=\theta_j-\phi_j$ is their lifted phase gap.
If $D_0\geq0$, then $D_j\geq0$ and

\begin{equation}\label{eq:overshoot}
 m\pi\leq D_{T_m}\leq m\pi+\delta.
\end{equation}
The bound is pathwise, uniform over all initial phases and
over all waiting times and crossing numbers. For sufficiently small
$\delta$, a single step cannot cross two new integer levels.

For any integer $k$, monotonicity, $\pi$-equivariance and
\eqref{eq:energyclose} give
\[
 D_j\geq k\pi\quad\Longrightarrow\quad
 \theta_{j+1}\geq F_{E'}(v_j,\phi_j)+k\pi
 \geq\phi_{j+1}+k\pi.

Taking $k=0$ and then $k=\lfloor D_j/\pi\rfloor$ proves the first

For the upper bound, whenever $D_j<m\pi$, write
\[
 \begin{aligned}
 D_{j+1}={}&\underbrace{F_{E'}(v_j,\theta_j)-F_{E'}(v_j,\phi_j)}_{<\,m\pi}\\
 &+\underbrace{F_{E'}(v_j,\phi_j)-F_{E'+\delta}(v_j,\phi_j)}_{\in[0,\delta]}

 \end{aligned}
\]
The first bound follows from $\theta_j<\phi_j+m\pi$ by monotonicity
and equivariance; the second is \eqref{eq:energyclose}. At a finite
first crossing, $j=T_m-1\geq0$ satisfies $D_j<m\pi\leq D_{j+1}$,
which proves \eqref{eq:overshoot}.

Finally, with $k=\lfloor D_j/\pi\rfloor$, the same estimate at
level $(k+1)\pi$ gives $D_{j+1}<(k+1)\pi+\delta<(k+2)\pi$ when
$\delta<\pi$, so at most one new integer level is crossed.

<!-- proposition-2: body_md -->
\label{prop:drift}
For the pair chain \eqref{eq:pair}, define
\begin{equation}\label{eq:W}
 W_j=\mathcal H_{E'}(\theta_j)-\mathcal H_{E'+\delta}(\phi_j).
\end{equation}
Then
\begin{equation}\label{eq:drift}
 \mathbb E[W_{j+1}-W_j\mid\mathcal F_j]=\pi \Delta N_{E',\delta}
      =\pi[N(E'+\delta)-N(E')].
\end{equation}
Before the first exit, $0\leq D_j<\pi$ gives
\begin{equation}\label{eq:Wbounds}
 -\omega_{\mathrm{en}}(\delta)\leq W_j
       \leq\pi+\omega_{\mathrm{en}}(\delta),\qquad j<T.
\end{equation}
At exit,
\begin{equation}\label{eq:Wexit}
 |W_T-\pi|\leq\omega_{\mathrm{exit}}(\delta)
 :=\omega_{\mathrm{ph}}(\delta)
                      +\omega_{\mathrm{en}}(\delta)\longrightarrow0.
\end{equation}
The deterministic exit error $\omega_{\mathrm{exit}}$ uses the
energy derivative bound $1$ from \eqref{eq:energyclose}; it
does not depend on the initial state, exit time, or crossing number.
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Since $v_j$ is independent of $\mathcal F_j$, Lemma~\ref{lem:harmonic},
specifically \eqref{eq:exactone} and \eqref{eq:rotationids}, gives
\[
 \begin{aligned}
 \mathbb E[W_{j+1}-W_j\mid\mathcal F_j]
 &=\pi(1-N(E'))-\pi(1-N(E'+\delta))\\
 &=\pi \Delta N_{E',\delta}.
 \end{aligned}
\]
The same fresh variable is used in both terms; linearity of conditional
expectation requires no independence between these two terms.
Before exit, monotonicity, $\pi$-equivariance, and the energy

$\varepsilon:=D_T-\pi$. Lemma~\ref{lem:overshoot} gives

By $\pi$-equivariance of $\mathcal H_{E'}$,

 \begin{aligned}
 W_T-\pi

  &\quad+[\mathcal H_{E'}(\phi_T)-\mathcal H_{E'+\delta}(\phi_T)].
 \end{aligned}
\]
The first and second brackets have absolute values at most
$\omega_{\mathrm{ph}}(\delta)$ and $\omega_{\mathrm{en}}(\delta)$,
respectively. This proves \eqref{eq:Wexit} pathwise, with a bound

<!-- lemma-5: body_md -->
\label{lem:contraction}
There are $L\geq1$, $r\in(0,1)$, $q_0\in(0,1)$ and a compact
neighborhood $I$ of $E$, all depending only on $\mu$ and the fixed
reference energy $E$, such that, uniformly over $E'\in I$ and
projective points $x,y$,
\begin{equation}\label{eq:blockcontract}
 \mathbb E d(M_{E',L}x,M_{E',L}y)^r\leq q_0d(x,y)^r.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
At the fixed energy $E$, let $P=P_E^+$ on $X=\mathbb {RP}^{1}$ and set
\[
 f_0([x])=\int\log\left\lVert A_E(v)x\right\rVert\,d\mu(v),\qquad \left\lVert x\right\rVert=1.
\]
Thus $f_0$ is the expected logarithmic growth in one step from the
direction $[x]$. The compact support of $\mu$ bounds both the matrices
and their inverses, so $f_0$ is continuous and bounded and $P$ is
Feller. For $n\geq1$, define the mean growth per step
\[
 b_n(z)=\frac1n\sum_{j=0}^{n-1}P^jf_0(z),\qquad z\in X.
\]
The cocycle identity and independence of each fresh potential give
$n b_n([x])=\mathbb E\log\left\lVert M_{E,n}x\right\rVert$ for every unit vector $x$.
By Proposition~\ref{prop:furstenberg-used},
$\int f_0\,d\nu>0$ for every stationary probability $\nu$ of $P$.

We first show that there are a single finite length $L\geq1$ and a
constant $a>0$ such that
\begin{equation}\label{eq:cesaro}
 \inf_{\left\lVert x\right\rVert=1}\mathbb E\log\left\lVert M_{E,L}x\right\rVert\geq a.
\end{equation}
Here the same $L$ and $a$ must work for every starting direction;
the energy is still fixed at $E$.

Suppose this were false. Then $\min_{z\in X}b_n(z)\leq0$ for every
$n$: a positive minimum for any $n$ would give \eqref{eq:cesaro}
with $L=n$ and $a=n\min_X b_n>0$. Since $b_n$ is continuous and
$X$ is compact, choose a
minimizer $z_n$ with $b_n(z_n)\leq0$. Start the projective chain at
$z_n$. Its law after $j$ steps is $(P^*)^j\delta_{z_n}$, where $P^*$
is the action of the transition kernel on probability measures.
Average these first $n$ laws to obtain the probability measure
\[
 \eta_n=\frac1n\sum_{j=0}^{n-1}(P^*)^j\delta_{z_n},
 \qquad \int f_0\,d\eta_n=b_n(z_n)\leq0.
\]
Because $X$ is a compact metric space, its probability measures are
weakly compact. Hence, along a subsequence $n_k\to\infty$,
$\eta_{n_k}$ converges weakly to a probability measure $\eta$.

The averaging makes this limit stationary: applying one further
transition cancels all the intermediate laws, leaving only the first
and last. More precisely, for every $g\in C(X)$,
\[
 \int(Pg-g)\,d\eta_n
 =\frac{P^ng(z_n)-g(z_n)}{n},
 \qquad
 \left|\int(Pg-g)\,d\eta_n\right|
 \leq\frac{2\left\lVert g\right\rVert_\infty}{n}\longrightarrow0.
\]
Since $P$ is Feller, $Pg-g$ is continuous. Passing to the weak limit
therefore gives $\int Pg\,d\eta=\int g\,d\eta$ for every continuous
$g$, which says precisely that $P^*\eta=\eta$. On the other hand,
continuity of $f_0$ gives
\[
 \int f_0\,d\eta
 =\lim_{k\to\infty}b_{n_k}(z_{n_k})\leq0.
\]
This contradicts the positive mean growth of every stationary
probability, proving \eqref{eq:cesaro}.

For this fixed $L$, the variables $Z_x=\log\left\lVert M_{E,L}x\right\rVert$ are
uniformly bounded. Taylor's inequality on their common bounded range
gives, uniformly in unit $x$,
\[
 \mathbb E e^{-2rZ_x}\leq1-2ra+Cr^2<1
\]
when $r>0$ is sufficiently small. Keep such an $r<1$ fixed.
The function
$G(E',[x])=\mathbb E\left\lVert M_{E',L}x\right\rVert^{-2r}$ is continuous jointly in
energy and direction, by continuity and the uniform bounds on a compact
energy neighborhood. Shrinking $I$ therefore gives
$\sup_{E'\in I,x}G(E',x)\leq q_0<1$, with $q_0>0$.
The determinant identity \eqref{eq:sl2distance} and Cauchy--Schwarz now
imply
\[
 \mathbb E d(M_{E',L}x,M_{E',L}y)^r
 \leq d(x,y)^r\bigl[G(E',x)G(E',y)\bigr]^{1/2}
 \leq q_0d(x,y)^r.\qedhere
\]

<!-- lemma-6: body_md -->
\label{lem:occupation}
With $L,r,q_0,I$ as in Lemma~\ref{lem:contraction}, define
\begin{equation}\label{eq:occupation-functions}
 f(\theta,\phi)=|\sin(\theta-\phi)|^r,
 \qquad V=\sum_{j=0}^{L-1}Q^jf.
\end{equation}
Then, uniformly over the allowed energy pairs,
\begin{equation}\label{eq:Vbounds}
 0\leq f\leq V\leq L,
\end{equation}
\begin{equation}\label{eq:Vdrift}
 QV-V\leq-(1-q_0)f+C\delta^r.
\end{equation}
For every deterministic initial state and every stopping time
$\tau\in L^\infty$ (equivalently, $0\leq\tau\leq N$ almost surely
for some deterministic integer $N$),
\begin{equation}\label{eq:stopped}
 (1-q_0)\mathbb E\sum_{j<\tau}f(\theta_j,\phi_j)
       \leq L+C\delta^r\mathbb E\tau.
\end{equation}
Here $C$ depends only on $\mu,I,L,r$, hence only on $\mu$ and the
fixed reference energy $E$ after the choices in Lemma~\ref{lem:contraction}.
It is independent of $E'$, $\delta$, the initial state, and the stopping
time $\tau$ (including its deterministic bound).

<!-- proof-7: body_md -->
First compare the two-energy pair with a pair at the single energy
$E'$. Starting from $(\theta,\phi)$, let $\widehat\phi_k$ evolve at
$E'$ from $\widehat\phi_0=\phi$, with the same potentials as
$\theta_k$ and $\phi_k$. Iterating the uniform one-step Lipschitz
bound and \eqref{eq:energyclose} over the fixed $L$ steps gives
$|\phi_L-\widehat\phi_L|\leq C\delta$.
Since $\bigl||\sin a|^r-|\sin b|^r\bigr|\leq|a-b|^r$,
Lemma~\ref{lem:contraction} yields
\[
 \begin{aligned}
 Q^Lf(\theta,\phi)
 &=\mathbb E|\sin(\theta_L-\phi_L)|^r\\
 &\leq\mathbb E|\sin(\theta_L-\widehat\phi_L)|^r+C\delta^r
 \leq q_0f(\theta,\phi)+C\delta^r.
 \end{aligned}
\]

The purpose of $V$ is to turn this $L$-step contraction into a
one-step drift estimate. Positivity of $Q$ and $Q1=1$ give
\eqref{eq:Vbounds}, while cancellation of the intermediate powers gives
\[
 QV-V=\sum_{k=0}^{L-1}(Q^{k+1}f-Q^kf)
      =Q^Lf-f\leq-(1-q_0)f+C\delta^r,
\]
proving \eqref{eq:Vdrift}. Each $Q^kf(\theta,\phi)$ is an
expectation with the initial phases fixed. Thus
$V_j:=V(\theta_j,\phi_j)$ is a function of the current phases and
is $\mathcal F_j$-measurable. The Markov property gives
\[
 \mathbb E[V_{j+1}-V_j\mid\mathcal F_j]
 =(QV-V)(\theta_j,\phi_j)
 \leq-(1-q_0)f(\theta_j,\phi_j)+C\delta^r.
\]

To derive \eqref{eq:stopped}, choose a deterministic integer $N$
with $\tau\leq N$. Since $\{j<\tau\}\in\mathcal F_j$, multiply this
conditional inequality by $\1_{\{j<\tau\}}$ and take expectations.
The pathwise identities
\[
 \sum_{j=0}^{N-1}\1_{\{j<\tau\}}(V_{j+1}-V_j)=V_\tau-V_0,
 \qquad \sum_{j=0}^{N-1}\1_{\{j<\tau\}}=\tau
\]
then give
\begin{equation}\label{eq:stopped-explicit}
 \begin{aligned}
 \mathbb E V_\tau-V_0
 &=\sum_{j=0}^{N-1}\mathbb E\!\left[
       \1_{\{j<\tau\}}(QV-V)(\theta_j,\phi_j)\right]\\
 &\leq-(1-q_0)\mathbb E\sum_{j<\tau}f(\theta_j,\phi_j)
       +C\delta^r\mathbb E\tau.
 \end{aligned}
\end{equation}
Rearranging and using $V_0\leq L$ and $V_\tau\geq0$ proves
\eqref{eq:stopped}. The constant $C$ comes from the fixed-$L$
comparison and is independent of $N$ and $\tau$.

<!-- lemma-8: body_md -->
\label{lem:mean}
Assume $E',E'+\delta\in I$ with $\Delta N_{E',\delta}>0$.
Start from deterministic phases with $0\leq D_0=\theta_0-\phi_0<\pi$,
and let $T=\inf\{j\geq0:D_j\geq\pi\}$ be the first crossing time.
Recall the initial value of the coordinate \eqref{eq:W}:
\[
 W_0=\mathcal H_{E'}(\theta_0)-\mathcal H_{E'+\delta}(\phi_0).
\]
The first crossing $T$ is integrable, $\pi \Delta N_{E',\delta}\mathbb E T\leq C$ uniformly, and
\[
 \left|\pi \Delta N_{E',\delta}\mathbb E T-(\pi-W_0)\right|
       \leq\omega_{\mathrm{exit}}(\delta).
\]
Here $C$ depends only on $\mu$ and the fixed interval $I$,
and is independent of $E'$, $\delta$, the initial phases, and $h$.
In particular, if the initial gap satisfies $D_0\geq\pi-h$,
where $0<h<\pi/2$, the remaining waiting time satisfies
\begin{equation}\label{eq:uppermean}
 \pi \Delta N_{E',\delta}\mathbb E T\leq\omega_{\mathrm{ph}}(h)

The mean identity follows by summing the conditional drift

\[
 \begin{aligned}
 \mathbb E W_{\tau_m}-W_0
 &=\sum_{j=0}^{m-1}\mathbb E\bigl[\1_{\{j<\tau_m\}}(W_{j+1}-W_j)\bigr]\\
 &=\pi \Delta N_{E',\delta}\sum_{j=0}^{m-1}\mathbb P(j<\tau_m)
 =\pi \Delta N_{E',\delta}\mathbb E\tau_m.
 \end{aligned}

expected total change in the coordinate $W$.

 \end{aligned}

<!-- lemma-9: body_md -->
\label{lem:killed}
Assume $E',E'+\delta\in I$ with $\Delta N_{E',\delta}>0$.
Uniformly in $\theta_0$ and $0\leq D_0<\pi$,
\begin{equation}\label{eq:gapoccupation}
 \pi \Delta N_{E',\delta}\mathbb E\sum_{j<T}|W_j|\longrightarrow0
 \quad(\delta\downarrow0).
\end{equation}
<!-- end lemma-9 -->
<!-- proof-10: body_md -->
We split the sum into three regions: near $0$, in the middle, and
near $\pi$. The coordinate is small in the first region; the time
spent in each of the other two is small on the IDS scale.
Throughout this proof, $C>0$ denotes a constant that may change from
line to line and depends only on $\mu$ and the fixed reference energy $E$.
In particular, it is independent of $E'$, $\delta$, the initial state,
and $h$.
Since $\mathbb E T<\infty$ and the stopped coordinate is uniformly bounded,
the bounded-stopping argument of Lemma~\ref{lem:mean}, applied at
$\sigma\wedge m$ and then passed to the limit, gives, for every
stopping time $\sigma\leq T$,
\begin{equation}\label{eq:stopped-drift-preexit}
 \pi \Delta N_{E',\delta}\mathbb E\sigma=\mathbb E W_\sigma-W_0.
\end{equation}
By Lemma~\ref{lem:mean}, we have
\begin{equation}\label{eq:scaled-mean-budget}
 \pi \Delta N_{E',\delta}\mathbb E T\leq C.
\end{equation}
Fix $0<h<\pi/4$.

\smallskip\noindent
\emph{1. Near $0$: the coordinate is small.}
If $D_j\leq h$, then
$|W_j|\leq\omega_{\mathrm{ph}}(h)+\omega_{\mathrm{en}}(\delta)$.
By \eqref{eq:scaled-mean-budget}, this region contributes at most
$C[\omega_{\mathrm{ph}}(h)+\omega_{\mathrm{en}}(\delta)]$
to the left side of \eqref{eq:gapoccupation}.

\smallskip\noindent
\emph{2. In the middle: contraction controls occupation.}
Apply \eqref{eq:stopped} to $T\wedge m$ and let $m\to\infty$.
Nonnegativity and monotone convergence give
\begin{equation}\label{eq:fkilled}
 \pi \Delta N_{E',\delta}\mathbb E\sum_{j<T}|\sin D_j|^r
 \leq\frac{L\pi \Delta N_{E',\delta}
                +C\delta^r(\pi \Delta N_{E',\delta}\mathbb E T)}{1-q_0}
 \leq C(\Delta N_{E',\delta}+\delta^r).
\end{equation}
On $h<D_j<\pi-h$, we have $|\sin D_j|\geq\sin h$,
while \eqref{eq:Wbounds} gives
$|W_j|\leq\pi+\omega_{\mathrm{en}}(\delta)$ for $j<T$.
Thus the middle region contributes at most
$C(\Delta N_{E',\delta}+\delta^r)/(\sin h)^r$.
For fixed $h>0$, this tends to zero as $\delta\downarrow0$;
the exponent $r$ from Lemma~\ref{lem:contraction} only needs to be
fixed and positive, and its numerical value is not needed.
The error in \eqref{eq:fkilled} is the rescaled quantity
$C\delta^r(\pi\Delta N_{E',\delta}\mathbb E T)\leq C\delta^r\to0$,
by \eqref{eq:scaled-mean-budget}. The unscaled $C\delta^r\mathbb E T$
need not tend to zero.

\smallskip\noindent
\emph{3. Near $\pi$: little remaining time on the IDS scale.}
Here $|W_j|$ need not be small, so we instead control how many
times the gap lies in $[\pi-h,\pi)$ before $T$. Let $\sigma_h$ be
the first time the gap reaches or exceeds $\pi-h$:
\[
 \sigma_h=\inf\{j\geq0:D_j\geq\pi-h\}.
\]
Since $D_T\geq\pi$ and $T<\infty$ almost surely, $\sigma_h\leq T$.
Every visit to $[\pi-h,\pi)$ before $T$ occurs at or after
$\sigma_h$, so pathwise
\[
 \sum_{j<T}\1_{\{D_j\geq\pi-h\}}\leq T-\sigma_h.
\]
We count all times from $\sigma_h$ to $T-1$, including times when
the gap has left this region. Thus this bound also covers later
returns, and it suffices to control $\mathbb E(T-\sigma_h)$.

At $\sigma_h$, we have
$\theta_{\sigma_h}\geq\phi_{\sigma_h}+\pi-h$. Monotonicity,
$\pi$-equivariance and the continuity moduli therefore give
\[
 \begin{aligned}
 W_{\sigma_h}
 &\geq\mathcal H_{E'}(\phi_{\sigma_h}+\pi-h)
                 -\mathcal H_{E'+\delta}(\phi_{\sigma_h})\\
 &=\pi+\mathcal H_{E'}(\phi_{\sigma_h}-h)
                 -\mathcal H_{E'+\delta}(\phi_{\sigma_h})\\
 &\geq\pi-\omega_{\mathrm{ph}}(h)-\omega_{\mathrm{en}}(\delta).
 \end{aligned}
\]
This calculation also applies when $\sigma_h=0$ or $\sigma_h=T$.
At exit, $W_T\leq\pi+\omega_{\mathrm{exit}}(\delta)$.
Subtracting the stopped-drift identities \eqref{eq:stopped-drift-preexit}
at $\sigma_h$ and $T$ now gives
\begin{equation}\label{eq:remaining}
 \pi\Delta N_{E',\delta}\mathbb E(T-\sigma_h)
 =\mathbb E W_T-\mathbb E W_{\sigma_h}
 \leq\omega_{\mathrm{ph}}(h)+\omega_{\mathrm{exit}}(\delta)
                              +\omega_{\mathrm{en}}(\delta).
\end{equation}
Thus, once the gap first reaches $\pi-h$, little mean waiting time
remains on the IDS scale. Since \eqref{eq:Wbounds} bounds $|W_j|$
uniformly before $T$, the contribution of this region is bounded by
\[
 \begin{aligned}
 \pi\Delta N_{E',\delta}\mathbb E\sum_{j<T}|W_j|
                           \1_{\{D_j\geq\pi-h\}}
 &\leq C\pi\Delta N_{E',\delta}\mathbb E(T-\sigma_h)\\
 &\leq C[\omega_{\mathrm{ph}}(h)+\omega_{\mathrm{exit}}(\delta)
                              +\omega_{\mathrm{en}}(\delta)].
 \end{aligned}
\]

\smallskip\noindent
\emph{Combining the three regions,} we obtain
\begin{equation}\label{eq:killed-error-ledger}
 \pi \Delta N_{E',\delta}\mathbb E\sum_{j<T}|W_j|
 \leq C\Bigl[\omega_{\mathrm{ph}}(h)
       +\frac{\Delta N_{E',\delta}+\delta^r}{(\sin h)^r}
       +\omega_{\mathrm{exit}}(\delta)
       +\omega_{\mathrm{en}}(\delta)\Bigr].
\end{equation}
The constant $C$ is independent of $h$. First let $\delta\to0$
at fixed $h$, then let $h\to0$. This proves the uniform limit.

<!-- theorem-2: body_md -->
\label{thm:exit}
Assume that the IDS $N$ is strictly increasing on an open interval
containing $E$. There is a compact neighborhood $I$ of $E$ inside
this interval such that the following statements hold uniformly
over $E',E'+\delta\in I$ with $\delta>0$, as $\delta\downarrow0$.
Here $\mathbb E_{\theta_0,D_0}$ averages over the common potentials with
deterministic initial phases $(\theta_0,\phi_0)=(\theta_0,\theta_0-D_0)$,
and $W_0$ is given by \eqref{eq:W}. For every fixed $q>0$,
\begin{equation}\label{eq:affineprofile}
 \sup_{\theta_0\in\mathbb R,\,0\leq D_0<\pi}
 \left|\mathbb E_{\theta_0,D_0}e^{-q\pi \Delta N_{E',\delta}T}
  -\frac{1+q W_0}{1+q\pi}\right|\longrightarrow0.
\end{equation}
Consequently, for every deterministic $h_\delta\downarrow0$,
uniformly in $\theta_0$ and $0\leq D_0\leq h_\delta$,
\begin{equation}\label{eq:exp}
 \Delta N_{E',\delta}T\ \Longrightarrow\ \operatorname{Exp}(1).
\end{equation}

<!-- proof-11: body_md -->
By Lemma~\ref{lem:mean}, $T$ is integrable and
$\pi\Delta N_{E',\delta}\mathbb E T\leq C$ uniformly.
We first prove the Laplace profile \eqref{eq:affineprofile}, then
deduce the exponential law \eqref{eq:exp} for small initial gaps.

\emph{The stopped drift gives a geometric discount.}
Fix $q>0$. We seek an affine function of $W$ whose discounted
expectation changes little up to $T$. Its conditional expectation
is determined by the exact drift \eqref{eq:drift}. The exit bound
\eqref{eq:Wexit} suggests normalizing this function to equal $1$
at $\pi$, giving $w\mapsto(1+qw)/(1+q\pi)$.
Set $\rho=(1+q\pi\Delta N_{E',\delta})^{-1}\in(0,1)$.
Then $\rho(1+q\pi\Delta N_{E',\delta})=1$ makes the drift cancel the
constant loss from discounting. The remaining term is proportional
to $W_j$, and Lemma~\ref{lem:killed} makes its accumulated contribution
before $T$ vanish. Comparing initial and terminal expectations will
therefore determine $\mathbb E\rho^T$.
Using \eqref{eq:drift} and $\mathcal F_j$-measurability of $W_j$,
\[
 \begin{aligned}
 &\mathbb E\bigl[\rho^{j+1}(1+qW_{j+1})-\rho^j(1+qW_j)\mid\mathcal F_j\bigr]\\
 &\quad=\rho^j\bigl\{\rho[1+q(W_j+\pi \Delta N_{E',\delta})]-(1+qW_j)\bigr\}\\
 &\quad=\rho^j q(\rho-1)W_j=-q^2\pi \Delta N_{E',\delta}\rho^{j+1}W_j,
 \end{aligned}
\]
where the last equality uses $\rho-1=-q\pi \Delta N_{E',\delta}\rho$.
Stop and telescope at $T\wedge m$. The uniform stopped bound on $W$
and $\mathbb E T<\infty$ justify dominated convergence on both sides, giving
\begin{equation}\label{eq:geometric-exit-identity}
 \mathbb E\bigl[\rho^T(1+qW_T)\bigr]-(1+qW_0)
       =-q^2\pi \Delta N_{E',\delta}\mathbb E\sum_{j<T}\rho^{j+1}W_j.
\end{equation}
Using $|W_T-\pi|\leq\omega_{\mathrm{exit}}(\delta)$ in
\eqref{eq:geometric-exit-identity} gives
\[
 \left|\mathbb E\rho^T-\frac{1+qW_0}{1+q\pi}\right|
 \leq\frac{q}{1+q\pi}\left[q\pi \Delta N_{E',\delta}\mathbb E\sum_{j<T}\rho^{j+1}|W_j|
              +\omega_{\mathrm{exit}}(\delta)\mathbb E\rho^T\right]\longrightarrow0.
\]
The limit follows from $0<\rho<1$ and Lemma~\ref{lem:killed}.

\emph{The mean bound allows an exponential discount.}
Write
\[
 a_{E',\delta}
   =\frac{\log(1+q\pi \Delta N_{E',\delta})}{\pi \Delta N_{E',\delta}}.
\]
Then $0<a_{E',\delta}\leq q$, $a_{E',\delta}\to q$, and
$\rho^T=e^{-a_{E',\delta}\pi \Delta N_{E',\delta}T}$.
The mean bound in Lemma~\ref{lem:mean} and the inequality
$|e^{-ax}-e^{-bx}|\leq|a-b|x$ for $a,b,x\geq0$ give
\[
 \left|\mathbb E\rho^T-\mathbb E e^{-q\pi \Delta N_{E',\delta}T}\right|
 \leq(q-a_{E',\delta})\pi \Delta N_{E',\delta}\mathbb E T\longrightarrow0.
\]
Combining these two limits proves \eqref{eq:affineprofile} uniformly.

\emph{Small initial gaps identify the exponential law.}
If $0\leq D_0\leq h_\delta\to0$, continuity of the coordinate gives
\[
 |W_0|\leq\omega_{\mathrm{ph}}(h_\delta)+\omega_{\mathrm{en}}(\delta)
             \longrightarrow0.
\]
For each $s>0$, take $q=s/\pi$ in \eqref{eq:affineprofile} to obtain,
uniformly over these initial states,
\[
 \mathbb E_{\theta_0,D_0}e^{-s \Delta N_{E',\delta}T}
       \longrightarrow\frac1{1+s}
       =\int_0^\infty e^{-st}e^{-t}\,dt.
\]
The limit is the Laplace transform of $\operatorname{Exp}(1)$.
The continuity theorem for Laplace transforms
\cite[Chapter~XIII, Section~1]{Feller1971} therefore gives
\eqref{eq:exp}. Applying this theorem along every allowed sequence
of energies and initial states proves the asserted uniformity.

Suppose $E'_{i,n}\to E$ for $0\leq i\leq s$, and
$n\Delta N_{i,n}\to w_i>0$ for $1\leq i\leq s$. Then the crossing
counts in the disjoint energy intervals $(E'_{i-1,n},E'_{i,n}]$,
$1\leq i\leq s$, are asymptotically independent:
\[
 (K_1(n),\ldots,K_s(n))\Longrightarrow (Z_1,\ldots,Z_s),
\]
where the $Z_i$ are independent Poisson variables of means $w_i$.

<!-- proof-17: body_md -->
The subdivision argument below is related to the phase-counting proofs
of independence in Allez and Dumaz \cite[Section~3.2]{AD14} and
Kotani and Nakano \cite[Section~5.3]{KN17}.

We divide the sites into $m$ blocks; after $n\to\infty$ at fixed $m$,
the total error from blocks with multiple crossings is $O(m^{-1})$,
which vanishes as $m\to\infty$.
With high probability, the single-crossing conditional probabilities
are approximately $w_i/m$, uniformly over the past, so successive
conditioning yields the independent Poisson limit.

\smallskip
\emph{Block starts and the full energy interval.}
The floor operation gives the exact identity
\begin{equation}\label{eq:joint-floor-identity}
 K_*(j)=\sum_{i=1}^s K_i(j)+\left\lfloor\frac{\sum_{i=1}^s R_i(j)}\pi\right\rfloor.
\end{equation}
Fix $m\geq1$ and define the block endpoints by
\[
 j_b=\left\lfloor\frac{bn}{m}\right\rfloor,\qquad 0\leq b\leq m.
\]
Block $b$ uses the potentials $v_j$ with
\[
 j_{b-1}\leq j<j_b,\qquad 1\leq b\leq m.
\]
Here $b$ labels spatial blocks, while $i$ labels energy windows.
We first let $n\to\infty$ with $m$ fixed.
Lemma~\ref{lem:residual-pinning} applies to each energy pair at the
finitely many positive times $(b-1)/m$, while $R_i(0)=0$.
Since $s$ and $m$ are fixed, choose a common deterministic
$h_n\downarrow0$ such that, for $1\leq b\leq m$, the events
\[
 G_{b,n}=\left\{\max_{1\leq i\leq s}R_i(j_{b-1})\leq h_n\right\}
          \in\mathcal F_{j_{b-1}}
\]
satisfy $\max_{1\leq b\leq m}\mathbb P(G_{b,n}^{\complement})\to0$.
On $G_{b,n}$, the restarted adjacent gaps are at most $h_n$,
and the gap for the full interval is at most $sh_n$.

Let $w_* =\sum_{i=1}^s w_i$ and define the block increments by
\[
 \Delta K_i^{(b)}=K_i(j_b)-K_i(j_{b-1}),\qquad
 \Delta K_*^{(b)}=K_*(j_b)-K_*(j_{b-1}).
\]
On $G_{b,n}$, $\sum_{i=1}^s R_i(j_{b-1})\leq sh_n<\pi$ for
large $n$, so the initial floor in \eqref{eq:joint-floor-identity}
vanishes. Subtracting that identity at the two block endpoints gives
\[
 \sum_{i=1}^s\Delta K_i^{(b)}
 =\Delta K_*^{(b)}
  -\left\lfloor\frac{\sum_{i=1}^s R_i(j_b)}\pi\right\rfloor
 \leq\Delta K_*^{(b)}.
\]
The terminal floor is nonnegative because each $R_i(j_b)\geq0$.

\smallskip
\emph{Crossings within one spatial block.}
The count $K_*$ lets us control, for every block $b$, two or more
crossings across the energy windows through the single event
$\{\Delta K_*^{(b)}\geq2\}$.
After removing the completed periods from $D_i$ at $j_{b-1}$,
$\Delta K_i^{(b)}\geq k$ means exactly that the restarted $T_k$
is at most $j_b-j_{b-1}$. Since
$\Delta N_{i,n}(j_b-j_{b-1})\to w_i/m$,
\eqref{eq:uniform-arrival-kernel}, also applied to the two extreme energies,
gives a common deterministic $\varepsilon_n(m)\to0$ such that,
on $G_{b,n}$,
\begin{align*}
 \left|\mathbb P(\Delta K_i^{(b)}\geq1\mid\mathcal F_{j_{b-1}})
                    -(1-e^{-w_i/m})\right|&\leq\varepsilon_n(m),\\
 \mathbb P(\Delta K_*^{(b)}\geq2\mid\mathcal F_{j_{b-1}})
                    &\leq\frac{w_*^2}{2m^2}+\varepsilon_n(m).
\end{align*}
The error is uniform in the restart phases, in gaps at most
$sh_n$, and in the finitely many block and window indices.
The deterministic telescoping identity
$\Delta N_{*,n}=\sum_{i=1}^s\Delta N_{i,n}$ gives
$n\Delta N_{*,n}\to w_*$. Thus the single-interval law
\eqref{eq:uniform-arrival-kernel}, applied directly to $E'_{0,n}$ and
$E'_{s,n}$, bounds the second probability by
$\Gamma_2(w_*/m)+\varepsilon_n(m)$, without assuming independence
between the smaller intervals. Finally, \eqref{eq:gamma-small-time}
gives $\Gamma_2(w_*/m)\leq w_*^2/(2m^2)$.

For fixed $t_1,\ldots,t_s\geq0$, define
$\Lambda=\sum_{i=1}^s w_i(1-e^{-t_i})$.
On $G_{b,n}$, separating zero or one crossing from multiple crossings gives
\[
 \begin{aligned}
 &\mathbb E\!\left[e^{-\sum_{i=1}^s t_i\Delta K_i^{(b)}}\mid\mathcal F_{j_{b-1}}\right]\\
 &\quad=1-\sum_{i=1}^s(1-e^{-t_i})
       \mathbb P(\Delta K_i^{(b)}\geq1\mid\mathcal F_{j_{b-1}})
       +O(m^{-2}+\varepsilon_n(m)).
 \end{aligned}
\]
The error covers multiple crossings, which require $\Delta K_*^{(b)}\geq2$.
Using $1-e^{-w_i/m}=w_i/m+O(m^{-2})$ yields
\begin{equation}\label{eq:joint-block-kernel}
 \left|\mathbb E\!\left[e^{-\sum_{i=1}^s t_i\Delta K_i^{(b)}}
                    \mid\mathcal F_{j_{b-1}}\right]-(1-\Lambda/m)\right|
 \leq C m^{-2}+C\varepsilon_n(m).
\end{equation}
Here $C$ depends only on the fixed windows and their limiting rates.

\smallskip
\emph{Combining the spatial blocks.}
Take $m>w_*$, so $1-\Lambda/m\in[0,1]$.
Take conditional expectations from the last block backwards, using
\eqref{eq:joint-block-kernel} on each $G_{b,n}$ and the bound $2$
on its complement. All earlier block factors are measurable and
bounded by one, giving
\begin{equation}\label{eq:joint-backward-error}
 \left|\mathbb E e^{-\sum_{i=1}^s t_iK_i(n)}-(1-\Lambda/m)^m\right|
 \leq C/m+Cm\varepsilon_n(m)
             +2\sum_{b=1}^m\mathbb P(G_{b,n}^{\complement}).
\end{equation}
First let $n\to\infty$ at fixed $m$, and then let $m\to\infty$.
The joint Laplace transform then converges to
$\exp[-\sum_{i=1}^s w_i(1-e^{-t_i})]$.
The multivariate Laplace continuity theorem
\cite[Proposition~B.10]{LastPenrose2018} proves the asserted
independent Poisson limit.

<!-- lemma-13: body_md -->
\label{lem:spectral-count-conversion}
Let the ordered endpoint arrays satisfy the hypotheses of
Proposition~\ref{prop:joint-crossing-counts}, and suppose that for
some fixed $k\in\{0,\ldots,s\}$ one has $E'_{k,n}=E$ for every $n$.
Denote the spectral counts by
\[
 C_i(n)=\#\{\lambda_{j,n}\in(E'_{i-1,n},E'_{i,n}]\}.
\]
Then
\[
 \mathbb P\bigl(C_i(n)=K_i(n)\text{ for every }1\leq i\leq s\bigr)
       \longrightarrow1.
\]
<!-- end lemma-13 -->
<!-- proof-18: body_md -->
Lemma~\ref{lem:sturm} expresses $C_i(n)$ as a difference of ceilings
of the endpoint phases, measured from the Dirichlet boundary
$\pi/2$ modulo $\pi$. Since $D_i(n)=\pi K_i(n)+R_i(n)$, taking
the integer $K_i(n)$ outside the ceiling gives
\begin{equation}\label{eq:exact-rounding-error}
 C_i(n)-K_i(n)
 =\left\lceil\frac{\theta_n(E'_{i,n})-\pi/2+R_i(n)}\pi\right\rceil
  -\left\lceil\frac{\theta_n(E'_{i,n})-\pi/2}\pi\right\rceil
 \in\{0,1\}.
\end{equation}
The difference can be nonzero only if the interval of length $R_i(n)$
starting at $\theta_n(E'_{i,n})$ contains a boundary angle. In particular,
\[
 C_i(n)\ne K_i(n)
 \quad\Longrightarrow\quad
 \operatorname{dist}_{\pi}(\theta_n(E'_{i,n}),\pi/2)\leq R_i(n).
\]

Fix $h>0$. If $\max_{1\leq i\leq s}R_i(n)\leq h$, consecutive
endpoint phases are at distance at most $h$ modulo $\pi$.
Starting from $E'_{k,n}=E$, at most $s$ such steps place every
endpoint phase within distance $sh$ of $\theta_n(E)$.
Thus a counting mismatch requires either a residual larger than $h$
or the fixed-energy phase to be within $(s+1)h$ of the boundary:
\[
 \mathbb P(\exists i:C_i(n)\ne K_i(n))
 \leq \mathbb P\!\left(\max_{1\leq i\leq s}R_i(n)>h\right)
  +\mathbb P\!\left(\operatorname{dist}_{\pi}
           (\theta_n(E),\pi/2)\leq(s+1)h\right).
\]
For fixed $h$, the first term tends to zero by
Lemma~\ref{lem:residual-pinning}.
Lemma~\ref{lem:forward-stationary-limit}, applied with $\beta=\pi/2$,
makes the limit superior of the second term tend to zero as
$h\downarrow0$. This proves the assertion.

<!-- proposition-5: body_md -->
\label{prop:bernoullibulk}
For a nondegenerate two-point potential taking values $a<b$, the
local strict-increase hypothesis of the point-process theorem holds
at every point in the interior of
\[
 \Sigma=[a-2,a+2]\cup[b-2,b+2].
\]
Consequently Theorem~\ref{thm:poisson}(\ref{part:ids-unfolding}) covers these
Bernoulli bulk centers without an additional regularity assumption on
the IDS.
<!-- end proposition-5 -->
<!-- proof-19: body_md -->
Fix $u<v$ such that $(u,v)$ meets $(c-2,c+2)$ for some
$c\in\{a,b\}$. On $L$ sites with potential equal to $c$ at every
site, the Dirichlet eigenvalues are
\[
 c+2\cos\frac{k\pi}{L+1},\qquad 1\leq k\leq L.
\]
This follows by substitution of the sine vectors
\cite[Section~1.3]{Teschl2000}. Since the angles have spacing
$\pi/(L+1)$, we can choose $L$ large enough that at least nine
eigenvalues lie in $(u,v)$. Fix this $L$ for the rest of the argument.

For $n$ a multiple of $L$, divide the $n$ sites into $n/L$ consecutive
intervals of length $L$, and select those on which every potential
equals $c$. Write $\alpha=\mu(\{c\})>0$. Each interval is selected
with probability $\alpha^L$, so the expected number selected is
$(n/L)\alpha^L$. Since $L$ is fixed, this probability stays positive
as $n$ grows.

Cut the bonds at the ends of the selected intervals. There are at
most two cuts per selected interval, and each cut changes the matrix
by rank at most two. Rank comparison at $u$ and $v$
\cite[Lemma~2.2]{Bai1999} therefore bounds the total change in the
number of eigenvalues in $(u,v]$ by eight times the number of selected
intervals. After cutting, each selected interval contributes at least
nine eigenvalues, and the remaining pieces contribute nonnegative
counts. Thus the original chain has at least as many eigenvalues in
$(u,v]$ as there are selected intervals. Taking expectations gives
\[
 \frac1n\mathbb E\#\{\lambda_{j,n}\in(u,v]\}
       \geq\frac1n\frac{n}{L}\alpha^L
       =\frac{\alpha^L}{L}>0.
\]
Here $\alpha>0$ and the finite integer $L$ are fixed independently of
$n$, so $\alpha^L/L$ is a fixed positive lower bound. Taking the IDS
limit along multiples of $L$ proves $N(v)-N(u)\geq\alpha^L/L>0$.

Finally, let $E\in\operatorname{int}\Sigma$ and choose an open interval
$I$ containing $E$ with $I\subset\Sigma$. Every nonempty open
subinterval of $I$ meets $(a-2,a+2)$ or $(b-2,b+2)$, so the preceding
argument gives $N(u)<N(v)$ for all $u<v$ in $I$. This includes the
case where the two closed bands touch at $E$.
Continuity was established in Lemma~\ref{lem:continuity}.

<!-- lemma-14: body_md -->
\label{lem:grid-poisson}
Let $\zeta_n$ be random locally finite counting measures on
$\mathbb R$. Suppose that, for every $R>0$ and every finite partition
\[
 -R=x_0<x_1<\cdots<x_s=R
\]
with $0\in\{x_0,\ldots,x_s\}$ and $x_{s-i}=-x_i$ for $0\leq i\leq s$,
one has
\[
 \bigl(\zeta_n((x_{i-1},x_i])\bigr)_{i=1}^{s}
 \Longrightarrow (Z_1,\ldots,Z_s),
\]
where $Z_1,\ldots,Z_s$ are independent and
$Z_i\sim\operatorname{Poisson}(x_i-x_{i-1})$ for $1\leq i\leq s$.
Then $\zeta_n\Longrightarrow\Pi$ in distribution for the vague topology.
<!-- end lemma-14 -->
<!-- proof-20: body_md -->
For $0\leq f\in C_c(\mathbb R)$, choose $R$ with
$\operatorname{supp}f\subset(-R,R)$. The two-cell count law makes
$X_n=\zeta_n((-R,R])$ tight. If $g$ is a nonnegative step function
constant on the intervals of such a partition, with
$\|f-g\|_\infty\leq\eta$ and zero outside $(-R,R]$, then
\begin{equation}\label{eq:laplace-tail-truncation}
 \left|\mathbb E e^{-\int f\,d\zeta_n}-\mathbb E e^{-\int g\,d\zeta_n}\right|
       \leq\eta M+\mathbb P(X_n>M).
\end{equation}
The joint count convergence gives the Laplace limit for $g$. Taking $M$ large
and then $\eta$ small, with the same bound for $\Pi$, gives the limit
for $f$. The standard Laplace-functional criterion
\cite[Proposition~3.19]{Resnick1987} proves the assertion.

\textit{IDS unfolding.}
Recall the IDS-unfolded point process
\[
 \xi_{n,E}=\sum_{j=1}^n
          \delta_{\,n[N(\lambda_{j,n})-N(E)]}.
\]
Fix $R>0$ and an integer $k\geq1$, set $s=2k$, and choose a partition
\[
 -R=x_0<\cdots<x_k=0<\cdots<x_s=R,\qquad
 x_{s-i}=-x_i\quad(0\leq i\leq s).
\]
Choose an open interval $J$ containing $E$ on which $N$ is continuous
and strictly increasing. For all sufficiently large $n$, define the
unique energies $E'_{i,n}\in J$ by
\[
 N(E'_{i,n})=N(E)+\frac{x_i}{n},\qquad 0\leq i\leq s.
\]
Continuity of the local inverse gives $E'_{i,n}\to E$, with
$E'_{k,n}=E$ exactly, and
\[
 n[N(E'_{i,n})-N(E'_{i-1,n})]=x_i-x_{i-1}>0,
 \qquad 1\leq i\leq s.
\]

Since $N(E)$ lies in the interior of $N(J)$, global monotonicity
excludes energies outside $J$ from the unfolded window $[-R,R]$
for large $n$. For each eigenvalue, strict monotonicity inside $J$ gives
\[
 \begin{aligned}
 \lambda_{j,n}\in(E'_{i-1,n},E'_{i,n}]
 &\iff N(E'_{i-1,n})<N(\lambda_{j,n})\leq N(E'_{i,n})\\
 &\iff x_{i-1}<n[N(\lambda_{j,n})-N(E)]\leq x_i.
 \end{aligned}
\]
Counting these eigenvalues yields
\[
 \xi_{n,E}((x_{i-1},x_i])
 =\#\{1\leq j\leq n:\lambda_{j,n}\in(E'_{i-1,n},E'_{i,n}]\},
 \qquad 1\leq i\leq s.
\]
Proposition~\ref{prop:joint-crossing-counts} and
Lemma~\ref{lem:spectral-count-conversion} show that these counts
converge jointly to independent Poisson variables with means
$x_i-x_{i-1}$. Lemma~\ref{lem:grid-poisson} therefore proves
$\xi_{n,E}\Longrightarrow\Pi$.
Proposition~\ref{prop:bernoullibulk} verifies the required IDS hypothesis
at every stated Bernoulli bulk center.

\textit{Linear scaling.}
If $0<\rho(E)=N'(E)<\infty$, use instead
\[
 \widetilde E'_n(x)=E+\frac{x}{n\rho(E)}.
\]
These endpoints again converge to $E$ and include the exact anchor
$\widetilde E'_n(0)=E$. Differentiability at this single point gives
\[
 n[N(\widetilde E'_n(x))-N(E)]\longrightarrow x.
\]
Every adjacent rate consequently tends to $x_i-x_{i-1}>0$.
Proposition~\ref{prop:joint-crossing-counts} and
Lemma~\ref{lem:spectral-count-conversion} also give the grid-count limits
for the linearly rescaled point process.
Lemma~\ref{lem:grid-poisson} completes the proof. No local inverse
or differentiability away from $E$ is used in this part.
