sequences of packet-generation intervals and system delays,
$(G_n,D_n)_{n=0,1,\ldots}$. In specific queueing models, however, a system is

<!-- assumption-1: body_md -->
\label{ass:regularity}
\begin{enumerate}
\par\hangindent 40\p@[(i)] The stationary mean AoI $\mathbb{E}_\lambda[A]$ is well defined, finite, and continuously differentiable on $\lambda \in \mathcal{L} = [\lambda_{\min}, \lambda_{\max}]$.

\par\hangindent 40\p@[(ii)] The gradient of $\mathbb{E}_\lambda[A]$ admits the representation \eqref{eq:g(lambda,theta)-def} for a known function $g$ and a parameter vector $\bm{\theta}_{\lambda} \in \Theta$, where $\Theta \subseteq \mathbb{R}^d$ denotes the parameter space.

\par\hangindent 40\p@[(iii)] There exists a unique $\lambda_* \in \mathcal{L}$ such that
\[
g(\lambda_*;\bm{\theta}_{\lambda_*}) + \kappa'(\lambda_*) = 0.
\]

\par\hangindent 40\p@[(iv)] There exists a constant $C_1 > 0$ such that, for all $\bm{\theta}, \bm{\theta}' \in \Theta$ and all $\lambda\in{\mathcal L}$,
\begin{equation} \label{eq:Lips}
|g(\lambda;\bm{\theta}) - g(\lambda;\bm{\theta}')| \le C_1 \|\bm{\theta} - \bm{\theta}'\|.
\end{equation}
\par\hangindent 40\p@[(v)] There exists a constant $C_2 > 0$ such that,  for all $\lambda\in{\mathcal L}$,
 
\begin{equation}\label{eq:theta_mixing}
\mathbb{E}_{\lambda^{\langle k\rangle}}\!\left[ \|\bm{\theta}_{\lambda^{\langle k\rangle}} - \hat{\bm{\theta}}^{\langle k\rangle}\|^2\, \Big|\, \,
\mathcal{F}^{\langle k-1\rangle}, \lambda^{\langle k\rangle} \right]
\le \frac{C_2}{n^{\langle k\rangle}}, \qquad k=1,2,\ldots,
\end{equation}
where $\mathcal{F}^{\langle k-1\rangle}$ denotes the sigma algebra generated by the observations in all iterations before the $k$-th decision epoch.
 
\end{enumerate}

<!-- theorem-1: body_md -->
\label{thm:SGD_convergence_2}
Suppose that Assumption~\ref{ass:regularity} holds. Then the proposed SGD algorithm, with step size $\gamma^{\langle k\rangle}$ and sample size $n^{\langle k\rangle}$ given in \eqref{eq:step-size} and \eqref{eq:sample-size}, with $\beta > 2(1-\alpha)$,
generates a sequence of update rates $(\lambda^{\langle k\rangle})_{k=0,1,\ldots}$ that converges almost surely to the minimizer $\lambda_*$ of \eqref{eq:objective}, i.e.,
 
\begin{align}
\lambda^{\langle k\rangle} \xrightarrow{\text{\rm a.s.}} \lambda_*.
\label{eq:lambda_convergence}
\end{align}

<!-- proof-1: body_md -->
{Let $B^{\langle k\rangle}$ denote the estimation bias of the gradient at iteration $k$, defined by
 
\begin{equation}\label{eq:bias_g}
B^{\langle k\rangle}
:=
g(\lambda^{\langle k\rangle}; \bm{\theta}_{\lambda^{\langle k\rangle}})
-
g(\lambda^{\langle k\rangle};\hat{\bm{\theta}}^{\langle k\rangle}),
\qquad k = 1,2,\ldots.
\end{equation}
 
Since both $\lambda^{\langle k\rangle}$ and $\hat{\bm{\theta}}^{\langle k\rangle}$ are random, $B^{\langle k\rangle}$ is a random variable.
The iteration \eqref{eq:update-lambda} can then be rewritten as
 
\begin{align*}
\lambda^{\langle k+1\rangle}
&=
\lambda^{\langle k\rangle}
-\gamma^{\langle k\rangle}
\left\{
g\!\left(\lambda^{\langle k\rangle};\bm{\theta}_{\lambda^{\langle k\rangle}}\right)
+\kappa'(\lambda^{\langle k\rangle})
\right\}
+\gamma^{\langle k\rangle} B^{\langle k\rangle}
+ Y^{\langle k\rangle},
\end{align*}
 
where the `projection residual' $Y^{\langle k\rangle}$ is defined via
 
\[
Y^{\langle k\rangle}
:=
\Pi_{\mathcal{L}}\!\left[\lambda^{\langle k\rangle} - \gamma^{\langle k\rangle} \hat{G}^{\langle k\rangle}\right]
-
\left(\lambda^{\langle k\rangle} - \gamma^{\langle k\rangle} \hat{G}^{\langle k\rangle}\right),
\quad
\hat{G}^{\langle k\rangle}
:=
g\!\left(\lambda^{\langle k\rangle};\hat{\bm{\theta}}^{\langle k\rangle}\right)
+\kappa'(\lambda^{\langle k\rangle}).
\]
 
The term $Y^{\langle k\rangle}$ is nonzero only when the unconstrained update leaves the feasible set $\mathcal{L}$, and in that case it represents the correction enforcing projection onto $\mathcal{L}$.

Let $\mathcal{F}^{\langle k-1\rangle}$ denote the sigma algebra generated by the observations in all iterations before the $k$-th decision epoch, i.e., 
 
\[
\mathcal{F}^{\langle k-1\rangle}:=\sigma\left((D_1^{\dag},G_1^{\dag}),(D_2^{\dag},G_2^{\dag})\ldots,(D^\dag_{\ell^{\langle k\rangle}},G^\dag_{\ell^{{\langle k\rangle}}}), \lambda^{\langle k-1\rangle}\right).
\]
 
Recall that $\lambda^{\langle k\rangle}$ is determined by $\lambda^{\langle k-1\rangle}$ and the data 
$((G_{\ell}^{\dag}, D_{\ell}^{\dag}))_{\ell \in \{\ell^{\langle k-1\rangle}+1,\ell^{\langle k-1\rangle}+2,\ldots,\ell^{\langle k\rangle}\}}$ observed in iteration $k-1$ (cf.\ \eqref{eq:hat-theta-def} and \eqref{eq:update-lambda}). 
Furthermore, we define $b^{\langle k\rangle}$ as the absolute mean bias:
\[
b^{\langle k\rangle} := \left| \mathbb{E}\!\left[ B^{\langle k\rangle} \, \Big|\, \lambda^{\langle k\rangle}, \mathcal{F}^{\langle k-1\rangle} \right] \right|.
\] 
Assumption~\ref{ass:regularity} (iii) ensures that 
the optimal $\lambda_*$ is the unique solution of \eqref{eq:zero_gradient}.
Therefore, owing to \cite[Page 127]{KY2003},
the almost sure convergence \eqref{eq:lambda_convergence} holds if the following conditions are satisfied:
   
\begin{enumerate}
 
\par\hangindent 40\p@[(i)] The step size sequence satisfies
\begin{align*}
\sum_{k=1}^\infty \gamma^{\langle k\rangle}=\infty, 
\quad 
\sum_{k=1}^\infty (\gamma^{\langle k\rangle})^2 < \infty.
\end{align*} 
 
\par\hangindent 40\p@[(ii)] There exists a constant $M<\infty$ such that for any $\lambda^{\langle k\rangle}\in[\lambda_{\min},\lambda_{\max}]$ and $n^{\langle k\rangle}$,
 
\begin{equation}\label{eq:variance_bound}
\mathbb{E}\!\left[\left(B^{\langle k\rangle}\right)^2 \, \Big|\,  \lambda^{\langle k\rangle}, \mathcal{F}^{\langle k-1\rangle} \right] \le M.
\end{equation}
 
\par\hangindent 40\p@[(iii)] The sample size sequence $(n^{\langle k\rangle})_{k=1,2,\ldots}$ satisfies, almost surely,
\begin{equation}\label{eq:bias}
\sum_{k=1}^\infty \gamma^{\langle k\rangle}|b^{\langle k\rangle}|<\infty .
\end{equation}
 
\end{enumerate} 
 
It is readily verified that $\gamma^{\langle k\rangle}$ given by \eqref{eq:step-size} satisfies (i).
Therefore, the proof requires verifying (ii) and (iii).


First of all, from Assumption~\ref{ass:regularity} (iv) and the definition
\eqref{eq:bias_g} of $B^{\langle k\rangle}$, we have
 
\begin{align*}
\mathbb{E}\!\left[ \left(B^{\langle k\rangle}\right)^2 \, \Big|\,
\mathcal{F}^{\langle k-1\rangle}, \lambda^{\langle k\rangle} \right]
&\le C_1^2 \,
\mathbb{E}\!\left[
\left\|
\bm{\theta}_{\lambda^{\langle k\rangle}}
-
\hat{\bm{\theta}}^{\langle k\rangle}
\right\|^2
\, \Big|\,
\mathcal{F}^{\langle k-1\rangle}, \lambda^{\langle k\rangle}
\right].
\end{align*}
 
Assumption~\ref{ass:regularity} (v) thus implies
 
\begin{align*}
\mathbb{E}\!\left[
\left(B^{\langle k\rangle}\right)^2 \, \Big|\,
\mathcal{F}^{\langle k-1\rangle},\lambda^{\langle k\rangle}
\right]
\leq
\frac{C_1^2 C_2}{n^{\langle k\rangle}}.
\end{align*}
 
Since $n^{\langle k\rangle}$ is an increasing sequence of positive integers, we
conclude that \eqref{eq:variance_bound} holds with
$M=C_1^2C_2$.
 
Furthermore, by the Cauchy--Schwarz inequality,
 
\begin{align*}
|b^{\langle k\rangle}|
&=
\left|
\mathbb{E}\!\left[
B^{\langle k\rangle} \, \Big|\,
\mathcal{F}^{\langle k-1\rangle}, \lambda^{\langle k\rangle}
\right]
\right|\\
&\le
\sqrt{
\mathbb{E}\!\left[
\left(B^{\langle k\rangle}\right)^2 \, \Big|\,
\mathcal{F}^{\langle k-1\rangle}, \lambda^{\langle k\rangle}
\right]
}
\leq
\frac{C_1\sqrt{C_2}}{\sqrt{n^{\langle k\rangle}}}.
\end{align*}
 
Therefore, we obtain \eqref{eq:bias} from \eqref{eq:step-size} and
\eqref{eq:sample-size}, as follows:
 
\begin{align*}
\sum_{k=1}^\infty \gamma^{\langle k\rangle}|b^{\langle k\rangle}|
&\leq
C_1\sqrt{C_2}
\sum_{k=1}^\infty
\frac{c_1 k^{-\alpha}}{\sqrt{c_2 k^{\beta}}}
=
\frac{C_1\sqrt{C_2}\,c_1}{\sqrt{c_2}}
\sum_{k=1}^\infty
\frac{1}{k^{\alpha+\beta/2}}
<\infty,
\end{align*}
 
recalling the condition $\beta > 2(1-\alpha)$.
}

For ease of presentation, we adopt the specific choices \eqref{eq:step-size} and \eqref{eq:sample-size} for the step size $\gamma^{\langle k\rangle}$ and sample size $n^{\langle k\rangle}$. These particular forms are not essential for the convergence result of Theorem~\ref{thm:SGD_convergence_2}. The proof only requires that the sequences satisfy
\[
\sum_{k=1}^{\infty} \gamma^{\langle k\rangle} = \infty,
\qquad
\sum_{k=1}^{\infty} (\gamma^{\langle k\rangle})^2 < \infty,
\qquad

<!-- lemma-5: body_md -->
\label{lem:ineq}
For any non-negative random variable $Y$ with $\mathbb{E}[Y] \leq 1$, we have
 
\begin{align}
\mathbb{E}[Ye^{-Y}] &\leq \mathbb{E}[e^{-Y}],
\label{eq:YexpY-bound}
\\
\mathbb{E}[Y^2e^{-Y}] &\leq \mathbb{E}[e^{-Y}].
\label{eq:Y2expY-bound}
\end{align}
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
Since $e^{-Y}$ is a decreasing function of $Y$, the random variables
$Y$ and $e^{-Y}$ have non-positive covariance (see, e.g., \cite{Behboodian1994}). Hence,
\[
    \mathbb{E}[Ye^{-Y}]
    \leq
    \mathbb{E}[Y]\mathbb{E}[e^{-Y}]
    \leq
    \mathbb{E}[e^{-Y}],
\]
where the last inequality follows from $\mathbb{E}[Y]\leq1$. This proves
\eqref{eq:YexpY-bound}.

To prove \eqref{eq:Y2expY-bound}, it suffices to show that
$
\mathbb{E}[(Y^2-1)e^{-Y}] \leq 0.$
Observe that
\begin{equation}
(y^2-1)e^{-y}\leq 2e^{-1}(y-1),
\qquad y\ge0.
\label{eq:linear-UB}
\end{equation}
Indeed,
\[
    \frac{(y^2-1)e^{-y}}{y-1}=(y+1)e^{-y},
\]
and the function $\gamma(y):=(y+1)e^{-y}$ is decreasing on $[0,\infty)$ since
$
\gamma'(y)=-ye^{-y}\le0.
$
Therefore,
\[
    (y+1)e^{-y}
    \begin{cases}
    \ge 2e^{-1}, & 0\le y\le1,\\
    \le 2e^{-1}, & y>1,
    \end{cases}
\]
which is equivalent to \eqref{eq:linear-UB}. Consequently,
\[
    \mathbb{E}[(Y^2-1)e^{-Y}]
    \le
    2e^{-1}\mathbb{E}[Y-1]
    =
    2e^{-1}(\mathbb{E}[Y]-1)
    \le0,
\]
thus establishing \eqref{eq:Y2expY-bound}.

(i)~Let $W_i$ denote the waiting time in the queue of tagged packet $i$. Then
    $D_i=W_i+H_i$. The sequence $(H_i)_{i\in\mathbb N}$ is i.i.d., while
    $(W_i)_{i\in\mathbb N}$ is regenerative, with regeneration epochs given
    by arrivals that find the queue empty. Hence, defining $X_i:=f(D_i,\lambda)$,
    the sequence $(X_i)_{i\in\mathbb N}$ is also regenerative, with the same
    regeneration epochs. Let $N_j$ denote the length of regeneration cycle $j$,
    and define
    \begin{align*}
    C_1&:=\left\{1, 2, \ldots, N_1\right\},\\
    C_j&:=\left\{\sum_{k=1}^{j-1}N_k+1,
    \sum_{k=1}^{j-1}N_k+2, \ldots, 
    \sum_{k=1}^{j}N_k\right\},
    \qquad j=2,3,\ldots,\\
    U_j&:=\sum_{i\in C_j}\left(X_i-\mathbb{E}_\lambda[X]\right).
    \end{align*}
    If $D_0\leq w$, then the initial workload is also bounded, since
    $W_0\leq D_0\leq w$. The distribution of the initial cycle
    $(N_1,U_1,C_1)$ may therefore depend on $D_0$, whereas the subsequent
    cycles $(N_2,U_2,C_2), (N_3,U_3,C_3),\ldots$  are i.i.d.\ and independent of the
    initial state.

    Since we assume $0 \leq f(d,\lambda) \leq1$ ($d \geq 0$), we have $0\leq X_i\leq1$ and
    $0\leq\mathbb{E}_\lambda[X]\leq1$. It then follows that
    $|U_j|\leq N_j$ and $\mathbb{E}_\lambda[U_j^2]\leq\mathbb{E}_\lambda[N_j^2]$.
    Moreover, $\mathbb{E}_\lambda[U_j]=0$ for $j=2,3,\ldots$.

    For the $n$-th observation, write $T_n$ for the index of the regeneration
    cycle containing it. Letting
     
    \[
        R_n:=\sum_{\substack{i\in C_{T_n}\\i>n}}
        \left(X_i-\mathbb{E}_\lambda[X]\right)
    \]
     
    denote the centered reward remaining until the end of that cycle, we have
    \[
        \sum_{i=1}^n\left(X_i-\mathbb{E}_\lambda[X]\right)
        =\sum_{j=1}^{T_n}U_j-R_n,
        \qquad |R_n|\leq N_{T_n}.
    \]
    For $j\geq2$, the cycle lengths $N_j$ are i.i.d.; write $N$ for their
    common distribution. Since $T_n\leq n$,
    \[
        \sum_{j=2}^{T_n}U_j
        =\sum_{j=2}^nU_j\mathbf{1}_{\{T_n\geq j\}},
        \qquad
        \{T_n\geq j\}
        =\left\{\sum_{r=1}^{j-1}N_r<n\right\}.
    \]
    Hence, $\mathbf{1}_{\{T_n\geq j\}}$ is determined by the regeneration
    cycles preceding cycle $j$. Thus, for $2\leq j<k\leq n$,
    \begin{align*}
    \mathbb{E}_\lambda\!\left[
    U_jU_k\mathbf{1}_{\{T_n\geq j\}}
    \mathbf{1}_{\{T_n\geq k\}}
    \,\middle|\,D_0=d_0\right]
    &=
    \mathbb{E}_\lambda\!\left[
    U_jU_k\mathbf{1}_{\{T_n\geq k\}}
    \,\middle|\,D_0=d_0\right]=0,
    \end{align*}
    where the first equality uses
    $\{T_n\geq k\}\subseteq\{T_n\geq j\}$. For the second equality, note that
    $\{T_n\geq k\}$ is determined by the cycles preceding cycle $k$, whereas
    $U_k$ is independent of these cycles and satisfies
    $\mathbb{E}_\lambda[U_k]=0$. Consequently,
    \[
        \mathbb{E}_\lambda\!\left[
        \left(\sum_{j=2}^{T_n}U_j\right)^2
        \,\bigg|\,D_0=d_0\right]
        =
        \sum_{j=2}^n
        \mathbb{E}_\lambda\!\left[
        U_j^2\mathbf{1}_{\{T_n\geq j\}}
        \,\middle|\,D_0=d_0\right]
        \leq(n-1)\mathbb{E}_\lambda[N^2].
    \]


    Let
    \begin{align*}
    m_w:=
    \sup_{\lambda\in\mathcal{L}}\sup_{d_0\leq w}
    \mathbb{E}_\lambda\left[N_1^2\, \Big|\, D_0=d_0\right].
    \end{align*}
    By a coupling argument, the length of an ordinary regeneration cycle is
    stochastically dominated by that of the first cycle when $W_0=w$. Hence, as $W_0\leq D_0\leq w$,
    \[
        \sup_{\lambda\in\mathcal L}\mathbb{E}_\lambda[N^2]\leq m_w.
    \]
    Furthermore, since $|R_n|\leq N_{T_n}$ and $T_n\leq n$,
    \begin{align*}
    \mathbb{E}_\lambda\!\left[R_n^2\,\middle|\,D_0=d_0\right]
    &\leq
    \mathbb{E}_\lambda\!\left[N_{T_n}^2\,\middle|\,D_0=d_0\right]\leq
    \mathbb{E}_\lambda\!\left[N_1^2\,\middle|\,D_0=d_0\right]
    +(n-1)\mathbb{E}_\lambda[N^2].
    \end{align*}
    Combining the above bounds with
    $(a-b)^2\leq2a^2+2b^2$ yields, uniformly over $\lambda\in\mathcal L$ and
    $d_0\leq w$,
    \begin{align*}
    \mathbb{E}_\lambda\!\left[
    \left|
    \frac{1}{n}\sum_{i=1}^n
    \left(X_i-\mathbb{E}_\lambda[X]\right)
    \right|^2
    \,\middle|\,D_0=d_0\right]
    &\leq \frac{6m_w}{n}.
    \end{align*}
    Since $m_w<\infty$, \eqref{eq:D_mixing} follows with $K_1=6m_w$.

    \smallskip
    \noindent (ii)~Define
    \[
        Z_n(\ell):=\frac{1}{n}\sum_{i=1}^n f_\ell(D_i,\lambda)-\mathbb{E}_\lambda[f_\ell(D,\lambda)].
    \]
    Due to $Z_n(\ell)=Z_n(\ell_1)+\int_{\ell_1}^{\ell}Z_n'(u)\,\mathrm{d} u$, Cauchy--Schwarz gives
    \[
        \sup_{\ell\in[\ell_1,\ell_2]}|Z_n(\ell)|^2
        \leq
        2|Z_n(\ell_1)|^2+
        2(\ell_2-\ell_1)\int_{\ell_1}^{\ell_2}|Z_n'(u)|^2\,\mathrm{d} u.
    \]
    Since, by assumption, $\left|\frac{\mathrm{d}}{\mathrm{d}\ell}f_\ell(d,\lambda)\right|\leq\bar m$, dominated
    convergence gives
    \[
        Z_n'(u)
        =
        \frac{1}{n}\sum_{i=1}^n\frac{\mathrm{d}}{\mathrm{d} u}f_u(D_i,\lambda)
        -\mathbb{E}_\lambda\!\left[\frac{\mathrm{d}}{\mathrm{d} u}f_u(D,\lambda)\right].
    \]
    Applying part~(i) to
    $g_u(d):=(\frac{\mathrm{d}}{\mathrm{d} u}f_u(d,\lambda)+\bar m)/(2\bar m)\in[0,1]$
    therefore yields, uniformly in $u\in[\ell_1,\ell_2]$ and
    $\lambda\in\mathcal L$,
    \[
        \mathbb{E}_\lambda\!\left[|Z_n'(u)|^2\,\middle|\,D_0=d_0\right]
        \leq\frac{4\bar m^2K_1}{n}.
    \]
    Together with part~(i) applied to $f_{\ell_1}$, this gives
    \[
        \mathbb{E}_\lambda\!\left[
        \sup_{\ell\in[\ell_1,\ell_2]}|Z_n(\ell)|^2
        \,\middle|\,D_0=d_0\right]
        \leq
        \frac{2K_1}{n}
        +\frac{8(\ell_2-\ell_1)^2\bar m^2K_1}{n},
    \]
    which completes the proof.

\noindent \noindent It is observed that the estimator $\hat{\lambda}_{\mathrm{bg}}$ is a
     $Z$-estimator truncated at zero. Define $A_0:=\{\Psi_n(0)<0\}$. We use the decomposition
     \begin{align*}
     \mathbb{E}|\hat{\lambda}_{\mathrm{bg}}-\lambda_{\mathrm{bg}}|^2
     &=\mathbb{E}\!\left[|\hat{\lambda}_{\mathrm{bg}}-\lambda_{\mathrm{bg}}|^2\mathbf{1}_{A_0}\right]
     +\mathbb{E}\!\left[|\hat{\lambda}_{\mathrm{bg}}-\lambda_{\mathrm{bg}}|^2\mathbf{1}_{A_0^c}\right].
     \end{align*}
     On $A_0$, $\hat{\lambda}_{\mathrm{bg}}=0$. Thus the first term vanishes when
     $\lambda_{\mathrm{bg}}=0$, while for $\lambda_{\mathrm{bg}}>0$ it equals
     $\lambda_{\mathrm{bg}}^2\mathbb{P}(A_0)$, where
     \[
         \mathbb{P}(A_0)
         =
         \mathbb{P}\!\left(
         \frac{1}{n}\sum_{i=1}^n e^{-\lambda D_i}
         >1-\lambda\mathbb{E}[H]\right).
     \]
     Define $w(x):=1-(\lambda+x)\mathbb{E}[H]-\mathbb{E}[e^{-(\lambda+x)D}]$ on
     $[-\lambda,\lambda_{\mathrm{bg}}]$. Then $w(-\lambda)=w(\lambda_{\mathrm{bg}})=0$,
     $w''(x)=-\mathbb{E}[D^2e^{-(\lambda+x)D}]<0$, and
     $w(0)=1-\lambda\mathbb{E}[H]-\mathbb{E}[e^{-\lambda D}]>0$. Moreover, with
     $c_0:=\tfrac{1}{2}\,{\lambda_{\min}}\inf_{\lambda\in\mathcal L}
     \mathbb{E}[D^2e^{-(\lambda+\lambda_{\mathrm{bg}})D}]$, the concavity argument gives
     $w(0)\geq c_0\lambda_{\mathrm{bg}}$. Consequently,
     \[
         \mathbb{P}(A_0)
         \leq
         \mathbb{P}\!\left(
         \Bigg|
         \frac{1}{n}\sum_{i=1}^n e^{-\lambda D_i}
         -\mathbb{E}[e^{-\lambda D}]
         \Bigg|>c_0\lambda_{\mathrm{bg}}\right).
     \]
     Since $\mathbb{E}[H^2]<\infty$, part~(i) of Lemma~\ref{lem:D_mixing} and Chebyshev's inequality give
     \begin{equation}\label{eq:truncation_error_bound}
     \mathbb{E}\!\left[|\hat{\lambda}_{\mathrm{bg}}-\lambda_{\mathrm{bg}}|^2\mathbf{1}_{A_0}\right]
     \leq \frac{K_1}{c_0^2n}.
     \end{equation}

     On $A_0^c$, $\hat{\lambda}_{\mathrm{bg}}$ is an interior solution that satisfies
     $\Psi_n(\hat{\lambda}_{\mathrm{bg}})=0$. For $\ell\leq\lambda_{\mathrm{bg}}$, the
     concavity argument gives $|\lambda_{\mathrm{bg}}-\ell|
     \leq |w(\ell;\lambda,\lambda_{\mathrm{bg}})|/c_0$, while for
     $\ell\geq\lambda_{\mathrm{bg}}$, monotonicity of $w'$ gives the same bound with
     $c_1:=\inf_{\lambda\in\mathcal L}
     \{\mathbb{E}[H]-\mathbb{E}[De^{-(\lambda+\lambda_{\mathrm{bg}})D}]\}$. Thus, with
     $c:=\min\{c_0,c_1\}$,
     $|\lambda_{\mathrm{bg}}-\ell|\leq c^{-1}|w(\ell;\lambda,\lambda_{\mathrm{bg}})|$
     for $\ell\in\Theta_\epsilon$. Taking $\ell=\hat{\lambda}_{\mathrm{bg}}$ and
     using $\Psi_n(\hat{\lambda}_{\mathrm{bg}})=0$ yields
     \[
         |\hat{\lambda}_{\mathrm{bg}}-\lambda_{\mathrm{bg}}|
         \leq
         \frac{1}{c}
         \left|
         \frac{1}{n}\sum_{i=1}^n e^{-(\lambda+\hat{\lambda}_{\mathrm{bg}})D_i}
         -\mathbb{E}[e^{-(\lambda+\hat{\lambda}_{\mathrm{bg}})D}]
         \right|.
     \]
     Let $f_\ell(d,\lambda)=e^{-\ell d}\in[0,1]$. By part~(ii)  of Lemma~\ref{lem:D_mixing},
     \begin{align*}
     \mathbb{E}\!\left[
     |\hat{\lambda}_{\mathrm{bg}}-\lambda_{\mathrm{bg}}|^2\mathbf{1}_{A_0^c}
     \right]
     &\leq
     \frac{1}{c^2}
     \mathbb{E}\!\left[
     \sup_{\ell\in\Theta_\epsilon}
     \left|
     \frac{1}{n}\sum_{i=1}^n e^{-(\lambda+\ell)D_i}
     -\mathbb{E}[e^{-(\lambda+\ell)D}]
     \right|^2
     \right]\leq \frac{K_2}{c^2n}.
     \end{align*}
     Combining \eqref{eq:truncation_error_bound} and the preceding bound gives
     \[
         \mathbb{E}|\hat{\lambda}_{\mathrm{bg}}-\lambda_{\mathrm{bg}}|^2
         \leq
         \left(
         \frac{K_1}{c_0^2}
         +\frac{K_2}{\min\{c_0,c_1\}^2}
         \right)\frac{1}{n},
     \]
     which implies the stated claim.
