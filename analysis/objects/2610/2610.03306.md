<!-- theorem--1: body_md -->
Consider full-batch Muon with momentum $\mu\in[0,1)$ and five tuned Newton--Schulz iterations, training the first layer of a width-$m$ two-layer ReLU network with fixed random output weights. Let the $n$ training inputs have unit norm, and suppose their limiting neural tangent kernel has smallest eigenvalue $\lambda_0>0$. Treat the output-weight scale and the average squared label magnitude as constants. For any target loss $\varepsilon>0$ and failure probability $\delta\in(0,1)$, take
\[
m=\Omega\!\left(\frac{n^5}{\lambda_0^4\delta^3}\right),
\qquad
\eta=\Theta\!\left(\frac{(1-\mu)\sqrt{\lambda_0\varepsilon}}{n}\right).
\]
Then, with probability at least $1-\delta$ over initialization, an iterate with empirical squared loss at most $\varepsilon$ is reached within
\[
T=\mathcal O\!\left(
\frac{n^{3/2}}{(1-\mu)\lambda_0\sqrt{\delta\varepsilon}}
\right)
\]
steps. The precise assumptions and constants are given in Theorem~\ref{thm:main_muon}.

<!-- theorem-1: body_md -->
\label{thm:main_muon}
Consider the model and initialization of
Section~\ref{sec:model}, and suppose
Assumption~\ref{ass:ns} holds with constants
$\kappa$ and $M$. Fix $\varepsilon>0$,
$\delta\in(0,1)$, and $\mu\in[0,1)$. If
\begin{equation}
    m\ge
    5\cdot10^{12}
    \left(\frac{M}{\kappa}\right)^8
    \frac{n^4}{\lambda_0^4\delta^3}
    \left(n+\frac{\|y\|_2^2}{\alpha^2}\right),
    \label{eq:muon_width}
\end{equation}
run (\ref{eq:muon_trajectory}) with
\begin{align}
    \eta
    &=(1-\mu)
    \frac{\kappa\sqrt{\lambda_0\varepsilon}}
         {8M^2n\alpha},
    \label{eq:muon_momentum_condition}\\
    T
    &=\left\lceil
    \frac{128M^2nR_0}
         {(1-\mu)\kappa^2\lambda_0\sqrt{\varepsilon}}
    \right\rceil.
    \label{eq:muon_budget}
\end{align}
Then, with probability at least $1-\delta$ over
initialization,
\begin{equation}
    \min_{0\le t\le T}L(W_t)\le\varepsilon.
    \label{eq:muon_hitting_conclusion}
\end{equation}
Thus, stopping at the first iterate with loss at
most $\varepsilon$ requires no more than $T$ updates.

\noindent\textbf{Part 1.}
For $Z\sim\mathcal N(0,1)$, $\mathbb E\sigma(Z)^2=\mathbb E[Z^2\mathds1\{Z\ge0\}]=1/2$.
Since the $a_r$ are independent of $W_0$ with $\mathbb E a_r=0$ and
$a_r^2=\alpha^2$, and $w_r(0)^\top x_i\sim\mathcal N(0,1)$,
\begin{equation}
 \mathbb E u(W_0)_i=0,\qquad
 \mathbb E u(W_0)_i^2=\frac1m\sum_{r=1}^m\alpha^2\,\mathbb E\sigma(w_r(0)^\top x_i)^2
 =\frac{\alpha^2}{2}.
 \label{eq:appendix_initial_moments}
\end{equation}
Then
\begin{align}
 \mathbb{P}(\mathcal E_1^c)
 &=\mathbb{P}(\|e_0\|_2^2>R_0^2)\nonumber\\
 &\le\frac{\mathbb E\|e_0\|_2^2}{R_0^2}
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[Markov's inequality]}\nonumber\\
 &=\frac{\|y\|_2^2+n\alpha^2/2}{R_0^2}
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[cross term vanishes by (\ref{eq:appendix_initial_moments})]}\nonumber\\
 &=\frac\delta3
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[definition (\ref{eq:muon_constants}) of $R_0$]}.
 \label{eq:appendix_E1_probability}
\end{align}

\noindent\textbf{Part 2.}
For fixed $i,j$,
\begin{equation}
 \frac{H(W_0)_{ij}}{\alpha^2}
 =\frac1m\sum_{r=1}^m Z_r^{ij},\qquad
 Z_r^{ij}:=x_i^\top x_j\,
 \mathds1\{w_r(0)^\top x_i\ge0,\ w_r(0)^\top x_j\ge0\}.
 \label{eq:appendix_kernel_summands}
\end{equation}
The $Z_r^{ij}$ are independent over $r$, have mean $H^\infty_{ij}$ by
(\ref{eq:muon_hinf}), and take values in an interval of length
$|x_i^\top x_j|\le1$. Since $\log z\le z$, $\kappa\le M$, $\lambda_0\le\frac12$
and $\delta<1$, the width condition (\ref{eq:muon_width}) gives
$m\ge48n^4/(\lambda_0^2\delta)\ge(8n^2/\lambda_0^2)\log(6n^2/\delta)$. Hence
\begin{align}
 \mathbb{P}\Bigl(\max_{i,j}\Bigl|\frac{H(W_0)_{ij}}{\alpha^2}-H^\infty_{ij}\Bigr|>\frac{\lambda_0}{4n}\Bigr)
 &\le n^2\cdot2\exp\Bigl(-\frac{m\lambda_0^2}{8n^2}\Bigr)
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[union bound, Hoeffding's inequality]}\nonumber\\
 &\le\frac\delta3
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[$m\ge(8n^2/\lambda_0^2)\log(6n^2/\delta)$]}.
 \label{eq:appendix_hoeffding}
\end{align}
Outside this event, $D:=H(W_0)/\alpha^2-H^\infty$ satisfies
\begin{align}
 \|D\|_2
 &\le\|D\|_F
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[operator norm $\le$ Frobenius norm]}\nonumber\\
 &\le n\max_{i,j}|D_{ij}|
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[$D$ has $n^2$ entries]}\nonumber\\
 &\le\frac{\lambda_0}{4},
 \label{eq:appendix_kernel_operator}
\end{align}
which is $\mathcal E_2$ after multiplying by $\alpha^2$. Hence
$\mathbb{P}(\mathcal E_2^c)\le\delta/3$.

\noindent\textbf{Part 3.}
The density of $\mathcal N(0,1)$ is at most $1/\sqrt{2\pi}$, so
$\mathbb{P}\bigl(|w_r(0)^\top x_i|\le\frac{\delta\kappa^2\lambda_0}{1620M^2n}\bigr)
\le\sqrt{2/\pi}\,\frac{\delta\kappa^2\lambda_0}{1620M^2n}$. Let $N$ be the double
sum in $\mathcal E_3$. Then
\begin{align}
 \mathbb{P}(\mathcal E_3^c)
 &=\mathbb{P}\Bigl(\frac Nm>\frac{\kappa^2\lambda_0}{648M^2}\Bigr)\nonumber\\
 &\le\frac{648M^2}{\kappa^2\lambda_0}\,\mathbb E\frac Nm
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[Markov's inequality]}\nonumber\\
 &\le\frac{648M^2}{\kappa^2\lambda_0}\cdot n\sqrt{\frac2\pi}\,
   \frac{\delta\kappa^2\lambda_0}{1620M^2n}
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[linearity of expectation]}\nonumber\\
 &\le\frac\delta3
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[$\sqrt{2/\pi}\le\tfrac56$]}.
 \label{eq:appendix_E3_probability}
\end{align}
The union bound over Parts~1--3 completes the proof.

\label{lem:muon_ball}
On $\mathcal E$, for every $W,V\in\mathbb B$ and every matrix $O$,

\end{align}

Fix an initialization in $\mathcal E$ and arbitrary
$W=(w_1,\ldots,w_m)$ and $V=(v_1,\ldots,v_m)$ in $\mathbb B$. Part 1
counts the gates that can flip; Part 2 turns that count into the Jacobian
bound (\ref{eq:muon_jacobian_close}) and the kernel bound
(\ref{eq:muon_kernel_lower}); Parts 3 and 4 turn it into the remainder bound
(\ref{eq:muon_remainder}) and the bound (\ref{eq:muon_jacobian_action}).

\noindent\textbf{Part 1.}
For $Z\in\{W,V\}$ and each $i$, define
\begin{equation}
 \mathcal S_i:=\Bigl\{r:|w_r(0)^\top x_i|\le\frac{\delta\kappa^2\lambda_0}{1620M^2n}\Bigr\},
 \qquad
 \mathcal I(Z):=\Bigl\{r:\|z_r-w_r(0)\|_2>\frac{\delta\kappa^2\lambda_0}{1620M^2n}\Bigr\}.
 \label{eq:appendix_margin_sets}
\end{equation}
If $r\notin\mathcal S_i\cup\mathcal I(Z)$, then, using $\|x_i\|_2=1$,
\begin{equation}
 |(z_r-w_r(0))^\top x_i|
 \le\|z_r-w_r(0)\|_2
 \le\frac{\delta\kappa^2\lambda_0}{1620M^2n}<|w_r(0)^\top x_i|,
 \label{eq:appendix_no_gate_change}
\end{equation}
so $z_r^\top x_i$ and $w_r(0)^\top x_i$ have the same nonzero sign and the
gate of unit $r$ on $x_i$ does not flip between $W_0$ and $Z$. Hence the set
\begin{equation}
 \mathcal A_i(W,V):=\{r:\text{the gate of $r$ on $x_i$ flips between
 $W_0$ and $W$, or between $W_0$ and $V$}\}
 \label{eq:appendix_bad_gate_set}
\end{equation}
satisfies
$\mathcal A_i(W,V)\subseteq\mathcal S_i\cup\mathcal I(W)\cup\mathcal I(V)$.
Each $r\in\mathcal I(Z)$ contributes more than
$(\delta\kappa^2\lambda_0/(1620M^2n))^2$ to $\|Z-W_0\|_F^2$, and $Z\in\mathbb B$,
so
\begin{equation}
 |\mathcal I(Z)|
 \le\Bigl(\frac{18M\sqrt n\,R_0}{\kappa\alpha\sqrt{\lambda_0}}\Bigr)^2
 \Bigl(\frac{1620M^2n}{\delta\kappa^2\lambda_0}\Bigr)^2
 \le\frac{8.51\cdot10^8\,M^6n^3R_0^2}{\kappa^6\alpha^2\lambda_0^3\delta^2}.
 \label{eq:appendix_large_move_count}
\end{equation}
Therefore
\begin{align}
 \frac1m\sum_{i=1}^n|\mathcal A_i(W,V)|
 &\le\frac1m\sum_{i=1}^n|\mathcal S_i|
   +\frac nm\bigl(|\mathcal I(W)|+|\mathcal I(V)|\bigr)
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[inclusion above]}\nonumber\\
 &\le\frac{\kappa^2\lambda_0}{648M^2}
   +\frac{1.71\cdot10^9\,M^6n^4R_0^2}{m\,\kappa^6\alpha^2\lambda_0^3\delta^2}
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[on $\mathcal E_3$; by (\ref{eq:appendix_large_move_count})]}\nonumber\\
 &\le\frac{\kappa^2\lambda_0}{324M^2}
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[width (\ref{eq:muon_width}), $R_0^2\le\tfrac{3\alpha^2}{\delta}\bigl(n+\tfrac{\|y\|_2^2}{\alpha^2}\bigr)$]}.
 \label{eq:appendix_bad_gate_average}
\end{align}
The last inequality uses (\ref{eq:muon_width}) together with
$648\cdot3\cdot1.71\cdot10^9<3.4\cdot10^{12}\le5\cdot10^{12}$.

\noindent\textbf{Part 2.}
Take $V=W_0$. A block of $J(W)-J(W_0)$ is nonzero only when the gate flips,
and then has squared norm $\alpha^2/m$. Hence
\begin{equation}
 \|J(W)-J(W_0)\|_F^2
 \le\frac{\alpha^2}{m}\sum_{i=1}^n|\mathcal A_i(W,W_0)|
 \le\frac{\kappa^2\alpha^2\lambda_0}{324M^2}
 \label{eq:appendix_jacobian_difference}
\end{equation}
by (\ref{eq:appendix_bad_gate_average}), which is (\ref{eq:muon_jacobian_close}).
For every $q\in\mathbb R^n$,
\begin{align}

 &\ge\|J(W_0)^\top q\|_2-\|J(W)-J(W_0)\|_2\|q\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[reverse triangle inequality]}\nonumber\\
  &\ge\sqrt{q^\top H(W_0)q}-\frac{\kappa\alpha\sqrt{\lambda_0}}{18M}\|q\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[by (\ref{eq:appendix_jacobian_difference})]}\nonumber\\
 &\ge\Bigl(\frac{\sqrt3}{2}-\frac1{18}\Bigr)\alpha\sqrt{\lambda_0}\,\|q\|_2

 \label{eq:appendix_ball_singular_value}
\end{align}
Here $\mathcal E_2$ gives $\lambda_{\min}(H(W_0))\ge\tfrac34\alpha^2\lambda_0$.
Since $(\sqrt3/2-1/18)^2>1/2$, squaring

\noindent\textbf{Part 3.}
Set
\begin{equation}
 \xi:=u(V)-u(W)-J(W)\operatorname{vec}(V-W),
 \qquad
 \xi_i=\frac1{\sqrt m}\sum_{r=1}^m a_r\,\rho\bigl(w_r^\top x_i,(v_r-w_r)^\top x_i\bigr),
 \label{eq:appendix_remainder_definition}
\end{equation}
where $\rho(z,h):=\sigma(z+h)-\sigma(z)-\mathds1\{z\ge0\}h$. The scalar
remainder satisfies $|\rho(z,h)|\le|h|$, and $\rho(z,h)=0$ unless the gates
at $z$ and $z+h$ differ. If neither $W$ nor $V$ flips a gate relative to
$W_0$, the gates at $W$ and $V$ agree, so only $r\in\mathcal A_i(W,V)$

\begin{align}

 &\le\alpha\sqrt{\frac{|\mathcal A_i(W,V)|}{m}}\,\|(V-W)^\top x_i\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[Cauchy--Schwarz over $r$]}\nonumber\\
 &\le\alpha\sqrt{\frac{|\mathcal A_i(W,V)|}{m}}\,\|V-W\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[$\|x_i\|_2=1$]}.
 \label{eq:appendix_coordinate_remainder}
\end{align}
Squaring and summing over $i$,
\begin{align}
 \|\xi\|_2^2
 &\le\alpha^2\|V-W\|_2^2\cdot\frac1m\sum_{i=1}^n|\mathcal A_i(W,V)|
   \nonumber\\
 &\le\frac{\kappa^2\alpha^2\lambda_0}{324M^2}\,\|V-W\|_2^2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[by (\ref{eq:appendix_bad_gate_average})]},
 \label{eq:appendix_vector_remainder}
\end{align}
which is (\ref{eq:muon_remainder}).

\noindent\textbf{Part 4.}
Write $o_r$ for the $r$-th column of $O$. The $i$-th entry of
$(J(W)-J(V))\operatorname{vec}(O)$ is

\begin{align}

 &\le\sum_{i=1}^n\alpha^2\frac{|\mathcal A_i(W,V)|}{m}\,\|O^\top x_i\|_2^2

 \label{eq:appendix_jacobian_action}
\end{align}
which is (\ref{eq:muon_jacobian_action}).

 &\le\frac{\alpha\eta}{\sqrt m}\sum_{r=1}^m|(O_t)_{\cdot,r}^\top x_i|

\noindent\textbf{Part 2.}

\begin{equation}

 =\frac\kappa2\bigl(e_{t-1}^\top H(W_{t-1})e_{t-1}\bigr)^{1/2}

\end{equation}
Let $\xi:=u(W_t)-u(W_{t-1})+\eta J(W_{t-1})\operatorname{vec}(O_t)$ be the
linearization remainder of the step. Both iterates lie in $\mathbb B$, so
(\ref{eq:muon_remainder}) and Lemma~\ref{lem:ns5} give

=\frac{\kappa\alpha\eta\sqrt{\lambda_0}}{18}$. Moreover,
\begin{equation}

 \label{eq:appendix_linear_gain_identity}

Since $e_t=e_{t-1}-(u(W_t)-u(W_{t-1}))$,
\begin{align}

\noindent\textbf{Part 3.}
Assume (\ref{eq:muon_local_stepsize}); since $\eta>0$ it forces
$\|e_{t-1}\|_2>0$. Multiplying (\ref{eq:muon_local_stepsize}) by
$nM^2\alpha^2\eta$ gives
$nM^2\alpha^2\eta^2\le\frac{\kappa\alpha\eta\sqrt{\lambda_0}}{8}\|e_{t-1}\|_2$, so
\begin{align}
 \|e_t\|_2^2
 &\le\|e_{t-1}\|_2^2-\frac{\kappa\alpha\eta\sqrt{\lambda_0}}{8}\|e_{t-1}\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[by (\ref{eq:appendix_squared_descent})]}\nonumber\\
 &=\|e_{t-1}\|_2^2\Bigl(1-\frac{\kappa\alpha\eta\sqrt{\lambda_0}}{8\|e_{t-1}\|_2}\Bigr).
 \label{eq:appendix_before_square_root}
\end{align}
The left side is nonnegative, so the bracket lies in $[0,1]$, and
\begin{align}
 \|e_t\|_2

which is (\ref{eq:muon_linear_descent}).

Write $J_k:=J(W_k)$. Unrolling $B_t=\mu B_{t-1}+G_t$ down to $B_0=0$ splits the
buffer into the current gradient and the change of the older ones,
\begin{equation}
 B_t=S\,G_t+D,\qquad
 S:=\sum_{j=0}^{t-1}\mu^j\ge1,\qquad
 D:=\sum_{j=1}^{t-1}\mu^j\bigl(G_{t-j}-G_t\bigr).
 \label{eq:appendix_buffer_split}
\end{equation}
Part 1 reduces the claim to a bound on $D$, Part 2 bounds one term of $D$, and
Part 3 adds the terms up.

\noindent\textbf{Part 1.}
The column space of $B_t$ lies in $\operatorname{span}\{x_1,\ldots,x_n\}$, so
\begin{align}
 S\langle G_t,O_t\rangle
 &=\langle B_t,O_t\rangle-\langle D,O_t\rangle
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[by (\ref{eq:appendix_buffer_split})]}\nonumber\\
 &\ge\kappa\|B_t\|_F-|\langle D,O_t\rangle|
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[Lemma~\ref{lem:ns5}]}\nonumber\\
 &\ge\kappa S\|G_t\|_F-\kappa\|D\|_F-|\langle D,O_t\rangle|
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[triangle inequality]}.
 \label{eq:appendix_momentum_reduction}
\end{align}
It therefore suffices to show
\begin{equation}
 \kappa\|D\|_F+|\langle D,O_t\rangle|\le\frac\kappa2\,S\|G_t\|_F.
 \label{eq:appendix_momentum_target}
\end{equation}

\noindent\textbf{Part 2.}
Fix $1\le j\le t-1$ and put $s=t-j$. By the gradient identity,
$\operatorname{vec}(G_s-G_t)
=-J_{s-1}^\top(e_{s-1}-e_{t-1})-(J_{s-1}-J_{t-1})^\top e_{t-1}$. Two facts
control the first term. By Part~1 of the proof of Lemma~\ref{lem:muon_step},
the residual moves by at most $\sqrt n\,M\alpha\eta$ per step, so
$\|e_{s-1}-e_{t-1}\|_2\le j\sqrt n\,M\alpha\eta$. And for every $W$ and every
matrix $O$, each entry of $J(W)\operatorname{vec}(O)$ is at most
$\alpha\|O^\top x_i\|_2\le\alpha\|O\|_2$ in magnitude, by Cauchy--Schwarz over
the hidden units, so $\|J(W)\operatorname{vec}(O)\|_2\le\alpha\sqrt n\,\|O\|_2$
and $\|J(W)\|_2\le\alpha\sqrt n$. Since $W_{s-1},W_{t-1}\in\mathbb B$,
\begin{align}
 \|G_s-G_t\|_F
 &\le\|J_{s-1}\|_2\|e_{s-1}-e_{t-1}\|_2+\|J_{s-1}-J_{t-1}\|_2\|e_{t-1}\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[triangle inequality]}\nonumber\\
 &\le jnM\alpha^2\eta+\frac{2\kappa\alpha\sqrt{\lambda_0}}{18M}\|e_{t-1}\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[facts above, (\ref{eq:muon_jacobian_close}) twice]},
 \label{eq:appendix_gradient_change_norm}\\
 |\langle G_s-G_t,O_t\rangle|
 &\le\|e_{s-1}-e_{t-1}\|_2\|J_{s-1}\operatorname{vec}(O_t)\|_2\nonumber\\
 &\quad+\|e_{t-1}\|_2\|(J_{s-1}-J_{t-1})\operatorname{vec}(O_t)\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[Cauchy--Schwarz]}\nonumber\\
 &\le jnM^2\alpha^2\eta+\frac{\kappa\alpha\sqrt{\lambda_0}}{18}\|e_{t-1}\|_2
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[facts above, (\ref{eq:muon_jacobian_action}), $\|O_t\|_2\le M$]}.
 \label{eq:appendix_gradient_change_inner}
\end{align}
Since $\kappa\le M$, the two bounds combine to
\begin{equation}
 \kappa\|G_s-G_t\|_F+|\langle G_s-G_t,O_t\rangle|
 \le2jnM^2\alpha^2\eta+\frac{3\kappa\alpha\sqrt{\lambda_0}}{18}\|e_{t-1}\|_2.
 \label{eq:appendix_gradient_change}
\end{equation}

\noindent\textbf{Part 3.}
Exchanging the order of summation and using $\sum_{k=0}^{t-1-i}\mu^k\le S$,
\begin{equation}
 \sum_{j=1}^{t-1}j\mu^j
 =\sum_{i=1}^{t-1}\sum_{j=i}^{t-1}\mu^j
 =\sum_{i=1}^{t-1}\mu^i\sum_{k=0}^{t-1-i}\mu^k
 \le S\sum_{i=1}^{t-1}\mu^i
 \le\frac{S\mu}{1-\mu},
 \label{eq:appendix_weighted_j}
\end{equation}
and also $\sum_{j=1}^{t-1}\mu^j\le S$. Summing (\ref{eq:appendix_gradient_change}) with
weights $\mu^j$,
\begin{align}
 \kappa\|D\|_F+|\langle D,O_t\rangle|
 &\le S\Bigl(\frac{2\mu}{1-\mu}\,nM^2\alpha^2\eta
   +\frac{3\kappa\alpha\sqrt{\lambda_0}}{18}\|e_{t-1}\|_2\Bigr)
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[triangle inequality, weight sums]}\nonumber\\
 &\le S\Bigl(\frac\kappa4\,\alpha\sqrt{\lambda_0\varepsilon}
   +\frac{3\kappa\alpha\sqrt{\lambda_0}}{18}\|e_{t-1}\|_2\Bigr)
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[bound on $\eta$, $\mu\le1$]}\nonumber\\
 &\le\frac\kappa2\,S\|G_t\|_F
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[see below]}.
 \label{eq:appendix_momentum_sum}
\end{align}
The last line uses the kernel bound (\ref{eq:muon_kernel_lower}) at $W_{t-1}$
and $\|e_{t-1}\|_2>\sqrt{2\varepsilon}$, which give
$\|G_t\|_F\ge\alpha\sqrt{\lambda_0/2}\,\|e_{t-1}\|_2>\alpha\sqrt{\lambda_0\varepsilon}$,
together with $\frac3{18}\le\frac1{4\sqrt2}$; each of the two terms is then at
most $\frac\kappa4\|G_t\|_F$. This is (\ref{eq:appendix_momentum_target}), and
with (\ref{eq:appendix_momentum_reduction}) it gives
$\langle G_t,O_t\rangle\ge\kappa\|G_t\|_F-\frac\kappa2\|G_t\|_F$. For $t=1$
the sums are empty, $D=0$, and the claim is Lemma~\ref{lem:ns5} directly.

\label{prop:jordan_constants}
For $p(\lambda)=a+b\lambda+c\lambda^2$ with $(a,b,c)=(3.4445,-4.7750,2.0315)$,
$\phi(x)=x\,p(x^2)$ and $\Phi=\phi^{\circ5}$,
\begin{equation}
 \Phi(x)\le\frac54\quad\text{and}\quad\Phi(x)\ge\frac23x\quad\text{for }x\in[0,1],
 \qquad

 \label{eq:appendix_jordan_claims}
\end{equation}

The proof uses four facts about the single quintic $\phi$, each an evaluation
of a polynomial. Let
$x_a\approx0.55453$ and $x_b\approx1.05014$ be the positive roots of
$\phi'(x)=a+3bx^2+5cx^4$.
\begin{enumerate}
 \item[(F1)] $p(z)\ge a-b^2/(4c)=0.6386\ldots>0$ for all $z$, and $p$ decreases
 on $[0,-b/(2c)]=[0,1.1752\ldots]$.
 \item[(F2)] $\phi$ increases on $[0,x_a]$, decreases on $[x_a,x_b]$ and
 increases on $[x_b,\frac54]$, with $\phi(x_a)=1.2023\ldots$,
 $\phi(x_b)=0.6818\ldots$ and $\phi(\frac54)=1.1790\ldots$.
 \item[(F3)] $p(x_a^2)=2.1682\ldots$.
 \item[(F4)] The first five iterates of $\frac1{400}$ under $\phi$ are
 $0.0086\ldots$, $0.0296\ldots$, $0.1020\ldots$, $0.3464\ldots$ and
 $1.0048\ldots$.
\end{enumerate}
By (F1) and (F2), $\phi$ maps $[0,\frac54]$ into $[0,1.2024]$ and maps
$[x_a,\frac54]$ into $[0.6818,1.2024]$. Fix $x\in(0,1]$ and let
$x_k=\phi^{\circ k}(x)$, so every $x_k$ lies in $[0,1.2024]$ and in particular
$\Phi(x)=x_5\le\frac54$.

\noindent\textbf{Part 1.}
Suppose $x_k\ge x_a$ for some $k\le4$. Then $x_k\in[x_a,\frac54]$, every later
iterate stays in $[0.6818,1.2024]\subseteq[x_a,\frac54]$, and so
$\Phi(x)\ge0.6818$. In particular $\Phi(x)\ge0.68$ and
$\Phi(x)\ge\frac23x$, since $x\le1$.

\noindent\textbf{Part 2.}
Suppose $x_k<x_a$ for every $k\le4$. Since $p$ decreases on $[0,x_a^2]$,
\begin{align}
 \frac{\Phi(x)}x
 &=\prod_{k=0}^4p(x_k^2)
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[$\phi(x)=x\,p(x^2)$]}\nonumber\\
 &\ge p(x_a^2)^5
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[(F1), $x_k^2<x_a^2$]}\nonumber\\
 &\ge1
   &&\text{\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\@setfontsize\footnotesize\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\@ixpt\@xpt\normalfont[(F3)]},
 \label{eq:appendix_small_orbit}
\end{align}
so $\Phi(x)\ge\frac23x$. If moreover $x\ge\frac1{400}$, then, since $\phi$
increases on $[0,x_a]$, induction on $k$ gives
$x_k\ge\phi^{\circ k}(\frac1{400})$ for $k\le4$, and so
$\Phi(x)=\phi(x_4)\ge\phi\bigl(\phi^{\circ4}(\frac1{400})\bigr)=1.0048\ldots\ge0.68$
by (F4).


Parts~1 and~2 cover all $x\in(0,1]$ and prove (\ref{eq:appendix_jordan_claims});
$x=0$ is trivial.

<!-- proposition-2: body_md -->
\label{prop:classical_ns}
Let $\Phi$ be the composition of $K$ classical polynomials
(\ref{eq:classical_ns}) of degrees $d_0,\ldots,d_{K-1}\ge1$. For $x\in[0,1]$,
\begin{equation}
 x\le\Phi(x)\le1
 \qquad\text{and}\qquad
 1-\Phi(x)^2\le(1-x^2)^{\prod_k(d_k+1)}.
 \label{eq:appendix_classical_claims}
\end{equation}
Hence Assumption~\ref{ass:ns} holds with $\kappa=M=1$ for every degree and
every number of steps. For five steps of the same degree, $\Phi(x)\ge0.68$ on
$[\frac17,1]$ when $d=1$ and on $[\frac1{19},1]$ when $d=2$.
<!-- end proposition-2 -->
<!-- proof-8: body_md -->
\noindent\textbf{Part 1.}
For $\lambda\in(0,1]$, the Taylor series
$\lambda^{-1/2}=\sum_{s\ge0}\frac{(2s)!}{4^s(s!)^2}(1-\lambda)^s$ converges and
has nonnegative terms, and $p^{(d)}$ is its truncation after the term $s=d$.
Hence $1\le p^{(d)}(\lambda)\le\lambda^{-1/2}$, and with $\lambda=x^2$,
\begin{equation}
 x\le x\,p^{(d)}(x^2)=\phi(x)\le1,\qquad x\in(0,1].
 \label{eq:appendix_classical_step}
\end{equation}
So each step maps $[0,1]$ into itself without decreasing its argument, and
composing $K$ steps gives $x\le\Phi(x)\le1$.

\noindent\textbf{Part 2.}
For a classical polynomial of degree $d$ and $u\in[0,1]$,
$1-(1-u)\,p^{(d)}(1-u)^2\le u^{d+1}$ \citep[Lemma~3]{kim2026muonns}. With
$u=1-x^2$ this reads $1-\phi(x)^2\le(1-x^2)^{d+1}$. By Part~1 every iterate
stays in $[0,1]$, so applying this bound at each of the $K$ steps gives the
second claim of (\ref{eq:appendix_classical_claims}).


The bound $\Phi(x)\ge\bigl(1-(1-x^2)^{\prod_k(d_k+1)}\bigr)^{1/2}$ increases in
$x$. For $K=5$ and $d=1$ the exponent is $32$ and
$(48/49)^{32}<0.52$, so $\Phi\ge0.69$ on $[\frac17,1]$. For $d=2$ the
exponent is $243$ and $(360/361)^{243}<0.51$, so $\Phi\ge0.70$ on
$[\frac1{19},1]$.

<!-- proposition-3: body_md -->
\label{prop:polar_optimal}
Let $B\ne0$ have compact singular value decomposition $B=U\Sigma V^\top$.
For every nonzero matrix $O$ of the same dimensions,
\begin{equation}
 \frac{\langle B,O\rangle}{\|O\|_2}\le\|B\|_*,
 \label{eq:polar_optimal}
\end{equation}
with equality for $O=UV^\top$.
<!-- end proposition-3 -->
<!-- proof-9: body_md -->
The trace inequality gives
$\langle B,O\rangle\le\sum_j s_j(B)s_j(O)
\le\|O\|_2\sum_j s_j(B)=\|O\|_2\|B\|_*$.
For $O=UV^\top$, $\|O\|_2=1$ and
$\langle B,O\rangle=\operatorname{tr}(\Sigma)=\|B\|_*$.
