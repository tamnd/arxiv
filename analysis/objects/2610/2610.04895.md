<!-- definition-1: body_md -->
\label{def:weak-solution}
		A field $u\in\mathcal{P}(\mathbb{B})$ is a weak solution of \eqref{eq:SHE}
		in $\mathbb{H}_r$, for $r\in\mathbb{R}$ and deterministic initial state $u_0\in \mathbb{H}_r$,
		if the following conditions hold on a stochastic basis $\mathbb{B}$
		carrying the noise $L$ of \eqref{eq:Levy-Ito}.
		\begin{enumerate}[label=(\roman*)]
			\item There exists an adapted process $(u(t))_{t\in[0,T]}$ with
			$(u(t))_{t\in[0,T]}\in\mathbb{D}([0,T];\mathbb{H}_r)$ and $u(0)=u_0$, $\mathbb{P}$-a.s., such that
			\[
			\mathbb{P}\!\left(
			\langle u(t),\varphi\rangle
			=\langle u(t,\cdot),\varphi\rangle
			\right)=1,
			\qquad t\in(0,T],\quad \varphi\in\mathcal{E}.
			\]
			
			\item For every $\varphi\in\mathcal{E}$, with probability one, the
			following identity holds for all $t\in[0,T]$:
			\begingroup
			\medmuskip=3mu plus 1mu minus 1mu
			\begin{equation}\label{eq:weak-form}
				\langle u(t),\varphi\rangle
				= \langle u_0,\varphi\rangle
				+\tfrac12\int_0^t\langle u(s),\varphi''\rangle\,\mathrm{d} s
				+\int_{\mathcal{Z}_t} b(u(s,y))\varphi(y)\,\mathrm{d} s\,\mathrm{d} y
				+\int_{\mathcal{Z}_t}\varphi(y)\sigma(u(s,y))L(\,\mathrm{d} s,\,\mathrm{d} y).
			\end{equation}
			\endgroup
			\item The field satisfies
			\begin{equation}\label{eq:weak-integrability}
				\int_{\mathcal{Z}_T}
				\bigl(|u(s,y)|+|b(u(s,y))|+|\sigma(u(s,y))|^\alpha\bigr)\,\mathrm{d} s\,\mathrm{d} y<\infty,
				\qquad \mathbb{P}\text{-a.s.}
			\end{equation}
		\end{enumerate}

\label{def:mild-solution}
		A predictable field $u\in\mathcal{P}(\mathbb{B})$ is a mild solution of
		\eqref{eq:SHE} if, for each fixed
		$(t,x)\in(0,T]\times D$, the integrals below are well defined and
		\begin{equation}\label{eq:mild}
			u(t,x)=P_tu_0(x)
			+\int_{\mathcal{Z}_t}\rho_{t-s}(x,y)b(u(s,y))\,\mathrm{d} s\,\mathrm{d} y
			+\int_{\mathcal{Z}_t}\rho_{t-s}(x,y)\sigma(u(s,y))L(\,\mathrm{d} s,\,\mathrm{d} y),
			\quad\mathbb{P}\text{-a.s.}
		\end{equation}
		We call a field $v \in \mathcal{P} (\mathbb{B})$ a density of an $\mathbb{H}_{-r}$-valued
		process $\widetilde u$ if, for each fixed $t\in(0,T]$,
		$\langle\widetilde u(t),\varphi\rangle
		=\int_D v(t,x)\varphi(x)\,\mathrm{d} x$ for all $\varphi\in\mathcal{E}$,
		$\mathbb{P}$-a.s. If $v$ also satisfies the mild
		formulation, we call it a mild density of $\widetilde u$.

The moment argument is standard; see Balan \cite{Balan14}*{Sections~5.2 and~6.1}
		and Chong--Dalang--Humeau \cite{CDH}*{Proposition~2.7} for related estimates.
		We give the calculation to track the dependence of $\mathcal{C}_0(q,K)$ and,
		in particular, its independence of $n$.
		
		Using the notation and martingale constant $B_q$ of Appendix~\ref{app:constants},
		fix $n$ and $K$, and put
		\[
		F_n(t):=1+\sup_{x\in D}\mathbb{E}|u_n^K(t,x)|^q,
		\qquad \beta_q:=\frac{q-1}{2}=1-\eta_q.
		\]
		For fixed $n$ and $K$, boundedness of the coefficients and continuity
		of the heat kernel give mean-square continuity of
		$x\mapsto u_n^K(t,x)$ for every $t>0$. Since $q\le2$, the function
		$x\mapsto\mathbb{E}|u_n^K(t,x)|^q$ is continuous. Its supremum over $D$
		therefore equals its supremum over $D\cap\mathbb{Q}$, which makes
		$F_n$ measurable on $(0,T]$. At $t=0$, it is a deterministic value.
		
		For fixed $n$, boundedness of $b_n$ and $\sigma_n$ gives
		$\sup_{t\le T}F_n(t)<\infty$.
		Using the sub-Markov property of $P_t$, the bound
		$\sup_x\int_D\rho_t(x,y)^q\,\mathrm{d} y\le h_qt^{-\beta_q}$,
		and the compensated-integral estimate of \cite{CDH}*{Lemma~A.2(2)},
		in the form \eqref{eq:compensated-maximal} with constant $B_q$,
		Jensen's inequality and
		\eqref{eq:approx-growth} yield
		\[
		\begin{aligned}
			F_n(t)\le a_q
			&+3^{q-1}2^{q-1}(2C_{\mathrm{gr}})^q
			\bigg[T^{q-1}\int_0^t F_n(s)\,\mathrm{d} s\\
			&\hspace{35mm}
			+B_qm_q(K)h_q\int_0^t(t-s)^{-\beta_q}F_n(s)\,\mathrm{d} s\bigg].
		\end{aligned}
		\]
		Since $1\le T^{\beta_q}(t-s)^{-\beta_q}$ for $0\le s<t\le T$, this implies
		\[
		F_n(t)\le a_q+d_{q,K}\int_0^t(t-s)^{\eta_q-1}F_n(s)\,\mathrm{d} s,
		\]
		with $a_q$ and $d_{q,K}$ as in \eqref{eq:C1-explicit}.
		Iterating the convolution inequality and using the beta integral yields
		\[
		F_n(t)\le a_q\sum_{j=0}^\infty
		\frac{\bigl[d_{q,K}\Gamma(\eta_q)t^{\eta_q}\bigr]^j}
		{\Gamma(1+j\eta_q)}
		=a_qE_{\eta_q}\bigl(d_{q,K}\Gamma(\eta_q)t^{\eta_q}\bigr).
		\]
		Taking the supremum over $t\le T$ and $n$ proves
		\eqref{eq:uniform-moment-bound}. Every constant in this last bound depends
		only on $q,T,C_{\mathrm{gr}},\|u_0\|_{\mathbb{L}^\infty(D)}$ and $m_q(K)$; the Lipschitz
		constants $\Lambda_n$ do not enter the estimate.

		\mathbb{E}\left|\int_0^t e^{-\lambda_k(t-s)}B_{n,k}^K(s)\,\mathrm{d} s\right|^q

		\le B_qm_q(K)D_q^K\int_0^t e^{-q\lambda_k(t-s)}\,\mathrm{d} s

		\le\frac{3^{q-1}}{\lambda_k}
		\left[\frac{\|u_0\|_{\mathbb{L}^\infty(D)}^q}{q}

		$\sum_{k\ge1}\lambda_k^{\eta q/2-1}
		=\lambda_1^{\eta q/2-1}\zeta(2-\eta q)$ proves (a).

		compensated-integral bound \eqref{eq:compensated-maximal},

		\mathbb{E}\sup_{t\le T}|X_{n,k}^K(t)|^q
		\le2^q\mathbb{E}\sup_{t\le T}|M_{n,k}^K(t)|^q
		\le2^qB_qm_q(K)TD_q^K.

		\mathbb{E}\sup_{t\le T}

		\le T^{q-1}\mathbb{E}\int_0^T|B_{n,k}^K(s)|^q\,\mathrm{d} s

			&\quad+2\sup_{\tau\le t\le\tau'}|M_{n,k}^K(t)-M_{n,k}^K(\tau)|.

			&\le2^{q/2}B_qm_q(K)

			&\le N_{q,q'}^K\delta^{1-q/q'}.

		\emph{Path space.} The embedding $\mathbb{H}_{-r_0}\hookrightarrow \mathbb{H}_{-r}$ is compact.
		By Lemma~\ref{lem2}(b) and Markov's inequality,
		\begin{equation}\label{eq:compact-containment}
			\sup_{n\ge1}\mathbb{P}\left(\sup_{t\in[0,T]}
			\|u_n^K(t)\|_{\mathbb{H}_{-r_0}}>R\right)
			\le C_{1,2}^K(q,r_0)R^{-q}.
		\end{equation}
		Here the modifications in the two Sobolev spaces agree at all times
		almost surely, by equality at rational times and at $T$, and their
		c\`adl\`ag paths. Since bounded balls of $\mathbb{H}_{-r_0}$ have compact
		closure in $\mathbb{H}_{-r}$, this gives compact containment.
		
		To prove tightness, we use Aldous's criterion \cite{Aldous}*{Theorem~1}
		in the metric-space form of \cite{LalleyTang20}*{Theorem~2.2}. Since
		the latter is stated for paths on $[0,\infty)$, we first extend the
		paths after $T$ by keeping them constant. Some care is needed when we
		come back to $[0,T]$. On $[0,\infty)$, the time changes in the
		$\mathrm{J}_1$ topology may move the point $T$, while on $[0,T]$ they
		must keep $T$ fixed. So, if the limit path jumps at $T$, convergence
		on $[0,\infty)$ does not give convergence of the restrictions to
		$[0,T]$. For this reason, we first show that every subsequential
		limit is almost surely continuous at $T$.
		
		Set $\widehat u_n^K(t):=u_n^K(t\wedge T)$ for all $t\ge0$, and keep the
		filtration equal to $\mathcal{F}_T$ after time $T$. Since $\widehat u_n^K(t)=u_n^K(t\wedge T)$, the bound
		\eqref{eq:compact-containment} also holds for $\widehat u_n^K$,
		with the supremum taken over $t\ge0$. Let $\tau$ be a bounded stopping time and
		$0<h\le1$. Then $\tau\wedge T$ is a stopping time bounded by $T$, and
		$\widehat u_n^K(\tau+h)-\widehat u_n^K(\tau)
		=u_n^K(((\tau\wedge T)+h)\wedge T)-u_n^K(\tau\wedge T)$. So

		\le C_{2,1}^K(q,q',r)h^\gamma.
		\]

		give the two conditions of \cite{LalleyTang20}*{Theorem~2.2}, so the
		laws of $\widehat u_n^K$ are tight in $\mathbb{D}([0,\infty);\mathbb{H}_{-r})$ with
		the $\mathrm{J}_1$ topology.
		
		By Prokhorov's theorem \cite{EK}*{Chapter~3, Theorem~2.2}, every
		subsequence of $(\widehat u_n^K)$ has a further subsequence that
		converges in distribution in $\mathbb{D}([0,\infty);\mathbb{H}_{-r})$. Let
		$\widehat u^K$ be the limit of such a subsequence, still indexed by
		$n$, defined on some probability space with expectation $\mathbb{E}_*$.

		$\mathrm{J}_1$ topology, so $\widehat u^K$ is also constant on
		$[T,\infty)$, almost surely. In particular, $T+1$ is an almost sure

		$\widehat u^K$. This is possible because a c\`adl\`ag process can jump

		with $\tau=T-\delta_j$ and $\delta=\delta_j$ gives

			\mathbb{E}\|\widehat u_n^K(T+1)-\widehat u_n^K(T-\delta_j)\|_{\mathbb{H}_{-r}}^q
			\le C_{2,1}^K(q,q',r)\delta_j^\gamma.
		\end{equation}
		Since $T+1$ and $T-\delta_j$ are continuity times of the limit, the
		difference $\widehat u_n^K(T+1)-\widehat u_n^K(T-\delta_j)$ converges
		in law to $\widehat u^K(T+1)-\widehat u^K(T-\delta_j)$. The function
		$x\mapsto\|x\|_{\mathbb{H}_{-r}}^q$ is continuous and nonnegative, so the
		expectation of its value at the limit is at most the $\liminf$ of the
		expectations; this follows from \cite{EK}*{Chapter~3, Theorem~3.1(e)},
		applied to the open sets $\{x:\|x\|_{\mathbb{H}_{-r}}^q>a\}$, $a\ge0$, and
		Fatou's lemma. By
		\eqref{eq:increment-at-T},
		\[
		\mathbb{E}_*\|\widehat u^K(T+1)-\widehat u^K(T-\delta_j)\|_{\mathbb{H}_{-r}}^q
		\le C_{2,1}^K(q,q',r)\delta_j^\gamma.
		\]
		Since $\widehat u^K(T+1)=\widehat u^K(T)$ and
		$\widehat u^K(T-\delta_j)\to\widehat u^K(T-)$ as $j\to\infty$,
		Fatou's lemma gives
		$\mathbb{E}_*\|\widehat u^K(T)-\widehat u^K(T-)\|_{\mathbb{H}_{-r}}^q=0$. Thus
		$\widehat u^K$ is almost surely continuous at $T$.

			e^{i\theta M_n^\varphi(t)}={}&1+\mathcal{M}_n^\varphi(t)\\
			&+\int_{\mathcal{Z}_t}e^{i\theta M_n^\varphi(s-)}
			\left[\int_{\mathbb{R}\setminus\{0\}}
			R_\theta(zH_n^\varphi(s,y))\nu_\alpha(\,\mathrm{d} z)\right]\,\mathrm{d} s\,\mathrm{d} y,
		\end{aligned}
		\]
		where
		\[
		\mathcal{M}_n^\varphi(t):=\int_{\mathcal{Q}_t}e^{i\theta M_n^\varphi(s-)}
		\bigl(e^{i\theta zH_n^\varphi(s,y)}-1\bigr)
		\widetilde N(\,\mathrm{d} s,\,\mathrm{d} y,\,\mathrm{d} z).
		\]
		This is a square-integrable complex-valued martingale. Indeed,
		$H_n^\varphi$ is bounded for fixed $n$, and the integrand has modulus at most
		$|\theta|\|H_n^\varphi\|_{\infty}|z|$ for $|z|\le1$, and at most $2$ for
		$|z|>1$. These bounds are square-integrable against the intensity
		on $\mathcal{Q}_T$, since $\alpha<2$.
		For every $h\in\mathbb{R}$, symmetry and the normalization of $L$ give
		\[
		\int_{\mathbb{R}\setminus\{0\}}R_\theta(zh)\nu_\alpha(\,\mathrm{d} z)
		=-|\theta h|^\alpha.
		\]
		The integral is absolutely convergent, since its integrand is of
		order $z^2$ near zero and at most of order $1+|z|$ at infinity,
		with $1<\alpha<2$. Its imaginary part vanishes by symmetry, and
		its real part is the stable characteristic exponent. Consequently,
		\begin{equation}\label{eq:ito-complex-martingale}
			e^{i\theta M_n^\varphi(t)}=1+\mathcal{M}_n^\varphi(t)
			-|\theta|^\alpha\int_0^t e^{i\theta M_n^\varphi(s-)}\,\mathrm{d} A_n^\varphi(s).
		\end{equation}
		By \eqref{eq:ito-complex-martingale} and the product rule, it follows that
		\[
		\,\mathrm{d} Z_n^{\varphi,\theta}(t)
		=e^{|\theta|^\alpha A_n^\varphi(t)}\,\mathrm{d}\mathcal{M}_n^\varphi(t).
		\]
		Since $M_n^\varphi(0)=A_n^\varphi(0)=0$, $Z_n^{\varphi,\theta}$ has the stochastic-integral
		representation
		\[
		Z_n^{\varphi,\theta}(t)
		=1+\int_0^t e^{|\theta|^\alpha A_n^\varphi(s)}\,\mathrm{d}\mathcal{M}_n^\varphi(s).
		\]
		The integrand is predictable and locally bounded, because $A_n^\varphi$ is
		adapted and continuous. Hence, this stochastic integral is a
		complex-valued local martingale, and $Z_n^{\varphi,\theta}(0)=1$.
		
		Since $A_n^\varphi$ is adapted and continuous,
		$\tau_{n,R}^\varphi$ is a stopping time and
		$A_n^\varphi(t\wedge\tau_{n,R}^\varphi)\le R$.
		Thus
		\[
		|Z_n^{\varphi,\theta}(t\wedge\tau_{n,R}^\varphi)|
		=\exp\bigl(|\theta|^\alpha
		A_n^\varphi(t\wedge\tau_{n,R}^\varphi)\bigr)
		\le e^{|\theta|^\alpha R}.
		\]
		The stopped real and imaginary parts are bounded local martingales,
		hence martingales. This proves the claim.

		\,\mathrm{d}\widetilde W^{\varphi,\theta}(s).
		\]
		Its integrand is predictable and locally bounded. This proves the
		claimed local martingale property.

		\|Q_nF-F\|_{\mathbb{L}^2(\widetilde\mathbb{P}\otimes\operatorname{Leb})}
		\longrightarrow0.

		\quad \widetilde\mathbb{P}\text{-a.s.},
		\]

		\int_{\mathcal{Z}_T}|F_{n_\ell}-F|^\alpha

		\qquad \widetilde\mathbb{P}\text{-a.s.}
		\]

		\[
		\int_{\mathcal{Z}_T}|F^{(m)}-F|^\alpha
		\,\mu_{\widetilde u}(\,\mathrm{d} s,\,\mathrm{d} y)
		\longrightarrow0,
		\qquad \widetilde\mathbb{P}\text{-a.s.}
		\]

		$G_m\in\mathcal{S}_{\widetilde M}$ such that
		\[
		\widetilde\mathbb{P}\left(
		\int_{\mathcal{Z}_T}|G_m-F^{(m)}|^\alpha
		\,\mu_{\widetilde u}(\,\mathrm{d} s,\,\mathrm{d} y)>2^{-m}
		\right)<2^{-m}.

		|G_m-F|^\alpha
		\le 2^{\alpha-1}
		\bigl(|G_m-F^{(m)}|^\alpha+|F^{(m)}-F|^\alpha\bigr)
		\]
		then proves the result with $F_\ell=G_\ell$.

Fix $\varepsilon>0$, $t>0$ and $x\in D$.
		
		\smallskip
		\noindent\textit{Step 1. Apply an integrating factor to each weak identity.}
		Since $\tfrac12e_k''=-\lambda_ke_k$, the weak equation gives
		\[
		\,\mathrm{d}\langle\widetilde u(s),e_k\rangle
		=-\lambda_k\langle\widetilde u(s),e_k\rangle\,\mathrm{d} s
		+\langle b(\widetilde u(s,\cdot)),e_k\rangle\,\mathrm{d} s
		+\,\mathrm{d}\widetilde M^{e_k}(s).
		\]
		The drift pairing is defined for almost every $s$, which suffices
		for its time integral. The process $\widetilde M^{e_k}$ is a local
		martingale. Subsection~\ref{subsec:stable-noise-reconstruction} gives
		$\widetilde M^{e_k}(a)=\int_{\mathcal{Z}_a}e_k(y)
		\sigma(\widetilde u(s,y))\widetilde L(\,\mathrm{d} s,\,\mathrm{d} y)$.
		The semimartingale product rule with the deterministic factor
		$e^{\lambda_ks}$ therefore yields
		\begin{equation}\label{eq:limit-coefficient-variation}
			\begin{aligned}
				\langle\widetilde u(t),e_k\rangle
				={}&e^{-\lambda_kt}\langle u_0,e_k\rangle\\
				&+\int_{\mathcal{Z}_t}e^{-\lambda_k(t-s)}e_k(y)
				b(\widetilde u(s,y))\,\mathrm{d} s\,\mathrm{d} y\\
				&+\int_{\mathcal{Z}_t}e^{-\lambda_k(t-s)}e_k(y)
				\sigma(\widetilde u(s,y))\widetilde L(\,\mathrm{d} s,\,\mathrm{d} y).
			\end{aligned}
		\end{equation}
		Writing the last term as a space--time integral uses the
		multiplication property of stochastic integration: it holds first
		for step functions of time, then for the factor $e^{-\lambda_k (t-s)}$ by uniform approximation. 
		
		\smallskip
		\noindent\textit{Step 2. Sum finitely many identities.}
		For $m\ge1$, put
		\[
		\rho_v^{(m)}(x,y):=\sum_{k=1}^m
		e^{-\lambda_kv}e_k(x)e_k(y),\qquad v>0,
		\qquad
		S_{\varepsilon,m}(t,x):=\sum_{k=1}^m
		e^{-\lambda_k\varepsilon}e_k(x)\langle\widetilde u(t),e_k\rangle.
		\]
		Multiplying \eqref{eq:limit-coefficient-variation} by
		$e^{-\lambda_k\varepsilon}e_k(x)$ and summing over $k\le m$ gives
		\begin{equation}\label{eq:regularized-finite-identity}
			\begin{aligned}
				S_{\varepsilon,m}(t,x)
				={}&\int_D\rho_{t+\varepsilon}^{(m)}(x,y)u_0(y)\,\mathrm{d} y\\
				&+\int_{\mathcal{Z}_t}\rho_{t-s+\varepsilon}^{(m)}(x,y)
				b(\widetilde u(s,y))\,\mathrm{d} s\,\mathrm{d} y\\
				&+\int_{\mathcal{Z}_t}\rho_{t-s+\varepsilon}^{(m)}(x,y)
				\sigma(\widetilde u(s,y))\widetilde L(\,\mathrm{d} s,\,\mathrm{d} y)
				=:\mathcal{T}_{\varepsilon,m}(\widetilde u)(t,x).
			\end{aligned}
		\end{equation}
		Taking the intersection
		of the probability-one events for the countably many $k$, the
		identities \eqref{eq:regularized-finite-identity} hold for every
		$m$ on one event of probability one.
		
		\smallskip
		\noindent\textit{Step 3. Pass to the limit on the right-hand side.}
		By $|e_k|\le\sqrt2$ and \eqref{eq:heat-kernel},
		\begin{equation}\label{eq:regularized-kernel-tail}
			\sup_{\substack{v\ge\varepsilon\\x,y\in D}}
			|\rho_v^{(m)}(x,y)-\rho_v(x,y)|
			\le 2\sum_{k>m}e^{-\lambda_k\varepsilon}
			=:d_{m,\varepsilon}\longrightarrow0 \quad\text{as } m\to\infty.
		\end{equation}

		respectively, by

		d_{m,\varepsilon}\|u_0\|_{\mathbb{L}^1(D)},

		\eqref{eq:weak-integrability}.

		\[
		H_{\varepsilon,m}^{t,x}(s,y)
		:=\bigl(\rho_{t-s+\varepsilon}^{(m)}(x,y)
		-\rho_{t-s+\varepsilon}(x,y)\bigr)
		\sigma(\widetilde u(s,y)).
		\]

			\int_{\mathcal{Z}_t}|H_{\varepsilon,m}^{t,x}(s,y)|^\alpha\,\mathrm{d} s\,\mathrm{d} y
			\le d_{m,\varepsilon}^\alpha
			\int_{\mathcal{Z}_t}|\sigma(\widetilde u(s,y))|^\alpha\,\mathrm{d} s\,\mathrm{d} y
			\longrightarrow0,\qquad\widetilde\mathbb{P}\text{-a.s.}

			&\widetilde\mathbb{P}\left(
			\left|\int_{\mathcal{Z}_t}H_{\varepsilon,m}^{t,x}(s,y)
			\widetilde L(\,\mathrm{d} s,\,\mathrm{d} y)\right|
			>\delta\right)\\
			&\quad\le

		The series converges since $\lambda_k=\pi^2k^2/2$ and

		\]
		Both terms on the right tend to zero. The probability on the left

		with respect to the natural filtration of $w$.

Let $\mathcal{F}_t^w:=\sigma(w(s):0\le s\le t)$.
		Set $w(0-):=w(0)$ and define
		\[
		S_nw(t,y)
		:= \mathfrak{s}_n(w(t-))(y) =
		\sum_{k=1}^n\left(1-\frac{k}{n+1}\right)
		\langle w(t-),e_k\rangle e_k(y).
		\]
		The left-limit coefficients are adapted and left-continuous,
		hence predictable. Therefore each $S_nw$ is measurable with respect
		to the predictable $\sigma$-field of $(\mathcal{F}_t^w)$ tensored with
		$\mathcal{B}(D)$.
		
		Define
		\[
		\bar w(\omega,t,y)
		:=
		\begin{cases}
			\displaystyle\lim_{n\to\infty}S_nw(\omega,t,y),
			&\text{if the limit exists in }\mathbb{R},\\
			0,&\text{otherwise}.
		\end{cases}
		\]
		Let $\mathcal{P}_0^w$ denote the predictable $\sigma$-field associated
		with $(\mathcal{F}_t^w)$. The set
		\[
		\bigl\{(\omega,t,y) \in \Omega \times [0,T] \times D :
		\lim_{n\to\infty}S_nw(\omega,t,y)
		\text{ exists in }\mathbb{R}\bigr\}
		\]
		belongs to $\mathcal{P}_0^w\otimes\mathcal{B}(D)$.
		Thus $\bar w$ is measurable with respect to this $\sigma$-field,
		and hence is a predictable field.
		
		Fix $\omega$ such that
		$v(\omega,\cdot,\cdot)\in\mathbb{L}^2((0,T)\times D)$,
		$w(\omega)$ is c\`adl\`ag, and
		\[
		\langle w(\omega,t),e_k\rangle
		=\int_D v(\omega,t,y)e_k(y)\,\,\mathrm{d} y
		\quad\text{for Leb-a.e. }t\in(0,T),\quad \forall k\ge1.
		\]
		These properties hold on an event of probability one.
		For this $\omega$, outside a Lebesgue-null set of times,
		$v(\omega,t,\cdot)\in\mathbb{L}^2(D)$, all the coefficient
		identities hold, and $w(\omega,t-)=w(\omega,t)$, since a c\`adl\`ag path
		has at most countably many jumps. Thus, for these times, $S_nw(t,\cdot)$ equals the weighted sine sum
		\[
		\sum_{k=1}^n\left(1-\frac{k}{n+1}\right)
		\left(\int_D v(t,z)e_k(z)\,\,\mathrm{d} z\right)e_k(\cdot),
		\]
		called the $n$th Fej\'er mean of $v(t,\cdot)$. Applied to its odd, $2$-periodic
		extension, \cite{Katznelson}*{Chapter~I, Theorem~3.2} gives
		\[
		S_nw(t,y)\longrightarrow v(t,y)
		\quad\text{for Leb-a.e. }y\in D.
		\]
		Tonelli's theorem therefore yields
		\[
		\mathbb{P}\bigl(\bar w=v
		\text{ Leb-a.e. on }\mathcal{Z}_T\bigr)=1.
		\]
		Hence $\bar w$ is a weak modification of $v$.
		
		The same approximations also converge in
		$\mathbb{L}^2((0,T)\times D)$, $\mathbb{P}$-a.s.
		Indeed, for the times considered above, put
		$c_k(t):=\int_D v(t,y)e_k(y)\,\,\mathrm{d} y$. Parseval's identity gives
		\[
		\|S_nw(t,\cdot)-v(t,\cdot)\|_{\mathbb{L}^2(D)}^2
		=
		\sum_{k=1}^n\left(\frac{k}{n+1}\right)^2|c_k(t)|^2
		+
		\sum_{k>n}|c_k(t)|^2
		\longrightarrow0.
		\]
		This expression is bounded by
		$\|v(t,\cdot)\|_{\mathbb{L}^2(D)}^2$.
		Since $v\in\mathbb{L}^2((0,T)\times D)$, dominated convergence gives
		\[
		\|S_nw-v\|_{\mathbb{L}^2((0,T)\times D)}
		\longrightarrow0,
		\qquad \mathbb{P}\text{-a.s.}
		\]
