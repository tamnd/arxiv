<!-- theorem-1: body_md -->
\label{thm:tree-profile-intro}
Let $T$ be a finite tree with $m:=|E(T)|\ge1$. Then
\begin{equation}
\label{eq:tree-feasible-range-intro}
\mathrm{DN}_{\mathrm{LLY}}^{k}(T)<\infty
\quad\Longleftrightarrow\quad
0\le k\le\frac{2}{m}.
\end{equation}
If $m=1$, then $T=K_2$ and
\[
\mathrm{DN}_{\mathrm{LLY}}^{k}(K_2)=1
\qquad(0\le k\le2).
\]
Assume $m\ge2$. For every $k\in[0,2/m]$, the map $F_{T,k}$ has a
unique normalized fixed point
$w_T^{\star,k}\in[1,\infty)^{E(T)}$, and
\[
\mathrm{DN}_{\mathrm{LLY}}^{k}(T)
=\max_{e\in E(T)}w_{T,e}^{\star,k}.
\]
This is the unique normalized $k$-optimal weight and satisfies
\[
(w_{T,e}^{\star,k}-1)
\bigl(\kappa_{\mathrm{LLY}}^{w_T^{\star,k}}(e)-k\bigr)=0
\qquad(e\in E(T)).
\]
For every edge $e$, both
\[
k\longmapsto w_{T,e}^{\star,k}
\quad\text{and}\quad
k\longmapsto\mathrm{DN}_{\mathrm{LLY}}^{k}(T)
\]
are nondecreasing and continuous on $[0,2/m]$. At $k=2/m$,
\[
\kappa_{\mathrm{LLY}}^{w_T^{\star,2/m}}(e)=\frac{2}{m}
\qquad(e\in E(T)).
\]

<!-- lemma-2: body_md -->
\cite[Lemma~3.4]{Xia2026InvariantI}
\label{lem:boundary-aggregation}
Let $G$ be finite and connected with nonempty $2$-core
$K:=\operatorname{core}_2(G)$, and let $A_G$ and $\widehat K_G$ be as above. Let $w$
be a positive edge weight on $G$. For $x\in A_G$, put
\[
b_x:=\sum_{u\in N_G(x)\setminus V(K)}w_{xu},
\]
and give $\widehat K_G$ the weight
\[
\widehat w_e=w_e\quad(e\in E(K)),
\qquad
\widehat w_{xx^\partial}=b_x.
\]
Then every core edge $x\sim y$ satisfies
\begin{equation}
\label{eq:boundary-aggregation}
\kappa_{\mathrm{LLY}}^{w,G}(x,y)
=
\kappa_{\mathrm{LLY}}^{\widehat w,\widehat K_G}(x,y).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let $x=x_0,\ldots,x_n=y$ be a shortest path from $x$ to $y$, and fix
$0<\varepsilon<k$. Since, by \eqref{eq:kappa_LLY},
\[
\kappa_{\mathrm{LLY}}^w(x_{i-1},x_i)
=
\lim_{\alpha\to1^-}
\frac{\kappa_\alpha^w(x_{i-1},x_i)}{1-\alpha}
\ge k
\]
for every $i=1,\ldots,n$, for $\alpha<1$ sufficiently close to $1$ we have,
simultaneously for all $i$,
\[
\frac{\kappa_\alpha^w(x_{i-1},x_i)}{1-\alpha}
\ge k-\varepsilon.
\]
Combining with \eqref{eq:kappa_alpha_W}, we have
\[
W_1(\mu_{x_{i-1}}^{\alpha,w},\mu_{x_i}^{\alpha,w})
\le
1-(1-\alpha)(k-\varepsilon).
\]
Writing $\delta_z$ for the Dirac measure at $z$, we have
$W_1(\delta_z,\mu_z^{\alpha,w})=1-\alpha$. Hence, by the triangle inequality,
\[
\begin{aligned}
d(x,y)=W_1(\delta_x,\delta_y)\le
W_1(\delta_x,\mu_x^{\alpha,w})
+\sum_{i=1}^n
W_1(\mu_{x_{i-1}}^{\alpha,w},\mu_{x_i}^{\alpha,w})
+W_1(\mu_y^{\alpha,w},\delta_y).
\end{aligned}
\]
Since $d(x,y)=n$, the preceding estimates give
\[
n\le2(1-\alpha)
+n\bigl(1-(1-\alpha)(k-\varepsilon)\bigr).
\]
Thus $n(k-\varepsilon)\le2$. Letting $\varepsilon\downarrow0$ gives
$d(x,y)\le2/k$.

<!-- theorem-4: body_md -->
\label{thm:rooted-response}
Let $k>0$, and let $G$ be a connected locally finite graph. Let $C$ be
a finite tree with $t:=|V(C)|\ge1$, rooted at $r$, and suppose that
$C$ is attached to $G-V(C)$ only through the boundary edge $e=xr$,
where $x\notin V(C)$. If a positive weight on $G$ makes $e$ and every
edge of $C$ have curvature at least $k$, then
\begin{equation}
\label{eq:rooted-share-lower}
p_x^w(e)=\frac{w_e}{d_x^w}\ge\frac{kt}{2}.
\end{equation}
If $A_x=d_x^w-w_e$ and $kt<2$, this is equivalent to
\begin{equation}
\label{eq:rooted-ratio-lower}
w_e\ge\frac{kt}{2-kt}A_x.
\end{equation}
If $kt=2$, it forces $A_x=0$.

Conversely, if the positive weights on all ambient edges incident with
$x$ are fixed and \eqref{eq:rooted-share-lower} holds, then the edges
of $C$ can be assigned positive weights so that every internal edge of
$C$ has curvature exactly $k$ and $e$ has curvature at least $k$.
Equality in \eqref{eq:rooted-share-lower} makes the boundary edge curvature
also equal to $k$.
<!-- end theorem-4 -->
<!-- proof-4: body_md -->
Put $c_k=1+k/2$. Since the boundary edge $e$ and every edge of $C$ are tree-like, \eqref{eq:share-curvature} applies. Summing it over the $t-1$ internal
edges of $C$ gives
\[
\sum_{uv\in E(C)}
\bigl(p_u^w(uv)+p_v^w(uv)\bigr)
=t-p_r^w(e)\ge(t-1)c_k,
\qquad
p_r^w(e)\le1-\frac{k(t-1)}2.
\]
Applying \eqref{eq:share-curvature} to $e=xr$ yields
\[
p_x^w(e)\ge c_k-p_r^w(e)\ge\frac{kt}{2},
\]
proving necessity; \eqref{eq:rooted-ratio-lower} is algebraically
equivalent.

For sufficiency, \eqref{eq:rooted-share-lower} implies $kt\le2$.
Root $C$ at $r$. For $v\in V(C)$ let $t_v$ be the number of vertices
in the descendant subtree rooted at $v$ and set
\[
\sigma_v:=1-\frac{k(t_v-1)}2.
\]
Since $t_v\le t$ and $kt\le2$, one has $\sigma_v>0$.
For each vertex $u\in V(C)$, let $e_u$ be its parent edge, with $e_r=e$. Set $[u]:=\{v\in V(C): v\ \text{is a child of}\ u\}$, and define recursively
\begin{equation}
\label{eq:direct-rooted-weight}
w_{uv}:=\frac{k t_v}{2\sigma_u}\,w_{e_u}.
\end{equation}
Since the child subtree sizes sum to $t_u-1$,
\[
d_u^w=w_{e_u}+\sum_{v\in [u]}w_{uv}=\frac{w_{e_u}}{\sigma_u},
\qquad
p_u^w(e_u)=\sigma_u,
\qquad
p_u^w(uv)=\frac{k t_v}{2}.
\]
Hence for every internal parent--child edge $uv$,
\[
p_u^w(uv)+p_v^w(uv)
=\frac{k t_v}{2}+1-\frac{k(t_v-1)}2
=1+\frac{k}{2},
\]
so its curvature is $k$ by \eqref{eq:share-curvature}. At the root,
$p_r^w(e)=1-k(t-1)/2$; together with
$p_x^w(e)\ge kt/2$, this gives curvature at least $k$ on $e$ by \eqref{eq:share-curvature}, with
equality when the inequality in \eqref{eq:share-curvature} is sharp.

<!-- definition-2: body_md -->
\label{def:k-loaded}
A positive weight $\widehat w$ on $\widehat K_G$ is
\emph{$k$-loaded admissible} if
\begin{align}
\kappa_{\mathrm{LLY}}^{\widehat w,\widehat K_G}(e)&\ge k
&& (e\in E(K)),\label{eq:kloaded-core}\\
\kappa_{\mathrm{LLY}}^{\widehat w,\widehat K_G}(xx^\partial)&\ge k h_G(x)
&& (x\in A_G).\label{eq:kloaded-boundary}
\end{align}
<!-- end definition-2 -->
<!-- proof-5: body_md -->
For $k=0$, the assertion is exactly the $2$-core reduction of
\cite{Xia2026InvariantI}, so assume $k>0$.

Let $w$ be $k$-admissible on $G$, and aggregate the first-layer weights
at each $x\in A_G$ by setting
\[
b_x:=\sum_{u\in N_G(x)\setminus V(K)}w_{xu}.
\]
By \cref{lem:boundary-aggregation}, core
curvatures are unchanged. If $e_C$ is the boundary edge of a component
$C\in\mathcal C_x$, \cref{thm:rooted-response} gives
\[
\frac{w_{e_C}}{d_x^w}\ge\frac{k|V(C)|}{2}.
\]
Summing over $C\in\mathcal C_x$ gives
$b_x/d_x^w\ge kh_G(x)/2$. Set 
\[
\widehat w_e:=w_e\quad(e\in E(K)),
\qquad
\widehat w_{xx^\partial}:=b_x.
\] 
Since aggregation preserves $d_x^w$, the
resulting proxy weight $\widehat w$ is $k$-loaded admissible.

Conversely, let $\widehat w$ be $k$-loaded admissible and fix
$x\in A_G$. Write $\mathcal C_x=\{C_1,\ldots,C_q\}$,
$t_i=|V(C_i)|$, and $b_x=\widehat w_{xx^\partial}$. By
\eqref{eq:kloaded-share}, choose positive $b_i$ with
\[
\sum_i b_i=b_x,
\qquad
\frac{b_i}{d_x^{\widehat w}}\ge\frac{k t_i}{2}.
\]
Assign $b_i$ to the boundary edge of $C_i$ and apply the converse part
of \cref{thm:rooted-response} independently to each component, then we can reconstruct a positive weight $w_G$ on G such that every edge of
$G\setminus K$ has curvature exactly $k$ and the boundary edge of $C_i$ has curvature at least $k$. The
total first-layer load at every core vertex remains $b_x$, so
\cref{lem:boundary-aggregation} preserves all core curvatures. The
reconstructed weight $w_G$ on $G$ is therefore $k$-admissible.

<!-- end proposition-3 -->
<!-- proof-7: body_md -->
If every edge has curvature at least $k$, then
\eqref{eq:total-curvature} gives $mk\le2$. Conversely, by the constant-curvature existence result of Lin--Liu \cite[Theorem~3.1]{LinLiu2026},
there is a positive weight $w$ on $T$ such that
\[
\kappa_{\mathrm{LLY}}^w(e)=\frac{\sum_{e\in E}\kappa_{\mathrm{LLY}}^w(e)}{m}=\frac{2(|V(T)|-|E(T)|)}{m}=\frac{2}{m}
\qquad(e\in E(T)).
\]
Hence this weight is $k$-admissible for every
$0\le k\le2/m$.

<!-- lemma-3: body_md -->
\label{lem:Psi-properties}
For $0\le k<2$, $\Psi_k$ is continuous, positively homogeneous of
degree one, and nondecreasing in each variable; for $k>0$ it is
strictly increasing in each variable.
<!-- end lemma-3 -->
<!-- proof-8: body_md -->
The first two assertions are immediate. For $k=0$,
\[
\Psi_0(a,b)=\sqrt{ab},
\]
which is nondecreasing in each variable.

Assume $k>0$. For $(a,b)\ne(0,0)$, direct differentiation gives
\[
\frac{\partial \Psi_k}{\partial a}
=
\frac{1}{2(2-k)}
\left(
k+
\frac{k^2(a-b)+8b}
{\sqrt{k^2(a-b)^2+16ab}}
\right).
\]
If $a\ge b$, this is clearly positive. If $a<b$, then
\[
\sqrt{k^2(a-b)^2+16ab}\ge k(b-a),
\]
and hence
\[
k\sqrt{k^2(a-b)^2+16ab}
+k^2(a-b)+8b
\ge 8b>0.
\]
Thus
\[
\frac{\partial \Psi_k}{\partial a}>0.
\]
By symmetry,
\[
\frac{\partial \Psi_k}{\partial b}>0.
\]
At the origin, strict increase along the coordinate axes follows from
\[
\Psi_k(a,0)=\frac{k}{2-k}a,
\qquad
\Psi_k(0,b)=\frac{k}{2-k}b.
\]
Therefore $\Psi_k$ is strictly increasing in each variable for $k>0$.

<!-- lemma-4: body_md -->
\label{lem:k-restoration}
Let $G$ be connected and locally finite, let $w$ be a positive edge
weight, and let $xy\inE_{\mathrm{tl}}(G)$. Put
\[
A_x=d_x^w-w_{xy},
\qquad
A_y=d_y^w-w_{xy}.
\]
For $0\le k<2$,
\begin{equation}
\label{eq:k-restoration-equivalence}
\kappa_{\mathrm{LLY}}^w(x,y)\ge k
\quad\Longleftrightarrow\quad
w_{xy}\ge\Psi_k(A_x,A_y),
\end{equation}
with equality on the right exactly when the curvature equals $k$.
<!-- end lemma-4 -->
<!-- proof-9: body_md -->
Substituting \eqref{eq:tree-like-curvature} and clearing denominators
gives
\[
(2-k)w_{xy}^2-k(A_x+A_y)w_{xy}-(2+k)A_xA_y\ge0.
\]
Its unique nonnegative root is \eqref{eq:Psi}.

<!-- definition-3: body_md -->
\label{def:k-restoration-map}
For $0\le k<2$, define
\[
F_{T,k}:[1,\infty)^{E(T)}\longrightarrow[1,\infty)^{E(T)}
\]
by
\begin{equation}
\label{eq:k-restoration-map}
(F_{T,k}(x))_{uv}
:=\max\left\{1,
\Psi_k(A_{u\mid v}(x),A_{v\mid u}(x))
\right\}.
\end{equation}
We call a vector $x\in [1,\infty)^{E(T)}$ a \emph{fixed point} of $F_{T,k}$ if 
\begin{equation}\label{eq:distortion-system}
    F_{T,k}(x)=x,
\end{equation}
and \emph{$k$-feasible} if $F_{T,k}(x)\le x$ coordinatewise. It is called \emph{normalized} if \[ \min_{e\in E(T)}x_e=1. \]
<!-- end definition-3 -->
<!-- theorem-5: body_md -->
\label{thm:k-tree-fixed-point}
Let $m:=|E(T)|\ge2$ and $0\le k\le2/m$. Then $F_{T,k}$ has a unique
normalized fixed point $w_T^{\star,k}$, which is coordinatewise minimal
among normalized $k$-admissible weights, and
\begin{equation}
\label{eq:k-tree-DN-fixed}
\mathrm{DN}_{\mathrm{LLY}}^{k}(T)=\max_{e\in E(T)}w_{T,e}^{\star,k}.
\end{equation}
For $T=K_2$, the unit weight is the unique normalized weight and
$\mathrm{DN}_{\mathrm{LLY}}^{k}(K_2)=1$ for $0\le k\le2$.
<!-- end theorem-5 -->
<!-- proof-10: body_md -->
By \cref{prop:tree-feasible-range}, choose a normalized
$k$-admissible vector $x^{(0)}$. Iterating
$x^{(n+1)}=F_{T,k}(x^{(n)})$ gives a coordinatewise decreasing sequence
bounded below by $\mathbf1$, because $F_{T,k}$ is order preserving.
Its limit is a fixed point by continuity. Since $x^{(0)}$ is normalized, some coordinate initially equals $1$;
that coordinate remains $1$ throughout the iteration. Hence the limit
is normalized.

For uniqueness, the case $k=0$ is exactly the finite-tree fixed-point
result of \cite{Xia2026InvariantI}, by \eqref{eq:F0-equals-F}. Let $k>0$ and let $x,y$ be normalized fixed
points. Set
\[
R:=\max_{e\in E(T)}\frac{x_e}{y_e}.
\]
Without loss of generality, let $R\ge1$. If $R>1$, choose $e=u\sim v$ attaining it. Then $x_e=Ry_e>1$, and
homogeneity and monotonicity of $\Psi_k$ give
\[
x_e
=
\Psi_k(A_{u\mid v}(x),A_{v\mid u}(x))
\le
R\Psi_k(A_{u\mid v}(y),A_{v\mid u}(y))
\le
Ry_e
=
x_e.
\]
Equality holds throughout. Strict monotonicity therefore forces
\[
A_{u\mid v}(x)=R A_{u\mid v}(y),
\qquad
A_{v\mid u}(x)=R A_{v\mid u}(y).
\]
The equality of the corresponding sums,
together with $x_f\le R y_f$ termwise, forces $x_f=R y_f$ on every
adjacent edge $f$. Repeating this along the connected edge set of the tree
gives $x=Ry$, contradicting normalization. Exchanging $x$ and $y$ gives $y\le x$, and hence $x=y$.

Starting the iteration from any normalized $k$-admissible vector $z$
produces the unique fixed point below $z$, proving coordinatewise
minimality. Since the fixed point is itself $k$-admissible,
\eqref{eq:k-tree-DN-fixed} follows by \cref{prop:attainment-monotonicity}. For $T=K_2$, the assertion is immediate by \cref{prop:tree-feasible-range}.

<!-- theorem-6: body_md -->
\label{thm:k-optimal-weight-uniqueness}
For $0\le k\le2/|E(T)|$, $w_T^{\star,k}$ is the unique normalized
$k$-optimal weight. Equivalently, every $k$-optimal
weight is $c\,w_T^{\star,k}$ for some $c>0$.
<!-- end theorem-6 -->
<!-- proof-11: body_md -->
At $k=0$ this is the uniqueness assertion in the finite-tree
fixed-point theorem of \cite{Xia2026InvariantI}. The case $T=K_2$ is immediate,
so assume $k>0$ and $|E(T)|\ge2$. Let $w$ be normalized and $k$-optimal.
By coordinatewise minimality,
\[
w\ge w_T^{\star,k}.
\]
Put $D=\mathrm{DN}_{\mathrm{LLY}}^{k}(T)=\max_e w_{T,e}^{\star,k}$. If $D=1$, both
weights are the unit weight.

Assume $D>1$ and suppose that
\[
w_f>w_{T,f}^{\star,k}
\]
for some $f\in E(T)$. Let $e=uv$ be adjacent to $f$, with
$f\ni u$, and assume $w_{T,e}^{\star,k}>1$. Since
$w\ge w_T^{\star,k}$,
\[
A_{u\mid v}(w)>A_{u\mid v}(w_T^{\star,k}),
\qquad
A_{v\mid u}(w)\ge A_{v\mid u}(w_T^{\star,k}).
\]
By $k$-admissibility of $w$ and strict monotonicity of $\Psi_k$,
\[
w_e\ge
\Psi_k\bigl(A_{u\mid v}(w),A_{v\mid u}(w)\bigr)>
\Psi_k\bigl(
A_{u\mid v}(w_T^{\star,k}),
A_{v\mid u}(w_T^{\star,k})
\bigr)=
w_{T,e}^{\star,k}.
\]
Thus strict inequality propagates from $f$ to every adjacent edge whose
canonical weight is greater than $1$.

For every non-pendant edge $e=uv$,
\[
A_{u\mid v}(w_T^{\star,k})\ge1,
\qquad
A_{v\mid u}(w_T^{\star,k})\ge1,
\]
and hence
\[
w_{T,e}^{\star,k}
\ge\Psi_k(1,1)>1.
\]
Choose $e_0$ such that
\[
w_{T,e_0}^{\star,k}=D.
\]
Let
\[
f=f_0,f_1,\ldots,f_r=e_0
\]
be a shortest sequence of pairwise adjacent edges. Every
$f_j$, $1\le j<r$, is non-pendant, and $w_{T,e_0}^{\star,k}=D>1$.
Applying the preceding comparison successively gives
\[
w_{f_j}>w_{T,f_j}^{\star,k}
\qquad(j=1,\ldots,r),
\]
and therefore
\[
w_{e_0}>w_{T,e_0}^{\star,k}=D,
\]
contrary to $\max_{e\in E(T)}w_e=D$. Hence
\[
w=w_T^{\star,k}.
\]
The unnormalized statement follows from scale invariance.

<!-- remark-1: body_md -->
Lin and Liu \cite[Proposition~3.2]{LinLiu2026} show that, on a finite
high-girth graph, prescribing the curvature of every edge determines the
weight up to scaling. In \cref{thm:k-optimal-weight-uniqueness}, the
individual edge curvatures are not prescribed; only the lower bound
\[

\qquad(e\in E(T))
\]
is imposed. Nevertheless, minimizing the distortion determines a unique
normalized weight. Moreover, this uniqueness phenomenon is special to

<!-- theorem-7: body_md -->
\label{thm:profile-regularity}
Let $m=|E(T)|$. For every edge $e$, the map
$k\mapsto w_{T,e}^{\star,k}$ is nondecreasing and continuous on
$[0,2/m]$. Consequently, $\mathcal D_T$ is nondecreasing and continuous
there. At $k=2/m$,
\begin{equation}
\label{eq:profile-endpoint}
\kappa_{\mathrm{LLY}}^{w_T^{\star,2/m}}(e)=\frac{2}{m}
\qquad(e\in E(T)).
\end{equation}
If $k<2/m$, at least one edge has curvature strictly larger than $k$.
<!-- end theorem-7 -->
<!-- proof-13: body_md -->
The case $K_2$ is immediate. Let $m\ge2$ and
$0\le k_1\le k_2\le2/m$. The $k_2$-canonical weight is
$k_1$-admissible, so minimality gives
\begin{equation}
\label{eq:canonical-monotonicity}
w_T^{\star,k_1}\le w_T^{\star,k_2}
\end{equation}
coordinatewise.

If $k_j\to k$, then by \eqref{eq:canonical-monotonicity},
\[
1\le w_{T,e}^{\star,k_j}\le w_{T,e}^{\star,2/m}
\qquad(e\in E(T)).
\]
Hence the sequence $\{w_T^{\star,k_j}\}$ lies in the compact set
\[
\prod_{e\in E(T)}
\left[1,w_{T,e}^{\star,2/m}\right].
\]
Let a convergent subsequence satisfy
\[
w_T^{\star,k_{j_\ell}}\longrightarrow w.
\]
Since
\[
F_{T,k_{j_\ell}}
\bigl(w_T^{\star,k_{j_\ell}}\bigr)
=
w_T^{\star,k_{j_\ell}}
\]
for every $\ell$, and since
$(k,x)\mapsto F_{T,k}(x)$ is continuous, passing to the limit gives
\[
F_{T,k}(w)=w.
\]
Thus $w$ is a fixed point of $F_{T,k}$. By uniqueness of the fixed point,
\[
w=w_T^{\star,k}.
\]
Thus every convergent subsequence of
$\{w_T^{\star,k_j}\}$ has the same limit $w_T^{\star,k}$.
Since the sequence lies in a compact set, it follows that
\[
w_T^{\star,k_j}\longrightarrow w_T^{\star,k}.
\]
Hence the map
\[
k\longmapsto w_{T}^{\star,k}
\]
is continuous on $[0,2/m]$.
In particular, the map
\[
k\longmapsto w_{T,e}^{\star,k}
\]
is continuous on $[0,2/m]$ for every $e\in E(T)$. Since $E(T)$ is finite and
\begin{equation}\label{eq:D_T_k}
    \mathcal D_T(k)=\max_{e\in E(T)}w_{T,e}^{\star,k},
\end{equation}
continuity of each $k\mapsto w_{T,e}^{\star,k}$ implies continuity of
$\mathcal D_T$. Since each map $k\longmapsto w_{T,e}^{\star,k}$ is nondecreasing, the function $\mathcal D_T$ is nondecreasing as well by \eqref{eq:D_T_k}.

At $k=2/m$, all curvatures are at least $2/m$ and their sum is $2$ by
\eqref{eq:total-curvature}, so all are equal. If $k<2/m$ and all
curvatures equaled $k$, the same identity would give $2=mk<2$, a
contradiction.
<!-- end proof-13 -->
<!-- proof-14: body_md -->
The sharp feasible interval is \cref{prop:tree-feasible-range}; the case
$K_2$ is included in \cref{thm:k-tree-fixed-point}. For $|E(T)|\ge2$,
the fixed-point formula and coordinatewise minimality are
\cref{thm:k-tree-fixed-point}, uniqueness of the normalized $k$-optimum is
\cref{thm:k-optimal-weight-uniqueness}, complementarity is
\cref{cor:complementarity}, and monotonicity, continuity, and the endpoint
constant-curvature statement are \cref{thm:profile-regularity}.

\label{thm:k-tree-like-comparison}
Let $G$ be connected and locally finite, let $k\ge0$, and let
$H\subseteq G$ be connected with $E(H)\subseteqE_{\mathrm{tl}}(G)$. Then
\begin{equation}
\label{eq:k-tree-like-comparison}

$\mathrm{DN}_{\mathrm{LLY}}^k(G)=\infty$, there is nothing to prove. Otherwise, restrict any finite-distortion $k$-admissible weight $w$ on
