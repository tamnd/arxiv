<!-- theorem-1: body_md -->
\label{theorem-intro1}
Let $F^\perp = (f_0, f_1, \ldots,f_n) \subseteq T$ be a complete intersection.
Let $\deg f_i = a_i$ and assume that  $a_0 \leqslant a_1\leqslant \ldots \leqslant a_n$. 
Assume that $f_i$ is non-reduced for some $i$. Let $k$ be the smallest 
such that $f_k$ is non-reduced. Assume that the subscheme 
$V(f_0,f_1,\ldots,f_{k-1})\subset \mathbb P^n$
is reduced.  
Let $J$ be the complete-intersection $(f_0,\ldots,f_{k-1},f_{k+1},\ldots,f_n)$
and let $HF(T/J,-)$ be the Hilbert function of $T/J$. Then ${\rm VSP}(F)$
is irreducible of dimension 
$$\sum_{\substack{i=0\\i\neq k}}^n HF(T/J,a_i-a_k)\,.$$
Moreover, if $\mathbb X\subset \mathbb P^n$ 
is a minimal apolar set to $F$, then
the ideal $I(\mathbb X)$ is a complete intersection of the form 
$$\big(f_i - \psi_if_k \;\big|\; i \in \{0,\ldots,n\}\setminus\{k\}\big)\,,$$
where $\psi_i$ is homogeneous of degree $a_i-a_k$.

\label{theorem-intro2}

<!-- proposition-1: body_md -->
\label{prop:hilbert_eq}
Let $F^\perp = (f_0, f_1, \ldots,f_n) \subseteq T$ be a complete intersection.
Let $\deg f_i = a_i$ and assume that  $a_0 \leqslant a_1\leqslant \ldots \leqslant a_n$. 
    Assume that $f_i$ is non-reduced for some $i$. Let $k$ be the smallest 
    such that $f_k$ is not reduced. Let $f_k=h^2g$ 
    for some $h$ with ${\rm deg}(h)\geqslant 1$. Assume that the subscheme 
    $V(f_0,f_1,\ldots,f_{k-1})\subset \mathbb P^n$
    is reduced.  
Let $J$ be the ideal $(f_0,\ldots,f_{k-1},f_{k+1},\ldots,f_n)$ and $\mathbb X \subset \mathbb{P}^n$ be a finite set such that $|\mathbb X|=\operatorname{Rk}(F)$.
Then for all $i \in \mathbb{Z}$, we have 
\[
\operatorname{HF}(T/I(\mathbb{X}), i) = \operatorname{HF}(T/J, i).
\]

<!-- proof-1: body_md -->
Recall the definition of $e$-computability from \cite[Definition 3.5]{CCCGW18}.
In Theorem \ref{thm2}, 
the rank of $F$ is computed by $h$, as can be see from 
equation (3.1) in the proof of \cite[Proposition 3.2]{BCMS26}. 
Applying \cite[Proposition 3.6]{CCCGW18}
we get
\begin{align}
    I(\mathbb{X}):(h) & = I(\mathbb{X}) \label{EQ:I:h} \\
    I(\mathbb{X}) +(h) & =F^\perp +(h)= J+(h). \label{EQ:I+h=J+h}
\end{align}
By Equation \eqref{EQ:I:h} we have $h$ is a nonzerodivisor on $I(\mathbb{X}).$
Hence we have 
\begin{equation}\label{EQ:HS1}
  (1-t^{\deg h}) H(T/I(\mathbb{X}), t)  = H(T/(I(\mathbb{X})+(h)), t),
\end{equation}
where $H(T/I(\mathbb{X}), t)$ denotes the Hilbert series of $T/I(\mathbb{X})$ in the variable $t.$
Similarly, $h$ is also a nonzerodivisor in $T/J.$ Therefore
\begin{equation}\label{EQ:HS2}
  (1-t^{\deg h}) H(T/J, t)  = H(T/(J+(h)), t).
\end{equation}
By Equation \eqref{EQ:I+h=J+h}, RHS of Equations \eqref{EQ:HS1} 
and \eqref{EQ:HS2} are equal. Hence we obtain
$$\operatorname{HF}(T/I(\mathbb{X}), i) = \operatorname{HF}(T/J, i)$$ 
for $i \geqslant 0.$
This completes the proof of the Proposition.

Let $J$ be the ideal $(f_0,\ldots,f_{k-1},f_{k+1},\ldots,f_n)$.
We first claim that 
\begin{equation}\label{EQ:I+h^2=J+h^2}
    I(\mathbb{X}) +(h^2) = J+(h^2).
\end{equation}
Note that we have $I(\mathbb{X}) \subset F^{\perp} \subset J+(h^2).$ 
Hence, $I(\mathbb{X}) +(h^2) \subseteq J+(h^2)$. 
By \eqref{EQ:I:h}, $h$ is a nonzerodivisor on $T/I(\mathbb X)$
and so the same is true for $h^2$. Clearly, $h^2$ is a 
nonzerodivisor on $T/J$. This gives
\begin{align*}
\operatorname{HF}(T/I(\mathbb{X}), i)- \operatorname{HF}(T/I(\mathbb{X}), i-2\deg(h))&=
    \operatorname{HF}(T/(I(\mathbb{X})+(h^2)), i)\\
    &\geqslant \operatorname{HF}(T/(J+(h^2)), i)\\
    &=\operatorname{HF}(T/J, i)- \operatorname{HF}(T/J, i-2\deg(h))\,.
\end{align*}
By Proposition \ref{prop:hilbert_eq}, the extremes are equal
and so we have equality everywhere in the above. This
shows that $I(\mathbb{X}) +(h^2) = J+(h^2)$.

Next we claim that $(F^\perp:h^2)=J+(g)$. If $h^2x=j+h^2gp$,
where $j\in J$, then $h^2(x-gp)\in J$, which shows that 
$x-gp\in J$. It follows that $(F^\perp:h^2)\subset J+(g)$. 
The other inclusion is obvious. This proves the claim. 

Since $F^{\perp} \subset J+(h^2)$, by 
equation \eqref{EQ:I+h^2=J+h^2} we obtain 
$F^{\perp}\subseteq I(\mathbb X)+(h^2)$. Hence, for 
$u \in F^{\perp}$, there exist 
$v \in I(\mathbb{X})$ and $h' \in T$ such that
\[
u = v +h^2h'.
\]
Note that $u-v \in F^{\perp}$. Therefore 
\begin{equation}\label{EQ:h^2colon}
    h^2h' \in (F^{\perp} \cap (h^2)) = h^2(F^{\perp}\colon (h^2)).
\end{equation}
Moreover, $F^{\perp}:(h^2) = J+(g).$ Substituting this in Equation \eqref{EQ:h^2colon} we obtain
\begin{equation}
    h^2h' \in h^2(J+(g)) = (h^2)J+(h^2g)=(h^2)J+(f_k).
\end{equation}
Hence we obtain $u \in I(\mathbb{X}) +(h^2)J+(f_k).$
Therefore, $F^{\perp} \subseteq I(\mathbb{X}) +(h^2)J+(f_k)$.
Since the reverse inclusion is trivial, we obtain the
equality
\begin{equation}\label{EQ:NAK}
    F^{\perp} = I(\mathbb{X})+(h^2)J +(f_k).
\end{equation}
Suppose $I(\mathbb{X}) +(f_k)\subsetneqq F^{\perp}$. 
Choose a homogeneous polynomial 
$H\in F^{\perp}\setminus (I(\mathbb{X}) +(f_k))$ of 
least degree possible. Using \eqref{EQ:NAK}, write 
$H=v+ h^2v'$ where $v\in I(\mathbb X)+(f_k)$ and $v' \in J$. 
Since $\deg v' <\deg H$, $v' \in I(\mathbb X)+(f_k).$ 
This means that $H=v+ h^2v'\in I(\mathbb X)+(f_k)$, 
a contradiction. This completes the proof of the Lemma.

\begin{equation}\label{eq:J-in-I-plus-C}

\end{equation}

\emph{Base case $d= a_k-1$.} Since $d<a_k$, the ideal 
$J_{\leqslant d}$ is generated by those $f_i$ for which 
$a_i<a_k$. Similarly, $\mathcal I_{\leqslant d}$ is generated

that $J_{\leqslant d}=\mathcal I_{\leqslant d}$.

holds in every degree strictly less than $d$,

    we obtain that all the $f_i$ are reduced for $0\leqslant i\leqslant n-1$.
    Now we apply Theorem \ref{Thm:VSP-main} to obtain the result.

\label{counterexample}
Let $I= (Y^2-Z^2, XY, X^5-Z^5) \subset T=\mathbb{C}[X,Y,Z]$. 
Note that $I$ is an Artinian complete intersection ideal. By Macaulay2 \cite{M2}
we obtain that for $F = (x^5+y^5)z+y^3z^3+yz^5$, we have $F^{\perp} = I$.
We claim that any minimal 
apolar ideal of $F$ is not a complete intersection.
\begin{proof}
In \cite[Example 6.1]{BCMS26} it is proved that $4 < \operatorname{Rk} (F) \leqslant 8$.
We now show that any minimal apolar 
ideal of $F$ is not a complete intersection. 
We prove this by contradiction. Suppose there 
exists a minimal apolar ideal 
$I(\mathbb{X}) \subset F^{\perp}$ such that 
$I(\mathbb{X}) = (g_1, g_2)$ is a complete 
intersection ideal. Since $I(\mathbb X)\subset F^\perp$,
$\deg (g_i)\geqslant 2$.
Without loss of generality assume 
$\deg (g_1) \leqslant \deg (g_2)$.
Since
$\deg (g_1) \cdot \deg (g_2) = |\mathbb{X}|=\operatorname{Rk}(F)$, 
possible degrees for $g_1, g_2$ are $(2,3), (2,4)$. 
But any such ideal 
$I(\mathbb{X}) \subset (XY, Y^2-Z^2)$ since 
$(F^{\perp})_i =(XY, Y^2-Z^2)_i$ for 
$2 \leqslant i \leqslant 4$. 
This implies $V(XY, Y^2-Z^2)\subset \mathbb X$ 
which is a contradiction as $\mathbb X$ is a reduced zero
dimensional scheme whereas $V(XY, Y^2-Z^2)$
is not reduced. Hence, in this case, no minimal 
apolar ideal of $F$ is a complete intersection.
\end{proof}
