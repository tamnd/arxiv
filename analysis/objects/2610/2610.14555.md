<!-- definition-1: body_md -->
\label{def:strong-solution}
Let $\tau>0$, $v_0\in V$, and $T_0\in V_T$. A pair $(v,T)$ is a strong solution on $[0,\tau]$ if
\begin{align*}
v&\in L^\infty(0,\tau;V)\cap L^2(0,\tau;D(A)),&\partial_tv&\in L^2(0,\tau;H),\\
T&\in L^\infty(0,\tau;V_T),&\nabla_HT&\in L^2(0,\tau;H^1(\Omega)),\quad\partial_tT\in L^2(0,\tau;L^2(\Omega)),
\end{align*}
and the reformulated equations hold almost everywhere with the stated boundary and initial conditions.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:maximal-time}
The maximal existence time is $\mathcal T^*:=\sup\{\tau>0:(v,T)\text{ is a strong solution on }[0,\tau]\}$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:global-solution}
A strong solution is global if it exists on $[0,\tau]$ for every finite $\tau>0$.
<!-- end definition-3 -->
<!-- theorem-1: body_md -->
\label{thm:global-well-posedness}
Let $(v_0,T_0)\in V\times V_T$. Then the primitive equations subject to the physical boundary conditions above admit a unique global strong solution.
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:linear-velocity}
\textbf{The velocity equation} Fixed $t_0$, and $\mathbf{v}^{n-1}$ is given in $X_{t_0}$ associated with $\mathbf{w^{n-1}}$ $\&$ ${T}^*$ is given in $T_{t_0}$, then the following equation has a unique solution $(v,w)$ with $v\in$ $X_{t_0}$.
\begin{align*}
\partial_t v + (\mathbf{v}^{n-1} \cdot \nabla_H) v + (\mathbf{w}^{n-1} \partial_z) v &+ \nabla_H p - \Delta v + f_0 k \times v = 0, \\
\partial_z p &+ T^* = 0, \\
\nabla_H \cdot \mathbf{v} &+ \partial_z \mathbf{w} = 0
\end{align*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Uniqueness follows from the $L^2$ energy estimate for the difference
of two solutions. To prove existence, let $\{V_k\}_{k\geq1}\subset D(A)$
be hydrostatic Stokes eigenfunctions satisfying $AV_k=\lambda_kV_k$.
They form an orthonormal basis of $H$ with respect to the
$L^2(\Omega)^2$ inner product and are orthogonal in $V$.
Let $P_m:H\to\operatorname{span}\{V_1,\ldots,V_m\}$ be the
$L^2$-orthogonal projection. For a fixed $m\in\mathbb N$, consider the Galerkin system,
\begin{align*}
\partial_t v^m
&+ P_m\mathbb{P}\Bigl(
    (v^{n-1}\cdot\nabla_H)v^m
    + w^{n-1}\partial_z v^m
  \Bigr)
  - \mathbb{P}\Delta v^m
  \notag\\
&= -P_m\mathbb{P}(f_0 k\times v^m)
   +P_m\mathbb{P}\left(
      \nabla_H\int_{-h}^{z}T^*\,d\xi
    \right),\\
v^m(0)&=P_m v_0.
\end{align*}

Take the inner product of the first equation above with $V_k$. Denote $(v^m, V_k) = g_k(t)$, $\beta_{kj} = \left( \mathbb{P} \left[ (v^{n-1} \cdot \nabla_H) V_j + (w^{n-1} \partial_z) V_j \right], V_k \right)$,$\quad \alpha_{j,k} = (f_0 k \times V_j, V_k), \eta_k=(\nabla_H\int_{-h}^zT^*,V_k)$, therefore, we have
\begin{align*}
g_k'(t) + \lambda_k g_k(t) + \sum g_j (\beta_{kj}+\alpha_{jk}) &= \eta_k
\end{align*}
Let $g^m=(g_1,\dots,g_m)^{T}$,$\alpha^m=(\alpha_{ij})_{1\leq i,j \leq m}$, $\beta^m=(\beta_{ij})_{1\leq i,j \leq m}$, $\eta^m=(\eta_1,\dots,\eta_m)^T$ and $\Lambda^m$ is the diagonal matrix with $j$-th diagonal element $\lambda_j$
\begin{align*}
\partial_t g^m(t) + [\Lambda^m+\beta^m(t)+(\alpha^m)^T] g^m(t)  &= \eta^m\\
g^m(0)&={g^m_0}
\end{align*}

\noindent where $g_0^m=
\bigl((v_0,V_1)_{L^2},\ldots,(v_0,V_m)_{L^2}
\bigr)^{\mathsf T}$,$\int_0^{t_0}|\Lambda^m+\beta^m(s)+(\alpha^m(s))^T|ds<\infty$. Therefore, because of the ODE theory, it has the unique solution $v^m$ on $[0,t_0]$.
Under classic consideration, we may show $v^m$ is uniformly bounded in $X_{t_0}$ with respect to m. Therefore, by passing to a subsequence, there exists a $v$ such that $v^m \rightarrow v$, which is the solution of,
\begin{align*}
\partial_t v + \mathbb{P} \left( (v^{n-1} \cdot \nabla_H) v + (w^{n-1} \partial_z) v \right) + Av &= - \mathbb{P} (f_0 k \times v)+\mathbb{P}(\nabla_H \int_{-h}^zT^* d\xi) \\
v(0) &= v_0
\end{align*}
which implies that $v$ is the solution of the original equation.

<!-- lemma-2: body_md -->
\label{lem:linear-temperature}
Let $t_0>0$ and suppose that
\[
v^{n-1}\in X_{t_0}
:=
L^\infty(0,t_0;V)
\cap
L^2(0,t_0;D(A)).
\]
\[
w^{n-1}(x,y,z,t)
:=
-\int_{-h}^{z}
\nabla_H\cdot v^{n-1}(x,y,\xi,t)\,d\xi .
\]
Then, for every $T_0\in V_T$, the problem
\begin{equation}
\label{eq:linear-temperature-original}
\left\{
\begin{aligned}
\partial_tT^n
&+
(v^{n-1}\cdot\nabla_H)T^n
+
w^{n-1}\partial_zT^n
-
\Delta_HT^n
=0
&&\text{in }\Omega\times(0,t_0),\\
T^n&=0
&&\text{on }\Gamma_s\times(0,t_0),\\
T^n(\cdot,0)&=T_0
&&\text{in }\Omega
\end{aligned}
\right.
\end{equation}
admits a unique solution satisfying
\[
T^n\in L^\infty(0,t_0;V_T),
\qquad
\nabla_HT^n\in L^2(0,t_0;H^1(\Omega)),
\qquad
\partial_tT^n\in L^2(0,t_0;L^2(\Omega)).
\]
Moreover,
\begin{equation}
\label{eq:linear-temperature-L2}
\|T^n(t)\|_{L^2}^2
+2\int_0^t\|\nabla_HT^n(s)\|_{L^2}^2\,ds
=\|T_0\|_{L^2}^2,
\qquad 0\leq t\leq t_0,
\end{equation}
and
\begin{align}
&\sup_{0\leq t\leq t_0}
\|\nabla T^n(t)\|_{L^2}^2
\nonumber\\
&\quad+
\int_0^{t_0}
\left(
\|\Delta_HT^n(t)\|_{L^2}^2
+
\|\nabla_H\partial_zT^n(t)\|_{L^2}^2
\right)\,dt
\leq C_{n-1}(t_0)\|T_0\|_{H^1}^2,
\label{eq:linear-temperature-H1}
\end{align}
where $C_{n-1}(t_0)$ depends only on $\Omega$, $t_0$, and
\[
\|v^{n-1}\|_{L^\infty(0,t_0;H^1)}
+
\|v^{n-1}\|_{L^2(0,t_0;H^2)}.
\]

To define the
temperature projection $P_N^T$, let $\{\phi_j\}_{j\geq1}$ be the
$L^2(M)$-orthonormal basis of horizontal Dirichlet problem, ordered by nondecreasing eigenvalues, i.e.
\[
 -\Delta_H\phi_j=\lambda_j\phi_j\quad\hbox{in }M,
 \qquad \phi_j=0\quad\hbox{on }\partial M.
\]
In the vertical variable, use the $L^2(-h,0)$-orthonormal cosine basis
\[
 \chi_0(z)=h^{-1/2},\qquad
 \chi_k(z)=\sqrt{\frac2h}
       \cos\!\left(\frac{k\pi(z+h)}h\right),\quad k\geq1.
\]

Set $\mu_k=(k\pi/h)^2$ for $k\geq0$, so that
$-\chi_k''=\mu_k\chi_k$ and $\chi_k'(-h)=\chi_k'(0)=0$.
The products $\phi_j(x,y)\chi_k(z)$ form an orthonormal
basis of $L^2(\Omega)$. This basis is the analog of the  basis used by Saal and Slav\'ik
\cite[Eq.~(2.17)]{SaalSlavik2023}. For $N\geq1$, define
\[
 \mathcal E_N:=\operatorname{span}
 \{\phi_j\chi_k:1\leq j\leq N,\ 0\leq k\leq N\}.
\]
Let $P_N^T:L^2(\Omega)\to\mathcal E_N$ be the $L^2$-orthogonal
projection; explicitly,
\[
 P_N^Tf
 =\sum_{j=1}^N\sum_{k=0}^N
    \left(\int_\Omega f(x,y,z)\phi_j(x,y)\chi_k(z)\,dx\,dy\,dz\right)
          \phi_j\chi_k.
\]

\[
 \|f\|_{H^1}^2
 =\sum_{j\geq1}\sum_{k\geq0}
 (1+\lambda_j+\mu_k)|(f,\phi_j\chi_k)|^2,
 \qquad f\in V_T,
\]
implies
\[
 \|P_N^Tf\|_{H^1}\leq\|f\|_{H^1},\qquad
 P_N^Tf\longrightarrow f\quad\text{in }H^1(\Omega).
\]
The Neumann condition satisfied by the basis is auxiliary:
this $H^1$ convergence imposes no vertical endpoint condition
on the limit.

We seek $T_N^n$ satisfying
\begin{equation}
 \begin{aligned}
 (\partial_tT_N^n,\psi)
 +(\nabla_HT_N^n,\nabla_H\psi)
 &+(v^{n-1}\cdot\nabla_HT_N^n
       +w^{n-1}\partial_zT_N^n,\psi)=0,
 \qquad \psi\in\mathcal E_N,\\
 &T_N^n(0)=P_N^TT_0.
 \end{aligned}
 \label{eq:linear-temperature-galerkin}
\end{equation}

This is a finite-dimensional linear system with integrable
time-dependent coefficients, and hence has a unique absolutely
continuous solution on $[0,t_0]$.
The divergence-free condition with boundary conditions gives,
\begin{equation}
 \frac12\frac{d}{dt}\|T_N^n\|_2^2
       +\|\nabla_HT_N^n\|_2^2=0.
 \label{eq:linear-temperature-approximate-L2}
\end{equation}
Testing \eqref{eq:linear-temperature-galerkin} by
$-\Delta_HT_N^n$ and $-\partial_{zz}T_N^n$, respectively, gives
\begin{align*}
 \frac12\frac{d}{dt}\|\nabla_HT_N^n\|_2^2
       +\|\Delta_HT_N^n\|_2^2
 &=\bigl(v^{n-1}\cdot\nabla_HT_N^n
          +w^{n-1}\partial_zT_N^n,\Delta_HT_N^n\bigr),\\
 \frac12\frac{d}{dt}\|\partial_zT_N^n\|_2^2
       +\|\nabla_H\partial_zT_N^n\|_2^2
 &=-\bigl(\partial_zv^{n-1}\cdot\nabla_HT_N^n,
                                      \partial_zT_N^n\bigr)\\
 &\quad+\int_\Omega(\nabla_H\cdot v^{n-1})
                                  |\partial_zT_N^n|^2.
\end{align*}
By Lemma~\ref{anisotropic inequality}, for every $\delta>0$,
\begin{align*}
 \left|\bigl(v^{n-1}\cdot\nabla_HT_N^n
          +w^{n-1}\partial_zT_N^n,\Delta_HT_N^n\bigr)\right|
 &\leq\delta\bigl(\|\Delta_HT_N^n\|_2^2
                   +\|\nabla_H\partial_zT_N^n\|_2^2\bigr)\\
 &+C_\delta\|v^{n-1}\|_{H^2}^2
                                  \|\nabla_HT_N^n\|_2^2\\
 &+C_\delta\bigl(1+\|v^{n-1}\|_{H^1}^4
             +\|v^{n-1}\|_{H^1}^2\|v^{n-1}\|_{H^2}^2\bigr)
                                  \|\partial_zT_N^n\|_2^2.
\end{align*}
And,
\begin{align*}
 &\left|\bigl(\partial_zv^{n-1}\cdot\nabla_HT_N^n,
                                  \partial_zT_N^n\bigr)\right|
 +\left|\int_\Omega(\nabla_H\cdot v^{n-1})
                                  |\partial_zT_N^n|^2\right|\\
 &\quad\leq\delta\bigl(\|\Delta_HT_N^n\|_2^2
                   +\|\nabla_H\partial_zT_N^n\|_2^2\bigr)
 +C_\delta(1+\|v^{n-1}\|_{H^2}^2)\|\nabla T_N^n\|_2^2.
\end{align*}
Choosing $\delta$ sufficiently small therefore yields
\begin{equation}
 \frac12\frac{d}{dt}\|\nabla T_N^n\|_2^2
 +\frac12\bigl(\|\Delta_HT_N^n\|_2^2
              +\|\nabla_H\partial_zT_N^n\|_2^2\bigr)
 \leq\Phi_{n-1}(t)\|\nabla T_N^n\|_2^2,
 \label{eq:linear-temperature-approximate-H1}
\end{equation}
where
\begin{equation}
 \Phi_{n-1}(t)
 :=C\left[1+\|v^{n-1}(t)\|_{H^1}^4
 +(1+\|v^{n-1}(t)\|_{H^1}^2)
                          \|v^{n-1}(t)\|_{H^2}^2\right]
 \in L^1(0,t_0).
 \label{eq:Phi-n}
\end{equation}
Gronwall's inequality gives
\begin{equation}
 \begin{aligned}
 &\sup_{0\leq t\leq t_0}\|T_N^n(t)\|_{H^1}^2
 +\int_0^{t_0}\bigl(\|\Delta_HT_N^n\|_2^2
                  +\|\nabla_H\partial_zT_N^n\|_2^2\bigr)\,dt\\
 &\hspace{30mm}\leq C_{n-1}(t_0)\|T_0\|_{H^1}^2.
 \end{aligned}
 \label{eq:linear-temperature-uniform}
\end{equation}
Here and below the constants are independent of $N$ and have
the dependence asserted in the lemma. Horizontal elliptic
regularity also bounds $\nabla_HT_N^n$ in $L^2(0,t_0;H^1)$.
Also,
\[
 \|\partial_tT_N^n\|_2
 \leq\|\Delta_HT_N^n\|_2
      +\|v^{n-1}\cdot\nabla_HT_N^n\|_2
      +\|w^{n-1}\partial_zT_N^n\|_2.
\]
Using the anisotropic inequality in Lemma~\ref{anisotropic inequality},
\begin{equation}
 \|\partial_tT_N^n\|_{L^2(0,t_0;L^2)}
 \leq C_{n-1}(t_0)\|T_0\|_{H^1}.
 \label{eq:temperature-time-uniform}
\end{equation}

By Lemma~\ref{lem:aubin-lions} in Appendix \ref{app:analytic-tools}, the preceding estimates yield a subsequence such that
\begin{align*}
 T_N^n&\longrightarrow T^n
       &&\text{in }C([0,t_0];L^2),\\
 T_N^n&\rightharpoonup^*T^n
       &&\text{in }L^\infty(0,t_0;V_T),\\
 \nabla_HT_N^n&\rightharpoonup\nabla_HT^n
       &&\text{in }L^2(0,t_0;H^1),\\
 \partial_tT_N^n&\rightharpoonup\partial_tT^n
       &&\text{in }L^2(0,t_0;L^2).
\end{align*}
In particular, $T^n(0)=T_0$, and $T^n$ has zero lateral trace.

To pass to the limit in the equation, write
\[
 G_N=v^{n-1}\cdot\nabla_HT_N^n
                  +w^{n-1}\partial_zT_N^n.
\]
The preceding estimates bound $G_N$ in $L^2(0,t_0;L^2)$.
For every smooth test function $\varphi$, integration by parts gives
\[
 \int_0^{t_0}(G_N,\varphi)\,dt
 =-\int_0^{t_0}
   (T_N^n,v^{n-1}\cdot\nabla_H\varphi
                  +w^{n-1}\partial_z\varphi)\,dt.
\]
Strong $L^2$ convergence of $T_N^n$ and
$v^{n-1},w^{n-1}\in L^2(0,t_0;L^2)$ identify the limit of this expression. Testing the Galerkin equation by $P_N^T\varphi$
and using $P_N^T\varphi\to\varphi$ in $L^2(0,t_0;H^1)$
therefore yields
\begin{align*}
 &\int_0^{t_0}\!\int_\Omega
 \bigl[-T^n\partial_t\varphi
       +\nabla_HT^n\cdot\nabla_H\varphi\\
 &\hspace{28mm}-T^n(v^{n-1}\cdot\nabla_H\varphi
                    +w^{n-1}\partial_z\varphi)\bigr]
 =\int_\Omega T_0\varphi(\cdot,0)
\end{align*}
whenever $\varphi|_{\Gamma_s}=0$ and
$\varphi(\cdot,t_0)=0$.
The values of $\varphi$ at $z=-h,0$ are unrestricted.
Thus $T^n$ solves \eqref{eq:linear-temperature-original}.

Weak lower semicontinuity yields
\eqref{eq:linear-temperature-H1}. The uniform $H^1$ bound and
$T^n\in C([0,t_0];L^2)$ give a weakly $H^1$-continuous
representative, so that the supremum can be taken over all
$t\in[0,t_0]$.
The transport bounds also hold for $T^n$, and hence every term
of its equation belongs to $L^2(0,t_0;L^2)$.
Therefore, we could multiply the equation by $T^n$, and derive the uniqueness and \eqref{eq:linear-temperature-L2}.

\label{prop:vn-L2}
Assume that $(v_0,T_0)\in V\times V_T$. Then, for every $n\geq1$ and $t>0$,

<!-- proof-3: body_md -->
Multiply the velocity equation by $v^n$, the transport and surface pressure terms vanish by incompressibility, the hydrostatic divergence constraint, and the boundary conditions. The Coriolis term also vanishes since $(k\times v^n)\cdot v^n=0$. It follows by the Cauchy Schwarz inequality, and Young's inequality,
\begin{align*}
\frac12\frac{d}{dt}\|v^n\|_{L^2}^2
+\|\nabla v^n\|_{L^2}^2
&=-\int_\Omega
\left(\int_{-h}^zT^n(\xi)\,d\xi\right)
\nabla_H\cdot v^n\,dx\\
&\leq\frac12\|\nabla v^n\|_{L^2}^2
+C\|T^n\|_{L^2}^2,
\end{align*}
Using \eqref{eq:linear-temperature-L2} and integrating in time, we obtain, for every $s\in[0,t]$,
\[
\|v^n(s)\|_{L^2}^2
+\int_0^s\|\nabla v^n(\tau)\|_{L^2}^2\,d\tau
\leq
\|v_0\|_{L^2}^2+C_hs\|T_0\|_{L^2}^2
=K_1(s).
\]
Since $K_1$ is nondecreasing and both terms on the left are nonnegative, it follows that
\[
\sup_{0\leq s\leq t}\|v^n(s)\|_{L^2}^2\leq K_1(t),
\qquad
\int_0^t\|\nabla v^n(s)\|_{L^2}^2\,ds\leq K_1(t).
\]
Adding these inequalities yields \eqref{eq:v-energy}.

The estimates for $v^n$ and $T^n$ are coupled. More precisely, the
higher-order estimate for $v^n$ involves the $H^1$-estimate of $T^n$,
whereas the latter depends only on the estimates of $v^{n-1}$.
Therefore, we prove the two propositions simultaneously by induction
on $n$.

For convenience, define
\begin{equation}
E_v^n(t)
:=
\sup_{0\le s\le t}
\|A^{1/2}v^n(s)\|_{L^2}^2
+
\int_0^t
\|Av^n(s)\|_{L^2}^2\,ds,
\label{eq:Ev-definition}
\end{equation}
and
\begin{equation}
E_T^n(t)
:=
\sup_{0\le s\le t}
\|\nabla T^n(s)\|_{L^2}^2
+
\int_0^t
\left(
\|\Delta_H T^n(s)\|_{L^2}^2
+
\|\nabla_H\partial_zT^n(s)\|_{L^2}^2
\right)\,ds.
\label{eq:ET-definition}
\end{equation}

\medskip
\noindent\textbf{Step 1. A conditional estimate for $T^n$.}
Let $K_2>0$ be a constant to be fixed in Step~2, and suppose that
\begin{equation}
E_v^{n-1}(t)\leq K_2,
\qquad 0\leq t\leq t^*\leq1.
\label{eq:conditional-velocity-bound}
\end{equation}
The temperature estimate follows directly from the calculation in
Lemma~\ref{lem:linear-temperature}. Indeed, by the norm equivalences
for $A$ and \eqref{eq:Phi-n},
\[
\int_0^t\Phi_{n-1}(s)\,ds
\leq C\bigl[(1+K_2^2)t+(1+K_2)K_2\bigr].
\]
Applying Gronwall's inequality to
\eqref{eq:linear-temperature-approximate-H1} and passing to the limit
as in that lemma therefore gives
\begin{equation}
E_T^n(t)\leq K_3(t)
:=2\|\nabla T_0\|_{L^2}^2
\exp\!\left\{C\bigl[(1+K_2^2)t+(1+K_2)K_2\bigr]\right\},
\label{eq:conditional-temperature-bound}
\end{equation}
where $C$ depends only on $\Omega$. Here the factor $2$ accounts for
the time supremum and the integrated dissipation. In particular,
for fixed $K_2$, this bound is independent of $n$ and
$tK_3(t)\to0$ as $t\downarrow0$.
We fix $K_2$ and close the joint induction in Step~2.

\medskip
\noindent
\textbf{Step 2. The higher-order estimate for $v^n$.}
We now derive the higher-order estimate for $v^n$. A direct
higher-order energy estimate is not sufficient because of the mismatch
of indices in the approximation scheme. We therefore introduce the
decomposition
\begin{equation}
\label{eq:velocity-decomposition}
v^n=\widetilde V+\widetilde v^n,
\qquad
w^n=\widetilde W+\widetilde w^n,
\end{equation}
where $\widetilde V$ is independent of $n$ and solves
\begin{equation}
\label{eq:linear-V}
\left\{
\begin{aligned}
\partial_t\widetilde V
-\Delta\widetilde V
+\nabla_H\widetilde P_s
+f_0k\times\widetilde V
&=0
&&\text{in }\Omega\times(0,t^*),\\
\nabla_H\cdot\int_{-h}^{0}\widetilde V\,dz
&=0
&&\text{in }M,\\
\partial_z\widetilde V&=0
&&\text{on }\Gamma_t,\\
\widetilde V&=0
&&\text{on }\Gamma_b\cup\Gamma_s,\\
\widetilde V(\cdot,0)&=v_0
&&\text{in }\Omega.
\end{aligned}
\right.
\end{equation}
The corresponding vertical velocity is defined by
\begin{equation}
\label{eq:linear-W}
\widetilde W(x,y,z,t)
:=
-\int_{-h}^{z}
\nabla_H\cdot\widetilde V(x,y,\xi,t)\,d\xi .
\end{equation}
In particular,
\[
\widetilde W=0
\qquad\text{on }\Gamma_t\cup\Gamma_b.
\]

For each $n\geq0$, we also set
\begin{equation}
\label{eq:remainder-w}
\widetilde w^n(x,y,z,t)
:=
-\int_{-h}^{z}
\nabla_H\cdot\widetilde v^n(x,y,\xi,t)\,d\xi .
\end{equation}
Then
\[
w^n=\widetilde W+\widetilde w^n.
\]

We choose the initial iterate to be the linear solution:
\begin{equation}
\label{eq:initial-iterate}
v^0:=\widetilde V,
\qquad
w^0:=\widetilde W.
\end{equation}
Consequently,
\begin{equation}
\label{eq:initial-remainder}
\widetilde v^0=0,
\qquad
\widetilde w^0=0.
\end{equation}
Notice that every iterate has the same initial value:
\[
v^n(\cdot,0)=v_0,
\qquad n\geq0,
\]
and hence
\begin{equation}
\label{eq:remainder-zero-initial}
\widetilde v^n(\cdot,0)=0,
\qquad n\geq0.
\end{equation}

Testing the projected form of \eqref{eq:linear-V} with
$\widetilde V$ and $A\widetilde V$, respectively, gives
\begin{align*}
\frac12\frac{d}{dt}\|\widetilde V\|_{L^2}^2
+\|A^{1/2}\widetilde V\|_{L^2}^2
&=0,\\
\frac12\frac{d}{dt}\|A^{1/2}\widetilde V\|_{L^2}^2
+\|A\widetilde V\|_{L^2}^2
&=-f_0(k\times\widetilde V,A\widetilde V)_{L^2}\\
&\leq\frac12\|A\widetilde V\|_{L^2}^2
+\frac{f_0^2}{2}\|\widetilde V\|_{L^2}^2.
\end{align*}
The first identity implies
$\|\widetilde V(t)\|_{L^2}\leq\|v_0\|_{L^2}$.
Integrating the second inequality therefore yields, for
$0\leq t\leq t^*\leq1$,
\begin{align}
E_{\widetilde V}(t)
&:=\sup_{0\leq s\leq t}\|A^{1/2}\widetilde V(s)\|_{L^2}^2
+\int_0^t\|A\widetilde V(s)\|_{L^2}^2\,ds
\nonumber\\
&\leq2\|A^{1/2}v_0\|_{L^2}^2
+2f_0^2\|v_0\|_{L^2}^2=:C_V.
\label{eq:CV-definition}
\end{align}
Moreover, since
$\partial_t\widetilde V=-A\widetilde V
-\mathbb{P}(f_0k\times\widetilde V)$, we have
\begin{equation}
\int_0^t\|\partial_t\widetilde V(s)\|_{L^2}^2\,ds
\leq2\int_0^t\|A\widetilde V(s)\|_{L^2}^2\,ds
+2f_0^2t\|v_0\|_{L^2}^2.
\label{eq:linear-V-time}
\end{equation}

We now derive the equation satisfied by the remainder.
Using
\[
v^{n-1}
=
\widetilde V+\widetilde v^{n-1},
\qquad
w^{n-1}
=
\widetilde W+\widetilde w^{n-1},
\]
and subtracting \eqref{eq:linear-V} from the $n$-th velocity
equation, we obtain
\begin{equation}
\label{eq:remainder-equation}
\left\{
\begin{aligned}
\partial_t\widetilde v^n
-\Delta\widetilde v^n
+\nabla_H\widetilde p_s^n
+f_0k\times\widetilde v^n
={}&
-\bigl(
  (\widetilde V+\widetilde v^{n-1})\cdot\nabla_H
 \bigr)
 (\widetilde V+\widetilde v^n)
\\
&-
(\widetilde W+\widetilde w^{n-1})
\partial_z(\widetilde V+\widetilde v^n)
+\nabla_H Q_T^n
&&\text{in }\Omega\times(0,t^*),\\
\nabla_H\cdot\int_{-h}^{0}\widetilde v^n\,dz
={}&0
&&\text{in }M,\\
\partial_z\widetilde v^n={}&0
&&\text{on }\Gamma_t,\\
\widetilde v^n={}&0
&&\text{on }\Gamma_b\cup\Gamma_s,\\
\widetilde v^n(\cdot,0)={}&0
&&\text{in }\Omega,
\end{aligned}
\right.
\end{equation}
where
\[
\widetilde p_s^n:=p_s^n-\widetilde P_s.
\]

The essential point is that the higher-order energy of
$\widetilde v^n$ vanishes initially. We therefore define
\begin{equation}
\label{eq:remainder-energy}
E_{\widetilde v}^n(t)
:=
\sup_{0\leq s\leq t}
\|A^{1/2}\widetilde v^n(s)\|_{L^2}^2
+
\int_0^t
\|A\widetilde v^n(s)\|_{L^2}^2\,ds.
\end{equation}

Fix
\[
K_2:=2(C_V+1).
\]
We prove by induction that, for a sufficiently small $t^*>0$
independent of $n$,
\begin{equation}
E_{\widetilde v}^n(t^*)\leq1,
\qquad n\geq0.
\label{eq:remainder-induction-goal}
\end{equation}
The case $n=0$ follows from \eqref{eq:initial-remainder}.
Assume the bound holds for $n-1$. Then, for $0\leq t\leq t^*$,
\[
E_v^{n-1}(t)
\leq2E_{\widetilde V}(t)+2E_{\widetilde v}^{n-1}(t)
\leq K_2,
\]
so Step~1 gives $E_T^n(t)\leq K_3(t)$.

For $a,b\in D(A)$, let
$w(a):=-\int_{-h}^z\nabla_H\cdot a(\xi)\,d\xi$.
The Sobolev embeddings and the anisotropic product estimate,
together with the norm equivalences for $A$, give
\[
\|(a\cdot\nabla_H)b+w(a)\partial_zb\|_{L^2}^2
\leq C\|A^{1/2}a\|_{L^2}\|Aa\|_{L^2}
       \|A^{1/2}b\|_{L^2}\|Ab\|_{L^2}.
\]
Testing \eqref{eq:remainder-equation} by $A\widetilde v^n$,
expanding the transport terms, and applying this estimate and
Young's inequality, we obtain
\begin{align*}
&\frac{d}{dt}\|A^{1/2}\widetilde v^n\|_{L^2}^2
+\|A\widetilde v^n\|_{L^2}^2\\
&\quad\leq C\Bigl(
1+\|A^{1/2}\widetilde V\|_{L^2}^2\|A\widetilde V\|_{L^2}^2
+\|A^{1/2}\widetilde v^{n-1}\|_{L^2}^2
 \|A\widetilde v^{n-1}\|_{L^2}^2\Bigr)
\|A^{1/2}\widetilde v^n\|_{L^2}^2\\
&\qquad+C\Bigl[
\|A^{1/2}\widetilde V\|_{L^2}^2\|A\widetilde V\|_{L^2}^2
+\|A^{1/2}\widetilde v^{n-1}\|_{L^2}
 \|A\widetilde v^{n-1}\|_{L^2}
 \|A^{1/2}\widetilde V\|_{L^2}\|A\widetilde V\|_{L^2}
+\|\nabla_HT^n\|_{L^2}^2\Bigr].
\end{align*}
Here and below $C$ depends only on $\Omega$ and $f_0$.
Set
\[
\rho(t):=\int_0^t\|A\widetilde V(s)\|_{L^2}^2\,ds.
\]
Since $\widetilde V$ is fixed, $\rho(t)\to0$ as $t\downarrow0$.
The induction hypothesis and the bound for $\widetilde V$ imply
\begin{align*}
\int_0^t\|A^{1/2}\widetilde v^{n-1}\|_{L^2}^2
             \|A\widetilde v^{n-1}\|_{L^2}^2\,ds
&\leq1,\\
\int_0^t\|A^{1/2}\widetilde V\|_{L^2}^2
             \|A\widetilde V\|_{L^2}^2\,ds
&\leq C_V\rho(t),\\
\int_0^t\|A^{1/2}\widetilde v^{n-1}\|_{L^2}
             \|A\widetilde v^{n-1}\|_{L^2}
             \|A^{1/2}\widetilde V\|_{L^2}
             \|A\widetilde V\|_{L^2}\,ds
&\leq\sqrt{C_V\rho(t)}.
\end{align*}

$\widetilde v^n$ yield
\[
E_{\widetilde v}^n(t)
\leq C\exp\!\bigl\{C[t+C_V\rho(t)+1]\bigr\}
\Bigl[C_V\rho(t)+\sqrt{C_V\rho(t)}+tK_3(t)\Bigr].

\[

\qquad

\]
which proves both propositions.

 \qquad
 \delta_\eta T(t):=\frac{T(t+\eta)-T(t)}{\eta}.
\]
We first bound the initial difference quotients.

Let $w_0=-\int_{-h}^z\nabla_H\cdot v_0(\xi)\,d\xi$.
The anisotropic estimates give,
\[
 \|w_0\partial_zv_0\|_2+\|w_0\partial_zT_0\|_2
 \leq C\|v_0\|_{H^2}
             \bigl(\|v_0\|_{H^2}+\|T_0\|_{H^2}\bigr).
\]
Hence $(F_0^v,F_0^T)$ defined following belong to $L^2(\Omega)$:
\[
\begin{aligned}
 &F_0^v:=Av_0+\mathbb P\left[
       (v_0\cdot\nabla_H)v_0+w_0\partial_zv_0+f_0k\times v_0
       -\int_{-h}^z\nabla_HT_0(\xi)\,d\xi\right],\\

\end{aligned}

\begin{align*}
\partial_t(v-v_0)+A(v-v_0)
+\mathbb P\Bigg[
  &v\cdot\nabla_H(v-v_0)+w\partial_z(v-v_0)\\
  &+(v-v_0)\cdot\nabla_Hv_0

    -\int_{-h}^{z}\nabla_H(T-T_0)(\xi)\,d\xi

\[
\begin{aligned}

 &\leq

\end{aligned}

\quad\text{for a.e. }t.

The right-hand side belongs to $L^1(0,t_0+\eta_0)$, so $Y$ is absolutely continuous. The initial traces give $Y(0)=0$.
Applying Lemma~\ref{lem:sqrt-gronwall} to the differential inequality $Y'(t)\leq C_0 Y+2 C_0 Y^{\frac{1}{2}}$, we obtain

\]

\end{aligned}
\]

$(v(t+\eta),T(t+\eta))$ and $(v(t),T(t))$ on $[0,t_0]$.

\eqref{eq:coupled-stability-coefficient} with
$(v^{(2)},T^{(2)})=(v(t),T(t))$, which is independent of $\eta$ and belongs to $L^1(0,t_0)$ by theorem~\ref{thm:global-well-posedness}.

controlling the supremum and the dissipation integral separately,
we obtain, for $0\leq t\leq t_0$,
\[
\begin{aligned}

       \bigl(\|\delta_\eta v(s)\|_2^2
                         +\|\delta_\eta T(s)\|_2^2\bigr)&+\int_0^t
       \bigl(\|\nabla\delta_\eta v\|_2^2
                         +\|\nabla_H\delta_\eta T\|_2^2\bigr)\,ds\\
 &\leq 2C_0^2\exp\!\left(C_0\eta+\int_0^tG(s)\,ds\right).
\end{aligned}
\]
Since $v,T\in H^1(0,t_0+\eta_0;L^2)$, their time

By the strong convergence of Steklov averages

\quad\text{strongly in }L^2(0,t_0;L^2)
\quad\text{as }\eta\downarrow0.

\end{aligned}
\]
The right-hand side defines a continuous nondecreasing $M_4$ on
$[0,\mathcal T^*)$.

 \end{aligned}

$\varepsilon=1/4$, and the initial energy is bounded by
$1+\|\partial_zT_0\|_2^2+\|\partial_{zz}T_0\|_2^2$.

$(\partial_zT)_N$ and $\partial_z(\partial_zT)_N$ in
$L^\infty_tL^2$, and for their horizontal gradients in $L^2_tL^2$.
Equation~\eqref{eq:flat-vertical-regularized}, written in divergence
form, also bounds $\partial_t(\partial_zT)_N$ in $L^2(0,t_0;V_T')$.
Here $V_T'$ is the dual of the scalar temperature space $V_T$,
with pivot $L^2(\Omega)$.
Indeed, the projection is bounded on that space,
$v,w,\partial_zw\in L^2_tL^\infty$, and
$\partial_zv\cdot\nabla_HT\in L^2_tL^2$ by
$T\in L^\infty_tH^1$ and \eqref{eq:flat-vertical-coefficients}.

\[

       \quad\hbox{strongly in }L^2_tL^2
       \quad\hbox{and in }C([0,t_0];L^2\hbox{-weak}).

\qquad
P(x,y,z,t)=\int_{-h}^{z}d(x,y,\xi,t)\,d\xi.

\[

\]
with zero initial data in the weak $L^2$ sense. The preceding

\[

       \cap L^2(0,t_0;H^1(\Omega)),
\qquad
\partial_zP=d\in L^\infty(0,t_0;L^2(\Omega)),
\]
and $P$ has zero lateral boundary condition. Let $\chi_a$ be a vertical cutoff of width

\[

\]

\[

\right|
\leq C_\varphi a^{1/2}
\|\partial_zw\|_{L^1_tL^\infty}
\|d\|_{L^\infty_tL^2}
\longrightarrow0.

\[
P_t-\Delta_HP+v\cdot\nabla_HP+w\partial_zP=\mathcal K_v P,
\qquad
\mathcal K_v P(z)
:=\int_{-h}^{z}
\partial_zv(\xi)\cdot\nabla_HP(\xi)\,d\xi.

&\leq C\left(

&\leq C\|v\|_{H^{11/4}}\|P\|_2.
\end{aligned}

\label{lemma:Gagliardo}
\noindent\textup{(i) Gagliardo--Nirenberg interpolation inequality.}
(see \cite{BREZIS20181355})Let $\Omega \subset \mathbb{R}^n$ be either the whole space, a half-space or a bounded Lipschitz domain. Let $1 \leq p, p_1, p_2 \leq +\infty$ be three positive extended real quantities and let $s, s_1, s_2$ be non-negative real numbers. Furthermore, let $\theta \in (0, 1)$ and assume that
\[
s_1 \leq s_2, \qquad s = \theta s_1 + (1 - \theta)s_2, \qquad \frac{1}{p} = \frac{\theta}{p_1} + \frac{1 - \theta}{p_2}
\]
hold. Then,
\[
\|u\|_{W^{s,p}(\Omega)} \leq C \|u\|_{W^{s_1,p_1}(\Omega)}^{\theta} \|u\|_{W^{s_2,p_2}(\Omega)}^{1-\theta}
\]
for any $u \in W^{s_1,p_1}(\Omega) \cap W^{s_2,p_2}(\Omega)$ if and only if
\[
\text{at least one of }
\begin{cases}
s_2 \in \mathbb{N} \text{ and } s_2 \geq 1, \\
p_2 = 1, \\
0 < s_2 - s_1 \leq 1 - \frac{1}{p_1}
\end{cases}
\text{ is false.}
\]
The constant $C > 0$ depends on the parameters $p, p_1, p_2, s, s_1, s_2, \theta$, on the domain $\Omega$, but not on $u$.

\medskip
\noindent\textup{(ii) Complex interpolation on bounded domains.}
Let $D\subset\mathbb R^n$ be a bounded Lipschitz domain. Let
$s_0,s_1\geq0$ with $s_0\neq s_1$, $1<p_0,p_1<\infty$, and
$0<\theta<1$, and set
\[
 s=(1-\theta)s_0+\theta s_1,\qquad
 \frac1p=\frac{1-\theta}{p_0}+\frac{\theta}{p_1}.
\]
Then
\begin{equation}
 [H_{p_0}^{s_0}(D),H_{p_1}^{s_1}(D)]_\theta=H_p^s(D)
 \label{eq:bounded-domain-complex-interpolation}
\end{equation}
with equivalent norms. Here $[\cdot,\cdot]_\theta$ denotes complex
interpolation, and $H_p^s(D)$ is the Bessel potential space defined
by restriction from $H_p^s(\mathbb R^n)$. These are the full spaces,
with no boundary trace conditions imposed.

In particular, if $m_0,m_1\in\mathbb N_0$, $m_0<m_1$, and
$(1-\theta)/p_0+\theta/p_1=1/2$, then
\begin{equation}
 [W^{m_0,p_0}(D),W^{m_1,p_1}(D)]_\theta
 =H^{(1-\theta)m_0+\theta m_1}(D)
 \label{eq:bounded-domain-integer-interpolation}
\end{equation}
with equivalent norms. We use here the integer-order identification
$H_p^m(D)=W^{m,p}(D)$ for $m\in\mathbb N_0$ and $1<p<\infty$,
and the identity $H_2^s(D)=H^s(D)$.

 =\{(x,y,z):(x,y)\in M,\ -h(x,y)<z<h(x,y)\}.
\]
By \eqref{eq:top-stokes-orthogonal}, its upper and lower faces meet the sidewall at right angles. The boundary has $C^3$ faces and edges, with no vertices, and the Dirichlet edge exponent is $2$.For background on the dependence of scalar elliptic corner regularity on the opening angles and boundary conditions, see \cite[Chapters~4--5]{Grisvard2011}.

\[
 \begin{cases}
 -\Delta(S\phi)=\phi&\text{in }\widetilde\Omega_h,\\

 \end{cases}

see \cite[Chapter~IV]{Galdi11} and \cite{BerghLofstrom76}.

$-\Delta v=f-\nabla_Hp_s$ and
\eqref{eq:top-stokes-scalar-shift} with $a=1/2$ give
\[
 \|v\|_{H^{5/2}(\Omega)}
 \leq C_h\left(
       \|f\|_{H^{1/2}(\Omega)}
       +\|\nabla_Hp_s\|_{H^{1/2}(M)}
       +\|v\|_{H^1(\Omega)}\right)
 \leq C_h\|f\|_{H^1(\Omega)}.
\]
The trace theorem now yields
\[
 \|\partial_zv|_{z=-h}\|_{H^1(M)}
 \leq C_h\|f\|_{H^1(\Omega)}.
\]
Applying the planar Stokes estimate once more to
\eqref{eq:barv}--\eqref{eq:barv-div}, with the same homogeneous
Dirichlet condition, gives
\[
 \|\nabla_Hp_s\|_{H^1(M)}
 \leq C_h\left(
       \|\bar f\|_{H^1(M)}
       +\|\partial_zv|_{z=-h}\|_{H^1(M)}\right)
 \leq C_h\|f\|_{H^1(\Omega)}.
\]
Finally, for every $0<\varepsilon<1$, apply
\eqref{eq:top-stokes-scalar-shift} with $a=1-\varepsilon$ to
$-\Delta v=f-\nabla_Hp_s$. Together with
$\|\partial_zv\|_{H^{2-\varepsilon}(\Omega)}
\leq C\|v\|_{H^{3-\varepsilon}(\Omega)}$, this proves
\[
 \|v\|_{H^{3-\varepsilon}(\Omega)}
 +\|\partial_zv\|_{H^{2-\varepsilon}(\Omega)}
 +\|\nabla_Hp_s\|_{H^{1-\varepsilon}(M)}
 \leq C_{h,\varepsilon}\|f\|_{H^1(\Omega)}.
\]


\emph{Variable depth.}
Fix $0<\varepsilon<1/2$ and use the normalized average $M_h$
defined in Subsection~\ref{subsec:topographic-notation}.
Tangential differentiation of the bottom condition gives
\[
 (\partial_i v)_b=(\partial_i h)(\partial_zv)_b,
 \qquad i=1,2,
\]
where the subscript $b$ denotes restriction to $z=-h(x,y)$.
Consequently, Leibniz' rule and the top condition $\partial_zv=0$
yield
\[
 \int_{-h(x,y)}^0\Delta v\,dz
 =\Delta_H(hM_hv)-(1+|\nabla_Hh|^2)(\partial_zv)_b.
\]
Integration of \eqref{eq:top-stokes-system} therefore gives
\begin{equation}
 \begin{split}
 -\Delta_H(hM_hv)+\nabla_H(hp_s)
 ={}&hM_hf-(1+|\nabla_Hh|^2)(\partial_zv)_b
       +p_s\nabla_Hh,\\
 \nabla_H\cdot(hM_hv)={}&0,
 \qquad (hM_hv)|_{\partial M}=0.
 \end{split}
 \label{eq:top-stokes-higher-average}
\end{equation}
The bottom derivative is retained; no homogeneous condition on
$(\partial_zv)_b$ is used.
Lemma~\ref{lem:top-stokes-H2}, the trace theorem, and the boundedness
of vertical averaging give
\[
 \|(\partial_zv)_b\|_{H^{1/2}(M)}
 +\|hM_hf\|_{H^1(M)}+\|p_s\nabla_Hh\|_{H^1(M)}
 \leq C_h\|f\|_{H^1(\Omega_h)}.
\]
Apply the planar Dirichlet Stokes estimate to
\eqref{eq:top-stokes-higher-average}, with velocity $hM_hv$ and
pressure $hp_s$. Using
$\nabla_Hp_s=h^{-1}[\nabla_H(hp_s)-p_s\nabla_Hh]$, we obtain
\[
 \|\nabla_Hp_s\|_{H^{1/2-\varepsilon}(M)}
 \leq C_{h,\varepsilon}\|f\|_{H^1(\Omega_h)};
\]
see \cite[Chapter~IV]{Galdi11} and \cite{BerghLofstrom76}.
Apply \eqref{eq:top-stokes-scalar-shift} with
$a=1/2-\varepsilon$ to $-\Delta v=f-\nabla_Hp_s$ and then take
the bottom trace. This gives
\[
 \|v\|_{H^{5/2-\varepsilon}(\Omega_h)}
 +\|(\partial_zv)_b\|_{H^{1-\varepsilon}(M)}
 \leq C_{h,\varepsilon}\|f\|_{H^1(\Omega_h)}.
\]
A second application of the planar Stokes estimate to
\eqref{eq:top-stokes-higher-average} gives the same bound for
$\|\nabla_Hp_s\|_{H^{1-\varepsilon}(M)}$. Finally,
\eqref{eq:top-stokes-scalar-shift} with $a=1-\varepsilon$ yields
\eqref{eq:top-stokes-higher}. The cases $1/2\leq\varepsilon<1$
follow by Sobolev embedding from any smaller loss, for example
$\varepsilon=1/4$.

\label{lem:sqrt-gronwall}
Let $\tau>0$, let $Y:[0,\tau]\to[0,\infty)$ be absolutely continuous, and let $a,b\in L^1(0,\tau)$ be nonnegative. Suppose that
\[
Y'(t)\leq a(t)Y(t)+2b(t)\sqrt{Y(t)}
\qquad\text{for a.e. }t\in(0,\tau).
\]
Then, for every $t\in[0,\tau]$,
\[
\sqrt{Y(t)}
\leq
\sqrt{Y(0)}
\exp\left(\frac12\int_0^t a(r)\,dr\right)
+
\int_0^t b(s)
\exp\left(\frac12\int_s^t a(r)\,dr\right)\,ds.
\]
In particular, if $Y(0)=0$, $a(t)\equiv a_0\geq0$,
and $b(t)\equiv b_0\geq0$, then
\[
\sqrt{Y(t)}\leq b_0t\,e^{a_0t/2},
\qquad
Y(t)\leq b_0^2t^2e^{a_0t},
\qquad 0\leq t\leq\tau.
\]
<!-- end lemma-8 -->
<!-- proof-23: body_md -->
For $\varepsilon>0$, set
\[
Z_\varepsilon(t):=\sqrt{Y(t)+\varepsilon}.
\]
Then $Z_\varepsilon$ is absolutely continuous and strictly
positive. The chain rule gives, for almost every $t$,
\[
\begin{aligned}
Z_\varepsilon'(t)
&=\frac{Y'(t)}{2\sqrt{Y(t)+\varepsilon}}\\
&\leq
\frac{a(t)}{2}
\frac{Y(t)}{\sqrt{Y(t)+\varepsilon}}
+
b(t)\frac{\sqrt{Y(t)}}{\sqrt{Y(t)+\varepsilon}}\\
&\leq \frac{a(t)}{2}Z_\varepsilon(t)+b(t).
\end{aligned}
\]
Multiplying by the integrating factor
$\exp(-\frac12\int_0^t a(r)\,dr)$ yields
\[
\frac{d}{dt}
\left[
\exp\left(-\frac12\int_0^t a(r)\,dr\right)
Z_\varepsilon(t)
\right]
\leq
b(t)\exp\left(-\frac12\int_0^t a(r)\,dr\right).
\]
Integrating from $0$ to $t$, and using
$Z_\varepsilon(0)=\sqrt{Y(0)+\varepsilon}$, we obtain
\[
Z_\varepsilon(t)
\leq
\sqrt{Y(0)+\varepsilon}
\exp\left(\frac12\int_0^t a(r)\,dr\right)
+
\int_0^t b(s)
\exp\left(\frac12\int_s^t a(r)\,dr\right)\,ds.
\]
Letting $\varepsilon\downarrow0$ proves the first assertion.
For constant coefficients and $Y(0)=0$, it follows that
\[
\sqrt{Y(t)}
\leq b_0\int_0^t e^{a_0(t-s)/2}\,ds
\leq b_0t\,e^{a_0t/2}.
\]
Squaring gives the remaining estimate.

\label{lem:anisotropic-isotropic}
Let $\Omega=M\times(-h,0)$, where $M\subset\mathbb{R}^2$ is a bounded
$C^3$ domain. Let $0<s\leq2$. Suppose that
\[
    f\in L^2((-h,0);H^s(M))
\]
and
\[
    f,\ \partial_z f,\ \partial_z^2f\in L^2(\Omega).
\]
Then $f\in H^s(\Omega)$ and
\begin{equation}
\label{eq:anisotropic-isotropic}
    \|f\|_{H^s(\Omega)}
    \le C
    \left(
        \|f\|_{L^2_zH^s_H}
        +
        \|f\|_{H^2_zL^2_H}
    \right),
\end{equation}
where
\[
    \|f\|_{H^2_zL^2_H}^2
    :=
    \|f\|_{L^2(\Omega)}^2
    +
    \|\partial_zf\|_{L^2(\Omega)}^2
    +
    \|\partial_z^2f\|_{L^2(\Omega)}^2.
\]
<!-- end lemma-9 -->
<!-- proof-24: body_md -->
Since $M$ is smooth and bounded and $(-h,0)$ is a bounded interval,
the standard Sobolev extension theorem, applied in the horizontal
and vertical variables, gives an extension $\widetilde f$ of $f$ to
$\mathbb{R}^3$ such that
\[
    \|\widetilde f\|_{L^2(\mathbb{R};H^s(\mathbb{R}^2))}
    \le
    C\|f\|_{L^2((-h,0);H^s(M))}
\]
and
\[
    \|\widetilde f\|_{H^2(\mathbb{R};L^2(\mathbb{R}^2))}
    \le
    C\|f\|_{H^2_zL^2_H}.
\]
We may therefore apply the Fourier transform to $\widetilde f$.
Writing $\xi=(\xi_H,\xi_3)$, where
$\xi_H=(\xi_1,\xi_2)$, and using $0<s\leq2$, we have
\[
    (1+|\xi_H|^2+|\xi_3|^2)^s
    \le
    C_s\left[
        (1+|\xi_H|^2)^s
        +(1+|\xi_3|^2)^2
    \right].
\]
Hence, by Plancherel's theorem,
\begin{align*}
    \|\widetilde f\|_{H^s(\mathbb{R}^3)}^2
    &\le
    C_s
    \int_{\mathbb{R}^3}
    (1+|\xi_H|^2)^s
    |\widehat{\widetilde f}(\xi)|^2\,d\xi
    \\
    &\quad
    +C_s
    \int_{\mathbb{R}^3}
    (1+|\xi_3|^2)^2
    |\widehat{\widetilde f}(\xi)|^2\,d\xi
    \\
    &\le
    C_s
    \left(
        \|\widetilde f\|_{L^2_zH^s_H}^2
        +
        \|\widetilde f\|_{H^2_zL^2_H}^2
    \right).
\end{align*}
Restricting $\widetilde f$ back to $\Omega$ gives
\eqref{eq:anisotropic-isotropic}.

\label{lem:weighted-vertical-temperature}
Let $Q_\ell=M\times(-\ell,0)$, where $\ell>0$ and $M$ is a bounded
$C^3$ domain, and let $\rho\in C^3(\overline M)$ be positive. Write
\[
 \|g\|_{2,\rho}^2=\int_{Q_\ell}\rho|g|^2,
 \qquad A_\rho=-\rho^{-1}\nabla_H\cdot(\rho\nabla_H),
\]
with horizontal Dirichlet form domain
$L^2((-\ell,0);H_0^1(M))$. Suppose that
\[
 v\in L^\infty(0,t_0;H^2(Q_\ell)^2)
       \cap L^2(0,t_0;H^{11/4}(Q_\ell)^2),
 \qquad v|_{\partial M\times(-\ell,0)}=0,
\]
and that
\[
 \omega(\sigma)=-\rho^{-1}\int_{-\ell}^{\sigma}
                       \nabla_H\cdot(\rho v)(s)\,ds,
 \qquad \omega|_{\sigma=0}=0.
\]
Let $\theta$ solve
\[
 \theta_t+A_\rho\theta+v\cdot\nabla_H\theta
                   +\omega\partial_\sigma\theta=0
\]
in distributions, with zero lateral trace, initial value
$\theta_0\in H^2(Q_\ell)$ with zero lateral trace, and
\[
 \theta\in L^\infty(0,t_0;H^1),\qquad
 \nabla_H\theta\in L^2(0,t_0;H^1),\qquad
 \theta_t\in L^2(0,t_0;L^2).
\]
No temperature condition is imposed at $\sigma=-\ell,0$.
Then
\[
 \partial_{\sigma\sigma}\theta\in L^\infty(0,t_0;L^2),
 \qquad
 \nabla_H\partial_{\sigma\sigma}\theta\in L^2(0,t_0;L^2).
\]
More precisely, for $0\leq t\leq t_0$,
\begin{equation}
 \begin{split}
 &\sup_{0\leq s\leq t}
       \|\partial_{\sigma\sigma}\theta(s)\|_{2,\rho}^2
       +\int_0^t\|\nabla_H\partial_{\sigma\sigma}\theta\|_{2,\rho}^2\,ds\\
 &\qquad\leq C\bigl(1+\|\partial_{\sigma\sigma}\theta_0\|_{2,\rho}^2\bigr)
        \exp\!\Bigg(C\int_0^t\|v(s)\|_{H^{11/4}}\\
 &\hspace{16mm}\cdot\bigl(1+\|\theta\|_{L^\infty(0,t_0;H^1)}
             +\|\nabla_H\theta(s)\|_{H^1}\bigr)\,ds\Bigg),
 \end{split}
 \label{eq:weighted-vertical-temperature-bound}
\end{equation}

<!-- proposition-17: body_md -->
\label{prop:appendix-basic-energy}
For every $0<t<\mathcal T^*$,
\[
 \sup_{0\leq s\leq t}\|v(s)\|_2^2
 +\int_0^t\|\nabla v(s)\|_2^2\,ds
 \leq C\left(\|v_0\|_2^2+\int_0^t\|T(s)\|_2^2\,ds\right)
 \leq C\left(\|v_0\|_2^2+t\|T_0\|_2^2\right),
\]
where $C$ depends only on $M$ and $h$. Moreover, whenever
$T_0\in L^q(\Omega)$, $2\leq q\leq\infty$,
\[
 \sup_{0\leq s\leq t}\|T(s)\|_q\leq\|T_0\|_q.
\]
<!-- end proposition-17 -->
<!-- proof-26: body_md -->
Set $Q_T(x,y,z)=\int_{-h}^zT(x,y,\xi)\,d\xi$.
Multiply the momentum equation with $v$, 
\[
 \int_\Omega\nabla_Hp_s\cdot v
 =-\int_Mp_s\nabla_H\cdot\left(\int_{-h}^0v\,dz\right)=0.
\]
Integration by parts, yields
\[
 \begin{aligned}
 \frac12\frac{d}{dt}\|v\|_2^2+\|\nabla v\|_2^2
 &=\int_\Omega\nabla_H Q_T\cdot v
 =-\int_\Omega Q_T\,\nabla_H\cdot v\\
 &\leq C_h\|T\|_2\|\nabla_Hv\|_2
 \leq\frac12\|\nabla_Hv\|_2^2+C_h\|T\|_2^2.
 \end{aligned}
\]
Consequently,
\begin{equation}
 \frac{d}{dt}\|v\|_2^2+\|\nabla_Hv\|_2^2
       +2\|\partial_zv\|_2^2\leq C_h\|T\|_2^2.
\end{equation}
For $2\leq q<\infty$, testing the temperature equation with
$|T|^{q-2}T$ gives
\begin{equation}
 \frac1q\frac{d}{dt}\|T\|_q^q
 +(q-1)\int_\Omega |T|^{q-2}|\nabla_HT|^2=0.
\end{equation}
Thus the $L^q$ norm is nonincreasing. For bounded $T_0$, letting
$q\to\infty$ on the bounded domain gives
\begin{equation}
 \sup_{0\leq s\leq t}\|T(s)\|_\infty\leq\|T_0\|_\infty.
\end{equation}
In particular,
\[
 \|T(t)\|_2^2+2\int_0^t\|\nabla_HT(s)\|_2^2\,ds
 \leq\|T_0\|_2^2.
\]
Integrating the velocity inequality and taking the time supremum
proves the first assertion. The identities are justified by the
energy approximations used in the construction.
