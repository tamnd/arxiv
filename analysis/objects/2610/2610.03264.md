<!-- assumption-1: body_md -->
\label{ass:regularity} For every $T>0$: 
\begin{enumerate}
\par\hangindent 40\p@ $f,\sigma_{i},\frac{\partial f}{\partial x},\frac{\partial\sigma_{i}}{\partial x}$
are jointly measurable in $(t,x)$, and $f(t,\cdot),\sigma_{i}(t,\cdot)$
are $C^{1}$ for almost every $t$; 
\par\hangindent 40\p@ 
\begin{equation}
\int^{T}_{0}\left(|f(t,0)|^{2}+\sum^{m}_{i=1}|\sigma_{i}(t,0)|^{2}\right)dt<\infty;
\end{equation}
\par\hangindent 40\p@ there exists $L_{T}>0$ such that 
\begin{equation}
\sup_{\substack{t\in[0,T]\\
x\in\mathbb{R}^{n}
}
}\left(\left\Vert \frac{\partial f(t,x)}{\partial x}\right\Vert +\sum^{m}_{i=1}\left\Vert \frac{\partial\sigma_{i}(t,x)}{\partial x}\right\Vert \right)\le L_{T};\label{eq:derivative-bound}
\end{equation}
\par\hangindent 40\p@ $\frac{\partial f}{\partial x}$ and $\frac{\partial\sigma_{i}}{\partial x}$
are uniformly continuous in $x$, uniformly for $t\in[0,T]$. 
\end{enumerate}

<!-- defn-1: body_md -->
The solution $X^{x}_{t}$ of the SDE \eqref{eq:ItoSDE} is \emph{mean-square
differentiable} with respect to $x$ in the direction $y\in\mathbb{R}^{n}$
if there exists $Y^{x,y}_{t}\in L^{2}(\Omega;\mathbb{R}^{n})$ such that 
\begin{equation}
\lim_{h\to0}\mathbb{E}\left|\frac{X^{x+hy}_{t}-X^{x}_{t}}{h}-Y^{x,y}_{t}\right|^{2}=0.\label{eq:mean square der}
\end{equation}
<!-- end defn-1 -->
<!-- prop-1: body_md -->
\label{prop:msdiff} Under Assumption~\ref{ass:regularity}, for
every $T>0$ and $x,y\in\mathbb{R}^{n}$ there exists an adapted process $Y^{x,y}$
such that 
\begin{equation}
\mathbb{E}\sup_{0\le t\le T}|Y^{x,y}_{t}|^{2}<\infty
\end{equation}
and 
\begin{equation}
\lim_{h\to0}\mathbb{E}\sup_{0\le t\le T}\left|\frac{X^{x+hy}_{t}-X^{x}_{t}}{h}-Y^{x,y}_{t}\right|^{2}=0.\label{eq:msdiff}
\end{equation}
Moreover, $Y^{x,y}$ is the unique solution of \eqref{eq:Ito_SDE_var}
with initial condition $Y^{x,y}_{0}=y$. The map $(x,y)\mapsto Y^{x,y}$
is continuous from $\mathbb{R}^{n}\times\mathbb{R}^{n}$ into $L_{2}(\Omega;C([0,T];\mathbb{R}^{n}))$.
<!-- end prop-1 -->
<!-- lem-1: body_md -->
\label{lem:FTC} Let $\gamma\in C^{1}([0,1];\mathbb{R}^{n})$. Under Assumption~\ref{ass:regularity},
\begin{equation}
X^{\gamma(1)}_{t}-X^{\gamma(0)}_{t}=\int^{1}_{0}Y^{\gamma(s),\gamma'(s)}_{t}\,ds\label{eq:L2FTC}
\end{equation}
holds in $L^{2}(\Omega;\mathbb{R}^{n})$. Consequently, 
\begin{equation}
\mathbb{E}|X^{\gamma(1)}_{t}-X^{\gamma(0)}_{t}|^{2}\le\int^{1}_{0}\mathbb{E}|Y^{\gamma(s),\gamma'(s)}_{t}|^{2}\,ds.\label{eq:L2FTC-Jensen}
\end{equation}
<!-- end lem-1 -->
<!-- proof-1: body_md -->
Proposition~\ref{prop:msdiff} implies that $x\mapsto X^{x}_{t}$
is continuously Fréchet differentiable as a map from $\mathbb{R}^{n}$ into
$L^{2}(\Omega;\mathbb{R}^{n})$, with directional derivative $Y^{x,y}_{t}$.
Hence $s\mapsto X^{\gamma(s)}_{t}$ is continuously differentiable
and 
\begin{equation}
\frac{d}{ds}X^{\gamma(s)}_{t}=Y^{\gamma(s),\gamma'(s)}_{t}\quad\text{in }L^{2}.
\end{equation}
The Banach-space fundamental theorem of calculus (see e.g., \cite[Chapter XIII]{lang1993real})
gives \eqref{eq:L2FTC}; Jensen's inequality gives \eqref{eq:L2FTC-Jensen}.

<!-- defn-2: body_md -->
The system is said to be Wasserstein contracting (or $W_{p}$ contracting
in short) if there exist two positive constants $K$, $\lambda$ such
that, for any $X_{0}\sim\mu_{0}$, $Y_{0}\sim\nu_{0}$, and $X_{t}\sim\mu_{t}$,
$Y_{t}\sim\nu_{t}$, there holds
\begin{equation}
W_{p}(\mu_{t},\nu_{t})\le Ke^{-\lambda t}W_{p}(\mu_{0},\nu_{0}),\quad t\ge0.\label{eq:w2-goal}
\end{equation}

<!-- thm-1: body_md -->
\label{thm:main} Suppose Assumption~\ref{ass:regularity} holds
and let $1\le p\le2$. Assume that there exist a function $V\in C([0,\infty)\times\mathbb{R}^{n}\times\mathbb{R}^{n})\cap C^{1,2,2}([0,\infty)\times\mathbb{R}^{n}\times(\mathbb{R}^{n}\backslash\{0\}))$
and constants $c_{1},c_{2},\lambda>0$ such that 
\begin{equation}
c_{1}|y|^{p}\le V(t,x,y)\le c_{2}|y|^{p}\label{eq:V-bounds}
\end{equation}
and 
\begin{equation}
\mathcal{A}V(t,x,y)\le-\lambda V(t,x,y)\label{eq:V-decay}
\end{equation}
for all $t\ge0$ and $x\in\mathbb{R}^{n}$, $y\in\mathbb{R}^{n}\backslash\{0\}$.
Then the system \eqref{eq:ItoSDE} is $W_{p}$ contracting.
<!-- end thm-1 -->
<!-- proof-2: body_md -->
The proof relies essentially on Dynkin's formula and Lemma \ref{lem:FTC}.

\emph{Step 1.} We treat the $p=2$ case first. If $V$ is $C^{2}$
on the whole of $[0,\infty)\times\mathbb{R}^{n}\times\mathbb{R}^{n}$,
then Dynkin's formula (see for example \cite[Chapter II]{Kushner1967})
together with the inquality \eqref{eq:V-decay} imply
\begin{equation}
\mathbb{E}V(t,X_{t},Y_{t})\le e^{-\lambda t}\mathbb{E}V(0,X_{0},Y_{0}).\label{eq:ineq-V}
\end{equation}
However, in our case, $V(\cdot,\cdot,y)$ is only continuous at $y=0$.
This can be easily handled through the following locolization procedure. 

Let $Y_{0}\neq0$. The variational equation (second line of \eqref{eq:lifted})
with initial condition $Y_{0}$ has the representation $Y_{t}=\Phi_{t}Y_{0}$,
where $\Phi_{t}$ is the its fundamental matrix which is almost surely
invertible. Consequently, $Y_{t}\neq0$ for all $t\ge0$ almost surely.

For $k\ge1$, define the stopping time 
\[
\tau_{k}:=\inf\left\{ t\ge0:|Y_{t}|\le k^{-1}\ \text{or}\ |X_{t}|+|Y_{t}|\ge k\right\} .
\]
On $[0,\tau_{k}]$, the process remains in a compact subset of $\mathbb{R}^{n}\times(\mathbb{R}^{n}\setminus\{0\})$,
where $V$ is $C^{1,2,2}$. Dynkin's formula applied to $e^{\lambda(t\wedge\tau_{k})}V(t\wedge\tau_{k},X_{t\wedge\tau_{k}},Y_{t\wedge\tau_{k}})$
therefore gives 
\[
\mathbb{E}\!\left[e^{\lambda(t\wedge\tau_{k})}V(t\wedge\tau_{k},X_{t\wedge\tau_{k}},Y_{t\wedge\tau_{k}})\right]\le V(0,x,y).
\]
Since $\tau_{k}\uparrow\infty$ almost surely as $k\to\infty$, Fatou's
lemma yields \eqref{eq:ineq-V}. 

For $Y_{0}=0$, the same estimate is trivial since $Y^{x,0}_{t}\equiv0$.
Using \eqref{eq:V-bounds}, we then arrive at
\begin{equation}
\mathbb{E}|Y^{x,y}_{t}|^{2}\le\frac{c_{2}}{c_{1}}e^{-\lambda t}|y|^{2}\label{eq:tangent-contraction}
\end{equation}
for every $t\ge0$, $x\in\mathbb{R}^{n}$, $y\in\mathbb{R}^{n}$.

Now for any two deterministic initial points $x_{1},x_{2}$, take
the segment $\gamma(s)=(1-s)x_{1}+sx_{2}$. Lemma~\ref{lem:FTC}
gives 
\begin{equation}
\mathbb{E}|X^{x_{2}}_{t}-X^{x_{1}}_{t}|^{2}\le\frac{c_{2}}{c_{1}}e^{-2\lambda t}|x_{2}-x_{1}|^{2}.\label{eq:point-contraction}
\end{equation}

\emph{Step 2}. Choose an optimal initial coupling $\pi_{0}\in\Pi_{\mathrm{opt}}(\mu_{0},\nu_{0})$,
sample $(X_{0},\widetilde{X}_{0})\sim\pi_{0}$ independently of $B$,
and evolve the two states with the same Brownian motion. Conditioning
on $(X_{0},\widetilde{X}_{0})$ and using \eqref{eq:point-contraction},
\begin{align}
\mathbb{E}|X_{t}-\widetilde{X}_{t}|^{2} & \le\frac{c_{2}}{c_{1}}e^{-2\lambda t}\mathbb{E}|X_{0}-\widetilde{X}_{0}|^{2}\nonumber \\
 & =\frac{c_{2}}{c_{1}}e^{-2\lambda t}W^{2}_{2}(\mu_{0},\nu_{0}).
\end{align}
The law of $(X_{t},\widetilde{X}_{t})$ is an admissible coupling
of $\mu_{t}$ and $\nu_{t}$, so $W^{2}_{2}(\mu_{t},\nu_{t})\le\mathbb{E}|X_{t}-\widetilde{X}_{t}|^{2}$.
Taking square roots proves \eqref{eq:w2-goal}. 

\emph{Step 3}. It remains consider $1\le p<2$. Since the underlying
measure is a probability measure, $L_{2}\subset L_{p}$ and $\|f\|_{p}\le\|f\|_{2}$.
Hence mean-square differentiability implies $L_{p}$-differentiability
for every $1\le p<2$, and the proof for $p=2$ carries over verbatim.

\label{lem:mani-esti1} Let $(x,v)\in TM$ and $\gamma:(-\epsilon,\epsilon)\to M$

variational equation of the coordinate SDE. Proposition \ref{prop:msdiff}

piecewise $C^{1}$ curve in $M$. Let $Z^{\gamma(s),\gamma'(s)}_{t}$

<!-- proof-7: body_md -->
It is enough to consider $\gamma\in C^{1}$. Define 
\[

By Minkowski's inequality, $\rho$ is a metric on random variables

 & \quad\le\rho\left(X^{\gamma(s+h)}_{t},\exp_{X^{\gamma(s)}_{t}}\left(hY^{\gamma(s),\gamma'(s)}_{t}\right)\right)\\

\end{align*}
The first term is $o(|h|)$. For the second term, the radial geodesic

\end{equation}
where $Z_{\varepsilon}$ is the normalization constant. 

(iii) As $\epsilon\to0$,
\begin{equation}
W_{2}(\pi_{\varepsilon},\delta_{x_{\star}})=O(\sqrt{\varepsilon}).\label{eq:pi-concentration}
\end{equation}
Consequently, the invariant distribution can be made arbitrarily concentrated
around $x_{\star}$ while preserving a strictly positive global contraction
rate.

<!-- rem-7: body_md -->
Proposition~\ref{prop:noise-assisted-sphere} separates convergence
and accuracy in a useful way. The parameter $k$ determines the time
scale of convergence, while $\varepsilon$ determines the amount of
noise remaining near the target. In particular, 
\[
\lim_{\varepsilon\to0}\lambda_{\varepsilon}=\frac{k}{4}>0,\quad\lim_{\varepsilon\to0}W_{2}(\pi_{\varepsilon},\delta_{x_{\star}})=0.
\]
Thus the invariant distribution can be made arbitrarily concentrated
around the target without causing the guaranteed global contraction
rate to vanish.
