{\rm
\begin{enumerate}
\item[{\rm(i)}]
Let $\calt$ denote the set of planar binary reduced rooted trees together with the trivial tree $\vcenter{\xymatrix@M=4pt@R=8pt@C=4pt{\ar@{-}[d]\\ \\}}$. If $t\in \calt$ has $n$ leaves, we call $t$ an {\bf $n$-tree}. The trivial tree $\vcenter{\xymatrix@M=4pt@R=8pt@C=4pt{\ar@{-}[d]\\ \\}}$ has one leaf.
\item[{\rm(ii)}]
 Let $\vertset$ be a set. By a {\bf decorated tree} we mean a tree $t$ of $\calt$ together with a decoration on the vertices of $t$ by elements of $\vertset$ and a decoration on the leaves of $t$ by distinct positive integers. Let $t(\vertset)$ denote the set of decorated trees for $t$ and denote by
\begin{equation*}
\calt(\vertset):=\coprod_{ t \in \calt} t(\vertset).
\end{equation*}
If $\tau\in t(\vertset)$ for an $n$-tree $t$, we call $\tau$ a {\bf labeled $n$-tree}.
\item[{\rm(iii)}]
For $\tau\in \calt(\vertset)$, we let $\vin(\tau)$ (resp. $\lin(\tau)$) denote the set (resp. ordered set) of labels of the vertices (resp. leaves) of $\tau$.
\item[{\rm(iv)}]
Let $\tau_{l},\tau_{r}\in \calt(\vertset)$ with disjoint sets of leaf labels and $\omega\in \vertset$. The {\bf grafting of $\tau_{l}$ and $\tau_{r}$ along} $\omega$ is denoted by $\tau=
  \tau_{l} \vee_{\omega} \tau_{r}$, which is an element in $\calt(\vertset)$.
\item[{\rm(v)}]
 For $\tau\in \calt(\vertset)$ with $|\lin(\tau)|>1$, we let $\tau= \tau_{l} \vee_{\omega} \tau_{r}$ denote the unique decomposition of $\tau$ as a grafting of $\tau_{l}$ and $\tau_{r}$
  in $\calt(\vertset)$ along $\omega \in \vertset$.
\end{enumerate}}

\label{def:anti-bisu}{\rm
Let $\gensp$ be a vector space with a basis $\genbas$.
\begin{enumerate}
\item[{\rm(i)}]
Define a vector space
\begin{equation}\label{genbi}
\genbi= \gensp \otimes (\mathbf{k} \prec \oplus~ \mathbf{k} \succ),
\end{equation}
where we denote $(\omega~\otimes \prec)$ (resp. $(\omega~\otimes \succ)$) by
$\begin{pmatrix}
  \omega \\
  \prec
  \end{pmatrix}$
(resp. $\begin{pmatrix}
  \omega \\
  \succ
  \end{pmatrix}$)

\begin{enumerate}
\item[{\rm(a)}] An {\bf $\mathbb{S}$-module} over $\mathbf{k}$ is a family
  $$ M:=\{ M(n)\}_{n \geq 0}=\{M(0),M(1),\dots,M(n),\dots\} $$
  of right $\mathbf{k} \mathbb{S}_{n}$-modules $M(n)$, where $\mathbb{S}_{n}$ is the symmetric group for all $n \geq 0$. If every $M(n)$ is the trivial $\mathbf{k} \mathbb{S}_{n}$-module, then $M$ is called an {\bf $\mathbb{N}$-module}.
\item[{\rm(b)}] A nonsymmetric operad is an $\mathbb{N}$-module $\opd=\{ \opd(n)\}_{n \geq 0}$ equipped with {\bf partial compositions}:
    $$ \circ_{i}:=\circ_{m,n,i}: \opd(m) \otimes \opd(n) \to \opd(m+n-1),\quad 1\leq i\leq m$$
    such that the following equalities hold for all $\eta\in \opd(l),\mu \in \opd(m)$ and $\nu\in \opd(n)$:
    \begin{enumerate}
  \item[{\rm(i)}] $(\eta \circ_i \mu) \circ_{i-1+j} \nu= \eta \circ_i (\mu \circ_j \nu), \quad 1\leq i\leq l,1\leq j\leq m.$
  \item[{\rm(ii)}]  $(\eta \circ_i \mu) \circ_{k-1+m} \nu= (\eta \circ_k \nu)\circ_i \mu,\quad 1 \leq i <k \leq l.$
  \item[{\rm(iii)}] There is an element ${\rm{Id}} \in \opd(1)$ such that $\mu \circ {\rm{Id}}=\mu$ and ${\rm{Id}} \circ \mu=\mu$.
     \end{enumerate}
\item[{\rm(c)}] A nonsymmetric operad $\opd=\{ \opd(n)\}_{n \geq 0}$ is called {\bf a (symmetric) operad}, if $\opd=\{ \opd(n)\}_{n \geq 0}$ is an $\mathbb{S}$-module and satisfies the following additional conditions
    \begin{enumerate}
  \item[{\rm(iv)}] For any $\sigma \in \mathbb{S}_{m}$, we have
  $$ \mu^{\sigma} \circ_i \nu=(\mu \circ_{\sigma(i)} \nu)^{\sigma'},\quad 1 \leq i\leq  m, $$
  where $\sigma'\in \mathbb{S}_{m+n-1}$ restricts to the bijection from $\{1,\dots,m+n-1\} \backslash \{i,\dots,i+n-1\}$ to $\{1,\dots,m+n-1\} \backslash \{\sigma(1),\dots,\sigma(i)+n-1\}$ according to
  $\sigma$ and restricts to the bijection from $\{i,\dots,i+m-1\}$ to $\{\sigma(i),\dots,\sigma(i)+m-1\}$ according to the identity (in the natural order).
  \item[{\rm(v)}] For any $\tau \in \mathbb{S}_{n}$, we have
  $$ \mu \circ_i \nu^{\tau}=(\mu \circ_{i} \nu)^{\tau'},\quad 1 \leq i\leq  m, $$
  where $\tau'\in \mathbb{S}_{m+n-1}$ is the permutation which acts by the identity, except on the block $\{i,\dots,i+n-1\}$ on which it acts via $\tau$.
  \end{enumerate}
\end{enumerate}

Theorem~\ref{pro:operad-sub-adj} gives an intrinsic explanation of the term \emph{anti-splitting}. \end{eqnarray*}

\begin{eqnarray*}
\eta: \antipostlie(2) \otimes \opd(2) \otimes \mathbf{k}. \mathrm{sgn}_{\mathbb{S}_2} &\to& \mathrm{ATSu}(\opd)(2)\\
\beta \otimes \omega \otimes 1 &\mapsto& \begin{pmatrix}

  \end{pmatrix},\\

\right.\\
\label{black-product-post-3}
& \left\{
\begin{aligned}

Thus we prove that $(\mathrm{ABSu}(\relsp))^{\perp}= (\mathrm{ADup}(\relsp^{\perp}))$.

As for {\rm(iii)}, by  \eqref{eq:annihilator}, for all $1\leq k \leq m$ and $1\leq l\leq 3n^2-m$, we have
$ \langle \st(f_k), \comp(g_l) \rangle=0, $
which implies that $(\st(\relsp))^{\perp}\supseteq (\comp(\relsp^{\perp}))$.
On the other hand, if
$$ h'=\sum_{i,j,u,v} x'_{i,j,u,v} \begin{pmatrix}
  e_i^{\vee}\\
  u
  \end{pmatrix} \circ_{\rmi} \begin{pmatrix}
  e_j^{\vee}\\
  v
  \end{pmatrix}+y'_{i,j,u,v} \begin{pmatrix}
  e_i^{\vee}\\
  u
  \end{pmatrix} \circ_{\rmii} \begin{pmatrix}
  e_j^{\vee}\\
  v
  \end{pmatrix}+z'_{i,j,u,v} \begin{pmatrix}
  e_i^{\vee}\\
  u
  \end{pmatrix} \circ_{\rmiii} \begin{pmatrix}
  e_j^{\vee}\\
  v
  \end{pmatrix}  $$
is in $(\st(R))^{\perp}$, where $u,v\in \{\dashv,\vdash\}$, then for all $1 \leq k \leq m$, we have
$ \langle h',\st(f_k)\rangle=0. $
For any $k$,
\begin{eqnarray*}
G &:=& \sum_{i,j}  a_{i,j}^k \begin{pmatrix}
  e_i \\
  \ast
  \end{pmatrix} \circ_{\rmi} \begin{pmatrix}
  e_j \\
  \ast
  \end{pmatrix}+b_{i,j}^{k}\begin{pmatrix}
  e_i \\
  \ast
  \end{pmatrix} \circ_{\rmii} \begin{pmatrix}
  e_j \\
  \ast
  \end{pmatrix}+c_{i,j}^k \begin{pmatrix}
  e_i \\
  \ast
  \end{pmatrix} \circ_{\rmiii} \begin{pmatrix}
  e_j \\
  \ast
  \end{pmatrix}
\end{eqnarray*}
is in $\st(f_k)$. Thus for all $1 \leq k \leq m$, we obtain
\begin{eqnarray*}
\sum_{i,j,u,v} a^k_{i,j} x'_{i,j,u,v}+b^k_{i,j} y'_{i,j,u,v}+c^k_{i,j} z'_{i,j,u,v}=\langle h',G \rangle=0.
\end{eqnarray*}
By \eqref{eq:anni-1}, we also have
$$ (\sum_{u,v} x'_{i,j,u,v}, \sum_{u,v} y'_{i,j,u,v}, \sum_{u,v} z'_{i,j,u,v})=\sum\limits_{l=1}^{3n^2-m} d'_l(\alpha_{i,j}^l,\beta_{i,j}^l,\gamma_{i,j}^l), $$
for some $d'_l \in \mathbf{k}$. Thus we obtain
\begin{eqnarray*}
h'= \sum_{l=1}^{3n^2-m} d'_l \Bigg(\sum_{i,j} \alpha^l_{i,j} \begin{pmatrix}
  e_i^{\vee}\\
  \dagger
  \end{pmatrix} \circ_{\rmi} \begin{pmatrix}
  e_j^{\vee}\\
  \dagger
  \end{pmatrix}+\beta^l_{i,j} \begin{pmatrix}
  e_i^{\vee}\\
  \dagger
  \end{pmatrix} \circ_{\rmii} \begin{pmatrix}
  e_j^{\vee}\\
  \dagger
  \end{pmatrix}+\gamma^l_{i,j} \begin{pmatrix}
  e_i^{\vee}\\
  \dagger
  \end{pmatrix} \circ_{\rmiii} \begin{pmatrix}
  e_j^{\vee}\\
  \dagger
  \end{pmatrix}\Bigg),
\end{eqnarray*}
which implies that $h$ is in the $\mathbb{S}$-module generated by $\comp(\relsp^{\perp})$. Thus we have $(\st(\relsp))^{\perp}= (\comp(\relsp^{\perp}))$. Therefore, we have
$ \mathrm{ADup}(\opd^!)=\mathrm{ABSu}(\opd)^!.$
