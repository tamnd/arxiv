<!-- theorem-1: body_md -->
\label{thm:main-noncompact}
Let $M$ be a noncompact rank-one symmetric space of real dimension $m=kn$.
For every bounded connected domain $\Omega\subset M$ with smooth boundary,
\begin{equation}\label{eq:main-noncompact}
  \sum_{j=1}^{m}\frac1{\mu_j(\Omega)}
  \ge \frac{m}{\mu_1(B_R)}.
\end{equation}
Equality holds if and only if $\Omega$ is a geodesic ball.
<!-- end theorem-1 -->
<!-- assumption-1: body_md -->
\label{ass:compact-localization}
The domain $\Omega$ is contained in a geodesic ball
$B_{i(M)/4}(q)$ of a compact rank-one symmetric space $M$.
<!-- end assumption-1 -->
<!-- theorem-2: body_md -->
\label{thm:main-compact}
Let $M$ be a compact rank-one symmetric space of real dimension $m=kn$, and
let $\Omega$ satisfy \cref{ass:compact-localization}.  Then
\begin{equation}\label{eq:main-compact}
  \sum_{j=1}^{m}\frac1{\mu_j(\Omega)}
  \ge \frac{m}{\mu_1(B_R)}
\end{equation}
with equality if and only if $\Omega$ is a geodesic ball, provided one of
 the following conditions holds:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2em]
  \item $k=1$;
  \item $n=1$;
  \item $1<k<m$ and
  \begin{equation}\label{eq:radius-condition}
     \cot^2 R\ge k-1.
  \end{equation}
\end{enumerate}
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor:compact-models}
Under \cref{ass:compact-localization}, the full estimate
\eqref{eq:main-compact} holds for every domain in $\mathbb{C}\mathrm P^n$ and $\mathbb{H}\mathrm P^n$.
For $M=\mathbb{O}\mathrm P^2$, it holds whenever
\begin{equation}\label{eq:Rstar}
  R\le R_*:=\arctan\frac1{\sqrt7}.
\end{equation}
In all these cases equality characterizes geodesic balls.
<!-- end corollary-1 -->
<!-- theorem-3: body_md -->
\label{thm:hopf-balanced-intro}
Under \cref{ass:compact-localization}, suppose the chosen Weinberger center
 satisfies $\Pi Z=0$.  Then the full estimate \eqref{eq:main-compact}
 holds, with equality only for geodesic balls, without any additional
 restriction on $R$.
<!-- end theorem-3 -->
<!-- lemma-1: body_md -->
\label{lem:radial-monotonicity}
The function $G$ is nondecreasing and $G'(r)=0$ for $r>R$.  In addition:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2em]
 \item in noncompact type, $G(r)/\sinh r$ is nonincreasing on $(0,\infty)$;
 \item in compact type, $G(r)/(\sin r\cos r)$ is nonincreasing on
 $(0,\pi/4)$.
\end{enumerate}
Consequently, the two angular weights in
\eqref{eq:A-compact} or \eqref{eq:A-noncompact}, after multiplication by
$G^2$, are nonincreasing on the relevant interval.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The monotonicity of $G$ follows from \eqref{eq:g-positive} and the constant
extension.  We include the quotient arguments, following the calculation
in \cite{MengWang2024}.

In noncompact type, set
\[
  s(r)=g'(r)-\coth r\,g(r).
\]
The regular expansion of the degree-one solution at the pole gives
$s(0+)=0$, while $s(R)=-\coth R\,g(R)<0$.  If $s$ were positive somewhere,
 it would have a positive interior maximum at a point $r_0$.  At that point
$s'(r_0)=0$, and combining this identity with \eqref{eq:radial-ode} gives
\begin{align*}
 0={}&\bigl(m\coth r_0+(k-1)\tanh r_0\bigr)g'(r_0)\\
 &+\left(\lambda_R-m\operatorname{csch}^2r_0+(k-1)\operatorname{sech}^2r_0\right)g(r_0).
\end{align*}
Since $s(r_0)>0$, we have $g'(r_0)>\coth r_0g(r_0)$.  Substitution in the
 preceding equality would imply
\[
 m\operatorname{csch}^2r_0-(k-1)\operatorname{sech}^2r_0-\lambda_R
 >m\coth^2r_0+(k-1),
\]
which is impossible.  Hence $s\le0$ and $g/\sinh r$ is nonincreasing on
$(0,R)$.  The constant extension preserves the conclusion for $r\ge R$.

In compact type, put
\[
  s(r)=g'(r)-(\cot r-\tan r)g(r).
\]
Again $s(0+)=0$ and $s(R)<0$.  At a hypothetical positive interior maximum
$r_0\in(0,R)$, the equations $s'(r_0)=0$ and
\eqref{eq:radial-ode} yield
\begin{align*}
 0={}&(m\cot r_0-k\tan r_0)g'(r_0)\\
 &+\left(\lambda_R-\frac{m-k}{\sin^2r_0}
 -\frac{k}{\sin^2r_0\cos^2r_0}\right)g(r_0).
\end{align*}
On $(0,\pi/4)$ the first coefficient is positive, and $s(r_0)>0$ implies

\[
  \lambda_R<2(m+k).
\]

nonincreasing.  It is strictly decreasing after $R$ because $G$ is constant

 \frac{G^2}{\sinh^2r\cosh^2r}
\]
are nonincreasing.  In compact type the second angular weight equals
$(G/(\sin r\cos r))^2$, and the first equals this function times
$\cos^2r$; both are nonincreasing on $(0,\pi/4)$.

<!-- theorem-4: body_md -->
\label{thm:hopf-projection}
There is a linear map
\[
  \Pi:\operatorname{Sym}(V)\longrightarrow\operatorname{Sym}(V)
\]
with the following properties.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2em]
 \item $\Pi$ is positive, unital, trace-preserving, Hilbert--Schmidt
 self-adjoint, and idempotent.
 \item For every $\xi\in S(V)$,
 \begin{equation}\label{eq:Q-via-Pi}
   Q_\xi=k\Pi(P_\xi)-P_\xi.
 \end{equation}
 \item The complex-linear extension of $\Pi$ to $M_m(\mathbb C)$ is a positive
 unital map.
\end{enumerate}
Consequently, if
\begin{equation}\label{eq:C-def}
  \mathcal C:=k\Pi-I,
\end{equation}
then $\mathcal C$ is positive on real positive semidefinite matrices,
$\mathcal C(P_\xi)=Q_\xi$, $\mathcal C(I)=(k-1)I$, and
\begin{equation}\label{eq:C-spectral-splitting}
 \mathcal C=(k-1)I\quad\text{on }\operatorname{Ran}\Pi,
 \qquad
 \mathcal C=-I\quad\text{on }\operatorname{Ker}\Pi.
\end{equation}

<!-- proof-2: body_md -->
We first treat $\mathbb K=\mathbb R,\mathbb C,\mathbb H$.  Regard $V$ as $\mathbb K^n$ with its standard
real inner product.  Let $J_0=I$ and let $J_1,\ldots,J_{k-1}$ be right
multiplication by an orthonormal basis of the imaginary part of $\mathbb K$.  Thus
all $J_\alpha$ are orthogonal, and the imaginary ones are skew-symmetric.
Define
\begin{equation}\label{eq:Pi-associative}
  \Pi(A)=\frac1k\sum_{\alpha=0}^{k-1}
  J_\alpha A J_\alpha^{\mathsf T}.
\end{equation}
For $\mathbb K=\mathbb C$ or $\mathbb H$, the conjugations
$A\mapsto J_\alpha A J_\alpha^{\mathsf T}$ form a finite group after signs
are discarded.  Hence their average is the orthogonal projection onto the
common fixed-point space.  This proves idempotence and Hilbert--Schmidt
self-adjointness.  Positivity, unitality, and trace preservation follow
immediately from \eqref{eq:Pi-associative}.  The same formula on complex
matrices is an average of unitary conjugations and is therefore positive and
unital.

For a unit vector $\xi$, the vectors
$J_0\xi,\ldots,J_{k-1}\xi$ form an orthonormal basis of the real $k$-plane
$\mathbb K\xi$.  Therefore
\[
  \Pi(P_\xi)=\frac1k\sum_{\alpha=0}^{k-1}P_{J_\alpha\xi}
  =\frac1kP_{\mathbb K\xi}.
\]
Since the Hopf space is $\mathcal V_\xi=\mathbb K\xi\ominus\mathbb R\xi$, we have
$Q_\xi=P_{\mathbb K\xi}-P_\xi$, proving \eqref{eq:Q-via-Pi}.

We next treat the Cayley plane.  Use the real spin representation
$V=\mathbb R^{16}$ of $\operatorname{Spin}(9)$.  Choose a symmetric Clifford system
$S_0,\ldots,S_8$ satisfying
\begin{equation}\label{eq:clifford-system}
  S_\alpha S_\beta+S_\beta S_\alpha
  =2\delta_{\alpha\beta}I.
\end{equation}
The matrices $I,S_0,\ldots,S_8$ are mutually orthogonal for the
Hilbert--Schmidt product, each with squared norm $16$.  Define
\begin{equation}\label{eq:Pi-cayley}
  \Pi(A)=\frac1{16}\left((\operatorname{tr} A)I+
  \sum_{\alpha=0}^{8}\operatorname{tr}(AS_\alpha)S_\alpha\right).
\end{equation}
Thus $\Pi$ is the Hilbert--Schmidt projection onto
$\operatorname{span}\{I,S_0,\ldots,S_8\}$; in particular it is self-adjoint and
idempotent.  It is unital and trace-preserving because
$\operatorname{tr} S_\alpha=0$.

For $\xi\in S^{15}$, put
$a_\alpha=\left\langle S_\alpha\xi,\xi\right\rangle$ and
$S(a)=\sum_{\alpha=0}^8a_\alpha S_\alpha$.  The standard Clifford-system
model of the octonionic Hopf fibration gives
\begin{equation}\label{eq:octonionic-hopf-identity}
  \sum_{\alpha=0}^{8}a_\alpha^2=1,
  \qquad S(a)\xi=\xi;
\end{equation}
see \cite{FerusKarcherMunzner1981,OrneaPartonPiccinniVuletescu2013}.
By \eqref{eq:clifford-system}, $S(a)^2=I$.  Its $+1$ eigenspace has real
dimension $8$ and is the octonionic line through $\xi$.  Denoting its
orthogonal projection by $P_{\mathbb O\xi}$, we obtain
\[
  P_{\mathbb O\xi}=\frac12(I+S(a)),
  \qquad
  \Pi(P_\xi)=\frac1{16}(I+S(a))=\frac18P_{\mathbb O\xi}.
\]
It follows that $8\Pi(P_\xi)-P_\xi=Q_\xi$, which is
\eqref{eq:Q-via-Pi} for $k=8$.

The last formula also proves positivity of $\Pi$ on real symmetric
matrices, because every positive semidefinite matrix is a nonnegative
linear combination of real rank-one projections.  To verify the positivity
needed for operator Jensen's inequality, extend \eqref{eq:Pi-cayley}
complex-linearly.  For a complex unit vector $z$, the numbers
$a_\alpha=z^*S_\alpha z$ are real.  Since
$S(a)^2=|a|^2I$ and
\[
  |a|^2=z^*S(a)z\le \left\lVert S(a)\right\rVert=|a|,
\]
we have $|a|\le1$.  Hence
\[
  \Pi(zz^*)=\frac1{16}(I+S(a))\succeq0.
\]
By spectral decomposition, the complex extension is positive on all
Hermitian positive semidefinite matrices.

Finally, \eqref{eq:C-spectral-splitting} follows from $\Pi^2=\Pi$.  Since
$\mathcal C(P_\xi)=Q_\xi\succeq0$, the real rank-one decomposition also shows
that $\mathcal C$ is positive on every real positive semidefinite matrix.

<!-- lemma-2: body_md -->
\label{lem:trace-ritz}
Let $\Omega$ be connected, and let $P_1,\ldots,P_m\in H^1(\Omega)$ be
linearly independent in $L^2(\Omega)$ and satisfy
$\int_\Omega P_i\,\mathrm dv=0$.  Define
\begin{equation}\label{eq:mass-stiffness-def}
  M_{ij}=\int_\Omega P_iP_j\,\mathrm dv,
  \qquad
  K_{ij}=\int_\Omega\left\langle \nabla P_i,\nabla P_j\right\rangle\,\mathrm dv.
\end{equation}
Then $M$ and $K$ are positive definite, and
\begin{equation}\label{eq:trace-ritz}
  \sum_{j=1}^{m}\frac1{\mu_j(\Omega)}
  \ge \operatorname{tr}(K^{-1}M).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
The matrix $M$ is the $L^2$ Gram matrix, hence is positive definite.  If
$a^{\mathsf T}Ka=0$, then the function
$P_a=\sum_i a_iP_i$ has zero weak gradient and is constant on the connected
domain $\Omega$.  Its mean is zero, so it vanishes.  Linear independence
then gives $a=0$, and $K$ is positive definite.

Let $0<\eta_1\le\cdots\le\eta_m$ be the Ritz values of the Neumann
quadratic form restricted to $\operatorname{span}\{P_1,\ldots,P_m\}$.  Since this trial
space is contained in the orthogonal complement of the constants, the
min--max principle gives $\mu_j(\Omega)\le\eta_j$.  The $\eta_j$ are the
generalized eigenvalues of $Ka=\eta Ma$, so their reciprocals are the
eigenvalues of $K^{-1/2}MK^{-1/2}$.  Therefore
\[
  \sum_{j=1}^{m}\frac1{\mu_j(\Omega)}
  \ge\sum_{j=1}^{m}\frac1{\eta_j}
  =\operatorname{tr}(K^{-1/2}MK^{-1/2})=\operatorname{tr}(K^{-1}M).
\]

\[
  \Phi(t)=\int_0^tG(s)\,\mathrm d s.
\]

$\mathscr F(p)=\int_\Omega\Phi(\operatorname{dist}(p,x))\,\mathrm dv(x)$.  Since $G$ is bounded

\label{lem:ray-rearrangement}

 F(y)\ge F(Y_R)+w(R)(y-Y_R)

 D=\operatorname{tr}\bigl((I-Y)^{-1}\Pi Y\bigr).
\]
Trace preservation and $\operatorname{tr} X=0$ give $\operatorname{tr} Y=0$.  If
$y_1,\ldots,y_m<1$ are the eigenvalues of $Y$, then strict convexity of
$t\mapsto t/(1-t)$ gives
\begin{equation}\label{eq:H-nonnegative}
  H=\sum_{i=1}^{m}\frac{y_i}{1-y_i}\ge0,
  \qquad H=0\Longleftrightarrow Y=0.
\end{equation}

We claim that $D\le H$.  On the open convex set $I-A\succ0$, let
\[
  \Phi(A)=-\log\det(I-A).
\]
Convexity and the derivative
$D\Phi(Y)[B]=\operatorname{tr}((I-Y)^{-1}B)$ give
\begin{equation}\label{eq:Phi-gradient}
 \operatorname{tr}\bigl((I-Y)^{-1}(Y-\Pi Y)\bigr)
 \ge \Phi(Y)-\Phi(\Pi Y).
\end{equation}
Since $\Pi$ is positive and unital,
$I-\Pi Y=\Pi(I-Y)\succ0$.  The function $t\mapsto-\log t$ is
operator convex.  The Davis--Choi--Jensen inequality
\cite{HansenPedersen2003} therefore yields
\[
 -\log(I-\Pi Y)
 \preceq\Pi\bigl(-\log(I-Y)\bigr).
\]
Taking traces and using trace preservation gives
$\Phi(\Pi Y)\le\Phi(Y)$.  The left-hand side of
\eqref{eq:Phi-gradient} is $H-D$, proving the claim.

Finally,
\begin{align*}
 \operatorname{tr}\bigl(T^{-1}(I+X)\bigr)-m
 &=\operatorname{tr}\bigl((I-Y)^{-1}(X+Y)\bigr)\\
 &=\left(1+\frac1u\right)H-
   \frac{v}{u(u+v)}D\\
 &\ge\left(1+\frac1{u+v}\right)H\ge0.
\end{align*}
This proves \eqref{eq:positive-projection-jensen}.  Equality forces $H=0$
and hence $Y=0$.  The linear operator $uI+v\Pi$ acts by $u+v$ on
$\operatorname{Ran}\Pi$ and by $u$ on $\operatorname{Ker}\Pi$, so it is invertible; consequently
$X=0$.  The converse is immediate.

<!-- lemma-9: body_md -->
\label{lem:positive-mass-closure}
Let $X\in\operatorname{Sym}_m$ satisfy $\operatorname{tr} X=0$ and put $A=I+X$.  Suppose
$\widehat M,T\in\operatorname{Sym}_m$ satisfy
\begin{equation}\label{eq:positive-mass-hypotheses}
 \widehat M\succeq0,
 \qquad \widehat M\succeq A,
 \qquad 0<T\preceq2I,
 \qquad \operatorname{tr}(TA)\le m.
\end{equation}
Then
\begin{equation}\label{eq:positive-mass-conclusion}
  \operatorname{tr}(T^{-1}\widehat M)\ge m.
\end{equation}
<!-- end lemma-9 -->
<!-- proof-12: body_md -->
Set
\[
 N=T^{-1/2}\widehat M T^{-1/2},
 \qquad B=T^{-1/2}AT^{-1/2}.
\]
Then $N\succeq0$ and $N\succeq B$.  Because $0<T\preceq2I$, the matrix
$P=2T-T^2$ satisfies $0\preceq P\preceqI$.  Therefore
\[
 \operatorname{tr} N\ge\operatorname{tr}(PN)\ge\operatorname{tr}(PB).
\]
The second inequality follows from $P\succeq0$ and $N-B\succeq0$.  Since
$P$ commutes with $T$, cyclicity of the trace gives
\[
 \operatorname{tr}(PB)=\operatorname{tr}\bigl((2I-T)A\bigr)
 =2\operatorname{tr} A-\operatorname{tr}(TA)\ge2m-m=m.
\]
Since $\operatorname{tr} N=\operatorname{tr}(T^{-1}\widehat M)$, the proof is complete.

There is a nonzero trace-free element of $\operatorname{Ran}\Pi$: in the complex and
quaternionic cases with $n\ge2$, one may take a real symmetric
$\mathbb K$-linear operator with $\mathbb K$-diagonal entries $1,-1,0,\ldots,0$; in the
Cayley case one may take any $S_\alpha$ from
\eqref{eq:clifford-system}.  For $X\in\operatorname{Ran}\Pi$,
$\mathcal C(X)=(k-1)X$, so
\begin{equation}\label{eq:T-I-plus-sX}
  T=I+sX,
  \qquad s=(k-1)\beta-\alpha>0.
\end{equation}
The ball lower bound $\lambda_R\ge2(m+k)$ and $R\le\pi/4$ imply
\[
  0<s<\frac{2(k-1)}{\lambda_R}<1.
\]
After multiplying a fixed nonzero trace-free element of $\operatorname{Ran}\Pi$ by a
sufficiently small scalar, both $I+X$ and $I+sX$ are positive definite.

Let $x_1,\ldots,x_m$ be the eigenvalues of $X$.  Then
$\sum_i x_i=0$ and
\[
 \operatorname{tr}\bigl(T^{-1}(I+X)\bigr)
 =\sum_{i=1}^{m}h_s(x_i),
 \qquad h_s(t)=\frac{1+t}{1+st}.
\]
Because $0<s<1$,
\[
  h_s''(t)=-\frac{2s(1-s)}{(1+st)^3}<0
\]
on the relevant interval.  Strict concavity and $X\ne0$ give
\[
  \frac1m\sum_{i=1}^{m}h_s(x_i)
  <h_s\left(\frac1m\sum_i x_i\right)=h_s(0)=1,
\]
which is \eqref{eq:obstruction-trace}.  Taking
$\widehat M=I+X$ and $K=T$ proves the final assertion.

<!-- lemma-10: body_md -->
\label{lem:imbalance-realization}
Let $A\in\operatorname{Sym}(V)$ be trace-free.  For every $R$ strictly below the maximal
polar radius under consideration, there are smooth, centrally symmetric,
star-shaped domains $\Omega_\varepsilon$ with $|\Omega_\varepsilon|=|B_R|$ such that
\begin{equation}\label{eq:Z-realization}
  Z_\varepsilon=\frac{2|\mathbb S^{m-1}|}{m(m+2)}\varepsilon A.
\end{equation}
For sufficiently small $|\varepsilon|$, the domains remain in the same compact
localization ball whenever $R<\pi/8$.
<!-- end lemma-10 -->
<!-- proof-19: body_md -->
Define the desired ray volume by
\[
  Y_\varepsilon(\xi)=Y_R+\varepsilon\left\langle A\xi,\xi\right\rangle.
\]
For small $|\varepsilon|$ this is positive and lies in the range of $V$.  Since
\[
  \int_{S(V)}\left\langle A\xi,\xi\right\rangle\,\mathrm d\sigma
  =\frac{|\mathbb S^{m-1}|}{m}\operatorname{tr} A=0,
\]
the total volume is unchanged.  Put
$r_\varepsilon(\xi)=V^{-1}(Y_\varepsilon(\xi))$ and take the radial graph
$0<r<r_\varepsilon(\xi)$.  It is smooth for small $\varepsilon$ and centrally symmetric
because $Y_\varepsilon(-\xi)=Y_\varepsilon(\xi)$; in particular $o$ is a Weinberger
center.

The fourth-moment identity on the sphere is
\[
 \int_{S^{m-1}}\xi_i\xi_j\xi_p\xi_q\,\mathrm d\sigma
 =\frac{|\mathbb S^{m-1}|}{m(m+2)}
 (\delta_{ij}\delta_{pq}+\delta_{ip}\delta_{jq}
 +\delta_{iq}\delta_{jp}).
\]
Using $\operatorname{tr} A=0$ in the definition of $Z$ gives
\eqref{eq:Z-realization}.  The localization statement follows by uniform
convergence $r_\varepsilon\to R$.

\label{rem:cayley-obstruction}
Let $M=\mathbb{O}\mathrm P^2$ and
\[
  R_*<R<\frac\pi8,
  \qquad R_*=\arctan\frac1{\sqrt7}.
\]
Take $X=tS_0$, where $S_0$ is one member of the Clifford system and
$0<t<1$.  Then $X\in\operatorname{Ran}\Pi$, $\operatorname{tr} X=0$, and the eigenvalues of $X$ are
$t$ and $-t$, each with multiplicity eight.  Since $\mathcal C(X)=7X$, put
$s=7\beta-\alpha\in(0,1)$.  Formula \eqref{eq:T-I-plus-sX} gives
\[
 \operatorname{tr}\bigl(T^{-1}(I+X)\bigr)
 =8\frac{1+t}{1+st}+8\frac{1-t}{1-st}<16.
\]
By \cref{lem:imbalance-realization}, the same matrix direction occurs for
arbitrarily small smooth star-shaped perturbations that remain inside the
original localization range.  This does not disprove the full spectral
inequality: the exact mass and stiffness matrices retain nonlinear raywise
remainders that are absent from the affine model.
