<!-- lemma-1: body_md -->
\label{lem:unisolv}
The linear forms \eqref{linear form points}-\eqref{linear forms moment} are uni-solvant on $V$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
See appendix \ref{app:lemma}.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\begin{equation}
    \label{curl}
\dfrac{d\textbf{\text{ROT }}\mathbf{u}_{\bm\sigma}}{dt}=0.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
We have
$$\vert C_{\bm\sigma}\vert \dfrac{d\textbf{\text{ROT }}\mathbf{u}_{\bm\sigma}}{dt}=\sum_{K{\bm\sigma}\in K}  \nabla\varphi_{\bm\sigma}^\bot \cdot \dfrac{d}{dt}\int_K  \mathbf{u} \,d\mathbf{x}=
-
\sum_{K{\bm\sigma}\in K}  \int_K \nabla\varphi_{\bm\sigma}^\bot \cdot \nabla p\; d\mathbf{x}
$$
because $\nabla\varphi_{\bm\sigma}$ is constant and 
$$\vert K\vert \dfrac{d\bar\mathbf{u}_K}{dt}=-\int_{\partial K} p\mathbf{n}\,d\gamma=-\int_K\nabla p\; d\mathbf{x}$$
because $\mathbf{u}$ is smooth in $K$.
Then we see that 
$$\int_K \nabla\varphi_{\bm\sigma}^\bot \cdot \nabla p\; d\mathbf{x}=\int_{K }\bigg (
\dfrac{\partial \varphi_{\bm\sigma}}{\partial y}\dfrac{\partial p}{\partial x}-\dfrac{\partial \varphi_{\bm\sigma}}{\partial x}\dfrac{\partial p}{\partial y}\bigg )\; d\mathbf{x}$$
and since
$$\dfrac{\partial \varphi_{\bm\sigma}}{\partial y}\dfrac{\partial p}{\partial x}-\dfrac{\partial \varphi_{\bm\sigma}}{\partial x}\dfrac{\partial p}{\partial y}=
\dfrac{\partial }{\partial y}\bigg ( \varphi_{\bm\sigma}\dfrac{\partial p}{\partial x}\bigg )-\dfrac{\partial }{\partial x}\bigg ( \varphi_{\bm\sigma}\dfrac{\partial p}{\partial y}\bigg )$$
because in $K$, we can apply Schwarz theorem on $p$ (\ldots), we get
using Stokes formula,
$$\int_K \nabla\varphi_{\bm\sigma}^\bot \cdot \nabla p\; d\mathbf{x}=\int_{\partial K} \nabla p\cdot {\bm\tau} \varphi_{\bm\sigma}\,d\gamma.$$
Then using the same arguments as before (the tangential derivative of $p$ is counted twice with opposite signs, $\varphi_{\bm\sigma}$ is continuous accorss edges and vanishes on the boundary of its support), we see that \eqref{curl} is true.
<!-- end proof-2 -->
<!-- remark-1: body_md -->
\label{rem:simplex}
$~$
\begin{enumerate}
    \item 
We get one involution per vertex. The proof uses the fact that the gradient of a piecewise linear function is constant per element and thus since the characteristic function of the triangle are used to compute the first moment (the average), we can use this gradient as a test function. 
 For mid points we cannot do the same.
 \item Similarly, the proof does not work for polygons, whatever degree because we would need the basis function attached to the ${\bm\sigma}$, and the moments are not expressed with this.
 \end{enumerate}
<!-- end remark-1 -->
<!-- proposition-2: body_md -->
\label{prop:curlHO}
 Let $k\geq 3$ and ${\bm {\mu}}=(\mu_1,\mu_2,\mu_3)$ with $\vert {\bm {\mu}}\vert=\mu_1+\mu_2+\mu_3=k-1\geq 2$.
 We set $$B_{\bm {\mu}}=\frac{k-1}{\mu_1!\mu_2!\mu_3!}{\bm {\lambda}}^{\bm {\mu}}_{k-1},$$ the B\'ezier polynomial on $K$
     Then the weak curls $\textbf{\text{ROT }}\mathbf{u}_{\bm\sigma}$ defined by
     $$
     \int_{\Omega}{\bm {\lambda}}_{k-2}^{{\bm {\mu}}}(\mathbf{x})\textbf{\text{ROT }}\mathbf{u}_{\bm\sigma}\,d\mathbf{x}=                                                                                             
     \int_\Omega  \nabla B_{\bm {\mu}} ^\bot \cdot \mathbf{u}\,d\mathbf{x}$$
     verify
      $$
      \dfrac{d}{dt} \textbf{\text{ROT }}\mathbf{u}_{\bm\sigma} =0.
      $$
      when $\mathbf{u}$ is the degree $k\ge 3$ active flux
          approximation of velocity in the  acoustics  system.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
Applying twice  \eqref{eq:grad_mom2} we can write
     as  
     $$  
     \dfrac{d}{dt}\int_K \big (\nabla{\bm {\lambda}}^{\bm {\mu}}_{k-1}\big )^\bot \cdot \mathbf{u}\,d\mathbf{x}=\sum\limits_{\vert \mu \vert =k-2} \mathbf{b}_{{\bm {\mu}}}\cdot 
      \dfrac{d}{dt}\int_K  {\bm {\lambda}}^{\bm {\mu}}_{k-2} \mathbf{u}\,d\mathbf{x}=
     - \sum\limits_{\vert \mu \vert =k-2} \mathbf{b}_{{\bm {\mu}}} \cdot \int_K  {\bm {\lambda}}^{\bm {\mu}}_{k-2} \nabla p\,d\mathbf{x}=
     -   \int_K \big (\nabla{\bm {\lambda}}^{\bm {\mu}}_{k-1}\big )^\bot \cdot \nabla p\,d\mathbf{x}
     $$
having also used
$$
 \dfrac{d}{dt}\int_K  {\bm {\lambda}}^{\bm {\mu}}_{k-2} \mathbf{u}\,d\mathbf{x} =
  - \int_K  {\bm {\lambda}}^{\bm {\mu}}_{k-2}\nabla p\,d\mathbf{x}
$$
This allows to write
     $$
     \dfrac{d}{dt}\int_K \big (\nabla{\bm {\lambda}}^{\bm {\mu}}_{k-1}\big )^\bot \cdot \mathbf{u}\,d\mathbf{x}=-\int_{\partial K} {\bm {\lambda}}^{\bm {\mu}}_{k-1} \dfrac{\partial p}{\partial {\bm\tau}}\; d\gamma.
     $$
             
                                                                        
     Then
     \begin{equation*}
     \begin{split}\int_\Omega {\bm {\lambda}}^{\bm {\mu}}_{k-2} \dfrac{d\textbf{\text{ROT }}\mathbf{u}_{\bm\sigma}}{dt}&=\sum_{K\subset \text{supp}(B_{\bm {\mu}})}\int_K \big (\nabla B_{\bm {\mu}}\big )^\bot\cdot \dfrac{d\mathbf{u}}{dt}\\
     &=-\sum_{K\subset \text{supp}(B_{\bm {\mu}})}\int_K \big (\nabla B_{\bm {\mu}}\big )^\bot\cdot \nabla p\; d\mathbf{x}\\
     &=-\frac{k-1}{\mu_1!\mu_2!\mu_3!}\sum_{K\subset \text{supp}(B_{\bm {\mu}})}\int_{\partial K}{\bm {\lambda}}^{\bm {\mu}}_{k-1} \dfrac{\partial p}{\partial {\bm\tau}}\\
     &=0
     \end{split}
     \end{equation*}
     because on the internal edges, ${\bm {\lambda}}^{\bm {\mu}}_{k-1}$ is continuous and $\dfrac{\partial p}{\partial {\bm\tau}}$ is counted twice with opposite sign, and on the boundary edges, ${\bm {\lambda}}^{\bm {\mu}}_{k-1}=0$.

<!-- remark-2: body_md -->
\label{rem:3d}
The above results provides a justification of why AF works so well on triangle for acoustics: in the quadratic, we get $n_v$ involutions ($n_v=3$ being the number of vertices). In the case of $k>2$, we get $n_v+(k-2)n_{edges}= 3 + 3(k-2)$ involutions, where $n_{edge}=3$ is the number of edges.

    This result can be easily be extended to the 3D case on thetrahedra. The number of involutions will be $n_v+\max(k-2,0)n_{edge}+\max(k-3,0)n_{faces}$. Here $n_v=4$, and $n_{face}=4$ is the number of faces, while $n_{edge}=6$.

<!-- proposition-3: body_md -->
\begin{equation}
    \label{div}
\dfrac{d\textbf{\text{DIV }}\mathbf{B}_{\bm\sigma}}{dt}=0.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-4: body_md -->
We have
$$\vert C_{\bm\sigma}\vert \dfrac{d\textbf{\text{DIV }}\mathbf{B}_{\bm\sigma}}{dt}=\sum_{K{\bm\sigma}\in K}  \nabla\varphi_{\bm\sigma} \cdot \dfrac{d}{dt}\int_K  \mathbf{B} \,d\mathbf{x}=
-
\sum_{K{\bm\sigma}\in K}  \int_K \nabla\varphi_{\bm\sigma} \cdot { \mathbf{ curl }\;} E\; d\mathbf{x}
$$
having used the evolution of $\bar\mathbf{B}_K$. 
Using Schwarz theorem for $E$, we see that 
$$\int_K \nabla\varphi_{\bm\sigma}  \cdot { \mathbf{ curl }\;} E\; d\mathbf{x}=\int_{K }\bigg (
\dfrac{\partial \varphi_{\bm\sigma}}{\partial x}\dfrac{\partial E}{\partial y}-\dfrac{\partial \varphi_{\bm\sigma}}{\partial y}\dfrac{\partial E}{\partial x}\bigg )\; d\mathbf{x}
= 
\dfrac{\partial }{\partial x}\bigg ( \varphi_{\bm\sigma}\dfrac{\partial E}{\partial y}\bigg )-\dfrac{\partial }{\partial y}\bigg ( \varphi_{\bm\sigma}\dfrac{\partial E}{\partial x}\bigg ).
$$
Finally we obtain
$$\int_K \nabla\varphi_{\bm\sigma}  \cdot { \mathbf{ curl }\;} E\; d\mathbf{x}=
\int_{\partial K} \nabla E\cdot {\bm\tau} \varphi_{\bm\sigma}.$$
Then using the same arguments as before (the tangential derivative of $E$ is counted twice with opposite signs, $\varphi_{\bm\sigma}$ is continuous across edges and vanishes on the boundary of its support), we see that \eqref{div} is true.

<!-- proposition-4: body_md -->
\label{prop:divHO}
 Let $k\geq 3$ and ${\bm {\mu}}=(\mu_1,\mu_2,\mu_3)$ with $\vert {\bm {\mu}}\vert=\mu_1+\mu_2+\mu_3=k-1\geq 2$.
 We set $$B_{\bm {\mu}}=\frac{k-1}{\mu_1!\mu_2!\mu_3!}{\bm {\lambda}}^{\bm {\mu}}_{k-1},$$ the B\'ezier polynomial on $K$
     Then the weak divs $\textbf{\text{DIV }}\mathbf{B}_{\bm\sigma}$ defined by
     $$
     \int_{\Omega}{\bm {\lambda}}_{k-2}^{{\bm {\mu}}}(\mathbf{x})\textbf{\text{DIV }}\mathbf{B}_{\bm\sigma}\,d\mathbf{x}=                                                                                             
     \int_\Omega  \nabla B_{\bm {\mu}} \cdot \mathbf{B}\,d\mathbf{x}$$
     verify
      $$
      \dfrac{d}{dt} \textbf{\text{DIV }}\mathbf{B}_{\bm\sigma} =0.
      $$
      when $\mathbf{B}$ is the degree $k\ge 3$ active flux
          approximation of Magnetic field in the  Maxwell  system.
<!-- end proposition-4 -->
<!-- proof-5: body_md -->
The proof is very similar to the one of \ref{prop:curlHO}:
  the repeated application  of \eqref{eq:grad_mom1}, and the evolution of the moments of the magnetic field   
   allow to show 
     $$
     \dfrac{d}{dt}\int_K \big (\nabla{\bm {\lambda}}^{\bm {\mu}}_{k-1}\big )  \cdot \mathbf{B}\,d\mathbf{x}=-\int_{\partial K} {\bm {\lambda}}^{\bm {\mu}}_{k-1} \dfrac{\partial E}{\partial {\bm\tau}}\; d\gamma.
     $$
<     Then, using the definition of the weak divergence moments, and the same continuity arguments of the tangent derivatives used before  we show that 
     \begin{equation*}
     \begin{split}\int_\Omega {\bm {\lambda}}^{\bm {\mu}}_{k-2} \dfrac{d\textbf{\text{DIV }}\mathbf{B}_{\bm\sigma}}{dt}
     &=-\frac{k-1}{\mu_1!\mu_2!\mu_3!}\sum_{K\subset \text{supp}(B_{\bm {\mu}})}\int_{\partial K}{\bm {\lambda}}^{\bm {\mu}}_{k-1} \dfrac{\partial E}{\partial {\bm\tau}}\\
     &=0.
     \end{split}
     \end{equation*}
