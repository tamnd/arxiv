\label{theo: main} Let $G$ be a non-compact connected semi-simple Lie group with finite center. Let $\Gamma$ be a lattice in $G$. With the notation as above, for any continuous functions $\varphi,\psi,\varphi',\psi'$ on $G/P$,
$$
\lim_{T\to\infty}\frac{1}{|\Gamma_{T}|}\sum_{\gamma\in \Gamma_{T}}\frac{\langle\pi(\gamma)\varphi,\psi\rangle\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi^{2}(\gamma)}=\langle \varphi,\varphi'\rangle\overline{\langle\psi,\psi'\rangle}.
$$

<!-- coro-1: body_md -->
\label{cor: equivalences in rank 1} Let $G$ be a connected semi-simple Lie group with finite center.  Assume  the real rank of $G$ equals $1$. Let $\Gamma$ be a lattice in $G$. The following three conditions are equivalent
\begin{enumerate}
\item For any  functions $\varphi,\psi,\varphi',\psi'$ in $L^{2}(G/P,\nu)$,
$$
\lim_{T\to\infty}\frac{1}{|\Gamma_{T}|}\sum_{\gamma\in \Gamma_{T}}\frac{\langle\pi(\gamma)\varphi,\psi\rangle\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi^{2}(\gamma)}=\langle \varphi,\varphi'\rangle\overline{\langle\psi,\psi'\rangle}.
$$
\item The restricted representation $\pi|_{\Gamma}$ satisfies property RD
with respect to $L|_{\Gamma}$. More precisely, there exists a constant
$C>0$ such that, for all unit vectors
$\varphi,\psi\in L^{2}(G/P,\nu)$ and all $T\geq1$,
\[
\sum_{\gamma\in\Gamma_{T+1}\setminus\Gamma_T}
|\langle\pi(\gamma)\varphi,\psi\rangle|^{2}
\leq CT^{2}.
\]
The exponent $2$ is optimal.


\item The lattice $\Gamma$ is uniform.
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lemma: half Roblin} Let $U,V$ be Borel subsets of $K/M$ with $\nu(\partial U)=\nu(\partial V)=0$.
Let $\theta>0$. Then
\[
	\lim_{T\to\infty}\frac{|\{\gamma\in\Gamma^{\theta}_T: {\hat \gamma}\in U\,\, \mbox{and}\,\, {\check{\gamma}}\in V \}|}{|\Gamma^{\theta}_T|}=\nu(U)\nu(V).
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $\tilde U=p^{-1}(U)$ and $\tilde V=p^{-1}(V)$. 
Let $N(A)$ be the normalizer of $A$ in $G$ and let $M'=N(A)\cap K$. Recall that the Weyl group is the quotient $W=M'/M$ and that it contains a unique element $s_0$ which sends $\frak{a}^{+}$ to $-\frak{a}^{+}$.
Let $m_0\in M'$ such that $m_0M=s_0$. Recall that the \emph{opposition involution}
\[
	\iota:\frak{a}^+\to\frak{a}^+,\quad H\mapsto -\mathrm{Ad}(m_0)H
\]
satisfies $\exp(-H)=m_0^{-1}\exp(\iota H)m_0$  for all $H\in\frak{a}^+$.
Let $g\in KA^{+}K$. It means there exists $k,l\in K$ and $H\in\frak a^{+}$ such that $g=k\exp(H)l$. Using the identity
$$g^{-1}=l^{-1}\exp(-H)k^{-1}=l^{-1}m_0^{-1}\exp(\iota H)m_0k^{-1},$$ 
it is elementary to check the equality
$$
\{g\in KA^{+}K: {\hat g}\in U\,\, \mbox{and}\,\, {\check{g}}\in V \}=\tilde UA^+m_0^{-1}(\tilde V)^{-1}.
$$
Let $T>0$. Using twice the inclusion $\Gamma^{\theta}_{T}\subset KA^{+}K$ and the above equality we obtain:
\begin{align*}
&\{\gamma\in\Gamma^{\theta}_T: {\hat \gamma}\in U\,\, \mbox{and}\,\, {\check{\gamma}}\in V \}\\
&=\Gamma^{\theta}_T\cap\{g\in KA^{+}K : {\hat g}\in U\,\, \mbox{and}\,\, {\check{g}}\in V \}\\
&=\Gamma^{\theta}_T\cap\tilde UA^+m_0^{-1}(\tilde V)^{-1}\\
&=\Gamma^{\theta}_T\cap\tilde U\overline{A^+}m_0^{-1}(\tilde V)^{-1}.
\end{align*}
In order to apply \cite[Theorem 1.6]{GorOh}, we put $\Omega_1=\tilde U$ and $\Omega_2=m_0^{-1}(\tilde V)^{-1}$. On one hand we have:
\begin{align*}
\tilde U\overline{A^+}m_0^{-1}(\tilde V)^{-1}&=\tilde UM\overline{A^+}m_0^{-1}(\tilde V)^{-1}=\tilde U\overline{A^+}Mm_0^{-1}(\tilde V)^{-1}\\
					&=\Omega_1\overline{A^+}M\Omega_2.
\end{align*} 
On the other hand, for $i=1,2$, $\mu(\partial\Omega_i)=0$ because $\nu(\partial U)=\nu(\partial V)=0$. Moreover, $\mu(\Omega_1 M)=\mu(\tilde U M)=\mu(\tilde U)=\nu(U)$, and, as $M$ is normal in $M'$,
\begin{align*}
\mu(M\Omega_2)&=\mu(Mm_0^{-1}(\tilde V)^{-1})=\mu(m_0Mm_0^{-1}(\tilde V)^{-1})\\
              &=\mu(M(\tilde V)^{-1})=\mu(\tilde V M)=\mu(\tilde V)=\nu(V).
\end{align*}
According to \cite[Theorem 1.6]{GorOh},
\[
\lim_{T\to\infty}
\frac{|(\Gamma\cap G_T)\cap\Omega_1\overline{A^+}M\Omega_2|\,\mathrm{vol}(G/\Gamma)}{\mathrm{vol}(G_T)}
=\nu(U)\nu(V).
\]
Notice that
\[
\frac{|(\Gamma\cap G_T)\setminus \Gamma^{\theta}_T|+|\Gamma^{\theta}_T|}{|\Gamma\cap G_T|}=1.
\]
According  to  \cite[Proposition 3.3 and its proof]{BLP}
\[
	\lim_{T\to\infty}\frac{|(\Gamma\cap G_T)\setminus \Gamma^{\theta}_T|}{|\Gamma\cap G_T|}=0.
\]
Finally, according to \cite{EskMcM}
\begin{equation}\label{equation: EskMcM}
	|\Gamma\cap G_T|\cdot \mathrm{vol}(G/\Gamma)\sim \mathrm{vol}(G_T).
\end{equation}
Combining these facts with the preceding normalized limit proves the lemma.

<!-- lemma-2: body_md -->
\label{lemma: behavior of normalized smooth coefficients} With the notation as above, let $G$ be a connected non-compact semi-simple Lie group with finite center. We fix a left $K$-invariant Riemannian metric  $d$ on $B=K/M$. For $b\in B$ and $r>0$, we denote 
$$
B(b,r)=\{a\in B: d(a,b)<r\}
$$
the open ball of center $b$ and radius $r$.
 Let $\varphi,\psi:B\to\mathbb C$ be bounded Borel functions. For any $g\in KA^{+}K$,
we define 
$$
R_{\varphi,\psi}(g)=\frac{\langle\pi(g)\varphi,\psi\rangle}{\Xi(g)}-\varphi\left({\check{g}}\right)\overline{\psi({\hat g})}.
$$ 
The following statements are true.
\begin{enumerate}

\item
\begin{align*}
R_{\varphi,\psi}(g)&=\frac{\langle\pi(g)\varphi,\psi-\psi({\hat g})\mathbf{#1}_B\rangle}{\Xi(g)}\\
&+\overline{\psi({\hat g})}\frac{\left\langle\varphi-\varphi\left({\hat {g^{-1}}}\right)\mathbf{#1}_B,\pi(g^{-1})\mathbf{#1}_B\right\rangle}{\Xi(g)},
\end{align*}

\item
\begin{align*}
\left|R_{\varphi,\psi}(g)\right|&\leq\|\varphi\|_{\infty}\mathcal{P}_0|\psi-\psi({\hat g})\mathbf{#1}_B|(g)\\
&+|\psi({\hat g})|\mathcal{P}_0\left|\varphi-\varphi\left({\check{g}}\right)\mathbf{#1}_B\right|(g^{-1}).
\end{align*}

\item
\begin{align*}
\mathcal{P}_0|\psi-\psi(\hat{g})\mathbf{#1}_B|(g)&\leq\sup_{b\in B(\hat g,r)}|\psi(b)-\psi(\hat{g})|\\
&+\sup_{b\in B}|\psi(b)-\psi(\hat{g})|\mathcal{P}_0\mathbf{#1}_{B\setminus B(\hat g,r)}(g).
\end{align*}

\item Assume $\theta>0$ is small enough so that $\overline{\frak a^{\theta}}\subset {\frak a}^{+}\cup\{0\}$. If $\psi$ is continuous  then
$$
\lim_{T\to\infty}\sup_{g\in (KA^{\theta}K)\setminus G_{T}}\mathcal{P}_0|\psi-\psi(\hat{g})\mathbf{#1}_B|(g)=0.
$$

\item Assume $\theta>0$ is small enough so that $\overline{\frak a^{\theta}}\subset {\frak a}^{+}\cup\{0\}$. If $\varphi$ and $\psi$ are continuous  then
$$
\lim_{T\to\infty}\sup_{g\in (KA^{\theta}K)\setminus G_{T}}\left|R_{\varphi,\psi}(g)\right|=0.
$$
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
In order to prove the first statement we compute:
\begin{align*}
\frac{\langle\pi(g)\varphi,\psi\rangle}{\Xi(g)}-\varphi\left({\check{g}}\right)
\overline{\psi({\hat g})}&=\frac{\langle\pi(g)\varphi,\psi-\psi({\hat g})\mathbf{#1}_B+\psi({\hat g})\mathbf{#1}_B\rangle}{\Xi(g)}\\
&\,\,\,\,\,\,\,\,-\varphi\left({\check{g}}\right)\overline{\psi({\hat g})}\\
&=\frac{\langle\pi(g)\varphi,\psi-\psi({\hat g})\mathbf{#1}_B\rangle}{\Xi(g)}+\overline{\psi({\hat g})}\frac{\langle\pi(g)\varphi,\mathbf{#1}_B\rangle}{\Xi(g)}\\
&\,\,\,\,\,\,\,\,-\varphi\left({\check{g}}\right)\overline{\psi({\hat g})}\\
&=\frac{\langle\pi(g)\varphi,\psi-\psi({\hat g})\mathbf{#1}_B\rangle}{\Xi(g)}+\overline{\psi({\hat g})}\frac{\langle\varphi,\pi(g^{-1})\mathbf{#1}_B\rangle}{\Xi(g)}\\
&\,\,\,\,\,\,\,\,-\varphi\left({\check{g}}\right)\overline{\psi({\hat g})}\frac{\langle\mathbf{#1}_B,\pi(g^{-1})\mathbf{#1}_B\rangle}{\Xi(g)}\\
&=\frac{\langle\pi(g)\varphi,\psi-\psi({\hat g})\mathbf{#1}_B\rangle}{\Xi(g)}\\
&\,\,\,\,\,\,\,\,+\overline{\psi({\hat g})}\frac{\left\langle\varphi-\varphi\left({\check{g}}\right)\mathbf{#1}_B,\pi(g^{-1})\mathbf{#1}_B\right\rangle}{\Xi(g)}.
\end{align*}
Taking norms proves the next statement:
\begin{align*}
\left|\frac{\langle\pi(g)\varphi,\psi\rangle}{\Xi(g)}-\varphi\left({\check{g}}\right)
\overline{\psi({\hat g})}\right|&\leq\|\varphi\|_{\infty}\frac{\langle\pi(g)\mathbf{#1}_B,|\psi-\psi({\hat g})\mathbf{#1}_B|\rangle}{\Xi(g)}\\
&\,\,\,\,\,\,\,\,+|\psi({\hat g})|\frac{\langle|\varphi-\varphi\left({\check{g}}\right)\mathbf{#1}_B|,\pi(g^{-1})\mathbf{#1}_B\rangle}{\Xi(g)}\\
&=\|\varphi\|_{\infty}\mathcal{P}_0|\psi-\psi({\hat g})\mathbf{#1}_B|(g)\\
&\,\,\,\,\,\,\,\,+|\psi({\hat g})|\mathcal{P}_0|\varphi-\varphi\left({\check{g}}\right)\mathbf{#1}_B|(g^{-1}).
\end{align*}
In order to prove the third statement we define
$$
K(g,b)=\frac{\sqrt{\frac{dg_{*}\nu}{d\nu}(b)}}{\Xi(g)}.
$$
Notice that $K(g,b)>0$ and
$\int_{B}K(g,b)d\nu(b)=1$.
We have
\begin{align*}
\mathcal{P}_0|\psi-\psi(\hat{g})\mathbf{#1}_B|(g)&=\int_{B}K(g,b)|\psi(b)-\psi(\hat{g})|d\nu(b)\\
&=\int_{B(\hat{g},r)}K(g,b)|\psi(b)-\psi(\hat{g})|d\nu(b)\\
&+\int_{B\setminus B(\hat g,r)}K(g,b)|\psi(b)-\psi(\hat{g})|d\nu(b)\\
&\leq\sup_{b\in B(\hat g,r)}|\psi(b)-\psi(\hat{g})|\\
&+\sup_{b\in B}|\psi(b)-\psi(\hat{g})|\mathcal{P}_0\mathbf{#1}_{B\setminus B(\hat g,r)}(g).
\end{align*}
To prove the next statement, notice first that according to \cite[Lemma 3.2]{BLP}, for each $r>0$ there exists a function $h_r:[0,\infty[\to[0,\infty[$ with the following properties:
\begin{enumerate}
	\item $h_r$ is non-increasing,
	\item $\lim_{s\to\infty}h_r(s)=0,$
	\item $\forall g\in KA^{\theta}K,$
\[
	\mathcal{P}_0\mathbf{#1}_{\{b\in B :\, d(b,\hat{g})\geq r\}}(g)\leq h_r(L(g)).
\]
\end{enumerate}
where $d$ and $\theta$ are as specified above. If $\epsilon>0$ is given, as $\psi$ is uniformly continuous on the compact space $B$, there exists $r>0$ small enough so that for any $g\in KA^{+}K$,
$$
\sup_{b\in B(\hat g,r)}|\psi(b)-\psi(\hat{g})|<\epsilon/2.
$$
Choose $T>0$ sufficiently large that 
$$
2\|\psi\|_\infty h_r(T)<\epsilon/2.
$$
 Applying the previously proved inequality stated in  the lemma, we deduce that
for all $T'\geq T$, for all $g\in(KA^{\theta}K)\setminus G_{T'}$
$$
\mathcal{P}_0|\psi-\psi(\hat{g})\mathbf{#1}_B|(g)<\epsilon.
$$
As already mentioned, the opposition involution preserves $\frak a^\theta$, moreover, since $L(g)=L(g^{-1})$, statements (2) and (4), applied to $g$ and $g^{-1}$, prove (5).

\label{lemma: balls and coronae} Let $X$ be a set and $(B_{T})_{T>0}$ be an increasing family of subset of $X$; that is, if $S<T$, then $B_{S}\subset B_{T}\subset X$. We assume each set $B_{T}$ is finite (and non-empty). Suppose
$$
\lim_{T\to\infty}\frac{|B_{T}|}{|B_{T+1}|}=c<1.
$$ 
Let $f:X\to\mathbb C$ be a function.
Then
$$
\lim_{T\to\infty}\frac{1}{|B_{T}|}\sum_{x\in B_{T}}f(x)=l,
$$ 
if and only if
$$
\lim_{T\to\infty}\frac{1}{|B_{T+1}\setminus B_{T}|}\sum_{x\in B_{T+1}\setminus B_{T}}f(x)=l.
$$
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We have:
\begin{align*}
\frac{1}{|B_{T+1}|}\sum_{x\in B_{T+1}}f(x)&=\frac{|B_{T}|}{|B_{T+1}|}\left(\frac{1}{|B_{T}|}\sum_{x\in B_{T}}f(x)\right)\\
&+\frac{|B_{T+1}\setminus B_{T}|}{|B_{T+1}|}\left(\frac{1}{|B_{T+1}\setminus B_{T}|}\sum_{x\in B_{T+1}\setminus B_{T}}f(x)\right).
\end{align*}
As $1-c>0$ this proves the lemma.

\label{lemma: Schur on theta-coronae} With the notation as above, let $\theta>0$ be small enough so that $\overline{\frak a^{\theta}}\subset {\frak a}^{+}\cup\{0\}$. For any continuous functions $\varphi,\varphi',\psi,\psi'$ on $G/P$,

<!-- proof-4: body_md -->

$$

$$
We define the function
$$
F:G/P\times G/P\to\mathbb C
$$
$$
(a,b)\mapsto\psi'(a)\overline{\psi(a)}\varphi(b)\overline{\varphi'(b)},
$$
and we obtain, for any $g\in KA^{+}K$,
\begin{align*}
\frac{\langle\pi(g)\varphi,\psi\rangle\overline{\langle\pi(g)\varphi',\psi'\rangle}}{\Xi^{2}(g)}&=F(\hat{g},\check{g})\\

&+R_{\varphi,\psi}(g)\overline{R_{\varphi',\psi'}(g)}.
\end{align*}
As the function $F$ is continuous, we may apply Proposition \ref{prop: equidistribution away from the walls}  and Lemma \ref{lemma: balls and coronae} to deduce that

\begin{align*}

\end{align*}
tend to zero if $T$ tends to infinity.

\begin{align*}
&\frac{1}{|E_{T}|}\sum_{\gamma\in E_{T}}\frac{\langle\pi(\gamma)\varphi,\psi\rangle\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi^{2}(\gamma)}\\
&=\frac{|\Gamma^{\theta}_{T}|}{|E_{T}|}\left(\frac{1}{|\Gamma^{\theta}_{T}|}\sum_{\gamma\in \Gamma^{\theta}_{T}}\frac{\langle\pi(\gamma)\varphi,\psi\rangle\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi^{2}(\gamma)}\right)\\
&+\frac{1}{|E_T|}\sum_{\gamma\in E_T\setminus\Gamma_T^\theta}
\frac{\langle\pi(\gamma)\varphi,\psi\rangle\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi^2(\gamma)}.
\end{align*}		
On one hand, there is a constant $c>0$ such that, for any $\theta>0$, 
$$
|\Gamma^{\theta}_{T}|\sim|\Gamma_{T}|\sim c T^{(r-1)/2}\cdot e^{\delta T},
$$
on the other hand, positivity of the kernel defining $\pi$ gives
\[
|\langle\pi(g)\varphi,\psi\rangle|
\leq\|\varphi\|_\infty\|\psi\|_\infty\Xi(g).
\]
Consequently, the exceptional contribution satisfies
\begin{align*}
&\left|\frac{1}{|E_T|}\sum_{\gamma\in E_T\setminus\Gamma_T^\theta}
\frac{\langle\pi(\gamma)\varphi,\psi\rangle
\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi(\gamma)^2}\right|\\
&\qquad\leq\frac{|E_T\setminus\Gamma_T^\theta|}{|E_T|}
\|\varphi\|_\infty\|\psi\|_\infty
\|\varphi'\|_\infty\|\psi'\|_\infty\longrightarrow0.
\end{align*}
This estimate remains valid when $E_T\setminus\Gamma_T^\theta$ is empty.
Hence, taking limits, and applying Lemma \ref{lemma: Schur on theta-coronae} and Lemma \ref{lemma: balls and coronae}, we obtain:
\begin{align*}
&\lim_{T\to\infty}\frac{1}{|E_{T}|}\sum_{\gamma\in E_{T}}\frac{\langle\pi(\gamma)\varphi,\psi\rangle\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi^{2}(\gamma)}\\
&=\lim_{T\to\infty}\frac{1}{|\Gamma^{\theta}_{T}|}\sum_{\gamma\in \Gamma^{\theta}_{T}}\frac{\langle\pi(\gamma)\varphi,\psi\rangle\overline{\langle\pi(\gamma)\varphi',\psi'\rangle}}{\Xi^{2}(\gamma)}\\
&=\langle \varphi,\varphi'\rangle\overline{\langle\psi,\psi'\rangle}.
\end{align*}
This proves the first 	statement of the theorem. The second statement follows from the first, thanks to Lemma \ref{lemma: balls and coronae}.

<!-- lemma-5: body_md -->
\label{lemma: HC}(Asymptotic behavior of the Harish-Chandra function in rank $1$.) Let $G$ be a connected semi-simple Lie group with finite center. Assume $\mbox{rank}_{\mathbb R}G=1$. There is a constant $c>0$ such that for any $g\in G$
$$
\Xi(g)\sim c\cdot L(g)e^{-\frac{\delta}{2}L(g)}
$$
when $L(g)$ goes to infinity.
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Choose the unit vector $H_{max}\in\frak a^+$ and let $\alpha$ be the unique positive indivisible root. By \cite[page 373]{Ank}, there is $c_2>0$ such that
\[
\Xi(\exp(tH_{max}))\sim c_2\,t\alpha(H_{max})e^{-t\rho(H_{max})}
\qquad(t\to\infty).
\]
The Cartan decomposition reads $G=K\overline{\exp\left(\frak a^+\right)}K$. Thus every $g\in G$ can be written
\[
g=k_1\exp(tH_{max})k_2,\qquad k_1,k_2\in K,\quad t\geq0.
\]
Since $k_2$ fixes $x_0$ and $k_1$ acts isometrically and fixes $x_0$, we have
\[
L(g)=d_X(\exp(tH_{max})x_0,x_0)=t.
\]
The $K$-bi-invariance of $\Xi$ gives $\Xi(g)=\Xi(\exp(tH_{max}))$. Finally, $\delta=2\rho(H_{max})$, so the assertion follows with $c=c_2\alpha(H_{max})>0$.

<!-- lemma-6: body_md -->
\label{lemma: L_{G} and d_{G}} Let $G$ be a non compact connected semi-simple Lie group with finite center. Using the notation introduced above, let $d_{G}$ be the left-invariant Riemannian metric on $G$, defined with the help of the Killing form and the chosen Cartan decomposition, and let $L$ be the length function defined with the help of the symmetric space $X=G/K$ associated to $G$. For all $g\in G$,
$$-\emph{diameter}(K)+d_G(e,g)\leq L(g)\leq d_{G}(e,g).$$
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
The canonical projection
$$p:G\to G/K=X$$
defines a right-$K$ principal bundle. The map $p$ is $G$-equivariant and is a Riemannian submersion relative to the Riemannian structures  associated with $d_{G}$ and $d_{X}$ (recall that $d_{G}$ is $G$-invariant on the left and $K$-invariant on the right). A geodesic in $G$ from $e$ to $g$
projects under $p$ to a path of no greater length  in  $X$ between $x_{0}$ and $gx_{0}$. This proves that $L(g)\leq d_G(e,g)$. Suppose $c$ is an arc length parametrized geodesic path in $X$ between $x_{0}$ and $gx_{0}$. The horizontal subspaces of the Riemannian submersion $p$ define a connection on the principal right $K$-bundle $G$. Let $\tilde{c}$ be the unique horizontal lift of $c$ starting at $e$. It is parametrized by its arc length and has the same length as $c$. The end point $\tilde{c}(L(g))$ differs from $g$ by an element of $K$, that is there exists $k\in K$ such that $\tilde{c}(L(g))k=g$, hence: 
$$d_{G}(e,g)\leq \mbox{length}(\tilde{c})+d_{G}(\tilde{c}(L(g)),\tilde{c}(L(g))k)=L(g)+d_{G}(e,k).$$
This proves that  
$$-\mbox{diameter}(K)+d_G(e,g)\leq L(g).$$
