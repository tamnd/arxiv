<!-- proposition-1: body_md -->
Scheme VA-I preserves
\begin{itemize}
    \item the total modified energy $\tilde{\mathcal{E}}_T^n= \frac{1}{2}\sum_{p=1}^{N_p} \omega_p (v_p^n)^2 + (r^n)^2-C_0$ for all $n\in \mathbb{N}$;
    \item the Gauss law $E_{i+1/2}^{n}-E_{i-1/2}^{n}=\Delta x (\rho_i^n-1)$ for all $n\in \mathbb{N}$.
\end{itemize}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For the kinetic energy, we multiply \eqref{VA1_v} by $\sum_{p=1}^{N_p} v_p^{n+1/2}$ to obtain
\begin{align}
    \mathcal{E}_K^{n+1}-\mathcal{E}_K^n&=\frac{1}{2}\sum_{p=1}^{N_p} \omega_p((v_p^{n+1})^2-(v_p^n)^2)=\sum_{i=0}^{N_x-1}\sum_{p=1}^{N_p}\omega_p v_p^{n+1/2} \Delta t E_{i+1/2}^{n+1/2} S^m(x_p^{n+1/2}-x_{i+1/2}).
\end{align}
For the  modified electric energy, we multiply \eqref{VA1_r} by $(r^n+r^{n+1})$ to obtain
\begin{align}
    \tilde{\mathcal{E}}_E^{n+1}-\tilde{\mathcal{E}}_E^n&=(r^{n+1})^2-(r^n)^2=-\sum_{i=0}^{N_x-1}\sum_{p=1}^{N_p}\omega_p v_p^{n+1/2} \Delta t E_{i+1/2}^{n+1/2} S^m(x_p^{n+1/2}-x_{i+1/2}).
\end{align}
Add both, we see that the total modified energy $\tilde{\mathcal{E}}_T^{n}=\mathcal{E}_K^n+\tilde{\mathcal{E}}_E^n$ is preserved.

To show the charge conservation, we assume that $\rho_i(t), x_p(t)$ are time-continuous functions satisfying $$\rho_i(t^n)=\rho_i^n,\quad \rho_i(t^{n+1})=\rho_i^{n+1},\quad x_p(t^n) = x_p^n, \quad x_p(t^{n+1})=x_p^{n+1},$$ 
then we have
\begin{align}
    \label{Charge1}
  \frac{\rho_i^{n+1}-\rho_i^{n}}{ \Delta t} &= \frac{1}{ \Delta t}\int_{t^n}^{t^{n+1}}\frac{\,\mathrm{d}\rho_i(t)}{\,\mathrm{d}t}\,\mathrm{d}t \nonumber \\
 &=\frac{1}{\Delta x \Delta t}\int_{t^n}^{t^{n+1}}\sum_{p=1}^{N_p} \omega_p (-\dot{x}_p(t)) (S^m)'(x_i-x_p(t)) \,\mathrm{d}t \nonumber\\
 &=\frac{1}{(\Delta x)^2\Delta t}\int_{t^n}^{t^{n+1}}\sum_{p=1}^{N_p} \omega_p \dot{x}_p(t)\Big(S^{m-1}(x_{i-1/2}-x_p(t))-S^{m-1}(x_{i+1/2}-x_p(t))\Big) \,\mathrm{d}t \nonumber\\
&=\frac{1}{(\Delta x)^2\Delta t} \sum_{p=1}^{N_p} \int_{x^n_p}^{x^{n+1}_p} \omega_p \Big(S^{m-1}(x_{i-1/2}-x_p)-S^{m-1}(x_{i+1/2}-x_p)\Big)\,\mathrm{d}x_p\nonumber\\
&=\nonumber\frac{-1}{(\Delta x)^2\Delta t} \sum_{p=1}^{N_p} \omega_p \left(\int_{x_{i-1/2}-x^n_p}^{x_{i-1/2}-x^{n+1}_p}S^{m-1}(y)\,\mathrm{d}y-\int_{x_{i+1/2}-x^n_p}^{x_{i+1/2}-x^{n+1}_p}S^{m-1}(y)\,\mathrm{d}y\right) \nonumber\\
    &=\frac{1}{\Delta x \Delta t}\sum_{p=1}^{N_p} \omega_p [\Phi^{m-1}(x_{i-1/2}-x_p^n)-\Phi^{m-1}(x_{i-1/2}-x_p^{n+1})\nonumber\\& \quad -\Phi^{m-1}(x_{i+1/2}-x_p^n)+\Phi^{m-1}(x_{i+1/2}-x_p^{n+1})]\nonumber\\
&=\nonumber\frac{\bar{J}_{i-1/2}-\bar{J}_{i+1/2}}{\Delta x}=\frac{1}{\Delta x}\left(\frac{E^{n+1}_{i+1/2}-E^{n+1}_{i-1/2}}{\Delta t}-\frac{E^{n}_{i+1/2}-E^{n}_{i-1/2}}{\Delta t}\right),
\end{align}
where we used \eqref{SmSm-1} in the third line and \eqref{Sm-1_prim} in the second last line above. Therefore, if the Gauss law $E_{i+1/2}^{n}-E_{i-1/2}^{n}=\Delta x (\rho_i^n-1)$ is satisfied at time $t^n$, then it is also satisfied at $t^{n+1}$, which means the charge is preserved.
<!-- end proof-1 -->
<!-- proposition-2: body_md -->
Scheme VA-II preserves
    \begin{itemize}
    \item the total modified energy $\tilde{\mathcal{E}}_T^n= \frac{1}{2}\sum_{p=1}^{N_p} \omega_p (v_p^n)^2 + (r^n)^2-C_0$ for all $n\in \mathbb{N}$;
    \item the Gauss law $E_{i+1/2}^{n}-E_{i-1/2}^{n}=\Delta x (\rho_i^n-1)$ for all $n\in \mathbb{N}$.
\end{itemize}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
For the energy conservation, since $v_p$ and $r$ are not updated in $\varphi_{\Delta t}^{[1]}$ and $\varphi_{\Delta t}^{[2]}$, we only need to check if $\varphi_{\Delta t}^{[3]}$ preserves the total modified energy. Indeed, multiplying \eqref{eq:vplus} by $v_p^{\text{mid}}$ and multiplying \eqref{eq:rplus} by $r^{\text{mid}}$ yield, respectively,
\begin{align}
    \mathcal{E}_K^+-\mathcal{E}_K^-&=\frac{1}{2}\sum_{p=1}^{N_p} \omega_p((v_p^{+})^2-(v_p^-)^2)=\sum_{i=0}^{N_x-1}\sum_{p=1}^{N_p}\omega_p v_p^\text{mid}\Delta t r^\text{mid} g_{i+1/2}^- S^m(x_p^--x_{i+1/2}),\\
        \tilde{\mathcal{E}}_E^+-\tilde{\mathcal{E}}_E^-&=(r^{+})^2-(r^-)^2=-\sum_{i=0}^{N_x-1}\sum_{p=1}^{N_p}\omega_p v_p^\text{mid}\Delta t r^\text{mid} g_{i+1/2}^- S^m(x_p^--x_{i+1/2}).
\end{align}
Adding the two equations gives $\tilde{\mathcal{E}}_T^+ = \tilde{\mathcal{E}}_T^-$, so the total modified energy is preserved.

Since the scheme VA-II shares the same $E$ update as the scheme VA-I, charge conservation can be proved exactly the same as before.
<!-- end proof-2 -->
<!-- remark-1: body_md -->
The two schemes, VA-I and VA-II, introduced above differ significantly. VA-I updates the original variables $(x_p,v_p,E_{i+1/2},r)$, but the update of $r$ is essentially decoupled from the rest of the scheme, whereas VA-II updates the transformed  variables $(x_p,v_p,g_{i+1/2}),r$, but the update of $E$ is decoupled from the main loop.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
Other time integrators can be used for the approximation of the semi-discretized system 
\begin{align*}
\begin{cases}
         \dot{x}_p(t) = v_p(t),\\
         \dot{v}_p(t) = E(t, x_p(t)) = r(t) g(t, x_p(t)), \\
         \dot{E}_{i+1/2}(t) = -{J}_{i+1/2}(t),\\
         \dot{r}(t) = -\frac{1}{2} \sum_{i=0}^{N_x} g_{i+1/2}(t) J_{i+1/2}(t) \Delta x. 
\end{cases}
\end{align*}
Indeed, to ensure the preservation of both charge and modified energy, the following splitting can be used  
 \begin{align}
 \label{va0_1}
     \varphi^{[1]}_{t} \quad &\begin{cases}
         \dot{x}_p(t) = v_p(t),\\
         \dot{E}_{i+1/2}(t) = -{J}_{i+1/2}(t),
     \end{cases}\\
     \label{va0_2}
     \varphi^{[2]}_t \quad &\begin{cases}
         \dot{v}_p(t) = r g_p(t),\\
         \displaystyle \dot{r}(t)
         =-\frac{1}{2}\sum_i g_{i+1/2}(t)J_{i+1/2}(t)\Delta x.  
     \end{cases}
 \end{align}
 To get second order accuracy in time, we use a Strang splitting 
 $\varphi^{[1]}_{\Delta t/2}\circ \varphi^{[2]}_{\Delta t}\circ \varphi^{[1]}_{\Delta t/2}$ combined with a midpoint scheme for \eqref{va0_2}. More precisely, for $\varphi^{[1]}_{\Delta t/2}$, we use the charge preserving technique 
 \begin{eqnarray*}
 x^+_p &=& x^-_p + \frac{\Delta t}{2} v_p^-, \;\;\;\; 
 {E}_{i+1/2}^+ = {E}_{i+1/2}^- -\frac{\Delta t}{2} \bar{J}_{i+1/2}^{\textrm{mid}},\nonumber\\
 \bar{J}_{i+1/2}^{\textrm{mid}} &=& \frac{1}{\Delta t}\sum_{p=1}^{N_p} \omega_p \Big[ \Phi^{m-1}(x_{i+1/2}-x_p^-) - \Phi^{m-1}(x_{i+1/2}-x_p^+) \Big]\nonumber. 
 \end{eqnarray*}
As a second order approximation of $g_{i+1/2}(t)$ (and $g_p(t)$) in $\varphi^{[2]}_t$, we compute $g^\star_{i+1/2}$ which denotes any first order approximation of $g_{i+1/2}(t^{n+1/2}) = E_{i+1/2}(t^{n+1/2})/r(t^{n+1/2})$, for example 
\begin{itemize}
    \item $g^\star_{i+1/2} = \frac{E^{+}_{i+1/2}}{r^- -\frac{\Delta t}{4}\sum_{i=0}^{N_x-1} g^-_{i+1/2}J^-_{i+1/2}\Delta x}$, 
        \item $g^\star_{i+1/2}= \frac{3}{2}g^n_{i+1/2} - \frac{1}{2} g^{n-1}_{i+1/2}$,  in the spirit of  \cite{sav_schro, sav_xavier}, 
    \item $g^\star_{i+1/2} = g^-_{i+1/2} + \frac{\Delta t}{2}\dot{g}^-_{i+1/2}  = g^-_{i+1/2} + \frac{\Delta t}{2}\Big( -\frac{J^-_{i+1/2}}{r^-_{i+1/2}} +\frac{g^-_{i+1/2}}{2r^-_{i+1/2}}\sum_{i=0}^{N_x-1} g^-_{i+1/2} J^-_{i+1/2} \Delta x\Big)$, using \eqref{VA_SAV_rg_g}. 
\end{itemize}  
Once $g^\star_{i+1/2}$ is computed,  
$\varphi^{[2]}_{\Delta t}$ is updated using the midpoint method to ensure modified energy preservation 
 \begin{eqnarray*}
  v_p^{+} &=& v_p^- + \Delta t r^{\textrm{mid}} g^\star_p, \;\;\;\; 
 r^{+} = r^- - \frac{\Delta t}{2}\sum_{p=1}^{N_p}\sum_{i=0}^{N_x-1} \omega_p g_{p}^\star v_p^{\textrm{mid}} S^m(x_{i+1/2}-x_p^-)\Delta x,\nonumber 
 \end{eqnarray*}
with $r^{\textrm{mid}} = (r^-+r^{+})/2$ and $v_p^{\textrm{mid}} = (v_p^-+v_p^{+})/2$,  
 which is solved as in Subsection \ref{sub:233}. 

For this second order time integrator, one can also prove charge preservation thanks to the current $\bar{J}_{i+1/2}$ in $\varphi^{[1]}_{t}$ and preservation of the modified total energy thanks to the midpoint method in $\varphi^{[2]}_{t}$, as for VA-I and VA-II. Of course, other time discretization may be derived from the SAV semi-discretization. We restricted ourselves to the presentation of VA-I and VA-II within this manuscript.

    \begin{align}
    \frac{1}{2}\sum_{p=1}^{N_p} \omega_p \left[ (v_{x,p}^{n+1})^2 - (v_{x,p}^n)^2  \right]
    &= \sum_{p=1}^{N_p} \omega_p v_{x,p}^{n+1/2} \Delta t E_x^{n+1/2}(x_p^{n+1/2}) \nonumber\\
    &\quad +\sum_{p=1}^{N_p} \omega_p v_{x,p}^{n+1/2} v_{y,p}^{n+1/2} \Delta t B^{n+1/2}(x_p^{n+1/2}),\\
    \frac{1}{2}\sum_{p=1}^{N_p} \omega_p \left[ (v_{y,p}^{n+1})^2 - (v_{y,p}^n)^2  \right]
    &= \sum_{p=1}^{N_p} \omega_p v_{y,p}^{n+1/2} \Delta t E_y^{n+1/2}(x_p^{n+1/2}) \nonumber\\

    (r_y^{n+1})^2 - (r_y^n)^2  & = -\Delta t \sum_{p=1}^{N_p} \omega_p E_{y}^{n+1/2} (x_p^{n+1/2}) v_{y,p}^{n+1/2} \nonumber\\
    &\quad - \Delta t \sum_{i=0}^{N_x-1} E_{y,i}^{n+1/2} (B^{n+1/2}_{i+1/2}-B^{n+1/2}_{i-1/2}), \\
    \frac{1}{2}\sum_{i=0}^{N_x-1} \left( (B^{n+1}_{i+1/2})^2 - (B^{n}_{i+1/2})^2 \right) \Delta x

    (r_y^+)^2 - (r_y^-)^2 & = -\Delta t r_y^\text{mid} \sum_{p=1}^{N_p} \omega_p v_{y,p}^\text{mid} g_{y,p}^-.
\end{align*}
Summing these equalities leads to $\tilde{\mathcal{E}}_T^+=\tilde{\mathcal{E}}_T^-$ since the electromagnetic field $B$ is unchanged during this step. 

In $\varphi^{[4]}_{\Delta t}$, we multiply \eqref{VM2_4_vx} by $v_{x,p}^\text{mid}$ and \eqref{VM2_4_vy} by $v_{y,p}^\text{mid}$, their sum vanishes, which means the midpoint preserves the kinetic energy. \begin{align}

\end{align}
Adding the two equations leads to $\tilde{\mathcal{E}}^{n+1}_T=\tilde{\mathcal{E}}^{n}_T$ for the fourth subsystem. Therefore, we conclude the full scheme preserves the total modified energy.

The Gauss law is preserved in the system, and the proof is identical to the VA system.

The extension of the SAV formulation to the 3D3V VM system is straightforward. For
$\alpha\in\{x,y,z\}$, we introduce one SAV for each component of the
electric field,
\begin{equation}
    r_\alpha^2(t)=\frac{1}{2}\int_\Omega E_\alpha^2(t,\boldsymbol{x})
    \,\mathrm{d}\boldsymbol{x}+C_0,
    \qquad
    \dot r_\alpha(t)=\frac{1}{2r_\alpha(t)}\int_\Omega
    E_\alpha\big((\nabla\times\boldsymbol{B})_\alpha-J_\alpha\big)
    \,\mathrm{d}\boldsymbol{x}.
\end{equation}
The corresponding modified energy is
\begin{equation}
    \widetilde{\mathcal{E}}_T
    =\frac{1}{2}\sum_{p=1}^{N_p}\omega_p|\boldsymbol{v}_p|^2
    +\sum_{\alpha}r_\alpha^2-3C_0
    +\frac{1}{2}\int_\Omega|\boldsymbol{B}|^2\,\mathrm{d}\boldsymbol{x}.
\end{equation}

Scheme VM-I extends by using the three-dimensional predictor
$\boldsymbol{v}_p^*-\boldsymbol{v}_p^n=\frac{\Delta t}{2}
(\boldsymbol{E}_p^n+\boldsymbol{v}_p^*\times\boldsymbol{B}_p^n)$ and then
applying the same midpoint field, velocity, and SAV updates.
For Scheme VM-II, we set $g_\alpha=E_\alpha/r_\alpha$ and apply the same symmetric splitting.

For the current used for charge conservation, we can adopt the current deposition from Esirkepov \cite{esirkepov-2001}. On a Yee's lattice, the current
$\boldsymbol{\overline{J}}_h^{n+1/2}$ is constructed to satisfy the discrete Gauss law
\begin{equation}
    \frac{\rho_{i,j,k}^{n+1}-\rho_{i,j,k}^{n}}{\Delta t}
    +D_x\overline{J}_{x,i,j,k}^{n+1/2}
    +D_y\overline{J}_{y,i,j,k}^{n+1/2}
    +D_z\overline{J}_{z,i,j,k}^{n+1/2}=0,
\end{equation}
where $D_\alpha$ is the directional finite-difference operator. 

Esirkepov's method writes 
\begin{align}
    \rho_p^{n+1}-\rho_p^n=W_{x,p}+W_{y,p}+W_{z,p},\quad D_\alpha\overline{J}_{\alpha,p}=-W_{\alpha,p}/\Delta t,
\end{align}
where the three $W_{\alpha,p}$ contain symmetric mixed products of the one-dimensional
shape-function differences. Since the discrete
operators on Yee's lattice satisfy $\nabla_h\cdot(\nabla_h\times\boldsymbol{B}_h)=0$,
taking the discrete divergence of the Amp\`ere update propagates
the discrete Gauss law from time level $n$ to $n+1$.  

For Scheme VM-II, we introduce the scaled electric fields
$g_\alpha=E_\alpha/r_\alpha$, $\alpha\in\{x,y,z\}$. 
The splitting in Subsection 
\ref{subsec:VM-II} extends to 
the 3D-3V system as follows, 
with  $\alpha\in\{x,y,z\}$:
\begin{subequations}
\begin{align}
\varphi_{\Delta t}^{[1]}\quad &
\begin{cases}
\dot{\boldsymbol{x}}_p=\boldsymbol{v}_p,\\
r_\alpha\dot{g}_\alpha=-J_\alpha,
\end{cases}\\
\varphi_{\Delta t}^{[2]}\quad &
\begin{cases}
r_\alpha\dot{g}_\alpha
=\dfrac{1}{2}g_\alpha
\displaystyle\int_\Omega g_\alpha J_\alpha\,\mathrm{d}\boldsymbol{x},
\end{cases}\\
\varphi_{\Delta t}^{[3]}\quad &
\begin{cases}
\dot{\boldsymbol{v}}_p
=\displaystyle\sum_{\alpha\in\{x,y,z\}}
r_\alpha g_{\alpha,p}\boldsymbol{e}_\alpha,\\
\dot{r}_\alpha
=-\dfrac{1}{2}\displaystyle\int_\Omega
g_\alpha J_\alpha\,\mathrm{d}\boldsymbol{x},
\end{cases}\\
\varphi_{\Delta t}^{[4]}\quad &
\begin{cases}
\dot{\boldsymbol{v}}_p
=\boldsymbol{v}_p\times\boldsymbol{B}_p,\\
\dot{\boldsymbol{B}}
=-\nabla\times
\left(\displaystyle\sum_{\alpha\in\{x,y,z\}}
r_\alpha g_\alpha\boldsymbol{e}_\alpha\right),\\
\dot{r}_\alpha
=\dfrac{1}{2}\displaystyle\int_\Omega
g_\alpha(\nabla\times\boldsymbol{B})_\alpha
\,\mathrm{d}\boldsymbol{x},\\
r_\alpha\dot{g}_\alpha
=(\nabla\times\boldsymbol{B})_\alpha
-\dfrac{1}{2}g_\alpha
\displaystyle\int_\Omega
g_\alpha(\nabla\times\boldsymbol{B})_\alpha
\,\mathrm{d}\boldsymbol{x},
\end{cases}
\end{align}
\end{subequations}
where $\boldsymbol{e}_\alpha$ are the Cartesian unit vectors, and
$g_{\alpha,p}$ and $\boldsymbol{B}_p$ denote the fields interpolated to
particle positions. In each subsystem, the variables that are not updated
remain constant. A symmetric second-order composition is then
\begin{equation}
\varphi_{\Delta t/2}^{[1]}
\circ\varphi_{\Delta t/2}^{[2]}
\circ\varphi_{\Delta t/2}^{[3]}
\circ\varphi_{\Delta t}^{[4]}
\circ\varphi_{\Delta t/2}^{[3]}
\circ\varphi_{\Delta t/2}^{[2]}
\circ\varphi_{\Delta t/2}^{[1]}.
\end{equation}
