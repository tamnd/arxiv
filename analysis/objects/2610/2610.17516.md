\label{Thm: Aperiodic} Let $(X,\mu, \mathcal{B}, T)$ be an aperiodic probability preserving $\mathbb{Z}$ action. For any $N \in \mathbb{N}$ and a finite measurable partition $\mathcal{P} = \{P_1,P_2,\ldots,P_t\}$ of $X$, there exist $B_N, B_{N+1} \in \mathcal{B}$ such that
\[

<!-- lemma-2: body_md -->
\label{lem: Aperiodic partittion} 
$\left\{X_F:\ F\in\mathscr{K}\right\}$ is a partition of $X$ to $T$ invariant sets so that for each $ F\in\mathscr{K}$ the following properties are satisfied: 
\begin{itemize}
    \item For all $j\in [t]$, $\mu\left(X_F\cap P_j\right)>0$ if and only if $j\in F$. Consequently $\mathcal{P}_F:=\left\{X_F\cap P_j:\ j\in F\right\}$ is a partition of $X_F$.
    \item For $\mu$ almost all $x\in X_F$, for all $j\in F$, 
    \[
    \lim_{n\to\infty}\frac{\left|\left\{x,Tx,\ldots,T^{n-1}x\right\}\cap \left(X_F\cap P_j\right)\right|}{n}>0.
    \]
\end{itemize}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
For almost every $x\in X$, $F_x:=\{j\in[t]:\varphi(x)_j\neq 0\}$ is a nonempty subset of $[t]$, thus we have $x\in X_{F_x}$. Let $F,G \in\mathscr{K}$ such that $F\neq G$. Then without loss of generality there exists $j \in F$ such that $j \notin G$. Thus $\varphi(x)_j\neq 0$ for a.e $x\in X_F$, while $\varphi(x)_j=0$ for a.e $x\in X_G$, hence $X_F\cap X_G=\emptyset$. This shows that $\{X_F:F\in\mathscr K\}$ is a partition of $X$. The sets $X_F$ are $T$ invariant because $\varphi$ is $\mathcal{I}$ measurable  and $$X_F:=\bigcap_{j\in F}\{\varphi(x)_j\neq 0\}\cap \bigcap_{j\notin F}\{\varphi(x)_j=0\}.$$

Now fix $ F\in \mathscr{K}$ and note that for all $j\in [t]$, 
\begin{align*}
\mu\left(X_F\cap P_j\right)&= \int 1_{X_F}1_{P_j}d\mu,\\
&= \int \mathbb{E}\left(\left.1_{X_F}1_{P_j}\right|\mathcal{I}\right)d\mu=\int_{X_F} \mathbb{E}\left(\left.1_{P_j}\right|\mathcal{I}\right)d\mu, \ \text{since $1_{X_F}$ is $\mathcal{I}$-measurable.}
\end{align*}
The first item follows from this, the positivity of conditional expectation and the definition of $X_F$. The last claim follows from the pointwise ergodic theorem since for every $j\in F$, for $\mu$ almost every $x\in X_F$
\[
\lim_{n\to\infty}\frac{1}{n}\sum_{k=0}^{n-1}1_{P_j}(T^k x)=\mathbb{E}\left(\left. 1_{P_j}\right|\mathcal{I}\right)(x)=\varphi(x)_j>0.
\]

<!-- lemma-3: body_md -->
\label{lem: aperiodic tower 1}
Let $F\in\mathscr{K}$ and $M\in\mathbb{N}$. For every $A\subset X_F$, a $T$ invariant set with $\mu_F(A)>0$, there exists $S\subset A$ with $\mu_F(S)>0$ such that for $\mu_F$ almost all $x\in S$ whose return time to $S$ is $n(x)$, 
\[
\min_{Q\in\mathcal{P}_F}\left|\left\{x,Tx,...,T^{n(x)-1}x\right\}\cap Q\right|>M.
\]
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Define for $K\in\mathbb{N}$,
\[
A_K:=\left\{x\in A:\ \min_{Q\in\mathcal{P}_F}\sum_{j=0}^{K-1}1_Q\circ T^j(x)> M\right\}
\]
By \Cref{lem: Aperiodic partittion}, $A_K\uparrow A$ (mod $\mu$ null sets) and consequently there exists $K\in \mathbb{N}$ such that the set $A_K$ is of positive $\mu$ measure. Note that for all $x\in A_K$ and every $Q\in\mathcal{P}_F$,
\[
\left|\left\{x,\ldots,T^{K-1}x\right\}\cap Q\right|>M.
\]
 By \cite[Lemma 1]{MR9703} (see also the proof of the Rokhlin lemma in \cite{MR97489}) there exists $S\subset A_K$ of positive $\mu$ measure such that for all $1\leq j\leq K-1$, $S\cap T^{j}S=\emptyset$, in other words for all $x\in S$ the return time to $S$ is larger or equal to $K$. The set $S$ is the desired set.

<!-- cor-2: body_md -->
\label{cor: Aperiodic tower}
Let $F\in\mathscr{K}$. For all  $M\in\mathbb{N}$, there exists a Kakutani tower for $\left(X_F,\mathcal{B}_F,\mu_F,T\right)$ with base $S=\bigsqcup_{j=Mt}^\infty S_j$ such that for every $x\in S_j$, 
\[
\min_{Q\in\mathcal{P}_F}\left|\left\{x,Tx,\ldots,T^{j-1}x\right\}\cap Q\right|>M.
\]
<!-- end cor-2 -->
<!-- proof-3: body_md -->
By \cref{lem: aperiodic tower 1} there exists $C^{(1)}\subset X_F$ of positive measure such that for all $x\in C^{(1)}$ whose return time to $C^{(1)}$ is $n(x)$, 
\[
\min_{Q\in\mathcal{P}_F}\left|\left\{x,Tx,...,T^{n(x)-1}x\right\}\cap Q\right|>M.
\]

If $\left[C^{(1)}\right]_T$ is not of full, consider now $A^{(1)}:=X_F\setminus \left[C^{(1)}\right]_T$ and note that it is a $T$ invariant set. If $A^{(1)}$ is a null set, then $S^{(1)}=C^{(1)}$ is the desired base set. Otherwise applying \cref{lem: aperiodic tower 1}, once again we obtain $C^{(2)}\subset A^{(1)}$ of positive measure such that for all $x\in C^{(2)}$ whose return time to $C^{(2)}$ is $n(x)$, 
\[
\min_{Q\in\mathcal{P}_F}\left|\left\{x,Tx,...,T^{n(x)-1}x\right\}\cap Q\right|>M.
\]
Now write $S^{(2)}:=C^{(1)} \bigsqcup C^{(2)}$ and note that by the way we chose $C^{(1)}$ and $C^{(2)}$ the return time of any $x\in S^{(2)}$ is the same as its return time to $C^{(1)}$ or $C^{(2)}$ according to which set it belongs, so $n(x)$ is its return time to $S^{(2)}$. Again if $\left[S^{(2)}\right]_T$ is of full $\mu_F$ measure, we are done.

Continuing in this way, by Zorn's lemma we obtain a maximal collection of sets $\left(C^{(\alpha)}\right)_{\alpha\in \mathcal{A}}$ such that the following holds:
\begin{itemize}
    
    \item For all $\alpha\in \mathcal{A}$,  $C^{(\alpha)}$ is a measurable subset of $X_F$ and 
     \[
\mu_F\left( C^{(\alpha)}\right)>0.
\]
    
    \item For every $\alpha_1\neq \alpha_2$, the $T$ saturations of $C^{(\alpha_1)}$ and $C^{(\alpha_2)}$ are disjoint. 
    \item For all $\alpha\in \mathcal{A}$, for all $x\in C^{(\alpha)}$ whose return time to $C^{(\alpha)}$ is $n(x)$, 
\[
\min_{Q\in\mathcal{P}_F}\left|\left\{x,Tx,...,T^{n(x)-1}x\right\}\cap Q\right|>M.
\] 
\end{itemize} 
As the $T$ saturations of the $C^{(\alpha)}$'s are of positive $\mu$ measure and pairwise disjoint, we deduce that $\mathcal{A}$ is countable and write simply $\mathcal{A}=\mathbb{N}$ and $(C^{(n)})_n$. 

We claim that $S:=\cup_n C^{(n)}$ is the required base of a tower. Indeed, if $\mu_F\left([S]_T\right)<1$, we can find using \Cref{lem: aperiodic tower 1} a set $C'\in\mathcal{B}$ which satisfies $\mu_F(C')>0$, $C'\cap [S]_T=\emptyset$ and for almost every $x\in C'$ whose return time to $C'$ is $n(x)$ one has  
\[
\min_{Q\in\mathcal{P}_F}\left|\left\{x,Tx,...,T^{n(x)-1}x\right\}\cap Q\right|>M.
\]
As this is a contradiction for the maximality of the family, we deduce that $\mu_F([S]_T)=1$. In addition, for every $x\in S$ there exists a unique $n\in\mathbb{N}$ such that $x\in C^{(n)}$. As the $T$ saturations of the subsets in the maximal family are pairwise disjoint, we deduce that $n(x)$, the return time of $x$ to $S$ is the same as the return time of $x$ to $C^{(n)}$. This and the choice of $C^{(n)}$'s implies that for all $x\in S$ whose return time to $S$ is $n(x)$, one has 
\[
\min_{Q\in\mathcal{P}_F}\left|\left\{x,Tx,...,T^{n(x)-1}x\right\}\cap Q\right|>M,
\]
as desired.

<!-- proof-4: body_md -->
Let $F\in\mathscr{K}$. By \Cref{cor: Aperiodic tower}, the aperiodic dynamical system $\left(X_F,\mathcal{B}_F,\mu_F,T\right)$ satisfies the conditions of \cref{fact: Kalikow} with respect to its partition $\mathcal{P}_F$. Consequently there exists $B_{N,F},B_{N+1,F}\subset X_F$ satisfying 
\begin{itemize}
\item[(a)] $
\{T^j B_{t,F} :  0\leq j < t, t\in \{N,N+1\}\}$ 
is a partition of $X_F$ up-to measure zero.
\item[(b)] $B_{N,F}$ and $B_{N+1,F}$ are independent of $\mathcal{P}_F$.
\item[(c)] $\mu_F\left(\bigsqcup_{j=0}^{N-1}T^j B_{N,F}\right)=\mu_{F}\left(\bigsqcup_{j=0}^{N}T^j B_{N+1,F}\right)=\frac{1}{2}$. As $T$ preserves $\mu_F$ this implies that for every  $t\in\{N,N+1\}$, $\mu_F\left(B_{t,F}\right)=\frac{1}{2t}$.
\end{itemize}
We claim that $B_N:=\bigsqcup_{F\in\mathscr{K}}B_{N,F}$ and $B_{N+1}:=\bigsqcup_{F\in\mathscr{K}}B_{N+1,F}$ form the bases of an $N,N+1$ Alpern tower independent of $\mathcal{P}$. 

The fact that $\{T^j B_{t} :  0\leq j < t, t\in \{N,N+1\}\}$ are pairwise disjoint follows from the fact that for each $F\in\mathscr{K}$, $\{T^j B_{t,F} :  0\leq j < t, t\in \{N,N+1\}$
are pairwise disjoint subsets of $X_F$ and that if $F_1,F_2\in\mathscr{K}$ are distinct then  $X_{F_1}\cap X_{F_2}=\emptyset$. To see that they form a partition of $X$ (mod $\mu$) note that \footnote{all equalities are modulo null sets}
\begin{align*}
\bigsqcup_{t\in \{N,N+1\}}\bigsqcup_{j=0}^{t-1}T^{j}B_t&=\bigsqcup_{F\in\mathscr{K}}\bigsqcup_{t\in \{N,N+1\}}\bigsqcup_{j=0}^{t-1}T^{j}B_{t,F}\\
&=\bigsqcup_{F\in\mathscr{K}}X_F=X.
\end{align*}
To complete the proof we need to show that both $B_N$ and $B_{N+1}$ are independent of $\mathcal{P}$. Let $P_i \in \mathcal{P}$, then
\[
\mu(P_i \cap B_N) = \sum_{F \in \mathscr{K}} \mu(P_i \cap B_N(F)).
\]
For all $F\in \mathscr{K}$ such that $i\notin F$, $\mu\left(P_i\cap X_F\right)=0$. In addition, if $i\in F$, then $P_i\cap X_F\in\mathcal{P}_F$. Combining these two insights we get,
\begin{align*}
\mu(P_i \cap B_N) &= \sum_{F\in\mathscr{K} : i \in F} \mu(X_F) \cdot \dfrac{\mu\left(P_i \cap B_{N,F}\right)}{\mu(X_F)}\\
&= \sum_{F\in\mathscr{K} : i \in F} \mu(X_F) \cdot \mu_F\left(\left(P_i\cap X_F\right) \cap B_{N,F}\right). 
\end{align*} 

Recall that $\mathcal{P}_F$ is a partition of $X_F$ which is independent of $B_{N,F}$, hence we get
\begin{align}
    \mu(P_i \cap B_N) &= \sum_{F\in\mathscr{K} : i \in F} \mu(X_F) \cdot \mu_F\left(P_i\right) \mu_F\left(B_{N,F}\right)\nonumber\\
    &= \dfrac{1}{2N} \sum_{F\in\mathscr{K} : i \in F} \mu(X_F) \cdot \mu_F(P_i)\nonumber\\
    &=\dfrac{1}{2N} \sum_{F\in\mathscr{K} : i \in F} \mu(X_F\cap P_i)=\frac{\mu\left(P_i\right)}{2N}.\label{eq: 1}
\end{align} 
Here we used the fact that $\mu_F$ measure of the $B_{N,F}$  is $1/2N$.  A similar argument shows that
\begin{align}
\mu\left(B_N\right)&=\sum_{F\in\mathscr{K}}\mu\left(X_F\right)\mu_F\left(B_{N,F}\right)\nonumber\\
&=\frac{1}{2N}\sum_{F\in\mathscr{K}}\mu\left(X_F\right)=\frac{1}{2N}.\label{eq: 2}
\end{align}



We deduce from \eqref{eq: 1} and \eqref{eq: 2}  that for all $1\leq i\leq t$,
\[
\mu(P_i \cap B_N)=\mu\left(P_i\right)\mu\left(B_N\right),
\]
in other words, $B_N$ is independent of $\mathcal{P}$. A similar argument shows that $B_{N+1}$ is independent of $\mathcal{P}$ as well.

\label{prop: copying lemma}
Let $(X,\mathcal{B},\mu,T)$ be an aperiodic invertible probability measure preserving transformation and $\xi$ a finite measurable partition of $X$. For every finite set $A$ and $X_1, X_2, .., X_l$ a

For every aperiodic, probability preserving system $(X,\mathcal{B},\mu,T)$, there exists a square integrable $f:X\to \mathbb{Z}$ with $\int fd\mu=0$ which satisfies the lattice local central limit theorem. That is there exists $\sigma^2>0$ such that 
\[
\sup_{x\in\mathbb{Z}}\left|\sqrt{n}\mu(S_n(f)=x)-\frac{1}{\sqrt{2\pi\sigma^2}}\exp\left(-\frac{x^2}{2n\sigma^2}\right)\right|\xrightarrow[n\to\infty]{}0,
\]
where $S_n(f):=\sum_{k=0}^{n-1}f\circ T^k$.
