			\item The essential norm of the Hardy--composition operator $C_{\varphi}$ is given by

\label{therorem: characterization of Sp for Cphi}
		Let $\varphi$ be an analytic self-mapping of $\mathbb{D}$  and $0 < p < \infty$.
		\begin{enumerate}[label = \textup{(\alph*)}]
			\item The Hardy--composition operator $C_{\varphi}$ belongs to the Schatten $p$-class $S_p (H^ 2 (\mathbb{D}))$ if and only if
			\begin{align*}
				\int_{\mathbb{D}} \left[ \frac{N_{\varphi} (z)}{\log{(1 / |z|)}} \right]^ {\frac{p}{2}} \, \frac{\mathrm{d} A(z)}{( 1 - |z|^ 2 )^ 2} < +\infty,

			\end{align*}

		\begin{align*}
			N_{\tau \circ \varphi} (z) =
			\begin{cases}
				\log{\frac{1}{|(\tau \circ \varphi)^ {-1} (z)|}}, & 0 < |(\tau \circ \varphi)^ {-1} (z)| < 1,\vspace{2mm} \\
				0, & \textrm{otherwise}.
			\end{cases}
		\end{align*}
		Putting
		\begin{align*}
			\Phi = (\tau \circ \varphi)^ {-1} = \varphi^ {-1} \circ \delta \circ \widetilde{\psi} \circ \delta,
		\end{align*}
		we can then rewrite $N_{\tau \circ \varphi}$ as
		\begin{align*}
			N_{\tau \circ \varphi} (z) =
			\begin{cases}
				\log{\frac{1}{|\Phi (z)|}}, & 0 < |\Phi (z)| < 1,\vspace{2mm} \\
				0, & \textrm{otherwise}.
			\end{cases}
		\end{align*}
		Letting $\widehat{\mathbb{C}}$ denote the extended complex plane, i.e., $\widehat{\mathbb{C}}=\mathbb C\cup\{\infty\}$, then
		\begin{align*}
			\widetilde{\psi} (\widehat{\mathbb{C}} \backslash \overline{\mathbb{D}}) = \widehat{\mathbb{C}} \backslash\widetilde{\psi} (\overline{\mathbb{D}}) \supseteq \widehat{\mathbb{C}} \backslash\overline{\mathbb{D}},
		\end{align*}
		and hence
		\begin{align*}
			\Phi (\mathbb{D}) = (\varphi^ {-1} \circ \delta \circ \widetilde{\psi} \circ \delta) (\mathbb{D}) = (\varphi^ {-1} \circ \delta \circ \widetilde{\psi}) (\widehat{\mathbb{C}} \backslash \overline{\mathbb{D}}) \supseteq (\varphi^ {-1} \circ \delta) (\widehat{\mathbb{C}}\backslash\overline{\mathbb{D}}) = \varphi^ {-1} (\mathbb{D}) \supseteq \mathbb{D}.
		\end{align*}
		Now we  consider the following two cases.
		
		\textbf{Case 1.} $\Phi (\mathbb{D}) \supseteq \overline{\mathbb{D}}$. In this case, we have
		\begin{align*}
			\int_{\mathbb{D}} \left[ \frac{N_{\tau \circ \varphi} (z)}{\log{(1 / |z|)}} \right]^ {\frac{p}{2}} \frac{\mathrm{d} A (z)}{(1 - |z|^ 2)^ 2} = \int_{\Phi^ {-1} (\mathbb{D})} \left[ \frac{N_{\tau \circ \varphi} (z)}{\log{(1 / |z|)}} \right]^ {\frac{p}{2}} \frac{\mathrm{d} A (z)}{(1 - |z|^ 2)^ 2},
		\end{align*}
		where $p$ is any positive number.
		Since the compact set $\Phi^ {-1} \left( \overline{\mathbb{D}} \right)$ is contained in $\Phi^ {-1} (\Phi (\mathbb{D})) = \mathbb{D}$, there exists some $\rho \in (0, 1)$ so that $\Phi^ {-1} \left( \overline{\mathbb{D}} \right) \subseteq \rho \mathbb{D}$. It follows that
		\begin{align*}
			\int_{\mathbb{D}} \left[ \frac{N_{\tau \circ \varphi} (z)}{\log{(1 / |z|)}} \right]^ {\frac{p}{2}} \frac{\mathrm{d} A (z)}{(1 - |z|^ 2)^ 2} & \leqslant \frac{1}{[\log{(1 / \rho)}]^ {\frac{p}{2}}} \frac{1}{(1 - \rho^ 2)^ 2} \int_{\Phi^ {-1} (\mathbb{D})} [N_{\tau \circ \varphi} (z)]^ {\frac{p}{2}} \,\mathrm{d} A(z).
		\end{align*}
		By Littlewood's inequality (\ref{ineq3}), the integral $$\int_{\mathbb{D}} \big[N_{\tau \circ \varphi} (z)\big]^{\frac{p}{2}} \,\mathrm{d} A(z)$$ is finite. We conclude by Lemma \ref{lemma: linear fractional symbols} and Theorem \ref{therorem: characterization of Sp for Cphi} that $C_{\varphi} C_{\psi}^ *$ is in $S_p (H^ 2)$ for $0 < p < \infty$.
		
		\textbf{Case 2.} $\Phi (\mathbb{D}) \nsupseteq \overline{\mathbb{D}}$ (i.e., the boundary of $\Phi (\mathbb{D})$ is externally tangent to $\mathbb{T}$). Then we can find some point $\mathrm{e}^ {\mathrm{i} \theta} \in \mathbb{T}$ such that $\Phi (\mathrm{e}^ {\mathrm{i} \theta}) = \omega \in \mathbb{T}$. By the definition of $\Phi$, we have
		\begin{align*}
			\frac{1}{\varphi (\omega)} = \delta \circ \varphi (\omega) = \widetilde{\psi} \circ \delta (\mathrm{e}^ {\mathrm{i} \theta}) = \overline{\psi (\mathrm{e}^ {\mathrm{i} \theta})}.
		\end{align*}
		Observing that $|\psi (\mathrm{e}^ {\mathrm{i} \theta})\varphi (\omega)| = 1$ and recalling that $\varphi ( \overline{\mathbb{D}})$ and $\psi ( \overline{\mathbb{D}} )$ are both  contained in $\overline{\mathbb{D}}$, we must have $|\psi (\mathrm{e}^ {\mathrm{i} \theta})| = |\varphi (\omega)| = 1$. It follows that
		\begin{align*}
			\varphi (\omega) = \psi (\mathrm{e}^ {\mathrm{i} \theta}) \in \mathbb{T}.
		\end{align*}
		By \cite[Theorem 2.1]{CZ}, $C_{\varphi} C_{\psi}^ *$ is not compact, and in particular, it is not in any $S_p (H^ 2)$ for $0 < p < \infty$.
		
		Therefore, for each $0 < p < \infty$, $C_{\varphi} C_{\psi}^ * \notin S_p (H^ 2)$ if and only if $\Phi (\mathbb{D}) \nsupseteq \overline{\mathbb{D}}$. In the discussion of Case 2, we have shown that $\Phi (\mathbb{D}) \nsupseteq \overline{\mathbb{D}}$ leads to
		\begin{align}
			\varphi (\mathbb{T}) \cap \psi (\mathbb{T}) \cap \mathbb{T} \neq \varnothing, \label{ineq11}
		\end{align}
		which gives the sufficiency part of (a).



		Conversely, if (\ref{ineq11}) holds, then there are points $\xi, \zeta$ and $\omega_0$ on $\mathbb{T}$ such that
		\begin{align*}
			\varphi (\xi) = \psi (\zeta) = \omega_0,
		\end{align*}
		and a direct calculation then gives $\Phi (\zeta) = \xi$. Consequently, $\Phi (\mathbb{D}) \nsupseteq \overline{\mathbb{D}}$ and hence $C_{\varphi} C_{\psi}^ * \notin S_p (H^ 2)$, to obtain the necessity part of (a).


This finishes the proof of Theorem \ref{theorem: characterization of Sp for linear fractional symbols}.

<!-- coro-1: body_md -->
\label{corollary: characterization of Sp for linear fractional symbols}
		Suppose that $\varphi$ and $\psi$ are linear fractional maps from $\mathbb{D}$ into itself. Then for any $0 < p < \infty$:
		\begin{enumerate}[label = \textup{(\alph*)}]
			\item $C_{\varphi} C_{\psi}^ *$ is in $S_p (H^ 2)$ if and only if $C_{\varphi} C_{\psi}^ *$ is compact, and
			\item $C_{\psi}^ * C_{\varphi}$ is in $S_p (H^ 2)$ if and only if $C_{\psi}^ * C_{\varphi}$ is compact.
		\end{enumerate}

<!-- thm-12: body_md -->
\label{proposition: sufficient condition for Cpsi*Cphi in Sp}
		Suppose $\varphi$ and $\psi$ are analytic self-mappings of $\mathbb{D}$, and $1 \leqslant  p < \infty$. Then  $C_{\psi}^ * C_{\varphi} \in S_p (H^ 2)$ provided that
		\begin{align}
			\int_{\mathbb{D}} \frac{1}{[1 - |\varphi (z)|]^ {\frac{3p}{2}} [1 - |\psi (z)|]^ {\frac{3p}{2}}} \frac{|\varphi' (z)\psi' (z)|}{1 - |\varphi (z)\psi (z)|} \,\mathrm{d} \sigma (z) < +\infty, \label{ineq6}
		\end{align}
where $\mathrm{d} \sigma (z) = \log{(1 / |z|^ 2)} \,\mathrm{d} A(z)$.
<!-- end thm-12 -->
<!-- proof-5: body_md -->
Assume that (\ref{ineq6}) holds. Let $P_n$ and $Q_n$  be defined as in the second paragraph of this subsection. It is easy to see that
		\begin{align*}
			C_{\psi}^ * C_{\varphi} - \left( C_{\psi}^ * C_{\varphi} P_n + P_n C_{\psi}^ * C_{\varphi} - P_n C_{\psi}^ * C_{\varphi} P_n \right) = Q_n C_{\psi}^ * C_{\varphi} Q_n
		\end{align*}
		and
		\begin{align*}
			\operatorname{rank}{\left( C_{\psi}^ * C_{\varphi} P_n + P_n C_{\psi}^ * C_{\varphi} - P_n C_{\psi}^ * C_{\varphi} P_n \right)} \leqslant 3 (n + 1).
		\end{align*}
		By Theorem \ref{theorem: characterization of Sp}, to show $C_{\psi}^ * C_{\varphi} \in S_p (H^ 2)$ it suffices to show that
		\begin{align*}
			\sum_{n = 1}^ {\infty} {\| Q_n C_{\psi}^ * C_{\varphi} Q_n \|^ p} < +\infty.
		\end{align*}


		Since
		\begin{align*}
			\| Q_n C_{\psi}^ * C_{\varphi} Q_n \| = \sup{\left\{ \left| \left\langle C_{\varphi} Q_n f, C_{\psi} Q_n g \right\rangle_2 \right| : f, g \in H^ 2 \textrm{ with } \| f \|_2\leqslant 1, \| g \|_2 \leqslant 1 \right\}},
		\end{align*}
		we turn our attention to estimating $\left\langle C_{\varphi} Q_n f, C_{\psi} Q_n g \right\rangle_2$, where $f$ and $g$ are functions in $H^ 2$ whose norms are at most $1$. Applying Theorem \ref{theorem: Littlewood-Paley identity}, we get
		\begin{align*}
			\left\langle C_{\varphi} Q_n f, C_{\psi} Q_n g \right\rangle_2 & = \underbrace{(Q_n f) (\varphi (0)) \overline{(Q_n g) (\psi (0))}}_{\alpha_n (f, g)} + \underbrace{\int_{\mathbb{D}} (Q_n f)' (\varphi (z)) \overline{(Q_n g)' (\psi (z))} \varphi' (z) \overline{\psi' (z)} \,\mathrm{d} \sigma (z)}_{\beta_n (f, g)}.
		\end{align*}
		By Lemma \ref{lemma: estimate of Qnf and (Qnf)'}, we have the following two estimates:
		\begin{align*}
			\left| \alpha_n (f, g) \right| \leqslant \frac{|\varphi (0)|^ {n + 1} |\psi (0)|^ {n + 1}}{\sqrt{1 - |\varphi (0)|^ 2} \sqrt{1 - |\psi (0)|^ 2}} =: \alpha_n
		\end{align*}
		and
		\begin{align*}
			\left| \beta_n (f, g) \right| \leqslant  C \sum_{k, l = 0}^ 2 {n^ {\frac{k + l}{2}} \int_{\mathbb{D}} \frac{|\varphi (z)|^ n |\psi (z)|^ n |\varphi' (z) \psi' (z)|}{\left[ 1 - |\varphi (z)|^2 \right]^ {\frac{3 - k}{2}} \left[ 1 - |\psi (z)|^ 2 \right]^ {\frac{3 - l}{2}}} \,\mathrm{d} \sigma (z)} =: \beta_n,
		\end{align*}
		where $C > 0$ is some constant. It follows that
		\begin{align*}
			\| Q_n C_{\psi}^ * C_{\varphi} Q_n \|\leqslant \alpha_n + \beta_n.
		\end{align*}
		The sequence $\{ \alpha_n \}_{n = 1}^ {\infty}$ is $p$-summable, so we only need to show that
		\begin{align}
			\sum_{n = 1}^ {\infty} {\beta_n^ p} < +\infty. \label{ineq7}
		\end{align}
		Note that
		\begin{align*}
			\int_{\mathbb{D}} |\varphi' (z)|~ |\psi' (z)| \,\mathrm{d} \sigma (z) & \leqslant \Big[ \int_{\mathbb{D}} |\varphi' (z)|^ 2 \,\mathrm{d} \sigma (z) \Big]^ {\frac{1}{2}} \Big[ \int_{\mathbb{D}} |\psi' (z)|^ 2 \,\mathrm{d} \sigma (z) \Big]^ {\frac{1}{2}} \\
			& = 2 \Big[ \int_{\mathbb{D}} N_{\varphi} (z) \,\mathrm{d} A(z) \Big]^ {\frac{1}{2}} \Big[ \int_{\mathbb{D}} N_{\psi} (z) \,\mathrm{d} A(z) \Big]^ {\frac{1}{2}},
		\end{align*}
		where the ``$=$" is due to Theorem \ref{theorem: a change of variables}. It follows from (\ref{ineq3}) that the measure $|\varphi' (z)\psi' (z)| \,\mathrm{d} \sigma (z)$ is finite. This allows us to apply H\"older's inequality to obtain that
		\begin{align*}
			& \sum_{n = 1}^ {\infty} {n^ {\frac{p (k + l)}{2}} \bigg\{ \int_{\mathbb{D}} \frac{|\varphi (z)|^ n |\psi (z)|^ n |\varphi' (z)\psi' (z)|}{\left[ 1 - |\varphi (z)|^ 2 \right]^ {\frac{3 - k}{2}} \left[ 1 - |\psi (z)|^ 2 \right]^ {\frac{3 - l}{2}}} \,\mathrm{d} \sigma (z) \bigg\}^ p} \\
			& \ \ \ \leqslant C' \sum_{n = 1}^ {\infty} {n^ {\frac{p (k + l)}{2}} \int_{\mathbb{D}} \bigg\{ \frac{|\varphi (z)|^ n |\psi (z)|^ n}{\left[ 1 - |\varphi (z)|^ 2 \right]^ {\frac{3 - k}{2}} \left[ 1 - |\psi (z)|^ 2 \right]^ {\frac{3 - l}{2}}} \bigg\}^ p |\varphi' (z)\psi' (z)| \,\mathrm{d} \sigma (z)} \\
			& \ \ \  \leqslant C'' \int_{\mathbb{D}} \frac{|\varphi' (z)\psi' (z)|}{\left[ 1 - |\varphi (z)|^ 2 \right]^ {\frac{p (3 - k)}{2}} \left[ 1 - |\psi (z)|^ 2 \right]^ {\frac{p (3 - l)}{2}} \left[ 1 - |\varphi (z)|^ p |\psi (z)|^ p \right]^ {1 + \frac{p (k + l)}{2}}} \,\mathrm{d} \sigma (z),
		\end{align*}
		where $k, l \in \{ 0, 1, 2 \}$ and $C', C'' > 0$ are constants. Since
		\begin{align*}
			\left( 1 - |\varphi (z)|^ p |\psi (z)|^ p \right)^ {\frac{p (k + l)}{2}} \geqslant \left( 1 - |\varphi (z)| \right)^ {\frac{p k}{2}} \left( 1 - |\psi (z)| \right)^ {\frac{p l}{2}},
		\end{align*}
		we further obtain that
		\begin{align*}
			&\sum_{n = 1}^ {\infty} {n^ {\frac{p (k + l)}{2}} \bigg\{ \int_{\mathbb{D}} \frac{|\varphi (z)|^ n |\psi (z)|^ n |\varphi' (z)\psi' (z)|}{\left[ 1 - |\varphi (z)|^ 2 \right]^ {\frac{3 - k}{2}} \left[ 1 - |\psi (z)|^ 2 \right]^ {\frac{3 - l}{2}}} \,\mathrm{d} \sigma (z) \bigg\}^ p} \\
			&\ \ \ \leqslant C'' \int_{\mathbb{D}} \frac{|\varphi' (z)\psi' (z)|}{\left[ 1 - |\varphi (z)|^ 2 \right]^ { \frac{3p}{2}} \left[ 1 - |\psi (z)|^ 2 \right]^ {\frac{3p}{2}} \left[ 1 - |\varphi (z)\psi (z)| \right]} \,\mathrm{d} \sigma (z),
		\end{align*}
		which is finite by (\ref{ineq6}). Since $k$ and $l$ in the above estimations are arbitrary numbers in $\{ 0, 1, 2 \}$, we obtain (\ref{ineq7}), and the proof is now complete.

<!-- prop-1: body_md -->
\label{proposition: sufficient condition for S2, Hardy space}
		Suppose that $\varphi$ and $\psi$ are analytic self-mappings of $\mathbb{D}$. If
		\begin{align}
			\int_{\mathbb{T}} \int_{\mathbb{T}} \frac{\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t}) \mathrm{d} m (\mathrm{e}^ {\mathrm{i} s})}{\big| 1 - \varphi (\mathrm{e}^ {\mathrm{i} t}) \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})} \big| \big| 1 - \overline{\psi (\mathrm{e}^ {\mathrm{i} t})} \psi (\mathrm{e}^ {\mathrm{i} s}) \big|} < +\infty, \label{ineq1}
		\end{align}
then  $C_{\psi}^ * C_{\varphi} \in S_2 (H^ 2)$.

<!-- proof-6: body_md -->
According to Theorem \ref{theorem: characterization of S2}, we only need to prove that
		\begin{align*}
			\sum_{j, k = 0}^ {\infty} {|\langle C_{\psi}^ * C_{\varphi} (z^ j), z^ k \rangle_2|^ 2} < +\infty
		\end{align*}
		if (\ref{ineq1}) holds.

		\begin{align*}
			\sum_{j, k = 0}^ N {|\langle C_{\psi}^ * C_{\varphi} (z^ j), z^ k \rangle_2|^ 2} & = \sum_{j, k = 0}^ N {|\langle C_{\varphi} (z^ j), C_{\psi} (z^ k) \rangle_2|^ 2} \\
			& = \sum_{j, k = 0}^ N {\left| \int_{\mathbb{T}} \varphi^ j \overline{\psi}^ k \,\mathrm{d} m \right|^ 2} \\
			& = \sum_{j, k = 0}^ N {\int_{\mathbb{T}} \int_{\mathbb{T}} \varphi (\mathrm{e}^ {\mathrm{i} t})^ j \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})}^ j \overline{\psi (\mathrm{e}^ {\mathrm{i} t})}^ k \psi (\mathrm{e}^ {\mathrm{i} s})^ k \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t}) \mathrm{d} m (\mathrm{e}^ {\mathrm{i} s})}.

		\end{align*}
		are null sets on $\mathbb{T} \times \mathbb{T}$ with respect to the measure $\mathrm{d} (m \times m)$. Thus,
		\begin{align*}
			&\sum_{j, k = 0}^ N {|\langle C_{\psi}^ * C_{\varphi} (z^ j), z^ k \rangle_2|^ 2} \\
&\ \ \ = \int_{\mathbb{T}} \int_{\mathbb{T}} \frac{1 - \varphi (\mathrm{e}^ {\mathrm{i} t})^ {N + 1} \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})}^ {N + 1}}{1 - \varphi (\mathrm{e}^ {\mathrm{i} t}) \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})}} \frac{1 - \overline{\psi (\mathrm{e}^ {\mathrm{i} t})}^ {N + 1} \psi (\mathrm{e}^ {\mathrm{i} s})^ {N + 1}}{1 - \overline{\psi (\mathrm{e}^ {\mathrm{i} t})} \psi (\mathrm{e}^ {\mathrm{i} s})} \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t}) \mathrm{d} m (\mathrm{e}^ {\mathrm{i} s}).
		\end{align*}
		The integrand on the right hand side is dominated by the function
		\begin{align*}
			\frac{4}{\big| 1 - \varphi (\mathrm{e}^ {\mathrm{i} t}) \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})} \big| \big| 1 - \overline{\psi (\mathrm{e}^ {\mathrm{i} t})} \psi (\mathrm{e}^ {\mathrm{i} s}) \big|},
		\end{align*}
		which is integrable over $\mathbb{T} \times \mathbb{T}$ due to  (\ref{ineq1}). Thus,  we conclude that
		\begin{align*}
			\sum_{j, k = 0}^ {\infty} {|\langle C_{\psi}^ * C_{\varphi} (z^ j), z^ k \rangle_2|^ 2} \leqslant 4 \int_{\mathbb{T}} \int_{\mathbb{T}} \frac{\mathrm{d} m (\mathrm{e}^ {\mathrm{i} t}) \mathrm{d} m (\mathrm{e}^ {\mathrm{i} s})}{\big| 1 - \varphi (\mathrm{e}^ {\mathrm{i} t}) \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})} \big| \big| 1 - \overline{\psi (\mathrm{e}^ {\mathrm{i} t})} \psi (\mathrm{e}^ {\mathrm{i} s}) \big|} < +\infty,
		\end{align*}
		as desired.

<!-- remark-2: body_md -->
Condition (\ref{ineq1}) is not only sufficient but also necessary in the case that $\varphi = \psi$. Suppose that $C_{\varphi}^ * C_{\varphi}$ belongs to $S_2 (H^ 2)$. In this case, the calculation in the proof of the  above proposition  shows that
		\begin{align*}
			\sum_{j, k = 0}^ N {|\langle C_{\varphi}^ * C_{\varphi} (z^ j), z^ k \rangle_2|^ 2} = \int_{\mathbb{T}} \int_{\mathbb{T}} \Big| \sum_{j = 0}^ N {\varphi (\mathrm{e}^ {\mathrm{i} t})^ j \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})}^ j} \Big|^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t}) \mathrm{d} m (\mathrm{e}^ {\mathrm{i} s}).
		\end{align*}
		Since $C_{\varphi}^ * C_{\varphi} \in S_2 (H^ 2)$, $C_{\varphi}$ is compact, which means that $|\varphi (\mathrm{e}^ {\mathrm{i} \theta})| < 1$ for $m$-a.e. $\mathrm{e}^ {\mathrm{i} \theta} \in \mathbb{T}$ (see \cite{Sch}). Therefore, the set
		\begin{align*}
			\left\{ (\mathrm{e}^ {\mathrm{i} t}, \mathrm{e}^ {\mathrm{i} s}) \in \mathbb{T} \times \mathbb{T}: \big| \varphi (\mathrm{e}^ {\mathrm{i} t}) \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})} \big| = 1 \right\}
		\end{align*}
		is a null set with respect to the measure $\mathrm{d} (m \times m)$. It follows that
		\begin{align*}
			\sum_{j, k = 0}^ N {|\langle C_{\varphi}^ * C_{\varphi} (z^ j), z^ k \rangle_2|^ 2} = \int_{\mathbb{T}} \int_{\mathbb{T}} \frac{\big| 1 - \varphi (\mathrm{e}^ {\mathrm{i} t})^ {N + 1} \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})}^ {N + 1} \big|^ 2}{\big| 1 - \varphi (\mathrm{e}^ {\mathrm{i} t}) \overline{\varphi (\mathrm{e}^ {\mathrm{i} s})} \big|^ 2} \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t}) \mathrm{d} m (\mathrm{e}^ {\mathrm{i} s}).
		\end{align*}

\label{lemma: characterization of the Hilbert--Schmidt class}
		A bounded linear operator $A$ on $L_a^ 2 (\mathrm{d} \sigma)$ belongs to $S_2 (L_a^ 2 (\mathrm{d} \sigma))$ if and only if
		\begin{align*}
			\int_{\mathbb{D}} \int_{\mathbb{D}} \left| A^ * K_z (w) \right|^ 2 \,\mathrm{d} \sigma (w) \mathrm{d} \sigma (z) < +\infty,
		\end{align*}
		where $K_z$ is the reproducing kernel for $L_a^ 2 (\mathrm{d} \sigma)$ at $z \in \mathbb{D}$.

<!-- proof-7: body_md -->
Consider the integral operator
		\begin{align*}
			\widetilde{A} f (z) = \int_{\mathbb{D}} f(w) \overline{A^ * K_z (w)} \,\mathrm{d} \sigma (w)
		\end{align*}
		defined on the Lebesgue space $L^ 2 (\mathbb{D}, \mathrm{d} \sigma)$. For every $f \in L^ 2 (\mathbb{D}, \mathrm{d} \sigma) \ominus L_a^ 2 (\mathrm{d} \sigma)$, we have
		\begin{align*}
			\widetilde{A} f (z) = \langle f, A^ * K_z \rangle_{L^ 2 (\mathbb{D}, \mathrm{d} \sigma)} = 0, \qquad z \in \mathbb{D},
		\end{align*}

\label{lemma: Aleksandrov--Clark measure of B}
		Suppose $B$ is a finite Blaschke product of degree $n$. Let $\{ \mathrm{d} \mu_{\alpha} : \alpha \in \mathbb{T} \}$ be the family of the Aleksandrov--Clark measures associated with $B$. Then we have
		\begin{align*}
			\mathrm{d} \mu_{\alpha} = \sum_{j = 1}^ n {\frac{1}{|B' (\zeta_j)|} \,\mathrm{d} \delta_{\zeta_j}},
		\end{align*}
		where $\zeta_1, \zeta_2, \cdots, \zeta_n$ are the $n$ distinct roots of the equation $B(z) = \alpha$ and each $\mathrm{d} \delta_{\zeta_j}$ is the point mass at $\zeta_j$.
<!-- end lem-7 -->
<!-- proof-11: body_md -->
According to (3) of \cite[Corollary 7.2]{Ros}, $\mathrm{d} \mu_{\alpha}$ is singular with respect to $\mathrm{d} m$ for each $\alpha \in \mathbb{T}$. For every $\alpha \in \mathbb{T}$,  we have
		\begin{align*}
			\frac{1 - |B(z)|^ 2}{|\alpha - B(z)|^ 2} = P_{B(z)} (\alpha) = \int_{\mathbb{T}} P_z (\mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} \mu_{\alpha} (\mathrm{e}^ {\mathrm{i} t}).
		\end{align*}
		Hence,  we obtain that
		\begin{align*}
			\lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}} P_{r \zeta} (\mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} \mu_{\alpha} (\mathrm{e}^ {\mathrm{i} t})} = 0
		\end{align*}
		if $\zeta \in \mathbb{T} \backslash \{ \zeta_1, \zeta_2, \dots, \zeta_n \}$. By (2) of \cite[Proposition 4.14]{Ros}, we have the inequality
		\begin{align*}
			(\underline{D} \mu_{\alpha}) (\zeta) \leqslant \varliminf_{r \rightarrow 1^ -} {\int_{\mathbb{T}} P_{r \zeta} (\mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} \mu_{\alpha} (\mathrm{e}^ {\mathrm{i} t})},
		\end{align*}
		where
		\begin{align*}
			(\underline{D} \mu_{\alpha}) (\mathrm{e}^ {\mathrm{i} \theta}) = \varliminf_{\delta \rightarrow 0^ +} {\frac{\mu_{\alpha} (\{ \mathrm{e}^ {\mathrm{i} t} \in \mathbb{T} : |\theta - t| < \delta \})}{m (\{ \mathrm{e}^ {\mathrm{i} t} \in \mathbb{T} : |\theta - t| < \delta \})}}
		\end{align*}
		is the lower symmetric derivative of $\mathrm{d} \mu_{\alpha}$ at $\mathrm{e}^ {\mathrm{i} \theta} \in \mathbb{T}$. Thus,
		\begin{align*}
			(\underline{D} \mu_{\alpha}) (\zeta) = 0, \qquad \zeta \in \mathbb{T} \backslash \{ \zeta_1, \zeta_2, \cdots, \zeta_n \}.
		\end{align*}
		On the other hand, according to (5) of \cite[Proposition 4.14]{Ros}, the set
		\begin{align*}
			\left\{ \zeta \in \mathbb{T} : (\underline{D} \mu_{\alpha}) (\zeta) = +\infty \right\}
		\end{align*}
		is a carrier for $\mathrm{d} \mu_{\alpha}$, which means that
		\begin{align*}
			\mu_{\alpha} (E) = \mu_{\alpha} (E \cap \left\{ \zeta \in \mathbb{T} : (\underline{D} \mu_{\alpha}) (\zeta) = +\infty \right\})
		\end{align*}
		for every Borel set $E$ in $\mathbb{T}$. Therefore, $\{ \zeta_1, \zeta_2, \cdots, \zeta_n \}$ is a carrier for $\mathrm{d} \mu_{\alpha}$, which yields  that $\mathrm{d} \mu_{\alpha}$ is a linear combination of point masses $\mathrm{d} \zeta_1, \mathrm{d} \zeta_2, \cdots, \mathrm{d} \zeta_n$.
		The assertion now follows from \cite[Corollary 9.2]{Ros}.

\label{proposition: sufficient condition for Cphi Cpsi* in Sp}
		Suppose that $2\leqslant  p < \infty$, $\varphi$ and $\psi$ are analytic self-mappings of $\mathbb{D}$. Then  $C_{\varphi} C_{\psi}^ * \in S_p (H^ 2)$ if
		\begin{align} \label{ineq2}
			\int_{\mathbb{D}} \int_{\mathbb{D}} \bigg[ \frac{\left| 1 + \overline{z} w \right|}{\left| 1 - \overline{z} w \right|^ 3} \sqrt{N_{\varphi} (z) N_{\psi} (w)} \bigg]^ p \,\mathrm{d} A(w) \mathrm{d} A(z) < +\infty.
		\end{align}
<!-- end thm-14 -->
<!-- proof-13: body_md -->
Assume  that (\ref{ineq2}) holds and denote $\Phi (z) = N_{\varphi} (z) / \log{(1 / |z|)}$. According to the discussion in the first paragraph of this subsection, it suffices to show that $D_{\psi} T_{\Phi} D_{\psi}^ *$ belongs to $S_{\frac{p}{2}} (L_a^ 2 (\mathrm{d} \sigma))$.
		Let $\{ e_n \}_{n = 1}^ {\infty}$ be an arbitrary orthonormal basis for $L_a^ 2 (\mathrm{d} \sigma)$. Then by Lemma \ref{lemma: Cphi* Cphi is a Toeplitz opeartor} we have
		\begin{align*}
			\sum_{n = 1}^ {\infty} {|\langle D_{\psi} T_{\Phi} D_{\psi}^* e_n, e_n \rangle_{L^2 (\mathbb{D}, \mathrm{d} \sigma)}|^ {\frac{p}{2}}} & = \sum_{n = 1}^ {\infty} {|\langle T_{\Phi} D_{\psi}^ * e_n, D_{\psi}^ * e_n \rangle_{L^ 2 (\mathbb{D}, \mathrm{d} \sigma)}|^ {\frac{p}{2}}} \\
			& = 2^ {\frac{p}{2}} \sum_{n = 1}^ {\infty} {\Big[ \int_{\mathbb{D}} |D_{\psi}^ * e_n (z)|^ 2 N_{\varphi} (z) \,\mathrm{d} A (z) \Big]^ {\frac{p}{2}}} \\
			& \leqslant 2^ {\frac{p}{2}} \sum_{n = 1}^ {\infty} {\int_{\mathbb{D}} |D_{\psi}^ * e_n (z)|^ {p} N_{\varphi} (z)^ {\frac{p}{2}} \,\mathrm{d} A(z)}.
		\end{align*}
		Using the elementary inequality
		\begin{align*}
			\Big( \sum_{n = 1}^ {\infty} {|c_n|} \Big)^ q \leqslant  \sum_{n = 1}^\infty {|c_n|^ q} \ \ \ \ \ \big(c_n \in \mathbb{C},\ 0 < q\leqslant 1\big),
		\end{align*}
		we obtain that
		\begin{align*}
			\sum_{n = 1}^ {\infty} {|D_{\psi}^ * e_n (z)|^ p} & \leqslant \Big[ \sum_{n = 1}^ {\infty} {|D_{\psi}^ * e_n (z)|^ 2} \Big]^ {\frac{p}{2}} = \Big[ \sum_{n = 1}^ {\infty} |\langle e_n, D_{\psi} K_z \rangle_{L^ 2 (\mathbb{D}, \mathrm{d} \sigma)}|^ 2 \Big]^ {\frac{p}{2}} = \| D_{\psi} K_z \|_{L^ 2 (\mathbb{D}, \mathrm{d} \sigma)}^ p,
		\end{align*}
		where $K_z$ is the reproducing kernel for $L_a^ 2 (\mathrm{d} \sigma)$ at $z$.  Furthermore, we have by Theorem \ref{theorem: a change of variables} that
		\begin{align*}
			\| D_{\psi} K_z \|_{L^ 2 (\mathbb{D}, \mathrm{d} \sigma)}^ 2 & = \int_{\mathbb{D}} |\psi' (w)|^ 2 \frac{|1 + \overline{z} \psi (w)|^ 2}{|1 - \overline{z} \psi (w)|^ 6} \,\mathrm{d} \sigma (w) \\
			& = 2 \int_{\mathbb{D}} \frac{|1 + \overline{z} w|^ 2}{|1 - \overline{z} w|^ 6} N_{\psi} (w) \,\mathrm{d} A(w),
		\end{align*}
	which yields that
		\begin{align*}
			\sum_{n = 1}^ {\infty} {|\langle D_{\psi} T_{\Phi} D_{\psi}^ * e_n, e_n \rangle_{L^ 2 (\mathbb{D}, \mathrm{d} \sigma)}|^ {\frac{p}{2}}} & \leqslant 2^ {\frac{p}{2}} \int_{\mathbb{D}} \bigg[ 2 \int_{\mathbb{D}} \frac{|1 + \overline{z} w|^ 2}{|1 - \overline{z} w|^ 6} N_{\psi} (w) \,\mathrm{d} A(w) \bigg]^ {\frac{p}{2}} N_{\varphi} (z)^ {\frac{p}{2}} \,\mathrm{d} A(z) \\
			& \leqslant 2^ p \int_{\mathbb{D}} \int_{\mathbb{D}} \frac{\left| 1 + \overline{z} w \right|^ p}{|1 - \overline{z} w|^ {3 p}} N_{\varphi} (z)^ {\frac{p}{2}} N_{\psi} (w)^ {\frac{p}{2}} \,\mathrm{d} A(w) \mathrm{d} A(z) \\
			& < +\infty.
		\end{align*}
		Now we conclude that $D_{\psi} T_{\Phi} D_{\psi}^ *$ belongs to $S_{\frac{p}{2}} (L_a^ 2 (\mathrm{d} \sigma))$, as desired.

\label{lemma: trace formula for positive operators}
		Suppose that $T$ is  a  bounded positive operator on the Hardy space $H^ 2$. Then $T$ belongs to $S_1 (H^ 2)$ if and only if the limit
		\begin{align}
			\lim_{r \rightarrow 1^ -} {\frac{1}{1 - r^ 2} \int_{\mathbb{T}} \mathcal{B} T (r \mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})} \label{for3}
		\end{align}
		exists. In this case, \textup{(\ref{for3})} is the trace of $T$.
<!-- end lem-8 -->
<!-- proof-14: body_md -->
Note that $T$ belongs to $S_1 (H^ 2)$ if and only if $\sqrt{T}$ belongs to $S_2 (H^ 2)$. It follows from Theorem \ref{theorem: characterization of S2} that $T \in S_1 (H^ 2)$ if and only if
		\begin{align*}
			\sum_{n = 1}^ {\infty} {\big\| \sqrt{T} e_n \big\|_2^ 2} < +\infty
		\end{align*}
		for some orthonormal basis  $\{ e_n \}_{n = 1}^ {\infty}$ of $H^ 2$.


Noting that
		\begin{align*}
			\big\| \sqrt{T} e_n \big\|_2^ 2 = \lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}} \big| \sqrt{T} e_n (r \mathrm{e}^ {\mathrm{i} t}) \big|^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})} = \lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}} \big| \big\langle \sqrt{T} e_n, K_{r \mathrm{e}^ {\mathrm{i} t}} \big\rangle_2 \big|^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})},
		\end{align*}
		we have
		\begin{align*}
			\sum_{n = 1}^ {\infty} {\big\| \sqrt{T} e_n \big\|_2^ 2} & = \sum_{n = 1}^ {\infty} {\lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}} \big| \big\langle \sqrt{T} e_n, K_{r \mathrm{e}^ {\mathrm{i} t}} \big\rangle_2 \big|^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})}} \\
			& = \lim_{r \rightarrow 1^ -} {\sum_{n = 1}^ {\infty} {\int_{\mathbb{T}} \big| \big\langle \sqrt{T} e_n, K_{r \mathrm{e}^ {\mathrm{i} t}} \big\rangle_2 \big|^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})}} \\
			& = \lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}} \sum_{n = 1}^ {\infty} {\big| \big\langle e_n, \sqrt{T} K_{r \mathrm{e}^ {\mathrm{i} t}} \big\rangle_2 \big|^ 2} \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})} \\
			& = \lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}}\big\| \sqrt{T} K_{r \mathrm{e}^ {\mathrm{i} t}} \big\|_2^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})} \\
			& = \lim_{r \rightarrow 1^ -} {\frac{1}{1 - r^ 2} \int_{\mathbb{T}} \mathcal{B} T (r \mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})}.
		\end{align*}
		Here, the second ``$=$'' above is due to the monotone convergence theorem, since the integral
		\begin{align*}
			\int_{\mathbb{T}} \big| \big\langle \sqrt{T} e_n, K_{r \mathrm{e}^ {\mathrm{i} t}} \big\rangle_2 \big|^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t}) = \int_{\mathbb{T}} \big| \sqrt{T} e_n (r \mathrm{e}^ {\mathrm{i} t}) \big|^ 2 \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})
		\end{align*}
		is increasing in $r\in [0, 1)$. Therefore, $T$ belongs to $S_1 (H^ 2)$ if and only if the limit (\ref{for3}) exists. In this case, by Theorem \ref{theorem: characterization of S2} we also have
		\begin{gather*}
			\lim_{r \rightarrow 1^ -} {\frac{1}{1 - r^ 2} \int_{\mathbb{T}} \mathcal{B} T (r \mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})} = \big\| \sqrt{T} \big\|_{S_2 (H^ 2)}^ 2 = \| T \|_{S_1 (H^ 2)} = \operatorname{tr}{(T)}.
		\end{gather*}
This finishes the proof of the lemma.

<!-- lem-9: body_md -->
\label{lemma: trace formula}
		Suppose that $T$ is a bounded linear operator on $H^ 2$. If $T$ belongs to $S_1 (H^ 2)$, then
		\begin{align}
			\lim_{r \rightarrow 1^ -} {\frac{1}{1 - r^ 2} \int_{\mathbb{T}} \mathcal{B} T (r \mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})} \label{for4}
		\end{align}
		exists. In this case, \textup{(\ref{for4})} is the trace of $T$.
<!-- end lem-9 -->
<!-- proof-15: body_md -->
Let us first consider the case that $T$ is self-adjoint. In this case, both $(|T| + T) / 2$ and $(|T| - T) / 2$ are positive and in $S_1 (H^ 2)$. Hence, according to Lemma \ref{lemma: trace formula for positive operators}, the limits
		\begin{align*}
			\lim_{r \rightarrow 1^ -} {\frac{1}{1 - r^ 2} \int_{\mathbb{T}} \mathcal{B} \Big( \frac{|T| \pm T}{2} \Big) (r \mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})}
		\end{align*}
		exist and equal $\operatorname{tr}{\left[ (|T| \pm T) / 2 \right]}$. It follows that
		\begin{align*}
			\operatorname{tr}{(T)} = \lim_{r \rightarrow 1^ -} {\frac{1}{1 - r^ 2} \int_{\mathbb{T}} \mathcal{B} T (r \mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})}.
		\end{align*}
		Now the assertion for general operators $T$ follows from the standard decomposition
		\begin{gather*}
			T = \frac{T + T^ *}{2} + \mathrm{i} \frac{T - T^ *}{2 \mathrm{i}},
		\end{gather*}
finishing the proof.

\label{proposition: trace formula, II}
		Suppose $\varphi$ and $\psi$ are analytic self-mappings of $\mathbb{D}$. If $C_{\varphi} C_{\psi}^ *$ belongs to $S_1 (H^ 2)$, then the limit
		\begin{align*}
			\lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}} \frac{\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})}{1 - \varphi (r \mathrm{e}^ {\mathrm{i} t}) \overline{\psi (r \mathrm{e}^ {\mathrm{i} t})}}}
		\end{align*}
		exists. In this case, the above quantity is the trace of $C_{\varphi} C_{\psi}^ *$.
<!-- end prop-5 -->
<!-- proof-16: body_md -->
First, note that
		\begin{align*}
			\left\langle C_{\varphi}^ * K_z, f \right\rangle_2 & = \left\langle K_z, C_{\varphi} f \right\rangle_2 = \overline{f (\varphi (z))} = \left\langle K_{\varphi (z)}, f \right\rangle_2,\ \ \  \ f\in H^2.
		\end{align*}
	This gives us that $C_{\varphi}^ * K_z = K_{\varphi (z)}$, and similarly $C_{\psi}^ * K_z = K_{\psi (z)}$.

		Then the  Berezin transform of $C_{\varphi} C_{\psi}^ *$ is given by
		\begin{align*}
			\mathcal{B} \left( C_{\varphi} C_{\psi}^ * \right) (z) & = \left\langle C_{\varphi} C_{\psi}^ * k_z, k_z \right\rangle_2 = \left\langle C_{\psi}^ * k_z, C_{\varphi}^ * k_z \right\rangle_2 = (1 - |z|^ 2) \left\langle K_{\psi (z)}, K_{\varphi (z)} \right\rangle_2 \\
			& = (1 - |z|^ 2) K_{\psi (z)} \left( \varphi (z) \right).
		\end{align*}
		 It follows from  Lemma \ref{lemma: trace formula} that the limit
		\begin{align*}
			\lim_{r \rightarrow 1^ -} {\frac{1}{1 - r^ 2} \int_{\mathbb{T}} \mathcal{B} \left( C_{\varphi} C_{\psi}^ * \right) (r \mathrm{e}^ {\mathrm{i} t}) \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})} = \lim_{r \rightarrow 1^ -} {\int_{\mathbb{T}} \frac{1}{1 - \varphi (r \mathrm{e}^ {\mathrm{i} t}) \overline{\psi (r \mathrm{e}^ {\mathrm{i} t})}} \,\mathrm{d} m(\mathrm{e}^ {\mathrm{i} t})}
		\end{align*}
		exists and is equal to the trace of $C_{\varphi} C_{\psi}^*$.

The ``only if'' part is a consequence of \cite[Theorem 1]{Nor}, which states that $C_{\psi}$ (with $\psi$ inner) is bounded below on $H^2$. \label{lemma: operator identities}
		Suppose $\varphi$ is an analytic self-mapping of $\mathbb{D}$ and $u$ is a bounded analytic function on $\mathbb{D}$. Let
		\begin{align*}
			\alpha = Q \varphi \qquad \textrm{and} \qquad \beta = Q \alpha,
		\end{align*}
		which are bounded and analytic on $\mathbb{D}$. Then on the Bergman space $L_a^ 2$:
		\begin{enumerate}[label = \textup{(\alph*)}]
			\item $C_{\varphi} T_u = T_{u \circ \varphi} C_{\varphi}$;
			\item $Q C_{\varphi} = T_{\alpha} C_{\varphi} Q$ and $Q T_{\alpha} = T_{\beta} + \alpha (0) Q$ if $\varphi(0)=0$.
		\end{enumerate}
<!-- end lem-12 -->
<!-- proof-19: body_md -->
Let $f$ be any function in $L_a^ 2$. Since
		\begin{align*}
			C_{\varphi} T_u f = (u \circ \varphi) (f \circ \varphi) = T_{u \circ \varphi} C_{\varphi} f,
		\end{align*}
		(a) holds.

		Using
		\begin{align*}
			Q C_{\varphi} f (z) = \frac{f(\varphi(z)) - f (0)}{z} = \frac{f(\varphi (z)) - f(0)}{\varphi (z)} \frac{\varphi (z)}{z} = T_{\alpha} C_{\varphi} Q f(z), \qquad z \in \mathbb{D},
		\end{align*}
		we obtain $Q C_{\varphi} = T_{\alpha} C_{\varphi} Q$. Finally, observing that
		\begin{align*}
			Q T_{\alpha} f (z) & = \frac{\alpha (z) f(z) - \alpha (0) f(0)}{z} = \frac{[\alpha (z) - \alpha (0)] f(z)}{z} + \alpha (0) \frac{f(z) - f(0)}{z} \\
			& = T_{\beta} f (z) + \alpha (0) Q f (z), \qquad z \in \mathbb{D},
		\end{align*}
		we get that $Q T_{\alpha} = T_{\beta} + \alpha (0) Q$. This gives (b).
<!-- end proof-19 -->
<!-- proof-20: body_md -->
Following the argument for the corresponding result in the  Hardy space setting, (a) and the necessity part of (b) can be established similarly (see \cite{CZ}) and are omitted. It remains to prove the sufficiency part of (b). Replacing $\varphi$ by $\frac{\varphi (0) - \varphi}{1 - \overline{\varphi (0)} \varphi}$ if necessary, we may assume that $\varphi (0) = 0$. And since the conclusion is trivial if $\psi$ is a constant function, we further assume that $\psi$ is non-constant.
		
		Adopting notation in Lemma \ref{lemma: representation of the adjoint}, we get that
		\begin{align*}
			C_{\psi}^ * C_{\varphi} = T_{g^ 2} C_{\tau} T_{h^ 2}^ * C_{\varphi} = T_{g^ 2} C_{\tau} (\overline{c} T_{\overline{z}} + \overline{d} I)^ 2 C_{\varphi},
		\end{align*}
		where $I$ is the identity operator on $L_a^ 2$. For any
		\begin{align*}
			f(z) = a_0 + a_1 z + a_2 z^ 2 + \cdots
		\end{align*}
		in $L_a^ 2$, we have
		\begin{align*}
			T_{\overline{z}} \left( a_0 + a_1 z + a_2 z^ 2 + \cdots \right) & = \frac{1}{2} a_1 + \frac{2}{3} a_2 z + \frac{3}{4} a_3 z^ 2 + \cdots \\
			& = \left( a_1 + a_2 z + a_3 z^ 2 + \cdots \right) - \left( \frac{1}{2} a_1 + \frac{1}{3} a_2 z + \frac{1}{4} a_3 z^ 2 + \cdots \right).
		\end{align*}
		Hence,
		\begin{align*}
			T_{\overline{z}} = Q + R,
		\end{align*}
		where $R$ is the compact operator defined by
		\begin{align*}
			R f = \sum_{j = 0}^ {\infty} {\frac{1}{\sqrt{(j + 1) (j + 2)}} (\sigma_j \otimes \sigma_{j + 1}) f}, \ \ \ \ f\in L_a^2.
		\end{align*}
		It follows that
		\begin{align*}
			(\overline{c} T_{\overline{z}} + \overline{d} I)^ 2 = (\overline{c} Q + \overline{c} R + \overline{d} I)^ 2 = (\overline{c} Q + \overline{d} I)^ 2 + K
		\end{align*}
		for some compact operator $K$. Therefore, $C_{\psi}^ * C_{\varphi}$ is compact if and only if the operator
		\begin{align*}
			\overline{c}^ 2 T_{g^ 2} C_{\tau} Q^ 2 C_{\varphi} + 2 \overline{c d} T_{g^ 2} C_{\tau} Q C_{\varphi} + \overline{d}^ 2 T_{g^ 2} C_{\tau} C_{\varphi}
		\end{align*}
		is compact.
		
		If $C_{\tau} C_{\varphi}$ is compact, then  $T_{g^ 2} C_{\tau} C_{\varphi}$ is compact. By Lemma \ref{lemma: operator identities}, we have
		\begin{align*}
			T_{g^ 2} C_{\tau} Q C_{\varphi} = T_{g^ 2} C_{\tau} T_{\alpha} C_{\varphi} Q = T_{g^ 2} T_{\alpha \circ \tau} C_{\tau} C_{\varphi} Q,
		\end{align*}
		where $\alpha = Q \varphi$. This implies that  $T_{g^ 2} C_{\tau} Q C_{\varphi}$ is compact. By Lemma \ref{lemma: operator identities} again, we obtain that
		\begin{align*}
			T_{g^ 2} C_{\tau} Q^ 2 C_{\varphi} & = T_{g^ 2} C_{\tau} Q T_{\alpha} C_{\varphi} Q = T_{g^ 2} C_{\tau} [T_{\beta} + \alpha (0) Q] C_{\varphi} Q \\

			& = T_{g^ 2} T_{\beta \circ \tau} C_{\tau} C_{\varphi} Q + \alpha (0) T_{g^ 2} T_{\alpha \circ \tau} C_{\tau} C_{\varphi} Q^ 2,
		\end{align*}
		where $\beta = Q \alpha$, which yields that $T_{g^ 2} C_{\tau} Q^ 2 C_{\varphi}$ is also compact. This completes the proof of the Lemma.

<!-- thm-18: body_md -->
\label{BS}
		Let $\varphi$ and $\psi$ be two analytic self-mappings of $\mathbb{D}$, and $1 \leqslant p < \infty$. Then  $C_{\psi}^ * C_{\varphi} \in S_p (L_a^ 2)$ if
		\begin{align*}
			\int_{\mathbb{D}} \frac{\mathrm{d} A(z)}{\left( 1 - |\varphi (z)| |\psi (z)| \right) \left( 1 - |\varphi (z)| \right)^ p \left( 1 - |\psi (z)| \right)^ p} < +\infty.

<!-- proof-24: body_md -->
The proof is analogous to that of Theorem \ref{proposition: sufficient condition for Cpsi*Cphi in Sp}. In the following,  we only provide an outline of the proof. Let $P_n$ be the orthogonal projection from $L_a^2$ onto $\operatorname{span}{\left\{ \sigma_0, \cdots, \sigma_n \right\}}$, where  $$\sigma_n (z) = \sqrt{n + 1} z^ n, \ \ \ n=0, 1, 2, \cdots.$$ Let $$Q_n = I - P_n,$$ where  $I$ is the identity operator on $L_a^2$ and $n$ is a positive integer. For each $f$ in $L_a^ 2$, we write
$$f(z) = \sum\limits_{j = 0}^ {\infty} {a_j \sigma_j}(z), \ \ \ \ z\in \mathbb D.$$
It follows that $$(Q_n f)(z) = \sum\limits_{j = n + 1}^ {\infty} {a_j \sigma_j}(z),$$
which yields
		\begin{align*}
			|(Q_n f) (z)| & \leqslant \left( \sum_{j = n + 1}^ {\infty} {|a_j|^ 2} \right)^ {\frac{1}{2}} \left( \sum_{j = n + 1}^ {\infty} {|\sigma_j (z)|^ 2} \right)^ {\frac{1}{2}} \\
			& \leqslant |z|^ {n + 1} \left[ \frac{1}{(1 - |z|^ 2)^ 2} + \frac{n + 1}{1 - |z|^ 2} \right]^ {\frac{1}{2}} \| f \|_{L^ 2 (\mathbb{D})} \\
			& \leqslant C |z|^ {n + 1} \left[ \frac{1}{1 - |z|^ 2} + \frac{n^ {\frac{1}{2}}}{(1 - |z|^ 2)^ {\frac{1}{2}}} \right] \| f \|_{L^ 2 (\mathbb{D})},
		\end{align*}
		where $C > 0$ is a constant.
		Now mimicking the proof of  Theorem \ref{proposition: sufficient condition for Cpsi*Cphi in Sp}, we arrive at the desired sufficient condition given in Theorem \ref{BS}.

Suppose that $\varphi$ and $\psi$ are analytic self-mappings of $\mathbb{D}$. If

			\int_{\mathbb{D}} \int_{\mathbb{D}} \frac{\mathrm{d} A(z) \mathrm{d} A(w)}{\big| 1 - \varphi (z) \overline{\varphi (w)} \big|^ 2 \big| 1 - \overline{\psi (z)} \psi (w) \big|^ 2} < +\infty,

then $C_{\psi}^ * C_{\varphi} \in S_2 (L_a^ 2)$.

<!-- proof-25: body_md -->
Fixing an $r \in (0, 1)$ and a positive integer $N$, we have
		\begin{align*}
			 &\sum_{j, k = 0}^ N {\Big| \int_{r \mathbb{D}} (C_{\varphi} \sigma_j) (z) \overline{(C_{\psi} \sigma_k) (z)} \,\mathrm{d} A(z) \Big|^ 2}\\
& = \sum_{j, k = 0}^ N {(j + 1) (k + 1) \Big| \int_{r \mathbb{D}} \varphi (z)^ j \overline{\psi (z)}^ k \,\mathrm{d} A(z) \Big|^ 2} \\
			& = \int_{r \mathbb{D}} \int_{r \mathbb{D}} \sum_{j, k = 0}^ N {(j + 1) (k + 1) \varphi(z)^ j \overline{\varphi (w)}^ j \overline{\psi (z)}^ k \psi (w)^ k \,\mathrm{d} A(z) \mathrm{d} A(w)} \\
			& = \int_{r \mathbb{D}} \int_{r \mathbb{D}} \bigg\{ \frac{1 - \varphi (z)^ {N + 1} \overline{\varphi (w)}^ {N + 1}}{\big[ 1 - \varphi (z) \overline{\varphi (w)} \big]^2} - (N + 1) \frac{\varphi (z)^ {N + 1} \overline{\varphi (w)}^ {N + 1}}{1 - \varphi (z) \overline{\varphi (w)}} \bigg\} \\
			& \, \quad  \times \bigg\{ \frac{1 - \psi (w)^ {N + 1} \overline{\psi (z)}^ {N + 1}}{\big[ 1 - \psi (w) \overline{\psi (z)} \big]^ 2} - (N + 1) \frac{\psi (w)^ {N + 1} \overline{\psi (z)}^ {N + 1}}{1 - \psi (w) \overline{\psi (z)}} \bigg\} \,\mathrm{d} A(z) \mathrm{d} A(w).
		\end{align*}
		Letting $N \rightarrow \infty$, we obtain that
		\begin{align*}
			\sum_{j, k = 0}^ {\infty} {\Big| \int_{r \mathbb{D}} (C_{\varphi} \sigma_j) (z) \overline{(C_{\psi} \sigma_k) (z)} \,\mathrm{d} A(z) \Big|^ 2} = \int_{r \mathbb{D}} \int_{r \mathbb{D}} \frac{\mathrm{d} A(z) \mathrm{d} A(w)}{\big[ 1 - \varphi (z) \overline{\varphi (w)} \big]^ 2 \big[ 1 - \psi (w) \overline{\psi (z)} \big]^ 2}.
		\end{align*}
		It follows from the assumption and Fatou's lemma that
		\begin{align*}
			\sum_{j, k = 0}^ {\infty} {\left| \langle C_{\psi}^ * C_{\varphi} \sigma_n, \sigma_n \rangle_{L^ 2 (\mathbb{D})} \right|^ 2} & = \sum_{j, k = 0}^ {\infty} {\Big| \int_{\mathbb{D}} (C_{\varphi} \sigma_j) (z) \overline{(C_{\psi} \sigma_k) (z)} \,\mathrm{d} A(z) \Big|^ 2} \\
			& \leqslant \varliminf_{n \rightarrow \infty} {\sum_{j, k = 0}^ {\infty} {\Big| \int_{r_n \mathbb{D}} (C_{\varphi} \sigma_j) (z) \overline{(C_{\psi} \sigma_k) (z)} \,\mathrm{d} A(z) \Big|^ 2}} \\
			& \leqslant \int_{\mathbb{D}} \int_{\mathbb{D}} \frac{\mathrm{d} A(z) \mathrm{d} A(w)}{\big| 1 - \varphi (z) \overline{\varphi (w)} \big|^ 2 \big| 1 - \psi (w) \overline{\psi (z)} \big|^ 2},
		\end{align*}
		where $\{ r_n \}_{n = 1}^ {\infty}$ is any sequence of positive numbers increasing to $1$. This completes the proof.
