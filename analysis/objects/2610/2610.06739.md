<!-- lemma-1: body_md -->
\label{lem:ordered}
Let $V_1,\ldots,V_n\subseteq V$, and let $\pi\in\mathfrak{S}_n$ be a permutation. Then
\begin{align}\label{ordered-Grassmann}
\dim\Bigl(\sum_{i\in[n]}V_i\Bigr)=\sum_{k\in[n]}
\Bigl[\dim(V_{\pi(k)})-\dim\Bigl(V_{\pi(k)}\cap\sum_{l<k}V_{\pi(l)}\Bigr)\Bigr],
\end{align}
where the sum over $l<1$ is understood to be $\{0\}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Set $U_k\coloneq\sum_{l=1}^{k}V_{\pi(l)}$ and $U_0\coloneq\{0\}$. By the classical Grassmann formula,
\begin{equation*}
\dim(U_k)-\dim(U_{k-1})=\dim(V_{\pi(k)})-\dim(U_{k-1}\cap V_{\pi(k)}).
\end{equation*}
Summing over $k\in[n]$ telescopes to \eqref{ordered-Grassmann}.
<!-- end proof-1 -->
<!-- theorem-1: body_md -->
\label{thm:symmetric-Grassmann}
Let $V_1,\ldots,V_n\subseteq V$. Then
\begin{equation}\label{sym-Grassmann}
\dim\Bigl(\sum_{i\in[n]}V_i\Bigr)=\sum_{i\in[n]}
\Bigl[\dim(V_i)-\sum_{\emptyset\neq S\subseteq[n]\setminus\{i\}}c_{n,|S|+1}\,\dim\Bigl(V_i\cap\sum_{j\in S}V_j\Bigr)\Bigr].
\end{equation}
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
Average \eqref{ordered-Grassmann} uniformly over all $n!$ permutations. Fix $i\in[n]$ and $S\subseteq[n]\setminus\{i\}$ with $|S|=s$. The condition $\operatorname{Pred}_{\pi}(i)=S$ means that precisely the elements of $S$ occur before $i$; they can be ordered in $s!$ ways, while the remaining $n-s-1$ elements can be ordered after $i$ in $(n-s-1)!$ ways. Thus, for a uniformly random permutation $\pi$, the probability that exactly the elements of $S$ precede $i$ is
\begin{equation*}
\mathtt{p}\bigl(\operatorname{Pred}_{\pi}(i)=S\bigr)=\frac{s!(n-s-1)!}{n!}=\frac{1}{n\binom{n-1}{s}}=\frac{1}{(s+1)\binom{n}{s+1}}=c_{n,s+1}.
\end{equation*}
Therefore the average contribution of the intersection loss associated with $(i,S)$ is $c_{n,|S|+1}\dim(V_i\cap\sum_{j\in S}V_j)$. Summing over $i$ and all nonempty $S$ gives \eqref{sym-Grassmann}.
<!-- end proof-2 -->
<!-- remark-1: body_md -->
\label{rem:n2}
For $n=2$, Theorem~\ref{thm:symmetric-Grassmann} reduces exactly to the classical Grassmann formula. Indeed, $c_{2,2}=1/2$, and the same intersection $V_1\cap V_2$ occurs once with distinguished index $1$ and once with distinguished index $2$.
<!-- end remark-1 -->
<!-- example-1: body_md -->
For $n=3$, Theorem~\ref{thm:symmetric-Grassmann}
recovers the identity in \cite[Exercise~20, Section~2C]{Axler24}:
\begin{equation}\label{sum-n=3}
\dim(V_1+V_2+V_3)=\sum_{i=1}^{3}\dim(V_i)
-\frac{1}{3}\sum_{1\leq i<j\le3}\dim(V_i\cap V_j)-\frac{1}{3}\sum_{i=1}^{3}
\dim\Bigl(V_i\cap\!\sum_{j\in[3]\setminus\{i\}}V_j\Bigr).
\end{equation}
Here the coefficient $1/3=2\,c_{3,2}$ arises by collecting
the identical terms corresponding to $V_i\cap V_j$ and
$V_j\cap V_i$. More generally, collecting the pairwise
terms over $i<j$ gives the coefficient
$2\,c_{n,2}=1/\binom{n}{2}$.
The first correction records pairwise overlap, while the
second captures, in particular, the genuinely non-distributive
phenomenon that a subspace may meet the sum of two others
nontrivially even when all pairwise intersections vanish.
<!-- end example-1 -->
<!-- example-2: body_md -->
\label{ex:three-lines}
Let $V=\mathbbm{F}^2$, with standard basis $\{e_1,e_2\}$, and consider the three distinct lines through the origin
\begin{equation*}
V_1=\operatorname{span}\{e_1\}, \qquad
V_2=\operatorname{span}\{e_2\}, \qquad
V_3=\operatorname{span}\{e_1+e_2\}.
\end{equation*}
Then $V_i\cap V_j=\{0\}$ for $i\neq j$, whereas $V_i\cap(V_j+V_k)=V_i$ for each permutation $(i,j,k)$ of $(1,2,3)$. Formula \eqref{sum-n=3} therefore gives
\begin{equation*}
\dim(V_1+V_2+V_3)=3-\frac{1}{3}(1+1+1)=2.
\end{equation*}
Thus the terms involving intersections with partial sums recover exactly the information missed by ordinary inclusion--exclusion.
<!-- end example-2 -->
<!-- theorem-2: body_md -->
\label{thm:weighted}
Let $V_1,\ldots,V_n\subseteq V$. Then
\begin{align}\label{weighted-Grassmann}
\dim\Bigl(\sum_{i\in[n]}V_i\Bigr)=\sum_{i\in[n]}\dim(V_i)
-\sum_{i\in[n]}\sum_{S\subseteq[n]\setminus\{i\}}\mathtt{p}_{\mu}(i,S)
\dim\Bigl(V_i\cap\sum_{j\in S}V_j\Bigr).
\end{align}
The term $S=\emptyset$ vanishes because the empty sum of subspaces is $\{0\}$.
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
For any fixed ordering $\pi$, Lemma~\ref{lem:ordered} adds the dimensions of the $V_i$ and subtracts, for each $i$, its intersection with the sum of the subspaces preceding it. Thus \eqref{ordered-Grassmann} can be written as
\begin{equation*}
\dim\Bigl(\sum_{i\in[n]}V_i\Bigr)=\sum_{i\in[n]}\dim(V_i)
-\sum_{i\in[n]}\dim\Bigl(V_i\cap\sum_{j\in\operatorname{Pred}_{\pi}(i)}V_j\Bigr).
\end{equation*}
The left-hand side and the first sum do not depend on $\pi$. Taking the expectation over $\pi$ with distribution $\mu$, and grouping the orderings by the set $S=\operatorname{Pred}_{\pi}(i)$, gives
\begin{equation*}
\mathtt{E}_{\mu}\Bigl[\dim\Bigl(V_i\cap\sum_{j\in\operatorname{Pred}_{\pi}(i)}V_j\Bigr)\Bigr]=
\sum_{S\subseteq[n]\setminus\{i\}}\mathtt{p}_{\mu}(i,S)
\dim\Bigl(V_i\cap\sum_{j\in S}V_j\Bigr).
\end{equation*}
Substituting this identity for each $i$ proves
\eqref{weighted-Grassmann}.

<!-- example-3: body_md -->
\label{ex:weighted}
Let $V=\mathbbm{F}^2$, $V_1=\operatorname{span}\{e_1\}$, and $V_2=V$. Give the ordering $(1,2)$ probability $3/4$ and the ordering $(2,1)$ probability $1/4$. In the first ordering, adding $V_1$ and then $V_2$ contributes $1+(2-1)=2$; in the second, adding $V_2$ and then $V_1$ contributes $2+(1-1)=2$. Their weighted average is
\begin{equation*}
\frac{3}{4}[1+(2-1)]+\frac{1}{4}[2+(1-1)]=2.
\end{equation*}
In the notation of the theorem, $\mathtt{p}_{\mu}(2,\{1\})=3/4$ and $\mathtt{p}_{\mu}(1,\{2\})=1/4$. Since $V_1\cap V_2=V_1$ has dimension $1$, formula \eqref{weighted-Grassmann} gives the same result:
\begin{equation*}
\dim(V_1+V_2)=1+2-\frac{3}{4}(1)-\frac{1}{4}(1)=2.
\end{equation*}
The weights change the average contributions of $V_1$ and $V_2$ to $3/4$ and $5/4$, respectively, while their sum remains $2$.

<!-- theorem-3: body_md -->
\label{thm:dual-Grassmann}
Let $V_1,\ldots,V_n\subseteq V$. Then
\begin{equation}\label{dual-Grassmann}
\dim\Bigl(\bigcap_{i\in[n]}V_i\Bigr)=\sum_{i\in[n]}\dim(V_i)-\sum_{i\in[n]}\sum_{\emptyset\neq S\subseteq[n]\setminus\{i\}}c_{n,|S|+1}\,\dim\Bigl(V_i+\bigcap_{j\in S}V_j\Bigr).
\end{equation}
<!-- end theorem-3 -->
<!-- proof-4: body_md -->
Apply Theorem~\ref{thm:symmetric-Grassmann} in $V^\vee$ to $V_1^{\circ},\ldots,V_n^{\circ}$. Let $d=\dim(V)$. Using \eqref{annihilator},
\begin{equation*}
\dim\Bigl(V_i^{\circ}\cap\sum_{j\in S}V_j^{\circ}\Bigr)=
d-\dim\Bigl(V_i+\bigcap_{j\in S}V_j\Bigr).
\end{equation*}
Moreover,
\begin{equation*}
\dim\Bigl(\sum_{i\in[n]}V_i^{\circ}\Bigr)=d-\dim\Bigl(\bigcap_{i\in[n]}V_i\Bigr),
\qquad \dim(V_i^{\circ})=d-\dim(V_i).
\end{equation*}
It remains only to observe that the sum of all coefficients in the double correction term equals $n-1$. Indeed,
\begin{align*}
\sum_{i\in[n]}
\sum_{\emptyset\neq S\subseteq[n]\setminus\{i\}}c_{n,|S|+1}=\sum_{i\in[n]}
\sum_{\emptyset\neq S\subseteq[n]\setminus\{i\}}
\frac{1}{(|S|+1)\binom{n}{|S|+1}}&=\sum_{i\in[n]}\sum_{s=1}^{n-1}
\binom{n-1}{s}\frac{1}{(s+1)\binom{n}{s+1}}\\
&=\sum_{i\in[n]}\sum_{s=1}^{n-1}\frac{1}{n}=n-1.
\end{align*}
The ambient-dimension terms therefore cancel, leaving \eqref{dual-Grassmann}.
<!-- end proof-4 -->
<!-- remark-3: body_md -->
For $n=2$, formula \eqref{dual-Grassmann} gives
\begin{equation*}
\dim(V_1\cap V_2)=\dim(V_1)+\dim(V_2)-\dim(V_1+V_2),
\end{equation*}
which is again the classical Grassmann identity.

<!-- theorem-4: body_md -->
\label{thm:TMS-comparison}
Let $V_1,\ldots,V_n\subseteq V$, and let $W_i$ and $Y$ be defined by \eqref{Wi-Y}. Then
\begin{equation}\label{TMS-comparison}
\sum_{i\in[n]}\sum_{\emptyset\neq S\subseteq[n]\setminus\{i\}}c_{n,|S|+1}\,\dim\Bigl(V_i\cap\sum_{j\in S}V_j\Bigr)=\sum_{i\in[n]}\dim(V_i\cap W_i)-\dim(Y).
\end{equation}
<!-- end theorem-4 -->
<!-- proof-5: body_md -->
Rearranging Theorem~\ref{thm:symmetric-Grassmann} shows that the left-hand side of \eqref{TMS-comparison} is
\begin{equation*}
\sum_{i\in[n]}\dim(V_i)-\dim\Bigl(\sum_{i\in[n]}V_i\Bigr).
\end{equation*}
On the other hand, rearranging \eqref{MS-corollary} gives
\begin{equation*}
\sum_{i\in[n]}\dim(V_i)-\dim\Bigl(\sum_{i\in[n]}V_i\Bigr)=
\sum_{i\in[n]}\dim(V_i\cap W_i)-\dim(Y),
\end{equation*}
which proves the claim.

<!-- remark-4: body_md -->
\label{rem:TMS-structural}
Theorem~\ref{thm:TMS-comparison} makes explicit that the two identities \eqref{sym-Grassmann} and \eqref{TMS-formula} resolve the same total dimension correction through different subspace data. The Tian--Makrides--Szemberg expression uses the maximal proper sums $W_i$ and their common intersection $Y$. The symmetric Grassmann-type identity instead uses intersections of a distinguished subspace with partial sums indexed by all nonempty subsets of the remaining indices. These terms can be grouped by cardinality to form the defect profile studied below.

<!-- proposition-1: body_md -->
\label{prop:Shapley}
For the rank function $\mathfrak{r}$ in \eqref{rank-function-S},
\begin{equation}\label{Shapley-value-Grassmann}
\phi_i(\mathfrak{r})=\dim(V_i)-
\sum_{\emptyset\neq S\subseteq[n]\setminus\{i\}}
c_{n,|S|+1}\,\dim\Bigl(V_i\cap\sum_{j\in S}V_j\Bigr).
\end{equation}
Consequently, Theorem~\ref{thm:symmetric-Grassmann} is precisely the Shapley efficiency identity \eqref{Shapley-efficiency} for the linearly representable polymatroid rank function $\mathfrak{r}$, expressed through Grassmann intersection losses.
<!-- end proposition-1 -->
<!-- proof-6: body_md -->
Substitute the Grassmann marginal identity \eqref{rank-marginal} into \eqref{Shapley-value} and use
\begin{equation*}
\frac{|S|!(n-|S|-1)!}{n!}=\frac{1}{(|S|+1)\binom{n}{|S|+1}}=c_{n,|S|+1}.
\end{equation*}
For each fixed $i$, these weights sum to one over $S\subseteq[n]\setminus\{i\}$, so the terms involving $\dim(V_i)$ combine to give $\dim(V_i)$. The term indexed by $S=\emptyset$ has zero intersection loss, yielding \eqref{Shapley-value-Grassmann}. Summing over $i$ and using \eqref{Shapley-efficiency} gives Theorem~\ref{thm:symmetric-Grassmann}.

<!-- remark-5: body_md -->
\label{rem:expected}
In the predecessor notation, the allocation to $V_i$ takes

<!-- example-4: body_md -->
\label{ex:polymatroid-Shapley}
Let $e_1,e_2,e_3$ be the standard basis of $\mathbbm{F}^3$, and set
\[
V_1=\operatorname{span}\{e_1,e_2\},\qquad
V_2=\operatorname{span}\{e_1,e_3\},\qquad
V_3=\operatorname{span}\{e_1,e_2+e_3\}.
\]
Each plane has dimension $2$, any two span $\mathbbm{F}^3$, and all pairwise and triple intersections equal $\operatorname{span}\{e_1\}$. Ordinary inclusion--exclusion incorrectly gives $6-3+1=4$. For distinct $i,j,l\in[3]$, one has $\dim(V_i\cap V_j)=\dim(V_i\cap V_l)=1$ and $\dim\bigl(V_i\cap(V_j+V_l)\bigr)=2$. Since $c_{3,2}=1/6$ and $c_{3,3}=1/3$, the symmetric Grassmann-type formula gives
\[
\dim(V_1+V_2+V_3)=\sum_{i=1}^{3}
\Bigl[2-\Bigl(\frac16+\frac16+\frac23\Bigr)\Bigr]=3.
\]
In a uniformly random ordering, each plane contributes $2$, $1$, or $0$ new dimensions with equal probabilities. Proposition~\ref{prop:Shapley} identifies each Grassmann contribution with its Shapley share:
\[
\underbrace{2-\Bigl(\frac{1}{6}+\frac{1}{6}+\frac{2}{3}\Bigr)}_{\text{dimension minus averaged intersection loss}}=\underbrace{\frac{1}{3}\,(2+1+0)}_{\text{average marginal dimension}}=\phi_i(\mathfrak{r})=1.
\]
The rank function satisfies $\mathfrak{r}(\{i\})=2$ and $\mathfrak{r}(S)=3$ for $|S|\geq2$. Its independence and base polytopes \cite{Fujishige05} are
\begin{align*}
P(\mathfrak{r})&\coloneq
\Bigl\{x\in\mathbbm{R}_{\geq0}^{3}\;\Big|\;
\sum_{i\in S}x_i\leq\mathfrak{r}(S) \quad\forall\,S\subseteq[3]\Bigr\}
=\bigl\{x\in[0,2]^3\mid x_1+x_2+x_3\leq3\bigr\}, \\
\mathcal{B}(\mathfrak{r})
&=\bigl\{x\in[0,2]^3\mid x_1+x_2+x_3=3\bigr\}.
\end{align*}
The six vertices of $\mathcal{B}(\mathfrak{r})$ are the ordered marginal allocations, the permutations of $(2,1,0)$. Their centre $(1,1,1)$ is the Shapley allocation, whose coordinate sum recovers the dimension $3$ by Shapley efficiency; see Figure~\ref{fig:polymatroid-Shapley}.

<!-- definition-1: body_md -->
\label{def:defect}
For $V_1,\ldots,V_n\subseteq V$, define
\begin{equation}\label{defect-dim}
\operatorname{def}(V_1,\ldots,V_n)\coloneq\sum_{i\in[n]}\dim(V_i)-
\dim\Bigl(\sum_{i\in[n]}V_i\Bigr).
\end{equation}
For $2\leq k\leq n$, define the $k$-th graded interaction defect by
\begin{equation}\label{kth-defect-Ak}
A_k(V_1,\ldots,V_n)\coloneq c_{n,k} \sum_{\substack{T\subseteq[n]\\|T|=k}}
\sum_{i\in T}\dim\Bigl(V_i\cap\sum_{j\in T\setminus\{i\}}V_j\Bigr).
\end{equation}
<!-- end definition-1 -->
<!-- corollary-1: body_md -->
\label{cor:defect-decomposition}
For every finite family $V_1,\ldots,V_n\subseteq V$,
\begin{equation}\label{defect-dim-decomposition}
\operatorname{def}(V_1,\ldots,V_n)=\sum_{k=2}^{n}A_k(V_1,\ldots,V_n).
\end{equation}
In particular, each contribution is nonnegative.
<!-- end corollary-1 -->
<!-- proof-7: body_md -->
Reindexing Theorem~\ref{thm:symmetric-Grassmann} by $T=S\cup\{i\}$, writing $k=|T|$, and using \eqref{defect-dim} and \eqref{kth-defect-Ak} gives \eqref{defect-dim-decomposition}.
<!-- end proof-7 -->
<!-- remark-6: body_md -->
Averaging the ordered identity \eqref{ordered-Grassmann} over uniformly random permutations gives
\begin{equation}
\operatorname{def}(V_1,\ldots,V_n)=\mathtt{E}\Bigl[\sum_{i\in[n]}
\dim\Bigl(V_i\cap\sum_{j\in\operatorname{Pred}_{\pi}(i)}V_j\Bigr)\Bigr].
\end{equation}
Thus the dimension defect is the expected total overlap encountered when the subspaces are inserted in a uniformly random order.
<!-- end remark-6 -->
<!-- theorem-5: body_md -->
\label{thm:Ak-monotone}
For every family $V_1,\ldots,V_n\subseteq V$,
\begin{equation}\label{kth-defect-Ak-monotone}
A_2\leq A_3\le\cdots\leq A_n.
\end{equation}
<!-- end theorem-5 -->
<!-- proof-8: body_md -->
Fix $i\in[n]$ and define $f_i(S)\coloneq\dim(V_i\cap\sum_{j\in S}V_j)$ for $S\subseteq[n]\setminus\{i\}$. The function $f_i$ is monotone under inclusion. Let $m=n-1$, and for $0\leq s\leq m$ let
\begin{equation*}
M_{i,s}\coloneq\frac{1}{\binom{m}{s}}
\sum_{\substack{S\subseteq[n]\setminus\{i\}\\|S|=s}}f_i(S).
\end{equation*}
We claim that $M_{i,s}\leq M_{i,s+1}$ for $0\leq s\leq m-1$. Double count pairs $(S,T)$ with $S\subset T$, $|S|=s$, and $|T|=s+1$. Since $f_i(S)\leq f_i(T)$, then $(m-s)\sum_{|S|=s}f_i(S)\leq(s+1)\sum_{|T|=s+1}f_i(T)$. Using $(m-s)\binom{m}{s}=(s+1)\binom{m}{s+1}$ gives $M_{i,s}\leq M_{i,s+1}$. Finally, since $k\binom{n}{k}=n\binom{n-1}{k-1}$, we may rewrite \eqref{kth-defect-Ak} as $A_k=\frac{1}{n}\sum_{i\in[n]}M_{i,k-1}$. For $2\leq k\leq n-1$, averaging the inequalities $M_{i,k-1}\leq M_{i,k}$ over $i$ proves \eqref{kth-defect-Ak-monotone}.
<!-- end proof-8 -->
<!-- corollary-2: body_md -->
\label{cor:defect-bounds}
Let $p\coloneq\frac{1}{\binom{n}{2}}\sum_{1\leq i<j\leq n}\dim(V_i\cap V_j)$ be the average pairwise intersection dimension, and let $q\coloneq\frac{1}{n}\sum_{i\in[n]}\dim(V_i\cap\sum_{j\neq i}V_j)$ be the average dimension of the intersection of each subspace with the sum of all the others. Then
\begin{equation}\label{defect-dim-bounds}
(n-1)p\leq\operatorname{def}(V_1,\ldots,V_n)\leq(n-1)q.
\end{equation}
<!-- end corollary-2 -->
<!-- proof-9: body_md -->
By definition, $A_2=p$ and $A_n=q$. Hence Theorem~\ref{thm:Ak-monotone} and Corollary~\ref{cor:defect-decomposition} imply
\begin{equation*}
(n-1)A_2\le\sum_{k=2}^{n}A_k\le(n-1)A_n.
\end{equation*}
<!-- end proof-9 -->
<!-- corollary-3: body_md -->
\label{cor:direct-sum}
The following are equivalent:
\begin{enumerate}[label=\textnormal{(\roman*)}]
\item $\dim(\sum_{i\in[n]}V_i)=\sum_{i\in[n]}\dim(V_i)$;
\item $A_k=0$ for every $2\leq k\leq n$;
\item $A_n=0$;
\item $V_i\cap\sum_{j\neq i}V_j=\{0\}$ for every $i\in[n]$;
\item $\sum_{i\in[n]}V_i=\bigoplus_{i\in[n]}V_i$.
\end{enumerate}
<!-- end corollary-3 -->
<!-- proof-10: body_md -->
The equivalence of (i) and (ii) follows from \eqref{defect-dim-decomposition} and nonnegativity. By monotonicity, (ii) and (iii) are equivalent. Since $A_n$ is the average of the nonnegative numbers
$\dim(V_i\cap\sum_{j\neq i}V_j)$, conditions (iii) and (iv) are equivalent. Finally, (iv) is the standard characterization of an internal direct sum, giving (iv)$\Leftrightarrow$(v).
<!-- end proof-10 -->
<!-- remark-7: body_md -->
\label{rem:n4}
For $n=4$, the coefficients in Theorem~\ref{thm:symmetric-Grassmann} are $\frac{1}{12}$ for $|S|=1$, $\frac{1}{12}$ for $|S|=2$, and $\frac{1}{4}$ for $|S|=3$. Thus, $\operatorname{def}(V_1,V_2,V_3,V_4)=A_2+A_3+A_4$, where $A_2\leq A_3\leq A_4$. The equality separates pairwise overlap, three-subspace overlap with a distinguished member, and one-versus-all-others overlap without introducing alternating inclusion--exclusion terms.

<!-- proposition-2: body_md -->
\label{prop:Bk-local}
For $2\leq k\leq n-1$,
\begin{equation}\label{defect-gap-Bk-local}
B_k=\frac{c_{n,k}}{n-k}
\sum_{i\in[n]}\sum_{\substack{S\subseteq[n]\setminus\{i\}\\|S|=k-1}}
\sum_{j\in[n]\setminus(S\cup\{i\})}\Bigl[\dim\Bigl(V_i\cap\Bigl(V_j+\sum_{l\in S}V_l\Bigr)\Bigr)-\dim\Bigl(V_i\cap\sum_{l\in S}V_l\Bigr)\Bigr].
\end{equation}
In particular, $B_k$ is the average additional overlap created by adjoining one further subspace to a $(k-1)$-element predecessor family.
<!-- end proposition-2 -->
<!-- proof-11: body_md -->
Fix $i\in[n]$. In the sum in \eqref{defect-gap-Bk-local}, each $k$-element set $T\subseteq[n]\setminus\{i\}$ occurs exactly $k$ times as $T=S\cup\{j\}$, once for each choice of $j\in T$. Each $(k-1)$-element set $S\subseteq[n]\setminus\{i\}$ occurs exactly $n-k$ times in the subtracted term, once for each admissible choice of $j$. Hence, using the definitions of $f_i$ and $M_{i,s}$ from the proof of Theorem~\ref{thm:Ak-monotone}, the sum over $S$ and $j$ for this fixed $i$ is
\begin{equation*}
k\binom{n-1}{k}M_{i,k}
-(n-k)\binom{n-1}{k-1}M_{i,k-1}=(n-k)\binom{n-1}{k-1}
(M_{i,k}-M_{i,k-1}),
\end{equation*}
where $k\binom{n-1}{k}=(n-k)\binom{n-1}{k-1}$. Multiplying by $c_{n,k}/(n-k)$ and summing over $i$ gives
\begin{equation*}
\frac{1}{n}\sum_{i\in[n]}(M_{i,k}-M_{i,k-1})
=A_{k+1}-A_k=B_k,
\end{equation*}
because $c_{n,k}=1/(n\binom{n-1}{k-1})$ and $A_\ell=\frac{1}{n}\sum_{i\in[n]}M_{i,\ell-1}$. This proves \eqref{defect-gap-Bk-local}.
<!-- end proof-11 -->
<!-- corollary-4: body_md -->
\label{cor:Bk-quotient}
For every $2\leq k\leq n-1$,
\begin{equation}\label{Bk-quotient}
B_k=\frac{1}{\binom{n}{k-1}\binom{n-k+1}{2}}
\sum_{\substack{S\subseteq[n]\\|S|=k-1}}
\sum_{\substack{j<l\\j,l\in[n]\setminus S}}
\dim(\overline{V}_j^{\,S}\cap\overline{V}_l^{\,S}).
\end{equation}
Consequently, $B_k=0$ if and only if
\begin{equation}\label{Bk-zero-quotient}
(V_j+U_S)\cap(V_l+U_S)=U_S
\end{equation}
for every $S\subseteq[n]$ with $|S|=k-1$ and every pair of
distinct indices $j,l\in[n]\setminus S$.
<!-- end corollary-4 -->
<!-- proof-12: body_md -->
Let $Q_S:V\to V/U_S$ be the quotient map. Its restriction to $V_j\cap(V_l+U_S)$ has image $\overline{V}_j^{\,S}\cap\overline{V}_l^{\,S}$ and kernel $V_j\cap U_S$. Hence
\[
\dim(\overline{V}_j^{\,S}\cap\overline{V}_l^{\,S})
=\dim\bigl(V_j\cap(V_l+U_S)\bigr)-\dim(V_j\cap U_S).
\]
Substitute this identity into Proposition~\ref{prop:Bk-local}. Each unordered pair occurs twice in its ordered-pair sum, and
\[
n\binom{n-1}{k-1}(n-k)=2\binom{n}{k-1}\binom{n-k+1}{2}.
\]
This gives \eqref{Bk-quotient}. Since every summand is nonnegative, $B_k=0$ precisely when all quotient intersections vanish. Taking their inverse images under $Q_S$ gives \eqref{Bk-zero-quotient}.
<!-- end proof-12 -->
<!-- theorem-6: body_md -->
\label{thm:exact-gap}
For every finite family of subspaces $V_1,\ldots,V_n\subseteq V$ with $n\geq2$,
\begin{equation}\label{lower-gap-exact}
\operatorname{def}(V_1,\ldots,V_n)=(n-1)A_2+\sum_{k=2}^{n-1}(n-k)B_k,
\end{equation}
and
\begin{equation}\label{upper-gap-exact}
\operatorname{def}(V_1,\ldots,V_n)=(n-1)A_n-\sum_{k=2}^{n-1}(k-1)B_k.
\end{equation}
Thus the errors in the two endpoint estimates of Corollary~\ref{cor:defect-bounds} are themselves resolved into nonnegative contributions indexed by interaction order.
<!-- end theorem-6 -->
<!-- proof-13: body_md -->
Substitute $A_k=A_2+\sum_{j=2}^{k-1}B_j$ into $\operatorname{def}=\sum_{k=2}^n A_k$. The gap $B_j$ occurs $n-j$ times, giving \eqref{lower-gap-exact}. Similarly, substituting $A_k=A_n-\sum_{j=k}^{n-1}B_j$ counts $B_j$ exactly $j-1$ times and gives \eqref{upper-gap-exact}.

<!-- theorem-7: body_md -->
\label{thm:common-core}
Let $V_1,\ldots,V_n\subseteq V$ and set $D\coloneq\bigcap_{i\in[n]}V_i$. The following conditions are equivalent:
\begin{enumerate}[label=\textnormal{(\roman*)}]
\item $\operatorname{def}(V_1,\ldots,V_n)=(n-1)A_2$;

\item $\operatorname{def}(V_1,\ldots,V_n)=(n-1)A_n$;

\item $A_2=A_3=\cdots=A_n$;

\item $V_i\cap\sum_{j\neq i}V_j=D$ for every $i\in[n]$;

\item the quotient subspaces $V_1/D,\ldots,V_n/D\subseteq V/D$ form an internal direct sum;

\item for every $i\in[n]$ there exists a complement $U_i$ of $D$ in $V_i$ such that $D\oplus U_1\oplus\cdots\oplus U_n$ is a direct sum.
\end{enumerate}
Whenever these conditions hold,
\begin{equation}\label{kth-defect-Ak-common-core}
A_2=\cdots=A_n=\dim(D)
\end{equation}
and
\begin{equation}\label{def-common-core}
\operatorname{def}(V_1,\ldots,V_n)=(n-1)\dim(D).
\end{equation}
<!-- end theorem-7 -->
<!-- proof-14: body_md -->
By Theorem~\ref{thm:exact-gap}, equality in either endpoint bound holds if and only if $B_k=0$ for every $2\leq k\leq n-1$. Hence conditions (i), (ii), and (iii) are equivalent. Assume (iii). Using $M_{i,s}$ and its monotonicity from the proof of Theorem~\ref{thm:Ak-monotone}, $A_n-A_2=0$ forces $M_{i,n-1}=M_{i,1}$ for every $i$. For fixed $i$ and every $j\neq i$, each $\dim(V_i\cap V_j)$ is at most $\dim(V_i\cap\sum_{k\ne i}V_k)=M_{i,n-1}$, while their average is $M_{i,1}=M_{i,n-1}$. Hence every one of these dimensions equals $M_{i,n-1}$, and containment gives $V_i\cap V_j=V_i\cap\sum_{k\ne i}V_k$ for all $j\ne i$. Since $V_i\cap V_j=V_j\cap V_i$, these spaces have a common value for all $i$, and that value is $D=\bigcap_{i\in[n]}V_i$. This proves (iv). Condition (iv) is equivalent to $(V_i/D)\cap\sum_{j\neq i}(V_j/D)=\{0\}$ for every $i$, which is the standard direct-sum criterion for the quotient family. Thus (iv) and (v) are equivalent. Choosing complements $U_i\subseteq V_i$ such that $V_i=D\oplus U_i$ shows that (v) and (vi) are equivalent. Finally, assume (iv). For every nonempty $S\subseteq[n]\setminus\{i\}$, we have $D\subseteq V_i\cap\sum_{j\in S}V_j\subseteq V_i\cap\sum_{j\neq i}V_j=D$. Thus every intersection dimension defining $A_k$ equals $\dim(D)$, so $A_2=\cdots=A_n=\dim(D)$. This proves (iii), completes the equivalence, and establishes \eqref{kth-defect-Ak-common-core}. Equation~\eqref{def-common-core} follows from Corollary~\ref{cor:defect-decomposition}.
<!-- end proof-14 -->
<!-- remark-8: body_md -->
For families of subspaces of equal dimension, condition~\textnormal{(iv)} is the defining property of a sunflower of maximal dimension in \cite[Section~2]{BSSV17}. For such families with constant pairwise intersection dimension, a characterization by successive rank increments is given in \cite[Remark~1.3]{LST22}. Theorem~\ref{thm:common-core} characterizes the common-core decomposition for arbitrary finite subspace families through equality in either averaged defect bound.

<!-- definition-2: body_md -->
\label{def:average-rank}
For $0\leq k\leq n$, define
\begin{equation}
R_k(V_1,\ldots,V_n)\coloneq\frac{1}{\binom{n}{k}}
\sum_{\substack{S\subseteq[n]\\|S|=k}}
\dim\Bigl(\sum_{i\in S}V_i\Bigr),
\end{equation}
where $R_0=0$.
<!-- end definition-2 -->
<!-- theorem-8: body_md -->
\label{thm:average-rank}
For every $2\leq k\leq n$,
\begin{equation}\label{kth-defect-Ak-Rk}
A_k=R_1+R_{k-1}-R_k.
\end{equation}
Equivalently, if
\begin{equation}
\Delta_k\coloneq R_k-R_{k-1}, \qquad 1\leq k\leq n,
\end{equation}
then
\begin{equation}\label{kth-defect-Ak-Delta}
A_k=R_1-\Delta_k.
\end{equation}
Moreover, for $2\leq k\leq n-1$,
\begin{equation}\label{defect-gap-Bk-Rk}
B_k=2R_k-R_{k-1}-R_{k+1}=\Delta_k-\Delta_{k+1}.
\end{equation}
Consequently,
\begin{equation}\label{Rk-concave}
R_{k+1}-R_k\leq R_k-R_{k-1}, \qquad 1\leq k\leq n-1,
\end{equation}
so the average rank profile $(R_k)_{k=0}^{n}$ is discretely concave.
<!-- end theorem-8 -->
<!-- proof-15: body_md -->
For $T\subseteq[n]$ with $|T|=k$ and $i\in T$,
the classical Grassmann formula gives
\begin{equation}\label{pointed-rank}
\dim\Bigl(V_i\cap\sum_{j\in T\setminus\{i\}}V_j\Bigr)=
\dim(V_i)+\dim\Bigl(\sum_{j\in T\setminus\{i\}}V_j\Bigr)
-\dim\Bigl(\sum_{j\in T}V_j\Bigr).
\end{equation}
Average \eqref{pointed-rank} over all pairs $(T,i)$ with $|T|=k$ and $i\in T$. The first term averages to $R_1$, the second to $R_{k-1}$, and the third to $R_k$. This proves \eqref{kth-defect-Ak-Rk}. Equation~\eqref{kth-defect-Ak-Delta} follows immediately. Taking the difference of \eqref{kth-defect-Ak-Rk} at levels $k+1$ and $k$ gives $B_k=A_{k+1}-A_k=2R_k-R_{k-1}-R_{k+1}=\Delta_k-\Delta_{k+1}$. Since $B_k\ge0$, one has $\Delta_2\geq\cdots\geq\Delta_n$. The remaining first step follows from $\Delta_1-\Delta_2=A_2\geq0$. This proves \eqref{Rk-concave}.
<!-- end proof-15 -->
<!-- remark-9: body_md -->
\label{rem:rank-curvature}
Equation~\eqref{defect-gap-Bk-Rk} shows that $B_k$ is the discrete curvature of the average-rank profile: $B_k=\Delta_k-\Delta_{k+1}$. It measures how much the average marginal dimension drops as the number of selected subspaces increases from $k$ to $k+1$. Hence $B_k=0$ means that the average marginal dimension is unchanged across those two steps, while $B_k>0$ records a drop at that scale. Thus the nonnegativity and monotonicity of the defect hierarchy, $0\leq A_2\leq\cdots\leq A_n$, are equivalent to the discrete concavity of $(R_k)_{k=0}^{n}$: $A_k$ measures average overlap at interaction order $k$, whereas $B_k$ measures the loss of average marginal rank when one passes from $k$ to $k+1$ participating subspaces. Notice also that
\begin{equation}
\operatorname{def}(V_1,\ldots,V_n)=nR_1-R_n.
\end{equation}
Under the equivalent conditions of Theorem~\ref{thm:common-core}, with common core $D$, one has
\begin{equation}
R_k=kR_1-(k-1)\dim(D), \qquad 1\leq k\leq n.
\end{equation}

\label{def:dual-defect}
Define
\begin{equation}
\operatorname{def}^{\vee}(V_1,\ldots,V_n)\coloneq
\sum_{i\in[n]}\operatorname{codim}(V_i)-

\begin{equation}
A_k^{\vee}\coloneq c_{n,k}
\sum_{\substack{T\subseteq[n]\\|T|=k}}

\end{equation}

\end{equation}

<!-- theorem-10: body_md -->
\label{thm:Kinser-slack-decomposition}
Let $V_1,\ldots,V_n\subseteq V$, with $n\ge4$, and set $K\coloneq\bigcap_{i=3}^{n}V_i$. Then
\begin{align}\nonumber
\mathcal{K}_n(V_1,\ldots,V_n)=&~[K+V_1+V_2:V_1+V_2]+[(V_1+V_3)\cap(V_1+V_n):V_1+K] \\
&+[(K\!+\!V_1)\cap(K\!+\!V_2):K]+
\sum_{i=4}^{n}\Bigl[(V_i\!+\!V_2)\cap(V_i\!+\!V_{i-1}):
V_i\!+\!\bigcap_{j=2}^{i-1}V_j\Bigr].
\label{Kinser-slack-decomposition}
\end{align}
In particular, every summand on the right-hand side is nonnegative.
<!-- end theorem-10 -->
<!-- proof-20: body_md -->
Following the auxiliary spaces and descending intersection chain in \cite[Section~3.1]{Kinser11}, set $\mathcal{L}_0\coloneq[K+V_1+V_2:K+V_1]$ and $\mathcal{R}_0\coloneq[K+V_2:K]$. By the Grassmann formula applied to $K+V_1$ and $K+V_2$,
\begin{equation}\label{Kinser-middle-slack}
\mathcal{R}_0-\mathcal{L}_0=[(K+V_1)\cap(K+V_2):K].
\end{equation}
Since $V_1+V_2\subseteq K+V_1+V_2$ and $K+V_1\subseteq(V_1+V_3)\cap(V_1+V_n)$, we obtain the exact identity
\begin{align}\nonumber
\mathcal{L}_0=&\dim(V_1+V_2)+\dim(V_1+V_3+V_n)-\dim(V_1+V_3)-\dim(V_1+V_n) \\
&+[K+V_1+V_2:V_1+V_2]+[(V_1+V_3)\cap(V_1+V_n):V_1+K].
\label{Kinser-left-exact}
\end{align}
Indeed,
\begin{equation*}
\dim\bigl((V_1+V_3)\cap(V_1+V_n)\bigr)=\dim(V_1+V_3)+\dim(V_1+V_n)-\dim(V_1+V_3+V_n).
\end{equation*}
On the other hand,
\begin{equation}\label{Kinser-chain}
\mathcal{R}_0=[V_2:V_2\cap K]=\sum_{i=3}^{n}
\Bigl[\bigcap_{j=2}^{i-1}V_j:\bigcap_{j=2}^{i}V_j\Bigr].
\end{equation}
For $i=3$, the corresponding term is
\begin{equation}\label{Kinser-i3}
[V_2:V_2\cap V_3]=\dim(V_2+V_3)-\dim(V_3).
\end{equation}
For $i\geq4$, put $X_i\coloneq\bigcap_{j=2}^{i-1}V_j$. Then $[X_i:X_i\cap V_i]=[V_i+X_i:V_i]$. Moreover, $V_i+X_i\subseteq(V_i+V_2)\cap(V_i+V_{i-1})$, and therefore
\begin{align}\nonumber
[X_i:X_i\cap V_i]=&\dim(V_2+V_i)+\dim(V_{i-1}+V_i)-\dim(V_2+V_{i-1}+V_i)
-\dim(V_i) \\
&-[(V_i+V_2)\cap(V_i+V_{i-1}):V_i+X_i].
\label{Kinser-chain-exact}
\end{align}
Substituting \eqref{Kinser-i3} and \eqref{Kinser-chain-exact} into \eqref{Kinser-chain}, and then using \eqref{Kinser-middle-slack} and \eqref{Kinser-left-exact}, gives \eqref{Kinser-slack-decomposition}.
<!-- end proof-20 -->
<!-- corollary-8: body_md -->
\label{cor:Kinser-equality}
Recall that $K=\bigcap_{i=3}^{n}V_i$. Then $\mathcal{K}_n(V_1,\ldots,V_n)=0$ if and only if all of the following conditions hold:
\begin{enumerate}
\item $K\subseteq V_1+V_2$,
\item $(V_1+V_3)\cap(V_1+V_n)=V_1+K$,
\item $(K+V_1)\cap(K+V_2)=K$,
\item $(V_i+V_2)\cap(V_i+V_{i-1})=V_i+\bigcap_{j=2}^{i-1}V_j$ for every $4\leq i\leq n$.
\end{enumerate}
<!-- end corollary-8 -->

Every term on the right-hand side of \eqref{Kinser-slack-decomposition} is nonnegative. Hence their sum vanishes if and only if each term vanishes. The four stated conditions are precisely the corresponding equality conditions.

<!-- theorem-11: body_md -->
\label{thm:symmetrized-Kinser}
For every arrangement $V_1,\ldots,V_n\subseteq V$, with $n\ge4$,
\begin{equation}
\frac{1}{n!}\sum_{\pi\in\mathfrak{S}_n}\mathcal{K}_n^{\pi}
=(n-2)B_2.
\end{equation}
Equivalently, in terms of the average rank profile,
\begin{equation}
\frac{1}{n!}\sum_{\pi\in\mathfrak{S}_n}\mathcal{K}_n^{\pi}
=(n-2)(2R_2-R_1-R_3).
\end{equation}
<!-- end theorem-11 -->

Under uniform permutation averaging, a rank term on $k$ distinct labels has average $R_k$. In \eqref{Kinser-slack}, the total coefficients of singleton, pair, and triple ranks are respectively $-(n-2)$, $3+2(n-3)-1=2(n-2)$, and $-(n-2)$. The average is therefore $(n-2)(2R_2-R_1-R_3)$, which equals $(n-2)B_2$ by \eqref{defect-gap-Bk-Rk}.

Thus the representability-specific information in the Kinser hierarchy is inherently nonsymmetric with respect to the labels, whereas its fully symmetric component is already encoded by the graded defect hierarchy.

<!-- theorem-12: body_md -->
\label{thm:conditioned-Kinser}
Let $n\ge4$. For every $2\leq k\leq n-2$,
\begin{equation}\label{defect-gap-Bk-Kinser-average}
B_k=\frac{1}{(n-k)\binom{n}{k-2}} \sum_{\substack{S\subseteq[n]\\|S|=k-2}}
\frac{1}{(n-k+2)!}\sum_{\pi\in\mathfrak{S}([n]\setminus S)}
\mathcal{K}_{n-k+2}^{S,\pi}.
\end{equation}
<!-- end theorem-12 -->
<!-- proof-23: body_md -->
Fix $S\subseteq[n]$ with $|S|=k-2$, and set $m\coloneq n-k+2\ge4$. Let $(R_j^{S})_{j=0}^{m}$ be the average rank profile of the quotient arrangement. Explicitly, for $0\leq j\leq m$,
\begin{equation*}
R_j^{S}=\frac{1}{\binom{m}{j}}
\sum_{\substack{T\subseteq[n]\setminus S\\|T|=j}}
\Bigl[\dim\Bigl(U_S+\sum_{i\in T}V_i\Bigr)-\dim(U_S)\Bigr].
\end{equation*}
Applying Theorem~\ref{thm:symmetrized-Kinser} to the $m$
quotient subspaces gives
\begin{equation}\label{fixed-S-Kinser}
\frac{1}{m!}\sum_{\pi\in\mathfrak{S}([n]\setminus S)}
\mathcal{K}_{m}^{S,\pi}=(n-k)(2R_2^{S}-R_1^{S}-R_3^{S}).
\end{equation}
For $0\leq j\leq m$, averaging over all $(k-2)$-element subsets $S$ and double counting disjoint pairs $(S,T)$ with $|T|=j$ yields
\begin{equation}\label{quotient-rank-average}
\frac{1}{\binom{n}{k-2}}
\sum_{\substack{S\subseteq[n]\\|S|=k-2}}R_j^{S}
=R_{k-2+j}-R_{k-2}.
\end{equation}
Indeed, every $(k-2+j)$-element set occurs as $S\cup T$
exactly $\binom{k-2+j}{k-2}$ times. Consequently, averaging \eqref{fixed-S-Kinser} and using \eqref{quotient-rank-average} gives
\begin{equation*}
\frac{1}{\binom{n}{k-2}}
\sum_{\substack{S\subseteq[n]\\|S|=k-2}}
\frac{1}{m!}\sum_{\pi\in\mathfrak{S}([n]\setminus S)}
\mathcal{K}_{m}^{S,\pi}=(n-k)(2R_k-R_{k-1}-R_{k+1})
=(n-k)B_k,
\end{equation*}
where the last equality follows from \eqref{defect-gap-Bk-Rk}. Dividing by $n-k$ proves \eqref{defect-gap-Bk-Kinser-average}.

<!-- corollary-9: body_md -->
\label{cor:Kinser-Br-zero}
For $2\leq k\leq n-2$, one has $B_k=0$ if and only if $\mathcal{K}_{n-k+2}^{S,\pi}=0$ for every $S\subseteq[n]$ with $|S|=k-2$ and every $\pi\in\mathfrak{S}([n]\setminus S)$. Equivalently, every ordered quotient arrangement obtained by contracting a set of $k-2$ subspaces attains equality in the corresponding Kinser inequality.
<!-- end corollary-9 -->
<!-- proof-24: body_md -->
Every quotient arrangement in \eqref{defect-gap-Bk-Kinser-average} is again linearly representable. Hence each Kinser slack $\mathcal{K}_{n-k+2}^{S,\pi}$ is nonnegative by \eqref{Kinser-nonnegative}. Formula~\eqref{defect-gap-Bk-Kinser-average} therefore expresses $B_k$ as a positive constant times the average of nonnegative quantities. The average vanishes if and only if every summand vanishes.

<!-- remark-12: body_md -->
\label{rem:Kinser-defect-bridge}
Theorem~\ref{thm:conditioned-Kinser} provides a direct bridge between two different hierarchies. The quantities $B_k$ arise from the symmetric dimension-defect hierarchy and measure the discrete curvature of the average rank profile, while the Kinser inequalities encode additional constraints imposed by linear representability. For $2\leq k\leq n-2$, the curvature $B_k$ equals $1/(n-k)$ times the average Kinser slack of quotient arrangements obtained by contracting $k-2$ subspaces, where the average is taken over both the contraction sets and the orderings of the remaining subspaces. In particular, the full symmetrization of a representability-specific inequality does not produce a new symmetric inequality beyond the submodular curvature $B_k\geq0$; instead, the additional representability information is retained in the individual, nonsymmetrized Kinser slacks. The final curvature $B_{n-1}$ has no counterpart in this construction, since the Kinser hierarchy begins with four subspaces.

<!-- example-6: body_md -->
\label{ex:final-curvature}
                                             
Let $n\geq4$, let $e_1,\ldots,e_{n-1}$ be the standard basis of $\mathbbm{F}^{n-1}$, and set
\[
V_i=\operatorname{span}\{e_i\}\quad(1\leq i\leq n-1),
\qquad\qquad
V_n=\operatorname{span}\{e_1+\cdots+e_{n-1}\}.
\]
Every proper selection of these lines is independent, whereas the full selection has rank $n-1$. Thus
\[
R_k=\min\{k,n-1\},\qquad A_2=\cdots=A_{n-1}=0,\qquad A_n=1,

<!-- proof-25: body_md -->
We proceed by induction on $n$. For $n=2$, each of the two pointed terms $\dim(V_1\cap V_2)$ and $\dim(V_2\cap V_1)$ has coefficient $c_{2,2}=1/2$. Together they give the classical Grassmann formula. Assume the formula holds for $n-1$ subspaces. For each $i\in[n]$, set
\begin{equation*}
W_{i}\coloneq\sum_{j\in[n]\setminus\{i\}}V_j.
\end{equation*}
Then $\sum_{j\in[n]}V_j=W_{i}+V_i$, so the two-subspace Grassmann formula gives
\begin{equation*}
\dim\Bigl(\sum_{j\in[n]}V_j\Bigr)=\dim(W_{i})+\dim(V_i)-\dim(W_{i}\cap V_i).
\end{equation*}
Since this identity holds for every $i\in[n]$, averaging over all $i$ yields
\begin{equation*}
\dim\Bigl(\sum_{j\in[n]}V_j\Bigr)=\frac1n\sum_{i\in[n]}
\Bigl[\dim(W_{i})+\dim(V_i)-\dim(W_{i}\cap V_i)\Bigr].
\end{equation*}
By the induction hypothesis, for each fixed $i$,
\begin{equation*}
\dim(W_{i})=\sum_{j\in[n]\setminus\{i\}}
\Bigl[\dim(V_j)-\sum_{\emptyset\neq S\subseteq[n]\setminus\{i,j\}}c_{n-1,|S|+1}\,\dim\Bigl(V_j\cap\sum_{l\in S}V_l\Bigr)\Bigr].
\end{equation*}
Substitute these expansions into the averaged identity. We determine the coefficient of each dimension term and each intersection term separately. The terms involving only individual dimensions simplify to
\begin{equation*}
\frac{1}{n}\sum_{i\in[n]}
\Bigl(\sum_{j\in[n]\setminus\{i\}} \dim(V_j)+\dim(V_i)\Bigr)
=\frac1n\sum_{i\in[n]}\sum_{j\in[n]}\dim(V_j)
=\sum_{j\in[n]}\dim(V_j).
\end{equation*}
Now fix $j\in[n]$ and a nonempty proper subset $S\subsetneq[n]\setminus\{j\}$. Put $k\coloneq|S|+1$; then $2\leq k<n$. We count the choices of the omitted index $i$ for which this fixed correction term appears. The term $\dim(V_j\cap\sum_{l\in S}V_l)$ occurs in the inductive expansion of $\dim(W_{i})$ exactly when $i\notin S\cup\{j\}$. There are $n-k$ such choices of $i$. Hence its coefficient in the averaged correction is
\begin{equation*}
\frac{n-k}{n}\,c_{n-1,k}=\frac{n-k}{n}\,\frac{1}{k\binom{n-1}{k}}=\frac{1}{k\binom{n}{k}}=c_{n,k}.
\end{equation*}
Finally, when $S=[n]\setminus\{j\}$, the term $\dim(W_{j}\cap V_j)$ occurs directly in the Grassmann formula for the single choice $i=j$. Its coefficient is $1/n=c_{n,n}$. Thus every correction term has the coefficient asserted in \eqref{sym-Grassmann}.

<!-- proposition-3: body_md -->
\label{prop:entropy-correspondence}
For $2\leq k\leq n$, the average mutual information is $A_k^{H}$:
\begin{equation}\label{Ak-mutual-information}
A_k^{H}=c_{n,k}\sum_{\substack{T\subseteq[n]\\|T|=k}} \sum_{i\in T} I(X_i;X_{T\setminus\{i\}}).
\end{equation}
Here $c_{n,k}\coloneq(k\binom{n}{k})^{-1}$. For $2\leq k\leq n-1$, the average conditional mutual information is $B_k^{H}$:
\begin{equation}\label{Bk-conditional-mutual-information}
B_k^{H}=\frac{c_{n,k}}{n-k}\sum_{i\in[n]} \sum_{\substack{S\subseteq[n]\setminus\{i\}\\|S|=k-1}} \sum_{j\in[n]\setminus(S\cup\{i\})}I(X_i;X_j\mid X_S).
\end{equation}
Moreover, the total correlation satisfies
\begin{equation}\label{total-correlation}
\sum_{i\in[n]}H(X_i)-H(X_1,\ldots,X_n)=\sum_{k=2}^{n}A_k^{H}.
\end{equation}
In particular, $A_k^{H}\geq0$ and $B_k^{H}\geq0$, and the average entropy profile $(\overline{H}_k)_{k=0}^n$ is discretely concave.
