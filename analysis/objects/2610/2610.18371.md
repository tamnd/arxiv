<!-- corollary-1: body_md -->
\label{main_corollary}
For any natural number $n$, one has
\begin{align}\label{Analogue of BS equation}
p^{(2)}(n) = \sum_{\pi \in \mathcal{D}(n)} (-1)^{ \# (\pi) }s(\pi) \left( \ell(\pi) - s(\pi) \right). 
\end{align}
<!-- end corollary-1 -->
<!-- theorem-1: body_md -->
\label{general_pk(n)}
 Let $p^{(k)}(n)$ be the number of partitions of $n$ into exactly $k$ distinct part sizes. Then, we have 
\begin{align}\label{general k identity}
p^{(k)}(n) = \sum_{\pi \in \mathcal{D}(n)} (-1)^{\#(\pi)-1} \binom{s(\pi)+k-1}{k} - \sum_{\substack{\lambda \in \mathcal{P}(n)\\ \nu(\lambda)\le k-1}} \binom{\ell(\lambda)+k-\nu(\lambda)-1} {k-\nu(\lambda)}.
\end{align}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
For $k=1$, the sum over $\lambda$ is empty and Theorem \ref{general_pk(n)} reduces to \eqref{different form_BS}.
For $k=2$, we recover identity \eqref{Analogue of BS equation}.
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lemma 1}
For any $m,n \in \mathbb{N}$, we obtain
\begin{align}\label{lemma 1 equation}
\sum_{\substack{\pi\in\mathcal D(n)\\ s(\pi)=m}}(-1)^{ \# (\pi) -1}
=\sum_{\lambda\in\mathcal P(n)}(-1)^{\ell(\lambda)-m}\binom{\nu(\lambda)}{\ell(\lambda)-m}.
\end{align}
<!-- end lemma-1 -->
<!-- remark-2: body_md -->
Identity \eqref{lemma 1 equation} is the coefficient of $c^{m}$, for $m \geq 1$, in the partition-theoretic form
\begin{align*}
\sum_{\pi\in\mathcal D(n)}(-1)^{\#(\pi)-1}\bigl(c^{s(\pi)}-1\bigr)
=\sum_{\lambda\in\mathcal P(n)}c^{\ell(\lambda)-\nu(\lambda)}(c-1)^{\nu(\lambda)}
\end{align*}
of an identity of Andrews, Garvan and Liang \cite[Theorem 3.5]{agl13}, which was recorded in this form by Dixit and Maji \cite[Equation~(2.6)]{DM}. The proofs in \cite{agl13,DM} use
$q$-series. To the best of our knowledge, Lemma \ref{lemma 1} gives the first
bijective proof of this refined identity.
<!-- end remark-2 -->
<!-- proof-1: body_md -->
We start by simplifying RHS of \eqref{lemma 1 equation}. Let $\mu$ be the conjugate of $\lambda$. Then $\ell (\lambda) = \#(\mu)$ and $\nu (\lambda) = \nu (\mu)$. Thus, we have 
\begin{align*}
\sum_{\lambda\in\mathcal P(n)}(-1)^{\ell(\lambda)-m}\binom{\nu(\lambda)}{\ell(\lambda)-m} = \sum_{\mu\in\mathcal P(n)}(-1)^{\#(\mu)-m}\binom{\nu(\mu)}{\#(\mu)-m}. 
\end{align*}
Let $B_{\mu}$ be the subset of set of distinct parts of $\mu$ with $|B_{\mu}|=\#(\mu)-m$.
Then $\binom{\nu(\mu)}{\#(\mu)-m}$ counts the total number of such subsets. 
This gives 
\begin{align}\label{wieghted_sum_of_RHS}
\sum_{\mu\in\mathcal P(n)}(-1)^{\#(\mu)-m}\binom{\nu(\mu)}{\#(\mu)-m} =   \sum_{\mu\in\mathcal P(n)}(-1)^{|B_{\mu}|}\binom{\nu(\mu)}{|B_{\mu}|}= \sum_{(\mu, B_{\mu})} (-1)^{|B_{\mu}|},
\end{align}
where the sum runs over the ordered pair $(\mu, B_{\mu})$, which counts the number of subsets $B_{\mu}$ corresponding to each partition $\mu$ with weight $(-1)^{|B_{\mu}|}$ . For all $b \in B_{\mu}$, remove one copy of each $b$ from $\mu$ and call it $\kappa$. Then one can see that $
|\kappa|+\sum_{b \in B_{\mu}}b = n$ and $\#(\kappa) = m.$ It is easy to check that there is a one-one correspondence between pairs $(\mu, B_{\mu})$ and  $(\kappa, B)$, where $\kappa$ is any partition with exactly $m$ part sizes and $B$ is any finite set of positive integers with $|\kappa| + \sum_{b \in B}b= n.$ Adding one copy of each $b \in B$ to $\kappa$ recovers $\mu$.  Again we conjugate $\kappa$ and then remove one copy of $m$ and call it $\rho$. Then $\rho$ is a partition with all parts $\leq m$ and  $|\rho|+m+\sum_{b \in B_{\mu}}b=n$. Next step is to split the set $B_{\mu}$ as $B_{\mu}=S\sqcup T$ with $S=B_{\mu}\cap[1,m]$ and $T=B_{\mu}\cap(m,\infty)$. Thus, the weight in the last sum in \eqref{wieghted_sum_of_RHS} is $(-1)^{|S|}(-1)^{|T|}$.

Since all parts of $\rho$ are $\leq m$ allows us to fix $T$ and work with pair $(\rho, S)$. Let $i$ be the smallest integer in $[1,m]$ that lies either in $S$ or is a part of $\rho$. If none of the above happens, the pair is $(\emptyset,\emptyset)$, which is fixed. Thus, we want to show that for $(\rho, S) \neq (\emptyset,\emptyset)$, This pair must be paired with another pair. If $i\in S$, remove $i$ from $S$ and add a part $i$ to $\rho$. If $i\notin S$, so $i$ is a part of $\rho$, remove one part $i$ from $\rho$ and put $i$ into $S$. This preserves $|\rho|+\sum_{s \in S} s$, changes $|S|$ by one, and hence the associated weight is  $(-1)^{|S|-1}(-1)^{|T|}$.
Consequently only the terms with $|\rho|+\sum_{s \in S}s =0$, i.e.\ $\sum_{t \in T}t=n-m$, survive, each with weight $(-1)^{|T|}$. The sets $T\subset(m,\infty)$ with $\sum_{t \in T}t=n-m$ correspond exactly to $\pi=\{m\}\cup T\in\mathcal D(n)$ with $s(\pi)=m$ and $|T|= \#(\pi)-1$. This is the left side of \eqref{lemma 1 equation}.
<!-- end proof-1 -->
<!-- proof-2: body_md -->
[]
Multiplying both sides of \eqref{lemma 1 equation} by $\binom{m+k-1}{k}$ and summing over all $m$, we obtain
\begin{align*}
\sum_{\pi \in \mathcal{D}(n)} (-1)^{ \# (\pi) -1} \binom{s(\pi)+k-1}{k} &= \sum_{m\geq 1} \sum_{\lambda\in \mathcal{P}(n)} (-1)^{\ell(\lambda)-m} \binom{\nu(\lambda)} {\ell(\lambda)-m} \binom{m+k-1}{k} \\
&= \sum_{\lambda\in P(n)} \sum_{m=1}^{\ell(\lambda)} (-1)^{\ell(\lambda)-m} \binom{\nu(\lambda)}
{\ell(\lambda)-m} \binom{m+k-1}{k}.
\end{align*}
Replacing $m$ by $\ell(\lambda)-m$, one has
\begin{align}\label{final_LHS}
\sum_{\pi \in \mathcal{D}(n)} (-1)^{ \# (\pi) -1} \binom{s(\pi)+k-1}{k} = \sum_{\lambda\in \mathcal{P}(n)} \sum_{m=0}^{\nu(\lambda)} (-1)^m \binom{\nu(\lambda)}{m} \binom{\ell(\lambda)-m+k-1}{k}.
\end{align}
The sum over $m$ runs till $\nu(\lambda)$ as $\binom{\nu(\lambda)}{m}$ vanishes for $m > \nu(\lambda)$. When $\nu(\lambda)=\ell(\lambda)$ the sum acquires the extra term $m=\ell(\lambda)$, which vanishes since $\binom{k-1}{k}=0$ for $k\ge 1$. Hence, the sum may be taken over $0\le m\le \nu(\lambda)$.
Let
\begin{align*}
f(m)=\binom{\ell(\lambda)-m+k-1}{k}.
\end{align*}
Then $f(m)$ is a polynomial of degree $k$ in $m$. Here we utilize the fact that if $f(x)$  is a polynomial of degree $d$, then
\begin{align*}
\sum_{j=0}^{n}(-1)^j
\binom{n}{j}f(j)=0,
\quad\text{if }n>d.
\end{align*}
Thus, \eqref{final_LHS} survives for $\nu(\lambda) \leq k$. This gives 
\begin{align}\label{general k first form}
\sum_{\pi \in \mathcal{D}(n)} (-1)^{ \# (\pi) -1} \binom{s(\pi)+k-1}{k} = \sum_{\substack{\lambda\in \mathcal{P}(n) \\ \nu(\lambda) \leq k}} \sum_{m=0}^{\nu(\lambda)} (-1)^m \binom{\nu(\lambda)}{m} \binom{\ell(\lambda)-m+k-1}{k}.
\end{align}
Now utilizing 
\begin{align*}
\sum_{m=0}^{v} (-1)^m \binom{v}{m} \binom{y-m}{k} = \binom{y-v}{k-v},
\end{align*}
with $v=\nu(\lambda)$ and $y=\ell(\lambda)+k-1$ in the right-hand side of \eqref{general k first form}, we obtain
\begin{align*}
\sum_{\pi \in \mathcal{D}(n)} (-1)^{ \# (\pi) -1} \binom{s(\pi)+k-1}{k} = \sum_{\substack{\lambda\in \mathcal{P}(n) \\ \nu(\lambda) \leq k}} \binom{\ell(\lambda)+k-\nu(\lambda)-1}{k-\nu(\lambda)}.
\end{align*}
Isolating the term corresponding to $\nu(\lambda)=k$ on the right-hand side of the above equation yields \eqref{general k identity}. This completes the proof of Theorem \ref{general_pk(n)}.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
[]
Letting  $k=2$ in \eqref{general k identity}, we obtain
\begin{align}\label{k=2_first_identity}
p^{(2)}(n) &=\sum_{\pi\in\mathcal{D}(n)} (-1)^{\#(\pi)-1}\binom{s(\pi)+1}{2} - \sum_{\substack{\lambda\in\mathcal{P}(n) \\ \nu(\lambda)=1}}\ell(\lambda) \nonumber \\
&=\sum_{\pi\in\mathcal{D}(n)} (-1)^{\#(\pi)-1}\binom{s(\pi)+1}{2} - \sigma(n).
\end{align}
Here, in the last step, we have utilized the fact that partitions with one distinct part size are $(d, d, …, d)$ with $d \mid n$, and their largest part is $d$. Utilize \eqref{BS-general} with $z=1$ to write $\sigma(n) = \sum_{\pi\in\mathcal{D}(n)} (-1)^{\#(\pi)-1}\sum_{j=1}^{s(\pi)} \bigl(\ell(\pi)-s(\pi)+j\bigr)$. Next, using the identity
\begin{equation*}
\sum_{j=1}^{s(\pi)} \bigl(\ell(\pi)-s(\pi)+j\bigr) - \binom{s(\pi)+1}{2} = s(\pi)\bigl(\ell(\pi)-s(\pi)\bigr),
\end{equation*}
on the RHS of \eqref{k=2_first_identity},  we obtain \eqref{Analogue of BS equation}. This completes the proof of Corollary \ref{main_corollary}
