<!-- definition-1: body_md -->
\label{def:stoch-bdd}

Consider random variables $Z_1$ and $Z_2$ with density $f_1$ and $f_2$, respectively. We say that $f_2$ (or $Z_2$) is stochastically bounded below by $f_1$ (or $Z_1)$, denoted $f_1 \prec f_2$ (or $Z_1 \prec Z_2$), if
$$ \mathsf{P}(Z_1 \geq t) \leq \mathsf{P}(Z_2 \geq t), \ \forall t \in \mathbb{R}. $$
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:stoch-bdd-law}

For each $i$, we say that the family of post-change laws $\mathcal{G}_i$ has a least favorable law (LFL) $\bar{G}_i$ with density $\bar{g}_i$ if
$$ \mathcal{L}(\phi (X), \bar{G}_i) \prec \mathcal{L}(\phi (X), G_i), \ \forall G_i \in \mathcal{G}_i, $$
where $\mathcal{L}(\phi (X), G_i)$ denotes the law or distribution of $\phi (X)$ when $X$ has distribution $G_i$.
<!-- end definition-2 -->
<!-- assumption-1: body_md -->
Each post-change family of distributions $\mathcal{G}_i$ has an LFL $\bar{g}_i$ for $i = 1, 2, \dots, m$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
All likelihood ratios appearing in this paper are continuous and have finite positive moments up to second order.
<!-- end assumption-2 -->

\label{alg:2e-rcusum}

Fix $A > 0$, $a_X > 0$, $a_Y > 0$, $N_0 > 0$, $N_X > 0$, and $\mu > 0$. The \textup{2E-RCUSUM} statistic $\{ D_n \}$ is defined as follows:

\begin{enumerate}
\item We start with $D_0 = 0$ and update $\{ D_n \}$ using the observations $\{ Y_n \}$ from the higher-quality experiment using
$$ D_{n + 1}  = D_n + \bar{\ell}_Y (Y_{n + 1}), $$
where $\bar{\ell}_Y (y) = \log \frac{\bar{g}_Y (y)}{f_Y (y)}$, until 
$$ \sigma = \sigma^Y (A) = \min \{ n \geq 1: D_n \notin [0, A) \}. $$
If $D_{\sigma} \geq A$, we stop and declare the change.

\item If $D_{\sigma} < 0$, we start using observations $\{ X_n \}$ from the lower-quality experiment.

    \begin{enumerate}
    \item We first scale the undershoot $U_Y = D_{\sigma}$ by a factor $a_Y$.
    
    \item We update the statistic $\{ D_n \}$ starting at time $\sigma$ using
    $$ D_{n + 1} = D_n + \bar{\ell}_X (X_{n + 1}), $$
    where $\bar{\ell}_X (X_{n + 1}) = \log \frac{\bar{g}_X (x)}{f_X (x)}$, until either the statistic falls below $a_Y U_Y$ at time
    $$ \sigma^X (a_Y U_Y) = \min \{ n \geq \sigma + 1: D_n \not \in [a_Y U_Y, 0) \} $$
    or $N_X$ number of observations of experiment $X$ are used. If $D_{\sigma^X (a_Y U_Y)} > 0$, we reset the statistic $D_n$ to $0$ and go to Step $1$.
    
    \item If $D_{\sigma^X (a_Y U_Y)} < a_Y U_Y$, we do not perform any experiment but still scale the undershoot $U_X~=~D_{\sigma^X (a_Y U_Y)}$ by a factor of $a_X$ and use $a_X U_X$ as the starting level for deterministically updating $D_n$ as
    $$ D_{n + 1} = D_n + \mu. $$
    This continues until either $D_n$ goes above $a_Y U_Y$ or $N_0$ number of deterministic updates are used. At this time, we reset $D_n$ to $a_Y U_Y$ and continue updating $D_n$ as in Step $\textup{c}$.
    \end{enumerate}

\item Stop at the stopping time
\begin{align*}
\tau_{\scriptscriptstyle \textup{2RC}}
& = \tau_{\scriptscriptstyle \textup{2RC}} (A, a_X, a_Y, N_0, N_X, \mu) \\
& = \inf \{ n \geq 1: D_n \geq A \}.
\end{align*}
\end{enumerate}

\label{alg:me-rcusum}

Fix $A > 0$, $a_i > 0$ ($i = 1, 2, \dots, m$), $N_j > 0$ ($j = 0, 1, \dots, m - 1$), and $\mu > 0$. The \textup{$m$E-RCUSUM} statistic $\{ D_n \}$ is defined as follows:

\begin{enumerate}
\item We start with $D_0 = 0$ and update $\{ D_n \}$ using the observations $\{ X_n^m \}$ from experiment $m$, the experiment with highest LFL quality, using
$$ D_{n + 1}  = D_n + \bar{\ell}_m (X_{n + 1}^m), $$
where $\bar{\ell}_m (x^m) = \log \frac{\bar{g}_m (x^m)}{f_m (x^m)}$, until 
$$ \sigma = \min \{ n \geq 1: D_n \notin [0, A) \}. $$
If $D_{\sigma} \geq A$, we stop and declare change.

\item If $D_{\sigma} < 0$, we start using observations $\{ X_n^{m - 1} \}$, $\{ X_n^{m - 2} \}$, \dots, $\{ X_n^1 \}$ from the lower-quality experiments and execute the truncated \textup{$(m - 1)$E-RCUSUM} algorithm for experiments $m - 1$, $m - 2$, up to $1$ as follows:

    \begin{enumerate}
    \item We first scale the undershoot $U_m = D_{\sigma}$ by a factor $a_m$, and use $a_m U_m$ as the new zero level.

    \item With $a_m U_m$ as the new zero level, the stopping threshold is set at $0$.

    \item We update the statistic starting at time $\sigma$ using the truncated \textup{$(m - 1)$E-RCUSUM} algorithm $\Psi_{\textup{T}(m - 1)\textup{RC}} (a_m U_m, 0, a_1, \dots, a_{m - 1}, N_0, \dots, \allowbreak N_{m - 1}, \mu)$.

    \item When $\Psi_{\textup{T}(m - 1)\textup{RC}} (a_m U_m, 0, a_1, \dots, a_{m - 1}, N_0,  \allowbreak \dots, N_{m - 1}, \mu)$ terminates, either because $D_n$ crosses $0$ from below or because $N_{m - 1}$ number of observations of experiment $m - 1$ have been used, we reset the statistic $D_n$ to $0$ and go to Step $1$.
    \end{enumerate}

\item Stop at the stopping time
\begin{align*}
\tau_{\scriptscriptstyle m\textup{RC}}
& = \tau_{\scriptscriptstyle m\textup{RC}} (A, a_1, \dots, a_m, N_0, \dots, N_{m - 1}, \mu) \\
& = \inf \{ n \geq 1: D_n \geq A \}.
\end{align*}
\end{enumerate}

<!-- theorem-1: body_md -->
\label{thm:lower-bound}
Suppose Assumption~1 and Assumption~2 are satisfied and 
$$ 0 < D(\bar{g}_1 \ \Vert \ f_1) \leq \dots \leq D(\bar{g}_m \ \Vert \ f_m) < \infty. $$
Let $\Psi$ be a policy satisfying $\mathsf{ARLFA}(\Psi) \geq \gamma$ and $\mathsf{POR}_i (\Psi) \leq \beta_i$ for $i = 1, 2, \dots, m$, for fixed $\gamma > 0$ and $0 < \beta_i < 1$ where $\displaystyle \sum_{i = 1}^m \beta_i < 1$. Then as $\gamma \to \infty$,
$$ \sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi) \geq \frac{\log \gamma}{D(\bar{g}_m \ \Vert \ f_m)} (1 + o(1)). $$
<!-- end theorem-1 -->
<!-- ieeeproof-1: body_md -->
By Assumption~1, an LFL $\bar{g}_i \in \mathcal{G}_i$ exists for each experiment $i = 1, \dots, m$. Consider $\bar{G} = (\bar{g}_1, \dots, \bar{g}_m) \in \mathcal{G}$. Thus, for any policy $\Psi$,
$$ \sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi) \geq \mathsf{WADD}_{\bar{G}}   (\Psi) \geq \inf_\Psi \ \mathsf{WADD}_{\bar{G}} (\Psi), $$
where the infimum is taken over all policies satisfying the ARLFA and POR constraints. 
By Corollary~1 in \cite{lubenia2026}, as $\gamma \to \infty$,
$$ \inf_\Psi \ \mathsf{WADD}_{\bar{G}} (\Psi) \sim \frac{\log \gamma}{D(\bar{g}_m \ \Vert \ f_m)}.$$
Thus, as $\gamma \to \infty$,
$$ \sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi) \geq \frac{\log \gamma}{D(\bar{g}_m \ \Vert \ f_m)} (1 + o(1)). $$
<!-- end ieeeproof-1 -->
<!-- theorem-2: body_md -->
\label{thm:ALRFA}
Consider the \textup{$m$E-RCUSUM} policy $\Psi_{m\textup{RC}}$ with
$$ 0 < D(\bar{g}_1 \ \Vert \ f_1) \leq \dots \leq D(\bar{g}_m \ \Vert \ f_m) < \infty. $$
For any fixed $a_i > 0$ ($i = 1, \dots, m$), $N_j > 0$ ($j = 0, \dots, \allowbreak m - 1$), and $\mu > 0$, setting $A = \log \gamma$ guarantees
$$ \mathsf{ARLFA}(\Psi_{m\textup{RC}}) \geq \gamma. $$
<!-- end theorem-2 -->
<!-- ieeeproof-2: body_md -->
The proof is similar to the proof of Theorem 1 in \cite{lubenia2026}. We provide the complete proof here for convenience. 

Since the RCUSUM algorithm for any experiment $i$ is the classical CUSUM algorithm with pre-change density $f_i$ and post-change density $\bar{g}_i$, we have the following well-known result for the lower bound of the RCUSUM statistic based on \cite{lorden1971}:
$$ \mathsf{E}_\infty [\tau_{\scriptscriptstyle \text{RC}}^i (A)] \geq e^A. $$
Each time the statistic $D_n$ cross zero from below, it is reset to zero. At this time, the process renews itself and the process from that point on has the same distribution as the process that started at time $1$. Since the obervations are independent, this implies that in the $m$E-RCUSUM algorithm, the sojourns of the statistic $D_n$ below zero (when lower-quality experiments are used or when no experiment is performed) only delay the statistic from reaching the threshold $A$. In fact, if we remove all the sojourns of the statistic $D_n$ below zero from 
the $m$E-RCUSUM algorithm, the algorithm will be statistically identical to the RCUSUM algrorithm for experiment $m$ (the one with the highest LFL information quality). 

Thus, conditioning on the event that the highest-quality experiment $m$ is used at every observation time decreases the expected stopping time:
\begin{align*}
\mathsf{ARLFA}(\Psi_{m\text{RC}})
& = \mathsf{E}_\infty [\tau_{\scriptscriptstyle m\text{RC}}] \\
& \geq \mathsf{E}_\infty [\tau_{\scriptscriptstyle m\text{RC}} \mid S_n^m = 1 \ \forall n] \\
& = \mathsf{E}_\infty [\tau_{\scriptscriptstyle \text{RC}}^m (A)] \geq e^A.
\end{align*}
Setting $A = \log \gamma$, we get
$$ \text{ARLFA}(\Psi_{m\text{RC}}) \geq e^{\log \gamma} = \gamma. $$
<!-- end ieeeproof-2 -->
<!-- theorem-3: body_md -->
\label{thm:POR}

Consider the \textup{$m$E-RCUSUM} policy $\Psi_{m\textup{RC}}$ with
$$ 0 < D(\bar{g}_1 \ \Vert \ f_1) \leq \dots \leq D(\bar{g}_m \ \Vert \ f_m) < \infty. $$
For any fixed threshold $A$, there exist values of $a_i > 0$ ($i = 1, \dots, m$), $N_j > 0$ ($j = 0, \dots, m - 1$), and $\mu > 0$, all independent of $A$, such that for each $i = 1, \dots, m$,
$$ \mathsf{POR}_i (\Psi_{m\textup{RC}}) \leq \beta_i $$
for any given constraint $\beta_i \in (0, 1)$ with $\displaystyle \sum_{i = 1}^m \beta_i < 1$.
<!-- end theorem-3 -->
<!-- ieeeproof-3: body_md -->
The statistic update defining the $m$E-RCUSUM policy $\Psi_{m\textup{RC}}$ is similar to that of the DE$m$E-CUSUM policy $\Psi_{\text{DE}m\text{E}}$ (Algorithm~9 in \cite{lubenia2026}) where the former uses the pre-change density $f_i$ and post-change density $\bar{g}_i$, while the latter uses $f_i$ and $g_i$, respectively, for each $i = 1, \dots, m$. Moreover, the structure of $\Psi_{m\textup{RC}}$ and $\Psi_{\text{DE}m\text{E}}$ is identical, except for the log-likelihood ratio used. Thus, the claim for $\Psi_{m\textup{RC}}$ follows immediately from Theorem~5(2) in \cite{lubenia2026} for $\Psi_{\text{DE}m\text{E}}$.

<!-- theorem-4: body_md -->
\label{thm:upper-bound}

Suppose Assumptions 1 and 2 hold. Then for any fixed choices of $a_i > 0$ ($i = 1, \dots, m$), $N_j > 0$ \linebreak ($j = 0, \dots, m - 1$), and $\mu > 0$, as $A \to \infty$,
$$ \sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi_{m\textup{RC}}) \leq \frac{A}{D(\bar{g}_m \ \Vert \ f_m)}(1 + o(1)). $$
<!-- end theorem-4 -->
<!-- ieeeproof-4: body_md -->
Since the $m$E-RCUSUM algorithm has the same structure as the DE$m$E-CUSUM algorithm, by Theorem~5(3) in \cite{lubenia2026}, when the statistic $D_n$ of the $m$E-RCUSUM algorithm falls below 0, due to the truncation parameters, the truncated $(m - 1)$E-RCUSUM algorithm resets $D_n$ to 0 after using at most
$$ K_{m - 1} = \sum_{r = 0}^{m - 1} \prod_{k = (m - 1) - r}^{m - 1} N_k $$
observations. This bound is independent of the threshold $A$, the time the statistic falls below 0, and the environment $G$.

We now obtain a bound on
$$ \mathsf{WADD}_G (\Psi_{\scriptscriptstyle m\text{RC}}) = \sup_{\nu \geq 1} \operatorname{ess\,sup} \mathsf{E}_\nu^G [(\tau_{\scriptscriptstyle m\text{RC}} - \nu + 1)^+ \mid I_{\nu - 1}] $$
for any fixed $G \in \mathcal{G}$.

We note that for a fixed $\nu$, the conditional expectation depends on $I_{\nu - 1}$ through the $m$E-RCUSUM statistic $D_{\nu - 1}$ only. As a result we have
\begin{equation}
\mathsf{E}_\nu^G [(\tau_{\scriptscriptstyle m\text{RC}} - \nu + 1)^+ \mid I_{\nu - 1}] \leq \mathsf{E}_1^G [\tau_{\scriptscriptstyle m\text{RC}}] + K_{m - 1}. \label{eq:ave-delay-bound}
\end{equation}
This is because the worst possible choice of the change point $\nu$ and the worst-possible information vector $I_{\nu-1}$ will make sure that when the change occurs at a certain time $\nu$, the algorithm is in the data-efficiency mode, i.e., performing no experiments. From this state, the time taken by the statistic $D_n$ to reach $0$ from below is bounded by $K_{m-1}$. Once the statistic reached $0$ from below, it is reset to $0$. From this point onward, the average detection delay will be $\mathsf{E}_1^G [\tau_{\scriptscriptstyle m\text{RC}}]$. 
Since the right-hand side of \eqref{eq:ave-delay-bound} does not depend on $I_{\nu - 1}$ and $\nu$, taking the supremum over $\nu$ of the essential supremum of both sides gives us
\begin{equation}
\mathsf{WADD}_G (\Psi_{m\text{RC}}) \leq \mathsf{E}_1^G [\tau_{\scriptscriptstyle m\text{RC}}] + K_{m - 1}. \label{eq:wadd-mrc-bound}
\end{equation}

We now determine a bound for $\mathsf{E}_1^G [\tau_{\scriptscriptstyle m\text{RC}}]$.

Let $\tau_{\scriptscriptstyle \text{RC}}^m$ be the stopping time of the RCUSUM algorithm in \cite{unnikrishnan2011} where experiment $m$ is used. Let the statistic of this algorithm be
\begin{align*}
& W_{n + 1} = \left(W_n + \log \frac{\bar{g}_m (X_{n + 1}^m)}{f_m (X_{n + 1}^m)}\right)^+,  \; \; \text{with} \ W_0 = 0, \\
& \tau_{\scriptscriptstyle \text{RC}}^m = \inf \{ n \geq 1: W_n \geq A \}.
\end{align*}
The evolution of the $m$E-RCUSUM statistic $D_n$ and the RCUSUM statistic $W_n$ are statistically identical, except for the sojourns of $D_n$ below zero.  To clarify this, we now define 
$$ \lambda_A = \inf \{ n \geq 1: D_n \notin [0, A) \} = \inf \{ n \geq 1: W_n \notin [0, A) \}, $$
the first time the $m$E-RCUSUM statistic $D_n$ (or the RCUSUM statistic $W_n$) either falls below 0 or crosses $A$ from below. Thus, if change occurs at time $\nu=1$, the starting at time $1$, the algorithms $\tau_{\scriptscriptstyle \text{RC}}^m $ and $\Psi_{m\text{RC}}$ are identical until either the statistics cross $A$ or go below zero. If $D_n$ goes below zero, as explained at the start of the proof, the number of observations made is bounded by $K_{m - 1}$.  In the algorithm $\tau_{\scriptscriptstyle \text{RC}}^m$, when the statistic $D_n$ crosses zero from below, the statistic is reset to zero. This starts a new renewal cycle. Consequently, the evolution of the statistic $D_n$ is statistically identical to that of the statistic $W_n$ starting at time $\lambda_A$ (conditioned on $W_{\lambda_A} < 0$). 

The number of such renewal cycles is a geometrically distributed random variable with probability $\mathsf{P}_1^G (D_{\lambda_A} \geq A)$ of successfully crossing $A$ from below. Thus, the average number of cycles is the mean of the said geometric random variable: 
$$
\frac{1}{\mathsf{P}_1^G (D_{\lambda_A} \geq A)}. 
$$
Applying the reasoning above to \eqref{eq:wadd-mrc-bound} gives us
\begin{align*}
\mathsf{WADD}_G (\Psi_{m\text{RC}})
& \leq \mathsf{E}_1^G [\tau_{\scriptscriptstyle m\text{RC}}] + K_{m - 1} \\
& \leq \mathsf{E}_1^{G_m} [\tau_{\scriptscriptstyle \text{RC}}^m] + \frac{K_{m - 1}}{\mathsf{P}_1^G (D_{\lambda_A} \geq A)} + K_{m - 1}.
\end{align*}
Here the term $ \frac{K_{m - 1}}{\mathsf{P}_1^G (D_{\lambda_A} \geq A)}$ appears as a bound on the time spent by the statistic $D_n$ below the level zero. 
Taking the supremum over $G \in \mathcal{G}$ of both sides, we get
\begin{align}
\sup_{G \in \mathcal{G}} \mathsf{WADD}_G (\Psi_{m\text{RC}}) \notag 
&\leq \sup_{G \in \mathcal{G}} \mathsf{E}_1^{G_m} [\tau_{\scriptscriptstyle \text{RC}}^m] + \frac{K_{m - 1}}{\displaystyle \inf_{G \in \mathcal{G}} \mathsf{P}_1^G (D_{\lambda_A} \geq A)} + K_{m - 1}\\
 &= \sup_{G_m \in \mathcal{G}_m} \mathsf{E}_1^{G_m} [\tau_{\scriptscriptstyle \text{RC}}^m] + \frac{K_{m - 1}}{\displaystyle \inf_{G \in \mathcal{G}} \mathsf{P}_1^G (D_{\lambda_A} \geq A)} + K_{m - 1}. \label{eq:sup-wadd}
\end{align}
Note that the supremum in the first term on the right-hand side of the quality is over the post-change law $G_m~\in~\mathcal{G}_m$ for experiment $m$ since the expectation is of the stopping time $\tau_{\scriptscriptstyle \text{RC}}^m$ of the RCUSUM algorithm. 

Now, due to Assumptions~1 and 2, and the proof of Theorem III.2 in \cite{unnikrishnan2011}, we have
\begin{equation}
\sup_{G_m \in \mathcal{G}_m} \ \mathsf{E}_1^{G_m} [\tau_{\scriptscriptstyle \text{RC}}^m] = \mathsf{E}_1^{\bar{G}_m} [\tau_{\scriptscriptstyle \text{RC}}^m]. \label{eq:sup-g}
\end{equation}
Also, by the same set of assumptions, and following the arguments in the proof of Theorem IV.4 in \cite{hou2025}, we have
\begin{align}
& \mathsf{P}_1^G (D_{\lambda_A} \geq A) \notag \\
& \quad \geq \mathsf{P}_1^G (D_n \geq 0 \ \text{for all} \ n) \notag \\
& \quad = \lim_{N \to \infty} \ \mathsf{P}_1^G (D_n \geq 0 \ \text{for all} \ n \leq N) \notag \\
& \quad = \lim_{N \to \infty} \ \mathsf{P}_1^{G_m} \left( \max_{1 \leq n \leq N} \max_{1 \leq k \leq n} \sum_{i = k}^n \bar{\ell}_m (X_i^m) \geq 0 \right) \notag \\
& \quad \geq \lim_{N \to \infty} \ \mathsf{P}_1^{\bar{G}_m} \left( \max_{1 \leq n \leq N} \max_{1 \leq k \leq n} \sum_{i = k}^n \bar{\ell}_m (X_i^m) \geq 0 \right) \notag \\
& \quad = \mathsf{P}_1^{\bar{G}_m} \left( \max_{1 \leq k \leq n} \sum_{i = k}^n \bar{\ell}_m (X_i^m) \geq 0 \ \text{for all} \ n \right). \label{eq:p-geq-q}
\end{align}
The probability on the extreme right, 
$$ q := \mathsf{P}_1^{\bar{G}_m} \left( \max_{1 \leq k \leq n} \sum_{i = k}^n \bar{\ell}_m (X_i^m) \geq 0 \ \text{for all} \ n \right) > 0. $$
This follows from the results of renewal theory \cite{woodroofe1982}. We note that the probability $q$ depends on $\bar{G}_m$ and not on any other $G \in \mathcal{G}$. Also, $q$ does not depend on threshold $A$.

Taking the infimum over $G \in \mathcal{G}$ of both sides of \eqref{eq:p-geq-q}, we get
\begin{equation}
\inf_{G \in \mathcal{G}} \ \mathsf{P}_1^G (D_{\lambda_A} \geq A) \geq q > 0. \label{eq:inf-p}
\end{equation}

Substituting \eqref{eq:sup-g} and \eqref{eq:inf-p} into \eqref{eq:sup-wadd}, we get,
\begin{align*}
 \sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi_{m\text{RC}}) & \quad \leq \sup_{G_m \in \mathcal{G}_m} \ \mathsf{E}_1^{G_m} [\tau_{\scriptscriptstyle \text{RC}}^m] + \frac{K_{m - 1}}{\displaystyle \inf_{G \in \mathcal{G}} \ \mathsf{P}_1^G (D_{\lambda_A} \geq A)} + K_{m - 1} \\
& \quad \leq \mathsf{E}_1^{\bar{G}_m} [\tau_{\scriptscriptstyle \text{RC}}^m] + \frac{K_{m - 1}}{q} + K_{m - 1} \\
& \quad \leq \frac{A}{D(\bar{g}_m \ \Vert \ f_m)}(1 + o(1)) + \frac{K_{m - 1}}{q} + K_{m - 1}, \quad \textup{ as $A \to \infty$}, \\
& \quad = \frac{A}{D(\bar{g}_m \ \Vert \ f_m)}(1 + o(1)), \quad \textup{ as $A \to \infty$}.
\end{align*}
Here, the last inequality follows from the analysis of the CUSUM algorithm in \cite{lorden1971}, and the last equality follows since $\frac{K_{m - 1}}{q} + K_{m - 1}$ is a constant which does not depend on $A$.

\label{thm:robust-optimality}

If Assumptions~1 and 2 hold, then the \textup{$m$E-RCUSUM} algorithm is asymptotically robust optimal for each fixed $\beta_i$ ($i = 1, \dots, m$) with $\displaystyle \sum_{i = 1}^m \beta_i < 1$, as $\gamma \to \infty$.
<!-- end theorem-5 -->
<!-- ieeeproof-5: body_md -->
By Theorem~\ref{thm:ALRFA}, setting $A = \log \gamma$ guarantees that $\mathsf{ARLFA}(\Psi_{m\text{RC}}) \geq \gamma$. By Theorem~\ref{thm:POR}, we can set the values of $a_i > 0$ ($i = 1, \dots, m$), $N_j > 0$ ($j = 0, \dots, m - 1$), and $\mu > 0$, all independent of $A$, to satisfy $\mathsf{POR}_i (\Psi_{m\textup{RC}}) \leq \beta_i$ for each $i = 1, \dots, m$. With these parameter values, Theorem~\ref{thm:upper-bound} gives us, as $A \to \infty$ (equivalently, $\gamma \to \infty$),
\begin{align*}
\sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi_{m\textup{RC}})
& \leq \frac{A}{D(\bar{g}_m \ \Vert \ f_m)}(1 + o(1)) \\
& = \frac{\log \gamma}{D(\bar{g}_m \ \Vert \ f_m)}(1 + o(1)).
\end{align*}
Now, by Theorem~\ref{thm:lower-bound}, as $\gamma \to \infty$,
$$ \sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi_{m\textup{RC}}) \geq \frac{\log \gamma}{D(\bar{g}_m \ \Vert \ f_m)}(1 + o(1)). $$
Therefore, as $\gamma \to \infty$,
$$ \sup_{G \in \mathcal{G}} \ \mathsf{WADD}_G (\Psi_{m\textup{RC}}) = \frac{\log \gamma}{D(\bar{g}_m \ \Vert \ f_m)}(1 + o(1)), $$
i.e., the $m$E-RCUSUM algorithm is first-order asymptotically optimal.
