\label{fi}
    Let $n$ be a positive integer. Then there exists a simply-connected $\pi$-finite space $X$ with rational homology of finite type such that  
    $$B\mathrm{aut}_1(X)\simeq_{{\mathbb Q}}K({\mathbb Q},n)\times K({\mathbb Q},n+2).$$

    $$n\in \{3,4\} \cup \{ 4m+2: m\geq1 \} .$$ 
    Moreover, such $X$ has the minimal Sullivan model $M_X$ exactly as follows. 
    \begin{enumerate}
        \item When $n=3$, there is a unique model: 
        \begin{equation*}
            \begin{split}
                M_X\cong\ &\big( \Lambda(a_2,b_2,x_2,z_3,w_3,y_4) ,d \big), \\
                &da=db=dx=0,\ \ dz=ax, \\
                &dw=-bx, \ \ dy=aw+bz. 
            \end{split}
        \end{equation*}
        
        \item When $n=4$, there is an infinite family and an additional type:

\label{sig}

(1) Since $\theta_1(y)=\theta_2(y)=0$ and $[\theta_1,\theta_2]=0$, by direct computation we have

            0&=\big(D[\theta_1,\theta_2]\big)(y) \\
            &= \big([D\theta_1,\theta_2]+(-1)^{|\theta_1|} [\theta_1,D\theta_2]  \big)(y) \\
            &= -(-1)^{(|\theta_1|-1)|\theta_2|} \theta_2\big( (D\theta_1)(y) \big) +(-1)^{|\theta_1|} \theta_1\big( (D\theta_2)(y) \big) 
        \end{split}
    \end{equation*}

    $$\widetilde{u} \in V^{<N} ,\ \ \partial u=\widetilde{u}-p^\prime_{\widetilde{u}}$$

    \begin{equation*}

            \Sigma L_{-1}\Phi(u)&= \big(L_{0} D\Phi(u)\big)(y)  \\

            &=(-1)^N \big(L_{0} \Phi(\widetilde{u}-p^\prime_{\widetilde{u}})\big)(y) .

    $$W_0={\mathbb Q} x \oplus {\mathbb Q} q,\ \ W_1={\mathbb Q} a\oplus {\mathbb Q} z\oplus {\mathbb Q} w,$$
    and then $V=W_0\oplus W_1\oplus {\mathbb Q} y$ is the weight decomposition of $V$. We set 
    $$Der_{-1}= (P\oplus Q,D) ,$$
    where $P$, $Q$ denote the derivation subspaces of $Der_{-1}$ 
    $$\mathrm{Hom}(W_1,\Lambda W_0),\ \ \mathrm{Hom}(y,W_1\otimes\Lambda W_0),$$
    respectively. Note that the differential $D$ satisfies 
    $$D\big(\theta,\xi\big)=\big(D_P \theta, D_Q\xi+ \varphi(\theta) \big),\ \ \ \varphi(\theta)=-(-1)^{|\theta|} \big(y, \theta(dy) \big) $$
    where $D_P,D_Q$ are the projections of $D$ onto $P,Q$, respectively. By direct computation, we have 
    $$\varphi(a^\ast)= 2(y,w),\ \ \varphi(z^\ast)=-2(y,z),\ \ \varphi(w^\ast)=-2(y,a), $$
    which implies
    $$\varphi: W_1^\ast  \otimes \Lambda W_0 \cong P \longrightarrow Q\cong (y,W_1)\otimes \Lambda W_0 $$ 
    is an isomorphism of $(\Lambda W_0,d)$-modules. The identity $D^2=0$ gives 
    $$D_Q \varphi+\varphi D_P=0.$$
    Thus $Der_{-1}$ is an acyclic mapping cone, which implies $H(Der_{-1}^\prime)=0$. 

    For $Der_{0}$, there are exactly six generators of positive-degree derivations: 
    \begin{equation*}\centerline{
    \begin{tabular}{c|c}
    degree & generators\\
    \hline
    3&$q^\ast$\\
    \hline
    2&$ x^\ast, (w,a)$ \\
    \hline
    1& $(q,x),(z,a),(w,z)$.\\
    \end{tabular}}\end{equation*}
    By direct computation, we have 
    \begin{equation*}
        \begin{split}
            Dq^\ast&= -(w,a), \\
            Dx^\ast&= -2(q,x)-(z,a)-(w,z), \\
            D(w,a)&= 0, \\
            D(q,x)&= -(w,xa), \\
            D(z,a)&= (w,xa)+2(y,az), \\
            D(w,z)&= (w,xa)-2(y,az).
        \end{split}
    \end{equation*}
    It follows that $H_3(Der_{0})=H_2(Der_{0})=H_1(Der_{0})=0$, which implies $H(Der_{0}^\prime)=0$. This completes the proof of Proposition \ref{geq5}.
