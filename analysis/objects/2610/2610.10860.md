<!-- theorem-1: body_md -->
\label{thm:register-lower}
There exists \(K_0\in\mathbb N\) such that, for all
\(0<\xi,\epsilon\leq1/4\), every \(k\geq K_0\), every
\(m\geq1\), all \(0<\delta<1/4\), and every estimator
\(\widehat {t}_m^\sharp\), then uniform risk at most $\delta$ implies, for each \(i\in\{0,1\}\),
\begin{equation}
 m-1\geq
 \frac{t_{\mathrm{mix},P_i^{\mathrm{dr}}}}
      {48\pi e^{34}\xi^2\epsilon^2}
 \log\frac1{4\delta}.
 \label{eq:register-rate}
\end{equation}
<!-- end theorem-1 -->
<!-- assumption-1: body_md -->
\label{asm:1}
Let \(R\geq100\) be divisible by ten and let \(D\geq64\) be divisible
by eight.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{asm:2}
Let \(R_0\) and \(D_0\) be sufficiently large positive integers. Assume the conditions in \Cref{asm:1} hold, \(R\geq R_0\), \(D\geq D_0\), with \(M\geq C_{\rm size}\ell^2\) and \(0<\epsilon\leq1/4\) satisfying $\epsilon\leq 20c_{\rm acc}/\ell$.
<!-- end assumption-2 -->
<!-- theorem-2: body_md -->
\label{thm:rr-confidence-component}
Under \Cref{asm:1}, for all
\(0<\epsilon,\delta\leq1/4\), every \(m\geq1\), and
every estimator \(\widehat t_m^\sharp\), uniform risk of at most $\delta$, implies, for each \(i\in\{0,1\}\),
\begin{equation}
 m-1\geq
 \frac{t_{\mathrm{mix},P_i}}{1320\,\epsilon^2}
 \log\frac1{4\delta}.
 \label{eq:rr-confidence-component-rate}
\end{equation}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:rr-support-component}
For $R,D,M,\ell$ and $0<\epsilon\leq 1/4$ satisfying \Cref{asm:2}, and every \(m\geq1\), every
\(0<\delta\leq19/40\), and every estimator \(\widehat t_m^\sharp\), uniform risk
at most $\delta$, implies
\begin{equation}
 m-1\geq \frac{5c_{\rm inf}}{8}
 \sup_{P\in\mathcal P_0\cup\mathcal P_1}
 \frac{t_{\mathrm{mix},P}\,
       \mathcal J_{\infty,\rm env}(P)}
      {\epsilon^2\log\mathcal J_{\infty,\rm env}(P)}.
 \label{eq:rr-support-component-rate}
\end{equation}
<!-- end theorem-3 -->
<!-- corollary-1: body_md -->
\label{cor:rr-matched-maximum}
Under \Cref{asm:2}, for every admissible $\epsilon$, every
\(m\geq1\), all \(0<\delta<1/4\), and every estimator
\(\widehat t_m^\sharp\), uniform risk at most $\delta$, implies
\begin{equation}
\label{eq:rr-theorem-intrinsic-rate}
 \begin{aligned}
 m-1
 &\geq
 \sup_{P\in\mathcal C_{R,D,\epsilon}}
 \frac{t_{\mathrm{mix},P}}
      {\epsilon^2}\\
 &\quad{}\times
 \max\left\{
  \frac{5c_{\rm inf}}{8}\frac{\mathcal J_{\infty,\rm env}(P)}
       {\log\mathcal J_{\infty,\rm env}(P)},
  \frac{\log\frac1{4\delta}}{1320}
 \right\}.
 \end{aligned}
\end{equation}

<!-- proposition-1: body_md -->
\label{prop:finite-binomial-inverse}
    Define the constant $L:=\sqrt{2\pi}e^{17}$. Then, there exists \(K_0\in\mathbb N\) such that, for \(k\geq K_0\) and \(0\leq b_0<b_1\leq3/8\), \begin{align}
    B_k^{-1}(b_1)-B_k^{-1}(b_0)
    &\leq L\frac{b_1-b_0}{\sqrt{k}},
    \label{eq:Bk-inverse-lipschitz}\\
    B_k^{-1}(b_1)&\leq L\frac{b_1}{\sqrt{k}}.
    \label{eq:Bk-inverse-size}
    \end{align}

\label{lem:h-equivalence}
Fix $k\in\mathbb N$.  For $0\leq a<1$, let $P_a$ be the product law of
$k$ Rademacher variables $Z_1,\ldots,Z_k$ of mean $a$, let
$P_0=U_k$, and put
\[
 S=\sum_{i=1}^k Z_i,
 \qquad X=\frac{k+S}{2}.
\]
For $0<a<1$, define
\[
 h(a):=\frac{-\log(1-a^2)}{\log((1+a)/(1-a))},
\]
and set $h(0):=0$.  Then $h$ is continuous and strictly increasing on
$[0,1)$, and
\[
 \frac a2\leq h(a)<a,
 \qquad 0<a<1.
\]
For $0\leq a<1$, let $r_k(a)$ be the smallest point of
$\{-k,-k+2,\ldots,k\}$ strictly larger than $kh(a)$, and put
$j_k(a)=(k+r_k(a))/2$.  For $0<a<1$,
\[
 \left\{\frac{dP_a}{dP_0}>1\right\}
 =\{S>kh(a)\}
 =\{S\geq r_k(a)\}
 =\{X\geq j_k(a)\}.
\]
Moreover, for $0\leq a<1$, with $p=(1+a)/2$,
\begin{equation}
 \begin{aligned}
 B_k(a)
 &=\|P_a-P_0\|_{\operatorname{TV}}\\
 &=\mathbb P\{\operatorname{Bin}(k,p)\geq j_k(a)\}
   -\mathbb P\{\operatorname{Bin}(k,1/2)\geq j_k(a)\}.
 \end{aligned}
 \label{eq:Bk-binomial-tail}
\end{equation}
<!-- end lemma-12 -->
<!-- proof-13: body_md -->
If a realisation has sum $S$, then it contains $(k+S)/2$ plus signs
and $(k-S)/2$ minus signs.  Independence therefore gives
\[
 \frac{dP_a}{dP_0}
 =(1+a)^{(k+S)/2}(1-a)^{(k-S)/2}.
\]
Taking logarithms and collecting the terms involving $k$ and $S$
yields
\[
 \log\frac{dP_a}{dP_0}
 =\frac{k}{2}\log(1-a^2)
  +\frac S2\log\frac{1+a}{1-a}.
\]

We next derive the integral representation for $h$ used
below.  Direct differentiation gives
\begin{align*}
 \frac{d}{du}\{-\log(1-u^2)\}
 &=-\frac{-2u}{1-u^2}
 =\frac{2u}{1-u^2},\\
 \frac{d}{du}\log\frac{1+u}{1-u}
 &=\frac{d}{du}\{\log(1+u)-\log(1-u)\}\\
 &=\frac1{1+u}+\frac1{1-u}
 =\frac{2}{1-u^2}.
\end{align*}
Both logarithmic expressions vanish at \(u=0\).  The Fundamental Theorem
of Calculus therefore gives
\[
 -\log(1-a^2)=\int_0^a\frac{2u}{1-u^2}\,du,
 \quad
 \log\frac{1+a}{1-a}=\int_0^a\frac{2}{1-u^2}\,du,
\]
and hence
\[
 h(a)=
 \frac{\displaystyle\int_0^a u\,w(u)\,du}
      {\displaystyle\int_0^a w(u)\,du},
 \qquad
 w(u):=\frac{2}{1-u^2}.
\]
Thus \(h(a)\) is the weighted average of \(u\) over \([0,a]\), with
positive increasing weight \(w\).  Since \(u\leq a\) and \(w(u)>0\),
\[
 u w(u)\leq a w(u),
\]
with strict inequality for \(0\leq u<a\).  Therefore
\[
 \int_0^a u w(u)\,du
 <a\int_0^a w(u)\,du.
\]
Dividing by the positive total weight \(\int_0^a w(u)\,du\) proves that
\(h(a)<a\).

We next prove that this weighted average is at least the midpoint \(a/2\).
Since \(\int_0^a w(u)\,du>0\), the desired inequality is equivalent to
\begin{align*}
 h(a)\geq\frac a2
 &\quad\Longleftrightarrow\quad
 \int_0^a u w(u)\,du
 \geq\frac a2\int_0^a w(u)\,du\\
 &\quad\Longleftrightarrow\quad
 \int_0^a\left(u-\frac a2\right)w(u)\,du\geq0.
\end{align*}
Thus it remains to prove that the last centred integral is nonnegative.
Split it at \(a/2\):
\begin{align*}
 I
 &:=
 \int_0^a\left(u-\frac a2\right)w(u)\,du=
 \int_0^{a/2}\left(u-\frac a2\right)w(u)\,du
 +\int_{a/2}^a\left(u-\frac a2\right)w(u)\,du.
\end{align*}
The first integral is nonpositive and the second is nonnegative.  To compare
their magnitudes, make the change of variables \(v=a-u\) in the second
integral.  Then \(u=a-v\), \(du=-dv\), and the transformed endpoints are
\(v=a/2\) when \(u=a/2\) and \(v=0\) when \(u=a\).  Hence
\begin{align*}
 \int_{a/2}^a\left(u-\frac a2\right)w(u)\,du
 &=
 -\int_{a/2}^{0}\left(\frac a2-v\right)w(a-v)\,dv\\
 &=
 \int_0^{a/2}\left(\frac a2-v\right)w(a-v)\,dv\\
 &=
 \int_0^{a/2}\left(\frac a2-u\right)w(a-u)\,du.
\end{align*}
The second equality uses
\(\int_b^c f(v)\,dv=-\int_c^b f(v)\,dv\): the minus sign from
\(du=-dv\) cancels the minus sign introduced by reversing the endpoints.
The final equality merely renames the dummy variable \(v\) as \(u\).
Since
\[
 \int_0^{a/2}\left(u-\frac a2\right)w(u)\,du
 =
 -\int_0^{a/2}\left(\frac a2-u\right)w(u)\,du,
\]
the two halves combine to give
\begin{align*}
 I
 &=
 \int_0^{a/2}\left(\frac a2-u\right)
       \{w(a-u)-w(u)\}\,du\geq0,
\end{align*}
because \(a-u\geq u\) on \([0,a/2]\) and \(w\) is increasing.  By the
equivalence above, \(I\geq0\) proves \(h(a)\geq a/2\).

Finally, write
\[
 N(a):=\int_0^a u w(u)\,du,
 \qquad
 D(a):=\int_0^a w(u)\,du,
\]
so that \(h(a)=N(a)/D(a)\).  The variable \(a\) appears in these
integrals through their upper endpoints.  The Fundamental Theorem of
Calculus therefore gives
\[
 N'(a)=a w(a),
 \qquad
 D'(a)=w(a).
\]
Since \(D(a)>0\), the quotient rule yields
\begin{align*}
 h'(a)
 &=
 \frac{N'(a)D(a)-N(a)D'(a)}{D(a)^2} =
 \frac{a w(a)D(a)-N(a)w(a)}{D(a)^2}\\
 &=
 \frac{w(a)}{D(a)}
 \left\{a-\frac{N(a)}{D(a)}\right\} =
 \frac{w(a)\{a-h(a)\}}{\int_0^a w(u)\,du}.
\end{align*}
Here \(w(a)>0\), \(D(a)>0\), and the previously proved inequality
\(h(a)<a\) gives \(a-h(a)>0\).  Consequently \(h'(a)>0\) for \(a>0\).
The defining quotient is continuous on $(0,1)$, while the bounds above
give $h(a)\to0=h(0)$ as $a\downarrow0$.  Thus $h$ is continuous on
$[0,1)$.  Together with $h'(a)>0$ for $a>0$ and
$h(a)>0=h(0)$, this proves that $h$ is strictly increasing and that
\[
 \frac a2\leq h(a)<a,
 \qquad 0<a<1.
\]

The definition of \(h\) makes its likelihood-ratio role explicit.  For
\(a>0\), the preceding formula for the log likelihood ratio can be
rewritten as
\[
 \log\frac{dP_a}{dP_0}
 =
 \frac12\log\frac{1+a}{1-a}\{S-kh(a)\}.
\]
Indeed, after multiplying out, the definition
\[
 h(a)=\frac{-\log(1-a^2)}{\log((1+a)/(1-a))}
\]
turns the term involving \(h(a)\) into
\((k/2)\log(1-a^2)\).  The factor outside the braces is positive, so
\[
 \frac{dP_a}{dP_0}>1
 \quad\Longleftrightarrow\quad
 S>kh(a).
\]
Since $S$ takes values in the lattice $\{-k,-k+2,\ldots,k\}$, the
definitions of $r_k(a)$, $j_k(a)$, and $X$ give
\[
 \{S>kh(a)\}=\{S\geq r_k(a)\}=\{X\geq j_k(a)\}.
\]
Thus the event on which the likelihood ratio exceeds one is a binomial
upper-tail event, namely $\{X\geq j_k(a)\}$.

For two probability mass functions $p$ and $q$ on a finite set,
\[
 \|p-q\|_{\operatorname{TV}}
 =\sum_{x:p(x)>q(x)}\{p(x)-q(x)\}.
\]
Because $P_0$ has full support, the event on which the likelihood ratio
exceeds one collects exactly the positive terms in the signed mass difference
and realises total variation.  Since the definition of $B_k$ gives
$B_k(a)=\|P_a-P_0\|_{\operatorname{TV}}$, it follows that
\[
 B_k(a)
 =P_a\{S>kh(a)\}-P_0\{S>kh(a)\}.
\]
Under $P_a$, $X\sim\operatorname{Bin}(k,(1+a)/2)$, whereas under
$P_0$, $X\sim\operatorname{Bin}(k,1/2)$.  The preceding event
equivalences therefore prove~\eqref{eq:Bk-binomial-tail} for $a>0$.
At $a=0$, both binomial upper-tail probabilities are equal and $B_k(0)=0$,
so the identity also holds at that endpoint.

\label{lem:binomial-equivalent-derivative}
For every $k\in\mathbb N$, $B_k$ is locally absolutely continuous on $[0,1)$.
At every $a\in(0,1)$ for which $r_k(a)$, equivalently $j_k(a)$, is
locally constant,
\begin{equation}
 B_k'(a)
 =\frac{k}{2}
 \mathbb P\left\{
 \operatorname{Bin}\left(k-1,\frac{1+a}{2}\right)=j_k(a)-1
 \right\}.
 \label{eq:Bk-derivative}
\end{equation}
Consequently, \eqref{eq:Bk-derivative} holds for almost every
$a\in(0,1)$.
<!-- end lemma-13 -->
<!-- proof-14: body_md -->
Put $p=(1+a)/2$, and write $P_p$ and $\mathbb E_p$ for probability and
expectation when $X\sim\operatorname{Bin}(k,p)$.  By
\eqref{eq:Bk-binomial-tail}, $B_k(a)$ is the difference of the two
corresponding binomial upper-tail probabilities.

Because \(h\) is strictly increasing, the lattice threshold \(r_k(a)\)
is piecewise constant and changes at only finitely many values of \(a\).
Fix an open interval on which \(r_k(a)\), and hence \(j_k(a)\), has a
constant value \(j\).  On this interval, the second term in
\eqref{eq:Bk-binomial-tail} is constant.  Term-by-term differentiation of
the first, finite binomial sum gives
\begin{align}
 \frac{d}{dp}P_p\{X\geq j\}
 &=\sum_{x=j}^k
   \frac{d}{dp}\left\{\binom{k}{x}p^x(1-p)^{k-x}\right\}\notag \\
   &=\sum_{x=j}^k\binom{k}{x}p^x(1-p)^{k-x}
   \left\{\frac{x}{p}-\frac{k-x}{1-p}\right\}\notag\\
   &=\frac{1}{p(1-p)}
   \mathbb E_p[(X-kp)\mathbf 1\{X\geq j\}].
 \label{eq:binomial-tail-score}
\end{align}
To evaluate the final expectation, the binomial identities
\[
 x\binom{k}{x}=k\binom{k-1}{x-1},
 \qquad
 (k-x)\binom{k}{x}=k\binom{k-1}{x}
\]
give
\begin{align*}
 \mathbb E_p[X\mathbf 1\{X\geq j\}]
 &=kp\mathbb P\{\operatorname{Bin}(k-1,p)\geq j-1\},\\
 \mathbb E_p[(k-X)\mathbf 1\{X\geq j\}]
 &=k(1-p)\mathbb P\{\operatorname{Bin}(k-1,p)\geq j\}.
\end{align*}
Since \(X-kp=(1-p)X-p(k-X)\), subtracting the two adjacent tails leaves
one atom:
\[
 \mathbb E_p[(X-kp)\mathbf 1\{X\geq j\}]
 =kp(1-p)\mathbb P\{\operatorname{Bin}(k-1,p)=j-1\}.
\]
Substitution into \eqref{eq:binomial-tail-score} therefore yields
\[
 \frac{d}{dp}P_p\{X\geq j\}
 =k\mathbb P\{\operatorname{Bin}(k-1,p)=j-1\}.
\]
Finally, $dp/da=1/2$.  The chain rule gives
\eqref{eq:Bk-derivative} whenever the lattice threshold is locally
constant.  There are only finitely many exceptional points at which the threshold changes.  Moreover, from its
definition as total variation, \(B_k\) is a finite sum of absolute values
of analytic functions of $a$.  It is therefore continuous and locally
absolutely continuous on $[0,1)$.  Thus \eqref{eq:Bk-derivative} holds
almost everywhere.

\label{lem:binomial-atom}
For every finite \(A>0\), there exist constants \(c_A>0\) and
\(n_A\in\mathbb N\) such that, whenever \(n\geq n_A\),
\(p\in[1/3,2/3]\), and \(j\in\{0,\ldots,n\}\) satisfy
\(|j-np|\leq A\sqrt n\),
\[
 \mathbb P\{\operatorname{Bin}(n,p)=j\}
 \geq\frac{c_A}{\sqrt n}.
\]
<!-- end lemma-14 -->
<!-- proof-15: body_md -->
Put \(q=j/n\).  The assumption
\(|j-np|\leq A\sqrt n\) implies that
\[
 |q-p|\leq\frac{A}{\sqrt n}.
\]
Choose
\[
 n_A=\left\lceil\max\{4,144A^2\}\right\rceil.
\]
If \(n\geq n_A\), then
\[
 |q-p|\leq\frac{A}{\sqrt n}\leq\frac1{12}.
\]
Since \(p\in[1/3,2/3]\), it follows that \(q\in[1/4,3/4]\).
Thus \(j\) and \(n-j\) are both positive and of order \(n\), and the
Stirling and Taylor bounds below hold uniformly.  In particular,
\(j,n-j\geq n/4\geq1\), so Robbins' formula applies to all three
factorials in \(n!/[j!(n-j)!]\) = $\binom{n}{j}$.

For every positive integer \(m\), write Robbins' formula as
\[
 m!=\sqrt{2\pi m}\left(\frac me\right)^m e^{r_m},
 \qquad
 \frac1{12m+1}<r_m<\frac1{12m}.
\]
Substitution into the binomial coefficient gives
\begin{align*}
 \binom nj
 &=
 \frac{1}{\sqrt{2\pi}}
 \sqrt{\frac{n}{j(n-j)}}
 \frac{n^n}{j^j(n-j)^{n-j}}
 \exp\{r_n-r_j-r_{n-j}\}.
\end{align*}
Here the factors involving \(e\) cancel because
\[
 e^{-n}e^je^{n-j}=1,
\]
while the three Robbins correction factors collect into
\(\exp\{r_n-r_j-r_{n-j}\}\).  For the square-root factor, the identities
\(j=qn\) and \(n-j=(1-q)n\) give
\[
 \frac{1}{\sqrt{2\pi}}\sqrt{\frac{n}{j(n-j)}}
 =
 \frac{1}{\sqrt{2\pi n q(1-q)}}.
\]
For the power factor, taking logarithms and making the same substitution
shows how the binary entropy arises:
\begin{align*}
 \log\left(\frac{n^n}{j^j(n-j)^{n-j}}\right)
 &=n\log n-j\log j-(n-j)\log(n-j)\\
 &=n\log n-qn\log(qn)-(1-q)n\log((1-q)n)\\
 &=n\log n-qn\{\log q+\log n\} -(1-q)n\{\log(1-q)+\log n\}\\
 &=-qn\log q-(1-q)n\log(1-q)\\
 &=nH(q),
\end{align*}
where \(H(q):=-q\log q-(1-q)\log(1-q)\).  Exponentiating therefore gives
\[
 \frac{n^n}{j^j(n-j)^{n-j}}=\exp\{nH(q)\}.
\]
Combining the square-root, power, and remainder factors yields
\[
 \binom nj
 =
 \frac{1}{\sqrt{2\pi n q(1-q)}}
 \exp\{nH(q)\}
 \exp\{r_n-r_j-r_{n-j}\}.
\]
Since \(q\in[1/4,3/4]\),
\[
 j=nq\geq\frac n4,
 \qquad
 n-j=n(1-q)\geq\frac n4.
\]
Moreover, \(r_n>0\), while the upper bound in Robbins' formula gives
\[
 r_j<\frac1{12j},
 \qquad
 r_{n-j}<\frac1{12(n-j)}.
\]
Consequently,
\begin{align*}
 r_n-r_j-r_{n-j}
 &>-\frac1{12j}-\frac1{12(n-j)}\geq-\frac1{3n}-\frac1{3n}=-\frac{2}{3n}\geq-\frac23.
\end{align*}
Also \(q(1-q)\leq1/4\), and hence
\begin{align*}
 \frac{1}{\sqrt{2\pi n q(1-q)}}
 &\geq\frac{1}{\sqrt{2\pi n(1/4)}}=\sqrt{\frac{2}{\pi n}}.
\end{align*}
Multiplying the preceding formula for \(\binom nj\) by
\(p^j(1-p)^{n-j}\), and using the exact identity
\begin{align*}
 H(q)+q\log p+(1-q)\log(1-p)
 &=-q\log\frac qp-(1-q)\log\frac{1-q}{1-p}=-\operatorname{kl}(q\Vert p),
\end{align*}
where \(\operatorname{kl}(q\Vert p)\) is the one-trial Bernoulli
relative entropy, therefore gives
\[
 \mathbb P\{\operatorname{Bin}(n,p)=j\}
 \geq\sqrt{\frac2\pi}\,\frac1{\sqrt n}
       \exp\left\{-\frac23-n\operatorname{kl}(q\Vert p)\right\}.
\]

For fixed \(p\), let \(f_p(x)=\operatorname{kl}(x\Vert p)\).  Then for all $x\in[1/4,3/4]$,
\[
 f_p(p)=f_p'(p)=0,
 \qquad
 f_p''(x)=\frac1{x(1-x)}\leq\frac{16}{3}.
\]
The segment between \(p\) and \(q\) lies in this interval, so Taylor's
theorem gives
\[
 \operatorname{kl}(q\Vert p)
 \leq\frac83(q-p)^2.
\]
Consequently \(n\operatorname{kl}(q\Vert p)\leq8A^2/3\), and the claimed
bound follows, for instance, with
\[
 c_A=\sqrt{\frac2\pi}
      \exp\left\{-\frac23-\frac83A^2\right\}.
\]
