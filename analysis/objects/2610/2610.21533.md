\label{lem:Variety_of_cone_volumes}
	Let $U\in\mathcal{U}(2,m)$, $\gamma\in C_{{\tt cv}}(U)$, and let
	$f_i$, $1\leq i\leq m$, be as in \eqref{eq:planarconevolumes}. Then
	\[
	X_\gamma
	\coloneqq \mathcal V_{\mathbb C}
	(f_1-\gamma_1,\ldots,f_{m-2}-\gamma_{m-2})
	\subset\mathbb C^m
	\]
	is a smooth irreducible algebraic set of complex dimension $2$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For $b\in X_\gamma$ we have
	\[
	b_i\neq0,\qquad i=1,\ldots,m-2,
	\]
	since otherwise \eqref{eq:planarconevolumes} would give
	$f_i(b)=0\neq\gamma_i$. Hence \eqref{eq:planar-recursion} shows that every
	$b\in X_\gamma$ is uniquely determined by the two coordinates $b_m$ and
	$b_1$.
	
	Set
	\[
	\Omega_1\coloneqq \mathbb C^2,\qquad
	r_0(x,y)\coloneqq x,\qquad r_1(x,y)\coloneqq y,
	\]
	and, for $i=1,\ldots,m-2$, define recursively
	\[
	\Omega_{i+1}\coloneqq \{(x,y)\in\Omega_i:r_i(x,y)\neq0\}
	\]
	and, for $(x,y)\in\Omega_{i+1}$,
	\[
	r_{i+1}(x,y)
	\coloneqq \frac{A_i r_i(x,y)-C_i r_{i-1}(x,y)}{B_i}
	+\frac{2\gamma_i}{B_i r_i(x,y)}.
	\]
	By induction, $r_i$ is regular on $\Omega_i$, and therefore
	$\Omega_{i+1}$ is Zariski open in $\Omega_i$, hence also in $\mathbb C^2$.
	Moreover, all these sets are nonempty. Indeed, since
	$\gamma\in C_{{\tt cv}}(U)$, there exists $b=(b_1,\ldots,b_m)\in S(U,\gamma)$,
	and \eqref{eq:planar-recursion} shows that $(b_m,b_1)\in\Omega_i$ for every
	$i$. Thus
	\[
	\Omega\coloneqq \Omega_{m-1}
	\]
	is a nonempty Zariski-open subset of $\mathbb C^2$. Consequently,
	$\Omega$ is smooth and irreducible and
	\[
	\dim_{\mathbb C}\Omega=2.
	\]
	
	By \eqref{eq:planar-recursion}, the map
	\[
	\Phi:\Omega\longrightarrow X_\gamma,\qquad
	(x,y)\longmapsto
	\bigl(r_1(x,y),\ldots,r_{m-1}(x,y),r_0(x,y)\bigr)
	\]
	is well defined and regular. Conversely, if $b\in X_\gamma$, then
	$b_i\neq0$ for $i=1,\ldots,m-2$, and
	\eqref{eq:planar-recursion} shows recursively that
	\[
	\Phi(b_m,b_1)=b.
	\]
	Thus $\Phi$ is bijective, with inverse
	\[
	\Phi^{-1}(b_1,\ldots,b_m)=(b_m,b_1),
	\]
	which is regular. Hence $\Phi$ is an isomorphism between $\Omega$ and
	$X_\gamma$. Therefore $X_\gamma$ is smooth and irreducible and
	\[
	\dim_{\mathbb C}X_\gamma=2.
	\]

<!-- lemma-4: body_md -->
\label{lem:cone_volume_vectors_with_different_last_entries}
	Let $U\in\mathcal{U}(2,m)$ and $\gamma\in C_{{\tt cv}}(U)$. Then there exists
	$\gamma'\in C_{{\tt cv}}(U)$ such that
	\[
	\gamma_i'=\gamma_i,\qquad i=1,\ldots,m-2, \text{ and } 
	\gamma_i'\neq\gamma_i,\qquad i=m-1,m.
	\]
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
If $U$ is reducible, then $m=4$ and, with the cyclic ordering of the outer
	unit normals, we have 
	$u_3=-u_1$ and $u_4=-u_2$. 
	For every cone-volume vector of a parallelogram one has
	\[
	\gamma_1+\gamma_3=\gamma_2+\gamma_4.
	\]
	The vector
	\[
	\gamma'\coloneqq (\gamma_1,\gamma_2,\gamma_3+1,\gamma_4+1)
	\]
	also satisfies the subspace concentration condition with respect to $U$. By the sufficiency of this
	condition for the discrete logarithmic Minkowski problem, proved by Chen, Li and Zhu
	\cite{ChenLiZhu2019}, it belongs to $C_{{\tt cv}}(U)$ and has the required
	properties.
	
	Assume now that $U$ is irreducible. Set
	\[
	\tau\coloneqq \sum_{i=1}^{m-2}\gamma_i,
	\qquad
	L_{m-1}\coloneqq \operatorname*{lin}\{u_{m-1}\},
	\qquad
	L_m\coloneqq \operatorname*{lin}\{u_m\},
	\]
	and
	\[
	\nu_{m-1}
	\coloneqq \sum_{\substack{1\leq i\leq m-2\\u_i\in L_{m-1}}}\gamma_i,
	\qquad
	\nu_m
	\coloneqq \sum_{\substack{1\leq i\leq m-2\\u_i\in L_m}}\gamma_i.
	\]
	Since $u_{m-1}$ and $u_m$ are adjacent outer unit normals,
	$L_{m-1}\neq L_m$. Moreover,
	\[
	\nu_{m-1}+\nu_m<\tau.
	\]
	Indeed, equality would imply, since all $\gamma_i$ are positive, that
	$u_1,\ldots,u_{m-2}\in L_{m-1}\cup L_m$. Hence
	$U\subset L_{m-1}\cup L_m$, contradicting the irreducibility of $U$.
	
	For every one-dimensional subspace $L\subset\mathbb{R}^2$, set
	\[
	\nu_L
	\coloneqq \sum_{\substack{1\leq i\leq m-2\\u_i\in L}}\gamma_i.
	\]
	Choose $q$ sufficiently large such that
	\[
	\nu_L
	<q+\frac{\tau-\nu_{m-1}-\nu_m}{2}
	\]
	for every one-dimensional subspace
	$L\neq L_{m-1},L_m$ with $L\cap U\neq\varnothing$, and, in addition,
	\[
	q>\gamma_{m-1}+\nu_{m-1},
	\qquad
	q>\gamma_m+\nu_m.
	\]
	Define
	\[
	\gamma_i'\coloneqq \gamma_i,\qquad i=1,\ldots,m-2, 
	\text{ and } 
	\gamma_{m-1}'\coloneqq q-\nu_{m-1},
	\,
	\gamma_m'\coloneqq q-\nu_m.
	\]
	By the choice of $q$ we have

\label{prop:irreducible-regular-point}
	Let $U\in\mathcal{U}(n,m)$. Then
	$U$ is irreducible if and only if $W(U)\neq\varnothing$.

<!-- proof-5: body_md -->
Suppose first that $U$ is irreducible. By
	\cite[Proposition~2.7]
	{baumbach2025polynomialinequalitiesconevolumespolytopes},
	the normalized cone-volume set, i.e., $C_{{\tt cv}}(U)\cap \{y\in\mathbb{R}^m : y_1+\dots+y_m=1\}$ 
	contains a relatively open subset of dimension $m-1$.
	Hence, $C_{{\tt cv}}(U)$ contains an open subset of dimension $m$. 
	
	If $W(U)=\varnothing$, then $\det^A\equiv0$ for every
	full-dimensional type-cone $A\in\mathcal A(U)$. Hence every point of
	$\operatorname*{int} A$ is a critical point of the polynomial map $f_{{\tt cv}}^A$,
	and Sard's theorem (see, e.g., \cite{979f1be4-becf-36f2-be30-0bebf5263aa4}) 
	implies that $f_{{\tt cv}}^A(\operatorname*{int} A)$ has Lebesgue
	measure zero. The remaining parts of the type-cone decomposition
	have dimension at most $m-1$, and their images under the corresponding
	polynomial maps have also 
	Lebesgue measure zero. Since there are only finitely many type-cones,
	$C_{{\tt cv}}(U)$ would have Lebesgue measure zero, contradicting the fact
	that it has nonempty interior. Thus $W(U)\neq\varnothing$.
	
	Conversely, suppose that $W(U)\neq\varnothing$ and choose $b\in W(U)$.
	Then $b\in\operatorname*{int} A$ for some type-cone $A$ and
	$\det Df_{{\tt cv}}^A(b)\neq0$. By the inverse function theorem,
	$C_{{\tt cv}}(U)$ contains a nonempty open subset of $\mathbb{R}^m$ which again by 
	 \cite[Proposition~2.7]
	{baumbach2025polynomialinequalitiesconevolumespolytopes}
	shows that $U$ is irreducible.

<!-- proposition-4: body_md -->
\label{prop:regular-points}
	Let $U\in\mathcal{U}(n,m)$ be irreducible. Then the following holds: 
	\begin{enumerate}
		\item For every $b\in W(U)$, the map $f^U_{{\tt cv}}$ is differentiable
		and locally invertible at $b$. 
		\item If $\gamma\in C_{{\tt cv}}(U)\setminus B(U)$, then
		$S(U,\gamma)\subset W(U)$ and $|S(U,\gamma)|<\infty$.
				
		\item $W(U)$ is open and dense in
		\[
		\bigcup_{A\in\mathcal A(U),\, 
				\det^A\not\equiv0}\operatorname*{int} A.
		\]
		\item $B(U)$ is semialgebraic and has Lebesgue measure zero in
		$\mathbb{R}^m$.
	\end{enumerate}
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
i) follows directly from the inverse function
	theorem. For ii), 
	if $\gamma\notin B(U)$, then by the definition of $B(U)$ every
	$b\in S(U,\gamma)$ belongs to $W(U)$. Hence
	$\det Df_{{\tt cv}}^U(b)\neq0$, and the inverse function theorem yields a
	neighborhood $V_b$ of $b$ such that
	\[
	S(U,\gamma)\cap V_b=\{b\}.
	\]
	Thus every point of $S(U,\gamma)$ is isolated. Since
	$S(U,\gamma)$ is semialgebraic, it follows that $S(U,\gamma)$ is
	finite (see, e.g., \cite[Sections~2.3 and~2.8]{bochnak2013real}).
	
	For iii), if ${\det}^A\not\equiv0$, then its zero set has empty interior,
	so $\operatorname*{int} A\setminus\mathcal V({\det}^A)$ is open and dense in
	$\operatorname*{int} A$.  
	Finally, for iv) we note that $B(U)$ is semialgebraic by the
	Tarski--Seidenberg theorem as the complement of $W(U)$ is a finite union
	of type-cone boundaries, zero sets $\mathcal V({\det}^A)$ with
	${\det}^A\not\equiv0$, and type-cones on which ${\det}^A\equiv0$. Except for the latter all these sets have semi-algebraic dimension at most $m-1$ 
	and so do their images under $f^U_{{\tt cv}}$. If $A$ is a full-dimensional type cone on which ${\det}^A\equiv0$ then, as before, 
	Sard’s theorem shows that $f^A_{\mathrm{cv}}(\operatorname{int}A)$ has Lebesgue measure zero. Since this image is semialgebraic, it has semialgebraic dimension at most $m-1$.  
	Hence $\dim^*B(U)\leq m-1$ and it has Lebesgue measure zero; see, for example,
	\cite[Section~2.8]{bochnak2013real}.

		\varepsilon(\bar{\mathfrak h})
		=(-1)^{m-1}\operatorname*{sgn}\det Df_{{\tt cv}}^U(b).

	\emph{Coordinates adapted to translations and scalings.} For
	$(y,t,s)$ in a neighbourhood of $(y_0,0,0)$ in
	$\mathcal Y\times\mathbb{R}^n\times\mathbb{R}$ let
	$R(y,t,s)=e^s\bigl(\Theta_\gamma(y)-U^\intercal t\bigr)$. The map $R$
	is smooth, $R(y_0,0,0)=b$, and it has the smooth local inverse
	\[
	b'\mapsto
	\Bigl(q(b'),\
	(UU^\intercal)^{-1}U\bigl(\Theta_\gamma(q(b'))-e^{-s}b'\bigr),\ s\Bigr),
	\qquad
	s=\log\frac{\langle\omega,b'\rangle}
	{\langle\omega,\Theta_\gamma(q(b'))\rangle},
	\]
	with $q$ as in the proof of Lemma~\ref{lem:parity-slice}; this follows
	from $q\circ\Theta_\gamma=\operatorname*{id}$ and
	$\langle\omega,U^\intercal t\rangle=0$. Hence $R$ is a local
	diffeomorphism. By scale invariance,
	$\Psi_\gamma\circ R(y,t,s)=F(y,t)
	\coloneqq\Psi_\gamma(\Theta_\gamma(y)-U^\intercal t)$ does not depend
	on $s$, and $F(y,0)=\psi_\gamma(y)$. Since
	$\Theta_\gamma(y)\in M_\gamma$, Lemma~\ref{lem:parity-functional}(iii)
	shows that $\partial_tF(y,0)=0$ for all $y$, and therefore
	$\partial_y\partial_tF(y_0,0)=0$. Moreover,
	$\partial_t^2F(y_0,0)=-\sum_i\gamma_ib_i^{-2}u_iu_i^\intercal$ is
	negative definite. Consequently,
	\[
	\mathrm{Hess}\,(\Psi_\gamma\circ R)(y_0,0,0)
	=\mathrm{Hess}\,\psi_\gamma(y_0)\oplus\partial_t^2F(y_0,0)\oplus(0),
	\]
	and since $\partial_sR(y_0,0,0)=b$, this is a Gram matrix of
	$\mathfrak h$ in which the last summand corresponds to the radical
	$\mathbb{R} b$. Thus
	$\bar{\mathfrak h}\cong\mathrm{Hess}\,\psi_\gamma(y_0)\oplus
	\partial_t^2F(y_0,0)$. As $\bar{\mathfrak h}$ is non-degenerate, so is
	$\mathrm{Hess}\,\psi_\gamma(y_0)$, and
	$\varepsilon(\bar{\mathfrak h})
	=(-1)^n\operatorname*{sgn}\det\mathrm{Hess}\,\psi_\gamma(y_0)$. Together with
	\eqref{eq:parity-eps} this proves the lemma.

<!-- lemma-9: body_md -->
\label{lem:parity-castro-lazer}
	Let $\phi\in C^1(\mathbb{R}^d)$ with $\phi(z)\to\infty$ as $|z|\to\infty$.
	Assume that $\nabla\phi$ has only finitely many zeros
	$z_1,\dots,z_k$, that $\phi$ is of class $C^2$ in a neighbourhood of
	each $z_j$, and that the Hessians $\mathrm{Hess}\,\phi(z_j)$ are
	non-singular. Then
	\[
	\sum_{j=1}^k\operatorname*{sgn}\det\mathrm{Hess}\,\phi(z_j)=1.
	\]
<!-- end lemma-9 -->
<!-- proof-18: body_md -->
By \cite{CastroLazer1979} (see \cite{McKenzie1984} for an elementary
	proof), the Brouwer degree of $\nabla\phi$ on the ball $B_r$ with
	respect to the value $0$ equals $1$ for all sufficiently large $r$. By
	the additivity of the degree, it equals the sum of the local indices
	of $\nabla\phi$ at the points $z_j$, and the local index at a
	non-degenerate zero is $\operatorname*{sgn}\det\mathrm{Hess}\,\phi(z_j)$; see, e.g.,
	\cite{Deimling1985}.

<!-- proof-19: body_md -->
As explained above, we may assume $\|\gamma\|_1=1$. By
	Proposition~\ref{prop:regular-points}(ii),
	Lemma~\ref{lem:parity-slice}(ii) and Lemma~\ref{lem:parity-index},
	$\psi_\gamma$ has finitely many critical points
	$y_j=\Theta_\gamma^{-1}(b^{(j)})$, where
	$S(U,\gamma)=\{b^{(1)},\dots,b^{(k)}\}$, and $\psi_\gamma$ is smooth
	with non-singular Hessian near each of them.

	If $d=0$, then $\mathcal Y$ is a point and $k=1$. If $d\geq1$, let
	$\sigma:\mathbb{R}^d\to\mathcal Y$ be a diffeomorphism; such a map exists
	since $\mathcal Y$ is an open convex subset of a $d$-dimensional
	affine space. Then $\phi=\psi_\gamma\circ\sigma$ satisfies the
	assumptions of Lemma~\ref{lem:parity-castro-lazer}: it tends to
	infinity at infinity by Lemma~\ref{lem:parity-coercive}, and at a
	critical point
	$\mathrm{Hess}\,\phi
	=D\sigma^\intercal\,(\mathrm{Hess}\,\psi_\gamma)\,D\sigma$. Hence, in
	both cases,
	$\sum_{j=1}^k\operatorname*{sgn}\det\mathrm{Hess}\,\psi_\gamma(y_j)=1$, and
	Lemma~\ref{lem:parity-index} gives
	\[
	\sum_{j=1}^k\operatorname*{sgn}\det Df_{{\tt cv}}^U(b^{(j)})=(-1)^{m+n-1}.
	\]
	A sum of $k$ terms $\pm1$ equals $\pm1$ only if $k$ is odd.

\label{thm:dev}
	Let $U\in\mathcal{U}(n,m)$ and let $A\in\mathcal A(U)$ be a proper type-cone.
	Write $f_i=f_i^A$. Then, for $b\in\operatorname*{int} A$,
	\[
	\frac{\partial f_i}{\partial b_j}(b)=
	\begin{cases}
		\displaystyle
		\frac{1}{n(n-1)}
		\sum_{k\in J^A(i)}
		\operatorname{vol}_{n-2}\bigl(F_i(b)\cap F_k(b)\bigr)
		\bigl(b_k\csc\theta_{ik}-n b_i\cot\theta_{ik}\bigr),
		& j=i,\\[3mm]
		\displaystyle
		\frac{1}{n}
		\operatorname{vol}_{n-2}\bigl(F_i(b)\cap F_j(b)\bigr)
		\,b_i\csc\theta_{ij},
		& j\in J^A(i),\\[2mm]
		0,
		& \text{ otherwise}.
	\end{cases}
	\]
<!-- end theorem-9 -->
<!-- proof-20: body_md -->
The standard variation formula for the volume of a polytope with
	respect to its support numbers gives
	\[
	\frac{\partial}{\partial h_{ij}}\operatorname{vol}_{n-1}(F_i(b))
	=
	\operatorname{vol}_{n-2}\bigl(F_i(b)\cap F_j(b)\bigr);
	\]
	see, e.g., \cite[Section~5.1]{Schneider2014}. Hence
	\[
	\frac{\partial}{\partial b_j}\operatorname{vol}_{n-1}(F_i(b))
	=
	\begin{cases}
		\displaystyle
		-\sum_{k\in J^A(i)}
		\operatorname{vol}_{n-2}\bigl(F_i(b)\cap F_k(b)\bigr)\cot\theta_{ik},
		& j=i,\\[2mm]
		\displaystyle
		\operatorname{vol}_{n-2}\bigl(F_i(b)\cap F_j(b)\bigr)\csc\theta_{ij},
		& j\in J^A(i),\\[2mm]
		0,
		& \text{ otherwise}.
	\end{cases}
	\]
	Since
	\[
	f_i(b)=\frac{b_i}{n}\operatorname{vol}_{n-1}(F_i(b)),
	\]
	the off-diagonal formulas follow immediately. For the diagonal term,
	we use the Minkowski formula in the $(n-1)$-dimensional affine hull of
	$F_i(b)$:
	\[
	(n-1)\operatorname{vol}_{n-1}(F_i(b))
	=
	\sum_{k\in J^A(i)}
	h_{ik}\,
	\operatorname{vol}_{n-2}\bigl(F_i(b)\cap F_k(b)\bigr).
	\]
	Substituting the expression for $h_{ik}$ gives
	\[
	\frac{\partial f_i}{\partial b_i}(b)
	=
	\frac{1}{n(n-1)}
	\sum_{k\in J^A(i)}
	\operatorname{vol}_{n-2}\bigl(F_i(b)\cap F_k(b)\bigr)
	\bigl(b_k\csc\theta_{ik}-n b_i\cot\theta_{ik}\bigr),
	\]
	as claimed.

<!-- lemma-10: body_md -->
\label{lem:Facet_subset_relation}
	Let $P\subset\mathbb{R}^n$ be a polytope, $\dim P=n$, let
	$u\in\mathbb{S}^{n-1}$, and for $t\in\mathbb{R}$ let 
	\[
	F(t)\coloneqq P\cap\{x\in\mathbb{R}^n:\langle u,x\rangle=t\}.
	\]
	Suppose that, for some $b\in\mathbb{R}$ and $\varepsilon>0$,
	\[
	\operatorname{vol}_{n-1}(F(b-\delta))
	=
	\operatorname{vol}_{n-1}(F(b))
	>0
	\qquad\text{for all }\delta\in[0,\varepsilon].
	\]
	Then $F(b-\delta)$ is a translate of $F(b)$ for every
	$\delta\in(0,\varepsilon)$.
<!-- end lemma-10 -->
<!-- proof-21: body_md -->
Set $H=\{x\in\mathbb{R}^n:\langle u,x\rangle=0\}$ and $K_t\coloneqq F(t)-tu\subset H$. Fix
	$\delta=\lambda\varepsilon$ with $\lambda\in(0,1)$. Convexity of $P$
	gives
	\[
	(1-\lambda)K_b+\lambda K_{b-\varepsilon}
	\subseteq K_{b-\delta}.
	\]
	All three sets have the same positive $(n-1)$-dimensional volume.
	Hence Brunn--Minkowski in $H$ gives equality, and therefore
	$K_b$ and $K_{b-\varepsilon}$ are homothetic. Since their volumes are
	equal, they are translates; write
	\[
	K_{b-\varepsilon}=K_b+v.
	\]
	It follows that $K_b+\lambda v\subseteq K_{b-\delta}$. The two sets
	again have the same volume, and hence
	\[
	K_{b-\delta}=K_b+\lambda v.
	\]
	Thus
	\[
	F(b-\delta)=F(b)+\lambda v-\delta u,
	\]
	which proves the assertion.

<!-- proposition-5: body_md -->
\label{thm:gamma_linear_iff_zylinder}
	Let $U\in\mathcal{U}(n,m)$ and let $A\in\mathcal A(U)$ be a proper type-cone.
	Then $f_i^A$ has degree 1 in the variable $b_i$
	if and only if
	$
	\dim\operatorname*{lin} \{u_j:j\in J^A(i)\} =n-1.$
<!-- end proposition-5 -->
<!-- proof-22: body_md -->
Suppose first that $f_i^A$ has degree 1 in $b_i$. Since
	\[
	f_i^A(b)=\frac{b_i}{n}\operatorname{vol}_{n-1}(F_i(b)),
	\]
	the polynomial $\operatorname{vol}_{n-1}(F_i(b))$ is independent of $b_i$.
	Fix $b\in\operatorname*{int} A$ and choose $\varepsilon>0$ such that
	$b-\delta e_i\in\operatorname*{int} A$ for $0\leq\delta\leq\varepsilon$.
	Since only the $i$-th support number changes,
	\[
	F_i(b-\delta e_i)
	=
	P(U,b)\cap\{x:\langle u_i,x\rangle=b_i-\delta\}.
	\]
	Hence Lemma~\ref{lem:Facet_subset_relation} gives
	$F_i(b-\delta e_i)=F_i(b)+t_\delta$ 
	for $0<\delta<\varepsilon$. Apparently,  $t_\delta\ne 0$ 
	and for every $j\in J^A(i)$ the $h_{F_i(b-\delta e_i)}(u_j)=b_j=h_{F_i(b)}(u_j)$, and therefore
	\[
	\langle u_j,t_\delta\rangle=0.
	\]
	Thus all normals adjacent to $u_i$ lie in the hyperplane
	$\{x\in\mathbb{R}^n : \langle t_\delta, x\rangle=0\}$.
	
	Conversely, suppose that the normals adjacent to $u_i$ are contained in
	an $(n-1)$-dimensional subspace $L$. Choose
	$v\ne 0$ orthogonal to $L$. As 
	$u_i\notin L$ we may assume that
	$\langle u_i,v\rangle=1$. Then for  $b\in\operatorname*{int} A$ and sufficiently small $s>0$ we get  
	\[
	F_i(b+s e_i)=F_i(b)+sv,
	\]
	because $\langle u_j,v\rangle=0$ for every $j\in J^A(i)$.
	Consequently, $\operatorname{vol}_{n-1}(F_i(b))$ is independent of $b_i$, and hence
	$f_i^A(b)$ has degree 1 in  $b_i$.

	\beta
	=
	\lim_{\lambda\to\infty}
	\lambda^{-n}\operatorname{vol}_n(P_\lambda).
	\]
	Set
	\[
	Q\coloneqq \bigcap_{j\neq i}H^-_{u_j,\overline{b}_j},
	\qquad
	C\coloneqq \bigcap_{j\neq i}H^-_{u_j,0},
	\]
	where for $t\in\mathbb{R}$, $H^-_{u_j,t}=\{x\in\mathbb{R}^n :\langle u_j,x\rangle\leq t\}$. 
	Then $ P_\lambda=Q\cap H^-_{u_i,\lambda}$.  
	Since $u_i$ continues to define a facet of $P_\lambda$ for arbitrarily
	large $\lambda$, we have $h_Q(u_i)=\infty$, and hence
	$h_C(u_i)=\infty$.
	
	We claim that the recession cone $C$ is full-dimensional. Suppose otherwise and put
	\[
	V\coloneqq (\operatorname*{lin} C)^\perp,
	\qquad d\coloneqq \dim V>0.
	\]
	The projection $Q\vert V$ is a $d$-dimensional bounded polytope in
	$V$, and therefore has at least $d+1$ facets. Let $v\in V$ be an outer
	unit normal of one of these facets. Since $v$ vanishes on $\operatorname*{lin} C$, we have $F_Q(v)=F_Q(v)+C$,
	and, in particular, $F_Q(v)$ is a facet of $Q$. 

	
	Moreover, since $h_C(u_i)=\infty$, for all sufficiently large
	$\lambda$ the truncation of $F_Q(v)$ by
	$H^-_{u_i,\lambda}$ is a facet of $P_\lambda$ adjacent to
	$F_i(P_\lambda)$. Thus $v$ is one of the normals
	$\{u_j:j\in J^A(i)\}$. Consequently, at least $d+1$ normals adjacent to $u_i$ lie in the
	$d$-dimensional subspace $V$, contradicting their general position.
	Hence
	\[
	\dim C=n.
	\]
	
	The set $C\cap H^-_{u_i,1}$ is bounded. Indeed, otherwise there exists a non-zero $w$ 
	satisfying 
	\[
	\langle u_j,w\rangle\leq0
	\qquad\text{for every }j=1,\ldots,m,
	\]
	which is impossible since $U$ positively spans $\mathbb{R}^n$. Since $C$ is
	full-dimensional, $C\cap H^-_{u_i,1}$ has positive $n$-dimensional
	volume. Moreover, $\overline{b}_j>0$ for all $j$, so $C\subseteq Q$ and
	\[
	P_\lambda
	\supseteq
	C\cap H^-_{u_i,\lambda}.
	\]
	By homogeneity of the cone $C$,
	\[
	\begin{aligned}
		\beta
		=
		\lim_{\lambda\to\infty}
		\lambda^{-n}\operatorname{vol}_n(P_\lambda)
		\geq
		\lim_{\lambda\to\infty}
		\lambda^{-n}
		\operatorname{vol}_n(C\cap H^-_{u_i,\lambda})
		=
		\operatorname{vol}_n(C\cap H^-_{u_i,1})
		>0.
	\end{aligned}
	\]
	Thus $f_i^A$ contains the monomial $b_i^n$ with positive coefficient.

The case distinction in Table~\ref{tab:liu-quadrilateral} provides
	a vector $\gamma^{(0)}\in C_{{\tt cv}}(U)$ for which
	$|S(U,\gamma^{(0)})|=3$. For example, if $0<\lambda<1$ and
	$\sigma>0$, choose $\gamma_1,\gamma_3>0$ and set
	\[
	\gamma_4=\frac{\gamma_1\gamma_3}{\gamma_2}.
	\]
	As $\gamma_2\to\infty$,
	\[
	\left(
	\frac{\gamma_1+\gamma_2-\gamma_3-\gamma_4}
	{\gamma_1+\gamma_2+\gamma_3+\gamma_4}
	\right)^2
	\longrightarrow1,
	\]
	while the exceptional equality $\lambda=\lambda_0$ can be avoided.
	Hence, for sufficiently large $\gamma_2$, one of the
	three-solution cases in the table applies. The other parameter
	ranges are treated analogously.
	
	It remains to observe that the three solutions persist under small
	perturbations of $\gamma^{(0)}$. Suppose, for instance, that
	$\sigma>0$. By \eqref{eq:liu-quartic}, they correspond to three
	distinct roots of $q_{\gamma^{(0)}}$ in $(1,\infty)$. Moreover,
	\[
	q_\gamma(0)=\gamma_3>0,
	\qquad
	q_\gamma(1)=-(\rho-1)^2\gamma_4<0,
	\]
	so there is a fourth root in $(0,1)$. Thus all four roots are
	simple. Since simple roots depend continuously on the coefficients,
	the three roots in $(1,\infty)$ persist for all $\gamma$ in a
	sufficiently small neighbourhood of $\gamma^{(0)}$. The case
	$\sigma<0$ is analogous. Shrinking the neighbourhood if necessary
	yields a nonempty open set
	$\Omega\subset C_{{\tt cv}}(U)$ on which $|S(U,\gamma)|=3$.
