<!-- proposition-1: body_md -->
\label{prop:relationfrompsres}
	Suppose that $\mathcal L$ is a closed linear relation in a Banach space $X$ whose resolvent is nonempty, $\rho(\mathcal L)\neq\emptyset$. Then,
    \begin{align*}
			\mathcal L  = \operatorname{ran}
			\begin{bmatrix}
				(\mathcal L-\mu)^{-1}\\ I+\mu(\mathcal L-\mu)^{-1}
			\end{bmatrix} \quad \text{and} \quad  \begin{matrix} \ker (\mathcal L-\mu)^{-1} &=& \operatorname{mul} \mathcal L \\ 
			\operatorname{ran} (\mathcal L-\mu)^{-1} &=& \operatorname{dom} \mathcal L,\end{matrix}
\end{align*}
for all $\mu\in\rho(\mathcal L)$.

<!-- proposition-2: body_md -->
\label{prop:finite_spectrum}
	Let $\mathcal L$ be a linear relation in $\mathbb C^n$ with $\rho(\mathcal L)\neq\emptyset$. Then the spectrum of $\mathcal L$ is purely point and finite. That is, $\sigma(\mathcal L) = \sigma_p(\mathcal L)$ and $\sigma(\mathcal L)$ is a finite set. Furthermore, if $\mathcal L=\operatorname{ran}\left[\begin{smallmatrix}
	    \mathcal{B}\\ \mathcal{A}
	\end{smallmatrix}\right]$ for some $\mathcal{A},\mathcal{B}\in\mathbb{C}^{n\times n}$, then the following is equivalent:
\begin{enumerate}[(a)]
\item $\lambda\in\sigma_p(\mathcal{L})\setminus\{\infty\}$;
\item $\det(\lambda\mathcal{B}-\mathcal{A})=0$;
\item $0\in\sigma(\lambda\mathcal{B}-\mathcal{A})$. 
\end{enumerate}
Furthermore, $\infty\in\sigma_p(\mathcal{L})$ if and only if $\det(\mathcal{B})=0$, or equivalently, $0\in\sigma(\mathcal{B})$.

<!-- definition-1: body_md -->
Let $\mathcal{L}$ be a linear relation in $X$. The \emph{approximate point spectrum} of $\mathcal{L}$ is given by 
	\begin{equation*}
		\sigma_{\operatorname{ap}}(\mathcal{L})=\{\lambda \in \mathbb C: \exists \ (x_n, \tilde{x}_n) \in\mathcal{L}, \ \ \textup{such that } \ \|x_n\|=1 \ \textup{and} \ \lim_{n\to\infty} \|\tilde{x}_n-\lambda x_n \|=0 \}
	\end{equation*} and $\infty\in\sigma_{\operatorname{ap}}(\mathcal{L}) $ if $0\in\sigma_{\operatorname{ap}}(\mathcal L^{-1}).$

\label{definition:truncated_LR}
	Let $\mathcal{A}:X\to X$ and $\mathcal{B}:X\to X$ be bounded linear operators in a Hilbert space $X$ with orthogonal decomposition $X=\bigoplus_{i=1}^nX_i$. For $f = (f_1, \ldots, f_n) \in \mathcal{S}_{X_1} \times \cdots \times \mathcal{S}_{X_n}$ with $f_i\in X_i$. Define the isometry
$$
F_f:\mathbb{C}^n\longrightarrow X,
\qquad
F_f c:=\sum_{j=1}^n c_jf_j,
$$ and $\mathcal{A}_{f}\in \mathcal{M}_n(\mathbb{C})$ by
	\begin{align*} \label{matrix A_f definition}
		\mathcal{A}_{f}:=
F_f^* \mathcal A F_f= \begin{bmatrix}
		\langle \mathcal A f_1,f_1\rangle  & \cdots& \langle \mathcal A f_n, f_1\rangle \\ \vdots &  & \vdots \\ \langle \mathcal A f_1,f_n \rangle  & \cdots& \langle \mathcal A  f_n , f_n \rangle
		\end{bmatrix} \in\mathbb{C}^{n\times n}.
	\end{align*}

\eqref{eq:ph-structure}--\eqref{eq:ph-assumptions}, and let $\alpha$

    W^2(\mathcal B,\mathcal A)\setminus\{\infty\}

    \bigl\{

        \operatorname{Re}\lambda\leqslant\alpha
    \bigr\}.
\end{equation}

$f_1\in\mathcal S_{X_1}$ and
$f_2\in\mathcal S_{X_2}$, set

    \qquad

Let $x=(x_1,x_2)\in X$ be a unit vector. Then
$\langle\mathcal Bx,x\rangle
    =
    \|x_1\|^2$. If $x_1\neq0$, the corresponding finite scalar compressed eigenvalue
is
$\lambda
    =
    \tfrac{\langle\mathcal Ax,x\rangle}{\|x_1\|^2}$.
Since $\mathcal J$ is skew-adjoint,
$\operatorname{Re}\langle\mathcal Jx,x\rangle=0$,
and hence
\begin{align*}
\operatorname{Re}\lambda
    =
    -
    \frac{
        \langle R_1x_1,x_1\rangle
        +
        \langle R_2x_2,x_2\rangle
    }{\|x_1\|^2}\leqslant
    -
    \frac{\langle R_1x_1,x_1\rangle}{\|x_1\|^2}
    \leqslant
    \alpha.
\end{align*}
If $x_1=0$, then
\[
    \langle\mathcal Bx,x\rangle=0,
    \qquad
    \operatorname{Re}\langle\mathcal Ax,x\rangle
    =
    -\langle R_2x_2,x_2\rangle
    <0.
\]
Thus the corresponding scalar compressed relation is purely
multivalued and contributes the point $\infty$, but no finite
spectral point. This proves \eqref{eq:ph-W1-halfplane}.

It remains to prove \eqref{eq:ph-W2-W1}. Let
$\lambda\in
    W^2(\mathcal B,\mathcal A)\setminus\{\infty\}$.
Then there exist unit vectors $f_1\in X_1$, $f_2\in X_2$ and a
nonzero vector $z=(z_1,z_2)^\top\in\mathbb C^2$ such that
\[
    \mathcal A_fz
    =
    \lambda\mathcal B_fz.
\]
Since $d\neq0$, the second row of this equation implies that
$z_1\neq0$. Set
$x=F_fz=z_1f_1+z_2f_2$.
Taking the inner product of the compressed eigenvalue equation with
$z$ gives
\[
    \langle\mathcal Ax,x\rangle
    =
    \lambda\langle\mathcal Bx,x\rangle
    =
    \lambda|z_1|^2.
\]
Since $z_1\neq0$, after normalizing $x$, this shows
\[
    \lambda=\frac{\langle\mathcal Ax,x\rangle}
         {\langle\mathcal Bx,x\rangle}\in
    W^1(\mathcal B,\mathcal A)\setminus\{\infty\}.
\]
Finally, $\infty$ belongs to both block numerical ranges because the
compressed matrix $\mathcal B_f$ is singular. Hence
\eqref{eq:ph-W2-W1} follows.
