<!-- head-1: body_md -->
Given functions $u$ and $v$, $u(x) = \mathcal{O}(v(x))$ implies $u(x) \le c_1 v(x)$ for some $c_1>0$, and $u(x) = \Omega(v(x))$ implies $u(x) \ge c_2 v(x)$ for some $c_2>0$. The notations $\tilde{\mathcal{O}}$ and $\tilde{\Omega}$ hide logarithmic factors in $\mathcal{O}$ and $\Omega$, respectively. Let $\mathcal{N}(\mu,\Sigma)$ denote the Gaussian distribution with mean $\mu$ and covariance $\Sigma$. For a matrix $A$, let $\|A\|_F$ denote its Frobenius norm and $\|A\|_2$ denote its spectral norm. $I_n$ denotes the $n \times n$ identity matrix  and $\mathop{\mathrm{Im}}\nolimits(\cdot)$ denotes the column space of a matrix.

<!-- definition-1: body_md -->
\label{markov}
 ~ From the true system $(A,B,C,D)$, the Markov parameter matrix required to recover the input-output mapping is denoted as $G^*$ and defined by 
    \begin{equation}\label{gstar}
        G^* = [D~CB~ CAB~ \dots ~ CA^{k-2}B].
    \end{equation}
    Note that the size of $G^*$ depends on $k$; however, we omit the explicit dependence on $k$ in the notation for simplicity.

<!-- assumption-2: body_md -->
\label{stability}
There exist constants $\psi > 0$ and $\rho \in [0, 1)$ such that $\|A^t\|_2 \le \psi \cdot \rho^t$ for all $t \ge 0$.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{maxnorm}
 Define a filtration {$\mathcal{F}_t = \bm{\sigma}\{x_0, x_1, \dots, x_t\}$.}
    There exists $\eta >0$ such that
  {   $x_0 = (x_{0,i})_{i=1}^n$ and $w_t = (w_{t,i})_{i=1}^e$ satisfies $\max_{i\in\{1,\dots,n\}}\|x_{0,i}\|_{\psi_2} \leq \eta$ and $\max_{i\in\{1,\dots,e\}}\|w_{t,i}\;|\;\mathcal{F}_t\|_{\psi_2} \leq \eta$ for all $t \geq 0$.}
<!-- end assumption-3 -->
<!-- assumption-4: body_md -->
\label{null}
 $w_t$ is an attack at each time $t$ with probability $p<0.5$ conditioned on $\mathcal{F}_t$; \textit{i.e.}, there exists a sequence $(\xi_t)_{t\geq 0}$ of independent $\text{Bernoulli}(p)$ variables such that 
    $\{\xi_t = 0\} \subseteq \{w_t=0\}$ for all $t\geq 0.$
   Under this restriction on attack times, $w_t$ can be chosen arbitrarily by an adversary with access to $\mathcal{F}_t$ at every attack time $t\ge 0$.

\end{align*}

<!-- lemma-3: body_md -->
\label{K_lowerbound}
Let the control inputs $\{u_t\}_{t=0}^{T+k-2}$ be independent with $u_t \sim \mathcal{N}(0, \sigma^2 I_m)$. Consider a fixed matrix $K \in \mathbb{R}^{e \times mk}$. Then, given $\delta \in (0,1]$, when

\end{align*}

\begin{align}\label{KU_T}
    \sum_{t\in\mathcal{T}_0} \|K \mathbf{U}_t^{(k)}\|_2 \ge \frac{(1-p)^{s-1}\sigma}{4} \sqrt{\frac{2}{\pi}} \|K\|_F T
\end{align}

<!-- lemma-5: body_md -->
\label{lem:psi_lower_bound}
Let the control inputs $\{u_t\}_{t=0}^{T+k-2}$ be independent with $u_t \sim \mathcal{N}(0, \sigma^2 I_m)$. For $K \in \mathbb{R}^{e \times mk}$ and $\tilde Q_s \in \mathbb{R}^{e\times e(s-1)}$, consider a function 
\begin{equation}\label{f_T_K_Q}
\begin{split}
    f_T(K,\tilde Q_s):=  &\sum_{t\in\mathcal{T}}\|K \mathbf{U}_t^{(k)}  + \tilde Q_s\mathbf{\tilde W}_t^{(s)}\|_2\\&\hspace{0mm}-2 \min\Bigl\{\|K \mathbf{U}_t^{(k)}  + \tilde Q_s\mathbf{\tilde W}_t^{(s)}\|_2,\|w_{t-1}\|_2\Bigr\}
    \end{split}
\end{equation}
where $\mathbf{\tilde W}_t^{(s)} := [w_{t-2}^T ~ \dots ~ w_{t-s}^T]^T$. 
Given $h>0$ and  $\delta \in (0, 1]$, when
\begin{equation}\label{eq:T_condition_psi}
\begin{split}
    T = \Omega\biggr(& \frac{k}{(1-p)^{s-1}} \log\left(\frac{1}{\delta}\right) \\&+ \frac{\eta^2 e \log(1/\delta)}{ h^2 \rho^{2s}} \log\left(\frac{1}{\delta} \cdot\frac{\eta^2 e \log(1/\delta)}{  \rho^{2s}  h^2} \right) \biggr),
    \end{split}
\end{equation}
the infimum $\Psi_T(K) := \inf_{\tilde Q_s} f_T(K, \tilde Q_s)$ satisfies
\begin{align}\label{psi_lowerbound}
    \Psi_T(K) \ge \frac{(1-2p)(1-p)^{s-1}\sigma}{4} \sqrt{\frac{2}{\pi}} \|K\|_F T -  \rho^s h T.
\end{align}
 with probability at least $1-\delta$.
<!-- end lemma-5 -->
<!-- lemma-6: body_md -->
\label{lem:universallowerbound}
 Let the control inputs $\{u_t\}_{t=0}^{T+k-2}$ be independent with $u_t \sim \mathcal{N}(0, \sigma^2 I_m)$.  Define $c_* := \frac{(1-2p)(1-p)^{s-1}\sigma}{4} \sqrt{\frac{2}{\pi}}$ and 
 $R:= \frac{4\rho^s h}{c_*}$ for $h>0$. Let $\mathcal{S}_R := \{K \in \mathbb{R}^{e \times mk} : \|K\|_F = R\}$ denote the Frobenius sphere of radius $R$. Given $\delta\in(0,1]$, when 
 \begin{align}\label{final_timebound}
    &T=\tilde \Omega\left(\frac{e  m k^2s }{(1-p)^{s-1}}  +\frac{\eta^2 e \cdot  (emks)^2 }{ \rho^{2s}h^2 } \right),
\end{align}
we have
 \begin{align}\label{goal}
    \Psi_T(K) \ge \frac{c_*}{4} RT,\quad \forall K\in \mathcal{S}_R 
    \end{align}
    with probability at least $1-\delta$.
<!-- end lemma-6 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Let the control inputs $\{u_t\}_{t=0}^{T+k-2}$ be independent with $u_t \sim \mathcal{N}(0, \sigma^2 I_m)$. Consider a minimizer $(\hat H, \hat J)$ to the constrained $\ell_2$-norm estimator \eqref{l2est}, and define $\hat K := \hat H G^* - \hat J$. Given $\delta \in (0, 1]$, when
\begin{align}\label{final_timebound2}
    T = \tilde{\Omega}\bigg( \frac{e m k^2 s}{(1-p)^{s-1}} + \frac{\eta^2 e\cdot (e m k s)^2}{\rho^{2s} (\|\hat H\|_2+\|H^*\|_2)^2} \bigg),
\end{align}
then  the estimation error satisfies
\begin{align}
\|\hat K\|_F = \mathcal{O}\bigg(\Big(\frac{\rho}{1-p}\Big)^s\frac{(1-p)(\|\hat H\|_2 + \|H^*\|_2)(1+\|C\|_2 M)}{(1-2p) \sigma} \bigg)  
\end{align}
with probability at least $1-\delta$, where $M$ is defined in Lemma \ref{alphabetabound}.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
For any scalar $\alpha \ge 1$ and $a, b \ge 0$, we have $\min\{\alpha a, b\} \le \alpha \min\{a, b\}$. Applying this inequality to $f_T$ yields $f_T(\alpha K, \alpha Q) \ge \alpha f_T(K, Q)$ for all $\alpha \ge 1$. Taking the infimum over $Q$ on both sides provides
\begin{equation}\label{alphalower}
\begin{split}
    \Psi_T (\alpha K ) &= \inf_{\tilde Q_s} f_T(\alpha K , \tilde Q_s)= \inf_{ Q} f_T(\alpha K , \alpha  Q)\\& \ge \alpha \inf_Qf_T(K,Q) = \alpha \Psi_T(K)
    \end{split}
\end{equation}
for all $\alpha\ge 1$.
Setting $h := \|\hat H\|_2 +\|H^*\|_2> 0$ and $R := \frac{4\rho^s (\|\hat H\|_2+\|H^*\|_2)}{c_*}$, Lemma \ref{lem:universallowerbound} guarantees that when $T$ satisfies \eqref{final_timebound2},
the universal lower bound \eqref{goal} holds
with probability at least $1-\delta$.

Now, for any matrix $\relax K$ satisfying $\|\relax K\|_F \ge R$, consider the scaling parameter $\alpha := \|\relax K\|_F / R \ge 1$. Since $\|\relax K / \alpha\|_F = R$, sequentially applying \eqref{alphalower} and \eqref{goal} yields
\begin{align}\label{psitkbar}

with probability at least $1-\delta$.

\begin{itemize}[leftmargin=0.4cm]

\end{align}

    \epsilon := \frac{c_* R}{4 \sigma \sqrt{mk}}.
\end{align*}
The covering number $N_\epsilon$ needed to cover $\mathcal{S}_R$ with $\epsilon$-balls in Frobenius norm satisfies
\begin{align*}
    N_\epsilon \le \left( 1 + \frac{2R}{\epsilon} \right)^{e m k} =\left( 1 +  \frac{16\sqrt{2\pi}\cdot \sqrt{mk}}{(1-2p)(1-p)^{s-1}} \right)^{e m k}.
\end{align*}

For each fixed grid point $\tilde K $ in the $\epsilon$-net, applying Lemma \ref{lem:psi_lower_bound} with failure probability $\frac{\delta}{2N_\epsilon}$ ensures that $\Psi_T(\tilde K) \ge c_* R T - \rho^s h T$ with probability at least $1 - \frac{\delta}{2N_\epsilon}$.

By a union bound over all $N_\epsilon$ grid points in the $\epsilon$-net, the pointwise lower bound holds simultaneously across the entire net with probability at least $1 - \frac{\delta}{2}$. Combining this with the  event in \eqref{Kinterbound} (which holds with probability at least $1 - \frac{\delta}{2}$), for any $K \in \mathcal{S}_R$, choosing a net point $\tilde K$ such that $\|K - \tilde K\|_F \le \epsilon$ yields
\begin{align*}
    \Psi_T(K) &\ge \Psi_T(\tilde K) - 2 T \sigma \sqrt{mk} \, \epsilon \\
    &\ge \left( c_* R T - \rho^s h T \right) - \frac{c_*}{2} R T = \frac{c_*}{4} R T
\end{align*}
with probability at least $1 - \delta$, where the last equality is from the definition of $R$.

To guarantee the union bound, we replace  $\delta $ in \eqref{eq:T_condition_psi} with  $\frac{\delta}{2N_\epsilon}$. Expanding the logarithm yields $\log(\frac{2N_\epsilon}{\delta}) = \mathcal{O}\big( e m k \log\big( \frac{mk}{(1-2p)(1-p)^{s-1}} \big) + \log(\frac{1}{\delta}) \big)=\mathcal{\tilde O}(emks)$, which leads to
the final time bound in \eqref{final_timebound}, hiding logarithmic factors. Note that  $T=\Omega\bigr( \frac{1}{m}\log\left(\frac{2}{\delta}\right)\bigr)$ required for \eqref{Kinterbound} is dominated by \eqref{final_timebound}. This completes the proof.
