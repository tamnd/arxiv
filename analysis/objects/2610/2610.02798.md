<!-- proposition-1: body_md -->
\label{prop:rotation_invariance}
Suppose that $\bm{\Theta}(t)$ follows either GF or Spectral GF dynamics under the embedding $\bm{\phi}(\cdot)$. Let $\mathbf{Q}\in\mathbb{R}^{d\times d}$ be an orthogonal matrix. Then $\widetilde\bm{\Theta}(t)=(\mathbf{Q}\mathbf{W}_{OV}(t)\mathbf{Q}^\top, \mathbf{Q}\mathbf{W}_{KQ}(t)\mathbf{Q}^\top)$ also follows the GF or Spectral GF dynamics under the embedding $\widetilde\bm{\phi}(\cdot)=\mathbf{Q}\bm{\phi}(\cdot)$.
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
\label{remark:signGF_embedding_dependent}
In general, Proposition \ref{prop:rotation_invariance} does not hold for Sign GF. A crucial property for Proposition~\ref{prop:rotation_invariance} to hold for Spectral GF is that for any matrix $\mathbf{M}$ and orthogonal matrices $\mathbf{U}$, $\mathbf{V}$, the spectral orthogonalization satisfies
$\operatorname{msign}(\mathbf{U}\mathbf{M}\mathbf{V}^\top)=\mathbf{U}\cdot\operatorname{msign}(\mathbf{M})\cdot\mathbf{V}^\top$ (Lemma~\ref{lemma:msign_rotation}).
However, this property does not hold for the element-wise sign operator in general. In Section~\ref{section:SignGF}, we will study how the Sign GF dynamics change across different orthonormal embeddings of the same dictionary.

<!-- proposition-2: body_md -->
\label{prop:prediction}
Suppose $\bm{\Theta}\in\mathcal{M}$. Then the prediction of the linear attention model satisfies that for any $(s', r')$,
\begin{align*}
\widehat p_{\bm{\Theta}}(a_*(s', r')|z_{1:T})=\frac{e^{\delta_{s', s(z_{1:T})}\cdot\alpha_S\beta_S/\sqrt R}}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{e^{\delta_{r', r(z_{1:T})}\cdot\alpha_R\beta_R/\sqrt S}}{R-1+e^{\alpha_R\beta_R/\sqrt S}}.
\end{align*}
Specifically, the model accuracy satisfies
\begin{align*}
\widehat p_{\bm{\Theta}}(z_{T+1}|z_{1:T})=\frac{1}{1+(S-1)e^{-\alpha_S\beta_S/\sqrt R}}\cdot\frac{1}{1+(R-1)e^{-\alpha_R\beta_R/\sqrt S}}.
\end{align*}

<!-- proposition-3: body_md -->
\label{prop:GF}
Suppose the model parameters $\bm{\Theta}\in\mathcal{M}$. Define $\widetilde\mathbf{v}_s$ and $\widetilde\mathbf{v}_r$ as
\begin{align}\label{eq:def_tildev}
\widetilde\mathbf{v}_s=\bar\mathbf{v}_s-\bar\mathbf{v}/\sqrt{S},\quad\widetilde\mathbf{v}_r=\bar\mathbf{v}_r-\bar\mathbf{v}/\sqrt R.
\end{align}
Then the gradient at $\bm{\Theta}$ satisfies
\begin{align*}
-\nabla_{\mathbf{W}_{OV}}L(\bm{\Theta})&=\frac{\beta_S/\sqrt R}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}(s)^\top+\frac{\beta_R/\sqrt S}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\sum_{r\in\mathcal{R}}\widetilde\mathbf{v}_r\bm{\phi}(r)^\top;\\
-\nabla_{\mathbf{W}_{KQ}}L(\bm{\Theta})&=\bigg[\frac{(S-1)\alpha_S/(S\sqrt R)}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\sum_{s\in{\mathcal{S}}}\bm{\phi}(s)+\frac{(R-1)\alpha_R/(R\sqrt S)}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\sum_{r\in\mathcal{R}}\bm{\phi}(r)\bigg]\bm{\phi}(\operatorname{EOS})^\top.
\end{align*}

<!-- theorem-1: body_md -->
\label{theorem:GF}
Assume that $S, R\ge2$, $\delta\le0.5$, and Assumption~\ref{assumption:init} holds with $\alpha_0, \beta_0<1$.
Then the learning times of the GF dynamics satisfy
\begin{align*}
T_S^\delta&\simeq\sqrt{SR(S-1)}\bigg\{\log\frac{\sqrt R\log(S-1)}{\alpha_0+\beta_0}+\frac{\delta^{-1}-1}{\log[(\delta^{-1}-1)(S-1)]}\bigg\},\\
T_R^\delta&\simeq\sqrt{SR(R-1)}\bigg\{\log\frac{\sqrt S\log(R-1)}{\alpha_0+\beta_0}+\frac{\delta^{-1}-1}{\log[(\delta^{-1}-1)(R-1)]}\bigg\}.
\end{align*}
<!-- end theorem-1 -->
<!-- remark-2: body_md -->
We notice from the ODE that the subject parameters $(\alpha_S, \beta_S)$ and the relation parameters $(\alpha_R, \beta_R)$ form two isolated systems (the dynamics are independent from each other). We hereby interpret the behavior of the subject and relation parameters in Theorem~\ref{theorem:GF} separately, each consisting of two phases. For the subject parameters $(\alpha_S, \beta_S)$, the term
\begin{align*}
\sqrt{SR(S-1)}\log\frac{\sqrt R\log(S-1)}{\alpha_0+\beta_0}\simeq T_S^{0.5}
\end{align*}
corresponds to \textbf{Phase I} where $q_S\le1$. In this phase, the term $S-1+e^{\alpha_S\beta_S/\sqrt R}$, which stems from the normalizing factor of softmax, satisfies $S-1\ge e^{\alpha_S\beta_S/\sqrt R}$, so the evolution of $(\alpha_S, \beta_S)$ is close to a linear ODE system. The term
\begin{align*}
\sqrt{SR(S-1)}\cdot\frac{\delta^{-1}-1}{\log[(\delta^{-1}-1)(S-1)]}
\end{align*}
corresponds to \textbf{Phase II} where $q_S\ge1$. In this phase, we have $e^{\alpha_S\beta_S/\sqrt R}\ge S-1$, so the increase of $\alpha_S$ and $\beta_S$ slows down due to the saturated softmax. It takes time $\widetilde\Theta(\delta^{-1})$, ignoring factors other than $\delta$, for the subject-dependent component in the model accuracy to reach $1-\delta$, which is consistent with prior works~\citep{soudry2018implicit, ji2018risk, ji2019implicit, lyu2019gradient}. Interpretations for the relation-dependent parameters $(\alpha_R, \beta_R)$ are similar.
<!-- end remark-2 -->
<!-- remark-3: body_md -->
We now consider the learning time ratio $T_S^\delta/T_R^\delta$. If the target error rate $\delta$ is viewed as a constant, then the terms in $T_S^\delta$ and $T_R^\delta$ corresponding to Phase II can be ignored. If we further assume that the initialization is in the order of constants, i.e., $\alpha_0+\beta_0=\Theta(1)$, then
\begin{align*}
\frac{T_S^\delta}{T_R^\delta}\simeq\sqrt{\frac{S-1}{R-1}}\cdot\frac{\log(\sqrt R\log(S-1))}{\log(\sqrt S\log(R-1))}.
\end{align*}
If we aim for high accuracy, i.e., $\delta^{-1}\ge\max\{S, R, \log(\alpha_0+\beta_0)^{-1}\}$, then the term corresponding to Phase II is dominating. In this case, the learning-time ratio satisfies
\begin{align*}
T_S^\delta/T_R^\delta\simeq\sqrt{(S-1)/(R-1)}.
\end{align*}
In both cases, the learning-time ratio is $\widetilde\Theta(\sqrt{S/R})$.

<!-- proposition-4: body_md -->
\label{prop:spectralGF}
Suppose the model parameters $\bm{\Theta}\in\mathcal{M}$. Let $\widetilde\mathbf{v}_s$ and $\widetilde\mathbf{v}_r$ be defined in \eqref{eq:def_tildev}. Then the spectral orthogonalization of the gradient at $\bm{\Theta}$ satisfies
\begin{align*}
&\operatorname{msign}(-\nabla_{\mathbf{W}_{OV}}L(\bm{\Theta}))=\mathop{\mathrm{sign}}(\beta_S)\cdot\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}(s)^\top+\mathop{\mathrm{sign}}(\beta_R)\cdot\sum_{r\in\mathcal{R}}\widetilde\mathbf{v}_r\bm{\phi}(r)^\top.\\
&\operatorname{msign}(-\nabla_{\mathbf{W}_{KQ}}L(\bm{\Theta}))=\bigg[\frac{(S-1)^2\alpha_S^2}{(S-1+e^{\alpha_S\beta_S/\sqrt R})^2}+\frac{(R-1)^2\alpha_R^2}{(R-1+e^{\alpha_R\beta_R/\sqrt S})^2}\bigg]^{-1/2}\\
&\quad\cdot\bigg[\frac{(S-1)/\sqrt S\cdot\alpha_S}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\sum_{s\in{\mathcal{S}}}\bm{\phi}(s)+\frac{(R-1)/\sqrt R\cdot\alpha_R}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\sum_{r\in\mathcal{R}}\bm{\phi}(r)\bigg]\cdot\bm{\phi}(\operatorname{EOS})^\top.
\end{align*}

<!-- theorem-2: body_md -->
\label{theorem:SpectralGF}
Suppose $S\ge R\ge 4$, and Assumption~\ref{assumption:init} holds with
$\alpha_0\le(SR)^{1/4}$ and $\beta_0\le(R/S)^{1/4}$.
Then for any target error rate $\delta\le0.5$, the learning times of spectral GF satisfy
\begin{align*}
T_S^\delta&\simeq(SR)^{1/4}\sqrt{\log[(\delta^{-1}-1)(S-1)]}.\\
T_R^\delta&\simeq\begin{cases}
(SR)^{1/4}\frac{\log[(\delta^{-1}-1)(R-1)]}{\sqrt{\log(R-1)}} & \text{if}\quad\log(\delta^{-1}-1)\le c\cdot\sqrt{\log(S-1)\cdot\log(R-1)};\\
(SR)^{1/4}\sqrt{\log[(\delta^{-1}-1)(S-1)]} & \text{otherwise}.
\end{cases}
\end{align*}
<!-- end theorem-2 -->
<!-- remark-4: body_md -->
Compared with the GF dynamics, the learning times under the Spectral GF dynamics has no polynomial dependence on $\delta^{-1}$, which is attributed to the spectral orthogonalization weakening the effect of softmax saturation.
Unlike GF, the phase transition behavior of the learning times $T_S^\delta$ and $T_R^\delta$ under Spectral GF is asymmetric, due to the coupled dynamics of $\beta_S$ and $\beta_R$. We show that $\beta_S$ increases almost linearly in $t$ (See Lemma~\ref{lemma:betaS_lower_betaR_upper}), so $T_S^\delta$ has no phase transition behavior.
The two phases of \(T_R^\delta\) are determined by whether the desired relation accuracy is reached before or after the subject-dependent component becomes well learned (See proof for details).
<!-- end remark-4 -->
<!-- remark-5: body_md -->
We now consider the learning-time ratio under the Spectral GF dynamics.
If the target error rate $\delta$ is viewed as a constant, then
\begin{align*}
T_S^\delta\simeq(SR)^{1/4}\sqrt{\log(S-1)},\quad T_R^\delta\simeq(SR)^{1/4}\sqrt{\log(R-1)},\quad\frac{T_S^\delta}{T_R^\delta}\simeq\sqrt{\frac{\log(S-1)}{\log(R-1)}}.
\end{align*}
We also consider the asymptotics of $\delta\to0$. In this case, the learning-time ratio is $T_S^\delta/T_R^\delta\simeq1$.
In both cases, we have $T_S^\delta/T_R^\delta=\widetilde\Theta(1)$. Therefore, compared with the GF dynamics, the learning-time ratio under the spectral GF dynamics has no polynomial dependence on $S/R$.
This effect is mainly attributed to the spectral orthogonalization reweighing the subject- and relation-dependent components in the gradient of $\mathbf{W}_{OV}$.

<!-- proposition-5: body_md -->
\label{prop:signGF}
In all possible combinations of the embeddings defined above, if $\Theta\in\mathcal{M}$, then the element-wise sign of the gradient satisfies $\mathop{\mathrm{sign}}(-\nabla L(\bm{\Theta}))\in\mathcal{M}$. Furthermore, if Assumptions~\ref{assumption:init} holds, then the Sign GF dynamics satisfy $\dot\alpha_S=C_{\alpha_S}$, $\dot\beta_S=C_{\beta_S}$, $\dot\alpha_R=C_{\alpha_R}$, and $\dot\beta_R=C_{\beta_R}$ (details given in Table~\ref{table:signGF}). If $\alpha_0$ and $\beta_0$ are sufficiently small, then the learning times satisfy
\begin{align*}
T_S^\delta=\widetilde\Theta(R^{1/4}(C_{\alpha_S}C_{\beta_S})^{-1/2}),\quad T_R^\delta=\widetilde\Theta(S^{1/4}(C_{\alpha_R}C_{\beta_R})^{-1/2}).
\end{align*}

<!-- proof-1: body_md -->
Let $\widetilde\mathbf{X}$ denote the embedding matrix under the embedding $\widetilde\bm{\phi}(\cdot)$, then
\begin{align*}
\widetilde\mathbf{X}=[\mathbf{Q}\bm{\phi}(z_1), \dots, \mathbf{Q}\bm{\phi}(T)]^\top=\mathbf{X}\mathbf{Q}^\top.
\end{align*}
We first notice that for any $a$, the logit satisfies
\begin{align*}
\langle\widetilde\bm{\phi}(a), \mathbf{f}(\widetilde\mathbf{X}; \widetilde\bm{\Theta})\rangle&=(\mathbf{Q}\bm{\phi}(a))^\top(\mathbf{Q}\mathbf{W}_{OV}\mathbf{Q}^\top)\cdot(\mathbf{X}\mathbf{Q}^\top)^\top(\mathbf{X}\mathbf{Q}^\top)\cdot(\mathbf{Q}\mathbf{W}_{KQ}\mathbf{Q}^\top)\cdot(\mathbf{Q}\bm{\phi}(\operatorname{EOS}))\\
&=\bm{\phi}(a)^\top\mathbf{W}_{OV}\mathbf{X}^\top\mathbf{X}\mathbf{W}_{KQ}\bm{\phi}(\operatorname{EOS})\\
&=\langle\bm{\phi}(a), \mathbf{f}(\mathbf{X}; \bm{\Theta})\rangle.
\end{align*}
where the first equality holds because $\mathbf{U}^\top\mathbf{U}=\mathbf{I}$. Therefore, the logits are invariant under the transformation, so
\begin{align*}
\widehat p_{\bm{\Theta}}(a|z_{1:T+1})=\widehat p_{\widetilde\bm{\Theta}}(a|z_{1:T+1}).
\end{align*}

We then calculate the gradient of $\ell(z_{1:T+1}; \bm{\Theta})$. The gradient w.r.t. $\mathbf{W}_{OV}$ satisfies
\begin{align}\label{eq:grad_ell_WOV}
-\nabla_{\mathbf{W}_{OV}}\ell(z_{1:T+1}; \bm{\Theta})=\bigg\{\sum_{a\in\mathcal{A}}\widehat p_{\bm{\Theta}}(a|z_{1:T})[\bm{\phi}(z_{T+1})-\bm{\phi}(a)]\bigg\}\bm{\phi}(\operatorname{EOS})^\top\mathbf{W}_{KQ}^\top\mathbf{X}^\top\mathbf{X}.
\end{align}
The gradient w.r.t. $\mathbf{W}_{KQ}$ satisfies
\begin{align}\label{eq:grad_ell_WKQ}
-\nabla_{\mathbf{W}_{KQ}}\ell(z_{1:T+1}; \bm{\Theta})=\mathbf{X}^\top\mathbf{X}\mathbf{W}_{OV}^\top\bigg\{\sum_{a\in\mathcal{A}}\widehat p_{\bm{\Theta}}(a|z_{1:T})[\bm{\phi}(z_{T+1})-\bm{\phi}(a)]\bigg\}\bm{\phi}(\operatorname{EOS})^\top.
\end{align}
Under the parameter transform, we notice that
\begin{align*}
&\widetilde\bm{\phi}(a)\widetilde\bm{\phi}(\operatorname{EOS})^\top\widetilde\mathbf{W}_{KQ}^\top\widetilde\mathbf{X}^\top\widetilde\mathbf{X}\\
&=(\mathbf{Q}\bm{\phi}(a))\cdot(\mathbf{Q}\bm{\phi}(\operatorname{EOS}))^\top\cdot(\mathbf{Q}\mathbf{W}_{KQ}\mathbf{Q}^\top)^\top\cdot(\mathbf{X}\mathbf{Q}^\top)^\top\cdot(\mathbf{X}\mathbf{Q}^\top)\\
&=\mathbf{Q}\cdot(\bm{\phi}(a)\bm{\phi}(\operatorname{EOS})\mathbf{W}_{KQ}^\top\mathbf{X}^\top\mathbf{X})\cdot\mathbf{Q}^\top,
\end{align*}
and
\begin{align*}
&\widetilde\mathbf{X}^\top\widetilde\mathbf{X}\widetilde\mathbf{W}_{OV}^\top\widetilde\bm{\phi}(a)\widetilde\bm{\phi}(\operatorname{EOS})^\top\\
&=(\mathbf{X}\mathbf{Q}^\top)^\top\cdot(\mathbf{X}\mathbf{Q}^\top)\cdot(\mathbf{Q}\mathbf{W}_{OV}\mathbf{Q}^\top)^\top\cdot(\mathbf{Q}\bm{\phi}(a))\cdot(\mathbf{Q}\bm{\phi}(\operatorname{EOS}))^\top\\
&=\mathbf{Q}\cdot(\mathbf{X}^\top\mathbf{X}\mathbf{W}_{OV}^\top\bm{\phi}(a)\bm{\phi}(\operatorname{EOS})^\top)\cdot\mathbf{Q}^\top.
\end{align*}
Combining the properties above with the fact that $\widehat p_{\bm{\Theta}}=\widehat p_{\widetilde\bm{\Theta}}$, we have
\begin{align*}
-\nabla_{\widetilde\mathbf{W}_{OV}}\ell(z_{1:T+1}; \widetilde\bm{\Theta})&=\mathbf{U}\cdot[-\nabla_{\mathbf{W}_{OV}}\ell(z_{1:T+1}; \bm{\Theta})]\cdot\mathbf{U}^\top,\\
-\nabla_{\widetilde\mathbf{W}_{KQ}}\ell(z_{1:T+1}; \widetilde\bm{\Theta})&=\mathbf{U}\cdot[-\nabla_{\mathbf{W}_{KQ}}\ell(z_{1:T+1}; \bm{\Theta})]\cdot\mathbf{U}^\top.
\end{align*}
Taking expectations of the sequence $\mathbb{E}_{z_{1:T+1}}$, we have
\begin{align*}
-\nabla_{\widetilde\mathbf{W}_{OV}}L(\widetilde\bm{\Theta})=\mathbf{Q}\cdot[-\nabla_{\mathbf{W}_{OV}}L(\bm{\Theta})]\cdot\mathbf{Q}^\top,\quad-\nabla_{\widetilde\mathbf{W}_{KQ}}L(\widetilde\bm{\Theta})=\mathbf{Q}\cdot[-\nabla_{\mathbf{W}_{KQ}}L(\bm{\Theta})]\cdot\mathbf{Q}^\top.
\end{align*}
Therefore, under the GF dynamics, we have
\begin{align*}
\frac{{\,\mathrm{d}}\widetilde\mathbf{W}_{OV}(t)}{{\,\mathrm{d}} t}&=\mathbf{Q}\cdot\frac{{\,\mathrm{d}}\mathbf{W}_{OV}(t)}{{\,\mathrm{d}} t}\cdot\mathbf{Q}^\top\\
&=\mathbf{Q}\cdot[-\gamma_{\mathrm{GF}}\cdot\nabla_{\mathbf{W}_{OV}}L(\bm{\Theta})]\cdot\mathbf{Q}^\top\\
&=-\gamma_{\mathrm{GF}}\cdot\nabla_{\widetilde\mathbf{W}_{OV}}L(\widetilde\bm{\Theta}).
\end{align*}
Similarly, we have
\begin{align*}
\frac{{\,\mathrm{d}}\widetilde\mathbf{W}_{KQ}(t)}{{\,\mathrm{d}} t}=-\gamma_{\mathrm{GF}}\cdot\nabla_{\widetilde\mathbf{W}_{KQ}}L(\widetilde\bm{\Theta}).
\end{align*}
Therefore, $\widetilde\bm{\Theta}$ follows the GF dynamics. Under the Spectral GF dynamics, we have
\begin{align*}
\frac{{\,\mathrm{d}}\widetilde\mathbf{W}_{OV}(t)}{{\,\mathrm{d}} t}&=\mathbf{Q}\cdot\frac{{\,\mathrm{d}}\mathbf{W}_{OV}(t)}{{\,\mathrm{d}} t}\cdot\mathbf{Q}^\top\\
&=\mathbf{Q}\cdot[-\gamma_{\mathrm{SpectralGF}}\cdot\operatorname{msign}(\nabla_{\mathbf{W}_{OV}}(L(\bm{\Theta})))]\cdot\mathbf{Q}^\top\\
&=-\gamma_{\mathrm{SpectralGF}}\cdot\operatorname{msign}(\mathbf{Q}\cdot\nabla_{\mathbf{W}_{OV}}(L(\bm{\Theta}))\cdot\mathbf{Q}^\top)\\
&=-\gamma_{\mathrm{SpectralGF}}\cdot\operatorname{msign}(\nabla_{\mathbf{W}_{OV}}L(\widetilde\bm{\Theta})),
\end{align*}
where the third equality holds by Lemma \ref{lemma:msign_rotation}. Similarly,
\begin{align*}
\frac{{\,\mathrm{d}}\widetilde\mathbf{W}_{KQ}(t)}{{\,\mathrm{d}} t}=-\gamma_{\mathrm{SpectralGF}}\cdot\operatorname{msign}(\nabla_{\mathbf{W}_{kQ}}L(\widetilde\bm{\Theta})).
\end{align*}
Therefore, $\widetilde\bm{\Theta}$ follows the Spectral GF dynamics.

We use $s(z_{1:T})$ and $r(z_{1:T})$ to denote the subject and relation tokens of the sequence $z_{1:T}$, respectively.
Let $C(z, z_{1:T})$ be the number of occurrences of the token $z$ in the sequence $z_{1:T}$. Then
\begin{align}\label{eq:XTX}
\mathbf{X}^\top\mathbf{X}=\bm{\phi}(s(z_{1:T}))\bm{\phi}(s(z_{1:T}))^\top+\bm{\phi}(r(z_{1:T}))\bm{\phi}(r(z_{1:T}))^\top+\sum_{n\in\mathcal{N}}C(n, z_{1:T})\bm{\phi}(n)\bm{\phi}(n)^\top.
\end{align}
Recall the definition of the manifold:
\begin{align*}
\mathbf{W}_{OV}&=\sum_{s\in{\mathcal{S}}}(\alpha_S\bar\mathbf{v}_s+\gamma_S\bar\mathbf{v})\bm{\phi}(s)^\top+\sum_{r\in\mathcal{R}}(\alpha_R\bar\mathbf{v}_r+\gamma_R\bar\mathbf{v})\bm{\phi}(r)^\top+\sum_{n\in\mathcal{N}}\gamma_N\bar\mathbf{v}\bm{\phi}(n)^\top,\\
\mathbf{W}_{KQ}&=\bigg(\beta_S\sum_{s\in{\mathcal{S}}}\bm{\phi}(s)+\beta_R\sum_{r\in\mathcal{R}}\bm{\phi}(r)+\beta_N\sum_{n\in\mathcal{N}}\bm{\phi}(n)\bigg)\bm{\phi}(\operatorname{EOS})^\top.
\end{align*}
We then calculate several quantities important for subsequent derivations. We have
\begin{align*}
\mathbf{W}_{OV}\mathbf{X}^\top\mathbf{X}&=(\alpha_S\bar\mathbf{v}_{s(z_{1:T})}+\gamma_S\bar\mathbf{v})\bm{\phi}(s(z_{1:T}))^\top+(\alpha_R\bar\mathbf{v}_{r(z_{1:T})}+\gamma_R\bar\mathbf{v})\bm{\phi}(r(z_{1:T}))^\top\\
&\quad+\sum_{n\in\mathcal{N}}\gamma_N\bar\mathbf{v}\cdot C(n, z_{1:T})\bm{\phi}(n)^\top,
\end{align*}
and
\begin{align*}
\mathbf{X}\mathbf{X}^\top\mathbf{W}_{KQ}\bm{\phi}(\operatorname{EOS})&=\beta_S\bm{\phi}(s(z_{1:T}))+\beta_R\bm{\phi}(r(z_{1:T}))+\beta_N\sum_{n\in\mathcal{N}}C(n, z_{1:T})\bm{\phi}(n).
\end{align*}
Finally,
\begin{align*}
\mathbf{f}(\mathbf{X}; \bm{\Theta})&=\mathbf{W}_{OV}\mathbf{X}^\top\mathbf{X}\mathbf{W}_{KQ}\bm{\phi}(\operatorname{EOS})\\
&=\beta_S(\alpha_S\bar\mathbf{v}_{s(z_{1:T})}+\gamma_S\bar\mathbf{v})+\beta_R(\alpha_R\bar\mathbf{v}_{r(z_{1:T})}+\gamma_R\bar\mathbf{v})+\beta_N\sum_{n\in\mathcal{N}}C(n, z_{1:T})\cdot\gamma_N\bar\mathbf{v}\\
&=\beta_S(\alpha_S\bar\mathbf{v}_{s(z_{1:T})}+\gamma_S\bar\mathbf{v})+\beta_R(\alpha_R\bar\mathbf{v}_{r(z_{1:T})}+\gamma_R\bar\mathbf{v})+\beta_N\gamma_N(T-3)\bar\mathbf{v},
\end{align*}
where the last equality holds because there are exactly $T-3$ noise tokens in any sequence $z_{1:T}$.

\noindent\textbf{Calculation of $\widehat p_{\bm{\Theta}}(a|z_{1:T})$.}
For any $(s', r')$, the logits corresponding to $a_*(s', r')$ is
\begin{align*}
\bm{\phi}(a_*(s', r'))^\top\mathbf{f}(\mathbf{X}; \bm{\Theta})&=\alpha_S\beta_S\cdot\bar\mathbf{v}_{s(z_{1:T})}^\top\bm{\phi}(a_*(s', r'))+\alpha_R\beta_R\cdot\bar\mathbf{v}_{r(z_{1:T})}^\top\bm{\phi}(a_*(s', r'))\\
&\quad+\underbrace{(\beta_S\gamma_S+\beta_R\gamma_R+(T-3)\beta_N\gamma_N)\bar\mathbf{v}^\top\bm{\phi}(a_*(s', r'))}_{:=C}\\
&=\frac{\alpha_S\beta_S}{\sqrt R}\cdot\delta_{s', s(z_{1:T})}+\frac{\alpha_R\beta_R}{\sqrt S}\cdot\delta_{r', r(z_{1:T})}+C,
\end{align*}
where the last equality holds due to Lemma~\ref{lemma:barv_properties}(a). Since $C$ is a constant for all $(s', r')$, we omit $C$ in subsequent calculations. The normalizing constant is
\begin{align*}
Z&=\sum_{s'\in{\mathcal{S}}}\sum_{r'\in\mathcal{R}}\exp(\alpha_S\beta_S/\sqrt R\cdot\delta_{s', s(z_{1:T})}+\alpha_R\beta_R/\sqrt S\cdot\delta_{r', r(z_{1:T})})\\
&=\bigg[\sum_{s'\in{\mathcal{S}}}\exp(\alpha_S\beta_S/\sqrt R\cdot\delta_{s', s(z_{1:T})})\bigg]\cdot\bigg[\sum_{r\in\mathcal{R}}\exp(\alpha_R\beta_R/\sqrt S\cdot\delta_{r', r(z_{1:T})})\bigg]\\
&=(S-1+e^{\alpha_S\beta_S/\sqrt R})(R-1+e^{\alpha_R\beta_R/\sqrt S}).
\end{align*}
Therefore, the model output satisfies
\begin{align*}
p(a_*(s', r')|z_{1:T})=\frac{\exp(\bm{\phi}(a_*(s', r'))^\top\mathbf{f}(\mathbf{X}; \bm{\Theta}))}{Z}=\frac{e^{\alpha_S\beta_S/\sqrt R\cdot\delta_{s', s(z_{1:T})}}}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{e^{\alpha_R\beta_R/\sqrt S\cdot\delta_{r', r(z_{1:T})}}}{R-1+e^{\alpha_R\beta_R/\sqrt S}}.
\end{align*}
\noindent\textbf{Calculation of $-\nabla_{\mathbf{W}_{OV}}L(\bm{\Theta})$.} We first calculate the stochastic gradients $-\nabla_{\mathbf{W}_{OV}}\ell(z_{1:T+1}; \bm{\Theta})$ as given in \eqref{eq:grad_ell_WOV}. We firstly have
\begin{align*}
&\sum_{a\in\mathcal{A}}\widehat p_{\bm{\Theta}}(a|z_{1:T})[\bm{\phi}(z_{T+1})-\bm{\phi}(a)]=\sum_{s'\in{\mathcal{S}}}\sum_{r'\in\mathcal{R}}\widehat p_{\bm{\Theta}}(a_*(s', r')|z_{1:T})[\bm{\phi}(z_{T+1})-\bm{\phi}(a_*(s', r'))]\\
&=\frac{1}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{1}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\cdot\sum_{s'\in{\mathcal{S}}}\sum_{r'\in\mathcal{R}}[\bm{\phi}(z_{T+1})-\bm{\phi}(a_*(s', r'))]\\
&\quad+\sum_{s'\in{\mathcal{S}}}\sum_{r'\in\mathcal{R}}\bigg(\widehat p_{\bm{\Theta}}(a_*(s', r')|z_{1:T})-\frac{1}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{1}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\bigg)\\
&\quad\cdot[\bm{\phi}(z_{T+1})-\bm{\phi}(a_*(s', r'))]\\
&=\frac{1}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{1}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\cdot\Big\{(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})\\
&\quad+(e^{\alpha_R\beta_R/\sqrt S}-1)\sum_{s'}[\bm{\phi}(z_{T+1})-\bm{\phi}(a_*(s', r(z_{1:T})))]\\
&\quad+(e^{\alpha_S\beta_S/\sqrt R}-1)\sum_{r'}[\bm{\phi}(z_{T+1})-\bm{\phi}(a_*(s(z_{1:T}), r'))]\Big\}\\
&=\frac{1}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{1}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\cdot[(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})\\
&\quad+(e^{\alpha_R\beta_R/\sqrt S}-1)(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})\\
&\quad+(e^{\alpha_S\beta_S/\sqrt R}-1)(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})]
\end{align*}
Therefore, the stochastic gradient w.r.t. $\mathbf{W}_{OV}$ satisfies
\begin{align}
&-\nabla  _{\mathbf{W}_{OV}}\ell(z_{1:T+1}; \bm{\Theta})=\bigg\{\sum_{a\in\mathcal{A}}\widehat p_{\bm{\Theta}}(a|z_{1:T})[\bm{\phi}(z_{T+1})-\bm{\phi}(a)]\bigg\}\bm{\phi}(\operatorname{EOS})^\top\mathbf{W}_{KQ}^\top\mathbf{X}^\top\mathbf{X}\nonumber\\
&=\frac{1}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{1}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\cdot[(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})\nonumber\\
&\quad+(e^{\alpha_R\beta_R/\sqrt S}-1)(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})\nonumber\\
&\quad+(e^{\alpha_S\beta_S/\sqrt R}-1)(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})]\nonumber\\
&\quad\cdot\bigg[\beta_S\bm{\phi}(s(z_{1:T}))+\beta_R\bm{\phi}(r(z_{1:T}))+\beta_N\sum_{n\in\mathcal{N}}C(n, z_{1:T})\bm{\phi}(n)\bigg]^\top.\label{eq:sgrad_WOV_manifold}
\end{align}
We then take the expectations over $z_{1:T+1}$ to obtain the negative gradient $-\nabla_{\mathbf{W}_{OV}}L(\bm{\Theta})$. We consider the three terms in the last bracket separately, and we first calculate the expectation of the noise term: for each $n\in\mathcal{N}$, the term $C(n, z_{1:T})\cdot\bm{\phi}(n)$ is independent with the tokens $s(z_{1:T})$, $r(z_{1:T})$ and $z_{T+1}$. Therefore,
\begin{align*}
&\mathbb{E}_{z_{1:T+1}}\big[(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})(C(n, z_{1:T})\cdot\bm{\phi}(n))^\top\big]\\
&=\mathbb{E}_{z_{1:T+1}}[SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v}]\cdot\mathbb{E}[C(n, z_{1:T})\cdot\bm{\phi}(n)^\top]={\bm 0};\\
&\mathbb{E}_{z_{1:T+1}}\big[(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})(C(n, z_{1:T})\cdot\bm{\phi}(n))^\top\big]\\
&=\mathbb{E}_{z_{1:T+1}}[S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})}]\cdot\mathbb{E}[C(n, z_{1:T})\cdot\bm{\phi}(n)^\top]={\bm 0};\\
&\mathbb{E}_{z_{1:T+1}}\big[(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})(C(n, z_{1:T})\cdot\bm{\phi}(n))^\top\big]\\
&=\mathbb{E}_{z_{1:T+1}}[R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})}]\cdot\mathbb{E}[C(n, z_{1:T})\cdot\bm{\phi}(n)^\top]={\bm 0}.
\end{align*}
Therefore,
\begin{align}
&\mathbb{E}_{z_{1:T}}\bigg[\big((SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})+(e^{\alpha_R\beta_R/\sqrt S}-1)(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})\nonumber\\
&\quad+(e^{\alpha_S\beta_S/\sqrt R}-1)(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})\big)\bigg(\sum_{n\in\mathcal{N}}C(n, z_{1:T})\bm{\phi}(n)\bigg)^\top\bigg]={\bm 0}.\label{eq:ee_noise_ov}
\end{align}
We then calculate the term corresponding to $\bm{\phi}(s(z_{1:T}))$. Note that
\begin{align*}
&\mathbb{E}_{z_{1:T+1}}[(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})\bm{\phi}(s(z_{1:T}))^\top]\\
&=\frac1{SR}\sum_{s\in{\mathcal{S}}}\bigg\{\sum_{r\in\mathcal{R}}[SR\cdot\bm{\phi}(a_*(s, r))-\sqrt{SR}\cdot\bar\mathbf{v}]\bigg\}\bm{\phi}(s)^\top\\
&=\frac1{SR}\sum_{s\in{\mathcal{S}}}(SR\cdot\sqrt R\cdot\bar\mathbf{v}_s-R\cdot\sqrt{SR}\cdot\bar\mathbf{v})\bm{\phi}(s)^\top=\sqrt R\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}(s)^\top;\\
&\mathbb{E}_{z_{1:T+1}}[(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})\bm{\phi}(s)^\top]\\
&=\frac{1}{SR}\sum_{s\in{\mathcal{S}}}\bigg\{\sum_{r\in\mathcal{R}}[S\cdot\bm{\phi}(a_*(s, r))-\sqrt S\cdot\bar\mathbf{v}_r]\bigg\}\bm{\phi}(s)^\top\\
&=\frac1{SR}\sum_{s\in{\mathcal{S}}}(S\cdot\sqrt R\cdot\bar\mathbf{v}_s-\sqrt R\cdot\sqrt S\cdot\bar\mathbf{v})\bm{\phi}(s)^\top=\frac1{\sqrt R}\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}(s)^\top;\\
&\mathbb{E}_{z_{1:T+1}}[(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})\bm{\phi}(s)^\top]\\
&=\frac1{SR}\sum_{s\in{\mathcal{S}}}\bigg\{\sum_{r\in\mathcal{R}}[R\cdot\bm{\phi}(a_*(s, r))-\sqrt R\cdot\bar\mathbf{v}_{s}]\bigg\}\bm{\phi}(s)^\top\\
&=\frac1{SR}\sum_{s\in{\mathcal{S}}}(R\cdot\sqrt R\cdot\bar\mathbf{v}_s-R\cdot\sqrt R\cdot\bar\mathbf{v}_s)\bm{\phi}(s)^\top={\bm 0}.
\end{align*}
Combining all the above, we have
\begin{align}
&\mathbb{E}_{z_{1:T+1}}\big\{\big[(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})+(e^{\alpha_R\beta_R/\sqrt S}-1)(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})\nonumber\\
&\quad+(e^{\alpha_S\beta_S/\sqrt R}-1)(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})\big]\bm{\phi}(s(z_{1:T}))^\top\big\}\nonumber\\
&=\Big(\sqrt{R}+\frac{e^{\alpha_R\beta_R/\sqrt S}-1}{\sqrt R}\Big)\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}(s)^\top=\frac{R-1+e^{\alpha_R\beta_R/\sqrt S}}{\sqrt R}\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}(s)^\top.\label{eq:ee_s_ov}
\end{align}
Similarly, for the term corresponding to $\bm{\phi}(r(z_{1:T}))$, we have
\begin{align}
&\mathbb{E}_{z_{1:T}}\big\{\big[(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})+(e^{\alpha_R\beta_R/\sqrt S}-1)(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})\nonumber\\
&\quad+(e^{\alpha_S\beta_S/\sqrt R}-1)(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})\big]\bm{\phi}(r(z_{1:T}))^\top\big\}\nonumber\\
&=\frac{S-1+e^{\alpha_S\beta_S/\sqrt R}}{\sqrt{S}}\sum_{r\in\mathcal{R}}\widetilde\mathbf{v}_r\bm{\phi}(r)^\top.\label{eq:ee_r_ov}
\end{align}
Plugging \eqref{eq:ee_noise_ov}, \eqref{eq:ee_s_ov}, and \eqref{eq:ee_r_ov} into the expectation of \eqref{eq:sgrad_WOV_manifold}, we have
\begin{align*}
-\nabla_{\mathbf{W}_{OV}}L(\bm{\Theta})=\frac{\beta_S/\sqrt R}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}(s)^\top+\frac{\beta_R/\sqrt S}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\sum_{r\in\mathcal{R}}\widetilde\mathbf{v}_r\bm{\phi}(r)^\top.
\end{align*}

\noindent\textbf{Calculation of $-\nabla_{\mathbf{W}_{KQ}}L(\bm{\Theta})$.}
We also start from the stochastic gradient. Similar to the analysis of the noise-related terms in $-\nabla_{\mathbf{W}_{OV}}\ell(z_{1:T+1}; \bm{\Theta})$, the noise-related term in $\mathbf{X}^\top\mathbf{X}\mathbf{W}_{OV}^\top$, i.e.,
\begin{align*}
\sum_{n\in\mathcal{N}}C(n, z_{1:T})\cdot\bm{\phi}(n)\bar\mathbf{v}^\top
\end{align*}
is independent from $s(z_{1:T})$, $r(z_{1:T})$, and $z_{T+1}$, and thus has zero contribution in the expectation. We also notice that due to Lemma~\ref{lemma:barv_properties}(a, b),
\begin{align*}
&\bar\mathbf{v}^\top(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})=SR\cdot\frac1{\sqrt{SR}}-\sqrt{SR}\cdot1=0\\
&\bar\mathbf{v}^\top(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})=S\cdot\frac1{\sqrt{SR}}-\sqrt S\cdot\frac1{\sqrt R}=0\\
&\bar\mathbf{v}^\top(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})=R\cdot\frac1{\sqrt{SR}}-\sqrt R\cdot\frac1{\sqrt{S}}=0.
\end{align*}
so we also drop the terms related to $\bar\mathbf{v}$ in $\mathbf{X}^\top\mathbf{X}\mathbf{W}_{OV}^\top$. Therefore,
\begin{align*}
&-\nabla_{\mathbf{W}_{KQ}}L(\bm{\Theta})=\mathbb{E}_{z_{1:T+1}}\Big\{(\alpha_S\bm{\phi}(s(z_{1:T}))\bar\mathbf{v}_{s(z_{1:T})}^\top+\alpha_R\bm{\phi}(r(z_{1:T}))\bar\mathbf{v}_{r(z_{1:T})}^\top)\\
&\quad\cdot\frac{1}{S-1+e^{\alpha_S\beta_S/\sqrt R}}\cdot\frac{1}{R-1+e^{\alpha_R\beta_R/\sqrt S}}\cdot[(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})\\
&\quad+(e^{\alpha_R\beta_R/\sqrt S}-1)(S\cdot\bm{\phi}(z_{T+1})-\sqrt S\cdot\bar\mathbf{v}_{r(z_{1:T})})\\
&\quad+(e^{\alpha_S\beta_S/\sqrt R}-1)(R\cdot\bm{\phi}(z_{T+1})-\sqrt R\cdot\bar\mathbf{v}_{s(z_{1:T})})]\cdot\bm{\phi}(\operatorname{EOS})^\top\Big\}.
\end{align*}
By Lemma~\ref{lemma:barv_properties}(a, b), we have
\begin{align*}
&\bar\mathbf{v}_{s(z_{1:T})}^\top(SR\cdot\bm{\phi}(z_{T+1})-\sqrt{SR}\cdot\bar\mathbf{v})=SR\cdot\frac1{\sqrt R}-\sqrt{SR}\cdot\frac1{\sqrt S}=(S-1)\sqrt R;\\

We first consider the ODE of $\alpha_S$ and $\beta_S$, and the ODE of $\alpha_R$ and $\beta_R$ can be analyzed similarly.
The derivative of the margin $q_S$ satisfies
\begin{align*}
\frac{{\,\mathrm{d}} q_S}{{\,\mathrm{d}} t}=\frac{\alpha_S}{\sqrt R}\cdot\frac{{\,\mathrm{d}}\beta_S}{{\,\mathrm{d}} t}+\frac{{\,\mathrm{d}}\alpha_S}{{\,\mathrm{d}} t}\cdot\frac{\beta_S}{\sqrt R}=\frac{(S-1)\alpha_S^2/S+\beta_S^2}{R(S-1+e^{\alpha_S\beta_S/\sqrt R})}=\frac{\alpha_S^2/S+\beta_S^2/(S-1)}{R(1+e^{q_S})}.
\end{align*}
We aim to express the right-hand side as a function of $q_S$. Notice that $\alpha_S^2/S-\beta_S^2/(S-1):=C_S$ is a constant because
\begin{align*}
\frac{{\,\mathrm{d}}[\alpha_S^2/S-\beta_S^2/(S-1)]}{{\,\mathrm{d}} t}=2\Big[\frac{{\,\mathrm{d}}\alpha_S}{{\,\mathrm{d}} t}\cdot\frac{\alpha_S}S-\frac{{\,\mathrm{d}}\beta_S}{{\,\mathrm{d}} t}\cdot\frac{\beta_S}{S-1}\Big]=0.
\end{align*}
We also notice that
\begin{align*}
\alpha_S^2/S+\beta_S^2/(S-1)&=\sqrt{[\alpha_S^2/S-\beta_S^2/(S-1)]^2+4\alpha_S^2\beta_S^2/(S(S-1))}\\
&=\sqrt{C_S^2+4R[q_S+\log(S-1)]^2/(S(S-1))}.
\end{align*}
Therefore, the derivative of $q_S$ can be written as the following expression with only $q_S$ and the constant $C_S$:
\begin{align*}
\frac{{\,\mathrm{d}} q_S}{{\,\mathrm{d}} t}=\frac{\sqrt{C_S^2/R^2+4[q_S+\log(S-1)]^2/(SR(S-1))}}{1+e^{q_S}}.
\end{align*}
Separating variables and taking the integral, we have
\begin{align*}
T_S^\delta&=\int_{\alpha_0\beta_0/\sqrt R-\log(S-1)}^{\log(\delta^{-1}-1)}\frac{1+e^{q_S}}{\sqrt{C_S^2/R^2+4[q_S+\log(S-1)]^2/(SR(S-1))}}{\,\mathrm{d}} q_S\\
&=\underbrace{\int_{\alpha_0\beta_0/\sqrt R-\log(S-1)}^{\log(\delta^{-1}-1)}\frac{{\,\mathrm{d}} q_S}{\sqrt{C_S^2/R^2+4[q_S+\log(S-1)]^2/(SR(S-1))}}}_{I_{S, 1}}\\
&\quad+\underbrace{\int_{\alpha_0\beta_0/\sqrt R-\log(S-1)}^{1-\log(S-1)}\frac{e^{q_S}{\,\mathrm{d}} q_S}{\sqrt{C_S^2/R^2+4[q_S+\log(S-1)]^2/(SR(S-1))}}}_{I_{S, 2}}\\
&\quad+\underbrace{\int_{1-\log(S-1)}^{\log(\delta^{-1}-1)}\frac{e^{q_S}{\,\mathrm{d}} q_S}{\sqrt{C_S^2/R^2+4[q_S+\log(S-1)]^2/(SR(S-1))}}}_{I_{S, 3}}
\end{align*}
For the term $I_{S, 1}$, we directly calculate the integral:
\begin{align*}
I_{S, 1}&=\frac{\sqrt{SR(S-1)}}{2}\bigg[\log\bigg(\log(\delta^{-1}-1)(S-1)+\sqrt{[\log(\delta^{-1}-1)(S-1)]^2+\frac{S(S-1)C_S^2}{4R}}\bigg)\\
&\quad-\log\bigg(\frac{\alpha_0\beta_0}{\sqrt R}+\sqrt{\frac{\alpha_0^2\beta_0^2}{R}+\frac{S(S-1)C_S^2}{4R}}\bigg)\bigg].
\end{align*}
Note that $S(S-1)C_S^2/(4R)$ is a lower-order term because
\begin{align}
\frac{S(S-1)C_S^2}{4R}&=\frac{S(S-1)}{4R}[\alpha_0^2/S-\beta_0^2/(S-1)]^2\nonumber\\
&\le\frac{S(S-1)}{4R}\max\Big\{\frac{\alpha_0^4}{S^2}, \frac{\beta_0^4}{(S-1)^2}\Big\}\nonumber\\
&\le\frac{S}{4R(S-1)}\le1,\label{eq:CS_bound}
\end{align}
where the first inequality holds because $|A-B|\le\max\{A, B\}$, the second inequality holds due to the assumption $\alpha_0, \beta_0\le1$, and the last inequality holds because $S, R\ge2$. We also have
\begin{align*}
\frac{\alpha_0\beta_0}{\sqrt R}+\sqrt{\frac{\alpha_0^2\beta_0^2}{R}+\frac{S(S-1)C_S^2}{4R}}=\frac1{2\sqrt R}\Big[\Big(\frac{S-1}{S}\Big)^{1/4}\alpha_0+\Big(\frac{S}{S-1}\Big)^{1/4}\beta_0\Big]^2\simeq\frac{(\alpha_0+\beta_0)^2}{2\sqrt R}
\end{align*}
Therefore, $I_{S, 1}$ is approximated as
\begin{align}\label{eq:IS1_final}
I_{S, 1}\simeq\sqrt{SR(S-1)}\log\frac{\sqrt R\log(\delta^{-1}-1)(S-1)}{\alpha_0+\beta_0}.
\end{align}

We then show that $I_{S, 2}$ is controlled by $I_{S, 1}$, i.e.,
\begin{align}\label{eq:IS2<IS1}
I_{S, 2}\lesssim I_{S, 1}.
\end{align}
It suffices to show that $e^{q_S}\lesssim1$, which holds because
\begin{align*}
e^{q_S}\le e/(S-1)\le e,
\end{align*}
where the first inequality holds because $q_S\le1-\log(S-1)$, and the second inequality holds because $S\ge2$.

For the term $I_{S, 3}$, we use the bound for $C_S$ in \eqref{eq:CS_bound} and have
\begin{align}\label{eq:IS3}
I_{S, 3}&=\frac{\sqrt{SR(S-1)}}2\int_{1-\log(S-1)}^{\log(\delta^{-1}-1)}\frac{e^{q_S}{\,\mathrm{d}} q_S}{\sqrt{\frac{S(S-1)C_S^2}{4R}+[q_S+\log(S-1)]^2}}\nonumber\\
&\simeq\sqrt{SR(S-1)}\int_{1-\log(S-1)}^{\log(\delta^{-1}-1)}\frac{e^{q_S}{\,\mathrm{d}} q_S}{q_S+\log(S-1)}\nonumber\\

&\simeq\sqrt{SR(S-1)}\cdot\frac{\delta^{-1}-1}{\log(\delta^{-1}-1)(S-1)},
\end{align}
where the first equality holds because $\frac{S(S-1)C_S^2}{4R}\lesssim1$ but $q_S+\log(S-1)\ge1$, and the last equality holds due to Lemma~\ref{lemma:Ei_estimate}. Combining \eqref{eq:IS1_final}, \eqref{eq:IS2<IS1}, and \eqref{eq:IS3}, we have
\begin{align*}
T_S^\delta\simeq\sqrt{SR(S-1)}\bigg[\log\frac{\sqrt R\log(S-1)}{\alpha_0+\beta_0}+\frac{\delta^{-1}-1}{\log(\delta^{-1}-1)(S-1)}\bigg].
\end{align*}
We similarly have
\begin{align*}

\end{align*}

<!-- proof-6: body_md -->
\noindent\textbf{Characterization of $T_S^\delta$.}
We notice that the derivative of $\beta_S$ satisfies
\begin{align*}
\frac{{\,\mathrm{d}}\beta_S}{{\,\mathrm{d}} t}=\frac{(1+e^{q_R})/\sqrt S}{\sqrt{(1+e^{q_S})^2+(1+e^{q_R})^2}}\le1/\sqrt S,
\end{align*}
so combining the above with the lower bound of $\beta_S$ in Lemma~\ref{lemma:betaS_lower_betaR_upper}, we have
\begin{align*}
\frac{t}{2\sqrt S}+\frac{1+\sqrt{R/S}}{2}\beta_0\le\beta_S(t)\le\frac{t}{\sqrt S}+\beta_0.
\end{align*}
Since $0<\sqrt{R/S}\le1$, we have $\beta_S(t)\simeq t/\sqrt S+\beta_0$.
According to the definition of $T_S^\delta$, we have
\begin{align*}
\sqrt R\log(\delta^{-1}-1)(S-1)=\alpha_S(T_S^\delta)\cdot\beta_S(T_S^\delta)\simeq(T_S^\delta+\alpha_0)(T_S^\delta/\sqrt S+\beta_0).
\end{align*}
Due to the assumptions $\alpha_0\le(SR)^{1/4}$ and $\beta_0\le(R/S)^{1/4}$, we have
\begin{align*}
T_S^\delta\simeq(SR)^{1/4}\sqrt{\log(\delta^{-1}-1)(S-1)}.
\end{align*}

\noindent\textbf{Characterization of $T_R^\delta$.} We develop the characterization of $T_R^\delta$ in 5 steps.

\textit{Step 1: Universal upper bound of $T_R^\delta$.}
Since $\beta_R$ is increasing, we have for any time $t\ge T_R^{0.5}$,
\begin{align*}
\beta_R(t)\ge\beta_R(T_R^{0.5})=\frac{\sqrt S\log(R-1)}{\alpha_R(T_R^{0.5})}.
\end{align*}
Combining the above with the definition of $T_R^\delta$, we have
\begin{align*}
\log(\delta^{-1}-1)(R-1)&=\frac{\alpha_R(T_R^\delta)\cdot\beta_R(T_R^\delta)}{\sqrt S}\\
&\ge\frac{\alpha_R(T_R^\delta)}{\alpha_R(T_R^{0.5})}\cdot\log(R-1)\\
&\simeq\frac{T_R^\delta}{T_R^{0.5}}\cdot\log(R-1)\\
&\simeq(SR)^{-1/4}\sqrt{\log(R-1)}\cdot T_R^\delta,
\end{align*}
where the third line holds because $\alpha_0=(SR)^{1/4}\lesssim T_R^{0.5}\le T_R^\delta$, and the last line holds due to Lemma~\ref{lemma:T0.5}. We thus have
\begin{align}\label{eq:TS_universal_upper_bound}
T_R^\delta\lesssim(SR)^{1/4}\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log(R-1)}}.
\end{align}

\textit{Step 2: The above upper bound is tight when $T_R^\delta<T_S^{0.5}$.}
The above upper bound relies on using $\beta_R(T_R^{0.5})$ to lower bound $\beta_R(T_R^\delta)$. We now show that when $T_R^\delta<T_S^{0.5}$, we have $\beta_R(T_R^\delta)\simeq\beta_R(T_R^{0.5})$, so the above upper bound is tight. Note that when $T_R^{0.5}\le t\le T_S^{0.5}$, we have $q_S\le0\le q_R$, and
\begin{align}\label{eq:dbetaR_dt_upper}
\frac{{\,\mathrm{d}}\beta_R}{{\,\mathrm{d}} t}=\frac{1/\sqrt R}{\sqrt{1+(\frac{1+e^{q_R}}{1+e^{q_S}})^2}}\le\frac{1/\sqrt R}{\sqrt{1+(\frac{1+e^{q_R}}{2})^2}}\le\frac{2}{\sqrt R e^{q_R}},
\end{align}
where the first inequality holds because $q_S\ge1$, and the second inequality holds because $1+(\frac{1+e^{q_R}}{2})^2\ge\frac{e^{2q_R}}{4}$.
We then derive a lower bound for $q_R$ when $T_R^{0.5}\le t\le T_S^{0.5}$:
\begin{align}
q_R(t)&=\frac{\alpha_R(t)\beta_R(t)}{\sqrt S}-\log(R-1)\nonumber\\
&=\frac{\alpha_R(t)\beta_R(t)-\alpha_R(T_R^{0.5})\beta_R(T_R^{0.5})}{\sqrt S}\nonumber\\
&\ge\frac{[\alpha_R(t)-\alpha_R(T_R^{0.5})]\beta_R(T_R^{0.5})}{\sqrt S}\nonumber\\
&=(t-T_R^{0.5})\cdot\frac{\log(R-1)}{\alpha_R(T_R^{0.5})},\label{eq:qR_lower}
\end{align}
where the second equality holds due to the definition of $T_R^{0.5}$, and inequality holds because $\beta_R(t)\ge\beta_R(T_R^{0.5})$ for $t\ge T_R^{0.5}$, and the last equality holds due to the definition of $T_R^{0.5}$ and $\alpha_R(t)=t+\alpha_0$. Plugging \eqref{eq:qR_lower} into \eqref{eq:dbetaR_dt_upper}, we have
\begin{align*}
\beta_R(T_R^\delta)-\beta_R(T_R^{0.5})&\le\int_{T_R^{0.5}}^{T_R^\delta}\frac{2{\,\mathrm{d}} t}{\sqrt R e^{\frac{\log(R-1)}{\alpha_R(T_R^{0.5})}\cdot(t-T_R^{0.5})}}\\
&\le\int_{T_R^{0.5}}^{\infty}\frac{2{\,\mathrm{d}} t}{\sqrt R e^{\frac{\log(R-1)}{\alpha_R(T_R^{0.5})}\cdot(t-T_R^{0.5})}}\\
&=\frac{2\alpha_R(T_R^{0.5})}{\sqrt R\log(R-1)}\\
&\frac{\simeq(S/R)^{1/4}}{\sqrt{\log(R-1)}}
\end{align*}
where the last equality holds due to Lemma~\ref{lemma:T0.5}. Since $\beta_R(T_R^{0.5})\simeq(S/R)^{1/4}\sqrt{\log(R-1)}$ according to Lemma~\ref{lemma:T0.5}, we have $\beta(T_R^\delta)\simeq\beta_R(T_R^{0.5})$, and the derivation of the upper bound is tight, i.e.,
\begin{align*}
T_R^\delta\simeq(SR)^{1/4}\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log R-1}}.
\end{align*}

\textit{Step 3: Lower bound of $T_R^\delta$ when $T_R^\delta\ge T_S^{0.5}$.}
Due to the definition of $T_R^\delta$, we have
\begin{align*}
\log(\delta^{-1}-1)(R-1)=\frac{\alpha_R(T_R^\delta)\beta_R(T_R^\delta)}{\sqrt S}\le\frac{T_R^\delta+\alpha_0}{\sqrt S}\cdot\Big(\frac{T_R^\delta}{\sqrt{2R}}+\frac{1+\sqrt{S/R}}{2}\beta_0\Big),
\end{align*}
where the last equality holds due to Lemma~\ref{lemma:betaS_lower_betaR_upper}. Since $\alpha_0\le(SR)^{1/4}$ and $\beta_0\le(R/S)^{1/4}$, we have
\begin{align*}
T_R^\delta&\gtrsim\max\{T_S^{0.5}, (SR)^{1/4}\sqrt{\log(\delta^{-1}-1)(R-1)}\}\\
&\simeq(SR)^{1/4}\sqrt{\max\{\log(S-1), \log(\delta^{-1}-1)+\log(R-1)\}}\\
&\simeq(SR)^{1/4}\sqrt{\log(\delta^{-1}-1)(S-1)}.
\end{align*}
where the first equality holds due to Lemma~\ref{lemma:T0.5}, and the last equality holds because $\max\{A, B\}\simeq A+B$ and $S\ge R$.

\textit{Step 4: Upper bound of $T_R^\delta$ when $T_R^\delta\ge T_S^{0.5}$.} In Step 3, we derived the lower bound of $T_R^\delta$ which has the same order as $T_S^\delta$. If $T_S^\delta\ge T_R^\delta$, then the bound is tight, so we focus on the case where $T_S^\delta<T_R^\delta$. We study the dynamics of $m(t):=\min\{q_S(t), q_R(t)\}$ for $t\ge\tau:=\max\{T_S^{0.5}, T_R^{0.5}\}$. Due to the definition and the assumption $T_S^\delta<T_R^\delta$, we immediately have
\begin{align}\label{eq:m(tau)=0}
m(\tau)=0,\quad m(T_R^\delta)=q_R(T_R^\delta)=\log(\delta^{-1}-1).
\end{align}
We then calculate the derivative of $m(t)$, and it suffices to consider the derivatives of $q_S$ and $q_R$. Suppose that $q_S(t)\le q_R(t)$ for a given $t$, then we have
\begin{align*}
\frac{{\,\mathrm{d}}\beta_S(t)}{{\,\mathrm{d}} t}=\frac{(1+e^{q_R})/\sqrt S}{\sqrt{(1+e^{q_S})^2+(1+e^{q_R})^2}}\ge\frac1{\sqrt{2S}}.
\end{align*}
The derivative of $m(t)$ thus satisfies
\begin{align*}
\frac{{\,\mathrm{d}} m(t)}{{\,\mathrm{d}} t}&=\frac{{\,\mathrm{d}} q_S(t)}{{\,\mathrm{d}} t}=\frac{{\,\mathrm{d}}\alpha_S(t)}{{\,\mathrm{d}} t}\cdot\frac{\beta_S(t)}{\sqrt R}+\frac{\alpha_S(t)}{\sqrt R}\cdot\frac{{\,\mathrm{d}}\beta_R(t)}{{\,\mathrm{d}} t}\\
&\ge\frac{\beta_S(t)}{\sqrt R}+\frac{t+t_0}{\sqrt{2RS}}\\
&=\frac{q_S(t)+\log(S-1)}{t+t_0}+\frac{t+t_0}{\sqrt{2RS}},
\end{align*}
where the last equality holds due to the definition of $q_S$. Similarly, when $q_S(t)\ge q_R(t)$, we have
\begin{align*}
\frac{{\,\mathrm{d}} m(t)}{{\,\mathrm{d}} t}\ge\frac{q_R(t)+\log(R-1)}{t+t_0}+\frac{t+t_0}{\sqrt{2RS}}.
\end{align*}
Combining the two cases, noting that $S\ge R$, we have
\begin{align*}
\frac{{\,\mathrm{d}} m(t)}{{\,\mathrm{d}} t}\ge\frac{m(t)+\log(R-1)}{t+t_0}+\frac{t+t_0}{\sqrt{2RS}}.
\end{align*}
We thus have
\begin{align*}
\frac{{\,\mathrm{d}}}{{\,\mathrm{d}} t}\bigg(\frac{m(t)+\log(R-1)}{t+\alpha_0}\bigg)&=\frac{1}{t+\alpha_0}\cdot\frac{{\,\mathrm{d}} m(t)}{{\,\mathrm{d}} t}-\frac{m(t)+\log(R-1)}{(t+\alpha_0)^2}\\
&\ge\frac{m(t)+\log(R-1)}{(t+\alpha_0)^2}+\frac1{\sqrt{2SR}}-\frac{m(t)+\log(R-1)}{(t+\alpha_0)^2}\\
&=\frac1{\sqrt{2SR}}.
\end{align*}
Taking the integral from $t=\tau$ to $t=T_R^\delta$, we have
\begin{align*}
\frac{T_R^\delta-\tau}{\sqrt{2SR}}&\le\frac{m(T_R^\delta)+\log(R-1)}{T_R^\delta+\alpha_0}-\frac{m(\tau)+\log(R-1)}{\tau+\alpha_0}\\
&=\frac{\log(\delta^{-1}-1)(R-1)}{T_R^\delta+\alpha_0}-\frac{\log(R-1)}{\tau+\alpha_0},
\end{align*}
where the equality holds due to \eqref{eq:m(tau)=0}. Solving the inequality, we have
\begin{align*}
T_R^\delta&\le\frac12\Big(\tau-\alpha_0-\frac{\sqrt{2SR}\log(R-1)}{\tau+\alpha_0}\Big)\\
&\quad+\sqrt{\frac14\Big(\tau+\alpha_0+\frac{\sqrt{2SR}\log(R-1)}{\tau+\alpha_0}\Big)^2+\sqrt{2SR}\log(\delta^{-1}-1)}\\
&\le\frac12\Big(\tau-\alpha_0-\frac{\sqrt{2SR}\log(R-1)}{\tau+\alpha_0}\Big)\\
&\quad+\frac12\Big(\tau+\alpha_0+\frac{\sqrt{2SR}\log(\delta^{-1}-1)}{\tau+\alpha_0}\Big)+\sqrt{\sqrt{2SR}\log(\delta^{-1}-1)}\\
&=\tau+(2SR)^{1/4}\sqrt{\log(\delta^{-1}-1)}\\
&\simeq(SR)^{1/4}(\sqrt{\log(S-1)}+\sqrt{\log(\delta^{-1}-1)})\\
&\simeq(SR)^{1/4}\sqrt{\log(\delta^{-1}-1)(S-1)},
\end{align*}
where the second inequality holds because $\sqrt{A^2+B^2}\le A+B$, and the second equality holds due to Lemma~\ref{lemma:T0.5}.

\textit{Step 5: Combining all results together.} We have so far derived matching upper and lower bounds of $T_R^\delta$ under two regimes $T_R^\delta<T_S^{0.5}$ and $T_R^\delta\ge T_S^{0.5}$. However, these two regimes are given by conditions that are not directly related to $S$, $R$, and $\delta$. We now discuss when these two approximations apply. We first notice that Steps 1-4 indicate that
\begin{align*}
T_R^\delta\simeq(SR)^{1/4}\cdot\min\bigg\{\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log(R-1)}}, \sqrt{\log(\delta^{-1}-1)(S-1)}\bigg\}.
\end{align*}
When we have
\begin{align*}
\log(\delta^{-1}-1)\le c\cdot\sqrt{\log(S-1)\cdot\log(R-1)},
\end{align*}
then
\begin{align*}
\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log(R-1)}}&\lesssim\frac{\sqrt{\log(S-1)\cdot\log(R-1)}+\log(R-1)}{\sqrt{\log(R-1)}}\\
&=\sqrt{\log(S-1)}+\sqrt{\log(R-1)}\\
&\simeq\sqrt{\log(S-1)}\\
&\le\sqrt{\log(\delta^{-1}-1)(S-1)},
\end{align*}
where the third line holds because $S\ge R$. On the other hand, we aim to prove that 
\begin{align*}
\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log(R-1)}}\gtrsim\sqrt{\log(\delta^{-1}-1)(S-1)}
\end{align*}
under the condition of
\begin{align*}
\log(\delta^{-1}-1)\ge c\cdot\sqrt{\log(S-1)\cdot\log(R-1)}.
\end{align*}
Following the similar logic of the previous deduction, we have
\begin{align*}
\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log(R-1)}}&\gtrsim\sqrt{\log(S-1)}.
\end{align*}
Therefore, if $\log(\delta^{-1}-1)\le\log(S-1)$, then
\begin{align*}
\sqrt{\log(S-1)}\simeq\sqrt{\log(\delta^{-1}-1)(S-1)}.
\end{align*}
Otherwise, when $\log(\delta^{-1}-1)\ge\log(S-1)$, we have
\begin{align*}
\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log(R-1)}}\lesssim\frac{\log(\delta^{-1}-1)}{\sqrt{\log(R-1)}}\lesssim\frac{\log(\delta^{-1}-1)}{\sqrt{\log(\delta^{-1}-1)}}\simeq\sqrt{\log(\delta^{-1}-1)(S-1)}
\end{align*}
where the first two inequalities hold because $\log(R-1)\le\log(S-1)\lesssim\log(\delta^{-1}-1)$, and the last equality holds because $\log(S-1)\le\log(\delta^{-1}-1)$. Therefore, the final result is
\begin{align*}
T_R^\delta\simeq\begin{cases}
(SR)^{1/4}\frac{\log(\delta^{-1}-1)(R-1)}{\sqrt{\log(R-1)}} & \text{if}\quad\log(\delta^{-1}-1)\le c\cdot\sqrt{\log(S-1)\cdot\log(R-1)};\\
(SR)^{1/4}\sqrt{\log(\delta^{-1}-1)(S-1)} & \text{otherwise}.
\end{cases}
\end{align*}

Let $\mathbf{V}_S\in\mathbb{R}^{d\times S}$ and $\mathbf{V}_R\in\mathbb{R}^{d\times R}$ be the matrices formed by horizontally stacking $\{\widetilde\mathbf{v}_s\}_{s\in{\mathcal{S}}}$ and $\{\widetilde\mathbf{v}_r\}_{r\in\mathcal{R}}$, respectively. \end{align*}
Note that $(\bm{\Phi}_S, \bm{\Phi}_R)$ forms an orthonormal basis:
\begin{align}\label{eq:Phi_properties}

Similar to the previous case, the first column of $\mathbf{H}_S-{\bm 1}_S\mathbf{e}_1^\top/\sqrt S$ is zero, and all other entries are $\pm1/\sqrt S$. Considering the additional $1/\sqrt R$ factor on ${\bm 1}/\sqrt R$, we have
\begin{align*}
\mathop{\mathrm{sign}}\bigg(\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}_S^{\mathrm{O}}(s)^\top\bigg)=\sqrt{SR}\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}_S^{\mathrm{O}}(s)^\top.
\end{align*}

\noindent\textbf{Subject Hadamard, answer Hadamard.} In this case,
\begin{align*}
\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}_S^{\mathrm{O}}(s)^\top&=\sum_{s\in{\mathcal{S}}}[(\mathbf{H}_S\mathbf{e}_s-\mathbf{e}_1/\sqrt S)\otimes\mathbf{e}_1](\mathbf{H}_S\mathbf{e}_s)^\top\\

&=(\mathbf{H}_S^2-\mathbf{e}_1{\bm 1}_S^\top\mathbf{H}_S/\sqrt S)\otimes\mathbf{e}_1.
\end{align*}
Since
\begin{align*}
\mathbf{H}_S^2=[(\mathbf{H}_2/\sqrt 2)^2]^{\otimes p}=(\mathbf{I}_2)^{\otimes p}=\mathbf{I}_S,
\end{align*}
we have
\begin{align*}
\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}_S^{\mathrm{O}}(s)^\top=(\mathbf{I}_S-\mathbf{e}_1\mathbf{e}_1^\top)\otimes\mathbf{e}_1.
\end{align*}
Therefore,
\begin{align*}
\mathop{\mathrm{sign}}\bigg(\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}_S^{\mathrm{O}}(s)^\top\bigg)=\sum_{s\in{\mathcal{S}}}\widetilde\mathbf{v}_s\bm{\phi}_S^{\mathrm{O}}(s)^\top.
\end{align*}

We first show that $\sqrt S\beta_S\ge\sqrt R\beta_R$. If $S=R$, then $\beta_S\equiv\beta_R$. We now assume that $S>R$. At initialization, we have $\sqrt S\beta_0>\sqrt R\beta_0$. Suppose there exists time when $\sqrt S\beta_S\le\sqrt R\beta_R$, then define
\begin{align*}
t_0=\inf\{t: \sqrt S\beta_S(t)\le\sqrt R\beta_R\}.
\end{align*}
Note that the derivative of $\sqrt S\beta_S-\sqrt R\beta_R$ satisfies
\begin{align*}
\frac{{\,\mathrm{d}}(\sqrt S\beta_S-\sqrt R\beta_R)}{{\,\mathrm{d}} t}&=\frac{e^{q_S}}{\sqrt{(1+e^{q_S})^2+(1+e^{q_R})^2}}(e^{q_R-q_R}-1)\\
&=\frac{e^{q_S}}{\sqrt{(1+e^{q_S})^2+(1+e^{q_R})^2}}\Big(\frac{S-1}{R-1}e^{\frac{t+\alpha_0}{\sqrt{SR}}(\sqrt S\beta_S-\sqrt R\beta_R)}-1\Big)
\end{align*}
where the last equality holds due to the definition of $q_S$ and $q_R$ as well as $\alpha_S(t)=\alpha_R(t)=t+\alpha_0$. Since at $t=t_0$, we have $\sqrt S\beta_S(t_0)-\sqrt R\beta_R(t_0)=0$, the derivative at $t=t_0$ satisfies
\begin{align*}
\frac{{\,\mathrm{d}}(\sqrt S\beta_S-\sqrt R\beta_R)}{{\,\mathrm{d}} t}\bigg|_{t=t_0}=\frac{e^{q_S}}{\sqrt{(1+e^{q_S})^2+(1+e^{q_R})^2}}\cdot\frac{S-R}{R-1}>0.
\end{align*}
This means that there exists $\epsilon>0$ such that $\sqrt S\beta_S(t_0-\epsilon)-\sqrt R\beta_R(t_0-\epsilon)<0$, which contradicts against definition of $t_0$. Therefore, $\sqrt S\beta_S\ge\sqrt R\beta_R$ at any time, and we have
\begin{align}\label{eq:sum_bounds_2}
2\sqrt R\beta_R(t)\le\sqrt S\beta_S(t)+\sqrt R\beta_R(t)\le2\sqrt S\beta_S(t).
\end{align}

We now get back to the proof of the upper and lower bounds, and we notice that
\begin{align*}
\Big[\frac{{\,\mathrm{d}}(\sqrt S\beta_S)}{{\,\mathrm{d}} t}\Big]^2+\Big[\frac{{\,\mathrm{d}}(\sqrt R\beta_R)}{{\,\mathrm{d}} t}\Big]^2=1.
\end{align*}
Note that for positive numbers $A, B$, we have $\sqrt{A^2+B^2}\le A+B\le\sqrt{2(A^2+B^2)}$. Therefore,
\begin{align*}
1\le\frac{{\,\mathrm{d}}(\sqrt S\beta_S+\sqrt R\beta_R)}{{\,\mathrm{d}} t}\le\sqrt2.
\end{align*}
Taking the integral, we have
\begin{align}\label{eq:sum_bounds_1}
t+(\sqrt S+\sqrt R)\beta_0\le\sqrt S\beta_S(t)+\sqrt R\beta_R(t)\le\sqrt2t+(\sqrt S+\sqrt R)\beta_0.
\end{align}
Combining \eqref{eq:sum_bounds_1} and \eqref{eq:sum_bounds_2}, we have
\begin{align*}
\beta_S(t)&\ge\frac{t}{2\sqrt S}+\frac{1+\sqrt{R/S}}{2}\beta_0,\\
\beta_R(t)&\le\frac{t}{\sqrt{2R}}+\frac{1+\sqrt{S/R}}{2}\beta_0.
\end{align*}

\label{lemma:msign_rotation}
Let $\mathbf{M}\in\mathbb{R}^{m\times n}$ be any matrix, and $\mathbf{Q}_1\in\mathbb{R}^{m\times m}$, $\mathbf{Q}_2\in\mathbb{R}^{n\times n}$ be any orthogonal matrices. Then
\begin{align*}
\operatorname{msign}(\mathbf{Q}_1\mathbf{M}\mathbf{Q}_2^\top)=\mathbf{Q}_1\cdot\operatorname{msign}(\mathbf{M})\cdot\mathbf{Q}_2^\top.
\end{align*}
<!-- end lemma-7 -->
<!-- proof-12: body_md -->
Let the SVD of $\mathbf{M}$ be $\mathbf{M}=\mathbf{U}\bm{\Sigma}\mathbf{V}^\top$. Then 
\begin{align*}
\mathbf{Q}_1\mathbf{M}\mathbf{Q}_2^\top=(\mathbf{Q}_1\mathbf{U})\cdot\bm{\Sigma}\cdot(\mathbf{Q}_2\mathbf{V})^\top.
\end{align*}
Note that $\mathbf{Q}_1\mathbf{U}$ and $\mathbf{Q}_2\mathbf{V}$ are left and right singular vectors because
\begin{align*}
(\mathbf{Q}_1\mathbf{U})^\top(\mathbf{Q}_1\mathbf{U})&=\mathbf{U}^\top(\mathbf{Q}_1^\top\mathbf{Q}_1)\mathbf{U}=\mathbf{U}^\top\mathbf{U}=\mathbf{I}_r,\\
(\mathbf{Q}_2\mathbf{V})^\top(\mathbf{Q}_2\mathbf{V})&=\mathbf{V}^\top(\mathbf{Q}_2^\top\mathbf{Q}_2)\mathbf{V}=\mathbf{V}^\top\mathbf{V}=\mathbf{I}_r.
\end{align*}
Therefore, the spectral orthogonalization of $\mathbf{Q}_1\mathbf{M}\mathbf{Q}_2^\top$ is
\begin{align*}
\operatorname{msign}(\mathbf{Q}_1\mathbf{M}\mathbf{Q}_2^\top)=(\mathbf{Q}_1\mathbf{U})(\mathbf{Q}_2\mathbf{V}^\top)=\mathbf{Q}_1\cdot\operatorname{msign}(\mathbf{M})\cdot\mathbf{Q}_2^\top.
\end{align*}

\label{lemma:barv_properties}
Let $\bar\mathbf{v}_s$, $\bar\mathbf{v}_r$, and $\bar\mathbf{v}$ be defined in~\eqref{eq:def_barv}, and $\widetilde\mathbf{v}_s$, $\widetilde\mathbf{v}_r$ be defined in~\eqref{eq:def_tildev}. Then the following properties hold
\begin{itemize}[leftmargin=*]
\item[(a).] For any $s', r'$, we have
\begin{align*}
\bar\mathbf{v}_s^\top\bm{\phi}(a_*(s', r'))=\delta_{s, s'}/\sqrt R,\quad\bar\mathbf{v}_r^\top\bm{\phi}(a_*(s', r'))=\delta_{r, r'}/\sqrt S,\quad\bar\mathbf{v}^\top\bm{\phi}(a_*(s', r'))=1/\sqrt{SR}.
\end{align*}
\item[(b).] For any $s, r$ and $s', r'$, we have
\begin{gather*}
\bar\mathbf{v}_{s'}^\top\bar\mathbf{v}_s=\delta_{s, s'},\quad\bar\mathbf{v}_{r'}^\top\bar\mathbf{v}_r=\delta_{r, r'},\quad\bar\mathbf{v}_s^\top\bar\mathbf{v}_r=\frac1{\sqrt{SR}},\\
\bar\mathbf{v}_s^\top\bar\mathbf{v}=1/\sqrt S,\quad\bar\mathbf{v}_r^\top\bar\mathbf{v}=1/\sqrt R,\quad\bar\mathbf{v}^\top\bar\mathbf{v}=1.
\end{gather*}
\item[(c).] For any $s, r$ and $s', r'$, we have
\begin{align*}
\widetilde\mathbf{v}_{s'}^\top\widetilde\mathbf{v}_s=\delta_{s, s'}-1/S,\quad\widetilde\mathbf{v}_{r'}^\top\widetilde\mathbf{v}_r=\delta_{r, r'}-1/R,\quad\widetilde\mathbf{v}_s^\top\widetilde\mathbf{v}_r=0.
\end{align*}
\end{itemize}

<!-- proof-13: body_md -->
\noindent\textbf{Part (a).} This part holds due to the definition in \eqref{eq:def_barv} and because embeddings $\bm{\phi}(a)$ are orthonormal.

\noindent\textbf{Part (b).}
We firstly have
\begin{align*}
\bar\mathbf{v}_{s'}\bar\mathbf{v}_s=\frac1{\sqrt R}\sum_{r'\in\mathcal{R}}\bm{\phi}(a_*(s', r'))^\top\bar\mathbf{v}_s=\frac1{\sqrt R}\cdot R\cdot\frac{\delta_{s, s'}}{\sqrt R}=\delta_{s, s'},
\end{align*}
where the second equality holds due to Part (a). Similarly, we have $\bar\mathbf{v}_{r'}^\top\bar\mathbf{v}_r=\delta_{r, r'}$. For $\bar\mathbf{v}_s^\top\bar\mathbf{v}_r$, we have
\begin{align*}
\bar\mathbf{v}_s^\top\bar\mathbf{v}_r=\frac1{\sqrt R}\sum_{r'\in\mathcal{R}}\bm{\phi}(a_*(s, r'))^\top\bar\mathbf{v}_r=\frac1{\sqrt R}\sum_{r'\in\mathcal{R}}\frac{\delta_{r, r'}}{\sqrt S}=\frac1{\sqrt{SR}},
\end{align*}
where the second equality holds due to Part (a). Furthermore,
\begin{align*}
\bar\mathbf{v}_s^\top\bar\mathbf{v}=\frac1{\sqrt R}\sum_{r'\in\mathcal{R}}\bm{\phi}(a_*(s, r'))^\top\bar\mathbf{v}=\frac1{\sqrt R}\cdot R\cdot\frac1{\sqrt{SR}}=\frac1{\sqrt S}.
\end{align*}
Similarly, we have $\bar\mathbf{v}_r^\top\bar\mathbf{v}=1/\sqrt R$. Finally, we have
\begin{align*}
\bar\mathbf{v}^\top\bar\mathbf{v}=\frac1{\sqrt{SR}}\sum_{s', r'}\bm{\phi}(a_*(s, r))^\top\bar\mathbf{v}=\frac1{\sqrt{SR}}\cdot SR\cdot\frac1{\sqrt{SR}}=1.
\end{align*}

\noindent\textbf{Part (c).} We firstly have
\begin{align*}
\widetilde\mathbf{v}_{s'}^\top\widetilde\mathbf{v}_s&=(\bar\mathbf{v}_{s'}-\bar\mathbf{v}/\sqrt S)^\top(\bar\mathbf{v}_s-\bar\mathbf{v}/\sqrt S)\\
&=\bar\mathbf{v}_{s'}^\top\bar\mathbf{v}_s-\frac{\bar\mathbf{v}_{s'}^\top\bar\mathbf{v}}{\sqrt S}-\frac{\bar\mathbf{v}_s^\top\bar\mathbf{v}}{\sqrt S}+\frac{\bar\mathbf{v}^\top\bar\mathbf{v}}{S}\\
&=\delta_{s, s'}-\frac1{\sqrt S}\cdot\frac1{\sqrt S}-\frac1{\sqrt S}\cdot\frac1{\sqrt S}+\frac1S\cdot1\\
&=\delta_{s, s'}-\frac1S,
\end{align*}
where the third equality holds due to Part (b).
Similarly, $\widetilde\mathbf{v}_{r'}^\top\widetilde\mathbf{v}_r=\delta_{r, r'}-1/R$. Finally,
\begin{align*}
\widetilde\mathbf{v}_s^\top\widetilde\mathbf{v}_r&=\bar\mathbf{v}_s^\top\bar\mathbf{v}_r-\frac{\bar\mathbf{v}_s^\top\bar\mathbf{v}}{\sqrt R}-\frac{\bar\mathbf{v}_r^\top\bar\mathbf{v}}{\sqrt S}+\frac{\bar\mathbf{v}^\top\bar\mathbf{v}}{\sqrt{RS}}\\
&=\frac1{\sqrt{SR}}
\end{align*}
where the second equality holds due to Part (c).

\label{lemma:Ei_estimate}
Assume that $M\ge2$, then
\begin{align*}
\int_1^M\frac{e^u}{u}{\,\mathrm{d}} u\simeq\frac{e^M}{M}.
\end{align*}
<!-- end lemma-9 -->
<!-- proof-14: body_md -->
Applying integral by parts, we have
\begin{align*}
\int_1^M\frac{e^u}{u}{\,\mathrm{d}} u=\frac{e^M}{M}-e+\int_1^M\frac{e^u}{u^2}{\,\mathrm{d}} u.
\end{align*}
Since $M\ge2$, we have
\begin{align*}
\int_1^M\frac{e^u}{u}{\,\mathrm{d}} u\ge\frac{e^M}{M}-e\simeq\frac{e^M}{M}.
\end{align*}
It then suffices to prove that
\begin{align*}
\int_1^M\frac{e^u}{u^2}{\,\mathrm{d}} u\lesssim\frac{e^M}{M}.
\end{align*}
To prove this, we notice that
\begin{align*}
\int_1^M\frac{e^u}{u^2}{\,\mathrm{d}} u=\underbrace{\int_1^{M/2}\frac{e^u}{u^2}{\,\mathrm{d}} u}_{J_1}+\underbrace{\int_{M/2}^M\frac{e^u}{u^2}{\,\mathrm{d}} u}_{J_2}.
\end{align*}
For the term $J_1$, we have
\begin{align*}
J_1\le e^{M/2}\int_1^{M/2}\frac{{\,\mathrm{d}} u}{u^2}\le e^{M/2}\int_1^\infty\frac{{\,\mathrm{d}} u}{u^2}=e^{M/2}\lesssim\frac{e^M}{M}.
\end{align*}
where the first inequality holds because $u\le M/2$. For the term $J_2$, we have
\begin{align*}
J_2&\le\int_{M/2}^M\frac{e^u{\,\mathrm{d}} u}{(M/2)^2}\le\frac{4e^M}{M^2}\lesssim\frac{e^M}{M}.
\end{align*}
where the first inequality holds because $u\ge M/2$, and the second inequality holds because $\int_{M/2}^Me^u{\,\mathrm{d}} u=e^M-e^{M/2}\le e^M$.
