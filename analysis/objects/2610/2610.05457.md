<!-- definition-1: body_md -->
\label{def:p-var}
We write $x\in V_p(\pi;[0,T])$ (or $x\in V_p(\pi)$ if the context is clear) if the measures
\[
 \mu_n^{(p)}=\sum_{I=[u,v]\in\pi^n}|\Delta_Ix|^p\delta_v
\]
converge weakly to a finite measure $\mu^{(p)}$ without atoms.
We define $[x]^{(p)}_\pi(t)=\mu^{(p)}([0,t])$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:index}
The variation index and roughness index along $\pi$ are
\begin{equation}\label{def.H-hat}
 p^\pi(x)=\inf\{p\geq1:\limsup_n S_n(p)=0\},
 \qquad H^\pi(x)=1/p^\pi(x),
\end{equation}
with the convention $\inf\emptyset=\infty$ and $1/\infty=0$. We also define the lower variation index
\[
 r^\pi(x)=\inf\{p\geq1:\liminf_n S_n(p)=0\}.
\]
Thus $1\leq r^\pi(x)\leq p^\pi(x)\leq\infty$.
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop:index}
Let $M_n=\max_{I\in\pi^n}|\Delta_Ix|$. Then $M_n\to0$, and for all
$1\leq q<r$,
\begin{equation}\label{eq:max-increment}
 S_n(r)\leq M_n^{r-q}S_n(q).                             
\end{equation}
Consequently the following conclusions hold.
\begin{enumerate}
\item If $0<\limsup_nS_n(p_0)<\infty$, then $p^\pi(x)=p_0$.
\item If $0<\liminf_nS_n(p_0)\leq\limsup_nS_n(p_0)<\infty$, then
\begin{equation}\label{eq:threshold}
 S_n(q)\longrightarrow
 \begin{cases}0,&q>p_0,\\+\infty,&1\leq q<p_0,\end{cases}
 \qquad r^\pi(x)=p^\pi(x)=p_0.
\end{equation}
\item If $x\in C^\alpha$, $0<\alpha\leq1$, then
$p^\pi(x)\leq1/\alpha$, so $H^\pi(x)\geq\alpha$.
\item More generally, $r^\pi(x)=p^\pi(x)=p_0<\infty$ implies the
terminal-time limits in \eqref{eq:threshold}.
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Inequality~\eqref{eq:max-increment} follows term by term; continuity gives
$M_n\leq\omega_x(|\pi^n|)\to0$. If the critical sums  are bounded above,
all higher sums tend to zero. If their limsup is positive, a subsequence is
bounded below by a positive constant, and \eqref{eq:max-increment} forces
every lower-power sum to diverge on that subsequence. This proves (1).
A positive liminf gives divergence along the full sequence, proving (2).
For (3), if $|\Delta_Ix|\leq C|I|^\alpha$ and $q>1/\alpha$, then
$S_n(q)\leq C^qT|\pi^n|^{\alpha q-1}\to0$.
Finally, the sets defining the two indices are upward closed by
\eqref{eq:max-increment}, so $q>p^\pi(x)$ implies $S_n(q)\to0$.
If $q<r^\pi(x)$ and $\liminf_nS_n(q)<\infty$, choose
$q<r<r^\pi(x)$. Along a subsequence where $S_n(q)$ is bounded,
\eqref{eq:max-increment} gives $S_n(r)\to0$, contradicting the definition
of $r^\pi(x)$. This proves (4).
<!-- end proof-1 -->
<!-- definition-3: body_md -->
\label{def:estimator}
Fix a parameter interval $P=[p_-,p_+]\subset(1,\infty)$.\\
 A variation index estimator (in $P$) is defined as a solution of 
\begin{equation}
    W(L,K,\pi,q,T,x)= T\qquad q\in P=[p_-,p_+]\label{eq.estimator}
\end{equation}
whenever all denominators are positive.  We denote the  set of solutions of \eqref{eq.estimator} by
\[
 \mathcal Z_{L,K}(x;P)=\{q\in P:W(L,K,\pi,q,T,x)=T\}
\]
and by $ \widehat p^\pi_{L,K}(x)\in\mathcal Z_{L,K}(x;P)$ a generic element of this set.
We call

 \widehat H^\pi_{L,K}(x)=\frac{1}{\widehat p^\pi_{L,K}(x)}\qquad \widehat p^\pi_{L,K}(x)\in\mathcal Z_{L,K}(x;P)
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:consistency}
Denote
\[
 N_K=N(\pi^K),\qquad
  M_{L(K)}=\max_{J\in\pi^{L(K)}}|\Delta_Jx|,\qquad
  \Lambda_K=M_{L(K)}N_K^{1/p_0}.
\]
Suppose Assumption~\ref{ass:local} holds, $p_-<p_0<p_+$, and
\begin{equation}\label{eq:power-separation} 
  \delta_K\le C|\pi^K|^a
  \qquad\text{for some constants }C>0\text{ and }a>1
  \text{ and  sufficiently large }K.
\end{equation}
Then $\Lambda_K\to0$, and for every
$0<\varepsilon<\min\{p_0-p_-,p_+-p_0\}$,
\begin{align}
  \sup_{q\in[p_-,p_0-\varepsilon]}W_K(q)&\longrightarrow 0,
  \label{eq:uniform-below}\\
\inf_{q\in[p_0+\varepsilon,p_+]}W_K(q)&\longrightarrow+\infty.
  \label{eq:uniform-above}
\end{align}
For $K$ sufficiently large, all fine denominators are positive,
$  W_K(p_-)<T<W_K(p_+)$ and
\begin{equation}\label{eq:all-roots}
  \emptyset\ne\mathcal Z_{L(K),K}(x;P)\subset(p_-,p_+),
  \qquad
  \sup_{q\in\mathcal Z_{L(K),K}(x;P)}|q-p_0|\longrightarrow0.
\end{equation}
There exist constants $C_2,C_3>0$, independent of $K$,
such that for $K$ sufficiently large 
\begin{equation}\label{eq:quantitative-root-bound}
  \sup_{q\in\mathcal Z_{L(K),K}(x;P)}|q-p_0|
  \le\frac{C_2}{\log(1/\Lambda_K)}
  \le\frac{C_3}{\log(1/|\pi^K|)}.
\end{equation}
In particular, every solution $\widehat p^\pi_{L(K),K}(x)\in \mathcal Z_{L(K),K}(x;P)$  satisfies
\[
  \widehat p^\pi_{L(K),K}(x)\longrightarrow p^\pi(x)=p_0,
  \qquad
  \widehat H^\pi_{L(K),K}(x)\longrightarrow H^\pi(x)=H.
\]
Moreover, $p_{\mathrm{var}}(x)=p_0$.

<!-- proof-2: body_md -->
For this proof, all estimates below are for sufficiently large $K$ so that the bounds in Assumption~\ref{ass:local} hold.  The local lower bound implies
$D_{L(K),I}(p_0)>0$ for every coarse interval $I$. Each such interval
therefore contains a nonzero fine increment, so all its denominators
$D_{L(K),I}(q)$ are positive for $q>0$. In particular, $M_{L(K)}>0$.
The balanced property gives
\[
  N_K\le\frac{BT}{|\pi^K|}.
\]
Fix $\beta\in(H/a,H)$. Since $x\in C^\beta$, there is a constant
$C_\beta$ such that $M_{L(K)}\le C_\beta\delta_K^\beta$.
Using $1/p_0=H$ and \eqref{eq:power-separation}, we obtain
\begin{equation}\label{eq:scale-factor-bound}
  0<\Lambda_K
  \le C_\beta(BT)^H\delta_K^\beta |\pi^K|^{-H}
  \le C_4 |\pi^K|^{a\beta-H}\longrightarrow0.
\end{equation}
Thus $0<\Lambda_K<1$ for all sufficiently large $K$.

H\"older's inequality applied to the $N_K$ coarse increments yields
\begin{equation}\label{eq:power-mean}
\begin{aligned}
  S_K(q)&\le N_K^{1-q/p_0}S_K(p_0)^{q/p_0},
       &&p_-\le q<p_0,\\
  S_K(q)&\ge N_K^{1-q/p_0}S_K(p_0)^{q/p_0},
       &&p_0<q\le p_+.
\end{aligned}
\end{equation}
The second inequality follows by applying the first norm comparison
with the roles of $q$ and $p_0$ interchanged and rearranging.
For $q<p_0$, comparison of each nonzero fine increment gives
\[
  D_{L(K),I}(q)
  \ge M_{L(K)}^{q-p_0}D_{L(K),I}(p_0).
\]
Zero increments contribute zero to both sums and can be omitted.
The local lower critical bound, \eqref{eq:power-mean}, and the coarse
upper critical bound imply
\begin{equation}\label{eq:below-bound}
\begin{aligned}
  W_K(q)
  &\le c_1^{-1}M_{L(K)}^{p_0-q}S_K(q)\\
  &\le c_1^{-1}C_0^{q/p_0}
       M_{L(K)}^{p_0-q}N_K^{1-q/p_0}
   =c_1^{-1}C_0^{q/p_0}\Lambda_K^{p_0-q}.
\end{aligned}
\end{equation}
For $q>p_0$, the fine-increment comparison reverses:
\[
  D_{L(K),I}(q)
  \le M_{L(K)}^{q-p_0}D_{L(K),I}(p_0).
\]
Using the local upper critical bound and the coarse lower critical
bound gives
\begin{equation}\label{eq:above-bound}
\begin{aligned}
  W_K(q)
  &\ge C_1^{-1}M_{L(K)}^{p_0-q}S_K(q)\\
  &\ge C_1^{-1}c_0^{q/p_0}
       M_{L(K)}^{p_0-q}N_K^{1-q/p_0}
   =C_1^{-1}c_0^{q/p_0}\Lambda_K^{p_0-q}.
\end{aligned}
\end{equation}
Since $P$ is compact, choose constants $C_*>T>c_*>0$, independent
of $K$ and $q$, such that
\[
  C_*\ge\max_{q\in P}c_1^{-1}C_0^{q/p_0},\qquad
  c_*\le\min_{q\in P}C_1^{-1}c_0^{q/p_0}.
\]
For any $\varepsilon$ in the stated range, it follows that
\[
  \sup_{q\in[p_-,p_0-\varepsilon]}W_K(q)
       \le C_*\Lambda_K^\varepsilon\longrightarrow0,
  \qquad
  \inf_{q\in[p_0+\varepsilon,p_+]}W_K(q)
       \ge c_*\Lambda_K^{-\varepsilon}\longrightarrow+\infty.
\]
This proves \eqref{eq:uniform-below}--\eqref{eq:uniform-above}.
In particular, the endpoint values strictly bracket $T$ for all
sufficiently large $K$. Continuity of $W_K$ in $q$ gives a root in
$(p_-,p_+)$, and the separation estimates exclude every root outside
$(p_0-\varepsilon,p_0+\varepsilon)$ for sufficiently large $K$.
This proves \eqref{eq:all-roots}.

At a root $q<p_0$, the bound $T\le C_*\Lambda_K^{p_0-q}$ gives
\[
  p_0-q\le\frac{\log(C_*/T)}{\log(1/\Lambda_K)}.
\]
At a root $q>p_0$, the bound $T\ge c_*\Lambda_K^{p_0-q}$ gives
\[
  q-p_0\le\frac{\log(T/c_*)}{\log(1/\Lambda_K)}.
\]
A root at $p_0$ satisfies either error bound trivially. Taking
$C_2=\max\{\log(C_*/T),\log(T/c_*)\}$ proves the first inequality
in \eqref{eq:quantitative-root-bound}. Equation~\eqref{eq:scale-factor-bound}
implies
\[
  \log(1/\Lambda_K)
  \ge(a\beta-H)\log(1/|\pi^K|)-\log C_4
  \ge\tfrac12(a\beta-H)\log(1/|\pi^K|)
\]
for all sufficiently large $K$, proving the second inequality.

Assumption~\ref{ass:local}(3) and Proposition~\ref{prop:index}(2)
give $p^\pi(x)=p_0$. For $q\in P$,
\[
  \left|\frac1q-\frac1{p_0}\right|
  =\frac{|q-p_0|}{q p_0}
  \le\frac{|q-p_0|}{p_-p_0},
\]
which proves the assertions for the reciprocal estimator.

<!-- proposition-2: body_md -->
\label{prop:normalization}
Under Assumption~\ref{ass:density}, $w^K(x,p_0,\pi)(t)\to t$ uniformly
on $[0,T]$. Denote $A_n^{(p_0)}$ as in \eqref{eq:power-sums} and
\[
 E_n=\max_{u\in\pi^n}|A_n^{(p_0)}(u)-[x]^{(p_0)}_\pi(u)|.
\]
If $L(K)>K$ is such that
\begin{equation}\label{eq:relative-error}
\frac{E_{L(K)}}{\underline{\pi^K}}\mathop{\longrightarrow}^{K\to\infty}0,
\end{equation}
then all fine denominators are  positive for $K$ large enough and
\begin{equation}\label{eq:critical-normalization}
\sup_{t\in[0,T]}|W(L(K),K,\pi,p_0,t,x)-t|\longrightarrow0.
\end{equation}
For fixed $K$, $W(L,K,\pi,p_0,\cdot,x)\to w^K(x,p_0,\pi)$ uniformly as
$L\to\infty$. Consequently, 
\[
\lim_{K\to\infty}\lim_{L\to\infty}
W(L,K,\pi,p_0,t,x)=t,
\]
uniformly for \(t\in[0,T]\).

Write $m=\min g>0$, $M=\max g$, and $\bar g_I=|I|^{-1}\int_I g$.
Uniform continuity gives
\[
 \left|\bar g_I^{-1}-g(v)^{-1}\right|
 \leq m^{-2}\omega_g(|\pi^K|),\qquad I=[u,v]\in\pi^K.
\]
Therefore \eqref{eq.weighted.p.var} differs uniformly from
$\sum_{v\leq t}g(v)^{-1}|\Delta_Ix|^{p_0}$ by at most
$m^{-2}\omega_g(|\pi^K|)S_K(p_0)\to0$.
Weak convergence of $\mu_K^{(p_0)}$ to $g(t)\mathrm d t$ implies weak convergence
of the positive measures $g^{-1}\mu_K^{(p_0)}$ to Lebesgue measure: test
against a continuous $f$ and use continuity of $f/g$. Their cumulative masses converge uniformly \cite[Lemma 1.3]{perkowski2019}.
This proves the first assertion.

Since coarse endpoints belong to the fine grid,
\[
 D_{L,I}(p_0)=A_L^{(p_0)}(v)-A_L^{(p_0)}(u),\qquad
 |D_{L,I}(p_0)-([x]^{(p_0)}_\pi(v)-[x]^{(p_0)}_\pi(u))|\leq2E_L.
\]
The relative error is at most
$\eta_K=2E_{L(K)}/(m\underline{\pi^K})\to0$.
For large $K$, each denominator lies between
$(1-\eta_K)([x]^{(p_0)}_\pi(v)-[x]^{(p_0)}_\pi(u))$ and $(1+\eta_K)([x]^{(p_0)}_\pi(v)-[x]^{(p_0)}_\pi(u))$.
Term-by-term comparison  gives
\[
 \sup_t|W(L(K),K,\pi,p_0,t,x)-w^K(x,p_0,\pi)(t)|
 \leq\frac{\eta_K}{1-\eta_K}w^K(x,p_0,\pi)(T)\longrightarrow0.
\]
For fixed $K$, the same argument applies because $E_L\to0$ and there are
finitely many positive coarse variation increments.

<!-- proposition-3: body_md -->
\label{prop:fixed-coarse-consistency}
Let $x\in V_{p_0}(\pi)$ for some $p_0>1$. Fix a coarse level $K_0$ such that
\begin{equation}\label{eq:fixed-coarse-nondegeneracy}
 [x]^{(p_0)}_\pi(v)-[x]^{(p_0)}_\pi(u)>0\quad\text{for every }I=[u,v]\in\pi^{K_0},
 \qquad S_{K_0}(p_0)>0.
\end{equation}
Let $P=[p_-,p_+]\subset(1,\infty)$ with $p_-<p_0<p_+$.
For $L$ large enough, all fine denominators are positive, 
\[
W(L,K_0,\pi,p_-,T,x)<T<W(L,K_0,\pi,p_+,T,x)
\]
and
\begin{equation}\label{eq:fixed-coarse-consistency}
 \sup_{q\in\mathcal Z_{L,K_0}(x;P)}|q-p_0|\longrightarrow0.
\end{equation}
Thus every root selection satisfies
$\widehat p^\pi_{L,K_0}(x)\to p^\pi(x)=p_0$ and
$\widehat H^\pi_{L,K_0}(x)\to1/p_0$.
Moreover, 
there is a constant $C<\infty$, independent of $L$, such that for all
sufficiently large $L$,
\begin{equation}\label{eq:fixed-coarse-rate}
 \sup_{q\in\mathcal Z_{L,K_0}(x;P)}|q-p_0|
 \leq\frac{C}{\log(1/M_L)}\qquad{\rm where}\quad M_L=\max_{J\in\pi^L}|\Delta_Jx|.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-6: body_md -->
Since the partitions are refining, for every  coarse interval
$I=[u,v]\in\pi^{K_0}$,
\[
 D_{L,I}(p_0)=A_L^{(p_0)}(v)-A_L^{(p_0)}(u)
 \longrightarrow [x]^{(p_0)}_\pi(v)-[x]^{(p_0)}_\pi(u)>0.
\]
where we use the notation $A_L^{(p_0)}$ as in \eqref{eq:power-sums}.
There are finitely many coarse intervals. Hence constants $0<m<M<\infty$
exist such that, for all sufficiently large $L$,
\[
 m\leq D_{L,I}(p_0)\leq M\qquad\text{for every }I\in\pi^{K_0}.
\]
Every block therefore contains a nonzero fine increment, making
$D_{L,I}(q)>0$ for every $q>0$. Continuity of $x$ and the vanishing fine
mesh imply $M_L\to0$; thus $0<M_L<1$ for $L$ large enough.
The function
\[
 B(q)=\sum_{I\in\pi^{K_0}}|I|\,|\Delta_Ix|^q
\]
is continuous and strictly positive on $P$, because
$S_{K_0}(p_0)>0$ means that at least one coarse increment is nonzero.
Consequently constants $b_-,b_+>0$ exist such that
$b_-\leq B(q)\leq b_+$ for every $q\in P$.
If $q<p_0$, comparison of each nonzero fine increment gives
\[
 D_{L,I}(q)\geq(M_L)^{q-p_0}D_{L,I}(p_0).
\]
Zero increments contribute zero and can be omitted. Summing the resulting
upper bounds for the terms of $W$ yields
\[
 W(L,K_0,\pi,q,T,x)\leq\frac{b_+}{m}(M_L)^{p_0-q}.
\]
For $q>p_0$, the fine-increment comparison reverses and gives
\[
 D_{L,I}(q)\leq(M_L)^{q-p_0}D_{L,I}(p_0),\qquad
 W(L,K_0,\pi,q,T,x)\geq\frac{b_-}{M}(M_L)^{p_0-q}.
\]
For  $0<\varepsilon<\min\{p_0-p_-,p_+-p_0\}$, these bounds imply
\begin{align*}
 \sup_{\substack{q\in P\\q\leq p_0-\varepsilon}}
 W(L,K_0,\pi,q,T,x)&\leq\frac{b_+}{m}(M_L)^\varepsilon\longrightarrow0,\\
 \inf_{\substack{q\in P\\q\geq p_0+\varepsilon}}
 W(L,K_0,\pi,q,T,x)&\geq\frac{b_-}{M}(M_L)^{-\varepsilon}
 \longrightarrow\infty.
\end{align*}
Since $p_0$ is interior to $P$, $T$ falls between the endpoint values for $L$ large. Continuity in $q$ gives a root by the intermediate value theorem,
and the same estimates exclude every root outside any prescribed
neighbourhood of $p_0$. This proves \eqref{eq:fixed-coarse-consistency}.

Choose constants $C_*>T>c_*>0$ with $C_*\geq b_+/m$ and
$c_*\leq b_-/M$. At a root $q<p_0$,
\[
 T\leq C_*(M_L)^{p_0-q}
 \quad\Longrightarrow\quad
 p_0-q\leq\frac{\log(C_*/T)}{\log(1/M_L)}.
\]
At a root $q>p_0$,
\[
 T\geq c_*(M_L)^{p_0-q}
 \quad\Longrightarrow\quad
 q-p_0\leq\frac{\log(T/c_*)}{\log(1/M_L)}.
\]
A root at $p_0$ already satisfies the bound. Taking the larger of the two
numerators proves \eqref{eq:fixed-coarse-rate}.
Finally, $S_L(p_0)\to [x]^{(p_0)}_\pi(T)>0$, so Proposition~\ref{prop:index}(2)
gives $p^\pi(x)=p_0$. Continuity of $q\mapsto1/q$ on $P$ gives the
convergence of the reciprocal estimator.

<!-- theorem-2: body_md -->
\label{thm:schauder}
Define $x\in C^H([0,1],\mathbb{R})$ as in \eqref{eq:schauder} and $p_0=1/H$. There exists constants
$c,C>0$  such that
\begin{equation}\label{eq:schauder-bounds}
 c|I|\leq D_{L,I}(p_0)\leq C|I|,
 \qquad I\in\mathbb T^K,\quad L>K\geq0,
\end{equation}
and $c\leq S_K(p_0)\leq C$ for $K\geq1$.
Consequently $p^{\mathbb T}(x)=p_0$, and the estimator is consistent for every
choice $K\to\infty$, $L(K)-K\to\infty$, with $p_0$ interior to $P$.
Its roots satisfy $$|\widehat p^{\mathbb T}_{L,K}-p_0|\leq \frac{C_P}{L-K}$$ for

Fix $I\in\mathbb T^K$ and choose one of its $2^{L-K}$ fine cells uniformly.
                                                                                                                               
We randomize only to average; we pick one of the $2^{L-K}$ fine cells of $I$ uniformly at random; $x$ itself stays deterministic.
All probabilities and expectations below refer solely to the uniform choice of a fine interval inside the fixed coarse interval $I$.  Successive
left/right choices in the dyadic tree give independent signs
$\varepsilon_m\in\{-1,1\}$, $K\leq m<L$. 
The tent selected at level
$m$ is determined by earlier choices; its coefficient is therefore predictable before $\varepsilon_m$ is revealed. Let $\rho=2^{-(1-H)}$.
The normalized increment on the chosen cell is exactly
\begin{equation}\label{eq:tree-increment}
 Y=2^{LH}\Delta_Jx
  =v_{K,L,I}+\sum_{m=K}^{L-1}\rho^{L-m}\theta_{m,k_m}\varepsilon_m.
\end{equation}

slope of a level-$m$ summand is

 |v_{K,L,I}|\leq C\rho^{L-K}.
\end{equation}

this finite tree and are pairwise orthogonal in $L^2$. It follows that
\[
 |Y|\leq B_0:=|b|+a_+\rho/(1-\rho),\qquad
 \mathbb E Y^2\geq\sum_{m=K}^{L-1}\rho^{2(L-m)}a_-^2
             \geq a_-^2\rho^2>0.
\]

$\mathbb E|Y|^{p_0}\geq(\mathbb E Y^2)^{p_0/2}$.
If $1<p_0<2$, use $Y^2\leq B_0^{2-p_0}|Y|^{p_0}$.

\]
This proves \eqref{eq:schauder-bounds}. Taking the single interval at

<!-- theorem-3: body_md -->
\label{thm:signed-rate}
Let $x\in\mathcal X^{p_0}$, $H=1/p_0$, $\rho=2^{-(1-H)}$, and
$P\subset(1,\infty)$ compact. Uniformly in $q\in P$ and in the sign array,
as $K\to\infty$ and $d=L-K\to\infty$,
\begin{equation}\label{eq:signed-asymptotic}
 W(L,K,\mathbb T,q,1,x)
 =2^{d(Hq-1)}\bigl(1+O_P(\rho^K+\rho^d)\bigr).
\end{equation}
If $p_0$ is interior to $P$, every root in $P$ satisfies
\begin{equation}\label{eq:signed-rate}
 |\widehat p^{\mathbb T}_{L,K}(x)-p_0|
 \leq C_P\frac{\rho^K+\rho^{L-K}}{L-K}.
\end{equation}
The same bound, with another constant, holds for
$|\widehat H^{\mathbb T}_{L,K}(x)-H|$.
In particular $L=2K$ gives an error bound
$O(2^{-K(1-H)}/K)$, and $W(L,K,\mathbb T,p_0,1,x)\to1$.

product value has probability $1/2$.
Consequently, for auxiliary independent Rademacher variables $Y_r$, the
tail in \eqref{eq:tree-increment} has the distribution of

 Z_d=\sum_{r=1}^d\rho^rY_r,

Both $|Z_\infty-Z_d|$ and $|v_{K,L,I}|$ are bounded by $C\rho^d$.

 \mathbb E|v_{K,L,I}+Z_d|^q=c_H(q)+O_P(\rho^d).
\end{equation}
The constants are independent of $I$ and of the sign array.

\label{rem:finite-sign-invariance}
In fact, for dyadic observations of $x\in\mathcal X^{p_0}$, the entire
finite-sample statistic is independent of the sign array. With $d=L-K$,

\begin{equation}\label{eq:exact-signed-statistic}
 W(L,K,\mathbb T,q,1,x)=
 2^{d(Hq-1)}\mathbb E\left[\frac{|Z_K|^q}{f_d(Z_K,q)}\right],\qquad q>0.
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:fgn}
Let $Z_j=B^H_j-B^H_{j-1}$, $1\leq j\leq m$, denote a vector of unit-scale
fractional Gaussian noise. Put $\kappa=\min(H,1-H)>0$.
For every $\varepsilon\in(0,\mu_H)$ there are $c_\varepsilon>0$ and
$m_\varepsilon$ such that
\begin{equation}\label{eq:fgn-concentration}
 \mathbb P\left(\left|m^{-1}\sum_{j=1}^m|Z_j|^{p_0}-\mu_H\right|>
 \varepsilon\right)
 \leq2\exp(-c_\varepsilon m^{2\kappa}),\qquad m\geq m_\varepsilon.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-12: body_md -->
The covariance matrix $\Gamma_m$ has entries $r_H(i-j)$, where
\[
 r_H(k)=\tfrac12\bigl(|k+1|^{2H}+|k-1|^{2H}-2|k|^{2H}\bigr).
\]
For $|k|\geq2$, the second-difference formula and the second derivative
of $u^{2H}$ show $|r_H(k)|\leq C_H|k|^{2H-2}$.
For $H<1/2$ this is summable; for $H=1/2$ the matrix is the identity;
for $H>1/2$ its absolute row sums are at most $C_Hm^{2H-1}$.
Thus, by the row-sum bound for a symmetric matrix,
\begin{equation}\label{eq:covariance-norm}
 \|\Gamma_m\|_{\mathrm{op}}\leq
 \begin{cases}C_H,&H\leq1/2,\\ C_Hm^{2H-1},&H>1/2.\end{cases}
\end{equation}
Represent the vector as $\Gamma_m^{1/2}G$ for a standard Gaussian vector
$G\in\mathbb R^m$, and define
$F_m(G)=m^{-1/p_0}\|\Gamma_m^{1/2}G\|_{p_0}$.
The reverse triangle inequality for the $\ell_{p_0}$ norm and
\[
 \|z\|_{p_0}\leq
 \begin{cases}\|z\|_2,&p_0\geq2,\\
 m^{1/p_0-1/2}\|z\|_2,&1<p_0<2
 \end{cases}
\]
give the Euclidean Lipschitz bound
\[
 \operatorname{Lip}(F_m)\leq
 \begin{cases}C_Hm^{-H},&H\leq1/2,\\
 C_Hm^{H-1},&H>1/2,
 \end{cases}
 \leq C_Hm^{-\kappa}.
\]
By the Gaussian concentration inequality
\cite{BoucheronLugosiMassart2013}, states any  $l$-Lipschitz function $F$ of
a standard Gaussian vector satisfies
$\mathbb P(|F-\mathbb E F|>u)\leq2\exp(-u^2/(2l^2))$.
Integration of this tail bound gives
$\|F_m-\mathbb E F_m\|_{L^{p_0}}\leq C m^{-\kappa}$.
Since $\mathbb E F_m^{p_0}=\mu_H$, the reverse triangle inequality in
$L^{p_0}$ implies
\[
 |\mathbb E F_m-\mu_H^{1/p_0}|
 \leq\|F_m-\mathbb E F_m\|_{L^{p_0}}\longrightarrow0.
\]
For fixed $\varepsilon\in(0,\mu_H)$, the event
$|F_m^{p_0}-\mu_H|>\varepsilon$ forces
$|F_m-\mu_H^{1/p_0}|>d_\varepsilon$ for a fixed $d_\varepsilon>0$.
For large $m$, it therefore forces
$|F_m-\mathbb E F_m|>d_\varepsilon/2$. Apply Gaussian concentration with the
Lipschitz bound above to obtain \eqref{eq:fgn-concentration}.

\label{thm:fbm}
Let $B^H$ be standard fractional Brownian motion with $H\in(0,1)$,
and put
\[
 p_0=\frac1H,
 \qquad \mu_H=\mathbb E|Z|^{p_0},\quad Z\sim N(0,1).
\]
Fix $a>1$ and $
 P=[p_-,p_+],\quad 1<p_-<p_0<p_+<\infty.$
For each integer $K\geq1$, use the coarse partition $\mathbb T^K$ and
the fine partition $\mathbb T^{L(K)}$ of $[0,1]$, where
\[
 L(K)=\lceil aK\rceil.
\]
There exists an event $\Omega_0$ with $\mathbb P(\Omega_0)=1$ such that,
for  $\omega\in\Omega_0$, the path $x=B^H(\omega)$ satisfies:
\begin{enumerate}
\item                                      
There exists $K_1=K_1(\omega)>0$ such that, for every $K\geq K_1$,
$D_{L(K),I}(q)>0$ for every $I\in\mathbb T^K,\ q\in P$
and
\[
 W(L(K),K,\mathbb T,p_-,1,x)
 <1<
 W(L(K),K,\mathbb T,p_+,1,x).
\]
Consequently, the estimating equation
\begin{equation}\label{eq.root}
     W(L(K),K,\mathbb T,q,1,x)=1
\end{equation}
has at least one solution in $(p_-,p_+)$ for any $K\geq K_1$.
\item                                        
 For each $\omega\in\Omega_0$, there are constants
$C(\omega)>0$ and $K_2(\omega)\geq K_1(\omega)$ such that 
\[
\forall K\ge K_2(\omega),\qquad \sup_{q\in\mathcal Z_{L(K),K}(x;P)}
 \left(|q-p_0|+\left|q^{-1}-H\right|\right)
 \le \frac{C(\omega)}{K}.
\]
so all solutions of \eqref{eq.root} are consistent estimators:
\[
 \lim_{K\to\infty}
 \sup_{q\in\mathcal Z_{L(K),K}(x;P)}|q-p_0|=0.
\]
In particular, for any sequence of solutions
$ \widehat p^{\mathbb T}_{L(K),K}(x)
 \in\mathcal Z_{L(K),K}(x;P),\quad K\geq K_1,$
\[
 \widehat p^{\mathbb T}_{L(K),K}(x)\mathop{\longrightarrow}^{K\to\infty}\frac1H,
 \qquad
 \widehat H^{\mathbb T}_{L(K),K}(x)
 =\frac1{\widehat p^{\mathbb T}_{L(K),K}(x)}
\mathop{\longrightarrow}^{K\to\infty} H.
\]
\item                                                
On the same event,
\begin{equation}\label{eq:fbm-block}
 \lim_{K\to\infty}
 \max_{I\in\mathbb T^K}
 \left|\frac{D_{L(K),I}(p_0)}{|I|}-\mu_H\right|=0,
 \qquad
 \lim_{K\to\infty}S_K(p_0)=\mu_H,
\end{equation}
and
\begin{equation}\label{eq:fbm-W}
 \lim_{K\to\infty}
 \sup_{t\in[0,1]}
 \left|W(L(K),K,\mathbb T,p_0,t,x)-t\right|=0.
\end{equation}
\end{enumerate}
The thresholds $K_1, K_2$ may depend on the sample path and on $a$ and $P$. All limits are taken with
$L(K)=\lceil aK\rceil$.

Each coarse interval contains $m_K=2^{L(K)-K}$ fine cells.
Stationarity and self-similarity imply that, for each fixed coarse interval,
\[
 \frac{D_{L(K),I}(p_0)}{|I|}
 \ \stackrel{\mathrm{law}}{=}\ m_K^{-1}\sum_{j=1}^{m_K}|Z_j|^{p_0}.

The equality uses $Hp_0=1$. There are $2^K$ blocks, so the union bound and

This bound is summable in $K$, since $L(K)-K\geq(a-1)K$.

 \leq C\epsilon_K S_K(p_0)\longrightarrow0.

Uniform convergence of the cumulative sums proves \eqref{eq:fbm-W}. The quantitative error bound follows from

<!-- proposition-4: body_md -->
\label{prop:stability}
Let $x$ satisfy Assumption~\ref{ass:local} with exponent $p_0=1/H$.
Then in the following cases $y$ also satisfies  Assumption~\ref{ass:local} for the same $p_0$, $\pi$, and $L(K)$:
\begin{enumerate}
\item $y(t)=x(t)+v(t)$ where $v\in C^\gamma([0,T])$ with $H<\gamma\leq1$;
\item  $y(t)=a(t)x(t)+b(t)$, where $a,b\in C^1([0,T])$
and $\min_t|a(t)|>0$;
\item $y(t)=F(x(t))$, where $F$ is continuously differentiable on an open interval
containing the range of $x$, and $\inf_{z\in x([0,T])}|F'(z)|>0$.
\end{enumerate}
The same statement holds with $C^H$ in place of $C^{H-}$ when $x\in C^H$.
Consequently Theorem \ref{thm:consistency} remains applicable to $y$.
<!-- end proposition-4 -->
<!-- proof-14: body_md -->
Write $p=p_0$, so $1/p=H$. For a block $I$, the $\ell_p$ norm of the
fine increments of $v\in C^\gamma$ is bounded by
\[
 \left(\sum_{J\subseteq I}|\Delta_Jv|^p\right)^{1/p}
 \leq C\left(\delta_K^{\gamma p-1}|I|\right)^{1/p}
 =C\delta_K^{\gamma-H}|I|^H.
\]
The corresponding norm for $x$ lies between $c_1^{1/p}|I|^H$ and
$C_1^{1/p}|I|^H$. The triangle and reverse triangle inequalities therefore
preserve the two-sided local bounds for $x+v$. On the whole coarse grid,
the norm of the perturbation is at most
$CT^H|\pi^K|^{\gamma-H}\to0$, whereas the norm for $x$ is bounded above and
away from zero. This proves the coarse bounds in (1).

For (2), put $y=ax+b$ and fix $I=[u,v]$. On a fine cell $J=[s,t]\subseteq I$,
\[
 \Delta_Jy=a(u)\Delta_Jx+e_J,\qquad
 e_J=(a(s)-a(u))\Delta_Jx+(a(t)-a(s))x(t)+\Delta_Jb.
\]
Since $a,b$ are Lipschitz and $x$ bounded, the $\ell_p$ norm of the error
over the fine cells is at most
\[
 C|I|D_{L(K),I}(p)^{1/p}
 +C\left(\sum_{J\subseteq I}|J|^p\right)^{1/p}
 \leq C(|\pi^K|+\delta_K^{1-H})|I|^H.
\]
This is $o(|I|^H)$ uniformly in $I$. The main term has norm
$|a(u)|D_{L(K),I}(p)^{1/p}$, bounded above and below by constant
multiples of $|I|^H$. The local bounds follow.
For the coarse sums, use
$\Delta_Iy=a(u)\Delta_Ix+(a(v)-a(u))x(v)+\Delta_Ib$.
The last two terms have global $\ell_p$ norm $O(|\pi^K|^{1-H})$;
the first has norm between
$(\min|a|)S_K(p)^{1/p}$ and $(\max|a|)S_K(p)^{1/p}$.
This proves the coarse bounds.

For (3), the range $x([0,T])$ is a compact interval. The mean value theorem
gives constants $0<c\leq C<\infty$ such that
\[
 c|\Delta_Jx|\leq|\Delta_J(F\circ x)|\leq C|\Delta_Jx|
\]
for every interval $J$. Summation immediately proves both critical bounds.
All three operations preserve $C^{H-}$, or $C^H$ in the exact case:
use the increment bounds above, $\gamma>H$, and the bounded derivative
of $F$ on the compact range. This verifies the remaining assumption.
