<!-- definition-1: body_md -->
\label{def1.1}
Let $d \ge 3$ and $T > 0$. Suppose $(u_0, w_0)$ satisfies \eqref{1.3}--\eqref{1.4}. Then $(u, w)$ of nonnegative functions defined in $\mathbb{R}^d \times (0, T)$ is called a weak solution if:
\begin{itemize}
\item[(i)] $(u, w) \in \left( C([0, T); L^1(\mathbb{R}^d)) \cap L^\infty(\mathbb{R}^d \times (0, T)) \right)^2$ and $\left(u^{\frac{2d}{d+2}}, w^{\frac{2d}{d+2}}\right) \in \left( L^2(0, T; H^1(\mathbb{R}^d)) \right)^2$.
\item[(ii)] $(u, w)$ satisfies
\begin{equation}\label{1.14}
\begin{split}
&\int_0^T \int_{\mathbb{R}^d} u \phi_{1t} \, dx \, dt + \int_{\mathbb{R}^d} u_0(x) \phi_1(x, 0) \, dx \\
&= \int_0^T \int_{\mathbb{R}^d} \left( \frac{\alpha_1}{2} \|u\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} \nabla\left(u^{\frac{2d}{d+2}}\right) - u\nabla v \right) \cdot \nabla \phi_1 \, dx \, dt, \\
&\int_0^T \int_{\mathbb{R}^d} w \phi_{2t} \, dx \, dt + \int_{\mathbb{R}^d} w_0(x) \phi_2(x, 0) \, dx \\
&= \int_0^T \int_{\mathbb{R}^d} \left( \frac{\alpha_2}{2} \|w\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} \nabla\left(w^{\frac{2d}{d+2}}\right) - w\nabla z \right) \cdot \nabla \phi_2 \, dx \, dt,
\end{split}
\end{equation}
for any test functions $\phi_i \in \mathcal{D}(\mathbb{R}^d \times [0, T))$ with $v = K * w$ and $z = K * u$.
\end{itemize}

<!-- definition-2: body_md -->
\label{def1.2}

Let $T > 0$. Then $(u, w)$ is called a free energy solution with initial data $(u_0, w_0)$ on $(0, T)$ if $(u, w)$ is a weak solution and moreover satisfies
\begin{equation}\label{1.15}
\left( u^{\frac{3d-2}{2(d+2)}}, w^{\frac{3d-2}{2(d+2)}} \right) \in \left( L^2(0, T; H^1(\mathbb{R}^d)) \right)^2,
\end{equation}
and
\begin{equation}\label{1.16}
\begin{aligned}
E_\alpha[u(t), w(t)] &+ \int_0^t \int_{\mathbb{R}^d} u \left| \nabla \left( \frac{\alpha_1}{2} \|u\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} u^{\frac{d-2}{d+2}} \right) - \nabla v \right|^2 \, dx \, ds \\
&+ \int_0^t \int_{\mathbb{R}^d} w \left| \nabla \left( \frac{\alpha_2}{2} \|w\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} w^{\frac{d-2}{d+2}} \right) - \nabla z \right|^2 \, dx \, ds \le E_\alpha[u_0, w_0],
\end{aligned}
\end{equation}
for all $t \in (0, T)$ with $v = K * w$ and $z = K * u$.

<!-- lemma-2: body_md -->
\label{lem:lower_bound}
Let $(u_\epsilon, w_\epsilon)$ be a smooth, non-negative solution to the regularized system \eqref{2.1} on $[0, T]$ satisfying mass conservation $\int_{\mathbb{R}^d} u_\epsilon(x,t)\,dx = M_1$ and a bounded second moment $I(t) \le I_0$. Then there exists a constant $c_{\min} = c_{\min}(M_1, I_0, d) > 0$ independent of $\epsilon$ such that
\begin{equation}
\|u_\epsilon(\cdot, t)\|_{\frac{2d}{d+2}} \ge c_{\min} > 0, \quad \forall t \in [0, T].
\end{equation}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Let $R > 0$ be a radial cutoff parameter. Splitting the total conserved mass inside and outside a ball $B_R(0)$, we have:
\begin{align*}
M_1 = \int_{\mathbb{R}^d} u_\epsilon(x,t)\,dx &= \int_{B_R(0)} u_\epsilon(x,t)\,dx + \int_{\mathbb{R}^d \setminus B_R(0)} u_\epsilon(x,t)\,dx \\
&\le \int_{B_R(0)} u_\epsilon(x,t)\,dx + \frac{1}{R^2} \int_{\mathbb{R}^d \setminus B_R(0)} |x|^2 u_\epsilon(x,t)\,dx \\
&\le \int_{B_R(0)} u_\epsilon(x,t)\,dx + \frac{I_0}{R^2}.
\end{align*}
Choosing $R = \sqrt{\frac{2I_0}{M_1}}$ yields $\int_{B_R(0)} u_\epsilon(x,t)\,dx \ge \frac{M_1}{2}$. Applying H{\"o}lder's inequality on $B_R(0)$ with exponents $p = \frac{2d}{d+2}$ and $q = \frac{2d}{d-2}$, we obtain:
\[ \frac{M_1}{2} \le \int_{B_R(0)} u_\epsilon(x,t)\,dx \le \|u_\epsilon\|_{L^{\frac{2d}{d+2}}(B_R(0))} |B_R(0)|^{\frac{d-2}{2d}} . \]
Since $|B_R(0)| = \omega_d R^d$, we isolate the norm:
\begin{equation}
\|u_\epsilon(\cdot, t)\|_{\frac{2d}{d+2}} \ge \frac{M_1}{2} \left( \omega_d \left(\frac{2I_0}{M_1}\right)^{d/2} \right)^{-\frac{d-2}{2d}} =: c_{\min} > 0.
\end{equation}
The same argument holds for $w_\epsilon(t)$ with mass $M_2$.

<!-- lemma-3: body_md -->
\label{lem2.2}
Let $T \in (0, T_\epsilon^{\max}]$. Suppose that there exists a constant $C > 0$ such that the solution satisfies $\|u_\epsilon(t)\|_{\frac{2d}{d+2}} \le C$ and $\|w_\epsilon(t)\|_{\frac{2d}{d+2}} \le C$ for $t \in (0, T)$. Then there exists a constant $C = C(d, u_0, w_0) > 0$ such that
\begin{equation}\label{2.4}
\|(u_\epsilon(t), w_\epsilon(t))\|_r \le C, \quad \|(v_\epsilon(t), z_\epsilon(t))\|_r + \|(\nabla v_\epsilon(t), \nabla z_\epsilon(t))\|_r \le C,
\end{equation}
for $r \in [1, \infty)$ and $t \in (0, T)$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For $p > 1$, we test the first equation in \eqref{2.1} by $u_\epsilon^{p-1}$ and integrate:
\begin{equation}\label{2.6}
\begin{split}
\frac{1}{p} \frac{d}{dt} \int_{\mathbb{R}^d} u_\epsilon^p \, dx &= -\alpha_1 \left(\frac{d - 2}{d}\right) \|u_\epsilon\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} \int_{\mathbb{R}^d} \nabla(u_\epsilon^{p-1}) \cdot \nabla(u_\epsilon + \epsilon)^{\frac{2d}{d+2}} \, dx \\
& \qquad + \int_{\mathbb{R}^d} u_\epsilon \nabla(u_\epsilon^{p-1}) \cdot \nabla v_\epsilon \, dx.
\end{split}
\end{equation}
Using the quantified lower bound from Lemma \ref{lem:lower_bound}, the first term is bounded away from zero, preventing structural disappearance:
\begin{align}\label{2.7}
I &\le -2\alpha_1 \frac{d - 2}{d + 2} (p - 1) c_{\min}^{\frac{4}{d+2}} \int_{\mathbb{R}^d} u_\epsilon^{p-2} (u_\epsilon + \epsilon)^{\frac{2d}{d+2} - 1} |\nabla u_\epsilon|^2 \, dx \nonumber \\
  &\le -4\alpha_1 \frac{d - 2}{d + 2} \frac{(p - 1)}{\left(p + \frac{d-2}{d+2}\right)^2} c_{\min}^{\frac{4}{d+2}} \int_{\mathbb{R}^d} \left| \nabla u_\epsilon^{\frac{p + \frac{d-2}{d+2}}{2}} \right|^2 \, dx.
\end{align}
The cross-attraction term satisfies $II = \frac{p - 1}{p} \int_{\mathbb{R}^d} u_\epsilon^p w_\epsilon \, dx$. Combining both population parameters, we use H{\"o}lder's inequality with carefully chosen constants $r_1, r_2 > 1$ satisfying the analytical interpolation bounds:
\begin{equation}\label{2.13}
\frac{2d}{d+2} < pr_1 < \left(p + \frac{d-2}{d+2}\right)\frac{d}{d-2}, \quad \text{and} \quad \frac{1}{r_1} > \max\left\{ \frac{d-2}{2d}, \frac{d-2}{d} \frac{p}{p + \frac{d-2}{d+2}} \right\}.
\end{equation}
These choices are not dummy variables but specific parameters necessary to invoke the Gagliardo-Nirenberg inequality, bounding the cross-terms by the structural dissipation. Integrating the resulting differential inequality tracks uniform control for all finite indices $r \in [1, \infty)$.

<!-- lemma-4: body_md -->
\label{lem2.3}
Under the assumptions of Lemma \ref{lem2.2} and utilizing the lower bound $c_{\min}$ from Lemma \ref{lem:lower_bound}, there exists a uniform constant $C_\infty > 0$ independent of $\epsilon$ such that:
\begin{equation}\label{2.31}
\|u_\epsilon(\cdot, t)\|_\infty + \|w_\epsilon(\cdot, t)\|_\infty \le C_\infty, \quad \forall t \in (0, T).
\end{equation}
Moreover, there exists a global weak solution $(u, w)$ of \eqref{1.1}--\eqref{1.2}.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Let $p_k = 2^k p$ for $k \ge 1$. Testing \eqref{2.1} with $u_\epsilon^{p_k - 1}$ and using $\|u_\epsilon\|_{\frac{2d}{d+2}} \ge c_{\min}$, the system energy balances into:
\begin{equation}
\begin{split}
\frac{d}{dt} \int_{\mathbb{R}^d} u_\epsilon^{p_k}\,dx 
&+ \frac{4 \alpha_1 (p_k-1) c_{\min}^{\frac{4}{d+2}}}{(p_k + \frac{d-2}{d+2})^2} \int_{\mathbb{R}^d} \left|\nabla u_\epsilon^{\frac{p_k + \frac{d-2}{d+2}}{2}}\right|^2\,dx \\
& \qquad \le p_k \int_{\mathbb{R}^d} u_\epsilon^{p_k} w_\epsilon\,dx.
\end{split}
\end{equation}
By the Gagliardo-Nirenberg inequality, the right-hand side satisfies:
\begin{equation}
\int_{\mathbb{R}^d} u_\epsilon^{p_k} w_\epsilon\,dx \le C_{GN} \left( \int_{\mathbb{R}^d} \left|\nabla u_\epsilon^{\frac{p_k}{2}}\right|^2\,dx \right)^{\theta} \left( \int_{\mathbb{R}^d} u_\epsilon^{p_{k-1}}\,dx \right)^{\gamma},
\end{equation}
where $\theta \in (0,1)$. Using Young's inequality to absorb the gradient bounds into the dissipation profile yields the inductive sequence:
\begin{equation}
\sup_{t \in (0,T)} \int_{\mathbb{R}^d} u_\epsilon^{p_k}\,dx \le \left( B^k C_0 \right) \left( \sup_{t \in (0,T)} \int_{\mathbb{R}^d} u_\epsilon^{p_{k-1}}\,dx \right)^{\sigma},
\end{equation}
where $B > 1$ penalizes high powers. Taking the $p_k$-th root and taking $k \to \infty$, the geometric product convergence ensures $\|u_\epsilon(\cdot, t)\|_\infty \le C_1 \|u_\epsilon(\cdot, t)\|_{L^2}$. Since Lemma \ref{lem2.2} bounds all finite norms uniformly, $C_\infty$ is securely established. Extensibility rules out early blow-up, allowing $\epsilon \to 0$ limits to solve \eqref{1.1} globally when \eqref{1.12} holds.

<!-- lemma-5: body_md -->
\label{lem2.4}
The global weak solution from Lemma \ref{lem2.3} is also a global free energy solution.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
To establish that the limiting weak solution $(u, w)$ is a free energy solution, we construct the passage to the limit $\epsilon \to 0$ across five sequential components:
\begin{itemize}
\item[\textbf{Phase 1:}] Testing \eqref{2.1} with the formal energy derivatives yields the exact identities:
\begin{equation}\label{energy_eps_appendix}
E_{\alpha,\epsilon}[u_\epsilon(t), w_\epsilon(t)] + \int_0^t \mathcal{D}_{1,\epsilon}(u_\epsilon, v_\epsilon) \, ds + \int_0^t \mathcal{D}_{2,\epsilon}(w_\epsilon, z_\epsilon) \, ds = E_{\alpha,\epsilon}[u_0^\epsilon, w_0^\epsilon],
\end{equation}
where $\mathcal{D}_{1,\epsilon}(u_\epsilon, v_\epsilon) = \int_{\mathbb{R}^d} u_\epsilon \left| \nabla \left( \frac{\alpha_1}{2} \|u_\epsilon\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} (u_\epsilon + \epsilon)^{\frac{d-2}{d+2}} \right) - \nabla v_\epsilon \right|^2 \, dx$ and $\mathcal{D}_{2,\epsilon}(w_\epsilon, z_\epsilon)$ is defined analogously.
\item[\textbf{Phase 2:}] Uniform bounds provide sequential spaces: $u_\epsilon \to u$ a.e. in $\mathbb{R}^d \times (0,T)$, $u_\epsilon \rightharpoonup^* u$ weakly-$\ast$ in $L^\infty(0,T; L^\infty(\mathbb{R}^d))$, and $\nabla u_\epsilon^{\frac{2d}{d+2}} \rightharpoonup \nabla u^{\frac{2d}{d+2}}$ weakly in $L^2(0,T; L^2(\mathbb{R}^d))$.
\item[\textbf{Phase 3:}] Expanding $\mathcal{D}_{1,\epsilon}$ using the standard relation $|a-b|^2 \ge \frac{1}{2}|a|^2 - |b|^2$ exposes the gradient regularity structure:
\begin{align*}
\mathcal{D}_{1,\epsilon} &\ge \frac{\alpha_1^2}{8} c_{\min}^{\frac{8}{d+2}} \left(\frac{d-2}{d+2}\right)^2 \int_{\mathbb{R}^d} u_\epsilon (u_\epsilon+\epsilon)^{-\frac{8}{d+2}} |\nabla u_\epsilon|^2 \, dx - \|u_\epsilon\|_\infty \|\nabla v_\epsilon\|_{L^2}^2 \\
&\ge \frac{\alpha_1^2}{8} c_{\min}^{\frac{8}{d+2}} \left(\frac{2(d-2)}{3d-2}\right)^2 \int_{\mathbb{R}^d} \left|\nabla u_\epsilon^{\frac{3d-2}{2(d+2)}}\right|^2 \, dx - C.
\end{align*}
Integrating tracks time-uniform boundedness, confirming $u_\epsilon^{\frac{3d-2}{2(d+2)}} \rightharpoonup u^{\frac{3d-2}{2(d+2)}}$ weakly in $L^2(0,T; H^1(\mathbb{R}^d))$, which achieves requirement \eqref{1.15}.
\item[\textbf{Phase 4:}] For the vector field $\xi_\epsilon := u_\epsilon^{1/2} \nabla \left( \frac{\alpha_1}{2} \|u_\epsilon\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} (u_\epsilon + \epsilon)^{\frac{d-2}{d+2}} \right) - u_\epsilon^{1/2} \nabla v_\epsilon$, testing against any $\Psi \in [C_c^\infty(\mathbb{R}^d \times (0,t))]^d$ yields after integrating by parts:
\begin{equation*}
\begin{split}
 \langle \xi_\epsilon, \Psi \rangle &= -\frac{\alpha_1}{2} \left( \frac{d-2}{3d-2} \right) \|u_\epsilon\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} \int_0^t \int_{\mathbb{R}^d} u_\epsilon^{\frac{3d-2}{2(d+2)}} \nabla \cdot \left[ \left( \frac{u_\epsilon}{u_\epsilon + \epsilon} \right)^{\frac{4}{d+2}} \Psi \right] \, dx \, ds \\
 & \qquad - \int_0^t \int_{\mathbb{R}^d} u_\epsilon^{1/2} \nabla v_\epsilon \cdot \Psi \, dx \, ds.
 \end{split}
 \end{equation*}

By Lebesgue's Dominated Convergence Theorem, $\left( \frac{u_\epsilon}{u_\epsilon + \epsilon} \right)^{\frac{4}{d+2}} \to \mathbf{1}_{\{u > 0\}}$ strongly. Taking $\epsilon \to 0$ recovers the exact limiting distribution $\langle \xi, \Psi \rangle$. Weak sequential lower semicontinuity then establishes the dissipation integration floors:
\begin{equation}\label{lsc_diss_main}
\begin{split}
&\int_0^t \int_{\mathbb{R}^d} u \left| \nabla \left( \frac{\alpha_1}{2} \|u\|_{\frac{2d}{d+2}}^{\frac{4}{d+2}} u^{\frac{d-2}{d+2}} \right) - \nabla v \right|^2 \, dx \, ds \\
& \qquad \qquad \qquad \le \liminf_{\epsilon \to 0} \int_0^t \mathcal{D}_{1,\epsilon}(u_\epsilon, v_\epsilon) \, ds.
\end{split}
\end{equation}

\item[\textbf{Phase 5:}] Convexity forces $\|u\|_{\frac{2d}{d+2}}^2 \le \liminf \|u_\epsilon+\epsilon\|_{\frac{2d}{d+2}}^2$. Concurrently, local uniform integrability and exterior spatial tightness via bounded second moments satisfy the parameters of the Vitali Convergence Theorem, verifying $\lim \int u_\epsilon (K*w_\epsilon) dx = \int u (K*w) dx$. Taking the $\liminf$ across \eqref{energy_eps_appendix} completes the proof of the lemma.
\end{itemize}

<!-- theorem-2: body_md -->
\label{thm2.5}
There exists a $T_{\max} \in (0, \infty]$ and a free energy solution $(u, w)$ over $\mathbb{R}^d \times (0, T_{\max})$ such that either $T_{\max} = \infty$ or $T_{\max} < \infty$ and $\lim_{t \to T_{\max}} (\|u(\cdot, t)\|_\infty + \|w(\cdot, t)\|_\infty) = \infty$. Moreover, if $\alpha_1, \alpha_2$ satisfy \eqref{1.12}, then if $T_{\max} < \infty$:
\begin{equation}\label{2.34}
\lim_{t \to T_{\max}} \|u(\cdot, t)\|_{\frac{2d}{d+2}} = \infty = \lim_{t \to T_{\max}} \|w(\cdot, t)\|_{\frac{2d}{d+2}}.
\end{equation}
<!-- end theorem-2 -->
<!-- proof-5: body_md -->
Local existence follows from regularized convergence limits. To verify \eqref{2.34}, we separate $H[u,w]$ using a precision parameter $\eta > 0$:
\begin{equation}\label{2.36}
\alpha_1\|u\|_{\frac{2d}{d+2}}^2 + \alpha_2\|w\|_{\frac{2d}{d+2}}^2 \le c_d \eta \|u\|_{\frac{2d}{d+2}}^{\frac{2d}{d+2}} + c_d C_\eta \|w\|_{\frac{2d}{d+2}}^{\frac{2d}{d+2}} + C.
\end{equation}
Isolating parameters yields the locked system behavior:
\begin{equation}\label{2.37}
\|u(t)\|_{\frac{2d}{d+2}}^{\frac{2d}{d+2}} \le C \|w(t)\|_{\frac{2d}{d+2}}^{\frac{2d}{d+2}} + C, \quad \|w(t)\|_{\frac{2d}{d+2}}^{\frac{2d}{d+2}} \le C' \|u(t)\|_{\frac{2d}{d+2}}^{\frac{2d}{d+2}} + C'.
\end{equation}
These bounds lock the critical norms together. By Gagliardo-Nirenberg, any uniform control on these critical norms would prevent $L^\infty$ explosion. Thus, if $L^\infty$ blows up at $T_{\max}$, the critical $L^{\frac{2d}{d+2}}$ norms must simultaneously scale to infinity.

<!-- theorem-3: body_md -->
\label{thm2.6}
If $\alpha_1, \alpha_2$ satisfy \eqref{1.12}, then \eqref{1.1} has a global free energy solution.
<!-- end theorem-3 -->
<!-- proof-6: body_md -->
We pass to the limit within the energy structural inequality bounds to establish that the maximal existence horizon is infinite ($T_{\max} = \infty$). By Lemma \ref{lem2.4}, our solution fulfills the free energy dissipation inequality \eqref{1.16}. Dropping the non-negative dissipation integrals yields:
\begin{equation}\label{energy_bound_thm6}
\alpha_1 \|u(t)\|_{\frac{2d}{d+2}}^2 + \alpha_2 \|w(t)\|_{\frac{2d}{d+2}}^2 - c_d H[u(t), w(t)] \le E_\alpha[u_0, w_0] < \infty.
\end{equation}
We invoke the sharp Hardy--Littlewood--Sobolev inequality \eqref{2.3} with indices $\lambda=2$, $p = \frac{2d}{d+2}$, and $q = \frac{2d}{d+2}$, confirming that $\frac{1}{p}+\frac{1}{q} = 1 + \frac{2}{d}$. The interaction operator satisfies:
\begin{equation}\label{hls_bound_thm6}
H[u(t), w(t)] \le C_{HLS} \|u(t)\|_{\frac{2d}{d+2}} \|w(t)\|_{\frac{2d}{d+2}}.
\end{equation}
Substituting \eqref{hls_bound_thm6} into \eqref{energy_bound_thm6} and completing the perfect square for the individual quadratic terms gives:
\begin{equation}\label{coercive_thm6}
\begin{split}
&\left( \sqrt{\alpha_1} \|u(t)\|_{\frac{2d}{d+2}} - \sqrt{\alpha_2} \|w(t)\|_{\frac{2d}{d+2}} \right)^2 \\
&\qquad + c_d \left( \frac{2\sqrt{\alpha_1\alpha_2}}{c_d} - C_{HLS} \right) \|u(t)\|_{\frac{2d}{d+2}} \|w(t)\|_{\frac{2d}{d+2}} \\
& \qquad \qquad  \qquad \le E_\alpha[u_0, w_0].
\end{split}
\end{equation}
We evaluate the parameters across two distinct algebraic settings dictated by the primary hypothesis \eqref{1.12}:\\

 
\textbf{Case A: Subcritical regime $\left (\frac{2\sqrt{\alpha_1\alpha_2}}{c_d} > C_{HLS} \right)$} \\


The cross-coefficient is strictly positive. Dropping the non-negative perfect square from \eqref{coercive_thm6} directly forces the bounds:
\[ \|u(t)\|_{\frac{2d}{d+2}} \|w(t)\|_{\frac{2d}{d+2}} \le \frac{E_\alpha[u_0, w_0]}{c_d \left( \frac{2\sqrt{\alpha_1\alpha_2}}{c_d} - C_{HLS} \right)} =: C^*. \]
Decoupling via the internal system containment traps \eqref{2.37} establishes uniform-in-time boundaries: $\|u(t)\|_{\frac{2d}{d+2}} \le C_A$ and $\|w(t)\|_{\frac{2d}{d+2}} \le C_A$ for all $t \in [0, T_{\max})$.\\

\textbf{Case B: Critical regime $\left(\frac{2\sqrt{\alpha_1\alpha_2}}{c_d} = C_{HLS}\right)$}\\

The cross-product coefficient vanishes identically. Taking the square root across the remaining terms yields the continuous proximity control:
\[ \left| \sqrt{\alpha_1} \|u(t)\|_{\frac{2d}{d+2}} - \sqrt{\alpha_2} \|w(t)\|_{\frac{2d}{d+2}} \right| \le \sqrt{E_\alpha[u_0, w_0]} =: C_0. \]
This metric restriction isolates the tracking distance. Bound couplings \eqref{2.37} prevent unilateral divergence over $\mathbb{R}^d$, fixing uniform bounds $\|u(t)\|_{\frac{2d}{d+2}} \le C_B$ and $\|w(t)\|_{\frac{2d}{d+2}} \le C_B$.


Assume $T_{\max} < \infty$. By Theorem \ref{thm2.5}, a finite termination horizon forces critical norm explosion $\lim_{t \to T_{\max}} \|u(t)\|_{\frac{2d}{d+2}} = \infty$. However, our uniform control profile establishes $\|u(t)\|_{\frac{2d}{d+2}} \le \max\{C_A, C_B\} < \infty$, causing a direct contradiction. Thus, $T_{\max} = \infty$.

<!-- lemma-6: body_md -->
\label{lem:2ndmonetevolution}
Let $(u_0, w_0)$ satisfy \eqref{1.3}--\eqref{1.4} and let $(u, w)$ be a free energy solution of \eqref{1.1} on $[0, T_{max})$ with $T_{max}  \in (0, \infty]$. Then,
\begin{equation*}
\frac{d}{dt} I(t) = G(t), \quad \forall  t \in (0, T_{max}),
\end{equation*}
where
\begin{equation*}
I(t) := \int_{{{\mathbb{R}}}^d} |x|^2 \left( u(x,t) + w(x,t)\right) \, dx
\end{equation*}
and
\begin{equation*}
\begin{split}
G(t)  &:= C(\alpha_1, u) \int_{{{\mathbb{R}}}^d} u(x,t)^{\frac{2d}{d+2}} \, dx + C(\alpha_2, w)\int_{{{\mathbb{R}}}^d} w(x,t)^{\frac{2d}{d+2}} \, dx \\
&-2c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{u(x,t)w(y,t)}{|x-y|^{d-2}} \, dy \, dx.
\end{split}
\end{equation*}
with
\begin{equation*}
\begin{split}
& C(\alpha_1, u):= 2(d-2)\alpha_1||u||_{\frac{2d}{d+2}}^{4/{d+2}},\\
&C(\alpha_2, w) := 2(d-2)\alpha_2||w||_{\frac{2d}{d+2}}^{4/{d+2}}.
\end{split}
\end{equation*}
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
Straightforward calculations lead to
\begin{equation*}
\begin{split}
&\frac{d}{dt} I(t) = \int_{{{\mathbb{R}}}^d} |x|^2 \left(  u(x,t) + w(x,t)  \right) \, dx\\
&= \int_{{{\mathbb{R}}}^d} |x|^2 \Big[C(\alpha_1, u)\Delta \left(u^{\frac{2d}{d+2}}\right)- \nabla \cdot \left(u \nabla v\right) \\
&\qquad  \qquad \qquad + C(\alpha_2, w)\Delta \left(w^{\frac{2d}{d+2}}\right)- \nabla \cdot \left(w \nabla z\right) \Big]\, dx \\
&= 2dC(\alpha_1, u) \int_{{{\mathbb{R}}}^d} u(x,t)^{\frac{2d}{d+2}} \, dx + C(\alpha_2, w)\int_{{{\mathbb{R}}}^d} w(x,t)^{\frac{2d}{d+2}} \, dx \\
&\quad + 2 \iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \left[   x \cdot \nabla \mathcal{K}(x-y)  \right]  u(x,t)w(y,t) \, dy \, dx \\
&\qquad + 2 \iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \left[   x \cdot \nabla \mathcal{K}(x-y)  \right]  u(y,t)w(x,t) \, dy \, dx,
\end{split}
\end{equation*}
with $\mathcal{K}(x) = \frac{c_d}{|x|^{d-2}}$ and $c_d$ is the surface area of the sphere $S^{d-1}$ in ${{\mathbb{R}}}^d$.
Now,
\begin{equation*}
\begin{split}
&2\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \left[x \cdot  \nabla \mathcal{K}(x-y) \right] u(x,t) w(y,t) \,dy \, dx \\
& = -2c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{(x-y) \cdot x}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx \\
& = -2c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{|x|^2}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx \\
& \quad + 2c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{y \cdot x}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx \\
&= -c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{|x|^2}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx \\
& \quad - c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{|y|^2}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx \\
& \qquad + 2c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{y \cdot x}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx.
\end{split}
\end{equation*}
Similarly,
\begin{equation*}
\begin{split}
&2\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \left[x \cdot  \nabla \mathcal{K}(x-y) \right] u(y,t) w(x,t) \,dy \, dx \\
&= -c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{|x|^2}{|x-y|^d} u(y,t) w(x,t) \,dy \, dx \\
& \quad - c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{|y|^2}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx \\
& \qquad + 2c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{y \cdot x}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx.
\end{split}
\end{equation*}
Combining, we deduce that
\begin{equation*}
\begin{split}
&\frac{d}{dt}\int_{{{\mathbb{R}}}^d} |x|^2 \left( u(x,t)+ w(y,t) \right) \, dx = 2dC(\alpha_1, u) \int_{{{\mathbb{R}}}^d} u^{\frac{2d}{d+2}} \, dx \\
&\qquad \qquad +  2dC(\alpha_2, w) \int_{{{\mathbb{R}}}^d} w^{\frac{2d}{d+2}} \, dx \\
&-c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{|x|^2+|y|^2}{|x-y|^d} u(x,t)w(y,t) \, dy \, dx \\
&-c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{|x|^2+|y|^2}{|x-y|^d} u(y,t)w(x,t) \, dy \, dx \\
&+ 4c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{y \cdot x}{|x-y|^d} u(x,t) w(y,t) \,dy \, dx\\
& =2dC(\alpha_1, u) \int_{{{\mathbb{R}}}^d} u^{\frac{2d}{d+2}} \, dx +  2dC(\alpha_2, w) \int_{{{\mathbb{R}}}^d} w^{\frac{2d}{d+2}} \, dx \\
&\qquad -2c_d(d-2)\iint_{{{\mathbb{R}}}^d \times {{\mathbb{R}}}^d} \frac{u(x,t) w(y,t) }{|x-y|^{d-2}} \, dy \, dx.
\end{split}
\end{equation*}

<!-- theorem-4: body_md -->
\label{thm3.2}
Let $\alpha_1, \alpha_2$ be such that \eqref{1.12} is violated. Then one can find initial data $(u_0, w_0)$ satisfying \eqref{1.3}--\eqref{1.4} such that the corresponding free energy solution blows up in finite time.
<!-- end theorem-4 -->
<!-- proof-8: body_md -->
We select initial configurations consisting of appropriate scale multiples of HLS optimizers such that the initial free energy is strictly negative, which directly implies $G(0) = 2(d-2)E_\alpha[u_0, w_0] < 0$. By the global energy dissipation inequality \eqref{1.16}, the functional $G(t)$ is monotonically non-increasing across the full existence horizon, which means:
\begin{equation}
G(t) \le G(0) < 0, \quad \forall t \in [0, T_{\max}).
\end{equation}
Integrating the second moment identity from Lemma \ref{lem:2ndmonetevolution} directly yields:
\begin{equation}
I(t) = I(0) + \int_0^t G(s) \, ds \le I(0) + G(0)t.
\end{equation}
Since $G(0) < 0$, the right-hand side becomes strictly negative at any finite time $t > -\frac{I(0)}{G(0)}$. However, the second moment $I(t)$ is inherently non-negative because the population densities $u$ and $w$ remain non-negative. This contradiction guarantees that the maximum existence time $T_{\max}$ cannot extend indefinitely, verifying finite-time blow-up.
