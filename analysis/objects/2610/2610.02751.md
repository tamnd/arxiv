<!-- definition-1: body_md -->
\label{def_stable}
Let $u$ be a weak solution of \eqref{eq_main} and let $K\subset\mathbb{R}^N$ be compact. We say that $u$ is \emph{stable outside $K$} if $Q_u(\varphi)\ge 0$ for every $\varphi\in C_c^1(\mathbb{R}^N\setminus K)$. We say that $u$ has \emph{finite Morse index} if there is an integer $k$ such that no subspace of $C_c^1(\mathbb{R}^N)$ of dimension larger than $k$ consists of functions $\varphi$ with $Q_u(\varphi)<0$ apart from $\varphi=0$, and we write $i(u)$ for the least such $k$, the \emph{Morse index} of $u$.
<!-- end definition-1 -->
<!-- remark-1: body_md -->
\label{rem_fmi}
Every finite Morse index solution is stable outside a compact set. Indeed, let $u$ have finite Morse index. The dimensions of the subspaces $Z\subset C_c^1(\mathbb{R}^N)$ such that $Q_u(\varphi)<0$ for every $\varphi\in Z\setminus\{0\}$ are bounded, so one of them, say $W$, has maximal dimension, the zero subspace being admissible. Since $W$ is finite dimensional and each of its elements has compact support, there is $R>0$ such that every element of $W$ is supported in $B_R$. Suppose $Q_u(\varphi)<0$ for some $\varphi\in C_c^1(\mathbb{R}^N\setminus\overline{B_R})$. Then $w$ and $\varphi$ have disjoint supports for every $w\in W$, so $Q_u(w+t\varphi)=Q_u(w)+t^2Q_u(\varphi)<0$ whenever $(w,t)\in W\times\mathbb{R}$ and $(w,t)\ne(0,0)$. Thus $W\oplus\mathbb{R}\varphi$ contradicts the maximality of $W$. Hence $Q_u\ge0$ on $C_c^1(\mathbb{R}^N\setminus\overline{B_R})$.
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{thm_main}
Let $N>p\ge 2$ and let $q$ satisfy
\[
p^*-1<q<q_c(N,p).
\]
Then every weak solution $u\in C^1(\mathbb{R}^N)$ of \eqref{eq_main} that is stable outside a compact set satisfies $u\equiv 0$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm_decay}
Let $N>p\ge2$ and $q>\frac{N(p-1)}{N-p}$. There is $\varepsilon_*=\varepsilon_*(N,p,q)>0$ with the following property. Let $u\in C^1(\mathbb R^N)$ be a weak solution of \eqref{eq_main} such that
\[
\limsup_{|x|\to\infty}|x|^{\alpha}|u(x)|<\varepsilon_* .
\]
Then
\[
u\in\mathcal D^{1,p}(\mathbb R^N),\qquad
\int_{\mathbb R^N}|\nabla u|^p\,dx=\int_{\mathbb R^N}|u|^{q+1}\,dx<\infty,
\]
and if moreover $q\ne p^*-1$, then $u\equiv0$.
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor_full}
Let $N>p\ge 2$ and let $q$ satisfy $p-1<q<q_c(N,p)$ and $q\ne p^*-1$. Then every weak solution $u\in C^1(\mathbb{R}^N)$ of \eqref{eq_main} that is stable outside a compact set satisfies $u\equiv 0$. In particular, every finite Morse index solution of \eqref{eq_main} in that range of exponents is trivial.
<!-- end corollary-1 -->
<!-- theorem-3: body_md -->
\label{thm_sing}
Let $N>p\ge2$ and $q>p^*-1$. Then every $u_{x_0}$ is a positive weak solution of \eqref{eq_main} belonging to $W^{1,p}_{\mathrm{loc}}(\mathbb{R}^N)\cap C^\infty(\mathbb{R}^N\setminus\{x_0\})$, and the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $Q_{u_{x_0}}(\varphi)\ge0$ for every $\varphi\in C_c^1(\mathbb{R}^N\setminus\{x_0\})$;
\item $u_{x_0}$ is stable outside some compact set;
\item $q\ge q_c(N,p)$.
\end{enumerate}

<!-- theorem-4: body_md -->
\label{thm_qc_hom}
Let $N>p\ge2$ and $q>p^*-1$. Let $U\in W^{1,p}_{\mathrm{loc}}(\mathbb R^N\setminus\{0\})\cap L^{q+1}_{\mathrm{loc}}(\mathbb R^N\setminus\{0\})$ be a $(-\alpha)$-homogeneous weak solution of \eqref{eq_main} in $\mathbb R^N\setminus\{0\}$ such that $Q_U(\varphi)\ge0$ for every $\varphi\in C_c^1(\mathbb R^N\setminus\overline{B_R})$ and some $R>0$. Then
\begin{enumerate}[label=\rm(\roman*)]
\item if $q<q_c(N,p)$, then $U\equiv0$;
\item if $q=q_c(N,p)$, then $U\equiv0$ or $U(x)=\pm c_s|x|^{-\alpha}$, with $c_s$ given by \eqref{eq_cs}.
\end{enumerate}

<!-- proposition-1: body_md -->
\label{prop_radial}
Let $N>p\ge2$ and $q>p^*-1$, and let $u\in C^1(\mathbb{R}^N)$ be a positive radial solution of \eqref{eq_main} with slow decay \eqref{eq_slow}. Then
\begin{equation}\label{eq_slowexact}
\lim_{|x|\to\infty}|x|^{\alpha}u(x)=c_s
\qquad\text{and}\qquad
\lim_{|x|\to\infty}|x|^{\alpha+1}|\nabla u(x)|=\alpha c_s .
\end{equation}
If moreover $q>q_c(N,p)$, then $u$ is stable outside a compact set.

<!-- corollary-2: body_md -->
\label{cor_radialsharp}
Let $N>p\ge2$ and $q>q_c(N,p)$. Then for every $a>0$ equation \eqref{eq_main} has a positive bounded radial solution $u_a\in C^1(\mathbb{R}^N)$ with $u_a(0)=a$, and every such solution satisfies \eqref{eq_slowexact} and is stable outside a compact set. In particular the upper endpoint in Theorem \ref{thm_main} cannot be moved past $q_c(N,p)$, even within the class of positive bounded radial solutions.
<!-- end corollary-2 -->
<!-- proof-1: body_md -->
For $q>p^*-1$ and every $a>0$ there is a positive radial solution of \eqref{eq_main} on $\mathbb R^N$ with $u_a(0)=a$ and $u_a'(0)=0$, and every such solution has slow decay \eqref{eq_slow}; this is the classification of radial solutions of \eqref{eq_main} due to Bidaut-V\'eron \cite{BidautVeron1989}, in the form given by Franca \cite[Cor.~3.14]{Franca2004}. Since $q>q_c(N,p)>p^*-1$, Proposition \ref{prop_radial} applies. Such a solution is radially decreasing, because the right hand side of \eqref{eq_radialint} below is positive, so it is bounded by $a$.

<!-- theorem-5: body_md -->
\label{thm_sob_endpoint}
Let $N>p\ge2$, $q=p^*-1$, and let $u\in C^1(\mathbb R^N)$ be a weak solution of \eqref{eq_main} which is stable outside a compact set. Then
\begin{equation}\label{eq_critical_energy}
u\in\mathcal D^{1,p}(\mathbb R^N),\qquad
\int_{\mathbb R^N}|\nabla u|^p\,dx
=\int_{\mathbb R^N}|u|^{p^*}\,dx<\infty.
\end{equation}
If $u$ has finite Morse index, then the number of its nodal regions is finite and does not exceed $i(u)$. In particular, if $i(u)\le1$, then either $u\equiv0$ or, up to sign, $u=u_{\lambda,x_0}$ for some $\lambda>0$ and $x_0\in\mathbb R^N$; and the same alternative holds for a solution of one sign, without assuming finite Morse index.

<!-- theorem-6: body_md -->
\label{thm_index}
Let $2<p<N$ and $q=p^*-1$. Then $i(u_{\lambda,x_0})=1$ for every $\lambda>0$ and every $x_0\in\mathbb{R}^N$.

<!-- lemma-1: body_md -->
\label{lem_comp}
Let $R>0$, let $f\in L^\infty_{\mathrm{loc}}(\overline{B_R^c})$ where $B_R^c=\{x:|x|>R\}$, and let
$v,w\in C(\overline{B_R^c})\cap W^{1,p}_{\mathrm{loc}}(B_R^c)$ satisfy
\[
\int_{B_R^c}|\nabla v|^{p-2}\nabla v\cdot\nabla\varphi\,dx\le\int_{B_R^c}f\varphi\,dx
\le\int_{B_R^c}|\nabla w|^{p-2}\nabla w\cdot\nabla\varphi\,dx
\]
for every nonnegative $\varphi\in C_c^1(B_R^c)$. Assume $v\le w$ on $\partial B_R$ and
$\limsup_{|x|\to\infty}\big(v(x)-w(x)\big)\le 0$. Then $v\le w$ in $B_R^c$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Fix $\lambda>0$ and set $D_\lambda=\{x\in B_R^c: v(x)-w(x)>\lambda\}$. Since $v-w$ is continuous on $\overline{B_R^c}$ and $v-w\le 0$ on $\partial B_R$, the set $D_\lambda$ has positive distance from $\partial B_R$. Since $\limsup_{|x|\to\infty}(v-w)\le 0$, the set $D_\lambda$ is bounded. Hence $(v-w-\lambda)^+$ belongs to $W^{1,p}(B_R^c)$ and has compact support in $B_R^c$, so it is an admissible test function by density. Subtracting the two inequalities and testing with it gives
\[
\int_{D_\lambda}\left(|\nabla v|^{p-2}\nabla v-|\nabla w|^{p-2}\nabla w\right)\cdot(\nabla v-\nabla w)\,dx\le 0 .
\]
The map $\xi\mapsto|\xi|^{p-2}\xi$ is strictly monotone on $\mathbb{R}^N$, so the integrand is nonnegative and vanishes only where $\nabla v=\nabla w$. Therefore $\nabla(v-w)=0$ almost everywhere in $D_\lambda$, so $v-w$ is constant on each connected component of $D_\lambda$. On the boundary of any such component one has $v-w=\lambda$ by continuity, while $v-w>\lambda$ inside. Hence $D_\lambda=\emptyset$. Letting $\lambda\to 0$ finishes the proof.
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lem_abs}
Let $u\in C^1(\mathbb{R}^N)$ be a weak solution of \eqref{eq_main}. Then $|u|$ is a weak subsolution of $-\Delta_p z=|u|^q$, that is,
\[
\int_{\mathbb{R}^N}\big|\nabla|u|\big|^{p-2}\nabla|u|\cdot\nabla\varphi\,dx\le\int_{\mathbb{R}^N}|u|^q\varphi\,dx
\qquad\text{for every nonnegative }\varphi\in C_c^1(\mathbb{R}^N).
\]
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
For $\varepsilon>0$ put $\theta_\varepsilon(s)=s(s^2+\varepsilon^2)^{-1/2}$, so that $\theta_\varepsilon$ is smooth, bounded by $1$ in absolute value, and $\theta_\varepsilon'\ge 0$. Let $\varphi\in C_c^1(\mathbb{R}^N)$ be nonnegative. The function $\theta_\varepsilon(u)\varphi$ is Lipschitz with compact support, hence admissible, and testing \eqref{eq_main} with it gives
\[
\int_{\mathbb{R}^N}|\nabla u|^{p-2}\theta_\varepsilon(u)\,\nabla u\cdot\nabla\varphi\,dx
+\int_{\mathbb{R}^N}|\nabla u|^{p}\,\theta_\varepsilon'(u)\,\varphi\,dx
=\int_{\mathbb{R}^N}|u|^{q-1}u\,\theta_\varepsilon(u)\,\varphi\,dx .
\]
The second term on the left is nonnegative, and $|u|^{q-1}u\,\theta_\varepsilon(u)\le|u|^q$ pointwise. Hence
\[
\int_{\mathbb{R}^N}|\nabla u|^{p-2}\theta_\varepsilon(u)\,\nabla u\cdot\nabla\varphi\,dx\le\int_{\mathbb{R}^N}|u|^q\varphi\,dx .
\]
As $\varepsilon\to 0$ we have $\theta_\varepsilon(u)\to \operatorname{sgn}(u)$ pointwise, with the convention $\operatorname{sgn}(0)=0$. Since $\nabla u=0$ almost everywhere on $\{u=0\}$ and $\nabla|u|=\operatorname{sgn}(u)\nabla u$ almost everywhere, the integrand converges almost everywhere to $\big|\nabla|u|\big|^{p-2}\nabla|u|\cdot\nabla\varphi$ and is dominated by $|\nabla u|^{p-1}|\nabla\varphi|$, which is integrable on the support of $\varphi$. Dominated convergence gives the claim.
<!-- end proof-3 -->
<!-- lemma-3: body_md -->
\label{lem_serrin}
Let $\Omega=B_4\setminus \overline{B_1}$ and $\Omega'=B_3\setminus\overline{B_2}$, let $p>1$ and let $k>N/p$. For every $M>0$ there is a constant $C=C(N,p,k,M)$ with the following property. If $c\in L^k(\Omega)$ with $\|c\|_{L^k(\Omega)}\le M$ and if $v\in W^{1,p}(\Omega)\cap C(\Omega)$ satisfies
\[
\int_{\Omega}|\nabla v|^{p-2}\nabla v\cdot\nabla\varphi\,dx=\int_{\Omega}c(x)|v|^{p-2}v\,\varphi\,dx
\qquad\text{for every }\varphi\in C_c^1(\Omega),
\]
then
\[
\sup_{\Omega'}|v|\le C\,\|v\|_{L^{p}(\Omega)} .
\]
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
\label{lem_grad}
Let $p>1$. There is $C=C(N,p)$ such that for every $v\in C^1(B_2)$ solving $-\Delta_p v=g$ weakly in $B_2$ with $g\in L^\infty(B_2)$ one has
\[
\sup_{B_1}|\nabla v|\le C\left(\|v\|_{L^\infty(B_2)}+\|g\|_{L^\infty(B_2)}^{\frac{1}{p-1}}\right).
\]
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Set $\lambda=\|v\|_{L^\infty(B_2)}+\|g\|_{L^\infty(B_2)}^{1/(p-1)}$, which we may assume positive, and put $V=v/\lambda$. Then $-\Delta_p V=\lambda^{1-p}g$ weakly in $B_2$, with $\|V\|_{L^\infty(B_2)}\le 1$ and $\|\lambda^{1-p}g\|_{L^\infty(B_2)}\le 1$. The standard interior $C^{1,\beta}$ estimates for degenerate quasilinear equations, as developed for instance by DiBenedetto \cite{DiBenedetto1983} and Tolksdorf \cite{Tolksdorf1984}, give $\sup_{B_1}|\nabla V|\le C(N,p)$. Multiplying by $\lambda$ gives the claim.

<!-- lemma-5: body_md -->
\label{lem_nem}
Let $\Omega\subseteq\mathbb{R}^N$ be open.
\begin{enumerate}[label=\rm(\roman*)]
\item For every $q>1$ the map $z\mapsto|z|^{q-1}$ acts continuously from $L^{q+1}(\Omega)$ into $L^{\frac{q+1}{q-1}}(\Omega)$.
\item Let $p>2$. Then $\xi\mapsto|\xi|^{p-2}\xi$ is of class $C^1$ on $\mathbb{R}^N$ with differential $\mathcal M$,
\begin{equation}\label{eq_Mbound}
|\mathcal M(\xi)|\le(p-1)|\xi|^{p-2}\qquad\text{for every }\xi\in\mathbb{R}^N ,
\end{equation}
and $\mathcal M$ acts continuously from $L^p(\Omega;\mathbb{R}^N)$ into $L^{\frac{p}{p-2}}(\Omega;\mathbb{R}^{N\times N})$.
\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
We first dispose of the pointwise statements in (ii). Away from the origin $\xi\mapsto|\xi|^{p-2}\xi$ is differentiable with differential $\mathcal M(\xi)$, and for a unit vector $e$ we have $|\mathcal M(\xi)e|\le|\xi|^{p-2}+(p-2)|\xi|^{p-2}$, which is \eqref{eq_Mbound}. At the origin, $\big||\xi|^{p-2}\xi\big|=|\xi|^{p-1}=o(|\xi|)$ because $p>2$, so the map is differentiable there with differential $0=\mathcal M(0)$; and the right hand side of \eqref{eq_Mbound} tends to $0$ as $\xi\to0$, so $\mathcal M$ is continuous at the origin. Hence the map is of class $C^1$ on $\mathbb{R}^N$.

Both superposition statements are instances of the following observation: if $F$ is a continuous map from $\mathbb{R}^d$ into a finite dimensional normed space with $|F(z)|\le C_F|z|^{a}$ for every $z$, and if $0<a\le b<\infty$ with $b\ge1$, then $F$ acts continuously from $L^{b}(\Omega;\mathbb{R}^d)$ into $L^{b/a}(\Omega)$. The growth bound gives $|F(z)|^{b/a}\le C_F^{b/a}|z|^{b}$, so $F$ does map $L^b$ into $L^{b/a}$. Let $z_j\to z$ in $L^b$. It is enough to produce, in every subsequence, a further subsequence along which $F(z_j)\to F(z)$ in $L^{b/a}$. Passing to a subsequence we may assume $\|z_{j+1}-z_j\|_{L^b}\le2^{-j}$, so that
\[
h:=|z_1|+\sum_{j\ge1}|z_{j+1}-z_j|
\]
satisfies $\|h\|_{L^b}\le\|z_1\|_{L^b}+1$ and $|z_j|\le h$ for every $j$. The series also converges almost everywhere, so $z_j\to z$ almost everywhere along this subsequence and $|z|\le h$ almost everywhere. By the continuity of $F$ we get $|F(z_j)-F(z)|^{b/a}\to0$ almost everywhere, while
\[
\big|F(z_j)-F(z)\big|^{b/a}\le\big(2C_Fh^{a}\big)^{b/a}=(2C_F)^{b/a}h^{b}\in L^1(\Omega),
\]
and dominated convergence concludes. Part (i) is the case $(a,b)=(q-1,q+1)$, $F(z)=|z|^{q-1}$, and the last assertion of (ii) is the case $(a,b)=(p-2,p)$, $F=\mathcal M$, the growth bound being \eqref{eq_Mbound}.

<!-- lemma-6: body_md -->
\label{lem_stab}
Let $\delta\in[1,\delta_0)$ and let $\gamma_0\ge \frac{p(q+\delta)}{q-p+1}$. There is $C=C(N,p,q,\delta,\gamma_0)$ such that
\begin{equation}
\int_{\mathbb{R}^N}|u|^{q+\delta}\eta^{\gamma_0}\,dx+\int_{\mathbb{R}^N}|u|^{\delta-1}|\nabla u|^p\eta^{\gamma_0}\,dx
\le C\int_{\mathbb{R}^N}|\nabla\eta|^{\frac{p(q+\delta)}{q-p+1}}\,dx
\end{equation}
for every $\eta\in C_c^\infty(\mathbb{R}^N\setminus K)$ with $0\le\eta\le 1$.
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Since $\delta\ge 1$ and $u\in C^1(\mathbb{R}^N)$, the function $|u|^{\delta-1}u\,\eta^{\gamma_0}$ belongs to $W^{1,p}(\mathbb{R}^N)$ and has compact support in $\mathbb{R}^N\setminus K$. Every integral written below is finite, because $u$ and $\nabla u$ are continuous and $\eta$ has compact support, so each of the absorption steps is legitimate. Testing \eqref{eq_main} with it gives
\begin{equation}\label{eq_test1}
\int|u|^{q+\delta}\eta^{\gamma_0}
=\delta\int|u|^{\delta-1}|\nabla u|^p\eta^{\gamma_0}
+\gamma_0\int|u|^{\delta-1}u\,|\nabla u|^{p-2}\nabla u\cdot\nabla\eta\ \eta^{\gamma_0-1},
\end{equation}
where all integrals are over $\mathbb{R}^N$. For $\varepsilon\in(0,1)$, Young's inequality with exponents $\frac{p}{p-1}$ and $p$ applied to the factorization
\[
|u|^{\delta-1}|u|\,|\nabla u|^{p-1}\eta^{\gamma_0-1}|\nabla\eta|
=\left(|u|^{\frac{(\delta-1)(p-1)}{p}}|\nabla u|^{p-1}\eta^{\frac{\gamma_0(p-1)}{p}}\right)
\left(|u|^{\frac{\delta-1}{p}+1}\eta^{\frac{\gamma_0}{p}-1}|\nabla\eta|\right)
\]
yields
\begin{equation}\label{eq_young1}
\gamma_0\left|\int|u|^{\delta-1}u\,|\nabla u|^{p-2}\nabla u\cdot\nabla\eta\ \eta^{\gamma_0-1}\right|
\le \varepsilon\int|u|^{\delta-1}|\nabla u|^p\eta^{\gamma_0}
+C_\varepsilon\int|u|^{\delta+p-1}\eta^{\gamma_0-p}|\nabla\eta|^p .
\end{equation}
Next we use stability. The function $\varphi=|u|^{\frac{\delta-1}{2}}u\,\eta^{\gamma_0/2}$ belongs to $C_c^1(\mathbb{R}^N\setminus K)$, so $Q_u(\varphi)\ge 0$. Since $p\ge 2$ we have $|\nabla u|^{p-4}(\nabla u\cdot\nabla\varphi)^2\le|\nabla u|^{p-2}|\nabla\varphi|^2$ pointwise, whence
\[
q\int|u|^{q+\delta}\eta^{\gamma_0}\le (p-1)\int|\nabla u|^{p-2}|\nabla\varphi|^2 .
\]
Expanding $\nabla\varphi$ and applying Young's inequality in the form $|a+b|^2\le(1+\varepsilon)a^2+(1+\varepsilon^{-1})b^2$,
\[
|\nabla\varphi|^2\le(1+\varepsilon)\frac{(\delta+1)^2}{4}|u|^{\delta-1}|\nabla u|^2\eta^{\gamma_0}
+C_\varepsilon|u|^{\delta+1}\eta^{\gamma_0-2}|\nabla\eta|^2 ,
\]
so that
\begin{equation}\label{eq_stab1}
q\int|u|^{q+\delta}\eta^{\gamma_0}
\le (p-1)(1+\varepsilon)\frac{(\delta+1)^2}{4}\int|u|^{\delta-1}|\nabla u|^p\eta^{\gamma_0}
+C_\varepsilon\int|u|^{\delta+1}|\nabla u|^{p-2}\eta^{\gamma_0-2}|\nabla\eta|^2 .
\end{equation}
If $p>2$, Young's inequality with exponents $\frac{p}{p-2}$ and $\frac p2$ applied to
\[
|u|^{\delta+1}|\nabla u|^{p-2}\eta^{\gamma_0-2}|\nabla\eta|^2
=\left(|u|^{\frac{(\delta-1)(p-2)}{p}}|\nabla u|^{p-2}\eta^{\frac{\gamma_0(p-2)}{p}}\right)
\left(|u|^{\delta+1-\frac{(\delta-1)(p-2)}{p}}\eta^{\frac{2\gamma_0}{p}-2}|\nabla\eta|^2\right)
\]
gives, since $\frac p2\left(\delta+1-\frac{(\delta-1)(p-2)}{p}\right)=\delta+p-1$,
\begin{equation}\label{eq_young2}
C_\varepsilon\int|u|^{\delta+1}|\nabla u|^{p-2}\eta^{\gamma_0-2}|\nabla\eta|^2
\le\varepsilon\int|u|^{\delta-1}|\nabla u|^p\eta^{\gamma_0}
+C\int|u|^{\delta+p-1}\eta^{\gamma_0-p}|\nabla\eta|^p ,
\end{equation}
where $C$ depends on $\varepsilon$. For $p=2$ the inequality \eqref{eq_young2} holds trivially with the first term on the right omitted.

From \eqref{eq_test1} and \eqref{eq_young1},
\begin{equation}\label{eq_grad_control}
\int|u|^{\delta-1}|\nabla u|^p\eta^{\gamma_0}
\le\frac{1}{\delta-\varepsilon}\left(\int|u|^{q+\delta}\eta^{\gamma_0}
+C_\varepsilon\int|u|^{\delta+p-1}\eta^{\gamma_0-p}|\nabla\eta|^p\right).
\end{equation}
Substituting \eqref{eq_young2} and \eqref{eq_grad_control} into \eqref{eq_stab1} we obtain
\[
\left(q-\frac{(p-1)(1+\varepsilon)\frac{(\delta+1)^2}{4}+\varepsilon}{\delta-\varepsilon}\right)
\int|u|^{q+\delta}\eta^{\gamma_0}\le C\int|u|^{\delta+p-1}\eta^{\gamma_0-p}|\nabla\eta|^p .
\]
The condition $q\delta>(p-1)\frac{(\delta+1)^2}{4}$ is equivalent to $(p-1)\delta^2-\big(4q-2(p-1)\big)\delta+(p-1)<0$, whose roots are
\[
\delta_\pm=\frac{2q-(p-1)\pm 2\sqrt{q(q-p+1)}}{p-1},
\]
so it holds precisely for $\delta\in(\delta_-,\delta_+)=(\delta_-,\delta_0)$. Since $q>p-1$ the value $\delta=1$ satisfies it strictly, so $\delta_-<1$ and the whole interval $[1,\delta_0)$ is allowed. Write $\kappa_\delta=q-\frac{(p-1)(\delta+1)^2}{4\delta}$, which is positive by the previous line. The coefficient on the left is a continuous function of $\varepsilon$ on $[0,\delta/2]$ whose value at $\varepsilon=0$ is $\kappa_\delta$, so there is $\varepsilon_\delta\in(0,\delta/2]$ such that this coefficient is at least $\kappa_\delta/2$ for every $\varepsilon\in(0,\varepsilon_\delta]$. Fixing such an $\varepsilon$ and using \eqref{eq_grad_control} we get
\begin{equation}\label{eq_almost}
\int|u|^{q+\delta}\eta^{\gamma_0}+\int|u|^{\delta-1}|\nabla u|^p\eta^{\gamma_0}
\le C\int|u|^{\delta+p-1}\eta^{\gamma_0-p}|\nabla\eta|^p .
\end{equation}
Finally, Young's inequality with exponents $\frac{q+\delta}{\delta+p-1}$ and $\frac{q+\delta}{q-p+1}$ applied to the factorization
\[
|u|^{\delta+p-1}\eta^{\gamma_0-p}|\nabla\eta|^p
=\left(|u|^{\delta+p-1}\eta^{\frac{\gamma_0(\delta+p-1)}{q+\delta}}\right)
\left(\eta^{\gamma_0-p-\frac{\gamma_0(\delta+p-1)}{q+\delta}}|\nabla\eta|^p\right)
\]
gives
\[
C\int|u|^{\delta+p-1}\eta^{\gamma_0-p}|\nabla\eta|^p
\le\frac12\int|u|^{q+\delta}\eta^{\gamma_0}
+C\int\eta^{\gamma_0-\frac{p(q+\delta)}{q-p+1}}|\nabla\eta|^{\frac{p(q+\delta)}{q-p+1}} .
\]
The exponent of $\eta$ in the last integral equals $\gamma_0-\frac{p(q+\delta)}{q-p+1}\ge 0$ by the choice of $\gamma_0$, so that factor is bounded by $1$. Absorbing the first term on the right into the left hand side of \eqref{eq_almost} completes the proof.

<!-- lemma-7: body_md -->
\label{lem_qc}
Let $N>p>1$ and define, for $q>p-1$,
\[
G(q)=q+\delta_0(q)-\frac{N(q-p+1)}{p},
\qquad \delta_0(q)=\frac{2q-(p-1)+2\sqrt{q(q-p+1)}}{p-1}.
\]
Then $G$ is strictly concave on $(p-1,+\infty)$ and $\lim_{q\to(p-1)^+}G(q)=p>0$. Moreover:
\begin{enumerate}[label=(\roman*)]
\item if $(p-1)N\le p(p+3)$, then $G(q)>0$ for every $q>p-1$;
\item if $(p-1)N> p(p+3)$, then $G$ has a unique zero in $(p-1,+\infty)$, it is given by the formula \eqref{eq_qc} for $q_c(N,p)$, and $G(q)>0$ for $p-1<q<q_c(N,p)$ while $G(q)<0$ for $q>q_c(N,p)$.
\end{enumerate}
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
Differentiating,
\[
G'(q)=1+\frac{2+h(q)}{p-1}-\frac{N}{p},
\qquad h(q)=\frac{2q-(p-1)}{\sqrt{q(q-p+1)}} .
\]
Since
\[
h(q)^2=\frac{4q^2-4(p-1)q+(p-1)^2}{q^2-(p-1)q}=4+\frac{(p-1)^2}{q^2-(p-1)q}
\]
and $q\mapsto q^2-(p-1)q$ is strictly increasing on $(p-1,+\infty)$, the function $h$ is strictly decreasing there, with $h(q)\to+\infty$ as $q\to(p-1)^+$ and $h(q)\to 2$ as $q\to+\infty$. Hence $G'$ is strictly decreasing, so $G$ is strictly concave, and
\[
\lim_{q\to+\infty}G'(q)=1+\frac{4}{p-1}-\frac Np=\frac{p+3}{p-1}-\frac Np .
\]
Also $G(p-1^+)=(p-1)+\frac{2(p-1)-(p-1)}{p-1}-0=p$.

If $(p-1)N\le p(p+3)$, then $\lim_{q\to\infty}G'\ge 0$, and since $G'$ is strictly decreasing with this limit we get $G'>0$ on $(p-1,+\infty)$. Hence $G$ is increasing and $G>p>0$, which is (i).

If $(p-1)N>p(p+3)$, then $\lim_{q\to\infty}G'<0$, so $G(q)\to-\infty$. Being concave with $G(p-1^+)=p>0$, the function $G$ has exactly one zero $q_c$ in $(p-1,+\infty)$, is positive to its left and negative to its right.

It remains to identify $q_c$. Put $B=(p-1)N-p$, $a=B-p^2$ and $b=(p-1)B$. In case (ii) we have $(p-1)N>p(p+3)>p$, so $B>0$ and $b>0$; also $a-2p=(p-1)N-p(p+3)>0$, so $a>2p>0$. A computation gives
\[
(p-1)\left(\frac{N(q-p+1)}{p}-q\right)-\big(2q-(p-1)\big)=\frac{aq-b}{p},
\]
so $G(q)=0$ is equivalent to
\begin{equation}\label{eq_root}
\frac{aq-b}{p}=2\sqrt{q(q-p+1)} .
\end{equation}
In particular $aq-b\ge 0$ at the zero, and squaring \eqref{eq_root} shows that $q_c$ is a root of
\[
P(q):=(a^2-4p^2)q^2+\big(4p^2(p-1)-2ab\big)q+b^2 .
\]
Here
\begin{align*}
a^2-4p^2
&=(a-2p)(a+2p)\\
&=\big((p-1)N-p(p+3)\big)\big((p-1)N-p(p-1)\big)\\
&=(p-1)(N-p)\big((p-1)N-p(p+3)\big)>0.
\end{align*}
and, using $a=B-p^2$ and $b=(p-1)B$,
\[
\left(\frac{4p^2(p-1)-2ab}{2}\right)^2-(a^2-4p^2)b^2
=4p^2\left(b^2-(p-1)ab+p^2(p-1)^2\right)
=4p^2(p-1)^2\left(B^2-aB+p^2\right).
\]
Since $a=B-p^2$ we have $B^2-aB+p^2=p^2B+p^2=p^2\big((p-1)N-p+1\big)=p^2(p-1)(N-1)$, so the quantity above equals $4p^4(p-1)^3(N-1)$ and its square root is $2p^2(p-1)\sqrt{(p-1)(N-1)}$. Also $ab=(p-1)(B^2-p^2B)$, hence
\[
\frac{2ab-4p^2(p-1)}{2}+2p^2(p-1)\sqrt{(p-1)(N-1)}
=(p-1)\left(B^2-p^2B-2p^2+2p^2\sqrt{(p-1)(N-1)}\right).
\]
Therefore the larger root of $P$ is
\[
\frac{B^2-p^2B-2p^2+2p^2\sqrt{(p-1)(N-1)}}{(N-p)\big((p-1)N-p(p+3)\big)} .
\]
Substituting $B=(p-1)N-p$ in $-p^2B-2p^2=-p^2(p-1)N+p^3-2p^2=-p^2(p-1)N+p^2(p-2)$ shows that this expression coincides with the right hand side of \eqref{eq_qc}. It is the larger root that corresponds to $q_c$. Indeed, \eqref{eq_root} gives $aq_c-b=2p\sqrt{q_c(q_c-p+1)}>0$, so $q_c>b/a$, while a direct computation using $(p-1)a-b=(p-1)(B-p^2)-(p-1)B=-(p-1)p^2$ gives
\[
P\!\left(\frac ba\right)=\frac{4p^2b}{a^2}\big((p-1)a-b\big)=-\frac{4p^4(p-1)b}{a^2}<0 .
\]
Since $P$ has positive leading coefficient, the point $b/a$ lies strictly between the two roots of $P$, so any root of $P$ larger than $b/a$ is the larger one. This proves (ii).

<!-- proposition-2: body_md -->
\label{prop_decay}
Let $N>p\ge 2$ and $p^*-1\le q<q_c(N,p)$, and let $u\in C^1(\mathbb{R}^N)$ be a weak solution of \eqref{eq_main} that is stable outside a compact set. Then
\begin{equation}\label{eq_decay}
\lim_{|x|\to+\infty}|x|^{\alpha}|u(x)|=0
\qquad\text{and}\qquad
\lim_{|x|\to+\infty}|x|^{\alpha+1}|\nabla u(x)|=0 .
\end{equation}
<!-- end proposition-2 -->
<!-- proof-9: body_md -->
We keep the notation of Section \ref{sec_stab}: $K$ is a compact set outside which $u$ is stable, and $R_0\ge1$ satisfies $K\subset B_{R_0}$. Put
\[
s_1=\frac{N(q-p+1)}{p},\qquad \delta_1=s_1-q .
\]
The inequality $\delta_1\ge 1$ is equivalent to $N(q-p+1)\ge p(q+1)$, that is to $q(N-p)\ge N(p-1)+p$, that is to $q\ge p^*-1$. The inequality $\delta_1<\delta_0$ is exactly $G(q)>0$ in the notation of Lemma \ref{lem_qc}, hence it holds because $q<q_c(N,p)$. Therefore $\delta_1\in[1,\delta_0)$, and the corresponding exponent in Corollary \ref{cor_annuli} is
\[
N-\frac{p(q+\delta_1)}{q-p+1}=N-\frac{p s_1}{q-p+1}=0 .
\]
By \eqref{eq_ball},
\begin{equation}\label{eq_Ls1}
\int_{\mathbb{R}^N\setminus B_{2R_0}}|u|^{s_1}\,dx\le C<+\infty .
\end{equation}
Since $\delta_1<\delta_0$ we may fix $\bar\delta\in(\delta_1,\delta_0)$ and set
\[
\eta_0=\frac{p(q+\bar\delta)}{q-p+1}-N>0 .
\]
By \eqref{eq_ann}, for every $r\ge 8R_0$,
\begin{equation}\label{eq_Lsbar}
\int_{B_{4r}\setminus B_r}|u|^{q+\bar\delta}\,dx\le C\,r^{-\eta_0} .
\end{equation}

For $r\ge 8R_0$ define $u_r(x)=r^{\alpha}u(rx)$ for $x\in B_4\setminus\overline{B_1}$. A change of variables shows that $u_r\in C^1$ is a weak solution of $-\Delta_p u_r=|u_r|^{q-1}u_r$ there. Since $\alpha(q+\bar\delta)=\frac{p(q+\bar\delta)}{q-p+1}=N+\eta_0$ and $\alpha s_1=N$, we get from \eqref{eq_Lsbar} and \eqref{eq_Ls1}
\begin{equation}\label{eq_scaled}
\int_{B_4\setminus B_1}|u_r|^{q+\bar\delta}\,dx=r^{\alpha(q+\bar\delta)-N}\int_{B_{4r}\setminus B_r}|u|^{q+\bar\delta}\,dx\le C,
\end{equation}
\begin{equation}
\int_{B_4\setminus B_1}|u_r|^{s_1}\,dx=\int_{B_{4r}\setminus B_r}|u|^{s_1}\,dx=:\omega(r)^{s_1},
\end{equation}
and $\omega(r)\to 0$ as $r\to+\infty$ because the integral in \eqref{eq_Ls1} converges and the annuli $B_{4r}\setminus B_r$ escape to infinity.

Write the equation for $u_r$ as $-\Delta_p u_r=c_r(x)|u_r|^{p-2}u_r$ with $c_r=|u_r|^{q-p+1}$, and set
\[
k=\frac{q+\bar\delta}{q-p+1}>\frac{q+\delta_1}{q-p+1}=\frac{s_1}{q-p+1}=\frac{N}{p} .
\]
Then $\|c_r\|_{L^k(B_4\setminus B_1)}^{k}=\int_{B_4\setminus B_1}|u_r|^{q+\bar\delta}\le C$ by \eqref{eq_scaled}, uniformly in $r$. Lemma \ref{lem_serrin} therefore gives a constant $C$ independent of $r$ with $\sup_{B_3\setminus B_2}|u_r|\le C\|u_r\|_{L^p(B_4\setminus B_1)}$. Since $q\ge p^*-1$ we have $s_1\ge\frac Np\cdot\frac{p^2}{N-p}=p^*>p$, so H\"older's inequality on the bounded set $B_4\setminus B_1$ turns this into
\[
\sup_{B_3\setminus B_2}|u_r|\le C\,\|u_r\|_{L^{s_1}(B_4\setminus B_1)}=C\,\omega(r) .
\]
Undoing the scaling, $\sup_{2r\le|x|\le 3r}|x|^{\alpha}|u(x)|\le 3^{\alpha}\sup_{B_3\setminus B_2}|u_r|\le C\omega(r)\to0$, and since the annuli $\{2r\le|x|\le3r\}$ cover a neighborhood of infinity as $r$ ranges over $[8R_0,\infty)$, the first limit in \eqref{eq_decay} follows.

For the gradient, set $M_r=\sup_{2r\le|x|\le 3r}|u|$, so that $r^{\alpha}M_r\to0$. Let $x_0$ with $|x_0|=\frac52 r$ and apply Lemma \ref{lem_grad} to $v(y)=u(x_0+\frac r4 y)$ on $B_2$, which solves $-\Delta_p v=(r/4)^p|v|^{q-1}v$. We obtain
\[
\frac r4\,|\nabla u(x_0)|\le C\left(M_r+\big((r/4)^pM_r^q\big)^{\frac1{p-1}}\right).
\]
Since $r^{\alpha}M_r\to 0$ we have
\[
r^pM_r^{q-p+1}=\big(r^{\alpha}M_r\big)^{q-p+1}\longrightarrow0.
\]
Hence, for $r$ large,
\[
\big(r^pM_r^q\big)^{1/(p-1)}
=\big(r^pM_r^{q-p+1}\big)^{1/(p-1)}M_r\le M_r.
\]
Therefore $|\nabla u(x_0)|\le CM_r/r$, and $|x_0|^{\alpha+1}|\nabla u(x_0)|\le C r^{\alpha}M_r\to0$. This proves the second limit in \eqref{eq_decay}.

<!-- lemma-8: body_md -->
\label{lem_super}
Let $0<\alpha<\gamma$ with $\gamma=\frac{N-p}{p-1}$, let $A,B>0$ and set $w(x)=A|x|^{-\alpha}+B|x|^{-\gamma}$ on $\mathbb{R}^N\setminus\{0\}$. Then $w$ is smooth there and
\begin{equation}\label{eq_superid}
-\Delta_p w(x)=\left(1+\frac{\gamma B}{\alpha A}|x|^{\alpha-\gamma}\right)^{p-2}
K\,A^{p-1}\,|x|^{-1-(\alpha+1)(p-1)},
\qquad K:=\alpha^{p-1}(p-1)(\gamma-\alpha)>0 .
\end{equation}
If moreover $p\ge2$ and $\alpha=\frac{p}{q-p+1}$, then $1+(\alpha+1)(p-1)=\alpha q$ and
\begin{equation}\label{eq_superineq}
-\Delta_p w(x)\ge K\,A^{p-1}\,|x|^{-\alpha q}\qquad\text{for } x\ne0 .
\end{equation}
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
Write $r=|x|$. Then $w'(r)<0$ and
\[
-w'(r)=\alpha A r^{-\alpha-1}+\gamma B r^{-\gamma-1}=\alpha A r^{-\alpha-1}(1+t(r)),
\qquad t(r)=\frac{\gamma B}{\alpha A}r^{\alpha-\gamma}.
\]
For radial functions, $-\Delta_p w=r^{1-N}\frac{d}{dr}\big[r^{N-1}(-w')^{p-1}\big]$. Set
\[
m=N-1-(\alpha+1)(p-1).
\]
Since $(p-1)\gamma=N-p$, we have $m=N-1-(p-1)-\alpha(p-1)=(N-p)-\alpha(p-1)=(p-1)(\gamma-\alpha)>0$. Next,
\[
r^{N-1}(-w')^{p-1}=(\alpha A)^{p-1}r^{m}(1+t)^{p-1},
\qquad t'(r)=(\alpha-\gamma)\frac{t}{r}=-\frac{m}{p-1}\cdot\frac tr .
\]
Hence
\[
\frac{d}{dr}\Big[r^{N-1}(-w')^{p-1}\Big]
=(\alpha A)^{p-1}\left[m r^{m-1}(1+t)^{p-1}+r^{m}(p-1)(1+t)^{p-2}t'\right]
=m(\alpha A)^{p-1}r^{m-1}(1+t)^{p-2},
\]
where we used $(1+t)^{p-1}-\frac{m}{p-1}\cdot\frac{(p-1)t}{m}(1+t)^{p-2}=(1+t)^{p-2}\big[(1+t)-t\big]=(1+t)^{p-2}$. Since $m-N=-1-(\alpha+1)(p-1)$ and $m=(p-1)(\gamma-\alpha)$, this gives \eqref{eq_superid}.

<!-- proposition-3: body_md -->
\label{prop_boot}
Let $N>p\ge2$ and $\frac{N(p-1)}{N-p}<q$, and set
\[
\kappa=\frac{q}{p-1}>1,\qquad \theta=\frac{\log 2}{\log\kappa} .
\]
There is $\varepsilon_*=\varepsilon_*(N,p,q)>0$ with the following property. If $u\in C^1(\mathbb{R}^N)$ is a weak solution of \eqref{eq_main} with
\begin{equation}\label{eq_smalldecay}
\limsup_{|x|\to\infty}|x|^\alpha|u(x)|<\varepsilon_* ,
\end{equation}
then there are $C>0$ and $R_*>1$ such that
\begin{equation}\label{eq_pharmonic}
|u(x)|\le C\,|x|^{-\gamma}\left(\log|x|\right)^{\theta}\qquad\text{for } |x|\ge R_* .
\end{equation}
<!-- end proposition-3 -->
<!-- proof-11: body_md -->
Let $K=\alpha^{p-1}(p-1)(\gamma-\alpha)$ be the constant of Lemma \ref{lem_super}, positive by \eqref{eq_alphagamma}, and put
\[
c_1=\log\left(2K^{-\frac{1}{p-1}}\right),\qquad \bar c=\frac{c_1}{\kappa-1} .
\]
The two conditions
\begin{equation}\label{eq_eps0}
2K^{-\frac1{p-1}}\varepsilon^{\kappa-1}\le\frac12
\qquad\text{and}\qquad -\log\varepsilon\ge \bar c+1
\end{equation}
both hold for all small $\varepsilon>0$, and $K$, $\kappa$ and $\bar c$ depend only on $N$, $p$ and $q$; let $\varepsilon_*=\varepsilon_*(N,p,q)\in(0,1)$ be a value such that \eqref{eq_eps0} holds for every $\varepsilon\in(0,\varepsilon_*]$. By \eqref{eq_smalldecay} we may fix $\varepsilon_0\in(0,\varepsilon_*]$ with \eqref{eq_eps0} and then $R_0\ge3$ with
\[
|u(x)|\le\varepsilon_0|x|^{-\alpha}\qquad\text{for } |x|\ge R_0 .
\] Define recursively
\begin{equation}\label{eq_rec}
A_n=K^{-\frac{1}{p-1}}\varepsilon_n^{\kappa},\qquad
\varepsilon_{n+1}=2A_n,\qquad
R_{n+1}=\left(\frac{\varepsilon_n}{A_n}\right)^{\frac{1}{\gamma-\alpha}}R_n .
\end{equation}
By \eqref{eq_eps0} and induction, $\varepsilon_{n+1}\le\frac12\varepsilon_n$ for all $n$, so $\varepsilon_n\le 2^{-n}\varepsilon_0$ decreases to $0$. Consequently $\varepsilon_n/A_n=2\varepsilon_n/\varepsilon_{n+1}\ge4$, so $R_{n+1}\ge 4^{1/(\gamma-\alpha)}R_n$ and $R_n$ increases to $+\infty$.

\smallskip
\emph{Step 1: the inductive comparison.} We claim that
\begin{equation}\label{eq_induct}
|u(x)|\le\varepsilon_n|x|^{-\alpha}\quad\text{for }|x|\ge R_n
\end{equation}
holds for every $n\ge0$. It holds for $n=0$. Assume it for some $n$ and set
\[
B_n=\varepsilon_n R_n^{\gamma-\alpha},\qquad w_n(x)=A_n|x|^{-\alpha}+B_n|x|^{-\gamma} .
\]
By Lemma \ref{lem_super} and the definition of $A_n$,
\[
-\Delta_p w_n\ge K A_n^{p-1}|x|^{-\alpha q}=\varepsilon_n^{q}|x|^{-\alpha q}\ge|u(x)|^{q}
\qquad\text{for }|x|>R_n ,
\]
where the last inequality uses \eqref{eq_induct}. By Lemma \ref{lem_abs}, $|u|$ is a weak subsolution of $-\Delta_p z=|u|^q$ in $\{|x|>R_n\}$. On $\partial B_{R_n}$,
\[
w_n\ge B_nR_n^{-\gamma}=\varepsilon_nR_n^{-\alpha}\ge|u| ,
\]
again by \eqref{eq_induct}. Finally $|u(x)|\to0$ and $w_n(x)\to0$ as $|x|\to\infty$. Lemma \ref{lem_comp}, applied with $v=|u|$, $w=w_n$, $f=|u|^q$ and $R=R_n$, gives
\begin{equation}\label{eq_compconc}
|u(x)|\le A_n|x|^{-\alpha}+B_n|x|^{-\gamma}\qquad\text{for }|x|\ge R_n .
\end{equation}
If $|x|\ge R_{n+1}$, then $|x|^{\gamma-\alpha}\ge\frac{\varepsilon_n}{A_n}R_n^{\gamma-\alpha}$, that is $B_n|x|^{-\gamma}\le A_n|x|^{-\alpha}$, so \eqref{eq_compconc} yields $|u(x)|\le 2A_n|x|^{-\alpha}=\varepsilon_{n+1}|x|^{-\alpha}$. This is \eqref{eq_induct} for $n+1$.

\smallskip
\emph{Step 2: the bookkeeping.} Taking logarithms in \eqref{eq_rec},
\[
\log R_{n+1}-\log R_n=\frac{\log2+\log\varepsilon_n-\log\varepsilon_{n+1}}{\gamma-\alpha},
\]
and summing from $0$ to $n-1$,
\[
(\gamma-\alpha)\big(\log R_n-\log R_0\big)=n\log 2+\log\varepsilon_0-\log\varepsilon_n .
\]
Exponentiating gives the identity
\begin{equation}\label{eq_identity}
\varepsilon_nR_n^{\gamma-\alpha}=\varepsilon_0R_0^{\gamma-\alpha}\,2^{\,n}=:c_0\,2^{\,n} ,
\end{equation}
so that $B_n=c_02^n$ and, since $A_n=\frac12\varepsilon_{n+1}$,
\begin{equation}\label{eq_An}
A_nR_{n+1}^{\gamma-\alpha}=\tfrac12\varepsilon_{n+1}R_{n+1}^{\gamma-\alpha}=\tfrac12 c_02^{\,n+1}=c_02^{\,n}.
\end{equation}
Let $n\ge0$ and $R_n\le|x|\le R_{n+1}$. Using \eqref{eq_compconc}, \eqref{eq_identity} and \eqref{eq_An},
\[
|u(x)|\le|x|^{-\gamma}\left(A_n|x|^{\gamma-\alpha}+B_n\right)
\le|x|^{-\gamma}\left(A_nR_{n+1}^{\gamma-\alpha}+B_n\right)=2c_0\,2^{\,n}|x|^{-\gamma} .
\]

\smallskip
\emph{Step 3: bounding $2^n$ by a power of $\log|x|$.} Put $e_n=-\log\varepsilon_n$. From $\varepsilon_{n+1}=2K^{-1/(p-1)}\varepsilon_n^\kappa$ we get $e_{n+1}=\kappa e_n-c_1$, hence $e_{n+1}-\bar c=\kappa(e_n-\bar c)$ and
\[
e_n=\bar c+\kappa^{\,n}(e_0-\bar c)\ge \bar c+\kappa^{\,n},
\]
by the second condition in \eqref{eq_eps0}. Combining with \eqref{eq_identity} and $\log R_0\ge0$, $n\log2\ge0$,
\[
(\gamma-\alpha)\log R_n\ge e_n-e_0\ge\kappa^{\,n}+\bar c-e_0 ,
\]
so that $\kappa^{\,n}\le(\gamma-\alpha)\log R_n+e_0-\bar c$. Set $R_*=\max\{R_1,e\}$ and let $|x|\ge R_*$, so that $|x|\in[R_n,R_{n+1}]$ for some $n\ge1$ and $\log|x|\ge\log R_n\ge1$. Then
\[
\kappa^{\,n}\le(\gamma-\alpha)\log R_n+(e_0-\bar c)\le C_1\log|x| ,
\]
where we used $\log R_n\le\log|x|$, $\log|x|\ge1$ and $e_0-\bar c\ge1$, and where $C_1=\gamma-\alpha+e_0-\bar c$. Whence $n\le\frac{\log(C_1\log|x|)}{\log\kappa}$ and
\[
2^{\,n}\le\exp\left(\frac{\log 2}{\log\kappa}\log\big(C_1\log|x|\big)\right)=\big(C_1\log|x|\big)^{\theta} .
\]
Together with Step 2 this gives \eqref{eq_pharmonic} with $C=2c_0C_1^{\theta}$.

Let $r\ge 2R_*$ and put $M_r=\sup_{\frac12 r\le|y|\le 2r}|u(y)|$, so that $M_r\le Cr^{-\gamma}(\log r)^{\theta}$ by \eqref{eq_pharmonic}. <!-- remark-5: body_md -->
The logarithmic factor in \eqref{eq_pharmonic} is an artifact of the iteration, and its exponent can be made as small as one wishes. The factor $2$ in $\varepsilon_{n+1}=2A_n$ is the price paid for passing from the two term bound \eqref{eq_compconc} to a single term at the radius $R_{n+1}$. Fix $\tau>0$ and choose $R_{n+1}$ instead so that at that radius the second term of \eqref{eq_compconc} is $\tau$ times the first, that is
\[
\varepsilon_{n+1}=(1+\tau)A_n,\qquad
R_{n+1}=\Big(\frac{\varepsilon_n}{\tau A_n}\Big)^{\frac1{\gamma-\alpha}}R_n .
\]
The induction of Step 1 goes through verbatim once $\varepsilon_0$ is small enough, depending on $\tau$, and the bookkeeping of Step 2 now gives
\[
\varepsilon_nR_n^{\gamma-\alpha}=c_0\lambda^{\,n},\qquad \lambda=\frac{1+\tau}{\tau},
\]
in place of \eqref{eq_identity}, whence $|u(x)|\le c_0\lambda^{\,n+1}|x|^{-\gamma}$ when $R_n\le|x|\le R_{n+1}$ and, by Step 3 with $\log\lambda$ in place of $\log 2$,
\[
|u(x)|\le C_\tau\,|x|^{-\gamma}\big(\log|x|\big)^{\theta_\tau},
\qquad \theta_\tau=\frac{\log\frac{1+\tau}{\tau}}{\log\frac{q}{p-1}} .
\]
Since $\theta_\tau\to0$ as $\tau\to\infty$, the exponent $\theta$ in \eqref{eq_pharmonic} may be replaced by any positive number, at the cost of a larger constant. It cannot be replaced by $0$ this way, because $C_\tau$ grows with $\tau$, and we do not know whether $u(x)=O(|x|^{-\gamma})$ holds. The question is of no consequence for Theorem \ref{thm_main}, since all the inequalities of Lemma \ref{lem_energy} are strict.

\label{lem_energy}
Let $N>p\ge2$ and $\frac{N(p-1)}{N-p}<q$, and let $u\in C^1(\mathbb{R}^N)$ satisfy \eqref{eq_pharmonic} and \eqref{eq_gradboot}, with $\theta$ as in Proposition \ref{prop_boot}. Then

<!-- proof-13: body_md -->
Since $(p-1)\gamma=N-p$ we have
\[
N-(\gamma+1)p=N-p-\gamma p=(N-p)\left(1-\frac{p}{p-1}\right)=-\frac{N-p}{p-1}<0 .
\]
Also $q+1>\frac{N(p-1)}{N-p}=\frac N\gamma$, hence $\gamma(q+1)>N$, that is $N-\gamma(q+1)<0$. Write $\mu_1=(\gamma+1)p-N>0$ and $\mu_2=\gamma(q+1)-N>0$. By \eqref{eq_pharmonic} and \eqref{eq_gradboot},
\[
\int_{|x|\ge R_*}|\nabla u|^p\,dx\le C\int_{R_*}^{\infty}r^{-(\gamma+1)p}(\log r)^{\theta p}r^{N-1}\,dr
=C\int_{R_*}^{\infty}r^{-1-\mu_1}(\log r)^{\theta p}\,dr<+\infty,
\]
and similarly $\int_{|x|\ge R_*}|u|^{q+1}\,dx\le C\int_{R_*}^{\infty}r^{-1-\mu_2}(\log r)^{\theta(q+1)}\,dr<+\infty$. The same computation applies to $|u|^{p^*}$, because $\gamma p^*=\frac{Np}{p-1}>N$. On the ball $B_{R_*}$ all three integrands are continuous, hence integrable.

<!-- lemma-10: body_md -->
\label{lem_poho}
Let $u\in C^1(\mathbb{R}^N)$ be a weak solution of \eqref{eq_main} with $|\nabla u|\in L^p(\mathbb{R}^N)$ and $u\in L^{q+1}(\mathbb{R}^N)$. Then
\begin{equation}
\frac{N}{q+1}\int_{\mathbb{R}^N}|u|^{q+1}\,dx=\frac{N-p}{p}\int_{\mathbb{R}^N}|\nabla u|^p\,dx .
\end{equation}
<!-- end lemma-10 -->
<!-- proof-14: body_md -->
Let $\varepsilon_*$ be the constant of Proposition \ref{prop_boot}. That proposition and Corollary \ref{cor_gradboot} give \eqref{eq_pharmonic} and \eqref{eq_gradboot}, and Lemma \ref{lem_energy} then gives
\[
u\in L^{q+1}(\mathbb{R}^N),\qquad |\nabla u|\in L^{p}(\mathbb{R}^N).
\]

Let $\eta_R\in C_c^\infty(B_{2R})$ with $0\le\eta_R\le1$, $\eta_R\equiv1$ on $B_R$ and $|\nabla\eta_R|\le C/R$. Testing \eqref{eq_main} with $u\eta_R$ gives
\[
\int_{\mathbb{R}^N}|\nabla u|^p\eta_R\,dx+\int_{\mathbb{R}^N}u\,|\nabla u|^{p-2}\nabla u\cdot\nabla\eta_R\,dx
=\int_{\mathbb{R}^N}|u|^{q+1}\eta_R\,dx .
\]
By \eqref{eq_pharmonic} and \eqref{eq_gradboot}, the middle term is bounded in absolute value by
\[
\frac{C}{R}\int_{B_{2R}\setminus B_R}|u|\,|\nabla u|^{p-1}\,dx
\le \frac{C}{R}\,R^{N}\,R^{-\gamma}R^{-(\gamma+1)(p-1)}(\log R)^{\theta p}
=C\,R^{\,N-p-\gamma p}(\log R)^{\theta p},
\]
and $N-p-\gamma p=-\frac{N-p}{p-1}<0$, so this tends to $0$. Letting $R\to\infty$ and using dominated convergence on the other two terms,
\begin{equation}\label{eq_id1}
\int_{\mathbb{R}^N}|\nabla u|^p\,dx=\int_{\mathbb{R}^N}|u|^{q+1}\,dx .
\end{equation}
The first and third bounds of Lemma \ref{lem_energy} give $u\in\mathcal D^{1,p}(\mathbb R^N)$, so \eqref{eq_id1} is the energy identity of the statement.

On the other hand, Lemma \ref{lem_poho} applies and gives
\begin{equation}\label{eq_id2}
\frac{N-p}{p}\int_{\mathbb{R}^N}|\nabla u|^p\,dx=\frac{N}{q+1}\int_{\mathbb{R}^N}|u|^{q+1}\,dx .
\end{equation}
Combining \eqref{eq_id1} and \eqref{eq_id2},
\[
\left(\frac{N-p}{p}-\frac{N}{q+1}\right)\int_{\mathbb{R}^N}|u|^{q+1}\,dx=0 .
\]
The factor $\frac{N-p}{p}-\frac{N}{q+1}$ vanishes only when $q+1=\frac{Np}{N-p}=p^*$. Hence, if $q\ne p^*-1$, then $\int_{\mathbb{R}^N}|u|^{q+1}=0$ and $u\equiv0$.

<!-- lemma-11: body_md -->
\label{lem_hardy}
Let $N\ge2$, $\sigma\ge0$ and $\beta\in\mathbb{R}$ with $N-2-\beta\ne0$. Then
\begin{equation}\label{eq_hardy}
\int_{\mathbb{R}^N}|x|^{-\beta}\Big[|\nabla\varphi|^2+\sigma(\partial_r\varphi)^2\Big]dx
\ \ge\ (1+\sigma)\left(\frac{N-2-\beta}{2}\right)^2\int_{\mathbb{R}^N}|x|^{-\beta-2}\varphi^2\,dx
\end{equation}
for every $\varphi\in C_c^1(\mathbb{R}^N\setminus\{0\})$.
<!-- end lemma-11 -->
<!-- proof-17: body_md -->
In polar coordinates $|\nabla\varphi|^2=(\partial_r\varphi)^2+r^{-2}|\nabla_\theta\varphi|^2$, where $\nabla_\theta$ is the gradient on $\mathbb{S}^{N-1}$. Discarding the angular term,
\[
\int_{\mathbb{R}^N}|x|^{-\beta}\Big[|\nabla\varphi|^2+\sigma(\partial_r\varphi)^2\Big]dx
\ \ge\ (1+\sigma)\int_{\mathbb{S}^{N-1}}\int_0^{\infty}\big(\partial_r\varphi\big)^2 r^{\,\nu}\,dr\,d\theta,
\qquad \nu=N-1-\beta .
\]
Fix $\theta$ and write $f(r)=\varphi(r\theta)$, a function in $C_c^1(0,\infty)$. Since $\nu\ne1$,
\begin{align*}
\int_0^\infty f^2r^{\nu-2}\,dr
&=\frac{1}{\nu-1}\int_0^\infty f^2\big(r^{\nu-1}\big)'\,dr \\
&=-\frac{2}{\nu-1}\int_0^\infty ff'r^{\nu-1}\,dr \\
&\le\frac{2}{|\nu-1|}
\left(\int_0^\infty f^2r^{\nu-2}\,dr\right)^{1/2}
\left(\int_0^\infty f'^2r^{\nu}\,dr\right)^{1/2},
\end{align*}
so that $\int_0^\infty f'^2r^\nu\,dr\ge\big(\frac{\nu-1}{2}\big)^2\int_0^\infty f^2r^{\nu-2}dr$. Since $\frac{\nu-1}{2}=\frac{N-2-\beta}{2}$, integrating over $\mathbb{S}^{N-1}$ gives \eqref{eq_hardy}.

<!-- lemma-12: body_md -->
\label{lem_algebra}
Let $N>p\ge2$ and $q>p^*-1$, and set $t=q-p+1$ and $M=N-p$. Then
\begin{equation}\label{eq_Hdef}
4(p-1)t^2\left[\left(\frac{N-2-\beta}{2}\right)^2-q\,\alpha(\gamma-\alpha)\right]=H(t),
\end{equation}
where
\begin{equation}\label{eq_H}
H(t)=M\big[(p-1)M-4p\big]\,t^2+2p^2(p-1)(2-M)\,t+p^4(p-1) .
\end{equation}
One has $H\big(\frac{p^2}{M}\big)=-\frac{4p^4}{M}<0$, and
\[
H(t)\ge0\iff (p-1)N>p(p+3)\ \text{ and }\ q\ge q_c(N,p).
\]
<!-- end lemma-12 -->
<!-- proof-18: body_md -->
By \eqref{eq_Nbeta}, $N-2-\beta=M-\alpha(p-2)$, and $\alpha=p/t$, $\gamma=M/(p-1)$, $q=t+p-1$. Hence
\begin{align*}
4t^2\left(\frac{N-2-\beta}{2}\right)^2
&=\big(Mt-p(p-2)\big)^2,\\
4t^2q\alpha(\gamma-\alpha)
&=4t^2(t+p-1)\frac pt\left(\frac{M}{p-1}-\frac pt\right)\\
&=\frac{4p}{p-1}(t+p-1)\big(Mt-p(p-1)\big).
\end{align*}
Multiplying the difference by $p-1$ and expanding,
\[
(p-1)\big(Mt-p(p-2)\big)^2-4p(t+p-1)\big(Mt-p(p-1)\big)
=\big[(p-1)M^2-4pM\big]t^2+c_1t+c_0 ,
\]
with
\begin{align*}
c_1
&=-2(p-1)Mp(p-2)-4p(p-1)M+4p^2(p-1)\\
&=-2p^2(p-1)M+4p^2(p-1)
 =2p^2(p-1)(2-M).
\end{align*}
and $c_0=(p-1)p^2(p-2)^2+4p^2(p-1)^2=p^2(p-1)\big[(p-2)^2+4(p-1)\big]=p^4(p-1)$. This is \eqref{eq_Hdef} with $H$ as in \eqref{eq_H}.

Next, with $t=p^2/M$,
\begin{align*}
\frac{M^2}{p^4}H\!\left(\frac{p^2}{M}\right)
&=M\big[(p-1)M-4p\big]+2(p-1)(2-M)M+(p-1)M^2\\
&=\big[(p-1)-2(p-1)+(p-1)\big]M^2-4pM+4(p-1)M\\
&=-4M.
\end{align*}
so $H(p^2/M)=-4p^4/M<0$. By \eqref{eq_alphasmall} the range $q>p^*-1$ is exactly $t>p^2/M$.

If $D>0$ then $H$ is a convex parabola and $H(p^2/M)<0$, so $p^2/M$ lies strictly between its two zeros and $H(t)\ge0$ on $(p^2/M,\infty)$ if and only if $t\ge t_+$, the larger zero. The discriminant of $H$ divided by $4$ equals
\begin{align*}
p^4(p-1)^2(M-2)^2-Dp^4(p-1)
&=p^4(p-1)\Big[(p-1)(M-2)^2-M\big((p-1)M-4p\big)\Big]\\
&=4p^4(p-1)\big(M+p-1\big)\\
&=4p^4(p-1)(N-1).
\end{align*}
where we used $(p-1)(M^2-4M+4)-(p-1)M^2+4pM=4M+4(p-1)$ and $M+p-1=N-1$. Hence
\[
t_+=\frac{p^2\Big[(p-1)(M-2)+2\sqrt{(p-1)(N-1)}\Big]}{D}.
\]
It remains to check that $p-1+t_+$ is the number $q_c(N,p)$ of \eqref{eq_qc}. Put $B=(p-1)N-p$, so that $(p-1)M=B+p(2-p)$ and $(p-1)M-4p=B-p^2-2p$. Then
\begin{align*}
(p-1)D+p^2(p-1)(M-2)
&=\big(B+p(2-p)\big)\big(B-p^2-2p\big)\\
&\quad +p^2\big(B+p(2-p)\big)-2p^2(p-1)\\
&=\big(B+2p-p^2\big)\big(B-2p\big)-2p^2(p-1).
\end{align*}
which expands to $B^2-p^2B-4p^2+2p^3-2p^3+2p^2=B^2-p^2B-2p^2$. Therefore
\begin{align*}
p-1+t_+
&=\frac{(p-1)D+p^2(p-1)(M-2)+2p^2\sqrt{(p-1)(N-1)}}{D}\\
&=\frac{B^2-p^2B-2p^2+2p^2\sqrt{(p-1)(N-1)}}
{(N-p)\big[(p-1)N-p(p+3)\big]}.
\end{align*}
and substituting $-p^2B-2p^2=-p^2\big((p-1)N-p\big)-2p^2=-p^2(p-1)N+p^2(p-2)$ turns the numerator into the numerator of \eqref{eq_qc}. Since $q=p-1+t$, the equivalence follows.

<!-- proof-19: body_md -->
Since $U$ is homogeneous of degree $-\alpha$, write
\[

\]
Set
\[
A(\theta)=\big(\alpha^2\psi^2+|\nabla_\theta\psi|^2\big)^{\frac{p-2}{2}},
\qquad m=(p-1)(\gamma-\alpha).
\]
The identities
\[
\nabla U=r^{-\alpha-1}\big(-\alpha\psi\,e_r+\nabla_\theta\psi\big),
\qquad
\alpha q=1+(\alpha+1)(p-1)
\]
show, by testing the weak equation with functions of the form $\eta(r)\zeta(\theta)$ and integrating the radial derivative by parts, that $\psi$ solves on $\mathbb S^{N-1}$
\begin{equation}\label{eq_sphere_qc}
-\operatorname{div}_\theta(A\nabla_\theta\psi)+\alpha m A\psi
=|\psi|^{q-1}\psi
\end{equation}
weakly. Testing \eqref{eq_sphere_qc} with $\psi$ gives
\begin{equation}\label{eq_sphere_energy}
I_1+\alpha m I_0=I_q,
\end{equation}
where
\[
I_1=\int_{\mathbb S^{N-1}}A|\nabla_\theta\psi|^2,\qquad
I_0=\int_{\mathbb S^{N-1}}A\psi^2,\qquad
I_q=\int_{\mathbb S^{N-1}}|\psi|^{q+1}.
\]
All these quantities are finite. Indeed, the homogeneity of $U$ and the assumptions on any fixed annulus imply
$\psi\in W^{1,p}(\mathbb S^{N-1})\cap L^{q+1}(\mathbb S^{N-1})$. In particular,
$A\in L^{p/(p-2)}(\mathbb S^{N-1})$ when $p>2$ (with the evident interpretation when $p=2$), and
$A\nabla_\theta\psi\in L^{p/(p-1)}(\mathbb S^{N-1})$. Smooth functions are dense in
$W^{1,p}(\mathbb S^{N-1})\cap L^{q+1}(\mathbb S^{N-1})$ for the sum norm. Hence one may first test with smooth approximations of $\psi$ and then pass to the limit by H\"older's inequality; this justifies both testing \eqref{eq_sphere_qc} with $\psi$ and the nonsmooth angular factor used in the stability test below.

We next note that the assumed stability outside $B_R$ upgrades, by homogeneity, to stability on the whole punctured space. Indeed, if $Q_U\ge0$ for test functions supported outside $B_R$ and $\varphi\in C_c^1(\mathbb R^N\setminus\{0\})$, then for $t$ sufficiently large the function $\varphi_t(x)=\varphi(x/t)$ is supported outside $B_R$. Since $\alpha(q-1)=2+(\alpha+1)(p-2)$, a change of variables gives
\[
Q_U(\varphi_t)=t^{N-2-(\alpha+1)(p-2)}Q_U(\varphi),
\]
so $Q_U(\varphi)\ge0$.

Put
\[
\beta=(\alpha+1)(p-2),\qquad
\mu=\frac{N-2-\beta}{2}.
\]
Choose $\eta\in C_c^1(\mathbb R)$ and use in the stability inequality the test function
\[
\varphi(r,\theta)=r^{-\mu}\eta(\log r)\psi(\theta).
\]
For $p\ge2$,
\[
|\nabla U|^{p-2}|\nabla\varphi|^2
 +(p-2)|\nabla U|^{p-4}(\nabla U\cdot\nabla\varphi)^2
\le (p-1)|\nabla U|^{p-2}|\nabla\varphi|^2.
\]
Using $\alpha(q-1)=\beta+2$ and $2\mu=N-2-\beta$, the radial powers cancel after the change of variables $s=\log r$, and stability yields
\[
q I_q\int_{\mathbb R}\eta^2\,ds
\le(p-1)\left[
I_0\int_{\mathbb R}(\eta'-\mu\eta)^2\,ds
+I_1\int_{\mathbb R}\eta^2\,ds
\right].
\]
Take $\eta_L\in C_c^1(\mathbb R)$ with $0\le\eta_L\le1$, $\eta_L\equiv1$ on $[1,L-1]$, $\operatorname{supp}\eta_L\subset[0,L]$ and $|\eta_L'|\le2$, so that $\int\eta_L^2\ge L-2$ and $\int\eta_L'^2\le8$. Since $\int\eta_L\eta_L'=0$, dividing by $\int\eta_L^2$ and letting $L\to\infty$ give
\begin{equation}\label{eq_sphere_stab}
q I_q\le(p-1)(\mu^2 I_0+I_1).
\end{equation}
Combining \eqref{eq_sphere_energy} and \eqref{eq_sphere_stab},
\begin{equation}\label{eq_sphere_cancel}
(q-p+1)I_1
\le\big[(p-1)\mu^2-q\alpha m\big]I_0.
\end{equation}
Since $m=(p-1)(\gamma-\alpha)$, the coefficient on the right of \eqref{eq_sphere_cancel} is
\[
(p-1)\mu^2-q\alpha m=(p-1)\big[\mu^2-q\alpha(\gamma-\alpha)\big]=\frac{H(t)}{4t^2},
\qquad t=q-p+1,
\]
by the identity \eqref{eq_Hdef} of Lemma \ref{lem_algebra}. That lemma also tells us the sign of $H(t)$ in the range $q>p^*-1$: it is negative for $q<q_c(N,p)$ and zero at $q=q_c(N,p)$.

Suppose first $q<q_c(N,p)$. Then the right hand side of \eqref{eq_sphere_cancel} is a negative multiple of $I_0$, while the left hand side is nonnegative. Hence $I_0=0$. Since $A\psi^2>0$ at every $\theta$ with $\psi(\theta)\ne0$, this forces $\psi=0$ almost everywhere, that is $U\equiv0$. This is (i).

Suppose now $q=q_c(N,p)$, so that the coefficient on the right of \eqref{eq_sphere_cancel} vanishes. Because $q-p+1>0$, we obtain $I_1=0$. For $p=2$ we have $A\equiv1$ and this already gives $\nabla_\theta\psi=0$ almost everywhere. For $p>2$ the pointwise inequality
\[
A|\nabla_\theta\psi|^2=\big(\alpha^2\psi^2+|\nabla_\theta\psi|^2\big)^{\frac{p-2}{2}}|\nabla_\theta\psi|^2\ \ge\ |\nabla_\theta\psi|^{p}
\]
turns $I_1=0$ into $\int_{\mathbb S^{N-1}}|\nabla_\theta\psi|^p=0$, so again $\nabla_\theta\psi=0$ almost everywhere. In either case $\psi$ is constant. Equation \eqref{eq_sphere_qc} then shows that either $\psi=0$ or
\[
|\psi|^{q-p+1}=\alpha^{p-1}(p-1)(\gamma-\alpha)=c_s^{q-p+1}.
\]
Thus $U\equiv0$ or $U=\pm c_s|x|^{-\alpha}$, which is (ii).

\label{prop_qc_omega}

$\mathcal X=W^{1,p}_{\mathrm{loc}}(\mathbb R^N\setminus\{0\})$ with its usual Fr\'echet topology and set

\begin{enumerate}[label=\rm(\roman*)]
\item every element of $\Omega_\infty(u)$ is homogeneous of degree $-\alpha$;
\item $\Omega_\infty(u)$ consists of a single element;
\item the full family $u_R$ converges in $\mathcal X$ as $R\to\infty$.
\end{enumerate}
Whenever these conditions hold, the limit is necessarily
\[

\emph{Step 4: the only equilibrium, and no cycles.} The equilibria of \eqref{eq_phase} in $\mathcal Q$ satisfy $F=W^{q}/m$ and $(\alpha W)^{p-1}=F$, hence $W^{q-p+1}=m\alpha^{p-1}=\alpha^{p-1}(p-1)(\gamma-\alpha)=c_s^{q-p+1}$. So $P=\big(c_s,(\alpha c_s)^{p-1}\big)$ is the unique equilibrium in $\mathcal Q$. The divergence of the vector field in \eqref{eq_phase} is the constant
\[
\alpha-m=\alpha-(p-1)(\gamma-\alpha)=p\alpha-(p-1)\gamma=p\alpha-(N-p)<0,
\]
the inequality being \eqref{eq_alphasmall}, that is $q>p^*-1$. Let $\Gamma$ be any periodic orbit of \eqref{eq_phase} and let $D$ be the region it encloses, so $D\subset\mathcal Q$ because $\mathcal Q$ is simply connected. The vector field is tangent to $\Gamma$, and Green's theorem gives $\int_D(\alpha-m)\,dWdF=0$, which is impossible since $|D|>0$. The same conclusion excludes a homoclinic loop at $P$: excise a disk $B_\rho(P)$, apply Green's theorem to the region enclosed by the loop with $B_\rho(P)$ removed, and let $\rho\downarrow0$; the flux across $\partial B_\rho(P)$ tends to zero because the vector field is $C^1$ and vanishes at $P$. Hence \eqref{eq_phase} has neither a periodic orbit nor a homoclinic loop in $\mathcal Q$.

\emph{Step 5: convergence.} By Step 3 the $\omega$-limit set of the forward orbit is a nonempty compact connected invariant subset of $\mathcal Q$. By the Poincar\'e--Bendixson theorem it is a periodic orbit, an equilibrium, or a union of equilibria and orbits connecting them. The first is excluded by Step 4. In the third case, $P$ being the only equilibrium in $\mathcal Q$, the set would contain a homoclinic loop at $P$, which Step 4 also excludes. Hence the $\omega$-limit set is an equilibrium, necessarily $P$. Therefore $W(t)\to c_s$ and $F(t)\to(\alpha c_s)^{p-1}$, which is exactly
\[
\lim_{r\to\infty}r^{\alpha}u(r)=c_s,\qquad \lim_{r\to\infty}r^{\alpha+1}\big(-u'(r)\big)=\alpha c_s ,
\]
and \eqref{eq_slowexact} follows because $|\nabla u|=-u'$.

\emph{Step 6: stability when $q>q_c(N,p)$.} Here Lemma \ref{lem_algebra} gives $H(q-p+1)>0$, that is
\[
(p-1)\left(\frac{N-2-\beta}{2}\right)^2>q\,\alpha(p-1)(\gamma-\alpha)=\frac{q\,c_s^{q-1}}{(\alpha c_s)^{p-2}} ,
\]
the last equality being \eqref{eq_Lambda}. Choose $\varepsilon\in(0,1)$ so small that
\[
(1-\varepsilon)^{p-2}(p-1)\left(\frac{N-2-\beta}{2}\right)^2\ \ge\ (1+\varepsilon)^{q-1}\,\frac{q\,c_s^{q-1}}{(\alpha c_s)^{p-2}} ,
\]
which is possible because the two sides are continuous in $\varepsilon$ and the inequality is strict at $\varepsilon=0$. By \eqref{eq_slowexact} there is $R_\varepsilon>0$ with
\[
u(x)\le(1+\varepsilon)c_s|x|^{-\alpha}
\qquad\text{and}\qquad
|\nabla u(x)|\ge(1-\varepsilon)\alpha c_s|x|^{-\alpha-1}
\qquad\text{for } |x|\ge R_\varepsilon .
\]
Let $\varphi\in C_c^1(\{|x|>R_\varepsilon\})$. Since $u$ is radial, $\nabla u=u'(|x|)\frac{x}{|x|}$ and therefore $|\nabla u|^{p-4}(\nabla u\cdot\nabla\varphi)^2=|\nabla u|^{p-2}(\partial_r\varphi)^2$, so
\[
Q_u(\varphi)=\int|\nabla u|^{p-2}\Big[|\nabla\varphi|^2+(p-2)(\partial_r\varphi)^2\Big]dx-q\int u^{q-1}\varphi^2dx .
\]
The bracket is nonnegative because $p\ge2$, so the first integral is at least
$\big((1-\varepsilon)\alpha c_s\big)^{p-2}\int|x|^{-\beta}\big[|\nabla\varphi|^2+(p-2)(\partial_r\varphi)^2\big]dx$, while the second is at most $q\big((1+\varepsilon)c_s\big)^{q-1}\int|x|^{-\beta-2}\varphi^2dx$, using $\alpha(q-1)=\beta+2$. Lemma \ref{lem_hardy} with $\sigma=p-2$ and the choice of $\varepsilon$ now give $Q_u(\varphi)\ge0$. Hence $u$ is stable outside $\overline{B_{R_\varepsilon}}$.

<!-- proof-24: body_md -->
Let $q=p^*-1$. Proposition \ref{prop_decay} applies at the endpoint and gives $|x|^\alpha|u(x)|\to0$, and $p^*-1>\frac{N(p-1)}{N-p}$, so Theorem \ref{thm_decay} applies and yields \eqref{eq_critical_energy}. Its last assertion is vacuous here, which is why the Sobolev endpoint survives.

Assume now that $u$ has finite Morse index. Since $u\in\mathcal D^{1,p}(\mathbb R^N)$ by \eqref{eq_critical_energy}, \cite[Thm.~1.4]{FMW2019} gives $n(u)\le i(u)$ when $p>2$. We record the corresponding elementary argument for $p=2$. Let $\omega_1,\ldots,\omega_k$ be distinct nodal regions of $u$. By the nodal-domain approximation of \cite[Lem.~3.1]{FMW2019}, for each $j$ there are functions $v_{j,n}\in C_c^1(\omega_j)$ converging in $\mathcal D^{1,2}(\mathbb R^N)$ to the restriction of $u$ to $\omega_j$, extended by zero. Testing the equation on the nodal region gives
\[
\int_{\omega_j}|\nabla u|^2=\int_{\omega_j}|u|^{2^*},
\]
and therefore
\[
Q_u(v_{j,n})\longrightarrow -(2^*-2)\int_{\omega_j}|u|^{2^*}<0.
\]
For $n$ large the $k$ functions $v_{j,n}$ have pairwise disjoint supports, so the quadratic form is negative definite on their span. Thus $k\le i(u)$. Since $k$ was arbitrary, the total number $n(u)$ of nodal regions is finite and satisfies $n(u)\le i(u)$ also for $p=2$.

Suppose $i(u)\le1$. Then $n(u)\le1$, so $u$ does not change sign. It remains to treat a solution of one sign, with or without the Morse index hypothesis. Either $u\equiv0$, or the strong maximum principle makes $u$ positive after a change of sign, and the classification of positive $\mathcal D^{1,p}(\mathbb R^N)$ solutions of the critical equation gives $u=u_{\lambda,x_0}$ for some $\lambda>0$ and $x_0\in\mathbb R^N$. That classification is \cite{Sciunzi2016} for $p>2$ and \cite{CGS1989} for $p=2$.

<!-- lemma-13: body_md -->
\label{lem_C2}
For $2<p<N$ the functionals $J$ and $\Phi$ are of class $C^2$ on $X$, and for every $u,v\in X$
\begin{equation}\label{eq_Jsecond}
J''(u)[v,v]=\int_{\mathbb{R}^N}|\nabla u|^{p-2}|\nabla v|^2dx+(p-2)\int_{\mathbb{R}^N}|\nabla u|^{p-4}(\nabla u\cdot\nabla v)^2dx-(p^*-1)\int_{\mathbb{R}^N}|u|^{p^*-2}v^2dx .
\end{equation}
In particular $J''(U)[v,v]=Q_U(v)$ for every $v\in X$, where $Q_U$ is the quadratic form of Section \ref{sec_intro} with $q=p^*-1$.
<!-- end lemma-13 -->
<!-- proof-25: body_md -->
Let $\mathcal M$ be the matrix \eqref{eq_Mdef}. We argue in three steps, for the functional $u\mapsto\int|\nabla u|^p$; the functional $u\mapsto\int|u|^{p^*}$ is treated in the same way, with $p^*>p>2$ in place of $p$.

\emph{Step 1: the bilinear form.} For $u\in X$ put
\[
B_u[v,w]=\int_{\mathbb{R}^N} \mathcal M(\nabla u)\nabla v\cdot\nabla w\,dx,\qquad v,w\in X .
\]
By \eqref{eq_Mbound} and H\"older with exponents $\frac{p}{p-2}$, $p$, $p$,
\[
\big|B_u[v,w]\big|\le(p-1)\,\|\nabla u\|_{L^p}^{p-2}\,\|\nabla v\|_{L^p}\,\|\nabla w\|_{L^p},
\]
so $B_u$ is a bounded bilinear form on $X$.

\emph{Step 2: $B$ is the Gateaux differential of $I'$, in the norm of $X^*$, and it is continuous in $u$.} Write $I(u)=\frac1p\int|\nabla u|^p$, so that $I'(u)[v]=\int|\nabla u|^{p-2}\nabla u\cdot\nabla v\,dx$. By Lemma \ref{lem_nem}(ii) the map $\eta\mapsto|\eta|^{p-2}\eta$ is of class $C^1$ on all of $\mathbb{R}^N$ with differential $\mathcal M$, so for $u,w\in X$ and $0<|t|\le1$ the fundamental theorem of calculus gives, pointwise,
\[
\frac1t\Big(|\nabla u+t\nabla w|^{p-2}(\nabla u+t\nabla w)-|\nabla u|^{p-2}\nabla u\Big)
=\int_0^1 \mathcal M(\nabla u+\tau t\nabla w)\,\nabla w\,d\tau .
\]
Let $\mathcal{A}(u):=\mathcal M(\nabla u)$. Pairing the last identity with $\nabla v$, integrating, and applying H\"older as in Step 1 together with Minkowski's integral inequality in the $\tau$ variable,
\[
\Big\|\tfrac1t\big(I'(u+tw)-I'(u)\big)-B_u[w,\cdot\,]\Big\|_{X^*}
\ \le\ \Big(\int_0^1\big\|\mathcal{A}(u+\tau tw)-\mathcal{A}(u)\big\|_{L^{\frac{p}{p-2}}}d\tau\Big)\,\|\nabla w\|_{L^p} .
\]
By Lemma \ref{lem_nem}(ii), $\mathcal A$ is continuous from $X$ into $L^{\frac{p}{p-2}}(\mathbb{R}^N)$. The integrand in the bracket therefore tends to $0$ for each $\tau$ as $t\to0$, and is bounded by $2(p-1)\big\||\nabla u|+|\nabla w|\big\|_{L^p}^{p-2}$ uniformly in $\tau$ and $|t|\le1$, so the bracket tends to $0$ by dominated convergence. Hence $I'$ is Gateaux differentiable at $u$ with differential $B_u$, and $u\mapsto B_u$ is continuous from $X$ into the space of bounded bilinear forms on $X$, again by the continuity of $\mathcal A$ and the H\"older bound of Step 1.

\emph{Step 3: conclusion.} A Gateaux differentiable map between Banach spaces whose Gateaux differential is continuous is Fr\'echet differentiable, with the same differential. Applying this to $I':X\to X^*$, Steps 1 and 2 give that $I$ is of class $C^2$ on $X$ with $I''(u)=B_u$, hence that $u\mapsto\int|\nabla u|^p=pI(u)$ is of class $C^2$ with second derivative $p\,B_u$. The corresponding statement for $u\mapsto\int|u|^{p^*}$ gives the second derivative $p^*(p^*-1)\int|u|^{p^*-2}vw$, and \eqref{eq_Jsecond} follows, as does the corresponding formula for $\Phi''$. The identification with $Q_U$ is a comparison of the two expressions, using $q-1=p^*-2$.

<!-- lemma-14: body_md -->
\label{lem_nehari}
$U$ minimizes $J$ on $\mathcal{N}$, and $\Phi'(U)\ne0$.
<!-- end lemma-14 -->
<!-- proof-26: body_md -->
Since $U\in X$ and $U$ solves the equation weakly in $X$, testing with $U$ itself is admissible and gives $\int|\nabla U|^p=\int U^{p^*}$, that is $\Phi(U)=0$, so $U\in\mathcal N$. For $u\in\mathcal N$ put $Y=\|\nabla u\|_{L^p}^p=\|u\|^{p^*}_{L^{p^*}}>0$. The Sobolev inequality gives $Y\ge S\|u\|_{L^{p^*}}^p=S\,Y^{p/p^*}$, hence
\[
Y^{1-\frac{p}{p^*}}\ge S,\qquad\text{that is}\qquad Y\ge S^{\frac{p^*}{p^*-p}}=S^{\frac Np},
\]
and $Y=S^{N/p}$ for $u=U$, because $U$ attains $S$ \cite{Talenti1976} and lies on $\mathcal N$. Since $J(u)=\big(\frac1p-\frac1{p^*}\big)Y$ on $\mathcal{N}$, the function $U$ is a global minimizer of $J$ on $\mathcal N$.

Finally
\[
\Phi'(U)[U]=p\int|\nabla U|^p-p^*\int U^{p^*}=(p-p^*)\int U^{p^*}<0 ,
\]
where we used $\int|\nabla U|^p=\int U^{p^*}$, so $\Phi'(U)\ne0$.
<!-- end proof-26 -->
<!-- proof-27: body_md -->
\emph{Lower bound.} The function $U$ is positive, so it has exactly one nodal region, and \cite[Thm.~1.4]{FMW2019} gives $1\le i(U)$.

\emph{Upper bound.} By Lemma \ref{lem_nehari} we may fix $e\in X$ with $\Phi'(U)[e]=1$, and $T:=\ker \Phi'(U)$ is a closed subspace of $X$ of codimension one. Let $v\in T$. Since $\Phi$ is $C^2$ by Lemma \ref{lem_C2} and $\frac{\partial}{\partial s}\Phi(U+tv+se)\big|_{t=s=0}=1$, the implicit function theorem provides $\varepsilon>0$ and a function $s\in C^2((-\varepsilon,\varepsilon))$ with $s(0)=0$ and
\[
\Phi\big(\sigma(t)\big)=0\quad\text{for }|t|<\varepsilon,\qquad \sigma(t):=U+tv+s(t)e .
\]
Differentiating $\Phi(\sigma(t))=0$ at $t=0$ gives $\Phi'(U)[v]+s'(0)=0$, hence $s'(0)=0$ because $v\in T$. Shrinking $\varepsilon$ if necessary we also have $\sigma(t)\ne0$ for $|t|<\varepsilon$, since $\sigma$ is continuous and $U\ne0$. Thus $\sigma$ is a $C^2$ curve in $\mathcal N$ with $\sigma(0)=U$ and $\sigma'(0)=v$.

By Lemma \ref{lem_nehari} the function $g(t)=J(\sigma(t))$ has a minimum at $t=0$, so $g''(0)\ge0$. On the other hand $U$ is a critical point of $J$, so $J'(U)=0$ and
\[
g''(0)=J''(U)[\sigma'(0),\sigma'(0)]+J'(U)\big[\sigma''(0)\big]=J''(U)[v,v] .
\]
Therefore
\begin{equation}\label{eq_nonneg}
Q_U(v)=J''(U)[v,v]\ \ge\ 0\qquad\text{for every }v\in T .
\end{equation}

Let now $V\subset C_c^1(\mathbb{R}^N)$ be a subspace with $Q_U(\varphi)<0$ for every $\varphi\in V\setminus\{0\}$. Since $C_c^1(\mathbb{R}^N)\subset X$, \eqref{eq_nonneg} forces $V\cap T=\{0\}$. The restriction to $V$ of the quotient map $X\to X/T$ is therefore injective, and $\dim(X/T)=1$, so $\dim V\le1$. Hence $i(U)\le1$, and with the lower bound $i(U)=1$.

Combining Theorems \ref{thm_sob_endpoint} and \ref{thm_index} with \cite[Thm.~1.5]{FMW2019} gives, for $2<p<N$, a complete description of the low-index part of the Sobolev endpoint without assuming finite energy in advance: a $C^1$ solution of $-\Delta_pu=|u|^{p^*-2}u$ which is stable outside a compact set and satisfies $i(u)\le1$ is either $0$ or one of the functions \eqref{eq_bubble}, and in the latter case $i(u)=1$ exactly.
