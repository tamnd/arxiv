<!-- lemma-1: body_md -->
\label{ideals}\normalfont
For $\sigma>1$, both 
\begin{equation*}-\frac{\zeta_K'}{\zeta_K}(\sigma)\quad \text{ and }\quad\bigg(\frac{\zeta_K'}{\zeta_K}\bigg)'(\sigma)
\end{equation*} 
are positive.
Also, for $\sigma_0,\sigma_1>1$,
\begin{equation*}
\log\frac{\zeta_K(\sigma_1)}{\zeta_K(\sigma_0)}=\sum_{\mathfrak p\in P_K^*}\log \left( \frac{1-(N\mathfrak p)^{-\sigma_0}}{1-(N\mathfrak p)^{-\sigma_1}}\right).
\end{equation*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The logarithmic identity follows directly from \eqref{Euler}. On the other hand, again by the Euler product expression \eqref{Euler} for $\zeta_K$, since $\Re s>1$, we have
\begin{equation*}
 -\frac{\zeta_K'}{\zeta_K}(s)=\sum_{\mathfrak p\in P_K^*}\sum_{m\geq 1}
\frac{\log N\mathfrak p}{N\mathfrak p^{ms}},\qquad 
\bigg(\frac{\zeta_K'}{\zeta_K}\bigg)'(s)=\sum_{\mathfrak p\in P_K^*}\sum_{m\geq 1}
\frac{m\log^2 N\mathfrak p}{N\mathfrak p^{ms}}.
\end{equation*}
By selecting $s=\sigma>1$, we obtain that the above expressions are positive.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{Hadamard:grouped}\normalfont
There exists a non-zero constant $C>0$ such that
\begin{equation}\label{pound}
\xi_K(s)=
C\prod_{\rho=\beta}\left(1-\frac{s}{\rho}\right)
\prod_{\substack{\rho=\beta+i\gamma\\ \gamma>0}}
\left(1-\frac{s}{\rho}\right) \left(1-\frac{s}{\overline{\rho}}\right),
\end{equation}
where the products are taken with multiplicity.
Consequently, for $s$ not a zero of $\zeta_K$,
\begin{equation}\label{stone}
\frac{\xi_K'}{\xi_K}(s) =
\sum_{\rho=\beta} \frac{1}{s-\rho}+\sum_{\substack{\rho=\beta+i\gamma\\ \gamma>0}}\left(\frac{1}{s-\rho}+ \frac{1}{s-\overline{\rho}}\right).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
In the Hadamard product \eqref{Hadamard} we separate the real zeros from the non-real zeros and group the latter in
conjugate pairs; this reordering is justified by absolute convergence. There are only finitely many real non-trivial zeros and for $\rho=\beta+i\gamma$, $\gamma>0$, we have
\begin{equation*}
e^{s/\rho}e^{s/\overline{\rho}}
=e^{s\frac{2\beta}{\beta^2+\gamma^2}}=e^{O\left(\frac{1}{\gamma^2}\right)}.
\end{equation*}
Hence, the corresponding sum over zeros $\rho=\beta+i\gamma$, $\gamma>0$, converges thanks to the zero counting estimate
\begin{equation}\label{N_K}
N_K(T)
:=\#\left\{\rho=\beta+i\gamma:\zeta_K(\rho)=0,\ 0<\beta<1,\ 0<\gamma\leq  T\right\}\ll T(\log T+2).
\end{equation} 
where zeros are counted with multiplicity.
Subsequently, all exponential correction factors combine with
$e^{bs}$ into a single exponential factor $e^{Bs}$.  Thus we may write
\begin{equation*}
\xi_K(s) =e^{a+Bs}\prod_{\rho=\beta} \left(1-\frac{s}{\rho}\right)\prod_{\substack{\rho=\beta+i\gamma\\ \gamma>0}} \left(1-\frac{s}{\rho}\right)\left(1-\frac{s}{\overline{\rho}}\right)=e^{a+Bs}P(s).
\end{equation*}

On the other hand, the functional equation
$\xi_K(s)=\xi_K(1-s)$ implies that the non-trivial zeros are invariant under
$\rho\longmapsto1-\rho$. Moreover, for a non-real zero
$\rho=\beta+i\gamma$ with $\gamma>0$, we have
\begin{equation*}
\left(1-\frac{1-s}{\rho}\right)\left(1-\frac{1-s}{\overline{\rho}}\right)=
\frac{|1-\rho|^2}{|\rho|^2}\left(1-\frac{s}{1-\rho}\right)\left(1-\frac{s}{1-\overline{\rho}}\right).
\end{equation*}
The map $\rho\longmapsto1-\overline{\rho}$ permutes the zeros in the upper half-plane. Therefore, reindexing the absolutely convergent product gives
\begin{align*}
P(1-s)=P(s)\prod_{\rho}\frac{|1-\rho|^2}{|\rho|^2}=
P(s)\prod_{\rho}\bigg(1+O^{*}\left(\frac{1}{\gamma^2}\right)\bigg)=cP(s),
\end{align*}
up to the finitely many analogous factors arising from the real zeros where
$c\in\mathbb{C}\setminus{\{0\}}$. Hence
\begin{equation*}e^{a+Bs}P(s)=\xi_K(s)=\xi_K(1-s)=e^{a+B(1-s)}P(1-s)=ce^{a+B(1-s)}P(s).
\end{equation*}
Since $P$ is not
identically zero, this forces $e^{B(2s-1)}=c$
for all $s\in\mathbb{C}$, and therefore $B=0$ and $c=1$. Setting $C=e^a$, we obtain \eqref{pound}  

On the other hand, taking logarithmic derivative gives
\begin{equation*}
\frac{\xi_K'}{\xi_K}(s) =\frac{P'(s)}{P(s)}.
\end{equation*}

<!-- lemma-6: body_md -->
\label{Taylor}\normalfont
We have the following explicit bounds.
\begin{enumerate}[label=\textbf{(\alph*)},ref=\textbf{(\alph*)},leftmargin=3em,itemsep=2pt,topsep=2pt]

\item\label{taylor:a}
\begin{equation*}
\log(1+u) =u-\frac{u^2}{2}+\frac{u^3}{3}+O^*\left(\frac{u^4}{4}\right),\qquad(0\leq  u\leq 1),
\end{equation*}

\item\label{taylor:b}
\begin{equation*}
 (1+u)^{-1} =1-u+u^2+O^*(u^3),\qquad(u\geq 0).
\end{equation*}

\item\label{taylor:c}
\begin{equation*}
 (1+u)^{-2} =1-2u+O^*(3u^2),\qquad(u\geq 0).
\end{equation*}

\item\label{taylor:d}
\begin{equation*}
(1+u)^{-3}=1+O^*(3u),\qquad(u\geq 0).
\end{equation*}

\item\label{taylor:e}
\begin{equation*}
\frac{1-u}{(1+u)^2}=1-3u+5u^2+O^*(7u^3),\qquad(u\geq 0).
\end{equation*}

\end{enumerate}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
\ref{taylor:a}. For $|u|\leq 1$, $u\neq -1$, we have the expansion
\begin{equation*}
 \log(1+u)=-\sum_{n\geq1}\frac{(-u)^n}{n}.
\end{equation*}
For $0\leq  u\leq 1$, the terms $u^n/n$ decrease monotonically to $0$. Since we have an alternating series, it follows that
\begin{equation*}
 \bigg| \log(1+u) -\left(u-\frac{u^2}{2}+\frac{u^3}{3}\right)\bigg|\leq \frac{u^4}{4}.
\end{equation*}
For \ref{taylor:b}, we observe that
\begin{equation*}
\bigg|\frac{1}{1+u}-(1-u+u^2)\bigg|=\bigg|-\frac{u^3}{1+u}\bigg|\leq  u^3\qquad(u\geq 0).
\end{equation*}
\ref{taylor:c}. We have
\begin{equation*}
 \bigg|\frac{1}{(1+u)^2}-(1-2u)\bigg|=\bigg|u^2\frac{3+2u}{(1+u)^2}\bigg|\leq  3u^2\qquad(u\geq 0).
\end{equation*}
\ref{taylor:d}. We have
\begin{equation*}
 \bigg|\frac{1}{(1+u)^3}-1\bigg|=\bigg|u\frac{3+3u+u^2}{(1+u)^3}\bigg|\leq  3u\qquad(u\geq 0).
\end{equation*}
Finally, for \ref{taylor:e}, a direct calculation gives
\begin{equation*}
\bigg|\frac{1-u}{(1+u)^2}-(1-3u+5u^2)\bigg|= \bigg|u^3\frac{7+5u}{(1+u)^2}\bigg|\leq  7u^3\qquad(u\geq 0).
\end{equation*}

<!-- lemma-9: body_md -->
\label{calculus}\normalfont
Let $\theta>0$. 
\begin{enumerate}[
    label=\textbf{(\alph*)},
    ref=\textbf{(\alph*)},
    leftmargin=3em,
    itemsep=6pt,
    topsep=6pt,
    parsep=0pt
]
\item\label{>=1/3} If $\theta\ge 1/3$, then $W_\theta(x)>0\qquad (x>0)$.
\item\label{<1/3} If $0<\theta<1/3$, then $W_\theta$ has exactly one positive
zero $x_\theta$. Furthermore, $W_\theta(x)<0$, $0<x<x_\theta$, and $W_\theta(x)>0$, $x>x_\theta$.
\end{enumerate}
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
For $x>0$, write $W_\theta(x)= \frac{x^2}{2}\left(I(x)-e^{-\theta x}\right)$
where
\begin{equation}\label{def:I}
 I(x):=\frac{2(x-1+e^{-x})}{x^2}.
\end{equation}
Since $e^t>1+t$ for all $t\in\mathbb{R}$, $t\neq 0$, we have $I(x)>0$, $x\in\mathbb{R}$, $x\neq 0$. Moreover, $I$ extends continuously to $x=0$ by defining $I(0)=\lim_{x\to0}I(x)=1$. Thus $I(x)>0$ for all  $x\in\mathbb{R}$. We may now define the infinitely differentiable function
\begin{equation*}
        \Theta:x\geq 0\mapsto
        -\frac{\log I(x)}{x}.
\end{equation*}
Then $I(x)=e^{-x\Theta(x)}$ and
\begin{equation*}
W_\theta(x) =\frac{x^2}{2}\left( e^{-x\Theta(x)}-e^{-\theta x}\right).
\end{equation*}
Hence $W_\theta(x)>0$ if and only if $\Theta(x)<\theta$
and $W_\theta(x)=0$ if and only if $ \Theta(x)=\theta$.
       
We next study the monotonicity of $\Theta$. By definition \eqref{def:I},
we have
\begin{align*}
 (-\log I(x))'&=\phantom{-}\frac{x-2+(x+2)e^{-x}}{x(x-1+e^{-x})},\\
(-\log I(x))''&= -\frac{x^2-4x+2+(x^3+x^2+4x-4)e^{-x}+ 2e^{-2x} } {x^2(x-1+e^{-x})^2 }.
\end{align*}
By calculus, we can prove that $x^2-4x+2+(x^3+x^2+4x-4)e^{-x}+ 2e^{-2x}>0$ for all $x\geq 0$. Thus $(-\log I(x))''<0$ and hence $(-\log I(x))'$ is strictly decreasing for $x\geq 0$. Moreover, since $(-\log I(0))'=0$, then $(-\log I(x))'<0$ for all $x>0$. Thus, the function $-\log I(x)$ is strictly decreasing for $x>0$. Hence $\Theta(x)$ is strictly decreasing for $x>0$.

Furthermore, we readily see that $
\lim_{x\to0^+}\Theta(x)= \frac{1}{3}.$
Hence, $\Theta$ maps $(0,\infty)$ strictly decreasingly onto
$(0,1/3)$ since
\begin{equation*}
 I(x) =\frac{2(x-1+e^{-x})}{x^2} \sim\frac{2}{x}\ \text{ and }\ \Theta(x)=-\frac{\log I(x)}{x} =
 \frac{\log x-\log2+o(1)}{x}\to 0.
                    
\end{equation*}
Subsequently,  if $\theta\ge1/3$, then $\Theta(x)< 1/3\leq  \theta$ for all $x>0$, and hence
$W_\theta(x)>0$ for all $x>0$, whence \ref{>=1/3}. Else, if $0<\theta<1/3$, then there exists
a unique $x_\theta>0$ such that $\Theta(x_\theta)=\theta$. Since $\Theta$ is strictly decreasing, $x_\theta$ is the
unique positive zero of $W_\theta$. Thereupon,
we have $\Theta(x)>\theta$ if $0<x<x_\theta$, and
$ \Theta(x)<\theta$ if $x>x_\theta$; consequently
$W_\theta(x)<0$ if $0<x<x_\theta$, and
$ W_\theta(x)>0$ if $x>x_\theta$. This gives \ref{<1/3}.
