<!-- proof-1: body_md -->
In view of the assumption of the theorem and applying Lemma \ref{lem-1}, we easily obtain that $||A_n||\leq\frac{1-|a_0|^2}{1+\gamma}$ for all $n\in\mathbb{N}$. Using this, it follows that
\begin{eqnarray*}
M_{T^{f}_{\beta}}(r)&:=& \sum_{n=0}^{\infty}\left( \frac{1}{n+1}\sum_{k=0}^{n}\frac{\Gamma\left( n-k +\beta\right)}{\Gamma\left(n-k+1\right)\Gamma(\beta)} ||A_k||\right)r^n\nonumber\\&=&\sum_{n=0}^{\infty}\left( \frac{1}{n+1}\frac{\Gamma\left( n+\beta\right)}{\Gamma\left(n+1\right)\Gamma(\beta)}||A_0|| \right)r^n\nonumber\\&+&\sum_{n=0}^{\infty}\left( \frac{1}{n+1}\sum_{k=1}^{n}\frac{\Gamma\left( n-k +\beta\right)}{\Gamma\left(n-k+1\right)\Gamma(\beta)} ||A_k||\right)r^n\nonumber\\&\leq&|a_0|\sum_{n=0}^{\infty}\left( \frac{1}{n+1}\frac{\Gamma\left(n+\beta\right)}{\Gamma\left(n+1\right)\Gamma(\beta)} \right)r^n\nonumber\\&+& \frac{1-|a_0|^2}{1+\gamma}\;\sum_{n=0}^{\infty}\left( \frac{1}{n+1}\sum_{k=1}^{n}\frac{\Gamma\left( n-k +\beta\right)}{\Gamma\left(n-k+1\right)\Gamma(\beta)}
\right)r^n.\end{eqnarray*}
From the above inequality, it is easily seen that \begin{eqnarray}\label{e4.1a}    M_{T^{f}_{\beta}}(r)&\leq&\frac{|a_0|}{r}\int_{0}^{r}\frac{1}{(1-t)^\beta}dt\;+\frac{(1-|a_0|^2)}{(1+\gamma)}\frac{1}{r}\int_0^{r}\frac{t}{(1-t)^{\beta+1}}\nonumber\\&\leq&\frac{|a_0|}{r}\int_{0}^{r}\frac{1}{(1-t)^\beta}dt\;+\frac{(1-|a_0|^2)}{(1+\gamma)}\frac{1}{r}\left(\int_0^{r}\frac{1}{(1-t)^{\beta+1}}-\int_0^{r}\frac{1}{(1-t)^{\beta}}\right)\nonumber\\&=&\left(\frac{|a_0|}{r}-\frac{1-|a_0|^2}{(1+\gamma)r}\right)\int_0^r\frac{1}{(1-t)^\beta}dt+ \frac{(1-|a_0|^2)}{(1+\gamma)}\frac{1}{r} \int_0^r\frac{1}{(1-t)^{\beta+1}}dt\nonumber\\&=&\left(\frac{|a_0|}{r}-\frac{1-|a_0|^2}{(1+\gamma)r}    \right)\frac{1-(1-r)^{1-\beta}}{1-\beta}+ \frac{(1-|a_0|^2)}{(1+\gamma)}\frac{1}{r} \left(\frac{1-(1-r^{-\beta})}{-\beta}\right)\nonumber\\&=&\frac{1}{r(1+\gamma)}\left(|a_0|(1+\gamma)-(1-|a_0|^2)\right)\frac{1-(1-r)^{1-\beta}}{1-\beta}\nonumber\\&+&(1-|a_0|^2)(\frac{(1-r)^{-\beta}-1}{\beta}):=\Phi(|a_0|,r).\end{eqnarray}
Now differentiating $\Phi(|a_0|,r)$ twice with respect to $|a_0|$, we get
\begin{eqnarray*}\Phi^{\prime}(|a_0|,r)=\frac{1}{r(1+\gamma)}\left((1+\gamma+2|a_0|)\frac{1-(1-r)^{1-\beta}}{1-\beta}  -2|a_0|\frac{(1-r)^{-\beta}-1}{\beta}\right)\end{eqnarray*} and 
\begin{eqnarray*} \Phi^{\prime\prime}(|a_0|,r)= \frac{2}{r(1+\gamma)}\left(\frac{1-(1-r)^{1-\beta}}{1-\beta}-\frac{(1-r)^{-\beta}-1}{\beta}\right).\end{eqnarray*}
\par  Let us set \begin{eqnarray*} F(x):=\frac{1-x^{1-\beta}}{1-\beta}-\frac{x^{-\beta}-1}{\beta},\;\;\text{where}\;\; x=1-r.\end{eqnarray*} Then one can easily see that $F(x)$ is monotonic increasing function of $x\in[0,1)$. This implies that $F(x)\leq F(1)=0$, and thus $\Phi^{\prime\prime}(|a_0|,r)\leq0$, which implies that \begin{eqnarray}\label{e4.1}\Phi'(|a_0|,r)\geq  \Phi'(1,r)=\frac{1}{r(1+\gamma)}\left((3+\gamma)\frac{1-(1-r)^{1-\beta}}{1-\beta} -2\frac{(1-r)^{-\beta}-1}{\beta}\right).\end{eqnarray}
\par Consider the function 
\begin{eqnarray*} G(x):= (3+\gamma)\frac{1-x^{1-\beta}}{1-\beta} -2\frac{x^{-\beta}-1}{\beta},\;\;\text{where}\;\;x=1-r.\end{eqnarray*}
A simple calculation shows that $G^{\prime}(x)\geq0$ for $r\leq(1+\gamma)/(3+\gamma)$, i.e., $G(x)$ is monotonic increasing for $r\leq(1+\gamma)/(3+\gamma)$. Hence $	\Phi'(1,r)$ is monotonic increasing for $r\leq(1+\gamma)/(3+\gamma)$.
\par Again we observe that $\Phi'(1,r)|_{r=0}=0$ and $\lim_{r \to1^-}\Phi'(1,r)\rightarrow-\infty$. Therefore, $\Phi'(1,r)\geq0$ for $r\leq R_{\beta}$, where $R_{\beta}$ is the smallest positive root of the equation \begin{eqnarray*} (3+\gamma)\frac{1-(1-x)^{1-\beta}}{1-\beta}-2\frac{(1-x)^{-\beta}-1}{\beta}=0.\end{eqnarray*}
\par Therefore, in view of \eqref{e4.1}, it follows that $\Phi'(|a_0|,r)\geq0$ for $0\leq r\leq R_\beta$. This implies that \begin{eqnarray*}\Phi(|a_0|,r) \leq \Phi(1)= \frac{1}{r}\left(\frac{1-(1-r)^{1-\beta}}{1-\beta}\right).\end{eqnarray*}
Hence in view of \eqref{e4.1a} and the above inequality, we obtain \begin{eqnarray*} M_{T^{f}_{\beta}}(r) \leq\frac{1}{r}\left(\frac{1-(1-r)^{1-\beta}}{1-\beta}\right)\end{eqnarray*} for $r\leq R_{\beta}$, where $R_{\beta}$ is the smallest positive root of the equation \begin{eqnarray*} (3+\gamma)\frac{1-(1-x)^{1-\beta}}{1-\beta}-2\frac{(1-x)^{-\beta}-1}{\beta}=0.\end{eqnarray*}
                                                                                                                              
\par In order to show that the radius is sharp, we consider the function
\begin{eqnarray*} \chi_\gamma(z)=\left(\frac{a-\gamma-(1-\gamma)z}{1-a\gamma-a(1-\gamma)z}\right)I,\end{eqnarray*} where  $z\in\Omega_\gamma$, $a\in(0,1)$, $0\leq\gamma<\frac{1}{2-a}$ and $I$ is the identity operator on $\mathcal{B}(\mathcal{H})$.
\par Let us define the function $L:\Omega_\gamma\to\mathbb{D}$ by $L(z):=(1-\gamma)z+\gamma$ and $\phi:\mathbb{D}\to\mathbb{D}$ by \begin{eqnarray*} \phi(z):=\frac{a-z}{1-az}\qquad\text{for }a\in(0,1).\end{eqnarray*}
\par It can be easily seen that $\chi_\gamma=\phi\circ L$ maps $\Omega_\gamma$ univalently onto
$\mathbb{D}$. After simple computations, we see that \begin{eqnarray}\label{e4.2}
\chi_\gamma(z)=\chi_a(z)I=A_0-\sum_{n=1}^{\infty}A_nz^n,\qquad z\in\mathbb{D},\end{eqnarray} where  $A_0=\left(\frac{a-\gamma}{1-a\gamma}\right)I$
and $A_n=\left(\frac{1-a^2}{a(1-a\gamma)}\left(\frac{a(1-\gamma)}{1-a\gamma}\right)^n\right)I\;\;\text{for }n\in\mathbb{N}.$
\par Now \begin{eqnarray}\label{e4.4} M_{T^{f}_{\beta}}(r)&=&\sum_{n=0}^{\infty}\left(\frac{1}{n+1}\sum_{k=0}^{n}	\frac{\Gamma(n-k+\beta)}{\Gamma(n-k+1)\Gamma(\beta)} ||A_k||\right)r^n\nonumber\\&=&\sum_{n=0}^{\infty}\frac{|| A_0||}{n+1}
\frac{\Gamma(n+\beta)}{\Gamma(n+1)\Gamma(\beta)}r^{n}+\sum_{n=0}^{\infty}\left(\frac{1}{n+1}\sum_{k=1}^{n}\frac{\Gamma(n-k+\beta)}{\Gamma(n-k+1)\Gamma(\beta)}||A_k||\right)r^n\nonumber\\&=&\frac{(1-a^2)(1-\gamma)}{(1-a\gamma)^2}\sum_{n=0}^\infty \left(\frac{1}{n+1} \sum_{k=1}^n \frac{\Gamma(n-k+\beta)}{\Gamma(n-k+1) \Gamma(\beta)} \left(\frac{a(1-\gamma)}{1-a\gamma}\right)^n\right)r^n\nonumber\\&+&\frac{a-\gamma}{(1-a\gamma)r}\left(\frac{1-(1-r)^{1-\beta}}{1-\beta}\right).\end{eqnarray}
\par A simple computation shows that \begin{eqnarray}\label{e4.5} && \frac{a-\gamma}{(1-a\gamma)r}\left(\frac{1-(1-r)^{1-\beta}}{1-\beta}\right)\nonumber\\&&=\frac{1}{r}\;\frac{1-(1-r)^{1-\beta}}{1-\beta}-\frac{1-a}{r(1-a\gamma)}\left[ (3+\gamma)\frac{1-(1-r)^{1-\beta}}{1-\beta}-2\frac{(1-r)^{-\beta}-1}{\beta}\right]\nonumber\\&&+\frac{2(1-a)}{r(1-a\gamma)}\left[ \frac{1-(1-r)^{1-\beta}}{1-\beta}-\frac{(1-r)^{-\beta}-1}{\beta}\right]\nonumber\\&&=\frac{1}{r}\;\frac{1-(1-r)^{1-\beta}}{1-\beta}-\frac{1-a}{r(1-a\gamma)}\left[ (3+\gamma)\frac{1-(1-r)^{1-\beta}}{1-\beta}-2\frac{(1-r)^{-\beta}-1}{\beta}\right]\nonumber\\&&+\frac{2(1-a)}{r(1-\gamma)}\left[ \frac{1-(1-r)^{1-\beta}}{1-\beta}-\frac{(1-r)^{-\beta}-1}{\beta}\right]+O(1-a)^2.\end{eqnarray}
\par In view of \eqref{e4.4} and \eqref{e4.5}, one can easily observe that $M_{T^{f}_{\beta}}(r)>\frac{1}{r}\;\frac{1-(1-r)^{1-\beta}}{1-\beta}$ whenever  $r>R_\beta$ and $a\rightarrow1^-$. This proves the sharpness.

<!-- proof-2: body_md -->
In view of Lemma \ref{lem-1}, we easily obtain that $||A_n||\leq\frac{1-|a_0|^2}{1+\gamma}$ for all $n\in\mathbb{N}$. Using this, we obtain 
\begin{eqnarray}\label{e5.10} M_V^f(r)&=&\sum_{n=0}^{\infty}\frac{||A_n||}{n+1}r^{n+1}=||A_0||r+\sum_{n=1}^{\infty}\frac{||A_n||}{n+1}r^{n+1}\nonumber\\&=& |a_0|r+\frac{1-|a_0|^2}{1+\gamma}\sum_{n=1}^{\infty}\frac{r^{n+1}}{n+1}=|a_0|r-\frac{1-|a_0|^2}{1+\gamma}\;[\ln(1-r)+r]\nonumber\\&:=&\Phi(a_0,r).\end{eqnarray}
\par Differentiating $\Phi(a_0,r)$ twice with respect to $a_0$, we have  
\begin{eqnarray*} \Phi^{\prime}(a_0,r)&=&r+\frac{2a_0}{1+\gamma}[\ln (1-r)+r],\\\Phi^{\prime\prime}(a_0,r)&=&\frac{2}{1+\gamma}[\ln (1-r)+r]\leq 0\;\;\text{for}\;\;0<r\leq 1.\end{eqnarray*}
\par This implies that $\Phi^{\prime}(a_0,r)\geq\Phi^{\prime}(1,r)=r+\frac{2}{1+\gamma}\;[\ln (1-r)+r]$, which upon simplification gives \begin{eqnarray*} \Phi^{\prime}(a_0,r)\geq\Phi^{\prime}(1,r) &=&\frac{1}{1+\gamma}\left[(3+\gamma)r+2\ln (1-r)\right]\\&:=&\frac{F(r)}{1+\gamma}.\end{eqnarray*}
\par We observe that $F(0)=0$, $F^{\prime}(r)\geq0$ for $r\leq(1+\gamma)/(3+\gamma)$, $\lim_{r\to 1^-}F(r)=-\infty$, and hence $F(x)=0$ must have a positive root, say $x=R_\gamma$ in $(0,1)$, which depens on $\gamma$. This implies that $\Phi^{\prime}(a_0,r)\geq0$ for $r\leq R_\gamma$. Therefore, in view of \eqref{e5.10}, we obtain \begin{eqnarray*} M_V^f(r)=\Phi(a_0,r)\leq \Phi(1,r)=r.\end{eqnarray*}
\par It remain to prove the sharpness of the result. For this we consider the same function as defined in \eqref{e4.2}, where $a\in(0,R_\gamma)$.
\par Now \begin{eqnarray}\label{e5.11} M_V^{\chi}(r)&=&\sum_{n=0}^{\infty}\frac{||A_n||}{n+1}r^{n+1}=||A_0||r+\sum_{n=1}^{\infty}\frac{||A_n||}{n+1}r^{n+1}\nonumber\\&=& ||A_0||r+\frac{1-||A_0||^2}{1+\gamma}\sum_{n=1}^{\infty}\frac{r^{n+1}}{n+1}\nonumber\\&=&\frac{r(a-r)}{1-ar}+\frac{1-a^2}{a(1-ar)}\sum_{n=1}^{\infty}\left(\frac{a(1-r)}{1-ar}\right)^n\frac{r^{n+1}}{n+1}\nonumber\\&=& r+\frac{r(a-1)(1+\gamma)}{1-a\gamma}+\frac{1-a^2}{a(1-ar)}\sum_{n=1}^{\infty}\left(\frac{a(1-r)}{1-ar}\right)^n\frac{r^{n+1}}{n+1}\nonumber\\&=& r+\frac{(a-1)(1+\gamma)^2}{(1-a\gamma)(3+\gamma)}\left(\frac{r(3+\gamma)}{1+\gamma}+\frac{2\ln (1-r)}{1+\gamma}\right)\nonumber\\&+&\frac{1-a^2}{a(1-ar)}\sum_{n=1}^{\infty}\left(\frac{a(1-r)}{1-ar}\right)^n\frac{r^{n+1}}{n+1}-\frac{2(a-1)(1+\gamma)}{(1-a\gamma)(3+\gamma)}\ln (1-r)\nonumber\\&=& r-\frac{(1-a)(1+\gamma)}{(1-a\gamma)(3+\gamma)}[(3+\gamma)r+2\ln (1-r)]+E_{a,\gamma}(r),\end{eqnarray} where 
\begin{eqnarray*} && E_{a,\gamma}(r)=\frac{1-a}{1-a\gamma}\left[\frac{1+a}{a}\sum_{n=1}^{\infty}\left(\frac{a(1-\gamma)}{1-a\gamma}\right)^n\frac{r^{n+1}}{n+1}+\frac{2(1+\gamma)}{3+\gamma}\ln (1-r)\right]\\&&= \frac{1-a}{1-a\gamma}\left[\frac{(1+a)(1-a\gamma)}{a^2(1-\gamma)}\left(-\ln(1-\frac{r(1-a\gamma)}{a(1-\gamma)})-\frac{r(1-a\gamma)}{a(1-\gamma)}\right)+\frac{2(1+\gamma)\ln (1-r)}{3+\gamma}\right].\end{eqnarray*}
\par Now after simple computations it is easy to observe that $E_{a,\gamma}(0)=0$ and $E_{a,\gamma}^{\prime}(r)> 0$ for $r>R_\gamma$. Therefore, $E_{a,\gamma}>0$ for $r>R_\gamma$. 
\par Hence in view of \eqref{e5.11} and the fact that $(3+\gamma)r+2\ln (1-r)<0$ for $r>R_\gamma$, it is easily seen that $M_V^\chi(r)>r$. This shows the sharpness.
