<!-- theo-1: body_md -->
\textbf{(Monge-Kantorovich)}
	Take a pair $\mu,\nu\in P(M)$ and let $p\geq 1$. Then the infimum
$$C_p(\mu,\nu):=\inf_{\pi\in\Pi(\mu,\nu)}\int_{M\times M}d^p(x,y)\, d\pi(x,y)$$
is attained. Any $\pi\in\Pi(\mu,\nu)$ attaining it is called an \textbf{optimal coupling} between $\mu$ and $\nu$.
<!-- end theo-1 -->
<!-- definition-1: body_md -->
The \textbf{Wasserstein distance} of order $p\geq 1$ between two probability measures $\mu$ and $\nu$ on $M$ is defined by
\begin{equation}\label{Wp}
	W_p(\mu,\nu):=\left(\inf_{\pi\in\Pi(\mu,\nu)}\int_{M^2}d(x,y)^p\, d\pi(x,y)\right)^{1/p}=C_p(\mu,\nu)^{1/p}.
\end{equation}
<!-- end definition-1 -->
<!-- theo-2: body_md -->
\cite[Theorem 6.9]{villani2009optimal}
    If $(\mu_k)_{k\in\mathbb{N}}$ is a sequence in ${P}_p(M)$ and $\mu$ is a measure in ${P}(M)$, then $(\mu_k)_{k\in\mathbb{N}}$ converges to $\mu$ in the weak topology if and only if $W_p(\mu_k,\mu)\xrightarrow[]{k\to\infty}0$.
<!-- end theo-2 -->
<!-- definition-2: body_md -->
A function $\psi:M\to\mathbb{R}\cup\{+\infty\}$ is said to be \textbf{$d^2/2$-convex} if it is not identically $+\infty$ and there is a function $\phi:M\to\mathbb{R}\cup\{\pm\infty\}$ such that for every $x\in M$, $\psi(x)=\sup_{y\in M}\{\phi(y)-d^2(x,y)/2\}$.
<!-- end definition-2 -->
<!-- theo-3: body_md -->
\label{theo1.1}
    Let $\mu_t$ be an absolutely continuous curve on $P(M)$, with $\mu_0$ absolutely continuous with respect to the volume measure. Then there exists a Borel vector field $Z_t$ on $M$ such that
    $$\int_0^1\int_M\left\lVertZ_t(x)\right\rVert^2_{T_xM}d\mu_tdt<+\infty$$
    and the following conservation of mass formula 
    \begin{equation}
        \label{conservation} \dot{\mu_t}+\nabla\cdot (\mu_t Z_t)=0
    \end{equation}
    holds in the sense of distributions. Uniqueness to (\ref{conservation}) holds if moreover $Z_t$ is imposed to be in
    $$\overline{\{\nabla\psi,\,\psi\in C^\infty(M)\}}^{L^2(\mu_t)}.$$
<!-- end theo-3 -->
<!-- prop-1: body_md -->
\label{prop:constant-field}
	Fix $\psi\in C^\infty(M)$ and denote by $\phi_t:M\to M$ the flow of $\nabla\psi$. Then, for any $\mu_0\in P(M)$, the curve $\mu_t:=(\phi_t)_*\mu_0$ has constant velocity field, that is, $\dot{\mu_t}=V_\psi|_{\mu_t}$ for every $t$.
<!-- end prop-1 -->
<!-- proof-1: body_md -->
Since $\mu_t=(\phi_t)_*\mu_0$ and $\phi_t$ is the flow of $\nabla\psi$,
	\begin{align*}
		\frac{d}{dt}\int_Mf(x)\mu_t(dx)&=\frac{d}{dt}\int_Mf(\phi_{t}(x))\mu_0(dx)=\int_M \langle\nabla f(\phi_t(x)),\nabla\psi(\phi_t(x))\rangle \mu_0(dx)\\
		&=\int_M \langle\nabla f(x), \nabla\psi(x)\rangle \mu_t(dx)=-\int_Mf\,\nabla\cdot(\mu_t \nabla\psi)(dx),
	\end{align*}
	so that $\dot{\mu_t}=V_{\psi}|_{\mu_t}$.
<!-- end proof-1 -->
<!-- prop-2: body_md -->
\label{prop:lie-bracket}
    Given $\phi_1,\phi_2\in C^\infty(M)$, the Lie bracket of constant vector fields is given by
    $$[V_{\phi_1},V_{\phi_2}]=V_{\theta_\mu}$$
    with
    $$\theta_\mu=\Delta_\mu^{-1}\operatorname{div}_\mu(\nabla\phi_1\Delta_\mu\phi_2-\nabla\phi_2\Delta_\mu\phi_1)=G_\mu d_\mu^*\flat(\nabla^2\phi_2(\nabla\phi_1)-\nabla^2\phi_1(\nabla\phi_2)).$$
<!-- end prop-2 -->
<!-- proof-2: body_md -->
Let $\phi^i_t$ denote the flow of $\nabla\phi_i$, so that by Proposition \ref{prop:constant-field} the curve $(\phi^i_t)_*\mu$ has constant velocity field $V_{\phi_i}$. Then
  	\begin{align*}
  		[V_{\phi_1},V_{\phi_2}]F_f(\mu)&=(V_{\phi_1}(V_{\phi_2}))F_f(\mu)-(V_{\phi_2}(V_{\phi_1}))F_f(\mu)\\
       &=\left.\frac{d}{dt}\right|_{t=0}(V_{\phi_2}F_f)((\phi^1_t)_*\mu)-\left.\frac{d}{dt}\right|_{t=0}(V_{\phi_1}F_f)((\phi^2_t)_*\mu)\\
       & =\left.\frac{d}{dt}\right|_{t=0}\int_M\langle\nabla f(x),\nabla\phi_2(x)\rangle ((\phi^1_t)_*\mu)(dx)\\
       & -\left.\frac{d}{dt}\right|_{t=0}\int_M\langle\nabla f(x),\nabla\phi_1(x)\rangle ((\phi^2_t)_*\mu)(dx)\\
       & =-\int_M\langle \nabla f,\nabla\phi_2\rangle\,\Delta_\mu \phi_1\, d\mu+\int_M\langle \nabla f,\nabla\phi_1\rangle\,\Delta_\mu \phi_2\, d\mu\\
	   &=\int_M\langle\nabla f,\nabla\phi_1\Delta_\mu\phi_2-\nabla\phi_2\Delta_\mu\phi_1\rangle\, d\mu.
\end{align*}
        By \cite[Proposition 3.2]{ding2021geometry}, the projection of $\mathcal{C}_{\phi_1,\phi_2}(\mu)=\nabla\phi_1\Delta_\mu\phi_2-\nabla\phi_2\Delta_\mu\phi_1$ onto $T_\mu P^\infty(M)$ is $V_{\theta_\mu}$ with
    $$\theta_\mu=\Delta_\mu^{-1}\operatorname{div}_\mu(\nabla\phi_1\Delta_\mu\phi_2-\nabla\phi_2\Delta_\mu\phi_1).$$
Thus,
    $$[V_{\phi_1},V_{\phi_2}]F_f(\mu)=V_{\theta_\mu}F_f(\mu).$$
    This formula can be extended to polynomials and by continuity to smooth functions so that we have the identification
    $$[V_{\phi_1},V_{\phi_2}]=V_{\theta_\mu}.$$

    Note that $\Delta_\mu^{-1}\operatorname{div}_\mu=G_\mu d^*_\mu\flat$, since $\Delta_\mu=-d^*_\mu d$ and $d^*_\mu\flat(\xi)=-\operatorname{div}_\mu(\xi)$, so that the two signs cancel. The identity
\begin{equation}\label{eq:bracket-potential}
	\theta_\mu=G_\mu d_\mu^*\flat(\nabla^2\phi_2(\nabla\phi_1)-\nabla^2\phi_1(\nabla\phi_2))
\end{equation}
is then \cite[Equation 5.11]{lott2006some}.
<!-- end proof-2 -->
<!-- lemma-1: body_md -->
\label{Lemma4}
    At $\mu=\rho\,d\operatorname{vol}$, we have $\overline{\nabla}_{V_{\phi_1}}V_{\phi_2}= V_\phi$, where
$$\phi= G_\mu d_\mu^*\flat(\nabla^2\phi_2(\nabla \phi_1)).$$
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
The proof can be found in \cite[Lemma 4]{lott2006some} in local coordinates. We present it here adapted to our intrinsic language. Let $\Pi_\mu$ denote the orthogonal projection onto $\overline{\operatorname{Im}(d)}\subset \Omega^1_{L^2}(M,\mu)$, so that $dG_\mu d^*_\mu=\Pi_\mu$ by the definition of $G_\mu$. Given $\phi_3\in C^\infty(M)$, and using that $d\phi_3\in\operatorname{Im}(d)$ to drop $\Pi_\mu$ in the third equality,
    \begin{align*}
        \langle V_{\phi_3},V_\phi\rangle_\mu & =\int_M\langle d\phi_3,dG_\mu d_\mu^*\flat(\nabla^2\phi_2(\nabla \phi_1))\rangle\, d\mu\\
        & =\int_M\langle d\phi_3,\Pi_\mu\flat(\nabla^2\phi_2(\nabla \phi_1))\rangle\, d\mu= \int_M\langle d\phi_3,\flat(\nabla^2\phi_2(\nabla \phi_1))\rangle\, d\mu\\
    	&=\int_M\langle \nabla\phi_3,\nabla^2\phi_2(\nabla \phi_1)\rangle\, d\mu=\int_M\langle \nabla\phi_1,\nabla^2\phi_2(\nabla \phi_3)\rangle\, d\mu=\langle V_{\phi_3},\overline{\nabla}_{V_{\phi_1}}V_{\phi_2}\rangle_\mu,
	\end{align*}
    where the second to last equality uses the symmetry of $\nabla^2\phi_2$ and the last one is (\ref{prodconnect}). Since this holds for every $\phi_3\in C^\infty(M)$ and the metric is non-degenerate on $T_\mu P^\infty(M)$, we conclude that $V_\phi=\overline{\nabla}_{V_{\phi_1}}V_{\phi_2}$.
<!-- end proof-3 -->
<!-- theo-4: body_md -->
\label{thm:parallelization}
	Let $\{\phi_i\}_{i\geq1}\subset C^\infty(M)$ be eigenfunctions of the Laplace-Beltrami operator, $\Delta\phi_i=-\lambda_i\phi_i$ with $\lambda_i>0$, such that $\{1\}\cup\{\phi_i\}_{i\geq1}$ is an orthogonal basis of $L^2(\operatorname{vol})$. Then, for every $\mu\in P^\infty(M)$, the following hold.
	\begin{enumerate}
		\item Every $X\in L^2_0(\mu)$ has an expansion $X=\sum_ic_i\nabla\phi_i$ converging in $L^2(\mu)$, and the coefficients are unique: if $\sum_ic_i\nabla\phi_i$ converges to $0$ in $L^2(\mu)$, then $c_i=0$ for every $i$. Equivalently, $\{V_{\phi_i}\}_{i\geq1}$ is a Schauder basis of the Hilbert space $T_\mu P(M)$ with Otto's inner product (\ref{ottoprod}).
		\item If $\psi\in C^\infty(M)$, then $V_\psi=\sum_ic_iV_{\phi_i}$ with $c_i=\langle\psi,\phi_i\rangle_{L^2(\operatorname{vol})}/\left\lVert\phi_i\right\rVert^2_{L^2(\operatorname{vol})}$, and $\sum_ic_i\phi_i$ converges to $\psi-\int_M\psi\,d\operatorname{vol}$ in every $C^k(M)$.
	\end{enumerate}
	Since the family does not depend on $\mu$, $\{V_{\phi_i}\}_{i\geq1}$ is a global frame for $TP^\infty(M)$.
<!-- end theo-4 -->
<!-- proof-4: body_md -->
Write $\mu=\rho\,\operatorname{vol}$ with $\rho\in C^\infty(M)$ positive. Since $M$ is compact and $\rho$ is continuous and positive, $m:=\inf_M\rho>0$ and $K:=\sup_M\rho<\infty$, so that
	$$m\left\lVert\xi\right\rVert^2_{L^2(\operatorname{vol})}\leq\left\lVert\xi\right\rVert^2_{L^2(\mu)}\leq K\left\lVert\xi\right\rVert^2_{L^2(\operatorname{vol})}$$
	for every $\xi\in L^2(\operatorname{vol})$. The norms $L^2(\operatorname{vol})$ and $L^2(\mu)$ are therefore equivalent, both on functions and on vector fields, and in particular $L^2_0(\mu)=L^2_0(\operatorname{vol})$ as sets.

	We prove (2) first. For $\psi\in C^\infty(M)$ and $k\geq0$, integrating by parts $k$ times gives $\langle\psi,\phi_i\rangle_{L^2(\operatorname{vol})}=(-\lambda_i)^{-k}\langle\Delta^k\psi,\phi_i\rangle_{L^2(\operatorname{vol})}$, so the coefficients $c_i$ decay faster than any power of $\lambda_i$. By Weyl's law and the bounds $\left\lVert\phi_i\right\rVert_{C^k}\leq C_k(1+\lambda_i)^{N_k}\left\lVert\phi_i\right\rVert_{L^2(\operatorname{vol})}$, which follow from elliptic regularity and the Sobolev embedding, the series $\sum_ic_i\phi_i$ converges in every $C^k(M)$. Its limit is $\psi-\int_M\psi\,d\operatorname{vol}$, because this is its limit in $L^2(\operatorname{vol})$. In particular $\nabla\psi=\sum_ic_i\nabla\phi_i$ uniformly on $M$, hence in $L^2(\mu)$, and $V_\psi=\sum_ic_iV_{\phi_i}$ since $V_\mu$ is an isometry by (\ref{ottoprod}).

	For (1), integration by parts gives $\langle\nabla\phi_i,\nabla\phi_j\rangle_{L^2(\operatorname{vol})}=\lambda_j\langle\phi_i,\phi_j\rangle_{L^2(\operatorname{vol})}$, so the gradients $\nabla\phi_i$ are orthogonal in $L^2(\operatorname{vol})$, with $\left\lVert\nabla\phi_i\right\rVert^2_{L^2(\operatorname{vol})}=\lambda_i\left\lVert\phi_i\right\rVert^2_{L^2(\operatorname{vol})}$. By (2) their closed span in $L^2(\operatorname{vol})$ contains every gradient of a smooth function, so it is $L^2_0(\operatorname{vol})$, and the normalized gradients form an orthonormal basis of $L^2_0(\operatorname{vol})$. Every $X\in L^2_0(\mu)=L^2_0(\operatorname{vol})$ is then the sum of its expansion in $L^2(\operatorname{vol})$, hence in $L^2(\mu)$ by the equivalence of norms. If $\sum_ic_i\nabla\phi_i$ converges to $0$ in $L^2(\mu)$, it converges to $0$ in $L^2(\operatorname{vol})$, and taking the inner product with $\nabla\phi_j$ gives $c_j\lambda_j\left\lVert\phi_j\right\rVert^2_{L^2(\operatorname{vol})}=0$, so $c_j=0$. The statement about $T_\mu P(M)$ follows because $V_\mu$ is an isometry.

\label{prop:parallel}
	Let $\mu_t=\rho_t\,d\operatorname{vol}$ be a smooth curve in $P^\infty(M)$ with $\dot{\mu}_t=V_{\psi_t}$, and let $\eta_t\in C^\infty(M)$ depend smoothly on $t$. The vector field $V_{\eta_t}$ along $\mu_t$ is parallel if and only if one of the following equivalent conditions holds for every $t$:
        \begin{enumerate}
            \item $\dot{\eta}_t+G_{\mu_t}d^*_{\mu_t}\flat\big(\nabla^2\eta_t(\nabla\psi_t)\big)$ is constant on $M$,
            \item $2\dot{\eta}_t+\langle\nabla\eta_t,\nabla\psi_t\rangle+\theta_t$ is constant on $M$,
        \end{enumerate}
    where $\theta_t=\Delta_{\mu_t}^{-1}\operatorname{div}_{\mu_t}(\nabla \psi_t\Delta_{\mu_t}\eta_t-\nabla\eta_t\Delta_{\mu_t} \psi_t)$ is the potential of $[V_{\psi_t},V_{\eta_t}]$ given by Proposition \ref{prop:lie-bracket}.

<!-- proof-5: body_md -->
For fixed $s$, $V_{\eta_s}$ is a vector field on $P^\infty(M)$, and the field along the curve is $t\mapsto V_{\eta_t}|_{\mu_t}$. Since $\eta\mapsto V_\eta$ is linear, the first term is $V_{\dot{\eta}_t}$. Lemma \ref{Lemma4} with $\phi_1=\psi_t$ and $\phi_2=\eta_t$ then gives

	Since $\rho_t>0$ and $M$ is connected, $V_\chi=0$ exactly when $\chi$ is constant, and this gives (1). By (\ref{theo3.3}) and Proposition \ref{prop:lie-bracket}, $\overline{\nabla}_{V_{\psi_t}}V_{\eta_t}=\frac12V_{\langle\nabla\psi_t,\nabla\eta_t\rangle}+\frac12V_{\theta_t}$, and this gives (2).

<!-- theo-5: body_md -->
\label{riemanniancurvature}
    Given $\phi_1,\phi_2,\phi_3,\phi_4\in C^\infty(M)$, the Riemannian curvature operator $\overline{R}$ of $P^\infty(M)$ is given by 
    \begin{align*}
        \langle \overline{R}(V_{\phi_1},V_{\phi_2})V_{\phi_3},V_{\phi_4}\rangle_\mu & = \int_M \langle R(\nabla\phi_1,\nabla\phi_2)\nabla\phi_3,\nabla\phi_4\rangle d\mu -2\langle T_{\phi_1\phi_2},T_{\phi_3\phi_4}\rangle_\mu\\
    &+\langle T_{\phi_2\phi_3},T_{\phi_1\phi_4}\rangle_\mu-\langle T_{\phi_1\phi_3},T_{\phi_2\phi_4}\rangle_\mu.
    \end{align*}
<!-- end theo-5 -->
<!-- proof-8: body_md -->
By (\ref{prodconnect}),
	$$V_{\phi_1}\langle\overline{\nabla}_{V_{\phi_2}}V_{\phi_3},V_{\phi_4}\rangle_\mu= V_{\phi_1}F_{\langle\nabla\phi_2,\nabla^2\phi_3(\nabla\phi_4)\rangle}(\mu)=F_{\langle \nabla\phi_1,\nabla\langle\nabla\phi_2,\nabla^2\phi_3(\nabla\phi_4)\rangle\rangle}(\mu).$$
    Analogously,
    $$V_{\phi_2}\langle\overline{\nabla}_{V_{\phi_1}}V_{\phi_3},V_{\phi_4} \rangle_\mu=F_{\langle \nabla\phi_2,\nabla\langle\nabla\phi_1,\nabla^2\phi_3(\nabla\phi_4)\rangle\rangle}(\mu).$$

        From Lemma \ref{Lemma4}, (\ref{defT}) and the definition of the metric of $P^\infty(M)$,
    \begin{align*}
    \langle\overline{\nabla}_{V_{\phi_2}}V_{\phi_3},\overline{\nabla}_{V_{\phi_1}}V_{\phi_4}\rangle_\mu &=\langle dG_\mu d_\mu^*\flat(\nabla^2\phi_3(\nabla \phi_2)), dG_\mu d_\mu^*\flat(\nabla^2\phi_4(\nabla \phi_1))\rangle_{L^2}\\
	&=\langle \Pi_\mu\flat(\nabla^2\phi_3(\nabla \phi_2)), \Pi_\mu\flat(\nabla^2\phi_4(\nabla \phi_1))\rangle_{L^2}\\
    &= \langle \flat(\nabla^2\phi_3(\nabla \phi_2) ),\flat(\nabla^2\phi_4(\nabla \phi_1))\rangle_{L^2}-\langle T_{\phi_2\phi_3},T_{\phi_1\phi_4}\rangle\\
    & =\int_M\langle \nabla^2\phi_3(\nabla \phi_2),\nabla^2\phi_4(\nabla \phi_1)\rangle d\mu-\langle T_{\phi_2\phi_3},T_{\phi_1\phi_4}\rangle \\
    &=F_{\langle \nabla^2\phi_3(\nabla \phi_2),\nabla^2\phi_4(\nabla \phi_1)\rangle}(\mu)-\langle T_{\phi_2\phi_3},T_{\phi_1\phi_4}\rangle.
    \end{align*}

    Similarly,
    $$\langle\overline{\nabla}_{V_{\phi_1}}V_{\phi_3},\overline{\nabla}_{V_{\phi_2}}V_{\phi_4}\rangle_\mu=F_{\langle \nabla^2\phi_3(\nabla \phi_1),\nabla^2\phi_4(\nabla \phi_2)\rangle}(\mu)-\langle T_{\phi_1\phi_3},T_{\phi_2\phi_4}\rangle$$

    Let us now calculate $\langle \overline{\nabla}_{[V_{\phi_1},V_{\phi_2}]}V_{\phi_3},V_{\phi_4}\rangle_\mu$. By (\ref{eq:bracket-potential}) we may write $[V_{\phi_1},V_{\phi_2}]=V_{\theta_\mu}$ with $$\theta_\mu=G_\mu d_\mu^*\flat(\nabla^2\phi_2(\nabla\phi_1)-\nabla^2\phi_1(\nabla\phi_2)).$$
    Thus, by (\ref{prodconnect}),
    \begin{align*}
        \langle \overline{\nabla}_{[V_{\phi_1},V_{\phi_2}]}V_{\phi_3},V_{\phi_4}\rangle_\mu & =\int_M\langle \nabla\theta_\mu,\nabla^2\phi_3(\nabla\phi_4)\rangle d\mu=\langle d\theta_\mu,\flat(\nabla^2\phi_3(\nabla\phi_4))\rangle_{L^2}\\
    &=\langle dG_\mu d_\mu^*\flat(\nabla^2\phi_2(\nabla\phi_1)-\nabla^2\phi_1(\nabla\phi_2)),\flat(\nabla^2\phi_3(\nabla\phi_4))\rangle_{L^2}\\
    &=\langle \Pi_\mu\flat(\nabla^2\phi_2(\nabla\phi_1)-\nabla^2\phi_1(\nabla\phi_2)),\Pi_\mu\flat(\nabla^2\phi_3(\nabla\phi_4))\rangle_{L^2}\\
        &=\int_M\langle\nabla^2\phi_2(\nabla\phi_1)-\nabla^2\phi_1(\nabla\phi_2),\nabla^2\phi_3(\nabla\phi_4)\rangle d\mu-\langle T_{\phi_1\phi_2},T_{\phi_4\phi_3}\rangle\\
    &+\langle T_{\phi_2\phi_1},T_{\phi_4\phi_3}\rangle\\
    &=\int_M\langle\nabla^2\phi_2(\nabla\phi_1)-\nabla^2\phi_1(\nabla\phi_2),\nabla^2\phi_3(\nabla\phi_4)\rangle d\mu+2\langle T_{\phi_1\phi_2},T_{\phi_3\phi_4}\rangle.
    \end{align*}

    The theorem then follows from the formula
    \begin{align*}
        \langle\overline{R}(V_{\phi_1},V_{\phi_2})V_{\phi_3},V_{\phi_4}\rangle_\mu
    &=V_{\phi_1}\langle\overline{\nabla}_{V_{\phi_2}}V_{\phi_3},V_{\phi_4}\rangle_\mu
    -\langle\overline{\nabla}_{V_{\phi_2}}V_{\phi_3},\overline{\nabla}_{V_{\phi_1}}V_{\phi_4}\rangle_\mu
    \\
    &-V_{\phi_2}\langle\overline{\nabla}_{V_{\phi_1}}V_{\phi_3},V_{\phi_4}\rangle_\mu+\langle\overline{\nabla}_{V_{\phi_1}}V_{\phi_3},\overline{\nabla}_{V_{\phi_2}}V_{\phi_4}\rangle_\mu\\
    &-\langle\overline{\nabla}_{[V_{\phi_1},V_{\phi_2}]}V_{\phi_3},V_{\phi_4}\rangle_\mu,
    \end{align*}
    with the convention $R(X,Y)=\nabla_X\nabla_Y-\nabla_Y\nabla_X-\nabla_{[X,Y]}$, both on $M$ and on $P^\infty(M)$. Substituting the five expressions above, the terms in $T$ give the last three terms of the statement, and it remains to show that the remaining integrands add up to $\langle R(\nabla\phi_1,\nabla\phi_2)\nabla\phi_3,\nabla\phi_4\rangle$. Write $X_i=\nabla\phi_i$ and let $H_i=\nabla^2\phi_i$, viewed as a symmetric endomorphism, so that $\nabla_YX_i=H_iY$. By the Leibniz rule,
    $$\langle X_1,\nabla\langle X_2,H_3X_4\rangle\rangle=\langle H_2X_1,H_3X_4\rangle+\langle(\nabla_{X_1}H_3)X_4,X_2\rangle+\langle H_3X_2,H_4X_1\rangle,$$
    and similarly with $1$ and $2$ exchanged. The terms $\langle H_3X_2,H_4X_1\rangle$ and $\langle H_3X_1,H_4X_2\rangle$ cancel against the second and the fourth terms of the formula, and $\langle H_2X_1-H_1X_2,H_3X_4\rangle$ cancels against the fifth. What remains is
    $$\langle(\nabla_{X_1}H_3)X_4,X_2\rangle-\langle(\nabla_{X_2}H_3)X_4,X_1\rangle=(\nabla_{X_1}\nabla^2\phi_3)(X_2,X_4)-(\nabla_{X_2}\nabla^2\phi_3)(X_1,X_4),$$
    by the symmetry of $\nabla^2\phi_3$. By the Ricci identity for the one-form $d\phi_3$, this equals
    $$-d\phi_3\big(R(X_1,X_2)X_4\big)=\langle R(X_1,X_2)X_3,X_4\rangle .$$

\label{prop:geodesic-circle}

\label{lem:complement}

In dimension one $\nabla^2\psi(\nabla\phi)=\psi''\phi'$, so by Lemma \ref{lem:complement} the form $T_{\phi\psi}$ is the multiple of $\rho^{-1}d\theta$ with coefficient $\big(\int_0^{2\pi}\psi''\phi'\,d\theta\big)/C_\mu$. This is an integral against $d\theta$, not against $\mu$, so the ordinary orthogonality relations of the Fourier basis apply. For $\phi=c_n$ and $\psi=s_n$ the integrand is $n^3s_n^2$, of integral $n^3\pi$. For every other pair of frame elements the integrand is of the form $c_pc_q$, $s_ps_q$ or $s_pc_q$ with $p\neq q$, and hence has integral zero. The antisymmetry is Lemma \ref{lem:T-antisym}.
<!-- end proof-14 -->
<!-- theo-7: body_md -->
\label{thm:curvature-general}
	Let $\mu=\rho\,d\theta\in P^\infty(S^1)$ and set
	$$\alpha_\mu:=\frac{\pi}{C_\mu}\,\rho^{-1}d\theta\in\Omega^1_{L^2}(S^1,\mu),\qquad \omega(u,v):=\sum_{n\geq1}n^3\big(u^c_nv^s_n-u^s_nv^c_n\big),$$
	where $u,v\in C^\infty(S^1)$ have Fourier expansions $u=\sum_n(u^c_nc_n+u^s_ns_n)$ and $v=\sum_n(v^c_nc_n+v^s_ns_n)$, so that the series defining $\omega(u,v)$ converges absolutely by the rapid decay of the Fourier coefficients of a smooth function. Then $T_{uv}=\omega(u,v)\,\alpha_\mu$, and consequently, for every plane in $T_\mu P^\infty(S^1)$ spanned by $V_u$ and $V_v$,
	\begin{equation}\label{eq:curvature-plane}
		\overline{K}\big(V_u,V_v\big)=\frac{3\pi^2\,\omega(u,v)^2}{C_\mu\left(\left\lVertV_u\right\rVert_\mu^2\left\lVertV_v\right\rVert_\mu^2-\langle V_u,V_v\rangle_\mu^2\right)}.
	\end{equation}
<!-- end theo-7 -->
<!-- proof-15: body_md -->
As in the proof of Theorem \ref{thm:curvature}, Lemma \ref{lem:complement} gives
	$$T_{uv}=\frac{1}{C_\mu}\left(\int_0^{2\pi}u'v''\,d\theta\right)\rho^{-1}d\theta .$$
	The Fourier series of $u'=\sum_nn(u^s_nc_n-u^c_ns_n)$ and $v''=-\sum_nn^2(v^c_nc_n+v^s_ns_n)$ converge in $L^2(d\theta)$, so by Parseval's identity
	$$\int_0^{2\pi}u'v''\,d\theta=\pi\sum_{n\geq1}n^3\big(u^c_nv^s_n-u^s_nv^c_n\big)=\pi\,\omega(u,v),$$
	and hence $T_{uv}=\omega(u,v)\,\alpha_\mu$.
	
	For the curvature, recall from the corollary to Theorem \ref{riemanniancurvature} that $\overline{K}=3\left\lVertT\right\rVert^2_\mu$ for an orthonormal pair spanning the plane, the first term vanishing here for two independent reasons: $S^1$ is flat, and, being one-dimensional, its gradients are everywhere proportional, so that the Gram factor $|\nabla\phi_1|^2|\nabla\phi_2|^2-\langle\nabla\phi_1,\nabla\phi_2\rangle^2$ vanishes pointwise. Since $\omega$ is alternating and the Gram determinant is quadratic in the same sense, both the numerator and the denominator of (\ref{eq:curvature-plane}) are multiplied by $\det^2$ under a change of basis of the plane, so the right hand side depends only on the plane and it suffices to check it for an orthonormal pair, where it reduces to $3\omega(u,v)^2\left\lVert\alpha_\mu\right\rVert^2_\mu$. Finally $\left\lVert\alpha_\mu\right\rVert^2_\mu=(\pi/C_\mu)^2\int_0^{2\pi}\rho^{-2}\rho\,d\theta=\pi^2/C_\mu$, which gives (\ref{eq:curvature-plane}).

<!-- corollary-7: body_md -->
\label{cor:isotropic}
	A plane in $T_\mu P^\infty(S^1)$ has vanishing sectional curvature if and only if it is isotropic for the symplectic form $\omega$. In particular, the plane spanned by any two frame fields of distinct frequencies is flat.
<!-- end corollary-7 -->
<!-- proof-16: body_md -->
By (\ref{eq:curvature-plane}) the curvature vanishes if and only if $\omega(u,v)=0$, since $C_\mu$ is finite and the Gram determinant of a genuine plane is strictly positive. For two frame fields of distinct frequencies, $\omega$ vanishes because it only couples indices of equal frequency.

<!-- corollary-8: body_md -->
\label{cor:curvature-fourier}
	For every $n\geq1$,
	\begin{equation}\label{eq:curvature-general}
		\overline{K}(V_{c_n},V_{s_n})=\frac{12\pi^2n^2}{C_\mu\left(1-|\widehat{\mu}(2n)|^2\right)},
	\end{equation}
	where $\widehat{\mu}(k):=\int_0^{2\pi}e^{ik\theta}\,d\mu=\pi(a_k+ib_k)$. In particular $P^\infty(S^1)$ is not flat.

<!-- lemma-4: body_md -->
\label{lem:sharp-cs}
	Let $\mu=\rho\,d\theta\in P^\infty(S^1)$ and let $k\geq1$. Then
	$$\left(\int_0^{2\pi}\rho^{-1}d\theta\right)\left(1-|\widehat{\mu}(k)|^2\right)\geq4\pi^2,$$
	with equality if and only if $\rho\equiv1/2\pi$.
<!-- end lemma-4 -->
<!-- proof-18: body_md -->
Write $m:=\widehat{\mu}(k)$ and set $g(\theta):=1-m\,e^{-ik\theta}$. Since $k\geq1$ we have $\int_0^{2\pi}e^{-ik\theta}d\theta=0$, so $\int_0^{2\pi}g\,d\theta=2\pi$. Moreover, using $\int\rho\,d\theta=1$ and $\int e^{-ik\theta}\rho\,d\theta=\overline{m}$,
	$$\int_0^{2\pi}|g|^2\rho\,d\theta=1-m\overline{m}-\overline{m}m+|m|^2=1-|m|^2.$$
	Applying the Cauchy-Schwarz inequality to the factorization $g=(g\sqrt\rho)\cdot\rho^{-1/2}$ yields
	$$4\pi^2=\left|\int_0^{2\pi}g\,d\theta\right|^2\leq\left(\int_0^{2\pi}|g|^2\rho\,d\theta\right)\left(\int_0^{2\pi}\rho^{-1}d\theta\right)=\left(1-|m|^2\right)\int_0^{2\pi}\rho^{-1}d\theta,$$
	which is the stated inequality. Equality holds if and only if $g\rho$ is constant, that is, $\rho=c/(1-m\,e^{-ik\theta})$ for some $c$. Integrating $g\rho=c$ over $[0,2\pi]$ gives $2\pi c=1-|m|^2$, so $c$ is real, and $c\neq0$ because $\rho>0$. Writing $m=|m|e^{i\alpha}$, the right hand side is real for every $\theta$ only if $|m|\sin(\alpha-k\theta)\equiv0$, which forces $m=0$ and hence $\rho$ constant. Conversely $\rho\equiv1/2\pi$ gives $m=0$ and $\int\rho^{-1}d\theta=4\pi^2$.

\label{cor:curvature-bound}

	$$0<\overline{K}(V_{c_n},V_{s_n})\leq3n^2,$$
	with equality on the right if and only if $\mu=\operatorname{vol}$. Moreover, along any family of measures whose densities degenerate, in the sense that $\int_0^{2\pi}\rho^{-1}d\theta\to\infty$, the sectional curvature of every such plane tends to $0$.

<!-- prop-7: body_md -->
\label{prop:metric-curvature}
	Let $\operatorname{vol}=d\theta/2\pi$ be the uniform measure on $S^1$, let $\phi,\psi\in C^\infty(S^1)$ and put $v:=\nabla\phi$, $w:=\nabla\psi$. Then
	$$\lim_{t\to0}\frac{3}{t^4}\left[t^2\left\lVertv-w\right\rVert^2_{L^2(\operatorname{vol})}-W_2^2\big((\mathrm{Id}+tv)_*\operatorname{vol},\ (\mathrm{Id}+tw)_*\operatorname{vol}\big)\right]=3\left(\int_{S^1} v\,w'\,d\operatorname{vol}\right)^2,$$
	and the right hand side equals $3\left\lVertT_{\phi\psi}\right\rVert^2_{\operatorname{vol}}$. The left hand side is computed from the definition of $W_2$ alone. If the expansion recalled above is applied to Otto's metric, the left hand side is $\big\langle\overline{R}(V_\phi,V_\psi)V_\psi,V_\phi\big\rangle_{\operatorname{vol}}$, and dividing it by the Gram determinant $\left\lVertV_\phi\right\rVert^2_{\operatorname{vol}}\left\lVertV_\psi\right\rVert^2_{\operatorname{vol}}-\langle V_\phi,V_\psi\rangle^2_{\operatorname{vol}}$ recovers $(\ref{eq:curvature-plane})$ at $\mu=\operatorname{vol}$. For $\phi=c_n$ and $\psi=s_n$ that quotient equals $3n^2$, the value given by Corollary \ref{cor:curvature-bound} at $\mu=\operatorname{vol}$, so that the non-flatness of $P^\infty(S^1)$ and the growth of its curvature with the frequency are already visible from the definition of $W_2$.
<!-- end prop-7 -->
<!-- proof-20: body_md -->
The idea is elementary. Both measures are images of $\operatorname{vol}$ under maps close to the identity. On the real line there would be nothing to decide, since the monotone map is optimal and its cost is exactly $t^2\left\lVertv-w\right\rVert^2_{L^2(\operatorname{vol})}$. On the circle that map may be composed with a rigid rotation, so there is a whole family of admissible costs, one for each rotation $\beta$, and $W_2^2$ is the smallest of them. Write $C(\beta)$ for the cost of the rotation by $\beta$, so that $C(0)$ is the value just quoted.

	Everything then depends on whether $\beta=0$ is optimal. It is not, as soon as $C'(0)\neq0$, and the gain is easy to estimate. Near the origin $C$ is a parabola,
	$$C(\beta)\approx C(0)+C'(0)\,\beta+\tfrac12C''(0)\,\beta^2,$$
	whose minimum value is $C(0)-C'(0)^2/2C''(0)$. Completing this square is the whole proof, and the two derivatives are computed in Step 2.

	The order in $t$ is decided there as well, and may be announced now. We will find $C''(0)=2+O(t^2)$, so the denominator is harmless. The term of order $t$ in $C'(0)$, on the other hand, is a multiple of $\int_{S^1}(v-w)\,d\operatorname{vol}$, which vanishes because $v$ and $w$ are derivatives of periodic functions. Hence $C'(0)$ is of order $t^2$, its square is of order $t^4$, and this is the fourth order defect of the statement.

	\emph{Step 1: the cost of a rotation.} Cut the circle at $\theta=0$ and let
	$$F_\mu(\theta):=\mu\big([0,\theta]\big)=\int_0^\theta\rho(\sigma)\,d\sigma\qquad(0\leq\theta<2\pi),$$
	extended to $\mathbb{R}$ by $F_\mu(\theta+2\pi)=F_\mu(\theta)+1$. It is smooth and strictly increasing, and we read its inverse in arclength by setting
	$$\widetilde{Q}_\mu(\theta):=F_\mu^{-1}\big(\theta/2\pi\big),$$
	the quantile function of $\mu$ in this parametrization, which satisfies $\widetilde{Q}_\mu(\theta+2\pi)=\widetilde{Q}_\mu(\theta)+2\pi$ and $\widetilde{Q}_{\operatorname{vol}}=\mathrm{Id}$. Equivalently, $\widetilde{Q}_\mu$ is the increasing map commuting with $\theta\mapsto\theta+2\pi$ that pushes $\operatorname{vol}$ forward to $\mu$, and it is unique up to a translation of its argument, which is the choice of cut point. The only property we use is that it turns a pushforward into the map itself: for $S$ increasing with $S(\theta+2\pi)=S(\theta)+2\pi$,
	$$\widetilde{Q}_{S_*\operatorname{vol}}=S .$$
	In particular $\widetilde{Q}_\mu=\mathrm{Id}+tv$ and $\widetilde{Q}_\nu=\mathrm{Id}+tw$, for $t$ small enough that both maps are increasing.

	By \cite{delon2010fast} the optimal coupling on the circle is monotone up to a rotation, and in this parametrization
	$$W_2^2(\mu,\nu)=\min_{\beta\in\mathbb{R}}C(\beta),\qquad C(\beta):=\int_{S^1}\big|\widetilde{Q}_\mu(\theta)-\widetilde{Q}_\nu(\theta+\beta)\big|^2d\operatorname{vol}(\theta),$$
	where $\beta$ is the rotation measured in arclength, the difference is computed on the lift, and $C$ is convex. The ambiguity in the cut point translates the argument of each quantile function, which the minimization over $\beta$ absorbs. Substituting the two quantile functions, the integrand becomes explicit and all trace of the construction disappears:
	$$C(\beta)=\int_{S^1}\big|t\,v(\theta)-\beta-t\,w(\theta+\beta)\big|^2d\operatorname{vol}(\theta).$$
	Write $L:=\max\left(\left\lVertv\right\rVert_\infty,\left\lVertw\right\rVert_\infty\right)$. Step 4 below checks that every minimizer of $C$ satisfies $|\beta|\leq5tL$, and that on that range the distance computed on the lift is the geodesic distance of $S^1$ once $t<\pi/7L$. All estimates that follow are taken on $|\beta|\leq5tL$.

	\emph{Step 2: the first two derivatives at $\beta=0$.} Differentiating under the integral sign twice,
	$$C'(\beta)=-2\int_{S^1}\big(t\,v(\theta)-\beta-t\,w(\theta+\beta)\big)\big(1+t\,w'(\theta+\beta)\big)\,d\operatorname{vol}(\theta),$$
	$$C''(\beta)=2\int_{S^1}\big(1+t\,w'(\theta+\beta)\big)^2d\operatorname{vol}(\theta)-2t\int_{S^1}\big(t\,v(\theta)-\beta-t\,w(\theta+\beta)\big)\,w''(\theta+\beta)\,d\operatorname{vol}(\theta).$$
	Since $v=\phi'$ and $w=\psi'$ with $\phi$ and $\psi$ periodic, the integral over $S^1$ of a derivative vanishes, which kills four terms at once:
	$$\int_{S^1}v\,d\operatorname{vol}=\int_{S^1}w\,d\operatorname{vol}=\int_{S^1}w'\,d\operatorname{vol}=0,\qquad \int_{S^1}w\,w'\,d\operatorname{vol}=\tfrac12\int_{S^1}\big(w^2\big)'\,d\operatorname{vol}=0 .$$
	Evaluating at $\beta=0$ and using these,
	$$C(0)=t^2\left\lVertv-w\right\rVert^2_{L^2(\operatorname{vol})},\qquad C'(0)=-2t^2\int_{S^1}v\,w'\,d\operatorname{vol},\qquad C''(0)=2+O(t^2).$$
	The middle identity is the one announced above. Before the cancellations it reads
	$$C'(0)=-2t\int_{S^1}(v-w)\,d\operatorname{vol}-2t^2\int_{S^1}(v-w)\,w'\,d\operatorname{vol},$$
	and it is the vanishing of the first integral that removes the order $t$ and leaves an order $t^2$. In particular $\beta=0$ is not optimal whenever $\int_{S^1}v\,w'\,d\operatorname{vol}\neq0$, and it is exactly this failure that the circle affords and the line does not.

	\emph{Step 3: the gain.} On $|\beta|\leq5tL$ the integrand and its $\beta$-derivatives are bounded. The function $t\,v(\theta)-\beta-t\,w(\theta+\beta)$ is itself $O(t)$ there, and each of its derivatives in $\beta$ beyond the first carries a factor $t$, so that $C'''=O(t)$ there. Since $C''(0)=2+O(t^2)$, the function $C'$ is strictly increasing on that range for $t$ small, hence has a single zero $\beta^*$, and
	$$\beta^*=-\frac{C'(0)}{C''(0)}+O(t^5)=t^2\int_{S^1}v\,w'\,d\operatorname{vol}+O(t^4).$$
	Taylor's formula at $\beta=0$ with the bound on $C'''$ now gives
	$$W_2^2(\mu,\nu)=C(\beta^*)=C(0)-\frac{C'(0)^2}{2C''(0)}+O(t^5)=t^2\left\lVertv-w\right\rVert^2_{L^2(\operatorname{vol})}-t^4\left(\int_{S^1}v\,w'\,d\operatorname{vol}\right)^2+O(t^5),$$
	which is the stated limit.

	\emph{Step 4: the two claims of Step 1.} Expanding the square and using $\big|t\,v(\theta)-t\,w(\theta+\beta)\big|\leq2tL$ gives $C(\beta)\geq\beta^2-4tL|\beta|$, while $C(0)\leq4t^2L^2$. Hence $C(\beta)>C(0)$ as soon as $\beta^2-4tL|\beta|>4t^2L^2$, which holds for $|\beta|\geq5tL$, so every minimizer satisfies $|\beta^*|\leq5tL$. For such a $\beta$ each particle is displaced by $\big|t\,v(\theta)-\beta-t\,w(\theta+\beta)\big|\leq7tL$, which is less than half a turn once $t<\pi/7L$, and in that regime the distance computed on the lift is the geodesic distance of $S^1$.

	\emph{Step 5: the right hand side.} Here $\int_{S^1}v\,w'\,d\operatorname{vol}=\frac{1}{2\pi}\int_0^{2\pi}\phi'\psi''\,d\theta$, while Lemma \ref{lem:complement} with $\rho\equiv1/2\pi$ gives $C_{\operatorname{vol}}=4\pi^2$ and
	$$T_{\phi\psi}=\frac{\int_0^{2\pi}\psi''\phi'\,d\theta}{4\pi^2}\,\rho^{-1}d\theta=\frac{1}{2\pi}\left(\int_0^{2\pi}\psi''\phi'\,d\theta\right)d\theta,$$
	whose squared $L^2(\operatorname{vol})$ norm is $\big(\frac{1}{2\pi}\int_0^{2\pi}\psi''\phi'\,d\theta\big)^2$. This is $\left(\int_{S^1}v\,w'\,d\operatorname{vol}\right)^2$, so the limit equals $3\left\lVertT_{\phi\psi}\right\rVert^2_{\operatorname{vol}}$.
<!-- end proof-20 -->
<!-- remark-1: body_md -->
Proposition \ref{prop:metric-curvature} has a purely metric consequence: no neighbourhood of $\operatorname{vol}$ in $P_2(S^1)$ is a CAT(0) space. For $t$ small the curves $\gamma_\phi(t):=(\mathrm{Id}+t\nabla\phi)_*\operatorname{vol}$ and $\gamma_\psi(t):=(\mathrm{Id}+t\nabla\psi)_*\operatorname{vol}$ are geodesics issuing from $\operatorname{vol}$, with
	$$W_2(\operatorname{vol},\gamma_\phi(t))=t\left\lVert\nabla\phi\right\rVert_{L^2(\operatorname{vol})},\qquad W_2(\operatorname{vol},\gamma_\psi(t))=t\left\lVert\nabla\psi\right\rVert_{L^2(\operatorname{vol})},$$
	and by the proposition $W_2(\gamma_\phi(t),\gamma_\psi(t))/t$ tends to $\left\lVert\nabla\phi-\nabla\psi\right\rVert_{L^2(\operatorname{vol})}$. Hence the comparison angle at $\operatorname{vol}$ of the triangle with vertices $\operatorname{vol}$, $\gamma_\phi(t)$, $\gamma_\psi(t)$ converges, as $t\to0$, to the angle between $\nabla\phi$ and $\nabla\psi$ in $L^2(\operatorname{vol})$, and this limit is the Alexandrov angle between the two geodesics at $\operatorname{vol}$. In a CAT(0) space the Alexandrov angle between two sides of a geodesic triangle is at most the corresponding angle of its comparison triangle \cite[Proposition II.1.7(4)]{BH99}, so the comparison angle would never be smaller than its limit, that is, $W_2(\gamma_\phi(t),\gamma_\psi(t))\geq t\left\lVert\nabla\phi-\nabla\psi\right\rVert_{L^2(\operatorname{vol})}$ for all small $t$. The proposition gives the strict opposite inequality whenever $\int_{S^1}\phi'\psi''\,d\operatorname{vol}\neq0$, for instance for $\phi=c_n$ and $\psi=s_n$.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
The computation above isolates what distinguishes the circle from the line, and the same dichotomy is visible on both sides of the paper.
	
	On the Riemannian side, the argument of Lemma \ref{lem:complement} applied to $\mathbb{R}$ would again produce $\rho^{-1}dx$ as the only candidate for a one-form orthogonal to all exact ones. But for every probability density on $\mathbb{R}$ one has $\left\lVert\rho^{-1}dx\right\rVert^2_{L^2(\mu)}=\int_{\mathbb{R}}\rho^{-1}dx=\infty$, so that candidate does not belong to $\Omega^1_{L^2}(\mathbb{R},\mu)$, the orthogonal complement of $\overline{\operatorname{Im}(d)}$ is trivial, and $T$ vanishes identically. Equivalently, on $\mathbb{R}$ every function is a derivative, whereas on $S^1$ only those of vanishing mean are. Thus $I-\Pi_\mu$ is of rank zero in the first case and of rank one in the second, and this is what $H^1(S^1)\neq0$ contributes.
	
	On the metric side, the same dichotomy appears as the rotation parameter $\beta$ of Proposition \ref{prop:metric-curvature}. On the line the monotone coupling between two absolutely continuous measures is unique, there is nothing to optimize, and the defect vanishes. On the circle the monotone coupling is determined only up to a rotation, and optimizing it produces the fourth order gain.
