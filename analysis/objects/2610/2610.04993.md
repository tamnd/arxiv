<!-- lemma-1: body_md -->
\label{lem:exact-formula}
For every integer \(n\ge1\),
\begin{equation}\label{eq:exact-formula}
  s_1(n)=\sum_{m\le n}a(m)\left\lfloor\frac nm\right\rfloor.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Note that $s_1(1) = a(1) = 0$. By \cite[Eq.~(4)]{BaraskarVukusic2026}, for \(n\geq 2\),
\begin{equation}\label{eq:least-prime-decomposition}
  s_1(n)=1+\sum_{p<n}|M_p(n)|,
\end{equation}
where
\[
  M_p(n):=
  \left\{
    m\in
    \left[
      n-\left\lfloor\frac{n-p}{p+1}\right\rfloor,
      n-1
    \right]:
    p\mid m
    \quad \text{and}\quad 
    P^-(m)\geq p
  \right\}.
\]
Thus, on writing an element of
\(M_p(n)\) as \(pa\), one obtains
\begin{equation}\label{eq:Mp-rough-interval}
  |M_p(n)|
  =\#\{a\in I_p\cap\mathbb Z:P^-(a)\geq p\},
  \qquad
  I_p:=\left(\frac n{p+1},\frac{n-1}{p}\right].
\end{equation}
M\"obius inversion gives
\[
  \1_{P^-(q)\geq p}
  =\sum_{\substack{d\mid q\\P^+(d)<p}}\mu(d).
\]
Thus,
\begin{equation*}\label{eq:local-mobius-count}
  |M_p(n)|=
  \sum_{\substack{d\ge1\\P^+(d)<p}}\mu(d)
  \left(
    \left\lfloor\frac{n-1}{pd}\right\rfloor
    -\left\lfloor\frac{n}{(p+1)d}\right\rfloor
  \right).
\end{equation*}

Before replacing the first floor, note that we may extend the prime sum from \(p<n\) to \(p\le n\).
Now using
\begin{align}\label{eqn: floor diff}
   \1_{v\mid n} 
  =\left\lfloor\frac nv\right\rfloor-\left\lfloor\frac{n-1}{v}\right\rfloor
  \qquad(v\ge1),
\end{align}
with \(v=pd\) and then summing
over \(p\) gives
\begin{align*}
  s_1(n)
  &=
  1+
  \sum_{\substack{p,d\geq1\\ p\ \mathrm{prime}\\P^+(d)<p}}
  \mu(d)
  \left(
    \left\lfloor\frac{n}{pd}\right\rfloor
    -\1_{pd\mid n}
    -\left\lfloor\frac{n}{(p+1)d}\right\rfloor
  \right)\\
  &=
  1+
  \sum_{\substack{p,d\geq1\\ p\ \mathrm{prime}\\P^+(d)<p}}
  \mu(d)
  \left(
    \left\lfloor\frac{n}{pd}\right\rfloor
    -\left\lfloor\frac{n}{(p+1)d}\right\rfloor
  \right)
  -C(n),
\end{align*}
where the correction term is
\[
  C(n):=
  \sum_{\substack{p,d\geq1\\ p\ \mathrm{prime}\\
                   P^+(d)<p\\pd\mid n}}
  \mu(d).
\]
If \(\mu(d)\ne0\), then \(v=pd\) is squarefree and \(p=P^+(v)\); conversely, every squarefree \(v\ge2\) occurs once. Since \(\mu(d)=-\mu(v)\), for $n>1$, \[ C(n)=-\sum_{\substack{v\mid n\\v\ge2}}\mu(v)=1. \]
Thus, we obtain
\begin{equation}\label{eq:post-correction}
  s_1(n)=
  \sum_{\substack{p,d\geq1\\ p\ \mathrm{prime}\\P^+(d)<p}}
  \mu(d)
  \left(
    \left\lfloor\frac{n}{pd}\right\rfloor
    -\left\lfloor\frac{n}{(p+1)d}\right\rfloor
  \right).
\end{equation}

Grouping the terms in \eqref{eq:post-correction} according to the
denominator of the floor function finishes the proof.

<!-- lemma-2: body_md -->
\label{lem:derivative-estimates}
There exists a fixed integer \(J\geq 0\) such that the following
statements hold.

For every integer \(j\geq 0\), every prime \(p\), and every \(w\) with
\(\Re w\geq 1\), one has
\begin{equation}\label{eq:Vp-derivative-bound}
  \bigl|V_p^{(j)}(w)\bigr|
  \ll_j
  (1+\mathcal L_p)^{J+j}.
\end{equation}
Moreover, for every \(\tau,R>0\) and all integers \(k\geq 0\),
uniformly for
\(
  1\leq \Re w\leq 1+\tau,
  \)
  \(|\Im w|\leq R,
\)
one has
\begin{align}
                            
                   
                                     
                                    
  \bigl|T_p^{(k)}(w)\bigr|
  &\ll_{k,\tau,R}
  \frac{(1+\mathcal L_p)^{J+k}}{p^2}.
  \label{eq:Tp-derivative-bound}
\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We first prove \eqref{eq:Vp-derivative-bound}. The prime \(p=2\)
causes no difficulty, so suppose that \(p\geq 3\). Fix \(w\) with
\(\Re w\geq 1\), and consider the circle
\[
  |z-w|=\frac{1}{\mathcal L_p}.
\]
For every \(z\) on this circle and every prime \(q<p\), we have
\[
  |q^{-z}|
  =q^{-\Re z}
  \leq \frac{q^{1/\mathcal L_p}}{q}
  \leq \frac{\mathrm e}{q}.
\]
It follows that
\[
  |V_p(z)|
  \leq
  \prod_{q<p}\bigl(1+|q^{-z}|\bigr)
  \leq
  \exp\left(\mathrm e\sum_{\substack{q<p\\q\text{ prime}}}\frac1q\right).
\]
By Mertens' estimate for the sum of the reciprocals of the primes,
there is a fixed integer \(J\geq 0\) such that
\[
  |V_p(z)|\ll (1+\mathcal L_p)^J.
\]
Cauchy's estimate on the circle of radius \(1/\mathcal L_p\) therefore
gives
\[
  |V_p^{(j)}(w)|
  \leq
  j!\,\mathcal L_p^j
  \max_{|z-w|=1/\mathcal L_p}|V_p(z)|
  \ll_j (1+\mathcal L_p)^{J+j}.
\]

We next estimate the derivatives of \(d_p\). Differentiating $j$ times
gives
\[
  d_p^{(j)}(w)
  =
  (-\log p)^j p^{-w}
  -
  \bigl(-\log(p+1)\bigr)^j(p+1)^{-w}.
\]
Fix \(j\) and \(w\). For $x\geq 2$, define
\[
  f_{j,w}(x):=(-\log x)^j x^{-w}.
\]
Then
\[
  d_p^{(j)}(w)=f_{j,w}(p)-f_{j,w}(p+1).
\]
For complex \(w\), we have the identity
\[
  d_p^{(j)}(w)
  =
  -\int_p^{p+1}f_{j,w}'(x)\,dx.
\]
A direct calculation gives
\[
  f_{j,w}'(x)
  =
  (-1)^j x^{-w-1}
  \left(
    j(\log x)^{j-1}-w(\log x)^j
  \right),
\]
where the first term is omitted when \(j=0\). If
\(1\leq\Re w\leq1+\tau\) and \(|\Im w|\leq R\), then
\(|w|\ll_{\tau,R}1\). Moreover, for \(p\leq x\leq p+1\),
\[
  x^{-\Re w-1}\leq p^{-2},
  \qquad
  1+\log x\ll 1+\mathcal L_p.
\]
Consequently,
\[
  |f_{j,w}'(x)|

\label{lem:one-sided-moments}
Assume either of the bounds in \eqref{eq:one-sided-error-bounds}.  Then,
for every integer \(m\geq0\),
\begin{equation*}\label{eq:absolute-log-moments}
  \int_0^\infty t^m|h(t)|\,\mathrm d t<\infty,
\end{equation*}
and \(\mathcal M_E\) has a \(C^\infty\) extension to \([0,\infty)\)
satisfying
\begin{equation*}\label{eq:Mellin-moment-identity}
  \mathcal M_E^{(m)}(0)
  =(-1)^m\int_0^\infty t^m h(t)\,\mathrm d t.
\end{equation*}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
The trivial estimate \(|E(n)|\ll n\) gives
\[
  |h(t)|=e^{-t}|E(\lfloor e^t\rfloor)|\ll1.
\]
Consequently, for every \(z>0\) and every integer \(m\geq0\), we may
differentiate under the integral in~\eqref{eq:Mellin-error-def}, obtaining
\[
  \mathcal M_E^{(m)}(z)
  =(-1)^m\int_0^\infty t^m h(t)e^{-zt}\,dt.
\]
Indeed, locally uniformly for \(z>0\), the integrand is dominated by
a constant multiple of \(t^m e^{-ct}\) for some \(c>0\).

By the discussion following Lemma~\ref{lem:derivative-estimates}, the function
\[
  F(x):=D(1+x)-\Xi,\qquad x\geq0,
\]
belongs to \(C^\infty([0,\infty))\), and \(F(0)=0\). Since
\(\Phi(x)=xQ(x)\) and \(Q(0)=1\), after restricting to a sufficiently
small neighborhood of \(0\) we have \(Q(x)\neq0\). Moreover,
\begin{equation}\label{eq:F-over-x}
  \frac{F(x)}x=\int_0^1F'(ux)\,du
  \qquad (x>0).
\end{equation}
The right-hand side defines a \(C^\infty\) function also at \(x=0\);
indeed, for every \(r\geq0\),
\[
  \frac{d^r}{dx^r}\int_0^1F'(ux)\,du
  =\int_0^1u^rF^{(r+1)}(ux)\,du.
\]
It follows that
\[
  \widetilde{\mathcal M}_E(x)
  :=\frac1{Q(x)}\int_0^1F'(ux)\,du
\]
is \(C^\infty\) near \(0\). By~\eqref{eq:factorized-Mellin} and~\eqref{eq:F-over-x}, it agrees with \(\mathcal M_E(x)\)
for \(x>0\), and hence gives a \(C^\infty\) extension of \(\mathcal M_E\) to
\(x=0\). In particular, \(\mathcal M_E^{(m)}(z)\) has a finite limit as
\(z\downarrow0\).

Suppose first that the upper bound in~\eqref{eq:one-sided-h-bounds} holds. Put
\[
  h_+(t):=\max\{h(t),0\},
  \qquad
  h_-(t):=\max\{-h(t),0\}.
\]
Then, for all sufficiently large \(t\),
\[
  0\leq h_+(t)
  \leq C_\beta e^{-\beta\sqrt{t\log t}}.
\]
Since the function on the right has finite polynomial moments, we have
\[
  \int_0^\infty t^m h_+(t)\,dt<\infty.
\]
For \(z>0\),
\[
  \int_0^\infty t^m h_-(t)e^{-zt}\,dt
  =
  \int_0^\infty t^m h_+(t)e^{-zt}\,dt
  -(-1)^m\mathcal M_E^{(m)}(z).
\]
The right-hand side has a finite limit as \(z\downarrow0\).
Monotone convergence on the left therefore gives
\[
  \int_0^\infty t^m h_-(t)\,dt<\infty.
\]
Hence
\[
  \int_0^\infty t^m|h(t)|\,dt<\infty.
\]
Dominated convergence now yields
\[
  \mathcal M_E^{(m)}(0)
  =\lim_{z\downarrow0}\mathcal M_E^{(m)}(z)
  =(-1)^m\int_0^\infty t^mh(t)\,dt.
\]

Under the lower bound in~\eqref{eq:one-sided-h-bounds}, the same argument applies with
\(h_+\) and \(h_-\) interchanged.

<!-- lemma-4: body_md -->
\label{lem:integral-representation}
Under either hypothesis in \eqref{eq:one-sided-error-bounds}, for every
\(k\geq1\),
\begin{equation}\label{eq:integral-representation}
  D_k=\int_0^\infty h(t)\mathcal{K}_k(t)\,\mathrm d t.
\end{equation}
Moreover, \(\mathcal{K}_k\) is a polynomial of degree at most \(k-1\).
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
By Lemma~\ref{lem:one-sided-moments}, \(\mathcal M_E\), regarded as a
function of the real variable \(z\geq0\), has a \(C^\infty\) extension to
\(z=0\). By the discussion following Lemma~\ref{lem:derivative-estimates},
the same is true of \(D(1+z)\), and
\[
  \left.\frac{d^k}{dz^k}D(1+z)\right|_{z=0+}=D_k.
\]
Hence both sides of~\eqref{eq:factorized-Mellin} have \(C^\infty\) extensions to \(z=0\).
Since they agree for \(z>0\), their \(k\)-th right derivatives at
\(z=0\) agree. Thus, for \(k\geq1\),
\[
  D_k
  =
  \sum_{j=0}^k
  \binom{k}{j}
  \Phi^{(j)}(0)\mathcal M_E^{(k-j)}(0).
\]
By Lemma~\ref{lem:one-sided-moments},
\[
  \mathcal M_E^{(k-j)}(0)
  =
  (-1)^{k-j}
  \int_0^\infty t^{k-j}h(t)\,dt.
\]
All these integrals are absolutely convergent, so the finite sum may
be moved inside the integral. Therefore
\[
  D_k
  =
  \int_0^\infty h(t)
  \left\{
    \sum_{j=0}^k
    \binom{k}{j}
    \Phi^{(j)}(0)(-t)^{k-j}
  \right\}\,dt
  =
  \int_0^\infty h(t)\mathcal{K}_k(t)\,dt.
\]
Finally,
\[
  \mathcal{K}_k(t)
  =
  \sum_{j=0}^k
  \binom{k}{j}
  \Phi^{(j)}(0)(-t)^{k-j}.
\]
Since \(\Phi(0)=0\), the term with \(j=0\), which would have degree
\(k\), vanishes. Hence \(\mathcal{K}_k\) has degree at most \(k-1\).

\[
  (-1)^{k-1}\mathcal{K}_k(t)
  =kt^{k-1}
   +\sum_{j=2}^k\binom{k}{j}\Phi^{(j)}(0)(-1)^{j-1}t^{k-j}.

\label{lem:saddle-points}
For all sufficiently large real \(x\), the equation
\begin{equation}\label{eq:saddle-equation-xi}
  \mathrm e^\xi=1+\xi-x
\end{equation}
has a unique solution \(\xi(x)\) satisfying
\[
  \bigl|\xi(x)-(\log x+i\pi)\bigr|
  \leq\frac{4\log x}{x}.
\]
Moreover,
\begin{align}
  \xi(x)
  &=\log x+i\pi
    -\frac{1+\log x+i\pi}{x}
    +O\!\left(\frac{(\log x)^2}{x^2}\right),
    \label{eq:xi-expansion}\\
  \xi'(x)
  &=\frac1{x-\xi(x)}.
    \label{eq:xi-derivative}
\end{align}

<!-- proof-10: body_md -->
Write
\[
  \xi=\log x+i\pi+u.
\]
Equation \eqref{eq:saddle-equation-xi} is equivalent to
\[
  u=
  \operatorname{Log}\left(1-\frac{1+\log x+i\pi+u}{x}\right),
\]
where the logarithm is taken near \(1\). If
\(|u|\leq4\log x/x\), then, for all sufficiently large \(x\),
\[
  \left|\frac{1+\log x+i\pi+u}{x}\right|
  \leq\frac{2\log x}{x}.
\]
Using \(|\operatorname{Log}(1-v)|\leq2|v|\) for \(|v|\leq1/2\), the right-hand side
maps the disk \(|u|\leq4\log x/x\) into itself. Its derivative with respect
to \(u\) is
\[
  -\frac1{x-1-\log x-i\pi-u}
  =O\left(\frac1x\right),
\]
so it is a contraction on this disk. This proves existence and
uniqueness. Expanding the logarithm gives
\[
  u
  =-\frac{1+\log x+i\pi}{x}
   +O\!\left(\Bigl(\frac{\log x}{x}\Bigr)^2\right),
\]
which proves \eqref{eq:xi-expansion}.

Differentiating \eqref{eq:saddle-equation-xi} gives
\[
  \mathrm e^{\xi(x)}\xi'(x)=\xi'(x)-1.
\]
Since \(\mathrm e^{\xi(x)}=1+\xi(x)-x\), it follows that
\[
  \bigl(x-\xi(x)\bigr)\xi'(x)=1,
\]
which proves \eqref{eq:xi-derivative}.

\label{lem:local-saddle-taylor}

\end{equation}

\[
  \Psi(z):=zH'(z)=1-z-\mathrm e^{-z} \quad\text{ and }\quad  z(t):=z_k\mathrm e^{it}.
\]
Since \(z_k\) is a saddle,
\[
  \Psi(z_k)=k.
\]
By the chain rule,
\[
  F_k'(t)
  =iz(t)H'(z(t))-ik
  =i\{\Psi(z(t))-k\}.
\]
Thus
\[
  F_k'(0)=0.
\]
Differentiating once more gives
\[
  F_k''(t)=-z(t)\Psi'(z(t)).
\]

\[
  \mathrm e^{-z_k}=\mathrm e^{\xi_k}=1+\xi_k-k,
\]
and hence using
\[
  \Psi'(z)=-1+\mathrm e^{-z},
\]
we get
\begin{align*}

\end{align*}

\[
  \Psi''(z)=-\mathrm e^{-z}.

\]
Using \(F_k(0)=H(z_k)\), \(F_k'(0)=0\), and
\(F_k''(0)=-B_k\), we obtain

\label{lem:common-factor-estimates}
There is an absolute constant \(C>0\) such that, for
\(0<\rho\leq\mathcal L_p/2\),
\begin{align}
  \max_{|z|\leq\rho}|D_p(z)|
  &\leq2\mathrm e^\rho,
    \label{eq:Dp-coarse-bound}\\
  \max_{|z|\leq\rho}|P_p(z)|
  &\leq
  \exp\left\{
    C\left(\frac{\mathrm e^\rho}{\rho}+1+\mathcal L_p\right)
  \right\}.
    \label{eq:Pp-coarse-bound}
\end{align}
Moreover, uniformly for \(|z|\leq\mathcal L_p/4\),
\begin{align}
  \operatorname{Log} P_p(z)
  &=\sum_{q<p}\frac{1-\mathrm e^{-z\log q/\mathcal L_p}}q
    +O\!\left(\frac{|z|}{\mathcal L_p}\right),
    \label{eq:Pp-first-order-decomposition}\\
  \operatorname{Log} D_p(z)+z
  &=\log\left(1+\frac{z}{\mathcal L_p}\right)
    +O\!\left(\frac{(1+|z|)^2}{p}\right).
    \label{eq:Dp-refined-expansion}
\end{align}
The second formula holds for all sufficiently large \(p\).

Since \(p\log(1+1/p)\leq1\), for \(|z|\leq\rho\) we have
\[
  p\delta_p|z|\leq\frac{\rho}{\mathcal L_p}\leq\frac12.
\]
Thus \(|1-\mathrm e^{-u}|\leq|u|\mathrm e^{|u|}\) and
\eqref{eq:D-factor} give \eqref{eq:Dp-coarse-bound}.

For the terms with \(m=1\) in \eqref{eq:log-Pp-expansion}, partial
summation and Chebyshev's estimate give absolute constants \(A,B>0\)
such that
\[
  \sum_{q<p}\frac{\exp(\rho\log q/\mathcal L_p)}q
  \leq A\log(2+\mathcal L_p)+B\frac{\mathrm e^\rho}{\rho}.
\]
For \(m\geq2\), the inequality \(\rho/\mathcal L_p\leq1/2\) gives
\[
  q^{-m}\left|\mathrm e^{-mz\log q/\mathcal L_p}\right|
  \leq q^{-m/2}.
\]
Together with \(\sum_{q<p}q^{-1}\ll1+\mathcal L_p\), these estimates
prove \eqref{eq:Pp-coarse-bound}.

Separating the term \(m=1\) in \eqref{eq:log-Pp-expansion} and using
\(|1-\mathrm e^{-u}|\leq |u|\mathrm e^{|u|}\) once again, we obtain, for
\(|z|\leq\mathcal L_p/4\),

   \left|1-\mathrm e^{-mz\log q/\mathcal L_p}\right|\\
  &\qquad\ll
  \frac{|z|}{\mathcal L_p}
  \sum_q(\log q)\sum_{m\geq2}q^{-3m/4}
  \ll\frac{|z|}{\mathcal L_p},
\end{align*}
which proves \eqref{eq:Pp-first-order-decomposition}.  Finally, observe that
\[
  p\delta_p
  =\frac1{\mathcal L_p}
   +O\!\left(\frac1{p\mathcal L_p}\right).

\label{lem:uniform-Gp}

  \qquad \Delta_p(0)=0,
\]

\end{equation}

<!-- proof-18: body_md -->
For sufficiently large \(p\), the disk in the statement lies in
\(|z|\leq\mathcal L_p/4\), and we may put
\begin{align*}
  \Delta_p(z)
  :=\operatorname{Log} D_p(z)+\operatorname{Log} P_p(z)-H(z)\\
  =\operatorname{Log} D_p(z)+\operatorname{Log} P_p(z)-\operatorname{Ein}(z)+z.
\end{align*}
In view of Lemma~\ref{lem:common-factor-estimates}, it suffices to estimate the term
\[
  \sum_{q<p}\frac{1-\mathrm e^{-z\log q/\mathcal L_p}}q
\]
in \eqref{eq:log-Pp-expansion}.  Define
\[
  \mathcal P(y):=\sum_{\log q\leq y}\frac{\log q}{q},
  \qquad
  \vartheta(x):=\sum_{q\leq x}\log q.
\]
Partial summation gives
\[
  \mathcal P(y)
  =\mathrm e^{-y}\vartheta(\mathrm e^y)
   +\int_2^{\mathrm e^y}\frac{\vartheta(t)}{t^2}\,\mathrm d t.
\]
The quantitative prime number theorem (see \cite[Theorem~6.9, equation~(6.13)]{MontgomeryVaughan2007}) implies that, for some
\(\kappa>0\),
\begin{equation}\label{eq:weighted-PNT}
  \mathcal P(y)=y+C_1+\mathcal Q(y),
  \qquad
  \mathcal Q(y)
  =O\bigl((1+y)\mathrm e^{-\kappa\sqrt y}\bigr).
\end{equation}
Put
\[
  \varphi_z(t):=
  \begin{cases}
    (1-\mathrm e^{-zt})/t,&t\neq0,\\
    z,&t=0.
  \end{cases}
\]
Stieltjes integration gives
\begin{align*}
  \sum_{q<p}\frac{1-\mathrm e^{-z\log q/\mathcal L_p}}q
  &\ =\frac1{\mathcal L_p}
  \int_{0-}^{\mathcal L_p-}
  \varphi_z(y/\mathcal L_p)\,\mathrm d\mathcal P(y)\\
  &=\operatorname{Ein}(z)+\frac{C_1z}{\mathcal L_p}
   +\frac{\varphi_z(1)\mathcal Q(\mathcal L_p-)}{\mathcal L_p}
   -\frac1{\mathcal L_p^2}\int_0^{\mathcal L_p}
    \varphi_z'(y/\mathcal L_p)\mathcal Q(y)\,\mathrm d y.
\end{align*}
Since
\[
  |\varphi_z'(t)|\leq\frac{|z|^2}{2}\mathrm e^{|z|t}
  \qquad(0\leq t\leq1),
\]
\eqref{eq:weighted-PNT} yields, uniformly for
\(|z|\leq C_0\log\mathcal L_p\),
\begin{equation}\label{eq:first-order-Ein-approximation}
  \sum_{q<p}\frac{1-\mathrm e^{-z\log q/\mathcal L_p}}q
  =\operatorname{Ein}(z)
   +O_{C_0}\!\left(\frac{\log\mathcal L_p}{\mathcal L_p}\right).
\end{equation}
Combining \eqref{eq:first-order-Ein-approximation} with
\eqref{eq:Pp-first-order-decomposition} and
\eqref{eq:Dp-refined-expansion} proves
\eqref{eq:Delta-p-bound}.

\label{lem:derivative-approximation}
Uniformly for primes satisfying \(\mathcal L_p\geq {\frac{k}{\log k}}\), \begin{equation*}\label{eq:derivative-approximation} G_p^{(k)}(0)-f^{(k)}(0)=o(2k!|Q_k|). \end{equation*}

large \(k\),
\[
  r_k
  \leq 2\log\frac{k}{\log k}
  \leq 2\log\mathcal L_p.
\]
Hence, uniformly for primes satisfying
\(\mathcal L_p\geq k/\log k\),
\[
  \sup_{|z|=r_k}|\Delta_p(z)|
  \ll
  \frac{\log(k/\log k)}{k/\log k}
  \ll
  \frac{(\log k)^2}{k}
  =o(1).
\]
It follows that, on writing
\[
  \epsilon_k
  :=
  \sup_{\substack{\mathcal L_p\geq k/\log k\\|z|=r_k}}
  \left|\mathrm e^{\Delta_p(z)}-1\right|,
\]
we have
\[
  \epsilon_k\ll \frac{(\log k)^2}{k}.
\]

Since \(G_p(z)=f(z)\mathrm e^{\Delta_p(z)}\), Cauchy's formula gives
\[
  |G_p^{(k)}(0)-f^{(k)}(0)|
  \leq
  \epsilon_k
  \frac{k!}{r_k^k}
  \max_{|z|=r_k}|f(z)|.

  \leq
  \epsilon_k
  \frac{k!}{r_k^k}
  |f(z_k)|.
\]
Using~\eqref{eq:Qk-Ak-def} and~\eqref{eq:Bk-real-size}, 
we obtain
\[
  |G_p^{(k)}(0)-f^{(k)}(0)|
  \ll
  \frac{(\log k)^{5/2}}{\sqrt{k}}\,
  k!|Q_k|
  =
  o(k!|Q_k|).
\]

<!-- lemma-17: body_md -->
\label{lem:weighted-prime-moment}
As \(k\to\infty\),
\begin{equation*}\label{eq:weighted-prime-moment}
  \sum_pT_p(1)\mathcal L_p^k
  =\bigl(\mathrm e^{-\gamma}+o(1)\bigr)\Gamma(k-1).
\end{equation*}
Additionally,
\begin{equation*}\label{eq:small-weight-tail}
  \sum_{\mathcal L_p<\frac{k}{\log k}}T_p(1)\mathcal L_p^k
  =o\bigl(\Gamma(k-1)\bigr).
\end{equation*}
<!-- end lemma-17 -->
<!-- proof-20: body_md -->
Mertens' product theorem gives
\begin{equation}\label{eq:mertens-weight-uniform}
  V(p)\log p=\mathrm e^{-\gamma}+o(1).
\end{equation}
We first evaluate
\[
  S_k:=\sum_p\frac{(\log p)^{k-1}}{p^2}.
\]
Writing \(\vartheta(x)=\sum_{p\leq x}\log p\), we have the exact
Stieltjes representation
\[
  S_k
  =\int_{(\log2)^-}^\infty t^{k-2}\mathrm e^{-2t}
    \,\mathrm d\vartheta(\mathrm e^t).
\]
The quantitative prime number theorem gives
\[
  \vartheta(\mathrm e^t)
  =\mathrm e^t+O\bigl(\mathrm e^t\mathrm e^{-\kappa\sqrt t}\bigr)
\]
for some \(\kappa>0\).  Integration by parts therefore yields
\begin{align*}
  S_k
  =\int_{(\log2)^-}^\infty t^{k-2}\mathrm e^{-t}\,\mathrm d t+O\left(
    \int_{(\log2)^-}^\infty
    t^{k-2}\mathrm e^{-t}\mathrm e^{-\kappa\sqrt t}
    \left(1+\frac{k}{t}\right)\,\mathrm d t
  \right).
\end{align*}
The error is \(o(\Gamma(k-1))\): on
\(k/2\leq t\leq2k\) this follows from the factor
\(\mathrm e^{-\kappa\sqrt{k/2}}\), while the two complementary ranges are
exponentially small relative to the gamma integral.  Thus
\begin{equation}\label{eq:prime-gamma-moment}
  S_k=(1+o(1))\Gamma(k-1).
\end{equation}
The same argument, or simply
\[
  \int_0^{{\frac{k}{\log k}}}t^{k-2}\mathrm e^{-t}\,\mathrm d t
  \leq\frac{1}{k-1}\Bigl(\frac{k}{\log k}\Bigr)^{k-1},
\]
shows that
\begin{equation}\label{eq:prime-gamma-left-tail}
  \sum_{\mathcal L_p<{\frac{k}{\log k}}}
  \frac{\mathcal L_p^{k-1}}{p^2}
  =o(\Gamma(k-1)).
\end{equation}

Now
\[
  T_p(1)\mathcal L_p^k
  =\frac{V(p)\mathcal L_p^k}{p(p+1)}.
\]
The estimate in \eqref{eq:mertens-weight-uniform} and
\eqref{eq:prime-gamma-left-tail} show that the contribution of
\(\mathcal L_p<{\frac{k}{\log k}}\) is \(o(\Gamma(k-1))\).  On the complementary
range, \({\frac{k}{\log k}}\to\infty\), so the estimate in
\eqref{eq:mertens-weight-uniform} and
\(p/(p+1)=1+o(1)\) hold uniformly.  Hence
\[
  \sum_{\mathcal L_p\geq {\frac{k}{\log k}}}T_p(1)\mathcal L_p^k
  =(\mathrm e^{-\gamma}+o(1))
  \sum_{\mathcal L_p\geq {\frac{k}{\log k}}}
  \frac{\mathcal L_p^{k-1}}{p^2}.
\]
Equations \eqref{eq:prime-gamma-moment} and
\eqref{eq:prime-gamma-left-tail} prove both assertions.

\label{lem:squarefree-smooth-bound}
Let \(x\geq z\geq3\), put
\[
  U:=\frac{\log x}{\log z},
\]
and suppose that \(\mathrm e^2\leq U<z\).  Then
\begin{equation}\label{eq:squarefree-smooth-bound}
  \Psi_{\mu^2}(x,z)
  \leq
  x\exp\left\{
    -U\log U
    +O\left(\frac{U}{\log U}+\log\log z\right)
  \right\}.
\end{equation}
The implied constant is absolute.
<!-- end lemma-20 -->
<!-- proof-25: body_md -->
Set
\[
  \eta:=\frac{\log U}{\log z},
  \qquad
  \sigma:=1-\eta.
\]
The hypotheses imply \(0<\sigma<1\).  Rankin's argument gives
\begin{align*}
  \Psi_{\mu^2}(x,z)
  \leq
  x^\sigma
  \sum_{\substack{d\geq1\\P^+(d)<z}}
  \frac{\mu^2(d)}{d^\sigma} =x^\sigma
  \prod_{q<z}\left(1+q^{-\sigma}\right).
\end{align*}
Since \(x=z^U\), we have
\[
  x^\sigma=x\exp\{-U\log U\}.
\]
Moreover,
\[
  \log\prod_{q<z}\left(1+q^{-\sigma}\right)
  \leq \sum_{q<z}q^{-1+\eta}.
\]
The standard Chebyshev bound \(\pi(t)\ll t/\log t\) and partial
summation give
\begin{align*}
  \sum_{q<z}q^{-1+\eta}\ll \frac{z^\eta}{\log z}
       +\int_2^z\frac{t^{-1+\eta}}{\log t}\,\mathrm d t \ll \log\log z+\frac{U}{\log U}.
\end{align*}
For the last estimate, split the integral at \(t=e^{1/\eta}\) and, in the
second part, make the change of variables \(w=\eta \log t\).  Combining these
estimates proves \eqref{eq:squarefree-smooth-bound}.
