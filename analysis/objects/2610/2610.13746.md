<!-- defn-1: body_md -->
Let $x\in \ensuremath{{\mathbb R}}^2$ and $\rho > 0.$ Given an $H^1$ map $u: \ensuremath{{\mathbb R}}^2 \to N,$  define the \emph{(outer) energy scale}                                                                                           
\begin{equation}\label{lambdadef}
\lambda_{\rho, x}(u) = \inf \left\{ \lambda \in [0,\rho] \,\, \vrule \,\, \sup_{\lambda<r<\rho} E( u, \mathbb{D}_r(x)\setminus \mathbb{D}_{r/2}(x) ) < \varepsilon_0 \right\}.
\end{equation}
Here $\varepsilon_0 > 0$ is a constant  which depends only on the geometry of $N$ and which is chosen small enough so that $\varepsilon$ regularity results such as \cite[Theorem 3.4]{songwaldron} are applicable.  
Further define
\begin{equation*}
    \underline{\lambda}_{\rho, x}(u) = \inf_{y \in \mathbb{D}_{\rho / 2}(x)} \lambda_{\rho,y}(u).
\end{equation*}

<!-- thm-1: body_md -->
\label{thm:main} Let $(N, h)$ be a compact real-analytic manifold. Let $u : \ensuremath{{\mathbb R}}^2 \times \left[ 0, T \right) \to N$ be a classical solution of harmonic map flow with uniformly bounded Dirichlet energy.
Fix $x \in \ensuremath{{\mathbb R}}^2$ and suppose that
\begin{equation}\label{main:assumption}
    \limsup_{t \nearrow T} \underline{\lambda}_{\sqrt{T - t}, x}(u(t)) \cdot \sup_{\mathbb{D}_{\sqrt{T - t}}(x)} |du(t)| < \infty.
\end{equation}
Then the body map $u(T)$ is continuous at $x$ and there is no neck between $u(T)$ and any bubble $\omega:\ensuremath{{\mathbb R}}^2\to N$ that forms at $x,$ i.e. we have $u(T,x)=\omega(\infty).$

<!-- prop-1: body_md -->
(Variant of \cite[Theorem $3$]{leonsimon})\label{prop:Loj-Simon}
Let $N$ be any closed analytic manifold and let
$\omega^* : S^2 \to N$ be any harmonic map.
Then there exist constants $\varepsilon_1>0$, $C<\infty$, and $\gamma\in(1,2]$ such that
\begin{equation}\label{est:Loj-Simon}
\vertE(u)-E(\omega^*)\vert
\leq
C \, \VertdE(u)\Vert_{H^{-1}(S^2, g_{S^2})}^\gamma
\end{equation}
holds for all $u\in H^1(S^2,N)$ with
$\Vertu-\omega^*\Vert_{L^\infty(S^2)}
+\Vert\nabla(u-\omega^*)\Vert_{L^2(S^2 , g_{S^2})}
<\varepsilon_1$.
<!-- end prop-1 -->
<!-- prop-2: body_md -->
\label{prop:Loj-L^2}
Let $\omega^* : S^2 \to N$ be a harmonic map into a closed analytic manifold.
Then there exist constants $\varepsilon\in(0,\varepsilon_1]$ and $C<\infty$ 
such that the following holds true for any map 
$u:\ensuremath{{\mathbb R}}^2\to N$ which
is constant outside of $\mathbb{D}_2$.

Suppose that 
there exist $a\in\mathbb{D}_1$ and a factor $\mu\geq 1$ such that
\begin{equation}
\label{ass:Loj-L2}
\Vert\nabla(u-\omega^*\circ \pi_{a,\mu})\Vert_{L^2(\ensuremath{{\mathbb R}}^2)} + \Vertu-\omega^*\circ \pi_{a,\mu}\Vert_{L^\infty(\ensuremath{{\mathbb R}}^2)}
<\varepsilon \text{ for } \pi_{a,\mu}(x):=\pi\bigl(\mu(x-a)\bigr)
\end{equation}
where $\pi:\ensuremath{{\mathbb R}}^2\to S^2$ denotes the inverse stereographic projection. Then
\begin{equation} \label{claim:Loj-L2}
\vertE(u)-E(\omega^*)\vert
\leq
C\,\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}^{\gamma}
\Bigl(1+\vert\log(\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)})\vert\Bigr)^{\gamma/2}.
\end{equation}
Here $\gamma\in (1,2]$ and $\varepsilon_1>0$ are as in Proposition \ref{prop:Loj-Simon}
<!-- end prop-2 -->
<!-- rmk-1: body_md -->
\label{rmk:mu-large}
We note that condition \eqref{ass:Loj-L2} implies that  for $\varepsilon>0$ chosen sufficiently small the factor $\mu$ is guaranteed to be larger than any fixed constant, as desired in our analysis of maps that are close to simple bubble trees. We also note that the claimed estimate is trivially true for maps whose tension is bounded away from zero and that the claim is invariant under translation. In the following, we can hence restrict our attention to maps with
$\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}\leq \delta_1$
which are constant outside of $\mathbb{D}_3$ and satisfy \eqref{ass:Loj-L2} for $a=0$ and $\mu\geq \mu_0$, where $\delta_1>0$ and $\mu_0 >1$ are sufficiently small/large constants that are chosen below and that only depend on $N$ and $\omega^*$.
<!-- end rmk-1 -->
<!-- proof-1: body_md -->
As the $L^\infty$ and $\dot H^1$ norms are conformally invariant the assumption \eqref{ass:Loj-L2} ensures that  
$ \widetilde u:=u\circ \pi_{a,\mu}^{-1}:S^2\to N$
satisfies the assumptions of Proposition~\ref{prop:Loj-Simon}. We hence deduce that 
\begin{equation}\begin{aligned} \label{est:Loj-dual}
\vertE(u)-E(\omega^*)\vert
&=
\vertE(\widetilde u)-E(\omega^*)\vert
\leq
C \VertdE(\widetilde u)\Vert_{H^{-1}(S^2, g_{S^2})}^\gamma
=
C \VertdE(u)\Vert_{*,(a,\mu)}^\gamma,
\end{aligned}\end{equation}
where $\Vert\cdot\Vert_{*,(a,\mu)}$ denotes the dual norm to the weighted
$H^1$ norm
\begin{equation} \label{def:weighted-H1}
\Vertv\Vert_{(a,\mu)}
:=
\Vertv\Vert_{H^1(\ensuremath{{\mathbb R}}^2,g_{a,\mu})} \text{ for }
g_{a,\mu}
=
\pi_{a,\mu}^*g_{S^2}.
\end{equation}

We note that
\begin{equation*}
\Vertv\Vert_{L^2(\mathbb{D}_2)}
\leq C
(\log\mu)^\frac{1}{2} \cdot \Vertv\Vert_{(a,\mu)},
\end{equation*}
compare e.g. (2.27) of \cite{rupflinharmonic} and hence that
\begin{equation*}
C\Vert\tau(u)\Vert_{L^2(\mathbb{D}_2)}
\geq
(\log\mu)^{-1/2}\VertdE(u)\Vert_{*,(a,\mu)}. 
\end{equation*}
The estimate \eqref{est:Loj-dual} hence allows us to deduce that 
\begin{equation} \label{est:Loj-with-logmu}
\vertE(u)-E(\omega^*)\vert
\leq
C(\log\mu)^{\gamma/2}
\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}^\gamma
\end{equation}
holds 
for any $u$ for which the assumptions of  Proposition~\ref{prop:Loj-L^2} are satisfied for $\varepsilon=\varepsilon_1$.

For maps $u$ for which $\log \mu$ is bounded by some fixed multiple of $\vert\log(\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)})\vert$, so in particular whenever
$
\mu
\leq
\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}^{-8}
$, 
we hence immediately get the estimate \eqref{claim:Loj-L2} claimed
in Proposition~\ref{prop:Loj-L^2}.

It thus remains to consider the case where 
$\mu
\geq
\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}^{-8}$. In this case
 we should expect to obtain stronger estimates since the
$L^2$ norm of the tension strongly weighs the
highly concentrated region while the contribution of the bulk should be controlled quadratically in
terms of the tension since constant maps are
non-degenerate critical points of the energy.
Indeed, we can show


\textbf{Claim:}
There exists $\varepsilon\in (0,\varepsilon_1]$ and $C$ so that 
the  estimate 
\begin{equation*}
\vertE(u)-E(\omega^*)\vert
\leq
C\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}^2
\end{equation*} 
holds for any map $u$ which satisfies the assumptions
of Proposition~\ref{prop:Loj-L^2} for some 
\begin{equation}\label{ass:case2}
\mu \geq \Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}^{-8}.
\end{equation}

For the proof of this claim we can use 
standard results on the energy
on neck regions of almost harmonic maps, as found e.g. in Lemma~2.9 of \cite{toppingannals}. These imply that there exist constants $\delta_{1,2}>0$ and $K$ depending only on $N$ so that for maps $v:\mathbb{D}_1\to N$ with $\Vert\tau(u)\Vert_{L^2(\mathbb{D}_1)}\leq\delta_1 $ 
the estimate 
\begin{equation}
E\bigl(v,\mathbb{D}_{e r}\setminus \mathbb{D}_{r}\bigr)
\leq
K r
\label{est:neck-Peter}
\end{equation} 
holds for all $r\in(0,e^{-1})$ for which  
\begin{equation}\label{ass:E-small-neck}
E\bigl(v,\mathbb{D}_1\setminus \mathbb{D}_{r^2}\bigr)<\delta_2. 
\end{equation}



\begin{proof}[Proof of claim]
As discussed above we can assume that $a=0$ and that $\Vert\tau(u)\Vert_{L^2(\ensuremath{{\mathbb R}}^2)}\leq \delta_1$  where $\delta_1=\delta_1(N)>0$ is as above. For $\varepsilon$ sufficiently small we immediately get from 
\eqref{ass:Loj-L2} that the

\label{thm:weightedloj} Let $(N, h)$ be a compact real-analytic manifold and let $E_0, K > 0$ be given. There exists a constant $C$, an exponent $\gamma\in (1,2]$ and a finite set of critical values $\mathcal{E}_0 \subset [ 0, E_0 ]$ of the Dirichlet energy, depending only on $N, K,$ and $E_0,$ so that the following holds.

\begin{equation}\label{weightedloj:assumption}
    \underline{\lambda}_{\tau, 0}(u) \sup_{\mathbb{D}_{\tau}(0)} |du| \leq K.
\end{equation}
Then there exists $E_1 \in \mathcal{E}_0$ such that $u$ obeys a \L ojasiewicz inequality of the form
\begin{equation}\label{weightedloj:loj}
    |\Phi_\tau(u) - E_1 | \leq C \left( 1 + \vert\log \left( \sqrt{\tau} \| { \widehat{{\mathcal T}} }(u) \|_\tau \right)\vert \right)^{\gamma / 2} \left( \sqrt{\tau} \| { \widehat{{\mathcal T}} }(u) \|_\tau \right)^\gamma.
\end{equation}

Rescaling by $\sqrt{\tau},$ we can assume without loss of generality that $\tau = 1.$ The assumption (\ref{weightedloj:assumption}) becomes
\begin{equation}\label{weightedloj:scaledassumption}
    \underline{\lambda}_{1, 0}(u) \sup_{\mathbb{D}_{1}(0)} |du| \leq K.
\end{equation}
Let now $\mathcal{H}_0$ be the set of  harmonic maps $\omega: \ensuremath{{\mathbb R}}^2 \to N$ that satisfy 
\begin{equation}\label{weightedloj:definitionofscrE0}
    \underline{\lambda}_{R, 0}(\omega) \sup_{\mathbb{D}_{R}(0)} |d\omega| \leq K
\end{equation}
for all $R \geq 1.$
We note that this set $\mathcal{H}_0$ is smoothly compact modulo scaling. Since $N$ is real-analytic, the standard \L ojasiewicz-Simon inequality \cite{leonsimon} implies that the corresponding set of critical values
$$\mathcal{E}_0=\lbrace E(\omega): \omega\in \mathcal{H}_0 \rbrace$$
is finite.

Let $u$ be any map as in the statement. First note that the weighted Poincar\'e inequality of \cite[Lemma 2.1]{waldronweighted} gives
\begin{equation}\label{Thatcontrolsr2duetc}
\int r^2 |du|^2 e^{-\frac{r^2}{4}}  + \| {\mathcal T}(u) \|^2_{L^2(\mathbb{D}_2)} \leq C \| { \widehat{{\mathcal T}} }(u) \|_1^2.
\end{equation}
In particular, we have
\begin{equation}\label{energycontrolonannulus}
E(u, \mathbb{D}_2 \setminus \mathbb{D}_1) \leq C\| { \widehat{{\mathcal T}} }(u) \|_1^2.
\end{equation}
Similarly, on any compact subset of $\ensuremath{{\mathbb R}}^2,$ the $L^2$-norm of the tension of $u$ is controlled by the $L^2$-norm of the weighted tension.

We now argue by contradiction: supposing that the result is false, there must exist a sequence of maps $u_i : \ensuremath{{\mathbb R}}^2 \to N$ as in the statement
but with
\begin{equation}\label{weightedloj:contradloj}

\end{equation}
for each $E_1 \in \mathcal{E}_0.$ In particular,
\begin{equation} 
\label{ass:tension-to-zero}\| { \widehat{{\mathcal T}} }(u_i) \|_1 \to 0\end{equation}
as $i \to \infty.$

Let $R \geq 1.$ In view of 
\eqref{Thatcontrolsr2duetc} and \eqref{ass:tension-to-zero} we clearly have
\begin{equation*}

\end{equation*}
as well as
\begin{equation*}
    \lambda_{R,0}(u_i) = \lambda_{1,0}(u_i)
\end{equation*}
for $i$ sufficiently large. Thus (\ref{weightedloj:scaledassumption}) gives
\begin{equation}\label{weightedloj:uiassumption}
    \underline{\lambda}_{R, 0}(u_i) \sup_{\mathbb{D}_{R}(0)} |du_i|  \leq K
\end{equation}
for $i$ sufficiently large.

In view of the tension control, we can pass to a subsequence that converges in the bubble-tree sense on $\ensuremath{{\mathbb R}}^2.$ Due to (\ref{Thatcontrolsr2duetc}), the origin is the only possible energy-concentration point.

\end{equation}

\begin{equation*}

\begin{equation*}

\end{equation*}
We further have
$$\left| E(u_i, \mathbb{D}_1) - \frac12 \int_{\mathbb{D}_1} |du_i|^2 e^{\frac{-r^2}{4}} \right| \leq \int_{\mathbb{D}_1} r^2 |du_i|^2 \lesssim \| { \widehat{{\mathcal T}} }(u_i) \|_1^{2}.$$
Finally, we have
$$\int_{\ensuremath{{\mathbb R}}^2 \setminus \mathbb{D}_1} |du_i|^2 e^{\frac{-r^2}{4}} \lesssim \int_{\ensuremath{{\mathbb R}}^2 \setminus \mathbb{D}_1} r^2 |du|^2 e^{\frac{-r^2}{4}} \lesssim \| { \widehat{{\mathcal T}} }(u_i) \|_1^{2}.$$
Combining the last three estimates, we obtain (\ref{weightedloj:loj}) for $u = u_i$ and $E_1 = E(\omega^*).$ But for $i$ sufficiently large, this contradicts (\ref{weightedloj:contradloj}).

We can take $x = 0$ without loss of generality. The assumption (\ref{main:assumption}) implies (\ref{weightedloj:assumption}), with $\tau = \sqrt{T - t},$ for times $t$ sufficiently close to $T.$ By the previous theorem, for each such $t,$ the \L ojasiewicz inequality (\ref{weightedloj:loj}) is satisfied for $u = u(t).$ Since the quantity $\Phi_{\sqrt{T - t}}(u(t))$ is monotone-decreasing (see \cite{struwehmsurfaces} or \cite[Theorem 3.2]{waldronweighted}) and the set $\mathcal{E}_0$ is finite, by further advancing $t,$ we can assume that (\ref{weightedloj:loj}) is satisfied for the same value $E_1 \in \mathcal{E}_0$ at all times. The hypothesis of \cite[Corollary 4.3]{waldronweighted} is therefore satisfied, giving the desired conclusions.
