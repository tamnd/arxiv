Let $\mathbb{T}$ denote the unit circle in the complex plane, and let $q$ be a real number lying in the interval $(0,1)$. Let  $\mathbb{N}_{0}=\mathbb{N}\cup \{0\}$. The standard bases of the Hilbert spaces $\ell^2(\mathbb{N}_0)$ and
$\ell^2(\mathbb{Z})$ will be denoted by $\{e_m : m \in \mathbb{N}_0\}$ and $\{e_m : m \in \mathbb{Z}\}$ respectively. The length of a Weyl word $w$ is denoted by $\ell(w)$. The number operator $e_m\mapsto me_m$ is denoted by $N$. The operator $S$ denotes the left shift  $e_m\mapsto e_{m-1}$. The $C^{\ast}$-subalgebra of $\mathcal{L}(\ell^2(\mathbb{N}_0))$ generated by $S$ is denoted by $\mathscr{T}$. 
Let $\sigma: \mathscr{T}\longrightarrow\mathbb{C} $
be the homomorphism for which $\sigma(S)=1$. For a Hilbert space $H$, the algebra of compact operators on $H$ will be denoted by $\mathcal{K}(H)$, or simply $\mathcal{K}$ when the Hilbert space is clear from the context. For $i,j\in \mathbb{N}_0$, the operator $p_{ij}$ denotes the operator on $\ell^2(\mathbb{N}_0)$ which sends the basis element $e_i$ to $e_j$.

\rm \label{Indexing of the irreducible representations}
For each $t\in\mathbb{T}$ and $0\leq m\leq 2n-1$, and via the natural embedding
$\mathscr{T}^{\otimes m} \subseteq \mathcal{L}\left(\ell^2\left(\mathbb{N}_0\right)^{\otimes m}\right)$, the irreducible representation
$\sigma_{t,m}^{(q)}$ coincides with the representation appearing in the classification of irreducible representations of $C(SO_q(2n+1)/SO_q(2n-1))$ in \cite{BhuBisSau-2026aa}, namely, $\sigma_{t,m}^{(q)}
= \phi_{t,\omega_{m+1}}$.
Using the diagram representations for $\phi_{t,\omega_{m+1}}$ (see \cite{BhuSau-2023aa}), we obtain:
\begin{itemize}
\item If $0\leq m\leq n-1$, then
$$\sigma_{t,m}^{(q)}(x_r)=0

    \item First, consider the case where $i \in \{1, \dots, 2n+1\} \setminus \{n+1\}$. Let $h \in \ker\pi\left(z_i\right)$. Applying the representation $\pi$ to relation~\eqref{r3.0} and evaluating at $h$ gives
$$
\left(\pi\left(z_i z_i^*\right) + \sum_{k>i} \pi\left(z_k z_k^*\right)\right)h = \pi\left(z_i^*z_i\right)h = 0.
$$
Taking the inner product of the left-hand side with $h$ yields a sum of squared norms:
$$
\left\|\pi\left(z_i^*\right)h\right\|^2 + \sum_{k>i} \left\|\pi\left(z_k^*\right)h\right\|^2 = 0.
$$
Thus $\pi\left(z_k^*\right)h=0$ for every $k\geq i$,
proving the first inclusion.

For the case $i=n+1$, the inclusion follows by a similar argument using relation \eqref{r4.0}.

   \item For the case $i=1$, relation~\eqref{r3.0} states that $z_1^*z_1 = \sum_{k=1}^{2n+1} z_k z_k^*$. By relation~\eqref{r5.0}, this sum is equal to $1$. Applying the representation $\pi$ yields $\pi(z_1)^*\pi(z_1) = I$, which proves that $\pi(z_1)$ is an isometry.

Next, consider $i \in \{2, \dots, 2n+1\}$ with $i \neq n+1$. Let $W_{i-1} := \bigcap_{k=1}^{i-1} \ker\pi\left(z_k^*\right)$ and let $h \in W_{i-1}$. Since $\pi(z_k^*)h = 0$ for all $k < i$, we have $\pi\left(\sum_{k=1}^{i-1} z_k z_k^*\right)h = 0$. We can then compute:
\begin{align*}
    \pi\left(z_i\right)^*\pi\left(z_i\right)h
    &= \pi\left(\sum_{k=i}^{2n+1} z_k z_k^*\right)h && \left(\text{by relation \eqref{r3.0}}\right) \\
    &= \left( \pi\left(\sum_{k=1}^{i-1} z_k z_k^*\right) + \pi\left(\sum_{k=i}^{2n+1} z_k z_k^*\right) \right) h && \left(\text{since } \pi\left(\sum_{k=1}^{i-1} z_k z_k^*\right)h = 0\right) \\
    &= \pi\left(\sum_{k=1}^{2n+1} z_k z_k^*\right)h = h && \left(\text{by relation \eqref{r5.0}}\right).
\end{align*}
This shows that the restriction $\pi\left(z_i\right)|_{W_{i-1}}$ is an isometry.

Finally, the proof for the case $i=n+1$ is analogous, using relation \eqref{r4.0} in place of relation \eqref{r3.0}.

    \item Since $\omega$ is a projection, $\sigma\left(\pi(\omega)\right) \subseteq \{0, 1\}$. Assume for contradiction that $\sigma\left(\pi(\omega)\right) = \{1\}$. This would imply $\pi(\omega) = I$. From Lemma \ref{Properties of omega for odd}, the relation $\omega z_i = 0$ gives $\pi\left(z_i\right) = \pi(\omega)\pi\left(z_i\right) = 0$ for all $i < 2n+1$. This yields a contradiction via relation \eqref{r4.0}:
    $$
    I = \pi(\omega) = \pi\left(z_{n+1}^* z_{n+1}\right) = \pi\left(z_{n+1}\right)^*\pi\left(z_{n+1}\right) = 0.
    $$
    The assumption that $\sigma\left(\pi(\omega)\right) = \{1\}$ is therefore false, which establishes the claim.
\end{enumerate}

\label{ON basis for odd}
Let $\pi$ be an irreducible representation of $\mathcal{B}_{2n+1}$ on a Hilbert space $H$ such that $\pi\left(z_{2n+1}\right)\neq 0$. \begin{cases}
\pi\left(z_{i+1}\right)^{\alpha_{i+1}}\cdots\pi\left(z_{2n}\right)^{\alpha_{2n}}u,& 1\leq i\leq 2n-1,\\

\end{cases}

\left\langle u_{\alpha}^{(2)},u_{\beta}^{(2)}\right\rangle$.

\rm
A subset $A \subset \mathbb{T} \times \{1, 2, \dots, k\}$ is said to be \emph{closed} under the topology $\mathcal{O}$ if:
\begin{enumerate}
	\item[\rm (1)] $A_{\text{ind}_A}$ is a closed set in $\mathbb{T}$, and
	\item[\rm (2)] $A_r = \mathbb{T}$ for all $r > \text{ind}_A$.
\end{enumerate}

<!-- thm-6: body_md -->
\label{K-groups-B}
For $1 \leq k \leq 2n+1$ with $k \neq n+1$, define $u_k=  z_k+1-z_kz_k^*$. 
Then one has 
\begin{IEEEeqnarray*}{rCl}
	K_0(\mathcal{B}^k_{2n+1})&=&\begin{cases}
		\langle[1]\rangle \cong \mathbb{Z} & \mbox{ if } 1 \leq  k \leq n, \cr 
		\langle[1]\rangle \oplus \langle[z_{n+1}^*z_{n+1}]\rangle  \cong \mathbb{Z} \oplus \mathbb{Z}/2\mathbb{Z} & \mbox{ if } n+2 \leq k \leq 2n+1, \cr 
	\end{cases} \\
	K_1(\mathcal{B}^k_{2n+1})&=&\langle [u_k] \rangle\cong \mathbb{Z}.
\end{IEEEeqnarray*}

<!-- thm-7: body_md -->
\label{mQDS}
For $1 \leq k \leq 2n+1$, we have
\[
\mathcal{B}^k_{2n+1} \;\cong\; 
\begin{cases}
	\Sigma^{2(k-1)} C(\mathbb{T}), & \text{if } 1 \leq k \leq n, \cr
	\Sigma^{2(n-1)} C(\mathbb{T}), & \text{if } k= n+1, \cr
	\Sigma^{2(k-n-1)} \,\Sigma_2^2 \,\Sigma^{2(n-1)} C(\mathbb{T}), & \text{if } n+2 \leq k \leq 2n. \cr 
\end{cases}
\]
In particular, 
\[
C\bigl(SO_0(2n+1)/SO_0(2n-1)\bigr) \;\cong\; \Sigma^{2(n-1)} \,\Sigma_2^2 \,\Sigma^{2(n-1)} C(\mathbb{T}).
\]
