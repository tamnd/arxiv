\label{def:localcone}
        A generator is a $k$-local fermionic Lindbladian, written $\mathcal L\in\mathfrak L_k$, if it has a decomposition
        \begin{equation}
            \mathcal L=\sum_{e:|e|\le k}\mathcal L_e
        \end{equation}
        with
        \begin{align}
            \mathcal L_e(\rho)
            =
            -i[H_e,\rho]
            +
            \sum_{p=0}^1
            \sum_{A,B\in\mathcal I_{e,p}}
            K^{(e,p)}_{A,B}
            \left(
                \Gamma_A\rho\Gamma_B
                -
                \frac12\{\Gamma_B\Gamma_A,\rho\}
            \right),\label{eq:local-gksl}
        \end{align}
        where $H_e$ is even, Hermitian, traceless, and supported on $e$, and each $K^{(e,p)}$ is PSD.

<!-- lemma-1: body_md -->
\label{lem:local-coefficient-norm}
        Let $1\le j\le k$ and write a parity-covariant $j$-mode generator in the local form \eqref{eq:local-gksl}, with Hamiltonian coefficients $h_A$, Kossakowski blocks $K^{(p)}$, and Majorana-superoperator coefficients $\chi_{A,B}$. Then
        \begin{equation*}
            \sum_A|h_A|
            +
            \sum_{p=0}^1\sum_{A,B}|K^{(p)}_{A,B}|
            +
            \sum_{A,B}|\chi_{A,B}|
            \le
            c_k\|\mathcal L^\dagger\|_{2\to2},
            \qquad c_k=2\cdot4^k.
        \end{equation*}
        The norm is taken on the full local operator algebra, including its even and odd subspaces, and $c_k$ is independent of the ambient system size.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Set $d_j=2^j$. For a single local term, \eqref{eq:G-from-chi}, \eqref{eq:h-from-chi}, and the triangle inequality give
        \begin{equation*}
            \sum_A|h_A|+\sum_{p=0}^1\sum_{A,B}|K^{(p)}_{A,B}|
            \le \sum_{A,B}|\chi_{A,B}|.
        \end{equation*}
        Let $M$ be the $d_j^2\times d_j^2$ matrix of $\mathcal L$ under column vectorization. Since vectorization is an isometry for the Hilbert--Schmidt norm, $\|M\|_{\rm op}=\|\mathcal L\|_{2\to2}$. The basis matrices $\Gamma_B^{\mathsf T}\otimes\Gamma_A$ are orthogonal and have Frobenius norm $d_j$, so
        \begin{equation*}
            \|M\|_{\rm F}^2=d_j^2\sum_{A,B}|\chi_{A,B}|^2.
        \end{equation*}
        If $s_\ell$ are the singular values of $M$, then
        \begin{equation*}
            \|M\|_{\rm F}^2
            =\sum_\ell s_\ell^2
            \le d_j^2\max_\ell s_\ell^2
            =d_j^2\|\mathcal L\|_{2\to2}^2.
        \end{equation*}
        Hence $\|M\|_{\rm F}\le d_j\|\mathcal L\|_{2\to2}$, and Cauchy--Schwarz over the $d_j^4$ coefficient pairs gives
        \begin{equation*}
            \sum_{A,B}|\chi_{A,B}|
            \le d_j\|M\|_{\rm F}
            \le d_j^2\|\mathcal L\|_{2\to2}
            =d_j^2\|\mathcal L^\dagger\|_{2\to2}.
        \end{equation*}
        The last equality uses invariance of the induced Hilbert--Schmidt norm under adjoints. The full coefficient sum is therefore at most $2d_j^2\|\mathcal L^\dagger\|_{2\to2}$, and $d_j^2=4^j\le4^k$ concludes the proof.

<!-- lemma-2: body_md -->
\label{lem:graded-diamond-amplification}
        Let $\mathcal T$ be a parity-covariant linear map on the system CAR algebra and let $\mathcal T_{\rm ext}$ be its fermionic extension by finitely many idle fermionic modes.  Then
        \begin{equation*}
            \|\mathcal T_{\rm ext}\|_{2\to2}
            =
            \|\mathcal T\|_{2\to2},
            \qquad
            \|\mathcal T_{\rm ext}(X)\|_1
            \le
            \|\mathcal T\|_\diamond\|X\|_1
        \end{equation*}
        for every $X$ on the enlarged algebra.  In particular, for parity-covariant Lindbladians $\mathcal L$ and $\mathcal K$,
        \begin{equation*}
            \|e^{t\mathcal L_{\rm ext}}-e^{t\mathcal K_{\rm ext}}\|_{1\to1}
            \le
            t\|\mathcal L-\mathcal K\|_\diamond ,
            \qquad t\ge0 .
        \end{equation*}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Order the system modes before the idle modes. Under the resulting identification of their Fock spaces, system Majorana monomials act as $\Gamma_A\otimes\mathds 1$, while the fermionic grading is carried by parity strings on odd ancillary operators. Thus the fermionic extension is represented on the full matrix algebra by the ordinary amplification $\mathcal T_{\rm ext}=\mathcal T\otimes\operatorname{id}$. Consequently,
        \begin{equation*}
            \|\mathcal T_{\rm ext}\|_{2\to2}=\|\mathcal T\|_{2\to2},
            \qquad
            \|\mathcal T_{\rm ext}\|_{1\to1}\le\|\mathcal T\|_\diamond.
        \end{equation*}
        These bounds are independent of the chosen fermionic ordering because CAR-mode permutations are unitarily implemented. Finally, Duhamel's formula gives
        \begin{equation*}
            e^{t\mathcal L_{\rm ext}}-e^{t\mathcal K_{\rm ext}}
            =
            \int_0^t
            e^{(t-s)\mathcal L_{\rm ext}}
            (\mathcal L-\mathcal K)_{\rm ext}
            e^{s\mathcal K_{\rm ext}}\,ds .
        \end{equation*}
        The two amplified semigroups are trace-norm contractions, so integration yields the stated estimate.

<!-- lemma-3: body_md -->
\label{lem:maj-completeness}
        For all matrix indices $p,q,r,s$,
        \begin{equation}\label{eq:majorana-completeness}
            \sum_{A\subseteq[2m]}
            (\Gamma_A)_{pq}(\Gamma_A)_{rs}
            =
            d\,\delta_{ps}\delta_{qr}.
        \end{equation}
        Consequently, for all matrices $X,Y,Z,W$ on the $d$-dimensional Fock space,
        \begin{equation}\label{eq:majorana-fierz}
            \sum_{C,D\subseteq[2m]}
            \operatorname{tr}[\Gamma_C X\Gamma_D Y]
            \operatorname{tr}[\Gamma_C Z\Gamma_D W]
            =
            d^2\operatorname{tr}[XW]\operatorname{tr}[YZ].
        \end{equation}

<!-- lemma-4: body_md -->
\label{lem:signed-transfer}
        For a parity-covariant map and every $C,D\subseteq[2m]$,
        \begin{equation}
            \Lambda_{C,D}
            =
            d^{-1}
            \sum_{\substack{A,B\\ \pi(A)=\pi(B)}}
            (-1)^{\pi(D)\pi(B)}
            \chi_{A,B}
            \operatorname{tr}[\Gamma_C\Gamma_A\Gamma_D\Gamma_B].
            \label{eq:signed-transfer}
        \end{equation}
        Both sides vanish whenever $\pi(C)\neq\pi(D)$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
If $\pi(C)\neq\pi(D)$, the ancillary factor contains one power of $\eta$, so $\Lambda_{C,D}=0$. The trace on the right also vanishes termwise: since $\pi(A)=\pi(B)$, the product $\Gamma_C\Gamma_A\Gamma_D\Gamma_B$ is odd. Now suppose $\pi(C)=\pi(D)=q$. In one summand,
        \begin{equation*}
            J_\eta(\Gamma_C)\Gamma_AJ_\eta(\Gamma_D)\Gamma_B
            =
            i^{2q}\Gamma_C\eta^q\Gamma_A\Gamma_D\eta^q\Gamma_B.
        \end{equation*}
        Moving the first $\eta^q$ through $\Gamma_A\Gamma_D$ contributes $(-1)^{q(\pi(A)+q)}$.  Since $i^{2q}=(-1)^q$ and $q^2=q$ modulo two, the total phase is $(-1)^{q\pi(A)}=(-1)^{\pi(D)\pi(B)}$.  The ancillary trace then contributes a factor of two, leaving the system trace in \eqref{eq:signed-transfer}.

<!-- lemma-5: body_md -->
\label{lem:signed-dressed-inversion}
        For every $A,B\subseteq[2m]$,
        \begin{equation}\label{eq:signed-dressed-inversion}
            \chi_{A,B}
            =
            d^{-3}
            \sum_{C,D\subseteq[2m]}
            (-1)^{\pi(D)\pi(B)}
            \Lambda_{C,D}\,
            \operatorname{tr}[\Gamma_C\Gamma_B\Gamma_D\Gamma_A].
        \end{equation}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
If $\pi(A)\neq\pi(B)$, then $\chi_{A,B}=0$ by parity covariance. Moreover, $\Lambda_{C,D}$ vanishes unless $\pi(C)=\pi(D)$, in which case $\Gamma_C\Gamma_B\Gamma_D\Gamma_A$ is odd and its trace vanishes. Thus both sides are zero. Assume henceforth that $\pi(A)=\pi(B)$.

        Insert \eqref{eq:signed-transfer} into the right-hand side of \eqref{eq:signed-dressed-inversion}, and denote the summation indices in \eqref{eq:signed-transfer} by $A',B'$.  The sign multiplying the two traces is
        \begin{equation*}
            (-1)^{\pi(D)\pi(B)}
            (-1)^{\pi(D)\pi(B')}
            =
            (-1)^{\pi(D)(\pi(B)+\pi(B'))}.
        \end{equation*}
        If $\pi(B')=\pi(B)$, this sign is one.  Applying \eqref{eq:majorana-fierz} to
        \begin{equation*}
            \operatorname{tr}[\Gamma_C\Gamma_{A'}\Gamma_D\Gamma_{B'}]
            \operatorname{tr}[\Gamma_C\Gamma_B\Gamma_D\Gamma_A]
        \end{equation*}
        gives $d^2\operatorname{tr}[\Gamma_{A'}\Gamma_A]\operatorname{tr}[\Gamma_{B'}\Gamma_B]$, hence only $A'=A$ and $B'=B$ survive after the prefactors.

        It remains to exclude the block $\pi(B')\neq\pi(B)$.  In that case the leftover factor is $(-1)^{\pi(D)}$, which can be written as $\Pi\Gamma_D\Pi$ inside the $D$-sum.  Applying \eqref{eq:majorana-fierz} then yields a product of traces
        \begin{equation*}
            d^2\operatorname{tr}[\Gamma_{A'}\Pi\Gamma_A]\,
            \operatorname{tr}[\Pi\Gamma_{B'}\Gamma_B],
        \end{equation*}
        but the second trace vanishes because $\Pi$ is even and $\pi(B')\neq\pi(B)$. Thus only the desired block survives.

<!-- theorem-1: body_md -->
\label{thm:diagonal-extension}
        For every $A,B\subseteq\mathcal M(R)$ with $\pi(A)=\pi(B)$,
        \begin{equation}
            \chi^{[R]}_{A,B}
            =
            \sum_{E\subseteq\mathcal M(R^c)}
            s_R(A,B;E)\,
            \chi_{A\sqcup E,\,B\sqcup E}.
            \label{eq:diagonal-extension}
        \end{equation}
        Under $k$-locality, only terms with
        $|\operatorname{supp}_{\rm f}(A,B)\cup\operatorname{supp}_{\rm f}(E)|\le k$ can be nonzero.
<!-- end theorem-1 -->
<!-- proof-6: body_md -->
Write arbitrary global labels as $A'=A_0\sqcup E$ and $B'=B_0\sqcup F$, with $A_0,B_0\subseteq\mathcal M(R)$ and $E,F\subseteq\mathcal M(R^c)$. Since $C,D$ are supported on $R$, the trace over $R^c$ contains $\operatorname{tr}_{R^c}[\Gamma_E\Gamma_F]$, which vanishes unless $E=F$ by Majorana-basis orthogonality. Using \eqref{eq:union-product}, commuting the complement monomial through the local factors, and tracing over $R^c$ gives
        \begin{align}
            d^{-1}\operatorname{tr}[\Gamma_C\Gamma_{A'}\Gamma_D\Gamma_{B'}]
            =
            \delta_{E,F}\,d_R^{-1}
            \omega(A_0,E)\omega(B_0,E)
            (-1)^{\pi(E)(\pi(D)+\pi(B_0))}
            \operatorname{tr}_R[\Gamma_C\Gamma_{A_0}\Gamma_D\Gamma_{B_0}].
            \label{eq:trace-factor}
        \end{align}
        The signed transfer relation contributes
        \begin{equation*}
            (-1)^{\pi(D)\pi(B')}
            =
            (-1)^{\pi(D)(\pi(B_0)+\pi(E))}.
        \end{equation*}
        Multiplying this with the sign in \eqref{eq:trace-factor} cancels the two copies of $\pi(D)\pi(E)$ and leaves
        \begin{equation*}
            \omega(A_0,E)\omega(B_0,E)
            (-1)^{\pi(E)\pi(B_0)}
            (-1)^{\pi(D)\pi(B_0)}.
        \end{equation*}
        The definition of $\chi^{[R]}_{A,B}$ supplies one more inversion sign,
        $(-1)^{\pi(D)\pi(B)}$.  Hence the $C,D$-dependent part is the local signed Fierz kernel with sign
        \begin{equation*}
            (-1)^{\pi(D)(\pi(B)+\pi(B_0))},
        \end{equation*}
        while the $C,D$-independent coefficient is
        \begin{equation*}
            \omega(A_0,E)\omega(B_0,E)(-1)^{\pi(E)\pi(B_0)}.
        \end{equation*}
        Applying Lemma~\ref{lem:signed-dressed-inversion} in the abstract $|R|$-mode Clifford algebra therefore leaves only $A_0=A$, $B_0=B$.  The remaining coefficient is exactly $s_R(A,B;E)$, proving \eqref{eq:diagonal-extension}.

        \begin{equation*}

        \end{equation*}

<!-- lemma-7: body_md -->
\label{lem:fermionic-stability}
        Suppose every restricted coefficient $\chi^{[R]}_{A,B}$ used by \eqref{eq:descending-recursion} is known to additive error at most $\varepsilon_L$.  For fixed $k$, the recovered coefficients satisfy
        \begin{equation}
            \max_{\substack{\pi(A)=\pi(B)\\ |\operatorname{supp}_{\rm f}(A,B)|\le k}}
            \left\lvert \widehat\chi_{A,B}-\chi_{A,B}\right\rvert
            \le
            C_k m^k\varepsilon_L
        \end{equation}
        for a constant $C_k$ depending only on $k$.  More precisely, on the punctured set of nonidentity coefficients,
        \begin{equation}
            \max_{\substack{\pi(A)=\pi(B)\\
            1\le |\operatorname{supp}_{\rm f}(A,B)|\le k}}
            \left\lvert \widehat\chi_{A,B}-\chi_{A,B}\right\rvert
            \le
            C_k m^{k-1}\varepsilon_L .

            \le

<!-- remark-1: body_md -->
\label{rem:number-conserving-identifiability}
        Some parity-preserving but number-nonconserving access is necessary for full identification. Our protocol provides it through pairing rotations.  Indeed, let $\mathcal L_0$ be $U(1)$ covariant, meaning that it is invariant under the global phase rotations generated by $\mathsf N$.  Equivalently, $\mathcal L_0$ commutes with $-i[\mathsf N,\cdot]$, and for states, effects, and controls commuting with $\mathsf N$,
        \begin{equation*}
            \operatorname{tr}\!\left[
                Oe^{t(\mathcal L_0-i\theta[\mathsf N,\cdot])}(\rho)
            \right]
            =
            \operatorname{tr}\!\left[Oe^{t\mathcal L_0}(\rho)\right].
        \end{equation*}
        Thus strictly number-conserving experiments cannot distinguish generators that differ by the chemical-potential term $-i\theta[\mathsf N,\cdot]$.  Pairing rotations remove this obstruction by coupling particle-number sectors that differ by two while preserving parity.  This does not imply that fixing this ambiguity suffices to identify every $U(1)$-covariant generator.

<!-- remark-2: body_md -->

Hence no arbitrary many-body state preparation or non-Gaussian control is needed, and for fixed locality the operation involves only a constant number of modes.  Our preparation scheme uses local control, including marker access, Gaussian mode mixing, and pairing rotations.

<!-- proof-10: body_md -->
Fix a query $\tau=(P,Q)$. For every accepted pairing $s$, Proposition~\ref{prop:pair-block-process-shadow} gives conditional mean $2^{-|W|}\operatorname{tr}[P\Phi(Q)]$ and conditional second moment at most $7^{w_*}$ for $Z_\tau^s$. Averaging over the pairing, with $X_\tau=0$ on rejection, therefore gives
        \begin{align*}
            \mathbb E X_\tau
            &=p_\tau^{-1}\Pr\{s\in\mathsf A_\tau\}\,2^{-|W|}\operatorname{tr}[P\Phi(Q)]
            =2^{-|W|}\operatorname{tr}[P\Phi(Q)],\\
            \mathbb E X_\tau^2
            &\le p_\tau^{-2}\Pr\{s\in\mathsf A_\tau\}\,7^{w_*}
            =p_\tau^{-1}7^{w_*}.
        \end{align*}
        Thus acceptance with probability $p_\tau$ cancels one factor of $p_\tau^{-1}$ in the second-moment bound.

        We next estimate all overlaps from repeated independent samples, redrawing the pairings each time. The variance of every $X_\tau$ is at most $p_*^{-1}7^{w_*}$. For $0<\varepsilon_M<1$, divide the data into $\mathcal O(\log(2|\mathcal T|/\delta))$ groups, each containing $\mathcal O(p_*^{-1}7^{w_*}\varepsilon_M^{-2})$ samples, with sufficiently large constants. The median-of-means bound \cite[Theorem~4.1]{DevroyeEtAl2016} gives error at most $\varepsilon_M$ with failure probability at most $\delta/|\mathcal T|$. A union bound gives simultaneous success with probability at least $1-\delta$ and the sample count \eqref{eq:random-pairing-cost}. The same groups serve all queries, since the union bound does not require independence between their estimates. For $\varepsilon_M\ge1$, zero estimates suffice because every target overlap has absolute value at most one.

<!-- proof-19: body_md -->
We first prove the inverse formula \eqref{eq:structure-signed-inverse} by applying inclusion--exclusion to the diagonal-extension identity \eqref{eq:diagonal-extension} of Theorem~\ref{thm:diagonal-extension}. This requires a composition rule for the extension signs.

        For extensions $E,F$ on disjoint additional mode sets, associativity and the product rule \eqref{eq:union-product} give $\omega(A,E)\omega(A\sqcup E,F)=\omega(A,E\sqcup F)\omega(E,F)$. Applying this to $A$ and $B$ produces the factor $\omega(E,F)^2=(-1)^{\pi(E)\pi(F)}$ by \eqref{eq:omega}. In the extension-sign definition \eqref{eq:extension-phase}, this cancels the extra factor $(-1)^{\pi(F)\pi(E)}$ arising from $\pi(B\sqcup E)=\pi(B)+\pi(E)$ modulo two. Hence, with $R'=R\cup\operatorname{supp}_{\rm f}(E)$,
        \begin{equation*}
            s_R(A,B;E)s_{R'}(A\sqcup E,B\sqcup E;F)
            =s_R(A,B;E\sqcup F).
        \end{equation*}
        We now apply \eqref{eq:diagonal-extension} on $R'$ to obtain
        \begin{equation*}
            y_{A\sqcup E,B\sqcup E}
            =\sum_{\substack{F\subseteq\mathcal M((R')^c)\\ |R'\cup\operatorname{supp}_{\rm f}(F)|\le k}}
            s_{R'}(A\sqcup E,B\sqcup E;F)\,
            \chi_{A\sqcup E\sqcup F,B\sqcup E\sqcup F}.
        \end{equation*}
        Substituting into the right-hand side of \eqref{eq:structure-signed-inverse}, using the sign identity above, and grouping by the final extension $H=E\sqcup F$ yields
        \begin{align*}
            &\sum_{\substack{E\subseteq\mathcal M(R^c),\ F\subseteq\mathcal M((R')^c)\\ |R'\cup\operatorname{supp}_{\rm f}(F)|\le k}}
            (-1)^{|\operatorname{supp}_{\rm f}(E)|}
            s_R(A,B;E)s_{R'}(A\sqcup E,B\sqcup E;F)\,
            \chi_{A\sqcup E\sqcup F,B\sqcup E\sqcup F}
            \\
            &\qquad\qquad\qquad=\sum_{\substack{H\subseteq\mathcal M(R^c)\\ |R\cup\operatorname{supp}_{\rm f}(H)|\le k}}
            s_R(A,B;H)\,\chi_{A\sqcup H,B\sqcup H}
            \sum_{V\subseteq\operatorname{supp}_{\rm f}(H)}(-1)^{|V|}
            =\chi_{A,B}.
        \end{align*}
        Indeed, $E$ and $F$ occupy disjoint mode sets. For fixed $H$, choosing the intermediate mode set $V\subseteq\operatorname{supp}_{\rm f}(H)$ uniquely fixes $E=H\cap\mathcal M(V)$ and $F=H\setminus E$. The final equality follows from inclusion--exclusion: the inner sum is zero unless $H=\varnothing$, when it equals one. This proves \eqref{eq:structure-signed-inverse}, whose exponent counts modes rather than Majorana labels.

        Next, we control error propagation through $\mathsf T$ and $\mathsf T^{-1}$. Since supports only shrink, maximum absolute column sums bound their induced local coefficient one-norms. Each input coefficient contributes to at most $2^k$ outputs by removing subsets of its occupied modes with matching Majorana labels in both indices. The corresponding entries of $\mathsf T$ and, by \eqref{eq:structure-signed-inverse}, $\mathsf T^{-1}$ are signs $\pm1$. Thus both induced norms and degree amplification are at most $2^k$, and the condition number, the product of these norms, is at most $4^k$.

        To relate these bounds to the physical coefficients, let $\mathsf F$ map $x=(h,G)$ to the Majorana-superoperator coefficients indexed by $\mathcal I_k^\chi$, and let $\Psi$ extract $(h,G)$ by \eqref{eq:G-from-chi} and \eqref{eq:h-from-chi}. In \eqref{eq:local-gksl}, each Hamiltonian coordinate contributes two entries of modulus $|h_A|$, and each dissipative coordinate contributes at most three entries of total modulus at most $2|G_{A,B}|$. These contributions do not enlarge supports, and extraction does not increase degree. Hence $\Psi\mathsf F=\mathrm{id}$ and
        \begin{equation}
            \|\mathsf F x\|_{\mathrm{loc},1}\le2\|x\|_{\mathrm{loc},1},
            \qquad
            \deg(\mathsf F x)\le3\deg(x),
            \qquad
            \|\Psi z\|_{\mathrm{loc},1}\le\|z\|_{\mathrm{loc},1},
            \label{eq:structure-coefficient-maps}
        \end{equation}
        We now transfer the sparse approximation in \eqref{eq:structure-approx-degree} to the restricted coefficients. Set $y^{\rm big}:=\mathsf T\mathsf F x^{\rm big}$. Since $y=\mathsf T\mathsf F x$, the preceding bounds give
        \begin{equation*}
            \deg(y^{\rm big})\le3\cdot2^k\mathsf d,
            \qquad
            \|y-y^{\rm big}\|_{\mathrm{loc},1}\le2^{k+1}b.
        \end{equation*}
        We next bound the error of the thresholded estimates $\widetilde y$ defined in \eqref{eq:structure-marginal-threshold}. On the support of $y^{\rm big}$, an estimate satisfying $|\widehat y_v|>2\nu$ has error at most $\nu$, while a discarded one has error $|y_v|\le|\widehat y_v|+\nu\le3\nu$. Outside that support, $y_v^{\rm big}=0$. A discarded entry has error $|y_v|$, while an entry with $|\widehat y_v|>2\nu$ has error at most $\nu<|y_v|$. Summing per mode yields
        \begin{equation*}
            \|\widetilde y-y\|_{\mathrm{loc},1}
            \le3\nu\deg(y^{\rm big})+\|y-y^{\rm big}\|_{\mathrm{loc},1}
            \le9\cdot2^k\mathsf d\nu+2^{k+1}b.
        \end{equation*}
        Applying the reconstruction map $\Psi\mathsf T^{-1}$ now gives
        \begin{equation*}
            \|\widehat x-x\|_{\mathrm{loc},1}
            \le2^k\|\widetilde y-y\|_{\mathrm{loc},1}
            \le9\cdot4^k\mathsf d\nu+2\cdot4^k b,
        \end{equation*}
        proving \eqref{eq:structure-threshold-error}.

        Finally, we bound the output degree. Every index with $|\widehat y_v|>2\nu$ outside the support of $y^{\rm big}$ has $|y_v|>\nu$, so the tail bound allows at most $2^{k+1}b/\nu$ such indices per mode. Together with $\deg(y^{\rm big})\le3\cdot2^k\mathsf d$, this gives $\deg(\widetilde y)=\mathcal O_k(\mathsf d+b/\nu)$. Inversion increases degree by at most $2^k$, and extraction does not increase it. Since the true $h$ is real and $G$ is Hermitian, taking real and Hermitian parts preserves the error bound and increases degree by at most a factor two.
