<!-- definition-1: body_md -->
\label{def:sym}
    Let $\ensuremath{\mathbf{v}}_k \in \mathbb{R}^{n^k}$ be a vector corresponding to a $k$-way tensor $\ensuremath{\mathbf{V}}_k \in \mathbb{R}^{n \times n \times \dots \times n}$, such that $\textnormal{vec}\ensuremath{\if$$ \else \left[\right]\fi}(\ensuremath{\mathbf{V}}_k) = \ensuremath{\mathbf{v}}_k$.
    We say that the vector $\ensuremath{\mathbf{v}}_k$ is \emph{symmetric} if $\ensuremath{\mathbf{V}}_k$ is a symmetric tensor, i.e., for any permutation $\pi$ of the indices $\{1,2,\dots,k\}$, the following holds:
    \begin{equation*}
        \ensuremath{\mathbf{V}}_k(i_1,i_2,\dots,i_k) = \ensuremath{\mathbf{V}}_k(i_{\pi(1)},i_{\pi(2)},\dots,i_{\pi(k)}), \quad \forall i_1,i_2,\dots,i_k \in \{1,2,\dots,n\},
    \end{equation*}
    where $\ensuremath{\mathbf{V}}_k(i_1,i_2,\dots,i_k)$ denotes the entry of $\ensuremath{\mathbf{V}}_k$ corresponding to the multi-index $(i_1,i_2,\dots,i_k)$.

<!-- theorem-1: body_md -->
\label{ch3:thm:albrekht}
    Assume that the functions $\ensuremath{\mathbf{f}}(\ensuremath{\mathbf{x}})$ and $\ensuremath{\mathbf{g}}(\ensuremath{\mathbf{x}})$ in the dynamics \cref{ch3:eq:FOM-NL}, along with the functions $\bm{Q}(\ensuremath{\mathbf{x}})$ and $\bm{R}(\ensuremath{\mathbf{x}})$ in the cost function \cref{ch3:eq:opt-contr-problem}, are analytic about $\ensuremath{\mathbf{x}}=\ensuremath{\mathbf{0}}$.
    Define the matrices $\ensuremath{\mathbf{A}} = \partial \ensuremath{\mathbf{f}}/\partial \ensuremath{\mathbf{x}}(\ensuremath{\mathbf{0}})$, $\ensuremath{\mathbf{B}} = \ensuremath{\mathbf{g}}(\ensuremath{\mathbf{0}})$, $\ensuremath{\mathbf{Q}}_2 = \bm{Q}(\ensuremath{\mathbf{0}})$, and $\ensuremath{\mathbf{R}}_0 = \bm{R}_0(\ensuremath{\mathbf{0}})$.
    Further assume that the pair $(\ensuremath{\mathbf{A}}, \ensuremath{\mathbf{B}})$ is stabilizable and the pair $(\ensuremath{\mathbf{A}}, \ensuremath{\mathbf{Q}}_2)$ is detectable.
    These conditions guarantee that the associated linear-quadratic regulator (LQR) problem admits a unique stabilizing solution and the corresponding closed-loop linearization is Hurwitz.
    Then the following statements hold for the nonlinear optimal control problem \eqref{ch3:eq:opt-contr-problem}:
    \begin{enumerate}
        \item the value function and the optimal feedback law are
              analytic about $\ensuremath{\mathbf{x}}=\ensuremath{\mathbf{0}}$ and can be locally approximated by polynomials;
        \item the lowest degree polynomial term in the value function is degree~$2$, whose polynomial coefficient
              is given by the unique stabilizing solution to the algebraic Riccati equation (ARE) associated with the LQR problem on the linearized dynamics;
        \item the remaining higher degree polynomial coefficients of the value function are solutions to
              linear algebraic equations that are entirely determined by the already-computed polynomial coefficients.
    \end{enumerate}

\label{ch3:thm:kiPoly}
    Consider the special case of the optimal control problem \cref{ch3:eq:FOM-Poly,ch3:eq:opt-contr-problem} given by the polynomial performance index \cref{ch3:eq:performanceIndex-Poly} and polynomial dynamics \cref{ch3:eq:FOM-Poly}.
    Assume that
    $\ensuremath{\mathbf{Q}}_2 \succeq \ensuremath{\mathbf{0}}$, $\ensuremath{\mathbf{R}}_0 \succ \ensuremath{\mathbf{0}}$,
    $\ensuremath{\mathbf{E}}$ is nonsingular,
    and assume that the the pair $(\ensuremath{\mathbf{E}}^{-1}\ensuremath{\mathbf{A}},\ensuremath{\mathbf{E}}^{-1}\ensuremath{\mathbf{B}})$ is stabilizable and $(\ensuremath{\mathbf{E}}^{-1}\ensuremath{\mathbf{A}},\ensuremath{\mathbf{Q}})$ is detectable.
    Let the optimal feedback law $\ensuremath{\mathbf{u}}(\ensuremath{\mathbf{x}})$ from~\cref{ch3:eq:Ki-coeffs} be expanded and truncated to $d-1$ terms, with the coefficients $\ensuremath{\mathbf{K}}_i \in \mathbb{R}^{m \times n^i}$ for $i=1,2,\dots,d-1$.
    Then $\overline\ensuremath{\mathbf{V}}_2$ and $\ensuremath{\mathbf{K}}_1$ are the unique stabilizing solution to the generalized algebraic Riccati equation (GARE) and the linear gain matrix, respectively, associated with the LQR problem:
    \begin{align}
         & \ensuremath{\mathbf{A}}^\top  \overline\ensuremath{\mathbf{V}}_2 \ensuremath{\mathbf{E}} +   \ensuremath{\mathbf{E}}^\top  \overline\ensuremath{\mathbf{V}}_2 \ensuremath{\mathbf{A}}- \ensuremath{\mathbf{E}}^\top  \overline\ensuremath{\mathbf{V}}_2 \ensuremath{\mathbf{B}} \ensuremath{\mathbf{R}}_0^{-1} \ensuremath{\mathbf{B}}^\top \overline\ensuremath{\mathbf{V}}_2 \ensuremath{\mathbf{E}}
        + \ensuremath{\mathbf{Q}}_2 = \ensuremath{\mathbf{0}}, \label{ch3:eq:ARE}                                                                                                  \\
         & \ensuremath{\mathbf{K}}_1      = - \ensuremath{\mathbf{R}}_0^{-1} \ensuremath{\mathbf{B}}^\top \overline\ensuremath{\mathbf{V}}_2 \ensuremath{\mathbf{E}}.\label{ch3:eq:K1}
    \end{align}
    For $3 \leq k \leq d$, the linear systems
    \begin{equation}
        \begin{split}
            \ensuremath{\mathcal{L}}^\ensuremath{\mathbf{E}}_{k}(\ensuremath{\mathbf{A}} + \ensuremath{\mathbf{B}}\ensuremath{\mathbf{K}}_1 )^\top \bar{\ensuremath{\mathbf{v}}}_k & =
            -\rm{symmetrize}\left(\ensuremath{\mathbf{b}}\right)  ,\label{ch3:eq:value-function-linear-system} \\
            \ensuremath{\mathbf{b}}                                           & \coloneqq
            \sum_{\mathclap{i+p=k+1}} \ensuremath{\mathcal{L}}^\ensuremath{\mathbf{E}}_{i} (\ensuremath{\mathbf{F}}_p)^\top \ensuremath{\mathbf{v}}_i
            + \sum_{\mathclap{i+p+q=k+1}} i
            \textnormal{vec}\ensuremath{\if$$ \else \left[\right]\fi}\left[\ensuremath{\mathbf{K}}_q^\top\ensuremath{\mathcal{G}}\ensuremath{\mathcal{V}}^\ensuremath{\mathbf{E}}_{p,i}\right]
            +  \ensuremath{\mathbf{q}}_k
            + \sum_{\mathclap{i+p+q=k}} \textnormal{vec}\ensuremath{\if$$ \else \left[\right]\fi}\left[\ensuremath{\mathbf{r}}_i^\top \left(\ensuremath{\mathbf{K}}_p \otimes \ensuremath{\mathbf{K}}_q \right)\right] ,
        \end{split}
    \end{equation}
    where $\ensuremath{\mathcal{G}}\ensuremath{\mathcal{V}}^\ensuremath{\mathbf{E}}_{p,i} = {\text{reshape}\left(\ensuremath{\mathbf{G}}_p^\top\overline\ensuremath{\mathbf{V}}_i {\ensuremath{\mathbf{E}}}^{\tikz[baseline=(char.base)]{\node(char)[shape=rounded rectangle,draw,inner sep=0.6pt,minimum height=1.5ex]{\tiny{\ensuremath{i-1}}};}},m,n^{k-q}\right)}$,
    are uniquely solvable, and their solutions $\bar{\ensuremath{\mathbf{v}}}_k \in \mathbb{R}^{n^k}$ are used to determine the gain coefficients
    $\ensuremath{\mathbf{K}}_{k-1}$ for $3 \leq k \leq d$ as
    \begin{align}
        \ensuremath{\mathbf{K}}_{k-1} & =  \ensuremath{\mathbf{R}}_0^{-1}  \left[- \sum_{\mathclap{\substack{p+q=k-1 \\q\neq0}}}  \ensuremath{\mathbf{R}}_q^\top \left(\ensuremath{\mathbf{K}}_p \otimes \ensuremath{\mathbf{I}}_{n^q}\right)   -\sum_{\mathclap{i+p=k}} \frac{1}{2} i
            \ensuremath{\mathcal{G}}\ensuremath{\mathcal{V}}^\ensuremath{\mathbf{E}}_{p,i}
            \right]. \label{ch3:eq:control-gain-linear-system}
    \end{align}

<!-- proof-2: body_md -->
The complexity of \cref{ch3:alg:full-PPR} can be assessed using a floating point operation (flop) count.
    Standard solvers for the ARE in line 1 require $O(n^3)$ operations (low-rank solvers are discussed in \cref{ch3:sec:lr-adaptation}).
    Once $\overline\ensuremath{\mathbf{V}}_2$ is known, $\ensuremath{\mathbf{K}}_1$ can be formed with $O(m^3n)$ operations by solving $n$ linear systems with respect to inverting $\ensuremath{\mathbf{R}}_0$.
    Often $\ensuremath{\mathbf{R}}_0^{-1}$ is known explicitly, in which case $\ensuremath{\mathbf{K}}_1$ can be formed with $O(m^2n)$ operations.
    The operations of forming and appropriately symmetrizing the quantities in lines 4-7 can be performed efficiently by reshaping Kronecker products as matrix multiplication operations.
    The main workhorse is the Kronecker-Vec identity $\left(\ensuremath{\mathbf{B}}^\top \otimes \ensuremath{\mathbf{A}} \right) \textnormal{vec}\ensuremath{\if$\ensuremath{\mathbf{D}}$ \else \left[\ensuremath{\mathbf{D}}\right]\fi} = \textnormal{vec}\ensuremath{\if$\ensuremath{\mathbf{A}}\ensuremath{\mathbf{D}}\ensuremath{\mathbf{B}}$ \else \left[\ensuremath{\mathbf{A}}\ensuremath{\mathbf{D}}\ensuremath{\mathbf{B}}\right]\fi}$, as described in \cite[Sec. 1.3.7]{Golub2013}.
    In the case of repeated Kronecker products, the Kronecker-Vec relation can be applied recursively.
    This allows the linear systems for $\bar\ensuremath{\mathbf{v}}_k$ and $\ensuremath{\mathbf{K}}_{k-1}$ to be formed with $O(n^{k+1})$ complexity.


    Solving the linear system \cref{ch3:eq:value-function-linear-system} for $\bar\ensuremath{\mathbf{v}}_k$ in line 6 using standard methods prohibitively expensive, with a cost of $O(n^{3k})$ due to the vector $\bar\ensuremath{\mathbf{v}}_k$ having $n^k$ unknowns.
    Instead, as described in \cref{ch3:sec:linear-solver}, we leverage structured Lyapunov solvers that scale as $O(n^{k+1})$.
    Once $\bar\ensuremath{\mathbf{v}}_k$ is known, $\ensuremath{\mathbf{K}}_{k-1}$ can be formed with $O(m^3n^{k-1})$ operations by solving the $n^{k-1}$ columns as linear systems with respect to inverting $\ensuremath{\mathbf{R}}_0$.
    Similar to the case for $\ensuremath{\mathbf{K}}_{1}$, which corresponds to $k=2$, when $\ensuremath{\mathbf{R}}_0^{-1}$ is known explicitly, $\ensuremath{\mathbf{K}}_{k-1}$ can be formed with $O(m^2n^{k-1})$ operations.

    Overall, the complexity of \cref{ch3:alg:full-PPR} for computing a degree $d$ value function approximation and degree $d-1$ approximation of the optimal feedback law is $O(n^{d+1})$.
    The memory requirements for this algorithm are $O(n^d)$, primarily due to the need to store the right-hands-side vector $\ensuremath{\mathbf{b}} \in \mathbb{R}^{n^d}$ in \cref{ch3:eq:value-function-linear-system} and the resulting value function coefficient $\bar\ensuremath{\mathbf{v}}_k \in \mathbb{R}^{n^d}$
    (the structured $k$-way Lyapunov solvers never form the matrix $\ensuremath{\mathcal{L}}_k^E(\ensuremath{\mathbf{A}} + \ensuremath{\mathbf{B}}\ensuremath{\mathbf{K}}_1)^\top$).
    The highest-order gain coefficient $\ensuremath{\mathbf{K}}_{d-1}$ requires $O(mn^{d-1})$ storage, so the memory usage is dominated by the value function computations.
<!-- end proof-2 -->

    The implied constant factor in the big-O notation can depend linearly on $d$, but the dominant contributor to the increase in computational complexity with respect to polynomial degree is the exponential factor $n^{d+1}$.

        \end{split}

    where $\ensuremath{\mathcal{G}}\ensuremath{\mathcal{V}}^r_{p,i} = {\text{reshape}\left({\ensuremath{\mathbf{G}}^r_p}^\top \ensuremath{\mathbf{V}}^r_i, m,r^{k-q}\right)}$, are uniquely solvable, and their solutions ${\ensuremath{\mathbf{v}}}^r_k \in \mathbb{R}^{r^k}$ are used to determine

<!-- remark-3: body_md -->
In practice, \cref{ch3:cor:vi-Ki-Poly_approxs} should not be invoked to obtain an approximation to the value function $V(\ensuremath{\mathbf{x}})$ and feedback law $\ensuremath{\mathbf{u}}(\ensuremath{\mathbf{x}})$, as it involves scaling the computed reduced-order coefficients back up to the full dimension.
    This is expensive both in terms of flops and memory.
    Rather, the reduced-order state approximation $\ensuremath{\mathbf{x}}_r = \ensuremath{\mathbf{T}}^\top \ensuremath{\mathbf{x}}$ should be used in the evaluation of the higher-order terms.
    For example, to compute a cubic approximation to $\ensuremath{\mathbf{u}}(\ensuremath{\mathbf{x}})$ for a given state $\ensuremath{\mathbf{x}}$, instead of evaluating $\ensuremath{\mathbf{u}}(\ensuremath{\mathbf{x}}) \approx \ensuremath{\mathbf{K}}_1 \ensuremath{\mathbf{x}} + \ensuremath{\mathbf{K}}_2^r {{{\ensuremath{\mathbf{T}}^\top}}^{\tikz[baseline=(char.base)]{\node(char)[shape=rounded rectangle,draw,inner sep=0.6pt,minimum height=1.5ex]{\tiny{\ensuremath{2}}};}}} {\ensuremath{\mathbf{x}}}^{\tikz[baseline=(char.base)]{\node(char)[shape=rounded rectangle,draw,inner sep=0.6pt,minimum height=1.5ex]{\tiny{\ensuremath{2}}};}} + \ensuremath{\mathbf{K}}_2^r {{{\ensuremath{\mathbf{T}}^\top}}^{\tikz[baseline=(char.base)]{\node(char)[shape=rounded rectangle,draw,inner sep=0.6pt,minimum height=1.5ex]{\tiny{\ensuremath{3}}};}}} {\ensuremath{\mathbf{x}}}^{\tikz[baseline=(char.base)]{\node(char)[shape=rounded rectangle,draw,inner sep=0.6pt,minimum height=1.5ex]{\tiny{\ensuremath{3}}};}}$, one should
    first form $\ensuremath{\mathbf{x}}_r = \ensuremath{\mathbf{T}}^\top \ensuremath{\mathbf{x}}$ and then evaluate $\ensuremath{\mathbf{u}}(\ensuremath{\mathbf{x}}) \approx \ensuremath{\mathbf{K}}_1 \ensuremath{\mathbf{x}} + \ensuremath{\mathbf{K}}_2^r {\ensuremath{\mathbf{x}}_r}^{\tikz[baseline=(char.base)]{\node(char)[shape=rounded rectangle,draw,inner sep=0.6pt,minimum height=1.5ex]{\tiny{\ensuremath{2}}};}} + \ensuremath{\mathbf{K}}_2^r  {\ensuremath{\mathbf{x}}_r}^{\tikz[baseline=(char.base)]{\node(char)[shape=rounded rectangle,draw,inner sep=0.6pt,minimum height=1.5ex]{\tiny{\ensuremath{3}}};}}$.

<!-- remark-5: body_md -->
Truncated polynomial feedback law approximations are not guaranteed to preserve global, and in some cases even local, stability of the closed-loop system.
    However, unlike other polynomial approximation techniques such as Galerkin approximations,
    Al'brekht's method \emph{does} guarantee local stabilization, which it inherits from the assumption that the linearized dynamics are stabilized by the solution to the Riccati equation.
    As such, the controllers computed using \cref{ch3:alg:full-PPR,ch3:alg:reduced-PPR} provide local stability guarantees, since they retain the full-order Riccati solution.
    Conversely, controllers computed using \cref{ch3:alg:lr-PPR} are only guaranteed to be locally stabilizing if the low-rank Riccati solver converges to a stabilizing solution.

The continuous-time generalized linear time-invariant (LTI) system
    \begin{equation}\label{eq:LTI}
        \begin{split}
            \dot{\ensuremath{\mathbf{x}}} & = \ensuremath{\mathbf{A}} \ensuremath{\mathbf{x}} + \ensuremath{\mathbf{B}} \ensuremath{\mathbf{u}}, \\
            \ensuremath{\mathbf{y}}       & = \ensuremath{\mathbf{C}} \ensuremath{\mathbf{x}},
        \end{split}
    \end{equation}
    is stabilizable if and only if
    \begin{align*}
        \text{rank}\begin{bmatrix}
                       \ensuremath{\mathbf{A}}-\lambda\ensuremath{\mathbf{I}} & \ensuremath{\mathbf{B}}
                   \end{bmatrix} = n, \qquad \forall \lambda \in \mathbb{C} : \mathfrak{R}[\lambda] \geq 0.
    \end{align*}
    The continuous-time generalized LTI system \cref{eq:LTI} is detectable if and only if
    \begin{align*}
        \text{rank}\begin{bmatrix}
                       \ensuremath{\mathbf{A}}-\lambda\ensuremath{\mathbf{I}} \\ \ensuremath{\mathbf{C}}
                   \end{bmatrix} = n, \qquad \forall \lambda \in \mathbb{C} : \mathfrak{R}[\lambda] \geq 0.
    \end{align*}
