\label{defi:SOLA}
    We say that $u\in W^{1,max\{1, p-1\}}_{0,\omega^{p'}}(\Omega)$ is a SOLA to the problem \eqref{main.eq} if  
\begin{equation*}\label{weak-0}
\int_{\Omega} |\mathbb{M} Du|^{p-2} \mathbb{M}^{2}Du\cdot D\varphi\,dx = \int_{\Omega} \varphi\, d\mu \quad\text{for all } \; \varphi\in C^{\infty}_{c}(\Omega),
\end{equation*}
and there exists a sequence of weak solutions $\{u_k\}_{k\ge1}\subset W^{1,p}_{0,\omega}(\Omega)$ to
\begin{equation}\label{weak-1}
  \left\{ \begin{aligned}
      -{\rm div} \big(|\mathbb{M}(x)Du_k|^{p-2} \mathbb{M}(x)^2 Du_k\big) &= \mu_k  && in\;   \Omega,\\
                      u_k &= 0 && on\;   \partial\Omega,
    \end{aligned} \right.
\end{equation}
such that 
\begin{equation}\label{weak-2}
u_k \to u
\quad \text{in } W^{1,\max\{1,p-1\}}_{0,\omega^{p'}}(\Omega)
\quad \text{as } k\to\infty ,
\end{equation}
where the sequence $\{\mu_k\}_{k\ge1}\subset L^{\infty}(\Omega)$ converges weakly to $\mu$
in the sense of measure and satisfies
\begin{equation}\label{weak-3}
 \limsup_{k\to\infty} |\mu_k|(V)\le |\mu|(\overline V)
\end{equation}
for each open set $V \Subset \Omega$.

<!-- proof-1: body_md -->
The strategy of the proof relies on a standard approximation argument, which is carried out in two steps.
We first derive uniform $L^{\tilde{q}}$-estimate for the weak solutions to \eqref{main.eq} with every $1<\tilde q<\frac{n(p-1)}{n-1}$ under the assumption that $\mu$ is regular.
In the second step, we construct a sequence of solutions to regularized problems; by a compactness argument based on the boundedness from the first step, we then show that a subsequence converges to a limit function, which is shown to be a SOLA to \eqref{main.eq}.

To obtain the uniform estimate on the domain $\Omega$ based on the {small log-BMO condition}, which is local property in nature, in the first step, we will use covering arguments.
Hence, for the simplicity of notation, the generic constant $c$ in this section depends only on $n,p, \tilde{q}, \Lambda, R_0, d$.


\medskip
\noindent\textbf{Step 1: Uniform $L^{\tilde{q}}$-estimate.}
In this step, we assume that $\mu$ belongs to $C^{\infty}(\Omega)\cap L^{\infty}(\Omega)$ to ensure the existence of a unique weak solution $u \in W^{1,p}_{0,\omega}(\Omega)$ to \eqref{main.eq}.
We now assert the following statement:
\begin{quote}
For every $1<\tilde q<\frac{n(p-1)}{n-1}$, there exists {a constant $\kappa_1=\kappa_1(n,p,\tilde q,\Lambda)>0$ such that if $[\log\mathbb{M}]_{\operatorname{BMO}_{R_0}(\mathbb{R}^n)}\le \kappa_1$ for some $R_0>0$}, then there is a constant $c=c(n, p, \tilde q, \Lambda, R_0, d)$ satisfying
\begin{equation}\label{eq.3.1}
\int_{\Omega} \bigl(\omega^{p'}|Du|\bigr)^{\tilde q}\,dx \le c\,|\mu|(\Omega)^{\frac{\tilde q}{p-1}}.
\end{equation}
\end{quote}

Define $u_{\pm}:=\max\{\pm u,0\}$, \(\Omega^{+}:=\{x\in\Omega: u(x)>0\}\), and \(\Omega^{-}:=\Omega\setminus\Omega^{+}\).
Testing \eqref{main.eq} with $\varphi = h^{1-\xi} - (h+u_{\pm})^{1-\xi}\in W^{1,p}_{0,\omega}(\Omega)$ for some constants $h>0$, $\xi>1$ chosen shortly, we have
\begin{equation*}\label{est.u.1}
\int_{\Omega} \frac{\omega^{p}|Du|^{p}}{(h+|u|)^{\xi}}\,dx
\le  c\frac{h^{1-\xi}}{\xi-1}|\mu|(\Omega).
\end{equation*}
Taking
$$
 h = \left(\int_\Omega {\omega}^\frac{\tilde q p'}{p-\tilde q}|u|^{\frac{\xi \tilde q}{p-\tilde q}}\,dx\right)^{\frac{p-\tilde q}{\xi \tilde q}}\left(\int_\Omega {\omega}^\frac{\tilde q p'}{p-\tilde q}\,dx\right)^{-\frac{p-\tilde q}{\tilde q\xi}}, 
 $$
H\"older's inequality gives
\begin{equation}\label{eq3-5}
\begin{aligned}
\int_\Omega ({\omega}^{p'}|Du|)^{\tilde q}\,dx
&\le \left(\int_\Omega \frac{\omega^p|Du|^p}{(h+|u|)^\xi}\,dx\right)^{\frac{\tilde q}{p}}
\left(\int_\Omega {\omega}^\frac{{(p'-1)p\tilde q}}{p-\tilde q}(h+|u|)^{\frac{\xi \tilde q}{p-\tilde q}}\,dx\right)^{\frac{p-\tilde q}{p}} \\
                                                                                                                                                                                                                                                                                                                  
&\le c  \left(\frac{|\mu|(\Omega)}{\xi-1}\right)^{\frac{\tilde q}{p}}h^\frac{\tilde q}{p}
\left( \int_\Omega {\omega}^\frac{\tilde q p'}{p-\tilde q}\,dx\right)^{\frac{p-\tilde q}{p}}.
\end{aligned}
\end{equation}

To establish the estimate \eqref{eq.3.1}, it remains to bound $h$.
Taking $\xi = \frac{n - \tilde{q} + n(p - \tilde{q})}{2(n-\tilde{q})}$, we have
\begin{equation}\label{param.xi}
\frac{n \tilde{q}(\xi -1) p'}{n(p-\tilde{q}) - \xi(n-\tilde{q})}=\frac{n\tilde{q}p'}{n-\tilde{q}} > \max \left\{ \frac{n\tilde{q}}{n-\tilde{q}},\frac{\tilde{q}p'}{p-\tilde{q}} \right\}.
\end{equation}
Assume
\begin{equation}\label{exist.M.cond}
{[\log\mathbb{M}]_{\mathrm{BMO}_{R_0}(\mathbb{R}^n)}} \leq \gamma \max \left\{\tilde{q}', \frac{n\tilde qp'}{n-\tilde q}\right\}^{-1} =: {\kappa_1},
\end{equation}
where $\gamma = \gamma(n, \Lambda) > 0$ is the constant given in Lemma~\ref{lem:logBMO}.
Recalling $2 - \frac{1}{n} < p \leq n$, we note basic inequalities
\begin{equation}\label{param.xi.1}
1 < \tilde{q} < \frac{n(p-1)}{n-1} \leq n, \quad 1 < \xi < \dfrac{n(p-\tilde{q})}{n-\tilde{q}}, \quad \text{and} \quad \frac{\xi \tilde{q}}{p-\tilde{q}} < \frac{n \tilde{q}}{n - \tilde{q}} .
\end{equation}
Now we apply the Poincar\'e type inequality given in Lemma~\ref{poincare.global} to get
\begin{equation}\label{exist.bd.1}
\begin{aligned}
\int_\Omega {\omega}^\frac{\tilde q p'}{p-\tilde q}|u|^{\frac{\xi \tilde q}{p-\tilde q}}\,dx
& = \int_{\Omega} \left( \omega^{p'} |u| \right)^{\frac{\xi \tilde{q}}{p-\tilde{q}}} \omega^{-\frac{\tilde{q} (\xi -1) p'}{p- \tilde{q}}} {\,dx} \\
& \leq c \left( \int_{\Omega} ( \omega^{p'} |u| )^{\frac{n \tilde{q}}{n- \tilde{q}}} {\,dx} \right)^{\frac{(n-\tilde{q})\xi}{n(p-\tilde{q})}} 
\left( \int_{\Omega} \omega^{- \frac{n\tilde{q}p'}{n-\tilde{q}}}\;dx \right)^{1-\frac{(n-\tilde{q})\xi}{n(p-\tilde{q})}} \\
& \leq c \left( \int_{\Omega} ( \omega^{p'} |Du| )^{\tilde{q}} {\,dx} \right)^{\frac{\xi}{p-\tilde{q}}} 
\left( \int_{\Omega} \omega^{- \frac{n\tilde{q}p'}{n-\tilde{q}}}\;dx \right)^{1-\frac{(n-\tilde{q})\xi}{n(p-\tilde{q})}}.
\end{aligned}
\end{equation}
It is worth mentioning that by Lemma~\ref{lem:logBMO} with \eqref{exist.M.cond}, the last line is finite.


Combining \eqref{eq3-5} and \eqref{exist.bd.1}, we have
\begin{equation}\label{eq13}
\begin{aligned}
\int_\Omega ({\omega}^{p'}|Du|)^{\tilde q}\,dx
& \leq c |\mu|(\Omega)^{\frac{\tilde q}{p}}
\left(\int_\Omega (\omega^{p'}|Du| )^{\tilde q} dx\right)^{\frac{1}{p}},
\end{aligned}
\end{equation}
where $c=c(n, p, \tilde q, \Lambda, R_0, d)>0.$ Here we have also used the fact that
\begin{align*}
\left(\int_\Omega \omega^{-\frac{n\tilde{q}p'}{n-\tilde{q}}} dx\right)^{\frac{(n-\tilde{q})\left(n(p-\tilde{q})-n+\tilde{q}\right)}
{np\left(n-\tilde{q}+n(p-\tilde{q})\right)}} 
\left(\int_\Omega \omega^{\frac{\tilde qp'}{p-\tilde q}} dx\right)^{\frac{(p-\tilde{q})\left(n(p-\tilde{q})-n+\tilde{q}\right)}
{p\left(n-\tilde{q}+n(p-\tilde{q})\right)}}
\leq c,
\end{align*}
which is a consequence of covering argument and Lemma~\ref{lem:logBMO} with the assumption \eqref{exist.M.cond}.
Absorbing the last integral term in \eqref{eq13} into the left-hand side, we finally arrive at \eqref{eq.3.1}.








\,

\noindent\textbf{Step 2: Approximation and Convergence.}

By a standard mollification, we construct a sequence $\{\mu_k\}_{k\ge 1} \subset C^\infty(\Omega)\cap L^{\infty}(\Omega)$ such that $\{ \mu_k \}$ converges to $\mu$ weakly in measure.
Then, there exist the corresponding weak solutions $\{u_k\}_{k\ge 1} \subset W^{1,p}_{0,\omega}(\Omega)$ such that each $u_k$ solves \eqref{main.eq} with the datum $\mu_k$.

Now we are going to prove that $\{ u_k \}_{k \ge 1}$ converges to a SOLA to \eqref{main.eq} up to non-relabeled subsequence.
To this end, we take
\begin{equation}\label{kappa.0}
\begin{aligned}
\kappa_0 := \gamma \max \left\{ \tilde{q}' , \frac{n \tilde{q}p' }{n - \tilde{q}} \right\}^{-1} \leq \gamma \max \left\{ 2, \frac{\tilde{q} p'}{\tilde{q} -2 +p} \right\}^{-1}
\end{aligned}
\end{equation}
with
\begin{equation}\label{tilde.q}
\tilde{q} := \frac12\left(\max\{1,p-1\} + \frac{n(p-1)}{n-1}\right) \in \left( \max\{1,p-1\} , \frac{n(p-1)}{n-1} \right),
\end{equation}
and assume
\begin{equation*}
\begin{aligned}
{[\log\mathbb{M}]_{\mathrm{BMO}_{R_0}(\mathbb{R}^n)}< \kappa_0.}
\end{aligned}
\end{equation*}
Then \eqref{eq.3.1} holds true for $u, \mu$ replaced by $u_k, \mu_k$ for each $k \in \mathbb{N}$.
By \eqref{weak-3}, for large enough $k \in \mathbb{N}$, we further have
\begin{equation}\label{ex.step2.bd}
\int_{\Omega} \bigl(\omega^{p'}|Du_k|\bigr)^{\tilde q}\,dx \le c\,|\mu|(\Omega)^{\frac{\tilde q}{p-1}}.
\end{equation}
Moreover, by Lemma~\ref{lem:logBMO} with \eqref{kappa.0}, $\| \omega \|_{L^{\gamma/\kappa_0}(B_{2d})} < \infty$.
We will use this fact later in this section without mentioning.
In the remaining part of this step, we use the temporary notation $c_{*}$ which stands for constants depending on $n, p, \Lambda, R_0, d, |\mu|(\Omega), \| \omega \|_{L^{\gamma/\kappa_0}(B_{2d})} $ and may vary from line to line.

From the fact that $\{u_k\}_{k\ge1}$ is uniformly bounded in $W^{1,\tilde q}_{0,\omega^{p'}}(\Omega)$, which is a reflexive Banach space when $\tilde q>1$, there exists a non-relabeled subsequence satisfying 
$$u_k \rightharpoonup u \quad\text{in } W^{1,\tilde q}_{0,\omega^{p'}}(\Omega).$$
Then the weighted Rellich--Kondrachov compactness theorem in \cite{K1997} implies that, after passing to a further subsequence if necessary, we have
$$
u_k \to u \text{ in } L^{\tilde q}_{\omega^{p'}}(\Omega)
\qquad\text{and}\qquad
u_k \to u \quad \text{a.e. in }\Omega.
$$
Consequently, $u_k$ converges to $u$ in measure with respect to the Lebesgue measure. That is, for any $\varepsilon > 0$ and sufficiently large $k$, we have
$$\mathrm{meas} \left\{x\in\Omega:|u_k-u|>\frac\varepsilon2 \right\}\to 0.$$
 Thus, for $l,m$ sufficiently large,
\begin{equation}\label{meas.est.mn}
\mathrm{meas}\left\{x\in\Omega:|u_l-u_m|>\varepsilon \right\} \le \mathrm{meas}\left\{x\in\Omega:|u_l-u|>\frac\varepsilon2 \right\}+\mathrm{meas}\left\{ x\in\Omega:|u-u_m|>\frac\varepsilon2 \right\} \leq \varepsilon.
\end{equation}


For a fixed $\varepsilon>0$, we consider the truncation function $\psi \in C(\mathbb{R},\mathbb{R})$ defined by
$$
\psi(s)=
\begin{cases}
\varepsilon  & \text{if } s>\varepsilon,\\[4pt]
s  & \text{if } |s|\le \varepsilon,\\[4pt]
-\varepsilon  & \text{if } s<-\varepsilon.
\end{cases}
$$
For any $l, m \in \mathbb{N}$, we test the corresponding equations for both $(\mu_l, u_l)$ and $(\mu_m, u_m)$ with $\varphi=\psi(u_l-u_m) \in W^{1,p}_{0,\omega}(\Omega)$, and then subtract the two equations to obtain
\begin{equation}\label{eq17}
\begin{aligned}
& \int_{D_{l,m,\varepsilon}} \langle|\mathbb{M} D u_l|^{p-2} \mathbb{M} D u_l - |\mathbb{M} D u_m|^{p-2} \mathbb{M} D u_m , \mathbb{M} D u_l - \mathbb{M} D u_m \rangle \,dx \\
& = \int_{\Omega} \psi'(u_l-u_m) \langle |\mathbb{M} D u_l|^{p-2} \mathbb{M} D u_l - |\mathbb{M}D u_m|^{p-2} \mathbb{M} D u_m , \mathbb{M} D u_l - \mathbb{M} D u_m \rangle \,dx \\
& = \int_{\Omega} \psi(u_l-u_m)\,d(\mu_l-\mu_m)
\leq 2 \varepsilon |\mu|(\Omega),
\end{aligned}
\end{equation}
where the set $D_{l,m,\varepsilon}$ is defined by
$$
D_{l,m,\varepsilon}
 := \left\{ x\in\Omega \,\big|\, |u_l(x)-u_m(x)| \le \varepsilon \right\}.
$$
Since the monotonicity estimate \eqref{mono.ineq2} takes different forms in the subquadratic and superquadratic cases, we estimate the left-hand side of \eqref{eq17} separately for $2-\frac{1}{n}<p\leq 2$ and $2<p\leq n.$


\textbf{Case 1:} $2-\frac1n<p\le2.$ 
By {\eqref{mono.ineq2}}, \eqref{Balance-1}, and \eqref{eq17},
\begin{equation*}\label{eq19}
\begin{aligned}
& \int_{D_{l,m,\varepsilon}} \omega^{p}(|Du_l| + |Du_m|)^{p-2}|Du_l-Du_m|^2\,dx \\
& \leq c \int_{D_{l,m,\varepsilon}} (|\mathbb{M}Du_l|+|\mathbb{M}Du_m|)^{p-2}|\mathbb{M}Du_l-\mathbb{M}Du_m|^2\,dx \\
& \leq c \int_{D_{l,m,\varepsilon}}
\langle |\mathbb{M}Du_l|^{p-2}\mathbb{M}Du_l-|\mathbb{M}Du_m|^{p-2}\mathbb{M}Du_m , \mathbb{M}Du_l-\mathbb{M}Du_m \rangle \,dx \\
& \leq c \varepsilon |\mu|(\Omega).
\end{aligned}
\end{equation*}
Hence, by H\"older's inequality,
\begin{equation}\label{eq20}
\begin{aligned}
& \int_{D_{l,m,\varepsilon}} \omega^{p'}|Du_l-Du_m|\,dx \\
&\le
\left(\int_{D_{l,m,\varepsilon}} \omega^{p} (|Du_{l}| + |Du_{m}|)^{p-2}|Du_l-Du_m|^2\,dx\right)^{\frac12} \cdot \\
& \qquad \cdot
\left(\int_{D_{l,m,\varepsilon}} \omega^{p'(3-p)} (|Du_l| + |Du_m|)^{2-p} \,dx\right)^{\frac12} \\
&\le c \varepsilon^{\frac12} |\mu|(\Omega)^{\frac{1}{2}}
\left(\int_{D_{l,m,\varepsilon}} \omega^{p'(3-p)} (|Du_l| + |Du_m|)^{2-p} \,dx\right)^{\frac12}.
\end{aligned}
\end{equation}
When $p=2$, we find
\begin{equation}\label{ex.step2.p2}
\int_{D_{l,m,\varepsilon}} \omega^{p'(3-p)} (|Du_l| + |Du_m|)^{2-p} \,dx \leq \int_{\Omega} \omega^2 \,dx < \infty.
\end{equation}
Moreover, when $2-\frac1n<p<2$, H\"older's inequality with exponents $\frac{\tilde q}{2-p}$ and $\frac{\tilde q}{\tilde q-2+p}$, and \eqref{ex.step2.bd} yield
\begin{equation}\label{ex.step2.ple2}
\begin{aligned}
& \int_{D_{l,m,\varepsilon}} \omega^{p'(3-p)} (|Du_l| + |Du_m|)^{2-p} \,dx \\
&\le
\left(\int_{D_{l,m,\varepsilon}} [\omega^{p' }(|Du_l|+|Du_m|)]^{\tilde q}\,dx\right)^{\frac{2-p}{\tilde q}}
\left(\int_{D_{l,m,\varepsilon}} \omega^{\frac{\tilde qp'}{\tilde q-2+p}}\,dx\right)^{\frac{\tilde q-2+p}{\tilde q}} \\
&\le
c |\mu|(\Omega)^{\frac{2-p}{p-1}} \left(\int_{\Omega} \omega^{\frac{\tilde qp'}{\tilde q-2+p}}\,dx\right)^{\frac{\tilde q-2+p}{\tilde q}}.
\end{aligned}
\end{equation}
Merging \eqref{eq20}, \eqref{ex.step2.p2}, and \eqref{ex.step2.ple2}, we deduce
\begin{equation}\label{eq21}
\begin{aligned}
\int_{D_{l,m,\varepsilon}} \omega^{p'}|Du_l-Du_m|\,dx
\le c_{*} \varepsilon^{\frac12}.
\end{aligned}
\end{equation}

On the other hand, by H\"older's inequality and \eqref{ex.step2.bd}, we have 
\begin{equation}\label{eq22}
\begin{aligned}
&\int_{\Omega\setminus D_{l,m,\varepsilon}} \omega^{p'}|Du_l-Du_m|\,dx\\
&\le
\left(\int_{\Omega\setminus D_{l,m,\varepsilon}} (\omega^{p'}|Du_l-Du_m|)^{\tilde q}\,dx\right)^{\frac{1}{\tilde q}}
\mathrm{meas}\{x\in\Omega:\ |u_l-u_m|>\varepsilon\}^{1-\frac{1}{\tilde q}} \\
&\le c |\mu|(\Omega)^{\frac{1}{p-1}}\,\mathrm{meas}\{x\in\Omega:\ |u_l-u_m|>\varepsilon\}^{1-\frac{1}{\tilde q}}.
\end{aligned}
\end{equation}
It then follows from \eqref{eq22} that
$$
\int_{\Omega\setminus D_{l,m,\varepsilon}} \omega^{p'}|Du_l-Du_m|\,dx
\le c \varepsilon^{1-\frac{1}{\tilde q}} |\mu|(\Omega)^{\frac{1}{p-1}} .
$$
Combining this with \eqref{eq21}, we conclude
$$
\int_\Omega \omega^{p'}|Du_l-Du_m|\,dx
\le c_{*} ( \varepsilon^{\frac12} + \varepsilon^{1-\frac{1}{\tilde q}}),
$$
which means $\{Du_k\}_{k\ge 1}$ is a Cauchy sequence in $L^1_{\omega^{p'}}(\Omega)$, and hence
\begin{equation}\label{eq23}
\begin{aligned}
\int_\Omega \omega^{p'}|Du_k-Du|\,dx\to 0.
\end{aligned}
\end{equation}

Finally, since $0<p-1\le1$, it follows from \eqref{Balance-1}, \eqref{mono.ineq1}, and \eqref{eq23} that
\begin{align*}
|\mathbb{M}Du_k|^{p-2}\mathbb{M}^2Du_k \to |\mathbb{M}Du|^{p-2}\mathbb{M}^2Du \quad \text{in} \quad L^{1}(\Omega)
\end{align*}
and therefore
$$
\int_\Omega |\mathbb{M}Du|^{p-2}\mathbb{M}^2Du\cdot D\varphi\,dx = \int_{\Omega} \varphi \, d \mu ,
\qquad \forall\,\varphi \in C_c^{\infty}(\Omega).
$$


\textbf{Case 2:} $2<p\le n.$ 
By \eqref{Balance-1}, \eqref{mono.ineq2}, and \eqref{eq17}, we have
\begin{equation*}             
\begin{aligned}
\int_{D_{l,m,\varepsilon}} \omega^{p}\,|D u_l - D u_m|^{p}\,dx
                                                                                                                                                                                                                                                                      
&\leq c \varepsilon |\mu|(\Omega).
\end{aligned}
\end{equation*}
By H\"older's inequality and Lemma~\ref{lem:logBMO} with \eqref{kappa.0}, it follows that
\begin{equation}\label{eq26}
\begin{aligned}
&\int_{D_{l,m,\varepsilon}} \omega^{p} |D u_l - D u_m|^{p-1}\,dx\\
&\le  \left( \int_{D_{l,m,\varepsilon}} \omega^{p} |D u_l - D u_m|^{p}\,dx
   \right)^{\frac{p-1}{p}}
   \left( \int_{D_{l,m,\varepsilon}} \omega^{p}\,dx \right)^{\frac1p} \\
&\le c \varepsilon^{\frac{p-1}{p}}|\mu|(\Omega)^{\frac{p-1}{p}}
   \left( \int_{\Omega} \omega^{p}\,dx \right)^{\frac1p} 
 \le c_* \varepsilon^{\frac{p-1}{p}}.
\end{aligned}
\end{equation}

On the other hand, combining \eqref{ex.step2.bd} and \eqref{meas.est.mn}, we obtain
\begin{equation}\label{eq28}
\begin{aligned}
&\int_{\Omega\setminus D_{l,m,\varepsilon}} \omega^{p}|Du_l-Du_m|^{p-1}\,dx\\
&\le
\left(\int_{\Omega\setminus D_{l,m,\varepsilon}} \left(\omega^{p'}|Du_l-Du_m|\right)^{\tilde q}dx\right)^{\frac{2(n-1)}{2n-1}}
\mathrm{meas}\{x\in\Omega:\ |u_l-u_m|>\varepsilon\}^{\frac{1}{2n-1}}\\
&\le c\varepsilon^{\frac{1}{2n-1}} |\mu|(\Omega).
\end{aligned}
\end{equation}
Hence, \eqref{eq26} and \eqref{eq28} imply that
$$
\int_\Omega \omega^{p}|Du_l-Du_m|^{p-1}\,dx
\le c_{*} (\varepsilon^{\frac{p-1}{p}} + \varepsilon^{\frac{1}{2n-1}}),
$$
which means that $\{Du_k\}_{k\ge1}$ is a Cauchy sequence in $L^{p-1}_{\omega^{p'}}(\Omega)$.
Therefore,
\begin{equation}\label{eq29}
\begin{aligned}
\int_\Omega \omega^{p}|Du_k-Du|^{p-1}\,dx\to 0.
\end{aligned}
\end{equation}

Finally, from \eqref{Balance-1}, \eqref{mono.ineq1}, \eqref{ex.step2.bd}, and \eqref{eq29}, we obtain
\begin{equation*}             
\begin{aligned}
&\int_{\Omega}\bigl||\mathbb{M} D u_k|^{p-2} \mathbb{M}^{2} D u_k-|\mathbb{M} D u|^{p-2} \mathbb{M}^{2} D u\bigr|\,dx\\
&\le c \left(\int_{\Omega}\omega^{p}\bigl(|Du_k|+|Du|\bigr)^{p-1} dx\right)^{\frac{p-2}{p-1}}\left(\int_{\Omega}\omega^{p}|Du_k-Du|^{p-1} dx\right)^{\frac{1}{p-1}} \\
&\le c\left(\int_{\Omega}\omega^{p}|Du_k|^{p-1} dx+\int_{\Omega}\omega^{p}|Du_k-Du|^{p-1} dx\right)^{\frac{p-2}{p-1}}\left(\int_{\Omega}\omega^{p}|Du_k-Du|^{p-1} dx\right)^{\frac{1}{p-1}}
 \to 0.
\end{aligned}
\end{equation*}
At this stage, by following the same reasoning given in the last part of {\bf Case 1}, we conclude that $u$ is a SOLA to \eqref{main.eq} in the case $2 < p \leq n$.
This completes the proof of Theorem~\ref{Existence.of.SOLA}.

\left(\frac{|\mu|(B_{{t} r})}{r^{n-1}}\right)^{\frac{1}{p-1}} 
+\frac{c}
{({t}-{s})^{\frac{n}{p-1}\left(2-\frac1{\tilde q}\right)}}
\left(\frac{|\mu|(B_{{t} r})}{r^{n-1}}\right)^\frac{1}{p-1}.
\end{aligned}
\end{equation*}
Lemma~\ref{lem-iteration} leads to \eqref{Reverse-G1}.

\label{comparison estimate}
Assume that \eqref{p} holds. Let $u\in W^{1,p}_{0,\omega}(\Omega)$ be the weak solution to \eqref{main.eq} and $v\in u+W^{1,p}_{0,\omega}(B_{2r})$ be the weak solution to \eqref{eq1}.
Let $\kappa_1=\kappa_1(n,p,\Lambda)>0$ be the constant defined in \eqref{exist.M.cond}, with $\tilde{q}$ as in \eqref{tilde.q}. If
$$[\log\mathbb{M}]_{\mathrm{BMO}({B_{3r}})}<\kappa_1,$$
then
\begin{equation}\label{est.lem.4.4}
\fint_{B_{2r}} |{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}}\,dx 
\le c\left\{\left(\frac{|\mu|(B_{2r})}{r^{n-1}}\right)^{\frac{\tilde{q}}{p-1}}+\chi_{\{p<2\}}\left(\frac{|\mu|(B_{2r})}{r^{n-1}}\right)^{\tilde{q}}\left(\fint_{B_{2r}} |{\mathbb{A}(Du)}|^{\tilde{q}}\,dx\right)^{2-p}\right\},
\end{equation}
where $c=c(n,p,\Lambda)>0$.
<!-- end lem-10 -->
<!-- proof-4: body_md -->
The proof is similar to that of {\bf Step 1} in Section~3. Hence we only emphasize the necessary modifications. 

We set $\xi=\frac{n-\tilde{q}+n(p-\tilde{q})}{2(n-\tilde{q})}.$ Since $\tilde{q}$ is defined as \eqref{tilde.q}, \eqref{param.xi} and \eqref{param.xi.1} are satisfied. For any constant $h>0$, testing \eqref{main.eq} and \eqref{eq1} with $\varphi = h^{1-\xi} - (h+(u-v)_{\pm})^{1-\xi} \in W^{1,p}_{0,\omega}(B_{2r})$, we obtain
\begin{equation}\label{eq38}
\fint_{B_{2r}} \frac{\langle|\mathbb{M} D u|^{p-2} \mathbb{M} Du - |\mathbb{M} Dv|^{p-2} \mathbb{M} Dv , \mathbb{M} Du - \mathbb{M} Dv \rangle }{(h+|u-v|)^{\xi}}\,dx
                                              
\le c \frac{h^{1-\xi}}{\xi-1} \frac{|\mu|(B_{2r})}{r^n}.
\end{equation}


When $p \geq 2$, it follows from \eqref{mono.ineq2} and \eqref{eq38} that
\begin{equation*}
\fint_{B_{2r}}
\frac{\omega^p |Du-Dv|^p}{(h+|u-v|)^\xi}\,dx
\le
c \frac{h^{1-\xi}}{\xi-1} \frac{|\mu|(B_{2r})}{r^n}.
\end{equation*}
At this point, the remaining part in this case follows the same lines as in {\bf Step 1} of the proof of Theorem~\ref{Existence.of.SOLA}, with $u$ replaced by $u-v$ and $\Omega$ replaced by $B_{2r}$. Consequently, we obtain
$$
\fint_{B_{2r}} |{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}}\,dx \le c\left(\frac{|\mu|(B_{2r})}{r^{n-1}}\right)^{\frac {\tilde{q}}{p-1}}.
$$

It remains to consider the case $2-{\frac1n}<p < 2$. In this case, we provide the proof in detail based on monotonicity estimate in \eqref{mono.ineq2}.
                                     
                              
                                                                            
                                                  
                
Since $\tilde q                          
<2$, H\"older's inequality with the conjugate exponents $\frac{2}{\tilde q}$ and $\frac{2}{2-\tilde q}$, together with \eqref{mono.ineq2} and \eqref{eq38}, yields
\begin{equation}\label{eq40}
\begin{aligned}
\fint_{B_{2r}} |{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}} \, dx
                                                                  
                                                                                                                                           
                                                                                                                                                                                                                  
&\le c\left[\fint_{B_{2r}}\frac{\omega^{p}|Du-Dv|^2}{(h+|u-v|)^\xi}(|Du|+|Dv|)^{p-2}\, dx\right]^\frac{\tilde{q}}{2}\mathbb{I}\\
&\le c \left[\frac{h^{1-\xi}}{\xi-1} \frac{|\mu|(B_{2r})}{r^n}\right]^\frac{\tilde{q}}{2}
\mathbb{I},
\end{aligned}
\end{equation}
where we denote
$$
\mathbb{I}:=\left(\fint_{B_{2r}}\omega^{\frac{(3-p)\tilde q p'}{2-{\tilde q}}}
(|Du|+|Dv|)^{\frac{(2-p){\tilde q}}{2-{\tilde q}}}
(h+|u-v|)^{\frac{\xi \tilde{q}}{2-\tilde{q}}}
\, dx\right)^\frac{2-\tilde{q}}{2}.
$$
To estimate $\mathbb{I}$, we take
$$
h=\left(\fint_{B_{2r}}\omega^{\frac{\tilde{q}p'}{p-\tilde{q}}}|u-v|^{\frac{\xi \tilde{q}}{p-\tilde{q}}}\,dx\right)^{\frac{p-\tilde{q}}{\xi \tilde{q}}}
\left(\fint_{B_{2r}}\omega^{\frac{\tilde{q}p'}{p-\tilde{q}}}\,dx\right)^{\frac{\tilde{q}-p}{\xi \tilde{q}}}.$$
Since $\tilde{q}<p$, by applying H\"older's inequality with exponents $\frac{2-\tilde{q}}{2-p}$ and $\frac{2-\tilde{q}}{p-\tilde{q}}$, we obtain
\begin{equation}\label{eq41}
\begin{aligned}
\mathbb{I}
&\le \left(\fint_{B_{2r}} \omega^{\tilde{q}p'}\bigl(|Du|+|Dv|\bigr)^{\tilde{q}}\,dx
\right)^{\frac{2-p}{2}}\left(\fint_{B_{2r}}\omega^{\frac{\tilde{q} p'}{p-\tilde{q}}}(h+|u-v|)^{\frac{\xi \tilde{q}}{p-\tilde{q}}}\,dx\right)^{\frac{p-\tilde{q}}{2}}\\
                                                                                                                                                             
                                                                                                                                                                                                                                                         
&\le c  h^\frac{\xi \tilde{q}}{2}
\left(\fint_{B_{2r}}\omega^{\frac{\tilde{q} p'}{p-\tilde{q}}}\,dx\right)^{\frac{p-\tilde{q}}{2}} 
\left[\left(\fint_{B_{2r}}{|{\mathbb{A}(Du)}|}^{\tilde{q}}\,dx\right)^{\frac{2-p}{2}}+\left(\fint_{B_{2r}}|{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}}\,dx\right)^{\frac{2-p}{2}}\right].
\end{aligned}
\end{equation}


It remains to estimate $h$. Repeating the argument used to derive \eqref{exist.bd.1}, we obtain
\begin{equation}\label{est.h.2}
h
\le c r \left( \fint_{B_{2r}} |{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}} {\,dx} \right)^{\frac{1}{\tilde{q}}}
\left( \fint_{B_{2r}} \omega^{- \frac{n\tilde{q}p'}{n-\tilde{q}}}\;dx \right)^{\frac{n(p-\tilde{q})-(n-\tilde{q})\xi}{n\tilde{q}\xi }}
\left(\fint_{B_{2r}}\omega^{\frac{\tilde{q}p'}{p-\tilde{q}}}\,dx\right)^{\frac{\tilde{q}-p}{\xi \tilde{q}}}.
\end{equation}


Combining \eqref{eq40}, \eqref{eq41}, and \eqref{est.h.2}, and then applying Young's inequality, we derive
\begin{equation*}\label{eq44}
\begin{aligned}
&\fint_{B_{2r}}|{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}}\,dx\\
                                                                                                                                                                                                                                                                                                                                                                              
                                                                                                                                                                                                                                                                                                                                                                                                   
                                                                                                                                                                          
                                                                                                                                                                                                                                                                                                                                                                                                                                                                
                                                                                                                                                                                                                                        
                                                                                                                                                                                                                                                                                                                      
&\le c\left(\frac{|\mu|(B_{2r})}{r^{n-1}}\right)^{\tilde{q}}
\left(\fint_{B_{2r}}|{\mathbb{A}(Du)}|^{\tilde{q}}\,dx\right)^{2-p}
+c\left(\frac{|\mu|(B_{2r})}{r^{n-1}}\right)^{\tilde{q}}
\left(\fint_{B_{2r}}|{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}}\,dx\right)^{2-p}\\
&\le
{\frac12} \fint_{B_{2r}}|{\mathbb{A}(Du)}-{\mathbb{A}(Dv)}|^{\tilde{q}}\,dx
+ c\left(\frac{|\mu|(B_{2r})}{r^{n-1}}\right)^\frac{\tilde{q}}{p-1}
+c\left(\frac{|\mu|(B_{2r})}{r^{n-1}}\right)^{\tilde{q}}
\left(\fint_{B_{2r}}|{\mathbb{A}(Du)}|^{\tilde{q}}\,dx\right)^{2-p},
\end{aligned}
\end{equation*}
where $c=c(n,p, \Lambda)>0$. Here in the first inequality, we also have to use the Lemma~\ref{lem:logBMO}.

By absorbing the corresponding term into the left-hand side, we obtain the desired estimate \eqref{est.lem.4.4} in case $2-{\frac1n}<p\le 2$.
This completes the proof of Lemma~\ref{comparison estimate}.
