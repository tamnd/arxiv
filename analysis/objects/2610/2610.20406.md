<!-- problem-1: body_md -->
\label{problem:minimal_solomon}
Is every foliation of $\mathbb{R}^{n+1}$, $n\leq 7$, by properly embedded minimal hypersurfaces a foliation by parallel affine hyperplanes?
<!-- end problem-1 -->
<!-- conjecture-1: body_md -->
\label{conjecture:cmc_solomon}
Every foliation of $\mathbb{R}^{n+1}$, $n\leq 7$, by (possibly varying) CMC hypersurfaces is a foliation by parallel affine hyperplanes.
<!-- end conjecture-1 -->
<!-- conjecture-2: body_md -->
\label{conjecture:yau-docarmo-solomon}
Every foliation of $\mathbb{R}^{n+1}$ by (possibly varying) CMC hypersurfaces is a foliation by minimal hypersurfaces.
<!-- end conjecture-2 -->
<!-- customthm-1: body_md -->
{\ref*{thm:main_euclidean}}
Let $\mathfrak{F}$ be a codimension one CMC foliation of $\mathbb{R}^{n+1}$. Then, each leaf of $\mathfrak{F}$ is a minimal and stable hypersurface.
<!-- end customthm-1 -->
<!-- customcor-1: body_md -->
{\ref*{thm:main_euclidean_low_dim}}
Let $\mathfrak{F}$ be a codimension one CMC foliation of $\mathbb{R}^{n+1}$. If $n\leq 6$, then $\mathfrak{F}$ consists entirely of parallel hyperplanes.
<!-- end customcor-1 -->
<!-- lemma-1: body_md -->
\label{lem:gradH}
$\nabla H=gN$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $\mathfrak{F}$ is a CMC foliation, $H$ is constant along each leaf. Hence, \(X(H)=\langle\nabla H,X\rangle=0\) for every vector field \(X\) tangent to the leaves. Therefore, \(\nabla H\) is orthogonal to the leaves and thus parallel to \(N\). Since \(\langle\nabla H,N\rangle=N(H)=g\) and \(|N|=1\), we conclude that \(\nabla H=gN\).
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:gT}
For every vector field $X$ tangent to a leaf $L$ of $\mathfrak F$, we have
\[
X(g)=g\langle T,X\rangle.
\]
Equivalently, the gradient of $g$ along each leaf satisfies
\[
\nabla_L g=gT.
\]
Consequently, $g$ has constant sign on each leaf $L$: either
$g\equiv0$ on $L$, or $g>0$ everywhere on $L$, or $g<0$
everywhere on $L$.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:divT}
$\operatorname{div} T=|A|^2-ng$.
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
\label{lem:divX}
Let $X$ be a $C^1$ vector field on $\mathbb{R}^{n+1}$ tangent to the leaves. Then, along every leaf $L$,
\begin{equation}\label{eq:divX}
\operatorname{div} X=\operatorname{div}_LX-\langle X,T\rangle .
\end{equation}
<!-- end lemma-4 -->
<!-- lemma-5: body_md -->
\label{lem:lapL}
For $f\in C^\infty(\mathbb R^{n+1})$ and any leaf $L\in \mathfrak{F}$, we have
\begin{equation}\label{eq:lapL}
\Delta_L f = \operatorname{tr}_{TL}\operatorname{Hess} f
+nH\,N(f) \qquad\text{on }L.
\end{equation}
In particular, for $\rho(x):=\sqrt{1+|x|^2}$, one has $\Delta_L\rho\le n(1+|H|).$
<!-- end lemma-5 -->
<!-- proof-2: body_md -->
Let $\{e_1,\ldots,e_n\}$ be a local orthonormal frame of $TL$.
By the definition of the Laplace--Beltrami operator on $L$,
$$
\Delta_L f
=
\sum_{i=1}^n
\left(
e_i(e_i f)-(\nabla^L_{e_i}e_i)(f)
\right),
$$
where $\nabla^L$ denotes the Levi-Civita connection of the induced
metric on $L$. On the other hand, the ambient Hessian satisfies
$$
\operatorname{Hess}f(e_i,e_i)=e_i(e_i f)-(\nabla_{e_i}e_i)(f),
$$
where $\nabla$ is the Euclidean connection of $\mathbb R^{n+1}$. Since $\nabla_{e_i}e_i$ decomposes into its tangential and normal
components, we have
$$
\nabla_{e_i}e_i=\nabla^L_{e_i}e_i+\langle\nabla_{e_i}e_i,N\rangle N.
$$
Differentiating $\langle e_i,N\rangle=0$ along $e_i$, we obtain
$$
\langle\nabla_{e_i}e_i,N\rangle=-\langle e_i,\nabla_{e_i}N\rangle=\langle Ae_i,e_i\rangle,
$$
where $A(X)=-\nabla_XN$ is the shape operator. Consequently,
$$
\nabla_{e_i}e_i=\nabla^L_{e_i}e_i+\langle Ae_i,e_i\rangle N.
$$
It follows that
\begin{equation}
e_i(e_i f)-(\nabla^L_{e_i}e_i)(f)=\operatorname{Hess}f(e_i,e_i)+\langle Ae_i,e_i\rangle N(f).
\end{equation}
Summing over $i$, we get

\begin{eqnarray*}
\Delta_L f & = & \sum_{i=1}^n\operatorname{Hess}f(e_i,e_i)+\left(\sum_{i=1}^n\langle Ae_i,e_i\rangle\right)N(f)\\
& = & \operatorname{tr}_{TL}\operatorname{Hess}f
+nH\,N(f),    
\end{eqnarray*}
where we used the equation $\operatorname{tr}(A|_{TL})=nH$ (see Equation \eqref{eqtraco}). This proves the Equation \eqref{eq:lapL}.


We now compute the gradient and the Hessian of
$$
\rho(x):=\sqrt{1+|x|^2},
\qquad x=(x_1,\ldots,x_{n+1})\in\mathbb R^{n+1}.
$$

By the chain rule, for every $i=1,\ldots,n+1$, we have $\frac{\partial\rho}{\partial x_i} =\frac{x_i}{\rho}$, and thus
$$
\nabla\rho=\frac{x}{\rho}.
$$
Moreover,
\[
\frac{\partial^2\rho}{\partial x_i\partial x_j}
=
\frac{\delta_{ij}}{\rho}
-\frac{x_ix_j}{\rho^3}.
\]
Hence, the Hessian matrix is
\[
\operatorname{Hess}\rho
=
\frac1\rho I-\frac1{\rho^3}x\otimes x,
\]
where $I$ denotes the identity operator and $x\otimes x$ is the linear
operator defined by
\[
(x\otimes x)(v)=\langle x,v\rangle x,
\qquad v\in\mathbb R^{n+1}.
\]
Equivalently, for any $v,w\in\mathbb R^{n+1}$,
\[
\operatorname{Hess}\rho(v,w)
=
\frac{\langle v,w\rangle}{\rho}
-
\frac{\langle x,v\rangle\langle x,w\rangle}{\rho^3}.
\]
In particular,
\[
\operatorname{Hess}\rho(v,v)
=
\frac{|v|^2}{\rho}
-
\frac{\langle x,v\rangle^2}{\rho^3}.
\]
Therefore,
\[
\begin{aligned}
\operatorname{Hess}\rho(v,v)
&\ge
\left(\frac{1}{\rho}-\frac{|x|^2}{\rho^3}\right)|v|^2\\
&=\frac{\rho^2-|x|^2}{\rho^3}|v|^2\\
&=\frac{1}{\rho^3}|v|^2\ge0.
\end{aligned}
\]
Moreover, since $\rho\ge1$, we have
\[
\operatorname{Hess}\rho(v,v)
\le \frac{|v|^2}{\rho}
\le |v|^2.
\]
Hence, in the sense of quadratic forms,
\[
0\le\operatorname{Hess}\rho\le I.
\]
Taking the trace over an orthonormal basis
$\{e_1,\ldots,e_n\}$ of $TL$, we obtain
\[
\operatorname{tr}_{TL}\operatorname{Hess}\rho
=
\sum_{i=1}^n\operatorname{Hess}\rho(e_i,e_i)
\le
\sum_{i=1}^n|e_i|^2

\label{lem:hessian}
Let $\chi\in C^{\infty}(\mathbb{R}^{n+1})$ and let $q\geq 4$ be an
integer. Then
\[
\operatorname{Hess}(\chi^q)
=
q(q-1)\chi^{q-2}\,d\chi\otimes d\chi
+
q\chi^{q-1}\operatorname{Hess}\chi.
\]
Here, the Hessian is regarded as a symmetric bilinear form, and
$d\chi\otimes d\chi$ denotes the bilinear form defined by
\[
(d\chi\otimes d\chi)(v,w)=d\chi(v)d\chi(w).
\]
<!-- end lemma-6 -->

Let $v,w\in\mathbb{R}^{n+1}$ be arbitrary vectors. By the
definition of the Hessian,
\[
\operatorname{Hess}(\chi^q)(v,w)
=
\left\langle
D_v\bigl(\nabla(\chi^q)\bigr),w
\right\rangle.
\]
By the chain rule,
\[
\nabla(\chi^q)=q\chi^{q-1}\nabla\chi.
\]
Differentiating this identity in the direction of $v$ and applying
the product rule, we obtain
\[
D_v\bigl(\nabla(\chi^q)\bigr)
=
q(q-1)\chi^{q-2}d\chi(v)\nabla\chi
+
q\chi^{q-1}D_v(\nabla\chi).
\]
Taking the inner product with $w$ gives
\begin{align*}
\operatorname{Hess}(\chi^q)(v,w)
={}&q(q-1)\chi^{q-2}
d\chi(v)\langle\nabla\chi,w\rangle
+q\chi^{q-1}
\langle D_v(\nabla\chi),w\rangle.
\end{align*}
Since
\[
d\chi(w)=\langle\nabla\chi,w\rangle
\]
and
\[
\operatorname{Hess}\chi(v,w)
=
\langle D_v(\nabla\chi),w\rangle,
\]
we conclude that
\[
\begin{aligned}
\operatorname{Hess}(\chi^q)(v,w)
={}&q(q-1)\chi^{q-2}d\chi(v)d\chi(w)
+q\chi^{q-1}\operatorname{Hess}\chi(v,w).
\end{aligned}
\]
By the definition of $d\chi\otimes d\chi$, this is equivalent to
\[
\operatorname{Hess}(\chi^q)
=
q(q-1)\chi^{q-2}\,d\chi\otimes d\chi
+
q\chi^{q-1}\operatorname{Hess}\chi.

Since $v$ and $w$ were arbitrary, the identity holds everywhere

\label{lem:divV}
The vector field

V:=H^pN+\frac{p}{n}H^{p-1}T
\end{equation}

\operatorname{div}V
=
\frac{p}{n}H^{p-1}|A|^2-nH^{p+1}

Since $H$, $N$, and $T$ are smooth and $p$ is an integer, the
vector field $V$ defined in \eqref{eq:V} is smooth on
$\mathbb R^{n+1}$. We first compute the divergence of each term in $V$. By the
product rule for the divergence,
\[
\operatorname{div}(H^pN)
=
\langle\nabla(H^p),N\rangle
+H^p\operatorname{div}N.
\]
Using the chain rule and the identities 
$\nabla H=gN$ (see Lemma \ref{lem:gradH}) and $\operatorname{div}N=-nH$, we obtain
\[
\begin{aligned}
\operatorname{div}(H^pN)
&=pH^{p-1}\langle\nabla H,N\rangle
-nH^{p+1}\\
&=pH^{p-1}g\langle N,N\rangle
-nH^{p+1}\\
&=pH^{p-1}g-nH^{p+1},
\end{aligned}
\]
where we used $|N|=1$.

Next, applying the product rule to the second term gives
\[
\operatorname{div}(H^{p-1}T)
=
\langle\nabla(H^{p-1}),T\rangle
+H^{p-1}\operatorname{div}T.
\]
Since
\[
\nabla(H^{p-1})
=(p-1)H^{p-2}\nabla H
=(p-1)H^{p-2}gN
\]
and $T\perp N$, we have
\[
\langle\nabla(H^{p-1}),T\rangle
=(p-1)H^{p-2}g\langle N,T\rangle=0.
\]
Therefore, by Lemma~\ref{lem:divT},
\[
\begin{aligned}
\operatorname{div}\left(\frac{p}{n}H^{p-1}T\right)
&=\frac{p}{n}H^{p-1}\operatorname{div}T\\
&=\frac{p}{n}H^{p-1}(|A|^2-ng)\\
&=\frac{p}{n}H^{p-1}|A|^2
-pH^{p-1}g.
\end{aligned}
\]

Adding the two divergence identities, we find that
$$
\operatorname{div}V=pH^{p-1}g-nH^{p+1}+\frac{p}{n}H^{p-1}|A|^2-pH^{p-1}g=\frac{p}{n}H^{p-1}|A|^2-nH^{p+1}.
$$
Thus, the terms involving $g$ cancel identically, and the
equality in \eqref{eq:divV} follows without any assumption on
the signs of $g$ or $H$.

It remains to establish the inequality. By the
Cauchy--Schwarz inequality,
\[
\bigl(\operatorname{tr}(A|_{TL})\bigr)^2
\le n\,|A|^2.
\]
Since $\operatorname{tr}(A|_{TL})=nH$, this yields
\begin{equation}\label{eq:CS}
|A|^2\ge nH^2.
\end{equation}
Because $p$ is odd, $p-1$ is even, and hence
$H^{p-1}\ge0$. Multiplying \eqref{eq:CS} by
$\frac{p}{n}H^{p-1}\ge0$, we obtain
\[
\frac{p}{n}H^{p-1}|A|^2
\ge pH^{p+1}.
\]
Consequently,
\[
\begin{aligned}
\operatorname{div}V
&=\frac{p}{n}H^{p-1}|A|^2-nH^{p+1}\\
&\ge pH^{p+1}-nH^{p+1}\\
&=(p-n)H^{p+1}.
\end{aligned}
\]
Finally, since $p+1$ is even,
\[
H^{p+1}=|H|^{p+1}.
\]
Moreover, $p>n$ implies $p-n>0$. Therefore,
\[
\operatorname{div}V
\ge(p-n)|H|^{p+1}\ge0,
\]
which completes the proof.

\[
\nabla_L\chi=\nabla\chi-N(\chi)N

is the orthogonal projection of $\nabla\chi$ onto the tangent

$
\langle\nabla_L\chi,N\rangle=0.

=
\langle\nabla_L\chi,T\rangle.

On each leaf $L$, the restriction of $\nabla_L\chi$ is the

\[
\operatorname{div}_L(\nabla_L\chi)=\Delta_L\chi.
\]

\[
\operatorname{div}(\nabla_L\chi)
=
\operatorname{div}_L(\nabla_L\chi)

\]
Equivalently,
\[
\langle\nabla_L\chi,T\rangle
=
\Delta_L\chi-\operatorname{div}(\nabla_L\chi).
\]
Consequently,
\[
\langle\nabla\chi,T\rangle
=
\Delta_L\chi-\operatorname{div}(\nabla_L\chi).

\Delta_L\chi
=
\operatorname{tr}_{TL}\operatorname{Hess}\chi
+nH\,N(\chi).

Substituting this identity, we conclude that
\[
\begin{aligned}

={}&\operatorname{tr}_{TL}\operatorname{Hess}\chi
+nH\,N(\chi)
-\operatorname{div}(\nabla_L\chi).
\end{aligned}

\[
\begin{aligned}

+nH\kappa\,N(\chi)-\kappa\operatorname{div}(\nabla_L\chi).
\end{aligned}

It remains to rewrite the last term. By the product rule

\[
\operatorname{div}(\kappa\nabla_L\chi)
=
\langle\nabla\kappa,\nabla_L\chi\rangle
+\kappa\operatorname{div}(\nabla_L\chi).
\]
Since $\kappa$ is constant along the leaves, its derivative
in every tangential direction vanishes. Thus,
\[
\langle\nabla\kappa,X\rangle=0
\qquad\text{for every }X\in TL.

Consequently,

\nabla\chi_R

\[
(u\otimes v)(w):=\langle v,w\rangle u.
\]
In particular,
\[
(\widehat{x}\otimes\widehat{x})(v)
=
\langle\widehat{x},v\rangle\widehat{x}.

Since $|\widehat{x}|=1$, the operator

\nabla\chi_R=a(r)\widehat{x}.

\operatorname{Hess}\chi_R(v)=D(\nabla\chi_R)(v) & = & Da(v)\,\widehat{x}+a(r)D\widehat{x}(v)\\
& = & a'(r)Dr(v)\,\widehat{x}+a(r)D\widehat{x}(v).

\[
a'(r)=\frac{1}{R^2}\chi_0''\left(\frac{r}{R}\right).

\operatorname{Hess}\chi_R=\alpha P+\beta Q.

The operators $P$ and $Q$ are orthogonal projections onto
complementary orthogonal subspaces. In particular, for every
$v\in\mathbb R^{n+1}$,
\[
v=Pv+Qv,\qquad

\]
and therefore
\[

It follows that
\[
\begin{aligned}
|\operatorname{Hess}\chi_R(v)|^2
&=|\alpha Pv+\beta Qv|^2\\
&=\alpha^2|Pv|^2+\beta^2|Qv|^2\\
&\le
\max\{|\alpha|^2,|\beta|^2\}
\left(|Pv|^2+|Qv|^2\right)\\
&=
\max\{|\alpha|^2,|\beta|^2\}|v|^2.
\end{aligned}
\]
Taking the supremum over all unit vectors $v$, we obtain
\[
\|\operatorname{Hess}\chi_R\|
\le\max\{|\alpha|,|\beta|\}.

Equality holds because the radial subspace and its orthogonal
complement are nontrivial eigenspaces of the Hessian. Hence,
\[
\|\operatorname{Hess}\chi_R\|
=

\frac{|\chi_0'(r/R)|}{Rr}
\right\}.

$\operatorname{Hess}\chi_R$ vanish outside the annulus $R\le r\le2R.$ On this annulus, $r\ge R$, so

\[
\begin{aligned}
\|\operatorname{Hess}\chi_R\|
&\le
\frac{1}{R^2}
\max\left\{
\sup_{s\ge0}|\chi_0''(s)|,
\sup_{s\ge0}|\chi_0'(s)|
\right\}\\
&\le\frac{C_0}{R^2}.
\end{aligned}
\]
This proves the second estimate in \eqref{eq:cutoff}, and
completes the proof of both claimed bounds.

a:=|H|\chi\geq 0.
\]
Recall that $q=p+1$, where $p>n$ is an odd integer. By the estimates in
\eqref{eq:cutoff}, we have
\[
|\nabla\chi|\leq \frac{C_0}{R},
\qquad
\|\operatorname{Hess}\chi\|\leq \frac{C_0}{R^2}.
\]
Since $0\leq\chi\leq 1$, the chain rule yields
\[
\nabla(\chi^q)=q\chi^{q-1}\nabla\chi
=q\chi^p\nabla\chi,
\]
and hence
\[
|\nabla(\chi^q)|
=q\chi^p|\nabla\chi|
\leq \frac{qC_0}{R}\chi^p.
\]

By Lemma \ref{lem:hessian}, we have
\[
\operatorname{Hess}(\chi^q)
=
q(q-1)\chi^{q-2}\,d\chi\otimes d\chi
+
q\chi^{q-1}\operatorname{Hess}\chi.
\]


Taking operator norms and using $q-1=p$ and $q-2=p-1$, we obtain
\begin{align*}
\|\operatorname{Hess}(\chi^q)\|
&\leq q(q-1)\chi^{p-1}|\nabla\chi|^2
+q\chi^p\|\operatorname{Hess}\chi\|\\
&\leq \frac{q(q-1)C_0^2}{R^2}\chi^{p-1}
+\frac{qC_0}{R^2}\chi^p\\
&\leq \frac{C_1}{R^2}\chi^{p-1},
\end{align*}
where $C_1:=q(q-1)C_0^2+qC_0.$ In the last inequality, we used $\chi^p\leq\chi^{p-1}$. Since $p$ is odd, both $p+1$ and $p-1$ are even. Therefore,
\[
|H|^p\chi^p=a^p,
\qquad
H^{p-1}\chi^{p-1}=a^{p-1},
\qquad
\chi^{p+1}H^{p+1}=a^{p+1}.
\]
Moreover, $\chi$ vanishes outside $\overline{B_{2R}}$, and thus
$a$ also vanishes there. Set
\[
I=I(R):=\int_{\mathbb{R}^{n+1}}\chi^{p+1}H^{p+1}
=\int_{\mathbb{R}^{n+1}}a^{p+1}.
\]
In particular, $I<\infty$. Define the constants
\[
A_1:=(p+1)qC_0,
\qquad
A_2:=pC_1.
\]
Applying Proposition~\ref{prop:int} and the preceding estimates, we obtain
\begin{align*}
(p-n)I
&\leq (p+1)\int_{\mathbb{R}^{n+1}}
|H|^p|\nabla(\chi^q)|
+p\int_{\mathbb{R}^{n+1}}
H^{p-1}\|\operatorname{Hess}(\chi^q)\|\\
&\leq \frac{A_1}{R}\int_{B_{2R}}a^p
+\frac{A_2}{R^2}\int_{B_{2R}}a^{p-1}.
\end{align*}

Let
\[
\mathcal V:=|B_{2R}|=\omega_{n+1}(2R)^{n+1},
\]
where $\omega_{n+1}$ denotes the volume of the unit ball in
$\mathbb{R}^{n+1}$. Since $a\geq0$ and
\[
\int_{B_{2R}}a^{p+1}
\leq \int_{\mathbb{R}^{n+1}}a^{p+1}=I,
\]
H\"older's inequality gives
\begin{align*}
\int_{B_{2R}}a^p
&\leq
\left(\int_{B_{2R}}a^{p+1}\right)^{\frac{p}{p+1}}
|B_{2R}|^{\frac{1}{p+1}}\\
&\leq I^{\frac{p}{p+1}}\mathcal V^{\frac{1}{p+1}},
\end{align*}
and, in the same way,
\begin{align*}
\int_{B_{2R}}a^{p-1}
&\leq
\left(\int_{B_{2R}}a^{p+1}\right)^{\frac{p-1}{p+1}}
|B_{2R}|^{\frac{2}{p+1}}\\
&\leq I^{\frac{p-1}{p+1}}\mathcal V^{\frac{2}{p+1}}.
\end{align*}
Now put
\[
X:=I^{\frac{1}{p+1}},
\qquad
Y:=\frac{\mathcal V^{\frac{1}{p+1}}}{R}.
\]
Both $X$ and $Y$ are nonnegative, and $Y>0$. Substituting the
H\"older estimates into the preceding integral inequality yields
\[
(p-n)X^{p+1}
\leq A_1X^pY+A_2X^{p-1}Y^2.
\]

If $X=0$, then $X\leq\Lambda Y$ holds trivially for every
$\Lambda>0$. Suppose, therefore, that $X>0$. Dividing the last
inequality by $X^{p-1}Y^2>0$, we obtain
\[
(p-n)\left(\frac{X}{Y}\right)^2
-A_1\frac{X}{Y}-A_2\leq0.
\]
Writing $s:=X/Y$, this becomes
\[
(p-n)s^2-A_1s-A_2\leq0.
\]
Since $p-n>0$ and $s\geq0$, the quadratic inequality implies
\[
s\leq
\frac{A_1+\sqrt{A_1^2+4(p-n)A_2}}{2(p-n)}
=:\Lambda.
\]
Consequently, in both cases $X=0$ and $X>0$, we have
\[
X\leq\Lambda Y.
\]
Raising this inequality to the power $p+1$ and recalling the
definitions of $X$ and $Y$, we conclude that
\[
I\leq\Lambda^{p+1}\frac{\mathcal V}{R^{p+1}}.
\]
Since $\mathcal V=\omega_{n+1}(2R)^{n+1}$, it follows that
\[
I(R)\leq
\Lambda^{p+1}\omega_{n+1}2^{n+1}R^{n-p}.
\]
The constant $\Lambda$ depends only on $n$, $p$, and the chosen
cutoff function $\chi_0$. Moreover, since $\chi=1$ on $B_R$ and
$H^{p+1}=|H|^{p+1}\geq0$, we have
\begin{equation*}
\int_{B_R}{|H|^{p+1}}\leq I(R)\leq
\Lambda^{p+1}\omega_{n+1}2^{n+1}R^{n-p}.
\label{eq:decay}
\end{equation*}

We now show that $H$ vanishes identically. Fix $R_0>0$. For every
$R\geq R_0$, the inclusion $B_{R_0}\subset B_R$ and the
nonnegativity of $|H|^{p+1}$ imply
\[
0\leq\int_{B_{R_0}}|H|^{p+1}
\leq\int_{B_R}|H|^{p+1}
\leq
\Lambda^{p+1}\omega_{n+1}2^{n+1}R^{n-p}.
\]
Because $p>n$, we have $n-p<0$, and therefore
\[
\lim_{R\to\infty}R^{n-p}=0.
\]
Letting $R\to\infty$ in the preceding inequality gives
\[
\int_{B_{R_0}}|H|^{p+1}=0.
\]
The function $|H|^{p+1}$ is continuous and nonnegative. Hence it
must vanish at every point of $B_{R_0}$; otherwise, continuity
would give a neighborhood on which it is strictly positive,
contradicting the vanishing of its integral. Thus $H=0$ on $B_{R_0}$. Since $R_0>0$ was arbitrary, we conclude that $H\equiv0$ on $\mathbb{R}^{n+1}.$ Therefore, every leaf of $\mathfrak F$ has vanishing mean curvature and is consequently a minimal hypersurface. Finally, stability follows from Theorem 1 of \cite{Oshikiri-1982}.
