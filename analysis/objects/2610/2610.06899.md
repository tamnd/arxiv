<!-- theorem-1: body_md -->
\label{thm:HJT}
If $G$ is a graph with domination number $\gamma(G)$, then $m_G[0,1)\le \gamma(G)$.
<!-- end theorem-1 -->
<!-- problem-1: body_md -->
Characterize all trees $T$ satisfying the equality $m_T[0,1)=\lceil(d(T)+1)/3\rceil$.
<!-- end problem-1 -->
<!-- conjecture-1: body_md -->
With the notation above,
$$\lim_{n\to\infty}\frac{|\mathcal T_n^*|}{|\mathcal T_n|}=0.$$
<!-- end conjecture-1 -->
<!-- theorem-2: body_md -->
\label{thm:classification-intro}
Let $T$ be a tree of diameter $d$.
Then $m_T[0,1)=\lceil(d+1)/3\rceil$ if and only if $T$ has a diametral path $P=v_0v_1\cdots v_d$ such that $T$ belongs to one of the following two classes.

\medskip

\noindent
\textbf{(i) Caterpillars.}
There is a minimum dominating set $D$ of the path $P$ such that $T$ is obtained from $P$ by adding any number of pendant vertices, possibly zero, only at internal vertices $v_i\in D$, and by attaching no other vertices.

\medskip

\noindent
\textbf{(ii) Trees with one attached star.}
Here $d=3k$ for some $k\ge2$.
There is an integer $r$ with $1\le r\le k-1$ and an integer $t\ge1$ such that $T$ is obtained as follows.
First take the path $P=v_0v_1\cdots v_{3k}$.
Take a star $K_{1,t}$ with center $u$ and add the edge $uv_{3r}$.
In addition, add any number of pendant vertices, possibly zero, only at vertices in
$ \{v_1,v_4,\ldots,v_{3r-2}\} \cup \{v_{3r+2},v_{3r+5},\ldots,v_{3k-1}\},$
and attach no other vertices.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:asymptotic-intro}
Let $\mathcal T_n$ be the set of unlabeled trees of order $n$, and let
$$ \mathcal T_n^* = \left\{ T\in\mathcal T_n: m_T[0,1)=\left\lceil\frac{d(T)+1}{3}\right\rceil \right\}.$$
Then $\lim_{n\to\infty}\frac{|\mathcal T_n^*|}{|\mathcal T_n|}=0$.
Equivalently, almost all trees satisfy $m_T[0,1)\ge \lceil(d(T)+1)/3\rceil+1$.
<!-- end theorem-3 -->
<!-- lemma-1: body_md -->
\label{lem:dom-lower}
If $T$ is a tree of diameter $d$, then $\gamma(T)\ge \lceil(d+1)/3\rceil$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:contraction}
Let $T$ be a tree on $n$ vertices with Laplacian eigenvalues $\mu_1(T)\le\cdots\le\mu_n(T)$, and let $T'$ be the tree obtained from $T$ by contracting an edge.
Then $T'$ has $n-1$ vertices and
$$ \mu_i(T)\le\mu_i(T')\qquad\text{for }1\le i\le n-1. $$
<!-- end lemma-2 -->
<!-- corollary-1: body_md -->
\label{cor:contraction}
Let $T$ be a tree, and suppose that $V(T)=U_1\cup U_2\cup\cdots\cup U_r$ is a partition into nonempty connected subsets.
Let $T_0$ be the graph obtained by contracting each $U_i$ to a single vertex.
Then $T_0$ is a tree and $m_T[0,1)\ge m_{T_0}[0,1)$.
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
Since $T$ is connected and each $U_i$ is connected, $T_0$ is connected.
Moreover, two distinct blocks cannot be joined by two different edges of $T$, because those two edges together with the unique paths inside the two blocks would form a cycle.
Similarly, any cycle in $T_0$ would lift to a cycle in $T$.
Hence $T_0$ is a tree.
Each $U_i$ is connected, so it induces a subtree of $T$; contracting the edges of these subtrees one by one produces $T_0$ from $T$.
By Lemma~\ref{lem:contraction}, contracting an edge never decreases any Laplacian eigenvalue, so it cannot increase the number of Laplacian eigenvalues strictly less than $1$.
Therefore $m_{T_0}[0,1)\le m_T[0,1)$.
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
\label{lem:spectral-lower}
If $T$ is a tree of diameter $d$, then $m_T[0,1)\ge \lceil(d+1)/3\rceil$.
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
\label{lem:joining-eigenvalue-one}
Let $T_1, T_2$ be two trees and $v_1 \in T_1, v_2 \in T_2$. Form a new tree $T$ by joining the vertices $v_1$, $v_2$ by an edge. Then the following hold:
\begin{enumerate}
\item[(1)] If $v_1$ is a nonzero vertex of $T_1$, then
\[
m_T(1) = m_{T_1}(1) + m_{T_2}(1) - 1.
\]
\item[(2)] If $v_1$ is not a nonzero vertex of $T_1$ and $v_2$ is not a nonzero vertex of $T_2$, then
\[
m_{T_1}(1) + m_{T_2}(1) \le m_T(1) \le m_{T_1}(1) + m_{T_2}(1) + 1.
\]
\end{enumerate}
<!-- end lemma-4 -->
<!-- lemma-5: body_md -->
\label{lem:cut-additivity}
Let $T_1$ and $T_2$ be two trees, and let $T$ be obtained by adding an edge $uv$, where $u\in V(T_1)$ and $v\in V(T_2)$.
If $u$ is a nonzero vertex of $T_1$, then
$$ m_T[0,1)=m_{T_1}[0,1)+m_{T_2}[0,1). $$
<!-- end lemma-5 -->
<!-- proof-2: body_md -->
Let $T^\ast=T_1\cup T_2$.
Since $u$ is a nonzero vertex of $T_1$, Lemma~\ref{lem:joining-eigenvalue-one} gives
$$ m_T(1)=m_{T_1}(1)+m_{T_2}(1)-1=m_{T^\ast}(1)-1. $$

Now $T^\ast$ has exactly $|V(T^\ast)|-m_{T^\ast}[0,1)-m_{T^\ast}(1)$ Laplacian eigenvalues greater than $1$.
If $b=e_u-e_v$ is the incidence vector of the added edge, then
$$ L(T)=L(T^\ast)+bb^{\top}. $$
The rank-one positive semidefinite interlacing inequalities imply
$$ m_{T^\ast}[0,1)-1\le m_T[0,1)\le m_{T^\ast}[0,1), $$
and they also imply that $T$ has at most $|V(T^\ast)|-m_{T^\ast}[0,1)-m_{T^\ast}(1)+1$ Laplacian eigenvalues greater than $1$.
If $m_T[0,1)=m_{T^\ast}[0,1)-1$, then $m_T(1)=m_{T^\ast}(1)-1$ would force $T$ to have
$$ |V(T)|-\bigl(m_{T^\ast}[0,1)-1\bigr)-\bigl(m_{T^\ast}(1)-1\bigr)=|V(T^\ast)|-m_{T^\ast}[0,1)-m_{T^\ast}(1)+2 $$
Laplacian eigenvalues greater than $1$, a contradiction.
Hence $m_T[0,1)=m_{T^\ast}[0,1)$.
Finally, the Laplacian spectrum of the disjoint union $T^\ast=T_1\cup T_2$ is the multiset union of the spectra of $T_1$ and $T_2$, and thus
$$ m_{T^\ast}[0,1)=m_{T_1}[0,1)+m_{T_2}[0,1). $$
Therefore $m_T[0,1)=m_{T_1}[0,1)+m_{T_2}[0,1)$.
This proves the result.
<!-- end proof-2 -->
<!-- lemma-6: body_md -->
\label{lem:depth-three}
Let $T$ be a tree of diameter $d$, and let $P=v_0v_1\cdots v_d$ be a diametral path.
If some vertex of $T$ has distance at least $3$ from $P$, then $m_T[0,1) \ge \lceil(d+1)/3\rceil+1$.
<!-- end lemma-6 -->
<!-- proof-3: body_md -->
Choose a vertex $t$ whose distance to $P$ is at least $3$, and let $v_j$ be the vertex of $P$ closest to $t$.
Let $v_j-w_1-w_2-w_3$ be the first three edges of the unique path from $v_j$ to $t$, where $w_1,w_2,w_3\notin V(P)$.

Delete the edges of $P$ and also delete $v_jw_1$, $w_1w_2$, and $w_2w_3$.
Let $U_i$ be the component containing $v_i$, and let $W_1,W_2,W_3$ be the components containing $w_1,w_2,w_3$, respectively.
These $d+4$ sets are precisely the components of the resulting forest.
Contracting all these components gives a tree $T^*$ obtained from $P_{d+1}$ by attaching a pendant path of length $3$ at $v_j$.
By Corollary~\ref{cor:contraction}, $m_T[0,1)\ge m_{T^*}[0,1)$.

Let $P_3=w_1w_2w_3$.
The vector $z$ defined by
$$ z(w_1)=1,\qquad z(w_2)=0,\qquad z(w_3)=-1 $$
satisfies $L(P_3)z=z$.
Hence $w_1$ is a nonzero vertex of $P_3$ for the eigenvalue $1$.
The Laplacian spectrum of $P_3$ is $\{0,1,3\}$, and therefore $m_{P_3}[0,1)=1$.

The tree $T^*$ is obtained from the vertex-disjoint trees $P_3$ and $P_{d+1}$ by adding the edge $w_1v_j$.
Lemma~\ref{lem:cut-additivity} gives
$$ m_{T^*}[0,1)=m_{P_3}[0,1)+m_{P_{d+1}}[0,1)=m_{P_{d+1}}[0,1)+1. $$
Since $P_{d+1}$ has diameter $d$, Lemma~\ref{lem:spectral-lower} yields $m_{P_{d+1}}[0,1)\ge \lceil(d+1)/3\rceil$.
Therefore
$$ m_T[0,1)\ge m_{T^*}[0,1)=m_{P_{d+1}}[0,1)+1 \ge \left\lceil\frac{d+1}{3}\right\rceil+1,$$
as required.
<!-- end proof-3 -->
<!-- corollary-2: body_md -->
\label{cor:distance-two}
If $T$ is a tree of diameter $d$ satisfying $m_T[0,1)=\lceil(d+1)/3\rceil$, then $T$ has a diametral path $P$ such that every vertex of $T$ has distance at most $2$ from $P$.
<!-- end corollary-2 -->
<!-- proof-4: body_md -->
Let $P$ be a diametral path.
If some vertex had distance at least $3$ from $P$, Lemma~\ref{lem:depth-three} would give $m_T[0,1) \ge \lceil(d+1)/3\rceil+1$, contrary to equality.
<!-- end proof-4 -->
<!-- lemma-7: body_md -->
\label{lem:pendant-monotonicity}
Let $G'$ be obtained from a graph $G$ by deleting a pendant vertex.
Then $m_G[0,1)\ge m_{G'}[0,1)$.
<!-- end lemma-7 -->
<!-- lemma-8: body_md -->
\label{lem:path-count}
For every $n\ge1$,
$$ m_{P_n}[0,1)=\left\lceil\frac{n}{3}\right\rceil. $$
If $n\equiv0\pmod3$, both end vertices of $P_n$ are nonzero vertices for the eigenvalue $1$.
<!-- end lemma-8 -->
<!-- proof-5: body_md -->
The Laplacian eigenvalues of $P_n$ are $2-2\cos(j\pi/n)$, where $j=0,1,\ldots,n-1$.
Such an eigenvalue is less than $1$ precisely when $j<n/3$, which gives the stated count.

Now assume $n\equiv0\pmod3$ and write the path as $v_0v_1\cdots v_{n-1}$.
Starting at $v_0$, assign the values $1,0$ and propagate by $z(v_{i+1})=z(v_i)-z(v_{i-1})$.
The values repeat with pattern $1,0,-1,-1,0,1$.
Since $n\equiv0\pmod3$, the equation at the opposite end is also satisfied.
Thus $L(P_n)z=z$ and $z(v_0)\ne0$.
The other end follows by symmetry.
<!-- end proof-5 -->
<!-- lemma-9: body_md -->
\label{lem:caterpillar-equality}
Let $T$ be obtained from a diametral path $P$ of length $d$ by adding pendant vertices only at vertices belonging to a minimum dominating set of $P$.
Then
$$ m_T[0,1)=\left\lceil\frac{d+1}{3}\right\rceil. $$
<!-- end lemma-9 -->
<!-- proof-6: body_md -->
The specified minimum dominating set of $P$ also dominates $T$.
Theorem~\ref{thm:HJT}, Lemma~\ref{lem:dom-lower} and Lemma~\ref{lem:spectral-lower} give
$$ \left\lceil\frac{d+1}{3}\right\rceil\le m_T[0,1)\le\gamma(T)\le\left\lceil\frac{d+1}{3}\right\rceil. $$
<!-- end proof-6 -->
<!-- lemma-10: body_md -->
\label{lem:nonzero-tail}
Let $T$ be obtained from a path $P=v_0v_1\cdots v_d$ by adding pendant vertices only to vertices of $P$.
Suppose that $v_p$ is the vertex of $P$ with the largest index having a pendant neighbour outside $V(P)$.
If $d-p\equiv1\pmod3$, then $v_d$ is a nonzero vertex of $T$.
Symmetrically, if $v_p$ is the vertex of $P$ with the smallest index having a pendant neighbour outside $V(P)$ and $p\equiv1\pmod3$, then $v_0$ is a nonzero vertex of $T$.
<!-- end lemma-10 -->
<!-- proof-7: body_md -->
We prove the first assertion.
Choose a pendant neighbour $w\in V(T)\setminus V(P)$ of $v_p$.
Set $z(v_d)=1$ and $z(v_{d-1})=0$, and propagate from right to left by $z(v_{i-1})=z(v_i)-z(v_{i+1})$ until $v_p$ is reached.
The values repeat with pattern $1,0,-1,-1,0,1$, so $d-p\equiv1\pmod3$ gives $z(v_p)=0$.
Set all coordinates to the left of $v_p$ equal to $0$, put $z(w)=-z(v_{p+1})$, and assign value $0$ to every other pendant vertex outside $V(P)$.
A direct substitution gives $L(T)z=z$ at every vertex, and $z(v_d)=1$.
The second assertion follows by reversing the path.
<!-- end proof-7 -->
<!-- lemma-11: body_md -->
\label{lem:wrong-leaf}
Let $d\equiv2\pmod3$, and let $T$ be obtained from a diametral path $P=v_0v_1\cdots v_d$ by adding one pendant vertex at $v_p$, where $1\le p\le d-1$ and $p\not\equiv1\pmod3$.
Then
$$ m_T[0,1)=\frac{d+1}{3}+1. $$
<!-- end lemma-11 -->
<!-- proof-8: body_md -->
Suppose first that $p\equiv0\pmod3$.
Delete the edge $v_{p-1}v_p$.
The component containing $v_0$ is a path of diameter $p-1\equiv2\pmod3$, and the component containing $v_d$ is a path of diameter $d-p+1\equiv0\pmod3$, with the added pendant vertex as one end.
The joining end of the component containing $v_0$ is nonzero by Lemma~\ref{lem:path-count}.
Lemmas~\ref{lem:cut-additivity} and \ref{lem:path-count} give
$$ m_T[0,1)=\frac{p}{3}+\left(\frac{d-p+1}{3}+1\right)=\frac{d+1}{3}+1. $$

If $p\equiv2\pmod3$, delete $v_pv_{p+1}$ and apply the same argument from the other end.
<!-- end proof-8 -->
<!-- lemma-12: body_md -->
\label{lem:F-structure}
Let $T$ have diameter $3k+1$, and suppose every vertex of $T$ has distance at most $2$ from $P$.
Then $T$ is a caterpillar whose pendant vertices are attached only at vertices of some $V_s$ if and only if $T$ contains none of \textup{(F1)--(F4)} as a subtree containing $P$.
<!-- end lemma-12 -->
<!-- proof-9: body_md -->
Suppose first that every component outside $P$ is a single pendant vertex, and let $I=\{i: v_i \text{ has an added pendant neighbour}\}$.
The set $\{v_i:i\in I\}$ is contained in some $V_s$ precisely when no $i\in I$ is congruent to $2$ modulo $3$, and every $i\in I$ congruent to $1$ occurs before every $i\in I$ congruent to $0$.
Failure of the first condition gives \textup{(F1)}, while failure of the second gives \textup{(F2)}.

Now suppose that a component outside $P$ is nontrivial.
Its center together with one of its leaves gives a pendant path of length two.
If its root is congruent to $2$ modulo $3$, retaining only the center already gives \textup{(F1)}.
If its root is congruent to $1$ or $0$ modulo $3$, the diameter condition excludes the boundary positions, and the component gives \textup{(F3)} or \textup{(F4)}, respectively.
The converse follows immediately from the description of the sets $V_s$.
<!-- end proof-9 -->
<!-- lemma-13: body_md -->
\label{lem:F-spectral}
If $T'$ is one of \textup{(F1)--(F4)}, then
$$ m_{T'}[0,1)=\left\lceil\frac{d+1}{3}\right\rceil+1. $$
<!-- end lemma-13 -->
<!-- proof-10: body_md -->
For \textup{(F1)}, let the added leaf be adjacent to $v_{3a-1}$.
The subtree induced by
$$ \{v_{3a-3},v_{3a-2},v_{3a-1},v_{3a},v_{3a+1}\} $$
together with the added leaf is $T_6$.
Cut $v_{3a-4}v_{3a-3}$ when $a\ge2$, and cut $v_{3a+1}v_{3a+2}$ when $a\le\frac{d-4}{3}$.
The outer components, when present, are paths of diameters $3a-4$ and $d-3a-2$, both congruent to $2$ modulo $3$.
Their joining ends are nonzero by Lemma~\ref{lem:path-count}.
Repeated use of Lemma~\ref{lem:cut-additivity} and the value $m_{T_6}[0,1)=3$ gives
$$ m_{T'}[0,1)=(a-1)+3+\frac{d-3a-1}{3}=\left\lceil\frac{d+1}{3}\right\rceil+1, $$
where an absent path contributes zero.

For \textup{(F2)}, let the two leaves be attached to $v_{3a}$ and $v_{3b+1}$.
Assume first that $a<b$, and delete $v_{3a+1}v_{3a+2}$.
The component containing $v_0$ has diameter $3a+1$ and is an extremal caterpillar, so Lemma~\ref{lem:caterpillar-equality} gives $a+1$ eigenvalues less than $1$.
Its joining vertex $v_{3a+1}$ is nonzero by Lemma~\ref{lem:nonzero-tail}.
The component containing $v_d$ has diameter $d-3a-2\equiv2\pmod3$, and its remaining leaf is attached at a position congruent to $2$ modulo $3$ relative to its displayed diametral path.
Lemma~\ref{lem:wrong-leaf} gives $\frac{d-3a+2}{3}$ eigenvalues less than $1$.
Hence Lemma~\ref{lem:cut-additivity} yields $$m_{T'}[0,1)=\left\lceil\frac{d+1}{3}\right\rceil+1.$$

If $a=b$, delete $v_{3a-1}v_{3a}$.
The component containing $v_0$ is a path of diameter $3a-1$ and contributes $a$.
In the component containing $v_d$, the path beginning with the leaf adjacent to $v_{3a}$ is diametral, has diameter $d-3a+1$, and the other leaf is attached at a forbidden residue class.
Lemma~\ref{lem:wrong-leaf} gives $\frac{d-3a+5}{3}$ eigenvalues less than $1$.
The joining end of the component containing $v_0$ is nonzero, and therefore $$m_{T'}[0,1)=a+\frac{d-3a+5}{3}=\left\lceil\frac{d+1}{3}\right\rceil+1.$$

For \textup{(F3)}, write the pendant path as $v_{3a+1}uw$ and delete $v_{3a-1}v_{3a}$.
The component containing $v_0$ is a path of diameter $3a-1$ and contributes $a$.
Since this component is $P_{3a}$, its joining end $v_{3a-1}$ is nonzero for the eigenvalue $1$ by Lemma~\ref{lem:path-count}.
In the other component, $wuv_{3a+1}v_{3a+2}\cdots v_d$ is a diametral path of length $d-3a+1$, while $v_{3a}$ is a leaf attached at a forbidden position, and Lemma~\ref{lem:wrong-leaf} gives $\frac{d-3a+5}{3}$ eigenvalues less than $1$.
Hence Lemmas~\ref{lem:cut-additivity} and \ref{lem:wrong-leaf} give
$$ m_{T'}[0,1)=a+\frac{d-3a+5}{3}=\left\lceil\frac{d+1}{3}\right\rceil+1. $$

For \textup{(F4)}, write the pendant path as $v_{3a}uw$ and delete $v_{3a-1}v_{3a}$.
The component containing $v_0$ is a path of diameter $3a-1$, while the other component is the path $wuv_{3a}v_{3a+1}\cdots v_d$ of diameter $d-3a+2$.
The joining end of the component containing $v_0$ is nonzero, so Lemmas~\ref{lem:cut-additivity} and \ref{lem:path-count} give
$$ m_{T'}[0,1)=a+\frac{d-3a+5}{3}=\left\lceil\frac{d+1}{3}\right\rceil+1. $$

<!-- corollary-3: body_md -->
\label{cor:d-one}
Let $T$ have diameter $d\equiv1\pmod3$, and suppose every vertex of $T$ has distance at most $2$ from a diametral path $P$.
If $m_T[0,1)=\left\lceil\frac{d+1}{3}\right\rceil$, then $T$ is a caterpillar whose pendant vertices are attached only at vertices contained in a minimum dominating set of $P$.
<!-- end corollary-3 -->
<!-- proof-11: body_md -->
Otherwise Lemma~\ref{lem:F-structure} gives a subtree $T'$ containing $P$ and having one of the forms \textup{(F1)--(F4)}.
Since $T'$ contains the whole diametral path $P$ and every component of $T-V(P)$ is a star with its center adjacent to $P$, the vertices of $T-T'$ can be restored successively as pendant vertices: for each component of $T-V(P)$ not contained in $T'$, first restore its center if necessary and then restore its leaves.
Lemmas~\ref{lem:pendant-monotonicity} and \ref{lem:F-spectral} give $m_T[0,1)\ge m_{T'}[0,1)=\left\lceil\frac{d+1}{3}\right\rceil+1$, a contradiction.

<!-- lemma-14: body_md -->
\label{lem:C-structure}
Let $T$ have diameter $3k$, where $k\ge2$, and suppose every vertex of $T$ has distance at most $2$ from $P$.
Then $T$ belongs to class \textup{(i)} or \textup{(ii)} of Theorem~\ref{thm:classification-intro} if and only if $T$ contains none of \textup{(C1)--(C3)}, whether $P$ is labeled from left to right or from right to left.
<!-- end lemma-14 -->
<!-- proof-12: body_md -->
If every component outside $P$ is a single pendant vertex, the attachment positions are contained in a minimum dominating set of $P$ precisely when their labels are nondecreasing.
Indeed, normalize a minimum dominating set as
$$ D=\{v_{a_1},\ldots,v_{a_{k+1}}\},\qquad a_1=1,\ a_{k+1}=3k-1, $$
replacing $v_0$ by $v_1$ and $v_{3k}$ by $v_{3k-1}$ if necessary, which removes no internal attachment vertex. Set $t_j=a_{j+1}-a_j$. Since $D$ dominates $P$, we have $1\le t_j\le3$, and
$$ \sum_{j=1}^{k}t_j=(3k-1)-1=3k-2,\qquad \text{so}\qquad \sum_{j=1}^{k}(3-t_j)=2. $$
Thus either one gap equals $1$ and the remaining gaps equal $3$, or two gaps equal $2$ and the remaining gaps equal $3$. In the former case the labels $\ell(a_j)$ change once, from $0$ to $2$; in the latter they increase from $0$ to $1$ and then to $2$; in both cases they are nondecreasing. Conversely, if the occupied positions have nondecreasing labels, then placing a gap of size $1$ where the label changes from $0$ to $2$, or two gaps of size $2$ where the label changes from $0$ to $1$ and from $1$ to $2$, and gaps of size $3$ elsewhere, yields a minimum dominating set of $P$ containing all of them.
Thus failure of class \textup{(i)} is exactly \textup{(C1)}.

Suppose now that $T-V(P)$ has a nontrivial component.
Selecting its center and one of its leaves produces a pendant path of length two.
If its root is not divisible by $3$, this is \textup{(C2)}.
Hence assume that its root is $v_{3a}$.
For $T$ to belong to class \textup{(ii)}, this must be the unique nontrivial component, and every ordinary pendant vertex must be attached in
$$ E_a=\{v_1,v_4,\ldots,v_{3a-2}\}\cup\{v_{3a+2},v_{3a+5},\ldots,v_{3k-1}\}. $$
A second nontrivial component rooted at a multiple of $3$, or an ordinary leaf at a multiple of $3$, gives \textup{(C3)} after a suitable choice of the labeling direction of $P$.
When only the center of the selected nontrivial component is retained, it becomes a pendant vertex adjacent to $v_{3a}$.
Thus an ordinary leaf to the left of $v_{3a}$ outside the first part of $E_a$, or to the right outside the second part, gives a label inversion with this pendant vertex and hence gives \textup{(C1)}.
These alternatives exhaust all failures of class \textup{(ii)}.
The converse is immediate from the two constructions in the theorem.

<!-- lemma-15: body_md -->
\label{lem:C-spectral}
If $T'$ is one of \textup{(C1)--(C3)}, then
$$ m_{T'}[0,1)\ge \left\lceil\frac{d+1}{3}\right\rceil+1. $$
<!-- end lemma-15 -->
<!-- proof-13: body_md -->
For \textup{(C1)}, the possible label inversions are $(1,0)$, $(2,1)$ and $(2,0)$.
Reversing $P$ transforms $(2,1)$ into $(1,0)$, so it is enough to treat $(1,0)$ and $(2,0)$.

First suppose $i\equiv0\pmod3$ and $j\equiv1\pmod3$.
Delete $v_{i-1}v_i$.
The left component is a path of diameter $d_1=i-1\equiv2\pmod3$, and its joining end is nonzero.
In the other component, the path beginning with the leaf at $v_i$ is diametral and has diameter $d_2=d-i+1\equiv1\pmod3$.
Relative to this path, the leaf at $v_j$ is attached at a position congruent to $2$ modulo $3$, so it is an \textup{(F1)} structure.
Lemma~\ref{lem:F-spectral} gives $m_{T'_2}[0,1)\ge\lceil(d_2+1)/3\rceil+1$.
Since $d_1+d_2=d$, Lemma~\ref{lem:cut-additivity} gives
$$ m_{T'}[0,1)\ge\frac{d_1+1}{3}+\frac{d_2+2}{3}+1=\left\lceil\frac{d+1}{3}\right\rceil+1. $$

Now suppose $i\equiv2\pmod3$ and $j\equiv1\pmod3$.
Delete $v_{j-1}v_j$.
The component $T'_1$ containing $v_0$ has diameter $d_1=j-1\equiv0\pmod3$ and is an extremal caterpillar, since its unique attachment position $v_i$ is contained in a minimum dominating set of its displayed diametral path.
The other component $T'_2$ is a path of diameter $d_2=d-j+1\equiv0\pmod3$.
The vertex $v_i$ is the rightmost vertex of the displayed path having a pendant neighbour outside the displayed path, and $j-1-i\equiv1\pmod3$.
Thus the joining vertex $v_{j-1}$ is nonzero by Lemma~\ref{lem:nonzero-tail}.
Lemmas~\ref{lem:caterpillar-equality}, \ref{lem:path-count} and \ref{lem:cut-additivity} yield
$$ m_{T'}[0,1)=\left(\frac{d_1}{3}+1\right)+\left(\frac{d_2}{3}+1\right)=\left\lceil\frac{d+1}{3}\right\rceil+1. $$

For \textup{(C2)}, write the pendant path as $v_iuw$.
If $i=2$ or $i=d-2$, reverse $P$ if necessary and assume $i=2$.
Delete $v_3v_4$.
The left component is $T_6$, and the right component is a path of diameter $d-4\equiv2\pmod3$ whose joining end is nonzero.
Hence
$$ m_{T'}[0,1)=3+\frac{d-3}{3}=\left\lceil\frac{d+1}{3}\right\rceil+1. $$

Assume now that the root is not a boundary position.
After reversing $P$ if necessary, we may suppose $i\equiv1\pmod3$ and $4\le i\le d-5$.
Delete $v_{i-2}v_{i-1}$ and $v_{i+2}v_{i+3}$.
The middle component is $T_6$, and the two outer components are paths of diameters $i-2$ and $d-i-3$, both congruent to $2$ modulo $3$.
Their joining ends are nonzero, so repeated use of Lemma~\ref{lem:cut-additivity} gives
$$ m_{T'}[0,1)=\frac{i-1}{3}+3+\frac{d-i-2}{3}=\left\lceil\frac{d+1}{3}\right\rceil+1. $$

For \textup{(C3)}, write the pendant path as $v_{3a}uw$.
Suppose first that $a<b$, and delete $v_{3a+1}v_{3a+2}$.
In the left component $T'_1$, the path $wuv_{3a}v_{3a-1}\cdots v_0$ is diametral and has diameter $d_1=3a+2\equiv2\pmod3$, while $v_{3a+1}$ is a leaf attached at its third vertex.
Lemma~\ref{lem:wrong-leaf} gives $m_{T'_1}[0,1)=a+2$.
The right component $T'_2$ has diameter $d_2=d-3a-2\equiv1\pmod3$ and is an extremal caterpillar, since its unique attachment position $v_{3b}$ is contained in a minimum dominating set of its displayed diametral path.
Thus $m_{T'_2}[0,1)=\frac{d-3a}{3}$.
The attachment vertex $v_{3b}$ lies at distance $3(b-a)-2\equiv1\pmod3$ from the joining end $v_{3a+2}$.
Lemma~\ref{lem:nonzero-tail} shows that $v_{3a+2}$ is nonzero.
Therefore $$m_{T'}[0,1)=a+2+\frac{d-3a}{3}=\left\lceil\frac{d+1}{3}\right\rceil+1.$$
If $a=b$, delete $v_{3a-1}v_{3a}$.
The component containing $v_0$ is a path of diameter $3a-1$ and contributes $a$.
In the other component, $wuv_{3a}v_{3a+1}\cdots v_d$ is a diametral path of length $d-3a+2\equiv2\pmod3$, and the additional leaf at $v_{3a}$ is attached at its third vertex.
Lemma~\ref{lem:wrong-leaf} gives $\frac{d-3a}{3}+2$ eigenvalues less than $1$.
The joining end of the left path is nonzero, and hence $$m_{T'}[0,1)=a+\frac{d-3a}{3}+2=\left\lceil\frac{d+1}{3}\right\rceil+1.$$

<!-- corollary-4: body_md -->
\label{cor:d-zero}
Let $T$ have diameter $d\equiv0\pmod3$ with $d\ge6$, and suppose every vertex of $T$ has distance at most $2$ from a diametral path $P$.
If $m_T[0,1)=\left\lceil\frac{d+1}{3}\right\rceil$, then $T$ belongs to class \textup{(i)} or \textup{(ii)} of Theorem~\ref{thm:classification-intro}.
<!-- end corollary-4 -->
<!-- proof-14: body_md -->
Otherwise Lemma~\ref{lem:C-structure} gives a subtree $T'$ containing $P$ and having one of the forms \textup{(C1)--(C3)}.
Since $T'$ contains the whole diametral path $P$ and every component of $T-V(P)$ is a star with its center adjacent to $P$, the vertices of $T-T'$ can be restored successively as pendant vertices: for each component of $T-V(P)$ not contained in $T'$, first restore its center if necessary and then restore its leaves.
Lemmas~\ref{lem:pendant-monotonicity} and \ref{lem:C-spectral} give $m_T[0,1)\ge m_{T'}[0,1)\ge \left\lceil\frac{d+1}{3}\right\rceil+1$, a contradiction.
<!-- end proof-14 -->
<!-- proof-15: body_md -->
We first prove sufficiency.

Assume that $T$ satisfies (i), and let $D$ be the minimum dominating set of $P$ appearing in the statement.
Since the added pendant vertices are adjacent only to internal vertices of $P$, the path $P$ remains diametral.
Indeed, the distance from such a vertex adjacent to $v_i$ to either endpoint of $P$ is at most $d$, and the distance between two such vertices adjacent to $v_i$ and $v_j$ is at most $1+|i-j|+1\le 1+(d-2)+1=d$.
Thus the diameter is exactly $d$.

Since all added pendant vertices are adjacent to vertices of $D$, the set $D$ dominates all of $T$.
Hence $\gamma(T)\le |D|=\lceil(d+1)/3\rceil$.
By Lemma~\ref{lem:dom-lower}, $\gamma(T)\ge \lceil(d+1)/3\rceil$, so $\gamma(T)=\lceil(d+1)/3\rceil$.
Theorem~\ref{thm:HJT} and Lemma~\ref{lem:spectral-lower} give $\lceil(d+1)/3\rceil \le m_T[0,1) \le \gamma(T) = \lceil(d+1)/3\rceil$.
Therefore $m_T[0,1)=\lceil(d+1)/3\rceil$.

Now assume that $T$ satisfies (ii), and let $u$ be the center of the star $K_{1,t}$.
First we check that the path $P=v_0v_1\cdots v_{3k}$ is a diametral path.
Let $w$ be any vertex outside $P$, and let $v_i$ be the vertex of $P$ closest to $w$.
If $w$ is a pendant vertex adjacent directly to $P$, then $1\le i\le3k-1$, so $d_T(w,v_0)=i+1\le3k$ and $d_T(w,v_{3k})=3k-i+1\le3k$.
If $w=u$, then $i=3r$, and $d_T(u,v_0)=3r+1\le3k$ and $d_T(u,v_{3k})=3k-3r+1\le3k$.
If $w$ is a pendant vertex of $K_{1,t}$, then $d_T(w,v_0)=3r+2\le3k$ and $d_T(w,v_{3k})=3k-3r+2\le3k$, because $1\le r\le k-1$.

It remains to check distances between two vertices outside $P$.
Two pendant vertices adjacent directly to $P$ can only occur at indices between $1$ and $3k-1$, so their distance is at most $1+(3k-2)+1=3k$.
A pendant vertex of $K_{1,t}$ and a pendant vertex adjacent on the left have distance at most $2+(3r-1)+1=3r+2\le3k$, and a pendant vertex of $K_{1,t}$ and a pendant vertex adjacent on the right have distance at most $2+(3k-1-3r)+1=3k-3r+2\le3k$.
The same estimates with $2$ replaced by $1$ cover the center $u$.
Thus every pair of vertices is at distance at most $3k$, while $d_T(v_0,v_{3k})=3k$.
Hence the diameter is exactly $d=3k$.

Define
$$ D = \{v_1,v_4,\ldots,v_{3r-2}\} \cup \{u\} \cup \{v_{3r+2},v_{3r+5},\ldots,v_{3k-1}\}.$$
The vertices $v_1,v_4,\ldots,v_{3r-2}$ dominate the left part of the path up to $v_{3r-1}$.
The vertex $u$ dominates $v_{3r}$ and every vertex of $K_{1,t}$.
The vertices $v_{3r+2},v_{3r+5},\ldots,v_{3k-1}$ dominate the right part of the path from $v_{3r+1}$ to $v_{3k}$.
They also dominate all pendant vertices adjacent directly to $P$.
Thus $D$ dominates $T$.
Its size is $r+1+(k-r)=k+1 = \lceil(3k+1)/3\rceil$.
As above, Lemma~\ref{lem:dom-lower}, Theorem~\ref{thm:HJT}, and Lemma~\ref{lem:spectral-lower} imply $m_T[0,1)=\lceil(d+1)/3\rceil$.

We now prove necessity.
Assume $m_T[0,1)=\lceil(d+1)/3\rceil$.
If $d=0$, then $T\cong P_1$, which satisfies (i).
If $d=1$, then $T\cong P_2$, which also satisfies (i).
Hence assume $d\ge2$.
By Corollary~\ref{cor:distance-two}, there is a diametral path $P=v_0v_1\cdots v_d$ such that every vertex of $T$ has distance at most $2$ from $P$.

If $d\equiv2\pmod3$, the characterization of Guo, Xue and Liu~\cite{GXL} shows that $T$ is obtained from $P$ by adding pendant vertices only at $v_1,v_4,\ldots,v_{d-1}$.
Thus $T$ satisfies (i).

If $d\equiv1\pmod3$, Corollary~\ref{cor:d-one} shows that $T$ satisfies (i).

Finally, suppose $d\equiv0\pmod3$.
If $d=3$, then $T$ is a double star and its two central vertices form a minimum dominating set of a diametral path, so $T$ satisfies (i).
If $d\ge6$, Corollary~\ref{cor:d-zero} shows that $T$ satisfies (i) or (ii).

The proof is complete.

<!-- lemma-16: body_md -->
\label{lem:counting}
For every $n\ge3$,
$$ |\mathcal T_n^*|\le(n+2)2^{n-3}. $$
<!-- end lemma-16 -->
<!-- proof-18: body_md -->
By Theorem~\ref{thm:classification-intro}, every tree in $\mathcal T_n^*$ belongs to class \textup{(i)} or class \textup{(ii)}.

We first count class \textup{(i)}.
Fix the diameter $d$ and choose an orientation $P=v_0v_1\cdots v_d$ of the diametral path occurring in the characterization.
Let $a_i$ be the number of pendant vertices attached to $v_i$.
For an upper bound, we disregard the restrictions imposed by the minimum dominating set and allow $a_i\ge0$ at every path vertex.
Then
$$ a_0+a_1+\cdots+a_d=n-d-1. $$
Therefore, the number of such vectors is
$ \binom{n-1}{d}. $
Every tree in class \textup{(i)} has at least one such representation, while different vectors or orientations may produce the same unlabeled tree.
Consequently, the number of trees in class \textup{(i)} is at most
$$ \sum_{d=0}^{n-1}\binom{n-1}{d}=2^{n-1}. $$

We next count class \textup{(ii)}.
Again fix an oriented path $P=v_0v_1\cdots v_d$, and deliberately disregard the congruence and position restrictions in the characterization.
There are at most $d+1$ choices for the path vertex to which the star center is joined.
Let $s\ge2$ be the order of the attached star component, including its center, and let $a_i\ge0$ be the number of pendant vertices attached directly to $v_i$.
The order condition becomes
$$ s-2+a_0+a_1+\cdots+a_d=n-d-3. $$
For fixed $d$, the number of nonnegative solutions is $\binom{n-2}{d+1}$.
Hence the number of trees in class \textup{(ii)} is at most
$$ \sum_{d=0}^{n-3}(d+1)\binom{n-2}{d+1}=(n-2)2^{n-3}. $$

Adding the two estimates gives
$$ |\mathcal T_n^*|\le2^{n-1}+(n-2)2^{n-3}=(n+2)2^{n-3}. $$

<!-- lemma-17: body_md -->
\label{lem:Otter}
Let $\mathcal T_n$ denote the set of unlabeled trees on $n$ vertices.
Then
$$ |\mathcal T_n| \sim C_0\alpha^n n^{-5/2}, $$
where $C_0>0$ and $\alpha\approx 2.955765\ldots>\frac52$ is Otter's tree constant.
<!-- end lemma-17 -->
<!-- proof-19: body_md -->
Lemma~\ref{lem:counting} gives $|\mathcal T_n^*|=O(n2^n)$.
On the other hand, Lemma~\ref{lem:Otter} gives $|\mathcal T_n| \sim C_0\alpha^n n^{-5/2}$, where $C_0>0$ and $\alpha\approx 2.955765\ldots>\frac52$.
Hence
$$ \frac{|\mathcal T_n^*|}{|\mathcal T_n|} = O\!\left(n^{7/2}\left(\frac{2}{\alpha}\right)^n\right) \xrightarrow[n\to\infty]{}0. $$
This proves the theorem.
