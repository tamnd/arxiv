<!-- definition-1: body_md -->
\label{d:weak_sol}
Fix $\mathbf{U}_o = (\rho_o,q_o) \in \mathbf{L^{\infty}}(\mathbb R; \mathcal{U})$.
A function $\mathbf{U}$ in $\mathbf{L^{\infty}}(\mathbb R_+ \times \mathbb R; \mathcal{U})\cap \mathbf{C^{0}}(\mathbb R_+ ;\mathbf{L^{1}_{loc}}(\mathbb R; \mathcal{U}))$ is a weak solution of the Cauchy problem \eqref{e:2x2} if for any test function $\varphi \in \mathbf{C_c^{\infty}} \left( [0,\infty)\times \mathbb R; \mathbb R\right)$ we have
\begin{equation*}
\int_{\mathbb R_+}\int_\mathbb R \left( \partial_t\varphi + c \, V(v(\mathbf{U})) \, \partial_x\varphi\right) \, \mathbf{U} \, {\rm{d}} x \, {\rm{d}} t + \int_\mathbb R \mathbf{U}_o(x) \, \varphi(0,x) \, {\rm{d}} x = \mathbf{0}.
\end{equation*}

<!-- theorem-1: body_md -->
\label{t:main}
Assume that $p \colon D_p \to \mathbb R$ and $V \colon D_V \to [0,\infty)$ are $\mathbf{C^{2}}$ and satisfy the conditions listed in \eqref{e:p}, \eqref{e:V}, \eqref{e:pV1} and \eqref{e:pV2}.
Then, for any given compact subsets $\mathcal{K}_c$ of $(0,\infty)$ and $\mathcal{K}_{\mathbf{U}}$ of $\mathcal{U}_e$ given in \eqref{e:Ue}, there exists a constant $\delta>0$ such that, for every coefficient function $c \in \mathbf{BV}\cap\mathbf{L^{1}_{loc}}(\mathbb R;(0,\infty))$ and initial datum $\mathbf{U}_o \in \mathbf{L^{1}_{loc}}(\mathbb R;\mathcal{U}_e)$ satisfying the following conditions
\begin{align}
\label{e:Uc_cond}
&\operatorname{TV}[\,\mathbf{U}_o\,] + \operatorname{TV}[\,c\,] \leq \delta,&
&\lim_{x\to-\infty} c(x) \in \mathcal{K}_c,&
&\lim_{x\to-\infty} \mathbf{U}_o(x) \in \mathcal{K}_{\mathbf{U}},
\end{align}
the corresponding Cauchy problem \eqref{e:2x2} admits a globally defined weak solution taking values in $\mathcal{U}_e$.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Consider the Cauchy problem for a $3\times3$ system of conservation laws
\begin{equation}
\label{e:3x3}
\begin{cases}
\partial_t\mathbf{Z}+\partial_xG(\mathbf{Z})=\mathbf{0},
\\
\mathbf{Z}(0,x)=\mathbf{Z}_o(x),
\end{cases}
\end{equation}
where both the conserved quantity $\mathbf{Z} \doteq (\mathbf{U},\zeta) = (\rho,q,\zeta)$ and the initial datum $\mathbf{Z}_o$ take values in the \emph{open} domain
\begin{equation}
\label{e:Ze}
\mathcal{Z}_e \doteq \{ (\rho,q,\zeta) \in \mathbb R^3 : \rho \in \operatorname{int}(D_p),\ v(\rho,q) \in \operatorname{int}(D_V),\ \zeta>0 \},
\end{equation}
and the flux $G \colon \mathcal{Z}_e \to \mathbb R^3$ is defined by $G(\mathbf{U},\zeta) \doteq 
\zeta \, V\left( v(\mathbf{U}) \right) 
(\mathbf{U},0)^T$, where $v$ is defined in \eqref{e:vw}\textsubscript{$1$}.
Observe that $\mathbf{0} \notin \mathcal{Z}_e$.
Moreover, $G$ is $\mathbf{C^{2}}$ in $\mathcal{Z}_e$.
The eigenvalues and the corresponding eigenvectors of the system of conservation laws in \eqref{e:3x3} are
\begin{gather*}
\lambda_1(\mathbf{Z})=\zeta \, \left(V\left(v(\mathbf{U})\right) - \rho \, p'(\rho) \, V'\left(v(\mathbf{U})\right) \right),
\quad
R_1(\mathbf{Z})=\left(\begin{array}{@{}c@{}}
\mathbf{U} \\
0 \\
\end{array}\right),
\\
\lambda_c(\mathbf{Z})=0,
\quad
R_c(\mathbf{Z})=\left(\begin{array}{@{}c@{}}
\mathbf{U} \\
\zeta \, \left( \frac{\rho \, p'(\rho) \, V'\left(v(\mathbf{U})\right)}{V\left(v(\mathbf{U})\right)} - 1 \right) \\
\end{array}\right),
\\
\lambda_2(\mathbf{Z})=\zeta \, V\left(v(\mathbf{U})\right),
\qquad
R_2(\mathbf{Z})=\left(\begin{array}{@{}c@{}}
\rho \\
q + \rho^2 \, p'(\rho) \\
0 \\
\end{array}\right).
\end{gather*}
System \eqref{e:3x3} is strictly hyperbolic in $\mathcal{Z}_e$ because $\lambda_1(\mathbf{Z}) < 0 = \lambda_c(\mathbf{Z}) <  \lambda_2(\mathbf{Z})$ by \eqref{e:V} and \eqref{e:pV2}. 
The characteristic fields related to $\lambda_c$ and $\lambda_2$ are linearly degenerate because $\nabla\lambda_c\cdot r_c \equiv 0$ and $\nabla\lambda_2\cdot r_2 \equiv 0$.
The characteristic field corresponding to $\lambda_1$ is genuinely non-linear because by \eqref{e:pV1} we have
\[
\nabla\lambda_1(\mathbf{Z}) \cdot r_1(\mathbf{Z}) = 
\zeta \, \rho \, \left(
\rho \, p'(\rho)^2 \, V''\left(v(\mathbf{U})\right) 
- \left( 2 p'(\rho) + \rho \, p''(\rho) \right) V'\left(v(\mathbf{U})\right) 
\right) \neq 0.
\]

System \eqref{e:3x3} satisfies the hypotheses of \cite[Theorem~2]{AnconaMarson-2007}, hence for any given compact subset $\mathcal{K}$ of $\mathcal{Z}$, there exists a constant $\delta>0$ such that, for every initial datum $\mathbf{Z}_o \in \mathbf{L^{1}_{loc}}(\mathbb R)$ satisfying the following conditions
\begin{align}
\label{e:Z_cond}
&\operatorname{TV}[\,\mathbf{Z}_o\,] \leq \delta,&
&\lim_{x\to-\infty} \mathbf{Z}_o(x) \in \mathcal{K},
\end{align}
the Cauchy problem~\eqref{e:3x3} admits a weak solution defined globally in time.

We now take $\mathbf{Z}_o = (\mathbf{U}_o,c) = (\rho_o, q_o, c)$.
By \eqref{e:Uc_cond} we have that $\mathbf{Z}_o \in \mathbf{L^{1}_{loc}}(\mathbb R)$ satisfies \eqref{e:Z_cond} with $\mathcal{K} = \mathcal{K}_{\mathbf{U}} \times \mathcal{K}_c$.
Furthermore, $\mathbf{Z}_o$ takes values in $\mathcal{Z}_e$ defined in \eqref{e:Ze} because $\mathbf{U}_o$ takes values in $\mathcal{U}_e$ given in \eqref{e:Ue} and $c$ in $(0,\infty)$.
The Cauchy problem \eqref{e:3x3} corresponding to such initial datum then admits a weak solution $\mathbf{Z} = (\mathbf{U},\zeta) = (\rho,q,\zeta)$ defined for all $t\geq0$, which then satisfies the following condition: for any test function $\varphi \in \mathbf{C_c^{\infty}}([0,\infty)\times\mathbb R;\mathbb R)$
\begin{align*}
&\int_{\mathbb R_+} \int_{\mathbb R} \left( \mathbf{U} \, \partial_t\varphi + \zeta \, V\left(\frac{q}{\rho}-p(\rho)\right) \, \mathbf{U} \, \partial_x\varphi \right) {\rm{d}} x \, {\rm{d}} t + \int_{\mathbb R} \mathbf{U}_o(x) \, \varphi(0,x) {\rm{d}} x = \mathbf{0},\\
&\int_{\mathbb R_+} \int_{\mathbb R} \zeta \, \partial_t\varphi \, {\rm{d}} x \, {\rm{d}} t + \int_{\mathbb R} c(x) \, \varphi(0,x) {\rm{d}} x = 0.
\end{align*}
The last condition implies that $\zeta \equiv c$ a.e.~in $\mathbb R_+ \times \mathbb R$.
Therefore, by substituting $c$ for $\zeta$ in the first two equations, we deduce that $\mathbf{U} = (\rho,q)$ is a weak solution of \eqref{e:2x2} in the sense of Definition~\ref{d:weak_sol} and taking values in $\mathcal{U}_e$.

\end{align*}

For any $w \in [\underline{w},\overline{w}]$, the function $F_{w}(\rho) \doteq \rho \, V\left( v(\rho,w) \right)$ given in \eqref{e:F} is strictly concave down and $F_{w}\left(p^{-1}( w )\right) = 0$.

<!-- proof-2: body_md -->
By \eqref{e:pV1} we have
\begin{align}
\label{e:Fwprimo}
F_{w}'(\rho) &= V\left( w - p(\rho) \right) - \rho \, p'(\rho) \, V'\left( w - p(\rho) \right),
\\
\label{e:Fwsecondo}
F_{w}''(\rho) &= -V'\left( w - p(\rho) \right) \, \left( 2p'(\rho)+\rho\,p''(\rho) \right) + \rho \, p'(\rho)^2 \, V''\left( w - p(\rho) \right) < 0,
\end{align}
hence $F_{w}$ is strictly concave down.
The latter claim follows from \eqref{e:v} and \eqref{e:V}\textsubscript{$1$}.

<!-- definition-3: body_md -->
\label{d:Peryphery}
Fix $\bar{c}>0$. 
The Riemann solver $\mathsf{RS}_{\bar{c}} \colon \mathcal{W}^2 \to \mathbf{BV}(\mathbb R;\mathcal{W})$ for the Riemann problem \eqref{e:Riemann} with $c^{\pm}=\bar{c}$ is defined as follows:
\begin{enumerate}[label={{\rm\bf(L.\arabic*)}}, leftmargin=*]

\item\label{L-C}
If $\mathbf{W}_L,\mathbf{W}_R \in \mathcal{W}$ with $w_L \neq w_R$ and $v(\mathbf{W}_L) = v(\mathbf{W}_R) = v$, then
\[
\mathsf{RS}_{\bar{c}}[\mathbf{W}_L,\mathbf{W}_R](\nu) \doteq
\begin{cases}
\mathbf{W}_L&\hbox{if }\nu/\bar{c} < V(v),
\\
\mathbf{W}_R&\hbox{if }\nu/\bar{c} \geq V(v)).
\end{cases}
\]

\item\label{L-S}
If $\mathbf{W}_L,\mathbf{W}_R \in \mathcal{W}$ with $w_L = w_R$ and $v(\mathbf{W}_L) > v(\mathbf{W}_R)$, then
\[
\mathsf{RS}_{\bar{c}}[\mathbf{W}_L,\mathbf{W}_R](\nu) \doteq
\begin{cases}
\mathbf{W}_L&\hbox{if } \nu/\bar{c} < s(\mathbf{W}_L,\mathbf{W}_R),
\\
\mathbf{W}_R&\hbox{if } \nu/\bar{c} \geq s(\mathbf{W}_L,\mathbf{W}_R),
\end{cases}
\]
where $s$ is given in \eqref{e:s}.

\item\label{L-R}
If $\mathbf{W}_L,\mathbf{W}_R \in \mathcal{W}$ with $w_L = w_R = w$ and $v(\mathbf{W}_L) < v(\mathbf{W}_R)$, then
\[
\mathsf{RS}_{\bar{c}}[\mathbf{W}_L,\mathbf{W}_R](\nu) \doteq
\begin{cases}
\mathbf{W}_L&\hbox{if } \nu/\bar{c} < \lambda_{w}(\rho_L),
\\
\left(\mathcal{H}_w\left(\nu/\bar{c}\right) , w\right)&\hbox{if } \lambda_{w}(\rho_L) \leq \nu/\bar{c} < \lambda_{w}(\rho_R),
\\
\mathbf{W}_R&\hbox{if }\nu/\bar{c} \geq \lambda_{w}(\rho_R),
\end{cases}
\]
where $\lambda_w$ is given in \eqref{e:lambdaw}.

\item\label{L-SC}
If $\mathbf{W}_L,\mathbf{W}_R \in \mathcal{W}$ with $w_L\neq w_R$ and $v(\mathbf{W}_L) > v(\mathbf{W}_R)$, then
\[
\mathsf{RS}_{\bar{c}}[\mathbf{W}_L,\mathbf{W}_R](\nu) \doteq
\begin{cases}
\mathbf{W}_L&\hbox{if } \nu/\bar{c} < s(\mathbf{W}_L,\mathbf{W}_M),
\\
\mathbf{W}_M&\hbox{if } s(\mathbf{W}_L,\mathbf{W}_M)\leq \nu/\bar{c} < V(v(\mathbf{W}_R)),
\\
\mathbf{W}_R&\hbox{if } \nu/\bar{c} \geq V(v(\mathbf{W}_R)),
\end{cases}
\]
where $\mathbf{W}_M  \doteq \mathbf{W}_*(\mathbf{W}_R,w_L)$.
\item\label{L-RC}
If $\mathbf{W}_L,\mathbf{W}_R \in \mathcal{W}$ with $w_L\neq w_R$ and $v(\mathbf{W}_L) < v(\mathbf{W}_R)$, then
\[
\mathsf{RS}_{\bar{c}}[\mathbf{W}_L,\mathbf{W}_R](\nu) \doteq
\begin{cases}
\mathbf{W}_L&\hbox{if }\nu/\bar{c} < \lambda_{w_L}(\rho_L),
\\
\left(\mathcal{H}_{w_L}\left(\nu/\bar{c}\right),w_L\right)&\hbox{if } \lambda_{w_L}(\rho_L) \leq \nu/\bar{c} < \lambda_{w_L}(\rho_M),
\\
\mathbf{W}_M&\hbox{if } \lambda_{w_L}(\rho_M) \leq \nu/\bar{c} < V(v(\mathbf{W}_R)),
\\
\mathbf{W}_R&\hbox{if } \nu/\bar{c} \geq V(v(\mathbf{W}_R)),
\end{cases}
\]
where $\lambda_w$ is given in \eqref{e:lambdaw} and $\mathbf{W}_M  \doteq \mathbf{W}_*(\mathbf{W}_R,w_L)$.
\item
If $\mathbf{W}_L=\mathbf{W}_R$, then $\mathsf{RS}_{\bar{c}}[\mathbf{W}_L,\mathbf{W}_R] \,\dot{\equiv}\, \mathbf{W}_{L,R}$.
\end{enumerate}

<!-- proposition-3: body_md -->
\label{p:OrbitCulture}
If $c^- > c^+ > 0$ and $\overline{v}^\pm = \overline{v} \in D_V$ are such that
\begin{align}
\label{e:OrbitCulture}
(\rho,w) \in \mathcal{W}(\overline{v}) &\Longrightarrow  \lambda_w(\rho) \leq 0,
\end{align}
then $D_{\mathsf{RS}} = \mathcal{W}(\overline{v}^-)\times\mathcal{W}(\overline{v}^+)$.
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
For simplicity of notation, we write $\mathbf{W}_*$ in place of $\mathbf{W}_*(\mathbf{W}_R,w_L)$.
By the hypothesis \eqref{e:OrbitCulture}, we have that
\begin{gather*}
\mathsf{Q}_-(\mathbf{W}_L) = \mathsf{F}_-(w_L),
\qquad
\mathsf{Q}_+(\mathbf{W}_R,w_L) = F_{w_L}(\rho_*),
\\
\mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R) = \min\left\{
c^- \, \mathsf{F}_-(w_L),
c^+ \, F_{w_L}(\rho_*)
\right\} = c^+ \, F_{w_L}(\rho_*),
\end{gather*}
see \eqref{e:Qm}, \eqref{e:Qp} and \eqref{e:Q}.
By \ref{i:trivcheck} we have that $\check{\mathbf{W}}(\mathbf{W}_L,\mathbf{W}_R) = \mathbf{W}_*(\mathbf{W}_R,w_L)$ is well defined and obviously belongs to $\mathcal{W}(\overline{v})$.
Case \ref{i:trivhat} is immediate.
We are then left to consider the case $\mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R) \neq c^- \, F_{w_L}(\rho_L)$ and to prove that there exists $\mathbf{W} \in \mathcal{W}(\overline{v})$ satisfying \eqref{e:hat}.
Observe that $F_{w_L} \colon [\mathsf{r}_-(w_L),p^{-1}(w_L)] \to [0,\mathsf{F}_-(w_L)]$ is invertible by \eqref{e:OrbitCulture}.
Furthermore, the hypotheses $c^->c^+$ and $\overline{v}^\pm=\overline{v}$ imply that $\frac{c^+}{c^-} \, F_{w_L}(\rho_*) \in [0,\mathsf{F}_-(w_L)]$.
Therefore, it is sufficient to take $\rho = F_{w_L}^{-1} \left( \frac{c^+}{c^-} \, F_{w_L}(\rho_*)\right) \in [\mathsf{r}_-(w_L),p^{-1}(w_L)]$.
This concludes the proof.
<!-- end proof-5 -->
<!-- remark-4: body_md -->
For any $(\mathbf{W}_L,\mathbf{W}_R) \in D_{\mathsf{RS}}$, $\mathbf{W}_\pm \doteq \mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R](0^\pm)$ satisfy the following conditions
\begin{align}
\label{e:Landmarks}
\mathbf{W}_- = \hat{\mathbf{W}}(\mathbf{W}_L,\mathbf{W}_R),\
\mathbf{W}_+ = \check{\mathbf{W}}(\mathbf{W}_L,\mathbf{W}_R),\
c^\pm \, F_{w_\pm}(\rho_\pm) = \mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R).
\end{align}
<!-- end remark-4 -->
<!-- remark-5: body_md -->
We stress that $u(t,x) \doteq \mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R](x/t)$ is a weak solution corresponding to the initial datum $(\mathbf{W}_L,\mathbf{W}_R)$.
This simply follows from Proposition~\ref{p:SnarkyPuppy} and \eqref{e:Landmarks}\textsubscript{$3$}.
<!-- end remark-5 -->
<!-- remark-6: body_md -->
In the case $c^-\neq c^+$, we have that $u(t,x) \doteq \mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R](x/t)$ is a stationary weak solution if and only if one of the following mutually exclusive cases occurs: either
\begin{equation*}
v(\mathbf{W}_L) = v(\mathbf{W}_R) = 0,
\end{equation*}
or
\begin{align*}
&w_L = w_R,&
&\rho_L \leq \mathsf{r}_-(w_L),&
&c^- \, F_{w_L}(\rho_L) = c^+ \, F_{w_R}(\rho_R),
\end{align*}
or
\begin{align*}
&w_L = w_R,&
&\rho_L > \mathsf{r}_-(w_L),&
&\rho_R > \mathsf{r}_+(w_R),&
&c^- \, F_{w_L}(\rho_L) = c^+ \, F_{w_R}(\rho_R).
\end{align*}
<!-- end remark-6 -->
<!-- remark-7: body_md -->
Definition~\ref{d:Haken} resembles \cite[Definition~2.1]{Colombo-Goatin_constraint} for a scalar conservation law with a point constraint on the flow at $x=0$, which in our case becomes $c^\pm \, F_{w_\pm}(\rho_\pm) \leq \mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R)$. 
The key difference is that we do not check whether the classical solution satisfies the constraint. Instead, we emphasize selecting, among the self-similar weak solutions that meet the constraint, the one that optimizes the flow.
<!-- end remark-7 -->
<!-- remark-8: body_md -->
For any $(\mathbf{W}_L,\mathbf{W}_R)$ such that $w_L\neq w_R$, the rightmost wave of $\mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R]$ is a $2$-contact discontinuity propagating with speed $c^+ \, V(v(\mathbf{W}_R))$.
We stress that $2$-contact discontinuities always separate the two families of vehicles, characterized by the Lagrangian markers $w_L$ and $w_R$.
<!-- end remark-8 -->
<!-- remark-9: body_md -->
$\mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R]$ expressed in the Riemann invariant coordinates \eqref{e:vw} may have total variation greater than that of the initial datum, that is $|v(\mathbf{W}_L)-v(\mathbf{W}_R)|+|w_L-w_R|$; moreover the maximum principle holds for the $w$-coordinate, but may fail for the $v$-coordinate.
Indeed, if $\mathbf{W}_L = \mathbf{W}_R$, $v(\mathbf{W}_L) = v(\mathbf{W}_R) \neq 0$ and $c^- \neq c^+$, then $\mathsf{RS}_{c^-,c^+}[\mathbf{W}_L, \mathbf{W}_L]$ is not constant, see Section~\ref{s:4} for explicit examples.
<!-- end remark-9 -->
<!-- remark-10: body_md -->
We have a stationary $2$-contact discontinuity at $\nu=0$ if and only if $v(\mathbf{W}_R)=0$.
If this condition is not satisfied, the presence of a stationary discontinuity at $\nu=0$ is instead due to the discontinuity of $c$.
In this case, we refer to such a discontinuity as a $c$-wave.
<!-- end remark-10 -->
<!-- proposition-4: body_md -->
The Riemann solver $\mathsf{RS}_{c^-,c^+} \colon D_{\mathsf{RS}} \to \mathbf{BV}(\mathbb R;\mathcal{W})$ is coherent and $\mathbf{L^{1}_{loc}}$-continuous with respect to the initial datum.
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
We first prove that the Riemann solver $\mathsf{RS}_{c^-,c^+}$ is coherent in $D_{\mathsf{RS}}$.
To do so, it is sufficient to show that for any $(\mathbf{W}_L,\mathbf{W}_R) \in D_{\mathsf{RS}}$ we have
\begin{equation}
\label{e:CoheedandCambria}
\mathsf{Q}(\mathbf{W}_-,\mathbf{W}_+) = \mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R)
\end{equation}
where $\mathbf{W}_\pm = \mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R](0^\pm)$.
Indeed, by \eqref{e:Landmarks}\textsubscript{$3$} and \eqref{e:CoheedandCambria} we have
\[c^\pm \, F_{w_\pm}\left(\rho_\pm\right) = \mathsf{Q}(\mathbf{W}_-,\mathbf{W}_+).\]
By \ref{i:trivhat}, \ref{i:trivcheck} and \eqref{e:Landmarks}\textsubscript{$1,2$} we have then $\mathsf{RS}_{c^-,c^+}[\mathbf{W}_-,\mathbf{W}_+](0^\pm) = \mathbf{W}_\pm$ and therefore $\mathsf{RS}_{c^-,c^+}$ is coherent at $(\mathbf{W}_L, \mathbf{W}_R)$, see \eqref{e:PorcupineTree}.
We are therefore left to prove \eqref{e:CoheedandCambria}.
Suppose, by contradiction, that there exists $(\mathbf{W}_L, \mathbf{W}_R) \in D_{\mathsf{RS}}$ such that \eqref{e:CoheedandCambria} is not satisfied, namely
\begin{equation}
\label{e:Blackfield}
\mathsf{Q}(\mathbf{W}_-,\mathbf{W}_+) \neq \mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R).
\end{equation}
By \ref{i:trivhat}, \eqref{e:hat}\textsubscript{$2$}, \eqref{e:w*}\textsubscript{$2$}, \ref{i:trivcheck} and \eqref{e:check}\textsubscript{$2$} we have $\hat{w}(\mathbf{W}_L,\mathbf{W}_R) = \check{w}(\mathbf{W}_L,\mathbf{W}_R) = w_L$, hence by \eqref{e:Landmarks}\textsubscript{$1,2$}
\begin{equation}
\label{e:Klone}    
w_\pm = w_L.
\end{equation}
This implies that $\mathbf{W}_*(\mathbf{W}_+,w_-) = \mathbf{W}_+$, see \eqref{e:w*}.
By \eqref{e:Qm}, \eqref{e:Qp}, \eqref{e:Q}, \eqref{e:Landmarks}\textsubscript{$3$} and \eqref{e:Blackfield} we have that $\mathsf{Q}(\mathbf{W}_-,\mathbf{W}_+) > 
\mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R) =
c^\pm \, F_{w_\pm}(\rho_\pm)$.
As a consequence, by \ref{i:hat}, \eqref{e:hat}\textsubscript{$1$}, \ref{i:check}, \eqref{e:check}\textsubscript{$1$} and \eqref{e:Klone} we have
\begin{align*}
&\rho_- > \mathsf{r}_-(w_L),&
&\rho_+ = \rho_*(\mathbf{W}_+,w_-) < \mathsf{r}_+(w_L).
\end{align*}
The above conditions imply that
\begin{align*}
\hbox{either } \quad F_{w_L}(\rho_L) > F_{w_-}(\rho_-) \quad \hbox{ or } \quad
&\begin{cases}
F_{w_L}(\rho_L) \leq F_{w_-}(\rho_-),
\\
\rho_L > \mathsf{r}_-(w_L), 
\end{cases}
\\
\hbox{either } \quad F_{w_*}(\rho_*) > F_{w_+}(\rho_+) \quad \hbox{ or } \quad
&\begin{cases}
F_{w_*}(\rho_*) \leq F_{w_+}(\rho_+),
\\
\rho_* < \mathsf{r}_+(w_L).
\end{cases}
\end{align*}
In any case, we get $\mathsf{Q}_-(\mathbf{W}_L) > F_{w_L}(\rho_-)$ and $\mathsf{Q}_+(\mathbf{W}_R,w_L) > F_{w_L}(\rho_+)$, hence $\mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R) > c^\pm \, F_{w_\pm}(\rho_\pm)$.
This gives a contradiction and therefore \eqref{e:CoheedandCambria} holds.
As a consequence, $\mathsf{RS}_{c^-,c^+}$ is coherent.

We now prove that the Riemann solver $\mathsf{RS}_{c^-,c^+}$ is $\mathbf{L^{1}_{loc}}$-continuous with respect to the initial data in $D_{\mathsf{RS}}$.
The case $c^- = c^+$ is trivial and is therefore omitted.
Observe first that  $\mathsf{Q}$ is continuous because $\mathsf{Q}_-$ and $\mathsf{Q}_+$ are so.
By the continuity of the speed of propagation of the waves, it is sufficient to consider initial data for which the traces at $\nu=0$ are discontinuous.
                    
           
                                                                                      

                                                                              
                                        
                                     
                                                               
                                                 
                                         
                                                  

                              
                                                                              
                                        
                                     
                                                  
                                                               
                                                        
                                     
            

                  
                                                              
                                                                                                                                              
                                                                        
                
             
More precisely, assume that $(\mathbf{W}_L^\varepsilon, \mathbf{W}_R^\varepsilon)$ converges to $(\mathbf{W}_L,\mathbf{W}_R)$ (as $\varepsilon$ tends to zero), but the traces $\mathbf{W}_\pm^\varepsilon \doteq \mathsf{RS}_{c^-,c^+}[\mathbf{W}_L^\varepsilon, \mathbf{W}_R^\varepsilon](0^\pm)$ do not converge to $\mathbf{W}_\pm \doteq \mathsf{RS}_{c^-,c^+}[\mathbf{W}_L, \mathbf{W}_R](0^\pm)$.
                                   
By \eqref{e:Landmarks}\textsubscript{$1,3$} and Definition~\ref{d:hat}, we have to consider the following cases:
\begin{itemize}[leftmargin=*]
\item
Assume that $\rho_L \leq \mathsf{r}_-(w_L)$ and $\mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R) = c^- \, F_{w_L}(\rho_L)$, so that $\mathbf{W}_- = \mathbf{W}_L$.
Consider now a perturbed initial datum $(\mathbf{W}_L^\varepsilon, \mathbf{W}_R^\varepsilon)$ such that $\rho_L^\varepsilon \leq \mathsf{r}_-(w_L^\varepsilon)$, $\mathsf{Q}(\mathbf{W}_L^\varepsilon,\mathbf{W}_R^\varepsilon) < c^- \, F_{w_L^\varepsilon}(\rho_L^\varepsilon)$, so that $\mathbf{W}_-^\varepsilon \neq \mathbf{W}_L^\varepsilon$.
Observe that $\mathsf{RS}_-[\mathbf{W}_L,\mathbf{W}_-]$ is constant, whereas $\mathsf{RS}_-[\mathbf{W}_L^\varepsilon,\mathbf{W}_-^\varepsilon]$ has just one wave, a $1$-shock, with speed of propagation that converges to zero as $\varepsilon \to 0$.
This ensures the $\mathbf{L^{1}_{loc}}$-convergence of $\mathsf{RS}_{c^-,c^+}[\mathbf{W}_L^\varepsilon,\mathbf{W}_R^\varepsilon]$ to $\mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R]$.
\item 
Assume that $\rho_* \geq \mathsf{r}_+(w_L)$ and $\mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R) = c^+ \, F_{w_L}(\rho_*)$, so that $\mathbf{W}_+ = \mathbf{W}_*$.
Consider now a perturbed initial datum $(\mathbf{W}_L^\varepsilon, \mathbf{W}_R^\varepsilon)$ such that $\rho_*^\varepsilon \geq \mathsf{r}_+(w_L^\varepsilon)$, $\mathsf{Q}(\mathbf{W}_L^\varepsilon,\mathbf{W}_R^\varepsilon) < c^+ \, F_{w_L^\varepsilon}(\rho_*^\varepsilon)$, so that $\mathbf{W}_+^\varepsilon \neq \mathbf{W}_*^\varepsilon$.
Observe that $\mathsf{RS}_+[\mathbf{W}_+,\mathbf{W}_*]$ is constant, whereas $\mathsf{RS}_+[\mathbf{W}_+^\varepsilon,\mathbf{W}_*^\varepsilon]$ has just one wave, a $1$-shock, with speed of propagation that converges to zero as $\varepsilon \to 0$.
This ensures the $\mathbf{L^{1}_{loc}}$-convergence of $\mathsf{RS}_{c^-,c^+}[\mathbf{W}_L^\varepsilon,\mathbf{W}_R^\varepsilon]$ to $\mathsf{RS}_{c^-,c^+}[\mathbf{W}_L,\mathbf{W}_R]$.
\end{itemize}
This concludes the proof.

\label{ex:3}
In \figurename~\ref{f:sol-Ber-Sinh} we show the construction of the weak solution $u$ in the case $p(\rho) = \left(\frac{1}{\rho}-1\right)^{-1}$, $V(v) = \sinh(v)$ and with
\begin{align}
\label{e:inivac}
&c^-=2,&
&c^+=3,&
&\rho_L = \frac{2}{5},&
&w_L = \underline{w} \doteq 1,&
&w_R = \frac{13}{10},&
&\rho_R = p^{-1}\left(w_R-\overline{v}^+\right) = \frac{3}{13},
\end{align}
where $\overline{v}^+ = \operatorname{arcsinh}\left(\lambda_{w_L}(0)\right) = \operatorname{arcsinh}\left(\frac{e^2-1}{2 e}\right)$ and $e \approx 2.71828$ is Euler’s number.
In this case $\rho_L > \mathsf{r}_-(w_L)$, $\mathbf{W}_*(\mathbf{W}_R,w_L) = (0,w_L)$ and $\rho_*(\mathbf{W}_R,w_L) < \mathsf{r}_+(w_L)$, therefore by definition, see \eqref{e:Qm}, \eqref{e:Qp} and \eqref{e:Q}, we have
\begin{align*}
&\mathsf{Q}_-(\mathbf{W}_L) = \mathsf{F}_-(w_L),&
&\mathsf{Q}_+(\mathbf{W}_R,w_L) = \mathsf{F}_+(w_L),&
&\mathsf{Q}(\mathbf{W}_L,\mathbf{W}_R) = c^- \, \mathsf{F}_-(w_L).
\end{align*}
By \ref{i:hat} and \ref{i:check} we have $\hat{\mathbf{W}}(\mathbf{W}_L,\mathbf{W}_R) = ( \mathsf{r}_-(w_L)  , w_L)$ and $\check{\mathbf{W}}(\mathbf{W}_L,\mathbf{W}_R) = ( F_{w_L}^{-1}(\frac{c^-}{c^+} \, \mathsf{F}_-(w_L)) , w_L )$.
Therefore, $u$ has a $1$-rarefaction in the cone $c^- \, \lambda_{w_L}(\rho_L) \leq x/t \leq c^- \, \lambda_{w_L}(\mathsf{r}_-(w_L)) = 0$, a stationary $c$-wave at $x=0$, a $1$-rarefaction in the cone $c^+ \, \lambda_{w_L}(\check{\rho}(\mathbf{W}_L,\mathbf{W}_R)) \leq x/t \leq c^+ \, \lambda_{w_L}(0)$ and a $2$-contact discontinuity at $x = c^+ \, V(v(\mathbf{W}_R))$.
Observe that $V(v(\mathbf{W}_R)) = \lambda_{w_L}(0) = V(\overline{v}^+)$.
