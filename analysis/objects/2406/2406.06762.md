<!-- theorem-1: body_md -->
Let $M$ be a smooth manifold and $g$ a Riemannian metric tensor on $M$ of regularity $C^k$, $k \geq 1$. Suppose that the following hold:
\begin{enumerate}
    \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,d_g)$ is a complete metric space, where $d_g$ is the Riemannian distance on $M$ induced by $g$.
    \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $\mathrm{Ric}(X,X) \geq 0$ in $\mathcal{D}'(M)$ for every compactly supported smooth vector field $X$ on $M$.
    \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} There exists a geodesic $c:\mathbb R \to M$ with the property $d_g(c(s),c(t)) = |s-t|$ for all $s,t \in \mathbb R$ (i.e.\ $c$ is a \emph{line}). 
\end{enumerate}
Then there exists an isometric $C^{k+1}$-diffeomorphism $\Phi:(M' \times \mathbb R, h+dt^2) \to (M,g)$, where $M'$ is a smooth manifold with a Riemannian metric $h \in C^k$ which satisfies $\mathrm{Ric}_h \geq 0$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
Let $M$ be a smooth manifold and $g$ a semi-Riemannian metric tensor on $M$ of constant signature $(l_1,l_2)$ and of regularity $C^k$, $k \geq 1$. Then the following are equivalent:
    \begin{enumerate}
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $\mathrm{Riem}_g = 0$ in $C^{k-2}(T^1_3 M)$ (where $C^{k-2} = \mathcal{D}^{'(1)}$ if $k=1$).
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,g)$ is $C^k$-frame flat, i.e.\ in a neighborhood of every point there exists a parallel orthonormal frame of regularity $C^k$.
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,g)$ is $C^{k+1}$-coordinate flat, i.e.\ each point in $M$ is contained in a neighborhood $U$ for which there exists an isometric $C^{k+1}$-diffeomorphism $\varphi:U \to \varphi(U) \subseteq \mathbb R^{l_1,l_2}$ onto an open subset of $\mathbb R^{l_1,l_2}$, the semi-Euclidean space of signature $(l_1,l_2)$.
    \end{enumerate}

<!-- definition-1: body_md -->
We call a map $\nabla:C^\infty(TM)\times C^\infty(TM) \to \mathcal{D}'(TM)$, 
    \emph{a distributional connection} if it satisfies
     for all $X,X',Y,Y'\in C^\infty(TM)$ and $f\in C^\infty(M)$
    \begin{enumerate}
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $\nabla_{fX+X'}Y = f\nabla_{X}Y+\nabla_{X'}Y$,
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $\nabla_X(Y+Y')= \nabla_X Y + \nabla_X Y'$,
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $\nabla_X (fY) = X(f) Y + f \nabla_X Y$.
    \end{enumerate}
    Given a functional (or distributional) space $\mathcal{F}$, we say a distributional connection $\nabla$ is an \emph{$\mathcal{F}$-connection} if the image of $\nabla$ is contained in $\mathcal{F}(TM)$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Fix a non-negative convolution kernel $\{\rho_\varepsilon\}_{\varepsilon>0}$ on $\mathbb R^n$ and a countable atlas $\{(U_\alpha, \psi_\alpha) \}_{\alpha\in \mathbb N}$ with relatively compact $U_\alpha$, a subordinate smooth partition of unity $\{\xi_\alpha\}_\alpha$ and functions $\chi_\alpha \in C^\infty_c(M)$ with $0\leq \chi_\alpha\leq 1$ and $\chi_\alpha \equiv 1$ on a neighborhood of $\operatorname{supp} \xi_\alpha $. Then for any $T\in \mathcal{D}'(T^r_s M)$ and $\varepsilon>0$ we define the smooth $(r,s)$-tensor
\begin{equation*}
    T \star_M \rho_\varepsilon := \sum_{\alpha\in\mathbb N}\chi_\alpha (\psi_\alpha)^*( ((\psi_\alpha)_*(\xi_\alpha T) )* \rho_\varepsilon),
\end{equation*}
where the convolution $((\psi_\alpha)_*(\xi_\alpha T) )* \rho_\varepsilon$ is to be understood in the component-wise sense, recalling that each component is in $\mathcal{D}'(\psi_\alpha (U_\alpha))$.
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop:standard_properties_of_starM}
    The following statements hold:
    \begin{enumerate}
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} If $T \in \mathcal{D}'(T^0_2 M)$ is symmetric and satisfies $T\geq 0$, then $T \star_M \rho_\varepsilon\ge 0$ for all $\varepsilon>0$.
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} If $T\in \mathcal{D}'(T^r_s M)$, then 
        \begin{equation*}
            \langle T \star_M \rho_\varepsilon , \mu \rangle \to \langle T,\mu \rangle \qquad \forall \mu\in C^\infty_c(T^s_r M\otimes \mathrm{Vol}\, M).
        \end{equation*}
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} If $T\in C^k$, for some $k\in \mathbb N$, then $T \star_M \rho_\varepsilon \to T$ in $C^k_\text{\normalfont loc}$.
    \end{enumerate}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
All of these claims follow from the definition of $\star_M$ and standard
properties of convolution on $\mathbb R^n$.

<!-- remark-1: body_md -->
\label{Remark: Approximatingmetrics}
    Given a continuous semi-Riemannian metric $g$ on $M$, by Proposition \ref{prop:standard_properties_of_starM} (iii), $g\star_M \rho_\varepsilon \to g$ locally uniformly for $\varepsilon\to 0$. Thus for any $K\Subset M$ there is an $\varepsilon_K>0$ such that for all $\varepsilon\in (0,\varepsilon_K]$
we have:
\begin{align}
g\star_M\rho_\varepsilon|_K \text{ is a semi-Riemannian metric of the same signature as $g$}. \label{geps1} 
\end{align}
Since \eqref{geps1} is stable with respect to decreasing $K$ and $\varepsilon$, we may apply \cite[Lem.\ 4.3]{HKS_wave} to obtain a smooth
map $(\varepsilon,x) \mapsto g_\varepsilon(x)$ that possesses this property globally on $(0,1] \times M$ (i.e., each $g_\varepsilon$ is a semi-Riemannian metric on $M$ of the same signature as $g$) and such that, in addition, for every $K\Subset M$ there exists some $\varepsilon_K>0$ with $g_\varepsilon(x) = g\star_M\rho_\varepsilon(x)$ for all $(\varepsilon,x)\in (0,\varepsilon_K]\times K$. 

In the special case (most relevant for our purposes here) of $g$ being Riemannian, \eqref{geps1} is automatically satisfied globally (since local convolution with $\rho_\varepsilon \ge 0$ preserves positive definiteness). In this signature, we additionally have that for each $K\Subset M$ there is an $\varepsilon_K>0$ such that for all $\varepsilon\in (0,\varepsilon_K]$
\begin{equation}\label{geps2} 
\frac{1}{2}\abs{v}_g \le \abs{v}_{g\star_M\rho_\varepsilon} \le 2\abs{v}_g \quad \forall v\in TM|_K,
\end{equation}
and again applying \cite[Lem.\ 4.3]{HKS_wave} we construct Riemannian metrics $g_\varepsilon$ satisfying 
\begin{equation}\label{geps3} 
\frac{1}{2}\abs{v}_g \le \abs{v}_{g_\varepsilon} \le 2\abs{v}_g \quad \forall v\in TM,
\end{equation}
Then in particular, for any  $C^1$-curve $\gamma$ connecting two points in $M$, for its lengths with respect to $g$ resp.\ $g_\varepsilon$ we have $\frac{1}{2}L_g(\gamma) \le L_{g_\varepsilon}(\gamma) \le 2L_g(\gamma)$, from which it follows that for the corresponding Riemannian distances
we have:
\begin{equation}\label{eq:distances_g_geps}
    \frac{1}{2} d_g(x,y) \le d_{g_\varepsilon}(x,y) \le 2d_g(x,y) \qquad \forall x, y \in M.
\end{equation}
In particular, for the corresponding metric balls this gives:
\begin{equation}\label{eq:balls_g_geps}
B^{d_g}_{r/2}(x) \subseteq B^{d_{g_\varepsilon}}_r(x) \subseteq B^{d_g}_{2r}(x)
\qquad (x\in M, r>0).
\end{equation}

\label{lem:W-1pCovar}
	Let $(M,g)$ be a Riemannian manifold with $g \in C^{0,1}_{\text{\normalfont loc}}(T^0_2 M )$. For $V\in L^{\infty^-}_\text{\normalfont loc}(TM)$ and $W\in W^{1,\infty^-}_\text{\normalfont loc}(TM)$, it holds that
	\begin{equation*}
		\nabla_V W \in L^{\infty^-}_\text{\normalfont loc}(TM),\qquad  \nabla_W V \in W^{-1,1+ \frac{1}{n}}_\text{\normalfont loc}(TM).
	\end{equation*}
 Moreover, for $X\in W^{-1,1+ \frac{1}{n}}_\text{\normalfont loc}(TM)$ and $Y\in W^{1,\infty^-}_\text{\normalfont loc}(TM)$, their scalar product is a well defined as a distribution
 \begin{equation*}
     g( X, Y ) \in W^{-1,1+ \frac{1}{n}}_\text{\normalfont loc}(M).
 \end{equation*}
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
Since these claims are local, we may without loss of generality suppose that 
 all vector fields are compactly supported and $M=\mathbb R^n$.
	Recall that in smooth local coordinates
	\begin{align*}
		(\nabla_V W )^\beta= V^\alpha(\partial_\alpha W^\beta + \Gamma_{\alpha\gamma}^\beta W^\gamma).
	\end{align*}
	Observing that $\Gamma_{\alpha\gamma}^\beta \in L^\infty_{\mathrm{loc}}(\mathbb R^n)$ and that $L^{\infty^-}(\mathbb R^n)$ is closed with respect to the pointwise product, we conclude that $(\nabla_V W)^\beta \in L^{\infty^-}(\mathbb R^n)$.\\
	Similarly, applying Remark \ref{rk:W-1pMultipl} to	\begin{align*}
		(\nabla_W V )^\beta= W^\alpha(\partial_\alpha V^\beta + \Gamma_{\alpha\gamma}^\beta V^\gamma).
	\end{align*}
we see that $W^\alpha\partial_\alpha V^\beta \in W^{-1,1+\frac{1}{n}}(\mathbb R^n)$, while $W^\alpha\Gamma_{\alpha\gamma}^\beta V^\gamma \in L^{\infty^-}(\mathbb R^n)$. We conclude by recalling that
\begin{equation*}
    L^{\infty^-}(\mathbb R^n)\subseteq L^{1+\frac{1}{n}}(\mathbb R^n) \cong (L^{n+1}(\mathbb R^n))' \subseteq (W^{1, n+1}(\mathbb R^n))' = W^{-1,1+\frac{1}{n} }(\mathbb R^n).
\end{equation*}
For the second statement observe that $g( X,Y ) = X^\alpha Y^\beta g_{\alpha\beta}$. Here $Y^\beta\in W^{1,\infty^-}(\mathbb R^n)$ and $g_{\alpha\beta}\in W^{1,\infty}_{\mathrm{loc}}(\mathbb R^n)$, thus $Y^\beta g_{\alpha\beta}\in W^{1,\infty^-}(\mathbb R^n)$. Thus another appeal to Remark \ref{rk:W-1pMultipl} concludes the proof.

<!-- theorem-3: body_md -->
\label{thm:BochWeitz}
    Let $(M,g)$ be a Riemannian manifold with metric $g \in C^{0,1}_{\text{\normalfont loc}}(T^0_2 M)$. Then for all vector fields $X\in W^{1,\infty^-}_\text{\normalfont loc}(TM)$ the following identity holds in $W^{-1,1+\frac{1}{n}}_{\text{\normalfont loc}}(M) \subseteq \mathcal{D}'(M)$: 
	\begin{equation}\label{eq:BochWeitz}
    	\mathrm{Ric}(X,X)= \Delta \frac{\abs{X}^2}{2} - \abs{\nabla X}^2 +
            g( \Delta_H X,X).
	\end{equation}
<!-- end theorem-3 -->
<!-- proof-4: body_md -->
Let us argue locally by fixing an ONF $\{e_i\}_i$, which consists of $C^{0,1}_\text{\normalfont loc}$-vector fields. Lemma \ref{lem:W-1pCovar} ensures that all objects in this proof are well-defined.\\
    We first compute $\text{\normalfont d}\delta X^\flat$ explicitly, evaluated on an arbitrary $W^{1,\infty^-}_\text{\normalfont loc}$-vector field $Y$. Observe that since $g\in W^{1,\infty}_\text{\normalfont loc}(T^0_2 M)$ and $X\in W^{1,\infty^-}_\text{\normalfont loc}(TM)$, then $X^\flat \in W^{1,\infty^-}_\text{\normalfont loc} (T^*M)$. 
    \begin{align*}
        \text{\normalfont d}\delta X^\flat (Y)&= \nabla_Y\Big(-\sum_i \nabla_{e_i} X^\flat (e_i)\Big) = -\sum_i \big(\nabla_Y (\nabla_{e_i} X^\flat)(e_i) +\nabla_{e_i} X^\flat (\nabla_Y e_i) \big)\\
        &=-\sum_i \big(\nabla^2_{Y,e_i}X^\flat(e_i) + \nabla_{e_i} X^\flat (\nabla_Y e_i) +\nabla_{\nabla_Y e_i} X^\flat (e_i)\big).
    \end{align*}
    Now we claim that 
    \begin{align*}
        \sum_i \big(\nabla_{e_i}  X^\flat (\nabla_Y e_i) +\nabla_{\nabla_Y e_i} X^\flat (e_i)\big) =0 .
    \end{align*}
    Indeed up to $L^1_\text{\normalfont loc}$-linearity observe that for $\nabla X^\flat = \alpha \otimes \beta$
    \begin{align*}
        &\sum_i \alpha\otimes\beta (e_i, \nabla_Y e_i) + \alpha\otimes\beta (\nabla_Y e_i , e_i) = \sum_i \alpha\otimes\beta ( \nabla_Y (e_i\otimes e_i))\\
        &= \nabla_Y\left[\sum_i \alpha\otimes\beta (e_i, e_i) \right] -\sum_i(\nabla_Y \alpha)\otimes \beta(e_i,e_i) -\sum_i \alpha\otimes(\nabla_Y \beta) (e_i,e_i)\\
        &=\nabla_Y (\alpha (\beta^\sharp)) - (\nabla_Y \alpha)(\beta^\sharp) - \alpha(\nabla_Y \beta^\sharp)=0.
    \end{align*}
    Let us denote by $S^T$ the transposed tensor of the $(0,2)$-tensor $S$, i.e. $S^T(V,W):= S(W,V)$. We can then compute
    \begin{align*}
        \delta\text{\normalfont d} X^\flat (Y) &= \delta\left[(V,W)\mapsto \nabla X^\flat (V,W) - \nabla X^{\flat}(W,V) \right] (Y)\\
        &=-\sum_i \nabla_{e_i} ( \nabla X^\flat - \nabla X^{\flat T})(e_i, Y) \\
        &=-\sum_i \nabla_{e_i} (\nabla X^\flat)(e_i, Y) + \sum_i \nabla_{e_i} ( \nabla X^{\flat})(Y,e_i) \\
        &= -\sum \nabla^2_{e_i,e_i} X^\flat (Y) + \sum_i \nabla^2_{e_i,Y} X^\flat (e_i).
    \end{align*}
    Adding together the two previous results, with $Y=X$, we get
    \begin{align*}
        \Delta_H X^\flat (X) = - \sum_i \nabla^2_{e_i,e_i}X^\flat (X) + \sum_i (\nabla^2_{e_i,X} X^\flat (e_i) - \nabla^2_{X,e_i} X^\flat (e_i)),
    \end{align*}
    which can be rewritten as
    \begin{align*}
        g( \Delta_H X, X) = -g( \sum_i \nabla^2_{e_i,e_i}X ,X) + \mathrm{Ric}(X,X).
    \end{align*}
    Indeed it holds that $\nabla V^\flat = (\nabla V)^\flat$ for $V\in L^{\infty^-}_\text{\normalfont loc}(TM)$, which is to be understood as a distributional identity. This can be verified in coordinates:
\begin{align*}
    &[\nabla V^\flat - (\nabla V)^\flat]_{\alpha\beta}\\
    &=\partial_\alpha (g_{\beta i} V^i) - \Gamma^j_{\alpha\beta}g_{ji}V^i - g_{\beta i}\partial_\alpha V^i -\Gamma_{\alpha i}^j g_{\beta j}V^i\\
    &= \frac{V^i}{2}\left( 2\partial_\alpha g_{\beta i} -\partial_\alpha g_{\beta i} -\partial_\beta g_{\alpha i} +\partial_i g_{\alpha\beta } -\partial_\alpha g_{\beta i} -\partial_i g_{\alpha \beta} +\partial_\beta g_{\alpha i}\right)\\
    &=0.
\end{align*}
Apart from $g_{\beta i} \partial_\alpha V^i$, which is admissible by Remark \ref{rk:W-1pMultipl}, all the above are pointwise multiplications of functions.\\
    Now observe that
    \begin{align*}
        g( \sum_i \nabla^2_{e_i,e_i}X ,X) 
        &= \sum_i (g(  \nabla_{e_i}\nabla_{e_i}X,X) -g(\nabla_{\nabla_{e_i}e_i}X,X))\\
        &=\sum_i (\nabla_{e_i}g(\nabla_{e_i}X,X) - g( \nabla_{e_i}X, \nabla_{e_i}X) -g(\nabla_{\nabla_{e_i}e_i}X,X))\\
        &=\sum_i \nabla_{e_i}\left[V\mapsto g(\nabla_V X,X )\right] (e_i) - \abs{\nabla X}^2\\
        &=\sum_i \nabla_{e_i}(\text{\normalfont d} \frac{\abs{X}^2}{2}) (e_i) - \abs{\nabla X}^2\\
        &= \Delta \frac{\abs{X}^2}{2} - \abs{\nabla X}^2.
    \end{align*}

Observe that by polarization of (\ref{eq:BochWeitz}) we obtain a complete characterization of $\mathrm{Ric}(X,Y)$ for $X,Y \in W^{1,\infty^-}_\text{\normalfont loc}(TM)$:

        \mathrm{Ric}(X,Y) &= \frac{1}{4}\left(\mathrm{Ric}(X+Y,X+Y)- \mathrm{Ric}(X-Y,X-Y)\right)\\

<!-- proposition-3: body_md -->
\label{prop:busemannSubharmonic}
    The Busemann functions $b^\pm$ are subharmonic in the distributional sense, i.e. for all non-negative $\varphi\in C^\infty_c(M)$ it holds that
    \begin{equation}
        \int_M b^\pm \Delta \varphi \,\text{\normalfont d}\mathrm{Vol} \geq 0.\label{eq:busemann_subharmonic}
    \end{equation}
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Let us  denote $d_t:=d(\,\cdot\, ,c(t))$ and $d_{i,t}:=d_{i}(\,\cdot\, ,c(t))$. Given $\varphi\in C^\infty_c(M)$, let $K \Subset M$ be a compact set with $\operatorname{supp}(\varphi) \subseteq K$. Then $d_{i,t} \to d_t$ uniformly on $K$ as $i\to \infty$. Moreover, due to $g_i \to g$ in $C^1_{loc}$ and $\mathrm{Ric} \geq 0$, there exists a sequence $\delta_i\searrow 0$ such that $\mathrm{Ric}_i \geq - (n-1) \delta_i g_i$ on $TM|_K$ (cf.\ Proposition \ref{prop:LpRicConvergence}). Thus 
    \begin{align*}
        \int_M d_{i,t} \Delta_i \varphi \, \text{\normalfont d}\mathrm{Vol}_i = \int_M (\Delta_{i} d_{i,t}) \varphi \, \text{\normalfont d}\mathrm{Vol}_i \leq (n-1) \int_M \sqrt{\delta_i} \coth(\sqrt{\delta_i} d_{i,t}) \varphi \, \text{\normalfont d}\mathrm{Vol}_i,
    \end{align*} 
    because integration happens on the compact set $K$ and standard Laplacian comparison (see e.g.\ \cite[Lem.\ 7.1.9]{petersen2006riemannian}) 
    works there due to the estimate on $K$ for $\mathrm{Ric}_i$. Also
    \begin{align*}
        \int_M d_{i,t} \Delta_i \varphi \, \text{\normalfont d}\mathrm{Vol}_i \longrightarrow \int_M d_t \Delta \varphi \,\text{\normalfont d}\mathrm{Vol}.
    \end{align*}
    Noting that $\lim_{i\to\infty}\sqrt{\delta_i}\coth(\sqrt{\delta_i} d_{i,t}) = \frac{1}{d_{t}}$ uniformly on $K$, we conclude that
    \begin{align*}
    \int_M d_t \Delta \varphi \, \text{\normalfont d}\mathrm{Vol} &= \lim_{i\to\infty} \int_M d_{i,t} \Delta_i \varphi \, \text{\normalfont d}\mathrm{Vol}_i \\
    &\leq \lim_{i\to\infty} (n-1) \int_M \sqrt{\delta_i} \coth(\sqrt{\delta_i} d_{i,t}) \varphi \, \text{\normalfont d}\mathrm{Vol}_i
    = \int_M \frac{n-1}{d_t} \varphi \, \text{\normalfont d}\mathrm{Vol}.    
    \end{align*}
    Note that for any $t \in \mathbb R$, $\int_M t \Delta \varphi \, \text{\normalfont d}\mathrm{Vol} = \int_M (\Delta t) \varphi \, \text{\normalfont d}\mathrm{Vol} = 0$, so
    \begin{align*}
        \int_M (t-d_t) \Delta \varphi \, \text{\normalfont d}\mathrm{Vol} \geq - \int_M \frac{n-1}{d_t} \varphi \, \text{\normalfont d}\mathrm{Vol}.
    \end{align*}
    The LHS converges to $\int_M b \Delta \varphi \, \text{\normalfont d}\mathrm{Vol}$ and the RHS to $0$ for $t \to \infty$, so we get that $b^+$ is subharmonic. An analogous proof establishes the same property for $b^-$.

\label{rk:distribLaplacian}
    Let us briefly note that $f\in C^{0,1}(M)$ being \emph{harmonic (subharmonic, superharmonic) in the distributional sense}, i.e.
    \begin{equation*}
        \int_M f \Delta \varphi \,\text{\normalfont d}\mathrm{Vol} =0 \quad(\text{resp. }\geq 0,\, \leq 0) \qquad \text{for all non-negative }\varphi\in C^\infty_c(M),
    \end{equation*}
    is equivalent to
    \begin{equation*}
        \Delta (f\circ \psi^{-1}) = \frac{1}{\sqrt{\det g}}\partial_\alpha ( \sqrt{\det g} \,g^{\alpha\beta} \partial_\beta( {f\circ\psi^{-1}})) =0 \quad(\text{resp. }\geq 0,\,\leq 0)  \qquad \text{in }\mathcal{D}'(\psi(U))
    \end{equation*}

Let $\varphi\in C^\infty_c(M)$ be non-negative and, without loss of generality, we can assume $\operatorname{supp} \varphi$ is fully contained in $U$. Let us denote $\Bar{f}:=f\circ\psi^{-1},\,\Bar{\varphi}:=\varphi \circ \psi^{-1}$. Then we have
    \begin{align*}

<!-- theorem-5: body_md -->
\label{thm:C1hopfrinow}
     Let $(M,g)$ be a connected Riemannian manifold with $g\in C^1(T^0_2 M)$. Then the following are equivalent:
     \begin{enumerate}
         \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,d_g)$ is a complete metric space.
         \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,d_g)$ is a proper metric space, i.e.\ closed bounded sets are compact.
         \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,g)$ is geodesically complete, i.e.\ every geodesic $\gamma:(a,b)\to M$ can be extended to a geodesic $\bar{\gamma}:\mathbb R\to M$.
     \end{enumerate}
<!-- end theorem-5 -->
<!-- proof-8: body_md -->
The equivalence of the first two conditions is ensured by the Hopf-Rinow-Cohn-Vossen Theorem \cite[Thm.\ 2.5.28]{bubuiv2001metric} (in particular this equivalence holds even for continuous metrics, cf.\ \cite{Burtscher_length}). Due to the same theorem, we also have that the first two conditions are equivalent to
     \begin{equation*}

<!-- proof-12: body_md -->
Since $\operatorname{Hess} b=0$, also $\nabla \nabla b=0$ and thus the integral curves of $\nabla b$ are geodesics. By Theorem \ref{thm:C1hopfrinow}, all integral curves of $\nabla b$ are defined on all of $\mathbb R$. Since $b \in C^2(M)$ and $\abs{\nabla b}=1$, all level sets $b^{-1}(t)$ are embedded $C^2$-hypersurfaces.\\
    Recall that for $(t,p)\in \mathbb R\times M'$, the flow $\text{\normalfont Fl}_t^{\nabla b} (p)$ of the vector field $\nabla b$ denotes the point $\gamma(t)\in M$, where $\gamma$ is the solution of
    \begin{equation*}
        \begin{cases}
            \Dot\gamma(t) = \nabla b(\gamma(t)),\\
            \gamma(0)=p.
        \end{cases}
    \end{equation*}
    It is a well-known consequence of the flow box theorem (\cite[Prop.\ 4.1.13]{Abraham_Marsden_Ratiu}) that $(t,p)\mapsto \Phi(t,p):= \text{\normalfont Fl}^{\nabla b}_t(p)$ is a $C^1$-map $\Phi:\mathbb R \times M' \to M$, while $t \mapsto \text{\normalfont Fl}^{\nabla b}_t(p)$ is $C^2$ for every fixed $p \in M$.\\
    Note that for $t \in \mathbb R$ and $p \in M'$
        \begin{equation*}
            b(\text{\normalfont Fl}_t^{\nabla b} (p) )=t,
        \end{equation*}and consequently
        \begin{equation*}
            \text{\normalfont Fl}_t^{\nabla b} (M' )= b^{-1}(t).
        \end{equation*}
    This immediately implies that $\Phi$ is a $C^1$-diffeomorphism. Let us first check that $\Phi$ is a $C^1$-isometry. Observe that
    \begin{equation}\label{eq:from_here}
       T_{(t,p)}\Phi(a,0) = a \nabla b(\Phi (t,p)).
    \end{equation}
    Fix $p\in M'$. Since $\nabla b(\Phi(t,p)) \perp T_{\Phi(t,p)} (\{b=t\})$ for each $t$, for any $(a,v)\in \mathbb R \times T_{p} M$ we have 
    \[g(T_{(t,p)}\Phi(a,v), T_{(t,p)}\Phi(a,v)) = a^2 |\nabla b(t)|^2 + |T_{(t,p)}\Phi(0,v)|^2
    = a^2 + |T_{(t,p)}\Phi(0,v)|^2.
    \] 

    Thus we may conclude the proof by showing that $\abs{T_{(t,p)}\Phi(0,v)}^2$ is independent of $t$ for any given $v\in T_{p}M'$ (because for $t=0$ it holds that $T_{(0,p)} \Phi (0,v) = v$). 
    On $\mathbb R\times M'$ let us index with latin letters the coordinates on $\{0\}\times M$ to distinguish them from the flow parameter $t$. Greek letters will be reserved to coordinates on $M$. Locally, by definition, we have
    \begin{align*}
        \partial_t \Phi^\beta (t,p) = (\nabla b)^\beta (\Phi(t,p))
    \end{align*}
    and by symmetry of the second distributional derivatives
    \begin{align*}
        \partial_t\partial_a \Phi^\beta (t,p)=\partial_a\partial_t \Phi^\beta (t,p)= \partial_a[(\nabla b)^\beta (\Phi(t,p))] = \partial_a \Phi^\mu(t,p) \partial_\mu(\nabla b)^\beta (\Phi(t,p)).
    \end{align*}
    Given that these mixed second derivatives are distributions represented by continuous functions, by \cite[Th.\  2.1.2]{Friedlander} the derivatives exist also in the classical sense. Now the following computation is admissible due to the fact that the products are merely multiplications between functions.
    \begin{align*}
        \dv{}{t} \abs{T_{(t,p)}\Phi(0,v)}^2&=\partial_t( g_{\alpha\beta}(\Phi(t,\,.\,))\partial_j\Phi^\alpha v^j\partial_k\Phi^\beta v^k) \\
        &=\partial_t\Phi^\mu \partial_\mu g_{\alpha\beta}\partial_j\Phi^\alpha v^j\partial_k\Phi^\beta v^k +
        2g_{\alpha\beta}\partial_t \partial_j\Phi^\alpha v^j\partial_k\Phi^\beta v^k\\
        &=(\nabla b)^\mu (\Gamma^\sigma_{\mu\alpha}g_{\sigma\beta} + \Gamma^\lambda_{\mu\beta}g_{\alpha\lambda})\partial_j\Phi^\alpha v^j\partial_k\Phi^\beta v^k +
        2g_{\alpha\beta}\partial_j \Phi^\nu \partial_\nu(\nabla b)^\alpha  v^j\partial_k\Phi^\beta v^k\\
        &=2 g(\nabla_{T_{(t,p)}\Phi(0,v)} \nabla b, T_{(t,p)}\Phi(0,v) ) \\
        &= 2(\operatorname{Hess} b)(T_{(t,p)}\Phi(0,v),T_{(t,p)}\Phi(0,v))\\
        &=0.
    \end{align*}
We conclude that $\Phi$ is a $C^1$-isometry. At this point, the regularity of the embedding of $M'$ in $M$ is no longer relevant. We endow $M'$ with a $C^2$-compatible $C^\infty$-structure (cf.\ \cite[Thm.\ 2.9]{Hirsch}). Then $\Phi$ can be understood as a $C^1$-isometry of $C^1$-metrics on smooth manifolds. Thus, a result of Taylor \cite[Thm.\ 2.1]{taylor2006existence} asserting the $C^2$-regularity of distance-preserving maps between $C^1$-Riemannian metrics concludes the proof.

\label{Proposition: Extendingtoparallelvf}
Let $(M,g)$ be a $C^1$ semi-Riemannian manifold satisfying $\mathrm{Riem}_g = 0$. Then for any $p \in M$ and $v \in T_pM$, there exists an open neighborhood $U$ around $p$ and a parallel vector field $V \in C^1(TU)$ with $V_p = v$.
\begin{proof}
As this is a local question, suppose $M=\mathbb R^n$ and $p=0$. Consider an open cube $\{x \in \mathbb R^n \mid |x^j| < \eta \quad \forall j\} =: C_{\eta}$ around $0$.\par 
We start with $v \in T_pM$ and parallel transport it along the $x^1$-axis (inside $C_{\eta}$) with respect to $g_{\varepsilon}$, then from each point on the $x^1$-axis we parallel transport the resulting vector field along the $x^2$-axis (again with respect to $g_{\varepsilon}$) and so on. By smooth dependence on initial data and the right hand side in the parallel transport equation, this construction yields a smooth vector field $V_{\varepsilon}$ on $C_{\eta}$. By construction, $\nabla^{\varepsilon}_{\partial_1} V_{\varepsilon} = 0$ along the $x^1$-axis in $C_{\eta}$, $\nabla^{\varepsilon}_{\partial_2} V_{\varepsilon} = 0$ in the $x^1x^2$-plane in $C_{\eta}$, and so on (where $\nabla^{\varepsilon}$ denotes the Levi-Civita connection with respect to $g_{\varepsilon}$). Generally, on $M_k:=\{x^{k+1} = \dots = x^n = 0\} \cap C_{\eta}$, we have $\nabla^{\varepsilon}_{\partial_k} V_{\varepsilon} = 0$. Let us now determine, e.g., $\nabla^{\varepsilon}_{\partial_1} V_{\varepsilon}$ in $M_2$. For this, we calculate
\begin{align*}
    \nabla^{\varepsilon}_{\partial_2}\nabla^{\varepsilon}_{\partial_1} V_{\varepsilon} = \nabla^{\varepsilon}_{\partial_1}\nabla^{\varepsilon}_{\partial_2} V_{\varepsilon} + \mathrm{Riem}_{\varepsilon}(\partial_2,\partial_1)V_{\varepsilon}.
\end{align*}
The first term vanishes in $M_2$, so we are left with estimating the curvature term. We take $K:=\overline{C_{\eta/2}}$, then, 
by the above observation
$V_\varepsilon$ converges in $C^0$ to the corresponding solution of the $g$-parallel transport equation. In particular there exists $C >0$ such that on $K$ (with $|\, .\,|$ denoting the Euclidean norm)
\begin{align*}
    |V_{\varepsilon}| \leq C \quad \forall \varepsilon.
\end{align*}
Now let $\delta > 0$ be arbitrary, then by Proposition \ref{Proposition: SmallnessofapproxRiemann} there exists $\varepsilon_0$ such that for all $\varepsilon < \varepsilon_0$ and all $1\le i,j \le n$, we have
\begin{align*}
    |\mathrm{Riem}_{\varepsilon}(\partial_i,\partial_j)V_{\varepsilon}| < \delta \quad \text{on } K.
\end{align*}
This shows that $\nabla^{\varepsilon}_{\partial_1} V_{\varepsilon}$ is almost $g_{\varepsilon}$-parallel along the $x^2$-lines. Now let $W_{\varepsilon}$ be the vector field in $M_2$ that arises by $g_{\varepsilon}$-parallel transporting $\nabla^{\varepsilon}_{\partial_1} V_{\varepsilon}$ from points on the $x^1$-axis along the $x^2$-lines. We will now make precise that $W_{\varepsilon}$ and $\nabla^{\varepsilon}_{\partial_1}V_{\varepsilon}$ are close. Since the difference of $W_{\varepsilon}$ and $\nabla^{\varepsilon}_{\partial_1} V_{\varepsilon}$ is the solution of the parallel transport equation with an additional error term $\mathrm{Riem}_{\varepsilon}(\partial_2,\partial_1)V_{\varepsilon}$ which tends to $0$ uniformly, the difference $W_{\varepsilon} - \nabla^{\varepsilon}_{\partial_1}V_{\varepsilon}$ converges in $C^0$ to the zero vector field.
But then necessarily, $\nabla^{\varepsilon}_{\partial_1}V_{\varepsilon}$ converges to the $g$-parallel transport along the $x^2$-lines of the vectors $\nabla_{\partial_1}^{\varepsilon}V_{\varepsilon}$ evaluated along the $x^1$-axis (because $W_{\varepsilon}$ does so). Moreover, by construction $\nabla^{\varepsilon}_{\partial_1}V_{\varepsilon} = 0$ on the $x^1$-axis, hence $W_{\varepsilon} = 0$, so $\nabla^{\varepsilon}_{\partial_1}V_{\varepsilon} \to 0$ in $M_2$. Iterating this procedure for all derivatives in all coordinate directions, one sees that $\nabla_{\partial_k}^{\varepsilon} V_{\varepsilon}$ is uniformly small on $C_{\eta/2}$. 
By construction, $V_{\varepsilon} \to V$ in $C^0$, where $V$ is the $C^1$-vector field obtained from $g$-parallel transporting $v$ along $x^1$, then along $x^2$ and so on (note that the $V$ so constructed is indeed $C^1$ by \eqref{eq:eps_parallel_transport}).

$\nabla^{\varepsilon}_{\partial_k}V_{\varepsilon} \to \nabla_{\partial_k} V$, and we obtain that $V$ is $g$-parallel in $C_{\eta/2}$, concluding the proof.
\end{proof}

\label{Theorem: FlatnesscriterionC1}
Let $(M,g)$ be an $n$-dimensional $C^1$ semi-Riemannian manifold of signature $(k,l)$. Then the following are equivalent:
\begin{enumerate}
    \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,g)$ is \textit{(rank one) distributionally curvature-flat}, i.e.\ $\mathrm{Riem}_g = 0 \in \mathcal{D}^{'(1)}(T^1_3 M)$.
    \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,g)$ is \textit{$C^1$ frame-flat}, i.e.\ for each $p \in M$ there exists a neighborhood $U$ of $p$ and an orthonormal frame $E_1,\dots,E_n$ on $U$ of regularity $C^1$ consisting of parallel vector fields.
    \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,g)$ is \textit{$C^2$ coordinate-flat}, i.e.\ for each $p \in M$ there is a coordinate neighborhood $(U,\varphi)$ containing $p$ such that the coordinate map $\varphi: U \to \varphi(U) \subseteq \mathbb R^n$ is a $C^2$-diffeomorphism from $U$ onto an open subset of $\mathbb R^n$ satisfying $\varphi^* g_{k,l} = g$, where $g_{k,l}$ is the semi-Euclidean metric of signature $(k,l)$.

We define

We can conclude that $\norm{f}_{W^{1,2}(M)} = \norm{f}_{W^{1,2}_*(M)}$.\\
    Observe that $C^{0,1}_c(M)$ is dense in both spaces: in $W^{1,2}(M)$ due to mollification and in $W^{1,2}_*(M)$ due to \cite[Lem. 6.2.12]{gigpas2020notes}, recalling that bounded supports are compact due to the completeness of $(M,g)$. \label{thm:RCDtodistributional}

<!-- proof-24: body_md -->
Take an arbitrary $X\in C^\infty_c(TM)\subseteq H^{1,2}_H(TM)$.  Due to their compact support and their $W^{-1,\infty^-}_\text{\normalfont loc}$ regularity, Remark \ref{rk:W-1pMultipl} ensures that we are allowed to evaluate all the following distributions on the non-compactly supported canonical volume density $\mathrm{Vol}\in C^{0,1}_{\mathrm{loc}}(\mathrm{Vol}\, M) =W^{1,\infty}_\text{\normalfont loc}(\mathrm{Vol}\, M)\subseteq W^{1,\infty^-}_\text{\normalfont loc}(\mathrm{Vol}\, M)$. First of all
    \begin{align*}
        \langle \Delta \abs{X}^2, \mathrm{Vol}\rangle = -\int_M g \left(\nabla \abs{X}^2, \nabla 1 \right) \,\text{\normalfont d}\mathrm{Vol}=0.
    \end{align*}
    Then by \cite[Thm. 6.5.1]{gigpas2020notes} and Theorem \ref{thm:BochWeitz} the following holds:
    \begin{align*}
        \bm{{\mathrm{Ric}}}(X,X)(M) &=\int_M \vert{\delta X^\flat}\vert^2 + \vert{\text{\normalfont d} X^\flat}\vert^2 - \abs{\nabla X}^2\,\text{\normalfont d}\mathrm{Vol}\\
        &=\left\langle \frac{1}{2} {\Delta}\abs{X}^2 +  g(\Delta_H X, X )- \abs{\nabla X}^2 ,\mathrm{Vol}\right\rangle \\
        &=\left\langle\mathrm{Ric}(X,X),\mathrm{Vol}\right\rangle.
    \end{align*}
    We also know from \cite[Thm. 6.5.1]{gigpas2020notes} that, for all $Y\in H^{1,2}_H(TM)$,
    \begin{align*}
        \bm{{\mathrm{Ric}}}(Y,Y) (M)\geq K \int_M \abs{Y}^2 \,\text{\normalfont d}\mathrm{Vol}.
    \end{align*}
    Given a non negative $\varphi \in C^\infty_c(M)$, we can always represent it as a finite sum of squares of $C^{1,1}_c(M)\subseteq W^{1,\infty^-}_\text{\normalfont loc}(M)$ functions, $\varphi = \sum_{i=1}^m \psi_i^2$. 
    A proof of this fact can be found in \cite[Lem. 4]{guan97mongeampere}. Then the following multiplication is justified by Remark \ref{rk:W-1pMultipl}: 
    \begin{align*}
        \left\langle\mathrm{Ric}(X,X),\varphi\right\rangle =  \left\langle (\sum_i \psi^2_i)\mathrm{Ric}(X,X),\mathrm{Vol}\right\rangle =  \sum_i \left\langle\mathrm{Ric} (\psi_i X, \psi_i X),\mathrm{Vol}\right\rangle.
    \end{align*}
    Due to \cite[Lem. 3.3]{bregig2024calculus},  we also know that $\psi_i X \in H^{1,2}_H(TM)$. We can thus conclude
    \begin{align*}
        \left\langle\mathrm{Ric}(X,X),\varphi\right\rangle &= \sum_i \left\langle\mathrm{Ric} (\psi_i X, \psi_i X),\mathrm{Vol}\right\rangle
        = \sum_i \bm{{\mathrm{Ric}}}(\psi_i X, \psi_i X) (M)\\
        &\geq K \sum_i \int_M \abs{\psi_i X}^2 \,\text{\normalfont d}\mathrm{Vol}
        = K \int_M \varphi \abs{X}^2 \,\text{\normalfont d}\mathrm{Vol}.
    \end{align*}

We call \emph{metric measure space} any triple $(X,d,\mathfrak{m})$ such that
    \begin{enumerate}
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(X,d)$ is a complete, separable metric space,
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $\mathfrak{m}$ is a Radon measure on $X$.

<!-- definition-4: body_md -->

        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $\sup_{y,z\in B_R(x_n)} \abs{d_n (y,z) - d( f^{R,\varepsilon}_n(y), f^{R,\varepsilon}_n(z))} < \varepsilon$,
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} the $\varepsilon$-neighborhood of $f^{R,\varepsilon}_n(B_R(x_n))$ contains $B_{R-\varepsilon}(x)$,
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(f^{R,\varepsilon}_n)_* (\mathfrak{m}_n\vert_{B_R(x_n)})$ weakly converges to $\mathfrak{m}\vert_{B_R(x)}$, i.e.
        \begin{gather*}
            \int_{B_R(x_n)} \varphi\circ f^{R,\varepsilon}_n \;\text{\normalfont d} \mathfrak{m}_n \longrightarrow \int_{B_R(x)}\varphi \;\text{\normalfont d} \mathfrak{m} \\
            \text{for all bounded }\varphi\in C(X) \text{ with bounded support,}
        \end{gather*}
        for a.e. $R>0$.
    \end{enumerate}

Let $n=\dim M$. Consider as before $g_i= g_{\varepsilon_i}$ as in Remark \ref{Remark: Approximatingmetrics} for some $\varepsilon_i\searrow 0$. It follows from Theorem \ref{thm:C1hopfrinow}, together with \eqref{eq:balls_g_geps} that  $(M,g_i)$ is a complete Riemannian manifold for every $i$. Observe that for all $i\in \mathbb N$ one can define the continuous function $\kappa_i:M\to \mathbb R $ assigning to every $p\in M$ the lowest eigenvalue of $\mathrm{Ric}_i (p)$ with respect to $g_i(p)$. Clearly, it holds that $\mathrm{Ric}_i \geq \kappa_i g_i$. 
    
    Fix any $x_0\in M$. In order to apply Theorem \ref{thm:kettererRiemannCD} we need to verify that, for some $p>\frac{n}{2}$,
    \begin{align}\label{eq:ketterer_to_verify}
        \frac{R^{2p}}{\mathrm{Vol}_i (B_1^i(x_0) )} \int_{B_R^i(x_0)} \left[ (\kappa_i - K)^- \right]^p \,\text{\normalfont d}\mathrm{Vol}_i\xrightarrow{i\to\infty}0 \qquad \forall R>0.
    \end{align}
To see this, note first that, by \eqref{eq:balls_g_geps},
for any fixed $R>0$ and all $i\in \mathbb N$ we have $B_R^i(x_0) \subseteq \overline{B_{2R}(x_0))} \Subset M$, and since $g_i\to g$ locally uniformly, for large $i$ we additionally have
$\mathrm{Vol}_i (B_1^i(x_0) )$ $\ge$ $\mathrm{Vol}_i(B_{1/2}(x_0)) \ge \frac{1}{2}\mathrm{Vol}(B_{1/2}(x_0))$. Thus to establish
\eqref{eq:ketterer_to_verify}, it remains to show that $(\kappa_i - K)^- \to 0$ in $L^p_{\mathrm{loc}}(M)$. Fix some $L\Subset M$, and without loss of generality suppose that $L$ is small enough to admit a smooth local frame $X_1,\dots,X_n$ in a neighborhood $V$ of $L$. Also, fix $\bar x\in L$. Then according to the Rayleigh quotient formula, for any $i$ there exists some $g_i$-unit vector $\bar X^{(i)}\in T_{\bar x}M$ with $(\kappa_i(\bar x)-K) = \mathrm{Ric}_{g_i}(\bar X^{(i)},\bar X^{(i)})-K$. 
Let $\bar X^{(i)}=\sum_{k=1}^n \alpha_k^{(i)} X_k(\bar x)$ ($\alpha_1^{(i)},\dots,\alpha_n^{(i)}\in \mathbb R$) and set
\begin{equation}\label{eq:Xidef}
    X^{(i)} := \sum_{k=1}^n \alpha_k^{(i)} X_k
\end{equation}
on $V$. Since $\bar X^{(i)}$ is a $g_i$-unit
vector it follows from \eqref{geps3} and the local equivalence of $\|\,\|_g$ with the Euclidean norm w.r.t.\ the frame $X_k$ that
there exists some
$C_L>0$ such that
\begin{equation}\label{eq:alpha_bounds}
|\alpha_k^{(i)}| \le C_L \qquad \forall i\in \mathbb N,\ k=1,\dots,n,
\end{equation}
independently of the choice of $\bar x\in L$ and $\bar X^{(i)}$ as above.
On $L$ we have
\begin{equation}\label{eq:Ric_i_decomposition}
\begin{split}
\mathrm{Ric}_i(X^{(i)},X^{(i)}) -& Kg_i(X^{(i)},X^{(i)}) = 
[\mathrm{Ric}_i - (\mathrm{Ric}\star_M \rho_i)](X^{(i)},X^{(i)}) \\
&+[(\mathrm{Ric}\star_M \rho_i)(X^{(i)},X^{(i)}) - \mathrm{Ric}(X^{(i)},X^{(i)})\star_M\rho_i]\\
&+[\mathrm{Ric}(X^{(i)},X^{(i)}) - Kg(X^{(i)},X^{(i)})]\star_M\rho_i \\
&+ [Kg(X^{(i)},X^{(i)})\star_M\rho_i - Kg_i(X^{(i)},X^{(i)})] \\
&=: A_i(X^{(i)},X^{(i)}) + B_i(X^{(i)},X^{(i)}) + C_i(X^{(i)},X^{(i)}) + D_i(X^{(i)},X^{(i)}).
\end{split}
\end{equation}
Since $\mathrm{Ric}(X,X)\geq K g(X,X)$ in distributions and $\rho\ge 0$, $C(X^{(i)},X^{(i)}) \ge 0$. Therefore,
\[
[\mathrm{Ric}_i(X^{(i)},X^{(i)}) - Kg_i(X^{(i)},X^{(i)})]^- \le |A_i(X^{(i)},X^{(i)})| + |B_i(X^{(i)},X^{(i)})| + |D_i(X^{(i)},X^{(i)})| 
\]
pointwise on $L$. Set $\hat A_i := \max\{|A_i(X_k,X_l)| \colon k,l = 1,\dots,n \}$, and analogously define $\hat B_i$ and
$\hat D_i$.
Inserting \eqref{eq:Xidef} into \eqref{eq:Ric_i_decomposition} and taking into account \eqref{eq:alpha_bounds}, it follows that there exists a constant $\tilde C_L$ such that
\[
[\mathrm{Ric}_i(X^{(i)},X^{(i)}) - Kg_i(X^{(i)},X^{(i)})]^- \le
\tilde C_L(\hat A_i + \hat B_i + \hat D_i),
\]
pointwise on $L$. 
Since all of the above arguments apply irrespective of the choice of $\bar x$ and $\bar X^{(i)}$ we conclude that $[\kappa_i-K]^-\le
\tilde C_L(\hat A_i + \hat B_i + \hat D_i)$, pointwise on $L$.
Proposition
\ref{prop:LpRicConvergence} shows that both $\hat A_i$ and $\hat B_i$ converge to $0$ in $L^p(L)$. Moreover, $\hat D_i\to 0$ uniformly on $L$. Altogether, this establishes \eqref{eq:ketterer_to_verify}.
 
From this, by applying Theorem \ref{thm:kettererRiemannCD}, we obtain that $\{[M, d_i, \mathfrak{m}_i, x_0]\}_{i\in\mathbb N}$ admits a subsequence that converges in the pmG-sense to the isomorphism class of a $\mathsf{CD}(K, \dim M)$ space. At the same time, due to the convergence $g_i\to g$ in $C^0_\text{\normalfont loc}$, we can easily see that $(M, d_i , \mathfrak{m}_i, x_0) \to (M,d,\mathfrak{m},x_0)$, where $\mathfrak{m}_{(i)}:= \mathrm{Vol}_{(i)}(B_1(x_0))^{-1} \mathrm{Vol}$, in the pmGH-sense. By Theorem \ref{thm:GMSpmG}, also their isomorphism classes converge in the pmG-sense. Due to uniqueness of the limits, and the fact that the $\mathsf{CD}(K,N)$ condition is invariant under isomorphisms of metric measure spaces, we have that $(M,d,\mathfrak{m},x_0)$ is a $\mathsf{CD}(K,N)$-space. Also the Riemannian structure of $(M,g)$ naturally ensures the infinitesimal Hilbertianity of $(X,d,\mathfrak{m})$ (cf.\ also the discussion on Sobolev spaces from the previous section), making it an $\mathsf{RCD}(K,N)$-space. Since $\mathrm{Vol}$ is simply a rescaling of $\mathfrak{m}$ and the $\mathsf{RCD}(K,N)$-condition is invariant under this operation, the claim follows.

<!-- proposition-14: body_md -->
Let $(M,g)$ be a complete, connected Riemannian manifold with $g\in C^{0,1}_\text{\normalfont loc}(T^0_2 M)$ and suppose that $\mathrm{Ric}\ge K$ in the distributional sense for some $K\in \mathbb R$. Then the following properties hold:
    \begin{enumerate}
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} Given any $p\in M$, for $\mathrm{Vol}$-a.e. $q\in M$ there exists a unique minimizing geodesic with endpoints $p$ and $q$.
        \refstepcounter{inlineenum}{(\roman{inlineenum})\hspace{\thelabelsep}} $(M,d)$ is a non-branching metric space, i.e.\ if there exist two minimizing geodesics $\gamma,\Tilde{\gamma}:[0,1]\to M$ and $t\in(0,1)$ such that $\gamma\vert_{[0,t]} = \Tilde{\gamma}\vert_{[0,t]}$, then $\gamma = \Tilde{\gamma}$.
    \end{enumerate}

<!-- theorem-13: body_md -->
Let $(X,d,\mathfrak{m})$ be an $\mathsf{RCD}(0,N)$ space with $1\leq N <\infty$. Assume $\operatorname{supp} \mathfrak{m}$ contains a line, i.e. a curve $c:\mathbb R \to X$ such that
    \begin{equation*}
        d(c(t),c(s))= \abs{t-s}\qquad \text{for all }t,s\in\mathbb R.
    \end{equation*}
    Then there exists a metric measure space $(X',d',\mathfrak{m}')$ such that $X$ is isomorphic to $X'\times \mathbb R$ endowed with the product measure $\mathfrak{m}'\times\mathcal{L}^1$ and the product distance
    \begin{equation*}
        d_{X'\times \mathbb R}((x',t)(y',s)):= \sqrt{d'(x',y')^2 + \abs{t-s}^2 }.
    \end{equation*}
    Furthermore if $N\geq 2$, then $(X',d',\mathfrak{m}')$ is an $\mathsf{RCD}(0,N-1)$ space. For $N\in [1,2)$, $X'$ is a singleton.
