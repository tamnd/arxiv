A spectral triple considered in the present paper, named as {\it modular spectral triple} according to the terminology in \cite{CM}, is a triplet ${\mathcal T}:=(\omega,{\mathcal L},{\mathcal E})$, where with a slight abuse of notation we write ${\mathcal E}\equiv{\mathcal E}_{\mathcal T}$, made of a modular state $\omega\in{\mathcal S}({\mathfrak A})$, an

First we put ${\mathcal L}^\star:=-*{\mathcal L}\,*$ and, for $a\in {\mathfrak A}$, we denote by $a_\omega\in {\mathcal H}_\omega$ the natural embedding of $a$ in ${\mathcal H}_\omega$, i.e.\ $a_\omega=\pi_\omega(a)\xi_\omega$. 
\begin{itemize}
\item [(i)]  With ${\mathcal H}_\omega\supset {\mathcal A}_\omega\ni a_\omega\mapsto L_\omega^\#a_\omega:=({\mathcal L}^\#a)_\omega\subset{\mathcal H}_\omega$,

\begin{equation}
\label{spqu1}
D^{(o)}_{\mathcal T}:=\begin{pmatrix} 
	 0 &\Delta_\omega^{-1}L_\omega\\
	L_\omega^\star\Delta_\omega^{-1}& 0\\
     \end{pmatrix} 
\end{equation}
acting on its own domain ${\mathcal D}_{D^{(o)}_{\mathcal T}}\subset{\mathcal H}_\omega\bigoplus{\mathcal H}_\omega$ uniquely defines a self-adjoint operator with compact resolvent: $D^{(o)}_{\mathcal T}$ is densely defined essentially self-adjoint

\item[(ii)] There exists a dense $*$-subalgebra ${\mathcal B}\subset{\mathfrak A}$, and a set ${\bf P}=(P_\iota)_{\iota\in{\mathcal I}}\subset{\mathcal B}({\mathcal H}_\omega)$ of (increasing) self-adjoint projections with

  D_{\mathcal T}\xi=D_{\mathcal T}^{(o)}\xi\in {\mathcal D}_{\Gamma_{\mathcal T}^{-1}\underline{A}\Gamma_{\mathcal T}}\subset{\mathcal D}_{\overline{\Gamma_{\mathcal T} \underline{A} \Gamma_{\mathcal T}^{-1}}} = {\mathcal H}_\omega \bigoplus {\mathcal H}_\omega \  ,

 \Gamma_{\mathcal T}^{-1} \underline{A}\Gamma_{\mathcal T}\, D_{\mathcal T}^{(o)} \xi =  \overline{\Gamma_{\mathcal T}^{-1} \underline{A}\Gamma_{\mathcal T}}\, D_{\mathcal T} \xi \ .

provides a compact operator for each $A\in{\mathcal E}_{\mathcal T}$.

<!-- thm-5: body_md -->
The real number $\lambda$ is an eigenvector of the deformed Dirac operator \eqref{ndsd} if and only 
if there exists $n\in{\mathbb Z}$ such that the homogeneous Hill equation
\begin{equation}
\label{ndsd3}
H''+\big(\lambda^2d_n^{2}(\theta)-n^2\big)H=0
\end{equation}
admits a periodic $C^2$ (and therefore necessarily $C^\infty$) nontrivial solution.
<!-- end thm-5 -->
<!-- proof-7: body_md -->
Taking into account \eqref{ndsd}, $\lambda$ is an eigenvalue of $D_\omega$ if, for some $n$, there exist non vanishing periodic functions $f,g$ on the unit circle, necessarily smooth, satisfying
\begin{equation}
\label{ndsd1}
\left\{\begin{array}{ll}
                     \!\!\!\!\!\!\!&\big(\imath z\frac{{\rm d}\,\,}{{\rm d} z}-n I\big)g=\lambda\delta_n f\,, \\[1ex]

               	\!\!\!\!\!\!\!&\big(-\imath z\frac{{\rm d}\,\,}{{\rm d} z}-n I\big)\delta_n^{-1}f=\lambda g\,.
                    \end{array}
                    \right.
\end{equation}

By replacing $g$ from the 2nd equation in the 1st of \eqref{ndsd1}, and defining $h:=\delta_n^{-1}f$, we obtain
\begin{equation*}
              
\Big(\Big(z\frac{{\rm d}\,\,}{{\rm d} z}\Big)^2+n^2\Big)h=\lambda^2\delta_n^{2}h\,.
\end{equation*}

By passing to $H(\theta):=h(e^{\imath\theta})$ and $d_n(\theta)=\delta_n(e^{\imath\theta})$, it becomes
$$
\Big(-\frac{{\rm d}^2\,\,}{{\rm d}\theta^2}+n^2\Big)H=\lambda^2d_n^{2}H\,,
$$
which leads to the homogeneous Hill equation
\eqref{ndsd3}.
\medskip

Suppose now that \eqref{ndsd3} admits a non zero smooth periodic solution for some $\lambda\neq0$ and $n\in{\mathbb Z}$, the case $\lambda=0$ being trivially satisfied.

By putting $H(\theta)d_n(\theta)=f(z)$ with $z=e^{\imath\theta}$, and
\begin{equation}
\label{hcdo1}
g(z):=\frac1{\lambda}\big(-\imath z\frac{{\rm d}\,\,}{{\rm d} z}-n I\big)\delta_n^{-1}f(z)\,,
\end{equation}
an easy calculation shows that the vector-valued function 
\begin{equation*}
             
F(z):=\bigoplus_{k\in{\mathbb Z}}\delta_{n,k} f(z)\bigoplus g(z) \in {\mathcal H}_\omega\bigoplus{\mathcal H}_\omega
\end{equation*}
provides an eigenvector corresponding to the eigenvalue $\lambda$. 

Indeed, for $\lambda\neq0$ and $n\in{\mathbb Z}$, \eqref{ndsd1} reads
$\delta_n^{-1}L_ng=\lambda f$ and $L_n^\star(\delta_n^{-1} f)=\lambda g$, where $L_n$ and $L_n^\star$ are the differential operators given in \eqref{ellcl}.

We first note that \eqref{hcdo1} is nothing else than the first condition above. Concerning the second, by using the Hill equation we get
\begin{align*}
\delta_n^{-1}L_ng=\frac1{\lambda\delta_n}(L_nL_n^\star)\delta_n^{-1}f
=\frac1{\lambda d_n}\Big(-\frac{{\rm d}^2\,\,}{{\rm d}\theta^2}+n^2\Big)H=\lambda d_n H=\lambda f\,.
\end{align*}

Summarising,
\begin{align*}
D_\omega F=&\begin{pmatrix} 
	 0 &\Delta_\omega^{-1}L\\
	L^\star\Delta_\omega^{-1}& 0\\
     \end{pmatrix}F=
\left(\bigoplus_{n\in{\mathbb Z}}
     \begin{pmatrix} 
	 0 &M_{\delta^{-1}_n}L_n\\
	L_n^\star M_{\delta^{-1}_n}& 0\\
     \end{pmatrix}\right)F\\
     =&\begin{pmatrix} 
	 0 &M_{\delta^{-1}_n}L_n\\
	L_n^\star M_{\delta^{-1}_n}& 0\\
     \end{pmatrix}\begin{pmatrix} 
	 f\\
	g\\
     \end{pmatrix}=\lambda\begin{pmatrix} 
	 f\\
	g\\\end{pmatrix}=\lambda F\,,
\end{align*}
and this concludes the proof.
