\thlabel{THM:HRUSH}
Let $1\leq p< \infty$ and $E \subseteq \partial\mathbb{D}$ be a Lebesgue measurable set. Then the following statements are all equivalent: 
\begin{enumerate}
    \item[(i)] $E$ does not satisfy the SA-property wrt $L^p_a(\mathbb{D})$.
    \item[(ii)] $E$ satisfies the Khinchin-Ostrowski property wrt $L^p_a(\mathbb{D})$.
    \item[(iii)]
    There exists a compact set $K$ of positive Lebesgue measure, which is almost contained in $E$, such that $K$ has finite Beurling-Carleson entropy:
\[
\sum_k \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{I_k} \log \frac{1}{\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{I_k}} < \infty
\]
where $\{I_n\}_n$ denotes the connected components of $\partial\mathbb{D} \setminus K$.
\item[(iv)] There exists a non-trivial integrable function $\phi$ on $\partial\mathbb{D}$ that vanishes off $E$, such that its Cauchy transform
\[
C(\phi)(z) := \int_E \frac{\zeta \phi(\zeta) dm(\zeta)}{\zeta -z}, \qquad z\in \mathbb{D}
\]
extends to a non-constant function in $C^\infty(\partial\mathbb{D})$.
\end{enumerate}

\thlabel{THM:SAsets} Let $w$ be a majorant and $\mathcal{B}_0(w)$ denote the corresponding Bloch-type space. Then there exists a set of positive Lebesgue measure which satisfies the SA-property wrt $\mathcal{B}_0(w)$ if and only if  
\begin{equation}\label{DiniDiv}
\int_0^1 \frac{w^2(t)}{t}dt = \infty.
\end{equation}
If condition \eqref{DiniDiv} does not hold, then any set of positive Lebesgue measure is a set of rigidity for $\mathcal{B}_0(w)$.

\thlabel{THM:Bstruc} Let $w$ be a majorant and $E \subset \partial\mathbb{D}$ be a set of positive Lebesgue measure. Then there exists a unique (modulo sets of Lebesgue measure zero) partition into subsets $A,S$ of $E$, depending only on $E$ and $w$, for which the following statements hold:
\begin{enumerate}
    \item[(i)] $A$ is a set of rigidity for $\mathcal{B}_0(w)$. 
    \item[(ii)] $S$ satisfies the SA-property wrt $\mathcal{B}_0(w)$.
\end{enumerate}
Moreover, the following isomorphic identification holds:
\begin{equation}\label{EQ:CHAUM}
    \mathcal{P}_{E} \mathcal{B}_0(w) \cong \mathcal{P}_A \mathcal{B}_0(w) \oplus L^\infty(S).
\end{equation}
Modulo sets of Lebesgue measure zero, the measurable sets $A,S$ appearing in the partition of $E$ are conformally invariant.

\thlabel{THM:CHAR1} Let $w$ be a majorant satisfying \eqref{DiniDiv} and $E \subseteq \partial\mathbb{D}$ be a set of positive Lebesgue measure. Then following statements are all equivalent.
\begin{enumerate}
    \item[(i)] $E$ does not satisfy the SA-property wrt $\mathcal{B}_0(w)$.
    \item[(ii)] $E$ satisfies the weak KO-property wrt $\mathcal{B}_0(w)$.
    \item[(iii)] $E$ almost contains a non-trivial subset $A$ of positive Lebesgue measure, such that $A$ is a set of rigidity for $\mathcal{B}_0(w)$.
    \item[(iv)] There exists a non-trivial pair of real-valued functions $u, v \in L^1(E)$ such that 
\[
u + H(v) \in \mathscr{B}_w(\partial\mathbb{D}) \setminus \{0\},
\]
where $H$ denotes the Hilbert transform on $\partial\mathbb{D}$.

\end{enumerate}

Furthermore, $E$ is a set of rigidity for $\mathcal{B}_0(w)$ if and only if $E$ is a carrier set of at least one of the functions $u,v$ appearing in $(iv)$.

\thlabel{THM:CHARREM} Let $K \subset \partial\mathbb{D}$ be a compact set of positive Lebesgue measure. Then the following statements are equivalent:
\begin{enumerate}
    \item[(i)] $K$ is removable for the analytic Sobolev space $W^1_a$.
    \item[(ii)] There exists no non-trivial function $g \in B^1(\partial\mathbb{D})$ (or in $\mathscr{B}_1(\partial\mathbb{D}))$, which is supported in $K$.
    \item[(iii)] $C^\infty_K(\partial\mathbb{D})$ is norm-dense in the little Zygmund class $\lambda_1(\partial\mathbb{D})$.
\end{enumerate}
Furthermore, if $K$ is negligible for any $\Lambda_1$-condenser capacity in the sense that condition
\begin{equation}\label{REMB1}

\end{equation}
holds for any arc $I\subseteq \partial\mathbb{D}$, then $K$ is removable for $W^1_a$.

<!-- proof-1: body_md -->
Note that a straightforward application of the Littlewood-Paley formula gives 
    \[
    \int_{\partial\mathbb{D}}f(r\zeta) \overline{g(r\zeta)} dm(\zeta) = f(0) \overline{g(0)} + r^2 \int_{\mathbb{D}} f'(rz) \overline{g'(rz)} \log \frac{1}{|z|} dA(z), \qquad 0<r<1
    \]
    whenever $f, g$ are holomorphic functions in $\mathbb{D}$. It follows easily from this formula and some standard estimates that every $g \in \mathcal{B}(w)$ induces a bounded linear functional on $W^1_a(w)$ and that every $f \in W^1_a(w)$ induces a bounded linear functional on $\mathcal{B}_0(w)$. To show the converse, we primarily note that a straightforward argument involving dilatation shows that the analytic polynomials are dense in $W^1_a(w)$. The next step is to verify that if $g$ is holomorphic in $\mathbb{D}$ for which 
    \[
    \ell_g(f) := \lim_{r\to 1-} \int_{\partial\mathbb{D}} f(r\zeta) \overline{g(r\zeta)} dm(\zeta), \qquad f\in W^1_a(w)
    \]
    induces a bounded linear functional on $W^1_a(w)$, then $g$ must necessarily belong to $\mathcal{B}(w)$. To this end, we apply $\ell_g$ to the rational functions $f_{\lambda}(\zeta) = (1-\overline{\lambda}\zeta)^2$ with $\lambda \in \mathbb{D}$ fixed and utilize the Cauchy integral formula to get
    \[
    \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{g'(\lambda)} = \lim_{r\to 1-} \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{\int_{\partial\mathbb{D}}\frac{g(r\zeta)}{(1-\overline{r\zeta}\lambda)^2} dm(\zeta) } = \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{\ell_g(f_\lambda)} \leq C \@ifstar{\@ifstar{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}}{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}}{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}*}*}{f_{\lambda}}_{W^1_a(w)}.
    \]
    Here $C>0$ is a constant, which arises from the assumption $\ell_g \in W^1_a(w)'$ and is independent of $f$. It remains to compute the $W^1_a(w)$-norm of $f_{\lambda}$, which by polar coordinates takes the form 
    \[
    \@ifstar{\@ifstar{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}}{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}}{\@ifstar{\@ifstar{\norm}{\norm*}}{\@ifstar{\norm}{\norm*}*}*}*}{f_{\lambda}}_{W^1_a(w)} = 1 + \int_{0}^1  \int_{\partial\mathbb{D}} \frac{dm(\zeta)}{\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{\zeta - r\lambda}^3}  w(1-r) rdr.
    \]
    Now the integral on $\partial\mathbb{D}$ is asymptotically equal to $(1-r\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{\lambda})^{-2}$ (cf. Chap. I, \cite{hedenmalmbergmanspaces}). Using this in conjunction with a change of variable, and the assumption that $w$ is a majorant, one gets
    \[
    \int_{0}^1  \int_{\partial\mathbb{D}} \frac{dm(\zeta)}{\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{\zeta - r\lambda}^3}  w(1-r) rdr \lesssim \int_{0}^1 \frac{w(1-r)}{(1-r|\lambda|)^2} dr \leq \int_{1-\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{\lambda}}^1 \frac{w(t)}{t^2} dt \lesssim \frac{w(1-|\lambda|)}{1-|\lambda|}.
    \]
    We therefore conclude that $g \in \mathcal{B}(w)$, hence $W^1_a(w) ' \cong \mathcal{B}(w)$. Now it is a general result that $\mathcal{B}_0(w)'$ is the unique pre-dual of $\mathcal{B}(w)$ (cf. Corollary 1.3 in \cite{perfekt2017m}), hence using the previously established relation, we conclude that $\mathcal{B}_0(w)' \cong W^1_a(w)$.

Recall that the pre-dual of $C(L^1)$, considered in the Cauchy-dual pairing, can be identified with $H^\infty$. Note that the pre-annihilator $\mathcal{M}_\perp$ of $\mathcal{M}$, regarded as a weak-star closed subspace of $H^\infty$, is invariant under the shift operator $M_z$. An application of Beurling's Theorem in $H^\infty$ (c.f Theorem 7.5 in \cite{garnett}) implies that there exists a unique inner function $\Theta$, such that $\mathcal{M}_\perp = \Theta H^\infty$. It remains to verify that $\mathcal{M}$ is in the kernel of $T_{\overline{\Theta}}$. However, this claim follows from the observation that the formal transpose map of the multiplication with $\Theta$ on $H^\infty$ equals $T_{\overline{\Theta}}$ on $C(L^1)$.

\par
  \addvspace{\medskipamount} 
  \noindent\emph{Step 1: Dual reformulation of sets of rigidity:}\par\nobreak
  \addvspace{\smallskipamount} 
  \@afterheading


Note that if the set $\mathcal{L}_A(X')$ is empty, then Khrushchev's lemma implies that $A$ satisfies the SA-property wrt $X$, and hence $\mathcal{P}_A(X)$ cannot be irreducible. Now suppose that the set $\mathcal{L}_A(X')$ is non-empty and we shall in fact argue that it must be dense in $L^1(A)$. Note that since $X$ was assumed to be $M_z$-invariant, a straightforward computation of its formal transpose in $X'$ reveals that $M_z ': X' \to X'$. Furthermore, a routine computation involving Cauchy kernels shows that
\[
M_z ' C(h)(z) = C (\overline{\zeta}h)(z), \qquad z\in \mathbb{D}
\]
whenever $h\in L^1(\partial\mathbb{D})$, hence we conclude that $\mathcal{L}_A(X')$ is invariant under the operation $h \mapsto \overline{\zeta}h(\zeta)$. Now if the non-empty set $\mathcal{L}_A(X')$ is not dense in $L^1(A)$, then there exists an element $f\in L^\infty(A)$ which annihilates it, thus
\[
\int_{A} f(\zeta) \zeta^n \overline{h(\zeta)} dm(\zeta) = 0 \qquad n\geq 0
\]
for any $h \in \mathcal{L}_A(X')$. According to the F. and M. Riesz Theorem, $f \overline{h}1_{A}$ must agree with the boundary values of an $H^1$-function, but since $A$ does not have full Lebesgue measure on $\partial\mathbb{D}$, we conclude that $f\overline{h}=0$ on $A$. Since $h\neq 0$ $dm$-a.e on $A$, we are forced to conclude that $f=0$ identically as an element in $L^\infty(A)$, which is absurd and  $\mathcal{L}_A(X')$ is therefore dense in $L^1(A)$. Now let $(Q_n)_n$ be a sequence of analytic polynomials with the property that $Q_n \to 0$ in $X$ and $Q_n \to f$ weak-star in $L^\infty(A)$. Then, for any $h \in L^1(A)$ with the property that $C(h) \in X'$, we have 
\[
\int_A f \overline{h} dm = \lim_n \int_{A} Q_n \overline{h} dm = \lim_n \lim_{r\to 1-} \int_{\partial\mathbb{D}} Q_n(r\zeta) \overline{C(h)(r\zeta)} dm=0.
\]
Now since $\mathcal{L}_A(X')$ was dense in $L^1(A)$, we conclude that $f=0$ on $A$, hence $A$ is a set of rigidity for $X$.

 
  \par
  \addvspace{\medskipamount} 
  \noindent\emph{Step 2: Density of $\mathcal{K}_A(X')$:}\par\nobreak
  \addvspace{\smallskipamount} 
  \@afterheading
 We primarily note that since $X$ is $M_z$-invariant, an simple computation of the transpose of $M_z$ implies that the Cauchy dual $X'$ is $M'_z$-invariant. This means that $\mathcal{K}_A(X')$ regarded as a subset of $C(L^1)$ is $M_z '$-invariant. Now suppose that $\mathcal{K}_A(X')$ is not dense in $C(L^1)$, then according to \thref{LEM:Mz*K} there exists a non-trivial inner function $\Theta$ such that 
\[
\textbf{Clos}\left(\mathcal{K}_A(X') \right)_{C(L^1)} = \left\{ f= C(h): T_{\overline{\Theta}}(h) =0, \, \, h\in L^1(\partial\mathbb{D}) \right\}
\]
This implies that for any $h \in L^1(A)$ with $C(h) \in X'$ satisfies 
\[
\sum_{n\geq 0} z^n \int_{\partial\mathbb{D}} \overline{\Theta(\zeta)} \zeta^{-n} h(\zeta) dm(\zeta) = T_{\overline{\Theta}}(h)(z)  = 0  \qquad z\in \mathbb{D}.
\]
By the F. and M. Riesz Theorem, $\Theta\overline{h}$ agrees with the boundary values of an $H^1$-function, but since $h$ vanishes off the set $A$ (a set of positive Lebesgue measure) and $\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{\Theta} =1$ $dm$-a.e on $\partial\mathbb{D}$, we conclude that $h \equiv 0$ for any $h \in L^1(A)$ with $C(h) \in X'$. But this implies that $\mathcal{K}_A(X') =\{0\}$, which according to Khrushchev's Lemma implies that $A$ satisfies the SA-property wrt $X$, contradicting the initial assumption that $\mathcal{P}_A(X)$ was irreducible. We therefore conclude that $\mathcal{K}_A(X') = C(L^1)$ and the proof is complete.

<!-- thm-12: body_md -->
\thlabel{THM:AbsThom}
Let $X$ be a holomorphic Banach space and $E$ be Lebesgue measurable subset of $\partial\mathbb{D}$. Then there exists a unique (modulo sets of Lebesgue measure zero) partition of $E$ into subsets $A,S$ satisfying the following properties:
\begin{enumerate}
    \item[(i)] $A$ is a set of rigidity for $X$.
    \item[(ii)] $S$ satisfies the SA-property wrt $X$. 
\end{enumerate}
Moreover,the following isometric isomorphic identification 
\begin{equation}\label{absThomdec}
    \mathcal{P}_E(X) \cong \mathcal{P}_A(X) \oplus L^\infty(S).
\end{equation}
<!-- end thm-12 -->
<!-- proof-4: body_md -->
Since the statement is phrased modulo sets of Lebesgue measure zero, and sets of full Lebesgue measure are always sets of rigidity for $X$, we may assume that $0<|E|<1$. 
 
  \par
  \addvspace{\medskipamount} 
  \noindent\emph{Step 1: Constructing sets of rigidity:}\par\nobreak
  \addvspace{\smallskipamount} 
  \@afterheading


Consider the quantity 
\begin{equation}\label{maxirr}
    \iota(X,E) := \sup \left\{ |A|: A \subseteq E, \, \, \, \mathcal{P}_{A}(X) \, \, \text{is irreducible}  \right\}.
\end{equation}
We shall for a moment assume that the above set is non-empty. First we note that it is straightforward to verify that if $A_1, A_2$ are both sets of rigidity for $X$, then so is their union $A_1 \cup A_2$. We may therefore pick a chain of subsets $A_N \subseteq A_{N+1}$ of $E$ with the property that $\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{A_N}$ increase up to $\iota(X,E)$. Setting $A := \cup_N A_N$
we see that $\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{A} = \iota(X,E)$ and it remains to verify that $\mathcal{P}_A(X)$ is irreducible. To this end, let $\{Q_n\}_n$ be analytic polynomials which converge to $(0,f)$ in $X \oplus L^\infty(A)$. Now since each $\mathcal{P}_{A_N}(X)$ is irreducible it follows that $f = 0$ on each $A_N$. But since $\{A_N\}_N$ is an exhaustion of $A$ we get that $1_{A_N} f \to 1_A f$ pointwise on $\partial\mathbb{D}$, thus $f=0$ on $A$ and $\mathcal{P}_A(X)$ is therefore irreducible.

 
  \par
  \addvspace{\medskipamount} 
  \noindent\emph{Step 2: The subset satisfying the SA-property:}\par\nobreak
  \addvspace{\smallskipamount} 
  \@afterheading

Set $S := E \setminus A$ in case $A$ is non-trivial and $S=E$ in case $A$ is the empty set, 
and it remains to verify that $S$ satisfies the SA-property wrt $X$. Arguing by duality using \thref{SAduality}, suppose the exists an element $h \in L^1(S)$ such that $C(h)$ is a non-trivial element in $X'$. Now if $S_0$ denotes a carrier set for $h$, then it follows from \thref{THM:Irred} that $\mathcal{P}_{S_0}(X)$ is irreducible. But then $\mathcal{P}_{A \cup S_0}(X)$ is also irreducible, which contradictions the maximality of $A$, since $\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{A} + \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{S_0} = \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{A \cup S_0} > \iota(X,E)$. We therefore conclude that $S$ satisfies the SA-property wrt $X$.

 
  \par
  \addvspace{\medskipamount} 
  \noindent\emph{Step 3: Verifying the Chaumat-type decomposition:}\par\nobreak
  \addvspace{\smallskipamount} 
  \@afterheading
 Note since $\mathcal{P}_A(X)$ was proved to be irreducible, we may $\mathcal{P}_E(X)$ regarded as a closed subspace of $\mathcal{P}_A(X) \oplus L^\infty(S)$, hence it remains to prove the reverse containment. Let $\ell \in \mathcal{P}_E(X)^\perp$ be an element in the annihilator of $\mathcal{P}_E(X) \subseteq X \oplus L^\infty(E)$, regarded as a subspace of $(X \oplus L^\infty(E) )' \cong X' \oplus L^1(E)$. Then $\ell = (f,h) \in X' \oplus L^1(E)$ and we seek the annihilator for which the carrier set of $h$ has maximal Lebesgue measure.                                
   
                                                        
   
                                                   
Note that elements $(f,h) \in \mathcal{P}_E(X)^\perp$ satisfy the property that 
\[
f_n = \lim_{r \to 1-} \int_{\partial\mathbb{D}} f(r\zeta) \zeta^{-n}dm(\zeta) = - \int_{\partial\mathbb{D}}h(\zeta) \zeta^{-n} dm(\zeta) = - \widehat{h}(n), \qquad n=0,1,2, \dots,
\]
where $f_n$ are the Taylor coefficients of $f$ and $\widehat{h}(n)$ are the Fourier coefficients of $h$. \thlabel{LEM:SA-RIG} Let $X$ be a holomorphic Banach space. Then the following properties hold.

\begin{description}

As an immediate consequence of \thref{THM:AbsThom}, it only remains to establish the claims on conformal invariance of the $SA$-sets and the sets of rigidity. To this end, let $E$ be a Lebesgue measurable set $E\subseteq \partial\mathbb{D}$ which is the carrier set of an element $h \in L^1(\partial\mathbb{D})$ with $C(h) \in W^1_a(w) \neq \{0\}$.     \[
    \phi_\lambda (z) = \frac{\lambda-z}{1-\overline{\lambda}z}, \qquad z\in \mathbb{D}

\thlabel{wcutoff}
Let $w$ be a majorant and $K \subset \partial\mathbb{D}$ be a set of finite $w$-entropy. Then there exists an outer function $f : \mathbb{D} \to \mathbb{D}$ such that $f$ extends analytically across $\partial\mathbb{D} \setminus K$ and satisfies the estimate  
\begin{equation}\label{fKest}
    \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{f(z)} \lesssim w \left( 1-|z| \right), \qquad z\in \partial \Gamma_K \cap \mathbb{D},
\end{equation}
where $\Gamma_K := \cup_{\zeta \in K} \Gamma(\zeta)$ denotes a Stolz-Privalov domain associated to $K$. Furthermore, the function $F:\partial\mathbb{D} \to \mathbb{C}$ defined by
\[
F(e^{it})=f(e^{it}) 1_{\partial\mathbb{D} \setminus K}(e^{it}) 
\]
belongs to $C_w(\partial\mathbb{D})$, and there exists a constant $C>0$, such that
\begin{equation}\label{Fest}
\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{F(e^{it})} \leq C  w(\textbf{dist}(e^{it},K))^N, \qquad 0\leq t \leq 2\pi,
\end{equation}
for any desirable $N>0$.

  \noindent\emph{Step 1: The construction of $f$:}\par\nobreak

Let $I$ be a connected component of $\partial\mathbb{D} \setminus K$ and consider the so-called Whitney decomposition of $I$ consisting of a collection of (half-open, say right end-point excluded) subarcs $\{I_n\}_{n \in \mathbb{Z}}$ of $I$ with the properties 
\[

<!-- thm-18: body_md -->
\thlabel{THM:Univ1} Let $X$ be a holomorphic Banach space and $E \subset \partial\mathbb{D}$ be a Lebesgue measurable set. Then there exists a  partition $A, S$ of $E$ (unique modulo sets of Lebesgue measure zero), which gives rise to the following distinct phenomenons.
                                                                                                                                                                 
\begin{enumerate}
    \item[(i)] There exists a generic subset $\mathcal{U}_X \subset X$ in the sense of Baire categories, such that every $f \in \mathcal{U}_X$ gives rise to a universal Menshov-pair $(f,S)$ from $X$. 
    
    \item[(ii)] Whenever $(f,f^*) \in \mathcal{P}_A(X)$ and there exists a subsequence $\{T_{n_k}f\}_k$ which converges pointwise $dm$-a.e on $A$, then they must converge to $f^*$.
 
\end{enumerate}

<!-- proof-19: body_md -->
\par
  \addvspace{\medskipamount} 
  \noindent\emph{Step 1: Statement $(i)$:}\par\nobreak
  \addvspace{\smallskipamount} 
  \@afterheading

According to \thref{THM:AbsThom}, there exists a unique (modulo sets of Lebesgue measure zero) partition of Lebesgue measurable subsets $A, S$ of $E$, such that $\mathcal{P}_A(X)$ is irreducible and $S$ satisfies the SA-property wrt $X$. We may assume that both $A, S$ are non-trivial, that is, they have positive Lebesgue measure, otherwise one can set one of them to be the empty-set, for which the verification of either $(i)$ or $(ii)$ becomes redundant. To prove $(i)$, we shall adapt the argument in \cite{beise2016generic}. Note that since any measurable function $f$ is the pointwise $dm$-a.e limit of its truncation by height $f_N := f 1_{\@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{f}\leq N}$, it suffices to approximate any essentially bounded functions on $S$. To this end, we shall regard the sequence of Taylor polynomials $\{T_n\}_n$ as continuous linear operators from $X$ into balls of radius $M>0$ in $L^\infty(S)$, which equipped with the weak-star topology are metrizable (for instance, see \cite{rudin1973functional}). We denote the ball by $B_M(L^\infty(A))$ and recall that the universality criterion in \thref{THM:Univ1} assures that it suffices to check that for any pair $(f,g) \in X \oplus B_M(L^\infty(A))$ there exists analytic polynomials $\{Q_k\}$ and a subsequence $\{T_{n_k}\}_k$ such that
\[
(Q_k, T_{n_k}(Q_k) ) \to (f,g) 
\]
in $X \oplus B_M(L^\infty(A))$. However, since $T_{n} (Q_k) = Q_k$ for any $n$ sufficiently large, the universality criterion actually follows from the assumption that $S$ satisfies the SA-property wrt $X$. By means of repeat the same argument for any ball $B_M(L^\infty(A))$, the claim follows.

 
  \par
  \addvspace{\medskipamount} 
  \noindent\emph{Step 2: Statement $(ii)$:}\par\nobreak
  \addvspace{\smallskipamount} 
  \@afterheading

Let $(f,f^*)\in \mathcal{P}_A(X)$ and assume that $\{T_{n_k}f\}_k$ is a subsequence of Taylor polynomials of $f$, which converges pointwise $dm$-a.e to a measurable function $g$. Pick a sequence of analytic polynomials $(Q_j)_j$ such that $(Q_j, Q_j)$ converges to $(f, f^*)$ in $X \oplus L^\infty(A)$. Fix an $\varepsilon >0$ and write
\[
m(\{ A: \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{g-Q_j} > \varepsilon \}) \leq m(\{ A: \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{g-T_{n_k}(f)} > \varepsilon/2 \}) + \frac{2}{\varepsilon} \sum_{m=0}^{n_k} \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{f_m -(Q_j)_m}.
\]
Using the assumption that $Q_j \to f$ in $X \hookrightarrow \text{Hol}(\mathbb{D})$, it follows by standard arguments involving uniform convergence on compacts, that the Taylor coefficients $(Q_j)_n$ of $Q_j$ converge to the Taylor coefficients $f_n$ of $f$. Sending $j$ to infinity, we obtain
\[
\limsup_{j \to \infty} m(\{ A: \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{g-Q_j} > \varepsilon \}) \leq m(\{ A: \@ifstar{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}}{\@ifstar{\@ifstar{\abs}{\abs*}}{\@ifstar{\abs}{\abs*}*}*}*}{g-T_{n_k}(f)} > \varepsilon/2 \}), \qquad \forall k, \varepsilon >0.
\]
Now since $T_{n_k}(f)$ converges pointwise $dm$-a.e to $g$ on $A$, they also converge in Lebesgue measure on $A$, hence we can send $k \to \infty$ and conclude that $Q_j \to g$ in Lebesgue measure on $A$. By means of passing to a subsequence and using the assumption that $Q_j \to f^*$ weak-star in $L^\infty(A)$, we conclude that $g=f^*$ on $A$, which proves $(ii)$. The proof is now complete.
