\label{prop:32}

\begin{enumerate}
  
\item The entries within matrices $\frac{1}{h} \mathbf{M}_h^p$, $h \mathbf{B}_h^p$ and $h^{2p-1} \mathbf{D}_h^p$ are independent of the mesh parameter $h$.
  
\item The matrices $\mathbf{M}_h^p$, $\mathbf{B}_h^p$ and $\mathbf{D}_h^p$ are persymmetric, i.e., they are symmetric about their northeast-southwest diagonal.

<!-- theorem-1: body_md -->
{\cite[Theorem 3]{AmodioBrugnano1996}} \label{th:41}
Let $\{\mathbf{C}_n\}_{n}$ represent a family of nonsingular Toeplitz band matrices, and let $q^{\mathbf{C}}$ in \eqref{eq:42} be the associated polynomial. Then, the family $\{\mathbf{C}_n\}_{n}$ exhibits the following behaviour:
\begin{itemize}
  
\item it is well-conditioned if $q^{\mathbf{C}}$ is of type $(m,0,k)$;
  
\item it is weakly well-conditioned if $q^{\mathbf{C}}$ is of type $(m_1,m_2,k)$ or $(m,k_1,k_2)$, where $m_1+m_2=m$ and $k_1+k_2=k$. In this scenario, $\kappa(\mathbf{C}_n)$ grows at most as $\mathcal{O}(n^\eta)$, with $\eta$ representing the highest multiplicity among the zeros of unit modulus of $q^{\mathbf{C}}$.
\end{itemize}
  
Additionally, if $q^{\mathbf{C}}$ is of type $(q_1,q_2,q_3)$ for $q_1,q_2$ and $q_3$ such that $q_1+q_3>0$ and simultaneously $q_1 \ne m$ and $q_3 \ne k$, then $\kappa(\mathbf{C}_n)$ grows exponentially as $n$ increases.
<!-- end theorem-1 -->
<!-- remark-3: body_md -->
Weakly well-conditioning may be true for families of type $(0,m+k,0)$. Consider  indeed the family $\{\mathbf{C}_n\}_n$ of symmetric tridiagonal Toeplitz matrices with $c_{-1} = 1, c_{0} = 2$, and $c_1 = 1$. The polynomial $q^{\mathbf{C}}(z) = 1+2z+z^2$ is of type $(0,2,0)$, and the eigenvalues of matrices of this family are explicitly known (see, e.g. \cite[Pag. 59]{Smith1985})
 
\begin{equation*}
    \left\{\lambda_j = 2+2 \cos\left( \frac{j \pi}{n+1} \right) \quad j = 1,\ldots,n\right\}.
\end{equation*}
 
From the latter expression we readily obtain that $\kappa_2(\mathbf{C}_n) = \mathcal{O}(n^2)$.
<!-- end remark-3 -->
<!-- remark-4: body_md -->
Property that $q^{\mathbf{C}}$ is of type $(m_1,m_2,k)$ or $(m,k_1,k_2)$ , where $m_1+m_2=m$ and $k_1+k_2=k$ actually implies nonsingularity, for $n$ sufficiently large, of the elements of the family $\{\mathbf{C}_n\}_n$ (see \cite[Remark 2]{AmodioBrugnano1996}).
<!-- end remark-4 -->
<!-- remark-5: body_md -->
Note that similar results can be obtained in the $2$-norm by recalling that the following inequality is valid (see, for example, \cite[Equations 2.3.11 and 2.3.12]{GolubVanLoan1996})
 
\begin{equation*}
    \frac{1}{\sqrt{n}} \| \mathbf{C} \|_1 \le \| \mathbf{C} \|_2 \le \sqrt{n} \| \mathbf{C} \|_1, \quad \text{for all~} \mathbf{C} \in \mathbb{R}^{n \times n}.
\end{equation*}
 
In particular, weakly well-conditioning with respect to a given norm $\| \cdot \|_1$, $\| \cdot \|_\infty$ or $\| \cdot \|_2$ is equivalent to weakly well-conditioning in any other of the two norms.

<!-- lemma-1: body_md -->
\label{lem:47}
Let the family of matrices $\{\mathbf{G}_n^p\}_n$ be as in \eqref{eq:44}, and, if $p>1$, assume that not all the zeros of $q^{\mathbf{G}^p}$, defined in \eqref{eq:45}, are on the boundary of the unit circle. Then, $\{\mathbf{G}_n^p\}_n$ is weakly well-conditioned if and only if the associated polynomial $q^{\mathbf{G}^p}$ has exactly two zeros of unit modulus.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Assume $p>1$, if $p=1$ the proof is analogous. According to Theorem \ref{th:41}, the matrix family $\{\mathbf{G}_n^p\}_n$ exhibits weakly well-conditioned, or well-conditioned, behaviour if and only if $q^{\mathbf{G}^p}$ falls into one of the following types: $(r_1,p+1-r_1,p-1)$ for some $r_1 \in \{0,\ldots,p+1\}$ or $(p+1,p-1-r_2,r_2)$ for some $r_2 \in \{0,\ldots,p-1\}$. It is worth noting that if $\xi$ is a zero of $q^{\mathbf{G}^p}$, then $\xi^{-1}$ is also a zero. Hence, $q^{\mathbf{G}^p}$ necessarily is a polynomial of type $(r,2p-2r,r)$ for some $r \in \{1,\ldots,p\}$. Consequently, the only admissible choice for weakly well-conditioning is $q^{\mathbf{G}^p}$ of type $(p-1,2,p-1)$.

Then, the family of matrices $\{ \mathbf{G}_n^p \}_n$ as defined in \eqref{eq:44} exhibits weakly well-conditioned behaviour if and only if the Chebyshev transform $\mathcal{T} q^{\mathbf{G}^p}$ \eqref{eq:46} of the polynomial $q^{\mathbf{G}^p}$ has exactly one real root in the interval $[-2,2]$.

<!-- lemma-3: body_md -->
\label{lem:49}
Consider $q^{\mathbf{G}^p}(z) = g_0 z^{2p} + g_1 z^{2p-1} + \ldots + g_1 z + g_0 $, a self-reciprocal polynomial of degree $2p$. Let $\{x_1,\ldots,x_p\}$ denote the roots of its Chebyshev transform $\mathcal{T} q^{\mathbf{G}^p}$. Assuming $g_0 \ne 0$, the following formulas are valid
 
\begin{align*}
   \prod_{j=1}^p (x_j-2) = (-1)^p \, \frac{q^{\mathbf{G}^p}(1)}{g_0}, \quad \quad \quad \prod_{j=1}^p (x_j+2) = \frac{q^{\mathbf{G}^p}(-1)}{g_0}.
\end{align*}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
To prove the first formula, let us consider $f(x) = \mathcal{T} q^{\mathbf{G}^p}(x+2)$. For all roots $x_j$ of $\mathcal{T} q^{\mathbf{G}^p}$, we have $f(x_j-2)=\mathcal{T} q^{\mathbf{G}^p}(x_j)=0$. Therefore, recalling Vieta's formulas the product $(x_1-2)\dots(x_p-2)$ is equal to $(-1)^p f(0)/g_0$. We can compute $f(0) = \mathcal{T} q^{\mathbf{G}^p}(2) = q^{\mathbf{G}^p}(1)$, the latter because $x(z) = z^{-1} + z = 2$ if and only if $z=1$. For the second formula, the proof follows a similar reasoning.

<!-- proposition-3: body_md -->
\label{prop:51}
For $p \ge 1$ and $\theta \in [-\pi,\pi]$, let the functions $M_p, B_p, D_p : [-\pi,\pi] \to \mathbb{R}$ be defined as
 
\begin{align*}
	M_p(\theta)  = (2-2\cos \theta)^{p+1} \sum_{j \in \mathbb{Z}} \frac{1}{(\theta + 2j\pi)^{2p+2}}, & \quad 
 B_p(\theta)  = -(2-2\cos \theta)^{p+1} \sum_{j \in \mathbb{Z}} \frac{1}{(\theta + 2j\pi)^{2p}},
	\\  & \hspace{-1.5cm} D_p(\theta) = (-1)^p (2-2\cos \theta)^p.
\end{align*}
 
Then, the following relations hold true 
 
\begin{align}
    \Phi_{2p+1}(p+1) + \sum_{j=1}^{p} \Phi_{2p+1}(p+1-j) (e^{-\mathrm{i} j \theta}+e^{\mathrm{i} j \theta}) & = M_p(\theta), \label{eq:51}
	\\ \partial_s^2 \Phi_{2p+1}(p+1) + \sum_{j=1}^{p}  \partial_s^2 \Phi_{2p+1}(p+1-j) (e^{-\mathrm{i} j \theta} + e^{\mathrm{i} j \theta}) & = B_p(\theta),
	\\ \partial_s^{2p} \Phi_{2p+1}(p+1) + \sum_{j=1}^{p} \partial_s^{2p} \Phi_{2p+1}(p+1-j) (e^{-\mathrm{i} j \theta} + e^{\mathrm{i} j \theta})  & = D_p(\theta).
\end{align}
  
Therefore, we can express $q^{\mathbf{K}^p(\rho,\delta)}(e^{\mathrm{i} \theta})$ as
 
\begin{equation} \label{eq:54}
	q^{\mathbf{K}^p(\rho,\delta)}(e^{\mathrm{i} \theta}) = e^{\mathrm{i} p \theta}\left( B_p(\theta) + \rho\left(M_p(\theta) + \delta (-1)^p D_p(\theta) \right) \right).
\end{equation}
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
There is already a proof for the explicit expressions of $M_p(\theta)$ and $B_p(\theta)$ in \cite[Lemmas 7 and 6]{GaroniManniPelosiSerraCapizzanoSpeleers2014}. For the sake of completeness, we report the details for $M_p(\theta)$, and present a different proof for $B_p(\theta)$, which is useful for calculating $D_p(\theta)$.

We recall that the Fourier transform of a cardinal spline $\Phi_p$ is
 
\begin{equation*}
	\widehat \Phi_p(\xi) = \left(\frac{1-e^{-\mathrm{i} \xi}}{\mathrm{i} \xi} \right)^{p+1}, \quad \left| \widehat \Phi_p(\xi) \right|^2 = \left(\frac{2-2\cos \xi}{\xi^2} \right)^{p+1}, \quad \quad \xi \in \mathbb{R}
\end{equation*}
 
(see \cite[Example 3.4]{Chui1992}), and satisfies the hypothesis of a Poisson summation formula \cite[Theorem 2.28]{Chui1992}, i.e., for all $\theta \in [-\pi,\pi]$
 
\begin{equation*}
	\sum_{j \in \mathbb{Z}} \left(\int_{\mathbb{R}} \Phi_p(s) \Phi_p(s+j) \dd s\right) e^{\mathrm{i} j \theta}  = \sum_{j \in \mathbb{Z}} \left| \widehat{\Phi}_p(\theta+2j\pi) \right|^2.
\end{equation*}
 
To show \eqref{eq:51}, recalling \eqref{eq:37} and \eqref{eq:38}, we compute for $\theta \in [-\pi,\pi]$, 
 
\begin{align*}
	\Phi_{2p+1}(p+1) & + \sum_{j=1}^{p}\Phi_{2p+1}(p+1-j) (e^{-\mathrm{i} j \theta}+e^{\mathrm{i} j \theta}) = \sum_{j \in \mathbb{Z}} \Phi_{2p+1}(p+1-j) e^{\mathrm{i} j \theta}
    \\ & = \sum_{j \in \mathbb{Z}} \left(\int_{\mathbb{R}} \Phi_p(s) \Phi_p(s+j) \dd s\right) e^{\mathrm{i} j \theta} = \sum_{j \in \mathbb{Z}} \left| \widehat{\Phi}_p\left(\theta+2j\pi\right) \right|^2
    \\ & = (2-2\cos \theta)^{p+1} \sum_{j \in \mathbb{Z}} \frac{1}{(\theta+2j\pi)^{2(p+1)}}.
\end{align*}
 
Using the recurrence relation for derivatives of cardinal splines \cite[Theorem 4.3]{Chui1992} 
 
\begin{equation*}
	\partial_s^k \Phi_p(s) = \partial_s^{k-1} \Phi_{p-1}(s) - \partial_s^{k-1} \Phi_{p-1}(s-1),
\end{equation*}
 
and considering that $e^{\mathrm{i} j \theta} - 2e^{\mathrm{i} (j+1) \theta} + e^{\mathrm{i} (j+2) \theta} = e^{\mathrm{i} j \theta} (1-e^{\mathrm{i} \theta})^2 = -e^{\mathrm{i} (j+1) \theta}(2-2\cos \theta)$, we can compute
 
\begin{align*}
	 \partial_s^2 \Phi_{2p+1}(p+1) & + \sum_{j=1}^{p} \partial_s^2 \Phi_{2p+1}(p+1-j) (e^{-\mathrm{i} j \theta}+e^{\mathrm{i} j \theta}) 
  \\ &= \sum_{j \in \mathbb{Z}} \partial_s^2 \Phi_{2p+1}(p+1-j) e^{\mathrm{i} j \theta}
    \\ & = \sum_{j \in \mathbb{Z}} \left(\partial_s \Phi_{2p}(p+1-j)-\partial_s \Phi_{2p}(p-j)\right) e^{\mathrm{i} j \theta}
	 \\ & = \sum_{j \in \mathbb{Z}} \left(\Phi_{2p-1}(p+1-j)-2\Phi_{2p-1}(p-j)+\Phi_{2p-1}(p-j-1)\right)  e^{\mathrm{i} j \theta}
	 \\ & = -(2-2\cos \theta)\sum_{j \in \mathbb{Z}} \Phi_{2p-1}(p+1-j) e^{\mathrm{i} (j+1) \theta} 
     \\ & = -(2-2\cos \theta)\sum_{j \in \mathbb{Z}} \Phi_{2p-1}(p-j) e^{\mathrm{i} j \theta}
     \\ & = -(2-2\cos \theta)^{p+1} \sum_{j \in \mathbb{Z}} \frac{1}{(\theta + 2j\pi)^{2p}},
\end{align*}
  
where in the last equality we used the same procedures as in the proof for $M_p(\theta)$. Similarly, we can iteratively make the same steps as before to show that 
 
\begin{align*}
	\partial_s^{2p} \Phi_{2p+1}(p+1) + \sum_{j=1}^{p} \partial_s^{2p} \Phi_{2p+1}(p+1-j) (e^{-\mathrm{i} j \theta}+e^{\mathrm{i} j \theta})
	& = (-1)^p (2-2\cos \theta)^p \sum_{j \in \mathbb{Z}} \Phi_1(1-j)e^{\mathrm{i} j \theta} 
	\\ & = (-1)^p (2-2\cos \theta)^{p}.
\end{align*}

<!-- lemma-4: body_md -->
\label{lem:59}
For $p \ge 1$, let $\delta_p$ and $\rho_p$ be defined as in \eqref{eq:56}. Define the function 
 
\begin{equation*}
	F_p(\theta;\rho,\delta) = e^{- \mathrm{i} p \theta} q^{\mathbf{K}^p(\rho,\delta)}(e^{\mathrm{i} \theta}) = B_p(\theta) + \rho(M_p(\theta) + \delta (-1)^p D_p(\theta))
\end{equation*}
 
with $M_p, B_p$ and $D_p$ as in Proposition \ref{prop:51}. Then, the following statements hold true.
 
\begin{itemize}
	\item The equation (in terms of $\theta$) $F_p(\theta;\rho,\delta) = 0$ has exactly two simple zeros in $[-\pi,\pi]$ for all $\rho > 0$ if and only if $\delta \le \delta_p$;
	\item The equation (in terms of $\theta$) $F_p(\theta;\rho,0) = 0$ has exactly two simple zeros in $[-\pi,\pi]$ if and only if $\rho \le \rho_p$.
\end{itemize}
<!-- end lemma-4 -->
<!-- proof-8: body_md -->
Let $p \ge 1$. We study the equation only in the interval $[0,\pi]$ since $F_p(\theta; \rho,\delta)$ is even in $\theta$. From \eqref{eq:55} and the definition in Proposition \ref{prop:51}, we observe that $M_p(\theta) > 0$ for all $\theta \in [0,\pi]$. Hence, the ratio $F_p(\theta;\rho,\delta)/M_p(\theta)$ is well defined for $\theta \in [0,\pi]$. Our objective is to show that for all $p \ge 1$, $\rho > 0$ and $\delta \le 0$, the function $F_p(\theta; \rho, \delta)/M_p(\theta)$ is strictly decreasing in $\theta$ for $\theta \in (0,\pi)$. We begin by computing
 
\begin{equation*}
	\frac{d}{d \theta} \frac{F_p(\theta;\rho,\delta)}{M_p(\theta)} = \frac{d}{d \theta} \frac{B_p(\theta)}{M_p(\theta)}  + \rho \delta (-1)^p \left( \frac{D'_p(\theta)M_p(\theta) - D_p(\theta)M_p'(\theta)}{(M_p(\theta))^2} \right).
\end{equation*}
 
It is shown in \cite[Theorem 2]{EkstromFurciGaroniManniSerraCapizzano2018} that $\frac{d}{d \theta}\left[B_p(\theta)/M_p(\theta)\right] < 0$ for all $p \ge 1$ and $\theta \in (0,\pi)$. Additionally, we calculate $(-1)^p D_p'(\theta) = 2p \sin \theta (2-2\cos \theta)^{p-1} > 0$ for all $\theta \in (0,\pi)$. Therefore, combining \cite[Lemma A.2.]{DonatelliGaroniManniSerraCapizzanoSpeleers2017} that tell us that $M_p'(\theta) < 0$ for all $p \ge 1$ and $\theta \in (0,\pi)$, and the fact that $M_p(\theta) > 0$, and $(-1)^p D_p(\theta) > 0$, we conclude that for all $\rho >0$ and $\delta \le 0$, $\frac{d}{d \theta} [F_p(\theta; \rho, \delta)/M_p(\theta)] < 0$ for all $\theta \in (0,\pi)$. This confirms that $F_p(\theta; \rho, \delta)/M_p(\theta)$ is strictly decreasing in $\theta$ for $\rho>0$ and $\delta \le 0$.

We can state that $F_p(\theta; \rho, \delta)$ has exactly one zero in $[0,\pi]$ if and only if $F_p(0; \rho, \delta) \ge 0$ and $F_p(\pi ; \rho, \delta) \le 0$. Both quantities have been computed in Corollary \ref{cor:54}, leading to the desired result.

Let us consider the function $F_p^k(\theta;\rho,\delta) = e^{- \mathrm{i} p \theta} q^{\mathbf{K}^{p,k}(\rho,\delta)}(e^{\mathrm{i} \theta})$. We aim to show that the equation (in terms of $\theta$) $F^k_p(\theta;\rho,\delta) = 0$ has exactly one simple zero in $[0,\pi]$ for all $\rho > 0$ if and only if $\delta \le \delta_p^k$. Establishing this fact leads to the conclusion by Lemma \ref{lem:47} and Theorem \ref{th:45}.

As in the proof of Lemma \ref{lem:59}, for all $p \ge 1$, $\rho > 0$, and $\delta < 0$, we show that the function $F_p^k(\theta; \rho, \delta)/M_p(\theta)$ is strictly decreasing in $\theta$. Explicitly, we can express it as
 
\begin{equation*}
	\frac{F_p^k(\theta ; \rho, \delta)}{M_p(\theta)} = \frac{B_p(\theta)}{M_p(\theta)} + \rho + \rho \delta (-1)^k \frac{D_p^k(\theta)}{M_p(\theta)}.
\end{equation*}
 
By virtue of \cite[Theorem 2]{EkstromFurciGaroniManniSerraCapizzano2018}, showing that the last term is a decreasing functions suffices. From Proposition \ref{prop:62}, 
 
\begin{equation*}
	(-1)^k \frac{D^k_p(\theta)}{M_p(\theta)} = (2-2\cos \theta)^k \frac{M_{p-k}(\theta)}{M_p(\theta)} = \prod_{j=0}^{k-1} (2-2\cos \theta) \frac{M_{p-k+j}(\theta)}{M_{p-k+j+1}(\theta)}.
\end{equation*}
 
Each factor here is a decreasing function (as established in \cite[Lemma A.2.]{DonatelliGaroniManniSerraCapizzanoSpeleers2017}) and positive, hence the product is decreasing as well. Consequently, $F_p^k(\theta; \rho, \delta)$ has only one simple zero in $[0,\pi]$ if and only if $F_p^k(0; \rho, \delta) \ge 0$ and $F_p^k(\pi; \rho, \delta) \le 0$. This holds true for all $\rho > 0$ if and only if $\delta \le \delta_p^k$, given that $F_p^k(0;\rho,\delta)= \rho$. The explicit expression of $\delta_p^k$ is a consequence of \eqref{eq:62}.

where $\mathbf{e}_\ell$ is the $\ell^{\operatorname{th}}$ vector of the canonical base of $\mathbb{R}^n$, $\mathbf{1}_{m+k} = (1,\ldots,1)^{\top} \in \mathbb{R}^{m+k}$, $\mathbf{Z}$ is defined in \eqref{eq:A6}, and the matrices $\mathbf{G}_1^{(0)} \in \mathbb{R}^{m \times k}$ and $\mathbf{G}_2^{(0)} \in \mathbb{R}^{k \times k}$ are defined by
 
\begin{equation*}
    \begin{pmatrix}
        \mathbf{G}_1^{(0)} \vspace{0.1cm} \\
        \mathbf{G}_2^{(0)}
    \end{pmatrix}
    = \mathbf{W}^{-1} \mathbf{X}_{11}^{(0)}[1:m+k,1:k], 
\end{equation*}
 
with $\mathbf{W}$ the Casorati matrix, and $\mathbf{X}_{11}^{(0)}[1:m+k,1:k]$ the $(m+k) \times k$ top-left block of $\mathbf{X}_{11}^{(0)}$ in standard indices notation. A similar expression is obtained for the $\ell^{\operatorname{th}}$ row of $\mathbf{X}_{11}^{(1)}$ with the difference that the top-left block of $\mathbf{C}_n^{(1)}$ in \eqref{eq:A2} must be also considered, from which the analogue of \eqref{eq:A9} only holds for $\ell \ge k+1$. With greater details, we obtain
 
\begin{equation*}
    \mathbf{e}_\ell^\top \mathbf{X}_{11}^{(1)} = \mathbf{1}_{m+k}^{\top} \mathbf{Z}^{\ell-1-k}
    \begin{pmatrix}
        \mathbf{G}_1^{(1)} \vspace{0.1cm} \\
        \mathbf{G}_2^{(1)}
    \end{pmatrix}, \quad \ell = k+1,\ldots, n
\end{equation*}
 
with the matrices $\mathbf{G}_1^{(1)} \in \mathbb{R}^{m \times k}$ and $\mathbf{G}_2^{(1)} \in \mathbb{R}^{k \times k}$ defined by
 
\begin{equation} \label{eq:A10}
    \begin{pmatrix}
        \mathbf{G}_1^{(1)} \vspace{0.1cm} \\
        \mathbf{G}_2^{(1)}
    \end{pmatrix}
    = \mathbf{W}^{-1} \mathbf{X}_{11}^{(1)}[k+1:m+2k,1:k].
\end{equation}
 
Note that the block $\mathbf{X}_{11}^{(1)}[k+1:m+2k,1:k]$ depends on the top-left block perturbation of size $(m+k) \times k$ of $\mathbf{C}_n^{(1)}$.

For $n$ sufficient large, similarly, we compute
 
\begin{equation} \label{eq:A11}
    \begin{aligned}
        \mathbf{X}_{21}^{(0)} & =
        \begin{pmatrix}
            \mathbf{W}_{11} & \mathbf{W}_{12}
        \end{pmatrix}
        \mathbf{Z}^n
        \begin{pmatrix}
            \mathbf{G}_1^{(0)} \vspace{0.1cm} \\
            \mathbf{G}_2^{(0)}
        \end{pmatrix}
     = \mathbf{W}_{12} \mathbf{Z}_2^n \mathbf{G}_2^{(0)} + \mathcal{O}\left(\left|\frac{z_m}{z_{m+1}}\right|^n\right),
    \\ 
        \mathbf{X}_{21}^{(1)} & =
        \begin{pmatrix}
            \mathbf{W}_{11} & \mathbf{W}_{12}
        \end{pmatrix}
        \mathbf{Z}^{n-k}
        \begin{pmatrix}
            \mathbf{G}_1^{(1)} \vspace{0.1cm} \\
            \mathbf{G}_2^{(1)}
        \end{pmatrix}
     = \mathbf{W}_{12} \mathbf{Z}_2^{n-k} \mathbf{G}_2^{(1)} + \mathcal{O}\left(\left|\frac{z_m}{z_{m+1}}\right|^{n-k}\right).
    \end{aligned}
\end{equation}
 
Since it is assumed that $\mathbf{C}_n^{(0)}$ and $\mathbf{C}_n^{(1)}$ are nonsingular, for sufficiently large $n$, by Lemma \ref{lem:A1} also $\mathbf{G}_2^{(1)}$ and $\mathbf{G}_2^{(0)}$ are.

Finally, $\mathbf{X}_{22}$ has been calculated in \cite[Equation (25)]{AmodioBrugnano1996}, and it holds true
 
\begin{equation} \label{eq:A12}
    \mathbf{X}_{22} \mathbf{e}_j =
    \begin{pmatrix}
        \mathbf{W}_{11} & \mathbf{W}_{12}
    \end{pmatrix}
    \mathbf{Z}^{n+m-j}
    \begin{pmatrix}
        \mathbf{G}_1^{(\cdot)} \vspace{0.1cm} \\
        \mathbf{G}_2^{(\cdot)}
    \end{pmatrix}, \quad j = 1,\ldots,n
\end{equation}
 
with the vectors $\mathbf{G}_1^{(\cdot)}$ and $\mathbf{G}_2^{(\cdot)}$ not depending on $j$, defined as
 
\begin{equation*}
    \begin{pmatrix}
        \mathbf{G}_1^{(\cdot)} \vspace{0.1cm} \\
        \mathbf{G}_2^{(\cdot)}
    \end{pmatrix}
    = 
    \mathbf{W}^{-1}
    \begin{pmatrix}
        0 \\ \vdots \\ 0 \\ c_k^{-1}
    \end{pmatrix}.
\end{equation*}
 
Combining \eqref{eq:A9}-\eqref{eq:A12}, for $\ell = k+1,\ldots,n$ and $j=1,\ldots,n$ we deduce from \eqref{eq:A8}, for $n$ sufficiently large,
 
\begin{align*}
    \mathbf{H}_n^{(1)}[\ell,j] & = \mathbf{e}_\ell^\top \bigl(\mathbf{X}_{11}^{(0)} (\mathbf{X}_{21}^{(0)})^{-1}-\mathbf{X}_{11}^{(1)} (\mathbf{X}_{21}^{(1)})^{-1} \bigr) \mathbf{X}_{22} \mathbf{e}_j 
    \\ & \hspace{-1cm} \approx \mathbf{1}_{m+k}^\top \mathbf{Z}^{\ell-1}
            \left(
            \begin{pmatrix}
            \mathbf{G}_1^{(0)} \\
            \mathbf{G}_2^{(0)}
            \end{pmatrix}
            (\mathbf{G}_2^{(0)})^{-1}
            - \mathbf{Z}^{-k}
            \begin{pmatrix}
            \mathbf{G}_1^{(1)} \\
            \mathbf{G}_2^{(1)}
            \end{pmatrix}
            (\mathbf{G}_2^{(1)})^{-1} \mathbf{Z}_2^k
            \right) \mathbf{Z}_2^{-n} \mathbf{W}_{12}^{-1} \mathbf{X}_{22} \mathbf{e}_j
    \\ & \hspace{-1cm} = \mathbf{1}_{m+k}^\top \mathbf{Z}_1^{\ell-1}
            \left( 
            \mathbf{G}_1^{(0)} (\mathbf{G}_2^{(0)})^{-1}
            - \mathbf{Z}_1^{-k} \mathbf{G}_1^{(1)} (\mathbf{G}_2^{(1)})^{-1} \mathbf{Z}_2^k
            \right) \mathbf{Z}_2^{-n} \mathbf{W}_{12}^{-1} \mathbf{X}_{22} \mathbf{e}_j
    \\ & \hspace{-1cm} \approx \mathbf{1}_{m+k}^\top \mathbf{Z}_1^{\ell-1}
            \left(
            \mathbf{G}_1^{(0)} (\mathbf{G}_2^{(0)})^{-1} 
            - \mathbf{Z}_1^{-k} \mathbf{G}_1^{(1)} (\mathbf{G}_2^{(1)})^{-1} \mathbf{Z}_2^k
            \right) \mathbf{Z}_2^{m-j} \mathbf{G}_2^{(\cdot)}
    \\ & \hspace{-1cm} = \mathcal{O} \left( \frac{|z_m|^\ell}{|z_{m+1}|^j} \right).
\end{align*}
 
From the latter, we deduce that the families $\{\mathbf{C}_n^{(0)}\}_n$ and $\{\mathbf{C}_n^{(1)}\}_n$ have the same conditioning behaviour.

\noindent\textbf{Step 2.} Consider the families of matrices $\{(\mathbf{C}_n^{(1)})^\top\}_n$ and $\{(\mathbf{C}_n^{(2)})^\top\}_n$. Since each difference $(\mathbf{C}_n^{(2)})^\top - (\mathbf{C}_n^{(1)})^\top$ has the same structure as each $\mathbf{C}_n^{(1)} - \mathbf{C}_n^{(0)}$, it is possible to proceed exactly as in \textbf{Step 1.} and bound component-wise the elements of the matrix $\mathbf{H}_n^{(2)}$. Note that the polynomial $q^{\mathbf{C}^\top}$ associated with the purely Toeplitz part of the family $\{(\mathbf{C}_n^{(1)})^\top\}_n$ has as roots the reciprocals of the roots of $q^{\mathbf{C}}$, while $m$ and $k$ now have an inverted role. Therefore, if $q^{\mathbf{C}}$ is of type $(m_1,m_2,k)$, then $q^{\mathbf{C}^\top}$ is of type $(k,m_2,m_1)$. Taking these facts into account, we deduce that, for $n$ sufficiently large,
 
\begin{equation*}
    \mathbf{H}_n^{(2)}[\ell,j] =  \mathcal{O}\left(\frac{|z_{n-k+1}|^{-j}}{|z_{n-k}|^{-\ell}}\right) = \mathcal{O}\left(\frac{|z_m|^\ell}{|z_{m+1}|^j}\right), 
\end{equation*}
 
for $\ell = m+1,\ldots,n$ and $j = k+1,\ldots, n$. As in the previous step, we conclude that the families $\{ \mathbf{C}_n^{(1)}\}_n$ and $\{ \mathbf{C}_n^{(2)}\}_n$ have the same conditioning behaviour.

\noindent\textbf{Step 3.} Let us define $\textbf{J}_n$ the exchange matrix of size $n \times n$, i.e.,
 
\begin{equation} \label{eq:A13}
    \textbf{J}_n = \begin{pmatrix}
    0 & \\
    1 & 0  \\
    0 & 1 & 0 \\
    \vdots &\ddots &\ddots & \ddots \\
    0 &\ldots& 0 & 1 & 0
    \end{pmatrix}_{n \times n} \hspace{-0.65cm}.
\end{equation}
 
Consider the families $\{\textbf{J}_n (\mathbf{C}_n^{(2)})^\top \textbf{J}_n\}_n$ and $\{\textbf{J}_n (\mathbf{C}_n^{(3)})^\top \textbf{J}_n \}_n$. Given a nonsingular matrix $\mathbf{C} \in \mathbb{R}^{n \times n}$, it holds valid $(\textbf{J}_n \mathbf{C} \textbf{J}_n)^{-1} = \textbf{J}_n \mathbf{C}^{-1} \textbf{J}_n$. Therefore, the conditioning behaviour of the families just defined is equal to that of $\{\mathbf{C}_n^{(2)}\}_n$ and $\{\mathbf{C}_n^{(3)}\}_n$, respectively. As in \textbf{Step 2.}, each difference $\textbf{J}_n (\mathbf{C}_n^{(3)})^\top \textbf{J}_n - \textbf{J}_n (\mathbf{C}_n^{(2)})^\top \textbf{J}_n$ has the same structure as each $\mathbf{C}_n^{(1)} - \mathbf{C}_n^{(0)}$, and it is possible to proceed as in \textbf{Step 1.} to bound component-wise the elements of $\mathbf{H}_n^{(3)}$. Note that in this case the polynomial $q^{\textbf{J} \mathbf{C}^\top \textbf{J}}$ associated with the Toeplitz part of the family $\{\textbf{J}_n (\mathbf{C}_n^{(2)})^\top \textbf{J}_n\}_n$ satisfies $q^{\textbf{J} \mathbf{C}^\top \textbf{J}} = q^{\mathbf{C}}$. We deduce, for $n$ sufficiently large,
 
\begin{equation*}
    (\textbf{J}_n (\mathbf{H}_n^{(3)})^\top \textbf{J}_n) [\ell,j] = \mathcal{O}\left(\frac{|z_m|^\ell}{|z_{m+1}|^j}\right), \quad \ell = k+1,\ldots,n \quad \text{and} \quad j = m+1,\ldots, n,
\end{equation*}
 
or equivalently
 
\begin{equation*}
    \mathbf{H}_n^{(3)} [\ell,j] = \mathcal{O}\left(\frac{|z_m|^{n-j+1}}{|z_{m+1}|^{n-\ell+1}}\right), \quad \ell = k+1,\ldots,n \quad \text{and} \quad j = m+1,\ldots, n.
\end{equation*}
 
We deduce the same conditioning behaviour for the families $\{ \mathbf{C}_n^{(2)}\}_n$ and $\{ \mathbf{C}_n^{(3)}\}_n$.

\noindent\textbf{Step 4.} Finally, we consider the families $\{\textbf{J}_n (\mathbf{C}_n^{(3)}) \textbf{J}_n\}_n$ and $\{\textbf{J}_n (\mathbf{C}_n^{(4)}) \textbf{J}_n \}_n$. Proceeding as in \textbf{Steps 2.} and \textbf{3.}, with estimates similar as in \textbf{Step 1.}, we obtain
 
\begin{equation*}
    \mathbf{H}_n^{(4)} [\ell,j] = \mathcal{O}\left(\frac{|z_m|^{n-j+1}}{|z_{m+1}|^{n-\ell+1}}\right), \quad \ell = m+1,\ldots,n \quad \text{and} \quad j = k+1,\ldots, n.
\end{equation*}
 
Combining the four steps concludes the proof.
\hfill \qedsymbol
