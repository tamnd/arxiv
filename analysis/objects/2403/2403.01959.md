For $N \ge 0$, $\mathbb{R}^{N+1}_+=\{(x,y): x \in \mathbb{R}^N, y>0\}$. We write $\nabla u, D^2 u$ for the gradient and the Hessian matrix of a function $u$ with respect to all $x,y$ variables and $\nabla_x u, D_y u, D_{x_ix_j }u, D_{x_i y} u$ and so on, to distinguish the role of $x$ and $y$.

<!-- prop-1: body_md -->
\label{prop-form}
	The forms $\mathfrak{a}$, $\mathfrak{a^*}$ are continuous, accretive and   sectorial.
<!-- end prop-1 -->
<!-- head-2: body_md -->
We consider only the form $\mathfrak{a}$ since the adjoint form can be handled similarly. If $u\in H^1_{c}$
\begin{align*}
	\textrm{\emph{Re}\,} \mathfrak{a}(u,u)\geq \|\nabla_x u\|^2_{L^2_c}+\|D_y u\|^2_{L^2_c}-2|a| \|\nabla_x u\|_{L^2_c}\|D_y u\|_{L^2_c}\geq (1-|a|)(\|\nabla_x u\|^2_{L^2_c}+\|D_y u\|^2_{L^2_c}).
\end{align*}
By the ellipticity assumption $|a|<1$, the accretivity follows.
Moreover 
\begin{align*}
	|\textrm{\emph{Im}\,} \mathfrak{a}(u,u)|\leq  2|a|\|\nabla_x u\|_{L^2_c}\|D_y u\|_{L^2_c}\leq |a|(\|\nabla_x u\|^2_{L^2_c}+\|D_y u\|^2_{L^2_c})\leq \frac{|a|}{(1-|a|)}\textrm{\emph{Re}\,} \mathfrak{a}(u,u).
\end{align*}
This proves the sectoriality and then the continuity of the form. \thinspace\null\nobreak\hfill\hbox{\vbox{\kern-.2pt\hrule

<!-- prop-2: body_md -->
\label{generation L2}
	$\mathcal L$ and $\mathcal L^*$  generate contractive analytic semigroups $e^{z \mathcal L}$, $e^{z \mathcal L^*}$, $z\in\Sigma_{\frac{\pi}{2}-\arctan \frac{|a|}{1-|a|}}$,  in $L^2_{c}$. Moreover the semigroups $(e^{t\mathcal L})_{t \geq 0}, (e^{t\mathcal L^*})_{t \geq 0}$ are positive and $L^p_c$-contractive for $1 \leq p \leq \infty$.
<!-- end prop-2 -->
<!-- head-3: body_md -->
We argue only for $\mathcal L$.
The  generation  result immediately follows from   Proposition \ref{prop-form} and \cite[Theorem 1.52]{Ouhabaz}.
The positivity follows by \cite[Theorem 2.6]{Ouhabaz} after observing that, if $u\in H^1_{c}$, $u$ real, then $u^+\in H^1_{c}$ and
\begin{align*}
	\mathfrak{a}(u^+,u^-)
	&:=
	\int_{\mathbb{R}^{N+1}_+} \langle \nabla u^+, \nabla u^-\rangle\,y^{c} dx\,dy+2\int_{\mathbb{R}^{N+1}_+}D_yu^+ a\cdot \nabla_xu^-\,y^{c} dx\,dy=0.
\end{align*}
Finally, the $L^\infty$-contractivity follows by \cite[Corolary 2.17]{Ouhabaz}  after observing that if $0\leq u\in H^1_{c}$,  then $1\wedge u, (u-1)^+\in H^1_{c}$ and, since $\nabla (1\wedge u)=\chi_{\{u<1\}}\nabla u$ 
	and 
	$\nabla (u-1)^+=\chi_{\{u>1\}}\nabla u$, one has 
	\begin{align*}
		\mathfrak{a}(1\wedge u,(u-1)^+)&=0.
	\end{align*}

<!-- prop-3: body_md -->
\label{Prop scaling} The scale homogeneity of $\mathcal L$ is  $2$:
		\begin{align}\label{scale.op}
			s^2\mathcal L=I_s^{-1}\mathcal L I_s, \qquad I_su(x,y)=u(sx,sy), \qquad s>0. 
		\end{align}
	Moreover the semigroup $e^{t\mathcal L}$ generated by $\mathcal L$  in $L^2_{c}$
		satisfies
		\begin{equation}\label{scale.semi}
			e^{s^2 z\mathcal L}=I_{s}^{-1}e^{-z\mathcal L}I_s, \qquad   z\in\Sigma_{\frac{\pi}{2}-\arctan \frac{|a|}{1-|a|}},\quad s>0. 
		\end{equation}
The same results  hold for the adjoint operator $\mathcal L^\ast$ and its generated semigroup.

<!-- lem-2: body_md -->
\label{Mazya 1}
	If $\alpha\in\mathbb{R}$ and  $q\geq 1$ satisfies
	\begin{align*}
		1\leq q\leq (N+1)'=\frac{N+1}{N},\qquad \beta=\alpha-1+\frac{q-1}{q}(N+1)>-\frac{1}{q}
	\end{align*}
	(in particular $\alpha+1>0$) then there exists a positive constant $C$ such that
	\begin{align}\label{Mazya 1 eq}
		\|y^{\beta} u\|_{L^q(\mathbb{R}^{N+1}_+)}\leq C\|y^{\alpha}\nabla u\|_{L^1(\mathbb{R}^{N+1}_+)},\quad \forall u\in C_c^\infty\left(\overline{\mathbb{R}^{N+1}_+}\right).
	\end{align}

\label{Mazya 2}
	Let  $q, \ \beta,\  \alpha\in \mathbb{R}$, satisfying the following assumptions:
	\begin{itemize}
		\item[(i)] $2\leq q\leq 2^{\ast}= 2\frac{N+1}{N-1}$ \quad ($2\leq q$ when $N=1$);
		\item[(ii)] $\beta+\frac 1 q>0$;
		\item[(iii)] $\beta= \alpha-1+(N+1)\left(\frac 1 2 -\frac 1 q\right)$ \quad i.e. \quad $\frac 1 q=\frac 1 2- \frac{\beta-\alpha+1}{N+1}$.
	\end{itemize} 
	Then one has  for some positive constant $C$
	\begin{align*}
		\| y^\beta u\|_{L^q\left(\mathbb{R}^{N+1}_+\right)}
		\leq C\|y^{ \alpha}\nabla u\|_{L^2\left(\mathbb{R}^{N+1}_+\right)},\qquad \text{for all }\quad u\in C_c^\infty\left(\overline{\mathbb{R}^{N+1}_+}\right).
	\end{align*}
<!-- end lem-3 -->
<!-- head-5: body_md -->
The proof follows by applying Lemma \ref{Mazya 1} with $q$, $\alpha$, $\beta$  replaced  respectively by $\frac{2q}{q+2}$, $\alpha+\frac{q\beta}2$, $\left(\frac q2+1\right)\beta$ to the function $|u|^{\frac{q}2+1}$ which yields
\begin{align*}
	\|y^{\beta} u\|^{\frac q 2+1}_{L^q(\mathbb{R}^{N+1}_+)}\leq C\frac{q+2}{2}\|y^{\alpha}|\nabla u|y^{\frac{q\beta}2}|u|^{\frac q 2}\|_{L^1(\mathbb{R}^{N+1}_+)}.
\end{align*}
Applying then the  Cauchy-Schwarz inequality to the right hand side yields
\begin{align*}
	\|y^{\beta} u\|^{\frac q 2+1}_{L^q(\mathbb{R}^{N+1}_+)}\leq C\frac{q+2}{2}\|y^\alpha\nabla u\|_{L^2(\mathbb{R}^{N+1}_+)}\|y^\beta u\|^{\frac q 2}_{L^q(\mathbb{R}^{N+1}_+)}
\end{align*}
which is the required claim.

		\| u\|_{L^2_c(\mathbb{R}^{N+1}_+)}\leq C\|\nabla u\|_{L^1_c(\mathbb{R}^{N+1}_+)},\quad \forall u\in C_c^\infty\left(\overline{\mathbb{R}^{N+1}_+}\right),\quad \mbox{supp}\,u\subseteq \mathbb{R}^N\times [0,1].
	\end{align*}
	Let $m\in\mathbb{N}$; applying the latter inequality to $|u|^{m-1}u$ instead of $u$ and using Holder's inequality  we obtain
	\begin{align*}
		\| u\|^m_{L^{2m}_c(\mathbb{R}^{N+1}_+)}&\leq Cm\|\nabla u|u|^{m-1}\|_{L^1_c(\mathbb{R}^{N+1}_+)}\leq Cm\|\nabla u\|_{L^2_c(\mathbb{R}^{N+1}_+)}\|u\|^{m-1}_{L^{2m-2}_c(\mathbb{R}^{N+1}_+)}
	\end{align*}
	that is 
	\begin{align}\label{Induzione1}
		\| u\|_{L^{2m}_c(\mathbb{R}^{N+1}_+)}& \leq (Cm)^{\frac 1 m}\|\nabla u\|_{L^2_c(\mathbb{R}^{N+1}_+)}^{\frac 1 m}\|u\|^{1-\frac 1 m}_{L^{2(m-1)}_c(\mathbb{R}^{N+1}_+)}
	\end{align}
	which is valid for any $u\in C_c^\infty\left(\overline{\mathbb{R}^{N+1}_+}\right)$ with $ \mbox{supp}\,u\subseteq \mathbb{R}^N\times [0,1]$. The inequality above and  an easy inductive argument then proves that, for some positive  constant $C=C(m)>0$, one has 
	\begin{align}\label{Induzione2}
		\| u\|_{L^{2m}_c(\mathbb{R}^{N+1}_+)}& \leq C \|\nabla u\|_{L^2_c(\mathbb{R}^{N+1}_+)}^{1-\frac 1 m}\|u\|^{\frac 1 m}_{L^{2}_c(\mathbb{R}^{N+1}_+)},\quad \forall u\in C_c^\infty\left(\overline{\mathbb{R}^{N+1}_+}\right),\quad \mbox{supp}\,u\subseteq \mathbb{R}^N\times [0,1]
	\end{align}
	which is the required claim for $q=2m$. Indeed for $m=1$ the latter inequality is simply $	\| u\|_{L^{2}_c(\mathbb{R}^{N+1}_+)} \leq C \|u\|_{L^{2}_c(\mathbb{R}^{N+1}_+)}$ which is obviously true. Let us suppose \eqref{Induzione2} true for $m=k$; then applying  \eqref{Induzione1} we get  for any $u\in C_c^\infty\left(\overline{\mathbb{R}^{N+1}_+}\right)$ with $ \mbox{supp}\,u\subseteq \mathbb{R}^N\times [0,1]$ and for some positive constant $C$ depending on $k$
\begin{align*}
		\| u\|_{L^{2(k+1)}_c(\mathbb{R}^{N+1}_+)}& \leq C\|\nabla u\|_{L^2_c(\mathbb{R}^{N+1}_+)}^{\frac 1 {k+1}}\|u\|^{1-\frac 1 {k+1}}_{L^{2k}_c(\mathbb{R}^{N+1}_+)}\\[1ex]
		& \leq C\|\nabla u\|_{L^2_c(\mathbb{R}^{N+1}_+)}^{\frac 1 {k+1}}\left(C \|\nabla u\|_{L^2_c(\mathbb{R}^{N+1}_+)}^{1-\frac 1 k}\|u\|^{\frac 1 k}_{L^{2}_c(\mathbb{R}^{N+1}_+)}\right)^{1-\frac 1 {k+1}}\\[1ex]
		&= C\|\nabla u\|_{L^2_c(\mathbb{R}^{N+1}_+)}^{1-\frac 1 {k+1}}\|u\|^{\frac 1 {k+1}}_{L^{2}_c(\mathbb{R}^{N+1}_+)}
\end{align*}
	which is the claim for $m=k+1$.
	
	 The claim for a general $q>2$  then  follows by interpolating $L^q_c(\mathbb{R}^{N+1}_+)$ between $L^{2m}_c(\mathbb{R}^{N+1}_+)$ and $L^{2m+2}_c(\mathbb{R}^{N+1}_+)$   for a suitable $m\in \mathbb{N}$.

and

\begin{align*}

	= &-\alpha^2	\left\langle \nabla \psi ,\nabla \psi (u-1)^+\right\rangle_{L^2_\nu}+\alpha \left\langle\nabla\psi ,\nabla (u-1)^+\right\rangle_{L^2_\nu}\\[1ex]

	 (u-1)^+\right\rangle_{L^2_\nu}

\end{align*}

 Using the properties of $\psi$ in \eqref{distortion} and  of $\phi$ in \eqref{Prop-phi} we get
 \begin{align*}
 	\mathfrak{b}_{\alpha\psi}(1\wedge u,(u-1)^+)
 	\geq  &\left(-\alpha^2-2|a|\alpha^2-C_0-2|\alpha|C_0|a|\right) \int_{\mathbb{R}^{N+1}_+} (u-1)^+d\nu\\[1ex]
 		&+\alpha \left\langle\nabla\psi ,\nabla (u-1)^+\right\rangle_{L^2_\nu}	+2\alpha	\left\langle D_y \psi ,a\cdot\nabla_x 
 	(u-1)^+\right\rangle_{L^2_\nu}+2\left\langle \frac{D_y\phi}{\phi},a\cdot\nabla_x (u-1)^+\right\rangle_{L^2_\nu}.
 \end{align*}
 
To estimate the second term  we  integrate by parts and we use again the properties of $\psi$ and  of $\phi$ (which implies in particular the estimate $\frac{|D_y\left(\phi^2 y^c\right)|}{\phi^2y^c}\leq C $, for some $C>0$) obtaining
\begin{align*}

&=\left|-\alpha \int_{\mathbb{R}^{N+1}_+}\left(\Delta\psi+\frac{D_y\psi D_y (\phi^2 y^c)}{\phi^2 y^c} \right) (u-1)^+d\nu\right|\\[1ex]

\end{align*}
Similarly we get
\begin{align*}

(u-1)^+\right\rangle_{L^2_\nu}\right|

	2\left\langle \frac{D_y\phi}{\phi},a\cdot\nabla_x (u-1)^+\right\rangle_{L^2_\nu}

\end{align*}
Putting together all the above estimates and using again the elementary inequality  $\alpha^2+c_1|\alpha|+c_2 \le 2\alpha^2+c_3$, we finally get, for some positive constant $C>0$,

& \geq -C\left(\alpha^2	+1 \right)\int_{\mathbb{R}^{N+1}_+} (u-1)^+d\nu.
\end{align*}

\label{gaussian-exp}
	For any $k\in\mathbb{R}^N\setminus\{0\}$ one has, for some positive constant $C>0$,
	\begin{align*}
		|x-ky|^2+y^2\geq C\left(|x|^2+y^2\right),\qquad \forall x\in\mathbb{R}^n,\, y\in\mathbb{R}.
	\end{align*}

Using the  Cauchy-Schwarz inequality we get for $\epsilon>0$
\begin{align*}
	|x-ky|^2+y^2&=|x|^2+(1+|k|^2)y^2-2x\cdot k y\geq|x|^2+(1+|k|^2)y^2-2|x||k| |y|\\[1ex]
	&\geq|x|^2+(1+|k|^2)y^2-\frac 1 \epsilon |x|^2-\epsilon {|k|^2} y^2\\[1ex]
	&=(1-\frac{1}\epsilon)|x|^2+(1+(1-\epsilon)|k|^2)y^2.
\end{align*}
The required claim then follows by choosing $1<\epsilon<1+\frac{1}{|k|^2}$.

	Let $v=(b,c)\in\mathbb{R}^{N+1}$ with $b=0$ if $c =0$, $

	\begin{align*}
		\mathcal L =\mbox{Tr }\left(QD^2u\right)+\frac{ v\cdot \nabla }{y}
	\end{align*} endowed with the boundary conditions
\begin{align*}
	&\lim_{y \to 0} y^{\frac c \gamma}\, v \cdot \nabla u=0,\qquad \text{(if $c\neq 0$)},\qquad \qquad 
	\lim_{y \to 0} y^{\frac c \gamma}\, D_y u=0,\qquad \hspace{2ex}\text{(if $c=0$)},
\end{align*}
 generates a contractive analytic semigroup  in $L^2_{\frac c\gamma}$. The semigroup $(e^{t{\mathcal L}})_{t> 0}$ consists of integral operators	and its heat kernel $p_{{\mathcal L}}$, written  with respect the measure $y^\frac{c}{\gamma}dz$, satisfies for some  $C,k>0$, 
 \begin{align*}

 	\leq C t^{-\frac{N+1}{2}} y_1^{-\frac{c}{2\gamma}} \left(1\wedge \frac {y_1}{\sqrt t}\right)^{\frac{c}{2\gamma}} y_2^{-\frac{c}{2\gamma}} \left(1\wedge \frac{y_2}{\sqrt t}\right)^{\frac{c}{2\gamma}}\,\exp\left(-\dfrac{|z_1-z_2|^2}{kt}\right).
 \end{align*}
