<!-- theorem-1: body_md -->
\label{main} Assume $N=\mathbb{S}^{L-1}$ or
$SO(m)$. Let $\phi\in \dot{H}^1(\mathbb R^n, N)$
and $\psi\in (L^\infty\cap {H}^1)(\mathbb R^n, \phi^*TN)$. For $\varepsilon>0$, let
$v_\varepsilon:\mathbb{R}^n\times (0,\infty)\to N$
be a minimizer of the energy
${E}_\varepsilon$ over the admissible space $\mathcal{H}$ together with boundary data $(\phi,\psi)$. Then there exists a subsequence $\varepsilon_i\to 0$ such that $v_i=v_{\varepsilon_i}$ weakly converges to 
a map $v:\mathbb{R}^n\times \mathbb{R}_+\to 
N$ in $H^1(\mathbb{R}^n\times \mathbb{R}_+)$ so that
$$
(\partial_tv, \nabla v)\in L^\infty(\mathbb{R}_+, L^2(\mathbb R^n)).
$$
And $v$ is a weak solution of the wave map equation:
\begin{align}
\partial_t^2v-\Delta v \perp T_v N \ 
\ {\rm{in}}\ \ \mathbb{R}^n\times (0,\infty),   
\end{align}
along with the initial condition
\begin{align}
v(x,0)=\phi(x),\  \partial_tv(x,0)=\psi(x),
\ \ \forall x\in\mathbb{R}^n.
\end{align}
Furthermore, $v$ satisfies the following energy inequality:
\begin{align}
E(v(t))=\int_{\mathbb{R}^n}
(|\partial_tv|^2+|\nabla v|^2)(x,t)\,dx
\le E(v(0))=\int_{\mathbb{R}^n}
(|\psi|^2+|\nabla \phi|^2)(x)\,dx.
\end{align}

<!-- lemma-1: body_md -->
\label{existenemin} Assume $\phi\in \dot{H}^1(\mathbb R^n,\mathbb S^{L-1})$ 
and $\psi\in ({H}^1\cap L^\infty)(\mathbb R^n,\mathbb \phi^*T\mathbb{S}^{L-1})$. Then there exists a map $u_\varepsilon\in\mathcal{H}_{\phi,\varepsilon\psi}$ such that
\begin{align}\label{minimizer}
I_\varepsilon(u_\varepsilon)\le I_\varepsilon(v), \ \forall v\in\mathcal{H}_{\phi,\varepsilon\psi}. \end{align}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
First, it follows from Lemma \ref{nonempty} below that
\[
\alpha:=\inf\Big\{I_\varepsilon(v): 
\ v\in\mathcal{H}_{\phi,\varepsilon\psi}\Big\}
\le \varepsilon^2\int_{\mathbb R^n}|\nabla\phi|^2+C\varepsilon^3<\infty,
\]
where $C>0$ depends on 
$\|\nabla\phi\|_{L^2(\mathbb{R}^n)}$,
$\|\nabla\psi\|_{L^2(\mathbb{R}^n)}$,
and $\|\psi\|_{L^4(\mathbb{R}^n)}$.

Now let $\{u_i\}\subset\mathcal{H}_{\phi,\varepsilon\psi}$ be a minimizing sequence of $\alpha$, i.e.,
$$
I_\varepsilon(u_i)\rightarrow \alpha.
$$
By the weak compactness of $\mathcal{H}_{\phi,\varepsilon\psi}$, there exists a map $u_\varepsilon:\mathbb{R}^n\times\mathbb R_+\to\mathbb R^L$,
with $\big[u\big]_{\mathcal{H}}<\infty$,
such that, after passing to a subsequence if necessary,
$$
(\partial_t^2u_i,\nabla u_i)\rightharpoonup
(\partial_t^2u_\varepsilon,\nabla u_\varepsilon) 
\ \ {\rm{in}}\ \ L^2(\mathbb R^n),
$$
and
$$
(u_i,\partial_tu_i)\rightarrow (u_\varepsilon, \partial_tu_\varepsilon)
\ \ {\rm{in}}\ \ L^2_{\rm{loc}}(\mathbb{R}^n\times \mathbb{R}_+)\ \ {\rm{and\ a.e.\ in }}
\ \ \mathbb{R}^n\times\mathbb R_+.
$$
In particular, $|u_\varepsilon|=1$ a.e. in 
\ $\mathbb{R}^n\times\mathbb R_+$, and $(u_\varepsilon,\partial_t u_\varepsilon)\big|_{t=0}=(\phi,\varepsilon\psi)$, that is, $u_\varepsilon\in \mathcal{H}_{\phi,\varepsilon\psi}$. 
Hence 
\[
I_\varepsilon(u_\varepsilon)\ge\alpha.
\]
On the other hand, by the lower semi continuity of $I_\varepsilon$
on $\mathcal{H}$, we have
\[
I_\varepsilon(u_\varepsilon)\le \liminf_{i}I_\varepsilon(u_i)=\alpha.
\]
Thus $I_\varepsilon(u_\varepsilon)=\alpha$. This
completes the proof.

<!-- lemma-2: body_md -->
\label{nonempty} Assume $\phi\in \dot{H}^1(\mathbb R^n,\mathbb S^{L-1})$ 
and $\psi\in ({H}^1\cap L^\infty)(\mathbb R^n,\mathbb \phi^*T\mathbb{S}^{L-1})$.
Then 
\begin{align}\label{energy_ineq1}
\alpha:=\inf\Big\{I_\varepsilon(u):\  u\in \mathcal{H}_{\phi,\varepsilon\psi}\Big\}
\le \varepsilon^2\int_{\mathbb{R}^n}
|\nabla\phi|^2+C\varepsilon^3.  
\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since 
$\phi(x)+t\varepsilon \psi(x)$ satisfies
$$
|\phi(x)+t\varepsilon \psi(x)|^2=
|\phi(x)|^2+t^2\varepsilon^2|\psi(x)|^2
\ge 1,
$$
we can define the $\mathbb{S}^{L-1}$-valued map by
$$w_\varepsilon(x)=\displaystyle\frac{\phi(x)+t\varepsilon \psi(x)}
{|\phi(x)+t\varepsilon \psi(x)|}, \ (x,t)\in\mathbb{R}^n\times\mathbb R_+.
$$
By the definition of $\alpha$, we have
$$
\alpha\le I_\varepsilon(w_\varepsilon).
$$
To estimate $I_\varepsilon(w_\varepsilon)$,
we first calculate
\begin{align*}
\nabla w_\varepsilon(\cdot,t)&=
\frac{(\nabla\phi+t\varepsilon\nabla\psi)
|\phi+t\varepsilon\psi|^2-(\phi+t\varepsilon\psi)\langle\phi+t\varepsilon\psi,\nabla\phi+t\varepsilon\nabla\psi\rangle}
{|\phi+t\varepsilon\psi|^3}\\
&=\frac{1}{|\phi+t\varepsilon\psi|}\Big(
(\nabla\phi+t\varepsilon\nabla\psi)-
\langle \nabla\phi+t\varepsilon\nabla\psi, \frac{\phi+t\varepsilon\psi}{|\phi+t\varepsilon\psi|}\rangle\frac{\phi+t\varepsilon\psi}{|\phi+t\varepsilon\psi|}\Big)
\end{align*}
so that
$$
|\nabla w_\varepsilon|(x,t)
\le |\nabla\phi+t\varepsilon\nabla\psi|
\le |\nabla\phi|+t\varepsilon|\nabla\psi|.
$$
Also we calculate
$$
\partial_tw_\varepsilon(\cdot,t)
=\frac{\varepsilon\psi}{|\phi+t\varepsilon\psi|}
-\frac{\varepsilon (\phi+t\varepsilon\psi)\langle \phi+t\varepsilon\psi,\psi\rangle}{|\phi+t\varepsilon\psi|^3}
$$
so that
\begin{align*}
 \partial_t^2w_\varepsilon(\cdot, t)
 &=-\frac{2\varepsilon^2\psi\langle \phi+t\varepsilon\psi,\psi\rangle}{|\phi+t\varepsilon\psi|^3}
 -\frac{\varepsilon^2|\psi|^2(\phi+t\varepsilon\psi)}
 {|\phi+t\varepsilon\psi|^3}\\ &\ \ \ +3\frac{\varepsilon^2(\phi+t\varepsilon\psi)\langle \phi+t\varepsilon\psi,\psi\rangle^2}
 {|\phi+t\varepsilon\psi|^5}.
\end{align*}
Hence
\begin{align*}
|\partial_t^2w_\varepsilon|(\cdot, t)
&\le 6\varepsilon^2|\psi|^2
\end{align*}
This implies that 
\begin{align*}
I_\varepsilon(w_\varepsilon)
 &\le  \iint_{\mathbb{R}^n\times\mathbb{R}_+} e^{-t}(|\partial_t^2w_\varepsilon|^2+\varepsilon^2|\nabla w_\varepsilon|^2)\,dxdt\\
 &\le \int_{\mathbb{R}^n}
(\varepsilon^2|\nabla\phi|^2+C\varepsilon^3|\nabla\phi||\nabla\psi|+C\varepsilon^4 (|\nabla\psi|^2+|\psi|^4))\,dxdt\\
&\le\varepsilon^2\int_{\mathbb{R}^n}|\nabla\phi|^2\,dx+O(\varepsilon^3).
\end{align*}
Here we have used the fact that
\[
\|\psi\|_{L^4(\mathbb{R}^n)}\le \|\psi\|_{L^2(\mathbb{R}^n)}^\frac12
\|\psi\|_{L^\infty(\mathbb{R}^n)}^\frac12.
\]
This completes the proof.

<!-- lemma-3: body_md -->
\label{energyestimate}  Assume $\phi\in \dot{H}^1(\mathbb R^n,\mathbb S^{L-1})$ 
and $\psi\in ({H}^1\cap L^\infty)(\mathbb R^n,\mathbb \phi^*T\mathbb{S}^{L-1})$.
Let $u_\varepsilon$ be a minimizer of $I_\varepsilon$ in $\mathcal{H}_{\phi,\varepsilon\psi}$. Then
$E_\varepsilon(t)$ satisfies
\begin{equation}\label{energy_identity}
 E_\epsilon'(t)=-2D_\varepsilon(t).    
\end{equation}
In particular, $E_\epsilon\in W^{1,1}_{\rm{loc}}(\mathbb{R}_+)$, and is monotonically non-increasing with respect to $t$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The idea of proof is based on the inner variation of $u_\varepsilon$ along the $t$-direction.
For the detail, we follow closely the argument by \cite[Proposition 3.1]{Serra-Tilli2012}.
For any $\eta\in C_0^\infty(\mathbb{R}_+)$, define
$$
g(t)=\int_0^t\eta(s)\,ds.
$$
Then $g\in C^\infty(\mathbb{R}_+)$ vanishes near $t=0$. 

Write $u$ for $u_\varepsilon$. For a sufficiently small $\delta\in\mathbb{R}$, define
$$\alpha_\delta(t)=t-\delta g(t) 
\ \ {\rm{and}}\ \ U_\delta(x,t )=u(x, \alpha_\delta(t)), 
\ (x,t)\in\mathbb{R}^n\times\mathbb{R}_+.$$
Then $U_\delta$ satisfies the boundary condition (2.2), and $u\equiv U_0$ so that by the minimality, we have 
$$
I_\varepsilon(u)\le I_\varepsilon(U_\delta), 
$$
and hence
$$
\frac{d}{d\delta}\big|_{\delta=0}I_\varepsilon
(U_\delta)=0.
$$
Since $\alpha_\delta$ is invertible when
$\delta$ is sufficiently small, we  denote $\beta_\delta$ as the inverse so that $\beta_\delta\circ\alpha_\delta(t)=t$.
By change of variables, we can rewrite
$I_\varepsilon(U_\delta)$ as
\begin{align*}
I_\varepsilon(U_\delta)
&=\iint_{\mathbb{R}^n\times\mathbb{R}_+}
(e^{-\beta_\delta(s)}\beta_\delta'(s))
\big(|\partial_t^2u(\alpha_\delta')^2+\partial_tu\alpha_\delta''|^2+\varepsilon^2 |\nabla u|^2(x,s)\big)\,dxds.
\end{align*}
Direct calculations yield
\[
\frac{d}{d\delta}\big|_{\delta=0}\big(e^{-\beta_\delta}\beta_\delta'\big)
=e^{-t}(g'(t)-g(t)),
\]
and
\[
\frac{d}{d\delta}\big|_{\delta=0}\alpha'_\delta(t)=g'(t),
\ \ \frac{d}{d\delta}\big|_{\delta=0}\alpha''_\delta(t)=g''(t).
\]
Using these identities, we can deduce
\begin{align*}
0&=\frac{d}{d\delta}\big|_{\delta=0}I_\varepsilon
(U_\delta)\\
&=\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}(g'-g)(|\partial_t^2u|^2+\varepsilon^2|\nabla u|^2)(t,x)\,dxdt\\
&\ \ -\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}(4|\partial_t^2u|^2g'+2\langle \partial_t^2u, \partial_t u\rangle g'')(t,x)\,dxdt\\
&=\int_0^\infty
e^{-t}(g'(t)-g(t))L_\varepsilon(t)\,dt-\int_0^\infty
e^{-t}(4D_\varepsilon(t) g'(t)+2 I_\varepsilon'(t) g''(t))\,dt.
\end{align*}
Observe that
\begin{align*}
-\int_0^\infty e^{-t} g(t)L_\epsilon(t)\,dt 
&=\int_0^\infty g(t)H_\varepsilon'(t)\,dt\\
&=g(t)H_\varepsilon(t)\big|_{0}^\infty-\int_0^\infty H_\varepsilon(t)g'(t)\,dt\\
&=-\int_0^\infty H_\varepsilon(t)g'(t)\,dt,
\end{align*}
since $g(0)=H_\varepsilon(\infty)=0$.
Observe also that
$$g'(t)=\eta(t)
\ \ {\rm{and}}\ \ g''(t)=\eta'(t).$$
Therefore, we obtain that
\begin{align}
 \int_0^\infty
\big(L_\varepsilon(t)-e^t H_\varepsilon(t)-4D_\varepsilon(t)\big) e^{-t}\eta(t)\,dt-2\int_0^\infty
I_\varepsilon'(t) e^{-t}\eta'(t))\,dt=0.
\end{align}
By integration by parts, we have
\[
\int_0^\infty I_\varepsilon'(t) e^{-t}\eta'(t)\,dt
=\int_0^\infty I_\varepsilon'(t) e^{-t}\eta(t)\,dt
-\int_0^\infty I_\varepsilon''(t) e^{-t}\eta(t)\,dt.
\]
Thus we arrive at
\[
 \int_0^\infty
\Big[\big(L_\varepsilon(t)-e^t H_\varepsilon(t)-4D_\varepsilon(t)-2I_\varepsilon'(t)\big)
+2I_\varepsilon''(t)\Big]e^{-t}\eta(t)\,dt=0.
\]
This, combined with
$(e^t H_\epsilon(t))'
=e^t H_\varepsilon(t)-L_\varepsilon(t)$,
yields
\begin{equation}\label{energy_ineq2}
E_\varepsilon'(t)=\Big(\frac12 e^t H_\epsilon(t)+I_\varepsilon(t)-I_\varepsilon'(t)\Big)'=-2D_\varepsilon(t)\le 0.
\end{equation}
In particular, $E_\varepsilon(t)$ is monotonically decreasing with respect to $t$. This completes the proof.

<!-- lemma-4: body_md -->
\label{dualestimate} Assume $\phi\in \dot{H}^1(\mathbb R^n,\mathbb S^{L-1})$ 
and $\psi\in ({H}^1\cap L^\infty)(\mathbb R^n,\mathbb \phi^*T\mathbb{S}^{L-1})$.
Let $u_\varepsilon$ be a minimizer of $I_\varepsilon$ in $\mathcal{H}_{\phi,\varepsilon\psi}$. There exists  $C>0$ such that for any $h\in L^\infty(\mathbb{R}^n)\cap \dot{H}^1(\mathbb{R}^n)$, if
$u_\varepsilon$ is a minimizer of $I_\varepsilon$, then for a.e. $t>0$, 
\begin{align}\label{dual_est}
 \Big|\int_{\mathbb{R}^n} 
 \partial_t^2u_\varepsilon(x,t)\times u_\varepsilon(x,t)\cdot h(x)\,dx\Big|\le Ce^{t}\varepsilon^2\Big(\|h\|_{L^\infty(\mathbb {R}^n)}+\|\nabla h\|_{L^2(\mathbb{R}^n)}\Big).
\end{align}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Since $u_\varepsilon$ is a minimizer of $I_\varepsilon$ over
$\mathcal{H}_{\phi,\psi}^\varepsilon$, it is a weak solution of the Euler-Lagrange equation:
\begin{align}\label{EL1}
\Big[\partial_t^2(e^{-t}\partial_t^2u_\varepsilon)-\nabla\cdot(\varepsilon^2 e^{-t}\nabla u_\varepsilon)\Big]\times u_\varepsilon=0
\ \ {\rm{in}}\ \ \mathbb{R}^n\times(0,+\infty). 
\end{align}
Assume $h\in C_0^\infty(\mathbb{R}^n)$. As in \cite[Theorem 4.2]{Serra-Tilli2012}, let $\eta\in C^{1,1}(\mathbb{R})$ be given by
\[
\eta(t)=\begin{cases} 0 & t\le 0,\\
t^2 & 0<t<1,\\
2t-1 & t\ge 1.  
\end{cases}
\]
For $T>0$ and $\delta>0$, define
$$
\eta_\delta(t)=\delta\eta(\frac{t-T}{\delta}).
$$
Testing \eqref{EL1} by $\eta_\delta(t)h(x)$, we have that
\begin{align*}
0&=\iint_{\mathbb{R}^n\times\mathbb{R}_+}
\Big[e^{-t}\partial_t^2u_\varepsilon\times \partial_t^2(u_\varepsilon \eta_\delta h)+\varepsilon^2 e^{-t}\nabla u_\varepsilon\times \nabla(u_\varepsilon\eta_\delta h)\Big]\,dxdt\\
&=\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}[\partial_t^2u_\varepsilon\times (\partial_t^2u_\varepsilon\eta_\delta +u_\varepsilon \eta_\delta''+2\partial_t u_\varepsilon\eta_\delta')h+\varepsilon^2 
\nabla u_\varepsilon\times (\nabla u_\varepsilon h+u_\varepsilon \nabla h) \eta_\delta]\,dxdt\\
&=\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}[\partial_t^2u_\varepsilon\times (u_\varepsilon \eta_\delta''+2\partial_tu_\varepsilon\eta_\delta')h+\varepsilon^2\nabla u_\varepsilon\times (u_\varepsilon \nabla h) \eta_\delta]\,dxdt.
\end{align*}
Thus, we have
\begin{align}\label{EL2}
\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}\partial_t^2u_\varepsilon\times u_\varepsilon \eta_\delta''h\,dxdt&+2\iint_{\mathbb{R}^n\times\mathbb{R}_+}e^{-t}\partial_t^2u_\varepsilon\times \partial_tu_\varepsilon\eta_\delta' h\,dxdt\nonumber\\
&+\varepsilon^2\iint_{\mathbb{R}^n\times\mathbb{R}_+}e^{-t}\nabla u_\varepsilon\times u_\varepsilon \nabla h \eta_\delta\,dxdt=0.
\end{align}
Since $\eta_\delta''(t)=2\delta^{-1}\chi_{(T,T+\delta)}(t)$, the first term of \eqref{EL2} becomes
\begin{align*}
\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}\partial_t^2u_\varepsilon\times u_\varepsilon \eta_\delta''h\,dxdt
&=\frac{2}{\delta}\int_{T}^{T+\delta} 
\int_{\mathbb{R}^n} e^{-t} \partial_t^2 u_\varepsilon\times u_\varepsilon h\,dxdt
\nonumber\\
&\rightarrow 2e^{-T}
\int_{\mathbb{R}^n} \partial_t^2u_\varepsilon\times u_\varepsilon h\,dx,
\ {\rm{as}}\ \ \delta\to 0.
\end{align*}
The second term of \eqref{EL2} can be estimated as follows.
\begin{align*}
&2\iint_{\mathbb{R}^n\times\mathbb{R}_+}e^{-t}
\partial_t^2u_\varepsilon\times \partial_tu_\varepsilon\eta_\delta' h\,dxdt\\
&
=4\int_{T}^{T+\delta} \frac{t-T}{\delta}
e^{-t}\int_{\mathbb{R}^n} \partial_t^2u_\varepsilon\times 
\partial_tu_\varepsilon h\,dxdt\\
&\quad+4\int_{T+\delta}^\infty  
e^{-t}\int_{\mathbb{R}^n} \partial_t^2u_\varepsilon\times 
\partial_tu_\varepsilon h\,dxdt,
\end{align*}  
hence we have
\begin{align*}
&\Big|2\iint_{\mathbb{R}^n\times\mathbb{R}_+}e^{-t}\partial_t^2u_\varepsilon\times \partial_tu_\varepsilon\eta_\delta' h\,dxdt\Big|\\
&\le 4\int_{T}^{\infty} e^{-t}\int_{\mathbb{R}^n} 
|\partial_t^2u_\varepsilon\times \partial_tu_\varepsilon| |h|\,dxdt\\
&\le C\|h\|_{L^\infty(\mathbb{R}^n)}
\Big(\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}|\partial_t^2u_\varepsilon|^2\,dxdt\Big)^\frac12 
\Big(\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}|\partial_tu_\varepsilon|^2\,dxdt\Big)^\frac12\\
&\le 
C\|h\|_{L^\infty(\mathbb{R}^n)}
\Big(\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}|\partial_t^2u_\varepsilon|^2\,dxdt\Big)^\frac12\\
&\quad\cdot\Big(\iint_{\mathbb{R}^n\times\mathbb{R}_+}
e^{-t}|\partial_t^2u_\varepsilon|^2\,dxdt+\int_{\mathbb{R}^n}|\partial_tu_\varepsilon(x,0)|^2\,dx\Big)^\frac12\\
&\le C\varepsilon^2 \|h\|_{L^\infty(\mathbb{R}^n)}.
\end{align*}  

The third term of \eqref{EL2} can be
estimated by 
\begin{align*}
&\lim_{\delta\to 0} 
\Big|\iint_{\mathbb{R}^n\times \mathbb{R}_+}e^{-t}\nabla u_\varepsilon\times u_\varepsilon \nabla h \eta_\delta\,dxdt\Big|\\
&=2\Big|\iint_{\mathbb{R}^n\times\mathbb{R}_+}e^{-t}(t-T)^+\nabla u_\varepsilon\times u_\varepsilon \nabla h\,dxdt\Big|\\
&\le 2\iint_{\mathbb{R}^n\times\mathbb{R}_+}e^{-t}(t-T)^+|\nabla u_\varepsilon||\nabla h|\,dxdt\\
&\le 2\sum_{k=0}^\infty
e^{-k}(k+1)\Big(\int_{T+k}^{T+k+1}\int_{\mathbb{R}^n}|\nabla u_\varepsilon|^2\,dxdt\Big)^\frac12 
\Big(\int_{T+k}^{T+k+1}\int_{\mathbb{R}^n}|\nabla h|^2\,dxdt\Big)^\frac12\\
&\le C\|\nabla h\|_{L^2(\mathbb{R}^n)}
\sup_{k\ge 0} \Big(\int_{T+k}^{T+k+1}\int_{\mathbb{R}^n}|\nabla u_\varepsilon|^2\,dxdt\Big)^\frac12\\
&\le C\|\nabla h\|_{L^2(\mathbb{R}^n)}.
\end{align*}
Substituting these estimates into \eqref{EL2} yields \eqref{dual_est}
holds for a.e. $T>0$.

<!-- lemma-5: body_md -->
\label{sharp_est0} Assume $\phi\in \dot{H}^1(\mathbb R^n,\mathbb S^{L-1})$ 
and $\psi\in ({H}^1\cap L^\infty)(\mathbb R^n,\mathbb \phi^*T\mathbb{S}^{L-1})$.
Let $u_\varepsilon$ be a minimizer of $I_\varepsilon$ in $\mathcal{H}_{\phi,\varepsilon\psi}$.  Then
\begin{align}\label{sharp_est}
 E_\varepsilon(0)
 =I_\varepsilon(0)-I_\varepsilon'(0)+\frac12 H_\varepsilon(0)\le
 \frac12\varepsilon^2\int_{\mathbb{R}^n}
 (|\nabla\phi|^2+|\psi|^2)(x)\,dx+C\varepsilon^3.
\end{align}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
From the definition, we know
\begin{align}\label{I-est}
I_\varepsilon(0)=\frac12\int_{\mathbb{R}^n}|\partial_tu_\varepsilon(x,0)|^2\,dx
=\frac12\int_{\mathbb{R}^n}\varepsilon^2|\psi(x)|^2\,dx. 
\end{align}
By \eqref{energy_ineq1}, we have 
\begin{align}\label{H-est}
\frac12H_\varepsilon(0)
=\frac12 I_\varepsilon(u_\varepsilon)
\le\frac12\big(\int_{\mathbb{R}^n}
\varepsilon^2|\nabla\phi(x)|^2\,dx+C\varepsilon^3\big).
\end{align}
Now we claim that
\begin{align}\label{I'-est}
 |I_\epsilon'(0)|
 \le C\varepsilon^3.
\end{align}
To do it, first observe that 
$|u_\varepsilon(x,t)|=1$ implies that
$\langle \partial_tu_\varepsilon, u_\varepsilon\rangle (x,t)=0$
and hence
\[
|\partial_tu_\varepsilon|^2(x,t)
=|\partial_tu_\varepsilon\times u_\varepsilon|^2(x,t),
\ (t,x)\in \mathbb{R}^n\times \mathbb{R}_+
\]
so that
\[
I_\varepsilon(t)=\int_{\mathbb{R}^n}
|\partial_tu_\varepsilon\times u_\varepsilon|^2(x,t)\,dx.
\]
Next we choose sufficiently small $\delta>0$ and estimate
\begin{align}\label{energy_est1}
\Big|\frac{1}{\delta}\int_0^\delta I_\varepsilon'(t)\,dt\Big|
&\le \Big|\frac{2}{\delta}\int_0^\delta
\int_{\mathbb{R}^n}\langle (\partial_t^2u_\varepsilon\times 
u_\varepsilon)(x,t), (\partial_tu_\varepsilon\times u_\varepsilon)(x,t)-(\partial_tu_\varepsilon\times u_\varepsilon)(x,0)\rangle\,dxdt\Big|\nonumber\\
&\ \ +\Big|\frac{2}{\delta}\int_0^\delta
\int_{\mathbb{R}^n}\langle (\partial_t^2u_\varepsilon\times u_\varepsilon)(x,t), (\partial_tu_\varepsilon\times u_\varepsilon)(x,0)\rangle\,dxdt\Big|\nonumber\\
&=A_\varepsilon+B_\varepsilon.
\end{align}
From \eqref{dual_est} 
\begin{align*}
 B_\varepsilon&=\Big|\frac{2\varepsilon}{\delta}\int_0^\delta
\int_{\mathbb{R}^n}\langle (\partial_t^2u_\varepsilon\times u_\varepsilon)(x,t), (\psi\times \phi)(x)\rangle\,dxdt\Big|\\
 &\le  
2\varepsilon\sup_{0\le t\le\delta}
\Big|\int_{\mathbb{R}^n}\langle (\partial_t^2u_\varepsilon\times u_\varepsilon)(x,t), (\psi\times \phi)(x)\rangle\,dx\Big|\\
&\le C\varepsilon^3\big(\|\nabla(\psi\times\phi)\|_{L^2(\mathbb{R}^n)}
+\|\psi\times\phi\|_{L^\infty(\mathbb{R}^n)}\big)\\
&\le C\varepsilon^3
\Big(\|\psi\|_{L^\infty(\mathbb{R}^n)}
+\|\nabla\psi\|_{L^2(\mathbb{R}^n)}
+\|\psi\|_{L^\infty(\mathbb{R}^n)}
\|\nabla\phi\|_{L^2(\mathbb{R}^n)}\Big)\\
&\le C\varepsilon^3.
\end{align*}
As for $A_\varepsilon$, we can estimate as follows
\begin{align*}
A_\varepsilon   
&=\Big|\frac{2}{\delta}\int_0^\delta
\int_{\mathbb{R}^m}\langle (\partial_t^2u_\varepsilon\times 
u_\varepsilon)(x,t), \int_0^t \partial_t(\partial_tu_\varepsilon\times u_\varepsilon)(x, \tau)\,d\tau\rangle\,dxdt\Big|\\
&=
\Big|\frac{2}{\delta}\int_0^\delta
\int_{\mathbb{R}^n}\langle (\partial_t^2u_\varepsilon\times 
u_\varepsilon)(x,t), \int_0^t (\partial_t^2u_\varepsilon\times u_\varepsilon)(x, \tau)\,d\tau\rangle\,dxdt\Big|\\
&\le \frac{2}{\delta}
\Big(\int_0^\delta
\int_{\mathbb{R}^n}|\partial_t^2u_\varepsilon(x,t)|^2\,dxdt\Big)^\frac12
\Big(\int_0^\delta
\int_{\mathbb{R}^n}t\int_0^t|\partial_t^2u_\varepsilon|^2(x,\tau)\,d\tau\,dxdt\Big)^\frac12\\
&\le 2\int_0^\delta
\int_{\mathbb{R}^n}|\partial_t^2u_\varepsilon(x,t)|^2\,dxdt
\to 0,\ \ {\rm{as}}\ \ \delta\to 0.
\end{align*}
Substituting the estimates for $A_\varepsilon$
and $B_\varepsilon$ into \eqref{energy_est1} and sending $\delta\to 0$ yields
\eqref{I'-est}.
It is readily seen that \eqref{sharp_est}
follows from \eqref{I-est}, \eqref{H-est},
and \eqref{I'-est}.

\end{align}

It is similar to that of Lemma \ref{dualestimate}. Here we give a sketch. First, as in the previous subsection, we know that $U_\varepsilon$ satisfies
\begin{align*}
0&=\iint_{\mathbb{R}^n\times\mathbb{R}_+}
\Big[\langle e^{-t}\partial_t^2U_\varepsilon, \partial_t^2(U_\varepsilon \eta_\delta h)\rangle +\varepsilon^2 \langle e^{-t}\nabla U_\varepsilon,  \nabla(U_\varepsilon\eta_\delta h)\rangle \Big]\,dxdt\\
&=\iint_{\mathbb{R}^n\times\mathbb{R}_+}
\Big[\langle e^{-t}\partial_t^2U_\varepsilon, (U_\varepsilon \eta_\delta''+2\partial_tU_\varepsilon\eta_\delta')h\rangle +
\varepsilon^2 \langle e^{-t}\nabla U_\varepsilon, U_\varepsilon \nabla h\rangle \eta_\delta\Big]\,dxdt,
\end{align*}
where $h\in C_0^\infty(\mathbb{R}^n, {\it so}(m))$ and $\eta_\delta\in C^{1,1}(\mathbb{R})$ is given by Lemma \ref{dualestimate}.

As in Lemma \ref{dualestimate}, we can bound
\begin{align*}
\iint_{\mathbb{R}^n\times (0,\infty)}
e^{-t}\langle \partial_t^2 U_\varepsilon, U_\varepsilon \eta_\delta''h\rangle\,dxdt
&=\frac{2}{\delta}\int_{T}^{T+\delta}\int_{\mathbb{R}^n} 
e^{-t}\langle \partial_t^2 U_\varepsilon, U_\varepsilon h\rangle \,dxdt\\
&\rightarrow  2e^{-T}
\int_{\mathbb{R}^n} \langle\partial_t^2 U_\varepsilon, U_\varepsilon h\rangle\,dx, \  {\rm{as}}\ \delta\to 0,
\end{align*}
\begin{align*}
&\Big|\iint_{\mathbb{R}^n\times (0,\infty)}
e^{-t}\langle \partial_t^2 U_\varepsilon, 2\partial_t U_\varepsilon \eta_\delta'h\rangle\,dxdt\Big|\\
&=\Big|4\int_T^{T+\delta} \frac{t-T}{\delta}e^{-t}\int_{\mathbb{R}^n}
\langle\partial_t^2 U_\varepsilon, \partial_t U_\epsilon h\rangle\,dxdt
+4\int_{T+\delta}^{\infty}e^{-t}\int_{\mathbb{R}^n}\langle\partial_t^2 U_\varepsilon, \partial_t U_\varepsilon h\rangle \,dxdt\Big|\\
&\le C\big\|h\big\|_{L^\infty(\mathbb{R}^n)}
\Big(\iint_{\mathbb{R}^n\times (0,\infty)} e^{-t}|\partial_t^2 U_\varepsilon|^2\,dxdt\Big)^\frac12
\Big(\iint_{\mathbb{R}^n\times (0,\infty)} e^{-t}|\partial_t U_\varepsilon|^2\,dxdt\Big)^\frac12\\
&\le C\varepsilon^2 \big\|h\big\|_{L^\infty(\mathbb{R}^n)},
\end{align*}
and
\begin{align*}
&\lim_{\delta\to 0}
\Big|\iint_{\mathbb{R}^n\times (0,\infty)} e^{-t}\langle\nabla U_\varepsilon, U_\varepsilon\nabla h\rangle\,dxdt\Big|\\
&=2\Big|\iint_{\mathbb{R}^n\times (0,\infty)}e^{-t}(t-T)^+\langle\nabla U_\varepsilon, U_\varepsilon\nabla h\rangle\,dxdt\Big|\\
&\le C\big\|\nabla h\big\|_{L^2(\mathbb{R}^n)} \sup_{k\ge 0}
\Big(\iint_{\mathbb{R}^n\times [T+k, T+k+1]}|\nabla U_\varepsilon|^2\,dxdt\Big)^\frac12\\
&\le C\big\|\nabla h\big\|_{L^2(\mathbb{R}^n)}.
\end{align*}
Putting all these estimates together yields \eqref{dualestimate2}.

\label{energyineq2}
Assume $\phi\in \dot{H}^1(\mathbb R^n,SO(m))$ 
and $\psi\in ({H}^1\cap L^\infty)(\mathbb R^n, \phi^*T(SO(m)))$.
Let $U_\varepsilon$ be a minimizer of $I_\varepsilon$ in $\mathcal{H}_{\phi,\varepsilon\psi}$.
Then \begin{align}\label{sharp_est2}
 E_\varepsilon(0)
 =I_\varepsilon(0)-I_\varepsilon'(0)+\frac12 H_\varepsilon(0)\le
 \frac12\varepsilon^2\int_{\mathbb{R}^n}
 (|\nabla\phi|^2+|\psi|^2)(x)\,dx+C\varepsilon^3.
\end{align}

<!-- proof-8: body_md -->
Similar to Lemma \ref{sharp_est0}. First, by definition
\[
I_\varepsilon(0)=\frac12\int_{\mathbb{R}^n}\varepsilon^2|\psi|^2\,dx,
\]
and by \eqref{energyupbound},
\[
\frac12 H_\varepsilon(0)=\frac12 I_\varepsilon(U_\varepsilon)
\le \frac12(\int_{\mathbb{R}^n}\varepsilon^2|\nabla\phi|^2\,dx+C\varepsilon^3).
\]
Thus, it remains to show
\[
|I_\varepsilon'(0)|\le C\varepsilon^3.
\]

Note that $U_\varepsilon:\mathbb{R}^n\times (0,\infty)\to SO(m)$, one has
\[
I_\varepsilon(t)=\int_{\mathbb{R}^n}|U_\varepsilon^T\partial_t U_\varepsilon|^2\,dx
\]
Hence, for $\delta>0$ small, we can estimate
\begin{align*}
\Big|\frac{1}{\delta}\int_0^\delta I_\varepsilon'(t)\,dt\Big|
&\le  
\Big|\frac{2}{\delta}\int_0^{\delta} \langle \partial_t(U_\varepsilon^T\partial_tU_\varepsilon), U_\varepsilon^T\partial_t U_\varepsilon-U_\varepsilon^T(x,0)\partial_t U_\varepsilon(x,0)\rangle\,dxdt\Big|\\
&+
\Big|\frac{2}{\delta}\int_0^{\delta} \langle \partial_t(U_\varepsilon^T\partial_tU_\varepsilon), U_\varepsilon^T(x,0)\partial_t U_\varepsilon(x,0)\rangle\,dxdt\Big|.   
\end{align*}
Now we can apply the same estimates as in Lemma \ref{sharp_est0}, with \eqref{dual_est} replaced by
\eqref{dual_est2}, to bound
\begin{align*}
&\Big|\frac{2}{\delta}\int_0^{\delta} \langle \partial_t(U_\varepsilon^T\partial_tU_\varepsilon), U_\varepsilon^T(x,0)\partial_t U_\varepsilon(x,0)\rangle\,dxdt\Big|\\
&\le C\varepsilon^3 \big(\|\psi\|_{L^\infty(\mathbb{R}^n)} 
+\|\nabla\psi\|_{L^2(\mathbb{R}^n)}+\|\psi\|_{L^\infty(\mathbb{R}^n)}\|\nabla\phi\|_{L^2(\mathbb{R}^n)}\big)\le C\varepsilon^3,
\end{align*}
and
\begin{align*}
&\Big|\frac{2}{\delta}\int_0^{\delta} \big\langle \partial_t(U_\varepsilon^T\partial_tU_\varepsilon), U_\varepsilon^T\partial_t U_\varepsilon-U_\varepsilon^T(x,0)\partial_t U_\varepsilon(x,0)\big\rangle\,dxdt\Big|\\
&\le C\int_0^{\delta}\int_{\mathbb{R}^n}|\partial_t^2 U_\varepsilon|^2\,dxdt\rightarrow 0, 
\end{align*}
as $\delta\to 0$.   Putting all these estimates together yields \eqref{sharp_est2}.
