<!-- theorem-1: body_md -->
\label{thm 1.1}
    Let $\gamma\in(1,3)$ and $\mu\in[\frac12,1)$.  Let $t_0<0$ with $|t_0|$ sufficiently small. There exists a finite-codimensional set of $C^{\lfloor\beta \rfloor, \beta- \lfloor\beta \rfloor }$ initial data at $t=t_0$ such that they launch strong  solutions to the vacuum boundary Euler system \eqref{Eulerh}-\eqref{IC} on $[t_0,0)$ transitioning to be $C^{1-\mu}$-H\"{o}lder continuous near $x=0$ at $t=0$. 
    The vacuum boundary remains at $x=0$ for $t\in [t_0,0)$ and the first gradient blow-up occurs at $(t,x)=(0,0)$.  The velocity and the speed of the sound are bounded in $L^2$ for each   $t\in [t_0,0)$  and the density remains positive $\rho>0$ for all $(t,x)\in[t_0,0)\times(0,\infty)$.
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
Initial data \eqref{IC} at $t_0<0$ 
leading to gradient blowups in Theorem \ref{thm 1.1}  
have the following asymptotics near the boundary $x=0$
    \begin{equation}
    \begin{split}\label{initial expansion}
         u_0(x) &=  \frac{2}{(\gamma+1)t_0} x + \sum_{i=2}^{\lfloor\beta \rfloor} a_i x^i + a_0 x^{\beta} + o(x^{\beta}),\\
        c_0(x)&=  -\frac{\gamma-1}{(\gamma+1)t_0} x + \sum_{i=2}^{\lfloor\beta \rfloor} b_i x^i -\frac{\gamma-1}{2} a_0 x^{\beta} + o(x^{\beta}),
    \end{split}
    \end{equation}
    where $a_0>0$ and $b_i:= b_i( a_2,\dots, a_i)$. Moreover, for given $0<x\ll 1$, as $t\to 0^-$ solutions constructed in Theorem \ref{thm 1.1} satisfy 
   \begin{equation}\begin{aligned}\label{1.7}
        u(t,x) = -\tilde a x^{1-\mu}+o(|t|),\  c(t,x) = \tilde b x^{1-\mu}+o(|t|)
   \end{aligned}\end{equation}
with $ \tilde a, \tilde b>0$.
<!-- end remark-1 -->
<!-- notation--1: body_md -->
Throughout the article, we use $\lfloor\beta \rfloor$  and  $\lceil\beta \rceil$ to denote the floor function  and the ceiling function respectively:  $\lfloor\beta \rfloor = \max\{ n \in \mathbb Z: n \le \beta \}$ and   $\lceil\beta \rceil = \min\{ n \in \mathbb Z: n \ge \beta \}$. Note that $\lceil\beta \rceil = \lfloor\beta \rfloor +1  $ if $\beta \notin \mathbb Z$ and  $\lceil\beta \rceil = \lfloor\beta \rfloor  $ if $\beta \in  \mathbb Z$.

\end{equation}

<!-- proposition-1: body_md -->
\label{prop bar z}
Let $\gamma \in (1,3)$ and $\mu \in \bigl[\tfrac12,1\bigr)$.  
The steady state $\bar z$ satisfies the following properties:
\begin{enumerate}
    \item[$(1)$] The first derivative is given by
    \[
        \partial_y \bar z = \frac{-4(\mu-1)\bar U}{(\gamma+1)\bar U-2}.
    \]
    Moreover, $\partial_y \bar z$ is monotone increasing from $\partial_y \bar z(0)=-\frac{4\mu}{\gamma+1}$ to $\lim_{y\to \infty} \partial_y \bar z(y)=0$. In particular, $ -\frac{4\mu}{\gamma+1} \le \partial_y \bar z <0$
    
    \item[$(2)$] For any $y \in [0,\infty)$, one has
    \[
        1 + \frac{\gamma+1}{4}\,\partial_y \bar z \in [\,1-\mu,\,1\,).
    \]
    
    \item[$(3)$] As $0<y\ll 1$, the function $\bar z$ admits the expansion
    \begin{equation}\label{z y<<1}
        \bar z(y) = -\frac{4\mu}{\gamma+1} y \;+\; c_1 y^{\beta} 
        \;+\; O\!\left(y^{2\beta -1 }\right),
    \end{equation}
    where $c_1=c_1 (\gamma,\mu,K)>0$.                                             
    Furthermore, there exists a constant $A>0$ such that, for $0<y\ll 1$,
    \begin{align}
        |\partial_y \bar z(y)| &\leq A, \notag \\ 
        |\partial_y^2 \bar z(y)| &\leq A\, y^{\beta-2}, \notag \\
        |\partial_y^i \bar z(y)| &\leq 
        \begin{cases}
            A\, y^{j(\beta-1)+1-i}, 
            & \text{if } \mu\in \mathfrak S,\; j\in \mathbb N_{\geq 1},\; (j-1)(\beta-1))+1 < i \leq j(\beta-1)+1, \\[6pt]
            A\, y^{\beta-i}, 
            & \text{if } \mu \notin \mathfrak S,
        \end{cases}\label{payi z y<<1}
    \end{align}
    for all $i \in \mathbb N_{\geq 3}$.
    
    \item[$(4)$] As $y \to \infty$, one has the asymptotic behavior
    \begin{equation}\label{z y>>1}
        \bar z(y) = -c_2 y^{1-\mu} + o\!\left(y^{1-\mu}\right),
    \end{equation}
    where $c_2:=c_2(\gamma,\mu,K)>0$. Moreover, there exists a constant $A>0$ such that, for $y \gg 1$,
    \begin{equation}\label{payi z y>>1}
        |\partial_y^i \bar z(y)| \leq A\, y^{1-\mu-i}, 
    \end{equation}
    for all $i \in \mathbb N_{\geq 1}$.
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For $(1)$, since $\bar z = -2y\bar U$, we compute
\[
    \partial_y \bar z = -2\Bigl(\bar U + y\,\partial_y \bar U \Bigr)
    = -2\Biggl(\bar U - \frac{[(\gamma+1)\bar U - 2\mu]\bar U}{(\gamma+1)\bar U-2}\Biggr)
    = \frac{-4(\mu-1)\bar U}{(\gamma+1)\bar U - 2}.
\]
Since the function $g(r) = \frac{-4(\mu-1)r}{(\gamma+1)r - 2}$ is strictly decreasing on the interval $r\in[0, \frac{2\mu}{\gamma+1}]$ and $\bar U(y)$ is strictly decreasing from $\frac{2\mu}{\gamma+1}$ to $0$ as $y$ increases from $0$ to $\infty$, $ \partial_y \bar z$ is increasing from $\partial_y \bar z(0)=-\frac{4\mu}{\gamma+1}$ to $\lim_{y\to \infty} \partial_y \bar z(y)=0$.

For $(2)$, one finds
\[
    1+\frac{\gamma+1}{4}\,\partial_y \bar z
    = \frac{(2-\mu)\bar U - \tfrac{2}{\gamma+1}}{\bar U - \tfrac{2}{\gamma+1}}.
\]
The function $f(r)=\frac{(2-\mu)r-\tfrac{2}{\gamma+1}}{r-\tfrac{2}{\gamma+1}}$ is decreasing for $r\in\bigl[0,\tfrac{2}{\gamma+1}\mu\bigr]$. Since $\bar U$ is decreasing in $y$, it follows that
\[
    1-\mu = \frac{(2-\mu)\bar U(0) - \tfrac{2}{\gamma+1}}{\bar U(0)-\tfrac{2}{\gamma+1}}
    < 1+\frac{\gamma+1}{4}\,\partial_y \bar z
    \leq \lim_{y\to\infty} \frac{(2-\mu)\bar U(y) - \tfrac{2}{\gamma+1}}{\bar U(y)-\tfrac{2}{\gamma+1}} = 1,
\]
where we used \eqref{Asy at C} in the last equality.

For $(3)$, the expansion \eqref{z y<<1} follows directly from \eqref{Asy at C}. Since $\beta \geq 2$, the bounds on $\partial_y \bar z$ and $\partial_y^2 \bar z$ are immediate. For higher-order derivatives, note that $\bar z \in C^{\infty}$ if $\mu \in \mathfrak S$, and $\bar z \in C^{[\beta],\beta-[\beta]}$ if $\mu \notin \mathfrak S$. This explains the dichotomy in \eqref{payi z y<<1}. A direct computation yields the stated bounds.

For $(4)$, the asymptotics \eqref{z y>>1} follow from \eqref{Asy at A}, and the derivative estimates \eqref{payi z y>>1} are then straightforward.

\label{le: supp chi}
Let $\chi$ be a smooth solution to                         
\eqref{pde of chi} with \eqref{def chi}. 
Then, $\chi$ has the following properties
\begin{enumerate}
\item[$(1)$]  $\text{supp}(\chi)\subset [0, 2y_0e^{\tau-\tau_0}) $,
\item[$(2)$] $\text{supp}(1-\chi)\subset [y_0e^{(1-\mu)(\tau-\tau_0)},\infty) $,
\item[$(3)$] $\chi(\tau,y)\in[0,1]$, $\partial_y \chi(\tau,y)\in[-\frac{2}{y_0},0]$ for $\tau>\tau_0$ and $y\geq0$,
\item[$(4)$] $|\partial_y^i \chi(\tau,y) |\leq A y^{-i}$ for $y\in[y_0,\infty)$ with $A=A(i)$.
\end{enumerate}

<!-- proof-3: body_md -->
(1) and (2) follow directly from Proposition \ref{prop Y(tau,p)}. 
    For (3),                     
    we first use \eqref{pde of chi} and \eqref{Lagrangian flow map} to derive
	\[
		\chi(\tau,Y(\tau,p)) = \chi(\tau_0,Y(\tau_0,p)) \in[0,1].
	\]
	Hence, the first claim in (3) follows. 
	 We then apply $\partial_y$ to \eqref{pde of chi} to obtain
	\begin{align*}
		\partial_\tau \partial_y\chi+(y+\frac{\gamma+1}{4}\bar z )\partial_y \partial_y\chi = -(1+\frac{\gamma+1}{4}\partial_y \bar z) \partial_y\chi.
	\end{align*}
	By using the flow map $Y(\tau,p)$ defined in \eqref{Lagrangian flow map}, we have 
	\begin{align*}
		\partial_\tau \partial_y\chi(\tau,Y(\tau,p))= -(1+\frac{\gamma+1}{4}\partial_y \bar z) \partial_y\chi(\tau,Y(\tau,p)).
	\end{align*}
	We then integrate the above ODE by using the integral factor 
	\begin{equation} \label{def IF}
	f(\tau) = \int (1+\frac{\gamma+1}{4}\partial_y \bar z (Y(\tau',p) )) d\tau' 
	\end{equation} 
	to obtain
	\begin{equation}\label{pa_y chi}
		\partial_y\chi(\tau,Y(\tau,p)) = \partial_y\chi(\tau_0,Y(\tau_0,p)) e^{-\int_{\tau_0}^\tau (1+\frac{\gamma+1}{4}\partial_y \bar z(Y(\tau',p)))\ d\tau'}.
	\end{equation}
    By using Proposition \ref{prop bar z} (2) and \eqref{range of flow map}, it then follows that
	\begin{align*}
		|\partial_y\chi(\tau,Y(\tau,p))| \leq |\partial_y\chi(\tau_0,Y(\tau_0,p))| e^{-(1-\mu)(\tau-\tau_0)} \leq |\partial_y\chi(\tau_0,Y(\tau_0,p))|.
	\end{align*}
	Together with \eqref{def chi}, it leads to the second claim in (3).  
    
    For (4), first note that since $\chi_0$ has a compact support, there exists a constant $A:=A(i)$ such that
	\begin{equation}\label{ineq pa_yi chi}
		\mid\partial_y^i \chi_0(y)\mid \leq A y^{-i} \text{ for any } y \ge y_0. 
	\end{equation}
	We then apply $\partial_y^i $ to  \eqref{pde of chi} and obtain
	\begin{align*}
		\partial_\tau \partial_y^i\chi+(y+\frac{\gamma+1}{4}\bar z )\partial_y \partial_y^{i}\chi = -i(1+\frac{\gamma+1}{4}\partial_y \bar z(y)) \partial_y^i\chi - \sum_{j=1}^{i-1} c_{ij} \partial_y^{j+1}\bar z (y) \partial_y^{i-j}\chi
	\end{align*}
	where the last summation appears only when $i\geq 2$ and $c_{ij}$ are constants. Along $Y(\tau,p)$, we have   
\begin{align*}
		\partial_\tau \partial_y^i\chi(\tau,Y(\tau,p)) = -i(1+\frac{\gamma+1}{4}\partial_y \bar z(Y(\tau,p))) \partial_y^i\chi (\tau,Y(\tau,p))- \sum_{j=1}^{i-1} c_{ij} \partial_y^{j+1}\bar z (Y(\tau,p)) \partial_y^{i-j}\chi(\tau,Y(\tau,p)).
	\end{align*}
	Then, by using the integral factor $if(\tau)$ where $f(\tau)$ is defined in \eqref{def IF}, we obtain
	\begin{equation}\label{eq payi chi}
    \begin{split}
		\partial_y^i\chi(\tau,Y(\tau,p)) &= \partial_y^i\chi(\tau_0,Y(\tau_0,p)) e^{i(f(\tau_0)-f(\tau))}\\
        &- \sum_{j=1}^{i-1}c_{ij}\int_{\tau_0}^\tau e^{i(f(\tau')-f(\tau))}\partial_y^{j+1}\bar z (Y(\tau',p)) \partial_y^{i-j}\chi(\tau',Y(\tau',p)) d\tau'. 
	\end{split}
    \end{equation}
We first observe that if $p\le y_0$, $\chi_0=1$. Thus $\chi (\tau, Y(\tau,p)) =1$ for each $\tau\ge \tau_0$ and $p\le y_0$, and hence for  $i\ge 1$, $\partial_y^i \chi(\tau,Y(\tau,p))=0$ for $\tau\ge \tau_0$ and $p<  y_0$. Therefore, to prove the claimed bound in (4), it is enough to consider $p \ge y_0$. 
	We will prove it by induction on $i$.  
    When $i=1$, by using \eqref{z = -2y U}, \eqref{B bc},  \eqref{eq Y(tau,p)} and \eqref{ineq pa_yi chi}, we have
	\begin{align*}
		|\partial_y \chi(\tau,Y(\tau,p))| &=| \partial_y\chi(\tau_0,Y(\tau_0,p)) |e^{-\int_{\tau_0}^\tau 1+\frac{\gamma+1}{4}\partial_y \bar z(Y(\tau,p))\ d\tau}\\
		&\leq A p^{-1} e^{-\int_{\tau_0}^\tau 1-\frac{\gamma+1}{2} \bar U(Y(\tau,p)) d\tau} e^{\int_{\tau_0}^\tau\frac{\gamma+1}{2} Y(\tau,p) \partial_y \bar U(Y(\tau,p))d\tau }\leq A (Y(\tau,p))^{-1}.
	\end{align*}
	We now assume that for any $i< m$, $|\partial_y^i \chi(\tau,y)|\leq A y^{-i}$. Then from \eqref{eq payi chi}, we obtain 
	 \begin{align*}
	 	|\partial_y^m \chi(\tau,Y(\tau,p))| &\leq A e^{m(f(\tau_0)-f(\tau))} p^{-m} +A  \int_{\tau_0}^\tau e^{m( f(\tau')-f(\tau))} \big(Y(\tau',p)\big)^{-\mu-m} d\tau'.
	 \end{align*}
     where we have also used Proposition \ref{prop bar z} (4) and \eqref{ineq pa_yi chi}. 
	 By using \eqref{z = -2y U}, \eqref{eq Y(tau,p)} and \eqref{def IF}, we may rewrite the right-hand side as  
	 \begin{align*}
	 	 &e^{m(f(\tau_0)-f(\tau))} p^{-m} +\int_{\tau_0}^\tau e^{m( f(\tau')-f(\tau))} \big(Y(\tau',p)\big)^{-\mu-m} d\tau'\\
	 &=  \big(Y(\tau,p)\big)^{-m}\Bigg( e^{m\int_{\tau_0}^{\tau} \frac{\gamma+1}{2} Y(\tau',p) \partial_y \bar U(Y(\tau',p)) d\tau'} + \int_{\tau_0}^\tau e^{m\int_{\tau'}^{\tau} \frac{\gamma+1}{2} Y(\tau'',p) \partial_y \bar U(Y(\tau'',p)) d\tau''} \big(Y(\tau',p)\big)^{-\mu} d\tau' \Bigg).
	 \end{align*}
    The desired bound follows once the boundedness of the terms in the bracket is shown. 
 The first term in the bracket is bounded for any $\tau\geq \tau_0$, since $Y(\tau',p) \partial_y \bar U(Y(\tau',p))\leq0$ holds for any $\tau'\geq \tau_0$. Concerning the second term, by \eqref{range of flow map}, we first observe that $\big(Y(\tau',p)\big)^{-\mu}\leq p^{-\mu} e^{-\mu(1-\mu) (\tau'-\tau_0)} \le y_0^{-\mu} e^{-\mu(1-\mu) (\tau'-\tau_0)}  $. 
	 We then further bound
	 \[
	 	 \int_{\tau_0}^\tau e^{m\int_{\tau'}^{\tau} \frac{\gamma+1}{2} Y(\tau'',p) \partial_y \bar U(Y(\tau'',p)) d\tau''} \big(Y(\tau',p)\big)^{-\mu} d\tau' \leq y_0^{-\mu} \int_{\tau_0}^\tau e^{-\mu(1-\mu) (\tau'-\tau_0)} d\tau' \le (\mu(1-\mu))^{-1}
	 \]
where we have used $y_0>1$ in the last inequality. 
     This completes the proof of (4).

<!-- remark-3: body_md -->
The choice of ${\bf{n}_{\mu}}$ \eqref{def ndmu} in our theorem is related to the dissipative property of the linearized operator $\mathcal L_z$ in \eqref{4.5} in $L^2_{(y^{-2{\bf{n}_{\mu}}}dy)}$ and $\dot H^{\bf{n}_{\mu}}$. The modulation component in the decomposition \eqref{sol decompose} is truncated at order ${\bf{n}_{\mu}}-1$. We distinguish three cases:
\begin{itemize}
    \item[(i)] if $\mu\in\mathfrak S$, then ${\bf{n}_{\mu}}-1= \beta$;
    \item[(ii)] if $\mu\notin\mathfrak S$ with $ \lfloor \beta+\frac12\rfloor=\lfloor \beta\rfloor $, then ${\bf{n}_{\mu}}-1 = \lfloor \beta\rfloor$;
    \item[(iii)] if $\mu\notin\mathfrak S$ with $ \lfloor \beta+\frac12\rfloor = \lfloor \beta+1\rfloor $, then ${\bf{n}_{\mu}}-1 =\lfloor \beta\rfloor+1$. Consequently, for such values of $\mu$, the modulation requires one additional degree of control compared to cases (i) and (ii). 
\end{itemize}
By Remark~\ref{remark 3.7}, for $\mu\notin\mathfrak S$, we have the control of modulation up to the order $\lceil \beta\rceil$, which is equal to $\lfloor \beta\rfloor+1$ for $\mu\notin \mathfrak S$. Hence, the maximal available order of modulation control coincides with the truncation order in case (iii). Hence, in all cases, the chosen truncation order ${\bf{n}_{\mu}}-1$ is compatible with the available modulation control.

    By Remark~\ref{remark 3.7}, we observe that for $\mu\in[\frac12,\frac{3}{5})$, there exist three unstable directions; see also Proposition~\ref{trivial mode}. These directions are associated with the constraints $z_1=w_1=0$ and $z_2=q_2(w_2)$.

		&\leq  \sum_{j=0}^\infty  \lambda_j^{1-2n}C_1 \Big( \left\lVert\partial_y^n f_j\right\rVert_{L^2([1,2])}^{\frac{2i}{n}}\left\lVertf_j\right\rVert_{L^2([1,2])}^{2-\frac{2i}{n}}+\left\lVertf_j\right\rVert_{L^2([1,2])} \Big)\\

	 \end{align*}
     where we have used the Gagliardo–Nirenberg inequality in the second inequality.
     We then use the H\"{o}lder inequality on the first summation and together with Lemma \ref{hardy 1}, we deduce the result.

<!-- lemma-13: body_md -->
\label{Linfty decay}
	Let $f \in\dot H^1([0,\infty)) \cap \dot H^n([0,\infty))$. Suppose $\partial_y ^j f(0) =0$ for some $j\in\{1,\dots, n-1\}$. Then, 
	\begin{equation}
		\left\lVert\partial_y^j f\right\rVert_{L^\infty} \lesssim \left\lVert\partial_y f\right\rVert_{L^2}^{1-\frac{2j-1}{2(n-1)}} \left\lVert\partial_y^n f\right\rVert_{L^2}^{\frac{2j-1}{2(n-1)}}. 
	\end{equation}
<!-- end lemma-13 -->
<!-- proof-19: body_md -->
Direct computation shows
    \begin{align*}
        (\partial_y^j f)^2 = \int_0^y 2\partial_y^j f \partial_y^{j+1} f d\tilde y.
    \end{align*}
    Hence, by using Holder inequality, we obtain
    \[
        \left\lVert\partial_y^j f\right\rVert_{L^\infty} \lesssim \left\lVert\partial_y^j f\right\rVert_{L^2}^{\frac{1}{2}}\left\lVert\partial_y^{j+1}f\right\rVert_{L^2}^{\frac{1}{2}}.
    \]
    By the Gagliardo–Nirenberg interpolation inequality
    \[
        \left\lVert\partial_y^jf\right\rVert_{L^2} \lesssim \left\lVert\partial_y f\right\rVert_{L^2}^{1-\frac{j-1}{n-1}} \left\lVert\partial_y^n f\right\rVert_{L^2}^{\frac{j-1}{n-1}}, \ \ \ 
        \left\lVert\partial_y^{j+1}f\right\rVert_{L^2} \lesssim \left\lVert\partial_y f\right\rVert_{L^2}^{1-\frac{j}{n-1}} \left\lVert\partial_y^n f\right\rVert_{L^2}^{\frac{j}{n-1}}, 
    \]
    we obtain the desired inequality.

<!-- lemma-14: body_md -->
\label{lemma w_1}
	Let $\gamma\in(1,3)$ and $\mu\in[\frac12,1)$. Suppose that the initial data satisfy $w_1(\tau_0) = \epsilon_0$ with $\epsilon_0>0$ sufficiently small. Then the solution to \eqref{ode w_1} exists for all $\tau>\tau_0$, is strictly decreasing to zero, and satisfies the exponential decay estimate
	\begin{equation}\label{estimate of w_1}
	0<w_1(\tau)< e^{\alpha^2 \tau_0}w_1(\tau_0)e^{-\alpha^2 \tau},\qquad \tau \geq \tau_0,
	\end{equation}
for some constant $\alpha \neq 0$. Furthermore, if $w_1(\tau_0) = 0$, then $w_1 \equiv 0$ for all $\tau > \tau_0$.
<!-- end lemma-14 -->
<!-- proof-21: body_md -->
We will prove it by a continuity argument. Since the $w_1(0)=\epsilon_0$ is sufficiently small, $\mu-\frac{3-\gamma}{4}w_1(0)>0$ initially. Consequently, $\partial_\tau w_1(\tau_0)<0$. By the Picard–Lindelöf theorem, there exists $\nu_0 > 0$ such that the solution to the ODE \eqref{ode w_1} with initial condition $w_1(\tau_0) = \epsilon_0$ exists on the interval $\tau \in [\tau_0,\tau_0+\nu_0)$. Moreover, $\nu_0$ can be chosen sufficiently small so that $w_1(\tau_0)-w_1(\tau)\ll 1$. It then follows that 
	\[
		1-\frac{3-\gamma}{\gamma+1}\frac{\mu}{\mu-\frac{3-\gamma}{4}w_1}>0, \ w_1\in(0,\epsilon) \text{ for }\tau\in[\tau_0,\tau_0+\nu_0).
	\]
	In particular 
	\[
		\partial_\tau w_1<0\text{ for }\tau\in[\tau_0,\tau_0+\nu_0).
	\]
	By a standard continuity argument, the solution can be extended to all $\tau \in [\tau_0,\infty)$, and $w_1$ is strictly decreasing. Furthermore, note that $w_1 \equiv 0$ is itself a solution to the ODE \eqref{ode w_1}. By uniqueness of solutions guaranteed by the Picard–Lindelöf theorem, this establishes the second part of the statement. Now, we remain to show \eqref{estimate of w_1}. Since $w_1\in[0,\epsilon_0)$. Then, there exists an nonzero real number $\alpha$ such that
	\[
		\partial_\tau w_1 < -\alpha^2 w_1.
	\]
	It then follows                                    
    
                                                                                       
    
                                                                                                         
	\[
		w_1(\tau)< w_1(\tau_0)e^{-\alpha^2 (\tau-\tau_0)}.
	\]
	This completes the proof.
