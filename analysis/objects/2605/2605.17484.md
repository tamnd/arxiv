\label{assumption:geometric}
    Throughout this paper, we impose the following geometric conditions, and the corresponding parameters are regarded as {\it a priori} data.
 \begin{itemize}

<!-- proposition-2: body_md -->
\label{prop:aa2}
        Let $U$ be a Lipschitz connected domain. Let $\Sigma\subset \partial U$ be an open portion of the boundary. Consider a harmonic function $w$ in $U$. We make the following assumptions 
        \begin{itemize}
            \item for any $y\in U$, there exists a rectifiable curve $\gamma$ in $U$ connecting $y$ and a point $x\in \Sigma$,
            \item the length of $\gamma$ is bounded by a uniform constant $L>0$,
            \item $\gamma$ does not touch the boundary of $U$ outside $\Sigma$, i.e., ${\rm dist}(\gamma,\partial U\setminus \Sigma)>4r$ for some $r>0$,
            \item $w$ is bounded with $\|w\|_{L^\infty(U)}=T>0$,
            \item $\|w\|_{H^{1/2}(\Sigma)}+\|\partial_{\nu} w\|_{H^{-1/2}(\Sigma)}\leq \varepsilon$ with $0< \varepsilon<1$.
        \end{itemize}
        Then, for all $y\in U$ and all sufficiently small $r>0$, one has
        \begin{align}\label{eq:propa-small-holder}
            |w(y)|\leq C T \varepsilon^{\widetilde{\alpha}^{L/r+1}},
        \end{align}
        with a constant $C$ depending only on {\it a priori} data.

We apply the integral identity \eqref{eq:int-id} with the CGO solution $u_0$ constructed in (\ref {eq:CGO}). The left-hand side of \eqref{eq:int-id} is estimated using \zcref{prop:lowerbound}, and the right-hand side is estimated using \zcref{prop:upperbound}. Collecting these estimates, we obtain

                \rho^{\lambda_1+\varepsilon_*-\mu_1} h_z \tau^{-\mu_1} &\lesssim \rho^{\lambda_1-1}h_z\tau^{-1} + (\rho+h_z)^{1-\mu_1}\tau^{-\mu_1} e^{-\frac{\tau \widetilde{a} h}{2}}\\
                & \quad + \rho^{-(s-\mu_1-1/2)}h_z \tau^{-(s-1/2)} + \rho^{-\widetilde{\mu_2}}h_z\tau^{-\widetilde{\mu_2}} + h_z\tau^{-1}\\
                & \quad + h^{-1}\tau^{-2}+\rho^{-\mu_1}\tau^{-(\mu_1+1)}+h_z e^{-\tau \widetilde{b} h} (h^{\mu_1}\rho^{-\mu_1}+1+h\tau)\\
                &\quad +h h_z \left(\|\nabla(u-u')\|_{L^\infty(S^e)}+\tau\|u-u'\|_{L^\infty(S^e)}\right).
            \end{align*}
            To estimate $\|u-u'\|_{L^\infty(S^e)}$ and $\|\nabla(u-u')\|_{L^\infty(S^e)}$, we apply propagation of smallness from \zcref{prop:propa-small}. Introduce the notation
            \begin{align*}
                \zeta(\varepsilon):=(\ln |\ln \varepsilon|)^{-\alpha}.
            \end{align*}
            Noting that the common H\"older norm $T$ depends only on {\it a priori} data and that the surface $S^e$ has a distance $1/\tau$ to $D$, we have,
            \begin{align*}
                \|u-u'\|_{L^\infty(S^e)} &\lesssim \zeta(\varepsilon),\\
                \|\nabla(u-u')\|_{L^\infty(S^e)} &\lesssim \tau \zeta(\varepsilon).
            \end{align*}
            Recall that $\rho,h$, and $h_z$ must satisfy the constraints \eqref{eq:constraint1},\eqref{eq:constraint2} and \eqref{eq:constraint3}, also that $D'$ is at distance at least $h$ from $\widetilde{\Omega}$. Thus, we choose
            \begin{align*}
                h_z=h,\quad \rho=\frac{h}{\ell},\quad h=\min\left\lbrace C_{\varepsilon_*}\left(1+\frac{1}{\ell}\right)^{-1} ,\frac{\ell^2}{2\ell+1},\;\frac{1}{2}d_H(D,D')\right\rbrace.
            \end{align*}
            Then the estimate becomes
            \begin{align*}
                h^{\lambda_1+\varepsilon_*-\mu_1+1} \tau^{-\mu_1} &\lesssim h^{\lambda_1}\tau^{-1} + h^{1-\mu_1}\tau^{-\mu_1} e^{-\frac{\tau \widetilde{a} h}{2}} + h^{3/2-s+\mu_1} \tau^{-(s-1/2)} + h^{1-\widetilde{\mu_2}}\tau^{-\widetilde{\mu_2}} + h\tau^{-1}\\
                & \quad + h^{-1}\tau^{-2}+h^{-\mu_1}\tau^{-\mu_1-1}+h e^{-\tau \widetilde{b} h} (1+h\tau) +h^2 \tau \zeta(\varepsilon).
            \end{align*}
            Next, we use the usual inequalities $e^{-x}\leq x^{-1}$ and $e^{-x}\leq x^{-2}$ for all $x>0$ to deduce that
            \begin{align*}
                h^{\lambda_1+\varepsilon_*-\mu_1+1} \tau^{-\mu_1} &\lesssim h^{\lambda_1}\tau^{-1} + h^{-\mu_1}\tau^{-\mu_1-1} + h^{3/2-s+\mu_1} \tau^{-(s-1/2)} + h^{1-\widetilde{\mu_2}}\tau^{-\widetilde{\mu_2}} + h\tau^{-1}\\
                & \quad + h^{-1}\tau^{-2}+h^{-\mu_1}\tau^{-\mu_1-1}+\tau^{-1} +h^2 \tau \zeta(\varepsilon).
            \end{align*}
            Recall the estimates on singular exponents: $0<\lambda_1<s-1/2$, $0<\mu_1<1$, and $3/2\leq s <\mu_1+1$. Combining these with $\tau\geq 1$ and $h\leq 1$, we deduce that
            \begin{align*}
                h^{\lambda_1+\varepsilon_*-\mu_1+1} \tau^{-\mu_1} &\lesssim h^{-1}\tau^{-1}+  h^{1-\widetilde{\mu_2}}\tau^{-\widetilde{\mu_2}}+h^2 \tau \zeta(\varepsilon).
            \end{align*}
            Recalling that $s=3/2+\epsilon$ with $0<\epsilon<\mu_1-1/2$ and using the definition of $\widetilde{\mu_2}$, we distinguish the following two cases.
            \begin{itemize}
                \item If $s-\mu_2\geq 1/2$, then $-\widetilde{\mu_2}=-\mu_2$ and $1-\widetilde{\mu_2}=1-\mu_2\geq 3/2-s=\epsilon>-1/2$.
                \item If $s-\mu_2< 1/2$, then $-\widetilde{\mu_2}=1/2-s=-1-\epsilon <-1$ and $1-\widetilde{\mu_2}=-\epsilon >-1/2$.
            \end{itemize}
            Taking into account that $\mu_2>1$, the estimate becomes
            \begin{align*}
                h^{\lambda_1+\varepsilon_*-\mu_1+1} \tau^{-\mu_1} &\lesssim h^{-1}\tau^{-1}+h^2 \tau \zeta(\varepsilon).
            \end{align*}
            By minimizing the right-hand side with respect to $\tau$, we choose
            \begin{align}
                \tau=\tau_e:=\left(h^3\zeta(\varepsilon)\right)^{-\frac{1}{2}}.
            \end{align}
            For $\varepsilon$ sufficiently small, one can verify that $\tau_e\geq \tau_0$, where $\tau_0$ is given in \eqref{eq:tau0}.

            Substituting this choice into the estimate and solving for $h$, we obtain
            \begin{align*}
                h \lesssim \zeta(\varepsilon)^{\frac{1}{2(\lambda_1+\varepsilon_*)+\mu_1+1}}\leq \zeta(\varepsilon)^{\frac{1}{3\lambda_1+\mu_1+1}},                                                   
            \end{align*}
            where the last inequality holds by choosing $\varepsilon_*<\lambda_1/2$.

            Finally, as $\displaystyle \zeta(\varepsilon)\underset{{\varepsilon}\rightarrow {0}}{\longrightarrow} 0$, we can deduce that for $\varepsilon$ small enough, the right-hand side above is less than both $C_{\varepsilon_*}\left(1+\frac{1}{\ell}\right)^{-1}$ and $\frac{\ell^2}{2\ell+1}$. Hence, the stability estimate \eqref{eq:stable} follows with the exponent $\kappa$ given by
            \begin{align*}
               \kappa:= \frac{-\alpha }{3\lambda_1+\mu_1+1}.
            \end{align*}
            The exponent $\kappa$ is bounded below by a constant depending only on {\it a priori} data, using $\alpha=\min\{\alpha_0,\epsilon,\lambda_1,\mu_1\}$ from \zcref{prop:propa-small} and the bounds $\mu_1\in (\frac{1}{2},1)$, $\lambda_1\in (1,\sqrt{2k})$.

            The proof is complete.

By a change of variables stretching $E$ into $2.4$ and the embedding \eqref{eq:embedding-weighted}, we have $\widetilde{\gamma}\in H^{s-\mu}(2.4)$. We compute $\partial_z u_{E,\mu}$ via the chain rule:
            \begin{align*}
                \frac{\partial}{\partial z}u_{E,\mu}(r,\theta,z)&= \left(\frac{\partial \widetilde{z}}{\partial z}\frac{\partial}{\partial \widetilde{z}}+\frac{\partial \widetilde{r}}{\partial z}\frac{\partial}{\partial \widetilde{r}}\right) u_{E,\mu}(r,\theta,z)\\
                &= \frac{1}{\delta(z)}\left(\frac{\partial}{\partial \widetilde{z}}\mathcal{K}[\gamma]\right)\widetilde{r}^{\mu}\varphi(\theta)-\frac{\delta'(z)}{\delta(z)} \left(\widetilde{r}^{\mu+1} \frac{\partial}{\partial \widetilde{r}}\mathcal{K}[\gamma]+\mu \widetilde{r}^{\mu}\mathcal{K}[\gamma]\right) \varphi(\theta)\\
                \frac{\partial}{\partial r} u_{E,\mu}(r,\theta,z) &= \frac{\partial \widetilde{r}}{\partial r}\frac{\partial}{\partial \widetilde{r}} u_{E,\mu}(r,\theta,z)=\frac{1}{\delta(z)}\left(\frac{\partial}{\partial \widetilde{r}}\mathcal{K}[\gamma]\right)\widetilde{r}^{\mu}\varphi(\theta)+\frac{\mu}{\delta(z)} \mathcal{K}[\gamma]\widetilde{r}^{\mu-1}\varphi(\theta),\\
                \frac{1}{r} \frac{\partial}{\partial \theta} u_{E,\mu}(r,\theta,z) &= \frac{1}{\delta(z)}\mathcal{K}[\gamma]\widetilde{r}^{\mu-1}\varphi'(\theta).
            \end{align*}
            By the convolution structure of $\mathcal{K}$, we have:
            \begin{align*}
                \frac{\partial}{\partial \widetilde{z}}\mathcal{K}[\gamma](\widetilde{r},z)& = \frac{\partial}{\partial \widetilde{z}}\int_2.4 \frac{1}{\widetilde{r}}\phi(\frac{\widetilde{z}-t}{\widetilde{r}})\widetilde{\gamma}(t)dt=\frac{1}{\widetilde{r}}\int_2.4 \frac{1}{\widetilde{r}}\phi'(\frac{t}{\widetilde{r}})\widetilde{\gamma}(\widetilde{z}-t) dt,\\
                \frac{\partial}{\partial \widetilde{r}}\mathcal{K}[\gamma](\widetilde{r},z)&= -\int_2.4 \left(\frac{1}{\widetilde{r}^2}\phi(\frac{t}{\widetilde{r}})+\frac{t}{\widetilde{r}^3}\phi'(\frac{t}{\widetilde{r}})\right)\widetilde{\gamma}(\widetilde{z}-t)dt=-\frac{1}{\widetilde{r}}\int_2.4 \frac{1}{\widetilde{r}}(t\phi)'(\frac{t}{\widetilde{r}})\gamma(\widetilde{z}-t)dt.
            \end{align*}
            Note that $\phi',\;(t\phi)'\in \mathcal{S}(2.4)$. We apply the Sobolev embedding $H^{s-\mu}(2.4)\hookrightarrow L^q(2.4)$ with $\frac{1}{q}=\frac{1}{2}-(s-\mu)$ if $s-\mu<1/2$ and $q=\infty$ if $s-\mu\geq 1/2$.
            \begin{itemize}
                \item Case $s-\mu <1/2$. Denoting by $q'$ the conjugate exponent of $q$, it follows from H\"older's inequality that
                \begin{align*}
                    \left|\mathcal{K}[\gamma](\widetilde{r},z)\right| &\leq \widetilde{r}^{-1/q} \|\phi\|_{L^{q'}(2.4)} \|\widetilde{\gamma}\|_{L^q(2.4)}=\widetilde{r}^{(s-\mu)-1/2} \|\phi\|_{L^{q'}(2.4)} \|\widetilde{\gamma}\|_{L^q(2.4)},\\
                    \left|\frac{\partial}{\partial \widetilde{z}}\mathcal{K}[\gamma](\widetilde{r},z)\right| &\leq \widetilde{r}^{(s-\mu)-3/2} \|\phi'\|_{L^{q'}(2.4)} \|\widetilde{\gamma}\|_{L^q(2.4)},\\
                    \left|\frac{\partial}{\partial \widetilde{r}}\mathcal{K}[\gamma](\widetilde{r},z)\right| &\leq \widetilde{r}^{(s-\mu)-3/2} \|(t\phi)'\|_{L^{q'}(2.4)} \|\widetilde{\gamma}\|_{L^q(2.4)}.
                \end{align*}
                \item Case $s-\mu\geq 1/2$. Using Young's inequality, we have
                \begin{align*}
                    |\mathcal{K}[\gamma](\widetilde{r},z)| &\leq \|\phi\|_{L^1(2.4)} \|\widetilde{\gamma}\|_{L^\infty(2.4)},\\
                    \left|\frac{\partial}{\partial \widetilde{z}}\mathcal{K}[\gamma](\widetilde{r},z)\right| &\leq \widetilde{r}^{-1} \|\phi'\|_{L^1(2.4)} \|\widetilde{\gamma}\|_{L^\infty(2.4)},\\
                    \left|\frac{\partial}{\partial \widetilde{r}}\mathcal{K}[\gamma](\widetilde{r},z)\right| &\leq \widetilde{r}^{-1} \|(t\phi)'\|_{L^1(2.4)} \|\widetilde{\gamma}\|_{L^\infty(2.4)}.
                \end{align*}
            \end{itemize}
            Define $\widetilde{\mu}:=\mu$ if $s-\mu\geq 1/2$ and $\widetilde{\mu}=s-1/2$ if $s-\mu<1/2$. Substituting these estimates into the expressions for derivatives of $u_{E,\mu}$, we obtain
            \begin{align*}
                \left|\frac{\partial}{\partial z} u_{E,\mu}(r,\theta,z)\right| &\leq \|\widetilde{\gamma}\|_{L^q(2.4)} \left(\frac{\widetilde{r}^{\widetilde{\mu}-1}}{\delta(z)} \|\phi'\|_{L^{q'}(2.4)} + \frac{|\delta'(z)| \widetilde{r}^{\widetilde{\mu}}}{\delta(z)}(\|(t\phi)'\|_{L^{q'}(2.4)}+\mu \|\phi\|_{L^{q'}(2.4)})\right).
            \end{align*}
            The same estimate holds for $\frac{\partial}{\partial r} u_{E,\mu}$.

            Note that the cut-off function $\eta$ and its support $\ell$ satisfy $\ell \leq 1/2$ and $\eta(\widetilde{r})=0$ if $\widetilde{r}\geq 2\ell$. Hence we may assume $\widetilde{r}\leq 1$. Also, $\delta$ is equivalent to the distance to the endpoints of $E$, which implies $\delta(z)\geq C_{\delta}\rho$ for some constant $C_\delta>0$. Then the estimates become
            \begin{align*}
                &\left|\frac{\partial}{\partial z} u_{E,\mu}(r,\theta,z)\right|,\; \left|\frac{\partial}{\partial r} u_{E,\mu}(r,\theta,z)\right|\\
                 &\leq \|\widetilde{\gamma}\|_{L^q(2.4)} C_\delta^{-\widetilde{\mu}}\left( \|\phi'\|_{L^{q'}(2.4)} + |\delta'(z)|(\|(t\phi)'\|_{L^{q'}(2.4)}+\mu \|\phi\|_{L^{q'}(2.4)})\right)\frac{r^{\widetilde{\mu}-1}}{\rho^{\widetilde{\mu}}},\\

            \end{align*}            
            
            Thus, the claim follows by combining these estimates with \eqref{eq:embedding-weighted}.
