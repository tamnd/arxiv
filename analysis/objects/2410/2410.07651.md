<!-- theorem-1: body_md -->
\cite{Stojnicflrdt23}
\label{thm:thmsflrdt1}  Consider large $n$ linear regime with  $\alpha\triangleq \lim_{n\rightarrow\infty} \frac{m}{n}$, remaining constant as  $n$ grows. Let ${\mathcal X}\subseteq {\mathbb R}^{n+1}$ and ${\mathcal Y}\subseteq {\mathbb R}^m$ be two given sets and let the elements of  $G\in{\mathbb R}^{m\times n}$
 be i.i.d. standard normals. Assume the complete sfl RDT frame from \cite{Stojnicsflgscompyx23} and consider a given function $f({\bf y}):R^m\rightarrow R$. Set
\begin{align}\label{eq:thmsflrdt2eq1}
   \psi_{rp} & \triangleq - \max_{{\bf x}\in{\mathcal X}} s \max_{{\bf y}\in{\mathcal Y}} \left ( f({\bf y})+{\bf y}^TG{\bf x} \right )
   \qquad  \mbox{(\textcolor{blue}{\textbf{random primal}})} \nonumber \\
   \psi_{rd}({\bf p},{\bf q},{\bf c},x,y,s) & \triangleq    \frac{x^2y^2}{2}    \sum_{k=2}^{r+1}\Bigg(\Bigg.
   {\bf p}_{k-1}{\bf q}_{k-1}
   -{\bf p}_{k}{\bf q}_{k}
  \Bigg.\Bigg)
{\bf c}_k
  - \psi_{S,\infty}(f({\bf y}),{\cal X},{\cal Y},{\bf p},{\bf q},{\bf c},x,y,s) \hspace{.03in} \mbox{(\textcolor{blue}{\textbf{fl random dual}})}. \nonumber \\
 \end{align}
Let $\hat{{\bf p}_0}\rightarrow 1$, $\hat{{\bf q}_0}\rightarrow 1$, and $\hat{{\bf c}_0}\rightarrow 1$, $\hat{{\bf p}}_{r+1}=\hat{{\bf q}}_{r+1}=\hat{{\bf c}}_{r+1}=0$, and let the non-fixed parts of $\hat{{\bf p}}\triangleq \hat{{\bf p}}(x,y)$, $\hat{{\bf q}}\triangleq \hat{{\bf q}}(x,y)$, and  $\hat{{\bf c}}\triangleq \hat{{\bf c}}(x,y)$ be the solutions of the following system
\begin{eqnarray}\label{eq:thmsflrdt2eq2}
   \frac{d \psi_{rd}({\bf p},{\bf q},{\bf c},x,y,s)}{d{\bf p}} =  0,\quad
   \frac{d \psi_{rd}({\bf p},{\bf q},{\bf c},x,y,s)}{d{\bf q}} =  0,\quad
   \frac{d \psi_{rd}({\bf p},{\bf q},{\bf c},x,y,s)}{d{\bf c}} =  0.
 \end{eqnarray}
 Then,
\begin{eqnarray}\label{eq:thmsflrdt2eq3}
    \lim_{n\rightarrow\infty} \frac{{\mathbb E}_G  \psi_{rp}}{\sqrt{n}}
  & = &
\min_{x>0} \max_{y>0} \lim_{n\rightarrow\infty} \psi_{rd}(\hat{{\bf p}}(x,y),\hat{{\bf q}}(x,y),\hat{{\bf c}}(x,y),x,y,s) \qquad \mbox{(\textcolor{blue}{\textbf{strong sfl random duality}})},\nonumber \\
 \end{eqnarray}
where $\psi_{S,\infty}(\cdot)$ is as in (\ref{eq:fl4})-(\ref{eq:fl5}).

\label{cor:cor1}  Assume the setup of Theorem \ref{thm:thmsflrdt1} with $r>0$, $\sigma >0$, $c_1\in [0,\sqrt{r_1}]$, and ${\mathcal X}={\mathcal X}(\sigma,c_1,r_1)$ and ${\mathcal Y}={\mathbb S}^m$, where ${\mathcal X}(\sigma,c_1,r_1)$ is as in (\ref{eq:ex13}) and ${\mathbb S}^m$ is the unit $m$-dimensional Euclidean sphere. Set $r_2=\sqrt{\sigma^2+(1-2c_1+r_1^2)}$ and
\begin{align}\label{eq:thmsflrdt2eq1a0}
   \psi_{rp} & \triangleq - \max_{{\bf x}\in{\mathcal X}} s \max_{{\bf y}\in{\mathcal Y}} \left ( {\bf y}^TG{\bf x}  \right )
   \qquad  \mbox{(\textcolor{blue}{\textbf{random primal}})} \nonumber \\
   \psi_{rd}({\bf p},{\bf q},{\bf c},r_2,1,s) & \triangleq    \frac{r_2^2}{2}    \sum_{k=2}^{r+1}\Bigg(\Bigg.
   {\bf p}_{k-1}{\bf q}_{k-1}
   -{\bf p}_{k}{\bf q}_{k}
  \Bigg.\Bigg)
{\bf c}_k
  - \psi_{S,\infty}(0,{\cal X},{\cal Y},{\bf p},{\bf q},{\bf c},r_2,1,s) \quad \mbox{(\textcolor{blue}{\textbf{fl random dual}})}. \nonumber \\
 \end{align}
Let the non-fixed parts of $\hat{{\bf p}}$, $\hat{{\bf q}}$, and  $\hat{{\bf c}}$ be the solutions of the following system
\begin{eqnarray}\label{eq:thmsflrdt2eq2a0}
   \frac{d \psi_{rd}({\bf p},{\bf q},{\bf c},r_2,1,s)}{d{\bf p}} =  0,\quad
   \frac{d \psi_{rd}({\bf p},{\bf q},{\bf c},r_2,1,s)}{d{\bf q}} =  0,\quad
   \frac{d \psi_{rd}({\bf p},{\bf q},{\bf c},r_2,1,s)}{d{\bf c}} =  0.
 \end{eqnarray}
 Then,
\begin{eqnarray}\label{eq:thmsflrdt2eq3a0}
    \lim_{n\rightarrow\infty} \frac{{\mathbb E}_G  \psi_{rp}}{\sqrt{n}}
  & = &
 \lim_{n\rightarrow\infty} \psi_{rd}(\hat{{\bf p}},\hat{{\bf q}},\hat{{\bf c}},r_2,1,s) \qquad \mbox{(\textcolor{blue}{\textbf{strong sfl random duality}})},\nonumber \\
 \end{eqnarray}
where $\psi_{S,\infty}(\cdot)$ is as in (\ref{eq:fl4})-(\ref{eq:fl5}).

  \end{eqnarray}

   -{\bf p}_{k}{\bf q}_{k}
  \Bigg.\Bigg)
{\bf c}_k

   {\bf q}_{k-1}
   -{\bf q}_{k}
  \Bigg.\Bigg)
{\bf c}_k \nonumber \\
& &
 - \nu_1 c_1
  -\gamma  r_1^2
     -\nu \beta
   - (1-\beta)\varphi(D_{10}^{(per)}(-1),{\bf c}) - \frac{1}{n}\sum_{i=1}^{k}\varphi(D_{i1}^{(per)}(-1),{\bf c}) \nonumber \\

  \end{eqnarray}
