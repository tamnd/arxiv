<!-- remark-1: body_md -->
As multiplicative-discrete type difference equations are also called $q$-difference equations, it is common to use the parameter $q$ for their shift parameters. 
However, in this paper, the parameter $q$ is used to denote the shift parameters of $q$-Painlev\'e equations, and we also consider the correspondence between the shift parameters of the $q$-Painlev\'e equations and those of the multiplicative dKdV equations.
Therefore, we use the parameter $\epsilon$ instead of the parameter $q$ for the multiplicative dKdV equations \eqref{eqn:mul_dKdV_1} and \eqref{eqn:mul_dKdV_2} to avoid confusion.

<!-- theorem-1: body_md -->
\label{theorem:A6}
The multiplicative dKdV equation \eqref{eqn:mul_dKdV_1} has the special solution
\begin{equation}
 u_{l,m}=\dfrac{\epsilon^{3/4}{\beta_{m}}^{1/2}(\alpha_{l}-\beta_{m})}{{\alpha_{l}}^{1/2}(h_{l,m}+\beta_{m+2})g_{l,m}}.
\end{equation}
Here, the functions $g_{l,m}$ and $h_{l,m}$ satisfy the system of ordinary difference equations for the $l$-direction:
\begin{equation}\label{eqn:theo_A6_l-direction}
 h_{l+1,m}h_{l,m}=\dfrac{\beta_{m+2}(1+g_{l,m})}{{g_{l,m}}^2},\quad
 g_{l+1,m}g_{l,m}=\dfrac{\beta_{m}(h_{l+1,m}+\alpha_{l+3})}{\alpha_{l+1}h_{l+1,m}(h_{l+1,m}+\beta_{m+2})},
\end{equation}
and that for the $m$-direction:
\begin{equation}\label{eqn:theo_A6_m-direction}
 f_{l,m+1}f_{l,m}=\dfrac{\alpha_l (h_{l,m}+\beta_{m+2})}{\beta_mh_{l,m}(h_{l,m}+\alpha_{l+2})},\quad
 h_{l,m+1}h_{l,m}=\dfrac{\alpha_{l+2}(1+f_{l,m+1})}{{f_{l,m+1}}^2},
\end{equation}
where 
\begin{equation}
 f_{l,m}=\dfrac{1}{g_{l,m}h_{l,m}}.
\end{equation}
Note that each of equations \eqref{eqn:theo_A6_l-direction} and \eqref{eqn:theo_A6_m-direction} is equivalent to the \text{\rm$q$-P$_{\rm III}^{D_7^{(1)}}$} \eqref{eqn:qp_appendix_A6_2}, which is a $q$-Painlev\'e equation of $A_6^{(1)}$-type. 
(See Appendix \ref{section:typical_examples_dPs} for details.)

<!-- theorem-2: body_md -->
\label{theorem:A5}
The multiplicative dKdV equation \eqref{eqn:mul_dKdV_1} has the special solution
\begin{equation}
 u_{l,m}=\dfrac{d(\alpha_{l}-\beta_{m})}{({\alpha_{l}}^{1/2}+{\beta_{m}}^{1/2}g_{l,m})\,h_{l,m}},
\end{equation}
where $d\in\mathbb{C}^\ast$ is an arbitrary parameter.
Here, the functions $g_{l,m}$ and $h_{l,m}$ satisfy 
the system of ordinary difference equations for the $l$-direction:
\begin{equation}\label{eqn:theo_A5_l-direction}
 g_{l+1,m}g_{l,m}
 =\dfrac{d^2({\alpha_{l}}^{1/2}+f_{l,m})}{f_{l,m}(1+{\alpha_{l}}^{1/2}f_{l,m})},\quad
 f_{l+1,m}f_{l,m}
 =\dfrac{d^2({\alpha_{l+1}}^{1/2}+{\beta_{m}}^{1/2}g_{l+1,m})}{g_{l+1,m}({\beta_{m}}^{1/2}+{\alpha_{l+1}}^{1/2}g_{l+1,m})},
\end{equation}
where 
\begin{equation}
 f_{l,m}=\dfrac{d^2}{g_{l,m}h_{l,m}},
\end{equation}
and that for the $m$-direction:
\begin{equation}\label{eqn:theo_A5_m-direction}
 h_{l,m+1}h_{l,m}
 =\dfrac{d^2({\beta_{m}}^{1/2}+{\alpha_{l}}^{1/2}g_{l,m})}{g_{l,m}({\alpha_{l}}^{1/2}+{\beta_{m}}^{1/2}g_{l,m})},\quad
 g_{l,m+1}g_{l,m}
 =\dfrac{d^2({\beta_{m}}^{1/2}+h_{l,m+1})}{h_{l,m+1}(1+{\beta_{m}}^{1/2}h_{l,m+1})}.
\end{equation}
Note that each of equations \eqref{eqn:theo_A5_l-direction} and \eqref{eqn:theo_A5_m-direction} is equivalent to the \text{\rm$q$-P$_{\rm III}$} \eqref{eqn:qp3_appendix_A5}, which is a $q$-Painlev\'e equation of $A_5^{(1)}$-type. 
(See Appendix \ref{section:typical_examples_dPs} for details.)

<!-- theorem-3: body_md -->
\label{theorem:A4}
The multiplicative dKdV equation \eqref{eqn:mul_dKdV_1} has the special solution
\begin{equation}
 u_{l,m}=\dfrac{{\beta_{m}}^{1/2}(\alpha_{l}-\beta_{m})h_{l,m}}{\epsilon^{1/4}{\alpha_{l}}^{3/2}d_1d_2(1+\beta_{m} g_{l,m})},
\end{equation}
where $d_1,d_2\in\mathbb{C}^\ast$ are arbitrary parameters.
Here, the functions $g_{l,m}$ and $h_{l,m}$ satisfy 
the system of ordinary difference equations for the $l$-direction:
\begin{equation}\label{eqn:theo_A4_l-direction}
\begin{cases}
 g_{l+1,m}g_{l,m}=\dfrac{(\beta_{m-1}{d_1}^2{d_2}^2+\alpha_lf_{l,m})(\beta_m+\alpha_lf_{l,m})}{{\alpha_l}^2{\beta_m}^2{d_2}^2({d_1}^2\beta_m+f_{l,m})},\\[1em]
 f_{l+1,m}f_{l,m}=\dfrac{{\beta_{m}}^2{d_1}^2{d_2}^2(1+\alpha_{l+1}g_{l+1,m})(1+\alpha_{l+1}\,\beta_m\, g_{l+1,m})}{{\alpha_{l+1}}^2(1+\beta_m g_{l+1,m})},
\end{cases}
\end{equation}
where
\begin{equation}
 f_{l,m}=\dfrac{{d_1}^2{d_2}^2(1+\alpha_l\,\beta_m\,g_{l,m})}{h_{l,m}},
\end{equation}
and that for the $m$-direction:
\begin{equation}\label{eqn:theo_A4_m-direction}
\begin{cases}
 h_{l,m+1}h_{l,m}=\dfrac{{\alpha_l}^2{d_1}^2{d_2}^2(1+\beta_m\,g_{l,m})(1+\alpha_l\,\beta_m\,g_{l,m})}{{\beta_m}^2(1+\alpha_l\,g_{l,m})},\\[1em]
 g_{l,m+1}g_{l,m}=\dfrac{(\alpha_{l+1}+\beta_{m+1}h_{l,m+1})(\alpha_l\,{d_1}^2{d_2}^2+\beta_{m+1}h_{l,m+1})}{{\alpha_l}^2\,{\beta_{m+1}}^2{d_2}^2(\alpha_l {d_1}^2+h_{l,m+1})}.
\end{cases}
\end{equation}

<!-- theorem-4: body_md -->
\label{theorem:A3}
The multiplicative dKdV equation \eqref{eqn:mul_dKdV_2} has the special solution
\begin{equation}
 u_{l,m}
 =\cfrac{\epsilon^{1/4}(\beta_m\gamma-\alpha_l)\left(f_{l,m}(d_1+x_{l,m})+{d_2}^{1/2}(1+d_1{d_2}^{-1}\,\beta_mx_{l,m})\right)}
 {{\alpha_l}^{1/2}{\beta_m}^{1/2}\gamma^{1/2}x_{l,m}\left(\beta_mf_{l,m}(d_1+x_{l,m})+{d_2}^{1/2}(1+d_1{d_2}^{-1}\,\beta_m x_{l,m})\right)},
\end{equation}
where $d_1,d_2\in\mathbb{C}^\ast$ are arbitrary parameters.
Here, the functions $f_{l,m}$ and $x_{l,m}$ satisfy 
the system of ordinary difference equations for the $l$-direction:
\begin{equation}\label{eqn:theo_A3_l-direction}
 \begin{cases}
 g_{l+1,m}g_{l,m}=\dfrac{(f_{l,m}+{d_1}^{-1}{d_2}^{1/2}{\alpha_l}^{-1}\gamma)(f_{l,m}+d_1{d_2}^{-1/2}\,{\alpha_l}^{-1})}{(f_{l,m}+{d_1}^{-1}{d_2}^{1/2})(f_{l,m}+d_1\,{d_2}^{-1/2})},\\[1em]
 
 f_{l+1,m}f_{l,m}=\dfrac{(g_{l+1,m}+{d_2}^{-1/2}{\alpha_l}^{-1}{\beta_{m-1}}^{1/2}\gamma^{1/2})(g_{l+1,m}+{d_2}^{1/2}{\alpha_l}^{-1}{\beta_{m+1}}^{-1/2}\gamma^{1/2})}{(g_{l+1,m}+{d_2}^{-1/2}{\beta_{m-1}}^{-1/2}\gamma^{-1/2})(g_{l+1,m}+{d_2}^{1/2}{\beta_{m-1}}^{1/2}\gamma^{1/2})},
 \end{cases}
\end{equation}
where
\begin{equation}
 g_{l,m}=\dfrac{{d_2}^{1/2}\gamma+d_1\,\alpha_l\,f_{l,m}}{\alpha_l\,{\beta_{m-1}}^{1/2}\gamma^{1/2}(d_1+{d_2}^{1/2}f_{l,m})x_{l,m}},
\end{equation}
and that for the $m$-direction:
\begin{equation}\label{eqn:theo_A3_m-direction}
 \begin{cases}
  y_{l,m+1}y_{l,m}=\dfrac{(x_{l,m}+{d_1}^{-1}{d_2}\,{\beta_m}^{-1})(x_{l,m}+d_1{d_2}^{-1}\, {\beta_{m-1}}^{-1}\gamma^{-1})}{(x_{l,m}+d_1)(x_{l,m}+{d_1}^{-1})},\\[1em]
  
 x_{l,m+1}x_{l,m}=\dfrac{(y_{l,m+1}+{\alpha_l}^{-1/2}{\beta_m}^{-1})(y_{l,m+1}+{\alpha_l}^{1/2}{\beta_m}^{-1}\gamma^{-1})}{(y_{l,m+1}+{\alpha_l}^{1/2})(y_{l,m+1}+{\alpha_l}^{-1/2})},
 \end{cases}
\end{equation}
where
\begin{equation}
 y_{l,m}=\dfrac{{\alpha_l}^{1/2}f_{l,m}(d_1+{d_2}\,\beta_{m-1}\,\gamma\, x_{l,m})}{{d_2}^{1/2}\beta_{m-1}\,\gamma(1+d_1\, x_{l,m})}.
\end{equation}
Note that each of equations \eqref{eqn:theo_A3_l-direction} and \eqref{eqn:theo_A3_m-direction} is equivalent to the \text{\rm$q$-P$_{\rm VI}$} \eqref{eqn:qP_appendix_A3}, which is a $q$-Painlev\'e equation of $A_3^{(1)}$-type. 
(See Appendix \ref{section:typical_examples_dPs} for details.)
