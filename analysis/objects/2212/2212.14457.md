\label{thm:Z-form}
Fix $P,L,N_0,\ldots, N_L\geq 1,\, \sigma^2>0$ as well as training data $X_{N_0},Y_{N_0}$. Fix ${\bf t}\in \mathbb R^{N_0}$ and write
\[
{\bf t} = {\bf t}_{||}+ {\bf t}_\perp,\qquad {\bf t}_{||}\in \mathrm{col}(X_{N_0}),\quad {\bf t}_{\perp}\in \mathrm{col}(X_{N_0})^\perp.
\]
Define $4M = \prod_{\ell=0}^L 2\sigma^2/N_\ell$
                              
                                                  
            
and introduce the following shorthand for the Meijer-G functions (see \S \ref{sec:back})                                                      
with parameters given by layer widths:
\begin{align*}
&\ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2},\frac{\mathbf{N}}{2}+k 
\end{array} \right)}:=\qquad\ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2},\frac{N_1}{2}+k,\ldots, \frac{N_L}{2}+k 
\end{array} \right)}.
\end{align*}
The partition function $Z_\infty({\bf t})$ of the predictive posterior defined in \eqref{eq:part-def} is
\begin{align}
&Z_\infty({\bf t}) =  \left(\frac{4\pi }{\left|\left|\theta_*\right|\right|^2}\right)^{\frac{P}{2}} \exp\left[-i\left \langle \theta_{*,N_0},{\bf t}\right\rangle \right]\prod_{\ell=1}^L \Gamma\left(\frac{N_\ell}{2}\right)^{-1} \nonumber\\
&\qquad \qquad \times \sum_{k=0}^\infty \frac{(-1)^k}{k!}\big|\big|{\bf t}_{\perp} \big|\big| ^{2k} M^{k} \ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2}, \frac{\bf{N}}{2}+k 
\end{array} \right)}.
\end{align}
In particular, the Bayesian model evidence $Z_\infty({\bf 0})$ equals
\begin{align}
\label{eq:evidence-form} \frac{(4\pi)^{P/2}}{\left|\left|\theta_*\right|\right|^{P}\prod_{\ell=1}^L \Gamma\left(\frac{N_\ell}{2}\right)} \ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2}, \frac{\bf{N}}{2} 
\end{array} \right)}.
                                                                                                                                                               
                                                                                                      
\end{align}
Further, given $x\in \mathbb R^{N_0}$, the mean of the predictive posterior is
\begin{equation}\label{eq:mean-G}
\mathbb E_{\text{post}}\left[f(x)\right]= (\theta_{*,N_0})^Tx,
\end{equation}
while the posterior variance $\mathrm{Var}_{\mathrm{post}}\left[f(x)\right]$ is 
\begin{equation}\label{eq:var-G}
2M\left|\left|x_{\perp}\right|\right|^2\frac{\ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2}, \frac{\bf{N}}{2}+1 
\end{array} \right)}}{\ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2}, \frac{\bf{N}}{2} 
\end{array} \right)}},    
\end{equation}
where $x_\perp$ is the projection of $x$ onto the orthogonal complement of the span $\mathrm{col}(X_{N_0})$ of the training data.

\label{thm:logG}
Set $N_1,\ldots,N_L = N$ and define $\mathbf{N} = (N_1, \dots, N_L)$. Suppose that the training data satisfies \eqref{eq:Vstar-def}. In different limiting cases such that $\{P, N\} \to \infty$ with fixed $P/N_0 = \alpha_0$, we evaluate the quantities
\begin{align*}
    \log G &:= \log \ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2},\frac{\mathbf{N}}{2}+k 
\end{array} \right)}\\
    \Delta(\log G)[k] &:= \log \ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2},\frac{\mathbf{N}}{2}+k 
\end{array} \right)} - \log \ensuremath{G_{0,L+1}^{L+1,0}\left( \frac{\left|\left|\theta_{*,N_0}\right|\right|^2}{4M} ~\bigg| \begin{array}{c}
     -  \\
     \frac{P}{2},\frac{\mathbf{N}}{2} 
\end{array} \right)}.
\end{align*}
We will see in each case that to leading order $\log G$ does not depend on $k$ while $ \Delta(\log G)[k] $ does.
\begin{enumerate}[label=(\alph*)]
\item Fix $L < \infty,\, \alpha,\sigma^2>0$. Suppose $P,N\rightarrow \infty$ with $P/N\rightarrow \alpha$. Then,
              
                                                                                                                                                                       
                                                                                                  
                          
            
\begin{align}
    \notag \log G &= \frac{N\alpha}{2}\left[\log\left(\frac{N\alpha}{2}\right) + \log\left(1+\frac{z_*}{\alpha}\right) - \left(1+\frac{z_*}{\alpha}\right)\right] \nonumber \\
    &\label{eq:logG-Lfinite}\quad + \frac{NL}{2}\left[\log\left(\frac{N}{2}\right) + \log\left(1+z_*\right) - (1+z_*)\right]
\end{align}
plus an error of size $\tilde O(1)$ and
\begin{align}
    \label{eq:DlogG-Lfinite} \Delta(\log G)[k] &= kL\left[\log\left(\frac{N}{2}\right) + \log(1+z_*)\right],
\end{align}
              
                                                                                                                                        
            
plus an error of size $\tilde O\left(1/N\right)$, where $z_*> \min\left\{-\alpha,-1\right\}$ is the unique solution to
\begin{equation}\label{eq:Psi-crit}
\left(1+\frac{z_*}{\alpha}\right)\left(1+z_*\right)^L = \frac{\left|\left|\theta_*\right|\right|^2}{\sigma^{2(L+1)}\alpha_0}.
\end{equation} 
\item Fix $\lambda_{\mathrm{prior}},\alpha>0$ and suppose $L=\lambda_{\mathrm{prior}} N, \, P,N\rightarrow \infty,\, P/N\rightarrow \alpha, \, \sigma^2=1$. Then,
\begin{align}
    \label{eq:logG-LN} \log G &= \frac{\lambda_{\mathrm{prior}} N^2}{2}\left[\log\left(\frac{N}{2}\right) - 1\right] + \frac{N\alpha}{2}\left[\log\left(\frac{N\alpha}{2}\right) - 1\right]\\
    &\quad + \frac{\lambda_{\mathrm{prior}} N}{2}\left[-\log\left(\frac{N}{2}\right) + \log(2\pi)\right]  + \tilde O(1)\nonumber
\end{align}
and, up to an error of size $\tilde O\left(1/N\right)$, 
\begin{align}
    \label{eq:DlogG-LN} \Delta(\log G)[k] &= k\left[\lambda_{\mathrm{prior}} N \log\left(\frac{N}{2}\right) + \log \left(\frac{\left|\left|\theta_*\right|\right|^2}{\alpha_0}\right)\right].
\end{align}
              
                                                                                                                                                    
                                               
            
\item Fix $\lambda_{\mathrm{post}} >0$. Suppose $N,P,L\rightarrow \infty$ with $LP/N\rightarrow \lambda_{\mathrm{post}}, \, \sigma^2=1$ and $L/N\to 0$. Then,
\begin{align}
    \label{eq:logG-PLN} \log G &= \frac{P}{2}\left[\log\left(\frac{P}{2}\right)-1\right] + \frac{\lambda_{\mathrm{post}} N}{2}\frac{N}{P}\left[\log\left(\frac{N}{2}\right)-1\right] \nonumber \\
    &\quad + \frac{P}{2}\left[\log(1+t_*)-t_*\left(1+\frac{\lambda_{\mathrm{post}} t_*}{2}\right)\right] \nonumber \\
    &\quad + \frac{\lambda_{\mathrm{post}} N}{2}\left[\frac{N}{P}\left(1+\frac{P}{N}t_*\right)\log\left(1+\frac{P}{N}t_*\right) - t_*\right] \nonumber\\
    &\quad + \tilde O(1),
\end{align}
and
\begin{align}
    \label{eq:DlogG-PLN}   \Delta(\log G)[k] &= k\left[L \log \left(\frac{N}{2}\right) + \lambda_{\mathrm{post}} t_*\right] + \tilde{O}\left(\frac{1}{N}\right),
\end{align}
where $t_*$ is the unique solution to
\begin{align}
    e^{\lambda_{\mathrm{post}} t_*}(1+t_*) &= \left|\left|\theta_*\right|\right|^2/\alpha_0.
\end{align}
\end{enumerate}
In all the estimates above, $\tilde O(1) = O(\max\{\log P, \log N\})$ suppresses lower-order terms of order 1, up to logarithmic factors; similarly, $\tilde O(1/N) = O(\max\{\log P, \log N\}/N)$. Suppressed terms are included in \S \ref{pf:logG}

\label{thm:bayesfeature}
Fix $L, N_1\dots, N_L \geq 1$, $\sigma^2 > 0$, $\alpha_0\in (0,1)$, and consider sequences of training data sets $X_{N_0}\in \mathbb R^{N_0\times P}, Y_{N_0}\in \mathbb R^{1\times P}$ such that
\begin{align*}
    P, N_0 \to \infty, \qquad P/N_0 \to \alpha_0
\end{align*}
and \eqref{eq:Vstar-def} holds. Let $Z_\infty({\bf 0})$ denote the limiting Bayesian model evidence. The following statements are equivalent:
\begin{enumerate}[label=(\alph*)]
    \item $\sigma^2$ maximizes $Z_\infty({\bf 0})$, and
    \item the posterior over predictions $f(x)$ converges weakly to a Gaussian $\mathcal{N}\left(\mu_*, \nu \Sigma_\perp\right)$.
\end{enumerate}

<!-- theorem-4: body_md -->
\label{thm:finite-L}
For each $N_0,P\geq 1$, consider training data $X_{N_0},Y_{N_0}$ satisfying \eqref{eq:Vstar-def}. Fix constants $L\geq 0,\,\sigma^2 > 0$ and suppose that
\begin{equation*}
P,N_0,N \rightarrow \infty,\quad P/N_0 \rightarrow \alpha_0\in (0,1),\quad P/N\rightarrow \alpha\in (0,\infty). 
\end{equation*}
In this limit, the posterior over predictions $f(x)$ converges weakly to a Gaussian
\begin{align*}
\mathcal{N}\left(\mu_*,~\nu\Sigma_\perp\left(1+\frac{z_*}{\alpha}\right)^{-1}\right),
\end{align*}
where $z_*$ is the unique solution to
\begin{equation}\label{eq:zstar}
\frac{\nu}{\sigma^{2(L+1)}} = \left(1+\frac{z_*}{\alpha}\right)\left(1+z_*\right)^L,\quad z_* > \max\left\{-1,-\alpha\right\}.
\end{equation}
Additionally, we have the following large-$P$ expansion of the Bayesian model evidence
\begin{align*}
\log Z_\infty({\bf 0}) &~\rightarrow~ \frac{P}{2}\Bigg[\log\left(\frac{P}{2}\right) + \log\left(1+\frac{z_*}{\alpha}\right) - \left(1+\frac{z_*}{\alpha}\right) \nonumber \\
&\quad - \log\left(\frac{\left|\left|\theta_*\right|\right|^2}{4\pi}\right)\Bigg] + \frac{NL}{2}\left[\log\left(1+z_*\right) - z_*\right] \nonumber\\
&\quad + O(\max\left\{\log P, \log N\right\}).
\end{align*}

<!-- corollary-1: body_md -->
\label{cor:Lfix-evid}
In the setting of Theorem \ref{thm:finite-L}  the Bayesian evidence $Z_\infty({\bf 0})$ satisfies: 
\begin{align}
\label{eq:sig-star}\sigma_*^2 &= \operatorname*{arg\,max}_{\sigma}Z_\infty({\bf 0}) = \nu^{\frac{1}{L+1}}\\
\label{eq:L-star} L_* &= \operatorname*{arg\,max}_{L}Z_\infty({\bf 0}) =\frac{\log(\nu)}{\log(\sigma^2)}-1.
\end{align}
In particular, given a prior variance with $\mathrm{sgn}\left(\nu-1\right)=\mathrm{sgn}\left(\sigma^2-1\right)$ satisfying $|\sigma^2-1| \leq \epsilon$, the optimal depth network satisfies $L_* \geq |\log(\nu)|/\epsilon$.

\label{thm:LN}
For each $N_0,P\geq 1$, consider training data $X_{N_0},Y_{N_0}$ satisfying \eqref{eq:Vstar-def}. Moreover, fix $\lambda_{\mathrm{prior}},\alpha \in (0,\infty),\, \alpha_0\in (0,1).$ Suppose that $N_1=\cdots=N_L=N$ and that
\begin{equation*}
P,N_\ell,L\rightarrow \infty,\quad P/N_0\rightarrow \alpha_0,\quad P/N\rightarrow \alpha,\quad L/N\rightarrow \lambda_{\mathrm{prior}}.
\end{equation*}
In this limit, the posterior over predictions $f(x)$ converges weakly to a Gaussian
\begin{align*}
\mathcal{N}\left(\mu_*,~\nu\Sigma_\perp\right),
\end{align*}
which is independent of $\alpha,\lambda_{\mathrm{prior}}$. Additionally, the evidence admits the following asymptotic expansion:
\begin{align*}
\log Z_\infty({\bf 0}) &= \frac{P}{2}\left[\log\left(\frac{P}{2}\right) - 1 - \log\left(\frac{\left|\left|\theta_*\right|\right|^2}{4\pi}\right)\right] + O(\max\left\{\log P, \log N\right\}).
\end{align*}

\label{cor:feature-learning}
In the setting of Theorem \ref{thm:LN}, the posterior distribution over regression weights $\theta$ is the same in the following two settings:
\begin{itemize}
	\item We fix $L\geq 0$, take the data-dependent prior variance $\sigma_*$ that maximizes the Bayesian model evidence as a function of the training data and network depth as in \eqref{eq:sig-star}, and send the network width $N$ to infinity.
	\item We fix $\lambda_{\mathrm{prior}} >0$, take a data-agnostic prior $\sigma^2=1$, and send both the network depth $L:=\lambda_{\mathrm{prior}} \cdot N$ and network width $N$ to infinity together.
\end{itemize}

<!-- theorem-6: body_md -->
\label{thm:PLN}
For each $N_0,P\geq 1$, consider training data $X_{N_0},Y_{N_0}$ satisfying \eqref{eq:Vstar-def}. Moreover, fix constants $\lambda_{\mathrm{post}} > 0,\, \alpha_0\in (0,1), \, \sigma^2=1$. Suppose that $N_1=\cdots = N_L=N$ and suppose that
\begin{align*}
P, N_\ell, L \to \infty, \; \frac{P}{N_0} \to \alpha_0 , \; \frac{P}{N} \to 0, \; \frac{L}{N} \to 0, \; \frac{LP}{N} \to \lambda_{\mathrm{post}}.
\end{align*}
In this limit, the posterior over predictions $f(x)$ converges weakly to a Gaussian $\mathcal{N}\left(\mu_*,~\nu\Sigma_\perp(1+t_*)^{-1}\right)$,
               
                                                    
             
where $t_*$ is the unique solution to
\begin{align*}
    \nu = (1+t_*)e^{\lambda_{\mathrm{post}} t_*}, \quad t_* > -1.
\end{align*}
Additionally, we have the following asymptotic expansion for the Bayesian model evidence
\begin{align*}
    \log Z_\infty({\bf 0}) &= \frac{P}{2}\left(\log\left(\frac{P}{2}\right)-1\right) - \frac{P}{2} \log\left(\frac{\left|\left|\theta_*\right|\right|^2}{4\pi}\right)\\
    &\quad + \frac{P}{2}\left[\log(1+t_*)-t_* - \frac{1}{2}\lambda_{\mathrm{post}} t_*^2\right] + \tilde O(\max\left\{\log P, \log N, \log L\right\}).
\end{align*}

\label{thm:scaling}

\end{align*}
where $C\in \mathbb R$ is a universal constant.

<!-- proposition-1: body_md -->
\label{prop:gamma-exp}
We have the following analytic asymptotic expansion for the Gamma function
\begin{equation}\label{eq:Gamma-exp}
\log \Gamma\left(z\right) \sim \left(z-\frac{1}{2}\right)\log(z) - z + \text{const}+O\left(\left|z\right|^{-1}\right),\quad \text{as }\left|z\right|\rightarrow \infty.
\end{equation}
In particular, we also have
\begin{equation}\label{eq:diGamma-exp}
\phi^{(1)}(z)=\frac{d}{dz}\log \Gamma\left(z\right) \sim \log(z) + O(\left|z\right|^{-1}),\quad \text{as }\left|z\right|\rightarrow \infty.
\end{equation}
Both expansions hold uniformly on sets of the form $\left|z\right|< \pi - \delta$ for a fixed $\delta >0$.

\begin{equation}\label{E:gamma-moments}
\mathbb E\left[\phi^t\right] = \theta^t\frac{\Gamma\left(k+t\right)}{\Gamma\left(k\right)}.

\mathbb E\left[e^{-it \log\phi}\right] = \exp\left[-it\log \theta + \log \Gamma\left(k-it\right)-\log \Gamma(k)\right]     
\end{equation}
is a meromorphic function of $t$ with poles on the negative imaginary axis:
\[
t = -i(\nu +k),\quad \nu = 0,1,2,\ldots.

<!-- lemma-5: body_md -->
\label{lem:Psi-est}
There exist $c,T_0>0$ such that
\begin{equation}\label{eq:Psi-est-1}
    \sup_{\substack{\left|z\right|>T\\ z\in \mathbb R}} \frac{\Re \Psi(z)}{1+\left|z\right|} \leq - c,\qquad \forall T\geq T_0.
\end{equation}
Moreover, for any $T,\delta>0$, writing
\begin{equation}\label{eq:S-def}
    S_{N,\delta,T}:=\left\{z\in \mathbb C~|~ \left|z\right|< T, \Im(z) >C_\delta\right\},
\end{equation}
there exists $C>0$ such that
\begin{equation}\label{eq:Psi-est-2}
    \sup_{z\in S_{N,\delta, T}}\max\left\{\left|\Psi(z)\right|,\left| \frac{d}{dz}\Psi(z)\right|\right\} \leq C.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
To show both \eqref{eq:Psi-est-1} and \eqref{eq:Psi-est-2} we will use the asymptotic expansion
\begin{equation}\label{eq:Gamma-exp-pf}
\log \Gamma\left(z\right) \sim \left(z-\frac{1}{2}\right)\log(z) - z + \text{const}+O\left(\left|z\right|^{-1}\right),\quad \text{as }\left|z\right|\rightarrow \infty,    
\end{equation}
which holds uniformly on sets of the form $\left|z\right|< \pi - \epsilon$ for a fixed $\epsilon >0$. With $\Psi_\ell$ defined in \eqref{eq:Psi-ell-def}, we find for each $\ell=1,\ldots, L$ that uniformly over $z\in \mathbb R$ 
\begin{align}
  \notag  \Psi_\ell(z)&=-iz\log\left(\frac{2\sigma^2}{N_\ell}\right) +\frac{1}{N}\left(\log \Gamma\left(\frac{N_\ell}{2}+k+1-iNz\right)-\log \Gamma\left(\frac{N_\ell}{2}+k+1\right)\right)\\
  \notag  &=-iz\left[-1+\log\left(\frac{2\sigma^2}{N_\ell}\right) + \log\left(\frac{N_\ell}{2}+k+1-iNz\right)\right]\\
\notag    &\qquad +\frac{1}{N}\left(\frac{N_\ell+1}{2}+k+1\right)\log\left(1-\frac{iNz}{\frac{N_\ell}{2}+k+1}\right) + O(N^{-1})\\
\notag    &=-iz\left[-1+\log(\sigma^2) + \log\left(1-\frac{2i N z}{N_\ell}\right)\right] + \frac{N_\ell}{2N}\log\left(1-\frac{2iNz}{N_\ell}\cdot\frac{1}{1+\frac{2(k+1)}{N_\ell}}\right)+O(N^{-1})\\
\notag   &=-iz\left[-1+\log(\sigma^2)+\log\left(1-\frac{2i N z}{N_\ell}\right)\right] + \frac{N_\ell}{2N}\log\left(1-\frac{2iNz}{N_\ell}\right)+O(N^{-1}).
\end{align}
Obtaining a similar expression for $\ell=0$ and summing over $\ell$ proves \eqref{eq:Psi-est-2}. Moreover, for $\ell=1,\ldots, L$ we obtain uniformly on $z\in \mathbb R$ 
\begin{align*}
    \Re\Psi_\ell(z) = z\arg\left(1-\frac{2iNz}{N_\ell}\right) + \frac{N_\ell}{2N}\log\left|1-\frac{2iNz}{N_\ell}\right|+O(N^{-1}).
\end{align*}
Note that for $T_0$ sufficiently large there we have
\begin{equation}\label{eq:RePsi-bound}
\left|z\right|> T_0\quad \Rightarrow \quad \arg\left(1-\frac{2iNz}{N_\ell}\right) \mathrm{sgn}\left(z\right) \leq  -\frac{\pi}{4}\quad  \Rightarrow \quad \Re\Psi_\ell(z) \leq -\frac{\pi}{8}\left|z\right|.
\end{equation}
A similar analysis applies to $\ell=0$ and completes the proof of \eqref{eq:Psi-est-1}.

\label{cor:optevd}
In the setting of Theorem 6, we have
\[
\lambda_{\mathrm{prior},*} = \sqrt{1 + \log(\nu)^2}-1 = \operatorname*{arg\,max}_{\lambda}\lim_{\substack{P,N_\ell \rightarrow \infty\\ P/N_0\rightarrow \alpha_0 \in (0,1)\\ \lambda_{\mathrm{prior}}(N_1,\ldots, N_L)\rightarrow \lambda}} Z_\infty\left({\bf 0}~|~L,N_\ell,\sigma^2=1,X_{N_0},Y_{N_0}\right).
\]
Moreover, for any $\lambda>0$, there exists $c>0$ such that we have
\begin{align*}
    c < \lim_{\substack{P,N_\ell \rightarrow \infty\\ P/N_0\rightarrow \alpha_0 \in (0,1)}} \frac{Z_\infty\left({\bf 0}~|~L=N\lambda,N_\ell=N,\sigma^2=1,X_{N_0},Y_{N_0}\right)}{ Z_\infty\left({\bf 0}~|~L=N\lambda_{\text{prior},*},N_\ell=N,\sigma^2=1,X_{N_0},Y_{N_0}\right)}\leq 1,
\end{align*}
i.e., the ratio of evidences is lower-bounded by a constant.

<!-- proof-7: body_md -->
In Corollary~\ref{cor:optevd}, we compute the Bayes-optimal neural network depth by maximizing Bayesian evidence~\cite{mackay1992bayesian}. A similar computation to the following is used to find Bayes-optimal parameters throughout the main text; we provide the proof to Corollary~\ref{cor:optevd} as an example. To evaluate $\partial \log Z / \partial \lambda_{\mathrm{prior}} = 0$ for the partition function given by Theorem \ref{thm:Z-form}, we take the partition function with $N_1 = \dots = N_L = N$ obtained from case (b) of Theorem \ref{thm:logG},
\begin{align*}
    \log Z_\infty(0) &= \frac{P}{2}\log\left(\frac{4\pi}{\|\theta_*\|^2}\right) + \frac{P}{2}\left[\log\left(\frac{P}{2}\right)-1\right] -\frac{1}{2}\log\left(\frac{P}{2}\right)\\
    &\quad  - \frac{1}{2}\log(2\lambda_{\mathrm{prior}}) - \frac{1}{4\lambda_{\mathrm{prior}}}\left(\lambda_{\mathrm{prior}} + \log\left(\frac{\|\theta_*\|^2}{\alpha_0}\right)\right)^2,
\end{align*}
and we evaluate the derivative with variable substitution $\nu = \|\theta_*\|^2/\alpha_0$, yielding
\begin{align*}
    \frac{\partial \log Z_\infty(0)}{\partial \lambda_{\mathrm{prior}}} &= \frac{\log(\nu)^2-\lambda_{\mathrm{prior}}(2+\lambda_{\mathrm{prior}})}{4\lambda_{\mathrm{prior}}^2} = 0.
\end{align*}
Solving gives
\begin{align*}
    \lambda_{\mathrm{prior},*} &= \sqrt{1 + \log^2 \nu} - 1 \geq 0.
\end{align*}
Moreover, the second derivative is
\begin{align*}
    \frac{\partial^2 \log Z_\infty(0)}{\partial \lambda_{\mathrm{prior}}^2} &= \frac{\lambda_{\mathrm{prior}}-\log^2\nu}{2\lambda_{\mathrm{prior}}^3},
\end{align*}
which at $\lambda_{\mathrm{prior},*}$ is
\begin{align*}
    \frac{\partial^2 \log Z_\infty(0)}{\partial \lambda_{\mathrm{prior}}^2} &= -\frac{\lambda_{\mathrm{prior},*}+1}{2\lambda_{\mathrm{prior},*}^2} < 0.
\end{align*}
Hence, Bayesian evidence is maximized at $\lambda_{\mathrm{prior},*} = O(1)$.

Moreover, evaluating the evidence at a different choice of fixed $\lambda_{\mathrm{prior}}$ such that $\lambda_{\mathrm{prior}} \neq \lambda_{\mathrm{prior},*}$, we see that the evidence ratio is
\begin{align*}
    \log Z_\infty(0; \lambda_{\mathrm{prior}}) - \log Z_\infty(0; \lambda_{\mathrm{prior},*}) &= -\frac{1}{2}\log\left(\frac{\lambda_{\mathrm{prior}}}{\lambda_{\mathrm{prior},*}}\right) - \frac{1}{4\lambda_{\mathrm{prior}}}\left(\lambda_{\mathrm{prior}} + \log\left(\frac{\|\theta_*\|^2}{\alpha_0}\right)\right)^2 \\
    &\quad + \frac{1}{4\lambda_{\mathrm{prior},*}}\left(\lambda_{\mathrm{prior},*} + \log\nu\right)^2,
\end{align*}
which is a constant.
