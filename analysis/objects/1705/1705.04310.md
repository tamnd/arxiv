\label{def:cwf.fiore}
  A \emph{category with families (à la Fiore)} consists of:
  \begin{enumerate} \setcounter{enumi}{-1}
  \item a category ${\mathcal{C}}$, together with
  \item presheaves $\ensuremath{\mathsf{Ty}}, \ensuremath{\mathsf{Tm}} : {\mathcal{C}}^{\mathrm{op}} \to \ensuremath{\ensuremath{\mathsf{Set}}}\xspace$; \label{item:cwf.presheaves}
  \item a natural transformation $\ensuremath{\mathsf{p}} : \ensuremath{\mathsf{Tm}} \to \ensuremath{\mathsf{Ty}}$; and
  \item  for each object $\Gamma : {\mathcal{C}}$ and $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$, a representation of the fiber of $\ensuremath{\mathsf{p}}$ over $A$, i.e. \label{item:cwf.fiber-rep}
    \begin{enumerate}
    \item an object $\ensuremath{{\Gamma}.{A}} : {\mathcal{C}}$ and map $\ensuremath{\mathsf{\pi}}_{A} : \ensuremath{{\Gamma}.{A}} \to \Gamma$, \label{item:cwf.first}
    \item an element $\ensuremath{\ensuremath{\mathsf{te}}}_A:\ensuremath{\ensuremath{\mathsf{Tm}}({\ensuremath{{\Gamma}.{A}}})}$, such that $\ensuremath{\mathsf{p}}_{\ensuremath{{\Gamma}.{A}}}(\ensuremath{\ensuremath{\mathsf{te}}}_A) = \ensuremath{\ensuremath{\mathsf{\pi}}_{A}^*{A}} : \ensuremath{\ensuremath{\mathsf{Ty}}({\ensuremath{{\Gamma}.{A}}})},$
    \item and such that the induced commutative square \label{item:cwf.last}
      \[
      \begin{xy}
        \xymatrix{
                **[l]\ensuremath{\mathsf{y}}_{}(\ensuremath{{\Gamma}.{A}})  \ar[r]^-{\widehat{\ensuremath{\ensuremath{\mathsf{te}}}_A}} \ar[d]_{\ensuremath{\mathsf{y}}_{}(\ensuremath{\mathsf{\pi}}_{A})}  \ar@{}[dr]|<<{\text{\pigpenfont A}} & \ensuremath{\mathsf{Tm}} \ar[d]^{\ensuremath{\mathsf{p}}}\\
                **[l]\ensuremath{\mathsf{y}}_{}(\Gamma) \ar[r]_{\widehat{A}}& \ensuremath{\mathsf{Ty}}
              }
      \end{xy}
      \]
      is a pullback.
       
      (Here e.g.\ $\widehat{A}$ denotes the Yoneda transpose of an element of a presheaf.)
       
    \end{enumerate}
  \end{enumerate}
   
  By a \emph{CwF structure} on a category ${\mathcal{C}}$, we mean the data of items \ref{item:cwf.presheaves}--\ref{item:cwf.fiber-rep} above.

\label{def:split.type.cat}

   A \emph{type-category} consists of:
  \begin{enumerate} \setcounter{enumi}{-1}
  \item a category ${\mathcal{C}}$, together with
  \item for each object $\Gamma : {\mathcal{C}}$, a type $\ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$,
  \item for each $\Gamma : {\mathcal{C}}$ and $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$, an object $\ensuremath{{\Gamma}.{A}} : {\mathcal{C}}$,
  \item for each such $\Gamma$, $A$, a morphism $\ensuremath{\mathsf{\pi}}_{A} : \ensuremath{{\Gamma}.{A}} \to \Gamma$,
  \item for each map $f : \Gamma' \to \Gamma$, a ‘reindexing’ function $\ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})} \to \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma'})}$, denoted $A \mapsto \ensuremath{f^*{A}} $,
  \item for each $\Gamma$, $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$, and $f : \Gamma' \to \Gamma$, a morphism
         $\ensuremath{\ensuremath{\mathsf{q}}({f},{A})} : \ensuremath{{\Gamma'}.{\ensuremath{f^*{A}}}} \to \ensuremath{{\Gamma}.{A}}$,
  \item such that for each such $\Gamma$, $A$, $\Gamma'$, $f$, the following square commutes and is a pullback:
         \[
           \xymatrix@C=5pc{
                        \ensuremath{{\Gamma'}.{\ensuremath{f^*{A}}}} \ar[r]^{\ensuremath{\ensuremath{\mathsf{q}}({f},{A})}} \ar[d]_{\ensuremath{\mathsf{\pi}}_{\ensuremath{f^*{A}}}} \ar@{}[dr]|<<{\text{\pigpenfont A}} &
                                                                        \ensuremath{{\Gamma}.{A}} \ar[d]^{\ensuremath{\mathsf{\pi}}_{A}} \\
                        \Gamma' \ar[r]_{f}& \Gamma
           }
         \]
  \setcounter{saveenumi}{\value{enumi}}
  \end{enumerate}

  \noindent A type-category is \emph{split} if:
  \begin{enumerate} \setcounter{enumi}{\value{saveenumi}}
   \item for each $\Gamma$, the type $\ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$ is a set;
   \item for each $\Gamma$ and $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$, we have equalities
     \begin{enumerate}
     \item $e : \ensuremath{1_{\Gamma}^*{A}} = A$, and
     \item $\ensuremath{\ensuremath{\mathsf{q}}({1_{\Gamma}},{A})} = \Delta_{e}\ :\ \ensuremath{{\Gamma}.{\ensuremath{1_\Gamma^*{A}}}} \to \ensuremath{{\Gamma}.{A}}$; and
     \end{enumerate}
   \item for $f' : \Gamma'' \to \Gamma')$, $f : \Gamma' \to \Gamma$, and $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$, we have equalities
     \begin{enumerate}
     \item $e' : \ensuremath{(\ensuremath{{f'}\cdot{f}})^*{A}} = \ensuremath{f'^*{\ensuremath{f^*{A}}}}$, and
     \item $\ensuremath{\ensuremath{\mathsf{q}}({\ensuremath{{f'}\cdot{f}}},{A})}
       = \ensuremath{{\ensuremath{{\Delta_{e'}}\cdot{\ensuremath{\ensuremath{\mathsf{q}}({f'},{\ensuremath{f^*{A}}})}}}}\cdot{\ensuremath{\ensuremath{\mathsf{q}}({f},{A})}}}
       \ :\ \ensuremath{{\Gamma''}.{\ensuremath{(\ensuremath{{f'}\cdot{f}})^*{A}}}} \to \ensuremath{{\Gamma}.{A}}.$
     \end{enumerate}
  \end{enumerate}

A \emph{(split) object extension structure} on a category ${\mathcal{C}}$ consists of:
  \begin{enumerate}
   \item a functor $\ensuremath{\mathsf{Ty}} : {\mathcal{C}}^{\mathrm{op}} \to \ensuremath{\ensuremath{\mathsf{Set}}}\xspace$;
   \item for each $\Gamma : {\mathcal{C}}$ and $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$,
      \begin{enumerate}
       \item an object $\ensuremath{{\Gamma}.{A}}$, and
       \item a \emph{projection} morphism $\ensuremath{\mathsf{\pi}}_{A} : {\mathcal{C}}(\ensuremath{{\Gamma}.{A}}, \Gamma)$.
      \end{enumerate}
  \end{enumerate}
   
  In general, one might want to loosen the setness and functoriality conditions on $\ensuremath{\mathsf{Ty}}$, and hence distinguish the present definition as the split version of a more general notion.
   
  In this paper, however, we do not consider the non-split case, so take \emph{object extension structures} to mean the split ones throughout.

\label{def:cartesian.generator}
  Let ${\mathcal{C}}$ be a category equipped with an object extension structure $X = (\ensuremath{\mathsf{Ty}},\ensuremath{{-}.{-}},\pi)$.
   
  A \emph{(functional) term-structure} over $X$ consists of:
 \begin{enumerate}
  \item a presheaf $\ensuremath{\mathsf{Tm}} : {\mathcal{C}}^{\mathrm{op}} \to \ensuremath{\ensuremath{\mathsf{Set}}}\xspace$, and natural transformation $\ensuremath{\mathsf{p}} : \ensuremath{\mathsf{Tm}} \to \ensuremath{\mathsf{Ty}}$;
  \item  for each object $\Gamma : {\mathcal{C}}$ and $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$, an element $\ensuremath{\ensuremath{\mathsf{te}}}_A:\ensuremath{\ensuremath{\mathsf{Tm}}({\ensuremath{{\Gamma}.{A}}})}$, such that $(\ensuremath{{\Gamma}.{A}}, \ensuremath{\mathsf{\pi}}_{A}, \ensuremath{\ensuremath{\mathsf{te}}}_A)$ form a representation of the fiber of $\ensuremath{\mathsf{p}}$ over $A$ as in item \ref{item:cwf.fiber-rep} of Def.~\ref{def:cwf.fiore}.
 \end{enumerate}
  
 One might say \emph{functional} to distinguish these from \emph{familial} term-structures, which would correspond analogously to CwF’s in the sense of Dybjer, with $\ensuremath{\ensuremath{\mathsf{Tm}}({\Gamma})}$ a family indexed by $\ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$.
  
 For the present paper, however, we work only with the functional ones, so call these simply \emph{term-structures}.

<!-- example-1: body_md -->
\label{ex:no-bijec-in-zf}
   Let ${\mathcal{C}}$ be the terminal category $\{ \bullet \}$, and $\ensuremath{\mathsf{Ty}} : {\mathcal{C}}^{\mathrm{op}} \to \ensuremath{\ensuremath{\mathsf{Set}}}\xspace$ be the terminal functor, $\ensuremath{\mathsf{Ty}}(\bullet) := \{ \star \}$.
    
   There is a unique object extension structure extending $({\mathcal{C}},\ensuremath{\mathsf{Ty}})$.

   A term-structure on this amounts to a set $\ensuremath{\ensuremath{\mathsf{Tm}}({\bullet})}$ and element $\ensuremath{\ensuremath{\mathsf{te}}}_\star \in \ensuremath{\ensuremath{\mathsf{Tm}}({\bullet})}$ such that
   \[
       \begin{xy}
        \xymatrix{
                           {\mathcal{C}}(\bullet,\bullet) \ar[r]^{\widehat{\ensuremath{\ensuremath{\mathsf{te}}}_\star}} \ar[d]   & \ensuremath{\ensuremath{\mathsf{Tm}}({\bullet})} \ar[d] \\
                           {\mathcal{C}}(\bullet,\bullet)  \ar[r] & \ensuremath{\ensuremath{\mathsf{Ty}}({\bullet})}
        }
       \end{xy}
   \]
   is a pullback square.
    
   So $\ensuremath{\mathsf{Ty}}(\bullet)$ must be a singleton set, and given that, the rest of the data is uniquely determined.
   
   On the other hand, to give a $\ensuremath{\mathsf{q}}$-morphism structure on $({\mathcal{C}}, \ensuremath{\mathsf{Ty}})$, there is no choice at all:
   all the required data consists of maps in ${\mathcal{C}}$, so is uniquely determined.

   In other words, there is exactly one $\ensuremath{\mathsf{q}}$-morphism structure on $({\mathcal{C}}, \ensuremath{\mathsf{Ty}})$, while term-structures on $({\mathcal{C}}, \ensuremath{\mathsf{Ty}})$ correspond precisely to singletons in $\ensuremath{\ensuremath{\mathsf{Set}}}\xspace$.
   
   In particular, in ZF(C) or similar set-theoretic settings, with the universe taken to be some Grothendieck universe, there are many singletons in $\ensuremath{\ensuremath{\mathsf{Set}}}\xspace$.
    
   So there is no bijection between term-structures and $\ensuremath{\mathsf{q}}$-morphism structures on $({\mathcal{C}},\ensuremath{\mathsf{Ty}})$: Problem~\ref{problem:weq-term-qmor} is insoluble.

\label{def:rel.univ.struct}
  Let $J : {\mathcal{C}} \to {\mathcal{D}}$ be a fully faithful functor.

  A \emph{$J$-universe structure} on a map $\ensuremath{\mathsf{p}} : \ensuremath{\mathsf{\tilde{U}}} \to \ensuremath{\mathsf{U}}$ of ${\mathcal{D}}$ is a function giving,
  for each object $X$ in ${\mathcal{C}}$ and map $f: J(X) \to \ensuremath{\mathsf{U}}$, a $J$-pullback $(\ensuremath{{X}.{f}}, \ensuremath{\ensuremath{\mathsf{p}}_{f}}, \ensuremath{\ensuremath{\mathsf{Q}}({X},{f})})$ of $\ensuremath{\mathsf{p}}$ along $f$.
   
  A \emph{universe relative to $J$}, or briefly a \emph{$J$-universe}, is a map $\ensuremath{\mathsf{p}}$ equipped with a $J$-universe structure.

  A \emph{weak universe relative to $J$} is a map $\ensuremath{\mathsf{p}} : \ensuremath{\mathsf{\tilde{U}}} \to \ensuremath{\mathsf{U}}$ such that for all suitable $X$, $f$, there exists some $J$-pullback of $\ensuremath{\mathsf{p}}$ along $f$.
<!-- end definition-9 -->
<!-- remark-2: body_md -->
In the conference version \cite{alv1-csl} of this article, the condition in 
 Definition~\ref{def:rel.univ.struct} on $J : {\mathcal{C}} \to {\mathcal{D}}$ being fully faithful was not imposed.
 
 This condition
 entails that the map $(X,f) \mapsto \ensuremath{{X}.{f}}$ can be equipped with a functorial action in a suitable sense, and that 
 the map $(X,f) \mapsto \ensuremath{\ensuremath{\mathsf{p}}_{f}}$ is natural with respect to this functorial action.
 More specifically, the type of this extra ``functoriality data'' on a $J$-universe $\ensuremath{\mathsf{p}}$ as above is a proposition when $J$ is faithful
 (\nolinkurl{isaprop_functorial_structure_relu}), 
 and is contractible when $J$ is fully faithful
 (\nolinkurl{iscontr_functorial_structure_relu}).
 
 If the restriction to fully faithful $J$ is dropped, 
 then the definition of $J$-universes should probably be expanded to include the functoriality data.
 
 However, our two motivating examples, $J=\ensuremath{\mathsf{Id}}_{{\mathcal{C}}}$ and $J=\ensuremath{\mathsf{y}}_{{\mathcal{C}}}$, are both fully faithful. 
 For the sake of simplicity, therefore, we give only this ``naïve'' definition of relative universes in the present work.

<!-- problem-7: body_md -->
\label{problem:universe.transfer.concrete}
  Given a square of functors commuting up to natural isomorphism, as in
  \[
      \begin{xy}
        \xymatrix{
                 {\mathcal{C}} \drtwocell<\omit>{\alpha} \ar[r]^{J} \ar[d]_{R}& {\mathcal{D}} \ar[d]^{S}\\
                 {\mathcal{C}}' \ar[r]_{J'} & {\mathcal{D}}',
        }
      \end{xy}
  \]
  such that
  \begin{itemize}
     \item $J$ and $J'$ are fully faithful,
     \item $S$ preserves pullbacks,
     \item $S$ is split full, and
     \item $R$ is split essentially surjective,
   \end{itemize}
   and given a $J$-relative universe structure on a map $\ensuremath{\mathsf{p}} : \ensuremath{\mathsf{\tilde{U}}} \to \ensuremath{\mathsf{U}}$ in ${\mathcal{D}}$,
   to construct a $J'$-universe structure on $S(\ensuremath{\mathsf{p}})$ in ${\mathcal{D}}'$.
<!-- end problem-7 -->
<!-- construction-7: body_md -->
{constr:universe.transfer.concrete}
  Given $X$ in ${\mathcal{C}}'$ and $f : J'(X) \to S\ensuremath{\mathsf{U}}$, we need to construct a $J'$-pullback of $S(\ensuremath{\mathsf{p}})$ along $f$.

  Split essential surjectivity of $R$ gives some $\bar{X} : {\mathcal{C}}$ and isomorphism $i : R(\bar{X}) \cong X$.
   
  We therefore have $\ensuremath{{\alpha_{\bar{X}}}\cdot{\ensuremath{{J'i}\cdot{f}}}} : SJ\bar{X} \to S\ensuremath{\mathsf{U}}$; so split fullness of $S$ gives us some $\bar{f} : J\bar{X} \to \ensuremath{\mathsf{U}}$ with $S \bar{f} = \ensuremath{{\alpha_{\bar{X}}}\cdot{\ensuremath{{J'i}\cdot{f}}}}$.
   
  Taking the given $J$-pullback of $\ensuremath{\mathsf{p}}$ along $\bar{f}$ and mapping it forward under $S$, we get a pullback square in ${\mathcal{D}}'$:
   
  \[
    \xymatrix@C=3.5pc{
      SJ(\ensuremath{{\bar{X}}.{\bar{f}}}) \ar[rrr]^{S(\ensuremath{\ensuremath{\mathsf{Q}}({\bar{X}},{\bar{f}})})} \ar[d]_{SJ\ensuremath{\ensuremath{\mathsf{p}}_{\bar{f}}}} \ar@{}[dr]|<<{\text{\pigpenfont A}}
      & & & S \ensuremath{\mathsf{\tilde{U}}} \ar[d]^{S \ensuremath{\mathsf{p}}}
      \\
      S J \bar{X} \ar[r]^{\alpha_{\bar{X}}}
      & J' R \bar{X} \ar[r]^{J'i}
      & J' X \ar[r]^f
      & S \ensuremath{\mathsf{U}}
    }
  \]

  The maps $\ensuremath{{R(\ensuremath{\ensuremath{\mathsf{p}}_{\bar{f}}})}\cdot{i}} : R(\ensuremath{{\bar{X}}.{\bar{f}}}) \to X$ and $\ensuremath{{\alpha_{\ensuremath{{\bar{X}}.{\bar{f}}}}^{-1}}\cdot{S(\ensuremath{\ensuremath{\mathsf{Q}}({\bar{X}},{\bar{f}})})}} : J'R(\ensuremath{{\bar{X}}.{\bar{f}}}) \to S \ensuremath{\mathsf{\tilde{U}}}$ now give the desired $J'$-pullback of $S\ensuremath{\mathsf{p}}$ along $f$, since the square they give---the right-hand square below---is isomorphic to the pullback obtained above.
   
                                                                   
  \[ \begin{gathered}[b] \xymatrix@C=3pc{
      SJ(\ensuremath{{\bar{X}}.{\bar{f}}}) \ar[r]^{\alpha_{\ensuremath{{\bar{X}}.{\bar{f}}}}}_{\cong} \ar[d]_{SJ\ensuremath{\ensuremath{\mathsf{p}}_{\bar{f}}}}
      & J'R(\ensuremath{{\bar{X}}.{\bar{f}}}) \ar[r]^{1}_{\cong} \ar[d]^{J'R\ensuremath{\ensuremath{\mathsf{p}}_{\bar{f}}}}
      & J'R(\ensuremath{{\bar{X}}.{\bar{f}}}) \ar[rr]^{\ensuremath{{\alpha_{\ensuremath{{\bar{X}}.{\bar{f}}}}^{-1}}\cdot{S(\ensuremath{\ensuremath{\mathsf{Q}}({\bar{X}},{\bar{f}})})}}} \ar[d]^{J' (\ensuremath{{R(\ensuremath{\ensuremath{\mathsf{p}}_{\bar{f}}})}\cdot{i}})} \ar@{}[dr]|<<{\text{\pigpenfont A}}
      & & S \ensuremath{\mathsf{\tilde{U}}} \ar[d]^{S \ensuremath{\mathsf{p}}}
      \\
      S J \bar{X} \ar[r]^{\alpha_{\bar{X}}}_{\cong}
      & J' R \bar{X} \ar[r]^{J'i}_{\cong}
      & J' X \ar[rr]^f
      & & S \ensuremath{\mathsf{U}}
  } \\[-\dp\strutbox] \end{gathered} \qedhere \]

<!-- definition-10: body_md -->
\label{def:rep_map}
 A map $\ensuremath{\mathsf{p}} : \ensuremath{\mathsf{Tm}} \to \ensuremath{\mathsf{Ty}}$ of presheaves on a category ${\mathcal{C}}$ is \emph{representable} if for each for each $\Gamma : {\mathcal{C}}$ and $A : \ensuremath{\ensuremath{\mathsf{Ty}}({\Gamma})}$, there exists some representation of the fiber of $\ensuremath{\mathsf{p}}$ over $A$, i.e.\ some $\ensuremath{\mathsf{\pi}}_{A} : \ensuremath{{\Gamma}.{A}} \to \Gamma$ and $\ensuremath{\ensuremath{\mathsf{te}}}_A \in \ensuremath{\ensuremath{\mathsf{Tm}}({\ensuremath{{\Gamma}.{A}}})}$ satisfying the conditions of Definition~\ref{def:cwf.fiore}, item \ref{item:cwf.fiber-rep}. 
  
 \footnote{Note that in \cite{awodey_natural_published}, Awodey takes \emph{representable map} to mean what we call a CwF structure, i.e.\ to include choices of representations of the fibers; cf.\ \cite[\textsection 1.1, Algebraic character]{awodey_natural_published}.}

  In other words, a representable map of presheaves on ${\mathcal{C}}$ is just like a CwF structure, except that the representations are merely assumed to exist, not included as chosen data.

<!-- problem-12: body_md -->
\label{problem:transfer_cwf}
 Given a weak equivalence $F : {\mathcal{C}} \to {\mathcal{D}}$, where ${\mathcal{D}}$ is univalent, to construct a map $\ensuremath{\mathsf{cwf}}({\mathcal{C}}) \to \ensuremath{\mathsf{cwf}}({\mathcal{D}})$.
<!-- end problem-12 -->
<!-- construction-12: body_md -->
{construction:transfer_cwf}
  We construct a map $\ensuremath{\mathsf{relu}}(\ensuremath{\mathsf{y}}_{{\mathcal{C}}}) \to \ensuremath{\mathsf{relu}}(\ensuremath{\mathsf{y}}_{{\mathcal{D}}})$ as an instance of Construction~\ref{constr:universe.struct.on} and obtain the desired map by composition with the equivalence
  to CwF structures
  of Construction~\ref{construction:equiv_cwf_relu}.
  Consider the diagram
  \[
        \xymatrix@R=3pc@C=4pc{
                 {\mathcal{C}} \drtwocell<\omit>{\alpha} \ar[r]^-{\ensuremath{\mathsf{y}}_{{\mathcal{C}}}} \ar[d]_{F}
                        & \ensuremath{\ensuremath{\mathsf{PreShv}}({{\mathcal{C}}})} \ar@/_/[d]_{\mathcal{S}}
                 \\ {\mathcal{D}} \ar[r]_-{\ensuremath{\mathsf{y}}_{{\mathcal{D}}}}
                        & \ensuremath{\ensuremath{\mathsf{PreShv}}({{\mathcal{D}}})} \ar@/_/[u]^{\simeq}_{F^{\circ}}
        }
  \]
  Here, the functor $F^{\circ}$ given by precomposition with $F^{\mathrm{op}}$ is a weak equivalence between univalent categories, and hence a
  strong equivalence with inverse $\mathcal{S}$.
  The isomorphism $\alpha$ is constructed as follows:
  Note that fully faithful functors reflect isomorphisms; we apply this for the functor $F^{\circ}$ of precomposition with $F^{\mathrm{op}}$.
  It hence suffices to construct a natural isomorphism from  $\ensuremath{{\ensuremath{\mathsf{y}}_{{\mathcal{C}}}}\cdot{\ensuremath{{\mathcal{S}}\cdot{F^{\circ}}}}} \simeq \ensuremath{\mathsf{y}}_{{\mathcal{C}}}$
     to $\ensuremath{{F}\cdot{\ensuremath{{\ensuremath{\mathsf{y}}_{{\mathcal{D}}}}\cdot{F^{\circ}}}}}$.
   But this is an instance of a general isomorphism:
   indeed, for any functor $G : {\mathcal{A}} \to {\mathcal{X}}$, we have natural transformation
   from $\ensuremath{\mathsf{y}}_{{\mathcal{A}}}$ to $\ensuremath{{G}\cdot{\ensuremath{{\ensuremath{\mathsf{y}}_{{\mathcal{X}}}}\cdot{G^{\circ}}}}}$,
   and this natural transformation is an isomorphism when $G$ is fully faithful.
   This ends the construction of the natural isomorphism $\alpha$.
   The functor $\mathcal{S}$ preserves pullbacks since it is fully faithful and essentially surjective.
   The hypotheses of Problem~\ref{problem:universe.struct.on} are easily checked,
   hence Construction~\ref{constr:universe.struct.on} applies.
