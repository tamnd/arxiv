\label{def_fruncation_operator}
We define the normalization operator
\begin{align}\label{eq_normalization_operator}
 \overline{(\cdot)}\,\colon\mathbb{R}^{{n+1}}\to\mathbb{R}^{n+1}\: \text{ defined such that }\: \mathbf{u} \mapsto \overline{\mathbf{u}}	:= \tfrac{\mathbf{u}}{u_0}
\end{align}
and the ``fruncation'' operator 
\begin{align}\label{eq_reduction_operator}
   	(\cdot)_\#\colon \mathbb{R}^{n+1}\to\mathbb{R}^{n}\:\text{ defined such that }\:  \mathbf{u} \mapsto \mathbf{u}_\# := [\mathbf{u}_1^\top,\dots,\mathbf{u}_N^\top]^\top \in \mathbb{R}^{n}\: .
\end{align}
Further, repeated application of the fruncation operator $(\cdot)_\#$ to the moment vector removes moments associated with the (currently) lowest degree basis functions, e.g. $({(\mathbf{u})_{\#}})_{\#}=[\mathbf{u}_2^\top,\dots,\mathbf{u}_N^\top]^\top$.

First, consider weak convergence to a single Dirac distribution $\delta_{{\mathbf{v}}^*}$ at an arbitrary point $\mathbf  v^*\in [-1,1]$ .
We show that the Maxwell-Boltzmann entropy of a Dirac sequence $f_n$ converging weakly to $\delta_{{\mathbf{v}}^*}$ diverges to infinity.
Consider
\begin{align}
	f_n({\mathbf{v}}) =n\mathds{1}_{B_{n}^{{\mathbf{v}}^*}}({\mathbf{v}}), \qquad n\in\mathbb{N}
\end{align}
where $B_{n}^{{\mathbf{v}}^*}\subset [-1,1]$ is a subdomain of measure $\frac{1}{n}$ and $\mathds{1}({\mathbf{v}})$ is the indicator function. Then $f_n$ is a Dirac sequence.
        
              
                                                      
            
    
              
                                                                                                                                                              
            
Moreover,
                              
                          
               
                                                                          
             
                                                                                            
\begin{align}
	\int_{[-1,1]}{f_n({\mathbf{v}})\log{f_n({\mathbf{v}})}-f_n({\mathbf{v}})}\;\mathrm{d} {\mathbf{v}} &=\int_{[-1,1]}n\mathds{1}_{B_{n}^{{\mathbf{v}}^*}}({\mathbf{v}})\log{n\mathds{1}_{B_{n}^{{\mathbf{v}}^*}}({\mathbf{v}})}-n\mathds{1}_{B_{n}^{{\mathbf{v}}^*}}({\mathbf{v}})\;\mathrm{d} {\mathbf{v}} \\

	&=\log{n}-1.
\end{align}
This term diverges to infinity as $n\rightarrow\infty$, which concludes the proof.

Then, the proof works analogously up to Eq.~\eqref{eq_helper1}, where we consider the sum of the integrals of the components of the linear combination. Then the intregrals over the corresponding sets
 can be considered individually and the remainder of the proof follows in analogy to the case $p=1$.

\left< {\mathbf{m}}\exp( {\boldsymbol{\alpha}}_{\overline{{\mathbf{u}}}}^\gamma \cdot {\mathbf{m}}) \right> + \gamma\left[0,({\boldsymbol{\alpha}}_{\overline{{\mathbf{u}}}}^\gamma)_\#^\top\right]^\top,

We next show that $\hat{h}^\gamma\colon \mathbb{R}^{n}\to\mathbb{R}$ is strictly convex. By the definition of $\hat{h}^\gamma$ and the formulation of $\hat{\phi}^\gamma$ in \eqref{eq_reduced_obj_reformulation},
\begin{equation}\label{eq_hat_h}
\begin{alignedat}{2}    \hat{h}^\gamma(\overline{\mathbf{u}}_{\#}) 
                                                                                                                                                                                                                               
    &= \sup_{{{\boldsymbol{\beta}}}\in\mathbb{R}^{n}} 
    \left[
      -\hat{\phi}^\gamma({\boldsymbol{\beta}};{\overline{\mathbf{u}}_{\#}}) \right]
      =\sup_{{{\boldsymbol{\beta}}}\in\mathbb{R}^{n}} 
    \left[
     {{\boldsymbol{\beta}}} \cdot \overline{\mathbf{u}}_{\#} - \bigg(\hat{\phi}^\gamma({\boldsymbol{\beta}};{\overline{\mathbf{u}}_{\#}})  +{{\boldsymbol{\beta}}} \cdot \overline{\mathbf{u}}_{\#}\bigg)\right] \:.
\end{alignedat}
\end{equation}
                                  
                                                              
                                                                                                                                                                                                                               
                                                
                                                                                                                                                                                                         
                 
                
Thus $\hat{h}^\gamma$ is the Legendre dual of the function ${\boldsymbol{\beta}} \mapsto \hat{\phi}^\gamma({\boldsymbol{\beta}};{\overline{\mathbf{u}}_{\#}})  +{{\boldsymbol{\beta}}} \cdot \overline{\mathbf{u}}_{\#}$,
                                                                                                                                          
which is strictly convex by Lemma~\ref{lem_reduced_dual_obj}. Therefore, $\hat{h}^\gamma$ is also strictly convex.
As for the gradient of $\hat{h}^\gamma$, since $\hat{h}^\gamma(\mathbf{w}):= -\hat{\phi}^\gamma({{\boldsymbol{\beta}}}_{\mathbf{w}}^\gamma;\mathbf{w})$ and $-\nabla_{\mathbf{w}} \hat{\phi}^\gamma( {\boldsymbol{\beta}};\mathbf{w}) = {\boldsymbol{\beta}}$,
\begin{equation}\label{eq_grad_h_hat}
\nabla_{\mathbf{w}}\hat{h}^\gamma(\mathbf{w}) 
                                                                                                                                                                                     
={{\boldsymbol{\beta}}}_{\mathbf{w}}^\gamma 
-\nabla_{\mathbf{w}}{{\boldsymbol{\beta}}}_{\mathbf{w}}^\gamma \left(\nabla_{{\boldsymbol{\beta}}_{\mathbf{w}}^\gamma}\hat{\phi}^\gamma({\boldsymbol{\beta}}_{\mathbf{w}}^\gamma;\mathbf{w})\right).
\end{equation}
Since ${\boldsymbol{\beta}}_{\mathbf{w}}^\gamma$ minimizes $\hat{\phi}^\gamma$, the last term in \eqref{eq_grad_h_hat} vanishes, which proves the claim.


To derive the relation between $h^\gamma$ and $\hat{h}^\gamma$, see Definition~\ref{def_entropy} and~\ref{def_reduced_values}, we start with
\begin{equation}\label{eq_h}

    = - \left< \exp(\bm\alpha_{{\mathbf{u}}}^\gamma \cdot \mathbf{m}) \right> + \bm\alpha_{{\mathbf{u}}}^\gamma\cdot \mathbf{u} - \frac{u_0\gamma}{2} \|(\bm\alpha_{{\mathbf{u}}}^\gamma)_{\#}\|^2,
\end{equation}
see Eq.~\eqref{eq_entropy_phi_def}. Plugging $\bm\alpha_{{\mathbf{u}}}^\gamma = [\alpha_{\overline{\mathbf{u}},0}^\gamma + \frac{\log \,u_0}{m_0}, (\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma)_{\#}^\top]^\top$, where $\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma = [\vartheta({{\boldsymbol{\beta}}}_{{\overline{\mathbf{u}}_{\#}}}^\gamma),({{\boldsymbol{\beta}}}_{{\overline{\mathbf{u}}_{\#}}}^\gamma)^\top]^\top$, into the above equation then gives
\begin{equation}\label{eq_h_extension}
\begin{alignedat}{2}
    h^\gamma(\mathbf{u})  &= - \left< \exp\left(\left[\alpha_{\overline{\mathbf{u}},0}^\gamma + \frac{\log \,u_0}{m_0}, (\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma)_{\#}^\top\right]^\top \cdot \mathbf{m}\right) \right> + \,\left[\alpha_{\overline{\mathbf{u}},0}^\gamma + \frac{\log \,u_0}{m_0}, (\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma)_{\#}^\top\right]^\top\cdot \overline{\mathbf{u}} u_0 - \frac{u_0\gamma}{2} \|(\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma)_{\#}\|^2\\
    &= - u_0 \left< \exp(\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma \cdot \mathbf{m}) \right> + \frac{u_0}{m_0}\log \,u_0 + u_0\,\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma\cdot \overline{\mathbf{u}} - \frac{u_0\gamma}{2} \|(\bm\alpha_{\overline{{\mathbf{u}}}}^\gamma)_{\#}\|^2\\

\end{alignedat}
\end{equation}
where the last equality uses the first equality in \eqref{eq_hat_h} and the definition of $\vartheta$ in \eqref{eq_ref_alpha_reduction}.
Finally, $h^\gamma(\overline{\mathbf{u}})= \hat{h}^\gamma(\overline{\mathbf{u}}_{\#})$ is a direct consequence of the relation between $h^\gamma$ and $\hat{h}^\gamma$ given in \eqref{eq_h_extension}.

For entropy dissipation, we first show that the integrability condition holds for the entropy/entropy-flux pair $h^\gamma$ and $\mathbf{j}^\gamma$ with the flux term 
$\mathbf{F}(\mathbf{u})=\left< \mathbf{v}\otimes\mathbf{m}f^\gamma_{\mathbf{u}} \right>$.
For $i=1,2,3$, let $j_{i}^\gamma:=\left< v_i \eta(f^\gamma_{\mathbf{u}}) \right>$ and $F_i(\mathbf{u}):=\left< v_i\,\mathbf{m}f^\gamma_{\mathbf{u}} \right>$, then 
                                                     
                      
                                                                          
                                                                        
                                                                                                                                                             
                                                                                                                
                                                      
                 
                
\begin{equation}\label{eq_entropy_entropyflux_pair}
	\nabla_\mathbf{u} j^\gamma_{i}(\mathbf{u})= \left< v_i\nabla_\mathbf{u} \eta(f^\gamma_{\mathbf{u}}) \right>
	= \left< v_i\eta^\prime(f^\gamma_{\mathbf{u}})\nabla_\mathbf{u} f^\gamma_{\mathbf{u}} \right> 
	= \left< v_i\eta^\prime(\eta_*^\prime(\mathbf{g}^\gamma_{\mathbf{u}}\cdot\mathbf{m}))\nabla_\mathbf{u} f^\gamma_{\mathbf{u}} \right> 
	= \mathbf{g}^\gamma_{\mathbf{u}}\cdot\left< v_i\mathbf{m}\,\nabla_\mathbf{u} f^\gamma_{\mathbf{u}} \right>
=\nabla_\mathbf{u} h^\gamma(\mathbf{u})\cdot \nabla_\mathbf{u} F_i(\mathbf{u}),
\end{equation}
where we have used the chain rule, the definition of $f^\gamma_{\mathbf{u}}$ in Eq.~\eqref{eq_ansatzes}, the Legendre duality, and the fact that $\mathbf{g}^\gamma_{\mathbf{u}}= \nabla_\mathbf{u} h^\gamma(\mathbf{u})$.
To derive the entropy dissipation law, we multiply the moment system from the left with $\nabla_\mathbf{u} h^\gamma(\mathbf{u})$ and obtain
\begin{equation}
	\nabla_\mathbf{u} h^\gamma(\mathbf{u})\cdot\partial_t \mathbf{u} + \nabla_\mathbf{u} h^\gamma(\mathbf{u})\cdot\sum_{i=1}^3\partial_{{x}_i}F_i(\mathbf{u})= \nabla_\mathbf{u} h^\gamma(\mathbf{u})\cdot\left< \mathbf{m} Q( {f}_{\mathbf{u}}^\gamma ) \right>\:.
\end{equation}
From the reverse chain rule and Eq.~\eqref{eq_entropy_entropyflux_pair}, this leads to 
\begin{equation}
	\partial_t h^\gamma(\mathbf{u}) + \nabla_\mathbf{x}\cdot \bm j^\gamma(\mathbf{u})= \nabla_\mathbf{u} h^\gamma(\mathbf{u})\cdot\left< \mathbf{m} Q( {f}_{\mathbf{u}}^\gamma) \right>.
\end{equation}
Entropy dissipation requires non-positivity of right-hand side term, which is given by
\begin{equation}
\nabla_\mathbf{u} h^\gamma(\mathbf{u})\cdot\left< \mathbf{m} Q( {f}_{\mathbf{u}}^\gamma ) \right>
= \left< \mathbf{g}^\gamma_{\mathbf{u}}\cdot\mathbf{m} Q( {f}_{\mathbf{u}}^\gamma ) \right>
= \left< \eta^\prime(\eta_*^\prime(\mathbf{g}^\gamma_{\mathbf{u}}\cdot\mathbf{m})) Q( {f}_{\mathbf{u}}^\gamma ) \right>
=\left< \eta^\prime({f}_{\mathbf{u}}^\gamma) Q( {f}_{\mathbf{u}}^\gamma ) \right>\leq0,
\end{equation}
where the equalities follow from similar properties used in Eq.~\eqref{eq_entropy_entropyflux_pair} in the reversed order, and the inequality is from the entropy dissipation law in Eq.~\eqref{eq_kinetic_entropy_dissipation}.



We next show that the closed moment system is hyperbolic. Note that the hyperbolicity follows from the entropy dissipation property of the moment system.
Consider again the flux function of the moment system
\begin{align}
	\mathbf{F}(\mathbf{u})=\left< \mathbf{v}\otimes\mathbf{m}f^\gamma_{\mathbf{u}} \right> =\left< \mathbf{v}\otimes\mathbf{m}\,\eta'_*\left(\mathbf{g}_{\mathbf{u}}^\gamma\cdot\mathbf{m}\right) \right>.
\end{align}
We define $j_{i,*}(\mathbf{g})=\left< v_i\eta_*(\mathbf{g}\cdot \mathbf{m}) \right>$, $i=1, 2, 3$, such that
$\nabla_{\mathbf{g}}j_{i,*}(\mathbf{g}^\gamma_{\mathbf{u}})=F_i(\mathbf{u})$, and denote
\begin{align}
	J_i =  {\nabla_\mathbf{g}^2}j_{i,*}(\mathbf{g}^\gamma_{\mathbf{u}}),\quad i = 1,2,3,\quad\text{and}\quad
    K = \nabla_\mathbf{u}^2 h^\gamma(\mathbf{u}) = \nabla_\mathbf{u} \mathbf{g}^\gamma_{\mathbf{u}},
\end{align}
where $K$ is symmetric positive definite by the strict convexity of $h^\gamma(\mathbf{u})$.
By using chain rule, the flux term in Eq.~\eqref{eq_hyper_moment_sys} can be written as
\begin{equation}
    \nabla_{\mathbf{x}}\left< \mathbf{v}\otimes\mathbf{m}f^\gamma_{\mathbf{u}} \right> = \sum_{i=1}^3 \partial_{x_i} F_i(\mathbf{u}) = \sum_{i=1}^3 \partial_{x_i} (\nabla_{\mathbf{g}}j_{i,*}(\mathbf{g}^\gamma_{\mathbf{u}})) = \sum_{i=1}^3 {\nabla_\mathbf{g}^2}j_{i,*}(\mathbf{g}^\gamma_{\mathbf{u}}) \cdot \nabla_\mathbf{u} \mathbf{g}^\gamma_{\mathbf{u}} \cdot \partial_{x_i}\mathbf{u}
    = \sum_{i=1}^3  J_i K \partial_{x_i}\mathbf{u}\:.
\end{equation}
                                                                                                                              
                                                                                                                                     
                            
                                       
             
                                                                                                                                                                                                                            
                                                                                                                                                                                                        
The symmetric positive definite matrix $K$ then symmetrizes \cite{friedrichs1971systems} the closed moment system \eqref{eq_hyper_moment_sys}; that is, after multiplication on the left by $K$, 
\begin{equation}
    K\partial_t\mathbf{u} +  \sum_{i=1}^3 K J_i K \partial_{x_i}\mathbf{u}\: = K \left< \mathbf{m} Q( {f}_{\mathbf{u}}^\gamma) \right>,
\end{equation}
where the matrix $K J_i K$ is symmetric for each $i \in \{1,2,3\}$.
