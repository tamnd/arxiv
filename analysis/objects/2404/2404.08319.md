<!-- thm-1: body_md -->
\label{t:Minkowski_Radon}
Let $K\subset\mathbb{R}^n$ be a convex body with non-empty interior and let $H$ be a hyperplane. If $K$ has centroid at the origin then
\begin{equation}\label{e:Minkowski_Radon}
\frac{\mathrm{vol}_1\bigl(K^-|H^\perp\bigr)}{\mathrm{vol}_1\bigl(K|H^\perp\bigr)}\geq \frac{1}{n+1}.
\end{equation}
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\label{t:Frad_Makai_Martini}
Let $K\subset\mathbb{R}^n$ be a convex body with non-empty interior, let $H$ be a hyperplane and let $f:[a,b]\longrightarrow \mathbb{R}_{\geq 0}$ be the function given by $f(t) = \mathrm{vol}_{n-1}\bigl({K \cap (tu + H)}\bigr)$.
If $K$ has centroid at the origin then
\begin{equation}\label{e:Frad_Makai_Martini}
\frac{f(0)}{\|f\|_{\infty}}\geq \left(\frac{n}{n+1}\right)^{n-1}.
\end{equation}
<!-- end thm-2 -->
<!-- thm-3: body_md -->
\label{t:Grunbaum_pgeq0}
Let $K\subset\mathbb{R}^n$ be a compact set with non-empty interior and with centroid at the origin. Let $H$ be a hyperplane such that the function $f:H^{\bot}\longrightarrow\mathbb{R}_{\geq0}$ given by $f(x)=\mathrm{vol}_{n-1}\bigl(K\cap(x+H)\bigr)$ is $p$-concave, for some $p\in(0,\infty)$. Then
\begin{equation}\label{e:Grunbaum_p>0}
\frac{\mathrm{vol}(K^{-})}{\mathrm{vol}(K)}\geq\left(\frac{p+1}{2p+1}\right)^{(p+1)/p}.
\end{equation}
The inequality is sharp.
<!-- end thm-3 -->
<!-- theorem-1: body_md -->
\label{t:p_concave_r_centroid}
Let $r\in[0,\infty)$ and let $K\subset\mathbb{R}^n$ be a compact set with non-empty interior having the point $\mathrm{g}_{r}\cdot u$, with respect to some direction $u\in\mathbb{S}^{n-1}$, at the origin. Let $H=\{x \in \mathbb{R}^{n}: \left\langle x,u\right\rangle=0\}$ be the hyperplane with normal vector $u$ and assume that the function $f:H^{\bot}\longrightarrow\mathbb{R}_{\geq0}$ given by $f(x)=\mathrm{vol}_{n-1}\bigl(K\cap(x+H)\bigr)$ is $p$-concave, for some $p\in(0,\infty)$. If $r\geq1$ then
\begin{equation*}
\frac{\mathrm{vol}(K^{-})}{\mathrm{vol}(K)}\geq\left(\frac{p+1}{2p+r}\right)^{(p+1)/p},
\end{equation*}
whereas if $0\leq r\leq1$ then
\begin{equation*}
\frac{\mathrm{vol}(K^{-})}{\mathrm{vol}(K)}\geq\left(\frac{p+r}{2p+r}\right)^{(p+1)/p}.
\end{equation*}

<!-- theorem-2: body_md -->
\label{t:functional_centroid_general_alpha_beta}
Let $h:[a,b]\longrightarrow[0,\infty)$ be a non-negative concave function, and let $\alpha,\beta>0$.
If $\beta\leq\alpha$ then
\begin{equation}\label{e:int_ineq_beta<alpha}
\frac{\int_{\mathrm{g}_{\alpha}(h)}^bh^\beta(t)\,\mathrm{d} t} {\int_{a}^bh^\beta(t)\,\mathrm{d} t} \geq\left(\frac{\beta+1}{\alpha+2}\right)^{\beta+1},
\end{equation}
whereas if $\alpha\leq\beta$ then
\begin{equation}\label{e:int_ineq_alpha<beta}
\frac{\int_{\mathrm{g}_{\alpha}(h)}^bh^\beta(t)\,\mathrm{d} t} {\int_{a}^bh^\beta(t)\,\mathrm{d} t} \geq\left(\frac{\alpha+1}{\alpha+2}\right)^{\beta+1}.
\end{equation}

<!-- remark-1: body_md -->
We observe that Theorem \ref{t:p_concave_r_centroid} is directly obtained from the previous result by just taking
$h=f^{p}$, $\beta=1/p$ and $\alpha=r\beta$ (where the case $r=0$ is derived when doing $\alpha\to0^+$).
Moreover, Theorem \ref{t:functional_centroid_general_alpha_beta} can be shown from Theorem \ref{t:p_concave_r_centroid} by just considering the set of revolution $K$ associated to the radius function $r=(1/\kappa_{n-1})f^{1/(n-1)}$, where $\kappa_{n-1}$ is the $(n-1)$-dimensional volume of the Euclidean unit ball in $\mathbb{R}^{n-1}$, $f=h^\beta$, $p=1/\beta$ and $r=\alpha/\beta$.
Therefore, in fact, both results (Theorems \ref{t:p_concave_r_centroid} and \ref{t:functional_centroid_general_alpha_beta}) are equivalent.

<!-- proposition-1: body_md -->
\label{p:midpoint}
Let $K\subset\mathbb{R}^n$ be a compact set with non-empty interior and with midpoint, with respect to some direction $u\in\mathbb{S}^{n-1}$, at the origin. Let $H=\{x \in \mathbb{R}^{n}: \left\langle x,u\right\rangle=0\}$ be the hyperplane with normal vector $u$ and assume that the function $f:H^{\bot}\longrightarrow\mathbb{R}_{\geq0}$ given by $f(x)=\mathrm{vol}_{n-1}\bigl(K\cap(x+H)\bigr)$ is $p$-concave, for some $p\in(0,\infty)$. Then
\begin{equation*}
\frac{\mathrm{vol}(K^{-})}{\mathrm{vol}(K)}\geq\left(\frac{1}{2}\right)^{(p+1)/p}.
\end{equation*}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
In the proof of \cite[Theorem~1.1]{MSYN} it is shown that
there exists a $p$-affine function $g_p:[-\gamma,\delta]\longrightarrow\mathbb{R}_{\geq0}$ given by
$g_p(t)=c(t+\gamma)^{1/p}$, for some $\gamma,\delta,c>0$,
such that $g_p(0)=f(0)$,
\begin{equation*}\label{e:eq_integ_f_g_p>0}
\int_{-\gamma}^0 g_p(t)\,\mathrm{d} t=\int_a^0 f(t)\,\mathrm{d} t \quad \text{and} \quad
\int_0^\delta g_p(t)\,\mathrm{d} t=\int_0^b f(t)\,\mathrm{d} t,
\end{equation*}
and further that $-\gamma\leq a<0<\delta\leq b$.

Here, since $K$ has its midpoint (w.r.t. $u$) at the origin, we have that $a=-b$ and then we get $-\gamma+\delta\leq0$. Thus
\[\frac{\mathrm{vol}(K^-)}{\mathrm{vol}(K)}=\frac{\int_{-\gamma}^0 g_p(t)\,\mathrm{d} t}{\int_{-\gamma}^\delta g_p(t)\,\mathrm{d} t}
\geq\frac{\int_{-\gamma}^{(-\gamma+\delta)/2} g_p(t)\,\mathrm{d} t}{\int_{-\gamma}^\delta g_p(t)\,\mathrm{d} t}
=\left(\frac{1}{2}\right)^{(p+1)/p},
\]
as desired.

<!-- thm-4: body_md -->
\label{t:BBL}
Let $\lambda\in(0,1)$. Let $-1/n\leq p\leq\infty$ and let $f,g,h:\mathbb{R}^n\longrightarrow\mathbb{R}_{\geq0}$ be measurable
functions with positive integrals such that
\begin{equation*}
h((1-\lambda)x + \lambda y)\geq \bigl((1-\lambda)f(x)^p+\lambda g(y)^p\bigr)^{1/p}
\end{equation*}
for all $x,y\in\mathbb{R}^n$ with $f(x)g(y)>0$. Then
\begin{equation}\label{e:BBL}
\int_{\mathbb{R}^n}h(x)\,\mathrm{d} x\geq
\left[(1-\lambda)\left(\int_{\mathbb{R}^n}f(x)\,\mathrm{d} x\right)^q+\lambda\left(\int_{\mathbb{R}^n}g(x)\,\mathrm{d} x\right)^q\right]^{1/q},
\end{equation}
where $q=p/(np+1)$.

<!-- corollary-1: body_md -->
\label{c:first_parcial_result}
Let $r\in(0,\infty)$ and let $K\subset\mathbb{R}^n$ be a compact set with non-empty interior having the point $\mathrm{g}_{r}\cdot u$, with respect to some direction $u\in\mathbb{S}^{n-1}$, at the origin. Let $H=\{x \in \mathbb{R}^{n}: \left\langle x,u\right\rangle=0\}$ be the hyperplane with normal vector $u$ and assume that the function $f:\mathbb{R}\longrightarrow\mathbb{R}_{\geq0}$ given by $f(t)=\mathrm{vol}_{n-1}\bigl(K\cap(tu+H)\bigr)$ is $p$-concave, for some $p\in(0,\infty)$. Then
\begin{equation}\label{e:first_parcial_result}
\frac{\mathrm{vol}(K^{-})}{\mathrm{vol}(K)} \geq \left(\frac{p}{2p + r}\right)^{(p+1)/p}.
\end{equation}

<!-- proposition-2: body_md -->
\label{p:first_parcial_result}
Let $K \subset \mathbb{R}^n$ be a convex body. Let $g : K \longrightarrow \mathbb{R}_{\geq 0}$ be a concave function and let $f : K \longrightarrow \mathbb{R}_{\geq 0}$ be a $p$-concave function, with $p>0$. Then
\begin{equation*}
g\left(\frac{\int_{K} x f(x) \,\mathrm{d} x}{\int_{K} f(x) \,\mathrm{d} x}\right) \geq \left(\frac{p}{(n + 1)p + 1}\right)\|g\|_{\infty}.
\end{equation*}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Let $\mu$ be the probability measure whose density function is given by
\[
\mathrm{d}\mu(x) = \frac{f(x)}{\int_K f(x) \,\mathrm{d} x}\mathrm{d} x.
\]
Since $g$ is concave, using Jensen's inequality we get that
\begin{equation*}
\begin{split}
    g\left(\frac{\int_{K} x f(x) \,\mathrm{d} x}{\int_{K} f(x) \,\mathrm{d} x}\right) = g\left(\int_K x \, \mathrm{d}\mu(x)\right) 
    &\geq \int_K g(x) \,\mathrm{d}\mu(x)\\ 
    &= \int_0^{\|g\|_{\infty}}\mu\bigl(\{x \in K : g(x) \geq t\}\bigr)\,\mathrm{d} t,
\end{split}
\end{equation*}
where in the last identity we have used Fubini's theorem. Now, since the density of $\mu$, with respect to the Lebesgue measure, is $p$-concave,  from the Borell-Brascamp-Lieb inequality \eqref{e:BBL} we have that
the function $\varphi(t) := \mu\bigl(\{x \in K : g(x) \geq t\}\bigr)$ is $\bigl(p/(np + 1)\bigr)$-concave. 
Indeed, it is enough to apply Theorem \ref{t:BBL} with the functions 
\[
\frac{f\cdot\chi_{_{\{x\in K\,:\, g(x)\geq t_1\}}}}{\int_K f(x) \,\mathrm{d} x}, \quad \frac{f\cdot\chi_{_{\{x\in K\,:\, g(x)\geq t_2\}}}}{\int_K f(x) \,\mathrm{d} x} \quad \text{ and } \quad \frac{f\cdot\chi_{_{\{x\in K\,:\, g(x)\geq (1-\lambda)t_1+\lambda t_2\}}}}{\int_K f(x) \,\mathrm{d} x}
\] 
for any $t_1,t_2\in\mathbb{R}$, where $\chi_{_M}$ denotes the characteristic function of the set $M$, one easily obtains the desired concavity of $\varphi$.
Hence, and taking into account that $\varphi(0)=1$ and $\varphi\bigl(\|g\|_{\infty}\bigr)\geq0$, we may assure that $\varphi(t)^{p/(np + 1)} \geq (1 - t/\|g\|_{\infty})$ for all $t \in [0,\|g\|_{\infty}]$. So, by integrating we obtain that
\begin{equation*}
\begin{split}
\int_0^{\|g\|_{\infty}}\mu\bigl(\{x \in K : g(x) \geq t\}\bigr)\,\mathrm{d} t 
&\geq \int_0^{\|g\|_{\infty}}(1 - t/\|g\|_{\infty})^{(np + 1)/p}\,\mathrm{d} t\\
&= \left(\frac{p}{(n + 1)p + 1}\right)\|g\|_{\infty},
\end{split}
\end{equation*}
from where the result immediately follows.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
Denoting by $[a,b]$ the support of $f$, let $\bar{f},g:[a,b]\longrightarrow\mathbb{R}_{\geq0}$ be the functions given by
\[
g(t)=\mathrm{vol}\bigl(K \cap (tu + H^{-})\bigr)^{p/(p+1)}=\left(\int_a^t f(s)\,\mathrm{d} s\right)^{p/(p+1)}
\] 
and 
\[
\bar{f}(t)=\mathrm{vol}_{n-1}\bigl(K\cap(tu+H)\bigr)^r=f(t)^r.
\]
By hypothesis, it is clear that $\bar{f}$ is $(p/r)$-concave, whereas from the Borell-Brascamp-Lieb inequality \eqref{e:BBL} we have that $g$ is concave on $[a,b]$. So, from Proposition \ref{p:first_parcial_result} applied to the functions $\bar{f}$ and $g$, and taking into account that $\mathrm{g}_{r}=0$, we get that
\begin{equation*}
\begin{split}
\mathrm{vol}(K^{-}) = g\left(\frac{\int_a^b t \bar{f}(t) \,\mathrm{d} t}{\int_a^b \bar{f}(t)\,\mathrm{d} t}\right)^{(p+1)/p} 
&\geq \left(\frac{p}{2p + r}\right)^{(p+1)/p}\|g\|_{\infty}^{(p+1)/p}\\
&= \left(\frac{p}{2p + r}\right)^{(p+1)/p}\mathrm{vol}(K),
\end{split}
\end{equation*}
as desired.
