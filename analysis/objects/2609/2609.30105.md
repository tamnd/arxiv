<!-- definition-1: body_md -->
\label{def:cert-bdd}
Let $m\ge2$ be even and $B_m>0$. A distribution $P$ on $\mathbb R^d$ with mean
$\mu$ is \emph{$(B_m,m)$-certifiably bounded} if there are polynomials
$q_1,\ldots,q_N$ of degree at most $m/2$ such that
\[
 B_m^m\|v\|_2^m-\mathbb E_{X\sim P}\langle X-\mu,v\rangle^m
 =\sum_{j=1}^N q_j(v)^2.
\]
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:logconcave}
A distribution is \emph{log-concave} if, on the affine span of its
support, it has a density $p(x)=e^{-V(x)}$ for a convex function
$V$ taking values in $\mathbb R\cup\{+\infty\}$.
For a positive definite matrix $B$, it is \emph{$B$-strongly
log-concave} if $V(x)-x^\top Bx/2$ is convex.
When $B=\alpha I_d$, we say \emph{$\alpha$-strongly log-concave}.
<!-- end definition-2 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Every isotropic log-concave distribution $P$ on $\mathbb R^d$ is
$(Cm,m)$-certifiably bounded for all even $m\ge2$, where $C>0$
is a universal constant.
<!-- end theorem-1 -->
<!-- definition-3: body_md -->
\label{def:cert-hyper}
Let $m\ge2$ be an even integer and $B_m>0$. A distribution $P$ on $\mathbb R^d$
with mean $\mu$ and covariance $\Sigma$ is \emph{$(B_m,m)$-hypercontractive}
if the polynomial
\begin{align*}
 B_m^m(v^\top\Sigma v)^{m/2}
 -\mathbb E_{X\sim P}\langle X-\mu,v\rangle^m
\end{align*}
is nonnegative on $\mathbb R^d$, and \emph{$(B_m,m)$-certifiably hypercontractive}
if it is a sum of squares.
<!-- end definition-3 -->
<!-- corollary-1: body_md -->
\label{cor:hyper}
Every log-concave distribution is $(Cm,m)$-certifiably hypercontractive for
all even $m\ge2$, with the same universal constant $C$ as in \Cref{thm:main}.
<!-- end corollary-1 -->
<!-- corollary-2: body_md -->
\label{cor:robust-mean}
There exist universal constants $0<c,c'<1/2$ and $C_0>0$ such that
the following holds. Let $P$ be a log-concave distribution on $\mathbb R^d$
with mean $\mu$ and covariance $\Sigma\succ0$.
Let $0<\varepsilon<c$ and let $m\ge4$ be even, with
$m^2\varepsilon^{1-2/m}\le c'$.
Given an $\varepsilon$-corrupted sample $S$ of size
$n\ge(md/\varepsilon)^{C_0m}$ from $P$, there is an algorithm that takes
$S$, $m$, and $\varepsilon$ as input, runs in time $(nd)^{\mathrm{poly}(m)}$,
and returns estimates $\widehat\mu$ and $\widehat\Sigma$ satisfying,
simultaneously with probability at least $0.9$:
\begin{itemizec}
 \item Mean estimation in Mahalanobis norm:
 $\|\Sigma^{-1/2}(\widehat\mu-\mu)\|_2
 \lesssim m\varepsilon^{1-1/m}$.
 \item Covariance estimation in relative spectral norm:
 $(1-\delta)\Sigma\preceq\widehat\Sigma\preceq(1+\delta)\Sigma$
 for $\delta\lesssim m^2\varepsilon^{1-2/m}$.
\end{itemizec}
<!-- end corollary-2 -->
<!-- head-1: body_md -->
We write $\|\cdot\|_{\mathrm{op}}$ and $\|\cdot\|_{\mathrm{HS}}$ for the operator and
Hilbert--Schmidt norms, and $M\preceq N$ when $N-M$ is positive semidefinite.
The space $\mathbb S^d$ of symmetric matrices has inner product
$\langle M,N\rangle_{\mathrm{HS}}=\operatorname{tr}(MN)$. We use $\lesssim$ to suppress
universal constants. For the localization process, $\mathbb E_{p_t}$ integrates
under $p_t$, whereas $\mathbb E_W$ averages over the Brownian motion.
Expectations of polynomials are taken coefficientwise.
<!-- end head-1 -->
<!-- definition-4: body_md -->
\label{def:subgaussian}
Let $s>0$. A distribution $P$ on $\mathbb R^d$ with mean $\mu$ is
\emph{$s$-subgaussian} if, for every $r\ge1$ and $v\in\mathbb R^d$,
\[
 \bigl(\mathbb E_{X\sim P}|\langle X-\mu,v\rangle|^r\bigr)^{1/r}
 \le C_0s\sqrt r\,\|v\|_2,
\]
where $C_0\ge2$ is a fixed universal constant.
<!-- end definition-4 -->
<!-- theorem-2: body_md -->
\label{thm:subgaussian}
There is a universal constant $C_{\mathrm{sg}}>0$ such that every
$s$-subgaussian distribution is $(C_{\mathrm{sg}}s\sqrt m,m)$-certifiably
bounded for every even $m\ge2$.
<!-- end theorem-2 -->
<!-- corollary-3: body_md -->
\label{cor:strong-moments}
There is a universal constant $K\ge2$ such that, for every
$B$-strongly log-concave distribution $Q$ with $B\succ0$ and mean $\mu$,
and every integer $k\ge1$,
\begin{equation}
\label{eq:strong-moments}
 \sststile{2k}{v}\,\mathbb E_{X\sim Q}\langle X-\mu,v\rangle^{2k}
       \le(Kk)^k(v^\top B^{-1}v)^k.
\end{equation}
<!-- end corollary-3 -->
<!-- proof-1: body_md -->
Let $X\sim Q$ and set $Y=B^{1/2}(X-\mu)$. Then $Y$ is $1$-strongly
log-concave and hence $1$-subgaussian. By \Cref{thm:subgaussian},
\begin{align*}
 \mathbb E\langle X-\mu,v\rangle^{2k}
 &=\mathbb E\langle Y,B^{-1/2}v\rangle^{2k}\\*
 &\le(C_{\mathrm{sg}}\sqrt{2k})^{2k}\|B^{-1/2}v\|_2^{2k}\\*
 &=(2C_{\mathrm{sg}}^2k)^k(v^\top B^{-1}v)^k.
\end{align*}
The substitution $u=B^{-1/2}v$ is linear, so this is a degree-$2k$
sum-of-squares proof. Taking
$K=\max\{2,2C_{\mathrm{sg}}^2\}$ proves the corollary.
<!-- end proof-1 -->
<!-- theorem-3: body_md -->
\label{thm:letwin}
Let $Y\in\mathbb R^d$ be isotropic and log-concave. Then, for every symmetric
$M\in\mathbb R^{d\times d}$,
\begin{equation}
\label{eq:letwin}
 \operatorname{Var}(Y^\top MY)\le8\|M\|_{\mathrm{HS}}^2.
\end{equation}
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:transfer}
Let $k\ge2$ be an integer and $L_k\ge1$. Suppose that every
$1$-strongly log-concave distribution is $(L_k,2k)$-certifiably bounded.
Then every compactly supported isotropic log-concave density $p_0$ satisfies
\begin{equation}
\label{eq:transfer}
 \sststile{2k}{v}\,\mathbb E_{p_0}\langle X,v\rangle^{2k}
 \le 2^{2k-1}e^{4k(k-1)T}
       \left(L_k^{2k}T^{-k}+1\right)\|v\|_2^{2k}
\end{equation}
for all $T>0$. In particular, $p_0$ is
$(4\sqrt{ek}\,L_k,2k)$-certifiably bounded.
<!-- end theorem-4 -->
<!-- lemma-1: body_md -->
\label{lem:fourth}
Let $Y\in\mathbb R^d$ be isotropic and log-concave. Then
\begin{equation}
\label{eq:fourth}
 \sststile{4}{u}\,\operatorname{Var}(\langle Y,u\rangle^2)\le8\|u\|_2^4.
\end{equation}
In particular, the law of $Y$ is $(\sqrt{3},4)$-certifiably bounded.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
By \Cref{thm:letwin}, the quadratic form
$8\|M\|_{\mathrm{HS}}^2-\operatorname{Var}(Y^\top MY)$ is nonnegative in the entries of $M$,
and hence a sum of squares of linear forms. Substituting $M=uu^\top$
gives~\eqref{eq:fourth}.
The fourth-moment bound follows from isotropy:
\begin{align*}
 \sststile{4}{u}\quad \mathbb E\langle Y,u\rangle^4
 &=\operatorname{Var}(\langle Y,u\rangle^2)+\|u\|_2^4
 \le9\|u\|_2^4.\qedhere
\end{align*}
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lem:projection}
Let $Y\in\mathbb R^d$ be isotropic with finite fourth moments. Then
\begin{equation}
\label{eq:projection}
 \sststile{4}{u}\,\left\|\mathbb E[Y\langle Y,u\rangle^2]\right\|_2^2
                  \le\operatorname{Var}(\langle Y,u\rangle^2).
\end{equation}
<!-- end lemma-2 -->
<!-- corollary-4: body_md -->
\label{cor:third-contraction}
Let $Y\in\mathbb R^d$ be isotropic and log-concave. Then
\begin{equation}
\label{eq:third-contraction}
 \sststile{4}{u}\,\left\|\mathbb E[Y\langle Y,u\rangle^2]\right\|_2^2
                  \le8\|u\|_2^4.
\end{equation}
<!-- end corollary-4 -->
<!-- definition-5: body_md -->
\label{def:controlled-localization}
For a continuous adapted matrix process $C_t\succ0$, define
\begin{equation}
 \begin{aligned}
 c_0=0,\qquad &\mathrm{d} c_t=C_t\mu_t\,\mathrm{d} t+C_t^{1/2}\mathrm{d} W_t,\\
 B_0=0,\qquad &\mathrm{d} B_t=C_t\,\mathrm{d} t,
 \end{aligned}
 \label{eq:param-sde}
\end{equation}
where $\mu_t$ and $A_t$ are the mean and covariance of the density
\begin{equation}
 p_t(x)=\frac{e^{c_t^\top x-\frac12x^\top B_tx}p_0(x)}
 {\int e^{c_t^\top y-\frac12y^\top B_ty}p_0(y)\,\mathrm{d} y},
 \qquad \mu_t=\mathbb E_{p_t}X,\qquad A_t=\operatorname{Cov}_{p_t}(X).
 \label{eq:localized-density}
\end{equation}
<!-- end definition-5 -->
<!-- fact-1: body_md -->
\label{fact:covariance-evolution}
The covariance satisfies
\begin{equation}
 \mathrm{d} A_t=\sum_{i=1}^d H_{t,i}(A_t^{-1/2}\mathrm{d} W_t)_i-A_t\,\mathrm{d} t.
 \label{eq:covariance-sde}
\end{equation}
<!-- end fact-1 -->
<!-- lemma-3: body_md -->
\label{lem:fluctuations}
For each realized density $p_t$,
\begin{equation}
\label{eq:fluctuations}
 \sststile{4}{v}\,\frac{\mathrm{d}}{\mathrm{d} t}[v^\top A_{\cdot}v]_t
       \le8(v^\top A_tv)^2.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Apply \Cref{cor:third-contraction} to $Y$ and substitute $u=A_t^{1/2}v$.
This linear substitution preserves the proof degree, and
$\|u\|_2^4=(v^\top A_tv)^2$, so~\eqref{eq:directional-variation} gives the claim.
<!-- end proof-3 -->
<!-- lemma-4: body_md -->
\label{lem:potential}
For every $T\ge0$,
\begin{align}
\label{eq:potential}
 \sststile{2k}{v}\,G_T(v)=\mathbb E_W[(v^\top A_Tv)^k]
       \le e^{\beta T}\|v\|_2^{2k}.
\end{align}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Taking expectations in It\^o's formula for $(v^\top A_tv)^k$ and using
\eqref{eq:directional-sde}, we obtain
\begin{align}
 G_t'(v)
 &=-kG_t(v)+\binom{k}{2}\mathbb E_W\left[
    (v^\top A_tv)^{k-2}
    \frac{\mathrm{d}}{\mathrm{d} t}[v^\top A_{\cdot}v]_t\right]\nonumber\\*
 &\le(\beta-k)G_t(v)
 \le\beta G_t(v)
 \label{eq:potential-derivative}
\end{align}
for almost every $t$. The stochastic integral has expectation zero, and
the first inequality follows from \Cref{lem:fluctuations}. Multiplying
its degree-four certificate by $(v^\top A_tv)^{k-2}$ and averaging gives
a degree-$2k$ certificate for $G_t'\le\beta G_t$.

Since $G_0(v)=\|v\|_2^{2k}$, integration gives
\begin{equation}
 e^{\beta T}\|v\|_2^{2k}-G_T(v)
 =\int_0^T e^{\beta(T-t)}
       \bigl(\beta G_t(v)-G_t'(v)\bigr)\,\mathrm{d} t.
 \label{eq:potential-certificate}
\end{equation}
The right-hand side is a sum of squares of degree $2k$ by
\Cref{lem:averaging}, which proves the claim.
<!-- end proof-4 -->
<!-- lemma-5: body_md -->
\label{lem:means}
For every $T\ge0$,
\begin{align}
\label{eq:means}
 \sststile{2k}{v}\,\mathbb E_W\langle\mu_T,v\rangle^{2k}
       \le e^{\beta T}\|v\|_2^{2k}.
\end{align}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Let $F_t(v)=\mathbb E_W\langle\mu_t,v\rangle^{2k}$. By
\Cref{fact:mean-evolution}, $\mathrm{d}\mu_t=A_t^{1/2}\mathrm{d} W_t$.
Thus It\^o's formula and Young's inequality give
\begin{align}
 F_t'(v)
 &=k(2k-1)\mathbb E_W[\langle\mu_t,v\rangle^{2k-2}(v^\top A_tv)]\nonumber\\*
 &\le\alpha F_t(v)+(2k-1)G_t(v),
 \label{eq:means-derivative}
\end{align}
where $\alpha=(k-1)(2k-1)$. Since $F_0=0$, integrating and using
\Cref{lem:potential} yields
\begin{align}
 F_T(v)
 &\le(2k-1)\int_0^T e^{\alpha(T-t)}G_t(v)\,\mathrm{d} t\nonumber\\*
 &\le(2k-1)\int_0^T e^{\alpha(T-t)+\beta t}\,\mathrm{d} t\,\|v\|_2^{2k}\nonumber\\*
 &=\frac{2k-1}{\beta-\alpha}
       \bigl(e^{\beta T}-e^{\alpha T}\bigr)\|v\|_2^{2k}.
 \label{eq:means-comparison}
\end{align}
As $\beta-\alpha=(k-1)(2k+1)\ge2k-1$, this is at most
$e^{\beta T}\|v\|_2^{2k}$. Young's inequality and the integrating-factor
argument have degree-$2k$ certificates by
\Cref{fact:young,lem:integration}.

<!-- lemma-6: body_md -->
\label{lem:precision}
For every $T>0$,
\begin{align}
\label{eq:precision}
 \sststile{2k}{v}\,\mathbb E_W[(v^\top B_T^{-1}v)^k]
       \le T^{-k}e^{\beta T}\|v\|_2^{2k}.
\end{align}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
We first bound $v^\top B_T^{-1}v$ by
$T^{-2}\int_0^T v^\top A_tv\,\mathrm{d} t$ for each realization. Indeed, the
identity $B_T=\int_0^T A_t^{-1}\,\mathrm{d} t$ gives
\begin{equation}
 \int_0^T v^\top A_tv\,\mathrm{d} t-T^2v^\top B_T^{-1}v
 =\int_0^T
   \left\|A_t^{1/2}v-TA_t^{-1/2}B_T^{-1}v\right\|_2^2\,\mathrm{d} t
 \ge0.
 \label{eq:precision-square}
\end{equation}

\begin{align}
 \mathbb E_W[(v^\top B_T^{-1}v)^k]

 &\le T^{-k-1}\int_0^T G_t(v)\,\mathrm{d} t\nonumber\\*
 &\le T^{-k-1}\int_0^T e^{\beta t}\,\mathrm{d} t\,\|v\|_2^{2k}\nonumber\\*
 &\le T^{-k}e^{\beta T}\|v\|_2^{2k}.
 \label{eq:precision-comparison}
\end{align}

 \label{eq:transfer-centered-moments}
\end{align}

    L_k^{2k}\mathbb E_W[(v^\top B_T^{-1}v)^k]

 \label{eq:transfer-average}
\end{align}

 &\le[16e(k-1)]^kL_k^{2k}\\*
 &\le\bigl(4\sqrt{ek}\,L_k\bigr)^{2k},
\end{align*}

\end{align*}

<!-- lemma-7: body_md -->
\label{lem:averaging}
In a fixed number of variables, the cone of sum-of-squares polynomials of degree at most $2k$ is closed under coefficientwise limits. If $p_\omega(v)$ belongs to this cone for almost every $\omega$ and its coefficients are measurable and integrable with respect to a finite positive measure $\nu$, then
\[
 \int p_\omega(v)\,\mathrm{d}\nu(\omega)
\]
also belongs to the cone.
<!-- end lemma-7 -->
<!-- proof-10: body_md -->
We first prove closedness. Suppose $p_j\to p$ coefficientwise, where each
$p_j$ is a sum of squares of degree at most $2k$. Let $z(v)$ be the vector
of monomials of degree at most $k$, and choose $Q_j\succeq0$ such that
$p_j(v)=z(v)^\top Q_jz(v)$.

For a standard Gaussian vector $G$, the matrix
$H=\mathbb E[z(G)z(G)^\top]$ is positive definite, since the monomials in $z$ are
linearly independent in $L_2(G)$. Consequently,
\[
 \lambda_{\min}(H)\operatorname{tr} Q_j
 \le\operatorname{tr}(HQ_j)
 =\mathbb E p_j(G)\longrightarrow\mathbb E p(G).
\]
The matrices $Q_j$ therefore have bounded trace and admit a convergent
subsequence, with limit $Q\succeq0$. It follows that $p(v)=z(v)^\top Qz(v)$.

The cone is convex as well as closed, so the averaging assertion follows
by separation. Indeed, for every linear functional $\ell$ nonnegative on
the cone, coefficientwise integrability gives
\[
 \ell\!\left(\int p_\omega\,\mathrm{d}\nu(\omega)\right)
 =\int\ell(p_\omega)\,\mathrm{d}\nu(\omega)\ge0.\qedhere
\]

\label{lem:integration}

\[

\]
for almost every $t$, then
\[
 \sststile{2k}{v}\,F_T(v)\le e^{aT}F_0(v)
               +\int_0^T e^{a(T-t)}H_t(v)\,\mathrm{d} t.
\]

\]
By assumption, $aF_t+H_t-F_t'$ is a sum of squares of degree at most $2k$ for almost every $t$. Averaging with the positive weight $e^{a(T-t)}$ proves the claim by \Cref{lem:averaging}.

<!-- fact-6: body_md -->
\label{lem:density}
The density $p_t$ satisfies
\begin{equation}
 \mathrm{d} p_t(x)=p_t(x)\langle x-\mu_t,A_t^{-1/2}\mathrm{d} W_t\rangle.
 \label{eq:density-sde}
\end{equation}
For $t>0$, it is $B_t$-strongly log-concave.
<!-- end fact-6 -->
<!-- proof-12: body_md -->
By~\eqref{eq:param-sde}, $B_t$ has finite variation. Applying It\^o's
formula to the logarithm of the normalizing integral
in~\eqref{eq:localized-density} gives
\begin{align}
 &\mathrm{d}\log\int e^{c_t^\top y-\frac12y^\top B_ty}p_0(y)\,\mathrm{d} y\nonumber\\*
 &\qquad=\langle\mu_t,\mathrm{d} c_t\rangle
       -\frac12\mathbb E_{p_t}[X^\top A_t^{-1}X]\,\mathrm{d} t
       +\frac12\operatorname{tr}(A_tA_t^{-1})\,\mathrm{d} t\nonumber\\*
 &\qquad=\langle\mu_t,\mathrm{d} c_t\rangle
       -\frac12\mu_t^\top A_t^{-1}\mu_t\,\mathrm{d} t,\nonumber
\end{align}
where the second equality follows from the definition of the covariance.
For $x$ with $p_0(x)>0$, the density formula therefore gives
\begin{align}
 \mathrm{d}\log p_t(x)
 &=\langle x-\mu_t,\mathrm{d} c_t\rangle
   -\frac12\bigl(x^\top A_t^{-1}x-\mu_t^\top A_t^{-1}\mu_t\bigr)\,\mathrm{d} t\nonumber\\*
 &=\langle x-\mu_t,A_t^{-1/2}\mathrm{d} W_t\rangle
   -\frac12\|A_t^{-1/2}(x-\mu_t)\|_2^2\,\mathrm{d} t.
 \label{eq:log-density-sde}
\end{align}
Applying It\^o's formula to the exponential gives
\begin{align}
 \mathrm{d} p_t(x)
 &=p_t(x)\mathrm{d}\log p_t(x)
   +\frac12p_t(x)\|A_t^{-1/2}(x-\mu_t)\|_2^2\,\mathrm{d} t\nonumber\\*
 &=p_t(x)\langle x-\mu_t,A_t^{-1/2}\mathrm{d} W_t\rangle.\nonumber
\end{align}
If $p_0(x)=0$, then $p_t(x)=0$ for all $t$, so the identity still holds.
Finally, the density formula gives
\begin{align}
 e^{\frac12x^\top B_tx}p_t(x)&\propto e^{c_t^\top x}p_0(x).\nonumber
\end{align}
The right-hand side is log-concave, so $p_t$ is $B_t$-strongly log-concave.

<!-- lemma-9: body_md -->
\label{lem:truncation}
Let $X$ be an isotropic log-concave random vector in $\mathbb R^d$. There are
compactly supported isotropic log-concave random vectors $Y_R$ such that
\[
 \mathbb E f(Y_R)\longrightarrow\mathbb E f(X)\qquad(R\to\infty)
\]
for every polynomial $f:\mathbb R^d\to\mathbb R$.
<!-- end lemma-9 -->
<!-- proof-16: body_md -->
Let $X_R$ have the law of $X$ conditioned on $\|X\|_2\le R$, with mean
$a_R$ and covariance $\Sigma_R$. Since $X$ has moments of all
orders~\cite{Bor74}, dominated convergence gives
\begin{align*}
 \mathbb E f(X_R)
 &=\frac{\mathbb E[f(X)\mathbf1_{\{\|X\|_2\le R\}}]}
          {\mathbb P(\|X\|_2\le R)}
 \longrightarrow\mathbb E f(X)
\end{align*}
for every polynomial $f$. In particular, $a_R\to0$ and $\Sigma_R\to I_d$.
For all sufficiently large $R$, we may therefore set
\[
 Y_R=\Sigma_R^{-1/2}(X_R-a_R).
\]
Conditioning on a convex set and applying an affine map preserve
log-concavity, so $Y_R$ is isotropic, log-concave, and compactly supported.
Finally, $\Sigma_R^{-1/2}\to I_d$, so for each fixed $f$ the coefficients of
$f(\Sigma_R^{-1/2}(x-a_R))$ converge to those of $f(x)$. Together with
the moment convergence of $X_R$, this proves the claim.

<!-- definition-6: body_md -->
\label{def:strong-contamination}
Let $P$ be a distribution on $\mathbb R^d$ and let $0<\varepsilon<1$.
A clean sample of $n$ independent points is drawn from $P$.
After inspecting the entire sample, a computationally unbounded
adversary may replace at most $\varepsilon n$ of its points by arbitrary
vectors in $\mathbb R^d$. The resulting sample is called an
\emph{$\varepsilon$-corrupted sample} from $P$.
The estimator observes this sample but is not told which points
were replaced.
<!-- end definition-6 -->
<!-- proof-17: body_md -->
Following~\cite{KotSte17}, we first give an information-theoretic
criterion and then sketch how to turn it into an efficient estimator
using SoS. Given the observations $y_1,\ldots,y_n$, we require a
candidate clean sample $x'_1,\ldots,x'_n$, with empirical mean
$\mu'$ and covariance $\Sigma'$, to satisfy
\begin{equation}
\label{eq:robust-constraints}
 \begin{gathered}
  |\{i:x'_i\ne y_i\}|\le\varepsilon n,\\[2pt]
  \frac1n\sum_{i=1}^n\langle x'_i-\mu',v\rangle^\ell
  \le(C\ell)^\ell(v^\top\Sigma'v)^{\ell/2}
 \end{gathered}
\end{equation}
for every $v\in\mathbb R^d$ and every even $2\le\ell\le m$, where $C$
is a sufficiently large universal constant.
The moment bounds are relative to $\Sigma'$, so the constraints
can be imposed when $\Sigma$ is unknown.
The estimator returns $(\mu',\Sigma')$ for any such sample.

Let $x_1,\ldots,x_n$ be the clean sample, with empirical mean
$\mu_0$ and covariance $\Sigma_0\succ0$.
By \Cref{cor:hyper} and the sampling result
in~\cite[Lemma 5.5]{KotSte17}, this sample satisfies the constraints
with high probability. On this event, fix any candidate sample
satisfying the constraints.

Both samples agree with the observations
at all but $\varepsilon n$ indices, so they differ only on a set
$B=\{i:x'_i\ne x_i\}$ of relative size $\rho=|B|/n\le2\varepsilon$.
Centering the identity
$\mu'-\mu_0=n^{-1}\sum_{i\in B}(x'_i-x_i)$ and applying
H\"older's inequality gives
\begin{equation}
\label{eq:robust-mean-overlap}
 \begin{aligned}
  (1-\rho)|\langle\mu'-\mu_0,v\rangle|
  &=\left|\frac1n\sum_{i\in B}
    \langle(x'_i-\mu')-(x_i-\mu_0),v\rangle\right|\\
  &\le Cm\rho^{1-1/m}
    \bigl(\sqrt{v^\top\Sigma'v}+\sqrt{v^\top\Sigma_0v}\bigr)\\
  &\lesssim m\varepsilon^{1-1/m}
    \sqrt{v^\top(\Sigma'+\Sigma_0)v}.
 \end{aligned}
\end{equation}
The mean bound still involves $\Sigma'$. To express the error
relative to $\Sigma_0$ alone, we also need to compare the covariances.
We apply the same overlap argument to second moments, using
pairwise differences to remove the dependence on the empirical means:
\[
 v^\top\Sigma_0v
 =\frac1{2n^2}\sum_{i,j}\langle x_i-x_j,v\rangle^2,
\]
and its analogue for $\Sigma'$.
Only pairs with an index in $B$ contribute to the difference
between the two sums. There are at most $2\rho n^2\le4\varepsilon n^2$
such pairs.
The moment bounds and the even-power triangle inequality give
\[
 \frac1{n^2}\sum_{i,j}\langle x_i-x_j,v\rangle^m
 \le(2Cm)^m(v^\top\Sigma_0v)^{m/2},
\]
and similarly for $x'_1,\ldots,x'_n$.
H\"older's inequality with exponent $m/2$, applied to the squared
pairwise differences on these pairs, therefore gives
\begin{equation}
\label{eq:robust-cov-overlap}
 |v^\top(\Sigma'-\Sigma_0)v|
 \lesssim m^2\varepsilon^{1-2/m}\,v^\top(\Sigma'+\Sigma_0)v.
\end{equation}
Since $m^2\varepsilon^{1-2/m}$ is sufficiently small, writing
$\Sigma'+\Sigma_0=2\Sigma_0+(\Sigma'-\Sigma_0)$
in~\eqref{eq:robust-cov-overlap} and rearranging gives
\begin{equation}
\label{eq:robust-cov-comparison}
 (1-\delta)\Sigma_0\preceq\Sigma'\preceq(1+\delta)\Sigma_0,
 \qquad \delta\lesssim m^2\varepsilon^{1-2/m}.
\end{equation}
In particular, \eqref{eq:robust-cov-comparison} gives
$\Sigma'+\Sigma_0\preceq(2+\delta)\Sigma_0$.
Substituting this bound into~\eqref{eq:robust-mean-overlap}
and using $1-\rho\ge1-2\varepsilon$, we get
\begin{equation}
\label{eq:robust-mean-comparison}
 \|\Sigma_0^{-1/2}(\mu'-\mu_0)\|_2
 =\sup_{v^\top\Sigma_0v=1}|\langle\mu'-\mu_0,v\rangle|
 \lesssim m\varepsilon^{1-1/m}.
\end{equation}

To obtain an efficient estimator, we strengthen
\eqref{eq:robust-constraints} by requiring degree-$\ell$ SoS
certificates in $v$ for the moment bounds. Introducing their
coefficients as auxiliary variables gives a finite polynomial
encoding of these bounds; see~\cite[Section 4.3.4]{FleKP19-sos}.
By \Cref{cor:hyper} and the same sampling result, the clean sample
still satisfies these constraints. The comparison above admits
a low-degree SoS proof.
The semidefinite relaxation of Kothari and
Steurer~\cite[Theorem 1.2]{KotSte17} therefore yields estimates
satisfying~\eqref{eq:robust-cov-comparison}
and~\eqref{eq:robust-mean-comparison} in the claimed running time.
At the stated sample size, the sampling errors between
$(\mu_0,\Sigma_0)$ and $(\mu,\Sigma)$ only change the constants
in these bounds, which proves the claim.
