<!-- assumption-2: body_md -->
\label{ass:local}
There are constants $r>0$ and $0<f_\alpha\le L_\alpha<\infty$ such that $F$ is continuous on $[-r,r]$ and, for $0\le t\le r$,
\begin{enumerate}
\item[(a)] $F(t)-(1-\alpha)\ge f_\alpha t$ and $(1-\alpha)-F(-t)\ge f_\alpha t$;
\item[(b)] $F(t)-(1-\alpha)\le L_\alpha t$ and $(1-\alpha)-F(-t)\le L_\alpha t$.
\end{enumerate}

<!-- proof-1: body_md -->
For any four real numbers, $|\max(a,b)-\max(c,d)|\le\max(|a-c|,|b-d|)$. Applying this inequality to the two endpoint residuals gives $|\widehat{S}^{Q}_t-S^{Q}_t|\leE^{Q}_{n}$ at every calibration time and at the test time. The $k$-th smallest values then differ by at most $E^{Q}_{n}$, also when ties occur: at least $k$ of the oracle scores $S^{Q}_{n+1},\ldots,S^{Q}_{n+m}$ are at most $S^{Q}_{(k)}$, the corresponding fitted scores are at most $S^{Q}_{(k)}+E^{Q}_{n}$, and therefore $\widehat{S}^{Q}_{(k)}\leS^{Q}_{(k)}+E^{Q}_{n}$; interchanging the fitted and the oracle scores gives the reverse inequality, so $|\widehat{S}^{Q}_{(k)}-S^{Q}_{(k)}|\leE^{Q}_{n}$.

For each $x$, write $w(x)=u_\alpha(x)-\ell_\alpha(x)\ge0$ and $\widehat{w}(x)=\widehat{u}(x)-\widehat{\ell}(x)$. Then $\operatorname{len}C^{Q}_{\alpha}(x)=w(x)$ and $\operatorname{len}\widehat{C}^{Q}_{\alpha}(x)=[\widehat{w}(x)+2\widehat{S}^{Q}_{(k)}]_+$. Since the positive-part map is $1$-Lipschitz,
\begin{align*}
|[\widehat{w}(x)+2\widehat{S}^{Q}_{(k)}]_+-w(x)|&\le|\widehat{w}(x)-w(x)|+2|\widehat{S}^{Q}_{(k)}|\\
&\le 2E^{Q}_{n}+2|\widehat{S}^{Q}_{(k)}-S^{Q}_{(k)}|+2|S^{Q}_{(k)}|\le 4E^{Q}_{n}+2|S^{Q}_{(k)}|.
\end{align*}
Taking the supremum over $x$ proves the length bound in \eqref{eq:comparison}, including the cases of quantile crossing and empty fitted intervals. Since the bound holds simultaneously for all $x$ in the stated domain, it applies to an adjacent, dependent test covariate without any train--test decoupling argument.

<!-- proposition-2: body_md -->
\label{prop:envelope}
For every $s$,
\begin{equation}\label{eq:envelope}
\mathrm{Var}\Big(\sum_{i=1}^m\mathbf{1}\{S_{n+i}\le s\}\Big)\le\sum_{j=-\infty}^{m}\Big(\sum_{i=\max(1,j)}^m\pi_{i-j}(s)\Big)^2\le m\Big(\sum_{j\ge0}\pi_j(s)\Big)^2.
\end{equation}
If the marginal density of $S_t$ is at most $L$, then
\begin{equation}\label{eq:projectionbound}
\|W_j(s)-W_j^*(s)\|_2\le\min\Big\{\{2\min(F(s),1-F(s))\}^{1/2},\ \sqrt{3}\,L^{q/[2(q+1)]}\delta_q(j)^{q/[2(q+1)]}\Big\},
\end{equation}
where $\delta_q(j)=\|S_j-S_j^*\|_q$ is the functional dependence measure of the score.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
Fix $s$ and set $I_t=\mathbf{1}\{S_t\le s\}-F(s)$. As a square-integrable causal function of \text{i.i.d.}\ innovations, $I_t$ has the orthogonal decomposition $I_t=\sum_{j=-\infty}^t\mathcal{P}_jI_t$ in $L^2$; the backward limit of the conditional expectations is zero by triviality of the tail sigma-field of the \text{i.i.d.}\ past. Summing over $t=1,\ldots,m$ and using orthogonality across $j$ gives
\[
\Big\|\sum_{t=1}^mI_t\Big\|_2^2=\sum_{j=-\infty}^m\Big\|\sum_{t=\max(1,j)}^m\mathcal{P}_jI_t\Big\|_2^2
\le\sum_{j=-\infty}^m\Big(\sum_{t=\max(1,j)}^m\pi_{t-j}(s)\Big)^2,
\]
which is the first inequality in \eqref{eq:envelope} after a shift of the time index by stationarity. Regrouping the projections by lag and applying Minkowski's inequality,
\[
\Big\|\sum_{t=1}^mI_t\Big\|_2\le\sum_{k\ge0}\Big\|\sum_{t=1}^m\mathcal{P}_{t-k}I_t\Big\|_2=\sqrt m\sum_{k\ge0}\pi_k(s),
\]
where the equality uses orthogonality over the distinct projection times $t-k$; this is the second inequality.

For \eqref{eq:projectionbound}, the representation $W_j(s)-W_j^*(s)=\mathbb{E}\{J_j(s)-J_j^*(s)\mid\mathcal{F}_0,\varepsilon_0'\}$ and Jensen's inequality give $\|W_j(s)-W_j^*(s)\|_2\le\|J_j-J_j^*\|_2$. The two indicators have identical marginal laws, so their disagreement probability is at most both $2F(s)$ and $2(1-F(s))$. For every bandwidth $h>0$, disagreement also implies either $|S_j-s|\le h$ or $|S_j-S_j^*|>h$, hence
\[
\|J_j-J_j^*\|_2^2\le2Lh+\delta_q(j)^q/h^q.
\]
For $\delta_q(j)>0$, the choice $h=\{\delta_q(j)^q/L\}^{1/(q+1)}$ makes the right-hand side $3L^{q/(q+1)}\delta_q(j)^{q/(q+1)}$; if the dependence coefficient is zero, let $h\downarrow0$. Taking square roots and combining the two bounds proves \eqref{eq:projectionbound}.

<!-- proposition-3: body_md -->
\label{prop:convolution}
If $\sup_{s}\pi_j(s)\le B(1\vee j)^{-\gamma}$ for all $j\ge0$ and some $\gamma>1/2$, then $\sigma_m\le C_\gamma B\rho_m$ with
\begin{equation}\label{eq:sigmarate}
\rho_m=\begin{cases}m^{-1/2}, & \gamma>1,\\ m^{-1/2}\log(m+1), & \gamma=1,\\ m^{1/2-\gamma}, & 1/2<\gamma<1,\end{cases}
\end{equation}
and $C_\gamma$ depending only on $\gamma$.
<!-- end proposition-3 -->
<!-- proof-4: body_md -->
Since $(1\vee j)^{-\gamma}\le2^\gamma(j+1)^{-\gamma}$, the envelope is at most $2^\gamma B(j+1)^{-\gamma}$. For projection indices $-m<j\le m$, each inner sum in \eqref{eq:envelope} is at most $2^\gamma B\sum_{k=0}^{2m}(k+1)^{-\gamma}$, and there are at most $2m$ such indices; their total contribution is bounded by $CB^2m$, $CB^2m\log^2(m+1)$, or $CB^2m^{3-2\gamma}$ in the three regimes $\gamma>1$, $\gamma=1$ and $1/2<\gamma<1$. For $j\le-m$, put $v=-j\ge m$ and bound the inner sum by $2^\gamma Bm(v+1)^{-\gamma}$; this contributes at most $4^\gamma B^2m^2\sum_{v\ge m}(v+1)^{-2\gamma}\le C_\gamma B^2m^{3-2\gamma}$, which is at most a constant times $m$ for $\gamma>1$ and is absorbed by $m\log^2(m+1)$ for $\gamma=1$. Dividing by $m^2$ and taking the square root gives the claim.

\label{prop:quantileform}

\begin{equation}\label{eq:twothreshold}

\end{equation}
If in addition $M_m\ge\|S_{(k)}\|_2$ is a deterministic number and $a_m<r$, then
\begin{equation}\label{eq:rankmoment}
\mathbb{E}|S_{(k)}|\le a_m+2b_m+M_m\min\{1,b_m/(r-a_m)\},
\end{equation}
and a valid choice is
\begin{equation}\label{eq:rankM}
M_m=\bigg\{\frac{m\,\mathbb{E}(S_0)^2}{\min(k,m-k+1)}\bigg\}^{1/2}.
\end{equation}
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
For $a_m<t\le r$, Assumption~\ref{ass:local}(a) gives $F_+(t)-d_m\ge f_\alpha(t-a_m)$ and $F_-(t)+d_m\ge f_\alpha(t-a_m)$, and each numerator in \eqref{eq:exacttwothreshold} is at most $m^2\sigma_m^2$. This proves \eqref{eq:twothreshold}. For \eqref{eq:rankmoment}, the tail-integral identity gives
\begin{align*}
\mathbb{E}\min(|S_{(k)}|,r)=\int_0^r\mathbb{P}(|S_{(k)}|>t)\,dt
&\le a_m+\int_{a_m}^r\min\{1,b_m^2/(t-a_m)^2\}\,dt\\
&\le a_m+\int_0^\infty\min\{1,b_m^2/u^2\}\,du=a_m+2b_m,
\end{align*}
with the integral interpreted as zero when $b_m=0$, and the Cauchy--Schwarz inequality bounds $\mathbb{E}(|S_{(k)}|-r)_+\le\mathbb{E}[|S_{(k)}|\mathbf{1}\{|S_{(k)}|>r\}]$ by $M_m\mathbb{P}(|S_{(k)}|>r)^{1/2}\le M_m\min\{1,b_m/(r-a_m)\}$. <!-- lemma-5: body_md -->
\label{lem:shiftedradius}
\begin{enumerate}
\item[(i)] $a$ is even and nondecreasing in $|g|$, and $\Phi^{-1}(1-\alpha)\le a(g)-|g|\le z_\alpha$ for every $g$.
\item[(ii)] $H_g'(w)\ge f_*$ for every $g$ and every $w$ with $|w-a(g)|\le w_*$.
\item[(iii)] $\tfrac12\kappa_\alpha\min(g^2,1)\le a(g)-z_\alpha\le C_\alpha g^2$ for every $g$, with $C_\alpha=\max\{\phi(1)/f_1,1\}$.
\item[(iv)] $a(g)=z_\alpha+\tfrac12z_\alpha g^2+O(g^4)$ as $g\to0$.
\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-15: body_md -->
(i) $H_g(w)$ is even in $g$, and $\partial_gH_g(w)=\phi(w+g)-\phi(w-g)\le0$ for $g,w\ge0$, so $H_g(w)$ is nonincreasing in $|g|$ and $a(g)$ is nondecreasing in $|g|$; this is Anderson's inequality for the intervals $[g-w,g+w]$. For $g\ge0$, $1-\alpha=H_g(a(g))\le\Phi(a(g)-g)$ gives the lower bound on $a(g)-g$, and $H_g(g+z_\alpha)=\Phi(z_\alpha)+\Phi(2g+z_\alpha)-1\ge2\Phi(z_\alpha)-1=1-\alpha$ gives $a(g)\le g+z_\alpha$; evenness covers $g<0$.

(ii) For $g\ge0$ and $w\ge0$, $|w-g|\le w+g$, so $H_g'(w)\ge\phi(w-g)=\phi(w-|g|)$, and the same holds for $g<0$ by evenness. If $|w-a(g)|\le w_*$, then by (i) $w-|g|\in[\Phi^{-1}(1-\alpha)-w_*,z_\alpha+w_*]$, on which $\phi\ge f_*$.

(iii) For $0\le g\le1$,
\[
H_0(z_\alpha)-H_g(z_\alpha)=\int_0^g\{\phi(z_\alpha-u)-\phi(z_\alpha+u)\}\,du\in\big[\kappa_\alpha g^2/2,\ \phi(1)g^2\big].
\]
Since $a(g)\ge z_\alpha$ and $H_g'<1$, $1-\alpha=H_g(a(g))\le H_g(z_\alpha)+a(g)-z_\alpha$, and $1-\alpha=H_0(z_\alpha)$, so $a(g)-z_\alpha\ge H_0(z_\alpha)-H_g(z_\alpha)\ge\kappa_\alpha g^2/2$; for $|g|\ge1$, monotonicity gives $a(g)-z_\alpha\ge a(1)-z_\alpha\ge\kappa_\alpha/2$. For the upper bound with $|g|\le1$, every $u\in[z_\alpha,a(g)]$ satisfies $u-|g|\in[z_\alpha-1,z_\alpha]$ by (i), so $H_g'(u)\ge\phi(u-|g|)\ge f_1$ there, and $H_g(a(g))-H_g(z_\alpha)=H_0(z_\alpha)-H_g(z_\alpha)\le\phi(1)g^2$ gives $a(g)-z_\alpha\le\phi(1)g^2/f_1$; for $|g|>1$, (i) gives $a(g)-z_\alpha\le|g|\le g^2$.

(iv) The function $(g,w)\mapsto H_g(w)$ is smooth with $\partial_wH_g(w)>0$, so $a$ is smooth by the implicit function theorem, and even, so $a'(0)=0$. Differentiating $H_g(a(g))=1-\alpha$ twice at $g=0$ gives $\partial_g^2H_0(z_\alpha)+\partial_wH_0(z_\alpha)a''(0)=0$ with $\partial_g^2H_0(z_\alpha)=2\phi'(z_\alpha)=-2z_\alpha\phi(z_\alpha)$ and $\partial_wH_0(z_\alpha)=2\phi(z_\alpha)$, whence $a''(0)=z_\alpha$, and the odd derivatives vanish.

\label{lem:quantilemse}
Let $\tau\in(0,1)$, $\zeta_\tau=\Phi^{-1}(\tau)$ and $j_n=\lceil n\tau\rceil$. Under \eqref{eq:linearprocess}, $\mathbb{E}(G_{(j_n)}-\zeta_\tau)^2\le C_qn^{1-2\beta}$ for all $n\ge1$, where $G_{(j_n)}$ is the $j_n$-th smallest of $G_1,\ldots,G_n$ and $C_q$ depends only on $\tau$, $\beta$ and $A$.

<!-- proof-19: body_md -->
Let $\Gamma_n$ be the correlation matrix of $(G_1,\ldots,G_n)$ and $\lambda_n$ its largest eigenvalue, so that $1\le\lambda_n\le\max_i\sum_j|\rho(i-j)|\le1+2C_\rho n^{2-2\beta}/(2-2\beta)\le C_\lambda n^{2-2\beta}$ by \eqref{eq:rhobound}--\eqref{eq:sums}, with $C_\lambda=1+C_\rho/(1-\beta)$. Fix $u\in\mathbb{R}$ and $t>0$, and let $f_i(x)=\exp[t\{\mathbf{1}\{x\le u\}-\Phi(u)\}]$. The Gaussian H\"older inequality of \citet[Theorem~1]{chen2015improved} states that for a centered Gaussian vector with covariance matrix $T$, nonnegative measurable functions $f_i$ and exponents $p_i>0$ such that $T\preceq\mathrm{diag}(p_1T_{11},\ldots,p_nT_{nn})$, one has $\mathbb{E}\prod_if_i(X_i)\le\prod_i\{\mathbb{E} f_i(X_i)^{p_i}\}^{1/p_i}$; for two standard Gaussian variables with correlation $\varrho$ and $p_1=p_2=p$ the condition reads $(p-1)^2\ge\varrho^2$, which is Nelson's hypercontractivity condition. Since $\Gamma_n$ has unit diagonal and $\Gamma_n\preceq\lambda_nI$, the condition holds with all exponents equal to $\lambda_n$, and the inequality gives
\[
\mathbb{E}\prod_{i=1}^nf_i(G_i)\le\prod_{i=1}^n\big\{\mathbb{E} f_i(G_i)^{\lambda_n}\big\}^{1/\lambda_n}\le\exp\Big(\frac{n\lambda_nt^2}8\Big),
\]
where the last step is Hoeffding's lemma for the centered variable $\mathbf{1}\{G_i\le u\}-\Phi(u)$, whose range has length one. With $F_n(u)=n^{-1}\sum_{i\le n}\mathbf{1}\{G_i\le u\}$, the Chernoff bound with $t=4x/\lambda_n$, applied also to $-\{\mathbf{1}\{x\le u\}-\Phi(u)\}$, yields
\begin{equation}\label{eq:trainingecdf}
\mathbb{P}\{|F_n(u)-\Phi(u)|\ge x\}\le2\exp(-2nx^2/\lambda_n),\qquad x>0.
\end{equation}
Let $f_\tau=\min\{\phi(v):|v-\zeta_\tau|\le1\}$, so that $\Phi(\zeta_\tau+t)-\tau\ge f_\tau t$ and $\tau-\Phi(\zeta_\tau-t)\ge f_\tau t$ for $0\le t\le1$, and note that $\tau\le j_n/n\le\tau+1/n$. For $2/(nf_\tau)\le t\le1$, the event $G_{(j_n)}>\zeta_\tau+t$ implies $F_n(\zeta_\tau+t)<j_n/n$, hence $F_n(\zeta_\tau+t)-\Phi(\zeta_\tau+t)<1/n-f_\tau t\le-f_\tau t/2$, and the event $G_{(j_n)}<\zeta_\tau-t$ implies $F_n(\zeta_\tau-t)\ge\tau$, hence $F_n(\zeta_\tau-t)-\Phi(\zeta_\tau-t)\ge f_\tau t$; by \eqref{eq:trainingecdf},
\[
\mathbb{P}(|G_{(j_n)}-\zeta_\tau|>t)\le4\exp\Big(-\frac{nf_\tau^2t^2}{2\lambda_n}\Big),\qquad \frac{2}{nf_\tau}\le t\le1.
\]
Integrating $2t$ times the tail probability, the interval $t<2/(nf_\tau)$ contributes at most $4/(nf_\tau)^2$, the interval up to $t=1$ contributes at most $\int_0^\infty8t\exp\{-nf_\tau^2t^2/(2\lambda_n)\}\,dt=8\lambda_n/(nf_\tau^2)$, and the remainder is $\mathbb{E}[(G_{(j_n)}-\zeta_\tau)^2\mathbf{1}\{|G_{(j_n)}-\zeta_\tau|>1\}]\le\{\mathbb{E}(G_{(j_n)}-\zeta_\tau)^4\}^{1/2}\,2\exp\{-nf_\tau^2/(4\lambda_n)\}$. The rank inequality of Step 2 in the proof of Lemma~\ref{lem:finitecal}, applied to fourth powers, gives $G_{(j_n)}^4\le\sum_iG_i^4/\min(j_n,n-j_n+1)\le\sum_iG_i^4/\{n\min(\tau,1-\tau)\}$, so $\mathbb{E}(G_{(j_n)}-\zeta_\tau)^4\le8\{3/\min(\tau,1-\tau)+\zeta_\tau^4\}$. Since $\lambda_n/n\le C_\lambda n^{1-2\beta}$, $1/n^2\le n^{1-2\beta}$ and $\exp(-cn^{2\beta-1})\le n^{1-2\beta}/(ce)$ for $c>0$, all three contributions are at most a constant multiple of $n^{1-2\beta}$; for the finitely many $n<2/f_\tau$ the bound follows from the fourth-moment bound by enlarging the constant.

\emph{Step 1: any center.} Write $g=\widehat{\mu}-\mu$, which is $\mathcal{F}_n$-measurable, and assume $\mathbb{E} g^2<\infty$, since otherwise there is nothing to prove. The calibrated radius $\widehat{S}^{M}_{(k)}$ is the $k$-th smallest of $|Y_{n+i}-\widehat{\mu}|=|G_{n+i}-g|$, so Lemma~\ref{lem:finitecal} applies, and since $m^{1-2\beta}\le r_m(\beta)$ it gives $\mathbb{E}|\widehat{S}^{M}_{(k)}-a(g)|\le2K_3\{r_m(\beta)+(m^{1-2\beta}\mathbb{E} g^2)^{1/2}\}$. Lemma~\ref{lem:shiftedradius}(iii) gives $0\le a(g)-z_\alpha\le C_\alpha g^2$, and \eqref{eq:gaussdecomp} gives
\begin{align*}
\mathbb{E}|\operatorname{len}\widehat{C}^{M}_{\alpha}-2z_\alpha|=2\mathbb{E}|\widehat{S}^{M}_{(k)}-z_\alpha|&\le2\mathbb{E}|\widehat{S}^{M}_{(k)}-a(g)|+2\mathbb{E}\{a(g)-z_\alpha\}\\
&\le4K_3\big\{r_m(\beta)+(m^{1-2\beta}\mathbb{E} g^2)^{1/2}\big\}+2C_\alpha\mathbb{E} g^2.
\end{align*}
Finally $(m^{1-2\beta}\mathbb{E} g^2)^{1/2}\le\{m^{1-2\beta}+\mathbb{E} g^2\}/2\le\{r_m(\beta)+\mathbb{E} g^2\}/2$, which proves \eqref{eq:secondorder}. The center enters the concentration of $\widehat{S}^{M}_{(k)}$ around $a(g)$ only through $m^{1-2\beta}\mathbb{E} g^2$: a shift $g$ breaks the evenness of the indicator $\mathbf{1}\{|G_{n+i}-g|\le w\}$, whose first Hermite coefficient is of order $g$ and carries the long-memory variance $m^{1-2\beta}$ (Lemma~\ref{lem:condvar}). The center is never treated as independent of the calibration block: the argument conditions on the training innovations and retains the conditional means $\Pi_i-g$ of the calibration residuals, whose interaction with the center appears in the term $(m^{1-2\beta}\mathbb{E} g^2)^{1/2}$.

\emph{Step 2: the three centers.} The sample mean is a function of $Y_1,\ldots,Y_n$ with $\widehat{\mu}-\mu=\bar{G}_n$, and
\[
v_n=\mathrm{Var}(\bar{G}_n)=\frac1{n^2}\sum_{s,t=1}^n\rho(s-t)\le\frac1n\Big\{1+2\sum_{h=1}^{n-1}|\rho(h)|\Big\}\le\frac{2C_\rho}{n}\,\frac{n^{2-2\beta}}{2-2\beta}=\frac{C_\rho}{1-\beta}\,n^{1-2\beta}
\]
by \eqref{eq:rhobound}--\eqref{eq:sums} and $C_\rho\ge1$. The empirical median is the $\lceil n/2\rceil$-th smallest of $Y_1,\ldots,Y_n$, so that $\widehat{\mu}-\mu=G_{(\lceil n/2\rceil)}$, and Lemma~\ref{lem:quantilemse} with $\tau=1/2$ gives $\mathbb{E}(\widehat{\mu}-\mu)^2\le C_qn^{1-2\beta}$. For the midpoint $\widehat c=(\widehat{\ell}+\widehat{u})/2$ of $\widehat{\ell}=Y_{(\lceil n\alpha/2\rceil)}=\mu+G_{(\lceil n\alpha/2\rceil)}$ and $\widehat{u}=Y_{(\lceil n(1-\alpha/2)\rceil)}=\mu+G_{(\lceil n(1-\alpha/2)\rceil)}$, since $\zeta_{\alpha/2}+\zeta_{1-\alpha/2}=0$ for the normal quantiles $\zeta_\tau=\Phi^{-1}(\tau)$,
\[
\widehat c-\mu=\tfrac12\big\{\big(G_{(\lceil n\alpha/2\rceil)}-\zeta_{\alpha/2}\big)+\big(G_{(\lceil n(1-\alpha/2)\rceil)}-\zeta_{1-\alpha/2}\big)\big\},
\]
and Lemma~\ref{lem:quantilemse} at the two levels gives $\mathbb{E}(\widehat c-\mu)^2\le C_qn^{1-2\beta}$, with $C_q$ the larger of the two constants. Step 1 with these three bounds proves the second claim. By the identity of Section~\ref{sec:gaussian}, which holds for any real endpoints, also when $\widehat{\ell}>\widehat{u}$, the CQR interval with the empirical quantiles as fitted endpoints is the CMR interval with center $\widehat c$. The same lemma gives $\mathbb{E}(E^{Q}_{n})^2\le2C_qn^{1-2\beta}$ for the training error $E^{Q}_{n}=\max\{|\widehat{\ell}-\ell_\alpha|,|\widehat{u}-u_\alpha|\}$ of these endpoints, which is the explicit rate mentioned after Assumption~\ref{ass:train}.

\emph{Step 1: any center.} Write $g=\widehat{\mu}-\mu$ and assume $\mathbb{E} g^2<\infty$, since otherwise the right-hand side of \eqref{eq:lowerrisk} is $-\infty$. By \eqref{eq:gaussdecomp}, Lemma~\ref{lem:shiftedradius}(iii) and Lemma~\ref{lem:finitecal},
\begin{align*}
\mathbb{E}|\operatorname{len}\widehat{C}^{M}_{\alpha}-2z_\alpha|&=2\mathbb{E}|\widehat{S}^{M}_{(k)}-z_\alpha|\ge2\mathbb{E}\{a(g)-z_\alpha\}-2\mathbb{E}|\widehat{S}^{M}_{(k)}-a(g)|\\
&\ge\kappa_\alpha\mathbb{E}\min(g^2,1)-2K_3\big\{r_m(\beta)+m^{1-2\beta}+(m^{1-2\beta}\mathbb{E} g^2)^{1/2}\big\},
\end{align*}
which is \eqref{eq:lowerrisk} with $c_1=\kappa_\alpha$ and $C=4K_3$, since $m^{1-2\beta}\le r_m(\beta)$.

\emph{Step 2: a common Gaussian component of the center.} Let $V=n^{-1/2}\sum_{s=-2n+1}^{-n}\varepsilon_s\sim N(0,1)$, a function of the innovations preceding the training block, and let $b_t=\mathrm{Cov}(G_t,V)=n^{-1/2}\sum_{s=-2n+1}^{-n}a_{t-s}$ for $1\le t\le n$. The indices $t-s$ range over $[t+n,t+2n-1]\subset[n+1,3n-1]$, so for $n\ge j_0$ the lower bound on the coefficients gives
\[
b_t\ge n^{-1/2}\cdot n\cdot c\,(3n)^{-\beta}=h_n,\qquad h_n=c\,3^{-\beta}n^{1/2-\beta}.
\]
Write $W_t=G_t-b_tV$. The vector $(W_1,\ldots,W_n)$ is jointly Gaussian with $V$ and uncorrelated with it, hence independent of $V$. Since $T$ is translation equivariant, $g=T(Y_1,\ldots,Y_n)-\mu=T(G_1,\ldots,G_n)=T(W+bV)$, where $b=(b_1,\ldots,b_n)^\top$. For fixed $W$, put $f_W(v)=T(W+bv)$; for $h\ge0$, every coordinate of $W+b(v+h)$ exceeds the corresponding coordinate of $W+bv+h_nh\mathbf1$, so monotonicity and equivariance give $f_W(v+h)\ge f_W(v)+h_nh$. If $f_W(0)\ge0$, then $g\ge h_n$ on the event $\{V\ge1\}$; if $f_W(0)<0$, then $g\le f_W(0)-h_n|V|\le-h_n$ on the event $\{V\le-1\}$. Since $V$ is independent of $W$ and $\mathbb{P}(V\ge1)=\mathbb{P}(V\le-1)=\Phi(-1)$,
\[
\mathbb{P}(|g|\ge h_n)\ge\Phi(-1),\qquad \mathbb{E}\min(g^2,1)\ge\Phi(-1)\min(h_n^2,1)=\Phi(-1)c^23^{-2\beta}n^{1-2\beta}
\]
for $n\ge n_0$, where $n_0\ge\max(j_0,2)$ is such that $h_n\le1$. No Gaussianity or unbiasedness of $g$ is used.

\emph{Step 3: a calibration block large relative to the training block.} Let $\nu=2\beta-1\in(0,1)$ and $c_2=\Phi(-1)c^23^{-2\beta}$. By Steps 1 and 2 and $\mathbb{E} g^2\le C_gn^{-\nu}$, for $n\ge n_0$ and $m\ge m_0$,
\[
\mathbb{E}|\operatorname{len}\widehat{C}^{M}_{\alpha}-2z_\alpha|\ge c_1c_2n^{-\nu}-C\big\{r_m(\beta)+C_g^{1/2}(nm)^{-\nu/2}\big\}.
\]
Let $m\ge Kn^{\max(1,2\nu)}\log(n+1)$ with $K\ge m_0$, and recall that $n\ge2$, so that $\log(n+1)\ge1$. If $\nu<1/2$, then $m\ge Kn$, so $r_m(\beta)=m^{-\nu}\le K^{-\nu}n^{-\nu}$ and $(nm)^{-\nu/2}\le K^{-\nu/2}n^{-\nu}$. If $\nu=1/2$, then, since $\log(m+1)/m$ is decreasing in $m$ and $Kn\log(n+1)+1\le K(n+1)^2$, $r_m(\beta)^2\le\{\log K+2\log(n+1)\}/\{Kn\log(n+1)\}\le(\log K+2)/(Kn)$, and $(nm)^{-1/4}\le K^{-1/4}n^{-1/2}$. If $\nu>1/2$, then $m\ge Kn^{2\nu}$, so $r_m(\beta)=m^{-1/2}\le K^{-1/2}n^{-\nu}$ and $(nm)^{-\nu/2}\le K^{-\nu/2}n^{-\nu(1+2\nu)/2}\le K^{-\nu/2}n^{-\nu}$, since $2\nu>1$. In each regime $r_m(\beta)+C_g^{1/2}(nm)^{-\nu/2}\le\epsilon_Kn^{-\nu}$ with $\epsilon_K\to0$ as $K\to\infty$, so for $K$ large the right-hand side of the last display is at least $c_1c_2n^{-\nu}/2$ and $r_m(\beta)\le n^{-\nu}$, which gives \eqref{eq:lowerbound} with $c_3=c_1c_2/4$.

<!-- corollary-1: body_md -->
\label{cor:coverage}
Assume \eqref{eq:linearprocess}, and let $\widehat{\mu}$ be as in Theorem~\ref{cor:rates}, with $\mathbb{E}(\widehat{\mu}-\mu)^2<\infty$. For all $n$ and all $m\ge m_0$, with $m_0$ as in Theorem~\ref{cor:rates},
\begin{equation}\label{eq:learnedcoverage}
\begin{aligned}
\big|\mathbb{P}\{Y_{n+m+1}\in\widehat{C}^{M}_{\alpha}\}-(1-\alpha)\big|&\le C\big[r_m(\beta)+\{m^{1-2\beta}\mathbb{E}(\widehat{\mu}-\mu)^2\}^{1/2}+m^{1-2\beta}\big]\\
&\le C\big\{\mathbb{E}(\widehat{\mu}-\mu)^2+r_m(\beta)\big\}.
\end{aligned}
\end{equation}
For the sample mean, the empirical median and the midpoint of Theorem~\ref{cor:rates}, the coverage error is therefore at most $C\{n^{-(2\beta-1)}+r_m(\beta)\}$.

Write $g=\widehat{\mu}-\mu$. The calibrated radius $\widehat{S}^{M}_{(k)}$ is the $k$-th smallest of $|G_{n+i}-g|$, $1\le i\le m$, and $Y_{n+m+1}\in\widehat{C}^{M}_{\alpha}$ exactly when $|G_{n+m+1}-g|\le\widehat{S}^{M}_{(k)}$. Let $\mathcal{F}_{n+m}$ denote the sigma-field generated by the innovations up to time $n+m$ and by the randomness of the training algorithm. Conditionally on $\mathcal{F}_{n+m}$, $G_{n+m+1}=a_0\varepsilon_{n+m+1}+\Pi'$ with $\Pi'=\sum_{j\ge1}a_j\varepsilon_{n+m+1-j}$ measurable with respect to $\mathcal{F}_{n+m}$, so $G_{n+m+1}$ is Gaussian with variance $a_0^2$, whereas $g$, $a(g)$ and $\widehat{S}^{M}_{(k)}$ are $\mathcal{F}_{n+m}$-measurable. Hence the conditional probability $\mathbb{P}(|G_{n+m+1}-g|\le q\mid\mathcal{F}_{n+m})=\Phi\{(q+g-\Pi')/|a_0|\}-\Phi\{(-q+g-\Pi')/|a_0|\}$ is Lipschitz in $q$ with constant $2\phi(0)/|a_0|$, and
\[
\big|\mathbb{P}\{|G_{n+m+1}-g|\le\widehat{S}^{M}_{(k)}\}-\mathbb{P}\{|G_{n+m+1}-g|\le a(g)\}\big|\le\frac{2\phi(0)}{|a_0|}\,\mathbb{E}|\widehat{S}^{M}_{(k)}-a(g)|.
\]
For the second probability, condition on $\mathcal{F}_n$ and use the decomposition \eqref{eq:decomposition} at $i=m+1$: $\mathbb{P}\{|G_{n+m+1}-g|\le a(g)\mid\mathcal{F}_n\}=\pi_{a(g)}(\Pi_{m+1}-g,s_{m+1})$, whereas $\pi_{a(g)}(-g,1)=H_g(a(g))=1-\alpha$ by the definition of $a(g)$. The bound on a single summand in the proof of Lemma~\ref{lem:condmean}, at $i=m+1$, and the Cauchy--Schwarz inequality give
\begin{align*}
\big|\mathbb{P}\{|G_{n+m+1}-g|\le a(g)\}-(1-\alpha)\big|&\le K_1\mathbb{E}\big(|g\Pi_{m+1}|+\Pi_{m+1}^2+\tau_{m+1}\big)\\
&\le K_1\big\{(\tau_{m+1}\mathbb{E} g^2)^{1/2}+2\tau_{m+1}\big\},
\end{align*}
and $\tau_{m+1}\le A^2m^{1-2\beta}/(2\beta-1)$ by \eqref{eq:sums}. Combining the two displays with Lemma~\ref{lem:finitecal} proves the first inequality in \eqref{eq:learnedcoverage}; the second follows from $m^{1-2\beta}\le r_m(\beta)$ and $2(xy)^{1/2}\le x+y$. For the three centers, Step~2 of the proof of Theorem~\ref{cor:rates} gives $\mathbb{E} g^2\le Cn^{1-2\beta}$.
