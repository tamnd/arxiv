<!-- definition-1: body_md -->
\label{def;k-col}
	\label{df;par-coll} Given any $\mathbf k \subset \mathbf n$ with $|\mathbf k| \ge 2$, we say a solution $q: (0, T) \to \mathbb E$, is $\mathbf k$-collision, if it is a collision-free solution for $0< t <T$ with $\lim_{t \to T} q(t) = q^*$, where
	$$ q^* \in \Delta \text{ with } \begin{cases}
		q^*_i = q^*_j, & \forall i \ne j \in \mathbf k; \\
		q^*_i \ne q^*_j, & \forall i \in \mathbf k, j \in \mathbf k':= \mathbf n \setminus \mathbf k. 
	\end{cases}
     $$
<!-- end definition-1 -->
<!-- remark-1: body_md -->
For any subset $\mathbf k$ of indices, $|\mathbf k|$ denotes its cardinality.
	A $\mathbf k$-collision solution is partial, if $|\mathbf k| < n$, and total, if $|\mathbf k|= n$. Notice that there may be more than one partial collision occurring at the same time.
<!-- end remark-1 -->
<!-- definition-2: body_md -->
\label{df;cc} $q_{\mathbf k} = (q_i)_{i \in \mathbf k} \in \mathbb E_{\mathbf k}$ is a central configuration or CC of the $\mathbf k$-subsystem, if there is a constant $\lambda >0$, such that 
$$ \nabla_{i} U_{\mathbf k}(q_{\mathbf k}) + \lambda m_i q_i = 0, \; \forall i \in \mathbf k.$$
We denote the set of all CCs of the $\mathbf k$-subsystem by $\mathscr{C}_{\mathbf k}$ and the subset of normalized CCs as 
$$ \hat{\mathscr{C}}_{\mathbf k} = \mathscr{C}_{\mathbf k} \cap S_{\mathbf k}. $$
<!-- end definition-2 -->
<!-- remark-2: body_md -->
$\hat{\mathscr{C}}_{\mathbf k}$ is precisely the set of critical points of partial potential $U_{\mathbf k}$ restricted on $S_{\mathbf k}$.
<!-- end remark-2 -->
<!-- remark-3: body_md -->
When $d=1$ it is well-known the answer to the above question is Yes, so below we will always assume $d \ge 2$.
<!-- end remark-3 -->
<!-- theorem-1: body_md -->
\label{thm;coll}  If $q(t)$ is a $\mathbf k$-collision solution, the normalized relative configuration $q^c_{\mathbf k}(t)$ converges to a particular CC in $\mathscr{C}_{\mathbf k}$ as the masses approach to the $\mathbf k$-collision. In particular there is no infinite spin.
<!-- end theorem-1 -->
<!-- definition-3: body_md -->
\label{def;k-para}
\label{df;k-para} Given any $\mathbf k \subset \mathbf n$ with $|\mathbf k| \ge 2$, we say a solution $q: (t_0, \infty) \to \mathbb E$ is $\mathbf k$-parabolic, if there exist positive constants $C_i$, $i =1, 2, 3$, such that when $t \to \infty$, 
$$ 
\begin{cases}
C_1 t^{\frac{2}{3}} \le r_{ij}(t) \le C_2 t^{\frac{2}{3}}, & \; \forall i \ne j \in \mathbf k; \\
r_{ij}(t) \ge C_3 t, & \; \forall i \in \mathbf k, \forall j \in \mathbf k':= \mathbf n \setminus \mathbf k.
\end{cases} 
$$
<!-- end definition-3 -->
<!-- remark-4: body_md -->
A $\mathbf k$-parabolic solution is partially parabolic, if $|\mathbf k| < n$, and complete parabolic, if $|\mathbf k|=n$. Notice that there may be more than one subsystem, such that inside it the motion is parabolic.
<!-- end remark-4 -->
<!-- theorem-2: body_md -->
\label{thm;para}  If $q(t)$ is a $\mathbf k$-parabolic solution, the normalized relative configuration $q^c_{\mathbf k}(t)$ converges to a particular CC in $\mathscr{C}_{\mathbf k}$, when time goes to infinity. In particular there is no infinite spin.
<!-- end theorem-2 -->
<!-- lemma-1: body_md -->
\label{lem;est-ft}
    If $q(t)$ is a $\mathbf k$-collision solution on $(0, T)$, then for $t_0<T$ sufficiently close to $T$, there exists a constant $C>0$ such that
    \begin{align}\label{eq;est-ft-col}
        \lVert f(t)\rVert_m\le C r(t), \qquad t_0\le t\le T.
    \end{align}
    If $q(t)$ is a $\mathbf k$-parabolic solution on $(0, \infty)$, then for sufficiently large $t_0$, there exists a constant $C>0$ such that
    \begin{align}
        \lVert f(t)\rVert_m\le C r(t)t^{-3}=O(t^{-7/3}), \qquad t\ge t_0.
    \end{align}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $G(y)=y/|y|^3$ be a vector value function and $y\in\mathbb R^d$ is a column vector. The differentiation gives
$$
 DG(y)=|y|^{-3}I-3|y|^{-5}yy^{\mathsf T},
 \qquad \|DG(y)\|\le 2|y|^{-3}.
$$
Here differential $DG$ can be seen as a matrix function, and $I$ is the identity matrix. 
From \eqref{eq;external-filed}, 
$$|F_t(q_i)-F_t(q_\ell)|\le \sum_{j\notin \mathbf k}m_j|G(q_j-q_i)-G(q_j-q_{\ell})|,\quad i,\ell\in\mathbf k.$$
In collision case, since $q_i(t)-q_{\ell}(t)\to 0$ and $|q_j(t)-q_{\ell}(t)|$ is bounded as $t\to T$, for $\theta\in (0,1)$, there exists $C_1>0$ and $t_0<T$ such that
$$
|\theta(q_j-q_i)+(1-\theta)(q_j-q_{\ell})|\ge |q_j-q_{\ell}|-\theta|q_i-q_{\ell}|\ge C_1,\qquad  t_0\le t\le T.
$$ 
Similarly in parabolic case, there exists $C_1>0$ and $t_0>0$ such that
$$
|\theta(q_j-q_i)+(1-\theta)(q_j-q_{\ell})|\ge C_1 t,\qquad t\ge t_0.
$$
Then the mean-value estimate gives 
$$
    |F_t(q_i)-F_t(q_\ell)|\le 
    \begin{cases}
    C_2|z_i-z_\ell|,\quad t_0\le t\le T  &\mathbf k\text{-collision},\\
 C_2t^{-3}|z_i-z_\ell|,\quad t\ge t_0 &\mathbf k\text{-parabolic},
 \end{cases}
$$
for some constant $C_2>0$.
Now write $f_i=m_{\mathbf k}^{-1}\sum_\ell m_\ell(F_t(q_i)-F_t(q_\ell))$ and use $|z_i|\le r/\sqrt{m_i}$, we have $\lVert f\rVert_m\le C r(t)$ in collision case and $\lVert f\rVert_m\le C r(t) t^{-3}$ in parabolic case. Finally, the identity
\begin{equation}\label{eq;pair-identity}
 r^2=m_{\mathbf k}^{-1}\sum_{\substack{i<j\\i,j\in \mathbf k}}m_im_j|z_i-z_j|^2
\end{equation}
and $|z_i|\le r/\sqrt{m_i}$ implies $r\sim t^{2/3}$ in the parabolic case. Here and below $A\sim B$ means $A/B$ are bounded below and above by finite constants. This gives the estimate $\lVert f\rVert_m=O(t^{-7/3})$ in parabolic case.

<!-- lemma-2: body_md -->
\label{lem;L-inq}
    Assume $\mathcal C$ is a compact subset of critical points on which $V=V_0$, which is a constant. There exists a neighborhood $N$ of $\mathcal C$, a constant $C>0$, and an exponent $\theta\in(0,1/2]$ such that

 |V(y)-V_0|^{1-\theta}\le C\lVert \nabla V(y)\rVert\quad y\in N.
\end{equation}

<!-- proof-2: body_md -->
The classical \L ojasiewicz gradient inequality states that for a critical point $x\in M$, there exists a neighborhood $N_x$, a constant $C_x$ and an exponent $\theta_x$ satisfying the inequality 
    $$
    |V(y)-V(x)|^{1-\theta_x}\le C_x\lVert \nabla V(y)\rVert\quad y\in N_x.
    $$
    We also assume that $|V(y)-V(x)|\le 1$ for $y\in N_x$. Since $\mathcal{C}$ is a compact subset and $V=V_0$ on $\mathcal{C}$, we can choose a finite cover of $\mathcal{C}$ by the neighborhoods $\{N_{x_i}\}$. Additionally, we have the corresponding constants $\{C_{x_i}\}$ and $\{\theta_{x_i}\}$. Let $C=\max\{C_{x_i}\}$, $\theta=\min\{\theta_{x_i}\}$ and $N=\cup N_{x_i}$, then we obtain this Lemma.

<!-- theorem-3: body_md -->
\label{thm;finite-length-analytic-gradient-equation}
Let $M$, $V$ and $\mathcal{C}$ be as above. Let $y:[\tau_0,\infty)\to M$ solve the equations
\begin{equation}\label{eq;sec-order-equation}
 u=y', \qquad D_\tau u=\nabla V(y)-a(\tau)u+e(\tau),
\end{equation}
where $D_\tau$ denotes covariant differentiation and $a\in C^0(\mathbb R,\mathbb R)$, $e\in C^0(\mathbb R,TM)$. Suppose
\begin{enumerate}[label=(\roman*)]
\item[$(i)$.] $\operatorname{dist}(y(\tau),\mathcal C):=\inf_{y\in \mathcal{C}} d(y(\tau),y)\to 0,\ \tau\to \infty$;
\item[$(ii)$.] $\lVert \nabla V(y(\tau))\rVert+\lVert u(\tau)\rVert\le1$ and $\lVert u(\tau)\rVert\to0,\ \tau\to \infty$;
\item[$(iii)$.] There exist constants $0<a_0\le a_1$ such that $a_0\le a(\tau)\le a_1$ when $\tau\ge\tau_0$ or $a_0\le -a(\tau)\le a_1$ when $\tau\ge \tau_0$;
\item[$(iv)$.] $e(\tau)\in T_{y(\tau)}M$ and $\lVert e(\tau)\rVert\le \min\{1,c e^{-\kappa\tau}\},\ \tau\ge\tau_0$ for some $c>0,\ \kappa>0$.
\end{enumerate}
Then
\begin{equation}\label{abstractlength}
 \int_{\tau_0}^{\infty}\bigl(\lVert u\rVert+\lVert \nabla V(y)\rVert\bigr)\,d\tau<\infty
\end{equation}
and therefore $y$ converges to a point of $\mathcal C$.
<!-- end theorem-3 -->
<!-- proof-3: body_md -->
Choose a neighborhood $N$ of $\mathcal{C}$ as described in the above Lemma. Without loss of generality, we assume that $N$ is a precompact neighborhood and $y(\tau)\in N$ for $\tau\ge \tau_0$ by assumption $(i)$. Set $g(\tau)=\nabla V(y(\tau))$, and choose a constant $L$ bounding $\lVert \operatorname{Hess} V\rVert$ in $N$. By the assumptions, we have $g\to0$, $V(y)\to V_0$, and $u\to0$ as $\tau\to\infty$. Let
$$
 H=\tfrac12\lVert u\rVert^2-(V(y)-V_0),\qquad
 E_0=H-\varepsilon\langle g,u\rangle,
$$
where $\varepsilon>0$ will be fixed sufficiently small. \eqref{eq;sec-order-equation} yields the identities
\begin{align*}
 H'&=-a\lVert u\rVert^2+\langle e,u\rangle,\\
 (\langle g,u\rangle)'&=\operatorname{Hess} V[u,u]+\lVert g\rVert^2-a\langle g,u\rangle+\langle g,e\rangle.
\end{align*}
Consequently
\begin{align}\label{Ederiv}
 E_0'=&-a\lVert u\rVert^2-\varepsilon\lVert g\rVert^2
 -\varepsilon\operatorname{Hess} V[u,u]+\varepsilon a\langle g,u\rangle\\
 &\hspace{9mm}+\langle e,u\rangle-\varepsilon\langle g,e\rangle.\nonumber
\end{align}
Here we assume that $a(\tau)$ is positive. If $a(\tau)$ is negative, we just let $E_0=-H-\varepsilon\langle g,u\rangle$. Apply Young's inequality in the forms
\begin{align*}
\varepsilon a_1\lVert g\rVert\lVert u\rVert&\le\tfrac\varepsilon4\lVert g\rVert^2+\varepsilon a_1^2\lVert U\rVert^2,\\
 \lVert e\rVert\lVert u\rVert&\le\tfrac{a_0}4\lVert u\rVert^2+a_0^{-1}\lVert e\rVert^2,\\
 \varepsilon\lVert g\rVert\lVert e\rVert&\le\tfrac\varepsilon4\lVert g\rVert^2+\varepsilon\lVert e\rVert^2.
\end{align*}
Choose $\varepsilon(L+a_1^2)\le a_0/4$. It follows that there are $c_0,c_1>0$ for which
\begin{equation}\label{eq;inq-H}
 E_0'\le- c_0(\lVert u\rVert^2+\lVert g\rVert^2)+c_1 e^{-2\kappa\tau}.
\end{equation}
Choose $B>(c_1+1)/(2\kappa)$ and define the corrected energy
\[
 E=E_0+B e^{-2\kappa\tau}.
\]
Then, with $b(\tau)=e^{-\kappa\tau}$ and a new constant $c_2>0$,
\begin{equation}\label{eq;Ecorrect}
 -E'\ge c_2(\lVert u\rVert^2+\lVert g\rVert^2+b^2).
\end{equation}
Also $E\to0$ as $\tau\to\infty$. Since $E$ decreases to zero, $E\ge0$. In fact, $E>0$ at every finite time because $b>0$. Thus, we can take its positive fractional powers.

Let $p=1-\theta\in[1/2,1)$, where $\theta$ is from Lemma~\ref{lem;L-inq}. Then \L  ojasiewicz inequality implies
\begin{align}\label{eq;Ep-bound}
 E^p&\le C_0\left(|V-V_0|^p+\lVert u\rVert^{2p}+\lVert g\rVert^p\lVert u\rVert^p+b^{2p}\right)\\
 &\le C_1\left(\lVert g\rVert+\lVert u\rVert+b\right).\nonumber
\end{align}
Here $2p\ge1$ gives $\lVert u\rVert^{2p}\le\lVert u\rVert$ and $b^{2p}\le b$. The mixed term is obtained by $\lVert g\rVert^p\lVert u\rVert^p\le(\lVert g\rVert+\lVert u\rVert)^{2p}\le \lVert g\rVert+\lVert u\rVert.$

Combining \eqref{eq;Ecorrect} and \eqref{eq;Ep-bound}, and using the elementary inequality $A^2+B^2+C^2\ge(A+B+C)^2/3$, we have
\begin{equation}\label{lengthineq}
 -(E^\theta)'=\theta\frac{-E'}{E^p}
 \ge c_3\bigl(\lVert u\rVert+\lVert g\rVert+b\bigr).
\end{equation}
Integration from $\tau_0$ to infinity proves the more precise estimate
\begin{equation}\label{tailestimate}
 \int_{\tau_0}^\infty(\lVert u\rVert+\lVert g\rVert+b)\,d\tau\le C E(\tau_0)^\theta.
\end{equation}
Thus $y$ has finite length and therefore has a limit, which belongs to $\mathcal C$ by assumption $(i)$.

<!-- proposition-1: body_md -->
\label{prop;MaC-col-eq}
$\gamma(\tau)$ satisfies the following time-dependent equations

\begin{equation}  \label{blowup-coll}
\begin{aligned}
 r'&=rv,\\
 v'&=\tfrac12v^2+\lVert w\rVert_m^2-U(s)+r^2\langle s,f(\tau)\rangle_m, \\
 s'&=w, \\
 w'&=\nabla_{S_{\mathbf k}} U_{\mathbf k}(s)-\tfrac12vw+r^2\bigl(f(\tau)-\langle s,f(\tau)\rangle_m s\bigr)-\lVert w\rVert_m^2s, 
\end{aligned} 
\end{equation}
where $\nabla_{S_{\mathbf k}}$ denotes the gradient on sphere $S_{\mathbf k}$. 

Moreover the following identities hold
\begin{equation}
     D_{\tau}w=\nabla_{S_{\mathbf k}}U_{\mathbf k}(s)-\tfrac12vw+e(\tau) \; \text{ where } \; e=r^2\bigl(f-\langle s,f\rangle_m s\bigr);
\end{equation}
\begin{equation}\label{eq;scaled-energy-coll}
 rh_{\mathbf k}=\tfrac12(v^2+\lVert w\rVert_m^2)-U_{\mathbf k}(s);
\end{equation}
\begin{equation}\label{eq;v-mono-coll}
 v'=\tfrac12\lVert w\rVert_m^2+rh_{\mathbf k}+r^2\langle s,f\rangle_m.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
Differentiating $r^2=\langle z,z\rangle_m$ twice gives 
$$
\dot{r}^2+r\ddot{r}=\langle \dot{r}s+r\dot{s},\dot{r}s+r\dot{s}\rangle_m+\langle rs,\ddot{z}\rangle_m.
$$
Combining $\langle s,\dot s\rangle_m=0$ and \eqref{eq;eq-subsystem}, we have
$$
\ddot r=r\lVert \dot s\rVert_m^2-r^{-2}U_{\mathbf k}(s)+\langle s,f\rangle_m.
$$
Here homogeneity yields $\nabla U_{\mathbf k}(rs)=r^{-2}\nabla U_{\mathbf k}(s)$ and $\langle s,\nabla U_{\mathbf k}(s)\rangle_m=-U_{\mathbf k}(s)$. Then 
$$v'=r^{3/2}(\frac{1}{2}r^{-1/2}\dot{r}+r^{1/2}\ddot{r})=\tfrac12v^2+\lVert w\rVert_m^2-U(s)+r^2\langle s,f(\tau)\rangle_m.$$
Again differentiating $z=rs$ twice implies
$$
\ddot{r}s+2\dot{r}\dot{s}+r\ddot{s}=\nabla U_{\mathbf k}+f,
$$
and 
$$
\ddot{s}=r^{-3}\nabla U_{\mathbf k}(s)+r^{-3}U_{\mathbf k}(s)s+r^{-1}f-2r^{-3}vw-r^{-1}\langle s,f\rangle_m s-\lVert \dot{s}\rVert_m^2s.
$$
The tangential projection gives
$$
D_t\dot s=\ddot{s}-\langle \ddot{s},s\rangle_m s=r^{-3}\nabla_{S_{\mathbf k}}U_{\mathbf k}(s)+r^{-1}(f-\langle s,f\rangle_m s)-2r^{-3}vw.
$$
Since $D_{t}w=r^{3/2}D_t \dot{s}+D_t(r^{3/2})\dot{s}$,
$$
 D_\tau w=\tfrac32vw+r^3D_t\dot s
 =-\tfrac12vw+\nabla_{S_{\mathbf k}}U_{\mathbf k}(s)+r^2\bigl(f-\langle s,f\rangle_m s\bigr).
$$
Differentiating $v=r^{1/2}\dot r$ proves the radial equation. Finally
$$
 \dot z=r^{-1/2}(vs+w),\qquad \langle s,w\rangle_m=0,
$$
which proves \eqref{eq;scaled-energy-coll} and \eqref{eq;v-mono-coll}.

<!-- proposition-2: body_md -->
\label{prop;asym-col}
Suppose $q(t)$ is a $\mathbf k$-collision solution for $t\in[0,T)$ and let $\gamma(\tau)=(r, v, s, w)(\tau)$ be the corresponding orbit of the $\mathbf k$-subsystem in MaGehee coordinates. Then when $\tau$ tends to $\infty$, there exists a negative constant $v_0$ such that

 \nabla_{S_{\mathbf k}} U_{\mathbf k}(s)\to0.
$$

By estimates \eqref{eq;asym-col} and identity $U_{\mathbf k}(z(t))=K_{\mathbf k}(t)-h_{\mathbf k}(t)=O(|T-t|^{-2/3})$, we obtain that
    $$U_{\mathbf k}(s)=rU_{\mathbf k}(z)=O(1).$$
Since the preimage of a bounded set for $U_{\mathbf k}$ is a compact subset of $S_{\mathbf k}$, $s(\tau)$ stays in a compact subset of $S_{\mathbf k}$. 

From Lemma~\ref{lem;est-energy-col}, $rh_{\mathbf k}\to 0$ as $\tau\to \infty$ and since $U_{\mathbf k}$ has a positive lower bound, \eqref{eq;scaled-energy-coll} implies that $|v|$ and $\lVert w\rVert_m$ are bounded. Combining the equation \eqref{eq;v-mono-coll} and estimate \eqref{eq;est-r2sf}, 
$$
v(\tau)-\int_{\tau_0}^{\tau}rh_{\mathbf k}+r^2\langle s,f\rangle_m\,d\eta
$$
is uniform bounded and nondecreasing. Thus $v(\tau)$ has a limit $v_0$ and $\int_{\tau_0}^{\infty}\lVert w\rVert_m^2d\tau<\infty$, which also implies $w\to 0$ when $\tau\to\infty$.
