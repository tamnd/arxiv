<!-- definition-1: body_md -->
\label{Def:Coordinate_Ring_of_A}
Let $\mathcal{A} \subset \mathbb{Z}^n$ be a finite set. For $s \in \mathbb{N}$ we define the sets 
\begin{equation*}
\mathcal{A}_s \coloneqq \Bigl\{ \: \sum_{i= 1}^m  \alpha_i  \: | \: m \leq s, \alpha_i \in \mathcal{A}  \Bigl\} \quad \text{ and } \quad \mathcal{A}_\infty \coloneqq \mathrm{lin}_\mathbb{N}(\mathcal{A}) = \left\{ \sum_{i = 0}^m \alpha_i \mid m \in \mathbb{N} , \alpha_i \in \mathcal{A}  \right\},
\end{equation*}
as well as the truncation
\begin{equation*}
\mathbb{R}[\mathcal{A}]_s \coloneqq \Bigl\{ \sum_{\alpha \in \mathcal{A}_s} c_{\alpha} x^{\alpha} \: | \: c_{\alpha} \in \mathbb{R} \Bigl\} \quad \text{ and the ring } \quad \mathbb{R}[\mathcal{A}] = \bigcup_{s \in \mathbb{N}} \mathbb{R}[\mathcal{A}]_s.
\end{equation*}



Further, the \emph{degree} of a polynomial $f \in \mathbb{R}[\mathcal{A}]$ with respect to $\mathcal{A}$ is defined as 
\begin{equation*}
\deg_{\mathcal{A}}(f) \coloneqq \min\{ s \in \mathbb{N} \: | \: f \in \mathbb{R}[\mathcal{A}]_s \}.
\end{equation*}
\label{DefinitionRA}
<!-- end definition-1 -->
<!-- example-1: body_md -->
Consider the set $\mathcal{A} = \{ e_1,\dots,e_n \} \subset \mathbb{N}^n$ and $\mathcal{B} = \{\pm e_1,\dots, \pm e_n\} \subset \mathbb{Z}^n$, where $e_i$ denotes the standard basis vector.
Then it holds
\begin{equation*}
\mathbb{R}[\mathcal{A}] = \mathbb{R}[x_1,\dots,x_n] \quad \text{ and } \quad \mathbb{R}[\mathcal{B}] = \mathbb{R}[x_1^{\pm 1},\dots,x_n^{\pm 1}].
\end{equation*}
<!-- end example-1 -->
<!-- remark-1: body_md -->
Observe that, as the exponents in \(\mathcal{A}\) might not be linear independent, \(\deg_\mathcal{A}\) is not necessarily a grading on \(\mathbb{R}[\mathcal{A}]\), that is, does not induce a valuation. Nevertheless, in this work we abuse notation and still call it graduation.
  
Given \(f,g \in \mathbb{R}[\mathcal{A}]\), we have that,
\begin{itemize}
    \item[i)] \(\deg_{\mathcal{A}}(f \cdot g) \leq \deg_{\mathcal{A}}(f) + \deg_{\mathcal{A}}(g)\),
    \item[ii)] \(\deg_{\mathcal{A}}(f+g) \leq \max\{\deg_{\mathcal{A}}(f),\deg_{\mathcal{A}}(g)\}\).
\end{itemize}

<!-- definition-2: body_md -->
For a finite set $\mathcal{A}\subset\mathbb{Z}^n$, define the (real) toric variety associated
with $\mathcal{A}$ as
\[
  \mathrm{Sper}(\mathbb{R}[\mathcal{A}])
  :=
  \bigl\{
  \mathfrak m \subset \mathbb{R}[\mathcal{A}]
  \;:\;
  \mathfrak m \text{ is a maximal real radical ideal}
  \bigr\}.
\]
For $I\subset\mathbb{R}[\mathcal{A}]$, we define its (real) variety as
\[
  \mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)
  :=
  \{\mathfrak m \in \mathrm{Sper}(\mathbb{R}[\mathcal{A}]) : I \subset \mathfrak m\}.
\]

<!-- remark-2: body_md -->
\label{rmk:homIsSper}
There is a natural identification
\[
  \mathrm{Sper}(\mathbb{R}[\mathcal{A}])
  =
  \bigl\{
  \Lambda \in \mathrm{Hom}_{\mathbb{R}\text{-alg}}(\mathbb{R}[\mathcal{A}],\mathbb{R})
  \bigr\},
\]
Every unital $\mathbb{R}$-algebra homomorphism $\Lambda:\mathbb{R}[\mathcal{A}]\to\mathbb{R}$ is
surjective and has a maximal real kernel. Conversely, if $\mathfrak m$ is a
maximal real ideal, then the residue field $\mathbb{R}[\mathcal{A}]/\mathfrak m$ is $\mathbb{R}$, and the
quotient map followed by this identification gives the corresponding
$\mathbb{R}$-algebra homomorphism.

<!-- lemma-2: body_md -->
\label{lem:Max_ideals_in_RA}
    Let $J \subset \mathbb{R}[\mathcal{A}]$ be a maximal real radical ideal. Then $J = \langle x^\alpha -\beta_\alpha : \alpha \in \mathcal{A} \rangle$ for some $\beta_\alpha \in \mathbb{R}$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Write $\mathcal{A} = \{ \alpha_1,\dots,\alpha_m \}$ and consider the surjective $\mathbb{R}$-algebra homomorphism $\psi_\mathcal{A}$ from \eqref{eq:kodairaMap}. The preimage $H := \psi_\mathcal{A}^{-1}(J)$ of $J$ is a maximal real radical ideal, since $\psi_\mathcal{A}$ is surjective.
      
    Thus, it can be written in the form $H = \langle z_i - \beta_i : i = 1,\dots,m \rangle$ for some $\beta_i \in \mathbb{R}$. Applying $\psi_\mathcal{A}$ gives the desired result.

<!-- proposition-1: body_md -->
\label{prop:toric_nullstellen}
    Let $I$ be an ideal in $\mathbb{R}[\mathcal{A}]$. Then the vanishing ideal of the real variety is the real radical of $I$:
    \[
        \mathcal{I}(\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)) = \sqrt[\mathbb{R}]{I}.
    \]
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
We shall prove that $\mathcal{I}(\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)) \subseteq \sqrt[\mathbb{R}]{I}$, as the opposite inclusion is trivial.
      
    As in the proof of Lemma~\ref{lem:Max_ideals_in_RA}, consider the preimage  $H := \psi_\mathcal{A}^{-1}(I)$ of $I$, where $\psi_\mathcal{A}$ is as defined in \eqref{eq:kodairaMap}.  
      
    From the previous lemma, we also see that there is an identification of maximal real radical ideals in $\mathbb{R}[\mathcal{A}]$ and the maximal real radical ideals of $\mathbb{R}[z_1,\dots,z_m]$ that contain $\mathrm{ker}(\psi_\mathcal{A})$.
      
    Hence, if $f$ vanishes at every $\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)$ and $g$
    is any lift of $f$ under $\psi_\mathcal{A}$, then $g$ vanishes on
    $\mathcal{V}_\mathbb{R}(H)$. By the classical real Nullstellensatz,
    $g\in\sqrt[\mathbb{R}]{H}$. Since real radicals commute with quotients under a
    surjective homomorphism,
    $\psi_\mathcal{A}(\sqrt[\mathbb{R}]{H})=\sqrt[\mathbb{R}]{I}$, and therefore
    $f\in\sqrt[\mathbb{R}]{I}$.

<!-- proposition-2: body_md -->
\label{prop:solsOnTorus}
Let $f \in \mathbb{R}[\mathcal{A}]$. Suppose there exists a point $p \in (\mathbb{R}^*)^n$ such that $f(p)=0$\footnote{The evaluation is defined by interpreting $f$ as a Laurent polynomial.}. Then there exists a real point $\Lambda \in \mathrm{Sper}(\mathbb{R}[\mathcal{A}])$ such that $\Lambda(f)=0$ and $\Lambda(x^\alpha) \neq 0$ for all $\alpha \in \mathcal{A}$. Moreover, if such a $\Lambda$ exists and the index of $\mathbb{Z}\mathcal{A}$ in $\mathbb{Z}^n$ is odd, then there exists $p \in (\mathbb{R}^*)^n$ such that $f(p)=0$.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
The map which evaluates polynomials at $p\in(\mathbb{R}^*)^n$ is a
homomorphism $\Lambda_p:\mathbb{R}[\mathcal{A}]\to\mathbb{R}$.  Its kernel is a maximal real ideal,
$\Lambda_p(f)=f(p)$, and $\Lambda_p(x^\alpha)=p^\alpha\ne0$ for every
$\alpha\in\mathcal{A}$.  This proves the first claim.

Conversely, suppose that $\Lambda(x^\alpha)\ne0$ for every $\alpha\in\mathcal{A}$.
Multiplicativity makes $\alpha\mapsto\Lambda(x^\alpha)$ a homomorphism from
the semigroup $\mathbb{N}\mathcal{A}$ to $\mathbb{R}^*$, and hence it extends uniquely to a group
homomorphism $\chi:\mathbb{Z}\mathcal{A}\to\mathbb{R}^*$.  Put the inclusion
$\mathbb{Z}\mathcal{A}\subseteq\mathbb{Z}^n$ in Smith normal form.  If its index is odd, all invariant
factors are odd, so the required roots in $\mathbb{R}^*$ exist and $\chi$ extends to
a character $\widetilde\chi:\mathbb{Z}^n\to\mathbb{R}^*$.  Setting
$p_i:=\widetilde\chi(e_i)$ gives $p\in(\mathbb{R}^*)^n$ and
$p^\alpha=\Lambda(x^\alpha)$ for all $\alpha\in\mathcal{A}$.  Consequently
$f(p)=\Lambda(f)=0$.
<!-- end proof-3 -->
<!-- example-2: body_md -->
\label{exp:Dim_not_equal_num_sol}
Consider the monomials
$x^2 y, xy^2
\in \mathbb{R}[x^2 y, x y^2] \subset \mathbb{R}[x,y]$.
The quotient ring $ \mathbb{R}[x,y] / {\sqrt[\mathbb{R}]{\langle x^2 y, x y^2 \rangle}} $ has Krull dimension $1$, while
$ \mathbb{R}[x^2 y, x y^2] / {\sqrt[\mathbb{R}]{\langle x^2 y, x y^2 \rangle}} $ has Krull dimension $0$ and defines a unique (real) point, the zero of the toric variety associated to $\mathbb{R}[x^2 y, x y^2]$.
  
Via the isomorphism from Rmk.~\ref{rmk:homIsSper}, this point correspond to $\Lambda \in \mathrm{Hom}(\mathbb{R}[x^2y,xy^2], \mathbb{R})$ such that $\Lambda(x^2 y) = \Lambda(x y^2) = 0$.
  
The discrepancy on the dimension comes from two real lines given by $x = 0$ and $y = 0$ in the variety associated to $\mathbb{R}[x,y] / {\sqrt[\mathbb{R}]{\langle x^2 y, x y^2 \rangle}} $.

Consider the monomials
$x^2, y^2 \in \mathbb{R}[x^2, y^2] \subset \mathbb{R}[x,y]$.
The quotient ring $ \mathbb{R}[x,y] / {\sqrt[\mathbb{R}]{\langle x^2 + y^2 \rangle}} $ has Krull dimension $0$ and its associated variety is the point zero in $\mathbb{R}^n$ as ${\sqrt[\mathbb{R}]{\langle x^2 + y^2 \rangle}} =  \langle x,y\rangle$ over $\mathbb{R}[x,y]$.
  
However, over $\mathbb{R}[x^2,y^2]$, the ideal $\langle x^2 + y^2 \rangle$ is real radical, so the quotient ring
$\mathbb{R}[x^2,y^2]/{\sqrt[\mathbb{R}]{\langle x^2+y^2\rangle}}$
has Krull dimension $1$.
  
We can visualize the lines contained in this variety using the isomorphism
from Rmk.~\ref{rmk:homIsSper}: for every $c\in\mathbb{R}$, there is a point
$\Lambda_c\in\mathrm{Hom}_{\mathbb{R}\text{-alg}}(\mathbb{R}[x^2,y^2],\mathbb{R})$ such that
$\Lambda_c(x^2)=-\Lambda_c(y^2)=c$. Observe that these points
admit complex lifts $(z,w)$ with $z^2=c$ and $w^2=-c$
in the complex variety associated to
$\mathbb{R}[x,y]/\langle x^2+y^2\rangle$, so real points of the toric variety can
correspond to complex solutions of the original system.

If we modify the previous example and consider the quotient ring $ \mathbb{R}[x^2,y^2] / {\sqrt[\mathbb{R}]{\langle x^4 + y^4 \rangle}}$, we observe that it corresponds to the point zero given by $\Lambda \in \mathrm{Hom}(\mathbb{R}[x^2,y^2], \mathbb{R})$, $\Lambda(x^2) = \Lambda(y^2) = 0$, similarly to the classical setting.

Let $\Lambda\in\mathbb{R}[\mathcal{A}]^*$ and let $\mathcal{B}$ be a set of monomials.
Then $\mathcal{B}$ indexes a maximal linearly independent set of columns of
$M(\Lambda)$ if and only if the residue classes of $\mathcal{B}$ form a basis of
$\mathbb{R}[\mathcal{A}]/\mathrm{ker}( M(\Lambda))$.
In particular,
\[
  \mathrm{rank}(M(\Lambda)) = \dim_\mathbb{R}\bigl(\mathbb{R}[\mathcal{A}]/\mathrm{ker}(M(\Lambda))\bigr).
\]
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
A finite linear combination among columns indexed by $\mathcal{B}$ is exactly a
polynomial $f$ in the linear span of $\mathcal{B}$ satisfying $M(\Lambda)\mathrm{vec}_\infty(f)=0$, that is,
$f\in\mathrm{ker}(M(\Lambda))$. Thus these columns are linearly independent if and only
if the residue classes of $\mathcal{B}$ are linearly independent in the quotient.
Likewise, a column indexed by a monomial $x^\alpha$ lies in the span of the
columns indexed by $\mathcal{B}$ if and only if the residue class of $x^\alpha$ lies in
the span of the residue classes of $\mathcal{B}$. The two maximality statements are
therefore equivalent, and the rank--dimension identity follows.

<!-- lemma-4: body_md -->
\label{LemmaKernelQF}
Let $\Lambda\in\mathbb{R}[\mathcal{A}]^*$.
If $\Lambda$ is positive, then the ideal $\mathrm{ker}(M(\Lambda))$ is real radical.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
First note that $\mathrm{ker}(M(\Lambda))$ is an ideal.
Assume $\Lambda$ is positive and let $\sum_{i=1}^m p_i^2 \in \mathrm{ker}(M(\Lambda))$ for some
$p_i\in\mathbb{R}[\mathcal{A}]$.
We show $p_i \in \mathrm{ker}(M(\Lambda))$ for every $i$.
For any $g\in\mathbb{R}[\mathcal{A}]$ we have
\[
  0
  =
  \Lambda\!\left(g^2 \sum_{i=1}^m p_i^2\right)
  =
  \sum_{i=1}^m \Lambda(g^2 p_i^2).
\]
Since each term $\Lambda(g^2 p_i^2)\ge 0$ by positivity, it follows that
$\Lambda(g^2 p_i^2)=0$ for all $i$ and all $g$.
It remains to show that $\Lambda(q^2)=0$ implies $\Lambda(q)=0$.
Fix $q\in\mathbb{R}[\mathcal{A}]$ with $\Lambda(q^2)=0$.
For any $t\in\mathbb{R}$,
\[
  0 \le \Lambda((q+t)^2)
  = \Lambda(q^2) + 2t\,\Lambda(q) + t^2 \Lambda(1)
  = 2t\,\Lambda(q) + t^2 \Lambda(1).
\]
This inequality for all $t\in\mathbb{R}$ forces $\Lambda(q)=0$.
  
Hence, if we consider $q=g \ p_i$, we get
$\Lambda(g \ p_i)=0$ for every $g\in\mathbb{R}[\mathcal{A}]$, and so $p_i\in\mathrm{ker}(M(\Lambda))$.

<!-- lemma-5: body_md -->
\label{LemmaEvaluationsAndKernels}
Let $\mathfrak m\subset\mathbb{R}[\mathcal{A}]$ be a maximal real radical ideal and let
$\Lambda_{\mathfrak m}$ be the corresponding element of $\mathrm{Sper}(\mathbb{R}[\mathcal{A}])$.
Then $M(\Lambda_{\mathfrak m})$ has rank $1$ and
$\mathrm{ker}(M(\Lambda_{\mathfrak m}))=\mathfrak m$.
More generally, let $\Lambda = \sum_{i=1}^r \lambda_i \Lambda_i$, where $\lambda_i\neq 0$ and $\Lambda_i\in\mathrm{Sper}(\mathbb{R}[\mathcal{A}])$ are pairwise distinct. Then $\mathrm{rank}(M(\Lambda))=r$.
If, in addition, all $\lambda_i>0$, then $\mathrm{ker}(M(\Lambda)) = \bigcap_{i=1}^r \mathrm{ker}(\Lambda_i)$.
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
It is immediate that $M(\Lambda_{\mathfrak m})$ has rank $1$.
Moreover,
\[
  \mathfrak m \subseteq \mathrm{ker}(M(\Lambda_{\mathfrak m})) \subsetneq \mathbb{R}[\mathcal{A}],
\]
and maximality of $\mathfrak m$ implies $\mathrm{ker}(M(\Lambda_{\mathfrak m}))=\mathfrak m$.

For the second claim, put $\mathfrak m_i=\mathrm{ker}(\Lambda_i)$. Distinct
maximal ideals are pairwise comaximal, so the Chinese remainder theorem gives
$p_1,\dots,p_r\in\mathbb{R}[\mathcal{A}]$ with $\Lambda_j(p_i)=\delta_{ij}$. In the bases induced by
these interpolation polynomials, the bilinear form
$(f,g)\mapsto\Lambda(fg)$ has diagonal matrix
$\operatorname{diag}(\lambda_1,\dots,\lambda_r)$. Since every $\lambda_i$ is
nonzero, this form has rank $r$, and so does $M(\Lambda)$.
Further, if each $M(\Lambda_i)\succeq 0$ and
$\Lambda=\sum_{i=1}^r \lambda_i\Lambda_i$ is a conic
combination of the $\Lambda_i$ with $\lambda_i>0$, we obtain
\[
  \mathrm{ker}(M(\Lambda))=\bigcap_{i=1}^r \mathrm{ker}(M(\Lambda_i))=\bigcap_{i=1}^r \mathrm{ker}(\Lambda_i).
\]
The equality follows form the identity 
\[
  \mathrm{ker}(\lambda A+\mu B) = \mathrm{ker}(A)\cap\mathrm{ker}(B),
\]
 for $\lambda,\mu >0$ and positive semidefinite matrices $A,B$.

<!-- lemma-6: body_md -->
\label{lem:Real_Nullstellensatz_Toric_variety}
Let $I\subset\mathbb{R}[\mathcal{A}]$ be a zero-dimensional real radical ideal, i.e.,
$|\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)|<\infty$.
Then
\[
  \dim(\mathbb{R}[\mathcal{A}]/I) = |\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)|
  \qquad\text{and}\qquad
  \mathbb{R}[\mathcal{A}]/I \cong \prod_{i=1}^{\dim(\mathbb{R}[\mathcal{A}]/I)} \mathbb{R}.
\]
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
Since $I$ is zero-dimensional, $\mathbb{R}[\mathcal{A}]/I$ is Artinian; see \cite[p.~92]{atiyah2018introduction}.
Hence $\mathbb{R}[\mathcal{A}]/I$ is finite-dimensional as an $\mathbb{R}$-vector space; write
$r:=\dim(\mathbb{R}[\mathcal{A}]/I)$.
Moreover, $\mathbb{R}[\mathcal{A}]/I$ is reduced because $I$ is real radical.
By \cite[Theorem~8.7]{atiyah2018introduction}, we obtain an isomorphism
\[
  \mathbb{R}[\mathcal{A}]/I \cong K_1 \times \dots \times K_l,
\]
where each $K_i$ is a finite field extension of $\mathbb{R}$.
Since $\mathbb{R}\subseteq \mathbb{R}[\mathcal{A}]/I$, each $K_i$ is a finite-dimensional field extension of
$\mathbb{R}$, hence $K_i\in\{\mathbb{R},\mathbb{C}\}$.
No factor can be $\mathbb C$: in such a factor the two nonzero
elements $1$ and $i$ have squares summing to zero. Multiplying them by the
idempotent supported on that factor would contradict the real-reduced
property of $\mathbb{R}[\mathcal{A}]/I$.
Thus $K_i=\mathbb{R}$ for all $i$, and therefore $l=r$ and
$\mathbb{R}[\mathcal{A}]/I\cong \mathbb{R}^r$.

Finally, the maximal ideals of $\mathbb{R}[\mathcal{A}]$ containing $I$ correspond to the coordinate
projections of $\mathbb{R}^r$, hence $|\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)|=r=\dim(\mathbb{R}[\mathcal{A}]/I)$.

<!-- theorem-3: body_md -->
\label{TheoremKernMRealRadicalIffLinearCombination}
\label{TheoremFiniteRankMomentMatrix}
Let $\Lambda\in\mathbb{R}[\mathcal{A}]^*$ be such that $\mathrm{rank}(M(\Lambda))=r<\infty$.
Then the following are equivalent:
\begin{enumerate}
  \item[i)] $\mathrm{ker}(M(\Lambda))$ is real radical.
  \item[ii)] $\Lambda = \sum_{i=1}^r \lambda_i \Lambda_i$, where
  $\lambda_i\in\mathbb{R}\setminus\{0\}$ and
  $\Lambda_i\in\mathrm{Sper}(\mathbb{R}[\mathcal{A}])$ are pairwise distinct.
\end{enumerate}
Furthermore, if $\Lambda$ is positive, then $\lambda_i>0$ and $\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(\mathrm{ker}(M(\Lambda))) = \{\Lambda_1,\dots,\Lambda_r\}$.
<!-- end theorem-3 -->
<!-- proof-8: body_md -->
Put $J:=\mathrm{ker}(M(\Lambda))$. If $J$ is real radical, then
Lemma~\ref{lem:rank_equals_dim_cooredinate_ring} and
Lemma~\ref{lem:Real_Nullstellensatz_Toric_variety} give an isomorphism
$\Theta:\mathbb{R}[\mathcal{A}]/J\to\mathbb{R}^r$. The functional induced by $\Lambda$ on $\mathbb{R}[\mathcal{A}]/J$ is
therefore of the form
\[
 \overline\Lambda=\sum_{i=1}^r\lambda_i\pi_i,
\]
where $\pi_i$ is the $i$-th coordinate projection. Every $\lambda_i$ is
nonzero, since the bilinear form induced by $M(\Lambda)$ on $\mathbb{R}[\mathcal{A}]/J$ is
nondegenerate and has diagonal matrix
$\operatorname{diag}(\lambda_1,\dots,\lambda_r)$ in the standard basis.
Pulling the coordinate projections back through $\Theta$ gives (ii).

Conversely, suppose that (ii) holds and put
$\mathfrak m_i:=\mathrm{ker}(\Lambda_i)$. Then
$\bigcap_i\mathfrak m_i\subseteq J$. The Chinese remainder theorem and
Lemma~\ref{lem:rank_equals_dim_cooredinate_ring} give
\[
 \dim\bigl(\mathbb{R}[\mathcal{A}]/\textstyle\bigcap_i\mathfrak m_i\bigr)=r
 =\dim(\mathbb{R}[\mathcal{A}]/J),
\]
so $J=\bigcap_i\mathfrak m_i$ and is real radical.

Finally, if $\Lambda$ is positive, then
$\lambda_i=\overline\Lambda(e_i)=\overline\Lambda(e_i^2)>0$. The last claim follows
from Lemma~\ref{LemmaEvaluationsAndKernels}.

<!-- theorem-5: body_md -->
\label{TheoremSparseFlatExtension}
Let $s\geq\rho_\mathcal{A}$ and let $\Lambda\in\mathbb{R}[\mathcal{A}]_{2s}^*$ satisfy
\[
 M_s(\Lambda)\succeq0,
 \qquad
 \mathrm{rank} (M_s(\Lambda))=\mathrm{rank} (M_{s-1}(\Lambda))=:r.
\]
Then there is a unique $\widetilde\Lambda\in\mathbb{R}[\mathcal{A}]^*$ such that
\[
 \widetilde\Lambda|_{\mathbb{R}[\mathcal{A}]_{2s}}=\Lambda,
 \qquad M(\widetilde\Lambda)\succeq0,
 \qquad \mathrm{rank} (M(\widetilde\Lambda))=r.
\]
<!-- end theorem-5 -->
<!-- proof-10: body_md -->
Let $K_s:=\mathrm{ker}(M_s(\Lambda))$. As $\mathrm{rank}(M_s(\Lambda))=\mathrm{rank}(M_{s-1}(\Lambda))$, we can choose a monomial column basis
$\mathcal{B}\subseteq\mathbb{R}[\mathcal{A}]_{s-1}$ of $M_s(\Lambda)$.
  
We consider the $\mathbb{R}$-vector space $V$ spanned by the monomials in $\mathcal{B}$.
  
Every element in $\mathbb{R}[\mathcal{A}]_s$ has a projection to $V$ given by the map be $\pi:\mathbb{R}[\mathcal{A}]_s\longrightarrow V$ which sends each $p\in\mathbb{R}[\mathcal{A}]_s$ to its unique projection $\pi(p)\in V$ characterized by
\begin{equation}
  p-\pi(p)\in K_s.                                  
  \label{eq:direct-reduction}
\end{equation}
Indeed, the residue classes of the monomials in $\mathcal{B}$ form a basis of $\mathbb{R}[\mathcal{A}]_s/K_s$: they are linearly independent by the choice of $\mathcal{B}$, and both spaces have dimension $r$. Thus $\mathbb{R}[\mathcal{A}]_s=V\oplus K_s$.
  
Observe that, if $h\in K_s\cap\mathbb{R}[\mathcal{A}]_{s-1}$, then
\begin{equation}
 x^{\alpha_i} \ h\in K_s.                                        
 \label{eq:direct-truncated-ideal}
\end{equation}
Indeed, if $q=\pi(q)+p\in\mathbb{R}[\mathcal{A}]_s$ with $p\in K_s$, then
$\Lambda((x^{\alpha_i}  \ h)\ q)
 =\Lambda\bigl(h \ x^{\alpha_i} \ \pi(q)\bigr)+\Lambda\bigl((x^{\alpha_i} \ h) \ p\bigr)=0.$ 
Here the first term vanishes because $h\in K_s$ and $x^{\alpha_i} \, \pi(q)\in\mathbb{R}[\mathcal{A}]_s$, while the second vanishes because $p \in K_s$ and $x^{\alpha_i} \, h\in\mathbb{R}[\mathcal{A}]_s$.
   
                              
  
                                                                

Given $p,q\in V$, consider the bilinear form
$\langle p,q\rangle_\Lambda:=\Lambda(pq)$. This is an inner product on $V$.
In fact, its Gram matrix is the principal submatrix of $M_s(\Lambda)$
indexed by $\mathcal{B}$.
  
This inner product is nondegenerate and its Gram matrix is positive definite because
$M_s(\Lambda)\succeq0$ and the corresponding columns are linearly independent.
  

For $i=1,\dots,m$, we define the multiplication maps
\begin{equation}
  X_i:V\longrightarrow V,\qquad X_i (p) :=\pi(x^{\alpha_i} \, p).       
  \label{eq:direct-multiplication}
\end{equation}
With respect to the inner product induced by $\Lambda$, these maps are self-adjoint,
\[
 \langle X_i (p),q\rangle_\Lambda
 =\Lambda(\pi(x^{\alpha_i} \, p) \ q) = \Lambda(x^{\alpha_i} \, p \, q)
 =\langle p,X_i(q)\rangle_\Lambda, \]
They also commute.  For $p,q\in V$,
\begin{align*}
 \langle X_i(X_j(p)),q\rangle_\Lambda
 =\langle X_j(p),X_i(q)\rangle_\Lambda
   =\Lambda(x^{\alpha_j} \, p \ x^{\alpha_i} \, q)
 =\Lambda(x^{\alpha_i} \, p \ x^{\alpha_j} \, q)
   =\langle X_j(X_i(p)),q\rangle_\Lambda.
\end{align*}
The nondegeneracy of the inner product gives $X_i \circ X_j=X_j \circ X_i$.

Let $e:=\pi(1)\in V$.  For $k \leq s$ and $w \in \{1\dots m\}^k$, define $x^w= \prod_{j = 1}^k x^{\alpha_{w_j}}$ and $X^w=X_{w_1} \circ \cdots \circ X_{w_k}$.
  
We claim that
\begin{equation}
  X^w(e)=\pi(x^w).
  \label{eq:direct-word-reduction}
\end{equation}
This can be proved by induction on $k$. The case $k=0$ is immediate. Inductively, if $x^{(w_1,\dots,w_{k-1})}-X^{(w_1,\dots,w_{k-1})}(e) \in K_s\cap\mathbb{R}[\mathcal{A}]_{s-1}$, by \eqref{eq:direct-truncated-ideal}, we have that
$x^{\alpha_{w_k}} (x^{(w_1,\dots,w_{k-1})}-X^{(w_1,\dots,w_{k-1})}(e)) \in K_s$.
Consequently, commutativity of the multiplication maps completes the induction as 
\begin{multline*}
    \pi(x^{(w_1,\ldots,w_k)})= \pi(x^{\alpha_{w_k}} \ x^{(w_1,\dots,w_{k-1})}) = \pi(x^{\alpha_{w_k}} \ X^{(w_1,\ldots,w_{k-1})}(e)) \\ =X_{w_k} \circ X^{(w_1,\ldots,w_{k-1})}(e) = X^{(w_1,\dots,w_{k})}(e).
\end{multline*}

The multiplication maps induce an algebra homomorphism $\Phi : \mathbb{R}[x_1,\dots,x_m] \rightarrow \operatorname{End}_\mathbb{R}(V)$ sending $x_i \mapsto \Phi(x_i) := X_i$.
  
Since $\mathbb{R}[\mathcal{A}]\simeq\mathbb{R}[x_1,\dots,x_m]/T_\mathcal{A}$, it remains to prove that the multiplication maps respect the relations in $T_\mathcal{A}=\mathrm{ker}(\psi_\mathcal{A})$.
  
Recall $G_\mathcal{A}$ is a generating set of $T_\mathcal{A}$.
   
Fix $g\in G_\mathcal{A}$ and $b \in \mathcal{B}$.
  
By \eqref{eq:direct-word-reduction}, there is $k\leq s-1$ and $w(b)\in \{1\dots m\}^k$ such that $b=X^{w(b)}(e)$.
  
Moreover, every monomial of $g$ has degree at most
$\rho_\mathcal{A}\leq s$, so by
\eqref{eq:direct-word-reduction} and $\psi_\mathcal{A}(g)=0$ we have that
\begin{multline*}
  \Phi(g) (b) = g(X_1,\dots,X_m) (b) = g(X_1,\dots,X_m) \circ X^{w(b)}(e) \\ =
   X^{w(b)} \circ g(X_1,\dots,X_m) (e) = 
  X^{w(b)}(\pi(\psi_\mathcal{A}(g))) = 0.
\end{multline*}
Thus $g(X_1,\dots,X_m)=0$ on $V$.  This holds for all $g\in G_\mathcal{A}$, and hence
for the whole toric ideal $T_\mathcal{A}$ as wanted.
  
Therefore, $\Phi$ factors through the quotient by $T_\mathcal{A}=\mathrm{ker}(\psi_\mathcal{A})$. 
We abuse notation and consider $\Phi : \mathbb{R}[\mathcal{A}]\simeq\mathbb{R}[x_1,\dots,x_m]/T_\mathcal{A} \rightarrow \operatorname{End}_\mathbb{R}(V)$. Observe that the image of $\Phi$ at any monomial in $\mathbb{R}[\mathcal{A}]$ does not depend on the chosen factorization of that monomial.

We now define the flat extension $\widetilde\Lambda \in \mathbb{R}[\mathcal{A}]^*$  as \[\widetilde\Lambda(f):=\Lambda\bigl(\Phi(f)(e)\bigr).\]
  
First observe that, as $1-e\in K_s$, for every $v\in V$,
$\Lambda(v)=\langle v,e\rangle_\Lambda.$
Moreover, since the maps $X_i$ commute and are self-adjoint, $\Phi(f)$ is self-adjoint for every $f\in\mathbb{R}[\mathcal{A}]$. It follows that, for all $f,g\in\mathbb{R}[\mathcal{A}]$,
\begin{equation}\label{eq:direct-full-gram}
    \widetilde\Lambda(fg) = \Lambda(\langle\Phi(fg)(e)) 
    =\bigl\langle\Phi(fg)(e),e\bigr\rangle_\Lambda
    =\bigl\langle\Phi(f)(e),\Phi(g)(e)\bigr\rangle_\Lambda.
\end{equation}
We now show that $\widetilde\Lambda$ extends $\Lambda$. 
  
Let $x^\tau\in\mathbb{R}[\mathcal{A}]_{2s}$ be a monomial. We can write $x^\tau=x^\alpha x^\beta$ with $x^\alpha,x^\beta\in\mathbb{R}[\mathcal{A}]_s$. By \eqref{eq:direct-word-reduction},
$\Phi(x^\alpha)(e)=\pi(x^\alpha)$ and $\Phi(x^\beta)(e)=\pi(x^\beta)$.
Therefore,
\[
    \widetilde\Lambda(x^\tau)
    =\bigl\langle\pi(x^\alpha),\pi(x^\beta)\bigr\rangle_\Lambda
    =\Lambda(x^\alpha x^\beta)
    =\Lambda(x^\tau),
\]
where the middle equality follows because $x^\alpha-\pi(x^\alpha)$ and $x^\beta-\pi(x^\beta)$ belong to $K_s$. By linearity, $\widetilde\Lambda|_{\mathbb{R}[\mathcal{A}]_{2s}}=\Lambda$.

Equation \eqref{eq:direct-full-gram} shows that the full moment matrix $M(\widetilde\Lambda)$ is the Gram matrix of the vectors $\{\Phi(x^\alpha)(e):x^\alpha\in\mathbb{R}[\mathcal{A}]\}$ in $V$ with respect to the inner product $\langle \bullet,\bullet \rangle_\Lambda$. Hence $M(\widetilde\Lambda)\succeq0$ and $\mathrm{rank} (M(\widetilde\Lambda))\leq\dim V=r$. On the other hand, if $b\in\mathcal{B}$, then \eqref{eq:direct-word-reduction} gives $\Phi(b)(e)=b$. Since $\mathcal{B}$ is a basis of $V$, these vectors are linearly independent, and therefore $\mathrm{rank} (M(\widetilde\Lambda))=r$. Thus, $\widetilde\Lambda$ is a flat extension of $\Lambda$.

For uniqueness, let $\widehat\Lambda\in\mathbb{R}[\mathcal{A}]^*$ be another positive semidefinite extension of $\Lambda$ with $\mathrm{rank}(M(\widehat\Lambda))=r$. The columns indexed by $\mathcal{B}$ remain linearly independent and hence form a basis of the column space of $M(\widehat\Lambda)$. For $b\in\mathcal{B}$ and $i=1,\ldots,m$, set $q_{i,b}:=x^{\alpha_i} \, b-X_i(b)\in K_s$. Since $\widehat\Lambda$ extends $\Lambda$ and is positive semidefinite,
\[
 \widehat\Lambda(q_{i,b}^2)=\Lambda(q_{i,b}^2)=0
 \quad\Longrightarrow\quad q_{i,b}\in\mathrm{ker}(M(\widehat\Lambda)).
\]
The same argument gives $1-e\in\mathrm{ker}(M(\widehat\Lambda))$. Thus, in the basis $\mathcal{B}$, multiplication by $x^{\alpha_i}$ is represented by $X_i$ and the column of $1$ is represented by $e$. Consequently, the column of every monomial $x^\alpha$ is represented by $\Phi(x^\alpha)(e)$, and
\[
 \widehat\Lambda(x^\alpha)=\bigl\langle\Phi(x^\alpha)(e),e\bigr\rangle_\Lambda
 =\widetilde\Lambda(x^\alpha).
\]
By linearity, $\widehat\Lambda=\widetilde\Lambda$.

\label{CorollaryDirectFlatKernel}

\[
 \mathrm{ker} (M(\widetilde\Lambda))=\langle\mathrm{ker} (M_s(\Lambda))\rangle.

<!-- proof-11: body_md -->

Since the maps in the image of $\Phi$ commute, $\Phi(p)(\Phi(f)(e))=\Phi(f)(\Phi(p)(e))=0$ for every $f$, and therefore $J\subseteq\mathrm{ker}(\Phi)$.

Conversely, the relations $1-e\in K_s$ and
\[
 x^{\alpha_i} \, b-X_i(b)=x^{\alpha_i} \, b-\pi(x^{\alpha_i} \, b)\in K_s\qquad(b\in\mathcal{B})
\]
reduce every monomial modulo $J$ to an element of $V$. Hence $\mathbb{R}[\mathcal{A}]=V+J$. If $v\in V\cap J$, then $v=\Phi(v)(e)=0$, so $\mathbb{R}[\mathcal{A}]=V\oplus J$ and $\mathrm{ker}(\Phi)=J$. Finally, by \eqref{eq:direct-full-gram} and the same spanning argument,
\[
 f\in\mathrm{ker}(M(\widetilde\Lambda))
 \iff \Phi(f)(e)=0
 \iff \Phi(f)=0.
\]
Thus $\mathrm{ker}(M(\widetilde\Lambda))=J$, and the residue classes of $\mathcal{B}$ form a basis of $\mathbb{R}[\mathcal{A}]/J$.

<!-- remark-4: body_md -->
In \cite{laurent2009generalized}, Laurent and Mourrain consider a finite
monomial set connected to $1$ in a polynomial ring.  This means that every
nonconstant monomial in the set can be obtained from $1$ by successive

<!-- lemma-8: body_md -->
\label{LemmaGenericLFinKt}
Let $\Lambda\in\mathcal{K}_t$. The following are equivalent:
\begin{enumerate}
\item[(i)] $M_{\lfloor t/2\rfloor}(\Lambda)$ has maximum rank among the moment
matrices associated with elements of $\mathcal{K}_t$;
\item[(ii)] $\mathrm{ker}(M_{\lfloor t/2\rfloor}(\Lambda)) \subseteq
\mathrm{ker}(M_{\lfloor t/2\rfloor}(\Lambda'))$ for every $\Lambda'\in\mathcal{K}_t$;
\item[(iii)] $\Lambda\in\operatorname{relint}(\mathcal{K}_t)$.
\end{enumerate}
<!-- end lemma-8 -->
<!-- proof-13: body_md -->
Write $M=M_{\lfloor t/2\rfloor}(\Lambda)$ and
$N=M_{\lfloor t/2\rfloor}(\Lambda')$. Since both are positive semidefinite,
\[
 \mathrm{ker}(M+N)=\mathrm{ker}(M)\cap\mathrm{ker}(N).
\]
If $M$ has maximum rank, then $(M+N)/2 \in \mathcal{K}_t$ cannot have larger
rank, so its kernel must equal $\mathrm{ker} (M)$; hence $\mathrm{ker} (M)\subseteq\mathrm{ker} (N)$.
The converse follows immediately from the rank--nullity theorem.

Assume (ii). 
                                                                               
$N$ vanishes on $\mathrm{ker}(M)$, because $\mathrm{ker} (M)\subseteq\mathrm{ker} (N)$. Thus, for sufficiently small
$\varepsilon>0$,
\[
 (1+\varepsilon)M-\varepsilon N\succeq0.
\]
All affine constraints defining $\mathcal{K}_t$ are preserved by the corresponding
functional $\Lambda+\varepsilon(\Lambda-\Lambda')$. The standard segment
characterization of relative interior now gives (iii). Conversely, if
$\Lambda$ is in the relative interior, the extension $\Lambda+\varepsilon_{\Lambda'}(\Lambda-\Lambda')$ is contained in $\mathcal{K}_t$ for 
every $\Lambda'\in\mathcal{K}_t$ and small enough $\varepsilon_{\Lambda'}$. For $u\in\mathrm{ker} (M)$, positivity of
$(1+\varepsilon)M-\varepsilon N$ gives
$0\le-\varepsilon u^TNu$, hence $Nu=0$. Therefore (ii) holds.

<!-- lemma-9: body_md -->
\label{lem:eventual-radical-kernel}
Let $F\subseteq\sqrt[\mathbb{R}]{I}$ be finite. There is $t_0$ such that, for every
$t\ge t_0$ and $s \in \mathbb{N}$, we have that
\[
 \{ x^\alpha \ f : f \in F, x^\alpha \in \mathbb{R}[\mathcal{A}]_s\} \subseteq\mathrm{ker} (M_{\lfloor t/2\rfloor + s}(\Lambda)) \qquad (\forall \Lambda\in\mathcal{K}_{t+2s}).
\]
<!-- end lemma-9 -->
<!-- proof-14: body_md -->
If $F=\emptyset$, the result is immediate, so assume that
$F\ne\emptyset$. Fix $s\in\mathbb{N}$ and $\Lambda\in\mathcal{K}_{t+2s}$, and put
$r=\lfloor t/2\rfloor$. We first record the truncated kernel rule used
below. If $p\in\mathrm{ker} (M_r(\Lambda))$ and
$\deg_\mathcal{A}(p)+\deg_\mathcal{A}(q)\le r-1$, then
$pq\in\mathrm{ker} (M_r(\Lambda))$.
To prove this directly, first let $q=x^{\alpha_i}$. Since
$x^{2\alpha_i}p\in\mathbb{R}[\mathcal{A}]_r$, we have
\[
 \Lambda\bigl((x^{\alpha_i}p)^2\bigr)
 =\Lambda\bigl(p\,x^{2\alpha_i}p\bigr)=0.
\]
Positivity of $M_r(\Lambda)$ gives
$x^{\alpha_i}p\in\mathrm{ker} (M_r(\Lambda))$. Iterating this argument proves the
claim when $q$ is a monomial, and the general case follows by linearity.

Fix $f\in F$. By Lemma~\ref{lem:Real_Nullstellensatz_Toric_variety} there are
$m\ge1$ and fixed
polynomials $q_j,u_i\in\mathbb{R}[\mathcal{A}]$ such that
\[
 f^{2m}+\sum_jq_j^2=\sum_i u_i f_i.
\]
For all sufficiently large $t$, the prolongation constraints imply
$f_i\in\mathrm{ker} (M_r(\Lambda))$: every product of $f_i$ with a row monomial of
$M_r(\Lambda)$ belongs to $\mathcal H_t\subseteq\mathcal H_{t+2s}$.
Increasing $t$ once more, the truncated
kernel rule gives $u_i f_i\in\mathrm{ker} (M_r(\Lambda))$. Hence the left-hand side of the
display lies in that kernel. Evaluating it at the row $1$ and using positivity
shows $\Lambda(f^{2m})=0$, so $f^m\in\mathrm{ker} (M_r(\Lambda))$. If $m>1$, put
$k=\lceil m/2\rceil$. Since $f^m$ is in the kernel, testing it against
$f^{2k-m}$ gives $\Lambda(f^{2k})=0$, and positivity gives
$f^k\in\mathrm{ker} (M_r(\Lambda))$. Repeating this descent yields
$f\in\mathrm{ker} (M_r(\Lambda))$. All polynomials used here are fixed, so one common
$t_0$ works for the finite set $F$.
Enlarge this $t_0$, if necessary, so that
$\left\lfloor\frac{t_0}{2}\right\rfloor
 \ge 1+\max_{f\in F}\deg_\mathcal{A}(f).$
Put $R:=\lfloor(t+2s)/2\rfloor=r+s$. For every $f\in F$, the inclusion
$f\in\mathrm{ker} (M_r(\Lambda))$ gives $\Lambda(f^2)=0$. Since
$M_R(\Lambda)\succeq0$, it follows that $f\in\mathrm{ker} (M_R(\Lambda))$. Moreover, for
every monomial $x^\alpha\in\mathbb{R}[\mathcal{A}]_s$,
\[
 \deg_\mathcal{A}(f)+\deg_\mathcal{A}(x^\alpha)
 \le \left\lfloor\frac{t_0}{2}\right\rfloor-1+s
 \le R-1.
\]
Applying the truncated kernel rule at order $R$ gives
$x^\alpha f\in\mathrm{ker} (M_R(\Lambda))$, which is the desired conclusion.

<!-- theorem-7: body_md -->
\label{TheoremNtEqualsRealRadicalWithRankCondition}
Let $D=\max_i\deg_\mathcal{A}(f_i)$ and $d=\lceil D/2\rceil$. Let
$\Lambda\in\mathcal{K}_t$ and suppose that either
\begin{enumerate}
\item[(i)] $\mathrm{rank} (M_s(\Lambda))=\mathrm{rank} (M_{s-1}(\Lambda))$ for some
$\max\{D,\rho_\mathcal{A}\}\le s\le\lfloor t/2\rfloor$, or
\item[(ii)] $\mathrm{rank} (M_s(\Lambda))=\mathrm{rank}(M_{s-d}(\Lambda))$ for some
$\max\{d,\rho_\mathcal{A}\}\le s\le\lfloor t/2\rfloor$.
\end{enumerate}
Then
\[
 \sqrt[\mathbb{R}]{I}\subseteq J:=\langle\mathrm{ker} (M_s(\Lambda)) \rangle.
\]
If $\Lambda$ is generic in $\mathcal{K}_t$, then $J=\sqrt[\mathbb{R}]{I}$, and every monomial set
indexing a column basis of $M_{s-1}(\Lambda)$ is a basis of
$\mathbb{R}[\mathcal{A}]/\sqrt[\mathbb{R}]{I}$.
<!-- end theorem-7 -->
<!-- proof-16: body_md -->
Since the ranks of the moment matrices are nondecreasing, either condition
implies
$\mathrm{rank} (M_s(\Lambda))=\mathrm{rank} (M_{s-1}(\Lambda))$.
Thus, Theorem~\ref{TheoremSparseFlatExtension} and
Corollaries~\ref{CorollaryDirectFlatKernel} and
\ref{CorollaryDirectFlatAtomic} show that $J$ is real radical and that the
flat extension of $\Lambda$ has the form
\[
 \widetilde\Lambda=\sum_{j=1}^r\lambda_j\Lambda_j,
 \qquad \lambda_j>0,
 \qquad \mathcal V_{\mathbb{R}[\mathcal{A}]}(J)=\{\Lambda_1,\dots,\Lambda_r\}.
\]

Under (i), for every $q\in\mathbb{R}[\mathcal{A}]_s$ we have
$\deg_\mathcal{A}(qf_i)\le s+D\le2s\le t$, so the prolongation constraints give
$f_i\in\mathrm{ker} (M_s(\Lambda))$. Hence $I\subseteq J$ and
$\sqrt[\mathbb{R}]{I}\subseteq J$.

Under (ii), choose a monomial column basis
$\mathcal{B}\subseteq\mathbb{R}[\mathcal{A}]_{s-d}$. By Corollary~\ref{CorollaryDirectFlatKernel},
$\mathcal{B}$ is a basis of $\mathbb{R}[\mathcal{A}]/J$, so there are interpolation polynomials
$p_1,\dots,p_r\in\mathrm{lin}_\mathbb{R}(\mathcal{B})$ such that $\Lambda_k(p_j)=\delta_{jk}$. Since
\[
 \deg_\mathcal{A}(p_j^2f_i)\le2(s-d)+D\le2s\le t,
\]
the prolongation constraints give
$0=\Lambda(p_j^2f_i)=\lambda_j\Lambda_j(f_i)$. Hence
$\mathcal V_{\mathbb{R}[\mathcal{A}]}(J)\subseteq\mathcal V_{\mathbb{R}[\mathcal{A}]}(I)$, and therefore
$\sqrt[\mathbb{R}]{I}\subseteq J$.

If $\Lambda$ is generic, Lemma~\ref{LemmaGenericLFinKt}, applied to the truncated
evaluations at the points of $\mathcal V_{\mathbb{R}[\mathcal{A}]}(I)$, gives
$\mathrm{ker} (M_s(\Lambda))\subseteq\sqrt[\mathbb{R}]{I}$. Hence
$J=\sqrt[\mathbb{R}]{I}$, and the last statement follows from
Corollary~\ref{CorollaryDirectFlatKernel}.

\label{TheoremCheckingExistenceOfRoots}

$1+\sum_j\Lambda(q_j^2)=0$, contradicting positivity. Conversely, a point of the
variety gives a truncated linear form in every $\mathcal{K}_t$.

 x^\delta k_\alpha\in
\mathrm{ker}\bigl(M_{\lfloor(t+2s_0)/2\rfloor+s-s_0}(\Lambda)\bigr)
 =\mathrm{ker}\bigl(M_{\lfloor t/2\rfloor+s}(\Lambda)\bigr)
\]
for every monomial

For $\mathcal{A}=\{e_1,e_2,e_3\} \subset \mathbb{N}^3$,  we have $\mathbb{R}[\mathcal{A}]=\mathbb{R}[x,y,z]$ is

Their homogenizations with respect to a new variable $w$ vanish on the
projective line $x=w=0$.

The real variety of this system consists only of the origin. We
claim that $\mathcal{K}_t$ is not a polyhedron for any $t\ge3$. Put
$d=\lfloor t/2\rfloor$ and let $L_t\subseteq\mathbb{R}[\mathcal{A}]_t^*$ be the affine subspace
of the functionals $\Lambda$ such that $\Lambda(1)=1$ and which vanish on every
nonconstant monomial except those of degree $2d$ in $y,z$.

For $\Lambda\in L_t$, we have that $\Lambda(f) = 0$ for every $f \in \mathcal{H}_t$ as
no polynomial in this vector space contain the monomials $1$ or any
monomial of degree $2d$ only involving $y$ and $z$.
  
Moreover, up to zero rows and columns,
\[
 M_d(\Lambda)=1\oplus H_d(\Lambda),
 \qquad \text{where }
 H_d(\Lambda)=
 \bigl(\Lambda(y^{2d-i-j}z^{i+j})\bigr)_{0\le i,j\le d}.
\]
Hence $L_t\cap\mathcal{K}_t$ is affinely isomorphic to the cone of positive
semidefinite Hankel matrices of size $d+1$. This cone is not
polyhedral: it is a classical result that it
has infinitely many distinct extreme rays given by the positive
semidefinite Hankel matrices of rank one,
\[
 \mathbb{R}_{\ge0} \cdot (V_d(u,v) \, V_d(u,v)^{\mathsf T}),
 \qquad
\text{where } V_d(u,v) := (u^d,u^{d-1}v,\dots,v^d)^{\mathsf T},
 \qquad \text{and } [u:v]\in\mathbb P^1(\mathbb{R}).
\]
Since the intersection of a polyhedron with an affine subspace is a
polyhedron, $\mathcal{K}_t$ is not a polyhedron for any $t\ge3$.

<!-- remark-8: body_md -->
We observe that this approach is closely related to  Lasserre’s hierarchy  \cite{lasserre2001global} for solving polynomial optimization problems over semialgebraic sets. Indeed, Lasserre’s hierarchy is based on the infinite-dimensional cone of moment functionals (in our case, $\mathcal{K}$), and considers its truncations to moments of degree at most $2t$ (which we called $\mathcal{K}_t$). This yields a sequence of semidefinite relaxations providing increasing lower bounds for the minimum of a polynomial function, since not every element of $\mathcal{K}_t$ admits an extension to an element of $\mathcal{K}$.
  
When the semialgebraic set is a set of points, Theorem~\ref{thm:KtIsPolytope} implies finite convergence for every fixed objective $c\in\mathbb{R}[\mathcal{A}]_{2s}$: if $s\geq s_0$ and $t\geq t_0+2s$, then
\[
 \min_{\Lambda\in\mathcal{K}_t}\Lambda(c)=\min_{\Lambda\in\mathcal{K}}\Lambda(c).
\]
  
Indeed, the objective depends only on the restriction to
$\mathbb{R}[\mathcal{A}]_{2s}$, and Theorem~\ref{thm:KtIsPolytope} gives
$\pi_{2s}(\mathcal{K}_t)=\pi_{2s}(\mathcal{K})$. Notice that this does not assert that every
element of $\mathcal{K}_t$ extends to an element of $\mathcal{K}$; only its restriction to
$\mathbb{R}[\mathcal{A}]_{2s}$ admits such an extension.
  
If $c$ has a unique minimizer on the real variety, the restriction
to $\mathbb{R}[\mathcal{A}]_{2s}$ of every optimizer in $\mathcal{K}_t$ is the evaluation at this point.

\label{prop:unique-nonnegative-solution}
Let
$ \mathcal V_{\mathbb{R}[\mathcal{A}]}(I)=\{\mathfrak m_+,\mathfrak m_1,\dots, \mathfrak m_{r-1}\} $
and define, for each solution, the vectors

 v(\mathfrak m):=
 \bigl(\Lambda_{\mathfrak m}(x^{\alpha_1}),\dots,
 \Lambda_{\mathfrak m}(x^{\alpha_m})\bigr)\in\mathbb{R}^m.
\]
   
Assume that $v(\mathfrak m_+)\ge0$ coordinatewise and that every
$v(\mathfrak m_j)$ has a negative coordinate. Then there is
$c\in\mathbb{R}[\mathcal{A}]_{r-1}$ with nonnegative coefficients such that
\[
 \Lambda_{\mathfrak m_+}(c)>
 \Lambda_{\mathfrak m_j}(c),
 \qquad j=1,\dots,r-1.
\]
Moreover, $c$ may be chosen with all its coefficients strictly positive.
<!-- end proposition-3 -->
<!-- proof-22: body_md -->
For every $j \in \{1,\dots,r-1\}$, choose $k(j) \in \{1,\dots,m\}$
  such that the $k(j)$-th coordinate of $v(\mathfrak m_j)$ is
  negative. Then, we define
  \[
    c :=
 \prod_{j=1}^{r-1}
 \left(x^{\alpha_{k(j)}} -v(\mathfrak m_j)_{k(j)}\right) \in\mathbb{R}[\mathcal{A}]_{r-1}.
\]
  
The factor indexed by $j$ vanishes at $\mathfrak m_j$, and hence
$\Lambda_{\mathfrak m_j}(c)=0$. On the other hand, every factor is strictly
positive at $\mathfrak m_+$, so $\Lambda_{\mathfrak m_+}(c)>0$.

of $\mathbb{R}[\mathcal{A}]_{r-1}$ and replace $c$ by
\[
 c_\varepsilon:=c+\varepsilon\sum_{b\in\mathcal B}b.
\]
For $\varepsilon>0$ sufficiently small, the strict inequalities above are
preserved.

\mathcal{B}_{\mathcal{A}}^{+} \coloneqq \mathcal{B} \cup \bigcup_{i = 1}^m x^{\alpha_i} \mathcal{B} = \mathcal{B} \cup \{x^{\alpha_i} b : b \in \mathcal{B}, i=1,\dots,m\}, \quad \partial \mathcal{B}_{\mathcal{A}} = \mathcal{B}_{\mathcal{A}}^+ \setminus \mathcal{B},

\label{DefinitionMiNProjectionGeneralized}
Consider a finite subset $\mathcal{A} = \{\alpha_1,\dots,\alpha_m \} \subset \mathbb{Z}^n$.
Let $N: \mathcal{L}^+ \rightarrow \mathcal{L}$ be a linear map.
We say that $N$ is a projection with respect to $\mathcal{A}$, if the following two conditions are satisfied:
\begin{itemize}
\item[i)] $N \cdot N = N$,
\item[ii)] Whenever elements $\beta_1,\dots,\beta_j,\gamma_1,\dots,\gamma_k \in \mathcal{A}$ satisfy $\sum_{i = 1}^j \beta_i = \sum_{l = 1}^k \gamma_l$, the following identity holds for all $f \in \mathcal{L}$
\begin{equation*}
N(x^{\beta_1} N( \dots N(x^{\beta_{j-1}} N( x^{\beta_j} f)) \dots ) = N(x^{\gamma_1} N( \dots N(x^{\gamma_{k-1}} N( x^{\gamma_k} f)) \dots ).
\end{equation*}
\end{itemize}
Further, for a projection $N$ with respect to $\mathcal{A}$, and $i \in \{1,\dots,m\}$ we define the \emph{multiplication operator} $\mathcal{M}_i$, as the linear map
\begin{equation*}
\mathcal{M}_i : \mathcal{L} \rightarrow \mathcal{L}, \quad f \mapsto N(x^{\alpha_i} f).
\end{equation*}

<!-- theorem-11: body_md -->
Let $I \subseteq \mathbb{R}[\mathcal{A}]$ be a real radical ideal with $1 \leq r = |\mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)| < \infty$.
Suppose $f \in \mathbb{R}[\mathcal{A}]$ is chosen such that the values $\Lambda_{\mathfrak m}(f)$ are distinct for $\mathfrak m \in \mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)$.
Further, let $\mathcal{B} = \{x^{\beta_1},\dots,x^{\beta_r} \}$ be a basis of $\mathbb{R}[\mathcal{A}] / I$, where $\beta_i \in \mathcal{A}_\infty$.
Then the left eigenspaces of the matrix $m_f$ are $1$-dimensional and are spanned by the row vectors $(\Lambda_{\mathfrak m}(x^{\beta_1}),\dots,\Lambda_{\mathfrak m}(x^{\beta_r}))$ for $\mathfrak m \in \mathcal{V}_{\mathbb{R}[\mathcal{A}]}(I)$.
\label{TheoremStickelberger2}

<!-- proof-24: body_md -->
By Lemma~\ref{lem:Real_Nullstellensatz_Toric_variety}, evaluation induces the
isomorphism
\[
 \Theta:\mathbb{R}[\mathcal{A}]/I\longrightarrow\mathbb{R}^r,
 \qquad [g]\longmapsto
 \bigl(\Lambda_{\mathfrak m}(g)\bigr)_{\mathfrak m\in\mathcal V_{\mathbb{R}[\mathcal{A}]}(I)}.
\]
Under this isomorphism, multiplication by $f$ corresponds to the diagonal
operator with entries $\Lambda_{\mathfrak m}(f)$. Hence the rows of the matrix of
$\Theta$ in the basis $\mathcal{B}$, namely
\[
 \bigl(\Lambda_{\mathfrak m}(x^{\beta_1}),\dots,
 \Lambda_{\mathfrak m}(x^{\beta_r})\bigr),
\]
are left eigenvectors of $m_f$. Since $\Theta$ is an isomorphism and the
values $\Lambda_{\mathfrak m}(f)$ are pairwise distinct, these are all the left
eigenspaces and each has dimension one.
