<!-- definition-1: body_md -->
\label{def:wave-packet-density}
Let $g\in\mathcal{S}(\mathbb{R}^n)$ satisfy
${\rm supp}\widehat{g}\subset B_1^n \setminus B_{1/2}^n$, and let
\[
    B_R^{n+1}(z_0)
    =
    B_R^{n+1}(x_0,t_0)
    \subset B_{\lambda}^{n+1}.
\]
The wave packet density of $g$ in $B_R^{n+1}(z_0)$ is defined by
\begin{equation}
\label{eq:wave-packet-density}
\begin{aligned}
    &\mathcal{W}(g,B_R^{n+1}(z_0)) \\
    &\qquad :=
    \sup_{R^{-1/2}\leq s\leq 1}
    \sup_{\tau\in\Theta_s}
    \sup_{V\parallel V_{\tau,R}}
    \left(
        \frac{1}{|V|}
        \sum_{\substack{
            \theta\in\Theta_{R^{-1/2}}\\
            \theta\subset\tau
        }}
        \sum_{\substack{
            T\in\mathbb{T}_{\theta}(z_0)\\
            T^{\flat}\subset V
        }}
        \|g_T\|_{L^2(\mathbb{R}^n)}^2
    \right)^{1/2}.
\end{aligned}
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:GHLW-density}
For every $\varepsilon>0$ and
\[
p \geq p(n):=
\begin{cases}
\displaystyle p_n^{+}, & n \text{ odd},\\[8pt]
\displaystyle 2+\frac{8}{3n-2-\omega(n)}, & n \text{ even},
\end{cases}
\qquad
\omega(n):=\frac{4(7n-2)}{3n^{2}+5n+14},
\]
one has
\begin{equation}\label{wave-packet-density-estimate}
    \|\mathcal{F}^{\lambda}f\|_{L^{p}(B_{R}^{n+1})}
    \lesssim_{\varepsilon}
    R^{(n-1)\left(\frac12-\frac1p\right)+\varepsilon}
    \|f\|_{2}^{\frac{2}{p}}
    \mathcal{W}(f,B_{R}^{n+1})^{1-\frac{2}{p}},
\end{equation}
for every $R \in [1, \lambda^{1-\varepsilon}]$ and all $f\in\mathcal{S}(\mathbb{R}^n)$ with
\begin{equation}\label{supp-f-hat}
\begin{aligned}
{\rm supp} \widehat{f}
\subset
\mathbb{A}^n(1)
:=
\bigl\{
(\xi_1,\ldots,\xi_n)\in B_1^n\setminus B_{1/2}^n :
&\ \xi_j>0 \text{ for } 1\leq j\leq n-1,\\
&\ |(\xi_1,\ldots,\xi_{n-1})|\leq \xi_n
\bigr\}.
\end{aligned}
\end{equation}

<!-- theorem-2: body_md -->
\label{thm:refined-GHLW-density}
Let $n\geq4$ be even, and let $\mathcal F$ be a type-$\mathbf1$ FIO
as above satisfying \eqref{quadratic-cone-condition}. For every
$\varepsilon>0$, the estimate
\eqref{wave-packet-density-estimate} holds for all $p\geq p'(n)$,
all $R\in[1,\lambda^{1-\varepsilon}]$, and all
$f\in\mathcal S(\mathbb R^n)$ satisfying \eqref{supp-f-hat}.

Without \eqref{quadratic-cone-condition}, the same estimate holds,
under the remaining hypotheses, in the following ranges:
\[
\begin{cases}
\displaystyle p\geq\frac{166}{57},
& n=4,\\[7pt]
\displaystyle p\geq2+\frac8{3n-2-8/n},
& n\geq16 \text{ even}.
\end{cases}
\]

Write $v(z;\xi)=\partial_z\phi(z;\xi)$.

\emph{Preservation of the quadratic relation.}
Consider a transformed phase of the form
\begin{equation}\label{eq:quadratic-phase-transformation}
\widetilde\phi(w;\eta)
=
c\,\phi(\Psi(w);\Xi(\eta))+\psi(\eta),
\end{equation}
where $c\neq0$, $\Psi$ is a spacetime diffeomorphism, $\Xi$ is a
homogeneous frequency diffeomorphism independent of $w$, and
$\psi$ depends only on $\eta$. Setting $J(w)=D\Psi(w)$, the chain
rule gives
\[
\widetilde v(w;\eta)
:=
\partial_w\widetilde\phi(w;\eta)
=
cJ(w)^T v(\Psi(w);\Xi(\eta)).
\]
Define
\begin{equation}\label{eq:quadratic-form-transformation}
\widetilde{\mathbf Q}_w
=
J(w)^{-1}\mathbf Q_{\Psi(w)}J(w)^{-T}.
\end{equation}
Then
\[
\widetilde v(w;\eta)^T
\widetilde{\mathbf Q}_w
\widetilde v(w;\eta)
=
c^2v(\Psi(w);\Xi(\eta))^T
\mathbf Q_{\Psi(w)}
v(\Psi(w);\Xi(\eta))
=
0.
\]
Moreover, $\widetilde{\mathbf Q}_w$ is symmetric, independent of
$\eta$, and nondegenerate, since
\[
\det\widetilde{\mathbf Q}_w
=
\frac{\det\mathbf Q_{\Psi(w)}}{(\det J(w))^2}
\neq0.
\]

The Lorentz rescaling used in \cite[Section~5.1]{GanHeLiWu}
is of the form \eqref{eq:quadratic-phase-transformation}.
To see this explicitly, write the center of the frequency cap as
$(\omega,1)$ and use the phase coordinates from
\cite[Section~2.5]{BeltranHickmanSogge2020}.
Let
\[
\Psi_\omega(y,t)=(X_\omega(y,t),t)
\]
be determined locally by
\[
\partial_{\xi'}\phi(X_\omega(y,t),t;\omega,1)=y',
\qquad
\phi(X_\omega(y,t),t;\omega,1)=y_n.
\]
Before the final bounded linear normalization, the rescaled phase
has the exact expression
\[
\widetilde\phi(y,t;\eta)
=
K^2\phi\left(
\Psi_\omega(K^{-1}y',K^{-2}y_n,t);
K^{-1}\eta'+\eta_n\omega,\eta_n
\right);
\]
see \cite[equation~(2.14)]{BeltranHickmanSogge2020}.
Thus \eqref{eq:quadratic-form-transformation} applies.
The subsequent linear changes of variables and localization
preserve the same condition. Likewise, subtracting $\phi(0;\xi)$
does not change $\partial_z\phi$, and the dilation
\[
\phi^\lambda(z;\xi)=\lambda\phi(z/\lambda;\xi)
\]
satisfies the quadratic-cone condition with defining matrix
$\mathbf Q_{z/\lambda}$.

\emph{Uniform nondegeneracy after normalization.}
The preceding argument establishes exact preservation, but an
arbitrary non-isotropic change of variables need not preserve the
condition number of a particular defining matrix. We therefore
normalize the defining form on each normalized phase patch.

At a fixed spacetime point, the usual positive-curvature
normalization allows the frozen cone to be parametrized, after a
uniformly bounded linear change of covector coordinates, by
\[
r(\eta,1,h(\eta)),
\qquad r>0,
\]
where, on a ball $|\eta|<a_0$ of fixed radius,
\[
h(0)=0,\qquad \nabla h(0)=0,
\qquad
c_0I\leq\nabla^2h(\eta)\leq C_0I.
\]
The constants $a_0,c_0,C_0$ depend only on the normalized phase
bounds; compare \cite[Lemma~2.5]{BeltranHickmanSogge2020} and the
normalization in \cite[Lemma~5.7]{GanHeLiWu}.

Let $\mathbf M$ be a nondegenerate symmetric matrix defining this
cone. At $\eta=0$, the cone has radial vector $e_n$ and tangent
space $\operatorname{span}\{e_1,\ldots,e_n\}$. Consequently,
$\mathbf M e_n$ is a nonzero multiple of $e_{n+1}$.
Multiplying $\mathbf M$ by a nonzero scalar, we may assume that
$\mathbf M e_n=-e_{n+1}$. Differentiating the quadratic relation
twice at $\eta=0$ then gives
\[
\mathbf M
=
\begin{pmatrix}
H&0&b\\
0&0&-1\\
b^T&-1&d
\end{pmatrix},
\qquad
H=\nabla^2h(0)\asymp I.
\]
The equation of the cone becomes
\begin{equation}\label{eq:normalized-quadratic-relation}
\eta^TH\eta-2h(\eta)
+2h(\eta)b\cdot\eta+d\,h(\eta)^2=0.
\end{equation}

Fix $0<a<a_0$, independently of the rescaling parameters, and set
\[
h_i^\pm:=h(\pm ae_i)\asymp a^2,
\qquad 1\leq i\leq n-1.
\]
Evaluating \eqref{eq:normalized-quadratic-relation} at $\pm ae_i$
and dividing by $2h_i^\pm$, we obtain
\[
\pm ab_i+\frac d2h_i^\pm
=
1-\frac{a^2H_{ii}}{2h_i^\pm}.
\]
The right-hand sides are uniformly bounded. Adding the two
identities first gives a uniform bound for $d$, since
$h_i^++h_i^-\asymp a^2$, and substitution then gives uniform
bounds for each $b_i$. Hence $\|\mathbf M\|\lesssim1$.
Since
\[
\det\mathbf M=-\det H
\]
and $H\asymp I$, we also have $\|\mathbf M^{-1}\|\lesssim1$.
Undoing the uniformly bounded covector change preserves these
bounds.

Finally, the Lorentz rescaling and subsequent localization return
the phase to the same quantitative class, as in
\cite[Lemma~2.3 and Remark~2.4]{BeltranHickmanSogge2020}.
Applying the preceding normalization to each resulting phase gives
uniform bounds for its defining quadratic form. These bounds depend
only on the fixed normalized phase constants and do not accumulate
with the number of rescaling steps.

We now describe Algorithm 1, which provides a procedure for refining a given triple $({\bf B},{\bf T};\mathcal {G})$, where ${\bf T}$ denotes the collection of wave packets and ${\bf B}$ denotes the integration domain. The output consists of a refined triple together with several auxiliary parameters.

\medskip

\noindent\textit{Input:}
Let $({\bf B},{\bf T};\mathcal {G})$ be an incidence triple, and suppose that there exists $\nu>0$ such that, for every $B\in{\bf B}$,
\[
\big\|\sum_{T\in{\bf T}(B;\mathcal {G})}\mathcal{F}^\lambda f_T\big\|_{L^p(B)}\sim \nu.
\]

\medskip

\noindent\textit{Pigeonholing and pruning:}
Pigeonhole the broad cardinality into the class zero and the positive
dyadic classes. If the zero class dominates, retain it as a terminal
narrow output: its local $k$-broad contribution vanishes. After the
horizontal localization in
Subsection~\ref{subsubsec:completingreduction}, this output is treated
by the narrow argument of \cite[Section 8.3]{GanHeLiWu}, without any
use of negative powers of $\mu$. The statements that follow concern
the positive class. By dyadic pigeonholing, there exist $\mu\lesssim R^{O(1)}$ and a subcollection ${\bf B}_1\subset{\bf B}$ such that
\[
    \#_{\textup{br},k}{\bf T}(B;\mathcal {G})\sim\mu
    \qquad\textup{for all $B\in{\bf B}_1$},
\]
and
\[
    L^p({\bf B},{\bf T};\mathcal {G};f)
    \lessapprox
    L^p({\bf B}_1,{\bf T};\mathcal {G};f).
\]

For dyadic numbers $W\lesssim R^{O(1)}$ and $\beta\lesssimK_\circ$, define
\[
    \mathcal {G}_W(T)
    :=
    \left\{
    \omega\in\mathcal {G}(T):
    \#\{B\in{\bf B}_1(T;\mathcal {G}):B\subset\omega\}\sim W
    \right\},
\]
and
\[
    {\bf T}_{W,\beta}
    :=
    \{T\in{\bf T}:\#\mathcal {G}_W(T)\sim\beta\}.
\]

Fix $B\in{\bf B}_1$. By dyadic pigeonholing, there exist $W(B)$ and $\beta(B)$ such that
\[
\nu
\sim
\big\|\sum_{T\in{\bf T}(B;\mathcal {G})}\mathcal{F}^\lambda f_T\big\|_{L^p(B)}
\lessapprox
\big\|\sum_{T\in{\bf T}_{W(B),\beta(B)}(B;\mathcal {G}_{W(B)})}
\mathcal{F}^\lambda f_T\big\|_{L^p(B)}.
\]
Applying dyadic pigeonholing once more, we may find dyadic numbers $W$, $\beta$, and $\nu_1$, together with a subcollection ${\bf B}_2\subset{\bf B}_1$, such that
\[
\begin{gathered}
W(B)\sim W,\qquad \beta(B)\sim\beta,\\
\left\|\sum_{T\in{\bf T}_{W,\beta}(B;\mathcal {G}_W)}
\mathcal{F}^\lambda f_T\right\|_{L^p(B)}\sim\nu_1
\quad\text{for all }B\in{\bf B}_2.
\end{gathered}
\]
and
\[
    L^p({\bf B}_1,{\bf T};\mathcal {G};f)
    \lessapprox
    L^p({\bf B}_2,{\bf T}_{W,\beta};\mathcal {G}_W;f).
\]

Finally, use the fixed $R^{1/2}$-grid in $B_R^{n+1}$. By dyadic pigeonholing, there exist dyadic numbers $m$, $\bar m$, and $\sigma$, together with a collection ${\mathcal Q}$ of $R^{1/2}$-cubes, such that, for every $Q\in{\mathcal Q}$,
\[
\#_{\textup{br},2}
\bigcup_{\substack{B\in{\bf B}_2\\B\subset Q}}
{\bf T}_{W,\beta}(B;\mathcal {G}_W)
\sim m,
\]
\[
\#_{\textup{br},2}
\bigcup_{\substack{B\in{\bf B}_2\\B\subset Q}}
{\bf T}(B;\mathcal {G})
\sim\bar m,
\]
and
\[
\#\{B \in \mathbf{B}_2, B \subset Q\} \sim\sigma.
\]
Moreover,
\[

    \lessapprox
    L^p({\bf B}_3,{\bf T}_{W,\beta};\mathcal {G}_W;f),
\]
where
\[
{\bf B}_3
:=
\{
B\in{\bf B}_2: B \subset Q \text{ for some } Q \in {\mathcal Q}\}
\]

:=
\sup_{T\in{\bf T}_{W,\beta}}

\begin{lemma}\label{sigma-new-bound}
Under the assumptions of this section, including
\eqref{quadratic-cone-condition}, and with $A$ sufficiently large,
\[

\lesssim
K^{O(1)}R^{\frac{n+1-k}{2}}\bar m^k\mu^{-k}.
\]
\end{lemma}

\begin{proof}
Fix $Q\in{\mathcal Q}$ and let $z_Q$ be its center. Set
\[
\mathcal T_0=
\bigcup_{\substack{B\in{\bf B}_3\\B\subset Q}}{\bf T}(B;\mathcal {G}),
\qquad
\mathcal T=(\mathcal T_0)_{\textup{br},2}.
\]
The nested-grid convention gives
\[
\#\mathcal T\sim\bar m,
\qquad
\#\{B\in{\bf B}_3:B\subset Q\}\sim\sigma.
\]
There is a cap $\tau_0$ such that
\[
\mathcal T
=
\{T\in\mathcal T_0:\theta(T)\not\subset3\tau_0\}.
\]
We show that each such $B$ supplies $k$ subcollections of
$\mathcal T(B;\mathcal {G})$, each of cardinality at least
$K^{-O(1)}\mu$, whose actual packet normals satisfy
\eqref{linear-independence} uniformly.

Fix $B\subset Q$, with center $z_B$. The coarse caps meeting
$3\tau_0$ form a family of cardinality at most $C_n$.
For each of these caps, choose a $(k-1)$-plane containing one
of its Gauss directions at $z_B$, and include these planes among
the exceptions in the broad-cardinality definition. Recall that
the angle between a cap and a plane is the infimum over all
Gauss directions in that cap. Thus every cap admissible for this
list is disjoint from $3\tau_0$, and its packets survive the
pruning.

By $\#_{\textup{br},k}{\bf T}(B;\mathcal {G})\sim\mu$, every choice of at most $A$
exceptional planes leaves an admissible coarse cap containing
$\gtrsim\mu$ packets. We choose $k$ rich subcollections
inductively. Suppose that $j<k$ representative frequencies
$\xi_1,\ldots,\xi_j$ have been selected. Choose a
$(k-1)$-plane $V_j$ containing
\[
\operatorname{span}\{
G^\lambda(z_B;\xi_1),\ldots,G^\lambda(z_B;\xi_j)
\},
\]
with an arbitrary choice when $j=0$. Add $V_j$ to the fixed
list of planes excluding the caps that meet $3\tau_0$.
This uses at most $C_n+1$ planes, so the choice is allowed
when $A\geq C_n+1$.

There is consequently an admissible coarse cap
$\tau_{j+1}$ containing $\gtrsim\mu$ packets, all with Gauss
directions at angle greater than $K^{-2}$ from $V_j$.
Subdivide $\tau_{j+1}$ into angular subcaps of radius $K^{-M}$,
where $M$ is a sufficiently large fixed constant depending only
on $n$ and the normalized phase bounds. Retain one containing
$\gtrsim K^{-O(1)}\mu$ packets, denote this subcollection by
$\mathcal T_{j+1}(B)$, and choose $\xi_{j+1}$ to be the frequency
direction of one of its packets.

After $k$ steps, the representative Gauss directions satisfy
\[
\left|
\bigwedge_{j=1}^{k}G^\lambda(z_B;\xi_j)
\right|
\gtrsim K^{-2(k-1)}.
\]
By Lemma~\ref{quadratic-gauss-normal}, the same polynomial
lower bound holds for their slab normals. Taking $M$ sufficiently
large makes this bound uniform over all choices of actual packet
directions from
$\mathcal T_1(B),\ldots,\mathcal T_k(B)$.

On $Q$, the variation of the slab normals is
\[
O(R^{1/2}/\lambda)\lesssim R^{-1/2}.
\]
By choosing $\varepsilon$ sufficiently small relative to the fixed
powers of $K$, the transversality therefore persists at $z_Q$.
Fix a constant $C_0$ for which the preceding lower bounds imply
\eqref{linear-independence}, and let $\mathcal A_Q$ be the set
of ordered $k$-tuples of packets in $\mathcal T$ satisfying
that condition at $z_Q$.

Write $T^+=N_{C K^2}(T)$, with $C$ large enough that
$B\cap T\neq\emptyset$ implies $B\subset T^+$.
For every $B\in{\bf B}_3$ contained in $Q$, the preceding choices
give at least $K^{-O(1)}\mu^k$ tuples in $\mathcal A_Q$,
and each corresponding enlarged intersection contains $B$.
The $B$'s are disjoint grid cubes of common volume
$v_K\sim K^{2(n+1)}$. Hence
\[
\begin{aligned}
K^{-O(1)}\mu^k\sigma v_K
&\lesssim
\sum_{(T_1,\ldots,T_k)\in\mathcal A_Q}
\left|Q\cap\bigcap_{i=1}^{k}T_i^+\right|\\
&\lesssim
K^{O(1)}R^{\frac{n+1-k}{2}}\#\mathcal A_Q\\
&\leq
K^{O(1)}R^{\frac{n+1-k}{2}}(\#\mathcal T)^k.
\end{aligned}
\]
The second inequality is Lemma~\ref{multilinear-estimates}.
Absorbing the powers of $K$ and using
$\#\mathcal T\sim\bar m$ proves the lemma.
\end{proof}

There is a cap $\tau_0$ such that
\[
\mathcal T
=
\{T\in\mathcal T_0:\theta(T)\not\subset3\tau_0\}.
\]
We show that each such $B$ supplies $k$ subcollections of
$\mathcal T(B;\mathcal {G})$, each of cardinality at least
$K^{-O(1)}\mu$, whose actual packet normals satisfy
\eqref{linear-independence} uniformly.

Fix $B\subset Q$, with center $z_B$. The coarse caps meeting
$3\tau_0$ form a family of cardinality at most $C_n$.
For each of these caps, choose a $(k-1)$-plane containing one
of its Gauss directions at $z_B$, and include these planes among
the exceptions in the broad-cardinality definition. Recall that
the angle between a cap and a plane is the infimum over all
Gauss directions in that cap. Thus every cap admissible for this
list is disjoint from $3\tau_0$, and its packets survive the
pruning.

By $\#_{\textup{br},k}{\bf T}(B;\mathcal {G})\sim\mu$, every choice of at most $A$
exceptional planes leaves an admissible coarse cap containing
$\gtrsim\mu$ packets. We choose $k$ rich subcollections
inductively. Suppose that $j<k$ representative frequencies
$\xi_1,\ldots,\xi_j$ have been selected. Choose a
$(k-1)$-plane $V_j$ containing
\[
\operatorname{span}\{
G^\lambda(z_B;\xi_1),\ldots,G^\lambda(z_B;\xi_j)
\},
\]
with an arbitrary choice when $j=0$. Add $V_j$ to the fixed
list of planes excluding the caps that meet $3\tau_0$.
This uses at most $C_n+1$ planes, so the choice is allowed
when $A\geq C_n+1$.

There is consequently an admissible coarse cap
$\tau_{j+1}$ containing $\gtrsim\mu$ packets, all with Gauss
directions at angle greater than $K^{-2}$ from $V_j$.
Subdivide $\tau_{j+1}$ into angular subcaps of radius $K^{-M}$,
where $M$ is a sufficiently large fixed constant depending only
on $n$ and the normalized phase bounds. Retain one containing
$\gtrsim K^{-O(1)}\mu$ packets, denote this subcollection by
$\mathcal T_{j+1}(B)$, and choose $\xi_{j+1}$ to be the frequency
direction of one of its packets.

After $k$ steps, the representative Gauss directions satisfy
\[
\left|
\bigwedge_{j=1}^{k}G^\lambda(z_B;\xi_j)
\right|
\gtrsim K^{-2(k-1)}.
\]
By Lemma~\ref{quadratic-gauss-normal}, the same polynomial
lower bound holds for their slab normals. Taking $M$ sufficiently
large makes this bound uniform over all choices of actual packet
directions from
$\mathcal T_1(B),\ldots,\mathcal T_k(B)$.

On $Q$, the variation of the slab normals is
\[
O(R^{1/2}/\lambda)\lesssim R^{-1/2}.
\]
By choosing $\varepsilon$ sufficiently small relative to the fixed
powers of $K$, the transversality therefore persists at $z_Q$.
Fix a constant $C_0$ for which the preceding lower bounds imply
\eqref{linear-independence}, and let $\mathcal A_Q$ be the set
of ordered $k$-tuples of packets in $\mathcal T$ satisfying
that condition at $z_Q$.

Write $T^+=N_{C K^2}(T)$, with $C$ large enough that
$B\cap T\neq\emptyset$ implies $B\subset T^+$.
For every $B\in{\bf B}_3$ contained in $Q$, the preceding choices
give at least $K^{-O(1)}\mu^k$ tuples in $\mathcal A_Q$,
and each corresponding enlarged intersection contains $B$.
The $B$'s are disjoint grid cubes of common volume
$v_K\sim K^{2(n+1)}$. Hence
\[
\begin{aligned}
K^{-O(1)}\mu^k\sigma v_K
&\lesssim
\sum_{(T_1,\ldots,T_k)\in\mathcal A_Q}
\left|Q\cap\bigcap_{i=1}^{k}T_i^+\right|\\
&\lesssim
K^{O(1)}R^{\frac{n+1-k}{2}}\#\mathcal A_Q\\
&\leq
K^{O(1)}R^{\frac{n+1-k}{2}}(\#\mathcal T)^k.
\end{aligned}
\]
The second inequality is Lemma~\ref{multilinear-estimates}.
Absorbing the powers of $K$ and using
$\#\mathcal T\sim\bar m$ proves the lemma.

Algorithm 2 repeatedly applies Algorithm 1 to obtain additional quantitative relationships among the parameters. The same iterative strategy was previously used in \cite{Wang2018}. By dyadic pigeonholing, we may assume that there exists a finitely overlapping collection ${\mathcal B}=\{B\}$ of $K^2$-balls such that the quantities $\|\mathcal{F}^\lambda f\|_{L^p(B)}$
are comparable for all $B\in{\mathcal B}$, and
\[
\|\mathcal{F}^\lambda f\|_{L^p(B_R^{n+1})}^p
\lessapprox
\sum_{B\in{\mathcal B}}
\|\mathcal{F}^\lambda f\|_{L^p(B)}^p.
\]

Set
\[
\mathbb{T}^{(0)}=\mathbb{T},
\qquad
{\mathcal B}^{(0)}={\mathcal B},
\]
and define
\[
\mathcal {G}^{(0)}(T)=\Omega
\qquad
\text{for every }T\in\mathbb{T}.
\]

We now apply Algorithm 1 iteratively, treating a terminal zero-broad
output by the narrow argument described above. On the positive branch,
use lower dyadic representatives for the cardinalities, so that the
monotonicity statements below hold. At the $i$-th step, we apply
Algorithm 1 to
\[
({\mathcal B}^{(i-1)},\mathbb{T}^{(i-1)};\mathcal {G}^{(i-1)};f),
\]
which produces the following:

\begin{enumerate}

    W^{(i-1)},\;
    \beta^{(i-1)},\;
    m^{(i-1)},\;
    l^{(i-1)},\;
    \sigma^{(i-1)}.
    \]

    \item Two collections of $K^2$-balls satisfying
    \[
    {\mathcal B}_3^{(i-1)}
    \subset
    {\mathcal B}_1^{(i-1)}
    \subset
    {\mathcal B}^{(i-1)}.
    \]

    \item A collection of planks
    \[
    \mathbb{T}  _{W^{(i-1)},\beta^{(i-1)}}^{(i-1)}
    \subset
    \mathbb{T}  ^{(i-1)},
    \]
    a shading map
    \[
    \mathcal {G}_{W^{(i-1)}}^{(i-1)}
    \subset
    \mathcal {G}^{(i-1)},
    \]
    and a collection ${\mathcal Q}^{(i-1)}$ of $R^{1/2}$-cubes.
\end{enumerate}

We then set
\[
{\mathcal B}^{(i)}
:=
{\mathcal B}_3^{(i-1)},
\qquad
\mathbb{T}  ^{(i)}
:=
\mathbb{T}  _{W^{(i-1)},\beta^{(i-1)}}^{(i-1)},
\qquad
\mathcal {G}^{(i)}
:=
\mathcal {G}_{W^{(i-1)}}^{(i-1)}.

\leq
\kappa\,\mu^{(N+1)},
\]

\bar{m}^{(N)}
\leq
\kappa\,\bar{m}^{(N+1)}
\leq
\kappa\,m^{(N)},
\]

\mathcal I({\mathcal B}_1^{(N)},\mathbb{T}  ^{(N)};\mathcal {G}^{(N)})
\leq
\kappa\,

l^{(N)},\;
{\mathcal B}^{(N+1)},\;
\mathbb{T}^{(N+1)},\;

\beta,\;
m,\;

\mathbb{T},\;
\mathcal {G},\;
\sigma,
\]
and ${\mathcal Q}$, respectively. We also redefine

f:=\sum_{T\in\mathbb{T}}f_T.

\eqref{incidence-0}, \eqref{incidence-1}, Lemma~\ref{hairbrush},
and Lemma~\ref{sigma-new-bound}, we obtain, whenever $\beta>100$,
\begin{equation}\label{cancel0}
lm\mu\lesssim K^{O(1)}\#\mathbb{T}  .
\end{equation}
Moreover,
\begin{equation}\label{cancel1}
\mu\lesssim\kappa^{O(1)}m,
\end{equation}

\begin{equation}\label{cancel2}
\sigma
\lesssim
(K\kappa)^{O(1)}
R^{\frac{n+1-k}{2}}m^k\mu^{-k}.
\end{equation}

\begin{lemma}\label{cancel}
We have
\[
\mu
\lesssim
K^{O(1)}
\left(\#\mathbb T\right)^{1/2}l^{-1/2}
\min\left\{
1,\frac{R^{\frac{n+1-k}{2}}}{\sigma}
\right\}^{\frac1{2k}}.
\]
\end{lemma}

\begin{proof}
Combining \eqref{cancel0} and \eqref{cancel1}, and absorbing
$\kappa^{O(1)}$ into $K^{O(1)}$, we obtain
\begin{equation}\label{cancel3}
\mu
\lesssim
K^{O(1)}\left(\#\mathbb T\right)^{1/2}l^{-1/2}.
\end{equation}
On the other hand, \eqref{cancel0} gives
\[
m\lesssim K^{O(1)}\frac{\#\mathbb T}{l\mu}.
\]
Substituting this into \eqref{cancel2}, we obtain
\[
\sigma
\lesssim
K^{O(1)}R^{\frac{n+1-k}{2}}
(\#\mathbb T)^k l^{-k}\mu^{-2k},
\]
and consequently
\[
\mu
\lesssim
K^{O(1)}
\left(\#\mathbb T\right)^{1/2}l^{-1/2}
\left(
\frac{R^{\frac{n+1-k}{2}}}{\sigma}
\right)^{\frac1{2k}}.
\]
Combining this estimate with \eqref{cancel3} proves the lemma.
\end{proof}
