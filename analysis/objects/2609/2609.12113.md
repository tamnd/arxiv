<!-- proposition-1: body_md -->
\label{prop:FPK}
    Assume the ground truth density $p_t(\mathbf{x})$ is sufficiently smooth on $\mathbb{R}^n\times[0,T]$ with its log-density denoted as $l_t(\mathbf{x}):=\log p_t(\mathbf{x})$. Then for all $(\mathbf{x},t)$, its log-density satisfies the PDE
\begin{equation}\label{eq:log_FPK}
    \partial_t l_t(\mathbf{x})
    =\mathbf{x}\cdot\nabla l_t(\mathbf{x})+n+\Delta l_t(\mathbf{x})+\|\nabla l_t(\mathbf{x})\|^2.
\end{equation}

<!-- definition-1: body_md -->
\label{def:ou_generator}
    The OU generator $\mathcal{L}$ acts on smooth functions $f\in C^2(\mathbb{R}^n)$ by
    \[\mathcal{L} f(\mathbf{x})=\Delta f(\mathbf{x})-\mathbf{x}\cdot\nabla f(\mathbf{x}).\]
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{def:ou_spectrum}
    Let $\gamma:=\mathcal{N}(0,I)$ be the standard Gaussian measure. In the Hilbert space $L^2(\gamma),$ $\mathcal{L}$ is self-adjoint and has discrete spectrum $\{0,-1,-2,...\}$ of non-positive integers.
    The eigenfunctions are the Hermite polynomials \cite{hermite1864nouveau} $(H_\alpha)_{\alpha\in\mathbb{N}^n}$, orthogonal in $L^2(\gamma),$ satisfying $\mathcal{L} H_\alpha=-|\alpha|H_\alpha$.
<!-- end theorem-1 -->
<!-- definition-2: body_md -->
\label{def:likelihood_pushforward}
Let $\mu$ be a probability measure on $\mathbb{R}^n$ with density $p$. Define the log-likelihood function
$l(\mathbf{x}) := \log p(\mathbf{x}).$
The pushforward measure of $\mu$ under $l$ is the probability measure
$L := l_{\#}\mu$
on $\mathbb{R}$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:lr_distribution}
Let $\mu$ be a probability measure on $\mathbb{R}^n$ with density $p$ and log-likelihood $l(\mathbf{x})=\log p(\mathbf{x})$. Let $L=l_{\#}\mu$.
Let $\eta$ be a probability measure on $\mathbb{R}$ such that $\eta \ll L$.
A probability measure $\nu$ on $\mathbb{R}^n$ is called a likelihood-reweighted measure of $\mu$ with target $\eta$ if
$\nu$ admits the disintegration
\[\nu(A)=\int_{\mathbb{R}} \mu(A \mid l(\mathbf{x})=u)\,\eta(du),
\quad A \subset \mathbb{R}^n \text{ Borel},\]
where $\mu(\cdot \mid l(\mathbf{x})=u)$ denotes a regular conditional probability of $\mu$ given $l(\mathbf{x})=u$, which exists for Borel probability measures on Polish spaces \cite{leao2004regular}.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:FOSD}
Let $\mu$ and $\nu$ be probability measures on $\mathbb{R}$. We say that $\nu$ first-order stochastically dominates $\mu$ and write
\[\mu\leq_{st}\nu\] if and only if $\mu([x,\infty))\leq \nu([x,\infty))$ for all $x\in\mathbb{R}$.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
\label{def:rho_outlier}
Let $\mu$ be a probability measure on $\mathbb{R}^n$ with log-likelihood pushforward $L=l_{\#}\mu$. 
A likelihood-reweighted measure $\nu$ with target $\eta$ is called a \emph{$\rho$-outlier measure} if
\[
\text{(1) }\eta \le_{st} L,\quad\text{ and \quad (2) } W_1(L,\eta) \geq \rho,
\]
                   
                          
                             
                 
where $\le_{st}$ represents FOSD (see Definition \ref{def:FOSD}) and $W_1(\cdot,\cdot)$ is the Wasserstein-1 distance (see Equation \eqref{eq:W1} below).
<!-- end definition-5 -->
<!-- remark-1: body_md -->
The notion of a $\rho$-outlier measure formalizes outliers at the distribution level, which is natural for generative models that approximate probability measures rather than individual samples.

Let $F_L$ and $F_\eta$ denote the CDFs of $L$ and $\eta$, and let $F_L^{-1}$ and $F_\eta^{-1}$ denote their quantile functions.
The condition $\eta \le_{st} L$ is equivalent to
$F_\eta^{-1}(w) \le F_L^{-1}(w)\quad \text{for all } w\in[0,1].$
Moreover, in one dimension the $W_1$ distance admits the quantile representation \cite{villani2021topics}
\begin{equation}\label{eq:W1}
W_1(L,\eta)=\int_0^1 |F_\eta^{-1}(w)-F_L^{-1}(w)|\,dw.
\end{equation}
Hence, under stochastic dominance,
\[W_1(L,\eta)=\int_0^1 \big(F_L^{-1}(w)-F_\eta^{-1}(w)\big)\,dw,\]
so the $\rho$-outlier condition quantifies a strict downward shift of likelihood quantiles, as expected of outliers.
<!-- end remark-1 -->
<!-- theorem-2: body_md -->
\label{thm:score-relationship}
Let $\mu$ be a probability measure on $\mathbb{R}^n$ with density $p$ and log-likelihood $l(\mathbf{x})=\log p(\mathbf{x})$.
Let $L=l_{\#}\mu$ be the pushforward measure of $\mu$ under $l$.
Let $\eta$ be a probability measure on $\mathbb{R}$ such that $\eta \ll L$, and define
\[
z(u):=\frac{d\eta}{dL}(u).
\]

Let $\nu$ be the likelihood-reweighted measure defined by
\[
\nu(A):=\int_{\mathbb{R}}\mu(A \mid l(\mathbf{x})=u)\,\eta(du),
\qquad A \subset \mathbb{R}^n \text{ Borel}.
\]

Then $\nu \ll \mu$ and
\begin{equation}
    \frac{d\nu}{d\mu}(\mathbf{x})=z(l(\mathbf{x}))=\frac{d\eta}{dL}(l(\mathbf{x})).
\end{equation}
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
Let $f:\mathbb{R}^n \to \mathbb{R}$ be bounded and measurable. By definition of $\nu$,
\[
\int f(\mathbf{x})\, \nu(d\mathbf{x})
= \int_{\mathbb{R}} \left[ \int f(\mathbf{x})\, p(d\mathbf{x} \mid l(\mathbf{x}) = u) \right] \eta(du).
\]
Since $\eta \ll L$ with density $z(u)$, we can write $\eta(du) = z(u)\,L(du)$
and obtain
\[
\int f(\mathbf{x})\, \nu(d\mathbf{x})
= \int_{\mathbb{R}} z(u)\, \int f(\mathbf{x})\,\mu(d\mathbf{x} \mid l(\mathbf{x}) = u)\, L(du).
\]
By the disintegration theorem \cite{dellacherie1978probabilities},
\[
\int_{\mathbb{R}}\int f(\mathbf{x})\,\mu(d\mathbf{x} \mid l(\mathbf{x})=u)L(du)=\int f(\mathbf{x})\,\mu(d\mathbf{x}).
\]
Applying the same identity to the measurable function $x\mapsto z(l(\mathbf{x}))f(\mathbf{x})$ gives
\[
\int f(\mathbf{x})\,z(l(\mathbf{x}))\,\mu(d\mathbf{x})
=\int_{\mathbb{R}}z(u)\int f(\mathbf{x})\,\mu(d\mathbf{x} \mid l(\mathbf{x})=u)L(du).
\]
Comparing the two expressions, we conclude
\[\int f(\mathbf{x})\,\nu(d\mathbf{x})=\int f(\mathbf{x})\,z(l(\mathbf{x}))\,\mu(d\mathbf{x}).\]
Since this holds for all bounded measurable $f$, it follows that
\[\frac{d\nu}{d\mu}(\mathbf{x})=z(l(\mathbf{x})).\]
<!-- end proof-1 -->
<!-- corollary-1: body_md -->
\label{corr:score-adjustmnt}
Let $\mu_t$ be a family of probability measures on $\mathbb{R}^n$ with densities $p_t$ and log-densities $l_t=\log p_t$.
Let $L_t=(l_t)_\#\mu_t$.
Fix a family of target measures $(\eta_t)_{t\in[0,T]}$ on $\mathbb{R}$ such that $\eta_t\ll L_t$ for each $t$, and define
\[z_t(u):=\frac{d\eta_t}{dL_t}(u).\]
Define $\nu_t$ to be the likelihood-reweighted measure of $\mu_t$ with target $\eta_t$ in the sense of Definition~\ref{def:lr_distribution}, and let $q_t$ denote its density.
Then $\nu_t\ll \mu_t$ and
\begin{equation}
    \frac{d\nu_t}{d\mu_t}(\mathbf{x}_t)=z_t(l_t(\mathbf{x}_t))=\frac{d\eta_t}{dL_t}(l_t(\mathbf{x}_t)).
\end{equation}
<!-- end corollary-1 -->

In this case, $\nu_T=\mu_T$, which approaches the standard Gaussian for sufficiently large $T$.
The score identity below characterizes each likelihood-reweighted density point-wise in time and motivates the controlled sampling ansatz introduced in Section \ref{sec:implementation}.

<!-- proposition-2: body_md -->
\label{prop:score_decomposition}
Let $(\mu_t,p_t,l_t, z_t, L_t, \eta_t,\nu_t)_{t\in[0,T]}$ be defined as in Corollary \ref{corr:score-adjustmnt}.

\end{equation}

\begin{equation}

\end{equation}

<!-- corollary-2: body_md -->
\label{corr:combined_score}
Let $(\eta_t)_{t\in[0,T]}$ be a family of target likelihood
distributions satisfying $\eta_t \ll L_t$ for all $t$.
Then the corresponding likelihood-reweighted densities
$q_t$ satisfy the score relation
                                                                                                                                                                                                                                            
\begin{equation}
\nabla \log q_t(\mathbf{x}) = \left[1 + \partial_u\left(\log\frac{d\eta_t}{dL_t}\right)(\log p_t(\mathbf{x}))\right] \nabla \log p_t(\mathbf{x}).
\end{equation}
Consequently, likelihood reweighting induces a multiplicative modification of the diffusion score function.
                                                                                                                                        
Furthermore, we choose a target likelihood distribution $\eta_0$ that satisfies
\[
\eta_0 \le_{st} L_0,
\quad
W_1(L_0,\eta_0) > \rho,
\]
matching our specification of $\rho$-outliers.

<!-- theorem-3: body_md -->
\label{thm:OU_convergence}
Let $\mathbf{x}_t$ solve the OU SDE \eqref{eq:forward_SDE}, where $\mathbf{x}_0\sim p$.
Let $\mu_t$ denote the time-marginal law of $\mathbf{x}_t$, and let $p_t$ denote its density.
The invariant measure is the standard Gaussian measure $\gamma=\mathcal{N}(0,I)$ with density $p_\gamma$. Define the log-densities
\[l_t(\mathbf{x}):=\log p_t(\mathbf{x}),\quad l^*(\mathbf{x}):=\log p_\gamma(\mathbf{x}).\]
Assume $\mu_t\ll\gamma$ and denote $h_t(\mathbf{x})$ as the RN derivative
$h_t(\mathbf{x}):=\frac{d\mu_t}{d\gamma}(\mathbf{x})$.
Assume $h_0 \in H^1(\gamma)$.                                
     
                   
                                                                   
                                                                                      
                                                  
                                                           
               
Then
\[\|h_t - 1\|_{L^2(\gamma)}^2
                                                           
     
                   
                             
\le e^{-2t}\|h_0 - 1\|^2_{L^2(\gamma)}.\]
Moreover,
\[
\|\nabla h_t\|_{L^2(\gamma)}\le e^{-t}\|\nabla h_0\|_{L^2(\gamma)}.
\]
Finally, suppose there exists $m>0$ such that $h_t(\mathbf{x})\ge m$ for $\mu_t$-almost all $\mathbf{x}$ and $t$.
Then
\[
\|\ell_t-\ell^*\|_{H^1(\gamma)}=\|\log h_t\|_{H^1(\gamma)}
                                        
    
               
    
                                 
\le\frac{e^{-t}}{m}\|h_0-1\|_{H^1(\gamma)}.
\]

<!-- proof-2: body_md -->
The density $p_t$ satisfies the FPK equation
\[\partial_t p_t = \nabla\cdot(\mathbf{x} p_t) + \Delta p_t.\]

Define the density ratio
$h_t(\mathbf{x}) := \frac{p_t(\mathbf{x})}{\gamma(\mathbf{x})}.$
Using the identities
\[
\nabla p_\gamma = -\mathbf{x} p_\gamma,
\quad
\Delta p_\gamma = (\|\mathbf{x}\|^2-n)p_\gamma,
\]
one verifies that $h_t$ satisfies
\[
\partial_t h_t = \mathcal{L} h_t,
\quad
\mathcal{L} := \Delta - \mathbf{x}\cdot\nabla,
\]
where $\mathcal{L}$ is the OU generator introduced in Definition~\ref{def:ou_generator}.
Let
$g_t := h_t - 1.$
Since $\int h_t\, d\gamma = 1$, we have $\int g_t\, d\gamma = 0$.
The evolution equation becomes
\begin{equation}\label{eq:OU_evolution}
\partial_t g_t = \mathcal{L} g_t,\quad g_t|_{t=0}=g_0.
\end{equation}
By Theorem~\ref{def:ou_spectrum}, the operator $\mathcal{L}$ is self-adjoint on $L^2(\gamma)$ with eigenfunctions given by Hermite polynomials \cite{hermite1864nouveau},
$\mathcal{L} H_\alpha = -|\alpha| H_\alpha .$
Expanding $g_0$ in the Hermite basis gives
\[
g_0 = \sum_{\alpha\neq 0} g_\alpha H_\alpha,
\quad
g_\alpha = \langle g_0,H_\alpha\rangle_{L^2(\gamma)}.
\]
Let $(P_t)_{t\ge 0}$ be the OU semigroup on $L^2(\gamma)$ generated by $\mathcal{L}$. Since $g_t$ satisfies Equation \eqref{eq:OU_evolution}, we identify $g_t$ with the unique semigroup solution $g_t=P_t g_0.$
Since $P_t$ acts diagonally on the Hermite basis, we can expand this solution
\[
g_t = \sum_{\alpha\neq 0} e^{-|\alpha|t} g_\alpha H_\alpha .
\]
Taking $L^2(\gamma)$ norms yields
\[
\|g_t\|_{L^2(\gamma)}^2=\sum_{k=1}^{\infty} e^{-2kt}\sum_{|\alpha|=k} g_\alpha^2 \le e^{-2t}\|g_0\|^2_{L^2(\gamma)},
\]
where the inequality
    
                                                       
    
follows immediately since $e^{-kt}\le e^{-t}$ for $k\ge1$.
By Lemma 1 of \cite{chen2025logsobolev}, we can bound the gradient:
                                     
    
                                                                           
    
                                   
\[
\|\nabla g_t\|^2_{L^2(\gamma)}\le e^{-2t}\|\nabla g_0\|^2_{L^2(\gamma)}.
\]
Because $g_t=h_t-1$, this is equivalent to
\[
\|\nabla h_t\|_{L^2(\gamma)}\le e^{-t}\|\nabla h_0\|_{L^2(\gamma)}.
\]

Now assume $h_t\ge m>0$. Since $\log$ is $1/m$-Lipschitz on $[m,\infty)$,
\[
|\log h_t(\mathbf{x})|\le \frac{1}{m}|h_t(\mathbf{x})-1|.
\]
Also, by expanding $\nabla \log h_t(\mathbf{x})=\frac{\nabla h_t(\mathbf{x})}{h_t(\mathbf{x})},$ we have
\[
\|\nabla \log h_t(\mathbf{x})\|\le \frac{1}{m}\|\nabla h_t(\mathbf{x})\|.
\]
Combining the two inequalities,
\[
\|\log h_t\|_{H^1(\gamma)}\le\frac{1}{m}\|h_t-1\|_{H^1(\gamma)}.
\]
Finally, using the $L^2$ and gradient estimates above,
\[
\|h_t-1\|_{H^1(\gamma)}\le e^{-t}\|h_0-1\|_{H^1(\gamma)}.
\]
Since
\[
\log h_t=\ell_t-\ell^*,
\]
the conclusion follows.

<!-- theorem-4: body_md -->
\label{thm:ct_interpolation}
Let $(\mathbf{x}_t)_{t\in[0,T]}$ be an OU diffusion \eqref{eq:forward_SDE}, and let $L_t$ and $\eta_t$ be two families of probability measures on $\mathds{R}$ which are absolutely continuous with respect to Lebesgue measure, converge exponentially to the Gaussian likelihood equilibrium, and satisfy:
\[
\eta_0\ll L_0,\quad\eta_0\leq_{st}L_0,\quad W_1(\eta_0,L_0)\geq\rho.
\]
Define the likelihood ratio
\[
z_t(u):=\frac{d\eta_t}{dL_t}(u),\quad c_t(u):=\partial_u \log z_t(u).
\]
Assume:
\begin{enumerate}
\item[(A1)] There exists $L>0$ such that for all $s,t\in[0,T]$,
\[
\|c_t-c_s\|_{L^2(L_t)} \le L|t-s|.
\]
\item[(A2)] There exists $K\ge 1$ such that for all $t\in[0,T]$ and all measurable $f$,
\[
K^{-1}\|f\|_{L^2(L_0)} \le \|f\|_{L^2(L_t)} \le K\|f\|_{L^2(L_0)}.
\]
\end{enumerate}
Define the interpolant
$\tilde c_t(u):=e^{-t}c_0(u).$
Then $\tilde c_0=c_0$, and for every $t\in[0,T]$,
\[
\|\tilde c_t-c_t\|_{L^2(L_t)}
\le
K\Big( (1-e^{-t})\|c_0\|_{L^2(L_0)} + Lt\Big).
\]
In particular,
$
\|\tilde c_t-c_t\|_{L^2(L_t)} = O(t),
$
<!-- end theorem-4 -->
<!-- proof-3: body_md -->
Fix $t\in[0,T]$. Add and subtract $c_0$:
\[
\tilde c_t-c_t = (e^{-t}c_0-c_0) + (c_0-c_t).
\]
Take $L^2(L_t)$ norms and apply the triangle inequality:
\[
\|\tilde c_t-c_t\|_{L^2(L_t)}
\le \|(1-e^{-t})c_0\|_{L^2(L_t)} + \|c_t-c_0\|_{L^2(L_t)}.
\]
\begin{align*}
\text{By (A2), }&\quad\|(1-e^{-t})c_0\|_{L^2(L_t)} \le K(1-e^{-t})\|c_0\|_{L^2(L_0)}.\\
\text{By (A1), }&\quad\|c_t-c_0\|_{L^2(L_t)} \le Lt.
\end{align*}
Combine the two bounds to obtain
\[
\|\tilde c_t-c_t\|_{L^2(L_t)}
\le
K\Big( (1-e^{-t})\|c_0\|_{L^2(L_0)} + Lt\Big).
\]
Finally, since $1-e^{-t}\le t$, the right-hand side is $O(t)$.

\label{thm:ct_interpolation2} Let $(\mathbf{x}_t)_{t\in[0,T]},L_t,\eta_t,\gamma,z_t(u),c_t(u)$ be defined as in Theorem \ref{thm:ct_interpolation}, and let $l_t,\mu_t,\nu_t$ be defined as in Corollary \ref{corr:score-adjustmnt}. Moreover, assume $c_0\in L^\infty$ and suppose the assumptions from Theorem \ref{thm:OU_convergence} are satisfied for $\mu_t$ and $\nu_t$.
Then the exponentially-interpolated controller 
$\tilde c_t(u):=e^{-t}c_0(u)$ 
satisfies the boundary condition $\tilde c_0=c_0$, and moreover 
\[
\|(\tilde c_t-c_t)(l_t(\cdot))\,\nabla \log p_t(\cdot)\|_{L^2(\gamma)}
\le
C e^{-t},
\quad t\in[0,T],
\]
for a constant $C>0$.
In particular,
\[
(\tilde c_t-c_t)(l_t(x))\,\nabla \log p_t(x)
\]
converges exponentially fast to zero in $L^2(\gamma)$.

<!-- proof-4: body_md -->
$\log z_t$ can be expanded as 
\[
\log z_t(l_t(\mathbf{x}))=\log\frac{d\nu_t}{d\mu_t}(\mathbf{x})=\log\frac{d\nu_t}{d\gamma}(\mathbf{x})-\log\frac{d\mu_t}{d\gamma}(\mathbf{x}).
\]
From Theorem \ref{thm:OU_convergence} and the triangle inequality, we have 
\begin{align*} 
\|\log z_t(l_t(\cdot))\|_{H^1(\gamma)} 
&\leq\left(\|\log\frac{d\nu_t}{d\gamma}\|_{H^1(\gamma)}+\|\log\frac{d\mu_t}{d\gamma}\|_{H^1(\gamma)}\right)\\
&\leq\frac{e^{-t}}{m}\left(\|\frac{d\nu_0}{d\gamma}-1\|_{H^1(\gamma)}+\|\frac{d\mu_0}{d\gamma}-1\|_{H^1(\gamma)}\right). 
\end{align*}
By definition of the $H^1(\gamma)$ norm, 
\begin{align*} 
\|\nabla\log z_t\|_{L^2(\gamma)} 
&\leq\frac{e^{-t}}{m}\left(\|\frac{d\nu_0}{d\gamma}-1\|_{H^1(\gamma)}+\|\frac{d\mu_0}{d\gamma}-1\|_{H^1(\gamma)}\right). 
\end{align*} 
Since $\nabla\log z_t(l_t(\mathbf{x}_t))$ can be expanded as $\partial_u\log z_t\nabla l_t(\mathbf{x}_t)$, 
\[ 
\|\nabla\log z_t(l_t(\cdot))\|_{L^2(\gamma)}=\|\partial_u\log z_t\nabla l_t(\mathbf{x}_t)\|_{L^2(\gamma)},
\]
which decays exponentially.
Finally, as $c_t=\partial_u \log z_t$ and $\tilde c_t=e^{-t}c_0$, we can write 
\[
\tilde c_t - c_t=e^{-t}\partial_u \log z_0 - \partial_u \log z_t.
\]
Using the triangle inequality,
\begin{align*} 
\|(\tilde c_t - c_t)(l_t(\cdot))\nabla l_t\|_{L^2(\gamma)} 
\leq&~e^{-t}\|c_0\|_{L^\infty}\|\nabla l_t\|_{L^2(\gamma)}\\&+\|\nabla\log z_t(l_t)\|_{L^2(\gamma)}.
                                                         
                                                    
\end{align*}
Since Theorem \ref{thm:OU_convergence} yields 
\[
\|\nabla l_t - \nabla \log \gamma\|_{L^2(\gamma)}
=\|\nabla l_t + x\|_{L^2(\gamma)}
\le C_1e^{-t},
\]
we obtain that $\|\nabla l_t\|_{L^2(\gamma)}$ is uniformly bounded over $t$.
Thus, 
\begin{align*}
\|(\tilde c_t&-c_t)(l_t(\cdot))\,\nabla \log p_t(\cdot)\|_{L^2(\gamma)}
\le
C e^{-t},\\
\text{where }C
=&~\|c_0\|_{L^\infty(\gamma)}\sup_{t\in[0,T]}\|\nabla l_t\|_{L^2(\gamma)}\\
&+\frac{1}{m}(\|\frac{d\nu_0}{d\gamma}-1\|_{H^1(\gamma)}+\|\frac{d\mu_0}{d\gamma}-1\|_{H^1(\gamma)})
\end{align*}
       
    
                                                                                                                                                    
    
                                                                                                                          
and therefore the right-hand side is $O(e^{-t})$.

\label{rem:logz_evolution}
Under the control $\tilde c_t(u)=c_0 e^{-t}$, where $c_0=-\frac{1}{2a}+\frac{1}{2b}$, the corresponding proxy log-RN derivative takes the approximate form
\[
\log z_t(u) \approx c_0 e^{-t} u + \beta(t)\implies z_t(u)\approx\exp(c_0e^{-t}u+\beta(t)),
\]
where $\beta(t)$ is determined by the normalization condition $\int z_t  dL_t = 1$ and the evolution of the measure $L_t$.
                                           
    
                                                         
    
                                                                                                                                 
This ensures a smooth interpolation between $z_T\approx 1$ and $z_0$ as given in Fact~\ref{fact:rn_derivative}. The linear dependence on $u$ greatly simplifies the implementation and analysis.
