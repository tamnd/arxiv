\label{Thm. A}
		Let $k\ge3$ be an integer and $0<\delta\le 1$ be a real number. Then for every sufficiently large prime $p$, any subset $A\subseteq\mathcal{P}_p$ with $\#A/\#\mathcal{P}_p\ge \delta$ contains a non-trivial arithmetic progression of length $k$ over $\mathbb{F}_p$.

\label{Thm. B}
		Fix an integer $k\ge3$ and a real number $\varepsilon\in(0,1)$. For every sufficiently large prime $p$, there is a subset $A\subseteq\mathcal{P}_p$ with $\#A=\lfloor (\#\mathcal{P}_p)^{1-\varepsilon}\rfloor$ such that $A$ contains no nontrivial arithmetic progression of length $k$.

<!-- definition-1: body_md -->
\label{Def. linear forms condition}
		Let $k$ be a positive integer and let $N$ be a sufficiently large integer. A nonnegative function $v_N: \mathbb{Z}/N\mathbb{Z}\rightarrow\mathbb{R}_{\ge 0}$ is said to obey the $k$-linear forms condition if one has 
		$$\mathbb{E}\left[\prod_{j=1}^{k}\prod_{\substack{{\bm \omega}=(\omega_i)\\ i\in[1,k]\setminus\{j\} \\ \omega_i\in \{0,1\}}}v_N\left(\sum_{i\in[1,k]\setminus\{j\}}(i-j)x_i^{(\omega_i)}\right)^{n_{j,{\bm \omega}}}: x_1^{(0)},x_1^{(1)},\cdots, x_k^{(0)},x_k^{(1)}\in\mathbb{Z}/N\mathbb{Z}\right]=1+o(1),$$
		for any choices of exponents $n_{j,{\bm \omega}}\in\{0,1\}$.
<!-- end definition-1 -->
<!-- theorem-3: body_md -->
\label{Thm. CFZ}
		For every integer $k\ge3$ and real number $\delta>0$, there exists $c>0$ such that if $v_N: \mathbb{Z}/N\mathbb{Z}\rightarrow\mathbb{R}_{\ge 0}$ satisfies the $k$-linear forms condition, $N$ is sufficiently large, and $f: \mathbb{Z}/N\mathbb{Z}\rightarrow\mathbb{R}_{\ge 0}$ satisfies $0\le f(x)\le v_N(x)$ for all $x\in \mathbb{Z}/N\mathbb{Z}$ and $\mathbb{E}[f(x): x\in \mathbb{Z}/N\mathbb{Z}]\ge \delta$, then 
		$$\mathbb{E}\left[f(x)f(x+d)\cdots f(x+(k-1)d): x, d\in\mathbb{Z}/N\mathbb{Z}\right]\ge c.$$

<!-- lemma-1: body_md -->
\label{Lem. characteristic function of P}
		Let $p$ be a prime. Then 
		$$1_{\mathcal{P}_p}(x)=\theta_{p-1}\sum_{d\mid p-1}\frac{\mu(d)}{\varphi(d)}\sum_{\substack{\chi\in\widehat{\mathbb{F}_p^*} \\ {\rm ord}(\chi)=d}} \chi(x)=\theta_{p-1}\sum_{\chi\in\widehat{\mathbb{F}_p^*}}c_{\chi}\cdot \chi(x)
		=\begin{cases}
			1  & \mbox{if}\ x\in\mathcal{P}_p,\\
			0 & \mbox{otherwise},
		\end{cases}$$
		where $\mu$ is the M\"obius function, $\varphi$ is the Euler totient function, $\theta_{p-1}=\frac{\varphi(p-1)}{(p-1)}$, and $c_{\chi}=\frac{\mu({\rm ord}(\chi))}{\varphi({\rm ord}(\chi))}$ for any $\chi\in\widehat{\mathbb{F}_p^*}$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{Lem. the Weil Bound}
		Let $\chi\in\widehat{\mathbb{F}_p^*}$ with ${\rm ord}(\chi)=d>1$, and let $f(t)\in\mathbb{F}_p[t]$ be a monic polynomial with $f(t)\neq g(t)^d$ for any $g(t)\in\mathbb{F}_p[t]$. Then, for any $a\in\mathbb{F}_p$ we have 
		$$\left|\sum_{x\in\mathbb{F}_p}\chi(af(x))\right|\le (r-1)\sqrt{p},$$
		where $r$ is the number of distinct roots of $f(t)$ in an algebraic closure $\overline{\mathbb{F}_p}$ of $\mathbb{F}_p$.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{Lem. Alon}
		Let $F$ be an arbitrary field, and let $f(x_1,\cdots,x_m)\in F[x_1,\cdots,x_m]$. Suppose $\deg(f)=t_1+t_2+\cdots+t_m$, where each $t_i$ is a nonnegative integer, and suppose that the coefficient of $x_1^{t_1}\cdots x_m^{t_m}$ in $f$ is nonzero. Then, if $S_1,\cdots,S_m$ are subsets of $F$ with $\#S_i>t_i$, there are $s_1\in S_1, s_2\in S_2,\cdots, s_m\in S_m$ so that 
		$$f(s_1,s_2,\cdots,s_m)\neq 0.$$

<!-- lemma-4: body_md -->
\label{Lem. estimates for nontrival character sums}
		Let $m,s\ge2$ be integers, and let $L_i(x_1,x_2,\cdots,x_m)\in\mathbb{Z}[x_1,x_2,\cdots,x_m]$ with $\deg(L_i)=1$ for any $1\le i\le s$. Suppose  that $L_i$ and $L_j$ are not proportional over $\mathbb{Q}$ for any $1\le i\neq j\le s$. Then, for every sufficiently large prime $p$ we have 
		\begin{equation}\label{Eq. inequality in Lemma estimates for nontrival character sums}
			\left|\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_i(L_i({\bm x}))\right|\le (s-1)\cdot p^{m-\frac{1}{2}}+\frac{s(s-1)}{2}\cdot p^{m-1},
		\end{equation}
		where $\chi_1,\chi_2,\cdots,\chi_s\in\widehat{\mathbb{F}_p^*}$ are not all trivial.
<!-- end lemma-4 -->
<!-- proof-1: body_md -->
For each integer $1\le i\le s$, let 
		$$L_i({\bm x})=a_{i1}x_1+a_{i2}x_2+\cdots+a_{im}x_m+b_i={\bm a}_i\cdot {\bm x}+b_i,$$
		where ${\bm a}_i=(a_{i1},a_{i2},\cdots,a_{im})\neq \0$, and let $\widetilde{{\bm a}_i}=(a_{i1},a_{i2},\cdots,a_{im},b_i)$. 
		
		For any prime $p$, we may view $L_1, L_2, \cdots,L_s$ as polynomials over $\mathbb{F}_p$. We first show that for sufficiently large $p$, when viewed as polynomials over $\mathbb{F}_p$, we still have $\deg(L_i)=1$ for any $1\le i\le s$ and $L_i$ and $L_j$ are not proportional over $\mathbb{F}_p$ for any $1\le i\neq j\le s$. In fact, given any $1\le i\neq j\le s$, since $L_i$ and $L_j$ are not proportional over $\mathbb{Q}$, the vectors $\widetilde{{\bm a}_i}$ and $\widetilde{{\bm a}_j}$ are linearly independent over $\mathbb{Q}$. From this, the rank of the matrix $\begin{bmatrix}
			\widetilde{{\bm a}_i}\\
			\widetilde{{\bm a}_j}
		\end{bmatrix}$ is equal to $2$, and hence this matrix
		has a nonzero $2\times 2$ minor $D_{ij}$. Applying this and noting that 
		$$d_i=\max\left\{|a_{ij}|: 1\le j\le m\right\}>0$$
		for each $1\le i\le s$, 
		 when 
		$$p>p_0=\max\left\{s+1, \prod_{1\le i<j\le s}|D_{ij}|\cdot \prod_{1\le i\le s}d_i\right\},$$
		it is clear that $\deg(L_i)=1$ for any $1\le i\le s$ and $L_i$ and $L_j$ are not proportional over $\mathbb{F}_p$ for any $1\le i\neq j\le s$.
		
		We next show that for any $p>p_0$, there exists a vector ${\bm v}\in\mathbb{F}_p^m$ such that ${\bm v}\cdot {\bm a}_i\neq 0$ for any $1\le i\le s$. In fact, since $p>p_0$, the vectors ${\bm a}_i\in\mathbb{F}_p^m\setminus\{\0\}$ for each $1\le i\le s$. Thus, 
		$$\left\{{\bm v}\in\mathbb{F}_p^m: {\bm v}\cdot {\bm a}_i=0\right\}$$
		is a subspace of $\mathbb{F}_p^m$ with dimension $m-1$ for any $1\le i\le s$. From this and noting that $p>p_0>s$, one can verify that 
		\begin{align*}
			\#\left\{{\bm v}\in\mathbb{F}_p^m: {\bm v}\cdot {\bm a}_i=0\ \text{for some $1\le i\le s$}\right\}
&=\#\bigcup_{i=1}^s\left\{{\bm v}\in\mathbb{F}_p^m: {\bm v}\cdot {\bm a}_i=0\right\}\\
&\le \sum_{i=1}^s\#\left\{{\bm v}\in\mathbb{F}_p^m: {\bm v}\cdot {\bm a}_i=0\right\}\\
&=sp^{m-1}\\
&<p^m.
		\end{align*}
		Hence there exists a vector ${\bm v}\in\mathbb{F}_p^m$ such that ${\bm v}\cdot {\bm a}_i\neq 0$ for any $1\le i\le s$ whenever $p>p_0$. 
		
		From now on, we assume $p>p_0$. Since $m\ge2$, there is a subspace $W$ of $\mathbb{F}_p^m$ with $\dim(W)=m-1$ such that 
		$$\mathbb{F}_p^m=W\bigoplus\mathbb{F}_p{\bm v}.$$
		This implies that every vector ${\bm x}\in\mathbb{F}_p^m$ can be uniquely written as 
		$${\bm x}={\bm w}+t{\bm v} \quad ({\bm w}\in W, t\in\mathbb{F}_p).$$
		Hence, for any ${\bm x}={\bm w}+t{\bm v}\in\mathbb{F}_p^m$ with ${\bm w}\in W$ and $t\in\mathbb{F}_p$ and any integer $1\le i\le s$, letting 
		\begin{equation}\label{Eq. definition of ci}
			c_i={\bm v}\cdot{\bm a}_i\neq 0,
		\end{equation}
		one can verify that 
		\begin{align}\label{Eq. expression of Li}
			L_i({\bm x})=L_i({\bm w}+t{\bm v})
&={\bm a}_i\cdot ({\bm w}+t{\bm v})+b_i\notag\\
&=L_i({\bm w})+c_it\notag\\
&=c_i(t-\alpha_i({\bm w})),
		\end{align}
where 
      \begin{equation}\label{Eq. definition of alpha i}
	\alpha_i({\bm w})=-\frac{L_i({\bm w})}{c_i}.
       \end{equation}


Define 
$$W_{{\rm bad}}=\left\{{\bm w}\in W: \alpha_i({\bm w})=\alpha_j({\bm w})\ \text{for some $1\le i\neq j\le s$}\right\}$$
and let $W_{{\rm good}}=W\setminus W_{{\rm bad}}$. We next obtain an upper bound for $\# W_{{\rm bad}}$. For any $1\le i\neq j\le s$, let the polynomial
$$L_{ij}=c_jL_i-c_iL_j.$$
Since $p>p_0$, by the above discussions, the polynomials $L_i$ and $L_j$ are not proportional over $\mathbb{F}_p$. From this and recalling that  $c_i,c_j\neq 0$ are defined by (\ref{Eq. definition of ci}), we immediately obtain $L_{ij}\neq 0$. On the other hand, for any ${\bm x}={\bm w}+t{\bm v}\in\mathbb{F}_p^m$ with ${\bm w}\in W$ and $t\in\mathbb{F}_p$, applying (\ref{Eq. expression of Li}) and (\ref{Eq. definition of alpha i}) one can verify that 
\begin{align}\label{Eq. Lij is independent of tv}
	L_{ij}({\bm x})=L_{ij}({\bm w}+t{\bm v})
&=c_jL_i({\bm w}+t{\bm v})-c_iL_j({\bm w}+t{\bm v})\notag\\
&=c_j\left(c_i(t-\alpha_i({\bm w}))\right)-c_i\left(c_j(t-\alpha_j({\bm w}))\right)\notag\\
&=-c_jc_i\alpha_i({\bm w})+c_ic_j\alpha_j({\bm w})\notag\\
&=c_jL_i({\bm w})-c_iL_j({\bm w})\notag\\
&=L_{ij}({\bm w}).
\end{align}
We now claim that there is at least one ${\bm w}\in W$ such that $L_{ij}({\bm w})\neq 0$. Suppose, to the contrary, that $L_{ij}({\bm w})=0$ for any ${\bm w}\in W$. Then (\ref{Eq. Lij is independent of tv}) implies that $L_{ij}({\bm x})=0$ for any ${\bm x}\in\mathbb{F}_p^m$. Thus, applying Lemma \ref{Lem. Alon} with $F=\mathbb{F}_p$ and $S_1=\cdots=S_m=\mathbb{F}_p$, we obtain $L_{ij}=0$, which contradicts the fact that $L_{ij}\neq 0$. By (\ref{Eq. Lij is independent of tv}) and the above discussions, 
$$\left\{{\bm w}\in W: L_{ij}({\bm w})=0\right\}\subsetneqq W$$
is either $\emptyset$ or a proper affine hyperplane in $W$. From this and noting that 
$$\left\{{\bm w}\in W: \alpha_i({\bm w})=\alpha_j({\bm w})\right\}=\left\{{\bm w}\in W: L_{ij}({\bm w})=0\right\},$$
we obtain 
$$\#\left\{{\bm w}\in W: \alpha_i({\bm w})=\alpha_j({\bm w})\right\}\le p^{\dim(W)-1}=p^{m-2}.$$
Hence 
\begin{align}\label{Eq. upper bound for W bad}
	\#W_{{\rm bad}}
&=\#\left\{{\bm w}\in W: \alpha_i({\bm w})=\alpha_j({\bm w})\ \text{for some $1\le i\neq j\le s$}\right\}\notag\\
&=\#\bigcup_{1\le i<j\le s}\left\{{\bm w}\in W: \alpha_i({\bm w})=\alpha_j({\bm w})\right\}\notag\\
&\le \sum_{1\le i<j\le s}\#\left\{{\bm w}\in W: \alpha_i({\bm w})=\alpha_j({\bm w})\right\}\notag\\
&\le \binom{s}{2}p^{m-2}.
\end{align}

Finally, we prove the inequality (\ref{Eq. inequality in Lemma estimates for nontrival character sums}). As $\chi_1,\cdots,\chi_s$ are not all trivial, we have 
$$l={\rm lcm}\{{\rm ord}(\chi_i): 1\le i\le s\}>1.$$
Fix a character $\psi\in\widehat{\mathbb{F}_p^*}$ with ${\rm ord}(\psi)=l$. Then there are integers $1\le e_1,\cdots, e_s\le l$ such that $\chi_i=\psi^{e_i}$ for any $1\le i\le s$. Since $\chi_1,\cdots,\chi_s$ are not all trivial, at least one of $e_1,\cdots,e_s$ is less than $l$. By this and (\ref{Eq. expression of Li}), one can verify that 
\begin{align}\label{Eq. sum is equal to good plus bad}
	\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_i(L_i({\bm x}))
&=\sum_{{\bm w}\in W}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\chi_i(L_i({\bm w}+t{\bm v}))\notag\\
&=\sum_{{\bm w}\in W}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\chi_i(c_i(t-\alpha_i({\bm w})))\notag\\
&=\sum_{{\bm w}\in W}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\notag\\
&=\sum_{{\bm w}\in W_{{\rm good}}}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))+\sum_{{\bm w}\in W_{{\rm bad}}}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w}))).
\end{align}
	
	For any ${\bm w}\in W$, let 
	$$f_{{\bm w}}(x)=\prod_{1\le i\le s}\left(x-\alpha_i({\bm w})\right)^{e_i}\in\mathbb{F}_p[x].$$
	If ${\bm w}\in W_{{\rm good}}$, then $\alpha_1({\bm w}),\alpha_2({\bm w}),\cdots, \alpha_s({\bm w})$ are pairwise distinct. Noting that $1\le e_1,\cdots,e_s\le l$ and $\min\{e_i: 1\le i\le s\}<l$, we have 
		$$f_{{\bm w}}(x)\neq g(x)^l$$
		for any $g(x)\in\mathbb{F}_p[x]$.  Applying Lemma \ref{Lem. the Weil Bound}, for each ${\bm w}\in W_{{\rm good}}$ we obtain 
$$\left|\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\right|=\left|\sum_{t\in\mathbb{F}_p}\psi\left(cf_{{\bm w}}(t)\right)\right|\le (s-1)\sqrt{p},$$
where 
$$c=\prod_{1\le i\le s}c_i^{e_i}\in\mathbb{F}_p^*.$$
Hence, 
\begin{align}\label{Eq. estimate for good w}
	     \left|\sum_{{\bm w}\in W_{{\rm good}}}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\right|
&\le  \sum_{{\bm w}\in W_{{\rm good}}}\left|\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\right|\notag\\
&\le \#W_{{\rm good}}\cdot (s-1)\cdot \sqrt{p}\notag\\
&\le \#W\cdot (s-1)\cdot \sqrt{p}\notag\\
&=(s-1)\cdot p^{m-\frac{1}{2}}.
\end{align}		
On the other hand, by (\ref{Eq. upper bound for W bad}) one can verify that 
\begin{align}\label{Eq. estimate for bad w}
	\left|\sum_{{\bm w}\in W_{{\rm bad}}}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\right|
&\le  \sum_{{\bm w}\in W_{{\rm bad}}}\left|\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\right|\notag\\
&\le \sum_{{\bm w}\in W_{{\rm bad}}}p\notag\\
&=\#W_{{\rm bad}}\cdot p \notag\\
&\le \binom{s}{2}\cdot p^{m-1}.
\end{align}
		
Combining (\ref{Eq. estimate for good w}) and (\ref{Eq. estimate for bad w}) with (\ref{Eq. sum is equal to good plus bad}), we obtain 
\begin{align*}
	 &\left|\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_i(L_i({\bm x}))\right|\\
\le &\left|\sum_{{\bm w}\in W_{{\rm good}}}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\right|+\left|\sum_{{\bm w}\in W_{{\rm bad}}}\sum_{t\in\mathbb{F}_p}\prod_{i=1}^{s}\psi^{e_i}(c_i(t-\alpha_i({\bm w})))\right|\\
\le & (s-1)\cdot p^{m-\frac{1}{2}}+\frac{s(s-1)}{2}\cdot p^{m-1}.
\end{align*}
		
In view of the above, we have completed the proof.

<!-- lemma-5: body_md -->
\label{Lem. W p-1 is very small}
		For any real number $\varepsilon>0$, if $p$ is sufficiently large, then $W_{p-1}\ll_{\varepsilon}p^{\varepsilon}$.
<!-- end lemma-5 -->
<!-- proof-2: body_md -->
Let 
$$\omega_{n}=\#\left\{q: q\mid n\ \text{and}\ q\ \text{is a prime}\right\}$$
     be the number of all distinct prime divisors of $n$. Robin \cite[Theorem 11]{Robin} proved that for any integer $n\ge 3$, we have 
     	$$\omega_{n}<1.3841\frac{\log (n)}{\log\log (n)}.$$
     	Thus, applying this with $n=p-1$, we obtain
     	$$W_{p-1}=2^{\omega_{p-1}}=e^{\omega_{p-1}\cdot \log 2}\le (p-1)^{\frac{1.3841\cdot\log 2}{\log\log(p-1)}}\ll_{\varepsilon}p^{\varepsilon}$$
     	for any sufficiently large prime $p$. This completes the proof.

<!-- lemma-6: body_md -->
\label{Lem. vp satisfies the k linear forms condition}
			Let $m,s$ be nonnegative integers with $m\ge2$, and let $L_i(x_1,x_2,\cdots,x_m)\in\mathbb{Z}[x_1,x_2,\cdots,x_m]$ with $\deg(L_i)=1$ for any $1\le i\le s$. Suppose  that $L_i$ and $L_j$ are not proportional over $\mathbb{Q}$ for any $1\le i\neq j\le s$. Then, for every sufficiently large prime $p$ we have 
			$$\mathbb{E}\left[\prod_{i=1}^{s}v_p(L_i({\bm x})): {\bm x}\in\mathbb{F}_p^m\right]=\frac{1}{p^m}\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}v_p(L_i({\bm x}))=1+o(1).$$
		   In particular, for any integer $k\ge3$, the function $v_p$ satisfies the $k$-linear forms condition.
<!-- end lemma-6 -->
<!-- proof-3: body_md -->
When $s=0$, i.e., no factor is selected, the expectation is exactly $1$. For $s=1$, as $p$ is sufficiently large, the reduction of $L_1$ modulo $p$ has degree $1$. Hence, for any $y\in\mathbb{F}_p$ we have 
		$$\#\left\{{\bm x}\in\mathbb{F}_p^m: L_1({\bm x})=y\right\}=p^{m-1}.$$
		Applying this and noting that
		$$\sum_{y\in\mathbb{F}_p}\chi(y)=
		\begin{cases}
			p-1 & \mbox{if}\ \chi=\chi_0,\\
			0    & \mbox{otherwise},
		\end{cases}$$
		we obtain 
		\begin{align*}
			\mathbb{E}\left[v_p(L_1({\bm x})): {\bm x}\in\mathbb{F}_p^m\right]
&=\frac{1}{p^m}\sum_{{\bm x}\in\mathbb{F}_p^m}\sum_{\chi\in\widehat{\mathbb{F}_p^*}}c_{\chi}\cdot \chi(L_1({\bm x}))\\
&=\frac{1}{p^m}\sum_{\chi\in\widehat{\mathbb{F}_p^*}}\sum_{{\bm x}\in\mathbb{F}_p^m}c_{\chi}\cdot \chi(L_1({\bm x}))\\
&=\frac{1}{p^m}\sum_{\chi\in\widehat{\mathbb{F}_p^*}}\sum_{y\in\mathbb{F}_p}c_{\chi}\cdot \chi(y)\#\left\{{\bm x}\in\mathbb{F}_p^m: L_1({\bm x})=y\right\}\\
&=\frac{1}{p}\sum_{\chi\in\widehat{\mathbb{F}_p^*}}c_{\chi}\sum_{y\in\mathbb{F}_p}\chi(y)\\
&=1-\frac{1}{p}\\
&=1+o(1).
		\end{align*}
		
		Suppose now $s\ge2$. By Lemma \ref{Lem. characteristic function of P} and (\ref{Eq. definition of vp}), one can verify that 
		\begin{align}\label{Eq. sum is trivial plus nontrivial}
			\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}v_p(L_i({\bm x}))
&=\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\sum_{\chi\in\widehat{\mathbb{F}_p^*}}c_{\chi}\cdot \chi(L_i({\bm x}))\notag\\
&=\sum_{{\bm x}\in\mathbb{F}_p^m}\sum_{\chi_1,\cdots,\chi_s\in\widehat{\mathbb{F}_p^*}}\prod_{i=1}^{s}c_{\chi_i}\cdot \chi_i(L_i({\bm x}))\notag\\
&=\sum_{\chi_1,\cdots,\chi_s\in\widehat{\mathbb{F}_p^*}}\prod_{i=1}^{s}c_{\chi_i} \sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_i(L_i({\bm x}))\notag\\
&=S_1+S_2,
		\end{align}
	where 
	$$S_1=\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_0(L_i({\bm x})),$$
	and
	$$S_2=\sum_{\substack{\chi_1,\cdots,\chi_s\in\widehat{\mathbb{F}_p^*}\\ \chi_1,\cdots,\chi_s\ \text{are not all trivial}}}\prod_{i=1}^{s}c_{\chi_i} \sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_i(L_i({\bm x})).$$
		
		
		For $S_1$, we clearly have 
		\begin{equation}\label{Eq. upper bound for trivial}
			\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_0(L_i({\bm x}))\le p^m.
		\end{equation}
		On the other hand, since $p$ is sufficiently large, the reduction of $L_i$ modulo $p$ has degree $1$. This implies 
		$$\#\left\{{\bm x}\in\mathbb{F}_p^m: L_i({\bm x})=0\right\}=p^{m-1}.$$
		From this, one can verify that 
		\begin{align}\label{Eq. lower bound for trivial}
			\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_0(L_i({\bm x}))
&=\#\left\{{\bm x}\in\mathbb{F}_p^m: L_i({\bm x})\neq 0\ \text{for any $1\le i\le s$}\right\}\notag\\
&=p^m-\#\left\{{\bm x}\in\mathbb{F}_p^m: L_i({\bm x})=0\ \text{for some $1\le i\le s$}\right\}\notag\\
&=p^m-\#\bigcup_{1\le i\le s}\left\{{\bm x}\in\mathbb{F}_p^m: L_i({\bm x})=0\right\}\notag\\
&\ge p^m-\sum_{1\le i\le s}\#\left\{{\bm x}\in\mathbb{F}_p^m: L_i({\bm x})=0\right\}\notag\\
&\ge p^m-s\cdot p^{m-1}.
		\end{align}
		
		We next consider $S_2$. For any positive divisor $d$ of $p-1$, it is known that
		$$\#\left\{\chi\in\widehat{\mathbb{F}_p^*}: {\rm ord}(\chi)=d\right\}=\varphi(d).$$
		Hence, by Lemma \ref{Lem. characteristic function of P} we obtain 
		\begin{align}\label{Eq. sum of c chi}
			\sum_{\chi\in\widehat{\mathbb{F}_p^*}}|c_{\chi}|
&=\sum_{\chi\in\widehat{\mathbb{F}_p^*}}\frac{|\mu({\rm ord}(\chi))|}{\varphi({\rm ord}(\chi))}\notag\\
&=\sum_{d\mid p-1}\frac{|\mu(d)|}{\varphi(d)}\cdot \#\left\{\chi\in\widehat{\mathbb{F}_p^*}: {\rm ord}(\chi)=d\right\}\notag\\
&=\sum_{d\mid p-1}|\mu(d)|\notag\\
&=W_{p-1}.
		\end{align}
		Assembling (\ref{Eq. sum of c chi}) and Lemma \ref{Lem. estimates for nontrival character sums} gives 
		\begin{align}\label{Eq. upper bound for the nontrivial}
	 |S_2|
&\le \sum_{\substack{\chi_1,\cdots,\chi_s\in\widehat{\mathbb{F}_p^*}\\ \chi_1,\cdots,\chi_s\ \text{are not all trivial}}}\prod_{i=1}^{s}|c_{\chi_i}| \left|\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}\chi_i(L_i({\bm x}))\right|\notag\\
&\le \left((s-1)\cdot p^{m-\frac{1}{2}}+\frac{s(s-1)}{2}\cdot p^{m-1}\right)\cdot \sum_{\chi_1,\cdots,\chi_s\in\widehat{\mathbb{F}_p^*}}\prod_{i=1}^{s}|c_{\chi_i}|\notag\\
&=\left((s-1)\cdot p^{m-\frac{1}{2}}+\frac{s(s-1)}{2}\cdot p^{m-1}\right)\cdot \prod_{i=1}^{s}\sum_{\chi_i\in\widehat{\mathbb{F}_p^*}}|c_{\chi_i}|\notag\\
&=\left((s-1)\cdot p^{m-\frac{1}{2}}+\frac{s(s-1)}{2}\cdot p^{m-1}\right)\cdot W_{p-1}^s.
		\end{align}
	Combining (\ref{Eq. upper bound for trivial}), (\ref{Eq. lower bound for trivial}) and (\ref{Eq. upper bound for the nontrivial}) with (\ref{Eq. sum is trivial plus nontrivial}), we obtain 
	$$ 1-\left(\frac{s-1}{\sqrt{p}}+\frac{s(s+1)}{2p}\right)\cdot W_{p-1}^s
	\le \frac{1}{p^m}\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}v_p(L_i({\bm x}))\le 1+\left(\frac{s-1}{\sqrt{p}}+\frac{s(s-1)}{2p}\right)\cdot W_{p-1}^s.$$
	Applying Lemma \ref{Lem. W p-1 is very small} to the above inequality, we obtain 
	$$\mathbb{E}\left[\prod_{i=1}^{s}v_p(L_i({\bm x})): {\bm x}\in\mathbb{F}_p^m\right]=\frac{1}{p^m}\sum_{{\bm x}\in\mathbb{F}_p^m}\prod_{i=1}^{s}v_p(L_i({\bm x}))=1+o(1).$$
		
	Finally, observe that each polynomial appearing in Definition \ref{Def. linear forms condition} has the form 
	$$L_{j,{\bm{\omega}}}=\sum_{i\in[1,k]\setminus\{j\}}(i-j)x_i^{(\omega_i)}.$$
	
	When $j\neq j'$, the polynomial $L_{j,{\bm{\omega}}}$ omits the variables $x_j^{(0)}$ and $x_j^{(1)}$, while $L_{j',{\bm{\omega}}'}$ contains exactly one of them. This implies that $L_{j,{\bm{\omega}}}$ and $L_{j',{\bm{\omega}'}}$ are not proportional. If $j=j'$ but $\bm{\omega}\neq \bm{\omega}'$, then for some $i\neq j$, one polynomial contains $x_i^{(0)}$ but not $x_i^{(1)}$, and the other precisely contains $x_i^{(1)}$ but not $x_i^{(0)}$. Hence, these two polynomials cannot be proportional. Applying the above discussions, we immediately see that $v_p$ satisfies the $k$-linear forms condition for any integer $k\ge3$.
		
	In view of the above, we have completed the proof.

\label{Lem. lift mod p to Z}
		Let $k\ge3$ be an integer and let $p$ be an odd prime. Let $S\subseteq\mathbb{Z}$ with $\#S\ge2$ and 
		$${\rm diam}(S)=\sup\left\{|x-y|: x,y\in S\ \text{and}\ x\neq y\right\}<\frac{p}{2}.$$
		Suppose that there are $b_0,b_1,\cdots,b_{k-1}\in S$ such that 
		$$\overline{b_0},\overline{b_1},\cdots,\overline{b_{k-1}}$$
		forms a nontrivial $k$-term arithmetic progression over $\mathbb{F}_p$. Then $b_0,b_1,\cdots,b_{k-1}$ is a nontrivial $k$-term arithmetic progression over $\mathbb{Z}$.
<!-- end lemma-8 -->
<!-- proof-4: body_md -->
Since $\overline{b_0},\overline{b_1},\cdots,\overline{b_{k-1}}$ forms a nontrivial $k$-term arithmetic progression over $\mathbb{F}_p$, there is an integer $d$ with $d\not\equiv 0\pmod{p}$ such that 
		$$b_j\equiv b_0+jd\pmod{p}$$
		for any $0\le j\le k-1$. Thus, for any integer $j\in[0,k-3]$ we have 
		$$b_j-2b_{j+1}+b_{j+2}\equiv 0\pmod{p}.$$
		Noting that 
		$$\left|b_j-2b_{j+1}+b_{j+2}\right|\le \left|b_j-b_{j+1}\right|+\left|b_{j+2}-b_{j+1}\right|\le 2\cdot {\rm diam}(S)<p,$$
		the above congruence yields 
		$$b_j-2b_{j+1}+b_{j+2}=0$$
		for any $0\le j\le k-3$. This, together with $b_1-b_0\equiv d\not\equiv 0\pmod{p}$, implies that 
		$$b_0,b_1,\cdots,b_{k-1}$$ 
		forms a nontrivial $k$-term arithmetic progression over $\mathbb{Z}$.

<!-- lemma-9: body_md -->
\label{Lem. the Behrend theorem}
		There is an absolute constant $C>0$ such that for every sufficiently large integer $N$, there is a subset 
		$$B\subseteq\left\{1,2,\cdots,N\right\}\subseteq\mathbb{Z}^+$$
		with $\#B\ge N\cdot\exp\left(-C\sqrt{\log N}\right)$ such that $B$ contains no nontrivial $3$-term arithmetic progression over $\mathbb{Z}$.
