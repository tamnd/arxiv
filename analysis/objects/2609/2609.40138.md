<!-- theorem-1: body_md -->
\label{thm:main}
Let $\mathcal {T}_n$ be a uniformly random (rooted Type~I) triangulation of the sphere with $n$ vertices, viewed as a metric space equipped with its graph distance. Then for any $\delta>0$,
\[\mathbf{P}(\mathrm{Dis}(\mathcal {T}_n) \ge (\log n)^{1/4-\delta}) \to 1 \quad\text{as}\quad n \to \infty.\]
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
\label{def:bubble}
Let $(i,j)$ be a vertex of $\mathcal {L}$ and let $k \ge 1$ be an integer.  The \emph{bubble} started at $(i,j)$ with \emph{initial height} $k$ is the sub-triangulation of $\mathcal {L}$ lying in between the bottom-most geodesics started from $(i,j)$ and from $(i,j+k)$, up to the time at which these two geodesics coalesce.  We call the bottom-most geodesic started from $(i,j)$ the \emph{bottom geodesic} of the bubble, the bottom-most geodesic started from $(i,j+k)$ its \emph{top geodesic}, and $i$ its \emph{starting time}.  For $t \ge 0$, the \emph{height of the bubble at time $t$} is the difference between the heights of its top and its bottom geodesic at time $i+t$. The \emph{lifetime} of the bubble is the first time at which its height vanishes, that is, the time taken by the two geodesics to coalesce.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:generating_function}
For $x \in [0,1)$, let $f(x)=\sum_{k=0}^{\infty} \theta(k)x^k$ be the generating function of $\theta$, let $f_1(x)=f(x)$, and for $n>1$, let $f_n(x)=f_{n-1}(f(x))$ be the $n$-th iterate of $f$ with itself. Then for all $n \ge 1$,
\begin{equation}\label{eq:nth_iterate}
f_n(x)=1-\left(n+\frac{1}{\sqrt{1-x}}\right)^{-2}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:genfn_1}\emph{Step 1. Expanding the right-hand side of~\eqref{eq:nth_iterate} for $n=1$.} Throughout the proof we take $x \in (0,1)$, so that we may divide by $x$; the case $x=0$ of~\eqref{eq:nth_iterate} then follows by continuity (or directly, since $f_n(0)=1-(n+1)^{-2}$ and $\theta(0)=3/4$). We first expand:
\begin{equation}\label{eq:expand_generating_function}
\begin{aligned}
1-\left(1+\frac{1}{\sqrt{1-x}}\right)^{-2}&=1-\frac{1-x}{(1+\sqrt{1-x})^2}=\frac{(1+\sqrt{1-x})^2-(1-x)}{(1+\sqrt{1-x})^2}\\
&=\frac{x^2-(1-x)(1-\sqrt{1-x})^2}{x^2}=\frac{3x-2+2\sqrt{1-x}-2x\sqrt{1-x}}{x^2}.
\end{aligned}
\end{equation}

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:genfn_2}\emph{Step 2. Computing $f$.} On the other hand, it is easy to check that
\[\theta(k)=\frac{\binom{2k}{k}}{4^k (k+1)}-\frac{\binom{2(k+1)}{k+1}}{4^{k+1} (k+2)};\]
indeed, $\binom{2k+2}{k+1}=\frac{2(2k+1)}{k+1}\binom{2k}{k}$, so that the right-hand side equals
\[\frac{\binom{2k}{k}}{2 \cdot 4^k (k+1)(k+2)}\big(2(k+2)-(2k+1)\big)=\frac{3\binom{2k}{k}}{2 \cdot 4^k(k+1)(k+2)},\]
which is $\theta(k)$ as defined in~\eqref{eq:theta_distribution}.
Since
\[c(x)=\frac{1-\sqrt{1-4x}}{2x}\]
is the generating function of the Catalan numbers, we have
\begin{align*}
f(x)&=\sum_{k=0}^{\infty} \theta(k)x^k=c\left(\frac{x}{4}\right)-\frac{1}{x}\left(c\left(\frac{x}{4}\right)-1\right)=\frac{2-2\sqrt{1-x}}{x}-\frac{2-2\sqrt{1-x}}{x^2}+\frac{1}{x}\\
&=\frac{3x-2+2\sqrt{1-x}-2x\sqrt{1-x}}{x^2}.
\end{align*}
This is exactly~\eqref{eq:expand_generating_function}. This proves~\eqref{eq:nth_iterate} for $n=1$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:genfn_3}\emph{Step 3. The induction.} The case $n>1$ follows by induction.  Indeed, the case $n=1$ can be rewritten as $(1-f(x))^{-1/2}=1+(1-x)^{-1/2}$, and hence, assuming~\eqref{eq:nth_iterate} for $n-1$, we get
\[f_n(x)=f_{n-1}(f(x))=1-\big(n-1+(1-f(x))^{-1/2}\big)^{-2}=1-\big(n+(1-x)^{-1/2}\big)^{-2},\]
as desired.  (The inductive hypothesis is applicable at the point $f(x)$ because $f(x) \in [\theta(0),1)=[3/4,1)$ for $x \in [0,1)$, $f$ being the generating function of a non-degenerate probability distribution on $\mathbf{Z}_{\ge0}$.)

<!-- lemma-2: body_md -->
\label{lem:tail_bound_maximum}
For all integers $x \ge y \ge 1$, we have
\begin{equation}\label{eq:upper_tail_bound_maximum}
\mathbf{P}_y\!\left(\sup_{j \ge 0} X_j \ge x\right) \le \frac{y}{x}.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since $\theta$ has mean 1, the process $X$ is a non-negative martingale. Thus by Doob's inequality,
\[\mathbf{P}_y\!\left(\max_{0 \le j \le n} X_j \ge x\right) \le \frac{\mathbf{E}_y(X_n)}{x}=\frac{y}{x}\]

<!-- lemma-3: body_md -->
\label{lem:tail_bound_minimum}
Let $x$ and $n$ be positive integers, and let $a \in (0,x)$. Then
\[\mathbf{P}_x\!\left(\min_{0 \le j \le n}X_j \ge a \right) \ge \left(1-\frac{1}{(n+1)^2}\right)^a-\left(1-\frac{1}{(2n+1)^2}\right)^x.\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:minimum_1}\emph{Step 1. A stopping time.} Let
\[\tau=\min \{k \ge 0: X_k<a\}.\]
Then $\tau$ is a stopping time, and $\{\tau>n\}=\{\min_{0 \le j \le n} X_j \ge a\}$, so that it suffices to bound $\mathbf{P}_x(\tau>n)$ from below.  Note also that $\tau \ge 1$, since $X_0=x>a$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:minimum_2}\emph{Step 2. Decomposition via the strong Markov property.} Since $0$ is an absorbing state for the branching process $X$, on the event $\{\tau \le n\}$ we have that the event $\{X_{2n}>0\}$ is contained in $\{X_{\tau+n}>0\}$, because $\tau+n \le 2n$ there. By the strong Markov property at the bounded stopping time $\min(\tau,n)$, we have
\begin{align*}
\mathbf{P}_x(X_{2n}>0)&\le \mathbf{P}_x(X_{2n}>0, \tau \le n)+\mathbf{P}_x(\tau>n)\\
&\le \mathbf{P}_x(X_{\tau+n}>0, \tau \le n )+\mathbf{P}_x(\tau>n)\\
&= \mathbf{E}_x\!\left(\mathbf{P}_{X_{\tau}}(X_n>0){\bf 1}_{\{\tau \le n\}}\right)+\mathbf{P}_x(\tau>n).
\end{align*}
Since $X_{\tau} < a$ on the event $\{\tau \le n\}$, and $t \mapsto 1-(f_n(0))^{t}$ is increasing in $t$, we have $\mathbf{P}_{X_{\tau}}(X_n>0) \le \mathbf{P}_a(X_n>0) := 1-(f_n(0))^a$ almost surely on $\{\tau \le n\}$. Therefore,
\begin{align*}
\mathbf{P}_x(X_{2n}>0) &\le \mathbf{P}_a(X_n>0)\mathbf{P}_x(\tau \le n)+\mathbf{P}_x(\tau>n)\\
&\le \mathbf{P}_a(X_n>0)+\mathbf{P}_x(\tau>n).
\end{align*}

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:minimum_3}\emph{Step 3. Conclusion.} Rearranging, we obtain
\begin{align*}
\mathbf{P}_x(\tau>n) &\ge \mathbf{P}_x(X_{2n}>0)-\mathbf{P}_a(X_n>0)\\
&= \left(1-\frac{1}{(n+1)^2}\right)^a- \left(1-\frac{1}{(2n+1)^2}\right)^x,
\end{align*}
where we used~\eqref{eq:bubble_lifetime} for both terms; this is the assertion of the lemma, by \Cref{step:minimum_1}.

<!-- lemma-4: body_md -->
\label{lem:expected_length_of_bubble}
Let $T=\min \{k \ge 1: X_k=0\}$ be the extinction time of~$X$. Then
\begin{equation}\label{eq:expected_length_of_bubble}
\mathbf{E}_n(T)=\sqrt{\pi n}+O(1) \quad\text{as}\quad n \to \infty.
\end{equation}
Also
\[\frac{3 \sqrt{n}}{4} \le \mathbf{E}_n(T) \le 2\sqrt{n} \quad\text{for all}\quad n \ge 1.\]
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:explen_1}\emph{Step 1. Reduction to a maximum of continuous random variables.} Under $\mathbf{P}_1$, the law of $T$ is given by
\[\mathbf{P}_1(T>k)=\mathbf{P}_1(X_{k}>0)=\frac{1}{(k+1)^2} \quad\text{for all integers } k \ge 0,\]
and so, since the $n$ initial individuals generate independent trees, under $\mathbf{P}_n$ the extinction time $T$ has the same law as $M_n=\max(T^{(1)},\dots, T^{(n)})$ where the $T^{(i)}$ are i.i.d.\ copies of $T$ under $\mathbf{P}_1$. Let $Y_1, Y_2, \dots$ be i.i.d.\ with
\[\mathbf{P}(Y_1>x)=\frac{1}{(x+1)^2} \quad\text{for}\quad  x \ge 0,\]
that is, they have density $2/(x+1)^3$ for $x \ge 0$. Then the $\lceil Y_i \rceil$ have the same distribution as the $T^{(i)}$. Then setting $T^{(i)}=\lceil Y_i \rceil$ and $M_n'=\max(Y_1, \dots, Y_n)$, we clearly have
\[Y_i \le T^{(i)} \le Y_i+1\]
and
\[M_n' \le M_n \le M_n'+1.\]
Thus it is enough to prove~\eqref{eq:expected_length_of_bubble} for $M_n'$ instead of $M_n$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:explen_2}\emph{Step 2. A recursion for $\mathbf{E}(M_n')$.} Now,
\begin{align*}
\mathbf{E}(M_n')&=\int_0^{\infty} \mathbf{P}(M_n'>x)dx=\int_0^{\infty} 1-\left(1-\frac{1}{(x+1)^2}\right)^n dx\\
&=\int_0^{\infty} 1-\left(1-\frac{1}{(x+1)^2}\right)^{n-1} dx+\int_0^{\infty} \frac{1}{(x+1)^2} \left(1-\frac{1}{(x+1)^2}\right)^{n-1} dx\\
&=\mathbf{E}(M_{n-1}')+\int_0^{\infty} \frac{1}{(x+1)^2} \left(1-\frac{1}{(x+1)^2}\right)^{n-1} dx.
\end{align*}
For this last integral, we perform the substitution $u=1-1/(x+1)^2$. The resulting integrand is proportional to the density of the Beta$(n,1/2)$ distribution, and so we find
\[\int_0^{\infty} \frac{1}{(x+1)^2} \left(1-\frac{1}{(x+1)^2}\right)^{n-1} dx=\frac{1}{2}\int_0^1 \frac{u^{n-1}}{\sqrt{1-u}} du=\frac{\sqrt{\pi}}{2}\frac{\Gamma(n)}{\Gamma(n+1/2)}.\]

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:explen_3}\emph{Step 3. Evaluation of the sum.} Therefore, by telescoping (with the convention $\mathbf{E}(M_0')=0$),
\[\mathbf{E}(M_n')=\sum_{k=1}^n \frac{\sqrt{\pi}}{2}\frac{\Gamma(k)}{\Gamma(k+1/2)}=\frac{\sqrt{\pi}\,\Gamma(n+1)}{\Gamma(n+1/2)}-1=\frac{4^n}{\binom{2n}{n}}-1.\]
The second equality can be checked easily by induction: it holds for $n=1$, both sides being equal to $1$, and if it holds for $n$ then, using $\Gamma(n+3/2)=(n+\frac12)\Gamma(n+\frac12)$,
\[\frac{\sqrt{\pi}\,\Gamma(n+1)}{\Gamma(n+1/2)}+\frac{\sqrt{\pi}}{2}\frac{\Gamma(n+1)}{\Gamma(n+3/2)}=\sqrt{\pi}\,\Gamma(n+1)\,\frac{(n+\frac12)+\frac12}{\Gamma(n+3/2)}=\frac{\sqrt{\pi}\,\Gamma(n+2)}{\Gamma(n+3/2)},\]
which is the corresponding expression for $n+1$.  The third equality follows from the identity $\Gamma(n+1/2)=\sqrt{\pi}\,(2n)!/(4^n n!)$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:explen_4}\emph{Step 4. Asymptotics and explicit bounds.} Stirling's formula yields~\eqref{eq:expected_length_of_bubble} immediately, and for the bounds, we can use the well-known inequalities
\[\frac{1}{2\sqrt{n}} \le \frac{1}{4^n} \binom{2n}{n} \le \frac{1}{\sqrt{\pi n}}\]
to conclude that
\[\mathbf{E}_n(T) \ge \mathbf{E}(M_n') \ge \sqrt{\pi n}-1 \ge (\sqrt{\pi}-1)\sqrt{n} \ge \frac{3\sqrt{n}}{4}\]
and
\[\mathbf{E}_n(T) \le \mathbf{E}(M_n')+1 \le 2\sqrt{n}.\qedhere\]

  \stepcounter{stepuid}

<!-- lemma-6: body_md -->
\label{lem:dropping_below_strong}
There exists a constant $c_{\ref{lem:dropping_below_strong}} > 0$ so that for each $\gamma \in (0,1)$ we have that
\[\mathbf{P}_{m^2}\!\left(\min_{0 \le j \le \gamma m} X_j \ge c_{\ref{lem:dropping_below_strong}} m^2 \right) \ge 1-e^{-c_{\ref{lem:dropping_below_strong}}  \gamma^{-2}} \quad \text{for all}\quad m \in \mathbf{N}. \]
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:dropstrong_1}\emph{Step 1. Splitting into $\gamma^{-2}$ independent groups.} Set $\zeta = c_{\ref{lem:dropping_below_initial}}/2$ and $p = c_{\ref{lem:dropping_below_initial}} - \zeta = c_{\ref{lem:dropping_below_initial}}/2$, so that all of the constants below are universal.  We can break a Galton--Watson process starting at height $m^2$ into a sum of $\gamma^{-2}$ i.i.d.\ Galton--Watson processes starting at height $m^2 \gamma^2$. \Cref{lem:dropping_below_initial} then implies that each of these has probability at least $p$ that its height never drops below $m^2 \gamma^2 \zeta$ before time $3 m \gamma$, and in particular before time $m \gamma$.  If a proportion of at least $p/2$ of these processes achieve this, then the total height will never have dropped below $m^2 p \zeta/2$ (there are then at least $\gamma^{-2}p/2$ of them, each contributing at least $m^2\gamma^2\zeta$).

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:dropstrong_2}\emph{Step 2. A binomial lower bound.} Thus,
\[\mathbf{P}_{m^2}\!\left(\min_{0 \le j \le  \gamma m} X_j \ge m^2 p \zeta/2 \right) \ge \mathbf{P}\!\left(Z \ge \gamma^{-2} p/2 \right),\]
where $Z \sim \mathrm{Bin}(\gamma^{-2}, p)$ (the number of sub-processes which stay above $m^2\gamma^2\zeta$ stochastically dominates $Z$, since they are independent and each succeeds with probability at least $p$).

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:dropstrong_3}\emph{Step 3. Conclusion.} By the Chernoff--Hoeffding inequality applied to the lower tail of $Z$, whose mean is $\gamma^{-2}p$,
\[\mathbf{P}\!\left(Z \ge \gamma^{-2} p/2 \right) \ge 1-\exp\left(-\mathcal {D}\left( p/2 \| p \right) \gamma^{-2} \right)\ge 1-\exp(-c_{\ref{lem:dropping_below_strong}} \gamma^{-2}),\]
where we may take $c_{\ref{lem:dropping_below_strong}} = \min\!\big(\mathcal {D}(p/2\,\|\,p),\, p\zeta/2\big)$, which is a universal constant.  Combining these estimates completes the proof of the lemma.

<!-- lemma-7: body_md -->
\label{lem:upper_bound_distances}
Let $x \ge 1$ be an integer, and let $A_x$ be the event that there is a path of length less than $(2r+1)x$ joining $(n,0)$ to $(n, r^2)$ that is entirely contained in the bubble of initial height $r^2$ started from $(n,0)$ and, moreover, in the strip $\mathcal {L}_{[n,n+r]}$. \[\mathbf{P}(A_x) \ge 1-\exp\left(-\frac{(x-1)^2}{2x}\right).\]

<!-- proof-7: body_md -->
\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:updist_1}\emph{Step 1. Reduction to $n=0$ and to $x \le r^2$.} The event $A_x$ is measurable with respect to $\mathcal {L}_{[n,\infty)}$, which has the same law as $\mathcal {L}$ (with $(n,0)$ playing the role of the origin, since the bottom-most geodesic started from $(0,0)$ passes through $(n,0)$).  Hence $\mathbf{P}(A_x)$ does not depend on $n$, and so we can assume without loss of generality that $n=0$.

We may also assume that $x \le r^2$: otherwise at most $r^2<x$ trees can survive, so that the event considered in the next step has probability one.

\noindent

  \refstepcounter{proofstep}\label{step:updist_2}\emph{Step 2. Few trees survive for $r$ generations.} There are $r^2$ trees in the skeleton between the points $(0,0)$ and $(0,r^2)$, and each one has probability $1/(r+1)^2 \le 1/r^2$ of surviving for at least $r$ generations, i.e., of having at least one individual at generation $r$.  Consequently, the number of trees which survive for at least $r$ generations is stochastically dominated by a $\mathrm{Bin}(r^2,1/r^2)$ random variable $Z$. Thus
\[\mathbf{P}(\text{fewer than $x$ trees survive at least $r$ generations}) \ge 1 - \mathbf{P}(Z \ge x).\]
By the Chernoff--Hoeffding inequality,
\[\mathbf{P}(Z \ge x) \le \exp\left(-\mathcal {D}\left(\frac{x}{r^2} \left\| \frac{1}{r^2} \right.\right)r^2\right) \le \exp\left(-\frac{\left(\frac{x}{r^2}-\frac{1}{r^2}\right)^2}{\frac{2x}{r^2}} \cdot r^2\right)=\exp\left(-\frac{(x-1)^2}{2x}\right).\]
Here $\mathcal {D}$ is as in~\eqref{eq:kl_divergence}, and for the second inequality we used~\eqref{eq:kl_lower_bound}.

  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:updist_3}\emph{Step 3. Construction of the path.} So now consider the event that fewer than $x$ trees survive, which implies the number of surviving trees $m$ satisfies $m \le x-1$, and hence $m+1 \le x$. Let $(0, i_1), \dots, (0, i_m)$ denote the points on the boundary between $(0,0)$ and $(0, r^2)$ such that the tree of the skeleton rooted at the slot $(0,i_k+\frac12)$ immediately above the point survives for at least $r$ generations, listed in increasing order; we set $i_0 = -1$ and $i_{m+1} = r^2$. We can then build a path of length at most $(2r+1)x - 1 < (2r+1)x$ between $(0,0)$ and $(0, r^2)$ as follows. For each of the $m+1$ intervals $0 \le j \le m$, the slots lying strictly between the points $(0,i_j+1)$ and $(0,i_{j+1})$ are precisely those at heights $i_j+\frac32,\dots,i_{j+1}-\frac12$, none of which is one of the surviving slots $(0,i_k+\frac12)$; hence the trees rooted at them are all extinct by generation $r$, and so the bottom-most geodesics starting from $(0, i_j+1)$ and $(0, i_{j+1})$ must coalesce before reaching the line $\{r\} \times \mathbf{Z}$, so the path that goes right along the bottom-most geodesic from $(0, i_j+1)$, then back left along the bottom-most geodesic from $(0, i_{j+1})$ has length at most $2r$. We do this $m+1$ times to get from $(0,0)$ to $(0, r^2)$, going right and returning left along all the bottom-most geodesics, and then to get from $(0, i_{j})$ to $(0, i_{j}+1)$ (of which there are $m$ transitions), we simply take a step up along the corresponding edge of the boundary of $\mathcal {L}$. This produces a path of length at most 
\[2r(m+1) + m = (2r+1)(m+1) - 1 \le (2r+1)x - 1 < (2r+1)x\]
between $(0,0)$ and $(0, r^2)$, and it is entirely contained in the bubble between these points.  Moreover, by construction, it is clearly contained in $\mathcal {L}_{[0,r]}$.

  \stepcounter{stepuid}

Since the law of $\mathcal {L}$ is moreover invariant under vertical translations, we may translate vertically so that $(s,h_{\Gamma}(s))$ becomes the origin; the restriction of $\Gamma$ to $[s,\infty)$ then becomes the bottom-most geodesic started from the origin, which is the line $\mathbf{Z}_{\ge 0}\times\{0\}$, and heights become heights above $\Gamma$. \[\mathbf{P}_{K'r^2}(X_{6r}>0)=1-\left(1-\frac{1}{(6r+1)^2}\right)^{K'r^2}.\]
This expression converges uniformly over $r \in [1,\infty)$ to 1 as $K' \to \infty$ (indeed $K'r^2/(6r+1)^2 \ge K'/49$ for $r \ge 1$, so that the expression is at least $1-e^{-K'/49}$), and so by choosing $K'$ larger if necessary, we also have that for all $r \ge 1$,
\[\mathbf{P}_{K'r^2}(X_{6r}>0) \ge 1-\frac{\kappa}{3}.\]
Note that enlarging $K'$ preserves the previous estimate as well, since it only shrinks the set of $j$ over which the infimum is taken.  Finally, \Cref{lem:tail_bound_maximum}, applied with $y=K'r^2$ and $x=Kr^2$, gives

\[\mathbf{P}_{K'r^2}\!\left(\sup_{j \ge 0} X_j<Kr^2\right) \ge 1-\frac{\kappa}{3}.\]

\item\label{it:sep_distance} We have $\inf_{j \ge K'r^2} d_{\mathrm{gr}}^{\mathcal {L}}((0,0),(0,j)) \ge 13r$.
\item\label{it:sep_survive} The trees in the skeleton started between $(0,0)$ and $(0, K'r^2)$ do not all die before generation $6r$.
\item\label{it:sep_population} The total population size of these trees never reaches $Kr^2$.
\end{enumerate}

By \Cref{step:lowdist_1} the path is contained in $\mathcal {L}_{[0,6r]}$; in particular $0 \le t \le 6r$.  By~\eqref{it:sep_survive} the starting point $(i,0)$ lies strictly below $\gamma$, and by~\eqref{it:sep_population} the endpoint $(t,h)$ lies strictly above it.  Now $\gamma$ crosses the strip $\mathcal {L}_{[0,6r]}$ from its left side to its right side and therefore separates it into two parts, and a path in a planar map cannot cross $\gamma$ without visiting one of its vertices; hence the path must cross $\gamma$. Let $(m, \ell)$ be the crossing point; note that $0 \le m \le 6r$, and that this point has distance exactly $m$ to $(0, K'r^2)$ since it is on its bottom-most geodesic. Then the path
\[(0,0) \to (i,0) \to (m, \ell) \to (0, K'r^2)\]
has length at most $12r$, since $i \le 5r$, the subpath from $(i,0)$ to $(m,\ell)$ has length less than $r$, and the subpath from $(m,\ell)$ to $(0,K'r^2)$ has length $m \le 6r$. This contradicts~\eqref{it:sep_distance} in the definition of $A$, and so no such path exists. This concludes the proof.

\[ \mathbf{P}(G_{\ell}\cap A) \le \mathbf{P}(G_{\ell}^1 \cap G^2 \cap A) = \mathbf{P}(G_{\ell}^1) \mathbf{P}(G^2 \cap A) \le \mathbf{P}(G_{\ell}^1) \mathbf{P}(A),\]

Suppose that we have fixed parameters $u, R, B, \epsilon > 0$ as in the statement of the proposition.


\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:botexp_1}\emph{Step 1. The bottom exploration does not go too low before time $T_{\max}$.} Let us first show that the bottom exploration has a positive chance of never going too low.  For $1 \le k \le k_{\max}$, consider the time interval between $\sqrt[4]{B}2^{k/2} \epsilon^2 n$ and $\sqrt[4]{B}2^{(k+1)/2} \epsilon^2 n$.  Then by the definition of the bottom exploration, we are adding in upward jumps of size $\sqrt{B}\epsilon^4 n^2$ every $\sqrt[4]{B}2^{-u k} \epsilon^2 n$ units of time during this interval.  In particular the time between consecutive slices is at most $(\sqrt{2}+1)\sqrt[4]{B}\epsilon^2 n \le 3\sqrt[4]{B}\epsilon^2 n$ for the initial slice (added at time $0$, the next one at the first multiple of $\sqrt[4]{B}2^{-u}\epsilon^2 n$ which is at least $\sqrt[4]{B}2^{1/2}\epsilon^2 n$), and at most twice the spacing $\sqrt[4]{B}2^{-uk}\epsilon^2 n$ for a slice added in the $k$-th interval, the factor of two accounting for the change of spacing at the endpoints. Let $k_0$ be a large positive integer. We break this into two cases depending on whether $k$ is larger or smaller than $k_0$.

\noindent\emph{Case 1. The early stages $k<k_0$.} Since the height of the bottom exploration at any time is at least the current height of the most recently added slice, it suffices to bound from below the probability that every slice individually stays above $R^3\epsilon^4n^2$ until the next slice is added.  By \Cref{lem:dropping_below_initial}, applied with $m=\sqrt[4]{B}\epsilon^2n$ (so that $m^2=\sqrt{B}\epsilon^4n^2$ is the initial height of a slice, and $3m$ bounds the time until the next slice is added, by the gap bounds noted above) and with $\zeta = R^3/\sqrt{B}$, each slice has probability at least $p_0$ of not dropping below $R^3\epsilon^4 n^2$ before the next slice is added, where $p_0 = c_{\ref{lem:dropping_below_initial}}-R^3/\sqrt{B}$ is some positive constant (provided that we assume $B$ is much larger than $R$; precisely, $B \ge C_0R^6$ gives $R^3/\sqrt{B} \le C_0^{-1/2} \le c_{\ref{lem:dropping_below_initial}}/2$ for $C_0$ large, whence the universal lower bound $p_0 \ge c_{\ref{lem:dropping_below_initial}}/2$). For each $k$, there are
\[2^{k(u+1/2)+1/2}-2^{k(u+1/2)}+1 \le 2^{k(u+1/2)+1}\]
slices added in the time interval between $\sqrt[4]{B}2^{k/2} \epsilon^2 n$ and $\sqrt[4]{B}2^{(k+1)/2} \epsilon^2 n$. So in total, up to time $\sqrt[4]{B}2^{k_0/2} \epsilon^2 n$, there are at most $k_0 2^{k_0(u+1/2)+1}$ slices added, including the initial slice. The slices are all independent, and so with probability at least
\[p_0^{k_0 2^{k_0(u+1/2)+1}}>0,\]
the bottom exploration never drops below $R^3\epsilon^4 n^2$ before time $\sqrt[4]{B}2^{k_0/2} \epsilon^2 n$.


\noindent\emph{Case 2. The later stages $k \ge k_0$.} \Cref{lem:dropping_below_strong} with $m = \sqrt[4]{B}\epsilon^2 n$ and $\gamma = 2^{1-u k}$ implies that the probability that a slice in the LHPT with initial height $m^2 = \sqrt{B}\epsilon^4 n^2$ drops below height $R^3\epsilon^4 n^2$ before the next slice is added (which by the gap bounds above happens within $2 \cdot \sqrt[4]{B}2^{-u k} \epsilon^2 n = \gamma m$ units of time) is at most $\exp(-c_{\ref{lem:dropping_below_strong}} 2^{2u k-2})$, the requirement $B \ge C_0R^6$ guaranteeing that the floor $c_{\ref{lem:dropping_below_strong}}\sqrt{B}\epsilon^4n^2$ provided by \Cref{lem:dropping_below_strong} is at least $R^3\epsilon^4n^2$. Taking a union bound over $k_0 \le k \le k_{\max}$ and $j \in \mathbf{N}$ such that
\[\sqrt[4]{B}2^{k/2} \epsilon^2 n \le \sqrt[4]{B}2^{-uk} j \epsilon^2 n \le \sqrt[4]{B}2^{(k+1)/2} \epsilon^2 n,\]
 we see that the probability that the bottom exploration does not drop below $R^3\epsilon^4 n^2$ at any time between $\sqrt[4]{B}2^{k_0/2}\epsilon^2 n$ and $T_{\max}$ is at least
\[1-\sum_{k=k_0}^{k_{\max}} \exp(-c_{\ref{lem:dropping_below_strong}} 2^{2u k-2})\cdot 2^{k(u+1/2)+1} \ge 1-\sum_{k=k_0}^{\infty} \exp(-c_{\ref{lem:dropping_below_strong}} 2^{2u k-2})\cdot 2^{k(u+1/2)+1}.\]
For a given $u>0$, we can pick $k_0$ sufficiently large (in particular $k_0 \ge 2/u$, so that $\gamma = 2^{1-uk}<1$ throughout Case~2) so that this last expression is positive.

Finally, we combine these two cases using the independence of slices. The probability that the bottom exploration never drops below $R^3\epsilon^4 n^2$ before time $T_{\max}$ is at least
\[p_0^{k_0 2^{k_0(u+1/2)+1}} \left(1-\sum_{k=k_0}^{\infty} \exp(-c_{\ref{lem:dropping_below_strong}} 2^{2u k-2})\cdot 2^{k(u+1/2)+1}\right),\]
which as already discussed is positive if $k_0$ (as a function of $u$) is large enough. Since $p_0 \ge c_{\ref{lem:dropping_below_initial}}/2$ and $c_{\ref{lem:dropping_below_strong}}$ are universal constants, this bound depends on $u$ (through $k_0$) but not on $R$, $B$, $\epsilon$, or $n$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:botexp_2}\emph{Step 2. The bottom exploration does not go too low between times $T_{\max}$ and $R\epsilon n$.} It remains to control the exploration on $[T_{\max}, R\epsilon n]$, where the slices are added at the constant spacing $\sqrt[4]{B}2^{-uk_{\max}}\epsilon^2 n$. Combining with the two cases above, the probability $q$ that the height never drops below $R^3\epsilon^4n^2$ up to time $R\epsilon n$ is bounded below by a positive constant depending only on $u$, for all $\epsilon$ small.  (That $k_0$ depends on $u$ alone is consistent with $B \ge C_0k_0^2 2^{k_0(1+2u)}$, since $B$ is chosen after $k_0$; see \Cref{fn:parameters}.)


\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:botexp_3}\emph{Step 3. The bottom exploration does not go too high.} Let us now show that the bottom exploration is unlikely to get too high.  First of all, the number of slices that we add to the bottom exploration over its course (that is, up to time $R\epsilon n$) is at most a constant (depending on $R$ and $B$) times $\sum_{k=1}^{k_{\max}} 2^{u k + k/2}$, which is of order $\epsilon^{-1- 2u}$: the sum counts the slices added before $T_{\max}$, and those added between $T_{\max}$ and $R\epsilon n$ number at most $(R\epsilon n)/(\sqrt[4]{B}\epsilon^{2+2u}n) = O(R\epsilon^{-1-2u})$, of the same order.   By \Cref{lem:tail_bound_maximum}, applied with $y = \sqrt{B}\epsilon^4n^2$ and $x = \epsilon^{3-3u}n^2$ (its hypothesis $x \ge y$ amounts to $\epsilon^{-1-3u} \ge \sqrt{B}$, valid for $\epsilon$ small), each slice has probability at most $\sqrt{B}\epsilon^{1+3u}$ of reaching a maximal height of at least $\epsilon^{3-3u}n^2$. So by a union bound, the probability that some slice hits $\epsilon^{3-3u}n^2$ is $O(\epsilon^u)$. 

Also, \Cref{lem:tail_bound_maximum} (this time with $x=\epsilon^{3+3u}n^2$, for which $x \ge y$ amounts to $\epsilon^{-1+3u} \ge \sqrt{B}$, again valid for $\epsilon$ small since $u<1/10$) implies that with probability at most $\sqrt{B}\epsilon^{1-3u}$, a slice has a maximal height of at least $\epsilon^{3+3u}n^2$.  Since the slices which make up the bottom exploration are i.i.d., the number of slices which have maximal height at least $\epsilon^{3+3u}n^2$ is stochastically dominated by $Z$, where $Z \sim \mathrm{Bin}(C\epsilon^{-1-2u},\sqrt{B}\epsilon^{1-3u})$ ($C$ is some positive constant). To summarize,
\begin{itemize}
\item There are $O(\epsilon^{-1-2u})$ slices added in total.
\item With probability $1-O(\epsilon^u)$, none of the slices has a maximal height greater than $\epsilon^{3-3u}n^2$.
\item With probability $1-O(\epsilon^u)$, there are fewer than $\epsilon^{-6u}$ slices that have a maximal height greater than $\epsilon^{3+3u}n^2$.
\end{itemize}

Thus with probability $1-O(\epsilon^u)$, the total maximal height of all these slices combined (i.e., if their maxima were all attained at the same time) is at most
\[C\epsilon^{-1-2u} \cdot \epsilon^{3+3u} n^2+\epsilon^{-6u} \cdot \epsilon^{3-3u} n^2=C\epsilon^{2+u}n^2+\epsilon^{3-9u}n^2,\]
which is of order $\epsilon^{2+u} n^2$ if $u>0$ is small enough (namely $u<1/10$, so that $3-9u \ge 2+u$).  Since the height of the bottom exploration is at all times at most the sum of the maximal heights of its slices, it follows that with probability $1-O(\epsilon^u)$ the height never exceeds $\epsilon^2 n^2$ up to time $R\epsilon n$, for all $\epsilon$ small.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:topsurv_1}\emph{Step 1. The bubble has probability at least $cB\epsilon^2$ of staying high up to time $2\epsilon n$.} By \Cref{lem:tail_bound_minimum}, the probability that the bubble of initial height $B\epsilon^4 n^2$ survives until time $2\epsilon n$ and never drops below $\sqrt{B}\epsilon^4 n^2$ before then is
\begin{align*}
\mathbf{P}_{B\epsilon^4 n^2}\!\left(\min_{0 \le j \le 2\epsilon n} X_j \ge \sqrt{B}\epsilon^4 n^2\right) &\ge \left(1-\frac{1}{(2\epsilon n+1)^2}\right)^{\sqrt{B}\epsilon^4 n^2}-\left(1-\frac{1}{(4\epsilon n+1)^2}\right)^{B\epsilon^4 n^2}\\
&\ge \left(1-\frac{\sqrt{B}\epsilon^4 n^2}{(2\epsilon n+1)^2}\right)-\left(1-\frac{B\epsilon^4 n^2}{(4\epsilon n+1)^2}+\frac{B^2 \epsilon^8 n^4}{2(4\epsilon n+1)^4}\right)\\
&\ge cB\epsilon^2
\end{align*}
for some positive constant $c$ (here $X_t$ is the height at time $t$ of the bubble of initial height $B\epsilon^4n^2$ started at $(0,0)$, which by \Cref{def:bubble} evolves as a Galton--Watson process started from $B\epsilon^4n^2$).  Indeed, using $(4\epsilon n+1)^{-2} \ge (4\epsilon n)^{-2}(1-(2\epsilon n)^{-1})$, $(2\epsilon n+1)^{-2} \le (2\epsilon n)^{-2}$ and $(4\epsilon n+1)^{-4} \le (4\epsilon n)^{-4}$, the right-hand side of the second line is at least
\[B\epsilon^2\left(\frac{1}{16}-\frac{1}{4\sqrt{B}}-\frac{B\epsilon^2}{512}-\frac{1}{32\epsilon n}\right),\]
the term in the parentheses is at least a universal constant for $B$ large, $B\epsilon^2 \le 1$ and $\epsilon n$ is large; the last of these holds for all small $\epsilon$ since $n \ge \epsilon^{-4}$.  The constant $c$ may therefore be taken to be universal.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:topsurv_2}\emph{Step 2. Reduction to the survival of a slice of the bottom exploration.} Now, suppose that this event happens, but the top exploration merges with the bottom before time $\epsilon n$. Then it means that some slice that is added in the bottom exploration before time $\epsilon n$ must have survived until time $2 \epsilon n$: at the merging time $t<\epsilon n$ the top exploration is empty, so every tree lying below the bottom-most geodesic started from $(0,B\epsilon^4n^2)$ at time $t$ belongs to a slice of the bottom exploration added at some time $\le t$; since $X_{2\epsilon n} \ge \sqrt{B}\epsilon^4n^2>0$, one of these trees has a descendant at time $2\epsilon n$. If we can show that the probability the latter happens is less than $c'B\epsilon^2$ for a constant $c'<c$, then this will prove the lemma.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:topsurv_3}\emph{Step 3. The probability that a slice of the bottom exploration survives until time $2\epsilon n$.} Each slice started before time $\epsilon n$ has probability at most
\[1-\left(1-\frac{1}{(\epsilon n+1)^2}\right)^{\sqrt{B}\epsilon^4 n^2} \le \sqrt{B}\epsilon^2\]
of surviving until time $2\epsilon n$. Consider the time interval between $\sqrt[4]{B}2^{k/2} \epsilon^2 n$ and $\sqrt[4]{B}2^{(k+1)/2} \epsilon^2 n$; since
\[T_{\text{max}}=\sqrt[4]{B}2^{(k_{\text{max}}+1)/2}\epsilon^2 n>B^{1/8}\epsilon n> \epsilon n,\]
every slice added before time $\epsilon n$ is added in one of those intervals for some $1 \le k \le k_{\text{max}}$. As already noted in the proof of \Cref{prop:bottom_exploration_good}, there are at most $2^{k(u+1/2)+1}$ slices added in this interval. Let $k_0$ be a large positive integer. We once again break things into two cases, according to whether $k$ is larger or smaller than $k_0$.

\noindent\emph{Case 1. The early stages $k<k_0$.} By a union bound, the probability that one of the slices between $\sqrt[4]{B}2^{k/2} \epsilon^2 n$ and $\sqrt[4]{B}2^{(k+1)/2} \epsilon^2 n$ survives until $2\epsilon n$ is at most $\sqrt{B}2^{k(u+1/2)+1}\epsilon^2$. Then taking a union bound over $k<k_0$ gives that with probability at most $\sqrt{B}k_0 2^{k_0(u+1/2)+1}\epsilon^2$, one of the slices added before time $\sqrt[4]{B}2^{k_0/2}\epsilon^2 n$ (including the initial slice, whose survival probability satisfies the same bound) survives until time $2\epsilon n$.

\noindent\emph{Case 2. The later stages $k \ge k_0$.} The event that the bubble survives until time $\sqrt[4]{B}2^{k/2}\epsilon^2 n$ is measurable with respect to the skeleton up to that time, while the survival of a slice added at a later time concerns the fresh subtrees rooted at that later time; by the branching property, the two events are therefore independent, and their probabilities may be multiplied. The first of these has probability
\[1-\left(1-\frac{1}{(\sqrt[4]{B}2^{k/2}\epsilon^2 n+1)^2}\right)^{B\epsilon^4 n^2} \le \frac{\sqrt{B}}{2^k}.\] 
As already explained above, there are at most $2^{k(u+1/2)+1}$ slices in this interval, and each one has probability no larger than $\sqrt{B}\epsilon^2$ of surviving until $2\epsilon n$. So by a union bound, the probability that the bubble survives until time $\sqrt[4]{B}2^{k/2}\epsilon^2 n$ and one of the slices in the interval from $\sqrt[4]{B}2^{k/2} \epsilon^2 n$ to $\sqrt[4]{B}2^{(k+1)/2} \epsilon^2 n$ survives until $2\epsilon n$ is at most
\[\frac{\sqrt{B}}{2^k} \cdot 2^{k(u+1/2)+1}\cdot \sqrt{B}\epsilon^2=B\epsilon^2 \cdot 2^{k(u-1/2)+1}.\]
Taking a union bound over $k \ge k_0$ yields a bound of
\[B\epsilon^2 \sum_{k=k_0}^{k_{\max}} 2^{k(u-1/2)+1} \le CB\epsilon^2 2^{k_0(u-1/2)+1}\]
for a universal constant $C$, since $u<1/2$ makes the sum a convergent geometric series with ratio $2^{u-1/2}$ bounded away from~$1$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:topsurv_4}\emph{Step 4. Conclusion.} Combining these cases, we therefore have that the probability that the bubble never drops below $\sqrt{B}\epsilon^4 n^2$ before time $2\epsilon n$ and the top merges with the bottom before time $\epsilon n$ is at most
\[\left(\sqrt{B}k_0 2^{k_0(u+1/2)+1}+CB 2^{k_0(u-1/2)+1}\right)\epsilon^2.\]
By picking $k_0$ very large, and then making sure that $B$ is much larger than $2^{k_0}$, this is less than $c'B\epsilon^2$ for a constant $c'$ that can be made as small as we want. Indeed, the second term is at most $c'B\epsilon^2/2$ once $2^{k_0(1/2-u)}$ is large compared with $1/c'$, which fixes $k_0$; the first is then at most $c'B\epsilon^2/2$ once $\sqrt{B}$ is large compared with $k_02^{k_0(u+1/2)}/c'$, that is, once $B \ge C_0k_0^2 2^{k_0(1+2u)}$, the constraint recorded in \Cref{fn:parameters}.  Taking $c'=c/2$, the probability that the top exploration merges after time $\epsilon n$ is therefore at least $(c-c')B\epsilon^2=cB\epsilon^2/2$, which proves the lemma with $c_{\ref{lem:top_exploration_survives_long}} = c/2$.

We must show that there is a constant $c_{\ref{prop:good_bubble_probability}}>0$ such that with probability at least $c_{\ref{prop:good_bubble_probability}}B \epsilon^2$, all the conditions in \Cref{def:bottom_good,def:top_good,def:bubble_good} hold.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:goodbubble_1}\emph{Step 1. Discarding condition~\eqref{it:max_height_top} of \Cref{def:top_good} and the upper bound on the merging time in~\eqref{it:top_survival} of \Cref{def:top_good}.} We begin with a few simplifications. First, by \Cref{lem:tail_bound_maximum}, together with the fact that the height process of the top exploration is stochastically dominated by a Galton--Watson process started from $B\epsilon^4n^2$, the probability that the height process of the top exploration ever exceeds $R\epsilon^2n^2/4$, i.e.\ that~\eqref{it:max_height_top} fails, is at most (the hypothesis $x \ge y$ of that lemma holding because $R/4 \ge B\epsilon^2$ for $\epsilon$ small)
\[\mathbf{P}_{B\epsilon^4 n^2}\!\left(\max_{j \ge 0} X_j \ge \frac{R\epsilon^2 n^2}{4}\right) \le \frac{4B \epsilon^2}{R}.\]
This is negligible compared to $c_{\ref{prop:good_bubble_probability}}B \epsilon^2$ if $R$ is large. Next, the height process of the top exploration is stochastically dominated by a Galton--Watson process with offspring distribution $\theta$, so the probability that it merges with the bottom after time $R\epsilon n$ is at most
\[\mathbf{P}_{B\epsilon^4 n^2}(X_{R\epsilon n}>0)=1-\left(1-\frac{1}{(R\epsilon n+1)^2}\right)^{B\epsilon^4 n^2}\le \frac{B\epsilon^2}{R^2}.\]
Once again, this is negligible compared to $c_{\ref{prop:good_bubble_probability}}B \epsilon^2$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:goodbubble_2}\emph{Step 2. Discarding condition~\eqref{it:initial_distance}.} We next take care of condition~\eqref{it:initial_distance} of \Cref{def:top_good}, namely that the distance between $(0,B\epsilon^4n^2)$ and $(0,\sqrt{B}\epsilon^4n^2)$ \emph{in the top exploration} is at most $\epsilon^{2-u}n$. The first slice to be removed from the top exploration and added to the bottom exploration is removed at the time $t_1$, the smallest multiple of $\sqrt[4]{B}2^{-u}\epsilon^2n$ which is at least $\sqrt[4]{B}2^{1/2}\epsilon^2 n$, so that $t_1 \ge \sqrt{2}\sqrt[4]{B}\epsilon^2 n$. Up to time $t_1$, but not after, the top exploration coincides with the region of the LHPT between the bottom-most geodesics started from $(0,\sqrt{B}\epsilon^4n^2)$ and $(0,B\epsilon^4n^2)$.  The path produced by \Cref{lem:upper_bound_distances} at scale $r$ has horizontal extent at most $r$, and $r=\sqrt{B-\sqrt{B}}\,\epsilon^2 n$ would exceed $t_1$ for large $B$; we therefore apply it at a smaller scale, splitting $[\sqrt{B}\epsilon^4n^2, B\epsilon^4n^2]$ into $M$ equal parts.  Set
\[M=\lceil B \rceil, \qquad \sigma=\sqrt{\frac{B-\sqrt{B}}{M}}\,\epsilon^2 n, \qquad h_i=\sqrt{B}\epsilon^4 n^2+(i-1)\sigma^2 \quad (1 \le i \le M+1),\]
so that $h_1=\sqrt{B}\epsilon^4n^2$, $h_{M+1}=B\epsilon^4 n^2$ and
\[\sigma \le \sqrt{\frac{B-\sqrt{B}}{B}}\,\epsilon^2 n \le \epsilon^2 n<\sqrt{2}\sqrt[4]{B}\epsilon^2 n \le t_1.\]
Apply \Cref{lem:upper_bound_distances} to each of the $M$ pairs $(0,h_i)$, $(0,h_{i+1})$, with $\sigma$ in place of $r$ and $\lceil \epsilon^{-u/2}\rceil$ in place of $x$, and use invariance of the law of the LHPT under vertical translations.  By a union bound, with probability at least $1-M\exp(-c\epsilon^{-u/2}) \ge 1-\exp(-c'\epsilon^{-u/2})$ we obtain, simultaneously for all $i$, a path from $(0,h_i)$ to $(0,h_{i+1})$ of length less than $(2\sigma+1)x$ and horizontal extent at most $\sigma$, contained in the bubble between the bottom-most geodesics started from these two points.  Each such bubble lies between the bottom-most geodesics started from $(0,\sqrt{B}\epsilon^4n^2)$ and $(0,B\epsilon^4n^2)$, and each path lies to the left of time $\sigma<t_1$, so each is contained in the top exploration.  Concatenating them gives a path from $(0,\sqrt{B}\epsilon^4n^2)$ to $(0,B\epsilon^4n^2)$ inside the top exploration, of length less than
\[M(2\sigma+1)x \le 2\sqrt{M(B-\sqrt{B})}\,\epsilon^2 n\,x+Mx \le CB\epsilon^{2-u/2}n+CB\epsilon^{-u/2} \le \epsilon^{2-u}n\]
for all $\epsilon$ sufficiently small, where $C$ is universal; here $\sqrt{M(B-\sqrt{B})}$ and $M$ are both at most $CB$ and $x \le 2\epsilon^{-u/2}$, and the last inequality holds because $\epsilon^{u/2}$ is small compared with $1/B$ and $n \ge \epsilon^{-4}$ (see \Cref{fn:parameters}).  As $\exp(-c'\epsilon^{-u/2})$ is of much smaller order than $B\epsilon^2$, we may ignore~\eqref{it:initial_distance} as well.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:goodbubble_3}\emph{Step 3. Condition~\eqref{it:merge_distance}, and reduction to four events.} Finally, let $E_1$ be the event that the bottom exploration is good \emph{and} the distance condition in~\eqref{it:merge_distance} of \Cref{def:bubble_good} holds.  The merging time $\ell$ is measurable with respect to the top exploration, which is independent of the bottom exploration, and by \Cref{lem:bottom_exploration_typical_height} (applied conditionally on the top exploration, with a bound uniform over $\ell \ge \epsilon n$) that distance condition fails with conditional probability $O(\epsilon^u)$.  Hence, conditionally on any top-exploration event on which $\ell \ge \epsilon n$, the event $E_1$ has probability at least $p_{\ref{prop:bottom_exploration_good}} - O(\epsilon^u) \ge p_{\ref{prop:bottom_exploration_good}}/2$ for $\epsilon$ small.  (Both~\eqref{it:merge_distance} and the goodness of the bottom exploration concern the bottom exploration, so their probabilities cannot be multiplied; this conditional bound is how~\eqref{it:merge_distance} is incorporated below.)

In summary, it thus remains to show that the probability of the intersection of the following four events is at least $cB\epsilon^2$ for some constant $c>0$:
\begin{itemize}
\item $E_1$: The bottom exploration is good \emph{and} the distance condition in~\eqref{it:merge_distance} holds;
\item $E_2$: The top exploration merges with the bottom exploration after time $\epsilon n$;
\item $E_3$: The height process of the top exploration at time $\epsilon n/2$ is at least $\epsilon^2n^2/R$ (i.e., condition~\eqref{it:middle_height} holds);
\item $E_4$: Condition~\eqref{it:middle_graph_distance} holds, i.e.\ the distance between $c_1$ and the bottom geodesic of the bubble is at least $\sqrt{D}\epsilon n$.
\end{itemize}

\noindent

  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:goodbubble_4}\emph{Step 4. Splitting the bubble at time $\epsilon n/4$.} Recall that $(Y_j)_{j \ge 0}$ denotes the height process for the top exploration, described before \Cref{def:top_good}. We of course have that $E_2$ is the event that $Y_{\epsilon n}>0$.

We have (note that $E_2 \subseteq \{Y_{\epsilon n/4}>0\}$)
\[\mathbf{P}(E_1 \cap E_2 \cap E_3 \cap E_4)=\mathbf{P}(Y_{\epsilon n/4}>0) \mathbf{P}(E_1 \cap E_2 \cap E_3 \cap E_4 \,|\, Y_{\epsilon n/4}>0).\]
By \Cref{lem:top_exploration_survives_long},
\[\mathbf{P}(Y_{\epsilon n/4}>0) \ge \mathbf{P}(Y_{\epsilon n}>0) \ge c_{\ref{lem:top_exploration_survives_long}} B \epsilon^2.\]

\begin{align*}
\mathbf{P}(E_1 \cap E_2 \cap E_3 \cap E_4 \,|\, Y_{\epsilon n/4}>0)&\ge \mathbf{P}(E_1 \cap E_2 \cap E_3 \,|\, Y_{\epsilon n/4}>0)-\mathbf{P}(E_3 \cap E_4^c \,|\, Y_{\epsilon n/4}>0)\\
&=\Pi_1-\Pi_2.
\end{align*}

\noindent
  \stepcounter{stepuid}

  \refstepcounter{proofstep}\label{step:goodbubble_5}\emph{Step 5. Upper bound for $\Pi_2$.} Let $\kappa>0$, let $K=K(\kappa)$ be the integer provided by \Cref{lem:lower_bound_distances}, and set $r=\sqrt{D}\epsilon n$; see \Cref{fig:high distance point}.  We assume that
\[D \le \min\left(\frac{1}{144},\ \frac{1}{KR}\right),\]
which is possible since in the order of \Cref{fn:parameters} $R$ is fixed before $D$, while $\kappa$ (hence $K$) is chosen in \Cref{step:goodbubble_9} in terms of $u$ alone.  With these choices $3r \le \epsilon n/4$, so that all of the paths considered below lie at times greater than $\epsilon n/4$, and $\epsilon n/2 \ge 3r$, as \Cref{lem:lower_bound_distances} requires.

On $E_3$ the vertex $c_1$ has height at least $Y_{\epsilon n/2} \ge \epsilon^2n^2/R \ge Kr^2$, since its height is the sum of the heights of the two explorations at time $\epsilon n/2$ and since $Kr^2=KD\epsilon^2n^2 \le \epsilon^2n^2/R$.  If $E_4$ fails, then some path of length less than $r$ started at $c_1$ visits a vertex $(t,h)$ with $h \le 0$.

A vertex of height $h<0$ lies strictly below the bottom-most geodesic $\mathbf{Z}_{\ge0}\times\{0\}$ started from the origin, whereas $c_1$ lies strictly above it, and a path cannot cross that geodesic without visiting one of its vertices; so the path visits a vertex $(t,0)$, necessarily with $|t-\epsilon n/2|<r$.  On $E_3$ this possibility is therefore contained in the event $N_{\downarrow}$ that some path of length less than $r$ started at a vertex $(\epsilon n/2,j)$ with $j \ge Kr^2$ visits a vertex of $\mathbf{Z}_{\ge0}\times\{0\}$.  Reversing the paths and applying \Cref{lem:lower_bound_distances} with $h_0=0$, this $r$ and $t_0=\epsilon n/2$ gives $\mathbf{P}(N_{\downarrow}) \le \kappa$.  Moreover $N_{\downarrow}$ is measurable with respect to $\mathcal {L}_{[\epsilon n/4,\infty)}$: the paths involved lie at times greater than $\epsilon n/4$, and the height labeling to the right of that time is determined by $\mathcal {L}_{[\epsilon n/4,\infty)}$, the bottom-most geodesic from the origin passing through $(\epsilon n/4,0)$.  Since $\{Y_{\epsilon n/4}>0\}$ is measurable with respect to $\mathcal {L}_{[0,\epsilon n/4]}$, which is independent of $\mathcal {L}_{[\epsilon n/4,\infty)}$, the conditioning may be removed, and this possibility contributes at most $\kappa$ to $\Pi_2$. We obtain that $\Pi_2 \le \kappa$, which is as small as we like once $\kappa$, and then $D$, is small.

\[\Pi_1=\mathbf{P}\!\big(E_1 \,|\, E_2 \cap E_3 \cap \{Y_{\epsilon n/4}>0\}\big)\,\mathbf{P}(E_2 \cap E_3 \,|\, Y_{\epsilon n/4}>0) \ge \frac{p_{\ref{prop:bottom_exploration_good}}}{2} \cdot \mathbf{P}(E_2 \cap E_3 \,|\, Y_{\epsilon n/4}>0).\] So now we can focus on lower bounding $\mathbf{P}(E_2 \cap E_3 \,|\, Y_{\epsilon n/4}>0)$. \[\mathbf{P}(E_2 \cap E_3 \,|\, Y_{\epsilon n/4}>0)=\mathbf{P}(E_2 \,|\, Y_{\epsilon n/4}>0)-\mathbf{P}(E_2 \cap E_3^c \,|\, Y_{\epsilon n/4}>0)=\Pi_3-\Pi_4.\]

\noindent
  \stepcounter{stepuid}

\[\mathbf{P}(Y_{\epsilon n/4}>0) \le \mathbf{P}_{B\epsilon^4 n^2}(X_{\epsilon n/4}>0)=1-\left(1-\frac{1}{(\epsilon n/4+1)^2}\right)^{B\epsilon^4 n^2} \le 16B\epsilon^2.\]

\[\Pi_3=\frac{\mathbf{P}(Y_{\epsilon n}>0)}{\mathbf{P}(Y_{\epsilon n/4}>0)} \ge \frac{c_{\ref{lem:top_exploration_survives_long}}}{16}.\]

\begin{align*}
\Pi_4&=\mathbf{P}(E_2 \cap E_3^c \,|\, Y_{\epsilon n/4}>0)=\sum_{1 \le j < \epsilon^2 n^2/R} \mathbf{P}(E_2 \cap \{Y_{\epsilon n/2}=j\} \,|\, Y_{\epsilon n/4}>0)\\

\end{align*}
Once again, using the fact that $Y$ is stochastically dominated by a Galton--Watson process and monotonicity in the initial height,
\[\mathbf{P}(E_2 \,|\, Y_{\epsilon n/2}=j) \le \mathbf{P}_j(X_{\epsilon n/2}>0) \le \mathbf{P}_{\epsilon^2 n^2/R}(X_{\epsilon n/2}>0)\]
for all $1 \le j < \epsilon^2 n^2/R$. Therefore
\begin{align*}
\Pi_4 &\le \mathbf{P}_{\epsilon^2 n^2/R}(X_{\epsilon n/2}>0)\mathbf{P}\!\left(1 \le Y_{\epsilon n/2}<\frac{\epsilon^2 n^2}{R} \,|\, Y_{\epsilon n/4}>0\right)\\

&=1-\left(1-\frac{1}{(\epsilon n/2+1)^2}\right)^{\epsilon^2 n^2/R} \le \frac{4}{R}.

\[\mathbf{P}(E_1 \cap E_2 \cap E_3 \cap E_4 \,|\, Y_{\epsilon n/4}>0) \ge \Pi_1 - \Pi_2 \ge \frac{p_0'\,c_0'}{32}-\frac{p_0'\,c_0'}{64}=\frac{p_0'\,c_0'}{64}.\]
Combining with the two displays of \Cref{step:goodbubble_4},
\[\mathbf{P}(E_1 \cap E_2 \cap E_3 \cap E_4) \ge c_0'B\epsilon^2 \cdot \frac{p_0'\,c_0'}{64} =: c_1 B \epsilon^2.\]
The event $E_1 \cap E_2 \cap E_3 \cap E_4$ encodes conditions~\eqref{it:bottom_good}, \eqref{it:merge_distance} and~\eqref{it:middle_graph_distance} of \Cref{def:bubble_good}, together with~\eqref{it:middle_height} and the first half of~\eqref{it:top_survival} of \Cref{def:top_good}.  It remains to take into account the conditions set aside in \Cref{step:goodbubble_1,step:goodbubble_2}, by subtracting the failure probabilities $4B\epsilon^2/R$ for~\eqref{it:max_height_top}, $B\epsilon^2/R^2$ for the second half of~\eqref{it:top_survival}, and $\exp(-c'\epsilon^{-u/2})$ for~\eqref{it:initial_distance}:

provided $R$ was also taken large enough that $4/R+1/R^2 \le c_1/4$, and $\epsilon$ is small enough.  The two requirements on $R$ are compatible, since $c_1 = p_0'(c_0')^2/64$ depends only on $u$.

\label{prop:no_good_bubbles}
There is a constant $c_{\ref{prop:no_good_bubbles}} > 0$ which depends only on $u$ (and not on $B, R$ or $D$) such that for all $\epsilon$ sufficiently small,

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:nogoodbub_1}\emph{Step 1. A renewal sequence of bubbles.} We begin by defining a sequence of random times $(S_k)_{k \ge 0}$, which will all be of the form $j\epsilon^2n$ for some $j \ge \epsilon^{-2}/3$, as follows. First we let $S_0=n/3$. Now assume that $S_{k-1}$ has been defined for some $k \ge 1$. We then define $S_k$ to be the smallest time of the form $j\epsilon^2n$ which is larger than or equal to the time when the bubble started at $S_{k-1}$ dies, i.e.\ larger than or equal to $S_{k-1}$ plus the lifetime of that bubble.  Note that the bubbles defined at the times $S_k$ are independent of each other (by the strong Markov property: $S_k$ is determined by the LHPT to the left of $S_k$, and by the branching property, the bubble started at $S_k$ is a function of fresh randomness).  In particular, the increments $(S_k-S_{k-1})_{k \ge 1}$ are i.i.d., and have the same law as
\[\left \lceil \frac{T}{\epsilon^2n} \right \rceil \epsilon^2n\]
where $T$ denotes the lifetime of a bubble of initial height $B\epsilon^4n^2$: by \Cref{def:bubble} its height is a Galton--Watson process started from $B\epsilon^4n^2$, so that $T$ is the extinction time of $X$ under $\mathbf{P}_{B\epsilon^4n^2}$ and its law is given by~\eqref{eq:bubble_lifetime}. We write $G$ for the event that this bubble is good, so that $p:=\mathbf{P}(G)$ is the probability appearing in \Cref{prop:good_bubble_probability}.

Next, let $N$ be the smallest integer $k$ such that the bubble started at $S_k$ is good, and let $\tau=S_{N+1}$. Thus $\tau$ is the first time of the form $j\epsilon^2n$ by which a good bubble started at one of the $S_k$ has been observed in its entirety.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:nogoodbub_2}\emph{Step 2. Reduction to an estimate for $\mathbf{E}(\tau-n/3)$.} If there are no good bubbles on $[n/3, 2n/3]$, then we must have $\tau \ge 2n/3$ (indeed, none of the bubbles started at the times $S_k \in [n/3,2n/3]$ is then good, so $S_N > 2n/3$). So by Markov's inequality,
\begin{equation}\label{eq:markov_no_good_bubbles}
\mathbf{P}(E^c) \le \mathbf{P}\!\left(\tau \ge \frac{2n}{3}\right) \le \frac{3\mathbf{E}(\tau-n/3)}{n}.
\end{equation}

The law of $\tau$ is the same as that of
\[\left \lceil \frac{W}{\epsilon^2n} \right \rceil \epsilon^2n+\sum_{k=1}^{N} \left \lceil \frac{V_k}{\epsilon^2n} \right \rceil \epsilon^2n+\frac{n}{3},\]
where $N$ is a geometric random variable on $\{0,1, \dots\}$ with success probability $p=\mathbf{P}(G)$, the $V_k$'s are i.i.d.\ with the law of $T$ conditioned on $G^c$, and $W$ has the law of $T$ conditioned on $G$, with all these variables independent.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:nogoodbub_3}\emph{Step 3. Estimates for $\mathbf{E}(N)$, $\mathbf{E}(V_1)$ and $\mathbf{E}(W)$.} By \Cref{prop:good_bubble_probability},
\[\mathbf{E}(N)=\frac{1}{p}-1\le \frac{1}{c_{\ref{prop:good_bubble_probability}} B\epsilon^2}.\]
Moreover,
\begin{align*}
\mathbf{E}(V_1)
&=\frac{\mathbf{E}(T {\bf 1}_{G^c})}{1-p}
 \le \frac{\mathbf{E}_{B\epsilon^4n^2}(T)}{1-B\epsilon^2} \quad\text{(by \Cref{prop:good_bubble_probability})}\\
&\le c\sqrt{B}\epsilon^2n \quad\text{(by \Cref{lem:expected_length_of_bubble})}
\end{align*}
for all $\epsilon > 0$ sufficiently small and a constant $c > 0$. Finally,
\begin{align*}
\mathbf{E}_{B\epsilon^4n^2}(T \cdot {\bf 1}_{\{T \ge \epsilon n\}})&\le \sum_{k<\epsilon n} \mathbf{P}_{B\epsilon^4n^2}(T \ge \epsilon n)+\sum_{k \ge \epsilon n} \mathbf{P}_{B\epsilon^4n^2}(T \ge k)\\
&\le B\epsilon^3n+\sum_{k \ge \epsilon n} \frac{B\epsilon^4n^2}{k^2} \quad\text{(by~\eqref{eq:bubble_lifetime}, \eqref{eq:bernoulli})}\\
&\le 3B\epsilon^3n,
\end{align*}
so that by \Cref{prop:good_bubble_probability} again,
\[\mathbf{E}(W)=\frac{\mathbf{E}(T {\bf 1}_{G})}{p} \le \frac{\mathbf{E}_{B\epsilon^4n^2}(T \cdot {\bf 1}_{\{T \ge \epsilon n\}})}{p} \le c \epsilon n,\]
possibly increasing the value of $c > 0$; here we used that $G \subseteq \{T \ge \epsilon n\}$: on $G$ the two explorations merge at a time $\ell \ge \epsilon n$ by~\eqref{it:top_survival} of \Cref{def:top_good}, and so the top exploration has positive height and the bubble has not died by time $\epsilon n$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:nogoodbub_4}\emph{Step 4. Conclusion.} Putting all this together, we thus find
\begin{equation}\label{eq:prob_no_good_bubbles}
\begin{aligned}
\mathbf{E}(\tau-n/3)&\le \mathbf{E}\!\left(W+\sum_{k=1}^{N} V_k +(N+1)\epsilon^2n\right)=\mathbf{E}(W)+\mathbf{E}(N)\mathbf{E}(V_1)+\mathbf{E}(N+1)\epsilon^2n\\
&\le c n\left(\epsilon+\frac{1}{\sqrt{B}}+\frac{1}{B}+\epsilon^2\right),
\end{aligned}
\end{equation}
possibly increasing $c > 0$ further at the end.

  \stepcounter{stepuid}

  \refstepcounter{proofstep}\label{step:condfail_1}\emph{Step 1. Bayes' rule and a lower bound for $\mathbf{P}(F)$.} By Bayes' rule,
\[\mathbf{P}(A \,|\, F) = \frac{\mathbf{P}(F \,|\, A)}{\mathbf{P}(F)} \mathbf{P}(A).\]
As explained above, we have that
\[\mathbf{P}(F) \ge \frac{c_1}{\sqrt{B}}\]

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:condfail_2}\emph{Step 2. An event $E_J^c$ which contains $F$ up to the extra Bernoulli, and is independent of $A$.} Now given $A$, the extra information that we get is at most that there are no good bubbles on the particular sub-segment of the starting segment that we are working with. We formalize this as follows.  Write $J=[a,b]$ for the time window of the sub-segment to which $A$ refers, and set $J^+=[a-2R\epsilon n, b)$.  Whether the bubble started at a time $t$ is good is determined by the restriction of the LHPT to the time window $[t, t+2R\epsilon n]$.  All of the conditions in \Cref{def:bubble_good} concern the two explorations up to time $R\epsilon n$ after $t$ (the merging time is at most $R\epsilon n$ after $t$ when the bubble is good, by~\eqref{it:top_survival} of \Cref{def:top_good}, and the height bounds of~\eqref{it:bottom_height_bound} of \Cref{def:bottom_good} exactly up to that time) together with paths of length at most $\epsilon^{2-4u}n$ or $\sqrt{D}\epsilon n$ started within this range, which move at most that much in the time direction and so stay inside the stated window.  In particular this window is disjoint from $J$ when $t \notin J^+$.  Let $E_J$ be the event that some good bubble is started at a time $t \in [n/3,2n/3]\setminus J^+$, the analog of the event $E$ of \Cref{prop:no_good_bubbles} for this smaller collection of starting times.  Then $F \subseteq E_J^c \cup \{\text{the extra Bernoulli equals } 1\}$, so that
\[\mathbf{P}(F \,|\, A) \le \mathbf{P}(E_J^c \,|\, A)+\frac{c_{\ref{prop:no_good_bubbles}}}{\sqrt{B}}.\]
By the preceding paragraph ${\bf 1}_{E_J^c}$ is the product of a function of $\mathcal {L}_{[0,a]}$ and a function of $\mathcal {L}_{[b,\infty)}$, whereas ${\bf 1}_A$ is a function of $\mathcal {L}_{[a,b]}$.  By the branching property of the skeleton, $\mathcal {L}_{[a,\infty)}$ is independent of $\mathcal {L}_{[0,a]}$ and $\mathcal {L}_{[b,\infty)}$ is independent of $\mathcal {L}_{[0,b]}$; conditioning successively, these two statements give the \emph{joint} independence of $\mathcal {L}_{[0,a]}$, $\mathcal {L}_{[a,b]}$ and $\mathcal {L}_{[b,\infty)}$.  Consequently $E_J^c$ is independent of $A$, and $\mathbf{P}(E_J^c \,|\, A) = \mathbf{P}(E_J^c)$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:condfail_3}\emph{Step 3. An upper bound for $\mathbf{P}(E_J^c)$.} To bound $\mathbf{P}(E_J^c)$ we run the chain $(S_k)$ from the proof of \Cref{prop:no_good_bubbles}, modified so as to skip over $J^+$: whenever the next starting time would fall in $J^+$, the next bubble is instead started at the first time of the form $j\epsilon^2n$ which is at least $b$.  Each $S_k$ is still a stopping time for the exploration of the LHPT from left to right, so by the strong Markov property the bubbles of the modified chain are still independent, with the increments distributed as before (the single skip contributes no increment).  On $E_J^c$ none of these bubbles started in $[n/3,2n/3]$ is good, so the first good one is started after time $2n/3$; as the only part of $[n/3,2n/3]$ which the chain does not traverse by means of bubble lifetimes is the skipped interval, of length at most $|J^+|+\epsilon^2n=n/6+2R\epsilon n+\epsilon^2n$, the accumulated increments must sum to at least
\[\frac{n}{3}-\frac{n}{6}-2R\epsilon n-\epsilon^2n \ge \frac{n}{7}\]
for $\epsilon$ sufficiently small.  Hence, in place of~\eqref{eq:markov_no_good_bubbles}, Markov's inequality gives
\[
\mathbf{P}(E_J^c)\le \mathbf{P}\!\left(\tau - \frac{n}{3} \ge \frac{n}{7}\right) \le \frac{7\mathbf{E}(\tau-n/3)}{n},
\]
with $\tau$ having the exact same distribution. Therefore the rest of the calculations in \Cref{prop:no_good_bubbles} are exactly the same, and we get the same bound up to a constant factor.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}

\[\mathbf{P}(A \,|\, F ) \le \frac{2 c_2\left(\epsilon+ 1/\sqrt{B}\right)+ c_1/\sqrt{B}}{c_1/\sqrt{B}} \mathbf{P}(A)=\left(\frac{2c_2}{c_1}\left(\epsilon \sqrt{B}+1\right)+1\right)\mathbf{P}(A)\le c_3\mathbf{P}(A)\]
if $\epsilon$ is sufficiently small.

<!-- definition-5: body_md -->
\label{def:L-eps_admissible}
Let $a \in \mathbf{Z}_{\ge 0} \times \mathbf{Z}$, and $b$ be a point on the bottom-most geodesic started from $a$, and let $L \ge 1$ and $\epsilon>0$. We will say that a function $f: P_{a,b} \to \ell^2$ is $(L,\epsilon)$-\emph{admissible} if the following conditions hold. See \Cref{fig:exploration around good bubble}.
\begin{enumerate}[(i)]
\item\label{it:adm_lipschitz} For every $x,y \in P_{a,b}$ which are connected to each other by a path entirely contained in $P_{a,b}$,
\[\|f(x)-f(y)\|_2 \le L\, d_{P_{a,b}}(x,y),\]
where $d_{P_{a,b}}(x,y)$ denotes the minimal length of such a path; and for every $x,y \in P_{a,b}$ such that $y$ lies on the bottom-most geodesic started from $x$ and the geodesic segment from $x$ to $y$ is entirely contained in $P_{a,b}$,
\[d(x,y) \le \|f(x)-f(y)\|_2.\]
\item\label{it:adm_bubble_start} For every bubble which is good relative to a starting segment of length $m$ (that is, a bubble arising from the bottom and top explorations started at a point of a geodesic segment of length $m$, which is good in the sense of \Cref{def:bubble_good} with $m$ in place of $n$) and which is entirely contained in $P_{a,b}$, if $a_1$ is the starting point of the bubble and $a_1'$ is the point right above the $B\epsilon^4m^2$ trees at the start of the bubble (that is, the starting point of the top geodesic of the bubble, which lies at height $B\epsilon^4m^2$ above $a_1$), then
\[\|f(a_1)-f(a_1')\|_2 \le 2L\epsilon^{2-u}m.\]
\item\label{it:adm_bubble_merge} For every bubble which is good relative to a starting segment of length $m$ and which is entirely contained in $P_{a,b}$, if $b_1'$ is the merging vertex, and $b_1$ is the point on the starting segment at the same time, then
\[\|f(b_1)-f(b_1')\|_2 \le L\epsilon^{2-4u}m.\]
\item\label{it:adm_bubble_middle} For every bubble which is good relative to a starting segment of length $m$ and which is entirely contained in $P_{a,b}$, if $c_1$ is the point from part~\eqref{it:middle_graph_distance} of \Cref{def:bubble_good}, and $c_2$ is the point on the starting segment at the same time, then
\[\|f(c_1)-f(c_2)\|_2 \ge \sqrt{D}\epsilon m.\]
\end{enumerate}

<!-- lemma-13: body_md -->
\label{lem:laakso_diamond}
Suppose that we have a good bubble on a starting segment of length $n$, and that the bubble is entirely contained in the parallelogram $P_{a,b}$ defined below.  Let $a$, $a_1$, $a_1'$, $b_1$, $b_1'$, $c_1$, $c_2$, $b$ be the following points:
\begin{itemize}
\item $a=(0,0)$ and $b=(n,0)$.
\item $a_1$ is the starting point of the good bubble on the starting segment.
\item $b_1'$ is the merging vertex, and $b_1$ is the point on the starting segment at the same time.
\item $a_1'$ is the point right above the $B\epsilon^4n^2$ trees at the start of the bubble, i.e., the starting point of its top geodesic.
\item $c_1$ is the point from part~\eqref{it:middle_graph_distance} of \Cref{def:bubble_good}, and $c_2$ is the point on the starting segment with the same time coordinate.
\end{itemize}
See \Cref{fig:exploration around good bubble}. Let
\begin{align*}
&\chi^{(1)}=\chi_{L,\epsilon}(a,a_1) \qquad \chi^{(4)}=\chi_{L,\epsilon}(c_1,b_1')\\
&\chi^{(2)}=\chi_{L,\epsilon}(b_1,b) \qquad \chi^{(5)}=\chi_{L,\epsilon}(a_1,c_2)\\
&\chi^{(3)}=\chi_{L,\epsilon}(a_1',c_1) \qquad \chi^{(6)}=\chi_{L,\epsilon}(c_2,b_1).
\end{align*}
Then
\[\chi_{L,\epsilon}(a,b) \le \max(\chi^{(1)}, \dots, \chi^{(6)})+L\beta-\frac{\eta}{L},\]
where
\[\beta= 2\epsilon^{2-4u} \quad\text{and}\quad \eta=\frac{D\epsilon}{16R}.\]

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:laakso_1}\emph{Step 1. Restrictions of admissible functions.} If no $(L,\epsilon)$-admissible function $P_{a,b} \to \ell^2$ exists, then $\chi_{L,\epsilon}(a,b)=-\infty$ and the claim holds trivially.  So suppose that at least one exists. Set $M=\max(\chi^{(1)}, \dots, \chi^{(6)})$ and let $f$ be an $(L,\epsilon)$-admissible function. By the remarks above, all six of the parallelograms $P_{a,a_1}$, $P_{b_1,b}$, $P_{a_1',c_1}$, $P_{c_1,b_1'}$, $P_{a_1,c_2}$, $P_{c_2,b_1}$ are contained in $P_{a,b}$. Also, each of these segments is either contained in the segment from $a$ to $b$, or is on the top geodesic of a good bubble; in particular, in each case the right endpoint lies on the bottom-most geodesic started from the left endpoint, so that the six parallelograms are of the form to which \Cref{def:L-eps_admissible} applies.  These observations imply that the restriction of $f$ to any of these six parallelograms is also an $(L,\epsilon)$-admissible function.  Indeed, for condition~\eqref{it:adm_lipschitz}, a path contained in one of the six parallelograms is contained in $P_{a,b}$ (so the minimal path length can only decrease when computed in $P_{a,b}$, which strengthens the required upper bound), and a geodesic segment contained in one of them is contained in $P_{a,b}$; and for conditions~\eqref{it:adm_bubble_start}--\eqref{it:adm_bubble_middle}, a good bubble contained in one of the six parallelograms is contained in $P_{a,b}$, so the corresponding requirements on $f$ are among those already imposed.  In particular, since an admissible function exists on $P_{a,b}$, one exists on each of the six parallelograms, and hence $1 \le \chi^{(j)} \le L$ for all $j$, so that $1 \le M \le L$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:laakso_2}\emph{Step 2. Decomposition into four terms.} By the triangle inequality, We have
\begin{align*}
\|f(a)-f(b)\|_2 &\le \|f(a)-f(a_1)\|_2+\left\|f(a_1)-\frac{f(c_1)+f(c_2)}{2}\right\|_2\\
&\phantom{=}+\left\|\frac{f(c_1)+f(c_2)}{2}-f(b_1)\right\|_2+\|f(b_1)-f(b)\|_2\\
&=:T_1+T_2+T_3+T_4.
\end{align*}
We now estimate each of these four terms below.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:laakso_3}\emph{Step 3. The terms $T_1$, $T_2$ and $T_4$.} For the first and fourth terms, we have
\[T_1 \le \chi^{(1)}d(a,a_1) \le Md(a,a_1)\]
and
\[T_4 \le \chi^{(2)}d(b_1,b) \le Md(b_1,b).\]
For the second term, we write 
\[f(a_1)-\frac{f(c_1)+f(c_2)}{2}=\frac12\big[f(a_1)-f(a_1')\big]+\frac12\big[f(a_1')-f(c_1)\big]+\frac12\big[f(a_1)-f(c_2)\big]\] 
and apply the triangle inequality, which gives
\begin{align*}
T_2 &\le \frac{\|f(a_1)-f(a_1')\|_2}{2}+\frac{\|f(a_1')-f(c_1)\|_2}{2}+\frac{\|f(a_1)-f(c_2)\|_2}{2}\\
&\le \frac{\|f(a_1)-f(a_1')\|_2}{2}+\frac{M}{2}d(a_1',c_1)+\frac{M}{2}d(a_1,c_2).
\end{align*}
Since $f$ is $(L,\epsilon)$-admissible (namely, by condition~\eqref{it:adm_bubble_start} of \Cref{def:L-eps_admissible}, applied to our good bubble, which is contained in $P_{a,b}$ by hypothesis),
\[\frac{\|f(a_1)-f(a_1')\|_2}{2} \le L\epsilon^{2-u}n\le \frac{L\beta}{2} d(a,b),\]
where the second inequality holds because $\epsilon^{2-u} \le \epsilon^{2-4u}$ and $d(a,b)=n$.
Furthermore, $d(a_1',c_1)=d(a_1,c_2)$. This is because $a_1'$ and $a_1$ have the same time coordinate, $c_1$ and $c_2$ have the same time coordinate, $c_1$ is on the bottom-most geodesic started from $a_1'$ and $c_2$ is on the bottom-most geodesic started from $a_1$, so that both distances are equal to the common difference of the time coordinates. Altogether,
\[T_2\le \frac{L\beta}{2} d(a,b)+Md(a_1,c_2).\]

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:laakso_4}\emph{Step 4. The term $T_3$ and the parallelogram law.} Next, for $T_3$, we have by the parallelogram law,
\begin{equation}\label{eq:parallelogram_law}
\left\|f(b_1)-\frac{f(c_1)+f(c_2)}{2}\right\|_2^2=\frac{\|f(b_1)-f(c_1)\|_2^2}{2}+\frac{\|f(b_1)-f(c_2)\|_2^2}{2}-\frac{\|f(c_1)-f(c_2)\|_2^2}{4}.
\end{equation}
Now, since $f$ is $(L,\epsilon)$-admissible (the first term below is bounded using the triangle inequality together with condition~\eqref{it:adm_bubble_merge} of \Cref{def:L-eps_admissible}, and the second using the definition of $\chi^{(4)}$ applied to the restriction of $f$ to $P_{c_1,b_1'}$), we have
\begin{align*}
\frac{\|f(b_1)-f(c_1)\|_2^2}{2} &\le \frac{1}{2} \left( \|f(b_1)-f(b_1')\|_2+\|f(b_1')-f(c_1)\|_2 \right)^2\\
&\le \frac{1}{2}\left(L\epsilon^{2-4u}n+Md(b_1',c_1)\right)^2\\
&= \frac{1}{2}\left(L\epsilon^{2-4u}n+Md(b_1,c_2)\right)^2.
\end{align*}
In the last line, we used the fact that $b_1$ and $b_1'$ have the same time coordinates, and likewise $c_1$ and $c_2$ have the same time coordinates. So because $b_1'$ is on the bottom-most geodesic started from $c_1$, and $b_1$ is on the one started from $c_2$, we have $d(b_1',c_1)=d(b_1,c_2)$.

We also have
\[\frac{\|f(b_1)-f(c_2)\|_2^2}{2}  \le \frac{(M d(b_1,c_2))^2}{2} \le \frac{1}{2}\left(L\epsilon^{2-4u}n+Md(b_1,c_2)\right)^2,\]
and by~\eqref{it:adm_bubble_middle} of \Cref{def:L-eps_admissible},
\[\frac{\|f(c_1)-f(c_2)\|_2^2}{4} \ge \frac{D\epsilon^2n^2}{4}.\]
Therefore \eqref{eq:parallelogram_law} reduces to
\[T_3 \le \sqrt{(L\epsilon^{2-4u}n+Md(b_1,c_2))^2-\frac{D\epsilon^2n^2}{4}}.\]
Note that the quantity under the square root is automatically non-negative, since by the above it is an upper bound for $T_3^2$.
The inequality
\[\sqrt{\lambda-\mu} \le \sqrt{\lambda}-\frac{\mu}{2\sqrt{\lambda}}\]
holds for any $\lambda>0$ and $\mu\le\lambda$, by concavity of the function $\mu \mapsto \sqrt{\lambda-\mu}$. Applying it with $\lambda=(L\epsilon^{2-4u}n+Md(b_1,c_2))^2$ and $\mu=D\epsilon^2n^2/4$, we get
\[T_3 \le L\epsilon^{2-4u}n+Md(b_1,c_2)-\frac{D\epsilon^2n^2}{8(L\epsilon^{2-4u}n+Md(b_1,c_2))}.\]
The distance between $b_1$ and $c_2$ is at most $R\epsilon n$ by part~\eqref{it:top_survival} of \Cref{def:top_good}. We have $M \le L$, and $\epsilon^{2-4u} n \le R \epsilon n$ for $\epsilon$ sufficiently small. These observations imply that
\[\frac{D\epsilon^2n^2}{8(L\epsilon^{2-4u}n+Md(b_1,c_2))} \ge \frac{D\epsilon^2n^2}{8(L\epsilon^{2-4u}n+LR\epsilon n)} \ge \frac{D\epsilon n}{16LR}.\]
So overall,
\[T_3 \le L\epsilon^{2-4u}n+Md(b_1,c_2)-\frac{D\epsilon n}{16LR}= \frac{L\beta}{2}d(a,b)+Md(b_1,c_2)-\frac{\eta}{L}d(a,b).\]

  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr5-1\relax}
  \refstepcounter{proofstep}\label{step:laakso_5}\emph{Step 5. Conclusion.} Putting all these estimates together, we thus obtain
\[\|f(a)-f(b)\|_2 \le M( d(a,a_1)+d(a_1,c_2)+d(c_2,b_1)+d(b_1,b))+L\beta d(a,b)-\frac{\eta}{L}d(a,b).\]
The path $a \to a_1 \to c_2 \to b_1 \to b$ is a geodesic from $a$ to $b$, so the term in parentheses is exactly $d(a,b)$. Hence, we finally obtain
\[\|f(a)-f(b)\|_2 \le \left(M+L\beta-\frac{\eta}{L}\right)d(a,b).\]
Since $f$ was an arbitrary $(L,\epsilon)$-admissible function, the result follows.

  \refstepcounter{proofstep}\label{step:distrect_1}\emph{Step 1. Setup and the choice of $\rho$.} For $\epsilon>0$ small, we set $n=e^{\epsilon^{-2}}$. \end{equation}
More specifically, we can pick $\rho$ as follows. Let $c_{\ref{prop:no_good_bubbles}}$ and $c_{\ref{lem:conditional_prob_given_failure}}$ be the constants appearing in \Cref{prop:no_good_bubbles} and \Cref{lem:conditional_prob_given_failure}, respectively; recall that these constants do not depend on the parameter $B$. \noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:distrect_2}\emph{Step 2. The induction: base cases and decomposition.} Let us now prove~\eqref{eq:tail_bound_chi}. We argue by induction on $n$. When $k=0$, this is trivial since the left-hand side of~\eqref{eq:tail_bound_chi} is 0; we always have $\chi_{L,\epsilon}(a,b) \le L$ by definition of $\chi_{L,\epsilon}$. If $k \ge 1$ and $n \le 16\epsilon^{-4}$, then this is also trivial because in that case the right-hand side of~\eqref{eq:tail_bound_chi} is at least 1.  (Indeed, $16\epsilon^{-4}<32\epsilon^{-5}=(2/\epsilon)^5$, so that $n<(2/\epsilon)^5$ and hence $\log n/\log(2/\epsilon)<5$; thus $g(n) \le 1 \le k$, where $g$ is defined below, and $\rho^{g(n)-k} \ge 1$ because $\rho<1$).

So we may assume from now on that $k \ge 1$ and $n>16\epsilon^{-4}$.  Let us stress that the induction applies the estimates of \Cref{sec:exploration_process} at the scale $n$, for every such $n$, with $\epsilon$ held fixed; this is legitimate precisely because the lower bound on $n$ required there is polynomial in $\epsilon^{-1}$ ($n \ge \epsilon^{-4}$ suffices, as explained in \Cref{fn:parameters}) and $n>16\epsilon^{-4}$ here.  It is only the largest scale, at which the proposition is finally applied, that is taken superpolynomially large in $\epsilon^{-1}$.

Now assume that for all $n'<n$, we have proven~\eqref{eq:tail_bound_chi} holds for all $k \ge 0$ (with $n'$ instead of $n$). For each $x$ and $\ell$, let $G_{x,\ell}$ be the event that the bubble started at $x$ is good and has merging time $\ell$, so that the point $b_1$ of \Cref{lem:laakso_diamond} lies at time $x+\ell$. Also let $F$ be the event that there is a failure (recall this means that either there are no good bubbles, or that an independent $Z \sim \mathrm{Ber}(c_{\ref{prop:no_good_bubbles}}/\sqrt{B})$ is $1$). Then
\[\mathbf{P}(A_{n,k})=\mathbf{P}\!\left( A_{n,k} \cap \left( \bigcup_{x,\ell} G_{x,\ell} \right) \cap \{Z=0\}\right)+\mathbf{P}(A_{n,k} \cap F)=: \Pi_1+\Pi_2. \]

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:distrect_3}\emph{Step 3. Bounding $\Pi_1$: the case of a good bubble.} We have that
\[\Pi_1 \le \mathbf{P}\!\left( A_{n,k} \cap \left( \bigcup_{x,\ell} G_{x,\ell} \right)\right)\le \sum_{x, \ell} \mathbf{P}(A_{n,k} \cap G_{x,\ell})=\sum_{x, \ell} \mathbf{P}(A_{n,k} \,|\, G_{x,\ell}) \mathbf{P}(G_{x,\ell}),\]
where the sum runs over the pairs $(x,\ell)$ with $\mathbf{P}(G_{x,\ell})>0$.
When we have a good bubble at $x$, we can repeat the exploration on the six segments around this good bubble shown in \Cref{fig:exploration around good bubble}. Then by \Cref{lem:laakso_diamond} (whose containment hypothesis is satisfied, since $x$ lies in the middle third of $[0,n]$),
\[\mathbf{P}(A_{n,k} \,|\, G_{x,\ell}) \le \mathbf{P}(\max(\chi^{(1)}, \dots, \chi^{(6)})-s>L-ks \,|\, G_{x,\ell}) \le \sum_{j=1}^6 \mathbf{P}(\chi^{(j)}>L-(k-1)s \,|\, G_{x,\ell}).\]
Now given $G_{x,\ell}$,
\begin{enumerate}[(i)]
\item\label{it:mkv_left} The law of the triangulation to the left of the bubble is an LHPT on top of the segment $[0,x]$, and is independent of everything after $x$; this is the region carrying $\chi^{(1)}$.
\item\label{it:mkv_right} The triangulation to the right of the bubble is an LHPT on top of the segment $[x+\ell,n]$, and is independent of everything before $x+\ell$; this is the region carrying $\chi^{(2)}$.
\item\label{it:mkv_top_first} The triangulation on top of the bubble over the segment $[x,x+\epsilon n/2]$ is an LHPT, and is independent of everything below the bubble, to the left of $x$ and to the right of $x+\epsilon n/2$; this is the region carrying $\chi^{(3)}$.
\item\label{it:mkv_top_second} The triangulation on top of the bubble over the segment $[x+\epsilon n/2, x+\ell]$ is an LHPT and is independent of everything below the bubble, to the left of $x+\epsilon n/2$ and to the right of $x+\ell$; this is the region carrying $\chi^{(4)}$.
\end{enumerate}
Thus, for $j=1,2,3,4$ and any $t$, the event $\{\chi^{(j)}>t\}$ is independent of the top exploration of the bubble started at $x$.  This holds for $j=1$ and $j=2$ by~\eqref{it:mkv_left} and~\eqref{it:mkv_right} since the parallelograms involved are to the left of time $x$ and to the right of time $x+\ell$ respectively, whereas the top exploration lies between these two times.  This holds for $j=3$ and $j=4$ by~\eqref{it:mkv_top_first} and~\eqref{it:mkv_top_second}, i.e., by the spatial Markov property above the top geodesic.  For $j=2$ it is \emph{not}, however, independent of $G_{x,\ell}$.  This is because being good also constrains the bottom exploration \emph{after} the merging time, both through~\eqref{it:bottom_height_bound} of \Cref{def:bottom_good}, whose height bounds are used up to time $R\epsilon n \ge \ell$, and through~\eqref{it:merge_distance} of \Cref{def:bubble_good}, which concerns a path lying to the right of time $\ell$, and the region which these two conditions constrain meets the parallelogram $P_{b_1,b}$ of the second segment.  We therefore use, for all four of these segments, the product bound which the last two segments require in any case.  Write $G^1_{x,\ell} \supseteq G_{x,\ell}$ for the event of \Cref{lem:conditional_prob_given_good} for the bubble started at $x$, namely that its top exploration is good and that its merging time equals $\ell$; this event is measurable with respect to the top exploration of that bubble, and hence, on the event that the merging time equals $\ell$, with respect to the restriction of the LHPT to $[x,x+\ell] \times \mathbf{Z}$.  Consequently
\[\mathbf{P}\!\big(\{\chi^{(j)}>L-(k-1)s\} \cap G_{x,\ell}\big) \le \mathbf{P}(\chi^{(j)}>L-(k-1)s)\,\mathbf{P}(G^1_{x,\ell}) \quad\text{for $j=1,\dots,4$,}\]
since $G_{x,\ell} \subseteq G^1_{x,\ell}$ and the two events on the right-hand side are independent.
For $\chi^{(5)}$ and $\chi^{(6)}$, we instead use \Cref{lem:conditional_prob_given_good}. For any $t$, the event $\{\chi^{(5)}>t\}$ only depends on the parallelogram above the segment from $a_1$ to $c_2$, and likewise for $\chi^{(6)}$, and by the discussion at the start of this section, those parallelograms have height at most $R^3 \epsilon^4 n^2$, so that the $\ell$-resolved bound~\eqref{eq:good_ell_resolved} applies, and we have
\[\mathbf{P}\!\big(\{\chi^{(j)}>L-(k-1)s\} \cap G_{x,\ell}\big)\le \mathbf{P}(\chi^{(j)}>L-(k-1)s)\,\mathbf{P}(G^1_{x,\ell}) \quad\text{for $j=5,6$}\]
as well.
Writing
\[g(n)=\frac{1}{4}\left \lfloor \frac{\log n}{\log(2/\epsilon)} \right \rfloor,\]
it therefore follows from the induction hypothesis (which applies to each of the four scales occurring below, namely $x$, $n-(x+\ell)$, $\epsilon n/2$ and $\ell-\epsilon n/2$, since each of them is smaller than $n$ (we have $x \le 2n/3$, $n-(x+\ell) \le 2n/3$, and $\ell \le R\epsilon n<n$ for $\epsilon$ small) and at least $\epsilon n/2>0$ (we have $x \ge n/3$, $n-(x+\ell) \ge n/3-R\epsilon n$ and $\ell \ge \epsilon n$)) that
\begin{align}
\Pi_1 &\le \sum_{x, \ell} \left( \mathbf{P}(A_{x,k-1})+\mathbf{P}(A_{n-(x+\ell),k-1})+2\mathbf{P}(A_{\epsilon n/2,k-1})+2\mathbf{P}(A_{\ell-\epsilon n/2,k-1})\right)\mathbf{P}(G^1_{x,\ell}) \notag\\
&\le \sum_{x, \ell} \left( \rho^{g(x)-(k-1)}+\rho^{g(n-(x+\ell))-(k-1)}+2\rho^{g(\epsilon n/2)-(k-1)}+2\rho^{g(\ell-\epsilon n/2)-(k-1)} \right)\mathbf{P}(G^1_{x,\ell})  \label{eq:P1_recursion}\\
\intertext{Using that $g$ is non-decreasing, that good bubbles always have merging time at least $\epsilon n$, and that we only look for good bubbles in the middle third of the starting segment (so $n/3 \le x \le 2n/3$) (whence $x \ge n/3 \ge \epsilon n/2$, $n-(x+\ell) \ge n/3-R\epsilon n \ge \epsilon n/2$ and $\ell-\epsilon n/2 \ge \epsilon n/2$ for $\epsilon$ small, so that all four exponents are at least $g(\epsilon n/2)$), we see that~\eqref{eq:P1_recursion} is bounded from above by}
&\phantom{\le}\qquad\qquad\qquad\qquad\qquad\qquad 6\rho^{g(\epsilon n/2)-(k-1)} \sum_{x, \ell} \mathbf{P}(G^1_{x, \ell}). \notag
\end{align}
Note also that by~\eqref{eq:good_top_decomposed} we have $\sum_{\ell}\mathbf{P}(G^1_{x,\ell}) \le B\epsilon^2$ for each $x$, and since we test for good bubbles at at most $\epsilon^{-2}$ starting times, this gives $\sum_{x,\ell}\mathbf{P}(G^1_{x,\ell}) \le B$.  (The same count applies to $\sum_{x,\ell}\mathbf{P}(G_{x,\ell})$, which is the expected number of good bubbles, since by \Cref{prop:good_bubble_probability} the probability that a bubble is good is at most $B\epsilon^2$.)  Hence $\Pi_1 \le 6B\rho^{g(\epsilon n/2)-(k-1)}$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:distrect_4}\emph{Step 4. Bounding $\Pi_2$: the case of a failure.} If we have a failure, we can still break the starting segment into its six sub-segments and do the exploration on each of them. Let $v_1, \dots, v_5$ be the endpoints of the six sub-segments, $v_0=(0,0)$, and $v_6=(n,0)$, and set in this case
\[\chi^{(j)}=\chi_{L,\epsilon}(v_{j-1},v_j).\]
Since the $v_j$'s all lie on a geodesic, we have that for any $(L,\epsilon)$-admissible function $f$ on $P_{v_0,v_6}$ (whose restriction to each $P_{v_{j-1},v_j}$ is again admissible, since these parallelograms are contained in $P_{v_0,v_6}$, their segments being sub-segments of the segment from $v_0$ to $v_6$),
\begin{align*}
\|f(v_0)-f(v_6)\|_2 &\le \|f(v_0)-f(v_1)\|_2+\dots+\|f(v_5)-f(v_6)\|_2\\
&\le \chi^{(1)}d(v_0,v_1)+\dots+\chi^{(6)}d(v_5,v_6)\\
&\le \max(\chi^{(1)},\dots,\chi^{(6)})( d(v_0,v_1)+\dots+d(v_5,v_6) )\\
&=\max(\chi^{(1)},\dots,\chi^{(6)}) d((0,0),(n,0)),
\end{align*}
so that
\[\chi_{L,\epsilon}((0,0),(n,0)) \le \max(\chi^{(1)},\dots,\chi^{(6)}).\]
Then
\begin{align*}
\Pi_2&\le \mathbf{P}(\max(\chi^{(1)},\dots,\chi^{(6)})>L-ks \,|\, F)\mathbf{P}(F)\\
&\le \left(\sum_{j=1}^6\mathbf{P}(\chi^{(j)}>L-ks \,|\, F)\right) \cdot 2c_{\ref{prop:no_good_bubbles}}\left(\frac{1}{\sqrt{B}}+\epsilon\right) \quad\text{(by \Cref{prop:no_good_bubbles})}\\
&\le 12 c_{\ref{prop:no_good_bubbles}}c_{\ref{lem:conditional_prob_given_failure}}\left(\frac{1}{\sqrt{B}}+\epsilon\right)\mathbf{P}(A_{n/6,k}) \quad\text{(by \Cref{lem:conditional_prob_given_failure})}\\
&\le 12c_{\ref{prop:no_good_bubbles}}c_{\ref{lem:conditional_prob_given_failure}}\left(\frac{1}{\sqrt{B}}+\epsilon\right) \rho^{g(n/6)-k} \quad\text{(the induction hypothesis)}\\
&\le 12 c_{\ref{prop:no_good_bubbles}}c_{\ref{lem:conditional_prob_given_failure}}\left(\frac{1}{\sqrt{B}}+\epsilon\right) \rho^{g(\epsilon n/2)-k}.
\end{align*}
Here the bound $\mathbf{P}(F) \le 2c_{\ref{prop:no_good_bubbles}}(1/\sqrt{B}+\epsilon)$ used in the second line follows from \Cref{prop:no_good_bubbles} together with the definition of a failure: 
\[\mathbf{P}(F) \le \mathbf{P}(E^c)+\mathbf{P}(Z=1) \le c_{\ref{prop:no_good_bubbles}}\left(\epsilon+\frac{1}{\sqrt{B}}\right)+\frac{c_{\ref{prop:no_good_bubbles}}}{\sqrt{B}} \le 2c_{\ref{prop:no_good_bubbles}}\left(\frac{1}{\sqrt{B}}+\epsilon\right).\]
In the third line, \Cref{lem:conditional_prob_given_failure} is applied separately to each of the six events $\{\chi^{(j)}>L-ks\}$, which is an event for the LHPT restricted to the $j$-th sub-segment of the starting segment, and we then use that $\mathbf{P}(\chi^{(j)}>L-ks)=\mathbf{P}(A_{n/6,k})$ for every $j$, since the part of the LHPT on top of each of the six sub-segments is again an LHPT.  In the last line we also used that $g$ is non-decreasing and $\epsilon n/2 \le n/6$.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr5-1\relax}
  \refstepcounter{proofstep}\label{step:distrect_5}\emph{Step 5. Completing the induction.} Putting all this together, we thus have, using the fact that $g(\epsilon n/2)-g(n)=-1/4$ and our choices of $\rho$ and $B$,
\begin{align*}
\mathbf{P}(A_{n,k}) &\le 6B \rho^{g(\epsilon n/2)-(k-1)}+ 12 c_{\ref{prop:no_good_bubbles}}c_{\ref{lem:conditional_prob_given_failure}}\left(\frac{1}{\sqrt{B}}+\epsilon\right) \rho^{g(\epsilon n/2)-k}\\
&=\left( 6B \rho^{\frac{3}{4}}+12 c_{\ref{prop:no_good_bubbles}}c_{\ref{lem:conditional_prob_given_failure}}\left(\frac{1}{\sqrt{B}}+\epsilon\right)\rho^{-\frac{1}{4}}\right) \rho^{g(n)-k} \le \rho^{g(n)-k}.
\end{align*}
This completes the proof of~\eqref{eq:tail_bound_chi}.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr6-1\relax}
  \refstepcounter{proofstep}\label{step:distrect_6}\emph{Step 6. Conclusion.} Now pick $L=\epsilon^{2\delta-1/2}$. Then
\[s=\frac{D\epsilon}{16 R L}-2L\epsilon^{2-4u}=\frac{D}{16 R}\epsilon^{\frac{3}{2}-2\delta}-2\epsilon^{\frac{3}{2}+2\delta-4u}.\]
So for all $\epsilon$ sufficiently small,
\[\frac{1}{2\Lambda}\epsilon^{\frac{3}{2}-2\delta} \le s \le \frac{1}{\Lambda}\epsilon^{\frac{3}{2}-2\delta},\]
where we have set $\Lambda=16R/D$ for convenience.  It is here that the choice $u<\delta$ made in \Cref{step:distrect_1} is used: the second exponent $\frac32+2\delta-4u$ exceeds the first exponent $\frac32-2\delta$ by $4(\delta-u)>0$, so the second term in the expression for $s$ is negligible with respect to the first as $\epsilon \to 0$. In particular, $s$ is positive, and so for this choice of $L$, we have with $k=\lceil L/s \rceil$ that $L-ks \le 0$. So on the event that $\chi_{L,\epsilon}((0,0),(n,0)) \le L-ks$, there is no $(L,\epsilon)$-admissible function from $P_{(0,0),(n,0)}$ into $\ell^2$ (if one existed, then, as noted after the definition of $\chi_{L,\epsilon}$, we would have $\chi_{L,\epsilon}((0,0),(n,0)) \ge 1 > 0 \ge L-ks$), and hence, by the discussion following \Cref{def:L-eps_admissible}, the $\ell^2$ distortion of $P_{(0,0),(n,0)}$ is at least $L$. Since the parallelogram $P_{(0,0),(n,0)}$ is contained in $Q_n$ for $\epsilon$ small (its height is $R\epsilon^2n^2 \le n^2$), and since the distortion of a subset is at most that of the whole set (a function on $Q_n$ satisfying~\eqref{eq:distortion_def} restricts to one on $P_{(0,0),(n,0)}$), it follows that the distortion of $Q_n$ is at least $L$:
\[\mathbf{P}(\mathrm{Dis}(Q_n) \ge L) \ge \mathbf{P}(A_{n,k}^c) \ge 1-\rho^{g(n)-k} \ge 1-\rho^{g(n)-\frac{L}{s}-1}.\]
But now
\[\Lambda\epsilon^{4\delta-2} \le \frac{L}{s} \le 2\Lambda\epsilon^{4\delta-2},\]
so recalling that we had set $n=e^{\epsilon^{-2}}$,
\[g(n)-\frac{L}{s}-1 \ge \frac{\epsilon^{-2}}{4\log(2/\epsilon)}-2\Lambda\epsilon^{4\delta-2}-2 \ge \epsilon^{3\delta-2}=(\log n)^{1-\frac{3\delta}{2}}\]
for $\epsilon$ sufficiently small. Also,
\[L=\epsilon^{2\delta-\frac{1}{2}}=(\epsilon^{-2})^{\frac{1}{4}-\delta}=(\log n)^{\frac{1}{4}-\delta}.\]
Hence
\begin{align*}
\mathbf{P}(\mathrm{Dis}(Q_n) \ge (\log n)^{\frac{1}{4}-\delta})&=\mathbf{P}(\mathrm{Dis}(Q_n) \ge L) \ge 1-\rho^{g(n)-\frac{L}{s}-1}\\
&\ge 1-\rho^{(\log n)^{1-\frac{3\delta}{2}}} \to 1.
\end{align*}
Note that what the computation bounds is the distortion of the parallelogram $P_{(0,0),(n,0)}$, that of $Q_n$ being deduced from it; this is the form in which the proposition will be applied.
This concludes the proof.

Every path of $\mathcal {S}$ is a path of $\mathfrak{M}$, so $d_{\mathrm{gr}}^{\mathfrak{M}} \le d_{\mathrm{gr}}^{\mathcal {S}}$ on $P$.  Conversely, let $x,y \in P$ and let $\pi$ be a path of $\mathfrak{M}$ from $x$ to $y$, which by~\eqref{it:inh_diameter} we may assume to be of length at most $D$.  A vertex of $\mathcal {S}$ which is not on its boundary has all of its neighbors in $\mathfrak{M}$ among the vertices of $\mathcal {S}$, the faces of $\mathcal {S}$ incident to it being faces of $\mathfrak{M}$ which already surround it; the path $\pi$ can therefore leave $\mathcal {S}$ only from a vertex of its boundary. The boundary of $\mathcal {S}$ consists of three parts: the starting vertical segment and the top and bottom geodesics run until they coalesce.  Every vertex $u$ of that boundary visited by $\pi$ carries a time coordinate which is within $D$ of the time coordinate of $x$, so that its time lies in $[t_0-D,t_0+m+D]$; by $D<t_0$ and~\eqref{it:inh_alive} it does not lie on the initial segment. By ~\eqref{it:inh_confined}, it cannot lie on the top geodesic. Hence, it must lie on the bottom geodesic.

Finally, if $\pi$ exits from the bottom geodesic at $u$ and then re-enters from $u'$, then we can modify the portion of $\pi$ from $u$ to $u'$ by simply following the bottom geodesic between them. Since it's a geodesic from $u$ to $u'$, this modification can only decrease the length, and the portion from $u$ to $u'$ is now in $\mathcal {S}$. Doing this to all the exits and re-entries of $\pi$ from the bottom geodesic gives that the length of $\pi$ is at least $d_{\mathrm{gr}}^{\mathcal {S}}(x,y)$.

Thus $d_{\mathrm{gr}}^{\mathcal {S}}(x,y) \le d_{\mathrm{gr}}^{\mathfrak{M}}(x,y)$.  The metric which $\mathfrak{M}$ induces on $P$ is therefore that of $\mathcal {S}$, and the $\ell^2$ distortion of a metric space is at least that of any of its subspaces.

The height of the slice is a Galton--Watson process with offspring distribution $\theta$ started from $n^2$, so by \Cref{lem:tail_bound_minimum}, applied with $x=n^2$, with $n^{3/4}$ in place of $n$ and with $n^{5/4}$ in the role of the parameter called $a$ there, the probability that~\eqref{it:slice_deep} fails is at most

  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:distuipt_1}\emph{Step 1. Choice of the parameter $a$.} Fix $\epsilon>0$, and assume for notational simplicity that $n$ is even, so that $n/2$ is an integer (otherwise replace $n/2$ by $\lfloor n/2\rfloor$ throughout). By \cite[Theorem 2]{cl2017peel}, we have that $L_n/n^2$ converges in distribution to the $\mathrm{Gamma}(3/2)$ distribution. It follows that if $a>0$ is sufficiently small, then for all $n$ sufficiently large,
\[\mathbf{P}(A_1) \ge 1-\frac{\epsilon}{2} \quad\text{for}\quad A_1=\{an^2 \le L_n \le a^{-1}n^2\} \cap \{an^2 \le L_{n/2} \le a^{-1}n^2\}.\]
We fix this choice of $a$ for the rest of the proof.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:distuipt_2}\emph{Step 2. The slice.} Let $\mathcal {S}$ be the slice spanned by the first $n$ trees of $\widetilde{\mathcal {F}}_{n/2,n}^{(1)}$, that is, the sub-triangulation of the annulus $B_n^\bullet(\mathcal {T}_\infty^{(1)}) \setminus B_{n/2}^\bullet(\mathcal {T}_\infty^{(1)})$ lying between the bottom-most geodesics started at the two ends of the corresponding arc of $\partial B_n^\bullet(\mathcal {T}_\infty^{(1)})$, built from those $n$ trees and from the Boltzmann triangulations filling their slots. Its initial height is $n=(\sqrt{n})^2$, which is the height for which \Cref{prop:slice_geometry} is stated with $\sqrt{n}$ in place of $n$.  Its vertex at time $t$ and height $h$ lies on the cycle $\partial B_{n-t}^\bullet(\mathcal {T}_\infty^{(1)})$, and its height $Z_t$ is the population at generation $t$ of the forest formed by those $n$ trees.  On $A_1$ we have $L_n \ge an^2 \ge n$ for $n$ large, so that $\mathcal {S}$ is then well defined.

Being a deterministic function of those $n$ trees and of the fillings of the slots which they index, which are i.i.d.\ Boltzmann triangulations independent of the forest both here and in the LHPT, the slice can be compared with its LHPT counterpart.  Summing the bound of \Cref{lem:lhpt_to_uipt} over the forests with the lemma applied with $r=n/2$, $s=n$ and with $a/4$ in place of $a$ (every pair $p,q \in [an^2,a^{-1}n^2]$ satisfies $(a/4)(n/2)^2<p,q\le(a/4)^{-1}(n/2)^2$ and on $A_1$ the forest belongs to $\mathbf{F}_{p,q,n/2}''$ for such a pair) and then integrating over the fillings shows that for every set $\mathbf{S}$ of slices,
\begin{equation}\label{eq:slice_comparison}
\mathbf{P}(\{\mathcal {S} \in \mathbf{S}\} \cap A_1) \le C_1\,\mathbf{P}(\mathcal {S}' \in \mathbf{S}),
\end{equation}
where $\mathcal {S}'$ is the slice spanned by the first $n$ trees of $([\tau_1]_{n/2},\dots,[\tau_{N}]_{n/2})$, $N=N_{n/2}^{(a/4)}$, with independent Boltzmann fillings.  Here $n<(a/4)(n/2)^2<N$ for $n$ large, so that $\mathcal {S}'$ is well defined, and $\mathcal {S}'$ is the slice $\mathcal {S}^{(n)}$ of initial height $n$ of an LHPT, up to the truncation of its trees at generation $n/2$.

  \refstepcounter{proofstep}\label{step:distuipt_3}\emph{Step 3. The set $\mathbf{G}_n$.} Let $E$ be the event of \Cref{prop:slice_geometry}, applied with $\sqrt{n}$ in place of $n$ and with $\delta/2$ in place of $\delta$.  Note that $\mathbf{P}(\mathcal {S}' \notin E) \to 0$.  Let $\mathbf{G}_n$ be the set of pairs consisting of a hull of radius $n$ and of a marked vertex on its boundary cycle for which the slice determined by the pair as in \Cref{step:distuipt_2} belongs to $E$.  By~\eqref{eq:slice_comparison},
\[\mathbf{P}\big((B_n^\bullet(\mathcal {T}_\infty^{(1)}),w_n) \notin \mathbf{G}_n\big) \le \mathbf{P}(A_1^c)+\mathbf{P}(\{\mathcal {S} \notin E\} \cap A_1) \le \frac{\epsilon}{2}+C_1\,\mathbf{P}(\mathcal {S}' \notin E)\]
for $n$ large.  For each fixed $\epsilon>0$ the second term tends to $0$. The left-hand side thus tends to $0$ since $\epsilon>0$ was arbitrary and $E$ does not depend on $\epsilon$.

  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:distuipt_4}\emph{Step 4. Conclusion.} Let $(\mathfrak{H},w) \in \mathbf{G}_n$, let $\mathfrak{S}$ be the slice which this pair determines, let $\rho$ be the root of $\mathfrak{H}$, and let $\mathfrak{M}$ be a triangulation containing $\mathfrak{H}$ as a sub-triangulation and in which the distances from $\rho$ to the vertices of $\mathfrak{H}$ are those of $\mathfrak{H}$.  Then $\mathfrak{S}$ is a sub-triangulation of $\mathfrak{M}$, and its vertex at time $t$ and height $h$ lies on the cycle $\partial B_{n-t}^\bullet(\mathfrak{H})$, hence is at distance $n-t$ from $\rho$ in $\mathfrak{H}$ and so also in $\mathfrak{M}$. Consequently $|d_{\mathrm{gr}}^{\mathfrak{M}}(\rho,u)-d_{\mathrm{gr}}^{\mathfrak{M}}(\rho,v)| \le d_{\mathrm{gr}}^{\mathfrak{M}}(u,v)$ for all vertices $u,v$ of $\mathfrak{S}$ carrying time coordinates.  Since $\mathfrak{S} \in E$, \Cref{lem:slice_distances} applies, with $t_0=\tfrac12 n^{3/8}$, $m=n^{1/8}$ and $D=4n^{1/4}$, the trees of $\mathfrak{S}$ being truncated at the generation $n/2>n^{3/8}$.  With $P$ the parallelogram of \Cref{prop:slice_geometry} we have that
\[\mathrm{Dis}(\mathfrak{M}) \ge \mathrm{Dis}(P) \ge (\log \sqrt{n})^{(1/4)-\delta/2} \ge (\log n)^{(1/4)-\delta}\]
for $n$ large, the distortion of $P$ computed with the distances of $\mathfrak{S}$, the second inequality being property~\eqref{it:slice_distortion} of $E$, and the third holding because the ratio of the last two quantities is $2^{-(1/4)+\delta/2}(\log n)^{\delta/2} \to \infty$.  Together with \Cref{step:distuipt_3}, and taking $\mathfrak{M}=\mathfrak{H}=B_n^\bullet(\mathcal {T}_\infty^{(1)})$, this proves both assertions of the proposition.

\label{lem:uipt_to_sphere}
There exists a constant $\overline{c}$ such that, for every $n \ge 1$, for every $r \ge 1$ and $\mathfrak{T} \in \mathbf{C}_{1,r}$ such that $n > N(\mathfrak{T})$,
\[ \mathbf{P}( B_r^\bullet(\overline{\mathcal {T}}_n^{(1)}) = \mathfrak{T}) \le \overline{c} \left( \frac{n}{n-N(\mathfrak{T})} \right)^{3/2} \mathbf{P}( B_r^\bullet(\mathcal {T}_\infty^{(1)}) = \mathfrak{T}).\]

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr1-1\relax}
  \refstepcounter{proofstep}\label{step:main_1}\emph{Step 1. Two events of high probability.} Fix $\epsilon>0$ and $\delta>0$.  We assume without loss of generality that $\delta<1/4$. By \cite[Theorem 1 (ii)]{miermont2006invprinc}, we have that $d_{\mathrm{gr}}(\rho_n,o_n)/n^{1/4}$ converges in distribution to a positive, non-degenerate random variable $Z$ (recall that $\rho_n$ denotes the root of $\overline{\mathcal {T}}_n^{(1)}$, and $o_n$ is a vertex picked uniformly at random from the set of vertices distinct from the root). It follows that there is an $a>0$ such that for all $n$ sufficiently large,
\[\mathbf{P}(A_1) \ge 1-\frac{\epsilon}{2} \quad\text{for}\quad A_1 = \{ d_{\mathrm{gr}}(\rho_n,o_n) \ge an^{1/4}\}.\]
By the convergence of the rescaled profile of distances to $o_n$ to a measure which assigns positive weight to any neighborhood of $0$, there is a $0<b<1$ such that for all sufficiently large $n$,
\[\mathbf{P}(A_2) \ge 1-\frac{\epsilon}{2} \quad\text{for}\quad A_2 = \left\{ \left|B_{\frac{an^{1/4}}{2}}(o_n) \right| \ge bn \right\}. \]
(here $B_r(o_n)$ denotes the ball of radius $r$ around $o_n$). See \cite[Equation (62)]{cl2019fpp} and the discussion surrounding it.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr2-1\relax}
  \refstepcounter{proofstep}\label{step:main_2}\emph{Step 2. The hull around the root is not too large.} On the event that both $A_1$ and $A_2$ occur, we have that $B_{n^{1/8}}^\bullet(\overline{\mathcal {T}}_n^{(1)})$ contains at most $(1-b)n+1$ vertices for all $n$ sufficiently large, since $B_{(an^{1/4})/2}(o_n)$ is connected and disjoint from the ball of radius $n^{1/8}$ around the root.

\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr3-1\relax}
  \refstepcounter{proofstep}\label{step:main_3}\emph{Step 3. Transfer from the UIPT to the sphere.} Write $m=n^{1/8}$. Now for any triangulation $\mathfrak{T}$ with $(1-b)n+1$ or fewer vertices, we have $N(\mathfrak{T}) \le (1-b)n$, hence $n-N(\mathfrak{T}) \ge bn$, and therefore
\[\mathbf{P}( B_{m}^\bullet(\overline{\mathcal {T}}_n^{(1)}) = \mathfrak{T}) \le \overline{c} \left( \frac{1}{b} \right)^{3/2} \mathbf{P}( B_{m}^\bullet(\mathcal {T}_\infty^{(1)}) = \mathfrak{T})\]
by \Cref{lem:uipt_to_sphere}. Let $\mathbf{G}_m$ be the set of pairs from the proof of \Cref{prop:distortion_uipt}, applied at radius $m$ and with $\delta/2$ in place of $\delta$, and let $w$ denote the vertex of the boundary cycle of the hull of radius $m$ which is selected by the uniform random rotation of its skeleton.  Conditionally on the hull, $w$ is uniform on that cycle in both models, so the pointwise comparison of hull laws extends to the pair consisting of the hull and $w$.  On the event $A_1 \cap A_2$ the hull $B_m^\bullet(\overline\mathcal {T}_n^{(1)})$ takes values among the triangulations with at most $(1-b)n+1$ vertices, by \Cref{step:main_2}, and summing the resulting bound over the pairs which do not belong to $\mathbf{G}_m$ therefore gives
\[\mathbf{P}\big(\{(B_m^\bullet(\overline\mathcal {T}_n^{(1)}),w) \notin \mathbf{G}_m\} \cap A_1 \cap A_2\big) \le \overline{c}\left(\frac{1}{b}\right)^{3/2}\mathbf{P}\big((B_m^\bullet(\mathcal {T}_\infty^{(1)}),w) \notin \mathbf{G}_m\big).\]
The right-hand side tends to $0$ as $n \to \infty$ by \Cref{prop:distortion_uipt}, so that the left-hand side is at most $\epsilon$ for all $n$ sufficiently large.


\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr4-1\relax}
  \refstepcounter{proofstep}\label{step:main_4}\emph{Step 4. From the hull to the whole triangulation.} The triangulation $\overline\mathcal {T}_n^{(1)}$ contains its hull $B_m^\bullet(\overline\mathcal {T}_n^{(1)})$ as a sub-triangulation, and the distances from the root to the vertices of that hull are the same in $\overline\mathcal {T}_n^{(1)}$ as in the hull itself.  \Cref{prop:distortion_uipt} therefore gives that on the event $\{(B_m^\bullet(\overline\mathcal {T}_n^{(1)}),w) \in \mathbf{G}_m\}$,
\[\mathrm{Dis}(\overline\mathcal {T}_n^{(1)}) \ge (\log m)^{(1/4)-\delta/2} \ge (\log n)^{(1/4)-\delta} \quad\text{(recall $m=n^{1/8}$)}\]
for $n$ large.  By \Cref{step:main_1,step:main_2,step:main_3}, the event above intersected with $A_1 \cap A_2$ has probability at least $1-2\epsilon$ for all $n$ sufficiently large.


\noindent
  \stepcounter{stepuid}
  \setcounter{proofstep}{\numexpr5-1\relax}
  \refstepcounter{proofstep}\label{step:main_5}\emph{Step 5. From the $1$-gon to the sphere.} Recall that the statement of \Cref{thm:main} is for a uniformly random rooted Type~I triangulation $\mathcal {T}_N$ of the sphere with $N$ vertices. It is thus left to transfer the distortion lower bound from $\overline\mathcal {T}_n^{(1)}$ to the case of $\mathcal {T}_N$.  Note that the marked points associated with $\overline\mathcal {T}_n^{(1)}$ do not affect the underlying graph, so that $\mathrm{Dis}(\overline\mathcal {T}_n^{(1)})=\mathrm{Dis}(\mathcal {T}_n^{(1)})$, and $\mathcal {T}_n^{(1)}$ is uniform on $\mathbf{T}_{n,1}$.  The distance-preserving bijection $\Phi$ between the set of rooted triangulations of the sphere with $n+1$ vertices and $\mathbf{T}_{n,1}$ described at the beginning of this section combined with the previous paragraph yields
\[\mathbf{P}\!\big(\mathrm{Dis}(\mathcal {T}_{n+1}) \ge (\log n)^{(1/4)-\delta}\big) \ge 1-2\epsilon\]
for all $n$ sufficiently large.  Now let $\delta' \in (0,1/4)$ be given and apply the above with $\delta=\delta'/2$ and with $N-1$ in place of $n$.  Then we obtain
\[\mathbf{P}\!\big(\mathrm{Dis}(\mathcal {T}_N) \ge (\log N)^{(1/4)-\delta'}\big) \ge 1-2\epsilon\]
for all $N$ sufficiently large.  Since $\epsilon$ was arbitrary, the proof of the theorem is complete.
