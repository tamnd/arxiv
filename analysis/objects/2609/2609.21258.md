\label{ergodic-th}
For any $t>0$, the process $\{u(t,x)\}_{x\in \mathbb{R}^d}$ is ergodic.

<!-- theorem-2: body_md -->
\label{cov-th}
For any $t_1,t_2>0$,
\begin{equation}
\label{def-K}
K(t_1,t_2):=\lim_{R \to \infty}\frac{1}{R^{\gamma}}\mathbb{E}[F_R(t_1) F_R(t_2)] \quad \mbox{is finite},
\end{equation}
where
\begin{equation}
\label{def-gamma}
\gamma:=\left\{
\begin{array}{ll} d & \mbox{in case (i)} \\
2d-\beta & \mbox{in case (ii) }.
\end{array} \right.
\end{equation}
In particular, $\sigma_R^2(t) \sim K(t,t) R^{\gamma}$ as $R \to \infty$, for any $t>0$.

<!-- theorem-3: body_md -->
\label{QCLT}

\[
d_{TV}\left( \frac{F_R(t)}{\sigma_R(t)},Z\right) \leq C_t R^{\frac{\gamma-2d}{2}},
\]

 \item[(i)] If $d = 1$ and $f \in L^1(\mathbb{R})$ or the noise is white, then

\]
\item[(ii)] If $d\in \{1,2\}$ and either (a) $f \in L^\ell(\mathbb{R}^d)$ for some $\ell >1$, or (b) $f(x) = |x|^{-\beta}$ for some $\beta \in (0,d)$, then 
\[
\int_{\mathbb{R}^d} \int_{\mathbb{R}^d} \varphi(x)\psi(y) f(x-y)K_t^{\alpha}(x-y)  dx dy \leq C_{T,\alpha,p',q} \|\varphi\|_{L^{2p'}(\mathbb{R}^d)}\|\psi\|_{L^{2p'}(\mathbb{R}^d)},
\]
for any $p'\in (q,1)$, where $q=\frac{\ell}{2\ell-1}$ in case (a), and $q=\frac{d}{2d-\beta}$ in case (b). 
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
(i) By \eqref{def-K1}, it follows that $K_t(x) \leq T$ for any $x \in \mathbb{R}$ and $t \in [0,T]$. Therefore, 
\[
 \int_{\mathbb{R}} \int_{\mathbb{R}} \varphi(x) \psi(y)  f(x-y)K_t^{\alpha}(x-y) dx dy \leq  T^{\alpha} \int_{\mathbb{R}} \int_{\mathbb{R}} \varphi(x) \psi(y) f(x-y) dx dy.
\]
The conclusion follows by applying \eqref{LHS-Lell} with $\ell=1$.  

(ii) Using \eqref{def-K1} and \eqref{def-K2}, it follows that for all $\varepsilon > 0$,
\begin{equation}
\label{bound-K}
 K_t(x) \leq C_{T,\epsilon} |x|^{-\varepsilon}.
\end{equation}

We fix $p' \in (q,1)$. We apply H\"older's inequality, followed by Young's inequality:
\begin{align*}
I &:=\int_{\mathbb{R}^d} \int_{\mathbb{R}^d}  \varphi(x) \psi(y) f(x-y)K_t^{\alpha}(x-y) dx dy 
= \int_{\mathbb{R}^d} (\varphi * fK_t^{\alpha})(y) \psi(y) dy \\
& \quad \leq \| \varphi* fK_t^n  \|_{L^a(\mathbb{R}^d)} \|\psi \|_{L^{2p'}(\mathbb{R}^d)}  \quad \text{with}\  \frac{1}{a} + \frac{1}{2p'} = 1, \\
& \quad \leq  \| fK_t^n\|_{L^{b}(\mathbb{R}^d)}  \|\varphi \|_{L^{2p'}(\mathbb{R}^d)}\|\psi \|_{L^{2p'}(\mathbb{R}^d)} \quad  \text{with } \  \frac{1}{b} +\frac{1}{2p'} = 1 + \frac{1}{a} = 2 - \frac{1}{2p'} .
\end{align*}
Note that $b=\frac{p'}{2p'-1}>1$ since $p'\in (\frac12,1)$. 

To estimate $\|K_t^{\alpha}\|_{L^{c}(\mathbb{R}^d)}$, we consider separately the two cases.
In case (a), we apply H\"older's inequality with $\frac{1}{p_1}+\frac{1}{p_2}=1$ and $p_1=\frac{\ell}{b}>1$. (This means that $\frac{1}{\ell}<\frac{1}{b}=2 - \frac{1}{p'}$, i.e. $p'>\frac{\ell}{2\ell -1}=q$.) Then
\begin{align*}
\| fK_t^{\alpha}\|_{L^{b}(\mathbb{R}^d)}  & \leq \| f\|_{L^\ell(\mathbb{R}^d)} \|K_t^{\alpha}\|_{L^{c}(\mathbb{R}^d)} \qquad \text{with } \frac{1}{l} + \frac{1}{c} = \frac{1}{b}.
\end{align*}
We use \eqref{bound-K} and the compact support of $K_t$. Then,
$\|K_t^{\alpha}\|_{L^{c}(\mathbb{R}^d)} \leq C_{T,\varepsilon}\int_{|x| < 2t} |x|^{-\alpha \varepsilon c} dx < \infty$, provided that we choose $\varepsilon > 0$ so that $\alpha \varepsilon  c< d$. This implies that
\[
I \leq C_{T,\varepsilon,\alpha,c,d} \| f\|_{L^\ell(\mathbb{R}^d)} \|\varphi\|_{L^{2p'}(\mathbb{R}^d)}\|\psi\|_{L^{2p'}(\mathbb{R}^d)}. 
\]

In case (b), using \eqref{bound-K} and the compact support of $K_t$, we have:
\[
\|fK_t^{\alpha}\|_{L^b}^b \leq C_{T,\varepsilon} \int_{\{|x|<2t\}}  |x|^{-\beta b-\alpha\varepsilon  b} dx<  \infty,
\]
provided that we choose $\varepsilon>0$ such that $(\beta+\alpha\varepsilon ) b< d$. This means that 
$$2-\frac{1}{p'}=\frac{1}{b}>\frac{\beta+\alpha \varepsilon}{d} \quad \mbox{i.e.} \quad \frac{\alpha \varepsilon}{d}< \frac{2d-\beta}{d}-\frac{1}{p'}.$$
This is possible since $p'>\frac{d}{2d-\beta} =q$. Then,
\[
I \leq C_{T,\varepsilon,\alpha,\beta,b,d} \|\varphi\|_{L^{2p'}(\mathbb{R}^2)}\|\psi\|_{L^{2p'}(\mathbb{R}^2)}. 
\]

|{\rm Cov}(F,G)| \leq \int_{\mathbb{R}_+} \int_{\mathbb{R}^{2d}} \|D_{r,y}F \|_2 \|D_{r,y'}G\|_2 f(y-y') dydy'dr.
\end{equation}

<!-- theorem-5: body_md -->
\label{key-Du}
For any $0\leq r\leq t \leq T$, $x,z \in \mathbb{R}^d$ and $p\geq 2$,
\[
\|D_{r,z}u(t,x)\|_p \leq C_{T,p} G_{t-r}(x-z),
\]
where $C_{T,p}>0$ is a constant that depends on $(T,p,q,\|b'\|_\infty,\|\sigma'\|_\infty)$.
<!-- end theorem-5 -->
<!-- proof-3: body_md -->
Using Heisenberg commutation rule \eqref{Heisenberg} followed by the chain rule \eqref{chain}, we infer that the sequence $(Du_n)_{n\geq 0}$ satisfies the following recurrence relation:
\begin{align}
\nonumber
D_{r,z}u_{n+1}(t,x)&=G_{t-r}(x-z)\sigma\big(u_n(r,z)\big)+\int_r^t \int_{\mathbb{R}^d}G_{t-s}(x-y) b'\big(u_n(s,y)\big)D_{r,z}u_n(s,y) dyds+\\
\label{rec-Dun}
&\quad \int_r^t \int_{\mathbb{R}^d} G_{t-s}(x-y)\sigma'\big(u_n(s,y)\big) D_{r,z}u_n(s,y) W(ds,dy),
\end{align}
for any $n\geq 0$. Note that $D_{r,z}u_{0}(t,x)=0$ since $u_0(t,x)=1$. 

We use Burkholder-Davis-Gundy (BDG) inequality and Minkowski inequality to estimate the $p$-th moment of the stochastic integral. Using \eqref{def-K}, it follows that for any $0<r<t\leq T$, $x,z \in \mathbb{R}^d$ and $n\geq 0$,
\begin{align}
\nonumber
&\|D_{r,z}u_{n+1}(t,x)\|_p^2  \leq B_{t} \left\{ G_{t-r}^2(x-z)+  \left(\int_r^t \int_{\mathbb{R}^d} G_{t-s}(x-y)\|D_{r,z}u_n(s,y)\|_pdyds\right)^2 \right.\\	
\nonumber
& \quad +\left.\int_r^t \int_{(\mathbb{R}^d)^2} G_{t-s}(x-y) G_{t-s}(x-y') \|D_{r,z}u_n(s,y)\|_p \|D_{r,z}u_n(s,y')\|_p  f(y-y') dydy'ds \right\} \\\
\label{recD1}
& \quad \quad \quad \quad \quad \quad  =: B_{t} \{ G_{t-r}^2(x-z)+  I+II \},
\end{align}
where $B_{t}$ is a constant that depends on $(t,p,\|b'\|_{\infty},\|\sigma'\|_{\infty})$ and is non-decreasing in $t$.

We will prove below that for any $0<r<t\leq T$, $x,z \in \mathbb{R}^d$ and $n\geq 1$,
\begin{equation}
\label{key-un}
\|D_{r,z}u_{n}(t,x)\|_p \leq C_{T,p} G_{t-r}(x-z).
\end{equation}

The conclusion will follow by Lemma A.1 of \cite{BS26}, since $\{Du_n(t,x)\}_{n\geq 1}$ converges to $Du(t,x)$ in the weak topology $L^2(\Omega;\mathcal{H})$ (by Lemma 1.2.3 of \cite{nualart06}).

\medskip

{\em Case 1.} $d=1$.
For the first integral on the RHS of \eqref{recD1}, we use H\"older's inequality with respect to the measure $\mu_t(ds,dy)=G_{t-s}(x-y)dsdy$ on $[r,t] \times \mathbb{R}$ whose total mass is $(t-r)^2/2$. For the second integral of \eqref{recD1}, we use \eqref{LHS-q}. We obtain that for any $0<r<t\leq T$, $x,z \in \mathbb{R}, n\geq 0$,
\begin{align*}
\|D_{r,z}u_{n+1}(t,x)\|_p^2  & \leq  B_{T} \left\{ G_{t-r}^2(x-z)+ \frac{T^2}{2}\int_r^t \int_{\mathbb{R}} G_{t-s}^2(x-y) \|D_{r,z}u_n(s,y)\|_p^2 dyds \right.\\
& \quad \quad \quad \left. c_{f,1}\int_r^t \left( \int_{\mathbb{R}}G_{t-s}^{2q}(x-y) \|D_{r,z}u_n(s,y)\|_p^{2q}dy\right)^{1/q}ds \right\},
\end{align*}
where $q$ is given by \eqref{def-q}. Using the fact that $G_t^{2q}(x)=c_{q} G_t^{2}(x)$ and applying H\"older's inequality with respect to the measure $G_{t-s}^2(x-y)dy$ whose total mass is $(t-s)/2$, it follows that:
\[
\left(\int_{\mathbb{R}} G_{t-s}^{2q}(x-y) \|D_{r,z}u_n(s,y)\|_p^{2q} dy\right)^{1/q} \leq c_{q}^{q} \frac{T}{2}\int_{\mathbb{R}} G_{t-s}^2(x-y)\|D_{r,z}u_n(s,y)\|_p^{2} dy.
\]
Hence
\begin{align*}
&\|D_{r,z}u_{n+1}(t,x)\|_p^2  \leq  A_T \left\{ G_{t-r}^2(x-z)+ \int_r^t \int_{\mathbb{R}} G_{t-s}^2(x-y) \|D_{r,z}u_n(s,y)\|_p^2 dyds \right\},
\end{align*}
where $A_T=B_{T} \big(1+\frac{T^2}{2}+c_{q}^{q} \frac{T}{2}\big)$.
By induction on $n\geq 1$, it follows that 
\[
\|D_{r,z}u_{n}(t,x)\|_p^2 \leq A_T \left\{ G_{t-r}^2(x-z)+ \sum_{k=1}^{n-1} A_T^k I_k(t,x,r,z)\right\},
\]
for any $0<r<t\leq T$ and $x,z \in \mathbb{R}$, where
\begin{align}
\label{def-Ik}
I_k(t,x,r,z)&=\int_{r<t_1<\ldots<t_k<t} \int_{\mathbb{R}^k}\prod_{i=1}^{k}G_{t_{i+1}-t_i}^2(x_{i+1}-x_i) G_{t_1-r}^2(x_1-z)d\pmb{x}_k d \pmb{t}_k,
\end{align}
with $\pmb{x}_k=(x_1,\ldots,x_k)$, $\pmb{t}_k=(t_1,\ldots,t_k)$, $t_{k+1}=t$ and $x_{k+1}=x$. Using the fact that $G_t^2(x)=\frac{1}{2}G_t(x)$ and inequality
\[
\int_{\mathbb{R}}G_{t-s}(x-y)G_{s-r}(y-z)dy \leq (t-r)G_{t-r}(x-z),
\]
it follows that 
\[
I_k(t,x,r,z)\leq \frac{(t-r)^{2k}}{2^k k!}G_{t-r}(x-z).
\]
Hence, for any $0<r<t\leq T$, $x,z \in \mathbb{R}$ and $n\geq 1$,
\[
\|D_{r,z}u_{n}(t,x)\|_p^2 \leq 2A_T \left(\sum_{k\geq 0} A_T^k \frac{T^{2k}}{2^k k!} \right) G_{t-r}^2(x-z)=:C_{T,p}^2 G_{t-r}^2(x-z).
\]

\medskip

{\em Case 2.} $d=2$. We proceed similarly to Lemma A.2 of \cite{BS26}. We fix $r \in [0,T]$ and $z \in \mathbb{R}^2$.
By induction on $n \geq 1$,  $D_{r,z}u_n(t,x)=0$ if $|x-z|\geq t-r$. We denote 
\[
f_n(t,x)=\|D_{r,z}u_n(t,x)\|_p \quad \mbox{and} \quad \widetilde{f}_n(t)=\sup_{|x-z|<t-r}\frac{f_n(t,x)}{G_{t-r}(x-z)}.
\]

We suppose first that $\widetilde{f}_n(t)<\infty$ for all $t \in [r,T]$, and we derive a recurrence relation for $\widetilde{f}_{n+1}$. 

We bound the terms on the right-hand side of \eqref{recD1}. We start with $I$, the term due to the drift. We have:
\begin{align*}
\int_r^t \int_{\mathbb{R}^2} G_{t-s}(x-y) f_n(s,y)dyds & \leq \int_r^t \widetilde{f}_n(s) \int_{\mathbb{R}^2} G_{t-s}(x-y) G_{s-r}(y-z) dy ds \\
& = \frac{1}{2}\int_r^t  \widetilde{f}_n(s) \int_{|t-2s+r|}^{t-r} G_{\theta}(x-z)d\theta ds,
\end{align*}
where for the second line we use the  identity \eqref{G*G}.


Considering separately the cases $\theta \in [r,\frac{t+r}{2}]$ and $\theta \in [\frac{t+r}{2},t-r]$, and using Fubini's theorem, it follows that
\[
\int_r^t \int_{\mathbb{R}^2} G_{t-s}(x-y) f_n(s,y)dyds \leq \frac{1}{2} \int_0^{t-r} \left(\int_{\frac{t+r-\theta}{2}}^{\frac{t+r+\theta}{2}} \widetilde{f}_n(s) ds \right) G_{\theta}(x-z)d\theta.
\]
Using Caucy-Schwarz inequality, we obtain that for any $\theta \in [0,t-r]$,
\begin{align*}
\int_{\frac{t+r-\theta}{2}}^{\frac{t+r+\theta}{2}} \widetilde{f}_n(s) ds  & \leq \theta^{1/2}
\left(\int_{\frac{t+r-\theta}{2}}^{\frac{t+r+\theta}{2}} \widetilde{f}_n^2(s) ds\right)^{1/2} \leq
\theta^{1/2}
\left(\int_{r}^{t} \widetilde{f}_n^2(s) ds\right)^{1/2}.
\end{align*}
Hence,
\[
\int_r^t \int_{\mathbb{R}^2} G_{t-s}(x-y) f_n(s,y)dyds \leq \frac{1}{2} \left(\int_{r}^{t} \widetilde{f}_n^2(s) ds\right)^{1/2}\int_0^{t-r} \theta^{1/2} G_{\theta}(x-z)d\theta.
\]
By direct calculation,
\begin{equation}
\label{theta-G}
\int_0^t \theta^{1/2} G_{\theta}(x) d\theta \leq 2t^{3/2} G_{t}(x) \quad \mbox{for any $t>0$}.
\end{equation}
Indeed,
\begin{align*}
\int_0^t \theta^{1/2} G_{\theta}(x) d\theta &=\frac{1}{2\pi}\int_{|x|}^t \sqrt{\frac{\theta}{\theta+|x|}} \cdot \frac{1}{\sqrt{\theta-|x|}} d\theta \leq \frac{1}{2\pi} \int_{0}^{t-|x|} \frac{1}{\sqrt{\eta}} d\eta=\frac{1}{\pi}\sqrt{t-|x|} \\
& \leq 2 \sqrt{(t-|x|)(t^2-|x|^2)} G_t(x) \leq 2t^{3/2} G_{t}(x).
\end{align*}
It follows that
\begin{equation}
\label{bound-I}
I \leq   (t-r)^{3} G_{t-r}^2(x-z) \int_{r}^{t} \widetilde{f}_n^2(s) ds.
\end{equation}

We now bound the term $II$, due to the noise. Using the definition of $\widetilde{f}_n(t)$ and inequality \eqref{LHS-q} with $q$ given by \eqref{def-q}, we have:
\begin{align*}
II & \leq \int_r^t \widetilde{f}_n^2(s) \int_{\mathbb{R}^2} \int_{\mathbb{R}^2} G_{t-s}(x-y) G_{t-s}(x-y') G_{s-r}(y-z) G_{s-r}(y'-z) f(y-y') dydy'ds \\
& \leq c_{f,2} \int_r^t \widetilde{f}_n^2(s) \big[ G_{t-s}^{2q}* G_{s-r}^{2q}(x-z)\big]^{1/q}ds.  
\end{align*}
We apply H\"older's inequality, with $\frac{1}{p}+\frac{1}{p'}=1$:
\begin{align*}
II \le c_{f,2} \left(\int_r^t \widetilde{f}_n^{2p}(s)ds \right)^{1/p} \left( \int_r^t \big[ G_{t-s}^{2q}* G_{s-r}^{2q}(x-z)\big]^{p'/q}ds \right)^{1/p'}.
\end{align*}
We now apply Lemma 4.3 of \cite{BNZ} with $\delta=p'/q$, which means that we must choose $1<p'<\frac{q}{2q-1}$. Note that this lemma remains valid for any $\delta(1-2q)+1>0$ and $q \in (\frac{1}{2},1)$. We obtain that
\begin{equation}
\label{bound-II}
II \leq C_{f,q}^2  (t-r)^{1-\delta(1-2q)} G_{t-r}^{\delta(2q-1)}(x-z) \left(\int_r^t \widetilde{f}_n^{2p}(s)ds \right)^{1/p},
\end{equation}
for some constant $C_{f,q}>1$.

We put together the bounds from \eqref{bound-I} and \eqref{bound-II}. First, by H\"older's inequality, 
\[
\int_r^t \widetilde{f}_n^2(s) ds \leq (t-r)^{1/p'} \left(\int_r^t \widetilde{f}_n^{2p}(s)ds \right)^{1/p}.
\]
Secondly, we use the fact that $G_{t-r}^{\delta(2q-1)}(x-z) \leq (t-r)^{2-\delta(2q-1)} G_{t-r}^{2}(x-z)$, due to the trivial bound:
\begin{equation}
\label{G-ab}
G_t^a(x) \leq t^{b-a} G_t^b(x) \quad \mbox{for any $t>0$ and $a<b$}.
\end{equation}
Hence,
\[
f_{n+1}^2(t,x) \le C_{f,q}^2 G_{t-r}^2(x-z) \left\{1+ [(t-r)^{3+\frac{1}{p'}}+(t-r)^{3-2\delta(2q-1)}] \left( \int_r^t \widetilde f_n^{2p}(s)ds\right)^{1/p}\right\}
\]
We divide by $G_{t-r}^2(x-z)$ and we take the supremum over $x$ with $|x-z|<t-r$. Then, we take power $p$. We obtain that:
\[
\widetilde{f}_{n+1}^{2p}(t) \le C_{f,q}^{2p} \left\{1+ [(t-r)^{3+\frac{1}{p'}}+(t-r)^{3-2\delta(2q-1)}]  \int_r^t \widetilde f_n^{2p}(s)ds\right\}.
\]

This relation is similar to (A.10) of \cite{BS26}. Proceeding as in Step 2 and 3 of the proof of Lemma A.1 of \cite{BS26}, we conclude that $\widetilde{f}_{n}(t) \leq C_{f,q} C_t$ where $C_t$ is a constant that is non-decreasing in $t$.

\label{key-D2u-int}
For any $0\leq r,\theta \leq t' \leq T$, $w \in \mathbb{R}^d$,  and $p\geq 2$,
\[
\int_{\mathbb{R}^d} \left[ \sup_{r \vee \theta \leq t \leq t'} \int_{\mathbb{R}^d} \|D_{(r,z),(\theta,w)}^2u(t,x)\|^{2p'}_p dx \right] dz \leq C_{T,p,p'},

where $p'=1$ if $d=1$ and $f \in L^1(\mathbb{R})$ or the noise is white, and  $p' \in (q,1)$ is arbitrary if $d=2$ and $f \in L^1(\mathbb{R}^2) \cap L^{\ell}(\mathbb{R}^2)$ for some $\ell>1$, or $f(x)=|x|^{-\beta}$ for some $\beta \in (0,d)$.

We first note that $D_{r,z}u(t,x) = 0$ whenever $t < r$ by \eqref{Mal-zero}. Furthermore, we use the convention $G_t(x) = 0$ whenever $t < 0$. Let $0\leq r,\theta \leq t \leq t' \leq T$. We denote $\rho = r \vee \theta$.

Taking the first and then second Malliavin derivative of the mild solution \eqref{mild soln} using \eqref{chain} and \eqref{Heisenberg}, 
\begin{align*}
D^2_{(r,z),(\theta,w)}u(t,x) & =  G_{t-r}(x-z) \sigma'\big(u(r,z)\big) D_{\theta,w} u(r,z) \\
& + G_{t-\theta}(x-w) \sigma'\big(u(\theta,w)\big) D_{r,z} u(\theta,w) \\
& + \int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y) b''\big(u(s,y)\big) D_{r,z}u(s,y) D_{\theta,w}u(s,y)  dyds \\
& + \int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y) b'\big(u(s,y)\big) D_{(r,z),(\theta,w)}^2 u(s,y) dyds \\
& + \int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y) \sigma''\big(u(s,y)\big) D_{r,z}u(s,y) D_{\theta,w}u(s,y)  W(ds,dy)\\
& + \int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y) \sigma'\big(u(s,y)\big) D_{(r,z),(\theta,w)}^2 u(s,y) W(ds,dy).
\end{align*}
Using BDG inequality and Minkowski's inequality,
\begin{align*}
\|D^2_{(r,z),(\theta,w)}u(t,x)\|_p^{2p'} & \lesssim  G^{2p'}_{t-r}(x-z) \|\sigma'\big(u(r,z)\big) D_{\theta,w} u(r,z)\|_p^{2p'} \\
& + G^{2p'}_{t-\theta}(x-w) \|\sigma'\big(u(\theta,w)\big) D_{r,z} u(\theta,w)\|_p^{2p'} \\
& + \left(\int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y) \|b''\big(u(s,y)\big) D_{r,z}u(s,y) D_{\theta,w}u(s,y)\|_p  dyds\right)^{2p'} \\
& + \left(\int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y) \|b'\big(u(s,y)\big) D_{(r,z),(\theta,w)}^2 u(s,y)\|_p dyds\right)^{2p'} \\
& + \bigg(\int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{t-s}(x-y) \|\sigma''\big(u(s,y)\big) D_{r,z}u(s,y) D_{\theta,w}u(s,y)\|_p f(y-y')  \\
& \qquad G_{t-s}(x-y') \|\sigma''\big(u(s,y')\big) D_{r,z}u(s,y') D_{\theta,w}u(s,y')\|_p dy dy' ds \bigg)^{p'}\\
& + \bigg( \int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{t-s}(x-y) \|\sigma'\big(u(s,y)\big) D_{(r,z),(\theta,w)}^2 u(s,y)\|_pf(y-y')  \\
& \qquad G_{t-s}(x-y') \|\sigma'\big(u(s,y')\big) D_{(r,z),(\theta,w)}^2 u(s,y')\|_p dy dy' ds \bigg)^{p'}.
\end{align*}
Using Theorem \ref{key-Du}, H\"older's inequality, and the boundedness of $b'$,$b''$,$\sigma'$ and $\sigma''$, 
\[
 \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'} \int_{\mathbb{R}^d} \|D^2_{(r,z),(\theta,w)}u(t,x)\|_{p}^{2p'} dx dz \lesssim \sum_{i=1}^5 \mathcal{B}_i
\]
where
\begin{align*}
 \mathcal{B}_1 & = \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'} \int_{\mathbb{R}^d} \Big(G^{2p'}_{t-r}(x-z)G^{2p'}_{r-\theta}(z-w) + G^{2p'}_{t-\theta}(x-w)G^{2p'}_{\theta-r}(w-z) \Big) dx dz, \\
 \mathcal{B}_2 & = \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'} \int_{\mathbb{R}^d} \left( \int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y)G_{s-r}(y-z)G_{s-\theta}(y-w) dy ds\right)^{2p'} dx dz, \\
 \mathcal{B}_3 & = \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \int_{\mathbb{R}^d} \left( \int_\rho^t \int_{\mathbb{R}^d} G_{t-s}(x-y) \|D^2_{(r,z),(\theta,w)}u(s,y)\|_{p}  dy ds\right)^{2p'} dx dz, \\
 \mathcal{B}_4 & = \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \int_{\mathbb{R}^d} \bigg( \int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{t-s}(x-y)G_{s-r}(y-z)G_{s-\theta}(y-w) f(y-y') \\
 & \quad G_{t-s}(x-y')G_{s-r}(y'-z)G_{s-\theta}(y'-w) dydy' ds\bigg)^{p'} dx dz, \\
 \mathcal{B}_5 & = \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \int_{\mathbb{R}^d} \bigg( \int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{t-s}(x-y) \|D^2_{(r,z),(\theta,w)}u(s,y)\|_{p} f(y-y') \\
 & \quad G_{t-s}(x-y') \|D^2_{(r,z),(\theta,w)}u(s,y')\|_{p} dy dy' ds\bigg)^{p'} dx dz. 
\end{align*}

We compute each term individually. For $\mathcal{B}_1$ we use \eqref{int G p}, noting that $2p'<2$ if $d=2$:
                                                    
\[
\mathcal{B}_1 \lesssim \int_{\mathbb{R}^d} \Big( G^{2p'}_{r-\theta}(z-w) + G^{2p'}_{\theta-r}(w-z)\Big) dz  \lesssim C_T.
\]

For $\mathcal{B}_2$ we use H\"older's inequality with respect to the finite measure $\mathbf 1_{\{\rho < s < t\}} \mathbf 1_{\{|x-y| \leq T\}}dyds$ and exponent $2p' > 1$:
\[
 \mathcal{B}_2 \lesssim \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d}  G^{2p'}_{t-s}(x-y)G^{2p'}_{s-r}(y-z)G^{2p'}_{s-\theta}(y-w) dx dy ds dz. \\
\]
Applying \eqref{int G p} with respect to $dx$, 
\[
 \mathcal{B}_2 \lesssim \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \int_\rho^t \int_{\mathbb{R}^d}G^{2p'}_{s-r}(y-z)G^{2p'}_{s-\theta}(y-w) dy ds dz \leq \int_\rho^{t'} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G^{2p'}_{s-r}(y-z)G^{2p'}_{s-\theta}(y-w) dz dy ds.
\]
 Applying \eqref{int G p} with respect to $dz$ and $dy$ then integrating in $ds$, 
\[
 \mathcal{B}_2 \lesssim \int_\rho^T \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G^{2p'}_{s-\theta}(y-w) dy ds \lesssim \int_\rho^T ds \lesssim C_T.
\]

Similarly for $\mathcal{B}_3$, we use H\"older's inequality with respect to the measure $\mathbf 1_{\{\rho \leq s \leq t\}} \mathbf 1_{\{|x-y| \leq T\}}dyds$:
\[
 \mathcal{B}_3 \lesssim \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'} \int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G^{2p'}_{t-s}(x-y) \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p}  dx dy ds dz. \\
\]
Applying \eqref{int G p} with respect to $dx$,
\[
 \mathcal{B}_3 \lesssim \int_{\mathbb{R}^d} \sup_{\rho \leq t\leq t'} \int_\rho^t \int_{\mathbb{R}^d} \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p} dy ds dz \lesssim \int_{\mathbb{R}^d} \int_\rho^{t'} \int_{\mathbb{R}^d} \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p} dy ds dz. \\
\]
Relabelling, 
\[
 \mathcal{B}_3 \lesssim \int_\rho^{t'} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} \|D^2_{(r,z),(\theta,w)}u(t,x)\|^{2p'}_{p} dx dz dt. \\
\]
Replacing the $dx$ integral (which depends on $t$) by its supremum over $s \in [\rho, t]$,
\[
 \mathcal{B}_3 \lesssim \int_\rho^{t'} \int_{\mathbb{R}^d} \sup_{\rho \leq s \leq t} \int_{\mathbb{R}^d} \|D^2_{(r,z),(\theta,w)}u(s,x)\|^{2p'}_{p} dx dz dt. 
\]
For $0 \leq t \leq t' \leq T$ we have $1 \leq T^{1-p'}(t'-t)^{p'-1}$. We will applying this trivial bound so that $\mathcal{B}_3$ matches the estimate for $\mathcal{B}_5$ below, 
\[
 \mathcal{B}_3 \lesssim \int_\rho^{t'} (t'-t)^{p'-1} \int_{\mathbb{R}^d} \sup_{\rho<s<t} \int_{\mathbb{R}^2} \|D^2_{(r,z),(\theta,w)}u(s,x)\|^{2p'}_{p} dx dz dt.
\]

For $\mathcal{B}_4$ we use Lemma \ref{Hol-lem} with respect to the measure $\mathbf 1_{\{|x-z| < T\}}dx$, if $p'< 1$. (If $p'  = 1$, the relation below is trivial. We obtain: 
\begin{align*}
 \mathcal{B}_4 & \lesssim \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \bigg(\int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{t-s}(x-y)G_{s-r}(y-z)G_{s-\theta}(y-w) f(y-y'), \\
 & \quad G_{t-s}(x-y') G_{s-r}(y'-z)G_{s-\theta}(y'-w) dx dy dy' ds \bigg)^{p'} dz.
 \end{align*}
Identifying the convolution $K_{t-s}(y-y')$, taking the supremum over $t \in [\rho,t']$, and using the fact that $t \to K_t(x)$ is non-decreasing, we have:
\begin{align*}
 \mathcal{B}_4 & \lesssim \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \bigg(\int_\rho^t \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{s-r}(y-z)G_{s-\theta}(y-w) f(y-y') K_{t-s}(y-y') \\
 & \quad G_{s-r}(y'-z)G_{s-\theta}(y'-w)  dy dy' ds \bigg)^{p'} dz \\
 & \leq \int_{\mathbb{R}^d} \bigg(\int_\rho^{t'} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{s-r}(y-z)G_{s-\theta}(y-w) f(y-y') K_{t'}(y-y') \\
 & \quad G_{s-r}(y'-z)G_{s-\theta}(y'-w)  dy dy' ds \bigg)^{p'} dz.
\end{align*}
We apply again Lemma \ref{Hol-lem} with respect to the measure $\mathbf 1_{\{|z-w| < 2T\}}dz$, if $p' < 1$. (If $p'=1$, the relation below is trivial.) We obtain:
\begin{align*}
 \mathcal{B}_4 & \lesssim \bigg(\int_\rho^{t'} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{s-r}(y-z)G_{s-\theta}(y-w) f(y-y')  K_{t'}(y-y') \\
 & \quad G_{s-r}(y'-z)G_{s-\theta}(y'-w) dz dy dy' ds \bigg)^{p'}.
\end{align*}
Identifying the convolution $K_{s-r}(y-y')$, and bounding it by $K_{t'}(y-y')$, we obtain:
\begin{align*}
 \mathcal{B}_4 & \lesssim \bigg(\int_\rho^{t'} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{s-\theta}(y-w) G_{s-\theta}(y'-w) f(y-y') K_{t'}^2(y-y')  dy dy' ds \bigg)^{p'}.
\end{align*}
By Lemma \ref{fK int},                              
\begin{align*}
 \mathcal{B}_4 & \lesssim \bigg(\int_\rho^{t'} \bigg( \int_{\mathbb{R}^d} G^{2p'}_{s-\theta}(y-w) dy \bigg)^{\frac{1}{p'}} ds\bigg)^{p'}.
\end{align*}
Using \eqref{int G p} then integrating with respect to $ds$,
\begin{align*}
 \mathcal{B}_4 & \lesssim \bigg(\int_\rho^{T} (s-\theta)^{\frac{2-2p'}{p'}} ds\bigg)^{p'} \lesssim C_{T}.
\end{align*}

For $\mathcal{B}_5$, we note that Corollary \ref{D2u-support} introduces the indicator $\mathbf 1_{\{|x-z| < T\}}$. We apply again Lemma \ref{Hol-lem} for the measure $\mathbf 1_{\{|x-z| < T\}}dx$, if $p'< 1$. (If $p'=1$, the relation below is trivial.) We obtain:
\begin{align*}
  \mathcal{B}_5 & \lesssim \int_{\mathbb{R}^d} \sup_{\rho \leq t \leq t'}  \bigg(\int_\rho^t  \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_{t-s}(x-y) \|D^2_{(r,z),(\theta,w)}u(s,y)\|_{p} f(y-y') \\
 & \quad G_{t-s}(x-y') \|D^2_{(r,z),(\theta,w)}u(s,y')\|_{p} dx dy dy' ds \bigg)^{p'} dz \\
 & \leq \int_{\mathbb{R}^d} \bigg(\int_\rho^{t'}  \int_{\mathbb{R}^d} \int_{\mathbb{R}^d}  \|D^2_{(r,z),(\theta,w)}u(s,y)\|_{p} f(y-y')K_{t'}(y-y')  \\
 & \quad \|D^2_{(r,z),(\theta,w)}u(s,y')\|_{p} dy dy' ds \bigg)^{p'} dz.
\end{align*}
Applying Lemma \ref{fK int}, we get:                              
\begin{align*}
  \mathcal{B}_5 & \lesssim \int_{\mathbb{R}^d} \bigg(\int_\rho^{t'}  \bigg(\int_{\mathbb{R}^d}  \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p} dy \bigg)^{\frac{1}{p'}} ds \bigg)^{p'} dz.
\end{align*}
The $dy$ integral depends on $s$. Taking the supremum of similar integrals for $s' \in [\rho,s]$, we obtain:
\begin{align*}
  \mathcal{B}_5 & \lesssim \int_{\mathbb{R}^d} \bigg( \int_\rho^{t'}  \bigg(\sup_{\rho \leq s'\leq s} \int_{\mathbb{R}^2}  \|D^2_{(r,z),(\theta,w)}u(s',y)\|^{2p'}_{p} dy \bigg)^{\frac{1}{p'}} ds \bigg)^{p'} dz. 
\end{align*}


We define $$M_{r,z,\theta,w}(t) = \sup_{\rho \leq s \leq t} \int_{\mathbb{R}^d}  \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p} dy.$$ 

Then $M_{r,z,\theta,w}(t)$ is non-decreasing in $t$. Therefore, by Lemma \ref{incr ineq}, we have: 
\begin{align*}
  \mathcal{B}_5 & \lesssim  \int_{\mathbb{R}^d} p'\int_\rho^{t'} (t'-s)^{p'-1} M_{r,z,\theta,w}(s) ds dz \\
  & \lesssim \int_\rho^{t'} (t'-t)^{p'-1} \int_{\mathbb{R}^d} \sup_{\rho \leq s \leq t} \int_{\mathbb{R}^d}  \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p} dy dz dt.
\end{align*}

Combining the bounds gives the recursion, 
\begin{align*}
 & \int_{\mathbb{R}^d} \sup_{\rho < t < t'} \int_{\mathbb{R}^d} \|D^2_{(r,z),(\theta,w)}u(t,x)\|_{p}^{2p'} dx dz \\
 & \quad \leq A_T + B_T\int_\rho^{t'} (t'-t)^{p'-1} \int_{\mathbb{R}^d} \sup_{\rho<s<t} \int_{\mathbb{R}^d}  \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p} dy dz dt
\end{align*}
Let 
\[
 H_{r,\theta,w}(t) = \int_{\mathbb{R}^d} \sup_{\rho<s<t} \int_{\mathbb{R}^d}  \|D^2_{(r,z),(\theta,w)}u(s,y)\|^{2p'}_{p} dy dz.  
\]
Applying the above recursion twice and integrating yields, 
\[
H_{r,\theta,w}(t') \leq A_T + B_T \int_\rho^{t'} (t'-t)^{2p'-1} H_{r,\theta,w}(t)dt \leq   A_T + B_T \int_\rho^{t'} H_{r,\theta,w}(t)dt. 
\]
Applying the classical Gronwall lemma gives the desired bound, 
\[
 H_{r,\theta,w}(t) \leq A_Te^{B_T(t-\rho)} \leq C_T.
\]

<!-- lemma-3: body_md -->
\label{ergodicity}
Let $\{Z(t,x) : t \geq 0,\ x \in \mathbb{R}^d \}$ be an adapted random field such that $\{ Z(t,x)\}_{x \in \mathbb{R}^d}$ is strictly stationary, and $Z(t,x) \in \mathbb{D}^{1,2}$ for any $t \geq 0$ and $x \in \mathbb{R}^d$. Assume that for any $0 < r < t$ and $x,z \in \mathbb{R}^d$,
\begin{equation}\label{DZ}
\| D_{r,z}Z(t,x)\|_2 \leq C_t G_{t-r}(x-z)
\end{equation}
where $C_t > 0$ is a constant that depends on $t$. Then $\{ Z(t,x)\}_{x \in \mathbb{R}^d}$ is ergodic.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
By Lemma 4.2 of \cite{BZ24}, it suffices to show that for any $k\geq 1$, and $b_1, \ldots , b_k, \zeta_1, \ldots, \zeta_k\in\mathbb{R}^d$ and $g \in \{x \to \sin(x), x \to \cos(x)\}$, the following relation holds:
\begin{align}
\label{ERG1}
\lim_{R\to\infty} \frac{1}{R^{2d}}
{\rm Var}\bigg(  \int_{[0,R]^d} g \bigg(\sum_{j=1}^k b_j  Z(t,x + \zeta_j)  \bigg) dx \bigg)
=0.
\end{align}
We denote 
\[
X_{b,\zeta}(t,x) := g \left(\sum_{j=1}^k b_j  Z(t,x + \zeta_j)  \right)
\]
and 
\[
 V_R := {\rm Var}\left(  \int_{[0,R]^d} X_{b,\zeta}(t,x) dx \right)
\]
By Poincar\'e inequality \eqref{Poincare}, \eqref{LHS-q'}, and Minkowski's inequality,
\begin{align*}
V_R & \leq \int_{\mathbb{R}_+} \int_{\mathbb{R}^{2d}} \left\| D_{r,y}\int_{[0,R]^d} X_{b,\zeta}(t,x) dx  \right\|_2 \left\| D_{r,y'}\int_{[0,R]^d} X_{b,\zeta}(t,x) dx  \right\|_2 f(y-y') dydy'dr \\
 & \lesssim \int_{\mathbb{R}_+} \left[\int_{\mathbb{R}^{d}} \left\| \int_{[0,R]^d} D_{r,y}X_{b,\zeta}(t,x) dx  \right\|_2^{2q'} dy\right]^{1/q'} dr \\
 & \lesssim \int_{\mathbb{R}_+} \left[\int_{\mathbb{R}^{d}} \left( \int_{[0,R]^d} \left\| D_{r,y}X_{b,\zeta}(t,x)  \right\|_2 dx\right) ^{2q'} dy\right]^{1/q'} dr.
\end{align*}
By the definition of $X_{b,\zeta}(t,x)$, the chain rule \eqref{chain}, and \eqref{DZ},
\[
 \left\| D_{r,y}X_{b,\zeta}(t,x)  \right\|_2 \leq C_t\sum_{j=1}^k |b_j|  G_t(x + \zeta_j - y).
\]
Therefore, by the above and H\"older's inequality, 
\begin{align*}
V_R & \lesssim  \int_{\mathbb{R}_+} \left[\int_{\mathbb{R}^{d}} \left( \int_{[0,R]^d} \sum_{j=1}^k |b_j| G_{t-r}(x + \zeta_j - y) dx\right) ^{2q'} dy\right]^{1/q'} dr \\
 & \lesssim \sum_{j=1}^k |b_j|^2 \int_{\mathbb{R}_+} \left[\int_{\mathbb{R}^{d}} \left( \int_{[0,R]^d} G_{t-r}(x + \zeta_j - y) dx\right) ^{2q'} dy\right]^{1/q'} dr.
\end{align*}
For $R \geq t$, since $|x| \leq \sqrt{d}R$, we have $|\zeta_j - y| \leq (1+\sqrt{d})R$ for each $j = 1,\cdots,k$. Therefore,
\[
V_R \lesssim \sum_{j=1}^k |b_j|^2 \int_{\mathbb{R}_+} \left[\int_{B_{(1+\sqrt{d})R}(\zeta_j)} \left( \int_{[0,R]^d} G_{t-r}(x + \zeta_j - y) dx\right) ^{2q'} dy\right]^{1/q'} dr.
\]
Bounding $[0,R]^d$ by $\mathbb{R}^d$ and integrating in the order of $dx$, $dy$ then $dr$ yields, 
\[
V_R \lesssim \sum_{j=1}^k |b_j|^2 R^{\gamma}.
\]
Since $\gamma < 2d$ this proves Lemma \ref{ergodicity} and in turn, Theorem \ref{ergodic-th}.

<!-- lemma-4: body_md -->
\label{Hol-lem}
Let $(X,\mathcal{X},\mu)$ be a finite measure space and $\phi:X \to \mathbb{R}_{+}$ be a measurable function. For any $p' \in (0,1)$,
\[
\int_{X} \big(\phi(x)\big)^{p'} \mu(dx) \leq \big[\mu(X)\big]^{1-p'} \left( \int_{X} \phi(x) \mu(dx)\right)^{p'}.
\]
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
Let $p=\frac{1}{p'}>1$ and $F(x)=\big(\phi(x)\big)^{p'}$. By H\"older's inequality,
\[
\left(\int_{X} F(x) \mu(dx)\right)^p \leq \big[\mu(X)\big]^{p-1}  \int_{X} \big(F(x) \big)^p \mu(dx).
\]
The conclusion follows taking power $1/p$.

<!-- lemma-5: body_md -->
\label{incr ineq}
Let $p' \in (\frac12,1]$ and $0 \leq r<t$. Let $M:[r,t] \to [0,\infty)$ be a non-decreasing right-continuous function. Then
\[
 \left( \int_r^t M(s)^{\frac{1}{p'}} ds \right)^{p'} \leq p' \int_r^t (t-s)^{p'-1} M(s) ds.
\]
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
The $p'= 1$ case is trivial.
We suppose that $p'\in (\frac12,1)$.  By a change of variable,
\[
\left( \int_r^t M(s)^{\frac{1}{p'}} ds \right)^{p'} =\left( \int_0^{t-r} M(t-s)^{\frac{1}{p'}} ds \right)^{p'}.                                                                     
\]
We denote $H(s)=M(t-s)$ for $s \in [0,t-r]$ and $L=t-r$. Then $H$ is non-increasing and left-continuous. It is enough to prove that
\begin{equation}
\label{H-ineq}
J:=\left( \int_0^L H(s)^{\frac{1}{p'}} ds \right)^{p'} \leq p' \int_0^L s^{p'-1} H(s) ds.
\end{equation}

By Minkowski's inequality,
\begin{align*}
J  &= \left[ \int_0^L \left( \int_0^{\infty} 1_{\{r\leq H(s)\}} dr\right)^{\frac{1}{p'}}ds \right]^{p'} = \left\| \int_0^{\infty}1_{\{r \leq H(s)\}} dr\right\|_{L^{\frac{1}{p'}}([0,L];ds)}\\
& \leq \int_0^{\infty} \|1_{\{r \leq H(s)\}} \|_{L^{\frac{1}{p'}}([0,L];ds)} dr=\int_0^{\infty} \|1_{\{s \leq H^{-1}(r)\}} \|_{L^{\frac{1}{p'}}([0,L];ds)} dr=\int_0^{\infty}\big(H^{-1}(r)\big)^{p'}dr,
\end{align*}
where $H^{-1}(y)=\sup\{x \in [0,L];H(x)\geq y\}$. On the other hand, by Fubini theorem,
\begin{align*}
p' \int_0^L s^{p'-1} H(s) ds&=p' \int_0^L s^{p'-1} \left(\int_0^{\infty} 1_{\{r \leq H(s)\}} dr \right) ds\\
& = p' \int_0^{\infty} \int_0^L s^{p'-1} 1_{\{s \leq H^{-1}(r)\}} dsdr=p' \int_0^{\infty} \left(\int_0^{H^{-1}(r)} s^{p'-1}ds \right) dr\\
&=\int_0^{\infty} \big(H^{-1}(r)\big)^{p'}dr.
\end{align*}
Relation \eqref{H-ineq} follows.
