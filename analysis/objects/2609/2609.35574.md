<!-- assumption-1: body_md -->
\label{ass: alfors}
    Assume $\rho \in \mathcal{P}_2(\mathbb{R}^d)$ such that for some $M>0$ and $\theta \in (0, 1]$,
    \begin{equation}
    \rho(B_\eta(x)) \leq M \eta^{d-1 + \theta} \quad\text{ for every } x \in \mathbb{R}^d \text{ and } \eta>0.
    \tag*{$A(M,\theta)$}
    \label{eq: assumption eqn}
    \end{equation}
<!-- end assumption-1 -->
<!-- theorem--1: body_md -->
Let $\mathcal{X}$ and $\mathcal{Y}$ be compact subsets of $\mathbb{R}^d$ for $d\geq2$, and let $\rho_0$ be a probability measure on $\mathcal{X}$ satisfying \ref{eq: assumption eqn}. Then for all $\rho_1 \in \mathcal{P}(\mathcal{X})$, all $\mu_0, \mu_1 \in \mathcal{P}(\mathcal{Y})$, and any $\gamma_{\rho_1 \to \mu_1} \in \Gamma(\rho_1, \mu_1)$,
    \begin{equation}
        W_2(\gamma_{\rho_0 \to \mu_0}, \gamma_{\rho_1 \to \mu_1}) \leq C\Big(W_1(\rho_0, \rho_1) + W_1(\mu_0, \mu_1)\Big)^{\frac{\theta}{2(1+\theta)}},
    \end{equation}
    for an explicit constant $C(\mathcal{X}, \mathcal{Y}, M)>0$ given in the statement of Theorem \ref{thrm: bi marginal plan stability}. Moreover, the bound is sharp in exponent for all such $\theta$.
<!-- end theorem--1 -->

<!-- lemma-1: body_md -->
\label{lem: diam to avg int lemma}
    Let $\phi : \mathbb{R}^d \to \mathbb{R}$ be a convex function. Then for any $x,c \in \mathbb{R}^d$ and any $\eta>0$,
    \begin{equation*}
        D_\eta \phi(x) \leq \frac{2(d+1)}{\beta_{d-1}\eta^d} \int_{B_{2\eta}(x)} \| \nabla \phi(z) - c\| \mathrm{d} z,
    \end{equation*}
    where $\beta_d := |B_1|$ is the volume of the $d$-dimensional unit ball.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Subtracting $\langle \cdot, c\rangle$ from $\phi$, it suffices to consider $c=0$. Choose $z \in \overline B_\eta(x)$ and $y \in \partial\phi(z)$. For all $z' \in B_\eta(z)$ and $y' \in \partial\phi(z')$, monotonicity implies
    \begin{equation*}
        \langle z' - z, y \rangle \leq \langle z'-z, y'\rangle \leq \eta \|y'\|.
    \end{equation*}
    Integrating in $z'$ over the half ball $A = \{ z' \in B_\eta(z) : \langle z' - z, y \rangle \geq 0 \} \subset B_{2\eta}(x)$ gives
    \begin{equation}
        \frac{\beta_{d-1}}{d+1} \eta^{d+1} \|y\| \leq \eta \int_{B_{2\eta}(x)} \| \nabla \phi(z')\| \mathrm{d} z'.
    \end{equation}
    Thus, for every $z \in \overline B_\eta(x)$ and $y \in \partial\phi(z)$,
    \begin{equation}
    \label{eq: L infinity control of convex functions}
        \|y\| \leq \frac{d+1}{\beta_{d-1} \eta^d} \int_{B_{2\eta}(x)} \| \nabla \phi(z')\| \mathrm{d} z'.
    \end{equation}
    Passing to the supremum over such $y$ gives the result.
<!-- end proof-1 -->
<!-- theorem-1: body_md -->
\label{thrm: quant reg}
Let $d \geq 2$, and let $\phi : \mathbb{R}^d \to \mathbb{R}$ be convex, with $\partial\phi(\mathbb{R}^d) \subset \mathcal{Y}$ for some compact convex $\mathcal{Y}\subset \mathbb{R}^d$. Let $R>0$, and $\tilde \rho \in \mathcal{P}(\overline B_R)$ satisfy \ref{eq: assumption eqn}. Then, for every $q \in [1,\infty)$,
\begin{equation}
\label{eq: Lq quant reg}
    \forall \eta \geq 0, \qquad \int_{\overline B_R} D_\eta \phi(x)^q\mathrm{d} \tilde \rho(x)
    \leq c_d M R^{d-1+\theta} D_\mathcal{Y}^q\left(\frac{\eta}{R}\right)^{\theta},
\end{equation}
where
\begin{equation*}
    c_d=\frac{4}{3}\frac{6^d d^2(d+1)\beta_d}{\beta_{d-1}}.
\end{equation*}
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
If $D_\mathcal{Y}=0$, then $\phi$ is affine and the result is immediate. Otherwise, fix $x \in \mathbb{R}^d$ and $\eta>0$, and set
\begin{equation*}
    c_x = \frac{1}{|B_{2\eta}(x)|}\int_{B_{2\eta}(x)} \nabla \phi(z) \mathrm{d} z.
\end{equation*}
Applying Lemma~\ref{lem: diam to avg int lemma},
\begin{equation*}
    D_\eta \phi(x) \leq \frac{2(d+1)}{\beta_{d-1}\eta^d} \|\nabla\phi - c_x\|_{L^1(B_{2\eta}(x))}.
\end{equation*}
The $BV$ Poincaré inequality \cite[Theorem 3.44]{ambrosio2000functions}, applied to each component with the diameter bound and sharp universal coefficient $1/2$ \cite{acosta2004optimal}, gives
\begin{equation*}
    \|\nabla\phi - c_x\|_{L^1(B_{2\eta}(x))} \leq 2 \eta \|D^2\phi\|_{1, 1}(B_{2\eta}(x)),
\end{equation*}
where $D^2\phi$ (and later $\Delta \phi$) is a Radon measure and
\begin{equation*}
    A \in \mathbb{R}^{d \times d}, \quad\|A\|_{1, 1} = \sum_{i=1}^d \sum_{j=1}^d |A_{ij}|
\end{equation*}
denotes the entry-wise $l^1$ norm of a matrix. For a symmetric positive semi-definite matrix $A \in \mathbb{R}^{d \times d}$, by the inequality $|A_{ij}| \leq \frac{A_{ii}+A_{jj}}{2}$, it holds that $\|D^2\phi\|_{1,1} \leq d \Delta \phi$ in the sense of measures. Thus
\begin{equation}
\label{eq: local Hessian measure bound}
    D_\eta \phi(x) \leq \frac{4(d+1)}{\beta_{d-1} \eta^{d-1}}
     \|D^2\phi\|_{1, 1}(B_{2\eta}(x)) \leq \frac{4d(d+1)}{\beta_{d-1}\eta^{d-1}} \Delta\phi(B_{2\eta}(x)).
\end{equation}
Integrating \eqref{eq: local Hessian measure bound} in $x$ and using Fubini's theorem gives
\begin{align*}
    \int_{\overline B_{R}}D_\eta \phi(x)\mathrm{d} \tilde \rho(x) &\leq \frac{4d(d+1)}{\beta_{d-1}\eta^{d-1}} \int_{\overline B_{R}} \int_{B_{2\eta}(x)} \mathrm{d} \Delta\phi(z) \mathrm{d} \tilde \rho(x)\\
    &= \frac{4d(d+1)}{\beta_{d-1}\eta^{d-1}}\int_{B_{R+2\eta}} \tilde \rho(\overline B_{R} \cap B_{2\eta}(z)) \mathrm{d} \Delta \phi(z)\\
    &\leq \frac{2^{d+2}d(d+1)}{\beta_{d-1}} M\Delta \phi(B_{R+2\eta})\eta^\theta.
\end{align*}
Let $\chi \in \mathcal{C}^\infty_c(B_1)$ be a non-negative mollifier with $\int \chi = 1$, set $\chi_\varepsilon(x) = \varepsilon^{-d} \chi(x/\varepsilon)$, and define $\phi_\varepsilon = \phi * \chi_\varepsilon$. If $\phi$ is convex and $L$-Lipschitz, integration by parts yields
\begin{equation*}
    \Delta \phi_\varepsilon(B_{R+2\eta})\leq \operatorname{Lip}(\phi_\varepsilon)\mathcal{H}^{d-1}(\partial B_{R+2\eta})
    \leq d\beta_d L(R+2\eta)^{d-1}.
\end{equation*}
Passing to the limit $\varepsilon \to 0$,
\begin{equation*}
    \Delta\phi(B_{R+ 2\eta}) \leq \liminf_{\varepsilon \to 0} \Delta \phi_\varepsilon(B_{R+2\eta}) \leq d\beta_d L(R+2\eta)^{d-1},
\end{equation*}
by the lower-semicontinuity of the measure of an open set for the convergence $\Delta \phi_\varepsilon \overset{\ast}{\rightharpoonup} \Delta \phi$. Noting that $\Delta\phi(A) = \Delta(\phi + \langle\cdot, c\rangle )(A)$ for any constant $c \in \mathbb{R}^d$, $L$ can be replaced by $D_\mathcal{Y}$. By assumption, for any $\eta>0$, $D_\eta \phi\leq D_\mathcal{Y}$, and hence
\begin{equation*}
    D_\eta \phi(x)^q\leq D_\mathcal{Y}^{q-1}D_\eta  \phi(x),
\end{equation*}
which proves that for all $\eta> 0$,
\begin{equation}
    \label{eq: local integral regularity}
    \int_{\overline B_{R}} D_\eta \phi(x)^q\mathrm{d} \tilde \rho(x)
    \leq 3^{1-d}c_d M (R+2 \eta)^{d-1} D_\mathcal{Y}^q\eta^{\theta}.
\end{equation}
If $\eta \leq R$, the stated bound is proven using $(R+ 2\eta) \leq 3R$; otherwise if $\eta \geq R$ the bound holds trivially since $1= \tilde \rho(\overline B_R) \leq M R^{d-1+ \theta}$ and $D_\eta \phi \leq D_\mathcal{Y}$ for all $\eta$. Finally, letting $\eta\downarrow0$ proves the case $\eta=0$.

\label{lem: Fenchel bound distance}
    Let $\phi:\mathbb{R}^d \to \mathbb{R}$ be convex. Then for any $(x, y) \in \mathbb{R}^d \times \mathbb{R}^d$, any $g \in \partial\phi(x)$, and any $\eta>0$, 
    \begin{equation*}
        \|y-g\| \leq D_{\eta}\phi(x) + \frac{G_\phi(x, y)}{\eta}.
    \end{equation*}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Write $(x_0,y_0)=(x,y)$. The conclusion is immediate if $G_\phi(x_0,y_0)=+\infty$. Otherwise, for any $(x_1,y_1)$ with $\phi^*(y_1)<+\infty$, Fenchel-Young inequalities give
    \begin{equation*}
        \phi(x_0) \geq \langle x_0, y_1\rangle - \phi^*(y_1), \quad \phi^*(y_0) \geq \langle x_1, y_0 \rangle - \phi(x_1),
    \end{equation*}
    so that
    \begin{align*}
        G_\phi(x_0, y_0) \geq& \langle x_0, y_1\rangle - \phi^*(y_1) +  \langle x_1, y_0 \rangle - \phi(x_1) - \langle x_0, y_0 \rangle \\
        =& - G_\phi(x_1, y_1) - \langle x_1 - x_0, y_1 - y_0 \rangle.
    \end{align*}
    In particular, when one takes $(x_1, y_1) \in \operatorname{graph}(\partial\phi)$, $G_\phi(x_1, y_1) = 0$ so that the first term on the right vanishes. Fix any $g_0 \in \partial \phi(x_0)$. If $y_0=g_0$, the result is immediate. Otherwise choose
    \begin{equation*}
        x_1 = x_0 + \eta\frac{y_0 - g_0}{\|y_0 - g_0\|},
    \end{equation*}
    and also choose any $y_1 \in \partial \phi(x_1)$. Then
    \begin{align*}
        G_\phi(x_0, y_0) \geq& \langle x_1 - x_0, y_0 - y_1 \rangle = \eta \Big\langle \frac{y_0 - g_0}{\|y_0 - g_0\|}, y_0 - g_0 + g_0- y_1 \Big\rangle\\
        \geq& \eta \left( \|y_0 - g_0\| - \|g_0 - y_1\|\right).
    \end{align*}
    Since $x_1 \in \overline B_\eta(x_0)$, rearranging,
    \begin{equation*}
        \|y_0 - g_0\| \leq D_\eta\phi(x_0) + \frac{G_\phi(x_0, y_0)}{\eta}.
    \end{equation*}

\label{lem: general coupling of plans}
    Let $q \in [1, +\infty)$, $\mathcal{X}, \mathcal{Y} \subset\mathbb{R}^d$ be compact and convex, and let $\phi : \mathbb{R}^d \to \mathbb{R}$ be a proper convex function with $\partial \phi(\mathbb{R}^d) \subset \mathcal{Y}$. Let $\tilde \rho, \rho_0, \rho_1 \in \mathcal{P}(\mathcal{X})$ and  $\mu_0, \mu_1 \in \mathcal{P}(\mathcal{Y})$, and suppose for $i=0, 1$ that $\gamma_i \in \Pi(\rho_i, \mu_i)$ satisfy $\operatorname{spt} \gamma_i \subset \operatorname{graph}(\partial\phi)$.             \int_{(\mathcal{X} \times \mathcal{Y})^2}\|y_0-y_1\|^q\mathrm{d}\pi_\beta  \leq& \int_{\mathcal{X}} D_{s+\eta} \phi(\tilde x)^q \mathrm{d} \tilde \rho(\tilde x)\\
                &+D_\mathcal{Y}^q\alpha(\|\tilde x-x_0\|>\eta) + D_\mathcal{Y}^q\beta(\|x_0-x_1\|>s).
    \end{align*}

                &+ \frac{D_\mathcal{Y}^{q-1}}{s}\int_{(\mathcal{X} \times \mathcal{Y})^2} G_\phi(x_0, y_1) \mathrm{d} \pi_\beta+D_\mathcal{Y}^q\alpha(\|\tilde x-x_0\|>\eta).
        \end{align*}

<!-- theorem-2: body_md -->
\label{thrm: quant unique general}
Let $q\in[1,\infty)$ and $r \in [1, \infty]$, and let $\mathcal{X}$ and  $\mathcal{Y}$ be compact and convex subsets of $\mathbb{R}^d$ for $d\geq2$. Let $\tilde\rho$ be a probability measure on $\mathcal{X}$ satisfying \ref{eq: assumption eqn}. Then for all $\rho\in\mathcal{P}(\mathcal{X})$ and $\mu\in\mathcal{P}(\mathcal{Y})$,
\begin{equation}
\label{eq: quant unique}
        \operatorname{diam}_{W_q}\Gamma(\rho,\mu)
        \leq C W_r(\rho,\tilde\rho)^{\nu},
\end{equation}
where $\nu \in (0, 1]$ is given by
\begin{equation*}
   \nu = \begin{cases}
            \frac{ r\theta}{q(r+\theta)} & r<\infty\\
            \frac{\theta}{q}& r= \infty, 
        \end{cases}
\end{equation*}
and $C = D_\mathcal{Y} \left(c_d M D_\mathcal{X}^{d-1+\theta}+ 1\right)^{1/q}D_\mathcal{X}^{-\nu}>0$.
<!-- end theorem-2 -->
<!-- proof-5: body_md -->
Fix two optimal plans $\gamma_0, \gamma_1 \in \Gamma(\rho, \mu)$. If $W_r(\rho, \tilde \rho)=0$ then $\rho = \tilde \rho$ and the result holds by the Brenier-McCann theorem, so assume that $W_r(\rho, \tilde \rho)>0$, with $\alpha$ the optimal coupling between them. Fix a Brenier potential $\phi$ for the transport from $\rho$ to $\mu$. We apply Lemma \ref{lem: general coupling of plans} with $\beta$ the identity coupling and $s=0$:
\begin{equation*}
    \int\|y_0-y_1\|^q\mathrm{d}\pi_\beta  \leq D_\mathcal{Y}^{q-1} \int_{\mathcal{X}} D_{\eta} \phi(\tilde x) \mathrm{d} \tilde \rho(\tilde x) +D_\mathcal{Y}^q\alpha(\|\tilde x-x_0\|>\eta).
\end{equation*}
We apply Theorem \ref{thrm: quant reg} and a Markov bound. For $\eta>0$,
\begin{equation*}
    \int\|y_0-y_1\|^q\mathrm{d}\pi_\beta \leq D_\mathcal{Y}^q \left( c_d M D_\mathcal{X}^{d-1+\theta} \left(\frac{\eta}{D_\mathcal{X}}\right)^\theta + \frac{W_r^r(\rho, \tilde \rho)}{\eta^r}\right).
\end{equation*}
We balance the terms containing $\eta$, choosing
\begin{equation*}
    \left(\frac{\eta}{D_\mathcal{X}}\right)^\theta = \frac{W_r^r(\rho, \tilde \rho)}{\eta^r}, \quad \text{ so that } \quad \eta = \left(W_r^r(\rho, \tilde \rho) D_\mathcal{X}^\theta\right)^{1/(r+\theta)}.
\end{equation*}
Consequently,
\begin{equation}
    \label{eq: uniqueness bound just before ry opt}
    \int\|y_0-y_1\|^q\mathrm{d}\pi_\beta \leq D_\mathcal{Y}^q \left(c_d M D_\mathcal{X}^{d-1+\theta}+ 1\right)D_\mathcal{X}^{-\frac{r\theta}{r+\theta}}W_r(\rho, \tilde \rho)^{\frac{r\theta}{r + \theta}}.
\end{equation}
This is the claimed bound. For $r=+\infty$, the proof is the same, choosing $\eta= W_\infty(\rho, \tilde \rho)$ without needing any Markov bound.

<!-- remark-2: body_md -->
\label{rmk: uniformity of constant in q r theta}
    The statements of the theorems in the introduction claim the existence of a constant depending on only $\mathcal{X}$, $\mathcal{Y}$ and $M$, and a priori, the above constant
    \begin{equation*}
        C = D_\mathcal{Y}\left(c_d M D_\mathcal{X}^{d-1+\theta}+ 1\right)^{1/q}D_\mathcal{X}^{-\nu}
    \end{equation*}
    does not satisfy this. Since any constant which holds for all measures on $\mathcal{X}'$ also holds for all measures on $\mathcal{X}\subset \mathcal{X}'$, with $D_{\mathcal{X}} \leq D_\mathcal{X}'$, we can freely replace instances of $D_\mathcal{X}$ by $\max(1, D_\mathcal{X})$; then $\max(1, D_\mathcal{X})^{-\nu} \leq 1$ and $\max(1, D_\mathcal{X})^{d-1+\theta} \leq \max(1, D_\mathcal{X})^d$. Furthermore, since the expression inside the $q$th root is larger than $1$, we can bound this by the same term without the $q$th root. Thus a choice of constant depending only on $\mathcal{X}, \mathcal{Y}$ and $M$ is given by
    \begin{equation*}
        C = D_\mathcal{Y}(1 + c_d M \max(1, D_\mathcal{X})^d).
    \end{equation*}
    The same principle applies to all the other theorems using similar calculations.

\label{thrm: same subgrad plan control}
    Let $q \geq1$ and $r \in [1, \infty]$, let $\mathcal{X}$ and $\mathcal{Y}$ be compact and convex subsets of $\mathbb{R}^d$ for $d\geq2$, and let $\rho_0$ be a probability measure on $\mathcal{X}$ satisfying \ref{eq: assumption eqn}. Then for any $\gamma_1 \in \mathcal{P}(\mathcal{X} \times \mathcal{Y})$ satisfying $\operatorname{spt} \gamma_1 \subset \operatorname{graph}(\partial \phi)$ with first marginal $\rho_1 \in \mathcal{P}(\mathcal{X})$,
    \begin{equation*}
        W_q(\gamma_0, \gamma_1) \leq  C W_r(\rho_0, \rho_1)^\nu,
    \end{equation*}

    \begin{equation*}
    \nu = \begin{cases}
                \frac{ r\theta}{q(r+\theta)} & r<\infty\\
                \frac{\theta}{q}& r= \infty, 
            \end{cases}
    \end{equation*}
    and $C^q= D_\mathcal{X}^{q(1-\nu)}+ D_\mathcal{Y}^q(1+ c_d M D_\mathcal{X}^{d-1+\theta}) D_\mathcal{X}^{-q\nu}$.

If $W_r(\rho_0, \rho_1)=0$, then $\rho_0=\rho_1$, and differentiability of $\phi$ $\rho_0$-almost everywhere implies $\gamma_0=\gamma_1$, so assume that $W_r(\rho_0, \rho_1)>0$.

        + D_\mathcal{Y}^q\beta(\|x_0 - x_1\|> s).
    \end{align*}
    For the last two terms, Theorem~\ref{thrm: quant reg} and Markov's inequality give, for finite $r$ and $0<s$,
    \begin{align*}
        \int_\mathcal{X} D_{s} \phi(x_0)^q\mathrm{d} \rho_0(x_0)
        + D_\mathcal{Y}^q\beta(\|x_0 - x_1\|> s)\\
        \leq D_\mathcal{Y}^q \left(c_d M D_\mathcal{X}^{d-1}s^\theta + \frac{W_r^r(\rho_0, \rho_1)}{s^r}\right).
    \end{align*}

        \left(\frac{s}{D_\mathcal{X}}\right)^\theta = \frac{W_r^r(\rho_0, \rho_1)}{s^r}, \quad \text{ so that } \quad s = \left(W_r^r(\rho_0, \rho_1) D_\mathcal{X}^\theta\right)^{1/(r+\theta)},
    \end{equation*}

        \label{eq: same subgradient diff marg before beta}
        W_q^q(\gamma_0, \gamma_1) \leq \int\|x_0 -x_1\|^q \mathrm{d} \beta + D_\mathcal{Y}^q(1+ c_d M D_\mathcal{X}^{d-1+\theta}) D_\mathcal{X}^{-q\nu} W_r(\rho_0, \rho_1)^{q\nu}.
    \end{equation}

        \int_{\mathcal{X}^2} \|x_0 - x_1\|^q \mathrm{d} \beta \leq \begin{cases}
            D_\mathcal{X}^{q-r}W_r^r(\rho_0, \rho_1) & 1 \leq r \leq q <\infty\\

        \end{cases}
    \end{equation}

    \end{equation*}
    which, after combining with \eqref{eq: same subgradient diff marg before beta}, gives the result.

        \leq& D_\mathcal{X}^{q-1}W_1(\rho_0, \rho_1) + D_\mathcal{Y}^{q-1}\int_{(\mathcal{X} \times \mathcal{Y})^2} \|y_0 - y_1\| \mathrm{d} \pi_\beta.
    \end{align*}
    Applying Lemma \ref{lem: Fenchel bound distance} with $x=x_0$, $y=y_1$ and $g=y_0$, followed by Theorem \ref{thrm: quant reg}, for $s>0$,
    \begin{align}
        \int_{(\mathcal{X} \times \mathcal{Y})^2} \|y_0 - y_1\| \mathrm{d} \pi_\beta \leq D_\mathcal{Y} c_d M R_\mathcal{X}^{d-1} s^\theta + \frac{\Delta}{s},
    \end{align}
    where
    \begin{equation*}
        \Delta = \int G_\phi(x_0, y_1) \mathrm{d} \pi_\beta.
    \end{equation*}
    If $\Delta =0$, passing $s \downarrow 0$ gives the result. Otherwise for $\Delta>0$ we balance the terms in $s$ on the unitless length scale, choosing
    \begin{equation*}
        \left(\frac{s}{R_\mathcal{X}}\right)^\theta = \frac{\Delta}{D_\mathcal{Y} s} \implies s^{\theta+1} = \frac{\Delta R_\mathcal{X}^{\theta}}{D_\mathcal{Y}},
    \end{equation*}
    so that 
    \begin{equation*}
        \frac{\Delta}{s} = R_\mathcal{X}^{-\theta/(1+\theta)}D_\mathcal{Y}^{1/(1+\theta)} \Delta^{\theta/(1+\theta)}, \qquad R_\mathcal{X}^{d-1}s^\theta = R_\mathcal{X}^{d-1 + \theta} R_\mathcal{X}^{-\theta/(1+\theta)} D_\mathcal{Y}^{-\theta/(1+\theta)} \Delta^{\theta/(1+ \theta)}.
    \end{equation*}
    Substituting gives
    \begin{equation*}
        \int_{(\mathcal{X} \times \mathcal{Y})^2} \|y_0 - y_1\|^q  \mathrm{d} \pi_\beta \leq D_\mathcal{Y}^{q}(1 + c_d M R_\mathcal{X}^{d-1+ \theta}) (D_\mathcal{Y} R_\mathcal{X})^{- \theta/(1+\theta)}\Delta^{\theta/(1+\theta)}.
    \end{equation*}
    The proof is completed by applying Lemma \ref{lem: direct control on the integral fenchel deficit by coupling} above, and absorbing the $W_1$ term into the constant via
    \begin{equation*}
        D_\mathcal{X}^{q-1}W_1(\rho_0, \rho_1) \leq D_\mathcal{X}^{q-1 +1/(1+\theta)} R_\mathcal{Y}^{-\theta/(1+\theta)} \Big(R_\mathcal{Y} W_1(\rho_0, \rho_1) \Big)^{\theta/(1+\theta)}.
    \end{equation*}

    \end{align*}

<!-- theorem-5: body_md -->
\label{thrm: coercivity rough measures}
    Let $q \geq 1$ and $r \in [1, \infty]$, and let $\mathcal{X}$ and $\mathcal{Y}$ be compact and convex subsets of $\mathbb{R}^d$ for $d \geq 2$. Fix $\tilde \rho$ a probability measure on $\mathcal{X}$ satisfying \ref{eq: assumption eqn}. Then for any $\rho \in \mathcal{P}(\mathcal{X})$ and $\mu \in \mathcal{P}(\mathcal{Y})$ and any optimal plan $\gamma_{\rho \to \mu} \in \Gamma(\rho, \mu)$,
    \begin{equation*}
        \forall \gamma \in \Pi(\rho, \mu), \quad W_q(\gamma_{\rho \to \mu}, \gamma) \leq C_0\left( \int_{{\mathcal{X} \times \mathcal{Y}}} \|x - y\|^2 \mathrm{d} \gamma - W_2^2(\rho, \mu) \right)^{\frac{\theta}{q(1+\theta)}} + C_1 W_r(\rho, \tilde \rho)^\nu,
    \end{equation*}
    where $\nu \in (0, 1]$ is given by
    \begin{equation*}
    \nu = \begin{cases}
                \frac{ r\theta}{q(r+\theta)} & r<\infty\\
                \frac{\theta}{q}& r= \infty, 
            \end{cases}
    \end{equation*}
    where $C_0^q = D_\mathcal{Y}^q(1 + c_d M D_\mathcal{X}^{d-1+ \theta}) (D_\mathcal{Y} D_\mathcal{X})^{- \theta/(1+\theta)}$ and $C_1>0$ is the constant from Theorem \ref{thrm: quant unique general}.

<!-- proof-9: body_md -->
Let $\phi$ be a Brenier potential for the transport $\rho \to \mu$. We apply Lemma \ref{lem: general coupling of plans} with $\tilde \rho = \tilde \rho$, $\rho_0 = \rho_1 = \rho$, $\alpha$ optimal for $W_r(\rho, \tilde \rho)$, and $\beta$ the identity coupling. Then
    \begin{equation*}
        W_q^q(\gamma_0, \gamma_1) \leq \int_{(\mathcal{X}\times \mathcal{Y})^2} \|y_0 - y_1\|^q \mathrm{d} \pi_\beta \leq D_\mathcal{Y}^{q-1}\int_{(\mathcal{X} \times \mathcal{Y})^2} \|y_0 - y_1\| \mathrm{d} \pi_\beta.
    \end{equation*}
    For $r<\infty$, applying Lemma \ref{lem: Fenchel bound distance} followed by Theorem \ref{thrm: quant reg}, the subadditivity of $(\cdot)^\theta$, and a Markov bound,
    \begin{align*}
        \int_{(\mathcal{X} \times \mathcal{Y})^2} \|y_0 - y_1\| \mathrm{d} \pi_\beta \leq& \int_\mathcal{X} D_{\eta+ s} \phi(\tilde x) \mathrm{d} \tilde \rho(\tilde x) + \frac{\Delta}{s} + D_\mathcal{Y}\alpha(\|\tilde x - x\| > \eta)\\
        \leq& D_\mathcal{Y} c_d M D_\mathcal{X}^{d-1} ( s^\theta + \eta^\theta)+ \frac{\Delta}{s} + D_\mathcal{Y}\frac{W_r^r(\rho, \tilde \rho)}{\eta^r}
    \end{align*}
    where as before
    \begin{equation*}
        \Delta = \int G_\phi(x_0, y_1) \mathrm{d} \pi_\beta.
    \end{equation*}
    When $\Delta = 0$ or $W_r(\tilde \rho, \rho)=0$ we collapse into cases covered by the previous theorems. Assuming both are positive, we choose $s$ as in the previous theorem, and $\eta$ as in the quantitative uniqueness theorem, which gives
    \begin{equation*}
        W_q^q(\gamma_0, \gamma_1) \leq C_0^q \Delta^{\theta/(1+\theta)} + C_1^q W_r(\rho, \tilde \rho)^{q\nu}
    \end{equation*}
    for the constants given in the statement of the theorem. We take the $q$th root then use subadditivity, before applying Lemma \ref{lem: direct control on the integral fenchel deficit by coupling} to complete the proof when $r<\infty$. Here we recover the quadratic cost gap directly since the plans have the same marginals, so that
    \begin{equation*}
        \int \langle x, y \rangle \mathrm{d} (\gamma_0 - \gamma_1) = \frac{1}{2}\int\|x-y\|^2 \mathrm{d} (\gamma_1 - \gamma_0).
    \end{equation*}
    When $r=\infty$, the same bounds hold directly, taking the same choice of $s$ and instead $\eta = W_\infty(\rho, \tilde \rho)$ without needing any Markov bound.

<!-- theorem-6: body_md -->
\label{thrm: map coercivity wrt dual suboptimality}
Let $q \in [1, +\infty)$, and let $\mathcal{X}$ and $\mathcal{Y}$ be compact and convex subsets of $\mathbb{R}^d$ for $d \geq2$. Let $\rho$ be a probability measure on $\mathcal{X}$ satisfying \ref{eq: assumption eqn}, and $\mu \in \mathcal{P}(\mathcal{Y})$. Then
\begin{equation*}
    \forall \psi \in \mathcal{C}^0(\mathcal{Y}),\quad\| \nabla \psi^* - T_{\rho \to \mu}\|_{L^q(\rho)}^q \leq C\left( \mathcal{K}_{\rho, \mu}(\psi) - \min_{\mathcal{C}^0(\mathcal{Y})} \mathcal{K}_{\rho, \mu} \right)^{\frac{\theta}{1+\theta}},
\end{equation*}
where $C= D_\mathcal{Y}^{q}(1 + c_d M D_\mathcal{X}^{d-1+ \theta}) (D_\mathcal{Y} D_\mathcal{X})^{- \theta/(1+\theta)}$.
<!-- end theorem-6 -->
<!-- proof-10: body_md -->
Denote $T = T_{\rho \to \mu}$. Since $\partial\psi^*(\mathcal{X}) \subset \mathcal{Y}$, we apply Lemma \ref{lem: general coupling of plans} with $\rho_0 = \rho_1 = \tilde \rho = \rho$, $\gamma_0 = (\operatorname{Id}, \nabla\psi^*)_\#\rho$, $\gamma_1 = (\operatorname{Id}, T)_\#\rho$, $\phi = \psi^*$, $\alpha, \beta$ the identity couplings, and $\eta= 0$. For this choice of coupling,
    \begin{equation*}
        \int_{(\mathcal{X} \times \mathcal{Y})^2} \|y_0 - y_1\|^q \mathrm{d} \pi_\beta = \|\nabla\psi^* - T\|_{L^q(\rho)}^q.
    \end{equation*}
    The lemma implies that for all $s>0$
    \begin{equation*}
        \|\nabla\psi^* - T\|_{L^q(\rho)}^q \leq D_\mathcal{Y}^{q-1}\left(\int_\mathcal{X} D_s \psi^*(x) \mathrm{d} \rho(x) + \frac{\Delta}{s} \right),
    \end{equation*}
    where
    \begin{equation*}
        \Delta = \int_\mathcal{X} G_{\psi^*}(x, T(x)) \mathrm{d} \rho(x).
    \end{equation*}
    Applying Theorem~\ref{thrm: quant reg}, for all $s>0$,
 \begin{equation*}
        \|\nabla\psi^* - T\|_{L^q(\rho)}^q \leq D_\mathcal{Y}^{q-1}\left(D_\mathcal{Y} c_d M D_\mathcal{X}^{d-1} s^\theta + \frac{\Delta}{s} \right).
    \end{equation*}
    If $\Delta = 0$, letting $s\downarrow 0$ completes the proof. Otherwise, balancing the terms in $s$ as in previous theorems, we choose
    \begin{equation*}
        \left(\frac{s}{D_\mathcal{X}}\right)^\theta = \frac{\Delta}{D_\mathcal{Y} s} \implies s = \left(\frac{\Delta D_\mathcal{X}^\theta}{D_\mathcal{Y}}\right)^{1/(1+\theta)}.
    \end{equation*}
    Substituting this value gives
    \begin{equation*}
        \|\nabla\psi^* - T\|_{L^q(\rho)}^q \leq D_\mathcal{Y}^{q}(1 + c_d M D_\mathcal{X}^{d-1+ \theta}) (D_\mathcal{Y} D_\mathcal{X})^{- \theta/(1+\theta)}\Delta^{\theta/(1+\theta)}.
    \end{equation*}
    The proof is completed by observing that $\Delta$ is a lower bound for the dual suboptimality gap. Since $\psi^{**} \leq \psi$,
    \begin{align*}
        \int_{\mathcal{X}} G_{\psi^*}(x, T(x)) \mathrm{d} \rho(x) \leq& \int_\mathcal{X} \psi^*(x) + \psi(T(x)) - \langle x, T(x)\rangle \mathrm{d} \rho(x)\\
        =& \mathcal{K}_{\rho, \mu}(\psi)  - \int_\mathcal{X}\langle x, T(x)\rangle \mathrm{d} \rho(x)\\
        =& \mathcal{K}_{\rho, \mu}(\psi) - \min_{\mathcal{C}^0(\mathcal{Y})} \mathcal{K}_{\rho, \mu},
    \end{align*}
    by optimality of $T$ and strong duality.

<!-- theorem-7: body_md -->
\label{thrm: bi marginal map stability}
Let $q \in [1, +\infty)$, let $\mathcal{X}$ and $\mathcal{Y}$ be compact subsets of $\mathbb{R}^d$, and let $\rho_0 \in \mathcal{P}(\mathcal{X})$ satisfy \ref{eq: assumption eqn}. Then for all $\rho_1 \in \mathcal{P}(\mathcal{X})$ and $\mu_0, \mu_1 \in \mathcal{P}(\mathcal{Y})$, and Brenier potentials $\phi_{\rho_i \to \mu_i}$
\begin{equation*}
    \|\nabla \phi_{\rho_0 \to \mu_0} - \nabla \phi_{\rho_1 \to \mu_1}\|_{L^q(\rho_0)}^q \leq C \Big(D_\mathcal{Y} W_1(\rho_0, \rho_1) +D_\mathcal{X} W_1(\mu_0, \mu_1)\Big)^{\frac{\theta}{1+\theta}}
\end{equation*}
where $C$ is the constant in Theorem~\ref{thrm: map coercivity wrt dual suboptimality}.
<!-- end theorem-7 -->
<!-- proof-11: body_md -->
Let $(\phi_0, \psi_0)$ and $(\phi_1, \psi_1)$ be pairs of optimal potentials for the transport between $\rho_i$ and $\mu_i$ respectively. We apply the dual coercivity Theorem~\ref{thrm: map coercivity wrt dual suboptimality} to the $\rho_0, \mu_0$ problem, using that $(\phi_1, \psi_1)$ is a competitor to the dual $\mathcal{K}_{\rho_0, \mu_0}$. Thus
\begin{equation}
    \label{eq: bi marignal map subopt gap}
    \| \nabla \phi_{\rho_0 \to \mu_0} - \nabla \phi_{\rho_1 \to \mu_1}\|_{L^q(\rho_0)}^q \leq C \left( \int \phi_1 \mathrm{d} \rho_0 + \int \psi_1 \mathrm{d} \mu_0 - \int \phi_0 \mathrm{d} \rho_0  -\int \psi_0 \mathrm{d} \mu_0 \right)^{\frac{\theta}{1+\theta}}.
\end{equation}
Adding the second dual suboptimality gap
\begin{equation*}
    \int \phi_0 \mathrm{d} \rho_1  + \int \psi_0 \mathrm{d} \mu_1 - \int \phi_1 \mathrm{d} \rho_1  - \int \psi_1 \mathrm{d} \mu_1  \geq 0
\end{equation*}
to the right hand side of \eqref{eq: bi marignal map subopt gap} inside the bracket,
\begin{equation*}
    \| \nabla \phi_{\rho_0 \to \mu_0} - \nabla \phi_{\rho_1 \to \mu_1}\|_{L^q(\rho_0)}^q \leq C \left( \int (\phi_1 - \phi_0) \mathrm{d}(\rho_0-\rho_1) + \int (\psi_1 - \psi_0) \mathrm{d}(\mu_0 - \mu_1) \right)^{\frac{\theta}{1+\theta}}.
\end{equation*}
Since $\nabla \phi_i \subset \mathcal{Y}$ and $\nabla \psi_i \subset \mathcal{X}$ for $i=0, 1$, then the functions $\phi_1 - \phi_0$ and $\psi_1 - \psi_0$ are $D_\mathcal{Y}$ and $D_\mathcal{X}$ Lipschitz respectively, so the result follows by Kantorovich-Rubinstein duality.
<!-- end proof-11 -->
<!-- remark-4: body_md -->
\label{rmk: statistical estimation of transport maps}
The theorem has the following implications for approximation of transport maps. It says that if one can solve the dual problem for $\rho_1, \mu_1$ to get some Brenier potential $\phi_{\rho_1 \to \mu_1}$ defined on all $\mathcal{X}$, then $\nabla \phi_{\rho_1 \to \mu_1}$ is a good approximator in $L^q(\rho_0)$ for the transport map $T_{\rho_0 \to \mu_0} = \nabla \phi_{\rho_0 \to \mu_0}$. This means that a map constructed out of the gradient of the Brenier potential $\phi_{\rho_1 \to \mu_1}$ will, on average, send samples from $\rho_0$ to a point near the image of the true optimal map $T_{\rho_0 \to \mu_0}$.

<!-- theorem-8: body_md -->
\label{thrm: bi marginal plan stability}
    Let $q \in [1, +\infty)$, and let $\mathcal{X}$ and $\mathcal{Y}$ be compact convex subsets of $\mathbb{R}^d$ for $d \geq 2$. Let $\rho_0$ be a probability measure on $\mathcal{X}$ satisfying \ref{eq: assumption eqn}, let $\mu_0 \in \mathcal{P}(\mathcal{Y})$, and let $\gamma_{\rho_0 \to \mu_0} \in \Gamma(\rho_0, \mu_0)$ be the unique optimal plan between them. Then for all $\rho_1 \in \mathcal{P}(\mathcal{X})$, all $\mu_1 \in \mathcal{P}(\mathcal{Y})$ and any optimal plan $\gamma_{\rho_1 \to \mu_1} \in \Gamma(\rho_1, \mu_1)$,
    \begin{equation*}
        W_q(\gamma_{\rho_0 \to \mu_0}, \gamma_{\rho_1 \to \mu_1}) \leq C\Big( W_1(\rho_0, \rho_1)+  W_1(\mu_0, \mu_1)\Big)^{\frac{\theta}{q(1+\theta)}},
    \end{equation*}
    where $C = 2\max(D_\mathcal{X}, D_\mathcal{Y})^{\theta/(q(1+\theta))}C'$ for $C'$ the constant in Theorem \ref{thrm: coercivity general marginals}, with $R_\mathcal{X}$ and $R_\mathcal{Y}$ replaced by $D_\mathcal{X}$ and $D_\mathcal{Y}$.
<!-- end theorem-8 -->
<!-- proof-12: body_md -->
Since all the quantities in the theorem are invariant under translation of source and target domains, we can assume without loss of generality that $0 \in \mathcal{X} \cap \mathcal{Y}$. We apply the arbitrary marginal coercivity theorem, Theorem \ref{thrm: coercivity general marginals}, with $\gamma_i = \gamma_{\rho_i \to \mu_i}$, so that
    \begin{equation}
        \label{eq: stability bound pre coercivity}
        W_q(\gamma_0, \gamma_1)\leq C' \left( \left|\int_{\mathcal{X} \times \mathcal{Y}} \langle x, y \rangle \mathrm{d}(\gamma_0 - \gamma_1)\right| + D_\mathcal{Y} W_1(\rho_0, \rho_1) + D_\mathcal{X} W_1(\mu_0, \mu_1) \right)^{\frac{\theta}{q(1+\theta)}},
    \end{equation}
    and we replaced the instances of $R_\mathcal{X}$ and  $R_\mathcal{Y}$ by $D_\mathcal{X}$ and  $D_\mathcal{Y}$, since $0 \in \mathcal{X} \cap \mathcal{Y}$. Let $(\phi_0, \psi_0)$ and $(\phi_1, \psi_1)$ be pairs of optimal maximum correlation potentials for the respective dual problems from $\rho_i$ to $\mu_i$. Write $\gamma_i=\gamma_{\rho_i\to\mu_i}$. The compatibility condition for each of the two transport problems implies
    \begin{align*}
        \int \langle x,y\rangle\mathrm{d}(\gamma_0-\gamma_1)
        =& \int\phi_0\mathrm{d}\rho_0+\int\psi_0\mathrm{d}\mu_0-\int\phi_1\mathrm{d}\rho_1-\int\psi_1\mathrm{d}\mu_1\\
        \leq& \int\phi_1\mathrm{d}(\rho_0-\rho_1)+\int\psi_1\mathrm{d}(\mu_0-\mu_1)\\
        \leq& D_\mathcal{Y} W_1(\rho_0,\rho_1)+D_\mathcal{X} W_1(\mu_0,\mu_1),
    \end{align*}
    where we used the suboptimality of $(\phi_1,\psi_1)$ in the $\rho_0$ to $\mu_0$ dual problem, followed by Kantorovich-Rubinstein duality. Interchanging $\gamma_0$ and $\gamma_1$ and applying a symmetric bound controls the absolute value $\left|\int \langle x, y \rangle \mathrm{d} (\gamma_0 - \gamma_1)\right|$. Thus the gap
    \begin{equation*}
        \Delta = \left|\int_{\mathcal{X} \times \mathcal{Y}} \langle x, y \rangle \mathrm{d} (\gamma_0 - \gamma_1) \right| + D_\mathcal{Y} W_1(\rho_0, \rho_1) + D_\mathcal{X} W_1(\mu_0, \mu_1)
    \end{equation*}
    in Theorem~\ref{thrm: coercivity general marginals} satisfies
    \begin{equation*}
        \Delta\leq 2D_\mathcal{Y} W_1(\rho_0,\rho_1)+2D_\mathcal{X} W_1(\mu_0,\mu_1) \leq 2\max(D_\mathcal{X}, D_\mathcal{Y})(W_1(\rho_0, \rho_1) + W_1(\mu_0, \mu_1)).
    \end{equation*}
    Substituting this bound into \eqref{eq: stability bound pre coercivity} completes the proof.

\label{prop: sharp exponent stability}
    Let $d\geq 2$ and $\theta \in (0, 1]$. Let $\mathcal{X}$ and  $\mathcal{Y}$ be compact convex subsets of $\mathbb{R}^d$ with nonempty interior. There exists a probability measure $\rho$ on $\mathcal{X}$ satisfying \ref{eq: assumption eqn} for some $M>0$, and measures $\mu_0, (\mu_i)_{i\geq 1} \in \mathcal{P}(\mathcal{Y})$ such that for any $q \geq 1$,  $r \in [1, \infty]$ and
    \begin{equation*}
        \alpha > \begin{cases}
        \frac{r\theta}{q(r+\theta)} & r< \infty,\\
        \frac{\theta}{q} & r = \infty,
        \end{cases}
    \end{equation*}
    denoting by $\gamma_i \in \Gamma(\rho, \mu_i)$ the unique optimal transport plan,
    \begin{equation}
        \label{eq: sup the stability fails}
        \limsup_{i \to \infty} \frac{W_q(\gamma_{\rho \to \mu_0}, \gamma_{\rho \to \mu_i})}{W_r(\mu_0, \mu_i)^\alpha} = \infty. 
    \end{equation}
    Consequently the exponents in Theorems~\ref{thrm: bi marginal map stability} and \ref{thrm: bi marginal plan stability} are sharp.

It suffices to construct one such compactly supported example in $\mathbb{R}^d$; then since the quadratic optimal transport problem is invariant with respect to dilations and translations of source and target, and $\mathcal{X}, \mathcal{Y}$ have non-empty interior, the same example can be dilated and translated inside these sets.  We use a modified form of the example given in \cite[Theorem 2]{letrouit2026unstable}. We construct $\rho$ out of multiple copies of cells of the form
    \begin{equation*}
        Q(l, h) := \left\{ z \in \mathbb{R}^d: 0 < z_1 < l, |z_2| <h/2, |z_k| < 1/2\; (3 \leq k \leq d)\right\}
    \end{equation*}
     for some $l, h>0$. (If $d=2$ we omit the last condition.) For a centre $c_i \in \mathbb{R}$ and offset $a_i>0$, a horizontal length $l_i>0$ and vertical thickness $h_i>0$, define
     \begin{align*}
        R_i^+ =& (c_i + a_i, 0,...,0) + Q(l_i, h_i),\\
        R_i^- =& (c_i - a_i, 0,...,0) - Q(l_i, h_i)
     \end{align*}
     in the sense of Minkowski. We will choose $a_i$ and $c_i$ at scales such that all cells are separated. Now for a collection of positive lengths $(l_i, h_i, a_i)_{i \in \mathbb{N}}$ and centres $(c_i)_{i \in \mathbb{N}}$ such that $\sum_{i \geq 1} l_i h_i^\theta < +\infty$ to be fixed later, we denote
     \begin{equation*}
        U := \bigcup_{i \geq 1} (R_i^+ \cup R_i^-), \quad \rho(x) := Z_\theta^{-1}|x_2|^{\theta - 1} \chi_U(x),
     \end{equation*}
     where 
     \begin{equation*}
        Z_\theta = \frac{2^{2-\theta}}{\theta} \sum_{i \geq 1} l_i h_i^\theta
     \end{equation*}
     is a normalising constant so that $\rho$ is a probability density. For any ball $B_\eta(x)$,
     \begin{align*}
        \rho(B_\eta(x)) \leq & \int_{x_2 - \eta}^{x_2 + \eta} \int_{B_\eta(x_{-2})} \rho(x) \mathrm{d} x_1 \mathrm{d} x_3 ... \mathrm{d} x_d \mathrm{d} x_2 \\
        \leq & \eta^{d-1}\beta_{d-1} Z_\theta^{-1} \int_{x_2 - \eta}^{x_2 +\eta} |x_2|^{\theta-1} \mathrm{d} x_2 \leq \frac{2}{\theta Z_\theta} \beta_{d-1} \eta^{d-1 + \theta},
     \end{align*}
     where we used that $\int_{x_2 - \eta}^{x_2 +\eta} |x_2|^{\theta-1} \mathrm{d} x_2$ is maximised at $x_2 = 0$. Hence $\rho$ satisfies \ref{eq: assumption eqn} with $M = 2\theta^{-1} Z_\theta^{-1} \beta_{d-1}$. Define
     \begin{equation*}
        \sigma_i := \rho(R_i^+) = \rho(R_i^-) = c_{\rho, \theta} l_i h_i^\theta, \quad \text{ where } c_{\rho, \theta} = \frac{2^{1-\theta}}{\theta Z_\theta}.
     \end{equation*}
     We introduce target points for the mass in the $i$th cells:
     \begin{equation*}
        y_i^+ = (c_i, a_i, 0,...,0), \quad\quad y_i^- = (c_i, -a_i, 0,...,0),
     \end{equation*}
     and their perturbations by $h_i$ in the $x_1$ coordinate:
     \begin{equation*}
        \hat y_i^+ = (c_i +h_i, a_i, 0,...,0), \quad\quad \hat y_i^- = (c_i -h_i, -a_i, 0,...,0).
     \end{equation*}
     We claim that every point $x^+ \in R_i^+$ is closer to $\hat y_i^+$ than $\hat y_i^-$, and similarly every $x^- \in R_i^-$ is closer to $\hat y_i^-$ than $\hat y_i^+$. Indeed, write $x^+ = (c_i + a_i, 0,...,0) + z_i^+$ for some $z_i^+ \in Q(l_i, h_i)$. The worst case is given by $z_i^+ = (0, -h_i/2,...)$ where the remaining coordinates contribute the common term $\sum_{k=3}^d|x_k^+|^2$ to both squared distances. In this limiting case,
     \begin{align*}
        \|x^+ - \hat y_i^+\|^2 =& (h_i- a_i)^2 + (a_i + h_i/2)^2 + \sum_{k=3}^d|x_k^+|^2,\\
        \|x^+ - \hat y_i^-\|^2 =& (h_i + a_i)^2 + (a_i - h_i/2)^2 + \sum_{k=3}^d|x_k^+|^2,
     \end{align*}
     so that
     \begin{equation*}
        \|x^+ - \hat y_i^-\|^2 - \|x^+ - \hat y_i^+\|^2 = 2a_i h_i >0. 
     \end{equation*}
     We define a reference target measure, as well as the $i$th perturbed target by 
     \begin{align*}
        \mu_0 =& \sum_{j \geq 1} \sigma_j (\delta_{y_j^+} + \delta_{y_j^-})\\
        \mu_i =& \mu_0 + \sigma_i(\delta_{\hat y_i^+} + \delta_{\hat y_i^-} -\delta_{y_i^+} - \delta_{y_i^-}).
     \end{align*}
     The perturbation $\mu_0 \to \mu_i$ simply moves the centred masses near the cells $R_i^+, R_i^-$ to their perturbed positions. We impose that
     \begin{align}
        \forall i \geq 2, \quad \min(c_i - c_{i-1}, c_{i+1} - c_i) \geq& 100 \max(l_i, h_i, a_i),\label{eq: constraint 1 on sequences}\\
         \forall i \geq 1, \quad a_i \geq& 100 h_i. \label{eq: constraint 2 on sequences}
     \end{align}
     The first ensures that clusters with different indices are well separated: the first two coordinate projections of the cells and targets indexed by $i$ lie much closer to each other than to those of any other cluster. The second guarantees that
     \begin{equation*}
        \|y_i^+ - \hat y_i^+\| \leq \|y_i^+ - \hat y_i^-\| \quad \text{ and } \quad \|y_i^- - \hat y_i^-\| \leq \|y_i^- - \hat y_i^+\|.
     \end{equation*}
     Consequently, by coupling  mass via $y_i^+ \leftrightarrow \hat y_i^+$ and $y_i^- \leftrightarrow \hat y_i^-$,
     \begin{equation*}
        W_r(\mu_0, \mu_i) \leq 2^{1/r} h_i \sigma_i^{1/r} \quad (r<+\infty), \qquad W_\infty(\mu_0,\mu_i)\leq h_i.
     \end{equation*}
     Since $\rho$ is absolutely continuous, the optimal plans from $\rho$ to $\mu_i$ are induced by maps. The optimal map from $\rho$ to $\mu_0$ is
     \begin{equation*}
        T_0(x) = \begin{cases}
            y_j^+ & x \in R_j^+ \cup R_j^- \text{ and } x_2 \geq 0,\\
            y_j^- & x \in R_j^+ \cup R_j^- \text{ and } x_2 < 0,
        \end{cases}
     \end{equation*}
     for each $j\geq1$, since every point is sent to the nearest point in $\operatorname{spt} \mu_0$. The optimal map from $\rho$ to $\mu_i$ is
     \begin{equation*}
        T_i(x) = \begin{cases}
            T_0(x) & x \notin R_i^+ \cup R_i^-,\\
            \hat y_i^+ & x \in R_i^+,\\
            \hat y_i^- & x \in R_i^-,\\
        \end{cases}
     \end{equation*}
     as again every point is sent to the nearest point in $\operatorname{spt} \mu_i$ by the above calculations.

    We now prove a lower bound on the $W_q$ distance between the transport plans $\gamma_i = (\operatorname{Id}, T_i)_\#\rho$. Unlike in the sharp quantitative uniqueness example, the supports are not cleanly separated. Set
    \begin{equation*}
        G_i = (R_i^+ \times \{ \hat y_i^+\}) \cup (R_i^- \times \{ \hat y_i^-\}), \quad \gamma_i(G_i) = 2 \sigma_i,
    \end{equation*}
    and
    \begin{align*}
        H_i = \big((R_i^+ \cap \{ x_2 \geq 0\}) \times \{ y_i^+\}\big) \cup& \big((R_i^- \cap \{ x_2 \leq 0\}) \times \{y_i^-\}\big), \\
        \gamma_0(H_i) =& \sigma_i.
    \end{align*}
    Consider any coupling $\pi \in \Pi(\gamma_0, \gamma_i)$, with variables $(x_0,y_0,x_i,y_i)$. For $\pi$-almost every such point with $(x_i,y_i)\in G_i$, if
    \begin{equation*}
        \max(\|x_0 - x_i\|, \| y_0 - y_i\|) < 2 a_i,
    \end{equation*}
    we must have $(x_0, y_0) \in H_i$ by the separation of cells and $a_i \geq 100 h_i$. It follows that
    \begin{equation*}
        \pi( \max(\|x_0 - x_i\|, \|y_0 - y_i\|) \geq 2a_i) \geq \sigma_i
    \end{equation*}
    for any $\pi \in \Pi(\gamma_0, \gamma_i)$ -- this is effectively a measure theoretic pigeonhole principle, for couplings rather than bijections. It follows that
    \begin{equation*}
        W_q(\gamma_0, \gamma_i) \geq 2 a_i \sigma_i^{1/q}.
    \end{equation*}
    We now fix $(l_i, h_i, c_i, a_i)_{i \in \mathbb{N}}$, taking
    \begin{equation*}
        a_i = l_i = (i+i_0)^{-2}, \quad h_i = 2^{-(i+i_0)}, \quad c_1=0,\quad c_{i+1} - c_i = 100(a_i + a_{i+1}).
    \end{equation*}
    We fix $i_0$ large enough that both conditions \eqref{eq: constraint 1 on sequences} and \eqref{eq: constraint 2 on sequences} hold for all $i\geq1$. All cells, sums and the normalising constant $Z_\theta$ use these sequences. Since
    \begin{equation*}
        \sum_{i \geq 1} a_i < \infty \quad \text{ and } \quad \sum_{i \geq1} l_i h_i^\theta < \infty,
    \end{equation*}
    $\rho$ is a compactly supported probability measure, with $\operatorname{spt} \rho = \overline U$. Assume that for some constant $C$ independent of $i$, and some $\alpha >0$,
    \begin{equation*}
        W_q(\gamma_0, \gamma_i) \leq C W_r(\mu_0, \mu_i)^\alpha. 
    \end{equation*}
    Then, up to constants independent of $i$,
    \begin{equation*}
        W_r(\mu_0, \mu_i)^\alpha \leq 2 c_{\rho, \theta, r} \big(l_i^{1/r}h_i^{(r+\theta)/r}\big)^\alpha 
    \end{equation*}
    and
    \begin{equation*}
        W_q(\gamma_0, \gamma_i) \geq 2 c_{\rho, \theta, q} l_i^{(q+1)/q} h_i^{\theta/q}
    \end{equation*}
    so that for all $i$,
    \begin{equation*}
        l_i^{\left(\frac{q+1}{q} - \frac{\alpha}{r}\right)} h_i^{\left(\frac{\theta}{q} - \frac{\alpha(r+\theta)}{r}\right)} \leq C_{\rho, \theta, r, q}.
    \end{equation*}
    Since $h_i$ decays exponentially fast compared to the polynomial $l_i$, the only way this is possible is if the exponent is nonnegative, i.e.
    \begin{equation*}
        \frac{\theta}{q} - \frac{\alpha(r+\theta)}{r} \geq 0 \implies \alpha \leq \frac{r\theta}{q(r+\theta)}.
    \end{equation*}
    When $r=\infty$, the bound $W_\infty(\mu_0,\mu_i)\leq h_i$ gives
    \begin{equation*}
        \frac{W_q(\gamma_0,\gamma_i)}{W_\infty(\mu_0,\mu_i)^\alpha}
        \geq 2c_{\rho,\theta}^{1/q}a_i l_i^{1/q}h_i^{\theta/q-\alpha}\longrightarrow\infty
        \quad\text{if }\alpha>\theta/q.
    \end{equation*}
    Consequently, conclusions of Theorems \ref{thrm: bi marginal map stability} and \ref{thrm: bi marginal plan stability} cannot hold with any exponent larger than the one proven.

\label{prop: sharp exponent coercivity}
    Let $d\geq 2$ and $\theta \in (0, 1]$. Let $\mathcal{X}$ and $\mathcal{Y}$ be compact convex sets in $\mathbb{R}^d$ with nonempty interior. There exists a probability measure $\rho$ on $\mathcal{X}$ satisfying \ref{eq: assumption eqn} for some $M>0$, a measure $\mu \in \mathcal{P}(\mathcal{Y})$, and transport plans $(\gamma_i)_{i \in \mathbb{N}} \in \Pi(\rho, \mu)$ such that for any $q \in [1, +\infty)$ and
    \begin{equation*}
    \alpha > \frac{\theta}{q(1+\theta)},
    \end{equation*}
    it holds that
    \begin{equation}
    \label{eq: sup the coercivity fails}
    \limsup_{i \to \infty} \frac{W_q(\gamma_{\rho \to \mu}, \gamma_i)}{\left(\int \langle x, y \rangle \mathrm{d} (\gamma_{\rho \to \mu} - \gamma_i)\right)^\alpha} = \infty. 
    \end{equation}
    Consequently the exponents in Theorems~\ref{thrm: coercivity general marginals} and \ref{thrm: coercivity rough measures} are sharp.

    \begin{equation*}
        W_q(\gamma_{\rho \to \mu}, \gamma_i) = 2 a_i \sigma_i^{1/q},
    \end{equation*}
    where the equality is a consequence of the upper bound by $\|T_0 - T_i\|_{L^q(\rho)} = 2 a_i \sigma_i^{1/q}$. To compute the excess transport cost, we compute 
    \begin{equation*}
        \Delta_i := \int \|T_i(x) - x\|^2 - \|T_0(x) - x\|^2 \mathrm{d} \rho(x) = \int \|x - y\|^2 \mathrm{d} (\gamma_i - \gamma_{\rho \to \mu}) = 2 \int \langle x, y \rangle \mathrm{d} (\gamma_{\rho \to \mu} - \gamma_i),

    \begin{align*}
        \|T_i(x) - x\|^2 - \|T_0(x) - x\|^2 =& \|x - y_i^+\|^2 - \|x - y_i^-\|^2\\
        =& (a_i - x_2)^2 - (a_i + x_2)^2 = 4 a_i |x_2|
    \end{align*}
    with the value on $R_i^- \cap \{ x_2 \geq 0\}$ equal by symmetry. Consequently,
    \begin{equation*}
        \Delta_i = 2a_i\int_{R_i^+ \cup R_i^-} |x_2| \mathrm{d} \rho(x) = \frac{4 a_i l_i}{Z_\theta} \int_{-h_i/2}^{h_i/2} |x_2|^\theta \mathrm{d} x_2 = \frac{4}{Z_\theta(1+ \theta)2^\theta} a_i l_i h_i^{1+\theta}.
    \end{equation*}

\label{thrm: strong convexity with general exponents}
    Let $q \in [1, +\infty)$, let $\mathcal{X}, \mathcal{Y} \subset \mathbb{R}^d$ be compact and convex for $d\geq2$, and let $\rho$ be a probability measure on $\mathcal{X}$ satisfying \ref{eq: assumption eqn}. Then for all $\psi_0, \psi_1 \in \mathcal{C}^0(\mathcal{Y})$,
    \begin{equation}
    \label{eq: kanto strong conv Lq}
        \|\nabla \psi_0^* - \nabla\psi^*_1\|_{L^q(\rho)}^{\frac{q(1 + \theta)}{\theta}}\leq C^{-1}J_\rho(\psi_0, \psi_1) 
    \end{equation}
    where
    \begin{equation*}
        C = \theta(1+\theta)^{-\frac{1+\theta}{\theta}} 4^{-\frac{1+\theta}{\theta}} \left(k_q D_\mathcal{Y}^{q+\theta(q-1)}c_d M D_\mathcal{X}^{d-1}\right)^{-\frac{1}{\theta}}
    \end{equation*}
    and $k_q$ is defined in the proof.

For $i=0,1$, $\phi_i = \psi_i^*$ is differentiable $\rho$-almost everywhere. Set $u(x) = \nabla \phi_1(x) - \nabla \phi_0(x)$ and $A = \|u\|_{L^q(\rho)}$. If $A=0$, the result follows from convexity of $\mathcal{K}_\rho$, so assume $A>0$. By definition,
    \begin{equation}
    \label{eq: inf convolution rep for jensen}
        J_\rho(\psi_0, \psi_1) = \int_\mathcal{X} j(x) \mathrm{d} \rho(x)
    \end{equation}
    where
    \begin{equation*}
        j(x) = \frac{\phi_0(x)+\phi_1(x)}{2} - \left(\frac{\psi_0+\psi_1}{2}\right)^*(x).
    \end{equation*}
    For every $h\in\mathbb{R}^d$, the definition of the conjugate gives
    \begin{equation*}
        \left(\frac{\psi_0+\psi_1}{2}\right)^*(x) \leq \frac{\phi_0(x+h)+\phi_1(x-h)}{2}.
    \end{equation*}
    Fix a point $x$ where both $\phi_i$ are differentiable. For some $h\in\mathbb{R}^d$ to be fixed later, choose $y_h^0\in\partial\phi_0(x+h)$ and $y_h^1\in\partial\phi_1(x-h)$. Subgradient inequalities give
    \begin{equation*}
        \phi_0(x) - \phi_0(x+h) \geq \langle y_h^0, -h \rangle \quad \text{and}\quad \phi_1(x) - \phi_1(x-h) \geq \langle y_h^1, h \rangle
    \end{equation*}
    so that
    \begin{align*}
        j(x) \geq& \frac{1}{2}\langle y_h^1 - y_h^0, h\rangle\\
        =& \frac{1}{2}\langle u(x), h\rangle + \frac{1}{2}\langle y_h^1 - \nabla \phi_1(x), h \rangle + \frac{1}{2}\langle \nabla\phi_0(x) - y_h^0, h \rangle.
    \end{align*}
    We choose, for some $\lambda>0$ to be fixed later,
    \begin{equation*}
        h_\lambda(x) = \begin{cases}
            \lambda\|u(x) \|^{q-2} u(x) & u(x) \neq 0, \\
            0& u(x) = 0.
        \end{cases}
    \end{equation*}
    Set $\eta_\lambda := D_\mathcal{Y}^{q-1}\lambda$, so that $\|h_\lambda(x)\|\leq\eta_\lambda$. Set $k_1 = 1$ and for $q>1$,
    \begin{equation*}
        k_q = \frac{4^{q-1}(q-1)^{q-1}}{q^q} \geq \frac{3}{4}.
    \end{equation*}
    For $q>1$, Young's inequality $a^{q-1}b \leq a^q/4 + k_q b^q$ implies
    \begin{equation}
        \label{eq: pointwise j rho lower bound}
        j(x) \geq \frac{\lambda}{4}\|\nabla \phi_1(x) - \nabla\phi_0(x) \|^{q} - \frac{k_q\lambda}{2} \left(\big(D_{\eta_{\lambda}} \phi_0(x)\big)^{q} + \big(D_{\eta_\lambda} \phi_1(x)\big)^{q} \right),
    \end{equation}
    and when $q=1$ the same inequality holds directly since $\|h_\lambda(x)\| \in \{0, \lambda\}$. For any $\lambda\geq 0$, Theorem~\ref{thrm: quant reg} gives
    \begin{align*}
         \int_{\mathcal{X}} k_q \big(D_{\eta_{\lambda}} \phi_i(x)\big)^{q} \mathrm{d} \rho(x) \leq& k_q D_\mathcal{Y}^q c_d M D_\mathcal{X}^{d-1} \eta_\lambda^\theta\\
        =& k_q D_\mathcal{Y}^{q+\theta(q-1)}c_d M D_\mathcal{X}^{d-1}\lambda^\theta = \overline C \lambda^\theta.
    \end{align*}
    Thus integrating \eqref{eq: pointwise j rho lower bound} with respect to $\rho$,
    \begin{equation}
        \label{eq: J rho lambda lower bound}
        J_{\rho}(\psi_0, \psi_1) \geq \frac{\lambda}{4}A^q - \overline C\lambda^{1+\theta}.
    \end{equation}
    We choose
    \begin{equation*}
        \lambda^\theta = \frac{A^q}{4(1+\theta)\overline C},
    \end{equation*}
    so that substituting the chosen value of $\lambda$ into \eqref{eq: J rho lambda lower bound} gives
    \begin{equation*}
        J_\rho(\psi_0, \psi_1) \geq  CA^{\frac{q(1+ \theta)}{\theta}}
    \end{equation*}
    for the $C$ given in the statement of the theorem, so the proof is complete.
