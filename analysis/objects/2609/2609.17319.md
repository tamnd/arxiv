  {\mathpalette\@transpose{}}

  {\mathpalette\@transpose{}}

\label{exa:PGL2}Suppose that $n=2$ and that $q$ is odd (so that
$d=2$) and that $q\not\equiv1\pmod 4$. We show that there is no
irreducible character of $\operatorname{PGL}_{2}(q)$ whose square contains every
irreducible character. On the other hand, we also show that there
does exist an irreducible character of $G=\operatorname{GL}_{2}(q)$ whose square
contains every irreducible character of $G$ that is trivial on the
centre. By inspection of the character table of $\operatorname{PGL}_{2}(q)$, one
sees that the sum of the degrees of all the irreducible characters
is $q^{2}+1$, so if $\rho\in\operatorname{Irr}(\operatorname{PGL}_{2}(q))$ has square that contains
every irreducible character, then $\rho$ must have degree $q+1$
(and in particular, the square of the Steinberg character does not
contain every irreducible character of $\operatorname{PGL}_{2}(q)$). Write $B$
for the upper triangular subgroup and $T$ for the diagonal subgroup,
so that $R^{G}_{T}=\operatorname{Ind}^{G}_{B}\operatorname{Inf}^{B}_{T}(-)$. The irreducible
characters of $\operatorname{PGL}_{2}(q)$ of degree $q+1$, that is, the irreducible
characters of $G$ trivial on $Z$, are of the form $R^{G}_{T}(\alpha,\alpha^{-1})$,
where $\alpha\in\operatorname{Irr}(\ensuremath{\mathbb{F}}^{\times}_{q})$ is such that $\alpha\neq\alpha^{-1}$
and $(\alpha,\alpha^{-1})\in\operatorname{Irr}(T)$ is trivial on $Z$. 

Let $\varepsilon\in\operatorname{Irr}(\ensuremath{\mathbb{F}}^{\times}_{q})$ be of order $2$, so that
$\varepsilon\circ\det$ is a linear character of $G$ of order $2$ that
is trivial on $Z$. We show that either $\mathds{1}$ or $\varepsilon\circ\det$
is not a contained in $R^{G}_{T}(\alpha,\alpha^{-1})^{2}$. Indeed,
\[
\langle\mathds{1},R^{G}_{T}(\alpha,\alpha^{-1})^{2}\rangle=\langle R^{G}_{T}(\alpha,\alpha^{-1}),\mkern 1.5mu\overline{\mkern-1muR^{G}_{T}(\alpha,\alpha^{-1})\mkern-1mu}\mkern 1.5mu\rangle,
\]
which is $1$ if and only if $\text{\ensuremath{R^{G}_{T}}(\ensuremath{\alpha},\ensuremath{\alpha^{-1}})}$
is self-dual. Assume that $R^{G}_{T}(\alpha,\alpha^{-1})$ is self-dual.
Then 
\begin{multline*}
\langle\varepsilon\circ\det,R^{G}_{T}(\alpha,\alpha^{-1})^{2}\rangle=\langle\varepsilon R^{G}_{T}(\alpha,\alpha^{-1}),R^{G}_{T}(\alpha,\alpha^{-1})\rangle\\
=\sum_{g\in B\backslash G/B}\langle\varepsilon\operatorname{Inf}^{B}_{T}(\alpha,\alpha^{-1})|_{B\cap\hspace{1pt}{\vphantom{B}}^{g}\hspace{-1pt}{B}},\hspace{1pt}{\vphantom{\operatorname{Inf}^{B}_{T}(\alpha,\alpha^{-1})}}^{g}\hspace{-1pt}{\operatorname{Inf}^{B}_{T}(\alpha,\alpha^{-1})}|_{B\cap\hspace{1pt}{\vphantom{B}}^{g}\hspace{-1pt}{B}}\rangle\\
=\langle(\varepsilon\circ\det|_{T})(\alpha,\alpha^{-1}),(\alpha,\alpha^{-1})\rangle+\langle(\varepsilon\circ\det|_{T})(\alpha,\alpha^{-1}),(\alpha^{-1},\alpha)\rangle.
\end{multline*}
Here the first term is $0$ since $\varepsilon\circ\det|_{T}\neq\mathds{1}_{T}$
and the second term is $0$ since if $\varepsilon\circ\det|_{T}=(\alpha^{-2},\alpha^{2})$,
then $\varepsilon(ab)=\alpha^{-2}(a)\alpha^{2}(b)$ for all $a,b\in\ensuremath{\mathbb{F}}^{\times}_{q}$,
so taking $a=b^{-1}$ implies that $\alpha^{4}=\mathds{1}$. But since
$q\not\equiv1\pmod 4$, there is no element of $\operatorname{Irr}(\ensuremath{\mathbb{F}}^{\times}_{q})$
of order $4$. Thus $\alpha^{2}=\mathds{1}$, contradicting $\alpha\neq\alpha^{-1}$.
Therefore $\varepsilon\circ\det$ is not contained in $R^{G}_{T}(\alpha,\alpha^{-1})^{2}$.

On the other hand, now drop the condition $q\not\equiv1\pmod 4$ and
consider the character $\sigma=R^{G}_{T}(\mathds{1},\varepsilon)$ of
$G$ (see Definition~\ref{def: Char sigma}; note that $\sigma$
is not trivial on $Z$). By direct computation of the character values
as in \cite[Theorem~3.1~(3)]{Aburto-Hageman-PantojaGL2} or by \cite[Theorem~1.1~(8)]{Gupta2025},
\[
\sigma^{2}=\operatorname{Ind}^{G}_{T}(\mathds{1},\varepsilon^{2})+R^{G}_{T}(\varepsilon,\varepsilon)=\operatorname{Ind}^{G}_{T}\mathds{1}+\varepsilon(\mathds{1}+\operatorname{St}_{2}).
\]
 It then follows from Theorem~\ref{thm: Restriction to T} that every
irreducible character of $\operatorname{PGL}_{2}(q)$ is either contained in $\operatorname{Ind}^{G}_{T}\mathds{1}$
or is linear (i.e., equal to $\varepsilon\circ\det$), thus in either
case is contained in $\sigma^{2}$.

it follows that $\operatorname{ord}(\chi)\mid\gcd(n,q-1)=d$. Thus $\chi$ lies
in the subgroup of $\operatorname{Irr}(\ensuremath{\mathbb{F}}^{\times}_{q})$ of order $d$  and by
assumption, $\varepsilon$ is a generator of that subgroup. There exists
a permutation matrix $g\in N_{G}(L)$ such that 
\[
\hspace{1pt}{\vphantom{\lambda}}^{g}\hspace{-1pt}{\lambda}=(\varepsilon^{d-j+i}\operatorname{St}_{m})^{d-1}_{j=0}=(\varepsilon^{d+i}\operatorname{St}_{m},\dots,\varepsilon^{i+1}\operatorname{St}_{m})=(\varepsilon^{i}\operatorname{St}_{m},\dots,\varepsilon^{i+1}\operatorname{St}_{m}).
\]
 Hence 
\[
\lambda\hspace{1pt}{\vphantom{\lambda}}^{g}\hspace{-1pt}{\lambda}=(\varepsilon^{i}\operatorname{St}^{2}_{m},\dots,\varepsilon^{i}\operatorname{St}^{2}_{m})=(\chi\circ\det)|_{L}(\operatorname{St}^{2}_{m},\ldots,\operatorname{St}^{2}_{m}).

and $L\cap U=\{1\}$. Let $H=P\cap\hspace{1pt}{\vphantom{P}}^{g}\hspace{-1pt}{P}$. Since $g\in N_{G}(L)$,
we have $H=L\bigl(U\cap\hspace{1pt}{\vphantom{U}}^{g}\hspace{-1pt}{U}\bigr)$. As both $\tilde{\lambda}|_{H}$

\[
\tilde{\lambda}|_{H}\hspace{1pt}{\vphantom{\tilde{\lambda}}}^{g}\hspace{-1pt}{\tilde{\lambda}}=\operatorname{Inf}^{H}_{L}(\lambda\hspace{1pt}{\vphantom{\lambda}}^{g}\hspace{-1pt}{\lambda})=(\chi\circ\det)|_{H}\,\operatorname{Inf}^{H}_{L}(\operatorname{St}^{2}_{m},\ldots,\operatorname{St}^{2}_{m}).

\sigma^{2} & \supseteq\operatorname{Ind}^{G}_{H}\big((\chi\circ\det)|_{H}\,\operatorname{Inf}^{H}_{L}(\operatorname{St}^{2}_{m},\ldots,\operatorname{St}^{2}_{m})\big)=\chi\operatorname{Ind}^{G}_{H}\operatorname{Inf}^{H}_{L}(\operatorname{St}^{2}_{m},\ldots,\operatorname{St}^{2}_{m})\\
 & \supseteq\chi\operatorname{Ind}^{G}_{H}\mathds{1}\supseteq\chi\circ\det.

\[

By assumption, the left-hand side is positive, hence there exists

\eta=(\eta_{1},\dots,\eta_{r}),\qquad\eta_{i}\in\operatorname{Irr}(\operatorname{GL}_{n_{i}}(q)).

whose square contains all the irreducible characters of $G$ with
central character $\zeta$ of order $2$. We identify the centre $Z$
of $G$ with $\ensuremath{\mathbb{F}}^{\times}_{q}$ and identify the corresponding characters.
Let $\chi$ be an irreducible character of $Z$ of order $4$, so
that $\operatorname{Irr}(Z)=\{\mathds{1},\zeta,\chi,\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\}$. Note that $\chi^{2}=\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu^{2}=\zeta$
and $\zeta\chi=\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu$.

We first find the irreducible characters of $G$ with central character
$\zeta$. Note that $\zeta$ extends to $\chi\circ\det\in\operatorname{Irr}(G)$,
since $\chi\circ\det|_{Z}=\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\circ\det|_{Z}=\chi^{2}=\zeta$.
Thus, $\rho\in\operatorname{Irr}(G)$ has central character $\zeta$ if and only
if $\chi\rho$ is trivial on $Z$. Thus the characters of $G$ with
central character $\zeta$ are precisely $C=\{\chi\tilde{\rho}\mid\rho\in\operatorname{Irr}(G/Z)\}$,
where $\tilde{\rho}$ is the inflation and we note, as is well-known,
that $G/Z\cong S_{5}$.

Next, assume that there exists a $\sigma\in\operatorname{Irr}(G)$ such that $\sigma^{2}$
contains all the characters $\chi\tilde{\rho}\in C$. Then, since
the degrees of the characters in $C$ are $1,1,4,4,5,5,6$, we must
have $\sigma^{2}(1)\geq26$, hence, as the degrees of irreducible
characters of $G$ are $1,4,5$ and $6$, we must have $\sigma(1)\geq6$.
Moreover, since every irreducible constituent of $\sigma^{2}$ must
have central character $\zeta$, it follows that $\sigma$ has central
character $\chi$ or $\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu$. We thus proceed to find the irreducible
characters of $G$ of degree $6$ and central character $\chi$ or
$\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu$.

Let $\rho\in\operatorname{Irr}(G)$ be of degree $6$. Then $\rho=R^{G}_{T}(\alpha,\beta)$,
for some $\alpha,\beta\in\operatorname{Irr}(\ensuremath{\mathbb{F}}^{\times}_{q})$ with $\alpha\neq\beta$,
and $\rho$ has central character $\alpha\beta$. Assuming that $\alpha\beta=\chi$,
so that $\beta=\mkern 1.5mu\overline{\mkern-1mu\alpha\mkern-1mu}\mkern 1.5mu\chi$, the possibilities for $(\alpha,\beta)$
are $(\mathds{1},\chi),(\zeta,\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu),(\chi,\mathds{1}),(\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu,\zeta)$.
This gives two distinct irreducible characters
\[
\rho_{1}=R^{G}_{T}(\mathds{1},\chi),\qquad\rho_{2}=R^{G}_{T}(\zeta,\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu).
\]
Similarly, the possible irreducible characters $R^{G}_{T}(\alpha,\beta)$
with central character $\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu$ are
\[
\rho_{3}=R^{G}_{T}(\mathds{1},\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu)=\mkern 1.5mu\overline{\mkern-1mu\rho\mkern-1mu}\mkern 1.5mu_{1},\qquad\rho_{4}=R^{G}_{T}(\zeta,\chi)=\mkern 1.5mu\overline{\mkern-1mu\rho\mkern-1mu}\mkern 1.5mu_{2}.
\]
Hence $\rho\in\{\rho_{1},\rho_{2},\rho_{3},\rho_{4}\}$.

Finally, we compute inner products of the squares of the $\rho_{i}$
with linear characters to derive a contradiction. Let $\varphi\in\operatorname{Irr}(G/Z)$
denote the nontrivial linear character. Then $\tilde{\varphi}=\zeta\circ\det$,
so $\chi\tilde{\varphi}=\chi\zeta\circ\det=\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\circ\det$. Thus
\begin{align*}
\langle\rho^{2}_{1},\chi\tilde{\varphi}\rangle & =\langle\rho_{1},\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\mkern 1.5mu\overline{\mkern-1mu\rho\mkern-1mu}\mkern 1.5mu_{1}\rangle=\langle\rho_{1},\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\rho_{3}\rangle=\langle\rho_{1},R^{G}_{T}(\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\circ\det)(\mathds{1},\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu)\rangle\\
 & =\langle\rho_{1},R^{G}_{T}(\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu,\zeta)\rangle=\langle\rho_{1},\rho_{2}\rangle=0
\end{align*}
and
\begin{align*}
\langle\rho^{2}_{2},\chi\tilde{\varphi}\rangle & =\langle\rho_{2},\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\mkern 1.5mu\overline{\mkern-1mu\rho\mkern-1mu}\mkern 1.5mu_{2}\rangle=\langle\rho_{2},\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\rho_{4}\rangle=\langle\rho_{2},R^{G}_{T}(\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\circ\det)(\zeta,\chi)\rangle\\
 & =\langle\rho_{2},R^{G}_{T}(\chi,\mathds{1})\rangle=\langle\rho_{2},\rho_{1}\rangle=0.
\end{align*}
Moreover, from these computations we also deduce
\begin{align*}
\langle\rho^{2}_{3},\chi\circ\det\rangle & =\langle\rho_{3},\chi\mkern 1.5mu\overline{\mkern-1mu\rho\mkern-1mu}\mkern 1.5mu_{3}\rangle=\langle\rho_{3},\chi\rho_{1}\rangle=\langle\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\rho_{3},\rho_{1}\rangle=0
\end{align*}
and
\begin{align*}
\langle\rho^{2}_{4},\chi\circ\det\rangle & =\langle\rho_{4},\chi\mkern 1.5mu\overline{\mkern-1mu\rho\mkern-1mu}\mkern 1.5mu_{4}\rangle=\langle\rho_{4},\chi\rho_{2}\rangle=\langle\mkern 1.5mu\overline{\mkern-1mu\chi\mkern-1mu}\mkern 1.5mu\rho_{4},\rho_{2}\rangle=0.
\end{align*}
Therefore every candidate square omits at least one of the two required
linear characters, which is a contradiction.
