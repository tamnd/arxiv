<!-- theorem-1: body_md -->
\label{thm:intro} There exists a set $S \subset (-E,E)$, locally finite in $(-E,E)$, such that for $\lambda \in (-E,E) \setminus S$, the equation $(\mathcal{D}-\lambda) u = 0$ has a solution $u \in H^1_{\operatorname{loc}}$ of the form
\begin{equation}\label{eq:intro-main}
    u(x) \; = \; \chi(-x \cdot e_-)\psi_{\lambda,-}(x) \; + \; t(\lambda)\cdot \chi(x \cdot e_+)\psi_{\lambda,+}(x) \; + \; \mathcal{E}_\nu,
\end{equation}
where $\nu > 0$ and $t(\lambda) \in \mathbb{C}$ has modulus one.
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
\label{def:domain-wall}
A horizontal domain wall is a function $\kappa_* : \mathbb{R}^2 \rightarrow \mathbb{R}$ of the form $\kappa_*(x) = k_*(x_2)$, where $k_* \in L^\infty(\mathbb{R},\mathbb{R})$ is such that
\begin{equation}\label{eq:domain-wall}
    k_*(x_2) = 1 \ \text{ for } x_2 \le -1, \qquad k_*(x_2) = -1 \ \text{ for } x_2 \ge 1.
\end{equation}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:dirac}
Given a horizontal domain wall $\kappa_*$, the unperturbed Dirac operator is
\begin{equation}\label{eq:dirac}
    \mathcal{D}_* \stackrel{\mathrm{def}}{=} D_1 \sigma_1 + D_2 \sigma_2 + \kappa_*(x)\sigma_3,
\end{equation}
acting on $L^2$ with domain $H^1$.
<!-- end definition-2 -->
<!-- lemma-1: body_md -->
\label{lem:zero-mode}
The kernel of $\mathcal{D}_\perp$ on $L^2(\mathbb{R},\mathbb{C}^2)$ is spanned by
\begin{equation}\label{eq:phi0}
    \phi_*(x_2) \stackrel{\mathrm{def}}{=} c_* \begin{bmatrix} 1 \\ 1 \end{bmatrix} e^{K_*(x_2)}, \qquad K_*(x_2) \stackrel{\mathrm{def}}{=} \int_0^{x_2} k_*(s)\,ds,
\end{equation}
where $c_* > 0$ is the constant normalizing $\phi_*$ in $L^2(\mathbb{R},\mathbb{C}^2)$. Moreover, $\sigma_1 \phi_* = \phi_*$ and $M_* \stackrel{\mathrm{def}}{=} \lVert e^{|x_2|}\phi_* \rVert_\infty < +\infty$, so that $|\phi_*(x_2)| \le M_* e^{-|x_2|}$ for all $x_2 \in \mathbb{R}$.
<!-- end lemma-1 -->
<!-- definition-3: body_md -->
\label{def:edge-projection}
The edge projection $P_*$ is the orthogonal projection of $L^2$ defined by
\begin{equation}\label{eq:edge-projection}
    (P_*f)(x) \stackrel{\mathrm{def}}{=} a_*(x_1)\,\phi_*(x_2), \qquad
    a_*(x_1) \stackrel{\mathrm{def}}{=} \int_\mathbb{R} \overline{\phi_*(x_2)}^{\top}\, f(x_1,x_2)\, dx_2 .
\end{equation}

<!-- lemma-2: body_md -->
\label{lem:decomposition}
The projection $P_*$ commutes with $\mathcal{D}_*$, and:
\begin{enumerate}[label=(\alph*)]
    \item on  $\mathcal{H}_*$, the operator $\mathcal{D}_*$ acts as $D_1$:
    \begin{equation}
        \mathcal{D}_*\left(\beta(x_1)\phi_*(x_2)\right) = (D_1\beta)(x_1)\,\phi_*(x_2).
    \end{equation}
    \item the restriction of $\mathcal{D}_*$ to $\mathcal{H}_*^\perp$ has spectrum contained in $\Omega_*^c = \mathbb{R}\setminus\left(-E_*,E_*\right)$.
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
A direct computation based on \Cref{lem:zero-mode} shows (a).

For (b), recall that $\sigma_1$ anticommutes with $\sigma_2$ and with $\sigma_3$. Hence
\begin{equation}\label{eq:chiral}
    \sigma_1 \mathcal{D}_\perp = -\mathcal{D}_\perp \sigma_1 .
\end{equation}
Expanding the square and cancelling the cross terms,
\begin{equation}\label{eq:fiber-square}
    \mathcal{D}_*^2 = D_1^2 + D_1\left(\sigma_1 \mathcal{D}_\perp + \mathcal{D}_\perp \sigma_1\right) + \mathcal{D}_\perp^2 = D_1^2 + \mathcal{D}_\perp^2 \geq \mathcal{D}_\perp^2.
\end{equation}
By \Cref{lem:zero-mode}, the kernel of $\mathcal{D}_\perp$ on $L^2(\mathbb{R},\mathbb{C}^2)$ is $\mathbb{C}\phi_*$; and by \eqref{eq:E-star}, the spectrum of $\mathcal{D}_\perp$ meets $\left(-E_*,E_*\right)$ only at $0$. Hence $\mathcal{D}_\perp^2 \geq E_*^2$ on $\left(\mathbb{C}\phi_*\right)^\perp$. By arguing fiberwise, we deduce that  $\mathcal{D}_\perp^2 \ge E_*^2$ as an operator on $\mathcal{H}_*^\perp$. Therefore, $\mathcal{D}_*^2 \geq E_*^2$ on $\mathcal{H}_*^\perp$. Assertion (b) follows.

<!-- lemma-3: body_md -->
\label{lem:edge-flux}
Let $\lambda \in \mathbb{R}$, let $\chi \in C^\infty(\mathbb{R})$ such that for some $T > 0$, $\chi(t) = 0$ when $t \le -T$ and $\chi(t) = 1$ when $t \ge T$, and let $\zeta = \left(\zeta_1,\zeta_2\right) \in \mathbb{R}^2$ be a unit vector with $\zeta_1 > 0$. Set
\begin{equation}\label{eq:edge-flux-w}
    w_*(x) \stackrel{\mathrm{def}}{=} \chi\left(x\cdot\zeta\right) e^{i\lambda x_1}\, \phi_*(x_2) .
\end{equation}
For $R > 0$, we have:
\begin{equation}\label{eq:edge-flux}
    \int_{\partial B_R(0)} J(w_*,w_*)\cdot n \ d\ell = 1+O\left(e^{-\zeta_1 R/2}\right).
\end{equation}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Because $\sigma_1\phi_* = \phi_*$ and $\overline{\phi_*}^\top\sigma_2\phi_* = 0$, we have
\begin{equation}\label{eq:edge-flux-J}
    J(w_*,w_*)(x) = \chi\left(x\cdot\zeta\right)^2 \begin{bmatrix} \overline{\phi_*(x_2)}^\top\sigma_1\phi_*(x_2) \\[4pt] \overline{\phi_*(x_2)}^\top\sigma_2\phi_*(x_2) \end{bmatrix}
    = \chi\left(x\cdot\zeta\right)^2 \left|\phi_*(x_2)\right|^2 e_1 .
\end{equation}


Because \eqref{eq:edge-flux-J} is carried by $e_1$, we have:
\begin{equation}\label{eq:edge-flux-divergence}
    \operatorname{div}J(w_*,w_*)(x) = \zeta_1\left(\chi^2\right)'\left(x\cdot\zeta\right)\left|\phi_*(x_2)\right|^2.
\end{equation}
The divergence theorem on $B_R(0)$ therefore gives
\begin{equation}\label{eq:edge-flux-div}
    \int_{\partial B_R(0)} J(w_*,w_*)\cdot n \ d\ell \; = \; \int_{B_R(0)} \zeta_1\left(\chi^2\right)'\left(x\cdot\zeta\right)\left|\phi_*(x_2)\right|^2 dx .
\end{equation}
For fixed $x_2$, the substitution $t = x\cdot\zeta$ has $dt = \zeta_1\,dx_1$; since $\chi^2$ runs from $0$ to $1$ and $\lVert \phi_* \rVert_{L^2(\mathbb{R})} = 1$, the same integrand over the whole plane gives
\begin{equation}\label{eq:edge-flux-full}
    \int_{\mathbb{R}^2} \zeta_1\left(\chi^2\right)'\left(x\cdot\zeta\right)\left|\phi_*(x_2)\right|^2 dx
    = \int_\mathbb{R} \left(\chi^2\right)'(t)\,dt \cdot \int_\mathbb{R} \left|\phi_*(x_2)\right|^2 dx_2 = 1 .
\end{equation}
It remains to discard the part of \eqref{eq:edge-flux-div} lying outside $B_R(0)$. Without loss of generality, assume $R \geq 2T/\zeta_1$. The function $\left(\chi^2\right)'$ is supported in $[-T,T]$, so the integral
\begin{equation}
    \int_{|x| > R} \zeta_1\left(\chi^2\right)'\left(x\cdot\zeta\right)\left|\phi_*(x_2)\right|^2 dx
\end{equation}
takes place on $\left|\zeta_1 x_1+\zeta_2 x_2\right| \le T$, where $\zeta_1\left|x_1\right| \le T + \left|x_2\right|$ and hence $\zeta_1 R \le \zeta_1|x| \le T+2|x_2|$; together with $R \ge 2T/\zeta_1$ this forces $|x_2| \ge \zeta_1 R/4$. Since $\phi_*$ decays exponentially in $x_2$, the same substitution gives
\begin{equation}
    \int_{|x| \ge R} \zeta_1\left|\left(\chi^2\right)'\left(x\cdot\zeta\right)\right| \left|\phi_*(x_2)\right|^2 dx \leq \int_\mathbb{R} \big| (\chi^2)'(t)\big| dt \cdot \int_{|x_2| \geq \zeta_1 R/4} \left|\phi_*(x_2)\right|^2 dx_2 = O\left(e^{-\zeta_1 R/2}\right).
\end{equation}
This proves \eqref{eq:edge-flux}.

<!-- proposition-1: body_md -->
\label{prop:edge-resolvent}
The family of operators $(\mathcal{D}_* - \lambda)^{-1}P_* : L^2_{\operatorname{comp}} \rightarrow L^2_{\operatorname{loc}}$, defined initially for $\operatorname{Im} \lambda > 0$, continues analytically to a family of operators
$A_*(\lambda) : L^2_{\operatorname{comp}} \rightarrow L^2_{\operatorname{loc}}$ for $\lambda \in \Omega_*$.


Moreover, for $\lambda \in \widetilde{\Omega}_*$ and $\gamma$ such that $|\operatorname{Im}\lambda| < \gamma \le \gamma_*(\lambda)$, the operator $A_*(\lambda)$ extends by density from $\mathcal{E}_\gamma$ to $L^2_{\operatorname{loc}}$, and we have the formula:
\begin{equation}\label{eq:edge-resolvent}
   A_*(\lambda)f(x) = i\int_{-\infty}^{x_1} e^{i\lambda(x_1-s)}a_*(s)\,ds \cdot \phi_*(x_2), \quad f\in \mathcal{E}_\gamma, \quad  P_*f(x) = a_*(x_1) \phi_*(x_2),
\end{equation}
where the above integral converges absolutely.
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
\textbf{1.} Before all, we justify that for $\lambda \in \widetilde{\Omega}_*$, $\gamma$ such that $|\operatorname{Im}\lambda| < \gamma \le \gamma_*(\lambda)$, $f \in \mathcal{E}_\gamma$, and $a_*$ as in \eqref{eq:edge-projection}, the integral in \eqref{eq:edge-resolvent} converges absolutely. 
    
    We claim that $a_* \in e^{-\gamma|x_1|}L^2(\mathbb{R})$. Indeed, using the formula \eqref{eq:edge-projection} for $a_*$, we have:
\begin{equation}
    e^{\gamma|x_1|}\left|a_*(x_1)\right| \; \le \; \int_\mathbb{R} \left|\phi_*(x_2)\right| e^{\gamma|x|}\left|f(x)\right| dx_2 \; \le \; \lVert \phi_* \rVert_{L^2(\mathbb{R})} \left(\int_\mathbb{R} e^{2\gamma|x|}\left|f(x)\right|^2 dx_2\right)^{1/2} .
\end{equation}
Since $\lVert \phi_* \rVert_{L^2(\mathbb{R})} = 1$, integrating in $x_1$ produces \eqref{eq:edge-data}:
\begin{equation}\label{eq:edge-data}
   \lVert e^{\gamma|x_1|}a_* \rVert_{L^2(\mathbb{R})}^2 = \int_\mathbb{R} e^{2\gamma|x_1|}\left|a_*(x_1)\right|^2 dx_1 \; \le \; \int_{\mathbb{R}^2} e^{2\gamma|x|}\left|f(x)\right|^2 dx = \|f\|_{\mathcal{E}_\gamma}^2.
\end{equation}

We now note that: 
\begin{equations}\label{eq:edge-integral-bound}
    \int_{-\infty}^{x_1} \left| e^{-i\lambda s} a_*(s) \right| ds \leq \int_{-\infty}^{x_1} \left| e^{|\operatorname{Im} \lambda|\,|s|}a_*(s) \right| ds
    \\
     = \int_{-\infty}^{x_1} \left| e^{ (|\operatorname{Im} \lambda|-\gamma)|s|} \cdot e^{ \gamma|s|} a_*(s) \right| ds \leq \dfrac{1}{\sqrt{\gamma-|\operatorname{Im} \lambda|}} \lVert e^{\gamma|x_1|}a_* \rVert_{L^2(\mathbb{R})}\leq \dfrac{1}{\sqrt{\gamma-|\operatorname{Im} \lambda|}} \lVert f \rVert_{\mathcal{E}_\gamma},
\end{equations}
where in the last line we used the Cauchy--Schwarz inequality and $|\operatorname{Im} \lambda| < \gamma$. This proves that the integral in \eqref{eq:edge-resolvent} converges absolutely.




    \textbf{2.} We now prove the core of the proposition. First work with $\lambda$ with $\operatorname{Im}\lambda>0$ and $f \in L^2_{\operatorname{comp}}$. We look for $u(x) = \beta(x_1)\phi_*(x_2) \in \mathcal{H}_*$ solving $(\mathcal{D}_*-\lambda)u = P_*f = a_*\phi_*$. Note that $a_* \in L^2_{\operatorname{comp}}(\mathbb{R})$: indeed, if $\operatorname{supp}(f) \subset B_R(0)$, for $|x_1| \geq R$ we have:
    \begin{equation}
        a_*(x_1) = \int_\mathbb{R} \overline{\phi_*(x_2)}^\top f(x_1,x_2) dx_2 = 0.
    \end{equation}
Because $a_* \in L^2(\mathbb{R})$ -- see \eqref{eq:edge-data} -- we deduce $a_* \in L^2_{\operatorname{comp}}$. By part (a) of \Cref{lem:decomposition}, solving $(\mathcal{D}_*-\lambda)u = a_*\phi_*$ in $\mathcal{H}_*$ reduces to the scalar ODE
\begin{equation}\label{eq:transport}
    (D_1 - \lambda)\beta = a_*.
\end{equation}

\textbf{3.} Because $f \in L^2_{\operatorname{comp}} \subset L^2$, $a_* \in L^2$. Moreover $D_1$ is selfadjoint on $L^2(\mathbb{R})$ and $\operatorname{Im} \lambda > 0$. Hence $\beta = (D_1-\lambda)^{-1} a_*$ is the unique solution to the equation \eqref{eq:transport} in $L^2(\mathbb{R})$. It is given by the integral formula
\begin{equation}\label{eq:edge-term}
    \beta(x_1) = (D_1-\lambda)^{-1} a_* = i\,e^{i\lambda x_1}\int_{-\infty}^{x_1} e^{-i\lambda s}a_*(s)\,ds
    = i\int_{-\infty}^{x_1} e^{i\lambda(x_1-s)}a_*(s)\,ds .
\end{equation}
Indeed, a straightforward calculation shows that \eqref{eq:edge-term} solves \eqref{eq:transport}. Moreover, \eqref{eq:edge-term} is square-integrable, because it is the convolution of $a_* \in e^{-\gamma|x_1|}L^2(\mathbb{R}) \subset L^2(\mathbb{R})$ against the kernel
\begin{equation}\label{eq:transport-kernel}
    k_\lambda(t) \stackrel{\mathrm{def}}{=} e^{i\lambda t}\,\1_{t \ge 0}, \qquad \text{which has} \qquad \lVert k_\lambda \rVert_{L^1(\mathbb{R})} = \int_0^{+\infty} e^{-t\operatorname{Im}\lambda}\,dt = \frac{1}{\operatorname{Im}\lambda} < +\infty .
\end{equation}
Young's convolution inequality therefore gives $\lVert \beta \rVert_{L^2(\mathbb{R})} \le (\operatorname{Im}\lambda)^{-1}\lVert a_* \rVert_{L^2(\mathbb{R})}$, in particular $\beta \in L^2(\mathbb{R})$. We deduce that for $\operatorname{Im} \lambda >0$ and $f \in L^2_{\operatorname{comp}}$,
\begin{equation}\label{eq:edge-resolvent-comp}
   A_*(\lambda)f(x) = i\int_{-\infty}^{x_1} e^{i\lambda(x_1-s)}a_*(s)\,ds \cdot \phi_*(x_2) .
\end{equation}

\textbf{4.} For $f \in L^2_{\operatorname{comp}}$ and $\lambda \in \Omega_*$, $a_* \in L^2_{\operatorname{comp}}(\mathbb{R})$, and the integral \eqref{eq:edge-resolvent-comp} defines an element of $L^2_{\operatorname{loc}}$: indeed,  for any $L$:
\begin{equations}
    \int_{-L}^L \left| ie^{i\lambda x_1} \int_{-\infty}^{x_1} e^{-i\lambda s}a_*(s)\,ds  \right|^2 dx_1  \leq \; 2L \cdot 2R\, e^{2\left|\operatorname{Im}\lambda\right|(L+R)} \| a_* \|_{L^2(\mathbb{R})}^2 \; < \; \infty,
\end{equations}
where $R$ is such that $\operatorname{supp}(a_*) \subset [-R,R]$, and where we used $\left|x_1-s\right| \le L+R$ on the domain of integration; the right-hand side is finite for every $\lambda \in \mathbb{C}$. Moreover, the formula \eqref{eq:edge-resolvent-comp} depends analytically on $\lambda$ as can be seen from an application of Morera's theorem. Because $\phi_* \in L^2(\mathbb{R})$ is normalized, this extension satisfies
\begin{equation}
\lVert A_*(\lambda) f \rVert^2_{L^2((-L,L) \times \mathbb{R})} = \int_{-L}^L \left| \int_{-\infty}^{x_1} e^{i\lambda(x_1-s)}a_*(s)\,ds  \right|^2 dx_1 \; < \; \infty .
\end{equation}
Therefore it defines an analytic family of operators $A_*(\lambda) : L^2_{\operatorname{comp}} \rightarrow L^2_{\operatorname{loc}}$ for $\lambda \in \Omega_*$, that extends $(\mathcal{D}_* - \lambda)^{-1}P_* : L^2_{\operatorname{comp}} \rightarrow L^2_{\operatorname{loc}}$ past $\left(-E_*,E_*\right)$.


\textbf{5.} Assume now $\lambda \in \widetilde{\Omega}_*$ and $\gamma$ such that $|\operatorname{Im}\lambda| < \gamma \le \gamma_*(\lambda)$. We justify that $A_*(\lambda)$ extends by density as an operator from $\mathcal{E}_\gamma$ to $L^2_{\operatorname{loc}}$. Let $f \in L^2_{\operatorname{comp}}$; in particular, $f \in \mathcal{E}_\gamma$ and $e^{\gamma|x_1|}a_* \in L^2(\mathbb{R})$. By \eqref{eq:edge-integral-bound}, we have, for any $L$:
\begin{align}\label{eq:edge-density-bound}
    \int_{-L}^L \left| ie^{i\lambda x_1} \int_{-\infty}^{x_1} e^{-i\lambda s}a_*(s)\,ds  \right|^2 dx_1 & \leq \int_{-L}^L e^{2|\operatorname{Im} \lambda| |x_1|} \cdot \dfrac{1}{\gamma-|\operatorname{Im} \lambda|} \lVert e^{\gamma|x_1|}a_* \rVert_{L^2(\mathbb{R})}^2 dx_1
    \\
&     \leq \dfrac{2L e^{2L}}{\gamma-|\operatorname{Im} \lambda|} \| f \|_{\mathcal{E}_\gamma}^2.
\end{align}
Because of \eqref{eq:edge-density-bound}, we conclude that $A_*(\lambda)$ maps $\mathcal{E}_\gamma$ to $L^2_{\operatorname{loc}}$ indeed. This completes the proof.

<!-- proposition-2: body_md -->
\label{prop:agmon} Let $\zeta \in \mathbb{R}^2$ be a unit vector, $|\lambda| < E_*$, and $0 < \varepsilon \leq 2\gamma_*(\lambda)$, with $\gamma_*(\lambda)$ given by \eqref{eq:def-gamma}.
For every $f \in \mathcal{H}_*^\perp$:
\begin{equation}\label{eq:agmon-estimate}
    e^{\varepsilon |x\cdot \zeta|} f \in L^2 \quad \Rightarrow \quad
    e^{\varepsilon|x \cdot \zeta|} B_*(\lambda) f \in L^2, \quad
    \big\| e^{\varepsilon|x \cdot \zeta|} B_*(\lambda) f \big\| \leq 4\varepsilon^{-1} \big\|e^{\varepsilon |x \cdot \zeta|} f\big\|.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
The proof relies on Agmon estimates, which are themselves an elaborate version of the identity
    \begin{equations}\label{eq:1e}
        \left\langle (S+\lambda-iT)v, (S-\lambda+iT)v \right\rangle
        \\
        = \| S v\|^2-|\lambda|^2 \| v\|^2 - \| T v\|^2 + 2i \operatorname{Re} \left\langle S v,Tv \right\rangle - 2i \operatorname{Im} \lambda \left\langle v,Sv \right\rangle + 2 \operatorname{Im} \lambda\left\langle v,Tv \right\rangle,
    \end{equations}
valid for selfadjoint operators $S$ and $T$ and $\lambda \in \mathbb{C}$. Taking the real part of \eqref{eq:1e} gives:
\begin{equation}\label{eq:1k}
        \operatorname{Re} \left\langle (S+\lambda-iT)v, (S-\lambda+iT)v \right\rangle 
        = \| S v\|^2-|\lambda|^2 \| v\|^2 - \| T v\|^2 + 2 \operatorname{Im} \lambda\left\langle v,Tv \right\rangle.
    \end{equation}


\textbf{1.} Let $|\lambda|<E_*$ and $0<\varepsilon \leq 2\gamma_*(\lambda)$ with $\gamma_*(\lambda)$ given by \eqref{eq:def-gamma}; note that $\varepsilon \in (0,1)$. Fix a unit vector $\zeta \in \mathbb{R}^2$, $N \in \mathbb{N}$, and define:
\begin{equations}\label{eq:1i}
    \varphi_k(x_k) = \begin{cases}
        -N & \quad \text{for } \varepsilon \zeta_k x_k \leq -N
        \\
        \varepsilon \zeta_k x_k & \quad \text{for } |\varepsilon \zeta_k x_k| \leq N
        \\
        N & \quad \text{for } \varepsilon \zeta_k x_k \geq N
    \end{cases}, \qquad
    \varphi(x) \stackrel{\mathrm{def}}{=} \varphi_1(x_1) + \varphi_2(x_2).
\end{equations}
The function $\varphi$ is bounded and Lipschitz (in particular it is differentiable almost everywhere) and $\| \nabla \varphi \|_\infty \leq \varepsilon$. Moreover, $|\varphi(x)| \leq \varepsilon |\zeta \cdot x|$. Indeed, write
\begin{equation}
    \varphi(x) = \pi_N(\varepsilon \zeta_1 x_1) + \pi_N(\varepsilon \zeta_2 x_2), \qquad \pi_N(s) = \max(-N,\min(N,s)).
\end{equation}
The map $\pi_N$ preserves sign. Therefore, if $\zeta_1 x_1$ and $\zeta_2 x_2$ have the same sign, so do $\pi_N(\varepsilon \zeta_1 x_1)$ and $\pi_N(\varepsilon \zeta_2 x_2)$. Hence, because $\pi_N$ is $1$-Lipschitz, 
\begin{equation}
    |\varphi(x)| = |\pi_N(\varepsilon \zeta_1 x_1)| + |\pi_N(\varepsilon \zeta_2 x_2)| \leq \varepsilon |\zeta_1 x_1| + \varepsilon |\zeta_2 x_2| = \varepsilon |x \cdot \zeta|.
\end{equation}
If on the other hand, $\zeta_1 x_1$ and $\zeta_2 x_2$ have different signs, so do $\pi_N(\varepsilon \zeta_1 x_1)$ and $\pi_N(\varepsilon \zeta_2 x_2)$. Because $|\pi_N(s)| = \min(N,|s|)$, we have $||\pi_N(s)|-|\pi_N(t)|| \leq ||s|-|t||$, so
\begin{equation}
    \big|\varphi(x) \big| \; = \; \Big| |\pi_N(\varepsilon \zeta_1 x_1)| - |\pi_N(\varepsilon \zeta_2 x_2)| \Big| \leq \varepsilon \big| |\zeta_1 x_1|- |\zeta_2 x_2| \big| = \varepsilon |\zeta \cdot x|.
\end{equation}
This proves that $|\varphi(x)| \leq \varepsilon |\zeta \cdot x|$ indeed.


Let $f \in \mathcal{H}_*^\perp$ such that $e^{\varepsilon |x\cdot \zeta|} f \in L^2$, and set $u = B_*(\lambda) f \in \mathcal{H}_*^\perp$.
The function $v=e^\varphi u$ satisfies $D_k v = e^\varphi D_ku -i \partial_k\varphi \cdot v$, so
\begin{equation}\label{eq:1q}
    (\mathcal{D}_*-\lambda + i (\sigma \cdot \nabla)\varphi) v = e^\varphi (\mathcal{D}_*-\lambda) u = e^\varphi f.
\end{equation}
Pair both sides with $(\mathcal{D}_*+\lambda - i (\sigma \cdot \nabla)\varphi) v = e^\varphi f+ 2\lambda v - 2i (\sigma \cdot \nabla)\varphi \cdot v$ to obtain
\begin{equation}\label{eq:1t}
    \left\langle (\mathcal{D}_*+\lambda - i (\sigma \cdot \nabla)\varphi) v, (\mathcal{D}_*-\lambda + i (\sigma \cdot \nabla)\varphi) v \right\rangle =  \big\| e^\varphi f \big\|^2 + 2\left\langle (\lambda - i(\sigma \cdot \nabla)\varphi) v, e^\varphi f \right\rangle.
\end{equation}
Take the real part of the left-hand-side and expand it according to  \eqref{eq:1k} to obtain:
\begin{equations}\label{eq:1l}
    \big\| \mathcal{D}_* v \big\|^2 - |\lambda|^2 \big\| v \big\|^2 - \big\| (\sigma \cdot \nabla)\varphi\, v \big\|^2 + 2 \operatorname{Im} \lambda \left\langle (\sigma \cdot \nabla)\varphi\, v, v \right\rangle
    \\
    =  \big\| e^\varphi f \big\|^2 + 2\operatorname{Re} \left\langle (\lambda - i(\sigma \cdot \nabla)\varphi) v, e^\varphi f \right\rangle.
\end{equations}

Recall that $|\lambda| < E_*$. We have the following inequalities:
\begin{align}
    \big| 2 \operatorname{Im} \lambda \left\langle (\sigma \cdot \nabla)\varphi\, v, v \right\rangle \big|
    & \leq \varepsilon^{-1} \big\| (\sigma \cdot \nabla)\varphi\, v \big\|^2 + \varepsilon \big\| v \big\|^2
    \leq \Big(\varepsilon^{-1} \big\| \nabla \varphi \big\|_\infty^2 + \varepsilon\Big) \big\| v \big\|^2 ,
    \\
    \big| 2 \operatorname{Re} \left\langle \lambda v, e^\varphi f \right\rangle \big|
    & \leq \varepsilon \big\| v \big\|^2 + \varepsilon^{-1} \big\| e^\varphi f \big\|^2 ,
    \\
    \big| 2 \operatorname{Re} \left\langle -i(\sigma \cdot \nabla)\varphi\, v, e^\varphi f \right\rangle \big|
    & \leq \varepsilon \big\| (\sigma \cdot \nabla)\varphi\, v \big\|^2 + \varepsilon^{-1} \big\| e^\varphi f \big\|^2
    \leq \varepsilon \big\| \nabla \varphi \big\|_\infty^2 \big\| v \big\|^2 + \varepsilon^{-1} \big\| e^\varphi f \big\|^2 .
\end{align}
Plug them in \eqref{eq:1l} to obtain:
\begin{equations}\label{eq:1m}
    \big\| \mathcal{D}_* v \big\|^2 - \Big(|\lambda|^2+2\varepsilon + (1+\varepsilon^{-1}+\varepsilon) \big\| \nabla \varphi \big\|_\infty^2 \Big) \big\| v \big\|^2
    \leq (1+2\varepsilon^{-1}) \big\| e^\varphi f \big\|^2.
\end{equations}
Because $\| \nabla \varphi \|_\infty \leq \varepsilon$ and $\varepsilon \in (0,1)$, we have $(1+\varepsilon^{-1}+\varepsilon) \| \nabla \varphi \|_\infty^2 \leq 3\varepsilon$ and $1+2\varepsilon^{-1} \leq 3\varepsilon^{-1}$. This implies
\begin{equations}\label{eq:1h}
    \big\| \mathcal{D}_* v \big\|^2 - \big(|\lambda|^2+5\varepsilon \big) \big\| v \big\|^2
    \leq 3\varepsilon^{-1} \big\| e^\varphi f \big\|^2.
\end{equations}


\textbf{2.} From \Cref{lem:decomposition}, we get:
\begin{equation}\label{eq:1u}
    \big\| \mathcal{D}_* v \big\|^2 \geq \big\| \mathcal{D}_* P_*^\perp v \big\|^2 \geq E_*^2\big\| P_*^\perp v \big\|^2 = E_*^2\Big(\big\| v \big\|^2 - \big\| P_*v \big\|^2\Big) .
\end{equation}
Plugging this in \eqref{eq:1h} produces
\begin{equation}\label{eq:1n}
    \big(E_*^2-|\lambda|^2-5\varepsilon \big) \big\| v \big\|^2 - E_*^2\big\| P_*v \big\|^2 \leq \big\| \mathcal{D}_* v \big\|^2 - \big(|\lambda|^2+5\varepsilon \big) \big\| v \big\|^2 \leq  3\varepsilon^{-1} \big\| e^\varphi f \big\|^2.
\end{equation}

\textbf{3.} We bound $\| P_*v \|$. Recall that  $\varphi = \varphi_1+\varphi_2$ was defined in \eqref{eq:1i}.
Set $w = e^{\varphi_2} u$ and note that $0 = P_*u = P_* e^{-\varphi_2} w$. Moreover, $P_*$ commutes with $e^{\varphi_1}$. Therefore,
\begin{equation}
    P_*v = P_* e^{\varphi_1+\varphi_2} u = e^{\varphi_1} P_* w = e^{\varphi_1} P_*(1-e^{-\varphi_2}) w.
 \end{equation}
It follows that:
\begin{equation}\label{eq:1p}
    \big\| P_*v \big\|^2 = \int_{\mathbb{R}^2} e^{2\varphi_1(x_1)} \left|\phi_*(x_2)\right|^2
    \left| \int_\mathbb{R} \overline{\phi_*(y_2)}^\top \left(1-e^{-\varphi_2(y_2)}\right) w(x_1,y_2)\,dy_2 \right|^2 dx .
\end{equation}
Below we use the notation $\Phi(y_2) = \overline{\phi_*(y_2)}^\top (1-e^{-\varphi_2(y_2)})$ and we estimate the integral \eqref{eq:1p}. First realize the integral in $x_2$, using that $\phi_*$ is normalized. Then use the Cauchy--Schwarz inequality on the integral in $y_2$ to obtain:
\begin{align}
    \big\| P_*v \big\|^2 & = \int_\mathbb{R} e^{2\varphi_1(x_1)}
    \left| \int_\mathbb{R} \Phi(y_2)\, w(x_1,y_2)\,dy_2 \right|^2 dx_1
    \\
    & \leq \big\| \Phi \big\|^2_{L^2(\mathbb{R})} \int_\mathbb{R} e^{2\varphi_1(x_1)}
    \big\| w(x_1,\cdot) \big\|^2_{L^2(\mathbb{R})} dx_1
     \; = \; \big\| \Phi \big\|^2_{L^2(\mathbb{R})} \big\| e^{\varphi_1} w \big\|^2 \; = \; \big\| \Phi \big\|^2_{L^2(\mathbb{R})} \big\| v \big\|^2 .
\end{align}

We now estimate $\| \Phi \|_{L^2(\mathbb{R})}$. The bounds $\left|1-e^{-s}\right| \leq |s|e^{|s|}$ and $|\varphi_2(x_2)| \leq \varepsilon |x_2|$ yield 
\begin{equation}\label{eq:Phi-pointwise}
    \big|1-e^{-\varphi_2(x_2)}\big| \; \leq \; \varepsilon\, |x_2|\, e^{\varepsilon|x_2|} .
\end{equation}
Recall from \Cref{lem:zero-mode} that $\left|\phi_*(x_2)\right| \leq M_* e^{-|x_2|}$. Since $\varepsilon \leq 1/10$, we deduce that:
\begin{equation}
    \big\| \Phi \big\|_{L^2(\mathbb{R})}^2 \; \leq \; M_*^2 \varepsilon^2 \int_\mathbb{R} x_2^2\, e^{-2(1-\varepsilon)|x_2|}\,dx_2 \; \leq \; M_*^2 \varepsilon^2 \int_\mathbb{R} x_2^2\, e^{-|x_2|}\,dx_2 \; = \; 4M_*^2 \varepsilon^2 .
\end{equation}
It follows that $\| P_*v \| \leq 2M_*\varepsilon \| v\|$.

\textbf{4.} Plugging this bound in \eqref{eq:1n} produces:
\begin{equation}\label{eq:1o}
    \big(E_*^2-|\lambda|^2-5\varepsilon - 4M_*^2 E_*^2\varepsilon^2 \big) \big\| v \big\|^2 \leq  3\varepsilon^{-1} \big\| e^\varphi f \big\|^2.
\end{equation}
Because $\varepsilon \leq 2\gamma_*(\lambda)$ with $\gamma_*(\lambda)$ given by \eqref{eq:def-gamma}, we have $\varepsilon \leq M_*^{-2}$ and $\varepsilon \leq \tfrac{E_*^2-|\lambda|^2}{10}$. Then $4M_*^2E_*^2\varepsilon^2 \leq 4E_*^2\varepsilon \leq 4\varepsilon$, so that
\begin{equation}\label{eq:1r}
    E_*^2-|\lambda|^2-5\varepsilon - 4M_*^2 E_*^2\varepsilon^2 \geq E_*^2-|\lambda|^2 - 9\varepsilon \geq \dfrac{E_*^2-|\lambda|^2}{10} \geq \varepsilon.
\end{equation}
Plugging \eqref{eq:1r} in \eqref{eq:1o} (and recalling that $|\lambda| < E_* \leq 1$), we obtain
\begin{equation}
    \big\| v \big\| \leq 2\varepsilon^{-1} \big\| e^\varphi f \big\|.
\end{equation}


\textbf{5.} Recall that $\varphi$ depends on $N$. We now emphasize this dependence by writing $\varphi = \varphi^{(N)}$ below. As $N \rightarrow \infty$, $\varphi^{(N)}$ converges to $\varepsilon \zeta\cdot x$. Therefore, by Fatou's lemma:
\begin{equations}
   \big\| e^{\varepsilon x \cdot \zeta} u \big\|^2 = \int_{\mathbb{R}^2} e^{2\varepsilon\, x \cdot \zeta} \left|u\right|^2 dx = \int_{\mathbb{R}^2} \liminf_{N \rightarrow \infty} e^{2\varphi^{(N)}} \left|u\right|^2 dx
   \\
   \leq \liminf_{N \rightarrow \infty} \int_{\mathbb{R}^2} e^{2\varphi^{(N)}} \left|u\right|^2 dx = \liminf_{N \rightarrow \infty} \big\| e^{\varphi^{(N)}} u \big\|^2 \leq 4\varepsilon^{-2} \cdot \liminf_{N \rightarrow \infty} \big\| e^{\varphi^{(N)}} f \big\|^2 \leq 4\varepsilon^{-2} \cdot\big\| e^{\varepsilon|x \cdot \zeta|} f \big\|^2,
\end{equations}
where we used $|\varphi (x)| \leq \varepsilon|x \cdot \zeta|$ in the last line.

In particular, $e^{\varepsilon x \cdot \zeta} u$ is in $L^2$. Changing $\zeta$ to $-\zeta$ produces $e^{-\varepsilon x \cdot \zeta} u \in L^2$ with the same bound. We can then conclude using the inequality $e^{\varepsilon |x \cdot \zeta|} |u| \leq e^{\varepsilon x \cdot \zeta} |u| +  e^{-\varepsilon x \cdot \zeta} |u|$.
It implies, by the triangle inequality on $L^2$:
\begin{equation}
    \big\| e^{\varepsilon |x \cdot \zeta|} u \big\| \leq \big\| e^{\varepsilon\, x \cdot \zeta} u \big\|  + \big\| e^{-\varepsilon x \cdot \zeta} u \big\| \leq 4\varepsilon^{-1} \cdot \big\| e^{\varepsilon |x \cdot \zeta|} f \big\|.
\end{equation}
This proves \eqref{eq:agmon-estimate} when $f \in \mathcal{H}_*^\perp$.

We have the identity
\begin{equation}\label{eq:cpt-resolvent-identity}
    B_+(\lambda)P_-^\perp - P_+^\perp B_-(\lambda) \ = \ B_+(\lambda)\,Z\,B_-(\lambda) , \qquad Z \stackrel{\mathrm{def}}{=} \left(\kappa_--\kappa_+\right)\sigma_3 = \mathcal{D}_- - \mathcal{D}_+.
\end{equation}
Indeed, for $\operatorname{Im}\lambda>0$ the operators $R_\pm(\lambda) = \left(\mathcal{D}_\pm-\lambda\right)^{-1}$ are bounded on $L^2$ and commute with $P_\pm$, so that the resolvent identity $R_+(\lambda)-R_-(\lambda) = R_+(\lambda)ZR_-(\lambda)$ gives
\begin{equation}
    B_+(\lambda)P_-^\perp - P_+^\perp B_-(\lambda) = P_+^\perp\left(R_+(\lambda)-R_-(\lambda)\right)P_-^\perp = P_+^\perp R_+(\lambda)\, Z\, R_-(\lambda)P_-^\perp = B_+(\lambda) Z B_-(\lambda) .
\end{equation}
Therefore \eqref{eq:cpt-resolvent-identity} holds for $\operatorname{Im}\lambda>0$; by analyticity, it holds for $|\lambda| < E$.

Combining \eqref{eq:cpt-AB}, \eqref{eq:cpt-B-split} and \eqref{eq:cpt-resolvent-identity}, we obtain the decomposition
\begin{equations}\label{eq:cpt-decomposition}
\begin{aligned}
    K_2(\lambda) &= \underbrace{W A_+(\lambda)}_{(1)} \; - \; \underbrace{W A_-(\lambda)}_{(2)} \; + \; \underbrace{W B_+(\lambda)\,Z\,B_-(\lambda)}_{(3)}
    \\[4pt]
    &\qquad + \; \underbrace{W B_+(\lambda)P_-}_{(4)} \; - \; \underbrace{W P_+B_-(\lambda)}_{(5)} .
\end{aligned}
\end{equations}
We prove in steps \textbf{3}--\textbf{6} below that each of $(1)$--$(5)$ is bounded from $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$, for some $\rho > \gamma$.


\textbf{2.} We record here a few weighted bounds on the operators $P_\pm, B_\pm(\lambda)$. \Cref{lem:projection-weight-tilted} applied to $P_\pm = P_{e_\pm}$ shows that the operators
\begin{equation}\label{eq:cpt-projection-bounds}
    P_\pm, P_\pm^\perp  \ : \ e^{-\mu |\zeta \cdot x|} L^2 \rightarrow e^{-\mu |\zeta \cdot x|} L^2 , \qquad |\zeta| = 1, \qquad \mu \in (0,1)
\end{equation}
are bounded. Together with (the tilted version of) \Cref{prop:agmon}, this implies  that
\begin{equation}\label{eq:cpt-agmon}
    B_\pm(\lambda) \ : \ e^{-\varepsilon |\zeta \cdot x|} L^2 \rightarrow e^{-\varepsilon |\zeta \cdot x|} L^2 , \qquad 0< \varepsilon \leq 2\gamma(\lambda), \qquad  |\zeta| = 1 
\end{equation}
is bounded.

Moreover, the operator
\begin{equation}\label{eq:cpt-projection-transverse}
    P_\pm \ : \ L^2 \ \longrightarrow \ e^{-\mu\left|x\cdot e_\pm^\perp\right|} L^2 , \qquad \mu \in (0,1)
\end{equation}
is bounded. Indeed,
\begin{equation}
    P_\pm f(x) = a_\pm\left(x\cdot e_\pm\right)\, \phi_\pm\left(x\cdot e_\pm^\perp\right) , \qquad a_\pm(s) \stackrel{\mathrm{def}}{=} \int_\mathbb{R} \overline{\phi_\pm(\tau)}^{\top} f\left(s\,e_\pm + \tau\, e_\pm^\perp\right) d\tau .
\end{equation}
Since $\lVert \phi_\pm \rVert_{L^2(\mathbb{R})} = 1$ and $P_\pm$ is an orthogonal projection, $\lVert a_\pm \rVert_{L^2(\mathbb{R})} = \lVert P_\pm f \rVert \leq \lVert f \rVert$. Thus
\begin{equation}
    \big\| e^{\mu\left|x\cdot e_\pm^\perp\right|} P_\pm f \big\| \ = \ \lVert a_\pm \rVert_{L^2(\mathbb{R})}\ \big\| e^{\mu\left|x\cdot e_\pm^\perp\right|}\phi_\pm \big\|_{L^2(\mathbb{R})} \ \leq \ \big\| e^{\mu\left|x\cdot e_\pm^\perp\right|}\phi_\pm \big\|_{L^2(\mathbb{R})}\, \lVert f \rVert ,
\end{equation}
and $\big\| e^{\mu|\tau|}\phi_\pm \big\|_{L^2(\mathbb{R})} < +\infty$ because $\mu < 1$ and $\left|\phi_\pm(\tau)\right| \leq M_\pm e^{-|\tau|}$ by \eqref{eq:phi-pm}. This proves \eqref{eq:cpt-projection-transverse}.


\textbf{3.} We work on the terms $W A_\pm(\lambda)$ in \eqref{eq:cpt-decomposition}. Heuristically, these map $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$ for some $\rho > \gamma$ because they add decay in the $e_\pm^\perp$-direction (due to $P_\pm$) and in the $e_0$-direction (due to $W$), and $\left(e_\pm^\perp, e_0\right)$ form a basis of $\mathbb{R}^2$. Let us be more specific. We recall from \eqref{eq:phi-pm} and \eqref{eq:edge-integral-bound} that
\begin{equations}\label{eq:cpt-edge-factors}
    \left|\phi_\pm\left(x\cdot e_\pm^\perp\right)\right| \ \le \ C\,e^{-\left|x\cdot e_\pm^\perp\right|},
\qquad
\left|\int_{-\infty}^{x\cdot e_\pm} e^{-i\lambda s}a_\pm(s)\,ds\right| \ \le \ \dfrac{\lVert f \rVert_{\mathcal{E}_\gamma}}{\sqrt{\gamma-\left|\operatorname{Im}\lambda\right|}}.
\end{equations}
This leads to:
\begin{align}
    \big|A_\pm(\lambda)f(x)\big| 
    & = \left| e^{i\lambda\, x\cdot e_\pm}\left(\int_{-\infty}^{x\cdot e_\pm} e^{-i\lambda s}\,a_\pm(s)\,ds\right)\phi_
    \pm\left(x\cdot e^\perp_\pm\right)\right|
    \\
    & \leq \ \dfrac{C\,\lVert f \rVert_{\mathcal{E}_\gamma}}{\sqrt{\gamma-\left|\operatorname{Im}\lambda\right|}}\; e^{\gamma|x| \, - \, \left|x\cdot e_\pm^\perp\right|} . \label{eq:cpt-edge-A-bound}
\end{align}

The function $W$ is supported in the strip $|x \cdot e_0| \leq 1$, and so $W(x) = O(e^{-|x \cdot e_0|})$. Combined with \eqref{eq:cpt-edge-A-bound}, this produces:
\begin{equation}\label{eq:cpt-amplitude}
    \big|W(x) A_\pm(\lambda)f(x)\big| 
     \leq \ \dfrac{C\,\lVert f \rVert_{\mathcal{E}_\gamma}}{\sqrt{\gamma-\left|\operatorname{Im}\lambda\right|}}\; e^{\gamma|x| \, - \, \left|x\cdot e_\pm^\perp\right| \, - \, \left|x\cdot e_0\right|} .
\end{equation}

We now study $\left|x\cdot e_\pm^\perp\right| + \left|x\cdot e_0\right|$. We have
\begin{equation}\label{eq:cpt-edge-geometry}
    \left|x\cdot e_\pm^\perp\right| + \left|x\cdot e_0\right| \ \ge \ \sqrt{\left(x\cdot e_\pm^\perp\right)^2 + \left(x\cdot e_0\right)^2} = \sqrt{x^\top M_\pm\, x}, \qquad M_\pm \stackrel{\mathrm{def}}{=} e_\pm^\perp\left(e_\pm^\perp\right)^\top + e_0\,e_0^\top .
\end{equation}
Set $c_\pm \stackrel{\mathrm{def}}{=} e_\pm^\perp \cdot e_0 = \mp \sin {\theta}_0$. The matrix $M_\pm$ maps $e_\pm^\perp$ to $e_\pm^\perp + c_\pm\,e_0$ and $e_0$ to $c_\pm\,e_\pm^\perp + e_0$. In that basis $M_\pm$ is represented by
\begin{equation}\label{eq:cpt-edge-matrix}
    \begin{bmatrix} 1 & c_\pm \\ c_\pm & 1 \end{bmatrix} ,
    \qquad\text{whose eigenvalues are}\qquad
    1 \pm c_\pm .
\end{equation}
The smallest eigenvalue of $M_\pm$ is therefore $1-\left|\sin\theta_0\right| \in (0,1]$, hence
\begin{equation}\label{eq:cpt-edge-sqrt}
    \left|x\cdot e_\pm^\perp\right| + \left|x\cdot e_0\right| \ \ge \sqrt{x^\top M_\pm\, x} \ \ge \ \sqrt{1-\left|\sin\theta_0\right|}\ |x| \ \ge \ \left(1-\left|\sin\theta_0\right|\right)|x|.
\end{equation}

Combining \eqref{eq:cpt-amplitude} and \eqref{eq:cpt-edge-sqrt}, we obtain 
\begin{equation}\label{eq:cpt-edge-radial}
    \left|W(x)\,A_\pm(\lambda)f(x)\right| \ \le \ \dfrac{C\,\lVert f \rVert_{\mathcal{E}_\gamma}}{\sqrt{\gamma-\left|\operatorname{Im}\lambda\right|}}\; e^{-\left(1-\left|\sin\theta_0\right|-\gamma\right)|x|} .
\end{equation}
In particular, $WA_\pm(\lambda)$ map $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$ when $\rho < 1 -\left|\sin\theta_0\right|-\gamma$.

\textbf{4.} We now work on the term $W B_+(\lambda) P_-$ in \eqref{eq:cpt-decomposition}. This term should also map $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$ for the same heuristic reason as the terms $(1)$ and $(2)$. By \eqref{eq:cpt-projection-transverse} with $\mu = \tfrac12$, and by \eqref{eq:cpt-agmon}, the operators
\begin{equation}\label{eq:cpt-term4-projection}
\begin{aligned}
    P_- \ &: \ L^2 \ \longrightarrow \ e^{-\frac12\left|x\cdot e_-^\perp\right|}L^2 \ \subset \ e^{-\gamma(\lambda) \left|x\cdot e_-^\perp\right|}L^2 ,
    \\
    B_+(\lambda) \ &: \ e^{-\gamma(\lambda) \left|x\cdot e_-^\perp\right|}L^2 \ \longrightarrow \ e^{-\gamma(\lambda) \left|x\cdot e_-^\perp\right|}L^2
\end{aligned}
\end{equation}
are bounded. Therefore, since $W$ is supported in the strip $|x \cdot e_0| \leq 1$ by \eqref{eq:parametrix} and \eqref{eq:parametrix-error}, we have:
\begin{equation}
    W B_+(\lambda) P_- \ : \ L^2 \ \longrightarrow \ e^{-|x\cdot e_0|-\gamma(\lambda) \left|x\cdot e_-^\perp\right|}L^2.
\end{equation}
By \eqref{eq:cpt-edge-sqrt}, and because $\gamma(\lambda) < 1$ by \eqref{eq:def_gamma}, we have
\begin{equation}
    |x\cdot e_0|+\gamma(\lambda) \left|x\cdot e_-^\perp\right| \geq \gamma(\lambda) \big( |x\cdot e_0| + \left|x\cdot e_-^\perp\right|\big) \geq \gamma(\lambda) (1-|\sin {\theta}_0|) |x|.
\end{equation}
Hence, for $\rho \leq \gamma(\lambda) (1-|\sin {\theta}_0|)$, the operator $W B_+(\lambda) P_-$ maps $L^2$ to $\mathcal{E}_\rho$, hence $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$.

\textbf{5.} We now work on the term $W P_+ B_-(\lambda)$ in \eqref{eq:cpt-decomposition}. Because $|\lambda| < E \le E_-$, $\lambda$ is not in the spectrum of $\mathcal{D}_-\big|_{\mathcal{H}_-^\perp}$; in particular, $B_-(\lambda)$ is bounded on $L^2$.

It therefore suffices to show that $W P_+$ is bounded from $L^2$ to $\mathcal{E}_\rho$ for some $\rho > \gamma$. By \eqref{eq:cpt-projection-transverse} with $\mu = \tfrac12$, and because $W$ is supported in the strip $\left|x\cdot e_0\right| \leq 1$, the operators
\begin{equation}\label{eq:cpt-term5-projection}
\begin{aligned}
    P_+ \ &: \ L^2 \ \longrightarrow \ e^{-\frac12\left|x\cdot e_+^\perp\right|}L^2 ,
    \\
    W \ &: \ e^{-\frac12\left|x\cdot e_+^\perp\right|}L^2 \ \longrightarrow \ e^{-\frac12\left(\left|x\cdot e_+^\perp\right| \, + \, \left|x\cdot e_0\right|\right)}L^2
\end{aligned}
\end{equation}
are bounded. By \eqref{eq:cpt-edge-sqrt}, we have
\begin{equation}
    \dfrac12\left(\left|x\cdot e_+^\perp\right| + \left|x\cdot e_0\right|\right) \ \geq \ \dfrac{1-\left|\sin{\theta}_0\right|}{2}\,|x| .
\end{equation}
For $\rho \leq \frac12\left(1-\left|\sin{\theta}_0\right|\right)$, we deduce that $W P_+ B_-(\lambda) : L^2 \rightarrow \mathcal{E}_\rho$ is bounded. In particular, $W P_+ B_-(\lambda)$ maps $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$.

\textbf{6.} We now work on $W B_+(\lambda)\,Z\,B_-(\lambda)$. This term adds decay in the direction $e_0^\perp$, as well as in all directions not in the support of $Z$; these cover a basis of $\mathbb{R}^2$, see \Cref{fig:supp-Z}. We first observe that $Z$ is supported at distance at most $1$ from the set
\begin{equation}\label{eq:1f}
    \{ x : \  \text{ $x\cdot e_+^\perp$ and $x\cdot e_-^\perp$ have opposite signs} \} .
\end{equation}
In particular, on the support of $Z$, we have $x \cdot e_+^\perp \leq 1$ and $x \cdot e_-^\perp \geq -1$; or $x \cdot e_+^\perp \geq -1$ and $x \cdot e_-^\perp \leq 1$. The vector $e_0^\perp$ decomposes in the basis $(e_\pm,e_\pm^\perp)$ as 
\begin{equation}\label{eq:1j}
    e_0^\perp = \sin \theta_0 \, e_+ + \cos {\theta}_0 \, e_+^\perp = -\sin \theta_0 \, e_- + \cos {\theta}_0 \, e_-^\perp .
\end{equation}

Therefore, if $x \cdot e_+^\perp \leq 1$ and $x \cdot e_-^\perp \geq -1$, we have:
\begin{equation}\label{eq:cpt-e0perp}
\begin{aligned}
    x \cdot e_0^\perp &= \sin {\theta}_0 \, x \cdot e_+ + \cos {\theta}_0 \, x \cdot e_+^\perp \leq \sin {\theta}_0 \, x \cdot e_+ + \cos {\theta}_0 ;
    \\
    x \cdot e_0^\perp &= -\sin {\theta}_0 \, x \cdot e_- + \cos {\theta}_0 \, x \cdot e_-^\perp \geq -\sin {\theta}_0 \, x \cdot e_- - \cos {\theta}_0 ;
    \\
    \Rightarrow \qquad \left|x \cdot e_0^\perp\right| &\leq \left|\sin {\theta}_0\right| \max\left(\left|x \cdot e_+\right| , \left|x \cdot e_-\right|\right) + \cos {\theta}_0 .
\end{aligned}
\end{equation}
A similar argument shows that \eqref{eq:cpt-e0perp} also holds when $x \cdot e_+^\perp \geq -1$ and $x \cdot e_-^\perp \leq 1$. Therefore, for all $x \in \operatorname{supp} Z$ and $\rho \leq \tfrac{\gamma}{|\sin {\theta}_0|}$,
\begin{align}
    \rho |x \cdot e_0^\perp| & \leq \rho |\sin \theta_0| \max (|x \cdot e_+| , |x \cdot e_-|) + \rho \cos {\theta}_0
    \\
    & \leq \gamma \max (|x \cdot e_+| , |x \cdot e_-|) + \rho. \label{eq:1g}
\end{align}
In particular, this proves that 
\begin{equation}
    \big| Z(x) e^{\rho |x \cdot e_0^\perp|} \big| \leq \| Z \|_\infty e^{\gamma \max (|x \cdot e_+| , |x \cdot e_-|) + \rho} \leq \| Z \|_\infty e^\rho (e^{\gamma |x \cdot e_+|} + e^{\gamma |x \cdot e_-|}).
\end{equation}
This implies that $Z$, as a multiplication operator, maps
$e^{-\gamma |x \cdot e_+|} L^2 \cap e^{-\gamma |x \cdot e_-|} L^2$ to $e^{-\rho |x \cdot e_0^\perp|} L^2$.

Recall $\gamma < \gamma(\lambda) (1-|\sin {\theta}_0|) < 2\gamma(\lambda)$ and assume that $0<\rho \leq \gamma(\lambda)(1-|\sin {\theta}_0|)$; in particular $0<\rho \leq 2\gamma(\lambda)$.  
Bringing in \Cref{prop:agmon} -- and the fact that $W$ is supported in $|x \cdot e_0| \leq 1$ -- we obtain that the operators
\begin{equation}
\begin{aligned}
    B_-(\lambda) \ &: \ \mathcal{E}_\gamma \ \longrightarrow \ e^{-\gamma \left|x \cdot e_+\right|} L^2 \cap e^{-\gamma \left|x \cdot e_-\right|} L^2 ,
    \\
    Z \ &: \ e^{-\gamma \left|x \cdot e_+\right|} L^2 \cap e^{-\gamma \left|x \cdot e_-\right|} L^2 \ \longrightarrow \ e^{-\rho \left|x \cdot e_0^\perp\right|} L^2 ,
    \\
    B_+(\lambda) \ &: \ e^{-\rho \left|x \cdot e_0^\perp\right|} L^2 \ \longrightarrow \ e^{-\rho \left|x \cdot e_0^\perp\right|} L^2 ,
    \\
    W \ &: \ e^{-\rho \left|x \cdot e_0^\perp\right|} L^2 \ \longrightarrow \ e^{-\rho \left|x \cdot e_0^\perp\right| - \rho \left|x \cdot e_0\right|} L^2 \ \subset \ \mathcal{E}_\rho
\end{aligned}
\end{equation}
are bounded. Chaining these together, we obtain that $W B_+(\lambda)\,Z\,B_-(\lambda)$ maps $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$. 

\textbf{7.} Therefore, if $\rho$ satisfies
\begin{equation}\label{eq:cpt-rho}
    \gamma \ < \ \rho \ < \ \min\left( 1-\left|\sin{\theta}_0\right|-\gamma \ , \ \ \gamma(\lambda)\left(1-\left|\sin{\theta}_0\right|\right) \ , \ \ \dfrac{1-\left|\sin{\theta}_0\right|}{2} \ , \ \ \dfrac{\gamma}{\left|\sin{\theta}_0\right|} \right)
\end{equation}
-- with the convention $\gamma/0 = +\infty$ when ${\theta}_0 = 0$ -- the terms $(1)$--$(5)$ from \eqref{eq:cpt-decomposition} are all bounded from $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$: the four upper bounds in \eqref{eq:cpt-rho} are the constraints produced in steps \textbf{3}, \textbf{4}, \textbf{5} and \textbf{6}, respectively. In particular,  if $\rho$ satisfies \eqref{eq:cpt-rho},
    $K_2(\lambda) : \mathcal{E}_\gamma \rightarrow \mathcal{E}_\rho$
 is bounded. 

It remains to check that there exist values $\rho$ satisfying \eqref{eq:cpt-rho}. We first note that
\begin{equation}
    \min\left( 1-\left|\sin{\theta}_0\right|-\gamma \ , \ \ \gamma(\lambda)\left(1-\left|\sin{\theta}_0\right|\right) \ , \ \ \dfrac{1-\left|\sin{\theta}_0\right|}{2} \right) = \gamma(\lambda)\left(1-\left|\sin{\theta}_0\right|\right)
\end{equation}
because $\gamma(\lambda) \leq 1/20$ and $\gamma < \tfrac{1-\left|\sin{\theta}_0\right|}{2}$. By assumption, $\gamma(\lambda)\left(1-\left|\sin{\theta}_0\right|\right) > \gamma$,
and $\tfrac{\gamma}{\left|\sin{\theta}_0\right|} > \gamma$ (because $|{\theta}_0| < \pi/2$), so the minimum in \eqref{eq:cpt-rho} is larger than $\gamma$ indeed. This completes the proof.

\label{lem:cpt-fredholm} Under the assumptions of \Cref{lem:cpt-gain}, the operators $K_1(\lambda), K_2(\lambda) : \mathcal{E}_\gamma \rightarrow \mathcal{E}_\gamma$ are compact.

\textbf{1.} The function $V_+$ is supported in
\begin{equation}
    \{ \kappa \neq \kappa_+ \} \cap \{ \chi_+ \neq 0 \}.
\end{equation}
These two sets lie within fixed distance of $\{ \pi < \arg x < \pi+{\theta}_+ \}$ and $\{ x \cdot e_0 > 0 \}$, respectively; see \Cref{fig:supp-V}. So $V_+$ is a bounded, compactly supported function. Let $\varphi \in C^\infty_0(\mathbb{R}^2)$ be equal to $1$ on a neighborhood of $\operatorname{supp} V_+$, so that $V_+ R_+(\lambda) = V_+ \cdot \varphi R_+(\lambda)$. For $\lambda \in \widetilde{\Omega}$, the operator $\varphi R_+(\lambda)$ maps $\mathcal{E}_\gamma$ boundedly to $H^1$, with values supported in the fixed compact set $\operatorname{supp}\varphi$; by the Rellich--Kondrachov theorem, it is therefore compact from $\mathcal{E}_\gamma$ to $L^2$. Since $V_+$ is bounded and compactly supported, the multiplication operator $V_+ : L^2 \rightarrow \mathcal{E}_\gamma$ is bounded. Being the composition of a compact and a bounded operator, $V_+R_+(\lambda)$ is thus compact as an operator on $\mathcal{E}_\gamma$. The same holds for  $V_- R_-(\lambda)$. It follows that $K_1(\lambda)$ is compact on $\mathcal{E}_\gamma$.

\textbf{2.} We now show that $K_2(\lambda) : \mathcal{E}_\gamma \rightarrow \mathcal{E}_\gamma$ is a limit (in operator norm) of compact operators on $\mathcal{E}_\gamma$; hence it is itself compact. For $n \in \mathbb{N}$, let $\varphi_n$ be a smooth function supported in $B_{2n}(0)$, equal to $1$ on $B_n(0)$, such that $0 \leq\varphi_n \leq 1$. By \Cref{lem:cpt-gain}, the operator $K_2(\lambda)$ maps $\mathcal{E}_\gamma$ to $\mathcal{E}_\rho$ for some $\rho > \gamma$, and thus:
\begin{align}
    \left\| K_2(\lambda) - \varphi_n K_2(\lambda) \right\|_{\mathcal{E}_\gamma \rightarrow \mathcal{E}_\gamma} & \leq \left\| 1 - \varphi_n \right\|_{\mathcal{E}_\rho \rightarrow \mathcal{E}_\gamma} \left\| K_2(\lambda) \right\|_{\mathcal{E}_\gamma \rightarrow \mathcal{E}_\rho}
    \\
    & \leq C \left\| (1 - \varphi_n) e^{(\gamma-\rho) |x|}\right\|_\infty \leq C e^{-(\rho - \gamma) n}.
\end{align}
Hence, as operators on $\mathcal{E}_\gamma$,
\begin{equation}
    K_2(\lambda) = \lim_{n \rightarrow \infty} \varphi_n K_2(\lambda).
\end{equation}
Because $R_\pm(\lambda)$ maps $\mathcal{E}_\gamma$ to $H^1_{\operatorname{loc}}$ and $W$ is smooth with bounded derivatives, $K_2(\lambda)$ maps $\mathcal{E}_\gamma$ to $H^1_{\operatorname{loc}}$, and hence $\varphi_n K_2(\lambda)$ maps $\mathcal{E}_\gamma$ to $H^1_{\operatorname{comp}}$, in particular it is a compact operator on $\mathcal{E}_\gamma$. This proves that $K_2(\lambda)$ is compact on $\mathcal{E}_\gamma$, and this completes the proof.

\textbf{1.}  We first claim that $\operatorname{Id}+K(\lambda)$ is invertible on $\mathcal{E}_\gamma$ for every $\lambda$ satisfying \eqref{eq:cpt-parameters}, with $\operatorname{Im}\lambda>0$. By \Cref{lem:cpt-fredholm}, $K(\lambda)$ is compact, so it suffices to show that $\operatorname{Id}+K(\lambda)$  is injective.


Let $f \in \mathcal{E}_\gamma$ with $\left(\operatorname{Id}+K(\lambda)\right)f = 0$. By \eqref{eq:parametrix-error}, we have
$(\mathcal{D}-\lambda) Q(\lambda)f = 0$. Because $\operatorname{Im} \lambda > 0$, $\mathcal{D}-\lambda$ is invertible on $L^2$ and hence $Q(\lambda) f = 0$. We write this identity as 
\begin{equation}
    0 = \chi_+u_++\chi_-u_-, \qquad u_\pm \stackrel{\mathrm{def}}{=} R_\pm(\lambda)f.
\end{equation}

Let $h \stackrel{\mathrm{def}}{=} u_+-u_- \in H^1$. Note that
\begin{equation}\label{eq:1s}
    \chi_- h = \chi_-u_+-\chi_-u_- =  \chi_-u_+ + \chi_+ u_+ = u_+.
\end{equation}
Therefore, from $u_\pm = R_\pm(\lambda)f$, we deduce that:
\begin{align}
      (\mathcal{D}_--\lambda) u_- = f &= (\mathcal{D}_+-\lambda) u_+  =  (\mathcal{D}_--\lambda) u_+ + (\mathcal{D}_+-\mathcal{D}_-) u_+ \\ & = (\mathcal{D}_--\lambda) u_+ + (\kappa_+-\kappa_-) \sigma_3 u_+ = (\mathcal{D}_--\lambda) u_+ + (\kappa_+-\kappa_-) \chi_- \sigma_3 h.
\end{align}
In particular, subtracting $(\mathcal{D}_--\lambda) u_-$ from both sides, we obtain:
\begin{equation}\label{eq:mero-selfadjoint}
    0 = \big(\mathcal{D}_- - \lambda + \chi_-\left(\kappa_+-\kappa_-\right)\sigma_3 \big) h.
\end{equation}
The operator $\mathcal{D}_- + \chi_-\left(\kappa_+-\kappa_-\right)\sigma_3$ is selfadjoint on $L^2$ with domain $H^1$. So \eqref{eq:mero-selfadjoint} is a spectral, selfadjoint problem with spectral parameter $\lambda$ with positive imaginary part. The only solution is $h=0$. This implies $u_+ =0$ by \eqref{eq:1s}, and so $f = (\mathcal{D}_+-\lambda)u_+ = 0$. Therefore $\operatorname{Id}+K(\lambda)$  is injective for $\operatorname{Im} \lambda > 0$, and hence $\operatorname{Id}+K(\lambda)$ is invertible when in addition $\lambda$ satisfies \eqref{eq:cpt-parameters}, by the Fredholm alternative.


\textbf{2.} Fix $ 0 < \gamma < \left(1-\left|\sin{\theta}_0\right|\right)\gamma(0)$ and set
\begin{equation}\label{eq:mero-domain}
    U_\gamma \stackrel{\mathrm{def}}{=} \left\{ \lambda \in \mathbb{C} \ : \ |\lambda| < E, \ \ \left|\operatorname{Im}\lambda\right| < \gamma < \left(1-\left|\sin{\theta}_0\right|\right)\gamma(\lambda) \right\} .
\end{equation}
For $\lambda \in U_\gamma$:
\begin{itemize}
    \item $K(\lambda) : \mathcal{E}_\gamma \rightarrow \mathcal{E}_\gamma$ is analytic;
    \item $K(\lambda) : \mathcal{E}_\gamma \rightarrow \mathcal{E}_\gamma$ is compact on $\mathcal{E}_\gamma$;
    \item $\operatorname{Id}+K(\lambda) : \mathcal{E}_\gamma \rightarrow \mathcal{E}_\gamma$ is invertible when $\operatorname{Im} \lambda > 0$.
\end{itemize}
Therefore, by analytic Fredholm theory \cite[Theorem C.8]{DZ}, $\left(\operatorname{Id}+K(\lambda)\right)^{-1}$ is a meromorphic family of bounded operators on $\mathcal{E}_\gamma$, $\lambda \in U_\gamma$, whose Laurent coefficients at each pole are finite rank.

\textbf{3.} Now observe that
\begin{equation}\label{eq:mero-formula}
    (\mathcal{D}-\lambda)^{-1} = Q(\lambda)\left(\operatorname{Id}+K(\lambda)\right)^{-1} \ : \ \mathcal{E}_\gamma \longrightarrow L^2_{\operatorname{loc}} ,
\end{equation}
valid when $\operatorname{Im} \lambda > 0$ and $\lambda$ satisfies \eqref{eq:mero-domain}, meromorphically continues to $\lambda \in U_\gamma$. This provides a meromorphic continuation of $(\mathcal{D}-\lambda)^{-1}$ from $\mathcal{E}_\gamma$ to $L^2_{\operatorname{loc}}$ past $(-E,E)$.

In particular this meromorphic continuation maps $L^2_{\operatorname{comp}}$ to $L^2_{\operatorname{loc}}$; we denote this family by $R(\lambda)$ below. A priori $R(\lambda)$ depends on $\gamma$, but since $L^2_{\operatorname{comp}}$ is dense in $\mathcal{E}_\gamma$, and they all continue the same operator for $\operatorname{Im} \lambda > 0$, the uniqueness of the meromorphic continuation shows that $R(\lambda)$ is $\gamma$-independent. In particular, we obtain a meromorphic continuation of $(\mathcal{D}-\lambda)^{-1}$ from $L^2_{\operatorname{comp}}$ to $L^2_{\operatorname{loc}}$, for 
\begin{equation}
    \lambda \in \bigcup_{0 < \gamma < (1-|\sin {\theta}_0|) \gamma(0)}U_\gamma = \widetilde{\Omega}.
\end{equation}

\textbf{4.} It remains to upgrade $L^2_{\operatorname{loc}}$ to $H^1_{\operatorname{loc}}$. Away from the poles, $u = R(\lambda)f$ satisfies $\left(\mathcal{D}-\lambda\right)u = f$ in the sense of distributions: this holds for $\operatorname{Im}\lambda>0$ and continues meromorphically to $\lambda \in \widetilde{\Omega}$. Hence $\left(\mathcal{D}-\lambda\right)u \in L^2_{\operatorname{loc}}$, and elliptic regularity for the first-order elliptic operator $\mathcal{D}$ gives $u \in H^1_{\operatorname{loc}}$. By the closed graph theorem, $R(\lambda)$ maps $L^2_{\operatorname{comp}}$, respectively $\mathcal{E}_\gamma$, to $H^1_{\operatorname{loc}}$. The proof is complete.

    \operatorname{div} J(v,w), \qquad J(v,w) \ \text{ as in } \eqref{eq:current} .
\end{equation}
In particular, if $u \in H^1_{\operatorname{loc}}$ satisfies $\mathcal{D} u = \lambda u$ for some $\lambda \in \mathbb{R}$, then

    \operatorname{div} J(u,u) = 0 .

Write $\langle a,b\rangle \stackrel{\mathrm{def}}{=} \overline{a}^\top b$ for the pairing of $\mathbb{C}^2$, conjugate linear in the first slot. We have
\begin{equation} \label{eq:current-second-half}
    \sum_j \sigma_j\partial_j w = i\left(\mathcal{D} w - \kappa\sigma_3 w\right), \qquad
    \sum_{j=1,2} \left\langle v, \sigma_j\partial_j w\right\rangle
    = i\left\langle v, \mathcal{D} w\right\rangle - i\kappa \left\langle v,\sigma_3 w\right\rangle .
\end{equation}
The matrices $\sigma_j$ being Hermitian, the same computation applied to $v$ gives
\begin{equation}\label{eq:current-first-half}
    \sum_{j=1,2}\left\langle \partial_j v, \sigma_j w\right\rangle
    = \left\langle \sum_{j=1,2}\sigma_j\partial_j v, w\right\rangle
    = \left\langle i\left(\mathcal{D} v-\kappa\sigma_3 v\right),w\right\rangle
    = -i\left\langle \mathcal{D} v,w\right\rangle + i\kappa\left\langle v,\sigma_3 w\right\rangle.
\end{equation}
Adding \eqref{eq:current-second-half} and \eqref{eq:current-first-half} gives \eqref{eq:current-identity}. Taking $v=w=u$, the components of $J(u,u)$ are real because $\sigma_1,\sigma_2$ are Hermitian, and \eqref{eq:current-identity} becomes 
\begin{equation}
\operatorname{div}J(u,u) = i\left(\lambda - \overline{\lambda}\right)|u|^2 = -2\,\operatorname{Im}\lambda  \cdot|u|^2  = 0.
\end{equation}
This completes the proof.

\begin{equation}\label{eq:B-v-v-flux}
    \int_{\partial B_r(0)} J(u_-,u_-)\cdot n \ d\ell
    \; = \; \int_{B_r(0)}\left(\rho^2\right)'(x_1)\left|\phi_-(x_2)\right|^2 dx
    \; \xrightarrow[\ r \rightarrow +\infty\ ]{} \; \int_\mathbb{R} \left(\rho^2\right)'(x_1)\,dx_1 \; = \; -1 .
\end{equation}
Likewise,
\Cref{lem:edge-flux-tilted} shows that
\begin{equation}\label{eq:Juplus}
    \int_{\partial B_r(0)} J(u_+,u_+)\cdot n \ d\ell
    \; \xrightarrow[\ r \rightarrow +\infty\ ]{} \; \big| \widehat{a_+}(\lambda) \big|^2.
\end{equation}



\textbf{5.} We now study the cross term $J(u_-,u_+)$. The function $u_-$ has support in $x_1 \leq 0$ and the function $u_+$ has support in $x \cdot e_0 \geq -1$. Therefore, on the support of $J(u_-,u_+)$, we have
\begin{equation}
    |x_1|\cos{\theta}_0 = -x_1\cos{\theta}_0 = -x \cdot e_0 + x_2\sin{\theta}_0 \leq 1 + |x_2|.
\end{equation}
Since moreover $u_-$ decays like $e^{-|x_2|}$, we deduce that 
\begin{equation}
    \big|J(u_-,u_+)(x)\big| \leq C e^{-|x_2|} \leq C \exp\left( {-\tfrac{|x_2|}{2} - \tfrac{|x_1|\cos{\theta}_0}{2}} \right).
\end{equation}
In particular, $J(u_-,u_+) \in L^1$. Going back to \eqref{eq:1v}, we obtain:
\begin{equation}\label{eq:1e-split}
    J(u,u) = J(u_+,u_+) + J(u_-,u_-) + h, \qquad h \in L^1.
\end{equation}

\textbf{6.} Note that
\begin{equation}
   +\infty > \int_{\mathbb{R}^2} |h| = \int_{\mathbb{R}^+} \left(\int_{|x| = r} |h| d\ell\right) dr.
\end{equation}
Therefore there exists a sequence $r_n \rightarrow + \infty$ such that
\begin{equation}\label{eq:h-flux}
    \int_{\partial B_{r_n}(0)} \left|h\right| \, d\ell \; \xrightarrow[\ n\to+\infty\ ]{} \; 0 .
\end{equation}


We deduce that as $n \rightarrow \infty$,
\begin{align}
   0 = \int_{\partial B_{r_n}(0)} J(u,u)\cdot n \ d\ell & = \int_{\partial B_{r_n}(0)} J(u_-,u_-)\cdot n \ d\ell + \int_{\partial B_{r_n}(0)} J(u_+,u_+)\cdot n \ d\ell   + o(1)
   \\
   & 
   \rightarrow -1+\big|\widehat{a_+}(\lambda)\big|^2.
\end{align}
In the first identity, we used \eqref{eq:1a}; in the second, we used the decomposition \eqref{eq:1e-split} as well as the asymptotics \eqref{eq:h-flux} for $h$; and in the third, we used the asymptotics \eqref{eq:B-v-v-flux} and \eqref{eq:Juplus} for $J(u_-,u_-)$ and $J(u_+,u_+)$, respectively. We therefore conclude that $| \widehat{a_+}(\lambda) |^2=1$. This completes the proof.
