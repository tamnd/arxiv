<!-- definition-1: body_md -->
A random vector $Y=(Y_i)_{i=1}^d$ has \emph{negatively associated}
(NA) coordinates if, for disjoint $I,J\subset[d]$ and bounded
coordinatewise increasing functions $f$ and $g$,
\[
 \mathbb E f(Y_I)g(Y_J)\le \mathbb E f(Y_I)\,\mathbb E g(Y_J).
\]
For nonnegative increasing functions the same inequality follows by
truncation whenever the expectations involved are finite.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
There exists a universal constant $c>0$ with the following property.
Let $X$ be a unconditional log-concave random vector in $\mathbb R^d$ and
assume that $(|X_i|)_{i=1}^d$ has negatively associated coordinates.
For every $P\ge1$, every finite $T\subset\mathbb R^d$ with $|T|\ge e^P$, and
every $a>0$ such that
\begin{equation}\label{eq:original-packing}
  \lVert \langle t-s,X\rangle\rVert_P\ge a
       \qquad(s,t\in T,\ s\ne t),
\end{equation}
one has
\begin{equation}\label{eq:main-smp}
  \mathcal W_X(T)\ge ca.
\end{equation}
The constant is independent of the dimension, the law, $P$, and $T$.
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:tail}
Let $Y$ be a log-concave random vector supported on $\mathbb R_+^d$, and set
\[
 Q_Y(v)=\mathbb P(Y\ge v),\qquad v\in\mathbb R_+^d,
\]
where inequalities between vectors are coordinatewise. The function
$Q_Y$ is log-concave and upper semicontinuous. In particular,
\begin{equation}\label{eq:tail-shrink}
 Q_Y(\theta v)\ge Q_Y(v)^\theta
       \qquad(0<\theta\le1).
\end{equation}
For $p,u>0$ and $A\subset[d]$, the set
\begin{equation}\label{eq:actual-body}
 K_A(p,u)=\left\{v\ge0:\ \operatorname{supp} v\subset A,\quad
           Q_Y(v)\ge e^{-p},\quad \lVert v\rVert_1\le u\right\}
\end{equation}
is nonempty, compact, convex, and downward closed in the positive orthant.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $v,w\ge0$ and $0<\theta<1$,
\[
 \theta(v+\mathbb R_+^d)+(1-\theta)(w+\mathbb R_+^d)
   =\theta v+(1-\theta)w+\mathbb R_+^d.
\]
Log-concavity of the measure gives log-concavity of $Q_Y$; unbounded
orthants can be treated by compact approximation. Since $Q_Y(0)=1$,
this also proves \eqref{eq:tail-shrink}.
If $v_n\to v$, then
$\limsup_n\mathbf 1_{\{Y\ge v_n\}}\le\mathbf 1_{\{Y\ge v\}}$ pointwise.
Reverse Fatou proves upper semicontinuity. The properties of
\eqref{eq:actual-body} now follow from its definition: the tail superlevel
set is closed and convex, decreasing thresholds can only increase its
probability, and the $\ell_1$ cap makes it bounded.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{input:bernoulli}
There is a universal $c_B>0$ such that, if $q\ge1$, a finite
$V\subset\mathbb R^d$ satisfies $|V|\ge e^q$, and
\[
 \lVert \sum_i(v_i-w_i)\varepsilon_i\rVert_q\ge b
          \qquad(v\ne w\text{ in }V),
\]
then
\[
 \mathbb E_\varepsilon\max_{v\in V}\left|\sum_i v_i\varepsilon_i\right|
       \ge c_B b.
\]
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{input:witness}
There is a universal $C_w\ge1$ such that the following holds.
Let $X$ be unconditional and log-concave, $P\ge1$, $|J|\le P$,
and $r_i\ge0$. There is $z\in\mathbb R_+^J$ with
\begin{equation}\label{eq:classical-witness}
 \mathbb P\bigl(|X_i|\ge z_i\text{ for all }i\in J\bigr)\ge e^{-P},
 \qquad
 \sum_{i\in J}r_i z_i
 \ge C_w^{-1}\lVert \sum_{i\in J}r_iX_i\rVert_P.
\end{equation}
<!-- end proposition-2 -->
<!-- proposition-3: body_md -->
\label{input:reduction}
Fix $\eta,\epsilon>0$. There are constants
$c_{\eta,\epsilon}>0$ and $P_0(\eta,\epsilon)<\infty$ such that, for
$P\ge P_0(\eta,\epsilon)$ and an unconditional log-concave $X$ satisfying
\eqref{eq:original-packing}, either
\begin{equation}\label{eq:easy-reduction-branch}
 \mathcal W_X(T)\ge c_{\eta,\epsilon}a,
\end{equation}
or there are $n=\lceil e^{P/4}\rceil$ distinct sets $A_t\subset[d]$
and weights $r_i\ge0$ such that
\begin{align}
 |A_t|&\le\eta P,\label{eq:reduction-support}\\
 \lVert \sum_i r_i(\mathbf 1_{A_t}(i)-\mathbf 1_{A_s}(i))X_i\rVert_P
       &\ge a/2\qquad(s\ne t),\label{eq:reduction-packing}\\
 M_{\rm bin}:=\mathbb E\max_t\left|\sum_{i\in A_t}r_iX_i\right|
       &\le2\mathcal W_X(T)+\epsilon a.\label{eq:reduction-transfer}
\end{align}

<!-- lemma-2: body_md -->
\label{lem:kraft}
For every admissible encoder and every $S\subset[d]$ with $|S|\le m$,
\begin{equation}\label{eq:kraft}
 \sum_{t\in T}\exp\bigl(-\tau_t^E(A_t\setminus S)\bigr)
       \le B_m,\qquad B_m=(m+1)4^m.
\end{equation}
In particular, $S$ need not be one of the encoded sub-supports.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Fix a length $k$. Further classify an ordered word $(i_1,\ldots,i_k)$
by the positions $J\subset[k]$ at which its coordinates belong to $S$,
and by the set $D\subset S$ of those coordinates. Necessarily
$|J|=|D|$. The ordering fixes which member of $D$ occupies each position
in $J$.

For one fixed pattern $(k,J,D)$, run the following probability
experiment. At a position in $J$, force the prescribed coordinate of $D$.
At any other position, sample according to the original prefix
conditional probabilities in the size-$k$ class. On a prefix absent
from the original tree, define arbitrary transition probabilities;
such a prefix cannot be extended into a valid word of that tree.

If a valid label $t$ has this pattern, the probability that this
experiment produces its word is
\[
 \prod_{j\notin J}\frac{n_j(t)}{n_{j-1}(t)}
   =\exp\bigl(-\tau_t^E(C_t\setminus S)\bigr).
\]
The accepted words are distinct, so their probabilities sum to at most
one. There are at most $2^k2^{|S|}$ choices of $(J,D)$ for each $k$,
and at most $m+1$ possible lengths. Summing over all patterns gives
$(m+1)4^m$. Finally, $\tau_t^E$ vanishes off $C_t\subset A_t$, so
$\tau_t^E(C_t\setminus S)=\tau_t^E(A_t\setminus S)$.

<!-- theorem-2: body_md -->
\label{thm:encoder}
Let $N=|T|\ge2$, $p=\log N$, and $|A_t|\le m$.
Let $K_t\subset\mathbb R_+^d$ be nonempty compact convex downward-closed
sets supported on $A_t$. Suppose $u>0$ and
\begin{equation*}
 \max\left\{
 h_{K_t}(\mathbf 1_{A_t\setminus A_s}),
 h_{K_s}(\mathbf 1_{A_s\setminus A_t})
 \right\}\ge u\qquad(s\ne t).
 \tag{H}\label{eq:H}
\end{equation*}
There is one probability law $\nu$ on admissible encoders such that
\begin{equation}\label{eq:common-law}
 a_t:=\frac{u}{2p}\mathbb E_\nu\tau_t^E\in K_t
                  \qquad\text{for every }t\in T.
\end{equation}
These vectors satisfy, for every $s\in T$,
\begin{equation}\label{eq:exponential-selection}
 \sum_{t\in T}
  \exp\left(-\frac{2p}{u}a_t(A_t\setminus A_s)\right)
       \le B_m.
\end{equation}
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
Condition \eqref{eq:H} forces the original $A_t$ to be distinct, so
the choice $C_t=A_t$ is admissible. There are finitely many admissible
families of sub-supports. Let
\[
 \mathcal C=\operatorname{conv}\left\{(\tau_t^E)_{t\in T}:E\text{ admissible}\right\},
 \qquad \mathcal K=\prod_{t\in T}K_t.
\]
Both are compact convex sets in the finite-dimensional space of
label-coordinate arrays. We prove
\begin{equation}\label{eq:intersection}
       \frac{u}{2p}\mathcal C\cap\mathcal K\ne\varnothing.
\end{equation}

Fix nonnegative costs $w_t$ on $A_t$, and put
\begin{equation}\label{eq:cheap-cores}
 H_t=h_{K_t}(w_t),\qquad
 C_t=\left\{i\in A_t:w_{t,i}\le\frac{2H_t}{u}\right\}.
\end{equation}
We first check that the $C_t$ are distinct. For a pair $s\ne t$,
condition \eqref{eq:H}, downward closure, and scaling down give either
a vector $v\in K_t$ of mass exactly $u$ supported on
$A_t\setminus A_s$, or the corresponding vector at endpoint $s$.
Consider the first case. Since $\langle w_t,v\rangle\le H_t$, the mass of $v$
outside $C_t$ is at most $u/2$ when $H_t>0$. If $H_t=0$, that mass
is zero. Hence
\[
   v(C_t\setminus A_s)\ge u/2>0.
\]
Thus $C_t\not\subset A_s$, whereas $C_s\subset A_s$, and $C_t\ne C_s$.
The other endpoint gives the same conclusion.

The family in \eqref{eq:cheap-cores} therefore defines an admissible
encoder $E$. Every coordinate used by label $t$ has cost at most
$2H_t/u$. By \eqref{eq:tau-budget},
\[
 \langle w_t,\tau_t^E\rangle\le\frac{2H_t}{u}\sum_i\tau_i^E(t)
                         \le\frac{2p}{u}H_t.
\]
It follows that, for every nonnegative cost array,
\begin{equation}\label{eq:dual-certificate}
 \frac{u}{2p}\min_{\tau\in\mathcal C}
     \sum_t\langle w_t,\tau_t\rangle
       \le\sum_t h_{K_t}(w_t)=h_{\mathcal K}(w).
\end{equation}

If \eqref{eq:intersection} failed, strict separation of the two compact
convex sets would give a linear functional $w$ with
\[
 \inf_{z\in(u/(2p))\mathcal C}\langle w,z\rangle>h_{\mathcal K}(w).
\]
All encoder arrays are nonnegative. Write $w_+$ for the
coordinatewise positive part of $w$. Downward closure of $\mathcal K$ gives
$h_{\mathcal K}(w)=h_{\mathcal K}(w_+)$: an optimizer for $w_+$ may have all
coordinates with negative $w$ set to zero. Replacing $w$ by $w_+$ can
only increase the infimum on the left. This contradicts
\eqref{eq:dual-certificate} and proves \eqref{eq:intersection}.

A point of $\mathcal C$ is the average of whole encoder arrays under one
finite probability law $\nu$. Thus \eqref{eq:intersection} is precisely
\eqref{eq:common-law}. Finally, Jensen's inequality and
\cref{lem:kraft}, used with $S=A_s$, give
\begin{align*}
 \sum_t e^{-(2p/u)a_t(A_t\setminus A_s)}
 &=\sum_t e^{-\mathbb E_\nu\tau_t^E(A_t\setminus A_s)}\\
 &\le\mathbb E_\nu\sum_t e^{-\tau_t^E(A_t\setminus A_s)}
 \le B_m.
\end{align*}

\label{prop:positive-count}
Let $F(Y)=\max_t\sum_{i\in A_t}Y_i$. Then
\begin{equation}\label{eq:positive-count}
 \mathbb E F(Y)\ge\beta(1-\Lambda^{-1})_+.
\end{equation}

<!-- proof-5: body_md -->
For a fixed $t$, put
\[
 F_t(Y)=\max_{s\ne t}\sum_{i\in A_s\setminus A_t}Y_i.
\]
This is increasing and depends only on $A_t^c=[d]\setminus A_t$,
while $B_t$ is
increasing and depends only on $A_t$. Also $F_t\le F$. Thus NA gives
\[
 \mathbb E[\mathbf 1_{B_t}F_t]\le q_t\mathbb E F_t\le q_t\mathbb E F.

\]
The assertion follows, with the case $\mathbb E F=\infty$ understood trivially.

<!-- lemma-3: body_md -->
\label{lem:signed-count}
Assume \eqref{eq:b-separation}, $|A_t|\le m$, and $m\ge1$. Let
\[
 M=\mathbb E_{Y,\varepsilon}\max_t
       \left|\sum_{i\in A_t}\varepsilon_iY_i\right|,
 \qquad L_m=2^m\lceil e^{2m}\rceil,
\]
where the signs are independent of $Y$. With the constant $c_B$ in
\cref{input:bernoulli},
\begin{equation}\label{eq:signed-count}
 M\ge c_B\beta\left(1-\frac{L_m}{\Lambda}\right)_+.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
We may assume $M<\infty$. For each fixed reference label $t$, define
\begin{equation}\label{eq:G-definition}
 G_t(y)=\mathbb E_\varepsilon\max_{s\in T}
       \left|\sum_{i\in A_s\setminus A_t}\varepsilon_i y_i\right|.
\end{equation}
As a function of one coordinate, this expectation is convex and even,
and hence nondecreasing on $[0,\infty)$. Thus $G_t$ is increasing on
the positive orthant and depends only on coordinates outside $A_t$.
Moreover, averaging first over the signs in $A_t$ and using convexity
of the maximum of absolute values shows, for every $y\ge0$, that
\[
 G_t(y)\le\mathbb E_\varepsilon\max_s
               \left|\sum_{i\in A_s}\varepsilon_i y_i\right|.
\]
Consequently, NA gives
\begin{equation}\label{eq:NA-G}
 \mathbb E[\mathbf 1_{B_t}G_t(Y)]\le q_t\mathbb E G_t(Y)\le q_tM.
\end{equation}
For an unbounded $G_t$, apply NA to $G_t\wedge R$ and let $R\to\infty$.

We claim that
\begin{equation}\label{eq:many-active}
 \mathcal N(y)>L_m\quad\Longrightarrow\quad
       G_t(y)\ge c_B\beta\quad\text{for every }t\in T.
\end{equation}
Fix such $y$ and a reference $t$. Partition the active labels
$\{s:y\in B_s\}$ according to $A_s\cap A_t$. There are at most
$2^{|A_t|}\le2^m$ classes. Some class $\mathcal S$ therefore has
more than $e^{2m}$ labels.

For $s\in\mathcal S$ define the deterministic vector
$v_s=y\mathbf 1_{A_s\setminus A_t}$. If $r,s\in\mathcal S$ are distinct,
their supports agree inside $A_t$, so
\begin{align}
 \lVert v_s-v_r\rVert_1
 &=y(A_s\setminus A_r)+y(A_r\setminus A_s)\notag\\
 &\ge b_s(A_s\setminus A_r)+b_r(A_r\setminus A_s)
 \ge2\beta.\label{eq:active-l1}
\end{align}
The difference has at most $2m$ nonzero coordinates. The event on which
every relevant Rademacher sign agrees with its coefficient therefore
gives
\begin{equation}\label{eq:bernoulli-sparse-moment}
 \lVert \sum_i(v_s-v_r)_i\varepsilon_i\rVert_{2m}
 \ge2^{-\lvert \operatorname{supp}(v_s-v_r)\rvert/(2m)}\lVert v_s-v_r\rVert_1
 \ge\beta.
\end{equation}
The vectors are distinct by \eqref{eq:active-l1}, and
$|\mathcal S|>e^{2m}$. Applying \cref{input:bernoulli} at $q=2m$
proves \eqref{eq:many-active}.

Now sum \eqref{eq:NA-G}. By \eqref{eq:many-active},
\begin{align}
 \Lambda=\mathbb E\mathcal N
 &\le L_m+\mathbb E[\mathcal N\mathbf 1_{\{\mathcal N>L_m\}}]\notag\\
 &\le L_m+\frac1{c_B\beta}\sum_t
                 \mathbb E[\mathbf 1_{B_t}G_t(Y)]\notag\\
 &\le L_m+\frac{M}{c_B\beta}\Lambda.\label{eq:count-master}
\end{align}
Rearrangement gives \eqref{eq:signed-count}.

<!-- corollary-2: body_md -->
\label{cor:orlicz}
Let $I_1,\ldots,I_k$ be a partition of $[d]$. For each $i$ let $\phi_i$
be a Young function, and for each $j$ let
$h_j:[0,\infty)\to[0,\infty)$ be log-concave and nonincreasing.
Assume that
\begin{equation}\label{eq:orlicz-density}
 f(x)=\frac1Z\prod_{j=1}^k
       h_j\!\left(\sum_{i\in I_j}\phi_i(|x_i|)\right)
\end{equation}
is a probability density, with $0<Z<\infty$ and $h_j(\infty)=0$.
Then a vector with density $f$ satisfies the conclusion of
\cref{thm:main}, with the same universal constant, independent of
$k$, the block dimensions, and the defining functions. In particular,
the conclusion holds for the uniform measure on every Cartesian
product of generalized Orlicz balls.
<!-- end corollary-2 -->
<!-- proof-9: body_md -->
Each factor in \eqref{eq:orlicz-density} is unconditional and
log-concave. Indeed, $-\log h_j$ is convex and nondecreasing on its
effective domain, so its composition with
$\sum_{i\in I_j}\phi_i(|x_i|)$ is convex.
Wojtaszczyk's theorem~\cite[Theorem~1.3]{Wojtaszczyk} gives NA of the
magnitudes in each block. The blocks are independent because the
density factors.

For completeness, independent unions of NA families are NA. For two
independent blocks $U,V$ and increasing functions $F,G$ depending on
disjoint coordinate sets, fix $V$ and apply NA in $U$. This bounds
$\mathbb E_U(FG)$ by $(\mathbb E_U F)(\mathbb E_U G)$. The two expectations are increasing
functions of disjoint coordinate sets of $V$, so NA in $V$ gives
$\mathbb E(FG)\le\mathbb E F\,\mathbb E G$. Iteration proves the assertion for all
blocks. Thus \cref{thm:main} applies.

Taking $h_j=\mathbf 1_{[0,R_j]}$ gives the uniform law on
\[
 \prod_{j=1}^k
 \left\{x_{I_j}:\sum_{i\in I_j}\phi_i(|x_i|)\le R_j\right\},
\]
whenever these sets are convex bodies.

\label{cor:covering}
Let $X$ satisfy the assumptions of \cref{thm:main}. There is a universal
$C>0$ such that, for every bounded nonempty $T\subset\mathbb R^d$ and every
$p\ge1$,
\begin{equation}\label{eq:moment-covering}
 N\bigl(T,d_p,C\mathcal W_X(T)\bigr)<e^p.
\end{equation}
In particular, suppose $X$ is nondegenerate and define its symmetric
$L_p$ centroid body by
\[
 h_{Z_p(X)}(t)=\lVert \langle t,X\rangle\rVert_p.
\]
For every origin-symmetric convex body $K\subset\mathbb R^d$,
\begin{equation}\label{eq:centroid-covering}
 N\bigl(K,C\mathbb E h_K(X)\,Z_p(X)^\circ\bigr)<e^p,
\end{equation}
where $N(K,L)$ is the least number of translates of $L$ covering $K$,
and $L^\circ$ denotes the polar body. Consequently, for every norm
on $\mathbb R^d$ with dual unit ball $B^*$,
\begin{equation}\label{eq:norm-covering}
 N\bigl(B^*,C\mathbb E\lVert X\rVert\,Z_p(X)^\circ\bigr)<e^p.
\end{equation}
<!-- end corollary-3 -->
<!-- proof-10: body_md -->
Let $c$ be the constant in \cref{thm:main}, and take $C=2/c$.
First suppose $W:=\mathcal W_X(T)>0$ and set $r=CW$. If a subset
$S\subset T$ of size $\lceil e^p\rceil$ had all its pairwise
$d_p$-distances greater than $r$, then \cref{thm:main} would give
\[
 W\ge\mathcal W_X(S)\ge cr=2W,
\]
a contradiction. A greedy choice of points at distance greater than
$r$ from all previously chosen points must therefore stop after
fewer than $e^p$ points. At that time the closed balls of radius $r$
cover $T$, proving \eqref{eq:moment-covering}.

If $W=0$, centering and symmetry imply, for every $s,t\in T$,
\[
 \mathbb E|\langle t-s,X\rangle|
   =2\mathbb E\max\{\langle t,X\rangle,\langle s,X\rangle\}\le2W=0.
\]
Thus $d_p(s,t)=0$ and one ball of radius zero suffices.
Finally, the unit ball of $t\mapsto\lVert \langle t,X\rangle\rVert_p$ is exactly
$Z_p(X)^\circ$. Apply \eqref{eq:moment-covering} to $T=K$ to obtain
\eqref{eq:centroid-covering}. Taking $K=B^*$ and using
$h_{B^*}(x)=\lVert x\rVert$ gives \eqref{eq:norm-covering}.
