We start with the case $\mathrm{T} = \mathrm{L}$.
We have
$$
\mathrm{L}_{hg}=\mathrm{L}_h\circ \mathrm{L}_g
$$
and
$$
\mathrm{L}_{(hg)^{-1}}\circ \mathrm{L}_h
=
\mathrm{L}_{g^{-1}h^{-1}}\circ \mathrm{L}_h
=
\mathrm{L}_{g^{-1}}.
$$
Hence, by the chain rule,

$$
\textnormal{d}_{hg}\mathrm{L}_{(hg)^{-1}}\circ\textnormal{d}_g\mathrm{L}_h
=
\textnormal{d}_g\mathrm{L}_{g^{-1}},
$$
and therefore
$$
\begin{aligned}
\operatorname{Exp}_{hg}\bigl(\textnormal{d}_g\mathrm{L}_h(v)\bigr)
&=
hg\exp\left(
\textnormal{d}_{hg}\mathrm{L}_{(hg)^{-1}}
\bigl(\textnormal{d}_g\mathrm{L}_h(v)\bigr)
\right)\\
&=
hg\exp\left(\textnormal{d}_g\mathrm{L}_{g^{-1}}(v)\right)\\
&=
h\operatorname{Exp}_g(v).
\end{aligned}
$$

The proof works analogously for right translations.

The proof proceeds by induction over $n$. By assumption,
$$
\tilde{\mathbf g}_0=\mathrm{T}_h(\mathbf g_0),
$$
so the assertion holds for $n=0$.

Suppose that
$$
\tilde{\mathbf g}_n=\mathrm{T}_h(\mathbf g_n)
$$
for some $n$. The update vector $\mathbf v$ is equivariant:
$$
\mathbf v\bigl(\mathrm{T}_h(\mathbf g_n),\mathrm{T}_h(\mathbf f),\mathbf t\bigr)
=
\textnormal{d}_{\mathbf g_n}\mathrm{T}_h
\bigl(
\mathbf v(\mathbf g_n,\mathbf f,\mathbf t)
\bigr);
$$
see the proof of~\cite[Theorem 1]{Schade2025Biinvariant} for a derivation.

Consequently,
$$
\begin{aligned}
\tilde{\mathbf g}_{n+1}
&=
\operatorname{Exp}_{\tilde{\mathbf g}_n}
\left(
\lambda\,
\mathbf v(\tilde{\mathbf g}_n,\mathrm{T}_h(\mathbf f),\mathbf t)
\right)
\\
&=
\operatorname{Exp}_{\mathrm{T}_h(\mathbf g_n)}
\left(
\lambda\,
\textnormal{d}_{\mathbf g_n}\mathrm{T}_h
\bigl(
\mathbf v(\mathbf g_n,\mathbf f,\mathbf t)
\bigr)
\right)
\\
&=
\mathrm{T}_h
\left(
\operatorname{Exp}_{\mathbf g_n}
\left(
\lambda\,
\mathbf v(\mathbf g_n,\mathbf f,\mathbf t)
\right)
\right)
\\
&=
\mathrm{T}_h(\mathbf g_{n+1}),
\end{aligned}
$$

where the third equality follows from Lemma~\ref{lem:equivariance_exp} because $\operatorname{Exp}$ works component-wise. Thus the assertion holds for $n+1$, completing the induction.

Note that
    \begin{equation*}
        \gamma_{t,g} = \mathrm{L}_g \circ \exp \circ\, t\log \circ\, \mathrm{L}_{g^{-1}}.
    \end{equation*}
    Thus, due to the chain rule,
    \begin{equation*}
        \partial_1 \gamma_t (g,f) = \textnormal{d}_{\exp(t\log(g^{-1}f))} \mathrm{L}_g \circ \textnormal{d}_{t\log(g^{-1}f)} \exp \circ\, t\, \textnormal{d}_{g^{-1}f}\log \circ\, \textnormal{d}_f\mathrm{L}_{g^{-1}}.
    \end{equation*}
    It follows that
    \begin{align*}
        \bigl( \partial_1 \gamma_t (g,f) \bigr)^{-1} &= \bigl( \textnormal{d}_{\exp(t\log(g^{-1}f))} \mathrm{L}_g \circ \textnormal{d}_{t\log(g^{-1}f)} \exp \circ\, t\, \textnormal{d}_{g^{-1}f}\log \circ\, \textnormal{d}_f\mathrm{L}_{g^{-1}} \bigr)^{-1} \\
        &= \bigl( \textnormal{d}_f\mathrm{L}_{g^{-1}} \bigr)^{-1} \,\circ\, \frac{1}{t}\, \bigl( \textnormal{d}_{g^{-1}f}\log \bigr)^{-1}  \\ 
        &\qquad \qquad \circ \bigl( \textnormal{d}_{t\log(g^{-1}f)} \exp \bigr)^{-1} \,\circ\, \bigl( \textnormal{d}_{\exp(t\log(g^{-1}f))} \mathrm{L}_g \bigr)^{-1} \\
        &= \textnormal{d}_{g^{-1}f}\mathrm{L}_g \,\circ\, \frac{1}{t}\, \textnormal{d}_{\log(g^{-1}f)}\exp \\ 
        &\qquad \qquad \circ\, \textnormal{d}_{\exp(t\log(g^{-1}f))} \log \,\circ\, \textnormal{d}_{g\exp(t\log(g^{-1}f))} \mathrm{L}_{g^{-1}}.
    \end{align*}
    
    When $\rho(\textnormal{ad}_{\log{(g^{-1}f})}) < 2\pi$, Equations~\eqref{eq:derivative_exp} and \eqref{eq:derivative_log} and the bilinearity of ad yield
    \begin{align*}
    \bigl( \partial_1 \gamma_t (g,f) \bigr)^{-1} =\
        &\textnormal{d}_{g^{-1}f}\mathrm{L}_g \\
        &\circ \frac{1}{t}\; \textnormal{d}_e \mathrm{L}_{g^{-1}f} \\ &\circ \sum_{n=0}^\infty \frac{(-1)^n \textnormal{ad}_{\log(g^{-1}f)}^n}{(n+1)!} \\
        &\circ \sum_{k=0}^\infty \frac{(-1)^k t^k B_k \textnormal{ad}_{\log(g^{-1}f)}^k}{k!} \\
        &\circ \textnormal{d}_{\exp(t\log(g^{-1}f))} \mathrm{L}_{\exp(t\log(g^{-1}f))^{-1}} \\
        &\circ\, \textnormal{d}_{g\exp(t\log(g^{-1}f))} \mathrm{L}_{g^{-1}}.
    \end{align*}

Lemma~\ref{lem:Psi_alt} ensures that there is a neighborhood $W_{\mathbf x} \times W_{\mathbf y} \times [0, \lambda_0]$ on which
\begin{align} \label{eq:psi_difference}
    \Psi_{\mathbf y}(\mathbf x, \lambda) - \Psi_{\mathbf y}(\widetilde{\mathbf x}, \lambda) = A_\lambda (\mathbf x - \widetilde{\mathbf x}) + \lambda \big( \mathcal R(\mathbf x,\mathbf y;\lambda) - \mathcal R(\widetilde{\mathbf x},\mathbf y;\lambda) \bigr).
\end{align}
for $\mathbf x, \widetilde{\mathbf x} \in W_{\mathbf x}$.

If $\lambda < 1/\mu_{\textnormal{max}}$ We know from Lemma~\eqref{eq:norm_A} that $A_\lambda$ is a contraction with contraction factor $1 - \lambda \mu_{\textnormal{min}} < 1$, that is,
\begin{align} \label{eq:contraction_A}
\| A_\lambda (\mathbf x - \widetilde{\mathbf x}) \|_2 \le (1 - \lambda \mu_{\textnormal{min}}) \| \mathbf x - \widetilde{\mathbf x} \|_2.
\end{align}
To combine \eqref{eq:psi_difference} and \eqref{eq:contraction_A}, we must assume $0 < \lambda \le \lambda^* = \textnormal{min}(\lambda_0, 1/\mu_{\textnormal{max}})$.

It remains to bound the rest.
Lemma~\ref{lem:dR} gives 
\begin{equation*} \label{eq:dr_bound}
        \|\partial_0\mathcal R(\mathbf x,\mathbf y;\lambda)\|_2
    \leq C'(\|\mathbf x\|_2+\|\mathbf y\|_2)
\end{equation*}
on a smaller compact neighborhood $\mathcal{C}_\mathbf x \times \mathcal{C}_\mathbf y \times [0, \lambda^*]$.
The constant~$C'$ can be chosen uniformly for $\lambda\in[0,\lambda^*]$.\footnote{We label the constant $C'$ to distinguish it from the constant that appeard in the proof of Lemma~\ref{lem:selfmap}.} This uniformity is important because the subsequent contraction estimate is required to hold simultaneously for all sufficiently small stepsizes.

From our assumptions, we now get
\begin{align} 
    \|\partial_0 \mathcal R(\mathbf x,\mathbf y;\lambda)\|_2 \le C'(R + \sqrt{N}r), \label{eq:bound_D1_R}
\end{align}
where $r > 0$ is sufficiently small such that $\mathbf x \in \mathcal{C}_\mathbf x$ and $\mathbf y \in \mathcal{C}_{\mathbf y}$.

We now want to bound 
$$\mathcal{R}(\mathbf x, \mathbf y;\lambda) - \mathcal{R}(\widetilde{\mathbf x}, \mathbf y;\lambda)$$
on $\mathcal{C}_x$.
Let 
$$\ell(s) = \widetilde{\mathbf x} + s(\mathbf x - \widetilde{\mathbf x}), \quad s \in [0,1]$$
be the straight line between $\mathbf x$ and $\widetilde{\mathbf x}$. It does not leave the domain because $\mathcal{C}_\mathbf x$ is convex.
Set 
$$F(s) = \mathcal{R}(\ell(s), \mathbf y; \lambda).$$
Then,
$$F(1) - F(0) = \mathcal{R}(\mathbf x,\mathbf y;\lambda) - \mathcal{R}(\widetilde{\mathbf x},\mathbf y;\lambda).$$
We also have
$$F(1) - F(0) = \int_0^1 F'(s) \textnormal{d} s.$$
Because 
$$F'(s) = \partial_0 \mathcal{R}(\ell(s), \mathbf y;\lambda)(\mathbf x - \widetilde{\mathbf x}),$$
it follows that
$$\mathcal{R}(\mathbf x,\mathbf y;\lambda) - \mathcal{R}(\widetilde{\mathbf x},\mathbf y;\lambda) = \int_0^1 \partial_0 \mathcal{R}(\ell(s), \mathbf y;\lambda)(\mathbf x - \widetilde{\mathbf x}) \textnormal{d} s.$$
Consequently,
$$\| \mathcal{R}(\mathbf x,\mathbf y;\lambda) - \mathcal{R}(\widetilde{\mathbf x},\mathbf y;\lambda) \|_2 \le \int_0^1 \| \partial_0 \mathcal{R}(\ell(s), y;\lambda) \|_2 \; \|(\mathbf x - \widetilde{\mathbf x})\|_2 \textnormal{d} s.$$
Equation~\eqref{eq:bound_D1_R} then yields
\begin{align}
    \left\| \mathcal{R}(\mathbf x,\mathbf y;\lambda) - \mathcal{R}(\widetilde{\mathbf x},\mathbf y;\lambda) \right\|_2 &\le \int_0^1 C' (R + \sqrt{N}r) \|(\mathbf x - \widetilde{\mathbf x})\|_2 \textnormal{d} s \nonumber \\
    &= C'(R + \sqrt{N}r) \|(\mathbf x - \widetilde{\mathbf x})\|_2. \label{eq:lip_R}
\end{align}
on $\mathcal{C}_x \times \mathcal{C}_y \times [0, \lambda^*]$.
\bigskip

We can finally investigate 
$$\Psi_{\mathbf y}(\mathbf x,\lambda) - \Psi_{\mathbf y}(\widetilde{\mathbf x},\lambda).$$
Equation \eqref{eq:psi_difference} yields
\begin{align*}
    \| \Psi_{\mathbf y}(\mathbf x, \lambda) - \Psi_{\mathbf y}(\widetilde{\mathbf x}, \lambda) \|_2 &\le \| A_\lambda(\mathbf x - \widetilde{\mathbf x}) \|_2 + \lambda \| \mathcal R(\mathbf x,\mathbf y;\lambda) - \mathcal R(\widetilde{\mathbf x},\mathbf y;\lambda) \|_2
\end{align*}
on $W_{\mathbf x} \times W_{\mathbf y} \times [0, \lambda_0]$.
Applying \eqref{eq:contraction_A} and \eqref{eq:lip_R} to this inequality gives
\begin{align*}
    \| \Psi_{\mathbf y}(\mathbf x, \lambda) - \Psi_{\mathbf y}(\widetilde{\mathbf x}, \lambda) \|_2 &\le \bigl( 1 - \lambda \mu_{\textnormal{min}}  + \lambda C'(R + \sqrt{N}r) \bigr) \| \mathbf x - \widetilde{\mathbf x} \|_2.
\end{align*}
on the (possibly smaller) domain $\mathcal{C}_x \times \mathcal{C}_y \times [0, \lambda^*]$.
Consequently,
$\Psi_{\mathbf y}$ is a contraction on $B^2_R(0) \subseteq \mathcal{C}_x$ when 
$$  1 - \lambda \mu_{\textnormal{min}}  + \lambda C'(R + \sqrt{N}r) < 1,$$
which is equivalent to
$$ C' \left(R + \sqrt{N} r \right) < \mu_{\textnormal{min}}.$$
Using $R = \alpha r$, we obtain
$$ C' \left(\alpha + \sqrt{N} \right) r < \mu_{\textnormal{min}},$$
or, equivalently,
$$ r < \frac{\mu_{\textnormal{min}}}{C' \left(\alpha + \sqrt{N} \right)}.$$
With $\widetilde r > 0$ fulfilling $B^2_{\alpha \widetilde r}(0) \subseteq \mathcal{C}_x$ as well as $\{(y_1,\dots,y_N) \in (T_eG)^N: \textnormal{max}(\| y_i \|) \le r \} \subseteq \mathcal{C}_y$ and $\epsilon << 1$ set 
$$r_c(\alpha) := (1- \epsilon) \; \textnormal{min}\left(\frac{\mu_{\textnormal{min}}}{C' \left(\alpha + \sqrt{N} \right)}, \widetilde r \right).$$
Then, $\Psi_\mathbf y$ is a contraction on $B^2_{\alpha r}(0)$ for all $0 < r \le r_c(\alpha)$.

Combining Lemmas \ref{lem:selfmap} and \ref{lem:contaction}, we see that under our assumptions there exists a positive number $r^*(\alpha) := \textnormal{min}(r_c(\alpha), r_s(\alpha)) > 0$ such that for all $0 < \lambda \le \lambda^*$ the update map
    $\Psi_{\mathbf y}$ is a contractive self-map on $B^2_{\alpha r}(0) \subseteq V^2$.
    In this case, Banach's Fixed Point Theorem applies: $\Psi_{\mathbf y}$ has a unique fixed point $\hat \mathbf x \in B^2_{\alpha r}(0)$.
    But since $\operatorname{Exp}$ is invertible in $U$ and $\lambda > 0$, this is only possible if 
    $$\mathbf v(\exp(\hat \mathbf x), \mathbf f, \mathbf t) = 0.$$ Therefore, $\hat \mathbf g := \exp(\hat \mathbf x)$ consists of the starting and end points of a bi-invariant estimator for geodesic regression for the given data. 
    
    Banach's theorem also implies that the sequence $\mathbf x_{n+1}:= \Psi_\mathbf y(\mathbf x_n, \lambda)$ converges at least with linear speed to $\hat \mathbf x$ for all initial values in $B^2_{\alpha r}(0)$. 
    Since the exponential map is smooth, it is Lipschitz on the compact neighborhood under consideration; hence linear convergence in logarithmic coordinates implies linear convergence of the corresponding group-valued iterates.
    Therefore, the sequence $\mathbf g_{n}:= \exp(\mathbf x_n)$ provided by Algorithm~\ref{alg:algorithm} converges at least with linear speed to $\hat \mathbf g$ for all initial values in $\exp(B^2_{\alpha r}(0))$.
