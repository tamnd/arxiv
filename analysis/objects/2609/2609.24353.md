<!-- lemma-1: body_md -->
\label{lem:partite-matching}
		For every $\ell>k\ge2$, there is $\varepsilon_0=\varepsilon_0(k,\ell)>0$ such that
		the following holds for all sufficiently large $n$. Let
		$0<\delta\le\varepsilon\le\varepsilon_0$, and let $G$ be an $\ell$-partite $k$-graph with
		vertex set $V_1\cup\cdots\cup V_\ell$ such that
		$(1+\delta)n\ge |V_\ell|\ge\cdots\ge|V_1|\ge n$.
		If
		$d_G(v)>(\binom{\ell-1}{k-1}-\varepsilon) n^{k-1}$ for every $v\in V(G)$,
		then $G$ has a matching covering all but at most $k-1$ vertices.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $K$ be the complete $\ell$-partite $k$-graph with parts $V_1, \cdots, V_\ell$. Then $G$ is a subgraph of $K$, and for every vertex $v$ of $K$, $d_K(v)\le\binom{\ell-1}{k-1}(1+\delta)^{k-1}n^{k-1}$. Since $(1+\delta)^{k-1}\leq 1+k\delta$ for sufficiently small $\delta$, we have
		\begin{equation}\label{eq:missing-link}
			|N_K(v)\setminus N_G(v)|=d_K(v)-d_G(v)\leq (k\delta\binom{\ell-1}{k-1}+\varepsilon)n^{k-1}
			\le 2k\binom{\ell-1}{k-1}\varepsilon n^{k-1}.
		\end{equation}
		
		
		Suppose, to the contrary, that $\nu(G)<\lfloor\frac{|V(G)|}{k}\rfloor$. Since $K$ has a matching of size $\lfloor\frac{\ell n}{k}\rfloor$,  we may assume that $\nu(G)\geq \lfloor\frac{ (k+1)n}{k}\rfloor-1> (1+\delta)n$ by adding edges if necessary.
		Choose a maximum matching $M$ of $G$ for which the number of parts containing
		uncovered vertices is as large as possible. Since $|M|> n$, there are $k$ parts such that $|M\cap E(K[V_{j_1}\cup\cdots\cup V_{j_k}])|> \frac{n}{\binom\ell k}$
		for all sufficiently large $n$. Let $M'=M\cap E(K[V_{j_1}\cup\cdots\cup V_{j_k}])$. 
		
		                                      
		\vspace{0.15cm}	
		{\bf Claim.}
		If $M^*\subseteq M'$ and
		$|M^*|\ge n/(2\binom{\ell}{k})$, then for every legal
		$k$-set $T$ disjoint from $V(M^*)$, $G[V(M^*)\cup T]$ has a matching of size $|M^*|+1$.
		            
		\vspace{0.15cm}	
		
		We now prove this claim. Write
		$T=\{v_{a_1},\ldots,v_{a_k}\}$, where
		$v_{a_i}\in V_{a_i}$ for $1\leq i\leq k$. For
		$M''\in\binom{M^*}{k-1}$, let
		\[
		\mathcal T(M'')
		=
		\left\{
		S\subseteq V(M''): S \text{ is legal and }
		|S\cap e|=1\text{ for every }e\in M''
		\right\}.
		\]
		We say that $M''$ is a \emph{$(k-1)$-neighbour} of
		$v_{a_i}$ if
		$\{v_{a_i}\}\cup S\in E(G)$
		for every $S\in\mathcal T(M'')$ with
		$S\cap V_{a_i}=\varnothing$. Denote the family of all
		$(k-1)$-neighbours of $v_{a_i}$ by $\mathcal M_{a_i}$.
		Note that for each $M''\in\binom{M^*}{k-1}\setminus\mathcal M_{a_i}$, there exists $S\in\mathcal T(M'')$ such that $S\in N_K(v_{a_i})\setminus N_G(v_{a_i})$.
		Since $M^*$ is a matching and $M''$ consists precisely of the edges of $M^*$ intersecting $S$, we have 
		$\left|
		\binom{M^*}{k-1}\setminus\mathcal M_{a_i}
		\right|
		\le 2k\binom{\ell-1}{k-1}\varepsilon n^{k-1}$
		by \eqref{eq:missing-link}.
		Consequently,
		$|\mathcal M_{a_i}|
		\ge
		\binom{|M^*|}{k-1}-2k\binom{\ell-1}{k-1}\varepsilon n^{k-1}$.
		
		Choose $M''\in\binom{M^*}{k-1}$ uniformly at random. Since $|M^*|\ge \frac{n}{2\binom{\ell}{k}}$, $n$ is sufficiently large  and $\varepsilon$  is sufficiently small, by the union bound,
		\begin{equation}
			\begin{aligned}
				\mathbb P\left(M''\in\bigcap_{i=1}^{k}\mathcal M_{a_i}\right)
				\ge 1-\sum_{i=1}^{k}\mathbb P(M''\notin\mathcal M_{a_i})
				\ge 1-\frac{2k^2\binom{\ell-1}{k-1}\varepsilon n^{k-1}}{\binom{|M^*|}{k-1}}>0.
			\end{aligned}
		\end{equation}
		Thus there exists
		an	$M_0\in\bigcap_{i=1}^{k}\mathcal M_{a_i}$.
		Write
		$M_0=\{e_1,\ldots,e_{k-1}\}$ and
		$e_t=\{x_{t,1},\ldots,x_{t,k}\}$, where
		$x_{t,i}\in V_{j_i}$ for $i\in [k]$. Since $T$ is legal, there is a bijection $\pi:T\to[k]$
		such that $\pi(u)=i$ whenever $u\in V_{j_i}$, $i\in[k]$. For each $i\in[k]$, define
		\[
		f_i=
		\{x_{1,i+1},x_{2,i+2},\ldots,x_{k-1,i+k-1}\},
		\]
		where the second subscripts are modulo $k$.
		Then $\{f_1,\ldots,f_k\}$ is a partition of $V(M_0)$ and $f_i\cap V_{j_i}=\varnothing$ for every
		$i\in[k]$. 
		For every $u\in T$, since $M_0$ is a $(k-1)$-neighbour of $u$, we have	$\{u\}\cup f_{\pi(u)}\in E(G)$.
		Therefore, $\{\{u\}\cup f_{\pi(u)}:u\in T\}$
		is a perfect matching of size $k$ in $G[V(M_0)\cup T]$. It follows that $\{\{u\}\cup f_{\pi(u)}:u\in T\}\cup (M^*\setminus M_0)$ is a matching of size $|M^*|+1$ in $G[V(M^*)\cup T]$. This proves the Claim.
		                                                        
		                                                               
		                                                                             
		                                                      
		                                                       
		                                                                              
		                                                  
		                                                                       
		
		
		                              
		                                                                            
		                                                                               
		                                                      
		                                              
		                                            
		                                                                          
		                                                                        
		                                                                     
		                                     
		
		Let $\mathcal V$ be the set of parts  containing  vertices uncovered by $M$. If
		$|\mathcal V|\ge k$, choose an uncovered legal $k$-set $T$. Applying the
		Claim  with $M^*=M'$, we can  find a matching of size $|M|+1$. Then we get a contradiction to the choice of $M$. Thus
			$|\mathcal V|\le k-1$.
		
		Since $|M|\leq\left\lfloor |V(G)|/k\right\rfloor-1$, at least $k$ vertices are
		uncovered by $M$. By $|\mathcal V|\le k-1$, some $V_t\in\mathcal V$
		contains two uncovered vertices. Fix one of them, say $y$. Recall that
		$|M|>(1+\delta)n\geq |V_t|$.  Thus there is an edge
		$e_0\in M$ avoiding $V_t$. As every edge intersects $k$ distinct parts and
		$|\mathcal V|\le k-1$, the edge $e_0$ intersects a part $V_i\notin\mathcal V$.
		Fix $x\in e_0\cap V_i$ and set $T_0=(e_0\setminus\{x\})\cup\{y\}.$
		This is a legal $k$-set disjoint from
		$V(M'\setminus\{e_0\})$. Applying the claim with $M^*=M'\setminus\{e_0\}$ and $T=T_0$,
		we obtain a matching $M_1$ of size $|M'\setminus\{e_0\}|+1$
		in $G[V(M'\setminus\{e_0\})\cup T_0]$.
		Note that $M_1$ covers every vertex of
		$V(M'\setminus\{e_0\})\cup T_0$.
		Let $M_2=M_1\cup\bigl(M\setminus(M'\cup\{e_0\})\bigr)$.
		Then $M_2$ is a matching with $|M_2|=|M|$ and
		$V(M_2)=(V(M)\setminus\{x\})\cup\{y\}$.
		Thus the number of parts
		containing vertices uncovered by $M_2$ is greater than
		the corresponding number for $M$, contradicting the choice
		of $M$. This completes the proof.

<!-- lemma-4: body_md -->
\label{lem:close-reduction}
		Let $k,\ell$ be integers such that $\ell>k\ge2$ and $\mathcal F$ be a finite family
		of $k$-graphs. There exist $\varepsilon,\beta>0$ and a sufficiently large integer $s_0$ such that the following holds. Let $n,s$ be
		integers with $\max\{s_0,n/k-\beta n\}<s<n/k$. If an $n$-vertex $k$-graph
		$G$ is $\mathcal F$-free, satisfies $\nu(G)\le s$, and is
		$\varepsilon$-close to $T_k(n,\ell)$, then
		$|E(G)|\le\mathrm{ex}_k(ks+k-1,\mathcal F)$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Since $G$ is $\varepsilon$-close to $T_k(n,\ell)$, there is a
		copy $T$ of $T_k(n,\ell)$ defined on $V(G)$ such that
		\[
		|E(G)\setminus E(T)|+|E(T)\setminus E(G)|
		\leq \varepsilon n^k.
		\]
		Let $X=\{x\in V(G):|N_T(x)\setminus N_G(x)|
		>\sqrt{\varepsilon}n^{k-1}\}.$
		We claim that $|X|\leq k\sqrt{\varepsilon}n$, otherwise
		\[
		|E(T)\setminus E(G)|
		\geq \frac{1}{k}\sum_{x\in X}|N_T(x)\setminus N_G(x)|
		>\varepsilon n^k,
		\]
		a contradiction.
		
		Let $U=V(G)\setminus X$ and let $H$ be the $k$-graph defined on $U$ with edge
		set $E(G[U])\cap E(T[U])$. Then $H$ is an $\ell$-partite $k$-graph.
		Denote the vertex partition of $H$ by $\{U_1,U_2,\ldots,U_\ell\}$. For each vertex
		$x\in U$, we have
		\begin{equation}\label{H-degree}
			\begin{split}
				d_H(x)
				&\geq |N_G(x)\cap N_T(x)|-|X|n^{k-2}\\
				&\geq d_T(x)-|N_T(x)\setminus N_G(x)|
				-k\sqrt{\varepsilon}n^{k-1}\\
				&\geq d_T(x)-(k+1)\sqrt{\varepsilon}n^{k-1}.
			\end{split}
		\end{equation}
		
		Let $F=\{e\in E(G):e\cap X\neq\emptyset\}$ and $M\subseteq F$ be a
		matching in $G$ which maximizes $|U\cup V(M)|$.  Note that $\{U_1,U_2,\ldots,U_\ell\}$ is the vertex partition of $T-X$. Thus $\frac{n}{\ell}-|X|-1\leq |U_i|\leq \frac{n}{\ell}+1$ for $i\in[\ell]$.
		Since every edge of $M$ intersects $X$ and $M$ is a matching, we have $|M|\leq |X|\leq k\sqrt{\varepsilon}n$. Thus
		\[
		\frac{n}{\ell}-(k+k^2)\sqrt{\varepsilon}n-1
		\leq |U_i\setminus V(M)|
		\leq \frac{n}{\ell}+1
		\]
		for each $i\in[\ell]$. Moreover, for every
		$x\in U\setminus V(M)$, inequality \eqref{H-degree} implies that
		\begin{equation}\label{H-minus-M-degree}
			\begin{split}
				d_{H[U\setminus V(M)]}(x)
				&\geq d_T(x)-(k+1)\sqrt{\varepsilon}n^{k-1}
				-|V(M)|n^{k-2}\\
				&\geq
				\binom{\ell-1}{k-1}
				\left(\frac{n}{\ell}-1\right)^{k-1}
				-(k^2+k+1)\sqrt{\varepsilon}n^{k-1}\\
				&\geq
				\left(\binom{\ell-1}{k-1}
				-2k^2\ell^{k}\sqrt{\varepsilon}\right)
				\left(\frac{n}{\ell}\right)^{k-1}.
			\end{split}
		\end{equation}
		 Therefore, by choosing $\varepsilon$ sufficiently
		small and $n$ sufficiently large, applying Lemma~\ref{lem:partite-matching}
		to $H[U\setminus V(M)]$ yields a matching $M'$ of size
		$|M'|=\left\lfloor\frac{|U\setminus V(M)|}{k}\right\rfloor.$
		
		If
		$|V(M)\cup U|\geq ks+k$, then
		\[
		\begin{split}
			|M\cup M'|
			=\frac{|V(M)|}{k}
			+\left\lfloor\frac{|U\setminus V(M)|}{k}\right\rfloor
			=\left\lfloor\frac{|V(M)\cup U|}{k}\right\rfloor
			\geq\frac{ks+k}{k}=s+1,
		\end{split}
		\]
		a contradiction. 	Thus $|V(M)\cup U|\leq ks+k-1$.
		
		Let $I=V(G)\setminus(V(M)\cup U)$ and $W=\{x\in V(M):d_G(x)\leq
		2k^2\sqrt{\varepsilon}n^{k-1}\}.$
		Let $M_1=\{e\in M:e\cap W=\emptyset\}$ and $M_2=M\setminus M_1$.
		
		\vspace{0.15cm}	
		{\bf Claim.}		
		                            
		Every edge of $G$ intersecting $I$ also intersects $V(M_2)$.
		\vspace{0.15cm}	
		
		Suppose, to the contrary,  that there is an edge $f_0\in E(G)$  such that $f_0\cap I\neq\emptyset $ and
		$f_0\cap V(M_2)=\emptyset$.
		Let $M_0=\{e\in M_1:f_0\cap e\neq\emptyset\}$. Since $M$ is a matching in $G$ which maximizes $|U\cup V(M)|$, $f_0\cap V(M_1)\neq \emptyset$. Thus $M_0\neq \emptyset$.
		Let $A=V(M_0)\setminus(f_0\cup U).$
		Since $M_0\subseteq M_1$, the definition of $M_1$ implies that $V(M_0)\cap W=\emptyset$. Thus $d_G(x)>2k^2\sqrt{\varepsilon}n^{k-1}$ for each $x\in A$.
		
		We assert that there exists a matching
		$M'_0=\{h_x:x\in A\}$ such that
		$h_x\cap A=\{x\}$ for every $x\in A$ and $V(M'_0)\setminus A
		\subseteq U\setminus\bigl(V(M)\cup f_0\bigr)$.
		Indeed, suppose that a matching $M''_0$ has already been chosen with respect to
		some vertices of $A$, and let $x\in A\setminus V(M''_0)$. Define
		$B_x=\bigl(V(M)\setminus\{x\}\bigr)
		\cup I\cup f_0\cup V(M''_0).$
		Since $|M|\leq k\sqrt{\varepsilon}n$, $|I|\leq|X|\leq
		k\sqrt{\varepsilon}n$ and $|M_0|\leq k$, we have
		$|B_x|<2k^2\sqrt{\varepsilon}n$
		for sufficiently large $n$. Hence the number of edges containing $x$
		and meeting $B_x$ is less than $2k^2\sqrt{\varepsilon}n^{k-1}$,  which is less than $d_G(x)$.
		We may therefore choose an edge
		$h_x$ containing $x$ and avoiding $B_x$. Since
		$X\subseteq V(M)\cup I,$
		all the other vertices of $h_x$ belong to
		$U\setminus(V(M)\cup f_0\cup V(M''_0))$. Continuing this process, we can find
		the desired matching $M'_0$.
		
		Let
		$\widetilde M=(M\setminus M_0)\cup\{f_0\}\cup M'_0$. Then $\widetilde M$
		is a matching contained in $F$, 
		$V(M)\setminus U\subseteq V(\widetilde M)$ and $ V(\widetilde M)\cap I\neq \emptyset$. Consequently,
		$|U\cup V(\widetilde M)|>|U\cup V(M)|,$
		a contradiction  to the choice of $M$. This completes the proof of the Claim.
		
		
		
		
	By the Claim, $f\cap V(M_2)\neq\emptyset$ for each $f\in\{e\in E(G):e\cap I\neq\emptyset\}$.
		If $M_2=\emptyset$, then there is no edge of $G$
		intersecting $I$. Hence the number of isolated vertices of $G$ is at least $n-(ks+k-1)$. Therefore
		$|E(G)|\leq\operatorname{ex}_k(ks+k-1,\mathcal F)$
		and the proof is done. Thus, in the following proof,
		we may assume that $M_2\neq\emptyset$.	Recall that for each $e\in M_2$, $e\cap W\neq \emptyset$. Let $M_2=\{e_1,\ldots,e_{|M_2|}\}$. For each
		$i\in[|M_2|]$, choose one vertex $x_i\in e_i\cap W$, and let
		$U_0=\{x_i:1\leq i\leq |M_2|\}.$
		Then $|U_0|=|M_2|$ and $|V(M_2)|=k|U_0|$.
		
		We will construct an $\mathcal F$-free $k$-graph $G'$  on $V(G)$ such that
		$\nu(G')\leq s$ and $|E(G)|<|E(G')|\leq\operatorname{ex}_k(ks+k-1,\mathcal F)$.
		Let $q=\max\{|V(F')|:F'\in\mathcal F\}$. Since $n$ is sufficiently large, we can choose $q$
		vertices from $U_1\setminus U_0$. Denote the set of these $q$ vertices
		by $S$. Let $N(S)=\bigcap_{x\in S}N_H(x)$ and
		$$\mathcal D=
		\left\{
		A\in\binom{U}{k-1}:
		A\cap U_1=\emptyset
		\text{ and }|A\cap U_i|\leq1\text{ for every }i\in[\ell]
		\right\}.$$
		Since all vertices of $S$ lie in $U_1$, we have $ N(S)\subseteq \mathcal D$. Moreover, the definition of $X$ gives
		$|\mathcal D\setminus N_H(x)|
		\leq\sqrt{\varepsilon}n^{k-1}$ for every $x\in S$.
		As $|U_i|\geq n/\ell-k\sqrt{\varepsilon}n-1$ for every $i\in[\ell]$, we have
		\begin{equation}\label{common-neighborhood}
			\begin{split}
				|N(S)|
				\geq |\mathcal D|
				-\sum_{x\in S}|\mathcal D\setminus N_H(x)|
				\geq |\mathcal D|-q\sqrt{\varepsilon}n^{k-1}
				\geq \frac{n^{k-1}}{2\ell^{k-1}},
			\end{split}
		\end{equation}
		provided that $\varepsilon$ is sufficiently small and $n$ is
		sufficiently large.
		
		We now construct $G'$ by deleting every edge of $G$
		intersecting $I\cup U_0$ and adding all sets
		$\{x\}\cup A$ with $A\in N(S)$ and $A\cap U_0=\emptyset$ for every $x\in U_0$.
		Then $I$ is an isolated vertex set in $G'$. Since
		$|U\cup V(M)|\leq ks+k-1$, it follows that $\nu(G')\leq s$.
		
		We next prove that $G'$ is $\mathcal F$-free. Suppose that $G'$ contains a copy
		$Q$ of some $F'\in\mathcal F$. If $V(Q)\cap U_0=\emptyset$, then every edge of
		$Q$ is an edge of $G$, and hence $G$ contains a copy of $F'$, a
		contradiction. Thus
		$C_0=V(Q)\cap U_0\neq\emptyset$.
		Every edge of $G'$ intersecting $U_0$ contains exactly one vertex of $U_0$.
		Since $|S|=q\geq |V(F')|$ and $S\cap U_0=\emptyset$, we have
		$|S\setminus V(Q)|
		\geq |S|-|V(Q)\setminus C_0|
		\geq |C_0|$.
		Hence there exists an injection
		\[
		\phi:C_0\longrightarrow S\setminus V(Q).
		\]
		Replacing every $x\in C_0$ by $\phi(x)$ transforms each new edge

		&\le(a+t)t_{k-1}(n-s,\ell-1)+ht_{k-1}(n-s,\ell-1)=st_{k-1}(n-s,\ell-1).

<!-- lemma-8: body_md -->
\label{prop:weak-structure}
	Let $n, \ell, k, s$ be integers such that $\ell\geq k\geq 3$ and $1\leq s<c_0n$. Suppose that $F$ is an $n$-vertex $k$-graph, $V_0:=\{v:d_F(v)\ge t_{k-1}(n-s,\ell-1)\}$ and $V_2:=\{v\in V(F):d_F(v)\le ks\binom{n-1}{k-2}\}$. If $F$ is $\mathcal K_{\ell+1}^k$-free and $\nu(F)\le s$, then
	$\sum_{x\in V_0}d_{F[V_0\cup V_2]}(x)
	\le |V_0|\,t_{k-1}(|V_2|,\ell-1)$.
<!-- end lemma-8 -->
<!-- proof-11: body_md -->
We first prove that $V_0$ is strongly independent. Suppose, to the contrary, that there is an edge
	$e\in E(F)$ containing two distinct vertices $x,y\in V_0$. By
	Observation~\ref{obs:weak-link}, $L_F(x)$ is
	$\mathcal K_\ell^{k-1}$-free. Define $V(L_F(u))=\cup_{f\in E(L_F(u))} f$ for every $u\in V(F)$. If $|V(L_F(x))|<n-s$,  Theorem \ref{thm:weak-turan} implies that
	$	d_F(x)\le t_{k-1}(n-s-1,\ell-1)<t_{k-1}(n-s,\ell-1)$,
	contradicting $x\in V_0$. Thus $|V(L_F(x))|\ge n-s$, and similarly
	$|V(L_F(y))|\ge n-s$. Hence
	$|V(L_F(x))\cap V(L_F(y))|\ge n-2s$.
	Let $W\subseteq(V(L_F(x))\cap V(L_F(y)))\setminus e$ be a set of size $n-2s-k$. By Lemma \ref{lem:numerical},
	\[
	|E(L_F(x)[W])|\ge t_{k-1}(n-s,\ell-1)-(2s+k)\binom{n-1}{k-2}
	>t_{k-1}(|W|,\ell-2).
	\]
	 for $\ell>k$ and Theorem~\ref{thm:weak-turan} implies that there is a member of
	$\mathcal K_{\ell-1}^{k-1}$ in $L_F(x)[W]$. For the case $\ell=k$, $|E(L_F(x)[W])|\ge t_{k-1}(n-s,\ell-1)-(2s+k)\binom{n-1}{k-2}>0$, so any edge of $L_F(x)[W]$ is
	a member of $\mathcal K_{k-1}^{k-1}$. In either case, let $G\in\mathcal K_{\ell-1}^{k-1}$ be a subgraph
	of $L_F(x)[W]$ with core $C$.
	Every pair in $C$ and every pair $\{x,u\}$ with $u\in C$ is covered by an edge of $F$ containing $x$.
	Since $C\subseteq W\subseteq V(L_F(y))$, every pair $\{y,u\}$ with $u\in C$ is also covered by an edge of $F$. Moreover, $\{x,y\}$ is covered by $e$. Hence $C\cup\{x,y\}$ is the core of a member of $\mathcal K_{\ell+1}^k$ in $F$, a contradiction.
	Thus $V_0$ is strongly independent.
	
	Now let $x\in V_0$. Since $V_0$ is strongly independent, every edge of
	$F[V_0\cup V_2]$ containing $x$ has its remaining $k-1$ vertices in $V_2$.
	Recall that $L_F(x)[V_2]$ is $\mathcal K_\ell^{k-1}$-free for every $x\in V_0$. Hence
	$d_{F[V_0\cup V_2]}(x)=|E(L_F(x)[V_2])|
	\le t_{k-1}(|V_2|,\ell-1)$.
	Summing over $x\in V_0$ proves the lemma.

	\frac{k-2}{64(k-1)^{k-2}(\ell-1)^2},

<!-- lemma-10: body_md -->
\label{lem:exp-structure}
	Let $n,s,\ell,k$ be integers such that $\ell\geq k\geq 3$. There exist positive constants
	$\rho,\varepsilon,\xi,\eta$ and integers $s_0=s_0(k,\ell)$ and
	$n_0=n_0(k,\ell)$ such that the following holds.
	Let $H$ be an $n$-vertex $H_{\ell+1}^k$-free $k$-graph with $\nu(H)\le s$, where $s_0\le s<c_0n$ and $n\ge n_0$.
	If $|E(H)|\ge st_{k-1}(n-s,\ell-1)$, then
	there exist a set $X\subseteq V(H)$ and pairwise disjoint sets
	$U_1,\ldots,U_{\ell-1}\subseteq V(H)\setminus X$ such that
	$|X|\ge s-\frac{k\binom{\ell+1}{2}}{\eta}$,
	$d_H(x)<t_{k-1}(n-s,\ell-1)$ for every $x\notin X$ and
	$d_H(x)\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}$ for every $x\in X$.
	Moreover:
	\begin{enumerate}[label=(\roman*)]
		\item $|U_i|\ge(1-\varepsilon)\frac{n-s}{\ell-1}$ for $i\in [\ell-1]$;\label{prop1}
		\item $|e\cap U_i|\le1$ for every $e\in E(H)$ and every $i\in [\ell-1]$;\label{prop2}
		\item $X$ is a 	strongly independent set;\label{prop3}
		\item every $x\in X$ has at most $(4k\ell)^k\rho n$ sparse neighbours in
		$\bigcup_{i\in [\ell-1]}U_i$;\label{prop4}
		\item every $u\in\bigcup_iU_i$ has at most $(4k\ell)^{k/2}\sqrt\rho\,|X|$ sparse neighbours in $X$;\label{prop5}
		\item for every $i\in[\ell-1]$ and $u\in U_i$, at most $\xi^{1/3}n$ vertices of $\bigcup_{j\ne i}U_j$ are sparse neighbours of $u$.\label{prop6}
	\end{enumerate}
<!-- end lemma-10 -->
<!-- proof-13: body_md -->
Let $\varepsilon,\rho,\xi,\eta$ be constants such that $0<\varepsilon<1/(256k^3(4\ell)^{k-2})$, 
	$0<\rho<\frac{\varepsilon^2}{16(\ell-1)^2(4k\ell)^k}$, $0<\xi<\rho^3/(4k\ell)^{3k}$ and $0<\eta\ll\xi$.
	Assume $s_0$ and $n_0$ are sufficiently large.
	Let $G$ be a spanning subgraph of $H$ obtained by deleting every edge that contains a sparse pair in $H$. Since there are at most $\binom n2$ pairs of vertices of $H$ and each sparse pair is contained in at most $k\binom{\ell+1}{2}n^{k-3}$ edges of $H$, we have
	$|E(H)\setminus E(G)|\le\binom n2 k\binom{\ell+1}{2}n^{k-3}$.
	Therefore
	\begin{equation}
		|E(G)|\ge |E(H)|-\binom n2 k\binom{\ell+1}{2}n^{k-3}
		\ge st_{k-1}(n-s,\ell-1)-k\binom{\ell+1}{2}n^{k-1}. \label{eq:52}
	\end{equation}
	Let
	$Y=\{y\in V(G):d_G(y)\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}\}$.
	By Lemma~\ref{obs:extension}, it is not difficult to see that $G$ is $\mathcal K_{\ell+1}^{k}$-free.
	Thus $L_G(y)$ is $\mathcal K_\ell^{k-1}$-free for every $y\in Y$ by Observation~\ref{obs:weak-link} . If
	$|V(L_G(y))|<n-s-\rho n$, then $d_G(y)\leq t_{k-1}(n-s-\rho n,\ell-1)< t_{k-1}(n-s,\ell-1)-\eta n^{k-1}$ by Lemma \ref{lem:exp-numerical} and Theorem \ref{thm:weak-turan}, contradicting $y\in Y$. Thus for each $y\in Y$,
	\begin{equation}
		|V(L_G(y))|\ge n-s-\rho n. \label{eq:53}
	\end{equation}
	
	We claim that $Y$ is a strongly independent set in $H$. Otherwise, there is an edge
	$e\in E(H)$ containing two vertices $x,y\in Y$. By \eqref{eq:53}, let
	$W\subseteq(V(L_G(x))\cap V(L_G(y)))\setminus e$ be a set such that
	$|W|=\lfloor n-2s-2\rho n-k\rfloor$.
	By Lemma \ref{lem:exp-numerical}, we have
	\[
	|E(L_G(x)[W])|\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}-(2s+2\rho n+k)\binom{n-1}{k-2}
	>t_{k-1}(|W|,\ell-2).
	\]
	for $\ell>k$. Then Theorem~\ref{thm:weak-turan} implies that there is a member of
	$\mathcal K_{\ell-1}^{k-1}$ in $L_G(x)[W]$ for $\ell>k$. For the case $\ell=k$, $|E(L_G(x)[W])|\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}-(2s+2\rho n+k)\binom{n-1}{k-2}
	>0$ and an edge of $L_G(x)[W]$ is
	a member of $\mathcal K_{k-1}^{k-1}$. In either case, let $C$ denote its core. Every pair in
	$C\cup\{x,y\}$ except possibly $\{x,y\}$ is contained in an edge of $G$ and is
	therefore dense in $H$. Notice that $\{x,y\}$ is covered by $e$, and
	$e\cap(C\cup\{x,y\})=\{x,y\}$.
	By Lemma~\ref{obs:extension}, $H$ has a copy of $H_{\ell+1}^k$, a
	contradiction.
	
	Let
	$W_1=\{v:ks\binom{n-1}{k-2}<d_G(v)<t_{k-1}(n-s,\ell-1)-\eta n^{k-1}\}$ and
	$W_2=V(G)\setminus(Y\cup W_1)$.
	By Lemma~\ref{lem:budget}(i), one can see that $|Y|+|W_1|+\nu(G[W_2])\le s$.
	Let $h=s-|Y|-|W_1|$. Then $h\ge0$. Using similar arguments as in the proof of Lemma~\ref{lem:budget}(ii),                        
	we can derive that
	$|E(G)|\leq|W_1|(t_{k-1}(n-s,\ell-1)-\eta n^{k-1})+|Y|(t_{k-1}(n-s,\ell-1)+h\binom{n-1}{k-2})+k^2sh\binom{n-1}{k-2}$.
	Since $(|Y|+k^2s)\binom{n-1}{k-2}\le(k^2+1)s\binom{n-1}{k-2}<t_{k-1}(n-s,\ell-1)-\eta n^{k-1}$ by Lemma \ref{lem:exp-numerical}, we have
	$|E(G)|\leq st_{k-1}(n-s,\ell-1)-(s-|Y|)\eta n^{k-1}$.
	Combining with inequality \eqref{eq:52}, we have
	$s-|Y|\le\frac{k\binom{\ell+1}{2}}{\eta}$.
	By the choice of $s_0$,
	$|Y|\ge s_0-k\binom{\ell+1}{2}/\eta\ge\binom{\ell+1}{2}$. Let
	$Y'\subseteq Y$ such that $|Y'|=\binom{\ell+1}{2}$, and let $y_0\in Y'$.
	Define $V_0=\{x\in V(H):d_H(x)\ge t_{k-1}(n-s,\ell-1)\}$ and $X=V_0\cup Y$. Then $|X|\geq s-\frac{k\binom{\ell+1}{2}}{\eta}$,
	$d_H(x)<t_{k-1}(n-s,\ell-1)$ for every $x\notin X$, and	$d_H(x)\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}$ for every $x\in X$  as required.
	
	Recall that $d_H(x)>ks\binom{n-1}{k-2}$ for every $x\in Y\cup W_1$. By Lemma~\ref{lem:budget}(i), we have
	$|(Y\cup W_1)\cup V_0|\le s$, and hence $|(W_1\cup V_0)\setminus Y|\le\frac{k\binom{\ell+1}{2}}{\eta}\le\eta n.$
	                                                      
	                                             
	Let $Q=W_2\setminus V_0$. Recall that for every $y\in Y'$, $L_G(y)$ is $\mathcal K_\ell^{k-1}$-free
	and
	\begin{equation}
		\begin{split}
			|E(L_G(y)[Q])|
			&\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}-|(W_1\cup V_0)\setminus Y|n^{k-2}\\
			&\ge t_{k-1}(n-s,\ell-1)-2\eta n^{k-1}\ge t_{k-1}(|Q|,\ell-1)-3\eta n^{k-1}.
		\end{split}
	\end{equation}
	The last inequality holds since
	$n-s\le|Q|\le n-s+k\binom{\ell+1}{2}/\eta$ and $n$ is sufficiently large.
	Since $H_\ell^{k-1}\in \mathcal K_\ell^{k-1}$, applying Theorem~\ref{thm:exp-stability} with parameters $k-1$ and $\ell-1$ when $k>3$, and Theorem~\ref{erdos-stability} when $k=3$, we conclude that $L_G(y)[Q]$ is $\xi$-close to $T_{k-1}(|Q|,\ell-1)$.
	Thus there is a partition $\{V_1^y, \ldots, V_{\ell-1}^y\}$ of $Q$ satisfying $|V_i^y|=\lfloor(|Q|+i-1)/(\ell-1)\rfloor$ for $i\in [\ell-1]$ such that $L_G(y)[Q]$ can be transformed into the complete $(\ell-1)$-partite $(k-1)$-graph $K_y$ on $V_1^y\cup \cdots\cup V_{\ell-1}^y$ by adding and deleting at most $\xi n^{k-1}$ edges.
	Since $s\leq c_0n$, we may assume that
	$|V_i^y|\ge n/(2\ell)$ for every $y\in Y'$ and
	$i\in[\ell-1]$.
	
	For $y\in Y'$, let
	$A_{y}=\{u\in Q:\{y,u\}\text{ is sparse in }H\}$.
	Recall that every edge of $G$ contains only dense pairs of $H$. Hence
	$u\in A_{y}$ implies $d_{L_G(y)[Q]}(u)=0$. Since every edge of $K_y$ containing such a vertex is missing from $L_G(y)[Q]$ and each missing edge is counted at most $k-1$ times, we have
	\[	\begin{aligned}
		\xi n^{k-1}
		\ge |E(K_{y})\setminus E(L_G(y)[Q])|
		\ge \frac{|A_{y}|}{k-1}
		\binom{\ell-2}{k-2}\left(\frac{n}{2\ell}\right)^{k-2}.
	\end{aligned}
	\]
	Consequently,
	\begin{equation}
		|A_{y}|\le (4k\ell)^k\xi n.  \label{eq:Ay}
	\end{equation}
	
	Let $y_0\in Y'$. With respect to the fixed partition $\{V_1^{y_0},\cdots, V_{\ell-1}^{y_0}\}$, let
	\[
	\mathcal S_0=\{\{u,v\}:u\in V_i^{y_0},\ v\in V_j^{y_0},\ i\ne j, \ i,j\in [\ell-1],
	\ \{u,v\}\text{ is sparse in }H\}.
	\]
	If $\{u,v\}\in\mathcal S_0$, then no edge of $L_G(y_0)[Q]$ contains $u$ and $v$.
	Every such pair is contained in at least
	$\binom{\ell-3}{k-3}(n/(2\ell))^{k-3}$ edges of $K_{y_0}$ and
	each missing edge contains at most $\binom{k-1}{2}$ pairs. Therefore
	\[
	\begin{aligned}
		\xi n^{k-1}
		\ge |E(K_{y_0})\setminus E(L_G(y_0)[Q])|
		\ge \frac{|\mathcal S_{0}|}{\binom{k-1}{2}}
		\binom{\ell-3}{k-3}\left(\frac{n}{2\ell}\right)^{k-3}.
	\end{aligned}
	\]
	It follows that
	$|\mathcal S_0|\le (4k\ell)^k\xi n^2$.
	
	For $u\in V_i^{y_0}$, $ i\in [\ell-1]$, let
	$d_{\mathcal S_0}(u)$ denote the number of vertices in
	$\bigcup_{j\ne i}V_j^{y_0}$ each of which together with $u$ forms a sparse pair in $\mathcal S_0$, and let
	$B_0=\{u\in Q:d_{\mathcal S_0}(u)>\xi^{1/3}n\}$.
	Double counting the pairs in $\mathcal S_0$ yields $|B_0|\xi^{1/3}n\leq 2|\mathcal S_0|$. Thus
	\begin{equation}
		|B_0|\le 2(4k\ell)^k\xi^{2/3}n. \label{eq:B0}
	\end{equation}
	For $i\in[\ell-1]$, let
	$U_i^0=V_i^{y_0}\setminus
	\left(B_0\cup\bigcup_{y\in Y'}A_y\right)$.
	By \eqref{eq:Ay}, \eqref{eq:B0}
	and by the choice of $\rho$ and $\xi$,
	\begin{equation}
		\min_i|U_i^0|\ge(1-\varepsilon/2)\frac{n-s}{\ell-1}
		\mbox{ and }
		\left|Q\setminus\bigcup_iU_i^0\right|\le\rho n. \label{eq:54}
	\end{equation}
	By the definition of $A_y$, every vertex of $\bigcup_iU_i^0$ is a dense neighbour of every vertex of $Y'$ in $H$. By the definition of $B_0$,
	each $u\in U_i^0$ has at most $\xi^{1/3}n$ sparse neighbours in
	$\bigcup_{j\ne i}U_j^0$. In particular, the family of sparse pairs
	joining distinct $U_i^0$'s is a subfamily of $\mathcal S_0$ and has at
	most $(4k\ell)^k\xi n^2$ members.
	
	\vspace{0.15cm}	
	{\bf Claim 1.} $|e\cap U_i^0|\le1$ for every $e\in E(H)$ and
	every $i \in [\ell-1]$.
	\vspace{0.15cm}
	
	Otherwise, after relabelling, we suppose that there is an edge $e$ that contains
	distinct $u_0,u_1\in U_1^0$. Since
	$|Y'|=\binom{\ell+1}{2}>k$, choose $y\in Y'\setminus e$. For
	$j=2,\ldots,\ell-1$, let
	$Z_j=\{u\in U_j^0\setminus e: \{u_0,u\}\text{ and } \{u_1,u\}
	\text{ are dense  in $H$} \}$.
	Thus $|Z_j|\ge|U_j^0|-2\xi^{1/3}n-k\ge|U_j^0|/2$. For the case $\ell=3$, let $z_2\in Z_2$. For the case $\ell>3$, applying
	Lemma~\ref{lem:selection} to the family of sparse pairs joining
	distinct $U_j^0$'s, there is a set $\{z_j\in Z_j: 2\le j\le \ell-1\}$ in which
	every pair  is dense in $H$. In either case, let $C:=\{y,u_0,u_1\}\cup\{z_j:j\ne1\}$. All pairs of
	$C$ are dense except $\{u_0,u_1\}$ since the
	vertex $y$ is  a dense neighbour of each of $u_0,u_1,z_2,\ldots,z_{\ell-1}$. Note that
	$e\cap C=\{u_0,u_1\}$. Lemma~\ref{obs:extension} gives a copy of
	$H_{\ell+1}^k$ in $H$, a contradiction. This completes the proof of Claim 1.
	
	Let $K^0$ be the complete $(\ell-1)$-partite $(k-1)$-graph on
	$U_1^0\cup \cdots \cup U_{\ell-1}^0$. We prove that for each $x\in X$ and $i\in[\ell-1]$,
	\begin{equation}\label{eq:coarse}
		|\{u\in U_i^0:\{x,u\}\text{ is sparse  in $H$ }\}|<\frac13|U_i^0|.
	\end{equation}
	Recall that every $x\in X$ satisfies
	$d_H(x)\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}$. Moreover,
	$V(H)\setminus Q$ is contained in
	$Y\cup\bigl((W_1\cup V_0)\setminus Y\bigr)$,
	where $|Y|\le s$ and
	$|(W_1\cup V_0)\setminus Y|\le k\binom{\ell+1}{2}/\eta$. Together with
	\eqref{eq:54}, we can derive that
	\[
	|E(L_H(x)[\textstyle\bigcup_iU_i^0])|
	\ge t_{k-1}(n-s,\ell-1)-\eta n^{k-1}
	-\left(s+\rho n+\frac{k\binom{\ell+1}{2}}{\eta}\right)\binom{n-1}{k-2}.
	\]
	Since $s<c_0n$,
	$|E(L_H(x)[\textstyle\bigcup_iU_i^0])|
	\ge t_{k-1}(n-s,\ell-1)-(c_0+\rho+2\eta)n^{k-1}.$
	Note that $L_H(x)[\textstyle\bigcup_iU_i^0]$ is a subgraph of $K^0$ and
	$|E(K^0)|\le t_{k-1}(|Q|,\ell-1)\le t_{k-1}(n-s,\ell-1)+\eta n^{k-1}$
	for sufficiently large $n$. Since $\rho+3\eta<c_0$, we have
	\begin{equation}\label{eq:coarse-missing}
		|E(K^0)\setminus E(L_H(x)[\textstyle\bigcup_iU_i^0])|<2c_0n^{k-1}.
	\end{equation}
	
	Let $m=\min_i|U_i^0|$. By \eqref{eq:54},
	$m^{k-1}\ge\frac12\left(\frac{n}{\ell-1}\right)^{k-1}$.
	Suppose that \eqref{eq:coarse} fails for some $x$ and $i$. Let
	$S_i=\{u\in U_i^0: \{x,u\}\text{ is sparse}\}$. Then $|S_i|\geq\frac13|U_i^0|\geq \frac13 m $. For each $u\in S_i$,
	$d_{K^0}(u)\ge\binom{\ell-2}{k-2}m^{k-2}$ and $d_{L_H(x)}(u)\le k\binom{\ell+1}{2}n^{k-3}$.
	Since no edge of $K^0$ contains distinct vertices of $U_i^0$,
	\begin{align*}
		|E(K^0)\setminus E(L_H(x)[\textstyle\bigcup_iU_i^0])|
		\ge |S_i|\left(\binom{\ell-2}{k-2}m^{k-2}-k\binom{\ell+1}{2}n^{k-3}\right)
		\ge\frac14\binom{\ell-2}{k-2}m^{k-1}
		\ge\frac18\frac{\binom{\ell-2}{k-2}}
		{ (\ell-1)^{k-1}}n^{k-1}.
	\end{align*}
	As
	$\binom{\ell-2}{k-2}\ge
	\left(\frac{\ell-1}{k-1}\right)^{k-2}$,
	the last expression is at least
	$\frac{n^{k-1}}{8(k-1)^{k-2}(\ell-1)}
	=(\ell-1)^2c_0n^{k-1}
	\ge4c_0n^{k-1}$,
	contradicting \eqref{eq:coarse-missing}. This proves \eqref{eq:coarse}.
	
	\vspace{0.15cm}
	{\bf Claim 2.} $X$ is a strongly independent set in $H$, and then (iii) follows.
	\vspace{0.15cm}
	
	Suppose, to the contrary,  that there is an edge
	$e$ of $H$ that contains distinct $a,b\in X$. For each $i\in[\ell-1]$, let
	$Z_i=\{u\in U_i^0\setminus e:
	\{a,u\}\text{ and }\{b,u\}\text{ are dense}\}$.
	By \eqref{eq:coarse}, $|Z_i|\ge|U_i^0|/3-k\ge|U_i^0|/4$.
	Applying Lemma~\ref{lem:selection} to the family of sparse pairs
	joining distinct $U_i^0$'s, there is a set $\{u_i\in Z_i: i\in[\ell-1]\}$ in which every
	pair  is dense in $H$. Thus
	$\{a,b,u_1,\ldots,u_{\ell-1}\}$ has all pairs dense except possibly $\{a,b\}$,
	while $e$ intersects this set exactly in $\{a,b\}$. Lemma~\ref{obs:extension}
	yields a copy of $H_{\ell+1}^k$  in $H$, a contradiction. This completes the proof of Claim 2.

		\frac{\rho}{2^{k-1}(k-1)^{k-2}}\,n^{k-1}
		>\eta n^{k-1}
	\end{aligned}
	\]
	for sufficiently large $n$.
	This proves $t_{k-1}(n-s,\ell-1)-\eta n^{k-1}>t_{k-1}(n-s-\rho n,\ell-1)$.
	
	Finally, by inequality \eqref{eq:numerical-gap}, we have
	\[
	\begin{aligned}
		&t_{k-1}(n-s,\ell-1)-\eta n^{k-1}
		-(2s+2\rho n+k)\binom{n-1}{k-2}
		-t_{k-1}(\lfloor n-2s-2\rho n-k\rfloor,\ell-2)\\
		>&
		n^{k-1}\left(
		\frac{k-2}{4(k-1)^{k-2}(\ell-1)^2}
		-\eta-\frac{2\rho+k/n}{(k-2)!}
		\right).
	\end{aligned}
	\]
	By the assumptions on $\rho$ and $\eta$, 
	$\eta+\frac{2\rho}{(k-2)!}<3\rho<\frac{3(k-2)}{64(k-1)^{k-2}(\ell-1)^2}$.
	Hence $t_{k-1}(n-s,\ell-1)-\eta n^{k-1}
	-(2s+2\rho n+k)\binom{n-1}{k-2}>t_{k-1}(\lfloor n-2s-2\rho n-k\rfloor,\ell-2)$.
