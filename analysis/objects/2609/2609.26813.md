<!-- theorem-1: body_md -->
\label{th:GKS-LagGraph-main}
		For each $\epsilon\in\ensuremath{\mathcal{E}}_r$, there is a
		one-to-one correspondence between $\epsilon$-unit generalized
		Killing spinors on a three-dimensional pseudo-Riemannian spin manifold $(M,g)$ and Lagrangian sections of almost pseudo-Hermitian structure $(\ensuremath{\mathcal{G}}_c,\ensuremath{\mathcal{J}}_c)$ on the total space of its principal spin bundle $\pi\colon P\to M$.
		Thus, in the Lorentzian case, the correspondence is two-to-one if
		spacelike and timelike spinors are considered together.
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
A linear map $ l_g \colon \ensuremath{\mathfrak{m}} \to \ensuremath{\mathfrak{h}} $ is given by 
		\begin{equation*}
			\lambda_*(l_g(X))Y = X\times Y, \quad \forall X,Y\in \ensuremath{\mathfrak{m}}.
		\end{equation*}
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lm:lg-Ad-isom}
		The linear map $l_g$ is an $H$-equivariant linear isomorphism, where $H$ acts on $\ensuremath{\mathfrak{m}}$ through $\lambda$ and on $\ensuremath{\mathfrak{h}}$ through the adjoint representation. More precisely,
		\begin{equation*}
			l_g(\lambda(h)X) = \operatorname{Ad}_h(l_g(X)), \quad \forall h\in H, \forall X\in \ensuremath{\mathfrak{m}}.
		\end{equation*}
		Moreover, $l_g$ intertwines the cross product on $\ensuremath{\mathfrak{m}}$ with the Lie bracket on $\ensuremath{\mathfrak{h}}$:
		\begin{equation}\label{eq:lg-cross}
			l_g(X\times Y ) = [l_gX,l_gY], \quad \forall X,Y\in \ensuremath{\mathfrak{m}}.
		\end{equation}
		Thus $l_g$ identifies the $H$-module $\ensuremath{\mathfrak{m}}$, equipped with the cross product, with $\ensuremath{\mathfrak{h}}$, equipped with its Lie bracket.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $\lambda_* $ is a Lie algebra isomorphism, $l_g\circ \lambda(h) = \operatorname{Ad}_h\circ l_g $ and 
		\begin{equation*}
			\lambda_*\circ l_g \colon\ensuremath{\mathfrak{m}}
			\ni X\mapsto * X\in \lambda_*(\ensuremath{\mathfrak{h}})=\mathfrak{so}(\ensuremath{\mathfrak{m}})\simeq \Lambda^2\ensuremath{\mathfrak{m}}
		\end{equation*}
		is a linear isomorphism.
		                   
			                        
			                  
				                                            
				                                                                                       
				                        
				                                       
				                                   
				                
			    
		For every $h\in H, $ and $X,Y\in \ensuremath{\mathfrak{m}}, $
		\begin{align*}
			\lambda_*(\operatorname{Ad}_hl_g(X))Y & = \lambda(h)\lambda_*(l_g(X))\lambda(h)^{-1}Y\\
			& = \lambda(h)(X\times\lambda(h)^{-1}Y)\\
			& = \lambda(h)X\times Y\\
			& = \lambda_*(l_g\lambda(h)(X))Y.
		\end{align*}
		
		Setting $L_X\coloneqq \lambda_*l_g(X) \in \mathfrak{so}(\ensuremath{\mathfrak{m}}), $ by Jacobi identity, for every $X,Y,Z\in \ensuremath{\mathfrak{m}},$
		\begin{align*}
			L_X(Y\times Z) & = L_X Y \times Z+Y\times L_XZ,\\
			[L_X,L_Y](Z) & = X\times(Y\times Z) - Y\times(X\times Z)\\
			& = L_{X\times Y} (Z).
		\end{align*}
		Therefore,
		\begin{align*}
			\lambda_*([l_gX,l_gY])(Z) & = [\lambda_*l_g(X),\lambda_*l_g(Y)](Z)\\
			& = \lambda_*l_g(X\times Y)(Z).
		\end{align*}
		Since $\lambda_*\colon \ensuremath{\mathfrak{h}}\to \mathfrak{so}(\ensuremath{\mathfrak{m}})$ is a linear isomorphism, 
		the assertion follows.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{pr:almost-pseudo-Hermitian}
		For every $c>0$, we define a bundle endomorphism
		$\ensuremath{\mathcal{J}}_c\in\Gamma(\operatorname{End}(TP))$ by
		\begin{equation*}
			\ensuremath{\mathcal{J}}_c(X_u^H+U_u^\#) = -c^{-1}(l_g^{-1}U)^H_u+c(l_g X)^\#_u,
		\end{equation*}
		and a pseudo-Riemannian metric $\ensuremath{\mathcal{G}}_c$ on $P$ by
		\begin{align*}
			\ensuremath{\mathcal{G}}_c(X_u^H+U_u^\#,Y_u^H+V_u^\#)
			&=
			c\,\left\langle X,Y\right\rangle
			+c^{-1}\left\langle l_g^{-1}U,l_g^{-1}V\right\rangle.
		\end{align*}
		Then $(\ensuremath{\mathcal{G}}_c,\ensuremath{\mathcal{J}}_c)$ is an almost
		pseudo-Hermitian structure on $P$.

<!-- remark-1: body_md -->
\label{rm:non-symplectic_Lagrangian}
		By \prettyref{lm:Hor and Ver Lie-bracket}, the fundamental two-form $\Omega$ is non-degenerate but never closed, therefore the corresponding section cannot be symplectic Lagrangian but almost symplectic Lagrangian.
		Indeed, for $\widehat{X},\widehat{Y},\widehat{Z} \in C^\infty(P,\ensuremath{\mathfrak{m}})^H $ and $ \widehat{U},\widehat{V},\widehat{W} \in C^\infty(P,\ensuremath{\mathfrak{h}})^H, $
		\begin{align*}
			d\Omega (\widehat{X}^H,\widehat{Y}^H,\widehat{Z}^H) & = -\sum_{\text{cyc}}{\Omega([\widehat{X}^H,\widehat{Y}^H],\widehat{Z}^H)}\\
			& = \sum_{\text{cyc}}{\Omega(\widetilde\Omega(\widehat{X}^H,\widehat{Y}^H)^\#,\widehat{Z}^H)}\\
			& = -\sum_{\text{cyc}}{\left\langle l_g^{-1}\widetilde\Omega(\widehat{X}^H,\widehat{Y}^H), \widehat{Z} \right\rangle},\\
			d\Omega (\widehat{X}^H,\widehat{Y}^H,\widehat{U}^\#) & = \widehat{X}^H\Omega(\widehat{Y}^H,\widehat{U}^\#) - \widehat{Y}^H\Omega(\widehat{X}^H,\widehat{U}^\#)\\
			& \quad  - \Omega([\widehat{X}^H,\widehat{Y}^H],\widehat{U}^\#) - \Omega([\widehat{Y}^H,\widehat{U}^\#], \widehat{X}^H) - \Omega([\widehat{U}^\#,\widehat{X}^H],\widehat{Y}^H)\\
			& = \widehat{X}^H\left\langle l_g^{-1}\widehat{U}, \widehat{Y} \right\rangle - \widehat{Y}^H\left\langle l_g^{-1}\widehat{U} , \widehat{X}\right\rangle \\
			& \quad -\left\langle l_g^{-1}\widehat{U},\widehat{X}^H(\widehat{Y}) - \widehat{Y}^H(\widehat{X}) \right\rangle + \left\langle l_g^{-1}\widehat{Y}^H(\widehat{U}),\widehat{X} \right\rangle -\left\langle l_g^{-1}\widehat{X}^H(\widehat{U}),\widehat{Y} \right\rangle\\
			& =0,\\
			d\Omega (\widehat{X}^H,\widehat{U}^\#,\widehat{V}^\#) & = - \widehat{U}^\#\Omega(\widehat{X}^H,\widehat{V}^\#) + \widehat{V}^\#\Omega(\widehat{X}^H,\widehat{U}^\#) - \Omega([\widehat{U}^\#,\widehat{V}^\#], \widehat{X}^H)\\
			                                             
			& = -\left\langle l_g^{-1}\widehat{U}\times l_g^{-1}\widehat{V},\widehat{X} \right\rangle ,\\
			d\Omega (\widehat{U}^\#,\widehat{V}^\#,\widehat{W}^\#) & = 0.
		\end{align*}
		Thus, the term of $d\Omega (\widehat{X}^H,\widehat{U}^\#,\widehat{V}^\#)$ does not vanish for some $\widehat{X},\widehat{U},\widehat{V}.$

<!-- lemma-2: body_md -->
\label{lm:Hor and Ver Lie-bracket}
		Let $ \widehat{X},\widehat{Y} \in C^\infty(P,\ensuremath{\mathfrak{m}})^H $ and $ \widehat{U}, \widehat{V} \in C^\infty(P,\ensuremath{\mathfrak{h}})^H $ be $H$-equivariant function.
		The following formulae hold.
		\begin{align*}
			\left[ \widehat{X}^H,\widehat{Y}^H\right] & = \left( \widehat{X}^H(\widehat{Y}) - \widehat{Y}^H(\widehat{X}) \right)^H - \widetilde\Omega(\widehat{X}^H,\widehat{Y}^H)^\#,\\
			\left[ \widehat{U}^\#,\widehat{Y}^H \right] & = -\left( \widehat{Y}^H(\widehat{U})\right)^\#,\\
			\left[ \widehat{U}^\#,\widehat{V}^\# \right] & = -[\widehat{U},\widehat{V}]^\#.
		\end{align*}
		\begin{proof}
			These formulae
			\[ 0 = d\theta+\lambda_*{\widetilde{\omega}}\wedge\theta, \quad \widetilde\Omega = d{\widetilde{\omega}} + {\widetilde{\omega}}\wedge{\widetilde{\omega}} \]
			lead our assertions by the following direct calculations
			\begin{align*}
				0 =d\theta(\widehat{X}^H,\widehat{Y}^H) & = \widehat{X}^H(\theta(\widehat{Y}^H)) - \widehat{Y}^H(\theta(\widehat{X}^H)) -\theta([\widehat{X}^H,\widehat{Y}^H]),\\
				\theta([\widehat{X}^H,\widehat{Y}^H]) & = \widehat{X}^H(\widehat{Y}) - \widehat{Y}^H(\widehat{X}),\\
				\widetilde\Omega(\widehat{X}^H,\widehat{Y}^H) & = d{\widetilde{\omega}}(\widehat{X}^H,\widehat{Y}^H) = -{\widetilde{\omega}}([\widehat{X}^H,\widehat{Y}^H]). 
			\end{align*}
			Therefore, 
			\begin{equation}\label{eq:hori-bracket}
				\left[ \widehat{X}^H,\widehat{Y}^H\right] = \left( \widehat{X}^H(\widehat{Y}) - \widehat{Y}^H(\widehat{X}) \right)^H - \widetilde\Omega(\widehat{X}^H,\widehat{Y}^H)^\#.
			\end{equation}
			The other formulae are also obtained by the similar calculations.
		\end{proof}

<!-- theorem-2: body_md -->
\label{th:integrability-complex}
		The almost complex structure $\ensuremath{\mathcal{J}}_c$ on $P$ is a complex structure if and only if $ (M,g) $ has constant sectional curvature $(-1)^{r-1}c^2$.
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
By \prettyref{lm:cal-Nijenhuis} and Newlander--Nirenberg theorem, the almost complex structure $\ensuremath{\mathcal{J}}_c$ is integrable, if and only if 
		\[ \widetilde\Omega(\widehat{X}^H,\widehat{Y}^H) = c^2[l_g\widehat{X}, l_g\widehat{Y}], \quad \forall \widehat{X},\widehat{Y}\in C^\infty(P,\ensuremath{\mathfrak{m}})^H. \]
		By \prettyref{eq:lg-cross} in \prettyref{lm:lg-Ad-isom}, for every section $s \in \Gamma(\ensuremath{\mathcal{U}},P), $ and put $ \ensuremath{\mathcal{X}} = [s, X], X,Y,Z\in \ensuremath{\mathfrak{m}}$
		\begin{align*}
			R(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}})\ensuremath{\mathcal{Z}} & = \left[ s,\lambda_*\left( s^*\widetilde\Omega\left( \ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}} \right) \right) Z \right],\\
			& = \left[ s,c^2[\lambda_*l_gX,\lambda_*l_gY]Z \right],\\
			& = \left[ s,c^2(X\times Y)\times Z \right],\\
			& = (-1)^rc^2 \left( g(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Z}})\ensuremath{\mathcal{Y}} - g(\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{Z}})\ensuremath{\mathcal{X}} \right).
		\end{align*}
		Therefore, the sectional curvature $\kappa$ of base manifold $(M,g)$ spanned by $\{\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}}\}$ is
		\[ \kappa = \frac{g(R(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}})\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{X}})}{g(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{X}})g(\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{Y}})-g(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}})^2} = (-1)^{r-1}c^2. \]

<!-- lemma-4: body_md -->
Let $\overline\nabla$ be the Levi-Civita connection of $(P,\mathcal G_c)$.
		For $\widehat X,\widehat Y\in C^\infty(P,\ensuremath{\mathfrak{m}})^H$ and
		$\widehat U,\widehat V\in C^\infty(P,\mathfrak h)^H$, the horizontal and
		vertical components of $\overline\nabla$ are given by
		\begin{align*}
			\overline\nabla_{\widehat{X}^H}\widehat{Y}^H & = \left( \widehat{X}^H(\widehat{Y}) \right)^H -\frac{1}{2}\widetilde\Omega\left( \widehat{X}^H,\widehat{Y}^H \right)^\#,\\
			\overline\nabla_{\widehat{X}^H}\widehat{V}^\# & = \ensuremath{\mathcal{T}}_c(\widehat{X},\widehat{V})^H + \left( \widehat{X}^H(\widehat{V}) \right)^\#,\\
			\overline\nabla_{\widehat{U}^\#}\widehat{Y}^H & = \ensuremath{\mathcal{T}}_c(\widehat{Y},\widehat{U})^H,\\
			\overline\nabla_{\widehat{U}^\#}\widehat{V}^\# & = -\frac{1}{2}[\widehat{U},\widehat{V}]^\#,
		\end{align*}
		where $\ensuremath{\mathcal{T}}_c(\widehat{X},\widehat{V}) \in C^\infty(P,\ensuremath{\mathfrak{m}})^H $ is given by
		\[ \left\langle \ensuremath{\mathcal{T}}_c(\widehat{X}, \widehat{V}),\widehat{Z}\right\rangle = \frac{1}{2c^2}\left\langle l_g^{-1}\widetilde\Omega\left( \widehat{X}^H,\widehat{Z}^H\right), l_g^{-1}\widehat{V}\right\rangle. \]

<!-- theorem-3: body_md -->
\label{th:NK-str}
		The almost pseudo-Hermitian structure $ (\ensuremath{\mathcal{G}}_c, \ensuremath{\mathcal{J}}_c) $ on $P$ is nearly pseudo-K\"ahler if and only if $(M,g)$ has constant sectional curvature $\frac{(-1)^{r}}{3}c^2.$
<!-- end theorem-3 -->
<!-- proof-5: body_md -->
Assume that $(P,\ensuremath{\mathcal{G}}_c,\ensuremath{\mathcal{J}}_c)$ is nearly pseudo-K\"ahler. Then covariant derivatives are given by
		\begin{align*}
			(\overline\nabla_{\widehat{X}^H}\ensuremath{\mathcal{J}}_c)\widehat{X}^H & = c\ensuremath{\mathcal{T}}_c(\widehat{X},l_g\widehat{X})^H,\\
			(\overline\nabla_{\widehat{U}^\#}\ensuremath{\mathcal{J}}_c)\widehat{U}^\# & = -c^{-1}\ensuremath{\mathcal{T}}_c(l_g^{-1}\widehat{U}, \widehat{U})^H,\\
			(\overline\nabla_{\widehat{U}^\#}\ensuremath{\mathcal{J}}_c)\widehat{X}^H + (\overline\nabla_{\widehat{X}^H}\ensuremath{\mathcal{J}}_c)\widehat{U}^\# & = -2cl_g\ensuremath{\mathcal{T}}_c(\widehat{X},\widehat{U})^\# + \frac{1}{2c}\widetilde\Omega(\widehat{X}^H,(l_g^{-1}\widehat{U})^H)^\# - \frac{1}{2}c[\widehat{U},l_g\widehat{X}]^\#
		\end{align*}
		all vanishes.
		For every $\ensuremath{\mathfrak{m}}$-valued two form $\alpha \in \Lambda^2\ensuremath{\mathfrak{m}}^*\otimes \ensuremath{\mathfrak{m}} $, there exists a unique $S\in \operatorname{End}(\ensuremath{\mathfrak{m}}) $ via the isomorphism $\Lambda^2\ensuremath{\mathfrak{m}} \ni X\wedge Y \mapsto X\times Y\in \ensuremath{\mathfrak{m}}$
		\begin{equation*}
			\alpha(X,Y) = S(X\times Y).
		\end{equation*}
		Hence, there exists a unique $\ensuremath{\mathcal{S}} \in C^\infty(P,\operatorname{End}(\ensuremath{\mathfrak{m}}))^H, $
		\begin{equation*}
			l_g^{-1}\widetilde\Omega(\widehat{X}^H,\widehat{Y}^H) = \ensuremath{\mathcal{S}}(\widehat{X}\times\widehat{Y}).
		\end{equation*}
		If $(P,\ensuremath{\mathcal{G}}_c,\ensuremath{\mathcal{J}}_c)$ is nearly pseudo-K\"ahler, then $ \ensuremath{\mathcal{T}}_c(\widehat{X},l_g\widehat{X}) = 0 $ that implies
		\[ \left\langle \ensuremath{\mathcal{T}}_c(\widehat{X}, l_g\widehat{X}),\widehat{Z}\right\rangle = \frac{1}{2c^2}\left\langle \ensuremath{\mathcal{S}}(\widehat{X}\times \widehat{Z}), \widehat{X}\right\rangle = 0, \quad \forall \widehat{X}, \widehat{Z} \in C^\infty(P,\ensuremath{\mathfrak{m}})^H. \]
		Then there exists a unique function $\rho\in C^\infty(P,\ensuremath{\mathbb{R}}), $
		\[ \ensuremath{\mathcal{S}} = \rho\operatorname{id}_\ensuremath{\mathfrak{m}}. \]
		From $ \left\langle X\times Y,Z\right\rangle = -\left\langle X\times Z,Y \right\rangle, $
		\begin{align*}
			\left\langle \ensuremath{\mathcal{T}}_c(\widehat{X},\widehat{U}),\widehat{Z}\right\rangle & = \frac{1}{2c^2}\left\langle l_g^{-1}\widetilde\Omega\left( \widehat{X}^H,\widehat{Z}^H \right),l_g^{-1}\widehat{U} \right\rangle\\
			& = -\frac{\rho}{2c^2}\left\langle \widehat{X}\times l_g^{-1}\widehat{U}, \widehat{Z} \right\rangle.
		\end{align*}
		then
		\[ \ensuremath{\mathcal{T}}_c(\widehat{X},\widehat{U}) = -\frac{\rho}{2c^2}\left( \widehat{X}\times l_g^{-1}\widehat{U}\right). \]
		By \prettyref{eq:lg-cross} in \prettyref{lm:lg-Ad-isom}, we have
		\begin{align*}
			(\overline\nabla_{\widehat{U}^\#}\ensuremath{\mathcal{J}}_c)\widehat{X}^H + (\overline\nabla_{\widehat{X}^H}\ensuremath{\mathcal{J}}_c)\widehat{U}^\# & = -2cl_g\ensuremath{\mathcal{T}}_c(\widehat{X},\widehat{U})^\# + \frac{1}{2c}\widetilde\Omega(\widehat{X}^H,(l_g^{-1}\widehat{U})^H)^\# - \frac{1}{2}c[\widehat{U},l_g\widehat{X}]^\#\\
			& = \left( \frac{3\rho}{2c}l_g\left( \widehat{X}\times l_g^{-1}\widehat{U}\right) -\frac{c}{2}[\widehat{U},l_g\widehat{X}] \right)^\#\\
			& = \frac{3\rho+c^2}{2c}l_g \left( \widehat{X}\times l_g^{-1}\widehat{U} \right)^\#=0.
		\end{align*}
		Then $\rho = -\frac{c^2}{3}.$
		It is also followed that if $\rho = -\frac{c^2}{3}, $ for every section $s \in \Gamma(\ensuremath{\mathcal{U}},P), $ and put $\ensuremath{\mathcal{X}} = [s, X], X,Y,Z\in \ensuremath{\mathfrak{m}},$ then
		\begin{align*}
			R(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}})\ensuremath{\mathcal{Z}} & = \left[ s,\lambda_*\left( s^*\widetilde\Omega\left( \ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}} \right) \right) Z \right],\\
			& = \left[ s,\rho(X\times Y)\times Z \right],\\
			& = \frac{(-1)^rc^2}{3} \left( g(\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{Z}})\ensuremath{\mathcal{X}} - g(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Z}})\ensuremath{\mathcal{Y}} \right).
		\end{align*}
		Thus, the sectional curvature of the base manifold is
		\[ \kappa = \frac{(-1)^r}{3}c^2. \]
		Conversely, if $(M,g)$ admits constant sectional curvature $\kappa = \frac{(-1)^rc^2}{3}, $ 
		\[ 	\widetilde\Omega\left( \widehat{X}^H,\widehat{Y}^H \right) = -\frac{c^2}{3}\left[ l_g\widehat{X},l_g\widehat{Y}\right]. \]
		It implies $(\overline\nabla_\ensuremath{\mathscr{X}} \ensuremath{\mathcal{J}}_c) \ensuremath{\mathscr{X}} = 0 $ for all $\ensuremath{\mathscr{X}} \in \Gamma(TP).$

<!-- lemma-5: body_md -->
\label{lm:lg-gamma}
		For every $X\in \ensuremath{\mathfrak{m}},$
		\begin{equation}\label{eq:lg-gamma}
			l_g(X)=\frac{(\sqrt{-1})^r}{2}\gamma(X).
		\end{equation}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Let $\{X_i\}_{i=0}^2 $ be an orthonormal basis of $\ensuremath{\mathfrak{m}},$ and put $\gamma_i=\gamma(X_i).$
		By the definition of $l_g, $
		\begin{equation*}
			\lambda_* \left( l_g(X_i)\right)(X_j) = X_i \times X_j = \epsilon_{ij}{}^k X_k.
		\end{equation*}
		Derivative of $\gamma(\lambda(h)X) = h\gamma(X)h^{-1} $ yields
		\begin{equation*}
			\gamma\left(\lambda_*(U)X\right)
			=
			[U,\gamma(X)],
			\quad U\in\ensuremath{\mathfrak{h}},\quad X\in\ensuremath{\mathfrak{m}}.
		\end{equation*}
		Moreover, since $ \frac{(\sqrt{-1})^r}{2}\gamma_i\in \ensuremath{\mathfrak{h}}, $
		\begin{align*}
			\left[ \frac{(\sqrt{-1})^r}{2}\gamma_i, \gamma_j \right] 
			& = \epsilon_{ij}{}^k\gamma_k,\\
			\gamma \left( \lambda_*\left( \frac{(\sqrt{-1})^r}{2}\gamma_i\right) X_j \right)  & = \gamma\left( X_i\times X_j \right).
		\end{align*}
		It implies that 
		\[ l_g(X_i) = \frac{(\sqrt{-1})^r}{2}\gamma_i. \]
		The assertion now follows by linearity.

<!-- remark-2: body_md -->
\label{rm:null-spinor}
		The non-null assumption is essential in the
		Lorentzian case. For a nowhere-vanishing null spinor field $\psi$,
		its annihilator
		$\operatorname{Ann}(\psi)=\{V\in TM:V\cdot\psi=0\}$ is a null line
		bundle. Hence a tensor field $A$ satisfying \eqref{eq:GKS}, even if it
		exists, is not unique. Indeed, if $K$ is a local nonzero section of
		$\operatorname{Ann}(\psi)$ and $A$ is symmetric, then
		$A+fK^\flat\otimes K$ is another symmetric solution for every smooth
		function $f$. We therefore exclude null spinors from the
		correspondence below.

<!-- lemma-7: body_md -->
\label{lm:scM-def}

		\begin{equation*}
			\ensuremath{\mathcal{M}}_u \colon \ensuremath{\mathfrak{m}} \ni X\mapsto dF_u(X^H_u)F(u)^{-1}\in \ensuremath{\mathfrak{h}}.

			& = [uh,X],

			& = h^{-1}dF_u(\left( \lambda(h)X\right)^H_u)F(u)^{-1}h\\

<!-- proposition-2: body_md -->
\label{pr:GKS-condition}

			A = \frac{1}{2}l_g^{-1}\ensuremath{\mathbf{M}}.

<!-- theorem-5: body_md -->
\label{th:invariant-GKS-unimodular}
		Let $G$ be a connected three-dimensional Lie group with a
		left-invariant pseudo-Riemannian metric and its $G$-invariant spin structure.
		Then for a fixed $\epsilon \in \ensuremath{\mathcal{E}}_r, $ the following conditions are equivalent:
		\begin{enumerate}
			\item $G$ is unimodular,
			\item some invariant $\epsilon$-unit spinor is a generalized Killing spinor,
			\item every invariant $\epsilon$-unit spinor is a generalized Killing spinor,
			\item some $G$-equivariant section of $P$ is Lagrangian,
			\item every $G$-equivariant section of $P$ is Lagrangian.
		\end{enumerate}
		Moreover, in this case all invariant generalized Killing spinors have
		the same associated symmetric endomorphism.

<!-- proof-15: body_md -->
Let $\{\ensuremath{\mathcal{X}}_k=[s,X_k]\}_{k=0}^2$ be the left-invariant orthonormal frame
		determined by $s$. 
		For every left-invariant vector field $\ensuremath{\mathcal{X}}$, the Levi-Civita connection of a left-invariant metric is given by
		\[ \nabla_\ensuremath{\mathcal{X}} \ensuremath{\mathcal{Y}}=\frac12[\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}}]+U(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}}),\]
		where $U$ is the symmetric bilinear map characterized by
		\[
		2g(U(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}}),\ensuremath{\mathcal{Z}})
		= g([\ensuremath{\mathcal{Z}},\ensuremath{\mathcal{X}}],\ensuremath{\mathcal{Y}})+g(\ensuremath{\mathcal{X}},[\ensuremath{\mathcal{Z}},\ensuremath{\mathcal{Y}}]), \quad \forall \ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{Z}} \in \ensuremath{\mathfrak{X}}^L(G) \simeq \ensuremath{\mathfrak{g}}.
		\]
		\begin{align}
			\operatorname{div} \ensuremath{\mathcal{X}}_j & = g^{lk}g\left( \frac{1}{2}[\ensuremath{\mathcal{X}}_l,\ensuremath{\mathcal{X}}_j]+U(\ensuremath{\mathcal{X}}_l,\ensuremath{\mathcal{X}}_j),\ensuremath{\mathcal{X}}_k \right) \notag\\
			& = g^{lk}g\left( [\ensuremath{\mathcal{X}}_l,\ensuremath{\mathcal{X}}_j],\ensuremath{\mathcal{X}}_k\right)\notag \\
			& = -\operatorname{tr}(\operatorname{ad}_{\ensuremath{\mathcal{X}}_j}), \quad \forall j\in \{0,1,2\}\label{eq:div-free-frame-Lie group}
		\end{align}
		Since $G$ is connected, it is unimodular if and only if for every $\ensuremath{\mathcal{X}}\in \ensuremath{\mathfrak{g}}, $
		\[ \tau(\ensuremath{\mathcal{X}}) = 0. \]
		This is equivalent to the left-invariant oriented orthonormal frame being divergence-free by \prettyref{eq:div-free-frame-Lie group}.
		Since the left-invariant trivialization determines a $G$-invariant spin structure $P=G\times H$, then its left action on itself lifts on $P.$
		Then a section of $\pi\colon P\to G$ defined as $s_f = s\cdot f$ in \prettyref{eq:section-torsor}:
		\begin{equation}\label{eq:s_f-Lie}
			s_f(q) = (q,f(q)) \quad \forall q\in G
		\end{equation} 
		is $G$-equivariant if and only if $f\in C^\infty(G,H) $ is constant.
		By \prettyref{cr:constant-map-GKS}, for every $h_0\in H, $ as the constant image of $f$, the following are equivalent:
		\begin{enumerate}\renewcommand{(\roman{enumi})}{(\roman{enumi})}
			\item the invariant unit-length spinor represented by $f = s^*F \in C^\infty(G,H) $ is a generalized Killing spinor,
			\item the section $s_f$ given in \prettyref{eq:s_f-Lie} is Lagrangian,
			\item $\operatorname{div}(\ensuremath{\mathcal{X}}_k)=0, $ for all $ k\in \{0,1,2\}$.
		\end{enumerate}
		Moreover, these conditions are independent of $h_0 \in H.$
		Consequently, if $G$ is unimodular, every left-invariant oriented orthonormal frame is divergence-free, and hence every invariant spinor of unit length is a generalized Killing spinor and every $G$-equivariant section is Lagrangian.
		Thus
		\[
		(1)\Longrightarrow(3)\Longrightarrow(2),
		\qquad
		(1)\Longrightarrow(5)\Longrightarrow(4).
		\]
		Conversely, the existence of either an invariant generalized Killing spinor of unit length or an equivariant Lagrangian section implies existence of a divergence-free left-invariant oriented orthonormal frame, 
		and therefore implies $G$ is unimodular.
		
		It remains to prove the final assertion. 
		For the representing $H$-valued function $F\in C^\infty(P,H)^H, $ its expression relative to $s$ is the constant map $ f(q) = F\circ s(q) = h_0. $
		Hence $df = 0, $ then by \prettyref{eq:scM-gauge} in \prettyref{lm:scM-def} and \prettyref{lm:lg-Ad-isom},
		\begin{align*}
			\ensuremath{\mathcal{M}}_{s_f(x)}(X_f) & = \operatorname{Ad}_{h_0^{-1}}\ensuremath{\mathcal{M}}_{s(x)}(\lambda(h_0)X_f), \quad X_f = s_f^*\theta_x(\ensuremath{\mathcal{X}})\in \ensuremath{\mathfrak{m}}\\
			&= \operatorname{Ad}_{h_0^{-1}}\ensuremath{\mathcal{M}}_{s(x)}(X), \quad X = s^*\theta_x(\ensuremath{\mathcal{X}})\in \ensuremath{\mathfrak{m}},\\
			[s_f(x),l_g^{-1}\ensuremath{\mathcal{M}}_{s_f(x)}(X_f)] & = [s(x)h_0,\lambda(h_0)^{-1}l_g^{-1}\ensuremath{\mathcal{M}}_{s(x)}(X)] = [s(x),l_g^{-1}\ensuremath{\mathcal{M}}_{s(x)}(X)]\in T_xG,
		\end{align*}
		where $\ensuremath{\mathcal{X}}|_x = [s_f(x), X_f] = [s(x),X]\in T_xG.$
		The symmetric endomorphism associated with the corresponding generalized Killing spinor is consequently given by \[ A(\ensuremath{\mathcal{X}}) = \frac{1}{2}\left[ s, l_g^{-1} (s^*{\widetilde{\omega}})(\ensuremath{\mathcal{X}})\right]. \]
		This expression depends only on the fixed invariant section $s$ and its induced frame, and is independent of $h_0\in H$ and $\epsilon\in \ensuremath{\mathcal{E}}_r.$ Thus all invariant generalized Killing spinors have the same associated symmetric endomorphism.
