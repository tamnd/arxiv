\label{teo}
		Assume that $\Omega\subset\mathbb R^2$ is smooth, bounded, contains the
		origin, and is invariant under the two coordinate reflections, see \eqref{simm}. Let
		$0<\lambda<\lambda_1(\Omega)$ and let $k\ge1$ be fixed. Then, for all
		$\varepsilon>0$ sufficiently small, problem \eqref{main} has a positive solution
		$u_\varepsilon$ with a tower of $k$ bubbles concentrating at the origin.
		
		More precisely, there exist scales
		\[
		d_k\ll d_{k-1}\ll\cdots\ll d_1\longrightarrow0 \quad 
		\]
		and heights 
		\[
		\gamma_j
		=
		\alpha_j|\log\varepsilon|^{1/2}\varepsilon^{1/2-j}(1+o(1)),
		\qquad
		\alpha_j=
		\frac{2^{j-1}(k-j)!}{(k-1)!}\sqrt{\frac2k}
		, \quad j=1, \ldots , k\]
		as $\varepsilon \to 0$, such that, for every fixed $j$ and every
		compact set $K\Subset\mathbb R^2\setminus\{0\}$,
		\[
		p\gamma_j^{p-1}
		\bigl[u_\varepsilon(d_jy)-\widehat\gamma_j\bigr]
		\longrightarrow
		w(y):=\log\frac8{(1+|y|^2)^2}
		\]
		uniformly for $y\in K$. Here $\widehat\gamma_j$ are the effective local levels  and satisfy 
		$\widehat\gamma_j/\gamma_j\to1$ as $\varepsilon \to 0$.
		
		The concentration scales satisfy, as $\varepsilon \to 0$,
		\[
		\log d_j^2
		=
		-\gamma_j^p+O\!\left(\varepsilon\gamma_j^p+\log\gamma_j\right),
		\]
		and in particular
		\[
		\frac{d_j}{d_{j-1}}\longrightarrow0,
		\qquad j=2,\ldots,k.
		\]

\label{lem:robust-scales}
		For every fixed $\delta>0$ and all $m\in[\delta,\delta^{-1}]^k$,
		equation \eqref{defmuj} has, for $\varepsilon$ sufficiently small, a unique solution
		satisfying
		\[
		\frac{L_j}{G_j}=O(\varepsilon)+O\!\left(\frac{\log G_j}{G_j}\right).
		\]
		Moreover
		\begin{align}
			\log\mu_j^2
			&=
			O\!\left(\varepsilon G_j+\log G_j\right),
			\label{robust-mu}\\
			\log d_j^2
			&=
			-G_j+
			O\!\left(\varepsilon G_j+\log G_j\right),
			\label{robust-d}
		\end{align}
		and consequently
		\[
		d_k\ll d_{k-1}\ll\cdots\ll d_1,
		\qquad
		\frac{d_j}{d_{j-1}}\to0.
		\]

The construction is sequential in $j$.  For $j=1$ one has $\eta_1=0$.
		Assume inductively that $L_i$, $\mu_i$, and $d_i$ have already been
		constructed for $i<j$ and satisfy \eqref{robust-mu}--\eqref{robust-d}.
		Then, we clearly have 
		\[
		-4\log d_i-h_\lambda(0)
		=
		2G_i+O(\varepsilon G_i+\log G_i),
		\]
		and therefore
		\[
		|\eta_j|
		\le
		C\sum_{i<j}\gamma_i
		\le C\gamma_{j-1}.
		\]
		Since
		\[
		\frac{\gamma_{j-1}}{\gamma_j}=\varepsilon m_j,
		\]
		we obtain, uniformly in the parameter box \eqref{emme},
		\begin{equation*}\label{eta-rough}
			\frac{\eta_j}{\gamma_j}=O(\varepsilon).
		\end{equation*}
		Also $\Gamma_j=O(\varepsilon)$ directly from \eqref{defGammaj}.
		
		Write $\ell=L/G_j$.  Dividing \eqref{Bjexact} by $G_j$ gives
		\[
		\frac{{\mathcal B}_j(L)}{G_j}
		=
		-2(1+\Gamma_j)\ell
		-\varepsilon-\frac{p\eta_j}{\gamma_j}
		+2\Gamma_j
		+
		O\!\left(\frac{\log G_j}{G_j}\right).
		\]
		Thus
		\[
		{\mathfrak F}_{j,\varepsilon}(\ell)
		:=
		\ell+
		\left[
		1+\frac1p\frac{{\mathcal B}_j(G_j\ell)}{G_j}
		\right]^p-1
		\]
		converges, uniformly for $\ell$ in a fixed small interval, 
		to $\ell+(1-\ell)^2-1.$ Clearly, the zero $\ell=0$ is nondegenerate.  
		
		The implicit function theorem applied to ${\mathfrak F}_{j,\varepsilon}(\ell)=0$, using \eqref{defmuj} gives a unique small branch and
		\[
		\ell
		=
		O(\varepsilon)+O\!\left(\frac{\log G_j}{G_j}\right).
		\]
		This proves \eqref{robust-mu} for the $j$-th scale.  Since
		\[
		\log\rho_j^2=-G_j-\log(\lambda pG_j),
		\]
		formula \eqref{robust-d} follows as well.  The induction is complete.
		
		Finally,
		\[
		\log\frac{d_j^2}{d_{j-1}^2}
		=
		-G_j+G_{j-1}
		+O(\varepsilon G_j+\log G_j).
		\]
		Because
		\[
		G_{j-1}=(\varepsilon m_j)^pG_j=o(G_j),
		\]
		we obtain
		\[
		\log\frac{d_j^2}{d_{j-1}^2}=-(1+o(1))G_j.
		\]
		Hence $d_j/d_{j-1}\to0$.  In fact, since every $G_j$ dominates
		$|\log\varepsilon|$, for every fixed $N>0$,
		\begin{equation*}\label{superalgebraic-separation}
			d_1=O(\varepsilon^N),
			\qquad
			\frac{d_j}{d_{j-1}}=O(\varepsilon^N),
			\quad j=2,\ldots,k.
		\end{equation*}
		This stronger form will be used repeatedly below.

\label{lem:core-expansion}
		Let
		\[
		t=\log\frac{|x|}{d_j},\qquad x=d_jy.
		\]
		Uniformly for
		\[
		|t|\le2R_\varepsilon,\qquad j=1,\ldots,k,
		\]
		one has
		\begin{equation}\label{coarse-local-U}
			\frac{U(d_jy)}{\gamma_j}
			=
			a_j+
			\frac{w(y)-4\Gamma_j\log|y|}{pG_j}
			+r_{j,\varepsilon}^{\,c}(y),
			\qquad
			G_j|r_{j,\varepsilon}^{\,c}(y)|=o(1).
		\end{equation}
		Consequently,
		\begin{align}
			\lambda Ue^{U^p}
			&=
			\frac{b_j}{p}V_j
			\left[
			1+
			O\!\left(
			\varepsilon(1+|t|)
			+\frac{1+t^2}{G_j}
			\right)
			\right],
			\label{core-f}\\
			f_+'(U)-\lambda
			&=
			V_j
			\left[
			1+
			O\!\left(
			\varepsilon(1+|t|)
			+\frac{1+t^2}{G_j}
			\right)
			\right].
			\label{core-fprime}
		\end{align}
		All remainders are uniform in the parameter box.

\label{lem:localization}
		Let $\omega\in\mathbb S^1$ and write
		\[
		U_\omega(s):=U(e^{-s}\omega).
		\]
		For $1\le j\le k-1$ set
		\[
		{\mathcal J}_j=[s_j,s_{j+1}],
		\qquad
		{\mathcal J}_j^{\rm tr}=[s_j+2R_\varepsilon,s_{j+1}-2R_\varepsilon].
		\]
		For all sufficiently small $\varepsilon$, uniformly in the parameter box and in
		$\omega$, the following properties hold.
		
		\begin{enumerate}
			\item The function
			\[
			{\mathcal H}_\omega(s)
			:=-2s+\log(\lambda U_\omega(s))+U_\omega(s)^p
			\]
			is strictly convex on the interval
			\[
			s_j+R_\varepsilon\le s\le s_{j+1}-R_\varepsilon.
			\]
			In particular it has at most one critical point on this interval,
			necessarily a minimum.
			
			\item There are constants $c,C>0$ such that
			\begin{equation}\label{Dtransition-small}
				\sup_{\omega\in\mathbb S^1}\sup_{s\in{\mathcal J}_j^{\rm tr}}
				e^{-2s}\lambda U_\omega(s)e^{U_\omega(s)^p}
				\le C e^{-cR_\varepsilon}(b_j+b_{j+1}).
			\end{equation}
			
			\item For every $j=1,\ldots,k-1$ and every
			$s\in{\mathcal J}_j=[s_j,s_{j+1}]$,
			\begin{equation}\label{betaUbound}
				\beta_\varepsilon(s)U_\omega(s)^{p-1}\le \frac{C}{\varepsilon}.
			\end{equation}
			
			\item Let $s_0=-\log r_0$.  In the outer bridge
			$[s_0,s_1-2R_\varepsilon]$, restricted to $U_\omega\ge1$,
			${\mathcal H}_\omega$ is strictly convex and
			\begin{equation}\label{outer-high-density}
				e^{-2s}\lambda U_\omega(s)e^{U_\omega(s)^p}
				\le C\left[b_1e^{-cR_\varepsilon}+e^{-c/b_1}\right].
			\end{equation}
		\end{enumerate}

Differentiating \eqref{juan}--\eqref{approxisol} along a ray gives, uniformly in $\omega$,
		\begin{equation}\label{Usbridge}
			\partial_sU_\omega(s)
			=
			\frac4p\sum_{i=1}^k b_i\frac{e^{-2s}}{d_i^2+e^{-2s}}
			+O\!\left(e^{-s}\sum_i b_i\right).
		\end{equation}
		The last term comes from the smooth functions $H_i^\lambda$; the reflection
		symmetries actually improve it to quadratic order at the origin, but the
		displayed estimate is sufficient.
		
		If $s_j+R_\varepsilon\le s\le s_{j+1}-R_\varepsilon$, the terms $i\le j$ are
		exponentially small, whereas
		\[
		\sum_{i\ge j+1}b_i=b_{j+1}(1+O(\varepsilon)).
		\]
		Using \eqref{Usbridge} and the scale separation, we first obtain the
		first-derivative estimate
		\begin{equation*}\label{Us-central}
			\partial_sU_\omega
			=
			\frac4p b_{j+1}(1+O(\varepsilon))+o(b_{j+1}).
		\end{equation*}
		Differentiating \eqref{Usbridge} with respect to $s$ gives, in addition,
		the following second-derivative estimate: for every fixed $N>0$,
		\begin{equation}\label{Uss-super-small}
			\sup_{\omega}
			\sup_{s_j+R_\varepsilon\le s\le s_{j+1}-R_\varepsilon}
			|\partial_{ss}U_\omega(s)|
			=
			o(b_{j+1}^{\,N}).
		\end{equation}
		Indeed, after this differentiation every bubble term
		contains a factor $e^{-2R_\varepsilon}$, while the smooth part contains the
		physical factor $e^{-s}$.  Since
		$R_\varepsilon=|\log\varepsilon|^2$ and the scale separation
		\eqref{superalgebraic-separation} is faster than every algebraic power of
		$\varepsilon$, both contributions are $o(b_{j+1}^{\,N})$ for every fixed $N$.
		Moreover $U_\omega\ge c\gamma_j$ in the central bridge.  A direct
		calculation yields
		\begin{align*}
			{\mathcal H}_\omega''
			={}&\frac{\partial_{ss}U_\omega}{U_\omega}
			-\frac{\partial_sU_\omega^2}{U_\omega^2}
			+p(p-1)U_\omega^{p-2}\partial_sU_\omega^2
			+pU_\omega^{p-1}\partial_{ss}U_\omega.
		\end{align*}
		The negative term is $o(b_{j+1}^2)$ because
		$U_\omega\ge c\gamma_j\to\infty$.  Moreover
		$U_\omega^{p-1}\le Cb_{j+1}^{-1}$, so
		\eqref{Uss-super-small} with, for instance, $N=3$ shows that both terms
		containing $\partial_{ss}U_\omega$ are $o(b_{j+1}^2)$.  On the other hand,
		\[
		p(p-1)U_\omega^{p-2}\partial_sU_\omega^2\ge cb_{j+1}^2.
		\]
		Thus ${\mathcal H}_\omega''>0$, uniformly in $\omega$.
		
		Lemma \ref{lem:core-expansion} near $s_j$ gives
		\[
		e^{-2s}\lambda U_\omega e^{U_\omega^p}
		\le Cb_j e^{-c|s-s_j|}
		\]
		for $|s-s_j|\le2R_\varepsilon$, uniformly in $\omega$; the analogous estimate
		holds near $s_{j+1}$.  Strict convexity shows that the density has no
		interior maximum in the central bridge and proves \eqref{Dtransition-small}.
		
		On ${\mathcal J}_j$,
		\[
		U_\omega\le C\gamma_{j+1},\qquad \beta_\varepsilon\le b_j.
		\]
		Since $\gamma_j/\gamma_{j+1}=\varepsilon m_{j+1}$, this gives
		\[
		\beta_\varepsilon U_\omega^{p-1}
		\le C\left(\frac{\gamma_{j+1}}{\gamma_j}\right)^{p-1}
		\le \frac C\varepsilon.
		\]
		
		Finally consider the outer bridge.  Away from the first core,
		\eqref{Usbridge} gives on the set $U_\omega\ge1$
		\[
		\partial_sU_\omega=\frac4p b_1(1+O(\varepsilon))+o(b_1).
		\]
		Exactly as above, for every fixed $N>0$,
		\begin{equation}\label{outer-Uss-super-small}
			\sup_{\omega}
			\sup_{\substack{s_0\le s\le s_1-R_\varepsilon\\U_\omega(s)\ge1}}
			|\partial_{ss}U_\omega(s)|
			=
			o(b_1^{\,N}).
		\end{equation}
		Since
		\[
		p(p-1)U_\omega^{p-2}-U_\omega^{-2}\ge c>0
		\quad\hbox{for }U_\omega\ge1,
		\]
		and $U_\omega^{p-1}\le Cb_1^{-1}$ on the outer bridge,
		\eqref{outer-Uss-super-small} with $N=3$ shows that the terms containing
		$\partial_{ss}U_\omega$ are negligible.  The same calculation therefore gives
		${\mathcal H}_\omega''>0$.  At the core-side
		endpoint the matching expansion gives $Cb_1e^{-cR_\varepsilon}$.  At the other
		endpoint of the high-amplitude region, where $U_\omega=1$, the estimate
		\[
		U_\omega(s)\le Cb_1(1+s)
		\]
		implies $s\ge c/b_1$, so the density is at most $Ce^{-c/b_1}$.  Convexity
		proves \eqref{outer-high-density}.

\label{lem:bridge-moments}
		Fix $R>1$.  For $j=2,\ldots,k$ set
		\[
		A_j=s_{j-1}+\log R,
		\qquad
		B_j=s_j-\log R,
		\qquad
		I_j^R=[A_j,B_j].
		\]
		Let
		\[
		{\mathcal V}_\varepsilon=\lambda+\sum_{i=1}^kV_i
		\]
		and
		\[
		g_j(s)
		=
		e^{-2s}\left[
		2{\mathcal V}_\varepsilon(e^{-s})\beta_\varepsilon(s)+\overline\varrho_\varepsilon(e^{-s})
		\right].
		\]
		Then $g_j$ admits a decomposition
		\[
		g_j=g_j^{\,L}+g_j^{\,R}+g_j^{\,M},
		\qquad g_j^{\,L},g_j^{\,R},g_j^{\,M}\ge0,
		\]
		where $g_j^{\,L}$ is supported in the left half of $I_j^R$,
		$g_j^{\,R}$ in the right half, and
		\begin{align}
			\int_{A_j}^{B_j}(t-A_j)g_j^{\,L}(t)\,dt
			&\le \eta_R\,\beta_\varepsilon(A_j),
			\label{leftmoment}\\
			\int_{A_j}^{B_j}(B_j-t)g_j^{\,R}(t)\,dt
			&\le \eta_R\,\beta_\varepsilon(B_j),
			\label{rightmoment}\\
			(B_j-A_j)^2\|g_j^{\,M}\|_\infty
			&\le o_\varepsilon(1)\min\{\beta_\varepsilon(A_j),\beta_\varepsilon(B_j)\},
			\label{middlemoment}
		\end{align}
		with
		\[
		\lim_{R\to\infty}\limsup_{\varepsilon\to0}\eta_R=0.
		\]
		
		On the outer bridge
		\[
		I_1^R=[s_0,s_1-\log R],
		\qquad s_0=-\log r_0,
		\]
		the same three estimates hold with $\beta_\varepsilon\equiv b_1$ and with a
		constant $\eta_R^{\rm out}(r_0)$ satisfying
		\begin{equation}\label{eta-outer}
			\limsup_{\varepsilon\to0}\eta_R^{\rm out}(r_0)
			\le Cr_0^2+\eta_R.
		\end{equation}
		Thus $\eta_R^{\rm out}(r_0)$ can be made arbitrarily small by choosing
		first $r_0$ small and then $R$ large.

We give the decomposition explicitly.  Let
		\[
		s_j^*:=\frac{A_j+B_j}{2},
		\qquad
		\ell_j:=B_j-A_j.
		\]
		On an inter-bubble bridge $I_j^R$, $j\ge2$, the low-amplitude transition
		density vanishes for small $\varepsilon$.  Since all the remaining terms in
		$\varrho_\varepsilon$ are radial except for ${\mathcal D}^{\rm hi}_\varepsilon$, write
		\begin{align}
			g_j^{\rm rad}(s)
			:={}&e^{-2s}\left[
			2{\mathcal V}_\varepsilon(e^{-s})\beta_\varepsilon(s)+b_1
			+\sum_{i=1}^k b_iV_i(e^{-s})
			\Theta_{i,\varepsilon}(s_i-s)
			\right],
			\label{qj-bridge}\\
			g_j^{\rm tr}(s)
			:={}&\varepsilon^{-2}e^{-2s}
			\sup_{\omega\in\mathbb S^1}
			{\mathcal D}^{\rm hi}_\varepsilon(e^{-s}\omega).
			\label{hj-bridge}
		\end{align}
		Then $g_j=g_j^{\rm rad}+g_j^{\rm tr}$.  We set
		\begin{equation*}\label{g-decomposition}
			g_j^{\,L}:={\bf 1}_{[A_j,s_j^*]}g_j^{\rm rad},
			\qquad
			g_j^{\,R}:={\bf 1}_{(s_j^*,B_j]}g_j^{\rm rad},
			\qquad
			g_j^{\,M}:=g_j^{\rm tr}.
		\end{equation*}
		This is the decomposition appearing in the statement.
		
		We first record two elementary facts about the affine bridge weight.
		From \eqref{robust-d}, \eqref{explicitgamma}, and
		$b_j=\gamma_j^{1-p}$,
		\begin{equation}\label{beta-endpoints}
			\beta_\varepsilon(A_j)=b_{j-1}(1+o(1)),
			\qquad
			\beta_\varepsilon(B_j)=b_j(1+o(1)),
		\end{equation}
		and
		\begin{equation*}\label{beta-slope}
			\frac{|\beta_\varepsilon'|}{b_j}
			=
			O\!\left(
			\frac{b_{j-1}/b_j}{s_j-s_{j-1}}
			\right)
			=o(1).
		\end{equation*}
		Indeed $b_{j-1}/b_j$ is only an algebraic power of $\varepsilon^{-1}$, whereas
		$s_j-s_{j-1}=\frac12G_j(1+o(1))$ grows faster than that power.  It follows
		that
		\begin{equation}\label{beta-left-right}
			\beta_\varepsilon(s)\le \beta_\varepsilon(A_j)
			\quad(A_j\le s\le s_j^*),
		\end{equation}
		and, on the right half,
		\begin{equation}\label{beta-right-growth}
			\beta_\varepsilon(s)
			\le
			C\beta_\varepsilon(B_j)\,[1+B_j-s].
		\end{equation}
		
		We now estimate the left moment.  The bubble adjacent to the left endpoint
		is the $(j-1)$-st one.  Put $\tau=s-s_{j-1}$.  Since
		$A_j=s_{j-1}+\log R$, one has $\tau\ge\log R$, and
		\begin{equation}\label{sechidentity}
			e^{-2s}V_{j-1}(e^{-s})
			=2\,\operatorname{sech}^2\tau
			\le Ce^{-2\tau}.
		\end{equation}
		By \eqref{beta-left-right}, \eqref{beta-endpoints}, and
		$1/(\varepsilon G_i)\le C$ for every $i$, the contribution of this bubble to the
		left moment is bounded by
		\begin{align}
			C\beta_\varepsilon(A_j)
			\bigg[&\int_{\log R}^{\infty}
			\tau\,(1+\tau+\tau^2)e^{-2\tau}\,d\tau
			\nonumber\\
			&+\varepsilon^{-1}\int_{R_\varepsilon}^{\infty}
			\tau e^{-2\tau}\,d\tau\bigg].
			\label{left-neighbour-estimate}
		\end{align}
		The first integral tends to zero as $R\to\infty$, uniformly in $\varepsilon$;
		the second is $o(\varepsilon^N)$ for every fixed $N$ because
		$R_\varepsilon=|\log\varepsilon|^2$.
		
		All bubbles with $i<j-1$ are farther to the left, while those with
		$i\ge j$ are separated from the left half by at least a fixed fraction of
		the logarithmic distance $s_j-s_{j-1}$.  Using
		\eqref{sechidentity} with the corresponding centers and the
		superalgebraic separation \eqref{superalgebraic-separation}, the sum of
		all these contributions is
		\begin{equation}\label{left-other-bubbles}
			o_\varepsilon(1)\,\beta_\varepsilon(A_j).
		\end{equation}

		$2\lambda e^{-2s}\beta_\varepsilon(s)$ satisfy

			\int_{A_j}^{s_j^*}(s-A_j)e^{-2s}

			&=o_\varepsilon(1)\beta_\varepsilon(A_j).
			\label{left-constant-terms}
		\end{align}

		proves \eqref{leftmoment}.

		minor difference.  The adjacent bubble is now the $j$-th one.  Put
		$\tau=s_j-s$; then $\tau\ge\log R$ on the right half and
		$B_j-s\le\tau$.  By \eqref{beta-right-growth},

		\beta_\varepsilon(s)

		b_j\le C\beta_\varepsilon(B_j).

			&+\varepsilon^{-1}\int_{R_\varepsilon}^{\infty}

		e^{-2s}

		\le
		Ce^{-cR_\varepsilon}(b_{j-1}+b_j).
		\]

		\begin{equation}\label{middle-direct}
			\ell_j^2\|g_j^{\,M}\|_\infty
			\le
			C\ell_j^2\varepsilon^{-2}e^{-cR_\varepsilon}(b_{j-1}+b_j).
		\end{equation}

		$\varepsilon^{-1}$, while $e^{-cR_\varepsilon}=e^{-c|\log\varepsilon|^2}$ decays faster than

		This completes the proof for $j=2,\ldots,k$.

		A_1=s_0,
		\qquad
		B_1=s_1-\log R,
		\qquad
		M_1=\frac{A_1+B_1}{2}.
		\]
		Here $\beta_\varepsilon\equiv b_1$.  In the definition of $q_1$ we include, in
		addition to the terms in \eqref{qj-bridge}, the low-amplitude contribution
		\[
		\varepsilon^{-1}e^{-2s}
		\sup_{\omega}{\mathcal D}^{\rm lo}_\varepsilon(e^{-s}\omega),
		\]

			\int_{A_1}^{M_1}(s-A_1)e^{-2s}b_1\,ds
			\le Cr_0^2b_1.

		\begin{equation}\label{outer-low-moment}
			\varepsilon^{-1}
			\int_{A_1}^{B_1}(1+s)e^{-2s}
			\sup_{\omega}{\mathcal D}^{\rm lo}_\varepsilon(e^{-s}\omega)\,ds

Let $P_j$ solve
		\[
		-P_j''=g_j,
		\qquad
		P_j(A_j)=P_j(B_j)=0,
		\]
		where $g_j$ is given in Lemma \ref{lem:bridge-moments}.  Positivity follows
		from the one-dimensional maximum principle.  The Green function on
		$[A_j,B_j]$ is
		\[
		G(s,t)=
		\begin{cases}
			\dfrac{(t-A_j)(B_j-s)}{B_j-A_j},&t\le s,\\[2mm]
			\dfrac{(s-A_j)(B_j-t)}{B_j-A_j},&t\ge s.
		\end{cases}
		\]
		For every $s,t\in[A_j,B_j]$ the Green function satisfies the two
		bounds
		\begin{equation*}\label{Green-affine-bounds}
			G(s,t)
			\le
			\frac{B_j-s}{B_j-A_j}(t-A_j),
			\qquad
			G(s,t)
			\le
			\frac{s-A_j}{B_j-A_j}(B_j-t).
		\end{equation*}
		Indeed, one inequality is an equality on each side of $t=s$, and the
		other follows from $t\le s$ or $t\ge s$, respectively.  Applying the first
		bound to $g_j^{\,L}$ and the second to $g_j^{\,R}$ gives
		\[
		P_j^{L}(s)
		\le
		\widehat\eta_R\frac{B_j-s}{B_j-A_j}\beta_\varepsilon(A_j),

		P_j^{R}(s)
		\le
		\widehat\eta_R\frac{s-A_j}{B_j-A_j}\beta_\varepsilon(B_j).
		\]
		For the middle contribution,
		\[
		\int_{A_j}^{B_j}G(s,t)\,dt
		=
		\frac{(s-A_j)(B_j-s)}2
		\le\frac{(B_j-A_j)^2}{8},
		\]

		$\beta_\varepsilon(s)\ge\min\{\beta_\varepsilon(A_j),\beta_\varepsilon(B_j)\}$.  Finally,
		$\beta_\varepsilon$ is affine on the interval:
		\[
		\beta_\varepsilon(s)
		=
		\frac{B_j-s}{B_j-A_j}\beta_\varepsilon(A_j)
		+
		\frac{s-A_j}{B_j-A_j}\beta_\varepsilon(B_j).
		\]
		After increasing $R$ and decreasing $\varepsilon_0$ if necessary, we conclude
		\[
		P_j\le\widehat\eta_R\beta_\varepsilon.
		\]
		The choice of $r_0$ and $R$ gives \eqref{Pbound}.
		
		For a radial function $F(s)$,
		\[
		\Delta F=e^{2s}F''.
		\]
		Since $\beta_\varepsilon''=0$ on a bridge and
		\[
		-P_j''

			L{\mathcal S}_j
			&=
			-2{\mathcal V}_\varepsilon\beta_\varepsilon-\overline\varrho_\varepsilon
			+{\mathcal V}_\varepsilon(\beta_\varepsilon+P_j)\\
			&=
			-{\mathcal V}_\varepsilon\beta_\varepsilon-\overline\varrho_\varepsilon+{\mathcal V}_\varepsilon P_j\\
			&\le
			-(1-\widehat\eta_R){\mathcal V}_\varepsilon\beta_\varepsilon-\overline\varrho_\varepsilon
			\le -(1-\widehat\eta_R){\mathcal V}_\varepsilon\beta_\varepsilon-\varrho_\varepsilon,
		\end{align*}

		\[

		\psi\ge0\quad\hbox{on }\partial D,

		{\mathcal S}_j\Delta\!\left(\frac\psi{{\mathcal S}_j}\right)

		\frac\psi{{\mathcal S}_j}L{\mathcal S}_j
		>0,
		\]
		a contradiction.  Thus the strict positive supersolution restores the
		maximum principle despite the positive zeroth-order potential.

Set
		\[
		\Lambda_\varepsilon:=e^{{\mathcal T}_\varepsilon+S_\varepsilon}.
		\]
		On the support of $\chi_{j,\varepsilon}$,
		\[
		\frac{d_j}{\Lambda_\varepsilon}
		\le |x|\le
		d_j\Lambda_\varepsilon.
		\]
		From the definition of $E$,
		\begin{equation}\label{Edecomp-proj}
			E
			=
			\frac{b_j}{p}V_j(Q_j-1)
			-\frac1p\sum_{i\ne j}b_iV_i
			-\lambda U.
		\end{equation}
		
		We first estimate the cross-bubble potentials.  If $i<j$, then
		$|x|\ll d_i$ throughout the support of $\chi_{j,\varepsilon}$ and
		\[
		V_i(x)\le\frac{C}{d_i^2}.
		\]
		Consequently
		\begin{equation*}\label{cross-outer}
			b_j^{-1}b_i
			\int_{\operatorname{supp}\chi_{j,\varepsilon}}V_i
			\le
			C\frac{b_i}{b_j}
			\left(\frac{d_j\Lambda_\varepsilon}{d_i}\right)^2
			=o(G_1^{-1}).
		\end{equation*}
		If $i>j$, then $d_i\ll |x|$ and
		\[
		V_i(x)\le C\frac{d_i^2}{|x|^4}.
		\]
		Hence
		\begin{equation*}\label{cross-inner}
			b_j^{-1}b_i
			\int_{\operatorname{supp}\chi_{j,\varepsilon}}V_i
			\le
			C\frac{b_i}{b_j}
			\left(\frac{d_i\Lambda_\varepsilon}{d_j}\right)^2
			=o(G_1^{-1}).
		\end{equation*}
		To justify the last relations, observe that every quotient $b_i/b_j$ is
		at most an algebraic power of $\varepsilon^{-1}$, whereas
		\eqref{robust-d} gives exponential separation of the $d_i$'s.  The factor
		\[
		\log\Lambda_\varepsilon
		=
		2|\log\varepsilon|+\frac12|\log\varepsilon|^2
		\]
		is negligible compared with the relevant $G_\ell$.
		
		The term $-\lambda U$ is even smaller.  The local expansion
		\eqref{local-U-exact} gives $|U|\le C\gamma_j$ on the support of
		$\chi_{j,\varepsilon}$; therefore
		\begin{equation*}\label{lambdaU-proj}
			\gamma_j^{p-1}
			\int_{\operatorname{supp}\chi_{j,\varepsilon}}\lambda|U|
			\le
			C G_jd_j^2\Lambda_\varepsilon^2
			=o(G_1^{-1}).
		\end{equation*}
		
		It remains to compare the principal term with $\Pi_j^\varepsilon$.  Since
		$\chi_{j,\varepsilon}\equiv1$ for $|t|\le {\mathcal T}_\varepsilon$, $t=\log(|x|/d_j)$,
		the difference is supported where
		\[
		{\mathcal T}_\varepsilon\le |t|\le {\mathcal T}_\varepsilon+S_\varepsilon.
		\]
		The exact core expansion used in Proposition \ref{prop:error} yields there
		\[
		|Q_j-1|
		\le
		C\left[
		\varepsilon(1+|t|)
		+\frac{1+t^2}{G_j}
		\right].
		\]
		In logarithmic coordinates,
		\[
		V_j(x)\,dx
		\le
		C e^{-2|t|}\,dt\,d\theta.
		\]
		Thus
		\begin{align}
			\int_{{\mathcal T}_\varepsilon\le|t|\le {\mathcal T}_\varepsilon+S_\varepsilon}
			V_j|Q_j-1|
			&\le
			C e^{-2{\mathcal T}_\varepsilon}
			\left[
			\varepsilon(1+{\mathcal T}_\varepsilon+S_\varepsilon)
			+\frac{1+({\mathcal T}_\varepsilon+S_\varepsilon)^2}{G_j}
			\right]
			\nonumber\\
			&=
			o(G_1^{-1}),
			\label{Qtail-proj}
		\end{align}
		because $e^{-2{\mathcal T}_\varepsilon}=\varepsilon^4$.  Combining
		\eqref{Edecomp-proj}--\eqref{Qtail-proj} proves
		\eqref{Eprojection-exact}.
