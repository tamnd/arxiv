<!-- proposition-1: body_md -->
\label{mb:prpsn1}
For each $\zeta \in \sigma$ the spaces $X_\zeta = P_\zeta X$ and $Y_\zeta = Q_\zeta Y$ are closed Banach subspaces with $X = \bigoplus_{\zeta \in \sigma} X_\zeta$ and $Y = \bigoplus_{\zeta \in \sigma}Y_\zeta$ represented by finite internal topological direct sums.  Equivalently the finite coordinate maps $\Phi_X: X \to \prod_{\zeta \in \sigma} X_\zeta$ and $\Phi_Y: Y \to \prod_{\zeta \in \sigma} Y_\zeta$ defined by $\mbox{$\mbox{\boldmath $x$}$} \mapsto ( \mbox{$\mbox{\boldmath $x$}$}_\zeta )_{\zeta \in \sigma}$ and $\mbox{$\mbox{\boldmath $y$}$} \mapsto ( \mbox{$\mbox{\boldmath $y$}$}_\zeta)_{\zeta \in \sigma}$, where $\mbox{$\mbox{\boldmath $x$}$}_\zeta = P_\zeta \mbox{$\mbox{\boldmath $x$}$}$ and $\mbox{$\mbox{\boldmath $y$}$}_\zeta = Q_\zeta \mbox{$\mbox{\boldmath $y$}$}$, with product space norms defined by $\lVert (\mbox{$\mbox{\boldmath $x$}$}_\zeta)_{\zeta \in \sigma}\rVert = \max_{\zeta \in \sigma} \lVert \mbox{$\mbox{\boldmath $x$}$}_\zeta \rVert$ and $\lVert (\mbox{$\mbox{\boldmath $y$}$}_\zeta)_{\zeta \in \sigma}\rVert = \max_{\zeta \in \sigma} \lVert \mbox{$\mbox{\boldmath $y$}$}_\zeta \rVert$, are Banach space isomorphisms with inverse mappings defined by $(\mbox{$\mbox{\boldmath $x$}$}_\zeta)_{\zeta \in \sigma} \mapsto \sum_{\zeta \in \sigma} \mbox{$\mbox{\boldmath $x$}$}_\zeta$ and $(\mbox{$\mbox{\boldmath $y$}$}_\zeta)_{\zeta \in \sigma} \mapsto \sum_{\zeta \in \sigma} \mbox{$\mbox{\boldmath $y$}$}_\zeta$.

<!-- head-1: body_md -->
The subspaces $X_\zeta = P_\zeta(X)$ and $Y_\zeta = Q_\zeta(Y)$ are closed because $P_\zeta$ and $Q_\zeta$ are bounded projections.  The partitions of unity $\sum_{\zeta \in \sigma} P_\zeta = I_X$ and $\sum_{\zeta \in \sigma} Q_\zeta = I_Y$ and mutual annihilation properties $P_\zeta P_\eta = \scalebox{0.9}{\mbox{\large $0$}}_X$ and $Q_\zeta Q_\eta = \scalebox{0.9}{\mbox{\large $0$}}_Y$ when $\eta \neq \zeta$ confirm the existence and uniqueness of the decompositions.  The boundedness of the coordinate map and its inverse follows from the finiteness of the spectral set and the open mapping theorem. Pointwise application of the coordinate maps for all $t \in {\mathbb Z}$ gives $X^{\mathbb Z} \cong \prod_{\zeta \in \sigma} X_\zeta^{\mathbb Z}$ and $Y^{\mathbb Z} \cong \prod_{\zeta \in \sigma} Y_\zeta^{\mathbb Z}$.  Because $\sigma$ is finite the external direct sum and Cartesian product coincide as vector spaces.  $\hfill \Box

<!-- lemma-1: body_md -->
\label{mr:lem1}
Let $U$ be a Banach space and let $G \in {\mathcal B}(U)$.  Define
\begin{equation*}
	{\mathcal C}_G(U) = \{ \mbox{$\mbox{\boldmath $u$}$}_0 \in U \mid \exists\ \{ \mbox{$\mbox{\boldmath $u$}$}_p \}_{p \in {\mathbb N}} \subset U \mbox{ with } \mbox{$\mbox{\boldmath $u$}$}_{p-1} = G(\mbox{$\mbox{\boldmath $u$}$}_p) \mbox{ for all } p \in {\mathbb N} \}.
\end{equation*}
Then:
\begin{enumerate}[(i)]
\item the algebraic core ${\mathcal C}_G(U)$ is a linear subspace;
\item $G({\mathcal C}_G(U)) = {\mathcal C}_G(U)$;
\item ${\mathcal C}_G(U)$ is the largest linear subspace $M \subset U$ satisfying $G(M) = M$;
\item if $G$ is injective on ${\mathcal C}_G(U)$ then $G\vert_{{\mathcal C}_G(U)}:{\mathcal C}_G(U) \to {\mathcal C}_G(U)$ is bijective and the compatible orbit $G^{-p}\mbox{$\mbox{\boldmath $u$}$}$ is algebraically uniquely defined\,\footnote{The space ${\mathcal C}_G(U)$ need not be closed and there is no assertion that $G^{-1}$ is bounded on ${\mathcal C}_G(U)$.} for each $\mbox{$\mbox{\boldmath $u$}$} \in {\mathcal C}_G(U)$;
\item ${\mathcal C}_G(U) \subset G^\infty(U) \coloneqq \bigcap_{n \in {\mathbb N}} G^n(U)$ with ${\mathcal C}_G(U) = G^\infty(U)$ whenever $G(G^\infty(U)) = G^\infty(U)$ which is true, in particular, if $G$ is injective on $U$;
\item if $G$ is quasinilpotent with $\lVert G^p \rVert^{1/p} \to 0$ as $p \to \infty$ and $\mbox{$\mbox{\boldmath $u$}$}_0 \neq \mbox{$\mbox{\boldmath $0$}$}$ with $\mbox{$\mbox{\boldmath $u$}$}_{p-1} = G \mbox{$\mbox{\boldmath $u$}$}_p$ for all $p \in {\mathbb N}$ then $\lVert \mbox{$\mbox{\boldmath $u$}$}_p \rVert^{1/p} \to \infty$ as $p \to \infty$ and the compatible orbit $\{ \mbox{$\mbox{\boldmath $u$}$}_p\}_{p \in {\mathbb N}}$ is not subexponential;
\item if $G$ is nilpotent ${\mathcal C}_G(U) = \{ \mbox{$\mbox{\boldmath $0$}$} \}$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- head-2: body_md -->
Let \(M={\mathcal C}_G(U)\).

(i) The zero sequence $\{ \mbox{$\mbox{\boldmath $u$}$}_p\}_{p \in {\mathbb N}-1}$ with $\mbox{$\mbox{\boldmath $u$}$}_p = \mbox{$\mbox{\boldmath $0$}$}$ for all $p \in \mathbb{N}-1$ satisfies $\mbox{$\mbox{\boldmath $u$}$}_{p-1} = G\mbox{$\mbox{\boldmath $u$}$}_p$, so $\mbox{$\mbox{\boldmath $0$}$} \in M$.  If \(\mbox{$\mbox{\boldmath $u$}$}_0, \mbox{$\mbox{\boldmath $v$}$}_0 \in M\), they have compatible orbits \(\{\mbox{$\mbox{\boldmath $u$}$}_p\}_{p \in {\mathbb N}}\) and \(\{\mbox{$\mbox{\boldmath $v$}$}_p\}_{p \in {\mathbb N}}\) satisfying $\mbox{$\mbox{\boldmath $u$}$}_{p-1} = G\mbox{$\mbox{\boldmath $u$}$}_p$ and $\mbox{$\mbox{\boldmath $v$}$}_{p-1} = G\mbox{$\mbox{\boldmath $v$}$}_p$.  Since \(G(\alpha \mbox{$\mbox{\boldmath $u$}$}_{p}+ \beta \mbox{$\mbox{\boldmath $v$}$}_{p})= \alpha G\mbox{$\mbox{\boldmath $u$}$}_{p}+ \beta G\mbox{$\mbox{\boldmath $v$}$}_{p}= \alpha \mbox{$\mbox{\boldmath $u$}$}_{p-1}+\beta \mbox{$\mbox{\boldmath $v$}$}_{p-1}\) it follows that $\alpha \mbox{$\mbox{\boldmath $u$}$}_0 + \beta \mbox{$\mbox{\boldmath $v$}$}_0$ also has a compatible orbit $\{\alpha \mbox{$\mbox{\boldmath $u$}$}_p + \beta \mbox{$\mbox{\boldmath $v$}$}_p\}_{p \in {\mathbb N}}$ and so $\alpha \mbox{$\mbox{\boldmath $u$}$}_0 + \beta \mbox{$\mbox{\boldmath $v$}$}_0 \in M$ for all $\alpha, \beta \in {\mathbb C}$.  Therefore $M$ is a linear subspace.

(ii) Let \(\mbox{$\mbox{\boldmath $u$}$}_0 \in M\). By the definition of $M$, there exists a compatible orbit \(\{\mbox{$\mbox{\boldmath $u$}$}_{p}\}_{p \in {\mathbb N}}\) for $\mbox{$\mbox{\boldmath $u$}$}_0$ such that $\mbox{$\mbox{\boldmath $u$}$}_{p-1}=G \mbox{$\mbox{\boldmath $u$}$}_p$.  The sequence \(\{\mbox{$\mbox{\boldmath $u$}$}_{p+1}\}_{p \in {\mathbb N}}\) forms a compatible orbit for \(\mbox{$\mbox{\boldmath $u$}$}_{1}\) because \(\mbox{$\mbox{\boldmath $u$}$}_p = G\mbox{$\mbox{\boldmath $u$}$}_{p+1}\) for all \(p \in {\mathbb N}\).  Therefore $\mbox{$\mbox{\boldmath $u$}$}_1 \in M \Rightarrow \mbox{$\mbox{\boldmath $u$}$}_0 = G \mbox{$\mbox{\boldmath $u$}$}_1 \in G(M)\).  Thus $M \subset G(M)$.  Let \(\mbox{$\mbox{\boldmath $v$}$} = G \mbox{$\mbox{\boldmath $u$}$}_0 \in G(M)\) for some \(\mbox{$\mbox{\boldmath $u$}$}_0 \in M\).  Since \(\mbox{$\mbox{\boldmath $u$}$}_0 \in M\), it has a compatible orbit \(\{\mbox{$\mbox{\boldmath $u$}$}_p \}_{p \in {\mathbb N}}\).  Therefore \(\mbox{$\mbox{\boldmath $v$}$}_0 = \mbox{$\mbox{\boldmath $v$}$}\) and \(\mbox{$\mbox{\boldmath $v$}$}_p = \mbox{$\mbox{\boldmath $u$}$}_{p-1}\) for \(p \in {\mathbb N}\) defines a compatible orbit $\{ \mbox{$\mbox{\boldmath $v$}$}_p \}_{p \in {\mathbb N}}$ for $\mbox{$\mbox{\boldmath $v$}$}$.  Hence \(\mbox{$\mbox{\boldmath $v$}$} \in M\).  Thus $G(M) \subset M$.  It follows that $G(M) = M$.

(iii) Let \(L \subset U\) be any linear subspace such that \(G(L) = L\). Let $\mbox{$\mbox{\boldmath $u$}$}_0 \in L$.  Because \(G(L) = L\) there exists \(\mbox{$\mbox{\boldmath $u$}$}_1 \in L\) such that \(G \mbox{$\mbox{\boldmath $u$}$}_1 = \mbox{$\mbox{\boldmath $u$}$}_0\).  By mathematical induction there exists \(\{\mbox{$\mbox{\boldmath $u$}$}_p\}_{p \in \mathbb{N}} \subset L\) such that \(\mbox{$\mbox{\boldmath $u$}$}_{p-1} = G \mbox{$\mbox{\boldmath $u$}$}_p\) for all \(p \in \mathbb{N}\).  Thus, \(\mbox{$\mbox{\boldmath $u$}$}_0 \in M\). It follows that \(L \subset M\).  Therefore \(M\) is the largest linear subspace with $G(M)=M$.

(iv) If $\mbox{$\mbox{\boldmath $u$}$} \in M$ there is a unique image $\mbox{$\mbox{\boldmath $w$}$} = G \mbox{$\mbox{\boldmath $u$}$} \in G(M)$ for all $\mbox{$\mbox{\boldmath $u$}$} \in M$.  Now suppose $\mbox{$\mbox{\boldmath $w$}$} \in G(M)$.  If $G$ is injective on $M$ then $G \mbox{$\mbox{\boldmath $v$}$} = G \mbox{$\mbox{\boldmath $u$}$} \Rightarrow \mbox{$\mbox{\boldmath $v$}$} = \mbox{$\mbox{\boldmath $u$}$}$.  Let $\mbox{$\mbox{\boldmath $u$}$} = G^{-1} \mbox{$\mbox{\boldmath $w$}$}$ denote the unique preimage for each $w \in G(M)$.  Thus $G\vert_M: M \to M$ is bijective.  Since $M = G(M)$ we can apply these arguments repeatedly to show that each $\mbox{$\mbox{\boldmath $u$}$} \in M$ has a collection of uniquely defined preimages $G^{-p} \mbox{$\mbox{\boldmath $u$}$}$ for all $p \in {\mathbb N}$.   

(v) If $G$ is injective on $U$ then $\mbox{$\mbox{\boldmath $v$}$} \in G^\infty(U) \subset U \Rightarrow \mbox{$\mbox{\boldmath $u$}$} = G^{-1}\mbox{$\mbox{\boldmath $v$}$}$ is uniquely defined and $\mbox{$\mbox{\boldmath $u$}$} \in G^\infty(U) \Rightarrow \mbox{$\mbox{\boldmath $v$}$} = G \mbox{$\mbox{\boldmath $u$}$} \in G(G^\infty(U)) \Rightarrow G^\infty(U) \subset G(G^\infty(U))$.  Now the obvious inclusion $G(G^\infty(U)) \subset G^\infty(U) \Rightarrow G^\infty(U) = G(G^\infty(U))$.  From (iii) the condition $G(G^\infty(U)) = G^\infty(U) \Rightarrow G^\infty(U) \subset M$.  If $\mbox{$\mbox{\boldmath $u$}$}_0 \in M$ there exists $\{ \mbox{$\mbox{\boldmath $u$}$}_p \}_{p \in {\mathbb N}} \subset U$ with $\mbox{$\mbox{\boldmath $u$}$}_0 = G^p \mbox{$\mbox{\boldmath $u$}$}_p$ for all $p \in {\mathbb N}$.  Therefore $M \subset  G^\infty(U)$.  Thus $G^\infty(U) = M$.

(vi) We have $G^p \mbox{$\mbox{\boldmath $u$}$}_p = \mbox{$\mbox{\boldmath $u$}$}_0 \neq \mbox{$\mbox{\boldmath $0$}$}$ for all $p \in {\mathbb N}$.  Therefore $\lVert \mbox{$\mbox{\boldmath $u$}$}_0 \rVert \leq \lVert G^p \rVert \cdot \lVert \mbox{$\mbox{\boldmath $u$}$}_p \rVert$.  If $\lVert G^p \rVert^{1/p} \to 0$ as $p \to \infty$ then $\lVert \mbox{$\mbox{\boldmath $u$}$}_p \rVert^{1/p} \geq \| \mbox{$\mbox{\boldmath $u$}$}_0 \|^{1/p} /\lVert G^p \rVert^{1/p} \to \infty$ as $p \to \infty$.

(vii) If $G$ is nilpotent there exists $q \in {\mathbb N}$ such that $G^p \mbox{$\mbox{\boldmath $u$}$} = \mbox{$\mbox{\boldmath $0$}$}$ for all $\mbox{$\mbox{\boldmath $u$}$} \in U$ and all $p \in {\mathbb N}+q$.  Therefore ${\mathcal C}_G(U) = \{ \mbox{$\mbox{\boldmath $0$}$} \}$.   $\hfill \Box

<!-- theorem-1: body_md -->
\label{mr:thm1}
	Choose $t_0 \in {\mathbb Z}$.
	\begin{enumerate}[(i)]
		\item If $\zeta \in \sigma$ and $\lvert \zeta \rvert \in (1, \infty)$ each $\mbox{$\mbox{\boldmath $c$}$}_{\zeta, t_0, -\infty} \in X_\zeta$ defines a unique forward flow solution
	\begin{equation}
		\label{mr:e1}
		\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t} = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\nabla$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t} + {H_\zeta}^{-(t-t_0)} \mbox{$\mbox{\boldmath $c$}$}_{\zeta,t_0,-\infty} \in X_\zeta \quad \mbox{ for all } t \in {\mathbb Z}
		\end{equation}
	where $\mbox{$\mbox{\boldmath $c$}$}_{\zeta,t_0,-\infty} = \lim_{r \rightarrow \infty} {H_\zeta}^{-r-t_0} \mbox{$\mbox{\boldmath $x$}$}_{\zeta,-r}$ represents a boundary value in the far distant past.  The flow is subexponential if and only if $\mbox{$\mbox{\boldmath $c$}$}_{\zeta, t_0, -\infty} = \mbox{$\mbox{\boldmath $0$}$}$.
	
	\item If $\zeta \in \sigma$ and $\lvert \zeta \rvert \in (0,1)$ each $\mbox{$\mbox{\boldmath $c$}$}_{\zeta, t_0, \infty} \in X_\zeta$ defines a unique backward flow solution
	\begin{equation}
		\label{mr:e2}
		\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t} = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\Delta$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t+k} + {H_\zeta}^{-(t-t_0)} \mbox{$\mbox{\boldmath $c$}$}_{\zeta,t_0,\infty} \in X_\zeta  \quad \mbox{ for all } t \in {\mathbb Z}
	\end{equation}
	where $\mbox{$\mbox{\boldmath $c$}$}_{\zeta, t_0, \infty} = \lim_{r \rightarrow \infty} {H_\zeta}^{r - t_0} \mbox{$\mbox{\boldmath $x$}$}_{\zeta, r}$ represents a boundary value in the far distant future.  The flow is subexponential if and only if $\mbox{$\mbox{\boldmath $c$}$}_{\zeta, t_0, \infty} = \mbox{$\mbox{\boldmath $0$}$}$.
	
	\item If $\zeta \in \sigma$ and $\lvert \zeta \rvert = 1$ each $\mbox{$\mbox{\boldmath $c$}$}_{\zeta,t_0} = \mbox{$\mbox{\boldmath $x$}$}_{\zeta,t_0} \in X_\zeta$ defines a unique outward flow solution
	\begin{equation}
		\label{mr:e3}
		\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t} = \left\{ \begin{array}{ll}
			\mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\nabla$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^+,t} + {H_\zeta}^{-(t-t_0)} \mbox{$\mbox{\boldmath $c$}$}_{\zeta,t_0} \in X_\zeta & \mbox{ for } t \in t_0 + {\mathbb N} \\
			\mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\Delta$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^-,t+k} + {H_\zeta}^{(t_0-t)} \mbox{$\mbox{\boldmath $c$}$}_{\zeta,t_0} \in X_\zeta & \mbox{ for } t \in t_0 - {\mathbb N} \end{array} \right.
	\end{equation}
	where $t_0 + {\mathbb N} = \{ t \in {\mathbb Z} \mid t > t_0 \}$ and $t_0 - {\mathbb N} = \{ t \in {\mathbb Z} \mid t < t_0\}$ and where $\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t_0}$ is the initial value.  The flow is always subexponential with ${\mbox{$\mbox{\boldmath $\nabla$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^+,t} = \mbox{$\mbox{\boldmath $0$}$}$ if $t \in t_0 - {\mathbb N} = \{ t \in {\mathbb Z} \mid t < t_0 \}$ and ${\mbox{$\mbox{\boldmath $\Delta$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^-,t+k} = \mbox{$\mbox{\boldmath $0$}$}$ if $t + k \in t_0+{\mathbb N} = \{ t + k \in {\mathbb Z} \mid t+k > t_0\}$.
	
	\item If $\infty \in \sigma$ the projected equation takes the form $\mbox{$\mbox{\boldmath $x$}$}_{\infty,t} = R_{\infty,0} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\infty,t} - G_\infty \mbox{$\mbox{\boldmath $x$}$}_{\infty,t-1}$ where $G_\infty = (R_{\infty,0}A_1)$.  This recursion propagates naturally in the forward direction.  Recovery of a compatible flow for $t \in t_0 - {\mathbb N} = \{ t \in {\mathbb Z} \mid t < t_0\}$ upstream from the observed value $\mbox{$\mbox{\boldmath $x$}$}_{\infty,t_0}$ requires generation of successive preimages under $G_\infty$.  In this context each $\mbox{$\mbox{\boldmath $c$}$}_{\infty, t_0, -\infty} = \lim_{r \rightarrow \infty} (-1)^{r+t_0} G_\infty^{r+t_0} \mbox{$\mbox{\boldmath $x$}$}_{\infty,-r} \in {\mathcal C}_{G_\infty}(X_\infty)$ represents a compatible boundary value in the far distant past and defines a unique forward flow for $t \in t_0 -1 + {\mathbb N} = \{ t \in {\mathbb Z} \mid t \geq t_0 \}$ with
	\begin{equation}
		\label{mr:e4}
		\mbox{$\mbox{\boldmath $x$}$}_{\infty,t} = \mbox{$\sum_{\ell \in {\mathbb N}-1}$} R_{\infty,\ell}\, \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\infty,t-\ell} + (-1)^{t-t_0} G_\infty^{\,t-t_0} \mbox{$\mbox{\boldmath $c$}$}_{\infty, t_0, -\infty} \in X_\infty.
	\end{equation}
	If the propagating operator $G_\infty\vert_{{\mathcal C}_{G_\infty}(X_\infty)}$ is injective then $\mbox{$\mbox{\boldmath $x$}$}_{\infty,t}$ is uniquely determined by $(\ref{mr:e4})$ for all $t \in {\mathbb Z}$ and the flow is subexponential if and only if $\mbox{$\mbox{\boldmath $c$}$}_{\infty, t_0, -\infty} = \mbox{$\mbox{\boldmath $0$}$}$.  If $G_\infty\vert_{{\mathcal C}_{G_\infty}(X_\infty)}$ is not injective $(\ref{mr:e4})$ is no longer a valid solution per se when $t \in t_0 - {\mathbb N} = \{ t \in {\mathbb Z} \mid t < t_0\}$ but rather provides a framework within which a valid solution can be constructed.  The flow is no longer uniquely defined and one must construct a compatible upstream flow $\mbox{$\mbox{\boldmath $x$}$}_{c,\infty,t} = (-1)^{t-t_0} \mbox{$\mbox{\boldmath $u$}$}_{t_0-t}$ for $t \in t_0 - {\mathbb N}$ by setting $\mbox{$\mbox{\boldmath $u$}$}_0 = \mbox{$\mbox{\boldmath $c$}$}_{\infty,t_0,-\infty}$ and choosing $\{ \mbox{$\mbox{\boldmath $u$}$}_p \}_{p \in {\mathbb N}}$ with $\mbox{$\mbox{\boldmath $u$}$}_{p-1} = G_\infty \mbox{$\mbox{\boldmath $u$}$}_p$ for all $p \in {\mathbb N}$.  The propagating operator $G_\infty$ is quasinilpotent and hence is not invertible.
	
	\item If $0 \in \sigma$ the projected equation takes the form $\mbox{$\mbox{\boldmath $x$}$}_{0,t} = R_{0,-1}\mbox{$\mbox{\boldmath $\varepsilon$}$}_{0,t+1} - G_0 \mbox{$\mbox{\boldmath $x$}$}_{0,t+1}$ where $G_0 = (R_{0,-1}A_0)$.  This recursion propagates naturally in the backward direction.  Recovery of a compatible flow for $t \in t_0 + {\mathbb N} = \{ t \in {\mathbb Z} \mid t > t_0\}$ upstream from the observed value $\mbox{$\mbox{\boldmath $x$}$}_{0,t_0}$ requires generation of successive preimages under $G_0$.  In this context each $\mbox{$\mbox{\boldmath $c$}$}_{0,t_0,\infty} = \lim_{r \rightarrow \infty} (-1)^{r-t_0} G_0^{r-t_0} \mbox{$\mbox{\boldmath $x$}$}_{0,r} \in {\mathcal C}_{G_0}(X_0)$ represents a compatible boundary value in the far distant future and defines a unique backward flow for $t \in t_0 +1 - {\mathbb N} = \{ t \in {\mathbb Z} \mid t \leq t_0\}$ with
	\begin{equation}
		\label{mr:e5}
		\mbox{$\mbox{\boldmath $x$}$}_{0,t} = \mbox{$\sum_{k \in {\mathbb N}}$} R_{0,-k}\, \mbox{$\mbox{\boldmath $\varepsilon$}$}_{0,t+k} + (-1)^{\,t_0-t} G_0^{t_0-t} \mbox{$\mbox{\boldmath $c$}$}_{0, t_0, \infty} \in X_0.
	\end{equation}
	If the propagating operator $G_0\vert_{{\mathcal C}_{G_0}(X_0)}$ is injective then $\mbox{$\mbox{\boldmath $x$}$}_{0,t}$ is uniquely determined by $(\ref{mr:e5})$ for all $t \in {\mathbb Z}$ and the flow is subexponential if and only if $\mbox{$\mbox{\boldmath $c$}$}_{0, t_0, \infty} = \mbox{$\mbox{\boldmath $0$}$}$.  If $G_0\vert_{{\mathcal C}_{G_0}(X_0)}$ is not injective $(\ref{mr:e5})$ is no longer a valid solution per se when $t \in t_0 + {\mathbb N} = \{ t \in {\mathbb Z} \mid t > t_0\}$ but rather provides a framework within which a valid solution can be constructed.  The flow is no longer uniquely defined and one must construct a compatible upstream flow $\mbox{$\mbox{\boldmath $x$}$}_{c,0,t} = (-1)^{t-t_0} \mbox{$\mbox{\boldmath $u$}$}_{t-t_0}$ for $t \in t_0 + {\mathbb N}$ by setting $\mbox{$\mbox{\boldmath $u$}$}_0 = \mbox{$\mbox{\boldmath $c$}$}_{0,t_0,\infty}$  and choosing $\{ \mbox{$\mbox{\boldmath $u$}$}_p \}_{p \in {\mathbb N}}$ with $\mbox{$\mbox{\boldmath $u$}$}_{p-1} = G_0 \mbox{$\mbox{\boldmath $u$}$}_p$ for all $p \in {\mathbb N}$.  The propagating operator $G_0$ is quasinilpotent and hence is not invertible.
	\end{enumerate}

<!-- head-4: body_md -->
If $\zeta \in \sigma$ and $\lvert \zeta \rvert \in (1, \infty)$ then Lemma~\ref{sar:lem1} (i) shows that $\mbox{$\mbox{\boldmath $x$}$}_\zeta$ satisfies the forward recursion $\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t} = -H_\zeta^{-1}R_{\zeta,-1}\mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t} + H_\zeta^{-1}\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t-1}$ for all $t \in {\mathbb Z}$.  Suppose $t \in {\mathbb Z}$ is fixed.  Repeated application of the recursion shows that
\begin{equation}
	\label{mr:e6}
	\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t} = (-1) \mbox{$\sum_{\ell=0}^{t+r-1}$} {H_\zeta}^{-\ell-1} R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t-\ell} + {H_\zeta}^{-r-t} \mbox{$\mbox{\boldmath $x$}$}_{\zeta,-r} \quad \mbox{ for all } r \in -t + {\mathbb N}.
\end{equation}
The expression $[z I_X - H_\zeta]^{-1} = [(z - \zeta)I_X + R_{\zeta,-1}A_{\zeta,0}]^{-1}$ is analytic for all $z \neq \zeta$ because $R_{\zeta,-1}A_{\zeta,0}$ is quasinilpotent so the Maclaurin series $[ z I_X - H_\zeta]^{-1} = (-1) \mbox{$\sum_{\ell \in {\mathbb N}-1}$} z^{\ell} {H_\zeta}^{-\ell-1}$ converges for $\lvert z \rvert < \lvert \zeta \rvert$ with $\lim_{\ell \to \infty} \lVert {H_\zeta}^{-\ell} \rVert^{1/\ell} = 1/\lvert \zeta \rvert < 1$.  If $\mbox{$\mbox{\boldmath $\varepsilon$}$}$ is subexponential $\sum_{\ell \in {\mathbb N}-1} {H_\zeta}^{-\ell-1} R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t-\ell}$ converges. (i) For $\zeta \neq 0, \infty$ the operator $R_{\zeta,-1}A_{\zeta,0}$ is quasinilpotent.  The Neumann expansion and the properties of the Laurent series coefficients  show that
\begin{align*}
{H_\zeta}^{-1}R_{\zeta,-1}A_0 & = [\zeta I_X - R_{\zeta,-1}A_{\zeta,0}]^{-1}R_{\zeta,-1}(A_{\zeta,0} - \zeta A_1) \\
& = [ \mbox{$\sum_{k \in {\mathbb N}}$}\, \zeta^{-k}(R_{\zeta,-1}A_{\zeta,0})^{k-1} ]R_{\zeta,-1}[A_{\zeta,0} - \zeta  A_1] \\
& = - R_{\zeta,-1}A_1 + \mbox{$\sum_{k \in {\mathbb N}}$}\, \zeta^{-k} [ (R_{\zeta,-1}A_{\zeta,0})^{k-1} R_{\zeta,-1}A_{\zeta,0} - (R_{\zeta,-1}A_{\zeta,0})^kR_{\zeta,-1}A_1 ] \\
& = - P_\zeta + \mbox{$\sum_{k \in {\mathbb N}}$}\, \zeta^{-k} (-1)^{k-1} [ R_{\zeta,-k}A_{\zeta,0} + R_{\zeta,-k-1} A_1 ] \\
& = -P_\zeta
\end{align*}
where we note from (\ref{mb:e2}) that $R_{\zeta,-k}A_{\zeta,0} + R_{\zeta,-k-1} A_1 = \scalebox{0.9}{\mbox{\large $0$}}$ for all $k \in {\mathbb N}$. \begin{equation}
\label{sar:e1}
- \mbox{$\mbox{\boldmath $x$}$}_{\zeta,t} + {H_\zeta}^{-1}\mbox{$\mbox{\boldmath $x$}$}_{\zeta,t-1} = {H_\zeta}^{-1}R_{\zeta,-1}\mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t} \quad \mbox{ for all } t \in {\mathbb Z}.
\end{equation}
Multiplying through by $H_\zeta$ and replacing $t$ by $t+1$ gives 
\begin{equation}
\label{sar:e2}
- H_\zeta \mbox{$\mbox{\boldmath $x$}$}_{\zeta,t+1} + \mbox{$\mbox{\boldmath $x$}$}_{\zeta,t} = R_{\zeta,-1}\mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t+1} \quad \mbox{ for all } t \in {\mathbb Z}.  \end{equation}
The desired recursions follow by rearranging (\ref{sar:e1}) and (\ref{sar:e2}).  

(ii) For $\zeta = \infty$ multiply (\ref{in:e1}) on the left by $R_{\infty,0}$ to give $R_{\infty,0}A_0 \mbox{$\mbox{\boldmath $x$}$}_t + R_{\infty,0}A_1 \mbox{$\mbox{\boldmath $x$}$}_{t-1} = R_{\infty,0}\mbox{$\mbox{\boldmath $\varepsilon$}$}_t$ for all $t \in {\mathbb Z}$.  Now use $P_\infty = R_{\infty,0}A_0$, $R_{\infty,0}A_1 = R_{\infty,0}A_1[R_{\infty,0}A_0 + R_{\infty,-1}A_1] = R_{\infty,0}A_1R_{\infty,0}A_0$ because $R_{\infty,0}A_1R_{\infty,-1} = R_{\infty,0}Q_\infty^c = \scalebox{0.9}{\mbox{\large $0$}}$, and $R_{\infty,0} = R_{\infty,0}Q_\infty$ to deduce
\begin{equation}
\label{sar:e3}
\mbox{$\mbox{\boldmath $x$}$}_{\infty,t} + G_\infty \mbox{$\mbox{\boldmath $x$}$}_{\infty,t-1} = R_{\infty,0}\mbox{$\mbox{\boldmath $\varepsilon$}$}_{\infty,t} \quad \mbox{ for all } t \in {\mathbb Z}.  \end{equation}
A simple rearrangement of (\ref{sar:e3}) gives the desired recursion. 

(iii) For $\zeta = 0$ multiply (\ref{in:e1}) on the left by $R_{0,-1}$ and replace $t$ by $t+1$ to give $R_{0,-1}A_0 \mbox{$\mbox{\boldmath $x$}$}_{t+1} + R_{0,-1}A_1\mbox{$\mbox{\boldmath $x$}$}_t = R_{0,-1}\mbox{$\mbox{\boldmath $\varepsilon$}$}_{t+1}$ for all $t \in {\mathbb Z}$.  Now use $P_0 = R_{0,-1}A_1$ and $R_{0,-1} = R_{0,-1}Q_0$ to give
\begin{equation}
\label{sar:e4}
G_0 \mbox{$\mbox{\boldmath $x$}$}_{0,t+1} + \mbox{$\mbox{\boldmath $x$}$}_{0,t} = R_{0,-1}\mbox{$\mbox{\boldmath $\varepsilon$}$}_{0,t+1} \quad \mbox{ for all } t \in {\mathbb Z}.
\end{equation}
A simple rearrangement of (\ref{sar:e4}) gives the desired recursion.  $\hfill \Box

\label{sar:lem2}
Suppose $\zeta \in \sigma$ with $\lvert \zeta\rvert \in (0,\infty)$.  The particular cumulation series $\mbox{$\mbox{\boldmath $x$}$}_{P, \zeta,t}$ can be written as a sum of inverse powers of weighted backward or forward differences.  The specific expressions are:
\begin{enumerate}[(i)]
\item if $\lvert \zeta \rvert \in (1,\infty)$ then $\mbox{$\mbox{\boldmath $x$}$}_{P, \zeta,t} = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\nabla$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t}$ for all $t \in {\mathbb Z}$;
\item if $\lvert \zeta \rvert \in (0,1)$ then $\mbox{$\mbox{\boldmath $x$}$}_{P, \zeta,t} = \mbox{$\sum_{k \in {\mathbb N}}$}\, (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\Delta$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+k}$ for all $t \in {\mathbb Z}$;
\item if $\lvert \zeta \rvert = 1$ then $\mbox{$\mbox{\boldmath $x$}$}_{p,\zeta,t} = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\nabla$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^+,t}$ for all $t \in t_0 + {\mathbb N}$;
\item if $\lvert \zeta \rvert = 1$ then $\mbox{$\mbox{\boldmath $x$}$}_{p,\zeta,t} = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\Delta$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^-,t+k}$ for all $t \in t_0 - {\mathbb N}$.
\end{enumerate}

<!-- head-10: body_md -->
We refer readers to Section~\ref{s:mb} for details of the Laurent series expansions for $R(z)$.  Changes in the order of summation are justified because the series involved are absolutely convergent.  For (i) and (ii) with $\lvert \zeta \rvert \in (0,\infty)$ the Laurent series
\begin{equation*}
R_{\zeta, \mbox{\scriptsize\textup{sg}}}(z) = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^{k-1}(R_{\zeta,-1}A_{\zeta,0})^{k-1}R_{\zeta,-1}(z - \zeta)^{-k}
\end{equation*}
converges for all $ z \neq \zeta$.  Therefore $\lim_{k \to \infty} \lVert (R_{\zeta,-1}A_{\zeta,0})^k R_{\zeta,-1} \rVert^{1/k} \to 0$ as $k \to \infty$.  Note that $H_\zeta = \zeta I_X - R_{\zeta,-1}A_{\zeta,0}$.

\vspace{0.2cm}
(i) If $\lvert \zeta \rvert \in (1,\infty)$ then
\begin{align*}
\mbox{$\mbox{\boldmath $x$}$}_{p,\zeta,t} & = (-1) \mbox{$\sum_{\ell \in {\mathbb N}-1}$} ( \zeta I_X - R_{\zeta,-1}A_{\zeta,0})^{-\ell-1} R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t-\ell} \\
& = (-1) \mbox{$\sum_{\ell \in {\mathbb N}-1}$} \left\{ \mbox{$\sum_{k \in {\mathbb N}}$} \mbox{$\binom{\ell+k-1}{k-1}$} ( \zeta^{-1}R_{\zeta,-1}A_{\zeta,0})^{k-1} R_{\zeta,-1} \right\}\zeta^{-\ell-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t-\ell} \\
& = (-1) \mbox{$\sum_{k \in {\mathbb N}}$} (R_{\zeta,-1}A_{\zeta,0})^{k-1} R_{\zeta,-1} \left\{  \mbox{$\sum_{\ell \in {\mathbb N}-1}$} \mbox{$\binom{k-1+\ell}{\ell}$} \zeta^{-k-\ell} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t-\ell} \right\} \\
& = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\nabla$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t} \quad \mbox{ for all } t \in {\mathbb Z}.
\end{align*}

(ii) If $\lvert \zeta \rvert \in (0,1)$ then
\begin{align*}
\mbox{$\mbox{\boldmath $x$}$}_{p,\zeta,t} & = \mbox{$\sum_{\ell \in {\mathbb N}}$}\, ( \zeta I_X - R_{\zeta,-1}A_{\zeta,0})^{\ell-1} R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+\ell} \\
& =  \mbox{$\sum_{\ell \in {\mathbb N}}$}\, \zeta^{\ell-1} \left\{ \mbox{$\sum_{k=1}^{\ell}$} \mbox{$\binom{\ell-1}{k-1}$} (-1)^{k-1}( \zeta^{-1}R_{\zeta,-1}A_{\zeta,0})^{k-1} R_{\zeta,-1} \right\} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+\ell} \\
& = (-1) \mbox{$\sum_{k \in {\mathbb N}}$}\, (R_{\zeta,-1}A_{\zeta,0})^{k-1} R_{\zeta,-1} \left\{  (-1)^k \mbox{$\sum_{r \in {\mathbb N}-1}$} \mbox{$\binom{k-1+r}{r}$} \zeta^r \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+k+r} \right\} \\
& = \mbox{$\sum_{k \in {\mathbb N}}$}\, (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\Delta$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+k}  \quad \mbox{ for all } t \in {\mathbb Z}.
\end{align*}

(iii). If $\lvert \zeta \rvert = 1$ then
\begin{align*}
\mbox{$\mbox{\boldmath $x$}$}_{p,\zeta,t} & = (-1) \mbox{$\sum_{\ell=0}^{t-t_0-1}$} ( \zeta I_X - R_{\zeta,-1}A_{\zeta, 0})^{-\ell-1} R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t-\ell} \\
& = (-1) \mbox{$\sum_{\ell=0}^{t-t_0-1}$}\, \zeta^{-\ell-1} \mbox{$\sum_{k \in {\mathbb N}} \binom{\ell+k-1}{k-1}$} ( \zeta^{-1}R_{\zeta,-1}A_{\zeta,0})^{k-1} R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t-\ell} \\
& = (-1) \mbox{$\sum_{k \in {\mathbb N}}$} (R_{\zeta,-1}A_{\zeta,0})^{k-1} R_{\zeta,-1} \mbox{$\sum_{\ell=0}^{t-t_0-1}\,  \binom{k-1+\ell}{\ell}$} \zeta^{-k-\ell} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t-\ell} \\
& = \mbox{$\sum_{k \in {\mathbb N}}$} (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\nabla$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^+,t} \quad \mbox{ for all } t \in t_0 + {\mathbb N}.
\end{align*}

(iv). If $\lvert \zeta \rvert = 1$ then
\begin{align*}
\mbox{$\mbox{\boldmath $x$}$}_{p,\zeta,t} & = \mbox{$\sum_{\ell=1}^{-(t-t_0)}$} ( \zeta I_X - R_{\zeta,-1}A_{\zeta,0})^{\ell-1} R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+\ell} \\
& = \mbox{$\sum_{\ell=1}^{-(t-t_0)} \zeta^{\ell - 1} \sum_{k=1}^{\ell}\, \binom{\ell-1}{k-1}$}\, (-1)^{k-1} ( \zeta^{-1} R_{\zeta,-1}A_{\zeta,0})^{k-1}  R_{\zeta,-1} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+\ell} \\
& = \mbox{$\sum_{k=1}^{-(t-t_0)}\, (-1)^{k-1} (R_{\zeta,-1}A_{\zeta,0})^{k-1}  R_{\zeta,-1} \sum_{\ell=k}^{-(t-t_0)}  \binom{\ell-1}{k-1}$}\, \zeta^{\ell - k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+\ell} \\
& = (-1) \mbox{$\sum_{k=1}^{-(t-t_0)}\, (R_{\zeta,-1}A_{\zeta,0})^{k-1}  R_{\zeta,-1} (-1)^k \sum_{r=0}^{-(t-t_0)-k}  \binom{k-1+r}{r}$}\, \zeta^r \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta, t+k+r} \\
& = \mbox{$\sum_{k=1}^{-(t-t_0)}$}\, (-1)^k R_{\zeta,-k}\, {\mbox{$\mbox{\boldmath $\Delta$}$}_{\zeta}}^{-k} \mbox{$\mbox{\boldmath $\varepsilon$}$}_{\zeta,t_0^-,t+k}  \quad \mbox{ for all } t \in t_0 - {\mathbb N}.
\end{align*}
In (iii) and (iv) the respective truncated forms of (\ref{mr:e0b}) and (\ref{mr:e0f}) remain valid when $\lvert \zeta \rvert = 1$.  $\hfill \Box
