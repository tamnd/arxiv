\label{wD1}
A weak martingale solution of problem \eqref{eq:MHD} and \eqref{energ_nerovnost} is the quantity 
\[
\mathcal{U} = \Big( (\Omega, \mathcal{A}, (\mathcal{A}_t)_{t \geq 0}, 
\mathbb{P} ) , \varrho, {\bf u}, {\bf B}, W^1, W^2 \Big), 
\] 	
where 
\begin{enumerate}
	\item 
	$(\Omega, \mathcal{A}, (\mathcal{A}_t)_{t \geq 0}, 
	\mathbb{P} )$ is a complete stochastic basis with a right--continuous filtration $(\mathcal{A}_t)_{t \geq 0}$.
	\item $(W^1, W^2)$ are independent  $(\mathcal{A}_t)_{t \geq 0}$ Wiener processes. 
\item 
\begin{align} 
\varrho &\in C_{\rm weak}([0,T]; L^\gamma (Q_F)),  \nonumber \\  
\varrho {\bf u} &\in C_{\rm weak}([0,T]; L^{\frac{2 \gamma}{\gamma + 1}}(Q_F; \mathbb{R}^3)),  \nonumber \\  
{\bf B} &\in C_{\rm weak}([0,T]; L^2(\mathbb{T}^3; \mathbb{R}^3)) 	
\nonumber
\end{align}
are $(\mathcal{A}_t)$ progressively measurable stochastic processes, 
\[
{\bf u} \in L^2(0,T; W^{1,2}_0(Q_F; \mathbb{R}^3)) 
\]
is an $(\mathcal{A}_t)$ adapted random distribution, where the proper definition of random distribution has been given in \cite{BrFeHobook}, Chapter 2.2.
\item The joint law of the initial data
\begin{align}
[\varrho(0, \cdot), (\varrho{\bf u})(0, \cdot), {\bf B}(0,\cdot)]  &\sim [\varrho_0, (\varrho {\bf u})_0, {\bf B}_0] \ \mbox{(equivalence in law)}  \nonumber \\  &\quad \mbox{in}\ L^1 (Q_F) 
\times L^1 (Q_F; \mathbb{R}^3) 
\times L^1(\mathbb{T}^3; \mathbb{R}^3).  
\nonumber  
\nonumber
\end{align}

\item The equation of continuity is satisfied in the weak sense $\mathbb{P}$-a.s.: 
\begin{equation} \label{w2} 
\left[ \int_{Q_F} \varrho \varphi \ dx \right]_{t=0}^{t = \tau} 
= \int_0^\tau \int_{Q_F} \Big[ \varrho \partial_t \varphi + \varrho {\bf u} \cdot 
\nabla_x \varphi \Big] \,{\rm d} {x} \,{\rm d} t  
\end{equation}	
for any $\varphi \in C^1([0,T] \times \overline{Q}_F)$. 	
\item The renormalized equation of continuity is satisfied in the weak sense $\mathbb{P}$-a.s.: 
\begin{equation}\label{w2re}
   \left[ \int_{Q_F}{ b(\varrho) \varphi \,{\rm d} {x}} \right]_{t=0}^{t = \tau}= \int_0^\tau\int_{Q_F}[b(\varrho) \partial_t \varphi + b(\varrho) {\bf u} \cdot \nabla_x \varphi+(b(\varrho)-b'(\varrho)\varrho){\rm div}_x {\bf u}\varphi]{\rm d} x\,  {\rm d} t, 
\end{equation}
for any $\varphi \in C^1([0,T] \times \overline{Q}_F)$ and for any $b(z)$ such that $b \in C^1([0,\infty)), b' \in C_c([0,\infty))$. 
\item
The momentum equation is satisfied in the weak sense $\mathbb{P}$-a.s.: 
\begin{align}
    \left[ \int_{Q_F}{ \varrho {\bf u} \cdot \boldsymbol{\varphi} } \right]_{t = 0}^{t = \tau}= &\int_0^\tau\int_{Q_F}[\varrho {\bf u}\cdot \partial_t \bm{\varphi} + (\varrho {\bf u}\otimes {\bf u}): \nabla_x \bm{\varphi}+p(\varrho){\rm div}_x \bm{\varphi}  \nonumber \\ 
    &  -\nu \nabla_x {\bf u} : \nabla_x \bm{\varphi}-(\mu+\nu) {\rm div}_x {\bf u}\, {\rm div}_x \bm{\varphi}-\varrho \nabla_x G \cdot \bm{\varphi}]{\rm d} x\, {\rm d} t  \nonumber \\ 
    & +\int_0^\tau\int_{Q_F}[{\bf curl}_x {\bf B}\times{\bf B}] \cdot \bm{\varphi}\, {\rm d} x\, {\rm d} t 	
\label{w3}
\end{align}	
for any $\boldsymbol{\varphi} \in C^1_c([0,T] \times Q_F; \mathbb{R}^3)$.
\item 
The induction equation is satisfied in the weak sense $\mathbb{P}$-a.s.:
\begin{align}

\label{MT1}
    

Let $\gamma > \frac{3}{2}$.	
Suppose that the torus $\mathbb{T}^3$ admits a decomposition into $Q_{\rm int}$,
