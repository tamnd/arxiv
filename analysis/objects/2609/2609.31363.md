<!-- definition-1: body_md -->
\label{def:wass-defn} The Wasserstein distance of order 2 between two distributions $\mathbb{P}, \widehat{\mathbb{P}}\in \mathcal{P}$ is given by  $\mathsf{W}(\mathbb{P}, \widehat{\mathbb{P}}) \coloneqq \inf \{ \mathbb{E}_{\gamma} [\|Z - \widehat{Z}\|_2^2] ^{1/2}: \gamma \in \Gamma(\mathbb{P}, \widehat{\mathbb{P}})\}$, where $\Gamma(\mathbb{P}, \widehat{\mathbb{P}})$ denotes the set of joint distributions or couplings of $Z$ and $\widehat Z$ with marginals $\mathbb{P}$ and $\widehat{\mathbb{P}}$, respectively. \looseness=-1
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:cm_functional_map}
Let ${\mathcal{S}}_k$ denote the set of all permutations of $\{1, \dots, k\}$. A set $\mathcal{C}\subseteq \mathbb R^m \times \mathbb R^m$ is called cyclical monotone if for any $k\in\mathbb{N}$, any collection of pairs $\{(\widehat{z}_1, z_1), \dots, (\widehat{z}_k, z_k)\} \subseteq\mathcal{C}$ and any $\sigma\in{\mathcal{S}}_k$, the following equivalent conditions hold:
\vspace{-0.099cm}
{ 
    \abovedisplayskip=1pt plus 1pt minus 1pt
    \belowdisplayskip=1pt plus 1pt minus 1pt
    \begin{equation*}
         \sum_{i=1}^{k} \|\widehat{z}_i - z_i\|_2^2 \leq \sum_{i=1}^{k} \|\widehat{z}_i - z_{\sigma(i)}\|_2^2 \;\; \iff \;\;  \sum_{i=1}^{k} \langle z_{\sigma(i)} - z_i, \widehat{z}_i \rangle \leq 0.
    \end{equation*}
} 
We say that a map $T\in{\mathcal{T}}$ is cyclically monotone on a set $\mathcal{Z}\subseteq \mathbb{R}^m$ if its graph $ \{(\widehat{z}, T(\widehat{z})) : \widehat{z} \in \mathcal{Z}\}$ is a cyclically monotone set. If $\mathcal{Z}=\mathbb{R}^m$, we simply say that $T$ is cyclically monotone.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:monge_gap} Given any $\widehat{\mathbb{P}}\in\mathcal{P}$, the Monge gap of a map $T\in{\mathcal{T}}$ is defined as
{ 
    \abovedisplayskip=0.8pt plus 0.8pt minus 0.8pt
    \belowdisplayskip=0.5pt plus 0.8pt minus 0.8pt
    \begin{equation*}
        \mathcal{M}_{\widehat{\mathbb{P}}}(T)\coloneqq \mathbb{E}_{\widehat{\mathbb{P}}}[\|\widehat Z -T(\widehat Z)\|_2^2] - \mathsf{W}^2(T_{\#}\widehat{\mathbb{P}}, \widehat{\mathbb{P}}).
    \end{equation*}
}
<!-- end definition-3 -->
<!-- proposition-1: body_md -->
\label{prop:convex_bound_inexact}
Assume that $f(\theta, z)$ is convex in $\theta$ with
uniformly bounded gradients $\|\nabla_\theta f(\theta, z)\|_2 \le L_\theta$
for all $\theta \in \Theta$ and $z \in \mathbb{R}^m$. Fix any~$\theta_0\in\Theta$, let
$\theta^\star \in \arg\min_{\theta \in \Theta} F(\theta)$, and consider $H$ projected gradient descent iterates satisfying
\[
    \theta_{t+1}=\operatorname{Proj}_\Theta\!\bigl(\theta_t - \alpha\, g_t\bigr), \quad\text{where}
    \quad
    g_t =\mathbb{E}_{\widehat{\mathbb{P}}} \bigl[\nabla_\theta f(\theta_t, T_t(\widehat Z))\bigr],
\]
where $T_t\in{\mathcal{T}}$ is a near-optimal adversarial map that approximately solves~\eqref{eq:maximize-over-maps} at~$\theta=\theta_t$. If the constant step size is set to $\alpha = \|\theta_0 - \theta^\star\|_2 / L_{\theta}\sqrt{H}$, the average iterate $\bar\theta_H \coloneqq \frac{1}{H}\sum_{t=0}^{H-1}\theta_t$ satisfies
\begin{equation*}
\label{eq:convex_bound_inexact}
F(\bar\theta_H) - F(\theta^\star)
\;\le\;
\frac{L_{\theta}\,\|\theta_0 - \theta^\star\|_2}{\sqrt{H}}
\;+\;
\frac{1}{H}\sum_{t=0}^{H-1}\varepsilon(\theta_t, T_t).
\end{equation*}
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{prop:wasted_transport}
We have \(
   \varepsilon(\theta, T)  \geq \lambda \mathcal{M}_{\widehat\mathbb{P}}(T)
\)
for all~$T \in \mathcal{T}$.
<!-- end proposition-2 -->
<!-- proposition-3: body_md -->
\label{prop:optimal_map_cm} The optimal adversarial map $T^\star$  is cyclically monotone, and thus $\mathcal{M}_{\widehat \mathbb{P}}(T^\star)=0$.
<!-- end proposition-3 -->
<!-- proposition-4: body_md -->
\label{prop:pa_monotonicity}
If $m=1$ and $\nabla_z f(\theta,\cdot)$ is $L_{zz}$-Lipschitz, then 
\begin{itemize}[leftmargin=2em]\vspace{-0.2cm}
    \item the continuous-time PA map $T_{\mathrm{pa}}(t,\cdot)$ is cyclically monotone for all $t\in\mathbb{R}_+$;
    \vspace{-0.2cm}
    \item the discrete-time PA map $T_{\mathrm{pa}}(t,\cdot)$ is cyclically monotone for all $t \in \mathbb{Z}_+$ if $\eta\le 1/(L_{zz}+2\lambda)$.
    \vspace{-0.2cm}
\end{itemize}

<!-- example-1: body_md -->

\(
    f(\theta, z) = \sum_{k=1}^2 a_k
    e^{-(z-\mu_k)^\top\Sigma_k^{-1}(z-\mu_k)}

\label{prop:convex-potential-representation}
A transport map $T \in\mathcal T$ is cyclically monotone if and only if there exists a convex potential function $\psi: \mathbb{R}^m \to \mathbb{R} \cup \{+\infty\}$ such that $T(\widehat{z}) \in \partial \psi(\widehat{z})$ for every $\widehat z\in \mathbb{R}^m$. <!-- proposition-5: body_md -->
\label{prop:measurability_Tstar}
The set-valued mapping $S(\widehat{z})$ in~\eqref{eq:T_star_argmax} is non-empty and closed-valued for all $\widehat{z} \in \mathbb{R}^m$. Furthermore, there exists a Borel-measurable map $T^\star : \mathbb{R}^m \to \mathbb{R}^m$ such that $T^\star(\widehat{z}) \in S(\widehat{z})$ for all $\widehat{z} \in \mathbb{R}^m$.

<!-- proof-1: body_md -->
Define $g(\widehat{z}, z) \coloneqq f(\theta, z) - \lambda \|z - \widehat{z}\|_2^2$ to simplify notation. Since $f(\theta, z)$ is Lipschitz continuous in $z$, the quadratic penalty dominates as $\|z\|_2 \to \infty$, and thus the continuous function~$-g(\widehat{z}, z)$ is coercive in~$z$. Therefore, $g(\widehat{z}, z)$ attains its maximum over $z$, which in turn implies that the set of maximizers $S(\widehat{z})$ is non-empty and closed for every $\widehat{z}\in\mathbb{R}^m$. To establish the measurability of the set-valued mapping~$S(\widehat z)$, note first that $g(\widehat{z}, z)$ is continuous throughout $\mathbb{R}^m \times \mathbb{R}^m$. This implies via \cite[Examples~14.15~\&~14.29]{rockafellar1998variational} that $g(\widehat{z}, z)$ is a normal integrand. By~\cite[Theorem~14.37]{rockafellar1998variational}, the argmax mapping $S(\widehat{z})=\arg\max_{z \in \mathbb{R}^m} g(\widehat{z}, z)$ thus constitutes a Borel-measurable set-valued mapping. We can therefore invoke~\cite[Corollary~14.6]{rockafellar1998variational} to conclude that $S$ admits a measurable selection. That is, there exists a Borel-measurable function $T^\star : \mathbb{R}^m \to \mathbb{R}^m$ with $T^\star(\widehat{z}) \in S(\widehat{z})$ for all $\widehat{z} \in \mathbb{R}^m$.

<!-- proposition-6: body_md -->
\label{prop:inner_deterministic_optimizer}
The adversarial risk satisfies~\eqref{eq:regularized_dro-dual}, the measure $\mathbb{P}^{\star}{\coloneqq} T^{\star}_\#\widehat{\mathbb{P}}$ solves the inner maximization problem in~\eqref{prob:regularized_dro}, and~$\gamma^\star {\coloneq} (T^\star, \mathrm{id})_{\#} \widehat{\mathbb{P}}$ is an optimal coupling of~$\widehat{\mathbb{P}}$ and~$\mathbb{P}^{\star}$.

<!-- proof-2: body_md -->
By rewriting the squared Wasserstein distance appearing in the definition of~$F(\theta)$ as an optimal transport problem over couplings $\gamma$, and by eliminating the candidate measure $\mathbb{P}$ through its representation as the first marginal of~$\gamma$, we obtain
\begin{equation}
\label{eq:V_coupling}
F(\theta)
\;=\;
\sup_{\substack{\gamma\in\mathcal{P}(\mathbb{R}^m\times\mathbb{R}^m)\\ (\pi_2)_\#\gamma=\widehat{\mathbb{P}}}}
\int_{\mathbb{R}^m\times\mathbb{R}^m}\!\!
\Bigl(f(\theta,z)-\lambda\|z-\widehat z\|_2^2\Bigr)\,\mathrm{d}\gamma(z,\widehat z).
\end{equation}
Note that the constraint $\mathbb{P} \in \mathcal{P}(\mathbb{R}^m)$ in the definition of~$F(\theta)$, which requires~$\mathbb P$ to have finite second moments, translates to the requirement $\gamma \in \mathcal{P}(\mathbb{R}^m \times \mathbb{R}^m)$ in~\eqref{eq:V_coupling}, which requires the coupling~$\gamma$ to have finite second moments.  Indeed, whenever $\widehat{\mathbb{P}}$ has finite second moments, $\mathbb{P}$ has finite second moments if and only if every coupling of $\mathbb{P}$ and $\widehat{\mathbb{P}}$ does. This follows from the relation
\[
    \mathbb{E}_{\gamma} \left[\|(Z,\widehat{Z})\|_2^2 \right] = \mathbb{E}_{(\pi_1)_\#\gamma}\left[\|Z\|_2^2 \right] + \mathbb{E}_{\widehat{\mathbb{P}}}\left[\|\widehat{Z}\|_2^2 \right],
\]
where $\pi_1:\mathbb{R}^m\times\mathbb{R}^m\to\mathbb{R}^m$  is the coordinate projection defined by $\pi_1(z,\widehat z)=z$. Next, we show that couplings with infinite second moments are strictly suboptimal in~\eqref{eq:V_coupling} and that $\mathcal{P}(\mathbb{R}^m \times \mathbb{R}^m)$ can be relaxed to $\mathcal{P}_0(\mathbb{R}^m \times \mathbb{R}^m)$. To this end, we first prove
that the optimal value of~\eqref{eq:V_coupling} is finite. 

We can start by bounding
the integrand in~\eqref{eq:V_coupling}. As $f(\theta,z)$ is $L_z$-Lipschitz continuous in~$z$, we have
\begin{align*}
    f(\theta,z) - \lambda\|z - \widehat z\|_2^2 
    &\leq f(\theta,\widehat z) + L_z\|z - \widehat z\|_2  - \lambda\|z - \widehat z\|_2^2 \\
    &= f(\theta,\widehat z) + \frac{L_z^2}{4\lambda} -\lambda\left(\|z - \widehat z\|_2 - \frac{L_z}{2\lambda}\right)^2
\end{align*}
for all $z, \widehat z \in \mathbb{R}^m$. Using the elementary inequality $(a-b)^2 \geq \frac{1}{2}a^2 - b^2$ for any $a,b\in\mathbb R$, we can further loosen this bound to isolate the quadratic transport cost. We thus obtain
\begin{align*}
    f(\theta,z) - \lambda\|z - \widehat z\|_2^2 
    &\leq f(\theta,\widehat z) + \frac{L_z^2}{2\lambda} -\frac{\lambda}{2}\|z - \widehat z\|_2^2 \\
    &\leq f(\theta,0) + L_z\|\widehat z\|_2 + \frac{L_z^2}{2\lambda} -\frac{\lambda}{2}\|z - \widehat z\|_2^2.
\end{align*}
Integrating both sides with respect to~$\gamma$ and recalling that the second marginal of~$\gamma$ equals~$\widehat\mathbb{P}$ yields
\begin{subequations}
\begin{align}
    \int\!\bigl(f(\theta,z) - \lambda\|z - \widehat z\|_2^2\bigr)\,\mathrm{d}\gamma (z,\widehat z) 
    &\leq 
    f(\theta,0) + \frac{L_z^2}{2\lambda} + L_z\mathbb{E}_{\widehat\mathbb{P}}\bigl[\|\widehat Z\|_2\bigr] - \frac{\lambda}{2}\!\int\!\|z - \widehat z\|_2^2\,\mathrm{d}\gamma(z,\widehat z) \label{eq:upper_bound_1} \\
    &\leq f(\theta,0) + \frac{L_z^2}{2\lambda} + L_z\mathbb{E}_{\widehat\mathbb{P}}\bigl[\|\widehat Z\|_2\bigr]. \label{eq:upper_bound_2}
\end{align}
\end{subequations}

As~$\widehat{\mathbb{P}}$ has finite second moments, the inequality~\eqref{eq:upper_bound_2} shows that  the integral is bounded above by a finite constant uniformly across all couplings~$\gamma$. Thus, we find~$F(\theta) < \infty$. Conversely, the trivial coupling $\gamma = (\mathrm{id},\mathrm{id})_\#\widehat\mathbb{P}$ is feasible in~\eqref{eq:V_coupling} and attains the finite objective value $\mathbb{E}_{\widehat{\mathbb{P}}}[f(\theta,\widehat{Z})]$. Consequently, we have~$F(\theta)>-\infty$. In summary, we may conclude that~$F(\theta)$ is indeed finite.

By~\eqref{eq:upper_bound_1}, the objective function in~\eqref{eq:V_coupling} evaluates to~$-\infty$ for any coupling~$\gamma$ with infinite transport cost. Moreover, $\gamma$ has infinite transport cost if and only if it has infinite second moments because
\[
	\int \frac{1}{2}\|z-\widehat{z}\|_2^2 \,\mathrm{d}\gamma (z,\widehat z) \leq  \int\left( \|z\|_2^2 + \|\widehat{z}\|_2^2 \right)\mathrm{d}\gamma (z,\widehat z)  \leq \int\left( 2\|z-\widehat{z}\|_2^2 + 4\|\widehat{z}\|_2^2\right) \mathrm{d}\gamma (z,\widehat z) .
\]
We may therefore relax $\mathcal{P}(\mathbb{R}^m \times \mathbb{R}^m)$ to $\mathcal{P}_0(\mathbb{R}^m \times \mathbb{R}^m)$ without increasing the supremum in~\eqref{eq:V_coupling}.

Next, we can decompose any feasible coupling $\gamma\in \mathcal{P}_0(\mathbb{R}^m \times \mathbb{R}^m)$ with respect to its second marginal. Indeed, by the disintegration theorem on Polish spaces \citep[Theorem~5.3.1] {ambrosio2005gradient}, $\gamma$ decomposes as
\begin{equation}
\label{eq:disintegration}
\gamma(\mathrm{d} z,\mathrm{d}\widehat z)
\;=\;
\nu_{\widehat z}(\mathrm{d} z)\otimes \widehat{\mathbb{P}}(\mathrm{d}\widehat z)
\end{equation}
for a $\widehat{\mathbb{P}}$-a.e.\ uniquely defined Borel family $(\nu_{\widehat z})_{\widehat z\in\mathbb{R}^m}\subseteq\mathcal{P}_0(\mathbb{R}^m)$. Conversely, any such family defines a feasible $\gamma$ via~\eqref{eq:disintegration}. Substituting \eqref{eq:disintegration} into the exact relaxation of~\eqref{eq:V_coupling} derived above and applying the Fubini-Tonelli theorem, which is justified in view of~\eqref{eq:upper_bound_2}, we obtain
\begin{equation*}
F(\theta)
\;=\;
\sup_{(\nu_{\widehat z})}
\int_{\mathbb{R}^m}\!\!\left[\int_{\mathbb{R}^m}\!\bigl(f(\theta,z)-\lambda\|z-\widehat z\|_2^2\bigr)\,\mathrm{d}\nu_{\widehat z}(z)\right]\mathrm{d}\widehat{\mathbb{P}}(\widehat z).
\end{equation*}
For each fixed $\widehat z$, the inner integral admits the upper bound
\[
\int_{\mathbb{R}^m}\bigl(f(\theta,z)-\lambda\|z-\widehat z\|_2^2\bigr)\,\mathrm{d}\nu_{\widehat z}(z) \;\le\; \max_{z\in\mathbb{R}^m}\;\bigl\{f(\theta,z)-\lambda\|z-\widehat z\|_2^2\bigr\},
\]
where equality holds whenever $\nu_{\widehat z}$ is supported on $S(\widehat z)$ defined in~\eqref{eq:T_star_argmax}. Note that the right-hand side of the above inequality is a pointwise maximum of continuous functions and thus Borel-measurable. By \autoref{prop:measurability_Tstar}, there exists a Borel selector $T^\star$ such that $T^\star(\widehat z)$ is optimal on the right-hand side for every~$\widehat z\in\mathbb R^m$. Clearly, the choice $\nu_{\widehat z}^\star\coloneqq\delta_{T^\star(\widehat z)}$ defines a feasible disintegration. Thus, we find
\[
F(\theta)
\;=\;
\mathbb{E}_{\widehat{\mathbb{P}}} \Big[\max_{z \in \mathbb{R}^m} \bigl\{ f(\theta, z) - \lambda \|z - \widehat{Z}\|_2^2 \bigr\}\Big],
\]
which establishes~\eqref{eq:regularized_dro-dual}. This shows that $\gamma^\star \coloneqq (\mathrm{id},T^\star)_\#\widehat{\mathbb{P}}$ solves~\eqref{eq:V_coupling} and that~$\mathbb{P}^\star = (\pi_1)_\#\gamma^\star = T^\star_\#\widehat{\mathbb{P}}$ solves the inner maximization problem in~\eqref{prob:regularized_dro}. We emphasize that $\mathbb{P}^\star \in \mathcal{P}(\mathbb{R}^m)$ has finite second moments for otherwise~$\gamma^\star$ would have infinite second moments and an objective value of~$-\infty$, thus contradicting its optimality. It remains to be shown that $\gamma^\star$ is an optimal coupling between~$\mathbb{P}^\star$ and~$\widehat{\mathbb{P}}$. Suppose for contradiction that there is $\widetilde\gamma\in\Gamma(\mathbb{P}^\star,\widehat{\mathbb{P}})$ with
\[
\int_{\mathbb{R}^m\times\mathbb{R}^m}\|z- \widehat z\|_2^2\,\mathrm{d}\widetilde \gamma(z,\widehat z)
\;<\;
\int_{\mathbb{R}^m\times\mathbb{R}^m}\|z-\widehat z\|_2^2\,\mathrm{d}\gamma^\star(z, \widehat z).
\]
Since $\widetilde\gamma$ and $\gamma^\star$ share the first marginal $\mathbb{P}^\star$, they yield identical loss integrals, that is,
\[
\int f(\theta,z)\,\mathrm{d}\widetilde\gamma(z, \widehat z)
\;=\;
\int f(\theta,z)\,\mathrm{d}\gamma^\star(z, \widehat z)
\;=\;
\mathbb{E}_{\mathbb{P}^\star}[f(\theta,Z)].
\]
Consequently, we find
\[
\int\bigl(f(\theta,z)-\lambda\|z-\widehat z\|_2^2\bigr)\,\mathrm{d}\widetilde\gamma(z,\widehat z)
\;>\;
\int\bigl(f(\theta,z)-\lambda\|z-\widehat z\|_2^2\bigr)\,\mathrm{d}\gamma^\star(z,\widehat z)
\;=\;
F(\theta),
\]
which contradicts~\eqref{eq:V_coupling}. We may thus conclude that
\[
\mathsf{W}^2(\mathbb{P}^\star,\widehat{\mathbb{P}})
\;=\;
\int_{\mathbb{R}^m\times\mathbb{R}^m}\|z-\widehat z\|_2^2\,\mathrm{d}\gamma^\star(z, \widehat z)
\;=\;
\mathbb{E}_{\widehat{\mathbb{P}}}\!\bigl[\|T^\star(\widehat Z)-\widehat Z\|_2^2\bigr],
\]
which confirms that $\gamma^\star$ is indeed an optimal coupling between~$\mathbb{P}^\star$ and~$\widehat{\mathbb{P}}$.

We first show that $g_t$ is an $\varepsilon(\theta_t, T_t)$-subgradient of $F$ at $\theta_t$. As $f(\theta,z)$ is convex in $\theta$, we may use the first-order condition of convexity to obtain
\[
f(\theta,T_t(\widehat z))-\lambda\|T_t(\widehat z)-\widehat z\|_2^2
\;\ge\;
f(\theta_t,T_t(\widehat z))-\lambda\|T_t(\widehat z)-\widehat z\|_2^2
+\langle \nabla_{\theta}f(\theta_t,T_t(\widehat z)),\,\theta-\theta_t\rangle
\]
for any fixed $\theta\in\Theta$ and all $\widehat z \in \mathbb{R}^m$. Taking expectations with respect to $\widehat{\mathbb{P}}$ on both sides implies that
\[
\mathbb{E}_{\widehat{\mathbb{P}}}\!\bigl[f(\theta,T_t(\widehat Z))-\lambda\|T_t(\widehat Z)-\widehat Z\|_2^2\bigr]
\;\ge\;
\mathbb{E}_{\widehat{\mathbb{P}}}\!\bigl[f(\theta_t,T_t(\widehat Z))-\lambda\|T_t(\widehat Z)-\widehat Z\|_2^2\bigr]
+\langle g_t,\,\theta-\theta_t\rangle.
\]
By the definition of the adversarial risk, the left-hand side is no larger than $F(\theta)$, 
and by the definition of the adversarial error, the first term on the right-hand side equals $F(\theta_t)-\varepsilon(\theta_t,T_t)$. Hence, we find
\begin{equation}
\label{eq:eps_subgrad_proof}
F(\theta)\;\ge\;F(\theta_t)+\langle g_t,\theta-\theta_t\rangle-\varepsilon(\theta_t,T_t).
\end{equation}
As $\theta\in\Theta$ was chosen freely, this shows that $g_t$ is indeed an $\varepsilon(\theta_t, T_t)$-subgradient of $F$ at $\theta_t$.

Next, we use the nonexpansiveness of the Euclidean projection operator `$\operatorname{Proj}_\Theta(\cdot)$' and the update rule $\theta_{t+1}=\operatorname{Proj}_\Theta (\theta_t - \alpha\, g_t)$ of the projected gradient descent algorithm to conclude that
\begin{equation}
\label{eq:improvement}
\|\theta_{t+1}-\theta^\star\|_2^2
\;\le\;\|\theta_{t}-\alpha\, g_t-\theta^\star\|_2^2
\;\le\;
\|\theta_t-\theta^\star\|_2^2
-2\alpha\langle g_t,\theta_t-\theta^\star\rangle
+\alpha^2\|g_t\|_2^2.
\end{equation}
As $g_t$ is defined at the expected value of a random variable whose norm is uniformly bounded by $L_\theta$, it is clear that
$\|g_t\|_2\le L_\theta$. Combining~\eqref{eq:eps_subgrad_proof} 
for $\theta =\theta^\star$ with~\eqref{eq:improvement} and rearranging terms we then find
\[
F(\theta_t)-F(\theta^\star)
\;\le\;
\frac{1}{2\alpha} \left( \|\theta_t-\theta^\star\|_2^2-\|\theta_{t+1}-\theta^\star\|_2^2\right)
+\frac{\alpha L_\theta^2}{2}
+ \varepsilon(\theta_t,T_t).
\]
Summing over $t=0,\dots,H-1$ and discarding the last term $-\|\theta_H-\theta^\star\|_2^2/(2\alpha)\leq 0$ in the telescoping sum then yields the estimate
\[
\frac{1}{H}\sum_{t=0}^{H-1}\bigl(F(\theta_t)-F(\theta^\star)\bigr)
\;\le\;
\frac{\|\theta_0-\theta^\star\|_2^2}{2\alpha H}
+\frac{\alpha L_\theta^2}{2}
+\frac{1}{H}\sum_{t=0}^{H-1}\varepsilon(\theta_t,T_t).
\]
The step size $\alpha=\|\theta_0-\theta^\star\|_2/(L_\theta\sqrt{H})$ 
balances the first two terms, ensuring that their sum reduces to 
$L_\theta\|\theta_0-\theta^\star\|_2/\sqrt{H}$. Since $F(\theta)$ is convex
(as the pointwise supremum of convex functions), Jensen's inequality implies that $F(\bar\theta_H)\le \frac{1}{H}\sum_{t=0}^{H-1}F(\theta_t)$, 
which yields the claimed bound.

<!-- proof-6: body_md -->
Since $m = 1$, the map $T_{\mathrm{pa}}(t,\cdot)$ is cyclical monotone if and only if it is monotonically non-decreasing~\citep[Chapter~2]{santambrogio2015optimal}. We therefore prove that the PA map $T_{\mathrm{pa}}(t,\cdot)$ is monotonically non-decreasing both in the continuous-time and discrete-time settings. 

    \textbf{Continuous-time PA.} By the Picard-Lindel\"of theorem, the defining ODE of the continuous-time PA map has a unique solution that depends continuously on the initial conditions. This implies that $T_{\mathrm{pa}}(t,\widehat z)$ is continuously differentiable in~$t$ and continuous in~$\widehat z$. In the remainder, we fix $t>0$.
    
    Suppose now for the sake of contradiction that $T_{\mathrm{pa}}(t,\cdot)$ fails to be monotonically non-decreasing. Thus, there exist $\widehat z_1 < \widehat z_2$ such that $T_{\mathrm{pa}}(t,\widehat z_1)>T_{\mathrm{pa}}(t,\widehat z_2)$. For ease of notation, we define the gap function $\delta(t') \coloneqq T_{\mathrm{pa}}(t', \widehat z_2) - T_{\mathrm{pa}}(t', \widehat z_1)$, which is continuously differentiable. By assumption, we have $\delta(0) = \widehat z_2 - \widehat z_1 > 0$ and $\delta(t)<0$. The intermediate value theorem then implies that there exists $t^\star < t$ with $\delta(t^\star) = 0$ and $\delta'(t^\star) \leq 0$. Next, an elementary calculation reveals that
    \begin{align*}
        \delta'(t^\star)
        &= \tfrac{\mathrm{d}}{\mathrm{d}t} T_{\mathrm{pa}}(t^\star, \widehat z_2) - \tfrac{\mathrm{d}}{\mathrm{d}t} T_{\mathrm{pa}}(t^\star, \widehat z_1) \\
        &= \bigl[\nabla_z f(\theta, T_{\mathrm{pa}}(t^\star, \widehat z_2)) - 2\lambda\bigl(T_{\mathrm{pa}}(t^\star, \widehat z_2) - \widehat z_2\bigr)\bigr] \\
        &\quad - \bigl[\nabla_z f(\theta, T_{\mathrm{pa}}(t^\star, \widehat z_1)) - 2\lambda\bigl(T_{\mathrm{pa}}(t^\star, \widehat z_1) - \widehat z_1\bigr)\bigr].
    \end{align*}
    Since $T_{\mathrm{pa}}(t^\star, \widehat z_1) = T_{\mathrm{pa}}(t^\star, \widehat z_2)$, the terms involving $\nabla_z f$ and $2\lambda\, T_{\mathrm{pa}}(t^\star, \cdot)$ cancel. Thus, we obtain
    \[
        \delta'(t^\star) = 2\lambda(\widehat z_2 - \widehat z_1)>0,
    \]
    which contradicts the defining property $\delta'(t^\star) \leq 0$ of the point~$t^\star$. Consequently, we may conclude that the continuous-time PA map $T_{\mathrm{pa}}(t,\cdot)$ must be monotonically non-decreasing.

    \textbf{Discrete-time PA.} Fix $\widehat z_1 < \widehat z_2$ and define the gap $\delta_t \coloneqq T_{\mathrm{pa}}(t, \widehat z_2) - T_{\mathrm{pa}}(t, \widehat z_1)$. We use induction on~$t$ to show that~$\delta_t\geq 0$ for every~$t\in\mathbb N$. As~$\widehat z_1$ and~$\widehat z_2$ with~$\widehat z_1 < \widehat z_2$ were chosen arbitrarily, this readily implies that $T_{\mathrm{pa}}(t,\widehat z)$ is monotonically non-decreasing in~$\widehat z$. The base case corresponding to~$t = 0$ holds trivially because $T_{\mathrm{pa}}(0, \cdot)$ is the identity map, which implies that $\delta_0 = \widehat z_2 - \widehat z_1 > 0$. As for the induction step, assume now that $\delta_t \geq 0$, and note that the gap obeys the recursion
    \begin{align*}
        \delta_{t+1}
        &= \delta_t + \eta\bigl[\nabla_z f(\theta, T_{\mathrm{pa}}(t, \widehat z_2)) - \nabla_z f(\theta, T_{\mathrm{pa}}(t, \widehat z_1))\bigr] - 2\lambda \eta\, \delta_t + 2\lambda \eta (\widehat z_2 - \widehat z_1) \\
        &= (1 - 2\lambda \eta)\, \delta_t + \eta\bigl[\nabla_z f(\theta, T_{\mathrm{pa}}(t, \widehat z_2)) - \nabla_z f(\theta, T_{\mathrm{pa}}(t, \widehat z_1))\bigr] + 2\lambda \eta (\widehat z_2 - \widehat z_1).
    \end{align*}
    We have $\nabla_z f(\theta, T_{\mathrm{pa}}(t, \widehat z_2)) - \nabla_z f(\theta, T_{\mathrm{pa}}(t, \widehat z_1)) \geq -L_{zz}(T_{\mathrm{pa}}(t, \widehat z_2) - T_{\mathrm{pa}}(t, \widehat z_1)) = -L_{zz}\, \delta_t$ because $\nabla_z f(\theta, z)$ is $L_{zz}$-Lipschitz continuous in~$z$. Substituting this bound into the recursion yields
    \[
        \delta_{t+1} \;\geq\; \bigl(1 - \eta(L_{zz} + 2\lambda)\bigr)\, \delta_t + 2\lambda \eta (\widehat z_2 - \widehat z_1).
    \]
    The step size condition $\eta \leq (L_{zz} + 2\lambda)^{-1}$ and the induction hypothesis ensure that the first term is nonnegative, while the second term is positive by construction. We may thus conclude that $\delta_{t+1} \geq 0$, which completes the induction step. Thus, the claim follows.

<!-- lemma-2: body_md -->
\label{lem:compact_analytic_subset}
    If~\autoref{ass:analytic_loss_function} holds, then $\mathcal{M}_i$ is a compact analytic subset of $\mathbb{R}^m$ for each $i \in [B]$.
<!-- end lemma-2 -->
<!-- proof-8: body_md -->
We first show that $\mathcal{M}_i$ is compact. By definition, any $z \in \mathcal{M}_i$ satisfies 
    \begin{equation*}
        0=\nabla_z f_i(z) = \nabla_z f(\theta, z) - 2\lambda(z - \widehat{z}_i),
    \end{equation*}
    which implies $2\lambda \|z - \widehat{z}_i\|_2 = \|\nabla_z f(\theta, z)\|_2$. Lipschitz continuity of $f(\theta,\cdot)$ ensures that $\|\nabla_zf(\theta, \cdot)\|_2\leq L_z$. As~$\lambda>0$, we thus find $\|z - \widehat{z}_i\|_2 \le \frac{L_z}{2\lambda}$. Consequently, $\mathcal{M}_i$ is bounded. Because $\nabla_z f_i(z)$ is continuous, its zero-set $\mathcal{M}_i$ is also closed, making $\mathcal{M}_i$ a compact subset of $\mathbb{R}^m$.

    Next, note that the penalized objective~$f_i(z)$ inherits real analyticity from~$f(\theta, \cdot)$. Therefore, the squared norm of its gradient, $\|\nabla_z f_i(z)\|_2^2$, is also a real analytic function on $\mathbb{R}^m$. The set~$\mathcal{M}_i$ of stationary points can be uniquely identified as the zero locus of this single analytic function, that is,
    \begin{equation*}
        \mathcal{M}_i = \left\{z \in \mathbb{R}^m : \|\nabla_z f_i(z)\|_2^2 = 0 \right\}.
    \end{equation*}
    By definition, this makes $\mathcal{M}_i$ an analytic subset of $\mathbb{R}^m$~\cite[Chapter~6]{krantz2002primer}.

<!-- corollary-1: body_md -->
\label{cor:finite_critical_values}
    If~\autoref{ass:analytic_loss_function} holds, then~$\mathcal{M}_i$ has finitely many connected components, and the set of critical values $f_i(\mathcal{M}_i)$ is finite for each~$i \in [B]$.
<!-- end corollary-1 -->
<!-- proof-9: body_md -->
By \autoref{lem:compact_analytic_subset}, $\mathcal{M}_i$ is a compact analytic subset of $\mathbb{R}^m$, which inherently belongs to the broader class of semianalytic sets. A fundamental topological property of semianalytic sets is that they are locally connected and that the family of their connected components is locally finite \cite[Theorem~6.5.12]{krantz2002primer}. Local finiteness means that for every point~$z \in \mathcal{M}_i$ there exists an open neighborhood $U_z$ that intersects only a finite number of connected components. The collection of all such neighborhoods, $\{U_z\}_{z \in \mathcal{M}_i}$, naturally forms an open cover of $\mathcal{M}_i$. Because $\mathcal{M}_i$ is compact, this open cover admits a finite subcover, say $\{U_{z_1}, \dots, U_{z_K}\}$. Since the entirety of $\mathcal{M}_i$ lies within the union of these $K$ finite neighborhoods, and each neighborhood intersects only finitely many components, the total number of connected components in $\mathcal{M}_i$ is finite.

    Let $C \subseteq \mathcal{M}_i$ be one such connected component. By \cite[Theorem~6.5.12]{krantz2002primer}, the connected component $C$ is itself a semianalytic set. A standard result in semianalytic geometry is that any two points of a connected semianalytic set can be joined by a semianalytic arc, which is real analytic except at finitely many points~\citep{gabrielov1968projections}. Therefore, for any two points $z, z' \in C$, there exists a piecewise smooth path $\gamma: [0,1] \to C$ connecting them. Because every point on $\gamma$ belongs to $\mathcal{M}_i$, we have $\nabla_z f_i(\gamma(t)) = 0$ for all $t \in [0,1]$. By the fundamental theorem of calculus along curves, we thus have 
    \begin{equation*}
        f_i(z') - f_i(z) = \int_0^1 \langle \nabla_z f_i(\gamma(t)), \tfrac{\mathrm{d}}{\mathrm{d} t}\gamma(t) \rangle \, \mathrm{d} t = 0.
    \end{equation*}
    Therefore, $f_i$ is constant on $C$. Because $\mathcal{M}_i$ consists of a finite number of connected components, it follows that the set of critical values $f_i(\mathcal{M}_i)$ is finite.

<!-- definition-5: body_md -->
\label{def:ascent-operator}
    For a fixed step size $0 < \eta < 2/(L_{zz}+ 2\lambda)$, we define the
    \begin{itemize}[leftmargin=6mm]
        \item[(i)] \textbf{$i$-th one-step ascent operator:} $\mathcal{A}_i:\mathbb{R}^m\to\mathbb{R}^m$,  $z\mapsto z + \eta \nabla f_i(z)$;
        \item[(ii)] \textbf{$i$-th $t$-step ascent operator:} $\mathcal{A}^t_i \coloneqq \mathcal{A}_i \circ \mathcal{A}_i^{t-1}$ for all $t\in \mathbb{N}$, where $\mathcal{A}^0_i \coloneqq \text{id}$;
        \item[(iii)] \textbf{$i$-th asymptotic ascent operator:} $\mathcal{A}_i^\infty:\mathbb{R}^m\to\mathbb{R}^m$,  $z\mapsto \lim_{t \to \infty} \mathcal{A}_i^t(z)$;
        \item[(iv)] \textbf{$t$-step ascent operator:} $\mathcal{A}^t:\mathbb{R}^{Bm}\to\mathbb{R}^{Bm}$,  ${\bf z}\mapsto (\mathcal{A}_1^t(z_1), \dots, \mathcal{A}_B^t(z_B))$ for all $t \in \mathbb{N} \cup \{\infty\}$.
    \end{itemize}
<!-- end definition-5 -->
<!-- lemma-3: body_md -->
\label{lem:ascent_convergence}
    If~\hyperref[ass:loss_function_smoothness]{Assumptions~\ref*{ass:loss_function_smoothness}} and~\ref{ass:analytic_loss_function} hold and $0<\eta<2/(L_{zz}+2\lambda)$,
    then the sequence $z^t=\mathcal{A}_i^t(z^0)$ converges to a stationary point
    $z^\star\in\mathcal{M}_i$ for every~$i\in[N]$ and initialization $z^0\in\mathbb{R}^m$.
<!-- end lemma-3 -->
<!-- proof-10: body_md -->
Fix $i\in[N]$, and set $L \coloneqq L_{zz}+2\lambda$. Since $\nabla f_i$ is $L$-Lipschitz by \autoref{ass:loss_function_smoothness}, the ascent lemma \cite[Proposition~6.1.2]{bertsekas2015convex} and the update rule $z^{t+1}=\mathcal{A}_i(z^t) = z^t+\eta\nabla f_i(z^t)$ yield
    \begin{equation}
        \label{eq:strong-descent-contidion}
        \begin{aligned}
        f_i(z^{t+1})-f_i(z^t)
        &\geq
        \left(\frac{1}{\eta}-\frac{L}{2}\right)
        \|z^{t+1}-z^t\|_2^2 \\
        &= \left(1-\frac{\eta L}{2}\right)
        \|\nabla f_i(z^t)\|_2
        \|z^{t+1}-z^t\|_2. 
    \end{aligned}
    \end{equation}
    As $\eta < 2/L$, a standard telescoping sum argument implies that $f_i(z^t)\ge f_i(z^0)$ for all~$t \ge 0$. Moreover, the $L_z$-Lipschitz continuity of $f(\theta,\cdot)$ implies that $f_i(z)$ is coercive, \textit{i.e.}, $f_i(z)\to-\infty$ as $\|z\|_2\to\infty$. Thus, the superlevel set $\{z\in\mathbb{R}^m \mid f_i(z)\ge f_i(z^0)\}$ is compact and contains the entire sequence $\{z^t\}_{t\in\mathbb{N}}$. To establish convergence, we note that the sequence satisfies the strong ascent conditions of \cite[Definition~3.1]{absil2005convergence}. This is an immediate consequence of~\eqref{eq:strong-descent-contidion}. In particular, the inequality in the first line of~\eqref{eq:strong-descent-contidion} ensures that $f_i(z^{t+1})=f_i(z^t)$  implies $z^{t+1}=z^t$. Because the sequence $\{z^t\}_{t\in\mathbb{N}}$ is bounded and $f_i$ is real analytic by \autoref{ass:analytic_loss_function}, \cite[Theorem~3.2]{absil2005convergence} guarantees that the sequence converges and thus has a limit~$z^\star$. 
    Finally, the update rule ensures that
    \begin{align*}
        \nabla f_i(z^t) = \frac{z^{t+1}-z^t}{\eta} \longrightarrow 0 \quad \text{as} \quad t \to \infty.
    \end{align*}
    By the continuity of $\nabla f_i$, we thus have $\nabla f_i(z^\star)=0$. This implies that $z^\star\in\mathcal{M}_i$.
<!-- end proof-10 -->
<!-- definition-6: body_md -->
\label{def:reassignment_operator}
    The reassignment operator $\mathcal{R} : \mathbb{R}^{Bm} \to \mathbb{R}^{Bm}$ is defined through $\mathcal{R}(\mathbf{z})=\mathbf{z}'$, where $z'_i=z_{\sigma(i)}$ and
    $\sigma(i) \in  \operatorname*{arg\,max}_{j \in [B]} f_i(z_j)$ for all $i\in[B]$. Ties among the maximizers are resolved using a self-favoring rule, \textit{i.e.} if $i \in \operatorname*{arg\,max}_{j} f_i(z_j)$, we set $\sigma(i) = i$. Otherwise, ties are broken by choosing the maximizer with the smallest index.
<!-- end definition-6 -->
<!-- lemma-4: body_md -->
\label{lem:mpa_operators}
    If~\hyperref[ass:loss_function_smoothness]{Assumptions~\ref*{ass:loss_function_smoothness}} and~\ref{ass:analytic_loss_function} hold and $0<\eta<2/(L_{zz}+2\lambda)$,  the ascent and reassignment operators have the following properties.
    \begin{itemize}[leftmargin=6mm]
        \item[(i)]\textbf{Stationarity:} $\mathcal{A}^\infty(\mathbf{z})$ is stationary, and $\mathcal{R}(\mathbf{z})$ is assignment-stationary for all $\mathbf{z} \in \mathbb{R}^{Bm}$.
        \item[(ii)] \textbf{Idempotence:} $\mathcal{A}^\infty \circ \mathcal{A}^\infty = \mathcal{A}^\infty$ and $\mathcal{R} \circ \mathcal{R} = \mathcal{R}$. 
        \item[(iii)] \textbf{Strict monotonicity:} $\mathbf{f}(\mathcal{A}^t(\mathbf{z})) \geq \mathbf{f}(\mathbf{z})$ and $\mathbf{f}(\mathcal{R}(\mathbf{z})) \geq \mathbf{f}(\mathbf{z})$ for all $\mathbf{z}\in\mathbb{R}^{Bm}$ and $t \in \mathbb{N} \cup \{\infty\}$, and these inequalities are strict in at least one coordinate if $\mathcal{A}^t(\mathbf{z}) \neq \mathbf{z}$ or $\mathcal{R}(\mathbf{z}) \neq \mathbf{z}$, respectively.
    \end{itemize}
<!-- end lemma-4 -->
<!-- proof-11: body_md -->
\textbf{Stationarity:} 
    Fix an arbitrary $\mathbf{z}\in\mathbb{R}^{Bm}$. \autoref{lem:ascent_convergence} readily implies that $\mathcal{A}^\infty(\mathbf{z})\in\mathcal{M}$ is stationary. Similarly, the definition of the reassignment operator readily implies that $\mathbf{z}' \coloneqq \mathcal{R}(\mathbf{z})$ is such that $z'_i$ maximizes $f_i$ over the candidate set $\{z_1, \dots, z_B\}$. As $\{z'_1, \dots, z'_B\} \subseteq \{z_1, \dots, z_B\}$, it immediately follows that $f_i(z'_i) \ge f_i(z'_j)$ for all $i,j \in [B]$, meaning that $\mathbf{z}'$ is assignment-stationary.

    \textbf{Idempotence:} 
    Set $z^\star_i \coloneqq \mathcal{A}_i^\infty(z_i)$. As $\nabla f_i(z^\star_i) = 0$ by virtue of \autoref{lem:ascent_convergence}, the point $z^\star_i$ is a fixed point of the one-step operator $\mathcal{A}_i$, meaning that $\mathcal{A}_i^\infty(z^\star_i) = z^\star_i$. As this is true for every $i\in[B]$, we have $\mathcal{A}^\infty \circ \mathcal{A}^\infty = \mathcal{A}^\infty$. 
    as for the reassignment operator, set $\mathbf{z}' \coloneqq \mathcal{R}(\mathbf{z})$. As $\mathbf{z}'$ is assignment-stationary, we have $i \in \operatorname*{arg\,max}_{j\in[B]} f_i(z'_j)$ for all $i\in[B]$. The self-favoring tie-breaking rule in \autoref{def:reassignment_operator} thus dictates that $\mathcal{R}(\mathbf{z}')_i = z'_i$ for every $i\in[B]$. Consequently, we have $\mathcal{R} \circ \mathcal{R} = \mathcal{R}$.

    \textbf{Strict monotonicity:} 
    Equation~\eqref{eq:strong-descent-contidion} in the proof of \autoref{lem:ascent_convergence} and the assumptions on~$\eta$ imply that $f_i(\mathcal{A}_i^t(z_i)) \geq f_i(z_i)$ and that this inequality is strict if $\mathcal{A}_i^t(z_i) \neq z_i$, for every~$i\in[B]$. Hence, the $t$-step ascent operator~$\mathcal{A}^t$ is strictly monotone for every~$t\in\mathbb{N}\cup\{\infty\}$. The reassignment operator~$\mathcal{R}$ is strictly monotone by construction. In particular, if $\mathbf{z}' \coloneqq \mathcal{R}(\mathbf{z})$ and there exists $i\in[B]$ with $z'_i\neq z_i$, then the self-favoring tie-breaking rule ensures that $f_i(z'_i) > f_i(z_i)$.
<!-- end proof-11 -->
<!-- theorem-4: body_md -->
\label{thm:mpa_finite}
    If~\hyperref[ass:loss_function_smoothness]{Assumptions~\ref*{ass:loss_function_smoothness}} and~\ref{ass:analytic_loss_function} hold, $0<\eta<2/(L_{zz}+2\lambda)$ and~$K=\infty$, then the full-batch MPA algorithm terminates after $R\leq 1 + \sum_{i=1}^N (|f_i(\mathcal{M}_i)| - 1)$ rounds in each epoch. The returned output $\mathbf{z}^{(R)}$ is both stationary and assignment-stationary.
<!-- end theorem-4 -->
<!-- proof-12: body_md -->
The first iterate $\mathbf{z}^{(1)} = \mathcal{A}^\infty(\mathcal{R}(\mathbf{z}^{(0)}))$ is stationary by \autoref{lem:mpa_operators}\,{\em (i)}, and thus \autoref{cor:finite_critical_values} implies that~$f_i(z^{(1)}_i)$ belongs to the finite set~$f_i(\mathcal{M}_i)$ for every~$i\in[B]$. For any subsequent non-terminal round $r \geq 2$, we have $\mathbf{z}^{(r)} \neq \mathbf{z}^{(r-1)}$. The definition of the exact MPA algorithm then~yields
    \begin{equation*}
        \mathbf{f}(\mathbf{z}^{(r)}) =  \mathbf{f}(\mathcal{A}^\infty(\mathcal{R}(\mathbf{z}^{(r-1)}))) \ge \mathbf{f}(\mathbf{z}^{(r-1)}),
    \end{equation*}
    where the inequality is strict for at least one coordinate~$i\in[B]$ by virtue of \autoref{lem:mpa_operators}\,{\em (iii)}. Because both~$\mathbf{z}^{(r)}$ and~$\mathbf{z}^{(r-1)}$ belong to~$\mathcal{M}$, this represents a strict jump between two values inside $f_i(\mathcal{M}_i)$. 
    For each coordinate, $f_i$ can jump to a strictly higher critical value at most $|f_i(\mathcal{M}_i)| - 1$ times. Summing this bound over all~$N$ coordinates, and adding~$1$ to account for the first round, yields the desired upper bound on the total number of rounds~$R$ until termination: $1 + \sum_{i=1}^N (|f_i(\mathcal{M}_i)| - 1)$. 
    
    The terminal iterate~$\mathbf{z}^{(R)}$ is generated by the exact ascent operator~$\mathcal{A}^\infty$, which guarantees stationarity by \autoref{lem:mpa_operators}\,{\em (i)}. The termination condition thus simplifies to $\mathcal{R}(\mathbf{z}^{(R)}) = \mathbf{z}^{(R)}$, which ensures via \autoref{lem:mpa_operators}\,{\em (i)} that $\mathbf{z}^{(R)}$ is also assignment-stationary. This observation completes the proof.

<!-- remark-3: body_md -->
\label{rem:mpa_caveats}
    In practice, an epoch of Algorithm~\ref{alg:implicit_mpa} may be interrupted before a fixed point of~$\mathcal{A}^K\circ\mathcal{R}$ is reached. In this case, the last iterate may fail to be assignment-stationary. Indeed, while~$\mathcal{R}$ induces assignment-stationarity, $\mathcal{A}^K$ can fail to preserve assignment-stationarity even if~$K=\infty$. In practice, it is therefore expedient to conclude the MPA algorithm with an extra reassignment step to enforce assignment-stationarity and cyclical monotonicity. We also emphasize that assignment-stationarity is enforced only locally on the given mini-batch.

<!-- lemma-5: body_md -->
\label{lem:optimal_adversary_bounded_second_moment}
    If~\autoref{ass:data_distribution} holds and $M^2 \coloneqq 2\rho^2 + L_z^2/(2\lambda^2)$, any optimal adversarial map $T^\star$ in the sense of \autoref{prop:measurability_Tstar} satisfies
    \begin{equation*}
                                                        
        \mathbb{E}_{\mathbb{P}_0}\bigl[\|T^\star(\widehat{Z}_i)\|_2^2\bigr] \leq M^2\quad \forall i\in[B].
    \end{equation*}
<!-- end lemma-5 -->
<!-- proof-13: body_md -->
By construction, the random vector $Z^\star\coloneq T^{\star}(\widehat{Z}_i)$ is a global maximizer of the random function~$f_i$ pointwise for every uncertainty realization. The stationarity condition $\nabla_z f(\theta,Z^{\star}) =2\lambda(Z^{\star} - \widehat{Z}_i)$ thus implies via the proof of~\autoref{lem:compact_analytic_subset} that $\|Z^{\star} - \widehat{Z}_i\|_2 \leq \frac{L_z}{2\lambda}$. Hence, we obtain
    \begin{equation*}
        \|Z^{\star}\|_2^2 
        \leq \left(\|\widehat{Z}_i\|_2 + \|Z^{\star} - \widehat{Z}_i\|_2\right)^2 
        \leq \left(\|\widehat{Z}_i\|_2 + \frac{L_z}{2\lambda}\right)^2 
        \leq 2\|\widehat{Z}_i\|_2^2 + \frac{L_z^2}{2\lambda^2}.
    \end{equation*}
    The claim then follows by replacing~$Z^\star$ with its definition~$T^\star(\widehat Z_i)$, taking expectations with respect to~$\mathbb{P}_0$ on both sides of the resulting inequality, and invoking the definition of~$M^2$.

<!-- lemma-6: body_md -->
\label{lem:outlier_fraction}
    If~\autoref{ass:data_distribution} holds, then we have
    \begin{equation*}
        \mathbb{P}_0^B\left(\frac{|O_\epsilon|}{B} > \frac{\lambda\epsilon}{L_z^2}\right) \leq \exp\left(-B\lambda\epsilon/6L_z^2\right).
    \end{equation*}
<!-- end lemma-6 -->
<!-- proof-14: body_md -->
The indicator variables $Y_i \coloneqq \mathbf{1}_{\{\|T^\star(\widehat{Z}_i)\|_2 > R_\epsilon\}}$, $i \in [B]$, form a sequence of i.i.d.\ Bernoulli trials. We denote their common success probability by~$p$ and assume temporarily that $p>0$. We also set $\bar{p}\coloneq \lambda\epsilon /(2L_z^2)$. Markov's inequality together with \autoref{lem:optimal_adversary_bounded_second_moment} then implies that 
    \begin{equation*}
        p = \mathbb{E}_{\mathbb{P}_0}[Y_i] = \mathbb{P}_0\left(\|T^\star(\widehat{Z}_i)\|^2_2 > R_\epsilon^2\right) \leq \frac{M^2}{R_\epsilon^2} = \bar p,
    \end{equation*}
    where the last equality exploits the definition of~$R_\epsilon$ in~\eqref{def:bulk_radius}. Re-expressing the number of outliers~$|O_\epsilon|$ as the sum of all Bernoulli trials and setting $\delta \coloneqq 2\bar{p}/p-1\geq 1$ therefore yields 
    \begin{align*}
    \mathbb{P}_0^B\left(\frac{|O_\epsilon|}{B} > \frac{\lambda\epsilon}{L_z^2}\right)
    &= \mathbb{P}_0^B\left(\sum_{i=1}^{B}Y_i > 2B\bar{p}\right)
    = \mathbb{P}_0^B\left(\sum_{i=1}^{B}Y_i > (1+\delta)Bp\right) \\
    &\leq \left(\frac{e^\delta}{(1+\delta)^{1+\delta}}\right)^{Bp} \leq e^{-Bp\delta/3} \leq e^{-B\bar{p}/3}.
    \end{align*}
    Here, the first two equalities follow from the definitions of~$\bar{p}$ and~$\delta$, respectively. The first inequality exploits Chernoff’s inequality for sums of independent Bernoulli random variables \cite[Theorem~2.3.1]{vershynin2018high}, and the second inequality holds because $(1+\delta)\ln(1+\delta)\geq 4\delta/3$ for all~$\delta \geq 1$. The last inequality holds because $p\delta = 2\bar{p} - p \ge \bar{p}$. Replacing $\bar{p}$ with its definition then yields the claim for $p>0$. If $p=0$, on the other hand, $\mathbb{P}_0^B(\sum_{i=1}^B Y_i>0)=0$, and the claim trivially holds.

<!-- lemma-7: body_md -->
\label{lem:positive_covering_mass}
    If~\autoref{ass:data_distribution} holds, then we have~$\underline{p}(\epsilon) > 0$.
<!-- end lemma-7 -->
<!-- proof-15: body_md -->
Introduce an auxiliary function~$\kappa:\mathbb{R}^m\to [0,1]$ defined via $\kappa(z)=\mathbb{P}_0(B_{r_\epsilon}(z))$. As $\epsilon>0$ and~$\mathbb{P}_0$ admits an everywhere positive density with respect to Lebesgue measure, we have~$\kappa(z)>0$ for every~$z\in\mathbb{R}^m$. To prove the lemma, we must show that~$\kappa(z)$ is bounded away from~$0$ on~$B_{R_\epsilon}(0)$.
    
    By \autoref{ass:data_distribution}, the data-generating distribution $\mathbb{P}_0$ is absolutely continuous with respect to Lebesgue measure on~$\mathbb{R}^m$. Thus, for any fixed~$z\in\mathbb{R}^m$, the boundary of~$B_{r_\epsilon}(z)$ has $\mathbb{P}_0$-measure zero, and the function $h_{\widehat z}(z)\coloneq \mathbf{1}_{\{\widehat z\in B_{r_\epsilon}(z)\}}$ is continuous (in fact, constant) at~$z$ for $\mathbb{P}_0$-almost every~$\widehat z\in\mathbb{R}^m$. Consequently, $\kappa(z)$ is continuous thanks to the dominated convergence theorem \cite[Theorem~3.31]{axler2020measure}. As the closed ball~$B_{R_\epsilon}(0)$ is compact, there exists a minimizer~$z^\star\in B_{R_\epsilon}(0)$ with $\underline{p}(\epsilon)=\kappa(z^\star)>0$ by Weierstrass' extreme value theorem. Thus, the claim follows.

<!-- lemma-8: body_md -->
\label{lem:uncovered_inlier_fraction}
    If~\autoref{ass:data_distribution} holds, then we have
    \begin{equation}
    \label{eq:uncovered_inlier_bound}
        \mathbb{P}_0^B\left(\frac{|U_\epsilon|}{B} > \frac{\lambda\epsilon}{L_z^2}\right) \leq \frac{L_z^2}{\lambda\epsilon} \exp\left(-(B-1)\underline p(\epsilon)\right).
    \end{equation}
<!-- end lemma-8 -->
<!-- proof-16: body_md -->
Fix any index $i \in [B]$, and note that
    \begin{align*}
        \mathbb{P}_0^B(i \in U_\epsilon) & = \mathbb{P}_0^B(i\in O_\epsilon) \,\mathbb{P}_0^B(i \in U_\epsilon | i\in O_\epsilon) + \mathbb{P}_0^B(i \not\in O_\epsilon) \,\mathbb{P}_0^B(i \in U_\epsilon | i\not\in O_\epsilon) \\
        & \leq \mathbb{P}_0^B(i \in U_\epsilon | i\not\in O_\epsilon) = \mathbb{E}_{\mathbb{P}_0^B} \big[\mathbb{P}_0^B(i \in U_\epsilon | \widehat Z_i,\, i\not\in O_\epsilon) \big| i\not\in O_\epsilon\big] ,
    \end{align*}
    where the two equalities follow from the law of total probability, and the inequality holds because~$O_\epsilon\cap U_\epsilon=\emptyset$, which implies that $\mathbb{P}_0^B(i \in U_\epsilon | i\in O_\epsilon)=0$. The event $i\not\in O_\epsilon$ is completely determined by~$\widehat Z_i$ (that is, it belongs to the $\sigma$-algebra generated by~$\widehat Z_i$). As the empirical samples $\widehat Z_j$, $j\neq i$, are mutually independent as well as independent of~$\widehat Z_i$, the definition of~$U_\epsilon$ in~\eqref{def:uncovered_inliers_set} then implies that
    \begin{align*}
        \mathbb{P}_0^B \big(i \in U_\epsilon \big| \widehat Z_i,\, i\not\in O_\epsilon \big) &= \prod_{j\in[B]:\, j\neq i}\mathbb{P}_0^B \Big(\|\widehat Z_j - T^\star(\widehat Z_i)\|_2 > r_{\epsilon} \Big| \widehat Z_i,\, i\not\in O_\epsilon \Big) \\
        & \leq (1-\underline p(\epsilon))^{B-1} \leq \exp\bigl(-(B-1)\underline p(\epsilon)\bigr).
    \end{align*}
    Here, the first inequality follows from the definition of~$\underline p(\epsilon)$, which ensures that
    \[
        \mathbb{P}_0\Big( \|\widehat Z_j- T^\star(\widehat Z_i)\|_2\leq r_{\epsilon} \Big| \widehat Z_i,\, i\not\in O_\epsilon \Big) \ge \underline{p}(\epsilon).
    \]
    Combining the above estimates finally yields
    \begin{align*}
        \mathbb{P}_0^B(i \in U_\epsilon) & \leq \exp\bigl(-(B-1)\underline p(\epsilon)\bigr) \quad \forall i\in[B].
    \end{align*}
    As $|U_\epsilon|=\sum_{i=1}^B \mathbf 1_{\{i\in U_\epsilon\}}$, the expected number of uncovered inliers satisfies
    \begin{equation*}
        \mathbb{E}_{\mathbb{P}_0^B}[|U_\epsilon|] = \sum_{i=1}^B \mathbb{P}_0^B(i \in U_\epsilon) \le B \exp\bigl(-(B-1)\underline p(\epsilon)\bigr).
    \end{equation*}
    Note that $|U_\epsilon|$ is a non-negative random variable. By Markov's inequality, we thus obtain
    \begin{equation*}
        \mathbb{P}_0^B\left( \frac{|U_\epsilon|}{B} > \frac{\lambda\epsilon}{L_z^2} \right) \leq \frac{\mathbb{E}_{\mathbb{P}_0^B}[|U_\epsilon| / B]}{\lambda\epsilon / L_z^2} \leq \frac{L_z^2}{\lambda\epsilon} \exp\left(-(B-1)\underline p(\epsilon)\right).
    \end{equation*}
    This observation completes the proof.

<!-- lemma-9: body_md -->
\label{lem:suboptimality_level_sets}
    Suppose that~\autoref{ass:loss_function_smoothness} holds, and define $r_\epsilon$ as in~\eqref{def:bulk_radius}. If~$z^\star,z\in\mathbb{R}^m$ with~$f_i(z^\star)=f_i^\star$ and~$\|z-z^\star\|_2\leq r_\epsilon$, then $f_i(z)\geq f_i^\star-\epsilon/2$ for all~$i \in [B]$.
<!-- end lemma-9 -->
<!-- proof-17: body_md -->
Note that both~$f_i$ as well as~$-f_i$ are  $(L_{zz}+2\lambda)$-smooth by virtue of \autoref{ass:loss_function_smoothness}. Hence, the standard quadratic upper bound for the smooth function $-f_i$ \cite[Proposition~6.1.2]{bertsekas2015convex} yields
    \begin{align*}
        f_i(z) &\geq f_i(z^\star) + \langle\nabla f_i(z^\star),z-z^\star \rangle - \frac{L_{zz}+2\lambda}{2}\|z-z^\star\|_2^2 \\
        &\geq f_i^{\star} - \frac{L_{zz}+2\lambda}{2} r_\epsilon^2=f_i^{\star} -\frac{\epsilon}{2},
    \end{align*}
    where the second inequality holds because~$z^\star$ is a global maximizer of~$f_i$ satisfying~$\nabla f_i(z^\star) = 0$ and because~$\|z-z^\star\|_2\leq r_\epsilon$, and the equality follows from the definition of~$r_\epsilon$.

<!-- theorem-5: body_md -->
\label{thm:mpa_global_optimality}
    If \hyperref[ass:loss_function_smoothness]{Assumptions~\ref*{ass:loss_function_smoothness}} and~\ref{ass:data_distribution} hold and $0<\eta<2/(L_{zz}+2\lambda)$, then for any $K \in \mathbb{N}\cup\{\infty\}$, $R \in\mathbb{N}$, and tolerance $\epsilon > 0$, we have
    \begin{equation*}
        \mathbb{P}_0^B\bigl( \varepsilon(\theta, T_{\mathrm{mpa}}) \leq \epsilon \bigr) \geq 1 - \beta(B, \epsilon),
    \end{equation*}
    where $\varepsilon(\theta, T)$ is the adversarial error of a map $T$ with respect to $\widehat\mathbb{P}=\tfrac{1}{B} \sum_{i=1}^B \delta_{\widehat Z_i}$, and where
                                                                     
    \begin{equation*}
        \beta(B, \epsilon) \coloneqq \exp\left(-B\lambda\epsilon/6L_z^2\right) + \frac{L_z^2}{\lambda\epsilon} \exp\left(-(B-1)\underline p(\epsilon)\right).
    \end{equation*}
<!-- end theorem-5 -->
<!-- proof-18: body_md -->
We first rewrite the adversarial error of $T_{\mathrm{mpa}}$ by expanding the empirical expectation. Using the definition~\eqref{eq:oracle_error} of the adversarial error and the per-sample formulation~\eqref{eq:regularized_dro-dual} of $F(\theta)$, we can write
    \begin{align*}
        \varepsilon(\theta, T_{\mathrm{mpa}}) 
        &= \frac{1}{B}\sum_{i=1}^{B} \left( \max_{z \in \mathbb{R}^m} \bigl\{ f(\theta, z)\! -\! \lambda \|z - \widehat{Z}_i\|_2^2 \bigr\} \! - \! \left( f(\theta, T_{\mathrm{mpa}}(\widehat{Z}_i))\! -\! \lambda \|T_{\mathrm{mpa}}(\widehat{Z}_i) - \widehat{Z}_i\|_2^2 \right) \right) \\
        &= \frac{1}{B}\sum_{i=1}^{B} \left(f_i^\star - f_i(T_{\mathrm{mpa}}(\widehat{Z}_i)) \! \right).
    \end{align*}
    Next, we introduce $G_i \coloneqq f_i^\star - f_i(T_{\mathrm{mpa}}(\widehat{Z}_i))$ as shorthand for the per-sample suboptimality gap of~$T_{\mathrm{mpa}}$ for every~$i \in [B]$. Using the batch partition~\eqref{eq:batch-partition}, the adversarial error can thus be recast as
    \begin{align}
        \label{eq:error-decomposition}
        \varepsilon(\theta, T_{\mathrm{mpa}}) = \frac{1}{B}\sum_{i \in O_\epsilon} G_i + \frac{1}{B}\sum_{i \in U_\epsilon} G_i + \frac{1}{B}\sum_{i \in C_\epsilon} G_i.
    \end{align}
    By~\autoref{lem:outlier_fraction}, the probability of the event $|O_\epsilon|/B>\lambda\epsilon/L_z^2$ is bounded by $\exp(-B\lambda\epsilon/6L_z^2)$. Similarly, \autoref{lem:uncovered_inlier_fraction} ensures that the probability of the event $|U_\epsilon|/B>\lambda\epsilon/L_z^2$ is bounded by $L_z^2/(\lambda\epsilon)\exp(-(B-1)\underline p(\epsilon))$. Using the union bound, one thus verifies that the desirable event in which both $|O_\epsilon|/B\leq \lambda\epsilon/L_z^2$ and $|U_\epsilon|/B\leq \lambda\epsilon/L_z^2$ occurs with probability at least $1-\beta(B,\epsilon)$.

    It remains to be shown that the adversarial error is bounded by~$\epsilon$ in this desirable event. To this end, recall first that the ascent and reassignment operators are strictly monotone thanks to \autoref{lem:mpa_operators}. Hence, the objective value of $T_{\mathrm{mpa}}(\widehat{Z}_i)$ dominates that of $\widehat Z_i$, that is, $f_i(T_{\mathrm{mpa}}(\widehat{Z}_i)) \geq f_i(\widehat{Z}_i)$ for all $i\in[B]$. As $f(\theta, z)$ is $L_z$-Lipschitz continuous in~$z$, we may thus conclude that
    \begin{align*}
        G_i &\leq f_i^\star - f_i(\widehat{Z}_i) = \max_{z \in \mathbb{R}^m} \bigl\{ f(\theta,z) - f(\theta,\widehat{Z}_i) - \lambda\|z-\widehat{Z}_i\|_2^2 \bigr\} \\
            &\leq \max_{z \in \mathbb{R}^m} \bigl\{ L_z\|z-\widehat{Z}_i\|_2  - \lambda\|z-\widehat{Z}_i\|_2^2 \bigr\} = \frac{L_z^2}{4\lambda}.
    \end{align*}
    Using this coarse estimate, we obtain
    \begin{subequations}
    \begin{equation}
        \label{eq:O_epsilon_U_epsilon-bound}
        \frac{1}{B}\sum_{i \in O_\epsilon} G_i \le \frac{|O_\epsilon|}{B} \frac{L_z^2}{4\lambda} \le \frac{\epsilon}{4} \quad \text{and} \quad \frac{1}{B}\sum_{i \in U_\epsilon} G_i \le \frac{|U_\epsilon|}{B} \frac{L_z^2}{4\lambda} \le \frac{\epsilon}{4}.
    \end{equation}
    If~$i\in C_\epsilon$, then we have $\|T^\star(\widehat{Z}_i)\|_2\leq R_\epsilon$, and there is $j\neq i$ with $\|\widehat Z_j - T^\star(\widehat Z_i)\|_2 \leq r_{\epsilon}$; see~\eqref{def:covered_inliers_set}. By \autoref{lem:suboptimality_level_sets}, we thus find $f_i(\widehat{Z}_j) \ge f^\star_i - \epsilon/2$ and~$G_i \le \epsilon/2$ for all $i\in C_\epsilon$, which implies that
    \begin{equation}
        \label{eq:C_epsilon-bound}
        \frac{1}{B}\sum_{i \in C_\epsilon} G_i \le \frac{|C_\epsilon|}{B} \frac{\epsilon}{2} \le \frac{\epsilon}{2}.
    \end{equation}
    \end{subequations}
    Substituting~\eqref{eq:O_epsilon_U_epsilon-bound} and~\eqref{eq:C_epsilon-bound} into~\eqref{eq:error-decomposition} yields $\varepsilon(\theta, T_{\mathrm{mpa}}) \leq \epsilon$, and thus the claim follows.

<!-- corollary-2: body_md -->
\label{cor:mpa_sample_complexity}
    If all assumptions of \autoref{thm:mpa_global_optimality} hold, $\beta\in(0,1)$, and the batch size satisfies
    \begin{equation*}
        B \geq \max\left\{ \frac{6L_z^2}{\lambda\epsilon}\log\left(\frac{2}{\beta}\right), \; 1 + \frac{1}{\underline{p}(\epsilon)}\log\left(\frac{2L_z^2}{\lambda\epsilon\beta}\right) \right\},
    \end{equation*}
    then we have $\varepsilon(\theta, T_{\mathrm{mpa}}) \leq \epsilon$ with probability at least $1-\beta$.
<!-- end corollary-2 -->
<!-- proof-19: body_md -->
To guarantee that the failure probability $\beta(B, \epsilon)$ from \autoref{thm:mpa_global_optimality} falls below $\beta$, it suffices to bound each of its two additive terms by $\beta/2$. As for the first term, we thus require
    \begin{equation*}
        \exp\left(-\frac{B\lambda\epsilon}{6L_z^2}\right) \leq \frac{\beta}{2} \quad\implies \quad B \geq \frac{6L_z^2}{\lambda\epsilon}\log\left(\frac{2}{\beta}\right).
    \end{equation*}
    Similarly, we require the second term to satisfy
    \begin{equation*}
        \frac{L_z^2}{\lambda\epsilon}\exp\bigl(-(B-1)\underline p(\epsilon)\bigr) \leq \frac{\beta}{2} \quad \implies \quad B \geq 1 + \frac{1}{\underline p(\epsilon)}\log\left(\frac{2L_z^2}{\lambda\epsilon\beta}\right).
    \end{equation*}
    
    Taking $B$ to be the maximum of these two lower bounds ensures both conditions hold simultaneously. This observation completes the proof.

<!-- remark-4: body_md -->
\label{rem:covering_mass}
The sample complexity bound in \autoref{cor:mpa_sample_complexity} is governed primarily by the inverse covering mass $1/\underline p(\epsilon)$, which conceals an exponential dependence on the ambient dimension~$m$. Indeed, one can construct at least $(R_\epsilon/2r_\epsilon)^m$ disjoint balls of radius~$r_\epsilon$ with centers in $B_{R_\epsilon}(0)$. As their masses sum to at most one, at least one of them has mass at most $(2r_\epsilon/R_\epsilon)^m$, and since $R_\epsilon$ grows when $\epsilon$ decreases, we conclude that
$p(\epsilon)=\mathcal{O}(r_\epsilon^m)=\mathcal{O}(\epsilon^{m/2})$ irrespective of the data-generating distribution~$\mathbb{P}_0 \in \mathcal{P}(\mathbb{R}^m)$. The factor $\epsilon^{m/2}$ thus reflects the volume of the covering ball rather than any tail behavior of~$\mathbb{P}_0$. One can show that if~$\mathbb{P}_0$ is a standard normal distribution, then $p(\epsilon)=\Omega(\epsilon^{m/2})$. Even in this best possible case, \autoref{cor:mpa_sample_complexity} requires a batch of size $B=\widetilde\mathcal{O}(\max\{\epsilon^{-1},\epsilon^{-m/2}\})$, which grows exponentially with the ambient dimension~$m$.
<!-- end remark-4 -->
<!-- remark-5: body_md -->
\label{rem:mpa_vs_pa}
The global optimality guarantee of \autoref{cor:mpa_sample_complexity} requires a large batch of sufficiently diverse empirical samples. In the absence of such a batch, MPA may not output a globally optimal adversarial map. In fact, as each round of MPA starts with a reassignment step, the adversarial error of the resulting MPA map can even exceed that of the corresponding PA map. To see this, suppose that $f_i(\theta,\widehat z_i)<f_i(\theta,\widehat z_j)$, while the stationary point in the basin containing~$\widehat z_j$ has a lower objective value than the stationary point in the basin containing~$\widehat z_i$. To construct the $i$-th adversarial sample, PA applies gradient ascent initialized at~$\widehat z_i$, whereas MPA moves from~$\widehat z_i$ to~$\widehat z_j$ due to the first reassignment step and then applies gradient ascent initialized at~$\widehat z_j$. In this case, PA ends up in the better stationary point and thus finds a better adversarial example than MPA. 

To ensure that MPA dominates PA, one could reverse the order of the reassignment and gradient ascent steps in the inner loop of Algorithm~\ref{alg:implicit_mpa}. The first gradient ascent of this alternative variant of MPA replicates the PA map exactly. Moreover, since both the reassignment operator and the $t$-step ascent operator are monotone by \autoref{lem:mpa_operators}\,{\em (iii)}, every subsequent MPA update can only improve upon PA, regardless of the batch size. The price of this guarantee, however, is the loss of a global optimality result analogous to \autoref{thm:mpa_global_optimality}. In particular, one can construct problem instances for which the alternative variant of MPA produces a strictly suboptimal adversarial map for every possible batch of empirical samples. We tested both variants of MPA in our experiments and found that the original version of Algorithm~\ref{alg:implicit_mpa} consistently attains the lower adversarial error. Hence, we do not pursue the ascent-first variant of MPA further.
