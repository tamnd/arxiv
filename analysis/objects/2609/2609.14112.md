\label{L: artin schreier preimage}
    For all $\eta \in X$, we have $\operatorname{AS}^{-1}[\lambda] = \operatorname{Spec} Z_\eta({\mathfrak g})$ as schemes. These are precisely the Harish-Chanda central characters associated to $U_X({\mathfrak g})$-modules.
<!-- end lemma-4 -->
<!-- proof-1: body_md -->
Since the Harish-Chandra centre $U({\mathfrak g})^G$ surjects onto $Z_\eta({\mathfrak g})$, we have that $Z_\eta({\mathfrak g}) = \operatorname{im}(U({\mathfrak g})^G \to U_\eta({\mathfrak g}))$.
    Therefore a maximal ideal of $Z_\eta({\mathfrak g})$ corresponds to a maximal ideal of $U({\mathfrak g})^G$ whose image remains proper in $Z_\eta({\mathfrak g})$.
    Thus
    \begin{equation}
        \operatorname{Spec} Z_\eta({\mathfrak g}) = \{ \psi \in \operatorname{Hom}(U({\mathfrak g})^G, \mathbbm{k}) \mid \ker(U({\mathfrak g})^G \to U_\eta({\mathfrak g})) \subseteq \ker \psi \}.
    \end{equation}
   Note also that $\ker(U({\mathfrak g})^G \to U_\eta({\mathfrak g})) = I_\eta U({\mathfrak g}) \cap U({\mathfrak g})^G$.

    The Artin-Schreier map  is the morphism associated to the inclusion $Z_p({\mathfrak g})^G \subseteq U({\mathfrak g})^G$, see \eqref{eq:ArtinSchreier}.
    Identify $Z_p({\mathfrak g})^G = {\mathbbm k}[(\mathfrak t^*)^{(1)}/W]$ via \eqref{eq:Chevalleyrestriction}.
    If $J_\lambda$ denotes the defining ideal of $[\lambda]$ in $Z_p({\mathfrak g})^G$ then $\operatorname{AS}^{-1}[\lambda]$ corresponds to the maximal ideals of $U({\mathfrak g})^G$ containing $J_\lambda U({\mathfrak g})^G$.
    Therefore
    \begin{equation}
        \operatorname{AS}^{-1}[\lambda] = \{ \psi \in \operatorname{Hom}(U({\mathfrak g})^G, \mathbbm{k}) \mid J_\lambda U({\mathfrak g})^G \subseteq \ker \psi \}.
    \end{equation}

    To prove the current lemma, it is enough to show that $J_\lambda U({\mathfrak g})^G = I_\eta U({\mathfrak g}) \cap U({\mathfrak g})^G$.
    Since $\eta \in X$, we have that $J_\lambda \subseteq I_\eta$ and so $J_\lambda = I_\eta^G$.
    Hence it is clear that $I_\eta^G U({\mathfrak g})^G \subseteq I_\eta U({\mathfrak g}) \cap U({\mathfrak g})^G$. Equivalently we have a surjection of algebras
    \begin{eqnarray}
    \label{eq:alghom}
        U({\mathfrak g})^G / I_\eta^G U({\mathfrak g})^G \twoheadrightarrow U({\mathfrak g})^G / I_\eta U({\mathfrak g}) \cap U({\mathfrak g})^G.
    \end{eqnarray}

    By \cite[Theorem 8.2]{Special Slices} we have that $\operatorname{im}(U({\mathfrak g})^G \to U_\eta({\mathfrak g}))$ has dimension $p^{\operatorname{rank} {\mathfrak g}}$.
    Further, $J_\lambda = I_\eta^G$ is a maximal ideal $Z_p({\mathfrak g})^G$ and from \cite[Theorem 3.5(2)]{Brown Gordon} we know that $U({\mathfrak g})^G$ is a free $Z_p({\mathfrak g})^G$-module of rank $p^{\operatorname{rank} {\mathfrak g}}$. The latter implies that the left hand side of \eqref{eq:alghom} also has dimension $p^{\operatorname{rank}{\mathfrak g}}.$
    This implies that \eqref{eq:alghom} is an isomorphism. This implies $J_\lambda U({\mathfrak g})^G = I_\eta U({\mathfrak g}) \cap U({\mathfrak g})^G$ and the proof is complete.

<!-- theorem-2: body_md -->
\label{T:blockdecomp}
    The diagonal map
    \begin{equation} \label{eq: block decomp phi}
        \phi \colon U_X({\mathfrak g}) \to \bigoplus_{\psi \in \operatorname{AS}^{-1}[\lambda]} U_X^{(\psi)} ({\mathfrak g})
    \end{equation}
    is an isomorphism.

<!-- proof-2: body_md -->
Write $U_\eta^{(\psi)}({\mathfrak g})$ for the quotient $U_\eta({\mathfrak g})/\ker(\psi)^dU_\eta({\mathfrak g})$. We first prove that the specialised map
\begin{equation} \label{eq: phi eta}
    \phi_\eta \colon U_\eta({\mathfrak g}) \to \bigoplus_{\psi \in \operatorname{AS}^{-1}[\lambda]} U_\eta^{(\psi)}({\mathfrak g}) 
\end{equation}
is an isomorphism for any $\eta\in X$.
By a result of M\"{u}ller \cite[Theorem 7]{Muller} (also see \cite[2.10]{Brown Gordon} for the theorem applied directly to our context) we have the decomposition into blocks
\begin{equation}
    U_\eta({\mathfrak g}) = \bigoplus_{i=1}^k U_\eta({\mathfrak g}) e_i
\end{equation}
where $e_1, \ldots, e_k$ are the primitive central idempotents in $Z_\eta({\mathfrak g})$.
For each $e_i$ we let $I_i$ be the unique maximal ideal in the local ring $Z_\eta({\mathfrak g}) e_i$, see \cite[Theorem 8.7]{AM}.

We then define
\begin{equation}
    J_i = I_i \oplus \bigoplus_{j \neq i} Z_\eta({\mathfrak g}) e_j.
\end{equation}
Since $I_i$ is nilpotent, with nilpotency index bounded above by $d = \dim U_\eta({\mathfrak g})$, we then have
\begin{equation}
    J_i^d = \bigoplus_{j \neq i} Z_\eta({\mathfrak g}) e_j
\end{equation}
so that $U_\eta({\mathfrak g}) / J_i^d U_\eta({\mathfrak g}) \cong U_\eta({\mathfrak g}) e_i$.
Thus by applying Lemma \ref{L: artin schreier preimage} we see that (\ref{eq: phi eta}) is an isomorphism.

{\it Claim 1:}    The map $\phi$ given in (\ref{eq: block decomp phi}) is surjective.

\label{thm: GF(X) equivariance}

    \end{eqnarray}

<!-- proof-14: body_md -->
Since $G^\chi$ is a closed subgroup scheme of $G$, and $G^{F(\chi)} = F^{-1}_G ((G^{\chi})^{(1)})$, we have an equivalence between $G^{F(\chi)} \!\operatorname{-mod}$ and $(U_0 ({\mathfrak g}), G^\chi)\!\operatorname{-mod}$ by Lemma~\ref{L:monoidalequivalence}.
    Thus, the category $U_\chi({\mathfrak g}) \!\operatorname{-mod}^{G^{F(\chi)}}_{\operatorname{fg}}$ is equivalent to $U_\chi({\mathfrak g}) \!\operatorname{-mod}^{(U_0({\mathfrak g}), G^\chi)}_{\operatorname{fg}}$, using notation introduced in Section \ref{ss:equivariantmodules}.
    We will show that $U_\chi({\mathfrak g}) \!\operatorname{-mod}^{(U_0({\mathfrak g}), G^\chi)}_{\operatorname{fg}}$ is equivalent to the category $\mathrm{HC}_\chi^{G^\chi} U({\mathfrak g})$.

    We recall the structures of modules in both categories.
    
    A module $M \in U_\chi({\mathfrak g}) \!\operatorname{-mod}^{(U_0({\mathfrak g}), G^\chi)}_{\operatorname{fg}}$ has three actions $\rho \colon U_\chi({\mathfrak g}) \rightarrow \operatorname{End}(M)$, $\sigma \colon U_0({\mathfrak g}) \rightarrow \operatorname{End}(M)$, and $\tau \colon G^\chi \rightarrow \mathrm{GL}(M)$. Both $\rho$ and $\sigma$ are $G^\chi$-equivariant, $\rho$ is $U_0({\mathfrak g})$-equivariant and $d_1\tau = \sigma|_{{\mathfrak g}^\chi}$.

    A module $M \in \mathrm{HC}_\chi^{G^\chi} U({\mathfrak g})$ has three actions $\rho \colon U_\chi({\mathfrak g}) \rightarrow \operatorname{End}(M)$, $\upsilon \colon U_{\chi}({\mathfrak g}) \to \operatorname{End}(M)$ and $\tau : G^\chi \to \mathrm{GL}(M)$. Both $\rho$ and $\upsilon$ are $G^\chi$-equivariant commuting actions, and $d_1 \tau = (\rho - \upsilon)|_{{\mathfrak g}^\chi}$.

    To construct quasi-inverse functors between these categories, we take $M \in U_\chi({\mathfrak g}) \!\operatorname{-mod}^{(U_0({\mathfrak g}), G^\chi)}_{\operatorname{fg}}$ and construct a Harish-Chandra bimodule with the same underlying vector space, the same actions $\rho, \tau$ and right action $\upsilon := \sigma - \rho$. The data $(\rho, \sigma, \tau)$ can be recovered from $(\rho, \upsilon, \tau)$, so it suffices to show that the functor is well-defined.
    
    We claim that $G^\chi$-equivariance of $\rho$ is equivalent to $\rho$ and $\upsilon$ commuting. Equivariance states that
    for $x, y \in {\mathfrak g}$ and $g \in G^\chi$ we have
    \begin{align}
        \sigma(x) \rho(y) &= \rho(\operatorname{ad}(x)y) + \rho(y)\sigma(x). \label{eq: sigma equivariance}
    \end{align}
    whilst $[\rho, \upsilon] = 0$ states that
    \begin{equation}
    \label{eq:piggletog}
        [(\sigma - \rho)(x), \rho(y)] = [\sigma(x), \rho(y)] - [\rho(x), \rho(y)] = \sigma(x) \rho(y) - \rho(y) \sigma(x) - \rho(\operatorname{ad}(x)y) = 0.
    \end{equation}
    Equations \eqref{eq: sigma equivariance} and \eqref{eq:piggletog} are equivalent, and this completes the proof.
