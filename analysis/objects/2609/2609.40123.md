<!-- thm-1: body_md -->
\label{thm:main}
Assume that $H_2(\mu)>12$, and let $H\ge 1$ such that $|\omega_j|\le H$ for all~$j$ almost surely. Then there are constants $C=C(\mu)$ and $n_0=n_0(\mu,H)$ such that
\[
  \mathbb{P}(\operatorname{disc}(R)\text{ is a square}) \le \frac{C(\mu)}{\log n}
\]
for all $n\ge n_0$.
<!-- end thm-1 -->
<!-- cor-1: body_md -->
\label{cor1}
Let $N>\exp(12)$ be an integer and assume that $\mu$ is supported uniformly on $N$ consecutive integers in $\mathbb{Z}$. Then there is a constant $C'(\mu)>0$ such that
\[
  \mathbb{P}(R\text{ is irreducible and has Galois group }\operatorname{Sym}(n) \,\ifnum\currentgrouptype=16\middle\fi|\, \omega_0\ne 0) \ge 1- \frac{C'(\mu)}{\log n}
\]
for all sufficiently large $n$.
<!-- end cor-1 -->
<!-- fact-1: body_md -->
For any two polynomials $P_1,P_2\in\mathbb{Z}[x]$ have degrees $m,n$ respectively, then we have
\[
|\operatorname{Res}(P_1,P_2)| \le \| P_1 \|_2^n \| P_2 \|^m \le  (m+1)^{n/2}(n+1)^{m/2}H(P_1)^nH(P_2)^m.
\]
<!-- end fact-1 -->
<!-- fact-2: body_md -->
\label{fact:disc-bound}
For any polynomial $P(x)\in\mathbb{Z}[x]$ of degree~$n$, the Mahler measure of $P$ satisfies
\[
M(P) \le \| P \|_2 \le \sqrt{n+1}H(P).
\]
Therefore, writing the discriminant as a determinant and using Hadamard's bound gives
\[
|\operatorname{disc}(P)| \le n^n M(P)^{2n-2} \le n^n(\sqrt{n+1}H(P))^{2n-2} \le (nH(P))^{2n}.
\]
<!-- end fact-2 -->
<!-- lem-1: body_md -->
\label{lem:count-inadmissible}
Let $p$ be a prime. Then
\[
\sum_{k=1}^{\infty}\left|\mathcal{B}_{p^k}\right| \le n^2.
\]
<!-- end lem-1 -->
<!-- proof-1: body_md -->
By definition, the sets $\left\{\mathcal{B}_{p^k}\right\}_{k=1}^{\infty}$ are pairwise disjoint, and their union is precisely the set of nonzero elements of $\overline{\mathbb{F}_p}$ of multiplicative order less than $n$. For each $1\le d\le n$ there are at most $\varphi(d)$ elements in $\overline{\mathbb{F}_p}$ of order $d$, hence
\[
\sum_{k=1}^{\infty}\left|\mathcal{B}_{p^k}\right| = \left|
    \mathop{\ifx\mathop\mathop\vphantom{\bigcup}\fi
        \mathpalette\mov@rlay{\bigcup\cr\mbox{\larger[2]$\cdot$}}
      }
    \ifx\mathop\mathop\expandafter\displaylimits\fi_{k=1}^{\infty}\mathcal{B}_{p^k}\right| \le \sum_{d=1}^{n-1} \varphi(d) \le n^2. \qedhere
\]
<!-- end proof-1 -->

<!-- lem-2: body_md -->
\label{lem:linear-transform}
For any $\alpha\in\mathcal{V}$ and integers $d\ge 1$ and $l,t\ge 0$, we have
\[
\nu_{\alpha}^{(d+t,d+l+t)} = (A_{\alpha}^t)_*\nu_{\alpha}^{(d,d+l)},
\]
where $A_{\alpha}$ is the linear operator defined on $\mathcal{V}^r$ by
\[
(A_{\alpha}v)^{(s)} = \alpha v^{(s)} + sv^{(s-1)}
\]
for every $v=(v^{(0)},\dots,v^{(r-1)})\in\mathcal{V}^r$ and $0\le s\le r-1$ (where $v^{(-1)}=0$).
<!-- end lem-2 -->
<!-- proof-2: body_md -->
For the operator $A_{\alpha}$ defined above, we have
\begin{align*}
  (A_{\alpha}\mathbf{v}_j(\alpha))^{(s)} &= \alpha(\mathbf{v}_j(\alpha))^{(s)} + s(\mathbf{v}_j(\alpha))^{(s-1)} = \mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{j}{s}\right|}}
  {\left(\genfrac{}{}{0pt}{}{j}{s}\right|}{\left(\genfrac{}{}{0pt}{}{j}{s}\right|}{\left(\genfrac{}{}{0pt}{}{j}{s}\right|}\alpha^{j-s+1} + s\mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{j}{s-1}\right|}}
  {\left(\genfrac{}{}{0pt}{}{j}{s-1}\right|}{\left(\genfrac{}{}{0pt}{}{j}{s-1}\right|}{\left(\genfrac{}{}{0pt}{}{j}{s-1}\right|}\alpha^{j-(s-1)} \\
  &= \mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{j+1}{s}\right|}}
  {\left(\genfrac{}{}{0pt}{}{j+1}{s}\right|}{\left(\genfrac{}{}{0pt}{}{j+1}{s}\right|}{\left(\genfrac{}{}{0pt}{}{j+1}{s}\right|}\alpha^{j+1-s} = (\mathbf{v}_{j+1}(\alpha))^{(s)}
\end{align*}
for every $j\ge 0$ and $0\le s\le r-1$. Therefore $A_{\alpha}\mathbf{v}_j(\alpha)=\mathbf{v}_{j+1}(\alpha)$, so by induction we have $A_{\alpha}^t\mathbf{v}_j(\alpha)=\mathbf{v}_{j+t}(\alpha)$ for any $t\ge 0$. Since $\omega_{d+t},\dots,\omega_{d+t+l}$ have the same joint law as $\omega_d,\dots,\omega_{d+l}$, then $\sum_{j=d+t}^{d+l+t}\omega_j\mathbf{v}_j(\alpha) = A_{\alpha}^t\sum_{j=d}^{d+l}\omega_{j+t}\mathbf{v}_j(\alpha)$ has the same law as $A_{\alpha}^t\sum_{j=d}^{d+l}\omega_j\mathbf{v}_j(\alpha)$, so the claim follows from \eqref{eq:expr-vj}.

<!-- lem-3: body_md -->
\label{lem:fourier-mu-trans}
For any $v,\xi\in\mathcal{V}^r$,
\[
\left|\widehat{\mu.\delta_v}(\xi)\right| \le \exp\left(-\sum_{a_1,a_2\in\mathbb{Z}}\frac{\mu(a_1)\mu(a_2)([\Psi\circ\operatorname{tr}(\left\langle(a_1-a_2)\xi,v\right\rangle)]^{\sim})^2}{Q^2}
\right)
\]
<!-- end lem-3 -->
<!-- proof-3: body_md -->
Since $\widehat{\mu.\delta_v}(\xi)=\sum\mu(s)\xi_\zeta(av)$ and since $\mu$ is real we get
\[
\left|\widehat{\mu.\delta_v}(\xi)\right|^2 = \sum_{a_1,a_2\in\mathbb{Z}}\mu(a_1)\mu(a_2)\chi_{\xi}((a_1-a_2)v).
\]
As $\left|\widehat{\mu.\delta_v}(\xi)\right|^2\in\mathbb{R}$ and $\operatorname{Re}(e_Q(a)) \le 1-2([a]^{\sim})^2/Q^2$ if $a\in\mathbb{Z}/Q\mathbb{Z}$, we may use the definition of $\chi_{\xi}$ and deduce
\begin{align*}
\left|\widehat{\mu.\delta_v}(\xi)\right|^2 & \le 1 - 2\sum_{a_1,a_2\in\mathbb{Z}}\frac{\mu(a_1)\mu(a_2)([\Psi\circ\operatorname{tr}(\left\langle\xi,(a_1-a_2)v\right\rangle)]^{\sim})^2}{Q^2} \\
&= 1 - 2\sum_{a_1,a_2\in\mathbb{Z}}\frac{\mu(a_1)\mu(a_2)([\Psi\circ\operatorname{tr}(\left\langle(a_1-a_2)\xi,v\right\rangle)]^{\sim})^2}{Q^2}.
\end{align*}
However, for any $t$ we have $1-t\le\exp(-t)$, so the claim follows.

<!-- cor-2: body_md -->
\label{cor:fourier-nu-bound}

\left|\widehat{\nu}_{\alpha}^{(d,d+l)}(\xi)\right| \le \exp\left(-\sum_{a_1,a_2\in\mathbb{Z}}\left(\mu(a_1)\mu(a_2)\sum_{j=d}^{d+l}\frac{([S_j((a_1-a_2)\xi)]^{\sim})^2}{Q^2}
\right)\right)
\]
<!-- end cor-2 -->
<!-- proof-4: body_md -->
The Fourier transform of a convolution is the product of the Fourier transforms, so
\[
\left|\widehat{\nu}_{\alpha}^{(d,d+l)}(\xi)\right| = \left|\prod_{j=d}^{d+l}\widehat{\mu.\delta_{\mathbf{v}_j(\alpha)}}(\xi)\right|.

We now apply {Lemma~\ref{lem:fourier-mu-trans}} to get

proving the corollary.

<!-- lem-4: body_md -->

\end{equation}
in $\mathbb{F}_p$ for every $0\le m\le r\sum_{l=1}^tk_l - 1$. Then $P(\alpha_l)=0$ for every $1\le l\le t$, and \eqref{eq:pol-apply} holds for all $m\ge 0$.

We first note that for any $m\ge 0$ and any $0\le s\le r-1$,
\begin{align*}
  \left(\sum_{i=0}^e b_i\mathbf{v}_{i+m}(\alpha_l)\right)_s &= \sum_{i=0}^e \mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{i+m}{s}\right|}}
  {\left(\genfrac{}{}{0pt}{}{i+m}{s}\right|}{\left(\genfrac{}{}{0pt}{}{i+m}{s}\right|}{\left(\genfrac{}{}{0pt}{}{i+m}{s}\right|} b_i\alpha_l^{i+m-s} \\
  &= \sum_{i=0}^e \sum_{u=0}^s \binom{s}{u}\mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}}
  {\left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}{\left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}{\left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}(m)_u b_i \alpha_l^{i+m-s} \\
  &= \sum_{u=0}^s \binom{s}{u}\mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{m}{u}\right|}}
  {\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|} \alpha_l^{m-u} \sum_{i=0}^e  \mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}}
  {\left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}{\left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}{\left(\genfrac{}{}{0pt}{}{i}{s-u}\right|}b_i\alpha_l^{i-(s-u)} \\
  &= \sum_{u=0}^s \binom{s}{u}\mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{m}{u}\right|}}
  {\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|} \alpha_l^{m-u} P^{(s-u)}(\alpha_l).
\end{align*}
For each $s\in\{0,\dotsc,r-1\}$ we use this to rewrite \eqref{eq:pol-apply} and sum over $s$ to get
\[
\sum_{l=1}^t \sum_{s=0}^{r-1} \sum_{u=0}^s \binom{s}{u}\mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{m}{u}\right|}}
  {\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|} \operatorname{tr}_{\mathbb{F}_{q_l}/\mathbb{F}_p}\left(\beta_l^{(s)}\alpha_l^{m-u} P^{(s-u)}(\alpha_l)\right) = 0,
\]
namely
\[
\sum_{l=1}^t \sum_{s=0}^{r-1} \sum_{u=0}^s \sum_{j=0}^{k_l-1} \binom{s}{u}\mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{m}{u}\right|}}
  {\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|} (\beta_l^{(s)})^{p^j}\alpha_l^{p^j(m-u)} P^{(s-u)}(\alpha_l^{p^j}) = 0.
\]
Changing the order of summation of the last equation, we may rewrite it as
\begin{equation}\label{eq:pol-apply2}
  \sum_{l=1}^t  \sum_{u=0}^{r-1} \sum_{j=0}^{k_l-1} \mathchoice{{\textstyle \left(\genfrac{}{}{0pt}{}{m}{u}\right|}}
  {\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|}{\left(\genfrac{}{}{0pt}{}{m}{u}\right|} \alpha_l^{p^j(m-u)} x_{l,u,j}= 0
\end{equation}
where $x_{l,u,j} = \sum_{s=u}^{r-1} \binom{s}{u} (\beta_l^{(s)})^{p^j} P^{(s-u)}(\alpha_l^{p^j})$ for all $l,u,j$.

We think of $\left\{x_{l,u,j}\right\}$ as variables, and put the coefficients of \eqref{eq:pol-apply2} for all $m=0,\dots,r\sum_{l=1}^t k_l-1$ in the columns of a square matrix $A$. Then $A$ is the confluent Vandermonde matrix corresponding to $\big\{\alpha_l^{p^j}\big\}$ of order $r$, and thus its determinant is a nonzero scalar multiple of $\prod_{(l,j)<(l',j')}\big(\alpha_l^{p^j} - \alpha_{l'}^{p^{j'}}\big)^{r^2}$. In particular, the matrix is non-singular thanks to the assumptions on $\alpha_1,\dots,\alpha_t$ (in particular, the fact that $\alpha$ is generic). This forces $x_{l,u,j}=0$ for all $l,u,j$.

Fix $1\le l\le t$, and let $u$ be the maximal index so that $\beta_l^{(u)}\ne 0$. Then $x_{l,u,0} = \beta_l^{(u)}P(\alpha_l)=0$, so $P(\alpha_l)=0$. Moreover, \eqref{eq:pol-apply} holds for all $m\ge 0$ since $x_{l,u,j}=0$ for all $l,u,j$, completing the proof.

<!-- lem-5: body_md -->
\label{lem:approx-ideal} If $\Lambda(X)$ contains a nonzero polynomial of degree at most $\frac{N}{2}$, then there is a unique (up to multiplication by $\pm 1$) polynomial $P_0$ in $\Lambda(X)$ of minimal degree and with relatively prime coefficients. In this case, any polynomial in $\Lambda(X)$ of degree at most $N-\deg P_0$ must be divisible by $P_0$.
<!-- end lem-5 -->
<!-- proof-6: body_md -->
We set $E=3\left\lfloor \log Q^{rD} \right\rfloor$, so that, after increasing $C_0$ if necessary,
\[
1\le E\le \frac{1}{3}L,\qquad 2^E>Q^{rD},\qquad\text{and}\qquad \left\lfloor L/6E \right\rfloor\ge 4\log E.
\]
We will first show that there exists a nonzero polynomial $P_1\in\Lambda\big(\{\widetilde{S_j}\}_{j=0}^L\big)$ of degree at most $E$.

We consider the set
\[
\Omega\coloneqq\left\{(\sigma_0,\dots,\sigma_E)\in\left\{\pm 1\right\}^{E+1} \,\ifnum\currentgrouptype=16\middle\fi|\, \left|\sum_{j=0}^E\sigma_j\widetilde{S_{j+m}}\right| < \frac{Q}{2}\text{ for every }0\le m\le L-E\right\}.
\]
Let $\sigma_0,\dots,\sigma_E$ be independent, unbiased $\pm 1$-valued random variables. By Hoeffding's inequality and our assumption, for any $0\le m\le L-E$ we have
\[
\mathbb{P}\left(\left|\sum_{j=0}^E\sigma_j\widetilde{S_{j+m}}\right| \ge \frac{Q}{2}\right) \le 2\exp\left(-\frac{(Q/2)^2}{2\sum_{j=0}^E\widetilde{S_{j+m}}^2}\right) \le \frac{1}{2L}.
\]
Therefore $|\Omega|\ge \frac{1}{2}2^{E+1}>Q^{rD}$. By the pigeonhole principle, there exist distinct $x,y\in\Omega$ such that
\[
\sum_{j=0}^E x_j\widetilde{S_{j+m}} = \sum_{j=0}^E y_j\widetilde{S_{j+m}}
\]
for all $0\le m\le rD-1$. Put $a_j=\frac{x_j-y_j}{2}\in\left\{0,\pm 1\right\}$. Then the coefficients $a_j$ are not all zero, and
\begin{equation}\label{eq:sum-aj-Sj}
\sum_{j=0}^E a_j\widetilde{S_{j+m}}=0
\end{equation}
for all $0\le m\le rD-1$.

We now apply Lemma~\ref{lem:apply-pol} separately for each prime $p_j$, using only those coordinates $l$ for which $\beta_{j,l}\ne 0$. Since $\alpha$ is generic, the hypotheses of {Lemma~\ref{lem:apply-pol}} hold for these primes. As for the hypothesis \eqref{eq:pol-apply}, we take \eqref{eq:sum-aj-Sj} modulo $Q$ (i.e.\ remove the tildes) and use the fact that the $p_j$ are distinct, hence $\Psi$ is one-to-one. This shows that \eqref{eq:pol-apply} holds and we conclude from Lemma~\ref{lem:apply-pol} that
\[
\sum_{j=0}^E a_jS_{j+m}=0
\]
in $\mathbb{Z}/Q\mathbb{Z}$ for every $m\ge 0$.

Since $x,y\in\Omega$, we can return the tildes (by the definition of $\Omega$, $|\sum a_j\tilde{S}_{j+m}|<Q/2$ and is congruent to $\sum a_jS_{j+m}$ modulo $Q$). We get
\[
\sum_{j=0}^E a_j\widetilde{S_{j+m}}=0  
\]
for every $0\le m\le L-E$. Thus $P_1(x)\coloneqq a_0+a_1x+\cdots+a_Ex^E$ belongs to
\[
\Lambda_E(\{\widetilde{S_j}\}_{j=0}^L) \subseteq \Lambda(\{\widetilde{S_j}\}_{j=0}^{\left\lceil 2L/3 \right\rceil})
\]
(the inclusion holds because $E\le L/3$).

Since $\left\lceil 2L/3 \right\rceil\ge 2E$, Lemma~\ref{lem:approx-ideal}  
gives a unique, up to sign, polynomial $P_0\in\Lambda(\{\widetilde{S_j}\}_{j=0}^{\left\lceil 2L/3 \right\rceil})$ of minimal degree and with relatively prime coefficients. It follows that $P_0\mid P_1$, so $\deg P_0\le E$.

Returning to our application of Lemma~\ref{lem:apply-pol}, we will now use the first conclusion of the lemma. We get, for each prime, that  
$P_0(\alpha_{j,l})=0$ whenever $\beta_{j,l}\ne 0$. Thus, for each such pair $(j,l)$, there is an irreducible factor $P_{j,l}$ of $P_0$ such that $P_{j,l}(\alpha_{j,l})=0$.

We now bound the Mahler measure of each $P_{j,l}$. Let $\zeta_1,\dots,\zeta_{\deg P_{j,l}}$ be the complex roots of $P_{j,l}$. Write $s=\left\lfloor L/6E \right\rfloor$, so that $s\ge 4\log E$. By \cite[Lemma 26]{BreuillardVarju19}, there is a prime $\ell\in(s,2s]$ such that $\zeta_a/\zeta_b$ is not an $\ell$-th root of unity for any $a\ne b$. Hence the numbers $\zeta_1^\ell,\dots,\zeta_{\deg P_{j,l}}^\ell$ are all distinct.

Using the same pigeonhole argument as above, we can find a nonzero polynomial
\[
P_2(x)=\widetilde{P}_2(x^\ell), \qquad \widetilde{P}_2(x)=b_0+b_1x+\cdots+b_Ex^E,
\]
with $b_j\in\left\{0,\pm 1\right\}$, such that
\[
P_2\in \Lambda_{E\ell}(\{\widetilde{S_j}\}_{j=0}^L) \subseteq \Lambda(\{\widetilde{S_j}\}_{j=0}^{\left\lceil 2L/3 \right\rceil}).
\]
Here the inclusion follows from $E\ell\le L/3$. By the divisibility property of $P_0$, we have $P_0\mid P_2$. In particular, $P_{j,l}\mid P_2$. We get  
\[
M(P_{j,l})^\ell \stackrel{(*)}{\le} M(\widetilde{P}_2)
\stackrel{(**)}{\le} (E+1)^{1/2},
\]
where $(*)$ follows because the roots $\zeta_a^\ell$ are all distinct and $(**)$ follows from bounding the Mahler measure by the $L^2$ norm of the coefficients.
Therefore
\[
  M(P_{j,l}) \le (E+1)^{1/2\ell} \le (E+1)^{3E/L} \le \exp\left(\frac{30}{C_0(\log\log Q^{rD})^3}\right).
\]

We recall Dobrowolski's theorem~\cite{Dobrowolski79}: there is a universal constant $c>0$ such that, if $P\in\mathbb{Z}[x]$ is a polynomial with
\[
M(P) < \exp\left(c\left(\frac{\log\log\deg P}{\log\deg P}\right)^3\right),
\]
then $P$ is a product of powers of $x$ and cyclotomic polynomials. In our case,
\[
M(P_{j,l}) \le \exp\left(\frac{30}{C_0(\log\deg P_{j,l})^3}\right).
\]
Thus, after increasing $C_0$ if necessary (without dependence on any of the parameters), every such irreducible polynomial $P_{j,l}$ is cyclotomic or equal to $x$. The latter is impossible, because $P_{j,l}(\alpha_{j,l})=0$ while $\alpha_{j,l}\ne 0$, so $P_{j,l}$ must be cyclotomic. Since the degree of the $m$-th cyclotomic polynomial is Euler's totient function and $\varphi(m)\gg \frac{m}{\log\log m}$, we get
\[
\frac{m}{\log\log m}\ll \phi(m)=\deg P\le E,
\]
or equivalently $m\ll E\log\log E$. It follows that every $\alpha_{j,l}$ with $\beta_{j,l}\ne 0$ has multiplicative order smaller than $C_1\log Q^{rD}\log\log\log Q^{rD}$. This proves the proposition.

\label{cor:final-fourier-nu-bound}
Let $\alpha\in\mathcal{V}$ and $\xi\in\mathcal{V}^r\setminus\left\{0\right\}$, and assume that $\alpha$ is generic. Suppose that there exists a pair $(j_0,l_0)$ such that $\xi_{j_0,l_0}\ne 0$ and $\alpha_{j_0,l_0}$ has multiplicative order at least $C_1\log Q^{rD}\log\log\log Q^{rD}$. \[
\left|\widehat{\nu}_{\alpha}^{(d,d+l)}(\xi)\right| < \exp\left(-\frac{1-\| \mu \|_2^2}{8\log(4L)}\right)

for all $d\ge 1$ and $l\ge L$.

We recall from Corollary~\ref{cor:fourier-nu-bound} that
\[
\left|\widehat{\nu}_{\alpha}^{(d,d+l)}(\xi)\right|
\le
\exp\left(-\sum_{a_1,a_2\in\mathbb{Z}}
\left(
\mu(a_1)\mu(a_2)
\sum_{j=d}^{d+l}
\frac{([S_j((a_1-a_2)\xi)]^{\sim})^2}{Q^2}
\right)\right).
\]
We first explain how to remove the shift by $d$. As in the proof of {Lemma~\ref{lem:linear-transform}}, we have
\[
\left\langle \xi,\mathbf{v}_{d+j}(\alpha)\right\rangle = \left\langle \xi,A_{\alpha}^d\mathbf{v}_j(\alpha)\right\rangle = \left\langle (A_{\alpha}^*)^d\xi,\mathbf{v}_j(\alpha)\right\rangle

for every $j\ge 0$ and every $\xi\in\mathcal{V}^r$. Consequently,
\[
S_{d+j}(\xi)=S_j((A_{\alpha}^*)^d\xi)
\]
for every $j\ge 0$. Moreover, for every coordinate $(j,l)$, $\xi_{j,l}\ne 0$ if and only if $ ((A_{\alpha}^*)^d\xi)_{j,l}\ne 0$.

Now fix $a_1\ne a_2$ in the support of $\mu$. Since $\operatorname{supp}(\mu)\subseteq\left(-\frac{1}{2}p_j,\frac{1}{2}p_j\right)$ for every $j$, the integer $a_1-a_2$ is nonzero modulo every prime $p_j$. Hence $\big((a_1-a_2)(A_{\alpha}^*)^d\xi\big)_{j_0,l_0}\ne 0$. Applying Proposition~\ref{prop:Sj-tilde-bound} to $\beta=(a_1-a_2)(A_{\alpha}^*)^d\xi$ and using the assumed lower bound on the multiplicative order of $\alpha_{j_0,l_0}$, we obtain

\]

\]
Since $l\ge L$, this gives
\[
\sum_{j=d}^{d+l} \frac{([S_j((a_1-a_2)\xi)]^{\sim})^2}{Q^2} > \frac{1}{8\log(4L)}.
\]

Returning to the Fourier bound, we get

\left|\widehat{\nu}_{\alpha}^{(d,d+l)}(\xi)\right| < \exp\left(-\frac{1}{8\log(4L)}\sum_{a_1\ne a_2}\mu(a_1)\mu(a_2)\right) = \exp\left(-\frac{1-\| \mu \|_2^2}{8\log(4L)}\right).
\]
This proves the corollary.

<!-- prop-2: body_md -->
\label{prop:full-mixing-bound}
There are absolute constants $C,c>0$ such that the following holds. Let $n\ge 1$ be a positive integer with
\begin{align*}
  n & \ge \frac{C}{1-\| \mu \|_2^2}MT\log Q^{rD}(\log\log Q^{rD})^6, \\
  \log Q^{rD} & \ge \frac{1}{1-\| \mu \|_2^2}.
\end{align*}
Suppose further that $\operatorname{supp}\mu\subseteq\left(-\frac{1}{2}p_j,\frac{1}{2}p_j\right)$ for all $1\le j\le M$. Then
\[
\left|\sum_{\alpha\in A}\nu_{\alpha}^{(n)}(0) - \frac{\left|A\right|}{\left|\mathcal{V}\right|^r}\right| \le \exp\left(-\frac{cn}{\log Q^{rD}(\log\log Q^{rD})^5}\right)
\]
for any nonempty $A\subseteq\mathcal{A}_{\mathcal{V}}$.
<!-- end prop-2 -->
<!-- proof-9: body_md -->
Fix $A\subseteq\mathcal{A}_{\mathcal{V}}$. Since $\nu(0)=\frac{1}{|\mathcal{V}|^r}\sum_{\xi\in\mathcal{V}^r}\widehat{\nu}(\xi)$ and $\widehat{\nu}(0)=1$ for any probability measure $\nu$ on $\mathcal{V}^r$, we have
\begin{equation}\label{eq:kon-mixing-eq1}
\left|\sum_{\alpha\in A}\nu_{\alpha}^{(n)}(0)-\frac{|A|}{|\mathcal{V}|^r}\right| \le \frac{1}{|\mathcal{V}|^r}\sum_{\alpha\in A}\sum_{\xi\in\mathcal{V}^r\setminus\{0\}} \left|\widehat{\nu}_{\alpha}^{(n)}(\xi)\right|.
\end{equation}

We begin by bounding
\[
\sum_{\alpha\in A}\|\nu_{\alpha}^{(d,d+l)}\|_2^2 \le \sum_{\alpha\in\mathcal{V}}\|\nu_{\alpha}^{(d,d+l)}\|_2^2.
\]
Let $\widetilde{R}_{d,d+l}$ be an independent copy of $R_{d,d+l}$. Then
\begin{align*}
  \sum_{\alpha\in\mathcal{V}}\|\nu_{\alpha}^{(d,d+l)}\|_2^2 &= \sum_{\alpha\in\mathcal{V}} \mathbb{P}\left(R_{d,d+l}^{(i)}(\alpha) = \widetilde{R}_{d,d+l}^{(i)}(\alpha)\text{ for every }0\le i\le r-1 \right)\\
  &\le \mathbb{E}\left|\left\{\alpha\in\mathcal{V} \,\ifnum\currentgrouptype=16\middle\fi|\, R_{d,d+l}(\alpha)=\widetilde{R}_{d,d+l}(\alpha)\right\}\right|.
\end{align*}
If $R_{d,d+l}\ne\widetilde{R}_{d,d+l}$, then $R_{d,d+l}-\widetilde{R}_{d,d+l}$ is nonzero in each of the fields $\mathbb{F}_{p_j}$ appearing in $\mathcal{V}$ (since every nonzero coefficient difference has absolute value smaller than $p_j$ for every $j$), so it has at most $l+1$ roots in each of these fields. Hence
\[
\left|\left\{\alpha\in\mathcal{V} \,\ifnum\currentgrouptype=16\middle\fi|\, R_{d,d+l}(\alpha)=\widetilde{R}_{d,d+l}(\alpha)\right\}\right| \le (l+1)^{MT}.
\]
Moreover, $\mathbb{P}\left(R_{d,d+l}=\widetilde{R}_{d,d+l}\right) = \|\mu\|_2^{2(l+1)}$, and thus
\[
\sum_{\alpha\in\mathcal{V}}\|\nu_{\alpha}^{(d,d+l)}\|_2^2 \le (l+1)^{MT}+|\mathcal{V}|\|\mu\|_2^{2(l+1)} = (l+1)^{MT}+|\mathcal{V}|\exp\left(-(l+1)H_2(\mu)\right).
\]
Set $l_0\coloneqq\left\lceil\frac{\log|\mathcal{V}|}{H_2(\mu)}\right\rceil$. Applying the previous estimate to intervals of length $l_0$ and using Parseval we obtain
\[
\frac{1}{|\mathcal{V}|^r}\sum_{\alpha\in\mathcal{V}}\sum_{\xi\in\mathcal{V}^r} \left|\widehat{\nu}_{\alpha}^{(d,d+l_0-1)}(\xi)\right|^2 = \sum_{\alpha\in\mathcal{V}} \|\nu_{\alpha}^{(d,d+l_0-1)}\|_2^2\le 2l_0^{MT}.
\]
Since $\nu_{\alpha}^{(1,2l_0)} = \nu_{\alpha}^{(1,l_0)} * \nu_{\alpha}^{(l_0+1,2l_0)}$, two applications of Cauchy--Schwarz give
\begin{align}
  \frac{1}{|\mathcal{V}|^r}\sum_{\alpha\in\mathcal{V}}\sum_{\xi\in\mathcal{V}^r} \left|\widehat{\nu}_{\alpha}^{(1,2l_0)}(\xi)\right|
  &\le \left(\frac{1}{|\mathcal{V}|^r}\sum_{\alpha\in\mathcal{V}}\sum_{\xi\in\mathcal{V}^r} \left|\widehat{\nu}_{\alpha}^{(1,l_0)}(\xi)\right|^2 \right)^{1/2}\notag\\
  &\qquad\cdot\left(\frac{1}{|\mathcal{V}|^r} \sum_{\alpha\in\mathcal{V}}\sum_{\xi\in\mathcal{V}^r} \left|\widehat{\nu}_{\alpha}^{(l_0+1,2l_0)}(\xi)\right|^2\right)^{1/2}\notag\\
  &\le 2l_0^{MT}.
\label{eq:fourier-bound1}
\end{align}
Put $l_1=n-2l_0-2$, so that $\nu_{\alpha}^{(2l_0+1,n-1)} = \nu_{\alpha}^{(2l_0+1,\,2l_0+1+l_1)}$. After increasing $C$ if necessary, the assumptions imply that
\[
l_1\ge \frac{n}{2} \qquad\text{and}\qquad l_1\ge C_0\log Q^{rD}(\log\log Q^{rD})^4.
\]
Moreover, since $\alpha\in\mathcal{A}_{\mathcal{V}}$, every coordinate of $\alpha$ has multiplicative order at least $n$. The lower bound on $n$ therefore implies that every coordinate has multiplicative order at least $C_1\log Q^{rD}\log\log\log Q^{rD}$. By Corollary \ref{cor:iterated-fourier-nu-bound}, for every $\alpha\in A$ and $\xi\in\mathcal{V}^r\setminus\{0\}$,
\[
\left|\widehat{\nu}_{\alpha}^{(2l_0+1,n-1)}(\xi)\right| \le \exp\left(-\frac{cn}{\log Q^{rD}(\log\log Q^{rD})^5}\right).
\]

The coefficients indexed by $\left\{1,\dots,2l_0\right\}$ and $\left\{2l_0+1,\dots,n-1\right\}$ are independent, while the remaining constant and leading coefficients contribute Fourier factors of absolute value at most $1$. Hence
\[
\left|\widehat{\nu}_{\alpha}^{(n)}(\xi)\right|
\le \left|\widehat{\nu}_{\alpha}^{(1,2l_0)}(\xi)\right| \left|\widehat{\nu}_{\alpha}^{(2l_0+1,n-1)}(\xi)\right|.
\]
Using \eqref{eq:fourier-bound1}, we obtain
\[
\frac{1}{|\mathcal{V}|^r} \sum_{\alpha\in A} \sum_{\xi\in\mathcal{V}^r\setminus\{0\}} \left|\widehat{\nu}_{\alpha}^{(n)}(\xi)\right| \le 2l_0^{MT}\exp\left(-\frac{cn}{\log Q^{rD}(\log\log Q^{rD})^5}\right).
\]

Finally, $l_0 \le 1+(1-\| \mu \|_2^2)^{-1}\log|\mathcal{V}| \le 1+(1-\| \mu \|_2^2)^{-1}\log Q^D$, and hence, using $\log Q^{rD}\ge(1-\| \mu \|_2^2)^{-1}$, we have $\log(2l_0^{MT}) \ll MT\log\log Q^{rD}$. The assumed lower bound on~$n$ allows this factor to be absorbed into the exponential. Thus, after decreasing~$c$ if necessary,
\[

\]
The proposition now follows from \eqref{eq:kon-mixing-eq1}.

<!-- prop-3: body_md -->
\label{prop:num-unique-double}
Let $0<\delta<\frac{1}{3}$, and let $\mathcal{P}$ be a finite nonempty set of primes satisfying \ref{item:P1} and such that $p \ge \frac{n^2}{\delta}$ for all $p\in\mathcal{P}$. Then
\[
\mathbb{P}\left(\mathcal{U}_{\mathcal{P}}(R) < (1-3\delta)S(\mathcal{P})\right) \le \frac{K}{\delta^2 S(\mathcal{P})}
\]
for some universal constant $K>0$.
<!-- end prop-3 -->
<!-- proof-11: body_md -->
For each $p\in\mathcal{P}$, let $N_p$ denote the number of admissible double roots of~$R$ in $\mathbb{F}_p$, and write $N\coloneqq\sum_{p\in\mathcal{P}} N_p$. By {Corollary~\ref{cor:mix-app1}},
\begin{align*}
  \left|\mathbb{E}[N_p] - \frac{|\mathcal{A}_p|}{p^2}\right| &\le p\exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right) \le \exp\left(\frac{C\sqrt{n}}{\log^3 n}\right)\exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right) \\
  &\le \exp\left(-\frac{c\sqrt{n}}{2\log^2 n}\right)
\end{align*}
for large enough $n$. By {Lemma~\ref{lem:count-inadmissible}}, we have $|\mathcal{A}_p|=p-|\mathcal{B}_p|-1\ge p-n^2\ge(1-\delta)p$ for every $p\in\mathcal{P}$. Therefore,
\begin{align*}
  \mathbb{E}[N] &= \sum_{p\in\mathcal{P}}\mathbb{E}[N_p] \ge \sum_{p\in\mathcal{P}}\frac{|\mathcal{A}_p|}{p^2} - \left|\mathcal{P}\right|\exp\left(-\frac{c\sqrt{n}}{2\log^2 n}\right) \\
  &\ge (1-\delta)S(\mathcal{P}) - \exp\left(\frac{C\sqrt{n}}{\log^3 n}\right)\exp\left(-\frac{c\sqrt{n}}{2\log^2 n}\right) = (1-\delta-o(1))S(\mathcal{P}).
\end{align*}

We next bound the variance of $N$. For each $p\in\mathcal{P}$, $N_p(N_p-1)$ counts the number of pairs $(\alpha,\beta)\in\mathcal{A}_p^2$ with $\alpha\ne\beta$ such that $\alpha,\beta$ are double roots of $R$, so {Corollary~\ref{cor:mix-app1}} gives
\[
\mathbb{E}[N_p(N_p-1)] \le \frac{|\mathcal{A}_p|^2}{p^4} + \exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right)
\]
so
\[
\operatorname{Var}(N_p) \le \mathbb{E}[N_p^2] = \mathbb{E}[N_p] + \mathbb{E}[N_p(N_p-1)] \le \frac{1}{p} + \frac{1}{p^2} + 2\exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right).
\]
The same corollary (clause (3)) shows that for distinct $p,p'\in\mathcal{P}$,
\[
\operatorname{Cov}(N_p,N_{p'}) \le 3\exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right).
\]
We thus have
\begin{align*}
  \operatorname{Var}(N) &\le \sum_{p\in\mathcal{P}}\left(\frac{1}{p} + \frac{1}{p^2} + 2\exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right)\right) + 3\left|\mathcal{P}\right|^2\exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right) \\
  &\le (2+o(1))S(\mathcal{P}).
\end{align*}
By Chebyshev's inequality,
\begin{align*}
  \mathbb{P}(N \le (1-2\delta)S(\mathcal{P})) &\le \mathbb{P}(\left|N-\mathbb{E}[N]\right| \ge (\delta-o(1)) S(\mathcal{P})) \\
  &\le \frac{(2+o(1))S(\mathcal{P})}{((\delta-o(1)) S(\mathcal{P}))^2} = \frac{2+o(1)}{\delta^2 S(\mathcal{P})}.
\end{align*}
Next, write $N'=\sum_{p\in\mathcal{P}}N_p(N_p-1)$. By the above,
\[
\mathbb{E}[N'] \le \sum_{p\in\mathcal{P}}\left(\frac{1}{p^2} + \exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right)\right) = O(1),
\]
so by Markov's inequality we have
\[
\mathbb{P}(N' > \delta S(\mathcal{P})) \le \frac{O(1)}{\delta S(\mathcal{P})}.
\]

Finally, we claim that $\mathcal{U}_{\mathcal{P}}(R) \ge N-N'$. Indeed, $N-N' = \sum_{p\in\mathcal{P}}N_p(2-N_p)$; thus every $p$ with a unique admissible double root in $\mathbb{F}_p$ contributes $1$ to this sum, whereas all other summands are nonpositive. Therefore
\[
\mathbb{P}\left(\mathcal{U}_{\mathcal{P}}(R) < (1-3\delta)S(\mathcal{P})\right) \le \mathbb{P}(N < (1-2\delta) S(\mathcal{P})) + \mathbb{P}(N' > \delta S(\mathcal{P})) \le \frac{K}{\delta^2 S(\mathcal{P})}
\]
for some universal constant $K>0$.

<!-- lem-6: body_md -->
\label{lem:double-root-lift}
Let $P\in\mathbb{Z}[x]$ be a monic polynomial. Let $p$ be a prime, and let $a\in\mathbb{F}_p$ be a double root of the reduction $\overline{P}$ of $P$ modulo $p$. If $a$ lifts to a root of $P$ modulo $p^2$, then $P(a_0)\equiv 0\pmod{p^2}$ for every lift $a_0$ of $a$ in $\mathbb{Z}/p^2\mathbb{Z}$. If moreover $a$ is not a triple root of $\overline{P}$, then there is a unique lift $a_0 \in \mathbb{Z}/p^2\mathbb{Z}$ with $P(a_0)\equiv P'(a_0) \equiv 0\pmod{p^2}$.
<!-- end lem-6 -->
<!-- proof-14: body_md -->
For every $b\in\mathbb{F}_p$, we have
\[
P(a+pb)\equiv P(a)+pbP'(a)\pmod{p^2}
\]
If $a$ is a double root of $P$ modulo $p$, we have $P'(a)\equiv 0\pmod p$, and therefore
\[
P(a+pb)\equiv P(a)\pmod{p^2}.
\]
This proves the first assertion.




Suppose now that $a$ is not a triple root of $\overline{P}$ and let $a_0\in \mathbb{Z}/p^2\mathbb{Z}$ be a lift of $a$ with $P(a_0)\equiv 0\pmod{p^2}$. Write $P'(a_0)=pu\pmod{p^2}$ for some $u\in\mathbb{F}_p$. Note that
\[
P'(a_0+pb)\equiv P'(a_0)+pbP''(a_0)\pmod{p^2}.
\]
So $P'(a_0+pb)\equiv 0\pmod{p^2}$ if and only if $u+bP''(a)=0\pmod p$. Since $a$ is not a triple root, $P''(a)\not\equiv 0\pmod p$, so this equation has a unique solution $b\in\mathbb{F}_p$.

<!-- prop-6: body_md -->
\label{prop:Sj-tilde-bound-p2}
Let $p$ be a sufficiently large prime, let $a\in\mathbb{F}_p^\times$, and let $0\ne\xi=(\xi_1,\xi_2)\in\mathbb{Z}/p^2\mathbb{Z}\times\mathbb{F}_p$. Let $L\ge C_0\log p^4(\log\log p^4)^4$, and suppose that

This shows that the multiplicative order of $a$ is at most $C_1\log p^2\log\log\log p^2$, and hence the claimed bound.

\begin{equation}\label{eq:recurrence-p2-first}

\end{equation}
for $m=0,1$. A direct calculation gives

\]

\Theta_d(\xi_1,\xi_2) = \left(a^d\xi_1+pda^{d-1}\xi_2,\,a^d\xi_2\right).

  \sum_{a\in\mathbb{F}_p^\times}\| \rho_a^{(d,d+l)} \|_2^2 &= \sum_{a\in\mathbb{F}_p^\times}\mathbb{P}\left(\begin{array}{c} R_{d,d+l}(a)=\widetilde{R}_{d,d+l}(a)\;(\mathrm{mod}\;p^2),\\
  R_{d,d+l}'(a)=\widetilde{R}_{d,d+l}'(a)\;(\mathrm{mod}\;p)
  \end{array}\right) \\
  &\le \mathbb{E}\left|\left\{a\in\mathbb{F}_p^\times \,\ifnum\currentgrouptype=16\middle\fi|\, R_{d,d+l}(a)=\widetilde{R}_{d,d+l}(a)\;(\mathrm{mod}\;p)\right\}\right|.
\end{align*}
If $R_{d,d+l}\ne\widetilde{R}_{d,d+l}$, then their difference remains nonzero modulo $p$, since $\operatorname{supp}(\mu)\subseteq\left(-\frac{p}{2},\frac{p}{2}\right)$. After dividing by $x^d$, this difference has degree at most $l$, and therefore has at most $l$ roots in $\mathbb{F}_p^\times$. Since $\mathbb{P}\left(R_{d,d+l}=\widetilde{R}_{d,d+l}\right) = \| \mu \|_2^{2(l+1)}$, we obtain
\begin{equation}\label{eq:L2-p2}
\sum_{a\in\mathbb{F}_p^\times}\| \rho_a^{(d,d+l)} \|_2^2 \le l+p\| \mu \|_2^{2(l+1)}.
\end{equation}

Set $l_0\coloneqq \left\lceil \frac{\log p}{H_2(\mu)} \right\rceil$. By \eqref{eq:L2-p2} and Plancherel's identity,
\[
\frac{1}{p^3}\sum_{a\in\mathbb{F}_p^\times}\sum_{\xi\in\mathcal{W}_p} \left|\widehat{\rho}_a^{(d,d+l_0-1)}(\xi)\right|^2 = \sum_{a\in\mathbb{F}_p^\times}\| \rho_a^{(d,d+l_0-1)} \|_2^2 \le 2l_0.
\]
Since $\rho_a^{(1,2l_0)} = \rho_a^{(1,l_0)}*\rho_a^{(l_0+1,2l_0)}$, Cauchy--Schwarz gives
\begin{equation}\label{eq:L1-p2}
\frac{1}{p^3}\sum_{a\in\mathbb{F}_p^\times}\sum_{\xi\in\mathcal{W}_p} \left|\widehat{\rho}_a^{(1,2l_0)}(\xi)\right| \le 2l_0.
\end{equation}
If $a\in\mathcal{A}_p$, then the multiplicative order of $a$ is at least $n$, and by the assumptions on $p$ and $n$ we have $n \ge C_1\log p^4\log\log\log p^4$ and that $n-2l_0-2\ge n/2$. We may therefore apply {Corollary~\ref{cor:fourier-p2}} to obtain
\[
\left|\widehat{\rho}_a^{(2l_0+1,n-1)}(\xi)\right| \le \exp\left(-\frac{cn}{\log p^4(\log\log p^4)^5}\right)
\]
for every $a\in A$ and every $0\ne\xi\in\mathcal{W}_p$.

The constant and leading coefficients contribute Fourier factors of absolute value at most $1$. Hence, by \eqref{eq:L1-p2},
\[
\frac{1}{p^3}\sum_{a\in A}\sum_{\xi\in\mathcal{W}_p\setminus\left\{0\right\}} \left|\widehat{\rho}_a^{(n)}(\xi)\right| \le 2l_0\exp\left(-\frac{cn}{\log p^4(\log\log p^4)^5}\right) \le \exp\left(-\frac{c\sqrt{n}}{\log^2 n}\right)
\]
where the last inequality follows from the assumed upper bound on $p$, after decreasing $c$ if necessary. The proposition now follows by Fourier inversion on $\mathcal{W}_p$.

<!-- defn-3: body_md -->
\label{def:CCCP} Let $G$ be an Abelian group. A symmetric coset progression is a set of the form $H+Q$ where $H$ is a subgroup of $G$ and $Q=\phi(\Lambda\cap \mathbb{Z}^r)$, where $\Lambda$ is a symmetric box in $\mathbb{R}^r$ and $\phi$ is a group homomorphism from $\mathbb{Z}^r$ to $G$. \label{lem:finite-group-ILO}
For every $A>0$ and $0<\varepsilon<1$ there are constants $r,B,m_0=O_{A,\varepsilon,\mu}(1)$ such that the following holds for every $m\ge m_0$. Let $\omega_1,\dots,\omega_m$ be i.i.d.\ $\mu$-random variables. Then for any prime $p$ outside a finite set depending only on~$\mu$, for any $k\ge 1$, and for any $\alpha_1,\dots,\alpha_m\in\mathbb{F}_{p^k}$, if
\[
\sup_{\beta\in\mathbb{F}_{p^k}}\bigg|\mathbb{P}\bigg(\sum_{j=1}^m\omega_j \alpha_j = \beta\bigg) - \frac{1}{p^k}\bigg| \ge m^{-A},
\]
then there is a proper symmetric coset progression $H+Q\subseteq\mathbb{F}_{p^k}$ of rank at most $r$ and cardinality at most $m^B$ which contains all but at most $\varepsilon m$ of the $\alpha_j$, counted with multiplicity.

<!-- proof-20: body_md -->
We may assume without loss of regularity that $\operatorname{supp}(\mu)$ is not contained in a coset of a proper subgroup of $\mathbb{Z}$ (indeed, the set of excluded primes is exactly the divisors of the maximal $d$ such that $\operatorname{supp}(\mu)\subset a+d\mathbb{Z}$). This gives that $|\widehat{\mu}(\xi)|$ takes the value 1 only at $\xi=0$ and hence $|\widehat{\mu}(\xi)|\le 1-c\sin^2(\xi/2)$ for some $c$ (we normalise the Fourier transform to have $\xi\in[0,1]$).


Fix $0<\lambda_{\mu}\le\frac{1}{4}$ such that $4\pi^2\lambda_{\mu}\le c_{\mu}$, and let $\eta_1,\dots,\eta_m$ be independent lazy Bernoulli random variables with parameter $\lambda_{\mu}$, so that $\mathbb{P}(\eta_j=0)=1-\lambda_{\mu}$ and $\mathbb{P}(\eta_j=1)=\mathbb{P}(\eta_j=-1)=\frac{\lambda_{\mu}}{2}$. Then  
\[
  \widehat{\eta_1}(\xi) = 1-\lambda_{\mu}+\lambda_{\mu}\cos(\xi) = 1-2\lambda_{\mu}\sin^2(\xi/2)
  \ge |\widehat{\mu}(\xi)|.
\]  
Writing $\operatorname{tr}$ for the trace from $\mathbb{F}_{p^k}$ to $\mathbb{F}_p$, Fourier inversion gives
\begin{align*}
  \lefteqn{\sup_{\beta\in\mathbb{F}_{p^k}} \bigg|\mathbb{P}\bigg(\sum_{j=1}^m\omega_j\alpha_j=\beta\bigg) - \frac{1}{p^k}\bigg| \le \frac{1}{p^k} \sum_{\xi\in\mathbb{F}_{p^k}^{\times}} \prod_{j=1}^m\left|\widehat{\mu}\left(\frac{\operatorname{tr}(\xi\alpha_j)}{p}\right)\right|}\qquad & \\
  &\le \frac{1}{p^k} \sum_{\xi\in\mathbb{F}_{p^k}^{\times}} \prod_{j=1}^m\widehat{\eta_1}\left(\frac{\operatorname{tr}(\xi\alpha_j)}{p}\right)
  = \mathbb{P}\bigg(\sum_{j=1}^m\eta_j\alpha_j=0\bigg) - \frac{1}{p^k} \\
  &\le \sup_{\beta\in\mathbb{F}_{p^k}}\bigg|\mathbb{P}\bigg(\sum_{j=1}^m\eta_j\alpha_j=\beta\bigg) - \frac{1}{p^k}\bigg|.
\end{align*}
This shows that
\[
\sup_{\beta\in\mathbb{F}_{p^k}}\bigg|\mathbb{P}\bigg(\sum_{j=1}^m\eta_j\alpha_j=\beta\bigg) - \frac{1}{p^k}\bigg| \ge m^{-A}.
\]

For all sufficiently large $m$, the fixed parameter $\lambda_{\mu}$ satisfies the assumptions of \cite[Theorem~1.6(i)]{KoenigNguyenPan24}, with $\varepsilon_0=\frac{1}{4}$. Taking the number of exceptional elements to be $\left\lfloor \varepsilon m \right\rfloor$, that theorem gives a proper symmetric coset progression of bounded rank containing all but at most $\varepsilon m$ of the~$\alpha_j$. Its cardinality is bounded by $m^B$ for some $B=O_{A,\varepsilon,\mu}(1)$.

<!-- prop-10: body_md -->
\label{prop:inadmissible-root}
For any $A,\lambda>0$ there are constants $C,d_0,m_0=O_{A,\lambda,\mu}(1)$ such that the following holds. For every $m\ge m_0$, there are at most $m^C$ primes $p$ for which there exist $k\ge 1$ and $\alpha\in\mathbb{F}_{p^k}^{\times}$ of multiplicative order $d$ with $d_0\le d\le \lambda m$, such that
\[
\sup_{\beta\in\mathbb{F}_{p^k}}\left|\mathbb{P}(R_{1,m}(\alpha)=\beta)-\frac{1}{p^k}\right| \ge \frac{1}{m^A}.
\]
<!-- end prop-10 -->
<!-- proof-21: body_md -->
We begin by applying the inverse Littlewood--Offord {Lemma~\ref{lem:finite-group-ILO}} with $\varepsilon=\frac{1}{2}$ to the elements $\alpha,\dots,\alpha^m$. After discarding a finite set of primes depending only on $\mu$, there is a proper symmetric coset progression $H+Q\subseteq\mathbb{F}_{p^k}$ of rank at most $r=O_{A,\mu}(1)$ and cardinality at most $m^B$, where $B=O_{A,\mu}(1)$, such that $\alpha^j\in H+Q$ for at least $\frac{1}{2}m$ values of $1\le j\le m$.

We discard all primes which are at most $m^B$. For any remaining prime we have $|H|\le |H+Q|\le m^B<p$. Since every nontrivial additive subgroup of $\mathbb{F}_{p^k}$ has cardinality at least $p$, it follows that $H=\left\{0\right\}$. Therefore $\alpha^j\in Q$ for at least $\frac{1}{2}m$ values of $1\le j\le m$. Each power of $\alpha$ can appear at most $\left\lceil m/d \right\rceil$ times among $\alpha,\dots,\alpha^m$. In particular, $Q$ contains at least $\frac{m}{2\left\lceil m/d \right\rceil} \ge \frac{1}{4}\min\left\{d,m\right\}$ distinct powers of $\alpha$.

Let $M=\prod_{\ell\le r+1}\ell=O_{A,\mu}(1)$ be the product of all primes $\ell\le r+1$. We claim that we can find $r+1$ distinct powers $\alpha^{t_1},\dots,\alpha^{t_{r+1}}\in Q$ such that $\alpha^{M(t_j-t_{j'})}\ne 1$ for every $j\ne j'$. Indeed, define an equivalence relation on the distinct powers of~$\alpha$ in $Q$ by $\alpha^{t_j}\sim\alpha^{t_{j'}}$ if $\alpha^{M(t_{j'}-t_j)}=1$. Every equivalence class has cardinality at most $\gcd(M,d)\le M$. Hence there are at least $\frac{1}{4M}\min\left\{d,m\right\}$ equivalence classes. We may therefore choose $d_0\ge 4M(r+1)$ so that, for all sufficiently large $m$, there are at least $r+1$ such classes. Choosing one power from each of $r+1$ distinct classes proves the claim. We take $0\le t_j<d$.

Write $Q=\left\{m_1g_1+\cdots+m_sg_s\,\ifnum\currentgrouptype=16\middle\fi|\, |m_i|\le N_i\right\}$, where $s\le r$. Since $Q$ is proper, $\prod_{i=1}^s(2N_i+1)=|Q|\le m^B$, and hence $N_i\le m^B$ for every $i$. Write $\alpha^{t_j}=m_{j,1}g_1+\cdots+m_{j,s}g_s$. The corresponding $r+1$ vectors in $\mathbb{Z}^s$ are linearly dependent, so there are primitive integers $c_1,\dots,c_{r+1}$, not all zero, such that $\sum_{j=1}^{r+1}c_j\alpha^{t_j}=0$ in $\mathbb{F}_{p^k}$. By taking a minimal linear dependence and expressing its coefficients as minors of the coordinate matrix, we have $|c_j|\le s!m^{Bs}$, so we may moreover assume that $|c_j|\le m^{B'}$ for some $B'=O_{A,\mu}(1)$. After discarding the terms for which $c_j=0$, define $F(X)=\sum_jc_jX^{t_j}$. Since $\alpha$ has multiplicative order $d$ and $p\nmid d$, we have $\Phi_d(\alpha)=0$. Thus $F$ and $\Phi_d$ have a common root over $\mathbb{F}_{p^k}$, and hence $p\mid\operatorname{Res}(\Phi_d,F)$.

We claim that $\operatorname{Res}(\Phi_d,F)\ne 0$. Otherwise, $F(\zeta)=0$ for some primitive $d$-th root of unity $\zeta\in\mathbb{C}$. Take a minimal vanishing subsum of $F(\zeta)=\sum_jc_j\zeta^{t_j}=0$. This subsum has at most $r+1$ terms. By a result of Mann \cite{Mann65} (see also the improvement of Conway--Jones \cite{ConwayJones76}), the order of the quotient of any two roots occurring in this subsum divides $M$. In other words, for some $j\ne j'$, $d\, |\, M(t_j-t_{j'})$. Since  
$\alpha$ has order $d$, this implies $\alpha^{M(t_j-t_{j'})}=1$, contrary to the choice of $t_1,\dots,t_{r+1}$. Therefore $\operatorname{Res}(\Phi_d,F)\ne 0$.

It remains to count the possible prime divisors of these resultants. Since $\Phi_d$ is monic,
\[
\left|\operatorname{Res}(\Phi_d,F)\right| = \prod_{\substack{\zeta\in\mathbb{C}\\ \operatorname{ord}(\zeta)=d}}|F(\zeta)| \le \bigg(\sum_j|c_j|\bigg)^{\varphi(d)} \le \left((r+1)m^{B'}\right)^{\varphi(d)}.
\]
Since $d\le \lambda m$, it follows that $\left|\operatorname{Res}(\Phi_d,F)\right|\le m^{Cm}$ for some $C=O_{A,\lambda,\mu}(1)$. Thus each nonzero resultant has at most $Cm\log m$ distinct prime divisors.

Finally, there are only polynomially many possibilities for $d$, for the exponents $t_1,\dots,t_{r+1}$, and for the coefficients $c_1,\dots,c_{r+1}$: their number is at most $m^{O_{A,\mu}(1)}$. Summing the number of prime divisors over all these possibilities, and adding the primes $p\le m^B$ and the finite set of primes excluded at the beginning, gives at most~$m^C$ possible primes, after increasing $C$ and $m_0$ if necessary.

<!-- lem-9: body_md -->
\label{lem:affine-subspace}
Let $p$ be a prime, let $2\le k\le \frac{m-2}{2}$ be a positive integer, let $a\in\mathbb{F}_p^{\times}$, and let~$\alpha\in\mathbb{F}^{\circ}_{p^k}$. Assume that $\operatorname{supp}(\mu)\subseteq(-p/2,p/2)$. Then, for every affine $\mathbb{F}_p$-subspace $U\subseteq\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2$ of dimension $d\le 2k+2$,
\[

Consider the linear map $\Phi\colon\mathbb{F}_p^{2k+2}\to\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2$ given by
\[
\Phi(c_1,\dots,c_{2k+2}) = \left(\sum_{j=1}^{2k+2} c_j a^j, \sum_{j=1}^{2k+2} jc_j a^{j-1}, \sum_{j=1}^{2k+2} c_j\alpha^j, \sum_{j=1}^{2k+2} jc_j\alpha^{j-1}\right).
\]
We claim that $\Phi$ is injective. Indeed, if $\Phi(c_1,\dots,c_{2k+2})=0$, then $P(a)=P'(a)=P(\alpha)=P'(\alpha)=0$ for $P(x)=\sum_{j=1}^{2k+2} c_jx^j$. Hence $(x-a)^2f_{\alpha}^2\mid P$, where $f_{\alpha}$ is the minimal polynomial of $\alpha$ over $\mathbb{F}_p$. But $\deg f_{\alpha}=k$ and $\deg P\le 2k+2$, so this forces $P=0$ or $P=c(x-a)^2f_{\alpha}^2$. The latter is impossible since $P(0)=0$, so we must have $P=0$, proving that~$\Phi$ is injective.

Condition on the values of $\omega_{2k+3},\dots,\omega_m$, and put
\[
y=\left(\sum_{j=2k+3}^m\omega_j a^j,\sum_{j=2k+3}^m j\omega_j a^{j-1},\sum_{j=2k+3}^m\omega_j \alpha^j,\sum_{j=2k+3}^m j\omega_j\alpha^{j-1}\right).
\]
We need to bound the probability that $\Phi(\omega_1,\dots,\omega_{2k+2})\in U-y$. Let $S\coloneqq\Phi^{-1}(U-y)$. If $S$ is empty there is nothing to prove. Otherwise, $S$ is an affine subspace of $\mathbb{F}_p^{2k+2}$ of some dimension $d_0\le d$.

Choose a set $J\subseteq\left\{1,\dots,2k+2\right\}$ of size~$d_0$ such that the coordinate projection $\pi_J\colon\mathbb{F}_p^{2k+2}\to\mathbb{F}_p^J$ is one-to-one on~$S$. For every $u\in\pi_J(S)$, let $c(u)$ denote the unique element of $S$ such that $\pi_J(c(u))=u$. It follows that
\begin{align*}
  \mathbb{P}\bigl((\omega_1,\dots,\omega_{2k+2})\in S\bigr) &= \sum_{u\in\pi_J(S)} \prod_{j=1}^{2k+2}\mu(c_j(u)) \le \| \mu \|_\infty^{2k+2-d_0}\sum_{u\in\pi_J(S)}\prod_{j\in J}\mu(u_j) \\
  &\le \| \mu \|_\infty^{2k+2-d_0} \le \| \mu \|_\infty^{2k+2-d}.
\end{align*}
The bound is uniform in the conditioned values of $\omega_{2k+3},\dots,\omega_m$. Averaging over these coefficients proves the lemma.

<!-- prop-11: body_md -->
\label{prop:dichotomy}
Let $m\ge 1$ and $2\le k\le \frac{m-2}{2}$ be positive integers, let $a\in\mathbb{F}_p^{\times}$, and let $\alpha\in\mathbb{F}^{\circ}_{p^k}$. Suppose that $\operatorname{supp}(\mu)\subseteq(-p/2,p/2)$.

Then, for any $K>2$,
\[
\text{either} \qquad \| \nu_{a,\alpha}^{(1,2m)} \|_2 < \frac{1}{K}\| \nu_{a,\alpha}^{(1,m)} \|_2 \qquad \text{or} \qquad \| \nu_{a,\alpha}^{(1,m)} \|_2^2 \le \frac{1}{p^{2k + 2 - D_K}}
\]
for a nonnegative real number $D_K=O_{\mu}(\log^{3+o(1)}K)$.
<!-- end prop-11 -->
<!-- proof-25: body_md -->
For convenience, write $\nu=\nu_{a,\alpha}^{(1,m)}$ and $A=A_{(a,\alpha)}$ of {Lemma~\ref{lem:linear-transform}}, so that $\nu_{a,\alpha}^{(m+1,2m)} = (A^m)_*\nu$. Then $\| \nu_{a,\alpha}^{(m+1,2m)} \|_2 = \| \nu \|_2$, and $\nu_{a,\alpha}^{(1,2m)} = \nu*((A^m)_*\nu)$. Suppose that
\[
\| \nu_{a,\alpha}^{(1,2m)} \|_2 \ge \frac{1}{K}\| \nu \|_2,
\]
i.e.,
\[
\| \nu*((A^m)_*\nu) \|_2 \ge \frac{1}{K}\| \nu \|_2.
\]
We apply Varj\'{u}'s version of the Balog--Szemer\'{e}di--Gowers theorem (\cite[Lemma 15]{Varju12}) with $\nu_1=\nu$ and $\nu_2=\nu_{a,\alpha}^{(m+1,2m)}$. We get a subset $H\subseteq\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2$ such that
\[
\frac{1}{CK^C\| \nu \|_2^2} \le \left|H\right| \le \frac{CK^C}{\| \nu \|_2^2},
\]
\begin{equation}\label{eq:HHH}
\left|H+H+H\right| \le K^C\left|H\right|,
\end{equation}
and
\[
\min_{y\in H}(\check{\nu}*\nu)(y) \ge \frac{1}{CK^C\left|H\right|}
\]
where $\check{\nu}(y)=\nu(-y)$. Since
\[
\sum_{y\in H}(\check{\nu}*\nu)(y) = \sum_{w\in\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2}\nu(w)\nu(H+w),
\]
it follows that
\[
\nu(H+w) \ge \frac{1}{CK^C}
\]
for some $w\in\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2$.

We next apply Sanders' quantitative version of Freiman's theorem \cite[Theorem~1.4]{Sanders13}. See also \cite{Raghavan25} for stronger results (the difference does not matter for our application). We apply it to the set $H$, and get from \eqref{eq:HHH} that there exist a set $X\subseteq\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2$, a centered convex progression $P\subseteq\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2$ (recall Definition \ref{def:CCCP}), and an $\mathbb{F}_p$-subspace $V\le\mathbb{F}_p^2\times\mathbb{F}_{p^k}^2$, with
\begin{equation}\label{eq:dichotomy-dimP}
\dim P \le \log^{3+o(1)}K,
\end{equation}
\[
\left|P+V\right| \le \exp(\log^{3+o(1)}K)\left|H\right|,
\]
and
\[
\left|X\right| \le \exp(\log^{3+o(1)}K),
\]
such that
\[
H \subseteq X+P+V.
\]
We claim that
\begin{equation}\label{eq:dichotomy-dimV}
\dim V \ge 2k+2 - D_K\qquad\text{where}\qquad D_K=O_{\mu}(\log^{3+o(1)}K).
\end{equation}
Indeed, if $\dim P + \dim V > 2k+2$, this follows from \eqref{eq:dichotomy-dimP} by taking $D_K=\dim P$. Otherwise, assume $\dim P + \dim V \le 2k+2$. Then $P+V$ (and thus also $P+V+x+w$ for any $x\in X$) is contained in an affine subspace of dimension at most $\dim P + \dim V \le 2k+2$, so by {Lemma~\ref{lem:affine-subspace}}
\[
\nu(P+V+x+w) \le \| \mu \|_{\infty}^{2k+2 - \dim P - \dim V}.
\]
for any $x\in X$. On the other hand, by the lower bound on $\nu(H+w)$, there exists an element $x_0\in X$ such that
\[
\nu(P+V+x_0+w) \ge \frac{1}{CK^C\left|X\right|}.
\]
Comparing the two bounds for $\nu(P+V+x_0+w)$ and taking logarithms, we obtain
\[
\dim V \ge 2k+2 - \dim P - \frac{\log(CK^C|X|)}{\log\left(\| \mu \|_{\infty}^{-1}\right)} \ge 2k+2 - D_K
\]
for $D_K \coloneqq \dim P + \frac{\log(CK^C|X|)}{\log\left(\| \mu \|_{\infty}^{-1}\right)}\le C_{\mu}\log^{3+o(1)}K$, where $C_{\mu}>0$ depends only on $\| \mu \|_{\infty}$, proving \eqref{eq:dichotomy-dimV}.

Finally,
\[
p^{2k+2-D_K} \le p^{\dim V} = \left|V\right| \le \left|P+V\right| \le \exp\left(\log^{3+o(1)}K\right)\left|H\right| \le \frac{K_0}{\| \nu \|_2^2}
\]
for
\[
K_0 \coloneqq CK^C\exp\left(\log^{3+o(1)}K\right) = \exp(\log^{3+o(1)}K).
\]
Therefore,
\[
\| \nu \|_2^2 \le \frac{K_0}{p^{2k+2 - D_K}} \le \frac{1}{p^{2k+2 - D_K - \log K_0 / \log p}},
\]
and the proposition follows by replacing $D_K$ with $D_K + \frac{\log K_0}{\log p}$.

\label{prop:no-double-roots-big-ext}

For convenience, we write $k_0\coloneqq\left\lceil \log^4 n \right\rceil$.

Fix a prime $p\in\mathcal{P}$ and an integer $k_0\le k\le k_{\mu}(p)$. We note that
\[
2k+2 \le \frac{2\log\| \mu \|_2^{-2m}}{\log p} = \frac{2H_2(\mu)}{\log p}m \le m
\]
by our assumptions. For any $a\in\mathcal{A}_p$ and $\alpha\in\mathbb{F}^{\circ}_{p^k}$,
\[
\nu_{a,\alpha}^{(n)} = \nu_{a,\alpha}^{(1,2m)}*\nu_{a,\alpha}^{(2m+1,3m)}*\lambda
\]
(for $\lambda$ which corresponds to all coefficients outside of $[1,3m]$), and thus by Young's inequality
\[
\mathbb{P}(R(a)=R'(a)=R(\alpha)=R'(\alpha)=0) \le \| \nu_{a,\alpha}^{(n)} \|_{\infty} \le \| \nu_{a,\alpha}^{(1,2m)} \|_2 \| \nu_{a,\alpha}^{(2m+1,3m)} \|_2.
\]
Recalling that
\[
\| \nu_{a,\alpha}^{(2m+1,3m)} \|_2 = \| \nu_{a,\alpha}^{(1,m)} \|_2
\]
we have
\[
\mathbb{P}(R(a)=R'(a)=R(\alpha)=R'(\alpha)=0) \le \| \nu_{a,\alpha}^{(1,m)} \|_2 \| \nu_{a,\alpha}^{(1,2m)} \|_2,
\]
and thus
\begin{equation}\label{eq:double-root-pk}
  \mathbb{P}(B_p^{\mathrm{ad}}\cap B_{p^k}) \le \sum_{a\in\mathcal{A}_p}\sum_{\alpha\in\mathbb{F}^{\circ}_{p^k}}\| \nu_{a,\alpha}^{(1,m)} \|_2 \| \nu_{a,\alpha}^{(1,2m)} \|_2.
\end{equation}

We set $K=n^3$, and assume that $n$ is sufficiently large so that, for $D_K$ of {Proposition~\ref{prop:dichotomy}}, we have
\[
D_K \le C_{\mu}\log^{3+o(1)}K \le \log^4 n - 1.
\]
By {Proposition~\ref{prop:dichotomy}}, for any $a\in\mathcal{A}_p$ and $\alpha\in\mathbb{F}^{\circ}_{p^k}$,
\[
\text{either} \qquad \| \nu_{a,\alpha}^{(1,2m)} \|_2 < \frac{1}{n^3}\| \nu_{a,\alpha}^{(1,m)} \|_2 \qquad \text{or} \qquad \| \nu_{a,\alpha}^{(1,m)} \|_2^2 \le \frac{1}{p^{2k+3 - \log^4 n}}.
\]

We also recall that
\[
\| \nu_{a,\alpha}^{(1,2m)} \|_2 \le \| \nu_{a,\alpha}^{(1,m)} \|_2.
\]
It therefore follows that
\[
\| \nu_{a,\alpha}^{(1,m)} \|_2\| \nu_{a,\alpha}^{(1,2m)} \|_2 \le \frac{1}{n^3}\| \nu_{a,\alpha}^{(1,m)} \|_2^2 + \frac{1}{p^{2k + 3 - \log^4 n}}
\]
for any $a\in\mathcal{A}_p$ and any $\alpha\in\mathbb{F}^{\circ}_{p^k}$. Substituting back in \eqref{eq:double-root-pk}, we have
\begin{align*}
  \mathbb{P}(B_p^{\mathrm{ad}}\cap B_{p^k}) &\le \sum_{a\in\mathcal{A}_p}\sum_{\alpha\in\mathbb{F}^{\circ}_{p^k}} \left(\frac{1}{n^3}\| \nu_{a,\alpha}^{(1,m)} \|_2^2 + \frac{1}{p^{2k+3 - \log^4 n}}\right) \\
  &\le \frac{1}{p^{k+2 - \log^4 n}} + \frac{1}{n^3}\sum_{a\in\mathcal{A}_p}\sum_{\alpha\in\mathbb{F}^{\circ}_{p^k}} \| \nu_{a,\alpha}^{(1,m)} \|_2^2
\end{align*}

Let $\widetilde{R}$ be an independent copy of $R$, and write $S=R_{1,m}-\widetilde{R}_{1,m}$. This is a random polynomial with coefficient law $\mu_S=\mu*\check{\mu}$, where $\check{\mu}(t)=\mu(-t)$. Therefore $\| \mu_S \|_2\le\| \mu \|_2$, and $\operatorname{supp}(\mu_S)\subseteq[-2H_{\mu},2H_{\mu}]$. Then
\begin{align*}
  \sum_{a\in\mathcal{A}_p}\sum_{\alpha\in\mathbb{F}^{\circ}_{p^k}} \| \nu_{a,\alpha}^{(1,m)} \|_2^2 &= \sum_{a\in\mathcal{A}_p}\sum_{\alpha\in\mathbb{F}^{\circ}_{p^k}} \mathbb{P}\left(\begin{array}{c}R_{1,m}(a)=\widetilde{R}_{1,m}(a), R_{1,m}'(a)=\widetilde{R}_{1,m}'(a), \\ R_{1,m}(\alpha)=\widetilde{R}_{1,m}(\alpha), R_{1,m}'(\alpha)=\widetilde{R}_{1,m}'(\alpha)\end{array}\right) \\
  &= \sum_{a\in\mathcal{A}_p}\sum_{\alpha\in\mathbb{F}^{\circ}_{p^k}} \mathbb{P}\left(\begin{array}{c} S(a)=S'(a)=0 \\ S(\alpha)=S'(\alpha)=0\end{array}\right) = \mathbb{E}[\mathcal{D}_{p,p^k}(S)],
\end{align*}
and thus
\[
\mathbb{P}(B_p^{\mathrm{ad}}\cap B_{p^k}) \le \frac{1}{n^3}\mathbb{E}[\mathcal{D}_{p,p^k}(S)] + \frac{1}{p^{k+2 - \log^4 n}}.
\]

Summing over $k$, we have
\[
\sum_{k=k_0}^{k_{\mu}(p)}\mathbb{P}(B_p^{\mathrm{ad}}\cap B_{p^k}) \le \frac{1}{n^3}\sum_{k=k_0}^{k_{\mu}(p)}\mathbb{E}[\mathcal{D}_{p,p^k}(S)] + \sum_{k=k_0}^{k_{\mu}(p)}\frac{1}{p^{k+2 - \log^4 n}}.
\]
The second term is simple to estimate:
\[
\sum_{k=k_0}^{k_{\mu}(p)}\frac{1}{p^{k+2 - \log^4 n}} \le \frac{2}{p^{k_0+2 - \log^4 n}} \le \frac{2}{p^2}.
\]
To estimate the remaining sum, we replace the sum over double roots in $\mathbb{F}_{p^k}$ with a crude bound of $n$, leaving only the randoness in the roots $a\in\mathbb{F}_p$. Formally, if $S\ne 0$, then
\begin{equation}\label{eq:Dpp}
\sum_{k=k_0}^{k_{\mu}(p)}\mathcal{D}_{p,p^k}(S) \le n\sum_{a\in\mathcal{A}_p}\mathbbm{1}_{\left\{S(a)=S'(a)=0\right\}},
\end{equation}
On the other hand, if $S=0$, which happens with probability $\mathbb{P}(S=0)=\| \mu \|_2^{2m}$, then $\mathcal{D}_{p,p^k}(S)\le p^{k+1}$. Therefore, by {Corollary~\ref{cor:mix-app1}} (applied to $S$),
\begin{align*}
  \sum_{k=k_0}^{k_{\mu}(p)}\mathbb{E}[\mathcal{D}_{p,p^k}(S)] &= \mathbb{E}\left[\sum_{k=k_0}^{k_{\mu}(p)}\mathcal{D}_{p,p^k}(S)\right] \\
  &\le \mathbb{E}\left[\sum_{k=k_0}^{k_{\mu}(p)}\mathcal{D}_{p,p^k}(S)\mathbbm{1}_{\left\{S\ne 0\right\}}\right] + \sum_{k=k_0}^{k_{\mu}(p)}p^{k+1}\mathbb{P}(S=0) \\
  &\stackrel{\mathclap{(*)}}{\le} n\mathbb{E}\left[\sum_{a\in\mathcal{A}_p}\mathbbm{1}_{\left\{S(a)=S'(a)=0\right\}}\right] + 2p^{k_{\mu}(p)+1}\| \mu \|_2^{2m} \\
  &\stackrel{\mathclap{(**)}}{\le} n\sum_{a\in\mathcal{A}_p}\mathbb{P}(S(a)=S'(a)=0) + \frac{2}{p} \\
  &\stackrel{\mathclap{(\dagger)}}{\le} n\sum_{a\in\mathcal{A}_p}\left(\frac{1}{p^2} + \exp\left(-c\frac{(1-\| \mu \|_2^2)\sqrt{n}}{\log^2 n}\right)\right) + \frac{2}{p} \\
  &\le \frac{n}{p} + np\exp\left(-c\frac{(1-\| \mu \|_2^2)\sqrt{n}}{\log^2 n}\right) + \frac{2}{p},
\end{align*}
where in $(*)$ we used \eqref{eq:Dpp} for the left term, in $(**)$ we used \eqref{eq:def kmu} for the right term, and in $(\dagger)$ we used Corollary \ref{cor:mix-app1} (applied to $S$).
Consequently, for sufficiently large $n$
\begin{align*}
  \sum_{k=k_0}^{k_{\mu}(p)}\mathbb{P}(B_p^{\mathrm{ad}}\cap B_{p^k}) &\le \frac{1}{n^2p} + \frac{p}{n^2}\exp\left(-c\frac{(1-\| \mu \|_2^2)\sqrt{n}}{\log^2 n}\right) + \frac{2}{n^3p} + \frac{2}{p^2} \\
  &\le \frac{2}{n^2p} + \frac{2}{p^2} + \exp\left(-c\frac{(1-\| \mu \|_2^2)\sqrt{n}}{2\log^2 n}\right).
\end{align*}
Finally, summing over $p$,
\begin{align*}
  \sum_{p\in\mathcal{P}}\sum_{k=k_0}^{k_{\mu}(p)}\mathbb{P}(B_p^{\mathrm{ad}}\cap B_{p^k}) &\le \frac{2}{n^2}S(\mathcal{P}) + 2S_2(\mathcal{P}) + \left|\mathcal{P}\right|\exp\left(-c\frac{(1-\| \mu \|_2^2)\sqrt{n}}{2\log^2 n}\right),
\end{align*}
and the last term is $o\left(\frac{1}{\log n}\right)$ since $\left|\mathcal{P}\right|\le\exp\left(\frac{C\sqrt{n}}{\log^3 n}\right)$.

\label{prop:p^2-divides-disc}
Let $P\in\mathbb{Z}[x]$ be a monic polynomial with $\operatorname{disc}(P)\ne 0$, and let $p$ be an odd prime. We write $\overline{P}$ for the reduction of $P$ modulo $p$. Suppose that $a\in\mathbb{F}_p$ is a double root of $\overline{P}$. Then $p^2\mid\operatorname{disc}(P)$ if and only if at least one of the following holds:
\begin{enumerate}
  \item\label{item:p^2-i} $a$ is a triple root of $\overline{P}$;
  \item\label{item:p^2-ii} there is a monic irreducible polynomial $Q\ne x-a$ such that $Q^2\mid\overline{P}$, in which case $v_p(\operatorname{disc}(P))\ge 1+\deg Q$;
  \item\label{item:p^2-iii} $a$ lifts to a double root of $P$ modulo $p^2$.
\end{enumerate}

<!-- lem-10: body_md -->
Let $P\in\mathbb{Z}[x]$ be a monic polynomial with $\operatorname{disc}(P)\ne 0$, and let $p$ be a prime. Consider the ideal $(P,P')$ generated by $P$ and $P'$ in $\mathbb{Z}_p[x]$. Then \[v_p(\operatorname{disc}(P)) = v_p(|\mathbb{Z}_p[x]/(P,P')|).\]
<!-- end lem-10 -->
<!-- proof-27: body_md -->
Consider the self-map $T_P\colon A \mapsto P'A$ on the $\mathbb{Z}_p$-module $M:=\mathbb{Z}_p[x]/(P)$.  Let $\mathbb{Q}_p$ be the field of $p$-adic fractions. We note that $\operatorname{disc}(P)\ne 0$ also over $\mathbb{Q}_p$, as the embedding $\mathbb{Q}\to\mathbb{Q}_p$ is one-to-one. If $\alpha_1,\ldots,\alpha_n$ are the roots of $P$ in an algebraic closure $\overline{\mathbb{Q}_p}$, they are distinct since $\operatorname{disc}(P)\neq 0$, so the map $A \mapsto (A(\alpha_1),\ldots,A(\alpha_n))$ realizes an isomorphism between $M\otimes \overline{\mathbb{Q}_p}$ and $\overline{\mathbb{Q}_p}^n$ (here $\otimes$ is as $\mathbb{Z}_p$-modules). Under this isomorphism $T_P$ becomes diagonal with eigenvalues $P'(\alpha_1),\ldots,P'(\alpha_n)$. In particular, $\det(T_P)=\prod_i P'(\alpha_i)=(-1)^{n(n-1)/2}\operatorname{disc}(P)$. So $v_p(\operatorname{disc}(P))=v_p(\det(T_P))$. On the other hand, $\mathbb{Z}_p[x]/(P,P')\cong M/\mathrm{Im}(T_P)$.
By Smith normal form over $\mathbb{Z}_p$, there are bases of the domain and codomain in which $T_P$ is diagonal with entries
$p^{a_1},\ldots,p^{a_n}$, where $a_i\ge 0$. Thus
\[
|M/\mathrm{Im}(T_P)| = \prod_{i=1}^n|\mathbb{Z}_p/p^{a_i}\mathbb{Z}_p| = p^{\sum_{i=1}^n a_i} = p^{v_p(\det(T_P))},
\]
hence the result follows.

From the previous lemma, it is clear that
\[
v_p(\operatorname{disc}(P)) \ge \deg\gcd(\overline{P},\overline{P}').
\]
Consequently, if there is another repeated irreducible factor $Q\ne x-a$ of $\overline{P}$, we get $v_p(\operatorname{disc}(P))\ge 1+\deg Q$. This proves the second part of (\ref{item:p^2-ii}). Also, if (\ref{item:p^2-i}) or (\ref{item:p^2-ii}) holds, then $v_p(\operatorname{disc}(P))\ge 2.$

Suppose then that neither (\ref{item:p^2-i}) nor (\ref{item:p^2-ii}) holds. We need to show that (\ref{item:p^2-iii}) holds if and only if $v_p(\operatorname{disc}(P))\ge 2.$ Choose a lift $a_0\in\mathbb{Z}_p$ of $a$ in the $p$-adic integers $\mathbb{Z}_p$. The results of \cite{FeldheimSen17} show that $\mathbb{P}(\operatorname{disc}(R)=0)\ll 1/n$ under the assumption that $\mu(i)<\frac12$ for all $i\in\mathbb{Z}$ (see also the recent \cite{MichelenYakir26} for non-quantitative results without any restriction on $\mu$). This condition is much weaker than ours, as $\mu(i)\ge 1/2$ implies $H_2(\mu)\le \log 4$. Thus it is enough to show that $\mathbb{P}(\operatorname{disc}(R)\textrm{ is a nonzero square})\le C/\log n$.

Let $\mathcal{P}_0$ denote the set of exceptional primes of {Corollary~\ref{cor:no-inadmissible-double-roots}}, which has size at most $Cn^C$ for some $C=C(\mu)>0$. Let $\mathcal{P}_1$ denote the set of all primes which lie between~$n^{B_0}$ and $\exp\left(\frac{\sqrt{n}}{\log^7 n}\right)$, and set $\mathcal{P}=\mathcal{P}_1\setminus\mathcal{P}_0$. Then $\mathcal{P}$ satisfies \ref{item:P1}. We further set $k_{\mathrm{sm}}=\left\lfloor \log^4 n \right\rfloor$, so that
\[
p^{k_{\mathrm{sm}}} \le \exp\left(\frac{\sqrt{n}}{\log^3 n}\right)
\]
for every $p\in\mathcal{P}$.

By Mertens' theorem,
\[
S(\mathcal{P}_1) = \sum_{p\in\mathcal{P}_1}\frac{1}{p} = \log\log\exp\left(\frac{\sqrt{n}}{\log^7 n}\right) - \log\log(n^{B_0}) + o(1) = \left(\frac{1}{2}+o(1)\right)\log n.
\]
On the other hand,
\begin{align*}
  S(\mathcal{P}_0\cap\mathcal{P}_1) &= \sum_{p\in\mathcal{P}_0\cap\mathcal{P}_1}\frac{1}{p} \le \sum_{n^{B_0}\le p\le n^{B_0+C+1}}\frac{1}{p} + \sum_{\substack{p\in\mathcal{P}_0 \\ p>n^{B_0+C+1}}}\frac{1}{p} \\
  &\le O_{\mu}(1) + \frac{Cn^C}{n^{B_0+C+1}} = O_{\mu}(1).
\end{align*}
Therefore $S(\mathcal{P}) = \left(\frac{1}{2}+o(1)\right)\log n$.

We choose $\varepsilon>0$ such that $\varepsilon<\frac{1}{2}-\frac{6}{H_2(\mu)}$ and write $m=\left\lfloor \frac{n-1}{3} \right\rfloor$. Combining previous results in the paper, we see that with probability at least $1-\frac{C'}{\log n}$ (where $C'$ depends only on $\mu,\varepsilon$), either $\operatorname{disc}(R)=0$ or the following hold:
\begin{enumerate}
  \item\label{item:proof1} There are at least $\left(\frac{1}{2}-\varepsilon\right)\log n$ primes in $\mathcal{P}$ such that there is a unique admissible double root of $R$ in $\mathbb{F}_p$ ({Proposition~\ref{prop:num-unique-double}} with $\delta=\frac{\varepsilon}{3}$);
  \item\label{item:proof2} There are no admissible triple roots of $R$ in $\mathbb{F}_p$ for any $p\in\mathcal{P}$ ({Proposition~\ref{prop:no-triple-roots}});
  \item \label{item:proof3} No admissible double root of $R$ lifts to a double root modulo $p^2$ for any $p\in\mathcal{P}$ ({Proposition~\ref{prop:no-double-root-lifts}});
  \item\label{item:proof4} There are no pairs of admissible double roots $(a,\alpha)\in\mathcal{A}_p\times\mathcal{A}_{p^k}$ of $R$ for any $p\in\mathcal{P}$ and $2\le k\le\log^4 n$ ({Proposition~\ref{prop:no-admissible-double-roots-ext}});
  \item\label{item:proof5} There are no pairs of double roots $(a,\alpha)\in\mathcal{A}_p\times\mathcal{B}_{p^k}$ of $R$ for any $p\in\mathcal{P}$ and $1\le k\le\log^4 n$, even for indamissible $\alpha$ ({Corollary~\ref{cor:no-inadmissible-double-roots}});
  \item\label{item:proof6} There are no pairs of double roots $(a,\alpha)\in\mathcal{A}_p\times\mathbb{F}^{\circ}_{p^k}$ of $R$ for any $p\in\mathcal{P}$ and any $k>\log^4 n$ with $p^{k+2}\le\| \mu \|_2^{-2m}$, whether $\alpha$ is admissible or not ({Proposition~\ref{prop:no-double-roots-big-ext}}).
\end{enumerate}

Suppose that $\operatorname{disc}(R)$ is a nonzero square, and let $\mathcal{P}(R)$ denote a set of precisely $\left\lfloor \left(\frac{1}{2}-\varepsilon\right)\log n \right\rfloor$ primes satisfying (\ref{item:proof1}). Fix a prime $p\in\mathcal{P}(R)$. By (\ref{item:proof1}), $R$ has a double root $a\in\mathbb{F}_p^{\times}$, so $p$ must divide $\operatorname{disc}(R)$. By our assumption, $\operatorname{disc}(R)$ is a square, so $p^2\mid\operatorname{disc}(R)$. Since $B_0\ge 3$ we have $p>n$, we may use {Proposition~\ref{prop:p^2-divides-disc}} to deduce that one of the conclusions of {Proposition~\ref{prop:p^2-divides-disc}} holds. The first conclusion is excluded by (\ref{item:proof2}), and the third by (\ref{item:proof3}). Therefore there is some irreducible $P\in\mathbb{F}_p[x]$ such that $P\ne x-a$ and $P^2\mid R$ in $\mathbb{F}_p[x]$.

We note that $P\ne x$. Indeed, we may assume that  $n$ is sufficiently large so that $p>H$ for all $p\in\mathcal{P}$. Then $x^2\mid R$ in $\mathbb{F}_p[x]$ forces $\omega_0=\omega_1=0$ as integers. But then $x^2\mid R$ in $\mathbb{Z}[x]$, which is impossible since we assumed $\operatorname{disc}(R)\ne 0$.

Suppose next that $\deg P=1$, i.e., $P=x-b$ for some $b\in\mathbb{F}_p$ with $b\ne 0,a$. If $b$ is admissible, then $a,b$ are two admissible double roots of $R$, contradicting (\ref{item:proof1}); if $b$ is inadmissible, we get a contradiction from (\ref{item:proof5}). This shows that $\deg P\ge 2$.

Let $k=\deg P$, and let $\alpha$ be a root of $P$. Then $\alpha$ is a double root of $R$ in $\mathbb{F}^{\circ}_{p^k}$. By (\ref{item:proof4}) and (\ref{item:proof5}), we must have $k > \log^4 n$. But then by (\ref{item:proof6}) we have
\[
p^{k+2} > \| \mu \|_2^{-2m} = \exp(H_2(\mu)m),
\]
so
\[
v_p(\operatorname{disc}(R))\ge 1+k \ge \left\lfloor \frac{H_2(\mu)m}{\log p} \right\rfloor.
\]

We can now conclude the proof. We define the (random) integer
\[
N = \prod_{p\in\mathcal{P}(R)} p^{\left\lfloor \frac{H_2(\mu)m}{\log p} \right\rfloor} \ge \prod_{p\in\mathcal{P}(R)} p^{\frac{H_2(\mu)m}{\log p} - 1} = \frac{\exp\left(H_2(\mu)m\left|\mathcal{P}(R)\right|\right)}{\prod_{p\in\mathcal{P}(R)}p}.
\]
By the above, $N\mid\operatorname{disc}(R)$, and
\begin{align*}
  \log N &\ge H_2(\mu)m\left|\mathcal{P}(R)\right| - \sum_{p\in\mathcal{P}(R)}\log p \\
  &\ge H_2(\mu)\left\lfloor \frac{n-1}{3} \right\rfloor\left\lfloor \left(\frac{1}{2}-\varepsilon\right)\log n \right\rfloor - \left(\frac{1}{2}-\varepsilon\right)\frac{\sqrt{n}\log n}{\log^7 n} \\
  &= \left(\frac{1}{6}-\frac{\varepsilon}{3}+o(1)\right)H_2(\mu)n\log n.
\end{align*}
On the other hand, since $H(R)\le H$, we have
\[
\left|\operatorname{disc}(R)\right| \le (Hn)^{2n} = \exp(2n\log n + 2n\log H),
\]
which is a contradiction when $n$ is sufficiently large since $\left(\frac{1}{6}-\frac{\varepsilon}{3}\right)H_2(\mu)>2$.
