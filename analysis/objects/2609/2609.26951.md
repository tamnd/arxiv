\label{example:SGD-classifier}
Let $Z_i=(X_i,Y_i)\in\mathbb{R}^D\times[K]$, $i=1,\dots,n+1$, be exchangeable, where $Z_1,\dots,Z_n$ arrive sequentially as training examples and $Z_{n+1}$ is the next test point. Consider a parametric class-index map
\begin{align*}
    f_\Theta:\mathbb{R}^D\to\mathbb{R}^K, \qquad
    f_\Theta(x)=\left({f_{\Theta,1}(x),\dots,f_{\Theta,K}(x)}\right),
\end{align*}
where $\Theta$ collects the trainable parameters. We refer to $f_\Theta(x)$ as the logit vector. In the linear multi-index model, $f_\Theta(x)=\Theta^\top x$ for $\Theta=(\theta_1,\dots,\theta_K)\in\mathbb{R}^{D\times K}$, and multinomial logistic regression models $\mathbb{P}(Y_i=k\mid X_i=x)=\mathsf{softmax}\left({{\Theta^\star}^\top x}\right)_k$, where $\mathsf{softmax}(v)_k:=\exp(v_k)/\sum_{\ell=1}^K\exp(v_\ell)$.

More generally, $f_\Theta$ may be a neural network, with $\Theta$ containing all network parameters. This formulation includes image-classification benchmarks such as CIFAR-10 ($K=10$) and CIFAR-100 ($K=100$) \citep{krizhevsky2009learning}, as well as the $1,000$-class ImageNet classification task \citep{russakovsky2015imagenet}. Starting from $\widehat{\Theta}_0$, suppose each arriving training example produces one SGD update,
\begin{align*}
    \widehat{\Theta}_i
    &=
    \widehat{\Theta}_{i-1}
    -
    \eta_i
    \left.
    \nabla_\Theta \ell(\Theta;Z_i)
    \right|_{\Theta=\widehat{\Theta}_{i-1}}, 
\end{align*}
where $\ell(\Theta;(x,y))=-\log\mathsf{softmax}\left({f_\Theta(x)}\right)_y$ is the cross-entropy loss. For $z=(x,y)$, one common choice are the cross-entropy score~\cite[Section~6.2]{goodfellow2016deep},
\begin{align*}
    s_i^{\mathsf{Ent}}(z;Z_{<i})
    =
    -\log\mathsf{softmax}\left({f_{\widehat{\Theta}_{i-1}}(x)}\right)_y.
\end{align*}
Another common choice is a running average of the logit-margin score,
\begin{align*}
    s_i^{\mathsf{Mar}}(z;Z_{<i})
    =
    \frac{1}{i\wedge T}
    \sum_{j=(i-T)_+}^{i-1}
    \left\{{
        \max_{k\in[K]\setminus\{y\}} f_{\widehat{\Theta}_j,k}(x)
        -
        f_{\widehat{\Theta}_j,y}(x)
    }\right\}.
\end{align*}
The margin loss in multiclass formulation is standard~\cite{crammer2001algorithmic}, and we adopt the suffix averaging method of the most recent iterates as is common in stochastic optimization~\cite{rakhlin2012making}. Our experiments in
Sections~\ref{sec:numerical-sgd-logistic} and~\ref{sec:numerical-mnist} will study concrete instances of this setup for multi-index logistic regression and one-pass neural-network training.

\label{example:continual-llm} Let $Z_i=(X_i,Y_i)$ be exchangeable prompt--response pairs for $i \in [n+1]$. Before observing $Z_i$, the current language-model pipeline $M_{i-1}$ assigns a conditional distribution $p_{i-1}(\cdot\mid x)$ over a fixed response space $\mathcal{Y}$ (e.g., $\mathcal{Y}$ may be the set of candidate responses). A natural choice of the conformal score is the length-normalized negative log-likelihood~\cite{malinin2021uncertainty},
\begin{align*}
    s_i^{\mathsf{NLL}}((x,y);Z_{<i})
    =
    -\frac{1}{|y|}
    \sum_{t=1}^{|y|}
    \log p_{i-1}\left({y_t\mid x,y_{<t}}\right).
\end{align*}
After $Z_i$ is scored, it may be used to update the pipeline. This update can be arbitrary (e.g., fine-tuning, replacing the model by a new checkpoint, or modifying its prompting components), as long as $M_{i-1}$ is determined from $Z_{<i}$.

Assume that $Z_1,\dots,Z_{n+1}\stackrel{\textup{iid}}{\sim} P$ for some distribution $P$, and assume that 
\[\textnormal{The conditional distribution of $s_i(Z;Z_{<i})\mid Z_{<i}$ is continuous, almost surely, when $Z\sim P$,}\]
to ensure that there are no ties among scores.
Define events $\mathcal{A}_1,\dots,\mathcal{A}_n$ and random variables $p_1,\dots,p_n$ as
\[\mathcal{A}_i = \left\{{s_i(Z_{n+1};Z_{<i})>s_i(Z_i;Z_{<i})}\right\},\qquad p_i =  \mathbb{P}\left({\mathcal{A}_i \mid Z_{<i+1}}\right).\]
Conditional on observing $Z_{<i}$, the random variables $s_i(Z_{n+1};Z_{<i})$ and $s_i(Z_i;Z_{<i})$ are i.i.d.\ and continuously distributed.
Therefore, by the standard CDF transform, it follows that $p_i \mid Z_{<i} \sim\mathsf{Unif}(0,1)$. Moreover, observe that $p_1,\dots,p_{i-1}$ are functions of $Z_{<i}$; consequently, by induction, we have shown that 
\[p_1,\dots,p_n \stackrel{\textup{iid}}{\sim} \mathsf{Unif}(0,1).\]
Moreover, since the event $\mathcal{A}_i$ depends only on $Z_1,\dots,Z_i,Z_{n+1}$, and is independent of $Z_{i+1},\dots,Z_n$, we also have
$p_i = \mathbb{P}\left({\mathcal{A}_i \mid Z_{<i+1}}\right) = \mathbb{P}\left({\mathcal{A}_i \mid Z_{<n+1}}\right)$.

Next we need a lemma:
\begin{lemma}\label{lem:pigeonhole}
    Let $\mathcal{A}_1,\dots,\mathcal{A}_n$ be any events. Suppose that there exists some random variable $W$ such that 
    $\mathbb{P}\left({\mathcal{A}_1 \mid W}\right),\dots,\mathbb{P}\left({\mathcal{A}_n \mid W}\right)\stackrel{\textup{iid}}{\sim} \mathsf{Unif}(0,1)$.
    Then for any $\Delta \geq 0$,
    \[\mathbb{P}\left({\sum_{i=1}^n \mathds{1}_{\mathcal{A}_i} \geq  n-\Delta}\right) \leq \frac{2\Delta +1}{n+1}.\]
\end{lemma}
\noindent Finally, by definition of the rolling-CP prediction set,
we have
\[Z_{n+1}\not\in\mathcal{C}_n \ \Longleftrightarrow \ \sum_{i=1}^n\mathds{1}_{\mathcal{A}_i} \geq (1-\alpha)(n+1).\]
Therefore, applying the lemma (with $\Delta=\alpha(n+1) - 1$, and $W=Z_{<n+1}$) completes the proof.

<!-- lemma-1: body_md -->
\label{lem:pigeonhole}
    Let $\mathcal{A}_1,\dots,\mathcal{A}_n$ be any events. Suppose that there exists some random variable $W$ such that 
    $\mathbb{P}\left({\mathcal{A}_1 \mid W}\right),\dots,\mathbb{P}\left({\mathcal{A}_n \mid W}\right)\stackrel{\textup{iid}}{\sim} \mathsf{Unif}(0,1)$.
    Then for any $\Delta \geq 0$,
    \[\mathbb{P}\left({\sum_{i=1}^n \mathds{1}_{\mathcal{A}_i} \geq  n-\Delta}\right) \leq \frac{2\Delta +1}{n+1}.\]
<!-- end lemma-1 -->
<!-- head-2: body_md -->
Let $p_i = \mathbb{P}\left({\mathcal{A}_i\mid W}\right)$.
Write $p_{(1)}\leq \dots \leq p_{(n)}$ for the order statistics of $p_1,\dots,p_n$, and let $\mathcal{A}_{(j)}$ denote the event associated with $p_{(j)}$. Fix any positive integer $k$. Then, by the pigeonhole principle, 
\[\textnormal{If }\sum_{i=1}^n \mathds{1}_{\mathcal{A}_i} \geq n- \Delta\textnormal{ then }\sum_{j=1}^k \mathds{1}_{\mathcal{A}_{(j)}} \geq k- \Delta.\]
Therefore,
\[\mathbb{P}\left({\sum_{i=1}^n \mathds{1}_{\mathcal{A}_i} \geq n- \Delta}\right)
\leq \mathbb{P}\left({\sum_{j=1}^k \mathds{1}_{\mathcal{A}_{(j)}} \geq k -  \Delta}\right)
= \mathbb{P}\left({\sum_{j=1}^k \mathds{1}_{\mathcal{A}_{(j)}} \geq k -  \lfloor\Delta\rfloor}\right)
= \mathbb{E}\left[{\mathbb{P}\left({\sum_{j=1}^k \mathds{1}_{\mathcal{A}_{(j)}} \geq k -  \lfloor\Delta\rfloor \,\bigg|\, W}\right)}\right],\]
where the second step holds since the sum $\sum_{j=1}^k \mathds{1}_{\mathcal{A}_{(j)}}$ is integer-valued. Applying Markov's inequality (and assuming $k > \lfloor \Delta\rfloor$), we then have
\[\mathbb{P}\left({\sum_{i=1}^n \mathds{1}_{\mathcal{A}_i} \geq n- \Delta}\right)
\leq \mathbb{E}\left[{\frac{\mathbb{E}\left[{\sum_{j=1}^k \mathds{1}_{\mathcal{A}_{(j)}}  \,\Big|\, W}\right]}{k-\lfloor \Delta\rfloor}}\right]=\mathbb{E}\left[{\frac{\sum_{j=1}^k p_{(j)}}{k-\lfloor \Delta\rfloor}}\right] = \frac{\sum_{j=1}^k \frac{j}{n+1}}{k-\lfloor \Delta\rfloor}= \frac{k(k+1)}{2(n+1)(k-\lfloor \Delta\rfloor)},\]
where the next-to-last step holds since $p_1,\dots,p_n\stackrel{\textup{iid}}{\sim} \mathsf{Unif}(0,1)$ and so $\mathbb{E}\left[{p_{(j)}}\right] = \frac{j}{n+1}$. The above calculation holds for any  integer $k$ with $\lfloor \Delta\rfloor < k \leq n$; choosing $k = 2\lfloor \Delta\rfloor + 1$ completes the proof.
\endproof

Let $X,Y\in\mathbb{R}$ be random variables. We define the stochastic order, the convex order, and the decreasing convex order, respectively, as
\begin{align*}
X\preceq_{\mathrm{st}} Y&\quad\Longleftrightarrow\quad\mathbb{E}[f(X)]\leq \mathbb{E}[f(Y)] & & \textnormal{for every nondecreasing function $f:\mathbb{R}\to\mathbb{R}$}; \\
X\preceq_{\mathrm{cvx}} Y&\quad\Longleftrightarrow\quad\mathbb{E}[f(X)]\leq \mathbb{E}[f(Y)] & & \textnormal{for every convex function $f:\mathbb{R}\to\mathbb{R}$}; \\
X\preceq_{\mathrm{dcx}} Y&\quad\Longleftrightarrow\quad\mathbb{E}[f(X)]\leq \mathbb{E}[f(Y)]&&\textnormal{for every nonincreasing convex function $f:\mathbb{R}\to\mathbb{R}$}.
\end{align*}
The comparisons are restricted to measurable functions for which both expectations are well-defined. In particular, we can compare the definitions as follows:
\begin{equation*}\label{eqn:st_implies_dcx}
\textnormal{If $X\succeq_{\mathrm{st}}Y$ then $X\preceq_{\mathrm{dcx}} Y$,}
\end{equation*}
and
\begin{equation*}\label{eqn:cvx_implies_dcx}
\textnormal{If $X\preceq_{\mathrm{cvx}}Y$ then $X\preceq_{\mathrm{dcx}} Y$.}
\end{equation*}

<!-- theorem-6: body_md -->
\label{thm:stability-score-training-conditional} 
Let $Z_1, \cdots, Z_n, Z_{n+1} \stackrel{\textup{iid}}{\sim} P$ and let $\alpha \in (0,1)$. Define the rolling-CP prediction set $\mathcal{C}_n$ as in~\eqref{eqn:iterative-conformal-set}, and suppose Assumption~\ref{assmp:score-stability} holds for some $m\in[n]$ and some $\nu\in[0,1]$. Let
    \[\alpha_P(Z_{< n+1}) = \mathbb{P}(Z_{n+1} \notin \mathcal{C}_{n} \mid Z_1, \dots, Z_n)\]
    as the training-conditional miscoverage rate of $\mathcal{C}_n$. Then, for any $\delta \in (0,1)$,
    \begin{equation*}
        \mathbb{P}\left({\alpha_P(Z_{<n+1}) \leq \alpha \cdot \frac{n+1}{n-m+2} + \sqrt{\frac{\log(1/\delta)}{2(n-m+2)}}+2 \sqrt[3]{\nu}}\right) \geq  1 - \delta - \sqrt[3]{\nu}.
    \end{equation*}
<!-- end theorem-6 -->
<!-- head-4: body_md -->
The proof combines ideas from the arguments behind Theorem~\ref{thm:training-conditional} (training-conditional coverage without stability), and Theorem~\ref{thm:stability-score} (marginal coverage, with stability).

Fix any $\gamma>0$. As in the proof of Theorem~\ref{thm:stability-score}, it holds that
\[\mathbb{P} \left({\left|\sum_{i=m}^n \mathds{1}\{s_{i}(Z_{n+1}; Z_{<i}) > s_{i}(Z_i; Z_{<i})\} -  \sum_{i=m}^n  \mathds{1}\{s_\star(Z_{n+1};Z_{<m}) > s_\star(Z_i;Z_{<m})\} \right| \geq (n-m+1)\gamma }\right)\leq \frac{\nu}{\gamma}.\]
Define a random variable
\[D = \sum_{i=m}^n \mathds{1}\{s_{i}(Z_{n+1}; Z_{<i}) > s_{i}(Z_i; Z_{<i})\} - \sum_{i=m}^n  \mathds{1}\{s_\star(Z_{n+1};Z_{<m}) > s_\star(Z_i;Z_{<m})\},\]
so that the bound above implies
\[\mathbb{P}\left({D \geq (n-m+1)\gamma}\right)\leq \frac{\nu}{\gamma}.\]
Since $Z_{n+1}\in\mathcal{C}_n$ if and only if $\sum_{i=1}^n \mathds{1}\{s_{i}(Z_{n+1}; Z_{<i}) > s_{i}(Z_i; Z_{<i})\} <(1-\alpha)(n+1)$, it holds that
\[\textnormal{If $Z_{n+1}\not\in\mathcal{C}_n$ then $\sum_{i=m}^n  \mathds{1}\{s_\star(Z_{n+1};Z_{<m}) > s_\star(Z_i;Z_{<m})\} + D \geq (1-\alpha)(n+1)-(m-1)$.}\]
Therefore,
\begin{multline*}\alpha_P(Z_{<n+1}) \leq \mathbb{P}\left({\sum_{i=m}^n  \mathds{1}\{s_\star(Z_{n+1};Z_{<m}) > s_\star(Z_i;Z_{<m})\} + D \geq (1-\alpha)(n+1)-(m-1) \,\Big|\, Z_{<n+1}}\right)\\
\leq \mathbb{P}\left({\sum_{i=m}^n  \mathds{1}\{s_\star(Z_{n+1};Z_{<m}) > s_\star(Z_i;Z_{<m})\}\geq (1-\alpha)(n+1) -(m-1)- (n-m+1)\gamma  \,\Big|\, Z_{<n+1}}\right)\\
+ \mathbb{P}\left({D \geq  (n-m+1)\gamma  \,\Big|\, Z_{<n+1}}\right).
\end{multline*}
By Markov's inequality, for any $c> 0$,
\[\mathbb{P}\left({ \mathbb{P}\left({D \geq  (n-m+1)\gamma  \,\Big|\, Z_{<n+1}}\right) \geq c}\right) \leq \frac{\mathbb{E}\left[{\mathbb{P}\left({D \geq (n-m+1)\gamma \,\Big|\, Z_{<n+1}}\right)}\right]}{c} = \frac{\mathbb{P}\left({D \geq  (n-m+1)\gamma}\right)}{c} \leq \frac{\nu}{c\gamma}.\]
Next note that $s_\star(Z_m;Z_{<m}),\dots,s_\star(Z_{n+1};Z_{<m})$ are i.i.d.\ conditional on $Z_{<m}$. Denote $S_i = s_\star(Z_i;Z_{<m})$ for each $i = m,\cdots, n+1$. Let $S_{(1)}\leq \dots \leq S_{(n-m+1)}$ be the order statistics of $S_m, \cdots, S_n$. 
Then
\[\sum_{i=m}^n  \mathds{1}\{s_\star(Z_{n+1};Z_{<m}) > s_\star(Z_i;Z_{<m})\}\geq (1-\alpha)(n+1) -(m-1)- (n-m+1)\gamma \ \Longleftrightarrow \ S_{n+1} > S_{(k)}\] where $k = \lceil (1-\alpha)(n+1) -(m-1)- (n-m+1)\gamma\rceil$. Writing $F$ as the CDF of $s_\star(Z;Z_{<m})$ (for $Z\sim P$, conditional on $Z_{<m}$), we then have
\[\mathbb{P}\left({S_{n+1} > S_{(k)}\mid Z_{<n+1}}\right) = 1 - F(S_{(k)}).\]
Since $S_{(k)}$ is an order statistic of $S_m,\dots,S_n$, which are i.i.d.\ draws from the CDF $F$ (after conditioning on $Z_{<m}$), 
\[F(S_{(k)})\succeq_{\mathrm{st}} U_{(k)}, \]
where $U_{(k)}$ is the $k$th order statistic of $n-m+1$ i.i.d.\ $\mathsf{Unif}(0,1)$ random variables, which satisfies $U_{(k)} \sim\mathsf{Beta}(k,n-m+2-k)$. Therefore, for $c'>0$,
\[\mathbb{P}\left({\mathbb{P}\left({S_{n+1} > S_{(k)}\mid Z_{<n+1}}\right)  \geq \frac{n-m+2-k}{n-m+2} + c'}\right)
\leq \mathbb{P}\left({ U_{(k)} \leq \frac{k}{n-m+2} - c'}\right) \leq e^{-2(n-m+2)c'{}^2},\]
by subgaussianity of the Beta distribution \citep{elder2016bayesian}. 

Combining everything,
\[\mathbb{P}\left({\alpha_P(Z_{<n+1}) \leq \frac{n-m+2-k}{n-m+2} + c' + c}\right) \geq  1 - \frac{\nu}{c\gamma} - e^{-2(n-m+2)c'{}^2}.\]
Plugging in our choice of $k$, and simplifying,
\[\mathbb{P}\left({\alpha_P(Z_{<n+1}) \leq \frac{\alpha(n+1) +(n-m+1)\gamma}{n-m+2} + c' + c}\right) \geq  1 - \frac{\nu}{c\gamma} - e^{-2(n-m+2) c'{}^2}.\]
Choosing $\gamma = c = \sqrt[3]{\nu}$ and $c' = \sqrt{\frac{\log(1/\delta)}{2(n-m+2)}}$, 
this simplifies to
\[\mathbb{P}\left({\alpha_P(Z_{<n+1}) \leq \alpha \cdot \frac{n+1}{n-m+2} + \sqrt{\frac{\log(1/\delta)}{2(n-m+2)}} + 2\sqrt[3]{\nu}}\right) \geq  1 -\delta- \sqrt[3]{\nu} ,\]
which completes the proof.
\endproof

\label{lem:V-n-k-solution}
For each $n\geq 1$ and $k\in\{0,\dots,n\}$, define
random variables $p_1(n),\dots,p_n(n)$ and $B_1(n,k),\dots,B_n(n,k)$ as

    \end{align*}

<!-- head-5: body_md -->
We first notice that $p_{i-1}(n-1)$ has exactly the same distribution as $p_i(n)$. Moreover, it is easy to see conditional on $B_1(n,k)=1$, the joint distributions below are identical:
    \begin{align*}
        &(B_2(n,k), p_2(n)), \dots, (B_n(n,k), p_n(n)) \nonumber \\
        & \qquad \stackrel{d}{=} (B_1(n-1,k-1), p_1(n-1)), \dots, (B_{n-1}(n-1,k-1), p_{n-1}(n-1)).
    \end{align*}
    While on $B_1(n,k)=0$, we have similarly
    \begin{align*}
        &(B_2(n,k), p_2(n)), \dots, (B_n(n,k), p_n(n)) \nonumber \\
        & \qquad \stackrel{d}{=} (B_1(n-1,k), p_1(n-1)), \dots, (B_{n-1}(n-1,k), p_{n-1}(n-1)).
    \end{align*}
    This implies
    \begin{align*}
        V(n,k) & = \mathbb{P}(B_1(n,k)=1) \left({\mathbb{E} \left[{p_1(n) \mid B_1(n,k)=1}\right] +  \mathbb{E} \left[{\sum_{i=2}^n B_i(n,k)p_i(n) \mid B_1(n,k)=1}\right]}\right) \nonumber \\
        & \qquad + \mathbb{P}(B_1(n,k)=0) \mathbb{E} \left[{\sum_{i=2}^n B_i(n,k)p_i(n) \mid B_1(n,k)=0}\right] \nonumber \\
        & \stackrel{\mathrm{(i)}}{=} \mathbb{P}(B_1(n,k)=1) \left({\mathbb{E} \left[{p_1(n) \mid B_1(n,k)=1}\right] +  \sum_{i=1}^{n-1}\mathbb{E} \left[{ B_i(n-1,k-1)p_i(n-1)}\right]}\right) \nonumber \\
        & \qquad + \mathbb{P}(B_1(n,k)=0) \sum_{i=1}^{n-1} \mathbb{E} \left[{ B_i(n-1,k)p_i(n-1)}\right] \nonumber \\
        & = \mathbb{P}(B_1(n,k)=1) \left({\mathbb{E} \left[{p_1(n) \mid B_1(n,k)=1}\right] + V(n-1,k-1)}\right) + \mathbb{P}(B_1(n,k)=0) V(n-1,k),
    \end{align*}
 in (i) we use the preceding two equalities in distribution. Substituting the explicit decision rule $B_1(n,k)$ and the distribution of $p_1(n)$, it is clear that
    \begin{align*}
        V(n,k) &= \frac{k+1}{n+1} \cdot \left({\frac{k}{2n} + V(n-1,k-1)}\right) + \frac{n-k}{n+1} V(n-1,k) \nonumber \\
        & = \frac{k(k+1)}{2n(n+1)} + \frac{k+1}{n+1} V(n-1, k-1) + \frac{n-k}{n+1} V(n-1,k). 
    \end{align*}
    The boundary conditions are given by $V(n,0)=0$ and $V(n,n)=n/2$. These conditions are sufficient to uniquely determine the doubly indexed recursive sequence $\{V(n,k)\}$. Indeed, one can verify that the ansatz
\begin{align*}
    V(n,k) = \frac{k(k+1)}{2(n+1)}
\end{align*}
satisfies the recursive relation exactly and thus is its unique solution.
\endproof
