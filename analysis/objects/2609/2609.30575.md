\label{lem:bkk}
	Let $k > 2$ be even and $D>0$ be a discriminant. Then
	\begin{align*}
	\mathbb{D}^{2k - 1}\mathcal{F}_{1-k,D}(z) &= -\frac{D^{k-\frac{1}{2}}\Gamma(2k - 1)}{(4\pi)^{2k-1}}f_{k,D}(z) \qquad \text{and}\\
	\xi_{2-2k}\mathcal{F}_{1-k,D}(z) &= D^{k-\frac{1}{2}}f_{k,D}(z)
	\end{align*}
	for $z \not \in E_D$.

		&= -\frac{(4\pi)^{2k-1}}{D^{k-\frac{1}{2}}\Gamma(2k - 1)}(-4\pi)^{1 - 2k}R_{2-2k,z}^{2k - 1}\mathcal{F}_{1-k,D}(z)\\
