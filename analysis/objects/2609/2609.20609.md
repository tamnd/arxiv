<!-- remark-1: body_md -->
\label{rem:Lh}
For the following it will be convenient to characterize $w$ satisfying \eqref{eq:udk-3a}, \eqref{eq:udk-3b} as
\begin{equation}
 w=L_g\big( (1-g)\xi\big)\,,
 \label{eq:Lhw}
\end{equation}
where 
$L_h:L^2(\Gamma)\to H^1(\Omega)\subset L^2(\Gamma)$, $h\in L^\infty(\Gamma)$, $h\geq 0$, $|\{h>0\}|>0$, is the family of linear operators introduced in  
  \IfNoValueTF{LNRV21}{ 
    \IfBooleanTF{#3}{\optcite{*}}{\biblatexcite{*}} 
  }{ 
    \biblatexcite[LNRV21]{*} 
  } 
{p.~1226}, i.e.~$z=L_h(v)$ satisfies
\begin{equation}
	0 =\Delta z \textrm{ in } \Omega, \qquad
	\frac{\partial z}{\partial n}  + hz = v \textrm{ on }\Gamma\,.
  \label{eq:DefLh} 
\end{equation}
By  
  \IfNoValueTF{LNRV21}{ 
    \IfBooleanTF{#3}{\optcite{*}}{\biblatexcite{*}} 
  }{ 
    \biblatexcite[LNRV21]{*} 
  } 
{Lemma 2.6} the map $L_h:L^2(\Gamma)\to H^1(\Omega)$ is continuous, self-adjoint and $L_h(h)=1$. 
Using the continuity of $H^1(\Omega)\hookrightarrow L^4(\Gamma)$, the continuity of the Neumann-to-Dirichlet map $L^4(\Gamma)\to W^{1,4}(\Gamma)$, see  
  \IfNoValueTF{NiRV20}{ 
    \IfBooleanTF{#3}{\optcite{*}}{\biblatexcite{*}} 
  }{ 
    \biblatexcite[NiRV20]{*} 
  } 
{Appendix A}, and the continuous embedding $W^{1,4}(\Gamma)\hookrightarrow C^{0,\frac{1}{2}}(\Gamma)$ we obtain 
\begin{align}
  L_h: L^4(\Gamma)\to W^{1,4}(\Gamma)\quad
  \textrm{ and }\quad L_h: L^4(\Gamma)\to C^{0,\frac{1}{2}}(\Gamma)
  \textrm{ are continuous.}
  \label{eq:LhCont}
\end{align}
In addition $h \mapsto L_h$ is monotone decreasing. 
This means for any $h_1,h_2\in L^\infty(\Gamma)$ with $0\leq h_1\leq h_2$ we have
\begin{equation}\label{Lmonotoneh}
  L_{h_1}(s)\geq L_{h_2}(s) \quad\textrm{ for all }s\in L^2(\Gamma),\, s\geq 0\,.
\end{equation}
We also note that it holds 
\begin{equation}\label{Lmonotones}
 L_h(s_1) \geq L_h(s_2) \qquad \mbox{ if } s_1 \geq s_2 \; \mbox{ a.e.}.
\end{equation}

Furthermore it has been shown in \cite[Proposition 2.7]{LNRV21}, that for a solution $(u,\xi,w)$ of \eqref{eq:udk-1}-\eqref{eq:udk-4}
\begin{equation}\label{Lformula}
 L_g\big( (1-g)\xi)= L_{g\chi} \big( (1-g) \chi\big)\,,
\end{equation}
where $\chi$ is the characteristic function of the set $\{x \in \Gamma, t>0 \,|\, u(x,t)>0\}$.

<!-- lemma-1: body_md -->
\label{L.aprioriwk}
For almost all $t>0$ we have
\begin{equation*}
 \alpha_*(t) \leq w_k(\cdot,t) \leq \alpha^*(t) \qquad \textrm{ in } \Gamma\,.
\end{equation*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We use \eqref{Lformula} with  the characteristic function $\chi_k$ of the set $\{x \in \Gamma, t>0 \,|\, u_k(x,t)>0\}$. 
Together with \eqref{Lmonotoneh}, \eqref{Lmonotones} and $L_h(h)=1$ it follows that 
\begin{align*}
 w_k&= L_{g\chi_k} \big( (1-g)\chi_k\big)  \geq L_{\mathbf{g}_{\max} \chi_k} \big( (1-\mathbf{g}_{\max}) \chi_k\big)\\
 &= \frac{1}{\mathbf{g}_{\max}} L_{\mathbf{g}_{\max} \chi_k} (\mathbf{g}_{\max} \chi_k) - L_{\mathbf{g}_{\max} \chi_k}(\mathbf{g}_{\max} \chi_k)= \frac{1}{\mathbf{g}_{\max}} -1 = \alpha_*\,.
\end{align*}
The upper bound follows analogously.

<!-- lemma-2: body_md -->
\label{L.uktozero}
 We have $u_k(\cdot,t) \to 0$ in $L^1(\Gamma)$ for all $t>0$ and $u_k \rightharpoonup 0$ in $W^{2,1}_2(\Gamma\times (\delta,T))$ for any $0<\delta<T$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We first notice that since $u_k$ is nonnegative we obtain from mass conservation and $m_k \to 0$ that $u_k(\cdot,t) \to 0$ in $L^1(\Gamma)$ for all $t \geq 0$. By standard parabolic regularity, since the right-hand side of \eqref{eq:udk-1} is due to Lemma \ref{L.aprioriwk} uniformly bounded in $L^{\infty}((0,\infty), L^2(\Gamma))$, it follows that $u_k$ is uniformly bounded in $W^{2,1}_2(\Gamma \times (\delta,T))$ for any $0<\delta<T$. Thus there exists a weakly convergent subsequence in this space. Since $u_k(\cdot,t)$ converges to zero in $L^1(\Gamma)$, the limit is zero and the whole sequence converges.

<!-- proposition-1: body_md -->
\label{P.wkconv}
It holds
\begin{align}
  w_k &\to \alpha_* \quad\textrm{ in }L^p(\Gamma_T)\textrm{ for any }T>0\,,p \in [1,\infty)\,.\label{eq:WkConv1}
\end{align}

<!-- proof-3: body_md -->
We fix $T>0$.
By testing \eqref{eq:udk-3a}-\eqref{eq:udk-3b} with $w_k$ and using the lower bound on $g$ we easily find that 
\begin{equation}\label{eq:nablawkbound}
  \sup_{t>0}\| w_k(\cdot,t)\|_{H^1(\Omega)} \leq C(g)\,.
\end{equation}
For a subsequence we then have weak convergence in $L^2(0,T;H^1(\Omega))$ to some limit $\overline{w}$.
Due to Lemma \ref{L.aprioriwk}, the continuity of the trace operator, and an identification argument we deduce that in addition 
\begin{equation} \label{wkconv}
 w_k \stackrel{\ast}{\rightharpoonup} \overline{w} \qquad \textrm{ in } L^\infty(\Gamma_T)
\end{equation}
holds for the traces of $w_k,\overline{w}$.
We are going to show that $\overline{w}=\alpha_*$, hence it suffices to consider the chosen subsequence.

\begin{steps}
\item For almost all $t>0$
\begin{equation*}
 \overline{w}(\cdot,t) = \overline{w}(t)\quad\textrm{ is constant in }\Omega\,.
\end{equation*}
Since we can write the right hand side of \eqref{eq:udk-3b} as $\partial_t u_k - \Delta u_k$, we see from the weak formulation of \eqref{eq:udk-3a}-\eqref{eq:udk-3b} and Lemma \ref{L.uktozero} that $\overline{w}$ is harmonic inside $\Omega$ with zero Neumann boundary data.
Hence the claim follows. 

\item We have 
\begin{equation}\label{limsupwk}
 \overline{w}(t)  \leq \alpha_*(t) \qquad \textrm{ for a.a. } t>0\,.
\end{equation}

To prove this claim we first multiply equation \eqref{eq:udk-1} with a smooth, nonnegative test function $\varphi$ with compact support in $\Gamma\times(0,T)$ to obtain
\begin{align*}
 0&= \int_0^{\infty} \int_{\Gamma} u_k \big( \Delta \varphi + \partial_t\varphi\big) - \big( ( 1-g)\xi_k - g w_k)\big)\varphi \,d\sigma\,dt\\
 &\geq \int_0^{\infty} \int_{\Gamma} u_k \big( \Delta \varphi + \partial_t\varphi\big) - \big((1-g) - gw_k \big) \varphi\,d\sigma\,dt \,.
\end{align*}
The first term on the right hand side converges to zero due to Lemma \ref{L.uktozero}.
Hence we find in the limit $k \to \infty$ that 
\begin{equation}
  0 \leq \int_0^T \int_{\Gamma} \big(1-(1+\overline{w})g\big)\varphi\,d\sigma\,dt\,.
  \label{eq:22a}
\end{equation}
Next fix an arbitrary Lebesgue point $t_0>0$ of $\overline{w}-\alpha_*$.
For any given $\varepsilon>0$ choose $x_0\in\Gamma$ with $g(x_0,t_0)>\mathbf{g}_{\max}(t_0)-\varepsilon$. 
By continuity of $g,\mathbf{g}_{\max}$ there exists $\delta=\delta(\varepsilon)$ with $0<\delta<\min\{t_0,T-t_0\}$ and
\begin{equation}
  g\geq \mathbf{g}_{\max}-\varepsilon \quad\textrm{ in }B(x_0,\delta)\times (t_0-\delta,t_0+\delta)\,.
  \label{eq:EstGmaxEps}
\end{equation}
Now consider any nonnegative $\eta\in C^\infty_c((-1,1))$, $\int_\mathbb{R} \eta=1$, let $\eta_\varepsilon=\frac{1}{\delta}\eta\big(\frac{\cdot-t_0}{\delta}\big)$ and choose an arbitrary nontrivial, nonnegative test function $\psi_\varepsilon\in C^\infty(\Gamma)$ with $\operatorname{spt} \psi_\varepsilon\subset B(x_0,\delta)$.

Setting $\varphi_\varepsilon(x,t)=\eta_\varepsilon(t)\psi_\varepsilon(x)$ and using $1 = (1+\alpha_*) \mathbf{g}_{\max}$ we obtain from \eqref{eq:22a} that
\begin{align}
  0 &\leq  \int_0^\infty\int_\Gamma \big(1-(1+\overline{w})(\mathbf{g}_{\max}-\varepsilon)\big)\varphi_\varepsilon\,d\sigma \,dt\nonumber\\
  &= \int_\Gamma\psi_\varepsilon\,d\sigma \int_0^\infty\eta_\varepsilon(t) \big((1+\overline{w})\varepsilon +(\alpha_*-\overline{w})\mathbf{g}_{\max}\big)\,dt\,,
\end{align}
which yields after dividing by $\int_\Gamma\psi_\varepsilon$, by letting $\varepsilon\to 0$, $\delta=\delta(\varepsilon)\to 0$, and by using that $t_0$ is a Lebesgue point of $\overline{w}-\alpha_*$ and that $\mathbf{g}_{\max}$ is continuous and positive
\begin{equation*}
  \overline{w}(t_0)-\alpha_*(t_0) \leq 0\,.
\end{equation*}
Since almost all $t_0\in (0,T)$ are Lebesgue points of $\overline{w}-\alpha_*$, we conclude that \eqref{limsupwk} holds.

\item We have
\begin{equation}\label{liminfwk}
 \overline{w}(t) = \alpha_*(t) \qquad \textrm{ for a.a. } t>0\,.
\end{equation}
Indeed, the inequality $\overline{w} \geq \alpha_*$ follows directly from the lower bound in Lemma \ref{L.aprioriwk}.
By the previous step \eqref{liminfwk} holds.

\item Finally, since $w_k\geq \alpha_*$ for all $k\in\mathbb{N}$ by Lemma \ref{L.aprioriwk} we deduce from the weak* convergence in $L^\infty$ that for any $T>0$
\begin{equation*}
  0 = \lim_{k\to\infty}\int_0^T\int_\Gamma (w_k -\alpha_*)\,d\sigma\,dt 
  =\lim_{k\to\infty} \| w_k -\alpha_*\|_{L^1(\Gamma_T)}\,, 
\end{equation*}
which gives the $L^1_{\operatorname{loc}}$-convergence of $w_k$ to $\alpha_*$.
Then, since $(w_k)$ is also uniformly bounded due to Lemma \ref{L.aprioriwk} and since the limit is a priori given we also get convergence in $L^p(\Gamma_T)$ for any $p \in [1,\infty)$ and the whole sequence $k\to\infty$.

\end{steps}

<!-- proposition-2: body_md -->
\label{P.localization}

Consequently, from \eqref{eq:65a}-\eqref{eq:66b} we obtain
\begin{align*}
  \partial_t u &= -(h-\varphi+h\varphi)\,\chi_{\{u>0\}} \quad\textrm{ in }\Gamma\times (0,\infty)\,,\\
  \varphi &= \mathbf{g}_{\max}\Big(L_{g\chi_{\{u>0\}}}\big((1-g)\chi_{\{u>0\}}\big)-w_\infty\Big)\,\\
  u &\geq 0\,,\quad 0 \le \xi \le 1\,,\quad u\,\xi = u\,.
   
\end{align*}

We now can follow the proof of Theorem \ref{thm:Conc2}.
The main difference is the additional space dependence of $\varphi$ but by the uniform convergence of $\varphi(\cdot,t)\to 0$ with $t\to\infty$ the arguments all remain valid.
