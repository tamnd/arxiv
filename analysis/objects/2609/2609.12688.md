<!-- remark-1: body_md -->
The positive results recalled above are often formulated for an \(H\)-cylindrical Wiener process, with integrands taking values in \(\mathscr{L}_2(H,X)\), the space of Hilbert--Schmidt operators.
Our counterexample is formulated for the special case of a real Brownian motion, corresponding to \(H=\mathbb R\) and the canonical identification \(\mathscr{L}_2(H,X) =  X\).
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
There exist a separable Hilbert space \(X\) and a
densely defined operator \(A\) on \(X\) with the following properties.
\begin{enumerate}[label=\textup{(\roman*)}]
\item\label{it:main1} \(-A\) generates a compact and exponentially stable analytic semigroup
\(S\); the operator \(A\) is sectorial of angle zero.
\item\label{it:main2} There exists a stochastic basis and a 
predictable process $g\in L^\infty\bigl(\Omega;L^2([0,1];X)\bigr)$
such that $U_g$ as defined in \eqref{eq:mild-solution}
satisfies
\begin{equation*}
 \mathbb P\Big(
  \sup_{t\in[0,1]\cap\mathbb Q}\lVert U_g(t)\rVert=\infty
 \Big)>0.
\end{equation*}
In particular, \(U_g\) has no version with almost surely bounded or continuous paths and \eqref{eq:intro-maximal-inequality} does not hold.
\end{enumerate}
<!-- end theorem-1 -->

\label{thm:main2}
There exist a separable Hilbert space \(X\) and a

\begin{enumerate}[label=\textup{(\roman*)}]

\item\label{it2:main2} $A$ has a bounded $H^\infty$-calculus with $\omega_{H^\infty}(A) = \pi/2$.

<!-- proposition-1: body_md -->
\label{prop:Maxsquare}
    Suppose that $\omega(A) < \pi / 2$ and that there exist $0 < c$ and $C < \infty$ such that we have the following two-sided square function estimate:
\begin{equation}
    \label{eq:sqrexp}
    c \lVert x\rVert
    \leq \lVert x\rVert_{\psi,A} \leq C \lVert x\rVert, \quad x \in X.
\end{equation}
Then for any $p \in (0,\infty)$, $S$ satisfies a maximal estimate \eqref{eq:maximal-inequality} on $\mathbb R_+$ with constant
\begin{equation*}
    \mathsf{M}_{p,\mathbb R_+}(S) \leq B_p\, c^{-1}C,
\end{equation*}
where $B_p$ is as in~\eqref{eq:BDG}.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Let $g$ be a predictable step process, and set $\mathscr{X} = L^2(\mathbb R_+;X)$.
    By expanding the definition of $\lVert x\rVert_{\psi,A}$ and using $e^{-tA} = S(t)$, we rewrite~\eqref{eq:sqrexp} into a more transparent form:
    \begin{align}
        \label{eq:sqrexpexplicit}
        c^2 \lVert x\rVert^2 \leq \int_0^{\infty}\lVert A^{1/2}S(r)x\rVert^2 \,\mathrm d r \leq C^2 \lVert x\rVert^2, \quad x \in X.
    \end{align}
    We now define the $\mathscr{X}$-valued predictable process
    \begin{align*}
        \Phi(s) \coloneq r \mapsto \1_{s < r}A^{1/2}S(r-s)g(s),
    \end{align*}
    as well as the $\mathscr{X}$-valued martingale
    \begin{align*}
        \mathcal{M}(t) \coloneq \int_0^t \Phi(s)\,\mathrm d \beta_s,
    \end{align*}
    and observe that for $r > t$, the semigroup identity yields
    \begin{equation*}
        \mathcal{M}(t)(r) = \int_0^t 1_{s < r}A^{1/2}S(r-s)g(s) \,\mathrm d \beta_s = A^{1/2}S(r-t)U_g(t).
    \end{equation*}
    From the lower estimate of~\eqref{eq:sqrexpexplicit}, we thus find for $t \in \mathbb R_+$
    \begin{align*}
        c^2 \lVert U_g(t)\rVert^2 &\leq \int_0^{\infty} \lVert A^{1/2} S(r)U_g(t)\rVert^2 \,\mathrm d r
        = \int_t^{\infty} \lVert A^{1/2} S(r-t)U_g(t)\rVert^2 \,\mathrm d r 
        \\&\quad= \int_t^{\infty}\lVert \mathcal{M}(t)(r)\rVert^2 \,\mathrm d r \leq \lVert \mathcal{M}(t)\rVert_{\mathscr{X}}^2.
    \end{align*}
    On the other hand, from the upper estimate of~\eqref{eq:sqrexpexplicit}, we find for $s \in \mathbb R_+$
    \begin{align*}
        \lVert \Phi(s)\rVert_{\mathscr{X}}^2 = \int_s^{\infty}\lVert  A^{1/2}S(r-s)g(s)\rVert^2 \,\mathrm d r
        = \int_0^{\infty}\lVert  A^{1/2}S(r)g(s)\rVert^2 \,\mathrm d r \leq C^2 \lVert g(s)\rVert^2.
    \end{align*}
    Thus, combining the two estimates above and using \eqref{eq:BDG} in between, we conclude that
    \begin{align*}
        \lVert U_g\rVert_{L^p(\Omega;C(\mathbb R_+;X))} 
        &\leq c^{-1}
        \lVert \mathcal{M}\rVert_{L^p(\Omega;C(\mathbb R_+;\mathscr{X}))} 
        \\ & \leq B_p c^{-1}
        \lVert \Phi\rVert_{L^p(\Omega;L^2(\mathbb R_+;\mathscr{X}))} 
        \\ & \leq B_p c^{-1}C\lVert g\rVert_{L^p(\Omega;L^2(\mathbb R_+;X))}. \qedhere
    \end{align*}

<!-- lemma-1: body_md -->
\label{lem:green-killed-bm}
Let $f : \mathbb R_+ \to \mathbb R$ be nonnegative and measurable. Then we have the identity
\begin{equation}
\label{eq:green-killed-bm}
    \mathbb E\int_0^{\tau^a}f(b^a_s)\,\mathrm d s
    =
    2\int_0^\infty (a\wedge r)f(r)\,\mathrm d r.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
It suffices to prove \eqref{eq:green-killed-bm} for nonnegative
continuous compactly supported functions $f:\mathbb R\to[0,\infty)$.
Indeed, evaluating the two sides at indicators defines two Borel
measures on $\mathbb R$. Once the identity is established for such test
functions,  \cite[Theorem~2.25]{Kallenberg2021} implies that the two measures coincide. The identity for arbitrary nonnegative Borel measurable
functions follows by approximation with simple functions and
monotone convergence.

For $f$ as above, we define $F(a)$ as the right-hand side of~\eqref{eq:green-killed-bm}.
Then, $F$ is bounded with two bounded derivatives, and satisfies
\[
    F(0)=0,
    \qquad
    F'(s)=2\int_s^\infty f(r)\,\mathrm d r,
    \qquad
    F''(s)=-2f(s), \qquad s \geq 0.
\]
Applying It\^o's formula to $F(b^a_{t \wedge \tau^a})$ then gives
\begin{equation*}
F(b^a_{t \wedge \tau^a}) = F(a) + \int_0^{t \wedge \tau^a} F'(b^a_s) \,\mathrm d \beta_s - \int_0^{t \wedge \tau^a}f(b^a_s) \,\mathrm d s, \qquad t \geq 0,
\end{equation*}
so that taking expectations and rearranging the terms results in
\begin{equation*}
    \mathbb E \, F(b^a_{t \wedge \tau^a}) + \mathbb E \int_0^{t \wedge \tau^a} f(b^a_s) \,\mathrm d s = F(a) , \qquad t \geq 0.
\end{equation*}
Taking $t \to \infty$ using dominated convergence, the first term on the left-hand side vanishes and we recover the desired identity.

<!-- lemma-2: body_md -->
\label{lem:integrable}
Let $a>0$ and $x\in \mathsf D(A)$. Then 
    \begin{equation}
        \label{eq:kernelidentity}
        \mathbb E \int_0^{\tau^a} \lVert C P_{b^a_s}x\rVert^2 \,\mathrm d s = 2 \int_0^{\infty} (a \wedge r) \lVert C P_{r}x\rVert^2 \,\mathrm d r < \infty.
    \end{equation}
    Moreover, we have the identity
    \begin{equation}
        \label{eq:limitidentity}
        \lim_{a\to \infty}\mathbb E \int_0^{\tau^a} \lVert C P_{b^a_s} x\rVert^2 \,\mathrm d s = 
        2 \int_0^{\infty} r \lVert C P_{r}x\rVert^2 \,\mathrm d r 
        = \lVert x\rVert_{\phi,A}^2.
    \end{equation}

The first identity of~\eqref{eq:kernelidentity} is a direct application of Lemma~\ref{lem:green-killed-bm}.
    The finiteness follows since $x \in \mathsf D(A)$ and $P_r$ is exponentially stable in the operator norm.

\begin{equation*}

<!-- lemma-4: body_md -->
\label{lem:stoppingidentity}
Suppose that $\mathsf{M}_{2,\mathbb R_+}(S) < \infty$. 
    For every $g \in L^2_{\mathcal P}(\Omega;L^2(\mathbb R_+;X))$ and $y \in \mathsf D(A^*)$ we have the identity
    \begin{equation}
        \label{eq:stoppingidentity}
        \mathbb E \langle U_g (\tau^a), y\rangle = -\mathbb E\int_0^{\tau^a} \langle g(s), C^* P_{b^a_s}^*y\rangle \,\mathrm d s.
    \end{equation}
<!-- end lemma-4 -->
<!-- remark-3: body_md -->
The expectation on the right-hand side is well-defined by Cauchy--Schwarz, \eqref{eq:kernelidentity}, and Remark~\ref{rem:integrable}.
<!-- end remark-3 -->
<!-- proof-5: body_md -->
By density and the maximal estimate it suffices to prove the identity for $g \in L^2_{\mathcal P}(\Omega;L^2(\mathbb R_+;\mathsf D(A)))$.
    In this case, $U_g$ has a continuous version which satisfies
    \begin{equation*}
        \,\mathrm d U_g(t) =-A U_g(t) \,\mathrm d t + g \,\mathrm d\beta_t
    \end{equation*}
    in the strong sense.
    Combining this with~\eqref{eq:itoPbt}, it follows by applying It\^o's formula to $f(U_g(t \wedge \tau^a), P^*_{b^a_{t \wedge \tau^a}}y)$ with $f(u,v) = \langle u, v\rangle$  that
    \begin{equation*}
    \begin{aligned}
        \,\mathrm d \langle U_g(t), P_{b^a_t}^*y\rangle = -&\langle A U_g, P_{b^a_t}^* y\rangle \,\mathrm d t + \langle g, P_{b^a_t}^* y\rangle \,\mathrm d \beta_t \\
        +& \langle U_g, A^* P^*_{b^a_t}y\rangle \,\mathrm d t - \langle U_g, C^* P^*_{b^a_t}y\rangle \,\mathrm d \beta_t \\
        -&\langle g, C^* P^*_{b^a_t}y \rangle \,\mathrm d t,
    \end{aligned}
    \end{equation*}
    up to time $\tau^a$.
    Observing the cancellation between the drifts, we conclude from the identity above that
    \begin{equation*}
        \langle U_g(t \wedge \tau^a), P_{b^a_{t\wedge \tau^a}}^*y\rangle + \int_0^{t \wedge \tau^a} \langle g(s), C^* P^*_{b^a_s}y \rangle \,\mathrm d s = 
        \int_0^{t \wedge \tau^a} \langle g(s), P_{b^a_s}^* y\rangle - \langle U_g(s), C^* P^*_{b^a_s}y\rangle \,\mathrm d \beta_s.
    \end{equation*}
    From the maximal estimate~\eqref{eq:maximal-inequality} and Remark~\ref{rem:integrable} applied to the left-hand side of this identity, it follows that the martingale on the right-hand side is uniformly integrable. Thus, we may take the expectation, use optional stopping, and let $t\to \infty$ to conclude~\eqref{eq:stoppingidentity}.
<!-- end proof-5 -->
<!-- proof-6: body_md -->
Fix $a > 0$, $y \in \mathsf D(A^*)$ and $g \in L^2_{\mathcal P}(\Omega;L^2(\mathbb R_+;X))$.
    Taking the squared modulus of~\eqref{eq:stoppingidentity}, and applying Jensen's inequality and the maximal estimate to the left-hand side, we find:
    \begin{equation*}
        \lvert \mathbb E \int_0^{\tau^a} \langle g(s),  C^* P_{b^a_s}^*y \rangle \,\mathrm d s\rvert^2 \leq \mathsf{M}_{2,\mathbb R_+}^2(S) \lVert y\rVert^2 \mathbb E \int_0^{\tau^a} \lVert g(s)\rVert^2 \,\mathrm d s.
    \end{equation*}
    By duality in $L^2_{\mathcal P}(\mathbb R_+ \times \Omega;X)$, we then obtain
    \begin{equation*}
        \mathbb E \int_0^{\tau^a} \lVert C^* P_{b^a_s}^* y\rVert^2 \,\mathrm d s \leq \mathsf{M}_{2,\mathbb R_+}^2(S) \lVert y\rVert^2.
    \end{equation*}
    Taking $a \to \infty$ using~\eqref{eq:limitidentity} then gives the adjoint upper square function estimate:
    \begin{equation*}
        \lVert y\rVert_{\phi,A^*}^2 \leq \mathsf{M}_{2,\mathbb R_+}^2(S) \lVert y\rVert^2.
    \end{equation*}
    Since $y \in \mathsf D(A^*)$ was arbitrary, the above extends by density to all $y \in X^*$.
    The desired lower square function estimate~\eqref{eq:necessary-lower-square} then follows from a standard duality argument for square functions (see \cite[p. 196]{Haase2006}). The constant is $2$ due to $\int_0^\infty \phi(t)^2 \frac{\,\mathrm d t}{t} =\frac12$.

<!-- lemma-6: body_md -->
For all finitely supported $(a_k)_{k \geq 1}$, it holds that
    \begin{equation}
    \lVert \sum_k a_k f_k\rVert_{\phi,A}^2 
    \lesssim \lVert (a_k)_{k \geq 1}\rVert_{\ell^2}^2 \label{eq:QC-upper-coefficient-estimate}.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
A direct integration yields
\[
  \lVert \sum_k a_kf_k\rVert_{\phi,A}^2
 =
 \sum_{j,k\geq 1}a_j\overline{a_k}B_{jk},
 \qquad
 B_{jk}
 :=
 \frac{2\mu^{j+k}}{(\mu^j+\mu^k)^2}\langle f_j,f_k\rangle.
\]
Since $\lVert f_n\rVert \lesssim 1$, we can estimate the matrix coefficients as $\lvert B_{jk}\rvert \lesssim \mu^{-\lvert j - k\rvert}$ uniformly in $j$ and $k$. As a consequence, we find the following bounds for the row and column sums:
\begin{equation}
 \sup_{j\geq 1}\sum_{k\geq 1}|B_{jk}| + \sup_{k\geq 1}\sum_{j\geq 1}|B_{jk}| \lesssim 1.
\end{equation}
By the (unweighted) Schur test, it follows that $\|B\|_{\mathcal L(\ell^2)} \lesssim 1$, which yields the desired estimate.

<!-- proof-8: body_md -->
The claims in the first sentence follow directly from \cite[Lemmas~9.1.1 and 9.1.2]{Haase2006} and \cite[Proposition 10.2.28]{HNVW2017}.
    To see the failure of the lower square-function estimate, let $n \geq 1$, define $a_{n,k} \coloneq \1_{k \leq 2n+1}$ for $k \geq 1$, and set
    \begin{equation}
        x_n(\theta) \coloneq \sum_k a_{n,k} f_k(\theta) = \sum_{\lvert \ell\rvert \leq n} e^{2 \pi i \ell \theta} = \frac{\sin\big((2n+1)\pi \theta \big)}
     {\sin(\pi \theta)}, \quad \theta \in \mathbb{T}.
    \end{equation}
    For a sufficiently small absolute constant $c>0$,
    it holds that $|x_n(\theta)|\gtrsim n$ whenever $0<\theta <c/n$.
    Consequently,
    \begin{align}\label{eq:normxn}
     \|x_n\|^2
     =
     \int_{\mathbb T}|x_n(\theta) |^2w(\theta)\,\,\mathrm d \theta
     \gtrsim
     n^2\int_0^{c/n}\theta^{-\alpha}\,\,\mathrm d\theta
     \gtrsim n^{1+\alpha}, \quad n \geq 1.
    \end{align}
    On the other hand, for the coefficients we have $\lVert (a_{n,k})_{k \geq 1}\rVert_{\ell^2}^2 \eqsim n$.
    Thus the estimate $\lVert x_n\rVert \lesssim \lVert (a_{n,k})_{k \geq 1}\rVert_{\ell^2}$ fails, which implies via~\eqref{eq:QC-upper-coefficient-estimate} that $\lVert x_n\rVert \lesssim \lVert x_n\rVert_{\phi,A}$ must also fail.

    For the upper estimate, let $x = \sum_k a_k f_k$ with $(a_k)_{k\geq 1}$ finitely supported and note that
\[\lVert x\rVert_{\phi,A}^2 
    \lesssim \lVert (a_k)_{k \geq 1}\rVert_{\ell^2}^2 = \lVert x\rVert_{L^2(\mathbb T)}^2 \leq \|x\|^2,\]
    where  we used \eqref{eq:QC-upper-coefficient-estimate}, Parseval's identity and $w(\theta)\geq 1$. The general case follows by density.

Write $M = \sup_{t \in I}\lVert S(t)\rVert_{\mathscr{L}(X)}$. Let $p>0$ and let $g$ be a predictable step process with values in $\mathsf D(A)$.  Let $\theta > \max\{1,M\}$ and $0 < \varepsilon < \theta^{-p}$. Set $\delta =(\theta - M)/ (2K_{\varepsilon})$.
Taking into account that the constant $K_{\varepsilon}$ can be chosen independent of the stochastic basis, an adaptation of the three stopping time argument of~\cite[Lemma 3.5]{coxvw2024sharp} (which originates from~\cite{Burkholder1973}), results in the following good-$\lambda$ inequality:
\begin{align}\label{eq:goodlambda}
    \mathbb P(\lVert U_g\rVert_{C(I;X)} > \theta \lambda, \lVert g\rVert_{L^2(I;X)} \leq \delta \lambda) \leq \varepsilon \mathbb P(\lVert U_g\rVert_{C(I;X)} \geq \lambda), \quad \lambda > 0.
\end{align}
For convenience of the reader, we briefly indicate the details. 

Fix $\lambda>0$. Let $\sigma$ be the first time that
$\lVert U_g\rVert$ reaches $\lambda$, with $\sigma= \sup I$ if no such time exists. Set $V_{\lambda}= \{\sigma<\sup I\}$. 
Then
\[
    \{\lVert U_g\rVert_{C(I;X)}>\theta\lambda\}\subseteq V_{\lambda}
    \subseteq \{\lVert U_g\rVert_{C(I;X)}\geq\lambda\}.
\]
If $\mathbb P(V_{\lambda})=0$, \eqref{eq:goodlambda} is immediate.
Otherwise, define $\mathbb Q_\lambda=\mathbb P(\,\cdot\mid V_{\lambda})$.
Since $V_\lambda\in\mathscr F_\sigma$, the strong Markov property implies that
$\widehat\beta_s\coloneq\beta_{\sigma+s}-\beta_\sigma$
defines a Brownian motion under $\mathbb Q_\lambda$ with respect to the
shifted filtration $(\mathscr F_{\sigma+s})_{s\geq0}$.
Extend $g$ by zero beyond the original time interval and put
$\widehat g(s)=g(\sigma+s)$. Define
\[
    \rho=\inf\left\{
        r\geq0:
        \int_0^r\lVert \widehat g(s)\rVert^2\,\,\mathrm d s
        \geq\delta^2\lambda^2
    \right\},
    \qquad
    h=\widehat g\,\1_{[0,\rho]},
\]
where $\inf\varnothing=\infty$. Then $h$ is predictable for the
shifted filtration and
\[
    \lVert h\rVert_{L^\infty(\mathbb Q_\lambda;L^2(I;X))}
    \leq\delta\lambda.
\]
Write
\[
    U_h(s)=\int_0^s S(s-r)h(r)\,\,\mathrm d\widehat\beta_r,
    \qquad s\in I.
\]
Since $g$ is in $L^2(I;\mathsf D(A))$ a.s., the same holds for $h$, and therefore, $U_h$ has a continuous version. By \ref{it3:extrap}, with the same constant $K_\varepsilon$ on this
new stochastic basis,
\[
    \mathbb Q_\lambda\left(
        \lVert U_h\rVert_{C(I;X)}
        >K_\varepsilon\delta\lambda
    \right)\leq\varepsilon.
\]

On the event $\{\lVert U_g\rVert_{C(I;X)}>\theta\lambda,\ \|g\|_{L^2(I;X)}\leq\delta\lambda\}$, one has $\rho = \sup I$, and hence
\[
    U_g(\sigma+s)=S(s)U_g(\sigma)+U_h(s),
    \qquad \sigma+s\in I.
\]
Moreover, continuity gives $\lVert U_g(\sigma)\rVert=\lambda$.
Consequently, on this event,
\[
    \lVert U_h\rVert_{C(I;X)}
    >(\theta-M)\lambda
    =2K_\varepsilon\delta\lambda.
\]
It follows that
\[
    \begin{aligned}
    \mathbb P(\lVert U_g\rVert_{C(I;X)}>\theta\lambda,\ \|g\|_{L^2(I;X)}\leq\delta\lambda)
    &\leq \mathbb P(V_\lambda)\,
        \mathbb Q_\lambda\left(
            \lVert U_h\rVert_{C(I;X)}
            >K_\varepsilon\delta\lambda
        \right)\\ &\leq\varepsilon\,\mathbb P(V_\lambda)\\ &\leq\varepsilon\,\mathbb P(\lVert U_g\rVert_{C(I;X)}\geq\lambda),
    \end{aligned}
\]
which proves~\eqref{eq:goodlambda}.
Since $\theta > 1$, $\delta > 0$, $\varepsilon > 0$, and $\varepsilon \theta^p < 1$, we may now apply~\cite[Lemma 7.1]{Burkholder1973} using~\eqref{eq:goodlambda} to find
\begin{align}
    \mathbb E\lVert U_g\rVert_{C(I;X)}^p \leq \frac{\theta^p \delta^{-p}}{1 - \theta^p \varepsilon} \mathbb E \lVert g\rVert_{L^2(I;X)}^p,
\end{align}
hence $\mathsf{M}_{p,I}(S) < \infty$.

<!-- theorem-5: body_md -->
\label{thm:chaining}
Let $I = [0,T]$ for $T > 0$.
For every predictable $g:[0,T]\times \Omega\to X$ such that
\[\int_0^T \|g(t,\cdot)\|_{L^\infty(\Omega;X)}^2 \,\mathrm d t<\infty,\]
the process $U_g$ has a continuous version.
Moreover, for every $p \in [1,\infty)$ one has
\begin{equation}
 \left\|
  \sup_{t \in I}\lVert U_g(t)\rVert
 \right\|_{L^p(\Omega)}
 \leq C_p M_I^2 \Big(\int_0^T \|g(s,\cdot)\|_{L^\infty(\Omega;X)}^2 \,\mathrm d s\Big)^{1/2},
 \label{eq:chaining-estimate}
\end{equation}
where $C_p = \frac{2^{3/4}B_{\max\{p,4\}}} {2^{1/4} - 1}$, $M_I = \sup_{t \in I}\lVert S(t)\rVert_{\mathscr{L}(X)}$, and $B_q$ is as in \eqref{eq:BDG}.
<!-- end theorem-5 -->
<!-- proof-18: body_md -->
By rescaling time, we only need to prove the case $T = 1$. We first prove  the result in the case that $g\in L^\infty([0,1]\times \Omega;X)$. 
Since $g$ is uniformly bounded, the continuity follows from the factorization method \cite{DaPratoKwapienZabczyk1987}. Therefore, it remains to prove the estimate \eqref{eq:chaining-estimate}.
By homogeneity  we may additionally assume that $\int_0^1 \|g(t,\cdot)\|_{L^\infty(\Omega;X)}^2 \,\mathrm d t = 1$.

For an interval $(a,b] = J \subset [0,1]$, we introduce by analogy with $U_g(t)$ the notation
\begin{align*}
    U_g(J) \coloneq \int_a^b S(b-s)g(s) \,\mathrm d \beta_s.
\end{align*}

We begin by partitioning the time interval in a way which is tailored to $g$.
We define $\theta \colon [0,1] \to [0,2]$ by
\begin{equation}
    \theta(t) =\int_0^t 1 + \lVert g(s,\cdot)\rVert^2_{L^{\infty}(\Omega;X)} \,\mathrm d s.
\end{equation}
Using dominated convergence, we see that $\theta$ is continuous, strictly increasing and we have $\theta(0) = 0$ and $\theta(1) = 2$.
Thus, $\theta$ has a continuous strictly increasing inverse $\varrho \colon [0,2] \to [0,1]$, from which we define the binary nested sequence of partitions
\begin{align*}
    J_{n,k} &\coloneq (\varrho(k 2^{-n+1}), \varrho((k+1)2^{-n+1})], \qquad 0 \leq k < 2^n, \\
    \mathcal{D}_n &\coloneq \{J_{n,k} \}_{0 \leq k < 2^n}.
\end{align*}
Finally, we set $\mathcal{T} \coloneq \{ \varrho(k 2^{-n+1}) : n \geq 0, \, 0 \leq k < 2^n \}$.
Note that $\mathcal{T}$ is dense in $[0,1]$, since $\varrho$ is continuous and surjective.

We record the fact that for any $J \in \mathcal{D}_n$, we have $\int_J \lVert g(t,\cdot)\rVert_{L^{\infty}(\Omega;X)}^2 \,\mathrm d t \leq 2^{-n+1}$ by construction.
Thus, for $q \in (0,\infty)$, the bound \eqref{eq:BDG} gives
\begin{equation}
    \label{eq:bdgchain}
    \lVert U_g(J)\rVert_{L^q(\Omega;X)} \leq \sqrt{2} B_q 2^{-n/2}M_{I}, \quad J \in \mathcal{D}_n.
\end{equation}
Note that the intervals $J \in \mathcal{D}_n$ do not depend on $\omega$, which is crucial for this inequality.

Now fix $t \in \mathcal{T}$.
It follows from our construction that $(0,t]$ can be partitioned into a finite set of intervals $\mathcal{A} \subset \bigcup_{n \geq 0}\mathcal{D}_n$ which contains at most one interval from each $\mathcal{D}_n$.
Using the semigroup identity and letting $r_{J}$ denote the right endpoint of $J$, it then follows that almost surely
\[
 U_g(t)=\sum_{J\in \mathcal{A}}S(t-r_J)U_g(J).
\]
Since $t$ was arbitrary, this results in the almost sure inequality
\begin{equation}
\label{eq:treebound}
 \sup_{t \in \mathcal{T}}\lVert U_g(t)\rVert
 \leq M_{I} \sum_{n \geq 0} \max_{J \in \mathcal{D}_n}\lVert U_g(J)\rVert.
\end{equation}
We want to estimate the right-hand side term by term in $L^p(\Omega)$.
To do this, note that for $q \in [p,\infty)$, the embeddings $L^q(\Omega) \hookrightarrow L^p(\Omega)$ and $\ell^q \hookrightarrow \ell^{\infty}$ combined with Fubini's theorem give 
\begin{align}
    \label{eq:Dnbound}
    \lVert \max_{J \in \mathcal{D}_n}\lVert U_g(J)\rVert\rVert_{L^p(\Omega)}
    \leq 2^{n/q} \max_{J \in \mathcal{D}_n} \lVert U_g(J)\rVert_{L^q(\Omega;X)} \leq \sqrt{2} B_q M_{I} 2^{n/q - n/2},
\end{align}
with the second step following from~\eqref{eq:bdgchain}.
Choosing $q = \max\{4,p\}$ the right-hand side of~\eqref{eq:Dnbound} becomes summable in $n$, so the triangle inequality applied to~\eqref{eq:treebound} results in
\begin{equation}
    \lVert \sup_{t \in \mathcal{T}} \lVert U_g(t)\rVert\rVert_{L^p(\Omega)} \leq 2^{1/2}M_{I}^2 B_{q} \sum_{n\geq 0}  2^{n/q - n/2} \leq \frac{2^{3/4}M_{I}^2 B_{q}} {2^{1/4} - 1}.
\end{equation}
By density, we may replace $\mathcal{T}$ by $[0,1]$ in the above estimate (using the continuous version of $U_g$), which concludes the proof of the case where $g$ is uniformly bounded.

To treat the general case we cannot use approximation by adapted step processes, since these are not dense in the norm we are using. Instead, we can use an approximation argument through truncations. Indeed, 
let $g_k(s) = g(s)\1_{\lVert g(s)\rVert \leq k}$. Applying \eqref{eq:chaining-estimate} to the bounded processes $g_k - g_\ell$, it follows that $(U_{g_k})_{k\geq 1}$ is a Cauchy sequence in $L^p(\Omega;C([0,1];X))$ and hence convergent. Since the limit is a version of $U_g$, this completes the proof of the continuity for general $g$. The estimate \eqref{eq:chaining-estimate} follows by taking limits as well.

By the same reductions as in the proof above, it is enough to prove~\eqref{eq:chaining-estimate} for $T = 1$ and $g\in L^\infty([0,1]\times \Omega;X)$ with the normalization $\int_0^1 \|g(t,\cdot)\|_{L^\infty(\Omega;X)}^2 \,\mathrm d t = 1$ (in which case $U_g$ has a continuous version).

To do this, we define the following objects:
\begin{align*}
    \widetilde{S}(t',s') &= S(\varrho(t') - \varrho(s')),  \qquad 0 \leq s' \leq t' \leq 2,\\

\end{align*}
Note that $\widetilde{S}$ is a $C_0$-evolution family (by continuity of $\varrho$), $\widetilde{\beta}$ is a Brownian motion with respect to a time-changed filtration (by L\'evy's characterization), and $\tilde{g}$ is predictable with respect to this filtration.
By the deterministic time change formula, we then obtain the identity
\begin{align*}
    U_g(\varrho(t')) = \int_0^{t'}\widetilde{S}(t',s')\widetilde{g}(s') \,\mathrm d \widetilde{\beta}_{s'}, \qquad t' \in [0,2].
\end{align*}
After observing that $\lVert \tilde{g}(s')\rVert \leq 1$ for all $s' \in [0,2]$, we may apply the factorisation method \cite{Seidler1993} to the right-hand side to find
\begin{align}
    \left\lVert \sup_{t' \in [0,2]} \lVert U_g(\varrho(t'))\rVert_{X}\right\rVert_{L^p(\Omega)} \lesssim M_I^2 \sqrt{p}.
\end{align}
Since $\varrho$ is surjective onto $[0,1]$, this exactly coincides with~\eqref{eq:chaining-estimate}. \qedhere
