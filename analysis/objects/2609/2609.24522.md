<!-- lemma-5: body_md -->
\citep{{rajendra-book}}
    For $\mathbf{X}\in \mathbb{R}^{m\times n}$, $\mathbf{Y}\in \mathbb{R}^{n\times s}$, and $\mathbf{Z}\in \mathbb{R}^{s\times t}$, if $\frac{1}{p}+\frac{1}{q}+\frac{1}{r}=1$, we have
    $$
        \left\|\mathbf{X}\mathbf{Y}\mathbf{Z}\right\|_{S_1}\leq \left\|\mathbf{X}\right\|_{S_p}\left\|\mathbf{Y}\right\|_{S_q}\left\|\mathbf{Z}\right\|_{S_r},
    $$
    where the Schatten-$p$ norm is defined as
    $$
          \|\mathbf{X}\|_{S_p}=\left\{\begin{array}{cc}
               \Big(\sum_{i=1}^{\min\{m,n\}}\sigma_i(\mathbf{X})^p\Big)^{1/p}, & 1\leq p<\infty,\\
          \sigma_1(\mathbf{X}), & p=\infty.
           \end{array}\right.
    $$

<!-- lemma-6: body_md -->
\label{trace-transform}
    For any diagonal matrix $\mathbf{X}\in\mathbb R^{n\times n}$ with positive entries, $\mathbf{Y}\in\mathbb R^{m\times n}$, and $p>1$, we have
    $$\left\|\mathbf{X}^{\frac{1}{p}}\right\|_{S_{p}}^{p}=\operatorname{tr} \mathbf{X},\qquad \|\mathbf{Y}\|_{S_{2}}^{2}=\|\mathbf{Y}\|_F^2.$$
<!-- end lemma-6 -->
<!-- proof-1: body_md -->
Recall that we denote $\sigma_i$ and $\lambda_i$ as the $i$-th singular value and eigenvalue, respectively. Therefore $\sigma_i\left(\mathbf{X}^{\frac{1}{p}}\right)=\sigma_i\left(\mathbf{X}\right)^{\frac{1}{p}}$ and $\sigma_i(\mathbf{X})=\lambda_i(\mathbf{X})$. So we have
    $$
        \left\|\mathbf{X}^{\frac{1}{p}}\right\|_{S_{p}}^{p}=\left(\sum_{i=1}^n \lambda_i\left(\mathbf{X}\right)^{\frac{1}{p}\cdot p}\right)^{\frac{1}{p}\cdot p}=\sum_{i=1}^n \lambda_i\left(\mathbf{X}\right)=\operatorname{tr} \mathbf{X},
    $$
    and
    $$
        \left\|\mathbf{Y}\right\|_{S_{2}}^{2}=\left(\sum_{i=1}^{\min\{m,n\}} \sigma_i\left(\mathbf{Y}\right)^2\right)^{\frac{1}{2}\cdot 2}=\operatorname{tr} \mathbf{Y}\mathbf{Y}^\top=\|\mathbf{Y}\|_F^2.
    $$
    \hfill $\blacksquare$
<!-- end proof-1 -->
<!-- remark-1: body_md -->
Throughout this paper, we define $\mathbf{X}^{\frac{1}{p}}$ as the element‑wise $\frac{1}{p}$-th power. To avoid having to define the matrix power, we prove the first equality only for diagonal $\mathbf{X}$, although it holds for any symmetric positive semi-definite matrix.

<!-- definition-1: body_md -->
\citep{{rajendra-book}}
A norm is unitarily invariant if it satisfies $\|\mathbf{P}\mathbf{A}\mathbf{Q}^\top\|=\|\mathbf{A}\|$ for orthogonal matrices $\mathbf{P}$ and $\mathbf{Q}$. Both the Frobenius norm and nuclear norm are unitarily invariant norms.
<!-- end definition-1 -->
<!-- proof-2: body_md -->

        =&7\sqrt{38(m+n)}K\hat{\sigma}\cdot
        \frac{\sqrt{K \hat{\sigma}^{2} L\left(f\left(\mathbf{X}_{1}\right)-f^{*}\right)}}{\delta}+49 \frac{K \hat{\sigma}^{2} L\left(f\left(\mathbf{X}_{1}\right)-f^{*}\right)}{\delta^2},
    \end{align*}
    and
    \begin{align*}
        \frac{1}{K}\sum_{k=1}^{K} \mathbb{E}\left[\left\|\nabla f(\mathbf{X}_{k})\right\|_{*}\right]
        \leq&7\sqrt[4]{\frac{(m+n)\hat{\sigma}^2}{\delta^2}}\sqrt[4]{\frac{\hat{\sigma}^{2}L\left(f(\mathbf{X}_{1})-f^{*}\right)}{K}}+7\sqrt{\frac{\hat{\sigma}^{2}L\left(f(\mathbf{X}_{1})-f^{*}\right)}{K\delta^2}}\\
        \leq& \left(7\left(\sqrt{m+n}\frac{\hat{\sigma}}{\delta}\right)^{\frac{1}{2}}+7\frac{\hat{\sigma}}{\delta}\right)\cdot \sqrt[4]{\frac{\hat{\sigma}^{2}L\left(f(\mathbf{X}_{1})-f^{*}\right)}{K}}\\
        \leq& \left(4\sqrt{m+n}+11\frac{\hat{\sigma}}{\delta}\right)\cdot \sqrt[4]{\frac{\hat{\sigma}^{2}L\left(f(\mathbf{X}_{1})-f^{*}\right)}{K}},
    \end{align*}
    where we use $\delta=\frac{\tau\hat{\sigma}}{\sqrt{m+n}}$. Substituting this setting yields the stated bound.
