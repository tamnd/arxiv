\label{free02}
Let $S$ be a commutative ai-semiring and $\Sigma$ an equational basis for $S$ that contains an infinite subset of $\Omega$.
If $\mathbf{u}^{(m)}$ is $\mathbf{t}$-free for every inequality $\mathbf{s} \preceq \mathbf{t}$ in $\Sigma \setminus \Omega$ and for every integer $m \geq 1$, then $S$ is nonfinitely based.

<!-- lem-1: body_md -->
\label{lem01}
Let $\mathbf{q}\preceq \mathbf{u}$ be a nontrivial inequality such that
$\mathbf{u}=\mathbf{u}_1+\mathbf{u}_2+\cdots+\mathbf{u}_n$ with $\mathbf{u}_i, \mathbf{q} \in X^+_c$ for $1\leq i \leq n$.
Then $\mathbf{q}\preceq \mathbf{u}$ is satisfied by $S_c(ab)$ if and only if $\mathbf{u}$ and $\mathbf{q}$ satisfy one of the following conditions\textup{:}
\begin{enumerate}[$(\rm i)$]
\item  $\ell(\mathbf{u}_i)\geq 3$ for some $\mathbf{u}_i\in \mathbf{u}$;

\item  $c(L_1(\mathbf{u}))\cap c(L_2(\mathbf{u}))\neq \emptyset$;

\item  the graph ${\mathbb G}_{\mathbf{u}}$ contains an odd cycle;

\item  $\mathbf{q}\in {\mathbb G}_{\mathbf{u}}^{odd}$.
\end{enumerate}

\label{lem:small-tests}
Let $\mathbf{u}\approx \mathbf{u}+\mathbf{q}$ be a nontrivial ai-semiring identity such that
$\mathbf{u}=\mathbf{u}_1+\cdots+\mathbf{u}_n$, where $\mathbf{u}_i, \mathbf{q}\in X^+$, $1\leq i \leq n$. Then
\begin{itemize}
\item[$(1)$] $\mathbf{q}\preceq \mathbf{u}$ holds in $M_2$ if and only if $c(\mathbf{q})\subseteq c(\mathbf{u})$.

\item[$(2)$] $\mathbf{q}\preceq \mathbf{u}$ holds in $D_2$ if and only if $c(\mathbf{u}_i)\subseteq c(\mathbf{q})$ for some $\mathbf{u}_i \in \mathbf{u}$.

\item[$(3)$] $\mathbf{q}\preceq \mathbf{u}$ holds in $T_2$ if and only if $\ell(\mathbf{u}_i)\geq 2$ for some $\mathbf{u}_i \in \mathbf{u}$.
\end{itemize}

By \cite[Corollary~2.5]{gmrz}, every subvariety of

\label{cor:limit-subvarieties}
The varieties $\mathsf{V}(S_c(abc))$ and $\mathsf{V}(SR_6)$
are the only limit subvarieties of $\mathsf{V}(S_7^0)$.

By \cite[Theorem~3.6]{rjzl} and \cite[Theorem~5.4]{lry},
$\mathsf{V}(S_c(abc))$ and $\mathsf{V}(SR_6)$ are limit varieties.
Let $\mathcal V$ be a limit subvariety of
$\mathsf{V}(S_7^0)$. If $S_c(abc)\in\mathcal V$, then
$\mathsf{V}(S_c(abc))\leq\mathcal V$. Since the former is
nonfinitely based, the minimality of $\mathcal V$ gives
$\mathcal V=\mathsf{V}(S_c(abc))$.

Proposition~\ref{profibres}, $\mathcal V$ belongs to one of the six

or the sixth fibre and is not its least member. Proposition
~\ref{prop:strictcontains} yields
$\mathsf{V}(SR_6)\leq\mathcal V$. The minimality of $\mathcal V$
now gives $\mathcal V=\mathsf{V}(SR_6)$.
