<!-- teo-1: body_md -->
\cite[Theorem 1.2]{Propervarietyfinitepresented}\label{theorem: propervarietiesfinitepresentation}
    Let $X$ be a connected scheme that is proper over an algebraically closed field $k$. Then $\pi^{et}_1(X)$ is topologically finitely presented.
<!-- end teo-1 -->
<!-- teo-2: body_md -->
Let \(\mathcal{X}\) be a connected algebraic stack with finite inertia that is proper over an algebraically closed field $k$. Then \(\pi_1^{et}(\mathcal{X})\) is topologically finitely presented.
<!-- end teo-2 -->
<!-- rmk-1: body_md -->
The assumption of finite inertia on a proper stack is equivalent to the assumption that the stack has affine stabilizers. Indeed, the diagonal of a proper stack is proper, so \(\mathcal{I}_{\mathcal{X}}\to \mathcal{X}\) is proper, and a proper morphism with affine fibers is finite.
<!-- end rmk-1 -->
<!-- defi-1: body_md -->
\cite[Definition 3.1]{Noohi2002FUNDAMENTALGO}
Let $x,x': \operatorname{Spec} k \to \mathcal{X}$ be two geometric points of an algebraic stack \(\mathcal{X}\). A hidden path from $x$ to $x'$, denoted $x \rightsquigarrow x'$, is a natural transformation, i.e a morphism in the category $\operatorname{Fun}(\operatorname{Spec} k ,\mathcal{X})$. The hidden fundamental groupoid of $\mathcal{X}$, denoted by $\Pi_1^h(\mathcal{X})$ is defined as follows:
\begin{center}
   $ \operatorname{Ob}(\Pi^h_1(\mathcal{X})) = \{x : \operatorname{Spec} k \to \mathcal{X}\} $\\
    $\operatorname{Mor}(x,x')= \{x \rightsquigarrow x'\} $
\end{center}
    The \textit{hidden fundamental group} of a geometric point $x: \operatorname{Spec} k \to \mathcal{X}$, denoted by $\pi_1^h(\mathcal{X},x)$, is defined as the automorphism group of the object $x \in \Pi_1^h(\mathcal{X})$.
<!-- end defi-1 -->
<!-- rmk-2: body_md -->
Note that the hidden fundamental groupoid is the groupoid of \(k\)-points of \(\mathcal{X}\) and the hidden fundamental group of a point \(x\in \mathcal{X}(k)\) is the \(k\)-points of the stabilizer group of \(x\), in particular, if \(\mathcal{X}\) has finite inertia, then the hidden fundamental group at any point is finite. In general \(\pi_1^h(\mathcal{X},x)\) is not profinite, for example if \(\mathcal{X}=B\mathbb{G}_m\), then \(\pi_1^h(\mathcal{X},x)=k^*\).

\cite[Definition 4.1]{Noohi2002FUNDAMENTALGO}
Let $\mathcal{X}$ be a connected algebraic stack and let $x: \operatorname{Spec} k \to \mathcal{X}$ be a geometric point.
\begin{enumerate}
    \item A morphism of connected algebraic stacks $f: \mathcal{Y}  \to \mathcal{X}$ is a \textit{covering space} if $f$ is finite representable \'etale map.

    \item The Galois category $\mathbf{C}_{\mathcal{X}}$ and the fundamental functor $F_x$ are defined as follows:
    \begin{enumerate}
        \item  The Galois category $\mathbf{C}_{\mathcal{X}}$ is the  associated $1$-category of the $(2,1)$- category of covering spaces over $\mathcal{X}$ by declaring the $2$-morphisms to be identity.
        \item The functor \begin{equation}
            F_x : \mathbf{C}_{\mathcal{X}} \to \operatorname{Sets}
        \end{equation}
        is defined by 
        \begin{equation}
            F_x(\mathcal{Y}) = \{(y,\phi) \,|\, y :\operatorname{Spec} k \to \mathcal{Y} \, ,\, \phi:  x \rightsquigarrow f(y) \}/ \sim
        \end{equation}
        where  $\sim$ is defined  by 
        \begin{equation}
            (y,\phi) \sim (y',\phi') \,\, \text{if}\, \exists \beta : y \rightsquigarrow y' \, \text{such that} \, f(\beta) \circ \phi = \phi'.
        \end{equation}
         \end{enumerate}
        \item  In a similar way as $\Pi^h_1(\mathcal{X})$, one defines the \textit{fundamental groupoid}, denoted by \(\Pi^1(\mathcal{X})\), of $\mathcal{X}$ as the groupoid with objects being geometric points of $\mathcal{X}$ and morphisms being natural transformations $F_x \to F_{x'}$. The \textit{fundamental group} of the algebraic stack $\mathcal{X}$ at a geometric point $x : \operatorname{Spec} k \to \mathcal{X}$, denoted by $\pi_1^{et}(\mathcal{X},x)$, is the automorphism group of the object $x$ in $\Pi^1(\mathcal{X})$.
        \item Given two points \(x,x'\), a morphism from \(x'\) to \(x\) in \(\Pi_1(\mathcal{X})\) is called a \emph{path} from \(x'\) to \(x\), denoted by \(x'\to x\).
    
    \end{enumerate}

<!-- rmk-3: body_md -->
\begin{enumerate}
\item Similar to the case of schemes, $\pi_1^{et}(\mathcal{X},x)$ is a profinite group and if \(\mathcal{X}\) is connected then \(\Pi_1(\mathcal{X})\) is a connected groupoid, that is, for every pair of points \(x,x'\in \mathcal{X}\) there is a path \(x'\to x\). Any such path induces an isomorphism $\pi_1^{et}(\mathcal{X},x')\cong \pi_1^{et}(\mathcal{X},x)$.
\item There is a natural morphism
\begin{equation}
    \Pi^h_1(\mathcal{X}) \to \Pi_1(\mathcal{X})
\end{equation}
given by identity on the level of objects and on the level morphisms, a hidden path $\gamma: x \rightsquigarrow x'$ is mapped to the natural transformation $F_x \to F_{x'}$, which, when evaluated at covering $\mathcal{Y}\to \mathcal{X} $ is the function
\begin{equation}
    F_x(\mathcal{Y}) \to F_{x'}(\mathcal{Y}) \,\, (y,\phi) \mapsto (y,\gamma^{-1}\phi).
 \end{equation} 
 In particular, to every hidden path there is an associated (potentially trivial) path.
 This induces a homomorphism (of abstract groups) on the fundamental groups 
 \begin{equation}
     \omega_x : \pi^h_1(\mathcal{X},x) \to \pi_1^{et}(\mathcal{X},x).
 \end{equation}
 \item For $\mathcal{X} = BG$ over $k = k^{\operatorname{sep}}$ and $G$ an affine algebraic group of finite type over $k$, Noohi (\cite[Example 4.2]{Noohi2002FUNDAMENTALGO}) shows that $\pi_1^{et}(\mathcal{X},x)= G/G^0$, $\pi^h_1(\mathcal{X},x)= G(k)$ and the map $\omega_x$ is the canonical quotient map.
  \end{enumerate}

\cite[Theorem 11.4]{Noohi2002FUNDAMENTALGO}\label{Theorem: existence of covering space}
    Let $\mathcal{X}$ be a Noetherian algebraic stack. Then there exists a covering space $\mathcal{Y} \to \mathcal{X}$ such that the image of $\omega_y:\pi_1^h(\mathcal{Y},y)\to\pi_1^{et}(\mathcal{Y},y)$ is the identity for all geometric points of $\mathcal{Y}$.
<!-- end teo-4 -->
<!-- rmk-4: body_md -->
The proof of the above theorem uses the fact that a Noetherian algebraic stack admits a stratification by monotonous gerbes \cite[Definition 10.1 and Theorem 11.3]{Noohi2002FUNDAMENTALGO} and \cite[Corollary 10.7]{Noohi2002FUNDAMENTALGO}, which controls the image of the hidden fundamental groups of locally closed substacks that are monotonous gerbes. If we impose the additional conditions that $\omega_x$ is injective for all geometric points and $\mathcal{X}$ is Deligne-Mumford, then $\mathcal{Y}$ is an algebraic space and $\mathcal{X} = [\mathcal{Y}/G]$ for a finite group $G$ (in other words, $\mathcal{X}$ is uniformizable) \cite[Remark after Theorem 11.4]{Noohi2002FUNDAMENTALGO}.

\label{Theorem: finite presentation for proper alg sp}
    Let $X$ be a connected proper algebraic space over an algebraically closed field $k$. Then $\pi_1^{et}(X,x)$ is finitely presented for all geometric points $x \in X$.
<!-- end teo-5 -->
<!-- proof-1: body_md -->
Let \(X'\to X\) be a proper surjective morphism from a proper scheme (which exists by Chow's lemma for algebraic spaces, \cite[tag 088P]{stacks-project}). As the diagonal of \(X\) is  proper and representable by schemes, it follows that \(X''=X'\times_XX'\) and \(X'''=X'\times_X X'\times_X\times X'\) are proper schemes. By \Cref{theorem: propervarietiesfinitepresentation}, the fundamental groups of the connected components of \(X', X''\) and \(X'''\) are all finitely presented, moreover each of their \(\pi_0\)'s are finite. 
    By \cite[Theoreme 4.12]{SGA1}, the category of finite \'etale covers satisfies descent for proper surjective morphisms of finite type, and by \cite[Exp. IX, Corollarie 5.3]{SGA1} the finite presentation of the fundamental groups of the components of \(X',X'',X'''\) implies that fundamental group of \(X\) also is finitely presented.

<!-- teo-6: body_md -->
\label{Theorem: finite presentation for alg stacks}
    Let $\mathcal{X}$ be a connected proper algebraic stack with finite inertia over an algebraically closed field $k$. Then  for any geometric point $x \in \mathcal{X}$, the \'etale fundamental group $\pi_1^{et}(\mathcal{X},x)$ is finitely presented.
<!-- end teo-6 -->
<!-- proof-2: body_md -->
Since \(\mathcal{X}\) is proper over a field, it is in particular Noetherian.
By \Cref{Theorem: existence of covering space}, there exists a covering space $\mathcal{Y} \to \mathcal{X}$ such that the images of $\omega_y:\pi_1^h(\mathcal{Y},y)\to \pi^{et}_1(\mathcal{Y},y)$ are trivial for all geometric points $y \in \mathcal{Y}$. As $\mathcal{Y} \to \mathcal{X}$ is finite \'etale and $\mathcal{X}$ is proper with finite inertia, $\mathcal{Y}$ is proper and has finite inertia. By the Keel-Mori theorem(\cite{keelmoritheorem}),  $\mathcal{Y}$ admits a connected coarse moduli space $p: \mathcal{Y} \to Y$ such that $Y$ is proper.  \\

By \Cref{Theorem: finite presentation for proper alg sp}, $\pi_1^{et}(Y,y)$ is finitely presented for all \(y\in Y(k)\).  As $\omega_y$ is trivial for all \(y\in \mathcal{Y}(k)\), the subgroup $N\subset\pi_1^{et}(\mathcal{Y},y)$ is zero. By \Cref{theorem: relation to cms} we see that $\pi_1^{et}(\mathcal{Y},y) = \pi^{et}_1(Y,y)$ for all geometric points $y$. Thus, $\pi_1^{et}(\mathcal{Y},y)$ is finitely presented. As $\mathcal{Y} \to \mathcal{X}$ is a covering space, we have an injection of fundamental groups
\begin{equation}
    \pi_1^{et}(\mathcal{Y},y) \hookrightarrow \pi_1^{et}(\mathcal{X},x)
\end{equation}

where $\pi_1^{et}(\mathcal{Y},y)$ is an open subgroup of $\pi_1^{et}(\mathcal{X},x)$. Hence by \cite[Proposition 2.3]{obstructionlifttochar0}, $\pi_1^{et}(\mathcal{X},x)$ is finitely presented.
