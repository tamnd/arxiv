<!-- thm-1: body_md -->
\label{thm:convergence}
There exists a function $f:\mathbb{R}_+\times\mathcal Q_1\to\mathbb{R}$
such that for every $t\geq 0$ and $\mathsf{q}\in\mathcal Q_1$,
\begin{equation}
  \lim_{N\to\infty}\bar F_N(t,\mathsf{q})=f(t,\mathsf{q}).
  \label{eq:convergence}
\end{equation}
<!-- end thm-1 -->
<!-- cor-1: body_md -->
\label{cor:critical-point}
For every $t\geq0$ and $\mathsf{q}\in\mathcal Q_2$, there exist
$\mathsf{q}'\in\mathcal Q_2$ and $\mathsf{p}\in\mathcal Q_\infty$ such that
\begin{equation}
  \mathsf{p}=\partial_{\mathsf{q}}\psi(\mathsf{q}'),\qquad
  \mathsf{q}'=\mathsf{q}+t\nabla\xi(\mathsf{p}),\qquad
  f(t,\mathsf{q})=\mathcal J_{t,\mathsf{q}}(\mathsf{q}',\mathsf{p}).
  \label{eq:critical-point}
\end{equation}
The path $\mathsf{q}'-\mathsf{q}$ is bounded. If $\mathsf{q}$ is bounded, so is $\mathsf{q}'$.
<!-- end cor-1 -->
<!-- proof-1: body_md -->
By Theorem~\ref{thm:convergence}, the hypothesis of
\cite[Theorem~1.2]{ChenMourrat2025} holds. Proposition~7.3 of
\cite{ChenMourrat2025}, together with the proof of Theorem~1.2 there,
gives a bounded path $\mathsf{p}\in\mathcal Q_\infty$ such that, with
$\mathsf{q}'=\mathsf{q}+t\nabla\xi(\mathsf{p})$, the first and last identities in
\eqref{eq:critical-point} hold. Since $\nabla\xi$ is increasing on
$S^D_+$ for every $\xi$ as in \eqref{eq:covariance}
\cite{ChenMourrat2025}, the path $\mathsf{q}'$ is increasing. Since $\mathsf{p}$ is
bounded and $\nabla\xi$ is continuous, the path $\mathsf{q}'-\mathsf{q}$ is
bounded. Hence $\mathsf{q}'\in\mathcal Q_2$, and $\mathsf{q}'$ is bounded whenever
$\mathsf{q}$ is.

<!-- prop-1: body_md -->
\label{prop:rate}
Suppose that $P_1$ has finite support in the unit ball. For every
$T,Q<\infty$, there exists a constant $C=C_{P_1,D,\xi,T,Q}>0$ such that
for all $n\geq1$ and $k\geq2$,
\begin{equation}
  \sup_{\substack{0\leq t\leq T,\ \mathsf{q}\in\mathcal Q_\infty\\
                   |\mathsf{q}|_{L^\infty}\leq Q}}
  \bigl|\bar F_{kn}(t,\mathsf{q})-\bar F_n(t,\mathsf{q})\bigr|
  \leq Cn^{-1/4}.
  \label{eq:rate}
\end{equation}

<!-- rem-1: body_md -->
\label{rem:rate}
Letting $k\to\infty$ in \eqref{eq:rate} and using
Theorem~\ref{thm:convergence}, we obtain
$|\bar F_n(t,\mathsf{q})-f(t,\mathsf{q})|\leq Cn^{-1/4}$ in the same range of
$(t,\mathsf{q})$. By the rescaling in Section~\ref{sec:proof-main}, the same
bound holds for every reference measure with finite support, with a constant
depending also on the radius of the support.

<!-- lem-1: body_md -->
\label{lem:restricted-continuity}
Let $\Sigma$ be a Borel set with $P_{kn}(\Sigma)>0$, and let
$\mathsf{q},\mathsf{q}'\in\mathcal Q_\infty$. Then
\begin{equation}
 |\bar F_{\theta,k,n}(t,\mathsf{q};\Sigma)
   -\bar F_{\theta,k,n}(t,\mathsf{q}';\Sigma)|
 \leq\int_0^1\|\mathsf{q}(s)-\mathsf{q}'(s)\|_{\mathrm{op}}\mathop{}\!\mathrm{d} s.
 \label{eq:restricted-path-continuity}
\end{equation}
The same estimate holds for $\bar F_N(t,\cdot\,;\Sigma)$.
<!-- end lem-1 -->
<!-- proof-2: body_md -->
We follow the proof of \cite[Proposition~5.1]{ChenMourrat2025}. For
$\lambda\in[0,1]$, set $\mathsf{q}_\lambda=\mathsf{q}'+\lambda\kappa$ with
$\kappa=\mathsf{q}-\mathsf{q}'$, so that $\mathsf{q}_\lambda\in\mathcal Q_\infty$.
Conditionally on the cascade, the covariance of
$\sqrt2\,W_{kn}^{\mathsf{q}_\lambda}(\alpha)\cdot\sigma$ and
$\sqrt2\,W_{kn}^{\mathsf{q}_\lambda}(\alpha')\cdot\sigma'$ has derivative
$2\kappa(\alpha\wedge\alpha')\cdot\sigma\sigma'^\intercal$ in
$\lambda$, and $-\mathsf{q}_\lambda(1)\cdot\sigma\sigma^\intercal$ is minus
one half of the variance. 
Let $\langle\cdot\rangle_\lambda$ be the Gibbs measure on
$\Sigma\times\mathfrak U$ with path $\mathsf{q}_\lambda$, and
$(\sigma^1,\alpha^1)$, $(\sigma^2,\alpha^2)$ are two independent
replicas. 
Then, the Gaussian integration by parts gives
\[
 \frac{\mathop{}\!\mathrm{d}}{\mathop{}\!\mathrm{d}\lambda}\bar F_{\theta,k,n}(t,\mathsf{q}_\lambda;\Sigma)
 =\frac1{kn}\mathbb{E}\bigl\langle
 \kappa(\alpha^1\wedge\alpha^2)\cdot\sigma^1(\sigma^2)^\intercal
 \bigr\rangle_\lambda.
\]
The columns of $\sigma^1,\sigma^2$ have norm at most $1$, so
the trace norm of $\sigma^1(\sigma^2)^\intercal$ is at most $kn$ and
\[
 \bigl|\kappa(\alpha^1\wedge\alpha^2)\cdot\sigma^1(\sigma^2)^\intercal\bigr|
 \leq kn\,\|\kappa(\alpha^1\wedge\alpha^2)\|_{\mathrm{op}}.
\]
Conditionally on the Hamiltonian, the logarithm of the spin integral
over $\Sigma$ is a Lipschitz function of the field
$W_{kn}^{\mathsf{q}_\lambda}(\alpha)$. The invariance of the cascade in
\cite[Proposition~4.8]{ChenMourrat2025} therefore applies, and
$\alpha^1\wedge\alpha^2$ is uniformly distributed on $[0,1]$ under
$\mathbb{E}\langle\cdot\rangle_\lambda$. Integrating in $\lambda$ gives
\eqref{eq:restricted-path-continuity}.

<!-- lem-4: body_md -->
\label{lem:parisi-representation}
Let $\mu$ be a nonzero finite measure on $\mathbb{R}^{D\times N}$, supported on
configurations with $|\sigma_{\cdot,i}|\leq1$ for every $i$. Let
$\mathsf{q}\in C^1([0,1];S^D_+)$ be increasing. The equation
\begin{equation}
\begin{cases}
 \partial_s\Phi+
 \displaystyle\sum_{i=1}^N\dot\mathsf{q}(s)\cdot\nabla^2_{x_{\cdot,i}}\Phi
 +s\,\dot\mathsf{q}(s)\cdot
       \bigl(\nabla_x\Phi(\nabla_x\Phi)^\intercal\bigr)=0,
 & s<1,\\[1ex]
 \displaystyle
 \Phi(1,x)=\log\int
 e^{x\cdot\sigma-\mathsf{q}(1)\cdot\sigma\sigma^\intercal}\mathop{}\!\mathrm{d}\mu(\sigma)
\end{cases}
\label{eq:parisi-pde}
\end{equation}
has a unique classical solution with bounded spatial gradient.
It is convex in $x$, its spatial derivatives of positive order
are bounded, and
\[
 |\nabla_{x_{\cdot,i}}\Phi(s,x)|\leq1,\qquad 1\leq i\leq N.
\]
Moreover,
\begin{equation}
 \mathbb{E}\Phi(0,\sqrt{2\mathsf{q}(0)}z_0)
 =\mathbb{E}\log\iint
 e^{\sqrt2 W_N^{\mathsf{q}}(\alpha)\cdot\sigma
       -\mathsf{q}(1)\cdot\sigma\sigma^\intercal}
 \mathop{}\!\mathrm{d}\mu(\sigma)\mathop{}\!\mathrm{d}\mathfrak R(\alpha),
 \label{eq:parisi-representation}
\end{equation}
where $z_0$ has independent standard Gaussian entries.
<!-- end lem-4 -->
<!-- proof-3: body_md -->
Apply \cite[Lemma~4.5 and Proposition~4.7]{ChenMourratSimultaneous}
in dimension $DN$, with covariance path $I_N\otimes\mathsf{q}$ and
distribution function $s\mapsto s$. Adding the independent field
$\sqrt{2\mathsf{q}(0)}z_0$ gives \eqref{eq:parisi-representation}, as in
\cite[Lemma~5.4]{ChenMourratSimultaneous}. Differentiating the
representation gives the bound on the gradient in each column, and
H\"older's inequality gives convexity. For uniqueness, the difference
of two solutions satisfies a linear backward equation with bounded
drift, zero terminal value, and at most linear growth, and It\^o's
formula with localization shows that it vanishes.
<!-- end proof-3 -->
<!-- lem-5: body_md -->
\label{lem:martingale}
Let $v$ be a bounded Borel function on $\operatorname{supp}\mu$, and let $\Phi^y$ be
the solution with terminal measure $e^{yv}\mu$. Along the unperturbed
diffusion,
\begin{equation}
\begin{aligned}
 \left.\partial_y\Phi^y(s,X_s)\right|_{y=0}
 &=\mathbb{E}[v(\sigma)\,|\,\mathcal G_s],\\
 \left.\partial_y^2\Phi^y(s,X_s)\right|_{y=0}
 &=s\operatorname{Var}(v(\sigma)\,|\,\mathcal G_s)
   +\mathbb{E}\left[\int_s^1\operatorname{Var}(v(\sigma)\,|\,\mathcal G_r)\mathop{}\!\mathrm{d} r
                \,|\,\mathcal G_s\right].
\end{aligned}
\label{eq:terminal-response}
\end{equation}
In particular, set $m_s=\mathbb{E}[\sigma\,|\,\mathcal G_s]$,
$V_s=\operatorname{Cov}(\sigma\,|\,\mathcal G_s)$, and $K_s=\nabla_x^2\Phi(s,X_s)$.
Then $m_s=\nabla_x\Phi(s,X_s)$ and
\begin{equation}
 K_s=sV_s+
 \mathbb{E}\left[\int_s^1 V_r\mathop{}\!\mathrm{d} r\,|\,\mathcal G_s\right].
 \label{eq:covariance-identity}
\end{equation}
If $\dot\mathsf{q}\succeq\varepsilon I_D$ for some $\varepsilon>0$, then
every deterministic orthogonal projection $\Pi$ on $\mathbb{R}^{DN}$ satisfies
\begin{equation}
 \int_0^1\mathbb{E}\operatorname{tr}(\Pi K_s^2)\mathop{}\!\mathrm{d} s
 \leq\frac{\mathbb{E}|\Pi\sigma|^2}{2\varepsilon}.
 \label{eq:parisi-covariance-energy}
\end{equation}
<!-- end lem-5 -->
<!-- proof-4: body_md -->
Put $\chi_1=\left.\partial_y\Phi^y\right|_{y=0}$ and
$\chi_2=\left.\partial_y^2\Phi^y\right|_{y=0}$, and differentiate the
PDE in $y$, keeping $X$ unperturbed. By It\^o's formula,
\[
 \mathop{}\!\mathrm{d}\chi_1(s,X_s)
 =\sum_{i=1}^N\nabla_{x_{\cdot,i}}\chi_1(s,X_s)\cdot
       \sqrt{2\dot\mathsf{q}(s)}\mathop{}\!\mathrm{d} B_{\cdot,i,s},
\]
while the drift of $\chi_2(s,X_s)$ is
\[
 -2s\sum_{i=1}^N
 |\sqrt{\dot\mathsf{q}(s)}\nabla_{x_{\cdot,i}}\chi_1(s,X_s)|^2.
\]
The terminal values are the conditional mean and variance of
$v(\sigma)$ given $\mathcal G_1$. Hence $\chi_1(s,X_s)$ is the
conditional mean of $v(\sigma)$ given $\mathcal G_s$. It follows that
the drift of $\operatorname{Var}(v(\sigma)\,|\,\mathcal G_s)$ is the same expression
without the factor $s$, and therefore that
\[
 \chi_2(s,X_s)-s\operatorname{Var}(v(\sigma)\,|\,\mathcal G_s)
 +\int_0^s\operatorname{Var}(v(\sigma)\,|\,\mathcal G_r)\mathop{}\!\mathrm{d} r
\]
is a martingale. Its terminal value gives
\eqref{eq:terminal-response}. The bounds on the derivatives of $\Phi^y$
justify the differentiation in $y$ and the use of It\^o's formula.

For $v(\sigma)=a\cdot\sigma$, translation in $x$ gives
$\Phi^y(s,x)=\Phi(s,x+ya)$. Varying $a$ yields
\eqref{eq:covariance-identity} and
\begin{equation}
 \mathop{}\!\mathrm{d} m_s=K_s\bigl(I_N\otimes\sqrt{2\dot\mathsf{q}(s)}\bigr)\mathop{}\!\mathrm{d} B_s.
 \label{eq:mean-martingale}
\end{equation}
By It\^o's isometry and Jensen's inequality,
\[
 2\varepsilon\int_0^1\mathbb{E}\operatorname{tr}(\Pi K_s^2)\mathop{}\!\mathrm{d} s
 \leq\mathbb{E}|\Pi m_1|^2-\mathbb{E}|\Pi m_0|^2
 \leq\mathbb{E}|\Pi\sigma|^2,
\]
as desired.

<!-- cor-2: body_md -->
\label{cor:correlation-identity}
For every Borel function $v$ bounded on $\operatorname{supp}\mu$,
\begin{equation}
 \mathbb{E}\langle v(\sigma^1)v(\sigma^2)\rangle
 =\int_0^1\mathbb{E}\left[\mathbb{E}[v(\sigma)\,|\,\mathcal G_s]^2\right]\mathop{}\!\mathrm{d} s.
 \label{eq:scalar-replicas}
\end{equation}
<!-- end cor-2 -->
<!-- proof-5: body_md -->
Differentiate \eqref{eq:parisi-representation} for the terminal
measure $e^{yv}\mu$. By \eqref{eq:terminal-response} at $s=0$,
\[
 \mathbb{E}\langle v(\sigma)\rangle=\mathbb{E} v(\sigma),\qquad
 \mathbb{E}\bigl[\langle v(\sigma)^2\rangle-\langle v(\sigma)\rangle^2\bigr]
 =\int_0^1\mathbb{E}\operatorname{Var}(v(\sigma)\,|\,\mathcal G_s)\mathop{}\!\mathrm{d} s.
\]
Apply the first identity to $v^2$ and expand the conditional variance
in the second. The terms involving one replica cancel.

\label{lem:replicas}
We have

 =\int_0^1\mathbb{E}_\theta|\mathsf{R}_s^{(1)}-\mathsf{R}_s^{(2)}|^2\mathop{}\!\mathrm{d} s.

<!-- prop-3: body_md -->
\label{prop:block-overlap}
Suppose that $P_1$ has finite support in the unit ball of $\mathbb{R}^D$. Fix
$\varepsilon>0$, and suppose that $\mathsf{q}\in C^1([0,1];S^D_+)$ satisfies
$\dot\mathsf{q}(s)\succeq\varepsilon I_D$ for every $s\in[0,1]$. There exists
$C_D>0$, depending only on $D$, such that for all $n\geq1$, $k\geq2$,
$t\geq0$, and $\theta\in[0,1]$,
\begin{equation}
 \int_0^1\mathbb{E}_\theta|\mathsf{R}_s^{(1)}-\mathsf{R}_s^{(2)}|^2\mathop{}\!\mathrm{d} s
 \leq C_D(n\varepsilon)^{-1/3}.
 \label{eq:block-overlap-bound}
\end{equation}
