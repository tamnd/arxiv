<!-- lemma-1: body_md -->
\label{lem:half}
		Let $r\in(0,\pi/2)$, $R\in\partial\overline{{\ensuremath{\mathcal{B}}}}_r(R_c)$,
		where \(\partial\) denotes the boundary,
		$Q\in\overline{{\ensuremath{\mathcal{B}}}}_r(R_c)$ with $L:=d(R,Q)>0$, and let
		$\hat e_{c},\hat e_{Q}$ be the unit body-coordinate directions at $R$
		toward $R_c$ and $Q$.
		Then
		$\hat e_c^\top\hat e_Q\geq
		\cot\tfrac{r}{2}\tan\tfrac{L}{4}>0$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
$\operatorname{SO}(3)$ with the bi-invariant metric has \emph{constant}
		sectional curvature $1/4$, so within the injectivity region the
		exact spherical law of cosines of the model sphere (radius $2$)
		applies~\cite{docarmo1992riemannian}: with $c:=d(R_c,Q)$,
		$\cos\tfrac{c}{2}=\cos\tfrac{r}{2}\cos\tfrac{L}{2}
		+\sin\tfrac{r}{2}\sin\tfrac{L}{2}\,
		(\hat e_c^\top\hat e_Q)$.
		Solving and using $c\leq r$,
		\(
		\hat e_c^\top\hat e_Q
		\geq\frac{\cos\tfrac{r}{2}
			\bigl(1-\cos\tfrac{L}{2}\bigr)}
		{\sin\tfrac{r}{2}\sin\tfrac{L}{2}}
		=\cot\tfrac{r}{2}\tan\tfrac{L}{4},
		\)
		by $(1-\cos x)/\sin x=\tan(x/2)$; positivity follows from
		$0<L\leq2r<\pi$.
		Equality holds precisely when $c=r$.
<!-- end proof-1 -->
<!-- assumption-1: body_md -->
\label{ass:tree}
		${\ensuremath{\mathcal{G}}}$ is a tree: connected with $|{\ensuremath{\mathcal{E}}}|=n-1$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:conv}
		On ${\ensuremath{\mathcal{C}}}$: (a)~$F$ is geodesically
		$\mu_F$-strongly convex, with $\mu_F>0$;
		(b)~$R^*:=\operatornamewithlimits{argmin}_{R\in{\ensuremath{\mathcal{C}}}}F(R)$ lies in the interior of
		${\ensuremath{\mathcal{C}}}$; (c)~$R_i(0)\in{\ensuremath{\mathcal{C}}}$ for every $i$. Fix any $M>0$
		satisfying $M\geq\max_{i\in{\ensuremath{\mathcal{V}}}}\max_{R\in{\ensuremath{\mathcal{C}}}}
		\|\nabla_Rf_i(R)\|$; such an $M$ exists since each $f_i$ is
		$C^2$ on the open set ${\ensuremath{\mathcal{O}}}\supset{\ensuremath{\mathcal{C}}}$ and ${\ensuremath{\mathcal{C}}}$ is
		compact.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:rad}
		$\hat e_{i,c}^\top\nabla_{R_i}f_i\leq 0$ for every
		$R_i\in\partial{\ensuremath{\mathcal{C}}}$ and every $i$, where
		$\hat e_{i,c}$ is the unit direction from $R_i$ toward $R_c$.
<!-- end assumption-3 -->
<!-- remark-1: body_md -->
\label{rem:class}
		For weighted squared distances
		$f_i=\tfrac{k_i}{2}d^2(\cdot,R_i^*)$ with $k_i>0$ and targets
		$R_i^*\in\overline{{\ensuremath{\mathcal{B}}}}_{r_0}(R_c)$, $r_0<\rho$, all
			cost-related conditions in Assumptions~\ref{ass:conv}
			and~\ref{ass:rad} are verified constructively; indeed,
			$\rho+r_0<\pi$, so the costs are $C^2$ on a common open
			neighborhood of ${\ensuremath{\mathcal{C}}}$.
		The distance identity~\eqref{eq:deriv} with $R_j\equiv R_i^*$ static gives
		$\nabla_{R}f_i=-k_i\,e_{i,R_i^*}$; Assumption~\ref{ass:rad} then
		follows from Lemma~\ref{lem:half}, and
		$\|\nabla f_i\|=k_i\,d(R,R_i^*)\leq k_i(\rho+r_0)=:M_i$ gives
		$M=\max_i M_i$.
		The classical Jacobi-field computation on the constant-curvature
		space $\kappa=1/4$~\cite{docarmo1992riemannian,
			tron2013riemannian} gives the Hessian spectrum of
		$\tfrac12 d^2(\cdot,p)$ as $\{1,\ \tfrac{r}{2}\cot\tfrac{r}{2}\}$
		at distance $r$, whence one may take
			$\mu_F=(\sum_i k_i)\tfrac{\rho+r_0}{2}
			\cot\tfrac{\rho+r_0}{2}>0$.
		Finally, interiority~(b) holds automatically, by a nested-ball
		argument: for any $R\in{\ensuremath{\mathcal{C}}}$ with $r:=d(R,R_c)>r_0$, every target lies
		strictly inside $\overline{{\ensuremath{\mathcal{B}}}}_r(R_c)$ (at distance
		$\leq r_0<r$ from $R_c$), so Lemma~\ref{lem:half} applied to
		the ball of radius $r$ gives
		$\hat e_c^\top\hat e_{i,R_i^*}\geq
		\cot\tfrac{r}{2}\tan\tfrac{L_i}{4}>0$ for every target with
		$L_i=d(R,R_i^*)>0$; hence
		$\hat e_c^\top\nabla_R F=-\sum_ik_iL_i\,
		\hat e_c^\top\hat e_{i,R_i^*}<0$, $F$ strictly decreases
		toward $R_c$ at every point outside
		$\overline{{\ensuremath{\mathcal{B}}}}_{r_0}(R_c)$, and its minimizer over
		${\ensuremath{\mathcal{C}}}$ satisfies $d(R^*,R_c)\leq r_0<\rho$.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
\label{rem:necessity}
		Assumption~\ref{ass:rad} is the natural hypothesis for an
		operating ball not centered at the minimizer, and strong
		convexity does not imply it. By~\eqref{eq:sc} the descent
		direction points toward $R^*$, so geodesic balls centered at
		$R^*$ are invariant under the gradient flow; for an off-center
		ball, anisotropy alone breaks invariance. Already in ${\ensuremath{\mathbb{R}}}^2$,
		for $f(x)=\tfrac12(x-x^*)^\top\operatorname{diag}(1,10)(x-x^*)$
		on the ball of radius $\rho$ about $c$ with

\label{thm:main}
		Suppose Assumptions~\ref{ass:tree}--\ref{ass:rad} hold,
		and let $\alpha,\gamma>0$. For the protocol~\eqref{eq:law},
		the following statements hold.
		\begin{enumerate}
			\item[(i)] \emph{Strong invariance and global existence.}
			From every initial condition $x(0)\in{\ensuremath{\mathcal{C}}}^n$, a Filippov
			solution exists. Every maximal solution from this initial
			condition is defined on $[0,\infty)$ and satisfies
			$x(t)\in{\ensuremath{\mathcal{C}}}^n$ for all $t\geq0$.
			\item[(ii)] \emph{Exact finite-time consensus.}
			If, additionally,
			\begin{equation}
				\frac{\alpha}{\gamma}>\frac{nM}{2},
				\label{eq:gain}
			\end{equation}
			then every such solution reaches and maintains exact
			consensus:
			\[
			R_i(t)=R_j(t),\qquad
			\forall i,j\in{\ensuremath{\mathcal{V}}},\quad t\geq T^*,
			\]
			for a settling time satisfying
			\begin{equation}
				T^*\leq
				\frac{nW(0)}{2(2\alpha-\gamma nM)}.
				\label{eq:settle}
			\end{equation}
			\item[(iii)] \emph{Aggregate gradient flow and exponential
				convergence.}
			Under~\eqref{eq:gain}, let $\bar R(t):=R_i(t)$ for
			$t\geq T^*$. Then
			\begin{equation}
				\dot{\bar R}
				=
				\bar R\Bigl(-\frac{\gamma}{n}
				\nabla_{\bar R}F\Bigr)^{\!\wedge}
				\quad\text{for a.e. }t\geq T^*,
				\label{eq:sliding}
			\end{equation}
			and
			\begin{equation}
				d(\bar R(t),R^*)
				\leq
				d(\bar R(T^*),R^*)\,
				e^{-\frac{\gamma\mu_F}{n}(t-T^*)},
				\quad t\geq T^*.
				\label{eq:rate}
			\end{equation}
			Here $T^*$ and $\bar R(T^*)$ may depend on the particular
			Filippov solution; for each solution, the common trajectory
			on $[T^*,\infty)$ is uniquely determined by $\bar R(T^*)$.
		\end{enumerate}

<!-- lemma-2: body_md -->
\label{lem:filippov}
		Let $x\in{\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$, with all neighborhoods below taken
		inside ${\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$.
		(i)~\emph{Antisymmetric containment:}
		 
	\begin{align}
		\begin{aligned}
			
			{\ensuremath{\mathcal{K}}}[h](x)\subseteq A(x):=\Bigl\{
			\Bigl(\alpha\!\sum_{j\in{\ensuremath{\mathcal{N}}}_i}\!u_{ij}
			-\gamma\nabla_{R_i}f_i\Bigr)_{\!i}\Bigr\},
			\label{eq:containment}
		
		\end{aligned}
	\end{align}

		where the family $u$ ranges over all assignments with
		$u_{ij}\in\overline{{\ensuremath{\mathbb{B}}}}^3$, $u_{ji}=-u_{ij}$ for every edge,
		and $u_{ij}=\mathrm{sgn}(e_{ij})$ whenever $e_{ij}\neq 0$.
		(ii)~\emph{Nominal membership:} the nominal value
		$h(x)=\bigl(\alpha s_i-\gamma\nabla_{R_i}f_i\bigr)_i$ (with
		$\mathrm{sgn}(0)=0$) belongs to ${\ensuremath{\mathcal{K}}}[h](x)$.
		(iii)~\emph{Isolated-pair membership:} if the zero-error cluster
		containing a zero edge $(i_0,j_0)$ is exactly the pair
		$\{i_0,j_0\}$, then for any unit $u\in{\ensuremath{\mathbb{R}}}^3$ the value obtained
		from $h(x)$ by setting $u_{i_0j_0}=u=-u_{j_0i_0}$ on that edge,
		with $u_{ij}=0$ on every other zero edge, also belongs to
		${\ensuremath{\mathcal{K}}}[h](x)$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
The set $A(x)$ is the image of a compact convex product of balls
		and points under an affine map, hence compact and convex.
		Shrinking $\delta$ so that every edge with $e_{ij}(x)\neq0$
		stays nonzero on ${\ensuremath{\mathcal{B}}}_\delta(x)$, each value $h(y)$,
		$y\in{\ensuremath{\mathcal{B}}}_\delta(x)$, has the form~\eqref{eq:containment}
		with the antisymmetric choice $u_{ij}=\mathrm{sgn}(e_{ij}(y))$ and
		gradients evaluated at $y$. Replacing $\mathrm{sgn}(e_{ij}(y))$ by
		$\mathrm{sgn}(e_{ij}(x))$ on the edges with $e_{ij}(x)\neq0$ and
		$\nabla f_i(y)$ by $\nabla f_i(x)$ gives an element of $A(x)$
		(the zero-edge slots are free), with an error that tends to
		$0$ uniformly on ${\ensuremath{\mathcal{B}}}_\delta(x)$ as $\delta\to0$, by
		continuity of the nonzero-edge directions and of the
		gradients. Hence the essential values of $h$ near $x$ lie in a
		shrinking neighborhood of the closed convex set $A(x)$; taking
		closed convex hulls and intersecting over $\delta$ gives
		${\ensuremath{\mathcal{K}}}[h](x)\subseteq A(x)$, proving~(i).

		For~(ii), fix $\delta_1,\ldots,\delta_n\in{\ensuremath{\mathbb{R}}}^3$ with
		$\delta_j\neq\delta_i$ for every zero edge $(i,j)$ (generic),
		and for $\epsilon>0$ consider the \emph{full-dimensional}
		perturbation families
		$x^{\pm}(\epsilon,w):
		R_i\mapsto R_i\exp\bigl((\pm\epsilon\delta_i+w_i)^\wedge\bigr)$
		over $\left\lVertw\right\rVert:=\max_i\left\lVertw_i\right\rVert\leq\epsilon^2$.
		A fixed curve would be a null set, possibly inside the excluded
		set; the family is not: the image of the open $w$-ball has
		positive Riemannian measure, so for \emph{every} null set
		${\ensuremath{\mathcal{S}}}$ and every $\epsilon$ there are parameters $w^{\pm}$
		with $x^{\pm}(\epsilon,w^{\pm})\notin{\ensuremath{\mathcal{S}}}$.
		Uniformly over $\left\lVertw\right\rVert\leq\epsilon^2$ and over the finitely
		many zero edges, the BCH expansion
		gives $e_{ij}^{\pm}=\pm\epsilon(\delta_j-\delta_i)
		+O(\epsilon^2)$ on zero edges, while every edge that is nonzero
		at $x$ remains nonzero for
		sufficiently small $\epsilon$, and its signum direction converges
		to its value at $x$; hence
		$h\bigl(x^{\pm}(\epsilon,w^{\pm})\bigr)\to V^{\pm}$ as
		$\epsilon\to0$, where $V^{\pm}$ are the values of the
		right-hand side of~\eqref{eq:containment} with
		$u_{ij}=\pm\widehat{(\delta_j-\delta_i)}
		=\pm(\delta_j-\delta_i)/\left\lVert\delta_j-\delta_i\right\rVert$ on zero
		edges and
		$\mathrm{sgn}(e_{ij})$ on the others.
		Thus, for every $\delta>0$ and every null ${\ensuremath{\mathcal{S}}}$, both
		$V^{\pm}$ lie in the closure of
		$h({\ensuremath{\mathcal{B}}}_\delta(x)\setminus{\ensuremath{\mathcal{S}}})$, hence in its closed
		convex hull, and so does
		$\tfrac12(V^++V^-)=h(x)$; intersecting over $\delta$ and
		${\ensuremath{\mathcal{S}}}$ gives $h(x)\in{\ensuremath{\mathcal{K}}}[h](x)$, proving~(ii).

		For~(iii), fix $s>0$ small and pass to $x_s$ by
		$R_{j_0}\mapsto R_{j_0}\exp(su^\wedge)$; since
		$R_{i_0}=R_{j_0}$ at $x$, the chosen edge carries the exact
		error $e_{i_0j_0}(x_s)=su$. Every other edge incident to $i_0$
		or $j_0$ is nonzero by the cluster hypothesis, remains nonzero
		for sufficiently small $s$,
		and has a signum direction converging to its value at $x$; the
		remaining zero edges are untouched.
		The argument for~(ii) applied at $x_s$, with the chosen edge
		now sign-pinned to $u$, places the corresponding value in
		${\ensuremath{\mathcal{K}}}[h](x_s)$; letting $s\to 0$ and using upper
		semicontinuity of ${\ensuremath{\mathcal{K}}}[h]$ with closed values yields
		membership in ${\ensuremath{\mathcal{K}}}[h](x)$.

\label{lem:clusters}
		Let $x$ be a Filippov solution in ${\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$.
		For almost every time $t$, all agents in each cluster
		$C\in{\ensuremath{\mathcal{P}}}_0(x(t))$ have a common body angular velocity
		$\bar\omega_C(t)$ satisfying
		 
	\begin{align}
		\begin{aligned}
			
			|C|\,\bar\omega_C=\alpha S_C-\gamma G_C,
			\ \
			S_C:=\!\sum_{i\in C}\!s_i,\
			G_C:=\!\sum_{i\in C}\!\nabla_{R_i}f_i,
			\label{eq:cluster}
		
		\end{aligned}
	\end{align}

		where $|C|$ denotes the number of agents in $C$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
First, let ${\ensuremath{\mathcal{A}}}\subseteq[0,T]$ be measurable with
		$R_i(t)=R_j(t)$ on ${\ensuremath{\mathcal{A}}}$ for a fixed pair $(i,j)$.
		The function $G:=R_i-R_j$ is absolutely continuous and vanishes
		on ${\ensuremath{\mathcal{A}}}$; at a.e.\ density point of ${\ensuremath{\mathcal{A}}}$ that is also a
		differentiability point, $\dot G=0$, i.e.\
		$R_i\omega_i^\wedge=R_j\omega_j^\wedge$ with $R_i=R_j$
		invertible, so $\omega_i=\omega_j$ a.e.\ on ${\ensuremath{\mathcal{A}}}$.
		Second, the zero-edge pattern
		$\{(i,j)\in{\ensuremath{\mathcal{E}}}:e_{ij}(t)=0\}$ takes values in the finite set
		$2^{{\ensuremath{\mathcal{E}}}}$; for each pattern the corresponding time set is
		Borel, by continuity of $t\mapsto e_{ij}(t)$, and the cluster
		partition is a deterministic function of the pattern
		(Definition~\ref{def:clusters}).
		Applying the first step to every zero edge of every pattern and
		propagating equality along paths within a cluster yields a null
		set outside of which all members of each cluster
		$C\in{\ensuremath{\mathcal{P}}}_0(x(t))$ share a velocity $\bar\omega_C(t)$.
		Third, by Lemma~\ref{lem:filippov}(i), a.e.\ velocities take the
		antisymmetric form~\eqref{eq:containment}.
		Summing over $i\in C$: intra-cluster edge contributions cancel
		in antisymmetric pairs; cross-cluster edges have $e_{ij}\neq0$,
		so their $u_{ij}$ are pinned to $\mathrm{sgn}(e_{ij})$ and sum to
		$S_C$ (intra-cluster edges contribute $\mathrm{sgn}(0)=0$ to $S_C$ as
		well).
		Replacing each $\omega_i$, $i\in C$, by $\bar\omega_C$
		gives~\eqref{eq:cluster}.

<!-- proof-6: body_md -->
\emph{Step 1: Strong invariance and global existence.}
		Let $x(0)\in{\ensuremath{\mathcal{C}}}^n$. Since ${\ensuremath{\mathcal{C}}}^n\subset{\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$,
		a solution exists by the existence statement of
		Section~\ref{sec:protocol}; let $x$ be any solution with
		maximal interval $[0,T_{\max})$.
		Pick $\delta>0$ with $\rho+\delta<\pi/2$ and
		$\overline{{\ensuremath{\mathcal{B}}}}_{\rho+\delta}(R_c)\subset{\ensuremath{\mathcal{O}}}$, so Lemmas~\ref{lem:filippov} and~\ref{lem:clusters} apply
		verbatim while
		$x(t)\in\overline{{\ensuremath{\mathcal{B}}}}_{\rho+\delta}(R_c)^n
		\subset{\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$.
		On the compact shell
		$\{\rho\leq d(R,R_c)\leq\rho+\delta\}$ the functions
		$g_i(R):=\hat e_{c}(R)^\top\nabla_{R}f_i(R)$ are $C^1$, hence
		$L$-Lipschitz for some common $L$; writing $\Pi_\rho(R)$ for
		the radial projection to $\partial{\ensuremath{\mathcal{C}}}$ along the geodesic
		through $R_c$ and using $g_i(\Pi_\rho(R))\leq0$
		(Assumption~\ref{ass:rad}) with
		$d(R,\Pi_\rho(R))=d(R,R_c)-\rho$,
		 
	\begin{align}
		\begin{aligned}
			
			g_i(R)\leq L\,\bigl(d(R,R_c)-\rho\bigr)
			\quad\text{on the shell.}
			\label{eq:shellbound}
		
		\end{aligned}
	\end{align}

		Set $r_i(t):=d(R_i(t),R_c)$, $r(t):=\max_i r_i(t)$, and
		$\tau:=\inf\{t\in[0,T_{\max}):r(t)=\rho+\delta\}$, with
		$\inf\varnothing:=T_{\max}$.
		On $[0,\tau)$ the functions $r_i$ and $r$ are absolutely
		continuous. Since each $r_i-r_j$ has derivative zero almost
		everywhere on its zero set, the active derivatives coincide
		almost everywhere, and $\dot r=\dot r_i$ for every active
		index outside a null set.
		At such a $t$ with $r(t)>\rho$, let $C$ be the cluster of an
		active agent, with common attitude $R_C$, $d(R_C,R_c)=r(t)$.
		Every neighbor lies in $\overline{{\ensuremath{\mathcal{B}}}}_{r(t)}(R_c)$ and
		every edge leaving $C$ carries a nonzero error, so
		Lemma~\ref{lem:half} at radius $r(t)<\rho+\delta<\pi/2$ gives
		$\hat e_{C,c}^\top\hat e_{ij}\geq0$ and hence
		$\hat e_{C,c}^\top S_C\geq0$; internal selections cancel by
		antisymmetry (Lemma~\ref{lem:clusters}).
		Therefore, by the distance identity~\eqref{eq:deriv}
		and~\eqref{eq:shellbound},
		 
	\begin{align}
		\begin{aligned}
			\dot r = -\hat e_{C,c}^\top\bar\omega_C
			&= -\tfrac{\alpha}{|C|}\hat e_{C,c}^\top S_C
			+\tfrac{\gamma}{|C|}\sum_{i\in C}g_i(R_C)\\
			&\leq\gamma L\,\bigl(r(t)-\rho\bigr).
		\end{aligned}\notag
	\end{align}

		Hence $y:=[r-\rho]_+$ satisfies $\dot y\leq\gamma Ly$ a.e.\ on
		$[0,\tau)$ (on $\{y=0\}$, $\dot y=0$ a.e.), and $y(0)=0$, so
		Gr\"onwall's inequality gives $y\equiv0$ on $[0,\tau)$.
		If $\tau<T_{\max}$, continuity would give both
		$r(\tau)=\rho+\delta$ and $r(\tau)\leq\rho$, a contradiction;
		thus $\tau=T_{\max}$ and $x(t)\in{\ensuremath{\mathcal{C}}}^n$ on its maximal
		interval.
		Global extension follows since ${\ensuremath{\mathcal{C}}}^n$ is compact with
		positive distance to $\partial{\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$ and velocities are
		bounded there. This proves~(i).

		\emph{Step 2: Exact finite-time consensus.}
		Assume~\eqref{eq:gain} and let $x$ be a solution with
		$x(0)\in{\ensuremath{\mathcal{C}}}^n$; by Step~1 it is defined on $[0,\infty)$
		with values in ${\ensuremath{\mathcal{C}}}^n$.
		Then $w(t):=W(x(t))$ is absolutely continuous ($W$ is locally
		Lipschitz on ${\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$ and $x$ is absolutely continuous
		with values in the compact ${\ensuremath{\mathcal{C}}}^n$).
		Fix a.e.\ $t$.
		On edges with $e_{ij}(t)\neq0$,
		the distance identity~\eqref{eq:deriv} gives
		$\tfrac{d}{dt}\|e_{ij}\|=\hat e_{ij}^\top(\omega_j-\omega_i)$;
		on the time set where an edge error vanishes, the absolutely
		continuous function $\|e_{ij}(\cdot)\|$ is zero, so its
		derivative vanishes a.e.\ there.
		Summing over edges and regrouping endpoint contributions,
		 
	\begin{align}
		\begin{aligned}
			
			\dot w
			= -\sum_{i\in{\ensuremath{\mathcal{V}}}}s_i^\top\omega_i
			= -\!\!\sum_{C\in{{\ensuremath{\mathcal{P}}}_0(x(t))}}\!\! S_C^\top\bar\omega_C
			\quad\text{a.e.},
		
		\end{aligned}\notag
	\end{align}

		using Lemma~\ref{lem:clusters} and $\sum_{i\in C}s_i=S_C$.
		Substituting~\eqref{eq:cluster},
		$\|G_C\|\leq|C|M$, and the Cauchy--Schwarz bound
		$\sum_C\|S_C\|^2/|C|\geq\Sigma^2/n$ with
		$\Sigma:=\sum_C\|S_C\|$,
		 
	\begin{align}
		\begin{aligned}
			
			\dot w
			\leq -\frac{\alpha}{n}\Sigma^2+\gamma M\Sigma
			=: \psi(\Sigma)
			\quad\text{a.e.\ on }\{w>0\}.
		
		\end{aligned}\notag
	\end{align}

		The concave parabola $\psi$ has vertex at
		$\Sigma=n\gamma M/(2\alpha)<1<2$
		by~\eqref{eq:gain}, so $\psi$ is decreasing on $[2,\infty)$;
		Lemma~\ref{lem:tree_bound} gives $\Sigma\geq2$ on $\{w>0\}$,
		hence
		$\dot w\leq\psi(2)=-\tfrac{2}{n}(2\alpha-\gamma nM)=:-c<0$
		a.e.\ on $\{w>0\}$.
		Define $T^*:=\inf\{t\geq0:w(t)=0\}$. If $w(0)>0$, then
		$w>0$ on $[0,T^*)$, and integration gives
		$0\leq w(t)\leq w(0)-ct$ for every $t<T^*$; hence
		$T^*\leq w(0)/c$. If $w(0)=0$, then $T^*=0$.
		Moreover, $w$ cannot leave zero afterwards: any excursion
		interval $(t_2,t_1]$ with $w(t_2)=0$ and $w>0$ on $(t_2,t_1]$
		would give $w(t_1)\leq-c(t_1-t_2)<0$.
		Finally, $w=0$ forces every edge error to vanish, and
		connectivity gives consensus for all $t\geq T^*$; substituting
		$c$ yields~\eqref{eq:settle}. This proves~(ii).

		\emph{Step 3: Post-consensus optimization.}
		By Step~2, agreement persists: $R_i(t)=R_j(t)$ for all $i,j$
		and all $t\geq T^*$, so $\bar R(t):=R_i(t)$ is well defined,
		and for $t\geq T^*$ the trajectory evolves in the consensus set
		${\ensuremath{\mathcal{M}}}=\{(R,\ldots,R):R\in{\ensuremath{\mathcal{C}}}\}$, a submanifold with
		boundary, so its a.e.\ body velocities are of the common form
		$(\bar\omega,\ldots,\bar\omega)$.
		By Lemma~\ref{lem:filippov}(i), the same velocities also admit
		the antisymmetric form~\eqref{eq:containment}; summing over all
		agents, every edge term cancels in antisymmetric pairs and
		$n\bar\omega=-\gamma\nabla_{\bar R}F$.
		This proves~\eqref{eq:sliding}; its right-hand side is
		locally Lipschitz, so~\eqref{eq:sliding} uniquely determines the
		common trajectory from $\bar R(T^*)$. The sliding velocity is
		radially admissible at $\partial{\ensuremath{\mathcal{C}}}$, since
		$\hat e_c^\top(-\tfrac{\gamma}{n}\nabla F)
		=-\tfrac{\gamma}{n}\sum_i\hat e_c^\top\nabla f_i\geq0$ by
		Assumption~\ref{ass:rad}.
		For the rate, set $V_2:=\tfrac12 d^2(\bar R,R^*)$; by the
		distance identity~\eqref{eq:deriv} with the static endpoint
		$R^*$ and $e_*:=(\log(\bar R^\top R^*))^\vee$,
		$\dot V_2=-e_*^\top\bar\omega
		=\tfrac{\gamma}{n}\langle\nabla_{\bar R}F,e_*\rangle
		\leq-\tfrac{\gamma}{n}\mu_F d^2=-\tfrac{2\gamma\mu_F}{n}V_2$
		a.e., using the interior-minimizer
		inequality~\eqref{eq:sc} (Assumption~\ref{ass:conv}(a)--(b)).
		Gr\"onwall's inequality and $d=\sqrt{2V_2}$
		give~\eqref{eq:rate}. This proves~(iii).

		\end{aligned}
	\end{align}

		Set $r_i(t):=d(R_i(t),R_c)$, $r(t):=\max_i r_i(t)$, and
		$\tau:=\inf\{t\in[0,T_{\max}):r(t)=\rho+\delta\}$, with
		$\inf\varnothing:=T_{\max}$.
		On $[0,\tau)$ the functions $r_i$ and $r$ are absolutely
		continuous. Since each $r_i-r_j$ has derivative zero almost
		everywhere on its zero set, the active derivatives coincide
		almost everywhere, and $\dot r=\dot r_i$ for every active
		index outside a null set.
		At such a $t$ with $r(t)>\rho$, let $C$ be the cluster of an
		active agent, with common attitude $R_C$, $d(R_C,R_c)=r(t)$.
		Every neighbor lies in $\overline{{\ensuremath{\mathcal{B}}}}_{r(t)}(R_c)$ and
		every edge leaving $C$ carries a nonzero error, so
		Lemma~\ref{lem:half} at radius $r(t)<\rho+\delta<\pi/2$ gives
		$\hat e_{C,c}^\top\hat e_{ij}\geq0$ and hence
		$\hat e_{C,c}^\top S_C\geq0$; internal selections cancel by
		antisymmetry (Lemma~\ref{lem:clusters}).
		Therefore, by the distance identity~\eqref{eq:deriv}
		and~\eqref{eq:shellbound},
		 
	\begin{align}
		\begin{aligned}
			\dot r = -\hat e_{C,c}^\top\bar\omega_C
			&= -\tfrac{\alpha}{|C|}\hat e_{C,c}^\top S_C
			+\tfrac{\gamma}{|C|}\sum_{i\in C}g_i(R_C)\\
			&\leq\gamma L\,\bigl(r(t)-\rho\bigr).
		\end{aligned}\notag
	\end{align}

		Hence $y:=[r-\rho]_+$ satisfies $\dot y\leq\gamma Ly$ a.e.\ on
		$[0,\tau)$ (on $\{y=0\}$, $\dot y=0$ a.e.), and $y(0)=0$, so
		Gr\"onwall's inequality gives $y\equiv0$ on $[0,\tau)$.
		If $\tau<T_{\max}$, continuity would give both
		$r(\tau)=\rho+\delta$ and $r(\tau)\leq\rho$, a contradiction;
		thus $\tau=T_{\max}$ and $x(t)\in{\ensuremath{\mathcal{C}}}^n$ on its maximal
		interval.
		Global extension follows since ${\ensuremath{\mathcal{C}}}^n$ is compact with
		positive distance to $\partial{\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$ and velocities are
		bounded there. This proves~(i).

Assume~\eqref{eq:gain} and let $x$ be a solution with
		$x(0)\in{\ensuremath{\mathcal{C}}}^n$; by Step~1 it is defined on $[0,\infty)$
		with values in ${\ensuremath{\mathcal{C}}}^n$.
		Then $w(t):=W(x(t))$ is absolutely continuous ($W$ is locally
		Lipschitz on ${\ensuremath{\mathcal{D}}}_{\ensuremath{\mathcal{O}}}$ and $x$ is absolutely continuous
		with values in the compact ${\ensuremath{\mathcal{C}}}^n$).
		Fix a.e.\ $t$.
		On edges with $e_{ij}(t)\neq0$,
		the distance identity~\eqref{eq:deriv} gives
		$\tfrac{d}{dt}\|e_{ij}\|=\hat e_{ij}^\top(\omega_j-\omega_i)$;
		on the time set where an edge error vanishes, the absolutely
		continuous function $\|e_{ij}(\cdot)\|$ is zero, so its
		derivative vanishes a.e.\ there.
		Summing over edges and regrouping endpoint contributions,
		 
	\begin{align}
		\begin{aligned}
			
			\dot w
			= -\sum_{i\in{\ensuremath{\mathcal{V}}}}s_i^\top\omega_i
			= -\!\!\sum_{C\in{{\ensuremath{\mathcal{P}}}_0(x(t))}}\!\! S_C^\top\bar\omega_C
			\quad\text{a.e.},
		
		\end{aligned}\notag
	\end{align}

		using Lemma~\ref{lem:clusters} and $\sum_{i\in C}s_i=S_C$.
		Substituting~\eqref{eq:cluster},
		$\|G_C\|\leq|C|M$, and the Cauchy--Schwarz bound
		$\sum_C\|S_C\|^2/|C|\geq\Sigma^2/n$ with
		$\Sigma:=\sum_C\|S_C\|$,
		 
	\begin{align}
		\begin{aligned}
			
			\dot w
			\leq -\frac{\alpha}{n}\Sigma^2+\gamma M\Sigma
			=: \psi(\Sigma)
			\quad\text{a.e.\ on }\{w>0\}.
		
		\end{aligned}\notag
	\end{align}

		The concave parabola $\psi$ has vertex at
		$\Sigma=n\gamma M/(2\alpha)<1<2$
		by~\eqref{eq:gain}, so $\psi$ is decreasing on $[2,\infty)$;
		Lemma~\ref{lem:tree_bound} gives $\Sigma\geq2$ on $\{w>0\}$,
		hence
		$\dot w\leq\psi(2)=-\tfrac{2}{n}(2\alpha-\gamma nM)=:-c<0$
		a.e.\ on $\{w>0\}$.
		Define $T^*:=\inf\{t\geq0:w(t)=0\}$. If $w(0)>0$, then
		$w>0$ on $[0,T^*)$, and integration gives
		$0\leq w(t)\leq w(0)-ct$ for every $t<T^*$; hence
		$T^*\leq w(0)/c$. If $w(0)=0$, then $T^*=0$.
		Moreover, $w$ cannot leave zero afterwards: any excursion
		interval $(t_2,t_1]$ with $w(t_2)=0$ and $w>0$ on $(t_2,t_1]$
		would give $w(t_1)\leq-c(t_1-t_2)<0$.
		Finally, $w=0$ forces every edge error to vanish, and
		connectivity gives consensus for all $t\geq T^*$; substituting
		$c$ yields~\eqref{eq:settle}. This proves~(ii).

By Step~2, agreement persists: $R_i(t)=R_j(t)$ for all $i,j$
		and all $t\geq T^*$, so $\bar R(t):=R_i(t)$ is well defined,
		and for $t\geq T^*$ the trajectory evolves in the consensus set
		${\ensuremath{\mathcal{M}}}=\{(R,\ldots,R):R\in{\ensuremath{\mathcal{C}}}\}$, a submanifold with

		$(\bar\omega,\ldots,\bar\omega)$.
		By Lemma~\ref{lem:filippov}(i), the same velocities also admit
		the antisymmetric form~\eqref{eq:containment}; summing over all
		agents, every edge term cancels in antisymmetric pairs and
		$n\bar\omega=-\gamma\nabla_{\bar R}F$.
		This proves~\eqref{eq:sliding}; its right-hand side is
		locally Lipschitz, so~\eqref{eq:sliding} uniquely determines the
		common trajectory from $\bar R(T^*)$. The sliding velocity is
		radially admissible at $\partial{\ensuremath{\mathcal{C}}}$, since
		$\hat e_c^\top(-\tfrac{\gamma}{n}\nabla F)
		=-\tfrac{\gamma}{n}\sum_i\hat e_c^\top\nabla f_i\geq0$ by
		Assumption~\ref{ass:rad}.

		distance identity~\eqref{eq:deriv} with the static endpoint
		$R^*$ and $e_*:=(\log(\bar R^\top R^*))^\vee$,
		$\dot V_2=-e_*^\top\bar\omega
		=\tfrac{\gamma}{n}\langle\nabla_{\bar R}F,e_*\rangle
		\leq-\tfrac{\gamma}{n}\mu_F d^2=-\tfrac{2\gamma\mu_F}{n}V_2$
		a.e., using the interior-minimizer

		Gr\"onwall's inequality and $d=\sqrt{2V_2}$
		give~\eqref{eq:rate}. This proves~(iii).

\label{rem:viab}
		In body coordinates, the Bouligand tangent cone of the domain
		splits as
		$T_{{\ensuremath{\mathcal{C}}}^n}(x)=\prod_iT_{\ensuremath{\mathcal{C}}}(R_i)$ with
		 
	\begin{align}
		\begin{aligned}
			
			T_{\ensuremath{\mathcal{C}}}(R_i)=
			\begin{cases}
				{\ensuremath{\mathbb{R}}}^3, & d(R_i,R_c)<\rho,\\
				\{\omega:\hat e_{i,c}^\top\omega\geq0\},
				& d(R_i,R_c)=\rho.
			\end{cases}
		
		\end{aligned}\notag
	\end{align}

		For the $n=2$ tree consisting of the single edge $\{1,2\}$
		with $R_1=R_2\in\partial{\ensuremath{\mathcal{C}}}$, Lemma~\ref{lem:filippov}(iii)
		places the free selection $u_{12}=-\hat e_{1,c}$ on the zero
		edge inside ${\ensuremath{\mathcal{K}}}[h](x)$, and no other edge is incident, so
		 
	\begin{align}
		\begin{aligned}
			
			\hat e_{1,c}^\top\omega_1
			=-\alpha-\gamma\,\hat e_{1,c}^\top\nabla_{R_1}f_1
			\leq-\alpha+\gamma M<0
		
		\end{aligned}\notag
	\end{align}

		by the gain condition~\eqref{eq:gain}, which for $n=2$ reads
		$\alpha/\gamma>M$: the selected velocity has a strictly
		outward radial component, hence
		${\ensuremath{\mathcal{K}}}[h](x)\not\subseteq T_{{\ensuremath{\mathcal{C}}}^n}(x)$ and the standard
		pointwise sufficient condition for strong invariance is
		inapplicable.
		Membership in ${\ensuremath{\mathcal{K}}}[h](x)$ does not imply that the selected
		vector can be sustained along a solution: on the coincidence
		set the common-velocity identity of Lemma~\ref{lem:clusters}
		holds almost everywhere, and the maximum-radius estimate of
		Step~1 excludes every outward excursion, as for the scalar
		analogue $\dot z\in-\mathrm{SGN}(z)$, $z\in{\ensuremath{\mathbb{R}}}$, whose value at $z=0$
		contains nonzero elements while its only solution from $z=0$
		is $z\equiv0$. The operating ball is thus strongly invariant
		although the pointwise tangency test fails; compare the weaker
		existential notion of viability~\cite{aubin2009viability}.
		Note that Step~1 uses neither acyclicity nor strong convexity.

<!-- remark-5: body_md -->
\label{rem:tree_vs_general}
		The proof of Lemma~\ref{lem:tree_bound} is specific to trees:
		contracting zero-error clusters again produces a tree, whose leaf
		clusters each carry exactly one intercluster edge and contribute
		one unit vector to $\sum_C\|S_C\|$, independently of the
		attitudes, the operating radius, and the tree shape. With cycles,
		several active edges may leave one cluster and partially cancel,
		so a different estimate would be needed. On a general connected
		graph the protocol can be run over a spanning tree computed once
		in a distributed manner, at the cost of discarding the remaining
		edges. This independence concerns~\eqref{eq:two} only; the
		settling bound~\eqref{eq:settle} still depends on $M$ and $W(0)$.
