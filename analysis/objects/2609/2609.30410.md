<!-- theorem-1: body_md -->
\label{Thm_maindensity}
Fix $k,s\in\mathbb{N}$.
There exists a subadditive density $d^{\text{\ding{71}}}$ on $\mathbb{N}$ such that, for every $E\subseteq\mathbb{N}$ with $d^{\text{\ding{71}}}(E)>0$ and every $P\in\mathbb{Z}[y]$ of degree $k$ and satisfying $P(0)=0$, there exist $x>y\geqslant 2$ such that

\end{equation*}

<!-- corollary-1: body_md -->
\label{Cor_main}
For every finite coloring of $\mathbb{N}$, every $s\in\mathbb{N}$, and every $P\in\mathbb{Z}[y]$ satisfying $P(0)=0$, there exist infinitely many $x,y$ with $x>y\geqslant 2$ such that $\{x,x+P(y),xy^s\}$ is monochromatic.

<!-- definition-1: body_md -->
\label{D: invariant means}
A \emph{mean} on $\ell^{\infty}(\mathbb{N})$ is a linear functional $\beta:\ell^\infty(\mathbb{N})\to \mathbb{C}$, such that:
\begin{enumerate}
[label=(\roman{enumi}),ref=(\roman{enumi}),leftmargin=*]
\item $\beta$ is positive, i.e.,  $f\geqslant 0$ implies $\beta(f)\geqslant 0$;

\item $\beta$ is normalized, i.e., $\beta(\1_\mathbb{N})=1$.
\end{enumerate}
A mean is called \emph{shift-invariant} if $\beta(\sigma f)=\beta(f)$ for all $f\in\ell^\infty(\mathbb{N})$.
The collection of shift-invariant means is a convex and weak$^{*}$ compact set. We call the extreme points of this set \emph{ergodic (shift-invariant) means}.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Fix $k\in \mathbb{N}$. A polynomial Bohr$_{0}^{\leqslant k}$ set is a set of the form
    \begin{equation*}
        \left\{n\in \mathbb{N}\colon |\lambda_i^{P_i(n)}-1|\leqslant \varepsilon_i\  \text{for all } i\in [\ell]\right\}
    \end{equation*}for some  $\ell\in \mathbb{N}$, $\varepsilon_1,\ldots, \varepsilon_{\ell}>0$, $\lambda_1,\ldots, \lambda_\ell \in \mathbb{S}^1$ and non-constant polynomials $P_1,\ldots, P_\ell\in \mathbb{Z}[x]$ of degree at most $k$ and with $P_i(0)=0, \ 1\leqslant i \leqslant \ell$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{D: limsup for filters}
Suppose $\mathcal{F}$ is a $\pi$-system on $\mathbb{N}$. Given a bounded function $f\colon \mathcal{F}\to\mathbb{R}$, we define
\[
\limsup_{A\in \mathcal{F}} f(A)= \inf_{B\in\mathcal{F}}\ \sup_{C\in \mathcal{F}} f(C\cap B).
\]
<!-- end definition-3 -->
<!-- remark-1: body_md -->
We will use the notation $\gamma_B$ only in the case when $\gamma$ is a shift-invariant mean and $B$ is a polynomial Bohr$_{0}^{ \leqslant k}$ set, and we claim that under these assumptions $\gamma_B$ is always well-defined.

Indeed, if $B$ is such a set, then $\1_B$ is \emph{almost convergent}: every shift-invariant mean assigns to $\1_B$ the same value, namely the natural density $d(B)$ of $B$.
By \cite{Lorentz48}, this is equivalent to the uniform Cesàro convergence
\begin{equation*}
\lim_{N-M\to\infty}\frac{1}{N-M}\sum_{M\leqslant n<N}\1_B(n)=d(B).
\end{equation*}
For polynomial Bohr$_{0,\leqslant k}$ sets, this property follows from a classical argument of Furstenberg: one realizes the sequence $\1_B(n)$ as the values of a Riemann-integrable function along the orbit of a point in a uniquely ergodic system; see, for instance, \cite[Section 4.4]{EW11}.
We leave the details to the interested reader.
It follows that $\gamma(\1_B)=d(B)>0$ for every shift-invariant mean $\gamma$, and hence $\gamma_B$ is well defined.
<!-- end remark-1 -->
<!-- definition-4: body_md -->
\label{def_new_density}
Fix $k,s\in\mathbb{N}$.
Let $\beta$ and $\gamma$ be shift-invariant means on $\ell^\infty(\mathbb{N})$, and assume that $\beta$ is ergodic.
For $E\subseteq\mathbb{N}$, define
\begin{equation}\label{eq:definition-density}
d^{\text{\ding{71}}}(E):=\limsup_{B\in \mathrm{PB}_0^{\leqslant k}}\gamma_B\big(m\mapsto \beta(n\mapsto \1_E(m^s n))\big).
\end{equation}

    \end{equation*}and

<!-- corollary-2: body_md -->

    Let $(X,\mathcal{X},\mu,T)$ be a measure-preserving system. Then, any function in $F\in L^2(\mu)$ admits a decomposition \begin{equation*}

<!-- proposition-1: body_md -->
\label{prop_optimal factor for 2 polynomials}Let $P_1, P_2$ be polynomials with integer coefficients and zero constant terms, and let $Q$ be a polynomial with real coefficients. Assume also that $P_1, P_2$ have the same degree and are essentially distinct and that the degree of $Q$ is smaller than or equal to the degree of $P_1$.
Then, for any invertible ergodic measure-preserving system $(X,\mathcal{X},\mu,T)$, and any functions $F,G\in L^{\infty}(\mu)$ such that at least one of $F,G$ is weak-mixing, we have \begin{equation*}
\lim_{M\to\infty}\Big\|  \mathbb{E}_{m\in \Phi_M} e( Q(m)) T^{P_1(m)}F\cdot T^{P_2(m)}G \Big\|_{L^2(\mu)}=0
    \end{equation*}
for any F\o{}lner\ sequence $(\Phi_M)_{M\in \mathbb{N}}$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Throughout the proof, we may assume $F$ is a weak-mixing function and that $Q(0)=0$. We abuse terminology and say a function $F\in L^{\infty}(\mu)$ is orthogonal to a factor $(Y, \mathcal{Y}, \nu,S)$, if the conditional expectation\footnote{Recalling that a factor corresponds to an invariant sub-$\sigma$-algebra, the conditional expectation with respect to a factor is defined to be equal to the conditional expectation with respect to this $\sigma$-algebra.} $\mathbb{E}_{\mu}(F|\mathcal{Y})$ is equal to zero. 
We consider two cases depending on whether the polynomials $P_1$ and $P_2$ are dependent or not. We fix $k$ as the degree of the polynomials $P_1,P_2$


Assume first that $P_1, P_2$ are linearly independent. Note that the real vector space $\mathbb{R}_k[x]$ of polynomials of degree at most $k$ and with zero constant term has dimension $k$. Thus, we can find a collection $P_3,\ldots, P_{k}$ of polynomials with integer coefficients, such that $\{P_1,\ldots, P_{k}\}$ spans
$\mathbb{R}_k[x]$.

Write 
\[
Q(m)=\sum_{i=1}^k \theta_i P_i(m)
\]where $\theta_i\in \mathbb{R}$. For each $\theta_i$ we consider a rotation $(\mathbb{T},\mu_i,S_i)$ by $\theta_i$.  


 We rewrite our averages as 
 \[
\left\| \mathbb{E}_{m\in \Phi_M} (T\times S_1)^{P_1(m)} F'\cdot (T\times S_2)^{P_2(m)} G'    \prod_{i=3}^k (T\times S_i)^{P_i(m)}H_i \right\|_{L^2(\nu)}
 \]
where $F'(x,y)=F(x)e(y)$, $G'(x,y)=G(x)e(y)$ and $H_i(x,y)=e(y)$ for every $3\leqslant i\leqslant k$ and every $(x,y)\in X\times\mathbb{T}$.

The transformations $T\times S_i$ are commuting and the polynomials $P_1,\ldots, P_k$ are linearly independent. Therefore, \cite[Theorem 2.10]{FK25} implies that if $F'$ is orthogonal to the rational Kronecker factor $\mathcal{K}_{\mathrm{rat}, T\times S_1}$, then the previous average is zero.

We justify the orthogonality by showing that for every $r\in \mathbb{N}$, we have that the averages
\begin{equation*}
    \mathbb{E}_{m\in \Phi_M} (T\times S_1)^{rm}F'
\end{equation*}
converge to $0$ in $L^2(\nu)$. Plugging in the formula for $F$, this reduces to proving that the averages
\begin{equation*}
    \mathbb{E}_{m\in \Phi_M} e(rm\theta_1)T^{rm}F
\end{equation*}
converge to zero in $L^2(\mu)$. But this follows from the fact that $F$ is a weak-mixing function and, thus, its spectral measure has no point masses.






We are left with the case that $P_1,P_2$ are dependent. Then, we can find a polynomial $P\in \mathbb{Z}_k[x]$ such that $P_1=r_1P$ and $P_2=r_2P$ for some $r_1,r_2\in \mathbb{Z}$.
Since $P$ has degree $k$, the set $\{P,n,n^2,\ldots, n^{k-1}\}$
forms a basis for $\mathbb{R}_k[x]$ and we can write 
\begin{equation*}
    Q(m)=\theta P(m) +\sum_{i=1}^{k-1}\theta_i m^i.
\end{equation*}
Consider the space $X\times \mathbb{T}^k$ with the transformation
\[
S(x,y)=\Big(Tx,y+\big(\frac{\theta}{r_1},\theta_1,\ldots, \theta_{k-1}  \big)\Big), \qquad x\in X, y\in \mathbb{T}^k
\]and the measure $\mu\times \lambda_{\mathbb{T}^k}$, where $\lambda_{\mathbb{T}^k}$ stands for the Lebesgue measure on $\mathbb{T}^k$. Since $T$ is ergodic with respect to $\mu$, almost every ergodic component of $\mu\times \lambda_{\mathbb{T}^k}$ projects to $\mu$ under the corresponding coordinate map $X\times \mathbb{T}^{k}\to X$. We fix $\nu$ to be such an ergodic measure on $X\times \mathbb{T}^k$.

We rewrite our averages as 
\[
\Big\|\mathbb{E}_{m\in \Phi_M} S^{r_1P(m)}F'\cdot S^{r_2P(m)}G'
\prod_{j=1}^{k-1}S^{m^j}H_j
\Big\|_{L^2(\nu)}
\]
where $F'(x,y_1,\ldots, y_k)=F(x)e(y_1)$, $G'(x,y_1,\ldots, y_k)=G(x)$ and $H_j(x,y_1,\ldots, y_k)=e(y_{j+1})$ for all $(x,y_1,\ldots, y_k)\in X\times \mathbb{T}^k$.

We claim that if $F'$ is orthogonal to the Kronecker factor, then the previous average is zero. Assuming this claim for now, we conclude the proof, since $F'$ is a weak-mixing function
on the product system. This follows from the identity 
\[
\left|\int \overline{F'}\cdot S^n F'\,d\nu\right|=\left|\int \overline{F}\cdot T^n F \, d\mu\right|
\]and the hypothesis that $F$ is a weak-mixing function.


In order to establish that the Kronecker factor is characteristic for our averages, it suffices by \cite[Corollary 1.14]{Ku23}
to show that the family of polynomials 
$$
\{r_1P(m),r_2P(m), m,\ldots, m^{k-1}\}
$$
which appears in the iterates has algebraic complexity at most 1. This means that if there are polynomials $Q_0,Q_1,\ldots, Q_{k+1}$
with real coefficients such that \begin{equation}\label{eq_polynomial relations}
    Q_0(x)+Q_1(x+r_1P(y))+Q_2(x+r_2P(y))+\dots +Q_{k+1}(x+y^{k-1})=0,
\end{equation}then the polynomials $Q_0,\ldots, Q_{k+1}$ have degree at most $1$.

Assuming that there exist polynomials satisfying \eqref{eq_polynomial relations} with degrees at least 2, then after differentiating sufficiently many times, we find polynomials $\widetilde{Q}_i$ of degree at most 2 for which \eqref{eq_polynomial relations} is satisfied. 
Therefore, it suffices to prove our claim when the polynomials $Q_i$ have degree at most 2.

Write $Q_i(x)=a_ix^2+b_ix+c_i$ with $a_i,b_i,c_i\in \mathbb{R}$ and assume that not all $a_i$ are zero.
Then, \begin{equation*}
    a_0x^2+a_1(x+r_1P(y))^2+a_2(x+r_2P(y))^2+\sum_{i=3}^{k+1} a_i(x+y^{i-2})^2 +\Big(\sum_{i=0}^{k+1}b_i\Big)x+R(y)=0,
\end{equation*}where $R(y)$ is a polynomial of degree at most $k$.

Comparing the coefficient of the term $x$, we get \begin{equation*}
    2(a_1r_1+a_2r_2)P(y) +2\sum_{i=3}^{k+1}a_iy^{i-2}+\sum_{i=0}^{k+1}b_i=0
\end{equation*}The polynomial $P(y)$ has degree $k$. Therefore, we get $a_1r_1+a_2r_2=a_3=\ldots=a_{k+1}=0$. 

Now we compute the coefficient of $y^{2k}$, which appears only in the terms with $a_1,a_2$, to infer that
\[
a_1r_1^2+a_2r_2^2=0.
\]
We conclude that $a_1=a_2=0$. Finally, the coefficient of $x^2$ yields $a_0=0$. 

To summarize, we have that $a_0=a_1=\ldots=a_{k+1}=0$, a contradiction. Thus, the polynomial family has algebraic complexity at most 1 and the conclusion follows.

<!-- definition-5: body_md -->
\label{def_means on algebras}
    A \emph{mean} on a unital $C^*$-subalgebra $\mathcal{A}$ of $\ell^{\infty}(\mathbb{N})$ is a linear functional $\beta:\mathcal{A}\to \mathbb{C}$, such that:
\begin{enumerate}
[label=(\roman{enumi}),ref=(\roman{enumi}),leftmargin=*]
\item $\beta$ is positive, i.e., $\beta(f)\geqslant 0$ whenever $f$ is a positive\footnote{An element in a $C^{*}$-algebra $\mathcal{A}$ is called positive if it takes the form $\sum_{i=1}^k b_i\cdot b_i^{*}$, for some $b_1,\ldots, b_k\in \mathcal{A}$.} element,

\item $\beta$ is normalized, i.e., $\beta(\1_\mathbb{N})=1$.
\end{enumerate}

A mean on a shift-invariant $C^*$-subalgebra $\mathcal{A}$ is called \emph{shift-invariant} if $\beta(\sigma f)=\beta(f)$ for all $f\in\mathcal{A}$.
The collection of shift-invariant means is a convex and weak$^{*}$ compact set. We call the extreme points of this set the \emph{ergodic (shift-invariant) means} of $\mathcal{A}$.

        \lim_{k\to+\infty} \frac{|\Phi_k\cap (\Phi_k-h)|}{|\Phi_k|}=1 \ \text{for all } h\in \mathbb{N}.
    \end{equation*}

<!-- lemma-1: body_md -->
\label{L: mean on separable is Folner}
    Let $\mathcal{A}$ be a unital, separable, shift-invariant $C^*$-subalgebra of $\ell^{\infty}(\mathbb{N})$
    and let $\beta$ be a shift-invariant mean on $\mathcal{A}$. Then, there exists a F\o{}lner\ sequence ${\bf \Phi} = (\Phi_k)_{k\in \mathbb{N}}$, such that $\beta(f)=A_{{\bf \Phi}}(f)$ for every $f\in \mathcal{A}$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Our result follows from \cite[Corollary 4.6]{HS9}, which asserts that every shift-invariant mean on $\ell^{\infty}(\mathbb{N})$ is a limit along a F\o{}lner\ net. This implies that any shift-invariant mean on $\mathcal{A}$ is also a limit along F\o{}lner\ nets since it is the restriction of a shift-invariant mean of $\ell^{\infty}(\mathbb{N})$ to $\mathcal{A}$ (by the invariant Hahn--Banach theorem).
Since $\mathcal{A}$ is a separable space, the subspace topology inherited from the weak*-topology on the unit ball of the dual of $\ell^{\infty}(\mathbb{N})$ is metrizable. Thus, the unit ball of the dual of $\mathcal{A}$ is a compact metric space.
Using the first countability, it is not then difficult to show that for any cluster point of a F\o{}lner\ net $(\Phi_i)_{i\in I}$, we can construct a ``subsequence'' $(\Phi_{i_n})_{n\in \mathbb{N}}$ with indices $i_n\in I$ satisfying $i_n\leqslant i_{n+1}$ for every $n\in \mathbb{N}$, which also converges to that point. 
We infer that any shift-invariant mean on $\mathcal{A}$ is also a limit along a F\o{}lner\ sequence.

<!-- lemma-2: body_md -->
\label{lem_ergodicity passes to factors}
    Let $\beta$ be an ergodic shift-invariant mean on $\ell^{\infty}(\mathbb{N})$. Let $\mathcal{A}$ be any unital, shift-invariant $C^*$-subalgebra of $\ell^{\infty}(\mathbb{N})$. Then, the restriction $\beta|_{\mathcal{A}}$ is an ergodic mean on $\mathcal{A}$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Suppose that $\beta|_{\mathcal{A}}$ is not an extreme point among the shift-invariant means on $\mathcal{A}$.
Then, there exist shift-invariant means $\beta_1,\beta_2$ on $\mathcal{A}$ and a number
$0<t<1$ such that
$$\beta|_\mathcal{A}=t\beta_1+(1-t)\beta_2.$$
Define
$
\gamma_1:=t\beta_1,
\gamma_2:=(1-t)\beta_2.
$
Then
$$
\gamma_1+\gamma_2=\beta|_\mathcal{A},
$$
and therefore
$$
0\leqslant \gamma_1(f)\leqslant \beta|_\mathcal{A}(f)
\ \text{for all } f\geqslant 0,
$$since $\gamma_2$ is a positive functional. 


By the ordered Hahn--Banach extension theorem\footnote{Since the functionals take complex values, while the ordered Hahn--Banach theorem is stated in most sources for real valued functionals, one may proceed as follows: on the (real) vector space of self-adjoint elements of $\mathcal{A}$, we note that $\gamma_1$ is dominated by the sub-linear functional $p(f)=\inf\{\beta(g), g\geqslant 0, g\geqslant f\}$. We then apply the Hahn--Banach theorem to construct a real functional on the real vector space of self-adjoint elements of $\ell^{\infty}(\mathbb{N})$, which is dominated by $\beta$ on positive elements, and then complexify it via routine procedure.} (\cite[Theorem 3.2]{Rudin1991}), there exists
a positive functional $\widetilde{\gamma}_1$ on $\ell^\infty(\mathbb{N})$ such that
\[
\widetilde{\gamma}_1|_\mathcal{A}=\gamma_1
\]
and
\[
 \widetilde{\gamma}_1(f)\leqslant \beta(f) 
\]for all positive $f\in \ell^{\infty}(\mathbb{N})$.
Note that the last inequality forces the functional $\widetilde{\gamma_1}$ on the $C^*$-algebra $\ell^{\infty}(\mathbb{N})$ to be bounded and combined with the fact that $\left\Vert \widetilde{\gamma}_1 \right\Vert=\left\Vert \gamma_1 \right\Vert=\gamma_1(1)=\widetilde{\gamma}(1)$, this implies the positivity of $\widetilde{\gamma}_1$. 
However, it is not necessarily shift-invariant. To overcome this problem, we take a shift-invariant mean
$\gamma$ on $\ell^\infty(\mathbb{N})$ and average along shifts. More specifically, for
$f\in\ell^\infty(\mathbb{N})$, define
$$
\gamma_1^*(f)
:=
\gamma\big(
n\mapsto \widetilde{\gamma}_1(\sigma^n f)
\big).
$$
Then $\gamma_1^*$ is a positive shift-invariant functional on $\ell^\infty(\mathbb{N})$.
Moreover, since
\[
0\leqslant \widetilde{\gamma}_1(\sigma^n f)
\leqslant \beta(\sigma^n f)
=\beta(f)
\qquad \text{for }f\geqslant 0,
\]
we have
\[
0\leqslant \gamma_1^*\leqslant \beta .
\]
Since $\gamma_1$ is already shift-invariant on $\mathcal{A}$, for every $f\in \mathcal{A}$,
\[
\gamma_1^*(f)
=
\gamma \big(
n\mapsto \widetilde{\gamma}_1(\sigma^n f)
\big)
=
\gamma\big(
n\mapsto \gamma_1(\sigma^n f)
\big)
=
\gamma\!\left(
n\mapsto \gamma_1(f)
\right)
=
\gamma_1(f).
\]
Hence
\[
\gamma_1^*|_\mathcal{A}=\gamma_1 .
\]
Set $\gamma_2^*:=\beta-\gamma_1^*$.
Since $0\leqslant \gamma_1^*\leqslant \beta$, the functional $\gamma_2^*$ is positive and
shift-invariant. 


We have established that the functionals $\gamma_1^*, \gamma_2^*$ are positive and shift-invariant, so we normalize them to make them means.
Note that
\[
\gamma_1^*(1)=\gamma_1(1)=t\in (0,1).
\]
Consequently,
\[
\beta
=
t\frac{\gamma_1^*}{t}
+
(1-t)\frac{\gamma_2^*}{1-t},
\]
where both $\gamma_1^*/t$ and $\gamma_2^*/(1-t)$ are shift-invariant means on
$\ell^\infty(\mathbb{N})$.
Since $\beta$ is extreme among shift-invariant means on $\ell^\infty(\mathbb{N})$, it
follows that
\[
\frac{\gamma_1^*}{t}=\beta.
\]
Restricting to $\mathcal{A}$ gives
\[
t\beta_1
=
\gamma_1
=
\gamma_1^*|_\mathcal{A}
=
t\,\beta|_\mathcal{A} ,
\]
which implies that
\[
\beta_1=\beta|_\mathcal{A} .
\]
The same argument applied to $\gamma_2$ yields
\[
\beta_2=\beta|_\mathcal{A} .
\]

In conclusion, every convex decomposition of $\beta|_A$ into shift-invariant means on
$\mathcal{A}$ is trivial. Hence $\beta|_A$ is an extreme point of the set of
shift-invariant means on $\mathcal{A}$.

<!-- proposition-2: body_md -->
\label{prop_correspondence principle}
     Let $\mathcal{A}$ be a separable, unital, shift invariant $C^{*}$-subalgebra of $\ell^{\infty}(\mathbb{N})$ and let $\beta$ be an ergodic shift-invariant mean on $\mathcal{A}$. Then there exists a compact metric space $X$, a continuous map $T\colon X\to X$, a $T$-invariant Borel probability measure $\mu$ on $X$ that is ergodic with respect to $T$, and an isometric $C^{*}$-isomorphism  $\Psi:\mathcal{A}\to C(X)$, such that for all positive integers $\ell$ and sequences $f_1,\ldots,f_{\ell} \in \mathcal{A}$, there exist $F_{1},\ldots, F_\ell\in C(X)$, such that for all non-negative $h_1\ldots, h_{\ell}\in \mathbb{Z}$, we have
        \begin{equation}\label{eq_furstenberg system identity0}
    \beta\bigg(n\mapsto f_1(n+h_1)\cdot\ldots\cdot f_{\ell}(n+h_{\ell})\bigg)=\int T^{h_1}F_1\cdot \ldots\cdot T^{h_{\ell}}F_{\ell}\, d\mu.
    \end{equation}
   In addition, if $f_1,\ldots, f_k$ are positive elements of $\mathcal{A}$, then the functions $F_1,\ldots, F_k$ are non-negative.
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Recall that $\sigma$ denotes the shift operator on $\ell^{\infty}(\mathbb{N})$.
    We let $$X=\{\phi:\mathcal{A}\to \mathbb{C}\colon \ \phi \text{ is a multiplicative linear functional}\}$$
be the Gelfand spectrum of $\mathcal{A}$. This is a compact Hausdorff space with the weak*-topology (the subspace topology inherited by the weak* topology on the dual of $\mathcal{A}$). The weak*-topology on the unit ball of $\mathcal{A}^{*}$ is metrizable since $\mathcal{A}$ is separable. Thus, $X$ is a compact metrizable space. Finally, by the Gelfand-Naimark theorem, there exists an isometric $C^*$-isomorphism $\Psi: \mathcal{A}\to C(X)$.

We denote by $T$ the transformation $$T: X\to X, \ T\phi(a)=\phi(\sigma(a))\ \text{for every } \phi\in X,$$ 
which is easily checked to be a well-defined, continuous function on $X$. 

To construct the measure on $X$, we consider the push-forward of the linear functional $\beta$. Namely, if we define $\rho=\beta\circ \Psi^{-1} $, then it is straightforward to check that this is a positive linear functional on the vector space $C(X)$ (with the supremum norm). Invoking the Riesz-Markov-Kakutani representation theorem, we can find a Radon measure $\mu$ on $X$ characterized by \begin{equation}\label{eq_characterization of mu}
    \rho(F)= \int F \, d\mu\ \text{for every } F\in C(X).
\end{equation} Since $\beta$ is shift-invariant, it follows that the measure $\mu$
is $T$-invariant.

We claim that the measure $\mu$ is ergodic.  It is known (see \cite[Theorem 4.1, Theorem 4.4]{EW11}) that the space of Borel $T$-invariant probability measures on the compact metric space $X$ is a non-empty, convex set and its extreme points are precisely the ergodic $T$-invariant probability measures on $X$. Arguing by contradiction, we suppose that $\mu$ is not ergodic and, thus, not an extreme point for the set of $T$-invariant probability measures on $X$. This implies that we can find $T$-invariant measures $\mu_1,\mu_2$ and $t\in (0,1)$ such that
$$\mu=t\mu_1+(1-t)\mu_2.$$
Naturally, it follows that for the functionals $\rho_1,\rho_2$ corresponding to $\mu_1,\mu_2$, we also have $$\rho=t\rho_1+(1-t)\rho_2.$$
Consequently, $$\beta=t\beta_1+(1-t)\beta_2,$$where  $\beta_1=\rho_1\circ \Psi $ and $\beta_2=\rho_2\circ \Psi$ are shift-invariant means on $\mathcal{A}$. This contradicts the ergodicity assumption on $\beta$, and the claim follows. 

Let $f_1\ldots, f_k\in \mathcal{A}$ be elements of $\mathcal{A}$. We define the functions $F_i=\Psi({f_i})$. We show that if the elements $f_i$ are positive, then the functions $F_i$ are also positive. This follows from the fact that $\Psi$ is a positive operator. More simply,
if the elements $f_i$ are positive, we can find elements $b_1,\ldots, b_k\in \mathcal{A}$ such that $f_i=b_1\cdot b_1^{*}+\dots+b_k\cdot b_k^{*}$. Since $\Psi$ is a $C^*$-algebra homomorphism, we derive $F_i=G_1\cdot \overline{G_1}+\dots+G_k\cdot \overline{G_k}$, where $G_i=\Psi(b_i)$. This confirms our claim. 


It remains to establish \eqref{eq_furstenberg system identity0}.
We note that the left-hand side of \eqref{eq_furstenberg system identity0} can be rewritten as \begin{equation*}
    \beta\left(\sigma^{h_1}{f_1}\cdot\ldots\cdot \sigma^{h_{\ell}}{f_{\ell}}\right).
\end{equation*}



The identity ${\Psi}(\sigma( a))=T({\Psi}(a))$ implies that the 
 image of $\sigma^{h_j}f_j$
is the function $T^{h_j}F_j$. 
Since ${\Psi}$ is an algebra homomorphism, it follows that the image of the sequence $\sigma^{h_1}f_1\cdot\ldots\cdot \sigma^{h_{\ell}}f_{\ell}$ is the function $T^{h_1}F_1\cdot\ldots \cdot T^{h_k}F_k$. The conclusion follows from \eqref{eq_characterization of mu}.

<!-- theorem-3: body_md -->
\label{prop_dynamical statement}
   Let $s,k\in \mathbb{N}$, $P\in \mathbb{Z}[x]$ of degree $k$ with $P(0)=0$ and \(\gamma\) be a shift-invariant mean. Assume $(X,\mathcal{X},\mu,T)$ is an invertible ergodic measure preserving system and $(F_m)_{m\in \mathbb{N}}$ is a sequence of measurable functions on $X$ taking values in $[0,1]$ such that the quantity 
\begin{equation}\label{eq_positivity of averages of measurable functions along Bohr sets}
         \limsup_{B\in \mathrm{PB}_0^{\leqslant k}}\gamma_B\Big(m\mapsto \int F_{m}\, d\mu\big)>0
    \end{equation}
   Then, we have $$\int F_{a}\cdot T^{\frac{1}{a^s} P(\frac{m}{a})}F_a\cdot  F_m\, d\mu> 0.$$ for infinitely many $a,m\in \mathbb{N}$ with $a^{s+1}\mid m$.

Specializing to the case $s=1$ and $P(m)=m$, consider the non-ergodic transformation\footnote{This is an adaptation of a construction by Moreira (personal communication).} $T(x,y)=(x,x+y)$ on $\mathbb{T}^2$ with the Lebesgue measure. Fix $s\in \mathbb{N}$, let $E$ and $P\in \mathbb{Z}[x]$ be as in the statement of \cref{Thm_maindensity} and let $\gamma$ and $\beta$ be as in \cref{def_new_density}. 

Let $\mathcal{A}$ be the affinely invariant $C^*$-algebra generated by $\1_{E}$. Applying Lemma~\ref{lem_ergodicity passes to factors}, we deduce that the restriction $\beta|_\mathcal{A}$ to $\mathcal{A}$ is ergodic. It is immediate from the formula for the density in \cref{def_new_density} that $E$ satisfies $$\limsup_{B\in \text{PB}_0^{\leqslant k}}\gamma_{B}\big(m\mapsto \beta|_{\mathcal{A}}(n\mapsto \,\1_{E}(m^sn))\big)>0,$$
    since the sequence $n\mapsto \1_{E}(m^sn)$ is also an element of $\mathcal{A}$.

We apply Proposition \ref{prop_correspondence principle} for the algebra $\mathcal{A}$ and the mean $\beta|_\mathcal{A}$. We let $(X,\mu, T)$ be the ergodic measure-preserving system and $\Psi\colon \mathcal{A}\to C(X)$ be the $C^{*}$-isomorphism provided by this proposition.
Furthermore, for every $a\in\mathbb{N}$, let $f_a(n)=1_E(a^sn)$ and 
define $F_{a}= \Psi(f_a)$. 
Note that $F_a$ is a continuous function on $X$ taking values in $[0,1]$.


The system provided by \cref{prop_correspondence principle} is not invertible, so iterates involving negative quantities do not necessarily make sense. To overcome this, we pass to the natural extension $(\widetilde{X},\widetilde{\mu},\widetilde{T})$ of $(X,\mu,T)$, where $\widetilde{X}$ is a compact metric space, $\widetilde{T}:\widetilde{X}\to \widetilde{X}$ is a homeomorphism
and $\widetilde{\mu}$ is an ergodic $\widetilde{T}$-invariant Borel probability measure. This is a standard dynamical construction. Any function $F\in C(X)$
taking values in $[0,1]$ has a lift in $C(\widetilde{X})$ taking values in $[0,1]$. In addition, the integral on the right-hand side of \eqref{eq_furstenberg system identity0} are preserved when replacing the functions by their lifts, $T$ with $\widetilde{T}$ and $\mu$ by $\mu$. This also allows us to extend \eqref{eq_furstenberg system identity0} to general integer shifts $h_1,\ldots, h_k$ using the shift-invariance of $\beta$. We omit the details. 










From now on, we assume that $(X,\mu,T)$ is replaced by the natural extension and the functions $F_a$ have been replaced by their lifts. We keep the same notation for convenience and just assume that $(X,\mu,T)$ is an invertible measure-preserving system.

We deduce from \eqref{eq_furstenberg system identity0} that $$\beta|_\mathcal{A}(n\mapsto 1_E(m^sn))=\int F_m\ d\mu$$ for every $m\in \mathbb{N}$, which implies that $$\limsup_{B\in \text{PB}_0^{\leqslant k}}\gamma_{B}\Big(m\mapsto \beta\big(n\mapsto \,\1_{E}(m^sn)\big)\Big)=\limsup_{B\in \text{PB}_0^{\leqslant k}}\gamma_B\left(m\mapsto \int F_m\, d\mu\right).$$

Our assumption on the positivity of the left-hand side implies that 
the sequence $F_m$ of continuous functions and the mean $\gamma$ satisfy \eqref{eq_positivity of averages of measurable functions along Bohr sets}.
It follows that there exist infinitely many $a,m\in \mathbb{N}$ with $a^{s+1}\mid m$
such that $$\int F_{a}\cdot T^{\frac{1}{a^s} P(\frac{m}{a})}    F_a\cdot  F_m\, d\mu> 0.$$
However, using the identity \eqref{eq_furstenberg system identity0} and the fact that $F_m$ is the image of $n\mapsto 1_E(m^sn)$, we deduce that \begin{equation*}
    \beta\left(n\mapsto \1_{E}(a^sn)\1_E\left(a^s\left(n+\tfrac{P(m/a)}{a^s}\right)\right)\1_{E}(m^sn)\right)=\int F_{a}\cdot  T^{\frac{1}{a^s} P(\frac{m}{a})}  F_a\cdot  F_m\, d\mu.
\end{equation*}
Therefore, we can find infinitely many triples $(a,m,n)$ with $a^{s+1}\mid m$ so that $$\1_{E}(a^sn)\1_E\left(a^sn+P(\tfrac{m}{a})  \right)\1_{E}(m^sn)>0.$$
Choosing $x=a^sn$ and $y=m/a$, we deduce that $\{x,x+P(y),xy^s\}\subseteq E$.

\label{prop_dynamical statement full}
    Let $k,s\in \mathbb{N}$, $P\in \mathbb{Z}[x]$ of degree $k$ with $P(0)=0$ and \(\gamma\) be a shift-invariant mean. <!-- proof-6: body_md -->
Our first objective is to construct the sequence of sets $(B_{\ell})_{\ell\in \mathbb{N}}$.
For every $m\in \mathbb{N}$, we use Corollary \ref{cor_decomposition via Kronecker} to write
$$F_{m}=F_{m,\mathrm{str}}+F_{m,\mathrm{unf}}$$
for functions $F_{m,\mathrm{str}},F_{m,\mathrm{unf}}\in L^{\infty}(\mu)$, such that $F_{m,\mathrm{str}}\in L^{\infty}(\mu)$ is a compact function and $F_{m,\mathrm{unf}}$ is a weak-mixing function.

For every $\ell\in\mathbb{N}$, we can find a polynomial Bohr$_0$ set $C_{\ell}\subseteq\mathbb{N}$, such that  for all $a\in\{1,\ldots,\ell\}$
\begin{align}
a^{s+1}\mid m,\qquad \text{for all } m\in C_{\ell}
\label{eqn_bohr_set_divisibility_assumption}\\
\sup_{m\in C_{\ell}} 
\int \big|F_{a,\mathrm{str}}\big(T^{\frac{1}{a^s}P(\frac{m}{a})}x)-  F_{a,\mathrm{str}}(x)\big|^2 d\mu(x)\ \leqslant \frac{1}{\ell}\label{eqn_bohr_set_of_almost_periods}.
\end{align}
To see why this is true, note first that the set of $m$ that are divisible by $(\ell!)^{s+1}$ is a PB$_0^{\leqslant k}$ set, and all elements of this set satisfy the first condition. For the second assertion, note that for every $a\in \mathbb{N}$, there exist $t_a\in \mathbb{N}$, eigenfunctions $g_{a,1},\ldots, g_{a,t_a}\in L^2(\mu)$, and coefficients $c_{a,1},\ldots, c_{a,t_a}$, such that \begin{equation}\label{eq_approximation of structured by eigenfunctions}
\Big\|F_{a,\mathrm{str}}-\sum_{t\leqslant t_a}c_{a,t}g_{a,t}\Big\|_{L^2(\mu)}\leqslant \frac{1}{3\ell}.
\end{equation}Additionally, let $\lambda_{a,t}\in \mathbb{C}$ denote the eigenvalue of the function $g_{a,t}$.
For any $a\in \mathbb{N}$, we let $Q_a(m)=a^{k}P(m/a)$, which is a polynomial with integer coefficients since $P$ has degree $k$.


Now, we define the PB$_0^{\leqslant k}$ set (we make an arbitrary choice for the root $\lambda_{a,t}^{1/a^{k+s}}$ here)
$${C}_{a,\ell}=\left\{m\in \mathbb{N}\colon \Big|\Big(\lambda_{a,t}^{\frac{1}{a^{k+s}}}\Big)^{Q_a(m)}-1\Big|\leqslant \frac{1}{3\ell\sum_{t\leqslant t_a}|c_{a,t}|} \ \text{for all } t\in [t_a]\right\}\cap (\ell!)^{s+1}\mathbb{N}.$$
Observe that since $Q_a$ takes integer values, we have
$\big(\lambda_{a,t}^{\frac{1}{a^{k+s}}}\big)^{Q_a(m)}=\lambda_{a,t}^{\frac{Q_a(m)}{a^{k+s}}}$.
We infer that for any $m\in C_{a,\ell}$, we have 
\begin{equation}\label{eq_Bohr property of eigenvalues}
    \big|\lambda_{a,t}^{\frac{1}{a^s} P(\frac{m}{a})}-1\big|\leqslant\frac{1}{3\ell\sum_{t\leqslant t_a}|c_{a,t}|}
\end{equation}   for all admissible values of $t$.
The triangle inequality implies that for any $m\in {C}_{a,\ell}$, we have 
\begin{align*}
\big\| & T^{\frac{1}{a^s}P(\frac{m}{a})} F^{}_{a,\mathrm{str}}-  F^{}_{a,\mathrm{str}}\big\|_{L^2(\mu)}\\
&\leqslant \Big\|T^{\frac{1}{a^s}P(\frac{m}{a})}F_{a,\mathrm{str}}-\sum_{t\leqslant t_a}c_{a,t}T^{\frac{1}{a^s}P(\frac{m}{a})}g_{a,t}\Big\|_{L^2(\mu)}+\Big\|F_{a,\mathrm{str}}-\sum_{t\leqslant t_a}c_{a,t}g_{a,t}\Big\|_{L^2(\mu)}  \\
&\qquad \qquad \qquad  \qquad  \qquad \qquad \ \ \ \ + \Big\|\sum_{t\leqslant t_a} c_{a,t}T^{\frac{1}{a^s}P(\frac{m}{a})}g_{a,t}-\sum_{t\leqslant t_a}c_{a,t}g_{a,t}\Big\|_{L^2(\mu)}\\
&\leqslant \frac{2}{3\ell}+
\sum_{t\leqslant t_a}|c_{a,t}|\cdot\|\lambda_{a,t}^{\frac{1}{a^s}P(\frac{m}{a})} g_{a,t}-g_{a,t}\|_{L^2(\mu)}\ \\
&\leqslant 3\cdot \frac{1}{3\ell}=\frac{1}{\ell}.
\end{align*}
In these inequalities, we used \eqref{eq_approximation of structured by eigenfunctions} for the first two terms (note that $T$ is an isometry on $L^2(\mu)$), and for the third term we used that $g_{a,t}$ are eigenfunctions and \eqref{eq_Bohr property of eigenvalues}. 

Finally, we need the last relation to hold for all $a\in [\ell]$, so we  consider the intersection \begin{equation*}
   C_{\ell}=\bigcap_{a=1}^{\ell} C_{a,\ell}\cap (\ell!)^{s+1}\mathbb{Z}.
\end{equation*}
It follows that the set $C_{\ell}\in \text{PB}_0^{\leqslant k}$ satisfies the desired properties
\eqref{eqn_bohr_set_divisibility_assumption} and \eqref{eqn_bohr_set_of_almost_periods}.

Now fix $\varepsilon>0$.
By \cref{D: limsup for filters}, we deduce that for all sets $C_{\ell}$ above we have 
\begin{equation*}
    \sup_{B\in \mathrm{PB}_0^{\leqslant k}} \gamma_{B\cap C_{\ell}}\Big(m\mapsto \int F_m\, d\mu\Big)\geqslant \delta.
\end{equation*}
This implies that we can find a sequence
$B_{\ell}\subseteq C_{\ell}$ of sets in the family PB$_0^{\leqslant k}$, such that \begin{equation}\label{eq_lower bound by delta/2 with mean}
\frac{\gamma\Big(m\mapsto \1_{B_{\ell}}(m)\int F_m\, d\mu \Big)}{\gamma(B_{\ell})}>\delta-\varepsilon.
\end{equation}for every $\ell\in \mathbb{N}$. This completes the construction of the sequence $(B_\ell)_{\ell\in\mathbb{N}}$.

Our next goal is to construct the F\o{}lner\ sequence $(\Phi_M)_{M\in\mathbb{N}}$. 
Let $\mathcal{B}$ denote the shift-invariant $C^*$-algebra generated by 
\begin{enumerate}
    \item $\1_{B_{\ell}}$ for $\ell\in \mathbb{N}$ and
    \item  the sequence $m\mapsto \int F_m \,d\mu$.
\end{enumerate} 
We apply Lemma~\ref{L: mean on separable is Folner} for the mean $\gamma$ and the separable $C^*$-algebra $\mathcal{B}$ to obtain a F\o{}lner\ sequence $(\Phi_{M})_{M\in \mathbb{N}}$, such that \begin{equation*}
    \gamma(m\mapsto f(m))=\lim_{M\to+\infty}\mathbb{E}_{m\in \Phi_M} f(m)\ \text{for every } f\in \mathcal{B}.
\end{equation*}
Since $\1_{B_{\ell}}$ and $m\mapsto \int F_m\, d\mu$ are elements of $\mathcal{B}$, we get
\begin{equation*}
    \frac{\gamma\left(m\mapsto \1_{B_{\ell}}(m)\int F_m \, d\mu\right)}{\gamma(B_{\ell})}=\frac{1}{\gamma(B_{\ell})}\lim_{M\to+\infty}\mathbb{E}_{m\in \Phi_M} \1_{B_{\ell}}(m)\int F_m\, d\mu.
\end{equation*}
Thus, \eqref{eq_lower bound by delta/2 with mean} implies \begin{equation}\label{eq_lower bound by delta/2}
   \lim_{M\to+\infty} \mathbb{E}_{m\in \Phi_{M}\cap B_{\ell}} \int F_m \,d\mu> \delta-\varepsilon
\end{equation}for every $\ell\in \mathbb{N}$.
This completes the construction of the F\o{}lner{} sequence $(\Phi_M)_{M\in\mathbb{N}}$.

Note that by passing to a subsequence of $B_{\ell}$ and $\Phi_M$ if necessary (which doesn't affect the validity of \eqref{eq_lower bound by delta/2}), we may assume that the iterated limits \begin{align*}
    L:=
 &\lim_{\ell_2\to\infty}
\lim_{M_2\to\infty}
\lim_{\ell_1\to\infty}
\lim_{M_1\to\infty} \mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}  \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}\ \int F_a\cdot T^{\frac{1}{a^s} P(\frac{m}{a})}F_a\cdot F_m\,d\mu
\end{align*}exist.
To make the presentation cleaner from now on, we will use the notation \begin{equation*}
    \lim_{(n_1,\ldots, n_k)\to\infty}a_{n_1,\ldots,n_k}
\end{equation*}for the sequence of iterated limits \begin{equation*}
    \lim_{n_1\to+\infty}\cdots \lim_{n_k\to+\infty}a_{n_1,\ldots,n_k}
\end{equation*}for a given multiply indexed sequence $a_{n_1,\ldots,n_k}:\mathbb{N}^k\to \mathbb{C}$, assuming that all limits in the previous expression exist.

We now apply the decomposition $F=F_{a,\mathrm{str}}+F_{a,\mathrm{unf}}$ to rewrite our averages as
\begin{align*}
L= \lim_{(\ell_2,M_2,\ell_1,M_1)\to\infty} \mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}&  \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}
\int F_a\cdot T^{\frac{1}{a^s} P(\frac{m}{a})}F_a   \cdot F_m\, d\mu
\\
& =\Sigma_1+\Sigma_2,
\end{align*}
where
\begin{align*}
\Sigma_1&=  \lim_{(\ell_2,M_2,\ell_1,M_1)\to\infty} &
\mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}  \mathbb{E}_{m\in B_{\ell_1}\cap\Phi_{M_1}}
\int F_a\cdot T^{\frac{1}{a^s} P(\frac{m}{a})}F_{a,\mathrm{str}}\cdot F_m\, d\mu,
\\
\Sigma_2&=  \lim_{(\ell_2,M_2,\ell_1,M_1)\to\infty} &
\mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}  \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}
 \int F_a \cdot T^{\frac{1}{a^s} P(\frac{m}{a})}F_{a,\mathrm{unf}}\cdot  F_m\,d\mu.
\end{align*}Here, we can pass to subsequences again if necessary, to ensure that the new iterated limits exist as well. 

We claim that $\Sigma_2=0$. 
Using the Cauchy-Schwarz inequality, we have
\begin{align*}
\Big|\mathbb{E}_{a\in B_{\ell_2}\cap  \Phi_{M_2}} &\mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}} 
 \int F_a \cdot T^{\frac{1}{a^s} P(\frac{m}{a})}F_{a,\mathrm{unf}}\cdot  F_m\,d\mu \Big|^2
\\
&\leqslant \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}
\int  \Big|\mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}} F_a\cdot T^{\frac{1}{a^s} P(\frac{m}{a})} F_{a,\mathrm{unf}} \Big|^2\,d\mu
\\
&=\mathbb{E}_{a,b\in B_{\ell_2}\cap\Phi_{M_2}} \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}
\int  F_a\cdot F_b \cdot T^{\frac{1}{a^s} P(\frac{m}{a})}F_{a,\mathrm{unf}}\cdot 
T^{\frac{1}{b^s} P(\frac{m}{b})}F_{b,\mathrm{unf}} \, d\mu.
\end{align*}



Note that the contribution of the diagonal terms $a=b$ is negligible, so we may assume throughout $a\neq b$.


For large enough $\ell_1$, we have $(ab)^{s+1}$ divides $m$ by the construction of the Bohr sets. Inserting the weight $1_{(ab)^{s+1}\mid m}$ into the average and doing the change of variables $m\to (ab)^{s+1}m$, it suffices to show that 
\begin{equation*}
   \limsup_{M\to\infty}\mathbb{E}_{m\in B_{\ell_1}/(ab)^{s+1}\cap \Phi_{M}/(ab)^{s+1}} \int  F_a\cdot F_b \cdot T^{P_a(m)}F_{a,\mathrm{unf}}\cdot 
T^{P_b(m)}F_{b,\mathrm{unf}} \, d\mu
\end{equation*}
converges to zero for every $\ell_1\in \mathbb{N}$, where $P_a,P_b$ are polynomials with integer coefficients defined by $P_a(m)=\frac{1}{a^s}P(a^{s}b^{s+1}m)$ and $P_b(m)=\frac{1}{b^s}P(a^{s+1}b^{s}m)$.





Since $B_{\ell_1}$ is a PB$_0^{\leqslant k}$ set, a standard approximation argument shows that it suffices to establish the estimate \begin{equation}\label{eq_final average with weak mixing}
    \limsup_{M\to\infty}\left|\mathbb{E}_{m\in \frac{\Phi_{M}}{(ab)^{s+1}}}  e\left( Q(m)\right)\int  H \cdot T^{P_a(m)}F_{a,\mathrm{unf}}
T^{P_b(m)}F_{b,\mathrm{unf}}\, d\mu\right|=0
\end{equation}for any function $H\in L^{\infty}(\mu)$, $\theta\in \mathbb{T}$ and any polynomial $Q$ with real coefficients and degree at most $k$.
Notice now that the assumptions of Proposition \ref{prop_optimal factor for 2 polynomials} are satisfied (since $\Phi_{M}/(ab)^{s+1}$ forms a F\o{}lner\ sequence, $P_a, P_b$ are essentially distinct as they have distinct leading coefficients and the degree assumption is satisfied), and thus, \eqref{eq_final average with weak mixing} follows.









































To conclude the proof, we show that the remaining term $\Sigma_1$ yields the desired lower bound. For this, recall that the original functions $F_a$ are non-negative, which also implies that the same holds for the functions $F_{a,\mathrm{str}}$. In particular, by passing to subsequences, if necessary, for the limit below to exist, we have the lower bound 
\begin{multline*}
    L=\Sigma_1\geqslant\\
    \lim_{(\ell_2,M_2,\ell_1,M_1)\to\infty} 
\mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}  \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}
\int F_a\cdot T^{\frac{1}{a^s} P(\frac{m}{a})} F_{a,\mathrm{str}}\cdot  F_m\cdot F_{m, \mathrm{str}}\, d\mu.
\end{multline*}
Using \eqref{eqn_bohr_set_of_almost_periods} for every $m\in B_{\ell_1}\subseteq C_{\ell_1}$ and noticing that the variable $m$ is very large compared to the numbers $a\in \Phi_{M_2}$, we have
\begin{equation}\label{eq_only structured parts remaining}
  L\geqslant  \lim_{(\ell_2,M_2,\ell_1,M_1)\to\infty}
\mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}  \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}\int 
F_a\cdot F_{a,\mathrm{str}} \cdot F_m\cdot F_{m, \mathrm{str}}\, d\mu.
\end{equation}



Next, consider the function
\[
\varphi(\ell_2,M_2,\ell_1,M_1)=
\mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}  \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}\int 
F_a\cdot F_{a,\mathrm{str}} \cdot F_m\cdot F_{m, \mathrm{str}}\, d\mu.
\]


Using Ramsey's theorem for $4$-sets and replacing $(B_\ell)_{\ell\in\mathbb{N}}$ and $(M_j)_{j\in\mathbb{N}}$ with subsequences of themselves if necessary, we can assume that there exists some $C_1\in[0,1]$ such that\begin{equation}\label{eq_first ramsey aprroximation}
    |\varphi(\ell_2,M_2,\ell_1,M_1)-C_1|\leqslant\varepsilon
\end{equation}
for all $M_1,\ell_1,M_2,\ell_2$ with $M_1>\ell_1>M_2>\ell_2$.
Using Ramsey's theorem for $2$-sets and replacing $(B_\ell)_{\ell\in\mathbb{N}}$ and $(M_j)_{j\in\mathbb{N}}$ by subsequences once more, we can also assume without loss of generality that there exists some $C_2\in[0,1]$ such that
\begin{equation}\label{eq_second ramsey approximation}
    \Big|\mathbb{E}_{m\in B_{\ell}\cap \Phi_M}  
 \int F_m\cdot F_{m, \mathrm{str}}\, d\mu -C_2\Big|\leqslant\varepsilon
\end{equation}
for all $M,\ell$ with $M>\ell$. 
Using the Cauchy-Schwarz inequality, we obtain for any $Q\in \mathbb{N}$
\begin{align*}
C_2^2&=
\Big( \mathbb{E}_{q\in [Q]}\mathbb{E}_{\ell\in(2^q,2^{q+1}]}\mathbb{E}_{M\in(2^{q+1},2^{q+2}]}  \mathbb{E}_{m\in B_{\ell}\cap \Phi_{M}}  
 \int F_m\cdot F_{m, \mathrm{str}}\, d\mu \Big)^2+{\rm O}(\varepsilon)
 \\
 &\leqslant
\int 
\Big(\mathbb{E}_{q\in [Q]}\mathbb{E}_{\ell\in(2^q,2^{q+1}]}\mathbb{E}_{M\in(2^{q+1},2^{q+2}]}  \mathbb{E}_{m\in B_{\ell}\cap \Phi_M}\     
   F_m\cdot F_{ m,\mathrm{str}}\Big)^2\,d\mu+{\rm O}(\varepsilon)
   \\&=
\frac{2}{Q^2}\sum_{\substack{p,q\in [Q]\\ p+1<q}}
\mathbb{E}_{\substack{\ell_2\in(2^p,2^{p+1}]\\ M_2\in(2^{p+1},2^{p+2}]}}
\mathbb{E}_{\substack{\ell_1\in(2^q,2^{q+1}]\\ M_1\in(2^{q+1},2^{q+2}]}}\,
\varphi(\ell_2,M_2,\ell_1,M_1)+{\rm O}\left(\varepsilon+\frac{1}{Q}\right)
\\
&=
C_1+{\rm O}\left(\varepsilon+\frac{1}{Q}\right).
\end{align*}
Taking $Q$ large enough compared to $\varepsilon^{-1}$ and combining \eqref{eq_first ramsey aprroximation} and \eqref{eq_second ramsey approximation}, this proves that
\begin{align*}
L&\geqslant \lim_{(\ell_2,M_2,\ell_1,M_1)\to\infty}
\mathbb{E}_{a\in B_{\ell_2}\cap \Phi_{M_2}}  \mathbb{E}_{m\in B_{\ell_1}\cap \Phi_{M_1}}\int 
F_a\cdot F_{a,\mathrm{str}} \cdot F_m\cdot F_{m, \mathrm{str}}\, d\mu
\\
& \geqslant \Big(\lim_{(\ell,M)\to\infty}
\mathbb{E}_{m\in B_{\ell}\cap \Phi_{M}}\int F_m\cdot F_{m, \mathrm{str}}\, d\mu
\Big)^2+{\rm O}(\varepsilon).
\end{align*}Now note that if we decompose $F_m$ into its structured and uniform parts, we will have $$\int F_{m,\mathrm{unf}}\cdot F_{m,\mathrm{str}}\, d\mu=0$$by orthogonality. We deduce that 
\begin{align*}
L&\geqslant \Big(\lim_{(\ell,M)\to\infty}\mathbb{E}_{m\in B_{\ell}\cap \Phi_{M}}\int F_m\cdot F_{m, \mathrm{str}}\, d\mu
\Big)^2+{\rm O}(\varepsilon)
\\
& \geqslant \Big(\lim_{(\ell,M)\to\infty}
\mathbb{E}_{m\in B_{\ell}\cap\Phi_{M}}\int F_{m, \mathrm{str}}^2\, d\mu
\Big)^2+{\rm O}(\varepsilon)\\
&\geqslant \Big(\lim_{(\ell,M)\to\infty}
\mathbb{E}_{m\in B_{\ell}\cap \Phi_{M}}\int F_{m, \mathrm{str}}\, d\mu
\Big)^4+{\rm O}(\varepsilon),
\end{align*}where the last inequality is proven via iterated applications of Cauchy-Schwarz. Now, we observe that $\int F_m\,\,d\mu=\int F_{m,\mathrm{str}}\, d\mu$, which follows from the fact that the weak-mixing part of any bounded measurable function has integral equal to zero (as it is orthogonal to the constant functions which are $T$-invariant). Combining this with the lower bound in \eqref{eq_lower bound by delta/2}, we conclude that   
\begin{align*}
L
\geqslant 
\Big(\lim_{(\ell,M)\to\infty}
\mathbb{E}_{m\in B_{\ell}\cap \Phi_{M}}\int F_{m}\, d\mu
\Big)^4+{\rm O}(\varepsilon)\geqslant \delta^4+{\rm O}(\varepsilon).
\end{align*}
Adjusting $\varepsilon$ accordingly, the claimed result follows.
