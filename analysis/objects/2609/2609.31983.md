<!-- head-1: body_md -->
We use small regular letters (e.g. $a$) to denote scalars, small bold-face letters (e.g. ${\bm{a}}$) to denote vectors, and capital bold-face letters (e.g. ${\bm{A}}$) to denote matrices. For a matrix ${\bm{M}}$, we use $\sigma_i({\bm{M}})$ to denote its $i$th singular value, $\sigma_{\min}({\bm{M}})$ to denote its minimum singular value, and $\left\|{\bm{M}}\right\|_2,\left\|{\bm{M}}\right\|_F$ to denote its operator and Frobenius norm, respectively. For a square matrix ${\bm{M}}$, we use $\lambda_i({\bm{M}})$ to denote its $i$th eigenvalue and $\lambda_{\min}({\bm{M}})$ to denote its smallest eigenvalue. For a symmetric matrix ${\bm{M}}$, we use $\text{span}({\bm{M}})$ to denote the row/column space of ${\bm{M}}$.

\label{thm:subspace_change}
    Let ${\bm{P}}\left(t\right) = {\bm{P}}\left(\bm{\theta}(t)\right)$ be defined in (\ref{eq:subspace_proj}). Consider the parameter dynamic induced by gradient flow on the loss $\mathcal{L}\left(\bm{\theta}\right)$, as given in (\ref{eq:gf}). Then we have that
    \[
        \left\|\dot{{\bm{P}}}(t)\right\|_F^2 \leq \frac{4}{\sigma_{\min}\left({\bm{G}}(t)\right)^2}\left((C-1)\left\|\left({\bm{I}} - {\bm{P}}(t)\right){\bm{H}}(t){\bm{g}}(t)\right\|_2^2 + \sum_{c=1}^{C-1}\left\|\left({\bm{I}}-{\bm{P}}(t)\right){\bm{H}}_c(t){\bm{g}}(t)\right\|_2^2\right)
    \]
    where ${\bm{H}}(t) = {\bm{H}}\left(\bm{\theta}(t)\right)$ and ${\bm{H}}_c(t)$ is the Hessian restricted to class-$c$ samples at $\bm{\theta}(t)$
    \[  
        {\bm{H}}_c(t) := \frac{1}{n_c}\sum_{i=1}^{n_c}\nabla^2_{\bm{\theta}}\ell\left(f\left(\bm{\theta},{\bm{x}}_{i,c}\right),y_c\right),\text{ s.t. } {\bm{H}}(t) = \sum_{c=1}^C\frac{n_c}{n}{\bm{H}}_c(t)
    \]

<!-- theorem-2: body_md -->
\label{thm:hessian_proj} 
    Fix any $t\geq 0$ and let ${\bm{H}}(t) = {\bm{H}}, {\bm{H}}_c(t) = {\bm{H}}_c$, ${\bm{P}}(t) = {\bm{P}}$. Assume that $\lambda_{\min}\left({\bm{H}}_c\right)\geq -\nu$ for all $c\in[C]$. Let $\varepsilon \geq \left\|\left({\bm{I}} - {\bm{P}}\right){\bm{H}}\left({\bm{I}} - {\bm{P}}\right)\right\|_2$. Then we have that $\lambda_{\min}\left({\bm{H}}\right)\geq -\nu$, and
