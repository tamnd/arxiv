\label{lem:pullback}
Let $f\in L^2(\mathrm{T},W)$. Define $g=f\circ\Psi$. Then
\begin{equation}\label{eq:pullback_identity}
\int_{\mathrm{T}} |f(x,y)|^2 \, W(x,y)\,\mathrm{d} x\,\mathrm{d} y
=
\int_0^1\int_0^1 |g(s,t)|^2\, s^{\gamma}(1-s)^\delta\, t^{\alpha}(1-t)^{\beta}\,\mathrm{d} s\,\mathrm{d} t.
\end{equation}
<!-- end lem-1 -->
<!-- proof-1: body_md -->
If $x=(2-t)s$, $y=t$, then $2-x-y=(2-t)(1-s)$ and $2-y=2-t$.  Thus,
\begin{align*}
W(\Psi(s,t))
&=
\big((2-t)s\big)^{\gamma}\big((2-t)(1-s)\big)^\delta\ t^{\alpha}(1-t)^{\beta}\ (2-t)^{-({\gamma}+\delta+1)}
\\&=
(2-t)^{-1}s^{\gamma}(1-s)^\delta t^{\alpha}(1-t)^{\beta}.
\end{align*}
Since $\mathrm{d} x \, \mathrm{d} y=(2-t)\,\mathrm{d} s \, \mathrm{d} t$, we get $W(\Psi(s,t))\,\mathrm{d} x\,\mathrm{d} y = s^{\gamma}(1-s)^\delta t^{\alpha}(1-t)^{\beta}\,\mathrm{d} s\,\mathrm{d} t$,
which yields \eqref{eq:pullback_identity}.

<!-- prop-1: body_md -->
\label{prop:energy_square_bilinear}
For $f,g\in C^2(Q)$,
\begin{equation}\label{eq:energy_square_bilinear}
{\langle} {\mathcal L} g,f{\rangle}_{L^2(Q,{\mathcal W})}
=
\int_Q
\Big(
s(1-s)(\partial_s g)(\partial_s f)
+
t(1-t)(\partial_t g)(\partial_t f)
\Big){\mathcal W}(s,t)\,\mathrm{d} s\,\mathrm{d} t.
\end{equation}
Consequently,
\begin{equation}\label{eq:symmetry_square}
{\langle} {\mathcal L} g,f{\rangle}_{L^2(Q,{\mathcal W})}
=
{\langle} g,{\mathcal L} f{\rangle}_{L^2(Q,{\mathcal W})},
\end{equation}
and the differential expression ${\mathcal L}$ is formally symmetric and nonnegative.
<!-- end prop-1 -->
<!-- proof-2: body_md -->
We integrate by parts separately in the two variables. For the $s$-part,
\begin{align*}
{\langle} {\mathcal J}_s(w_{{\gamma},\delta},g(\bullet,t)),f{\rangle}_{L^2(Q,{\mathcal W})}
&=
-\int_0^1\int_0^1
\partial_s\!\left(
w_{{\gamma},\delta}(s)s(1-s)\partial_s g
\right)
f\,w_{{\alpha},{\beta}}(t)\,\mathrm{d} s\,\mathrm{d} t
\\
&=
\int_Q
s(1-s)(\partial_s g)(\partial_s f){\mathcal W}(s,t)\,\mathrm{d} s\,\mathrm{d} t.
\end{align*}
The boundary terms vanish because all endpoint exponents are positive and the functions and their first derivatives are bounded. The same argument applies to the $t$-part. Summing the two identities gives
\eqref{eq:energy_square_bilinear}; symmetry and nonnegativity follow immediately.

<!-- prop-2: body_md -->
\label{prop:2d}
For $n\in{\mathbb N}_0$ and $0\le k\le n$, define
\begin{equation}\label{eq:Phi_mk}
\Phi_k^n(s,t)
:=
\mathsf P_k^{({\gamma},\delta)}(s)
\mathsf P_{n-k}^{({\alpha},{\beta})}(t).
\end{equation}
Then $\{\Phi_k^n:0\le k\le n\}$ is a mutually orthogonal basis of
${\mathcal V}_n(Q,{\mathcal W})$, and
\begin{equation}\label{eq:2d_eigs}
{\mathcal L}\Phi_k^n
=
\Lambda_{n,k}\Phi_k^n,
\qquad
\Lambda_{n,k}
=
k(k+{\gamma}+\delta+1)
+
(n-k)(n-k+{\alpha}+{\beta}+1).
\end{equation}
<!-- end prop-2 -->
<!-- proof-3: body_md -->
The eigenvalue relation follows directly from \eqref{eq:1d_eigs}, since the two
one-dimensional Jacobi expressions act in separate variables. The orthogonality
and completeness are the standard product-weight construction; see
\cite[Proposition~2.2.1]{DunklXu}.

\label{BforSq}
Let $n\in{\mathbb N}$. Then every polynomial $g\in\Pi_n$ satisfies
\begin{equation}\label{eq:Bernstein_square_energy}
\|\sqrt{s(1-s)}\,\partial_s g\|^2_{L^2(Q,{\mathcal W})}
+
\|\sqrt{t(1-t)}\,\partial_t g\|^2_{L^2(Q,{\mathcal W})}
\le
M(n,{\mathcal W})\|g\|^2_{L^2(Q,{\mathcal W})},
\end{equation}
where
\[
M(n,{\mathcal W})
=
\max\{\Lambda_{n,k}:0\le k\le n\}
=
n\bigl(n+1+\max\{\alpha+\beta,\gamma+\delta\}\bigr).
\]
The constant $M(n,{\mathcal W})$ is sharp. Equality is attained by $\Phi_0^n$ if
$\alpha+\beta>\gamma+\delta$, by $\Phi_n^n$ if
$\alpha+\beta<\gamma+\delta$, and by every nonzero element of
$\operatorname{span}\{\Phi_0^n,\Phi_n^n\}$ if
$\alpha+\beta=\gamma+\delta$.

Moreover, if $g$ is a polynomial such that $\deg_s g\le n$, then
\begin{equation}\label{eq:Bernstein_square_s}
\|\sqrt{s(1-s)}\,\partial_s g\|_{L^2(Q,{\mathcal W})}^2
\le
n(n+{\gamma}+\delta+1)\|g\|_{L^2(Q,{\mathcal W})}^2.
\end{equation}
Similarly, if $\deg_t g\le n$, then
\begin{equation}\label{eq:Bernstein_square_t}
\|\sqrt{t(1-t)}\,\partial_t g\|_{L^2(Q,{\mathcal W})}^2
\le
n(n+{\alpha}+{\beta}+1)\|g\|_{L^2(Q,{\mathcal W})}^2.
\end{equation}
<!-- end thm-1 -->
<!-- proof-4: body_md -->
Expand
\[
g=\sum_{j=0}^n\sum_{k=0}^j c_{j,k}\Phi_k^j.
\]
By \eqref{eq:energy_square_bilinear} and \eqref{eq:2d_eigs},
\[
\langle {\mathcal L} g,g\rangle_{L^2(Q,{\mathcal W})}
=
\sum_{j=0}^n\sum_{k=0}^j
\Lambda_{j,k}|c_{j,k}|^2\|\Phi_k^j\|_{L^2(Q,{\mathcal W})}^2.
\]
Since $\Lambda_{j,k}$ is convex in $k$, its maximum over $0\le k\le j\le n$ is attained at an endpoint, which gives
\[
\Lambda_{j,k}\le n\bigl(n+1+\max\{\alpha+\beta,\gamma+\delta\}\bigr)=M(n,{\mathcal W}).
\]
Together with \eqref{eq:energy_square_bilinear}, this proves
\eqref{eq:Bernstein_square_energy}. The equality cases follow from the
endpoint eigenfunctions described above.

Finally, \eqref{eq:Bernstein_square_s} and \eqref{eq:Bernstein_square_t}
follow by applying the one-dimensional sharp Bernstein inequality
\eqref{eq:1d_bernstein} to one variable at a time and integrating in the
other variable.

\label{prop:energy_D}
Let $f,h\in C^2(\mathrm{T})$, and set $g=f\circ\Psi$ and
$q=h\circ\Psi$. Then
\begin{equation}\label{eq:conjugation_identity}
{\langle} \widetilde{{\mathcal L}}f,h{\rangle}_{L^2(\mathrm{T},W)}
=
{\langle} {\mathcal L} g,q{\rangle}_{L^2(Q,{\mathcal W})}.
\end{equation}
Consequently, by \eqref{eq:symmetry_square}, $\widetilde{{\mathcal L}}$ is symmetric and
nonnegative on $C^2(\mathrm T)$ with respect to the weighted inner product.
Moreover,
\begin{align}\label{eq:energy_D}
{\langle} \widetilde{{\mathcal L}}f,f{\rangle}_{L^2(\mathrm{T},W)}
=&
\int_\mathrm{T} |\Phi_x(x,y) \partial_x f(x,y)|^2\,W(x,y)\,\mathrm{d} x\,\mathrm{d} y
\\&+
\int_\mathrm{T} |\Phi_y(x,y) D_y f(x,y)|^2\,W(x,y)\,\mathrm{d} x\,\mathrm{d} y, \nonumber
\end{align}
where we recall $D_y f=\partial_y f-\frac{x}{2-y}\,\partial_x f$.
<!-- end prop-3 -->
<!-- proof-5: body_md -->
Identity \eqref{eq:conjugation_identity} follows from the isometry
\eqref{eq:pullback_identity} and the definition \eqref{eq:L_def}. Together
with \eqref{eq:symmetry_square}, it gives the symmetry of $\widetilde{{\mathcal L}}$.
To derive \eqref{eq:energy_D}, set $h=f$, so that $q=g$, and use
\eqref{eq:energy_square_bilinear}:
\[
\langle {\mathcal L} g,g\rangle_{L^2(Q,{\mathcal W})}
=
\int_Q
\Big(
s(1-s)\,|\partial_s g(s,t)|^2
+
t(1-t)\,|\partial_t g(s,t)|^2
\Big){\mathcal W}(s,t)\,\mathrm{d} s\,\mathrm{d} t.
\]
By \eqref{eq:Phi_pullback}, \eqref{eq:derivs_st}, and
\eqref{lem:Dy_is_dt}, we have
\[
s(1-s) |\partial_s g|^2
=
(\Phi_x(\Psi))^2 |(\partial_x f)\circ\Psi|^2,
\]
and
\[
t(1-t) |\partial_t g|^2
=
(\Phi_y(\Psi))^2 |(D_y f)\circ\Psi|^2.
\]
Multiplying by ${\mathcal W}(s,t)\,\mathrm{d} s\,\mathrm{d} t$ and using
\eqref{eq:pullback_identity} gives \eqref{eq:energy_D}. Nonnegativity follows
from the right-hand side of \eqref{eq:energy_D}.

\label{prop:div_form}
For $f \in C^2(\mathrm{T})$, we have
\begin{align}\label{eq:div_form}
\widetilde{{\mathcal L}} f=&
-\frac{1}{x^{\gamma}(2-x-y)^\delta}\,\partial_x\Big(x^{{\gamma}+1}(2-x-y)^{\delta+1}\,\partial_x f\Big)
\\&-\frac{1}{y^{\alpha}(1-y)^{\beta}}\,D_y\Big(y^{{\alpha}+1}(1-y)^{{\beta}+1}\,D_y f\Big). \nonumber
\end{align}
<!-- end prop-4 -->
<!-- proof-6: body_md -->
Let $g(s,t)=(f\circ\Psi)(s,t)=f((2-t)s,t)$. By definition,
\[
(\widetilde{{\mathcal L}} f)\circ\Psi={\mathcal L}(f\circ\Psi)={\mathcal J}_s\!\left(w_{{\gamma},\delta},g(\bullet,t)\right) +{\mathcal J}_t\!\left(w_{{\alpha},{\beta}},g(s,\bullet)\right).
\]
We treat ${\mathcal J}_s\!\left(w_{{\gamma},\delta},g(\bullet,t)\right)$ first. Using \eqref{eq:Js},
\[
{\mathcal J}_s\!\left(w_{{\gamma},\delta},g(\bullet,t)\right)=-\frac{1}{w_{{\gamma},\delta}(s)}\partial_s\big(w_{{\gamma},\delta}(s)\,s(1-s)\,\partial_s g(s,t)\big).
\]
Since
$
\partial_s g(s,t)
=
(2-t)(\partial_x f \circ\Psi )(s,t),
$
\[
s^{\gamma+1}(1-s)^{\delta+1}\partial_s g(s,t)
=
(2-t)\,
s^{\gamma+1}(1-s)^{\delta+1}
(\partial_x f \circ\Psi )(s,t).
\]
Let $\tilde{f}(x,y):=(2-y)\frac{x^{\gamma+1}(2-x-y)^{\delta+1}}{(2-y)^{\gamma+\delta+2}} \partial_x f(x,y)$. Thus
\[
s^{\gamma+1}(1-s)^{\delta+1}\partial_s g(s,t)
= (\tilde{f} \circ\Psi)(s,t).
\]
Differentiating once more with respect to $s$, we obtain
\[
\partial_s
\left(
s^{\gamma+1}(1-s)^{\delta+1}\partial_s g
\right)
=
(2-t)
\Big[
\partial_x
\Big(
\frac{x^{\gamma+1}(2-x-y)^{\delta+1}}{(2-y)^{\gamma+\delta+1}} \partial_x f
\Big)
\Big]\circ\Psi .
\]
Substituting into the definition of the $s$-component yields
\begin{align*}
{\mathcal J}_s\!\left(w_{{\gamma},\delta},g(\bullet,t)\right)
&=
-(2-t)
\frac{1}{s^\gamma(1-s)^\delta}
\Big[
\partial_x
\Big(
\frac{x^{\gamma+1}(2-x-y)^{\delta+1}}{(2-y)^{\gamma+\delta+1}} \partial_x f
\Big)
\Big]\circ\Psi \\ &= \left[-\frac{1}{x^{\gamma}(2-x-y)^\delta}\,\partial_x\Big(x^{{\gamma}+1}(2-x-y)^{\delta+1}\,\partial_x f \Big) \right] \circ\Psi.
\end{align*}
Hence
\[
\left[{\mathcal J}_s\!\left(w_{{\gamma},\delta},g(\bullet,t)\right)\right]\circ\Psi^{-1}(x,y)
=
-\frac{1}{x^{\gamma}(2-x-y)^\delta}\,\partial_x\Big(x^{{\gamma}+1}(2-x-y)^{\delta+1}\,\partial_x f\Big).
\]
Similarly, using \eqref{eq:Js} and \eqref{lem:Dy_is_dt}, we obtain
\[
\left[{\mathcal J}_t\!\left(w_{{\alpha},{\beta}},g(s,\bullet)\right)\right]\circ\Psi^{-1}(x,y)
=
-\frac{1}{y^{\alpha}(1-y)^{\beta}}\,D_y\Big(y^{{\alpha}+1}(1-y)^{{\beta}+1}\,D_y f\Big).
\]
This completes the proof of \eqref{eq:div_form}.

\label{thm:diag_D}
The family $\{{\mathbb P}^n_k : 0 \leq k \leq n\}$ is a mutually orthogonal basis of ${\mathcal V}_n(\mathrm T,W)$.
Moreover,
\begin{align*}
\widetilde{{\mathcal L}} {\mathbb P}^n_0=n(n+{\alpha} + {\beta} + 1)\,{\mathbb P}^n_0.
\end{align*}
<!-- end prop-5 -->
<!-- proof-7: body_md -->
By construction,
$
{\mathbb P}^n_0(\Psi(s,t))=\mathsf{P}_0^{({\gamma},\delta)}\!\left(s\right)\,
\mathsf{P}_{n,0}^{({\alpha},{\beta})}(t)
$.
Then, using \eqref{eq:Js} and Proposition  \ref{prop:2d},
\begin{align*}
(\widetilde{{\mathcal L}} {\mathbb P}^n_{0})\circ\Psi= {\mathcal L}({\mathbb P}^n_{0}\circ\Psi)=\mathsf{P}_{0}^{({\gamma},\delta)}(s) \,{\mathcal J}_t \left(w_{{\alpha},{\beta}},\mathsf{P}_{n,0}^{({\alpha},{\beta})}\right)=\lambda^{({\alpha},{\beta})}_n
{\mathbb P}^n_{0}\circ\Psi.
\end{align*}
Orthogonality and completeness follow from Proposition 2.6.1 of \cite{DunklXu}.
<!-- end proof-7 -->
<!-- rem-1: body_md -->
In general, the polynomials
${\mathbb P}_k^n$ are not eigenfunctions of $\widetilde{{\mathcal L}}$. The endpoint $k=0$ is an
exception, since $\mathsf P_{n,0}^{({\alpha},{\beta})}$ is, up to normalization, the
shifted Jacobi polynomial $\mathsf P_n^{({\alpha},{\beta})}$.
For $k=n$, however,
\[
({\mathbb P}_n^n\circ\Psi)(s,t)
=
c_n(2-t)^n\mathsf P_n^{({\gamma},\delta)}(s),
\]
where $c_n\ne0$ is a constant, and the $t$-part of ${\mathcal L}$ acts nontrivially
on $(2-t)^n$. Consequently, ${\mathbb P}_n^n$ is not an eigenfunction of
$\widetilde{{\mathcal L}}$ for $n\ge1$.

More generally, $\widetilde{{\mathcal L}}$ does not preserve the algebraic polynomial spaces
$\Pi_n$. For instance,
\begin{equation*}
\widetilde{{\mathcal L}}x
=
({\gamma}+\delta+2)x-({\gamma}+1)(2-y)
+
\frac{x}{2-y}
\big({\alpha}+1-({\alpha}+{\beta}+2)y\big),
\end{equation*}
which is not a polynomial for ${\alpha},{\beta}>-1$. Thus, unlike on the simplex and
the unit ball, the sharp Bernstein inequalities below do not arise from a spectral decomposition of $\widetilde{{\mathcal L}}$.

The two endpoint elements ${\mathbb P}_0^n$ and ${\mathbb P}_n^n$ of the orthogonal basis
\eqref{eq:Pmk} will nevertheless play a distinguished role. Their behavior
with respect to $\widetilde{{\mathcal L}}$ is essentially different: ${\mathbb P}_0^n$ is an
eigenfunction of $\widetilde{{\mathcal L}}$, whereas ${\mathbb P}_n^n$ not an eigenfunction for $n \geq 1$.
Nevertheless, both endpoint polynomials turn out to be extremizers:
${\mathbb P}_0^n$ for the inequality involving $\Phi_yD_y$ and ${\mathbb P}_n^n$ for the
inequality involving $\Phi_x\partial_x$. Hence the sharpness of the two
inequalities cannot be attributed solely to the spectral properties of
$\widetilde{{\mathcal L}}$. Instead, the extremal behavior is inherited separately from
the one-dimensional Jacobi structure in the $t$- and $s$-variables,
respectively, under the parametrization $\Psi$.

<!-- prop-6: body_md -->
\label{prop:bernstein_T}
Let ${\alpha},{\beta},{\gamma},\delta>-1$ and $n\in{\mathbb N}$. Then, for every $f\in\Pi_n$,
\begin{align}
\|\Phi_x\partial_x f\|_{L^2(\mathrm{T},W)}^2
&\le
n(n+{\gamma}+\delta+1)
\|f\|_{L^2(\mathrm{T},W)}^2,
\label{eq:bernstein_T_x}
\\[0.5em]
\|\Phi_yD_y f\|_{L^2(\mathrm{T},W)}^2
&\le
n(n+{\alpha}+{\beta}+1)
\|f\|_{L^2(\mathrm{T},W)}^2.
\label{eq:bernstein_T_y}
\end{align}
Both constants are sharp. Equality in \eqref{eq:bernstein_T_x} is attained
by ${\mathbb P}_n^n$, whereas equality in \eqref{eq:bernstein_T_y} is attained
by ${\mathbb P}_0^n$.
<!-- end prop-6 -->
<!-- proof-8: body_md -->
Let $g=f\circ\Psi$. Since $f\in\Pi_n$, we have
$
\deg_s g\le n,
\deg_t g\le n.
$
By \eqref{eq:Phi_pullback}, \eqref{eq:derivs_st}, \eqref{lem:Dy_is_dt}, and the pullback identity,
\[
\|\Phi_x\partial_x f\|_{L^2(\mathrm{T},W)}^2
=
\|\sqrt{s(1-s)}\,\partial_s g\|_{L^2(Q,{\mathcal W})}^2
\]
and
\[
\|\Phi_yD_y f\|_{L^2(\mathrm{T},W)}^2
=
\|\sqrt{t(1-t)}\,\partial_t g\|_{L^2(Q,{\mathcal W})}^2.
\]
The two inequalities now follow directly from
\eqref{eq:Bernstein_square_s} and \eqref{eq:Bernstein_square_t},
respectively.

It remains to verify sharpness. From \eqref{eq:Pmk},
\[
({\mathbb P}_n^n\circ\Psi)(s,t)
=
(2-t)^n
\mathsf P_n^{({\gamma},\delta)}(s)
\mathsf P_{0,n}^{({\alpha},{\beta})}(t).
\]
Since $\mathsf P_{0,n}^{({\alpha},{\beta})}$ is a nonzero constant, for every fixed
$t$ the dependence on $s$ is a scalar multiple of
$\mathsf P_n^{({\gamma},\delta)}(s)$. Equality therefore holds in the one-dimensional Bernstein inequality
for Jacobi polynomials used above, and hence in \eqref{eq:bernstein_T_x}. Thus the constant
$n(n+{\gamma}+\delta+1)$ is sharp. Similarly,
$
({\mathbb P}_0^n\circ\Psi)(s,t)
=
\mathsf P_0^{({\gamma},\delta)}(s)
\mathsf P_{n,0}^{({\alpha},{\beta})}(t).
$
For $k=0$, the weight defining $\mathsf P_{n,0}^{({\alpha},{\beta})}$ is simply
$w_{{\alpha},{\beta}}$; hence $\mathsf P_{n,0}^{({\alpha},{\beta})}$ is, up to normalization,
the shifted Jacobi polynomial of degree $n$. Therefore equality is attained in the one-dimensional sharp Bernstein inequality for Jacobi polynomials in the \(t\)-variable, and consequently also in \eqref{eq:bernstein_T_y}. This proves the sharpness of the constant \(n(n+{\alpha}+{\beta}+1)\).

<!-- cor-1: body_md -->
\label{cor}
Let ${\alpha},{\beta},{\gamma},\delta>-1$ and $n\in{\mathbb N}$. Define
\[
\widetilde W(x,y)
:=
(2-y)W_{{\alpha},{\beta},{\gamma},\delta}(x,y)
=
x^{\gamma}(2-x-y)^\delta y^{\alpha}(1-y)^{\beta}
(2-y)^{-({\gamma}+\delta)}
\]
and
\[
\widetilde\Phi_x(x,y)
:=
\frac{\Phi_x(x,y)}{\sqrt{2-y}}
=
\sqrt{\frac{x(2-x-y)}{2-y}}.
\]
Then, for every $f\in\Pi_n$,
\begin{equation}\label{DxT}
\|\widetilde\Phi_x\partial_xf\|_{L^2(\mathrm{T},\widetilde W)}
\le
\sqrt{n(n+{\gamma}+\delta+1)}\,
\|\widetilde f\|_{L^2(\mathrm{T},\widetilde W)},
\end{equation}
where
\[
\widetilde f(x,y):=\frac{f(x,y)}{\sqrt{2-y}}.
\]
The constant is sharp, and equality is attained for $f={\mathbb P}_n^n$.
<!-- end cor-1 -->
<!-- rem-2: body_md -->
It is worth noting that the sharpness of \eqref{DxT}  is obtained
after the factor $(2-y)^{-1/2}$ is incorporated into the function. If one
keeps the polynomial $f$ itself on the right-hand side, the same argument
still yields a Bernstein inequality, but it does not determine its optimal
constant.

Indeed, since
$
\frac{1}{\sqrt{2}}\,
\|f\|_{L^2(\mathrm{T},\widetilde W)}
\le
\|\widetilde f\|_{L^2(\mathrm{T},\widetilde W)}
\le
\|f\|_{L^2(\mathrm{T},\widetilde W)},
$
Corollary~\ref{cor} gives
\[
\sqrt{\frac{n(n+{\gamma}+\delta+1)}{2}}
\le
\sup_{\substack{f\in\Pi_n\\ f\neq0}}
\frac{
\|\widetilde\Phi_x\partial_xf\|_{L^2(\mathrm{T},\widetilde W)}
}{
\|f\|_{L^2(\mathrm{T},\widetilde W)}
}
\le
\sqrt{n(n+{\gamma}+\delta+1)}.
\]
Moreover, the extremal polynomial ${\mathbb P}_n^n$ from Corollary~\ref{cor}
satisfies
\[
\lim_{n\to\infty}
\frac{
\|\widetilde\Phi_x\partial_x{\mathbb P}_n^n\|_{L^2(\mathrm{T},\widetilde W)}
}{
\sqrt{n(n+{\gamma}+\delta+1)}
\|{\mathbb P}_n^n\|_{L^2(\mathrm{T},\widetilde W)}
}
=
\frac{1}{\sqrt2}.
\]
Thus, when measured against the norm of the polynomial itself, the extremizer
of the weighted formulation asymptotically attains the lower bound in the
above two-sided estimate. The last limit follows from
Corollary~\ref{cor} and the following lemma.

<!-- lem-2: body_md -->
\label{lem:weighted-extremizer-limit}
Let
\[
\widetilde{{\mathbb P}_n^{n}}(x,y):=\frac{{\mathbb P}_n^{n}(x,y)}{\sqrt{2-y}}.
\]
Then
\[
\lim_{n\to\infty}
\frac{\big\|\widetilde{{\mathbb P}_n^{n}}\big\|^2_{L^2(\mathrm{T},\widetilde W)}}
{\|{\mathbb P}_n^{n}\|^2_{L^2(\mathrm{T},\widetilde W)}}
=
\lim_{n\to\infty}
\frac{\displaystyle\int_0^1(2-y)^{2n}w_{{\alpha},{\beta}}(y)\,\mathrm{d} y}
{\displaystyle\int_0^1(2-y)^{2n+1}w_{{\alpha},{\beta}}(y)\,\mathrm{d} y}
=
\frac12.
\]
<!-- end lem-2 -->
<!-- proof-9: body_md -->
Using \eqref{eq:Pmk} and the change of variables \eqref{eq:Psi}, the factors
depending on $s$ cancel in the quotient of the two norms, which gives the
integral representation in the statement.

For fixed $\varepsilon\in(0,1)$,
\[
\frac12\le \frac{1}{2-y}\le \frac{1}{2-\varepsilon}
\qquad\text{for }y\in[0,\varepsilon],
\quad\text{and}\quad
\frac{1}{2-\varepsilon}\le \frac{1}{2-y}\le1
\qquad\text{for }y\in[\varepsilon,1].
\]
Set
\[
R_n:=
\frac{\big\|\widetilde{{\mathbb P}_n^{n}}\big\|^2_{L^2(\mathrm{T},\widetilde W)}}
{\|{\mathbb P}_n^{n}\|^2_{L^2(\mathrm{T},\widetilde W)}}.
\]
Then
\[
\frac12
\le R_n
\le
\frac{1}{2-\varepsilon}
+
\frac{\displaystyle\int_\varepsilon^1(2-y)^{2n}w_{{\alpha},{\beta}}(y)\,\mathrm{d} y}
{\displaystyle\int_0^1(2-y)^{2n+1}w_{{\alpha},{\beta}}(y)\,\mathrm{d} y}.
\]
For any fixed $\eta\in(0,\varepsilon)$,
\[
0\le
\frac{\displaystyle\int_\varepsilon^1(2-y)^{2n}y^{\alpha}(1-y)^{\beta}\,\mathrm{d} y}
{\displaystyle\int_0^1(2-y)^{2n+1}y^{\alpha}(1-y)^{\beta}\,\mathrm{d} y}
\le
\left(\frac{2-\varepsilon}{2-\eta}\right)^{2n}
\frac{\displaystyle\int_\varepsilon^1y^{\alpha}(1-y)^{\beta}\,\mathrm{d} y}
{\displaystyle\int_0^\eta(2-y)y^{\alpha}(1-y)^{\beta}\,\mathrm{d} y}
\longrightarrow0.
\]
Thus,
\[
\frac12
\le
\liminf_{n\to\infty}R_n
\le
\limsup_{n\to\infty}R_n
\le
\frac{1}{2-\varepsilon}.
\]
Letting $\varepsilon\downarrow0$, we conclude that $R_n\to\frac12$.

<!-- prop-7: body_md -->
\label{prop:bernstein_y}
Let ${\alpha} > -1$, and for $(x,y) \in \mathrm{T}$ let
\[
Y = Y_{{\alpha}}(x,y)=
\begin{cases}
y^{\alpha} (1-y), & \text{if } x\le 1,\\[4pt]
y^{\alpha} , & \text{if } 1 < x.
\end{cases}
\]
Define, for $(x,y) \in \mathrm{T}$,
\begin{equation*}

\label{lem:adjacent-jacobi-integral}
For \(\alpha>-1\) and \(n\ge 0\),
\[
\int_{-1}^1 (1-x)^\alpha \bigl(P_n^{(\alpha,1)}(x)\bigr)^2\, \mathrm{d} x
=
\frac{2^{\alpha+1}(n+1)}{n+\alpha+1}.
\]
<!-- end lem-3 -->
<!-- proof-11: body_md -->
We start from the adjacent-parameter relation for Jacobi polynomials
(see, e.g., \cite[(18.9.5)]{NIST}):
\[
(2m+\alpha+1)P_m^{(\alpha,0)}(x)
=
(m+\alpha+1)P_m^{(\alpha,1)}(x)
+
(m+\alpha)P_{m-1}^{(\alpha,1)}(x),
\qquad m\ge 1.
\]
Equivalently,
\[
(m+\alpha+1)P_m^{(\alpha,1)}(x)
=
(2m+\alpha+1)P_m^{(\alpha,0)}(x)
-
(m+\alpha)P_{m-1}^{(\alpha,1)}(x).
\]
Iterating this identity for \(m=n,\ldots,1\) and using $P_0^{(\alpha,0)}=P_0^{(\alpha,1)}=1$, one obtains
\[
(n+\alpha+1)P_n^{(\alpha,1)}(x)
=
\sum_{k=0}^n (-1)^{\,n-k}(2k+\alpha+1)P_k^{(\alpha,0)}(x).
\]
Now square this identity and integrate on \([-1,1]\) with the weight \((1-x)^\alpha\). Since the family
\(\{P_k^{(\alpha,0)}\}_{k\ge 0}\) is orthogonal with respect to \((1-x)^\alpha\), all cross terms vanish, and therefore
\[
(n+\alpha+1)^2
\int_{-1}^1 (1-x)^\alpha \bigl(P_n^{(\alpha,1)}(x)\bigr)^2\, \mathrm{d} x
=
\sum_{k=0}^n (2k+\alpha+1)^2 h_k^{(\alpha,0)},
\]
where
\[
h_k^{(\alpha,0)}
:=
\int_{-1}^1 (1-x)^\alpha \bigl(P_k^{(\alpha,0)}(x)\bigr)^2\, \mathrm{d} x.
\]
By the standard Jacobi norm formula,
\[
h_k^{(\alpha,0)}=\frac{2^{\alpha+1}}{2k+\alpha+1}.
\]
Hence
\[
(n+\alpha+1)^2
\int_{-1}^1 (1-x)^\alpha \bigl(P_n^{(\alpha,1)}(x)\bigr)^2\, \mathrm{d} x
=
2^{\alpha+1}\sum_{k=0}^n (2k+\alpha+1).
\]
Therefore,
\[
(n+\alpha+1)^2
\int_{-1}^1 (1-x)^\alpha \bigl(P_n^{(\alpha,1)}(x)\bigr)^2\, \mathrm{d} x
=
2^{\alpha+1}(n+1)(n+\alpha+1),
\]
Thus
\[
\int_{-1}^1 (1-x)^\alpha \bigl(P_n^{(\alpha,1)}(x)\bigr)^2\, \mathrm{d} x
=
\frac{2^{\alpha+1}(n+1)}{n+\alpha+1},
\]
which completes the proof.

\label{lem:jacobi-moments}
Let ${\alpha} > -1$. Then
\[
\int_{0}^{1} (1- y) \left|
\partial_y \mathsf{P}_n^{({\alpha},1)}(y)  \right|^2 y^{{\alpha}+2} \,  \mathrm{d} y = \frac{n(2n+{\alpha})(n+{\alpha}+2)}{2(2n+{\alpha}+2)} \int_{0}^{1} \left| \mathsf{P}_n^{({\alpha},1)}(y)\right|^2 y^{{\alpha}}  \, \mathrm{d} y,
\]
and
\[
\int_{0}^{1} \left| \mathsf{P}_n^{({\alpha},1)}(y)\right|^2 y^{{\alpha}+1}  \, \mathrm{d} y = \frac{2n+{\alpha}+1}{2n+{\alpha}+2}  \int_{0}^{1} \left| \mathsf{P}_n^{({\alpha},1)}(y)\right|^2 y^{{\alpha}}  \, \mathrm{d} y.
\]

Set
\[
Q_n(y):=P_n^{(\alpha,1)}(1-2y)=\mathsf P_n^{(\alpha,1)}(y),
\]
and define
\[
I_n:=\int_0^1 y^{\alpha+1} (Q_n(y))^2\, \mathrm{d} y,
\qquad
A_n:=\int_0^1 (1-y)y^\alpha (Q_n(y))^2\, \mathrm{d} y.
\]
Then
\[
D_n:=\int_0^1 y^\alpha Q_n(y)^2\, \mathrm{d} y=I_n+A_n.
\]

We compute \(A_n\) and \(D_n\) separately. Using the change of variables
$
x=1-2y,
$
we obtain
\[
A_n
=
2^{-(\alpha+2)}
\int_{-1}^1 (1-x)^\alpha(1+x)\bigl(P_n^{(\alpha,1)}(x)\bigr)^2\, \mathrm{d} x.
\]
Now we apply the standard Jacobi norm formula
\[
\int_{-1}^1 (1-x)^\alpha(1+x)^\beta
\bigl(P_n^{(\alpha,\beta)}(x)\bigr)^2\, \mathrm{d} x
=
\frac{2^{\alpha+\beta+1}}{2n+\alpha+\beta+1}
\frac{\Gamma(n+\alpha+1)\Gamma(n+\beta+1)}
{n!\,\Gamma(n+\alpha+\beta+1)}.
\]
With \(\beta=1\), this gives
\[
A_n
=
2^{-(\alpha+2)}
\cdot
\frac{2^{\alpha+2}}{2n+\alpha+2}
\frac{\Gamma(n+\alpha+1)\Gamma(n+2)}
{n!\,\Gamma(n+\alpha+2)}.
\]
Since
$
\Gamma(z+1)=z\Gamma(z),
$
we get
\[
A_n=\frac{n+1}{(2n+\alpha+2)(n+\alpha+1)}.
\]
By the change of variables \(x=1-2y\) and Lemma~\ref{lem:adjacent-jacobi-integral}, we have
\[
D_n
=
2^{-(\alpha+1)}
\int_{-1}^1 (1-x)^\alpha \bigl(P_n^{(\alpha,1)}(x)\bigr)^2\, \mathrm{d} x = 2^{-(\alpha+1)}\cdot \frac{2^{\alpha+1}(n+1)}{n+\alpha+1}
=
\frac{n+1}{n+\alpha+1}.
\]
Since \(D_n=I_n+A_n\), we have
\[
I_n=D_n-A_n =  \frac{n+1}{n+\alpha+1}
\left(1-\frac{1}{2n+\alpha+2}\right).
\]
Hence
\[
I_n
=
\frac{2n+\alpha+1}{2n+\alpha+2}\cdot
\frac{n+1}{n+\alpha+1}
=
\frac{2n+\alpha+1}{2n+\alpha+2}\,D_n.
\]
This proves the second identity. It remains to establish the first one. Since $Q_n=\mathsf P_n^{(\alpha,1)}$, the Jacobi differential equation gives
\[
-\frac{\mathrm{d}}{\mathrm{d} y}
\left(y^{\alpha+1}(1-y)^2Q_n'(y)\right)
=
\lambda_n^{({\alpha},1)} y^\alpha(1-y)Q_n(y).
\]
Multiplying by $\dfrac{y}{1-y}Q_n(y)$ and integrating by parts, we obtain
\[
\lambda_n^{({\alpha},1)} I_n
=
\int_0^1 y^{\alpha+2}(1-y)|Q_n'(y)|^2\,\mathrm{d} y
+
\int_0^1 y^{\alpha+1}Q_n(y)Q_n'(y)\,\mathrm{d} y.
\]
Moreover, using $Q_n(1)=P_n^{(\alpha,1)}(-1)=(-1)^n(n+1)$,
\[
\int_0^1 y^{\alpha+1}Q_n(y)Q_n'(y)\,\mathrm{d} y
=
\frac{(n+1)^2}{2}-\frac{\alpha+1}{2}D_n.
\]
Consequently,
\[
\int_0^1 y^{\alpha+2}(1-y)|Q_n'(y)|^2\,\mathrm{d} y
=
\lambda_n^{({\alpha},1)} I_n-\frac{(n+1)^2}{2}+\frac{\alpha+1}{2}D_n.
\]
Using the formulas for $I_n$ and $D_n$ obtained above and simplifying, we get
\[
\int_0^1 y^{\alpha+2}(1-y)|Q_n'(y)|^2\,\mathrm{d} y
=
\frac{n(2n+\alpha)(n+\alpha+2)}{2(2n+\alpha+2)}D_n,
\]
which proves the first identity and completes the proof.

where $Q=[0,1]^2$. On $Q$, the weight is
$\Upsilon=y^{\alpha}$, so the one-dimensional Jacobi inequality
in the $y$-variable has parameters $({\alpha},0)$ and gives

\begin{align*}

&\le
n(n+{\alpha}+1)\|f\|_{L^2(Q,\Upsilon)}^2
+
n(n+{\alpha}+2)\|f\|_{L^2(\triangle,\Upsilon)}^2
\\
&\le
n(n+{\alpha}+2)\|f\|_{L^2(\mathrm T,\Upsilon)}^2.
\end{align*}
For the lower bound, take
$

integration with respect to $x$ gives
\begin{align*}
\|\widetilde\Phi_y\partial_y Q_n\|_{L^2(\mathrm T,\Upsilon)}^2
&=
\int_0^1 y^{{\alpha}+1}
\bigl(2(1-y)+y\log y\bigr)|Q_n'(y)|^2\,\mathrm{d} y.
\end{align*}
For $0<y\le1$,
\[
\log y\ge \frac12\left(y-\frac1y\right),
\]
and hence
\[
2(1-y)+y\log y
\ge
\frac32(1-y)^2+y(1-y).
\]
Therefore,
\begin{align*}
\|\widetilde\Phi_y\partial_y Q_n\|_{L^2(\mathrm T,\Upsilon)}^2
\ge
\frac32
\int_0^1y^{{\alpha}+1}(1-y)^2|Q_n'(y)|^2\,\mathrm{d} y
+
\int_0^1y^{{\alpha}+2}(1-y)|Q_n'(y)|^2\,\mathrm{d} y.
\end{align*}

\[
D_n:=\int_0^1 (Q_n(y))^2 y^{\alpha}\,\mathrm{d} y.
\]

\[
\int_0^1 y^{{\alpha}+1}(1-y)^2|Q_n'(y)|^2\,\mathrm{d} y
=
\lambda_n^{({\alpha},1)} \int_0^1 |Q_n(y)|^2 y^{\alpha}(1-y)\,\mathrm{d} y
=
\frac{\lambda_n^{({\alpha},1)}}{2n+{\alpha}+2}D_n.
\]
Moreover, by Lemma~\ref{lem:jacobi-moments},
\[
\int_0^1y^{{\alpha}+2}(1-y)|Q_n'(y)|^2\,\mathrm{d} y
=
\frac{\lambda_n^{({\alpha},1)} (2n+{\alpha})}{2(2n+{\alpha}+2)}D_n.

\|\widetilde\Phi_y\partial_y Q_n\|_{L^2(\mathrm T,\Upsilon)}^2
\ge
\frac{\lambda_n^{({\alpha},1)} (2n+{\alpha}+3)}{2(2n+{\alpha}+2)}D_n.

\begin{align*}

=
\int_0^1(2-y) |Q_n(y)|^2 y^{\alpha}\,\mathrm{d} y
=
\frac{2n+{\alpha}+3}{2n+{\alpha}+2}D_n.
\end{align*}

\ge
