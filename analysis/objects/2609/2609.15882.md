<!-- theorem-1: body_md -->
\label{thmmain} 
		Let $\lambda > 0$ and let $N, p, q$ satisfy one of the following assumptions: 
		\begin{itemize} 
			\item[(1)] $N \ge 5$, $1 < q < p < 2^* -1$; 
			\item[(2)] $N = 2,3,4$, $1 < q < p \le p_{\text{alg}}(N) := \frac{4 + 2\sqrt{2N + 4}}{N}$; 
			\item[(3)] $N = 2,3,4$, $1 < q < p < 2^* -1$, $p - q \le 1$. 
		\end{itemize} 
		Then \eqref{doublepower} admits a unique solution up to translations.

We are preparing a paper to prove it rigorously.	Once it is proved, in order to obtain uniqueness result of \eqref{doublepower} in dimension $2$, additional assumptions on $p$ and $q$ are necessary.

\label{thmnondegenerate} 
			Let $\lambda > 0$, $N \ge 2$ and $1 < q < p < 2^*-1$. Assume that either one of the conditions in Theorem \ref{thmmain} holds or 
			\begin{align*} 
				N(p-1) \le 2(q+1). 
			\end{align*} 
		Then every radial solution of \eqref{doublepower} having Morse index $1$ is nondegenerate in $H^1_{rad}(\mathbb{R}^N)$ and nondegenerate in $H^1(\mathbb{R}^N)$.

<!-- remark-2: body_md -->
\label{rmkstability} 
		Define the energy functional 
		\begin{align} 
			I(u) := \frac{1}{2}\int_{\mathbb{R}^N}|\nabla u|^2dx - \frac{1}{p+1}\int_{\mathbb{R}^N}|u|^{p+1}dx - \frac{1}{q+1}\int_{\mathbb{R}^N}|u|^{q+1}dx. 
		\end{align} 
		When $1 < q < p < 1+4/N$ and $c > 0$, Lions showed in \cite{Lions} that the set 
		\begin{align*} 
			M := \big\{u \in S_c: I(u) = e_0\big\}, \quad e_0 := \inf_{S_c}I 
		\end{align*} 
		is not empty. The same result holds when $1 < q < p = 1+4/N$ and $c \in (0,c_0)$. It is clear that any element in $M$ is a positive (multiplying $-1$ if it is negative) solution of \eqref{eqmass}. Let 
		\begin{align} 
			M_0 := \big\{u \in H^1(\mathbb{R}^N;\mathbb{C}): \int_{\mathbb{R}^N}|u|^2dx = c, I(u) = e_0\big\}. 
		\end{align} 
		By classical arguments developed by Cazenave and Lions in \cite{CLions}, the set $M_0$ is orbitally stable. As an application of Theorem \ref{thmmain2}, we get 
		\begin{align*} 
			M_0 = \big\{e^{i\theta} u_c(\cdot + y): \theta \in \mathbb{R}, y \in \mathbb{R}^N \big\}, 
		\end{align*} 
		and any standing wave solution corresponding to an element in $M_0$ is orbitally stable, where $u_c$ is the unique radial positive solution given in Theorem \ref{thmmain2}.

\label{lemestimate} 
		Suppose $N \ge 2$ and $1 < q < p < 2^* -1$. Then 
		\begin{align} \label{usefulestimate1}

		for $r > 0$.

<!-- proof-1: body_md -->
Let 
		\begin{align*} 
			\mathcal{P}_0 = r^N\left(v_r^2 + 2\mathcal{F}(v)\right) + (N-2)r^{N-1}vv_r 
		\end{align*} 
		where 
		\begin{align*} 
			\mathcal{F}(t) = -\frac{\eta}{2}t^2 + \frac{1-\theta}{p+1}t^{p+1} + \frac{\theta}{q+1}t^{q+1}. 
		\end{align*} 
		From direct computations we derive that 
		\begin{align*} 
			\mathcal{P}_0'(r) = r^{N-1}v^2\left(-2\eta + (1-\theta)\frac{N+2-p(N-2)}{p+1}v^{p-1} + \theta\frac{N+2-q(N-2)}{q+1}v^{q-1}\right). 
		\end{align*} 
		Since $v(r)$ is decreasing in $r > 0$ and $q < p < 2^* -1$, the function 
		\begin{align*} 
			-2\eta + (1-\theta)\frac{N+2-p(N-2)}{p+1}v(r)^{p-1} + \theta\frac{N+2-q(N-2)}{q+1}v(r)^{q-1} 
		\end{align*} 
		is decreasing in $r > 0$ and has at most one zero point. Furthermore, using $\mathcal{P}_0(0) = 0$ and $\mathcal{P}_0(r) \to 0$ as $r \to \infty$, we obtain $\mathcal{P}_0(r) > 0$ for all $r > 0$. On the other hand, 
		\begin{align*} 
			\mathcal{P}_0 = r^{N-2}v^2\left(ry_r - r^2J\right). 
		\end{align*} 
		Therefore, 
		\begin{align*} 
			y_r > rJ > 0. 
		\end{align*} 
 
		Next we prove $h_r < 0$. Since $v(0) = 1$, we have 
		\begin{align*} 
			h(0) = - \lim_{r \to 0}\frac{v_r(r)}{r} = -v_{rr}(0) =: h_0 > 0. 
		\end{align*} 
		Using \eqref{eqofv} we get 
		\begin{align*} 
			-N v_{rr}(0) = (1-\theta)v(0)^{p} + \theta v(0)^{q} - \eta v(0) = 1 - \eta. 
		\end{align*} 
		Thus $h_0 = (1-\eta)/N$. By the regularity and radial symmetry of $v(x)$, we can write 
		\begin{align*} 
			h(r) = h_0 + \mu r^2 + o(r^2), \quad h_r(r) = 2\mu r + o(r). 
		\end{align*} 
		Using $\xi_r = rh$, we get 
		\begin{align*} 
			\xi(r) = \int_0^r sh(s)ds = \int_0^r s\left(h_0 + \mu s^2 + o(s^2)\right)ds = \frac{h_0}{2}r^2 + O(r^4). 
		\end{align*} 
		Moreover, $F_\xi(0) = -P(0)$. Thus 
		\begin{align*} 
			F(\xi) = F(0) + F_\xi(0)\xi + o(\xi) = 1 - P(0)\xi  + o(\xi) = 1 - \frac{h_0P(0)}{2}r^2 + o(r^2). 
		\end{align*} 
		By $v_r/v = -rh$, we have 
		\begin{align*} 
			\frac{v_{rr}}{v} = -h - rh_r + r^2h^2. 
		\end{align*} 
		On the other hand, 
		\begin{align*} 
			-v_{rr} - \frac{N-1}{r}v_r + \eta v = F(\xi)v. 
		\end{align*} 
		Therefore, we can derive that 
		\begin{align} \label{eqofh_r} 
			rh_r = r^2h^2 + F -\eta - Nh. 
		\end{align} 
		Note that 
		\begin{align*} 
			& rh_r = r(2\mu r + o(r)) = 2\mu r^2 + o(r^2), \\ 
			& r^2h^2 = r^2(h_0 + \mu r^2 + o(r^2))^2 = h_0^2 r^2 + o(r^2), 
		\end{align*} 
		and 
		\begin{align*} 
			F - \eta - Nh & = 1 - \frac{h_0P(0)}{2}r^2 + o(r^2) - \eta - N (h_0 + \mu r^2 + o(r^2)) \\ 
			& = 1 - \eta - Nh_0 - \frac{h_0P(0) + 2N\mu}{2}r^2 + o(r^2) \\ 
			& = - \frac{h_0P(0) + 2N\mu}{2}r^2 + o(r^2). 
		\end{align*} 
		Comparing the coefficient of $r^2$, one can see that 
		\begin{align*} 
			2\mu = h_0^2 - \frac{h_0P(0) + 2N\mu}{2}, 
		\end{align*} 
		namely 
		\begin{align*} 
			\mu = \frac{h_0(2h_0 - P(0))}{2(N+2)}. 
		\end{align*} 
		Thus 
		\begin{align} \label{eqextending} 
			h(r) = h_0 + \frac{h_0(2h_0 - P(0))}{2(N+2)} r^2 + o(r^2), \quad h_r(r) = \frac{h_0(2h_0 - P(0))}{N+2} r + o(r). 
		\end{align} 
 
		We claim that $2h_0 \le P(0)$. If it is not in this case, when $r > 0$ is small enough we have $h'(r) > 0$. On the other hand, as $r \to \infty$, 
		\begin{align*} 
			0 < h(r) = -\frac{v_r(r)}{rv(r)} \le \frac{C}{r} \to 0. 
		\end{align*} 
		This yields the existence of $R > 0$ such that 
		\begin{align*} 
			h_r(r) > 0 ~~ (0 < r < R), \quad\quad h_r(R) = 0. 
		\end{align*} 
		We also have $h_{rr}(R) \le 0$. Now we introduce 
		\begin{align*} 
			Q(r) := 2h(r) - P(\xi(r)). 
		\end{align*} 
		Direct computations show that 
		\begin{align*} 
			Q_r = 2h_r - P_\xi \xi_r = 2h_r - rh P_\xi. 
		\end{align*} 
		Since $h > 0$ and $P_\xi < 0$, we get $Q_r > 0$ in $r \in (0,R)$ and so $Q(R) > Q(0) = 2h_0 - P(0) > 0$. Differentiating \eqref{eqofh_r} with respect to $r$, we deduce that 
		\begin{align*} 
			rh_{rr} + h_r = 2r^2hh_r + 2rh^2 + F_\xi \xi_r - Nh_r = 2r^2hh_r + 2rh^2 - rhP - Nh_r, 
		\end{align*} 
		namely 
		\begin{align} \label{eqofhrr} 
			h_{rr} = hQ + \left(2rh - \frac{N+1}{r}\right)h_r. 
		\end{align} 
		This shows $h_{rr}(R) = h(R)Q(R) > 0$, a contradiction! Thus the claim holds true. 
 
		In the following we aim to prove $h_r \le 0$ for $r > 0$. Suppose by contradiction that the set 
		\begin{align*} 
			E = \big\{r > 0: h_r(r) > 0\big\} 
		\end{align*} 
		is not empty. Take a connected component $(\alpha,\beta) \subset E$. It is clear that $h_r(r) > 0$ for $\alpha < r < \beta$. Since $h > 0$ and $h(r) \to 0$ as $r \to \infty$, we have $\beta < \infty$. At the right endpoint $r = \beta$, 
		\begin{align*} 
			h_r(\beta) = 0, \quad\quad h_{rr}(\beta) \le 0. 
		\end{align*} 
		If $\alpha > 0$, at the left endpoint $r = \alpha$ we also have 
		\begin{align*} 
			h_r(\alpha) = 0, \quad\quad h_{rr}(\alpha) \ge 0. 
		\end{align*} 
		Using \eqref{eqofhrr} we get 
		\begin{align*} 
			h_{rr}(\alpha) = h(\alpha)Q(\alpha). 
		\end{align*} 
		Together with $h(\alpha) > 0$ we obtain $Q(\alpha) \ge 0$. Reasoning as before we have 
		\begin{align*} 
			Q(\beta) > Q(\alpha) \ge 0. 
		\end{align*} 
		Using \eqref{eqofhrr} again, 
		\begin{align*} 
			h_{rr}(\beta) = h(\beta)Q(\beta) > 0, 
		\end{align*} 
		contradicting $h_{rr}(\beta) \le 0$. If $\alpha = 0$, we have proved that $Q(0) = 2h_0 - P(0) \le 0$. If $Q(0) < 0$, by \eqref{eqextending} one gets $h_r(r) < 0$ near $r = 0$, implying that $(0,\beta) \not\subset E$. Therefore, $Q(0) = 0$ and it still holds that 
		\begin{align*} 
			Q(\beta) > Q(0) = 0 
		\end{align*} 
		and 
		\begin{align*} 
			h_{rr}(\beta) = h(\beta)Q(\beta) > 0 
		\end{align*} 
		a contradiction! Thus the set $E$ must be empty, namely $h_r \le 0$ in $r > 0$. 
 
		Further, we will eliminate the possibility that $h_r$ has a zero point. Arguing by contradiction, we assume that $h_r(R) = 0$ for some $R > 0$. Together with $h_r \le 0$ we get $h_{rr}(R) = 0$. By \eqref{eqofhrr} and using $h(R) > 0$ we have $Q(R) = 0$. At $r = R$, 
		\begin{align*} 
			Q_r(R) = 2h_r(R) - Rh(R)P_\xi(\xi(R)) = - Rh(R)P_\xi(\xi(R)) > 0. 
		\end{align*} 
		Differentiating \eqref{eqofhrr} with respect to $r$, we deduce 
		\begin{align*} 
			h_{rrr} = h_rQ + hQ_r + A_r(r)h_r + A(r)h_{rr} 
		\end{align*} 
		where 
		\begin{align*} 
			A(r) = 2rh - \frac{N+1}{r}. 
		\end{align*} 
		At $r = R$, 
		\begin{align*} 
			h_{rrr}(R) = h(R)Q_r(R) > 0. 
		\end{align*} 
		Expanding $h_r$ at $r = R$ we have 
		\begin{align*} 
			h_r(R + \epsilon) = h_r(R) + h_{rr}(R)\epsilon + \frac12 h_{rrr}(R)\epsilon^2 + o(\epsilon^2) = \frac12 h_{rrr}(R)\epsilon^2 + o(\epsilon^2). 
		\end{align*} 
		For small $\epsilon > 0$ we derive that $h_r(R + \epsilon) > 0$, contradicting $h_r \le 0$. This contradiction yields that $h_r < 0$ for all $r > 0$. 
 
		Finally, by 
		\begin{align*} 
			y_r = (rk)_r = r k_r + k = rk_r + rh > rJ 
		\end{align*} 
		we get $k_r > J - h$. It remains to prove $k_r < J$. Let 
		\begin{align*} 
			\mathcal{P}_* = r^N\left(v_r^2 + 2\mathcal{F}(v)\right) + (N-1)r^{N-1}vv_r. 
		\end{align*} 
	    Then we have 
		\begin{align*} 
			\mathcal{P}_* = r^Nv^2 \left(k_r - J\right), \quad \mathcal{P}_*'(r) = r^{N-1}v^2 \Sigma 
		\end{align*} 
		where 
		\begin{align*} 
			\Sigma(r) = k^2 - \eta + (1-\theta)\frac{N+1-p(N-1)}{p+1}v^{p-1} + \theta\frac{N+1-q(N-1)}{q+1}v^{q-1}. 
		\end{align*} 
		We first prove $\mathcal{P}_* \le 0$ in $r > 0$. Otherwise, since $\mathcal{P}_*(0) = 0$ and $\mathcal{P}_*(r) \to 0$ as $r \to \infty$, we can assume the existence of $R > 0$ such that 
		\begin{align*} 
			\mathcal{P}_*(R) > 0, \quad\quad \mathcal{P}_*'(R) = 0. 
		\end{align*} 
		Clearly, $\Sigma(R)  = 0$. Since 
		\begin{align*} 
			k_r = k^2 - \frac{v_{rr}}{v} = k^2 - \eta + (1-\theta)v^{p-1} + \theta v^{q-1} - (N-1)h, 
		\end{align*} 
		at $r = R$ we get 
		\begin{align*} 
			k_r(R) = N J(\xi(R)) - (N-1)h(R), \quad Rh_r(R) = k_r(R) - h(R) = N\left(J(\xi(R)) - h(R)\right). 
		\end{align*} 
		By $h_r < 0$ we get $h(R) > J(\xi(R))$. Thus 
		\begin{align*} 
			\mathcal{P}_*(R) = (N-1)R^{N}v(R)^2\left(J(\xi(R)) - h(R)\right) < 0, 
		\end{align*} 
		a contradiction! 
 
		Now we suppose by contradiction that there exists $r_0 > 0$ such that $\mathcal{P}_*(r_0) = 0$. In view of $\mathcal{P}_* \le 0$, we know $\mathcal{P}_*'(r_0) = 0$ immediately. Arguing as before, we get $\mathcal{P}_*(r_0) < 0$ and find a contradiction. This shows $\mathcal{P}_* < 0$, namely $k_r < J$ in $r > 0$. The proof is complete.

<!-- lemma-5: body_md -->
\label{lemtransformation} 
		Suppose $N \ge 3$. For any radial function $f \in C_c^\infty(\mathbb{R}^N)$, let $f(r) = f(x)$ with $r = |x|$ and rewrite $f(r) = \phi(r)g(r)$. Then 
		\begin{align} \label{indentity} 
			Q_L(f) = \omega_{N-1} \int_{\mathbb{R}} \rho(s)\left(g_s^2 - (N-1)g^2\right)ds, 
		\end{align} 
		where $\omega_{N-1} = |\mathbb{S}^{N-1}|$ and 
		\begin{align*} 
			Q_L(f) := \int_{\mathbb{R}^N}\left(|\nabla f|^2 + \eta f^2 - (1-\theta)pv^{p-1}f^2 - \theta qv^{q-1}f^2\right)dx. 
		\end{align*} 
		The identity \eqref{indentity} can be extended to the case of $f \in H^1_{rad}(\mathbb{R}^N)$.
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
Multiplying both sides of \eqref{eqofphi} by $r^{N-1}g^2\phi$ and integrating on $(\epsilon,R)$ yields that 
		\begin{align*} 
			\left[\phi\phi_r g^2r^{N-1}\right]_{\epsilon}^R - \int_\epsilon^R \left(2\phi\phi_r gg_r + \phi_r^2 g^2\right)r^{N-1}dr = \int_\epsilon^R\left(\frac{N-1}{r^2} + \eta - (1-\theta)pv^{p-1} - \theta q v^{q-1}\right)\phi^2 g^2 r^{N-1}dr. 
		\end{align*} 
		Then we have 
		\begin{align} \label{eqeptoR} 
			& \int_\epsilon^R \left(f_r^2 + (\eta - (1-\theta)pv^{p-1} - \theta q v^{q-1})f^2\right)r^{N-1}dr \\ 
			& ~~~ = \int_\epsilon^R \phi^2g_r^2 r^{N-1}dr - (N-1)\int_\epsilon^R \phi^2g^2r^{N-3}dr + \left[\phi\phi_r g^2r^{N-1}\right]_{\epsilon}^R. 
 		\end{align} 
		Recall that $h_0 = -v_{rr}(0) > 0$. Since $v(x) \in C^4(\mathbb{R}^N)$ is radial, we know 
		\begin{align*} 
			\phi(r) = h_0r + O(r^3), \quad \phi_r(r) = h_0 + O(r^2), \quad \rho(s) = h_0^2r^{N}(1+O(r^2)). 
		\end{align*} 
		Moreover, since $f$ is smooth and radial, we have 
		\begin{align*} 
			f(r) = f(0) + O(r^2), \quad f_r(r) = f_{rr}(0)r + O(r^3). 
		\end{align*} 
		Therefore, 
		\begin{align*} 
			g(r) = \frac{f}{\phi} = \frac{f(0)}{h_0r} + O(r), \quad g_r(r) = \frac{\phi f_r - f\phi_r}{\phi^2} = -\frac{f(0)}{h_0r^2} + O(1). 
		\end{align*} 
		Recalling $s = \ln r$, one gets $g_s = rg_r$, yielding that 
		\begin{align*} 
			g_s = r\frac{\phi f_r - f\phi_r}{\phi^2} = -\frac{f(0)}{h_0r} + O(r). 
		\end{align*} 
		Thus, near $r = 0$, both $g$ and $g_s$ belong to $L^2(\rho ds)$ for $N > 2$. If $N = 2$ and $f(0) \neq 0$, both integrals diverge. We also notice that 
		\begin{align*} 
			\phi\phi_r g^2r^{N-1} = f(0)^2r^{N-2} + O(r^N) \to 0 \quad \text{ as } r \to 0. 
		\end{align*} 
		Choose $R > 0$ large enough such that $\text{supp~}f(x) \subset B_R(0)$. Then sending $\epsilon$ to $0$ in \eqref{eqeptoR} and performing variable substitution $s = \ln r$, we derive that 
		\begin{align*} 
			Q_L(f) & = \int_{\mathbb{S}^{N-1}}d\sigma \int_0^\infty\left(f_r^2 + (\eta - (1-\theta)pv^{p-1} - \theta q v^{q-1})f^2\right)r^{N-1}dr \\ 
			& = \omega_{N-1} \int_{\mathbb{R}} \rho(s)\left(g_s^2 - (N-1)g^2\right)ds. 
		\end{align*} 
Using the radial Hardy inequality 
		\begin{align*}

We define the action functional $J_\lambda : H^1(\mathbb{R}^N) \to \mathbb{R}$ as 
		\begin{align*} 
			J_\lambda(u) = \frac{1}{2}\int_{\mathbb{R}^N}\left(|\nabla u|^2 + \lambda u^2\right)dx - \frac{1}{p+1}\int_{\mathbb{R}^N}|u|^{p+1}dx - \frac{1}{q+1}\int_{\mathbb{R}^N}|u|^{q+1}dx. 
		\end{align*} 
		The functional $J_\lambda$ satisfies the Palais-Smale condition in the radial space $H_{rad}^1(\mathbb{R}^N)$. Any critical point of $J_\lambda$ constrained on $H_{rad}^1(\mathbb{R}^N)$ is a weak solution of 
		\begin{align} \label{eqofweaksolution} 
			-\Delta u + \lambda u = |u|^{p-1}u + |u|^{q-1}u \quad \text{ in } \mathbb{R}^N. 
		\end{align} 
		If the critical point is positive, it is indeed a solution of \eqref{doublepower}. Define the radial Nehari manifold 
		\begin{align*} 
			\mathcal{N}_\lambda := \big\{u \in H_{rad}^1(\mathbb{R}^N) \backslash \{0\}: \int_{\mathbb{R}^N}\left(|\nabla u|^2 + \lambda u^2\right)dx = \int_{\mathbb{R}^N}(|u|^{p+1} + |u|^{q+1})dx \big\}, 
		\end{align*} 
		and set 
		\begin{align*} 
			c_\lambda := \inf_{\mathcal{N}_\lambda}J_\lambda. 
		\end{align*} 
		Then we can find a non-negative critical point $u_\lambda \in H_{rad}^1(\mathbb{R}^N)$ of $J_\lambda$ at level $c_\lambda$, namely 
		\begin{align*} 
			J_\lambda(u_\lambda) = c_\lambda. 
		\end{align*} 
		Using the strong maximum principle, one sees that $u_\lambda$ is positive. Since the codimension of $\mathcal{N}_\lambda$ in $H_{rad}^1(\mathbb{R}^N)$ is $1$, we have $m_{rad}(u_\lambda) \le 1$. Together with 
		\begin{align*} 
			\langle L_{u_\lambda}u_\lambda,u_\lambda \rangle =(1-p)\int_{\mathbb{R}^N}|u_\lambda|^{p+1}dx + (1-q)\int_{\mathbb{R}^N}|u_\lambda|^{q+1}dx < 0, 
		\end{align*} 
		we derive that $m_{rad}(u_\lambda) = 1$. Then, using the decomposition in terms of spherical harmonics, we also obtain that $m(u_\lambda) = 1$. 
 
		For any $\lambda_* > 0$, Theorem \ref{thmnondegenerate} yields that $u_{\lambda_*}$ is nondegenerate in $H_{rad}^1(\mathbb{R}^N)$. Applying the implicit function theorem to the map 
		\begin{align*} 
			H^1_{rad}(\mathbb{R}^N) \times (0,\infty) \to \left(H^1_{rad}(\mathbb{R}^N)\right)^*: ~~ (u,\lambda) \mapsto -\Delta u + \lambda u - |u|^{p-1}u - |u|^{q-1}u, 
		\end{align*} 
		we can establish a local branch parameterized by $\lambda$ 
		\begin{align*} 
			(\lambda_*-\epsilon,\lambda_*+\epsilon) \to H_{rad}^1(\mathbb{R}^N), \quad \lambda \mapsto u(\lambda), 
		\end{align*} 
		such that $\epsilon > 0$ is small, $u(\lambda_*) = u_{\lambda_*}$, the map $\lambda \mapsto u(\lambda)$ is $C^1\left((\lambda_*-\epsilon,\lambda_*+\epsilon), H_{rad}^1(\mathbb{R}^N)\right)$, $u(\lambda)$ solves \eqref{eqofweaksolution} and $m(u(\lambda)) = 1$. Define the operator 
		\begin{align*} 
			L_{\lambda,-} := -\Delta + \lambda - |u(\lambda)|^{p-1} - |u(\lambda)|^{q-1}. 
		\end{align*} 
		Note that $L_{\lambda,-}u(\lambda) = 0$ and $\sigma_{ess}(L_{\lambda,-}) = [\lambda,\infty)$. Since $u(\lambda_*)$ is positive, $0$ is the first eigenvalue of $L_{\lambda_*,-}$ and is simple. It is clear that $L_{\lambda,-} \to L_{\lambda_*,-}$ in norm-resolvent sense. Taking $\epsilon > 0$ small enough we can get that $0$ is the first eigenvalue of $L_{\lambda,-}$ and $u(\lambda)$ is positive for $\lambda \in (\lambda_*-\epsilon,\lambda_*+\epsilon)$. Next, we extend the branch to $\lambda \to \infty$. For this, we define 
		\begin{align*} 
			\lambda^* := \sup \big\{\Lambda: \text{ for } \lambda \in [\lambda_*,\Lambda), & ~ u(\lambda) \text{ is a positive solution of } \eqref{eqofweaksolution}, m(u(\lambda)) = 1,\\ 
			& ~~~ u(\lambda_*) = u_{\lambda_*}, \lambda \mapsto u(\lambda) \text{ is } C^1\left([\lambda_*,\Lambda), H_{rad}^1(\mathbb{R}^N)\right) \big\} 
		\end{align*} 
		We claim $\lambda^* = \infty$. Suppose by contradiction that $\lambda^* < \infty$. Next, we prove that $u(\lambda)$ is uniformly bounded for $\lambda \in [\lambda_*,\lambda^*)$. Define 
		\begin{align*} 
			g(\lambda) := \frac{p-1}{p+1}\int_{\mathbb{R}^N}|u(\lambda)|^{p+1}dx + \frac{q-1}{q+1}\int_{\mathbb{R}^N}|u(\lambda)|^{q+1}dx. 
		\end{align*} 
		By the Nehari identity we have $C_1g(\lambda) \le \|u(\lambda)\|_{H^1(\mathbb{R}^N)}^2 \le C_2g(\lambda)$ and 
		\begin{align*} 
			\lambda \int_{\mathbb{R}^N}u(\lambda)^2dx \le \frac{q+1}{q-1} g(\lambda). 
 		\end{align*} 
		Let $v_\lambda = \frac{d}{d \lambda}u(\lambda)$ and we have 
		\begin{align*} 
			-\Delta v_\lambda + \lambda v_\lambda + u(\lambda) = p|u(\lambda)|^{p-1}v_\lambda + q|u(\lambda)|^{q-1}v_\lambda. 
		\end{align*} 
		Then we get 
		\begin{align*} 
			g'(\lambda) = \int_{\mathbb{R}^N}u(\lambda)^2 dx \le \frac{q+1}{q-1}\lambda^{-1} g(\lambda), 
		\end{align*} 
		yielding 
		\begin{align*} 
			g(\lambda) \le \left(\frac{\lambda}{\lambda_*}\right)^{\frac{q+1}{q-1}}g(\lambda_*) \le \left(\frac{\lambda^*}{\lambda_*}\right)^{\frac{q+1}{q-1}}g(\lambda_*). 
		\end{align*}

		\begin{align*} 
			\|u(\lambda)\|_{H^1(\mathbb{R}^N)} \le C. 
		\end{align*} 
		One can check that $c_\lambda > 0$ is continuous and increasing in $\lambda > 0$, see for example \cite{Song}. Take $\{\lambda_n\} \subset [\lambda_*,\lambda^*)$ such that $\lambda_n \to \lambda^*$. The sequence $\{u(\lambda_n)\}$ is a bounded Palais-Smale sequence of $J_{\lambda^*}$. By the Palais-Smale condition, up to a subsequence, there exists $\tilde{u} \in H_{rad}^1(\mathbb{R}^N)$ such that $u(\lambda_n) \to \tilde{u}$ strongly in $H^1(\mathbb{R}^N)$.  		\end{align*} 
		we deduce that $\tilde{u} \neq 0$. By the strong maximum principle, $\tilde{u}$ is positive. Combining 
		\begin{align*} 
			m(\tilde{u}) \le \liminf_{n}m(u(\lambda_n)), \quad \langle L_{\tilde{u}}\tilde{u},\tilde{u} \rangle < 0, 
		\end{align*} 
		and the decomposition in terms of spherical harmonics, we can check that $m(\tilde{u}) = 1$. By Theorem \ref{thmnondegenerate}, $\tilde{u}$ is nondegenerate. Applying the implicit function theorem at $(\tilde{u}, \lambda^*)$ with $u(\lambda^*) = \tilde{u}$, we find a contradiction with the definition of $\lambda^*$. 
 
		Denote $u_\lambda = u(\lambda)$ for $\lambda \in [\lambda_*,\infty)$. We consider 
		\begin{align*} 
			w_\lambda(x) := \lambda^{-\frac{1}{p-1}}u_\lambda\left(\frac{x}{\sqrt{\lambda}}\right), 
		\end{align*} 
		which satisfies the following equation 
		\begin{align} \label{scalingequation} 
			-\Delta w + w = w^p + \lambda^{\frac{q-p}{p-1}}w^q \quad \text{ in } \mathbb{R}^N, \quad w > 0. 
		\end{align} 
		Let $w_0$ be the unique radial positive solution of 
		\begin{align*} 
			-\Delta w + w = w^p, \quad w \in H^1(\mathbb{R}^N). 
		\end{align*} 
		One can see that $w_\lambda \to w_0$ in $H^1(\mathbb{R}^N)$ as $\lambda \to \infty$, see \cite[Theorem 4.10]{JZZ}. 
		 
		If at $\lambda =\lambda_*$, \eqref{doublepower} admits another radial solution $\widehat{u}_{\lambda_*} \neq u_{\lambda_*}$ with $m(\widehat{u}_{\lambda_*}) = 1$, then we obtain another branch 
		\begin{align*} 
			[\lambda_*,\infty) \to H^1_{rad}(\mathbb{R}^N), \quad \lambda \mapsto \widehat{u}(\lambda) 
		\end{align*} 
		such that $\widehat{u}(\lambda_*)= \widehat{u}_{\lambda_*}$, the map $\lambda \mapsto \widehat{u}(\lambda)$ is $C^1$, $\widehat{u}(\lambda)$ is a positive solution of \eqref{eqofweaksolution} and $m(\widehat{u}(\lambda)) = 1$. Denote $\widehat{u}_\lambda = \widehat{u}(\lambda)$ and we consider

			\widehat{w}_\lambda(x) := \lambda^{-\frac{1}{p-1}}\widehat{u}_\lambda\left(\frac{x}{\sqrt{\lambda}}\right), 
		\end{align*} 
		which is also a solution of \eqref{scalingequation}. As $\lambda \to \infty$, we have $\widehat{w}_\lambda \to w_0$ in $H^1(\mathbb{R}^N)$. Since $w_\lambda \neq \widehat{w}_\lambda$, using the nondegeneracy of $w_0$ and applying the implicit function theorem, we find a contradiction and complete the proof.

		\end{align*} 
		Hence, 
		\begin{align*} 
			\int_{\mathbb{R}^N}u_\lambda^2dx \sim \lambda^{\frac{4-N(p-1)}{2(p-1)}}\int_{\mathbb{R}^N}w_0^2dx \quad \text{ as } \lambda \to \infty. 
		\end{align*} 
		If $p <  1 + 4/N$, 
		\begin{align*} 
			\int_{\mathbb{R}^N}u_\lambda^2dx \to \infty \text{ as } \lambda \to \infty. 
		\end{align*} 
		If $p = 1 + 4/N$, 
		\begin{align*} 
			\lim_{\lambda \to \infty}\int_{\mathbb{R}^N}u_\lambda^2dx = c_0, 
		\end{align*} 
		where $c_0$ is given in Section \ref{secintroduction}.  Similarly (see 
		Theorem 2.5 and Theorem 4.6 in \cite{JZZ}), 
		\begin{align*} 
			\lambda^{-\frac{1}{q-1}}u_\lambda\left(\frac{x}{\sqrt{\lambda}}\right) \to W_q \quad \text{ in } H^1(\mathbb{R}^N) \quad \text{ as } \lambda \to 0, 
		\end{align*} 
		where $W_q$ is the unique radial positive solution of 
		\begin{align*} 
			-\Delta W + W = W^q, \quad W \in H^1(\mathbb{R}^N). 
		\end{align*} 
		Hence, 
		\begin{align*} 
			\int_{\mathbb{R}^N}u_\lambda^2dx \sim \lambda^{\frac{4-N(q-1)}{2(q-1)}} \int_{\mathbb{R}^N}W_q^2dx \to 0 \quad \text{ as } \lambda \to 0. 
		\end{align*} 
		Recalling \eqref{ulavla}, we further have 
		\begin{align*} 
			\frac{d}{d\lambda}\int_{\mathbb{R}^N}u_\lambda^2dx > 0 
		\end{align*} 
		and conclude the results presented in Theorem \ref{thmmain2}.
