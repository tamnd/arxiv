\label{defn:left_right_duals_of_endofunctors}
    Let $F \in \hom(\mathcal{C},\mathcal{C})$ be an $A$-linear endofunctor of a dualizable $A$-linear category $\mathcal{C}$.
    \begin{enumerate}
        \item Say that $F$ is \emph{left dualizable} if $\Xi_{\mathcal{C}}^{-1}(F)$ admits an $A$-linear right adjoint.
        Then define the \emph{left dual of $F$} by
        \[
            F^! := \Theta_{\mathcal{C}}(\tau\circ\Xi_{\mathcal{C}}^{-1}(F)^R) \in \hom(\mathcal{C},\mathcal{C}).
        \]
        \item Say that $F$ is \emph{right dualizable} if $\Theta_{\mathcal{C}}^{-1}(F)$ admits an $A$-linear right adjoint.
        Then define the \emph{right dual of $F$} by
        \[
            F^{\ast} := \Xi_{\mathcal{C}}(\tau\circ\Theta_{\mathcal{C}}^{-1}(F)^R) \in \hom(\mathcal{C},\mathcal{C}).
        \]
    \end{enumerate}

\label{prop:dualizable_implies_compact}
    Let $\mathcal{V} \in \mathrm{CAlg}(\mathrm{Pr}^L_{\omega})$ be a compactly-generated presentable symmetric monoidal category.
    Then the following statements hold:
    \begin{enumerate}
        \item The category $\mathrm{Pr}^L_{\omega,\mathcal{V}}$ is compactly generated, i.e. $\aleph_0$-presentable, and symmetric monoidal.
        \item The symmetric monoidal structure on $\mathrm{Pr}^L_{\omega,\mathcal{V}}$ preserves small colimits separately in each variable.
        \item The symmetric monoidal unit of $\mathrm{Pr}^L_{\omega,\mathcal{V}}$ is compact.
        \item Dualizable objects of $\mathrm{Pr}^L_{\omega,\mathcal{V}}$ are compact.
    \end{enumerate}
    \begin{proof}
        For (1) and (2), note that $\mathrm{Pr}^L_{\omega}$ is $\aleph_0$-presentable \cite{scholze_gestalten}*{Theorem 2.9} and symmetric monoidal \cite{scholze_gestalten}*{Definition/Proposition 3.6}.
        Therefore by \cite{lurie_HA}*{\S 4.2, \S 4.5}, we know that $\mathrm{Pr}^L_{\omega,\mathcal{V}}$ is $\aleph_0$-presentable and symmetric monoidal.
        Furthermore, the symmetric monoidal structure on $\mathrm{Pr}^L_{\omega}$ preserves small colimits separately in each variable, so the same is true for the symmetric monoidal structure on $\mathrm{Pr}^{L}_{\omega,\mathcal{V}}$ \cite{lurie_HA}*{4.8.5.21}.

        Now we prove item (3).
        Observe that \cite{ramzi_rigidity}*{Example 4.38} implies that the free functor $\mathrm{Pr}^L_{\omega} \to \mathrm{Pr}^L_{\omega,\mathcal{V}}$ endows $\mathrm{Pr}^L_{\omega,\mathcal{V}}$ with the structure of a rigid $\mathrm{Pr}^L_{\omega}$-algebra.
        In particular, the unit $\mathbf{1}_{\mathrm{Pr}^L_{\omega,\mathcal{V}}} = \mathcal{V}$ is $\mathrm{Pr}^L_{\omega}$-atomic, so it suffices to show that the symmetric monoidal unit of $\mathrm{Pr}^L_{\omega}$ is compact.
        This is done in Lemma \ref{lem:ani_compact}.

        Finally for (4), suppose that $\mathcal{C} \in \mathrm{Pr}^L_{\omega,\mathcal{V}}$ is dualizable.
        It is a general fact that for any symmetric monoidal $\infty$-category with colimit-preserving tensor product and compact unit, the dualizable objects are compact.
        (Indeed, the functor of taking maps out of a dualizable object is the composite of tensoring with the dual and then taking maps out of the unit).
        So we conclude (4) by applying items (2) and (3).
    \end{proof}

\label{lem:ani_compact}
    The symmetric monoidal unit $\mathrm{Ani}$ of $\mathrm{Pr}^L_{\omega}$ is compact.
    \begin{proof}
        Let $\mathcal{D}$ be a compactly-generated presentable category.
        Recall that morphisms in $\mathrm{Pr}^L_{\omega}$ are functors that preserve small colimits and compact objects. 

        We claim that evaluation at the point determines an equivalence of $\infty$-categories
        \[
            \mathrm{Map}_{\mathrm{Pr}^L_{\omega}}(\mathrm{Ani},\mathcal{D}) \simeq \mathcal{D}^c,

\label{prop:perfect_and_proper_support_implies_pseudo-perfect}
    Assume that $p: X \to S$ is a smooth morphism of classical varieties over $\mathbb{C}$.
    Let $T$ be an eventually coconnective scheme over $S$, and let $\mathcal{F} \in \mathrm{Perf}(X\times_S T)$ be a perfect complex that has proper support over $T$.

        We prove the statement for $T = S$, and the general case follows from this one by base change.

        Let $\mathcal{F} \in \mathrm{Perf}(X)$ be a perfect complex, and suppose that $Z$ is a derived scheme with a closed immersion $i: Z \to X$ such that $\mathcal{F}|_{X\setminus Z} \simeq 0$ and such that the composite $Z \to S$ is proper.

        The sheaf $\hom_X(\mathcal{G},\mathcal{F})$ is perfect and moreover properly supported on $Z$ \cite{lurie_SAG}*{Corollary 7.1.5.6}.
        Therefore, it suffices to show that if $\mathcal{E} \in \mathrm{Perf}(X)$ is properly supported, then $p_{\ast}\mathcal{E} \in \mathrm{Perf}(S)$.

        The closed immersion $i: Z \to X$ induces a morphism $\hat{\imath}: X^{\wedge}_Z \to X$ from the formal completion of $X$ along $Z$ to $X$.

        This is the content of the following Lemma \ref{lem:end_of_compactness_proposition}.
    \end{proof}

\label{lem:end_of_compactness_proposition}
    Suppose that $p: X \to S$ is a smooth morphism of classical varieties over $\mathbb{C}$.
    Let $i: Z \to X$ be a closed immersion from a derived scheme of finite type such that the composite $Z \to S$ is proper.
    Write $X^{\wedge}_Z$ for the formal completion of $X$ along $Z$, and let $\mathcal{E}$ be a compact object in $\mathrm{QCoh}(X^{\wedge}_Z)$.
    Then the object $p_{\ast}\hat{\imath}_?\mathcal{E}$ is compact in $\mathrm{QCoh}(S)$.
    \begin{proof}
       Consider the space \begin{equation}\label{eq:colimit_inside_map}
            \mathop{\mathrm{Map}}\nolimits_{\mathrm{QCoh}(S)}(p_{\ast}\hat{\imath}_?\mathcal{E},\mathrm{colim}_i \mathcal{F}_i).
        \end{equation}
        The functor $\Xi_S: \mathrm{QCoh}(S) \to \mathrm{IndCoh}(S)$ is fully faithful \cite{gaitsgory_indcoh}*{Proposition 1.5.3}, so we have an equivalence
        \[
            \mathop{\mathrm{Map}}\nolimits_{\mathrm{QCoh}(S)}(p_{\ast}\hat{\imath}_?\mathcal{E},\mathrm{colim}_i \mathcal{F}_i) \simeq \mathop{\mathrm{Map}}\nolimits_{\mathrm{IndCoh}(S)}(\Xi_Sp_{\ast}\hat{\imath}_?\mathcal{E},\Xi_S\mathrm{colim}_i\mathcal{F}_i).
        \]
        We may ``move the functor $\Xi$ past the two pushforward morphisms'' via the following two equivalences
        \begin{equation*}
            \Xi_S \circ p_{\ast} \simeq p_{\ast}^{\mathrm{IndCoh}}\circ \Xi_X, \quad\quad
            \Xi_X \circ \hat{\imath}_? \simeq \hat{\imath}_{\ast}^{\mathrm{IndCoh}}\circ \Xi_{X^{\wedge}_Z},
        \end{equation*}
        (see \cite{gaitsgory_indcoh}*{Proposition 3.6.7} and \cite{GR_dgindschemes}*{\S 7.6.2} respectively).
        Thus the space (\ref{eq:colimit_inside_map}) is equivalent to the space
        \[
            \mathop{\mathrm{Map}}\nolimits_{\mathrm{IndCoh}(S)}((p\circ\hat{\imath})_{\ast}^{\mathrm{IndCoh}}\Xi_{X^{\wedge}_Z}\mathcal{E},\Xi_S\mathrm{colim}_i\mathcal{F}_i).
        \]
        The map $p\circ\hat{\imath}$ is ind-proper, so we have an adjunction $(p\circ\hat{\imath})_{\ast}^{\mathrm{IndCoh}} \dashv (p\circ\hat{\imath})^!$.
        Furthermore, we may ``move the functor $\Xi$ past the two pullback morphisms'' via the following two equivalences
        \begin{equation*}
            p^!\circ \Xi_S \simeq \Xi_X \circ p^{\mathrm{QCoh},!},\quad\quad 
            \hat{\imath}^!\circ \Xi_X \simeq \Xi_{X^{\wedge}_Z}\circ \hat{\imath}^{\ast},
        \end{equation*}
        (see \cite{gaitsgory_indcoh}*{Proposition 7.4.7} and \cite{GR_dgindschemes}*{Proposition 7.6.4} respectively).
        By the Adjoint Functor Theorem, the functors $\hat{\imath}^{\ast}$, $p^{\mathrm{QCoh},!}$ commute with colimits, since they have right adjoints $\hat{\imath}_{\ast}$ and $p_{\ast}\hom(\omega_{X/S},\mhyphen)$ respectively.

        There is an adjunction $\Upsilon_{X^{\wedge}_Z}\dashv \Xi^{\vee}_{X^{\wedge}_Z}$ with colimit-preserving right adjoint \cite{GR_dgindschemes}*{\S 7.6.2}, which dualizes to an adjunction $\Xi_{X^{\wedge}_Z}\dashv \Upsilon^{\vee}_{X^{\wedge}_Z}$ with colimit-preserving right adjoint.
        Therefore, the functor $\Xi_{X^{\wedge}_Z}$ preserves compact objects, so we may conclude that the space (\ref{eq:colimit_inside_map}) is equivalent to
        \begin{align*}
            \mathrm{colim}_i \mathop{\mathrm{Map}}\nolimits_{\mathrm{IndCoh}(X^{\wedge}_Z)}(\Xi_{X^{\wedge}_Z}\mathcal{E},\hat{\imath}^!p^!\Xi_S \mathcal{F}_i) &\simeq \mathop{\mathrm{Map}}\nolimits_{\mathrm{IndCoh}(S)}(\Xi_S p_{\ast}\hat{\imath}_?\mathcal{E},\Xi_S\mathcal{F}_i) \\
            &\simeq \mathop{\mathrm{Map}}\nolimits_{\mathrm{QCoh}(S)}(p_{\ast}\hat{\imath}_?\mathcal{E},\mathcal{F}_i).
        \end{align*}
        Thus we have shown that $p_{\ast}\hat{\imath}_?\mathcal{E}$ is compact.
    \end{proof}
