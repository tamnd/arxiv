\label{Lemma:FTPoissonsum}
	Let $L, P \in 2\mathbb N$. For $f, g \in {\mathcal W}_+$ with $\hat{f}_{\cos}, \, \hat{g}_{\sin} \in {\mathcal W}_+$ and $g(0)=0$ we have for $x \in \big[0,\,\tfrac{L}{2}\big]$
	\begin{align} \label{eq:Poisson1}
		 \textstyle f(x) + \sum\limits_{k = 1}^{\infty}\big[ f(x + kL) + f(kL-x)\big]
             &=  \textstyle \frac{2}{L}\,{\hat f}_{\cos}(0) + \frac{4}{L}\,\sum\limits_{n = 1}^{\infty}{\hat f}_{\cos}\big(\tfrac{n}{L
          }\big)\, \cos\frac{2\pi nx}{L}\, ,
		\\
		\label{eq:Poisson3}
		 \textstyle g(x) + \sum\limits_{k = 1}^{\infty}\big[ g(x + kL) - g(kL-x)\big]
             &=  \textstyle  \frac{4}{L}\,\sum\limits_{n = 1}^{\infty} \hat{g}_{\sin}\big(\tfrac{n}{L}\big)\, \sin\frac{2\pi nx}{L}\,,
             \end{align}
             and for $v \in \big[0,\,\tfrac{P}{2}\big]$
\begin{align}
		\label{eq:Poisson2}
		 \textstyle  {\hat f}_{\cos}(v) + \sum\limits_{k = 1}^{\infty} \big[{\hat f}_{\cos}(v + kP)+ {\hat f}_{\cos}(kP-v)\big]
                         &=  \textstyle  \frac{1}{2P}\,f(0) + \frac{1}{P}\sum\limits_{n = 1}^{\infty}f\big(\tfrac{n}{P}\big)\, \cos \frac{2\pi nv}{P}\,,\\
                         \label{eq:Poisson4}
                 \textstyle   \hat{g}_{\sin}(v) + \sum\limits_{k = 1}^{\infty} \big[ \hat{g}_{\sin}(v + kP)-  \hat{g}_{\sin}(kP-v)\big]
                         &=  \textstyle  \frac{1}{P}\sum\limits_{n = 1}^{\infty}g\big(\tfrac{n}{P}\big)\, \sin \frac{2\pi nv}{P}\,,
	\end{align}	
		and all series converge absolutely and uniformly.

Let $h$ be defined on $\mathbb R$ by $h(x) := \frac12 (f(x)+g(x))$ and $h(-x) := \frac12 (f(x)-g(x))$
for $x \in \mathbb R_{+}$, so that $f$ is the even and $g$ the odd part of $h$ on ${\mathbb R}_+$. We observe that
$$
\textstyle  \hat{h}(v) = \int_{\mathbb R} h(x) \, \mathrm e^{-2\pi \mathrm i xv} \,\mathrm{d} x = {\hat f}_{\cos}(v) - \mathrm i\, \hat{g}_{\sin}(v), \quad
\hat{h}(-v) =  {\hat f}_{\cos}(v) + \mathrm i\, \hat{g}_{\sin}(v), \quad v \in {\mathbb R}_{+}.
$$
 In particular, $h$ satisfies $ \sum\limits_{j \in \mathbb Z} \max\limits_{x \in [0,\,1]} |h(x + j)| < \infty$
as well as $ \sum\limits_{j \in \mathbb Z} \max\limits_{v \in [0,\,1]} |\hat{h}(v + j)| < \infty$,
and the formulas of

<!-- lemma-2: body_md -->
\label{Lemma:polydecay}
	Let $L \in 2\mathbb N$ be given. Assume that $f \in C({\mathbb R}_{+})$ has \emph{polynomial
	decay}
	$$
	\textstyle \sup\limits_{x \in {\mathbb R}_{+}} |f(x)|\,(1 + x)^a \leq c < \infty\,,
	\quad a > 1\,.
	$$
	Then $f \in {\mathcal W}_{+}$ and
	\begin{equation}
		\label{eq:deltapoly}
	\textstyle	\delta_{+}(f,L) \leq c\,(2^a - 1)\zeta(a)\,L^{-a}\,,
	\end{equation}
	where $\zeta$ denotes the \emph{Riemann zeta function}
	$$
	\textstyle \zeta(a) := \sum\limits_{n=0}^{\infty} (n+1)^{-a} \leq 1 + \frac{1}{a-1}\,,
	\quad a > 1\,.
	$$
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The assumption on  the  decay of $f$ implies  for all $k \in \mathbb N$ and
$|x| < \frac{L}{2}$
$$
\textstyle \big|f(x + kL)\big| \leq c\,\big(1 + x + kL\big)^{-a}
\leq c\,\big(1 + (k - \frac{1}{2})L\big)^{-a}\,,
$$
and hence
\begin{eqnarray*}
	\delta_{+}(f,L) &\leq&  \textstyle c\, \sum\limits_{k=1}^{\infty}
	\big(1 +  (k - \frac{1}{2}) L\big)^{-a} =
	c\,L^{-a}\,\sum\limits_{k=1}^{\infty} \big(k - \frac{1}{2} + \tfrac{1}{L}\big)^{-a} \\
	&<&  \textstyle c\,L^{-a}\sum\limits_{n=0}^{\infty} \big(n + \tfrac{1}{2}\big)^{-a}
	= c\,\zeta\big(a, \tfrac{1}{2}\big)\,L^{-a}
\end{eqnarray*}
with the \emph{Hurwitz zeta function} $\zeta\big(a, \frac{1}{2}\big) :=
\sum\limits_{n=0}^{\infty}\bigl(n + \frac{1}{2}\bigr)^{-a} = (2^a - 1)\,\zeta(a)$, see e.g.
\cite[Formula 25.11.11]{DLMF}.
In particular,
$$
\textstyle \| f\|_{{\mathcal W}_{+}} = \sum\limits_{k=0}^{\infty} \max\limits_{x\in [0,\,1]} |f(x+k)| \leq c\,\sum\limits_{k=0}^{\infty} (k+1)^{-a} = c\,\zeta(a) < \infty\,.
$$
finishes the proof.

<!-- lemma-3: body_md -->
\label{Lemma:expodecay}
	Let $L \in 2\mathbb N$ be given. Let $f \in C({\mathbb R}_{+})$ have \emph{exponential decay}
	$$
	\textstyle \sup\limits_{x \in {\mathbb R}_{+}} |f(x)|\,\mathrm e^{r\,x^{\alpha}}
	\leq c < \infty\,, \quad r,\, \alpha > 0\,.
	$$
	Then $f \in {\mathcal W}_{+}$ and, with the abbreviation $u := r\,\big(\frac{L}{2}\big)^{\alpha}$ and the
	upper incomplete gamma function $\Gamma\big(\tfrac{1}{\alpha},u\big) := \int_u^{\infty} t^{\frac{1}{\alpha}-1}\,\mathrm e^{-t}\,\,\mathrm{d} t$  satisfies
	\begin{equation}
		\label{eq:deltaexpo}
		\textstyle \delta_{+}(f,L) \leq  c\,\mathrm e^{-u}   + \frac{c}{\alpha\,L\,r^{1/\alpha}}\,\Gamma\big(\tfrac{1}{\alpha},\,u\big).
	\end{equation}
In particular, if $\alpha$, $r$ and $L$ satisfy $2\alpha\big(u+1-\frac1\alpha\big)\ge 1$, we have
\begin{align}\label{eq:deltaexpo1}
\textstyle  \delta_{+}(f,L) \leq  2c\,\mathrm e^{-r\, \big(\frac{L}{2}\big)^{\alpha}}.
\end{align}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The assumed  exponential decay of $f$ implies for all $k \in \mathbb N$ and
$|x| \leq \frac{L}{2}$ that
$$
\textstyle \bigl|f(x + k L)\bigr| \leq c\,\mathrm e^{-r\,((k- \frac{1}{2})L)^{\alpha}}\,.
$$
Since $t \mapsto \mathrm e^{-r((t-\frac12)L)^{\alpha}}$ is decreasing on $[1,\infty)$, the
integral test yields
\begin{align*}
	\textstyle \delta_{+}(f,L) &\leq \textstyle
	c\,\sum\limits_{k=1}^{\infty}\mathrm e^{-r\,((k- \frac{1}{2})L)^{\alpha}}
	\;\le\; \textstyle c\, \mathrm e^{-r \big(\frac{L}{2}\big)^\alpha} + c\int\limits_{1}^{\infty}\mathrm e^{-r\,((t-\frac12)L)^{\alpha}}\,\,\mathrm{d} t\\
	& = \textstyle c\,\mathrm e^{-u} + \frac{c}{L}\int\limits_{L/2}^{\infty}\mathrm e^{-r\,y^{\alpha}}\,\,\mathrm{d} y
	= c\,\mathrm e^{-u} + \frac{c}{\alpha\,L\,r^{1/\alpha}}\,\Gamma\big(\tfrac1\alpha,\,u\big)\,.
\end{align*}
This proves
\eqref{eq:deltaexpo}. If $u > \frac1\alpha - 1$, then \cite[Proposition 2.7]{Pi20} gives
$\Gamma\big(\frac1\alpha, u\big) \le \frac{u^{1/\alpha}}{u+1-\frac1\alpha}\,\mathrm e^{-u}$.
Inserting $u^{1/\alpha} = r^{1/\alpha}\frac{L}{2}$

yields \eqref{eq:deltaexpo1}.
In particular,
$$
\textstyle \| f\|_{{\mathcal W}_{+}} \leq c \,\sum\limits_{k=0}^{\infty} \mathrm e^{-r k^{\alpha}} \leq c + c\,\int\limits_0^{\infty} \mathrm e^{- r t^{\alpha}}\,\,\mathrm{d} t
	= c + \frac{c}{\alpha}\,r^{-1/\alpha}\,\Gamma\big(\tfrac{1}{\alpha}\big) < \infty\,,
$$
implies  that  $f \in {\mathcal W}_{+}$.
Finally, the condition $2\alpha\big(u+1-\frac1\alpha\big) \ge 1$ is satisfied for
sufficiently large $L$ and arbitrary $r$, $\alpha > 0$.

\label{Thm:samplingcospolynomial}
Let $L$, $P \in 2 \mathbb N$ be fixed and let $f \in {\mathcal W}_{+}$ with ${\hat f}_{\cos} \in {\mathcal W}_{+}$ be given.
Then ${\hat f}_{\cos}$ can be uniformly approximated on $[0,\,P/2]$ by the $P$-periodic
sampling cosine polynomial $c_{P,L}$ in  \eqref{eq:approx}
and  the error
estimate
\begin{equation}
\label{eq:cosmain}
 \textstyle E^{\cos}_{P,L}(f) =  \textstyle \frac{1}{\sqrt{L}} \max\limits_{v \in [0,\frac{P}{2}]} |\hat{f}_{\cos} (v)\! - \!c_{P,L}(v)|\leq \frac{2}{\sqrt L}\,\delta_{+}\big({\hat f}_{\cos}, P\big) + \sqrt L \,\delta_{+}(f,L)
\end{equation}
holds with the (one-sided) decay rates defined in \eqref{eq:decayrate+} for $f$ and ${\hat f}_{\cos}$.

Furthermore, for $f \in {\mathcal W}_{+}$ with ${\hat f}_{\sin} \in {\mathcal W}_{+}$ and $f(0)=0$  we similarly  have
\begin{equation}
\label{eq:sinmain}
 \textstyle E^{\sin}_{P,L}(f)= \textstyle \frac{1}{\sqrt{L}} \max\limits_{v \in [0,\frac{P}{2}]} |\hat{f}_{\sin} (v)\! - \!s_{P,L}(v)|
  \leq \frac{2}{\sqrt L}\,\delta_{+}\big({\hat f}_{\sin}, P\big) + \sqrt L \,\delta_{+}(f,L).
\end{equation}
The additional assumption $f(0)=0$ in the sine case is needed for the continuity of the odd
extension of $f$ on $\mathbb R$.

<!-- proof-4: body_md -->
Splitting the two series in the Poisson summation formula
\eqref{eq:Poisson2} into the principal terms and the remainders, we obtain
${\hat f}_{\cos}(v) + \sigma_1(v)= c_{P,L}(v) + \sigma_2(v)$ for all $v \in [0, \frac{P}{2}]$ with
$$
\textstyle \sigma_1(v) := \sum\limits_{k =1}^{\infty} \big[{\hat f}_{\cos}(v + k P) + {\hat f}_{\cos}(k P- v)\big]\,,\qquad
\sigma_2(v) := \frac{1}{P}\sum\limits_{n = 1 + LP/2}^{\infty} f\big(\tfrac{n}{P}\big)\,\cos \frac{2\pi nv}{P}\,,
$$
and hence $\sqrt L\,E^{\cos}_{P,L}(f) \leq \max\limits_{v \in [0,P/2]} |\sigma_1(v)| +
\max\limits_{v \in [0,P/2]} |\sigma_2(v)|$. The definition of $\delta_{+}\big({\hat f}_{\cos},
P\big)$ implies

\textstyle \max\limits_{v \in [0,\, P/2]} |\sigma_1(v)| \leq \max\limits_{v \in [0,\, P/2]} \sum\limits_{k=1}^{\infty}
\big|{\hat f}_{\cos}(v + k P)\big| + \max\limits_{v \in [0,\, P/2]} \sum\limits_{k=1}^{\infty} \big|{\hat f}_{\cos}(k P - v)\big|
\leq 2\,\delta_{+}\big({\hat f}_{\cos}, P\big)\,.
$$
Substituting $n = m + k LP$ with $m = 1-\tfrac{LP}{2},\,\ldots,\,\tfrac{LP}{2}$
and $k \in \mathbb N$ in $\sigma_2$, we obtain
$$
\textstyle \sigma_2(v) = \frac{1}{P}\,\sum\limits_{m=1-LP/2}^{LP/2}\,\sum\limits_{k=1}^{\infty} f\big(\tfrac{m}{P} + k L\big)\, \cos\frac{2\pi\,(m+k LP)\,v}{P}
$$
and therefore

$$
Thus (\ref{eq:cosmain}) follows. The estimate (\ref{eq:sinmain}) can be shown similarly.

Let $L$, $P \in 2 \mathbb N$ be fixed and let $f \in {\mathcal W}_{+}$ with ${\hat f}_{\cos} \in {\mathcal W}_{+}$ be given.\\

 \textstyle    \frac{1}{\sqrt{L}} \big|{\hat f}_{\cos}(\frac{P}{2}) \big| - \sqrt{L} \,\delta_{+}(f,L)

\begin{align*} \textstyle \sqrt{L} E^{\cos}_{P,L}(f) &= \textstyle \max\limits_{v \in [0,\, P/2]} |{\hat f}_{\cos}(v) - c_{P,L}(v)| = \max\limits_{v \in [0,\, P/2]} |\sigma_2(v) - \sigma_1(v)| \\
&\ge  \textstyle \max\{ \big|\sigma_2\big(\frac{P}{2}\big)\big| - \big|\sigma_1\big(\frac{P}{2}\big)\big|, \,  \big|\sigma_1\big(\frac{P}{2}\big)\big|-\big|\sigma_2\big(\frac{P}{2}\big)\big| \}. \end{align*}

\label{neu}
For $f$ of the form $(\ref{rational})$ its Fourier cosine transform is given by
$$  \textstyle {\hat f}_{\cos}(v) = \sum\limits_{j=1}^J \sum\limits_{k=1}^{k_j} c_{k,j}
\Big(\frac{\pi v}{\mathrm i z_j} \Big)^{k - \frac{1}{2}} \frac{\sqrt{\pi}}{(k-1)!}   \,
K_{k- \frac{1}{2}}(2\pi \mathrm i v z_j)  ,
\qquad v \in {\mathbb R}_{+},
$$
where $K_n$ with $n >0$ denotes the modified Bessel function of second kind.
In particular,
$$ \textstyle |{\hat f}_{\cos}(v)|  \le d_0\, (1+v)^{k_{\max}-1} \, \mathrm e^{- 2\pi s_0 v}, $$
for $v \in {\mathbb R}_{+}$, where the constant $d_0>0$ depends on $c_{k,j}$, $z_j$, and $k_j$.

<!-- proof-6: body_md -->
We employ the following integral formula  (see \cite[Formula 3.773(8)]{GR82})
\begin{align}\label{eq:ratcosK} \textstyle \int\limits_{0}^{\infty} \frac{\cos(ax)}{(x^2+\beta^2)^{n+1/2}} \,\mathrm{d} x  = \frac{\sqrt{\pi}}{2^n\beta^n
\Gamma(n+\frac{1}{2})}  a^n \, K_n(a \beta) \quad \text{for} \; a>0, \; \Re(\beta)>0, \, n > -\frac{1}{2},
\end{align}
with the modified Bessel function of second kind $K_n$. Here we set $a=2\pi v$, $\beta= \mathrm i z_j$ such that $\beta^2 = -z_j^2$,
and $n+ \frac{1}{2} = k$. Then
\begin{align*}
\textstyle \int\limits_0^{\infty} (x^2-z_j^2)^{-k} \, \cos(2\pi xv) \,\mathrm{d} x &=
\textstyle \frac{\sqrt{\pi}}{2^{k - \frac{1}{2}} (\mathrm i z_j)^{k - \frac{1}{2}} \Gamma(k)}  (2\pi v)^{k - \frac{1}{2}} \,
K_{k- \frac{1}{2}}(2\pi \mathrm i v z_j)\\
&= \textstyle  \Big(\frac{\pi v}{\mathrm i z_j} \Big)^{k - \frac{1}{2}} \frac{\sqrt{\pi}}{(k-1)!}   \,
K_{k- \frac{1}{2}}(2\pi \mathrm i v z_j).
\end{align*}
This formula is a generalization of \cite[p.~8, Formula 2.7]{Ob90}.
For shortness, we set
$$
\textstyle \varphi_{k,j}(v) := \Big(\frac{\pi v}{\mathrm i z_j}\Big)^{k-\frac12}\,\frac{\sqrt\pi}{(k-1)!}\,
K_{k-\frac12}(2\pi\mathrm i v z_j)\,\mathrm e^{2\pi s_0 v}\,(1 + v)^{1-k}\,,\quad v > 0\,.
$$
Using the asymptotic expansions (see \cite[9.6.9 and 9.7.2]{AS72})
$$ \textstyle
K_{k-\frac12}(w) = \left\{\begin{array}{ll}
\frac{1}{2}\,\Gamma\big(k-\frac12\big)\big(\frac{2}{w}\big)^{k-\frac12}\big(1 + {\mathcal O}(|w|^2)\big)\,, & w \to 0\,,\\[1mm]
\sqrt{\frac{\pi}{2w}}\, \mathrm e^{-w}\big(1 + {\mathcal O}(|w|^{-1})\big)\,, & w \to \infty\,,
\end{array}\right.
$$
with $w = 2\pi\mathrm i v z_j$, we see that finite limits of $\varphi_{k,j}(v)$ for $v \to +0$ and
$v \to \infty$ exist. Hence each function $\varphi_{k,j}$ is continuous and bounded on ${\mathbb R}_{+}$.
Estimating
\begin{eqnarray*}
\textstyle
|{\hat f}_{\cos}(v)|\,\mathrm e^{2\pi s_0 v}\,(1+v)^{1-k_{\max}} &\leq& \sum\limits_{j=1}^J \sum\limits_{k=1}^{k_j} |c_{k,j}| \,|\varphi_{k,j}(v)|\, (1 + v)^{k- k_{\max}}\\
&\leq& d_0 := \sup\limits_{v\in {\mathbb R}_{+}}\, \sum\limits_{j=1}^J \sum\limits_{k=1}^{k_j} |c_{k,j}| \,|\varphi_{k,j}(v)|\,,
\end{eqnarray*}
we obtain the assertion of Theorem \ref{neu}, since $(1+v)^{k - k_{\max}} \leq 1$ for all $v \in {\mathbb R}_{+}$.

<!-- proof-7: body_md -->
The proof of (\ref{eq:raterror}) follows directly by application of Theorem \ref{Thm:samplingcospolynomial} together with Table \ref{table1}. To show (\ref{eq:errorlow}) we apply the inequality (\ref{eq:cosbelow2}) in Corollary \ref{corlower}.  By assumption $f$ in $(\ref{rational})$  is real-valued and monotonically decreasing, and with $A := \frac{L}{2}+\frac{1}{P}$ we find
\begin{align*} \textstyle \int\limits_{A}^{\infty} f(x) \, \,\mathrm{d} x
&=   \textstyle \sum\limits_{j=1}^J c_{j}  \int\limits_{A}^{\infty} (x^2-z_j^2)^{-1} \, \,\mathrm{d} x
=  \sum\limits_{j=1}^J \frac{c_{j}}{|z_j|} \, \mathrm{arccot}\big(\tfrac{A}{|z_j|} \big)
\ge  \sum\limits_{j=1}^J \frac{c_{j}}{|z_j|} \, \frac{1}{ 1+\frac{A}{|z_j|}} \\
&= \textstyle \sum\limits_{j=1}^J \frac{c_j}{A + |z_j|}
= \frac{2}{L}\sum\limits_{j=1}^J \frac{c_j}{1 + \frac{2}{L}\big(\frac1P+|z_j|\big)}
\;\ge\; \frac{1}{L} \sum\limits_{j=1}^J c_j\,,
\end{align*}
where we have used that $\mathrm{arccot}(x) \ge \frac{1}{x+1}$ for $x \ge 0$ and

<!-- proof-8: body_md -->
The function $f(x) := \operatorname{sech}(\pi x) = \frac{2}{e^{\pi x} + e^{-\pi x}}$ is meromorphic with the
	simple poles $\mathrm i\,(k+\tfrac12)$, $k \in \mathbb Z$, and similarly as in (\ref{s0}) we set $s_0:=  \min\limits_{k \in {\mathbb Z}} |\Im( \mathrm i\,(k+\tfrac12))| = \frac{1}{2}$. The  Fourier cosine transform is given by
	$$
	{\hat f}_{\cos}(v) =  \textstyle \int\limits_0^{\infty} \operatorname{sech}(\pi x)\,\cos(2\pi x v)\,\,\mathrm{d} x
	= \tfrac12\,\operatorname{sech}(\pi v)\,, \quad v \in {\mathbb R}_{+}\,.
	$$
As in the proofs of Theorem \ref{Thm:samplingcospolynomial} and Corollary \ref{corlower},
$\sqrt{P}\,E^{\cos}_{P,P}(f) = \max_{v \in [0, P/2]} |\sigma_2(v)-\sigma_1(v)|$, so that
\begin{align}\label{eq:ung}
 \textstyle
\big|\sigma_1\big(\frac{P}{2}\big)\big| - \big|\sigma_2\big(\frac{P}{2}\big)\big| \le \sqrt{P} \, E^{\cos}_{P,P}(f)
\le   \max\limits_{v \in [0,\, P/2]} \big(\big|\sigma_1\big(v)\big| + \big|\sigma_2\big(v)\big| \big).
\end{align}
We observe that  $ \operatorname{sech} (t)
 \in [ \mathrm e^{-t} , \,  2 \mathrm e^{-t}]$ is monotonically decreasing
 for $t \ge 0$, such that
$$
 \textstyle \big|\sigma_1\big(\tfrac P2\big)\big|= {\hat f}_{\cos}\big(\tfrac P2\big) + 2 \sum\limits_{k=1}^\infty {\hat f}_{\cos}(\frac{P}{2} + kP) \ge  \tfrac12\operatorname{sech}\tfrac{\pi P}{2} \;\geq\; \tfrac12\,\mathrm e^{-\pi P/2}\,,
$$
Since $\operatorname{sech}''(t) = \operatorname{sech}(t)\big(1-2\operatorname{sech}^2 t\big)>0$
for $t > \operatorname{arsech}\frac{1}{\sqrt2} = 0.8814$, the function $\operatorname{sech}$ is
convex on $[0.89,\infty)$. For $P \ge 8$, $k \in \mathbb N$ and $v \in [0,\frac P2]$ all arguments
$\pi(kP\pm v)$ lie in $[ 4 \pi,\infty)$, so that
$v \mapsto {\hat f}_{\cos}(v+kP)+{\hat f}_{\cos}(kP-v)$ is convex
 and increasing in $v$ on $[0,\frac P2]$, hence maximal at
$v=\frac P2$. Summing over $k$ gives $|\sigma_1(v)| \le \sigma_1\big(\frac P2\big)$ and
\begin{align*}
 \textstyle \big|\sigma_1(v)\big| &\le   \textstyle
 \tfrac{1}{2}\operatorname{sech}\tfrac{\pi P}{2} +  \sum\limits_{k=1}^\infty \operatorname{sech}\big(\pi(\tfrac P2 + kP)\big)
 \le  \mathrm e^{-P\pi/2} + 2 \sum\limits_{k=1}^\infty \mathrm e^{-\frac{P\pi}{2}(1+2k)}
 < \mathrm e^{-P\pi/2} \big(1 + 10^{-10}\big)\,.
 \end{align*}
On the other hand, $\sigma_2\big(\tfrac P2\big) = \frac1P\sum_{n>P^2/2}(-1)^n\operatorname{sech}\tfrac{\pi n}{P}$
is an alternating series with monotonically decreasing terms, whence
$\big|\sigma_2\big(\tfrac P2\big)\big| \le \tfrac1P\operatorname{sech}\tfrac{\pi P}{2} \le \tfrac2P\mathrm e^{-\pi P/2}$.
 To obtain an upper bound for $\big|\sigma_2\big(v\big)\big|$ for all $v \in [0,\frac P2]$, we use
$\operatorname{sech}(t) \le 2\mathrm e^{-t}$ and estimate for every $v \in [0,\frac P2]$,
$$
\textstyle |\sigma_2(v)| \le \frac2P \sum\limits_{n>P^2/2}\mathrm e^{-\pi n/P}
= \frac{2}{P}\,\frac{\mathrm e^{-\pi P/2}}{\mathrm e^{\pi/P}-1}
= \Big( \frac{2}{\pi}\,\frac{t}{\mathrm e^{t}-1}\Big|_{t = \pi/P} \Big) \, \mathrm e^{-\pi P/2}
\le \frac{2}{\pi}\,\mathrm e^{-\pi P/2}\,,
$$
since $t \mapsto \frac{t}{\mathrm e^t-1}$ is decreasing with limit $1$ at $t=0$.
Thus we obtain from (\ref{eq:ung}) the estimate
\begin{align*}
 \textstyle
\frac{1}{\sqrt{P}} \big( \frac{1}{2} \mathrm e^{-\pi P/2}  -  \frac{2}{P} \mathrm e^{-\pi P/2} \big) \le   E^{\cos}_{P,P}(f)
\le   \frac{1}{\sqrt{P}}  \big( \tfrac{2}{\pi} \mathrm e^{-\pi P/2} + \mathrm e^{-\pi P/2} (1 + 10^{-10}) \big),
\end{align*}
and the assertion follows for $P\ge 8$, since $\frac12 - \frac2P \ge \frac14$ and
$\frac2\pi + 1 + 10^{-10} < 1.64$.

\label{Lemma:aliasingcheb}
	Let $P \in \mathbb N \setminus \{1\}$ be fixed. If $h \in C(I)$ satisfies
	$\sum_{n=0}^{\infty}|a_n[h]| < \infty$, then for every $n =0,\ldots,P$
	the aliasing formula
	\begin{equation}
		\textstyle a_n^{(P)}[h]
		= a_n[h] + \sum\limits_{k=1}^{\infty}\bigl(a_{n+2kP}[h]+a_{2kP-n}[h]\bigr)\,,
		\label{eq:aliasingcheb}
	\end{equation}
	is satisfied, where both series converge absolutely. In particular, for $n=0$ and $n=P$ we have
\begin{eqnarray*}
\textstyle a_0^{(P)}[h] &=& \textstyle a_0[h] + 2\,\sum\limits_{k=1}^{\infty} a_{2kP}[h]\,,\\
\textstyle a_P^{(P)}[h] &=& \textstyle 2\, \sum\limits_{k=1}^{\infty}a_{(2k-1)P}[h]\,.
\end{eqnarray*}
<!-- end lemma-4 -->
<!-- proof-9: body_md -->
By assumption, the Chebyshev series \eqref{eq:ChebSeries}
converges absolutely and uniformly on $I$ to $h$ (see
\cite[Chapter~6]{PlPoStTa23}), such that on the $(P+1)$-point Chebyshev grid we have
$$
\textstyle h\big(x_j^{(P)}\big) = \tfrac12\,a_0[h] + \sum\limits_{k=1}^{\infty} a_k[h]\,\cos \tfrac{jk \pi}{P}\,, \quad j=0,\ldots, P\,.
$$
Inserting this into \eqref{eq:DCTcoeffs} and using $2\cos\varphi\cos\psi =
\cos(\varphi-\psi) + \cos(\varphi+\psi)$, we obtain
$$
\textstyle 	a_n^{(P)}[h] = \tfrac{1}{P}\, a_0[h]\, \mathop{{\sum}''}\limits_{\!\!\!j=0}^{\!\!\!P} \cos \tfrac{\pi j n}{P} + \frac{1}{P}\sum\limits_{k=1}^{\infty} a_k[h]  \, \mathop{{\sum}''}\limits_{\!\!\!j=0}^{\!\!\!P} \Big(\cos \tfrac{\pi j (k-n)}{P} + \cos \tfrac{\pi j (k+n)}{P} \Big)\,.
$$
 We apply the relation
$$
\textstyle \tfrac{1}{P} \mathop{{\sum}''}\limits_{\!\!\!j=0}^{\!\!\!P}\cos \tfrac{\pi j m}{P} = \left\{ \begin{array}{ll} 1 &\quad m \in 2P\,\mathbb Z\,,\\
0 & \quad m \in \mathbb Z \setminus 2P\,\mathbb Z \,, \end{array}\right.
$$
 for $m = n$, $m = k-n$, and $m = k+n$, and obtain exactly the summation indices $k$ of the
form $k = n + 2mP$ or $k = 2mP - n$ with $m \in \mathbb N$,
 this yields
\eqref{eq:aliasingcheb}.

<!-- theorem-4: body_md -->
\label{Cor:Ech}
	Let $P \in \mathbb N \setminus \{1\}$ and $L \le P$ be fixed and let $h \in C(I)$ satisfy
	$\sum_{n=0}^{\infty}|a_n[h]| < \infty$. Then the parametric quadrature  error \eqref{eq:Ech}
	satisfies
	$$
	E^{\mathrm{Ch}}_{P}(h) \leq 2\,\delta_{\mathrm{Ch}}(h,P)\,.
	$$
	Furthermore, the reconstruction error $R^{\mathrm{Ch}}_{P,L}(h) = \|h - t_{P,L}\|_{C(I)} $ in \eqref{eq:EN} satisfies
	\begin{equation}
		\label{eq:mainestimate}
		 \textstyle R^{\mathrm{Ch}}_{P,L}(h) \leq \delta_{\mathrm{Ch}}(h,L) + \delta_{\mathrm{Ch}}(h,P).
	\end{equation}
	 In the special case $L = P$ we have
	$R^{\mathrm{Ch}}_{P,P}(h) \leq 2\,\delta_{\mathrm{Ch}}(h,P)$.
<!-- end theorem-4 -->
<!-- proof-12: body_md -->
1.\ The stated bound for  $E^{\mathrm{Ch}}_{P}(h)$ directly follows from
\eqref{eq:aliasrem} by the triangle inequality. For $n \leq P-1$, the occurring
indices satisfy $n+2kP \geq 2P > P$ and $2kP-n \geq 2P - (P-1) = P+1$, so that
each of the two sums is bounded by $\delta_{\mathrm{Ch}}(h,P)$. For $n=P$ we have
$\tfrac12\,a_P^{(P)}[h] - a_P[h] = \sum_{k=1}^{\infty} a_{(2k+1)P}[h]$, whose
indices exceed $P$ as well.

2.\ We estimate the reconstruction error $R^{\mathrm{Ch}}_{P,L}(h)$. We consider first the case $L<P$.
Inserting $a_n^{(P)}[h] =
a_n[h] + r_n^{(P)}[h]$ into \eqref{eq:samplingpoly} yields
the
error decomposition
\begin{equation}
	\label{eq:errordecomp}
	 \textstyle h(x) - t_{P,L}(x) = \underbrace{ \textstyle \sum\limits_{n=L+1}^{\infty}a_n[h]\,T_n(x)}_{\text{degree truncation}}
	- \underbrace{ \textstyle \mathop{{\sum}'}\limits_{n=0}^L r_n^{(P)}[h]\,T_n(x)}_{\text{sampling aliasing}}\,,
\end{equation}
where the prime indicates that the first term of the second sum is multiplied by $\tfrac12$.

The degree-truncation term is bounded by
$\sum_{n=L+1}^\infty|a_n[h]| = \delta_{\mathrm{Ch}}(h,L)$  since $|T_n(x)| \le 1$ for $x \in I$. For the sampling aliasing term
each remainder $r_n^{(P)}[h]$, $n =0,\ldots,L$, is a sum of coefficients $a_m[h]$
with $m=n+2kP\geq 2P$ and $m=2kP-n \geq 2P-L\geq P+1$, $k\in\mathbb N$.
Since $|T_n(x)|\le 1$ on $I$, the sampling aliasing term is bounded by
$\frac12\big|r_0^{(P)}[h]\big| + \sum_{n=1}^{L}\big|r_n^{(P)}[h]\big|$, and by \eqref{eq:aliasrem}
this is at most
$$
\textstyle \sum\limits_{k=1}^{\infty}\big|a_{2kP}[h]\big|
+ \sum\limits_{n=1}^{L}\sum\limits_{k=1}^{\infty}\Big(\big|a_{n+2kP}[h]\big| + \big|a_{2kP-n}[h]\big|\Big)\,,
$$
 where all occurring indices are pairwise disjoint.
Consequently
$$
\textstyle \mathop{{\sum}'}\limits_{n=0}^L \,\big|r_n^{(P)}[h]\big| \leq \sum\limits_{m=P+1}^\infty|a_m[h]| = \delta_{\mathrm{Ch}}(h,P)\,.
$$
For $L=P$  we obtain similarly with $
\textstyle \tilde{r}_P^{(P)}[h] := \tfrac12\,a_P^{(P)}[h] - a_P[h] $
that
\begin{equation}
	\label{eq:errordecompP}
\textstyle 	h(x) - t_{P,P}(x) = \sum\limits_{n=P+1}^{\infty}a_n[h]\,T_n(x)
	- \mathop{{\sum}'}\limits_{n=0}^{P-1} r_n^{(P)}[h]\,T_n(x)
	- \tilde{r}_P^{(P)}[h]\,T_P(x)\,,
\end{equation}
again with the prime indicating that the term $n=0$ is halved.
The same argument applies to $r_n^{(P)}[h]$, $n =0,\ldots,P-1$, where
now $m=2kP-n\geq 2P-(P-1)=P+1$, the extra term obeys
$\big|\tilde{r}_P^{(P)}[h]\big|\leq\sum_{k=1}^{\infty}\big|a_{(2k+1)P}[h]\big|$,
where the indices  in the sum are odd multiples of $P$ and hence are disjoint from the
previous ones, which
lie in $(2kP-P,2kP+P)$. Thus the
sampling aliasing term is again bounded by $\delta_{\mathrm{Ch}}(h,P)$  and the error estimates follow.

<!-- remark-6: body_md -->
\label{Rem:comparison}
1. The estimate $E^{\mathrm{Ch}}_{P}(h) \le 2\,\delta_{\mathrm{Ch}}(h,P)$ of Theorem~\ref{Cor:Ech}
 is the exact analogue of
Theorem~\ref{Thm:samplingcospolynomial}. In all three settings the quadrature  error
is bounded by twice a transform-side tail, $\delta_{+}({\hat f}_{\cos},P)$, $\delta_{+}({\hat f}_{\sin},P)$ and

L\,\delta_{+}(f,L)$ of \eqref{eq:cosmain} stems from the truncation of the sampling sum

\eqref{eq:DCTcoeffs} uses all available samples. Consequently no scaling factor
$L^{-1/2}$ is needed in \eqref{eq:Ech}.\\
For examples of Chebyshev expansions of special functions we refer to \textnormal{\cite[Table~A.2]{PlPoStTa23}}.

$\tfrac12 a_L[h]\,T_L(x)$ to \eqref{eq:errordecomp}, and $a_L[h]$ belongs neither to the

\eqref{eq:mainestimate} would fail for rapidly decaying coefficients. For $L=P$, by contrast,
halving is required. It yields the interpolating polynomial and, through
$a_P[h]-\tfrac12 a_P^{(P)}[h]=-\sum_{k\ge1} a_{(2k+1)P}[h]$, replaces the otherwise unbounded
contribution $a_P[h]$ by genuine aliasing of indices larger than  $P$.

\label{Thm:rational}
	Let $h$ be given as in $(\ref{simplepole})$. Assume that $w_j :=z_j+ s_j$ with $s_j := \sqrt{z_j^2-1}$, where the  branch of the square root $s_j$ is chosen in such a way that
	$|w_j|>1$. Then for
	$\rho_0:=\min\{|w_j|:\,j=1,\ldots,J\}>1$  we have
	\begin{equation}
		\label{eq:rationalcoeffs}
		 \textstyle a_n[h] = - 2\,\sum\limits_{j=1}^{J}\frac{c_j}{s_j}\,w_j^{-n}\,, \quad
		|a_n[h]| \leq \gamma_0\,\rho_0^{-n}\,,\quad
		\gamma_0 := 2\,\sum\limits_{j=1}^{J}\frac{|c_j|}{|s_j|}\,,
	\end{equation}
	for all $n\in{\mathbb N}_0$. Consequently, the quadrature error $E^{\mathrm{Ch}}_{P}(h)$ and the reconstruction error $R^{\mathrm{Ch}}_{P,L}(h)$
	satisfy
$$
	\textstyle 	E^{\mathrm{Ch}}_{P}(h) \leq \frac{2\gamma_0}{\rho_0-1}\,\rho_0^{-P}\,, \quad R^{\mathrm{Ch}}_{P,L}(h) \leq \frac{\gamma_0}{\rho_0-1}
		\bigl(\rho_0^{-L}+\rho_0^{-P}\bigr)\,.
$$
<!-- end theorem-5 -->
<!-- proof-14: body_md -->
We consider only  $h(x):= \frac{1}{x-z_j}$ with $z_j \in\mathbb C\setminus I$ for arbitrary $j \in \{1,\ldots,J\}$. Let $w_j = z_j+ s_j$ with $s_j =\sqrt{z_j^2-1}$ and $|w_j| >1$.
We study the Chebyshev expansion (\ref{eq:ChebSeries}) with $a_n:=
\tfrac{-2}{s_j}\,w_j^{-n}$, $n \in {\mathbb N}_0$  and prove that this expansion converges to $h$, thereby showing
$a_n[h]:=
a_n$ for $n \in {\mathbb N}_0$.

Since $|a_n| = \frac{2}{|s_j|}\,|w_j|^{-n}$ with $|w_j|>1$, we have $\sum_{n=0}^\infty |a_n| < \infty$, i.e.,
the Chebyshev series converges absolutely and uniformly on $I$ to a function
$g \in C(I)$. Using $x\,T_n(x) = \tfrac12\,\big(T_{n-1}(x) + T_{n+1}(x)\big)$,
we obtain
$$ \textstyle a_n[(x- z_j)\,g(x)] = \tfrac12\,\big(a_{n-1}[g] + a_{n+1}[g]\big) -
z_j\,a_n[g]\,,  \quad n \in {\mathbb N}_0\,, $$
with $a_{-1}[g] := a_1[g]$. For $n=0$ we conclude with $(z_j + s_j)^{-1} = z_j - s_j$
$$
\textstyle a_0[(x- z_j)\,g(x)] = a_1[g] - z_j\,a_0[g] = \tfrac{-2}{s_j}\,(z_j - s_j) + z_j\,\tfrac{2}{s_j} = 2\,,
$$
and for $n \in \mathbb N$,
$$
a_n[(x- z_j)\,g(x)] = - \tfrac{w_j^{1-n}}{s_j}\,\big(1 + (z_j - s_j)^2 - 2 z_j\,(z_j - s_j)\big) = 0\,.
$$
Hence $(x- z_j)\,g(x) = 1$ for all $x \in I$, i.e., $g = h$.
The representation of $a_n[h]$ in (\ref{eq:rationalcoeffs}) follows by linearity. The corresponding

Observe that by definition of $\rho_0$,  there exists one $w_j$ with $\rho_0 = |w_j|$, and therefore
the decay $\rho_0^{-n}$  in $(\ref{eq:rationalcoeffs})$ cannot be improved.
In other words, $\rho_0$ is the parameter of the greatest Bernstein region ${\mathcal E}_{\rho_0}$ in which

\textstyle r_n^{(P)}[h_j] = \frac{-2c_j}{s_j}\,\big(w_j^{n}+w_j^{-n}\big)\,\frac{w_j^{-2P}}{1-w_j^{-2P}}\,,

$$
which has decay rate $\rho_0^{-P}$ and matches Theorem $\ref{Thm:rational}$ up to the constant
factor $\frac{2\rho_0}{\rho_0-1}$.

\label{Lemma:powercauchykernel}
	Let $k \in \mathbb{N}$ be given. For fixed $z_0\in\mathbb{C}\setminus I$,
	let $s_0 := \sqrt{z_0^2-1}$ and $w_0 := z_0 + s_0$, where the
	branch of the square root $s_0$ is chosen in such a way that $|w_0|>1$.\\
   Then the Chebyshev coefficients of the $k$th power of the Cauchy kernel
   $h_k(x) := \tfrac{1}{(x-z_0)^k}$, $x \in I$,
    are given as
   \begin{equation}
   \label{eq:an[hk]}
   \textstyle a_n[h_k] =  \frac{2}{\pi} \int\limits_0^{\pi} \frac{\cos(n\theta)}{(\cos\theta -z_0)^k} \,\mathrm{d} \theta =
   \tfrac{2\,(-1)^k}{(k-1)!\,s_0^k}\,q_k(n)\,w_0^{-n}\,,\quad n\in {\mathbb N}_0\,, \;k\in \mathbb N\,,
   \end{equation}
   with a monic polynomial $q_k(n)$ of degree $k-1$, which satisfies  the recurrence relation
   \begin{equation}
   \label{eq:qk(n)}
  \textstyle  - (k-1)\,s_0\,q_{k-1}(n) = \tfrac{w_0}{2}\,q_k(n-1) + \tfrac{1}{2w_0}\,q_k(n+1) - z_0\,q_k(n)\,,\quad q_1(n) = 1\,,
   \end{equation}
   with the initial condition
   \begin{equation}
   \label{eq:initialcond}
  \textstyle  - (k-1)\,s_0\,q_{k-1}(0) = w_0^{-1}\,q_k(1) - z_0\,q_k(0)\,, \quad k \in {\mathbb N}\setminus \{1\}\,.
   \end{equation}
   Further the Chebyshev coefficients $a_n[h_k]$ can be estimated by
   \begin{equation}
   \label{eq:|an[hk]|}
   \big|a_n[h_k]\big| \leq c_k\,(n+1)^{k-1}\,|w_0|^{-n}\,, \quad n \in {\mathbb N}_0\,,
   \end{equation}
   with some $c_k > 0$.\\
   Let $P\in \mathbb{N}\setminus \{1\}$ and $L \in \mathbb N$ with $L \leq P$
   and $(L + 1)\,\ln (|w_0|) > k -1$
   be fixed.
 Then the quadrature error $E^{\mathrm{Ch}}_{P}(h_k)$ and the reconstruction error $R^{\mathrm{Ch}}_{P,L}(h_k)$
	satisfy
$$
	\textstyle 	E^{\mathrm{Ch}}_{P}(h_k) \leq \frac{2c_k\,(P+1)^k  |w_0|^{-P}}{(P+1) \,\ln (|w_0|) +1 -k}\,,
	\quad R^{\mathrm{Ch}}_{P,L}(h_k) \leq \frac{c_k\, (P+1)^k  |w_0|^{-P}}{(P+1)\, \ln (|w_0|) + 1 - k} +
		\frac{c_k (L+1)^k  |w_0|^{-L}}{(L+1)\, \ln (|w_0|) + 1 - k}\,.
$$

1. We show \eqref{eq:an[hk]} by induction with respect to $k \in \mathbb N$. Obviously, \eqref{eq:an[hk]} is true in the case $k =1$ with $q_1(n)=1$ by Theorem \ref{Thm:rational}. Assume now that
\eqref{eq:an[hk]} holds for some $k \in \mathbb N$. Let $g$ be the function with the Chebyshev coefficients
$$
a_n[g] = \tfrac{2\,(-1)^{k+1}}{k!\,s_0^{k+1}}\,q_{k+1}(n)\,w_0^{-n}\,,\quad n\in {\mathbb N}_0\,.
$$
Since $\sum_{n\ge0}|a_n[g]| < \infty$, the corresponding Chebyshev series \eqref{eq:ChebSeries}
converges absolutely and uniformly on $I$ to some $g \in C(I)$. For $g_1(x) := (x-z_0)\,g(x)$,
the recurrence $x\,T_n(x) = \frac12\big(T_{n-1}(x)+T_{n+1}(x)\big)$ gives 
$$ \textstyle a_n[g_1] = \frac12 a_{n-1}[g] + \frac12 a_{n+1}[g] - z_0\,a_n[g], \qquad n \in {\mathbb N}_0, $$
 with
$a_{-1}[g] = a_1[g]$.   For $n=0$ it follows by \eqref{eq:initialcond} that
\begin{align*}
\textstyle a_0[g_1] &= a_1[g] \!- \!z_0\,a_0[g] =
 \tfrac{2(-1)^{k+1}}{k!s_0^{k+1}}\,\big(w_0^{-1}q_{k+1}(1) \!- \!z_0q_{k+1}(0)\big)
= \tfrac{2(-1)^k}{(k-1)!s_0^k}\,q_k(0) = a_0[{h_k}].
\end{align*}
For arbitrary $n \in \mathbb N$ we obtain by \eqref{eq:qk(n)} that
\begin{align*}
a_n[g_1] &=   \textstyle \tfrac12\,a_{n-1}[g] + \tfrac12\,a_{n+1}[g] - z_0\,a_n[g]\\
&= \textstyle  \tfrac{2\,(-1)^{k+1}}{k!\,s_0^{k+1}}\,\big(\tfrac{w_0}{2}\,q_{k+1}(n-1) + \tfrac{w_0^{-1}}{2}\,q_{k+1}(n+1) - z_0\,q_{k+1}(n)\big)\,w_0^{-n}\,,\\
&= \textstyle  \tfrac{2\,(-1)^k}{(k-1)!\,s_0^k}\,q_k(n)\,w_0^{-n} = a_n[{h_k}]\,.
\end{align*}

Hence
$g_1 = h_k$ on $I$, i.e.\ $g = h_{k+1}$.\\
2. Formula (\ref{eq:an[hk]}) yields  $|a_n[h_k]| = \tfrac{2}{(k-1)!\,|s_0|^k}\,|q_k(n)|\,|w_0|^{-n}$.
Since $q_k(n)$ is a monic polynomial of degree $k-1$, the supremum
$c_k:= \sup_{n \in {\mathbb N}_0} \frac{|a_n[h_k]| }{(n+1)^{k-1} \,|w_0|^{-n}}$ is finite
and \eqref{eq:|an[hk]|} holds. Note that the function $x^{k-1}\,|w_0|^{-x}$, $x\in [P+1, \infty)$, is decreasing since

$$ \textstyle \frac{\,\mathrm{d}}{\,\mathrm{d} x}\big(x^{k-1}|w_0|^{-x}\big) = x^{k-2}|w_0|^{-x}\big(k-1-x\ln (|w_0|)\big) \le 0$$
 for $x \ge P+1$, which follows from the assumption $(P+1)\,\ln (|w_0|) \geq (L+1)\,\ln (|w_0|) > k-1$. Thus we can estimate
\begin{align*}
\textstyle \delta_{\mathrm{Ch}}(h_k,P)&=  \textstyle \sum\limits_{n=P+1}^{\infty} |a_n[h_k]| \le
c_k \sum\limits_{n=P+1}^{\infty} (n+1)^{k-1} |w_0|^{-n}  = c_k |w_0| \sum\limits_{n=P+2}^{\infty} n^{k-1} |w_0|^{-n}\\
&\le \textstyle c_k |w_0| \int\limits_{P+1}^{\infty} x^{k-1} \,|w_0|^{-x} \, \,\mathrm{d} x =
\frac{c_k |w_0|}{(\ln (|w_0|))^k} \int\limits_{(P+1) \ln (|w_0|)}^{\infty} u^{k-1} \, \mathrm e^{-u} \, \,\mathrm{d} u\\
&= \textstyle  \frac{c_k |w_0|}{(\ln (|w_0|))^k}\,\Gamma\big(k,(P+1) \ln (|w_0|)\big)\,.
\end{align*}
Using \cite[Proposition 2.7]{Pi20}, the upper incomplete gamma function can be estimated for
$w_1 := (P+1)\ln (|w_0|) > k-1$ by
$$
\textstyle \Gamma(k,w_1) := \int_{w_1}^{\infty} u^{k-1}\,\mathrm e^{-u}\,\,\mathrm{d} u \leq \frac{w_1^k}{w_1 + 1 - k}\,\mathrm e^{-w_1}\,.
$$
Thus we obtain with $|w_0|\,\mathrm e^{-w_1} = |w_0|^{-P}$ that
$$
\textstyle \delta_{\mathrm{Ch}}(h_k,P) \leq \frac{c_k\,(P+1)^k\,|w_0|^{-P}}{(P+1)\,\ln (|w_0|) + 1 -k}\,,
$$
where we have $(P+1)\,\ln (|w_0|) + 1 -k > 0$ by assumption.
The estimates for the quadrature error and the reconstruction error now follow from Theorem \ref{Cor:Ech}.

	\label{Ex:runge}
	The Runge function $h(x)=\tfrac{1}{1+x^2}$ is real-analytic with simple poles
	$z_{1,2} = \pm\mathrm i$ and $w_{1,2} = \pm\mathrm i(1+\sqrt2)$, so that $\rho_0 = 1+\sqrt2$.
	We obtain
	$$
	\textstyle a_n[h] = \tfrac{1}{\sqrt 2}\,\big(w_1^{-n} + w_2^{-n}\big) = \sqrt 2\,(1 + \sqrt 2)^{-n}\, \cos\tfrac{n \pi}{2}\,, \quad n \in {\mathbb N}_0\,,

	a_n[h]\big| \leq \sqrt 2\,\rho_0^{-n}$. The Chebyshev decay rate for even
