\label{thm: Weighted functional First estimation}
        There exist constants $C_{\text{Res}}, C_r>0$ independent of $(\tilde{u},\tilde{\lambda})$ such that for all $\theta\in \Lambda_\rho$
         
        \begin{equation}\label{eq: First upper error estimate}
            \Vert u-\tilde{u} \Vert_V^2+\Vert {\lambda}-\tilde{\lambda} \Vert_{W}^2\leq C_{\text{Res}}\:\Vert \bm{A}\tilde{u}-\bm{\ell}+\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}^2+C_r\:\bm{E}(\theta;\tilde{u},\tilde{\lambda}).

By applying \eqref{eq: Contiuity & Ellipticity of A} and \eqref{Mixed Formulation Gleichung} as well as writing $\langle \bm{B}{\lambda}\circ\gamma,\tilde{u}\rangle_V = \langle \bm{B}{\lambda},\gamma(\tilde{u})\rangle_W$, we obtain
         
        \begin{align*}
            C_a\Vert u-\tilde{u} \Vert_V^2 &\leq \langle \bm{A}(u-\tilde{u}),u-\tilde{u}\rangle_V\\
            & = \langle \bm{\ell}-\bm{B}{\lambda}\circ \gamma-\bm{A}\tilde{u},u-\tilde{u}\rangle_V\\
            & = \langle \bm{\ell} -\bm{B}\tilde{\lambda}\circ \gamma -\bm{A}\tilde{u},u-\tilde{u}\rangle_V- \langle \bm{B}{\lambda}\circ\gamma,u-\tilde{u}\rangle_V + \langle \bm{B}\tilde{\lambda}\circ\gamma,u-\tilde{u}\rangle_V.
        \end{align*}
     
        Then, by the linearity of $\bm{B}$ and $\gamma$, the definition of the dual norm and Young's inequality we have
        \begin{equation*}
        \aligned
             C_a\Vert u-\tilde{u} \Vert_V^2 &\leq \Vert \bm{A}\tilde{u}-\bm{\ell}+\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}\Vert u-\tilde{u} \Vert_V +\langle \bm{B}(\tilde{\lambda}-{\lambda}),\gamma(u-\tilde{u})\rangle_W\\
            &\leq \frac{1}{4\delta} \Vert \bm{A}\tilde{u}-\bm{\ell}+\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}^2 + \delta\Vert u-\tilde{u} \Vert_V^2 + \langle \bm{B}(\tilde{\lambda}-{\lambda}),\gamma(u-\tilde{u})\rangle_W
            \endaligned
        \end{equation*}
     
        for all $\delta>0$.
        This implies
     
        \begin{equation}\label{u-v Estimate}
            \Vert u-\tilde{u} \Vert_V^2 \leq \frac{1}{4\delta(C_a-\delta)}\Vert \bm{A}\tilde{u}-\bm{\ell}+\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}^2 +\frac{1}{C_a-\delta}\langle \bm{B}(\tilde{\lambda}-{\lambda}),\gamma(u-\tilde{u})\rangle_W
        \end{equation}
     
        for $C_a>\delta>0$.
        Exploiting \eqref{eq: Definition of B}, \eqref{eq: conditions on lambda and gamma u} and $\lambda \in\Lambda$, we obtain for any $\theta \in \Lambda_\rho$ that
     
        \begin{equation*}
            \aligned
                &\langle \bm{B}(\tilde{\lambda}-{\lambda}),\gamma(u-\tilde{u})\rangle_W\\
                &\quad =  \langle \bm{B}\tilde{\lambda},\gamma(u-\tilde{u})\rangle_W -  \langle \bm{B}\lambda,\gamma(u-\tilde{u})\rangle_W + \bigl(\theta,\mathfrak{L}(\gamma(u-\tilde{u}))\bigr)_{0,G}- \bigl(\theta,\mathfrak{L}(\gamma(u-\tilde{u}))\bigr)_{0,G}\\
                &\quad = \bigl(\mathfrak{L}(\tilde{\lambda})-\theta,\mathfrak{L}(\gamma(u-\tilde{u}))\bigr)_{0,G} + \bigl(\theta-\mathfrak{L}(\lambda),\mathfrak{L}(\gamma(u))\bigr)_{0,G}-\bigl(\theta,\mathfrak{L}(\gamma(\tilde{u}))\bigr)_{0,G} + \langle \bm{B}\lambda,\gamma(\tilde{u})\rangle_W\\
                &\quad \leq \bigl(\mathfrak{L}(\tilde{\lambda})-\theta,\mathfrak{L}(\gamma(u-\tilde{u}))\bigr)_{0,G} + j(\tilde{u}) -\bigl(\theta,\mathfrak{L}(\gamma(\tilde{u}))\bigr)_{0,G} +\bigl(\theta,\mathfrak{L}(\gamma(u))\bigr)_{0,G} - j(u).
            \endaligned
        \end{equation*}
         
        Then, as $\theta \in \Lambda_\rho$, we obtain with \eqref{eq: j-mu geq 0} and \eqref{eq: Continuity of gamma and L} that
         
        \begin{equation*}
        \aligned
             &\langle \bm{B}(\tilde{\lambda}-{\lambda}),\gamma(u-\tilde{u})\rangle_W\\
                &\quad \leq \bigl(\mathfrak{L}(\tilde{\lambda})-\theta,\mathfrak{L}(\gamma(u-\tilde{u}))\bigr)_{0,G} + j(\tilde{u}) -\bigl(\theta,\mathfrak{L}(\gamma(\tilde{u}))\bigr)_{0,G}+\bigl(\theta,\mathfrak{L}(\gamma(u))\bigr)_{0,G} - j(u)\\
                &\quad \leq \bigl(\mathfrak{L}(\tilde{\lambda})-\theta,\mathfrak{L}(\gamma(u-\tilde{u}))\bigr)_{0,G} + j(\tilde{u}) -\bigl(\theta,\mathfrak{L}(\gamma(\tilde{u}))\bigr)_{0,G}\\
                &\quad \leq \Vert \mathfrak{L}(\tilde{\lambda})-\theta \Vert_{0,G}\Vert \mathfrak{L}(\gamma(u-\tilde{u})) \Vert_{0,G}+j(\tilde{u}) -\bigl(\theta,\mathfrak{L}(\gamma(\tilde{u}))\bigr)_{0,G}\\
                &\quad \leq C_LC_\gamma\Vert \theta-\mathfrak{L}(\tilde{\lambda}) \Vert_{0,G}\Vert u-\tilde{u} \Vert_V+j(\tilde{u}) -\bigl(\theta,\mathfrak{L}(\gamma(\tilde{u}))\bigr)_{0,G}.
                \endaligned
        \end{equation*}
         
        We combine the above estimate with Young's inequality, which results in
         
        \begin{equation}\label{Kappa Lambda duale Paarung}
            \aligned
                 \langle \bm{B}(\tilde{\lambda}-{\lambda}),\gamma(u-\tilde{u})\rangle_W
                &\leq \frac{1}{4\varepsilon}\Vert \theta-\mathfrak{L}(\tilde{\lambda}) \Vert_{0,G}^2+\varepsilon C_L^2C_\gamma^2\Vert u-\tilde{u} \Vert_V^2+j(\tilde{u}) -\bigl(\theta,\mathfrak{L}(\gamma(\tilde{u}))\bigr)_{0,G}\\
                &\leq \varepsilon C_L^2C_\gamma^2\Vert u-\tilde{u} \Vert_V^2 + \frac{1}{\varepsilon}\bm{E}(\theta;\tilde{u},\tilde{\lambda})
            \endaligned
        \end{equation}
        
        for all $\varepsilon\in (0,1)$ as \eqref{eq: j-mu geq 0} holds. Inserting \eqref{Kappa Lambda duale Paarung} into \eqref{u-v Estimate} yields the inequality
         
        \begin{equation}\label{eq: u-v intermediate estimate}
            \Vert u\!-\!\tilde{u} \Vert_V^2 \leq \frac{1}{4\delta(C_a\!-\!\delta\!-\!\varepsilon  C_L^2C_{\gamma}^2) }\Vert \bm{A}\tilde{u}\!-\!\bm{\ell}\!+\!\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}^2 +\frac{1}{\varepsilon( C_a\!-\!\delta\!-\!\varepsilon C_L^2C_{\gamma}^2)}\, \bm{E}(\theta;\tilde{u},\tilde{\lambda})
        \end{equation}
         
        for $0<\varepsilon<\min \{1,(C_a-\delta)(C_L^2C_{\gamma}^2)^{-1}\}$.
        Next, by the inf-sup condition of Theorem~\ref{thm:inf-sup} and \eqref{Mixed Formulation Gleichung} we obtain
         
        \begin{equation*}
            \aligned
                \beta \Vert \lambda-\tilde{\lambda} \Vert_W & \leq \sup\limits_{\substack{v\in V\\ \Vert v \Vert_V=1}}\langle \bm{B}(\lambda-\tilde{\lambda}),\gamma(v)\rangle_W\\
                &=\sup\limits_{\substack{v\in V\\ \Vert v \Vert_V=1}}\langle \bm{B}(\lambda-\tilde{\lambda})\circ \gamma,v\rangle_V\\
                &= \Vert \bm{B}(\lambda-\tilde{\lambda})\circ\gamma \Vert_{V^*}\\
                &=  \Vert \bm{\ell}-\bm{A} u-\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}\\
                &\leq \Vert \bm{A} \tilde{u}-\bm{\ell}+\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}+\Vert \bm{A}(u-\tilde{u}) \Vert_{V^*}.
            \endaligned
        \end{equation*}
         
        Then, we use \eqref{eq: Contiuity & Ellipticity of A} and have
         
        \begin{equation}\label{lambda-kappa estimate}
        \aligned
                \Vert \lambda-\tilde{\lambda} \Vert_W^2 &\leq \left(\frac{1}{\beta}\right)^2 \bigl(\Vert \bm{A} \tilde{u}-\bm{\ell}+\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}+C_A\Vert u-\tilde{u} \Vert_{V}\bigr)^2\\
                &\leq 2\left(\frac{1}{\beta}\right)^2\Vert \bm{A} \tilde{u}-\bm{\ell}+\bm{B}\tilde{\lambda}\circ\gamma \Vert_{V^*}^2+2\left(\frac{C_A}{\beta}\right)^2\Vert u-\tilde{u} \Vert_V^2.
                \endaligned
        \end{equation}
         
        Adding \eqref{lambda-kappa estimate} to \eqref{eq: u-v intermediate estimate} yields \eqref{eq: First upper error estimate} with 
         
        \begin{equation*}
        \aligned
            C_{\text{Res}}\!&:=\!\frac{\beta^2\!+\!2\,C_A^2\!+\!8\delta(C_a\!-\!\delta\!-\!\varepsilon C_L^2C_\gamma^2)}{\beta^24\delta(C_a\!-\!\delta\!-\!\varepsilon  C_L^2C_{\gamma}^2 )} > 0,\qquad 
            C_r\!&:=\!\frac{\beta^2\!+\!2\,C_A^2}{\varepsilon \beta^2(C_a\!-\!\delta\!-\!\varepsilon  C_L^2C_{\gamma}^2)} > 0
        \endaligned
        \end{equation*}
         
        for $C_a>\delta>0$ and $\min\{(C_a-\delta)(C_L^2C_{\gamma}^2)^{-1},1\}>\varepsilon>0$.

It is clear that $0\in\Lambda$ and, thus, $\Lambda\neq\emptyset$. Furthermore, let $\zeta\in[0,1]$ and $w,z\in\Lambda$. Then it holds by \eqref{eq: Definition of B} and the linearity of $\mathfrak{L}$ that
     
    \begin{equation*}
    \aligned
        \langle \bm{B} (\zeta w+(1-\zeta)z), \gamma(v)\rangle_W&=\left(\mathfrak{L}(\zeta w+(1-\zeta)z),\mathfrak{L}(\gamma(v))\right)_{0,G}\\
        &=\zeta\;(\mathfrak{L}(w),\mathfrak{L}(\gamma(v)))_{0,G}+(1-\zeta)(\mathfrak{L}(z),\mathfrak{L}(\gamma(v)))_{0,G}\\
        &\leq j(v),
    \endaligned
    \end{equation*}
     
    which means that $\Lambda$ is convex. Next, let $(w_n)_{n\geq1}\subset \Lambda$ be a sequence converging to $w\in W$. By the continuity of the $\mathfrak{L}$ and the inner product it holds that
     
    \begin{equation*}
    \aligned
        \langle \bm{B} w, \gamma(v)\rangle_W &= (\lim\limits_{n\to\infty}\mathfrak{L}(w_n),\mathfrak{L}(\gamma(v))_{0,G}=\lim\limits_{n\to\infty}(\mathfrak{L}(w_n),\mathfrak{L}(\gamma(v))_{0,G}\leq j(v),
    \endaligned
    \end{equation*}
     
    which implies that $w\in \Lambda$. Therefore, $\Lambda$ is closed.

Let $w\in W$ be arbitrary.
    As $\gamma:V\to \gamma(V)$ is surjective and $\gamma(V)$ is a Banach space with $\Vert \cdot \Vert_{\gamma(V)}$, it follows from \cite[Lem.~2.3]{Burg2015} that there exists a subspace $V_\gamma\subset V$ with $\gamma(V_\gamma)=\gamma(V)$ and a constant $c_\gamma>0$ such that
         
        \begin{equation*}
            \Vert v \Vert_V\leq c_\gamma\,\Vert \gamma(v) \Vert_{\gamma(V)}
        \end{equation*}
         
        for all $v\in V_\gamma$.
        Additionally, as $\gamma(V)$ is dense in $W$ it holds that $\Vert \bm{B} w \Vert_{W^*}=\Vert \bm{B} w|_{\gamma(V)} \Vert_{\gamma(V)^*}$, see e.g.~\cite[Prop.~5.4]{Stein2011}.
        Therefore, we obtain 
         
        \begin{equation*}
            \aligned
            \Vert \bm{B} w \Vert_{W^*}=
            \Vert \bm{B} w\!|_{\gamma(\!V\!)} \Vert_{\gamma\!(\!V\!)^*}&\!=\!\sup\limits_{z\in \gamma(\!V\!)\backslash\{0\}}\frac{\langle\bm{B} w,z\rangle_W}{\Vert z \Vert_{\gamma(\!V\!)}}\\
             &\!=\! \sup\limits_{v_\gamma\in V_\gamma\backslash\{0\}}\frac{\langle\bm{B} w,\gamma(v_\gamma)\rangle_W}{\Vert \gamma(\!v_\gamma\!) \Vert_{\gamma(V)}}\\
             &\leq c_\gamma\!\sup\limits_{v_\gamma\in V_\gamma\backslash\{0\}}\frac{\langle\bm{B} w,\gamma(\!v_\gamma\!)\rangle_W}{\Vert v_\gamma \Vert_{V}}\\
            &\leq c_\gamma\!\sup\limits_{v\in V\backslash\{0\}}\frac{\langle\bm{B} w,\gamma(\!v\!)\rangle_W}{\Vert v \Vert_V},
            \endaligned

        that together with $C_b \Vert w \Vert_W \leq  \Vert \bm{B} w \Vert_{W^*}$, which follows from \eqref{eq: B cont and ell}, yields the assertion.

<!-- proof-14: body_md -->
For the proof we use similar arguments as in \cite[Thm.~2.2]{Burg2015}.
    Let $u\in V$ be the solution of the variational inequality \eqref{Variational Inequality}. First, we show that $\tilde{\bm{q}} \in \gamma(V)^*$ given by
         
        \begin{align*}
            \langle \tilde{\bm{q}}, \gamma(v)\rangle_{\gamma(V)}= \langle\bm{\ell}-\bm{A}u,v\rangle _V
        \end{align*}
         
    is well-defined. To that end, let $v_0,v_1\in V$ such that $\gamma(v_0)=\gamma(v_1)$.
     As $v_0-v_1+u\in V$ with $\gamma(v_0-v_1+u) = \gamma(u)$ for $u$ the solution of the variational inequality \eqref{Variational Inequality} it holds that
         
        \begin{align*}
            0&\leq \langle \bm{A}u-\bm{\ell}, v_0-v_1+u-u\rangle_V + j(v_0-v_1+u)-j(u)\\
            &=\langle \bm{A}u-\bm{\ell}, v_0-v_1\rangle_V + j(u) -j(u) \\
            &=\langle \bm{A}u-\bm{\ell}, v_0-v_1\rangle_V.
        \end{align*}
         
        Conversely, as $v_1-v_0+u\in V$ it follows in the same way that $0\leq\langle \bm{A}u-\bm{\ell}, v_1-v_0\rangle_V.$ This implies that $\langle \bm{A}u-\bm{\ell}, v_0-v_1\rangle_V =0$ and, thus,
         
        \begin{align*}
            \langle \tilde{\bm{q}}, \gamma(v_0)\rangle_{\gamma(V)}=\langle \bm{\ell}-\bm{A}u,v_0\rangle_V=\langle \bm{\ell}-\bm{A}u,v_1\rangle_V= \langle \tilde{\bm{q}}, \gamma(v_1)\rangle_{\gamma(V)}
        \end{align*}
         
        for all $v_0,v_1\in V$ with $\gamma(v_0)=\gamma(v_1)$, which gives the well-definedness of $\tilde{\bm{q}}$.\\
        Next, the density of $\gamma(V)$ in $W$ implies the existence of a unique extension $\bm{q}\in W^*$ of $\tilde{\bm{q}}$, see e.g.~\cite[Prop.~5.4]{Stein2011}.
        Consequently, it holds that
         
        \begin{align*}
            \langle \bm{q}, \gamma(v)\rangle_W = \langle \tilde{\bm{q}}, \gamma(v)\rangle_{\gamma(V)} = \langle\bm{\ell}-\bm{A}u,v\rangle_V
        \end{align*}
         
        for all $v\in V$. As the linear operator $\bm{B}$ is continuous and elliptic, there       
        there exists a unique $\lambda\in W$ such that $\bm{B} \lambda = \bm{q}$, see e.g.~\cite[Thm.~5.10]{Han2013}. Thus, it holds that
         
        \begin{align*}
            \langle \bm{B} \lambda, \gamma(v)\rangle_W = \langle \bm{q}, \gamma(v)\rangle_W = \langle\bm{\ell}-\bm{A}u,v\rangle_V
        \end{align*}
            
        for all $v\in V$, which is \eqref{Mixed Formulation Gleichung}.
        In order to show that $\lambda\in \Lambda$, let $v\in V$ be arbitrary. It holds by \eqref{Variational Inequality} and the triangle inequality that 
         
        \begin{align*}
            \langle \bm{B} \lambda,\gamma(v)\rangle_W &= \langle \bm{\ell}-\bm{A} u,v+u-u\rangle_V
            \leq j(v+u)-j(u)
            \leq j(v)+j(u)-j(u)
            = j(v)
        \end{align*}
         
        and, thus, $\lambda\in\Lambda$.

        To show \eqref{eq: conditions on lambda and gamma u}, we set $v=0$ in \eqref{Variational Inequality}, which gives
         
        \begin{align*}
            \langle \bm{B} \lambda,\gamma(u)\rangle_W = \langle \bm{\ell} -\bm{A} u, u\rangle_V \geq j(u),
        \end{align*}
         
        but as it also holds that $\langle \bm{B} \lambda,\gamma(u)\rangle_W\leq j(u)$ as $\lambda\in\Lambda$, we obtain
         
        \begin{align*}
            j(u) = \langle \bm{B} \lambda,\gamma(u)\rangle_W.
        \end{align*}
         
        To show \eqref{Mixed Formulation Ungleichung}, let $\mu\in\Lambda$. Then we have with \eqref{eq: conditions on lambda and gamma u} that
         
        \begin{align*}
            \langle \bm{B} (\mu-\lambda),\gamma(u)\rangle_W \leq j(u) - \langle \bm{B} \lambda,\gamma(u)\rangle_W = j(u)-j(u) = 0.
        \end{align*}
         
        Consequently, $(u,\lambda)\in V\times \Lambda$ solves the problem \eqref{Mixed Formulation}.

        For the opposite direction let us assume that $(u,{\lambda})$ is the solution of the mixed formulation \eqref{Mixed Formulation}. From $\lambda \in \Lambda$ and \eqref{Mixed Formulation Ungleichung} we obtain as above that \eqref{eq: conditions on lambda and gamma u} is true.     
        Then, as $u-v\in V$ for any $v\in V$, it holds that
         
        \begin{align*}
            0&= \langle \bm{A}u-\bm{\ell},v-u\rangle_V+\langle\bm{B} \lambda,\gamma(v-u)\rangle_W\\
            &= \langle\bm{A}u-\bm{\ell},v-u\rangle_V + \langle\bm{B} \lambda,\gamma(v)\rangle_W - j(u)\\
            &\leq  \langle\bm{A}u-\bm{\ell},v-u\rangle_V + j(v) - j(u).
        \end{align*}
         
        Therefore, $u$ is a solution of the variational inequality \eqref{Variational Inequality}.

    Due to the above proven equivalence of \eqref{Variational Inequality} and \eqref{Mixed Formulation}, the uniqueness of $u$ carries over from \eqref{Variational Inequality}. Let $(u,\lambda_1)$ and $(u,\lambda_2)$ be both solutions of \eqref{Mixed Formulation}. By subtracting the corresponding equalities in \eqref{Mixed Formulation Gleichung} we obtain
         
        \begin{align*}                             
            \langle \bm{B}(\lambda_1-\lambda_2),\gamma(v)\rangle_W = 0
        \end{align*}
         
        for all $v\in V$.
        The uniqueness of $\lambda$ now follows with the inf-sup condition \eqref{lem:dense subset norm estimation}.
