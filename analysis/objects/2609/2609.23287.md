\label{defn: t-mot}
Let $F$ be a field extension of $k$.
A \emph{$t$-motive over $F$} is an $F[t,\tau]$-module $M$ which is free of finite rank over $F[t]$ and satisfies
\begin{equation}\label{eqn: *-con}
(t-\theta)^n M \subset F \cdot (\tau M), \quad \text{ for }  n \gg 0.
\end{equation}

\label{sec: good-red}

Let $M$ be a $t$-motive over $F$, where $F$ is a finite extension of $k_{v}$ in $\mathbb{C}_{v}$.
Denote by $O_{F}$ the ring of integers in $F$ and $\mathfrak{p}_F$ the unique maximal ideal of $O_{F}$.
Let $\mathbb{F}_F\mathrel{\vcenter{\baselineskip0.5ex \lineskiplimit0pt
                     \hbox{\scriptsize.}\hbox{\scriptsize.}}} 
                     = O_{F}/\mathfrak{p}_F$, the residue field of $F$, which is naturally regarded as a subfield of $O_{F}$ under the Teichm\"uller embedding
 $\matheur{T}: \mathbb{F}_F \hookrightarrow O_F$ 
given as follows:
\begin{equation}\label{eqn: Teichmuller}
\mathbb{F}_F= \frac{O_F}{\mathfrak{p}_F}\ni \ c\bmod \mathfrak{p}_F \ \longmapsto \ \lim_{n\rightarrow\infty } c^{\#(\mathbb{F}_F)^n} \ \in O_F.
\end{equation}
A \emph{model of $M$} is a left $O_{F}[t,\tau]$-submodule $\CMcal{M}$ of $M$ which is locally-free of finite rank over $O_{F}[t]$ and satisfies $F \CMcal{M} = M$.
By Seshadri's theorem (i.e.~Quillen--Suslin theorem over a principal ideal domain, see \cite{Qui76}, \cite{Sus76} and \cite[Ch.~5, Theorem 2.7]{Lam78}), one has that $\CMcal{M}$ must be free over $O_F[t]$.
Define $\overline{\CMcal{M}} \mathrel{\vcenter{\baselineskip0.5ex \lineskiplimit0pt
                     \hbox{\scriptsize.}\hbox{\scriptsize.}}} 
                     = \CMcal{M} / \mathfrak{p}_F \CMcal{M}$, which is naturally equipped with a left $\mathbb{F}_F[t,\tau]$-module structure.

\begin{defn}\label{defn: model}
Keep the above notation. The model $\CMcal{M}$ of $M$ is \emph{good} if the map $\tau: \overline{\CMcal{M}} \rightarrow \overline{\CMcal{M}}$ is injective.
We say that \emph{$M$ has good reduction} if there exists a good model of $M$.
\end{defn}

\begin{rem}\label{rem: good-model}
Let $M$ be a $t$-motive over a finite extension $F$ of $k_{v}$ in $\mathbb{C}_{v}$ as above.
\begin{itemize}
    \item[(1)] It can be seen that a model $\CMcal{M}$ of $M$ is good if and only if there exists a sufficiently large integer $n$ such that
    \[
    (t-\bar{\theta})^n \overline{\CMcal{M}} \subset \tau \overline{\CMcal{M}}.
    \]
    To see this, let $\{m_1,...,m_r\}$ be an $O_F[t]$-base for $\CMcal{M}$. There exists $U \in \operatorname{Mat}_r(O_F[t])$ so that 
    \[
    \begin{pmatrix}
        \tau m_1 \\ 
        \vdots \\
        \tau m_r
    \end{pmatrix}
    = U \cdot \begin{pmatrix}
        m_1 \\ \vdots \\ m_r
    \end{pmatrix}.
    \]
    The condition~\eqref{eqn: *-con} implies that there exists a sufficiently large integer $n$ and $c \in O_F$ such that $\det(U) = c(t-\theta)^n$.
    Then $\tau: \overline{\CMcal{M}}\rightarrow \overline{\CMcal{M}}$ is injective if and only if $c \in O_F^{\times}$, which is equivalent to $(t-\theta)^n\overline{\CMcal{M}} \subset \tau \overline{\CMcal{M}}$ for $n\gg 0$.
    \item[(2)] Suppose $M$ has good reduction.
    Then $M$ has a unique good model up to a unique isomorphism, see \cite[Proposition 2.13~(i)]{Gar03}.
    \item[(3)] Given a morphism $f: M_1\rightarrow M_2$ of $t$-motives over $F$, suppose $M_1$ and $M_2$ both have good reduction (with good models $\CMcal{M}_1$ and $\CMcal{M}_2$, respectively). Then $f$ can be `extended' to an $O_F[t,\tau]$-module homomorphism $f_0:\CMcal{M}_1\rightarrow \CMcal{M}_2$, see \cite[Proposition 2.13~(ii)]{Gar03}.
\end{itemize} 
\end{rem}

Let $L_w$ be the closure of $L$ in $\mathbb{C}_{v}$, regarding as the completion of $L$ at $w$.
We say that a $t$-motive $M$ over $L$ \emph{has good reduction at $w$} if $M_{/L_w}$ has a good model.

Keep the above notation. For each positive integer $\ell$,

\[
\frac{L\cdot \tau^{\ell} M_{/L}}{(t-\theta)\big(L\cdot \tau^{\ell} M_{/L}\big)} \overset{\sim}{\longrightarrow}

It suffices to prove the case when $L=F$.
The statement clearly holds when $\ell=1$. Suppose $\ell>1$. 
From the definition of $t$-motives one has that
\[
(t-\theta)^n M \subset F\cdot (\tau M) \quad \text{ for sufficiently large integer $n$.}
\]
Multiplying $\tau^{i}$, $1\leq i \leq \ell-1$ on both sides, we get
\[
\prod_{i=1}^{\ell-1}(t-\theta^{q^i})^n \cdot (F\cdot \tau M) \subset F \cdot (\tau^{\ell} M).
\]
Take $a,b \in F[t]$ with 
\[
a(t-\theta)^n + b\prod_{i=1}^{\ell-1}(t-\theta^{q^i})^n = 1.

<!-- defn-5: body_md -->
\label{defn: CM type}
Let $J_{\underline{K}}$ be the set of $\mathbb{F}_q$-algebra embeddings $\xi:\underline{K}\rightarrow \bar{k}$ with $\xi(t) = \theta$,
and $I_{\underline{K}}$ be the free abelian group generated by $J_{\underline{K}}$.
A \emph{CM type of $\underline{K}$} is a sum $\xi_1+\cdots + \xi_d \in I_{\underline{K}}$ satisfying $d = [\underline{K}^+:\underline{k}]$ and $J_{\underline{K}^+} = \big\{\xi_1\big|_{\underline{K}^+},...,\xi_d\big|_{\underline{K}^+}\big\}$. We call $\Xi \in I_{\underline{K}}$ a \emph{generalized CM type of $\underline{K}$} if $\Xi = \sum_{i=1}^w \Xi_i$, where $\Xi_i$ is a CM type of $\underline{K}$ for $1\leq i \leq w$, and $w$ is called the \emph{weight of $\Xi$} and denoted by $\operatorname{wt}(\Xi)$.

<!-- defn-6: body_md -->
\label{defn: CM t-motive}
(cf.~\cite[Proposition 4.3.4]{BCPW22})
Let $\underline{K}$ be a CM field over $\underline{k}$ and $\Xi$ be a generalized CM type of $\underline{K}$. A CM $t$-motive with CM type $(\underline{K},\Xi)$ over $\bar{k}$ is a $t$-motive $M$ of rank $[\underline{K}:\underline{k}]$ over $\bar{k}$ satisfying:
\begin{itemize}
    \item[(1)] There exists an $\underline{A}$-algebra homomorphism $\underline{O}_{\underline{K}}\hookrightarrow \operatorname{End}(M)$, where $\underline{O}_{\underline{K}}$ is the integral closure of $\underline{A}$ in $\underline{K}$.
    \item[(2)] Write $\Xi = \sum_{\xi \in J_{\underline{K}}} m_{\xi} \xi \in I_{\underline{K}}$. Regarding $M$ as a projective module of rank one over $\bar{k}\otimes_{\mathbb{F}_q}\underline{O}_{\underline{K}}$, we have
    \[
    \frac{M}{\tau M} \cong \prod_{\xi \in J_{\underline{K}}}\frac{\bar{k}\otimes_{\mathbb{F}_q}\underline{O}_{\underline{K}}}{\mathfrak{P}_{\xi}^{m_{\xi}}},
    \]
    where $\mathfrak{P}_{\xi}$ is the kernel ideal of the $\bar{k}$-algebra homomorphism
    $\bar{k}\otimes_{\mathbb{F}_q} \underline{O}_{\underline{K}}\rightarrow \bar{k}$ sending $c\otimes \underline{\alpha}$ to $c \cdot \xi(\underline{\alpha})$ for every $c \in \bar{k}$ and $\underline{\alpha} \in \underline{O}_{\underline{K}}$.
    \item[(3)] $M$ is pure.
\end{itemize}

It is also natural to ask whether the monomial relations of $v$-adic geometric special gamma values arised from the standard functional equations in Proposition~\ref{prop: FE}, together with the additional relation coming from the Gross--Koblitz--Thakur formula stated in Theorem~\ref{thm: G-K-T-formula-geo}, explain all the algebraic relations among these values in question.

Recall that given $\underline{\mathfrak{n}} \in \underline{A}_+$ with $\deg \underline{\mathfrak{n}}>0$, $\mathscr{A}_{\underline{\mathfrak{n}}}^{\operatorname{geo}}$ is the free abelian group generated by elements in $\mathfrak{n}^{-1}A$ (where $\mathfrak{n} = \underline{\mathfrak{n}}(\theta)=\iota(\underline{\mathfrak{n}})$).
Let $\mathscr{A}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}\mathrel{\vcenter{\baselineskip0.5ex \lineskiplimit0pt
                     \hbox{\scriptsize.}\hbox{\scriptsize.}}} 
                     = \mathscr{A}_{\underline{\mathfrak{n}}}^{\operatorname{geo}}\otimes_{\mathbb{Z}}\mathbb{Q}$ and $\mathscr{D}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ be the subspace of $\mathscr{A}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ generated by
\[
[x+a]-[x], \quad \forall\  x \in \mathfrak{n}^{-1}A,\ a \in A,
\]
and
\[
[\mathfrak{m} x] - \sum_{a \in A,~\deg a<\deg \mathfrak{m}} [x+\frac{a}{\mathfrak{m}}], \quad \forall \mathfrak{m} \in A_+ \text{ with } \mathfrak{m} \mid \mathfrak{n}.
\]
The quotient $\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}\mathrel{\vcenter{\baselineskip0.5ex \lineskiplimit0pt
                     \hbox{\scriptsize.}\hbox{\scriptsize.}}} 
                     = \mathscr{A}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}/\CMcal{D}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ is called the \emph{universal ordinary distribution} on $\mathfrak{n}^{-1}A/A$.
Suppose $v \nmid \mathfrak{n}$. Let $\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ be the subspace of $\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ generated by the `multiplication' relation
\[
\sum_{\epsilon \in \mathbb{F}_q^{\times}} [\epsilon x], \quad \forall x \in \mathfrak{n}^{-1}A,
\]
and let $\mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ be the subspace of $\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ generated by the following `Gross--Koblitz--Thakur' relation:
\[
\sum_{i=0}^{\ell-1} [v^ix], \quad \forall x \in \mathfrak{n}^{-1}A, 
\quad \text{where $\ell$ is the order of $\bar{v} \mathrel{\vcenter{\baselineskip0.5ex \lineskiplimit0pt
                     \hbox{\scriptsize.}\hbox{\scriptsize.}}} 
                     = (v \bmod \mathfrak{n}) \in (A/\mathfrak{n})^{\times}$.}
\]
Then:
\begin{lem}\label{lem: dim-U}
Keep the notation as above.
Let $H_v$ be the subgroup of $(A/\mathfrak{n})^{\times}$ generated by $\bar{v}$ and $\mathbb{F}_q^{\times}$.
We have
\[
\dim_{\mathbb{Q}}\left(\frac{\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}}{\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}+\mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}}\right) =    
\Big(1- \frac{1}{q-1}- \frac{1}{\ell} + \frac{1}{\#H_v}\Big) \cdot \#\Big(\frac{A}{\mathfrak{n}}\Big)^{\times}.
\]
\end{lem}

\begin{proof}
Observe that the action of $(A/\mathfrak{n})^{\times}$ on $\mathscr{A}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ given in \eqref{eqn: a-act} induces an action on $\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$, and it is known that (see \cite[Theorem 6.1.3]{ABP04}) 
\begin{equation}\label{eqn: dim-U}
\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}} \cong \mathbb{Q}[(A/\mathfrak{n})^{\times}] \quad \text{ as  $\mathbb{Q}[(A/\mathfrak{n})^{\times}]$-modules.}
\end{equation}
In particular, one has $\dim_{\mathbb{Q}} \mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}} = \#(A/\mathfrak{n})^{\times}$.
Moreover,  $\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ coincides with the subspace of $\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ consisting of elements fixed by every $\epsilon \in \mathbb{F}_q^{\times}$, and  $\mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ equals to the subspace of $\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}$ consisting of elements fixed by $\bar{v} \in (A/\mathfrak{n})^{\times}$.
Hence
\[
\dim_{\mathbb{Q}}(\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}) = \frac{1}{q-1}\cdot \#\left(\frac{A}{\mathfrak{n}}\right)^{\times},\quad
\dim_{\mathbb{Q}}(\mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}) = \frac{1}{\ell}\cdot \#\left(\frac{A}{\mathfrak{n}}\right)^{\times},
\]
and
\[\dim_{\mathbb{Q}}(\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}\cap \mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}) = \frac{1}{\#H_v}\cdot \#\left(\frac{A}{\mathfrak{n}}\right)^{\times}. 
\]
Therefore
\begin{align*}
\dim_{\mathbb{Q}}\left(\frac{\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}}{\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}+\mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}}\right) &=  
\dim_{\mathbb{Q}}(\mathscr{U}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}})-
\left(\dim_{\mathbb{Q}}(\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}})+\dim_{\mathbb{Q}}(\mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}})
-\dim_{\mathbb{Q}}(\mathscr{V}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}}\cap \mathscr{W}_{\underline{\mathfrak{n}},\mathbb{Q}}^{\operatorname{geo}})\right)\\
&= \Big(1- \frac{1}{q-1}- \frac{1}{\ell} + \frac{1}{\#H_v}\Big) \cdot \#\Big(\frac{A}{\mathfrak{n}}\Big)^{\times}.
\end{align*}
\end{proof}
