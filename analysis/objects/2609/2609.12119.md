<!-- definition-1: body_md -->
The iteration of SGD is given by
    \begin{equation}
        \mathbf{x}_{t+1}=\mathbf{x}_t-\alpha_t \mathbf{g}_t,
    \end{equation}
    where $\mathbf{g}_t=\nabla f(\mathbf{x}_t;\xi_t)$ is the stochastic gradient at $\mathbf{x}_t$ with a random process $\xi_t$ and $\alpha_t$ is a step size. Throughout this paper, we assume the stochastic gradient is unbiased and denote $\mathbb{E}_t[\mathbf{g}_t]=\nabla f(\mathbf{x}_t)$ as the true gradient.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:DP_SGD}
    SGD-CN is a modification of SGD, where gradients are clipped and noise is added to the clipped gradients.
    \begin{equation}\label{eq:DP_SGD}
        \mathbf{x}_{t+1}=\mathbf{x}_t-\alpha_t \mathbf{g}_t^{CN},
    \end{equation}
    where the clipped noisy stochastic gradient $\mathbf{g}_t^{CN}$ is given by
    \begin{equation}
        \mathbf{g}_t^{CN}=\text{clip}_q(\nabla f(\mathbf{x}_t;\xi_t))+q\zeta_t,
    \end{equation}
    where the $\text{clip}_q$ function is defined for $q>0$ by
    \begin{equation}
        \text{clip}_q(\nabla f(\mathbf{x}_t;\xi_t))=\min\left(1,\frac{q}{\|\nabla f(\mathbf{x}_t;\xi_t)\|}\right)\nabla f(\mathbf{x}_t;\xi_t),
    \end{equation}
    where $\zeta_t\sim\mathcal{N}(0,\sigma_{CN}^2I)$ is independent zero-mean Gaussian noise with variance $\sigma_{CN}^2$
<!-- end definition-2 -->
<!-- remark-1: body_md -->
SGD-CN can be adapted to other stochastic gradient methods such as stochastic heavy ball and stochastic Nesterov accelerated gradient.
    In the differential privacy (DP) literature, the parameters $q$ and $\sigma_{CN}$ are chosen to satisfy $(\epsilon,\delta)$-DP \cite{dwork2006calibrating}, but our analysis does not rely on this connection.
<!-- end remark-1 -->
<!-- assumption-1: body_md -->
\label{as:L_smoothness}
    $f$ is bounded from below by $f^*:=\min_{\mathbf{x}\in\mathbb{R}^d}f(\mathbf{x})$ and its gradient $\nabla f$ is $L$-Lipschitz i.e. 
    \begin{equation}
        \|\nabla f(\mathbf{x})-\nabla f(\mathbf{y})\|\leq L\|\mathbf{x}-\mathbf{y}\|,
    \end{equation}
    for all $\mathbf{x},\mathbf{y}\in\mathbb{R}^d$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{as:mu_strongly_convex}
    There exists a positive constant $\mu>0$ such that
    \begin{equation}
        f(\mathbf{y})\geq f(\mathbf{x})+\langle\nabla f(\mathbf{x}),\mathbf{y}-\mathbf{x}\rangle+\frac{\mu}{2}\|\mathbf{y}-\mathbf{x}\|^2,
    \end{equation}
    for all $\mathbf{x},\mathbf{y}\in\mathbb{R}^d$
    A consequence of $f$ being $\mu$-strongly convex is that
    \begin{equation}\label{eq:strong_convex2}
        \frac{1}{2\mu}\|\nabla f(\mathbf{x})\|^2\geq f(\mathbf{x})-f^*.
    \end{equation}
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{as:bounded_noise}
There exists a constant $\sigma_g\geq 0$ such that
\begin{equation}
    \left\|
    \nabla f(\mathbf{x};\xi)-\nabla f(\mathbf{x})
    \right\|
    \leq \sigma_g
    \quad \text{a.s.}
\end{equation}
for every $\mathbf{x}\in\mathbb{R}^d$.
<!-- end assumption-3 -->
<!-- remark-2: body_md -->
Assumption~\ref{as:bounded_noise} has been extensively studied in \cite{koloskova2023revisiting} and provides a theoretical lower bound on the clipping threshold.
<!-- end remark-2 -->
<!-- proposition-1: body_md -->
\label{prop:1}
    Let $\{X_t\}$, $\{Y_t\}$, and $\{Z_t\}$ be three sequences of random variables that are adapted to a filtration $\{\mathcal{F}_t\}$. Let $\{\gamma_t\}$ be a sequence of nonnegative real numbers such that $\prod_{t=1}^{\infty} (1 + \gamma_t) < \infty$. Suppose that the following conditions hold:
    \begin{enumerate}
        \item $X_t, Y_t, Z_t$ are nonnegative for all $t \geq 1$.
        \item $\mathbb{E}[Y_{t+1} | \mathcal{F}_t] \leq (1 + \gamma_t) Y_t - X_t + Z_t$ for all $t \geq 1$.
        \item $\sum_{t=1}^{\infty} Z_t < \infty$ holds a.s.
    \end{enumerate}
    Then, we have
    \begin{equation}
        \sum_{t=1}^{\infty} X_t < \infty \quad \text{a.s.,}
    \end{equation}
    and $Y_t$ converges a.s.
<!-- end proposition-1 -->
<!-- lemma-1: body_md -->
\label{lem:jy}
Let $\{X_t\}$ be a sequence of nonnegative real numbers and $\{\alpha_t\}$ be a decreasing sequence of positive real numbers such that the following conditions hold:
\begin{equation}
    \sum_{t=1}^{\infty} \alpha_t X_t < \infty, \quad \sum_{t=1}^{\infty} \frac{\alpha_t}{\sum_{i=1}^{t-1} \alpha_i} = \infty.
\end{equation}
Then,
\begin{equation}
\min_{1 \leq i \leq t} X_i = o \left( \frac{1}{\sum_{i=1}^{t-1} \alpha_i} \right).
\end{equation}

<!-- proposition-2: body_md -->
\label{prop:clipped_alignment}
    Suppose Assumption~\ref{as:bounded_noise} holds and define
    \begin{equation}
        \Psi_q(r):=\min\{r^2,qr\},
        \quad r\geq 0.
        \label{eq:Psi_def}
    \end{equation}
    Then, for every $\mathbf{x}\in\mathbb{R}^d$,
    \begin{align}
        &\left\langle
        \nabla f(\mathbf{x}),
        \mathbb{E}_t\left[
            \operatorname{clip}_q(\nabla f(\mathbf{x};\xi))
            | \mathbf{x}
        \right]
        \right\rangle \notag\\
        &\quad\geq
        \|\nabla f(\mathbf{x})\|^2
        -
        \|\nabla f(\mathbf{x})\|
        \left(
            \|\nabla f(\mathbf{x})\|+\sigma_g-q
        \right)_+ .
        \label{eq:general_clipped_alignment}
    \end{align}
    In particular, if $q>\sigma_g$, then
    \begin{equation}
        \left\langle
        \nabla f(\mathbf{x}),
        \mathbb{E}_t\left[
            \operatorname{clip}_q(\nabla f(\mathbf{x};\xi))
            | \mathbf{x}
        \right]
        \right\rangle
        \geq
        \left(1-\frac{\sigma_g}{q}\right)
        \Psi_q\left(\|\nabla f(\mathbf{x})\|\right).
        \label{eq:clipped_alignment}
    \end{equation}
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
Fix $\mathbf{x}\in\mathbb{R}^d$ and write
    \begin{equation*}
        \mathbf{g}:=\nabla f(\mathbf{x}),
        \quad
        \mathbf{g}_\xi:=\nabla f(\mathbf{x};\xi),
        \quad
        r:=\|\mathbf{g}\|.
    \end{equation*}
    Define $\mathbf{h}_\xi:=\operatorname{clip}_q(\mathbf{g}_\xi)$.
    By conditional unbiasedness,
    $\mathbb{E}_t[\mathbf{g}_\xi|\mathbf{x}]=\mathbf{g}$, and hence
    \begin{align}
        \left\langle
        \mathbf{g},
        \mathbb{E}_t[\mathbf{h}_\xi|\mathbf{x}]
        \right\rangle
        &=
        \left\langle
        \mathbf{g},
        \mathbb{E}_t[\mathbf{g}_\xi|\mathbf{x}]
        \right\rangle
        -
        \left\langle
        \mathbf{g},
        \mathbb{E}_t[\mathbf{g}_\xi-\mathbf{h}_\xi|\mathbf{x}]
        \right\rangle \notag\\
        &=
        r^2
        -
        \left\langle
        \mathbf{g},
        \mathbb{E}_t[\mathbf{g}_\xi-\mathbf{h}_\xi|\mathbf{x}]
        \right\rangle \notag\\
        &\geq
        r^2
        -
        r\,\mathbb{E}_t[
            \|\mathbf{g}_\xi-\mathbf{h}_\xi\|
            |\mathbf{x}
        ].
        \label{eq:clip_bias_step}
    \end{align}
    By the definition of the clipping operator,
    \begin{equation*}
        \|\mathbf{g}_\xi-\mathbf{h}_\xi\|
        =
        \left(\|\mathbf{g}_\xi\|-q\right)_+.
    \end{equation*}
    Assumption~\ref{as:bounded_noise} further gives
    \begin{equation*}
        \|\mathbf{g}_\xi\|
        \leq
        \|\mathbf{g}\|+\|\mathbf{g}_\xi-\mathbf{g}\|
        \leq
        r+\sigma_g
        \quad\text{a.s.}
    \end{equation*}
    Therefore,
    \begin{equation*}
        \|\mathbf{g}_\xi-\mathbf{h}_\xi\|
        \leq
        (r+\sigma_g-q)_+
        \quad\text{a.s.}
    \end{equation*}
    Substituting this into \eqref{eq:clip_bias_step} yields
    \begin{equation*}
        \left\langle
        \mathbf{g},
        \mathbb{E}_t[\mathbf{h}_\xi|\mathbf{x}]
        \right\rangle
        \geq
        r^2-r(r+\sigma_g-q)_+,
    \end{equation*}
    proving \eqref{eq:general_clipped_alignment}.
    Now suppose $q>\sigma_g$. If $r+\sigma_g\leq q$,
    then
    \begin{equation*}
        \left\langle
        \mathbf{g},
        \mathbb{E}_t[\mathbf{h}_\xi|\mathbf{x}]
        \right\rangle
        \geq r^2
        \geq
        \left(1-\frac{\sigma_g}{q}\right)\Psi_q(r).
    \end{equation*}
    Otherwise, $r+\sigma_g>q$, and
    \begin{equation*}
        r^2-r(r+\sigma_g-q)
        =
        r(q-\sigma_g).
    \end{equation*}
    If $r\leq q$, then
    \begin{equation*}
        r(q-\sigma_g)
        \geq
        \left(1-\frac{\sigma_g}{q}\right)r^2,
    \end{equation*}
    whereas if $r>q$, then
    \begin{equation*}
        r(q-\sigma_g)
        =
        \left(1-\frac{\sigma_g}{q}\right)qr.
    \end{equation*}
    Since $\Psi_q(r)=\min\{r^2,qr\}$, the desired result
    follows.

<!-- remark-3: body_md -->
\label{rem:clipping_threshold}
    Proposition~\ref{prop:clipped_alignment} holds for arbitrary
    $q>0$. However, the condition $q>\sigma_g$ guarantees that
    the expected clipped stochastic gradient is uniformly aligned
    with the true gradient. In particular, if $q\geq 2\sigma_g$, then
    \begin{equation}
        \left\langle
        \nabla f(\mathbf{x}),
        \mathbb{E}_t\left[
            \operatorname{clip}_q(\nabla f(\mathbf{x};\xi))
            |\mathbf{x}
        \right]
        \right\rangle
        \geq
        \frac{1}{2}
        \min\left\{
            \|\nabla f(\mathbf{x})\|^2,
            q\|\nabla f(\mathbf{x})\|
        \right\}.
    \end{equation}
    More generally, the alignment coefficient is
    $1-\sigma_g/q$, explicitly quantifying the effect of the
    clipping threshold relative to the stochastic-gradient noise.

<!-- theorem-1: body_md -->
\label{thm:SGD_CN}
    Consider the iterates of SGD-CN in
    \eqref{eq:DP_SGD}.
    Suppose Assumptions~\ref{as:L_smoothness} and
    \ref{as:bounded_noise} hold, let $q>\sigma_g$, and suppose
    that the step sizes satisfy \eqref{eq:stepsize_rate}.
    Then
    \begin{equation}
        \sum_{t=1}^{\infty}
        \alpha_t
        \Psi_q(\|\nabla f(\mathbf{x}_t)\|)
        <\infty
        \quad\text{a.s.}
        \label{eq:SGD_weighted_sum}
    \end{equation}
    and $f(\mathbf{x}_t)$ converges almost surely to a finite random
    variable.
    Furthermore,
    \begin{equation}
        \min_{1\leq i\leq t}
        \Psi_q(\|\nabla f(\mathbf{x}_i)\|)
        =
        o\left(
            \left(\sum_{i=1}^{t-1}\alpha_i\right)^{-1}
        \right)
        =
        o(t^{-\theta})
        \quad\text{a.s.}
        \label{eq:SGD_Psi_rate}
    \end{equation}
    and hence,
    \begin{equation}
        \min_{1\leq i\leq t}
        \|\nabla f(\mathbf{x}_i)\|
        =
        o(t^{-\theta/2})
        \quad\text{a.s.}
    \end{equation}
    If, in addition, Assumption~\ref{as:mu_strongly_convex}
    holds, then
    \begin{equation}
        \min_{1\leq i\leq t}
        \bigl(f(\mathbf{x}_i)-f^*\bigr)
        =
        o(t^{-\theta})
        \quad\text{a.s.}
        \label{eq:SGD_SC_rate}
    \end{equation}

<!-- proof-2: body_md -->
By $L$-smoothness,
    \begin{align*}
        f(\mathbf{x}_{t+1})
        \leq\;&
        f(\mathbf{x}_t)
        -
        \alpha_t
        \langle\nabla f(\mathbf{x}_t),\mathbf{g}_t^{CN}\rangle
        +
        \frac{L\alpha_t^2}{2}
        \|\mathbf{g}_t^{CN}\|^2.
                                   
    \end{align*}
    Taking conditional expectation and applying
    \eqref{eq:gCN_alignment} and
    \eqref{eq:gCN_second_moment} yields
    \begin{align*}
        \mathbb{E}_t[f(\mathbf{x}_{t+1})-f^*]
        \leq\;&
        f(\mathbf{x}_t)-f^*\nonumber\\
        &-
        \kappa_q\alpha_t
        \Psi_q(\|\nabla f(\mathbf{x}_t)\|)
        +
        \frac{LQ^2}{2}\alpha_t^2.
                                     
    \end{align*}

    We apply Proposition~\ref{prop:1} with
    \begin{equation*}
        Y_t:=f(\mathbf{x}_t)-f^*,
        \quad
        X_t:=
        \kappa_q\alpha_t
        \Psi_q(\|\nabla f(\mathbf{x}_t)\|),
        \quad
        Z_t:=\frac{LQ^2}{2}\alpha_t^2.
    \end{equation*}
    Since $Y_t,X_t,Z_t\geq0$ and
    $\sum_t\alpha_t^2<\infty$, Proposition~\ref{prop:1}
    implies that $f(\mathbf{x}_t)-f^*$ converges almost surely and
    \begin{equation*}
        \sum_{t=1}^{\infty}
        \alpha_t
        \Psi_q(\|\nabla f(\mathbf{x}_t)\|)
        <\infty
        \quad\text{a.s.},
    \end{equation*}
    proving \eqref{eq:SGD_weighted_sum}.
    We next apply Lemma~\ref{lem:jy} pathwise with
    \begin{equation*}
        X_t=\Psi_q(\|\nabla f(\mathbf{x}_t)\|).
    \end{equation*}
    The step-size conditions required by the lemma hold because
    $\{\alpha_t\}$ is positive and nonincreasing, and
    \begin{equation*}
        \alpha_t
        \sum_{i=1}^{t-1}\alpha_i
        =
        \Theta(t^{-1+2\theta}),
    \end{equation*}
    whose sum diverges for every $\theta>0$.
    Therefore,
    \begin{equation*}
        \min_{1\leq i\leq t}
        \Psi_q(\|\nabla f(\mathbf{x}_i)\|)
        =
        o\left(
            \left(\sum_{i=1}^{t-1}\alpha_i\right)^{-1}
        \right)
        =
        o(t^{-\theta}),
    \end{equation*}
    almost surely.

    To obtain the gradient-norm rate, define
    \begin{equation*}
        r_t^*
        :=
        \min_{1\leq i\leq t}
        \|\nabla f(\mathbf{x}_i)\|.
    \end{equation*}
    Since $\Psi_q$ is nondecreasing,
    \begin{equation*}
        \Psi_q(r_t^*)
        =
        \min_{1\leq i\leq t}
        \Psi_q(\|\nabla f(\mathbf{x}_i)\|)
        \to0.
    \end{equation*}
    Hence $r_t^*\to0$, and for all sufficiently large $t$,
    $r_t^*\leq q$. Therefore
    $\Psi_q(r_t^*)=(r_t^*)^2$, which together with
    \eqref{eq:SGD_Psi_rate} gives
    \begin{equation*}
        r_t^*=o(t^{-\theta/2})
        \quad\text{a.s.}
    \end{equation*}

    Finally, suppose $f$ is $\mu$-strongly convex.
    Then, as a result of Assumption \ref{as:mu_strongly_convex},
    \begin{align*}
        \Psi_q(\|\nabla f(\mathbf{x}_t)\|)
        &\geq
        \Psi_q\left(
            \sqrt{2\mu(f(\mathbf{x}_t)-f^*)}
        \right)
        \nonumber\\
        &=
        \min\left\{
            2\mu(f(\mathbf{x}_t)-f^*),
            q\sqrt{2\mu(f(\mathbf{x}_t)-f^*)}
        \right\}.
        \label{eq:SC_Psi}
    \end{align*}
    Applying Lemma~\ref{lem:jy} to the right-hand side shows that its best-iterate value is
    $o(t^{-\theta})$ almost surely.
    Since this quantity converges to zero, its first branch
    eventually applies to the best iterate, and hence
    \begin{equation*}
        \min_{1\leq i\leq t}
        (f(\mathbf{x}_i)-f^*)
        =
        o(t^{-\theta})
        \quad\text{a.s.}
    \end{equation*}
    This completes the proof.

    \end{align*}
    Since $\{\alpha_t'\}$ is bounded,
    \eqref{eq:last_previous} and \eqref{eq:last_x_weighted}
    imply
    \begin{equation}
        \sum_{t=1}^{\infty}
        \alpha_t'\|\nabla f(\mathbf{z}_t)\|^2
        <\infty.
        \label{eq:last_z_weighted}
    \end{equation}
    It remains to verify the oscillation condition in
    Lemma~\ref{lem:orabona}. Define
    \begin{equation*}
        \mathbf{h}_t
        :=
        \text{clip}_q(\nabla f(\mathbf{x}_t;\xi_t)),
        \quad
        \bar{\mathbf{h}}_t
        :=
        \mathbb{E}_t[\mathbf{h}_t],
        \quad
        \mathbf{u}_t
        :=
        \mathbf{h}_t-\bar{\mathbf{h}}_t.
        \label{eq:last_decomp}
    \end{equation*}
    We first note that
    \begin{equation}
        \|\bar{\mathbf{h}}_t\|
        \leq
        c_q\|\nabla f(\mathbf{x}_t)\|,
        \quad
        c_q:=\frac{q}{q-\sigma_g}.
        \label{eq:last_mean_clip_bound}
    \end{equation}
    Indeed, if $\|\nabla f(\mathbf{x}_t)\|\leq q-\sigma_g$, then
    Assumption~\ref{as:bounded_noise} implies that clipping is
    inactive almost surely and
    $\bar{\mathbf{h}}_t=\nabla f(\mathbf{x}_t)$. Otherwise,
    $\|\bar{\mathbf{h}}_t\|\leq q
    \leq c_q\|\nabla f(\mathbf{x}_t)\|$.
    Let
    \begin{equation*}
        b_t:=\|\nabla f(\mathbf{z}_t)\|,
        \quad
        c_\beta:=\frac{\beta}{1-\beta}.
    \end{equation*}
    By $L$-smoothness and \eqref{eq:last_zx},
    \begin{equation*}
        \|\bar{\mathbf{h}}_t\|
        \leq
        c_q b_t+c_qLc_\beta\|\mathbf{v}_t\|.
        \label{eq:last_mean_z}
    \end{equation*}
    Moreover,
    \begin{equation*}
        \mathbf{g}_t^{CN}
        =
        \bar{\mathbf{h}}_t+\mathbf{u}_t+q\zeta_t.
    \end{equation*}
    Therefore, for every $\tau\geq1$,
    \begin{align}
        |b_{t+\tau}-b_t|
        &\leq
        L\|\mathbf{z}_{t+\tau}-\mathbf{z}_t\|
        \nonumber\\
        &\leq
        Lc_q
        \sum_{i=t}^{t+\tau-1}\alpha_i'b_i
        +
        Lc_qLc_\beta
        \sum_{i=t}^{t+\tau-1}\alpha_i'\|\mathbf{v}_i\|
        \nonumber\\
        &\quad+
        L\left\|
            \sum_{i=t}^{t+\tau-1}
            \alpha_i'(\mathbf{u}_i+q\zeta_i)
        \right\|.
        \label{eq:last_oscillation}
    \end{align}
    Now define the augmented random vector
    \begin{equation}
        \mathbf{w}_t
        :=
        \begin{bmatrix}
            \mathbf{u}_t+q\zeta_t\\
            c_qLc_\beta\|\mathbf{v}_t\|
        \end{bmatrix}.
        \label{eq:last_augmented_noise}
    \end{equation}
    Since $\mathbf{u}_t$ is a martingale difference and
    $\mathbb{E}_t\|\mathbf{u}_t\|^2\leq q^2$, while $\zeta_t$ is an
    independent zero-mean Gaussian vector,
    \begin{equation*}
        \sum_{t=1}^{\infty}
        \alpha_t'(\mathbf{u}_t+q\zeta_t)
    \end{equation*}
    converges almost surely by the martingale convergence
    theorem \cite{williams1991probability}, since $\sum_t(\alpha_t')^2<\infty$.
    Furthermore, by Cauchy--Schwarz and
    \eqref{eq:last_previous},
    \begin{align*}
        \sum_{t=1}^{\infty}\alpha_t'\|\mathbf{v}_t\|
        &\leq
        \left(
            \sum_{t=1}^{\infty}(\alpha_t')^2
        \right)^{1/2}
        \left(
            \sum_{t=1}^{\infty}\|\mathbf{v}_t\|^2
        \right)^{1/2}
        <\infty.
        \label{eq:last_v_abs}
    \end{align*}
    Hence
    \begin{equation*}
        \sum_{t=1}^{\infty}\alpha_t'\mathbf{w}_t
    \end{equation*}
    converges almost surely.
    Finally, using
    $a+b\leq\sqrt{2(a^2+b^2)}$ in
    \eqref{eq:last_oscillation}, there exists a constant
    $C>0$ such that
    \begin{equation}
        |b_{t+\tau}-b_t|
        \leq
        C\left(
            \sum_{i=t}^{t+\tau-1}\alpha_i'b_i
            +
            \left\|
                \sum_{i=t}^{t+\tau-1}\alpha_i'\mathbf{w}_i
            \right\|
        \right).
        \label{eq:last_orabona}
    \end{equation}
    Together with \eqref{eq:last_z_weighted},
    $\sum_t\alpha_t'=\infty$, and
    Lemma~\ref{lem:orabona} with $p=2$, this yields
    \begin{equation*}
        \|\nabla f(\mathbf{z}_t)\|\to0
        \quad\text{a.s.}
    \end{equation*}
    Finally, $\mathbf{v}_t\to0$ and \eqref{eq:last_zx} imply
    $\|\mathbf{z}_t-\mathbf{x}_t\|\to0$, and therefore
    \begin{equation*}
        \|\nabla f(\mathbf{x}_t)\|
        \leq
        \|\nabla f(\mathbf{z}_t)\|
        +
        L\|\mathbf{x}_t-\mathbf{z}_t\|
        \to0
        \quad\text{a.s.}
    \end{equation*}
    Setting $\beta=0$ recovers the SGD-CN result. The proof of convergence for SNAG-CN is similar, following \eqref{eq:SNAG_velocity_expansion}.
