A complete table of notation is provided in \cref{tab:notation,tab:notation:cont}. For a matrix $\mathbf{A} \in \mathbb{C}^{m \times n}$, let $\operatorname{Re} \mathbf{A} \in \mathbb{R}^{m \times n}$ and $\operatorname{Im} \mathbf{A} \in \mathbb{R}^{m \times n}$ be the real and imaginary parts of $\mathbf{A}$. The absolute value of a complex number is defined as $\lverta\rvert = \sqrt{(\operatorname{Re} a)^2 + (\operatorname{Im} a)^2}$. We use the notation $\overline{\mathbf{A}}$ to represent the complex conjugate of $\mathbf{A}$. An eigenvalue of a matrix is denoted as $\nu$. We define the $\ell_1$, $\ell_2$, and $\ell_{\infty}$ norms of a matrix $\mathbf{A} = (a_{ij})$ as $\left\lVert\mathbf{A}\right\rVert_1 = \sum_{i,j} \lverta_{ij}\rvert$, $\lVert\mathbf{A}\rVert_2 = (\sum_{i,j} \lverta_{ij}\rvert^2)^{1/2}$, and $\left\lVert\mathbf{A}\right\rVert_{\infty} = \max_{i,j} |a_{ij}|$, respectively. The $\ell_0$ ``norm'', $\lVert\mathbf{A}\rVert_0$, is defined as the number of nonzero elements of $\mathbf{A}$. For $q \geq 1$, we denote the $L_q$-norm of a random variable $X$ as $\left\lVertX\right\rVert_{L_q} = ( \mathbb{E}\lvertX\rvert^q )^{1/q}$ where $\mathbb{E}$ is the expected value. The Kronecker product between two matrices is denoted as $\mathbf{A} \otimes \mathbf{B}$. We also use $\operatorname{vec}(\mathbf{A})$ to denote the operator that stacks the columns of $\mathbf{A}$. The notation $x_t \asymp y_t$ indicates that $x_t$ and $y_t$ are asymptotically of the same order while $x \lesssim y$ denotes that $x \leq Cy$ for some constant $C > 0$ and $x \lesssim_{\alpha} y$ indicates that $x \leq C_{\alpha} y$ where $C_{\alpha}$ is a constant that depends on the problem parameter $\alpha$. The $p \times p$ identity matrix is denoted as $\mathbf{I}_p$. The gradient of $L_D$ with respect to the parameters $\Delta$ is denoted as $\nabla L_D(\Delta; \hat\Sigma_1, \hat\Sigma_2)$; arguments after the semicolon are treated as fixed or conditioning quantities. \label{lem:BE:Dtrace:abbr}

\begin{gather}

\begin{equation} \label{cond:lowdim}

\end{equation*}

\item For $\omega \in \{0, \pi\}$,
\begin{equation*}
\sqrt{\frac{T}{B}} \begin{pmatrix} \hat{F}_{ij} - f^*_{ij} \\ \hat{F}_{kl} - f^*_{kl} \end{pmatrix} \xrightarrow{d} \mathcal{N}\left(0, \begin{bmatrix} f^*_{ii} f^*_{jj} + f^{*2}_{ij} & f^*_{ik} f^*_{lj} + f^*_{il} f^*_{kj} \\ f^*_{ki} f^*_{jl} + f^*_{il} f^*_{kj} & f^*_{kk} f^*_{ll} + f^{*2}_{kl} \end{bmatrix}\right).
\end{equation*}
\end{enumerate}

<!-- proposition-1: body_md -->
\label{prop:asympcov}
Suppose $\mathbf{Z} \in \mathbb{C}^{p \times p}$ is a complex random matrix such that for any $i, j, k, l \in [p]$,
\begin{gather*}
\operatorname{Cov}\Big(Z_{i, j}, Z_{k, l}\Big) = \mathbb{E}\Big[\left(Z_{i, j} - \mathbb{E}{Z_{i, j}}\right) \overline{\left(Z_{k, l} - \mathbb{E}{Z_{k, l}}\right)}\Big] = f^*_{i, k} f^*_{l, j}, \\
\operatorname{Cov}\Big(Z_{i, j}, \overline{Z_{k, l}}\Big) = \mathbb{E}\Big[\left(Z_{i, j} - \mathbb{E}{Z_{i, j}}\right) \left(Z_{k, l} - \mathbb{E}{Z_{k, l}}\right)\Big] = f^*_{i, l} f^*_{k, j},
\end{gather*}
i.e., $\mathbf{Z}$ has the same covariance and pseudo-covariance structure as the limiting covariance and pseudo-covariance in the first part of \cref{thm:asympnormal}. Then, for any $\mathbf{w} \in \mathbb{R}^{4p^2}$, $\sigma^2 = \operatorname{Var}\left(\mathbf{w}^\top \mathbf{P}^\top \left[\begin{smallmatrix} \operatorname{vec} \operatorname{Re} \mathbf{Z} \\ \operatorname{vec} \operatorname{Im} \mathbf{Z} \end{smallmatrix}\right]\right) = \left(\mathbf{P} \mathbf{w}\right)^\top \Psi \mathbf{P} \mathbf{w}$, where
\begin{equation} \label{eq:VarSh}
\begin{aligned}
\Psi
&= \operatorname{Var}\left(\begin{bmatrix} \operatorname{vec} \operatorname{Re} \mathbf{Z} \\ \operatorname{vec} \operatorname{Im} \mathbf{Z} \end{bmatrix}\right) \\
&= \frac 1 2 \begin{bmatrix}
\left(\mathbf{I}_{p^2} + \mathbf{K}\right) \left(\operatorname{Re} \mathbf{F}^* \otimes \operatorname{Re} \mathbf{F}^*+\operatorname{Im} \mathbf{F}^* \otimes \operatorname{Im} \mathbf{F}^*\right) &
\left(\mathbf{I}_{p^2} + \mathbf{K}\right) \left(\operatorname{Im} \mathbf{F}^* \otimes \operatorname{Re} \mathbf{F}^* - \operatorname{Re} \mathbf{F}^* \otimes \operatorname{Im} \mathbf{F}^*\right) \\
\left(\mathbf{I}_{p^2} - \mathbf{K}\right) \left(\operatorname{Re} \mathbf{F}^* \otimes \operatorname{Im} \mathbf{F}^* - \operatorname{Im} \mathbf{F}^* \otimes \operatorname{Re} \mathbf{F}^*\right) &
\left(\mathbf{I}_{p^2} - \mathbf{K}\right) \left(\operatorname{Re} \mathbf{F}^* \otimes \operatorname{Re} \mathbf{F}^* + \operatorname{Im} \mathbf{F}^* \otimes \operatorname{Im} \mathbf{F}^*\right)
\end{bmatrix}.
\end{aligned}
\end{equation}

\label{prop:asympcov:real}
Suppose $\mathbf{Z} \in \mathbb{R}^{p \times p}$ is a real random matrix such that for any $i, j, k, l \in [p]$,
\begin{equation*}
\operatorname{Cov}\big(Z_{i, j}, Z_{k, l}\big) = \mathbb{E}\Big[\left(Z_{i, j} - \mathbb{E}{Z_{i, j}}\right) \left(Z_{k, l} - \mathbb{E}{Z_{k, l}}\right)\Big] = f^*_{i, k} f^*_{l, j} + f^*_{i, l} f^*_{k, j},
\end{equation*}
i.e., $\mathbf{Z}$ has the same covariance structure as the limiting covariance in the second part of \cref{thm:asympnormal}. Then, for any $\mathbf{w} \in \mathbb{R}^{p^2}$, $\sigma^2 = \operatorname{Var}(\mathbf{w}^\top \operatorname{vec} \mathbf{Z}) = \mathbf{w}^\top \Psi \mathbf{w}$, where
\begin{equation} \label{eq:VarSh:real}
\Psi = \operatorname{Var}(\operatorname{vec} \mathbf{Z}) = (\mathbf{I}_{p^2} + \mathbf{K}) (\mathbf{F}^* \otimes \mathbf{F}^*).
\end{equation}

Then, the Berry-Esseen condition for the sum of projected Welch's estimators in \cref{cond:BE} holds for the following specifications of $N$, $\sigma^2$, and $\varepsilon_{\textnormal{BE}}$:

\item The asymptotic variance is $\sigma^2 = (N/N_1) \sigma^2_1 + (N/N_2) \sigma^2_2$;

\begin{multline*}
\varepsilon_{\textnormal{BE}} \lesssim_{\mathbf{w}^*_1, \mathbf{w}^*_2, \sigma, \rho_1, \rho_2, \alpha_1, \alpha_2} \frac{1}{\sqrt{N}} + d^{-1}
+ \lVertX_{1, \cdot}\rVert_{\psi_{\alpha_1}}^4 (\log d)^{1 + 2 \alpha_1} \sqrt{\frac{\log B_1}{B_1}} + \Phi_{1, 2}^2 \sqrt{\frac{T_1}{B_1^3}} \\
+ \lVertX_{2, \cdot}\rVert_{\psi_{\alpha_2}}^4 (\log d)^{1 + 2 \alpha_2} \sqrt{\frac{\log B_2}{B_2}} + \Phi_{2, 2}^2 \sqrt{\frac{T_2}{B_2^3}}.
\end{multline*}
\end{itemize}

\label{thm:BE:SDD}
Let $s_{\text{tot}} = \left\lVert\Delta^*\right\rVert_1 \left\lVert\mathbf{v}^*\right\rVert_1 + s_{\Delta^*} + s_{\mathbf{v}^*}$, and suppose $B_1, B_2 \geq s_{\text{tot}}$. Then, in the combined setting of \cref{lem:BE:Dtrace:abbr} and \cref{thm:BE:projWelch},
\begin{multline} \label{eq:BE:SDD}
\sup_z \left\lvert\mathbb{P}\left\{\sigma^{-1} \sqrt{N} \left(\tilde{\Delta}_0-\Delta^*_0\right) \leq z\right\} - \Phi(z)\right\rvert \lesssim_{\Sigma_1^*, \Sigma_2^*, \mathbf{w}^*_1, \mathbf{w}^*_2, \sigma, \rho_1, \rho_2, \alpha_1, \alpha_2} d^{-1} \\
+ \max\left\{\lVertX_{1, \cdot}\rVert_{\psi_{\alpha_1}}^2, \lVertX_{1, \cdot}\rVert_{\psi_{\alpha_1}}^4\right\} s_{\text{tot}} (\log d)^{2 + 4 \alpha_1} \sqrt{\frac{B_1}{T_1}} + \lVertX_{1, \cdot}\rVert_{\psi_{\alpha_1}}^4 (\log d)^{1 + 2 \alpha_1} \sqrt{\frac{\log B_1}{B_1}} + \max\left\{\Phi_{1, 2}^2, \Phi_{1, 2}^4\right\} \sqrt{\frac{T_1}{B_1^3}} \\
+ \max\left\{\lVertX_{2, \cdot}\rVert_{\psi_{\alpha_2}}^2, \lVertX_{2, \cdot}\rVert_{\psi_{\alpha_2}}^4\right\} s_{\text{tot}} (\log d)^{2 + 4 \alpha_2} \sqrt{\frac{B_2}{T_2}} + \lVertX_{2, \cdot}\rVert_{\psi_{\alpha_2}}^4 (\log d)^{1 + 2 \alpha_2} \sqrt{\frac{\log B_2}{B_2}} + \max\left\{\Phi_{2, 2}^2, \Phi_{2, 2}^4\right\} \sqrt{\frac{T_2}{B_2^3}}.
\end{multline}

\label{cor:BE:SDD:abbr}
In the setting of \cref{thm:BE:SDD}, suppose the window sizes are chosen as in \cref{eq:BE:optB}. Then, provided that
\begin{gather*}
\max\left\{\lVertX_{1, \cdot}\rVert_{\psi_{\alpha_1}}^2, \lVertX_{1, \cdot}\rVert_{\psi_{\alpha_1}}^4\right\} s_{\text{tot}} (\log d)^{\frac{3}{2} + 3 \alpha_1} = o\left(T_1^{\frac 1 4}\right), \\
\max\left\{\lVertX_{2, \cdot}\rVert_{\psi_{\alpha_2}}^2, \lVertX_{2, \cdot}\rVert_{\psi_{\alpha_2}}^4\right\} s_{\text{tot}} (\log d)^{\frac{3}{2} + 3 \alpha_2} = o\left(T_2^{\frac 1 4}\right),
\end{gather*}
we have
\begin{equation*}
\sup_z \left\lvert\mathbb{P}\left\{\sigma^{-1} \sqrt{N} \left(\tilde{\Delta}_0-\Delta^*_0\right) \leq z\right\} - \Phi(z)\right\rvert \to 0 \quad \text{as} \quad d, \, T_1, \, T_2 \to \infty.
\end{equation*}

<!-- fact-1: body_md -->
\label{fact:kronspec}
Let $\mathbf{A}$ and $\mathbf{B}$ be square matrices. If $\nu_{\mathbf{A}}$ is an eigenvalue of $\mathbf{A}$ and $\nu_{\mathbf{B}}$, an eigenvalue of $\mathbf{B}$, then $\nu_{\mathbf{A}} \nu_{\mathbf{B}}$ is an eigenvalue of $\mathbf{A} \otimes \mathbf{B}$.
<!-- end fact-1 -->
<!-- proof-1: body_md -->
Let $\mathbf{v}$ be an eigenvector of $\mathbf{A}$ associated with $\nu_{\mathbf{A}}$, and $\mathbf{w}$, an eigenvector of $\mathbf{B}$ associated with $\nu_{\mathbf{B}}$. Then,
\begin{equation*}
\left(\mathbf{A} \otimes \mathbf{B}\right) \left(\mathbf{v} \otimes \mathbf{w}\right) = \left(\mathbf{A} \mathbf{v}\right) \otimes \left(\mathbf{B} \mathbf{w}\right) = \left(\nu_{\mathbf{A}} \mathbf{v}\right) \otimes \left(\nu_{\mathbf{B}} \mathbf{w}\right) = \nu_{\mathbf{A}} \nu_{\mathbf{B}} \left(\mathbf{v} \otimes \mathbf{w}\right).
\end{equation*}
<!-- end proof-1 -->
<!-- fact-2: body_md -->
\label{fact:equal_evals}
Let $\mathbf{F} = \mathbf{A} + \imath \mathbf{B} \in \mathbb{C}^{p \times p}$ be a complex matrix. Let $\Sigma$ be the realification of $\mathbf{F}$: $\Sigma = \left[\begin{smallmatrix} \mathbf{A} & -\mathbf{B} \\ \mathbf{B} & \phantom{-}\mathbf{A} \end{smallmatrix}\right]$. If $\nu$ is an eigenvalue of $\mathbf{F}$, then $\nu$ and $\bar{\nu}$ are both eigenvalues of $\Sigma$.
<!-- end fact-2 -->
<!-- proof-2: body_md -->
Let $\mathbf{v}$ be an eigenvector of $\mathbf{F}$ associated with $\nu$: $\mathbf{F} \mathbf{v} = \mathbf{A} \mathbf{v} + \imath \mathbf{B} \mathbf{v} = \nu \mathbf{v}$. Then, $\bar{\mathbf{F}} \bar{\mathbf{v}} = \mathbf{A} \bar{\mathbf{v}} - \imath \mathbf{B} \bar{\mathbf{v}} = \bar{\nu} \bar{\mathbf{v}}$, and
\begin{gather*}
\Sigma \begin{bmatrix} \phantom{-\imath} \mathbf{v} \\ -\imath \mathbf{v} \end{bmatrix}
= \begin{bmatrix} \mathbf{A} & -\mathbf{B} \\ \mathbf{B} & \phantom{-}\mathbf{A} \end{bmatrix} \begin{bmatrix} \phantom{-\imath} \mathbf{v} \\ -\imath \mathbf{v} \end{bmatrix}
= \begin{bmatrix} \mathbf{A} \mathbf{v} + \imath \mathbf{B} \mathbf{v} \\ \mathbf{B} \mathbf{v} - \imath \mathbf{A} \mathbf{v} \end{bmatrix}
= \begin{bmatrix} \phantom{-\imath} \mathbf{F} \mathbf{v} \\ -\imath \mathbf{F} \mathbf{v} \end{bmatrix}
= \nu \begin{bmatrix} \phantom{-\imath} \mathbf{v} \\ -\imath \mathbf{v} \end{bmatrix}, \\
\Sigma \begin{bmatrix} \phantom{\imath} \bar{\mathbf{v}} \\ \imath \bar{\mathbf{v}} \end{bmatrix}
= \begin{bmatrix} \mathbf{A} & -\mathbf{B} \\ \mathbf{B} & \phantom{-}\mathbf{A} \end{bmatrix} \begin{bmatrix} \phantom{\imath} \bar{\mathbf{v}} \\ \imath \bar{\mathbf{v}} \end{bmatrix}
= \begin{bmatrix} \mathbf{A} \bar{\mathbf{v}} - \imath \mathbf{B} \bar{\mathbf{v}} \\ \mathbf{B} \bar{\mathbf{v}} + \imath \mathbf{A} \bar{\mathbf{v}} \end{bmatrix}
= \begin{bmatrix} \phantom{\imath} \bar{\mathbf{F}} \bar{\mathbf{v}} \\ \imath \bar{\mathbf{F}} \bar{\mathbf{v}} \end{bmatrix}
= \bar{\nu} \begin{bmatrix} \phantom{\imath} \bar{\mathbf{v}} \\ \imath \bar{\mathbf{v}} \end{bmatrix}.
\end{gather*}
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lem:RSC}
Let $\nu_1$ and $\nu_2$ be the minimum eigenvalues of $\Sigma_1^*$ and $\Sigma_2^*$, respectively. Suppose
\begin{equation*}
\left\lVert\hat\Sigma_1-\Sigma_1^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 1}, \quad \left\lVert\hat\Sigma_2-\Sigma_2^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 2},
\end{equation*}
where $\delta_{\textnormal{LD}, 1} = \delta_{\textnormal{LD}, 1}(d, T_1) > 0$ and $\delta_{\textnormal{LD}, 2} = \delta_{\textnormal{LD}, 2}(d, T_2) > 0$ satisfy
\begin{equation*}
2 \left(1+\kappa\right)^2 s \left(\left\lVert\Sigma_2^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 1} + \left\lVert\Sigma_1^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 2} + \delta_{\textnormal{LD}, 1} \delta_{\textnormal{LD}, 2}\right) \leq \nu_1 \nu_2
\end{equation*}
for some $0 \leq s \leq d$ and $\kappa > 0$. Then,
\begin{equation*}
\mathbf{v}^\top \hat{\mathbf{H}} \mathbf{v} \geq \frac{\nu_1 \nu_2}{2} \lVert\mathbf{v}\rVert_2^2 \quad \text{for all} \quad \mathbf{v} \in \mathcal{C}(s, \kappa).
\end{equation*}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
For any $\mathbf{v} \in \mathcal{C}(s, \kappa)$,
\begin{equation*}
\left\lVert\mathbf{v}\right\rVert_1 = \left\lVert\mathbf{v}_{\mathcal{S}}\right\rVert_1 + \left\lVert\mathbf{v}_{\mathcal{S}^c}\right\rVert_1 \leq \left(1+\kappa\right) \left\lVert\mathbf{v}_{\mathcal{S}}\right\rVert_1 \leq \left(1+\kappa\right) \sqrt{s} \left\lVert\mathbf{v}_{\mathcal{S}}\right\rVert_2 \leq \left(1+\kappa\right) \sqrt{s} \lVert\mathbf{v}\rVert_2.
\end{equation*}
By H\"{o}lder's inequality,
\begin{equation*}
\mathbf{v}^\top \hat{\mathbf{H}} \mathbf{v} = \mathbf{v}^\top \mathbf{H}^* \mathbf{v} + \mathbf{v}^\top \left(\hat{\mathbf{H}}-\mathbf{H}^*\right) \mathbf{v} \geq \mathbf{v}^\top \mathbf{H}^* \mathbf{v}  - \left\lVert\hat{\mathbf{H}}-\mathbf{H}^*\right\rVert_{\infty} \lVert\mathbf{v}\rVert_1^2.
\end{equation*}
By the definition of $\mathbf{H}^*$, \cref{fact:kronspec}, and in the case of $\omega \notin \{0, \pi\}$, \cref{fact:equal_evals},
\begin{equation*}
\mathbf{v}^\top \mathbf{H}^* \mathbf{v} \geq \nu_1 \nu_2 \lVert\mathbf{v}\rVert^2.
\end{equation*}
Thus,
\begin{equation*}
\mathbf{v}^\top \hat{\mathbf{H}} \mathbf{v} \geq \left(\nu_1 \nu_2 - \left(1+\kappa\right)^2 s \left\lVert\hat{\mathbf{H}}-\mathbf{H}^*\right\rVert_{\infty}\right) \lVert\mathbf{v}\rVert_2^2.
\end{equation*}
By \cref{eq:Hessdiff}, under the condition of the lemma,
\begin{equation*}
\left(1+\kappa\right)^2 s \left\lVert\hat{\mathbf{H}}-\mathbf{H}^*\right\rVert_{\infty} \leq \left(1+\kappa\right)^2 s \left(\left\lVert\Sigma_2^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 1} + \left\lVert\Sigma_1^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 2} + \delta_{\textnormal{LD}, 1} \delta_{\textnormal{LD}, 2}\right) \leq \frac{\nu_1 \nu_2}{2}.
\end{equation*}
Therefore,
\begin{equation*}
\mathbf{v}^\top \hat{\mathbf{H}} \mathbf{v} \geq \frac{\nu_1 \nu_2}{2} \lVert\mathbf{v}\rVert_2^2.
\end{equation*}
<!-- end proof-3 -->
<!-- lemma-3: body_md -->
\label{lem:Dh}
Let $\nu_1$ and $\nu_2$ be the minimum eigenvalues of $\Sigma_1^*$ and $\Sigma_2^*$, respectively. Let $s_{\Delta^*} = \lVert\Delta^*\rVert_0$. Suppose
\begin{equation*}
\left\lVert\hat\Sigma_1-\Sigma_1^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 1}, \quad \left\lVert\hat\Sigma_2-\Sigma_2^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 2},
\end{equation*}
where $\delta_{\textnormal{LD}, 1} = \delta_{\textnormal{LD}, 1}(d, T_1) > 0$ and $\delta_{\textnormal{LD}, 2} = \delta_{\textnormal{LD}, 2}(d, T_2) > 0$ satisfy
\begin{equation*}
\delta_{\textnormal{LD}, H} = \left\lVert\Sigma_2^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 1} + \left\lVert\Sigma_1^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 2} + \delta_{\textnormal{LD}, 1} \delta_{\textnormal{LD}, 2} \leq \frac{\nu_1 \nu_2}{32 s_{\Delta^*}}.
\end{equation*}
Suppose $\lambda_1$ satisfies
\begin{equation*}
\lambda_1 \geq 2 \left(\left\lVert\Delta^*\right\rVert_1 \delta_{\textnormal{LD}, H} + \delta_{\textnormal{LD}, 1} + \delta_{\textnormal{LD}, 2}\right).
\end{equation*}
Then,
\begin{equation*}
\left\lVert\hat\Delta-\Delta^*\right\rVert_1 \lesssim \frac{s_{\Delta^*} \lambda_1}{\nu_1 \nu_2}.
\end{equation*}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
The lemma can be proved by the standard analysis for the LASSO, e.g., Corollary 1 of \citet{negahban2012unified}.
<!-- end proof-4 -->
<!-- lemma-4: body_md -->
\label{lem:dbdh}
Let $\nu_1$ and $\nu_2$ be the minimum eigenvalues of $\Sigma_1^*$ and $\Sigma_2^*$, respectively. Let $s_{\mathbf{v}^*} = \lVert\mathbf{v}^*\rVert_0$. Suppose
\begin{equation*}
\left\lVert\hat\Sigma_1-\Sigma_1^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 1}, \quad \left\lVert\hat\Sigma_2-\Sigma_2^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 2},
\end{equation*}
where $\delta_{\textnormal{LD}, 1} = \delta_{\textnormal{LD}, 1}(d, T_1) > 0$ and $\delta_{\textnormal{LD}, 2} = \delta_{\textnormal{LD}, 2}(d, T_2) > 0$ satisfy
\begin{equation*}
\delta_{\textnormal{LD}, H} = \left\lVert\Sigma_2^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 1} + \left\lVert\Sigma_1^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 2} + \delta_{\textnormal{LD}, 1} \delta_{\textnormal{LD}, 2} \leq \frac{\nu_1 \nu_2}{8 s_{\mathbf{v}^*}}.
\end{equation*}
Suppose $\lambda_2$ satisfies
\begin{equation*}
\lambda_2 \geq \left\lVert\mathbf{v}^*\right\rVert_1 \delta_{\textnormal{LD}, H}.
\end{equation*}
Then,
\begin{equation*}
\left\lVert\hat{\mathbf{v}}-\mathbf{v}^*\right\rVert_1 \lesssim \frac{s_{\mathbf{v}^*} \lambda_2}{\nu_1 \nu_2}.
\end{equation*}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
The proof follows the standard analysis for Dantzig selectors (cf., \citet{candes2007dantzig,bickel2009simultaneous}). On the event of the hypothesis, $\mathbf{v}^*$ satisfies the Dantzig constraint. Thus, on one hand,
\begin{equation} \label{eq:Dantzig:graddiff}
\left\lVert\hat{\mathbf{H}} \left(\hat{\mathbf{v}}-\mathbf{v}^*\right)\right\rVert_{\infty} = \left\lVert\hat{\mathbf{H}} \hat{\mathbf{v}} - \mathbf{e}_0 - \hat{\mathbf{H}} \mathbf{v}^* + \mathbf{e}_0\right\rVert_{\infty} \leq 2 \lambda_2.
\end{equation}
On the other hand, writing $d\hat{\mathbf{v}} = \hat{\mathbf{v}}-\mathbf{v}^*$,
\begin{multline*}
\lVert\mathbf{v}^*\rVert_1 \geq \lVert\hat{\mathbf{v}}\rVert_1
= \lVert\mathbf{v}^* + d\hat{\mathbf{v}}\rVert_1
= \lVert\mathbf{v}^*_{\mathcal{S}_{\mathbf{v}^*}} + \mathbf{v}^*_{\mathcal{S}_{\mathbf{v}^*}^c} + d\hat{\mathbf{v}}_{\mathcal{S}_{\mathbf{v}^*}} + d\hat{\mathbf{v}}_{\mathcal{S}_{\mathbf{v}^*}^c}\rVert_1 \\
\geq \lVert\mathbf{v}^*_{\mathcal{S}_{\mathbf{v}^*}}\rVert_1 + \lVertd\hat{\mathbf{v}}_{\mathcal{S}_{\mathbf{v}^*}^c}\rVert_1 - \lVertd\hat{\mathbf{v}}_{\mathcal{S}_{\mathbf{v}^*}}\rVert_1
= \lVert\mathbf{v}^*\rVert_1 + \lVertd\hat{\mathbf{v}}_{\mathcal{S}_{\mathbf{v}^*}^c}\rVert_1 - \lVertd\hat{\mathbf{v}}_{\mathcal{S}_{\mathbf{v}^*}}\rVert_1,
\end{multline*}
and hence,
\begin{equation*}
\lVert\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)_{\mathcal{S}_{\mathbf{v}^*}^c}\rVert_1 \leq \lVert\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)_{\mathcal{S}_{\mathbf{v}^*}}\rVert_1.
\end{equation*}
In other words, $d\hat{\mathbf{v}} = \hat{\mathbf{v}}-\mathbf{v}^* \in \mathcal{C}(s_{\mathbf{v}^*}, 1)$, and
\begin{multline} \label{eq:Dantzig:dbdhdiff:1}
\lVert\hat{\mathbf{v}}-\mathbf{v}^*\rVert_1
= \lVert\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)_{\mathcal{S}_{\mathbf{v}^*}}\rVert_1 + \lVert\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)_{\mathcal{S}_{\mathbf{v}^*}^c}\rVert_1 \\
\leq 2 \lVert\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)_{\mathcal{S}_{\mathbf{v}^*}}\rVert_1
\leq 2 \sqrt{s_{\mathbf{v}^*}} \lVert\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)_{\mathcal{S}_{\mathbf{v}^*}}\rVert_2
\leq 2 \sqrt{s_{\mathbf{v}^*}} \lVert\hat{\mathbf{v}}-\mathbf{v}^*\rVert_2.
\end{multline}

Now, on one hand,
\begin{equation} \label{eq:Dantzig:dbdhdiff:2}
\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)^\top \hat{\mathbf{H}} \left(\hat{\mathbf{v}}-\mathbf{v}^*\right)
\leq \left\lVert\hat{\mathbf{H}} \left(\hat{\mathbf{v}}-\mathbf{v}^*\right)\right\rVert_{\infty} \left\lVert\hat{\mathbf{v}}-\mathbf{v}^*\right\rVert_1
\leq 4 \sqrt{s_{\mathbf{v}^*}} \lambda_2 \left\lVert\hat{\mathbf{v}}-\mathbf{v}^*\right\rVert_2,
\end{equation}
where we have used \cref{eq:Dantzig:graddiff,eq:Dantzig:dbdhdiff:1}. On the other hand, since $\hat{\mathbf{v}}-\mathbf{v}^* \in \mathcal{C}(s_{\mathbf{v}^*}, 1)$,
\begin{equation} \label{eq:Dantzig:dbdhdiff:3}
\left(\hat{\mathbf{v}}-\mathbf{v}^*\right)^\top \hat{\mathbf{H}} \left(\hat{\mathbf{v}}-\mathbf{v}^*\right) \geq \frac{\nu_1 \nu_2}{2} \left\lVert\hat{\mathbf{v}}-\mathbf{v}^*\right\rVert_2^2.
\end{equation}
Combining \cref{eq:Dantzig:dbdhdiff:2,eq:Dantzig:dbdhdiff:3},
\begin{equation*}
\lVert\hat{\mathbf{v}}-\mathbf{v}^*\rVert_2 \leq \frac{8 \sqrt{s_{\mathbf{v}^*}} \lambda_2}{\nu_1 \nu_2}.
\end{equation*}
Again using \cref{eq:Dantzig:dbdhdiff:1},
\begin{equation*}
\lVert\hat{\mathbf{v}}-\mathbf{v}^*\rVert_1 \leq \frac{16 s_{\mathbf{v}^*} \lambda_2}{\nu_1 \nu_2}.
\end{equation*}
<!-- end proof-5 -->
<!-- lemma-5: body_md -->
\label{lem:BE:Dtrace}
Let $\nu_1$ and $\nu_2$ be the minimum eigenvalues of $\Sigma_1^*$ and $\Sigma_2^*$, respectively. Let $s_{\Delta^*} = \lVert\Delta^*\rVert_0$ and $s_{\mathbf{v}^*} = \lVert\mathbf{v}^*\rVert_0$. Suppose
\begin{gather*}
\sup_z \left\lvert\mathbb{P}\left\{\sigma^{-1} \sqrt{N} \left\{\mathbf{w}^{*\top}_1 \operatorname{vec}\left(\hat\Sigma_1-\Sigma_1^*\right) + \mathbf{w}^{*\top}_2 \operatorname{vec}\left(\hat\Sigma_2-\Sigma_2^*\right)\right\} \leq z\right\} - \Phi(z)\right\rvert \leq \varepsilon_{\textnormal{BE}}, \\
\mathbb{P}\left\{\left\lVert\hat\Sigma_1-\Sigma_1^*\right\rVert_{\infty} > \delta_{\textnormal{LD}, 1} \text{ or } \left\lVert\hat\Sigma_2-\Sigma_2^*\right\rVert_{\infty} > \delta_{\textnormal{LD}, 2}\right\} \leq \varepsilon_{\textnormal{LD}}
\end{gather*}
for some $\sigma^2 > 0$, $N = N(T_1, T_2) > 0$, $\delta_{\textnormal{LD}, 1} = \delta_{\textnormal{LD}, 1}(d, T_1) > 0$, $\delta_{\textnormal{LD}, 2} = \delta_{\textnormal{LD}, 2}(d, T_2) > 0$, $\varepsilon_{\textnormal{BE}} = \varepsilon_{\textnormal{BE}}(d, T_1, T_2) \in (0, 1)$, $\varepsilon_{\textnormal{LD}} = \varepsilon_{\textnormal{LD}}(d, T_1, T_2) \in (0, 1)$ such that
\begin{equation} \label{cond:SC:RSC}
\delta_{\textnormal{LD}, H} = \left\lVert\Sigma_2^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 1} + \left\lVert\Sigma_1^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 2} + \delta_{\textnormal{LD}, 1} \delta_{\textnormal{LD}, 2} \leq \frac{\nu_1 \nu_2}{\max\{32 s_{\Delta^*}, 8 s_{\mathbf{v}^*}\}}.
\end{equation}

If the regularization parameters satisfy
\begin{equation*}
\lambda_1 \geq 2 \left(\left\lVert\Delta^*\right\rVert_1 \delta_{\textnormal{LD}, H} + \delta_{\textnormal{LD}, 1} + \delta_{\textnormal{LD}, 2}\right), \quad \lambda_2 \geq \left\lVert\mathbf{v}^*\right\rVert_1 \delta_{\textnormal{LD}, H},
\end{equation*}
then
\begin{equation*}
\sup_z \left\lvert\mathbb{P}\left\{\sigma^{-1} \sqrt{N} \left(\tilde{\Delta}_0-\Delta^*_0\right) \leq z\right\} - \Phi(z)\right\rvert \lesssim \varepsilon_{\textnormal{BE}} + \varepsilon_{\textnormal{LD}} + \delta_{\textnormal{LD}}
\end{equation*}
with
\begin{equation*}
\delta_{\textnormal{LD}} = \frac{\sqrt{N}}{\sigma} \left(\left\lVert\Delta^*\right\rVert_1 \left\lVert\mathbf{v}^*\right\rVert_1 \delta_{\textnormal{LD}, 1} \delta_{\textnormal{LD}, 2} + \frac{\left(s_{\Delta^*} + s_{\mathbf{v}^*}\right) \lambda_1 \lambda_2}{\nu_1 \nu_2}\right) + \frac{\lambda_2}{1-\lambda_2}.
\end{equation*}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
By the decomposition in \cref{eq:decomposition},
\begin{equation*}
\sigma^{-1} \sqrt{N} \left(\tilde{\Delta}_0-\Delta^*_0\right) = \frac{\sigma^{-1} \sqrt{N} \left(U + V\right)}{1+W},
\end{equation*}
where
\begin{gather*}
U = \mathbf{w}^{*\top}_1 \operatorname{vec}\left(\hat\Sigma_1-\Sigma_1^*\right) + \mathbf{w}^{*\top}_2 \operatorname{vec}\left(\hat\Sigma_2-\Sigma_2^*\right), \quad V = V_1 + V_2 + V_3, \quad W = \left[\hat{\mathbf{H}} \hat{\mathbf{v}}\right]_0-1,
\end{gather*}
with
\begin{gather*}
V_1 = -\mathbf{v}^{*\top} \frac 12 \left\{\left(\hat\Sigma_1-\Sigma_1^*\right) \otimes \left(\hat\Sigma_2-\Sigma_2^*\right) + \left(\hat\Sigma_2-\Sigma_2^*\right) \otimes \left(\hat\Sigma_1-\Sigma_1^*\right)\right\} \operatorname{vec}(\Delta^*), \\
V_2 = -\left[\hat{\mathbf{H}} \hat{\mathbf{v}}\right]_u^\top \left(\hat\Delta_u-\Delta^*_u\right), \quad V_3 = \left(\hat{\mathbf{v}}-\mathbf{v}^*\right)^\top \nabla L_D\left(\Delta^*; \hat\Sigma_1, \hat\Sigma_2\right).
\end{gather*}

Consider the event
\begin{equation*}
\left\lVert\hat\Sigma_1-\Sigma_1^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 1}, \quad \left\lVert\hat\Sigma_2-\Sigma_2^*\right\rVert_{\infty} \leq \delta_{\textnormal{LD}, 2}.
\end{equation*}

On this event, all of the following statements hold:
\begin{enumerate}
\item By \cref{eq:Hessdiff},
\begin{equation*}
\left\lVert\hat{\mathbf{H}}-\mathbf{H}^*\right\rVert_{\infty} \leq \left\lVert\Sigma_2^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 1} + \left\lVert\Sigma_1^*\right\rVert_{\infty} \delta_{\textnormal{LD}, 2} + \delta_{\textnormal{LD}, 1} \delta_{\textnormal{LD}, 2} = \delta_{\textnormal{LD}, H}.
\end{equation*}

\item By \cref{eq:graddiff},
\begin{equation} \label{eq:infnorm:gradD}
\left\lVert\nabla L_D(\Delta^*; \hat\Sigma_1, \hat\Sigma_2)\right\rVert_{\infty} \leq \left\lVert\Delta^*\right\rVert_1 \delta_{\textnormal{LD}, H} + \delta_{\textnormal{LD}, 1} + \delta_{\textnormal{LD}, 2} = \frac{\lambda_1}{2}.
\end{equation}

\item By the definition of $\mathbf{v}^*$,
\begin{equation} \label{eq:infnorm:gradH}
\left\lVert\hat\mathbf{H} \mathbf{v}^* - \mathbf{e}_0\right\rVert_{\infty} = \left\lVert\hat{\mathbf{H}} \mathbf{v}^* - \mathbf{H}^* \mathbf{v}^*\right\rVert_{\infty} \leq \left\lVert\mathbf{v}^*\right\rVert_1 \delta_{\textnormal{LD}, H} = \lambda_2.
\end{equation}

\item Under the condition of \cref{cond:SC:RSC}, by \cref{lem:RSC},
\begin{equation*}
\mathbf{v}^\top \hat{\mathbf{H}} \mathbf{v} \geq \frac{\nu_1 \nu_2}{2} \lVert\mathbf{v}\rVert_2^2,
\end{equation*}
for any $\mathbf{v} \in \mathcal{C}(s_{\Delta^*}, 3) \cup \mathcal{C}(s_{\mathbf{v}^*}, 1)$, where $\mathcal{C}(s, \kappa)$ is the set defined in \cref{eq:cone}.
\end{enumerate}

These, in turn, imply the following statements:
\begin{enumerate}
\setcounter{enumi}{4}
\item By \cref{lem:Dh},
\begin{equation} \label{eq:onenorm:Ddiff}
\left\lVert\hat\Delta-\Delta^*\right\rVert_1 \lesssim \frac{s_{\Delta^*} \lambda_1}{\nu_1 \nu_2}.
\end{equation}

\item By \cref{lem:dbdh},
\begin{equation} \label{eq:onenorm:vdiff}
\left\lVert\hat{\mathbf{v}}-\mathbf{v}^*\right\rVert_1 \lesssim \frac{s_{\mathbf{v}^*} \lambda_2}{\nu_1 \nu_2}.
\end{equation}
\end{enumerate}

Combining the implications of \cref{eq:infnorm:gradD,eq:infnorm:gradH,eq:onenorm:Ddiff,eq:onenorm:vdiff} via H\"{o}lder's inequality,

\end{equation*}
Note that for each $i', j' \in [p]$, $\mathbf{E}_{i'j'} \otimes \mathbf{E}_{j'i'}$ has a single nonzero component 1 in the $((i'-1)p+j', (j'-1)p+i')$-th position. Thus,

\paragraph{Convergence of $\hat{\mathbf{M}}_l$ to $\mathbf{M}^*_l$.} We establish convergence for $l = 1$ as $l = 2$ follows with the same logic.
\begin{align*}
    \left\lVert\hat{\mathbf{M}}_1 - \mathbf{M}^*_1\right\rVert_1 & = \left\lVert\hat\Sigma_2 \hat\Delta \otimes \mathbf{I}_{2p} + \mathbf{I}_{2p} \otimes \hat\Sigma_2 \hat\Delta - \Sigma_2^* \Delta^* \otimes \mathbf{I}_{2p} - \mathbf{I}_{2p} \otimes \Sigma_2^* \Delta^*\right\rVert_1 \\

    & = 4p \left\lVert(\hat\Sigma_2 - \Sigma_2^*) (\hat\Delta - \Delta^*) + (\hat\Sigma_2 - \Sigma_2^*)\Delta^* + \Sigma_2^* (\hat\Delta - \Delta^*)\right\rVert_1 \\

where we used that if $\mathbf{A}, \mathbf{B} \in \mathbb{R}^{2p \times 2p}$, then $\left\lVert\mathbf{A} \mathbf{B}\right\rVert_1 \leq 2p\left\lVert\mathbf{A}\right\rVert_{\infty}\left\lVert\mathbf{B}\right\rVert_1$. Then, by \cref{lem:Dh},
\begin{align*}
    \delta_{\textnormal{M},1} = \left\lVert\hat{\mathbf{M}}_1 - \mathbf{M}^*_1\right\rVert_1 & \lesssim 8p^2 \left(\frac{ (\delta_{\textnormal{LD}, 2} + \left\lVert\Sigma_2^*\right\rVert_{\infty} )s_{\Delta^*} \lambda_1}{\nu_1 \nu_2} + \delta_{\textnormal{LD}, 2} \left\lVert\Delta^*\right\rVert_1\right) \\

where $\lambda_1$ is defined in \cref{lem:Dh}.

\paragraph{Convergence of $\hat{\mathbf{V}}_l$ to $\mathbf{V}^*_l$.} Next we will establish the rate of $\left\lVert\hat{\mathbf{V}}_l - \mathbf{V}^*_l\right\rVert_{\infty}$. Recall that $\mathbf{F}^*_l = \operatorname{Re} \mathbf{F}^*_l + \imath \operatorname{Im} \mathbf{F}^*_l = \mathbf{A}^*_l + \imath \mathbf{B}^*_l $ and in \cref{prop:asympcov},
\begin{equation*}
    \mathbf{V}^*_l = \mathbf{P}^{\top} \Psi_l \mathbf{P} \, ,
\end{equation*}
where $\Psi_l$ is given by \cref{eq:VarSh}
\begin{align*}
    \Psi_l = \frac 1 2 \begin{bmatrix}
    \left(\mathbf{I}_{p^2} + \mathbf{K}\right) \left( \mathbf{A}^*_l \otimes \mathbf{A}^*_l + \mathbf{B}^*_l \otimes \mathbf{B}^*_l\right) &
    \left(\mathbf{I}_{p^2} + \mathbf{K}\right) \left( \mathbf{B}^*_l \otimes \mathbf{A}^*_l -  \mathbf{A}^*_l \otimes \mathbf{B}^*_l\right) \\
    \left(\mathbf{I}_{p^2} - \mathbf{K}\right) \left( \mathbf{A}^*_l \otimes \mathbf{B}^*_l - \mathbf{B}^*_l \otimes \mathbf{A}^*_l\right) &
    \left(\mathbf{I}_{p^2} - \mathbf{K}\right) \left(\mathbf{A}^*_l \otimes \mathbf{A}^*_l + \mathbf{B}^*_l \otimes \mathbf{B}^*_l\right)
    \end{bmatrix}.
\end{align*}
We can remove the left and right multiplication of $\mathbf{P}$ as each row and column only has one $1$. In other words, these matrices simply select one element from the matrix so their multiplication does not change the max norm. Similarly the rows and columns of $\mathbf{I}_{p^2} + \mathbf{K}$ have either one entry with a value of $2$ or two entries each with a value of $1$ so the max norm with these matrices is at most the max norm without them multiplied by 2. Thus 
\begin{align*}
    & \left\lVert\hat{\mathbf{V}}_l - \mathbf{V}^*_l\right\rVert_{\infty} \\
    & \quad = \left\lVert\begin{bmatrix}
    (\hat{\mathbf{A}}_l \otimes \hat{\mathbf{A}}_l + \hat{\mathbf{B}}_l \otimes \hat{\mathbf{B}}_l) - (\mathbf{A}^*_l \otimes \mathbf{A}^*_l + \mathbf{B}^*_l \otimes \mathbf{B}^*_l) &
    (\hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l - \hat{\mathbf{A}}_l \otimes \hat{\mathbf{B}}_l) - (\mathbf{B}^*_l \otimes \mathbf{A}^*_l - \mathbf{A}^*_l \otimes \mathbf{B}^*_l) \\
    (\hat{\mathbf{A}}_l \otimes \hat{\mathbf{B}}_l - \hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l) - (\mathbf{A}^*_l \otimes \mathbf{B}^*_l - \mathbf{B}^*_l \otimes \mathbf{A}^*_l) &
    (\hat{\mathbf{A}}_l \otimes \hat{\mathbf{A}}_l + \hat{\mathbf{B}}_l \otimes \hat{\mathbf{B}}_l) - (\mathbf{A}^*_l \otimes \mathbf{A}^*_l + \mathbf{B}^*_l \otimes \mathbf{B}^*_l)
    \end{bmatrix}\right\rVert_{\infty} \, ,
\end{align*}
which is equivalent to the maximum of the $(1,1)$ and $(2,1)$ blocks. We will consider each of these blocks separately. Consider just the (1,1) entry,
\begin{equation*}
    \left\lVert (\hat{\mathbf{A}}_l \otimes \hat{\mathbf{A}}_l + \hat{\mathbf{B}}_l \otimes \hat{\mathbf{B}}_l) - (\mathbf{A}^*_l \otimes \mathbf{A}^*_l + \mathbf{B}^*_l \otimes \mathbf{B}^*_l) \right\rVert_{\infty} \leq \left\lVert\hat{\mathbf{A}}_l \otimes \hat{\mathbf{A}}_l - \mathbf{A}^*_l \otimes \mathbf{A}^*_l\right\rVert_{\infty} + \left\lVert\hat{\mathbf{B}}_l \otimes \hat{\mathbf{B}}_l - \mathbf{B}^*_l \otimes \mathbf{B}^*_l\right\rVert_{\infty} \,.
\end{equation*}
Studying the first entry on the right hand side,
\begin{align*}
    \left\lVert\hat{\mathbf{A}}_l \otimes \hat{\mathbf{A}}_l - \mathbf{A}^*_l \otimes \mathbf{A}^*_l\right\rVert_{\infty} & = \left\lVert (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) \otimes \hat{\mathbf{A}}_l + \mathbf{A}^*_l \otimes (\hat{\mathbf{A}}_l - \mathbf{A}^*_l)\right\rVert_{\infty} \\
    & = \left\lVert (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) \otimes (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) + (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) \otimes \mathbf{A}^*_l + \mathbf{A}^*_l \otimes (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) \right\rVert_{\infty} \\
    & \leq \left\lVert (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) \otimes (\hat{\mathbf{A}}_l - \mathbf{A}^*_l)\right\rVert_{\infty} + 2 \left\lVert(\hat{\mathbf{A}}_l - \mathbf{A}^*_l)\right\rVert_{\infty} \left\lVert\mathbf{A}^*_l\right\rVert_{\infty} \\
    & \leq  \left\lVert (\hat{\mathbf{A}}_l - \mathbf{A}^*_l)\right\rVert_{\infty}^2 + 2 \left\lVert(\hat{\mathbf{A}}_l - \mathbf{A}^*_l)\right\rVert_{\infty} \left\lVert\mathbf{A}^*_l\right\rVert_{\infty} \, ,
\end{align*} 
where we have used the fact that $\left\lVert\mathbf{X} \otimes \mathbf{Y}\right\rVert_{\infty} \leq \left\lVert\mathbf{X}\right\rVert_{\infty} \left\lVert\mathbf{Y}\right\rVert_{\infty}$ since the Kronecker product multiplies every entry in $\mathbf{X}$ with every entry in $\mathbf{Y}$. Using that $\left\lVert\hat{\mathbf{A}}_l - \mathbf{A}^*_l\right\rVert_{\infty}$, $\left\lVert\hat{\mathbf{B}}_l - \mathbf{B}^*_l\right\rVert_{\infty}$ $\leq \left\lVert\hat\Sigma_l - \Sigma^*_l\right\rVert_{\infty}$ and also that $\left\lVert\mathbf{A}^*_l\right\rVert_{\infty}$, $\left\lVert\mathbf{B}^*_l\right\rVert_{\infty} <  \left\lVert\Sigma^*_l\right\rVert_{\infty}$ and repeating the same process for $\left\lVert\hat{\mathbf{B}}_l \otimes \hat{\mathbf{B}}_l - \mathbf{B}^*_l \otimes \mathbf{B}^*_l\right\rVert_{\infty}$,
\begin{align*}
    & \left\lVert\hat{\mathbf{A}}_l \otimes \hat{\mathbf{A}}_l - \mathbf{A}^*_l \otimes \mathbf{A}^*_l\right\rVert_{\infty}  \leq \delta_{\textnormal{LD},l}^2 + 2 \delta_{\textnormal{LD},l} \left\lVert\Sigma^*_l\right\rVert_{\infty} \\
    & \left\lVert\hat{\mathbf{B}}_l \otimes \hat{\mathbf{B}}_l - \mathbf{B}^*_l \otimes \mathbf{B}^*_l\right\rVert_{\infty}  \leq \delta_{\textnormal{LD},l}^2 + 2 \delta_{\textnormal{LD},l} \left\lVert\Sigma^*_l\right\rVert_{\infty} \\
    & \left\lVert (\hat{\mathbf{A}}_l \otimes \hat{\mathbf{A}}_l + \hat{\mathbf{B}}_l \otimes \hat{\mathbf{B}}_l) - (\mathbf{A}^*_l \otimes \mathbf{A}^*_l + \mathbf{B}^*_l \otimes \mathbf{B}^*_l) \right\rVert_{\infty} \leq 2 \delta_{\textnormal{LD},l}^2 + 4 \delta_{\textnormal{LD},l} \left\lVert\Sigma^*_l\right\rVert_{\infty} \, .
\end{align*}
Next we consider the $(2,1)$ entry,
\begin{equation*}
    \left\lVert(\hat{\mathbf{A}}_l \otimes \hat{\mathbf{B}}_l - \hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l) - (\mathbf{A}^*_l \otimes \mathbf{B}^*_l - \mathbf{B}^*_l \otimes \mathbf{A}^*_l)\right\rVert_{\infty} \leq \left\lVert\mathbf{B}^*_l \otimes \mathbf{A}^*_l - \hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l\right\rVert_{\infty} + \left\lVert \hat{\mathbf{A}}_l \otimes \hat{\mathbf{B}}_l - \mathbf{A}^*_l \otimes \mathbf{B}^*_l\right\rVert_{\infty}  \, .
\end{equation*}
For the first entry on the right hand side,
\begin{align*}
    \left\lVert\mathbf{B}^*_l \otimes \mathbf{A}^*_l - \hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l\right\rVert_{\infty} & =  \left\lVert\hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l - \mathbf{B}^*_l \otimes \mathbf{A}^*_l\right\rVert_{\infty}  \\
    & = \left\lVert (\hat{\mathbf{B}}_l - \mathbf{B}^*_l) \otimes \hat{\mathbf{A}}_l + \mathbf{B}^*_l \otimes \hat{\mathbf{A}}_l - \mathbf{B}^*_l \otimes \mathbf{A}^*_l\right\rVert_{\infty} \\
    & = \left\lVert(\hat{\mathbf{B}}_l - \mathbf{B}^*_l) \otimes \hat{\mathbf{A}}_l + \mathbf{B}^*_l \otimes (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) \right\rVert_{\infty} \\
    & = \left\lVert(\hat{\mathbf{B}}_l - \mathbf{B}^*_l) \otimes (\hat{\mathbf{A}}_l - \mathbf{A}^*_l) + (\hat{\mathbf{B}}_l - \mathbf{B}^*_l) \otimes \mathbf{A}^*_l + \mathbf{B}^*_l \otimes (\hat{\mathbf{A}}_l - \mathbf{A}^*_l))\right\rVert_{\infty} \\
    & \leq \left\lVert\hat{\mathbf{B}}_l - \mathbf{B}^*_l\right\rVert_{\infty} \left\lVert\hat{\mathbf{A}}_l - \mathbf{A}^*_l\right\rVert_{\infty} + \left\lVert\hat{\mathbf{B}}_l - \mathbf{B}^*_l\right\rVert_{\infty} \left\lVert\mathbf{A}^*_l\right\rVert_{\infty} + \left\lVert\hat{\mathbf{A}}_l - \mathbf{A}^*_l\right\rVert_{\infty} \left\lVert\mathbf{B}^*_l\right\rVert_{\infty} \, .
\end{align*}
Repeating the same process for $ \left\lVert\hat{\mathbf{A}}_l \otimes \hat{\mathbf{B}}_l - \mathbf{A}^*_l \otimes \mathbf{B}^*_l\right\rVert_{\infty}$,
\begin{align*}
    & \left\lVert\hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l - \mathbf{B}^*_l \otimes \mathbf{A}^*_l\right\rVert_{\infty} \leq \delta_{\textnormal{LD},l}^2 + 2 \delta_{\textnormal{LD},l} \left\lVert\Sigma^*_l\right\rVert_{\infty} \\
    & \left\lVert\hat{\mathbf{A}}_l \otimes \hat{\mathbf{B}}_l - \mathbf{A}^*_l \otimes \mathbf{B}^*_l\right\rVert_{\infty} \leq \delta_{\textnormal{LD},l}^2 + 2 \delta_{\textnormal{LD},l} \left\lVert\Sigma^*_l\right\rVert_{\infty} \\
    & \left\lVert (\hat{\mathbf{B}}_l \otimes \hat{\mathbf{A}}_l - \hat{\mathbf{A}}_l \otimes \hat{\mathbf{B}}_l) - (\mathbf{B}^*_l \otimes \mathbf{A}^*_l - \mathbf{A}^*_l \otimes \mathbf{B}^*_l) \right\rVert_{\infty} \leq 2 \delta_{\textnormal{LD},l}^2 + 4 \delta_{\textnormal{LD},l} \left\lVert\Sigma^*_l\right\rVert_{\infty}  \, .
\end{align*}
and we conclude
\begin{align*}
    \delta_{\textnormal{V},1} & = \left\lVert\hat{\mathbf{V}}_1 - \mathbf{V}^*_1\right\rVert_{\infty} \leq 2 \delta_{\textnormal{LD}, 1}^2 + 4 \delta_{\textnormal{LD}, 1} \left\lVert\Sigma_1^*\right\rVert_{\infty} \\
    \delta_{\textnormal{V},2} & = \left\lVert\hat{\mathbf{V}}_2 - \mathbf{V}^*_2\right\rVert_{\infty} \leq 2 \delta_{\textnormal{LD}, 2}^2 + 4 \delta_{\textnormal{LD}, 2} \left\lVert\Sigma_2^*\right\rVert_{\infty} \, .
\end{align*}

\paragraph{Convergence of $\mathbf{I}_{4p^2} \otimes \hat{\mathbf{M}}_l \hat{\mathbf{V}}_l$ to $\mathbf{I}_{4p^2} \otimes \mathbf{M}^*_l \mathbf{V}^*_l$.}
\begin{align*}
    \left\lVert\mathbf{I}_{4p^2} \otimes \hat{\mathbf{M}}_l \hat{\mathbf{V}}_l - \mathbf{I}_{4p^2} \otimes \mathbf{M}^*_l \mathbf{V}^*_l\right\rVert_{\infty} & = \left\lVert\hat{\mathbf{M}}_l \hat{\mathbf{V}}_l -  \mathbf{M}^*_l \mathbf{V}^*_l\right\rVert_{\infty} \\
    & = \left\lVert(\hat{\mathbf{V}}^{\top}_l \otimes \mathbf{I}_{4p^2}) \operatorname{vec}(\hat{\mathbf{M}}_l) - (\mathbf{V}^{* \top}_l \otimes \mathbf{I}_{4p^2}) \operatorname{vec}(\mathbf{M}^*_l)\right\rVert_{\infty} \\
    & = \left\lVert(\hat{\mathbf{V}}^{\top}_l \otimes \mathbf{I}_{4p^2} - \mathbf{V}^{* \top}_l \otimes \mathbf{I}_{4p^2}) \operatorname{vec}(\hat{\mathbf{M}}_l) - (\mathbf{V}^{* \top}_l \otimes \mathbf{I}_{4p^2}) \operatorname{vec}(\mathbf{M}^*_l - \hat{\mathbf{M}}_l)\right\rVert_{\infty} \\
    & \leq \left\lVert(\hat{\mathbf{V}}^{\top}_l \otimes \mathbf{I}_{4p^2} - \mathbf{V}^{* \top}_l \otimes \mathbf{I}_{4p^2}) \operatorname{vec}(\hat{\mathbf{M}}_l - \mathbf{M}^*_l)\right\rVert_{\infty} \\ 
    & \quad + \left\lVert(\hat{\mathbf{V}}^{\top}_l \otimes \mathbf{I}_{4p^2} - \mathbf{V}^{* \top}_l \otimes \mathbf{I}_{4p^2}) \operatorname{vec}(\mathbf{M}^*_l)\right\rVert_{\infty} \\
    & \quad - \left\lVert(\mathbf{V}^{* \top}_l \otimes \mathbf{I}_{4p^2}) \operatorname{vec}(\mathbf{M}^*_l - \hat{\mathbf{M}}_l)\right\rVert_{\infty} \\
    & \leq \left\lVert \hat{\mathbf{V}}_l - \mathbf{V}^*_l\right\rVert_{\infty} \left\lVert\hat{\mathbf{M}}_l - \mathbf{M}^*_l\right\rVert_1 + \left\lVert \hat{\mathbf{V}}_l - \mathbf{V}^*_l\right\rVert_{\infty} \left\lVert\mathbf{M}^*_l\right\rVert_1 \\
    & \quad + \left\lVert\mathbf{V}^*_l\right\rVert_{\infty} \left\lVert\hat{\mathbf{M}}_l - \mathbf{M}^*_l\right\rVert_1 \, .
\end{align*}
Then,
\begin{align*}
    \left\lVert\mathbf{I}_{4p^2} \otimes \hat{\mathbf{M}}_1 \hat{\mathbf{V}}_1 - \mathbf{I}_{4p^2} \otimes \mathbf{M}^*_1 \mathbf{V}^*_1\right\rVert_{\infty} & \lesssim  \delta_{\textnormal{V},1} \delta_{\textnormal{M},1} + \delta_{\textnormal{V},1} \left\lVert\mathbf{M}^*_1\right\rVert_1 + \left\lVert\mathbf{V}^*_1\right\rVert_{\infty} \delta_{\textnormal{M},1} \\
    \left\lVert\mathbf{I}_{4p^2} \otimes \hat{\mathbf{M}}_2 \hat{\mathbf{V}}_2 - \mathbf{I}_{4p^2} \otimes \mathbf{M}^*_2 \mathbf{V}^*_2\right\rVert_{\infty} & \lesssim  \delta_{\textnormal{V},2} \delta_{\textnormal{M},2} + \delta_{\textnormal{V},2} \left\lVert\mathbf{M}^*_2\right\rVert_1 + \left\lVert\mathbf{V}^*_2\right\rVert_{\infty} \delta_{\textnormal{M},2} \, .
\end{align*}
\paragraph{Convergence of $\hat{\mathbf{G}}_l$ to $\mathbf{G}^*_l$.}
\begin{align*}
\left\lVert\hat{\mathbf{G}}_l - \mathbf{G}^*_l\right\rVert_{\infty} & = \left\lVert\hat{\mathbf{M}}_l \hat{\mathbf{V}}_l \hat{\mathbf{M}}^{\top}_l - \mathbf{M}^*_l \mathbf{V}^*_l \mathbf{M}^{* \top}_l\right\rVert_{\infty} \\
    & = \left\lVert(\mathbf{I}_{4p^2} \otimes \hat{\mathbf{M}}_l \hat{\mathbf{V}}_l) \operatorname{vec}(\hat{\mathbf{M}}^{\top}_l) - (\mathbf{I}_{4p^2} \otimes \mathbf{M}^*_l \mathbf{V}^*_l) \operatorname{vec}(\mathbf{M}^{* \top}_l)\right\rVert_{\infty} \\
    & \leq \left\lVert(\mathbf{I}_{4p^2} \otimes \hat{\mathbf{M}}_l \hat{\mathbf{V}}_l - \mathbf{I}_{4p^2} \otimes \mathbf{M}^*_l \mathbf{V}^*_l) \operatorname{vec}(\hat{\mathbf{M}}^{\top}_l)\right\rVert_{\infty} + \left\lVert(\mathbf{I}_{4p^2} \otimes \mathbf{M}^*_l \mathbf{V}^*_l) \operatorname{vec}(\hat{\mathbf{M}}^{\top}_l - \mathbf{M}^{* \top}_l)\right\rVert_{\infty} \\
    & \leq \left\lVert\hat{\mathbf{M}}_l \hat{\mathbf{V}}_l - \mathbf{M}^*_l \mathbf{V}^*_l\right\rVert_{\infty} \left\lVert\operatorname{vec}(\hat{\mathbf{M}}^{\top}_l - \mathbf{M}^{* \top}_l)\right\rVert_1 + \left\lVert\hat{\mathbf{M}}_l \hat{\mathbf{V}}_l - \mathbf{M}^*_l \mathbf{V}^*_l\right\rVert_{\infty} \left\lVert\operatorname{vec}(\mathbf{M}^{* \top}_l)\right\rVert_1 \\
    & \quad + \left\lVert\mathbf{M}^*_l \mathbf{V}^*_l\right\rVert_{\infty} \left\lVert\operatorname{vec}(\hat{\mathbf{M}}^{\top}_l - \mathbf{M}^{* \top}_l)\right\rVert_1 \, .
\end{align*}
Using the same simplification for $\left\lVert\hat{\mathbf{G}}_2 - \mathbf{G}^*_2\right\rVert_{\infty}$ and based on the rates above,
\begin{align*}
    \delta_{\textnormal{G},1} & = \left\lVert\hat{\mathbf{G}}_1 - \mathbf{G}^*_1\right\rVert_{\infty} \lesssim  \left(\delta_{\textnormal{V},1} \delta_{\textnormal{M},1} + \delta_{\textnormal{V},1} \left\lVert\mathbf{M}^*_1\right\rVert_1 + \left\lVert\mathbf{V}^*_1\right\rVert_{\infty} \delta_{\textnormal{M},1}\right) \left(\delta_{\textnormal{M},1} + \left\lVert\mathbf{M}^*_1\right\rVert_1\right) + \left\lVert\mathbf{M}^*_1 \mathbf{V}^*_1\right\rVert_{\infty} \delta_{\textnormal{M},1} \\
    \delta_{\textnormal{G},2} & = \left\lVert\hat{\mathbf{G}}_2 - \mathbf{G}^*_2\right\rVert_{\infty} \lesssim  \left(\delta_{\textnormal{V},2} \delta_{\textnormal{M},2} + \delta_{\textnormal{V},2} \left\lVert\mathbf{M}^*_2\right\rVert_1 + \left\lVert\mathbf{V}^*_2\right\rVert_{\infty} \delta_{\textnormal{M},2}\right) \left(\delta_{\textnormal{M},2} + \left\lVert\mathbf{M}^*_2\right\rVert_1\right) + \left\lVert\mathbf{M}^*_2 \mathbf{V}^*_2\right\rVert_{\infty} \delta_{\textnormal{M},2} \, .
\end{align*}

\paragraph{Convergence of plug-in variance estimator.} Now we are ready to show that our plug-in estimate of the variance is consistent. To show that $\hat{\sigma}^2$ is consistent we will show that 
\begin{equation*}
    \left\lvert\hat{\sigma}^2 - \sigma^2\right\rvert = o_p(1) \,.
\end{equation*}
We proceed by studying $\lvert\hat{\mathbf{v}}^{\top} \hat{\mathbf{G}}_l \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_l \mathbf{v}^*\rvert$,
\begin{align*}
    & \left\lvert\hat{\mathbf{v}}^{\top} \hat{\mathbf{G}}_l \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_l \mathbf{v}^*\right\rvert \\ 
    & \quad = \left\lvert\hat{\mathbf{v}}^{\top} \hat{\mathbf{G}}_l \hat{\mathbf{v}} - \mathbf{v}^{*\top} \hat{\mathbf{G}}_l \hat{\mathbf{v}} + \mathbf{v}^{*\top} \hat{\mathbf{G}}_l \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_l \mathbf{v}^*\right\rvert \\
    & \quad = \left\lvert (\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}) \hat{\mathbf{G}}_l \hat{\mathbf{v}} + \mathbf{v}^{*\top} \hat{\mathbf{G}}_l \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_l \mathbf{v}^*\right\rvert \\
    & \quad = \left\lvert (\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}) \hat{\mathbf{G}}_l (\hat{\mathbf{v}} - \mathbf{v}^*) + (\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}) \hat{\mathbf{G}}_l \mathbf{v}^* + \mathbf{v}^{*\top} \hat{\mathbf{G}}_l \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_l \mathbf{v}^*\right\rvert \\
    & \quad \leq \left\lvert(\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}) \hat{\mathbf{G}}_l (\hat{\mathbf{v}} - \mathbf{v}^*)\right\rvert + \left\lvert(\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}) \hat{\mathbf{G}}_l \mathbf{v}^*\right\rvert + \left\lvert\mathbf{v}^{*\top} \hat{\mathbf{G}}_l (\hat{\mathbf{v}} - \mathbf{v}^*)\right\rvert + \left\lvert\mathbf{v}^{*\top} (\hat{\mathbf{G}}_l - \mathbf{G}^*_l) \mathbf{v}^*\right\rvert \\
    & \quad \leq \left\lVert\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}\right\rVert_1 \left\lVert\hat{\mathbf{G}}_l (\hat{\mathbf{v}} - \mathbf{v}^*)\right\rVert_{\infty} + 2 \left\lVert\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}\right\rVert_1  \left\lVert\hat{\mathbf{G}}_l \mathbf{v}^*\right\rVert_{\infty} + \left\lVert\mathbf{v}^*\right\rVert_1 \left\lVert(\hat{\mathbf{G}}_l - \mathbf{G}^*_l) \mathbf{v}^*\right\rVert_{\infty} \\
    & \quad \leq \left\lVert\hat{\mathbf{v}} - \mathbf{v}^*\right\rVert_1^2 \left\lVert\hat{\mathbf{G}}_l\right\rVert_{\infty}  + 2 \left\lVert\hat{\mathbf{v}}^{\top} - \mathbf{v}^{*\top}\right\rVert_1  \left\lVert (\hat{\mathbf{G}}_l - \mathbf{G}^*_l) \mathbf{v}^* + \mathbf{G}^*_l \mathbf{v}^*\right\rVert_{\infty} + \left\lVert\mathbf{v}^*\right\rVert_1^2 \left\lVert\hat{\mathbf{G}}_l - \mathbf{G}^*_l\right\rVert_{\infty} \\
    & \quad \leq \left\lVert\hat{\mathbf{v}} - \mathbf{v}^*\right\rVert_1^2 \left\lVert\hat{\mathbf{G}}_l - \mathbf{G}^*_l\right\rVert_{\infty} +  \left\lVert\hat{\mathbf{v}} - \mathbf{v}^*\right\rVert_1^2 \left\lVert\mathbf{G}^*_l\right\rVert_{\infty} + 2 \left\lVert\hat{\mathbf{v}} - \mathbf{v}^*\right\rVert_1 \left\lVert\hat{\mathbf{G}}_l - \mathbf{G}^*_l\right\rVert_{\infty} \left\lVert\mathbf{v}^*\right\rVert_1 \\
    & \qquad + 2 \left\lVert\hat{\mathbf{v}} - \mathbf{v}^*\right\rVert_1 \left\lVert\mathbf{G}^*_l \mathbf{v}^*\right\rVert_{\infty} + \left\lVert\mathbf{v}^*\right\rVert_1^2 \left\lVert\hat{\mathbf{G}}_l - \mathbf{G}^*_l\right\rVert_{\infty} \, .
\end{align*}
Comparing the rate for $\left\lVert\hat{\mathbf{v}} - \mathbf{v}^*\right\rVert_1$ from \cref{lem:dbdh} with $\delta_{\textnormal{G},1}, \delta_{\textnormal{G},2}$ it can be seen that $\delta_{\textnormal{G},1}, \delta_{\textnormal{G},2}$ are slower and thus dominate the error rate. Closer examination of $\delta_{\textnormal{G},1}, \delta_{\textnormal{G},2}$ and their components reveal that $\delta_{\textnormal{M},1}, \delta_{\textnormal{M},2}$, specifically $8p^2 \delta_{\textnormal{LD}, 1}$ and  $8p^2 \delta_{\textnormal{LD}, 2}$, are the dominant terms. Recall from \cref{thm:BE:projWelch} that the effective sample size is $N = N_1 + N_2$ where $N_1 = T_1/B_1$, $N_2 = T_2/B_2$. Then, if $N_1 \asymp N_2$, and under \cref{assm}, by \cref{thm:B_rates}, we see that $\delta_{\textnormal{LD}, 1}, \delta_{\textnormal{LD}, 2}$ have rates given by \cref{eq:Sxhrate,eq:Syhrate} with high probability. Plugging in these rates, and simplifying we conclude that 
\begin{align*}
    \left\lvert\hat{\mathbf{v}}^{\top} \hat{\mathbf{G}}_1 \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_1 \mathbf{v}^*\right\rvert & = O_p\left(p^2 \sqrt{\frac{B_2}{T_2}} \right) \\
    \left\lvert\hat{\mathbf{v}}^{\top} \hat{\mathbf{G}}_2 \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_2 \mathbf{v}^*\right\rvert & = O_p\left(p^2 \sqrt{\frac{B_1}{T_1}} \right) \, ,
\end{align*}
and
\begin{align*}
    \left\lvert\hat{\sigma}^2 - \sigma^2\right\rvert & \leq \frac{N}{4N_1} \left\lvert\hat{\mathbf{v}}^{\top} \hat{\mathbf{G}}_1 \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_1 \mathbf{v}^*\right\rvert + \frac{N}{4N_2} \left\lvert\hat{\mathbf{v}}^{\top} \hat{\mathbf{G}}_2 \hat{\mathbf{v}} - \mathbf{v}^{*\top} \mathbf{G}^*_2 \mathbf{v}^*\right\rvert \\
    & = O_p\left(p^2 \left(\sqrt{\frac{B_2}{T_2}} + \sqrt{\frac{B_1}{T_1}}\right)\right) \\
    & = o_p(1) \, .
\end{align*}

\label{lem:BE}
Suppose
\begin{equation*}
\sup_z \left\lvert\mathbb{P}\left\{U \leq z\right\} - \Phi(z)\right\rvert \leq \varepsilon_U, \quad \mathbb{P}\left\{|V| \leq \delta_V,\ |W| \leq \delta_W\right\} \geq 1-\varepsilon_{VW}
\end{equation*}
for some $\delta_V, \delta_W, \varepsilon_U, \varepsilon_{VW} \in [0, 1)$. Then,
 \begin{equation*}
 \sup_z \left\lvert\mathbb{P}\left\{\frac{U+V}{1+W} \leq z\right\} - \Phi(z)\right\rvert \leq \delta_V + \frac{\delta_W}{1-\delta_W} + \varepsilon_U + \varepsilon_{VW}.
 \end{equation*}
