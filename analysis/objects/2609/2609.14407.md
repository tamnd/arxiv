<!-- question-1: body_md -->
\label{Q1}
	Is the Jacobian $\mathcal{J}: \dot{W}^{1,np}(\mathbb{R}^n,\mathbb{R}^n)\rightarrow \mathscr{H}^p(\mathbb{R}^n)$ surjective?
<!-- end question-1 -->
<!-- theorem-1: body_md -->
\label{theorem-1}
	Let $f:\mathbb{R}^2\rightarrow\mathbb{R}$ be a compactly supported function such that $x\mapsto f(x,y)$ is a bounded measurable for all $y\in\mathbb{R}$ and $y\mapsto f(x,y)$ is $C^1$ for all $x\in\mathbb{R}$. For each $p>1$, there exists a function $u\in \dot{W}^{1,2p}(\mathbb{R}^2)\cap L^\infty(\mathbb{R}^2)$ such that
	\begin{equation}
		\det(\nabla u(x,y))= f(x,y) \mbox{ for a.e. }(x,y)\in \mathbb{R}^2.
	\end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
Let $f\in C_c^\infty(\mathbb{R}^2)$. For each $p>1$, there exists a function $u\in \dot{W}^{1,2p}(\mathbb{R}^2)\cap L^\infty(\mathbb{R}^2)$ such that
	\begin{equation}
		\det(\nabla u(x,y))= f(x,y) \mbox{ for all }(x,y)\in \mathbb{R}^2.
	\end{equation}
	Furthermore, it holds that $u\in C^\infty(\mathbb{R}^2)$.
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
Let $R_0>0$ be such that $\mbox{supp}(f)\subset (-R_0,R_0)\times (-R_0,R_0)$. We wish to construct functions $u_i:\mathbb{R}^2\rightarrow\mathbb{R}$ for $i=1,2$ such that $\det(\nabla u)=f$ where $u=(u_1,u_2)$. Let us consider $u_2(x,y)$ to be a Lipschitz function on $\mathbb{R}^2$ such that
	\begin{enumerate}[(i.)]
		\item For $|x|\leq 2R_0$ and $|y|\geq 2R_0$, $|u_2(x,y)|\geq R_0$.
		\item For $|x|\leq 2R_0$ and $|y|\leq 2R_0$, $u_2(x,y)=y$. 
	\end{enumerate}
	We will give an explicit construction of such $u_2$ later. Suppose that such $u_2$ exists. Similar to \cite{Moser65}, we then set
	\begin{equation*}
		u_1(x,y):=	\int_{-R_0}^{x}f(t,u_2(x,y))dt.
	\end{equation*}
	From our choice of $u_2$, we get
	\begin{equation*}
		u_1(x,y)=	\int_{-R_0}^{x}f(t,R_0)dt=0\mbox{ when }|x|\leq R_0\mbox{ and } |y|\geq R_0,
	\end{equation*}
	since $f(t,z)=0$ for any $t\in\mathbb{R}$ and $|z|\geq R_0$. We can now calculate
	\begin{align}
		\partial_x u_1(x,y) &=f(x,u_2(x,y))+\partial_xu_2(x,y)\int_{-R_0}^{x}\partial_2f(t,u_2(x,y))dt,\label{derivative-u1-x}\\
		\partial_y u_1(x,y) &=\partial_yu_2(x,y)\int_{-R_0}^{x}\partial_2f(t,u_2(x,y))dt.\label{derivative-u1-y}
	\end{align}
	Here $\partial_2$ denotes the partial derivative of the function $f$ with respect to second argument. Now, we may calculate $\det(\nabla u)$ as follows
	\begin{align*}
		\det(\nabla u(x,y))&=\partial_x u_1(x,y)\partial_y u_2(x,y)-\partial_y u_1(x,y)\partial_x u_2(x,y)\\
		&=f(x,u_2(x,y))\partial_y u_2(x,y)+\partial_xu_2(x,y)\partial_y u_2(x,y)\int_{-R_0}^{x}\partial_2f(t,u_2(x,y))dt\\
		&-\partial_x u_2(x,y)\partial_yu_2(x,y)\int_{-R_0}^{x}\partial_2f(t,u_2(x,y))dt\\
		&=f(x,u_2(x,y))\partial_y u_2(x,y).
	\end{align*}
	Observe that the following
	\begin{enumerate}
		\item when $|x|\geq R_0$ we have $f(x,u_2(x,y))=0$.
		\item For $|x|\leq R_0$ and $|y|\geq R_0$ from property (i) of $u_2$ we again get $f(x,u_2(x,y))=0$ since support of $f$ is contained in side $(-R_0,R_0)\times (-R_0,R_0)$. 
		\item When $|x|\leq R_0$ and $|y|\leq R_0$ , we have $\partial_y u_2(x,y)=1$ from property (ii).
	\end{enumerate} 
	Hence, we have
	\begin{equation}
		\det(\nabla u(x,y))=f(x,y)\mbox{ for all }(x,y)\in\mathbb{R}^2. 
	\end{equation}
	We define the now, the function $u_2$ as follows
	\begin{enumerate}
		\item For $-\infty<x\leq -R_0$,
		\begin{equation*}
			u_2(x,y)=\left\{
			\begin{array}{cl}
				\frac{2R_0^2y}{R_0^2+x^2}&\text{ for }-\infty<x\leq -R_0,|y|\leq R_0,\\
				\frac{2R_0^3\left(3R_0-x-y\right)}{(R_0^2+x^2)(2R_0-x)}+\frac{R_0(y-R_0)}{2R_0-x}&\text{ for }-\infty<x\leq -R_0, R_0\leq y\leq 3R_0-x,\\
				R_0&\text{ for }-\infty<x\leq -R_0, y\geq 3R_0-x,\\
				-\frac{2R_0^3\left(3R_0-x+y\right)}{(R_0^2+x^2)(2R_0-x)}+\frac{R_0(y+R_0)}{2R_0-x}&\text{ for }-\infty<x\leq -R_0, -3R_0+x\leq y\leq -R_0,\\
				-R_0&\text{ for }-\infty<x\leq -R_0, y\leq -3R_0+x.
			\end{array}\right.
		\end{equation*}
		\item For $-R_0\leq x\leq R_0$,
		\begin{equation*}
			u_2(x,y)=\left\{
			\begin{array}{cl}
				y&\text{ for }-R_0\leq x\leq R_0,|y|\leq R_0,\\
				R_0&\text{ for }-R_0\leq x\leq -R_0, R_0\leq y<+\infty,\\ 
				-R_0&\text{ for }-R_0\leq x\leq -R_0, -\infty<y\leq -R_0.
			\end{array}\right.
		\end{equation*}
		\item For $R_0\leq x<+\infty$,
		\begin{equation*}
			u_2(x,y)=\left\{
			\begin{array}{cl}
				\frac{2R_0^2y}{R_0^2+x^2}&\text{ for }R_0\leq x< +\infty,|y|\leq R_0,\\
				\frac{2R_0^3\left(3R_0+x-y\right)}{(R_0^2+x^2)(2R_0+x)}+\frac{R_0(y-R_0)}{2R_0+x}&\text{ for }R_0\leq x<+\infty, R_0\leq y\leq 3R_0+x,\\
				R_0&\text{ for }R_0\leq x<+\infty, y\geq 3R_0+x,\\
				-\frac{2R_0^3\left(3R_0+x+y\right)}{(R_0^2+x^2)(2R_0+x)}+\frac{R_0(y+R_0)}{2R_0+x}&\text{ for }R_0\leq x<+\infty, -3R_0-x\leq y\leq -R_0,\\
				-R_0&\text{ for }R_0\leq x<+\infty, y\leq -3R_0-x.
			\end{array}\right.
		\end{equation*}
	\end{enumerate}
	One can check that $u_2\in Lip(\mathbb{R}^2)$. Next we show that $\nabla u\in \dot{W}^{1,2p}(\mathbb{R}^2)$. From \eqref{derivative-u1-x} and \eqref{derivative-u1-y}, we can estimate $||\cdot||_{L^{2p}(\mathbb{R}^2)}$ of $\nabla u_1$ as follows.
	\begin{align*}
		&\int_{\mathbb{R}^2}(|\partial_xu_1(x,y)|^{2p}+|\partial_yu_1(x,y)|^{2p})dxdy\\
		&\leq C_p\int_{\mathbb{R}}\int_{-R_0}^{R_0}|f(x,u_2(x,y))|^{2p}dxdy+C_p\int_{-R_0}^{\infty}\int_{\mathbb{R}}|\partial_xu_2(x,y)|^{2p}\int_{-R_0}^{x}|\partial_2f(t,u_2(x,y))|^{2p}dtdydx\\
		&+\int_{-R_0}^{\infty}\int_{\mathbb{R}}|\partial_yu_2(x,y)|^{2p}\int_{-R_0}^{x}|\partial_2f(t,u_2(x,y))|^{2p}dtdydx,
                                                                        
                                                                                                                                                                                                 
                                                                                                                                                                                    
	\end{align*}
	for some constant $C_p>1$. Here we have used Jensen's inequality. Therefore, we get
	\begin{align*}
		&\int_{\mathbb{R}^2}(|\partial_xu_1(x,y)|^{2p}+|\partial_yu_1(x,y)|^{2p})dxdy\\
                                                                                                                                                               
                                                                                                  
		&\leq C_p\int_{-R_0}^{R_0}\int_{-R_0}^{R_0}|f(x,u_2(x,y))|^{2p}dxdy\\
		&+C_p\left(\sup\limits_{z\in \mathbb{R}}\int_{-R_0}^{R_0}|\partial_2f(t,z)|^{2p}dt\right)\left(\int_{-R_0}^{\infty}\int_{\mathbb{R}}|\partial_xu_2(x,y)|^{2p}dydx+\int_{-R_0}^{\infty}\int_{\mathbb{R}}|\partial_yu_2(x,y)|^{2p}dydx\right)\\
		&\leq C_p\int_{-R_0}^{R_0}\int_{-R_0}^{R_0}|f(x,u_2(x,y))|^{2p}dxdy+C_p\left(\sup\limits_{z\in \mathbb{R}}\int_{-R_0}^{R_0}|\partial_2f(t,z)|^{2p}dt\right)||\nabla u_2||^{2p}_{L^{2p}(\mathbb{R}^2)}.
	\end{align*}
Now, it remains to prove that $\nabla u_2\in L^{2p}(\mathbb{R}^2)$. We can obtain
\begin{align*}
			&\int_{\mathbb{R}^2}(|\partial_xu_2(x,y)|^{2p}+|\partial_yu_2(x,y)|^{2p})dxdy\\
			&\leq \int_{\mathbb{R}}\int_{-\infty}^{-R_0}(|\partial_xu_2(x,y)|^{2p}+|\partial_yu_2(x,y)|^{2p})dxdy\\
			&+\int_{\mathbb{R}}\int_{-R_0}^{R_0}(|\partial_xu_2(x,y)|^{2p}+|\partial_yu_2(x,y)|^{2p})dxdy\\
			&+\int_{\mathbb{R}}\int_{R_0}^{\infty}(|\partial_xu_2(x,y)|^{2p}+|\partial_yu_2(x,y)|^{2p})dxdy\\
			&=:I_1+I_2+I_3.
\end{align*}
Since $\nabla u_2\in L^\infty(\mathbb{R}^2)$, we can conclude that $I_2<+\infty$. Next we are going to show that $I_1<+\infty$. Proof for $I_3<+\infty$ follows in a similar way and we will omit here. From the definition, we can calculate the derivative of $u_2$ as follows.
\begin{enumerate}
	\item For $-\infty<x< -R_0,|y|< R_0$, we have
	\begin{equation*}
		\partial_x u_2(x,y)=-\frac{4R_0^2xy}{(R^2_0+x^2)^2}\mbox{ and }\partial_y u_2(x,y)=\frac{2R_0^2}{R_0^2+x^2}.
	\end{equation*}
	\item For $-\infty<x< -R_0, R_0< y<3R_0-x$, it follows
	\begin{align*}
		\partial_x u_2(x,y)&=-\frac{2R_0^3}{(R_0^2+x^2)(2R_0-x)}+\frac{2R_0^3\left(3R_0-x-y\right)}{(R_0^2+x^2)(2R_0-x)^2}\\
		&-\frac{4R_0^3x\left(3R_0-x-y\right)}{(R_0^2+x^2)^2(2R_0-x)}+\frac{R_0(y-R_0)}{(2R_0-x)^2},\\
		\partial_y u_2(x,y)&=-\frac{2R_0^3}{(R_0^2+x^2)(2R_0-x)}+\frac{R_0}{2R_0-x}.
	\end{align*}
	\item For $-\infty<x< -R_0, y> 3R_0-x$, we get
	\begin{equation*}
		\partial_xu_2(x,y)=0\mbox{ and }\partial_yu_2(x,y)=0.
	\end{equation*}
	\item For $-\infty<x< -R_0, -3R_0+x< y<-R_0$, it follows
	\begin{align*}
			\partial_x u_2(x,y)&=\frac{2R_0^3}{(R_0^2+x^2)(2R_0-x)}-\frac{2R_0^3\left(3R_0-x+y\right)}{(R_0^2+x^2)(2R_0-x)^2}\\
			&+\frac{4R_0^3x\left(3R_0-x+y\right)}{(R_0^2+x^2)^2(2R_0-x)}+\frac{R_0(y+R_0)}{(2R_0-x)^2},\\
			\partial_y u_2(x,y)&=-\frac{2R_0^3}{(R_0^2+x^2)(2R_0-x)}+\frac{R_0}{2R_0-x}.
	\end{align*}
	\item For $-\infty<x\leq -R_0, y< -3R_0+x$, we have
	\begin{equation*}
		\partial_xu_2(x,y)=0\mbox{ and }\partial_yu_2(x,y)=0.
	\end{equation*}
\end{enumerate}
Then, it follows
\begin{align*}
	&\int_{-\infty}^{-R_0}\int_\mathbb{R} (|\partial_xu_1(x,y)|^{2p}+|\partial_yu_1(x,y)|^{2p})dxdy\\
	&\leq 	C_p\int_{-\infty}^{-R_0}\int_{-R_0}^{R_0}\left(\frac{(4R_0^2)^{2p}|x|^{2p}|y|^{2p}}{(R^2_0+x^2)^{4p}}+\frac{(2R_0^2)^{2p}}{(R_0^2+x^2)^{2p}}\right)dydx\\
	&+C_p\int_{-\infty}^{-R_0}\int_{R_0}^{3R_0-x}\left(\frac{(2R_0^3)^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{(2R_0^3)^{2p}|3R_0-x-y|^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{4p}}\right)dydx\\
	&+C_p\int_{-\infty}^{-R_0}\int_{R_0}^{3R_0-x}\left(\frac{(4R_0^3)^{2p}|x|^{2p}|3R_0-x-y|^{2p}}{(R_0^2+x^2)^{4p}(2R_0-x)^{2p}}+\frac{R_0^{2p}|y-R_0|^{2p}}{(2R_0-x)^{4p}}\right)dydx\\
	&+C_p\int_{-\infty}^{-R_0}\int_{R_0}^{3R_0-x}\left(\frac{(2R_0^3)^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{R_0^{2p}}{(2R_0-x)^{2p}}\right)dydx\\
	&+C_p\int_{-\infty}^{-R_0}\int_{-3R_0+x}^{-R_0}\left(\frac{(2R_0^3)^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{(2R_0^3)^{2p}|3R_0-x+y|^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{4p}}\right)dydx\\
	&+C_p\int_{-\infty}^{-R_0}\int_{-3R_0+x}^{-R_0}\left(\frac{(4R_0^3)^{2p}|x|^{2p}|3R_0-x+y|^{2p}}{(R_0^2+x^2)^{4p}(2R_0-x)^{2p}}+\frac{R_0^{2p}|y+R_0|^{2p}}{(2R_0-x)^{4p}}\right)dydx\\
	&+C_p\int_{-\infty}^{-R_0}\int_{-3R_0+x}^{-R_0}\left(\frac{(2R_0^3)^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{R_0^{2p}}{(2R_0-x)^{2p}}\right)dydx\\
	&=I_{11}+I_{12}+I_{13}+I_{14}+I_{15}+I_{16}+I_{17}.
\end{align*}
Next we estimate $I_{11},I_{12},I_{13}$ and $I_{14}$. 
\begin{align*}
		I_{11}&=C_p\int_{-\infty}^{-R_0}\int_{-R_0}^{R_0}\left(\frac{(4R_0^2)^{2p}|x|^{2p}|y|^{2p}}{(R^2_0+x^2)^{4p}}+\frac{(2R_0^2)^{2p}}{(R_0^2+x^2)^{2p}}\right)dydx\\
				&\leq 	C_p2^{4p}R_0^{4p}\int_{-\infty}^{-R_0}\int_{-R_0}^{R_0}\left(\frac{|x|^{2p}|y|^{2p}}{(R^2_0+x^2)^{4p}}+\frac{1}{(R_0^2+x^2)^{2p}}\right)dydx\\
				&\leq 	C_p2^{4p+1}R_0^{4p+1}\int_{-\infty}^{-R_0}\frac{2}{(R^2_0+x^2)^{2p}}dx\\
				&\leq 	C_p2^{4p+1}R_0^{4p+1}\int_{-\infty}^{-R_0}\frac{2^{2p+1}}{(R_0-x)^{2p}}dx\quad(\mbox{since }2(R^2_0+x^2)\geq (R_0-x)^2)\\
				&=\frac{C_p}{2p-1}2^{6p+2}R_0^{2p+2}.
\end{align*}
Then, we analyze $I_{12}$.
\begin{align*}
	I_{12}&=C_p\int_{-\infty}^{-R_0}\int_{R_0}^{3R_0-x}\left(\frac{(2R_0^3)^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{(2R_0^3)^{2p}|3R_0-x-y|^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{4p}}\right)dydx\\
	&=C_p\int_{-\infty}^{-R_0}\left(\frac{(2R_0^3)^{2p}(2R_0-x)}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{1}{2p+1}\frac{(2R_0^3)^{2p}|2R_0-x|^{2p+1}}{(R_0^2+x^2)^{2p}(2R_0-x)^{4p}}\right)dx\\
	&\leq C_p(2R_0^3)^{2p}\int_{-\infty}^{-R_0}\frac{2}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p-1}}dx\\
	&\leq C_p(2R_0^3)^{2p}\int_{-\infty}^{-R_0}\frac{2^{2p+1}}{(R_0-x)^{2p}}\frac{1}{(3R_0)^{2p-1}}dx\quad(\mbox{since }2(R^2_0+x^2)\geq (R_0-x)^2)\\
	&=\frac{C_p(2R_0^3)^{2p}\cdot 2^{2p+1}}{(2p-1)(2R_0)^{2p-1}(3R_0)^{2p-1}}.
\end{align*}
Now, we estimate $I_{13}$ as follows.
\begin{align*}
	I_{13}&=C_p\int_{-\infty}^{-R_0}\int_{R_0}^{3R_0-x}\left(\frac{(4R_0^3)^{2p}|x|^{2p}|3R_0-x-y|^{2p}}{(R_0^2+x^2)^{4p}(2R_0-x)^{2p}}+\frac{R_0^{2p}|y-R_0|^{2p}}{(2R_0-x)^{4p}}\right)dydx\\
	&=\frac{C_p}{2p+1}\int_{-\infty}^{-R_0}\left(\frac{(4R_0^3)^{2p}|x|^{2p}(2R_0-x)^{2p+1}}{(R_0^2+x^2)^{4p}(2R_0-x)^{2p}}+\frac{R_0^{2p}(2R_0-x)^{2p+1}}{(2R_0-x)^{4p}}\right)dx\\
&=\frac{C_p}{2p+1}\int_{-\infty}^{-R_0}\left(\frac{(4R_0^3)^{2p}|x|^{2p}(2R_0-x)}{(R_0^2+x^2)^{4p}}+\frac{R_0^{2p}}{(2R_0-x)^{2p-1}}\right)dx\\
&\leq \frac{C_p}{2p+1}\int_{-\infty}^{-R_0}\left(\frac{2(4R_0^3)^{2p}(R_0-x)}{(R_0^2+x^2)^{2p}}+\frac{R_0^{2p}}{(2R_0-x)^{2p-1}}\right)dx\\
&\leq \frac{C_p}{2p+1}\int_{-\infty}^{-R_0}\left(\frac{2^{2p+1}(4R_0^3)^{2p}}{(R_0-x)^{4p-1}}+\frac{R_0^{2p}}{(2R_0-x)^{2p-1}}\right)dx\quad(\mbox{since }2(R^2_0+x^2)\geq (R_0-x)^2)\\
&= \frac{C_p}{2p+1} \left(\frac{1}{4p-2}\frac{2^{2p+1}(4R_0^3)^{2p}}{(2R_0)^{4p-2}}+\frac{1}{2p-2}\frac{R_0^{2p}}{(3R_0)^{2p-2}}\right).
\end{align*}
Finally, we bound $I_{14}$ as follows.
\begin{align*}
	I_{14}&=C_p\int_{-\infty}^{-R_0}\int_{R_0}^{3R_0-x}\left(\frac{(2R_0^3)^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{R_0^{2p}}{(2R_0-x)^{2p}}\right)dydx\\
	&=C_p\int_{-\infty}^{-R_0}\left(\frac{(2R_0^3)^{2p}(2R_0-x)}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p}}+\frac{R_0^{2p}(2R_0-x)}{(2R_0-x)^{2p}}\right)dx\\
		&=C_p\int_{-\infty}^{-R_0}\left(\frac{(2R_0^3)^{2p}}{(R_0^2+x^2)^{2p}(2R_0-x)^{2p-1}}+\frac{R_0^{2p}}{(2R_0-x)^{2p-1}}\right)dx\\
		&\leq C_p\int_{-\infty}^{-R_0}\left(\frac{2^{2p}(2R_0^3)^{2p}}{(R_0-x)^{2p}(3R_0)^{2p-1}}+\frac{R_0^{2p}}{(2R_0-x)^{2p-1}}\right)dx\\
		&=C_p\left(\frac{2^{2p}(2R_0^3)^{2p}}{(2p-1)(2R_0)^{2p-1}(3R_0)^{2p-1}}+\frac{R_0^{2p}}{(2p-2)(3R_0)^{2p-2}}\right).
\end{align*}
Note that here we have used the fact that $p>1$. The proof of bounds for $I_{15},I_{16}$ and $I_{17}$ follows by similar arguments as for $I_{12}$, $I_{13}$ and $I_{14}$ respectively. We omit the details here. This completes the proof of Theorem \ref{theorem-1}.

		For any $h\in \mathcal{S}(\mathbb{R}^2)$ there exists $u=(u_1,u_2)\in C^1(\mathbb{R}^2)$ such that

\label{prop:discont}
	Let $K\subset\mathbb{R}^d$ with $d\geq2$ be a convex compact set. Then there exists $u\in \dot{W}^{1,dp}(\mathbb{R}^d)\cap L^\infty(\mathbb{R}^d)$ such that
	\begin{equation}
		\det(\nabla u)=\mathbbm{1}_{K}.
	\end{equation}

<!-- proof-3: body_md -->
Define Minkowski functional $\mu_k:\mathbb{R}^d\rightarrow[0,\infty)$ as follows
	\begin{equation}
		\mu_K(x)=\inf\left\{t>0;\,t^{-1}x\in K\right\}.
	\end{equation}
	Observe that for any $x\in K$ we have $\mu_K(x)\leq1$ and when $x\in K^c$ it follows $\mu_K(x)>1$. Furthermore, Lipschitz continuity of the map $x\mapsto\mu_K(x)$ follows. We note that $\{x\in\mathbb{R}^d;\,\mu_K(x)\leq1\}=K$. Consider $u:\mathbb{R}^d\rightarrow\mathbb{R}^d$ defined as follows
	\begin{equation}
		u^+(x)=\left\{\begin{array}{cl}
			x&\mbox{ for }\mu_K(x)<1,\\
			\frac{x}{\mu_K(x)}&\mbox{ for }\mu_K(x)>1.
		\end{array}\right.
	\end{equation}
	By a simple calculation, it follows
	\begin{equation}
		\nabla u^+(x)=\left\{\begin{array}{cl}
			I_d&\mbox{ for }\mu_K(x)\leq1,\\
			\frac{1}{\mu_K(x)}I_d-\frac{x\otimes \nabla\mu_K(x)}{\mu^2_K(x)}&\mbox{ for }\mu_K(x)>1.
		\end{array}\right.
	\end{equation}
	To conclude about $\det(\nabla u)$, we next study $\nabla \mu_K$. Note that $\mu_K(sx)=s\mu_K(x)$. We invoke Euler’s Homogeneous Function Theorem to conclude
	\begin{equation*}
		x\cdot \nabla\mu_K(x)=\mu_K(x)
	\end{equation*}
	Then, it follows $[x\otimes \nabla\mu_K(x)]x=(x\cdot \nabla\mu_K(x))x=\mu_K(x)x$. Therefore, $$x\in\mbox{Ker}\left(\mu_K(x)I_d-x\otimes \nabla\mu_K(x)\right.$$ Subsequently,
	\begin{equation}
		\det(\nabla u^+(x))=\left\{\begin{array}{cl}
			1&\mbox{ for }\mu_K(x)<1,\\
			0&\mbox{ for }\mu_K(x)>1.
		\end{array}\right.
	\end{equation}
	This completes the proof of Proposition \ref{prop:discont}.
