<!-- definition-1: body_md -->
A \textbf{Hom-Lie algebra} is a triple $(\mathfrak{g},[\cdot,\cdot],\alpha)$,
		where $\mathfrak{g}$ is a $\mathbb{K}$-vector space,
		$[\cdot,\cdot]:\mathfrak{g}\times\mathfrak{g}\to\mathfrak{g}$ is a skew-symmetric bilinear map,
		and $\alpha:\mathfrak{g}\to\mathfrak{g}$ is a linear map satisfying the Hom-Jacobi identity
		$$
		[\alpha(x),[y,z]]+[\alpha(y),[z,x]]+[\alpha(z),[x,y]]=0, \qquad \forall x,y,z\in\mathfrak{g}, 
		$$
		and
		\begin{equation}
			\alpha([x,y])=[\alpha(x),\alpha(y)], \qquad \forall x,y\in\mathfrak{g}. \label{Liemultiplicativity}
		\end{equation}
		Identity~\eqref{Liemultiplicativity} is called the
		\textbf{multiplicativity condition}. A Hom-Lie algebra is called
		\textbf{regular} if $\alpha$ is bijective.
<!-- end definition-1 -->
<!-- remark-1: body_md -->
We use the term non-multiplicative Hom-Lie algebra when the
		multiplicativity condition is not imposed.
<!-- end remark-1 -->
<!-- definition-2: body_md -->
Let $(\mathfrak{g},[\cdot,\cdot],\alpha)$ be a Hom-Lie algebra. A linear map $d:\mathfrak{g}\to\mathfrak{g}$
		is called an \textbf{$\alpha$-derivation} if $d\circ\alpha=\alpha\circ d$
		and
		$$
		d([x,y])=[d(x),\alpha(y)]+[\alpha(x),d(y)], \qquad \forall x,y\in\mathfrak{g}.
		$$
		The quadruple $(\mathfrak{g},[\cdot,\cdot],\alpha,d)$ is then called a
		\textbf{Hom-Lie algebra with an $\alpha$-derivation}.
<!-- end definition-2 -->
<!-- remark-2: body_md -->
This is the case $k=1$ of the $\alpha^k$-derivations introduced in
		\cite{AlphaKDerivations}.
<!-- end remark-2 -->
<!-- definition-3: body_md -->
\label{def:hom-ass-nonass}
		A \textbf{Hom-nonassociative algebra} is a triple $(A,\mu,\alpha)$,
		where $A$ is a $\mathbb{K}$-vector space, $\mu:A\times A\to A$ is a bilinear
		map, written $\mu(a,b)=ab$, and $\alpha:A\to A$ is a linear map such that
		\begin{equation}
		\alpha(ab)=\alpha(a)\alpha(b), \qquad \forall a,b\in A. \label{jiehemultiplicativity}
		\end{equation}
		Identity~\eqref{jiehemultiplicativity} is called the
		\textbf{multiplicativity condition}. A \textbf{Hom-associative algebra}
		is a Hom-nonassociative algebra satisfying the Hom-associativity identity
		$$
		\alpha(a)(bc)=(ab)\alpha(c), \qquad \forall a,b,c\in A. 
		$$
		It is called \textbf{regular} if $\alpha$ is bijective.
<!-- end definition-3 -->
<!-- remark-3: body_md -->
We use the term non-multiplicative Hom-nonassociative algebra when
		the multiplicativity condition is not imposed.
<!-- end remark-3 -->
<!-- proposition-1: body_md -->
\label{prop:semidirect-HNA}
		The triple $(A\oplus A,\star,\widetilde{\alpha})$ is a Hom-nonassociative algebra.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
The map $\widetilde{\alpha}$ is linear and $\star$ is bilinear by
		construction. To verify multiplicativity, let $(a,u),(b,v)\in A\oplus A$.
		Using multiplicativity of $\alpha$, we obtain
		\begin{align*}
			\widetilde{\alpha}((a,u)\star(b,v))
			&=\widetilde{\alpha}(ab,u\alpha(b)+\alpha(a)v)\\
			&=(\alpha(ab),\alpha(u\alpha(b)+\alpha(a)v))\\
			&=(\alpha(a)\alpha(b),\alpha(u)\alpha^2(b)+\alpha^2(a)\alpha(v))\\
			&=(\alpha(a),\alpha(u))\star(\alpha(b),\alpha(v))\\
			&=\widetilde{\alpha}(a,u)\star \widetilde{\alpha}(b,v).
		\end{align*}
		Therefore, $(A\oplus A,\star,\widetilde{\alpha})$ is a
		Hom-nonassociative algebra.
<!-- end proof-1 -->
<!-- proposition-2: body_md -->
\label{prop:HomLie-direct-sum}
		The triple $(\mathfrak{g}\oplus\mathfrak{g},[\cdot,\cdot]_{\mathfrak{g}\oplus\mathfrak{g}},\widetilde{\alpha})$ is a Hom-Lie algebra.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
The bracket is bilinear and skew-symmetric. For
		$(x,u),(y,v),(z,w)\in\mathfrak{g}\oplus\mathfrak{g}$, multiplicativity follows from
		\begin{align*}
			\widetilde{\alpha}([(x,u),(y,v)]_{\mathfrak{g}\oplus\mathfrak{g}})
			&=\bigl(\alpha([x,y]),\alpha([u,\alpha(y)]+[\alpha(x),v])\bigr)\\
			&=\bigl([\alpha(x),\alpha(y)],
			[\alpha(u),\alpha^2(y)]+[\alpha^2(x),\alpha(v)]\bigr)\\
			&=[\widetilde{\alpha}(x,u),\widetilde{\alpha}(y,v)]_{\mathfrak{g}\oplus\mathfrak{g}}.
		\end{align*}
		For the Hom-Jacobi identity, we compute
		\begin{align*}
			&[\widetilde{\alpha}(x,u),[(y,v),(z,w)]_{\mathfrak{g}\oplus\mathfrak{g}}]_{\mathfrak{g}\oplus\mathfrak{g}}
			+[\widetilde{\alpha}(y,v),[(z,w),(x,u)]_{\mathfrak{g}\oplus\mathfrak{g}}]_{\mathfrak{g}\oplus\mathfrak{g}}\\
			&\quad
			+[\widetilde{\alpha}(z,w),[(x,u),(y,v)]_{\mathfrak{g}\oplus\mathfrak{g}}]_{\mathfrak{g}\oplus\mathfrak{g}}\\
			&=
			\left(
			\begin{array}{@{}l@{}}
				[\alpha(x),[y,z]]
				+[\alpha(y),[z,x]]
				+[\alpha(z),[x,y]],\\[2pt]
				\begin{aligned}
					&[\alpha(u),[\alpha(y),\alpha(z)]]
					+[\alpha^2(x),[v,\alpha(z)]]
					+[\alpha^2(x),[\alpha(y),w]]\\
					&+[\alpha(v),[\alpha(z),\alpha(x)]]
					+[\alpha^2(y),[w,\alpha(x)]]
					+[\alpha^2(y),[\alpha(z),u]]\\
					&+[\alpha(w),[\alpha(x),\alpha(y)]]
					+[\alpha^2(z),[u,\alpha(y)]]
					+[\alpha^2(z),[\alpha(x),v]]
				\end{aligned}
			\end{array}
			\right)\\
			&=
			\left(
			\begin{array}{@{}l@{}}
				0,\\[2pt]
				\begin{aligned}
					&\bigl([\alpha(u),[\alpha(y),\alpha(z)]]
					+[\alpha^2(y),[\alpha(z),u]]
					+[\alpha^2(z),[u,\alpha(y)]]\bigr)\\
					&+\bigl([\alpha(v),[\alpha(z),\alpha(x)]]
					+[\alpha^2(z),[\alpha(x),v]]
					+[\alpha^2(x),[v,\alpha(z)]]\bigr)\\
					&+\bigl([\alpha(w),[\alpha(x),\alpha(y)]]
					+[\alpha^2(x),[\alpha(y),w]]
					+[\alpha^2(y),[w,\alpha(x)]]\bigr)
				\end{aligned}
			\end{array}
			\right)\\
			&=(0,0).
		\end{align*}
		Here the first component and each parenthesized expression in the second
		component vanish by the Hom-Jacobi identity in $\mathfrak{g}$. Therefore,
		$(\mathfrak{g}\oplus\mathfrak{g},[\cdot,\cdot]_{\mathfrak{g}\oplus\mathfrak{g}},\widetilde{\alpha})$ is a
		Hom-Lie algebra.

<!-- proposition-3: body_md -->
\label{diyigelizi}
		Let $(\mathfrak{g},[\cdot,\cdot]_0)$ be a Lie algebra, let $\delta:\mathfrak{g}\to\mathfrak{g}$
		be a derivation, and let $\alpha:\mathfrak{g}\to\mathfrak{g}$
		be a Lie algebra endomorphism such that
		$\alpha\circ\delta=\delta\circ\alpha$.
		Then $(\mathfrak{g},\alpha\circ[\cdot,\cdot]_0,\alpha)$ is a Hom-Lie algebra,
		and $\alpha\circ\delta$ is an $\alpha$-derivation of this algebra.
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
Set $[\cdot,\cdot]_\alpha:=\alpha\circ[\cdot,\cdot]_0$ and
		$d:=\alpha\circ\delta$. By \cite[Corollary~2.5]{HomAlgebraTwists},
		$(\mathfrak{g},[\cdot,\cdot]_\alpha,\alpha)$ is a Hom-Lie algebra. The relation
		$\alpha\delta=\delta\alpha$ gives
		$d\alpha=\alpha\delta\alpha=\alpha^2\delta=\alpha d$.
		Since $\delta$ is a derivation of $(\mathfrak{g},[\cdot,\cdot]_0)$, for
		$x,y\in\mathfrak{g}$ we obtain
		\begin{align*}
			d([x,y]_\alpha)
			&=
			\alpha\delta(\alpha([x,y]_0))
			=
			\alpha^2\delta([x,y]_0)\\
			&= \alpha^2([\delta(x),y]_0+[x,\delta(y)]_0)\\
			&= \alpha([\alpha\delta(x),\alpha(y)]_0+[\alpha(x),\alpha\delta(y)]_0)\\
			&= [\alpha\delta(x),\alpha(y)]_\alpha+[\alpha(x),\alpha\delta(y)]_\alpha\\
			&= [d(x),\alpha(y)]_\alpha+[\alpha(x),d(y)]_\alpha.
		\end{align*}
		Thus $(\mathfrak{g},[\cdot,\cdot]_\alpha,\alpha,d)$ is a Hom-Lie algebra with an
		$\alpha$-derivation.

<!-- proposition-5: body_md -->
\label{prop:commutator}
		The quadruple $(A,[\cdot,\cdot]_A,\alpha,D)$ is an object of $\mathbf{HomLieDer}_\alpha$.

<!-- proof-5: body_md -->
By \cite[Proposition~1.6]{MakhloufSilvestrov},
		$(A,[\cdot,\cdot]_A,\alpha)$ is a Hom-Lie algebra. It remains to verify
		that $D$ is an $\alpha$-derivation of the commutator bracket. The relation
		$D\alpha=\alpha D$ holds because $(A,\mu,\alpha,D)$ is an object of
		$\mathbf{HomAsDer}_\alpha$. For any $a,b\in A$, the twisted Leibniz rule gives
		\begin{align*}
			D([a,b]_A)
			&=D(ab-ba)\\
			&=D(ab)-D(ba)\\
			&=D(a)\alpha(b)+\alpha(a)D(b)
			-D(b)\alpha(a)-\alpha(b)D(a)\\
			&=(D(a)\alpha(b)-\alpha(b)D(a))+(\alpha(a)D(b)-D(b)\alpha(a))\\
			&=[D(a),\alpha(b)]_A+[\alpha(a),D(b)]_A.
		\end{align*}
		Therefore, $(A,[\cdot,\cdot]_A,\alpha,D)$ is an object of
		$\mathbf{HomLieDer}_\alpha$.

<!-- lemma-1: body_md -->
\label{lem:multiplicativity-quotient}
		With the notation above, $(\mathcal{F}_{\mathrm{m}}(V),\mu_{\mathrm{m}},\alpha_{\mathrm{m}})$
		is a Hom-nonassociative algebra.
<!-- end lemma-1 -->
<!-- proof-7: body_md -->
First, we show that $M_\infty$ is a two-sided ideal satisfying
		$\alpha_F(M_\infty)\subseteq M_\infty$.
		Since $M_1$ is a two-sided ideal and $M_{n+1}$ is the two-sided ideal generated by
		$M_n\cup\alpha_F(M_n)$, each $M_n$ is a two-sided ideal and
		$M_n\subseteq M_{n+1}$. Hence $M_\infty=\bigcup_{n\ge 1}M_n$ is a two-sided ideal. For $x\in

		$$

		$$
		Thus $\alpha_F(M_\infty)\subseteq M_\infty$, so

		\end{align*}
		Thus $\Theta_1$ is a morphism of Hom-nonassociative algebras.
		Since $D_A$ is an $\alpha_A$-derivation of $A$, we also have
		\begin{align*}
		\Theta_2(uv)
		&=(G(uv),D_A(G(uv)))\\
		&=(G(u)G(v),D_A(G(u))\alpha_A(G(v))+\alpha_A(G(u))D_A(G(v)))\\
		&=\Theta_2(u)\star\Theta_2(v),
		\end{align*}

		\Theta_2(\alpha_{\mathrm{m}}(u))
		&=(G(\alpha_{\mathrm{m}}(u)),D_A(G(\alpha_{\mathrm{m}}(u))))\\
		&=(\alpha_A(G(u)),\alpha_A(D_A(G(u))))\\
		&=\widetilde\alpha_A(\Theta_2(u)).
		\end{align*}
		Hence $\Theta_2$ is a morphism of Hom-nonassociative algebras.
		The two morphisms agree on the generators: for every $x\in \mathfrak{g}$,
		$$
		\Theta_1(j(x))=(f(x),f(d(x)))=\Theta_2(j(x)),
		$$
		because $f\circ d=D_A\circ f$. By uniqueness in the universal property of
		$\mathcal{F}_{\mathrm{m}}(\mathfrak{g})$, we obtain $\Theta_1=\Theta_2$. Hence $G\circ\widehat d=D_A\circ G$. For

		$$
		h(D_U(\bar u))=h(\pi_I(\widehat d(u)))=G(\widehat d(u))=D_A(G(u))=D_A(h(\bar u)).
		$$
		Therefore, $h\circ D_U=D_A\circ h$, so $h$ is a morphism in
		$\mathbf{HomAsDer}_\alpha$.

		$\mathbf{HomAsDer}_\alpha$ satisfying $h'\circ \iota_U=f$. Then
		$h'\circ\pi_I:\mathcal{F}_{\mathrm{m}}(\mathfrak{g})\to A$ is a morphism of
		Hom-nonassociative algebras extending $f$. By the uniqueness of $G$,
		$h'\circ\pi_I=G=h\circ\pi_I$. Since $\pi_I$ is surjective, $h'=h$.
		
		To prove uniqueness up to isomorphism, let
		$(\mathcal{U}_{\mathrm{m}}'(\mathfrak{g}),\mu'_U,\alpha'_U,D'_U)$ be another
		universal enveloping algebra of
		$(\mathfrak{g},[\cdot,\cdot],\alpha,d)$, with canonical morphism
		$\iota'_U:\mathfrak{g}\to\mathcal{U}_{\mathrm{m}}'(\mathfrak{g})$. The two universal properties give
		morphisms $\Phi:\mathcal{U}_{\mathrm{m}}(\mathfrak{g})\to\mathcal{U}_{\mathrm{m}}'(\mathfrak{g})$ and
		$\Psi:\mathcal{U}_{\mathrm{m}}'(\mathfrak{g})\to\mathcal{U}_{\mathrm{m}}(\mathfrak{g})$ in $\mathbf{HomAsDer}_\alpha$ such that
		$\Phi\circ\iota_U=\iota'_U$ and $\Psi\circ\iota'_U=\iota_U$. Hence
		$$
		(\Psi\circ\Phi)\circ\iota_U=\iota_U,
		\qquad
		(\Phi\circ\Psi)\circ\iota'_U=\iota'_U.
		$$
		By uniqueness in the respective universal properties,
		$$
		\Psi\circ\Phi=\mathrm{id}_{\mathcal{U}_{\mathrm{m}}(\mathfrak{g})},
		\qquad
		\Phi\circ\Psi=\mathrm{id}_{\mathcal{U}_{\mathrm{m}}'(\mathfrak{g})}.
		$$
		Therefore, $\Phi$ is an isomorphism in $\mathbf{HomAsDer}_\alpha$.

<!-- proposition-14: body_md -->
\label{prop:reg-lie}
		Let $(\mathfrak{g},[\cdot,\cdot],\alpha,d)$ be a regular Hom-Lie algebra with an
		$\alpha$-derivation $d$. Then $(\mathfrak{g},\alpha^{-1}\circ[\cdot,\cdot])$
		is a Lie algebra, and $\alpha^{-1}\circ d$ is a derivation of this algebra.
<!-- end proposition-14 -->
<!-- proof-16: body_md -->
Set $[x,y]_0:=\alpha^{-1}([x,y])$ and $\delta:=\alpha^{-1}\circ d$.
		The bracket $[\cdot,\cdot]_0$ is skew-symmetric because
		$[\cdot,\cdot]$ is skew-symmetric. Since $\alpha$ is multiplicative and
		bijective, we have
		$$
		[x,\alpha^{-1}(u)]=\alpha^{-1}([\alpha(x),u])
		\qquad \forall x,u\in\mathfrak{g}.
		$$
		Using this identity and the Hom-Jacobi identity, we obtain, for all $x,y,z\in\mathfrak{g}$,
		\begin{align*}
		&[x,[y,z]_0]_0+[y,[z,x]_0]_0+[z,[x,y]_0]_0\\
		&\quad=\alpha^{-2}\bigl([\alpha(x),[y,z]]+[\alpha(y),[z,x]]+[\alpha(z),[x,y]]\bigr)=0.
		\end{align*}
		Thus $(\mathfrak{g},[\cdot,\cdot]_0)$ is a Lie algebra.
		
		To show that $\delta$ is a derivation, let $x,y\in\mathfrak{g}$. Then
		\begin{align*}
			\delta([x,y]_0)
			&=
			\alpha^{-1}d(\alpha^{-1}([x,y]))\\
			&=
			\alpha^{-2}d([x,y])
			\qquad (\text{since } d\alpha=\alpha d)\\
			&=
			\alpha^{-2}([d(x),\alpha(y)]+[\alpha(x),d(y)])\\
			&=
			\alpha^{-1}([\alpha^{-1}d(x),y])+\alpha^{-1}([x,\alpha^{-1}d(y)])\\
			&=
			[\delta(x),y]_0+[x,\delta(y)]_0.
		\end{align*}
		Thus $\delta$ is a derivation, as required.

\label{cor:regular-pbw}
		Let $(\mathfrak{g},[\cdot,\cdot],\alpha,d)$ be an object of
		$\mathbf{HomLieDer}_\alpha$ with $\alpha$ bijective. Then the filtration
		$\{F_p\mathcal{U}_{\mathrm{m}}(\mathfrak{g})\}_{p\ge 0}$ is preserved by $\alpha_U$,
		$\Delta_U$, and $D_U$. With respect to the untwisted product $\ast$,
		there is an isomorphism of graded associative algebras
		$$
		\operatorname{gr}\big(\mathcal{U}_{\mathrm{m}}(\mathfrak{g}),\ast\big)
		\cong
		S^+(\mathfrak{g}_0).
		$$
		If $\mathfrak{g}$ is finite-dimensional and $x_1<\cdots<x_n$ is an ordered basis
		of its underlying vector space, then the monomials
		$$
		\iota_U(x_{i_1})\ast\cdots\ast\iota_U(x_{i_r})
		\quad
		(r\ge 1,\ 1\le i_1\le\cdots\le i_r\le n)
		$$
		form a basis of $\mathcal{U}_{\mathrm{m}}(\mathfrak{g})$.

<!-- proof-21: body_md -->
By Theorem~\ref{thm:regular-comparison}, $\Phi$ induces an isomorphism of
		associative algebras with derivations
		$$
		\Phi^\sharp:
		(\mathcal{U}_{\mathrm{m}}(\mathfrak{g}),\ast,\Delta_U)
		\longrightarrow
		(\mathcal{U}^+(\mathfrak{g}_0),\widetilde{\delta}),
		$$
		with $\Phi^\sharp\circ\iota_U=\iota_0$. By
		Lemma~\ref{thm:pbw-lie},
		$$
		\operatorname{gr}\mathcal{U}^+(\mathfrak{g}_0)\cong S^+(\mathfrak{g}_0).
		$$
		Transporting the filtration along $\Phi^\sharp$ therefore gives
		$$
		\operatorname{gr}\big(\mathcal{U}_{\mathrm{m}}(\mathfrak{g}),\ast\big)
		\cong S^+(\mathfrak{g}_0).
		$$
		If $\mathfrak{g}$ is finite-dimensional, $(\Phi^\sharp)^{-1}$ sends the ordered PBW
		monomials in $\mathcal{U}^+(\mathfrak{g}_0)$ to the stated monomials formed with $\ast$. Hence the
		latter form a basis of $\mathcal{U}_{\mathrm{m}}(\mathfrak{g})$.

		It remains to prove that the three maps preserve the filtration.
		The map $\widetilde{\alpha}$ sends a product of $r$ generators to a product
		of $r$ generators and therefore preserves the filtration of
		$\mathcal{U}^+(\mathfrak{g}_0)$. The Leibniz rule shows that $\widetilde{\delta}$ preserves
		the same filtration. Under $\Phi^\sharp$, these maps correspond to
		$\alpha_U$ and $\Delta_U$, respectively. Hence $\alpha_U$ and $\Delta_U$
		preserve the filtration of $\mathcal{U}_{\mathrm{m}}(\mathfrak{g})$. Since $D_U=\alpha_U\circ\Delta_U$, $D_U$
		preserves the filtration as well.

<!-- remark-8: body_md -->
\label{rem:pbw-comparison}
		Every involutive Hom-Lie algebra is regular because
		$\alpha^2=\mathrm{id}$ implies that $\alpha$ is bijective. Hence
		Theorem~\ref{cor:regular-pbw} applies to the involutive Hom-Lie
		algebras considered in \cite{GuoZhangZheng} when equipped with
		$\alpha$-derivations. The bijectivity assumption is required for the
		untwisting used in the theorem.
