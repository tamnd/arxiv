<!-- lemma-1: body_md -->
\label{lem:numberofprefsforeachcar}
Given a permutation $\pi=\pi_1\cdots\pi_n\in\mathfrak{S}_n$, define
\[\ell(\pi_i)=\max\{k: \, \pi_j \leq \pi_i \text{ for all } i-k+1\leq j\leq i\}.\]
The number of parking functions with parking permutation $\pi$ is 
\[\prod_{i=1}^n \ell(\pi_i).\]
<!-- end lemma-1 -->
<!-- example-1: body_md -->
\label{ex: count from 23154}
Let $n=5$ and $\pi=23154$ be the permutation that parks the cars on the street. Then, $\ell(\pi_1)=1, \ell(\pi_2)=2, \ell(\pi_3)=1, \ell(\pi_4)=4$ and $\ell(\pi_5)=1$. Moreover, the product of $1\cdot 2 \cdot 1 \cdot 4 \cdot 1=8$ is the total number of parking functions with parking permutation $\pi=23154$.
<!-- end example-1 -->
<!-- definition-1: body_md -->
\label{def:pvector}
Let $\pi=\pi_1\pi_2\cdots\pi_n\in\mathfrak{S}_n$. For all $i\in[n]$, define the sequence $p=(p_1,p_2,\ldots,p_n)\in [n]^n$ by
\[p_i=\ell(\pi_i)=\max\{k: \, \pi_j \leq \pi_i \text{ for all } i-k+1\leq j\leq i\}.\] 
We call $p$ the \emph{parking table} and often write $p$ in one-line notation.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:reduction of perm}
Let $[k]^n$ denote the set of words of length $n$ in the alphabet $\{1,2,\ldots,k\}$. Let $w\in [k]^n$ and $w=w_1w_2\cdots w_n$. The reduction (or standardization) of $w$, denoted $\operatorname{red}(w)$, is the unique word of length $n$ obtained by replacing the $i^{th}$ smallest distinct letter of $w$ with $i$, for each $i$. We refer to $\operatorname{red}(w)$ as the \emph{reduced permutation}.
<!-- end definition-2 -->
<!-- example-2: body_md -->
\label{ex:reductionex1}
If $\pi=23154$, deleting the entry $\pi_3=1$ yields the word $2354$, whose reduced permutation is $\operatorname{red}(2354)=1243$.
<!-- end example-2 -->
<!-- definition-3: body_md -->
Let $\pi=\pi_1\pi_2\cdots\pi_n\in \mathfrak{S}_n$ and let $\sigma=\sigma_1\sigma_2\cdots\sigma_k\in \mathfrak{S}_k$. We say that $\pi$ contains $\sigma$ if there exist $1\leq i_1<i_2<\cdots <i_k\leq n$ so that $\operatorname{red}(\pi_{i_1}\pi_{i_2}\cdots\pi_{i_k})=\sigma$. Otherwise $\pi$ avoids $\sigma$.
<!-- end definition-3 -->
<!-- claim-1: body_md -->
\label{claim:reduavoid}
Let $k\leq n \in \mathbb{N}$ and $\Pi$ be a collection of patterns. If $\pi\in \operatorname{Av}_n(\Pi)$ then $\operatorname{red}(w)\in \operatorname{Av}_{k}(\Pi)$ for any subsequence $w$ of $\pi$ of length $k$.
<!-- end claim-1 -->
<!-- claim-2: body_md -->
\label{lem:parkingtables}
Let $w\in[k]^n$ with distinct elements and $\sigma\in \mathfrak{S}_n$. If $\sigma=\operatorname{red}(w)$ for a word $w$ with distinct elements then the parking tables of $w$ and $\sigma$ coincide.
<!-- end claim-2 -->
<!-- example-3: body_md -->
The parking table of the word $2354$ is $p=(1,2,3,1)$ which is the same parking table of its reduced permutation $1243$.
<!-- end example-3 -->
<!-- definition-4: body_md -->
\label{def:suffix}
Let $\pi=\pi_1\cdots \pi_n\in \mathfrak{S}_n$ be a parking permutation and $p=p_1p_2\cdots p_n\in[n]^n$ be its parking table. For $1\leq  k \leq j \leq n$, we call a pair $(\tau, p')$ a \emph{bi-suffix} of $\pi$ if \[\tau=\pi_k\pi_{k+1}\cdots \pi_j \in[n]^{j-k+1},\] and \[p'=p_kp_{k+1}\cdots p_j \in [n]^{j-k+1}.\]
<!-- end definition-4 -->
<!-- example-4: body_md -->
\label{ex:suffix1}
Let $\pi=2471635 \in \mathfrak{S}_7$ and $k=4,j=7$. The parking table of $\pi$ is given by $p=1231212$. Then the bi-suffix of $\pi$ is the tuple $(\tau,p)$ where $\tau=1635$ and $p=1212$.

<!-- lemma-2: body_md -->
Given $\pi=\pi_1\cdots\pi_n\in \mathfrak{S}_n$ with bi-suffix $(i,r)$, where $\pi_n=i$ and $\ell(\pi_n=i)=r$. Let $w\in[n]^{n-1}$ be the word obtained by deleting $\pi_n=i$ and $\operatorname{red}(w)=\tau=\tau_1\cdots\tau_{n-1}\in \mathfrak{S}_{n-1}$.
Then
\[ \prod_{i=1}^n \ell(\pi_i)=r\prod_{i=1}^{n-1}\ell(\tau_i).\]
In other words, the number of parking functions with parking permutation $\pi$ is the same as the product of $r$ and the number of parking functions with parking permutation $\tau$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
This follows by applying Lemma~\ref{lem:parkingtables} to $w=\pi_1\pi_2\cdots\pi_{n-1}$ and $\operatorname{red}(w)=\tau$. Since the parking tables of both $w$ and $\tau$ are the same, then 
\[\prod_{i=1}^{n-1}\ell(\tau_i)=\prod_{i=1}^{n-1} w_i= \prod_{i=1}^{n-1} \ell(\pi_i).\]
Thus, \[ \prod_{i=1}^n \ell(\pi_i)=\ell(\pi_n)\prod_{i=1}^{n-1}\ell(\tau_i)=r\prod_{i=1}^{n-1}\ell(\tau_i).\]

<!-- definition-5: body_md -->
Let 
$\Pi=\{\sigma_1,\sigma_2,\ldots, \sigma_k\}$ be collection of patterns. For $m\geq 1$, let $B_n^{(m)}(\Pi)$ denote the set of all bi-suffixes $(\tau, p')$, with $\tau, p'\in [n]^m$, that arise from permutations $\pi \in \operatorname{Av}_n(\Pi)$ and its corresponding parking table $p\in[n]^n$. That is,  
\[ 
B_{n}^{(m)}(\Pi)=\{(\tau, p'): \, (\tau, p') \text{ is a bi-suffix of } \pi \in \operatorname{Av}_n(\Pi) \text{ with parking table } p \}.
\]
We refer to $\tau$ and $p'$ as the permutation component and parking table component, respectively.

For each $(\tau,p')\in B_{n}^{(m)}(\Pi)$, we define
\[
B_{n}^{(m+1)}(\Pi;\, (\tau,p'))
=
\left\{
(\mu,q):
\begin{array}{l}
(\mu,q)\text{ is a bi-suffix of }\pi\in\operatorname{Av}_n(\Pi)
\text{ with parking table }p,\\
\mu_2\cdots\mu_{m+1}=\tau
\text{ and }
q_2\cdots q_{m+1}=p'
\end{array}
\right\}.
\]

\textit{Warning:} We abuse notation whenever $\tau=i$ and $p'=r$ are length $1$ and simply write $(i,r)$ instead of $((i), (r))$.

<!-- proposition-1: body_md -->
\label{prop:123recurrence}
Let $a(n)$ be the number of $123$-avoiding permutations and $a(n,i)$ be those permutations that end with the element $i$. Then
\[a(n,i)=
\begin{cases}
    a(n-1) & \text{if } i=1 \text{ or } i=2 \\
    a(n-1,i-1)+\sum_{j=i}^{n-1} a(n-1,j) & \text{otherwise}
\end{cases}
\]
with initial conditions $a(1,1)=1$ and $a(1,i)=0$ if $i\geq 2$.

<!-- lemma-3: body_md -->
The set of bi-suffixes in $B_{n}^{(1)}(123)$ is given by
\[
B_{n}^{(1)}(123)=\{(i,r): \, 1\leq r \leq i \leq n-1\} \cup \{(n,n)\}.
\]
In other words, if a $123$-avoiding parking permutation ends in $i<n$, then the corresponding parking table can end in any $r\in [i]$, while if it ends in $n$, the parking table necessarily ends in $n$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For any permutation $\pi\in \operatorname{Av}_n(123)$, the last element $\pi_n=i$ can range from any value from $1$ to $n$.

If $\pi_n=n$ then all elements to the left of $\pi_n$ are less than $n$ so $\ell(\pi_n)=p_n=n$ in the parking table. Hence, $(n,n)\in B_{n}^{(1)}(123)$. 
If $\pi_n=i$ and $i<n$, by definition, if $\ell(\pi_n)=r$ then there are $r-1$ elements to the left of $\pi_n$ that are all at most $i$. Since there are only $i$ elements of $[n]$ that are at most $i$, we have $r\leq i$. Thus, $(i,r)\in B_{n}^{(1)}(123)$ for $1\leq r\leq i \leq n-1$.

We now assume $ 1\leq r\leq i\leq n-1$ and construct a $123$-avoiding permutation whose last element $\pi_n=i$ and whose last entry in the parking table $p_n=r$.
Consider 
\[\pi= (i-1)(i-2)\cdots r n (n-1)\cdots (i+1)(r-1)(r-2)\cdots 1 i.\]
Notice that $\pi$ can be decomposed into four blocks. The first block $B_1$ contains the elements $r, r+1, \ldots , i-2, i-1$. If $i=r$, then $B_1$ is an empty block, otherwise, if $r<i$ then there are $i-r$ elements in block $B_1$ in decreasing order.
The second block $B_2$ contains the elements $i+1,i+2,\ldots, n-1, n$ in decreasing order, and there are $n-i$ of them. The third block $B_3$ contains the elements $1,2,\ldots, r-2,r-1$ in decreasing order and there are $r-1$ of them. The last block $B_4$ is the last element $i$. Altogether, there are $n$ elements in the permutation $\pi$.

Furthermore, $\ell(\pi_n)=r$ because the first element $j$ to the left of $\pi_n=i$ such that $j>i$ is at position $n-r$. Mainly, $\pi_{n-r}=i+1$ and $\pi_{n-r}>\pi_n=i$. For any $n-r+1\leq k \leq n-1 $, $\pi_k \leq r-1 < i$ and so $p_n=\ell(\pi_n)=r$.
Moreover, $\pi$ avoids the pattern $123$. Indeed, elements in blocks $B_1$, $B_2$ and $B_3$ are decreasing. Also, for any $b_1\in B_1, b_2\in B_2$ and $b_3\in B_3$, $b_3<b_1<b_2$ and $b_1<i<b_2$. Thus, there are no increasing subsequences as $b_3<b_1<i<b_2$.

<!-- lemma-4: body_md -->
\label{lem:lengthtwopairssuffixes}
Given $(i,r)\in B_{n}^{(1)}(123)$, the set of pairs $(j,s)$ for which
\[((j,i),(s,r)) \in B_{n}^{(2)}(123;\, (i,r))\]
is characterized as follows:
\[
\{(j,s): \, ((j,i),(s,r))\in B_{n}^{(2)}(123;\, (i,r))\}=\begin{cases}
   \{(j,s): \,  1\leq s <j, i< j <n   \} \cup \{(n,n-1)\} & \text{if } r=1\\
   \{(1,1)\} & \text{otherwise.}
\end{cases}
\]

<!-- proof-3: body_md -->
We first prove that if $(i,r)\in B_{n}^{(1)}(123)$ then $(\tau, p')\in B_{n}^{(2)}(123;\, (i,r) )$ is of the desired form. 

Let $\pi=\pi_1\pi_2\cdots \pi_{n-2}ji\in \operatorname{Av}_n(123)$ and $p=(p_1,p_2,\ldots, p_{n-2},s,r)$ be its parking table.
By the suffix scheme from Section~\ref{subsec:123perm}, if $\pi$ ends with $ji$ then either $j>i$ or $j<i$. 

If $j>i$ then $\ell(\pi_n=i)=1$ and so $r=1$ in the parking table $p$. We have the following cases:
\begin{enumerate}
    \item Suppose that $j=n$. Then, $\pi=\pi_1\pi_2\cdots \pi_{n-2}ni$ and $s=\ell(\pi_{n-1}=n)=n-1$ since for all $1\leq k \leq n-2$, $\pi_k<n$. Thus, $\{( (n,i),(n-1,1))\}\in B_{n}^{(2)}(123; \, (i,r))$ whenever $r=1$ and $\pi_{n-1}=n,\pi_n=i$.
    \item If $j<n$ then $i+1\leq j\leq n-1$ as $j>i$. We show that $1\leq s=\ell(\pi_{n-1}=j)\leq j-1$. If $s=\ell(\pi_{n-1}=j)$ then the only possible elements $S$ contiguously to the left of $\pi_{n-1}=j$ such that $1\leq s \leq j-1$ are if $S=\{1,2,\ldots, i-1\}\cup \{i+1,i+2,\ldots, j-1\}$, where $|S|=j-2$. Hence, $1\leq s \leq j-2+1=j-1$. It follows that $\{(\tau, p'): \,  1\leq s \leq j-1, i+1\leq j \leq n-1   \} \in B_{n}^{(2)}(123;\, (i,r))$ whenever $r=1$ and $\pi_{n-1}<n, \pi_n=i$. 
\end{enumerate}

If $j<i$ then $r=\ell(\pi_n=i)>1$. But $j=1$ by the suffix scheme from Section~\ref{subsec:123perm}. Therefore, $\pi_{n-2}>\pi_{n-1}=1$. It follows that $s=\ell(\pi_{n-1}=1)=1$. Thus, $\{((1,i), (1, r)\} \in B_{n}^{(2)}(123;\, (i,r))$ whenever $\pi_{n-1}=j,\pi_n=i$ with $j<i$ and $p_n=r>1$.

Now, we construct a permutation $\pi$ such that its bi-suffix is an element of $B_{n}^{(2)}(123;\, (i,r))$ and $\pi\in \operatorname{Av}_n(123)$. If $r=1$ then construct $\pi$ such that 
\[ \pi=\underbrace{(n-1)(n-2)\cdots (i+1)}_{n-i-1}\underbrace{(i-1)(i-2)\cdots 321}_{i-1}ni \]
or
\[ \pi=\underbrace{n(n-1)\cdots (j+2)(j+1)}_{n-j}\underbrace{(j-1)(j-2)\cdots (i+1)}_{j-i-1}\underbrace{(i-1)(i-2)\cdots 321}_{i-1}ji\]
with $j>i$. Both permutations are $123$-avoiding and both satisfy the condition that if $\pi_{n-1}=j$ then either $j=n$ or $i<j<n$ with $s=\ell(\pi_{n-1})=n-1$ or $1\leq s=\ell(\pi_{n-1}=j)\leq j-2+1=j-1$, respectively.

$S=(i,r)\in B_{n}^{(1)}(123)$, let $a(n,S)$ denote the number of parking functions whose parking permutations $\pi$ have bi-suffix $S$. Then
\[
a(n)=\sum_{S\in B_{n}^{(1)}(123)} a(n,S),

\]

\[
a(n,S)
=
\sum_{\substack{(j,s)\text{ such that}\\
((j,i),(s,r))\in
B_{n}^{(2)}(123;\, (i,r))}}
a(n,S,S').

Furthermore,
\[a(n,S,S')=a(n,(i,r),(j,s))=
\begin{cases}
a(n-1,(j-1,s)),

<!-- lemma-5: body_md -->
\label{lemm:item1-r1}
Let $\pi\in \operatorname{Av}_n(1234,2134)$ and $\pi_{n-1}< \pi_n$. 
Let $ k \in [n-2]$ and $\pi_k=1$. If $\pi_{n-1}=2$ and $\pi_n=n$ then
\[ (1, \pi_{k+1},\pi_{k+2}, \ldots, \pi_{n-2},2,n)=(1,\underbrace{r_1,r_1-1\ldots, 3}_{r_1-2},2,n),\]

<!-- proof-5: body_md -->

Let $\Pi=\{1234,2134\}$ and $\pi\in \operatorname{Av}_n(\Pi)$ such that $\pi_{n-1}=j,\pi_n=i$ and $p=(p_1,p_2,\ldots, p_n)$ is the parking table of $\pi$ such that and $p_n=r$. 
We prove the recurrence via the following cases on the value of $r$:
\begin{enumerate}
    \item If $r=1$: In this case, $\pi_{n-1}>\pi_{n}$ and so $\pi_{n-1},\pi_n$ reduce to $2,1$. Hence, $i+1\leq j \leq n$. By the permutation scheme, delete $\pi_n=i$ from $\pi$ and apply the reduction map to obtain a permutation of length $n-1$. The resulting word after deletion is $w=\pi_1\pi_2\cdots \pi_{n-2}j$ and reducing $w$ to a permutation of length $n-1$ results in a permutation whose last element is $j-1$. By applying Lemma~\ref{ex:suffix1} to the reduced permutation, we obtain
    \[  \operatorname{pk}(n,i,r) = \sum_{j=i}^{n-1} \operatorname{pk}(n-1,j).\]
    \item If $r=2$: In this case, if $r=2$ then $j=\pi_{n-1}<\pi_n=i$ and $\pi_n<\pi_{n-2}$. Hence, we proceed with the cases $\pi_n=2$ or $\pi_n\neq 2$ from the permutation scheme.
    \begin{enumerate}
        \item If $\pi_n=2$ then $\pi_{n-1}=1$ and the parking table is $p=(p_1,p_2,\ldots, 1,2)$. By the permutation scheme, delete $\pi_{n-1}=1$ from $\pi$ and apply the reduction map to obtain a permutation of length $n-1$. The resulting word after deletion is $w=\pi_1\pi_2\cdots \pi_{n-2}2$ and reducing $w$ to a permutation of length $n-1$ results in $\operatorname{red}(w)=w_1w_2\cdots w_{n-2}1$. The parking table of $w$ and $\operatorname{red}(w)$ coincide for the first $n-2$ elements by Claim~\ref{lem:parkingtables}. The corresponding parking table of $\operatorname{red}(w)$ is $p'=(p_1,p_2,\ldots, p_{n-2},1)$ since the last element of $\operatorname{red}(w)$ equals to one. Furthermore, since $\operatorname{red}(w)$ ends with the element one, deleting it and reducing again yields a permutation of length $n-2$ and its parking table is $p''=(p_1,p_2,\ldots, p_{n-2})$. 
        In particular, to recover the product $p_1p_2\cdots p_{n-2}12$, we multiply the product $p_1p_2\cdots p_{n-2}$ by $2$, which results in the number of parking functions whose parking permutations avoid $\Pi$. Thus, $\operatorname{pk}(n,i,r)=2\operatorname{pk}(n-2)$.
        \item If $\pi_n\neq 2$ with $j<i$ and $r=2$ then $j=1$ or $j=2$ by the permutation scheme. Since $r=2$ then $\pi_{n-2}>\pi_n=i$.\label{case2b}

Thus, $\operatorname{red}(w)=w_1w_2\cdots w_{n-2}(i-1)$. The corresponding parking table of $\operatorname{red}(w)$ is $p'=(p_1,p_2,\ldots, p_{n-2},1)$ since $w_{n-2}\geq i>i-1$ as $\pi_{n-2}>\pi_n=i$.     Altogether, for case~(\ref{item:case5}), we obtain 
    \begin{align*}
        \operatorname{pk}(n,i,r)&=\frac{r}{r-1}\operatorname{pk}(n-1,i-1,r-1) + r\operatorname{pk}(n-r+1,i-r+1,1) \\
        &\qquad + \sum_{r_1=2}^{r-1} \frac{2r}{r-r_1}\operatorname{pk}(n-r_1,i-r_1,r-r_1).
    \end{align*}
\end{enumerate}
In all of the arguments above, by Claim~\ref{claim:reduavoid}, each reduced permutation avoids both patterns $1234, 2134$. In addition, it is easy to check that the initial conditions are satisfied for $n\leq 3$.
