<!-- theorem-1: body_md -->
\label{thm:spec-gap}
Suppose that for any function $f:{\cal S}\to\mathbb R$ satisfying $\mathbb{E}_{\mu} f = 0$, 
$$ \langle \mathcal L f,f\rangle_\mu\geqslant \varepsilon\langle f,f\rangle_\mu,$$
where $\mathcal L=nI- \mathcal P$ and for two functions $f,g:{\cal S}\to \mathbb R$, $\langle f,g\rangle_\mu=\mathbb{E}_{X\sim\mu}[f(X)g(X)].$ Then, the mixing time of $\frac{1}{n}\mathcal P$ is bounded by $$O\Bigg(\frac{n\cdot \log\Big(\frac{1}{\mu_{\min}}\Big)}{\varepsilon}\Bigg)$$
where $\mu_{\min}=\min_x\mu(x)$.
<!-- end theorem-1 -->
<!-- head-1: body_md -->
Given the $n$-particle spin system, and function $f:S_1\times\dots \times S_n\to\mathbb R$ let $P_if$ be the function which averages $f$ on $S_i$ such that the spin of all other particles are fixed, namely:
$$ P_if(x_1,\dots,x_n)=\mathbb{E}_\mu[f(X_1,\dots,X_n) | X_j=x_j,\forall j\neq i].$$
<!-- end head-1 -->
<!-- fact-1: body_md -->
\label{fact:LLvsL}
    If for any function $f:{\mathcal S}\to \mathbb R$,
    $$ \langle \mathcal L f,\mathcal L f\rangle\geqslant \varepsilon\langle \mathcal L f,f\rangle=\varepsilon\cdot \mathcal E(f),$$
    then the spectral gap of the Glauber dynamics $\frac1{n}\mathcal P$ is at least $\varepsilon/n.$
<!-- end fact-1 -->
<!-- proof-1: body_md -->
Given an arbitrary function $g$, since $\mathcal L\succeq 0$, we let $f=\mathcal L^{-1/2} g$ (to be precise, we need to take the pseudo-inverse of $\mathcal L$ since $\mathcal L$ has a 0 eigenvalue). 
Having this,
    $$ \langle \mathcal L g,g\rangle=\langle \mathcal L f,\mathcal L f\rangle\geqslant\varepsilon\cdot\langle \mathcal L f,f\rangle = \langle g,g\rangle,$$
    where all the inner-products are with respect to $\mu$. This concludes the proof.
<!-- end proof-1 -->
<!-- definition-1: body_md -->
\label{def:rank-1-perturbed}
    We say $\mathcal P$ exhibits a rank-1-perturbed trickledown condition if there exists a family of vectors $\{\mathbf m(X)\} \in \mathbb R^n,$ for all $X\in {\mathcal S}$ such that for any function $f:{\mathcal S}\to\mathbb R$ and any $1\leqslant i<j\leqslant n$ and any (feasible) spin $x_{-i-j}$ (for all particles except $i,j$) we have
    \begin{align} \label{eq:rank1perturbed}&\mathbb{E}_{X\sim\mu}[L_if(X)L_jf(X) | x_{-i-j}] \geqslant -a_{i,j}\mathbb{E}[L_if(X)L_jf(X)m_i(X)m_j(X)|x_{-i-j}] \\
    &\quad- b_{i,j} \mathbb{E}[L_if(X)^2+L_jf(X)^2|x_{-i-j}]
    \end{align}
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{prop:rank-1-perturbed}
    Suppose $\mathcal P$ exhibits rank-1-perturbed trickledown condition w.r.t.\ vectors ${\bf m}(\mathbf x)$ with corresponding constants $\{a_{i,j},b_{i,j}\}_{1\leqslant i,j\leqslant n}$. If for every $i$,
    \begin{equation}\label{eq:rank-1-perturbed-condition}
        \sum_j b_{i,j} \mathbb{E}[L_if(X)^2]+ \lambda_{\max}(A)\mathbb{E}[m_i(X)^2L_if(X)^2] \leqslant (1-\varepsilon)\mathbb{E}[L_if(X)^2],
    \end{equation}
    where $A\in \mathbb R^{n\times n}$ has entries $a_{i,j}$, then $\langle \mathcal L f, \mathcal L f\rangle\geqslant \varepsilon\cdot\mathcal E(f)$.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Let $w_i(X)=L_if(X) m_i(X)$. Multiplying both sides of \eqref{eq:rank1perturbed} by $\sum_{X\in{\mathcal S}:X_{-i-j}=\sigma_{-i-j}}\mu(X)$ and summing up over all $i,j$ we get
    \begin{align*}
        \langle\mathcal L f,\mathcal L f\rangle&\geqslant -\mathbb{E}[{\bf w} {\bf w}^\top]\bullet A+ \sum_i (1-\sum_j b_{i,j})\mathbb{E}[\mathcal L_i f(X)^2]
                                                         
        \\
        &\geqslant -\lambda_{\max}(A)\operatorname{tr}(\mathbb{E}[{\bf w}{\bf w}^\top])+\sum_i (1-\sum_j b_{i,j})\mathbb{E}[\mathcal L_i f(X)^2]\\
        &= -\lambda_{\max}(A)\cdot \mathbb{E}[\|{\bf w}\|^2]+\sum_i (1-\sum_j b_{i,j})\mathbb{E}[\mathcal L_i f(X)^2]\\
        &=\sum_i \left(\left(1-\sum_j b_{i,j}\right)\mathbb{E}[\mathcal L_i f(X)^2]-\lambda_{\max}(A)\mathbb{E}[m_i(X)^2\mathcal L_if(X)^2]\right)\\
        &\geqslant \varepsilon\cdot \mathcal E(f),
    \end{align*}
    where the second inequality follows by Lemma~\ref{lem:amgm}. \qedhere

<!-- theorem-2: body_md -->
\label{thm:gap-L}
    Let $\mathcal L_J=\sum_i L_i=\sum_i (I-P_i)$. There exists an absolute constant $\varepsilon \geqslant 5 \cdot 10^{-5}$ with the following property.     where $C$ is a function of $\varepsilon$.

<!-- corollary-1: body_md -->
\label{cor:gap-L}
     There exists an absolute constant $\varepsilon >0$ with the following property. When $\beta<1/2 + \varepsilon$, the spectral gap of $\frac{1}{n}\mathcal P_J$ is $\Omega(1/n)$, and therefore Glauber dynamics for the SK model mixes in time $O(n^2)$.
<!-- end corollary-1 -->
<!-- proof-3: body_md -->
We will show that with probability $1 - o_n(1)$, we have
   \begin{equation}\label{eq:pi-min}
       \mu_{\min} := \min_{\mathbf x \in \{\pm 1 \}^n}\mu_{J}(\mathbf x)  \geqslant \exp(-(\ln 2 + 4\beta)n).
   \end{equation}

   With \eqref{eq:pi-min}, it follows from Theorem~\ref{thm:spec-gap} that the mixing time is $O(\log(1/\mu_{\min})/\Omega(1/n)) = O(n^2)$, as desired.
   
  It remains to prove $\eqref{eq:pi-min}$. By \eqref{eq:muJ}, a step towards this is to show that with probability $1 - o_n(1)$, we have for all $\mathbf x \in \{ \pm 1 \}^n$,
  \begin{equation}\label{eq:bounded-energy}
      |\mathbf x^\top J \mathbf x| \leqslant 2\beta n.
  \end{equation}
 Assuming that we are on the high probability event for which \eqref{eq:bounded-energy} is true, we can bound $Z_n = \sum_{\mathbf x \in \{ \pm 1 \}^n} \exp\Big(\frac12\mathbf x^\top J \mathbf x\Big) \leqslant 2^n \exp(\beta n) $. Consequently, this gives us \eqref{eq:pi-min}: \[\min_{\mathbf x \in \{\pm 1 \}^n}\mu_{J}(\mathbf x) \geqslant \frac{\exp(-\beta n)}{2^n \exp(\beta n)}= \exp(-(\ln 2 + 2 \beta)n).\] 
  
  Finally, it remains to prove \eqref{eq:bounded-energy}. We fix $\mathbf x \in \{ \pm 1 \}^n$.  Since $\operatorname{Var}(\mathbf x^\top J \mathbf x)  = \frac{4\beta^2}{n} \sum_{i<j}\operatorname{Var}(g_{ij}x_ix_j) = \frac{4\beta^2}{n}\binom n2 = 2\beta^2(n-1)$ where $g_{ij} \sim \mathcal{N}(0,1)$ for $i,j \in [n]$, we have $\mathbf x^\top J \mathbf x \sim \mathcal{N}(0, 2\beta^2(n-1))$. Consequently, we may apply the Gaussian tail bound to the standard normal random variable $\frac{\mathbf x^\top J x}{\beta\sqrt{2(n-1)}}$ to obtain,                                         
  \[\mathbb{P}\bigl[|\mathbf x^\top J\mathbf x|>2\beta n\bigr] \leqslant \mathbb{P}\Bigg[\frac{|\mathbf x^\top J\mathbf x|}{\beta\sqrt{2(n-1)}}>\sqrt{2n}\Bigg]
\leqslant 2\exp\left(
        \frac{-(\sqrt{2n})^2}{2}
    \right)  \leqslant 2e^{-n}.\] 
    Taking a union bound over all $\mathbf x \in \{ \pm 1 \}^n$, 
   \[ \mathbb{P}\big(\max_{\mathbf x \in \{ \pm 1 \}^n}|\mathbf x^\top J \mathbf x| > 2n \big) \leqslant 2^{n+1} \exp(-n) = o_n(1). \]
   This shows \eqref{eq:bounded-energy}.

For ${\bf z} \in \mathbb{S}^{n-1}$, we define the centered Gaussian processes 
    \[ X_{{\bf z}} = {\bf z}^\top J {\bf z} \quad\text{and}\quad Y_{{\bf z}} = \frac{2\beta}{\sqrt{n}} \mathbf g^\top {\bf z} \quad\text{where} \; \mathbf g\sim \mathcal{N}(0,I_n). \]

    We note that for ${\bf z},\tilde{{\bf z}} \in \mathbb{S}^{n-1} $ that 
        \begin{align*}
        \mathbb{E}\big[(X_{{\bf z}}-X_{\tilde{{\bf z}}})^2 \big] &=4\,\mathbb{E}\bigg[\big(\sum_{i<j}J_{ij}(z_iz_j-\tilde{z}_i\tilde{z}_j)\big)^2\bigg]\\&\underset{}{=} \frac{4 \beta^2}{n} \sum_{i < j} (z_i z_j - \tilde{z}_i\tilde{z}_j)^2 \leqslant \frac{2\beta^2}{n} \big\| {\bf z}{\bf z}^\top - \tilde{{\bf z}}\tilde{{\bf z}}^\top \big \|_F^2 \\
        &= \frac{4\beta^2}{n} \big[\big(1 - ({\bf z}^\top \tilde{{\bf z}})^2 \big) \big] \underset{1-x^2 \leqslant 2(1-x)}{\leqslant} \frac{4\beta^2}{n} \big[ 2\big(1-{\bf z}^\top \tilde{{\bf z}} \big) \big] \\
        &= \frac{4 \beta^2}{n} \|{\bf z} - \tilde{{\bf z}} \|_2^2 = \mathbb{E} \big[ (Y_{{\bf z}} - Y_{\tilde{{\bf z}}})^2 \big].
    \end{align*}

    \begin{theorem}[Variant of Sudakov-Fernique comparison inequality {\cite{V00}}]\label{thm:Sudakov-Fernique}
    Let $\mathcal I$ be a countable index set. Suppose $\{X_i\}_{i \in \mathcal I}$ and $\{Y_i \}_{i \in \mathcal I}$ are two mean-zero Gaussian processes  such that 
    \[ \mathbb{E}(X_i - X_j)^2 \leqslant \mathbb{E}(Y_i - Y_j)^2 \; \forall i,j\in\mathcal I.\]
    Let $\{m_i \}_{i \in \mathcal I}$ be such that $m_i \in \mathbb R$ for $i \in \mathcal I$ independent of the underlying probability measure. Then 
    \[ \mathbb{E} \sup_i (X_i + m_i) \leqslant \mathbb{E} \sup_i (Y_i + m_i)\; \forall i\in\mathcal I.\]
\end{theorem}

    \end{equation}
    We note that $\mathbb{S}^{n-1}$ is not enumerable but we can apply the theorem to a countable dense subset of $\mathbb{S}^{n-1}.$ First, we bound the RHS:
\begin{claim}\label{claim:expectedXz}
    $$ \mathbb{E} \sup_{{\bf z} \in \mathbb{S}^{n-1}} \bigg \{ \frac{2\beta}{\sqrt{n}}\mathbf g^\top {\bf z} - \rho  \norm{{\bf z}}^2_{\leqslant \delta n} \bigg\} \leqslant 2\beta-\frac{\pi \beta\rho}{120(\beta+\rho)}\delta^3$$
\end{claim}
\begin{proof}
Let $\bar{\mathbf g}=\mathbf g/\sqrt{n}$. First, using $\norm{{\bf z}}=1$ we observe that
\begin{align}\label{eq:gtzbound}
    2\beta\bar{\mathbf g}^\top {\bf z} - \rho \norm{{\bf z}}^2_{\leqslant \delta n}=\beta(1+\norm{\bar{\mathbf g}}^2)-(\beta\norm{\bar{\mathbf g}-{\bf z}}^2+\rho\norm{{\bf z}}^2_{\leqslant \delta n})\leqslant \beta(1+\norm{\bar{\mathbf g}}^2)-\frac{\beta\rho}{\beta+\rho}\norm{\bar{\mathbf g}}^2_{\leqslant \delta n}
\end{align}
To see the inequality, let $S$ be the smallest $\lfloor\delta n \rfloor$ coordinates of ${\bf z}$ (in absolute value). Then, by triangle inequality,
$$ \norm{\bar{\mathbf g}}_{\leqslant \delta n} = \sqrt{\sum_{i\in S} \bar{g}_i^2} \underset{\text{triangle ineq}}{\leqslant} \sqrt{\sum_{i\in S} (\bar{g}_i-z_i)^2} + \sqrt{\sum_{i\in S} z_i^2} \leqslant \norm{\bar{\mathbf g}-z} + \norm{{\bf z}}_{\leqslant \delta n}  $$
Therefore, by the Cauchy-Schwarz inequality, 
$$ \norm{\bar{\mathbf g}}^2_{\leqslant \delta n} \leqslant \bigg(\frac{1}{\beta}+\frac{1}{\rho} \bigg)(\beta\norm{\bar{\mathbf g}-{\bf z}}^2+\rho\norm{{\bf z}}_{\leqslant \delta n}^2)$$
Having \eqref{eq:gtzbound} and using $\mathbb{E}\norm{\bar{\mathbf g}}^2=1$, we have that
$$\mathbb{E} \sup_{{\bf z} \in \mathbb{S}^{n-1}} \bigg \{ \frac{2\beta}{\sqrt{n}}\mathbf g^\top {\bf z} - \rho  \norm{{\bf z}}^2_{\leqslant \delta n} \bigg\} \leqslant 2\beta-\frac{\beta\rho}{\beta+\rho}\frac{\mathbb{E}\norm{\mathbf g}^2_{\leqslant \delta n}}{n} 
$$
So, to finish the proof it is enough to show that $\mathbb{E}\norm{\mathbf g}^2_{\leqslant \delta n}\geqslant n \frac{\pi\delta^3}{6}.$ 

For every $t > 0$, 

\begin{align*}
    \frac{\mathbb{E} \|\mathbf g \|_{\leqslant \delta n}^2}{n} &\geqslant \frac{\mathbb{E} \min_{S \subset [n]:|S|=\lfloor\delta n \rfloor} \sum_{i\in S} \min \{ g_i^2 ,t^2 \}}{n} \\

    &= \delta t^2 - \mathbb{E}_{g \sim \mathcal{N}(0,1)} (t^2 - g^2)_+ = \delta t^2 - \int_{-t}^t (t^2 - x^2) \frac{\exp(-x^2/2)}{\sqrt{2\pi}}dx \\
    &\underset{e^{-x^2/2}\leqslant 1}{\geqslant} \delta t^2 - \frac{4t^3}{3\sqrt{2} \pi} \underset{t = \sqrt{\pi/2}\delta}{= }\frac{\pi}{6} \cdot \delta^3. 
\end{align*}
where we write $(a)_+=\max\{a,0\}.$
\end{proof}

  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 + s^2  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \mathbb{E}[Y \varphi(Y)] + \mathbb{E}[\varphi(Y)]\mathbb{E}[X] + s  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0\mathbb{E}[Y\varphi(Y)] \mathbb{E}[X]\right) \notag \\
     &\hspace{3em} -\sigma_{Y_0}\left( \mathbb{E} [\varphi(X)] s 
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 + s^2  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 \mathbb{E}[X \varphi(X)] + \mathbb{E}[\varphi(X)]\mathbb{E}[Y] + s  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0\mathbb{E}[X\varphi(X)] \mathbb{E}[Y]\right) \notag \\
    &= -2(s^2 + s) \sigma_{X_0} \sigma_{Y_0}  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0.
\end{align}
Second, we compute the coefficient coming from $\big(XY -  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \big)M(0)R_X(0)R_Y(0)$ using $\mathbb{E}_{\mu_{0,a,b}}[\varphi(X)] = \mathbb{E}_{\mu_{0,a,b}}[\varphi(Y)] = 0$ as follows 
\begin{align}\label{eq:second-MRXRY}
    \mathbb{E}_{\mu_{0,a,b}} \big[\big(XY -  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \big) \big(X + s  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 \big) \varphi(X) \big(Y + s  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \big)  \varphi(Y)\big] &\underset{\eqref{eq:constant-MRXRY}}{} \mathbb{E}_{\mu_{0,a,b}}\big[s^2XY  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \varphi(X) \varphi(Y)\big] -  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \sigma_{X_0} \sigma_{Y_0} \notag \\
    &= (s^2 -1) \sigma_{X_0} \sigma_{Y_0} 
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0.
\end{align}

We add \eqref{eq:first-MRXRY} and \eqref{eq:second-MRXRY} to get the term linear in $ 
  \mkern1mu\overline{\mkern-1mu J \mkern-1mu}\mkern1mu$, and combine with the constant term \eqref{eq:constant-MRXRY}, which gives the first assertion. The second assertion of (v) follows by interchanging the roles of $X$ and $Y$ above.

\textbf{(vii).} Let $F = MR_XR_Y\varphi(X)$. By Lemma~\ref{lem:bounded-second-der}, the conditions \eqref{eq:F-conditions} of  Lemma~\ref{lem:master} hold, so we apply Lemma~\ref{lem:master} to $F$. 
\begin{align}\label{eq:constant-MRXRYphi}
    &\mathbb{E}_{\mu_{0,a,b}} [F(0)] = \mathbb{E}_{\mu_{0,a,b}}\big[ \big(X + s  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 \big) \varphi(X)^2\big] \cdot \mathbb{E}_{\mu_{0,a,b}}\big[ \big( Y + s  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \big) \varphi(Y) \big] \notag \\
    &= \big(\mathbb{E}_{\mu_{0,a,b}}[X \varphi(X)^2]+ s  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0\mathbb{E}_{\mu_{0,a,b}}[\varphi(X)^2] \big)\big(\mathbb{E}_{\mu_{0,a,b}}[Y \varphi(Y)]+ s  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0\mathbb{E}_{\mu_{0,a,b}}[\varphi(Y)] \big) \notag \\
    &\underset{\eqref{eq:Xphisquare}}{=} (s-1) 
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 \cdot \sigma_{Y_0}.
\end{align}
As in (ii), using another application of Lemma~\ref{lem:bounded-second-der}, the linear term in $ 
  \mkern1mu\overline{\mkern-1mu J \mkern-1mu}\mkern1mu$ of $ MR_XR_Y\varphi(X)$ is $O\big( 
  \mkern1mu\overline{\mkern-1mu J \mkern-1mu}\mkern1mu\big)$. Interchanging the roles of $X$ and $Y$, we obtain the second assertion similarly. 

\textbf{(viii).} Let $F = MR_XR_Y\varphi(X)\varphi(Y)$. By Lemma~\ref{lem:bounded-second-der}, the conditions \eqref{eq:F-conditions} of  Lemma~\ref{lem:master} hold, so we apply Lemma~\ref{lem:master} to $F$. 
\begin{align}\label{eq:constant-MRXRYphisquare}
    &\mathbb{E}_{\mu_{0,a,b}} [F(0)] \\&= \mathbb{E}_{\mu_{0,a,b}}\big[ \big(X + s  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0 \big) \varphi(X)^2\big] \cdot \mathbb{E}_{\mu_{0,a,b}}\big[ \big( Y + s  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0 \big) \varphi(Y)^2 \big] \notag \\
    &= \big(\mathbb{E}_{\mu_{0,a,b}}[X \varphi(X)^2]+ s  
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0\mathbb{E}_{\mu_{0,a,b}}[\varphi(X)^2] \big)\big(\mathbb{E}_{\mu_{0,a,b}}[Y \varphi(Y)^2]+ s  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0\mathbb{E}_{\mu_{0,a,b}}[\varphi(Y)^2] \big) \notag \\
    &\underset{\eqref{eq:Xphisquare}}{=} (s-1)^2 
  \mkern1mu\overline{\mkern-1mu X \mkern-1mu}\mkern1mu_0  
  \mkern1mu\overline{\mkern-1mu Y \mkern-1mu}\mkern1mu_0.
\end{align}

As in (ii), using another application of Lemma~\ref{lem:bounded-second-der}, the linear term in $ 
  \mkern1mu\overline{\mkern-1mu J \mkern-1mu}\mkern1mu$ of $ MR_XR_Y\varphi(X)\varphi(Y)$ is $O\big( 
  \mkern1mu\overline{\mkern-1mu J \mkern-1mu}\mkern1mu\big)$. \qedhere
