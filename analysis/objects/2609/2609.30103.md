<!-- maintheorem-1: body_md -->
\label{thm:mainA}
Let $U$ be a wandering Fatou component of a transcendental entire function $f$. Then $\bigcup_{n\geqslant0}f^n(U)$ is unbounded.
<!-- end maintheorem-1 -->
<!-- maintheorem-2: body_md -->
\label{thm:mainB}
Let $U$ be a wandering Fatou component of a transcendental meromorphic function $f\colon\mathbb C\to\widehat{\mathbb C}$. 
Then
\[\limsup\limits_{n \to \infty} |f^n(z)| = \infty \qquad \text{for every }z \in U.\]
In particular, $f$ has no orbitally bounded wandering domains.
<!-- end maintheorem-2 -->
<!-- remark-1: body_md -->
No restriction is imposed on the number of poles or on the boundedness or
    connectivity of the components $U_n$. 
    Moreover, note that the above statement is equivalent to the existence of a sequence
    $n_k\to\infty$ such that
    $f^{n_k}\longrightarrow\infty$ locally uniformly on $U$. This answers a question posed by Bergweiler in \cite[Section~4.5, Question~8]{Bergweiler} affirmatively.
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lem:recovery}
Let $E\subset\mathbb C$ be a closed set. Suppose that $P_j\subset E$ are increasing finite sets, $\#|P_1|\geqslant2$, that exhaust $E$:
\[
\overline{\bigcup_{j\geqslant1}P_j}=E.
\]
Then on every connected component $G$ of $\mathbb C\setminus E$, we have the monotone convergence
\[
\rho_{\mathbb C \setminus P_j}\big|_G \uparrow \rho_G \quad \text{locally uniformly in $G$.}
\] 
In particular, for every Borel set $A\subset G$,
\[
\lim_{j \to \infty} {\mathcal A}_{\rho_{\mathbb C \setminus P_j}}(A) = {\mathcal A}_{\rho_G}(A).
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The proof is a corollary of the pointed Carath\'eodory convergence of domains $(\mathbb C \setminus P_j, z_0) \to (G,z_0)$ for any choice $z_0 \in G$; see, e.g., \cite[Proposition 4.4]{McM}.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:punctures}
Let $G\subset\mathbb C$ be a hyperbolic domain, and let $E\subset G$ be a finite set. Then
\begin{equation}\label{eq:puncture-cost}
0\leqslant
\int_{G\setminus E}
\left(\rho_{G\setminus E}^2-\rho_G^2\right)\,\mathrm dx\,\mathrm dy
\leqslant\frac{\pi}{2}\#|E|.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Choose increasing finite sets $P_j\subset\mathbb C\setminus G$, with $\#|P_1| \geqslant 2$, that exhaust $\mathbb C\setminus G$, i.e.,
\[
\overline{\bigcup_{j \geqslant 1} P_j} = \mathbb C \setminus G.
\]
By \eqref{eq:finite-area},
\begin{equation*}
\begin{aligned}
&{\mathcal A}_{\rho_{\mathbb C \setminus P_j}}(\mathbb C \setminus P_j) = \frac{\pi}{2}(\#|P_j|-1), \\
&{\mathcal A}_{\rho_{\mathbb C \setminus (P_j \cup E)}}(\mathbb C \setminus (P_j \cup E)) = \frac{\pi}{2}(\#|P_j \cup E|-1) = \frac{\pi}{2}(\#|P_j| + \#|E|-1),   
\end{aligned}
\end{equation*}
where the last equality follows since the sets $P_j$ and $E$ are disjoint. Hence
\[
\int_{\mathbb C \setminus (P_j \cup E)} \rho_{\mathbb C \setminus (P_j \cup E)}^2\,\mathrm dx\,\mathrm dy - \int_{\mathbb C \setminus P_j} \rho_{\mathbb C \setminus P_j}^2\,\mathrm dx\,\mathrm dy  = \frac{\pi}{2} \#|E|.
\]
Furthermore, since $E$ is finite (and hence has measure zero), we obtain
\[
\int_{\mathbb C \setminus P_j} \rho_{\mathbb C \setminus P_j}^2\,\mathrm dx\,\mathrm dy = \int_{\mathbb C \setminus (P_j \cup E)} \rho_{\mathbb C \setminus P_j}^2\,\mathrm dx\,\mathrm dy.
\]
Combining the last two estimates, we get
\[
\int_{\mathbb C \setminus (P_j \cup E)} \left(\rho_{\mathbb C \setminus (P_j \cup E)}^2 - \rho_{\mathbb C \setminus P_j}^2\right)\,\mathrm dx\,\mathrm dy = \frac{\pi}{2} \#|E|.
\]
The integrand is nonnegative by monotonicity. Lemma~\ref{lem:recovery} gives the convergence of $\rho_{\mathbb C\setminus(P_j \cup E)}$ and $\rho_{\mathbb C\setminus P_j}$ to the corresponding intrinsic hyperbolic densities $\rho_{G \setminus E}$ and $\rho_G$ on compacts in $G \setminus E$. Restricting the integrals to $G\setminus E$ and applying Fatou's lemma proves \eqref{eq:puncture-cost}.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:localization}
Fix a finite set $Q\subset\mathbb C$, with $\#|Q|\geqslant2$, an open Euclidean disk $D \subset \mathbb C$, and a compact set $K\subset D$. Then there exists a constant $C=C(Q,D,K)<\infty$ such that, for every finite set $P\supset Q$,
\[
\int_{K\setminus P}
\left(\rho_{D\setminus P}^2-\rho_{\mathbb C\setminus P}^2\right)\,\mathrm dx\,\mathrm dy
\leqslant C.
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Set $X_P :=\mathbb C\setminus P$ and define the function
\[
u_P :=\log\frac{\rho_{D\setminus P}}{\rho_{X_P}}
\quad\text{on }D\setminus P.
\]
Domain monotonicity and \eqref{eq:curvature} give
\[
u_P\geqslant0,
\qquad
\Delta u_P=4\left(\rho_{D\setminus P}^2-\rho_{X_P}^2\right)\geqslant0,
\]
so $u_P$ is a non-negative subharmonic function on $D \setminus P$.

Fix a compact set $T \subset D$. Choose a closed circular annulus inside $D$, separating $T$ from $\partial D$, which avoids the finite set $Q$. The positive continuous function $\rho_{\mathbb C\setminus Q}$ has a positive minimum on this annulus. Every path from $T\setminus P$ to $X_P\setminus D$ crosses the annulus. Since $\rho_{X_P} = \rho_{\mathbb C \setminus P} \geqslant\rho_{\mathbb C\setminus Q}$, there is $\varepsilon_T>0$, independent of $P$, such that
\begin{equation}\label{eq:distance-bound}
d_{X_P}(z,X_P\setminus D)\geqslant\varepsilon_T
\qquad \forall z\in T\setminus P,
\end{equation}
where $d_{X_P}$ is the hyperbolic distance in $X_P = \mathbb C \setminus P$.

Let $\pi\colon\mathbb D\to X_P$ be a universal cover with $\pi(0)=z\in T\setminus P$, and put $a_T :=\tanh\varepsilon_T$; note that we can choose $a_T < 1$. By covering invariance, every radial segment in $a_T\mathbb D$ has image of hyperbolic length less than $\varepsilon_T$. By~\eqref{eq:distance-bound},
\[
\pi(a_T\mathbb D)\subset D\setminus P.
\]
By the Schwarz--Pick Lemma applied to the mapping $\pi \colon a_T \mathbb D \to D \setminus P$ at $0$, we obtain
\[
\rho_{D \setminus P}(z) |\pi'(0)| \leqslant \rho_{a_T \mathbb D}(0) = \frac{\rho_{\mathbb D}(0)}{a_T} = \coth \varepsilon_T.
\]
Using the same idea for the starting covering $\pi\colon\mathbb D\to X_P$, we get $\rho_{X_P}(z)|\pi'(0)|=1$. Therefore,
\begin{equation}\label{eq:uniform-log-bound}
0\leqslant u_P(z)\leqslant\log\coth\varepsilon_T \qquad \forall z\in T\setminus P.
\end{equation}
Thus $u_P$ is locally bounded near each puncture $P\cap D$. A subharmonic function locally bounded above near an isolated puncture extends subharmonically across that puncture. Thus $u_P$ extends to a nonnegative subharmonic function on $D$. By \eqref{eq:uniform-log-bound}, these extensions are uniformly bounded on each compact subset of $D$.


Since $u_P$ extends to a non-negative subharmonic function on $D$, we may mollify it in the usual way. Let $\eta\in C_c^\infty(\mathbb C)$ be a non-negative standard mollifier, supported in the unit disk and normalized by $\int_{\mathbb C}\eta\,\,\mathrm dx\,\mathrm dy=1$. For $\delta>0$, let
\[
u_{P,\delta}(z)
:=
\int_{\mathbb C}u_P(z-\zeta)\eta_\delta(\zeta)\,\,\mathrm dx\,\mathrm dy(\zeta),
\]
where $\eta_\delta(\zeta)=\delta^{-2}\eta(\zeta/\delta)$ is a re-scaled mollifier. In this way, $u_{P,\delta}$ is well defined and smooth on
\[
D_\delta:=\{z\in D:\operatorname{dist}(z,\partial D)>\delta\}
\]
(where $\operatorname{dist}$ stands for the standard Euclidean distance). Moreover, since $u_P$ is subharmonic,
\[
\Delta u_{P,\delta}
=
(\Delta u_P)*\eta_\delta\geqslant 0.
\]

Choose $\varphi\in C_c^\infty(D)$ such that $0\leqslant \varphi\leqslant 1$ and $\varphi\equiv 1$ on a neighbourhood of $K$. Taking $\delta>0$ sufficiently small, we may assume that $\operatorname{supp}\varphi\subset D_\delta$. Since $u_P$ is smooth
on $D\setminus P$, we have
\[
\Delta u_{P,\delta}
\longrightarrow
\Delta u_P
=
4\left(\rho_{D\setminus P}^2-\rho_{\mathbb C\setminus P}^2\right)
\]
locally uniformly on $D\setminus P$. As $\Delta u_{P,\delta}\geqslant0$, Fatou's
lemma gives
\[
4\int_{K\setminus P}
\left(\rho_{D\setminus P}^2-\rho_{\mathbb C\setminus P}^2\right)\,\,\mathrm dx\,\mathrm dy
\leqslant
\liminf_{\delta\to0}
\int_D \varphi\,\Delta u_{P,\delta}\,\,\mathrm dx\,\mathrm dy .
\]

Since $\varphi$ has compact support in $D$, integration by parts yields
\[
\int_D \varphi\,\Delta u_{P,\delta}\,\,\mathrm dx\,\mathrm dy = \int_D u_{P,\delta}\,\Delta\varphi\,\,\mathrm dx\,\mathrm dy .
\]
By the uniform bound obtained above, there exists a constant $M>0$,
depending only on $Q$, $D$, and $\operatorname{supp}\varphi$ (and hence on $K$), such that $0\leqslant u_P\leqslant M$ on a neighbourhood of $\operatorname{supp}\varphi$, uniformly in $P\supset Q$. Hence, for
all sufficiently small $\delta$, we have the same bound $0\leqslant u_{P,\delta}\leqslant M$ on $\operatorname{supp}\varphi$. Therefore,
\[
\int_D u_{P,\delta}\,\Delta\varphi\,\,\mathrm dx\,\mathrm dy
\leqslant
M\int_D |\Delta\varphi|\,\,\mathrm dx\,\mathrm dy .
\]
Consequently,
\[
\int_{K\setminus P}
\left(\rho_{D\setminus P}^2-\rho_{\mathbb C\setminus P}^2\right)\,\,\mathrm dx\,\mathrm dy
\leqslant
\frac{M}{4}\int_D |\Delta\varphi|\,\,\mathrm dx\,\mathrm dy .
\]
The right-hand side is independent of $P$, which proves the required
uniform bound.

<!-- lemma-4: body_md -->
\label{lem:local-charts}
Let $f\colon\mathbb C\to\widehat{\mathbb C}$ be a nonconstant meromorphic map. Every point
$a\in\mathbb C\setminus\operatorname{Pol}(f)$ has arbitrarily small neighborhoods
$A\Subset\mathbb C\setminus\operatorname{Pol}(f)$ such that
\[
 f\colon A\longrightarrow D
\]
is a proper map of finite degree, where $D$ is a Euclidean disk. The restriction
has no critical value or has the single critical value $f(a)$. Consequently, every compact set $K\subset\mathbb C\setminus\operatorname{Pol}(f)$ admits finitely many such charts $A_\ell$ and compact sets $K_\ell\Subset A_\ell$ such that $\bigcup K_\ell \supset K$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Fix $a\in\mathbb C\setminus\operatorname{Pol}(f)$. Since $f(a)\in\mathbb C$, the local normal form for a nonconstant holomorphic map gives a neighborhood $U$ of $a$, a conformal coordinate $\zeta\colon U\to \zeta(U)$ with $\zeta(a)=0$, and an integer $m\geqslant 1$ such that
\[
f(z)=f(a)+\zeta(z)^m,\qquad z\in U.
\]
Choosing $r>0$ arbitrarily small with $\overline{D(0,r)}\subset\zeta(U)$, set
\[
A:=\zeta^{-1}(D(0,r)), \qquad D:=D(f(a),r^m).
\]
Then $A\Subset\mathbb C\setminus\operatorname{Pol}(f)$, and, in the coordinate $\zeta$, the restriction $f\colon A\to D$ is conjugate to
\[
w\longmapsto w^m\colon D(0,r)\to D(0,r^m).
\]
Hence it is a proper map of degree $m$. If $m=1$, it has no critical points; if $m\geqslant2$, its unique critical point is $a$, and its unique critical value is $f(a)$. Since $r$ may be chosen arbitrarily small, such charts form a neighborhood basis at $a$.

For the final assertion, choose for each $a\in K$ such a chart $A_a$ and an open neighborhood $U_a$ of $a$ with ${U_a}\Subset A_a$. By compactness, finitely many $U_{a_\ell}$ cover $K$. Setting
\[
A_\ell:=A_{a_\ell},
\qquad
K_\ell:=\overline{U_{a_\ell}},
\]
gives $K_\ell\subset A_\ell$ and $\bigcup_\ell K_\ell\supset K$.

<!-- lemma-5: body_md -->
\label{lem:deficit}
Let $f\colon\mathbb C\to\widehat{\mathbb C}$ be a nonconstant meromorphic function. Fix a finite set $Q\subset\mathbb C$, with $\#|Q|\geqslant2$, and a compact set $K\subset\mathbb C\setminus\operatorname{Pol}(f)$. Then there exists a positive constant $C = C(f, Q, K)<\infty$ such that, for every finite, forward-invariant set $P\supset Q$, it follows that
\begin{equation}\label{eq:deficit}
\int_K\left[\rho_{\mathbb C \setminus P}^2-(f^*\rho_{\mathbb C \setminus P})^2\right]_+\,\mathrm dx\,\mathrm dy\leqslant C.
\end{equation}
The discrete exceptional sets on which the displayed densities are undefined are omitted from the integral.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Choose finitely many proper charts
\[
f\colon A_\ell\to D_\ell
\]
and compact sets $K_\ell\Subset A_\ell$ covering $K$, as in
Lemma~\ref{lem:local-charts}. Let $V_\ell\subset D_\ell$ denote the set of
critical values of $f|_{A_\ell}$, and put
\[
Z_{\ell,P}:=D_\ell\setminus(P\cup V_\ell).
\]
Then
\[
f\colon A_\ell\setminus f^{-1}(P\cup V_\ell)\longrightarrow Z_{\ell,P}
\]
is a covering. Since $f(P)\subset P$, we have
\[
A_\ell\setminus f^{-1}(P\cup V_\ell)\subset\mathbb C\setminus P.
\]
Hence, by domain monotonicity and covering invariance of the hyperbolic
metric,
\[
\rho_{\mathbb C\setminus P}
\leqslant
\rho_{A_\ell\setminus f^{-1}(P\cup V_\ell)}
=
f^*\rho_{Z_{\ell,P}}.
\]
Thus, we have
\[
\rho_{\mathbb C\setminus P}(z)^2 \leqslant |f'(z)|^2\rho_{Z_{\ell,P}}(f(z))^2 \qquad \text{ for every } z\in K_\ell\setminus f^{-1}(P\cup V_\ell).
\]
Since
\[
\rho_{Z_{\ell,P}}\geqslant \rho_{\mathbb C\setminus P},
\]
it follows that, almost everywhere on $K_\ell$,
\begin{equation}\label{eq:local-deficit-bound}
\left[\rho_{\mathbb C\setminus P}^2-(f^*\rho_{\mathbb C\setminus P})^2\right]_+
\leqslant
|f'|^2
\left(\rho_{Z_{\ell,P}}^2-\rho_{\mathbb C\setminus P}^2\right)\circ f .
\end{equation}

We decompose the difference on the target as
\[
\rho_{Z_{\ell,P}}^2-\rho_{\mathbb C\setminus P}^2
=
\left(\rho_{D_\ell\setminus P}^2-\rho_{\mathbb C\setminus P}^2\right)
+
\left(\rho_{Z_{\ell,P}}^2-\rho_{D_\ell\setminus P}^2\right).
\]
Both terms are nonnegative. By Lemma~\ref{lem:localization}, the integral
of the first over $f(K_\ell)\Subset D_\ell$ is bounded by a constant
$C_\ell$ independent of $P$. By Lemma~\ref{lem:punctures}, applied in
$D_\ell\setminus P$ to the finite set $V_\ell\setminus P$, the second has
total integral at most
\[
\frac{\pi}{2}\#|V_\ell\setminus P|
\leqslant \frac{\pi}{2}\#|V_\ell|.
\]

Let $d_\ell:=\deg(f\colon A_\ell\to D_\ell)$. Applying the area formula to \eqref{eq:local-deficit-bound}, and using that the multiplicity of $f|_{K_\ell}$ is at most $d_\ell$, gives
\[
\int_{K_\ell}
\left[\rho_{\mathbb C\setminus P}^2-(f^*\rho_{\mathbb C\setminus P})^2\right]_+\,\mathrm dx\,\mathrm dy
\leqslant
d_\ell\left(C_\ell+\frac{\pi}{2}\#|V_\ell|\right).
\]
Summing over the finitely many sets $K_\ell$ yields \eqref{eq:deficit},
with a constant independent of $P$.

<!-- lemma-7: body_md -->
\label{lem:simple-connectivity}
Let $V$ be a Fatou component of $f$ containing a point with a bounded orbit. Then $V$ is simply connected.  \qed
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
Let $U$ be a wandering domain of $f$, and suppose that the orbit of $U$ is contained in a compact disk $K$. In particular, $U$ is bounded. Applying Lemma~\ref{lem:proper-component} inductively gives
\begin{equation}\label{eq:bounded-union-components}
U_n :=f^n(U)\subset K,
\qquad f\colon U_n\longrightarrow U_{n+1}
\quad\text{proper and onto}.
\end{equation}
All point orbits in the domains $U_n$ stay in $K$, so every $U_n$ is simply connected by Lemma~\ref{lem:simple-connectivity}.

The compact set $K$ contains only finitely many zeros of $f'$. Since the domains $U_n$ are pairwise disjoint, there is $N$ such that $f'$ has no zero in $U_n$ for $n\geqslant N$. For these $n$, the restriction in \eqref{eq:bounded-union-components} is a proper unramified covering of a simply connected domain. Hence it is a conformal isomorphism.

Use the densities $\rho_j$ from \eqref{eq:rho-j}. Lemma~\ref{lem:deficit}, with $Q=P_1$, gives a constant $C<\infty$ (independent of $j$) such that
\begin{equation}
\label{eq:uniform-deficit-K}
\int_K\left[\rho_j^2-(f^*\rho_j)^2\right]_+\,\mathrm dx\,\mathrm dy\leqslant C
\qquad(j\geqslant1).
\end{equation}
Fix $j$ and put $a_{j,n} := {\mathcal A}_{\rho_j}(U_n)$. These quantities are finite. For $n\geqslant N$, the change-of-variables formula for the conformal isomorphism $f\colon U_n\to U_{n+1}$ gives
\begin{align*}
a_{j,n}-a_{j,n+1}
&=\int_{U_n}\left(\rho_j^2-(f^*\rho_j)^2\right)\,\mathrm dx\,\mathrm dy\\
&\leqslant\int_{U_n}\left[\rho_j^2-(f^*\rho_j)^2\right]_+\,\mathrm dx\,\mathrm dy.
\end{align*}
Summing from $n=N$ to some $n=L > N$, and using disjointness and containment in $K$, yields
\begin{equation}\label{eq:main-telescoping}
a_{j,N}-a_{j,L+1}\leqslant C.
\end{equation}
On the other hand, for any fixed $j$, we have
\[
\sum_{n\geqslant N}a_{j,n} \leqslant {\mathcal A}_{\rho_j}(\mathbb C\setminus P_j)<\infty,
\]
so $a_{j,L+1}\to0$ as $L \to \infty$. Sending $L$ to infinity in \eqref{eq:main-telescoping} gives ${\mathcal A}_{\rho_j}(U_N)\leqslant C$.

Now let $j\to\infty$. By \eqref{eq:fatou-recovery} and monotone convergence,
\[
{\mathcal A}_{\rho_{U_N}}(U_N)\leqslant C.
\]
But $U_N$ is simply connected and hyperbolic. Thus ${\mathcal A}_{\rho_{U_N}}(U_N)=\infty$, a contradiction.

<!-- lemma-8: body_md -->
\label{lem:shrinking}
For every fixed $t>0$,
\[
 \sup_{w\in B_n(t)}|w-z_n|\longrightarrow0.
\]
Consequently, $\operatorname{diam} B_n(t)\to0$ and $B_n(t)\subset K$ for all sufficiently
large $n$. Moreover,
\begin{equation}\label{eq:ball-inclusion}
 f(B_n(t))\subset B_{n+1}(t)\qquad(n\geqslant0).
\end{equation}
<!-- end lemma-8 -->
<!-- proof-7: body_md -->
Fix two distinct finite points of $J(f)$. Every map $\pi_n$ omits them as well as
infinity, so the family $(\pi_n)$ is normal by the Montel's theorem. 
Given any subsequence of $(\pi_n)$, pass to
a further subsequence such that $z_n\to a\in S$ and $\pi_n$ converges
spherically locally uniformly to a limit $\pi$. Since $\pi(0)=a$, the limit function is not infinity.

We claim that $\pi$ is a constant function. If not, we can choose $\zeta_0\in\mathbb D$ with
$\pi'(\zeta_0)\neq0$, set $w=\pi(\zeta_0)$, and find a sufficiently small
closed disk centered at $\zeta_0$ on whose boundary $\pi-w$ is nonzero.
Rouch\'e's theorem implies that all sufficiently late $\pi_n$ in this
subsequence attain $w$ in that disk. 
It follows, that $w$ lies in infinitely many distinct Fatou components $U_n$, which is a contradiction.  

Thus every such limit is the constant $a$. It follows, by uniform convergence, that for each $r<1$,
\[
 \sup_{|\zeta|\leqslant r}|\pi_n(\zeta)-z_n|\longrightarrow0.
\]
The result now follows by using \eqref{eq:ball-uniformization}.
To obtain the forward inclusion, we apply the Schwarz--Pick lemma for the holomorphic map $f\colon U_n\to U_{n+1}$.

<!-- lemma-9: body_md -->
\label{lem:local-filling}
Fix $t>0$. For all sufficiently large $n$, every Jordan curve
$\gamma\subset B_n(t)$ has its bounded interior contained in $U_n$.
<!-- end lemma-9 -->
<!-- proof-8: body_md -->
By Lemma~\ref{lem:shrinking}, we can choose $N = N(t)$ large enough such that
\[
 B_m(t)\subset D(z_m,\eta)\qquad(m\geqslant N).
\]
Fix $n\geqslant N$. Let $\gamma\subset B_n(t)$ be a Jordan curve and let $\Omega$ denote its bounded interior. 
Since $\gamma\Subset D(z_n,\eta)$, the maximum modulus principle applied to the identity
map gives
\[
 \overline\Omega\subset D(z_n,\eta).
\]

We prove by induction that, for every $k\geqslant0$,
\begin{equation}\label{eq:filled-orbit}
 \begin{gathered}
 f^k\text{ is holomorphic on a neighborhood of }\overline\Omega,\\
 f^k(\overline\Omega)\subset D(z_{n+k},\eta).
 \end{gathered}
\end{equation}
The case $k=0$ follows by our choice of $\Omega$, 
so suppose that our assertion holds for $k$.
Then $f^k(\overline\Omega)$ lies in the pole-free disk $D(z_{n+k},\eta)$ and the composition 
$f^{k+1}=f\circ f^k$ is holomorphic on a neighborhood of $\overline\Omega$. 
On the boundary \eqref{eq:ball-inclusion} gives
\[
 f^{k+1}(\gamma)\subset B_{n+k+1}(t)
 \subset D(z_{n+k+1},\eta),
\]
and since the boundary image is compact, the maximum principle gives
\[
 \max_{w\in\overline\Omega}|f^{k+1}(w)-z_{n+k+1}|
 \leqslant\max_{w\in\gamma}|f^{k+1}(w)-z_{n+k+1}|<\eta,
\]
which proves the induction.

By \eqref{eq:filled-orbit}, every iterate is defined and holomorphic on
$\Omega$, and
\[
 |f^k(w)|\leqslant M+\eta\qquad(w\in\Omega,\ k\geqslant0).
\]
Thus $\Omega\subset F(f)$ by normality. 
Since the domain $\Omega$ is connected and meets $U_n$, we have $\Omega\subset U_n$.

<!-- lemma-10: body_md -->
\label{lem:embedded-balls}
For every fixed $t>0$, the restriction
\[
 \pi_n|_{r_t\mathbb D}\colon r_t\mathbb D\longrightarrow B_n(t),
 \qquad r_t=\tanh t,
\]
is a conformal isomorphism for all sufficiently large $n$. In particular,
these balls are simply connected and
\begin{equation}\label{eq:ball-area}
 {\mathcal A}_{\rho_{U_n}}(B_n(t))=\pi\sinh^2t
 \quad\text{for all sufficiently large }n.
\end{equation}
<!-- end lemma-10 -->
<!-- proof-9: body_md -->
Take $n$ sufficiently large for the conclusion of Lemma~\ref{lem:local-filling} to hold.
Since $B_n(t)$ is a plane domain, its fundamental group is generated by classes represented by Jordan curves. By Lemma~\ref{lem:local-filling}, every Jordan curve in $B_n(t)$ bounds a simply connected region in $U_n$, and hence is null-homotopic in $U_n$. 

Since the inclusion $B_n(t) \hookrightarrow U_n$ induces the trivial homomorphism on fundamental groups, the lifting criterion for the universal cover $\pi_n$ gives a lift
\[
 s\colon B_n(t)\longrightarrow\mathbb D,\qquad s(z_n)=0,
 \qquad\pi_n\circ s=\iota,
\]
where $\iota \colon B_n(t) \to U_n$ is the inclusion.
The lift is holomorphic since $\pi_n$ is locally biholomorphic. 
On $r_t\mathbb D$, the map $s\circ\pi_n$ and the inclusion $\iota \colon r_t\mathbb D \to \mathbb D$ both fix $0$ and are both lifts of the same map with respect to $\pi_n$.
The uniqueness of lifts thus yields
\[
 s\circ\pi_n=\operatorname{id}_{r_t\mathbb D},
\]
hence $\pi_n$ is injective on $r_t\mathbb D$. Surjectivity onto the ball follows
from \eqref{eq:ball-uniformization}.

Using covering invariance and change of variables, we obtain
\begin{align*}
 {\mathcal A}_{\rho_{U_n}}(B_n(t))
 &=\int_{r_t\mathbb D}\rho_{\mathbb D}^2\,\mathrm dx\,\mathrm dy
 =2\pi\int_0^{\tanh t}\frac{r}{(1-r^2)^2}\,\mathrm d r\\
 &=\frac{\pi\tanh^2t}{1-\tanh^2t}
 =\pi\sinh^2t.
\end{align*}

\label{lem:injectivity}
For every fixed $t>0$, there is $N(t)$ such that $f|_{B_n(t)}$ is injective
for all $n\geqslant N(t)$.

Suppose otherwise. Choose $n_k\to\infty$ such that $f|_{B_{n_k}(t)}$ is
noninjective. Pass to a subsequence with $z_{n_k}\to a\in S$. The point $a$

<!-- proof-11: body_md -->
The proof is similar to the proof of Theorem~\ref{thm:mainA}.

Suppose that some $z_0\in U$ has a bounded orbit. 
Let $K$ be the compact pole-free set from \eqref{eq:pole-free-K}.
Lemma~\ref{lem:deficit}, applied with $Q=P_1$, gives a constant $C<\infty$ such that
\begin{equation}\label{eq:uniform-deficit-K}
 \int_K\left[\rho_j^2-(f^*\rho_j)^2\right]_+\,\mathrm dx\,\mathrm dy\leqslant C
 \qquad(j\geqslant1).
\end{equation}
Choose $t>0$ so that
\begin{equation}\label{eq:choose-t}
 \pi\sinh^2t>C.
\end{equation}
By Lemmas~\ref{lem:shrinking}, \ref{lem:embedded-balls}, and
\ref{lem:injectivity}, choose $N$ such that for every $n\geqslant N$,
\begin{equation}\label{eq:tail-properties}
 \begin{gathered}
 B_n(t)\subset K,\qquad f|_{B_n(t)}\text{ is injective},\\
 {\mathcal A}_{\rho_{U_n}}(B_n(t))=\pi\sinh^2t.
 \end{gathered}
\end{equation}
We also have $f(B_n(t))\subset B_{n+1}(t)$.

Fix $j$ and put $a_{j,n}:={\mathcal A}_{\rho_j}(B_n(t))$. These quantities are finite
by \eqref{eq:finite-area-j}. Forward inclusion, injectivity, and change of
variables give, for $n\geqslant N$,
\begin{align}
 a_{j,n}-a_{j,n+1}
 &\leqslant {\mathcal A}_{\rho_j}(B_n(t))-{\mathcal A}_{\rho_j}(f(B_n(t)))\notag\\
 &=\int_{B_n(t)}\bigl(\rho_j^2-(f^*\rho_j)^2\bigr)\,\mathrm dx\,\mathrm dy\notag\\
 &\leqslant\int_{B_n(t)}
 \left[\rho_j^2-(f^*\rho_j)^2\right]_+\,\mathrm dx\,\mathrm dy.
 \label{eq:ball-one-step}
\end{align}
The balls lie in pairwise disjoint Fatou components and are contained in $K$.
Summing \eqref{eq:ball-one-step} from $n=N$ to $n=L > N$ and using
\eqref{eq:uniform-deficit-K} yields
\begin{equation}\label{eq:ball-telescoping}
 a_{j,N}-a_{j,L+1}\leqslant C.
\end{equation}
For this fixed $j$, disjointness and finite total area imply
\[
 \sum_{n\geqslant N}a_{j,n}
 \leqslant {\mathcal A}_{\rho_j}(\mathbb C\setminus P_j)<\infty.
\]
Hence $a_{j,L+1}\to0$ as $L \to \infty$. Letting $L\to\infty$ in
\eqref{eq:ball-telescoping} gives
\[
 {\mathcal A}_{\rho_j}(B_N(t))\leqslant C\qquad(j\geqslant1).
\]
Now let $j\to\infty$. By \eqref{eq:fatou-recovery} and monotone convergence,
\[
 {\mathcal A}_{\rho_{U_N}}(B_N(t))\leqslant C.
\]
This contradicts \eqref{eq:tail-properties} and \eqref{eq:choose-t}.
Therefore, no point of $U$ has a bounded orbit, proving
Theorem \ref{thm:mainB}.
