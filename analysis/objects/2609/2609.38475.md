<!-- proposition-1: body_md -->
For any test function $\varphi(v)\in C^\infty(\mathbb{R}^3)$, we have the following identity:
\begin{multline} \label{weak formula for summation of boltzmann collision integrals}
\eta \int_{\mathbb{R}^3} \mathcal{B}_1[f,F](w) \varphi(w)\, \mathrm{d}w 
+ \int_{\mathbb{R}^3} \mathcal{B}_2[F,f](v) \varphi(v)\, \mathrm{d}v \\
= \int_{\mathbb{R}^3 \times \mathbb{S}^2 \times \mathbb{R}^3} 
f(w) F(v) b(v - w, n) 
\left[\eta\varphi(w') + \varphi(v') -\eta \varphi(w) - \varphi(v) \right]
\, \mathrm{d}n \, \mathrm{d}v \, \mathrm{d}w.
\end{multline}

<!-- proposition-2: body_md -->
\label{prop:entropy inequality for BB system}
Let $f$ and $F$ be classical solutions of \eqref{coupled boltzmann equations}. Assume that $(x,v)\mapsto F(t,x,v)$ and $(x,w)\mapsto f(t,x,w)$ decay rapidly at infinity, while $(x,w)\mapsto \ln f(t,x,w)$ and $(x,v)\mapsto \ln F(t,x,v)$ have at most polynomial growth at infinity. Then
\begin{equation} 
    \frac{\mathrm{d}}{\mathrm{d}t} S_{\mathrm{BB}}\le 0.
\end{equation}

<!-- theorem-1: body_md -->
\label{prop:formal derivation of coupled Enskog equations}
Consider the two-species hard-sphere system described in Section \ref{sec:derive coupled kinetic equations}, with elastic collisions and indistinguishable particles within each species.
Assume that its joint distribution satisfies the Liouville equation \eqref{liouville equation} and the collision boundary conditions \eqref{boundary condition for g}.

Let $f$ and $F$ denote the one-particle mass density distribution functions of the gas molecules and particles, respectively.
Assume that the two-particle distributions satisfy the Enskog closure relations \eqref{def:correlation functions}, including at the collision boundaries.  
Assume sufficient regularity, integrability, and decay for the integrations and changes of variables used in the derivation.

Then the first equations of the two-species BBGKY hierarchy
formally reduce to the coupled Enskog system
\begin{equation}\label{coupled Enskog equations}
\left\{
\begin{aligned}
    &\partial_tf+w\cdot\nabla_{x}f=\mathcal{E}_0[f]+\mathcal{E}_1[f,F],\\
    &\partial_tF+v\cdot\nabla_{x}F=\mathcal{E}_2[F,f]+\mathcal{E}_3[F],
\end{aligned}\right.
\end{equation}
where the collision operators are defined in
\eqref{def:collision integral E_0},
\eqref{def:collision integral E_1},
\eqref{def:collision integral E_2}, and
\eqref{def:collision integral E_3}.

<!-- proposition-3: body_md -->
\label{prop:weak formula for enskog collision integral}
Assume that the cross-collision operators in \eqref{def:collision integral E_1} and \eqref{def:collision integral E_2} have unit contact factors and obey the collision laws \eqref{collision formula between different species}. For any smooth scalar or vector test function $\varphi(x,v)$, we have
\begin{multline}\label{weak formula for summation of enskog collision integrals}
\int_{\mathbb{R}^3}\mathcal{E}_1[f,F](x,w)\varphi(x,w)\mathrm{d} w
+\int_{\mathbb{R}^3}\mathcal{E}_2[F,f](x,v)\varphi(x,v)\mathrm{d} v
+\operatorname{div}_x I[f,F;\varphi](x)\\
=\frac{(r_g+r_p)^2}{m_g}
\int_{\mathbb{R}^3\times\mathbb{R}^3\times\mathbb{S}^2}
f(x+(r_g+r_p)n,w)F(x,v)[(v-w)\cdot n]_+\\
\times\bigl[\eta\bigl(\varphi(x+(r_g+r_p)n,w')-\varphi(x+(r_g+r_p)n,w)\bigr)
+\varphi(x,v')-\varphi(x,v)\bigr]\mathrm{d} n\mathrm{d} v\mathrm{d} w,
\end{multline}
where $[z]_+=\max\{z,0\}$ and
\begin{multline}\label{def:cross-collisional-flux}
I[f,F;\varphi](x):=\frac{(r_g+r_p)^2}{m_p}
\int_0^{r_g+r_p}\int_{\mathbb{R}^3\times\mathbb{R}^3\times\mathbb{S}^2}
n[(v-w)\cdot n]_+\\
\times\bigl(\varphi(x+sn,w')-\varphi(x+sn,w)\bigr)
f(x+sn,w)\\
\times F(x-(r_g+r_p-s)n,v)\mathrm{d} n\mathrm{d} v\mathrm{d} w\mathrm{d} s.
\end{multline}

<!-- proposition-4: body_md -->
\label{prop:entropy inequality for EE system}
Let $f$ and $F$ be classical solutions of \eqref{coupled Enskog equations}. Assume that $(x,v)\mapsto F(t,x,v)$ and $(x,w)\mapsto f(t,x,w)$ decay rapidly at infinity, while $(x,w)\mapsto \ln f(t,x,w)$ and $(x,v)\mapsto \ln F(t,x,v)$ have at most polynomial growth at infinity. Then, for the Boltzmann--Enskog collision integrals ($Y^{(1,1)}=Y^{(0,2)}=Y^{(2,0)}\equiv 1$), we have 
\begin{equation} 
    \frac{\mathrm{d}}{\mathrm{d}t} S_{\mathrm{EE}}=-\Lambda\le 0,
\end{equation}
where $\Lambda$ is defined by
\begin{multline*}
\Lambda:=-\frac{1}{2m_g^2}\int_{(\mathbb{R}^3)^3\times\mathbb{S}^2}f(t,x+2r_gn,w_1)\lambda\left(\frac{f(t,x,w^o)f(t,x+2r_gn,w_1^o)}{f(t,x,w)f(t,x+2r_gn,w_1)}\right)\\
    \times f(t,x,w)(w-w_1)\cdot nH((w-w_1)\cdot n)\mathrm{d}  x\mathrm{d}  n\mathrm{d}  w\mathrm{d}  w_1\\
    -\frac{1}{2m_p^2}\int_{(\mathbb{R}^3)^3\times\mathbb{S}^2}F(t,x,v)F(t,x+2r_pn,v_1)\lambda\left(\frac{F(t,x,v^o)F(t,x+2r_pn,v_1^o)}{F(t,x,v)F(t,x+2r_pn,v_1)}\right)\\
    \times (v-v_1)\cdot nH((v-v_1)\cdot n)\mathrm{d}  x\mathrm{d}  n\mathrm{d}  v\mathrm{d}  v_1\\
    -\frac{1}{m_gm_p}\int_{(\mathbb{R}^3)^3\times\mathbb{S}^2}f(t,x+(r_g+r_p)n,w)\lambda\left(\frac{f(t,x+(r_g+r_p)n,w')F(t,x,v')}{f(t,x+(r_g+r_p)n,w)F(t,x,v)}\right)\\
    \times F(t,x,v)(v-w)\cdot nH((v-w)\cdot n)\mathrm{d}  x\mathrm{d}  n\mathrm{d}  v\mathrm{d}  w,
\end{multline*}
where $v',w',v_*^o,v^o,w^o,w_*^o$ are given by \eqref{collision formula between same species}, \eqref{collision formula between different species}, and $\lambda(z):=z-1-\ln z$.

<!-- proof-1: body_md -->
\label{sec:proof of entropy inequality}
Similarly, we directly compute the time derivative of $S_{\mathrm{EE}}$. We decompose it into $3$ terms:
\[
\frac{\mathrm{d}}{\mathrm{d}t} S_{\mathrm{EE}}=\Gamma_1+\Gamma_2+\Gamma_3,
\]
where
\begin{align*}
    \Gamma_1:=&\int_{\mathbb{R}^3\times\mathbb{R}^3}\mathcal{E}_0[f]\ln f(t,x,w)\mathrm{d}  x\mathrm{d}  w+\frac{\mathrm{d}}{\mathrm{d}t}\left( \frac{1}{2m_g^2}\int_{\mathbb{R}^3}\beta_{2r_g}(|\cdot|)\star \rho_g \rho_g(t,x) \mathrm{d}  x\right)\\
    \Gamma_2:=&\int_{\mathbb{R}^3\times\mathbb{R}^3}\mathcal{E}_3[F]\ln F(t,x,v)\mathrm{d}  x\mathrm{d}  v+\frac{\mathrm{d}}{\mathrm{d}t}\left( \frac{1}{2m_p^2}\int_{\mathbb{R}^3}\beta_{2r_p}(|\cdot|)\star \rho_\ell \rho_\ell(t,x) \mathrm{d}  x\right)\\
    \Gamma_3:=&\int_{\mathbb{R}^3\times \mathbb{R}^3}\mathcal{E}_1[f,F]\ln f\mathrm{d}  x\mathrm{d}  w+\int_{\mathbb{R}^3\times \mathbb{R}^3}\mathcal{E}_2[F,f]\ln F\mathrm{d}  x\mathrm{d}  v\\
    &\qquad +\frac{1}{m_gm_p}\frac{\mathrm{d}}{\mathrm{d}t} \int_{\mathbb{R}^3\times \mathbb{R}^3}\beta_{r_g+r_p}(|x-y|)\rho_\ell(x)\rho_g(y)\mathrm{d}  x\mathrm{d}  y.
\end{align*}
We begin by considering $\Gamma_3$.
\begin{align*}
    &m_p\int_{\mathbb{R}^3\times \mathbb{R}^3}\mathcal{E}_1[f,F]\ln f\mathrm{d}  x\mathrm{d}  w+m_g\int_{\mathbb{R}^3\times \mathbb{R}^3}\mathcal{E}_2[F,f]\ln F\mathrm{d}  x\mathrm{d}  v\\
    =&(r_g+r_p)^2\int_{(v-w)\cdot n>0} (v-w)\cdot nf(x+(r_g+r_p)n,w)F(x,v)\\
    &\qquad\qquad\times\ln\frac{f(x+(r_g+r_p)n,w')F(x,v')}{f(x+(r_g+r_p)n,w)F(x,v)}\mathrm{d}  x\mathrm{d}  v\mathrm{d}  w\mathrm{d}  n. 
\end{align*}
By the following fundamental inequality
\begin{equation}\label{fundamental inequality}
\lambda(z):=z-1-\ln z\ge0, \text{for~}z>0 ,
\end{equation}
the quantity above is no larger than
\begin{align*}
    &(r_g+r_p)^2 \int_{(v-w)\cdot n>0}[f(x+(r_g+r_p)n,w')F(x,v')-f(x+(r_g+r_p)n,w)F(x,v)]\\
    &\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\times(v-w)\cdot n\mathrm{d}  x\mathrm{d}  v\mathrm{d}  w\mathrm{d}  n\\
    =&-(r_g+r_p)^2\int_{\mathbb{R}^3\times\mathbb{R}^3\times\mathbb{R}^3\times\mathbb{S}^2}(v-w)\cdot n f(x+(r_g+r_p)n,w)F(x,v)\mathrm{d}  x\mathrm{d}  v\mathrm{d}  w\mathrm{d}  n\\
    =&\int_{(\mathbb{R}^3)^4}(v-w)\cdot \frac{x-y}{|x-y|} f(y,w)F(x,v)\delta(|x-y|-(r_g+r_p))\mathrm{d}  x\mathrm{d}  y\mathrm{d} v\mathrm{d} w\\
    =&\int_{(\mathbb{R}^3)^4}\rho_\ell(x)\rho_g(y)\frac{x-y}{|x-y|}\cdot [u_\ell(x)-u_g(y)]\delta(|x-y|-(r_g+r_p))\mathrm{d}  x\mathrm{d}  y\mathrm{d} v\mathrm{d} w.
\end{align*}
On the other hand, the equations \eqref{local conservation laws for EE system}, together with integration by parts, yield
\begin{align*}
    &\frac{\mathrm{d}}{\mathrm{d}t} \int_{\mathbb{R}^3\times \mathbb{R}^3}\beta_{r_g+r_p}(|x-y|)\rho_\ell(x)\rho_g(y)\mathrm{d}  x\mathrm{d}  y\\
    =&-\int_{\mathbb{R}^3\times \mathbb{R}^3}\beta_{r_g+r_p}(|x-y|)\operatorname{div}_x(\rho_\ell(x)u_\ell(x))\rho_g(y)\mathrm{d}  x\mathrm{d}  y\\
    &\quad-\int_{\mathbb{R}^3\times \mathbb{R}^3}\beta_{r_g+r_p}(|x-y|)\rho_\ell(x)\operatorname{div}_y(\rho_g(y)u_g(y))\mathrm{d}  x\mathrm{d}  y\\
    =&\int_{\mathbb{R}^3\times \mathbb{R}^3}\rho_\ell(x)\rho_g(y)\{\nabla_x[\beta_{r_g+r_p}(|x-y|)]\cdot u_\ell(x)+\nabla_y[\beta_{r_g+r_p}(|x-y|)]\cdot u_g(y)\}\mathrm{d}  x\mathrm{d}  y\\
    =&\int_{\mathbb{R}^3\times \mathbb{R}^3}\rho_\ell(x)\rho_g(y)\frac{x-y}{|x-y|}\cdot [u_\ell(x)-u_g(y)]\beta_{r_g+r_p}'(|x-y|)\mathrm{d}  x\mathrm{d}  y.
\end{align*}
Recall that $\beta_s'(r)=-\delta(r-s)$. Therefore, we can rewrite the $\Gamma_3$ term as
 \begin{multline}\label{cpt:Gamma_3}
    \Gamma_3=\frac{(r_g+r_p)^2}{m_gm_p}\int_{(\mathbb{R}^3)^3\times\mathbb{S}^2}f(t,x+(r_g+r_p)n,w)\lambda\left(\frac{f(t,x+(r_g+r_p)n,w')F(t,x,v')}{f(t,x+(r_g+r_p)n,w)F(t,x,v)}\right)\\
    \times F(t,x,v)(v-w)\cdot nH((v-w)\cdot n)\mathrm{d}  x\mathrm{d}  n\mathrm{d}  v\mathrm{d}  w\le 0.
\end{multline} 
For the same reason, or by Theorem 3.4 in \cite{CCG2026}, we have the following estimates:
\begin{multline}\label{cpt:Gamma_1}
    \Gamma_1=\frac{2r_g^2}{m_g^2}\int_{(\mathbb{R}^3)^3\times\mathbb{S}^2}f(t,x,w)f(t,x+2r_gn,w_1)\lambda\left(\frac{f(t,x,w^o)f(t,x+2r_gn,w_1^o)}{f(t,x,w)f(t,x+2r_gn,w_1)}\right)\\
    (w-w_1)\cdot nH((w-w_1)\cdot n)\mathrm{d}  x\mathrm{d}  n\mathrm{d}  w\mathrm{d}  w_1\le0,
\end{multline}
    and
 \begin{multline}\label{cpt:Gamma_2}
    \Gamma_2=\frac{2r_p^2}{m_p^2}\int_{(\mathbb{R}^3)^3\times\mathbb{S}^2}F(t,x,v)F(t,x+2r_pn,v_1)\lambda\left(\frac{F(t,x,v^o)F(t,x+2r_pn,v_1^o)}{F(t,x,v)F(t,x+2r_pn,v_1)}\right)\\
    (v-v_1)\cdot nH((v-v_1)\cdot n)\mathrm{d}  x\mathrm{d}  n\mathrm{d}  v\mathrm{d}  v_1\le0.
\end{multline}  
Finally, we conclude the result by combining \eqref{cpt:Gamma_1}, \eqref{cpt:Gamma_2}, and \eqref{cpt:Gamma_3}.
