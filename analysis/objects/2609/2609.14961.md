\label{definitionCFro}    
    A \textbf{ discrete $C^\ast$-Frobenius algebra} in Rep$(\mathcal{A})$ is a tuple 
    \[
     \left(\mathcal{H} \cong \bigoplus_{i\in\Lambda}\mathcal{H}_i, \{\mu_i^{\mathfrak{L}}\}_{i\in\Lambda}, \{\mu_j^{\mathfrak{R}}\}_{j\in\Lambda }, \{T_k\}_{k\in\Lambda}\right),
    \]
    where each $ \mathcal{H}_i$ is a simple object in $\operatorname{Rep}^{\operatorname{d}}(\mathcal{A}), \mu_i^{\mathfrak{L}}\in\operatorname{Hom}_{\mathcal{A}}(\mathcal{H}_i\boxtimes \mathcal{H}, \mathcal{H}), \mu_j^{\mathfrak{R}} \in \operatorname{Hom}_{\mathcal{A}}(\mathcal{H} \boxtimes\mathcal{H}_j, \mathcal{H}), T_k\in\operatorname{Hom}_{\mathcal{A}}(\mathcal{H}, \mathcal{H}_k)$, for $i,j,k \in\Lambda$ ($\Lambda$ is called to be the \textbf{labeling set} of the discrete $C^*$-Frobenius algebra), such that the following conditions are satisfied: 
   \begin{enumerate}
       \item (\textbf{Unit}):  for any $i,j \in \Lambda$, $T_i T_j^* = \delta_{i,j}\operatorname{id}_i$, $T_i^*T_i$ is a projection and $\sum_{i\in\Lambda} T_i^*T_i = 1$ with respect to the strong operator topology. In addition, the following identities are satisfied:
       \begin{align*}
           \mu^{\mathfrak{L}}_i(\operatorname{id}_i \boxtimes T_0^* )&= T_i^* \\ 
           \mu_j^{\mathfrak{R}}(T_0^* \boxtimes \operatorname{id}_j) &= T^*_j.
       \end{align*}
       \item (\textbf{Associativity and Frobenius condition}): the following diagrams commute adjointly for any $i,j\in\Lambda$: 
       \[\begin{tikzcd}
\mathcal{H}_i\boxtimes \mathcal{H} \boxtimes \mathcal{H}_j \arrow{r}{\mu_i^{\mathfrak{L}}\boxtimes\operatorname{id}_j} \arrow[swap]{d}{\operatorname{id}_i\boxtimes\mu_j^{\mathfrak{R}}} & \mathcal{H}\boxtimes\mathcal{H}_j \arrow{d}{\mu_j^{\mathfrak{R}}} \\ 
\mathcal{H}_i\boxtimes \mathcal{H} \arrow{r}{\mu_i^{\mathfrak{L}}}& \mathcal{H}.\\
\end{tikzcd}
\]
     \end{enumerate}

<!-- remark-1: body_md -->
The ordinary $C^*$-Frobenius algebra \cite[Definition 2.1]{gui2025connes} is a discrete $C^*$-Frobenius algebra whose labeling set $F$ is a finite set. Let 
    $\left(\mathcal{H}_a \cong \bigoplus_{i\in F}\mathcal{H}_i, \mu , \iota \right)$ be a $C^*$-Frobenius algebra in $\operatorname{Rep}(\mathcal{A})$, where $F$ is a finite set. Choose
    $\{T_i\}_{i\in F}$ such that $\iota = T_0^*$, for any $i\in F$,  $T_iT_j^* = \delta_{i,j}\operatorname{id}_i$, $T^*_iT_i$ is a projection and $\sum_{i\in F}T^*_iT_i = 1$.
    
    For $i\in F$, define: 
    \begin{align*}
        \mu_i^\mathfrak{L} &= \mu \circ (T^*_i \boxtimes \operatorname{id}_a)\in \operatorname{Hom}_\mathcal{A}(\mathcal{H}_i \boxtimes \mathcal{H}_a, \mathcal{H}_a), \\
        \mu_i^\mathfrak{R} &= \mu \circ(\operatorname{id}_a \boxtimes T_i^*) \in \operatorname{Hom}_\mathcal{A}(\mathcal{H}_a \boxtimes \mathcal{H}_i, \mathcal{H}_a). 
    \end{align*}
    We will show that  $\left(\mathcal{H}_a \cong \bigoplus_{i\in F}\mathcal{H}_i, \{\mu^\mathfrak{L}_i\}_{i\in F}, \{\mu^\mathfrak{R}_i\}_{i\in F} , \{T_i\}_{i\in F}   \right)$ is a discrete $C^*$-Frobenius algebra. 
    For any $i\in F$, by the unit condition of the ordinary $C^*$-Frobenius algebra, we have the unit condition: 
\begin{align*}
    \mu_i^\mathfrak{L}(\operatorname{id}_i \boxtimes T_0^*) &= \mu\circ (\operatorname{id}_a \boxtimes T_0^*)(T^*_i \boxtimes \operatorname{id}_0) = T_i^* \\
   \mu_i^\mathfrak{R}(T_0^* \boxtimes \operatorname{id}_i) &= \mu \circ (T_0^*\boxtimes \operatorname{id}_a)(\operatorname{id}_0\boxtimes T_i^*) = T_i^*. 
\end{align*}
The associativity and Frobenius condition follows from the adjoint commutativity of the following diagram:
\[
\begin{tikzcd}
    \mathcal{H}_i\boxtimes \mathcal{H}_a \boxtimes \mathcal{H}_j \  \arrow{r}{T_i^*\boxtimes\operatorname{id}\boxtimes\operatorname{id}} \arrow[swap]{d}{\operatorname{id}\boxtimes \operatorname{id} \boxtimes T_j^*} &\mathcal{H}_a\boxtimes \mathcal{H}_a\boxtimes \mathcal{H}_j \arrow{r}{\mu\boxtimes\operatorname{id}} \arrow{d}{\operatorname{id} \boxtimes \operatorname{id} \boxtimes T_j^*}  & \mathcal{H}_a\boxtimes \mathcal{H}_j \arrow{d}{\operatorname{id}\boxtimes T_j^*}\\
    \mathcal{H}_i \boxtimes \mathcal{H}_a \boxtimes \mathcal{H}_a  \arrow{r}{T_i^*\boxtimes\operatorname{id} \boxtimes \operatorname{id}} \arrow{d}{\operatorname{id}\boxtimes\mu}    &\mathcal{H}_a \boxtimes \mathcal{H}_a \boxtimes \mathcal{H}_a \arrow{r}{\mu\boxtimes\operatorname{id}} \arrow{d}{\operatorname{id}\boxtimes\mu}     & \mathcal{H}_a \boxtimes \mathcal{H}_a \arrow{d}{\mu} \\
    \mathcal{H}_i \boxtimes \mathcal{H}_a  \arrow{r}{T_i^* \boxtimes \operatorname{id}}                   &\mathcal{H}_a \boxtimes \mathcal{H}_a  \arrow{r}{\mu}       & \mathcal{H}_a,
\end{tikzcd}
\]
where subscripts on identity morphisms are suppressed for simplicity.. The adjoint commutativity of the lower-right diagram is nothing but the 
associativity and the Frobenius condition of the ordinary $C^*$-Frobenius algebra. It is easy to see the remaining diagrams commute adjointly.

<!-- proof-1: body_md -->
By associativity, we have the following commuting diagram: 
 \[\begin{tikzcd}

\mathcal{H}_0\boxtimes \mathcal{H} \arrow{r}{\mu_0^{\mathfrak{L}}}& \mathcal{H} ,
\end{tikzcd}
\]
which implies 
\[
\mu_i^{\mathfrak{R}}\left( \mu_0^{\mathfrak{L}}(\operatorname{id}_0\boxtimes T_0^*)\boxtimes\operatorname{id}_i\right)  = 
\mu_0^{\mathfrak{L}}\left( \operatorname{id_0}\boxtimes \mu_i^{\mathfrak{R}}(T_0^*\boxtimes\operatorname{id}_i) \right).
\]
Then by the unit condition, we have $\mu_0^{\mathfrak{L}}(\operatorname{id}_0 \boxtimes T_i^*) = T_i^*$. The other equality,  $T_i^*  =  \mu_0^\mathfrak{R}(T_i^*\boxtimes \operatorname{id}_0)$ is proved similarly.

<!-- definition-3: body_md -->
A \textbf{(local) M\"{o}bius covariant extension} of $\mathcal{A}$  is a family of von Neumann algebras $\left\{\mathcal{B}(\widetilde{I}) \right\}_{\widetilde{I}\in\widetilde{\mathcal{I}}}$ on a fixed  Hilbert space
    $\mathcal{H}$ such that $\mathcal{H}$ is an $\mathcal{A}$-module and the following conditions are satisfied:
    \begin{enumerate}[(A)]
        \item (Extension) For any $\widetilde{I}\in\widetilde{\mathcal{I}},\ \pi_{\mathcal{H},I}\left(\mathcal{A}(I)\right) \subset \mathcal{B}(\widetilde{I}) .$ 
        \item (Isotony) If $\widetilde{I} \subset \widetilde{J}$, then $\mathcal{B}(\widetilde{I}) \subset \mathcal{B}(\widetilde{J}). $ 
         \item (Locality) If $I \cap J = \emptyset$, then $[\mathcal{B}(\widetilde{I}), \mathcal{B}(\widetilde{J})] = \{0\}$.
        \item (M\"{o}bius covariance) There is a strongly continuous unitary representation $U$  of $\widetilde{PSU}(1,1)$ on $\mathcal{H}$ such that 
        \[
        U(g)\mathcal{B}(\widetilde{I})U(g)^*  = \mathcal{B}(g\widetilde{I}),
        \]
        for any $g\in \widetilde{PSU}(1,1), \widetilde{I} \in \widetilde{\mathcal{I}}.$
        \item (Positive energy) The generator of the rotation subgroup is positive.
        \item (Vacuum) There is a $\widetilde{PSU}(1,1)$-invariant vector $\widetilde{\Omega}$, called the vacuum vector, such that 
        $\widetilde{\Omega}$ is cyclic for $\bigvee_{\widetilde{I}\in\widetilde{\mathcal{I}}}\mathcal{B}(\widetilde{I}).$
    \end{enumerate}

<!-- proof-6: body_md -->

<!-- theorem-4: body_md -->
\label{discreteLRconstruction}
    The family of  von Neumann algebras $\left\{ \mathcal{B}(\widetilde{I} \times \widetilde{J} )\right\}_{\widetilde{I},\ \widetilde{J} \in \widetilde{\mathcal{I}}}$  satisfies the following properties:  
    \begin{enumerate}
        \item (Extension) For any $\widetilde{I}, \widetilde{J} \in \widetilde{\mathcal{I}}$, 
        we have  $ \pi_{\mathcal{H}_\bullet, I \times J}(\mathcal{A}(I)\otimes\mathcal{A}^\mathfrak{r}(J)) \subset  \mathcal{B}(\widetilde{I}\times\widetilde{J})  $. 
        \item (Isotony) If $\widetilde{I_1} \subset \widetilde{I_2}$ and $\widetilde{J_1} \subset \widetilde{J_2}$, then $\mathcal{B}(\widetilde{I_1} \times \widetilde{J_1}) \subset \mathcal{B}(\widetilde{I_2} \times \widetilde{J_2})$.
        \item (M\"{o}bius covariance) There is a strongly continuous unitary representation $U$
of 
        $\widetilde{PSU}(1,1) \times \widetilde{PSU}(1,1) $ on $\mathcal{H}_\bullet$ such that for any $(g_1,g_2)\in \widetilde{PSU}(1,1)\times \widetilde{PSU}(1,1)$ and $\widetilde{I}, \widetilde{J}\in\widetilde{\mathcal{I}}$, 
        \[
        U(g_1, g_2) \mathcal{B}(\widetilde{I}\times \widetilde{J} ) U(g_1, g_2) ^* = \mathcal{B}(g_1\widetilde{I}\times g_2\widetilde{J} ).
        \]
        \item (Positive energy) The generator of the one-parameter rotation subgroup is positive.
        \item (Locality)  Let $\widetilde{I_1}, \widetilde{J_1}, \widetilde{I_2}, \widetilde{J_2}\in \widetilde{\mathcal{I}}$. If $\widetilde{I_1}$ is anticlockwise to  $\widetilde{I_2}$ and $\widetilde{J_1} $ is clockwise to $ \widetilde{J_2}$ , then  
    \[
   [ \mathcal{B}(\widetilde{I_1}\times\widetilde{J_1}), \mathcal{B}(\widetilde{I_2}\times\widetilde{J_2})    ] =\{0\}.
    \]
        \item (Vacuum) $\widetilde{T_0}^*(\Omega\otimes\Omega)$ is a $\widetilde{PSU}(1,1) \times \widetilde{PSU}(1,1)$ invariant vector and is cyclic for $\mathcal{B}(\widetilde{I}\times \widetilde{J} )$, for any  $\widetilde{I} , \widetilde{J} \in \widetilde{\mathcal{I}}.$
    \end{enumerate}
    In addition, for any $\widetilde{I}, \widetilde{J} \in\widetilde{\mathcal{I}}$, the von Neumann algebra $\mathcal{B}(\widetilde{I}\times\widetilde{J})$ depends only on $I, J$ and not on the choices of the arg-functions.  The strongly continuous unitary representation $U$ of $\widetilde{PSU}(1,1) \times \widetilde{PSU}(1,1) $ on $\mathcal{H}_\bullet$ factors through $PSU(1,1) \times PSU(1,1)$.
