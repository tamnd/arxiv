<!-- assumption-1: body_md -->
\label{ass:filtering}
The functions
\[
 b:\mathbb R^d\times U\to\mathbb R^d,
 \qquad
 \sigma:\mathbb R^d\times U\to\mathbb R^{d\times r},
 \qquad
 h:\mathbb R^d\times U\to\mathbb R^m
\]
are Borel measurable.  There is a constant $K\ge1$ such that, for all $x,y\in\mathbb R^d$ and $v\in U$,
\begin{align}
 &\left\lvert b(x,v)-b(y,v)\right\rvert+\left\lVert \sigma(x,v)-\sigma(y,v)\right\rVert
   \le K\left\lvert x-y\right\rvert,
   \label{eq:filtering-lipschitz}\\
 &\left\lvert b(x,v)\right\rvert+\left\lVert \sigma(x,v)\right\rVert+\left\lvert h(x,v)\right\rvert
   \le K(1+\left\lvert x\right\rvert+\left\lvert v\right\rvert).
   \label{eq:filtering-growth}
\end{align}

<!-- lemma-1: body_md -->
\label{lem:signal-moments}
Suppose \Cref{ass:filtering} holds.  Let $q\ge2$, and let $\widetilde{\mathbb P}$ be a probability measure equivalent to $\mathbb P$ such that $W$ remains an $\mathbb R^r$-valued Brownian motion under $\widetilde{\mathbb P}$.  If
\[
    \mathbb E^{\widetilde{\mathbb P}}|X_0|^q<\infty,
\]
then
\begin{equation}
  \mathbb E^{\widetilde{\mathbb P}}\sup_{0\le s\le\tau_k}\left\langle X_s\right\rangle^{q}
  \le C_{k,q,T,K}\left(1+\mathbb E^{\widetilde{\mathbb P}}\left\lvert X_0\right\rvert^{q}\right).
  \label{eq:signal-moment}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Set $\eta_n:=\inf\{t\in[0,T]:|X_t|\ge n\}\wedge T$ and first stop the
state equation at $\tau_k\wedge\eta_n$.  In the estimates displayed below,
each occurrence of $\tau_k$ is temporarily understood as
$\tau_k\wedge\eta_n$.  The linear-growth bound, H\"older's inequality in
time, and the Burkholder--Davis--Gundy inequality then apply without assuming
in advance the finiteness of the quantity being estimated, and their
constants are independent of $n$.

\begin{align*}
 \mathbb E^{\widetilde{\mathbb P}}\sup_{r\le t\wedge\tau_k}|X_r|^q
 \le C_q\Bigg(&\mathbb E^{\widetilde{\mathbb P}}|X_0|^q
 +\mathbb E^{\widetilde{\mathbb P}}\Big(\int_0^{t\wedge\tau_k}
        (1+|X_s|+|u_s|)\,\mathrm d s\Big)^q\\
 &+\mathbb E^{\widetilde{\mathbb P}}\Big(\int_0^{t\wedge\tau_k}
        (1+|X_s|+|u_s|)^2\,\mathrm d s\Big)^{q/2}\Bigg).
\end{align*}
The control terms are bounded pathwise by
\[
 \int_0^{\tau_k}|u_s|\,\mathrm d s\le T^{1/2}k^{1/2},
 \qquad
 \left(\int_0^{\tau_k}|u_s|^2\,\mathrm d s\right)^{q/2}\le k^{q/2}.
\]
For the state terms, H\"older's inequality yields
\[
 \left(\int_0^{t\wedge\tau_k}|X_s|\,\mathrm d s\right)^q
 +\left(\int_0^{t\wedge\tau_k}|X_s|^2\,\mathrm d s\right)^{q/2}
 \le C_{q,T}\int_0^t\sup_{a\le s\wedge\tau_k}|X_a|^q\,\mathrm d s.
\]
Consequently,
\[
 \mathbb E^{\widetilde{\mathbb P}}\sup_{r\le t\wedge\tau_k}|X_r|^q
 \le C_{k,q,T,K}\left(1+\mathbb E^{\widetilde{\mathbb P}}|X_0|^q
 +\int_0^t\mathbb E^{\widetilde{\mathbb P}}\sup_{a\le s\wedge\tau_k}|X_a|^q\,\mathrm d s\right).
\]
Gronwall's lemma gives the bound uniformly in $n$.  Letting
$n\uparrow\infty$ and applying Fatou's lemma proves
\eqref{eq:signal-moment}.  Notice that only the $L^2$-energy of $u$ is used; no $L^q$-in-time assumption on $u$ is required.  See \citet[Chapter~2]{KaratzasShreve1991} for the standard SDE moment argument.

\label{prop:filtering-likelihood}
Under \Cref{ass:filtering}, $\Lambda^k$ is a uniformly integrable $\mathbb P$-martingale.  In particular,
\begin{equation*}
  \mathbb E^{\mathbb P}\Lambda_t^k=1,
  \qquad 0\le t\le T.
  \label{eq:Lambda-expectation}
\end{equation*}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
For $j\in\mathbb N$, set
\begin{equation*}
 \rho_j
 :=\inf\left\{t\in[0,T]:
       \int_0^{t\wedge\tau_k}|H_s|^2\,\mathrm d s\ge j\right\}\wedge\tau_k
 \label{eq:rho-j}
\end{equation*}
and
\[
 \Lambda_t^{k,j}
 :=\mathcal E\left(-\int_0^{\,\cdot\wedge\rho_j}H_s^\top\,\mathrm d V_s\right)_t.
\]
Since the quadratic variation of its stochastic logarithm is bounded by $j$, Novikov's criterion implies that $\Lambda^{k,j}$ is a uniformly integrable martingale.  Define $\mathbb P^j$ by
\begin{equation*}
 \frac{\,\mathrm d\mathbb P^j}{\,\mathrm d\mathbb P}\bigg|_{\mathcal F_T}=\Lambda_T^{k,j}=\Lambda_{\rho_j}^{k,j}.
 \label{eq:Pj}
\end{equation*}
The multidimensional Girsanov theorem gives that
\[
 \widehat V_t^j:=V_t+\int_0^{t\wedge\rho_j}H_s\,\mathrm d s
\]
is an $\mathbb R^m$-valued Brownian motion under $\mathbb P^j$.  The density is driven only by $V$ and $[W,V]=0$, hence $W$ remains an $\mathbb R^r$-valued Brownian motion under $\mathbb P^j$. It is also easy to show that the initial distribution is preserved. The state equation is therefore unchanged under $\mathbb P^j$.

Applying \Cref{lem:signal-moments} under $\mathbb P^j$ yields
\begin{equation}
 \sup_{j\ge1}\mathbb E^{\mathbb P^j}
       \sup_{0\le s\le\tau_k}|X_s|^2
 \le C_k\bigl(1+\mathbb E^{\mathbb P}|X_0|^2\bigr).
 \label{eq:Pj-state-bound}
\end{equation}
By \eqref{eq:filtering-growth}, \eqref{eq:tau-k}, and \eqref{eq:Pj-state-bound},
\begin{equation*}
 \sup_{j\ge1}\mathbb E^{\mathbb P^j}
       \int_0^{\rho_j}|H_s|^2\,\mathrm d s
 \le C_k\bigl(1+\mathbb E^{\mathbb P}|X_0|^2\bigr).
 \label{eq:Pj-H-bound}
\end{equation*}
Under $\mathbb P^j$,
\[
 \log\Lambda_{\rho_j}^{k,j}
 =-\int_0^{\rho_j}H_s^\top\,\mathrm d\widehat V_s^j
   +\frac12\int_0^{\rho_j}|H_s|^2\,\mathrm d s.
\]
The stopped stochastic integral is square-integrable.  Hence
\begin{equation*}
 \mathbb E^{\mathbb P}\!
 \left[\Lambda_{\rho_j}^{k,j}\log\Lambda_{\rho_j}^{k,j}\right]
 =\frac12\mathbb E^{\mathbb P^j}\int_0^{\rho_j}|H_s|^2\,\mathrm d s
 \le C_k\bigl(1+\mathbb E^{\mathbb P}|X_0|^2\bigr).
 \label{eq:filtering-entropy}
\end{equation*}
Together with $\mathbb E^{\mathbb P}\Lambda_{\rho_j}^{k,j}=1$, this bounds
$\mathbb E\Phi(\Lambda_{\rho_j}^{k,j})$ uniformly in $j$ for
$\Phi(z)=z\log z-z+1$.  Since $\Phi(z)/z\to\infty$, the de la Vall\'ee--Poussin criterion implies uniform integrability of
$\{\Lambda_{\rho_j}^{k,j}:j\ge1\}$.

The continuity of $X$ and \eqref{eq:tau-k} imply
$\int_0^{\tau_k}|H_s|^2\,\mathrm d s<\infty$ a.s..  Thus $\rho_j\uparrow\tau_k$ and
$\Lambda_{\rho_j}^{k,j}\to\Lambda_T^k$ a.s..  Uniform integrability gives convergence in $L^1(\mathbb P)$ and
$\mathbb E^{\mathbb P}\Lambda_T^k=1$.  A nonnegative local martingale whose terminal expectation equals its initial value is closed by its terminal value.  Consequently
$\Lambda_t^k=\mathbb E^{\mathbb P}[\Lambda_T^k\mid\mathcal F_t]$ and $\Lambda^k$ is uniformly integrable.

<!-- lemma-2: body_md -->
\label{lem:reference-stopped-prp}
\label{lem:reference-stopping-functional}
For each $k$, the stopping time $\tau_k$ has an $\widetilde{\mathbb G}^k$-stopping-time
version.  With that version,
\begin{equation}
 \mathcal G_t^k
 =\widetilde{\mathcal G}^k_{t\wedge\tau_k}
 =\mathcal F^Y_{t\wedge\tau_k},
 \qquad 0\le t\le T,
 \label{eq:stopped-observation-identity}
\end{equation}
where the $\sigma$-fields at stopping times are completed by $\mathcal N$.
Moreover, $\mathbb G^k$ has the predictable representation property (PRP) with
respect to $Y^k_{\cdot\wedge\tau_k}$: every $\mathbb G^k$-local martingale
$N$ can be written
\begin{equation}
 N_t=N_0+\int_0^{t\wedge\tau_k}\zeta_s^\top\,\mathrm d Y_s^k
 \label{eq:stopped-reference-prp}
\end{equation}
for a $\mathbb G^k$-predictable integrand $\zeta$ which is locally square
integrable on $[0,\tau_k]$.

<!-- proof-3: body_md -->
Let $\mathsf C_m=C([0,T];\mathbb R^m)$, write $y^s=y_{\cdot\wedge s}$, and equip
$\mathsf C_m$ with its raw coordinate filtration.  Since $u$ is predictable
for the usual augmentation of the natural filtration of $Y$, predictable
factorization supplies a canonical-predictable Borel map
\[
 \mathfrak u:[0,T]\times\mathsf C_m\longrightarrow U,
 \qquad
 \mathfrak u(s,y)=\mathfrak u(s,y^s),
\]
such that
\begin{equation}
 u_s(\omega)=\mathfrak u(s,Y(\omega))
 \quad \,\mathrm d s\otimes\,\mathrm d\mathbb P\text{-a.e.}
 \label{eq:canonical-control-factorization}
\end{equation}
The same equality holds $\,\mathrm d s\otimes\,\mathrm d\mathbb Q^k$-a.e. because
$\mathbb Q^k\sim\mathbb P$.  By Fubini, outside one common null set the two accumulated
energies agree for every time.

For $y\in\mathsf C_m$, define the  nondecreasing continuous
functional
\begin{equation*}
 A_t(y):=\int_0^t|\mathfrak u(s,y^s)|^2\,\mathrm d s,
 \qquad
 \vartheta_k(y):=\inf\{t\in[0,T]:A_t(y)\ge k\}\wedge T.
 \label{eq:canonical-energy-hitting}
\end{equation*}
For $t<T$, the event $\{\vartheta_k\le t\}$ coincides with
$\{A_t\ge k\}$ and depends only on $y^t$; at $t=T$ it is the whole path
space.  Hence $\vartheta_k$ is a stopping-time  for the raw
coordinate filtration.  Equation
\eqref{eq:canonical-control-factorization} and Fubini give
\begin{equation}
 \tau_k=\vartheta_k(Y)
 \quad \mathbb P\text{- and }\mathbb Q^k\text{-a.s.}
 \label{eq:tau-as-Y-functional}
\end{equation}

Put $\tau=\vartheta_k(Y)$ and $\tau'=\vartheta_k(Y^k)$ on the full-probability
set on which \eqref{eq:tau-as-Y-functional} holds.  By the definition of
$Y^k$, the paths $Y$ and $Y^k$ agree on $[0,\tau]$.  Nonanticipativity of
$\mathfrak u$ therefore gives
\[
 A_t(Y^k)=A_t(Y),\qquad 0\le t\le\tau.
\]
It is then easy to see that
\begin{equation*}
 \tau_k=\vartheta_k(Y^k)\qquad \mathbb Q^k\text{-a.s.}
 \label{eq:tau-as-Yk-functional}
\end{equation*}
Replacing $\tau_k$ on a common null set by the right-hand side makes it an
$\widetilde{\mathbb G}^k$-stopping time without changing any stopped process.

Galmarino's test may now be applied in the reference Brownian filtration and
gives
\[
 \sigma(Y^k_{s\wedge\tau_k}:0\le s\le t)\vee\mathcal N
 =\widetilde{\mathcal G}^k_{t\wedge\tau_k}.
\]
On the other hand, $Y^k_{\cdot\wedge\tau_k}
=Y_{\cdot\wedge\tau_k}$.  Applying the same stopped-path identity in the
usual natural filtration of the physical observation gives
\eqref{eq:stopped-observation-identity}, including the common completion.

It remains to verify the representation assertion.  If
$\xi\in L^2(\mathcal G_T^k,\mathbb Q^k)=L^2(\widetilde{\mathcal G}^k_{\tau_k},\mathbb Q^k)$ and
\[
 \widetilde N_t:=\mathbb E^{\mathbb Q^k}[\xi\mid\widetilde{\mathcal G}_t^k],
\]
the Brownian predictable representation property in $\widetilde{\mathbb G}^k$ gives
\[
 \widetilde N_t=\widetilde N_0+\int_0^t\zeta_s^\top\,\mathrm d Y_s^k.
\]
Optional sampling and \eqref{eq:stopped-observation-identity} yield
\[
 \mathbb E^{\mathbb Q^k}[\xi\mid\mathcal G_t^k]
 =\widetilde N_{t\wedge\tau_k}
 =\widetilde N_0+
   \int_0^{t\wedge\tau_k}\zeta_s^\top\,\mathrm d Y_s^k.
\]
The stopped integrand has a $\mathbb G^k$-predictable version.  This proves
the representation for square-integrable martingales; localization and
pasting of the stopped integrands prove \eqref{eq:stopped-reference-prp} for
local martingales.

<!-- theorem-1: body_md -->
\label{thm:stopped-zakai}
For every $\varphi\in C_c^2(\mathbb R^d)$,
\begin{align}
 \left\langle \mu_t^k,\varphi\right\rangle
  ={}&\left\langle \nu_0,\varphi\right\rangle
  +\int_0^{t\wedge\tau_k}\left\langle \mu_s^k,L_s\varphi\right\rangle\,\mathrm d s
  +\sum_{a=1}^m\int_0^{t\wedge\tau_k}
       \left\langle \mu_s^k,\varphi h^a(\cdot,u_s)\right\rangle\,\mathrm d Y_s^a.
 \label{eq:zakai-stopped}
\end{align}
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
We first record the integrability which permits projection without an
$L^2(\mathbb Q)$ assumption on the density.  The growth bound, the pathwise control
energy bound, and \Cref{lem:signal-moments} under $\mathbb P$ give
\begin{equation}
 \mathbb E^{\mathbb P}\int_0^{\tau_k}|H_s|^2\,\mathrm d s<\infty.
 \label{eq:H2-under-P}
\end{equation}
Under $\mathbb P$, using $\,\mathrm d Y_s=H_s\,\mathrm d s+\,\mathrm d V_s$ on
$[0,\tau_k]$, we have
\[
 \log M^k_{\tau_k}
 =\int_0^{\tau_k}H_s^\top\,\mathrm d V_s
   +\frac12\int_0^{\tau_k}|H_s|^2\,\mathrm d s.
\]
The stochastic integral is square-integrable by \eqref{eq:H2-under-P} and has
mean zero.  Change of measure therefore yields
\begin{equation*}
 \mathbb E^{\mathbb Q}\!\left[M^k_{\tau_k}\log M^k_{\tau_k}\right]
 =\mathbb E^{\mathbb P}\!\left[\log M^k_{\tau_k}\right]
 =\frac12\mathbb E^{\mathbb P}\int_0^{\tau_k}|H_s|^2\,\mathrm d s<\infty.
 \label{eq:M-LlogL}
\end{equation*}
The change-of-measure equality is legitimate also for the absolute value:
\[
\mathbb E^{\mathbb Q}[M^k_{\tau_k}|\log M^k_{\tau_k}|]
=\mathbb E^{\mathbb P}|\log M^k_{\tau_k}|<\infty.
\]
Doob's $L\log L$ maximal inequality now gives
\begin{equation}
 \mathbb E^{\mathbb Q}\sup_{0\le s\le\tau_k}M_s^k<\infty.
 \label{eq:M-H1}
\end{equation}

Fix $\varphi\in C_c^2(\mathbb R^d)$.  Compactness of the supports of
$\varphi$, $D\varphi$, and $D^2\varphi$, together with
\eqref{eq:filtering-growth} and \eqref{eq:tau-k}, gives the pathwise bound
\begin{equation}
 \int_0^{\tau_k}\!\left(
   |\varphi(X_s)H_s|^2
   +\|D\varphi(X_s)\sigma(X_s,u_s)\|^2
   +|L_s\varphi(X_s)|
 \right)\,\mathrm d s
 \le C_{\varphi,k,T,K}.
 \label{eq:compact-test-energy}
\end{equation}
Indeed, each integrand on the left is bounded by a constant times
$1+|u_s|^2$ whenever it is nonzero.  Hence \eqref{eq:M-H1}, the
Burkholder--Davis--Gundy inequality, and \eqref{eq:compact-test-energy} imply
\begin{equation}
 \mathbb E^{\mathbb Q}\!\left(\int_0^{\tau_k}
       |M_s^k\varphi(X_s)H_s|^2\,\mathrm d s\right)^{1/2}
 +\mathbb E^{\mathbb Q}\!\left(\int_0^{\tau_k}
       \|M_s^kD\varphi(X_s)\sigma(X_s,u_s)\|^2\,\mathrm d s\right)^{1/2}
 <\infty,     \label{eq:two-H1}
 \end{equation}
 and
 \begin{equation*}   \mathbb E^{\mathbb Q}\int_0^{\tau_k}M_s^k|L_s\varphi(X_s)|\,\mathrm d s<\infty.
 \label{eq:drift-L1}
\end{equation*}
Thus both stochastic integrals in the It\^o decomposition below are
$H^1(\mathbb Q)$ martingales, and its finite-variation part is integrable.

It\^o's formula, \eqref{eq:M-k-SDE}, and $[W,Y]=0$ give
\begin{align*}
 M^k_{t\wedge\tau_k}\varphi(X_{t\wedge\tau_k})
 ={}&\varphi(X_0)
 +\int_0^{t\wedge\tau_k}M_s^kL_s\varphi(X_s)\,\mathrm d s\\
 &+\int_0^{t\wedge\tau_k}M_s^k\varphi(X_s)H_s^\top\,\mathrm d Y_s
 +\int_0^{t\wedge\tau_k}
       M_s^kD\varphi(X_s)\sigma(X_s,u_s)\,\mathrm d W_s.
\end{align*}
Take the optional projection onto $\mathbb G^k$ and the dual predictable
projection of the integrable drift.  By
\eqref{eq:predictable-pairing}, these projections are respectively
$\left\langle \mu_s^k,L_s\varphi\right\rangle$ and
$\left\langle \mu_s^k,\varphi h(\cdot,u_s)\right\rangle$.  The last $W$-integral has zero optional
projection.  Indeed, it is an $H^1$ martingale by \eqref{eq:two-H1}; every
$\mathbb G^k$-local martingale is an integral with respect to
$Y_{\cdot\wedge\tau_k}$ by the stopped PRP; and $[W,Y]=0$, so the $W$-integral
is strongly orthogonal to every $\mathbb G^k$-local martingale.  Testing
against bounded stopped $\mathbb G^k$-martingales, followed by the $H^1$
localization, shows that its optional projection is identically zero.

The standard optional-projection identity for an $H^1$ stochastic integral
with respect to the $\mathbb G^k$-Brownian motion gives
\[
 {}^{o,\mathbb G^k}\!\left(
   \int_0^{\,\cdot\wedge\tau_k}
       M_s^k\varphi(X_s)H_s^\top\,\mathrm d Y_s
 \right)
 =\int_0^{\,\cdot\wedge\tau_k}
       \left\langle \mu_s^k,\varphi h(\cdot,u_s)\right\rangle\,\mathrm d Y_s.
\]
Equivalently, this identity follows by pairing both sides with bounded
$\mathbb G^k$-predictable elementary integrands and using the stopped PRP;
\eqref{eq:two-H1} permits removal of the localization.  Finally,
$\mathcal G_0^k$ is trivial up to $\mathcal N$, and $X_0$ has law $\nu_0$
under $\mathbb Q$.  We obtain \eqref{eq:zakai-stopped} with the predictable versions
specified in \eqref{eq:predictable-pairing}.

<!-- lemma-3: body_md -->
\label{lem:mass-process}
The optional kernel $\mu^k$ can be modified on an evanescent set, and
$\overline Z^k$ can be replaced by an indistinguishable version $Z^k$, so
that
\[
 Z_t^k=\left\langle \mu_t^k,1\right\rangle
\]
for all $t$ outside one fixed null set and $Z^k$ is a strictly positive
continuous $\mathbb G^k$-martingale.  With these synchronized versions,
\begin{equation}
 \pi_t^k(\,\mathrm d x):=\frac{\mu_t^k(\,\mathrm d x)}{Z_t^k}
 \label{eq:KS-formula}
\end{equation}
is a $\mathbb G^k$-optional probability kernel and, for every bounded Borel
$\varphi$,
\[
 \left\langle \pi_t^k,\varphi\right\rangle
 =\mathbb E^{\mathbb P}\!\left[
     \varphi(X_{t\wedge\tau_k})\mid\mathcal G_t^k
   \right]
 \quad\text{a.s. for every }t.
\]
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Because $M^k$ is stopped at $\tau_k$ and is closed by $M^k_{\tau_k}$, the
tower property and \eqref{eq:mu-definition} give
\begin{equation}
 \overline Z_t^k
 =\mathbb E^{\mathbb Q}[M^k_{t\wedge\tau_k}\mid\mathcal G_t^k]
 =\mathbb E^{\mathbb Q}[M^k_{\tau_k}\mid\mathcal G_t^k].
 \label{eq:mass-closed-projection}
\end{equation}
Thus $\overline Z^k$ is the optional version of a closed
$\mathbb G^k$-martingale.  The stopped PRP in
\Cref{lem:reference-stopped-prp} supplies a continuous martingale version
$Z^k$ and a $\mathbb G^k$-predictable, locally square-integrable process
$\beta^k$, chosen to vanish after $\tau_k$, such that
\begin{equation}
 Z_t^k=1+\int_0^{t\wedge\tau_k}(\beta_s^k)^\top\,\mathrm d Y_s.
 \label{eq:mass-prp-beta}
\end{equation}

This continuous version is strictly positive simultaneously for all times.
Indeed, let
\[
 \zeta:=\inf\{t\in[0,T]:Z_t^k=0\}.
\]
On $\{\zeta<T\}$, optional sampling in
\eqref{eq:mass-closed-projection} gives
\[
 \mathbb E^{\mathbb Q}\!\left[
   \mathbf 1_{\{\zeta<T\}}M^k_{\tau_k}
 \right]
 =\mathbb E^{\mathbb Q}\!\left[
   \mathbf 1_{\{\zeta<T\}}Z_\zeta^k
 \right]=0.
\]
Since $M^k_{\tau_k}>0$ a.s.,
$\mathbb Q(\zeta<T)=0$.  Similarly,
\[
 \mathbb E^{\mathbb Q}[\mathbf 1_{\{Z_T^k=0\}}M^k_{\tau_k}]
 =\mathbb E^{\mathbb Q}[\mathbf 1_{\{Z_T^k=0\}}Z_T^k]=0,
\]
so $Z_T^k>0$ a.s. as well.  On the union of these null events set
$Z_t^k=1$ for every $t$.  The common completion makes the resulting process
adapted; it remains an indistinguishable continuous martingale version and is
now positive on every path used below.

Both $\overline Z^k=\langle\mu^k,1\rangle$ and $Z^k$ are optional versions of
the same optional projection.  By the optional section theorem their
disagreement set is evanescent.  On that set replace
$\mu_t^k$ by $Z_t^k\delta_0$, leaving it unchanged elsewhere.  This is one
optional-kernel modification and it gives
$\langle\mu_t^k,1\rangle=Z_t^k$ simultaneously for all $t$, without altering
any conditional-expectation or predictable-pairing identity.  Formula
\eqref{eq:KS-formula} now defines a probability kernel simultaneously in
time.  The fixed-time conditional-law assertion follows from the ordinary
Bayes formula
\[
 \mathbb E^{\mathbb P}[\varphi(X_{t\wedge\tau_k})\mid\mathcal G_t^k]
 =\frac{
   \mathbb E^{\mathbb Q}[M^k_{t\wedge\tau_k}
      \varphi(X_{t\wedge\tau_k})\mid\mathcal G_t^k]
 }{
   \mathbb E^{\mathbb Q}[M^k_{t\wedge\tau_k}\mid\mathcal G_t^k]
 }.
\]
See \citet[Chapter~3]{Kallianpur1980},
\citet[Chapters~3--5]{BainCrisan2009}, or
\citet[Chapter~5]{Xiong2008}.

<!-- lemma-4: body_md -->
Define, componentwise, the $\mathbb Q$-predictable projection
\begin{equation}
 c_s^k
 :={}^{p,\mathbb G^k,\mathbb Q}\!\left(
      M_s^kH_s\mathbf 1_{\{s\le\tau_k\}}
    \right),

Write
\[
 \delta\psi^n:=P_n\psi-\psi,
 \qquad \delta A^n:=P_nA-A,
 \qquad \delta B^n:=P_nB-B.

the explicit common localization
\begin{equation*}
 \vartheta_\ell
 :=\rho\wedge\inf\left\{t\ge0:
     \langle\mu_t,1\rangle
     +\int_0^{t\wedge\rho}(1+|u_s|^2)\,\mathrm d s
     \ge c_\ell\right\}.
 \label{eq:finite-rank-closure-localizer}
\end{equation*}
Narrow continuity of $\mu$, continuity of the time integral, and the stopped
control-energy bound imply $\vartheta_\ell\uparrow\rho$ a.s..  On
$[0,\vartheta_\ell]$ the mass is bounded by $c_\ell$, and, since all fields
are supported in $K$,
\[
 \sup_{x\in K}\bigl(|b(x,u_s)|+|a(x,u_s)|+|h(x,u_s)|^2\bigr)
 \le C_K(1+|u_s|^2).
\]

\begin{equation}
 \mathbb E^\mathbb Q\sup_{t\le\vartheta_\ell}
   \left|\langle\mu_t,\delta\psi_t^n\rangle\right|^2
 \le C_{K,\ell}\mathbb E^\mathbb Q\sup_{t\le\rho}
       \|\delta\psi_t^n\|_{H^{s_0}(K)}^2
 \longrightarrow0,
 \label{eq:finite-rank-closure-boundary}
 \end{equation}
 \begin{equation}
 \mathbb E^\mathbb Q\int_0^{\vartheta_\ell}
  \left|\left\langle\mu_s,
       L_s\delta\psi_s^n+\delta A_s^n+h(\cdot,u_s)^\top\delta B_s^n
       \right\rangle\right|\,\mathrm d s
 \longrightarrow0,
 \label{eq:finite-rank-closure-drift}
 \end{equation}
 and
 \begin{eqnarray}
 \mathbb E^\mathbb Q\int_0^{\vartheta_\ell}
  \left|\left\langle\mu_s,
       \delta B_s^n+\delta\psi_s^n h(\cdot,u_s)
       \right\rangle\right|^2\,\mathrm d s
 &\le& C_{K,\ell}
     \mathbb E^\mathbb Q\int_0^\rho\|\delta B_s^n\|_{H^{s_0}(K)}^2\,\mathrm d s\notag\\
 &&+C_{K,\ell}
     \mathbb E^\mathbb Q\sup_{s\le\rho}\|\delta\psi_s^n\|_{H^{s_0}(K)}^2
 \longrightarrow0.
 \label{eq:finite-rank-closure-martingale}
\end{eqnarray}
For \eqref{eq:finite-rank-closure-drift}, use the
$L^2(\Omega;L_t^1H^{s_0})$ convergence for $\delta A^n$, the
uniform-in-time convergence for $\delta\psi^n$, and Cauchy--Schwarz in time
for the term containing $\delta B^n$.  The endpoint estimate
\eqref{eq:finite-rank-closure-boundary} also applies at
$t\wedge\vartheta_\ell$, including the crossing time, because the mass
process is continuous.

Apply the finite-rank identity to $(P_n\psi,P_nA,P_nB)$ stopped at
$\vartheta_\ell$.  Equations
\eqref{eq:finite-rank-closure-boundary}--
\eqref{eq:finite-rank-closure-martingale}, together with It\^o's isometry,

are consistent and prove the asserted localized formula on $[0,\rho]$.  If
the terms in \eqref{eq:finite-rank-product-formula} are globally integrable
up to $\rho$, dominated convergence for the drift and BDG for the stochastic
term also permit $\ell\to\infty$ directly.

Let $w(x):=\left\langle x\right\rangle^{q_0}$ and choose the increasing stopping times
$\eta_n\uparrow\theta$ from \eqref{eq:local-q0-moment}.  Extend
$\mu$ constantly after $\theta$ and, for $r\in\mathbb N$, set
\[
 Q_t^{(r)}:=\left\langle \mu_{t\wedge\theta},w\wedge r\right\rangle,
 \qquad
 Q_t:=\sup_{r\in\mathbb N}Q_t^{(r)}
       =\left\langle \mu_{t\wedge\theta},w\right\rangle.
\]
Each $Q^{(r)}$ is continuous and adapted by narrow continuity; hence
$Q$ is progressively measurable.  Since the initial measure is fixed,
$Q_0<\infty$ is deterministic.  Put
\begin{equation*}
 c_n:=n+Q_0,\qquad
 \beta_n:=\inf\{t\in[0,\theta]:Q_t>c_n\}\wedge\theta,
 \qquad
 \alpha_n:=\eta_n\wedge\beta_n.
 \label{eq:product-localizers}
\end{equation*}
The debut theorem for progressive sets shows that $\beta_n$ and
$\alpha_n$ are stopping times.  The sequence is increasing and
$\alpha_n\uparrow\theta$: for every $t<\theta$, admissibility bounds
$\sup_{s\le t}Q_s$ after some $\eta_N$, so eventually
$t<\alpha_n$.  Moreover,
\begin{equation}
 Q_s\le c_n\qquad\text{for every }s<\alpha_n.
 \label{eq:product-moment-before-stop}
\end{equation}
The endpoint $Q_{\alpha_n}$ need not be bounded.  The mass process
$Z_t^\mu:=\left\langle \mu_{t\wedge\theta},1\right\rangle$, however, is continuous and
$Z_s^\mu\le Q_s$.  Taking a limit from the left gives
\begin{equation}
 \left\langle \mu_{\alpha_n},1\right\rangle\le c_n.
 \label{eq:product-mass-at-stop}
\end{equation}
Thus the $q_0$-moment is used only under time integrals, while
random-time boundary pairings are controlled by the mass.

Let $\rho_\varepsilon$ be a standard nonnegative spatial mollifier
and write $f^\varepsilon=f*\rho_\varepsilon$ and
$g^\varepsilon=g*\rho_\varepsilon$.  Choose
$\chi\in C_c^\infty$ equal to one on $B_1$ and zero outside $B_2$,
and put $\chi_R(x)=\chi(x/R)$.  For fixed $\varepsilon,R$, define
\[
 \psi^{\varepsilon,R}:=\chi_Rf^\varepsilon,
 \qquad
 A^{\varepsilon,R}:=-\chi_R\bigl[(L_sf_s)^\varepsilon
                         +(h_s^\top g_s)^\varepsilon\bigr],
\]
\[
 B^{\varepsilon,R}:=\chi_Rg^\varepsilon,
 \qquad h_s(x):=h(x,u_s).
\]
Choose a bounded smooth domain $K_R$ whose interior contains
$\operatorname{supp}\chi_R$.  For every fixed $s_0>d/2+2$, spatial smoothing,
\Cref{thm:main-bspde}, the
coefficient bounds, and the control-energy bound give
\begin{align*}
 &\mathbb E^\mathbb Q\sup_{s\le\theta}
       \|\psi_s^{\varepsilon,R}\|_{H_0^{s_0}(K_R)}^2
 +\mathbb E^\mathbb Q\left(\int_0^\theta
       \|A_s^{\varepsilon,R}\|_{H_0^{s_0}(K_R)}\,\mathrm d s\right)^2\\
 &\qquad
 +\mathbb E^\mathbb Q\int_0^\theta
       \|B_s^{\varepsilon,R}\|_{H_0^{s_0}(K_R)}^2\,\mathrm d s<\infty.
\end{align*}

We justify convolution of the stochastic term at the Hilbert-space level.
Without loss of generality take $0<\varepsilon\le1$, and define the
deterministic smoothing operator
\[
 T_{\varepsilon,R}v:=\chi_R(v*\rho_\varepsilon).
\]
For fixed $(\varepsilon,R,s_0)$, convolution followed by multiplication by
$\chi_R$ is a bounded linear map from $H^{-2}(B_{2R+1})$ into
$H_0^{s_0}(K_R)$; on nonnegative Sobolev orders the same statement follows
a fortiori.  In particular, the estimates above imply
\begin{align*}
 &\mathbb E^\mathbb Q\int_0^\theta
   \|T_{\varepsilon,R}g_s\|_{H^{s_0}(K_R;\mathbb R^m)}^2\,\mathrm d s<\infty,\\
 &\mathbb E^\mathbb Q\left(\int_0^\theta
   \|T_{\varepsilon,R}(L_sf_s+h_s^\top g_s)\|_{H^{s_0}(K_R)}\,\mathrm d s\right)^2<\infty.
\end{align*}
Apply $T_{\varepsilon,R}$ to the distribution-valued weak BSPDE.  A
deterministic bounded linear operator commutes with Bochner integration.  It
also commutes with the Hilbert-valued It\^o integral: this is immediate for
elementary predictable integrands, and the general case follows from It\^o's
isometry and the first estimate above.  Consequently, outside one null set
and for every $t\in[0,T]$,
\begin{equation}
 \psi_{t\wedge\theta}^{\varepsilon,R}
 =\psi_0^{\varepsilon,R}
  +\int_0^{t\wedge\theta}A_s^{\varepsilon,R}\,\mathrm d s
  +\int_0^{t\wedge\theta}(B_s^{\varepsilon,R})^\top\,\mathrm d Y_s
 \quad\text{in }H_0^{s_0}(K_R).
 \label{eq:smoothed-linked-Hilbert-identity}
\end{equation}
Equivalently, on $[0,\theta]$,
\[
 \,\mathrm d f_s^\varepsilon
 =-\bigl[(L_sf_s)^\varepsilon+(h_s^\top g_s)^\varepsilon\bigr]\,\mathrm d s
 +(g_s^\varepsilon)^\top\,\mathrm d Y_s
\]
after multiplication by $\chi_R$.  Thus
\eqref{eq:Sobolev-field-hypotheses}--
\eqref{eq:linked-Sobolev-decomp} hold, with the linked identity understood in
the precise Hilbert-valued sense of
\eqref{eq:smoothed-linked-Hilbert-identity}.

On $[0,\alpha_n]$,
\eqref{eq:product-moment-before-stop} controls every time-integrated
pairing and \eqref{eq:product-mass-at-stop} controls the boundary
pairing.  Lemma~\ref{lem:finite-rank-product} therefore applies and
yields the following identity:
\begin{eqnarray*}
 \left\langle \mu_{t\wedge\alpha_n},\chi_Rf_{t\wedge\alpha_n}^\varepsilon\right\rangle
 &={}&\left\langle \mu_0,\chi_Rf_0^\varepsilon\right\rangle
 +\int_0^{t\wedge\alpha_n}
   \left\langle \mu_s,\chi_R(g_s^\varepsilon+f_s^\varepsilon h_s)\right\rangle^\top\,\mathrm d Y_s
 \notag\\
& &+\int_0^{t\wedge\alpha_n}
   \left\langle \mu_s,C_s^{\varepsilon,R}+\chi_Rr_s^\varepsilon+\chi_Rs_s^\varepsilon\right\rangle\,\mathrm d s,
 \label{eq:eps-R-product}
\end{eqnarray*}
where
\begin{align*}
 C_s^{\varepsilon,R}&:=L_s(\chi_Rf_s^\varepsilon)-\chi_RL_sf_s^\varepsilon,\\
 r_s^\varepsilon&:=L_sf_s^\varepsilon-(L_sf_s)^\varepsilon,\\
 s_s^\varepsilon&:=h_s^\top g_s^\varepsilon-(h_s^\top g_s)^\varepsilon.
\end{align*}

\medskip
\noindent\emph{Step 1: $\varepsilon\downarrow0$ at fixed $R,n$.}

By \Cref{prop:local-time-continuity}, the path
$s\mapsto f_s$ is continuous with values in
$H^{n_*-2}(B_{2R+1})$, and $n_*-2>d/2$.  Uniform convergence of an
approximate identity on the compact range of this path, followed by
Sobolev embedding, gives
\begin{equation}
 \sup_{s\le\theta}\sup_{x\in B_{2R}}
 |f_s^\varepsilon(x)-f_s(x)|\longrightarrow0
 \quad\mathbb Q\text{-a.s.}
 \label{eq:mollifier-C0-uniform-time}
\end{equation}
The difference is bounded by $2\left\lVert \gamma\right\rVert_\infty$.  In particular,
for every fixed $t$,
\begin{equation*}
 \mathbb E^\mathbb Q\left|
 \left\langle \mu_{t\wedge\alpha_n},\chi_R(f_{t\wedge\alpha_n}^\varepsilon
                  -f_{t\wedge\alpha_n})\right\rangle
 \right|^2\longrightarrow0,
 \label{eq:mollifier-random-time-boundary}
\end{equation*}
by \eqref{eq:product-mass-at-stop}; the same holds at time zero.

For terms integrated against $\,\mathrm d s$ or $\,\mathrm d Y_s$,
\Cref{thm:main-bspde} and $n_*>d/2+2$ give
\begin{align}
 f^\varepsilon&\longrightarrow f
 &&\text{in }L^2(\mathbb Q\otimes\,\mathrm d s;C^2(B_{2R})),
 \notag\\
 g^\varepsilon&\longrightarrow g
 &&\text{in }L^2(\mathbb Q\otimes\,\mathrm d s;C^2(B_{2R};\mathbb R^m)).
 \label{eq:mollifier-local-time-integrated}
\end{align}
The coefficient
assumptions and commutator identities imply
\begin{eqnarray*}
 \sup_{x\in B_{2R}}|r_s^\varepsilon(x)|
 &\le& C_R\varepsilon(1+|u_s|)
   \sup_{x\in B_{2R+1}}(|Df_s(x)|+|D^2f_s(x)|),
 \label{eq:r-eps-bound}\\
 \sup_{x\in B_{2R}}|s_s^\varepsilon(x)|
 &\le& C\varepsilon\sup_{x\in B_{2R+1}}|g_s(x)|.
 \label{eq:s-eps-bound}
\end{eqnarray*}
For instance, the second-order commutator is the integral of
$[a^{ij}(x,u_s)-a^{ij}(x-y,u_s)]D_{ij}f_s(x-y)
\rho_\varepsilon(y)$, and on the fixed ball
$|D_xa|\le C_R(1+|u_s|)$.  By
\eqref{eq:product-moment-before-stop}, the pairings of the
right-hand sides are bounded by an integrable multiple of
$\varepsilon[(1+|u_s|)K_s^f+K_s^g]$.  Hence
\begin{equation*}
 \mathbb E^\mathbb Q\int_0^{t\wedge\alpha_n}
 \left|\left\langle \mu_s,\chi_Rr_s^\varepsilon
                         +\chi_Rs_s^\varepsilon\right\rangle\right|\,\mathrm d s
 \longrightarrow0.
 \label{eq:mollifier-drift-errors}
\end{equation*}
Expanding $C_s^{\varepsilon,R}$ as in
\eqref{eq:commutator-expanded}, using
\eqref{eq:mollifier-C0-uniform-time} for zeroth-order terms and
\eqref{eq:mollifier-local-time-integrated} for the first-order term,
gives
\begin{equation*}
 \mathbb E^\mathbb Q\int_0^{t\wedge\alpha_n}
 \left|\left\langle \mu_s,C_s^{\varepsilon,R}-C_s^R\right\rangle\right|\,\mathrm d s
 \longrightarrow0,
 \qquad
 C_s^R:=L_s(\chi_Rf_s)-\chi_RL_sf_s.
 \label{eq:mollifier-cutoff-commutator}
\end{equation*}
Here dominated convergence uses \eqref{eq:Kfg-integrability} and the
control-energy bound.  Finally,
\eqref{eq:mollifier-C0-uniform-time}--
\eqref{eq:mollifier-local-time-integrated} imply
\begin{equation*}
 \mathbb E^\mathbb Q\int_0^{t\wedge\alpha_n}
 \left|\left\langle \mu_s,\chi_R\bigl[(g_s^\varepsilon-g_s)
                   +(f_s^\varepsilon-f_s)h_s\bigr]\right\rangle\right|^2\,\mathrm d s
 \longrightarrow0.
 \label{eq:mollifier-stochastic-error}
\end{equation*}
It\^o's isometry and the preceding convergences yield
\begin{eqnarray*}
 \left\langle \mu_{t\wedge\alpha_n},\chi_Rf_{t\wedge\alpha_n}\right\rangle
 &={}&\left\langle \mu_0,\chi_Rf_0\right\rangle
 +\int_0^{t\wedge\alpha_n}
   \left\langle \mu_s,\chi_R(g_s+f_sh_s)\right\rangle^\top\,\mathrm d Y_s
 \notag\\
& &+\int_0^{t\wedge\alpha_n}\left\langle \mu_s,C_s^R\right\rangle\,\mathrm d s.
 \label{eq:R-product}
\end{eqnarray*}


\medskip
\noindent\emph{Step 2: $R\uparrow\infty$ at fixed $n$.}
Since $a$ is symmetric,
\begin{equation}
 C_s^R=f_s b(\cdot,u_s)\cdot D\chi_R
 +\frac12f_sa(\cdot,u_s):D^2\chi_R
 +(a(\cdot,u_s)D\chi_R)\cdot Df_s.
 \label{eq:commutator-expanded}
\end{equation}
It is supported on $A_R=\{R\le|x|\le2R\}$ and
\begin{equation*}
 |C_s^R(x)|\le C\mathbf 1_{A_R}(x)(1+|u_s|^2)K_s^f\left\langle x\right\rangle^{\lambda+1}.
 \label{eq:commutator-growth}
\end{equation*}
The order $\lambda+1$ comes from
$a\sim\left\langle x\right\rangle^2$, $D\chi_R\sim R^{-1}$, and
$Df\sim\left\langle x\right\rangle^\lambda$.  For $s<\alpha_n$,
\eqref{eq:product-moment-before-stop} gives
\begin{align*}
 \left|\left\langle \mu_s,C_s^R\right\rangle\right|
 &\le C(1+|u_s|^2)K_s^f
   \left\langle \mu_s,\mathbf 1_{A_R}\left\langle \cdot\right\rangle^{\lambda+1}\right\rangle\\
 &\le Cc_n(1+|u_s|^2)K_s^f.
\end{align*}
The first line tends to zero for
$\,\mathrm d\mathbb Q\otimes\,\mathrm d s$-almost every $(\omega,s)$ because
$q_0\ge\lambda+1$, and the second is integrable by
\eqref{eq:Kfg-integrability} and the control-energy bound.  Thus
\begin{equation*}
 \mathbb E^\mathbb Q\left|\int_0^{t\wedge\alpha_n}
          \left\langle \mu_s,C_s^R\right\rangle\,\mathrm d s\right|\longrightarrow0.
 \label{eq:commutator-vanish}
\end{equation*}


For the boundary pairings, no $q_0$-moment at
$\alpha_n$ is used.  For each sample point the narrowly continuous
image $\{\mu_s:0\le s\le\alpha_n\}$ is a compact, hence uniformly
tight, family of finite measures.  Together with
\eqref{eq:f-bounded}, this gives, uniformly in $t$,
\[
 \left\langle \mu_{t\wedge\alpha_n},(1-\chi_R)f_{t\wedge\alpha_n}\right\rangle\longrightarrow0.
\]
The variables are bounded by $\left\lVert \gamma\right\rVert_\infty c_n$ using
\eqref{eq:product-mass-at-stop}, so the convergence also holds in
$L^1(\mathbb Q)$; the initial pairing is analogous.
The two stochastic tails are controlled at the quadratic-variation level by
\begin{eqnarray*}
 \int_0^{\alpha_n}
 \left|\left\langle \mu_s,(1-\chi_R)g_s\right\rangle\right|^2\,\mathrm d s
 &\le& Cc_n^2 R^{2(\lambda-q_0)}
 \int_0^{\alpha_n}(K_s^g)^2\,\mathrm d s,
 \label{eq:g-tail-qv}\\
 \int_0^{\alpha_n}
 \left|\left\langle \mu_s,(1-\chi_R)f_sh_s\right\rangle\right|^2\,\mathrm d s
 &\le& C\|\gamma\|_\infty^2c_n^2
 \left(TR^{2(1-q_0)}+kR^{-2q_0}\right).
 \label{eq:fh-tail-qv}
\end{eqnarray*}
Indeed, on $|x|\ge R$,
$\left\langle x\right\rangle^\lambda\le R^{\lambda-q_0}\left\langle x\right\rangle^{q_0}$, while integration against $\mu_s$ gives
\[
 \left\langle \mu_s,\mathbf 1_{\{|x|\ge R\}}(1+|x|+|u_s|)\right\rangle
 \le C\bigl(R^{1-q_0}+|u_s|R^{-q_0}\bigr)
 \left\langle \mu_s,\left\langle \cdot\right\rangle^{q_0}\right\rangle.
\]
Here \eqref{eq:product-moment-before-stop} is used
only for $s<\alpha_n$; the value at the single endpoint is irrelevant.
Because $q_0\ge\lambda+1$ and $q_0\ge2$, both right-hand sides tend
to zero in $L^1(\mathbb Q)$.  BDG therefore passes to the limit in the
stochastic integral.  We obtain
\begin{equation}
 \left\langle \mu_{t\wedge\alpha_n},f_{t\wedge\alpha_n}\right\rangle
 =\left\langle \mu_0,f_0\right\rangle
 +\int_0^{t\wedge\alpha_n}\left\langle \mu_s,g_s+f_sh_s\right\rangle^\top\,\mathrm d Y_s.
 \label{eq:localized-final-product}
\end{equation}

\medskip
\noindent\emph{Step 3: remove the localizations and take expectations.}


Set
\[
 H_s^\mu:=\left\langle\mu_s,g_s+f_sh_s\right\rangle,
 \qquad
 \mathcal N_t^{(n)}:=\int_0^{t\wedge\alpha_n}(H_s^\mu)^\top\,\mathrm d Y_s.
\]
The stochastic integrals are consistent under stopping and hence define a
continuous local martingale $\mathcal N$ on the stochastic interval
$[0,\theta)$, localized by $(\alpha_n)$.  For each sample point and every
$t<\theta$, one has $t<\alpha_n$ eventually.  The same compact-tightness
argument used in \Cref{lem:terminal-pairing-convergence} shows that
$t\mapsto\langle\mu_t,f_t\rangle$ is continuous on $[0,\theta)$.  Taking a
countable intersection for rational $t$ and then using continuity in the
stopped identities therefore shows that, simultaneously for all $t<\theta$,
\begin{equation}
 \left\langle\mu_t,f_t\right\rangle
 =\left\langle\mu_0,f_0\right\rangle+\mathcal N_t.
 \label{eq:product-before-terminal}
\end{equation}
The pathwise argument in \Cref{lem:terminal-pairing-convergence} applies to
an arbitrary sequence $t_j\uparrow\theta$: narrow continuity of $\mu$,
local-uniform time continuity of $f$, and $|f|\le\|\gamma\|_\infty$ give
\begin{equation*}
 \left\langle\mu_t,f_t\right\rangle
 \longrightarrow \left\langle\mu_\theta,\gamma\right\rangle
 \quad\text{as }t\uparrow\theta,
 \qquad\mathbb Q\text{-a.s.}
 \label{eq:product-terminal-pathwise-limit}
\end{equation*}
Thus \eqref{eq:product-before-terminal} gives a finite pathwise limit for
$\mathcal N_t$ as $t\uparrow\theta$.  The continuous-local-martingale
convergence theorem (equivalently, the Dambis--Dubins--Schwarz
representation) then implies
\[
 \langle\mathcal N\rangle_\theta
 =\int_0^\theta|H_s^\mu|^2\,\mathrm d s<\infty
 \quad\mathbb Q\text{-a.s.}
\]
and $\mathcal N$ extends continuously to $\theta$ by
$\mathcal N_\theta:=\lim_{t\uparrow\theta}\mathcal N_t$.  Passing to the
limit in \eqref{eq:product-before-terminal} proves
\eqref{eq:measure-product} at the terminal time, and the stopped form gives

$\bar\mu=\mu^1-\mu^2$.  Fix $t\in[0,T]$ and
$\varphi\in C_c^{n_*+1}(\mathbb R^d)$, and put
$\theta=t\wedge\tau_k$.  The random variable
\begin{equation*}
 \xi:=\operatorname{sgn}\left\langle \bar\mu_\theta,\varphi\right\rangle
 \label{eq:sign-xi}
\end{equation*}
is bounded and $\mathcal F^Y_\theta$-measurable.  Hence
$\gamma(\omega,x)=\xi(\omega)\varphi(x)$ belongs to
$\mathscr T_\theta^{n_*+1}$.  Let $(f,g)$ solve the corresponding BSPDE \eqref{eq:main-bspde}.

\[
 \mathbb E^\mathbb Q\left\langle \bar\mu_\theta,\xi\varphi\right\rangle=0.
\]
By the definition of $\xi$,
\begin{equation*}
 \mathbb E^\mathbb Q\left|\left\langle \bar\mu_{t\wedge\tau_k},\varphi\right\rangle\right|=0.
 \label{eq:pairing-zero}
\end{equation*}
Thus the pairing vanishes almost surely.

$\mathcal D_0\subset C_c^{n_*+1}(\mathbb R^d)$.  Intersecting the
full-probability events obtained above for $t\in\mathbb Q\cap[0,T]$ and

 \left\langle\mu^1_{t\wedge\tau_k},\varphi\right\rangle
 =\left\langle\mu^2_{t\wedge\tau_k},\varphi\right\rangle
 \quad\text{for every }(t,\varphi)\in

\]
on one event of probability one.  The determining property and narrow
continuity of both stopped paths then imply
$\mu^1_{t\wedge\tau_k}=\mu^2_{t\wedge\tau_k}$ for every $t\in[0,T]$ on
that same event; hence the two solutions are indistinguishable on
$[0,\tau_k]$.

time.  Stopping an admissible solution at $\vartheta$ preserves the stopped
weak equation, narrow continuity, and the class-$D$ mass condition, while

      \mu^1_{t\wedge\vartheta}-\mu^2_{t\wedge\vartheta},\varphi

gives equality at every rational stopped time for the same class
$\mathcal D_0$.  Narrow continuity again upgrades this equality to
indistinguishability on $[0,\vartheta]$.

The whole unnormalization argument is carried out under the stopped reference
probability $\mathbb Q^k$.  On $[0,\tau_k]$ the process denoted by $Y$ is the
reference Brownian motion $Y^k$, and the Kushner--Stratonovich identity is the
same pathwise continuous-semimartingale identity under $\mathbb Q^k$ and under the
equivalent physical probability $\mathbb P$.  We transfer the final
indistinguishability statement back to $\mathbb P$ by equivalence.

Let $\pi^i$, $i=1,2$, be two solutions.  Put

\begin{equation*}
\begin{aligned}
 m_t^i&:=\left\langle \pi_t^i,h(\cdot,u_t)\right\rangle,\\
 R_t^i&:=\mathcal E\left(
      \int_0^{\,\cdot}(m_s^i)^\top\,\mathrm d Y_s\right)_t,
 \qquad
 \mu_t^i:=R_t^i\pi_t^i,\quad 0\le t<\tau_k.
\end{aligned}
 \label{eq:KS-unnormalization}
\end{equation*}

These processes are understood on their local
square-integrability intervals.  If
$\int_0^{\tau_k}|m_s^i|^2\,\mathrm d s<\infty$, they are extended
continuously to $\tau_k$; no value of $R_{\tau_k}^i$ is otherwise
needed.  The stochastic exponentials are strictly positive continuous
local martingales on these local domains.  Apply It\^o's
formula to $R_t^i\left\langle \pi_t^i,\varphi\right\rangle$ on any such interval.  With
$B_t^i(\varphi)=\left\langle \pi_t^i,\varphi h\right\rangle-\left\langle \pi_t^i,\varphi\right\rangle m_t^i$, the drift terms involving $m^i$ cancel:
\begin{align*}
 \,\mathrm d\left(R_t^i\left\langle \pi_t^i,\varphi\right\rangle\right)
 ={}&R_t^i\left\langle \pi_t^i,L_t\varphi\right\rangle\,\mathrm d t\\
 &+R_t^i\left(B_t^i(\varphi)
          +\left\langle \pi_t^i,\varphi\right\rangle m_t^i\right)^\top\,\mathrm d Y_t\\
 ={}&\left\langle \mu_t^i,L_t\varphi\right\rangle\,\mathrm d t
   +\left\langle \mu_t^i,\varphi h(\cdot,u_t)\right\rangle^\top\,\mathrm d Y_t.
\end{align*}
Thus $\mu^i$ solves the Zakai equation.


For $i=1,2$, choose increasing localizers
$\eta_n^i\uparrow\tau_k$ as in \eqref{eq:KS-local-m}.

Define on all of $[0,T]$
\[
 \bar A_t^i:=\int_0^{t\wedge\tau_k}|m_s^i|^2\,\mathrm d s,
\]
where the value at $\tau_k$ is the monotone limit in
$[0,+\infty]$.  Every stochastic exponential below is first defined up to
the indicated energy stop and then held constant after that stop.

For $n\ge2$, define
\begin{align}
 \kappa_n
 &:=\inf\{t\in[0,T]:\bar A_t^1+\bar A_t^2\ge n\}\wedge\tau_k,
 \notag\\
 \delta_n
 &:=\inf\left\{t\ge0:
       R_{t\wedge\kappa_n}^1\notin(n^{-1},n)
       \quad\text{or}\quad
       R_{t\wedge\kappa_n}^2\notin(n^{-1},n)\right\}\wedge\kappa_n,
 \notag\\
 \rho_n&:=\eta_n^1\wedge\eta_n^2\wedge\delta_n.
 \label{eq:KS-common-localizers}
\end{align}
If $\kappa_n=\tau_k$, the increasing energy has a finite limit no
larger than $n$, so the local exponentials extend continuously through
$\kappa_n$; if $\kappa_n<\tau_k$, their brackets up to $\kappa_n$ are
bounded by $n$.  Thus $R^i$ is defined throughout the interval used
for $\delta_n$.  The quantities in
\eqref{eq:KS-common-localizers} are genuine stopping times, the
sequence is increasing, and $\rho_n\uparrow\tau_k$.  Indeed, on every
compact subinterval of $[0,\tau_k)$ the energies are finite and each
strictly positive continuous path $R^i$ is bounded above and away
from zero.

On $[0,\rho_n]$, continuity at first exits gives
$n^{-1}\le R^i\le n$ and $\bar A^1+\bar A^2\le n$.  Moreover,
\[
 \sup_{s\le\rho_n}
 \left\langle \mu_s^i,\left\langle \cdot\right\rangle^{q_0}\right\rangle
 \le n\sup_{s\le\eta_n^i}
       \left\langle \pi_s^i,\left\langle \cdot\right\rangle^{q_0}\right\rangle<\infty.
\]
Thus $\mu^i$ satisfies every condition of
\Cref{def:admissible-zakai} on the common horizon $\rho_n$, and the
common initial law gives $\mu_0^1=\mu_0^2$.
\Cref{thm:zakai-uniqueness} yields $\mu^1=\mu^2$ there.  Equality as
finite measures gives $R^1=R^2$ directly. Strict positivity then gives
$\pi^1=\pi^2$ on $[0,\rho_n]$.  Letting $n\to\infty$ and using narrow
continuity at $\tau_k$, first on a countable convergence-determining
class and then simultaneously in time, proves indistinguishability
on $[0,\tau_k]$.

\label{thm:filter-stability}
Suppose that
\begin{align}
 X^n&\longrightarrow X
 &&\text{in $\mathbb Q$-probability in }C([0,T];\mathbb R^d),
 \label{eq:stability-X-convergence}\\
 \int_0^T|H_t^n-H_t|^2\,\mathrm d t&\longrightarrow0
 &&\text{in $\mathbb Q$-probability},
 \label{eq:stability-H-convergence}
\end{align}
and that
\begin{equation}
 \{\mathsf{Z}_T^n:n\ge1\}
 \quad\text{is uniformly integrable under }\mathbb Q.
 \label{eq:stability-Z-UI}
\end{equation}
Then $\mathsf{Z}$ is a uniformly integrable $\mathbb Q$-martingale,
\begin{equation}
 \mathsf{Z}_T^n\longrightarrow\mathsf{Z}_T
 \quad\text{in }L^1(\mathbb Q),
 \qquad
 \|\mathbb P^n-\mathbb P\|_{\mathrm{TV}}\longrightarrow0,
 \label{eq:stability-density-TV}
\end{equation}
and, for every fixed $t\in[0,T]$,
\begin{equation}
 d_{\mathrm{BL}}(\pi_t^n,\pi_t)
 \longrightarrow0
 \quad\text{in $\mathbb Q$-probability}.
 \label{eq:stability-BL-conclusion}
\end{equation}
In particular, for every $f\in C_b(\mathbb R^d)$,
\begin{equation}
 \left\langle \pi_t^n,f\right\rangle\longrightarrow\left\langle \pi_t,f\right\rangle
 \quad\text{in $\mathbb Q$-probability and in }L^p(\mathbb Q)
 \quad\text{for every }1\le p<\infty.
 \label{eq:stability-Cb-conclusion}
\end{equation}
If, in addition, the corresponding physical state processes satisfy
\begin{equation}
 \sup_{n\ge1}\mathbb E^{\mathbb P^n}
 \left[\sup_{0\le s\le T}|X_s^n|^2\right]
 +\mathbb E^{\mathbb P}
 \left[\sup_{0\le s\le T}|X_s|^2\right]
 <\infty,
 \label{eq:stability-second-moment}
\end{equation}
then, for every fixed $t\in[0,T]$,
\begin{equation}
 W_1(\pi_t^n,\pi_t)\longrightarrow0
 \quad\text{in $\mathbb Q$-probability}.
 \label{eq:stability-W1-conclusion}
\end{equation}
Consequently, convergence in probability of the filter pairings remains valid for every $f\in C(\mathbb R^d)$ satisfying

 |f(x)|\le C_f(1+|x|).
 \label{eq:stability-linear-growth-test}
\end{equation}
The probability convergences in \eqref{eq:stability-BL-conclusion} and \eqref{eq:stability-W1-conclusion} also hold under the limiting physical probability $\mathbb P$, and in the sense that the probabilities of the corresponding exceptional events under $\mathbb P^n$ tend to zero.

<!-- proof-18: body_md -->
We divide the proof into three steps.

\medskip
\noindent\emph{Step 1: convergence of the likelihoods.}
Set
\[
 A_n:=\int_0^T|H_s^n-H_s|^2\,\mathrm d s,
 \qquad
 N_t^n:=\int_0^t(H_s^n-H_s)^\top\,\mathrm d Y_s.
\]
For $\delta>0$, let
\[
 \rho_{n,\delta}
 :=\inf\left\{t\in[0,T]:
       \int_0^t|H_s^n-H_s|^2\,\mathrm d s>\delta\right\}\wedge T.
\]
On $\{A_n\le\delta\}$, $N_T^n=N_{\rho_{n,\delta}}^n$.  Hence, by It\^o's isometry and Chebyshev's inequality, for every $\varepsilon>0$,
\begin{eqnarray*}
 \mathbb Q(|N_T^n|>\varepsilon)
 &\le& \mathbb Q(A_n>\delta)
   +\mathbb Q(|N_{\rho_{n,\delta}}^n|>\varepsilon)\notag\\
 &\le& \mathbb Q(A_n>\delta)+\frac{\delta}{\varepsilon^2}.
 \label{eq:stability-stochastic-integral}
\end{eqnarray*}
First let $n\to\infty$ and then $\delta\downarrow0$.  It follows that $N_T^n\to0$ in probability.  Moreover,
\begin{eqnarray*}
 \left|\int_0^T(|H_s^n|^2-|H_s|^2)\,\mathrm d s\right|
 &\le& A_n^{1/2}
 \left(\left(\int_0^T|H_s^n|^2\,\mathrm d s\right)^{1/2}
       +\left(\int_0^T|H_s|^2\,\mathrm d s\right)^{1/2}\right)\notag\\
 &\le& A_n^{1/2}
 \left(A_n^{1/2}
       +2\left(\int_0^T|H_s|^2\,\mathrm d s\right)^{1/2}\right),
 \label{eq:stability-energy-difference}
\end{eqnarray*}
which also converges to zero in probability.  Therefore
\[
 \log\mathsf{Z}_T^n-\log\mathsf{Z}_T
 =N_T^n-\frac12\int_0^T(|H_s^n|^2-|H_s|^2)\,\mathrm d s
 \longrightarrow0
\]
in probability, and hence $\mathsf{Z}_T^n\to\mathsf{Z}_T$ in probability.  Uniform integrability and Vitali's theorem yield
\begin{equation}
 \mathbb E^\mathbb Q|\mathsf{Z}_T^n-\mathsf{Z}_T|\longrightarrow0.
 \label{eq:stability-Z-L1}
\end{equation}
Since $\mathbb E^\mathbb Q\mathsf{Z}_T^n=1$, we obtain $\mathbb E^\mathbb Q\mathsf{Z}_T=1$.  The positive local martingale $\mathsf{Z}$ is therefore a uniformly integrable martingale.  In particular,
\[
 \mathsf{Z}_t^n=\mathbb E^\mathbb Q[\mathsf{Z}_T^n\mid\mathcal F_t],
 \qquad
 \mathsf{Z}_t=\mathbb E^\mathbb Q[\mathsf{Z}_T\mid\mathcal F_t],
\]
and
\begin{equation}
 \mathbb E^\mathbb Q|\mathsf{Z}_t^n-\mathsf{Z}_t|
 \le\mathbb E^\mathbb Q|\mathsf{Z}_T^n-\mathsf{Z}_T|
 \longrightarrow0.
 \label{eq:stability-Zt-L1}
\end{equation}
The total-variation assertion in \eqref{eq:stability-density-TV} follows from
\[
 \|\mathbb P^n-\mathbb P\|_{\mathrm{TV}}
 =\frac12\mathbb E^\mathbb Q|\mathsf{Z}_T^n-\mathsf{Z}_T|.
\]

\medskip
\noindent\emph{Step 2: bounded continuous tests and weak convergence.}
For $f\in C_b(\mathbb R^d)$, put
\begin{equation*}
 \mathsf{N}_t^n(f)
 :=\mathbb E^\mathbb Q[\mathsf{Z}_t^n f(X_t^n)\mid\mathcal F^Y_t],
 \qquad
 \mathsf{D}_t^n:=\mathsf{N}_t^n(1),
 \label{eq:stability-ND-n}
\end{equation*}
and define $\mathsf{N}_t(f)$ and $\mathsf{D}_t$ analogously.  The Kallianpur--Striebel formula gives
\begin{equation}
 \left\langle \pi_t^n,f\right\rangle=\frac{\mathsf{N}_t^n(f)}{\mathsf{D}_t^n},
 \qquad
 \left\langle \pi_t,f\right\rangle=\frac{\mathsf{N}_t(f)}{\mathsf{D}_t}.
 \label{eq:stability-KS-ratio}
\end{equation}
By \eqref{eq:stability-X-convergence}, $f(X_t^n)\to f(X_t)$ in probability.  Together with \eqref{eq:stability-Zt-L1},
\begin{align*}
 &\mathbb E^\mathbb Q|\mathsf{Z}_t^n f(X_t^n)-\mathsf{Z}_t f(X_t)|\\
 &\quad\le
 \|f\|_\infty\mathbb E^\mathbb Q|\mathsf{Z}_t^n-\mathsf{Z}_t|
 +\mathbb E^\mathbb Q\big[\mathsf{Z}_t|f(X_t^n)-f(X_t)|\big]
 \longrightarrow0.
\end{align*}
The second term tends to zero because its bounded factor converges in probability under the finite measure $\mathsf{Z}_t\,\mathrm d\mathbb Q$.  Conditional expectation is an $L^1$ contraction, so
\begin{equation*}
 \mathsf{N}_t^n(f)\longrightarrow\mathsf{N}_t(f),
 \qquad
 \mathsf{D}_t^n\longrightarrow\mathsf{D}_t
 \quad\text{in }L^1(\mathbb Q).
 \label{eq:stability-ND-L1}
\end{equation*}
Since $\mathsf{Z}_t>0$ almost surely, $\mathsf{D}_t=\mathbb E^\mathbb Q[\mathsf{Z}_t\mid\mathcal F^Y_t]>0$ almost surely.  Taking ratios proves \eqref{eq:stability-Cb-conclusion} in probability.  The difference is bounded by $2\|f\|_\infty$, so convergence also holds in every finite $L^p(\mathbb Q)$.

Choose a countable convergence-determining family $\{f_j:j\ge1\}\subset C_b(\mathbb R^d)$.  From every subsequence one may extract a further subsequence along which $\left\langle \pi_t^n,f_j\right\rangle\to\left\langle \pi_t,f_j\right\rangle$ a.s. for every $j$.  On the resulting common full-probability event, the deterministic measures $\pi_t^n$ converge weakly to $\pi_t$, and hence their bounded--Lipschitz distance tends to zero.  The subsequence criterion for convergence in probability proves \eqref{eq:stability-BL-conclusion}.

\medskip
\noindent\emph{Step 3: first Wasserstein convergence.}
Assume \eqref{eq:stability-second-moment}.  For $R>0$, set
\[
 \psi_R(x):=|x|\mathbf 1_{\{|x|>R\}}.
\]

Although \eqref{eq:stability-KS-ratio} was stated for bounded continuous
tests, its underlying Bayes identity holds for every bounded Borel test.
Apply that identity first to $\psi_{R,M}:=\psi_R\wedge M$ and then use
conditional monotone convergence as $M\uparrow\infty$.  The physical
second-moment bound makes the limiting numerator integrable, since
\[
 \mathbb E^\mathbb Q[\mathsf Z_t^n\psi_R(X_t^n)]
 =\mathbb E^{\mathbb P^n}[\psi_R(X_t^n)]
 \le R^{-1}\mathbb E^{\mathbb P^n}|X_t^n|^2<\infty,
\]
uniformly in $n$, with the analogous estimate under $\mathbb P$.

For $a,\varepsilon>0$, Markov's inequality and the martingale property of $\mathsf{Z}^n$ give
\begin{eqnarray*}
 \mathbb Q\left(\left\langle \pi_t^n,\psi_R\right\rangle>\varepsilon\right)
 &\le& \mathbb Q(\mathsf{D}_t^n<a)
 +\mathbb Q\left(
   \mathbb E^\mathbb Q[\mathsf{Z}_t^n\psi_R(X_t^n)\mid\mathcal F^Y_t]
   >a\varepsilon\right)\notag\\
 &\le& \mathbb Q(\mathsf{D}_t^n<a)
 +\frac{1}{a\varepsilon}
   \mathbb E^\mathbb Q[\mathsf{Z}_T^n\psi_R(X_t^n)]\notag\\
 &\le& \mathbb Q(\mathsf{D}_t^n<a)
 +\frac{C}{a\varepsilon R},
 \label{eq:stability-filter-tail}
\end{eqnarray*}
where $C$ is independent of $n$, because $\psi_R(x)\le |x|^2/R$.  Since $\mathsf{D}_t^n\to\mathsf{D}_t>0$ in probability,
\begin{equation*}
 \lim_{a\downarrow0}\limsup_{n\to\infty}
 \mathbb Q(\mathsf{D}_t^n<a)=0.
 \label{eq:stability-denominator-away-zero}
\end{equation*}
Consequently,
\begin{equation}
 \lim_{R\to\infty}\limsup_{n\to\infty}
 \mathbb Q\left(\left\langle \pi_t^n,\psi_R\right\rangle>\varepsilon\right)=0.
 \label{eq:stability-uniform-tail}
\end{equation}
The same argument gives $\left\langle \pi_t,\psi_R\right\rangle\to0$ in probability as $R\to\infty$.

Let $T_R:\mathbb R^d\to\overline B_R$ be the radial projection,
\[
 T_R(x)=x\quad\text{if }|x|\le R,
 \qquad
 T_R(x)=R\frac{x}{|x|}\quad\text{if }|x|>R.
\]
Then
\begin{align}
 W_1(\pi_t^n,\pi_t)
 \le{}&W_1((T_R)_\#\pi_t^n,(T_R)_\#\pi_t)\notag\\
 &+\int|x-T_R(x)|\,\pi_t^n(\,\mathrm d x)
 +\int|x-T_R(x)|\,\pi_t(\,\mathrm d x).
 \label{eq:stability-W1-truncation}
\end{align}
For fixed $R$, \eqref{eq:stability-BL-conclusion} implies weak convergence in probability of the two pushforward measures.  Since they are supported on the compact ball $\overline B_R$, weak convergence there is equivalent to convergence in $W_1$.  Hence the first term in \eqref{eq:stability-W1-truncation} tends to zero in probability for fixed $R$.  The remaining terms are bounded by the corresponding $\psi_R$-moments, which vanish in probability by \eqref{eq:stability-uniform-tail} and its limiting analogue.  First choosing $R$ large and then $n$ large proves \eqref{eq:stability-W1-conclusion}.

For completeness, from every subsequence of the random measures one may extract an almost surely $W_1$-convergent subsubsequence.  The deterministic characterization of $W_1$ convergence then gives convergence against every continuous function satisfying \eqref{eq:stability-linear-growth-test}; the subsequence criterion returns convergence in probability for the original sequence.  Since $\mathbb P$ is equivalent to $\mathbb Q$, convergence in $\mathbb Q$-probability implies convergence in $\mathbb P$-probability.  If $A_n$ denotes either exceptional event, then
\[
 \mathbb P^n(A_n)
 \le \mathbb P(A_n)+\|\mathbb P^n-\mathbb P\|_{\mathrm{TV}}\longrightarrow0,
\]
which gives the final assertion.

<!-- remark-1: body_md -->
\label{rem:stability-inputs}
The intrinsic assumption is \eqref{eq:stability-H-convergence}.  Under the hypotheses used elsewhere in this paper, $h$ is only Borel measurable in its input variable, so $\mathbb E\int_0^T|u_n(t)-u(t)|\,\mathrm d t\to0$ does not by itself imply \eqref{eq:stability-H-convergence}.  A simple sufficient condition is
\[
 |h(x,v)-h(y,w)|\le C(|x-y|+|v-w|)
\]
and
\[
 \mathbb E^\mathbb Q\sup_{t\le T}|X_t^n-X_t|^2
 +\mathbb E^\mathbb Q\int_0^T|u_n(t)-u(t)|^2\,\mathrm d t\longrightarrow0.
\]
If the controls are uniformly bounded, then their $L^1$ convergence in expectation implies the required $L^2$ convergence.  At a common control-energy stopping level, the entropy estimate of \Cref{prop:filtering-likelihood} also verifies \eqref{eq:stability-Z-UI}, provided that its entropy bound is uniform in $n$.

<!-- definition-4: body_md -->
\label{def:bspde-solution}
Let $\theta\le\tau_k$ be an $\mathbb F^Y$-stopping time and
$\gamma\in\mathscr G_\theta^{n_*+1}$.  A pair $(f,g)$ is a bounded weighted Sobolev solution of \eqref{eq:main-bspde}
if, for some $\ell>0$,
\[f\text{ is }\mathbb F^Y\text{-adapted},\quad g\text{ is }\mathbb F^Y\text{-predictable},\quad
   f\in L^\infty(\Omega\times[0,\theta]\times\mathbb R^d),
 \label{eq:solution-meas}\]
 \begin{equation}\mathbb E^\mathbb Q\operatorname*{ess\,sup}_{s\le\theta}\left\lVert f_s\right\rVert_{H_\ell^{n_*}}^2
   +\mathbb E^\mathbb Q\int_0^\theta\left\lVert g_s\right\rVert_{H_\ell^{n_*}(\mathbb R^d;\mathbb R^m)}^2\,\mathrm d s
   <\infty,
 \label{eq:solution-norm}
\end{equation}
and, for every $\varphi\in C_c^\infty(\mathbb R^d)$ and $t\in[0,T]$,
\begin{align}
 \left\langle f_{t\wedge\theta},\varphi\right\rangle
 ={}&\left\langle \gamma,\varphi\right\rangle
 +\int_{t\wedge\theta}^{\theta}\left\langle f_s,L_s^*\varphi\right\rangle\,\mathrm d s
 \notag\\
 &+\int_{t\wedge\theta}^{\theta}
      \left\langle h(\cdot,u_s)^\top g_s,\varphi\right\rangle\,\mathrm d s
 -\int_{t\wedge\theta}^{\theta}\left\langle g_s,\varphi\right\rangle^\top\,\mathrm d Y_s.
 \label{eq:weak-bspde}
\end{align}
Here $L_s^*$ is the distributional adjoint of $L_s$.

<!-- theorem-8: body_md -->
\label{thm:main-bspde}
Suppose \Cref{ass:bspde-regularity} hold.  Fix $k\in\mathbb N$, an $\mathbb F^Y$-stopping time $\theta\le\tau_k$, and
$\gamma\in\mathscr G_\theta^{n_*+1}$.  Set
\begin{equation}
 \ell_0:=\frac d2+2.
 \label{eq:ell-zero}
\end{equation}

Then \eqref{eq:main-bspde} admits a unique bounded weighted Sobolev
solution $(f,g)$ with $\ell=\ell_0$ in \eqref{eq:solution-norm}.  The
component $f$ is $\mathbb F^Y$-adapted/predictable and has a single jointly
measurable, locally space--time continuous representative for which,
on one event of full $\mathbb Q$-probability,
\begin{equation}
 |f_s(x)|\le\left\lVert \gamma\right\rVert_{L^\infty(\Omega\times\mathbb R^d)},
 \qquad 0\le s\le\theta,\quad x\in\mathbb R^d.
 \label{eq:f-bounded}
\end{equation}

For every
\begin{equation*}
 \lambda>d+2
 \label{eq:lambda-threshold}
\end{equation*}
and every multi-index $\alpha$ with $|\alpha|\le2$,
\begin{align}
 \mathbb E^\mathbb Q\Bigg[
 &\operatorname*{ess\,sup}_{0\le s\le\theta}\sup_{x\in\mathbb R^d}
     \frac{|D^\alpha f_s(x)|^2}{\left\langle x\right\rangle^{2\lambda}}
 +\int_0^\theta\sup_{x\in\mathbb R^d}
     \frac{|D^\alpha g_s(x)|^2}{\left\langle x\right\rangle^{2\lambda}}\,\mathrm d s
 \Bigg]<\infty.
 \label{eq:weighted-sup-2}
\end{align}
In addition, $f$ admits a version that is continuous in time with values in $C(B_R)$ for every $R<\infty$.  Consequently, in \eqref{eq:weighted-sup-2} with $\alpha=0$ the essential time supremum may be replaced by the ordinary supremum.  The numerical exponents $\ell_0$ and the threshold $d+2$ are independent of $k$; only the constants in the estimates depend on the stopping level.

<!-- proposition-4: body_md -->
\label{prop:local-time-continuity}
Every weighted Sobolev solution has a single jointly measurable,
adapted modification, fixed simultaneously on all integer balls,
such that, for every $R<\infty$,
\begin{equation}
 f\in C\bigl([0,\theta];H^{n_*-2}(B_R)\bigr)
 \hookrightarrow C\bigl([0,\theta];C(\overline B_R)\bigr),
 \label{eq:local-time-continuity}
\end{equation}
where the process is extended constantly after $\theta$ when
convenient.  This modification remains jointly measurable and
$\mathbb G^k$-adapted.  For the constructed solution, fix this
representative once and for all and use it henceforth.
