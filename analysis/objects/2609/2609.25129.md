\label{monNIPequiv}  The following are equivalent for a complete theory $T$.
\begin{enumerate}
\item  $T$ is monadically NIP;
\item  $T$ is dp-minimal and has endless indiscernible triviality (EIT);
\item  $T$ has the f.s.\ dichotomy.
\end{enumerate}

\label{start}   ($T$ NIP)   Suppose $M\preceq{\mathfrak  C}$ is small.  An $L({\mathfrak  C})$ formula  $\varphi(\overline{x},\bar{d})$ does not fork over $M$ if and only if there is a global type
$p(\overline{x})\in S({\mathfrak  C})$ that is $M$-invariant and $\varphi(\overline{x},\bar{d})\in p$.

$p(\overline{x})$ is $A$-invariant.

\label{dptrichotomy}   [$T$ dp-minimal]  Let $p(x)\in S_1({\mathfrak  C})$ be a global, $M$-invariant 1-type.  
Then one of the three possibilities holds.
\begin{itemize}
\item   $p$ is finitely satisfied in $M$ but not $M$-definable;
\item  $p$ is $M$-definable but not finitely satisfied in $M$; or
\item  $p$ is generically stable over $M$ (i.e., both finitely satisfied in $M$ and $M$-definable).
\end{itemize}

<!-- proposition-1: body_md -->
\label{EITprop}  Suppose $M\preceq{\mathfrak  C}\preceq{\mathfrak  C}^+$ with $M$ small and ${\mathfrak  C}$ a monster model.
Suppose $p(\overline{x}),q(\bar{y})\in S({\mathfrak  C})$ are both global, $M$-invariant types.  Then for every $\bar{a},\bar{b}$ from ${\mathfrak  C}^+$, if ${\rm tp}(\bar{a}/{\mathfrak  C})=p$ and ${\rm tp}(\bar{b}/C)=q$,
then ${\rm tp}(\bar{a}\bar{b}/{\mathfrak  C})$ is $M$-invariant as well.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Checking that a global type is $M$-invariant is finitary.   Thus, choose any $\bar{c},\bar{c}'$ from ${\mathfrak  C}$ such that ${\rm tp}(\bar{c}/M)={\rm tp}(\bar{c}'/M)$.  It suffices to show
that ${\rm tp}(\bar{c}/M\bar{a}\bar{b})={\rm tp}(\bar{c}'/M\bar{a}\bar{b})$.  Trivially, ${\rm tp}(\bar{c}/M)$ is finitely satisfied in $M$, so let  $r(\overline{z})\in S({\mathfrak  C})$ be a global type extending ${\rm tp}(\bar{c}/M)$ that
is finitely satisfied in $M$.  Choose any $\bar{d}$ from ${\mathfrak  C}$ realizing the restriction $r|M\bar{c}\bar{c}'$.  
We will show that ${\rm tp}(\bar{c}/M\bar{a}\bar{b})={\rm tp}(\bar{d}/M\bar{a}\bar{b})={\rm tp}(\bar{c}'/M\bar{a}\bar{b})$, which, as noted above, suffices.
The argument for each of these equalities is symmetric, so we concentrate on the first one.   
Note that the two-element sequence $\bar{c},\bar{d}$ is the beginning of a Morley sequence in $r(\overline{z})$ over $M$.   Thus, 
by compactness and homogeneity, the Morley pair $(\bar{c},\bar{d})$ can be included in  an $M$-indiscernible sequence
$(\bar{e}_i:i\in{\mathbb Z})$ from ${\mathfrak  C}$, indexed by ${\mathbb Z}$.

\medskip
\noindent{\bf Claim.}   $(\bar{e}_i:i\in{\mathbb Z})$ is indiscernible over both $M\bar{a}$ and $M\bar{b}$.

\begin{proof}  The arguments are symmetric, so we concentrate on the first.   Choose any $i_1<\dots<i_n$ and $j_1<\dots<j_n$ from ${\mathbb Z}$.
As $(\bar{e}_i:i\in{\mathbb Z})$ is $M$-indiscernible, 
$${\rm tp}(\bar{e}_{i_1},\dots,\bar{e}_{i_n}/M)={\rm tp}(\bar{e}_{j_1},\dots,\bar{e}_{j_n}/M)$$
Thus, since ${\rm tp}(\bar{a}/{\mathfrak  C})$ is $M$-invariant, we conclude 
$${\rm tp}(\bar{e}_{i_1},\dots,\bar{e}_{i_n}/M\bar{a})={\rm tp}(\bar{e}_{j_1},\dots,\bar{e}_{j_n}/M\bar{a})$$
completing the proof of the Claim.
\qed
\end{proof}

Thus, by endless indiscernible triviality, we have $(\bar{e}_i:i\in{\mathbb Z})$ indiscernible over $M\bar{a}\bar{b}$.  As $\bar{c},\bar{d}$ are contained in the sequence, it follows
that  ${\rm tp}(\bar{c}/M\bar{a}\bar{b})={\rm tp}(\bar{d}/M\bar{a}\bar{b})$, as required.

The verification of the second equality is symmetric, by choosing a separate $M$-indiscernible sequence $(\bar{e}_i':i\in{\mathbb Z})$ containing both $\bar{c}'$ and $\bar{d}$
and arguing as above.
\qed

\label{iterateMinv}  [$T$  EIT]  Suppose $M\preceq{\mathfrak  C}$ is small.
For any $n$, a global type $p(x_1,\dots,x_n)\in S_n({\mathfrak  C})$ is $M$-invariant if and only if $p\mathord\restriction_{x_i}(x_i)\in S_1({\mathfrak  C})$ is $M$-invariant for each $i$.

<!-- fact-5: body_md -->
\label{charMdef}  [$T$ any complete theory]   Suppose $M\preceq{\mathfrak  C}$.   The following are equivalent for a global type $p(\overline{x})\in S_{\overline{x}}({\mathfrak  C})$.
\begin{enumerate}
\item $p$ is $M$-definable;
\item  $p$ is the unique heir of $p\mathord\restriction_{M}$;
\item  $p$ is $M$-invariant and  $p(\overline{x})\otimes q(\bar{y})=q(\bar{y})\otimes p(\overline{x})$ for all global types $q(\bar{y})\in S_{\bar{y}}({\mathfrak  C})$ that are finitely satisfied in $M$.
\end{enumerate}

$(3)\Rightarrow (1)$, which appeals to  $(1)\Leftrightarrow(2)$ is proved in Lemma~2.3 of \cite{dpInv}.  
\qed

realizing $q(\overline{x})$ and $\bar{b}$ realizing $r(\bar{y})$.

\label{partition}

Note that if $i\in D$, then since $p_i:={\rm tp}(a_i/MC)$ does not fork over $M$, there is a global 1-type $P_i(x_i)\in S_1({\mathfrak  C})$  extending $p_i$ that is $M$-invariant.  By Fact~\ref{dptrichotomy} $P_i$ is either $M$-definable or else finitely satisfied in $M$.  However, if $P_i$ were finitely satisfied in $M$, then $p_i$ would be as well, contradicting $i\not\in F^*$.  Thus, $P_i$ and hence
$p_i$ is $M$-definable.  Thus, by Lemma~\ref{overMC}, ${\rm tp}(\bar{d}/MC)$ is $M$-definable.  

Next, we know that each singleton $f_i\in\overline{f}$ has ${\rm tp}(f_i/MC)$ finitely satisfied in $M$, so by Fact~\ref{3.3}(1), ${\rm tp}(\overline{f}/MC)$ is finitely satisfied in $M$ as well.
To finish, we must prove that ${\rm tp}(\overline{f}/MC\bar{d})$ is finitely satisfied in $M$, which we prove by induction on $\lg(\bar{d})$ (so if $D=\emptyset$ there is nothing to prove).
The argument is akin to the proof of the Claim in Proposition~\ref{orthprop}(2).
 Write $\bar{d}=(d_i:i<k)$ and we argue by induction on $i\le k$ that ${\rm tp}(\overline{f}/MC\bar{d}_{<i})$ is finitely satisfied in $M$.
To see this, assume $i<k$ and ${\rm tp}(\overline{f}/MC\bar{d}_{<i})$ is finitely satisfied in $M$.  We apply the f.s.\ dichotomy to this and the element $d_i$.  The point is that
if ${\rm tp}(\overline{f} d_i/MC\bar{d}_{<i})$ were finitely satisfied in $M$, then ${\rm tp}(d_i/MC)$ would be finitely satisfied in $M$, contradicting our choice of $d_i$.
Thus, we must have ${\rm tp}(\overline{f}/MC\bar{d}_{\le i})$ finitely satisfied in $M$, completing our inductive step.
\qed

\noindent{\bf Case 1.}  $p$ is finitely satisfied in $M$.  
\medskip

In this case, we apply the f.s.\ dichotomy directly.  We know that ${\rm tp}(a/{\mathfrak  C})$ is finitely satisfied in $M$.  But, ${\rm tp}(ac/{\mathfrak  C})$ cannot be finitely satisfied in $M$, since this would imply ${\rm tp}(c/{\mathfrak  C})$ finitely satisfied in $M$, hence $M$-invariant.
Thus, by the f.s.\ dichotomy, ${\rm tp}(a/{\mathfrak  C} c)$ is finitely satisfied in $M$.   As ${\mathfrak  C}$ is full for $M$,  Fact~\ref{fullstationary} gives
$a$ realizes $p^*|{\mathfrak  C} c$, as required.

\medskip
\noindent{\bf Case 2.}  $p$ is  $M$-definable.
\medskip

First, since $q(y)$ is not $M$-invariant, there is a sequence $I=(\bar{e}_i:i\in \omega)$ that is $M$-indiscernible, but not $Mc$-indiscernible.  The argument is similar to part of the proof of Proposition~\ref{EITprop}.  As $q(y)$ is not $M$-invariant, choose tuples $\bar{b},\bar{b}'$ from ${\mathfrak  C}$ with ${\rm tp}(\bar{b}/M)={\rm tp}(\bar{b}'/M)$, but ${\rm tp}(\bar{b}/Mc)\neq{\rm tp}(\bar{b}'/Mc)$.    Let $r(\overline{z})={\rm tp}(\bar{b}/M)$.  As $r$ is (trivially)
finitely satisfied in $M$, choose a global $r^*(\overline{z})\in S({\mathfrak  C})$ extending $r$ and finitely satisfied in $M$.  In ${\mathfrak  C}$, choose
a realization $\bar{d}$ of $r^*|M\bar{b}\bar{b}'$.  From the above inequality, we must have either ${\rm tp}(\bar{b}/Mc)\neq {\rm tp}(\bar{d}/Mc)$ or
${\rm tp}(\bar{b}'/Mc)\neq{\rm tp}(\bar{d}/Mc)$.   As the arguments are symmetric, assume the former.  Then $(\bar{b},\bar{d})$ starts a Morley sequence in $r^*$ over $M$.   By compactness, there is an infinite, $M$-indiscernible sequence $I=(\bar{e}_i:i\in \omega)\subseteq{\mathfrak  C}$ with $\bar{b},\bar{d}$ occurring in $I$.  The presence of $\bar{b},\bar{d}\in I$ implies that $I$ is not $Mc$-indiscernible.  

Next, as $p$ is $M$-definable, so is the global extension $p^*(x)\in S({\mathfrak  C}^*)$.  By way of contradiction, assume $a$ does not realize $p^*|{\mathfrak  C} c$.   
Then there is some $L$-formula $\varphi(x,y,\overline{z})$, whose definition is an $L(M)$-formula $\theta(y,\overline{z})$, and $\overline{h}$ from ${\mathfrak  C}$  such that
$${\mathfrak  C}^*\models \neg(\varphi(a,c,\overline{h})\leftrightarrow \theta(c,\overline{h}))$$
Choose a sequence $J=(a_j:j\in\omega)\subseteq{\mathfrak  C}^*$ where $a_0=a$ and $a_j$ realizes $p^*|{\mathfrak  C} c \bar{a}_{<j}$ for each $j>0$.  Note that by forgetting $c$, $J$ is a Morley sequence in $p^*$ over ${\mathfrak  C}$, hence $J$ is indiscernible over ${\mathfrak  C}$.   Also, by
the properties of being a $p^*$-definition, $${\mathfrak  C}^*\models \varphi(a_j,c,\overline{k})\leftrightarrow \theta(c,\overline{k})\quad \hbox{for all $j>0, \overline{k}\in{\mathfrak  C}^{\lg(\overline{z})}$}$$
As a technique for obviating the role of the parameter $\overline{h}$, we pass to an expansion ${\mathfrak  C}^*_U=({\mathfrak  C}^*,U)$ of ${\mathfrak  C}^*$ by a unary predicate interpreted as $U^{{\mathfrak  C}^*_U}={\mathfrak  C}$.  Let $L(U)=L\cup\{U\}$.   As $Th({\mathfrak  C}^*)$ is monadic NIP, so is $Th({\mathfrak  C}^*_U)$.  We will get a contradiction by showing that the two sequences $I$ and $J$ defined above are mutually indiscernible over $M$ in the language $L(U)$, but neither $I$ nor $J$ is indiscernible over $Mc$.  This configuration directly contradicts that $Th({\mathfrak  C}^*_U)$ is dp-minimal.  

To see the  point of expanding the language, look at the $L(U)$-formula (with parameters from $M$)
$$\psi(x,y):=\exists \overline{z}\left(\bigwedge_{z_i\in\overline{z}} U(z_i)\wedge \neg[\varphi(x,y,\overline{z})\leftrightarrow \theta(y,\overline{z})]\right)$$
Then ${\mathfrak  C}^*_U\models\psi(a_0,c)\wedge\neg \psi(a_j,c)$ for any $j>0$, so $J$ is not $Mc$-indiscernible in $L(U)$.  
 
It remains to show that $J$ is $MI$-indiscernible in ${\mathfrak  C}^*_U$ and $I$ is $MJ$-indiscernible in ${\mathfrak  C}^*_U$.  The former is easy.  From above, $J$ is ${\mathfrak  C}$-indiscernible in ${\mathfrak  C}^*$.  However, any $\sigma\in  Aut({\mathfrak  C}^*)$ fixing ${\mathfrak  C}$ pointwise is an automorphism of the expansion ${\mathfrak  C}^*_U$.  Thus, $J$ is ${\mathfrak  C}$-indiscernible (and hence $MI$-indiscernible) in ${\mathfrak  C}^*_U$.
Thus, we are done once we prove the following Claim.

\medskip
\noindent{\bf Claim.}  $I$ is $MJ$-indiscernible in the expansion ${\mathfrak  C}^*_U$.  

\begin{proof}    Choose increasing sequences $i_1<\dots <j_n<\omega$ and $k_1<\dots<k_n<\omega$.  We begin by working in the original language $L$.  From above we know there is an automorphism $\sigma_0\in\operatorname{Aut}({\mathfrak  C})$ fixing $M$ pointwise with
$\sigma_0(\bar{e}_{i_j})=\bar{e}_{k_j}$ for each $1\le j\le n$.  Since ${\rm tp}(a_j/{\mathfrak  C})=p$ is $M$-invariant for each $j\in\omega$,
${\rm tp}(J/{\mathfrak  C})$ is $M$-invariant by Corollary~\ref{iterateMinv}. 
Because   $\sigma_0$ permutes ${\mathfrak  C}$ while fixing $M$ pointwise, $\sigma_0\cup id_{J}$ is a 
partial $L$-elementary map in ${\mathfrak  C}^*$.
Since ${\mathfrak  C}^*$ is a monster model for ${\mathfrak  C}$ (in the original language)
there is an $L$-automorphism $\sigma^*\in\operatorname{Aut}({\mathfrak  C}^*)$ extending $\sigma_0$ fixing $J$ pointwise.   
 In particular,  $\sigma^*$ permutes the set ${\mathfrak  C}$ while fixing $MJ$ pointwise.
Hence,  $\sigma^*$ is an $L(U)$-automorphism of ${\mathfrak  C}^*_U$ fixing $MJ$ pointwise and
$\sigma^*(\bar{e}_{i_j})=\bar{e}_{k_j}$ for each $j$.   
Thus,   $I$ is $MJ$-indiscernible in the expanded language.
\qed
\end{proof}

Putting all of this together, in the expanded ${\mathfrak  C}^*_U$, $I$ and $J$ are mutually indiscernible over $M$, with neither $I$ nor $J$
$L(U)$-indiscernible over $Mc$.  Thus, $\hbox{dp-rk}_{L(U)}{\rm tp}_{L(U)}(c/M)\ge 2$, contradicting dp-minimality of $Th({\mathfrak  C}^*_U)$.
\qed

{\em  A theory $T$ has {\em density of definable types among non-forking types} if, for every small model $M\preceq{\mathfrak  C}$ and every formula
$\varphi(\overline{x},\bar{c})$, $\varphi(\overline{x},\bar{c})$ does not fork over $M$ if and only if there is an $M$-definable global type $p(\overline{x})$ with $\varphi(\overline{x},\bar{c})\in p$.\\

}
