<!-- question-1: body_md -->
\label{Rump-qn} 
	Does every finite quasi-linear cycle set of size $>1$ have non-trivial socle?
<!-- end question-1 -->
<!-- conjecture-1: body_md -->
\label{main-conj} 
Let $A$ be a finite abelian group, and let $\tau$, $\sigma \in \mathrm{Sym}_0(A)$. If $\sigma$ is a magma automorphism of $(A,\tau)$ then either $\mathrm{Soc}(A,\tau)\neq \{0\}$ or $\mathrm{Soc}(A,\sigma)\neq \{0\}$.
<!-- end conjecture-1 -->
<!-- definition-1: body_md -->
\label{def-c-ex} 
Let $A$ be an abelian group and let $\tau \in \mathrm{Sym}_0(A)$. Then $\tau$ is a {\em counterexample for} $A$ if $\mathrm{Soc}(A,\tau)=\{0\}$ and there exists some $\sigma \in \mathrm{Sym}_0(A)$ such that  $\mathrm{Soc}(A,\sigma)=\{0\}$ and $\sigma$ is a magma automorphism of $(A,\tau)$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{mag-cat} 
\begin{itemize} 
\item[(i)]  
A {\em pointed magma} $(X, \cdot,0)$ consists of a set $X$, a binary operation 
$\cdot$ on $X$, and a distinguished element $0 \in X$. 
\item[(ii)] Let $(X, \cdot_X,0_X)$ and $(Y, \cdot_, 0_Y)$ be pointed magmas. A {\em homomorphism of pointed magmas} from $X$ to $Y$ is a function $f:X \to Y$ such that $f(u \cdot_X v) = f(u) \cdot_Y f(v)$ for all $u$, $y \in X$, and $f(0_X)=0_Y$. 
\item[(iii)] A {\em pointed submagma} of a pointed magma $(X,\cdot,0)$ is a subset $Y \subseteq X$ such that $0 \in Y$ and $u \cdot v \in Y$ for all $u$, $v \in Y$.  
\end{itemize}

<!-- proposition-1: body_md -->
\label{tilde-comm} 
If $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$ then $\sigma$ commutes with $\tau$ and $\tilde{\tau}$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For all $a$, $b \in A$, we have  
$\sigma(a \cdot_\tau b) = \sigma(a) \cdot_\tau \sigma(b)$, so that  
\begin{equation} \label{mag-aut-eqn} 
	 \sigma( \tau(b-a) - \tau(-a)) = \tau(\sigma(b)-\sigma(a)) - \tau( - \sigma(a)).  
\end{equation}  
Setting $a=0$ in (\ref{mag-aut-eqn}), we obtain $\sigma(\tau(b))=\tau(\sigma(b))$ 
for all $b$. Setting $b=a$  in (\ref{mag-aut-eqn}), we obtain $\sigma(0-\tau(-a))=0-\tau(-\sigma(a))$. Thus $\sigma(\tilde{\tau}(a))=\tilde{\tau}(\sigma(a))$ for all $a$.
<!-- end proof-1 -->
<!-- proposition-2: body_md -->
\label{autom} 
If $\tau \in \mathrm{Aut}_{\mathrm{gp}}(A)$ then  
$$ \mathrm{Aut}_\mathrm{mag}(A,\tau) = \{ \sigma \in \mathrm{Sym}_0(A) : \sigma \tau = \tau \sigma \}. $$
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
By Proposition \ref{tilde-comm} it suffices to prove that if  $\tau \in \mathrm{Aut}_{\mathrm{gp}}(A)$ and $\sigma \tau = \tau \sigma$ then $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$. Now  
\begin{eqnarray*} 
 \sigma(a \cdot_\tau b) & = & \sigma(\tau(b-a)-\tau(-a)) \\ 
                    &  = & \sigma(\tau(b)-\tau(a)+\tau(a)) \\ 
                    & = &  \sigma(\tau(b)) \\ 
                    & = &	\tau(\sigma(b)),  
\end{eqnarray*}   
while 
$$ \sigma(a) \cdot_\tau \sigma(b) = \tau(\sigma(b)-\sigma(a)) - 
\tau(-\sigma(a)) = \tau(\sigma(b)). $$ 
Hence $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$.
<!-- end proof-2 -->
<!-- proposition-3: body_md -->
\label{recip} 
For arbitrary $\tau$, $\sigma \in \mathrm{Sym}_0(A)$,  
$$ \sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau) \Leftrightarrow \tau \in \mathrm{Aut}_\mathrm{mag}(A,\sigma). $$
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
By symmetry, it suffices to prove one implication. So suppose that $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$. Again, (\ref{mag-aut-eqn}) holds for all $a$, $b \in A$. Writing $c=-a$ and $d=b-a$, we therefore have 
\begin{equation} \label{mag-autom1} 
	\sigma( \tau(d) - \tau(c)) = \tau(\sigma(d-c)-\sigma(-c)) - \tau( - 
	\sigma(-c))	\mbox{ for all } c, d \in A.  
\end{equation} 
But by Proposition \ref{tilde-comm},  
$$ \tau(-\sigma(-c)) = -\tilde{\tau}(\sigma(-c)) = \sigma(\tilde{\tau}(-c)) = \sigma(-\tau(c)).$$ 
Thus  
$$ \sigma( \tau(d) - \tau(c)) - \sigma(-\tau(c)) = \tau(\sigma(d-c)-\sigma(-c))  
\mbox{ for all } c, d \in A, $$ 
so that $\tau(c) \cdot_\sigma \tau(d) = \tau(c 
\cdot_\sigma d)$. Hence $\tau \in \mathrm{Aut}_\mathrm{mag}(A,\sigma)$.
<!-- end proof-3 -->
<!-- proposition-4: body_md -->
\label{conjug} 
Let $\tau \in \mathrm{Sym}_0(A)$, let $\alpha=\mathrm{Aut}_{\mathrm{gp}}(A)$, and let $\omega=\alpha \tau \alpha^{-1}$. Then $\alpha: (A,\tau) \to (A,\omega)$ is an isomorphism of pointed magmas.
<!-- end proposition-4 -->
<!-- proof-4: body_md -->
Let $a$, $b \in A$, and set $u=\alpha(a)$, $v=\alpha(b)$. Then  
\begin{eqnarray*}  
 \alpha(a \cdot_\tau b) & = & \alpha(\tau(b-a))-\alpha(\tau(-a)) \\ 
					  & = & \alpha \tau (\alpha^{-1}(v) -\alpha^{-1}(u) ) - \alpha \tau(-\alpha^{-1}(u)) \\ 
                      & = & \omega(v-u) - \omega(-u) \\ 
                      & = & u \cdot_\omega v. 
\end{eqnarray*}

<!-- proposition-5: body_md -->
\label{inv-tau}

<!-- proposition-6: body_md -->
\label{Soc-k-prop} 
	Let $A$ be a finite cyclic group, and let $\tau \in \mathrm{Sym}_0(A)$. For all $k \geq 1$, we have 
	\begin{equation} \label{Soc-k}  
		\mathrm{Soc}(A,\tau) \subseteq \mathrm{Soc}(A,\tau^k)  
	\end{equation} 
	with equality if $k$ is coprime to the order of $\tau$.
<!-- end proposition-6 -->
<!-- proof-7: body_md -->
It follows from (\ref{def-soc}) that restricting $\tau$ to $\mathrm{Soc}(A,\tau)$ gives an injective group homomorphism $\mathrm{Soc}(A,\tau) \to A$. Its image is therefore a subgroup of $A$ isomorphic to $\mathrm{Soc}(A,\tau)$. Since $A$ is cyclic, we have $\tau(\mathrm{Soc}(A,\tau))=\mathrm{Soc}(A,\tau)$. Inductively, for all $k \geq 1$ we have 
	$$ \tau^k(\mathrm{Soc}(A,\tau)) = \mathrm{Soc}(A,\tau) $$ 
	and 
	\begin{equation}  \label{tau-k} 
		\tau^k(a+b)=\tau^k(a) + \tau^k(b) \mbox{ for } a \in \mathrm{Soc}(A,\tau), b \in A.  
	\end{equation} 
	 
Now (\ref{Soc-k}) follows immediately from (\ref{tau-k}). If $\tau$ has order $m$ and $\gcd(k,m)=1$, take $h \geq 1$ with $kh \equiv 1 \pmod{m}$. Then 
$$ 	\mathrm{Soc}(A,\tau) \subseteq \mathrm{Soc}(A,\tau^k) \subseteq \mathrm{Soc}(A,(\tau^k)^h)=\mathrm{Soc}(A,\tau), $$   
giving equality in (\ref{Soc-k}).

<!-- corollary-2: body_md -->
\label{cyclic-powers} 
Let $A$ be a finite cyclic group and let $\tau$ be a counterexample for $A$. Let $m$ be the order of $\tau$. If $\gcd(k,m)=1$ then $\tau^k$ is also a counterexample for $A$.
<!-- end corollary-2 -->
<!-- proof-8: body_md -->
Let $\sigma \in \mathrm{Sym}_0(A)$ be such that $\sigma$, $\tau$ satisfy (\ref{st-conds}). Then $\tau \in \mathrm{Aut}_\mathrm{mag}(A,\sigma)$ by Proposition \ref{recip}, and, since $\mathrm{Aut}_\mathrm{mag}(A,\sigma)$ is a group, $\tau^k \in \mathrm{Aut}_\mathrm{mag}(A.\sigma)$. Using Proposition \ref{recip} again, we have $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau^k)$, and, by Proposition \ref{Soc-k-prop}, $\mathrm{Soc}(A,\tau^k)=\{0\}$. Hence $\tau^k$ is a counterexample for $A$.

<!-- proposition-7: body_md -->
Let $A$ be a group of prime order, and let $\tau \in \mathrm{Sym}_0(A)$. Then Conjecture \ref{main-conj} for $(A,\tau)$ is equivalent to the following description of $\mathrm{Aut}_\mathrm{mag}(A,\tau)$: 
$$ \mathrm{Aut}_\mathrm{mag}(A,\tau) = \begin{cases} 
 \{\sigma \in \mathrm{Sym}_0(A) : \tau \sigma = \sigma \tau\} & \mbox{if } \tau \in \mathrm{Aut}_{\mathrm{gp}}(A), \\ 
 \{\sigma \in \mathrm{Aut}_{\mathrm{gp}}(A) : \tau \sigma = \sigma \tau\} & \mbox{if } \tau \not \in \mathrm{Aut}_{\mathrm{gp}}(A). 
\end{cases} $$
<!-- end proposition-7 -->
<!-- proof-9: body_md -->
For arbitrary $A$, Proposition \ref{autom} shows unconditionally that if $\tau \in \mathrm{Aut}_{\mathrm{gp}}(A)$ then $\mathrm{Aut}_\mathrm{mag}(A,\tau) = \{\sigma \in \mathrm{Sym}_0(A) : \tau \sigma = \sigma \tau\}$. 
 
Now suppose $A$ has prime order and that $\tau \not \in \mathrm{Aut}_{\mathrm{gp}}(A)$. Then $\mathrm{Soc}(A,\tau) \neq A$, so $\mathrm{Soc}(A,\tau)=\{0\}$. If  
$\mathrm{Aut}_\mathrm{mag}(A,\tau) = \{\sigma \in \mathrm{Aut}_{\mathrm{gp}}(A) : \tau \sigma = \sigma \tau\}$ then, for any $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$, we have  $\mathrm{Soc}(A,\sigma)=A \neq \{0\}$. Thus Conjecture \ref{main-conj} holds for $(A,\tau)$. Conversely, assume that Conjecture \ref{main-conj} holds for $(A,\tau)$. If $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$ then $\mathrm{Soc}(A,\sigma) \neq \{0\}$, so $\mathrm{Soc}(A,\sigma)=A$ and $\sigma \in \mathrm{Aut}_{\mathrm{gp}}(A)$. Moreover $\tau \sigma = \sigma \tau$ by Proposition \ref{tilde-comm}. This shows the inclusion $\mathrm{Aut}_\mathrm{mag}(A,\tau) \subseteq  \{\sigma \in \mathrm{Aut}_{\mathrm{gp}}(A) : \tau \sigma = \sigma \tau\}$. For the reverse inclusion, let $\sigma \in \mathrm{Aut}_{\mathrm{gp}}(A)$ with $\tau \sigma =\sigma \tau$. Then $\tau \in \mathrm{Aut}_\mathrm{mag}(A,\sigma)$ by the first case of the Proposition (with $\tau$ and $\sigma$ interchanged), so $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$ by Proposition \ref{recip}.

<!-- proposition-8: body_md -->
\label{S-lem} 
Let $A$ be an arbitrary abelian group and let $\tau \in \mathrm{Sym}_0(A)$. Let 
$$ S= \mathrm{Supp}(\tilde{\tau}) = \{ a \in A : -\tau(-a) \neq a \},  $$ 
and let $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$. Then, for each $b \in A$, we have   
	$$ \sigma(b+S) \backslash S =(\sigma(b)+S) \backslash S. $$
<!-- end proposition-8 -->
<!-- proof-10: body_md -->
First observe that $\sigma(S)=S$ since, by Proposition \ref{tilde-comm}, $\sigma$ commutes with $\tilde{\tau}$.  
                                                                                         
 
For $a \in A \backslash S$ and $b \in A$, we have  
\begin{eqnarray*} 
  a \in b+S & \Leftrightarrow & a-b \in S \\ 
  			& \Leftrightarrow & \tau(b-a) \neq b-a \\ 
            & \Leftrightarrow & \tau(b-a) - \tau(-a) \neq b-a -(-a) \\ 
            & \Leftrightarrow & a \cdot_\tau b \neq b. 
\end{eqnarray*} 
Fix $b \in A$. For $a \in A$ we have $\sigma(a) \not \in S$ if and only if $a \not \in S$. Thus if $\sigma(a) \not \in S$, we have  
\begin{eqnarray*} 
	\sigma(a) \in \sigma(b+S) & \Leftrightarrow & a \in b+S \\ 
	                          & \Leftrightarrow & a \cdot_\tau b \neq b \\ 
	                          & \Leftrightarrow & \sigma(a) \cdot_\tau \sigma(b) \neq \sigma(b) \\ 
	                          & \Leftrightarrow & \sigma(a)  \in \sigma(b) + S. 
\end{eqnarray*} 
Hence $\sigma(b+S) \backslash S = (\sigma(b)+S) \backslash S$.

<!-- theorem-1: body_md -->
\label{trans-thm} 
Let $A=C_n$ with $n>2$, and let $\tau \in \mathrm{Sym}_0(A)$ be a transposition $\tau=(ab)$ where $\gcd(b-a,n)=1$. Then	 
\begin{equation} \label{transp-aut} 
		\mathrm{Aut}_\mathrm{mag}(A,\tau) = \begin{cases} \{\mathrm{id}\} & \mbox{if } b \neq -a; \\ 
		\{\mathrm{id}, \mathrm{inv}\} & \mbox{if } b =-a; 
	\end{cases}  
\end{equation} 
where $\mathrm{inv}(x)=-x$. In particular, Conjecture \ref{main-conj} holds for $(A, \tau)$.
<!-- end theorem-1 -->
<!-- proof-11: body_md -->
We begin with a preliminary observation. If $b=2a$ and $a=2b$ in $A$ then $3(b-a)=0$ and the coprimality hypothesis forces $n=3$. For $n=3$, we necessarily have $\tau=(12)=\mathrm{inv}$ and $\mathrm{Aut}_\mathrm{mag}(A,\tau)=\{\mathrm{id}, \tau \}=\mathrm{Aut}_{\mathrm{gp}}(A,\tau)$. Thus the Theorem holds for $n=3$. We henceforth assume that $n>3$, and further (by swapping $a$ and $b$ if necessary) that $b \neq 2a$.  
 
In the notation of Proposition \ref{S-lem}, we have  
$$ S= \mathrm{Supp}(\tilde{\tau})=	\{-a,-b\}. $$  
For $0 \leq j \leq n-1$, let $T_j=a+j(a-b)+S = \{j(a-b), (j+1)(a-b)\}$. In particular,  
$$  	T_0 =  \{ 0, a-b\},  \qquad 	T_{n-1} = \{ b-a, 0 \}.  $$  
The sets $T_j$ are pairwise distinct by the coprimality hypothesis, and for each $c \neq 0$ in $A$ there is a unique index $i$ such that  
$$ c \in T_j \Leftrightarrow j=i \mbox{ or } i+1. $$  
Moreover, there is a unique $j_0$ with $1 \leq j_0 \leq n-2$ such that $T_{j_0} = S= \{-a,-b\}$. We then have $T_{j_0-1} = \{b-2a, -a\}$ and $T_{j_0+1} = \{ -b, a-2b \}$.  
 
Next consider the sets $T'_j=T_j \backslash S$. We have $T'_{j_0}= \emptyset$ and  
$$ T'_{j_0-1}=\{b-2a\}=\{(j_0-1)(a-b)\}, \quad T_{j_0+1} = \{(j_0+2)(a-b)\} = \{a-2b \}, $$  
while $T'_j=T_j$ if $j \neq j_0$, $j_0 \pm 1$.  
 
Now let $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$. By Proposition \ref{S-lem}, $\sigma$ permutes the sets $T'_j$. In particular, as $\sigma(0)=0$, the two sets $T'_0$, $T'_{n-1}$ containing $0$ must either be fixed by $\sigma$ or swapped by $\sigma$.  
We consider these two possibilities separately. 
	 
First suppose that $\sigma(T'_0)=T'_0$ and $\sigma(T'_{n-1}) = T'_{n-1}$. We will show that in this case $\sigma=\mathrm{id}$. 
We first show by induction that for $0 \leq k \leq j_0-1$ we have  
\begin{equation} \label{transp-ind} 
	\sigma(T'_k)=T'_k, \qquad \sigma(k(a-b))=k(a-b). 
\end{equation} 
This is true for $k=0$. Suppose that (\ref{transp-ind}) holds for some $k<j_0-1$. Then $T'_k=\{k(a-b),(k+1)(a-b)\}$ is fixed by $\sigma$. Since $\sigma$ fixes $k(a-b)$, it must also fix $(k+1)(a-b)$. Since $(k+1)(a-b) \in T'_j$ if and only if $j=k$ or $k+1$, and $\sigma$ fixes $T'_k$, it must also fix $T'_{k+1}$. Hence (\ref{transp-ind}) also holds for $k+1$, completing the induction. A similar argument, using descending induction and starting from $\sigma(T'_{n-1})=T'_{n-1}$, shows that (\ref{transp-ind}) also holds for $n-1 \geq k \geq j_0+2$. Thus $\sigma(c)=c$ for all $c \in A$ with the possible exceptions of $c=j_0(b-a)=-a$ and $c=(j_0+1)(b-a)=-b$. Hence $\sigma=\mathrm{id}$ or $\sigma=(-a,-b)=\tilde{\tau}$. We will show that $\tilde{\tau}$ cannot be a magma automorphism when $n>3$. Let $x=-2a$ and $y=b-2a$. Since $a \neq b$ and we have assumed that $b \neq 2a$, we have $\tau(-x)=-x$ and $\tilde{\tau}(x)=x$. As $n>2$ and $\gcd(b-a,n)=1$, we have $2(b-a) \neq 0$, so that $b-2a \neq -a$, $-b$. Thus $\tilde{\tau}(y)=y$. We now calculate 
\begin{eqnarray*} 
	 \tilde{\tau}( x \cdot_\tau y) & = & \tilde{\tau}(\tau(y-x)-\tau(-x)) \\ 
	                         & = & \tilde{\tau}(\tau(b)-\tau(-x)) \\ 
	                         & = & \tilde{\tau}(a-2a) \\ 
	                         & = & -b 
\end{eqnarray*} 
and 
\begin{eqnarray*} 
	\tilde{\tau}(x) \cdot_\tau \tilde{\tau}(y) & = & \tau( \tilde{\tau}(y)-\tilde{\tau}(x)) - \tau( -\tilde{\tau}(x)) \\  
								& = & \tau(y-x) - \tau(-x) \\ 
								& = & \tau(b)-\tau(-x) \\ 
								& = & a -(-x) \\ 
								& = & -a.						 
\end{eqnarray*}         
As $-b \neq -a$ in $A$, this shows that $\tilde{\tau} \not \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$. 
 
Now suppose that $T'_0$ and $T'_{n-1}$ are swapped by the magma automorphism $\sigma$. We will show that in this case the only possibility is $\sigma =\mathrm{inv}$, and this can only occur if $b=-a$.  
We first show by induction that for $0 \leq k \leq j_0-1$, the following hold: 
\begin{equation} \label{inv-bound} 
	n-1-k \geq j_0+1;  
\end{equation} 
\begin{equation} \label{inv-ind} 
	\sigma(k(a-b))=k(b-a), \qquad  	\sigma(k(b-a))=k(a-b). 
\end{equation} 
\begin{equation} \label{inv-set} 
	\sigma(T'_k) = T'_{n-1-k}, \qquad \sigma(T'_{n-1-k})=T'_k;  
\end{equation} 
Clearly, these all hold when $k=0$. Suppose that they hold for some $k < j_0-1$. Then $|T'_k|=2$ and $T'_k=\{k(a-b), (k+1)(a-b)\}$. By (\ref{inv-set}), $\sigma$ swaps $T'_k$ and $T'_{n-1-k}$. Hence $|T'_{n-1-k}|=2$ as well, and $T'_{n-1-k}=\{k(b-a),(k+1)(b-a)\}$. In particular, $n-1-k \neq j_0+1$. Together with (\ref{inv-bound}), this shows that $n-1-k \geq j_0+2$, so that (\ref{inv-bound}) holds for $k+1$. Moreover, it follows from (\ref{inv-ind}) that $\sigma$ swaps $(k+1)(a-b)$ and $(k+1)(b-a)$, so (\ref{inv-ind}) holds for $k+1$. Finally, since  
$$  (k+1)(a-b) \in T'_j \Leftrightarrow j=k \mbox{ or } j=k+1 $$ 
and 
$$  (k+1)(b-a) \in T'_j \Leftrightarrow j=n-1-k \mbox{ or } j= n-2-k, $$ 
it follows that $\sigma$ swaps $T'_{k+1}$ and $T'_{n-2-k}$. This shows that (\ref{inv-set}) holds for $k+1$, completing the induction. Now taking $k=j_0-1$ in (\ref{inv-bound}) and (\ref{inv-set}), we have that $n-j_0 \geq j_0+1$ and that $\sigma$ swaps $T'_{j_0-1}$ and $T'_{n-j_0}$. Thus $|T'_{n-j_0}|=|T'_{j_0-1}|=1$. Hence $n-j_0=j_0+1$, so that $n$ is odd and $j_0=\frac{1}{2}(n-1)$. As $T_{j_0}=\{j_0(a-b), (j_0+1)(a-b)\}=\{-a,-b\}$, we have $b=-a$. Since (\ref{inv-ind}) holds for $0 \leq k \leq \frac{1}{2}(n-1)$, we now know that $\sigma(c)=-c$ for all $c \in A$ with the possible exceptions of $c=\frac{1}{2}(n-1)(a-b)=-a=b$ and $c=\frac{1}{2}(n+1)(a-b)=-b=a$. Thus either $\sigma=\mathrm{inv}$ or $\sigma=\tilde{\tau} \circ \mathrm{inv}$, where in this case $\tilde{\tau}=\tau$. Now $\mathrm{inv} \in \mathrm{Aut}_{\mathrm{gp}}(A)$ and $\mathrm{inv}$ commutes with $\tau$, so $\mathrm{inv} \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$ by Proposition \ref{autom}. We have already seen in the first case that $\tilde{\tau} \not \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$. Thus $\tilde{\tau} \circ \mathrm{inv} \not \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$.  
 
We have now shown both cases of (\ref{transp-aut}). In particular, $\mathrm{Aut}_\mathrm{mag}(A,\tau) \subseteq \mathrm{Aut}_{\mathrm{gp}}(A)$, so Conjecture \ref{main-conj} holds for $(A, \tau)$.

<!-- lemma-1: body_md -->
\label{submags} 
	Let $A$ be a finite abelian group, and let $\tau \in \mathrm{Sym}_0(A)$. 
	\begin{itemize} 
		\item[(i)] Any submagma of $(A,\tau)$ is a union of orbits of the subgroup $\langle \tau, \tilde{\tau}\rangle$ of $\mathrm{Sym}_0(A)$. In particular, if $\langle \tau, \tilde{\tau}\rangle$ is transitive on $A\backslash\{0\}$ then the only submagmas are $\{0\}$ and $A$. 
		\item[(ii)] The submagmas of size $2$ are the sets $\{0,a\}$ where $a$ satisfies $\tau(a)=\tilde{\tau}(a)=a$. 
		\item[(iii)] If $B$ is a subgroup of $A$ with $\tau(B)=B$ then $B$ is a submagma of $(A,\tau)$.  
		\item[(iv)] If $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$ then the set of fixed points of $\sigma$ 
		$$   \mathrm{Fix}(\sigma)=\{ a \in A : \sigma(a) =a \} $$ 
		is a submagma of $(A,\tau)$.  
		\item[(v)] If $\tau \in \mathrm{Aut}_{\mathrm{gp}}(A)$ and $B$ is any union of $\tau$-orbits containing $0$, then $B$ is a submagma of $(A,\tau)$.  
	\end{itemize}
<!-- end lemma-1 -->
<!-- proof-12: body_md -->
(i) If $B$ is a submagma of $(A,\tau)$ and $a \in B$ then $B$ contains $0 \cdot a = \tau(a)$ and $a \cdot a = \tilde{\tau}(a)$. 
	 
	(ii) If $B=\{0,a\}$ is a submagma of size $2$ then $\tau(a)=0 \cdot a$ and $\tilde{\tau}(a) = a \cdot a$ are in $A \backslash\{0\}$, so $\tau(a)=\tilde{\tau}(a)=a$. Conversely, if $\tau(a)=\tilde{\tau}(a)=a$ then 
	$$ 0 \cdot 0 = a \cdot 0 = 0, \quad 0 \cdot a = a \cdot a =a, $$ 
	so $\{0,a\}$ is a submagma. 
	 
	(iii) This is clear from the definition of $\cdot_\tau$. 
	 
	(iv) $0 \in \mathrm{Fix}(\sigma)$, and if $a$, $b \in \mathrm{Fix}(\sigma)$ then  
	$$ a \cdot_\tau b = \sigma(a) \cdot_\tau \sigma(b) = \sigma(a \cdot_\tau b), $$ 
	so $a \cdot_\tau b \in \mathrm{Fix}(\sigma)$.  
	 
	(v) If $\tau \in \mathrm{Aut}_{\mathrm{gp}}(A)$ and $B$ is a union of $\tau$-orbits then for $a$, $b \in B$ we have  
	$$ a \cdot_\tau b = \tau(b-a)-\tau(-a) = \tau(b)-\tau(a)+\tau(a)=\tau(b) \in B. $$

<!-- lemma-2: body_md -->
\label{long-cycle} 
Let $A$ be a finite abelian group of odd order $n \geq 5$, and let $\tau$, $\sigma \in \mathrm{Sym}_0(A)$ satisfy (\ref{st-conds}). Suppose that the order of $\tau$ is divisible by a prime $q  \geq (n+1)/2$.  Then $\sigma$ is supported on a set of cardinality at most $n-2-q$.
<!-- end lemma-2 -->
<!-- proof-13: body_md -->
Clearly $q$ is odd and $\tau$ contains exactly one cycle $\zeta$ of length $q$. Let $X=A \backslash (\mathrm{Supp}(\zeta) \cup \{0\})$, a set of cardinality $n-1-q$. Then $\tau = \zeta \pi$, where $\pi$ is a permutation with $\mathrm{Supp}(\pi) \subseteq X$. Now $\sigma$ commutes with $\tau$, and it follows from Proposition \ref{perm-comm} that $\sigma = \zeta^a \rho$ for some $a$ with $0 \leq a \leq q-1$ and some permutation $\rho$ with $\mathrm{Supp}(\rho) \subseteq X$. Thus $\rho$ cannot contain a cycle of length $q$. But $\sigma$ also commutes with $\tilde{\tau}=\tilde{\zeta} \tilde{\pi}$, and $\mathrm{Supp}(\tilde{\zeta}) \neq \mathrm{Supp}(\zeta)$ as $q$ is odd. Hence $a=0$, and $\rho$ must fix $\mathrm{Supp}(\zeta) \cup \mathrm{Supp}(\tilde{\zeta})$, a set of cardinality at least $q+1$. It is therefore supported on a set of cardinality at most $n-2-q$.

<!-- head-1: body_md -->
Throughout this section, let $A=C_{p^d}$ be a cyclic group of order $p^d$, with $p$ prime and $d \geq 1$. In this case, $A$ has a unique minimal (nontrivial) subgroup $M$, namely the subgroup of order $p$. Thus if $\mathrm{Soc}(A,\tau) \neq \{0\}$ then $M \subseteq \mathrm{Soc}(A,\tau)$. We therefore have 
\begin{equation}  \label{M-Soc} 
 \mathrm{Soc}(A,\tau) \neq \{0\} \Leftrightarrow \tau(a+b)=\tau(a)+\tau(b) \mbox{ for all } a \in M, b \in A.  
\end{equation} 
This allows us to improve on Proposition \ref{Soc-k-prop}. 
 
\begin{proposition}  \label{compose-soc} 
If $\rho$, $\tau \in \mathrm{Sym}_0(A)$ with $\mathrm{Soc}(A,\rho) \neq \{0\} \neq \mathrm{Soc}(A,\tau)$ then $\mathrm{Soc}(A,\rho \tau) \neq \{0\}$. 
\end{proposition} 
\begin{proof} 
For any $\sigma \in \mathrm{Sym}_0(A)$ we have  
$$ \mathrm{Soc}(A,\sigma) \neq \{0\} \Leftrightarrow M \subseteq \mathrm{Soc}(A,\sigma). $$ 
In particular, $\rho$ and $\tau$ satisfy these equivalent conditions. Thus, using (\ref{M-Soc}), for $a \in M$ and $b \in A$ we have   
$$ \rho(\tau(a+b))=\rho(\tau(a)+\tau(b))=\rho(\tau(a)) + \rho(\tau(b)), $$ 
showing that $a \in \mathrm{Soc}(A,\rho \tau)$. Hence $M \subseteq \mathrm{Soc}(A,\rho \tau)$.  
\end{proof} 
	 
\begin{definition} 
We call $\sigma \in \mathrm{Sym}_0(A)$ {\em special} if it satisfies the following properties: 
\begin{itemize} 
	\item[(i)] $\sigma$ has order $q^e$ where $q$ is prime and $e \geq 2$;  
	\item[(ii)] $\mathrm{Soc}(A,\sigma^q) \neq \{0\}$. 
\end{itemize} 
\end{definition} 
 
\begin{proposition} \label{reduce-to-pp} 
Let $\tau \in \mathrm{Sym}_0(A)$ and let $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$ with $\mathrm{Soc}(A,\sigma)=\{0\}$. Then there is some power $\sigma^k$ of $\sigma$ such that $\mathrm{Soc}(A,\sigma^k)=\{0\}$ and $\sigma^k$ either has prime order or is a special permutation. 
\end{proposition} 
\begin{proof}  
Let $\sigma$ have order $m$ with prime factorisation $m=q_1^{e_1} \cdots q_r^{e_r}$. For $1 \leq i \leq r$, let $m_i \in \mathbb{Z}$ satisfy  $m_i \equiv 1 \pmod{q_i^{e_i}}$ and $m_i \equiv 0 \pmod{q_j^{e_j}}$ for $j \neq i$. Set $\sigma_i = \sigma^{m_i} \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$. Then $\sigma_i$ has order $q_e^{e_i}$ and $\sigma=\sigma_1 \cdots \sigma_r$. If $\mathrm{Soc}(A,\sigma_i) \neq \{0\}$ for all $i$ then, applying Proposition \ref{compose-soc} repeatedly, we have $\mathrm{Soc}(A,\sigma) \neq \{0\}$, contrary to our hypotheses. Thus, for some $i$, we have $\mathrm{Soc}(A,\sigma_i)=\{0\}$. To simplify notation, write $q=q_i$, $e=e_i$, so $\sigma_i$ has order $q^e$. If $\mathrm{Soc}(A,\sigma_i^{q^{e-1}})=\{0\}$ then take $k=m_i q^{e-1}$ so that $\sigma^k$ has prime order $q$. Otherwise let $h$ be minimal such that $\mathrm{Soc}(A,\sigma_i^{q^h})\neq \{0\}$. Then $1 \leq h <e$ and, taking $k=m_i q^{h-1}$, we have that $\sigma^k$ is a special permutation of order $q^{e-h+1}$.  
\end{proof} 
 
\begin{proposition} \label{prime-special} 
If Conjecture \ref{main-conj} does not hold for $A$, then there are $\tau$, $\sigma$ satisfying (\ref{st-conds}) such that each of $\tau$, $\sigma$ either has prime order or is a special permutation. 
\end{proposition} 
\begin{proof} 
As Conjecture \ref{main-conj} does not hold for $A$, there are $\sigma$, $\tau$ satistifying (\ref{st-conds}). 
By Proposition \ref{reduce-to-pp}, we may replace $\sigma$ by a suitable power $\sigma^k$ so that $\sigma$ either has prime order or is a special permutation. We still have $\sigma \in \mathrm{Aut}_\mathrm{mag}(A,\tau)$, so, by Proposition \ref{recip}, also $\tau \in \mathrm{Aut}_\mathrm{mag}(A,\sigma)$. Applying Proposition \ref{reduce-to-pp} again (with $\sigma$ and $\tau$ interchanged), we can replace $\tau$ by a suitable power $\tau^j$ so that also either $\tau$ has prime order or $\tau$ is a special permutation. 
\end{proof} 
 
\begin{corollary} \label{special_or_smallprime} 
Let $A$ be cyclic of order $p^d \geq 5$, with $p \geq 3$. If Conjecture \ref{main-conj} does not hold for $A$ then there is a counterexample $\tau$ for $A$ which either has prime order $q \leq (p^d-1)/2$ or is a special permutation. 	 
\end{corollary} 
 \begin{proof} 
 Let $\sigma$, $\tau$ be as in Proposition \ref{prime-special}. Then $\tau$ is as required unless $\tau$ has prime order $q \geq (p^d+1)/2$. In that case, $\sigma$ is either a special permutation or has prime order $r \leq (p^d-1)/2$ by Lemma \ref{long-cycle}, and $\tau \in \mathrm{Aut}_\mathrm{mag}(A,\sigma)$ by Proposition \ref{recip}. Thus $\sigma$ is a counterexample with the required properties.	 
 \end{proof} 
  
 We now describe the special permutations in the case that $A$ has prime order. 
  
 \begin{proposition} \label{special-p} 
 Let $A$ be a cyclic group of order $p$ and let $\sigma \in \mathrm{Sym}_0(A)$ be a special permutation of order $q^e$ with $q$ prime and $e \geq 2$. Then $\sigma^q \in \mathrm{Aut}_{\mathrm{gp}}(A)$, $q^e$ divides $p-1$, and $\sigma$ consists of $(p-1)/q^e$ cycles of length $q^e$. 
 \end{proposition} 
 \begin{proof} 
 Since $\mathrm{Soc}(A,\sigma^q) \neq \{0\}$, we have $\mathrm{Soc}(A,\sigma^q)=A$, so $\sigma^q$ is an automorphism of $A$ of order $q^{e-1}>1$. Thus $\sigma^q$ has no fixed points except $0$ and consists of $(p-1)/q^{e-1}$ cycles of length $q^{e-1}$. It follows that $\sigma$ must consist of $(p-1)/q^e$ cycles of length $q^e$. In particular, $q^e$ divides $p-1$.  
 \end{proof}

<!-- lemma-3: body_md -->
\label{ad hoc} 
Let $A$ be the cyclic group of order $p=17$ or $19$. Suppose that $A$ does not satisfy Conjecture \ref{main-conj}. Then there is a counterexample $\tau$ for $A$ satisfying one of the following conditions: 
\begin{enumerate} 
	\item[(i)] $\tau$ is a special permutation; 
	\item[(ii)] $\tau$ has order $2$ or $3$; 
	\item[(iii)] $\tau$ is a single cycle of length $q=5$ or $7$; 
	\item[(iv)] $\tau$ contains a cycle $\zeta$ of length $q=5$ or $7$ for which $\mathrm{Supp}(\zeta)$ and $\mathrm{Supp}(\tilde{\zeta})$ are disjoint, and $\tau = \zeta \tilde{\zeta}^h$ with $1 \leq h \leq q-1$.  
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-19: body_md -->
We treat the cases $p=17$ and $p=19$ simultaneously. Again let $\sigma$, $\tau$ satisfy (\ref{st-conds}). By Proposition \ref{prime-special} we may assume that each of $\sigma$ and $\tau$ either has prime order or is a special permutation, and we are free to swap $\sigma$ and $\tau$ by Proposition \ref{recip}. If either $\sigma$ or $\tau$ is a special permutation, or has order $2$ or $3$, we have a counterexample satisfying (i) or (ii). From now on we assume that $\tau$, $\sigma$ have prime orders $q$, $r \geq 5$ respectively. 
 
If $q \geq 11$ then, by Lemma \ref{long-cycle}, $\sigma$ is supported on a set of cardinality at most $6$, so $r \leq 5$. If $r \neq 5$ then $\sigma$ satisfies (ii). If $r=5$ then $\sigma$ is a single cycle of length $5$, so satisfies (iii). Similarly, if $r \geq 11$ then $\tau$ satisfies (ii) or (iii).  
 
It remains to consider the cases $q$, $r \in \{5,7\}$. Moreover, we assume each of $\tau$, $\sigma$ contains at least two cycles, since otherwise $\tau$ or $\sigma$ satisfies (iii). We treat the cases $q=5$ and $q=7$ separately 
 
 Suppose that $q=5$. Then  $\tau=\zeta_1 \zeta_2 \zeta_3$ or $\tau=\zeta_1 \zeta_2$ where the $\zeta_i$ are pairwise disjoint cycles of length $5$.  
 
First let $\tau = \zeta_1 \zeta_2 \zeta_3$. Since $\sigma$ commutes with $\tau$ and has order $5$ or $7$, it follows from Proposition \ref{perm-comm} that $\sigma= \zeta_1^a \zeta_2^b \zeta_3^c$ where $0 \leq a, b, c \leq q-1$. Relabelling the $\zeta_i$ and replacing $\sigma$ by a power, we assume without loss of generality that $a=1$. Then $\sigma= \zeta_1 \zeta_2^b \zeta_3^c$. As $\sigma$ also commutes with $\tilde{\tau} = \tilde{\zeta}_1 \tilde{\zeta}_2 \tilde{\zeta}_3$, we have $\zeta_1^d=\tilde{\zeta}_i$ for some some $d \in \{1,2,3,4\}$ and $i \in \{1,2,3\}$. Since $\zeta_1$ has odd length, we cannot have $i=1$, so without loss of generality, $i=2$. Thus we have 
$$ \tau = \zeta_1 \tilde{\zeta}_1^d \zeta_3, \qquad \sigma = \zeta_1 \tilde{\zeta}_1^{db} \zeta_3^c. $$ 
Now $\zeta_3$ is disjoint from $\zeta_1$ and $\zeta_2=\tilde{\zeta}_1^d$, and hence $\tilde{\zeta}_3$ is also disjoint from $\zeta_1$ and $\zeta_2$. As $p \leq 19$, it follows that $\zeta_3$ and $\tilde{\zeta}_3$ cannot be disjoint.  
Since $\sigma$ commutes with $\tilde{\tau}$, this means that $c=0$. Thus $\sigma$ is as in (iv).  
 
Now let $\tau=\zeta_1 \zeta_2$. By Proposition \ref{perm-comm}, $\sigma = \zeta_1^a \zeta_2^b \pi$ where $\pi$ has support of size at most $8$. If $a=b=0$ then $\sigma=\pi$, and, since $\sigma$ has order $5$ or $7$, it must consist of a single cycle, contrary to hypothesis. So, without loss of generality, we may assume that $a=1$. Then $r \neq 7$, so $r=5$ and either $\pi=\mathrm{id}$ or $\pi$ is a single cycle of length $5$ (disjoint from $\zeta_1$ and $\zeta_2$). However, if $\pi$ is a cycle of length $5$ and $b \neq 0$ then $\sigma$ consists of three cycles of length $5$, and, interchanging $\sigma$ and $\tau$, we are reduced to the previous case. We may therefore assume that $\sigma$ contains exactly two cycles, and either $\sigma = \zeta_1 \zeta_2^b$ with $b \neq 0$ or $\sigma = \zeta_1 \zeta'$ where $\zeta'$ is a further $5$-cycle disjoint from $\zeta_1$ and $\zeta_2$. Moreover $\sigma$ commutes with $\tilde{\tau} = \tilde{\zeta}_1 \tilde{\zeta}_2$. 
 
If $\sigma = \zeta_1 \zeta_2^b$ then $\zeta_2=\tilde{\zeta}_1^d$ for some $d$, since the cycles $\zeta_1$, $\zeta_2$, $\tilde{\zeta}_1$, $\tilde{\zeta}_2$ cannot be pairwise disjoint. So $\tau$ is as in (iv).  
 
If $\sigma = \zeta_1 \zeta'$ then, for some $d$, we have $\zeta_1^d=\tilde{\zeta}_2$ or $\zeta'=\tilde{\zeta}_1^d$ or $\zeta'=\tilde{\zeta}_2^d$, so that, respectively, $\tau=\zeta_1 \tilde{\zeta}_1^d$ or $\sigma=\zeta_1 \tilde{\zeta}_1^d$ or  
$\sigma=\zeta_1 \tilde{\zeta}_2^d$. In the first two of these cases, either $\tau$ or $\sigma$ is as in (iv). We show that the remaining case $\tau=\zeta_1 \zeta_2$, $\sigma=\zeta_1 \tilde{\zeta}_2^d$ cannot occur. Indeed, for $\tau$ and $\sigma$ to commute, $\zeta_1$, $\zeta_2$, $\tilde{\zeta}_2$ must be pairwise disjoint, and if $\sigma$ and $\tilde{\tau}$ commute then also $\tilde{\sigma}$ and $\tau$ commute, so $\tilde{\zeta}_1$, $\zeta_2$, $\zeta_1$ are pairwise disjoint. This would give $4$ pairwise disjoint $5$-cycles, which is impossible. 
 
Finally, suppose that $q=7$. We may assume that $\tau=\zeta_1 \zeta_2$, the product of two cycles of length $7$. Then $\sigma = \zeta_1^a \zeta^b \pi$ where the support of $\pi$ has cardinality at most $4$. If $a=b=0$ then $r \leq 3$. So without loss of generality, $\sigma=\zeta_1 \zeta_2^b$. If $b=0$ then $\sigma$ satisfies (iii), so  
we can assume $1 \leq b \leq 6$. Since $\sigma$ commutes with $\tilde{\tau} = \tilde{\zeta}_1 \tilde{\zeta}_2$ and there cannot be more than $2$ mutually disjoint cycles of length $7$, we have that $\tilde{\zeta}_1$ is a power of $\zeta_2$ and $\tau$ satisfies (iv).
