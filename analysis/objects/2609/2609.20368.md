\label{sec:linearized_cone_def}
  Let $x^\star\in\mathds{R}^n$ be a feasible point of~\eqref{eq:mpbvc}.
  Then the linearized cone of~\eqref{eq:mpbvc} at $x^\star$ is equal to

<!-- proof-1: body_md -->
The proof is based on the proof of~\cite[Lemma~4]{mpvc}.
  We note that the linearized cone of~\eqref{eq:mpbvc} is defined as
  \begin{align*}
    \left\{
    d\in\mathds{R}^n \biggm|
    \begin{array}{rl}
      \nabla g_k(x^\star)^Td=0\,&\mathrm{for}\,k\,\in\mathcal{N}_g,\\
      \nabla h_l(x^\star)^Td\geq0\,&\mathrm{for}\,l\,\in\mathcal{I}_{h}(x^\star),\\
      \nabla H_{i}(x^\star)^Td\,\geq 0\,&\mathrm{for}\,i\,\in\mathcal{I}_{0}^H(x^\star),\\
      \nabla \theta_{ij}(x^\star)^Td \geq 0\,&\mathrm{for}\,(i,j)\,\in\mathcal{I}_{0}(x^\star)\cup\mathcal{I}_{(+,0)}(x^\star)
    \end{array}
    \right\},
  \end{align*}
  where $\theta_{ij}(x)\coloneqqG_{i,j}(x)H_{i}(x)$.
  We distinguish the following cases with respect to the gradients $\nabla\theta_{ij}(x^\star)$:
\begin{itemize}
    \item For $(i,j) \in \mathcal{I}_{(0,-)}$ it holds that $\nabla\theta_{ij}(x^\star)^Td = G_{i,j}(x^\star)\nabla H_{i}(x^\star)^Td \geq 0$
    which is equivalent to ${\nabla H_{i}(x^\star)^Td \leq 0}$, since $G_{i,j}(x^\star)<0$. We obtain with $\nabla H_{i}(x^\star)^Td\geq 0$ that $\nabla H_{i}(x^\star)^Td=0$. 
    \item For $(i,j) \in \mathcal{I}_{(0,0)}$ it holds that $\nabla\theta_{ij}(x^\star)^Td = 0\geq 0$, i.e.,\xspace we can exclude this case, since it provides no information.
    \item For $(i,j) \in \mathcal{I}_{(0,+)}$ it holds that $\nabla\theta_{ij}(x^\star)^Td = G_{i,j}(x^\star)\nabla H_{i}(x^\star)^Td\geq 0$ which is equivalent to  $\nabla H_{i}(x^\star)^Td \geq 0$. This requirement is already included.
    \item For $(i,j) \in \mathcal{I}_{(+,0)}$ it holds that $\nabla\theta_{ij}(x^\star)^Td =H_{i}(x^\star)\nabla G_{i,j}(x^\star)^Td \geq 0$. Since $H_{i}(x^\star)\geq 0$, it follows that ${\nabla G_{i,j}(x^\star)^Td\geq 0}$.
\end{itemize}

\label{vanishing: def: ACQ}
    Let $x$ be a feasible point of a standard NLP with feasible set $X$. We define the standard \textit{tangent cone} $\mathcal{T}(X,x)$ at $x \in X$
    and the \textit{polar cone} $M^{-}$ of a set $M$ as
    \begin{align}
        \mathcal{T}(X,x) &\coloneqq \bigg\{ d\in \mathds{R}^n \Bigm| \exists\{x^{(k)}\}\subset X,\{t^{(k)}\}\to0\,:\, x^{(k)}\to x\text{ and }\frac{x^{(k)}-x}{t^{(k)}}\to d \bigg\},\label{vanishing: T(x)}\\
        M^{-} &\coloneqq \{d \in \mathds{R}^{\tilde{n}} \mid \langle d, x \rangle \leq 0 \text{ for all } x \in M\}.
    \end{align}

  \phantomsection

) if $\mathcal{L}(x)^{-}=\mathcal{T}(X,x)^{-}$.

<!-- proof-4: body_md -->
This proof is analogously to the proof of~\cite[Lemma~2.4]{on_the_abadie}. For completeness we state the proof for MPBVC.
  We start with~\ref{vanishing: lem. T(x) and L(x) beta: caseA}.
   Assume $d \in \mathcal{T}(X,x^\star)$, then there exists a sequence
    $\{x^{(k)}\}$ of points feasible for MPBVC such that $x^{(k)} \to x^\star$
    for $k \to \infty$ which together with a sequence $\{t^{(k)}\}$ of
    step lengths converges to $d$. Our goal is to identify a subsequence
    which is contained entirely in the feasible set of
    $\operatorname{NLP}(\mathcal{B}^H_{+},\mathcal{B}^H_{0},x^\star)$ for a suitably chosen partition.
    Due to the feasibility of the iterates $x^{k}$ it holds that
    $H_{i}(x^{(k)}) \geq 0$. From the continuity of the
    constraint functions it follows that starting from a sufficiently
    large index $K$ it holds that
    $H_{i}(x^{(k)})=0$ for $i \in \mathcal{I}_{(0,-)}^H(x^\star)$ and
    $G_{i,j} (x^{(k)}) > 0$ for
    $(i,j) \in \mathcal{I}_{(+,+)}(x^\star) \cup \mathcal{I}_{(0,+)}(x^\star)$
    for all $k \geq K$ (we can assume that $K = 1$).
    To find a suitable partition of the bi-active indices at $x^\star$ we set  for all $k\in\mathds{N}$
     
    \begin{align*}
      \mathcal{B}^H_{0,k}& \coloneqq \{i\,\mid\,\text{there exists } (i,j) \in \mathcal{I}_{(0,0)}(x^\star)~\text{with}~ (i, j) \in \mathcal{I}_{(0,-)}(x^{k})\}, \\
      \mathcal{B}^H_{+,k}& \coloneqq \mathcal{I}_{(0,0)}^H(x^\star)  \setminus\mathcal{B}^H_{0,k}.
    \end{align*}
     
   We know that the number of partitions of $\mathcal{I}_{(0,0)}(x^\star)$ is finite. We can therefore choose an infinite
    subset $\mathcal{K}\subset\mathds{N}$, such that $\widebar{\mathcal{B}^H_{+}} \coloneqq\mathcal{B}^H_{+,k}$ and $\widebar{\mathcal{B}^H_{0}}\coloneqq\mathcal{B}^H_{0,k}$
    for all $k \in \mathcal{K}$. We see that $x^{(k)}$ is feasible for all $k\in\mathcal{K}$ for $\operatorname{NLP}(\widebar\mathcal{B}^H_{+},\widebar\mathcal{B}^H_{0})$.
    Therefore, the subsequence based on $\mathcal{K}$ shows that $d$ is contained in $\mathcal{T}^{\operatorname{NLP}(\widebar\mathcal{B}^H_{+},\widebar\mathcal{B}^H_{0})}(x^\star)$.

    Conversely, for a value $d$ in the union of tangential cones there exists a given partition $(\mathcal{B}^H_{+},\mathcal{B}^H_{0})$
    such that $d\in\mathcal{T}^{\operatorname{NLP}(\mathcal{B}^H_{+},\mathcal{B}^H_{0},x^\star)}(x^\star)$ with corresponding sequences $\{x^{(k)}\},\{t^{(k)}\}$. Since the feasible set
    of $\operatorname{NLP}(\mathcal{B}^H_{+},\mathcal{B}^H_{0},x^\star)$ is contained within that of~\eqref{eq:mpbvc}, the 
    $x^{(k)}$ are all feasible for \eqref{eq:mpbvc}, showing that $d \in \mathcal{T}(X,x^\star)$.

  We carry on with~\ref{vanishing: lem. T(x) and L(x) beta: caseB}.
    Assume $d\in\mathcal{L}^\mathrm{BVC}(x^\star)$ and take the partition
    \begin{align*}
      \mathcal{B}^H_{0} & \coloneqq \{i\,\mid\,\text{there exists }(i,j)\in\mathcal{I}_{(0,0)}(x^\star)~\textrm{s.t.\xspace}~ \nabla G_{i,j}(x^\star)^Td<0\},\\
      \mathcal{B}^H_{+} & \coloneqq \mathcal{I}_{(0,0)}^H(x^\star) \setminus\mathcal{B}^H_{0}.
    \end{align*} From the product $(\nabla H_{i}(x^\star)^Td)\cdot(\nabla G_{i,j}(x^\star)^Td)\geq0$ and $\nabla H_{i}(x^\star)^Td\geq0$ for all $(i,j)\in\mathcal{I}_{(0,0)}(x^\star)$
    we obtain with the choice of $\mathcal{B}^H_{0}$ that $\nabla H_{i}(x^\star)^Td=0$ for all $i\in\mathcal{B}^H_{0}$. For $\mathcal{B}^H_{+}$ we obtain $\nabla H_{i}(x^\star)^Td\geq0$
    and $\nabla G_{i,j}(x^\star)^Td\geq0$ for all $(i,j)\in\mathcal{B}_{+}$.
 The converse direction is clear by definitions of the respective cones.
 For~\ref{vanishing: lem. T(x) and L(x) beta: caseC} we know that $\mathcal{T}^{\operatorname{NLP}(\mathcal{B}^H_{+},\mathcal{B}^H_{0},x^\star)}(x^\star)\subseteq\mathcal{L}^{\operatorname{NLP}(\mathcal{B}^H_{+},\mathcal{B}^H_{0},x^\star)}(x^\star)$
  for all partitions $\mathcal{B}^H_{+} \mathbin{\mathaccent\cdot\cup} \mathcal{B}^H_{0} = \mathcal{I}_{(0,0)}^H(x^\star)$. If follows
  \begin{align*}
   	\mathcal{T}(X,x^\star) \overset{\ref{vanishing: lem. T(x) and L(x) beta: caseA}}{=} & \bigcup_{\mathcal{B}^H_{+} \mathbin{\mathaccent\cdot\cup} \mathcal{B}^H_{0} = \mathcal{I}_{(0,0)}^H(x^\star)} \mathcal{T}^{\operatorname{NLP}(\mathcal{B}^H_{+},\mathcal{B}^H_{0},x^\star)}(x^\star)
    \\ \subseteq &
    \bigcup_{\mathcal{B}^H_{+} \mathbin{\mathaccent\cdot\cup} \mathcal{B}^H_{0} = \mathcal{I}_{(0,0)}^H(x^\star)} \mathcal{L}^{\operatorname{NLP}(\mathcal{B}^H_{+},\mathcal{B}^H_{0},x^\star)}(x^\star) \overset{(b)}{=}\mathcal{L}^{\mathrm{BVC}}(x^\star).
  \end{align*}
  This is our assertion.

<!-- proof-6: body_md -->
This proof is based on the Implicit Function Theorem (IFT) and modified from~\cite{mpvc,mpbvc_pal}.
    In the first step we use the IFT to show the existence of a continuously differentiable function.
    We then show that $x(t)$ is feasible for all $t\in[0,\varepsilon)$.
    Let a partition $\mathcal{B}^H_{+}\mathbin{\mathaccent\cdot\cup}\mathcal{B}^H_{0}=\mathcal{I}_{(0,0)}^H(x^\star)$ be given and fixed.\\

    \textsc{Step 1.} We define the mapping $z:\mathds{R}^n\to\mathds{R}^q$ with \newline $q\coloneqq\#\mathcal{N}_g+
    \#\mathcal{I}_{(0,-)}^H(x^\star)+ \#\mathcal{B}^H_{0}$ and
    \begin{align*}
        z(x)\coloneqq\begin{pmatrix}
          g(x) \\
          H_{\mathcal{I}_{(0,-)}^H(x^\star)}(x) \\
          H_{\mathcal{B}^H_{0}}(x)\\
        \end{pmatrix}.
    \end{align*}We denote with $z_j$ the $j$-th component of the mapping $z$. 
    Let further $t \in \mathds{R}, y\in\mathds{R}^q$, $d\in\mathds{R}^n$ from~\ref{word:BVC_MFCQ} and 
    define $\hat{h}: \mathds{R}^{q} \times \mathds{R} \to \mathds{R}^q$ with
    \begin{align*}
      \hat{h}_j(y,t)\coloneqq z_j(x^\star + td + J_z(x^\star)^Ty) \text{ for all }j=1,\dots,q,
    \end{align*}where $J_z(x^\star)$ denotes the Jacobian of $z$ evaluated at $x^\star$.
    Then the (nonlinear) system $\hat{h}(y,t)=0$ has a solution $(y^\star,t^\star)\coloneqq(0,0)$ and the
    partial Jacobian $\frac{\partial}{\partial y}\hat{h}(0,0)=J_z(x^\star)J_z(x^\star)^T$ has full 
    rank since $x^\star$ satisfies~\ref{word:BVC_MFCQ}.
    It follows from the IFT that there exists an $\varepsilon>0$ and a continuously differentiable function
    $y:(-\varepsilon,\varepsilon)\to\mathds{R}^q$ such 
    that $y(0)=0$ and $\hat{h}(y(t),t)=0$ for all $t\in(-\varepsilon,\varepsilon)$. 
    Its derivative is given by \begin{align*}
        \frac{d}{dt} y(t) = -\left(\frac{\partial}{\partial y}\hat{h}(y(t),t)\right)^{-1}\frac{\partial}{\partial t}\hat{h}(y(t),t)\;\text{ for all }t\in(-\varepsilon,\varepsilon).
    \end{align*} Moreover, we obtain with the chain rule
    \begin{align*}
    \frac{d}{dt} y(0) = -\left(\frac{\partial}{\partial y}\hat{h}(y(0),0)\right)^{-1}\frac{\partial}{\partial t}\hat{h}(y(0),0) 
    = -\left(\frac{\partial}{\partial y}\hat{h}(y(0),0)\right)^{-1} J_z(x^\star) d =0,
    \end{align*}
    since $J_z(x^\star) d=0$ from~\ref{word:BVC_MFCQ}. 
    We set \begin{align*}
        x(t)\coloneqq x^\star + td + J_z(x^\star)^Ty(t),
    \end{align*} which is by definition continuously differentiable on the interval $(-\varepsilon,\varepsilon)$. 
    Furthermore it holds that
    \begin{align*}
        x(0) &= x^\star + 0\cdot d + J_z(x^\star)^Ty(0) = x^\star + J_z(x^\star)^T\cdot 0=x^\star \text{, and}\\
         \frac{d}{dt}x(0)&= d+ J_z(x^\star)^T  \frac{d}{dt}y(0) = d + J_z(x^\star)^T\cdot 0 = d.
    \end{align*}

       H_{i}(x(t)) &=0 \text{ for all }i\in\mathcal{I}_{(0,-)}^H(x^\star)\cup\mathcal{B}^H_{0} \text{ and } t\in (-\varepsilon,\varepsilon).
      \end{aligned}
    \end{equation} Furthermore, by continuity it holds that $H_{i}(x(t))\geq0$ 
    for all $i \in \mathcal{I}_{(+,0)}^H(x^\star) \cup \mathcal{I}_{(+,+)}^H(x^\star)$ 
    and sufficiently small $t>0$.

    It follows that $\frac{d}{dt} \phi(t) = \nabla_xH_{i}(x(t))^T \frac{d}{dt} x(t)$ and so

     $H_{i}(x(t))\geq0$ for $i\in\mathcal{N}_H$ and sufficiently small $t\geq0$. 
     Analogously we obtain $h_l(x(t))\geq0$ for all $l\in\mathcal{N}_h$ and $G_{i,j}(x(t))\geq0$ for $(i,j)\in\mathcal{I}_{(+,0)}(x^\star)\cup\mathcal{B}_{+}$.

        \end{align*}

        for $\delta>0$.

        conditions for sufficiently small $\delta$.

        such that $x_{\delta}(0)=x^\star$, $\frac{d}{dt} x_{\delta}(0)=d_{\delta}$
        and $x_{\delta}(t)$ is feasible for all $t\geq0$ sufficiently small. We take a sequence $\{t^{(k)}\}$ with $t^{(k)}$ sufficiently small,
        $t^{(k)}\to0$ for $k\to\infty$ and define $x^{(k)}\coloneqq x_{\delta}(t^{(k)})$.

        Since the cone $\mathcal{T}(X,x^\star)$ is closed we can let $\delta \to 0$ and get $ d_{\delta}\to d\in\mathcal{T}(X,x^\star)$.

    \item The system $B^T\xi =c$ has a solution $\xi\geq0$.
    \item $c^Td\geq0$ holds for all elements in the set $\{d\in\mathds{R}^n\,\mid\,Bd\geq0\}$.
\end{itemize}

            0, &\hspace{-9em} \text{ if }(i,j)\in\mathcal{I}_{(0,0)},\\
           H_{i}(x^\star)\nabla G_{i,j}(x^\star), &\hspace{-9em}\text{ if }(i,j)\in\mathcal{I}_{(+,0)}, \\
            \nabla H_{i}(x^\star)G_{i,j}(x^\star), &\hspace{-9em}\text{ if }(i,j)\in\mathcal{I}_{(0,+)}\cup\mathcal{I}_{(0,-)}, \\
            \nabla H_{i}(x^\star)G_{i,j}(x^\star) +H_{i}(x^\star)\nabla G_{i,j}(x^\star), &\hspace{-0.75em}\text{ if }(i,j)\in\mathcal{I}_{(+,+)}.\\
        \end{cases}
    \end{align*}
    We introduce the multipliers \begin{align}\label{vanishing: construction-lagrange-multiplier}
        \mu^H_i &= \lambda^H_i + \sum_{j\in\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star) & \text{and} && \mu^G_{ij} =\lambda^{GH}_{ij}H_{i}(x^\star).
    \end{align}
    We now verify the conditions~\eqref{vanishing: kkt-lagrange-equation} and~\eqref{vanishing: kkt-lagrange-multiplier}.
    \begin{align*}
    &\nabla_xL(x,\mu^H,\mu^G)\\
        &= \nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\mu^H_i\nabla H_{i}(x^\star)-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \mu^{g}_{ij}\nabla G_{i,j}(x^\star)\\
        &= \nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\big[\lambda^H_i+\sum_{j\in\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star)\big]\nabla H_{i}(x^\star)\\
        &\quad-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \lambda^{GH}_{ij}\big(H_{i}(x^\star)\nabla G_{i,j}(x^\star)\big)\\
        &= \nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\lambda^H_i\nabla H_{i}(x^\star)\\
        &\quad - \sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star)\nabla H_{i}(x^\star) -\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \lambda^{GH}_{ij}\big(H_{i}(x^\star)\nabla G_{i,j}(x^\star)\big)\\
        &= \nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\lambda^H_i\nabla H_{i}(x^\star)-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \lambda^{GH}_{ij}\nabla \big(H_{i}(x^\star)G_{i,j}(x^\star)\big) \\&= 0.
    \end{align*}
    By construction from~\eqref{vanishing: construction-lagrange-multiplier} it follows that \begin{align*}
        \mu^H_i &=\lambda^H_i + \sum_{j\in\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star) = \lambda^H_i, \text{ since } \lambda^{GH}_{ij}G_{i,j}(x^\star)=0,\\
        \mu^G_{ij} &= \lambda^{GH}_{ij}H_{i}(x^\star) \begin{cases}
            \geq 0 &\text{ if } (i,j)\in\mathcal{I}_{(+,0)} \\
            =0 & \text{otherwise}
        \end{cases}
    \end{align*}
    Conversely, since~\ref{word:GCQ} holds we know that the local minimum $x^\star$ satisfies~\eqref{vanishing: kkt-lagrange-equation} 
    and~\eqref{vanishing: kkt-lagrange-multiplier}
    with the Lagrange multipliers $\mu^H\in\mathds{R}^{n_H}$ and $\mu^G\in\mathds{R}^{n_G\cdot n_H}$.
    We set \begin{align*}
        \lambda^{GH}_{ij} &= \frac{\mu^G_{ij}}{H_{i}(x^\star)} &&\text{if }(i,j)\in\mathcal{I}_{(+,0)}, \\
        \lambda^{GH}_{ij} &= 0 &&\text{if }(i,j)\in\mathcal{I}_{(+,+)}, \\
        \lambda^{GH}_{ij} &\geq \max\bigg\{0,\frac{\mu^H_i}{G_{i,j}(x^\star)}\bigg\} &&\text{if }(i,j)\in\mathcal{I}_{(0,-)}, \\
        \lambda^{GH}_{ij} &\in\bigg[0, \frac{\mu^H_i}{G_{i,j}(x^\star)}\bigg]\neq\emptyset &&\text{if }(i,j)\in\mathcal{I}_{(0,+)}, \\
        \lambda^{GH}_{ij} &\geq 0  &&\text{if }(i,j)\in\mathcal{I}_{(0,0)}, \\
        &\text{and} &&\\
        \lambda^H_i &= \mu^H_i - \sum_{j\in\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star). &&
    \end{align*}
For~\eqref{vanishing: kkt-lagrange-equation-classic} if follows that
\begin{align*}
    &\vspace{-1.5em}\nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\lambda^H_i\nabla H_{i}(x^\star)-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \lambda^{GH}_{ij}\nabla \big(H_{i}(x^\star)G_{i,j}(x^\star)\big) \\
    &=\nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\big[\mu^H_i - \sum_{j\in\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star)\big]\nabla H_{i}(x^\star)\\
    &\quad-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \lambda^{GH}_{ij}\nabla \big(H_{i}(x^\star)G_{i,j}(x^\star)\big) \\
    &=\nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\mu^H_i\nabla H_{i}(x^\star) 
     +\sum_{i\in\mathcal{N}_H}\sum_{j\in\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star)\nabla H_{i}(x^\star)\\&\quad-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G}\lambda^{GH}_{ij}G_{i,j}(x^\star)\nabla H_{i}(x^\star)-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \lambda^{GH}_{ij}H_{i}(x^\star)\nabla G_{i,j}(x^\star)\\
    &=\nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\mu^H_i\nabla H_{i}(x^\star) -\sum_{(i,j)\in\mathcal{I}_{(+,0)}} \frac{\mu^G_{ij}}{H_{i}(x^\star)}\cdotH_{i}(x^\star)\nabla G_{i,j}(x^\star)\\
    &=\nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\mu^H_i\nabla H_{i}(x^\star)-\sum_{(i,j)\in\mathcal{I}_{(+,0)}} \mu^G_{ij}\nabla G_{i,j}(x^\star)\\
    &=\nabla f(x^\star) -\sum_{i\in\mathcal{N}_H}\mu^H_i\nabla H_{i}(x^\star)-\sum_{(i,j)\in\mathcal{N}_H\times\mathcal{N}_G} \mu^G_{ij}\nabla G_{i,j}(x^\star)\\
    &=0.
\end{align*}
By construction it holds that $\lambda^{GH}_{ij}\geq0$ and $\lambda_i\geq0$ for all $(i,j)\in\mathcal{N}_H\times\mathcal{N}_G$. The equations in~\eqref{vanishing: kkt-lagrange-multiplier-classic} are therefore satisfied.
