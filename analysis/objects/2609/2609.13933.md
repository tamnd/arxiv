			\mathcal{E}_{\Omega}(w_-,w_-)

			&\geq
			\sum_{x\in\Omega}\left(\overline H_0(x)+\mu H_0(x)
			\right) w_-^2(x),
		\end{align*}

		\[
		\overline{H}_0(x)+\mu H_0(x)
		=
		\overline{H}_0(x)-H_0(x)
		+
		(\mu+1)H_0(x) \geq 0.
		\]
		Consequently,
		\[
		0\geq
		\sum_{x\in\Omega}
		\left(
		\overline{H}_0(x)+\mu H_0(x)
		\right)
		w_-^2(x)
		\geq0 ,
		\]
		which implies $w_-(x)=0$ in $\Omega$.
		Hence,
		\[
		w\geq0
		\qquad\text{in }\Omega .
		\]

		$x_0\in\Omega$ is such that $w(x_0)=0$, then, together with $w\geq0$ in $\Omega\cup \partial \Omega$,

		=
		\sum_{y\sim x_0}
		-w(y) \leq 0.
		\]

		\[
		w(y)=0,
		\qquad \forall y\sim x_0 .
		\]
		Repeating the argument along the connected graph $\Omega$, we obtain $w\equiv0$
		in $\Omega$.

			{\cal J}_{\mu}(v) 
			& \geq \sum_{x\in \mathbb{Z}^d}  \big( \overline{H}_0(x)+ \mu H_{0}(x) \big) v^2(x)-v(\xi) \\

			\frac{d}{dt}
