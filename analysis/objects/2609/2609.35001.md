<!-- rmk-1: body_md -->
Here and henceforth, we define the adjoint operators with respect to the scalar product in $L^2(0,T;X)$.  We remark that we do not consider 
adjoint operators with duality product $_{Y^*}\langle\cdot, \, \cdot\rangle_{Y}$ where $X$ is a pivot space and $Y \hookrightarrow X \hookrightarrow Y^*$ is a prescribed Gel'fand triple (see e.g., Brezis \cite{Bre}, Section 6 of Chapter 1 in Yagi \cite{Ya}). Through such a duality product, in Yamamoto \cite{Y22}, we can 
define $(\partial_t^{\alpha})^*$ in Sobolev spaces of negative orders, but in this article we do not consider adjoint operators in this way.
<!-- end rmk-1 -->
<!-- defn-1: body_md -->
\label{defws}
We call $u=u(t)$ a weak solution to \eqref{(2.2)} if $u\in L^2(0,T;X)$ 
satisfies
\begin{equation}\label{(2.3)}
(u-a,\, (\partial_t^{\alpha})^*\varphi)_{L^2(0,T;X)}
= (u,\, \mathcal{A}^*\varphi)_{L^2(0,T;X)}
\end{equation}
for all $\varphi \in \mathcal{D}((\partial_t^{\alpha})^*) \cap \mathcal{D}(\mathcal{A}^*)$.
<!-- end defn-1 -->
<!-- defn-2: body_md -->
\label{defss}
We call $u=u(t)$ a strong solution to \eqref{(2.2)} if
\begin{equation*}
u\in L^2(0,T;\mathcal{D}(A)), \qquad u-a\in H_\alpha(0,T;X),
\end{equation*}
and \eqref{(2.2)} is satisfied in $L^2(0,T;X)$.
<!-- end defn-2 -->
<!-- thm-1: body_md -->
\label{thm1}
We assume that $A$ is a densely defined closed operator. In order that for each $a \in X$, there exists a unique weak solution to \eqref{(2.2)} satisfying
$u \in C([0,T];X)$ and  
\begin{equation}\label{(2.4)}
\Vert u(t)\Vert_X \le \Vert a\Vert_X \quad \text{ for all } t \ge 0,
\end{equation}
Condition $(\mathcal{M})$ is necessary and sufficient.
<!-- end thm-1 -->
<!-- rmk-2: body_md -->
This theorem is a generalization to $0<\alpha<1$ of the classical Hille-Yosida theorem for $\alpha=1,$ which asserts the equivalence of Condition $(\mathcal{M})$ and the well-posedness of $u'(t) = Au$ with $u(0) = a\in X,$ where $A$ generates a contraction C$_0$ semigroup (e.g., Pazy \cite{Pa} and Tanabe \cite{Ta}). We refer to Da Prato and Iannelli \cite{Da80} and Sforza \cite{Sf86} 
for some related results for integro-differential equations.
<!-- end rmk-2 -->
<!-- thm-2: body_md -->
\label{thmcont}
We assume \eqref{(1.2)}.
If $u\in L^2(0,T;X)$ is a weak solution of \eqref{(2.2)}
for $a \in X$, then
$$
u\in C([0,T];X), \qquad u(0)=a.
$$
<!-- end thm-2 -->
<!-- thm-3: body_md -->
\label{cor1.4}
We assume that there exists a unique weak solution $u$ to $\partial_t^{\alpha} (u-a) = Au$ in $X$ for each $a\in X$.
\begin{itemize}
    \item[(i)] For $a \in \mathcal{D}(A)$, the weak solution $u$ satisfies $u-a \in H_\alpha(0,T;X)$, $u\in L^2(0,T;\mathcal{D}(A))$ and
$\partial_t^{\alpha} (u(t)-a) = Au(t)$ in $X$ for almost all $t\in (0,T)$.
    \item[(ii)] Let $a \in \mathcal{D}(A^m)$ with $m\in \mathbb{N}$. Then the weak solution $u$ satisfies $A^mu \in L^2(0,T;X)$. 
\end{itemize}
<!-- end thm-3 -->
<!-- cor-1: body_md -->
\label{cor1}
Let $a \in \mathcal{D}(A)$. Then there exists a unique strong solution to \eqref{(2.2)} satisfying \eqref{(2.4)} and 
\begin{equation}\label{eq*}
u, \, Au \in C([0,T];X), \quad
\Vert Au(t)\Vert_X \le \Vert Aa\Vert_X \quad \mbox{for all $t \in [0,T]$.}
\end{equation}
<!-- end cor-1 -->
<!-- thm-4: body_md -->
\label{thmpert}
Let us assume \eqref{(1.2)} and let $a \in X$.  Then there exists a unique weak solution 
$u\in L^2(0,T;X)$ to \eqref{(pert)} such that

\label{llem3.1}
We assume \eqref{(1.2)}. Let $u\in L^2(0,T;X)$ be a weak solution to $\partial_t^{\alpha} u = Au$ in $X$

\\
First, we assume that $u \in L^2(0,T;\mathcal{D}(A))\cap \mathcal{D}(\partial_t^{\alpha})$ is a weak solution to $\partial_t^{\alpha} u = Au$. Then, we have $(\partial_t^{\alpha} u, u)_{L^2(0,T;X)} \ge \frac{T^{-\alpha}}{2\Gamma(1-\alpha)}\Vert u\Vert^2_{L^2(0,T;X)}$ (e.g., Theorem 3.3 (ii) in \cite{KRY}).
Since $(Au(t),\, u(t))_X \le 0$ for almost all $t \in (0,T)$ by Lemma \ref{lem9.1}, we obtain $0 \ge \frac{T^{-\alpha}}{2\Gamma(1-\alpha)}\Vert u\Vert^2_{L^2(0,T;X)},$ and so $u=0$ in $L^2(0,T;X)$. Thus the uniqueness is proved within $L^2(0,T;\mathcal{D}(A)) \cap \mathcal{D}(\partial_t^{\alpha})$.
 
Next, let $w\in L^2(0,T;X)$ satisfy 
\begin{equation}\label{(4.1)}
(w, \, (\partial_t^\alpha)^* \varphi)_{L^{2}(0,T;X)}
= (w, \mathcal{A}^* \varphi)_{L^{2}(0,T;X)}
\end{equation}
for all $\varphi \in \mathcal{D}((\partial_t^\alpha)^*) \cap L^2(0,T;\mathcal{D}(A^*))$. We need to prove that $w=0$.

Fix $\lambda>0$ and define $z:=(\lambda -A)^{-1} w$. Then,
\begin{equation}\label{(4.2)}
z\in L^2(0,T;\mathcal{D}(A)), \quad Az=\lambda z-w \quad\text{in }
L^2(0,T;X).
\end{equation}
We will show that
\begin{equation}\label{(4.3)}
z\in \mathcal{D}(\partial_t^\alpha) \quad\text{and}\quad 
\partial_t^\alpha z=Az.
\end{equation}
Let $\psi\in \mathcal{D}((\partial_t^\alpha)^*)$.
Since $\mathcal{D}((\partial_t^\alpha)^*) = J_{\alpha}L^2(0,T;X)$ by 
Lemma \ref{lem3.2} (ii), 
there exists $g\in L^2(0,T;X)$ such that $\psi=J_{\alpha}g$.
Because $(\lambda-A^*)^{-1}$ is bounded on $X$ and acts only in $X$, 
it commutes with the integral defining $J_{\alpha}$. Hence,
$$
(\lambda-A^*)^{-1}\psi = (\lambda-A^*)^{-1}J_{\alpha}g = J_{\alpha}
(\lambda-A^*)^{-1}g.
$$
Consequently, $(\lambda-A^*)^{-1}\psi \in \mathcal{D}((\partial_t^\alpha)^*)$ and
\begin{equation}\label{(4.4)}
(\partial_t^\alpha)^*(\lambda-A^*)^{-1}\psi 
= (\lambda-A^*)^{-1}(\partial_t^\alpha)^*\psi.
\end{equation}
Moreover, $(\lambda-A^*)^{-1}\psi\in L^2(0,T;\mathcal{D}(A^*)),$ so that 
$(\lambda-A^*)^{-1}\psi$ is an admissible test function in \eqref{(4.1)}.

Using \eqref{(4.1)} and \eqref{(4.4)}, we obtain
\begin{align*}
& (z,\, (\partial_t^\alpha)^*\psi)_{L^2(0,T;X)} 
= ((\lambda-A)^{-1}w,\, (\partial_t^\alpha)^*\psi)_{L^2(0,T;X)} \\
= & (w,\, (\lambda-A^*)^{-1}(\partial_t^\alpha)^*\psi)_{L^2(0,T;X)} 
  = (w,\, (\partial_t^\alpha)^* (\lambda-A^*)^{-1}\psi)
_{L^2(0,T;X)}                                     \\ 
= & (w,\, A^*(\lambda-A^*)^{-1}\psi)_{L^2(0,T;X)}.
\end{align*}
For the last equality, we used \eqref{(4.1)}, that is, the fact that 
$w$ is a weak solution.
Hence, since $((\lambda-A^*)^{-1})^* = (\lambda - A)^{-1}$, we have 
\begin{align*}
& (z,\, (\partial_t^\alpha)^*\psi)_{L^2(0,T;X)} 
= (w,\, \lambda(\lambda-A^*)^{-1}\psi - \psi)_{L^2(0,T;X)} \\
= & (\lambda (\lambda - A)^{-1}w - w,\, \psi)_{L^2(0,T;X)} 
= (\lambda z-w,\, \psi)_{L^2(0,T;X)} = (Az,\, \psi)_{L^2(0,T;X)}.
\end{align*}
That is, we obtain
$$
(z, \, (\partial_t^{\alpha})^*\psi)_{L^2(0,T;X)} = (Az,\, \psi)_{L^2(0,T;X)}
\quad \mbox{for all $\psi \in \mathcal{D}((\partial_t^{\alpha})^*)$}.
$$
Thus, by the definition of the adjoint operator, we have shown that $z\in \mathcal{D}(((\partial_t^{\alpha})^*)^*)$ and $((\partial_t^{\alpha})^*)^*z=Az$. Since $\partial_t^{\alpha}$ is densely defined and closed, $((\partial_t^{\alpha})^*)^*=\partial_t^{\alpha}$. Therefore, we have proved \eqref{(4.3)}.
By the uniqueness of the strong solution, which we already proved, 
we obtain $z=0$. Since $(\lambda - A)^{-1}$ is injective, we reach $w=0$. 
This proves the uniqueness of the weak solution.
$\blacksquare$

To construct $u_\lambda,$ we write \eqref{(5.1)} as
\begin{align*}
& u_{\lambda}(t) = a + \int^t_0 \frac{1}{\Gamma(\alpha)}
(t-s)^{\alpha-1} A_{\lambda} u_{\lambda}(s) ds   \\
&= a + \int^t_0 \frac{1}{\Gamma(\alpha)}
s^{\alpha-1} A_{\lambda} u_{\lambda}(t-s) ds, \quad 0<t<T.
\end{align*}
Henceforth, we denote
$$
u_{\lambda}'(t) := \frac{du_{\lambda}}{dt}(t), \quad \mbox{etc.}
$$
Formally, we differentiate both sides with respect to 
$t$, so that we can obtain a candidate equation for 
$u_{\lambda}'$:
$$
u_{\lambda}'(t) = \frac{1}{\Gamma(\alpha)}t^{\alpha-1} A_{\lambda} a
+ \int^t_0 \frac{1}{\Gamma(\alpha)}
  s^{\alpha-1} A_{\lambda} u_{\lambda}'(t-s) ds.
$$
Thus, to prove the existence of $u_{\lambda}' \in L^1(0,T;X)$,
we introduce the following iteration:
$$
\left\{ \begin{array}{rl}
& w_0(t) := 0, \quad w_1(t) := \frac{1}{\Gamma(\alpha)}t^{\alpha-1} A_{\lambda} a, \\
& w_{n+1}(t) = \frac{1}{\Gamma(\alpha)}t^{\alpha-1} A_{\lambda} a
+ \int^t_0 \frac{1}{\Gamma(\alpha)} s^{\alpha-1} A_{\lambda} w_{n}(t-s) ds,
\quad n\in \mathbb{N}.
\end{array}\right.
$$
We note that $A_{\lambda}: X \longrightarrow X$ is a bounded operator and 
$\Vert A_{\lambda}\Vert_{X\to X} \le 2\lambda$.

We can estimate
\begin{align*}
& \Vert (w_{n+2} - w_{n+1})(t)\Vert 
\le \frac{1}{\Gamma(\alpha)}\int^t_0 s^{\alpha-1} \Vert A_{\lambda}\Vert_{X\to X}
\Vert (w_{n+1} - w_n)(t-s)\Vert ds \\
\le & C\int^t_0 (t-s)^{\alpha-1} 
\Vert (w_{n+1} - w_n)(s)\Vert ds.
\end{align*}
Here $C>0$ depends on $\lambda$.

First,
$$
\Vert (w_1-w_0)(s)\Vert_X = \Vert w_1(s)\Vert_X
= \left\Vert \frac{1}{\Gamma(\alpha)}(A_{\lambda} a)s^{\alpha-1} 
\right\Vert_X \le Ms^{\alpha-1},
$$ 
where we set $M:=\frac{1}{\Gamma(\alpha)}\Vert A_{\lambda} a\Vert_X$.
Next,
$$
\Vert (w_2-w_1)(t)\Vert_X 
\le CM\int^t_0 (t-s)^{\alpha-1}s^{\alpha-1} ds
= CM\frac{\Gamma(\alpha)^2}{\Gamma(2\alpha)}t^{2\alpha-1},
$$
and
$$
\Vert (w_3-w_2)(t)\Vert_X 
\le C^2M\frac{\Gamma(\alpha)^2}{\Gamma(2\alpha)}
\int^t_0 (t-s)^{\alpha-1}s^{2\alpha-1} ds
= \frac{C^2M\Gamma(\alpha)^3}{\Gamma(3\alpha)}t^{3\alpha-1}
$$
for $0<t<T$.
Continuing the estimation, we obtain
$$
\Vert (w_{n+1} - w_n)(t)\Vert_X 
\le \frac{M\Gamma(\alpha)(C\Gamma(\alpha))^n}
{\Gamma((n+1)\alpha)}t^{(n+1)\alpha-1},
$$
and so 
$$
\Vert w_{n+1} - w_n\Vert_{L^1(0,T;X)} 
\le \frac{M\Gamma(\alpha)(C\Gamma(\alpha))^nT^{(n+1)\alpha}}
{\Gamma((n+1)\alpha+1)}
\le C_0\frac{C_1^n}{\Gamma((n+1)\alpha+1)}
$$
for all $n\in \mathbb{N}$, where $C_0, C_1 > 0$ are constants independent of
$n\in \mathbb{N}$.
Hence, for any $N', N\in \mathbb{N}$ with $N'>N$, we have 
\begin{align*}
& \Vert w_{N'}- w_N\Vert_{L^1(0,T;X)} 
= \Vert (w_{N'} - w_{N'-1}) + \cdots + (w_{N+1}-w_N)\Vert_{L^1(0,T;X)} \\
\le & \Vert w_{N'} - w_{N'-1}\Vert_{L^1(0,T;X)} 
  + \cdots + \Vert w_{N+1} - w_N\Vert_{L^1(0,T;X)} \\
\le& \sum_{k=N}^{N'-1} \frac{C_0C_1^k}{\Gamma((k+1)\alpha+1)} .
\end{align*}
Stirling's formula yields that 
$$
\lim_{N',N\to\infty} \Vert w_{N'} - w_N\Vert_{L^1(0,T;X)} = 0.
$$
Therefore, there exists $w \in L^1(0,T;X)$ such that
$\lim_{N\to\infty} w_N = w=:u'_\lambda$ in $L^1(0,T;X)$. This means that $u_{\lambda} \in W^{1,1}(0,T;X)$. The uniqueness of $u_\lambda$ follows from the coercivity Lemma \ref{lem5.2}.

Moreover, the Sobolev embedding implies that 
$\Vert u_{\lambda}\Vert_{L^{\infty}(0,T;X)} < \infty$.
Consequently, 
\begin{align*}
& \left\Vert \int^t_0 \frac{1}{\Gamma(\alpha)}
(t-s)^{\alpha-1}A_{\lambda} u_{\lambda}(s) ds\right\Vert_X  
\le C\int^t_0 (t-s)^{\alpha-1}\Vert A_{\lambda}\Vert_{X\to X}
\Vert u_{\lambda}(s)\Vert_X ds\\  
\le & C\Vert A_{\lambda}\Vert_{X\to X} 
\Vert u_{\lambda}\Vert_{L^{\infty}(0,T;X)} \int^t_0 (t-s)^{\alpha-1}ds
\, \longrightarrow 0
\end{align*}
as $t \to 0$, which implies $u_{\lambda}(0) = a$.
$\blacksquare$
