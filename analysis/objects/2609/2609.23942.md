<!-- definition-1: body_md -->
\label{def:lee-yang}
The class of \emph{fermionic Lee-Yang Hamiltonians} is defined as
\begin{subequations}
    \label{eq:free_fermion_and_interactions}
    \begin{gather}
        H = H_0 + V \\[6pt]
        \text{where} \quad H_0 := \sum_{i,j} A_{ij}\, a^{\dagger}_i a_j \,+\, \frac{1}{2}\sum_{i,j}\,\left(\,B_{ij}\, a^{\dagger}_i a^{\dagger}_j + B_{ij}^{*}\, a_j a_i\,\right),\label{eq:free_fermion}\\
        V := - \sum_{i\,<\,j}\, |U_{ij}| \left(n_i -\frac{1}{2}\right)\left(n_j -\frac{1}{2}\right)
    \end{gather}
\end{subequations}
with a positive-semidefinite (PSD) hopping matrix $A = A^{\dagger} \succeq 0$, an arbitrary antisymmetric pairing matrix $B = -B^T$, and attractive interactions with arbitrary strength $|U_{ij}|$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:Lee-Yang_theorem}
    For any Hamiltonian $H$ from the class in \cref{def:lee-yang}, in the complex plane of $h$, all the zeros of the partition function $\mathcal{Z}(h)$ satisfy $ \mathrm{Re}(h) \in [-\lambda_{\max}(A), 0].$
    In particular, when $A = 0$, they all lie on the imaginary axis.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:spectral_gap}
    For any Hamiltonian $H$ from the class in \cref{def:lee-yang}, $H + h \sum_i n_i$ with $h>0$ has a spectral gap of at least $h/8$ above the ground state. Consequently, there is an efficient quantum algorithm, running in time $\mathrm{poly}(N, \|H\|_{\mathrm{op}}, 1/\epsilon)$, to estimate the ground-state energy of $H$ to within any $\epsilon > 0$ additive precision, where $N$ is the number of fermion modes.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:zero-free_free_fermion}
    In the open unit polydisc, i.e., $|u_{i,m}| < 1$ for all $i$ and $m$, the Trotterized partition function $\mathcal{Z}_{\epsilon}(H_0, \bm{u})$, defined in \cref{eq:free_fermion_partition_function}, is nonzero for any free-fermion Hamiltonian $H_0$ defined in \cref{eq:free_fermion_and_interactions}.
<!-- end theorem-3 -->
<!-- proof-1: body_md -->
Let 
    \begin{align}
        Q :=
        \begin{pmatrix}
            A & B \\
            B^{\dagger} & -A^*
        \end{pmatrix}
        \quad \text{such that} \quad Q = Q^{\dagger},
    \end{align}
    and rewrite the free-fermion Hamiltonian from \cref{eq:free_fermion} as
    \begin{align}
        H_0 = \frac{1}{2}\bm{\psi}^{\dagger} Q \bm{\psi} + \frac{1}{2} \mathrm{Tr}(A) \quad \text{where} \quad \bm{\psi}^\dagger := (a_0^\dagger,\dots,a_{N-1}^\dagger,a_0,\dots,a_{N-1}).
    \end{align}
    Let $W := e^{-\epsilon Q}$, $E_m \,:=\, \operatorname{diag}\big(\{u_{i,m}\}_{i}, \bm{1}_N\big)$, $F_m \,:=\, \operatorname{diag}\big(\bm{1}_N, \{u_{i,m}\}_{i}\big)$, and $G_m := E_m F^{-1}_m$. Rewrite the square of the partition function as
    \begin{align}
        \mathcal{Z}_\epsilon(H_0,\bm{u})^2 &\,=\, e^{-\beta \,\mathrm{Tr}(A)}\left(\prod_{i,m} u_{i,m} \right)\mathrm{Tr}\left[\rule{0cm}{0.85cm}\,\prod_{m}\, \left[e^{- \frac{1}{2}\epsilon\,\bm{\psi}^{\dagger} Q \bm{\psi}}\, e^{\frac{1}{2}\bm{\psi}^{\dagger} \log(G_m) \bm{\psi}} \right]\,\right]^2 \\[6pt]
        &\,=\, e^{-\beta \,\mathrm{Tr}(A)}\left(\prod_{m} \det(F_m) \right) \det\left[\mathbb{I} + \prod_{m}W G_{m}\right].
    \end{align} 
    The determinant of $\mathbb{I}$ plus a product of matrices can be written as the determinant of a single block matrix\footnote{This follows from Schur complement: the top-left $(M-1)\times(M-1)$ block of $\mathfrak{B}$ is block-triangular with unit diagonal, and eliminating it leaves exactly $\det[\mathbb{I} + \prod_{m \in \mathbb{Z}_M}WG_m]$.}
    \begin{align}
        \det\left[\mathbb{I} + \prod_{m}W G_m\right] \,=\, \det \mathfrak{B} \quad \text{where} \quad \mathfrak{B} \,:=\,
        \begin{pmatrix}
            \mathbb{I}      & -W G_1 &        &       \\
                    & \mathbb{I}     & \ddots &      \\
                    &        & \ddots & -W G_{M-1}     \\
            W G_0   &        &        & \mathbb{I}
        \end{pmatrix},
    \end{align}
    with all the unspecified blocks equal to zero. Multiplying with $\prod_{m} \det(F_m)$, we get
    \begin{align}
        \label{eq:pencil_representation}
        \mathcal{Z}_\epsilon(H_0,\bm{u})^2  = e^{-\beta \,\mathrm{Tr}(A)}\det \mathfrak{D} \quad \operatorname{where} \quad 
        \mathfrak{D} = \begin{pmatrix}
            F_0     & -W E_1 &        &       \\
                    & F_1    & \ddots &      \\
                    &        & \ddots & -W E_{M-1}     \\
            W E_0   &        &        & F_{M-1}
        \end{pmatrix}.
    \end{align}
    Although the derivation above assumes $u_{i,m}\neq 0$ for all $i,m$,
    Eq.~\eqref{eq:pencil_representation} remains valid when some variables vanish,
    since both sides are polynomials in $\bm{u}$.

    We now show that the matrix $\mathfrak{D}(\bm u)$ is non-singular everywhere in the open unit polydisc. Suppose $\mathfrak{D}(\bm u)$ is singular and that $\phi = (\phi_0, \dots, \phi_{M-1})$ is a corresponding null vector. Then the structure of the matrix implies
    \vspace{-0.3cm}
    \begin{align}
        \label{eq:balance_equation}
        F_{m}\phi_{m} \,=\, \pm\, W E_{m+1}\, \phi_{m+1} \quad \text{for all }\, m
    \end{align}
    where the index $m$ is taken mod $M$ and the minus sign occurs only at $m = M-1$. Define a diagonal matrix $\Gamma := \operatorname{diag}( +\bm{1}_N,\, -\bm{1}_N)$, which satisfies $\{\Gamma, Q\} = 2\operatorname{diag}(A, A^*) \succeq 0$.
    Since $\partial_{\epsilon} W\Gamma W = -W\{\Gamma,Q\}W \preceq 0$ for all $\epsilon \in \mathbb{R}$, this implies $W \Gamma W \preceq \Gamma$. With $\Gamma$, define a quadratic form
    \begin{align}
        \braket{\psi, \psi}_{\Gamma} \,:=\, \braket{\psi, \Gamma \psi} \,=\, \sum_{j\in \mathbb{Z}_N} |\psi_j|^2 - \sum_{j\in \mathbb{Z}_{2N}\backslash \mathbb{Z}_N} |\psi_j|^2.
    \end{align}
    The quadratic form is monotonically non-increasing under the action of $W$ since
    \begin{align}
        \braket{W\psi, W\psi}_{\Gamma} \,=\,  \braket{W\psi, \Gamma W\psi} \,=\, \braket{\psi, W \Gamma W \psi} \,\leq\, \braket{\psi, \Gamma\psi} = \braket{\psi,\psi}_{\Gamma},
    \end{align} 
    where we used $W^{\dagger} = W$ since $Q$ is Hermitian. Taking the quadratic form of both sides of Eq.~\eqref{eq:balance_equation}, the sign squares away and we get
    \begin{align}
        \sum_{j\in \mathbb{Z}_N} |\phi_{m,j}|^2 \,-\, \sum_{j\in \mathbb{Z}_{2N}\backslash\mathbb{Z}_N} |u_{j-N,m}|^2|\phi_{m,j}|^2 \;\leq\;  \sum_{j\in \mathbb{Z}_N} |u_{j,m+1}|^2|\phi_{m+1,j}|^2 \,-\, \sum_{j\in \mathbb{Z}_{2N}\backslash\mathbb{Z}_N} |\phi_{m+1,j}|^2
    \end{align}
    Putting these together by summing over $m \in \mathbb{Z}_M$ and shifting the summation index $m \to m-1$ on the right-hand side, all the terms combine into
    \begin{align}
        \sum_{m\in \mathbb{Z}_M}\, \sum_{j\in \mathbb{Z}_{2N}} \left(1 - |u_{j (\mathrm{mod}\ N),\ m}|^2\right) |\phi_{m,j}|^2 \;\leq\; 0.
    \end{align}
    In the open unit polydisc, every coefficient $1 - |u_{j,m}|^2$ is strictly positive, which implies $\phi_m = 0$ for all $m$, i.e., $\phi = 0$. Hence $\mathfrak{D}(\bm{u})$ is non-singular, and by Eq.~\eqref{eq:pencil_representation}, $\mathcal{Z}_\epsilon(H_0,\bm{u})^2 = e^{-\beta \,\mathrm{Tr}(A)}\det \mathfrak{D}(\bm{u}) \neq 0$ everywhere in the open unit polydisc.

<!-- lemma-2: body_md -->
\label{thm:stability_of_schur_product}
    For each $X \subseteq \{1,2,\ldots,k\}$, denote $z^{X} = \prod_{i \in X} z_{i}$ where $z_i$ are complex variables. Suppose that the polynomials $a(\bm{z}) = \sum_{X} a_{X} z^{X}$ and $b(\bm{z}) = \sum_{X} b_{X} z^{X}$ are nonzero on the open unit polydisc. Then the Schur product of the polynomials $a(\bm{z})$ and $b(\bm{z})$,
    \begin{align}
        a(\bm{z}) \,\odot\,b(\bm{z}) \,:=\,  \sum_{X} a_{X} b_{X} z^{X}
    \end{align}
    is also nonzero on the open unit polydisc.
<!-- end lemma-2 -->
<!-- theorem-4: body_md -->
\label{thm:unit_disk_zero_free_interacting_fermions}
    In the open unit polydisc, the Trotterized partition function $\mathcal{Z}_\epsilon(H, \bm{u})$, defined in \cref{eq:interacting_fermion_partition_function}, is nonzero for any interacting fermionic Hamiltonian defined in \cref{eq:free_fermion_and_interactions}.
<!-- end theorem-4 -->
<!-- proof-2: body_md -->
Let $X$ be a subset of $\mathbb{Z}_N \times \mathbb{Z}_M$, and denote $u^{X}$ to be the monomial that is the product of all the $u_{i,m}$ with $(i,m) \in X$. Let 
    \begin{align}
        \mathcal{Z}_\epsilon(H_0,\bm{u}) \,=\, \sum_X p_X u^X, \quad \mathcal{Z}_\epsilon(V,\bm{u}) \,=\, \sum_X q_X u^X \quad \text{and} \quad \mathcal{Z}_\epsilon(H,\bm{u}) \,=\, \sum_X r_X u^X.
    \end{align}
    We claim that, for all $X$,
    \vspace{-0.4cm}
    \begin{align}
        r_X = p_X q_X.
    \end{align}
    To see this, expand the diagonal term $\prod_{i}(\mathbb{I}-n_i+u_{i,m}\,n_i)$ as $\sum_{S \subseteq \mathbb{Z}_N}\big(\prod_{i\in S}u_{i,m}\big)P_S$, where $P_S$ projects onto the basis state with the modes in $S$ occupied. 
    Since $V$ is diagonal in this basis, 
    \begin{align}
        e^{-\epsilon V}\prod_{i}(\mathbb{I}-n_i+u_{i,m}\,n_i) = \sum_{S \subseteq \mathbb{Z}_N}\left(\prod_{i\in S}u_{i,m}\right)e^{-\epsilon V (S)}P_S    
    \end{align}
    with $V(S)$ the classical interaction energy of the configuration $S$. 
    Every monomial $u^X$ in $\mathcal{Z}_\epsilon(H,\bm{u})$ therefore carries, relative to $\mathcal{Z}_\epsilon(H_0,\bm{u})$, the additional scalar factor 
    \begin{align}
        \prod_m e^{-\epsilon V(X_m)} \quad \operatorname{where} \quad X_m := \left\{i : (i,m)\in X\right\}
    \end{align}
    and by \cref{eq:classical_partition_function} this factor is exactly $q_X$.
    
    Therefore the polynomial $\mathcal{Z}_\epsilon(H,\bm{u})$ is a Schur product of the polynomials $\mathcal{Z}_\epsilon(H_0,\bm{u})$ and $\mathcal{Z}_\epsilon(V,\bm{u})$,
    \begin{align}
        \mathcal{Z}_\epsilon(H,\bm{u}) = \mathcal{Z}_\epsilon(H_0,\bm{u}) \,\odot\, \mathcal{Z}_\epsilon(V,\bm{u}).
    \end{align}
    By \cref{thm:stability_of_schur_product}, this implies that $\mathcal{Z}_\epsilon(H, \bm{u})$ is nonzero in the open unit polydisc.
