<!-- theorem-1: body_md -->
\label{thm:main}
  For every $n$ unit vectors $v_1, \dots, v_n \in \mathbb{R}^d$ with $n \ge d$, there exist signs $\varepsilon_1, \dots, \varepsilon_n \in \left\{{\pm1}\right\}$ such that
  \[
    \norm*{\varepsilon_1 v_1 + \dots + \varepsilon_n v_n} \ge \sqrt{2n-d}.
  \]
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
\label{prop:equalities}
  Let $n \ge d$ be positive integers, and let
  \[
    \mathbb{R}^d=V_0\oplus V_1\oplus\dots\oplus V_{n-d}
  \]
  be an orthogonal decomposition such that $V_1,\dots,V_{n-d}$ have positive even dimensions, with $V_0$ possibly zero dimensional. If $v_1,\dots,v_n$ consist of an orthonormal basis of $V_0$ and the vertices of a centered regular simplex inscribed in the unit sphere of $V_i$ for each $i \in \left\{{1, \dots, n-d}\right\}$, then
  \[
    \norm*{\varepsilon_1 v_1 + \dots + \varepsilon_n v_n} \le \sqrt{2n-d}, \text{ for every } \varepsilon_1, \dots, \varepsilon_n \in \left\{{\pm1}\right\}.
  \]
<!-- end proposition-1 -->
<!-- theorem-2: body_md -->
\label{thm:equalities}
  Let $v_1, \dots, v_n \in \mathbb{R}^d$ be unit vectors with $n \ge d$. If
  \[
    \norm*{\varepsilon_1 v_1 + \dots + \varepsilon_n v_n} \le \sqrt{2n-d}, \text{ for every } \varepsilon_1, \dots, \varepsilon_n \in \left\{{\pm1}\right\},
  \]
  then, after switching, the vectors form a configuration described in \cref{prop:equalities}.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:bp1}
  For every subspace $K$ of $\mathbb{R}^n$, there exists a random vector $V$ supported on $[-1,1]^n \cap K$ such that $\mathbb{E}[VV^\intercal] \succeq \operatorname{Proj}_K$. \qed
<!-- end theorem-3 -->
<!-- lemma-1: body_md -->
\label{lem:rounding}
  For every hollow matrix $M$ of order $n$, and every vector $v \in [-1,1]^n$, there exists a vector $\varepsilon \in \left\{{\pm 1}\right\}^n$ such that $\varepsilon^\intercal M \varepsilon \ge v^\intercal M v$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $V$ be a random vector supported on $\left\{{\pm 1}\right\}^n$ whose coordinates $V_1, \dots, V_n$ are independent $\pm 1$ random variables such that $\mathbb{E}[V_i] = v_i$ for every $i$. Since $M$ is hollow, we have
  \[
    \mathbb{E}[V^\intercal M V] = v^\intercal M v,
  \]
  which warrants a vector $\varepsilon \in \left\{{\pm 1}\right\}^n$ that satisfies $\varepsilon^\intercal M \varepsilon \ge v^\intercal M v$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:main}
  For every symmetric hollow matrix $M$, there exists a vector $\varepsilon \in \left\{{\pm 1}\right\}^n$ such that $\varepsilon^\intercal M \varepsilon$ is at least the sum of positive eigenvalues of $M$, which, in particular, is at least the multiplicity of $-1$ as an eigenvalue of $M$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let $M = Q + R$ be the decomposition of $M$ into its positive and negative parts, that is,
  \[
    Q = \sum_{i \colon \lambda_i > 0}\lambda_i u_i u_i^\intercal\quad \text{and} \quad R = \sum_{i \colon \lambda_i < 0}\lambda_i u_i u_i^\intercal,
  \]
  where $u_1, \dots, u_n$ form an eigenbasis of $M$, and $\lambda_1, \dots, \lambda_n$ are the corresponding eigenvalues. In particular, $Q$ is positive semidefinite, $\operatorname{im} Q \subseteq \ker R$, and $\operatorname{tr} Q$ is the sum of the positive eigenvalues of $M$.

  Set $K = \operatorname{im} Q$, and let $V$ be the random vector supported on $[-1,1]^n \cap K$ given by \cref{thm:bp1} for $K$. Since $Q$ is positive semidefinite and $\mathbb{E}[VV^\intercal]\succeq\operatorname{Proj}_K$, taking the trace against $Q$ gives\footnote{For positive semidefinite matrices $A$ and $B$, $\operatorname{tr}(AB)=\operatorname{tr}\mleft(A^{1/2}BA^{1/2}\mright)\ge0$.}
  \[
    \mathbb{E}[V^\intercal Q V] = \operatorname{tr}(Q\mathbb{E}[VV^\intercal]) \ge \operatorname{tr}(Q\operatorname{Proj}_K) = \operatorname{tr} Q.
  \]
  Thus some $v \in [-1,1]^n \cap K$ satisfies $v^\intercal Qv\ge\operatorname{tr} Q$. Since $v \in K \subseteq \ker R$, we have
  \[
    v^\intercal M v = v^\intercal Q v \ge \operatorname{tr} Q.
  \]
  According to \cref{lem:rounding}, there exists $\varepsilon \in \left\{{\pm 1}\right\}^n$ such that $\varepsilon^\intercal M \varepsilon \ge \operatorname{tr} Q$. Finally, if $m$ is the multiplicity of $-1$ as an eigenvalue of $M$, then $\operatorname{tr} M=0$ because $M$ is hollow, and hence $\operatorname{tr} Q=-\operatorname{tr} R\ge m$.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
Let $G$ be the Gram matrix of $v_1,\dots,v_n$. The matrix $M=G-I_n$ is hollow, and the multiplicity of $-1$ as an eigenvalue of $M$ is $n-\operatorname{rank} G$. By \cref{lem:main}, there exists $\varepsilon \in \left\{{\pm 1}\right\}^n$ such that $\varepsilon^\intercal M \varepsilon$ is at least the sum of the positive eigenvalues of $M$, hence at least $n-\operatorname{rank} G\ge n-d$. Therefore,
  \[
    \norm*{\varepsilon_1 v_1 + \dots + \varepsilon_n v_n}^2 = \varepsilon^\intercal G \varepsilon = n + \varepsilon^\intercal M \varepsilon \ge 2n-d. \qedhere
  \]
<!-- end proof-3 -->
<!-- proof-4: body_md -->
Every signed sum of an orthonormal basis of $V_0$ has squared norm $\dim V_0$. By orthogonality of the subspaces, it suffices to show that every signed sum of the vertices $v_0,\dots,v_k$ of a centered regular simplex on the unit sphere of an even-dimensional subspace $V_i$ has squared norm at most $k+2$, where $k$ is the dimension of $V_i$. Indeed, the squared norm of any signed sum in \cref{prop:equalities} would therefore be at most
  \[
    \dim V_0+\sum_{i=1}^{n-d}(\dim V_i+2)=d+2(n-d)=2n-d.
  \]

  Note that distinct vertices satisfy $\ip{v_i,v_j}=-1/k$. Since $k+1$ is odd, any signs $\varepsilon_0,\dots,\varepsilon_k \in \left\{{\pm 1}\right\}$ satisfy $\abs*{\varepsilon_0 + \dots + \varepsilon_k}\ge1$. Hence
  \[
    \norm*{\varepsilon_0v_0 + \dots + \varepsilon_kv_k}^2 = k+1-\frac{1}{k}\sum_{i \ne j}\varepsilon_i\varepsilon_j = \frac{1}{k}\left((k+1)^2-\left(\sum_i \varepsilon_i\right)^2\right) \le k+2. \qedhere
  \]

<!-- lemma-3: body_md -->
\label{lem:extremal}
  Let $Q$ be a positive semidefinite matrix of order $n$, and let $K=\operatorname{im} Q$. If
  \[
    x^\intercal Q x \le \operatorname{tr} Q, \text{ for every } x \in [-1,1]^n \cap K,
  \]
  then there exists a random vector $V$ supported on $\left\{{0,\pm 1}\right\}^n \cap K$ such that
  \[
    \mathbb{E}\left[VV^\intercal\right] = \operatorname{Proj}_K \quad\text{and}\quad
    V^\intercal Q V = \operatorname{tr} Q \quad\text{almost surely}.
  \]
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
According to \cref{thm:bp1}, let $V$ be a random vector on $[-1,1]^n \cap K$ such that $\mathbb{E}[VV^\intercal] \succeq \operatorname{Proj}_K$. Since $V^\intercal QV\le\operatorname{tr} Q$ almost surely, we have
  \[
    \operatorname{tr} Q
    \le \operatorname{tr}(Q\mathbb{E}[VV^\intercal])
    = \mathbb{E}[V^\intercal QV]
    \le \operatorname{tr} Q.
  \]
  Since $V$ is supported on $K$, the operator $\mathbb{E}[VV^\intercal] - \operatorname{Proj}_K$ acts on $K$. It is positive semidefinite, and $Q$ is positive definite on $K$, so equality forces $\mathbb{E}[VV^\intercal] = \operatorname{Proj}_K$.\footnote{If $A$ is positive definite and $B$ is positive semidefinite, then $\operatorname{tr}(AB)=0$ implies that $B$ is zero: the positive semidefinite matrix $A^{1/2}BA^{1/2}$ has trace zero, hence vanishes, and $A^{1/2}$ is invertible.} Furthermore, $V^\intercal Q V = \operatorname{tr} Q$ almost surely.

  It remains to show that $V$ is supported on $\left\{{0,\pm 1}\right\}^n$. Note that every support point of $V$ achieves the maximum $\operatorname{tr} Q$ of $x\mapstox^\intercal Q x$ over $[-1,1]^n \cap K$. Since this quadratic form is strictly convex on $K$, each such maximizer is a vertex of $[-1,1]^n \cap K$. Write $d=\dim K$. A vertex must have at least $d$ coordinates equal to $\pm1$: otherwise, a nonzero direction in $K$ vanishing on those coordinates would allow a perturbation in both directions within $[-1,1]^n \cap K$. However,
  \[
    \mathbb{E}\norm{V}^2  = \mathbb{E}[\operatorname{tr}\mleft(VV^\intercal\mright)] = \operatorname{tr} \mathbb{E}[VV^\intercal] = \operatorname{tr} \operatorname{Proj}_K = d.
  \]
  Hence $V$ has exactly $d$ nonzero coordinates, and they are $\pm1$, almost surely.
<!-- end proof-5 -->
<!-- lemma-4: body_md -->
\label{lem:maximizer}
  For every symmetric hollow matrix $M$ of order $n$, and every maximizer $v$ of $x \mapsto x^\intercal M x$ over $[-1,1]^n$, if $v \in \left\{{0,\pm 1}\right\}^n$, then
  \[
    v_i(Mv)_i \ge \sum_{j \colon v_j = 0}\abs{M_{ij}}, \text{ for every }i,
  \]
  and moreover $(Mv)_j = 0$ for every $j$ with $v_j = 0$.
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
If $v_j=0$, then $v\pm e_j\in[-1,1]^n$, so maximality and $M_{jj}=0$ give 
  \[
    0\ge(v\pm e_j)^\intercal M (v\pm e_j)-v^\intercal M v=\pm2(Mv)_j,
  \]
  which implies $(Mv)_j=0$.

  Let $V$ agree with $v$ at every nonzero coordinate of $v$, and choose its remaining coordinates independently and uniformly at random from $\left\{{\pm1}\right\}$. Since $M$ is hollow, the expectation satisfies $\mathbb{E}[V^\intercal M V]=v^\intercal M v$. Thus every support point of $V$ achieves the maximum $v^\intercal M v$ of $x\mapstox^\intercal M x$ over $[-1,1]^n$. Fix an arbitrary $i \in \left\{{1,\dots, n}\right\}$. We break the rest of the proof into two cases.

  \medskip\noindent\textbf{Case 1: $v_i=0$.} Fix $j$ with $v_j=0$. If $j=i$, then $M_{ij}=0$ because $M$ is hollow. Otherwise, choose a support point $u$ of $V$ with $u_i=u_j=-1$. The four vectors $u$, $u+2e_i$, $u+2e_j$, and $u+2e_i+2e_j$ are all support points and hence attain this maximum. Therefore,
  \[
    0=(u+2e_i+2e_j)^\intercal M (u+2e_i+2e_j)-(u+2e_i)^\intercal M (u+2e_i)-(u+2e_j)^\intercal M (u+2e_j)+u^\intercal M u=8M_{ij}.
  \]
  Thus $M_{ij}=0$ whenever $v_j=0$, and the desired inequality holds with equality.

  \medskip\noindent\textbf{Case 2: $v_i \ne 0$.} Choose a support point $w$ of $V$ so that $v_iM_{ij}w_j=-\abs{M_{ij}}$ whenever $v_j=0$. Since $w$ attains this maximum, $w_i=v_i$, and $M_{ii}=0$, we have
  \[
      0 \le w^\intercal M w-(w-2v_ie_i)^\intercal M (w-2v_ie_i)=4v_i(Mw)_i =4\left(v_i(Mv)_i-\sum_{j\colon v_j=0}\abs{M_{ij}}\right). \qedhere
  \]

<!-- lemma-5: body_md -->
\label{lem:switching}
  Let $M$ be a symmetric hollow matrix whose negative eigenvalues are all equal to $-1$. Suppose there exists a nonnegative vector $q$ such that $\abs{M}q\le q$ and $q_i>0$ whenever the $i$-th row of $M$ is nonzero. Then there exists a diagonal matrix $D$ with $\pm1$ diagonal entries such that $DMD=-\abs{M}$.
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
Using the connected components of the support graph of $M$, we can partition $M$ into a block diagonal matrix, where each diagonal block is either a zero matrix or a nonzero irreducible matrix. By focusing on a diagonal block, we may assume that $M$ itself is nonzero and irreducible, and hence $q$ is positive. Let $y$ be a positive Perron vector of $\abs{M}$, so that $\abs{M}y=\rho(\abs{M})y$, where $\rho(\abs{M})$ is the spectral radius of $\abs{M}$. Note that
  \[
    \rho(\abs{M})q^\intercal y = q^\intercal \abs{M} y \le q^\intercal y.
  \]
  Since $q$ is positive, we have $q^\intercal y > 0$, which implies $\rho(\abs{M}) \le 1$.

\label{lem:rigidity}
  For every symmetric irreducible hollow matrix $M$ of order $n$ with nonpositive entries, if $-1$ is the unique negative eigenvalue of $M$, and $M$ satisfies
  \[
    \varepsilon^\intercal M \varepsilon \le 1, \text{ for every }\varepsilon \in \left\{{\pm 1}\right\}^n,

By the Perron--Frobenius theorem, $1$ is the Perron--Frobenius eigenvalue of $-M$. Let $u$ be the corresponding unit eigenvector with positive coordinates. Set $Q=M+uu^\intercal$ and $K=\operatorname{im} Q$. Since $-1$ is the unique negative eigenvalue of $M$, we know that $Q$ is positive semidefinite and $u\in\ker Q$.   Applying \cref{lem:rounding} to $M$, we know that $x^\intercal M x\le1$ for every $x\in[-1,1]^n$.   \[
    \mathbb{E}[VV^\intercal]=\operatorname{Proj}_K \quad\text{and}\quad V^\intercal Q V=\operatorname{tr} Q=1\quad\text{almost surely}.
  \]

  \begin{claim*}
    We have $\abs{M}\bm{1}=\bm{1}$ and $K=\bm{1}^\perp$. Almost surely, $V$ has a unique zero coordinate $J$, which is uniformly distributed on $\left\{{1,\dots,n}\right\}$, and
    \[
      (MV)_i=V_i\abs{M_{iJ}} \text{ for every }i.
    \]

  \begin{claimproof}[Proof of Claim]
    By \cref{lem:maximizer}, for every $i$,
    \begin{equation} \label{eq:mij-ineq}
      V_i(MV)_i\ge\sum_{j\colon V_j=0}\abs{M_{ij}} \quad\text{almost surely}.
    \end{equation}
    Set $s_j=\sum_i\abs{M_{ij}}$. Summing over $i$ gives
    \begin{equation} \label{eq:s-ineq}
      1=V^\intercal M V\ge\sum_{j\colon V_j=0}s_j \quad\text{almost surely}.
    \end{equation}
    Since $K\subseteq u^\perp$, we have $\operatorname{Proj}_K+uu^\intercal\preceq I_n$. Thus $\mathbb{E}[VV^\intercal]=\operatorname{Proj}_K$ gives for every $j$ that
    \begin{equation} \label{eq:pos-p}
      \Pr(V_j=0)=1-(\operatorname{Proj}_K)_{jj}\ge u_j^2>0.
    \end{equation}
    Consequently $s_j\le1$ for every $j$. But $\abs{M}u=u$, so
    \[
      \sum_j u_js_j=u^\intercal\abs{M}\bm{1}=u^\intercal\bm{1}=\sum_j u_j.
    \]
    Since every $u_j$ is positive, all $s_j=1$. Symmetry gives $\abs{M}\bm{1}=\bm{1}$, and irreducibility gives $u=\bm{1}/\sqrt n$.

    Since $s_j=1$ for every $j$, \eqref{eq:s-ineq} shows that $V$ has at most one zero coordinate almost surely. On the other hand, using \eqref{eq:pos-p}, we obtain
    \[
      \mathbb{E}[\abs{\left\{{j}\colon{V_j=0}\right\}}] = \sum_j \Pr(V_j = 0) = n - \operatorname{tr} \operatorname{Proj}_K \ge \sum_j u_j^2 = 1.
    \]

    \[
      \Pr(J=j)=1-(\operatorname{Proj}_K)_{jj}=u_j^2=\frac1n.
    \]
    Since $s_J=1$ and $V^\intercal M V=1$ almost surely, equality holds in \eqref{eq:s-ineq}, hence equality holds in \eqref{eq:mij-ineq} for every $i$. For $i\ne J$, since $V_i\in\left\{{\pm1}\right\}$, this gives $(MV)_i=V_i\abs{M_{iJ}}$. For $i=J$, \cref{lem:maximizer} gives $(MV)_J=0=V_J\abs{M_{JJ}}$.
  \end{claimproof}

  \[
    \mathbb{E}\norm{MV}^2=\mathbb{E}\sum_i M_{iJ}^2=\frac{1}{n}\operatorname{tr}(M^2).
  \]
  On the other hand, $\mathbb{E}[VV^\intercal]=\operatorname{Proj}_K=I_n-uu^\intercal$ and $Mu=-u$, so
  \[
    \mathbb{E}\norm{MV}^2=\operatorname{tr}(M^2\operatorname{Proj}_K)=\operatorname{tr}(M^2)-1.
  \]
  Thus $\sum_{i\ne j}M_{ij}^2=\operatorname{tr}(M^2)=n/(n-1)$. Since $\abs{M}\bm{1}=\bm{1}$, we also have $\sum_{i\ne j}\abs{M_{ij}}=n$. Equality holds in the Cauchy--Schwarz inequality over the off-diagonal entries of $\abs{M}$, forcing $\abs{M_{ij}}=1/(n-1)$ for every $i\ne j$. Therefore $M=(I_n-J_n)/(n-1)$.

Let $G$ be the Gram matrix of $v_1,\dots,v_n$. If $\operatorname{rank} G<d$, we apply \cref{thm:main} in the span of these vectors to obtain a signed sum with squared norm at least $2n-\operatorname{rank} G>2n-d$, a contradiction.

By \cref{lem:main}, some vector $\varepsilon \in \left\{{\pm 1}\right\}^n$ satisfies $\varepsilon^\intercal M \varepsilon\ge\operatorname{tr} Q\ge m$.
