<!-- definition-1: body_md -->
\label{definition:passivity}
A stable transfer function $\mathcal T(s)$ is {\em strictly positive-real} \/if $\Phi_c(\imath \omega)\succ 0 $ for all $ \omega\in \mathbb{R}$ and 
it is \emph{positive-real} if $\Phi_c(\imath \omega)\succeq 0 $ for all $\ \omega\in \mathbb{R}$.\footnote{In the literature, a different definition is used, but for stable and rational continuous-time transfer functions, both definitions are equivalent.}

The transfer function $\mathcal T(s)$ is {\em asymptotically stable} if the eigenvalues of $sE-A$ are in the open left half plane, and it is 
{\em stable} if the eigenvalues of $sE-A$ are in the closed left half plane, with any eigenvalues occurring on the imaginary axis being simple.

The transfer function $\mathcal T(s)$ is {\em strictly passive} \/if  it is strictly positive-real and asymptotically stable and it is 
\emph{passive} if it is positive-real and stable.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:ph}
A linear time-invariant \emph{port-Hamiltonian (pH) descriptor system} has the generalized state-space form
 
\begin{equation} \label{eq:pH}
 \begin{array}{rcl} E \dot x(t)  & = & (J-R)Q x(t) + (G-K) u(t),\\
y(t) &=& (G+K)^{\mathsf{H}}Q x(t)+(S+N)u(t),
\end{array}
\end{equation}
 
and the system model matrices satisfy the structural conditions
 
\[
\mathcal V:= \left[ \begin{array}{cccc} -J & -G \\ \!G^{\mathsf{H}} \! & N \end{array} \right]=-\mathcal V^{\mathsf{H}},\
\mathcal W:= \!\left[ \begin{array}{cccc} R & \!K \! \\ K^{\mathsf{H}} & S \end{array} \right] \!=\mathcal W^{\mathsf{H}}\succeq 0, \ E^{\mathsf{H}}Q=Q^{\mathsf{H}}E \succeq 0.
\]

<!-- definition-3: body_md -->
A stable transfer function $\mathcal T(z)$ is {\em strictly positive-real} \/if $\Phi_d(e^{\imath \omega})\succ 0 $ for all $\ \omega\in [-\pi,\pi]$ and 
it is \emph{positive-real} if $\Phi_d(e^{\imath \omega})\succeq 0 $ for all $\ \omega\in [-\pi,\pi]$.\footnote{In the literature, a different definition is used, but for stable and rational discrete-time transfer functions, both definitions are equivalent.}

The transfer function $\mathcal T(z)$ is {\em asymptotically stable} if the eigenvalues of $zE-A$ are in the open unit disc, and it is 
{\em stable} if the eigenvalues of $zE-A$ are in the closed unit disc, with any eigenvalues occurring on the unit circle being simple.

The transfer function $\mathcal T(z)$ is {\em strictly passive} \/if  it is strictly positive-real and asymptotically stable and it is 
\emph{passive} if it is positive-real and stable.

 -c^2 E & 0 \\ 0 & I_m

<!-- theorem-1: body_md -->
\label{thm:mingamma}
	Consider the matrices $F, G, W_0$ as in~\eqref{WDelta} or~\eqref{WDeltahat} and the pointwise positive semi-definite matrix function
 
\begin{equation}\label{defmgamma}
M(\gamma):=\left[ \begin{array}{cc} \gamma F \\ G / \gamma\end{array}\right] W_0^{-1}
\left[ \begin{array}{cc} \gamma  F^\mathsf{H} & G^\mathsf{H} / \gamma \end{array}\right]
\end{equation}
 
in the real parameter $\gamma$. Then the largest eigenvalue $\lambda_{\max}(M(\gamma))$ is a \emph{unimodal function} of $\gamma$, ({i.e.} it is first monotonically decreasing and then monotonically increasing with growing $\gamma$). At the minimizing value $\underline \gamma$,  $M(\underline{\gamma})$ has an eigenvector $z$, {i.e.}
 
\[
 M(\underline{\gamma}) z = \underline\lambda_{\max} z, \quad z:=\left[ \begin{array}{cc} u \\ v \end{array}\right],
 \]
  
where
$  \|u\|_2^2=\|v\|_2^2=1$.
 
The minimum norm perturbation $\Delta$ is of rank $1$ and is given by $\Delta=vu^{\mathsf{H}}/\underline{\lambda}_{\max}$. It has norm $1/\underline{\lambda}_{\max}$
both in the spectral norm and in the Frobenius norm.

<!-- corollary-1: body_md -->
\label{cor:lev} Consider the matrices $F, G, W_0$ in \eqref{WDelta}\eqref{WDeltahat} and the pointwise positive semi-definite matrix function $M(\gamma)$ as in (\ref{defmgamma}). The largest eigenvalue of $M(\gamma)$ is also the largest eigenvalue of
 
\begin{equation} \label{maxeig}
\gamma^2 R^\mathsf{-H} F^\mathsf{H}F R^{-1} + 
R^\mathsf{-H} G^\mathsf{H}G R^{-1}/\gamma^2.
\end{equation}
 
An upper bound for $\underline{\lambda}_{\max}$ is given by $\underline{\lambda}_{\max}\le \frac{2}{\alpha\beta}$ where $\alpha^{-2}:=\lambda_{\max}(F W_0^{-1} F^\mathsf{H})$ and $\beta^{-2}=\lambda_{\max}(G W_0^{-1} G^\mathsf{H})$. The corresponding lower bound for $\| \Delta \|_2$ and $\| \Delta \|_F$ is given by
 
\[
  \rho_\mathcal{M}(X)  \ge \alpha\beta/2.
\]

<!-- theorem-2: body_md -->
\label{thm:Xpassivity}
Let $\mathcal{M}=\{A,B,C,D,E\}$ be a given model and assume that we are given a matrix $X\in  {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $, then the $X$-passivity radius $\rho_\mathcal{M}(X)$ is bounded by
 
\[
 \alpha\beta/2 \le \rho_\mathcal{M}(X) \le  \alpha\beta/(1+|v^{\mathsf{H}}w|),
 \]
  
where $v$, $\hat v$, $w$ and $\hat w$ are vectors of norm $1$, satisfying
 
\[\alpha^{-2}=\lambda_{\max}(F W_0^{-1} F^\mathsf{H}), \;\beta^{-2}=\lambda_{\max}(GW_0^{-1} G^\mathsf{H}), \; R^\mathsf{-H}F^\mathsf{H} \hat v=v/\alpha, \;  R^\mathsf{-H}G^\mathsf{H}\hat w=w/\beta.
\]
 
Moreover if $v$ and $w$ are linearly dependent, then $\rho_\mathcal{M}(X)=\alpha\beta/2$.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{th:optimal}
The normalized $X$-passivity radius $\rho_{\mathcal{M}_T}(c^2I_n)$ is larger or equal to the 
corresponding $X$-passivity radius $\rho_\mathcal{M}(X)$, for both the CT and DT cases, provided we choose $c^2:=\lambda_{\min}(X)$ as normalization constant.
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
We only give the proof for the CT case since both proofs are essentially the same. Notice that $c^2:=\lambda_{\min}(X)$ implies $\sigma_{\min}(T)=1$ and hence $T^\mathsf{H}T\succeq I_n$. 
We then start from the formulations of the passivity radii given in \eqref{maxeig}~:
\[  \rho_{\mathcal{M}}^{-1}(X)=\min_{\gamma}\| \gamma^2 R^\mathsf{-H}F^\mathsf{H}FR^{-1}  + R^\mathsf{-H}G^\mathsf{H}GR^{-1}/ \gamma^2 \|^2,
\]
and
\[  \rho_{\mathcal{M}_T}^{-1}(c^2I_n)=\min_{\gamma}\| \gamma^2 R_T^\mathsf{-H}F_T^\mathsf{H}F_TR_T^{-1}  + R_T^\mathsf{-H}G_T^\mathsf{H}G_TR_T^{-1}/ \gamma^2 \|^2.
\]
It follows from \eqref{gssPH} and \eqref{defWX} that 
\[ \hat T = \left[\begin{array}{cc} T & 0 \\ 0 & I_m \end{array}\right], \quad R_T^\mathsf{-H}F_T^\mathsf{H}=R^\mathsf{-H}F^\mathsf{H}\hat T^{-1}, \quad   R_T^\mathsf{-H}G_T^\mathsf{H}=R^\mathsf{-H}G^\mathsf{H}\hat T^{-1},\]
which implies that
\[  \rho_{\mathcal{M}_T}^{-1}(c^2I_n)=\min_{\gamma}\| \gamma^2 R^\mathsf{-H}F^\mathsf{H}\hat T^{-1} \hat T^\mathsf{-H}FR^{-1}  + R^\mathsf{-H}G^\mathsf{H}\hat T^{-1} \hat T^\mathsf{-H}GR^{-1}/ \gamma^2 \|^2,
\]
and since $\hat T^\mathsf{H}\hat T \succeq I_{n+m}$ the right hand side is non-increasing for all $\gamma$, including the 
minimizing $\gamma$. Therefore $\rho_{\mathcal{M}_T}(c^2I_n) \ge \rho_\mathcal{M}(X)$.

But it is possible that the coefficient $c$ is quite small. In such a case, we recommend to increase $c$ until reaching $\rho_{\mathcal{M}_T}(c^2I_n)=\rho_\mathcal{M}(X)$.

<!-- theorem-4: body_md -->
\label{thm:maxoverX}
Let $\mathcal{M}:=\{A,B,C,D,E\}$ be a minimal realization of a passive system with non-singular $E$, and let $X$ be a fixed certificate in the corresponding set $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $>$}}}} $. Then there is a unique $\xi^*(X)\ge 0$ which is maximal for the matrix inequality $$W_c(X,\mathcal{M}) \succeq \xi \ensuremath{\mathop{\mathrm{diag}}\!\left(E^{\mathsf{H}}XE,I_m\right)},$$ 
and is equal to the smallest eigenvalue of the positive definite pencil 
$W_c(X,\mathcal{M}) -\xi \ensuremath{\mathop{\mathrm{diag}}\!\left(E^{\mathsf{H}}XE,I_m\right)}$. This maximal value $\xi^*(X)$ equals zero if $X$ is a boundary point of $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $>$}}}} $ and is positive if and only if $X$ is in $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $.  Moreover, $X$ is a valid certificate for the passivity of the shifted system 
\begin{equation} \label{eq:shiftedCT}
\mathcal{M}_\xi=\{A+\frac{\xi}{2}E,B,C,D-\frac{\xi}{2}I_m,E\}.
\end{equation}
<!-- end theorem-4 -->
<!-- proof-3: body_md -->
If $X$ is a boundary point of $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $>$}}}} $ then $\det W_c(X,\mathcal{M})=0$ and for those $X$, we have $\xi^*(X)=0$.
If $X$ belongs to $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $, then $W_c(X,\mathcal{M})\succ 0$ and $\ensuremath{\mathop{\mathrm{diag}}\!\left(E^{\mathsf{H}}XE,I_m\right)}\succ 0$.
Therefore the pencil $W_c(X,\mathcal{M}) -\xi \ensuremath{\mathop{\mathrm{diag}}\!\left(E^{\mathsf{H}}XE,I_m\right)}$ is positive definite and all its eigenvalues are then positive. Its smallest eigenvalue is clearly equal to $\xi^*(X)$ and hence $\xi^*(X)>0$. Conversely, if $\xi^*(X)>0$ then $W_c(X,\mathcal{M})\succ 0$ and $X\in  {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $. 
Next, we have that for a given $X$ in $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $, the passivity LMI $W_c(X,\mathcal{M}_\xi)$ for the modified model $\mathcal{M}_\xi :=\{A+\frac{\xi}{2} E,B,C,D-\frac{\xi}{2} I_m,E\}$ is given by
 
\begin{equation*}                  
W_c(X,\mathcal{M}_\xi) := \! \left[\begin{array}{cc} \! -(A^\mathsf{H}+\frac{\xi}{2} E^{\mathsf{H}})XE-E^{\mathsf{H}}X(A+\frac{\xi}{2} E) &   C^{\mathsf{H}}-XB \\    C-B^{\mathsf{H}}X & D^\mathsf{H}+ D-\xi I_m \end{array}\right] \succeq 0.
\end{equation*}
Clearly this inequality follows from \eqref{xi} for $0 \le \xi \le \xi ^*(X)$ and its determinant is zero if $ \xi =\xi ^*(X)$. \hfill
<!-- end proof-3 -->
<!-- lemma-3: body_md -->
\label{inclusionc}
For every $X\succ 0$ in $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $ and any $\xi_-$ and $\xi_+$ satisfying $0\le \xi_- < \xi_+ \le \xi^*(X)$, the  systems $\mathcal{M}_{\xi_-}$ and $\mathcal{M}_{\xi_+}$ are passive. Moreover, the whole solution set of
$W_c(X,\mathcal{M}_{\xi_+})\succeq 0$ is included in the solution set of  $W_c(X,\mathcal{M}_{\xi_-})\succ 0$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
The LMIs for two different values $\xi_-<\xi_+$ are related as
 
\[
 W_c(X,\mathcal{M}_{\xi_+}) = W_c(X,\mathcal{M}_{\xi_-}) - (\xi_+-\xi_-) \ensuremath{\mathop{\mathrm{diag}}\!\left(E^{\mathsf{H}}XE,I_m\right)}.
\]
 
Since $X\in {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $, $\xi^*(X)>0$ and $\ensuremath{\mathop{\mathrm{diag}}\!\left(E^{\mathsf{H}}XE,I_m\right)}\succ 0$, it follows that
 
\begin{equation} \label{ineqsc}
W_c(X,\mathcal{M}) \succeq W_c(X,\mathcal{M}_{\xi_-}) \succ W_c(X,\mathcal{M}_{\xi_+})\succeq W_c(X,\mathcal{M}_{\xi^*(X)}) \succeq 0. \hfill
\end{equation}
 
Hence, the systems $\mathcal{M}_{\xi_-}$ and $\mathcal{M}_{\xi_+}$ are passive, since the associated LMIs have a nonempty solution set. Now consider {\em any} $X$ for which $W_c(X,\mathcal{M}_{\xi_+})\succ 0$.
Since $\xi_+$ is positive, so is $\xi^*(X)$ and hence $X\in  {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $. It then follows from \eqref{ineqsc} that $W_c(X,\xi_-)\succ 0$. Hence,
the solution set of $W_c(X,\mathcal{M}_{\xi_+}) \succeq 0$ is included in the solution set of $W_c(X,\mathcal{M}_{\xi_-}) \succ 0$.
<!-- end proof-4 -->
<!-- theorem-5: body_md -->
\label{th:pasrad}
Let $\mathcal{M}$ be a given passive model with transfer function ${\cal T}(s)$ and let $X=c^2T^\mathsf{H}T$ be a certificate for the corresponding LMI $W_c(X,\mathcal{M})\succeq 0$ with invertible $E$, then the extremal value $\xi^*(X)$ of the LMI $W_c(X,\mathcal{M}) - \xi \ensuremath{\mathop{\mathrm{diag}}\!\left(X,I_m\right)}\succ 0$ is the smallest eigenvalue of the positive definite pencil
\begin{equation} \label{eq:posdef}
    W_c(X,\mathcal{M})-\xi \ensuremath{\mathop{\mathrm{diag}}\!\left(E^\mathsf{H}XE,I_m\right)}. 
\end{equation} 
Since $E$ is invertible, this is also the smallest eigenvalue of the normalized matrix
$$ \widetilde W(c^2I,\widetilde \mathcal{M}_T):=\left[\begin{array}{cc} -c^2(A_TE_T^{-1})^\mathsf{H} - c^2 (A_TE_T^{-1}) & (C_TE_T^{-1})^\mathsf{H}-c^2B_T \\ C_TE_T^{-1} - c^2B_T^\mathsf{H} & D_T+D_T^\mathsf{H}\end{array}\right],
$$
corresponding to the model $\widetilde \mathcal{M}:=\{A_TE_T^{-1},B_T,C_TE_T^{-1},D_T,I_n\}$ with transfer function ${\cal T}(s)$.
<!-- end theorem-5 -->
<!-- proof-5: body_md -->
The matrices $W_c(X,\mathcal{M})$ and $\xi \ensuremath{\mathop{\mathrm{diag}}\!\left(E^\mathsf{H}XE,I_m\right)}$ are positive definite, which implies that all eigenvalues are positive, and are equal to the eigenvalues of 
$$ \widetilde W_c(c^2I,\widetilde \mathcal{M}_T)= \ensuremath{\mathop{\mathrm{diag}}\!\left((TE)^{-\mathsf{H}},I_m\right)}
W_c(X,\mathcal{M})\ensuremath{\mathop{\mathrm{diag}}\!\left((TE)^{-1},I_m\right)}.
$$  
The smallest value of $\xi$ for which the pencil \eqref{eq:posdef} becomes singular is therefore the smallest eigenvalue of that matrix.

<!-- theorem-6: body_md -->

Let $\mathcal{M}:=\{A,B,C,D,E\}$ be a minimal realization of a passive system with non-singular $E$, and let $X$ be a fixed certificate in the corresponding set $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $>$}}}} $. $ \widehat W_d(X,\mathcal{M}) - \xi \ensuremath{\mathop{\mathrm{diag}}\!\left(X,E^{\mathsf{H}}XE,I_m\right)}$. This maximal value $\xi^*(X)$ equals zero if $X$ is a boundary point of $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $>$}}}} $ and is positive if and only if $X$ is in $ {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $\gg$}}}} $. \mathcal{M}_\xi=\{A_\xi,B_\xi,C_\xi,D_\xi,E_\xi\}
:=\left\{\frac{A}{(1-\xi)},\frac{B}{(1-\xi)},\frac{C}{(1-\xi)},\frac{D-\xi I_m/2}{(1-\xi)},E\right\}.
\end{equation}

<!-- proof-6: body_md -->
The proof is completely analogous to the proof of Theorem \ref{thm:maxoverX} in the CT case, except for the shifted model \eqref{eq:shiftedDT}, which is obtained as follows.
The equality in the left hand side of the LMI
\begin{equation*}

\end{equation*} 
is clearly satisfied by the matrix
$$
\widehat W_d(X,\mathcal{M}_\xi):= \left[\begin{array}{ccc} X  &  XA_\xi & XB_\xi \\ 
A_\xi^{\mathsf{H}}X & E^{\mathsf{H}}XE & C_\xi^{\mathsf{H}} \\   B_\xi^{\mathsf{H}}X & C_\xi & D_\xi^{\mathsf{H}}+D_\xi \end{array}\right]
$$ 
which corresponds to the modified model \eqref{eq:shiftedDT}, and also implies that $\xi$ must be smaller than 1.

<!-- theorem-8: body_md -->
\label{thm:Xid} Let $\mathcal{M}$ be a strictly passive and minimal system. Then there is a bounded supremum $\Xi:=\sup_\xi \{\xi \; | \; \mathcal{T}_\xi(\lambda) \mathrm{\; is \; strictly \; passive}\}$ where the following properties hold
\begin{enumerate}
\item  $\mathcal{T}_\Xi(\lambda)$ is passive,
\item the solution set of $W(X,\mathcal{M}_\Xi)\succeq 0$ is not empty,
\item the solution set of $W(X,\mathcal{M}_\Xi)\succ 0$ is empty,
\item for any $\xi < \Xi$ the solution set of $W(X,\mathcal{M}_\xi)\succ 0$ is non-empty,
\item  $\Xi:=\sup_X \xi^*(X)$ for all $X\in  {\mathbb{X}^{\raisebox{0.2em}{{\fontsize{3}{2}\selectfont $>$}}}} $.
\end{enumerate}
