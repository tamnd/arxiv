<!-- definition-1: body_md -->
\label{solution}
For $(u_0,u_1)\in V\times H$, an energy solution on $[0,T]$ satisfies
\begin{equation}\label{regularity}
\begin{gathered}
u\in C([0,T];V),\quad u_t\in C([0,T];H)\cap L^2(0,T;V),\quad 
u_{tt}\in L^2(0,T;V^*)
\end{gathered}
\end{equation}
with $u(0)=u_0,~u_t(0)=u_1,$ and, for every $z\in V$ and almost every $t\in(0,T)$,
\begin{equation*}
\langle u_{tt},z\rangle+(\nabla u,\nabla z)+(\nabla u_t,\nabla z)
-\int_0^t g(t-s)(\nabla u(s),\nabla z)\,ds
=\int_\Omega |u|^{p(x)-2}u\log|u|z\,dx.
\end{equation*}

\label{local}
Assume \textup{(H1)--(H2)}. For every $(u_0,u_1)\in V\times H$, problem~\eqref{problem} admits a unique maximal energy solution on $[0,T_{\max})$, where $T_{\max}\in(0,\infty]$. The solution depends continuously on the initial data on every compact subinterval of its existence interval. If $T_{\max}<\infty$, then
\begin{equation}\label{28}
\limsup_{t\uparrowT_{\max}}\left(\lVert u(t)\rVert_V+\lVert u_t(t)\rVert_2\right)=\infty.
\end{equation}
The energy is locally absolutely continuous, and
\begin{equation}\label{ei}
E'(t)=-\lVert \nabla u_t(t)\rVert_2^2-\frac12g(t)\lVert \nabla u(t)\rVert_2^2
+\frac12\int_0^t g'(t-s)\lVert \nabla u(t)-\nabla u(s)\rVert_2^2\,ds
\end{equation}
for almost every $t<T_{\max}$.

\label{blowup}
Assume \textup{(H1)--(H2)}, and let $u$ be the solution in Theorem~\ref{local}.
\begin{enumerate}
\item \textbf{Negative shifted initial energy.}
Suppose \eqref{kw} holds and $\mathcal E(0)<0$. Then $T_{\max}<\infty$ and \eqref{28} holds. Put $\beta:=-\mathcal E(0)>0$. For every $\sigma>0$ such that
\begin{equation*}
(p_--2)\bigl[(u_0,u_1)+\beta\sigma\bigr]>2\lVert \nabla u_0\rVert_2^2,
\end{equation*}
one has
\begin{equation}\label{30}
T_{\max}\le
\frac{2(\lVert u_0\rVert_2^2+\beta\sigma^2)}
{(p_--2)[(u_0,u_1)+\beta\sigma]-2\lVert \nabla u_0\rVert_2^2}.
\end{equation}
In particular, minimizing the right-hand side of \eqref{30} over all admissible $\sigma$ gives
\begin{equation}\label{31}
T_{\max}\le
\frac{4\left[\sqrt{d_0^2+(p_--2)^2\beta\lVert u_0\rVert_2^2}-d_0\right]}
{(p_--2)^2\beta},
\end{equation}
where 
\begin{equation*}
d_0:=(p_--2)(u_0,u_1)-2\lVert \nabla u_0\rVert_2^2.
\end{equation*}

\item \textbf{Zero shifted initial energy.}
Suppose \eqref{ks} holds, $\mathcal E(0)=0$, and $(u_0,u_1)>0$. Then $T_{\max}<\infty$ and \eqref{28} holds. Its upper bound is given below by \eqref{35}, with $L_0:=(u_0,u_1)$.

\item \textbf{Positive shifted initial energy.}
Suppose \eqref{ks} holds and
\begin{equation}\label{he}
0<\mathcal E(0)<\frac C{p_-}(u_0,u_1).
\end{equation}
Then $T_{\max}<\infty$ and \eqref{28} holds. The upper bound is again given by \eqref{35},
without any additional restriction on the initial energy.
\end{enumerate}
For the bounds in \textup{(ii)} and \textup{(iii)}, define, in both cases,
\begin{equation}\label{32}
L_0:=(u_0,u_1)-\frac{p_-}{C}\mathcal E(0)>0,\quad
\beta:=\frac{CL_0}{2p_-}.
\end{equation}
Let $\tau$ be the unique positive solution of
\begin{equation}\label{33}
CL_0\tau^2+
\left[L_0-\frac{2(\mathcal E(0)+\beta)}{\lambda_1}\right]\tau
-\lVert u_0\rVert_2^2=0.
\end{equation}
              
                                      
                         
                                      
                                                                                
         
                
Set
\begin{equation}\label{34}
\begin{split}
U_\tau:=\left(\lVert u_0\rVert_2+
\sqrt{\frac{\tau(\mathcal E(0)+\beta)}{\lambda_1}}\right)^2,
\quad G_\tau:=\left(\lVert \nabla u_0\rVert_2+
\sqrt{\tau(\mathcal E(0)+\beta)}\right)^2,
\end{split}
\end{equation}
and \[d_\tau:=\frac{p_--2}{2}L_0-2G_\tau.\]
Then
\begin{equation}\label{35}
T_{\max}\le\tau+
\frac{4\left[\sqrt{d_\tau^2+(p_--2)^2\beta U_\tau}-d_\tau\right]}
{(p_--2)^2\beta}.
\end{equation}

<!-- lemma-1: body_md -->
\label{correctionlemma}
For every $q\ge p_->2$ and $r\in\mathbb R$,
\begin{equation}\label{36}
|r|^q\left(
\frac{q-p_-}{q}\log|r|+\frac{p_-}{q^2}
\right)\ge -c_{p_-}(q),
\end{equation}
where the expression at $r=0$ is defined by continuity.
For each fixed $q$, $c_{p_-}(q)$ is the smallest
nonnegative constant for which
\eqref{36} holds for all $r\in\mathbb R$.
Consequently, every energy solution satisfies
\begin{equation}\label{37}
\begin{aligned}
\int_\Omega |u|^{p(x)}\log|u|\,dx
\ge{}&
\frac{p_-}{2}\lVert u_t\rVert_2^2
+\frac{p_-}{2}
\left(1-\int_0^t g(s)\,ds\right)
\lVert \nabla u\rVert_2^2\\
&+\frac{p_-}{2}(g\circ\nabla u)(t)
-p_-\mathcal E(t).
\end{aligned}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $q=p_-$, the left-hand side of
\eqref{36} equals
$|r|^{p_-}/p_-\ge0$.
Thus the optimal nonnegative correction is
$c_{p_-}(p_-)=0$.
For  $q>p_-$ and $r\ne0$.
Setting $y=\log|r|$, we obtain
\[
\frac{d}{dy}
\left[
e^{qy}\left(
\frac{q-p_-}{q}y+\frac{p_-}{q^2}
\right)
\right]
=e^{qy}\bigl[(q-p_-)y+1\bigr].
\]
The derivative is negative for $y<-1/(q-p_-)$
and positive for $y>-1/(q-p_-)$.
Hence the global minimum is attained at
$y=-1/(q-p_-)$ and equals
\[
-\frac{q-p_-}{q^2}
e^{-\frac{q}{q-p_-}}
=-c_{p_-}(q).
\]
The expression in \eqref{36} tends
to zero as $r\to0$, so the inequality also holds
at $r=0$.
Moreover, equality holds when
$|r|=e^{-1/(q-p_-)}$, which proves optimality.

By the definition of $F$, taking $q=p(x)$
in \eqref{36} and integrating over
$\Omega$ gives
\[
\int_\Omega |u|^{p(x)}\log|u|\,dx
\ge p_-\int_\Omega F(x,u)\,dx-C_*.
\]
Using \eqref{energy} and
$p_-\mathcal E(t)=p_-E(t)+C_*$ yields
\eqref{37}.

\label{growthlemma}
Assume \textup{(H1)--(H2)} and \eqref{ks}.
Define
\begin{equation*}
L(t)=(u(t),u_t(t))-\frac{p_-}{C}\mathcal E(t).
\end{equation*}
Then $L$ is locally absolutely continuous on
$[0,T_{\max})$ and satisfies
\begin{equation*}
L(t)\ge L(0)e^{Ct}
\quad\text{for }0\le t<T_{\max}.
\end{equation*}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Fix $T<T_{\max}$.
The regularity \eqref{regularity} and the energy
identity imply that $L$ is absolutely continuous
on $[0,T]$. Using $\eqref{problem}_1$ gives
\begin{equation}\label{39}
\begin{aligned}
\frac{d}{dt}(u,u_t)
={}&\lVert u_t\rVert_2^2
-\left(1-\int_0^t g(s)\,ds\right)
 \lVert \nabla u\rVert_2^2
-(\nabla u,\nabla u_t)\\
&+\int_0^t g(t-s)
  (\nabla u(t),\nabla u(s)-\nabla u(t))\,ds
+\int_\Omega |u|^{p(x)}\log|u|\,dx
\end{aligned}
\end{equation}
for almost every $t\in(0,T)$.
By Cauchy--Schwarz and Young's inequalities, one has
\begin{equation}\label{40}
\begin{aligned}
\int_0^t g(t-s)
(\nabla u(t),\nabla u(s)-\nabla u(t))\,ds
\ge
-\frac{1}{2p_-}
 \left(\int_0^t g(s)\,ds\right)
 \lVert \nabla u(t)\rVert_2^2
-\frac{p_-}{2}(g\circ\nabla u)(t),
\end{aligned}
\end{equation}
\[
-(\nabla u,\nabla u_t)
\ge
-\frac{C}{4p_-}\lVert \nabla u\rVert_2^2
-\frac{p_-}{C}\lVert \nabla u_t\rVert_2^2.
\]
Substituting these estimates and
\eqref{37} into
\eqref{39}, we get

\begin{equation*}
\begin{aligned}
\frac{d}{dt}(u,u_t)
={}&\frac{p_-+2}{2}\lVert u_t\rVert_2^2
+\frac{2[(p_--1)^2\ell-1]-C}{4p_-}
 \lVert \nabla u\rVert_2^2\\
&-\frac{p_-}{C}\lVert \nabla u_t\rVert_2^2+\frac{p_-}{2}\lVert u_t\rVert_2^2
-p_-\mathcal E(t).
\end{aligned}
\end{equation*}
Here we used
\[
\begin{aligned}
\frac{p_--2}{2}
 \left(1-\int_0^t g(s)\,ds\right)
-\frac{1}{2p_-}\int_0^t g(s)\,ds
-\frac{C}{4p_-}\ge
\frac{2[(p_--1)^2\ell-1]-C}{4p_-}.
\end{aligned}
\]
Note that
$\mathcal E'(t)\le-\lVert \nabla u_t(t)\rVert_2^2$, one gets
\begin{equation}\label{41}
L'(t)\ge
\frac{p_-+2}{2}\lVert u_t\rVert_2^2
+\frac{2[(p_--1)^2\ell-1]-C}{4p_-}
 \lVert \nabla u\rVert_2^2
-p_-\mathcal E(t).
\end{equation}
By \eqref{29},
\[
C\le p_-+2,
\quad
\lambda_1
\frac{2[(p_--1)^2\ell-1]-C}{4p_-}
\ge\frac{C}{2}.
\]
Consequently, Poincar\'e's inequality \eqref{poincare} and
$2(u,u_t)\le\lVert u\rVert_2^2+\lVert u_t\rVert_2^2$ yield
\[
\begin{aligned}
L'(t)
\ge \frac{C}{2}
\left(\lVert u_t\rVert_2^2+\lVert u\rVert_2^2\right)
-p_-\mathcal E(t)\ge C(u,u_t)-p_-\mathcal E(t)
=CL(t).
\end{aligned}
\]
Hence $(e^{-Ct}L(t))'\ge0$ almost everywhere.
Integration gives $L(t)\ge e^{Ct}L(0)$ on $[0,T]$.
Since $T<T_{\max}$ was arbitrary, the proof
is complete.

<!-- lemma-3: body_md -->
\label{concavity}
Let $M\in W^{2,1}(0,R)\cap C^1([0,R])$ be positive and satisfy
\[
MM''-(1+\theta)(M')^2\ge0\quad\text{a.e. on }(0,R)
\]
for some $\theta>0$. If $M'(0)>0$, such a function cannot exist on $[0,R]$ when
\begin{equation*}
R>\frac{M(0)}{\theta M'(0)}.
\end{equation*}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Since $M\in C^1([0,R])$ and $M>0$ on the compact interval $[0,R]$, there exists
$m_0>0$ such that
\[
M(s)\ge m_0\quad\text{for all }s\in[0,R].
\]
 The Sobolev chain rule gives
\[
(M^{-\theta})''=-\theta M^{-\theta-2}
\bigl[MM''-(1+\theta)(M')^2\bigr]\le0,
\]
which reveals that $M^{-\theta}$ is concave on $[0,R]$.
Hence $M(s)^{-\theta}\le M(0)^{-\theta}-\theta M(0)^{-\theta-1}M'(0)s$. Equivalently,
\[
M(s)^{-\theta}
\le
M(0)^{-\theta}
\left(
1-\frac{\theta M'(0)}{M(0)}s
\right).
\]
If
$
s>\frac{M(0)}{\theta M'(0)},
$
then the right-hand side is strictly negative. This contradicts the positivity of the left-hand side, since $M(s)>0$ implies
$
M(s)^{-\theta}>0.
$
Therefore no point $s$ in the interval $[0,R]$ can exceed
$M(0)/(\theta M'(0))$. This completes the proof.

\label{negativelemma}
Assume \textup{(H1)--(H2)} and \eqref{kw}. If $\mathcal E(t_0)<0$ for some $t_0<T_{\max}$, put $\beta=-\mathcal E(t_0)>0$. For every $\sigma>0$ satisfying

\end{equation}

\frac{2(\lVert u(t_0)\rVert_2^2+\beta\sigma^2)}

\end{equation}

<!-- proof-5: body_md -->
Fix an admissible $\sigma$. If \eqref{43} fails, let us choose $R$ satisfying
\[\frac{2\bigl(\|u(t_0)\|_2^2+\beta\sigma^2\bigr)}
{(p_--2)\bigl[(u(t_0),u_t(t_0))+\beta\sigma\bigr]
-2\|\nabla u(t_0)\|_2^2}
<R<T_{\max}-t_0.\]
 For $0\le s\le R$, put $t=t_0+s$ and define
\begin{equation*}
M(s)=\lVert u(t)\rVert_2^2+\int_{t_0}^t\lVert \nabla u(r)\rVert_2^2\,dr
+(R-s)\lVert \nabla u(t_0)\rVert_2^2+\beta(s+\sigma)^2.
\end{equation*}
\emph{Step 1: the concavity functional and its regularity.}
The term $\beta(s+\sigma)^2$ makes $M$ strictly positive.
The product rule used in \eqref{39} and $u\in H^1(t_0,t_0+R;V)$ imply $M\in C^1([0,R])\cap W^{2,1}(0,R)$. In particular,
\begin{equation*}
M'(s)=2\left[(u,u_t)+\int_{t_0}^t(\nabla u,\nabla u_r)\,dr+\beta(s+\sigma)\right].
\end{equation*}
 $M'$ is continuous on $[0,R]$ because
$u\in C([t_0,t_0+R];V)$ and
$u_t\in C([t_0,t_0+R];H)$.
Moreover, the weak product rule gives
\begin{equation}
\label{1}
M''(s)
=2\lVert u_t(t)\rVert_2^2
+2\langle u_{tt}(t),u(t)
+2(\nabla u(t),\nabla u_t(t))
+2\beta
\end{equation}
for almost every $s\in(0,R)$.
All terms belong to $L^1(0,R)$.

\label{stoppinglemma}

\begin{gathered}

<!-- proof-7: body_md -->
Suppose $T_{\max}>\tau$. Since $\mathcal E(0)\ge0$ and $L_0>0$, we have $(u_0,u_1)
=L_0+\frac{p_-}{C}\mathcal E(0)>0$, and hence $u_0\ne0$. Obviously, \eqref{33}  has exactly one positive root.

\emph{Step 1: }
For any $t\le\tau$ for which $\mathcal E(s)\ge-\beta$ on $[0,t]$, \eqref{dissipation} directly illustrates 
\begin{equation*}
\int_0^t\lVert \nabla u_s\rVert_2^2\,ds\le\mathcal E(0)-\mathcal E(t)\le\mathcal E(0)+\beta.
\end{equation*}
The representation $u(t)=u_0+\int_0^t u_s\,ds$, Poincar\'e's inequality \eqref{poincare}, and Cauchy--Schwarz imply
\begin{equation}\label{54}
\begin{split}
\lVert u(t)\rVert_2&\le\|u_0\|_2+\int_0^t\|u_s(s)\|_2\,ds\le\|u_0\|_2+
\sqrt{\frac t{\lambda_1}}
\left(\int_0^t\|\nabla u_s(s)\|_2^2\,ds\right)^{1/2}\\
&\le\|u_0\|_2+
\sqrt{\frac{t(\mathcal E(0)+\beta)}{\lambda_1}}
,\\
\lVert \nabla u(t)\rVert_2&\le\|\nabla u_0\|_2+
\int_0^t\|\nabla u_s(s)\|_2\,ds
\le\lVert \nabla u_0\rVert_2+
\sqrt{t(\mathcal E(0)+\beta)}.
\end{split}
\end{equation}

\emph{Step 2: }
Suppose, for a contradiction, that $\mathcal E(s)\ge-\beta$ for all $s\in[0,\tau]$. Lemma~\ref{growthlemma} gives
\[

=2L(t)+\frac{2p_-}{C}\mathcal E(t)
\ge2L_0e^{Ct}-\frac{2p_-\beta}{C}=2L_0e^{Ct}-L_0.
\]
Therefore
\begin{equation*}
\lVert u(t)\rVert_2^2\ge\lVert u_0\rVert_2^2+\frac{2L_0}{C}(e^{Ct}-1)-L_0t.
\end{equation*}
Using $e^{Ct}-1>Ct+(Ct)^2/2$ for $t>0$, then
\[
\lVert u(\tau)\rVert_2^2>
\lVert u_0\rVert_2^2+L_0\tau+CL_0\tau^2
=2\lVert u_0\rVert_2^2+\frac{2\tau(\mathcal E(0)+\beta)}{\lambda_1},
\]
where the equality follows from \eqref{33}. In contrast, squaring the first inequality in \eqref{54} and using $(a+b)^2\le2a^2+2b^2$ yields
\[
\lVert u(\tau)\rVert_2^2\le
2\lVert u_0\rVert_2^2+\frac{2\tau(\mathcal E(0)+\beta)}{\lambda_1}.
\] 
This is a contradiction.

\emph{Step 3: }
The contradiction in Step~2 shows that
$\mathcal E(s_0)<-\beta$ for some $s_0\in(0,\tau]$.
If $s_0=\tau$, continuity gives the same strict inequality
at a time slightly earlier than $\tau$.
Thus there exists $s_1\in(0,\tau)$ such that
$\mathcal E(s_1)<-\beta$.
Since $\mathcal E(0)\ge0\ge-\beta$, the intermediate value theorem
provides $t_\beta\in(0,s_1)$ with
$\mathcal E(t_\beta)=-\beta$.

The energy is nonincreasing, so
$\mathcal E(t)\ge\mathcal E(t_\beta)=-\beta$
for every $t\in[0,t_\beta]$.
Hence \eqref{54} applies at $t_\beta$.
Using $t_\beta<\tau$ and \eqref{34}, we obtain
\[
\begin{aligned}
\|u(t_\beta)\|_2^2
&\le
\left(
\|u_0\|_2+
\sqrt{\frac{t_\beta(\mathcal E(0)+\beta)}{\lambda_1}}
\right)^2
\le U_\tau,\\
\|\nabla u(t_\beta)\|_2^2
&\le
\left(
\|\nabla u_0\|_2+
\sqrt{t_\beta(\mathcal E(0)+\beta)}
\right)^2
\le G_\tau.
\end{aligned}
\]
Moreover, the definition of $L$, Lemma~\ref{growthlemma},
and \eqref{32} establish 
\[
\begin{aligned}
(u(t_\beta),u_t(t_\beta))=L(t_\beta)+\frac{p_-}{C}\mathcal E(t_\beta)
=L(t_\beta)-\frac{p_-\beta}{C}\ge L_0e^{Ct_\beta}-\frac{L_0}{2}
\ge\frac{L_0}{2}.
\end{aligned}
\]
Together with $\mathcal E(t_\beta)=-\beta$ and
$0<t_\beta<\tau$, these estimates prove \eqref{53}.

\[
\frac{b_s^2}{s^2}=\frac{2}{\lVert \varphi\rVert_2^2}
\left[\kappa+\frac1{s^2}\int_\Omega F(s\varphi)\,dx-
\frac12\lVert \nabla\varphi\rVert_2^2\right]\longrightarrow\infty\quad\text{as }s\to\infty.
\]
Consequently,
\[
\frac{E(0)}{(u_0,u_1)}=
\frac{\kappa s}{b_s\lVert \varphi\rVert_2^2}\longrightarrow0\quad\text{as }s\to\infty,
\]
which proves the right inequality in \eqref{aa3} for large $s$. The choice of $\kappa$ yields the left inequality for every such $s$. Since $C_*=0$, Theorem~\ref{blowup}\textup{(iii)} applies.
