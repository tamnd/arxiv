<!-- head-1: body_md -->
We denote by ${\mathcal{C}^\infty(\mathbb{R},\mathbb{R}^q)}$ the space of infinitely differentiable functions mapping from $\mathbb{R}$ to $\mathbb{R}^q$. We define 
\begin{equation*} 
\mathcal{D}(\mathbb{R},\mathbb{R}^q):=\left\{ w \in \mathcal{C}^\infty(\mathbb{R},\mathbb{R}^q) \;\middle| \; w \text{ has compact support} \right\}. 
\end{equation*}We denote by $\mathbb{W}^{\mathbb{T}}$ the collection of all maps from $\mathbb{T}$ to $\mathbb{W}$.  
Moreover, ${\mathbb{R}^{p \times q}[\xi]}$ denotes the set of real polynomial matrices of size $p \times q$ in the indeterminate $\xi$. 
Let $f : \mathbb{R} \to \mathbb{R}^q$ and $K \in \mathbb{R}^{r \times q}$. We denote by $Kf$ the function $t \mapsto Kf(t)$ and, if $r=q$, by $f^\top Kf$ the function $t\mapsto f(t)^\top Kf(t)$.
<!-- end head-1 -->
<!-- definition-1: body_md -->
A \emph{dynamical system} is defined as a triple $\Sigma=(\mathbb{T}, \mathbb{W},\mathcal{B})$ with $\mathbb{T} \subseteq \mathbb{R}$ the \emph{time set}, $\mathbb{W}$ the \emph{signal alphabet}, and $\mathcal{B} \subseteq \mathbb{W}^{\mathbb{T}}$ the \emph{behavior} of the system.

A behavior $\mathcal{B} \in \mathcal{L}^q$ is \emph{controllable} if for each $w_1,w_2 \in \mathcal{B}$ there exists a $t_1 \geq 0$ and a trajectory $w\in \mathcal{B}$ such that $w(t)=w_1(t)$ for $t\leq0$ and ${w(t)=w_2(t-t_1)}$ for $t\geq t_1$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
Consider $\Phi \in \mathbb{R}_s^{q \times q}[\zeta, \eta]$. We call the QDF $Q_\Phi$ \emph{nonnegative}, denoted by $Q_\Phi \geq 0$, if ${Q_\Phi(w) \geq 0}$ for all $w\in \mathcal{C}^\infty(\mathbb{R},\mathbb{R}^q)$, that is, ${\left(Q_\Phi(w)\right)(t) \geq 0}$ for all ${w\in \mathcal{C}^\infty(\mathbb{R},\mathbb{R}^q)}$ and all $t\in \mathbb{R}$. Moreover, we call $Q_\Phi$ \emph{positive}, denoted by $Q_\Phi>0$, if ${Q_\Phi \geq 0}$ and if the only ${w\in \mathcal{C}^\infty (\mathbb{R},\mathbb{R}^q)}$ for which ${Q_\Phi(w)=0}$, that is, ${\left(Q_{\Phi}(w)\right)(t)=0}$ for all $t\in \mathbb{R}$, is $w=0$.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
Consider $\Phi \in \mathbb{R}_s^{q \times q}[\zeta, \eta]$ and $\mathcal{B}\in \mathcal{L}^q$. We call the QDF $Q_\Phi$ \emph{zero} on $\mathcal{B}$, denoted by $Q_\Phi \stackrel{\mathcal{B}}{=}0$, if \\$Q_\Phi(w)=0$ for all $w\in \mathcal{B}$. Moreover, we call it \emph{nonnegative} on $\mathcal{B}$, denoted by $Q_\Phi \stackrel{\mathcal{B}}{\geq} 0$, if $Q_\Phi(w) \geq 0$ for all $w\in \mathcal{B}$, and \emph{positive} on $\mathcal{B}$, denoted by $Q_\Phi\stackrel{\mathcal{B}}{>}0$, if $Q_\Phi \stackrel{\mathcal{B}}{\geq} 0$ and if the only $w\in\mathcal{B}$ for which $Q_\Phi(w)=0$ is $w=0$.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
The behavior $\mathcal{B}\in \mathcal{L}^q$ is \emph{autonomous} if for any $w_1,w_2 \in \mathcal{B}$ such that $w_1(t)=w_2(t)$ for $t<0$ we have that $w_1(t)=w_2(t)$ for all $t \in \mathbb{R}$.
<!-- end definition-5 -->
<!-- definition-6: body_md -->
Consider an autonomous behavior ${\mathcal{B} \in \mathcal{L}^q}$. We call $\mathcal{B}$ \emph{stable} if every $w\in \mathcal{B}$ is bounded on the half-line $[0, \infty)$ and \emph{asymptotically stable} if ${\lim_{t \to \infty}w(t)=0}$ for all $w\in \mathcal{B}$.
<!-- end definition-6 -->
<!-- theorem-1: body_md -->
Let ${\mathcal{B} \in \mathcal{L}^q}$ be autonomous. 
	\begin{enumerate}[label=\roman*)] 
		\item  $\mathcal{B}$ is stable if and only if there exists $\Psi \in \mathbb{R}_s^{q \times q}[\zeta, \eta]$ such that $Q_\Psi \stackrel{\mathcal{B}}{>} 0$ and $\frac{d}{dt}Q_\Psi \stackrel{\mathcal{B}}{\leq} 0$. 
		\item $\mathcal{B}$ is asymptotically stable if and only if there exists $\Psi \in \mathbb{R}_s^{q \times q}[\zeta, \eta]$ such that $Q_\Psi \stackrel{\mathcal{B}}{\geq} 0$ and $\frac{d}{dt}Q_\Psi \stackrel{\mathcal{B}}{<} 0$. 
	\end{enumerate} 
	\label{t: lyap stab}
<!-- end theorem-1 -->
<!-- definition-7: body_md -->
We call $\mathcal{B}$ \emph{dissipative} with respect to $Q_\Phi$ if 
	\begin{equation*} 
		\int_{-\infty}^\infty Q_\Phi(w)dt \geq 0  
	\end{equation*} 
	for all $w\in \mathcal{B} \cap \mathcal{D}(\mathbb{R},\mathbb{R}^q)$. We call $\mathcal{B}$ \emph{half-line dissipative} with respect to $Q_\Phi$ if 
	\vspace{-0.5ex} 
	\begin{equation*} 
		\int_{-\infty}^0 Q_\Phi(w)dt \geq 0  
	\end{equation*} 
	for all $w\in \mathcal{B} \cap \mathcal{D}(\mathbb{R},\mathbb{R}^q)$. Moreover, if these inequalities hold strictly for all nonzero $w \in \mathcal{B} \cap \mathcal{D}(\mathbb{R},\mathbb{R}^q)$, we call $\mathcal{B}$ \emph{strictly dissipative} with respect to $Q_\Phi$ and \emph{strictly half-line dissipative} with respect to $Q_\Phi$, respectively.
<!-- end definition-7 -->
<!-- definition-8: body_md -->
Let $\mathcal{B} \in \mathcal{L}^q$ and $\Phi \in \mathbb{R}_s^{q \times q}[\zeta,\eta]$. We call the QDF $Q_\Psi$ induced by $\Psi \in \mathbb{R}_s^{q \times q}[\zeta,\eta]$ a \emph{storage function for $(\mathcal{B}, Q_\Phi)$} if it satisfies the \emph{dissipation inequality} 
	\vspace{-0.5ex} 
	\begin{equation} 
		\frac{d}{dt}Q_\Psi \stackrel{\mathcal{B}}{\leq} Q_\Phi. 
		\label{diss ineq} 
	\end{equation} 
	Moreover, if \eqref{diss ineq} holds strictly, we say $Q_{\Psi}$ satisfies the dissipation inequality strictly.
<!-- end definition-8 -->
<!-- remark-1: body_md -->
Note that the dissipation inequality can be seen as finding a QDF $Q_\Psi$ such that the QDF $Q_\Phi -\frac{d}{dt}Q_\Psi$ is nonnegative on $\mathcal{B}$. This can be verified by solving an LMI for a coefficient matrix of $Q_\Psi$; for example, see \cite[Algorithm~8]{belur_algorithmic_2002}.
<!-- end remark-1 -->
<!-- theorem-2: body_md -->
Let $\mathcal{B} \in \mathcal{L}^q$ be controllable and consider ${\Phi\in \mathbb{R}_s^{q \times q}[\zeta, \eta]}$.  
	The following statements hold. 
	\begin{enumerate}[label=\roman*)] 
		\item $\mathcal{B}$ is dissipative with respect to $Q_\Phi$ if and only if there exists a storage function $Q_\Psi$ for $(\mathcal{B}, Q_\Phi)$. \label{first stat of thm diss char} 
		\item  $\mathcal{B}$ is half-line dissipative with respect to $Q_\Phi$ if and only if there exists a storage function $Q_\Psi$ for $(\mathcal{B}, Q_\Phi)$ such that $Q_\Psi \stackrel{\mathcal{B}}{\geq }0$. \label{second stat of thm diss char} 
	\end{enumerate} 
	\label{t: diss char}

<!-- remark-2: body_md -->
Unlike in the original definition of dissipativity in \cite{willems_dissipative_1972}, we do not define dissipativity in terms of the existence of nonnegative storage functions. This is in line with other dissipativity studies within the behavioral approach  \cite{trentelman_every_1997, willems_quadratic_1998}. We note that dissipativity with a possibly indefinite storage function is sometimes referred to as \emph{cyclo-dissipativity}, see e.g. \cite[Section~3.1]{van_der_schaft_dissipative_2017}.

Consider  $\bar{\mathcal{B}}_1\in \mathcal{L}^{m_1+p_1}$ and $\bar{\mathcal{B}}_2\in \mathcal{L}^{m_2+p_2}$. For $i=1,2$, partition ${w_i\in \bar{\mathcal{B}}_i}$ as ${w_i=(u_i,y_i)}$, where ${u_i\in \left(\mathbb{R}^{m_i}\right)^\mathbb{R}}$ and ${y_i\in \left(\mathbb{R}^{p_i}\right)^\mathbb{R}}$. Consider the interconnection obtained by setting 
	\vspace{-1ex} \begin{equation}\begin{bmatrix} 
			u_1 \\ u_2 
		\end{bmatrix}=K\begin{bmatrix} 
			y_1 \\ y_2 
		\end{bmatrix},\label{K law}\end{equation} where ${K\in \mathbb{R}^{(m_1+m_2)\times(p_1+p_2)}}$. We call such an interconnection a \emph{$K$-interconnection}. 
	Let $q:=p_1+p_2$. A natural choice for a behavior of the interconnection is the following:

Let $\mathcal{B}_1,\mathcal{B}_2 \in \mathcal{L}^q$ and ${\Phi_1, \Phi_2 \in \mathbb{R}_s^{q \times q}[\zeta,\eta]}$. 		\vspace{-1ex}

		\item $\mathcal{B}$ is asymptotically stable if $Q_{\Psi_1}$ satisfies the dissipation inequality strictly. \label{second stat first diss inter thm} 
	\end{enumerate}

\label{ex: arbitrary sr} 
Assume that $K_{12}$ has full column rank. 
For $i=1,2$, let \[{\bar{\Phi}_i(\zeta,\eta)=X^i=\begin{bmatrix} 
		X_{11}^i & X_{12}^i \\ X_{21}^i & X_{22}^i 
\end{bmatrix}},\] 
where $X_{11}^i\in \mathbb{R}^{m_i \times m_i}$, $X_{12}^i\in \mathbb{R}^{m_i \times p_i}$, $X_{21}^i\in \mathbb{R}^{p_i \times m_i}$, and $X_{22}^i\in \mathbb{R}^{p_i \times p_i}$. Suppose that there exists a storage function $Q_{\bar{\Psi}_i}$ for $(\bar{\mathcal{B}}_i,Q_{\bar{\Phi}_i})$, for $i=1,2$, such that $Q_{\bar{\Psi}_1}\stackrel{\bar{\mathcal{B}}_1}{>}0$ and  $Q_{\bar{\Psi}_2}\stackrel{\bar{\mathcal{B}}_2}{\geq}0$. Moreover, in line with \cite{arcak_stability_2016}, assume that there exist $\alpha_1>0$ and $\alpha_2\geq 0$ such that $\Xi(\alpha_1,\alpha_2)\leq0$, where $\Xi(\alpha_1,\alpha_2)$ is defined as 
	\begin{equation} 
		\begin{bmatrix} 
			K \\ I 
		\end{bmatrix}^\top \begin{bmatrix} 
			\alpha_1X^{11}_1 & 0  &\alpha_1X_1^{12} & 0\\ 
			0 & \alpha_2 X_2^{11} & 0 & \alpha_2X_2^{12}\\ 
			\alpha_1X^{21}_1 & 0  &\alpha_1X_1^{22} & 0\\ 
			0 & \alpha_2 X_2^{21} & 0 & \alpha_2X_2^{22} 
		\end{bmatrix}\begin{bmatrix} 
			K \\ I 
		\end{bmatrix}. 
		\label{Xi def} 
	\end{equation} 
	For $i=1,2$, define 
	\begin{equation} 
		\Phi_i(\zeta,\eta):=K_i^\top \bar{\Phi}_i(\zeta,\eta)K_i, \;\; \Psi_i(\zeta,\eta):=K_i^\top \bar{\Psi}(\zeta,\eta)K_i, 
		\label{bar Phi and bar Psi} 
	\end{equation} 
	where $K_1, K_2$ are as in \eqref{K1} and \eqref{K2}, respectively. 
	Then, by Lemma~\ref{lemma barB dissipativity} parts \ref{lemma stat 1} and \ref{lemma stat 3}, $Q_{\Psi_i}$ is a storage function for $(\mathcal{B}_i,Q_{\Phi_i})$ such that $Q_{\Psi_i}\stackrel{\mathcal{B}_i}{\geq}0$. Moreover, as $K_{12}$ has full column rank, also $K_1$ 
	has full column rank. Then, by Lemma~\ref{lemma barB dissipativity} part \ref{lemma stat 4}, $Q_{\Psi_1}\stackrel{\mathcal{B}_1}{>}0$.  Moreover, for $\Xi$ as in \eqref{Xi def} and all $w\in \left(\mathbb{R}^q\right)^\mathbb{R}$, we have that  
	\begin{equation*} 
		\alpha_1Q_{\Phi_1}(w)+\alpha_2Q_{\Phi_2}(w)= w^\top \Xi(\alpha_1,\alpha_2) w\leq 0. 
	\end{equation*} 
	Hence, by Theorem~\ref{t: first diss inter thm}, the interconnected behavior $\mathcal{B}$ is stable. 
	 
	Therefore, Theorem~\ref{t: first diss inter thm} allows us to show a similar result to the generalized interconnection result for networks of dissipative systems with static supply rates in \cite[Proposition~2.1]{arcak_stability_2016}, adapted to our behavioral setting and applied to an interconnection of two LTI systems.  
	 
	Unlike in \cite[Proposition~2.1]{arcak_stability_2016}, we do not require that the interconnection is well-posed or that both of the storage functions are positive. However, we add an extra rank assumption on the $K$ matrix. We note that the full column rank assumption on $K_{12}$ is not too restrictive, as common examples of $K$-interconnections like positive/negative feedback (with $K=K_+$ or $K_-$ in \eqref{feedback Ks}) satisfy it. 
	 
	Lastly, we note that with this extra assumption, one can similarly show that, if in addition, $Q_{\Psi_1}$ satisfies the dissipation inequality strictly, the interconnected behavior $\mathcal{B}$ will be asymptotically stable.

		-\delta^2 I & 0 \\0 & \hat{\gamma}_1^2 I\\ 
	\end{bmatrix}.\]

	\[\frac{d}{dt}Q_{\Psi_1} \stackrel{\mathcal{B}_1}{\leq} 
	Q_{\Phi_1}< Q_{\hat{\Phi}_1},\] 
	as for all $w=(u,y)\in \left(\mathbb{R}^q\right)^\mathbb{R} $, 
	\[Q_{\Phi_1}(w)=\gamma_1^2 \|y\|^2 - \|u\|^2\leq \hat{\gamma_1}^2\|y\|^2-\delta^2 \|u\|^2 = Q_{\Phi_1}(w),\] 
	and $Q_{\Phi_1}(w) = Q_{\hat{\Phi}_1}(w)$ if and only if $w=0$.

		\alpha_1 Q_{\hat{\Phi}_1}(w)+\alpha_2Q_{\Phi_2}(w) =  
		                                                                                                                                                                         
		w^\top \begin{bmatrix} 
			\left( \hat{\gamma}_1^2\gamma_2^2-\delta^2\right) I & 0 \\ 0 & 0 
		\end{bmatrix}w \leq 0. 
	\end{gather*}

\label{r: another behavioral} 
	In \cite{takaba_stability_2005}, the stability of an interconnection of two controllable, dissipative behaviors was studied. Our theorem is also applicable to this case. 
	 
	Let $\mathcal{B}_1,\mathcal{B}_2 \in \mathcal{L}^q$ be controllable and consider ${\Phi \in \mathbb{R}_s^{q \times q}[\zeta,\eta]}$. Suppose that there exists $\varepsilon>0$ such that 
	\begin{equation} 
		\int_{-\infty}^0 \left(Q_{\Phi}(w)\right)(\tau) d\tau\geq \epsilon \int_{-\infty}^0 \|w(\tau)\|^2 d\tau, 
		\label{other def strict diss} 
	\end{equation} 
	for all $w \in \mathcal{B}_1 \cap \mathcal{D}(\mathbb{R}, \mathbb{R}^q)$, and that $\mathcal{B}_2$ is half-line dissipative with respect to $Q_{-\Phi}$. <!-- theorem-4: body_md -->
Let $\mathcal{B}_1,\mathcal{B}_2 \in \mathcal{L}^q$ and ${\Phi_1, \Phi_2 \in \mathbb{R}_s^{q \times q}[\zeta,\eta]}$.  
	Suppose there exists a storage function $Q_{\Psi_i}$ for $(\mathcal{B}_i, Q_{\Phi_i})$, for $i=1,2$. Define $\mathcal{B} := \mathcal{B}_1 \cap \mathcal{B}_2$ and assume that there exist scalars $\alpha_1>0$ and $\alpha_2\geq 0$, and $\Gamma\in \mathbb{R}_s^{q \times q}[\zeta,\eta]$ such that  
	\begin{equation} 
		\frac{d}{dt}Q_\Gamma \stackrel{\mathcal{B}}{\geq}\alpha_1Q_{\Phi_1}+\alpha_2Q_{\Phi_2}. 
		\label{main thm cond 1} 
		\vspace{-1ex} 
	\end{equation} 
	Then, 
	\begin{enumerate}[label=\roman*)] 
		\item $\mathcal{B}$ is stable if 
		\vspace{-1.5ex} 
		\begin{equation} 
			\alpha_1Q_{\Psi_1}+\alpha_2Q_{\Psi_2}-Q_\Gamma \stackrel{\mathcal{B}}{>} 0; 
			\label{main thm cond strict} 
		\end{equation} \label{main thm stat 1} 
		\vspace{-2ex} 
		\item $\mathcal{B}$ is asymptotically stable if $Q_{\Psi_1}$ satisfies the dissipation inequality strictly and 
		\vspace{-1.5ex} 
		\begin{equation} 
			\alpha_1Q_{\Psi_1}+\alpha_2Q_{\Psi_2}-Q_\Gamma \stackrel{\mathcal{B}}{\geq} 0. 
			\label{main thm cond non strict} 
		\end{equation}\label{main thm stat 2} 
	\end{enumerate}  
	\label{t: main diss inter thm}

<!-- proof-1: body_md -->
First, we show \ref{main thm stat 1}. Consider the candidate Lyapunov function 
	\begin{equation} 
		Q_\Psi:= \alpha_1 Q_{\Psi_1}+\alpha_2 Q_{\Psi_2}-Q_\Gamma. 
		\label{main thm lyap} 
	\end{equation} 
	By assumption, we have that $Q_\Psi \stackrel{\mathcal{B}}{>}0$. Moreover,  
	\begin{equation*} 
		\begin{aligned} 
			\frac{d}{dt}Q_\Psi &= \alpha_1\frac{d}{dt}Q_{\Psi_1}+\alpha_2 \frac{d}{dt}Q_{\Psi_2}  -\frac{d}{dt}Q_\Gamma \\ 
			&\stackrel{\mathcal{B}}{\leq} \alpha_1 Q_{\Phi_1} + \alpha_2 Q_{\Phi_2}-\frac{d}{dt}Q_\Gamma \stackrel{\mathcal{B}}{\leq} 0. 
		\end{aligned} 
	\end{equation*} 
	Thus, by Theorem~\ref{t: lyap stab}, $\mathcal{B}$ is stable. 
	 
	Now we show \ref{main thm stat 2}. By assumption, the candidate Lyapunov function \eqref{main thm lyap} satisfies $Q_\Psi \stackrel{\mathcal{B}}{\geq }0$. Moreover,  
	\begin{equation*} 
		\begin{aligned} 
			\frac{d}{dt}Q_\Psi &= \alpha_1\frac{d}{dt}Q_{\Psi_1}+\alpha_2 \frac{d}{dt}Q_{\Psi_2}  -\frac{d}{dt}Q_\Gamma \\ 
			&\stackrel{\mathcal{B}}{<} \alpha_1 Q_{\Phi_1} + \alpha_2 Q_{\Phi_2}  -\frac{d}{dt}Q_\Gamma \stackrel{\mathcal{B}}{\leq} 0, 
		\end{aligned} 
	\end{equation*} 
	showing asymptotic stability by Theorem~\ref{t: lyap stab}.

<!-- lemma-1: body_md -->
Let $\bar{\mathcal{B}} \in \mathcal{L}^q$ and $\bar{\Phi} \in \mathbb{R}_s^{q \times q}[\zeta,\eta]$. Suppose that $Q_{\bar{\Psi}}$ is a storage function for $(\bar{\mathcal{B}},Q_{\bar{\Phi}})$. Let $N\in \mathbb{R}^{r \times q}$ and define $\mathcal{B}:=\left\{ w \;\middle| \; Nw \in \bar{\mathcal{B}} \right\}$. Moreover, define 
	\vspace{-0.5ex} 
	\begin{equation*} 
		\Phi(\zeta,\eta) := N^\top \bar{\Phi}(\zeta,\eta)N \quad \text{and} \quad \Psi(\zeta,\eta) := N^\top \bar{\Psi}(\zeta,\eta)N. 
	\end{equation*} 
	The following statements hold. 
	\begin{enumerate}[label=\roman*)] 
		\item The QDF $Q_\Psi$ is a storage function for $(\mathcal{B},Q_{\Phi})$. \label{lemma stat 1} 
		\item If $N$ has full column rank, then  
		\[\frac{d}{dt}Q_{\bar{\Psi}} \stackrel{\bar{\mathcal{B}}}{<}Q_{\bar{\Phi}} \implies \frac{d}{dt}Q_\Psi \stackrel{\mathcal{B}}{<}Q_\Phi. \vspace{-1.5ex}\] \label{lemma stat 2} 
		\item If $Q_{\bar{\Psi}} \stackrel{\bar{\mathcal{B}}}{\geq}0$, then $Q_\Psi \stackrel{\mathcal{B}}{\geq}0$. \label{lemma stat 3} 
		\item If $Q_{\bar{\Psi}} \stackrel{\bar{\mathcal{B}}}{>}0$ and $N$ has full column rank, then $Q_\Psi \stackrel{\mathcal{B}}{>}0$. \label{lemma stat 4} 
	\end{enumerate} 
	\label{lemma barB dissipativity}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
First, we show \ref{lemma stat 1}. We have that  
	\begin{equation}\frac{d}{dt}Q_{\bar{\Psi}} \stackrel{\bar{\mathcal{B}}}{\leq}Q_{\bar{\Phi}}.\label{lemma p diss ineq}\end{equation} Then, for all $w\in \mathcal{B}$ we have that  
	\begin{equation}\frac{d}{dt}Q_{\Psi}(w) =\frac{d}{dt}Q_{\bar{\Psi}}(Nw) \leq Q_{\bar{\Phi}}(Nw)=Q_\Phi(w). \label{bar B storage lemma ineq}\end{equation} 
	Hence, $Q_{\Psi}$ is a storage function for $(\mathcal{B},Q_{\Phi})$. 
	 
	Now, we show \ref{lemma stat 2}. Let $w\in \mathcal{B}$. As $Q_{\bar{\Psi}}$ satisfies the dissipation inequality \eqref{lemma p diss ineq} strictly, \eqref{bar B storage lemma ineq} holds and 
	\begin{equation*}\frac{d}{dt}Q_{\Psi}(w) =Q_{\Phi}(w)\end{equation*}  
	if and only if $Nw=0$. As $N$ has full column rank, this is equivalent to $w=0$. Hence, $Q_{\Psi}$ satisfies  
	\begin{equation*}\frac{d}{dt}Q_{\Psi} \stackrel{\mathcal{B}}{<}Q_{\Phi}. \vspace{-0.5ex}\end{equation*} 
	 
	To show \ref{lemma stat 3} suppose that $Q_{\bar{\Psi}} \stackrel{\bar{\mathcal{B}}}{\geq}0$. Then, for all $w\in \mathcal{B}$, $Q_{\Psi}(w)=Q_{\bar{\Psi}}(Nw)\geq 0$. Hence, $Q_{\Psi} \stackrel{\mathcal{B}}{\geq}0$.  
	 
	Lastly, suppose that $Q_{\bar{\Psi}} \stackrel{\bar{\mathcal{B}}}{>}0$ and that $N$ has full column rank. Let $w\in \mathcal{B}$. Then, as was shown for \ref{lemma stat 3}, $Q_{\Psi}\stackrel{\mathcal{B}}{\geq}0$, and $Q_{\Psi}(w)=0$ if and only if $Nw=0$, which is equivalent to $w=0$. Hence, $Q_{\Psi} \stackrel{\mathcal{B}}{>}0$. This shows \ref{lemma stat 4}.
