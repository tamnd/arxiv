<!-- definition-1: body_md -->
\label{def:solution}
Let $T>0$, $u_0\in H^{-3/4}(\mathbb R^+)$, $f\in H^{1/12}(\mathbb R^+)$. A function $u$ defined on $\mathbb R\times[-T,T]$ is a \emph{distributional solution} of \eqref{IBVP} on $[0,T]$ if
\begin{enumerate}
\item[(a)] (\emph{well-defined nonlinearity}) $u\in X_T$; by Proposition \ref{prop:kishimoto} below, $\partial_x(u^2)$ is then a well-defined distribution on $\mathbb R\times(-T,T)$;
\item[(b)] $\partial_t u+\partial_x^3u+\partial_x(u^2)=0$ in the sense of distributions on $(0,+\infty)\times(0,T)$;
\item[(c)] (\emph{space traces}) $u\in C([-T,T];H^{-3/4}(\mathbb R))$ and $u(\cdot,0)|_{\mathbb R^+}=u_0$ in $H^{-3/4}(\mathbb R^+)$;
\item[(d)] (\emph{time traces}) $u\in C(\mathbb R_x;H^{1/12}(0,T))$ and $u(0,\cdot)=f$ in $H^{1/12}(0,T)$.
\end{enumerate}
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Let $u_0\in H^{-3/4}(\mathbb R^+)$ and $f\in H^{1/12}(\mathbb R^+)$. There exist $T>0$, depending only on $\|u_0\|_{H^{-3/4}(\mathbb R^+)}+\|f\|_{H^{1/12}(\mathbb R^+)}$, and a distributional solution $u$ of \eqref{IBVP} on $[0,T]$ in the sense of Definition \ref{def:solution}. The map $(u_0,f)\mapsto u$ is Lipschitz (indeed analytic) from a neighbourhood of the data in $H^{-3/4}(\mathbb R^+)\times H^{1/12}(\mathbb R^+)$ into $X_T\cap C([0,T];H^{-3/4}(\mathbb R))\cap C(\mathbb R_x;H^{1/12}(0,T))$.
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:JK}
If $-\frac12<\alpha<\frac12$, then $\|\chi_{(0,+\infty)}g\|_{H^\alpha(\mathbb R)}\le c(\alpha)\,\|g\|_{H^\alpha(\mathbb R)}$ for all $g\in H^\alpha(\mathbb R)$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:cutoff}
Let $\alpha\ge0$ and $\theta\in C_c^\infty(\mathbb R)$. Then $\|\theta g\|_{H^\alpha(\mathbb R)}\le C\|\theta\|_{H^{\alpha+1}(\mathbb R)}\|g\|_{H^\alpha(\mathbb R)}$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
$\widehat{\theta g}=(2\pi)^{-1}\widehat\theta*\widehat g$ and $\langle\tau\rangle^\alpha\le\langle\tau-\tau'\rangle^\alpha\langle\tau'\rangle^\alpha$, so $\|\theta g\|_{H^\alpha}\le(2\pi)^{-1}\|\langle\tau\rangle^\alpha\widehat\theta\|_{L^1}\|g\|_{H^\alpha}$, and $\|\langle\tau\rangle^\alpha\widehat\theta\|_{L^1}\le\|\langle\tau\rangle^{-1}\|_{L^2}\|\theta\|_{H^{\alpha+1}}$.
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
\label{lem:embedding}
$\|u\|_{L^\infty_tH^s_x}\le(2\pi)^{-1}\|\langle\xi\rangle^s\widehat u\|_{L^2_\xi L^1_\tau}\le C\|u\|_{X^{s,\frac12,1}}$, and $X^{s,\frac12,1}\hookrightarrow C(\mathbb R_t;H^s(\mathbb R_x))$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
The first inequality is Minkowski's. For the second, by Cauchy--Schwarz in $\tau$ on each $B_k$, $\|\widehat u(\xi,\cdot)\|_{L^1_\tau(B_k)}\le 2^{(k+1)/2}\|\widehat u(\xi,\cdot)\|_{L^2_\tau(B_k)}$; summing in $k$ and taking the $L^2_\xi(A_j)$ norm with Minkowski's inequality gives $$\|\langle\xi\rangle^s\widehat u\|_{L^2_\xi(A_j)L^1_\tau}\le\sqrt2\sum_k 2^{k/2}\|\langle\xi\rangle^s\widehat u\|_{L^2(A_j\cap B_k)};$$ the $\ell^2_j$ sum gives the claim. Continuity in $t$ follows by density of $\mathcal S$.
<!-- end proof-2 -->
<!-- proposition-1: body_md -->
\label{prop:kishimoto}
There is $C>0$ such that for all $u,v\in X$,
$$\|\partial_x(uv)\|_{\mathcal N}\le C\|u\|_X\|v\|_X .$$
In particular the bilinear map $(u,v)\mapsto\partial_x(uv)$, defined on $\mathcal S(\mathbb R^2)$, extends continuously to $X\times X$ with values in the space of tempered distributions $G$ with $\|G\|_{\mathcal N}<\infty$.
<!-- end proposition-1 -->
<!-- lemma-4: body_md -->
\label{lem:kishimoto-linear}
For $0<T\le1$, with constants independent of $T$,
\begin{align}
\|e^{-t\partial_x^3}u_0\|_{X_T}+\sup_{|t|\le T}\|e^{-t\partial_x^3}u_0\|_{H^{-3/4}}&\lesssim\|u_0\|_{H^{-3/4}},\label{eq:K41}\\
\|\mathcal{K} G\|_{X_T}+\sup_{|t|\le T}\|\mathcal{K} G(t)\|_{H^{-3/4}}&\lesssim\|G\|_{\mathcal N}.\label{eq:K42}
\end{align}
<!-- end lemma-4 -->
<!-- lemma-5: body_md -->
\label{lem:KPV}
Let $s\in\mathbb R$, $\gamma=\frac{s+1}3\in[0,\frac12)$, and $\theta\in C_c^\infty(\mathbb R)$ with $\operatorname{supp}\theta\subset[-2,2]$. Then
$$\sup_{x\in\mathbb R}\|\theta(t)e^{-t\partial_x^3}\phi(x,\cdot)\|_{H^\gamma(\mathbb R_t)}\le C\|\theta\|_{H^1(\mathbb R)}\|\phi\|_{H^s(\mathbb R)},$$
and $x\mapsto\theta(t)e^{-t\partial_x^3}\phi(x,\cdot)$ is continuous into $H^\gamma(\mathbb R_t)$.
<!-- end lemma-5 -->
<!-- lemma-6: body_md -->
\label{lem:trace}
Let $-1<s<\frac12$, $\gamma=\frac{s+1}3$, and let $\|\cdot\|_{\mathcal N_s}$ be defined as in \eqref{eq:Nnorm} with $-\frac34$ replaced by $s$ in \eqref{eq:X}. Then for every $G$ with $\|G\|_{\mathcal N_s}<\infty$,
$$\sup_{x\in\mathbb R}\|\psi(t)\mathcal{K} G(x,\cdot)\|_{H^\gamma(\mathbb R_t)}\le C\|G\|_{\mathcal N_s},$$
and $x\mapsto\psi(t)\mathcal{K} G(x,\cdot)$ is continuous into $H^\gamma(\mathbb R_t)$.
<!-- end lemma-6 -->
<!-- proof-3: body_md -->
Set $W:=\mathcal{F}^{-1}[\langle\sigma\rangle^{-1}\widehat G]$, so that $\|G\|_{\mathcal N_s}=\|W\|_{X}+\|W\|_{Y}$ (with $s$ in place of $-\frac34$). By density it suffices to prove the bound for $G\in\mathcal S(\mathbb R^2)$; the continuity in $x$ then follows from the uniform bound and the explicit formulas below. Let $\psi_0\in C_c^\infty(\mathbb R)$ with $\psi_0=1$ on $[-1,1]$ and $\operatorname{supp}\psi_0\subset[-2,2]$, applied to the variable $\sigma$. Taking the Fourier transform in $x$,
$$\mathcal{F}_x {\mathcal{K} G}(\xi,t)=\frac{1}{2\pi}\, e^{it\xi^3}\int_\mathbb R\frac{e^{it\sigma}-1}{i\sigma}\,\widehat G(\xi,\tau)\,d\tau ,$$
and we decompose $\mathcal{K} G=I+II+III$ with
\begin{align*}
I&=\frac{1}{(2\pi)^2}\iint e^{ix\xi+it\xi^3}\frac{e^{it\sigma}-1}{i\sigma}\psi_0(\sigma)\widehat G\,d\tau d\xi
=\sum_{k\ge1}\frac{i^{k-1}t^k}{k!}\,e^{-t\partial_x^3}\phi_k,\\ 
&\quad \quad \widehat{\phi_k}(\xi)=\frac1{2\pi}\int\sigma^{k-1}\psi_0(\sigma)\widehat G\,d\tau,\\
&II=\frac{1}{(2\pi)^2}\iint e^{ix\xi+it\tau}\frac{1-\psi_0(\sigma)}{i\sigma}\widehat G\,d\tau d\xi,\\
&III=-\,e^{-t\partial_x^3}\phi_3,\qquad \widehat{\phi_3}(\xi)=\frac1{2\pi}\int\frac{1-\psi_0(\sigma)}{i\sigma}\widehat G\,d\tau .
\end{align*}

\emph{Term $I$.} By Lemma \ref{lem:KPV} with $\theta_k=t^k\psi$, $\|\theta_k\|_{H^1}\le C^k$, we get $$\|\psi I(x,\cdot)\|_{H^\gamma}\le\sum_k\frac{C^k}{k!}\|\phi_k\|_{H^s},$$ and since $|\sigma^{k-1}\psi_0(\sigma)|\le 2^{k-1}\langle\sigma\rangle^{-1}\cdot\sqrt5$,
$$\|\phi_k\|_{H^s}\le C2^{k}\big\|\langle\xi\rangle^s\langle\sigma\rangle^{-1}\widehat G\big\|_{L^2_\xi L^1_\tau}=C2^k\|W\|_Y .$$

\emph{Term $III$.} Since $(1-\psi_0(\sigma))/|\sigma|\le\sqrt2\langle\sigma\rangle^{-1}$, $\|\phi_3\|_{H^s}\le C\|W\|_Y$, and Lemma \ref{lem:KPV} gives $\|\psi\, III(x,\cdot)\|_{H^\gamma}\le C\|W\|_Y$.

\emph{Term $II$.} By Lemma \ref{lem:cutoff} it suffices to bound $\|II(x,\cdot)\|_{H^\gamma}$. The time Fourier transform of $II(x,\cdot)$ is $(2\pi)^{-1}F_x(\tau)$, where
$$F_x(\tau):=\int_\mathbb R e^{ix\xi}\frac{1-\psi_0(\sigma)}{i\sigma}\widehat G(\xi,\tau)\,d\xi .$$
All bounds below are in terms of $|\widehat G|$, hence uniform in $x$; we drop $e^{ix\xi}$. Set
$$D_\tau:=\int_\mathbb R|\widehat G(\xi,\tau)|^2\langle\xi\rangle^{2s}\langle\sigma\rangle^{-1}d\xi,\qquad \int_\mathbb R D_\tau\,d\tau\le 2\|W\|^2_{X^{s,\frac12}}\le2\|W\|_X^2 .$$

\emph{(i) $|\tau|\le8$.} By Cauchy--Schwarz, $|F_x(\tau)|^2\le D_\tau\int\langle\xi\rangle^{-2s}\langle\sigma\rangle^{-1}(1-\psi_0(\sigma))^2d\xi$. For $|\xi|\le4$ the last integrand is bounded, and for $|\xi|\ge4$ one has $|\sigma|\ge|\xi|^3-8\ge\frac12|\xi|^3$, so the integral is $\le C\int_{|\xi|\ge4}|\xi|^{-2s-3}d\xi+C\le C$ because $s>-1$. Hence $\int_{|\tau|\le8}\langle\tau\rangle^{2\gamma}|F_x|^2d\tau\le C\|W\|_X^2$.

\emph{(ii) $|\tau|\ge8$.} Let $r=r(\tau)$ be the real cube root of $\tau$, so $|r|\ge2$, and split $\mathbb R_\xi=R_1\cup R_2\cup R_3$,
$$R_1=\{|\xi-r|\le\tfrac12|r|\},\qquad R_3=\{|\xi|\ge\tfrac32|r|\},\qquad R_2=\mathbb R\setminus(R_1\cup R_3).$$
Note that $\xi\in R_1$ means that $\xi$ has the sign of $r$ and $\frac12|r|\le|\xi|\le\frac32|r|$. Hence a point of $R_2$ either satisfies $|\xi|<\frac12|r|$, whence $|\xi^3|<\frac18|\tau|$ and $|\sigma|=|\tau-\xi^3|\ge\frac78|\tau|$, or has the sign opposite to $r$ (otherwise, having $\frac12|r|\le|\xi|<\frac32|r|$, it would lie in $R_1$), whence $\tau$ and $\xi^3$ have opposite signs and $|\sigma|=|\tau|+|\xi|^3\ge|\tau|$. In both cases $|\sigma|\ge\frac78|\tau|$ on $R_2$. On $R_3$, $|\sigma|\ge|\xi|^3-|\tau|\ge\frac{19}{27}|\xi|^3$. On $R_1$, $\frac12|r|\le|\xi|\le\frac32|r|$, hence $\langle\xi\rangle\sim|r|=|\tau|^{1/3}$ and $|\partial_\xi\sigma|=3\xi^2\ge\frac34r^2$.

On $R_2$, by Cauchy--Schwarz and $s<\frac12$,
$$|F_{x,R_2}(\tau)|^2\le D_\tau\,\frac{C}{|\tau|}\int_{|\xi|\le\frac32|r|}\langle\xi\rangle^{-2s}d\xi\le C D_\tau|\tau|^{-1}|\tau|^{\frac{1-2s}3}=CD_\tau\langle\tau\rangle^{-2\gamma},$$
since $-1+\frac{1-2s}3=-\frac{2s+2}3=-2\gamma$. On $R_3$, by Cauchy--Schwarz and $s>-1$,
$$|F_{x,R_3}(\tau)|^2\le D_\tau\, C\int_{|\xi|\ge\frac32|r|}|\xi|^{-2s-3}d\xi\le CD_\tau|r|^{-2s-2}=CD_\tau\langle\tau\rangle^{-2\gamma}.$$
Integrating in $\tau$, the contributions of $R_2$ and $R_3$ to $\|\langle\tau\rangle^\gamma F_x\|_{L^2}^2$ are $\le C\|W\|_X^2$.

On $R_1$ we use the Besov structure. For $k\ge0$ let $E_k(\tau)=\{\xi\in R_1:2^k\le\langle\sigma\rangle<2^{k+1}\}$, and let $|E_k(\tau)|$ denote its Lebesgue measure in $\xi$. We claim that
\begin{equation}\label{eq:Ek}
|E_k(\tau)|\le \tfrac{16}{3}\,2^{k}|\tau|^{-2/3}.
\end{equation}
Indeed, the condition $2^k\le\langle\sigma\rangle<2^{k+1}$ means $$\sigma\in J_k:=\{\sigma\in\mathbb R: 2^{2k}-1\le\sigma^2<2^{2k+2}-1\},$$ which is the union of two intervals (one in $\sigma>0$, one in $\sigma<0$) each of length at most $2^{k+1}$, so $|J_k|\le2^{k+2}$. On $R_1$ the map $\xi\mapsto\sigma(\xi)=\tau-\xi^3$ is strictly monotone ($\xi$ has the sign of $r$ and $|\xi|\ge\frac12|r|$), with $|\sigma'(\xi)|=3\xi^2\ge\frac34r^2=\frac34|\tau|^{2/3}$. Hence, changing variables,
$$
|E_k(\tau)|=\int_{E_k(\tau)}d\xi=\int_{\sigma(E_k(\tau))}\frac{d\sigma}{|\sigma'(\xi(\sigma))|}\le\frac{|J_k|}{\frac34|\tau|^{2/3}}\le\frac{2^{k+2}}{\frac34|\tau|^{2/3}},
$$
which is \eqref{eq:Ek}. (Geometrically, $E_k(\tau)$ consists of two intervals on either side of $r$, at distance $\approx2^k/(3r^2)$ from $r$ and of width $\approx2^k/(3r^2)$: the shell $B_k$ cut at height $\tau$. It is empty when $2^k\gg|\tau|$, since $|\sigma|\le|\tau|+|\xi|^3\le\frac{35}{8}|\tau|$ on $R_1$.) Moreover $\langle\xi\rangle^{-2s}\le C|\tau|^{-2s/3}$ on $R_1$, because $\frac12|r|\le|\xi|\le\frac32|r|$ and $|r|\ge2$. Hence, by Cauchy--Schwarz and \eqref{eq:Ek},
\begin{equation}
\begin{split}
\Big|\int_{E_k(\tau)}\frac{1-\psi_0(\sigma)}{i\sigma}\widehat G\,d\xi\Big|&\le 2^{1-k}\|\langle\xi\rangle^s\widehat G(\cdot,\tau)\|_{L^2(E_k(\tau))}\Big(\int_{E_k(\tau)}\langle\xi\rangle^{-2s}d\xi\Big)^{1/2}
\\&\le C2^{-k/2}|\tau|^{-\frac{s+1}3}\|\langle\xi\rangle^s\widehat G(\cdot,\tau)\|_{L^2(E_k(\tau))}.
\end{split}
\end{equation}
Let $j(\tau)$ be defined by $2^{j(\tau)}\le\langle r\rangle<2^{j(\tau)+1}$; then $$E_k(\tau)\times\{\tau\}\subset\bigcup_{|j-j(\tau)|\le2}(A_j\cap B_k),$$ and since $|\xi|\ge1$ on $R_1$, these sets lie outside $D$. Therefore
$$\langle\tau\rangle^\gamma|F_{x,R_1}(\tau)|\le C\sum_{|j-j(\tau)|\le2}\ \sum_{k\ge0}2^{-k/2}\|\langle\xi\rangle^s\widehat G(\cdot,\tau)\|_{L^2_\xi(A_j\cap B_k)} .$$
Fix $j_0\ge0$. Taking the $L^2_\tau$ norm over $\{\tau:|\tau|\ge8,\ j(\tau)=j_0\}$ and using Minkowski's inequality,
$$\big\|\langle\tau\rangle^\gamma F_{x,R_1}\big\|_{L^2(\{j(\tau)=j_0\})}\le C\sum_{|j-j_0|\le2}\sum_{k\ge0}2^{-k/2}\|\langle\xi\rangle^s\widehat G\|_{L^2_{\tau,\xi}(A_j\cap B_k)}\le C\sum_{|j-j_0|\le2}a_j,$$
where $a_j:=\sum_k2^{k/2}\|\langle\xi\rangle^s\mathbf 1_{\mathbb R^2\setminus D}\widehat W\|_{L^2(A_j\cap B_k)}$; here we used $\langle\sigma\rangle\le2^{k+1}$ on $B_k$, so that $2^{-k/2}|\widehat G|\le 2\cdot 2^{k/2}\langle\sigma\rangle^{-1}|\widehat G|=2\cdot2^{k/2}|\widehat W|$. Summing the squares over $j_0$,
$$\big\|\langle\tau\rangle^\gamma F_{x,R_1}\big\|_{L^2(|\tau|\ge8)}^2\le C\sum_{j_0}\Big(\sum_{|j-j_0|\le2}a_j\Big)^2\le C\sum_ja_j^2\le C\|W\|_X^2 .$$
Collecting (i) and (ii), $\|II(x,\cdot)\|_{H^\gamma}\le C\|W\|_X$ uniformly in $x$, which completes the proof.

<!-- lemma-7: body_md -->
\label{lem:glue}
Let $m\ge0$, $n\ge m+2$, and $c=-i\omega^{-m}(\omega+i)^n$. There is a unique polynomial $P$ of degree $\le n-1$ such that $g$ in \eqref{eq:ext} is of class $C^{n-m-1}$ on $\mathbb R$ and has vanishing moments $\int_\mathbb R y^jg(y)\,dy=0$ for $0\le j\le m-1$. For this $P$,
\begin{equation}\label{eq:ghat}
\widehat g(\eta)=\int_\mathbb R e^{-iy\eta}g(y)\,dy=\frac{a(\eta)}{\eta-\omega},\qquad a(\eta)=c\,\frac{\eta^m}{(\eta+i)^n},
\end{equation}
and $P$ is determined by $\dfrac{Q(\eta)}{(\eta+i)^n}:=\dfrac{c\,\eta^m+i(\eta+i)^n}{(\eta-\omega)(\eta+i)^n}=\displaystyle\sum_{j=0}^{n-1}(-1)^jj!\,i^{j+1}\frac{p_j}{(\eta+i)^{j+1}}$, where $P(y)=\sum_{j=0}^{n-1}p_jy^j$.
<!-- end lemma-7 -->
<!-- proof-4: body_md -->
Since $\operatorname{Im}\omega>0$, $\int_0^\infty e^{-iy\eta}e^{i\omega y}dy=\frac{-i}{\eta-\omega}$. For $j\ge0$, substituting $y=-z$ and using $1-i\eta=-i(\eta+i)$,
$$\int_{-\infty}^0e^{-iy\eta}y^je^{y}dy=\int_0^\infty e^{-(1-i\eta)z}(-z)^jdz=\frac{(-1)^jj!}{(1-i\eta)^{j+1}}=\frac{(-1)^jj!\,i^{j+1}}{(\eta+i)^{j+1}} .$$
Hence, for any $P$ of degree $\le n-1$, $\widehat g(\eta)=\frac{-i}{\eta-\omega}+\frac{Q(\eta)}{(\eta+i)^n}$ with $Q$ a polynomial of degree $\le n-1$, and conversely every such $Q$ arises from exactly one $P$ (expand $Q$ in powers of $\eta+i$). Thus
$$\widehat g(\eta)=\frac{N(\eta)}{(\eta-\omega)(\eta+i)^n},\qquad N(\eta):=-i(\eta+i)^n+Q(\eta)(\eta-\omega),\quad \deg N\le n .$$
Now $g$ is bounded with compactly supported singularities, so $g\in C^{n-m-1}$ with piecewise smooth pieces is equivalent to $\widehat g(\eta)=O(|\eta|^{-(n-m+1)})$ at infinity, i.e.\ to $\deg N\le m$; and $\int y^jg=0$ for $j<m$ is equivalent to $\widehat g^{(j)}(0)=0$ for $j<m$, i.e.\ to $\eta^m\mid N$. Both together mean $N(\eta)=c'\eta^m$ for a constant $c'$, and evaluating at $\eta=\omega$ (where $N(\omega)=-i(\omega+i)^n$) gives $c'=c$. Conversely, given $c$, $Q(\eta)=\big(c\eta^m+i(\eta+i)^n\big)/(\eta-\omega)$ is a polynomial of degree $n-1$ precisely because the numerator vanishes at $\omega$; this determines $Q$, hence $P$, uniquely.

<!-- lemma-8: body_md -->
\label{lem:algebra}
Let $\tau\in\mathbb R\setminus\{0\}$.
\begin{enumerate}
\item[(i)] $\tau-\xi^3=\rho(\tau)^3(1-\zeta^3)=-\rho(\tau)^3(\zeta-1)(\zeta-\omega)(\zeta-\bar\omega)$ for all $\xi\in\mathbb{C}$.
\item[(ii)] As $\xi$ runs over $\mathbb R$, $\zeta$ runs over the line $\Gamma_\tau:=e^{-i\theta_\tau}\mathbb R$, i.e.\ $\Gamma_\tau=\mathbb R$ if $\tau>0$ and $\Gamma_\tau=e^{i\pi/3}\mathbb R$ if $\tau<0$, and $d\xi=\rho(\tau)\,d\zeta$. As $\xi$ runs over the open upper half-plane, $\zeta$ runs over the open half-plane $H_\tau$ to the left of $\Gamma_\tau$ (oriented by increasing $\xi$), namely $H_\tau=\{\arg\zeta\in(-\theta_\tau,\pi-\theta_\tau)\}$.
\item[(iii)] $\operatorname{dist}(\omega,\Gamma_\tau)=\frac{\sqrt3}2$ and $\omega\in H_\tau$; $\operatorname{dist}(-i,\overline{H_\tau})=\operatorname{dist}(-i,\Gamma_\tau)\ge\frac12$, so $-i\notin\overline{H_\tau}$.
\item[(iv)] The real root of $\tau-\xi^3$ is $\xi=\tau^{1/3}$ (real cube root); it corresponds to $\zeta=1$ if $\tau>0$ and to $\zeta=\bar\omega$ if $\tau<0$. In particular $K(\cdot,\tau)$ has no singularity on $\mathbb R_\xi$, and
\begin{equation}\label{eq:Kbound}
|K(\xi,\tau)|\le C_{m,n}\,|\tau|^{-1/3}\,|\zeta|^m\langle\zeta\rangle^{-n}\qquad(\xi\in\mathbb R),
\end{equation}

\end{enumerate}
<!-- end lemma-8 -->
<!-- proof-5: body_md -->
(i) follows from $\rho^3=\tau$. (ii): $\zeta=\xi/\rho$ is a rotation by $-\theta_\tau$ followed by a dilation; $H_\tau$ is the image of $\{\operatorname{Im}\xi>0\}$. (iii): $\arg\omega=\frac{2\pi}3\in(0,\pi)\cap(\frac\pi3,\frac{4\pi}3)$, and the distance from $\omega$ to the line through $0$ with direction $e^{-i\theta_\tau}$ is $|\sin(\frac{2\pi}3+\theta_\tau)|=\frac{\sqrt3}2$ in both cases; the distance from $-i=e^{-i\pi/2}$ to the same line is $|\sin(-\frac\pi2+\theta_\tau)|\in\{1,\frac12\}$, and $\arg(-i)=\frac{3\pi}2\notin[-\theta_\tau,\pi-\theta_\tau]$. (iv): for $\tau<0$, $\tau^{1/3}=-|\tau|^{1/3}$ and $\zeta=-e^{i\pi/3}=e^{i4\pi/3}=\bar\omega$. For $\xi\in\mathbb R$ we have $\zeta\in\Gamma_\tau$, so $|\zeta-\omega|\ge\frac{\sqrt3}2$ by (iii). Moreover $|\zeta+i|\ge\frac12$ on $\overline{H_\tau}$, whence $|\zeta+i|\ge\max(\frac12,|\zeta|-1)\ge\frac14\langle\zeta\rangle$ and
\begin{equation}\label{eq:abound}
|a(\zeta)|\le C|\zeta|^m\langle\zeta\rangle^{-n}\qquad\text{on }\overline{H_\tau};
\end{equation}
\eqref{eq:Kbound} follows.

<!-- proposition-2: body_md -->
\label{prop:L}
Let $m,n$ satisfy \eqref{eq:mn} and $h\in C^\infty_{0,c}(\mathbb R^+)$. Then:
\begin{enumerate}
\item[(a)] (\emph{Regularity and time traces}) $\mathcal{L} h\in C(\mathbb R^2)$, and for every $\alpha\in\mathbb R$,
$$\sup_{x\in\mathbb R}\|\mathcal{L} h(x,\cdot)\|_{H^\alpha(\mathbb R_t)}\le C\|h\|_{H^\alpha(\mathbb R)},$$
with $x\mapsto\mathcal{L} h(x,\cdot)$ continuous into $H^\alpha(\mathbb R_t)$. Moreover $\mathcal{L} h(0,t)=h(t)$ for all $t\in\mathbb R$.
\item[(b)] (\emph{Causality}) $\mathcal{L} h(x,t)=0$ for $t\le0$; in particular $\mathcal{L} h(x,0)=0$.
\item[(c)] (\emph{equation on $x>0$}) $F:=(\partial_t+\partial_x^3)\mathcal{L} h$ is a continuous function on $\mathbb R^2$ with $F(x,t)=0$ for $x>0$. Hence $\mathcal{L} h$ solves $\partial_tu+\partial_x^3u=0$ in $(0,+\infty)\times\mathbb R$.
\item[(d)] (\emph{Explicit form on $x>0$}) For $x>0$,
$$\mathcal{L} h(x,t)=\frac1{2\pi}\int_\mathbb R e^{it\tau}\,\widehat h(\tau)\,e^{ix\omega\rho(\tau)}\,d\tau,\qquad |e^{ix\omega\rho(\tau)}|=e^{-\frac{\sqrt3}2|\tau|^{1/3}x}.$$
\item[(e)] (\emph{Bourgain--Besov norm}) Let $s\ge-1$ and assume in addition
\begin{equation}\label{eq:mn2}
m>-s-\tfrac12,\qquad n>m+s+2,\qquad \operatorname{supp} h\subset[0,2].
\end{equation}
Then
$$\|\mathcal{L} h\|_{X^{s,\frac12,1}}\le C\|h\|_{H^{\frac{s+1}3}(\mathbb R)},\qquad\text{hence}\qquad \|\mathcal{L} h\|_{C(\mathbb R_t;H^s(\mathbb R_x))}\le C\|h\|_{H^{\frac{s+1}3}(\mathbb R)} .$$
\end{enumerate}
For $s=-\frac34$ one may take $m=1$, $n=5$, and then $\|\mathcal{L} h\|_X\le2\|\mathcal{L} h\|_{X^{-\frac34,\frac12,1}}\le C\|h\|_{H^{1/12}}$.

<!-- proof-6: body_md -->
Since $\widehat h\in\mathcal S(\mathbb R)$ and, by \eqref{eq:Kbound} and the change of variables $\xi=\rho\zeta$,
\begin{equation}\label{eq:L1}
\int_\mathbb R|K(\xi,\tau)|\,d\xi\le C|\tau|^{-1/3}\int_{\Gamma_\tau}|\zeta|^m\langle\zeta\rangle^{-n}|\rho|\,|d\zeta|=C_1<\infty
\end{equation}
uniformly in $\tau\ne0$ (because $n\ge m+2$), the integrand in \eqref{eq:Ldef} is in $L^1(\mathbb R^2)$ and $\mathcal{L} h$ is a bounded continuous function.

\emph{(a)} Define
\begin{equation}\label{eq:kappa}
\kappa(x,\tau):=\frac1{2\pi}\int_\mathbb R e^{ix\xi}K(\xi,\tau)\,d\xi=\frac1{2\pi}\int_{\Gamma_\tau}e^{ix\rho(\tau)\zeta}\,\frac{a(\zeta)}{\zeta-\omega}\,d\zeta ,
\end{equation}
where we used $d\xi=\rho\,d\zeta$ and $\rho\cdot\rho^{-1}=1$. On $\Gamma_\tau$, $\rho\zeta=\xi\in\mathbb R$, so $|e^{ix\rho\zeta}|=1$ and by Lemma \ref{lem:algebra}(iii) $|\kappa(x,\tau)|\le C_0$ for all $(x,\tau)$. By Fubini, $$\mathcal{L} h(x,t)=\frac1{2\pi}\int e^{it\tau}\widehat h(\tau)\kappa(x,\tau)d\tau,$$ then  $\mathcal F_t\mathcal{L} h(x,\cdot)=\widehat h(\tau)\kappa(x,\tau)$, whence $\|\mathcal{L} h(x,\cdot)\|_{H^\alpha}\le C_0\|h\|_{H^\alpha}$. Continuity in $x$: $\kappa(x,\tau)\to\kappa(x_0,\tau)$ for each $\tau$ by dominated convergence in \eqref{eq:kappa}, and then $\int\langle\tau\rangle^{2\alpha}|\widehat h|^2|\kappa(x,\tau)-\kappa(x_0,\tau)|^2d\tau\to0$ by dominated convergence again.

For the trace, we compute $\kappa(0,\tau)$ by residues. The function $\zeta\mapsto a(\zeta)/(\zeta-\omega)$ is meromorphic with a simple pole at $\omega$ and a pole at $-i$, and is $O(|\zeta|^{m-n-1})=O(|\zeta|^{-2})$ at infinity. By Lemma \ref{lem:algebra}(iii), $\omega\in H_\tau$ and $-i\notin\overline{H_\tau}$. Closing $\Gamma_\tau$ by a large arc in $H_\tau$ (which lies to the left of the oriented line, so the contour is positively oriented) we get
$$\kappa(0,\tau)=\frac1{2\pi}\,2\pi i\,\operatorname{Res}_{\zeta=\omega}\frac{a(\zeta)}{\zeta-\omega}=i\,a(\omega)=1 .$$
Hence $\mathcal{L} h(0,t)=\frac1{2\pi}\int e^{it\tau}\widehat h(\tau)d\tau=h(t)$.

\emph{(b)} Fix $\xi\in\mathbb R$. For $z$ in the open lower half-plane $\Pi_-=\{\operatorname{Im} z<0\}$ put $\rho(z)=z^{1/3}$, $\zeta=\xi/\rho(z)$ and $K(\xi,z)=\rho(z)^{-1}a(\zeta)/(\zeta-\omega)$. Since $\arg\rho(z)\in(-\frac\pi3,0)$, we have $\arg\zeta\in(0,\frac\pi3)$ if $\xi>0$ and $\arg\zeta\in(\pi,\frac{4\pi}3)$ if $\xi<0$; in both cases $\zeta\ne\omega,-i$, so $z\mapsto K(\xi,z)$ is holomorphic in $\Pi_-$, continuous on $\overline{\Pi_-}\setminus\{0\}$, and $|K(\xi,z)|\le C|z|^{-1/3}$ (because $|\zeta|^m\langle\zeta\rangle^{-n}\le1$ and the distances of $\omega$, $-i$ to the sectors above are positive). Since $\operatorname{supp} h\subset[0,\infty)$, $\widehat h(z)=\int_0^\infty e^{-itz}h(t)dt$ is holomorphic in $\Pi_-$, continuous on $\overline{\Pi_-}$, and integrating by parts $N$ times, $|\widehat h(z)|\le C_N(1+|z|)^{-N}$ on $\overline{\Pi_-}$. For $t\le0$ and $z\in\overline{\Pi_-}$, $|e^{itz}|=e^{-t\operatorname{Im} z}\le1$. Apply Cauchy's theorem to $z\mapsto e^{itz}\widehat h(z)K(\xi,z)$ on the region $\{\varepsilon<|z|<R,\ \operatorname{Im} z<0\}$: the integral over the large semicircle is $O(R\cdot R^{-N}R^{-1/3})\to0$, that over the small semicircle is $O(\varepsilon\cdot\varepsilon^{-1/3})\to0$, and we conclude $\int_\mathbb R e^{it\tau}\widehat h(\tau)K(\xi,\tau)d\tau=0$ for every $\xi$ and every $t\le0$. Integrating in $\xi$ (Fubini) gives $\mathcal{L} h(x,t)=0$ for $t\le0$.

\emph{(c)} Since $(\partial_t+\partial_x^3)^\wedge=i(\tau-\xi^3)$, Lemma \ref{lem:algebra}(i) and $\rho^3/\rho=\rho^2=(\tau-i0)^{2/3}$ give
\begin{equation}\label{eq:Fhat}
\begin{split}
&\widehat F(\xi,\tau)=i(\tau-\xi^3)\widehat h(\tau)K(\xi,\tau)=-i\,\widehat h(\tau)\,(\tau-i0)^{2/3}\,b(\zeta),\\& b(\zeta):=a(\zeta)(\zeta-1)(\zeta-\bar\omega)=c\,\frac{\zeta^m(\zeta-1)(\zeta-\bar\omega)}{(\zeta+i)^n}.
\end{split}
\end{equation}
By Lemma \ref{lem:algebra}(iii), $|\zeta+i|\ge\frac14\langle\zeta\rangle$ on $\overline{H_\tau}$, hence $|b(\zeta)|\le C\langle\zeta\rangle^{m+2-n}\le C\langle\zeta\rangle^{-2}$ on $\overline{H_\tau}$. Consequently $\widehat F(\cdot,\tau)\in L^1(\mathbb R)$ with $$\int|\widehat F(\xi,\tau)|d\xi\le C|\widehat h(\tau)||\tau|^{2/3}|\rho(\tau)|=C|\widehat h(\tau)||\tau|,$$ so $\widehat F\in L^1(\mathbb R^2)$ and $F=\mathcal{F}^{-1}\widehat F$ is a bounded continuous function with $(\partial_t+\partial_x^3)\mathcal{L} h=F$ in $\mathcal S'(\mathbb R^2)$. Fix $\tau\ne0$ and $x>0$. The function $\xi\mapsto\widehat F(\xi,\tau)$ extends holomorphically to $\{\operatorname{Im}\xi>0\}$ (as $\zeta=\xi/\rho$ then lies in $H_\tau$, where $b$ is holomorphic), with $|\widehat F(\xi+i\eta,\tau)|\le C|\widehat h(\tau)||\tau|^{2/3}\langle(\xi+i\eta)/\rho\rangle^{-2}\le C|\widehat h(\tau)||\tau|^{2/3}\langle\xi/\rho\rangle^{-2}$ uniformly in $\eta\ge0$. By Cauchy's theorem applied to the rectangle with vertices $\pm R$, $\pm R+i\eta$, letting $R\to\infty$ (the vertical sides are $O(R^{-2})$),
\begin{equation*}\begin{split}
\int_\mathbb R e^{ix\xi}\widehat F(\xi,\tau)d\xi&=\int_\mathbb R e^{ix(\xi+i\eta)}\widehat F(\xi+i\eta,\tau)d\xi,\\
\Big|\int_\mathbb R e^{ix(\xi+i\eta)}\widehat F(\xi+i\eta,\tau)d\xi\Big|&\le e^{-x\eta}\,C|\widehat h(\tau)||\tau| .
\end{split}\end{equation*}
Letting $\eta\to+\infty$ we get $\int_\mathbb R e^{ix\xi}\widehat F(\xi,\tau)d\xi=0$ for $x>0$, $\tau\ne0$, and therefore $F(x,t)=\frac1{(2\pi)^2}\int e^{it\tau}\big(\int e^{ix\xi}\widehat F(\xi,\tau)d\xi\big)d\tau=0$ for $x>0$.

\emph{(d)} For $x>0$ and $\zeta\in\overline{H_\tau}$, $\rho\zeta$ lies in the closed upper half-plane, so $|e^{ix\rho\zeta}|\le1$; as $a(\zeta)/(\zeta-\omega)=O(|\zeta|^{-2})$, we may close $\Gamma_\tau$ in \eqref{eq:kappa} by a large arc in $H_\tau$ and obtain $\kappa(x,\tau)=i\,a(\omega)e^{ix\rho\omega}=e^{ix\omega\rho(\tau)}$. Finally $\operatorname{Im}(\omega\rho(\tau))=|\tau|^{1/3}\sin(\frac{2\pi}3+\theta_\tau)=\frac{\sqrt3}2|\tau|^{1/3}$ in both cases $\theta_\tau\in\{0,-\frac\pi3\}$.

\emph{(e)} Write $W:=\widehat h\,K=\widehat{\mathcal{L} h}$ and, for $l\ge1$, $S_l:=\{\tau:2^l\le|\tau|<2^{l+1}\}$, $S_0:=\{|\tau|<2\}$, $\widehat h_l:=\mathbf 1_{S_l}\widehat h$. Set $\beta:=\frac{s+1}3$ and $c_l:=2^{\beta l}\|\widehat h_l\|_{L^2}$, so that $\sum_{l\ge0}c_l^2\le C\|h\|^2_{H^\beta}$. For $j,k\ge0$ and $\tau\in\mathbb R$ let $B_k(\tau):=\{\xi:(\tau,\xi)\in B_k\}$ and
$$M_{jk}(\tau):=\|\langle\xi\rangle^s\langle\sigma\rangle^{\frac12}K(\cdot,\tau)\|_{L^2_\xi(A_j\cap B_k(\tau))}.$$
Since the sets $A_j\cap B_k\cap(\mathbb R\times S_l)$, $l\ge0$, are disjoint and cover $A_j\cap B_k$, the triangle inequality in $L^2$ gives $\|F\|_{L^2(A_j\cap B_k)}\le\sum_{l\ge0}\|F\|_{L^2(A_j\cap B_k\cap(\mathbb R\times S_l))}$ for every $F$; and since $|W|=|\widehat h_l||K|$ on $\mathbb R\times S_l$, $$\|\langle\xi\rangle^s\langle\sigma\rangle^{1/2}W\|_{L^2(A_j\cap B_k\cap(\mathbb R\times S_l))}\le\|\widehat h_l\|_{L^2}\sup_{\tau\in S_l}M_{jk}(\tau).$$ Summing over $k$, separating $l=0$ from $l\ge1$ and using $(a+b)^2\le2a^2+2b^2$, we have
\begin{equation}\label{eq:assemble}
\|\mathcal{L} h\|_{X^{s,\frac12,1}}^2\le 2\sum_{j\ge0}\Big(\sum_{l\ge1}\|\widehat h_l\|_{L^2}\,\Theta_{jl}\Big)^2+2\sum_{j\ge2}\|\widehat h_0\|_{L^2}^2\,\Theta_{j0}^2+2R_0,\qquad \Theta_{jl}:=\sum_{k\ge0}\sup_{\tau\in S_l}M_{jk}(\tau),
\end{equation}
where $R_0:=\sum_{j\in\{0,1\}}\big(\sum_k\|\langle\xi\rangle^s\langle\sigma\rangle^{\frac12}W\|_{L^2(A_j\cap B_k\cap(\mathbb R\times S_0))}\big)^2$ collects the two blocks $j\in\{0,1\}$, $l=0$, for which $\sup_{\tau\in S_0}M_{jk}(\tau)=+\infty$ (because $|K(\cdot,\tau)|\sim|\tau|^{-1/3}$ on $|\xi|\lesssim|\tau|^{1/3}$) and which are therefore estimated directly below. We bound $\Theta_{jl}$ using \eqref{eq:Kbound} and $|\zeta|=|\xi||\tau|^{-1/3}$. Recall $2^{j-1}\le|\xi|<2^{j+1}$ on $A_j$, $j\ge1$, and $|\xi|<\sqrt3$ on $A_0$. Let $\delta_1:=\frac{2s+2m+1}6>0$ and $\delta_2:=\frac{n-m-s-2}3>0$ by \eqref{eq:mn2}.

\emph{Case A: $l\ge3j+4$.} Then $|\xi|^3<2^{3j+3}\le\frac12|\tau|$ on $A_j$, so $\frac12|\tau|\le|\sigma|\le\frac32|\tau|$ and $M_{jk}(\tau)=0$ unless $|k-l|\le2$. Also $|\zeta|\le 2^{j+1}2^{-l/3}\le1$, so $|K|\le C|\tau|^{-1/3}(2^{j+1}2^{-l/3})^m$ and, using $|A_j|\le2^{j+2}$,
$$M_{jk}(\tau)\le C\,2^{sj}\,2^{l/2}\,2^{-l/3}\,2^{(j-l/3)m}\,2^{j/2}=C\,2^{(s+m+\frac12)j}\,2^{(\frac16-\frac m3)l}=C\,2^{\beta l}\,2^{-\delta_1(l-3j)} ,$$
where the last identity is checked by comparing exponents: $$(s+m+\frac12)j+(\frac16-\frac m3)l-\beta l=-\frac{2s+2m+1}{6}(l-3j).$$ Hence $\Theta_{jl}\le C2^{\beta l}2^{-\delta_1(l-3j)}$.

\emph{Case B: $1\le l\le3j-5$.} Then $|\tau|<2^{l+1}\le2^{3j-4}\le\frac12|\xi|^3$ on $A_j$, so $\frac12|\xi|^3\le|\sigma|\le\frac32|\xi|^3$ and $M_{jk}(\tau)=0$ unless $|k-3j|\le4$. Also $|\zeta|\ge2^{j-1}2^{-(l+1)/3}\ge1$, so $|K|\le C|\tau|^{-1/3}|\zeta|^{m-n}\le C\,2^{(m-n)j}\,2^{(n-m-1)l/3}$, and, with $\langle\sigma\rangle^{1/2}\le C2^{3j/2}$,
\begin{equation*}\begin{split}M_{jk}(\tau)&\le C\,2^{sj}\,2^{3j/2}\,2^{(m-n)j}\,2^{(n-m-1)l/3}\,2^{j/2}\\&=C\,2^{(s+2+m-n)j}\,2^{(n-m-1)l/3}=C\,2^{\beta l}\,2^{-\delta_2(3j-l)},\end{split}\end{equation*}
since $(s+2+m-n)j+\frac{n-m-1}3l-\beta l=-\frac{n-m-s-2}3(3j-l)$. Hence $\Theta_{jl}\le C2^{\beta l}2^{-\delta_2(3j-l)}$.

\emph{Case C: $3j-5<l<3j+4$, $l\ge1$} (at most $8$ values of $l$ for each $j$, and $j\ge1$ here unless $l\le3$). On $A_j\times S_l$ we have $\langle\sigma\rangle\le1+|\tau|+|\xi|^3<1+2^{l+1}+2^{3j+3}<2^{l+8}$ (as $3j+3\le l+7$), so $M_{jk}=0$ for $k>l+7$; $|K|\le C|\tau|^{-1/3}\le C2^{-l/3}$ by \eqref{eq:Kbound} and $m\le n$; and for $j\ge1$, since $\xi\mapsto\tau-\xi^3$ is monotone on $A_j\cap\{\pm\xi>0\}$ with $|\partial_\xi(\tau-\xi^3)|\ge3\cdot2^{2j-2}$, the set $A_j\cap B_k(\tau)$ (on which $|\sigma|$ lies in an interval of length $2^{k+2}$) has measure $\le C2^{k-2j}$. Therefore
\begin{equation*}\begin{split}M_{jk}(\tau)&\le C\,2^{sj}\,2^{k/2}\,2^{-l/3}\,(2^{k-2j})^{1/2}=C\,2^{(s-1)j}\,2^{k}\,2^{-l/3},\\
\Theta_{jl}&\le C\,2^{(s-1)j}\,2^{-l/3}\sum_{k\le l+7}2^k\le C\,2^{(s-1)j}\,2^{2l/3}\le C\,2^{\beta l},\end{split}\end{equation*}
using $2^j\sim2^{l/3}$. For $j=0$ and $l\le3$ the same bound holds trivially since $|K|\le C|\tau|^{-1/3}\le C$ on $S_l$, $l\ge1$, and $k\le 10$.

\emph{Low temporal frequencies: $l=0$.} For $j\ge2$ and $|\tau|<2$ we are in the situation of Case B ($|\tau|<2\le\frac12|\xi|^3$), and $|K|\le C|\tau|^{-1/3}(|\xi||\tau|^{-1/3})^{m-n}=C|\xi|^{m-n}|\tau|^{(n-m-1)/3}\le C|\xi|^{m-n}$ because $n-m-1\ge0$; thus $\Theta_{j0}\le C2^{(s+2+m-n)j}$, which is square-summable in $j$ by \eqref{eq:mn2}. For $j\in\{0,1\}$ and $|\tau|<2$ we have $|\xi|<\sqrt{15}$, hence $\langle\sigma\rangle<1+2+15^{3/2}<64$ and only $k\le5$ occur; thus the $\ell^1_k$ sum in $R_0$ has at most six terms and $\sum_{k\le5}\|F\|_{L^2(A_j\cap B_k\cap(\mathbb R\times S_0))}\le6\|F\|_{L^2(A_j\times S_0)}$, so that $R_0\le C\int_{S_0}|\widehat h(\tau)|^2\int_{|\xi|<\sqrt{15}}|K(\xi,\tau)|^2d\xi\,d\tau$. By \eqref{eq:Kbound} and the change of variables $\xi=\rho\zeta$,
$$\int_{\mathbb R}|K(\xi,\tau)|^2d\xi\le C|\tau|^{-2/3}\int_{\Gamma_\tau}|\zeta|^{2m}\langle\zeta\rangle^{-2n}|\rho|\,|d\zeta|=C|\tau|^{-1/3}.$$
Hence $$R_0\leq C\int_{|\tau|<2}|\widehat h(\tau)|^2|\tau|^{-1/3}d\tau\le C\|\widehat h\|^2_{L^\infty}\le C\|h\|^2_{L^1}\le C\|h\|_{L^2}^2,$$ where we used $\operatorname{supp} h\subset[0,2]$; and $\sum_{j\ge2}\|\widehat h_0\|^2_{L^2}\Theta_{j0}^2\le C\sum_{j\ge2}2^{2(s+2+m-n)j}\|h\|^2_{L^2}\le C\|h\|_{L^2}^2$.

\emph{Summation.} Inserting the bounds of Cases A--C into \eqref{eq:assemble} for $l\ge1$, we obtain
$$\sum_{j\ge0}\Big(\sum_{l\ge1}\|\widehat h_l\|_{L^2}\Theta_{jl}\Big)^2\le C\sum_{j\ge0}\Big(\sum_{l\ge1}\theta_{jl}\,c_l\Big)^2,\qquad
\theta_{jl}:=\begin{cases}2^{-\delta_1(l-3j)},& l\ge3j+4,\\ 2^{-\delta_2(3j-l)},& l\le3j-5,\\ 1,&\text{otherwise.}\end{cases}$$
Since $\sup_l\sum_j\theta_{jl}+\sup_j\sum_l\theta_{jl}\le C(\delta_1,\delta_2)$, then $$\sum_j(\sum_l\theta_{jl}c_l)^2\le C\sum_lc_l^2\le C\|h\|^2_{H^\beta}.$$ Together with the $l=0$ contribution this proves $\|\mathcal{L} h\|_{X^{s,\frac12,1}}\le C\|h\|_{H^\beta}$ (note $\|h\|_{L^2}\le\|h\|_{H^\beta}$ since $\beta\ge0$). The $C_tH^s$ bound follows from Lemma \ref{lem:embedding}. For $s=-\frac34$, \eqref{eq:mn} and \eqref{eq:mn2} read $m>\frac14$, $n>m+\frac54$, $n\ge m+4$, satisfied by $m=1$, $n=5$.

<!-- remark-3: body_md -->
\label{rem:support}
The support assumption on $h$ in (e) is not a technicality: $\mathcal{L}$ is not bounded from $H^\beta(\mathbb R)$ into $X^{s,\frac12}$. Indeed $\int_\mathbb R|K(\xi,\tau)|^2d\xi=C|\tau|^{-1/3}$, so for $\widehat h=\varepsilon^{-1/2}\mathbf 1_{|\tau|<\varepsilon}$ one has $\|h\|_{H^\beta}\sim1$ while $\|\mathcal{L} h\|_{X^{s,\frac12}}\gtrsim\varepsilon^{-1/6}$. The mechanism is physical: a boundary datum of temporal frequency $\tau$ produces a layer of width $|\tau|^{-1/3}$, which for $|\tau|\to0$ is not square integrable in $x$ uniformly. Since the fixed point only requires boundary values on a bounded time interval, restricting to $\operatorname{supp} h\subset[0,2]$ costs nothing (Section \ref{sec:proof}).

<!-- remark-4: body_md -->
\label{rem:holmer}
All boundary forcing operators in the literature agree with $\mathcal{L}$ on $x>0$, by uniqueness for the linear problem (compare \cite[Lemma 8.1]{Holmer}): there they reduce to the Laplace-transform formula of Proposition \ref{prop:L}(d). They differ only in the continuation to $x<0$, and this is what decides the admissible $b$.

\emph{Point sources.} Holmer's $\mathcal{L}^\lambda_+$ \cite[Section 3]{Holmer} (and the operator of \cite{CK}) is a Duhamel integral of a source $x_-^{\lambda-1}/\Gamma(\lambda)$ at the boundary --- the classical boundary potential of Cattabriga \cite{Cat}, used by Faminskii \cite{Fa2} in Bourgain-type spaces to obtain global well-posedness for $s\ge0$. Its space-time Fourier transform is, up to a free solution, $(\xi-i0)^{-\lambda}(\tau-i0)^{\frac\lambda3+\frac23}\widehat h(\tau)(\tau-\xi^3)^{-1}$ (proof of Lemma 5.8(d) there): a simple pole at the real root $\xi=\tau^{1/3}$ with nonvanishing residue, so that $\int\langle\xi\rangle^{2s}\langle\sigma\rangle^{2b}|\cdot|^2d\xi$ converges only for $b<\frac12$ and the $\ell^1_k$ sum in \eqref{eq:Xsb1} diverges at $b=\frac12$. In physical space, $\mathcal{L}^\lambda_+h$ contains left-travelling free waves emitted from $x=0$. In \eqref{eq:K} the real root is not a pole and the trace is carried by the complex root $\omega\rho(\tau)$.

\emph{Reflections.} The operators $BI^{m1},BI^{m2}$ of \cite{BSZ2} extend the solution on $x>0$ by even and odd reflections, combined through the cutoff $\Theta(\xi,\tau)=\chi(|\xi|-\delta|\tau|^{1/3})$ and a corrector $\omega(\tau)$ fixed by the implicit relation (2.22) there. Our computation in the proof of Proposition \ref{prop:L}(e) reproduces their results exactly:
\begin{enumerate}
\item[(i)] At fixed $\tau$, for $s>-\frac12$ and all $b$ for which the integral converges, $$\int\langle\xi\rangle^{2s}\langle\sigma\rangle^{2b}|K|^2d\xi\sim|\tau|^{\frac{2s-1}3+2b},$$ i.e.\ $\|\mathcal{L} h\|_{X^{s,b}}\sim\|h\|_{H^{\frac{3b+s-1/2}3}}$ --- the regularity of Theorem 1.1 in \cite{BSZ2}, sharp in $b$, with $b=\frac12$ the value at which the datum has its natural regularity $\frac{s+1}3$. Convergence at $|\xi|\to\infty$ requires $n-m>s+2$, i.e.\ a $C^{n-m-1}$ junction at $x=0$ (Lemma \ref{lem:glue}); this is the role played in $BI^{m1}$ by the corrector $\omega(\tau)$, which removes a jump $O(\xi^{-1})$, and the restriction $b<\frac12-\frac s3$ there.
\item[(ii)] For $s<-\frac12$ and $m=0$ the region $|\xi|\le1$ dominates and yields only $\|h\|_{H^{\frac{3b-1}3}}$ --- the estimate of Appendix II of \cite{BSZ2} for the even extension $BI^e$, and the reason $BI^{m1}$ is introduced there. In $BI^{m1}$ the even part is switched off for $|\xi|\lesssim\delta|\tau|^{1/3}$ by $\Theta$; in $\mathcal{L}$ the same effect is produced by the factor $\zeta^m$, i.e.\ by the $m$ vanishing moments of the profile $g$ in \eqref{eq:ext}.
\end{enumerate}
Thus, for $-\frac34<s\le0$, Theorem 1.1 of \cite{BSZ2} and Proposition \ref{prop:L}(e) with $\ell^2_k$ in place of $\ell^1_k$ are the same statement. What \cite{BSZ2} does not address is the $\ell^1_k$ summability, which in the proof of Proposition \ref{prop:L}(e) comes from the pointwise bound \eqref{eq:Kbound} alone; the rational form of $K$ is what makes \eqref{eq:Kbound}, the trace identity and the support property elementary consequences of residue calculus.

\emph{Cutoffs.} A third family, going back to \cite{ET}, multiplies the decaying mode by a smooth cutoff $\rho(a_I\xi x)$ at the scale of the layer: Himonas--Yan \cite{HY} for KdV with Robin data, Compaan--Tzirakis \cite{CT} for the quartic gKdV, Gallego--Kwak \cite{GK} for higher-order KdV. The symbol is $|\tau|^{-1/3}\widehat\eta(\xi/(a_I|\tau|^{1/3}))\widehat h(\tau)$ with $\eta\in\mathcal S(\mathbb R)$ (times $\xi$ for Robin data): the structure of \eqref{eq:K} with a Schwartz profile in place of the rational one, and $|\widehat\eta(\xi/(a_I|\tau|^{1/3}))|\le C_N(|\tau|^{1/3}/(|\xi|+|\tau|^{1/3}))^N$ \cite[Lemma 2.2]{HY} is the counterpart of \eqref{eq:Kbound}. The resulting $X^{s,b}$ bounds \cite[Lemma 2.3]{HY}, \cite[Lemma 3.6]{CT}, \cite[Lemma 4.3]{GK} have the exponent of (i). The Besov norm is not considered in these works; the nonlinear arguments of \cite{HY,GK} stay at $b<\frac12$ with the correction $D_\alpha$ and the temporal space $Y^{s,-b}$ of \cite{Holmer}, and that of \cite{CT} places a linear part in $X^{s_0,b}$, $b>\frac12$, $s_0<s$, recovering the loss by nonlinear smoothing, which vanishes for KdV at $s=-\frac34$.
