\label{thm:main-result} 
Fix an even integer $k\ge 2$. Let $P$, $X$, and $Y$ be parameters going to infinity with $X,Y\in \mathbb{R}_{+}$ and $P$ prime; assume further that $Y=(1+o(1))X^{1-\delta_2},$ and $P\ll X^{1+\delta_1}$ for some $\delta_1,\delta_2$ with $0<\delta_1<\frac{1}{11}$, $2\delta_1<\delta_2<\frac{1}{13} (4-18\delta_1).$ Let $y := \frac{P}{X}$. Then 
\begin{align}
    &\frac{
    \sum_{N\in[X,X+Y]}^\square \sum_{f\in H^{\rm new}(N,k)}\sqrt{P}\lambda_f(P)\varepsilon(f)
    }{
    \sum_{N\in[X,X+Y]}^\square \sum_{f\in H^{\rm new}(N,k)}1 
    }  \\
    &=\frac{\alpha(-1)^{\frac{k}{2}-1}}{k-1}\sum_{1\leq r\leq 2\sqrt{y}}U_{k-2}\!\left(\frac{r}{2\sqrt{y}}\right)\nu(r)\sqrt{4y-r^2} +\frac{\beta}{k-1}\sqrt{y}-\gamma \delta_{k=2}y  \\
    &\quad\ +O_\varepsilon\!\left(X^{-\delta'+\varepsilon}+\frac{1}{P}\right)
\end{align}
where $\delta'>0$ is explicitly expressible through $\delta_1,\delta_2,$ and 
\begin{align}
\alpha= 2\pi \prod_{p}\frac{1-p-2p^2+p^4}{p^4-2p^2+p}, \quad \beta=2\pi\prod_{p}\frac{-1+p^2+p^3}{p(-1+p+p^2)}, \quad 
\gamma=12\prod_{p}\frac{p(1+p)}{-1+p+p^2}.
\end{align}

<!-- proof-1: body_md -->
Observe that 
    \begin{align}
        & \frac{1}{XY} \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y] \\ P\nmid N}}\sum_{1\leq r \leq 2\sqrt{\frac{P}{N}}} U_{k-2}\!\left(\frac{r\sqrt{N}}{2\sqrt{P}}\right) H_1(r^2N^2-4PN) \\
                                                  
        &=\frac{1}{XY}
        \sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}}\sideset{}{^\square} \sum_{\substack{N\in [X,X+Y] \\ P \nmid N}}
        U_{k-2}\!\left(\frac{r\sqrt{N}}{2\sqrt{P}}\right) H_1(r^2N^2-4PN)\\
                                
        &  \quad + \frac{1}{XY}
        \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y] \\ P\nmid N}}
        \sum_{2\sqrt{\frac{P}{X+Y}}\le r \le 2\sqrt{\frac{P}{N}}}  U_{k-2}\!\left(\frac{r\sqrt{N}}{2\sqrt{P}}\right) H_1(r^2N^2-4PN). \\
                                                  
        &= \frac{1}{XY}\sum_{1\leq r< 2\sqrt{\frac{P{}}{X+Y}}} U_{k-2}\!\left(\frac{r\sqrt{X}}{2\sqrt{P}}\right)
        \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y]\\ P\nmid N}} H_1(r^2N^2-4PN) \label{eq:initial-double-sum-main} \\
                                 
        &\quad + \frac{1}{XY}\sum_{1\leq r< 2\sqrt{\frac{P{}}{X+Y}}}\,\, \sideset{}{^\square }\sum_{\substack{N\in[X,X+Y]\\ P\nmid N}} E(r,N)H_1(r^2N^2-4PN)  \label{eq:initial-double-sum-first-error} \\
                                 
        &\quad + \frac{1}{XY}
        \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y] \\ P\nmid N}}
        \sum_{2\sqrt{\frac{P}{X+Y}}\le r \le 2\sqrt{\frac{P}{N}}}  U_{k-2}\!\left(\frac{r\sqrt{N}}{2\sqrt{P}}\right) H_1(r^2N^2-4PN),
        \label{eq:initial-double-sum-second-error}
    \end{align}
    where 
    \begin{align} \label{eqn:def-E(r,N)}
        E(r,N) := U_{k-2}\!\left(\frac{r}{2}\frac{\sqrt{N}}{\sqrt{P}}\right)-
        U_{k-2}\!\left(\frac{r}{2}\frac{\sqrt{X}}{\sqrt{P}}\right).
    \end{align}
    We show that \eqref{eq:initial-double-sum-main} constitutes the main term of the desired result (up to the claimed error term), and that
    \eqref{eq:initial-double-sum-first-error}
    and
    \eqref{eq:initial-double-sum-second-error}
    are bounded by the claimed error term.


        
    
    First, we have that
    \begin{align}
        \eqref{eq:initial-double-sum-main} 
        &= \frac{1}{XY}\sum_{1\leq r<2\sqrt{\frac{P{}}{X+Y}}} U_{k-2}\!\left(\frac{r\sqrt{X}}{2\sqrt{P}}\right)
        \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y]\\ P\nmid N}} H_1(r^2N^2-4PN) \\
                                                
        &= \frac{1}{XY} \sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}} 
        U_{k-2}\!\left(\frac{r\sqrt{X}}{2\sqrt{P}}\right) XYB_0\nu(r)\sqrt{4y-r^2} \\ 
        &\quad + \frac{1}{XY} \sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}}  O(k) 
        \cdot O_\varepsilon\!\left(P^\frac{1}{2} Y^\frac{3}{2} \mathcal{E} \right)
        \qquad \text{(by Lemma \ref{lem:bound-sum-class-num})}
        \\
                                                
        &= B_0 \sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}} 
        U_{k-2}\!\left(\frac{r\sqrt{X}}{2\sqrt{P}}\right)\nu(r)\sqrt{4y-r^2} \\ 
        &\quad +
         O_\varepsilon\bigg(k \Bigg[
        \frac{1}{XY} \cdot \sqrt{\frac{P}{X}} \cdot P^\frac{1}{2} Y^\frac{3}{2} \mathcal{E}
        \bigg]
        \Bigg)  \\ 
                                               
        &= B_0 \sum_{1\leq r\le 2\sqrt{y}} U_{k-2}\!\left(\frac{r}{2\sqrt{y}}\right)\nu(r) \sqrt{4y-r^2} + O\!\left(k\!\left[\frac{PY^\frac{3}{2}}{X^{\frac{5}{2}}}+\frac{P^\frac{1}{2}Y^\frac{1}{2}}{X}\right]\right) \\ 
        &\quad +
         O_\varepsilon\bigg(k \Bigg[
        \frac{P Y^\frac{1}{2}}{X^\frac{3}{2}} \mathcal{E}
        \bigg] 
        \Bigg) \hspace{45mm} \text{(by Lemma \ref{lem:Uk-split-main-error})} \\ 
                                               
        &= B_0 \sum_{1\leq r\le 2\sqrt{y}} U_{k-2}\!\left(\frac{r}{2\sqrt{y}}\right)\nu(r) \sqrt{4y-r^2}   +
        O_\varepsilon\!\left(k\!\left[
            \frac{PY^\frac{3}{2}}{X^{\frac{5}{2}}}
            +
            \frac{P^\frac{1}{2}Y^\frac{1}{2}}{X} 
            +
            \frac{P Y^\frac{1}{2}}{X^\frac{3}{2}} \mathcal{E}
        \right]\right) \\
        &= B_0 \sum_{1\leq r\le 2\sqrt{y}} U_{k-2}\!\left(\frac{r}{2\sqrt{y}}\right)\nu(r) \sqrt{4y-r^2}   +
        O_\varepsilon\!\left(k\!\left[
            \frac{P^\frac{1}{2}Y^\frac{1}{2}}{X} 
            +
            \frac{P Y^\frac{1}{2}}{X^\frac{3}{2}} 
        \right] \mathcal{E}
        \right) \\
        & \hspace{45mm} \text{\Big(since $\tfrac{PY^\frac{3}{2}}{X^{\frac{5}{2}}} \le \tfrac{P Y^\frac{1}{2}}{X^\frac{3}{2}} \le \tfrac{P Y^\frac{1}{2}}{X^\frac{3}{2}} \mathcal{E}$ and $\tfrac{P^\frac{1}{2}Y^\frac{1}{2}}{X} \le \tfrac{P^\frac{1}{2}Y^\frac{1}{2}}{X} \mathcal{E}$ \Big)} \\
        &= B_0 \sum_{1\leq r\le 2\sqrt{y}} U_{k-2}\!\left(\frac{r}{2\sqrt{y}}\right)\nu(r) \sqrt{4y-r^2}   +
        O_\varepsilon \!\left(k\!\left[
            1
            +
            \frac{P^\frac{1}{2}}{X^\frac{1}{2}}
        \right] \frac{P^\frac{1}{2}Y^\frac{1}{2}}{X} \mathcal{E}\right),
    \end{align}
    which constitutes the claimed main term.

    

    Second, we have that
    \begin{align}
        &|\eqref{eq:initial-double-sum-first-error}| \\
        &=
        \!\left\lvert 
        \frac{1}{XY}\sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}} \ \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y]\\ P\nmid N}} E(r,N)H_1(r^2N^2-4PN) 
         \right\lvert\\
        &\le 
        \frac{1}{XY}\sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}} \ \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y]\\ P\nmid N}} 
        \frac{12k}{4y-r^2} \frac{PY}{X^2}
        \cdot
        |H_1(r^2N^2-4PN)|
        \quad\ \text{(by Lemma \ref{lem:E(r,N)-bound})}
        \\
        &= \frac{1}{XY}\sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}} \frac{12k}{4y-r^2} \frac{PY}{X^2} \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y]\\ P\nmid N}} H_1(r^2N^2-4PN) \\
        &\qquad \text{\big(since $r^2N^2-4PN \ne 0$ because $NP$ is squarefree\big)} \\
        &\ll_\varepsilon \frac{1}{XY}\sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}} \frac{12k}{4y-r^2} \frac{PY}{X^2} \cdot  
            XY \sqrt{4y-r^2} \mathcal{E} \qquad\qquad \text{(by Lemma \ref{lem:bound-sum-class-num})}
        \\
        &= \frac{12kPY}{X^2} 
            \mathcal{E}
        \sum_{1\leq r< 2\sqrt{\frac{P}{X+Y}}} \frac{1}{\sqrt{4y-r^2}} \\
        &\le \frac{12kPY}{X^2} 
            \mathcal{E}
        \!\left[
            \int_1^{2 \sqrt{\frac{P}{X+Y}}} \frac{1}{\sqrt{4y-r^2}} dr
            + 
            \frac{1}{\sqrt{4y-4\frac{P}{X+Y}}}
        \right] \\
        &\le \frac{12kPY}{X^2} 
            \mathcal{E}
        \!\left[
            \int_0^{2 \sqrt{y}} \frac{1}{\sqrt{4y-r^2}} dr
            + 
            \frac{1}{2 \sqrt{\frac{P}{X}} \sqrt{1-\!\left(1+\frac{Y}{X}\right)^{-1}}}
        \right] \\
        &\le \frac{12kPY}{X^2} 
            \mathcal{E}
        \!\left[
            \frac{\pi}{2}
            + 
            \frac{1}{2 \sqrt{\frac{P}{X}} \cdot 
            \sqrt{\frac{1}{2} \frac{Y}{X}}
            }
        \right] 
        \qquad 
        \text{(by a first-order Taylor approximation)}
        \\
        &\ll_\varepsilon  k 
        \!\left[
            \frac{PY}{X^2}
            + 
            \frac{P^\frac{1}{2}Y^\frac{1}{2}}{X}
        \right] \mathcal{E} \\
        &= k \!\left[\frac{P^\frac{1}{2}Y^\frac{1}{2}}{X}+1\right]\frac{P^\frac{1}{2}Y^\frac{1}{2}}{X} \mathcal{E} \\
        &\ll_\varepsilon  k \!\left[\frac{P^\frac{1}{2}}{X^\frac{1}{2}}+1\right]\frac{P^\frac{1}{2}Y^\frac{1}{2}}{X} \mathcal{E},
    \end{align}
    which is bounded by the claimed error term.


    Third, observe that for $r$ in the range $2\sqrt{\frac{P}{X+Y}}\leq r\leq 2\sqrt{\frac{P}{X}}$ and $N\in[X,X+Y],$ we have the bound
    \begin{align}
        &\!\left\lvert r^2N^2-4PN \right\lvert \leq 4PN\!\left(1-\frac{N}{X+Y}\right)\leq 4P(X+Y)\!\left(1-\frac{X}{X+Y}\right)\ll PY, \\
        &\text{so that} \qquad
        H_1(r^2N^2-4PN) \ll_\varepsilon  (PY)^{\frac{1}{2}+\varepsilon}. \label{eqn:H-bound-from-r2N24pN-disc-bound}
    \end{align}
    Hence
    we can bound \eqref{eq:initial-double-sum-second-error} by 
    \begin{align}
        \eqref{eq:initial-double-sum-second-error}
        &= \frac{1}{XY}
        \sideset{}{^\square}\sum_{\substack{N\in[X,X+Y] \\ P\nmid N}}
        \sum_{2\sqrt{\frac{P}{X+Y}}\le r \le 2\sqrt{\frac{P}{N}}}  U_{k-2}\!\left(\frac{r\sqrt{N}}{2\sqrt{P}}\right) H_1(r^2N^2-4PN) \\
        &\ll_\varepsilon  \frac{1}{XY} \sideset{}{^\square}\sum_{\substack{N\in [X,X+Y]\\P\nmid N}} \sum_{2\sqrt{\frac{P}{X+Y}}\le r \le 2\sqrt{\frac{P}{N}}} k(PY)^{\frac{1}{2}+\varepsilon}  
        \qquad  \text{\big(since $U_{k-2}(x)\ll k$ and by \eqref{eqn:H-bound-from-r2N24pN-disc-bound}\big)}
        \\ 
        & \ll_\varepsilon  \frac{1}{XY}\cdot Y\!\left[2\sqrt{\frac{P}{X}}-2\sqrt{\frac{P}{X+Y}}+1\right] k(PY)^{\frac{1}{2}+\varepsilon} \\ 
        &\ll_\varepsilon  \frac{k}{X} \!\left[\sqrt{\frac{P}{X}}
        \!\left(1-\sqrt{\frac{X}{X+Y}}\right)+1\right] (PY)^{\frac{1}{2}+\varepsilon} \\
        & \le \frac{k}{X} \!\left[\sqrt{\frac{P}{X}} \!\left(\frac{1}{2} \frac{Y}{X}\right) + 1\right] (PY)^{\frac{1}{2}+\varepsilon} 
        \qquad \text{(by a first order Taylor approximation)}

<!-- lemma-4: body_md -->
\label{lem:lipschitz-const-at-x0-for-U}
    Fix a natural number $K \ge 1$, and let $x_0 \in (-1,1)$. Then for all $x \in [-1,1]$,
    \begin{align}
        \label{eqn:lipschitz-const-goal}
        |U_{K-1}(x) - U_{K-1}(x_0)| \le \frac{3K}{1-|x_0|} |x-x_0|.
    \end{align}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
We divide into two cases.

\textbf{Case 1: $|x-x_0| > \frac{2}{3} (1-|x_0|)$. \\}
Recall the global bound for Chebyshev polynomials
\begin{align}
    -K \le U_{K-1}(x) \le K.
\end{align}
This bound then implies that
\begin{align}
    |U_{K-1}(x)-U_{K-1}(x_0)| \le 2K = \frac{3K}{1-|x_0|} \cdot \frac{2}{3} (1-|x_0|) \le \frac{3K}{1-|x_0|} |x-x_0|,
\end{align}
verifying \eqref{eqn:lipschitz-const-goal}.


\textbf{Case 2: $|x-x_0| \le \frac{2}{3} (1-|x_0|)$. \\}
Observe that for every $t$ between $x_0$ and $x$, 
\begin{align}
    1-|t|
    &\ge
    1-|x_0|-|t-x_0| \\
    &\ge
    1-|x_0|-\frac{2}{3}(1-|x_0|) \\
    &=
    \frac{1}{3}(1-|x_0|).
    \label{eq:xi-away-from-endpoint}
\end{align}


Additionally, we have the derivative bound
\begin{align}
    |U_{K-1}'(t)|
    &=
    \!\left\lvert 
        \frac{1}{-\sin \theta}
        \cdot
        \frac{d}{d\theta}
        \!\left[
            \frac{\sin(K\theta)}{\sin \theta}
        \right]
     \right\lvert \qquad \qquad\quad
    \text{(where $t=\cos \theta$)}
    \\
    &=
    \!\left\lvert 
    \frac{
        K\cos(K\theta)\sin\theta
        -
        \sin(K\theta)\cos\theta
    }{\sin^3\theta}
     \right\lvert \\
    &\le
    \frac{
        K\sin\theta
        +
        K\sin\theta\,|\cos\theta|
    }{\sin^3\theta}
    \qquad\qquad\quad
    \text{(since $|\sin(K\theta)|\le K\sin\theta$)}
    \\
    &=
    \frac{K(1+|t|)}{1-t^2} \\
    &=
    \frac{K}{1-|t|}.
    \label{eq:U-derivative-bound}
\end{align}



Hence by the Mean Value Theorem, we have that
\begin{align}
    |U_{K-1}(x)-U_{K-1}(x_0)|
    &=
    |U_{K-1}'(t)|\,|x-x_0| 
    \qquad \text{(for some $t$ between $x_0$ and $x$)} 
    \\
    &\le
    \frac{K}{1-|t|}|x-x_0| 
    \qquad\quad\ \text{(by \eqref{eq:U-derivative-bound})}
    \\
    &\le
    \frac{3K}{1-|x_0|}|x-x_0|, 
    \qquad\ \,  \text{(by \eqref{eq:xi-away-from-endpoint})}
\end{align}
verifying \eqref{eqn:lipschitz-const-goal}.

Observe that
    \begin{align*} b(n)=\mu(n)^2\prod_{p\mid n}c_p,\qquad \text{where } c_p:=\frac{p^4}{p^4-2p^2-p+1}.
    \end{align*}
    Note that $c_p-1\ll\frac{1}{p^2}$. 
    Define the multiplicative function $g$ by
    \[
        g(p^e)=(-1)^{e-1}(c_p-1)
        \qquad \text{for } e\ge1.
    \]
    Then we have that $b = \mu^2 * g$ 
    because of the formal power series identity
    \begin{align*}
        \sum_{e\ge0}b(p^e)x^e
        &=
        1+c_px \\
        &=
        (1+x)
        \left(
            1+\frac{(c_p-1)x}{1+x}
        \right) \\
        &=
        \left(\sum_{e\ge0}\mu^2(p^e)x^e\right)
        \left(\sum_{e\ge0}g(p^e)x^e\right).
    \end{align*}

    \begin{align} \label{eq:h(n)-convergence}
        \sum_{n\ge1}\frac{|g(n)|}{\sqrt{n}}
        =
        \prod_p
        \sum_{e\ge0} \frac{|g(p^e)|}{p^{\frac{e}{2}}}
        =\prod_p\left(1+(c_p-1)\frac{p^{\frac{-1}{2}}}{1-p^{\frac{-1}{2}}}\right)
        < \infty,
    \end{align}
    since each local factor is $1+O\big(p^{\frac{-5}{2}}\big)$.

    \begin{align}
        B(x)
        \,\,&\!\!:=\sum_{n\le x}b(n) \\
        &= \sum_{n \le x} \sum_{d \mid n} g(d) \mu^2\!\left(\frac{n}{d}\right) 
        \\
        &=\sum_{d\le x}g(d)\sum_{m\le \frac{x}{d}}\mu^2(m)\\
        &= \sum_{d\le x}g(d) \!\left[
            \frac{1}{\zeta(2)} \frac{x}{d} + O\!\left(\sqrt{\frac{x}{d}}\right)
        \right] 
        \\
        &\qquad \text{(since $\textstyle \sum_{n\le X} \mu^2(n) = \frac{X}{\zeta(2)} + O(\sqrt X)$ by \cite[Exercise 4.4.10]{murty})}
        \\
        &= \frac{x}{\zeta(2)}\sum_{d\le x}\frac{g(d)}{d}+O\left(\sqrt{x}\sum_{d\le x}\frac{|g(d)|}{\sqrt{d}}\right) \\
        &= \!\left[
            \frac{x}{\zeta(2)} \sum_{d \ge 1} \frac{g(d)}{d} - \frac{x}{\zeta(2)} \sum_{d > x} \frac{g(d)}{d}

        + O(\sqrt{x}) 
        \qquad\qquad \text{(by \eqref{eq:h(n)-convergence})}

        \le
        \sqrt{x}
        \sum_{d>x}\frac{|g(d)|}{\sqrt d}
        \ll \sqrt{x}
        $
        by \eqref{eq:h(n)-convergence}\big)}
    \end{align}

First, we have that
\begin{align} \label{eq:small-d}
\!\left\lvert \frac1a\sum_{d\le a^{\frac{2}{3}}}b(d) F\Big(\frac{d}{a}\Big) \right\lvert
&\ll_k
\frac1{a}\sum_{d\le a^{\frac{2}{3}}}\!\left(\frac{d}{a}\right)^{\frac{-1}{2}} 
\qquad \text{(by \eqref{eqn:b(d)<<1} and \eqref{eq:F-bounds})}
\\
&\ll_k
\frac{a^{\frac{1}{2}}}{a} \cdot 
\!\left(a^{\frac{2}{3}}\right)^{\frac{1}{2}} 
\\
&=
a^{\frac{-1}{6}},
\end{align}
which is bounded by the claimed error term.
















Second, we have by partial summation from $a^\frac{2}{3}$ to $\infty$ that
\begin{align}
&\frac1a\sum_{d>a^{\frac{2}{3}}}b(d)F\!\left(\frac{d}{a}\right) \\
&=
\frac{1}{a}
\!\left[
    \lim_{x\to\infty} B(x) F\!\left(\frac{x}{a}\right)
    -B(a^{\frac{2}{3}}) F\bigg(\frac{a^\frac{2}{3}}{a}\bigg)
    -\frac1{a}\int_{a^{\frac{2}{3}}}^\infty B(t)F'\!\left(\frac{t}{a}\right)\,dt
\right]
\\
&=
\frac{1}{a}
\!\left[
    -\!\left[\Delta a^{\frac{2}{3}} + E(a^{\frac{2}{3}}) \right] F\big(a^{\frac{-1}{3}}\big)
    -\frac1{a}\int_{a^{\frac{2}{3}}}^\infty \big[\Delta t + E(t)\big] F'\!\left(\frac{t}{a}\right)\,dt
\right]
\\
& \qquad \text{(where $E(t) := B(t) - \Delta t \ll \sqrt t$ by Lemma \ref{lem:B-approx})} \\
&= - \Delta a^{\frac{-1}{3}} F\big(a^{\frac{-1}{3}}\big) - \frac{\Delta}{a^2} \int_{a^{\frac{2}{3}}}^\infty t F'\!\left(\frac{t}{a}\right)\,dt + \!\left[
    -\frac{E(a^{\frac{2}{3}})}{a} F\big(a^{\frac{-1}{3}}\big)
    - \frac{1}{a^2} \int_{a^{\frac{2}{3}}}^\infty E(t) F'\!\left(\frac{t}{a}\right)\,dt
\right] 
\\
&=:
- \Delta a^{\frac{-1}{3}} F\big(a^{\frac{-1}{3}}\big)
-\Delta\int_{a^{\frac{-1}{3}}}^\infty x F'(x)\,dx
+ \hat E(a) 
\\
&=
\Delta\int_{a^{\frac{-1}{3}}}^\infty F(x)\,dx
+ \hat E(a)
\qquad\qquad\quad\ \ \,
\text{\big(by IBP, since $\lim_{x \to \infty} xF(x) = 0$ by \eqref{eq:F-bounds}\big)}
\\
&= \Delta\int_0^\infty F(x)\,dx
+O_k\big(a^{\frac{-1}{6}}\big) +\hat E(a)
\quad\,
\text{\big(since 
  $\textstyle
    \int_0^{a^{\frac{-1}{3}}} \!\lvert F(x)\rvert dx
    \ll
    \int_0^{a^{\frac{-1}{3}}} \!\!x^{\frac{-1}{2}}dx
    =
    2a^{\frac{-1}{6}}
  $\big)} 
\\
&=
\Delta\int_0^\infty \frac{J(u)}{u}\,du
+O_k\big(a^{\frac{-1}{6}}\big) + \hat E(a)
\quad
\text{\big(letting $u=\tfrac{1}{x}$\big)}
\\
&=
-\frac{\Delta}{4}
+O_k\big(a^{\frac{-1}{6}}\big) + \hat E(a),
\qquad\qquad\quad\ 
\text{(by \eqref{eq:J-integral})}
\end{align}
which gives the main term of the desired result.






Additionally, observe that
\begin{align}
\!\left\lvert \hat E(a) \right\lvert
&= 
\!\left\lvert 
    -\frac{E(a^{\frac{2}{3}})}{a} F\big(a^{\frac{-1}{3}}\big)
    - \frac{1}{a^2} \int_{a^{\frac{2}{3}}}^\infty E(t) F'\!\left(\frac{t}{a}\right)\,dt
 \right\lvert \\
&\ll_k
\frac{|E(a^{\frac{2}{3}})|}{a} |F\big(a^{\frac{-1}{3}}\big)|
+
\frac{1}{a} 
\int_{a^{\frac{-1}{3}}}^\infty
\!\left\lvert 
    E\!\left(ax\right)
 \right\lvert
|F'(x)|\,dx
\\
&\ll_k
\frac{\sqrt{a^\frac{2}{3}}}{a} \!\left(a^\frac{-1}{3}\right)^\frac{-1}{2}
+
\frac{1}{a}
\int_{a^{\frac{-1}{3}}}^\infty
\sqrt{ax}
\cdot
x^\frac{-5}{2}\,dx
\qquad
\text{(by \eqref{eq:F-bounds} and \eqref{eq:F'-bounds})}
\\
&=
a^{\frac{-1}{2}}
+
a^{\frac{-1}{2}}
\left(a^\frac{-1}{3}\right)^{-1}
\\
&\ll_k
a^{\frac{-1}{6}},
\end{align}
which is bounded by the claimed error term. This
completes the proof.

<!-- lemma-10: body_md -->
\label{lem:Mk-convergence}
    The double summation
    \eqref{eq:Mk-bessel} converges absolutely and locally uniformly for
    $y\in(0,\infty)$. Moreover,
    \begin{align} \label{eq:Mk-at-zero}
        \mathcal{M}_k(y)\ll_k y^{\frac{1}{4}}
        \qquad \text{uniformly over $y \in (0,\infty)$},
    \end{align}
    so $\mathcal{M}_k$ extends continuously to $y=0$ by setting $\mathcal{M}_k(0)=0$.
<!-- end lemma-10 -->
<!-- proof-10: body_md -->
Applying the bounds
$J_{k-1}(x)\ll_k x^{\frac{-1}{2}}$ and $Q(d)=\frac{b(d)}{d^2}\ll d^{-2}$, we obtain that
\begin{align}
\sum_{s\ge1}
\sum_{d\ge 1}
\left|
\alpha\sqrt y\,\frac{Q(d)}s
J_{k-1}\left(\frac{4\pi s\sqrt y}{d}\right)
\right|
&\ll_k y^{\frac{1}{4}}
\left(\sum_{d\ge1}Q(d)\sqrt d\right)
\left(\sum_{s\ge1}s^{\frac{-3}{2}}\right) \\
&\ll_k y^{\frac{1}{4}} \qquad \text{uniformly over $y \in (0,\infty)$.}
\end{align}
This proves the desired result.

<!-- theorem-4: body_md -->
For $c>1$ and $k\ge6$ even, define the smoothing of $\mathcal{M}_k(y)$ as 
    \begin{align}
        \mathcal{M}_k^{c}(y)
        :=
        \frac{\displaystyle\int_1^c \mathcal{M}_k\!\left(\frac{y}{u}\right)u\, du}
        {\displaystyle\int_1^c u\, d u}.
    \end{align}
    Then $\mathcal{M}_k^{c}$ is continuous on $[0,\infty)$ with $\mathcal{M}_k^{c}(0)=0$ and as $y\to\infty$,
    \begin{align}
        \mathcal{M}_k^{c}(y)=\frac{1}{2}+O_{k,c}\!\left(y^{\frac{-1}{12}}\right).
    \end{align}
<!-- end theorem-4 -->
<!-- proof-11: body_md -->
Continuity follows immediately from Lemma \ref{lem:Mk-convergence}. So it remains to determine the behavior of $\mathcal{M}^{c}_k(y)$ as $y\to\infty$.
We have
\begin{align}
&\int_1^c
\mathcal{M}_k\!\left(\frac{y}{u}\right)u\,du \\
&=
\int_1^c
\alpha
\sqrt{\frac{y}{u}}
\sum_{s\ge1} \sum_{d\ge1}
\frac{Q(d)}{s}
J_{k-1}\!\left(
\frac{4\pi s}{d}
\sqrt{\frac{y}{u}}
\right)
u\,du
\qquad \text{(by \eqref{eq:Mk-bessel})}
\\
&=
\alpha \sqrt{y}
\sum_{s\ge1} \sum_{d\ge1}
\frac{Q(d)}{s}
\int_1^c
\sqrt{u}\,
J_{k-1}\!\left(\frac{4\pi s\sqrt{y}}{d\sqrt{u}}\right)\,du
\\
&=
2\alpha \sqrt y
\sum_{s\ge1} \sum_{d\ge1}
\frac{Q(d)}{s}
\int_{\frac{1}{\sqrt c}}^{1}
\frac{1}{w^4}
J_{k-1}\!\left(\frac{4\pi s \sqrt{y}}{d} w\right)\,dw
\qquad\quad
\text{\big(letting $w=\tfrac{1}{\sqrt u}$\big)}
\\
&=
2 \alpha \sqrt y
\sum_{s\ge1} \sum_{d\ge1}
\frac{Q(d)}{s} \!\left(\frac{4\pi s \sqrt{y}}{d}\right)^3
\int_{\frac{4\pi s\sqrt y}{d\sqrt c}}^{\frac{4\pi s\sqrt y}{d}}
\frac{J_{k-1}(t)}{t^4}\,dt
\qquad\ \,
\text{\big(letting $t=\tfrac{4\pi s\sqrt y}{d} w$\big)}
\\
&=
\frac{\alpha}{2\pi}
\sum_{s\ge1} \sum_{d\ge1}
\frac{d Q(d)}{s^2} \!\left(\frac{4\pi s \sqrt{y}}{d}\right)^4
\!\left[
\frac{
J\!\left(\frac{4\pi s\sqrt y}{d}\right)
}{
\!\left(\frac{4\pi s\sqrt y}{d}\right)^4
}
-
\frac{
J\!\left(\frac{4\pi s\sqrt y}{d\sqrt c}\right)
}{
\!\left(\frac{4\pi s\sqrt y}{d\sqrt c}\right)^4
}
\right]
\qquad\quad\!\!
\text{(by \eqref{eq:J-definition})}
\\
&=
\frac{\alpha}{2\pi}
\sum_{s\ge1} \sum_{d\ge1}
\frac{dQ(d)}{s^2}
\!\left[
J\!\left(\frac{4\pi s\sqrt y}{d}\right)
-
c^2
J\!\left(\frac{4\pi s\sqrt y}{d\sqrt c}\right)
\right]
\\
&=
\frac{\alpha}{2\pi}
\sum_{s\ge1}\frac{1}{s^2}
\!\left[
    S\!\left(4\pi s\sqrt y\right)
    -
    c^2S\!\left(\frac{4\pi s \sqrt y}{\sqrt c}\right)
\right] \\
&=
\frac{\alpha}{2\pi}
\sum_{s\ge1}\frac{1}{s^2}
\!\left[
    \frac{-\Delta}{4} 
    - c^2 \frac{-\Delta}{4}
    +O_{k,c}\!\left(\!\left(s \sqrt y\right)^{\frac{-1}{6}}\right)
\right] 
\qquad \text{(by Proposition \ref{prop:S-estimate})}
\\
&=
\frac{\alpha}{2\pi} \cdot \zeta(2) \cdot \frac{(c^2-1)\Delta}{4} + O_{k,c}\!\left(y^\frac{-1}{12}\right) \\
&= 
\frac{c^2-1}{4}
+O_{k,c}\!\left(y^{\frac{-1}{12}}\right). \qquad \text{(since $\textstyle \Delta = \frac{2\pi}{\alpha \zeta(2)}$)}
\end{align}

This then yields
\begin{align}
    \mathcal{M}_k^{\,c}(y)
    =
    \frac{
        \displaystyle
        \int_1^c
        \mathcal{M}_k\!\left(\frac{y}{u}\right)u\,du
    }{
        \displaystyle
        \int_1^c u\,du
    }
    =
    \frac{
        \displaystyle
        \frac{c^2-1}{4}
        +O_{k,c}\!\left(y^{\frac{-1}{12}}\right)
    }{
        \displaystyle
        \frac{c^2-1}{2}
    }
    =
    \frac12
    +
    O_{k,c}\!\left(y^{\frac{-1}{12}}\right),
\end{align}
as desired.
