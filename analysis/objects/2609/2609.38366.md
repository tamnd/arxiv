<!-- theorem-1: body_md -->
We have ${\rm Spec}'_R({\rm ZM}(m,n,r))=$
\begin{equation}
\left\{\!\displaystyle \frac{1}{m}\displaystyle \sum_{v=0}^{m-1}\!\!\!\!\!\!\!\!\!\sum_{^{\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,0\leq u\leq n-1}_{^{\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\frac{n}{(n,x-y)}\mid u}_{j_{(y_1,y_2,y)}(u,v)\mid i_{(x_1,x_2)}(u,v)}}}\!\!\!\!\!\!\!\!\!\!\!\frac{(m,[yu]_r)}{o_{\frac{(m,[yu]_r)}{j_{(y_1,y_2,y)}(u,v)}}(r)}; (x_1,x_2,x),(y_1,y_2,y)\in A_{(m,n,r)}\!\right\}\!.\nonumber\vspace{2mm}
\end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
If $n$ is prime, then we have
\begin{equation}
{\rm Spec}'_R({\rm ZM}(m,n,r))={\rm Spec}_R({\rm ZM}(m,n,r))=\left\{n-1+\displaystyle \frac{S}{n}\right\}\!,\nonumber
\end{equation}where
\begin{equation}
S=\displaystyle \sum_{u=0}^{n-1}\,(m,[u]_r).\nonumber
\end{equation}
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
Let $\varphi=\varphi_{(x_1,x_2,x)},\psi=\psi_{(y_1,y_2,y)}\in{\rm Aut}({\rm ZM}(m,n,r))$. In order to compute $R(\varphi,\psi)$, we will apply Burnside's lemma to the action of ${\rm ZM}(m,n,r)$ on ${\rm ZM}(m,n,r)$ given by
\begin{equation}
g\circ x=\psi(g)x\varphi(g)^{-1}, \forall\, g,x\in{\rm ZM}(m,n,r).\nonumber
\end{equation}Let $g=b^ua^v$ and $x=b^{\alpha}a^{\beta}$, where $0\leq u,\alpha<n$ and $0\leq v,\beta<m$. We have
\begin{align*}
&Fix(g)=\{x\in{\rm ZM}(m,n,r); \varphi(g)=x^{-1}\psi(g)x\}\\
&=\{b^{\alpha}a^{\beta}\in{\rm ZM}(m,n,r); b^{xu}a^{x_1v+x_2[u]_r}=b^{yu}a^{r^{\alpha}(y_1v+y_2[u]_r)-\beta(r^{yu}-1)}\}\\
&=\{b^{\alpha}a^{\beta}\in{\rm ZM}(m,n,r); b^{xu}=b^{yu} \mbox{ and } a^{x_1v+x_2[u]_r}=a^{r^{\alpha}(y_1v+y_2[u]_r)-\beta(r^{yu}-1)}\}.\nonumber
\end{align*}We remark that
\begin{equation}
b^{xu}=b^{yu}\Leftrightarrow n\mid (x-y)u\Leftrightarrow\frac{n}{(n,x-y)}\mid u\nonumber
\end{equation}and
\begin{equation}
a^{x_1v+x_2[u]_r}\!\!=a^{r^{\alpha}(y_1v+y_2[u]_r)-\beta(r^{yu}-1)}\Leftrightarrow m\mid (r^{\alpha}y_1-x_1)v-\beta(r^{yu}-1)+(r^{\alpha}y_2-x_2)[u]_r.\nonumber
\end{equation}For fixed $u,v$ and $\alpha$, the last congruence has solutions $\beta$ if and only if 
\begin{equation}
(m,[yu]_r)\mid (r^{\alpha}y_1-x_1)v+(r^{\alpha}y_2-x_2)[u]_r,
\end{equation}since $r^{yu}-1=[yu]_r(r-1)$ and $(m,r-1)=1$. Moreover, if it has solutions, then there are exactly $(m,[yu]_r)$ solutions. Note that (1) is equivalent with
\begin{equation}
(m,[yu]_r)\mid r^{\alpha}i_{(y_1,y_2)}(u,v)-i_{(x_1,x_2)}(u,v).
\end{equation}For fixed $u,v$, (2) has solutions $\alpha$ if and only if 
\begin{equation}
(m,[yu]_r,i_{(y_1,y_2)}(u,v))\mid i_{(x_1,x_2)}(u,v),\nonumber
\end{equation}which means
\begin{equation}
j_{(y_1,y_2,y)}(u,v)\mid i_{(x_1,x_2)}(u,v).\nonumber
\end{equation}Also, it is easy to see that (2) has the same number of solutions $\alpha$ as
\begin{equation}
\frac{(m,[yu]_r)}{j_{(y_1,y_2,y)}(u,v)}\mid r^{\alpha}-1,\nonumber
\end{equation}that is as\vspace{-2mm}
\begin{equation}
o_{\frac{(m,[yu]_r)}{j_{(y_1,y_2,y)}(u,v)}}(r)\mid\alpha.
\end{equation}Obviously, (3) has $\frac{n}{o_{\frac{(m,[yu]_r)}{j_{(y_1,y_2,y)}(u,v)}}(r)}$ solutions and so we obtain
\begin{align*}
&\hspace{40mm}R(\varphi,\psi)=\displaystyle \frac{1}{mn}\displaystyle \sum_{g\in{\rm ZM}(m,n,r)}\text{card}(Fix(g))\\
&=\displaystyle \frac{1}{mn}\displaystyle \sum_{(u,v)}\text{card}(\{(\alpha,\beta); b^{xu}=b^{yu} \mbox{ and } a^{x_1v+x_2[u]_r}=a^{r^{\alpha}(y_1v+y_2[u]_r)-\beta(r^{yu}-1)}\})\\
&=\displaystyle \frac{1}{mn}\sum_{v=0}^{m-1}\sum_{^{\,0\leq u\leq n-1}_{\frac{n}{(n,x-y)}\mid u}}\sum_{\alpha=0}^{n-1}\text{card}(\{0\leq\beta<m; m\!\mid\! (r^{\alpha}y_1{-}x_1)v{-}\beta(r^{yu}{-}1){+}(r^{\alpha}y_2{-}x_2)[u]_r\})\\
&=\displaystyle \frac{1}{mn}\sum_{v=0}^{m-1}\sum_{^{\,0\leq u\leq n-1}_{\frac{n}{(n,x-y)}\mid u}}\!\sum_{^{\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,0\leq \alpha\leq n-1}_{\,\,\,(m,[yu]_r)\mid (r^{\alpha}y_1-x_1)v+(r^{\alpha}y_2-x_2)[u]_r}}\!\!\!(m,[yu]_r)\\
&=\displaystyle \frac{1}{mn}\sum_{v=0}^{m-1}\sum_{^{\,0\leq u\leq n-1}_{\frac{n}{(n,x-y)}\mid u}}\!\!\!(m,[yu]_r)\text{card}(\{0\leq\alpha<n; (m,[yu]_r)\mid (r^{\alpha}y_1{-}x_1)v{+}(r^{\alpha}y_2{-}x_2)[u]_r\})\\
&=\displaystyle \frac{1}{mn}\displaystyle \sum_{v=0}^{m-1}\!\!\!\!\!\!\!\!\!\sum_{^{\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,0\leq u\leq n-1}_{^{\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\frac{n}{(n,x-y)}\mid u}_{j_{(y_1,y_2,y)}(u,v)\mid i_{(x_1,x_2)}(u,v)}}}\!\!\!\!\!\!\!\!\!\!\!(m,[yu]_r)\,\frac{n}{o_{\frac{(m,[yu]_r)}{j_{(y_1,y_2,y)}(u,v)}}(r)}\\
&=\displaystyle \frac{1}{m}\displaystyle \sum_{v=0}^{m-1}\!\!\!\!\!\!\!\!\!\sum_{^{\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,0\leq u\leq n-1}_{^{\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\,\frac{n}{(n,x-y)}\mid u}_{j_{(y_1,y_2,y)}(u,v)\mid i_{(x_1,x_2)}(u,v)}}}\!\!\!\!\!\!\!\!\!\!\!\frac{(m,[yu]_r)}{o_{\frac{(m,[yu]_r)}{j_{(y_1,y_2,y)}(u,v)}}(r)}\,,\nonumber
\end{align*}as desired.
<!-- end proof-1 -->
<!-- proof-2: body_md -->
If $n$ is prime, then $d=n$ and so the conditions $0\leq x,y<n$ and $x,y\equiv 1\, ({\rm mod}\, d)$ imply $x=y=1$ for any $\varphi=\varphi_{(x_1,x_2,x)},\psi=\psi_{(y_1,y_2,y)}\in{\rm Aut}({\rm ZM}(m,n,r))$. Thus, we have 
\begin{equation}
\frac{n}{(n,x-y)}=1\mid u, \mbox{ for all } 0\leq u<n.\nonumber 
\end{equation}We also obtain that
\begin{equation}
j_{(y_1,y_2,1)}(u,v)=(m,[u]_r,y_1v+y_2[u]_r)=(m,[u]_r,y_1v)=(m,[u]_r,v)\nonumber
\end{equation}divides $([u]_r,v)$ and consequently it divides
\begin{equation}
i_{(x_1,x_2)}(u,v)=x_1v+x_2[u]_r,\nonumber
\end{equation}for all $0\leq v<m$ and $0\leq u<n$. Then
\begin{equation}
R(\varphi,\psi)=\displaystyle \frac{1}{m}\displaystyle \sum_{v=0}^{m-1}\sum_{u=0}^{n-1}\frac{(m,[u]_r)}{o_{\frac{(m,[u]_r)}{(m,[u]_r,v)}}(r)}=n-1+\displaystyle \frac{S}{n}\nonumber
\end{equation}by Corollary 1.3 of \cite{2}. This completes the proof.
