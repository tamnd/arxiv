We write $\mathbb{N} \triangleq \{1, 2, \ldots\}$ and $\mathbb{N}_0 \triangleq \{0, 1, 2, \ldots\}$ for the positive and non-negative integers, respectively. 
$\mathbb{R}$ denotes the set of real numbers. 
We use calligraphic letters, as in $\mathcal{P}$ or $\mathcal{C}$, for matrices/operators on the space of Q-functions, with $\mathcal{I}$ denoting the identity matrix/operator.  
For a vector $V\in\mathbb{R}^d$, we use $\|V\|_p$ to denote its $p$-norm for $p\in\{1,2,\infty\}$. 
For a matrix $\mathcal{M} \in \mathbb{R}^{d\times m}$, we use $\|\mathcal{M}\|_{\infty}$ for its operator $\infty$-norm, that is, the maximum absolute row sum of $\mathcal{M}$.

\label{lem:convergence-h}
	Consider the $n$-step Harutyunyan operator~\eqref{eq:off-multi-lambda} and the corresponding iterations in~\eqref{eq:HQL-truncated-def}.
	\begin{itemize}
		\item \emph{(Policy evaluation)}
            The iterates of \emph{HI($\pi$)} satisfy $\left\Vert Q_\ell - Q^\pi \right\Vert_\infty \leq \alpha_{\mathrm{e}}^\ell \left\Vert Q_0 - Q^\pi \right\Vert_\infty$ for all $\ell \in \mathbb{N}_0$, where
            \[
            \alpha_{\mathrm{e}} \triangleq (\gamma\lambda)^{n-1}\bigl(\gamma - \eta_{\epsilon}\bigr) + \eta_{\epsilon}.
            \]
            In particular, for $n\geq2$, $\alpha_{\mathrm{e}} < 1$ and hence $Q_\ell \to Q^\pi$ if
            \begin{equation}\label{eq:epsilon-HI-pi}
            \epsilon < \epsilon_{\max} \triangleq \frac{1-\gamma}{\gamma\lambda}\cdot\frac{1-(\gamma\lambda)^n}{1-(\gamma\lambda)^{n-1}}. 
            \end{equation}
		\item \emph{(Control)}
            The iterates of \emph{HI} satisfy $\left\Vert Q_\ell - Q^\star \right\Vert_\infty \leq \beta_{\mathrm{c}}^\ell \left\Vert Q_0 - Q^\star \right\Vert_\infty$ for all $\ell \in \mathbb{N}_0$, where
            \[
            \beta_{\mathrm{c}} \triangleq \eta_2 - \frac{(1+\gamma)(\gamma\lambda)^n}{1-\gamma\lambda}.
            \]
            In particular, $\beta_{\mathrm{c}} < 1$ and hence $Q_\ell \to Q^\star$ if 
	            \begin{equation}\label{eq:lambda-HI-imp}
                \lambda < \frac{1-\gamma}{2\gamma} + \frac{(1+\gamma)(\gamma\lambda)^n}{2\gamma}.
	            \end{equation}
			\end{itemize}

<!-- rem-1: body_md -->
The implicit condition~\eqref{eq:lambda-HI-imp} can be characterized explicitly as 
\begin{equation}\label{eq:lambda-HI-exp}
    \lambda < \lambda_{\max} \triangleq \left\{\begin{array}{ll}
         1/\gamma, &  n=1, \\
         \overline{x}/\gamma \in\big(\frac{1-\gamma}{2\gamma},\frac{1-\gamma}{\gamma(1+\gamma)}\big], &  n\geq 2,
    \end{array}\right.
\end{equation}
where, for $n\geq 2$, $\overline{x}>0$ uniquely solves the equation $\sum_{i=1}^{n-1} x^{i} = \frac{1-\gamma}{1+\gamma}$.
Observe that the upper bound $\lambda_{\max}$ is a decreasing function of $n$. 
Moreover, $\lambda_{\max} > 1$, i.e., the condition becomes trivial, if any of the following conditions holds:
\begin{equation*}
    \text{\emph{(i)} }n=1, \quad  \text{\emph{(ii)} }\gamma \leq\frac{1}{3},\quad  \text{\emph{(iii)} }n < \frac{\log(3\gamma-1)- \log(1+\gamma)}{\log\gamma}\ \;\; \text{for} \;\; \gamma>\frac{1}{3}.
\end{equation*}

\label{lem:lk-aperiodicity}
	Let $\pi\in\bm{\mathrm{\Pi}}$ and suppose that $\mathcal{P}^\pi$ is aperiodic with mixing rate $\sigma_{\pi} \in (0,1)$ and mixing time $n_{\pi} \in\mathbb{N}$.

\label{thm:lkql-value-learn}
	Consider the iterates $\{(Q_k, U_k)\}_{k \geq 0}$ of \emph{LKQL}~\eqref{eq:lkql-update} in a finite MDP with parameters $n\in\mathbb{N}$, $\lambda\in(0,1]$, $\tau\in[0,1)$, initialization $Q_0\in\bm{\mathrm{Q}}$, $U_0=0\in\bm{\mathrm{Q}}$, and TD errors \eqref{eq:td-error-q} and \eqref{eq:u-td-error} computed at each iteration $k\in\mathbb{N}_0$ along a rollout generated from reset distribution $\rho\in\Delta(\bm{\mathrm{S}}\times\bm{\mathrm{A}})$ and behavior policy $\mu\in\bm{\mathrm{\Pi}}$.
	Assume further that
	\begin{enumerate}
		\item the step sizes $\{\alpha_k\}, \{\beta_k\}$ are positive and satisfy $\begingroup\textstyle \sum\endgroup_k \alpha_k = \begingroup\textstyle \sum\endgroup_k \beta_k = \infty$, $\begingroup\textstyle \sum\endgroup_k(\alpha_k^2+\beta_k^2) < \infty$, and $\alpha_k/\beta_k \to 0$;

		\item $\lambda < \frac{1-\gamma}{2\gamma}$.
	\end{enumerate}
	Then, $Q_k \to Q^\star$ and $U_k \to 0$ almost surely.

<!-- thm-4: body_md -->
\label{thm:ttsa-ode}
    \emph{\cite[\S8.1]{borkar2023stochastic}}
	Let $\{\bm{\mathrm{F}}_k\}_{k\geq0}$ be a filtration, and consider iterates $(X_k,Y_k)\in\mathbb{R}^d\times\mathbb{R}^m$ satisfying
	\begin{align*}
		X_{k+1} & = X_k + \alpha_k\bigl(H(X_k,Y_k)+M^X_{k+1}\bigr), \\
		Y_{k+1} & = Y_k + \beta_k\bigl(G(X_k,Y_k)+M^Y_{k+1}\bigr).
	\end{align*}
	Assume:
	\begin{enumerate}[label=(\roman*)]
		\item The step sizes are positive and satisfy
		      \begin{equation*}
			      \begingroup\textstyle \sum\endgroup_k \alpha_k = \begingroup\textstyle \sum\endgroup_k \beta_k = \infty, \qquad
			      \begingroup\textstyle \sum\endgroup_k(\alpha_k^2 + \beta_k^2) < \infty, \qquad
			      \alpha_k/\beta_k \to 0.
		      \end{equation*}
		\item The maps $H$ and $G$ are Lipschitz.
		\item $\{M^X_{k+1}\}$ and $\{M^Y_{k+1}\}$ are martingale-difference noise terms with respect to $\{\bm{\mathrm{F}}_k\}$ and, for some $K<\infty$,
		      \begin{equation*}
			      \mathds{E}\bigl[\left\Vert M^X_{k+1} \right\Vert^2 + \left\Vert M^Y_{k+1} \right\Vert^2 \mid \bm{\mathrm{F}}_k\bigr]
			      \leq K\bigl(1+\left\Vert X_k \right\Vert^2+\left\Vert Y_k \right\Vert^2\bigr).
		      \end{equation*}
		\item The iterates are almost surely bounded, that is, $\sup_k(\left\Vert X_k \right\Vert+\left\Vert Y_k \right\Vert)<\infty$.
		\item For every fixed $X$, the ODE $\dot{Y}(t)=G(X,Y(t))$ has a unique globally asymptotically stable equilibrium $Y^\star(X)$, and $Y^\star$ is Lipschitz.
		\item The ODE $\dot{X}(t)=H(X(t),Y^\star(X(t)))$ has a unique globally asymptotically stable equilibrium $X^\star$.
	\end{enumerate}
	Then $X_k\to X^\star$ and $Y_k\to Y^\star(X^\star)$ almost surely.

\label{lem:fast-gas}
	For any $V\in\bm{\mathrm{Q}}$ and diagonal operator $\mathcal{B}:\bm{\mathrm{Q}}\to\bm{\mathrm{Q}}$ with diagonal entries in $(0,1]$, the ODE
	\begin{equation*}
		\dot{U}(t)=\mathcal{B}\bigl(V+\tau\mathcal{P}^{\mu}U(t)-U(t)\bigr)
	\end{equation*}
	has the unique globally asymptotically stable equilibrium $U^\dagger=(\mathcal{I}-\tau\mathcal{P}^{\mu})^{-1}V$.

Recall that $\tau\in[0,1)$ and $\mathcal{P}^{\mu}$ is stochastic.
	It follows that, for fixed $V$, the map $U \mapsto V+\tau\mathcal{P}^{\mu}U$ is a $\tau$-contraction in $\infty$-norm with fixed point $U^\dagger$.
	Applying the argument in \citet[\S6.4]{borkar2023stochastic} for diagonally scaled ODEs, we conclude that $U^\dagger$ is the unique globally asymptotically stable equilibrium.

\label{lem:slow-gas}
	Let $\widehat{\mathcal{T}}\colon \bm{\mathrm{Q}} \to \bm{\mathrm{Q}}$ be $\gamma$-Lipschitz in $\infty$-norm with fixed point $Q^\dagger$, and $\mathcal{B}:\bm{\mathrm{Q}}\to\bm{\mathrm{Q}}$ be a diagonal operator with diagonal entries in $(0,1]$.
	If $\lambda < (1-\gamma)/(2\gamma)$, the ODE
	\begin{equation*}
		\dot{Q}(t) = \mathcal{B}\,\bigl(\mathcal{C}_{n,\lambda}^\mu + \xi\mathcal{A}_\tau^\mu\bigr)\bigl(\widehat{\mathcal{T}}Q(t) - Q(t)\bigr)
	\end{equation*}
	has $Q^\dagger$ as the unique globally asymptotically stable equilibrium.
<!-- end lem-4 -->
<!-- proof-2: body_md -->
The map $Q \mapsto Q + (\mathcal{C}_{n,\lambda}^\mu + \xi\mathcal{A}_\tau^\mu)(\widehat{\mathcal{T}}Q - Q)$ is an $\eta_2$-contraction in $\infty$-norm with fixed point $Q^\dagger$, using the same arguments as in the proof of Theorem~\ref{thm:convergence-control}.
	In particular, observe that the operator $\mathcal{A}_\tau^\mu$ shares the same norm identities used there for $\mathcal{P}_\infty^\mu$.
	Furthermore, by \citet[Thm.~4.2]{borkar1997analog}, for any $\infty$-norm contraction $\mathcal{F}$ with fixed point $X^\star$ and any scalar $a > 0$, the ODE $\dot{X} = a(\mathcal{F}X - X)$ has $X^\star$ as its unique globally asymptotically stable equilibrium.
	Combining this with the argument in \citet[\S6.4]{borkar2023stochastic} for diagonally scaled ODEs, we conclude that $Q^\dagger$ is globally asymptotically stable.
