<!-- problem-1: body_md -->
Given a failure probability $\delta \in (0,1)$, and $T$ gradient queries (samples), compute an estimate $x_T$ of $x^*$, and quantify an error bound of the form $f(x_T)- f(x^*) \leq e(T, \delta)$, which holds with probability at least $1-\delta.$
<!-- end problem-1 -->
<!-- assumption-1: body_md -->
\label{ass:smoothness}
    The objective function $f$ is $L$-smooth, i.e., for any $x,y\in\mathbb{R}^d$, the following holds:
\begin{equation}
    \left\| \nabla f(x)-\nabla f(y)\right\|_2\leq L\left\| x-y\right\|_2,
\end{equation}
where $\nabla(\cdot)$ denotes the (full) gradient operator.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:SC}
The objective function $f$ is $\mu$-strongly convex, i.e., for any $x,y\in\mathbb{R}^d$, we have
\begin{equation}
    f(y)\geq f(x)+\left\langle \nabla f(x),\, y-x\right\rangle + \frac{\mu}{2}\left\| x-y\right\|_2^2.
\end{equation}
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:unbiased}
The stochastic gradient $g(x,\xi)$ is unbiased, i.e., 
\begin{equation}
    \mathbb{E}_\xi\left[ g(x,\xi)\right]=\nabla f(x), \forall x \in \mathbb{R}^d. \label{eqn:unbiased}
\end{equation}
<!-- end assumption-3 -->
<!-- assumption-4: body_md -->
\label{ass:subG}
The norm of the noise in the stochastic gradient $g(x,\xi)$ is sub-Gaussian, i.e., for any given $x\in\mathbb{R}^d$, the following holds:
\begin{equation}
    \mathbb{E}_\xi\left[ \exp\left( \left\| g(x,\xi)-\nabla f(x)\right\|_2^2/\sigma^2\right)\right]\leq \exp\left( 1\right),\label{eqn:subG}
\end{equation}
where $\sigma^2>0$  is some constant known as the variance proxy.
<!-- end assumption-4 -->
<!-- lemma-1: body_md -->
\label{lem:onestep}
The following holds for any $t\ge0$ by choosing $\alpha L\le 1/2$:
\begin{equation}
\medmath{
    r_{t+1}\leq r_t - \frac{\alpha}{2}\left\| \nabla f(x_t)\right\|_2^2-\alpha\left\langle \nabla f(x_t),\, e_t\right\rangle+L\alpha^2\left\| e_t\right\|_2^2.}\label{eqn:onestep}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since the objective function $f$ is $L$-smooth, we have
\begin{equation}
\label{eqn:OG_onestep}
\medmath{
\begin{aligned}
    r_{t+1}&\leq r_t+\left\langle \nabla f(x_t),\, x_{t+1}-x_t\right\rangle+\frac{L}{2}\left\| x_{t+1}-x_t\right\|_2^2\\
    &\overset{(a)}{=}r_t-\alpha \left\langle \nabla f(x_t),\, g(x_t,\xi_t)\right\rangle+\frac{L\alpha^2}{2}\left\| g(x_t,\xi_t)\right\|_2^2\\
    &\overset{(b)}{\leq} r_t-\alpha\left\| \nabla f(x_t)\right\|_2^2-\alpha\left\langle \nabla f(x_t),\, e_t\right\rangle\\
    &\ +L\alpha^2\left\| \nabla f(x_t)\right\|_2^2+L\alpha^2\left\| e_t\right\|_2^2\\
    &\overset{(c)}{\leq} r_t - \frac{\alpha}{2}\left\| \nabla f(x_t)\right\|_2^2-\alpha\left\langle \nabla f(x_t),\, e_t\right\rangle+L\alpha^2\left\| e_t\right\|_2^2,
\end{aligned}}
\end{equation}
where $(a)$ follows from the update rule in~\eqref{eqn:SGD}, $(b)$ holds due to the definition of $e_t$ and Jensen's inequality, and $(c)$ is the result of choosing $\alpha L\leq 1/2$. The proof is complete.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:sn} Given any $\delta\in(0, 1)$, with probability (w.p.) at least $1-\delta$, we have:
\begin{equation}
    \left\| e_t\right\|_2^2\leq \sigma^2\log \left(\frac{T\exp(1)}{\delta} \right),\quad \forall t\ge 0\label{eqn:sqnm}
\end{equation}
where $\sigma^2$ is as defined in Assumption~\ref{ass:subG}.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Denote $\mathcal{F}_t:=\sigma\left( \xi_0,\xi_1,\cdots, \xi_t\right)$ as the sigma-algebra capturing all randomnesses up to time-step $t$ ($\mathcal{F}_{-1}=\varnothing$ when $t=0$), $\mathbb{P}_t(\cdot):=\mathbb{P}(\cdot\mid\mathcal{F}_{t-1})$, and $\mathbb{E}_t[\cdot]:=\mathbb{E}[\cdot\mid\mathcal{F}_{t-1}]$ as the probability and expectation operators conditioned on $\mathcal{F}_{t-1}$. Then, for any fixed $t\geq 0$ and $s>0$, we have:
\begin{equation}
\medmath{
\begin{aligned}
    \mathbb{P}_t\left( \left\| e_t\right\|_2^2\geq s\right)&=\mathbb{P}_t\left( \exp\left( \frac{\left\| e_t\right\|_2^2}{\sigma^2}\right)\geq\exp\left( \frac{s}{\sigma^2}\right)\right)\\
    &\overset{(a)}{\leq} \exp\left( -\frac{s}{\sigma^2}\right)\mathbb{E}_t\left[ \exp\left( \frac{\left\| e_t\right\|_2^2}{\sigma^2}\right)\right]\\
    &\overset{(b)}{\leq} \exp\left( -\frac{s}{\sigma^2}+1\right),
\end{aligned}
}
\end{equation}
where $(a)$ holds due to Markov's inequality, and $(b)$ uses Assumption~\ref{ass:subG} and the fact that $x_t$ is deterministic conditioned on $\mathcal{F}_{t-1}$. It then readily follows from the tower property that
\begin{equation}
\medmath{
    \mathbb{P}\left( \left\| e_t\right\|_2^2\geq s\right)=\mathbb{E}\left[ \mathbb{P}_t\left( \left\| e_t\right\|_2^2\geq s\right)\right]\leq \exp\left( -\frac{s}{\sigma^2}+1\right).}\label{eqn:subg_proof}
\end{equation}
Setting the R.H.S. of~\eqref{eqn:subg_proof} to $\delta$ yields $s = \sigma^2\log\frac{\exp(1)}{\delta}$. The proof follows by union-bounding over all $T$ iterations.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:onestep_sc}
Define $r^{(k)}:=f(x^{(k)})-f(x^*)$. Suppose Assumptions~\ref{ass:smoothness} to~\ref{ass:subG} hold. Then, given any $\delta\in(0,1)$, there exists a universal constant $c$ such that with $\alpha\leq1/(2L)$, the following holds for \texttt{Batched SGD} w.p. at least $1-\delta$: 
\begin{equation}
\label{eqn:main_decomp}
    r^{(K)} \leq \underbrace{\left( 1-\frac{\alpha\mu}{2}\right)^Kr^{(0)}}_{T_1}+\underbrace{\frac{4c^2\sigma^2}{\mu H}\log\frac{2dT}{\delta}}_{T_2}.
\end{equation}
<!-- end lemma-3 -->
<!-- remark-1: body_md -->
Lemma~\ref{lem:onestep_sc} provides an upper-bound on the function sub-optimality error after $K$ batches (equivalently $T$ samples). The upper-bound in~\eqref{eqn:main_decomp} comprises an exponentially decaying optimization term $T_1$, and a statistical term $T_2$ capturing the effect of noise. A key feature of our approach, enabled by the variance-reduction effect of batching, is that the noise term $T_2$ gets scaled down by the batch size $H$, which is a tunable parameter. When $H=1$, we recover the standard SGD algorithm, where the noise term $T_2$ can no longer be controlled, leading to vacuous bounds, as we had discussed earlier in Section~\ref{subsec:challenges}. 

To see how the additional degree of freedom provided by $H$ can be exploited, observe that $T_1$ decays exponentially in $K$, while the noise term $T_2$ decays linearly with $H$. As such, to balance these two terms, a $K$ on the order of $\mathcal{O}(\log(T))$ suffices. Since $T=KH$, this implies that the batch size $H$ is essentially $T$ up to log factors.
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{thm:sc}
Suppose the conditions in Lemma~\ref{lem:onestep_sc} hold. Then, given any $\delta \in (0,1)$, 
by choosing $K=4\kappa\log T$ and $\alpha=1/(2L)$, the following holds for \texttt{Batched SGD} w.p. at least $1-\delta$:
\begin{equation}
\label{eqn:final_sc}
    r^{(K)}\leq \frac{r^{(0)}}{T} + \frac{16c^2\kappa\sigma^2}{\mu T}\log T\log\frac{2dT}{\delta}, 
\end{equation}
where $\kappa:=L/\mu$ is the condition number.
<!-- end theorem-1 -->
<!-- head-1: body_md -->
We first state a variant of~\cite[Corollary 7]{jin2019short}, which pertains to variance-reduction of norm-sub-Gaussian random vectors.
<!-- end head-1 -->
<!-- lemma-4: body_md -->
\label{lem:VR_subG}
Let random vectors $X_1,\cdots,X_n\in\mathbb{R}^d$ be independent, zero-mean, and norm-sub-Gaussian with variance proxy $\sigma^2>0$, i.e., $\mathbb{E}\left[ X_i\right]=0$, $\mathbb{E}\left[ \exp\left( \left\| X_i\right\|_2^2/\sigma^2\right)\right]\leq \exp(1)$, $\forall i\in[n]$. Then, there exists a universal constant $c$ such that for any $\delta\in(0,1)$, with probability at least $1-\delta$:
\begin{equation}
    \left\| \frac{1}{n}\sum_{i=1}^nX_i\right\|_2\leq c\sqrt{\frac{\sigma^2}{n}\log\frac{2d}{\delta}}.\label{eqn:VR_subG}
\end{equation}
<!-- end lemma-4 -->
<!-- lemma-5: body_md -->
\label{lem:VR_sn}
Define $n_k:=g_k-\nabla f(x^{(k)})$, where $g_k$ is as in~\eqref{eqn:refined_gradient}. Given any $\delta\in(0, 1)$, w.p. at least $1-\delta$, the following holds (for \texttt{Batched SGD}) simultaneously for all $k=0, 1, \ldots, K-1$:
\begin{equation}
\label{eqn:VR_noise}
    \left\| n_k\right\|_2^2\leq \frac{c^2\sigma^2}{H}\log\frac{2dT}{\delta},
\end{equation}
where $c$ is some universal constant.
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
First, fix a batch index $k$. As per the definition of $g_k$ in~\eqref{eqn:refined_gradient}, we can decompose the noise $n_k$ as
\begin{equation}
    n_k=\frac{1}{H}\sum_{\ell=0}^{H-1}n_{k,\ell},
\end{equation}
where each component $n_{k,\ell}$ is defined as
\begin{equation}
    n_{k,\ell} := g(x^{(k)}, \xi_{kH+\ell})-\nabla f(x^{(k)}).
\end{equation}
We then argue that $\{n_{k,\ell}\}_{\ell=0}^{H-1}$ satisfies the conditions in Lemma~\ref{lem:VR_subG}. To that end, first condition on $\mathcal{F}_{kH-1}$ such that $x^{(k)}$ becomes deterministic, and the sole source of randomness in $\{n_{k,\ell}\}_{\ell=0}^{H-1}$ comes from $\{\xi_{kH+\ell}\}_{\ell=0}^{H-1}$, which is an \textbf{independent} sequence. In the special case of $k=0$, we do not need to condition since $x^{(0)}$ is already deterministic.

According to Assumption~\ref{ass:unbiased}, the stochastic gradient $g(x^{(k)}, \xi_{kH+\ell})$ is an unbiased estimate of the true gradient $\nabla f(x^{(k)})$, implying that the noise component $n_{k,\ell}$ is zero-mean conditioned on $\mathcal{F}_{kH-1}$. Moreover, Assumption~\ref{ass:subG} states that the noise in each stochastic gradient is norm-sub-Gaussian with variance proxy $\sigma^2$, i.e.,
\begin{equation} \mathbb{E}_{kH}\left[ \exp\left( \frac{\left\| n_{k,\ell}\right\|_2^2}{\sigma^2}\right)\right]\leq \exp(1),\quad \forall \ell=0,\cdots,H-1,
\end{equation}
where recall that $\mathbb{E}_{kH}[\cdot]:=\mathbb{E}[\cdot\mid\mathcal{F}_{kH-1}]$.
We have thus proved that $\{n_{k,\ell}\}_{\ell=0}^{H-1}$ satisfies the conditions in Lemma~\ref{lem:VR_subG}. Applying Lemma~\ref{lem:VR_subG} then yields: for any $\delta\in(0,1)$,
\begin{equation}
\medmath{
\mathbb{P}_{kH}\left( \left\| \frac{1}{H}\sum_{\ell=0}^{H-1}n_{k,\ell}\right\|_2\leq c\sqrt{\frac{\sigma^2}{H}\log\frac{2d}{\delta}}\right)\geq 1-\delta.
}
\end{equation}
It then follows from using the tower property (similarly as in~\eqref{eqn:subg_proof}) that with probability at least $1-\delta$,
\begin{equation}
\medmath{
    \left\| n_k\right\|_2\leq c\sqrt{\frac{\sigma^2}{H}\log\frac{2d}{\delta}}.} 
\end{equation}
Union bounding over all $K\leq T$ batches and squaring both sides then yields the desired claim.
<!-- end proof-3 -->
<!-- theorem-2: body_md -->
\label{thm:nc}
Suppose Assumptions~\ref{ass:smoothness},~\ref{ass:unbiased},~\ref{ass:subG} hold. Then, given any $\delta\in(0,1)$, there exists a universal constant $c$ such that with $K=\sqrt{T}$ and $\alpha\leq1/(2L)$, the following holds for \texttt{Batched SGD} w.p. at least $1-\delta$: 
\begin{equation}
\frac{1}{K}\sum_{k=0}^{K-1}\left\| \nabla f(x^{(k)})\right\|_2^2\leq \frac{8Lr^{(0)}}{\sqrt{T}}+\frac{8c^2\sigma^2}{\sqrt{T}}\log\frac{2dT}{\delta}.\label{eqn:nonconvex_sgd}
\end{equation}
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
First, note that arriving at~\eqref{eqn:thm_batch_int} only requires smoothness of $f$. As such, telescoping~\eqref{eqn:thm_batch_int} from $k=0$ to $K-1$ and dividing both sides by $\alpha K/4$ yields:
\begin{equation}
\frac{1}{K}\sum_{k=0}^{K-1}\left\| \nabla f(x^{(k)})\right\|_2^2\leq \frac{4\left( r^{(0)}-r^{(K)}\right)}{\alpha K}+\frac{8}{K}\sum_{k=0}^{K-1}\left\| n_k\right\|_2^2.\label{eqn:nc_int}
\end{equation}
Removing the negative term on the R.H.S., selecting $\alpha=1/(2L)$, and plugging in~\eqref{eqn:VR_noise} from Lemma~\ref{lem:VR_sn} into~\eqref{eqn:nc_int} yields w.p. at least $1-\delta$,
\begin{equation}
\frac{1}{K}\sum_{k=0}^{K-1}\left\| \nabla f(x^{(k)})\right\|_2^2\leq \frac{8Lr^{(0)}}{K}+\frac{8c^2\sigma^2}{H}\log\frac{2dT}{\delta}.
\end{equation}
Selecting $K=\sqrt{T}$ then establishes the desired claim.
<!-- end proof-4 -->
<!-- remark-2: body_md -->
Interestingly, to achieve near-optimal rates, while $K=\mathcal{O}(\log(T))$ updates sufficed for the strongly-convex case, the non-convex case requires $K=\sqrt{T}$ updates (from Theorem~\ref{thm:nc}). Thus, favorable structural properties of $f$ allow for less frequent parameter updates.
<!-- end remark-2 -->
<!-- assumption-5: body_md -->
\label{ass:unbiased_fed}
For each agent $i\in[M]$, the stochastic gradient $g_i(x, \xi)$ is unbiased, i.e., $\mathbb{E}_\xi [g_i(x,\xi)] = \nabla f_i(x), \forall x\in\mathbb{R}^d.$
<!-- end assumption-5 -->
<!-- assumption-6: body_md -->
\label{ass:norm_subG_fed}
For each agent $i\in[M]$, the noise in $g_i(x, \xi)$ is norm-sub-Gaussian, i.e., 
\begin{equation}
    \mathbb{E}_\xi\left[ \exp\left( \left\| g_i(x,\xi)-\nabla f_i(x)\right\|_2^2/\sigma^2\right)\right]\leq \exp\left( 1\right), \forall x \in \mathbb{R}^d, 
    \label{eqn:subG_fed}
\end{equation}
where $\sigma^2 > 0$ is the variance proxy.
<!-- end assumption-6 -->
<!-- assumption-7: body_md -->
\label{ass:indep}
The stochastic processes $\{\xi_{i,t}\}$ and $\{\xi_{j,t}\}$ are statistically independent $\forall i\neq j\in[M]$.
<!-- end assumption-7 -->
<!-- remark-3: body_md -->
Departing fundamentally from the standard FL template~\cite{mcmahan, konevcny, mitra2021linear, karimireddy2020scaffold, zhuachieving, mangold2024scafflsa} where multiple local updates are made per round using noisy gradients, the essence of \texttt{Batched FL} lies in making a single precise update using a direction that enjoys an $H$-fold lower variance. This ensures that the local models are always \textbf{synchronized}, thereby preventing biases from heterogeneity in local loss functions, while preserving collaborative benefits. We later show that this mechanism only requires \textbf{logarithmically} many communication rounds for strongly convex objectives.
<!-- end remark-3 -->
<!-- theorem-3: body_md -->
\label{thm:fl_sc}
Suppose the global objective function $f$ in~\eqref{eqn:global_f} satisfies Assumptions~\ref{ass:smoothness},~\ref{ass:SC}, and suppose Assumptions~\ref{ass:unbiased_fed},~\ref{ass:norm_subG_fed},~\ref{ass:indep} hold. Then, given any $\delta\in(0,1)$, there exists a universal constant $c$ such that with $K=4\kappa\log MT$ and $\alpha=1/(2L)$, the following holds for \texttt{Batched FL} w.p. at least $1-\delta$: 
\begin{equation}
    r^{(K)}\leq \frac{r^{(0)}}{MT} + \frac{16c^2\kappa\sigma^2}{\mu MT}\log MT\log\frac{2dT}{\delta}.\label{eqn:sc_fl}
\end{equation}
<!-- end theorem-3 -->
<!-- proof-5: body_md -->
Similar to Lemma~\ref{lem:onestep}, using smoothness of $f$, defining $g_k:=(1/M)\sum_{i\in[M]}g_{i,k}$, $n_k:=g_k - \nabla f(\bar x^{(k)})$ and selecting $\alpha L\leq 1/2$, we have the one-step descent
\begin{equation}
\begin{aligned}
    r^{(k+1)}&\leq r^{(k)}-\frac{\alpha}{4}\left\| \nabla f(\bar x^{(k)})\right\|_2^2+2\alpha\left\| n_k\right\|_2^2.\label{eqn:one-step-fl}
\end{aligned}
\end{equation}

Next, we need to bound the squared-norm term $\left\| n_k\right\|_2^2$, which follows a similar process as in the proof of Lemma~\ref{lem:VR_sn}. Fixing a round $k$, as per the definition of $n_k$, we have 
\begin{equation}
\medmath{
\begin{aligned}
    n_k&=\frac{1}{M}\sum_{i\in[M]}\left( \frac{1}{H}\sum_{\ell=0}^{H-1}g_i(\bar x^{(k)}, \xi_{i,kH+\ell}) - \nabla f_i(\bar x^{(k)})\right)\\
    &=\frac{1}{MH}\sum_{i\in[M]}\sum_{\ell=0}^{H-1}n_{i,k,\ell},
\end{aligned}
}
\end{equation}
where we define $n_{i,k,\ell}:=g_i(\bar x^{(k)}, \xi_{i,kH+\ell}) - \nabla f_i(\bar x^{(k)})$. We now verify that $\{n_{i,k,\ell}\}_{i,\ell}$ satisfies the conditions for invoking the variance-reduction effect in  Lemma~\ref{lem:VR_subG}. To do so, let $\mathcal{F}_{kH-1}:=\{\xi_{i,t}\}_{i\in[M], t\leq kH-1}$ be the sigma-algebra generated by all the randomness across all agents in $[M]$ up to time-step $kH-1$ ($\mathcal{F}_{-1}=\varnothing$). Observe that $\bar x^{(k)}$ is $\mathcal{F}_{kH-1}$-measurable. Conditioned on $\mathcal{F}_{kH-1}$ then, from Assumption~\ref{ass:indep},  $\{n_{i,k,\ell}\}_{i,\ell}$ is \textbf{independent} since the only randomness lies in $\{\xi_{i,kH+\ell}\}_{i,\ell}$; from Assumption~\ref{ass:unbiased_fed}, it is \textbf{zero-mean} due to the unbiasedness of $g_i$'s. Finally, Assumption~\ref{ass:norm_subG_fed} states that each $n_{i,k,\ell}$ is $\sigma^2$\textbf{-norm-sub-Gaussian}, i.e.,
\begin{equation}
    \mathbb{E}\left[ \exp\left( \frac{\left\| n_{i,k,\ell}\right\|_2^2}{\sigma^2}\right)\mid \mathcal{F}_{kH-1}\right]\leq \exp(1),\quad \forall i,\ell.
\end{equation}
Now that we have shown that the noise sequence $\{n_{i,k,\ell}\}_{i,\ell}$ satisfies all the requirements for Lemma~\ref{lem:VR_subG}, invoking it and using the tower property to get rid of the conditioning on $\mathcal{F}_{kH-1}$ yields for any $\delta\in(0,1)$, w.p. at least $1-\delta$:
\begin{equation}
    \left\| n_k\right\|_2\leq c\sqrt{\frac{\sigma^2}{MH}\log\frac{2d}{\delta}}.
\end{equation}
Union bounding over all $K\leq T$ rounds and squaring both sides yields w.p. at least $1-\delta$,
\begin{equation}
    \left\| n_k\right\|_2^2\leq \frac{c^2\sigma^2}{MH}\log\frac{2dT}{\delta}.\label{eqn:sn_fl}
\end{equation}
Plugging this into~\eqref{eqn:one-step-fl} and using the gradient domination property of $f$ yields w.p. at least $1-\delta$,
\begin{equation}
\medmath{
}    r^{(k+1)}\leq \left( 1-\frac{\alpha\mu}{2}\right)r^{(k)}+\frac{2c^2\sigma^2\alpha}{MH}\log\frac{2dT}{\delta}.
\end{equation}
Iterating from $k=0$ to $K-1$ and selecting $\alpha=1/(2L)$ yields w.p. at least $1-\delta$,
\begin{equation}
\medmath{
    r^{(K)}\leq \exp\left( -\frac{K}{4\kappa}\right)r^{(0)}+\frac{4c^2\sigma^2}{\mu MH}\log\frac{2dT}{\delta}.}
\end{equation}
Selecting $K=4\kappa\log MT$ yields the desired claim.
