\label{thm:2-2-level_set_lemma}
  Under \Cref{assum:2-1-minimizers} and \ref{assum:2-2-Lipschitz}, with $\pi_0(\cdot | ξ)$ denoting the uniform distribution $\mathsf{Unif}\prn[\big]{ \mathcal{X}(ξ) }$, we have

    \geq C_0 \cdot \alpha^{d_x},
  \end{equation*}
  where constant $C_0 := \pi^{d_x/2} / \prn[\big]{ \Gamma(d_x/2+1) \mathrm{Vol}\prn[\big]{ \mathcal{X}(ξ) } K^{d_x} }$ is uniform for all parameters $ξ \in \varXi$.

<!-- proof-4: body_md -->
By \Cref{assum:2-2-Lipschitz}, given any $x^{\star} \in L_0(ξ)$, we have
  \begin{equation*}
    \norm*{f(x; ξ) - f(x^{\star}; ξ)} \leq K \norm{x - x^{\star}} \leq \alpha,~ \forall x \in \mathbb{B}\prn*{ x^{\star}, \tfrac{\alpha}{K} }.
  \end{equation*} 
  Therefore, we have $\mathbb{B}\prn*{ x^{\star}, \tfrac{\alpha}{K} } \subset L_{\alpha}$, and thus
  \begin{equation*}
    \pi_0\prn[\big]{ L_{\alpha}(ξ) \big| ξ }
    \geq \pi_0\prn[\big]{ \mathbb{B}\prn[\big]{ x^*, \tfrac{\alpha}{K} } \big| ξ }
    = \frac{\pi^{d_x/2} \prn[\big]{\frac{\alpha}{K}}^d}{\Gamma \prn[\big]{ \tfrac{d_x}{2}+1 } \mathrm{Vol}\prn[\big]{ \mathcal{X}(ξ) }},
  \end{equation*}
  which completes the proof.

<!-- proof-5: body_md -->
Fix a parameter $ξ \in \varXi$, and we proceed to lower bound $\pi^{\star}\prn[\big]{ L_{\alpha}(ξ) \big| ξ }$. Indeed, by definition we have
  \begin{align}
    &\pi^{\star}\prn[\big]{ L_{\alpha}(ξ) \big| ξ } 
    ={} \int_{L_{\alpha}(ξ)} \pi^{\star}(x | ξ) \mathrm{d} x \nonumber\\
    ={}& \frac{\int_{L_{\alpha}(ξ)} \pi_0(x | ξ) \exp\prn[\big]{ -\varLambda f(x; ξ) } \mathrm{d} x}{\int_{\mathcal{X}(ξ)} \pi_0(x | ξ) \exp\prn[\big]{ -\varLambda f(x; ξ) } \mathrm{d} x} \nonumber\\
                                                                                                                                                                                                                                                                                                                                         
    \geq{}& 1 - \frac{\int_{\mathcal{X}(ξ) \setminus L_{\alpha}(ξ)} \pi_0(x | ξ) \exp\prn[\big]{ -\varLambda f(x; ξ) } \mathrm{d} x}{\int_{L_{\alpha}(ξ)} \pi_0(x | ξ) \exp\prn[\big]{ -\varLambda f(x; ξ) } \mathrm{d} x} \nonumber\\
    \geq{}& 1 - \frac{\int_{\mathcal{X}(ξ) \setminus L_{\alpha}(ξ)} \pi_0(x | ξ) \exp\prn[\big]{ -\varLambda f(x; ξ) } \mathrm{d} x}{\int_{L_{\alpha/2}(ξ)} \pi_0(x | ξ) \exp\prn[\big]{ -\varLambda f(x; ξ) } \mathrm{d} x} \nonumber\\
    \geq{}& 1 - \frac{\exp\prn[\big]{ -\varLambda (f^{\star}(ξ) + \alpha) } \pi_0\prn[\big]{ \mathcal{X}(ξ) \setminus L_{\alpha}(ξ) \big| ξ }}{\exp\prn[\big]{ -\varLambda (f^{\star}(ξ) + \alpha/2) } \pi_0\prn[\big]{ L_{\alpha/2}(ξ) \big| ξ }} \nonumber\\
    \geq{}& 1 - \frac{1}{\pi_0\prn[\big]{ L_{\alpha/2}(ξ) \big| ξ }} \cdot \exp(- \alpha \varLambda / 2), \label{eq:3-1-sampling_ideal-pi_bound}
  \end{align}
  where the second inequality is established by the fact that $L_{\alpha / 2}(ξ) \subseteq L_{\alpha}(ξ)$; the third inequality is due to the definition of $\alpha$-level sets such that
  \begin{equation*}
    f(x;ξ) \leq f^{\star}(ξ) + \alpha,~ \forall x \in L_{\alpha}(ξ);
  \end{equation*}
  and the fourth inequality is because $\pi_0\prn[\big]{ \mathcal{X}(ξ) \setminus L_{\alpha}(ξ) \big| ξ } \leq 1$. Note that, by \Cref{thm:2-2-level_set_lemma}, we have a lower bound on the size of the level sets under \Cref{assum:2-1-minimizers} and \ref{assum:2-2-Lipschitz}, i.e.,
  \begin{equation*}
    \pi_0\prn[\big]{ L_{\alpha/2}(ξ) \big| ξ }
    \geq C_0 \cdot \prn*{\frac{\alpha}{2}}^{d_x}.
  \end{equation*}
  Plugging the lower bound back into \eqref{eq:3-1-sampling_ideal-pi_bound}, we have
  \begin{equation}
    \pi^{\star}\prn[\big]{ L_{\alpha}(ξ) \big| ξ }
    \geq 1 - \frac{1}{C_0} \prn*{\frac{2}{\alpha}}^{d_x} \cdot \exp(-\alpha \varLambda / 2).
  \end{equation}
  Therefore, if we select $\varLambda$ such that
  \begin{equation*}
    \frac{1}{C_0} \prn*{\frac{2}{\alpha}}^{d_x} \cdot \exp(-\alpha \varLambda / 2) < \delta,
  \end{equation*}
  or equivalently,
  \begin{equation*}
    \varLambda > \frac{2}{\alpha} \prn*{ d_x \log\prn*{\frac{2}{\alpha}} + \log\prn*{\frac{1}{C_0 \delta}} } = O\prn*{\frac{d_x}{\alpha} \log\prn*{ \frac{1}{\alpha \delta}}},
  \end{equation*}
  we shall then guarantee that
  \begin{equation*}
    \pi^{\star}\prn[\big]{ L_{\alpha}(ξ) \big| ξ } > 1-\delta.
  \end{equation*}
  The proof is done since the above holds for all $ξ \in \varXi$.

    &\hspace{4em} {}+ \prn[\big]{ \widehat{\mathcal{L}}_k(\tilde{s}^{\star}_k) - \widetilde{\mathcal{L}}_k(\tilde{s}^{\star}_k) } \\
    &\leq \prn[\big]{ \widetilde{\mathcal{L}}_k(\hat{s}^{\star}_k) - \widehat{\mathcal{L}}_k(\hat{s}^{\star}_k) } + \prn[\big]{ \widehat{\mathcal{L}}_k(\tilde{s}^{\star}_k) - \widetilde{\mathcal{L}}_k(\tilde{s}^{\star}_k) } \\
    &\leq \sum_{s \in \mathcal{S}} \abs[\big]{ \widetilde{\mathcal{L}}_k(s) - \widehat{\mathcal{L}}_k(s) },
  \end{align*}
  where $\mathcal{S} :=  
  \IfBooleanTF{\hat{s}^{\star}_k, \tilde{s}^{\star}_k}{                            
    \IfNoValueTF{#4}{\brac*{.}}{\raw@set*{.}{#4}} 
  }{ 
    \IfNoValueTF{$}{ 
      \IfNoValueTF{#4}{\brac{.}}{\raw@set{.}{#4}} 
    }{ 
      \IfNoValueTF{#4}{\brac[$]{.}}{\raw@set[$]{.}{#4}} 
    } 
  } 
 For the upper bound, we would like to apply the well-known bound involving Rademacher complexity (\Cref{thm:lemma-Rademacher_property}). For this purpose, we first have to show that the individual losses are uniformly bounded. Note that
  \begin{equation*}
     \norm[\big]{s(x; ξ) - s^{\star}_k(x; ξ)}^2
    \leq 2 \norm[\big]{s(x; ξ)}^2 + 2 \norm[\big]{s^{\star}_k(x; ξ)}^2,~ \forall s \in \mathcal{S},
  \end{equation*}
  where we apply $\norm{a+b}^2 \leq 2\norm{a}^2 + 2\norm{b}^2$ again. Now the first term is directly bounded by \Cref{assum:4-D-diffusion-hypothesis_class}, while the second term can be directly computed as
  \begin{align*}
    \norm[\big]{s^{\star}_k(x; ξ)}
    &= \norm*{\frac{x_k - \exp(-t_k) x_0}{\sigma_k^2}}
    \leq \frac{\norm{x_k} + \norm{x_0}}{\sigma_k^2} \\
    &\leq \frac{2C_x}{1 - \exp(-2t_0)}
  \end{align*}
  by Assumption \ref{assum:2-1-minimizers}. As a conclusion, for any $s \in \mathcal{S}$,
  \begin{equation*}
    \norm[\big]{s(x; ξ) - s^{\star}_k(x; ξ)}^2
    \leq 2 C_{\mathscr{F}}^2 + 2 \prn*{\frac{2C_x}{1 - \exp(-2t_0)}}^2
    =: C_{\mathrm{sc}},
  \end{equation*}
  and consequently $g(ξ) \cdot \norm{s(x; ξ) - s^{\star}_k(x; ξ)}^2 \leq \gamma C_{\mathrm{sc}}$. Then by \Cref{thm:lemma-Rademacher_property} we have, with probability at least $1-\delta/2$,
  \begin{equation*}
    \abs[\big]{ \widetilde{\mathcal{L}}_k(s) - \widehat{\mathcal{L}}_k(s) } \leq 2R(\mathcal{S}) + \gamma C_{\mathrm{sc}} \sqrt{\frac{2 \log(4/\delta)}{N_k}},~ \forall s \in \mathcal{S},
  \end{equation*}
  where $R(\mathcal{S})$ is the Rademacher complexity of the finite hypothesis class $\mathcal{S}$. It is well-known that (see \Cref{thm:lemma-Rademacher_finite_class}), the Rademacher complexity of a bounded finite class $\mathcal{S}$ has a natural upper bound
  \begin{equation*}
    R(\mathcal{S}) \leq \gamma C_{\mathrm{sc}} \sqrt{\frac{\log 4}{N_k}} < \gamma C_{\mathrm{sc}} \sqrt{\frac{2 \log(4/\delta)}{N_k}}.
  \end{equation*}
  Therefore, with probability at least $1-\delta/2$, we have
  \begin{equation*}
    \abs[\big]{ \widetilde{\mathcal{L}}_k(s) - \widehat{\mathcal{L}}_k(s) } \leq 3 \gamma C_{\mathrm{sc}} \sqrt{\frac{2 \log(4/\delta)}{N_k}},~ \forall s \in \mathcal{S},
  \end{equation*}
  and the overall bound now becomes
  \begin{align*}
    \norm[\big]{\hat{s}^{\star}_k(x; ξ) - \tilde{s}^{\star}_k(x; ξ)}^2
    \leq 6 μ \gamma K_{\varTheta} C_{\mathrm{sc}} \sqrt{\frac{2 \log(4/\delta)}{N_k}},
  \end{align*}
  and consequently,
  \begin{equation*}
    E_k(ξ) \leq 12 μ \gamma K_{\varTheta} C_{\mathrm{sc}} \sqrt{\frac{2 \log(4/\delta)}{N_k}} + 2\varepsilon_{\mathscr{F}}.
  \end{equation*}
  In other words, if we take
  \begin{equation*}
    N_h > \frac{288 μ^2 \gamma^2 K_{\varTheta}^2 C_{\mathrm{sc}}^2 \log(4/\delta)}{\varepsilon^2},
  \end{equation*}
  using union bound we can guarantee that,
  \begin{equation*}
    E_k(ξ) < \varepsilon + 2\varepsilon_{\mathscr{F}}
  \end{equation*}
  with probability at least $1-\delta$. This completes the proof.

  Given a hypothesis class $\mathcal{H} =  
  \IfBooleanTF{h}{

 a data distribution $x \sim p$, and a loss function $\ell(h, x)$ that is uniformly bounded by $\abs{\ell(h, x)} \leq c$, consider the expected and empirical losses $\mathcal{L}(h) = \E[x \sim p]{\ell(h, x)}$ and $\widehat{\mathcal{L}}(h) = \tfrac{1}{N} \textstyle\sum_{i=1}^{N} \ell(h, x_i)$. Then, with probability at least $1-\delta$, the error bound
  \begin{equation*}
    \mathcal{L}(h) - \widehat{\mathcal{L}}(h) \leq 2R_N(\mathcal{H}) + c \sqrt{\frac{\log(2/\delta)}{N}}
  \end{equation*}
