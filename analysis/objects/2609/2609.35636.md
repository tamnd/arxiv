<!-- assumption-1: body_md -->
\label{ass:conformity}
On each non-empty interface $\Gamma_{s,r} := \partial\Omega_{s} \cap \partial\Omega_{r}$, with $s \neq r$, we assume that the trace spaces coincide:
 
\begin{equation}
	\mathrm{tr}(X_h^1(\Omega_{s})|_{\Gamma_{s,r}} = 
	\mathrm{tr}(X_h^1(\Omega_{r})|_{\Gamma_{s,r}}.
\end{equation}
<!-- end assumption-1 -->
<!-- definition-1: body_md -->
Let $\mathbb{S}^3_+$ denote the set of symmetric positive definite $3 \times 3$ matrices, equipped with the Euclidean norm. Given a function $\mathcal{A} \in L^{\infty}(\widehat{\Omega},\mathbb{S}^3_+)$ such that
 
\begin{equation*}
	\inf_{\widehat{\boldsymbol{x}} \in \widehat{\Omega}}\left[\lambda_{\mathrm{min}}\left(\mathcal{A}(\widehat{\boldsymbol{x}})\right)\right] >0,
\end{equation*}
 
we define the preconditioner $\mathbf{P}_{\mathcal{A}}$ as the symmetric and positive definite matrix associated to the bilinear form
 
\begin{equation*}
	\mathbf{u}^{\top} \mathbf{P}_{\mathcal{A}} \mathbf{v} = \sum_{k,\ell=1}^{3} \int_{\widehat{\Omega}} u_{k}(\widehat{\boldsymbol{x}}) v_{\ell}(\widehat{\boldsymbol{x}}) \mathcal{A}_{k,\ell}(\widehat{\boldsymbol{x}}) \, \mathrm{d}\widehat{\boldsymbol{x}}
\end{equation*}
 
where $\mathbf{u} = \begin{bmatrix} \mathbf{u}_1^{\top} & \mathbf{u}_2^{\top} & \mathbf{u}_3^{\top} \end{bmatrix}^{\top}$, $\mathbf{v} = \begin{bmatrix} \mathbf{v}_1^{\top} & \mathbf{v}_2^{\top} & \mathbf{v}_3^{\top} \end{bmatrix}^{\top}$, and for $k=1,2,3,$ $\mathbf{u}_k,\mathbf{v}_k$ are the coordinate vectors of $u_k,v_k \in \operatorname{span}(\mathcal{B}_{\mathbf{p}}(\boldsymbol \Xi;k)) $ with respect to the basis $\mathcal{B}_{\mathbf{p}}(\boldsymbol \Xi;k)$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:spectral_equiv}
There exist two positive constants $C_1$ and $C_2$, independent of $\mathbf{\Xi}$ and $\mathbf{p}$, such that  
 
\begin{equation}\label{eq:spectral_equiv}
	C_1 \leq \lambda_{\mathrm{min}}\left(\mathbf{P}_{\mathcal{A}}^{-\frac{1}{2}} {\mathbf M}_\epsilon \mathbf{P}_{\mathcal{A}}^{-\frac{1}{2}}\right) 
	\leq \lambda_{\mathrm{max}}\left(\mathbf{P}_{\mathcal{A}}^{-\frac{1}{2}} {\mathbf M}_\epsilon \mathbf{P}_{\mathcal{A}}^{-\frac{1}{2}}\right) 
	\leq C_2.
\end{equation}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
A generic 1-form in $X^1_{h,0}$ can be seen as $(\iota^1)^{-1}(v)$, with $v \in \,\hspace{-1pt}\widehat{X}^{1}_{h,0}\hspace{-1pt}\,(\Omega)$ defined as
 
\begin{equation*}
	v =\sum_{k=1}^3 v_{k} \mathrm{d}\widehat{x}_k, \qquad v_{k} \in \operatorname{span}(\mathcal{B}_{\mathbf{p}}(\boldsymbol \Xi;k)).
\end{equation*}
 
For $k=1,2,3$, let $\mathbf{v}_k$ be the coordinate vector of $v_{k}$ with respect to the basis ${\mathcal B}_{\mathbf p}(\boldsymbol \Xi;k)$.
By the Courant-Fischer theorem, \eqref{eq:spectral_equiv} is equivalent to find $C_1$ and $C_2$ such that
 
\begin{equation*}
	C_1 \leq \frac{\mathbf{v}^{\top} {\mathbf M}_\epsilon\mathbf{v}}{\mathbf{v}^{\top} \mathbf{P}_{\mathcal{A}} \mathbf{v}} \leq C_2, \qquad \text{for any} \qquad \mathbf{v} = [\mathbf{v}_1^{\top},\mathbf{v}_2^{\top},\mathbf{v}_3^{\top}]^{\top}.
\end{equation*}
 
Using \eqref{eq:mass_eps} and \eqref{eq:mass_eps_kl}, we observe that 
 
\begin{equation*}
	\mathbf{v}^{\top} {\mathbf M}_\epsilon\mathbf{v} = \sum_{k,\ell=1}^{3} \int_{\widehat{\Omega}} v_{k}(\widehat{\boldsymbol{x}}) v_{\ell}(\widehat{\boldsymbol{x}}) \mathcal{C}_{k,\ell}(\widehat{\boldsymbol{x}}) \, \mathrm{d}\widehat{\boldsymbol{x}}.
\end{equation*}
 
Due to the regularity and orientation-preserving assumptions on $\textbf{F}$, and the positivity of $\epsilon$, for all $\widehat{\boldsymbol{x}} \in \widehat \Omega$, the $3 \times 3$ matrix 
 
\begin{equation*}
	\mathcal{C}(\widehat{\boldsymbol{x}}) := \begin{bmatrix}
		\mathcal{C}_{1,1}(\widehat{\boldsymbol{x}}) & \mathcal{C}_{1,2}(\widehat{\boldsymbol{x}}) & \mathcal{C}_{1,3}(\widehat{\boldsymbol{x}}) \\
		\mathcal{C}_{2,1}(\widehat{\boldsymbol{x}}) & \mathcal{C}_{2,2}(\widehat{\boldsymbol{x}}) & \mathcal{C}_{2,3}(\widehat{\boldsymbol{x}}) \\
		\mathcal{C}_{3,1}(\widehat{\boldsymbol{x}}) & \mathcal{C}_{3,2}(\widehat{\boldsymbol{x}}) & \mathcal{C}_{3,3}(\widehat{\boldsymbol{x}})
	\end{bmatrix}
\end{equation*}
 
is symmetric positive definite and we can write
 
\begin{align*}
	\inf_{\widehat{\boldsymbol{x}} \in \widehat{\Omega}}\left[\lambda_{\mathrm{min}}\left(\mathcal{C}(\widehat{\boldsymbol{x}}) \mathcal{A}^{-1}(\widehat{\boldsymbol{x}})\right)\right]	\mathbf{v}^{\top} \mathbf{P}_{\mathcal{A}}\mathbf{v} &\leq \sum_{k,\ell=1}^{3} \int_{\widehat{\Omega}} v_{k}(\widehat{\boldsymbol{x}}) v_{\ell}(\widehat{\boldsymbol{x}}) \mathcal{A}_{k,\ell}(\widehat{\boldsymbol{x}})  \frac{\displaystyle \sum_{k',\ell'=1}^{3} v_{k'}(\widehat{\boldsymbol{x}}) v_{\ell'}(\widehat{\boldsymbol{x}}) {\mathcal{C}}_{k',\ell'}(\widehat{\boldsymbol{x}})}{ \displaystyle \sum_{k',\ell'=1}^{3} v_{k'}(\widehat{\boldsymbol{x}}) v_{\ell'}(\widehat{\boldsymbol{x}}) \mathcal{A}_{k',\ell'}(\widehat{\boldsymbol{x}})} \, \mathrm{d}\widehat{\boldsymbol{x}} \\
	&=\mathbf{v}^{\top} {\mathbf{M}}_{\epsilon}\mathbf{v}.
\end{align*}	
 
Similarly, 
 
\begin{align*}
	\mathbf{v}^{\top} {\mathbf{M}}_{\epsilon}\mathbf{v} &= \sum_{k,\ell=1}^{3} \int_{\widehat{\Omega}} v_{k}(\widehat{\boldsymbol{x}}) v_{\ell}(\widehat{\boldsymbol{x}}) \mathcal{A}_{k,\ell}(\widehat{\boldsymbol{x}}) \frac{ \displaystyle \sum_{k',\ell'=1}^{3} v_{k'}(\widehat{\boldsymbol{x}}) v_{\ell'}(\widehat{\boldsymbol{x}}) {\mathcal{C}}_{k',\ell'}(\widehat{\boldsymbol{x}})}{ \displaystyle \sum_{k',\ell'=1}^{3} v_{k'}(\widehat{\boldsymbol{x}}) v_{\ell'}(\widehat{\boldsymbol{x}}) \mathcal{A}_{k',\ell'}(\widehat{\boldsymbol{x}})} \, \mathrm{d}\widehat{\boldsymbol{x}} \\
	&\leq \sup_{\widehat{\boldsymbol{x}} \in \widehat{\Omega}}\left[\lambda_{\mathrm{max}}\left(\mathcal{C}(\widehat{\boldsymbol{x}}) \mathcal{A}^{-1}(\widehat{\boldsymbol{x}})\right)\right]	\mathbf{v}^{\top} \mathbf{P}_{\mathcal{A}}\mathbf{v}.
\end{align*}	
 
Hence, the thesis holds with
 
\begin{equation*}
	C_1 = \inf_{\widehat{\boldsymbol{x}} \in \widehat{\Omega}}\left[\lambda_{\mathrm{min}}\left(\mathcal{C}(\widehat{\boldsymbol{x}}) \mathcal{A}^{-1}(\widehat{\boldsymbol{x}})\right)\right] \qquad
	C_2 = \sup_{\widehat{\boldsymbol{x}} \in \widehat{\Omega}}\left[\lambda_{\mathrm{max}}\left(\mathcal{C}(\widehat{\boldsymbol{x}}) \mathcal{A}^{-1}(\widehat{\boldsymbol{x}})\right)\right].
\end{equation*}

<!-- remark-1: body_md -->
\label{rem:conditioning}
Theorem~\ref{thm:spectral_equiv} provides the following bound on the condition number of ${\mathbf M}_\epsilon$ under symmetric preconditioning by $\mathbf{P}_{\mathcal{A}}$:
 
\begin{equation*}
	\kappa_2 \left( \mathbf{P}_{\mathcal{A}}^{-\frac{1}{2}} {\mathbf M}_\epsilon \mathbf{P}_{\mathcal{A}}^{-\frac{1}{2}} \right) 
	\leq 
	\frac{\displaystyle \sup_{\widehat{\boldsymbol{x}} \in \widehat{\Omega}} \lambda_{\mathrm{max}} \left( \mathcal{C}(\widehat{\boldsymbol{x}}) \mathcal{A}^{-1}(\widehat{\boldsymbol{x}}) \right)}
	{\displaystyle \inf_{\widehat{\boldsymbol{x}} \in \widehat{\Omega}} \lambda_{\mathrm{min}} \left( \mathcal{C}(\widehat{\boldsymbol{x}}) \mathcal{A}^{-1}(\widehat{\boldsymbol{x}}) \right)},
\end{equation*}
 
with the right-hand side independent of both $\mathbf{\Xi}$ and $\mathbf{p}$.  
As a consequence, the number of iterations required by the preconditioned conjugate gradient (PCG) method to solve a system with ${\mathbf M}_\epsilon$ remains uniformly bounded with respect to $\mathbf{\Xi}$ and $\mathbf{p}$. Moreover, the bound becomes tighter as $\mathcal{A}(\widehat{\boldsymbol{x}})$ more closely approximates $\mathcal{C}(\widehat{\boldsymbol{x}})$.

<!-- theorem-2: body_md -->
\label{thm:additive_schwarz}
	There exists a positive constant $C$, independent of the meshes defined by $\mathbf{\Xi}^{(s)}$ for $s = 1,\ldots,N_{\mathrm{patch}}$, the total number of patches $N_{\mathrm{patch}}$, and the parameters $N_{\mathrm{adj}}$ and $N_{\mathrm{int}}$, but possibly dependent on the spline degrees $\mathbf{p}$, the geometric mapping $\mathbf{F}$, and the physical parameter $\epsilon$, such that
	 
	\begin{equation*}
		\kappa_2 \left( \mathbf{P}_{\mathrm{ad}}^{-1/2} \mathbf{M}_\epsilon \, \mathbf{P}_{\mathrm{ad}}^{-1/2} \right) \leq C N_{\mathrm{adj}} (N_{\mathrm{int}} +1).
	\end{equation*}
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
If the \emph{stable decomposition}, \emph{strengthened Cauchy-Schwarz inequalities}, and \emph{local stability} conditions hold, see~\cite[Assumptions 2.2, 2.3, and 2.4]{toselli2006domain}, then the conclusion follows directly from~\cite[Theorem 2.7]{toselli2006domain}.
 
\begin{itemize}
\item \textbf{Stable decomposition.}
 
We have to prove that there exists $C_0>0$, independent of the meshes $\mathbf{\Xi}^{(s)}$ but possibly dependent on $\mathbf{p}$, $\mathbf{F}$, and $\epsilon$, such that every 1-form $u \in X_{h,0}^1$ admits a decomposition
 
\begin{equation*}
u = \sum_{s=1}^{N_{\mathrm{patch}}} u^{(s)}, \qquad u^{(s)} \in V_{s},
\end{equation*}
 
satisfying
 
\begin{equation}\label{eq:stab_dec}
	\sum_{s=1}^{N_{\mathrm{patch}}} \widetilde{a}_{s}(u^{(s)},u^{(s)}) \leq C_0^2\, a(u,u).
\end{equation}
  
Any $u \in X_{h,0}^1$ can be written as
 
\begin{equation*}
	u = \sum_{i=1}^{N_{\mathrm{dof}}} u_i \beta_i, \quad \beta_i \in \mathcal{B}_{\mathrm{mp}}.
 
\end{equation*}
For each patch $s=1,\ldots,N_{\mathrm{patch}}$, we define
 
\begin{equation*}
	u^{(s)} := \sum_{i=1}^{N_{\mathrm{dof}}} u_i^{(s)} \beta_i,
	\qquad
	u_i^{(s)} :=
	\begin{cases}
		0, & \text{if } \operatorname{supp}(\beta_i) \cap \Omega_{s} = \emptyset,\\[1mm]
		\frac{u_i}{n_i}, & \text{otherwise},
	\end{cases}
\end{equation*}
 
where 
 
\begin{equation*}
	n_i := \#\{ r : \operatorname{supp}(\beta_i) \cap \Omega_{r} \neq \emptyset \}.
\end{equation*}
 
By construction, $u = \sum_{s=1}^{N_{\mathrm{patch}}} u^{(s)}$.
  
The global $L^2_{\epsilon}\Lambda^1$-norm can be written as a sum over the patches:
 
\begin{equation*}
	a(u,u) = (u,u)_{L^2_{\epsilon}\Lambda^1(\Omega)} = \sum_{s=1}^{N_{\mathrm{patch}}} (u|_{\Omega_{s}}, u|_{\Omega_{s}})_{L^2_{\epsilon}\Lambda^1(\Omega_{s})}.
\end{equation*}
 
Since the restriction of each $\beta_i$ to $\Omega_{s}$ coincides, up to a sign, with an element of the local patch basis $\mathcal{B}_{s}^1$, we have
 
\begin{equation*}
	(u|_{\Omega_{s}}, u|_{\Omega_{s}})_{L^2_{\epsilon}\Lambda^1(\Omega_{s})}
	= \mathbf{u}^{\top} \mathbf{R}_{s}^{\top} \mathbf{G}_{s}^{\top} \mathbf{M}^{s}_\epsilon \mathbf{G}_{s} \mathbf{R}_{s} \mathbf{u},
\end{equation*}
 
where $\mathbf{u}$ is the coordinate vector of $u$ with respect to $\mathcal{B}_{\mathrm{mp}}$.\\
 
For the local component $u^{(s)}$, we similarly have
 
\begin{equation*}
	\widetilde{a}_{s}(u^{(s)}, u^{(s)})
	= (u^{(s)}|_{\Omega_{s}}, u^{(s)}|_{\Omega_{s}})_{L^2_{\epsilon}\Lambda^1(\Omega_{s})}
	= \mathbf{u}^{\top} \mathbf{E}^{\top} \mathbf{R}_{s}^{\top} \mathbf{G}_{s}^{\top} \mathbf{M}^{s}_\epsilon \mathbf{G}_{s} \mathbf{R}_{s} \mathbf{E} \mathbf{u},
\end{equation*}
 
where $\mathbf{E} := \mathrm{diag}(1/n_1, \dots, 1/n_{N_{\mathrm{dof}}})$. 
 

Recalling that the restriction matrix $\mathbf{R}_{s}$ selects the degrees of freedom corresponding to $V_s$, we define the local vector $\mathbf{u}_{s} = [u_{i_1}, \dots, u_{i_{N_{\mathrm{dof},s}}}]^\top$ as
 
\begin{equation*}
	\mathbf{u}_{s} := \mathbf{R}_{s} \mathbf{u}.
\end{equation*}
 
Next, we introduce the local diagonal matrix $\mathbf{E}_{s} \in \mathbb{R}^{N_{\mathrm{dof},s} \times N_{\mathrm{dof},s}}$, given by
 
\begin{equation*}
	\mathbf{E}_{s} := \mathrm{diag}(1/n_{i_1}, \dots, 1/n_{i_{N_{\mathrm{dof},s}}}).
\end{equation*}
 
This allows us to write
 
\begin{equation*}
	\mathbf{R}_{s} \mathbf{E} \mathbf{u} = \mathbf{E}_{s} \mathbf{u}_{s}.
\end{equation*}
 
Consequently, we obtain the representations
 
\begin{equation*}
	\begin{aligned}
		(u|_{\Omega_{s}}, u|_{\Omega_{s}})_{L^2_{\epsilon}\Lambda^1(\Omega_{s})}
		&= \mathbf{u}_{s}^{\top} \mathbf{G}_{s}^{\top} \mathbf{M}^{s}_\epsilon \mathbf{G}_{s} \mathbf{u}_{s},\\
		\widetilde{a}_{s}(u^{(s)}, u^{(s)})
		&= \mathbf{u}_{s}^{\top} \mathbf{E}_{s}^{\top} \mathbf{G}_{s}^{\top} \mathbf{M}^{s}_\epsilon \mathbf{G}_{s} \mathbf{E}_{s} \mathbf{u}_{s}.
	\end{aligned}
\end{equation*}
 
Since $\mathbf{G}_{s}$ is diagonal with entries equal to $\pm 1$, it follows that
 
\begin{equation*}
	\frac{\mathbf{v}_{s}^{\top} \mathbf{E}_{s}^{\top} \mathbf{G}_{s}^{\top} \mathbf{M}^{s}_\epsilon \mathbf{G}_{s} \mathbf{E}_{s} \mathbf{v}_{s}}
	{\mathbf{v}_{s}^{\top} \mathbf{G}_{s}^{\top} \mathbf{M}^{s}_\epsilon \mathbf{G}_{s} \mathbf{v}_{s}}
	\leq \frac{\lambda_{\max}(\mathbf{E}_{s})^2 \lambda_{\max}(\mathbf{G}_{s})^2 \lambda_{\max}(\mathbf{M}^{s}_\epsilon)}
	{\lambda_{\min}(\mathbf{G}_{s})^2 \lambda_{\min}(\mathbf{M}^{s}_\epsilon)}
	\leq \kappa_2(\mathbf{M}^{s}_\epsilon).
\end{equation*}
 
Using arguments analogous to Theorem~\ref{thm:spectral_equiv}, we have
 
\begin{equation*}
	\lambda_{\max}(\mathbf{M}^{s}_\epsilon) \leq \sup_{\widehat{\mathbf{x}}\in\widehat{\Omega}} \lambda_{\max}(\mathcal{C}^{(s)}(\widehat{\mathbf{x}})) \max_{k=1,2,3} \lambda_{\max}(\widehat{\mathbf{M}}_{k,k}), 
	\quad
	\lambda_{\min}(\mathbf{M}^{s}_\epsilon) \geq \inf_{\widehat{\mathbf{x}}\in\widehat{\Omega}} \lambda_{\min}(\mathcal{C}^{(s)}(\widehat{\mathbf{x}})) \min_{k=1,2,3} \lambda_{\min}(\widehat{\mathbf{M}}_{k,k}).
\end{equation*}
 
Applying~\cite[Lemma~1]{LOLI2022245}, we conclude that
 
\begin{equation*}
	C_0^2 := \max_{s=1,\dots,N_{\mathrm{patch}}}  \left(\sup_{\widehat{\mathbf{x}}\in\widehat{\Omega}} \kappa_2(\mathcal{C}^{(s)}(\widehat{\mathbf{x}}))\right) \max_{k=1,2,3} \kappa_2(\widehat{\mathbf{M}}_{k,k})
\end{equation*}
 
is a positive constant independent of the meshes $\mathbf{\Xi}^{(s)}$, but possibly dependent on $\mathbf{p}$, $\mathbf{F}$, and $\epsilon$, so that inequality~\eqref{eq:stab_dec} holds.
 
\item \textbf{Strengthened Cauchy-Schwarz inequalities.}
We have to prove that for any $s,r = 1,\ldots,N_{\mathrm{patch}}$ there exists $\varepsilon_{s,r} \in [0,1]$ such that
\begin{equation*}
	a(v^{(s)},v^{(r)}) \leq \varepsilon_{s,r} a(v^{(s)},v^{(s)})^{1/2} a(v^{(r)},v^{(r)})^{1/2},
	\qquad
	\forall\, v^{(s)} \in V_{s},\ v^{(r)} \in V_{r}.
\end{equation*}
We define
 
\begin{equation*}
	\varepsilon_{s,r} :=
	\begin{cases}
		1, & \text{if } r \in \mathcal{N}_{\mathrm{int}, s}, \\
		0, & \text{otherwise (non-interacting patches)}.
	\end{cases}
\end{equation*}
 
For interacting patches, the estimate follows from the Cauchy--Schwarz inequality. For non-interacting patches, the supports are disjoint and the bilinear form vanishes.\\
 
Let $\mathcal{E} = \{\varepsilon_{i,j}\}$ denote the resulting interaction matrix.  
Since each patch interacts with at most $N_{\mathrm{int}}$ patches, every row of $\mathcal{E}$ contains at most $N_{\mathrm{int}}$ nonzero entries, all equal to~1, and hence
 
\begin{equation*}
	\rho(\mathcal{E}) \leq N_{\mathrm{int}}.
\end{equation*}
 
\item \textbf{Local stability.}
We have to prove that there exists a constant $\omega>0$ such that, for every $s=1,\ldots,N_{\mathrm{patch}}$ and for every $u^{(s)} \in V_{s}$,
 
\begin{equation} \label{eq:local_stability}
	a(u^{(s)},u^{(s)}) \leq \omega\, \widetilde{a}_{s}(u^{(s)},u^{(s)}).
\end{equation}
 
For any $u^{(s)} \in V_{s}$, the support of $u^{(s)}$ is contained in the patch $\Omega_{s}$ and the neighboring patches indexed by $\mathcal{N}_{s}$. Therefore,
 
\begin{equation*}
	a(u^{(s)},u^{(s)}) = \|u^{(s)}\|^{2}_{L^{2}_{\epsilon}\Lambda^{1}(\Omega)}
	= \sum_{r \in \mathcal{N}_{s}} \left\| u^{(s)}\big|_{\Omega_{r}} \right\|^{2}_{L^{2}_{\epsilon}\Lambda^{1}(\Omega_{r})}.
\end{equation*}
 
Let $\mathbf{u}$ denote the coordinate vector of $u^{(s)}|_{\Omega_{s}}$ with respect to the single patch 1-form basis $\mathcal{B}^1$ defined on $\Omega_{s}$. Then
 
\begin{equation*}
\widetilde{a}_{s}(u^{(s)},u^{(s)}) = \left\| u^{(s)}\big|_{\Omega_{s}} \right\|^{2}_{L^{2}_{\epsilon}\Lambda^{1}(\Omega_{s})} = \mathbf{u}^{\top}\,\mathbf{M}^{s}_{\epsilon}\,\mathbf{u}.
\end{equation*}
 
For an adjacent patch $\Omega_{r}$, the $L^{2}_{\epsilon}\Lambda^{1}(\Omega_{r})$-norm of the restriction $u^{(s)}|_{\Omega_{r}}$ can be represented as
 
\begin{equation*}
	\left\| u^{(s)}\big|_{\Omega_{r}} \right\|^{2}_{L^{2}_{\epsilon}\Lambda^{1}(\Omega_{r})}
	= \mathbf{u}^{\top}\, \mathbf{J}_{r,s}^{\top}\, \mathbf{M}^{r}_{\epsilon}\, \mathbf{J}_{r,s}\, \mathbf{u},
\end{equation*}
 
where $\mathbf{J}_{r,s} \in \mathbb{R}^{N_{\mathrm{dof},r} \times N_{\mathrm{dof},s}}$ is a rectangular matrix whose only nonzero entries (equal to $\pm1$) appear in the rows and columns corresponding to the degrees of freedom on the interface between $\Omega_{s}$ and $\Omega_{r}$, with at most one such entry per row and column.
Consequently, $\mathbf{J}_{r,s}^{\top}\mathbf{J}_{r,s}$ is diagonal with diagonal entries equal to $0$ or $\pm 1$, and thus
 
\begin{equation*}
	\lambda_{\max}\left(\mathbf{J}_{r,s}^{\top}\mathbf{J}_{r,s}\right) \leq 1.
\end{equation*}
 
Applying~\cite[Lemma~1]{LOLI2022245} to both patches $\Omega_{s}$ and $\Omega_{r}$, we obtain
\begin{align*}
	\mathbf{u}^{\top}\mathbf{J}_{r}^{\top}\, \mathbf{M}^{r}_{\epsilon}\, \mathbf{J}_{r}\mathbf{u}
	&\leq \lambda_{\max} \left(\mathbf{J}_{r}^{\top}\mathbf{J}_{r}\right)\,
	\lambda_{\max}\left(\mathbf{M}^{r}_{\epsilon}\right)\,
	\|\mathbf{u}\|^{2}_{2} \\
	&\leq \frac{\lambda_{\max}\left(\mathbf{M}^{r}_{\epsilon}\right)}
	{\lambda_{\min}\left(\mathbf{M}^{s}_{\epsilon}\right)}
	\mathbf{u}^{\top}\mathbf{M}^{s}_{\epsilon}\mathbf{u} \leq C_{s,r}\, \mathbf{u}^{\top}\mathbf{M}^{s}_{\epsilon}\mathbf{u},
\end{align*}
for a constant $C_{s,r}>0$ that is independent of the mesh $\mathbf{\Xi}$, but may depend on $\mathbf{p}$, $\mathbf{F}$, and $\epsilon$.

Summing over all adjacent patches $r \in \mathcal{N}_{s}$, we obtain
 
\begin{equation*}
	a(u^{(s)},u^{(s)}) \leq \left( \sum_{r \in \mathcal{N}_{s}} C_{s,r} \right)
	\mathbf{u}^{\top}\mathbf{M}^{s}_{\epsilon}\mathbf{u}.
\end{equation*}
 
Finally, using the definition
 
\begin{equation}\label{eq:omega}
	\omega := N_{\mathrm{adj}} \max_{s,r=1,\ldots,N_{\mathrm{patch}}} C_{s,r},
\end{equation}
 
we conclude that inequality~\eqref{eq:local_stability} holds for all $u^{(s)} \in V_{s}$.
\end{itemize}
 
Finally,~\cite[Theorem 2.7]{toselli2006domain} implies
 
\begin{equation*}
	\kappa_2 \left( \mathbf{P}_{\mathrm{ad}}^{-\frac{1}{2}}{\mathbf M}_\epsilon \mathbf{P}_{\mathrm{ad}}^{-\frac{1}{2}} \right) \leq \omega C_0^2 (\rho(\mathcal{E})+1) \leq C N_{\mathrm{adj}} (N_{\mathrm{int}} +1),
\end{equation*}
 
where we have set $\displaystyle C=C_0^2 \max_{s,r=1,\ldots,N_{\mathrm{patch}}} C_{s,r}$.

<!-- remark-6: body_md -->
\label{rem:additive_schwarz_performance}
As in the single-patch case, the preceding result implies that the number of iterations required for the conjugate gradient method to converge, when preconditioned with $\mathbf{P}_{\mathrm{ad}}$, is independent of the mesh size. However, unlike the single-patch setting, the iteration count in the continuous Galerkin multi-patch formulation may depend on the spline degree. Furthermore, the proposed additive Schwarz preconditioner is \emph{scalable}: the condition number of the preconditioned operator does not depend on the total number of patches, but solely on the maximum number of adjacent patches.

<!-- remark-7: body_md -->
In a time-marching scheme, at each time step, we need to solve a linear system associated with the same matrices, $\mathbf{M}_{\epsilon}$ and $\widetilde{\mathbf{M}}_{\mu}$. Consequently, the same preconditioner can be used throughout the entire process. This implies that the setup procedures for the preconditioner only need to be performed once at the beginning of the time-marching scheme. As a result, the computational cost of these setup procedures becomes negligible compared to the application procedures, which must be executed at every time step and every PCG iteration.
