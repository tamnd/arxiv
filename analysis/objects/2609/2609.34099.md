\label{ass:convex}
\leavevmode
\begin{itemize}
\setlength{\itemsep}{2pt}
    \item The safeguard set is $\mathcal B=\{x:\left\|x\right\|\leq1\}$. The feasible set
    $\mathcal C$ is nonempty, closed, and contained in $\mathcal B$. A point $x_0\in\mathcal C$
    is supplied.
    \item The function $f$ is $\ell$-weakly convex and $G_f$-Lipschitz on
    an open convex neighborhood of $\mathcal B$, where $\ell,G_f>0$.
    With fresh randomness at each call, the oracle satisfies
    \begin{equation}
        \mathbb E[\widetilde v(x;\xi)\mid\mathcal F]
        =v(x)\in\partial f(x),
        \qquad \left\|\widetilde v(x;\xi)\right\|\leq G_f\quad\text{a.s.}
        \label{eq:oracle}
    \end{equation}
    Here $x$ is measurable with respect to the past history $\mathcal F$.
    \item The function $g$ is convex and continuously differentiable on
    that neighborhood. For constants $G_g,\sigma>0$,
    \[
        \left\|\nabla g(x)\right\|\leq G_g\quad(x\in\mathcal B),
        \qquad
        \left\|\nabla g(z)\right\|\geq\sigma\quad(z\in\mathcal B,\ g(z)=0).
    \]
    Both $g(x)$ and $\nabla g(x)$ can be evaluated exactly.
\end{itemize}

\label{thm:convex-smooth}
Suppose Assumptions~\ref{ass:convex} and \ref{ass:convex-smooth} hold. For any
$0<\epsilon\leq1$, run Algorithm~\ref{alg:one-projection} with
$p=p_\gamma$, $\beta=2\ell$, $m=\ell$, the penalty parameters in
\eqref{eq:convex-settings}, and $\mathcal I=\{1,\ldots,T\}$.
There are choices
\begin{equation}
    T=O\!\left(1+\frac{\ell\Delta_f}{\epsilon^2}\right),
    \qquad K=\widetilde O(\epsilon^{-2}),
    \label{eq:convex-smooth-rate}
\end{equation}
specified in Appendix~\ref{app:convex-schedule}, for which the output
satisfies
\[
    \hat x\in\mathcal C,
    \qquad
    \Pr\{\mathcal G_\ell(\hat x)\leq\epsilon\}\geq\frac23.
\]
For fixed problem parameters, the algorithm uses
$TK=\widetilde O(\epsilon^{-4})$ stochastic gradient calls and exactly
one projection onto $\mathcal C$.

all assumptions and oracle access of Theorem~\ref{thm:convex-smooth}

\label{thm:nonconvex}
Suppose Assumption~\ref{ass:nonconvex} holds and use
\eqref{eq:nc-parameters}--\eqref{eq:nc-settings}. Let
$\Delta_\lambda:=\Psi_\lambda(x_0)-\min_{x\in\mathcal B}\Psi_\lambda(x)$.
For $0<\epsilon<\min\{h_{\rm feas},\beta R,1\}$, there are choices
\begin{equation}
    T=O\!\left(1+\frac{\beta\Delta_\lambda}{\epsilon^2}\right),
    \qquad K=\widetilde O(\epsilon^{-2}),
    \label{eq:nc-rate}
\end{equation}
specified in Appendix~\ref{app:nc-schedule}, with the following guarantee.
The output is exactly feasible. With probability at least $2/3$, the final
projection is unique, $\mathcal R_{\lambda,\beta}(x_J)\leq\epsilon$, and
there exist $y\in\mathcal C$, $v\in\partial f(y)$, and $\mu\in[0,\lambda]$ such that
\begin{equation}
    \left\|\hat x-y\right\|\leq\frac{2\epsilon}{\beta},\qquad
    \left\|v+\mu\nabla g(y)\right\|\leq\epsilon,\qquad \mu g(y)=0.
    \label{eq:nc-guarantee}
\end{equation}
For fixed problem parameters, the algorithm uses
$TK=\widetilde O(\epsilon^{-4})$ stochastic subgradients and exactly one
projection onto $\mathcal C$.
<!-- end theorem-3 -->
<!-- remark-3: body_md -->
Here we also achieve the standard
$\epsilon^{-4}$ rate.
The stochastic-oracle exponent is unchanged, although its constants now
depend on the infeasible-slope margin and the penalty's weak convexity.
Unlike the convex-constraint results above, this theorem does not require a feasible initial
point. It uses no projections onto local feasible-set models and does not
keep the intermediate iterates feasible. It does, however, assume an oracle
that returns an \emph{exact nearest point} in the last line. Compactness
ensures such a point exists on every outcome; uniqueness is guaranteed on
the stated success event. Local uniqueness is not a computational algorithm
for finding that point. Thus our bound counts one call to a potentially
expensive projection oracle and does not claim polynomial-time nonconvex projection.
For smooth $f$, replace $v$ in \eqref{eq:nc-guarantee} by $\nabla f(y)$ and we can keep the same rate.

<!-- lemma-1: body_md -->
\label{lem:inner}
Let $H$ be $m$-strongly convex on a neighborhood of $\mathcal B$, where $m>0$.
Suppose $\widetilde h_s$ is a stochastic subgradient at $z_s\in\mathcal B$ with
$\mathbb E[\widetilde h_s\mid\mathcal F_s]\in\partial H(z_s)$ and
$\left\|\widetilde h_s\right\|\leq G$ almost surely, for some $G>0$.
Starting at any $z_1\in\mathcal B$, set
\[
    z_{s+1}=\mathcal P_{\mathcal B}(z_s-\eta_s\widetilde h_s),
    \qquad \eta_s=\frac{2}{m(s+1)},\qquad
    \bar z=\frac1K\sum_{s=1}^Kz_s.
\]
For $0<\delta<1$, with probability at least $1-\delta$,
\begin{equation}
    H(\bar z)-\min_{z\in\mathcal B}H(z)
    \leq U_K(m,G,\delta)
    :=\frac{m+(G^2/m)\{\log(K+1)+2\log(1/\delta)\}}{K}.
    \label{eq:inner-bound}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $u$ minimize $H$ over $\mathcal B$, put $D_s=\left\|z_s-u\right\|$, and write
$h_s=\mathbb E[\widetilde h_s\mid\mathcal F_s]$ and
$X_s=\left\langleh_s-\widetilde h_s,z_s-u\right\rangle$.
Nonexpansiveness of the ball projection and strong convexity imply
\begin{align*}
    H(z_s)-H(u)
    &\leq \left\langleh_s,z_s-u\right\rangle-\frac m2D_s^2\\
    &\leq\frac{D_s^2-D_{s+1}^2}{2\eta_s}
            -\frac m2D_s^2+\frac{\eta_sG^2}{2}+X_s.
\end{align*}
Summing with the stated stepsize and dropping the negative terminal term gives
\begin{equation}
    \sum_{s=1}^K(H(z_s)-H(u))
    \leq\frac m4D_1^2-\frac m4\sum_{s=1}^KD_s^2
       +\frac{G^2}{m}\sum_{s=1}^K\frac1{s+1}+\sum_{s=1}^KX_s.
    \label{eq:inner-telescope}
\end{equation}
Conditionally on $\mathcal F_s$, $X_s$ is centered and its range has length
at most $2GD_s$. The conditional bounded-variable exponential inequality gives
\[
    \mathbb E[\exp(\theta X_s)\mid\mathcal F_s]
       \leq\exp(\theta^2G^2D_s^2/2).
\]
Iterating conditional expectations and applying Markov's inequality shows
that, with probability at least $1-\delta$,
\[
    \sum_{s=1}^KX_s
       \leq \frac{\theta G^2}{2}\sum_{s=1}^KD_s^2
                   +\frac{\log(1/\delta)}{\theta}.
\]
Set $\theta=m/(2G^2)$. The squared-distance sum cancels the negative sum
in \eqref{eq:inner-telescope}. Since $D_1\leq2$ and
$\sum_{s=1}^K(s+1)^{-1}\leq\log(K+1)$, division by $K$ and convexity of
$H$ prove \eqref{eq:inner-bound}.

<!-- lemma-2: body_md -->
\label{lem:convex-eb}
For every $x\in\mathcal B$,
\begin{equation}
    \operatorname{dist}(x,\mathcal C)\leq\frac{[g(x)]_+}{\sigma}.
    \label{eq:convex-error-bound}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The claim is immediate for $x\in\mathcal C$. Otherwise let $p=\mathcal P_{\mathcal C}(x)$.
Compactness of $\mathcal C$ gives existence, and convexity gives uniqueness.
The point $p$ lies on $g=0$. Because $\nabla g(p)\ne0$, the normal cone
is $N_{\mathcal C}(p)=\{\alpha\nabla g(p):\alpha\geq0\}$.
The projection condition gives $x-p=\alpha\nabla g(p)$ for some
$\alpha>0$. Convexity then yields
\[
    g(x)\geq g(p)+\left\langle\nabla g(p),x-p\right\rangle
       =\left\|\nabla g(p)\right\|\left\|x-p\right\|
       \geq\sigma\operatorname{dist}(x,\mathcal C).
\]
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:convex-inner}
Fix $x_t\in\mathcal B$, let $y_t=Q_\ell(x_t)$, and run the inner routine with the
convex settings. With conditional probability at least $1-\delta$,
\begin{align}
    F_t(x_{t+1})+\lambda_0[g(x_{t+1})]_+
       &\leq F_t(y_t)+e_K(\delta),\label{eq:cv-inner-gap}\\
    \operatorname{dist}(x_{t+1},\mathcal C)&\leq e_K(\delta)/G_F,\label{eq:cv-inner-dist}\\
    \left\|x_{t+1}-y_t\right\|^2&\leq2e_K(\delta)/\ell.\label{eq:cv-prox-error}
\end{align}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The function $F_t$ is $\ell$-strongly convex. Since $p_\gamma$ is convex
and nondecreasing and $g$ is convex,
$H_t:=F_t+p_\gamma\circ g$ is also $\ell$-strongly convex.
The stochastic subgradient used in Algorithm~\ref{alg:one-projection}
has norm at most
\[
    G_f+2\ell\left\|z_s-x_t\right\|+\lambda_0G_g
       \leq G_f+4\ell+\lambda_0G_g=G_H.
\]
Apply Lemma~\ref{lem:inner} and compare the minimum of $H_t$ to $H_t(y_t)$.
The elementary bounds
\[
    \lambda_0[u]_+\leq p_\gamma(u)
       \leq\lambda_0[u]_++\gamma\log2
\]
and $\gamma=1/K$ prove \eqref{eq:cv-inner-gap}.

For completeness, the exact-penalty objective
$P_t(z):=F_t(z)+\lambda_0[g(z)]_+$ has the same unique minimizer $y_t$
over $\mathcal B$ as $F_t$ does over $\mathcal C$. Indeed, for any $z\in\mathcal B$, set
$p=\mathcal P_{\mathcal C}(z)$ and $d=\left\|z-p\right\|$. The function $F_t$ is $G_F$-Lipschitz
on $\mathcal B$, so Lemma~\ref{lem:convex-eb} gives
\begin{align}
    P_t(z)&\geq F_t(p)-G_Fd+\lambda_0\sigma d\nonumber\\
          &\geq F_t(y_t)+G_Fd.\label{eq:cv-exact-penalty}
\end{align}
This proves exactness and, together with \eqref{eq:cv-inner-gap},
\eqref{eq:cv-inner-dist}. Finally, $P_t$ is $\ell$-strongly convex.
At its minimizer over the convex set $\mathcal B$, quadratic growth gives
\[
    \frac\ell2\left\|x_{t+1}-y_t\right\|^2
       \leq P_t(x_{t+1})-P_t(y_t)\leq e_K(\delta),
\]
proving \eqref{eq:cv-prox-error}.

<!-- lemma-4: body_md -->
\label{lem:cv-descent}
Suppose the conclusions of Lemma~\ref{lem:convex-inner} hold in every
outer iteration with a common error bound $e$. Then
\begin{equation}
    \frac1T\sum_{t=0}^{T-1}s_t^2
    \leq\frac{\Delta_f}{\ell T}
       +\frac{1+G_f/G_F}{\ell}e+\frac{e^2}{G_F^2}.
    \label{eq:cv-average-steps}
\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Dropping the nonnegative penalty in \eqref{eq:cv-inner-gap}, using
$F_t(y_t)\leq F_t(p_t)$, and using Lipschitz continuity of $f$ give
\begin{align*}
    f(x_{t+1})+\ell s_t^2
    &\leq F_t(y_t)+e\\
    &\leq f(p_t)+\ell d_t^2+e\\
    &\leq f(x_t)+G_f d_t+\ell d_t^2+e.
\end{align*}
This proves \eqref{eq:overview-descent}, including its quadratic distance
term. Now sum over $t$, use $d_0=0$ and $d_t\leq e/G_F$ for $t\geq1$,
and use $f(x_T)\geq\min_{\mathcal B}f$.

<!-- lemma-5: body_md -->
\label{lem:cv-prox-lipschitz}
The point $Q_\ell(x)$ is unique and $Q_\ell$ is $2$-Lipschitz. Moreover,
$\Phi_\ell$ is differentiable with
$\nabla\Phi_\ell(x)=2\ell(x-Q_\ell(x))$, and
\eqref{eq:moreau-intuition} holds.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
The function $f+\delta_{\mathcal C}$ is proper, lower semicontinuous and
$\ell$-weakly convex. Compactness gives existence of the proximal minimizer;
strong convexity gives uniqueness. For $q_i=Q_\ell(x_i)$, the optimality
conditions and monotonicity of the subdifferential of
$f+\delta_{\mathcal C}+\ell\left\|\cdot\right\|^2/2$ give
\[
    \left\langle2\ell(x_1-q_1)-2\ell(x_2-q_2),q_1-q_2\right\rangle
        \geq-\ell\left\|q_1-q_2\right\|^2.
\]
Consequently,
$\left\|q_1-q_2\right\|^2\leq2\left\langlex_1-x_2,q_1-q_2\right\rangle$, which proves the
Lipschitz bound. Differentiating the minimum with its unique continuous
minimizer yields the stated envelope gradient.
The proximal optimality condition is
\[
    2\ell(x-Q_\ell(x))\in
         \partial f(Q_\ell(x))+N_{\mathcal C}(Q_\ell(x)).
\]
The sum rule is valid since $f$ is locally Lipschitz and weakly convex on
a neighborhood of $\mathcal C$; equivalently it is the convex sum rule after
adding $\ell\left\|\cdot\right\|^2/2$. This proves
\eqref{eq:moreau-intuition}.

<!-- lemma-7: body_md -->
\label{lem:geometry}
Let $R=\sigma_{\rm bd}/L_g$. For every $p,y\in\mathcal C$ and
$n\in N_{\mathcal C}(p)$,
\begin{equation}
    \left\langlen,y-p\right\rangle\leq\frac{\left\|n\right\|}{2R}\left\|y-p\right\|^2.
    \label{eq:normal-inequality}
\end{equation}
Consequently, $\mathcal P_{\mathcal C}(x)$ is unique whenever $\operatorname{dist}(x,\mathcal C)<R$.
In particular, $\mathcal C$ is $R$-proximally smooth.
<!-- end lemma-7 -->
<!-- proof-9: body_md -->
At an interior point of $\mathcal C$ the normal cone is $\{0\}$.
At a boundary point $p$, the nonzero gradient condition yields the smooth
inequality normal formula
\[
    N_{\mathcal C}(p)=\{\alpha\nabla g(p):\alpha\geq0\}.
\]
Indeed, the local tangent cone is the halfspace
$\{u:\left\langle\nabla g(p),u\right\rangle\leq0\}$, whose polar is this normal ray.
Since $g(p)=0$ and $g(y)\leq0$, the smooth lower bound along the segment
in $\mathcal B$ gives
\[
    g(y)\geq\left\langle\nabla g(p),y-p\right\rangle
                   -\frac{L_g}{2}\left\|y-p\right\|^2,
    \qquad
    \left\langle\nabla g(p),y-p\right\rangle\leq\frac{L_g}{2}\left\|y-p\right\|^2.
\]
Multiply by $\alpha$ and use
$\alpha\leq\left\|n\right\|/\sigma_{\rm bd}$ to obtain
\eqref{eq:normal-inequality}.

To see uniqueness directly, suppose $p_1,p_2$ are projections of $x$,
with common distance $d<R$. Then $n_i=x-p_i\in N_{\mathcal C}(p_i)$.
Applying \eqref{eq:normal-inequality} twice and adding yields
\[
    \left\|p_1-p_2\right\|^2
       \leq\frac dR\left\|p_1-p_2\right\|^2.
\]
Thus $p_1=p_2$. Projections exist by compactness, so this proves the
unique-projection tube characterization of proximal smoothness.

<!-- lemma-8: body_md -->
\label{lem:nc-transfer}
For $x\in\mathcal B$, write $y=Q_{\lambda,\beta}(x)$ and
$r=\mathcal R_{\lambda,\beta}(x)$. If $r<h_{\rm feas}$, then $y\in\mathcal C$ and
there are $v\in\partial f(y)$ and $\mu\in[0,\lambda]$ with
\begin{equation}
    \beta(x-y)=v+\mu\nabla g(y),\qquad \mu g(y)=0.
    \label{eq:nc-witness}
\end{equation}
Every $p\in\mathcal P_{\mathcal C}(x)$ then satisfies
\begin{equation}
    \operatorname{dist}(x,\mathcal C)\leq\frac r\beta,\qquad
    \left\|p-y\right\|\leq\frac{2r}{\beta},\qquad
    \operatorname{dist}(0,\partial f(y)+N_{\mathcal C}(y))\leq r.
    \label{eq:nc-transfer}
\end{equation}
If additionally $r<\beta R$, the projection is unique.

<!-- proof-10: body_md -->
Convex subdifferential calculus after a quadratic shift, or the equivalent
weakly convex calculus, gives
\begin{equation}
    \beta(x-y)=v+\lambda\theta\nabla g(y)+n,
    \qquad v\in\partial f(y),\quad n\in N_{\mathcal B}(y),
    \label{eq:nc-prox-opt}
\end{equation}
where $\theta=1$ for $g(y)>0$, $\theta=0$ for $g(y)<0$, and
$\theta\in[0,1]$ for $g(y)=0$.
If $g(y)>0$, then $N_{\mathcal B}(y)$ is a cone and
\[
    r=\left\|v+\lambda\nabla g(y)+n\right\|
       \geq\lambda\operatorname{dist}(0,\nabla g(y)+N_{\mathcal B}(y))-G_f
       \geq h_{\rm feas},
\]
a contradiction. Thus $y\in\mathcal C\subset\operatorname{int}\mathcal B$ and $n=0$.
Set $\mu=\lambda\theta$. This proves \eqref{eq:nc-witness}, including
nonnegativity and exact complementarity. The boundary normal formula
from Lemma~\ref{lem:geometry} shows that
$\mu\nabla g(y)\in N_{\mathcal C}(y)$, proving the stationarity claim.

Finally $\operatorname{dist}(x,\mathcal C)\leq\left\|x-y\right\|=r/\beta$. For any nearest point $p$,
the triangle inequality gives
$\left\|p-y\right\|\leq\left\|p-x\right\|+\left\|x-y\right\|\leq2r/\beta$.
Lemma~\ref{lem:geometry} gives uniqueness when $r<\beta R$.

<!-- lemma-9: body_md -->
\label{lem:nc-descent}
Suppose every inner run returns $x_{t+1}\in\mathcal B$ such that
\begin{equation}
    H_t(x_{t+1})-H_t(y_t)\leq e,\qquad
    H_t(z):=\Psi_\lambda(z)+\frac\beta2\left\|z-x_t\right\|^2,\qquad
    y_t=Q_{\lambda,\beta}(x_t).
    \label{eq:nc-inner-gap}
\end{equation}
Then we have  
\begin{equation}
    \frac1T\sum_{t=0}^{T-1}\mathcal R_{\lambda,\beta}(x_t)^2
    \leq \frac{4\beta\Delta_\lambda}{T}
         +4\beta\left(1+\frac\beta m\right)e. 
\end{equation}
<!-- end lemma-9 -->
<!-- proof-11: body_md -->
Strong convexity and \eqref{eq:nc-inner-gap} imply
$\left\|x_{t+1}-y_t\right\|^2\leq2e/m$. Since
$H_t(y_t)\leq H_t(x_t)=\Psi_\lambda(x_t)$, we also have
\[
    \Psi_\lambda(x_{t+1})+\frac\beta2\left\|x_{t+1}-x_t\right\|^2
       \leq\Psi_\lambda(x_t)+e.
\]
Sum this inequality and use the definition of $\Delta_\lambda$ to obtain
\[
    \frac1T\sum_{t=0}^{T-1}\left\|x_{t+1}-x_t\right\|^2
        \leq\frac{2\Delta_\lambda}{\beta T}+\frac{2e}{\beta}.
\]
Finally,
\[
    \mathcal R_{\lambda,\beta}(x_t)^2
       \leq2\beta^2\left\|x_t-x_{t+1}\right\|^2
              +2\beta^2\left\|x_{t+1}-y_t\right\|^2.
\]
Averaging concludes the proof.
