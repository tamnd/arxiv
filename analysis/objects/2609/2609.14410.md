(\cite[Section 2.15.1]{Ammari2018PhotonicsPhononics})\label{thm:greenexpansionsingle}
The Green's function $\bm{G}^{\omega}=(G_{ij}^{\omega})_{i,j=1}^3$ admits the following expansion with respect to the frequency $\omega$:

\end{align*}

<!-- theorem-2: body_md -->
\label{thm:greenexpansion}
For non-zero quasi-momentum $\bm\alpha$, the quasi-periodic Green's function ${\bm G}^{\bm\alpha,\omega}$ admits an expansion in even powers of $\omega$ of the form
$$
    \bm{G}^{\bm{\alpha},\omega} = \bm{G}^{\bm{\alpha},0} + \sum\limits_{j=1}^{+\infty} \omega^{2j}\bm{G}_{j}^{\bm{\alpha},0},
$$
where the coefficients $\bm G_{j}^{\bm\alpha,0}$, $j=0,1,2,\ldots$, are independent of $\omega$.
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
It follows from the representation formulas (SM2.1)--(SM2.2), (SM2.4)--(SM2.6), (SM2.8)--(SM2.10) in Section~SM2 of the supplementary material that the dependence of ${\bm G}^{\bm\alpha,\omega}$ on $\omega$ occurs only through $\omega^2$, i.e., $\bm{G}^{\bm{\alpha},\omega} = \bm{G}^{\bm{\alpha},-\omega}$. Thus ${\bm G}^{\bm\alpha,\omega}$ is an even function of $\omega$. Consequently, its expansion contains only even powers of $\omega$, and all odd order terms vanish.

\label{thm:resonantfreq}
Let $\bm{\alpha}\in Y^*$ and assume that $|\bm{\alpha}|^2\geq\rho\omega^2/(\lambda+2\mu)$. For $i,i'=1,\ldots,N$ and $j,j'=1,\ldots,6$, define
\[
\displaystyle\widehat{\bm{C}}_{i,i'}^{(j,j')}:= -\int_{\partial D} \bm{f}_{i,\mathrm{per}}^{(j)}\cdot\bm{\psi}_{i'}^{(j')}\mathrm{d}\sigma, \quad \displaystyle\bm{M}_{i,i'}^{(j,j')}:=\int_{D}\bm{\psi}_i^{(j)}\cdot\bm{\psi}_{i'}^{(j')}\mathrm{d}\bm{x}.
\]
Let $\mathfrak{M}\in\mathbb{R}^{6N\times 6N}$ and $\widehat{\mathfrak{C}}\in\mathbb{C}^{6N\times 6N}$ be the matrices whose entries are given by
\[
 \mathfrak{M}_{6(i-1)+j, 6(i'-1)+j'} = \bm{M}_{i,i'}^{(j,j')},\quad \widehat{\mathfrak{C}}_{6(i-1)+j, 6(i'-1)+j'} = \widehat{\bm{C}}_{i,i'}^{(j,j')}.
\]
Then both $\mathfrak M$ and $\widehat{\mathfrak C}$ are positive definite.

Define the diagonal matrices $\uptau:=\mathrm{diag}(\tau_1^2,\cdots, \tau_N^2)\in\mathbb{R}^{6N\times 6N}$ and  $\updelta:=\mathrm{diag}(\delta_1, \allowbreak \cdots, \delta_N)\in\mathbb{R}^{6N\times 6N}$, where each $\tau_i^2$ and $\delta_i$ is repeated six consecutive times. Let $\lambda_k^{\bm\alpha}$, $k=1,\ldots,6N$, be the eigenvalues of $\widehat{\mathfrak{C}}\updelta(\mathfrak{M}\uptau)^{-1}$. Then the subwavelength resonant frequencies $\omega_k^{\bm{\alpha}} = \omega_k^{\bm{\alpha}}(\bm{\delta})$ of $\mathcal{A}_{\bm{\delta}}^{\bm{\alpha},\omega}$ satisfy
$$
\omega_k^{\bm{\alpha}} := (\lambda_k^{\bm{\alpha}}/\rho)^{1/2} + \mathcal{O}(\omega^2) + \mathcal{O}(\bm{\delta}) = \mathcal{O}({\bm\delta}^{1/2}),\quad k=1, \ldots, 6N.
$$

<!-- lemma-5: body_md -->
For $\bm\eta\in H^{1/2}(\partial D)^3$ and $\bm q\in\mathbb C^3$, the system
\begin{equation}\label{eq:constrained}
 \mathcal S_D^{0,0}[\bm\phi]+\bm a=\bm\eta,
 \qquad \int_{\partial D}\bm\phi\mathrm{d}\sigma=\bm q
\end{equation}
has a unique solution in $H^{-1/2}(\partial D)^3\times\mathbb C^3$, depending boundedly on the data. Here $\bm a$ is a constant boundary field.
<!-- end lemma-5 -->
<!-- proof-2: body_md -->
Define $\mathcal T:(H^{-1/2}(\partial D))^3\times\mathbb{C}^3\rightarrow (H^{1/2}(\partial D))^3\times\mathbb{C}^3$ by $\mathcal T(\phi,\bm a):=(\mathcal S_D^{0,0}[\phi]+\bm a, \int_{\partial D}\phi\mathrm{d}\sigma)$.
The difference between $\bm{G}^0$ and $\bm{G}^{0,0}$ is smooth on $\partial D\times\partial D$, $\mathcal S_D^{0,0} - \mathcal S_D^0$ is compact from $H^{-1/2}(\partial D)^3$ to $H^{1/2}(\partial D)^3$. The remaining difference between $\mathcal T$ and $\mathcal T_0(\phi,\bm a):=(\mathcal S_D^0[\phi], \bm a)$ is finite rank. Hence $\mathcal T-\mathcal T_0$ is compact. Since $\mathcal T_0$ is invertible, $\mathcal T$ is Fredholm of index zero.

Suppose $(\bm\phi_0,\bm a_0)$ solves \eqref{eq:constrained} with $\bm\eta=0, \bm q=0$. Set $\bm v=\mathcal S_D^{0,0}[\bm\phi]$. When $|\bm x|\rightarrow\infty$, $\bm v(\bm x)=O(r^{-1})$ and $\nabla\bm v=O(|\bm x_0|^{-2})$ for $d_l=1$. For $d_l=2$, since $\bm q=0$, 
\begin{align*}
\bm v(\bm x)= \mp\frac{1}{2}\operatorname{diag}(\mu^{-1},\mu^{-1},(\lambda+2\mu)^{-1})\int_{\partial D}y_3\bm\phi(\bm y)\mathrm{d}\sigma(\bm y)+\mathcal{O}(e^{-\kappa|x_3|}).
\end{align*}
Green's identity then gives, for both $d_l=1,2$ cases,
$$
 \int_Y\left(\lambda|\nabla\cdot\bm v|^2+
       \frac\mu2|\nabla\bm v+\nabla\bm v^\top|^2\right)d\bm x
 =-\int_{\partial D}\bm\phi\cdot\overline{\bm v}\,d\sigma=0.
$$
Hence $\bm v$ is a bounded global rigid field, thus a constant. The limit of $\bm v$ when $x_3\rightarrow\infty$ forces $\bm v=0$. The jump of conormal derivative across $\partial D$ then yields $\bm\phi=0$, and then $\bm a=0$. This finishes the proof of $\operatorname{dim}\operatorname{Ker}\mathcal T=0$. The Fredholm alternative completes the proof.

 \mathcal S_D^{0,0}[\bm\phi]+\bm a+\mathcal R_\omega[\bm\phi]=\bm\eta,
 \qquad \int_{\partial D}\bm\phi\mathrm{d}\sigma=(K_0\ln\omega+K_1^{\alpha_0})^{-1}\bm a,
\end{equation}
where $\bm a:=(K_0\ln\omega+K_1^{\alpha_0})\int_{\partial D}\bm\phi\mathrm{d}\sigma$. Since $\|\mathcal R_\omega\|=\mathcal O(|\omega\ln\omega|)$, this is a small perturbation of \eqref{eq:constrained} with $\bm q=0$.

Let $\alpha = \omega\alpha_0$, where $\alpha_0$ is independent of $\omega$ and satisfies $|\alpha_0|^2<\rho/(\lambda+2\mu)$.
Let $\mathfrak{M}\in\mathbb{R}^{6N\times 6N}$ be defined as in Theorem~\ref{thm:resonantfreq}, $\widehat{\mathfrak{C}}_0\in\mathbb{C}^{6N\times 6N}$ be defined as in \eqref{eq:c0-c1}. $$
    0=\lambda^\alpha_{1,0}=\lambda^\alpha_{2,0}=\lambda^\alpha_{3,0}<\lambda^\alpha_{4,0}\leq\cdots\leq\lambda^\alpha_{6N,0}.
$$
The subwavelength resonant frequencies $\omega_{k,0}^{\alpha} = \omega_{k,0}^{\alpha}(\bm{\delta})$ associated with $\mathcal{A}_{\bm{\delta}}^{\omega\alpha_0,\omega}$ satisfy
\begin{equation*}
\omega^{\alpha}_{k,0} := (\lambda^{\alpha}_{k,0}/\rho)^{1/2} +
\left\{
\begin{aligned}
    &\mathcal{O}(\sqrt{\delta_M}/|\ln\delta_M|),&& d_l=1,\\
    &\mathcal{O}(\delta_M), && d_l=2,
\end{aligned}
\right.
\quad k=4, \ldots, 6N.
\end{equation*}

Let $m_\delta:=\sum_i\tau_i^2|D_i|/\delta_i$. For $d_l=1$, let $\nu_j(\bm\delta)$ be the eigenvalues of $\bm K_0^{-1}/(\rho m_\delta)$. Three further nonzero resonances satisfy

 \omega_j=\sqrt{\frac{2\nu_j(\bm\delta)}{|\ln\delta_M|}}
 \left[1+\mathcal O\left(\frac{\ln|\ln\delta_M|}{|\ln\delta_M|}\right)\right],

For $d_l=2$, let $\zeta_j(\bm\delta)$ be the eigenvalues of $-(\bm B^{\alpha_0})^{-1}/(\rho m_\delta)$. Three further nonzero resonances satisfy

\end{equation*}

<!-- proof-5: body_md -->
When $\omega$ is a resonant frequency, the following system admits a nontrivial normalized solution $\bm{\psi}, \bm{\phi}\in (H^{-1/2}(\partial D))^3$:
    \begin{equation}\label{eq:bandsys1}
    \begin{cases}
    \mathcal S_{D_i}^{\tau_i\omega}[\bm\psi_i]
       -\mathcal S_D^{\omega\bm\alpha_0,\omega}[\bm\phi]=0,\\[1mm]
    \left(-\frac12\mathcal I+(\mathcal K_{D_i}^{\tau_i\omega})^*\right)[\bm\psi_i]
       -\delta_i\left(\tfrac12\mathcal I+
          (\mathcal K_D^{\omega\bm\alpha_0,\omega})^*\right)[\bm\phi]=0
                    \quad\text{on }\partial D_i.
    \end{cases}
    \end{equation}
    
    The factor $\delta_i$ multiplies the exterior traction on the observation inclusion. Substitution of the correct operator expansions gives, for $d_l=1$,
    \begin{equation*}                    
    \begin{aligned}
    &\left[-\dfrac12\mathcal I+(\mathcal K_{D_i}^0)^*+\tau_i^2\omega^2(\mathcal K_{D_i,2}^0)^*+\tau_i^3\omega^3(\mathcal K_{D_i,3}^0)^*+\mathcal O(\omega^4)\right][\bm\psi_i]\\
    &\quad=\delta_i\bigl[\dfrac12\mathcal I+(\mathcal K_D^{0,0})^*+\omega\log\omega(\mathcal K_{D,1,\log}^{\alpha_0})^*+\omega(\mathcal K_{D,1}^{\alpha_0})^*\\
    &\qquad\quad+\omega^2\log\omega(\mathcal K_{D,2,\log}^{\alpha_0})^*+\omega^2(\mathcal K_{D,2}^{\alpha_0})^*+\mathcal O\bigl(\omega^3(1+|\log\omega|)\bigr)\bigr][\bm\phi].
    \end{aligned}
    \end{equation*}
    For $d_l=2$, the right-hand bracket is instead
    \begin{equation*}                          
    \dfrac12\mathcal I+(\mathcal K_D^{0,0})^*+(\mathcal K_{D,H}^{\bm\alpha_0})^*+\omega(\mathcal K_{D,1}^{\bm\alpha_0})^*+\omega^2(\mathcal K_{D,2}^{\bm\alpha_0})^*+\mathcal O(\omega^3).
    \end{equation*}

    Letting $\omega\to 0$ and $\bm{\delta}\to 0$ in the second equation of \eqref{eq:bandsys1}, we conclude that the leading-order term of $\bm{\psi}_i$ lies in the span of ${\bm{f}_i^{(j)}}$, which forms a basis of $\operatorname{Ker}(-\frac{1}{2}\mathcal{I}+(\mathcal{K}_D^0)^*)$. Consequently, we may write $\bm{\psi}_i = \sum_{i,j} \tilde{c}_i^{(j)}\bm{f}_i^{(j)} + \mathcal{O}(\omega^2+\bm{\delta})$ on $\partial D_i$ with constants $\tilde{c}_i^{(j)}$.

    Taking the inner product on $\partial D$ of both sides of the second equation in \eqref{eq:bandsys1} with $\bm{\psi}_{i'}^{j'}$. On one hand, we have
    \begin{equation*}                    
    \begin{aligned}
        &\int_{\partial D_i}\left(-\frac{1}{2}\mathcal I+(\mathcal K_{D_i}^{\tau_i\omega})^*\right)[\bm\psi_i]\cdot\bm\psi_i^{(j')}\mathrm{d}\sigma \\
        =&\tau_i^2\omega^2\int_{\partial D_i}(\mathcal K_{D_i,2}^0)^*[\bm\psi_i]\cdot\bm\psi_i^{(j')}\mathrm{d}\sigma + \tau_i^3\omega^3\int_{\partial D_i}(\mathcal K_{D_i,3}^0)^*[\bm\psi_i]\cdot\bm\psi_i^{(j')}\mathrm{d}\sigma + \mathcal{O}(\omega^4) \\
        =&-\rho\tau_i^2\omega^2\int_{D_i}(\mathcal S_{D_i}^0+\tau_i\omega\mathcal S_{D_i,1}^0)[\bm\psi_i]\cdot\bm\psi_i^{(j')}\mathrm{d}\bm x+\mathcal O(\omega^4),
    \end{aligned}
    \end{equation*}
    where the last equality uses Lemma SM2.2 in the supplementary material. Combining with the expression of $\bm\psi_{i'}^{j'}$ and using \eqref{eq:potentialsingle} we have
    \begin{equation*}
    \begin{aligned}
        (\mathcal S_{D_i}^0+\tau_i\omega\mathcal S_{D_i,1}^0)[\bm\psi_i] =&\sum\limits_{j=1}^6 \widetilde{c}_i^{(j)}\bm\psi_i^{(j)} + \tau_i\omega\sum\limits_{j=1}^6\widetilde{c}_i^{(j)}\mathcal S_{D_i,1}^0(\mathcal S_{D_i}^0)^{-1}[\bm\psi_i^{(j)}] + \mathcal O(\omega^2+\bm\delta)\|\widetilde{\bm{c}}\|, \\
        =:&\sum\limits_{j=1}^6 c_i^{(j)}\bm\psi_i^{(j)} + \mathcal O(\omega^2+\bm\delta)\|\widetilde{\bm c}\|,
    \end{aligned}    
    \end{equation*}
    where for $\ell=1,2,3$,
    $$
        c_i^{(\ell)}:=\widetilde c_i^{(\ell)}-\frac{\mathrm{i}\tau_i\omega}{12\pi\rho}(\frac{2}{c_s^3}+\frac{1}{c_p^3})\sum_{j=1}^6\widetilde c_i^{(j)}\left[\int_{\partial D_i}(\mathcal S_{D_i}^0)^{-1}[\bm\psi_i^{(j)}]d\sigma\right]_{\ell},
    $$
    and $c_i^{(l)} := \widetilde c_i^{(l)}$ for $l=4,5,6$. Hence
    \begin{equation}\label{eq:omega0-2}
    \begin{aligned}
        \frac1{\delta_i}\int_{\partial D_i}\left(-\dfrac12\mathcal I+(\mathcal K_{D_i}^{\tau_i\omega})^*\right)[\bm\psi_i]\cdot\bm\psi_i^{(j')}\mathrm d\sigma
        =-\frac{\rho\tau_i^2\omega^2}{\delta_i}\sum_j\bm M_{i,i}^{(j,j')}c_i^{(j)}+\mathcal{O}\left(|\omega|^2+\frac{|\omega|^4}{\bm\delta}\right)\|\bm c\|.
    \end{aligned}
    \end{equation}

    On the other hand, from \eqref{eq:expansionsingle} and \eqref{eq:potentialsingle}, we obtain the expansion
    \begin{equation*}
        \mathcal{S}_D^{\bm{\tau}\omega}[\bm{f}_i^{(j)}] = \bm{\psi}_i^{(j)} - \sum\limits_{k=1}^N \tau_k\omega\frac{\mathrm{i}}{12\pi\rho}\Big(\frac{2}{c_s^3}+\frac{1}{c_p^3}\Big)\int_{\partial D_k}\bm{f}_i^{(j)}(\bm{y})\mathrm{d}\sigma(\bm{y})\chi_{\partial D_k}+\mathcal{O}(\omega^2),
    \end{equation*}
    and hence
    \begin{equation*}
    \begin{aligned}
        &\mathcal{S}_D^{\omega\alpha_0,\omega}[\bm{\phi}] = \mathcal{S}_D^{\bm{\tau}\omega}[\bm{\psi}] = \sum\limits_{i,j} \widetilde{c}_i^{(j)}\mathcal{S}_D^{\bm{\tau}\omega}[\bm{f}_i^{(j)}] + \mathcal{O}(\omega^2+\bm{\delta}) \\
        &= \sum\limits_{i,j} \widetilde{c}_i^{(j)}\bigg(\bm{\psi}_i^{(j)} - \frac{\mathrm{i}}{12\pi\rho}\Big(\frac{2}{c_s^3}+\frac{1}{c_p^3}\Big)\sum\limits_{k=1}^N \tau_k\omega\int_{\partial D_k} \bm{f}_i^{(j)}(\bm{y})\mathrm{d}\sigma(\bm{y})\chi_{\partial D_k}\bigg) +\mathcal{O}(\omega^2+\bm{\delta}).
    \end{aligned}
    \end{equation*}
    This implies that
    \begin{equation*}                      
        \bm\phi=\sum_{i,j}c_i^{(j)}(\mathcal S_D^{\omega\bm\alpha_0,\omega})^{-1}[\bm\psi_i^{(j)}]
        +\mathcal{O}(|\omega|^2+\bm\delta)\|\bm c\|.
    \end{equation*}
    
    Notice that $\bm{\psi}_{i}^{(j)}\in\mathrm{Ker}(-\frac{1}{2}\mathcal{I}+\mathcal{K}_D^{0,0})$. Therefore,
    \begin{equation}\label{eq:omega0-3}
        \int_{\partial D} (\frac{1}{2}\mathcal I+(\mathcal{K}_D^{\omega\bm\alpha_0,\omega})^*)[\bm{\phi}]\cdot\bm{\psi}_{i'}^{(j')}\mathrm{d}\sigma =-\sum\limits_{i,j}\widehat{\bm C}_{i,i'}^{(j,j')}(\omega)c_i^{(j)}+\mathcal{O}(|\omega|^2+\bm\delta)\|\bm c\|.
    \end{equation}
 
    Combining equations \eqref{eq:omega0-2} and \eqref{eq:omega0-3} we conclude that the resonance frequency $\omega$ must satisfy
    $$
       \operatorname{det}\left[(\widehat{\mathfrak{C}}(\omega) + \mathfrak{R}(\omega,\bm\delta)^\top)\updelta(\mathfrak M\uptau)^{-1}-\rho\omega^2\bm I_{6N}\right]=0,
    $$
    where $\|\mathfrak{R}(\omega,\bm{\delta})\|=\mathcal{O}(\bm{\delta}+|\omega|^2+|\omega|^4/\bm{\delta})$. Then the leading-order part of resonant frequencies are in the form of $(\lambda_{k,0}^\alpha/\rho)^{1/2}$, where $\lambda_{k,0}^\alpha$ are eigenvalues of $\widehat{\mathfrak{C}}_0\updelta(\mathfrak M\uptau)^{-1}$.
    
    For $d_l=1$, from \eqref{eq:capacitance-corrections} we know that $\widehat{\mathfrak{C}}(\omega)-\widehat{\mathfrak{C}}_0 = \mathcal{O}(|\ln\omega|^{-1})$, and $\omega = \mathcal{O}(\sqrt{\bm\delta})$, so $\omega_k = \sqrt{\lambda_{k,0}^\alpha/\rho} + \mathcal{O}(\sqrt{\bm\delta}/|\ln{\bm\delta}|)$. The result for $d_l=2$ can be concluded similarly from \eqref{eq:capacitance-corrections}.
    
    To determine the order of resonant frequencies corresponding to zero-eigenvalue of $\widehat{\mathfrak{C}}_0$, we define $T:\mathbb{C}^3\rightarrow\mathbb{C}^{6N}$ by $T\bm b:=(\bm b, \bm b, \cdots, \bm b)$ for any constant vector $\bm b\in\mathbb{C}^3$. Then we have $\operatorname{Ker}\widehat{\mathfrak{C}}_0=\operatorname{Ran}T$. Furthermore, from the definition of $\mathfrak{M}$ it follows that for any $\bm b\in\mathbb{C}^3$, 
    $$
        (T\bm b)^\top\mathfrak{M}\uptau\updelta^{-1}(T\bm b) = \left(\sum_i \frac{\tau_i^2|D_i|}{\delta_i}\right)|\bm b|^2.
    $$
    Define $m_\delta:=\sum\limits_{i=1}^N\tau_i^2|D_i|/\delta_i$, then $T^\top\mathfrak{M}\uptau\updelta^{-1}T = m_\delta I_3$. This implies that $m_\delta=\mathcal{O}(\bm\delta^{-1})$.
    
    For $d_l=1$, recall
    $$
        \widehat{\mathfrak{C}}(\omega) = \widehat{\mathfrak{C}}_0-\widehat{\mathfrak{C}}_1(\ln\omega)^{-1} + o(|\ln\omega|^{-1}).
    $$
    Furthermore, $T^\top\widehat{\mathfrak{C}}_1T=K_0^{-1}$. We then have
    $$
    T^\top\widehat{\mathfrak{C}}(\omega)T-T^\top\rho\omega^2(\mathfrak{M}\uptau)\updelta^{-1}T\bm = -K_0^{-1}(\ln\omega)^{-1} - \rho\omega^2m_\delta \bm I_3+o(|\ln\omega|^{-1}).
    $$ 
    The equation above implies that $\omega^2/\bm\delta=\mathcal{O}(|(\ln\omega)^{-1}|)$. Taking logarithm on both sides yields $\ln(1/\omega) = |\ln\bm\delta|/2+\mathcal{O}(\ln|\ln\bm\delta|)$. This proves the formula for resonant frequencies \eqref{eq:resonantfrequency-dl1}. The result for $d_l=2$ follows similarly. From (SM2.15), eigenvalues of $\bm B^{\bm\alpha_0}$ are negative-imaginary numbers, which means that $\zeta_j(\bm\delta)$ are also negative-imaginary numbers.
