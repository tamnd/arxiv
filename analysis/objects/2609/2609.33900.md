<!-- theorem-1: body_md -->
\label{t:main}
    Let $P$ be an increasing stochastic kernel on $(\mathsf X, \mathcal B)$.  Then $P$ is
    globally stable if and only if $P$ satisfies the weak monotone mixing
    condition.  In this case, if $n \in \mathbbm N$ and $\varepsilon > 0$ are such that
    $P^n_b$ is partially dominated by $P^n_a$ at level $\varepsilon$, then the
    stationary distribution $\mu^*$ obeys
     
    \begin{equation}\label{eq:rate}
        \kappa(\mu P^t, \mu^*)
        \leq 4 (1 - \varepsilon)^{\lfloor t/n \rfloor}
        \qquad \text{ for every $\mu \in \mathcal P$ and every $t \geq 0$}.
    \end{equation}
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{l:straddle}
    If $s \in \mathsf X$ is a straddle point for $(\mu, \nu) \in \mathcal P \times \mathcal P$,
    then $\mu$ is partially dominated by $\nu$ at level
    $\min \{ \mu([a,s]), \, \nu([s,b]) \}$.
<!-- end lemma-1 -->
<!-- theorem-2: body_md -->
\label{t:mmc}
    Let $\mathsf X = [a,b] \subset \mathbbm R$ with the usual order and let $P$ be an
    increasing stochastic kernel on $(\mathsf X, \mathcal B)$.  Then $P$ is globally stable
    if and only if $P$ satisfies the \textup{MMC}.
<!-- end theorem-2 -->
<!-- lemma-2: body_md -->
\label{l:binding}
    There exist $n \in \mathbbm N$ and $p > 0$ such that
    $\mathbbm P\{k_n = 0 \, | \, x_0 = b\} \geq p$.

<!-- proof-1: body_md -->
Let $c(x) \coloneqq wz + (1+r)k - g(x)$ be the consumption policy.  It is
    continuous by the theorem of the maximum.  Moreover, $wz \geq \underline w
    \, \underline z > 0$ at every $x \in \mathsf X$, so the Inada condition implies
    $c(x) > 0$; since $\mathsf X$ is compact, $\underline c \coloneqq \min_{x \in
    \mathsf X} c(x) > 0$.
    Suppose, contrary to the claim, that $\mathbbm P\{k_n = 0 \, | \, x_0 = b\} =
    0$ for every $n \in \mathbbm N$.  Then, starting from $b$, we have $0 < k_{t+1}
    < \bar k$ almost surely at every date, so the Euler equation holds with
    equality:  $u'(c_t) = \beta \, \mathbbm E_t [(1 + r_{t+1}) \, u'(c_{t+1})]$.
    Iterating gives
     
    \begin{equation*}
        u'(c_0)
        = \mathbbm E_0 \left[ \prod_{i=1}^n \beta (1 + r_i) \, u'(c_n) \right]
        \leq (\beta (1 + \bar r))^n \, u'(\underline c)
        \qquad (n \in \mathbbm N) .
    \end{equation*}
     
    Since $\beta(1 + \bar r) < 1$, letting $n \to \infty$ gives $u'(c_0)
    \leq 0$, a contradiction.

<!-- proof-3: body_md -->
Suppose first that $P$ is globally stable, and let $\mu^*$ be its
    stationary distribution.  Since $P^t_b = \delta_b P^t$ and $P^t_a = \delta_a P^t$, global
    stability gives
     
    \begin{equation*}
        \kappa(P^t_b, P^t_a)
        \leq \kappa(P^t_b, \mu^*) + \kappa(\mu^*, P^t_a)
        \to 0
        \quad (t \to \infty) .
    \end{equation*}
     
    Since $\alpha \leq 1$, definition \eqref{eq:tov} gives $1 - \alpha(P^t_b,
    P^t_a) \leq \gamma(P^t_b, P^t_a)$, and $\gamma \leq \kappa$ by
    \eqref{eq:gk}.  Hence
     
    \begin{equation}\label{eq:atone}
        \alpha(P^t_b, P^t_a) \to 1
        \quad (t \to \infty) ,
    \end{equation}
     
    and in particular $P$ satisfies the WMMC.

    Suppose now that $P$ satisfies the WMMC, and fix $n \in \mathbbm N$ such that $P^n_b$ is
    partially dominated by $P^n_a$.  By \eqref{eq:alf}, this means that
    $\alpha_n \coloneqq \alpha(P^n_b, P^n_a) > 0$.  We claim
    that
     
    \begin{equation}\label{eq:corner}
        \alpha(P^n_x, P^n_{x'}) \geq \alpha_n
        \qquad (x, x' \in \mathsf X) .
    \end{equation}
     
    To see this, fix $x, x' \in \mathsf X$.  Since $x \preceq b$ and $a \preceq x'$,
    we have $\delta_x \preceq \delta_b$ and $\delta_a \preceq \delta_{x'}$.  As
    $P^n$ is increasing, it follows that $\delta_x P^n \preceq \delta_b P^n$ and
    $\delta_a P^n \preceq \delta_{x'} P^n$, so \eqref{eq:amon} gives
     
    \begin{equation*}
        \alpha_n
        = \alpha(\delta_b P^n, \delta_a P^n)
        \leq \alpha(\delta_x P^n, \delta_{x'} P^n)
        = \alpha(P^n_x, P^n_{x'}) ,
    \end{equation*}
     
    which is \eqref{eq:corner}.  Since equality holds at $(x, x') = (b, a)$, the
    infimum of $\alpha(P^n_x, P^n_{x'})$ over $\mathsf X \times \mathsf X$ is exactly
    $\alpha_n > 0$.  This is the hypothesis of Corollary~5.1 of
    \cite{kamihigashi2019unified}, which therefore applies:  the kernel $P$ has a
    unique stationary distribution $\mu^* \in \mathcal P$, and
     
    \begin{equation}\label{eq:grate}
        \gamma(\mu P^t, \mu^*)
        \leq (1 - \alpha_n)^{\lfloor t/n \rfloor} \gamma(\mu, \mu^*)
        \qquad (\mu \in \mathcal P, \; t \geq 0) .
    \end{equation}
     
    In particular, by \eqref{eq:gk}, $\kappa(\mu P^t, \mu^*) \to 0$ for every
    $\mu \in \mathcal P$, so $P$ is globally stable.

    Turning to the rate bound, let $n \in \mathbbm N$ and $\varepsilon > 0$ be such
    that $P^n_b$ is partially dominated by $P^n_a$ at level $\varepsilon$.  Then $\alpha_n \coloneqq
    \alpha(P^n_b, P^n_a) \geq \varepsilon > 0$ by \eqref{eq:alf}, so the argument
    leading to \eqref{eq:grate} applies at this $n$.  Combining
    \eqref{eq:grate} with $\alpha_n \geq \varepsilon$ and \eqref{eq:gk} gives
     
    \begin{equation*}
        \kappa(\mu P^t, \mu^*)
        \leq 2 \gamma(\mu P^t, \mu^*)
        \leq 2 (1 - \varepsilon)^{\lfloor t/n \rfloor} \gamma(\mu, \mu^*)
        \leq 4 (1 - \varepsilon)^{\lfloor t/n \rfloor}
    \end{equation*}
     
    for all $\mu \in \mathcal P$ and $t \geq 0$, where the last step uses
    $\gamma(\mu, \mu^*) \leq \kappa(\mu, \mu^*) \leq 2$.  This is
    \eqref{eq:rate}.

Sufficiency is immediate from \cref{l:straddle}.  For necessity we argue by
    contraposition.  Assume that, for every $s \in [a,b]$, either $\mu([a,s]) =
    0$ or $\nu([s,b]) = 0$.  Set
     
    \begin{equation*}
        \tau \coloneqq \sup \left( \{a\} \cup
         \{ r \in [a,b] : \mu([a,r]) = 0 \}  \right) .
    \end{equation*}
     
    Since $r \mapsto \mu([a,r])$ is nondecreasing, the set in question is an
    interval containing its smaller points, so $\mu([a,r]) = 0$ for every $r < \tau$
    and $\mu([a,r]) > 0$ for every $r > \tau$.  In particular $\mu([a,\tau)) = 0$.
    Define $I$ as $[\tau, b]$ if $\mu(\{\tau\}) > 0$ and $(\tau, b]$ otherwise.
    In either case $I$ is an increasing Borel subset of $\mathsf X$ with $\mu(I) = 1$.
    We claim that $\nu(I) = 0$.  Given the claim, $\mu(I) - \nu(I) = 1$, so the
    supremum in \eqref{eq:dual} equals one and $\alpha(\mu, \nu) = 0$.  By
    \eqref{eq:alf}, $\mu$ is then not partially dominated by $\nu$ at any
    positive level, and the proof is complete.

    Suppose first that $\mu(\{\tau\}) > 0$.  Then $\mu([a,\tau]) > 0$, so $\nu([\tau,b])
    = 0$ by hypothesis, and $I = [\tau,b]$. Suppose instead that $\mu(\{\tau\}) = 0$.
    Then $\mu([a,\tau]) = 0$, and since $\mu([a,b]) = 1$ this forces $\tau < b$.  For
    every $s \in (\tau, b]$ we have $\mu([a,s]) > 0$ and hence $\nu([s,b]) = 0$.
    Letting $s$ decrease to $\tau$ along $(\tau,b]$ gives $\nu(I) = \nu((\tau,b]) = 0$.
