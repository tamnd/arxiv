<!-- theorem-1: body_md -->
\label{thm:norm-inflation}
Let $d \le 3$, $\sigma\in\{-1,1\}$, and $1\leq p<\infty$.
There exist a compact set $K_x\subset\mathbb R^d$, a radius $R>0$, a sequence
$t_n\downarrow0$, and data $f_{0,n}\in\mathcal D$ such that
\begin{equation}\label{eq:theorem-data}
 0\leq f_{0,n}\leq1,\qquad
 \operatorname{supp}f_{0,n}\subset K_x\times B_R,\qquad
 \left\lVert f_{0,n}\right\rVert_{X_p}\to0,\qquad
 \left\lVert f_{0,n}\right\rVert_{L^1_{x,v}}\to0,
\end{equation}
while the associated classical solutions satisfy
\begin{equation}\label{eq:theorem-inflation}
 \left\lVert f_n(t_n)\right\rVert_{X_p}\geq1.
\end{equation}
In particular, for every $T>0$, the trajectory map $\mathcal S_T$ is
discontinuous at zero.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:fixed-time}
Let $d \le 3$ and $\sigma\in\{-1,1\}$.
There exist $t_*>0$, $R>0$, and a compact set $K_x\subset\mathbb R^d$, independent
of $p$, $t$, and $n$, with the following property.  For every
$1\leq p<\infty$ and $t\in(0,t_*)$ there are data
$f_{0,n}^{(t)}\in\mathcal D$ satisfying
\begin{equation}\label{eq:fixed-data}
 \begin{gathered}
 0\leq f_{0,n}^{(t)}\leq1,\qquad
 \operatorname{supp}f_{0,n}^{(t)}\subset K_x\times B_R,\qquad 
 \left\lVert f_{0,n}^{(t)}\right\rVert_{X_p}\longrightarrow0,\qquad
 \left\lVert f_{0,n}^{(t)}\right\rVert_{L^1_{x,v}}\longrightarrow0,
 \end{gathered}
\end{equation}
but
\begin{equation}\label{eq:fixed-conclusion}
 \left\lVert f_n^{(t)}(t)\right\rVert_{X_p}\geq1
\end{equation}
for every $n$.  Hence $S_t:(\mathcal D,\left\lVert \cdot\right\rVert_{X_p})\to X_p$ is
discontinuous at zero.
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
The condition $d \le 3$ enter only through the global classical
	existence theory for smooth compactly supported data. In higher dimensions, the same argument applies whenever the
	classical solutions exist up to their observation times.
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lem:field}
Let $d\geq1$ and let $\rho\in L^1(\mathbb R^d)\cap L^\infty(\mathbb R^d)$.  Then
\begin{equation}\label{eq:field-bound}
 \left\lVert K_d*\rho\right\rVert_{L^\infty}
 \leq C_d
 \left\lVert \rho\right\rVert_{L^1}^{1/d}
 \left\lVert \rho\right\rVert_{L^\infty}^{1-1/d}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $x\in\mathbb R^d$ and a radius $s>0$.  Since
$\left\lvert K_d(z)\right\rvert\leq C_d\left\lvert z\right\rvert^{1-d}$, splitting the convolution into
$\left\lvert x-y\right\rvert\leq s$ and $\left\lvert x-y\right\rvert>s$ gives
\begin{align}
 \left\lvert (K_d*\rho)(x)\right\rvert
 &\leq C_d\left\lVert \rho\right\rVert_{L^\infty}
       \int_{\left\lvert z\right\rvert\leq s}\left\lvert z\right\rvert^{1-d}\,\mathrm d z
   +C_d s^{1-d}\left\lVert \rho\right\rVert_{L^1} \notag\\
 &\leq C_d\left(
       s\left\lVert \rho\right\rVert_{L^\infty}
       +s^{1-d}\left\lVert \rho\right\rVert_{L^1}\right).
 \label{eq:field-split}
\end{align}
If $\rho=0$, there is nothing to prove.  Otherwise choose
\begin{equation*}
 s=\left(\frac{\left\lVert \rho\right\rVert_{L^1}}
                 {\left\lVert \rho\right\rVert_{L^\infty}}\right)^{1/d}.
\end{equation*}
Substitution in \eqref{eq:field-split} gives \eqref{eq:field-bound}.

\label{lem:characteristics}
Let $T>0$ and let $f$ be the classical solution of \eqref{eq:VP}
with nonnegative initial datum $f_0\in C_c^\infty(\mathbb R^{2d})$.
Let $R_0>0$, and suppose
\begin{equation}\label{eq:char-assumptions}
 0\leq f_0\leq1,
 \qquad
 \operatorname{supp}_v f_0\subset B_{R_0},
 \qquad
 M:=\iint_{\mathbb R^{2d}}f_0\,\mathrm d x\,\mathrm d v<\infty.
\end{equation}
There are constants $c_0=c_0(d,R_0)>0$ and
$C_0=C_0(d,R_0)>0$ such that, if
\begin{equation}\label{eq:smallness-TM}
 T M^{1/d}\leq c_0,
\end{equation}
then, for $0\leq s\leq T$,
\begin{equation}\label{eq:P-E-bounds}
 \operatorname{supp}_v f(s)\subset B_{R_0+1},
 \qquad
 \left\lVert E_f(s)\right\rVert_{L^\infty}\leq C_0M^{1/d}.
\end{equation}
Moreover, every backward characteristic ending at $(x,v)$ at time
$t\leq T$ satisfies
\begin{align}

 &\leq C_0(t-s)M^{1/d},
 &&0\leq s\leq t, \label{eq:V-error}\\
 \left\lvert X(0;t,x,v)-(x-tv)\right\rvert
 &\leq \frac{C_0}{2}t^2M^{1/d}.
 \label{eq:X-error}
\end{align}

<!-- proof-2: body_md -->
The transport identity \eqref{eq:transport} and incompressibility of the
phase-space vector field imply
\begin{equation}\label{eq:conservation}
 0\leq f(s)\leq1,
 \qquad
 \iint f(s,x,v)\,\mathrm d x\,\mathrm d v=M
\end{equation}
for every time for which the classical solution exists.
If $M=0$, nonnegativity and \eqref{eq:conservation} imply
$f\equiv0$, so the conclusions hold.  We may therefore assume $M>0$.

Define the nondecreasing velocity-support function
\begin{equation}\label{eq:Pdef}
 P(s)=\sup\left\{\left\lvert v\right\rvert:
       f(\tau,x,v)\neq0\text{ for some }0\leq\tau\leq s
       \text{ and }x\in\mathbb R^d\right\}.
\end{equation}
Let $(X(\tau;0,x_0,v_0),V(\tau;0,x_0,v_0))$ denote the forward
characteristic from $(x_0,v_0)$.  Since the flow is continuous and
$\operatorname{supp}f_0$ is compact,
\[
 P(s)=\max_{\substack{(x_0,v_0)\in\operatorname{supp}f_0\\0\leq\tau\leq s}}
       \left\lvert V(\tau;0,x_0,v_0)\right\rvert.
\]
Thus $P$ is finite and continuous, by uniform continuity of $V$ on
the compact set of initial points and times.

By \eqref{eq:conservation},
\begin{equation}\label{eq:rho-P}
 \rho_f(s,x)
 =\int_{\left\lvert v\right\rvert\leq P(s)}f(s,x,v)\,\mathrm d v
 \leq \left\lvert B_1\right\rvert P(s)^d.
\end{equation}
Mass conservation gives $\left\lVert \rho_f(s)\right\rVert_{L^1}=M$.  Lemma~\ref{lem:field}
therefore yields
\begin{equation}\label{eq:E-P}
 \left\lVert E_f(s)\right\rVert_{L^\infty}
 \leq C_dM^{1/d}P(s)^{d-1}.
\end{equation}

Set $R_1=R_0+1$, and let
\begin{equation*}
 T_1=\sup\left\{\tau\in[0,T]:P(s)\leq R_1
                    \text{ for every }0\leq s\leq\tau\right\}.
\end{equation*}
This number is positive because $P(0)\leq R_0$.  For $0\leq s\leq T_1$,
\eqref{eq:E-P} gives
\begin{equation}\label{eq:E-bootstrap}
 \left\lVert E_f(s)\right\rVert_{L^\infty}
 \leq C_dR_1^{d-1}M^{1/d}=:C_0M^{1/d}.
\end{equation}
Along any characteristic issuing from the initial support,
\begin{equation*}
 \left\lvert V(s)\right\rvert
 \leq R_0+\int_0^s\left\lVert E_f(\tau)\right\rVert_{L^\infty}\,\mathrm d\tau
 \leq R_0+C_0TM^{1/d}.
\end{equation*}
Choose $c_0>0$ so that $C_0c_0\leq1/2$.  Under
\eqref{eq:smallness-TM}, we then have $P(s)\leq R_0+1/2<R_1$ on
$[0,T_1]$.  By continuity, $T_1<T$ would contradict the definition of
$T_1$.  Hence $T_1=T$, proving \eqref{eq:P-E-bounds}.

For a backward characteristic with $(X(t),V(t))=(x,v)$, integration of the
second equation in \eqref{eq:characteristics} gives
\begin{equation*}
 V(s)-v=-\sigma\int_s^tE_f(\tau,X(\tau))\,\mathrm d\tau.
\end{equation*}
The field bound in \eqref{eq:P-E-bounds} proves \eqref{eq:V-error}.  Since
\begin{equation*}
 X(0)=x-\int_0^tV(s)\,\mathrm d s,
\end{equation*}
we also obtain
\begin{align*}
 \left\lvert X(0)-(x-tv)\right\rvert
 &\leq\int_0^t\left\lvert V(s)-v\right\rvert\,\mathrm d s\\
 &\leq C_0M^{1/d}\int_0^t(t-s)\,\mathrm d s
 =\frac{C_0}{2}t^2M^{1/d},
\end{align*}
which is \eqref{eq:X-error}.

\label{prop:packets}
For $d \le 3$ and either $\sigma\in\{-1,1\}$, there exist
$h_*\in(0,1]$ and $r_*>0$, depending only on $d$, $\eta$,
and $\psi$, such that for every
\[
 0<h<h_*,
 \qquad
 0<r<\min\{h/4,r_*\},
\]
the datum \eqref{eq:f-hr} has the following properties for every
$1\leq p<\infty$:
\begin{align}
 &f_0^{h,r}\in C_c^\infty(\mathbb R^{2d}),
 \qquad
 0\leq f_0^{h,r}\leq1,
 \qquad
 \operatorname{supp}_vf_0^{h,r}\subset B_R,
 \label{eq:packet-basic}\\
 &\left\lVert f_0^{h,r}\right\rVert_{X_p}
 \leq C_{d,p,\eta}\left(\frac rh\right)^{d/p},
 \label{eq:packet-Xp}\\
 &M_{h,r}:=\left\lVert f_0^{h,r}\right\rVert_{L^1_{x,v}}
 \leq C_{d,\eta,\psi}\left(\frac rh\right)^d.
 \label{eq:packet-mass}
\end{align}
If $f^{h,r}$ is the corresponding classical solution, then
\begin{equation}\label{eq:packet-output-pointwise}
 \mathop{\rm ess\,sup}_{v\in\mathbb R^d}f^{h,r}(h,x,v)=1
 \qquad\text{for every }x\in Q.
\end{equation}
In particular,
\begin{equation}\label{eq:packet-output}
 \left\lVert f^{h,r}(h)\right\rVert_{X_p}\geq\left\lvert Q\right\rvert^{1/p}=1.
\end{equation}

<!-- proof-3: body_md -->
The number of grid points satisfies
\[
 \#\Gamma_h\leq C_dh^{-d}.
\]
Because the supports of the terms in \eqref{eq:a-hr} are disjoint,
$0\leq a_{h,r}\leq1$, which gives \eqref{eq:packet-basic}.
Since $h<1$ and $r<h/4<1/4$, we also have the common spatial support bound
\begin{equation}\label{eq:common-support}
 \operatorname{supp}a_{h,r}\subset K_x:=[-2,3]^d.
\end{equation}
Moreover,
\begin{align}
 \left\lVert f_0^{h,r}\right\rVert_{X_p}^p
 &=\left\lVert a_{h,r}\right\rVert_{L^p_x}^p \notag\\
 &=(\#\Gamma_h)r^d\left\lVert \eta\right\rVert_{L^p}^p
 \leq C_{d,\eta}h^{-d}r^d.
 \label{eq:packet-Xp-proof}
\end{align}
Taking the $p$th root proves \eqref{eq:packet-Xp}.  Similarly,
\begin{align}
 M_{h,r}
 &=\left(\int_{\mathbb R^d}a_{h,r}(x)\,\mathrm d x\right)
   \left(\int_{\mathbb R^d}\psi(v)\,\mathrm d v\right)\notag\\
 &=(\#\Gamma_h)r^d\left\lVert \eta\right\rVert_{L^1}\left\lVert \psi\right\rVert_{L^1}
 \leq C_{d,\eta,\psi}\left(\frac rh\right)^d,
 \label{eq:packet-mass-proof}
\end{align}
which proves \eqref{eq:packet-mass}.

To estimate the displacement caused by the field, note first that
\eqref{eq:packet-mass} gives
\begin{equation}\label{eq:M-root}
 M_{h,r}^{1/d}\leq C_1\frac rh.
\end{equation}
Take $T=h$ in Lemma~\ref{lem:characteristics}.  Its smallness condition is
satisfied once
\begin{equation*}
 hM_{h,r}^{1/d}\leq C_1r\leq c_0.
\end{equation*}
Write $E^{h,r}:=E_{f^{h,r}}$.  Under this condition, for $0\leq s\leq h$,
\begin{equation}\label{eq:packet-field}
 \left\lVert E^{h,r}(s)\right\rVert_{L^\infty}\leq C_2\frac rh.
\end{equation}
For every backward characteristic ending at $(x,v)$ at time $h$,
Lemma~\ref{lem:characteristics} and \eqref{eq:M-root} give
\begin{align}
 \left\lvert V(0;h,x,v)-v\right\rvert
 &\leq C_3hM_{h,r}^{1/d}\leq C_3r,
 \label{eq:packet-V-error}\\
 \left\lvert X(0;h,x,v)-(x-hv)\right\rvert
 &\leq C_3h^2M_{h,r}^{1/d}\leq C_3hr.
 \label{eq:packet-X-error}
\end{align}
Here $C_3=C_0C_1$ is admissible, with $C_0$ as in
Lemma~\ref{lem:characteristics} for $R_0=R$.

Fix $x\in Q$.  For $h<1$, a nearest point of the full lattice
$h\mathbb Z^d$ to $x$ belongs to $[-1,2]^d$.  Hence there is a point
$z=z_h(x)\in\Gamma_h$ such that
\begin{equation}\label{eq:nearest-grid}
 \left\lvert x-z\right\rvert\leq\frac{\sqrt d}{2}h.
\end{equation}
Define the final velocity
\begin{equation}\label{eq:target-velocity}
 v_h(x)=\frac{x-z_h(x)}h.
\end{equation}
Then
\begin{equation}\label{eq:target-velocity-bound}
 \left\lvert v_h(x)\right\rvert\leq\frac{\sqrt d}{2}.
\end{equation}
For free transport, the backward trajectory with terminal point
$(x,v_h(x))$ lands exactly at
\[
 (x-hv_h(x),v_h(x))=(z_h(x),v_h(x)).
\]
The spatial component is the center of the $z_h(x)$ packet, and the
velocity component lies strictly inside the plateau of $\psi$.  We now
verify that the nonlinear characteristic stays within these two
plateaux.

Let $(X(s),V(s))$ be the backward characteristic satisfying
$(X(h),V(h))=(x,v_h(x))$.  Set
\[
 h_*=\min\left\{1,\frac{1}{4C_3}\right\},
 \qquad
 r_*=\min\left\{\frac{c_0}{2C_1},\frac{\sqrt d}{2C_3}\right\}.
\]
These choices give
\begin{equation}\label{eq:hstar-choice}
 C_3h_*<\frac12
\end{equation}
and
\begin{equation}\label{eq:rstar-choice}
 C_1r_*\leq c_0,
 \qquad
 C_3r_*<\sqrt d.
\end{equation}
Because $x-hv_h(x)=z_h(x)$, the spatial error estimate and
\eqref{eq:hstar-choice} give
\begin{equation}\label{eq:initial-X-plateau}
 \left\lvert X(0)-z_h(x)\right\rvert<\frac r2,
 \qquad
 a_{h,r}(X(0))=1.
\end{equation}
Indeed, the first inequality places $X(0)$ in the ball on which the
$z_h(x)$ summand in \eqref{eq:a-hr} equals one; disjointness and
$0\leq a_{h,r}\leq1$ then give the second equality.  For the velocity
component, \eqref{eq:target-velocity-bound},
\eqref{eq:packet-V-error}, and \eqref{eq:rstar-choice} yield the explicit
margin
\[
 \left\lvert V(0)\right\rvert
 \leq \left\lvert v_h(x)\right\rvert+C_3r
 <\frac{\sqrt d}{2}+\sqrt d
 =\frac{3\sqrt d}{2}<2\sqrt d.
\]
Hence
\[
 \psi(V(0))=1.
\]
The plateau estimates are uniform in $x\in Q$ and $r<r_*$.  Conservation
along characteristics therefore gives
\begin{equation}\label{eq:value-one}
 f^{h,r}(h,x,v_h(x))
 =f_0^{h,r}(X(0),V(0))=1.
\end{equation}

The solution is continuous and satisfies $0\leq f^{h,r}\leq1$.
For each $0<\varepsilon<1$, \eqref{eq:value-one} and continuity in
velocity give an open ball on which $f^{h,r}(h,x,v)>1-\varepsilon$.
Since this ball has positive measure,
\begin{equation*}
 \mathop{\rm ess\,sup}_{v\in\mathbb R^d}f^{h,r}(h,x,v)=1.
\end{equation*}
This is \eqref{eq:packet-output-pointwise}.  Integrating over $Q$ gives
\eqref{eq:packet-output}.

<!-- corollary-1: body_md -->
\label{cor:free-transport}
Let $d \le 3$ and $1\leq p<\infty$.  For every $h\in(0,1)$, the
fixed-time free-transport map
\[
 U(h):(\mathcal D,\left\lVert \cdot\right\rVert_{X_p})\longrightarrow X_p,
 \qquad
 U(h)g(x,v)=g(x-hv,v),
\]
is discontinuous at zero.  Moreover, for every $T>0$, the trajectory map
\[
 \mathcal U_T:(\mathcal D,\left\lVert \cdot\right\rVert_{X_p})
 \longrightarrow L^\infty([0,T];X_p),
 \qquad
 \mathcal U_Tg=U(\cdot)g,
\]
is discontinuous at zero.
In both assertions, the approximating data may be chosen nonnegative,
bounded by one, supported in a common compact subset of phase space, and
convergent to zero in $L^1_{x,v}$.

Fix $h\in(0,1)$ and choose $r_n\downarrow0$ with $r_n<h/4$.  Set
$g_n=f_0^{h,r_n}$.  By construction, $0\leq g_n\leq1$ and

 =a_{h,r_n}(z_h(x))\psi(v_h(x))=1.
