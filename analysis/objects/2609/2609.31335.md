<!-- proposition-1: body_md -->
\label{prop.upgen}
Suppose that $G$ is a discrete Abelian group and $A \subset G$ has size $n$. Then
\begin{equation*}
\|\widehat{1_A}\|_1 \leq \sqrt{n-1}+\frac{1}{n + \sqrt{n-1}}.
\end{equation*}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
This improvement rests on the fact, used in \cite[(3.3), p.~61]{bou::7}, that
\begin{align}\nonumber 
\|\widehat{1_A}\|_4^4 =\sum_{z \in G}{1_A \ast 1_{-A}(z)^2}& \geq 1_A \ast 1_{-A}(0)^2 + \sum_{z \neq 0}{1_A \ast 1_{-A}(z)}\\ \label{eqn.l4}  & = n^2 + n(n-1) = 2n^2-n.
\end{align}
The conclusion of the proposition is certainly true if $n=1$ since both sides are $1$ in that case. Assume $n>1$ and write $r:=\sqrt{n-1}$ and
\begin{equation*}
\alpha_0:=\frac{nr(n+2r)}{2(n+r)^2}, \alpha_1:=\frac{n^2+2nr+3r^2}{2r(n+r)^2}, \alpha_2:= \frac{1}{2r(n+r)^2}.
\end{equation*}
Then
\begin{equation*}
\alpha_0+\alpha_1t^2-\alpha_2t^4-t=\frac{(n-t)(t-r)^2(n+2r+t)}{2r(n+r)^2} \geq 0 \text{ for all }t \in [0,n].
\end{equation*}
Since $|\widehat{1_A}(\gamma)| \in [0,n]$ for all $\gamma \in\Gamma$ we may take $t=|\widehat{1_A}(\gamma)|$ in this inequality and integrate over $\gamma$ to get
\begin{align*}
\|\widehat{1_A}\|_1 & \leq \alpha_0 +\alpha_1\|\widehat{1_A}\|_2^2 -\alpha_2\|\widehat{1_A}\|_4^4\\ & \leq \alpha_0 +\alpha_1n -\alpha_2(2n^2-n)=\sqrt{n-1}+\frac{1}{n + \sqrt{n-1}},
\end{align*}
where we used Parseval's theorem to compute $\|\widehat{1_A}\|_2^2$ and (\ref{eqn.l4}) and non-negativity of $\alpha_2$ to bound $-\alpha_2\|\widehat{1_A}\|_4^4$. The result is proved.
<!-- end proof-1 -->
<!-- remark-1: body_md -->
The values of $\alpha_0$, $\alpha_1$, and $\alpha_2$ in the argument above can be found by solving the optimisation problem
\begin{equation*}
\min\{\alpha_0+\alpha_1n-\alpha_2(2n^2-n): \alpha_2 \geq 0, t \leq \alpha_0+\alpha_1t^2-\alpha_2t^4\text{ for all }t \in [0,n]\}.
\end{equation*}
<!-- end remark-1 -->
<!-- example-1: body_md -->
\label{ex.g}
For $A \subset G$ a perfect difference set, $\|\widehat{1_A}\|_1$ attains the bound in Proposition \ref{prop.upgen}. Indeed, taking the Fourier transform of (\ref{eqn.ft}), we get
\begin{equation}\label{eqn.flat}
|\widehat{1_A}(\gamma)|=\sqrt{n-1} \text{ for all }\gamma \neq 0.
\end{equation}
Hence
\begin{equation*}
 \|\widehat{1_A}\|_1 = \frac{1}{|G|}(n + (|G|-1)\sqrt{n-1}) =\sqrt{n-1}+\frac{1}{n + \sqrt{n-1}}.
\end{equation*}

Suppose that $A$ is defined as in (\ref{eqn.ruzsa}), $F$ as in (\ref{eqn.Fmod}), $N=p(p-1)$, and $k\in \mathbb{N}$. Then 
\begin{equation}\label{eqn.mainest2}
\int_0^1{|f_A(\theta)|^{2k}\,\mathrm{d}\theta} = \int_0^1{\mathop{\mathbb{E}}|F(\theta)|^{2k}\,\mathrm{d}\theta}+o_{k;p\to\infty}(1).

Write
\begin{equation}\label{eqn.Hjj}
H(j,j') :=\int_0^1{\prod_{i=1}^k\widehat{1_{[N]}}\left(\theta-\frac{j_i}{N}\right) \overline{\prod_{i=1}^{k}\widehat{1_{[N]}}\left(\theta-\frac{j_i'}{N}\right)} \,\mathrm{d}\theta} \text{ for }j,j' \in\mathbb{Z}^{k}.
\end{equation}
The definition is invariant under translation of any $j_i$ (or $j_i'$) by a multiple of $N$, and so we also think of $H$ as a function on $\mathbb{Z}_N^k\times \mathbb{Z}_N^k$. 

From the definition of (\ref{eqn.Fmod}) and the fact that $\mathop{\mathbb{E}}{X_i^r\overline{X_i}^s} =1_{[r=s]}$, we have
\begin{equation}\label{eqn.Fmomentexpansion}
\int_0^1{\mathop{\mathbb{E}}{|F(\theta)|^{2k}}\,\mathrm{d}\theta} = \frac1{N^{2k}} \sum_{\substack{j,j' \in (\mathbb{Z}_N^*)^k\\ \exists \pi \in S_k \text{ s.t.\ }j=\pi^*(j')}}{H(j,j')}.
\end{equation}
Here, we have written $\pi^*(j')$ to mean the vector whose $i$th component is $j_{\pi(i)}'$ . Another way of saying $j=\pi^*(j')$ is that $\{j_1,\dots,j_k\}=\{j_1',\dots,j_k'\}$ as multisets.

Note that $j$ and $j'$ are taken in $(\mathbb{Z}_N^*)^k$, where $\mathbb{Z}_N^*:=\mathbb{Z}_N\setminus \{0\}$, rather than $\mathbb{Z}_N^k$ because the sum in (\ref{eqn.Fmod}) is over non-zero residues.

To understand $H$ we will use two standard estimates:
\begin{equation}\label{eqn.dirichletbound2}
|\widehat{1_{[N]}}(\theta)| = O\left(\min\left(N,\frac1{\|\theta\|_{\mathbb{T}}}\right)\right) \text{ where }\|\theta\|_\mathbb{T}:=\min\{|\theta-z|: z \in \mathbb{Z}\},
\end{equation}
and hence
\begin{equation}\label{eqn.dirichletsum2}
\sum_{j \in \mathbb{Z}_N}{\left|\widehat{1_{[N]}}\left(\theta-\frac{j}{N}\right)\right|} =O( N\log N).
\end{equation}
Later, in (\ref{eqn.fA}), we shall see that $\|f_A\|_{L_{2k}}^{2k}$ is a weighted sum of $H(j,j')$. Most of the diagonal terms, meaning those pairs $(j,j')$ with $j=\pi^*(j')$ for some $\pi \in S_k$, get weight $1$ which gives a contribution of the same size as (\ref{eqn.Fmomentexpansion}). The off-diagonal terms get small weight and to show that this makes their total contribution small we need the $\ell_1$-norm of $H$ to be small. This is what the next two claims show in a form that is useful to us.
\begin{claim*}
For $r \in [k]$ we have
\begin{equation}\label{Hjj}
\sum_{\substack{j,j' \in \mathbb{Z}_N^k\\  p \mid j_r}}{|H(j,j')|} ,\sum_{\substack{j,j' \in \mathbb{Z}_N^k\\  p-1 \mid j_r}}{|H(j,j')|} = O_k(p^{-1}N^{2k}\log^{2k} N).
\end{equation}
\end{claim*}
\begin{proof}
This is two separate, but similar estimates. We start by proving the upper bound for the first sum. The proof of the second is the same with the roles of $p$ and $p-1$ swapped. Use (\ref{eqn.dirichletsum2}) in the $2k-1$ variables $j_1,\dots,j_{r-1},j_{r+1},\dots,j_k,j_1',\dots,j_k'$ and write $j_r=px$ to get
\begin{equation*}
\sum_{\substack{j,j' \in \mathbb{Z}_N^k\\  p \mid j_r}}{|H(j,j')|} = O(N\log N)^{2k-1}\sum_{x \in \mathbb{Z}_{p-1}}{\int_0^1{\left|\widehat{1_{[N]}}\left(\theta-\frac{x}{p-1}\right)\right|\,\mathrm{d} \theta}}.
\end{equation*}
The integral is then the $L_1$-norm of the Dirichlet kernel (after a change of variables $\theta-\frac{x}{p-1} \mapsto \theta$) and so is $O(\log N)$, and $|\mathbb{Z}_{p-1}|=p-1=\frac{N}{p}$. Combining all this gives the claim.
\end{proof}
Similarly, but harder, we have:
\begin{claim*}
For $r,s \in [k]$ we have
\begin{equation}\label{Hjj2}
\sum_{\substack{j,j' \in \mathbb{Z}_N^k\\  j_r \neq j_s' \text{ and }p \mid j_r-j_s'}}{|H(j,j')|} = O_k(p^{-1}N^{2k}\log^{2k} N).
\end{equation}
\end{claim*}
\begin{proof}
Write $S$ for the sum on the left in (\ref{Hjj2}). By (\ref{eqn.dirichletsum2}) in the $2k-2$ variables $j_1,\dots,j_{r-1},j_{r+1},\dots,j_k,j_1',\dots,j_{s-1}',j_{s+1}',\dots j_k'$ we have
\begin{equation*}
S= O(N\log N)^{2k-2}\cdot\sum_{\substack{j_r,j_s' \in \mathbb{Z}_N\\ j_r \neq j_s'\text{ and }p \mid j_r-j_s'}}{\int_0^1{ \left|\widehat{1_{[N]}}\left(\theta - \frac{j_r}{N}\right)\right|\left|\widehat{1_{[N]}}\left(\theta - \frac{j_s'}{N}\right)\right|\,\mathrm{d} \theta}}.
\end{equation*}
By change of variables we have
\begin{equation*}
\int_0^1{ \left|\widehat{1_{[N]}}\left(\theta - \frac{j_r}{N}\right)\right|\left|\widehat{1_{[N]}}\left(\theta - \frac{j_s'}{N}\right)\right|\,\mathrm{d} \theta} = \int_0^1{ \left|\widehat{1_{[N]}}\left(\theta\right)\right|\left|\widehat{1_{[N]}}\left(\theta - \frac{j_r-j_s'}{N}\right)\right|\,\mathrm{d} \theta}.
\end{equation*}
Hence, writing $px=j_r-j_s'$, we have
\begin{align}\label{eqn.Su}
S& =O(N\log N)^{2k-2} \cdot N \cdot \sum_{x\in \mathbb{Z}_{p-1}^*}{\int_0^1{|\widehat{1_{[N]}}(\theta)|\left|\widehat{1_{[N]}}\left(\theta - \frac{x}{p-1}\right)\right|\,\mathrm{d} \theta}}.
\end{align}
Now split the range of integration into
\begin{equation*}
 I_1:=\{\theta \in [0,1]: \|\theta\|_\mathbb{T}\leq \frac1{2(p-1)}\},
 \end{equation*}
 \begin{equation*}
 I_2:=\left\{\theta\in [0,1]: \left\|\theta - \frac x{p-1}\right\|_\mathbb{T}\leq \frac1{2(p-1)}\right\},
 \end{equation*}
 and 
 \begin{equation*}
 I_3:=[0,1]\setminus (I_1 \cup I_2).
 \end{equation*}
 For $x \in \mathbb{Z}_{p-1}^*$ write $d(x):=(p-1)\left\|\frac{x}{p-1}\right\|_\mathbb{T}$. By the triangle inequality for $\|\cdot \|_\mathbb{T}$, if $\theta \in I_1$ then $\left\|\theta - \frac x{p-1}\right\|_\mathbb{T}\geq \frac{d(x)}{2(p-1)}$, and similarly if $\theta \in I_2$ then $\left\|\theta \right\|_\mathbb{T}\geq \frac{d(x)}{2(p-1)}$. It follows from (\ref{eqn.dirichletbound2}) that
\begin{align}
\nonumber \int_{I_1}{|\widehat{1_{[N]}}(\theta)|\left|\widehat{1_{[N]}}\left(\theta - \frac{x}{p-1}\right)\right|\,\mathrm{d} \theta}&  = O\left(\frac{2(p-1)}{d(x)}\right)\cdot \int_{I_1}{\min\left\{N,\frac{1}{\|\theta\|_\mathbb{T}}\right\}\,\mathrm{d} \theta}\\ \label{eqn.I1e} &=O\left(\frac{p}{d(x)}\log N\right),
\end{align}
and a similar estimate for the integral over $I_2$. Finally, the triangle inequality tells us that $\left\| \frac x{p-1}\right\|_\mathbb{T}\leq \|\theta\|_\mathbb{T}+\left\|\theta - \frac x{p-1}\right\|_\mathbb{T}$, which rearranges to
\begin{equation*}
\frac1{\|\theta\|_\mathbb{T}\left\|\theta - \frac x{p-1}\right\|_\mathbb{T}} \leq \frac{1}{\left\| \frac x{p-1}\right\|_\mathbb{T}}\left(\frac{1}{\|\theta\|_\mathbb{T}} + \frac{1}{\left\|\theta - \frac x{p-1}\right\|_\mathbb{T}}\right).
\end{equation*}
Hence by (\ref{eqn.dirichletbound2}) we have
\begin{align}\nonumber 
\int_{I_3}{|\widehat{1_{[N]}}(\theta)|\left|\widehat{1_{[N]}}\left(\theta - \frac{x}{p-1}\right)\right|\,\mathrm{d} \theta}& =O\left(\frac{p-1}{d(x)}\right)\left(\int_{I_3}{\frac{1}{\|\theta\|_\mathbb{T}}\,\mathrm{d}\theta} +\int_{I_3}{\frac{1}{\left\|\theta-\frac{x}{p-1}\right\|_\mathbb{T}}\,\mathrm{d}\theta} \right)\\ \label{eqn.I3e} &= O\left(\frac{p}{d(x)}\log p\right)=O\left(\frac{p}{d(x)}\log N\right).
\end{align}
Inserting the bounds from (\ref{eqn.I1e}), its analogue for $I_2$ in place of $I_1$, and (\ref{eqn.I3e}) into (\ref{eqn.Su}) gives
\begin{align*}
S& =O(N\log N)^{2k-2} \cdot N \cdot \sum_{x\in \mathbb{Z}_{p-1}^*}{O\left(\frac{p}{d(x)}\log N\right)}.
\end{align*}
Summing over $x \in \mathbb{Z}_{p-1}^*$, $d(x)$ takes each of the values $1,2,\dots,\frac{p-1}{2}$ twice, so the sum is $O(p\log N\log p)=O(p^{-1}N\log^2 N)$ which gives the claim.
\end{proof}
Now we turn to the $L_{2k}$-norm of $f_A$. For $j \in \mathbb{Z}_N^k$ write
\begin{equation*}
g(j) :=\begin{cases} \prod_{i=1}^k {\frac{\widehat{1_A }\left(\frac{j_i}{N}\right)}{\sqrt p}} & \text{ for }j \in (\mathbb{Z}_N^*)^k\\ 0 & \text{ otherwise}.\end{cases}
\end{equation*}
Given $j \in \mathbb{Z}_N^k$ there are unique $a \in \mathbb{Z}_{p-1}^k$ and $b \in \mathbb{Z}_p^k$ such that $j_i \equiv pa_i - (p-1)b_i \bmod N$ for all $i \in [k]$ (and similarly for $j'$, $a'$, and $b'$). This is the same instance of the Chinese Remainder Theorem implicitly mentioned after the definition (\ref{eqn.ruzsa}). Then
\begin{equation}\label{eqn.expg}
g(j)=p^{-\frac k2}\prod_{i=1}^k{G(\psi_1,\chi^{a_i})\chi^{-a_i}(b_i)} \text{ for all }j \in \mathbb{Z}_N^k.
\end{equation}
To see this, (\ref{eqn.above}) deals with $j \in \mathbb{Z}_N^*$, and for $j \in \mathbb{Z}_N^k \setminus (\mathbb{Z}_N^*)^k$ we use the fact that $\chi^{-0}(0)=0$ by definition. Note that $\chi$ in (\ref{eqn.expg}) is the particular choice of $\chi$ just before (\ref{eqn.ruzsa}), \emph{i.e.\ }$\chi(s)=\exp(-2\pi i/(p-1))$ where $s$ is a primitive root modulo $p$, fixed for our definition of $A$ just before (\ref{eqn.ruzsa}).

The function $g$ arises from (\ref{eqn.flarge}) which gives
\begin{equation}\label{eqn.fA}
\int_0^1|f_A(\theta)|^{2k}\,\mathrm{d}\theta
=
\frac1{N^{2k}}
\sum_{j,j' \in \mathbb{Z}_N^k}{g(j)\overline{g(j')}H(j,j')}.
\end{equation}

Recalling the definition of $H$ in (\ref{Hjj}), the change of variables $\theta \mapsto \theta + \frac{h}{N}$ shows that
\begin{equation}\label{eqn.hshift}
H(j+h\textbf 1, j'+h\textbf 1)=H(j,j') \text{ for all }h \in \mathbb{Z}_N, j,j' \in \mathbb{Z}_N^k,
\end{equation}
where $\textbf 1 = (1,\dots,1)$. Let 
\begin{equation*}
\mathcal{S}:=\{(j,j') \in \mathbb{Z}_N^k\times \mathbb{Z}_N^k: \exists \pi \in S_k \text{ s.t.\ }b=\pi^*(b')\}.
\end{equation*}
Note that $\mathcal{S}$ is closed under $(j,j')\mapsto (j+h\textbf 1,j'+h\textbf 1)$, and hence so is $\mathcal{S}^c=\mathbb{Z}_N^k \times \mathbb{Z}_N^k \setminus \mathcal{S}$.
\begin{claim*}
Either there is $\pi \in S_k$ such that $b=\pi^*(b')$; or else for at least a proportion $1-O(kp^{-1})$ of the $u\in \mathbb{Z}_{p-1}$ we have
\begin{equation}\label{eqn.weilaverage}
\left| \sum_{t\in\mathbb{Z}_p} \prod_{i=1}^k \chi^{-a_i-u}(b_i+t) \overline{\prod_{i=1}^{k}\chi^{-a_i'-u}(b_i'+t)}\right|=O(k\sqrt{p}).
\end{equation}
\end{claim*}
\begin{proof}
To see this we will apply the Weil bound (Theorem \ref{thm.weilbound}) using that $\chi$ has order $p-1$. First, let $\xi_1,\dots,\xi_d$ be the distinct values of $-b_1,\dots,-b_k,-b_1',\dots,-b_k'$. The summand in (\ref{eqn.weilaverage}) then has the form $\chi(R(t))$ where $R(t)$ is a product of $d \leq 2k$ factors $(t-\xi_i)^{e_i}$ and
\begin{equation*}
e_i(u):=\sum_{l \in [k]:-b_l'=\xi_i}{(a_l'+u)}-\sum_{l \in [k]:-b_l=\xi_i}{(a_l+u)} .
\end{equation*}
Suppose that there is no $\pi \in S_k$ such that $b=\pi^*(b')$. Then there is some $i \in [d]$ such that $m_i:=|\{l \in [k]: -b_l'=\xi_i\}| - |\{l \in [k]: -b_l=\xi_i\}|$ has $m_i \neq 0$. If $e_i(u) \not \equiv 0 \bmod{p-1}$ then by Theorem \ref{thm.weilbound} we have (\ref{eqn.weilaverage}) for that particular $u$, so set
\[E:=\{u \in \mathbb{Z}_{p-1}: e_i(u) \equiv 0 \bmod{p-1}\}.\]
If $|E|>2k$ then there are $u,u' \in E$ with $u \neq u'$ and $\left\|\frac{u-u'}{p-1}\right\|_\mathbb{T}<\frac 1{2k}$. Since $|m_i| \leq k$, we have
\begin{equation*}
0<\left\|m_i\frac{u-u'}{p-1}\right\|_\mathbb{T}=\left\|\frac{e_i(u)-e_i(u')}{p-1}\right\|_\mathbb{T}=0
\end{equation*}
which is a contradiction. Hence $|E| \leq 2k$, and the result is proved.\end{proof}

Suppose that $(j,j') \in \mathcal{S}^c$. Writing $h=pu-(p-1)t$ for $h\in \mathbb{Z}_N$, $u \in \mathbb{Z}_{p-1}$ and $t \in \mathbb{Z}_p$ we have
\begin{align*}
\sum_{ h \in \mathbb{Z}_N}{g(j+h\textbf1)\overline{g(j'+h\textbf 1)}} & = p^{-k}\sum_{u \in \mathbb{Z}_{p-1}}{\prod_{i=1}^k{G(\psi_1,\chi^{a_i+u})\overline{G(\psi_1,\chi^{a_i'+u})}}}\\ & \qquad \qquad \qquad \times  \sum_{t \in \mathbb{Z}_p}{\prod_{i=1}^k\chi^{-a_i-u}(b_i+t)\overline{\chi^{-a_i'-u}(b_i'+t)}}.
\end{align*}
Inserting the bound $|G(\psi_1,\chi^{a_i+u})| \leq \sqrt{p}$ along with the claim (where we are in the second case since $(j,j') \in \mathcal{S}^c$) gives
\begin{equation*}
\sum_{ h \in \mathbb{Z}_N}{g(j+h\textbf1)\overline{g(j'+h\textbf 1)}}  \leq p^{-k}\cdot (p-1)\cdot \sqrt{p}^{2k}\cdot O_k(\sqrt{p}) + p^{-k}\cdot O_k(1) \cdot \sqrt{p}^{2k}\cdot p =O_k(p^{\frac{3}{2}}).
\end{equation*}
Using this in (\ref{eqn.fA}), the fact that $\mathcal{S}^c$ is closed under $(j,j')\mapsto (j+h\textbf 1,j'+h\textbf 1)$ and $H(j,j')$ is invariant under $(j,j')\mapsto (j+h\textbf 1,j'+h\textbf 1)$, we get
\begin{align}
\nonumber \int_0^1|f_A(\theta)|^{2k}\,\mathrm{d}\theta
&=
\frac1{N^{2k}}\left(\sum_{(j,j') \in \mathcal{S}}{g(j)\overline{g(j')}H(j,j')}\right.\\ \nonumber & \qquad\qquad \qquad \left. + \sum_{(j,j') \in \mathcal{S}^c}{H(j,j')\frac{1}{N}\sum_{h \in \mathbb{Z}_N}{g(j+h \textbf1)\overline{g(j'+h\textbf 1)}}}\right)\\ \nonumber
&=
\frac1{N^{2k}}\left(\sum_{(j,j') \in \mathcal{S}}{g(j)\overline{g(j')}H(j,j')} + \sum_{j,j' \in \mathbb{Z}_N^k}{\frac{1}{N}O_k(p^{\frac32})|H(j,j')|}\right)\\ & = \frac1{N^{2k}}\sum_{(j,j') \in \mathcal{S}}{g(j)\overline{g(j')}H(j,j')} + O_k(p^{-\frac12}\log^{2k}N).
\end{align}
The passage to the last line uses $N=p(p-1)$ and (\ref{eqn.dirichletsum2}) $2k$ times, once for each of the variables $j_1$, $\dots$, $j_k$, $j_1'$, $\dots$, $j_k'$. (This is similar to the proof of (\ref{Hjj}).) Finally, if $j,j' \in \mathbb{Z}_N^k$ and $\pi \in S_k$ is such that $b=\pi^*(b')$ but $j\neq \pi^*(j')$, then $b=\pi^*(b')$ implies $j_i \equiv j_{\pi(i)}' \bmod p$ for all $i \in [k]$, and hence there is some $i \in [k]$ such that $j_i \neq j_{\pi(i)}'$ and $p \mid j_i - j_{\pi(i)}'$. By (\ref{Hjj2}) (and the fact that $|g(j)|,|g(j')| \leq 1$) we conclude that
\begin{align*}
\left|\sum_{\substack{j,j' \in \mathbb{Z}_N^k\\\exists \pi \in S_k \text{ s.t. }b=\pi^*(b')\\ \text{ and }j \neq \pi^*(j')}}{g(j)\overline{g(j')}H(j,j')}\right| & \leq \sum_{\pi \in S_k}{\sum_{i \in [k]}{\sum_{\substack{j,j' \in \mathbb{Z}_N^k \\ j_i \neq j'_{\pi(i)}\\ p \mid j_i - j'_{\pi(i)}}}{|g(j)||g(j')||H(j,j')|}}}\\
 &= O_k(p^{-1}N^{2k}\log^{2k}N).
\end{align*}
Hence
\begin{equation*}
\int_0^1|f_A(\theta)|^{2k}\,\mathrm{d}\theta=\frac{1}{N^{2k}}\sum_{\substack{j,j' \in \mathbb{Z}_N^k\\\exists \pi \in S_k \text{ s.t. }j= \pi^*(j')}}{g(j)\overline{g(j')}H(j,j')}+ O_k(p^{-\frac12}\log^{2k}N).
\end{equation*}
Finally, for $j,j' \in \mathbb{Z}_N^k$ with $j=\pi^*(j')$ we have $g(j)\overline{g(j')}=1$ if $p \nmid j_r$ and $p-1 \nmid j_r$ for all $r \in [k]$, and otherwise $|g(j)\overline{g(j')}| \leq 1$. The result then follows from (\ref{Hjj}) and (\ref{eqn.Fmomentexpansion}).

Write $\Omega$ for the common sample space for the $Y_m$s.  The random variables $Y_m\operatorname{sinc} (\pi(s-m))$ are independent mean zero random variables with summable variance and so
\begin{equation}\label{eqn.asc}
S_M(\omega,s):=\sum_{|m| \leq M}{Y_m\operatorname{sinc} (\pi(s-m))} \rightarrow S(\omega,s):=\sum_{m \in \mathbb{Z}}{Y_m\operatorname{sinc} (\pi(s-m))}
\end{equation}
for every $s \in [0,1)$ and almost every $\omega \in \Omega$.  It follows by Fubini's theorem that $S_M \rightarrow S$ a.e.\ $(\omega,s) \in \Omega \times [0,1)$.

Write $t=N\theta$ to see that
\begin{equation}\label{eqn.firstcalc}
\int_0^1{\mathop{\mathbb{E}}{|F(\theta)|}\,\mathrm{d}\theta}= \frac{1}{N}\int_{0}^N{ \mathop{\mathbb{E}}{\left| \sum_{j=1}^{N-1}{X_j\frac{\widehat{1_{[N]}}((t-j)/N)}{N}}\right|}\,\mathrm{d} t}.
\end{equation}
Now, for any $\theta \in (-1,1)$ we have
\begin{equation}\label{eqn.dk}
\widehat{1_{[N]}}(\theta)=\exp(-\pi i (N+1)\theta)\frac{N\operatorname{sinc} (\pi N \theta)}{\operatorname{sinc} (\pi \theta)}.
\end{equation}
Since $X_1$, \dots, $X_{N-1}$ are independent uniform $S^1$-valued random variables, $\omega_1X_1$, \dots, $\omega_{N-1}X_{N-1}$ are independent uniform $S^1$-valued random variables for any $\omega_1,\dots,\omega_{N-1} \in S^1$. Hence from (\ref{eqn.dk}), putting $\omega_j:=\exp(-\pi i \frac{N+1}{N}(t-j))$, we have
\begin{equation*}
 \mathop{\mathbb{E}}{\left| \sum_{j=1}^{N-1}{X_j\frac{\widehat{1_{[N]}}((t-j)/N)}{N}}\right|}= \mathop{\mathbb{E}}{\left| \sum_{j=1}^{N-1}{X_j \frac{\operatorname{sinc}(\pi (t-j))}{\operatorname{sinc}(\pi(t-j)/N)}}\right|}.
\end{equation*}
Inserting this in (\ref{eqn.firstcalc}) and writing $t=k+s$ for $s \in [0,1)$ and $k \in \mathbb{Z}$ we get
\begin{align}
\int_0^1{\mathop{\mathbb{E}}{|F(\theta)|}\,\mathrm{d}\theta}
&= \frac{1}{N}\sum_{k=0}^{N-1}{\int_0^1{ \mathop{\mathbb{E}}{\left| \sum_{j=1}^{N-1}{X_j \frac{\operatorname{sinc}(\pi(k+s-j))}{\operatorname{sinc}(\pi(k+s-j)/N)}}\right|}\,\mathrm{d} s}} \label{eqn.calc}.
\end{align}
To estimate this it is useful to have an auxiliary estimate: for $S \subset [N-1]$ we have
\begin{align}\nonumber 
\left(\mathop{\mathbb{E}}{\left| \sum_{ j \in S}{X_j \frac{\operatorname{sinc}(\pi (k+s-j))}{\operatorname{sinc}(\pi(k+s-j)/N)}}\right|}\right)^2 & \leq \mathop{\mathbb{E}}{\left| \sum_{ j \in S}{X_j \frac{\operatorname{sinc}(\pi (k+s-j))}{\operatorname{sinc}(\pi(k+s-j)/N)}}\right|^2}\\ \nonumber &= \sum_{j \in S}{\frac{\operatorname{sinc}^2( \pi (k+s-j))}{\operatorname{sinc}^2 (\pi (k+s-j)/N)}}\\& \leq \frac{1}{N^2}\sum_{ j \in S}{\frac{1}{\sin^2(\pi (k+s-j)/N)}}\nonumber \\

\end{align}
Fix $M$, to be sent to infinity later. We delete those terms on the right in (\ref{eqn.calc}) with $k \leq M$ and $k \geq N-M$. By (\ref{eqn.l2dump}) with $S=[N-1]$ these contribute $O(M/N)$ to the right of (\ref{eqn.calc}). For each $M<k<N-M$ we then use the triangle inequality to delete those $j$s in the inner sum with $|k-j|>M$. By (\ref{eqn.l2dump}) with $S:=\{j \in [N-1]: |k-j|>M\}$ these contribute $O(M^{-1/2})$ to the right of (\ref{eqn.calc}). Combining all this we get
\begin{align*}

\end{align*}

\begin{equation*}

\end{equation*}
In particular it is independent of $k$ and so we have
\begin{align}\label{eqn.mainest}
\int_0^1{\mathop{\mathbb{E}}{|F(\theta)|}\,\mathrm{d}\theta} &= \left(1-\frac{2M+1}{N}\right)\int_0^1{ \mathop{\mathbb{E}}{\left| \sum_{m=-M}^{M}{Y_m\frac{\operatorname{sinc}(\pi (s-m))}{\operatorname{sinc}(\pi(s-m)/N)}}\right|}\,\mathrm{d} s} \\ &\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad + O(MN^{-1}+M^{-1/2}). \nonumber
\end{align}

\begin{equation*}
\sum_{m=-M}^{M}{Y_m(\omega)\frac{\operatorname{sinc}(\pi (s-m))}{\operatorname{sinc}(\pi(s-m)/N)}} \rightarrow S_M(\omega,s)
\end{equation*}

Once $N>2M$ the left hand side is also bounded independently of $N$, $\omega$, and $s$, and so by bounded convergence we have
\begin{equation*}
\int_0^1{ \mathop{\mathbb{E}}{\left| \sum_{m=-M}^{M}{Y_m\frac{\operatorname{sinc}(\pi (s-m))}{\operatorname{sinc}(\pi(s-m)/N)}}\right|}\,\mathrm{d} s} \rightarrow \int_0^1{ \mathop{\mathbb{E}}{\left| S_M(\omega,s)\right|}\,\mathrm{d} s}
\end{equation*}

\begin{equation*}

Finally, from (\ref{opening}),

\label{thm.zyg}
There is $C>1$ and for each $p \in (0,\infty)$ there is $C_p>1$ such that if $N \in \mathbb{N}^*$, and $P$ is a degree $N-1$ polynomial then
\begin{equation}\label{thm.pt1}
\left(\frac{1}{N}\sum_{r=0}^{N-1}{|P(\exp(2\pi i r/N))|^p}\right)^{1/p} \leq C\left(\int_0^1{|P(\exp(2\pi i \theta))|^p\,\mathrm{d}\theta}\right)^{1/p} \text{ for all }1 \leq p \leq \infty;
\end{equation}
and
\begin{equation}\label{thm.pt2}
\left(\int_0^1{|P(\exp(2\pi i \theta))|^p\,\mathrm{d}\theta}\right)^{1/p} \leq C_p^{1/p}\left(\frac{1}{N}\sum_{r=0}^{N-1}{|P(\exp(2\pi i r/N))|^p}\right)^{1/p}  \text{ for all }1 < p < \infty.
\end{equation}
