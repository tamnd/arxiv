<!-- theorem-1: body_md -->
\label{thm:intro-local}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be a small $R$-linear extriangulated category and $\mathcal X=\operatorname{add}\mathcal X$ be an $n$-rigid subcategory of $\mathcal C$.
\begin{enumerate}[{\rm(1)}]
\item If $\mathcal X$ is strongly covariantly finite, then
\[
 \mathcal X
 =
 \operatorname{Tr}^{+}_{n}(\mathcal X)
 \cap
 \bigcap_{i=1}^{n}\mathcal X^{\perp_i}.
\]
\item If $\mathcal X$ is strongly contravariantly finite, then
\[
 \mathcal X
 =
 \operatorname{Tr}^{-}_{n}(\mathcal X)
 \cap
 \bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X.
\]
\end{enumerate}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:intro-main}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be a small $R$-linear extriangulated category and
 $\mathcal X=\operatorname{add}\mathcal X$ be a strongly functorially finite $n$-rigid
subcategory of $\mathcal C$. Then the following conditions are equivalent:
\begin{enumerate}[{\rm(1)}]
\item
\[
\mathcal X=
\bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X
=
\bigcap_{i=1}^{n}\mathcal X^{\perp_i}.
\]
\item $\Sigma_{\mathcal X}^{n}\cong\mathbf 0$ as endofunctors of $\mathcal C/[\mathcal X]$.
\item $\Omega_{\mathcal X}^{n}\cong\mathbf 0$ as endofunctors of $\mathcal C/[\mathcal X]$.
\end{enumerate}
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor:intro-triangulated}
Let $\mathcal T$ be a triangulated category and $\mathcal X=\operatorname{add}\mathcal X$ be a
functorially finite $n$-rigid subcategory of $\mathcal T$. Then the canonical pretriangulated
quotient $\mathcal T/[\mathcal X]$ is $n$-truncated if and only if $\mathcal X$ is $(n+1)$ cluster
tilting.

<!-- theorem-3: body_md -->
\label{thm:intro-sharp}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be a small $R$-linear extriangulated category. Assume that $\mathcal C$ is nonsplit and that $\mathcal X$ is strongly functorially finite and two-sided maximally $n$-orthogonal. Then the
nilpotency index of $\Sigma_{\mathcal X}$ is exactly $n$, and
$\mathbb E^{n+1}(\mathcal X,\mathcal X)\neq0$.
<!-- end theorem-3 -->
<!-- definition-1: body_md -->
\label{def:nrigid}
A subcategory $\mathcal X$ is \emph{$n$-rigid} if
\[
        \mathbb E^i(\mathcal X,\mathcal X)=0,
        \hspace{2.5mm}1\leq i\leq n.
\]
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:maxorth}
Let $n\geq1$. A subcategory $\mathcal X=\operatorname{add}\mathcal X$ is said to satisfy the \emph{two-sided maximal $n$-orthogonality condition} (with respect to the positive extensions $\mathbb E^i$) if
\[
\mathcal X
=
\bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X
=
\bigcap_{i=1}^{n}\mathcal X^{\perp_i}.
\]
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:cluster-tilting}
A subcategory $\mathcal X=\operatorname{add}\mathcal X$ is \emph{cluster tilting} if it is strongly functorially finite and
\[
 \mathcal X=\{A\in\mathcal C\mid\mathbb E(\mathcal X,A)=0\}
 =\{A\in\mathcal C\mid\mathbb E(A,\mathcal X)=0\}.
\]
This is the convention of Zhou--Zhu \cite[Definition 2.10]{ZZct}.
<!-- end definition-3 -->
<!-- theorem-4: body_md -->
\label{thm:known-onesided}
Let $\mathcal X$ be an additive subcategory of an extriangulated category $\mathcal C$.
\begin{enumerate}[{\rm(1)}]
\item If $\mathcal X$ is strongly covariantly finite, then $\mathcal C/[\mathcal X]$ carries a canonical right triangulated structure. Its suspension, denoted $\Sigma_{\mathcal X}$, is induced by left $\mathcal X$-approximation $\mathbb E$-triangles.
\item Dually, if $\mathcal X$ is strongly contravariantly finite, then $\mathcal C/[\mathcal X]$ carries a canonical left triangulated structure. Its loop, denoted $\Omega_{\mathcal X}$, is induced by right $\mathcal X$-approximation $\mathbb E$-triangles.
\end{enumerate}
<!-- end theorem-4 -->
<!-- head-1: body_md -->
Part \textup{(1)} is the $n=1$ specialization of the construction of He--He--Zhou \cite[Lemma 3.1]{HHZright}. Part \textup{(2)} follows by applying the same result to the opposite extriangulated category. In particular, the canonical suspension or loop is available under the corresponding one-sided strong finiteness assumption.  \qed
<!-- end head-1 -->
<!-- theorem-5: body_md -->
\label{thm:known-pretri}
If $\mathcal X$ is strongly functorially finite, then the two one-sided structures of Theorem~\ref{thm:known-onesided} combine to a canonical pretriangulated structure on $\mathcal C/[\mathcal X]$, and $(\Sigma_{\mathcal X},\Omega_{\mathcal X})$ is an adjoint pair.
<!-- end theorem-5 -->
<!-- head-2: body_md -->
This is the construction of Zhou--Zhu \cite[Lemma 3.21]{ZZtri}. See also Zheng--Cao--Wei \cite{ZCW} for a subfactor version in extriangulated categories and Beligiannis--Reiten \cite{BR} for the pretriangulated formalism. We use these known quotient constructions as black boxes.  \qed
<!-- end head-2 -->
<!-- lemma-1: body_md -->
\label{lem:stable-right}
Let $(\mathcal R,\Sigma,\Delta)$ be a right triangulated category. If $\Sigma$ is an autoequivalence, then $(\mathcal R,\Sigma,\Delta)$ is a triangulated category.
<!-- end lemma-1 -->
<!-- head-3: body_md -->
This is part of the standard right triangulated formalism recalled by Lin--Wang \cite[Section 2.1.1]{LW}. We use it only for the canonical right triangulated structure on an ideal quotient.  \qed
<!-- end head-3 -->
<!-- theorem-6: body_md -->
\label{thm:known-abelian}
Let $\mathcal C$ be an extriangulated category with enough projectives and enough injectives, and let $\mathcal X$ be a cluster tilting subcategory. Then $\mathcal C/[\mathcal X]$ is equivalent to $\mathrm{mod}\,(\mathcal X/[\mathcal P])$ and in particular is abelian.
<!-- end theorem-6 -->
<!-- head-4: body_md -->
This is due to Zhou--Zhu \cite[Corollary 3.5]{ZZct}. The triangulated specialization of Zhou--Zhu \cite[Corollary 3.6]{ZZct} recovers the theorem of Koenig--Zhu \cite[Theorem 3.3]{KZ}. We do not repeat these known abelian quotient results.  \qed

<!-- theorem-7: body_md -->
\label{thm:known-triangulated}
Assume the Auslander--Reiten duality hypotheses just stated, and let $\mathcal X$ be strongly functorially finite. Then the following are equivalent:
\begin{enumerate}[{\rm(1)}]
\item $(\mathcal C,\mathcal C)$ is an $\mathcal X$-mutation pair.
\item the canonical pretriangulated category $\mathcal C/[\mathcal X]$ is triangulated.
\item $\tau\underline{\mathcal X}=\overline{\mathcal X}$.
\item the canonical suspension $\Sigma_{\mathcal X}$ is an autoequivalence of $\mathcal C/[\mathcal X]$.
\end{enumerate}
If $\mathcal C$ is triangulated, this specializes to
\[
        \mathcal C/[\mathcal X]\text{ triangulated}
        \quad\Longleftrightarrow\quad
        \tau\mathcal X=\mathcal X.
\]
<!-- end theorem-7 -->
<!-- head-5: body_md -->
The equivalence of conditions \textup{(1)}--\textup{(3)} is due to Zhou--Zhu \cite[Theorem 4.3]{ZZtri}. The implication \textup{(2)}$\Rightarrow$\textup{(4)} is immediate, while \textup{(4)}$\Rightarrow$\textup{(2)} follows from Lemma~\ref{lem:stable-right} applied to the canonical right triangulated structure. The triangulated specialization of Zhou--Zhu \cite[Corollary 4.4]{ZZtri} recovers J\o rgensen \cite[Theorem 3.3]{Jor}. We use these facts only as a cited endpoint.  \qed

<!-- lemma-2: body_md -->
\label{lem:objectwise-zero}
Let $F\colon\mathcal A\to\mathcal B$ be a functor and suppose that $\mathcal B$ has a zero object. Then the following are equivalent:
\begin{enumerate}[{\rm(1)}]
\item $F(A)$ is a zero object for every $A\in\mathcal A$.
\item $F\cong\mathbf 0$ naturally.
\end{enumerate}
If a distinguished zero object has been fixed and $F(A)=0$ literally for every $A$, then $F$ is literally the zero functor.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
For every $A$, the unique morphism $F(A)\to0$ is an isomorphism because both objects are zero. These morphisms are automatically natural, again by uniqueness. Hence $F\cong\mathbf 0$. The final assertion follows because the only morphism $0\to0$ is the zero morphism.
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
\label{lem:adjoint-zero}
Let $F$ and $G$ be additive endofunctors such that $F$ is left adjoint to $G$.
For every $n\geq 1$,
\[
F^n\cong\mathbf 0
\quad\Longleftrightarrow\quad
G^n\cong\mathbf 0.
\]
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For every $n\geq 1$, the functor $F^n$ is left adjoint to $G^n$.
Suppose that $F^n\cong\mathbf 0$. Then, for all objects $A$ and $B$,
\[
\operatorname{Hom}(A,G^nB)\cong\operatorname{Hom}(F^nA,B)=0.
\]
Taking $A=G^nB$, we obtain $\operatorname{id}_{G^nB}=0$. Hence $G^nB$ is a zero
object for every $B$. Lemma~\ref{lem:objectwise-zero} then gives
$G^n\cong\mathbf 0$. The converse follows by the same argument.
<!-- end proof-2 -->
<!-- corollary-2: body_md -->
\label{cor:sigma-omega}
Assume that $\mathcal X$ is strongly functorially finite. For the canonical pretriangulated quotient,
\[
        \Sigma_{\mathcal X}^n\cong\mathbf 0
        \quad\Longleftrightarrow\quad
        \Omega_{\mathcal X}^n\cong\mathbf 0.
\]
<!-- end corollary-2 -->
<!-- proof-3: body_md -->
By Theorem~\ref{thm:known-pretri}, the relevant functors form an adjoint pair.
The result now follows immediately from Lemma~\ref{lem:adjoint-zero}.
<!-- end proof-3 -->
<!-- lemma-4: body_md -->
\label{lem:zero-object-quotient}
Assume $\mathcal X=\operatorname{add}\mathcal X$. Then $\pi_{\mathcal X} A$ is a zero object in $\mathcal C/[\mathcal X]$ if and only if $A\in\mathcal X$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
If $A\in\mathcal X$, then $\operatorname{id}_A$ factors through $\mathcal X$, so $\pi_{\mathcal X} A$ is zero. Conversely, if $\pi_{\mathcal X} A$ is zero, then $\operatorname{id}_A$ factors as $A\to X\to A$ with $X\in\mathcal X$. Hence $A$ is a direct summand of $X$ and belongs to $\mathcal X$.
<!-- end proof-4 -->
<!-- lemma-5: body_md -->
\label{lem:dimension-shifting}
Let $\mathcal C$ be a small $R$-linear extriangulated category. Assume that $\mathcal X$ is $n$-rigid, and let
\[
 A_j\xrightarrow{x_j}X_j\longrightarrow A_{j+1}\dashrightarrow,
 \hspace{2.5mm} X_j\in\mathcal X,
\]
be one of the left $\mathcal X$-approximation $\mathbb E$-triangles in \eqref{eq:tower}. Then, for every $X\in\mathcal X$, the following hold.
\begin{enumerate}[{\rm(1)}]
\item $\mathbb E(A_{j+1},X)=0$.
\item For $2\le r\le n$ there is a natural isomorphism
\[
        \mathbb E^r(A_{j+1},X)\cong \mathbb E^{r-1}(A_j,X).
\]
\item For $1\le r\le n-1$ there is a natural isomorphism
\[
        \mathbb E^r(X,A_{j+1})\cong \mathbb E^{r+1}(X,A_j).
\]
\end{enumerate}
Consequently,
\[
 \mathbb E^r(A_{j+1},X)=0
  \hspace{2mm}(0\le j\le n-1,\ 1\le r\le \min\{j+1,n\}),
\]
and, if $A_0\in\bigcap_{i=1}^n\mathcal X^{\perp_i}$, then
\[
        \mathbb E(X,A_j)=0 \hspace{2mm}(0\le j\le n-1).
\]
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Fix $X\in\mathcal X$. Applying the long exact sequence of Gorsky--Nakaoka--Palu \cite[Theorem 3.6]{GNP} to the $\mathbb E$-triangle
\[
        A_j\xrightarrow{x_j}X_j\longrightarrow A_{j+1}\dashrightarrow
\]
gives the exact sequence
\[
 \mathcal C(X_j,X)\longrightarrow\mathcal C(A_j,X)
 \longrightarrow\mathbb E(A_{j+1},X)\longrightarrow\mathbb E(X_j,X).
\]
The first map is surjective because $x_j$ is a left $\mathcal X$-approximation, and the last term vanishes by rigidity. This proves (1).

For $2\le r\le n$, the same long exact sequence contains
\[
 \mathbb E^{r-1}(X_j,X)\longrightarrow \mathbb E^{r-1}(A_j,X)
 \longrightarrow \mathbb E^r(A_{j+1},X)
 \longrightarrow \mathbb E^r(X_j,X).
\]
Both outer terms vanish by $n$-rigidity, proving (2).

For $1\le r\le\min\{j+1,n\}$, iterate (2) through the preceding $\mathbb E$-triangles. The result terminates at the degree one vanishing in (1):
\[
 \mathbb E^r(A_{j+1},X)\cong\mathbb E(A_{j-r+2},X)=0.
\]

To prove \textup{(3)}, the long exact sequence contains, for $1\le r\le n-1$,
\[
 \mathbb E^r(X,X_j)\longrightarrow \mathbb E^r(X,A_{j+1})
 \longrightarrow \mathbb E^{r+1}(X,A_j)
 \longrightarrow \mathbb E^{r+1}(X,X_j).
\]
Again the outer terms vanish by $n$-rigidity, which proves (3).

If $A_0\in\bigcap_{i=1}^n\mathcal X^{\perp_i}$, then for $1\le j\le n-1$ repeated use of (3) gives
\[
        \mathbb E(X,A_j)\cong\mathbb E^{j+1}(X,A_0)=0,
\]
while the case $j=0$ is part of the assumption. Notice that the largest degree used here is $j+1\le n$ and vanishing in degree $n+1$ is not required.
<!-- end proof-5 -->
<!-- remark-1: body_md -->
The small $R$-linear hypothesis is a convenient standard assumption ensuring that this calculus is available from Gorsky--Nakaoka--Palu \cite[Theorem 3.6]{GNP}.
<!-- end remark-1 -->
<!-- definition-4: body_md -->
\label{def:local-trunc}
For $m\geq0$, assume first that $\mathcal X$ is strongly covariantly finite and define
\[
 \operatorname{Tr}^{+}_{m}(\mathcal X)
 :=\{A\in\mathcal C\mid \Sigma_{\mathcal X}^{m}(\pi_{\mathcal X} A)\cong0\}.
\]
Dually, when $\mathcal X$ is strongly contravariantly finite, define
\[
 \operatorname{Tr}^{-}_{m}(\mathcal X)
 :=\{A\in\mathcal C\mid \Omega_{\mathcal X}^{m}(\pi_{\mathcal X} A)\cong0\}.
\]
Here the zeroth power is the identity.  If $\mathcal X=\operatorname{add}\mathcal X$, then Lemma~\ref{lem:zero-object-quotient} gives
$\operatorname{Tr}^{+}_{0}(\mathcal X)=\operatorname{Tr}^{-}_{0}(\mathcal X)=\mathcal X$.
Since the canonical suspension and loop are additive, these loci are closed under finite direct sums and direct summands, and
\[
 \operatorname{Tr}^{\pm}_{m}(\mathcal X)\subseteq
 \operatorname{Tr}^{\pm}_{m+1}(\mathcal X)
 \hspace{2.5mm}(m\geq0).
\]
<!-- end definition-4 -->
<!-- proposition-1: body_md -->
\label{prop:left-to-sigma}
Let $\mathcal X=\operatorname{add}\mathcal X$ be strongly covariantly finite and $n$-rigid. If
\[
        \mathcal X=\bigcap_{i=1}^n{}^{\perp_i}\mathcal X,
\]
then
\[
        \Sigma_{\mathcal X}^n\cong\mathbf 0.
\]
<!-- end proposition-1 -->
<!-- proof-6: body_md -->
Let $A\in\mathcal C$, set $A_0=A$, and consider the $\mathbb E$-triangle \eqref{eq:tower}. By Lemma~\ref{lem:dimension-shifting}, for every $X\in\mathcal X$,
\[
        \mathbb E^r(A_n,X)=0,
        \qquad 1\le r\le n.
\]
Hence
\[
        A_n\in\bigcap_{r=1}^n{}^{\perp_r}\mathcal X=\mathcal X.
\]
Equation \eqref{eq:sigma-tower} therefore gives
\[
        \Sigma_{\mathcal X}^n(\pi_{\mathcal X} A)\cong\pi_{\mathcal X}(A_n)\cong0
\]
for every $A\in\mathcal C$. By Lemma~\ref{lem:objectwise-zero}, $\Sigma_{\mathcal X}^n\cong\mathbf 0$.
<!-- end proof-6 -->
<!-- theorem-8: body_md -->
\label{thm:local-reconstruction}
Let $\mathcal X=\operatorname{add}\mathcal X$ be strongly covariantly finite and $n$-rigid. Then
\[
 \mathcal X
 =
 \operatorname{Tr}^{+}_{n}(\mathcal X)
 \cap
 \bigcap_{i=1}^{n}\mathcal X^{\perp_i}.
\]
Equivalently, for every $A\in\bigcap_{i=1}^{n}\mathcal X^{\perp_i}$,
\[
 \Sigma_{\mathcal X}^n(\pi_{\mathcal X} A)\cong0
 \quad\Longleftrightarrow\quad
 A\in\mathcal X.
\]
<!-- end theorem-8 -->
<!-- proof-7: body_md -->
The inclusion from left to right follows from $n$-rigidity and the fact that every object of $\mathcal X$ is zero in $\mathcal C/[\mathcal X]$.

For the reverse inclusion, let
\[
 A=A_0\in\operatorname{Tr}^{+}_{n}(\mathcal X)
       \cap\bigcap_{i=1}^{n}\mathcal X^{\perp_i}
\]
and consider the $\mathbb E$-triangle \eqref{eq:tower}.  By \eqref{eq:sigma-tower}, the local truncation condition gives
\[
 \pi_{\mathcal X}(A_n)\cong\Sigma_{\mathcal X}^n(\pi_{\mathcal X} A)\cong0.
\]
Hence $A_n\in\mathcal X$ by Lemma~\ref{lem:zero-object-quotient}.  On the other hand, Lemma~\ref{lem:dimension-shifting} gives
\[
 \mathbb E(X,A_j)=0
  \hspace{2.5mm}
 (X\in\mathcal X,\ 0\leq j\leq n-1).
\]
We show by descending induction that $A_j\in\mathcal X$ for every $j$.  If $A_{j+1}\in\mathcal X$, then the extension class of
\[
 A_j\longrightarrow X_j\longrightarrow A_{j+1}\dashrightarrow
\]
lies in $\mathbb E(A_{j+1},A_j)=0$.  The $\mathbb E$-triangle therefore splits, so $A_j$ is a direct summand of $X_j\in\mathcal X$ and hence belongs to $\mathcal X$.  Starting from $A_n\in\mathcal X$ yields $A=A_0\in\mathcal X$.
<!-- end proof-7 -->
<!-- corollary-3: body_md -->
\label{prop:sigma-to-right}
Let $\mathcal X=\operatorname{add}\mathcal X$ be strongly covariantly finite and $n$-rigid. If
\[
 \Sigma_{\mathcal X}^n\cong\mathbf 0,
\]
then
\[
 \mathcal X=\bigcap_{i=1}^n\mathcal X^{\perp_i}.
\]
<!-- end corollary-3 -->
<!-- proof-8: body_md -->
Global truncation means that
\[
\operatorname{Tr}^{+}_{n}(\mathcal X)=\mathcal C.
\]
Hence every object of $\mathcal C$ satisfies the local truncation condition appearing
in Theorem~\ref{thm:local-reconstruction}. The conclusion therefore follows
directly from that theorem.
<!-- end proof-8 -->
<!-- theorem-9: body_md -->
\label{thm:local-reconstruction-dual}
Let $\mathcal X=\operatorname{add}\mathcal X$ be strongly contravariantly finite and $n$-rigid. Then
\[
 \mathcal X
 =
 \operatorname{Tr}^{-}_{n}(\mathcal X)
 \cap
 \bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X.
\]
Equivalently, for every $A\in\bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X$,
\[
 \Omega_{\mathcal X}^n(\pi_{\mathcal X} A)\cong0
 \quad\Longleftrightarrow\quad
 A\in\mathcal X.
\]
<!-- end theorem-9 -->
<!-- proof-9: body_md -->
The inclusion from left to right follows from $n$-rigidity and the fact that
objects of $\mathcal X$ are zero in the quotient. Conversely, let
\[
 A=B_0\in\operatorname{Tr}^{-}_{n}(\mathcal X)
       \cap\bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X
\]
and use the right approximation $\mathbb E$-triangle \eqref{eq:right-tower}. Equation
\eqref{eq:omega-tower} and Lemma~\ref{lem:zero-object-quotient} give
$B_n\in\mathcal X$.

Fix $X\in\mathcal X$. The long exact sequences of Gorsky--Nakaoka--Palu
\cite[Theorem 3.6]{GNP}, applied to \eqref{eq:right-tower}, give
\[
 \mathbb E(B_j,X)\cong\mathbb E^{j+1}(B_0,X)=0
 \hspace{2.5mm}(0\leq j\leq n-1).
\]
Suppose inductively that $B_{j+1}\in\mathcal X$. The extension class of
\[

belongs to $\mathbb E(B_j,B_{j+1})=0$. Hence the triangle splits and $B_j$ is a
direct summand of $X^j$. Since $\mathcal X=\operatorname{add}\mathcal X$, it follows that $B_j\in\mathcal X$.
Descending induction from $B_n\in\mathcal X$ yields $A=B_0\in\mathcal X$.

<!-- theorem-11: body_md -->
\label{thm:recognition}

\item $\mathcal X$ satisfies the two-sided maximal $n$-orthogonality condition of {\rm Definition~\ref{def:maxorth}}.
\item $\Sigma_{\mathcal X}^n\cong\mathbf 0$ on $\mathcal C/[\mathcal X]$.
\item $\Omega_{\mathcal X}^n\cong\mathbf 0$ on $\mathcal C/[\mathcal X]$.
\end{enumerate}

<!-- remark-3: body_md -->
\label{rem:hypotheses}
The following observations separate the rigidity and approximation assumptions in Theorem~\ref{thm:recognition}.
\begin{enumerate}[{\rm(1)}]
	\item The $n$-rigidity hypothesis cannot be omitted.
	
	Let $\Lambda=k[\varepsilon]/(\varepsilon^2)$ and $\mathcal C=\mathrm{mod}\text{-}\Lambda$ with its usual exact structure. Put $\mathcal X=\mathcal C$. Then $\mathcal C/[\mathcal X]=0$, so every power of the canonical suspension and loop is zero. However $\mathcal X$ is not rigid. For the simple module $S=\Lambda/(\varepsilon)$, the following nonsplit sequence
		\[
		0\longrightarrow S\longrightarrow\Lambda\longrightarrow S\longrightarrow0
		\]
		gives $\operatorname{Ext}^1_\Lambda(S,S)\neq0$. This is a nonsplit exact
		example that is extriangulated but not triangulated. It shows that
		nilpotency of the canonical suspension and loop alone cannot recognize cluster tilting.
	\item The two approximation directions are independent data.
		
	Truncation does not by itself recover missing approximation theory. Approximation finiteness is an independent part of higher cluster tilting theory in infinite type. Holm--J\o rgensen \cite[Theorems D and E]{HJ} distinguish weakly $d$ cluster tilting subcategories from $d$ cluster tilting subcategories by separate left and right approximation conditions. In their $A_\infty$ model, weakly $d$ cluster tilting subcategories need not be functorially finite. Holm--J\o rgensen \cite[Propositions 3.4 and 3.5]{HJ} characterize the two approximation properties separately. Thus maximal higher orthogonality by itself does not supply approximation finiteness.
		
	Accordingly, Theorem~\ref{thm:one-sided} gives the natural one-sided statement. Under strong covariant finiteness, the condition $\Sigma_{\mathcal X}^n\cong\mathbf 0$ transfers maximal orthogonality from the left to the right. The dual statement transfers it in the opposite direction. The proof of the two-sided recognition theorem uses both approximation directions. This observation explains the formulation of the theorem. It is not asserted here as a counterexample showing that no weaker two-sided hypothesis can exist.
\end{enumerate}

\label{cor:triangulated-ntruncated}
Let $\mathcal T$ be a triangulated category and $\mathcal X=\operatorname{add}\mathcal X$ be a functorially finite subcategory satisfying

\hspace{2.5mm}

\item $\mathcal T/[\mathcal X]$ is $n$-truncated in the sense of Mochizuki--Nakaoka--Ogawa \cite[Definition 1.2.3]{MNO}.
\item $\Sigma_{\mathcal X}^n\cong\mathbf 0$.
\item $\Omega_{\mathcal X}^n\cong\mathbf 0$.
\end{enumerate}

\[
 A\in\mathcal X
 \quad\Longleftrightarrow\quad
 \left\{
 \begin{array}{l}
 \mathcal T(\mathcal X,A[i])=0\quad(1\leq i\leq n),\\
 \Sigma_{\mathcal X}^n(\pi_{\mathcal X} A)\cong0,
 \end{array}
 \right.
\]
and
\[
 A\in\mathcal X
 \quad\Longleftrightarrow\quad
 \left\{
 \begin{array}{l}
 \mathcal T(A,\mathcal X[i])=0\quad(1\leq i\leq n),\\
 \Omega_{\mathcal X}^n(\pi_{\mathcal X} A)\cong0.
 \end{array}
 \right.
\]

Hence \textup{(1)}, \textup{(3)}, and \textup{(4)} are equivalent. Thus \textup{(2)} is equivalent to \textup{(3)} and \textup{(4)}. <!-- proposition-3: body_md -->
\label{prop:exact-two-resolutions}
	Let $\mathcal E$ be an exact category and $\mathcal X=\operatorname{add}\mathcal X$ satisfy
	\[
	\operatorname{Ext}^i_{\mathcal E}(\mathcal X,\mathcal X)=0,
	\hspace{2.5mm} 1\leq i\leq n.
	\]
	Then the following are equivalent.
	\begin{enumerate}[{\rm(1)}]
		\item $\mathcal X$ is an $(n+1)$ cluster tilting subcategory of $\mathcal E$.
		\item Every $E\in\mathcal E$ admits admissible exact sequences
		\[
		0\longrightarrow E\longrightarrow X^0\longrightarrow X^1\longrightarrow\cdots\longrightarrow X^n\longrightarrow0
		\]
		and
		\[
		0\longrightarrow X_n\longrightarrow\cdots\longrightarrow X_1\longrightarrow X_0\longrightarrow E\longrightarrow0
		\]
		with all terms $X^i,X_i$ in $\mathcal X$.
	\end{enumerate}
	Here an admissible exact sequence means a sequence obtained by splicing
	conflations in the exact structure of $\mathcal E$.
	More precisely, the first family of sequences exists for every $E$ if and
	only if $\mathcal X$ is strongly covariantly finite and $\Sigma_{\mathcal X}^n\cong\mathbf 0$.
	Dually, the second family exists for every $E$ if and only if $\mathcal X$ is
	strongly contravariantly finite and $\Omega_{\mathcal X}^n\cong\mathbf 0$.
<!-- end proposition-3 -->
<!-- proof-14: body_md -->
The equivalence of \textup{(1)} and \textup{(2)} is the criterion of
	Kvamme \cite[Proposition 4.4]{Kvamme}, applied with $d=n+1$. We do not reprove that
	result. It remains to verify the two one-sided assertions.
	
	Consider the first admissible exact sequence and write it as
	\[
	0\longrightarrow A_0=E\longrightarrow X^0\longrightarrow X^1
	\longrightarrow\cdots\longrightarrow X^n\longrightarrow0.
	\]
	For $0\le j\le n-1$, let $A_{j+1}$ be the cokernel of the admissible monomorphism $A_j\to X^j$. Exactness gives conflations
	\[
	0\longrightarrow A_j\longrightarrow X^j\longrightarrow A_{j+1}\longrightarrow0,
	\hspace{2.5mm} 0\le j\le n-1,
	\]
	with $A_n\cong X^n\in\mathcal X$. We claim that each $A_j\to X^j$ is a left $\mathcal X$-approximation. Fix $X\in\mathcal X$. Dimension shifting along the tail of the resolution, together with the $n$-rigidity of $\mathcal X$, gives
	\[
	\operatorname{Ext}^1_{\mathcal E}(A_{j+1},X)
	\cong
	\operatorname{Ext}^{\,n-j}_{\mathcal E}(X^n,X)=0.
	\]
	Applying $\operatorname{Hom}_{\mathcal E}(-,X)$ to
	$0\to A_j\to X^j\to A_{j+1}\to0$ therefore shows that
	\[
	\operatorname{Hom}_{\mathcal E}(X^j,X)
	\longrightarrow
	\operatorname{Hom}_{\mathcal E}(A_j,X)
	\]
	is surjective. Thus the $n$ conflations above form a valid tower of left approximations for the canonical one-sided suspension, and
	\[
	\Sigma_{\mathcal X}^n(\pi_{\mathcal X} E)\cong\pi_{\mathcal X}(A_n)=0.
	\]
	Since this holds for every $E$, Lemma~\ref{lem:objectwise-zero} gives $\Sigma_{\mathcal X}^n\cong\mathbf 0$.
	The same argument also proves strong covariant finiteness, because the first
	conflation supplies a left approximation inflation for every $E$.
	
	Conversely, if the canonical one-sided suspension is defined and $\Sigma_{\mathcal X}^n\cong\mathbf 0$, its defining tower consists of $n$ conflations
	\[
	A_j\longrightarrow X^j\longrightarrow A_{j+1},
	\hspace{2.5mm} 0\le j\le n-1.
	\]
	By Lemma~\ref{lem:zero-object-quotient}, the terminal object $A_n$ lies in $\mathcal X$. Splicing these conflations gives the admissible exact sequence
	\[
	0\longrightarrow E\longrightarrow X^0\longrightarrow\cdots
	\longrightarrow X^{n-1}\longrightarrow A_n\longrightarrow0,
	\]
	which is the first resolution after renaming $A_n$ as $X^n$. This proves
	the first one-sided equivalence. The statement for $\Omega_{\mathcal X}^n$ and the second
	resolution is dual.

<!-- proposition-4: body_md -->
\label{prop:rigidity-collapse}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be a small $R$-linear extriangulated category and $\mathcal X=\operatorname{add}\mathcal X$.
\begin{enumerate}[{\rm(1)}]
\item Suppose that $\mathcal X$ is strongly contravariantly finite,
\[
 \mathcal X=\bigcap_{i=1}^{n}\mathcal X^{\perp_i},
\]
and
\[
 \mathbb E^{n+1}(\mathcal X,\mathcal X)=0.
\]
Then $\mathcal C=\mathcal X$ and $\mathbb E=0$.
\item Dually, the same conclusion holds if $\mathcal X$ is strongly covariantly finite,
\[
 \mathcal X=\bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X,
\]
and $\mathbb E^{n+1}(\mathcal X,\mathcal X)=0$.
\end{enumerate}
<!-- end proposition-4 -->
<!-- proof-15: body_md -->
We prove \textup{(1)}.  The displayed maximal orthogonality implies
$\mathbb E^i(\mathcal X,\mathcal X)=0$ for $1\leq i\leq n$. Combined with the last hypothesis, this yields that \(\mathcal{X}\) is \((n+1)\)-rigid. By the first assertion of Proposition~\ref{prop:dual},
$\Omega_{\mathcal X}^n\cong\mathbf 0$.

Fix $A\in\mathcal C$, put $B_0=A$, and choose an iterated right $\mathcal X$-approximation $\mathbb E$-triangle
\[
 B_{j+1}\longrightarrow X_j\longrightarrow B_j\dashrightarrow,
\hspace{2.5mm} X_j\in\mathcal X,
\hspace{2.5mm} 0\leq j\leq n-1.
\]
The canonical loop construction gives
$\Omega_{\mathcal X}^n(\pi_{\mathcal X} A)\cong\pi_{\mathcal X}(B_n)$. Hence
$B_n\in\mathcal X$ by Lemma~\ref{lem:zero-object-quotient}. Fix $X\in\mathcal X$. For each $0\leq j\leq n-1$, the long exact sequence of Gorsky--Nakaoka--Palu \cite[Theorem 3.6]{GNP} attached to
\[
 B_{j+1}\longrightarrow X_j\longrightarrow B_j\dashrightarrow
\]
contains
\[
 \mathbb E^{j+1}(X,X_j)\longrightarrow
 \mathbb E^{j+1}(X,B_j)\longrightarrow
 \mathbb E^{j+2}(X,B_{j+1})\longrightarrow
 \mathbb E^{j+2}(X,X_j).
\]
Since $1\leq j+1\leq n$ and $2\leq j+2\leq n+1$, both outer terms vanish by $(n+1)$-rigidity.  Hence
\[
 \mathbb E^{j+1}(X,B_j)\cong\mathbb E^{j+2}(X,B_{j+1})
 \hspace{2.5mm}(0\leq j\leq n-1).
\]
Iterating these isomorphisms gives
\[
 \mathbb E(X,A)
 \cong\mathbb E^2(X,B_1)
 \cong\cdots\cong
 \mathbb E^{n+1}(X,B_n)=0,
\]
where the last term vanishes because $B_n\in\mathcal X$.
Thus every object of $\mathcal X$ is projective in $\mathcal C$.

By the construction of positive extensions in Gorsky--Nakaoka--Palu
\cite[Section 3]{GNP}, $\mathbb E^m=\mathbb E^{\diamond m}$. Thus projectivity of $X\in\mathcal X$ implies
$\mathbb E^m(X,-)=0$ for every $m\geq1$.  Hence every $A\in\mathcal C$ belongs to
$\bigcap_{i=1}^{n}\mathcal X^{\perp_i}=\mathcal X$, so $\mathcal C=\mathcal X$.  Finally
$\mathbb E(\mathcal X,\mathcal X)=0$, and therefore $\mathbb E=0$ on $\mathcal C$.  Part \textup{(2)} is dual.

<!-- theorem-12: body_md -->
\label{thm:sharp-truncation}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be a small $R$-linear extriangulated category with $\mathbb E\neq0$, and let $\mathcal X=\operatorname{add}\mathcal X$ be strongly functorially finite.  Assume that $\mathcal X$ satisfies the two-sided maximal $n$-orthogonality condition.  Then
\[
 \mathbb E^{n+1}(\mathcal X,\mathcal X)\neq0
\]
and
\[
 \nu_{\mathcal X}(\mathcal C)=n.
\]
In particular, the positive global dimension of $\mathcal C$ in the sense of Gorsky--Nakaoka--Palu \cite[Definition 3.30]{GNP} is at least $n+1$ (or infinite).
<!-- end theorem-12 -->
<!-- proof-16: body_md -->
If $\mathbb E^{n+1}(\mathcal X,\mathcal X)=0$, either part of Proposition~\ref{prop:rigidity-collapse} would imply $\mathbb E=0$, contrary to the hypothesis.  This proves the first assertion.

By Theorem~\ref{thm:recognition}, $\Sigma_{\mathcal X}^n\cong\mathbf 0$, so
$\nu_{\mathcal X}(\mathcal C)\leq n$.  For $n=1$ this already gives
$\nu_{\mathcal X}(\mathcal C)=1$.  Assume $n\geq2$ and suppose that
$\nu_{\mathcal X}(\mathcal C)=m<n$.  Since maximal $n$-orthogonality implies $n$-rigidity, $\mathcal X$ is $m$-rigid. Applying Theorem~\ref{thm:recognition} with $m$ in place of $n$ shows that $\mathcal X$ is two-sided maximally $m$-orthogonal.  The original $n$-rigidity also gives
$\mathbb E^{m+1}(\mathcal X,\mathcal X)=0$.  Proposition~\ref{prop:rigidity-collapse}, now at level $m$, forces $\mathbb E=0$, again a contradiction.  Hence $\nu_{\mathcal X}(\mathcal C)=n$.

Finally, $\mathbb E^{n+1}(\mathcal X,\mathcal X)\neq0$ means that the bifunctor $\mathbb E^{n+1}$ is nonzero, so the positive global dimension cannot be at most $n$.

Equivalently, the canonical quotient is $n$-truncated and, for $n\geq2$, it is not $(n-1)$-truncated.

$\mathcal T(A,B[1])=0$ for all $A,B$, then essential surjectivity of the shift gives $\mathcal T(A,C)=0$ for all $A,C$, forcing $\mathcal T=0$. Hence the contradiction argument in Theorem~\ref{thm:sharp-truncation} yields the stated exact nilpotency index. The equivalence with $n$-truncatedness follows from Mochizuki--Nakaoka--Ogawa \cite[Definition 1.2.3]{MNO}.

<!-- corollary-8: body_md -->
\label{cor:minimal-level}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be a small $R$-linear extriangulated category and let
$\mathcal X=\operatorname{add}\mathcal X$ be strongly functorially finite and $n$-rigid for some $n\geq1$.
For $1\leq m\leq n$, the following are equivalent:
\begin{enumerate}[{\rm(1)}]
\item $\nu_{\mathcal X}(\mathcal C)=m$.
\item $\mathcal X$ satisfies the two-sided maximal $m$-orthogonality condition, and it satisfies no two-sided maximal $r$-orthogonality condition for $1\leq r<m$.
\end{enumerate}
If $\mathcal C$ has enough projectives and injectives, this is equivalently the least $m$ for which
$\mathcal X$ is an $(m+1)$ cluster tilting subcategory.
<!-- end corollary-8 -->
<!-- proof-19: body_md -->
Assume first that $\nu_{\mathcal X}(\mathcal C)=m$.  Since $m\leq n$, the subcategory $\mathcal X$ is $m$-rigid, and Theorem~\ref{thm:recognition} applied with $n=m$ shows that $\mathcal X$ satisfies two-sided maximal $m$-orthogonality.  If it satisfied the corresponding condition for some $r<m$, then another application of Theorem~\ref{thm:recognition} would give $\Sigma_{\mathcal X}^r\cong\mathbf 0$, contrary to the minimality of $m$.

Conversely, suppose that the condition in \textup{(2)} holds. Theorem~\ref{thm:recognition}, with $n=m$, gives $\Sigma_{\mathcal X}^m\cong\mathbf 0$, so $\nu_{\mathcal X}(\mathcal C)\leq m$. If $\nu_{\mathcal X}(\mathcal C)=r<m$, then $n$-rigidity implies $r$-rigidity and Theorem~\ref{thm:recognition}, now with $n=r$, would force two-sided maximal $r$-orthogonality, a contradiction. Hence $\nu_{\mathcal X}(\mathcal C)=m$. The final assertion follows from Corollary~\ref{cor:classical}.

<!-- corollary-9: body_md -->
\label{cor:nil-index-invariant}
Let $(\mathcal C,\mathcal X)$ and $(\mathcal D,\mathcal Y)$ be pairs for which the canonical pretriangulated ideal quotients are defined.  Suppose
\[
 \Phi\colon \mathcal C/[\mathcal X]\xrightarrow{\sim}\mathcal D/[\mathcal Y]
\]
is an equivalence compatible with the canonical suspensions, in the sense that
$\Phi\Sigma_{\mathcal X}\cong\Sigma_{\mathcal Y}\Phi$ naturally.  Then
\[
 \nu_{\mathcal X}(\mathcal C)=\nu_{\mathcal Y}(\mathcal D).
\]
In particular, under the hypotheses of Corollary~\ref{cor:minimal-level} on both pairs, an equivalence of the ideal quotients that is compatible with the canonical suspensions preserves the minimal higher maximal orthogonality level. In the classical higher extension setting it preserves the minimal cluster tilting level.
<!-- end corollary-9 -->
<!-- proof-20: body_md -->
For every $m\geq1$, iteration of the compatibility isomorphism gives
$\Phi\Sigma_{\mathcal X}^m\cong\Sigma_{\mathcal Y}^m\Phi$.  Since $\Phi$ is an equivalence,
$\Sigma_{\mathcal X}^m\cong\mathbf 0$ if and only if $\Sigma_{\mathcal Y}^m\cong\mathbf 0$.
Thus the two sets of nilpotency exponents coincide. In particular, either both are infinite or they have the same least element.  This proves the first assertion, and the remaining statements follow from Corollary~\ref{cor:minimal-level}.

<!-- proposition-5: body_md -->
\label{prop:no-internal-rigidity}
	Fix $n\geq1$. There exist two exact, hence extriangulated, pairs $(\mathcal C_0,\mathcal X_0)$ and $(\mathcal C_1,\mathcal X_1)$ such that their canonical pretriangulated ideal quotients are equivalent to the zero category, while $\mathcal X_0$ satisfies the two-sided maximal $n$-orthogonality condition (and is standard $(n+1)$ cluster tilting in this exact setting), whereas $\mathcal X_1$ is not even rigid.
<!-- end proposition-5 -->
<!-- proof-21: body_md -->
Let $k$ be a field and take $\mathcal C_0=\mathrm{mod}\text{-}k$ with its usual exact structure and $\mathcal X_0=\mathcal C_0$. Since $\mathcal C_0$ is semisimple, all positive higher extensions vanish. Hence
	\[
	\mathcal X_0
	=\bigcap_{i=1}^{n}{}^{\perp_i}\mathcal X_0
	=\bigcap_{i=1}^{n}\mathcal X_0^{\perp_i},
	\]
	and $\mathcal X_0$ is strongly functorially finite. Thus $\mathcal X_0$ satisfies the two-sided maximal $n$-orthogonality condition and, in this exact setting, is a standard $(n+1)$ cluster tilting subcategory.
	
	Now let $\Lambda=k[\varepsilon]/(\varepsilon^2)$, put $\mathcal C_1=\mathrm{mod}\text{-}\Lambda$, and take $\mathcal X_1=\mathcal C_1$. Again $\mathcal X_1$ is strongly functorially finite. In both cases every identity morphism factors through an object of $\mathcal X_i$, so
	\[
	\mathcal C_0/[\mathcal X_0]=0=\mathcal C_1/[\mathcal X_1]
	\]
	as canonical pretriangulated categories. On the other hand, if $S=\Lambda/(\varepsilon)$, the nonsplit exact sequence
	\[
	0\longrightarrow S\longrightarrow \Lambda\longrightarrow S\longrightarrow0
	\]
	shows that $\mathbb E(S,S)=\operatorname{Ext}^1_\Lambda(S,S)\neq0$. Thus $\mathcal X_1$ is not rigid. The two quotient categories therefore carry identical quotient-internal information while the ambient cluster tilting behaviour is different.

<!-- corollary-10: body_md -->
\label{cor:zero-endpoint}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be an extriangulated category, and let $\mathcal X=\operatorname{add}\mathcal X$ be a strongly functorially finite rigid subcategory. Then
\[
        \mathcal X\text{ is cluster tilting}
        \quad\Longleftrightarrow\quad
        \Sigma_{\mathcal X}\cong\mathbf 0
        \quad\Longleftrightarrow\quad
        \Omega_{\mathcal X}\cong\mathbf 0.
\]
Whenever the hypotheses of an established abelian quotient theorem are satisfied, this endpoint has an abelian ideal quotient. In particular, if $\mathcal C$ has enough projective and enough injective objects, then $\mathcal C/[\mathcal X]$ is abelian by Zhou--Zhu \cite[Corollary 3.5]{ZZct}.
<!-- end corollary-10 -->
<!-- proof-22: body_md -->
By Lemma~\ref{lem:adjoint-zero}, we obtain
$\Sigma_{\mathcal X}\cong\mathbf 0$ if and only if $\Omega_{\mathcal X}\cong\mathbf 0$, since
$\Sigma_{\mathcal X}$ is left adjoint to $\Omega_{\mathcal X}$. Suppose first that $\mathcal X$ is cluster tilting. For $A\in\mathcal C$, choose a left $\mathcal X$-approximation $\mathbb E$-triangle
\[
 A\longrightarrow X_A\longrightarrow A^+\dashrightarrow.
\]
For every $X\in\mathcal X$, the standard exact sequence associated with this $\mathbb E$-triangle contains
\[
 \mathcal C(X_A,X)\longrightarrow\mathcal C(A,X)
 \longrightarrow\mathbb E(A^+,X)\longrightarrow\mathbb E(X_A,X).
\]
The first map is surjective and the last term vanishes by rigidity. Hence $\mathbb E(A^+,\mathcal X)=0$, so $A^+\in\mathcal X$ by cluster tilting. Thus $\Sigma_{\mathcal X}(\pi_{\mathcal X} A)=\pi_{\mathcal X}(A^+)=0$ for every $A$, and Lemma~\ref{lem:objectwise-zero} gives $\Sigma_{\mathcal X}\cong\mathbf 0$.

Conversely, assume $\Sigma_{\mathcal X}\cong\mathbf 0$ and let $A$ satisfy $\mathbb E(\mathcal X,A)=0$. In the same approximation $\mathbb E$-triangle, $\pi_{\mathcal X}(A^+)=0$. Hence $A^+\in\mathcal X$ by Lemma~\ref{lem:zero-object-quotient}. Its extension class belongs to $\mathbb E(A^+,A)=0$, so the $\mathbb E$-triangle splits and $A$ is a direct summand of $X_A$. Therefore $A\in\mathcal X$, and rigidity gives
\[
        \mathcal X=\mathcal X^{\perp_1}.
\]
Since $\Omega_{\mathcal X}\cong\mathbf 0$, the dual argument gives $\mathcal X={}^{\perp_1}\mathcal X$. Together with strong functorial finiteness, these equalities are precisely the cluster tilting condition. The final abelian assertion is the cited theorem.

<!-- theorem-13: body_md -->
\label{thm:unified-endpoints}
Let $(\mathcal C,\mathbb E,\mathfrak s)$ be an extriangulated category and let $\mathcal X=\operatorname{add}\mathcal X$ be strongly functorially finite. Equip $\mathcal C/[\mathcal X]$ with its canonical pretriangulated structure.
\begin{enumerate}[{\rm(1)}]
\item \textup{\bf (Zero endpoint)} If $\mathcal X$ is rigid, then
\[
        \Sigma_{\mathcal X}\cong\mathbf 0
        \quad\Longleftrightarrow\quad
        \Omega_{\mathcal X}\cong\mathbf 0
        \quad\Longleftrightarrow\quad
        \mathcal X\text{ is cluster tilting}.
\]
Under the hypotheses of the standard extriangulated abelian quotient theorem, this implies that $\mathcal C/[\mathcal X]$ is abelian. For example, this is the theorem of Zhou--Zhu \cite[Corollary 3.5]{ZZct} when $\mathcal C$ has enough projective and injective objects.

\item \textup{\bf (Stable endpoint)} Assume in addition the Auslander--Reiten duality hypotheses of Zhou--Zhu \cite[Theorem 4.3]{ZZtri}. Then the following conditions are equivalent:
\begin{enumerate}[{\rm(a)}]
\item $(\mathcal C,\mathcal C)$ is an $\mathcal X$-mutation pair.
\item the canonical pretriangulated category $\mathcal C/[\mathcal X]$ is triangulated.
\item $\tau\underline{\mathcal X}=\overline{\mathcal X}$.
\item the canonical suspension $\Sigma_{\mathcal X}$ is an autoequivalence.
\end{enumerate}

\end{enumerate}
Consequently, within extriangulated categories, the cluster tilting and stable regimes are detected by the two extreme behaviours
\[
        \Sigma_{\mathcal X}\cong0
        \hspace{2.5mm}\text{and}\hspace{2.5mm}
        \Sigma_{\mathcal X}\text{ invertible}
\]
of one canonical endofunctor on one ideal quotient.  The comparison is summarized by
\[
\begin{array}{c@{\qquad\qquad\qquad}c}
\Sigma_{\mathcal X}\cong0 & \Sigma_{\mathcal X}\text{ invertible}\\[1mm]
\Updownarrow & \Updownarrow {\scriptstyle (\mathrm{AR})}\\[1mm]
\mathcal X\text{ cluster tilting} & (\mathcal C,\mathcal C)\text{ is an }\mathcal X\text{-mutation pair}\\[1mm]
\Downarrow\ {\scriptstyle (\mathrm{AQ})} & \Updownarrow\ {\scriptstyle (\mathrm{AR})}\\[1mm]
\mathcal C/[\mathcal X]\text{ abelian} & \mathcal C/[\mathcal X]\text{ triangulated}.
\end{array}
\]
Here \textup{(AQ)} denotes the additional hypotheses of the applicable extriangulated abelian quotient theorem, for instance the enough projectives and enough injectives hypothesis of Zhou--Zhu \cite[Corollary 3.5]{ZZct}. The symbol \textup{(AR)} denotes the Auslander--Reiten duality hypotheses of Zhou--Zhu \cite[Theorem 4.3]{ZZtri}. Thus only the upper equivalence in the left column is unconditional under the assumptions of part \textup{(1)}. The abelian conclusion is a cited consequence under \textup{(AQ)}.
<!-- end theorem-13 -->
<!-- proof-23: body_md -->
The equivalences in part \textup{(1)} are Corollary~\ref{cor:zero-endpoint}. The abelian assertion is the cited quotient theorem under \textup{(AQ)}. For part \textup{(2)}, the equivalence of \textup{(a)}--\textup{(c)} is due to Zhou--Zhu \cite[Theorem 4.3]{ZZtri}. Condition \textup{(b)} implies \textup{(d)} because the suspension of a triangulated category is an autoequivalence, while \textup{(d)} implies \textup{(b)} by Lemma~\ref{lem:stable-right} applied to the canonical right triangulated quotient.

<!-- remark-6: body_md -->
\label{rem:scope-endpoints}
\begin{enumerate}[{\rm(1)}]
	\item In degree one, the zero endpoint and stable endpoint use only the ordinary extriangulated extension bifunctor and therefore require neither smallness nor $R$-linearity.
	
	\item The left column of Theorem~\ref{thm:unified-endpoints} is specific to the canonical ideal quotient and its ambient extriangulated structure. It is not a formal statement that a pretriangulated category with zero suspension is abelian. Indeed, in the formalism of Beligiannis--Reiten \cite[Chapter II, Section 1]{BR}, an additive category with kernels and cokernels carries a pretriangulated structure with $\Sigma=\Omega=0$, without being abelian in general. The content of the zero endpoint is therefore the recognition
	\[
	\Sigma_{\mathcal X}\cong0
	\quad\Longrightarrow\quad
	\mathcal X\text{ cluster tilting}
	\]
	under rigidity and approximation hypotheses, followed, under \textup{(AQ)}, by the established abelian quotient theorem.  By contrast, invertibility of the suspension of a right triangulated category is already sufficient to recover a triangulated structure by Lemma~\ref{lem:stable-right}.
\end{enumerate}

<!-- corollary-11: body_md -->
\label{cor:triangulated-specialization}
Let $\mathcal T$ be a triangulated category with Auslander--Reiten translation $\tau$, and let $\mathcal X=\operatorname{add}\mathcal X$ be a functorially finite rigid subcategory. Denote by $\Sigma_{\mathcal X}$ the canonical suspension of the pretriangulated ideal quotient $\mathcal T/[\mathcal X]$.
\begin{enumerate}[{\rm(1)}]
\item
\[
 \mathcal X\text{ is cluster tilting}
 \quad\Longleftrightarrow\quad
 \Sigma_{\mathcal X}\cong0.
\]
In this case $\mathcal T/[\mathcal X]$ is abelian by Koenig--Zhu \cite[Theorem 3.3]{KZ}.
See also Zhou--Zhu \cite[Corollary 3.6]{ZZct}.
\item
\[
 \tau\mathcal X=\mathcal X
 \quad\Longleftrightarrow\quad
 \mathcal T/[\mathcal X]\text{ is triangulated}
 \quad\Longleftrightarrow\quad
 \Sigma_{\mathcal X}\text{ is an autoequivalence}.
\]
The first equivalence is J\o rgensen's theorem \cite[Theorem 3.3]{Jor}, in the formulation of Zhou--Zhu \cite[Corollary 4.4]{ZZtri}. The equivalence with invertibility of $\Sigma_{\mathcal X}$ follows from the canonical right triangulated structure and Lemma~\ref{lem:stable-right}.
\end{enumerate}
Hence the familiar triangulated diagram
\[
\begin{array}{c@{\qquad\qquad\qquad}c}
\Sigma_{\mathcal X}\cong0 & \Sigma_{\mathcal X}\text{ invertible}\\[1mm]
\Updownarrow & \Updownarrow\\[1mm]
\mathcal X\text{ cluster tilting} & \tau\mathcal X=\mathcal X\\[1mm]
\Downarrow & \Updownarrow\\[1mm]
\mathcal T/[\mathcal X]\text{ abelian} & \mathcal T/[\mathcal X]\text{ triangulated}
\end{array}
\]
is a special case of the extriangulated theorem above.

<!-- remark-7: body_md -->
If an endofunctor $F$ of an additive category satisfies $F^p\cong\operatorname{Id}$ for some $p\geq1$, then $F^{p-1}$ is an inverse equivalence. If, in addition, $F^m\cong\mathbf 0$ for some $m\geq1$, composing with an inverse of $F^m$ gives $\operatorname{Id}\cong\mathbf 0$, so the category is zero. Applied to $F=\Sigma_{\mathcal X}$, this shows that a nonzero canonical quotient cannot be both truncated and periodic.
