\label{thm:general}
		Let $(M,g)$ be a smooth closed Riemannian manifold of dimension
		$n\geq3$, and let $\Sigma\subset M$ be a smooth embedded closed
		submanifold of codimension 2. There are constants
		$B=B(M,g,\Sigma)$ and $A_m=A_m(M,g,\Sigma)$, $m=2,3,\ldots$, such that
		every eigenfunction $e_\lambda$ with $\lambda\geq2$ satisfies
		\begin{equation}\label{eq:general-quant}
			\left\lVert e_\lambda\right\rVert_{L^2(\Sigma)}^2
			\leq\left(A_m\lambda+\frac Bm\lambda\log\lambda\right)\left\lVert e_\lambda\right\rVert_{L^2(M)}^2.
		\end{equation}
		In particular, as $\lambda\to\infty$,
		\begin{equation}\label{eq:general-littleoh}
			\left\lVert e_\lambda\right\rVert_{L^2(\Sigma)}
			=o\!\left(\lambda^{1/2}\sqrt{\log\lambda}\right)\left\lVert e_\lambda\right\rVert_{L^2(M)}.
		\end{equation}
		
		If $(M,g)$ and the embedding $\Sigma\hookrightarrow M$
		are real analytic, then the log-free estimate \eqref{logfree} holds.

\label{upper:polynomial}
Let $d\ge0,\ k\ge1$, let $\Theta\subset\mathbb R^d$ be compact, and fix
$\psi_0,\psi_1\in C_c^\infty(\mathbb R^{k+1})$.
Suppose $\gamma_\theta(t,\tau,\xi)\in\mathbb R^k$ is jointly real analytic
near
\[

\begin{equation}\label{upper:polynomial-operator}
\begin{aligned}
[\mathcal P_{\theta,\epsilon}F](t,\xi)
={}&\psi_0(t,\xi)\int_\mathbb R
\frac{\chi_1(2\tau/\rho)(1-\chi_1(\tau/\epsilon))}{\tau}\,

\end{equation}

\[

(\theta,t-\tau,\gamma_\theta(t,\tau,\xi))
\]

[\mathcal P_0U](\theta,t,\xi)
=\psi_0(t,\xi)\operatorname{p.v.}\int_\mathbb R\frac{\chi_1(2\tau/\rho)}\tau

 e^{ivt/\epsilon}\mathcal P_0(e^{-ivt/\epsilon}U)\,d v.

 Hence

\label{upper:transport}
Let $k\ge1$, and let $a(t,\tau,\xi)\in\mathbb C$ and
$\gamma(t,\tau,\xi)\in\mathbb R^k$ be fixed smooth functions. Assume that,
for $|\tau|\leq r_0$, $a$ vanishes outside a fixed compact set
$K\subset\mathbb R^{k+1}$ in $(t,\xi)$, and
\[
\gamma(t,0,\xi)=\xi,
\qquad
\left\lVert \partial_\xi\gamma(t,\tau,\xi)-I\right\rVert_{\mathbb R^k\to\mathbb R^k}
\leq\tfrac12.
\]
Define
\begin{equation}\label{upper:Radon}
[\mathcal R_\lambda F](t,\xi)
=\int_\mathbb R k_\lambda(\tau)a(t,\tau,\xi)
F(t-\tau,\gamma(t,\tau,\xi))\,d\tau.
\end{equation}
There is $B$ independent of $m$ such that, for every $m\geq2$,
\begin{equation}\label{upper:transportestimate}
\begin{aligned}
\left\lVert \mathcal R_\lambda F\right\rVert_{L^2(\mathbb R^{k+1})}
\leq{}&\left(C_m+\frac Bm\log\lambda\right)
\left\lVert F\right\rVert_{L^2(\mathbb R^{k+1})}
+C_m\lambda^{-1}\left\lVert \nabla_\xi F\right\rVert_{L^2(\mathbb R^{k+1})},
\qquad\lambda\geq2.
\end{aligned}
\end{equation}
If $\gamma$ is real analytic near
$\{(t,0,\xi):(t,\xi)\in K\}$, then
\begin{equation}\label{upper:Radon-analytic}
\left\lVert \mathcal R_\lambda F\right\rVert_{L^2(\mathbb R^{k+1})}
\leq C\left\lVert F\right\rVert_{L^2(\mathbb R^{k+1})},\qquad\lambda\geq2.
\end{equation}

\qquad \gamma(t,\tau,\xi)=\xi+\tau G(t,\tau,\xi).

\delta=c_m\lambda^{-1/m},\qquad 0<c_m<r_0/4.

$Q_j=(t_j,\xi_j)+\delta Q$ be the disjoint grid cubes meeting $K$.

G_j(t,\tau,\xi)

\frac{(t-t_j,\tau,\xi-\xi_j)^\mu}{\mu!}
\partial^\mu G(t_j,0,\xi_j),
\qquad \mu\in\mathbb N_0^{k+2}.
\]
For $(t,\xi)\in Q_j$, $|\tau|\leq2\delta$, and $0\leq u\leq1$, let
\[
\gamma_{j,u}(t,\tau,\xi)
=\xi+\tau\bigl((1-u)G_j(t,\tau,\xi)+uG(t,\tau,\xi)\bigr).
\]
Taylor's formula and a sufficiently small $c_m$ give
\begin{equation}\label{upper:Taylor}
\begin{gathered}
|G-G_j|\leq C_m\delta^{m-1},\quad
\left\lVert \partial_\xi\gamma_{j,u}-I\right\rVert_{\mathbb R^k\to\mathbb R^k}
\leq C_m|\tau|\leq\tfrac12,\quad
|\gamma_{j,u}-\xi|\leq C_m\delta.
\end{gathered}
\end{equation}
The maps $(t,\xi)\mapsto(t-\tau,\gamma_{j,u})$ are injective on $Q_j$ and the Jacobian  satisfies
\[
\left|\det \partial_{(t,\xi)}(t-\tau,\gamma_{j,u})\right|
=|\det\partial_\xi\gamma_{j,u}|\geq2^{-k}.
\]
Thus their images lie in fixed concentric enlargements $Q_j^*$, with
$\sum_j\mathbf 1_{Q_j^*}\leq C_m$, and change of variables gives
\begin{equation}\label{upper:composition}
\left\lVert F(t-\tau,\gamma_{j,u}(t,\tau,\xi))\right\rVert_{L^2(Q_j)}
\leq2^{k/2}\left\lVert F\right\rVert_{L^2(Q_j^*)}.
\end{equation}
The same estimate holds with $F$ replaced by $\nabla_\xi F$.

On $Q_j$, define
\[
[\mathcal P_jF](t,\xi)
=\int_\mathbb R k_\lambda(\tau)\chi_1(\tau/\delta)
F(t-\tau,\xi+\tau G_j(t,\tau,\xi))\,d\tau.
\]
Define the coefficient vectors
\[
\theta_\mu
=\frac{\delta^{|\mu|}}{\mu!}\partial^\mu G(t_j,0,\xi_j)
\in\mathbb R^k,
\qquad
\theta=(\theta_\mu)_{|\mu|\leq m-2}.
\]
The rescaled polynomial is
\[
G_\theta(s,\tau,\xi)
:=\sum_{|\mu|\leq m-2}\theta_\mu(s,\tau,\xi)^\mu
=G_j(t_j+\delta s,\delta\tau,\xi_j+\delta\xi).
\]
For fixed $m$, these coefficient vectors lie in a compact box
independent of $j$ and $\lambda$. Viewing $\theta$ as an independent
parameter, the map
\[
(\theta,s,\tau,\xi)\longmapsto\xi+\tau G_\theta(s,\tau,\xi)
\]
is polynomial.
Choose fixed $\psi_0,\psi_1$ equal to one on $Q$ and, respectively,
on all its images for these coefficients and $|\tau|\leq2$.
Enlarge $Q_j^*$ so its rescaling contains $\operatorname{supp}\psi_1$, and set
\[

(t_j+\delta s,\xi_j+\delta\xi),\qquad
\left\lVert F_j\right\rVert_{L^2(\mathbb R^{k+1})}=\left\lVert F\right\rVert_{L^2(Q_j^*)}.

\[

\begin{equation}\label{upper:local-polynomial-bound}
\begin{aligned}
\left\lVert \mathcal P_jF\right\rVert_{L^2(Q_j)}
&\leq\left(C_m+2^{k/2}
\int_{\rho_m/2\leq|\tau|\leq2}\frac{\,d\tau}{|\tau|}\right)
\left\lVert F\right\rVert_{L^2(Q_j^*)}
\leq C_m\left\lVert F\right\rVert_{L^2(Q_j^*)}.
\end{aligned}
\end{equation}

\emph{Taylor error.}
Since $|a(t,\tau,\xi)-a(t,0,\xi)|\leq C|\tau|$ and
\[
\begin{aligned}
&F(t-\tau,\gamma(t,\tau,\xi))
-F(t-\tau,\xi+\tau G_j(t,\tau,\xi))\\
&\quad=\tau\int_0^1(G-G_j)(t,\tau,\xi)\cdot
\nabla_\xi F(t-\tau,\gamma_{j,u}(t,\tau,\xi))\,d u,

\begin{equation}\label{upper:Taylorerror}
\begin{aligned}
\left\lVert \mathcal R_{\lambda,\delta}F
-a(t,0,\xi)\mathcal P_jF\right\rVert_{L^2(Q_j)}&\leq C_m\int_{|\tau|\leq2\delta}
\left(\left\lVert F\right\rVert_{L^2(Q_j^*)}
+\delta^{m-1}\left\lVert \nabla_\xi F\right\rVert_{L^2(Q_j^*)}\right)\,d\tau\\
&\leq C_m\left(\delta\left\lVert F\right\rVert_{L^2(Q_j^*)}
+\delta^m\left\lVert \nabla_\xi F\right\rVert_{L^2(Q_j^*)}\right).
\end{aligned}
\end{equation}
Squaring, summing over the disjoint $Q_j$, and using
\eqref{upper:local-polynomial-bound} and bounded overlap yield
\begin{equation}\label{upper:near}
\left\lVert \mathcal R_{\lambda,\delta}F\right\rVert_{L^2(\mathbb R^{k+1})}
\leq C_m\left(\left\lVert F\right\rVert_{L^2(\mathbb R^{k+1})}
+\delta^m\left\lVert \nabla_\xi F\right\rVert_{L^2(\mathbb R^{k+1})}\right).
\end{equation}

\[
\left\lVert a(t,\tau,\xi)F(t-\tau,\gamma(t,\tau,\xi))\right\rVert_{L^2(\mathbb R^{k+1})}
\leq C\left\lVert F\right\rVert_{L^2(\mathbb R^{k+1})},
\]
where $C$ is independent of $m$. Consequently,
\begin{equation}\label{upper:far}
\begin{aligned}
\left\lVert \mathcal R_\lambda-\mathcal R_{\lambda,\delta}\right\rVert
_{L^2(\mathbb R^{k+1})\to L^2(\mathbb R^{k+1})}
\leq2C\int_\delta^{r_0}\frac{\,d\tau}{\tau}=\frac{2C}{m}\log\lambda+2C\log\frac{r_0}{c_m}.
\end{aligned}
\end{equation}
Since $\delta^m=c_m^m\lambda^{-1}\leq\lambda^{-1}$,
\eqref{upper:near} proves \eqref{upper:transportestimate} with $B=2C$.

If $\gamma$ is real analytic, replace $a(t,\tau,\xi)$ by
$a(t,0,\xi)$ as above, at a uniformly bounded cost.
Instead of taking Taylor polynomials, apply
Lemma~\ref{upper:polynomial} directly to $\gamma$, with $d=0$,
$\epsilon=C_0/\lambda$. This uniformly bounds the part
with cutoff $\chi_1(2\tau/\rho)$, where $\rho>0$ is fixed
and sufficiently small that $\chi_0=1$ on $[-\rho,\rho]$.
The remaining part is supported where
$\rho/2\leq|\tau|\leq r_0$ and is bounded, as in
\eqref{upper:far}, by
$2C\log\frac{2r_0}{\rho}.$
This proves
\eqref{upper:Radon-analytic}.

Write $q=q_N$, $\gamma=\gamma_{\nu,r,q}$, and
	$b_s=\sqrt{1-(r/\nu)q(s)^2}$. All constants below are independent
	of $N,r,\nu$. Lemma~\ref{lem:profiles} gives
	\[
	\left\lVert q\right\rVert_{L^\infty(\mathbb R)}\leq C,\qquad
	\left\lVert q'\right\rVert_{L^\infty(\mathbb R)}\leq C\sqrt N.
	\]
	Since $\nu r\geq N^2$, for sufficiently large $N$ we have
	$1/2\leq b_s\leq1$ and
	\[
	b_s-1=-\frac{rq(s)^2}{\nu(1+b_s)}=O(r/\nu),\qquad
	b_s'=-\frac{rq(s)q'(s)}{\nu b_s}.
	\]
	Differentiating \eqref{eq:graph} gives
	\begin{equation}\label{eq:local-speed}
		\frac{v(s)^2}{r^2}
		=b_s^2+\frac{(b_s')^2}{r^2}+\frac{q'(s)^2}{\nu r}
		=b_s^2+\frac{q'(s)^2}{\nu r b_s^2}
		=1+O\!\left(\frac N{\nu r}\right).
	\end{equation}
	Since $b_s>0$ and $0<r<1/2$, the first two coordinates of
	$\gamma(s)$ determine $s$, so $\gamma$ is injective.
	
	For $0\leq t<s\leq1$, let
$h=s-t,\  d=d(\gamma(s),\gamma(t)),\ 
	P(s,t)=(q(s)-q(t))^2/(2h).$
	We have
	\[
	\begin{gathered}
		|q(s)-q(t)|\leq C\sqrt N\,h,\qquad
		|b_sb_t-1|\leq Cr/\nu,\\
		b_s-b_t=-\frac{r(q(s)-q(t))(q(s)+q(t))}{\nu(b_s+b_t)},
		\qquad |b_s-b_t|\leq Cr\sqrt N\,h/\nu.
	\end{gathered}
	\]
	Consequently,
	\[
	4|b_sb_t-1|\sin^2(rh/2)+(b_s-b_t)^2
	\leq C(r^3h^2/\nu
	+r^2Nh^2/\nu^2)
	\leq Cr^3h^2/\nu.
	\]
	The squared Euclidean distance therefore gives
	\begin{equation}\label{eq:chord-transfer}
		\begin{aligned}
			4\sin^2(d/2)
			&=|\gamma(s)-\gamma(t)|^2=4b_sb_t\sin^2(rh/2)
			+\frac r\nu(q(s)-q(t))^2+(b_s-b_t)^2\\
			&=4\sin^2(rh/2)+\frac r\nu(q(s)-q(t))^2
			+O\!\left(\frac{r^3h^2}{\nu}\right).
		\end{aligned}
	\end{equation}
	The exact identity above and \eqref{eq:local-speed} imply
	\[
	crh\leq d\leq\int_t^s v(u)\,d u
	=rh\left(1+O\!\left(\frac N{\nu r}\right)\right)<1.
	\]
	Since $(4\sin^2(u/2))'=2\sin u$, the mean value formula gives
	\[
	|d-rh|
	\leq\frac C{rh}\left[
	\frac r\nu(q(s)-q(t))^2+\frac{r^3h^2}{\nu}\right]
	\leq C\frac{Nh}{\nu}.
	\]
	Taylor's formula now yields
	\[
	2\sin(rh)(d-rh)
	=\frac r\nu(q(s)-q(t))^2
	+O\!\left(\frac{r^3h^2}{\nu}
	+\frac{N^2h^2}{\nu^2}\right).
	\]
	Moreover,
	\[
	hP(s,t)=\tfrac12(q(s)-q(t))^2\leq C,\qquad
	\left|\frac{rh}{\sin(rh)}-1\right|P(s,t)
	\leq Cr^2h^2P(s,t)\leq Cr^2h.

	\begin{equation}\label{eq:transfer-phase-error}
		\begin{aligned}
			\nu d
			&=\nu rh+\frac{rh}{\sin(rh)}P(s,t)
			+O\!\left(h\left(r^2+\frac{N^2}{\nu r}\right)\right)\\
			&=\nu rh+P(s,t)
			+O\!\left(h\left(r^2+\frac{N^2}{\nu r}\right)\right).
		\end{aligned}
	\end{equation}
	Also, \eqref{eq:local-speed} and the bound for $d-rh$ give
	\begin{equation}\label{eq:transfer-amplitude-bounds}
		\frac{h\sqrt{v(s)v(t)}}{\sin d}
		=\frac{\sqrt{v(s)v(t)}}r\frac{rh}d\frac d{\sin d}
		=1+O\!\left(r^2h^2+\frac N{\nu r}\right).
	\end{equation}
	
	By \eqref{eq:sphere-kernel}, pairing $s>t$ and $s<t$ in
	\eqref{eq:normalized-Q} gives
	\[
	2\pi^2\mathcal Q_{\nu,r}(q)
	=2\int_{0<t<s<1}
	\frac{\sqrt{v(s)v(t)}\phi(s)\phi(t)}{\sin d}
	\sin(\nu d)\cos(\nu rh)\,d t\,d s.
	\]

	\[
	J=\int_0^1\int_0^1\frac{\phi(t+h)\phi(t)}h
	\sin\bigl(2\nu rh+P(t+h,t)\bigr)\,d h\,d t.

	Since $\nu d\leq C\nu rh$, \eqref{eq:transfer-phase-error} gives
	\[
	\begin{gathered}
		|\sin(\nu d)|\leq C\min\{\nu rh,1\},\\
		|\sin(\nu d)-\sin(\nu rh+P)|
		\leq Ch\left(r^2+\frac{N^2}{\nu r}\right),\\
		2\sin(\nu rh+P)\cos(\nu rh)
		=\sin P+\sin(2\nu rh+P).
	\end{gathered}
	\]
	Together with \eqref{eq:transfer-amplitude-bounds}, these formulas yield
	\begin{equation}\label{eq:split-Q}

			|2\pi^2\mathcal Q_{\nu,r}(q)-I_0(q)-J|&\leq C\int_0^1\left[
			\frac{r^2h^2+N/(\nu r)}h\min\{\nu rh,1\}
			+r^2+\frac{N^2}{\nu r}\right]\,d h\\
			&\leq C\left[r^2+\frac{N^2}{\nu r}
			+\frac N{\nu r}\bigl(1+\log(\nu r)\bigr)\right]\leq C.
		\end{aligned}

	\[\partial_hP(t+h,t)

		\qquad |\partial_hP(t+h,t)|\leq CN.

		\frac{\phi(t+h)\phi(t)e^{iP(t+h,t)}}h\right]=\phi(t)e^{iP(t+h,t)}

		\frac{\phi(t+h)\phi(t)e^{iP(t+h,t)}}h\right]\right|

	\end{aligned}

	$h=(\nu r)^{-1}$. Integration by parts against $e^{2i\nu rh}$ gives
	\[

		|J|
		&\leq C\nu r\int_0^{(\nu r)^{-1}}\,d h
		+\frac C{\nu r}\left[\nu r+
		\int_{(\nu r)^{-1}}^1
		\left(\frac Nh+\frac1{h^2}\right)\,d h\right]\\
		&\leq C\left(1+\frac{N\log(\nu r)}{\nu r}\right)\leq C.
	\end{aligned}
	\]
	Combining this with \eqref{eq:split-Q} and Lemma~\ref{lem:profiles},
	\[
	2\pi^2\mathcal Q_{\nu,r}(q_N)
	\geq I_0(q_N)-|J|-C\geq c\log N-C.
	\]
	Increasing $N_0$ proves \eqref{eq:local-lower}.
