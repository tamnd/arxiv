<!-- definition-1: body_md -->
\label{def:standard_pac_complexity}
For $\varepsilon,\delta\in(0,1)$ and $\Delta\in[0,1]$, the standard
sample complexity $m_{\mathrm{std}}^{(k)}
    ({\mathcal{R}},\varepsilon,\delta,\Delta)$ 
is the smallest nonnegative integer $m_0$ for which there exists a
learning rule such that, for every $m\ge m_0$, every context
distribution ${\mathcal{D}}$, every target reward $r^\star\in{\mathcal{R}}$, and every
demonstrator $\pi_{\mathrm{demo}}$ satisfying
$
    \Delta_{r^\star}\le\Delta,
$
the learned policy satisfies
\[
    \mathbb P\!\left\{
        {\mathcal{E}}_{r^\star}^k(\widehat\pi,\pi_{\mathrm{demo}})>\varepsilon
    \right\}
    \le\delta.
\]

        \sup_{r\in{\mathcal{R}}}

    \right\}
    \le\delta.
\]

\label{thm:standard_passk_upper}
For every $k\ge2$, there exists a learning rule ${\mathcal{A}}_k$ with
the following guarantee. For every $m\ge1$, $\delta\in(0,1)$,
context distribution ${\mathcal{D}}$, demonstrator $\pi_{\mathrm{demo}}$, and
evaluation reward $r^\star\in{\mathcal{R}}$ fixed before sampling,
the policy $\widehat\pi={\mathcal{A}}_k({\mathcal{R}},S)$ satisfies
\[
    \mathbb P_S\!\left\{
        {\mathcal{E}}_{r^\star}^k(\widehat\pi,\pi_{\mathrm{demo}})
        >
        \frac{3\log|{\mathcal{R}}|}{m\log k}
        +\frac{3\log(1/\delta)+1}{m}
    \right\}
    \le\delta.
\]

Consequently, for every $\varepsilon>0$, $\delta\in(0,1)$,
and $\Delta\in[0,1]$,
\[
    m_{\mathrm{std}}^{(k)}
        ({\mathcal{R}},\varepsilon,\delta,\Delta)
    \le
    \left\lceil
        \frac{3\log|{\mathcal{R}}|/\log k+3\log(1/\delta)+1}
             {\varepsilon}
    \right\rceil.
\]

<!-- theorem-7: body_md -->
\label{thm:online-regret}
For integers $k\geq2$ and $L,T\geq1$,
Algorithm~\ref{alg:passk-general} satisfies, for every
$r\in{\mathcal{R}}$ and every sequence of contexts and demonstrations,
\[
    \sum_{t=1}^T
    \left[r(x_t,y_t)-\max_{i\in[k]}r(x_t,\hat y_t^{(i)})\right]
    \leq\frac{\log|{\mathcal{R}}|}{\log(1+k/4)}+\frac TL
    \leq\frac{2\log|{\mathcal{R}}|}{\log k}+\frac TL.
\]
If all rewards are binary and $L=1$, the terms $T/L$ may be omitted.

<!-- proof-1: body_md -->
Fix $r\in{\mathcal{R}}$ and a sequence of contexts and demonstrations.
If $|{\mathcal{Y}}|<k$, the learner returns every response, so its regret
is nonpositive. Assume henceforth that $|{\mathcal{Y}}|\geq k$.

\paragraph{Reduction to quantized regret.}
To control the regret, first replace rewards by their quantized
values. Define the signed quantized regret at round $t$ by
\begin{equation}
\label{eq:quantized_round_regret}
    \ell_t^r:=q_L\bigl(r(x_t,y_t)\bigr)
      -q_L\!\left(\max_{i\in[k]}r(x_t,\hat y_t^{(i)})\right).
\end{equation}
The rounding property of~equation~\ref{eq:threshold_quantization} gives
\begin{equation}
\label{eq:online_regret_rounding}
    \sum_{t=1}^T
    \left[r(x_t,y_t)-\max_{i\in[k]}r(x_t,\hat y_t^{(i)})\right]
    \leq\sum_{t=1}^T\ell_t^r+\frac TL.
\end{equation}
It therefore suffices to show that
$\sum_{t=1}^T\ell_t^r\leq\log|{\mathcal{R}}|/\log(1+k/4)$.
We relate this sum to the average log-weight associated with $r$,
then use greedy coverage to bound the total weight available to
its thresholds.
\paragraph{From quantized regret to log-weights.}
For the fixed reward $r$, the thresholds covered by either the
demonstration or the learner form an initial segment of
$u_1,\ldots,u_L$. These two segments are therefore nested, and
their shared thresholds cancel in the difference defining
$\ell_t^r$. Thus, $[\ell_t^r]_+$ is the fraction of thresholds
covered only by the demonstration, while $[-\ell_t^r]_+$ is
the fraction covered only by the learner, where
$[v]_+:=\max\{v,0\}$.

The update rule increases the log-weight of each threshold
covered only by the demonstration by $\log(1+k/4)$ and decreases
that of each threshold covered only by the learner by
$\log(4/3)$. All other log-weights remain unchanged.
Averaging these increments over the $L$ thresholds and summing
over rounds gives, since $w^{(1)}(r,j)=1$,
\begin{equation}
\label{eq:log_weight_regret}
\begin{aligned}
    \frac1L\sum_{j=1}^L\log w^{(T+1)}(r,j)
    &=\sum_{t=1}^T\frac1L\sum_{j=1}^L
       \log\frac{w^{(t+1)}(r,j)}{w^{(t)}(r,j)}\\
    &=\sum_{t=1}^T\left\{
       \log(1+k/4)[\ell_t^r]_+
       -\log(4/3)[-\ell_t^r]_+\right\}\\
    &\geq\log(1+k/4)\sum_{t=1}^T\ell_t^r.
\end{aligned}
\end{equation}
For the last inequality, use
$\ell_t^r=[\ell_t^r]_+-[-\ell_t^r]_+$ and
$\log(1+k/4)\geq\log(4/3)$ for $k\geq2$.
In particular, when $\ell_t^r<0$, the average log-weight decreases
by at most $\log(1+k/4)|\ell_t^r|$, so the same inequality
also holds on rounds when the learner outperforms the demonstrator.
It remains to bound the average final log-weight by $\log|{\mathcal{R}}|$.
                                                   
                                                             
                                                             
                                                               
                                                             
                         
                  
                              
                 
                                            
                           
                                
                                          
                                           
               
                
                                                              
                                                                 
                                                                  
                                                            

\paragraph{Controlling the total weight.}
An upper bound on the total weight
$W_t:=w^{(t)}({\mathcal{S}}_L)$ also bounds the arithmetic mean of the
threshold weights for any fixed reward. The greedy selection rule
gives the comparison
\begin{equation}
\label{eq:greedy_weight_comparison}
    w^{(t)}(U_t\setminus D_t)
    \geq k\,w^{(t)}(D_t\setminus U_t),
\end{equation}
whose proof is given in Appendix~\ref{appen:standard_passk_upper}.
It ensures that the weight removed from pairs covered only by
the learner offsets the weight added to pairs covered only by
the demonstration. Indeed, the update gives
\begin{equation}
\label{eq:online_total_weight}
    W_{t+1}-W_t
    =\frac{k}{4}w^{(t)}(D_t\setminus U_t)
      -\frac14w^{(t)}(U_t\setminus D_t)\leq0.
\end{equation}
Consequently, $W_{T+1}\leq W_1=L|{\mathcal{R}}|$. Applying the
arithmetic--geometric mean inequality to the $L$ threshold
weights associated with $r$ now yields
\begin{equation}
\label{eq:online_log_weight_bound}
    \frac1L\sum_{j=1}^L\log w^{(T+1)}(r,j)
    \leq\log\!\left(\frac1L\sum_{j=1}^Lw^{(T+1)}(r,j)\right)
    \leq\log|{\mathcal{R}}|.
\end{equation}
Averaging over the thresholds removes the factor $L$ from the
total-weight bound, avoiding a $\log L$ contribution to regret.

\paragraph{Concluding the regret bound.}
Combining~equation~\ref{eq:log_weight_regret} with
equation~\ref{eq:online_log_weight_bound} gives the required bound on
$\sum_t\ell_t^r$. Substituting it into
equation~\ref{eq:online_regret_rounding} proves the first inequality
in the theorem. The identity
$1+k/4-\sqrt{k}=(\sqrt{k}-2)^2/4\geq0$ gives
$\log(1+k/4)\geq\tfrac12\log k$ and hence the second inequality.
For binary rewards with $L=1$, quantization is exact and the
rounding term is absent. This completes the proof.

<!-- lemma-1: body_md -->
\label{lem:exponential_moment}
For the learner above, every integer $m\geq1$ and $L\geq1$,
and every reward $r\in{\mathcal{R}}$ fixed before sampling,
\begin{equation}
\label{eq:conditional_excess_moment}
    \mathbb E_S\exp\!\left\{\frac13\sum_{t=1}^m g_t^r\right\}
    \leq\exp\!\left\{
       \frac{\log(3/2)}{\log(1+k/4)}\log|{\mathcal{R}}|\right\}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Fix a round $t$. If $|{\mathcal{Y}}|<k$, the learner returns every
response, so $D_t\subseteq U_t$ and the inequality is immediate.
Assume henceforth that $|{\mathcal{Y}}|\geq k$.

Let $\mathcal U_0:=\varnothing$ and, for $i\in[k]$, define
\[
    \mathcal U_i:=\bigcup_{s=1}^i A_t^{\hat y_t^{(s)}},
    \qquad
    G_i:=\mathcal U_i\setminus\mathcal U_{i-1}.
\]
Thus, $\mathcal U_i$ is the set of pairs covered by the first
$i$ responses, $G_i$ consists of the pairs newly covered at
step $i$, and $\mathcal U_k=U_t$.
The greedy rule gives
\[
    w^{(t)}(G_i)
    \geq w^{(t)}(D_t\setminus\mathcal U_{i-1}).
\]
Indeed, if $y_t$ has not yet been selected, it is an available
candidate whose uncovered weight cannot exceed that of the
greedy choice. If it has already been selected, then
$D_t\subseteq\mathcal U_{i-1}$, so the right-hand side is zero.

To isolate the newly covered weight outside $D_t$, subtract
$w^{(t)}(G_i\cap D_t)$ from both sides. Since
$D_t\setminus\mathcal U_{i-1}$ is the disjoint union of
$G_i\cap D_t$ and $D_t\setminus\mathcal U_i$, we obtain
\[
\begin{aligned}
    w^{(t)}(G_i\setminus D_t)
    &=w^{(t)}(G_i)-w^{(t)}(G_i\cap D_t)\\
    &\geq w^{(t)}(D_t\setminus\mathcal U_{i-1})
          -w^{(t)}(G_i\cap D_t)\\
    &=w^{(t)}(D_t\setminus\mathcal U_i)\\
    &\geq w^{(t)}(D_t\setminus U_t).
\end{aligned}
\]
The last inequality uses $\mathcal U_i\subseteq U_t$:
the weight still uncovered after step $i$ is at least the
weight left uncovered after all $k$ steps.
Finally, the sets $G_i\setminus D_t$ are pairwise disjoint
and have union $U_t\setminus D_t$. Summing over $i$ therefore gives
\[
    w^{(t)}(U_t\setminus D_t)
    =\sum_{i=1}^k w^{(t)}(G_i\setminus D_t)
    \geq k\,w^{(t)}(D_t\setminus U_t),
\]
as claimed.

<!-- proof-3: body_md -->
Fix a reward $r\in{\mathcal{R}}$ before sampling. If $|{\mathcal{Y}}|<k$, the
learner returns every response, so $\ell_t^r\leq0$ and
$g_t^r\leq0$ for every $t$. The left-hand side
of~equation~\ref{eq:conditional_excess_moment} is then at most one,
and the claim follows. Assume henceforth that $|{\mathcal{Y}}|\geq k$.

We first express the weight bound in a form suited to controlling
the exponential moment. Set
\[
    \theta_k:=\frac{\log(3/2)}{\log(1+k/4)}\in(0,1],
\]
where the inclusion follows from $k\geq2$, and define the
scaled average log-weight increment
\begin{equation}
\label{eq:scaled_log_increment}
    z_t^r:=\log(3/2)[\ell_t^r]_+
           -\theta_k\log(4/3)[-\ell_t^r]_+.
\end{equation}
By the log-weight identity in~equation~\ref{eq:log_weight_regret} and
the bound~equation~\ref{eq:online_log_weight_bound}, every realized
sample satisfies
\[
    \sum_{t=1}^m z_t^r
    =\frac{\theta_k}{L}\sum_{j=1}^L
        \log w^{(m+1)}(r,j)
    \leq\theta_k\log|{\mathcal{R}}|.
\]
It remains to relate these increments to the conditional
shortfalls $g_t^r=\mathbb E[\ell_t^r\mid{\mathcal{F}}_{t-1}]$.

The scaling is chosen so that $e^{-z_t^r}$ admits a linear
upper bound in $\ell_t^r$. Indeed, for $u\in[0,1]$, convexity
bounds each exponential by the line joining its endpoint values:
\[
\begin{aligned}
    (2/3)^u
    &\leq(1-u)+\frac23u
      =1-\frac u3,\\
    (4/3)^{\theta_k u}
    &\leq(1-u)+u(4/3)^{\theta_k}
      \leq1+\frac u3.
\end{aligned}
\]
The last inequality uses $\theta_k\leq1$.
If $\ell_t^r\geq0$, apply the first bound with $u=\ell_t^r$;
if $\ell_t^r<0$, apply the second with $u=-\ell_t^r$.
Since $|\ell_t^r|\leq1$, both cases give
\begin{equation}
\label{eq:one_step_exponential}
\begin{aligned}
    e^{-z_t^r}
    &\leq1-\frac{\ell_t^r}{3},\\
    \mathbb E[e^{-z_t^r}\mid{\mathcal{F}}_{t-1}]
    &\leq1-\frac{g_t^r}{3}
      \leq e^{-g_t^r/3}.
\end{aligned}
\end{equation}
Here the second line follows by conditional expectation and
$1-u\leq e^{-u}$.

To combine these conditional bounds across rounds, define
\[
    K_0:=1,\qquad
    K_t:=\exp\!\left\{
        \sum_{s=1}^t\left(\frac{g_s^r}{3}-z_s^r\right)
    \right\},
    \qquad t\in[m].
\]
This process is adapted and integrable, since
$|g_t^r|\leq1$ and $|z_t^r|\leq\log(3/2)$.
Moreover, $g_t^r$ and $K_{t-1}$ are
${\mathcal{F}}_{t-1}$-measurable, so~equation~\ref{eq:one_step_exponential} gives
\[
    \mathbb E[K_t\mid{\mathcal{F}}_{t-1}]
    =K_{t-1}e^{g_t^r/3}
       \mathbb E[e^{-z_t^r}\mid{\mathcal{F}}_{t-1}]
    \leq K_{t-1}.
\]
Thus $(K_t)_{t=0}^m$ is a nonnegative supermartingale, and
iterated expectation yields $\mathbb E_S K_m\leq1$.

Finally, the pathwise bound on $\sum_t z_t^r$ implies
\[
\begin{aligned}
    \mathbb E_S\exp\!\left\{\frac13\sum_{t=1}^m g_t^r\right\}
    &=\mathbb E_S\!\left[
        K_m\exp\!\left\{\sum_{t=1}^m z_t^r\right\}
      \right]\\
    &\leq
      \exp\!\left\{\theta_k\log|{\mathcal{R}}|\right\}
      \mathbb E_S K_m\\
    &\leq
      \exp\!\left\{
        \frac{\log(3/2)}{\log(1+k/4)}\log|{\mathcal{R}}|
      \right\}.
\end{aligned}
\]
This proves~equation~\ref{eq:conditional_excess_moment}.

The lower bound has two components. We use the same context space,
response space, and reward class for both. The first construction
distributes the difficulty across several contexts and gives the
dependence on the reward-class size. The second concentrates the
difficulty on one context and gives the dependence on the confidence
level. In both constructions, an unobserved context has an unknown
correct response among $2k$ possibilities, while a pass@$k$ policy can
include at most $k$ distinct responses. We prove the resulting
probability bounds separately and then combine them. Every demonstrator
used below is optimal for its selected evaluation reward.

\paragraph{Common reward class.}
Fix integers $k\geq1$ and $N\geq2k$, and set
\[
    q:=2k,
    \qquad
    d:=\left\lfloor\frac{\log N}{\log q}\right\rfloor\geq1,
    \qquad
    {\mathcal{X}}_0:=\{0,1,\ldots,d\},
    \qquad
    {\mathcal{Y}}:=\{0,1,\ldots,q\}.
\]
We allow $k=1$ so that the same construction also applies in
Appendix~\ref{appen:standard_pass1_lower}.
For each $\theta\in[q]^d$, define
\[
    f_\theta(0):=0,
    \qquad
    f_\theta(j):=\theta_j,\quad j\in[d],
    \qquad
    r_\theta(x,y):=\mathds{1}\{y=f_\theta(x)\}.
\]
These give $q^d$ distinct core rewards on ${\mathcal{X}}_0\times{\mathcal{Y}}$.

To obtain exactly $N$ rewards, let $t:=N-q^d$ and introduce
auxiliary contexts ${\mathcal{X}}_{\mathrm{aux}}:=\{z_1,\ldots,z_t\}$,
disjoint from ${\mathcal{X}}_0$. Set
\[
    {\mathcal{X}}:={\mathcal{X}}_0\sqcup{\mathcal{X}}_{\mathrm{aux}},
    \qquad
    f_\theta(z_s):=0,\quad s\in[t],
\]
and extend each $r_\theta$ using the same graph-indicator definition.
For each $s\in[t]$, define an additional target function and reward by
\[
    \widetilde f_s(x):=
    \begin{cases}
        1,&x\in[d]\cup\{z_s\},\\
        0,&\text{otherwise},
    \end{cases}
    \qquad
    \widetilde r_s(x,y):=\mathds{1}\{y=\widetilde f_s(x)\}.
\]
The common reward class is
\[
    {\mathcal{R}}:=\{r_\theta:\theta\in[q]^d\}
        \cup\{\widetilde r_s:s\in[t]\}.
\]
The added reward $\widetilde r_s$ differs from every core reward
at $(z_s,1)$, and two added rewards differ at their corresponding
auxiliary contexts. Hence $|{\mathcal{R}}|=q^d+t=N$. If $t=0$, the auxiliary
context set and the added reward family are empty. Every reward is
binary and has optimal value one. All distributions below assign
zero mass to the auxiliary contexts, and the spaces and reward class
depend only on $k$ and $N$.

\paragraph{Two statements to prove.}
Fix $m\geq1$, $0<\delta<1/16$, and an arbitrary, possibly randomized
learner. First, we will find a context distribution ${\mathcal{D}}$, a fixed
evaluation reward $r^\star\in{\mathcal{R}}$, and a demonstrator $\pi_{\mathrm{demo}}$ optimal
for $r^\star$ such that
\begin{equation}
\label{eq:standard_class_lower_component}
    \mathbb P\!\left\{
        {\mathcal{E}}_{r^\star}^k(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{32}\min\!\left\{
            1,\frac{\log N}{m\log(2k)}
        \right\}
    \right\}\geq\frac17>2\delta.
\end{equation}
Second, on the same reward class, we will find a context distribution
and an optimal-demonstrator instance satisfying
\begin{equation}
\label{eq:standard_confidence_lower_component}
    \mathbb P\!\left\{
        {\mathcal{E}}_{r^\star}^k(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{32}\min\!\left\{
            1,\frac{\log(1/\delta)}m
        \right\}
    \right\}\geq\frac{8\delta}{3}>2\delta.
\end{equation}
In each statement, $\widehat\pi={\mathcal{A}}_k({\mathcal{R}},S)$ is the policy returned
from $m$ demonstrations under the selected instance. The instance may
depend on the learner, but its distribution, demonstrator, and evaluation
reward are fixed before the sample is drawn. All probabilities include
the learner's training randomization; prediction-time randomness is
averaged in the policy values. Since each selected demonstrator is
optimal, both statements apply for every budget $\Delta\in[0,1]$.

\paragraph{The pass@$k$ inclusion constraint.}
For a realized output policy, define, for $j\in{\mathcal{X}}$ and $a\in{\mathcal{Y}}$,
\[
    C_j(a):=
    \mathbb P_{\mathbf y\sim\widehat\pi(\cdot\mid j)}
    \{a\in\{y^{(1)},\ldots,y^{(k)}\}\}.
\]
This is the probability that the learner's response tuple includes
$a$. Since a tuple contains at most $k$ distinct responses,
\begin{equation}
\label{eq:standard_inclusion_budget}
    \sum_{a=1}^q C_j(a)
    =\mathbb E_{\mathbf y\sim\widehat\pi(\cdot\mid j)}
        \bigl|[q]\cap\{y^{(1)},\ldots,y^{(k)}\}\bigr|
    \leq k.
\end{equation}
The inequality holds for arbitrarily dependent responses within
the tuple. We use it in both constructions below.

\subsubsection{Dependence on the reward-class size}

\paragraph{Context distribution and demonstrators.}
Let $b:=\min\{m,d\}$ and put
\[
    {\mathcal{D}}(j):=\frac1{2m},\quad j\in[b],
    \qquad
    {\mathcal{D}}(0):=1-\frac b{2m},
\]
with zero mass elsewhere. Since $b\leq m$, the mass at context
$0$ is at least $1/2$. For each $\theta\in[q]^d$, let
$\pi_{\mathrm{demo}}^{(\theta)}$ return $f_\theta(x)$ deterministically,
and select $r_\theta$ as the evaluation reward.

\paragraph{Admissibility and the target bound.}
The selected reward equals one on every demonstrated response. Thus
$V_{r_\theta}(\pi_{\mathrm{demo}}^{(\theta)})=V_{r_\theta}^\star=1$,
and $\Delta_{r_\theta}=0$. Define the excess under this fixed reward by
\[
    Z_\theta:=
        {\mathcal{E}}_{r_\theta}^k(\widehat\pi,\pi_{\mathrm{demo}}^{(\theta)}).
\]
A tuple succeeds under $r_\theta$ exactly when it contains the
response $f_\theta(j)$ at context $j$. For any distribution supported
on ${\mathcal{X}}_0$, this gives
\begin{equation}
\label{eq:standard_passk_graph_loss}
    Z_\theta=\sum_{j=0}^d{\mathcal{D}}(j)
        \bigl[1-C_j(f_\theta(j))\bigr].
\end{equation}
Under the present distribution, the contribution from context $0$
is nonnegative, so
\[
    Z_\theta\geq W_\theta,
    \qquad
    W_\theta:=\frac1{2m}\sum_{j=1}^b[1-C_j(\theta_j)],
    \qquad
    0\leq W_\theta\leq\frac b{2m}.
\]
We will show that, for some fixed $\theta$, the quantity $W_\theta$
exceeds $b/(16m)$ with probability at least $1/7$.

\paragraph{Unobserved contexts and the inclusion constraint.}
For the analysis, draw $\theta$ uniformly from $[q]^d$, then draw
the sample from ${\mathcal{D}}$ and $\pi_{\mathrm{demo}}^{(\theta)}$, and run
the learner. In this paragraph, $\mathbb P$ and $\mathbb E$ include
this auxiliary draw of $\theta$ as well as the sample and training
randomization. For each $j\in[b]$, let $U_j$ be the event that
context $j$ is absent from the sample. Bernoulli's inequality gives
\[
    \mathbb P(U_j)
    =\left(1-\frac1{2m}\right)^m
    \geq1-m\frac1{2m}=\frac12.
\]
On $U_j$, the sample reveals no information about $\theta_j$.
Because the coordinates of $\theta$ are independent and uniform,
$\theta_j$ remains uniform on $[q]$ conditional on $S$.
The learner's independent randomization supplies no further information,
so the same holds conditional on $S$ and $\widehat\pi$. Consequently,
on $U_j$,
\[
    \mathbb E[1-C_j(\theta_j)\mid S,\widehat\pi]
    =1-\frac1q\sum_{a=1}^q C_j(a)
    \geq1-\frac{k}{q}=\frac12,
\]
where the inequality is \cref{eq:standard_inclusion_budget}.
Writing $W:=W_\theta$, we obtain
\begin{align*}
    \mathbb E W
    &=\frac1{2m}\sum_{j=1}^b
        \mathbb E[1-C_j(\theta_j)]\\
    &\geq\frac1{2m}\sum_{j=1}^b
        \mathbb E\!\left[
            \mathds{1}\{U_j\}\,
            \mathbb E[1-C_j(\theta_j)\mid S,\widehat\pi]
        \right]\\
    &\geq\frac1{4m}\sum_{j=1}^b\mathbb P(U_j)
     \geq\frac b{8m}.
\end{align*}
The first inequality keeps only the nonnegative contributions from
unobserved contexts; the second uses the conditional lower bound above.

\paragraph{A probability bound for a fixed instance.}
Since $0\leq W\leq b/(2m)$, splitting its expectation at $b/(16m)$
gives
\[
    \frac b{8m}
    \leq\mathbb E W
    \leq \frac b{16m}
        +\frac{7b}{16m}\mathbb P\!\left\{W>\frac b{16m}\right\}.
\]
It follows that $\mathbb P\{W>b/(16m)\}\geq1/7$.
Let $\mathbb P_\theta$ denote probability over the sample and training
randomization when the parameter is fixed at $\theta$. Averaging over
the auxiliary parameter gives
\[
    \frac1{q^d}\sum_{\theta\in[q]^d}
        \mathbb P_\theta\!\left\{W_\theta>\frac b{16m}\right\}
    =\mathbb P\!\left\{W>\frac b{16m}\right\}
    \geq\frac17.
\]
Therefore at least one fixed $\theta$ satisfies this probability bound.
For that same parameter, $Z_\theta\geq W_\theta$. Moreover,
$d\geq\log N/(2\log q)$, so
\[
    \frac b{16m}
    =\frac1{16}\min\!\left\{1,\frac dm\right\}
    \geq\frac1{32}\min\!\left\{1,\frac{\log N}{m\log q}\right\}.
\]
Taking $r^\star=r_\theta$ and $\pi_{\mathrm{demo}}=\pi_{\mathrm{demo}}^{(\theta)}$
proves \cref{eq:standard_class_lower_component} because $q=2k$.
The selected parameter is fixed before the actual training sample
is drawn.

\subsubsection{Dependence on the confidence level}

\paragraph{Context distribution and demonstrators.}
For this construction, only contexts $0$ and $1$ have positive mass.
Set
\[
    \eta:=\min\!\left\{
        \frac12,\frac{\log(1/(8\delta))}{2m}
    \right\},
    \qquad
    {\mathcal{D}}(1):=\eta,
    \qquad
    {\mathcal{D}}(0):=1-\eta,
\]
with zero mass elsewhere. For each $\theta\in[q]^d$, select
$r_\theta$ and use $\pi_{\mathrm{demo}}^{(\theta)}$. The demonstrator
returns zero at context $0$ and $\theta_1$ at context $1$.

\paragraph{Admissibility and the target bound.}
Each demonstrator is again optimal for its selected reward, so
$\Delta_{r_\theta}=0$. Write $Z_\theta$ for the excess under
the new context distribution. By \cref{eq:standard_passk_graph_loss},
\[
    Z_\theta=(1-\eta)[1-C_0(0)]+\eta[1-C_1(\theta_1)]
        \geq\eta H_\theta,
    \qquad
    H_\theta:=1-C_1(\theta_1)\in[0,1].
\]
We will find a fixed $\theta$ for which $H_\theta>1/4$ with
probability at least $8\delta/3$. This implies that $Z_\theta>\eta/4$
with at least the same probability.

\paragraph{The event of receiving no informative demonstration.}
Again draw $\Theta$ uniformly from $[q]^d$ for the analysis, and let
$E$ be the event that context $1$ is absent from the sample. On $E$,
every demonstration is the pair $(0,0)$ and reveals nothing about
$\Theta_1$. Since $0<\eta\leq1/2$ and
$\log(1-\eta)\geq-2\eta$,
\[
    \mathbb P(E)=(1-\eta)^m
    \geq e^{-2m\eta}\geq8\delta.
\]
The last inequality follows from
$2m\eta\leq\log(1/(8\delta))$.

\paragraph{A probability bound for a fixed instance.}
On $E$, the label $\theta_1$ is uniform on $[q]$ conditional on
$S,\widehat\pi$. Thus the same inclusion argument as above gives
\[
    \mathbb E[H_\theta\mid S,\widehat\pi]
    =1-\frac1q\sum_{a=1}^q C_1(a)\geq\frac12.
\]
Since $0\leq H_\theta\leq1$, we also have, on $E$,
\[
    \frac12
    \leq\mathbb E[H_\theta\mid S,\widehat\pi]
    \leq\frac14+\frac34
        \mathbb P\{H_\theta>1/4\mid S,\widehat\pi\}.
\]
The conditional probability on the right is therefore at least $1/3$.
Using $Z_\theta\geq\eta H_\theta$ and integrating over $E$, we obtain
\begin{align*}
    \mathbb P\{Z_\theta>\eta/4\}
    &\geq\mathbb P(E\cap\{H_\theta>1/4\})\\
    &=\mathbb E\!\left[
        \mathds{1}\{E\}\,
        \mathbb P\{H_\theta>1/4\mid S,\widehat\pi\}
    \right]\\
    &\geq\frac13\mathbb P(E)\geq\frac{8\delta}{3}.
\end{align*}
Averaging over $\theta$ now selects a fixed $\theta$ for which
$\mathbb P_\theta\{Z_\theta>\eta/4\}\geq8\delta/3$.
Because $\delta<1/16$, we have
$\log(1/(8\delta))\geq\tfrac14\log(1/\delta)$, and hence
\[
    \frac\eta4
    =\min\!\left\{\frac18,\frac{\log(1/(8\delta))}{8m}\right\}
    \geq\frac1{32}\min\!\left\{1,\frac{\log(1/\delta)}m\right\}.
\]
This proves \cref{eq:standard_confidence_lower_component} for the
fixed evaluation reward $r_\theta$ and its optimal demonstrator.

\paragraph{Combining the two statements.}
Fix any learner and write
\[
    A:=\frac{\log N}{\log(2k)},
    \qquad
    B:=\log\frac1\delta.
\]
If $A\geq B$, use the first construction; otherwise, use the second.
Both give failure probability greater than $2\delta$, so the selected
optimal-demonstrator instance satisfies
\[
    \mathbb P\!\left\{
        {\mathcal{E}}_{r^\star}^k(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{32}\min\!\left\{1,\frac{\max\{A,B\}}m\right\}
    \right\}\geq2\delta.
\]
Since $\max\{A,B\}\geq(A+B)/2$,
\[
    \frac1{32}\min\!\left\{1,\frac{\max\{A,B\}}m\right\}
    \geq\frac1{64}\min\!\left\{1,\frac{A+B}m\right\}.
\]
Lowering the threshold can only increase the probability of exceeding
it. Substituting the definitions of $A$ and $B$ gives
\begin{equation}
\label{eq:standard_passk_lower_tail}
    \mathbb P\!\left\{
        {\mathcal{E}}_{r^\star}^k(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{64}\min\!\left\{
            1,\frac1m\left(
                \frac{\log N}{\log(2k)}+\log\frac1\delta
            \right)
        \right\}
    \right\}\geq2\delta.
\end{equation}
Both constructions use the same class of exactly $N$ rewards. For each
learner, the selected distribution, demonstrator, and evaluation reward
are fixed before the sample and training randomization are drawn.
Thus \cref{eq:standard_passk_lower_tail} is a lower bound for standard
evaluation under a fixed reward.

\paragraph{Sample complexity.}
Let $0<\varepsilon\leq1/128$. For every integer
\[
    1\leq m<\frac1{64\varepsilon}\left(
        \frac{\log N}{\log(2k)}+\log\frac1\delta
    \right),
\]
both entries of the minimum in \cref{eq:standard_passk_lower_tail}
are strictly larger than $64\varepsilon$: the first is one, whereas
$64\varepsilon\leq1/2$, and the second exceeds $64\varepsilon$
by the displayed condition on $m$. The excess threshold is therefore
strictly larger than $\varepsilon$, and every learner fails the PAC
requirement at each such sample size.

The upper limit on $m$ above is greater than one, since
$N\geq2k$, $\delta<1/16$, and $\varepsilon\leq1/128$.
In particular, every learner fails at $m=1$. A sample-complexity
threshold of zero is also impossible, because it would require success
at every sample size, including $m=1$.
All selected demonstrators have $\Delta_{r^\star}=0$ and are therefore
admissible for every $\Delta\in[0,1]$. By
\cref{def:standard_pac_complexity}, for every $k\geq1$,
\[
    m_{\mathrm{std}}^{(k)}({\mathcal{R}},\varepsilon,\delta,\Delta)
    \geq\frac1{64\varepsilon}\left(
        \frac{\log N}{\log(2k)}+\log\frac1\delta
    \right).
\]
Finally, when $k\geq2$, the inequality $\log(2k)\leq2\log k$
implies
\[
    m_{\mathrm{std}}^{(k)}({\mathcal{R}},\varepsilon,\delta,\Delta)
    \geq\frac1{128\varepsilon}\left(
        \frac{\log N}{\log k}+\log\frac1\delta
    \right).
\]
This proves the theorem with $c=c_0=1/128$ for a reward class of
cardinality $N$ for every $N\geq2k$.

The lower bound has two components. We use the same context space,
response space, and reward class for both. The first component captures
the dependence on the reward-class size, and the second captures the
dependence on the confidence level. Each component contains a fast term,
obtained using optimal demonstrators, and a suboptimality-dependent term.
We use Assouad's lemma and list-decoding Fano for the reward-class
component, and Le Cam's method and the Bretagnolle--Huber inequality for
the confidence component. We prove the corresponding sample-size bounds
and then combine them.

\paragraph{Common reward class.}
Fix an integer $N\geq2$ and set
\[
    d:=\left\lfloor\frac{\log N}{\log2}\right\rfloor\geq1,
    \qquad
    {\mathcal{Y}}:=\{0,1,2\}.
\]
Take the class of exactly $N$ rewards constructed in
Appendix~\ref{appen:standard_passk_lower} with $k=1$. Write
${\mathcal{X}}_{\mathrm{base}}$ for its context space, including the auxiliary
contexts, and enumerate its rewards as
$r_1^{\mathrm{base}},\ldots,r_N^{\mathrm{base}}$.
On the core contexts $\{0,1,\ldots,d\}$, this class contains the rewards
\[
    r_\theta^{\mathrm{base}}(x,y)
        =\mathds{1}\{y=f_\theta(x)\},
    \qquad
    f_\theta(0)=0,
    \qquad
    f_\theta(i)=\theta_i,\quad i\in[d],
\]
indexed by $\theta\in\{1,2\}^d$. Every reward prefers response zero
at context $0$. The auxiliary contexts allow the class to have exactly
$N$ rewards even when $N$ is not a power of two.

Define the extended context space by
\[
    {\mathcal{X}}:={\mathcal{X}}_{\mathrm{base}}\sqcup\mathcal Z,
    \qquad
    \mathcal Z:=\{-1,+1\}^N,
\]
where $\sqcup$ denotes disjoint union. For each $j\in[N]$, define
the extended reward $r_j:{\mathcal{X}}\times{\mathcal{Y}}\to\{0,1\}$ by
\[
    r_j(x,y):=
    \begin{cases}
        r_j^{\mathrm{base}}(x,y),&x\in{\mathcal{X}}_{\mathrm{base}},\\[2pt]
        \displaystyle
        \frac{1+x_j}{2}\mathds{1}\{y=1\}
        +\frac{1-x_j}{2}\mathds{1}\{y=2\},&x\in\mathcal Z.
    \end{cases}
\]
Thus, at a new context $x\in\mathcal Z$, reward $r_j$ accepts
response $1$ if $x_j=+1$ and response $2$ if $x_j=-1$.
Let $\mu$ be the uniform distribution on $\mathcal Z$, and set
${\mathcal{R}}:=\{r_j:j\in[N]\}$. We continue to write $r_\theta$ for the
extension of a core reward. The rewards remain distinct on
${\mathcal{X}}_{\mathrm{base}}$, so $|{\mathcal{R}}|=N$. Every reward is binary and
has optimal value one. The spaces and reward class depend only on $N$,
and all distributions below assign zero mass to the auxiliary contexts.

\paragraph{Two statements to prove.}
Fix $\Delta\in[0,1]$, $0<\varepsilon,\delta\leq1/64$, and an
arbitrary learner using $m\geq1$ demonstrations. We will show that,
if this learner satisfies the standard PAC requirement for every
admissible instance on ${\mathcal{R}}$, then its sample size must satisfy
\begin{equation}
\label{eq:standard_pass1_class_requirement}
    m\geq\frac1{1088}\left[
        \frac{\log N}{\varepsilon}
        +\frac{\Delta}{\varepsilon^2}
            \log\!\left(1+\frac{N\varepsilon^2}
                                {(\Delta+\varepsilon)^2}\right)
    \right]
\end{equation}
and
\begin{equation}
\label{eq:standard_pass1_confidence_requirement}
    m\geq\frac1{256}\left[
        \frac{\log(1/\delta)}{\varepsilon}
        +\frac{\Delta\log(1/\delta)}{\varepsilon^2}
    \right].
\end{equation}
For each construction, admissibility means that the demonstrator's
suboptimality under the selected evaluation reward is at most $\Delta$.
We write $P_\eta$ for the law of one demonstration under an instance
indexed by $\eta$, and $\mathbb P_\eta$ and $\mathbb E_\eta$ for
probability and expectation under $P_\eta^{\otimes m}$, including
the learner's training randomization. Prediction-time randomness is
averaged in the policy values. Each instance and its evaluation reward
are fixed before the sample is drawn.

The next two subsections establish these statements. In the first
subsection, a parameter regime in which the reward-class term is small
will be covered by the two-point estimate proved in the second.

\subsubsection{Dependence on the reward-class size}

\paragraph{Context distribution and optimal demonstrators.}
For the fast term, put
\[
    p:=\frac{8\varepsilon}{d},
    \qquad
    {\mathcal{D}}(i):=p,\quad i\in[d],
    \qquad
    {\mathcal{D}}(0):=1-dp=1-8\varepsilon,
\]
with zero mass elsewhere. For each $\theta\in\{1,2\}^d$, let
$\pi_{\mathrm{demo}}^{(\theta)}$ return $f_\theta(x)$ deterministically
on the core contexts, and select $r_\theta$ as the evaluation reward.
The demonstrator may be defined arbitrarily outside the support of
${\mathcal{D}}$.

\paragraph{Admissibility and the target bound.}
The selected reward is one on every demonstrated response, so
$\Delta_{r_\theta}=0\leq\Delta$. For the returned policy, define
\[
    Z_\theta:=
        {\mathcal{E}}_{r_\theta}^1(\widehat\pi,\pi_{\mathrm{demo}}^{(\theta)}),
    \qquad
    \widehat\theta_i:=
    \begin{cases}
        1,&\widehat\pi(1\mid i)\geq1/2,\\
        2,&\widehat\pi(1\mid i)<1/2.
    \end{cases}
\]
Whenever $\widehat\theta_i\ne\theta_i$, the learner assigns
probability at most $1/2$ to the correct response. Writing
$H(\widehat\theta,\theta):=\sum_{i=1}^d
\mathds{1}\{\widehat\theta_i\ne\theta_i\}$, we obtain
\begin{equation}
    Z_\theta
    =(1-dp)\bigl[1-\widehat\pi(0\mid0)\bigr]
        +p\sum_{i=1}^d\bigl[1-\widehat\pi(\theta_i\mid i)\bigr]\geq\frac p2H(\widehat\theta,\theta).
        \label{eq:standard_pass1_assouad_reduction}
\end{equation}
It therefore suffices to find a fixed $\theta$ for which the Hamming
error exceeds $d/4$ with probability greater than $\delta$: on this
event, $Z_\theta>pd/8=\varepsilon$.

\paragraph{Proof using Assouad's lemma.}
Let $\theta^{(i)}$ differ from $\theta$ only at coordinate $i$.
The sample laws $P_\theta^{\otimes m}$ and
$P_{\theta^{(i)}}^{\otimes m}$ agree when context $i$ is absent
and have disjoint supports otherwise. Hence
\[
    \operatorname{TV}(P_\theta^{\otimes m},P_{\theta^{(i)}}^{\otimes m})
    =1-(1-p)^m\leq mp.
\]
Adjoining the learner's independent randomization leaves this distance
unchanged. If $m\leq d/(32\varepsilon)$, then $mp\leq1/4$.
Assouad's lemma in Hamming loss
\citep[Theorem~2.12]{tsybakov2009introduction} gives
\[
    \max_{\theta\in\{1,2\}^d}
        \mathbb E_\theta H(\widehat\theta,\theta)
    \geq\frac d2\left[
        1-\max_{\theta,i}
            \operatorname{TV}(P_\theta^{\otimes m},P_{\theta^{(i)}}^{\otimes m})
    \right]
    \geq\frac{3d}{8}.
\]
Choose a fixed $\theta$ attaining this lower bound and write
$H=H(\widehat\theta,\theta)$. Since $0\leq H\leq d$,
\[
    \frac{3d}{8}
    \leq\mathbb E_\theta H
    \leq\frac d4+\frac{3d}{4}\mathbb P_\theta\{H>d/4\}.
\]
Thus $\mathbb P_\theta\{H>d/4\}\geq1/6>\delta$, and
\cref{eq:standard_pass1_assouad_reduction} implies that the learner
fails the PAC requirement on this fixed instance. Consequently,
every successful sample size must satisfy
\begin{equation}
\label{eq:standard_pass1_fast_class_assouad}
    m\geq\frac{d}{32\varepsilon}
      \geq\frac{\log N}{64\varepsilon},
\end{equation}
where we used $d\geq\log N/(2\log2)$ and $\log2\leq1$.

\paragraph{Context distribution and suboptimal demonstrators.}
For the suboptimality-dependent term, suppose $\Delta\geq8\varepsilon$
and set $h:=2\varepsilon/\Delta\leq1/4$. Put mass $1-\Delta$
on context $0$ and distribute mass $\Delta$ uniformly over $\mathcal Z$:
\[
    {\mathcal{D}}(0):=1-\Delta,
    \qquad
    {\mathcal{D}}(z):=\Delta\mu(z),\quad z\in\mathcal Z,
\]
with zero mass elsewhere. For each $j\in[N]$, select $r_j$ as the
evaluation reward and define
\[
    \pi_{\mathrm{demo}}^{(j)}(0\mid0)=1,
    \qquad
    \pi_{\mathrm{demo}}^{(j)}(1\mid z)=\frac12+h z_j,
    \qquad
    \pi_{\mathrm{demo}}^{(j)}(2\mid z)=\frac12-h z_j.
\]
Thus the demonstrator slightly favors the response accepted by $r_j$
at every context in $\mathcal Z$.

\paragraph{Admissibility and the target bound.}
Under $r_j$, the demonstrator receives expected reward $1/2+h$ at
every $z\in\mathcal Z$ and reward one at context $0$. Therefore
\[
    \Delta_{r_j}
    =1-\left[(1-\Delta)+\Delta\left(\frac12+h\right)\right]
    =\Delta\left(\frac12-h\right)\leq\Delta.
\]
For the returned policy, write
\[
    Z_j:={\mathcal{E}}_{r_j}^1(\widehat\pi,\pi_{\mathrm{demo}}^{(j)}),
    \qquad
    \widehat f(z):=\widehat\pi(1\mid z)-\widehat\pi(2\mid z),
    \qquad
    c_j:=\mathbb E_{z\sim\mu}[\widehat f(z)z_j].
\]
Here $|\widehat f(z)|\leq1$. The learner's expected reward at $z$ is
\[
    \frac{\widehat\pi(1\mid z)+\widehat\pi(2\mid z)
              +\widehat f(z)z_j}{2}
    \leq\frac{1+\widehat f(z)z_j}{2}.
\]
Since the excess contribution from context $0$ is nonnegative,
\[
    Z_j\geq\Delta\left(h-\frac{c_j}{2}\right).
\]
Because $\varepsilon=\Delta h/2$, the event $Z_j\leq\varepsilon$
requires $c_j\geq h$. Successful learning therefore places the true
reward index in the candidate list
\[
    \mathcal L(\widehat\pi):=\{j\in[N]:c_j\geq h\}.
\]

\paragraph{Proof using Fano's inequality with list decoding.}
A policy may perform well for several rewards, so we bound the number
of indices in this list. The coordinate functions $z\mapsto z_j$
are orthonormal in $L^2(\mu)$. Bessel's inequality gives
\begin{equation}
\label{eq:standard_pass1_list_size}
    h^2|\mathcal L(\widehat\pi)|
    \leq\sum_{j=1}^N c_j^2
    \leq\mathbb E_{z\sim\mu}[\widehat f(z)^2]
    \leq1.
\end{equation}
Every realized output policy therefore yields a list of at most
$\lfloor h^{-2}\rfloor$ candidate rewards.

                                                 
                                                                        
                                                                    
                                                                  
                  
                                     
                                        
                
                                                           
                                           
                                                    
                                               
                                                           
                                              
                                                                
                                                                      
                                                                       
                                                


Draw $J$ uniformly from $[N]$ and, conditional on $J=j$, draw
$S\sim P_j^{\otimes m}$. Let $U$ be the learner's independent
training randomization and set $W:=(S,U)$. The output policy and its
candidate list are functions of $W$. Since successful learning requires
the correct index to belong to the list, the PAC requirement gives
\[
    \mathbb P\{J\notin\mathcal L(\widehat\pi)\}
    \leq\frac1N\sum_{j=1}^N\mathbb P_j\{Z_j>\varepsilon\}
    \leq\delta.
\]
Let $P_0$ be the one-demonstration law obtained by replacing $h$ with
zero. The auxiliary-distribution bound on mutual information
\citep[Lemma~4]{scarlett2019introductory} yields
\begin{align*}
    I(J;W)=I(J;S)
    &\leq\frac1N\sum_{j=1}^N
        D_{\mathrm{KL}}(P_j^{\otimes m}\|P_0^{\otimes m})\\
    &\overset{(a)}{=}\frac{m\Delta}{N}\sum_{j=1}^N
        \mathbb E_{z\sim\mu}\!\left[
            \operatorname{kl}\left(\frac12+hz_j,\frac12\right)
        \right]\\
    &\overset{(b)}{\leq}4m\Delta h^2.
\end{align*}
Step $(a)$ uses the product and common-context-marginal identities from
\citet[Exercises~14.10 and~14.12]{lattimore2019bandit}: the laws agree at context $0$ and differ
only on $\mathcal Z$. Step $(b)$ uses the identity from~\citet[Appendix~A.2]{de2021bandits}.

Suppose first that $Nh^2\geq4$, and set
$L:=\lfloor h^{-2}\rfloor<N$. The candidate list has at most
$L$ indices, and its error probability
$e:=\mathbb P\{J\notin\mathcal L(\widehat\pi)\}$ satisfies
$e\leq\delta$. Consequently,
\begin{align*}
    4m\Delta h^2
    &\geq I(J;W)\\
    &\overset{(a)}{\geq}(1-e)\log\frac NL-\log2\\
    &\geq(1-\delta)\log\frac NL-\log2\\
    &\geq(1-\delta)\log(Nh^2)-\log2\\
    &\geq\frac14\log(Nh^2).
\end{align*}
Here step $(a)$ is due to Theorem~\ref{thm:list_decoding_fano}.
The next two inequalities use $e\leq\delta$ and $L\leq h^{-2}$,
respectively. The last inequality follows from $Nh^2\geq4$
and $\delta\leq1/64$.
                                      
                                   
                                                                     
                
                  
                                         
                                       
                             
              
                                                      
Substituting $h^2=4\varepsilon^2/\Delta^2$, we obtain
\[
    m\geq\frac{\Delta}{64\varepsilon^2}\log(Nh^2)
      \geq\frac{\Delta}{64\varepsilon^2}
          \log\!\left(1+\frac{N\varepsilon^2}
                              {(\Delta+\varepsilon)^2}\right),
\]
where the second inequality follows from
\[
    1+\frac{N\varepsilon^2}{(\Delta+\varepsilon)^2}
    \leq1+\frac{Nh^2}{4}\leq Nh^2.
\]

If $Nh^2<4$, the logarithm in the target bound is at most $\log2$.
The two-point estimate \cref{eq:standard_pass1_slow_component},
proved in the next subsection, then gives
\[
    m\geq\frac{\Delta\log(1/\delta)}{128\varepsilon^2}
      \geq\frac{\Delta}{128\varepsilon^2}
          \log\!\left(1+\frac{N\varepsilon^2}
                              {(\Delta+\varepsilon)^2}\right).
\]
Thus, in both regimes,
\begin{equation}
\label{eq:standard_pass1_class_component}
    m\geq\frac{\Delta}{128\varepsilon^2}
        \log\!\left(1+\frac{N\varepsilon^2}
                            {(\Delta+\varepsilon)^2}\right),
    \qquad \Delta\geq8\varepsilon.
\end{equation}

\paragraph{Conclusion of the reward-class statement.}
When $\Delta\geq8\varepsilon$, combining
\cref{eq:standard_pass1_fast_class_assouad,eq:standard_pass1_class_component}
gives
\[
    m\geq\frac1{256}\left[
        \frac{\log N}{\varepsilon}
        +\frac{\Delta}{\varepsilon^2}
            \log\!\left(1+\frac{N\varepsilon^2}
                                {(\Delta+\varepsilon)^2}\right)
    \right].
\]
When $0\leq\Delta<8\varepsilon$, the suboptimality-dependent term
is bounded by the fast term up to a constant:
\[
    \frac{\Delta}{\varepsilon^2}
        \log\!\left(1+\frac{N\varepsilon^2}
                            {(\Delta+\varepsilon)^2}\right)
    \leq\frac8\varepsilon\log(1+N)
    \leq\frac{16\log N}{\varepsilon}.
\]
Here we used $N\geq2$, so that $\log(1+N)\leq2\log N$.
The bracket in \cref{eq:standard_pass1_class_requirement} is therefore
at most $17\log N/\varepsilon$. Applying
\cref{eq:standard_pass1_fast_class_assouad} and using $17\cdot64=1088$
proves \cref{eq:standard_pass1_class_requirement} in this regime as
well.

\subsubsection{Dependence on the confidence level}

\paragraph{Context distribution and optimal demonstrators.}
Choose two core rewards $r_1^\circ,r_2^\circ$ whose preferred
responses at context $1$ are one and two, respectively. For the fast
term, put mass $w:=4\varepsilon$ on context $1$, mass $1-w$ on
context $0$, and zero mass elsewhere. Under instance $a\in\{1,2\}$,
the demonstrator returns $a$ at context $1$ and zero at context $0$;
the evaluation reward is $r_a^\circ$. Both demonstrators are optimal
for their selected rewards and are admissible for every $\Delta$.

\paragraph{Admissibility and the target bound.}
Write $Z_a:={\mathcal{E}}_{r_a^\circ}^1
(\widehat\pi,\pi_{\mathrm{demo}}^{(a)})$. Decode $\widehat a=1$ if
$\widehat\pi(1\mid1)\geq1/2$ and $\widehat a=2$ otherwise.
A decoding error means that the learner assigns probability at most
$1/2$ to the correct response at context $1$. Consequently,
\[
    Z_a=(1-w)\bigl[1-\widehat\pi(0\mid0)\bigr]
        +w\bigl[1-\widehat\pi(a\mid1)\bigr]
    \geq\frac w2=2\varepsilon
    \quad\text{whenever }\widehat a\ne a.
\]
The PAC requirement therefore bounds each decoding error probability
by $\delta$.

\paragraph{Proof using Le Cam's method.}
The sample laws overlap exactly when no demonstration has context $1$,
an event of probability $(1-w)^m$ under both instances. Thus
\[
    \operatorname{TV}(P_1^{\otimes m},P_2^{\otimes m})=1-(1-w)^m.
\]
Using the Pinsker inequality~\citep{pinsker1964information} and the identity from~\citet[Eq.~(14.13)]{lattimore2019bandit} in the Le Cam's testing bound, gives
\begin{align*}
    \frac12\mathbb P_1\{\widehat a\ne1\}
        +\frac12\mathbb P_2\{\widehat a\ne2\}
    &\geq\frac12\left[1-\operatorname{TV}(P_1^{\otimes m},P_2^{\otimes m})\right]\\
    &=\frac12(1-w)^m
    \geq\frac12e^{-2mw}.
\end{align*}
The last inequality uses $w\leq1/2$. At least one fixed instance has
decoding error at least this average. Since the PAC requirement makes
the average at most $\delta$, we conclude that
\begin{equation}
\label{eq:standard_pass1_fast_confidence_lecam}
    m\geq\frac{\log(1/(2\delta))}{8\varepsilon}
      \geq\frac{\log(1/\delta)}{16\varepsilon}.
\end{equation}

\paragraph{Context distribution and suboptimal demonstrators.}
Suppose now that $\Delta\geq8\varepsilon$ and again set
$h:=2\varepsilon/\Delta\leq1/4$. Put mass $\Delta$ on context $1$,
mass $1-\Delta$ on context $0$, and zero mass elsewhere. For
$\sigma\in\{-1,+1\}$, define
\[
    \pi_{\mathrm{demo}}^{(\sigma)}(0\mid0)=1,
    \qquad
    \pi_{\mathrm{demo}}^{(\sigma)}(1\mid1)=\frac12+\sigma h,
    \qquad
    \pi_{\mathrm{demo}}^{(\sigma)}(2\mid1)=\frac12-\sigma h,
\]
and select $r_+:=r_1^\circ$ and $r_-:=r_2^\circ$ as the
corresponding evaluation rewards. The one-demonstration laws are
\[
    P_\sigma(x,y)=
    \begin{cases}
        1-\Delta,&(x,y)=(0,0),\\
        \Delta(1/2+\sigma h),&(x,y)=(1,1),\\
        \Delta(1/2-\sigma h),&(x,y)=(1,2),\\
        0,&\text{otherwise}.
    \end{cases}
\]
Write $P_+:=P_{+1}$ and $P_-:=P_{-1}$. These laws share the same
context distribution and differ only in the conditional response law
at context $1$.

\paragraph{Admissibility and the target bound.}
Each demonstrator chooses the response accepted by its selected reward
with probability $1/2+h$ at context $1$. Hence
$\Delta_{r_\sigma}=\Delta(1/2-h)\leq\Delta$.
For the returned policy, write
\[
    Z_\sigma:=
        {\mathcal{E}}_{r_\sigma}^1(\widehat\pi,\pi_{\mathrm{demo}}^{(\sigma)}),
    \qquad
    \widehat\sigma:=
    \begin{cases}
        +1,&\widehat\pi(1\mid1)\geq1/2,\\
        -1,&\widehat\pi(1\mid1)<1/2.
    \end{cases}
\]
On a decoding error, the learner assigns probability at most $1/2$
to the rewarded response at context $1$, while the demonstrator
assigns probability $1/2+h$. The contribution from context $0$ is
nonnegative, so
\begin{equation}
\label{eq:standard_pass1_testing_reduction}
    \{\widehat\sigma\ne\sigma\}
    \subseteq\{Z_\sigma\geq\Delta h=2\varepsilon\}
    \subseteq\{Z_\sigma>\varepsilon\}.
\end{equation}

\paragraph{Proof using the Bretagnolle--Huber inequality.}
The product identity in~\citet[Exercises~14.10 and~14.12]{lattimore2019bandit} gives
\begin{align*}
    D_{\mathrm{KL}}(P_+^{\otimes m}\|P_-^{\otimes m})
    &=mD_{\mathrm{KL}}(P_+\|P_-)\\
    &\overset{(a)}{=}m\Delta\,\operatorname{kl}(1/2+h,1/2-h)\\
    &\overset{(b)}{\leq}16m\Delta h^2
      =\frac{64m\varepsilon^2}{\Delta}.
\end{align*}
Step $(a)$ follows by conditioning on the common context marginal;
the conditional response laws agree at context $0$. Step $(b)$ is due the identity from~\citet[Appendix~A.2]{de2021bandits}. These bounds remain valid after adjoining
the learner's independent training randomization.
The Bretagnolle--Huber inequality \citetext{\citet[Theorem~14.2]{lattimore2019bandit};
\cite{tsybakov2009introduction}} yields
\begin{align*}
    \frac12\mathbb P_+\{\widehat\sigma=-1\}
        +\frac12\mathbb P_-\{\widehat\sigma=+1\}
    &\geq\frac14\exp\!\left\{
        -D_{\mathrm{KL}}(P_+^{\otimes m}\|P_-^{\otimes m})
    \right\}\\
    &\geq\frac14\exp\!\left\{-\frac{64m\varepsilon^2}{\Delta}\right\}.
\end{align*}
At least one fixed sign has decoding error at least this average.
By \cref{eq:standard_pass1_testing_reduction}, the PAC requirement
bounds each decoding error by $\delta$. Therefore
\[
    \delta\geq\frac14\exp\!\left\{-\frac{64m\varepsilon^2}{\Delta}\right\}.
\]
Using $\log(1/(4\delta))\geq\tfrac12\log(1/\delta)$, we obtain
\begin{equation}
\label{eq:standard_pass1_slow_component}
    m\geq\frac{\Delta\log(1/\delta)}{128\varepsilon^2},
    \qquad \Delta\geq8\varepsilon.
\end{equation}
This is also the estimate used above when $Nh^2<4$.

\paragraph{Conclusion of the confidence statement.}
If $\Delta\geq8\varepsilon$, the bounds
\cref{eq:standard_pass1_fast_confidence_lecam,eq:standard_pass1_slow_component}
imply
\[
    m\geq\frac1{256}\left[
        \frac{\log(1/\delta)}{\varepsilon}
        +\frac{\Delta\log(1/\delta)}{\varepsilon^2}
    \right].
\]
If $0\leq\Delta<8\varepsilon$, the bracket is at most
$9\log(1/\delta)/\varepsilon$, so the fast bound alone gives the
same conclusion, even with the larger constant $1/144$.
This proves \cref{eq:standard_pass1_confidence_requirement} for
every $\Delta\in[0,1]$.

\paragraph{Combining the two statements.}
Let $A$ and $B$ denote the bracketed expressions in
\cref{eq:standard_pass1_class_requirement,eq:standard_pass1_confidence_requirement},
respectively. Both statements apply to every learner that satisfies
the PAC requirement on the same class ${\mathcal{R}}$. Thus
$A\leq1088m$ and $B\leq256m$, which give
\[
    m\geq\frac{A+B}{1344}\geq\frac{A+B}{2048}.
\]
The hard instances used to establish the two bounds need not coincide:
the PAC requirement must hold for every admissible instance on this
single reward class.

\paragraph{Sample complexity.}
Zero samples cannot suffice either. In the optimal two-point
construction, the empty-sample laws are identical. Hence the average
decoding error is at least $1/2$, and at least one fixed instance has
$Z_a\geq2\varepsilon$ with probability at least $1/2>\delta$.
Since every successful positive sample size satisfies the preceding
bound, \cref{def:standard_pac_complexity} gives
\begin{equation}
\label{eq:standard_pass1_lower_samples}
\begin{aligned}
    m_{\mathrm{std}}^{(1)}({\mathcal{R}},\varepsilon,\delta,\Delta)
    \geq\frac1{2048}\Bigg[
        &\frac{\log N+\log(1/\delta)}{\varepsilon}\\
        &+\frac{\Delta}{\varepsilon^2}
        \left\{
            \log\frac1\delta
            +\log\!\left(1+\frac{N\varepsilon^2}
                                {(\Delta+\varepsilon)^2}\right)
        \right\}
    \Bigg].
\end{aligned}
\end{equation}
The constructed class has exactly $N$ rewards for every integer
$N\geq2$. Each testing or averaging argument identifies an admissible
instance and a fixed evaluation reward before the sample is drawn,
as required for standard evaluation. This completes the proof.

The lower bound has two components. We use the same context space,
response space, and reward class for both. The first construction places
the difficulty across several blocks of contexts. A second-moment argument
shows that a block contains an unobserved context with constant probability,
and the pass@$k$ inclusion constraint then forces a nonzero probability of
missing its hidden response. This gives the dependence on $\log N$.
The second construction concentrates the difficulty on a single rare
context and gives the dependence on $\log(1/\delta)$. We prove these two
bounds separately and then combine them.

\paragraph{Common reward class.}
Fix integers $k\geq2$ and $N\geq2k^2$, and set
\[
    q:=2k,
    \qquad
    b:=\left\lfloor\frac{\log N}{\log(2k^2)}\right\rfloor\geq1,
    \qquad
    {\mathcal{Y}}:=\{0,1,\ldots,q\}.
\]
We first define the rewards on the core context space
${\mathcal{X}}_0:=\{0\}\cup([b]\times[k])$. For each
$v=((j_\ell,a_\ell))_{\ell=1}^b\in([k]\times[q])^b$, let
\[
    r_v(0,y):=1,
    \qquad
    r_v((\ell,j),y):=
    \begin{cases}
        \mathds{1}\{y=a_\ell\},&j=j_\ell,\\
        1,&j\ne j_\ell.
    \end{cases}
\]
Thus $r_v$ selects one context in each block and rewards only the
preferred response $a_\ell$ there; it gives reward one at all other
core contexts. These are $(kq)^b=(2k^2)^b$ distinct binary rewards.

To obtain exactly $N$ rewards, set $t:=N-(2k^2)^b$ and introduce
$t$ distinct auxiliary contexts $z_1,\ldots,z_t$ outside ${\mathcal{X}}_0$.
Let
\[
    {\mathcal{X}}:={\mathcal{X}}_0\cup\{z_1,\ldots,z_t\},
    \qquad
    r_v(z_s,y):=\mathds{1}\{y=0\},\quad s\in[t].
\]
Fix one core index $v^0$. For each $s\in[t]$, define an additional
reward $\widetilde r_s$ that agrees with $r_{v^0}$ everywhere except
at $z_s$, where $\widetilde r_s(z_s,y):=\mathds{1}\{y=1\}$. Then
\[
    {\mathcal{R}}:=\{r_v:v\in([k]\times[q])^b\}
        \cup\{\widetilde r_s:s\in[t]\}
\]
contains exactly $N$ distinct binary rewards. When $t=0$, no auxiliary
contexts or rewards are added. Every reward has optimal value one.
Both constructions below assign zero probability to the auxiliary
contexts. Consequently, each additional reward has the same value as
$r_{v^0}$ under every policy, and maximizing either the suboptimality
or the excess over ${\mathcal{R}}$ is equivalent to maximizing over the core
rewards. The spaces and reward class depend only on $k$ and $N$.

\paragraph{Two statements to prove.}
Fix $m\geq1$, $0<\Delta\leq1$, and $0<\delta<1/16$.
We will prove two probability lower bounds on this fixed reward class.
First, for any learner, there exist a context distribution ${\mathcal{D}}$ and
a demonstrator $\pi_{\mathrm{demo}}$ satisfying $\Delta_{{\mathcal{R}}}\leq\Delta$ such that
\begin{equation}
\label{eq:robust_proof_class_tail}
    \mathbb P\!\left\{
        \sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^k(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{96}\min\!\left\{
            \Delta,\frac1k,\frac{\log N}{m}
        \right\}
    \right\}\geq\frac17>2\delta.
\end{equation}
Second, for any learner, there exist a context distribution ${\mathcal{D}}$ and
a demonstrator $\pi_{\mathrm{demo}}$ satisfying $\Delta_{{\mathcal{R}}}\leq\Delta$ such that
\begin{equation}
\label{eq:robust_proof_confidence_tail}
    \mathbb P\!\left\{
        \sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^k(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{32}\min\!\left\{
            \Delta,\frac{\log(1/\delta)}m
        \right\}
    \right\}\geq\frac{8\delta}{3}>2\delta.
\end{equation}
In each statement, $\widehat\pi$ is the policy returned by the learner
from $m$ demonstrations under the selected instance. The instance may
depend on the learner, but is fixed before the sample is drawn.
All probabilities include the learner's training randomization, while
prediction-time randomness is averaged in the policy values.

\paragraph{The pass@$k$ inclusion constraint.}
Fix an arbitrary learner. For a realized output policy $\widehat\pi$,
define the probability that its response tuple contains $a\in[q]$ by
\[
    C_x(a):=
    \mathbb P_{{\bm{y}}\sim\widehat\pi(\cdot\mid x)}
       \{a\in\{y^{(1)},\ldots,y^{(k)}\}\}.
\]
Since a tuple contains at most $k$ distinct responses,
\begin{equation}
\label{eq:robust_proof_inclusion_budget}
    \sum_{a=1}^q C_x(a)
    =\mathbb E_{{\bm{y}}\sim\widehat\pi(\cdot\mid x)}
       \bigl|[q]\cap\{y^{(1)},\ldots,y^{(k)}\}\bigr|
    \leq k.
\end{equation}
This constraint holds for arbitrary dependence among the responses.
Both lower bounds use it to show that $k$ responses cannot cover a
uniformly distributed hidden response from $[q]$ with probability
greater than $k/q=1/2$.

\subsubsection{Dependence on the reward-class size}

\paragraph{Context distribution and demonstrators.}
Choose
\[
    p:=\min\!\left\{
        \frac{\Delta}{b},\frac1{2bk},\frac{\log k}{2m}
    \right\},
    \qquad
    {\mathcal{D}}((\ell,j)):=p,
    \qquad
    {\mathcal{D}}(0):=1-bkp,
\]
and give every auxiliary context zero mass. The condition
$p\leq1/(2bk)$ ensures that ${\mathcal{D}}(0)\geq1/2$, so this is a
probability distribution. For each $\theta\in[q]^{bk}$, let
$\pi_{\mathrm{demo}}^{(\theta)}$ return $\theta_{\ell,j}$ at $(\ell,j)$ and zero elsewhere.
The parameter $\theta$ therefore specifies one hidden response at
each core context other than $0$.

\paragraph{Admissibility and the robust excess.}
For each core reward $r_v$, the demonstrator receives reward one at
context $0$ and at the $k-1$ unselected contexts in each block.
At the selected context $(\ell,j_\ell)$, its reward is
$\mathds{1}\{a_\ell=\theta_{\ell,j_\ell}\}$. Hence
\begin{align*}
    V_{r_v}^\star-V_{r_v}(\pi_{\mathrm{demo}}^{(\theta)})
    &=1-(1-bkp)
        -p\sum_{\ell=1}^b\sum_{j=1}^k
            r_v((\ell,j),\theta_{\ell,j})\\
    &=bkp-p\sum_{\ell=1}^b
        \left[(k-1)+\mathds{1}\{a_\ell=\theta_{\ell,j_\ell}\}\right]\\
    &=p\sum_{\ell=1}^b
        \mathds{1}\{a_\ell\ne\theta_{\ell,j_\ell}\}.
\end{align*}
The last sum is at most $b$, and equality is attained by choosing
a mismatching preferred response in every block. The additional
rewards have the same values as a core reward on the support of
${\mathcal{D}}$, so $\Delta_{{\mathcal{R}}}=bp\leq\Delta$. All members of the family
are therefore admissible.

For demonstrations from $\pi_{\mathrm{demo}}^{(\theta)}$, let $\widehat\pi$ be the
learner's returned policy. At a selected context, the expected
pass@$k$ reward is $C_{(\ell,j_\ell)}(a_\ell)$. Thus
\begin{align*}
    V_{r_v}(\pi_{\mathrm{demo}}^{(\theta)})
    &=(1-bkp)+p\sum_{\ell=1}^b
        \left[(k-1)+\mathds{1}\{a_\ell=\theta_{\ell,j_\ell}\}\right],\\
    V_{r_v}^k(\widehat\pi)
    &=(1-bkp)+p\sum_{\ell=1}^b
        \left[(k-1)+C_{(\ell,j_\ell)}(a_\ell)\right].
\end{align*}
Subtracting these expressions gives
\[
    {\mathcal{E}}_{r_v}^k(\widehat\pi,\pi_{\mathrm{demo}}^{(\theta)})
    =p\sum_{\ell=1}^b
        \left[\mathds{1}\{a_\ell=\theta_{\ell,j_\ell}\}
              -C_{(\ell,j_\ell)}(a_\ell)\right].
\]
At any selected context, choosing the preferred response to be its
hidden response gives $1-C_{(\ell,j_\ell)}(\theta_{\ell,j_\ell})
\geq0$; every other choice gives a nonpositive contribution.
We may then choose the context with the largest miss probability
independently in each block. Consequently,
\begin{equation}
\label{eq:robust_proof_block_loss}
\begin{aligned}
    Z_\theta
    &:=\sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^k(\widehat\pi,\pi_{\mathrm{demo}}^{(\theta)})\\
    &=p\sum_{\ell=1}^b\max_{j\in[k]}
        \bigl[1-C_{(\ell,j)}(\theta_{\ell,j})\bigr],
    \qquad 0\leq Z_\theta\leq bp.
\end{aligned}
\end{equation}

\paragraph{Unobserved contexts and the inclusion constraint.}
Temporarily draw the coordinates of $\theta$ independently and
uniformly from $[q]$, and then generate the training sample under
${\mathcal{D}}$ and $\pi_{\mathrm{demo}}^{(\theta)}$. Until we select a fixed parameter below,
probabilities and expectations also average over this prior.
Let $U_\ell$ count the contexts in block $\ell$ that do not appear
in the sample. Each context is absent with probability $(1-p)^m$.
Since $p\leq1/(2bk)\leq1/4$ and $2mp\leq\log k$,
\[
    \lambda:=\mathbb E U_\ell
    =k(1-p)^m
    \geq ke^{-2mp}
    \geq1,
\]
where we used $\log(1-p)\geq-2p$ for $p\leq1/2$.
For two distinct contexts, the probability that both are absent is
$(1-2p)^m\leq(1-p)^{2m}$. Expanding the second moment therefore gives
\[
    \mathbb E U_\ell^2
    =k(1-p)^m+k(k-1)(1-2p)^m
    \leq\lambda+\lambda^2.
\]
The second-moment inequality, obtained by applying Paley--Zygmund inequality, now yields
\[
    \mathbb P\{U_\ell>0\}
    \geq\frac{(\mathbb E U_\ell)^2}{\mathbb E U_\ell^2}
    \geq\frac{\lambda}{1+\lambda}
    \geq\frac12.
\]
Thus every block contains an unobserved context with probability
at least $1/2$.

On $\{U_\ell>0\}$, let $J_\ell$ be the smallest index of an
unobserved context in that block. This choice depends only on the
sampled contexts. Conditional on $S$ and $\widehat\pi$, the hidden
response $\theta_{\ell,J_\ell}$ remains uniform on $[q]$: the sample
reveals only the responses at observed contexts, and the learner
depends on $\theta$ only through that sample. Hence, on this event,
\begin{align*}
    \mathbb E\!\left[
        1-C_{(\ell,J_\ell)}(\theta_{\ell,J_\ell})
        \mid S,\widehat\pi
    \right]
    &=1-\frac1q\sum_{a=1}^q C_{(\ell,J_\ell)}(a)\\
    &\geq1-\frac{k}{q}=\frac12,
\end{align*}
by \cref{eq:robust_proof_inclusion_budget}. The maximum in
\cref{eq:robust_proof_block_loss} is at least the miss probability
at this unobserved context. On the complementary event it is
nonnegative. Taking expectations therefore gives
\[
    \mathbb E Z_\theta
    \geq\frac p2\sum_{\ell=1}^b\mathbb P\{U_\ell>0\}
    \geq\frac{bp}{4}.
\]

\paragraph{A probability bound for a fixed instance.}
We next convert this expectation bound into a probability bound.
Since $0\leq Z_\theta\leq bp$, splitting the expectation at $bp/8$
gives
\begin{align*}
    \frac{bp}{4}
    \leq\mathbb E Z_\theta
    &\leq\frac{bp}{8}\mathbb P\{Z_\theta\leq bp/8\}
        +bp\,\mathbb P\{Z_\theta>bp/8\}\\
    &=\frac{bp}{8}
        +\frac{7bp}{8}\mathbb P\{Z_\theta>bp/8\}.
\end{align*}
It follows that $\mathbb P\{Z_\theta>bp/8\}\geq1/7$.
This is an average over the finite prior, so at least one fixed
$\theta\in[q]^{bk}$ satisfies
\[
    \mathbb P_\theta\{Z_\theta>bp/8\}\geq\frac17,
\]
where $\mathbb P_\theta$ now averages only over the training sample
and the learner's training randomization.

To express the threshold in terms of $N$, use
$\lfloor u\rfloor\geq u/2$ for $u\geq1$ and
$\log(2k^2)\leq3\log k$. These imply
\[
    b\log k
    \geq\frac{\log N\,\log k}{2\log(2k^2)}
    \geq\frac{\log N}{6}.
\]
Consequently,
\[
    \frac{bp}{8}
    =\min\!\left\{
        \frac{\Delta}{8},\frac1{16k},\frac{b\log k}{16m}
    \right\}
    \geq\frac1{96}\min\!\left\{
        \Delta,\frac1k,\frac{\log N}{m}
    \right\}.
\]
For the fixed parameter selected above, $Z_\theta$ therefore
exceeds the threshold in \cref{eq:robust_proof_class_tail} with
probability at least $1/7$. This proves the first statement.

\subsubsection{Dependence on the confidence level}

\paragraph{Context distribution and demonstrators.}
Use the same spaces and reward class, and set
\[
    \eta:=\min\!\left\{
        \Delta,\frac12,\frac{\log(1/(8\delta))}{2m}
    \right\},
    \qquad x_\bullet:=(1,1).
\]
Put mass $\eta$ on $x_\bullet$, mass $1-\eta$ on context $0$,
and zero mass on every other context. For each $\theta\in[q]$,
let $\pi_{\mathrm{demo}}^{(\theta)}$ return $\theta$ at $x_\bullet$ and zero elsewhere.
A core reward selecting $x_\bullet$ and a preferred response
distinct from $\theta$ has suboptimality $\eta$, and no reward
has larger suboptimality. Thus $\Delta_{{\mathcal{R}}}=\eta\leq\Delta$.
Maximizing the excess as in \cref{eq:robust_proof_block_loss} gives
\[
    Z_\theta
    :=\sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^k(\widehat\pi,\pi_{\mathrm{demo}}^{(\theta)})
    =\eta\bigl[1-C_{x_\bullet}(\theta)\bigr].
\]

\paragraph{The event of receiving no informative demonstration.}
Temporarily draw $\theta$ uniformly from $[q]$, and let $E$ be the
event that $x_\bullet$ is absent from the sample. Since
$0<\eta\leq1/2$ and $2m\eta\leq\log(1/(8\delta))$,
\[
    \mathbb P(E)=(1-\eta)^m
    \geq e^{-2m\eta}
    \geq8\delta.
\]
On $E$, the sample contains only copies of $(0,0)$ and reveals
no information about $\theta$. Thus $\theta$ remains uniform
conditional on $S$ and $\widehat\pi$. By
\cref{eq:robust_proof_inclusion_budget}, the random variable
$H:=1-C_{x_\bullet}(\theta)\in[0,1]$ satisfies, on $E$,
\[
    \mathbb E[H\mid S,\widehat\pi]
    =1-\frac1q\sum_{a=1}^q C_{x_\bullet}(a)
    \geq\frac12.
\]
Splitting this conditional expectation at $1/4$ gives
\[
    \frac12
    \leq\mathbb E[H\mid S,\widehat\pi]
    \leq\frac14+\frac34
        \mathbb P\{H>1/4\mid S,\widehat\pi\}.
\]
The conditional probability is therefore at least $1/3$ on $E$.
Since $Z_\theta=\eta H$, averaging over the sample and training
randomization yields
\[
    \mathbb P\{Z_\theta>\eta/4\}
    \geq\mathbb P\bigl(E\cap\{H>1/4\}\bigr)
    \geq\frac{\mathbb P(E)}3
    \geq\frac{8\delta}{3}.
\]
As before, averaging over the finite prior selects a fixed
$\theta\in[q]$ for which the same probability bound holds over
training alone.

For $\delta<1/16$, we have
$\log(1/(8\delta))\geq\tfrac14\log(1/\delta)$. Since also
$\Delta\leq1$,
\[
    \frac\eta4
    =\min\!\left\{
        \frac{\Delta}{4},\frac18,
        \frac{\log(1/(8\delta))}{8m}
    \right\}
    \geq\frac1{32}\min\!\left\{
        \Delta,\frac{\log(1/\delta)}m
    \right\}.
\]
For the selected fixed parameter, $Z_\theta$ exceeds the threshold
in \cref{eq:robust_proof_confidence_tail} with probability at least
$8\delta/3$. This proves the second statement.

\paragraph{Combining the two statements.}
Fix any learner and write
\[
    A:=\min\!\left\{\Delta,\frac1k,\frac{\log N}{m}\right\},
    \qquad
    B:=\min\!\left\{\Delta,\frac{\log(1/\delta)}m\right\}.
\]
If $A\geq B$, use the first construction. If $B>A$, use the
second construction. Both failure probabilities exceed $2\delta$,
and using the smaller threshold constant $1/96$ in both statements
shows that the selected admissible instance satisfies
\[
    \mathbb P\!\left\{
        Z>\frac1{96}\max\{A,B\}
    \right\}\geq2\delta,
    \qquad
    Z:=\sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^k(\widehat\pi,\pi_{\mathrm{demo}}).
\]
Since $\max\{A,B\}\geq(A+B)/2$, lowering the threshold and
substituting the definitions of $A$ and $B$ gives
\begin{equation}
\label{eq:robust_proof_combined_tail}
\begin{aligned}
    \mathbb P\Bigg\{
        Z>\frac1{192}\Bigg[
            &\min\!\left\{
                \Delta,\frac1k,\frac{\log N}{m}
            \right\}\\
            &+\min\!\left\{
                \Delta,\frac{\log(1/\delta)}m
            \right\}
        \Bigg]\Bigg\}\geq2\delta.
\end{aligned}
\end{equation}
Both constructions use the same class of exactly $N$ rewards.
For each learner, the selected context distribution, demonstrator,
and hidden parameter are fixed before the sample and training
randomization are drawn.

\paragraph{Sample complexity.}
Suppose
\[
    0<\varepsilon
    \leq\frac1{384}\min\!\left\{\Delta,\frac1k\right\}.
\]
For every integer
\[
    1\leq m\leq\frac{\log N+\log(1/\delta)}{384\,\varepsilon},
\]
the sum of the two truncated terms in
\cref{eq:robust_proof_combined_tail} satisfies
\begin{align*}
    &\min\!\left\{\Delta,\frac1k,\frac{\log N}{m}\right\}
        +\min\!\left\{\Delta,\frac{\log(1/\delta)}m\right\}\\
    &\qquad\geq\min\!\left\{
        \Delta,\frac1k,\frac{\log N+\log(1/\delta)}m
    \right\}
    \geq384\,\varepsilon.
\end{align*}
The excess threshold is therefore at least $2\varepsilon$, so
every learner fails the robust PAC requirement with probability
at least $2\delta>\delta$ on an admissible instance. This holds
at every such positive integer sample size, including the upper
endpoint when it is an integer. The displayed upper limit exceeds
one under the stated assumptions, so failure at $m=1$ also rules
out a successful learner using zero samples: such a learner could
simply ignore one available demonstration.

Definition~\ref{def:robust_pac_complexity} consequently gives
\[
    m_{\mathrm{rob}}^{(k)}
        ({\mathcal{R}},\varepsilon,\delta,\Delta)
    >\left\lfloor
        \frac{\log N+\log(1/\delta)}{384\,\varepsilon}
    \right\rfloor.
\]
This establishes \cref{eq:robust_passk_lower_samples} for every
$N\geq2k^2$, including the integer endpoint, and completes the proof.

The lower bound has two components. We use the same context space, response space, and reward class for both.
The first construction shows that the learner must obtain information about
many contexts in order to perform well simultaneously for all rewards, which
gives the dependence on $\log N$. The second construction concentrates the
difficulty on a single context and shows that the learner cannot make the
failure probability arbitrarily small without additional samples, which gives
the dependence on $\log(1/\delta)$. We prove these two bounds separately and
then combine them.

\paragraph{Common reward class.}
Fix an integer $N\geq2$ and set
\[
    d:=\left\lfloor\frac{\log N}{\log2}\right\rfloor\geq1,
    \qquad
    {\mathcal{X}}:=\{0,1,\ldots,d\},
    \qquad
    {\mathcal{Y}}:=\{-1,+1\}.
\]
The contexts $1,\ldots,d$ will carry the unknown response biases.
At context $0$, every demonstrator constructed below will always
output $+1$. For each $v\in\{-1,+1\}^d$, define
\[
    f_v(0):=+1,
    \qquad
    f_v(i):=v_i,\quad i\in[d],
    \qquad
    r_v(x,y):=\mathds{1}\{y=f_v(x)\}.
\]
These form a core class
${\mathcal{R}}_0:=\{r_v:v\in\{-1,+1\}^d\}$ of $2^d$ distinct binary rewards.
Since $2^d\leq N<2^{d+1}$, choose a subset
$\mathcal V\subseteq\{-1,+1\}^d$ with
$|\mathcal V|=N-2^d$. For every $v\in\mathcal V$, define
\[
    \widetilde r_v(x,y):=
    \begin{cases}
        1,&x=0,\\
        r_v(x,y),&x\in[d],
    \end{cases}
    \qquad
    {\mathcal{R}}:={\mathcal{R}}_0\cup\{\widetilde r_v:v\in\mathcal V\}.
\]
The added rewards are distinct from all core rewards because they
equal one at $(0,-1)$, and they are distinct from one another because
their preferred responses on $[d]$ differ. Thus $|{\mathcal{R}}|=N$.
When $N=2^d$, take $\mathcal V=\varnothing$.

Every reward in ${\mathcal{R}}$ has optimal value one. Moreover, for every
demonstrator $\pi_{\mathrm{demo}}$ used below and every policy $\pi$,
\[
    V_{\widetilde r_v}(\pi_{\mathrm{demo}})=V_{r_v}(\pi_{\mathrm{demo}}),
    \qquad
    V_{\widetilde r_v}(\pi)\geq V_{r_v}(\pi),
    \qquad v\in\mathcal V.
\]
The first equality holds because the demonstrator always outputs
$+1$ at context $0$; the inequality follows from
$\widetilde r_v\geq r_v$ pointwise. Therefore the added rewards
change neither the maximum demonstrator suboptimality nor the
robust excess: both maxima over ${\mathcal{R}}$ equal the corresponding
maxima over ${\mathcal{R}}_0$. Both constructions below use this same class
of $N$ rewards on ${\mathcal{X}}=\{0,\ldots,d\}$ and ${\mathcal{Y}}=\{-1,+1\}$.

\paragraph{Two statements to prove.}
Fix $m\geq1$, $0<\Delta\leq1$, and $0<\delta<1/16$.
We will prove two probability lower bounds on this fixed reward class.
First, for any learner, there exist a context distribution ${\mathcal{D}}$ and
a demonstrator $\pi_{\mathrm{demo}}$ satisfying $\Delta_{{\mathcal{R}}}\leq\Delta$ such that
\begin{equation}
\label{eq:robust_proof_pass1_class_tail}
    \mathbb P\!\left\{
        \sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^1(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{128}\min\!\left\{
            \Delta,\sqrt{\frac{\Delta\log N}{m}}
        \right\}
    \right\}\geq\frac18>2\delta.
\end{equation}
Second, for any learner, there exist a context distribution ${\mathcal{D}}$ and
a demonstrator $\pi_{\mathrm{demo}}$ satisfying $\Delta_{{\mathcal{R}}}\leq\Delta$ such that
\begin{equation}
\label{eq:robust_proof_pass1_confidence_tail}
    \mathbb P\!\left\{
        \sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^1(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{16}\min\!\left\{
            \Delta,\sqrt{\frac{\Delta\log(1/\delta)}{m}}
        \right\}
    \right\}\geq2\delta.
\end{equation}
In each statement, $\widehat\pi$ is the policy returned by the learner
from $m$ demonstrations under the selected instance. The instance may
depend on the learner, but is fixed before the sample is drawn.
All probabilities include the learner's training randomization, while
prediction-time randomness is averaged in the policy values.
The next two subsections prove these statements. We then select the
larger logarithmic contribution and convert the resulting probability
bound into the claimed sample-complexity lower bound.

\subsubsection{Dependence on the reward-class size}
Fix an arbitrary learner. We first construct the context distribution and the family of demonstrators. 
We put total mass $\Delta$ equally on the
hard contexts $1,\ldots,d$ and put the remaining mass on context $0$:
\[
    {\mathcal{D}}(0)=1-\Delta,
    \qquad
    {\mathcal{D}}(i)=\frac{\Delta}{d},\quad i\in[d].
\]
With this context distribution fixed, we construct a family of
demonstrators indexed by $\theta\in\{-1,+1\}^d$.
Let $h\in(0,1/4]$ be a bias magnitude, which we will choose in the
Assouad argument below. For each $\theta$, the policy $\pi_{\mathrm{demo}}^{(\theta)}$
always outputs $+1$ at context $0$. At a hard context $i$, for $t\in\{-1,+1\}$, we set 
\begin{equation}
\label{eq:demo_t_i}
    \pi_{\mathrm{demo}}^{(\theta)}(t\mid i)
    =\frac12+t\theta_i h
    =\frac12+t\left(p_{\theta,i}-\frac12\right),
\end{equation}
where in the last identity we define $p_{\theta,i}:=\pi_{\mathrm{demo}}^{(\theta)}(+1\mid i)$.
Thus $\theta$ specifies the signs of $d$ response biases, while $h$
sets their common magnitude. It is enough to find one member of this
family for which the fixed learner has large robust excess with the
probability required in \cref{eq:robust_proof_pass1_class_tail}.

We first verify that every demonstrator in this family satisfies the required
suboptimality constraint. Every reward has optimal value one,
and at context $0$ the demonstrator receives reward one under every
reward. For each $v\in\{-1,+1\}^d$,
\begin{align*}
    V_{r_v}^\star-V_{r_v}(\pi_{\mathrm{demo}}^{(\theta)})
    &=1-\sum_{x=0}^d
        {\mathcal{D}}(x)\pi_{\mathrm{demo}}^{(\theta)}(f_v(x)\mid x)\\
    &=1-(1-\Delta)
        -\frac{\Delta}{d}\sum_{i=1}^d\pi_{\mathrm{demo}}^{(\theta)}(v_i\mid i)\\
    &=\Delta-\frac{\Delta}{d}
        \sum_{i=1}^d\left(\frac12+v_i\theta_i h\right)\\
    &=\frac{\Delta}{d}\sum_{i=1}^d
        \left(\frac12-v_i\theta_i h\right).
\end{align*}
This is largest when $v=-\theta$, giving
\begin{equation}
\label{eq:robust_proof_pass1_suboptimality}
    \Delta_{{\mathcal{R}}}=\Delta(1/2+h)\leq\Delta.
\end{equation}
All members of the family are therefore admissible. 



Now we are ready to prove a high probability lower bound on the excess risk using Assouad's lemma. For demonstrations
from $\pi_{\mathrm{demo}}^{(\theta)}$, let $\widehat\pi$ be the learner's returned policy
and define its robust excess by
\[
    Z_\theta:=\sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^1(\widehat\pi,\pi_{\mathrm{demo}}^{(\theta)}).
\]
We write $\mathbb P_\theta$ and $\mathbb{E}_\theta$ for probability and
expectation under this demonstrator and the fixed context distribution.
In this notation, our task is to find a fixed $\theta$ for which
$Z_\theta$ exceeds the threshold in
\cref{eq:robust_proof_pass1_class_tail} with probability at least
$1/8$. Such a member of the family gives the context distribution and
admissible demonstrator required by the first statement.

\paragraph{Proof using Assouad's lemma.}
We first relate $Z_\theta$ to the error in estimating the response
probabilities, then decode the signs of these estimates. Assouad's
lemma will show that, for at least one $\theta$, many signs are decoded
incorrectly. Write $q_i:=\widehat\pi(+1\mid i)$ for
$i\in\{0,\ldots,d\}$. Maximizing over the rewards in ${\mathcal{R}}$ gives
\begin{align}
    Z_\theta
    &=\sup_{v\in\{-1,+1\}^d}
        \left[V_{r_v}(\pi_{\mathrm{demo}}^{(\theta)})-V_{r_v}(\widehat\pi)\right]
        \notag\\
    &=\sup_{v\in\{-1,+1\}^d}
        \sum_{x=0}^d{\mathcal{D}}(x)
        \left[\pi_{\mathrm{demo}}^{(\theta)}(f_v(x)\mid x)
              -\widehat\pi(f_v(x)\mid x)\right]\notag\\
    &\overset{(a)}{=}(1-\Delta)(1-q_0)
        +\frac{\Delta}{d}
          \sup_{v\in\{-1,+1\}^d}
          \sum_{i=1}^dv_i(p_{\theta,i}-q_i)\notag\\
    &=(1-\Delta)(1-q_0)
        +\frac{\Delta}{d}\sum_{i=1}^d|p_{\theta,i}-q_i|.
        \label{eq:robust_proof_pass1_loss}
\end{align}
For step $(a)$, the context-$0$ term follows from
$f_v(0)=+1$ and $\pi_{\mathrm{demo}}^{(\theta)}(+1\mid0)=1$. For the remaining terms,
\[
    \widehat\pi(t\mid i)
    =\begin{cases}
        q_i,&t=+1,\\
        1-q_i,&t=-1
      \end{cases}
    =\frac12+t\left(q_i-\frac12\right),
    \qquad t\in\{-1,+1\}.
\]
Combining this identity with \cref{eq:demo_t_i} gives
$\pi_{\mathrm{demo}}^{(\theta)}(v_i\mid i)-\widehat\pi(v_i\mid i)
=v_i(p_{\theta,i}-q_i)$. The last equality in
\cref{eq:robust_proof_pass1_loss} follows by choosing each $v_i$
independently to maximize its summand. The first term is nonnegative
because $q_0\leq1$. Hence the estimation error on the hard contexts
gives a lower bound on the robust excess.

To turn this estimation error into a sign-decoding error, let
$\widehat\theta_i=+1$ when $q_i\geq1/2$ and
$\widehat\theta_i=-1$ otherwise. Write
$H(\widehat\theta,\theta):=\sum_{i=1}^d
\mathds{1}\{\widehat\theta_i\ne\theta_i\}$ for the number of incorrectly
decoded signs. Every incorrect sign implies
$|p_{\theta,i}-q_i|\geq h$, including a tie at $q_i=1/2$ when
$\theta_i=-1$. Applying this observation to
\cref{eq:robust_proof_pass1_loss} gives
\begin{align}
    Z_\theta
    &\geq\frac{\Delta}{d}\sum_{i=1}^d
        \left|\frac12+h\theta_i-q_i\right|\notag\\
    &=\frac{\Delta}{d}\sum_{i=1}^d
        \left|h\theta_i+\left(\frac12-q_i\right)\right|\notag\\
    &\geq\frac{\Delta h}{d}H(\widehat\theta,\theta).
        \label{eq:robust_proof_pass1_decode}
\end{align}
We will use Assouad's lemma to show that, for some fixed $\theta$,
the Hamming error exceeds $d/8$ with probability at least $1/8$.

For this purpose, we choose $h$ so that neighboring sign vectors
produce training samples that are difficult to distinguish. Let
$P_\theta(x,y):={\mathcal{D}}(x)\pi_{\mathrm{demo}}^{(\theta)}(y\mid x)$ be the law of one
demonstration, and let $\theta^{(i)}$ be obtained from $\theta$
by flipping its $i$-th coordinate. The corresponding demonstrators
differ only at context $i$, which has probability $\Delta/d$.
For $0<h\leq1/4$,
\begin{align}
    D_{\mathrm{KL}}(P_\theta^{\otimes m}\parallel
        P_{\theta^{(i)}}^{\otimes m})
    &=mD_{\mathrm{KL}}(P_\theta\parallel P_{\theta^{(i)}})\notag\\
    &\overset{(a)}{=}m\sum_{x=0}^d{\mathcal{D}}(x)
        D_{\mathrm{KL}}\!\left(\pi_{\mathrm{demo}}^{(\theta)}(\cdot\mid x)\parallel
            \pi_{\mathrm{demo}}^{(\theta^{(i)})}(\cdot\mid x)\right)\notag\\
    &=m{\mathcal{D}}(i)
        D_{\mathrm{KL}}\!\left(\pi_{\mathrm{demo}}^{(\theta)}(\cdot\mid i)\parallel
            \pi_{\mathrm{demo}}^{(\theta^{(i)})}(\cdot\mid i)\right)\notag\\
    &=\frac{m\Delta}{d}\operatorname{kl}(1/2+h,1/2-h)\notag\\
    &\overset{(b)}{\leq}\frac{16m\Delta h^2}{d}.
        \label{eq:robust_proof_pass1_neighbor_KL}
\end{align}
The first equality is the product identity in from~\citet[Exercises~14.10 and~14.12]{lattimore2019bandit}. Step $(a)$ uses the common-context-marginal
identity from the same identity.
                                       
                                  
                                 
                                                         
The step $(b)$ follows from~\citet[Appendix~A.2]{de2021bandits}.
Taking
\[
    h:=\min\!\left\{\frac14,\sqrt{\frac{d}{32m\Delta}}\right\}
\]
makes every neighboring divergence at most $1/2$.
These bounds also apply when the learner is randomized, since
adjoining its independent training randomization leaves the KL
divergence unchanged by the result from~\citet[Exercise~14.12]{lattimore2019bandit}.
                           
Assouad's lemma in Hamming loss
\citetext{\citealp{assouad1983deux};
          \citealp[Theorem~2.12]{tsybakov2009introduction}}
therefore gives
\begin{align}
    \max_{\theta\in\{-1,+1\}^d}
        \mathbb{E}_\theta H(\widehat\theta,\theta)
    &\geq\frac d2\left[
        1-\max_{\theta,i}
          \operatorname{TV}(P_\theta^{\otimes m},P_{\theta^{(i)}}^{\otimes m})
    \right]\notag\\
    &\overset{(a)}{\geq}\frac d2\left[
        1-\sqrt{\frac12\max_{\theta,i}
            D_{\mathrm{KL}}(P_\theta^{\otimes m}\parallel
                P_{\theta^{(i)}}^{\otimes m})}
    \right]\geq\frac d4.
        \label{eq:assouad_lem}
\end{align}
Here step $(a)$ is Pinsker's inequality~\citep{pinsker1964information}.

Consequently, there exists a fixed $\theta\in\{-1,+1\}^d$ such
that $\mathbb{E}_\theta H(\widehat\theta,\theta)\geq d/4$. We fix this
parameter for the rest of the argument. Since $0\leq H\leq d$,
writing $H=H(\widehat\theta,\theta)$ gives
\[
    \frac d4
    \leq\mathbb{E}_\theta H
    \leq\frac d8+d\,\mathbb P_\theta\!\left\{H>\frac d8\right\}.
\]
Therefore,
\[
    \mathbb P_\theta\!\left\{H(\widehat\theta,\theta)>
        \frac d8\right\}\geq\frac18.
\]
For this same $\theta$, on the above event,
\cref{eq:robust_proof_pass1_decode} gives $Z_\theta>\Delta h/8$.
Since $d=\lfloor\log N/\log2\rfloor$ and $N\geq2$, we have
$d\geq\log N/(2\log2)\geq\log N/4$, so our choice of $h$
satisfies
\[
    \frac{\Delta h}{8}
    =\min\!\left\{\frac{\Delta}{32},
            \sqrt{\frac{\Delta d}{2048m}}\right\}
    \geq\frac1{128}\min\!\left\{
        \Delta,\sqrt{\frac{\Delta\log N}{m}}
    \right\}.
\]
Thus $Z_\theta$ exceeds the threshold in
\cref{eq:robust_proof_pass1_class_tail} with probability at least
$1/8$. This proves the first statement.

\subsubsection{Dependence on the confidence level}

\paragraph{Context distribution and demonstrators.}
For the confidence bound, we use the same reward class but put all
of the hard-context mass on context $1$: let ${\mathcal{D}}(1)=\Delta$,
${\mathcal{D}}(0)=1-\Delta$, and give every other context zero mass.
For each sign $\theta\in\{-1,+1\}$, construct a demonstrator that
outputs $+1$ with probability $1/2+\theta h$ at context $1$ and
always outputs $+1$ elsewhere. Here we take
\[
    h:=\min\!\left\{\frac14,
        \sqrt{\frac{\log(1/(8\delta))}{16m\Delta}}\right\}.
\]
The two demonstrators differ only in which response is slightly more
likely at context $1$.

\paragraph{Admissibility and the target bound.}
Both demonstrators satisfy $\Delta_{{\mathcal{R}}}=\Delta(1/2+h)\leq\Delta$,
by the same calculation as in
\cref{eq:robust_proof_pass1_suboptimality}. We again write $Z_\theta$
for the robust excess under the selected demonstrator and the new
context distribution, and write $P_\theta$ for the law of one
demonstration. Explicitly,
\[
    P_\theta(x,y):={\mathcal{D}}(x)\pi_{\mathrm{demo}}^{(\theta)}(y\mid x)
    =\begin{cases}
        1-\Delta,&(x,y)=(0,+1),\\
        \Delta(1/2+\theta h),&(x,y)=(1,+1),\\
        \Delta(1/2-\theta h),&(x,y)=(1,-1),\\
        0,&\text{otherwise}.
      \end{cases}
\]
Write $P_+:=P_{+1}$ and $P_-:=P_{-1}$. These laws share the same
context distribution and differ only in the conditional response law
at context $1$. Since the demonstrations are independent, their joint
law is $P_\theta^{\otimes m}$.
To prove \cref{eq:robust_proof_pass1_confidence_tail}, we must find
at least one fixed sign for which $Z_\theta$ exceeds its stated
threshold with probability at least $2\delta$.

\paragraph{Proof using the Bretagnolle--Huber inequality.}
Fix an arbitrary learner. By the same divergence calculation as in
\cref{eq:robust_proof_pass1_neighbor_KL}, using the results from~\citet[Exercises~14.10 and~14.12]{lattimore2019bandit}
                                             
                                  
                                       
                                  
                                 
                                                         
and ~\citet[Appendix~A.2]{de2021bandits},
                                            
\[
    \begin{aligned}
        D_{\mathrm{KL}}(P_+^{\otimes m}\parallel P_-^{\otimes m})
        &=m\Delta\,\operatorname{kl}(1/2+h,1/2-h)\\
        &\leq16m\Delta h^2
        \leq\log\frac1{8\delta}.
    \end{aligned}
\]
The last inequality follows from the choice of $h$.
Write $q_i:=\widehat\pi(+1\mid i)$ for $i\in\{0,1\}$ and,
as before, decode the sign as
\[
    \widehat\theta=
    \begin{cases}
        +1,&q_1\geq1/2,\\
        -1,&q_1<1/2.
    \end{cases}
\]
Here $\mathbb P_\theta$ denotes probability under sample law
$P_\theta^{\otimes m}$, including the learner's training
randomization. Write $\mathbb P_+:=\mathbb P_{+1}$ and
$\mathbb P_-:=\mathbb P_{-1}$. The Bretagnolle--Huber inequality~\citetext{\citet[Theorem~14.2]{lattimore2019bandit};
\cite{tsybakov2009introduction}}, gives
\[
    \begin{aligned}
        \frac12\mathbb P_+\{\widehat\theta=-1\}
        +\frac12\mathbb P_-\{\widehat\theta=+1\}
        &\geq\frac14\exp\!\left\{
            -D_{\mathrm{KL}}(P_+^{\otimes m}\parallel P_-^{\otimes m})
        \right\}\\
        &\geq2\delta.
    \end{aligned}
\]
It follows that, for at least one fixed sign $\theta$, the decoding
error probability is at least $2\delta$. We choose this sign before
the training sample is drawn. The same maximization over rewards as
in \cref{eq:robust_proof_pass1_loss} gives
\[
    Z_\theta=(1-\Delta)(1-q_0)
        +\Delta\left|\frac12+\theta h-q_1\right|
    \geq\Delta\left|\frac12+\theta h-q_1\right|.
\]
If $\theta=+1$, a decoding error means $q_1<1/2$, whereas
if $\theta=-1$, it means $q_1\geq1/2$. In either case,
$|1/2+\theta h-q_1|\geq h$. Thus every decoding error implies
$Z_\theta\geq\Delta h$, and hence
\[
    \mathbb P_\theta\{Z_\theta\geq\Delta h\}
    \geq\mathbb P_\theta\{\widehat\theta\ne\theta\}
    \geq2\delta.
\]
Since $\delta<1/16$,
$\log(1/(8\delta))\geq\tfrac14\log(1/\delta)$, and the chosen $h$
satisfies
\[
    \Delta h
    \geq\min\!\left\{\frac{\Delta}{4},
        \sqrt{\frac{\Delta\log(1/\delta)}{64m}}\right\}
    >\frac1{16}\min\!\left\{
        \Delta,\sqrt{\frac{\Delta\log(1/\delta)}{m}}
    \right\}.
\]
Every decoding error therefore gives robust excess strictly above
the threshold in \cref{eq:robust_proof_pass1_confidence_tail}.
For the chosen sign, this happens with probability at least $2\delta$,
which proves the second statement.

\paragraph{Combining the two statements.}
Fix any learner and write
\[
    A:=\log N,
    \qquad
    B:=\log(1/\delta).
\]
If $A\geq B$, use the first construction, which gives a bound
involving $A$ with probability at least $1/8>2\delta$.
If $B>A$, use the second construction, which gives a bound
involving $B$ with probability at least $2\delta$.
Using the smaller threshold constant $1/128$ in both
\cref{eq:robust_proof_pass1_class_tail,eq:robust_proof_pass1_confidence_tail},
the selected admissible instance satisfies
\[
    \mathbb P\!\left\{
        Z>\frac1{128}\min\!\left\{
            \Delta,\sqrt{\frac{\Delta\max\{A,B\}}{m}}
        \right\}
    \right\}\geq2\delta,
\]
where $Z:=\sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^1(\widehat\pi,\pi_{\mathrm{demo}})$.
Since $\max\{A,B\}\geq(A+B)/2$,
\[
    \begin{aligned}
        \frac1{128}\min\!\left\{
            \Delta,\sqrt{\frac{\Delta\max\{A,B\}}{m}}
        \right\}
        &\geq\frac1{128\sqrt2}\min\!\left\{
            \Delta,\sqrt{\frac{\Delta(A+B)}{m}}
        \right\}\\
        &\geq\frac1{256}\min\!\left\{
            \Delta,\sqrt{\frac{\Delta(A+B)}{m}}
        \right\}.
    \end{aligned}
\]
Lowering the threshold can only increase the probability of exceeding
it. Substituting the definitions of $A$ and $B$, we obtain
\begin{equation}
\label{eq:robust_proof_pass1_combined_tail}
    \mathbb P\!\left\{
        \sup_{r\in{\mathcal{R}}}{\mathcal{E}}_r^1(\widehat\pi,\pi_{\mathrm{demo}})
        >\frac1{256}\min\!\left\{
            \Delta,
            \sqrt{\frac{\Delta[\log N+\log(1/\delta)]}{m}}
        \right\}
    \right\}\geq2\delta.
\end{equation}
Both constructions use the same class of $N=|{\mathcal{R}}|$ rewards on
${\mathcal{X}}=\{0,\ldots,d\}$. For each learner, the selected instance and
its hidden parameter are fixed before the sample and training
randomization are drawn.

Suppose $0<\varepsilon\leq\Delta/256$. For every integer
\[
    1\leq m\leq
    \frac{\Delta[\log N+\log(1/\delta)]}
         {262144\,\varepsilon^2},
\]
the two entries of the minimum in
\cref{eq:robust_proof_pass1_combined_tail} are at least
$256\varepsilon$ and $512\varepsilon$, respectively.
The excess threshold is therefore at least
\[
    \frac1{256}\min\{256\varepsilon,512\varepsilon\}
    =\varepsilon,
\]
and every learner fails the robust PAC requirement at each such
sample size, including the upper endpoint when it is an integer.

Zero samples also cannot suffice. In the two-sign construction,
take $h=1/4$. Without data, the decoded sign is independent of a
uniform hidden sign, so its average error probability is $1/2$.
For at least one fixed sign, the robust excess is therefore at least
$\Delta/4>\varepsilon$ with probability at least $1/2$.
Combining these failures with \cref{def:robust_pac_complexity} gives
\[
    m_{\mathrm{rob}}^{(1)}
        ({\mathcal{R}},\varepsilon,\delta,\Delta)
    >\left\lfloor
        \frac{\Delta[\log N+\log(1/\delta)]}
             {262144\,\varepsilon^2}
    \right\rfloor,
\]
which establishes the lower bound in
\cref{eq:robust_pass1_lower_samples} for the constructed class
of cardinality $N=|{\mathcal{R}}|$ for every $N\geq2$, including the integer endpoint.
This completes the proof.
