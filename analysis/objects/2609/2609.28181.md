<!-- definition-1: body_md -->
Let $F$ be a distribution with density $f$. For $\lambda\in[0,1]$, define the generalised hazard rate $r_\lambda(x)=f(x)/(1-F(x))^{1+\lambda}$. We say $F$ is \emph{$\lambda$-regular} if $r_\lambda$ is non-decreasing on the support of $F$. The set of all $\lambda$-regular distributions is denoted by $\mathcal{P}_\lambda$.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
A distribution $F$ is $\lambda$-regular if and only if it can be written as
    $$
    F(x) = 1-\left(1+\lambda R_\lambda(x)\right)^{-1/\lambda}
    $$
     where $R_\lambda(x)=\int_0^xr_\lambda(t)\mathrm{d} t$ is an increasing, convex function with $R_{\lambda}(0) = 0$.
<!-- end lemma-1 -->
<!-- theorem-1: body_md -->
\label{thm:mhr-pricing}
	For $n\ge 1$, 
$$
Q_{n,0}:=\sup_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\inf_{F\in\mathcal{P}_0}\frac{\mathbb{E}[\mathrm{Rev}(F,Y_{V_1,\dots,V_n})]}{\mathbb{E}[V]}=\left(\frac{n}{n+1}\right)^{n+1},
$$
where $V,V_1\dots,V_n\overset{\text{i.i.d.}}{\sim} F$, $Y_{v_1,\dots,v_n}\sim\kappa((v_1,\dots,v_n))$. The supremum is attained by the pricing rule that is the sample mean, i.e., $Y_{V_1,\dots,V_n} = \frac{1}{n}\sum_{i=1}^n V_i$.
	\refstepcounter{theorempart}\label{thm:mhr-pricing-rule} 
    	\refstepcounter{theorempart}\label{thm:mhr-pricing-impossibility}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Recall that $F$ is $0$-regular, i.e., has a monotone hazard rate, if $r_0(x)=f(x)/(1-F(x))$ is non-decreasing.
    Consider the pricing rule which sets the price equal to the sample mean. The expected revenue for this pricing rule is
    \begin{align*}
        \mathrm{Rev}_{\text{SM}}(F)&:=\mathbb{E}\left[\mathrm{Rev}\left(F,\frac{1}{n}\sum_{i=1}^n V_i\right)\right]\\
        &=\int_{\mathbb{R}_+^n}\left(\frac{1}{n}\sum_{i=1}^nv_i\right)S\left(\frac{1}{n}\sum_{i=1}^nv_i\right)\left(\prod_{i=1}^nf(v_i)\right)\mathrm{d} v_1\dots\mathrm{d} v_n,
    \end{align*}
    where $S(x)=1-F(x)$ is the survival function. Let $H(x)=\int_0^xr_0(t)\mathrm{d} t$. Because $-S'(t)/S(t)=r_0(t)$ we have that $S(x)=\exp(-H(x))$. Moreover, observe that $H$ is convex since $r_0$ is non-decreasing. By this convexity,
    $$
        H\left(\frac{1}{n}\sum_{i=1}^nv_i\right)\le \frac{1}{n}\sum_{i=1}^nH\left(v_i\right)\Rightarrow S\left(\frac{1}{n}\sum_{i=1}^nv_i\right)\ge \prod_{i=1}^nS(v_i)^{1/n},
    $$
    because $S$ is a decreasing function in $x$.
    Consequently,
    \begin{align*}
        \mathrm{Rev}_{\text{SM}}(F)&\ge \int_{\mathbb{R}_+^n}\left(\frac{1}{n}\sum_{i=1}^nv_i\right)\prod_{i=1}^nS(v_i)^{1/n}\left(\prod_{i=1}^nh(v_i)S(v_i)\right)\mathrm{d} v_1\dots\mathrm{d} v_n\\
    &=\int_{\mathbb{R}_+^n}\left(\frac{1}{n}\sum_{i=1}^nv_i\right)\left(\prod_{i=1}^nh(v_i)S(v_i)^{1+1/n}\right)\mathrm{d} v_1\dots\mathrm{d} v_n\\
    &=\frac{1}{n}\sum_{i=1}^n\int_{\mathbb{R}_+^n}v_i\left(\prod_{i=1}^nh(v_i)S(v_i)^{1+1/n}\right)\mathrm{d} v_1\dots\mathrm{d} v_n\\
    &=\int_{\mathbb{R}_+^n} v_1\left(\prod_{i=1}^nh(v_i)S(v_i)^{1+1/n}\right)\mathrm{d} v_1\dots\mathrm{d} v_n\\
    &=\int_0^\infty v h(v)S(v)^{1+1/n}\mathrm{d} v\left(\int_0^\infty h(v)S(v)^{1+1/n}\mathrm{d} v\right)^{n-1}.
    \end{align*}
    Using $f(v_i)=h(v_i)S(v_i)$ in the first step. The penultimate step follows from symmetry. Recall that $S'(v)=-h(v)S(v)$, so with the substitution $t=S(v)$ we obtain
    \begin{align*}
        \int_0^\infty h(v)S(v)^{1+1/n}\mathrm{d} v&=-\int_0^\infty S(v)^{\frac{1}{n}}S'(v)\mathrm{d} v=-\int_1^0t^{\frac{1}{n}}\mathrm{d} t=\frac{n}{n+1}.
    \end{align*}
    The antiderivative of $S'(v)S(v)^{1/n}$ is $\frac{n}{n+1}S(v)^{1+1/n}$, so by integration by parts,
    \begin{align*}
        \int_0^\infty vh(v)S(v)^{1+1/n}\mathrm{d} v&=-\int_0^\infty v S'(v)S^{\frac{1}{n}}\mathrm{d} v\\
        &= \int_0^\infty \frac{n}{n+1}S(v)^{1+1/n}\mathrm{d} v-\left[v\frac{n}{n+1}S(v)^{1+1/n}\right]_0^\infty\\
        &=  \frac{n}{n+1} \int_0^\infty S(v)^{1+1/n}\mathrm{d} v\\
        &\ge \frac{n}{n+1}\int_0^\infty S\left(\frac{n+1}{n}v\right)\mathrm{d} v&\text{(MHR )} \\
        &=\left(\frac{n}{n+1}\right)^2\int_0^\infty S(w)\mathrm{d} w=\left(\frac{n}{n+1}\right)^2\mathbb{E}[V].
    \end{align*}
    Combining the expressions yields
    $$
        \int_0^\infty v h(v)S(v)^{1+1/n}\mathrm{d} v\left(\int_0^\infty h(v)S(v)^{1+1/n}\mathrm{d} v\right)^{n-1}\ge \left(\frac{n}{n+1}\right)^{n+1}\mathbb{E}[V]
    $$
    The above holds for all $F$, so the sup-inf problem is at least $\left(n/(n+1)\right)^{n+1}$.

<!-- theorem-2: body_md -->
\label{thm:single-sample-det}
	Let $\lambda\in(0,1)$, then
	$$
	   Q^{\mathrm{det}}_{1,\lambda}:=\sup_{\Psi:\mathbb{R}\to\mathbb{R}}\inf_{F\in\mathcal{P}_\lambda}\frac{\mathbb{E}[\mathrm{Rev}(F,\Psi(V_1))]}{\mathbb{E}[V]}=\frac{1-\lambda}{4-2\lambda},
    $$
    where $\Psi$ is a deterministic pricing rule mapping the observed sample to a posted price.
    The supremum is attained by the pricing rule $\Psi(x)=x$.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
We first establish in the following lemma that we can restrict to pricing rules that scale the sample. 

\begin{lemma}
    \label{lemma:scaling-reduction}
    Under the (mild) regularity assumption that $\lim_{t\to 0}\Psi(t)/t$ exists,\footnote{This is the same regularity assumption as in \cite[Proposition 1]{allouah2022pricing}.} it holds for $\lambda \in (0,1)$ that
    $$
    	Q^{\mathrm{det}}_{1,\lambda}:=\sup_{\Psi:\mathbb{R}\to\mathbb{R}}\inf_{F\in\mathcal{P}_\lambda}\frac{\mathbb{E}[\mathrm{Rev}(F,\Psi(V_1))]}{\mathbb{E}[V]}=\sup_{\gamma > 0}\inf_{F\in\mathcal{P}_\lambda}\frac{\mathbb{E}[\mathrm{Rev}(F,\gamma V_1)]}{\mathbb{E}[V]}.
    $$
\end{lemma}
\begin{proof}
See Appendix \ref{appendix:scaling-proof}.
\end{proof}

\noindent Based on Lemma \ref{lemma:scaling-reduction}, we next split into three cases: $\gamma=1$, $\gamma>1$ and $\gamma <1$. 

    \noindent\textit{Case 1: $\gamma = 1$.} Fix $F\in\mathcal{P}_\lambda$, then
    $$
        \frac{\mathbb{E}[\mathrm{Rev}(F, \gamma V_1)]}{\mathbb{E}[V]}=\frac{\mathbb{E}[V_1(1-F(V_1))]}{\mathbb{E}[V]}=\frac{1}{2}\frac{\mathbb{E}[\min\{V,V_1\}]}{\mathbb{E}[V]}\ge\frac{1-\lambda}{4-2\lambda},
    $$
    where the last inequality follows from \cite[Proposition 10]{schweizer2019performance}.
    Next, take the distribution $G$ with $R_\lambda(x)=x/\lambda$, then
    \begin{align*}
        &\mathbb{E}[V]=\int_0^\infty(1-G(v))\mathrm{d} v=\int_0^\infty(1+v)^{-1/\lambda}\mathrm{d} v=\left[\frac{1}{1-1/\lambda}(1+v)^{1-1/\lambda}\right]_0^\infty=\frac{\lambda}{1-\lambda}\\
        &\mathbb{E}[\min\{V,V_1\}]=\int_0^\infty(1-G(v))^2\mathrm{d} v=\frac{\lambda}{2-\lambda}, 
    \end{align*}
    so $\mathbb{E}[\min\{V,V_1\}]/(2\mathbb{E}[V])=(1-\lambda)/(4-2\lambda)$. Therefore
    $$
    \inf_{F\in\mathcal{P}_{\lambda}}\frac{\mathbb{E}[\mathrm{Rev}(F, V_1)]}{\mathbb{E}[V]}=\frac{1-\lambda}{4-2\lambda}.
    $$

    \noindent\textit{Case 2: $\gamma > 1$.} For $\alpha > 0$, let $G_\alpha$ be the distribution with $R_\lambda(x)=\max\{0,\alpha (x-1)\}$.\footnote{The everywhere differentiable choice $R_\lambda(x)=x^\alpha$ also works, but is harder to calculate with.} 
    Then $1-G_\alpha(x)=(1+\lambda\max\{0,\alpha(x-1)\})^{-1/\lambda}$ for $x\ge 0$ and $g(x)=\alpha (1+\lambda \alpha(x-1) )^{-1/\lambda-1}$ for $x\ge 1$ and zero otherwise. 
    The mean is
    $$
    \mathbb{E}[V]=\int_0^\infty(1-G_\alpha(v))\mathrm{d} v=1+\int_1^\infty(1+\lambda\alpha(v-1))^{-1/\lambda}\mathrm{d} v=1+\frac{1}{\alpha(1-\lambda)}
    $$
    and the revenue is bounded by 
    \begin{align*}
        \mathbb{E}[\mathrm{Rev}(G_\alpha, \gamma V_1)]&=\int_0^\infty\gamma v(1-G_\alpha(\gamma v))g(v)\mathrm{d} v=\gamma\int_1^\infty v(1-G_\alpha(\gamma v))g(v)\mathrm{d} v\\
        &\le \gamma(1-G_\alpha(\gamma ))\int_1^\infty vg(v)\mathrm{d} v= \gamma(1-G_\alpha(\gamma ))\mathbb{E}[V].
    \end{align*}
    Therefore 
    $$
    \lim_{\alpha\to \infty}\frac{\mathbb{E}[\mathrm{Rev}(G_\alpha, \gamma V_1)]}{\mathbb{E}[V]}\le \lim_{\alpha\to \infty}\gamma(1-G_\alpha(\gamma ))=  \lim_{\alpha\to \infty}\gamma (1+\lambda\alpha(\gamma-1))^{-1/\lambda}=0.
    $$
    Consequently, for $\gamma > 1$ there is no performance guarantee, that is,
    $$
    \sup_{\gamma > 1}\inf_{F\in\mathcal{P}_{\lambda}}\frac{\mathbb{E}[\mathrm{Rev}(F, \gamma V_1)]}{\mathbb{E}[V]}=0.
    $$

    \textit{Case 3: $\gamma < 1$.} 
    Consider the distribution $G$ again with $R_\lambda(x)=x/\lambda$ and let
        \begin{align*}
        I(\gamma)= \frac{\mathbb{E}[\mathrm{Rev}(G, \gamma V_1)]}{\mathbb{E}[V]} &=\frac{1}{\mathbb{E}[V]}\int_0^\infty \gamma v(1+\gamma v)^{-1/\lambda}\frac{1}{\lambda}(1+v)^{-1/\lambda-1}\mathrm{d} v\\
        &=\frac{1-\lambda}{\lambda^2}\gamma\int_0^\infty v(1+\gamma v)^{-1/\lambda}(1+v)^{-1/\lambda-1}\mathrm{d} v
    \end{align*}
    If $I(\gamma)$ is an increasing function, then 
    \begin{equation*}
        \sup_{\gamma \in(0,1)}\inf_{F\in\mathcal{P}_{\lambda}}\frac{\mathbb{E}[\mathrm{Rev}(F, \gamma V_1)]}{\mathbb{E}[V]}\le \sup_{\gamma \in(0,1)}\frac{\mathbb{E}[\mathrm{Rev}(G, \gamma V_1)]}{\mathbb{E}[V]}=I(\gamma)<I(1)=\frac{1-\lambda}{4-2\lambda},
    \end{equation*}
    from which it follows, in combination with Case 1 that no $\gamma < 1$ can attain the supremum in $Q_{1,\lambda}^\mathrm{det}$. It therefore remains to prove  that $I(\gamma)$ is increasing on $(0,1)$, which is done in the following lemma.  
    \begin{lemma}
    \label{lemma:special-function-monotonicity}
    For fixed $\lambda\in(0,1)$, the function 
    $$
    I(\gamma) = \frac{1-\lambda}{\lambda^2}\gamma\int_0^\infty v(1+\gamma v)^{-1/\lambda}(1+v)^{-1/\lambda-1}\mathrm{d} v
    $$ is increasing on $(0,1)$.
\end{lemma}
The proof of Lemma \ref{lemma:special-function-monotonicity} is deferred to Appendix \ref{app:hyper-geometric}. In a nutshell, we can rewrite $I(\gamma)$ as the product of $\gamma$ and a hypergeometric function by a variable substitution and applying an identity due to Euler. We then use a derivative formula for hypergeometric functions together with a so-called contiguous relation, in the spirit of Gauss' contiguous relations for hypergeometric functions, to argue about the derivative of $I(\gamma)$.
This completes the proof of the theorem.

<!-- cleverproof-1: body_md -->
First of all, by Lemma \ref{lemma:C-unimodality} the map $r\mapsto C(n,r,\lambda)$ is unimodal, so it is maximised for $r^*=\max\{r:C(n,r,\lambda)>C(n,r-1,\lambda)\}$. Lemma \ref{lemma:const-gen-hazard-order-stat} gives an upper bound on the inner infimum, so
    $$
    \max_{r\in\{1,\dots,n\}}\inf_{F \in \mathcal{P}_\lambda} \frac{\mathbb{E}\big[\mathrm{Rev}(F,V_{r:n} ))\big]}{\mathbb{E}[V]}\le \max_{r\in\{1,\dots,n\}}C(n,r,\lambda)=C(n,r^*,\lambda).
    $$
    We now derive a matching lower bound, meaning we need to show for all $F \in \mathcal{P}_{\lambda}$ and $r = r^*$ that
    \begin{align}
    \mathbb{E}[V_{r:n}(1-F(V_{r:n}))]-C(n,r,\lambda)\mathbb{E}[V] \geq 0.
        \label{eq:C-lower-bound}
    \end{align}
    In fact, we will show \eqref{eq:C-lower-bound} for every $r \leq r^*$, and because the maximum of $C(n,r,\lambda)$ is attained at $r = r^*$, $C(n,r^*,\lambda)$ it is the best lower bound we can obtain in this way.
    We first define 
     \begin{align*}
    g(x):=A(1-F(x))(1-F(x))^{-\lambda}f(x) \text{ with } A(t)=r\binom{n}{r}\frac{B(t;n-r+2,r)}{t}-C(n,r,\lambda),
     \end{align*}
    and $h(x) := 1/r_{\lambda}(x) = (1-F(x))^{\lambda + 1}/f(x)$ so that, using Lemma \ref{lemma:const-gen-hazard-order-stat} in the first equality,
    \begin{flalign*}
    & \mathbb{E}[V_{r:n}(1-F(V_{r:n}))]-C(n,r,\lambda)\mathbb{E}[V]\\
     & \ \  = r\binom{n}{r}\int_0^\infty B(1-F(x); n-r+2,r)\mathrm{d} x-C(n,r,\lambda)\int_0^\infty(1-F(x))\mathrm{d} x\\
      & \ \  =\int_0^\infty A(1-F(x))(1-F(x))\mathrm{d} x\\
       & \ \   =\int_0^\infty A(1-F(x))\frac{(1-F(x))^{-\lambda}f(x)}{r_\lambda(x)}\mathrm{d} x\\
        & \ \  = \int_0^\infty h(x)g(x)\mathrm{d} x.
    \end{flalign*}
To show that $\int_0^\infty h(x)g(x)\mathrm{d}x \geq 0$, which establishes \eqref{eq:C-lower-bound} for a given $r$, we employ the following lemma, whose proof is included in Appendix \ref{appendix:order-statistic-policy} for completeness.
\begin{lemma}[Chapter 4, Lemma 7.1 in \cite{barlow1981statistical}]
    \label{lemma:barlow}
    Let $g$ be a function with $g(0)\ge 0$, $\int_0^\infty g(x)\mathrm{d} x=0$ and that changes sign at most once in $[0,\infty)$ and let $h$ be a non-increasing function, then
    $$
    \int_0^\infty h(x)g(x)\mathrm{d} x\ge 0.
    $$
\end{lemma}

\noindent The remainder of the proof is now devoted to showing that our $g$ and $h$ as defined above satisfy the conditions of Lemma \ref{lemma:barlow}. First, note that the function $r_{\lambda}$ is non-decreasing by definition, and so the function $h$ is therefore non-increasing. We next prove that the three properties required for $g$ hold true.

Note that $g(0) = A(1)f(0)$, because $F(0) = 0$, and so $g(0) \geq 0$ if and only if $A(1) \geq 0$ as $f(0) \geq 0$ is always true for a probability density function. Observe that
    \begin{align}
    \label{eq:A-1-condition}
        A(1)  = \; &r\binom{n}{r}B(n-r+2,r)-C(n,r,\lambda)>0\nonumber\\
        \Leftrightarrow \; &r\binom{n}{r}B(n-r+2,r)-\frac{1-\lambda}{\lambda}r\binom{n}{r}\left(B(n+2-r-\lambda, r)-B(n+2-r, r)\right)>0\nonumber\\
        \Leftrightarrow \; &B(n+2-r,r) > (1-\lambda)B(n+2-r-\lambda,r)\nonumber\\
        \Leftrightarrow \; &C(n,r,\lambda) > C(n,r-1,\lambda),
    \end{align}
where the first equivalence follows by plugging in the definition of $C(n,r,\lambda)$, the second by simplification, and the third by Lemma \ref{lemma:constant-beta-equivalence}. By the unimodality of $r \mapsto C(n,r,\lambda)$ established in Lemma \ref{lemma:C-unimodality} it follows \eqref{eq:A-1-condition} is satisfied for all $r \leq r^*$. To conclude, for all $r \leq r^*$ we have $g(0) > 0$.

Next, using the substitution $s = 1 - F(x)$, so that $\mathrm{d}s = -f(x)dx$, we obtain
    \begin{align*}
        \int_0^\infty g(x)\mathrm{d} x&=\int_0^\infty A(1-F(x))(1-F(x))^{-\lambda}f(x) \mathrm{d} x\\
        &= - \int_1^0 A(s)s^{-\lambda} \mathrm{d} s
        = \int_0^1 A(s)s^{-\lambda} \mathrm{d} s\\
        &= r\binom{n}{r}\int_0^1 B(s,n-r+2,r)s^{-\lambda-1} \mathrm{d} s+\frac{C(n,r,\lambda)}{1-\lambda}\\
        &= r\binom{n}{r}\int_0^1 \int_0^s t^{n-r+1}(1-t)^{r-1}s^{-\lambda-1}\mathrm{d} t \mathrm{d} s+\frac{C(n,r,\lambda)}{1-\lambda}\\
        &= r\binom{n}{r}\int_0^1  t^{n-r+1}(1-t)^{r-1}\int_t^1s^{-\lambda-1}\mathrm{d} s \mathrm{d} t+\frac{C(n,r,\lambda)}{1-\lambda}\\
        &= \frac{r}{\lambda}\binom{n}{r}\int_0^1  t^{n-r+1}(1-t)^{r-1}(t^{-\lambda}-1)\mathrm{d} t+\frac{C(n,r,\lambda)}{1-\lambda}\\
        &= \frac{r}{\lambda}\binom{n}{r}\left(B(n+2-r-\lambda, r)-B(n+2-r, r)\right)+\frac{C(n,r,\lambda)}{1-\lambda}=0.
    \end{align*}
where the last equality holds by definition of $C(n,r,\lambda)$.

It remains to show that $g$ changes sign at most once on $[0,\infty)$. Because $(1-F(x))^{-\lambda}f(x)\ge 0$, the sign of $g$ is uniquely determined by the sign of the function $A$.  To argue about the sign of $A$ we use the following lemma.

\begin{lemma}
    \label{lemma:unimodal-beta-average}
    The function $\alpha(t)=\frac{1}{t}B(t;a,b)$ with $a>1,b\ge 1$ is unimodal on the interval $[0,1]$ with a maximum attained at some $t^* \in (0,1)$. 
\end{lemma}

Because $1 - F(x)$ is non-increasing, the above lemma implies that also the mapping $x\mapsto A(1-F(x))$ is unimodal. This is equivalent to saying that the mapping $t \mapsto A(t)$ for $t \in (0,1]$ is unimodal, because $0 \leq 1 - F(x) \leq 1$. Together with the already established fact that $A(1) > 0$, this implies that the mapping $t \mapsto A(t)$ changes sign at most once, because in order for this mapping to change sign twice, it is necessary that $A(0) < 0$ and $A(1) < 0$.

To conclude, all conditions on $g$ in Lemma \ref{lemma:barlow} are satisfied as long as $r \leq r^*$, and so we have established \eqref{eq:C-lower-bound} for every such $r$, and therefore in particular for $r = r^*$.

Let $(\Omega_1,\mathcal{A}_1)$ and $(\Omega_2,\mathcal{A}_2)$ be measurable spaces. 
    A Markov kernel is a map $\kappa:\Omega_1\times\mathcal{A}_2\to[0,1]$ satisfying the following two properties.
    \begin{itemize}
        \item [(i)] For each $\omega_1\in\Omega_1$, the map $\kappa(\omega_1,\cdot):\mathcal{A}_2\to[0,1]$ is a probability measure.
        \item [(ii)] For each $A_2\in\mathcal{A}_2$, the map $\kappa(\cdot, A_2):\Omega_1\to[0,1]$ is $\mathcal{A}_1$-measurable.
    \end{itemize}

<!-- proof-4: body_md -->
Let $F_\theta(x)=1-e^{-\theta x}$ denote the distribution of the $\mathrm{Expo}(\theta)$ distribution. Note that it has a constant hazard rate, so $F_\theta\in\mathcal{P}_0$. Since $\{F_{\theta}:\theta>0\}\subset\mathcal{P}_0$ we have that
$$
Q_{n,0}\le\sup_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\inf_{F\in\{F_{\theta}:\theta>0\}}\frac{\mathbb{E}[\mathrm{Rev}(F,Y_{V_1,\dots,V_n})]}{\mathbb{E}[V]}.
$$
This sup-inf problem on the right hand side can be rewritten as the following infinite linear program.
    \begin{align*}
        &\sup && C\\
        &\text{subject to} &&\frac{\mathbb{E}[\mathrm{Rev}(F,Y_{Z_1\dots,Z_n})]}{\mathbb{E}[Z]}\ge C\text{ for all } \theta>0,
        \\
        &&& \kappa((z_1,\dots,z_n))\in\mathcal{P}(\mathbb{R}_+)\text{ for all }(z_1,\dots,z_n)\in\mathbb{R}_+^n\text{, } C\in\mathbb{R},
    \end{align*}
where $Z,Z_1,\dots,Z_n\sim \mathrm{Expo}(\theta)$.
    The pointwise constraint on $\kappa$ is simply the definition of $\kappa\in\mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)$.
    The left hand side of the first expectation can be worked out
    \begin{align*}
        h(\theta,\kappa)&:=\frac{\mathbb{E}[\mathrm{Rev}(F,Y_{Z_1\dots,Z_n})]}{\mathbb{E}[Z]}=\theta \int_{\mathbb{R}_+^n}\prod_{i=1}^n\theta e^{-\theta z_i}\int_0^\infty se^{-\theta s} \kappa((z_1,\dots,z_n), \mathrm{d} s)\mathrm{d} \boldsymbol{z}\\
        &=\int_{\mathbb{R}^n_+}\int_0^\infty\theta^{n+1}se^{-\theta\left(s+\sum_{i=1}^nz_i\right)} \kappa(\boldsymbol{z},\mathrm{d} s)\mathrm{d} \boldsymbol{z}.
    \end{align*}
    The constraint then becomes $h(\theta,\kappa)\ge C$ for all $\theta>0$. The dual of the above LP can be written down directly, but as it is quite an intricate LP the derivation is provided.
    The Lagrangian of the problem is 
    $$\mathcal{L}(C,\kappa,\mu)=C+\int_0^\infty (h(\theta,\kappa)-C)\mathrm{d} \mu(\theta)=C(1-\mu((0,\infty)))+\int_0^\infty h(\theta,\kappa)\mathrm{d} \mu(\theta)$$
    with $\mu$ a non-negative measure on $(0,\infty)$. 
    The dual function is $$\phi(\mu)=\sup_{C\in\mathbb{R},\kappa\in\mathcal{K}(\mathbb{R}^n_+,\mathbb{R}_+)}\mathcal{L}(C,\kappa,\mu).$$
    By weak duality, $D :=\inf_{\mu\ge 0}\phi(\mu)$ is an upper bound on the primal problem. Observe that $\phi(\mu)=+\infty$ if $\mu((0,\infty))\ne1$, so the infimum can be taken over $\mu\in\mathcal{P}((0,\infty))$. Consequently,
    \begin{align*}
        D&=\inf_{\mu\in\mathcal{P}((0,\infty))}\sup_{\kappa\in\mathcal{K}(\mathbb{R}^n_+,\mathbb{R}_+)}\int_0^\infty 
        \int_{\mathbb{R}^n_+}\int_0^\infty\theta^{n+1}se^{-\theta\left(s+\sum_{i=1}^nz_i\right)} \kappa(\boldsymbol{z},\mathrm{d} s)\mathrm{d} \boldsymbol{z}
        \mathrm{d} \mu(\theta)\\
        &=\inf_{\mu\in\mathcal{P}((0,\infty))} \sup_{\kappa\in\mathcal{K}(\mathbb{R}^n_+,\mathbb{R}_+)}\int_{\mathbb{R}^n_+}\int_0^\infty\int_0^\infty\theta^{n+1}se^{-\theta\left(s+\sum_{i=1}^nz_i\right)}\mathrm{d} \mu(\theta) \kappa(\boldsymbol{z},\mathrm{d} s)\mathrm{d} \boldsymbol{z}&\text{(Tonelli)}\\
        &\le\inf_{\mu\in\mathcal{P}((0,\infty))} \int_{\mathbb{R}^n_+}\sup_{s\ge 0}\int_0^\infty\theta^{n+1}se^{-\theta\left(s+\sum_{i=1}^nz_i\right)}\mathrm{d} \mu(\theta)\mathrm{d} \boldsymbol{z}.
    \end{align*}
    We use pointwise maximisation and that $\kappa(\boldsymbol{z})$ is a probability measure in the second step.
    \footnote{Equality can be established, but is omitted here as only an upper bound on the primal is of interest.
    One would need to verify that the conditions of the Kuratowski and Ryll-Nardzewski measurable selection theorem hold
\cite[Theorem 18.13]{aliprantis2006infinite}.}
    With standard linearisation, the dual permits the following formulation:
    \begin{align*}
        &\inf && \int_{\mathbb{R}_+^n}\lambda(\boldsymbol{z})\mathrm{d} \boldsymbol{z}\\
        &\text{subject to} &&
        \lambda(\boldsymbol{z})\ge\int_0^\infty \theta^{n+1}s\exp\left(-\theta\left(s+\sum_{i=1}^nz_i\right)\right)\mathrm{d} \mu(\theta)\text{ for all }\boldsymbol{z}\in\mathbb{R}_+^n\text{ and }s\ge 0,
        \\
        &&& \mu\in\mathcal{P}((0,\infty))\text{, } \lambda:\mathbb{R}_+^n\to\mathbb{R}.
    \end{align*} 
    The infimum over all $\lambda$ is upper bounded by the infimum over all $\lambda$ that can be written as $\lambda(\boldsymbol{z})=\beta\left(\sum_{i=1}^nz_i\right)$. An upper bound on the dual is therefore
    \begin{align*}
        &\inf && \int_{\mathbb{R}_+^n}\beta\left(\sum_{i=1}^nz_i\right)\mathrm{d} \boldsymbol{z}\\
        &\text{subject to} &&
        \beta(y)\ge\int_0^\infty \theta^{n+1}se^{-\theta(s+y)} \mathrm{d} \mu(\theta)\text{ for all }y\ge 0\text{ and }s\ge 0,
        \\
        &&& \mu\in\mathcal{P}((0,\infty))\text{, }\beta:\mathbb{R}_+\to\mathbb{R}.
    \end{align*}
    The objective can be simplified.
    Define a measure $\nu$ on $\mathbb{R}_+$ by 
    $$
        \nu(A):=\mathrm{Leb}\left\{\boldsymbol{z}\in\mathbb{R}_+^n:\sum_{i=1}^nz_i\in A\right\}.
    $$
    A direct computation shows that for $y\ge 0$ that $\nu([0,y])=y^n/n!$ (see Lemma \ref{lemma:simplex-volume} below), so $\mathrm{d} \nu(y)=\frac{y^{n-1}}{(n-1)!}\mathrm{d} y$. The objective hence simplifies to
    \begin{equation}
        \label{eq:objective-expo-killing-dual}
        \int_{\mathbb{R}_+^n}\beta\left(\sum_{i=1}^nz_i\right)\mathrm{d} \boldsymbol{z}=\int_0^\infty \beta(y)\mathrm{d} \nu(y)=\frac{1}{(n-1)!}\int_0^\infty y^{n-1}\beta(y)\mathrm{d} y.
    \end{equation}
    Now, consider the following dual feasible solution
    \begin{align}
        \label{eq:dual-solution-measure}
        \mathrm{d} \mu_m(\theta)&=\frac{1}{\log m}\int_1^me^{-t\theta}\mathrm{d} t\mathrm{d} \theta\\
        \label{eq:dual-solution-constraint}
        \beta_m(y)&=\begin{cases}
            \frac{(n-1)!}{\log m},&\text{if }y\in[0,1),\\
            \frac{(n-1)!}{\log m}C_ny^{-n},&\text{if }y\in[1,m),\\
            \frac{n!}{\log m}\frac{m}{y^{n+1}},&\text{if }y\ge m.
        \end{cases}
    \end{align}
    First the feasibility of $(\mu_m,\beta_m)$ will be verified and then its associated objective will be computed. 
    To start, $\mathrm{d} \mu_m(\theta)\ge 0$ and
    $$
        \mu_m((0,\infty))=\int_0^\infty\frac{1}{\log m}\int_1^me^{-t\theta}\mathrm{d} t\mathrm{d} \theta=\frac{1}{\log m}\int_1^m\int_0^\infty e^{-t\theta}\mathrm{d} \theta\mathrm{d} t=\frac{1}{\log m}\int_1^m\frac{1}{t}\mathrm{d} t=1,
    $$
    so $\mu_m$ is a probability measure on $(0,\infty)$. Next the $\beta$ constraints. Define the right hand side of this constraint as
    \begin{align*}
        \xi_m(s,y)&:=\int_0^\infty \theta^{n+1}se^{-\theta(s+y)}\mathrm{d} \mu_m(\theta)=\frac{1}{\log m}\int_0^\infty\theta^{n+1}se^{-\theta(s+y)} \int_1^me^{-t\theta}\mathrm{d} t\mathrm{d} \theta\\
        &=\frac{1}{\log m}\int_1^ms\int_0^\infty \theta^{n+1}e^{-\theta(s+y+t)}\mathrm{d} \theta\mathrm{d} t=\frac{(n+1)!}{\log m}\int_1^m\frac{s}{(s+y+t)^{n+2}}\mathrm{d} t.
    \end{align*}
    Using definition of the Gamma function in the last step.
    Observe that $\beta_m(y)\ge \xi_m(s,y)$ for all $s, y\ge0$ ensures feasibility. This is established through three general bounds. 
    Observe that $\xi_m(s,y)$ is decreasing in $y$, with this observation we recover the bound
    \begin{align*}
        \xi_m(s,y)&\le\xi_m(s,0)=\frac{(n+1)!}{\log m}\int_1^m\frac{s}{(s+t)^{n+2}}\mathrm{d} t\\
        &\le \frac{(n+1)!}{\log m}\int_1^\infty\frac{s}{(s+t)^{n+2}}\mathrm{d} t=\frac{n!}{\log m}\frac{s}{(s+1)^{n+1}}\\
        &\le \frac{n!}{\log m}\frac{1/n}{(1/n+1)^{n+1}}&\text{(maximum at }s=1/n\text{)}\\
        &= \frac{(n-1)!}{\log m}\left(\frac{n}{n+1}\right)^{n+1}<\frac{(n-1)!}{\log m}.

To start, fix $\lambda\in(0,1)$ and let $F_\alpha(x)=1-\left(1+\frac{\lambda}{1-\lambda}\frac{x}{\alpha}\right)^{-1/\lambda}$. Note that $F_\alpha\in \mathcal{P}_\lambda$ and that $\{F_\alpha:\alpha>0\}\subset \mathcal{P}_\lambda$, so
                                                                                                                                                                                                                                                                                                                                                                                                                    
$$
Q_{n,\lambda}\le\sup_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\inf_{F\in\{F_\alpha:\alpha>0\}}T(\kappa,F)=(1-\lambda)^{1/\lambda}-\inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\sup_{\alpha>0}\left((1-\lambda)^{1/\lambda}-T(\kappa,F_\alpha)\right).
$$
Using that the mean of $V \sim F_{\alpha}$ is $\alpha$, which can be checked by computing $\mathbb{E}[V] = \int_0^\infty (1-F_\alpha(x))\mathrm{d}x$, we find that
$$
(1-\lambda)^{1/\lambda}-T(\kappa,F_\alpha) = (1-\lambda)^{1/\lambda} - \frac{\mathbb{E}[\mathrm{Rev}(F,Y_{V_1,\dots,V_n})]}{\mathbb{E}[V]} =\mathbb{E}[\ell_\alpha(Y_{\boldsymbol{V}})],
$$
where 
$$\ell_\alpha(t)=(1-\lambda)^{1/\lambda}-\frac{1}{\alpha}\mathrm{Rev}(F_\alpha,t)=(1-\lambda)^{1/\lambda}-\frac{t}{\alpha}\left(1+\frac{\lambda}{1-\lambda}\frac{t}{\alpha}\right)^{-1/\lambda}.$$
and $Y_{\boldsymbol{V}} = Y_{V_1,\dots,V_n}$ a (possibly randomised) pricing rule. Define $\ell(t)=\ell_\alpha(\alpha t)$

In what comes next, we will sometimes use the following result on the loss function $\ell$, that yields $\ell_\alpha(\alpha+t)\ge c_0\cdot(t/\alpha)^2$ for $t\in[-\alpha,\alpha/4]$ with $c_0$ a constant that depends only on $\lambda$. 
                                                                                                                                                                                                                                                                                                                                 
\begin{lemma}
    \label{lemma:ell-quad-bound}
    Let $\lambda\in(0,1)$ and define
    $$\ell(t):=(1-\lambda)^{1/\lambda}-t\left(1+\frac{\lambda}{1-\lambda}t\right)^{-1/\lambda}.$$
    Then for all $t\in[-1,1/4]$, we have that $\ell(1+t)\ge \frac{1}{2}\ell''(5/4)t^2$.
\end{lemma}







Let $\alpha_0=1$, $\alpha_n=1+t_n$, $t_n=h/\sqrt{n}$, and $h=1/2$. Let $m_n=(\alpha_0+\alpha_n)/2$ and define $\phi_n=\mathbbm{1}\!\!\left\{Y_{\boldsymbol{V}}\ge m_n\right\}$. We use the so-called test $\phi_n$ to derive lower bounds on $\mathbb{E}[\ell_{a_j}(Y_{\boldsymbol{V}})]$ for $j = 0,n$. There are two cases to consider.
First, consider the event $\{\phi_n=1\}$. In that case, 
$$Y_{\boldsymbol{V}}\mathbbm{1}\!\!\left\{\phi_n = 1\right\}\ge m_n\mathbbm{1}\!\!\left\{\phi_n = 1\right\}=(\alpha_0+t_n/2)\mathbbm{1}\!\!\left\{\phi_n = 1\right\}.$$ 
Because the loss function $\ell_{\alpha_0}$ is increasing after $\alpha_0$, we have that 
$$\ell_{\alpha_0}(Y_{\boldsymbol{V}})\mathbbm{1}\!\!\left\{\phi_n = 1\right\}\ge\ell_{\alpha_0}(\alpha_0+t_n/2)\mathbbm{1}\!\!\left\{\phi_n = 1\right\}.$$ 
Note that $t_n/2=1/(4\sqrt{n})\le 1/4= \alpha_0/4$, so $\ell_{\alpha_0}(\alpha_0+t_n/2)\ge \frac{1}{4}c_0(t_n/\alpha_0)^2=\frac{1}{4}c_0t_n^2$. This yields
\begin{align*}
    \mathbb{E}[\ell_{\alpha_0}(Y_{\boldsymbol{V}})]&\ge \mathbb{E}[\ell_{\alpha_0}(Y_{\boldsymbol{V}})\mathbbm{1}\!\!\left\{\phi_n=1\right\}]
    \\
    &\ge \mathbb{E}\left[\frac{c_0}{4}t_n^2\mathbbm{1}\!\!\left\{\phi_n=1\right\}\right]=\frac{c_0}{4}t_n^2\mathbb{P}_{\alpha_0}(\phi_n=1).
\end{align*}
Second, consider the event $\{\phi_n=0\}$. In that case, $Y_{\boldsymbol{V}}<m_n=\alpha_n-t_n/2$. Because the loss function $\ell_{\alpha_n}$ is decreasing before $\alpha_n$, we have that 
$$
\ell_{\alpha_n}(Y_{\boldsymbol{V}})\mathbbm{1}\!\!\left\{\phi_n = 0\right\}\ge \ell_{\alpha_n}(\alpha_n-t_n/2)\mathbbm{1}\!\!\left\{\phi_n = 0\right\}\ge\frac{1}{4}c_0\frac{t_n^2}{a_n^2}\mathbbm{1}\!\!\left\{\phi_n = 0\right\}\ge \frac{1}{9}c_0t_n^2\mathbbm{1}\!\!\left\{\phi_n = 0\right\}.
$$
Using $\alpha_n\le3/2$ for the last inequality. We then analogously find
$$
\mathbb{E}[\ell_{\alpha_n}(Y_{\boldsymbol{V}})]\ge \frac{1}{9}c_0t_n^2\mathbb{P}_{\alpha_n}(\phi_n=0).
$$
With these steps we can do the main step.
\begin{align*}
    \inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\sup_{\alpha>0}\mathbb{E}[\ell_{\alpha}(Y_{\boldsymbol{V}})]&\ge \inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\max\{\mathbb{E}[\ell_{\alpha_0}(Y_{\boldsymbol{V}})],\mathbb{E}[\ell_{\alpha_n}(Y_{\boldsymbol{V}})]\}\\
    &\ge \frac{1}{2}\inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}(\mathbb{E}[\ell_{\alpha_0}(Y_{\boldsymbol{V}})]+\mathbb{E}[\ell_{\alpha_n}(Y_{\boldsymbol{V}})])\\
    &\ge \frac{1}{2}\inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\left(\frac{c_0}{4}t_n^2\mathbb{P}_{\alpha_0}(\phi_n=1)+\frac{c_0}{9}t_n^2\mathbb{P}_{\alpha_n}(\phi_n=0)\right)\\
    &\ge \frac{c_0}{18}t_n^2\inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\left(\mathbb{P}_{\alpha_0}(\phi_n=1)+\mathbb{P}_{\alpha_n}(\phi_n=0)\right)\\
    &= \frac{c_0}{18}t_n^2\inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\left(1+\mathbb{P}_{\alpha_0}(\phi_n=1)-\mathbb{P}_{\alpha_n}(\phi_n=1)\right)\\
    &= \frac{c_0}{18}t_n^2\left(1-\sup_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\left(\mathbb{P}_{\alpha_n}(\phi_n=1)-\mathbb{P}_{\alpha_0}(\phi_n=1)\right)\right)\\
    &\ge  \frac{c_0}{18}t_n^2\left(1-\vert\vert P_{\alpha_n}-P_{\alpha_0} \vert\vert_{\mathrm{TV}}\right).
\end{align*}
The last step follows from the fact that the total variation distance provides an upper bound on the difference between the probabilities of any given event under two distributions. Using Pinsker's inequality and the additivity of the Kullback-Leibler divergence yields
$$
\vert\vert P_{\alpha_n}-P_{\alpha_0}\vert\vert_{\mathrm{TV}}\le\sqrt{\frac{1}{2}D_{\mathrm{KL}}(P_{\alpha_0}||P_{\alpha_n})}=\sqrt{\frac{1}{2}nD_{\mathrm{KL}}(F_{\alpha_0}||F_{\alpha_n})}.
$$
We next use the following lemma to upper bound the Kullback-Leibler divergence.

\begin{lemma}
    \label{lemma:kl-div-bound}
    Let $\lambda\in(0,1)$, $F_\alpha(x)=1-\left(1+\frac{\lambda}{1-\lambda}\frac{x}{\alpha}\right)^{-1/\lambda}$ and let $\varepsilon \in (0, 1)$, then
    $$
    D_{\mathrm{KL}}(F_{1}||F_{1+\varepsilon})\le \frac{\varepsilon^2}{1-\varepsilon}.
    $$
\end{lemma}
Applying Lemma \ref{lemma:kl-div-bound} with $\varepsilon = t_n$, we obtain
$$
\vert\vert P_{\alpha_n}-P_{\alpha_0}\vert\vert_{\mathrm{TV}}\le\sqrt{\frac{n}{2}\frac{t_n^2}{1-t_n}}=\sqrt{\frac{h^2}{2\left(1-h/\sqrt{n}\right)}}\le \sqrt{\frac{h^2}{2(1-h)}}=\frac{1}{2}.
$$
Using $h=1/2$ in the last step. With this bound on the total variation distance we find that
$$
    \inf_{\kappa\in \mathcal{K}(\mathbb{R}_+^n,\mathbb{R}_+)}\sup_{\alpha>0}\mathbb{E}[\ell_{\alpha}(Y_{\boldsymbol{V}})]\ge  \frac{c_0}{36}t_n^2=\frac{c_0}{144}\frac{1}{n}.
$$
This shows that
\begin{align*}
    Q_{n,\lambda}&\le (1-\lambda)^{1/\lambda}-\frac{c_0}{144}\frac{1}{n}=(1-\lambda)^{1/\lambda}-\frac{\ell''(5/4)}{288}\frac{1}{n}\\

\end{align*}
