<!-- assumption-1: body_md -->
\label{asm:domain}
The set $\mathcal Y\subseteq\mathbb R^{d_y}$ is nonempty, closed, and convex, and contains the origin.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{asm:smooth}
The function $f$ is continuously differentiable on a neighborhood of $\mathbb R^{d_x}\times\mathcal Y$ and is jointly $L$-smooth on this domain:
\[
\left\|\nabla f(x,y)-\nabla f(x',y')\right\|\le L\left\|[x-x';y-y']\right\|
\]
for all $x,x'\in\mathbb R^{d_x}$ and $y,y'\in\mathcal Y$.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{asm:concave}
For every $x\in\mathbb R^{d_x}$, the function $f(x,\cdot)$ is $\mu$-strongly concave on $\mathcal Y$, where $0<\mu\le L$:
\[
f(x,y)\le f(x,y')+\left\langle\nabla_yf(x,y'),y-y'\right\rangle-\frac\mu2\left\|y-y'\right\|^2
\]
for all $y,y'\in\mathcal Y$.
<!-- end assumption-3 -->
<!-- assumption-4: body_md -->
\label{asm:gap}
The primal function $\Phi(x)=\max_{y\in\mathcal Y}f(x,y)$ is bounded below and satisfies
\[
\Phi(0)-\inf_{x\in\mathbb R^{d_x}}\Phi(x)\le\Delta.
\]
<!-- end assumption-4 -->
<!-- definition-1: body_md -->
\label{def:oracle-model}
An oracle $\mathcal O$ for $f$ has variance at most $\sigma^2$ if, for every feasible query $(x,y)$,
\begin{equation}\label{eq:oracle-model}
\mathbb E_\xi[\mathcal O(x,y;\xi)]=\nabla f(x,y),\qquad
\mathbb E_\xi\left\|\mathcal O(x,y;\xi)-\nabla f(x,y)\right\|^2\le\sigma^2.
\end{equation}
Each call uses a fresh independent sample $\xi$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:zero}
An algorithm starts at $(x^0,y^0)=(0,0)$ and has zero-query output $\hat x^0=0$. Every query $(x^t,y^t)$ lies in $\mathbb R^{d_x}\times\mathcal Y$, and every output $\hat x^t$ lies in $\mathbb R^{d_x}$. Write $(g_x^s,g_y^s)=\mathcal O(x^s,y^s;\xi^s)$. The algorithm is zero-respecting if, for every $t\ge1$, its next query and its output after $t$ calls satisfy
\begin{align}\label{eq:zero}
\operatorname{supp}(x^t)\cup\operatorname{supp}(\hat x^t)&\subseteq\bigcup_{s=0}^{t-1}\operatorname{supp}(g_x^s),&
\operatorname{supp}(y^t)&\subseteq\bigcup_{s=0}^{t-1}\operatorname{supp}(g_y^s).
\end{align}
Queries may be adaptive and may use internal randomness independent of the oracle samples.

<!-- theorem-1: body_md -->
\label{thm:sapd-tight}
Let $L,\mu,G,\varepsilon>0$ and $\sigma\ge0$ satisfy
\begin{equation}\label{eq:tight-regime}
\kappa=\frac L\mu\ge C,\qquad
\varepsilon^2\le\frac{LG}{440C},\qquad \sigma\ge\varepsilon.
\end{equation}
For $\eta=1/(2L)$, the worst-case stochastic first-order oracle complexity over $\mathcal M(L,\mu,G)$ is
\begin{equation}\label{eq:main-tight}
\mathfrak N_{\rm env}(\mathcal M(L,\mu,G),\sigma,\varepsilon)
=\Theta\!\left(\frac{\kappa LG\sigma^2}{\varepsilon^4}\right).
\end{equation}
The upper bound is attained by SAPD+ with a uniformly selected outer iterate. The lower bound holds even when the primal component of the oracle is deterministic.

<!-- theorem-2: body_md -->
\label{thm:lower}
Let $C=370$. For any $L,\mu,\Delta,\varepsilon>0$ and $\sigma\ge0$ satisfying
\begin{equation}\label{eq:regime}
\kappa=\frac L\mu\ge C,\qquad
\varepsilon^2\le\frac{L\Delta}{96C},
\end{equation}
there exist a problem $f\in\mathcal F(L,\mu,\Delta)$ with a bounded dual box and an oracle of variance at most $\sigma^2$ such that every zero-respecting algorithm satisfies $\mathbb E\left\|\nabla\Phi(\hat x^N)\right\|>\varepsilon$ whenever
\begin{equation}\label{eq:lower-budget}
N\le c_0\left(\frac{L\Delta}{\varepsilon^2}
+\frac{\kappa L\Delta\sigma^2}{\varepsilon^4}\right),
\qquad c_0=\frac{1}{1536e^2C^2}.
\end{equation}
The primal component of the oracle is deterministic.

<!-- lemma-6: body_md -->
\label{lem:comparison}
Suppose $f\in\mathcal M(L,\mu,G)$ and $\left\|\nabla^2_{xy}f\right\|\le b$ throughout its domain. Then $\Phi$ is $(L+b^2/\mu)$-smooth. For $\eta=1/(2L)$,
\begin{equation}\label{eq:comparison-general}
\left\|\nabla\Phi(x)\right\|\le\left(1+\eta\left(L+\frac{b^2}{\mu}\right)\right)
\left\|\nabla\Phi_\eta(x)\right\|\qquad\text{for every }x.
\end{equation}
In particular, if $b\le\sqrt{\mu L}$, then
\begin{equation}\label{eq:comparison}
\left\|\nabla\Phi(x)\right\|\le2\left\|\nabla\Phi_\eta(x)\right\|.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-4: body_md -->
Let $y=y^*(x)$ and $y'=y^*(x')$. The optimality inequalities for the two constrained maximizations and $\mu$-strong concavity imply
\[
\mu\left\|y-y'\right\|^2
\le\left\langle\nabla_y f(x,y')-\nabla_y f(x',y'),y-y'\right\rangle
\le b\left\|x-x'\right\|\left\|y-y'\right\|.
\]
Thus $y^*$ is $(b/\mu)$-Lipschitz, including at the boundary of the box. Danskin's formula gives
\begin{align*}
\left\|\nabla\Phi(x)-\nabla\Phi(x')\right\|
&\le L\left\|x-x'\right\|+b\left\|y^*(x)-y^*(x')\right\|\\
&\le(L+b^2/\mu)\left\|x-x'\right\|.
\end{align*}
For $z=\operatorname{prox}_{\eta\Phi}(x)$, the first-order optimality condition is
\[
\nabla\Phi(z)=\frac{x-z}{\eta}=\nabla\Phi_\eta(x).
\]
Combining this identity with the smoothness just proved yields
\[
\left\|\nabla\Phi(x)\right\|
\le\left\|\nabla\Phi(z)\right\|+(L+b^2/\mu)\left\|x-z\right\|
=\big(1+\eta(L+b^2/\mu)\big)\left\|\nabla\Phi_\eta(x)\right\|.
\]
If $b^2\le\mu L$, the factor is at most $2$.

The maximum of $4e\exp(-s^{-2})s^{-3}$ is $\sqrt{54/e}$. For the second derivative, let $r=s^{-2}$. The stationary points of $(2r^3-3r^2)e^{-r}$ on $(0,\infty)$ occur at

Evaluation at these two points, together with the limits at zero and infinity, gives

The bounds on $\psi$ and its vanishing derivatives on $(-\infty,1/2]$ follow directly from~\eqref{eq:scalar}.

For the second scalar function,

The stated derivative bounds follow by maximizing these expressions. The Gaussian integral gives $0<\varphi(t)<\sqrt{2\pi e}$, and symmetry gives $\varphi(-t)=2\varphi_0-\varphi(t)$. Thus $\bar\varphi$ is odd and $|\bar\varphi(t)|<\varphi_0$.

\[

and, for $i<T$, the outgoing contribution is
\[
-\psi'(-u_i)\varphi(-u_{i+1})-\psi'(u_i)\varphi(u_{i+1}).
\]
At $i=1$, the initial contribution is $-\varphi'(u_1)$. If $|u_T|<1$, let $i$ be the first index with $|u_i|<1$. If $i=1$, then $\partial_1F_T(u)\le-\varphi'(u_1)<-1$. Otherwise $|u_{i-1}|\ge1$, so

Since $\varphi'$ is even and $\varphi'(u_i)>1$, the incoming contribution to $\partial_iF_T(u)$ is less than $-1$. The other contributions are nonpositive. Hence $\left\|\nabla F_T(u)\right\|>1$ in either case.

\end{align}

We prove the following invariant until the first time $P_t=T$: every discovered dual link has index at most $P_t$. Fix a time $t$ for which $P=P_t<T$ and the invariant holds. All primal indices below lie in $[T]$, and all dual-link indices lie in $[T-1]$. The support restrictions give $x_k^t=0$ and $y_k^{t,\pm}=0$ for indices $k>P$.

If link $P$ has not been discovered, then $y_P^{t,\pm}=0$ as well. Equation~\eqref{eq:gx} shows that $g_{x_k}^t=0$ for every $k\ge P+1$, so $P_{t+1}=P$. Equation~\eqref{eq:gy} shows that no dual link with index larger than $P$ can be revealed. At link $P$, its exact dual gradient is $c\psi(\pm u_P)$. If $x_P^t=0$, both components vanish. Otherwise $j(x^t)=P$, and the oracle returns the two components as $(\xi^t/p)c\psi(\pm u_P)$. Hence link $P$ can first be discovered only when $\xi^t=1$. The invariant is preserved.

If link $P$ has already been discovered, then~\eqref{eq:gx} gives $g_{x_k}^t=0$ for all $k\ge P+2$, and therefore $P_{t+1}\le P+1$. For each $k\ge P+1$, the entries $u_k,u_{k+1},y_k^{t,\pm}$ vanish, so~\eqref{eq:gy} gives zero on link $k$. The perturbation in~\eqref{eq:oracle} also vanishes there because $j(x^t)\le P$. Thus no dual link with index larger than $P$ is discovered on this call, and the invariant again holds.

Every increase from $P$ to $P+1$ therefore requires a distinct earlier success. Indeed, link $P$ cannot be discovered before primal coordinate $P$, or on the same call that reveals that coordinate. Its subsequent discovery requires $\xi^t=1$ while $P_t=P$. Since $P_t$ is nondecreasing, counting these discoveries gives $P_N-1\le\sum_{t=0}^{N-1}\xi^t$ until $P_N=T$. Thereafter $P_N=T$ and at least $T-1$ successes have already occurred, so the inequality continues to hold. This proves~\eqref{eq:progress}.

\label{lem:terminal}
For $M=10\sqrt T$ and $x_{\mathrm{good}}=\lambda M\mathbf1_T$,
\begin{equation}\label{eq:terminal}
\left\|\nabla\Phi(x_{\mathrm{good}})\right\|\le\frac{\varepsilon}{10T}<\frac\varepsilon2,
\qquad \left\|\nabla\Phi(0)\right\|=2\sqrt e\,\varepsilon.
\end{equation}
<!-- end lemma-7 -->
<!-- proof-11: body_md -->
Since $M\ge10$, all negative-argument $\psi$ and $\psi'$ terms vanish. Differentiating~\eqref{eq:chain} gives, for each $i\in[T]$,
\[
|[\nabla F_T(M\mathbf1_T)]_i|
\le e\sqrt e\,e^{-M^2/2}+\sqrt{2\pi e}\,\psi'(M).
\]
We have $\psi'(M)=4\psi(M)/(2M-1)^3\le4e/M^3$ and $e^{-M^2/2}\le M^{-3}$. Using $e\sqrt e+4e\sqrt{2\pi e}<50$, we obtain
\[
\left\|\nabla F_T(M\mathbf1_T)\right\|
\le\frac{50\sqrt T}{M^3}=\frac{1}{20T}.
\]
Multiplication by $L_0\lambda=2\varepsilon$ proves the first claim. The second follows from $\nabla F_T(0)=-\sqrt e\,e_1$.

\enlargethispage{\baselineskip}
The lower bound in~\eqref{eq:pair-finite} follows from the proof of Theorem~\ref{thm:lower}, including the zero-query case. For the upper bound, run Algorithm~\ref{alg:discovery}, which truncates the discovery procedure after $N=\lceil8m\rceil$ calls. The output is $x_{\mathrm{good}}$ if discovery finishes and zero otherwise. The terminal point is legal because all primal coordinates have been discovered; zero is always a legal output. By~\eqref{eq:waiting} and Markov's inequality,
\[
\mathbb P(\tau>N)\le\frac{m}{N}\le\frac18.
\]
Lemma~\ref{lem:terminal} therefore gives
\[
\mathbb E\left\|\nabla\Phi(\hat x)\right\|
\le\frac\varepsilon2+\frac18(2\sqrt e\,\varepsilon)
=\left(\frac12+\frac{\sqrt e}{4}\right)\varepsilon<\varepsilon.
\]
This proves~\eqref{eq:pair-finite} with a deterministic call budget.

Finally, $T\ge2$ and $0<p\le1$ imply that both bounds in~\eqref{eq:pair-finite} are of order $T/p$. By~\eqref{eq:parameters} and~\eqref{eq:p},
\[
T=\Theta\!\left(\frac{L\Delta}{\varepsilon^2}\right),\qquad
\frac1p=1+\frac{\kappa\sigma^2}{4e^2C\varepsilon^2}.
\]
The constant $C=370$ is fixed, so their product gives~\eqref{eq:pair-rate} with absolute implicit constants.
