<!-- theorem-1: body_md -->
\label{thm:main}
\begin{enumerate}
\item There is an absolute constant $c>0$ such that, for every $H>0$, there is a finite ONS $\Phi=(\phi_n)_{n=1}^N$ on a probability space $(\Omega,\mu)$, with $|\phi_n|=1$, such that every permutation $\tau$ admits a sum $f=\sum_{n=1}^N a_n\phi_n$ with real coefficients and $\sum_{n=1}^N|a_n|^2=1$ satisfying
\[
 \mu\{\mathcal M_\tau f>H\}\geq c.
\]
\item There is a complete ONS $\{\psi_n\}_{n\geq1}$ on a probability space $(\Omega,\mu)$, with $|\psi_n|=1$, such that for every permutation $\tau$ of $\mathbb N$ there are real coefficients $(a_n)\in\ell^2$ for which
\[
 \sup_{q\geq1}\left|\sum_{j=1}^q a_{\tau(j)}\psi_{\tau(j)}(x)\right|=\infty
\]
for almost every $x\in\Omega$.
\end{enumerate}

<!-- theorem-2: body_md -->
\label{thm:quantitative}
There are absolute constants $c,c_0>0$ such that, for every sufficiently large $N$, there is an ONS $\Phi_N=(\phi_n)_{n=1}^N$, with $\phi_n(x)\in\{-1,1\}$, having the following property. For every permutation $\tau$ of $[N]$, there is a sum $f=\sum_{n=1}^N a_n\phi_n$ with real coefficients and $\sum_{n=1}^N|a_n|^2=1$ satisfying
\begin{equation}\label{eq:sharp-lower}
 \|\mathcal M_\tau f\|_2\geq c\log\log N
\end{equation}
and
\begin{equation}\label{eq:weak-lower}
 \mu\{\mathcal M_\tau f>c\sqrt{\log\log N}\}\geq c_0.
\end{equation}
When $N$ is a power of two, the system consists of two differently labeled copies of the $N$ Walsh characters.

\label{lem:two}
Let $\pi$ be a permutation of $[m]$. There exist $N\geq m$ and a permutation $\sigma$ of $[N]$ such that, for every permutation $\tau$ of $[N]$, there is an $m$-term arithmetic progression $x_1<\cdots<x_m$ in $[N]$ for which $x_{\pi(1)},\ldots,x_{\pi(m)}$ is a subsequence of at least one of

 \qquad\text{and}\qquad

<!-- lemma-2: body_md -->
\label{lem:count}
Fix integers $m\geq2$ and $K\geq m$, and a permutation $\pi$ of $[m]$. The number of colorings $c:[N]\to[K]$ containing no $m$-term arithmetic progression $x_1<\cdots<x_m$ with
\begin{equation}\label{eq:pattern}
 c(x_{\pi(1)})<\cdots<c(x_{\pi(m)})
\end{equation}
is at most
\begin{equation}\label{eq:count}
 (m-1)^N\exp(o(N)).
\end{equation}
Here $N\to\infty$ with $m$, $K$, and $\pi$ fixed.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Let $\mathcal F_N$ denote the family in the statement. We count its members by deleting the points $N,N-1,\ldots,1$ in turn. Consider a family $\mathcal A$ of colorings of $[r]$, and let $\mathcal A_i$ be the colorings of $[r-1]$ which extend to a member of $\mathcal A$ by giving $r$ the color $i$. If each remaining coloring had at most $m-1$ extensions, deletion would cost at most a factor $m-1$. In general, we have the following inequality, where the intersections account for the additional extensions.
\begin{equation}\label{eq:recursion}
 |\mathcal A|\leq(m-1)\left|\bigcup_{i=1}^K\mathcal A_i\right|
 +\sum_{\substack{Q\subseteq[K]\\|Q|=m}}
 \left|\bigcap_{i\in Q}\mathcal A_i\right|.
\end{equation}
To see this, consider a coloring of $[r-1]$ with $t\geq1$ extensions. It contributes $t$ to the left-hand side and $m-1+\binom tm$ to the right-hand side, and $t\leq m-1+\binom tm$.

Apply \eqref{eq:recursion} to $\mathcal F_N$, then to each family on the right, and continue until all points have been deleted. For one nonzero term in the resulting sum, let $S$ be the points at which an intersection was chosen and let $Q_x$ be its set of $m$ colors at $x\in S$. Each of the other deletions contributes a factor $m-1$, so the term has weight $(m-1)^{N-|S|}$. It remains to bound the number of points in $S$.

We claim that any choice of a color from $Q_x$ at each point $x\in S$ extends to a single coloring in $\mathcal F_N$. To see this, restore the points in reverse order. At a point of $S$, membership in the intersection allows every color in $Q_x$. At any other point, membership in the union allows at least one color. We can therefore make each of the chosen assignments as that point is restored. The colors outside $S$ need not be the same for different assignments on $S$.

Fix a set $Q=\{q_1<\cdots<q_m\}\subseteq[K]$. We next observe that the set of $x\in S$ with $Q_x=Q$ contains no $m$-term arithmetic progression. Otherwise, given such a progression $x_1<\cdots<x_m$, we could assign color $q_j$ to $x_{\pi(j)}$, make arbitrary allowed choices at the other points of $S$, and extend to a member of $\mathcal F_N$. This contradicts the defining property of $\mathcal F_N$.

Let $r_m(N)$ denote the maximal size of a subset of $[N]$ containing no $m$-term arithmetic progression, and let $R:=\binom Km$. Each of the $R$ possible sets $Q$ occurs at no more than $r_m(N)$ points of $S$, so $|S|\leq Rr_m(N)$. Summing over the choices of $S$ and the sets $Q_x$, we obtain
\begin{equation}\label{eq:explicit-count}
 |\mathcal F_N|\leq
 \sum_{j=0}^{\min(N,Rr_m(N))}
 \binom Nj R^j(m-1)^{N-j}.
\end{equation}
By Szemer\'edi's 1975 theorem \cite{Szemeredi}, $r_m(N)=o(N)$. Since $m$ and $K$ are fixed, \eqref{eq:explicit-count} is at most $(m-1)^N\exp(o(N))$, which completes the proof.
<!-- end proof-1 -->
<!-- proof-2: body_md -->
We may assume $m\geq2$, since the case $m=1$ is immediate. Let $K=m^2$ and take $N$ divisible by $K$. We call a coloring \emph{balanced} if each color occurs $N/K$ times, and let $\mathcal B_N$ denote the family of balanced colorings. Stirling's formula gives
\[
 |\mathcal B_N|=\frac{N!}{((N/K)!)^K}=K^N\exp(o(N)).
\]
Choose $\sigma$ uniformly at random. For a fixed $c\in\mathcal B_N$, the coloring $c\circ\sigma^{-1}$ is uniformly distributed on $\mathcal B_N$. Writing $\mathcal F_N$ for the family in Lemma~\ref{lem:count}, the expected number of balanced colorings such that both $c$ and $c\circ\sigma^{-1}$ belong to $\mathcal F_N$ is
\[
 \frac{|\mathcal F_N\cap\mathcal B_N|^2}{|\mathcal B_N|}
 \leq\left(\frac{(m-1)^2}{m^2}\right)^N\exp(o(N)).
\]
For $N$ sufficiently large this is less than one. Thus there is a choice of $\sigma$ for which no balanced coloring has this property.

Given a permutation $\tau$, define the balanced coloring $c$ by
\[
 c(\tau(j))=\ell
 \quad\text{when}\quad
 (\ell-1)N/K<j\leq\ell N/K.
\]
The corresponding coloring for $\sigma\circ\tau$ is $c\circ\sigma^{-1}$. By the choice of $\sigma$, at least one lies outside $\mathcal F_N$ and hence contains a progression satisfying \eqref{eq:pattern}. Its terms occur in the required order.

<!-- lemma-3: body_md -->
\label{lem:fourier}
There is $\delta>0$ such that, for every $H>0$, there are $m$, a permutation $\pi$ of $[m]$, and real coefficients $b_1,\ldots,b_m$ satisfying
\begin{equation}\label{eq:fourier}
 \sum_{j=1}^m b_j^2=1,\qquad
 \left|\left\{x\in\mathbb T:\max_{q\leq m}
 \left|\sum_{j\leq q}b_j e(\pi(j)x)\right|>H\right\}\right|\geq\delta.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Nakata's Lemma~4 \cite{Nakata1971} gives, for arbitrarily large integers $k$, trigonometric polynomials $S_1,\ldots,S_L$ with pairwise disjoint Fourier supports such that
\[
 \int_{\mathbb T}\sum_{j=1}^L |S_j(x)|^2\,dx\lesssim k^{-1},
\]
and, on an interval of fixed positive measure, for every $x$ there are $1\leq u\leq v\leq L$ for which
\[
 \left|\sum_{j=u}^v S_j(x)\right|\gtrsim1.
\]
Order the Fourier characters blockwise according to $S_1,\ldots,S_L$. The last display is the difference of the partial sums ending after blocks $v$ and $u-1$, so one of these two partial sums has absolute value bounded below by an absolute constant. By Parseval and the disjointness of the supports, the squared sum of all coefficients is $O(k^{-1})$. After normalization the maximal sum is therefore $\gtrsim\sqrt k$ on a set of fixed positive measure.

The coefficients in the exponential expansion may be complex. Write the normalized coefficient vector as $c_j=\alpha_j+i\beta_j$, with $\alpha_j,\beta_j$ real. On the set where the corresponding maximal sum is greater than $2H$, at least one of the maximal sums associated to $(\alpha_j)$ and $(\beta_j)$ is greater than $H$. One of these two sets has at least half the measure. The selected real coefficient vector has squared sum at most one, and normalization can only increase its maximal sum. Taking $k$ sufficiently large, translating the frequencies to positive integers, and inserting missing frequencies with coefficient zero gives the lemma.

<!-- lemma-4: body_md -->
\label{lem:walsh-example}
For each integer $r\geq1$, there is an ordering $p_1,\ldots,p_m$ of $\mathbb F_2^{6r}$, where $m=2^{6r}$, and real coefficients $b_1,\ldots,b_m$ with $\sum_j b_j^2=1$ such that
\begin{equation}\label{eq:walsh-seed}
 \left\|\max_{0\leq q\leq m}
 \left|\sum_{j=1}^q b_j\chi_{p_j}\right|\right\|_2
 \geq\frac{r}{\sqrt{48}}
\end{equation}
and
\begin{equation}\label{eq:walsh-seed-weak}
 \mu\left\{\max_{0\leq q\leq m}
 \left|\sum_{j=1}^q b_j\chi_{p_j}\right|>\frac{\sqrt r}{8}\right\}
 \geq\frac1{12}.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
We construct polynomials on $(\mathbb Z/4\mathbb Z)^{2r}$ with disjoint Fourier supports, then express them as Walsh polynomials with disjoint supports. Their partial sums can then be realized as partial sums of a rearranged Walsh series by placing the characters of each polynomial consecutively.

Let $x=(x_0,\ldots,x_{r-1})$ and $y=(y_0,\ldots,y_{r-1})$ denote independent uniformly distributed elements of $(\mathbb Z/4\mathbb Z)^r$, and define
\[
 \epsilon(0)=\epsilon(2)=0,\qquad
 \epsilon(1)=1,\qquad\epsilon(3)=-1.
\]
For $1\leq n\leq r$ and a word $a=(a_0,\ldots,a_{n-1})\in\{0,1,2,3\}^n$, let
\begin{equation}\label{eq:triangular}
 z_j(a):=y_j+\sum_{i=0}^j a_i x_{j-i}\pmod4,
 \qquad 0\leq j<n,
\end{equation}
and
\begin{equation}\label{eq:quaternary-functions}
 F_{n,a}(x,y):=\epsilon(z_{n-1}(a))
 \prod_{j=0}^{n-2}\mathbf 1_{\{z_j(a)=0\}}.
\end{equation}
For $n=1$ the product is omitted. For fixed $x$ and $a$, the variables $z_j(a)$ are independent and uniformly distributed as $y$ varies. It follows that
\begin{equation}\label{eq:block-energy}
 \|F_{n,a}\|_2^2=2\cdot4^{-n},\qquad
 \sum_{n=1}^r\sum_{a\in\{0,1,2,3\}^n}\|F_{n,a}\|_2^2=2r.
\end{equation}

To verify disjointness of the Fourier supports, expand the indicators in \eqref{eq:quaternary-functions} and use $\epsilon(z)=(i^z-i^{-z})/(2i)$. Each character which occurs has exponent
\[
 \sum_{j=0}^{n-1}u_jz_j(a),\qquad
 u_j\in\mathbb Z/4\mathbb Z,\quad u_{n-1}\in\{1,3\}.
\]
The coefficients of the $y$-coordinates determine $n$ and the $u_j$. Since $u_{n-1}$ is odd, the coefficient $u_{n-1}a_0$ of $x_{n-1}$ then determines $a_0$. Comparing the remaining $x$-coefficients in reverse order determines $a_1,a_2,\ldots$ in turn. Thus no character occurs in two different functions $F_{n,a}$. By Parseval and \eqref{eq:block-energy}, the squared coefficient norm of their sum is $2r$.

We order the functions lexicographically in $a$, putting a word before its proper extensions. Suppose first that $x_0$ is odd. The equations $z_j(a)=0$ can then be solved successively, giving a unique word $w=(w_0,\ldots,w_{r-1})$ of length $r$. We take the partial sum ending just before $w$. At length $n$, a nonzero term must have first $n-1$ digits $w_0,\ldots,w_{n-2}$, and it precedes $w$ precisely when its last digit is less than $w_{n-1}$. The term with last digit $w_{n-1}$ is zero. The contribution from words of length $n$ is therefore
\[
 \sum_{b<w_{n-1}}\epsilon(x_0(b-w_{n-1}))
 =-\epsilon(x_0)\mathbf 1_{\{w_{n-1}\in\{1,2\}\}}.
\]
Writing
\[
 D_r(w):=|\{j:w_j\in\{1,2\}\}|,
\]
this partial sum has absolute value $D_r(w)$, since all the contributions have the same sign. For fixed $x$ with $x_0$ odd, the word $w$ is uniformly distributed as $y$ varies. In particular,
\begin{equation}\label{eq:digit-moment}
 \mathbb E D_r=r/2,\qquad \mathbb E D_r^2\geq r^2/4.
\end{equation}

This gives a lower bound of order $\sqrt r$ after normalization by \eqref{eq:block-energy}. We obtain a larger $L^2$ lower bound by also considering points where some initial coordinates of $x$ and $y$ vanish. On these sets the same construction occurs in fewer coordinates, with each term repeated many times.

For $0\leq k<r$, we define
\[
 E_k:=\{x_0=\cdots=x_{k-1}=y_0=\cdots=y_{k-1}=0,
                  \ x_k\text{ odd}\}.
\]
The sets $E_k$ are disjoint and have measures $16^{-k}/2$. On $E_k$, let $x'=(x_k,\ldots,x_{r-1})$ and $y'=(y_k,\ldots,y_{r-1})$. The functions with $n\leq k$ vanish, and for $1\leq\ell\leq r-k$ the definition gives
\[
 F_{k+\ell,a}(x,y)
 =F_{\ell,(a_0,\ldots,a_{\ell-1})}(x',y').
\]
The right-hand side does not depend on the last $k$ digits of $a$. Consequently, each function in the construction on $(x',y')$ occurs $4^k$ times.

Let $w$ denote the solution word for $(x',y')$, and consider the partial sum ending just before the length-$r$ word obtained by appending $k$ zeros to $w$. If a word in the shorter construction is not a prefix of $w$, then its first differing digit determines whether it precedes the cutoff, independently of the $k$ appended digits. If the word is a prefix of $w$, its function is zero. Thus this partial sum is exactly $4^k$ times the partial sum previously considered for $(x',y')$.

Let $S_r$ denote this partial sum on each $E_k$, and set $S_r=0$ outside their union. The factor $16^k$ from squaring cancels the measure factor $16^{-k}$, so \eqref{eq:digit-moment} gives
\begin{equation}\label{eq:sum-events}
 \|S_r\|_2^2
 \geq\sum_{k=0}^{r-1}\frac{16^{-k}}2\,16^k\,
                 \frac{(r-k)^2}{4}
 =\frac18\sum_{j=1}^r j^2
 \geq\frac{r^3}{24}.
\end{equation}

It remains to express the construction in Walsh characters. To preserve disjointness of Fourier supports, we use three bits for each coordinate modulo four. Let $\alpha,\beta,\gamma$ denote independent uniform bits and define
\[
 z:=\alpha+2\beta+\gamma\pmod4,\qquad
 A:=(-1)^\alpha,\quad B:=(-1)^\beta,\quad D:=(-1)^\gamma.
\]
The variable $z$ is uniform modulo four and independent of $\gamma$, and a direct computation gives
\[
\begin{array}{c|c|c}
 h&i^\gamma i^{hz}&\text{Walsh support}\\ \hline
 0&i^\gamma&\{1,D\}\\
 1&BD\,i^\alpha&\{BD,ABD\}\\
 2&A\,i^{-\gamma}&\{A,AD\}\\
 3&B\,i^{-\alpha}&\{B,AB\}.
\end{array}
\]
The supports follow by expanding factors such as $i^\alpha=(1+i)/2+(1-i)A/2$, and are pairwise disjoint.

Make this substitution independently in each of the $2r$ coordinates of $(x,y)$, writing $\gamma_1,\ldots,\gamma_{2r}$ for the additional bits, and set $\rho:=i^{\gamma_1+\cdots+\gamma_{2r}}$. Multiplication by $\rho$ factors coordinatewise, so the table shows that distinct characters on $(\mathbb Z/4\mathbb Z)^{2r}$ give disjoint Walsh supports. It follows that the $\rho F_{n,a}$ have disjoint Walsh supports on $6r$ bits. Taking real parts gives
\[
 G_{n,a}:=(\operatorname{Re}\rho)F_{n,a}.
\]
Since Walsh characters are real-valued, taking real parts does not introduce any new frequencies, so the supports remain disjoint. Moreover, $\operatorname{Re}\rho$ is independent of $(x,y)$ and has absolute value one with probability $1/2$, and zero otherwise. Combining this with \eqref{eq:block-energy} and \eqref{eq:sum-events} gives
\[
 \sum_{n,a}\|G_{n,a}\|_2^2=r,\qquad
 \|(\operatorname{Re}\rho)S_r\|_2^2\geq r^3/48.
\]

We expand each $G_{n,a}$ in Walsh characters, place its characters consecutively, and order these blocks according to the ordering of the $F_{n,a}$. Any unused characters are included with coefficient zero. At the ends of the blocks, the partial sums agree with those of the $G_{n,a}$, while the squared coefficient norm is $r$. Dividing the coefficients by $\sqrt r$ proves \eqref{eq:walsh-seed}.

On $E_0$ we have $|S_r|=D_r$. Since $\mathbb E D_r=r/2$ and $D_r\leq r$, it follows that $\mathbb P(D_r\geq r/4)\geq1/3$. Together with $\mu(E_0)=1/2$, this gives $|S_r|\geq r/4$ on a set of measure at least $1/6$. Independently, $|\operatorname{Re}\rho|=1$ with probability $1/2$. Thus the normalized maximal sum is at least $\sqrt r/4$ on a set of measure at least $1/12$, proving \eqref{eq:walsh-seed-weak}.

<!-- lemma-5: body_md -->
\label{lem:cube}
Let $m=2^s$, where $s\geq1$, and let $N=2^d$. A subset of $\mathbb F_2^d$ containing no affine $s$-dimensional subspace has size less than $2N^{1-1/m}$.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Let $A\subseteq\mathbb F_2^d$ have density $\delta$, and let $x,v_1,\ldots,v_s$ be independent uniformly distributed elements of $\mathbb F_2^d$. We write $C_s(A)$ for the probability that
\[
 x+\omega_1v_1+\cdots+\omega_sv_s\in A
 \quad\text{for every }\omega\in\mathbb F_2^s.
\]
Fixing $v_s$, this is the condition that an $(s-1)$-dimensional cube be contained in $A\cap(A+v_s)$. Starting with $C_0(A)=\delta$, induction and Jensen's inequality give
\[
 C_s(A)=\mathbb E_v C_{s-1}(A\cap(A+v))
 \geq\mathbb E_v\left(\frac{|A\cap(A+v)|}{N}\right)^{2^{s-1}}
 \geq\delta^{2^s}.
\]
Here we used the identity $\mathbb E_v|A\cap(A+v)|/N=\delta^2$.

If the directions are linearly independent, the cube is an affine $s$-dimensional subspace. Under the hypothesis on $A$, every cube counted by $C_s(A)$ must therefore have dependent directions. On the other hand, each nonzero binary linear combination of the directions vanishes with probability $1/N$, so the probability of dependence is at most $(2^s-1)/N$. We conclude that $\delta^m<m/N$, and hence
\[
 |A|<m^{1/m}N^{1-1/m}\leq2N^{1-1/m}.\qedhere
\]

<!-- lemma-6: body_md -->
\label{lem:walsh-two}
There is an absolute constant $C$ such that the following holds. Let $s\geq1$, let $m=2^s$, and fix an ordering $p_1,\ldots,p_m$ of $\mathbb F_2^s$. If
\begin{equation}\label{eq:dimension}
 d\geq C m^2\log m,
\end{equation}
then there is a permutation $\sigma$ of $X=\mathbb F_2^d$ with the following property. For every ordering $\tau$ of $X$, an affine image of $p_1,\ldots,p_m$ occurs as a subsequence of $\tau$ or of $\sigma\circ\tau$.
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Let $N=2^d$, $K=16m^2$, and $R=\binom Km$. We apply \eqref{eq:general-count} to the ordered affine images of $p_1,\ldots,p_m$. Since $\log R\leq m\log(16em)$, taking $C$ sufficiently large in \eqref{eq:dimension} ensures that
\[
 N^{1/m}>2R^2.
\]
It follows from Lemma~\ref{lem:cube} that every set containing no affine $s$-dimensional subspace has size less than $N/R^2$.

Let $\mathcal F$ denote the family of colorings $c:X\to[K]$ for which there is no affine injection $L:\mathbb F_2^s\to X$ satisfying
\[
 c(L(p_1))<\cdots<c(L(p_m)).
\]
Applying \eqref{eq:general-count} with $r<N/R^2$, we have
\begin{align*}
 |\mathcal F|
 &\leq\sum_{0\leq j\leq N/R}\binom NjR^j(m-1)^{N-j}\\
 &\leq R^{N/R}\sum_{j=0}^N\binom Nj(m-1)^{N-j}
 =R^{N/R}m^N\leq(2m)^N.
\end{align*}
The last inequality follows from $R^{1/R}\leq2$.

Increasing $C$ if necessary, \eqref{eq:dimension} gives $d\geq2s+4$, and hence $K\mid N$. The number of balanced colorings is at least
\[
 \frac{N!}{((N/K)!)^K}\geq(K/e)^N.
\]
For a uniformly random permutation $\sigma$, the expected number of balanced colorings for which both $c$ and $c\circ\sigma^{-1}$ belong to $\mathcal F$ is therefore bounded by
\[
 \frac{(2m)^{2N}}{(K/e)^N}=(e/4)^N<1.
\]
We may therefore choose $\sigma$ such that no balanced coloring has this property. For any ordering $\tau$, color its consecutive blocks of size $N/K$ as in Lemma~\ref{lem:two}. At least one of the colorings corresponding to $\tau$ and $\sigma\circ\tau$ does not belong to $\mathcal F$, and the resulting ordered affine image is the required subsequence.
