<!-- theorem-1: body_md -->
\label{th:wp-KNM}
	Let $\varepsilon>0$ and $T>0$ be fixed. Assuming \eqref{eq:assumption_F}, for any initial data $\mathbf{u}_0^\varepsilon:=(u_{i,0}^\varepsilon,u_{e,0}^\varepsilon,s_0^\varepsilon) \in  \mathbf{X}^\varepsilon$ satisfying \eqref{eq:compatibility}, there exists a unique solution
	\[
	\mathbf{u}^\varepsilon:=(u_i^\varepsilon,u_e^\varepsilon,s^\varepsilon)\in C^1([0,T]; \mathbf{X}^\varepsilon)
	\]
	of the (KNM$-\varepsilon$) system, satisfying $\mathbf{u}^\varepsilon(0)=\mathbf{u}_0^\varepsilon$ and the a priori estimates:
	\begin{equation}
		\label{apriori-estimate}
		\begin{aligned}
			\sup_{t \in [0,T]} \left( b^\varepsilon(\bar{u}^\varepsilon(t)) + a^\varepsilon(\bar{u}^\varepsilon(t)) +\phi^\varepsilon(\bar{u}^\varepsilon(t)) +  |s^\varepsilon(t)|^2_\varepsilon  \right) 
			\hspace{4cm} \\
			+ \int_0^T \left( |\partial_t v^\varepsilon(t)|_\varepsilon^2 + |\partial_t s^\varepsilon(t)|_\varepsilon^2 \right) dt 
			\leq C \left( b^\varepsilon(\bar{u}_0^\varepsilon) + \phi^\varepsilon(\bar{u}_0^\varepsilon) + a^\varepsilon(\bar{u}_0^\varepsilon) +|s_0^\varepsilon|^2_\varepsilon \right),
		\end{aligned}
	\end{equation}
	where $v^\varepsilon=u_i^\varepsilon-u_e^\varepsilon$, $\bar u^\varepsilon = (u_i^\varepsilon,u_e^\varepsilon)$, and the constant $C > 0$ is independent of $\varepsilon$. In addition, for each $t \in [0,T]$, the potentials $(u_i^\varepsilon, u_e^\varepsilon)$ attain the minimum constrained energy configuration for $a^\varepsilon$, that is: $a^\varepsilon(\bar{u}^\varepsilon(t)) = \min \left\{ a^\varepsilon({u}) : {u}_i - {u}_e = v^\varepsilon(t) \right\}.$

<!-- theorem-2: body_md -->
\label{th:main}
	Let $(\varepsilon_j)_{j\in \mathbb{N}}$ be a sequence of positive real numbers converging to $0$. Let $\mathbf{u}_0 := (u_{i,0}, u_{e,0}, s_0) \in H^1(\Omega) \times H_\circ^1(\Omega) \times L^2(\Omega)$. Suppose that $\mathbf{u}_0^{\varepsilon_j} := (u_{i,0}^{\varepsilon_j}, u_{e,0}^{\varepsilon_j}, s_0^{\varepsilon_j}) \in \mathbf{X}^{\varepsilon_j}$ is a sequence of initial data satisfying \eqref{eq:compatibility} such that, as $j \to \infty$, 
	\[
	\Pi_{\varepsilon_j}\mathbf{u}_0^{\varepsilon_j} \to \mathbf{u}_0\quad \text{strongly in } L^2(\Omega)^3\quad \text{and}\quad
	\limsup_{j\to \infty} a^{\varepsilon_j}(\bar u_0^{\varepsilon_j}) +\phi^{\varepsilon_j}(\bar u_0^{\varepsilon_j}) < \infty. 
	\]
	Then, there exists a subsequence $(\varepsilon_{j_k})$ and two positive-definite conductivity tensors $M_i, M_e \in L^\infty(\Omega;  \mathbb{R}^{3 \times 3}_{sym})$ such that the sequence of discrete solutions $\mathbf{u}^{\varepsilon_{j_k}}$ of the KNM$-\varepsilon_{j_k}$ system satisfies 
	\[
	\Pi_{\varepsilon_{j_k}}\mathbf{u}^{\varepsilon_{j_k}}(t) \to \mathbf{u}(t)\quad \text{strongly in } L^2(\Omega)^3
	\] 
	for every $t \in [0,T]$, where $\mathbf{u}$ is the unique solution of the macroscopic bidomain model (BD), complemented by \eqref{eq:FHN}--\eqref{eq:zeromean}, with conductivity tensors $M_i$ and $M_e$.

<!-- remark-1: body_md -->
\label{rem:laplacian}
	We collect here some general properties of the graph Laplacian. We refer to \cite{CH97} for a complete study. 
	
	\begin{itemize}
		\item[(i)] The operator \(\mathcal{L}_G\) is linear, symmetric, and positive semidefinite: for all $u,w \in X$
		\begin{align}
			\langle \mathcal{L}_G u, w \rangle &= \sum_{k \in \mathcal{V}} \sum_{j \in \mathcal{N}_k} G^{j,k} (u^k - u^j) w^k   \\
			&=  \sum_{\{j,k\}\in \mathcal{E}} G^{j,k} (u^k - u^j) (w^k - w^j) =\langle u, \mathcal{L}_G w \rangle.\label{5}
		\end{align}
		In particular,
		\[
		\langle \mathcal{L}_G u, u \rangle 	=  \sum_{\{j,k\}\in \mathcal{E}} G^{j,k} (u^k - u^j)^2 \geq 0
		\] 
		and, since $\mathcal{G}$ is connected, $\mathcal L_Gu =0$ if and only if $u$ is constant. 
		\item[(ii)] \, Let $X_0:=\left\{ w\in X : \sum_{k \in \mathcal{V}}w^k=0\right\}$. Then $\mathcal{L}_G(X)=X_0$. Indeed, let $f \in X$ be such that  $f^k=1$ for all $k\in \mathcal{V}$. For any \(w \in X\)
		\[
		\sum_{k \in \mathcal{V}} (\mathcal{L}_G w)^k = \langle \mathcal{L}_G w, f \rangle =  \langle w, \mathcal{L}_G  f \rangle =0.
		\]
		\item[(iii)] \, The restriction of $\mathcal L_G$ to $X_0$ is positive definite and the map $\mathcal{L}_G|_{X_0} : X_0 \to X_0$ is a linear isomorphism and is thus invertible. 
	\end{itemize}	
	When $G=1$,  $\mathcal L_G$ can be interpreted as the discrete divergence of a discrete gradient, which makes it the natural graph analogue of the continuous Laplacian.

<!-- lemma-1: body_md -->
\label{lemma:g}
	There exists $G>0$ (independent of $\varepsilon$) such that the linear  form $\mathbf{g}^\varepsilon$ satisfies:
	\begin{equation*}
		(\mathbf{g}^\varepsilon(\mathbf{u}, \widehat{\mathbf{u}}))^2 \leq G^2 \mathbf{b}^\varepsilon(\mathbf{u}) \mathbf{b}^\varepsilon(\widehat{\mathbf{u}}), \quad \forall\, \mathbf{u}, \widehat{\mathbf{u}} \in \mathbf{X}^\varepsilon,
	\end{equation*}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The inequality follows directly from the definition of $\mathbf{g}^\varepsilon$ and $\mathbf{b}^\varepsilon$, via Cauchy-Schwarz inequality. The constant $G$ depends on the physical parameters ($C_m, \lambda_F, \delta, \sigma$) and is independent of $\varepsilon$.

<!-- lemma-2: body_md -->
\label{lemma:coercive}
	The bilinear forms $\mathbf{a}^\varepsilon$ and $\mathbf{b}^\varepsilon$ are continuous, symmetric, and positive semidefinite. Moreover, there exists a constant $\nu > 0$ (independent of $\varepsilon$) such that:
	\begin{equation}
		\label{eq:ab_coercive}
		\mathbf{a}^\varepsilon(\mathbf{u}) + \mathbf{b}^\varepsilon(\mathbf{u}) \geq \nu \|\mathbf{u}\|^2_{\varepsilon}, \quad \forall \,\,\mathbf{u} \in \mathbf{X}^\varepsilon.
	\end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
The first part of the statement follows directly from the definition of $\mathbf{a}^\varepsilon$ and $\mathbf{b}^\varepsilon$ and by Remark \ref{rem:laplacian}. Since $u_e \in X_0^\varepsilon$, by \eqref{eq:coercivity},
	\[	
	\mathbf{a}^\varepsilon(\mathbf{u}) + \mathbf{b}^\varepsilon(\mathbf{u}) 
	\geq \bar\alpha |u_e|_\varepsilon^2 + C_m |u_i - u_e|_\varepsilon^2 + (1+\gamma) |s|_\varepsilon^2.
	\]
	Let $C_1 := \min \{ \bar \alpha/2, C_m \}$ and observe that
	\[
	\bar\alpha |u_e|_\varepsilon^2 + C_m |u_i - u_e|_\varepsilon^2 
	\geq \frac{\bar\alpha}{2} |u_e|_\varepsilon^2 + C_1\left(|u_e|_\varepsilon^2 + |u_i - u_e|_\varepsilon^2 \right)
	\geq \frac{\bar\alpha}{2} |u_e|_\varepsilon^2 + \frac{C_1}{2} |u_i|_\varepsilon^2,
	\]
	so we can choose the coercivity constant $\nu: = \min \{ \frac{C_1}{2}, 1+\gamma \}$.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
We first derive estimate \eqref{apriori-estimate}. By testing \eqref{KNM-abstract} with $\widehat{\mathbf{u}} = \partial_t \mathbf{u}^\varepsilon$, and exploiting the symmetry of $\mathbf{a}^\varepsilon$ and the relation $f = \varphi'$, we derive the following energy identity:
	\begin{equation}
		\label{energy-id-detail}
		\mathbf{b}^\varepsilon(\partial_t \mathbf{u}^\varepsilon) + \frac{d}{dt} \left( \frac{1}{2}\mathbf{a}^\varepsilon(\mathbf{u}^\varepsilon) + \phi^\varepsilon(\mathbf{u}^\varepsilon) \right) = \mathbf{g}^\varepsilon(\mathbf{u}^\varepsilon, \partial_t \mathbf{u}^\varepsilon).
	\end{equation}
	In view of Lemma \ref{lemma:g} and Young's inequality, the right-hand side is bounded by $\frac{G^2}{2} \mathbf{b}^\varepsilon(\mathbf{u}^\varepsilon) + \frac{1}{2} \mathbf{b}^\varepsilon(\partial_t \mathbf{u}^\varepsilon)$. Integrating \eqref{energy-id-detail} over the interval $(0, t)$ yields:
	\begin{equation}
		\label{pre-gronwall}
		\frac{1}{2}\int_0^t \mathbf{b}^\varepsilon(\partial_t \mathbf{u}^\varepsilon) dz + \frac{1}{2}\mathbf{a}^\varepsilon(\mathbf{u}^\varepsilon(t)) + \phi^\varepsilon(\mathbf{u}^\varepsilon(t)) \leq E^\varepsilon(\mathbf{u}_0^\varepsilon) + \frac{G^2}{2} \int_0^t \mathbf{b}^\varepsilon(\mathbf{u}^\varepsilon) dz.
	\end{equation}
	The uniform coercivity \eqref{eq:ab_coercive}, together with the non-negativity of $\phi^\varepsilon$, implies the estimate 
	\[
	\mathbf{b}^\varepsilon(\mathbf{u}^\varepsilon(t)) \leq \widehat{C}(E^\varepsilon(\mathbf{u}_0^\varepsilon) + \int_0^t \mathbf{b}^\varepsilon(\mathbf{u}^\varepsilon) dz.
	\]	 
	Consequently, Gr\"onwall's inequality ensures that $\sup_{t \in [0,T]} \mathbf{b}^\varepsilon(\mathbf{u}^\varepsilon(t)) \leq C E^\varepsilon(\mathbf{u}_0^\varepsilon)$. By substituting this result back into \eqref{pre-gronwall}, the individual bounds for $\mathbf{a}^\varepsilon, \phi^\varepsilon$ and the $L^2$-norm of $(\partial_t v_\varepsilon, \partial_t s^\varepsilon)$ are recovered from the non-negativity of the remaining terms on the left-hand side, thereby concluding \eqref{apriori-estimate}.
	
	In Proposition \ref{prop:local}, we stated the local existence and uniqueness of a solution of \eqref{KNM-abstract} on a maximal interval $[0, t_{\max})$. If $t_{\max} < T$, the norm $\|\mathbf{u}^\varepsilon(t)\|_\varepsilon$ would necessarily blow up as $t \to t_{\max}^-$. However, the uniform energy estimate \eqref{apriori-estimate} implies:
	\[ 
	\sup_{t \in [0, t_{\text{max}} )} \left[ \mathbf{b}^\varepsilon(\mathbf{u}^\varepsilon(t)) + \mathbf{a}^\varepsilon(\mathbf{u}^\varepsilon(t)) + \mathbf{\phi}^\varepsilon(\mathbf{u}^\varepsilon(t)) \right]\leq C\left( \mathbf{b}^\varepsilon(\mathbf{u}_0^\varepsilon) + \mathbf{\phi}^\varepsilon(\mathbf{u}_0^\varepsilon) + \mathbf{a}^\varepsilon(\mathbf{u}_0^\varepsilon) \right).
	\]
	This uniform bound prevents finite-time blow-up, allowing the local solution to be extended up to $T$ by standard ODE theory \cite{CL55}.
	
	Finally, the minimum energy property is a direct consequence of the Kirchhoff constraint $\mathcal{L}_{G_i^\varepsilon} u_i^{\varepsilon} + \mathcal{L}_{G_e^\varepsilon} u_e^{\varepsilon} = 0$, which acts as the Euler-Lagrange optimality condition for $\mathbf{a}^\varepsilon$ under the relation $u_i^{\varepsilon} - u_e^{\varepsilon} = v^\varepsilon$.

<!-- proposition-2: body_md -->
For a fixed $\tau > 0$ and any given $\mathbf{U}_{n-1}^{\varepsilon,\tau} \in \mathbf{X}^\varepsilon$, the incremental minimization problem $(\mathcal{P}_n^{\varepsilon,\tau})$ admits a unique minimizer $\mathbf{U}_n^{\varepsilon,\tau} \in \mathbf{X}^\varepsilon$ for each $n \in \{ 1, \dots, N \}$. Furthermore, the sequence $\{\mathbf{U}_n^{\varepsilon,\tau}\}_{n=0}^N$ generated by these minimizations satisfies the Euler scheme $(\mathcal{P}^{\varepsilon,\tau})$. Conversely, any solution to $(\mathcal{P}^{\varepsilon,\tau})$ minimizes the functional $\Psi_n^{\varepsilon,\tau}$ at each time step.
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Fix $n \in \{ 1, \dots, N \}$ and consider the functional $\Psi_n^{\varepsilon,\tau}(\mathbf{U})$ defined in \eqref{MM-eps}. By Lemma \ref{lemma:coercive}, the quadratic form $\mathbf{a}^\varepsilon + \mathbf{b}^\varepsilon$ is coercive on $\mathbf{X}^\varepsilon$. Since $\mathbf{\phi}^\varepsilon$ is a strictly convex functional, $\Psi_n^{\varepsilon,\tau}$ is strictly convex and coercive on $\mathbf{X}^\varepsilon$. This implies the existence and uniqueness of a minimizer $\mathbf{U}_n^{\varepsilon,\tau}$. The first-order optimality condition for this minimization problem reads:
	\[
	\frac{1}{\tau} \mathbf{b}^\varepsilon(\mathbf{U}_n^{\varepsilon,\tau} - \mathbf{U}_{n-1}^{\varepsilon,\tau}, \mathbf{W}) 
	+ \mathbf{a}^\varepsilon(\mathbf{U}_n^{\varepsilon,\tau}, \mathbf{W}) 
	+ \langle \mathbf{\phi}'(\mathbf{U}_n^{\varepsilon,\tau}), \mathbf{W} \rangle - \mathbf{g}^\varepsilon(\mathbf{U}_{n-1}^{\varepsilon,\tau}, \mathbf{W}) = 0,
	\]
	for all $\mathbf{W} \in \mathbf{X}^\varepsilon$. Recognizing that $\langle \mathbf{\phi}'(\mathbf{U}), \mathbf{W} \rangle = \mathcal{F}^\varepsilon(\mathbf{U}, \mathbf{W})$, this identity coincides with the weak formulation of the Euler scheme $(\mathcal{P}^{\varepsilon,\tau})$. Conversely, any solution of the Euler scheme satisfies this Euler-Lagrange condition, which implies the minimization of $\Psi_n^{\varepsilon,\tau}$.

<!-- lemma-3: body_md -->
\label{lemma:tau_convergence}
	For fixed $\varepsilon>0$, let $\mathbf{u}^\varepsilon = (u_i^\varepsilon,u_e^\varepsilon,s^\varepsilon):(0,T)\to\mathcal{V}_0^\varepsilon$ be the unique solution of \eqref{KNM-abstract}. For all sequences $(\tau_k)_{k \in \mathbb{N}}$, $\tau_k\to 0$, 
	\begin{align}
		\widetilde{\mathbf{U}}^{\varepsilon, \tau_k} \to \mathbf{u}^\varepsilon\quad \text{strongly in }L^2(0,T; \mathcal{V}_0^\varepsilon),\label{eq:strongL2}\\
		\left(\widetilde V^{\varepsilon,\tau_k},\widetilde s^{\varepsilon,\tau_k}\right) \to \left(v^\varepsilon,s^\varepsilon\right)\quad 
		\text{strongly in }C^0([0,T]; X^\varepsilon \times X^\varepsilon), \label{eq:strongC0}\\
		\frac{d}{dt}\left(\widetilde V^{\varepsilon,\tau_k},\widetilde s^{\varepsilon,\tau_k}\right) \to \frac{d}{dt}\left(v^\varepsilon,s^\varepsilon\right)\quad 
		\text{weakly in }L^2(0,T; X^\varepsilon \times X^\varepsilon),\label{eq:weakL2}
	\end{align}	
	where $\widetilde V^{\varepsilon,\tau}:= \widetilde{U}_i^{\varepsilon, \tau}-\widetilde{U}_e^{\varepsilon, \tau}$ and $v^\varepsilon:=u_i^\varepsilon -u_e^\varepsilon$.
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
Let $(\tau_k)_{k \in \mathbb{N}}$ be fixed. By definition of $\mathbf{b}^\varepsilon$ and \eqref{cauchy-estimate-knm}, $(\widetilde V^{\varepsilon,\tau_k})_k$ and $(\widetilde{s}^{\varepsilon,\tau_k})_k$ are Cauchy sequences in $C^0([0,T]; X^\varepsilon)$. Moreover, by \eqref{eq:ab_coercive} and \eqref{cauchy-estimate-knm}
	\[
	\int_0^T  \left|\overline{\mathbf{U}}^{\varepsilon, \tau}(t) - \overline{\mathbf{U}}^{\varepsilon, \eta}(t))\right|_\varepsilon^2 \, dt \le C(\tau + \eta) E^\varepsilon(\mathbf{u}_0^\varepsilon),
	\]
	so that $(\overline{\mathbf{U}}^{\varepsilon, \tau_k})_k$ is Cauchy in $L^2(0,T;\mathcal{V}_0^\varepsilon)$. Finally, by \eqref{continuous-stability-tilde}, $\frac{d}{dt}\widetilde V^{\varepsilon,\tau_k}$ and $\frac{d}{dt}\widetilde{s}^{\varepsilon,\tau_k}$ are bounded in $L^2(0,T;X^\varepsilon)$. Up to extracting further subsequences, not relabeled, we can find $\mathbf{u}^\varepsilon = (u_i^\varepsilon,u_e^\varepsilon,s^\varepsilon):(0,T)\to\mathcal{V}_0^\varepsilon$ such that \eqref{eq:strongL2}, \eqref{eq:strongC0}, and \eqref{eq:weakL2} hold. Since $(\mathbf{U}^{\varepsilon,\tau_k}_n)_n$ is a solution of $(\mathcal{P}^{\varepsilon,\tau})$, integrating the Euler Scheme \eqref{P-eps-tau} in time yields
	\begin{equation}
		\label{eq:test}
		\begin{split}
			\int_0^T \mathbf{b}^\varepsilon &\left( \frac{d}{dt} \widetilde{\mathbf{U}}^{\tau_k,\varepsilon}(t), \widehat{\mathbf{U}}(t) \right) + \mathbf{a}^\varepsilon\left(\overline{\mathbf{U}}^{\varepsilon,\tau_k}(t), \widehat{\mathbf{U}}(t)\right)dt\\
			& +\int_0^T \mathcal{F}^\varepsilon\left(\overline{\mathbf{U}}^{\varepsilon,\tau_k}(t), \widehat{\mathbf{U}}(t)\right) -\mathbf{g}^\varepsilon\left(\underline{\mathbf{U}}^{\varepsilon,\tau_k}(t), \widehat{\mathbf{U}}(t)\right)dt=0,
		\end{split}
	\end{equation}
	for all $\widehat{\mathbf{U}} \in L^2(0,T;\mathbf{X}^\varepsilon)$. Since the piecewise constant interpolants of $V_n^{\varepsilon,\tau}:= U_{i,n}^{\varepsilon, \tau}-U_{e,n}^{\varepsilon, \tau}$ converge strongly in $L^\infty(0,T;X^\varepsilon)$, we can pass to the limit in \eqref{eq:test} and obtain
	\begin{equation*}
		\begin{split}
			\int_0^T \mathbf{b}^\varepsilon \left( \frac{d}{dt} \mathbf{u}^\varepsilon (t), \widehat{\mathbf{U}}(t) \right) & + \mathbf{a}^\varepsilon\left(\mathbf{u}^\varepsilon(t), \widehat{\mathbf{U}}(t)\right)dt\\
			& +\int_0^T \mathcal{F}^\varepsilon\left(\mathbf{u}^\varepsilon, \widehat{\mathbf{U}}(t)\right) -\mathbf{g}^\varepsilon\left(\mathbf{u}^\varepsilon(t), \widehat{\mathbf{U}}(t)\right)dt=0.
		\end{split}
	\end{equation*}
	Since $\widehat{\mathbf{U}}$ is arbitrary, we conclude that $\mathbf{u}^\varepsilon$ is the solution of \eqref{KNM-abstract}, which is unique by Theorem \ref{th:wp-KNM}. Finally, since the limit is unique and independent of the choice of subsequence, a standard compactness argument ensures that the entire sequence of interpolants $\{\widetilde{\mathbf{U}}^{\varepsilon, \tau}\}_{\tau > 0}$ converges to $\mathbf{u}^\varepsilon$ as $\tau \to 0$.

<!-- theorem-3: body_md -->
Let $\mathbf{u}^\varepsilon$ be the unique solution of the time-continuous problem \eqref{KNM-abstract} and let $\widetilde{\mathbf{U}}^{\varepsilon, \tau}$ be the piecewise linear interpolant of the solutions $\{\mathbf{U}^{\varepsilon,\tau}_n\}_{n=0}^N$ of the minimizing movement scheme \eqref{MM-eps}. 	\end{equation}

Then there exists a unique solution $\mathbf{u} \in C^0([0, T]; \mathbf{X})$ to \eqref{Bidomain-abstract}. Furthermore, the solution satisfies the following a priori estimate:

<!-- theorem-6: body_md -->
\label{th:ac04}
	For every sequence $(\varepsilon_j)$ of positive real numbers converging to 0, there exists a subsequence $(\varepsilon_{j_k})$ and two Carath\'eodory functions $g_{i,e}:\Omega \times \mathbb{R}^d \to \mathbb{R}$, satisfying
	\[

<!-- theorem-7: body_md -->
\label{theorem:gamma}
	Let the time step $\tau > 0$ and the time level $n \in \{1, \dots, N\}$ be fixed. Let $\varepsilon_j \to 0$ be a sequence such that $\mathcal H_{\varepsilon_j}$ $\Gamma$-converges to $\mathcal H$ as in Theorem \ref{th:ac04} and assume that the sequence of vectors $(\mathbf{U}_{n-1}^{\varepsilon_j,\tau}) \subset \mathbf{X}^{\varepsilon_j}$ converges to $\mathbf{U}_{n-1} \in L^2(\Omega)^3$ in the $s$-$s$-$w$ topology. If
	\begin{equation}
		\label{eq:ASn-1}
		\sup_{\varepsilon_j>0} \left(  \mathbf{a}^{\varepsilon_j}(\mathbf{U}_{n-1}^{\varepsilon_j,\tau}) + \phi^{\varepsilon_j}(\mathbf{U}_{n-1}^{\varepsilon_j,\tau}) \right) < \infty,
	\end{equation}
	then the following properties hold:
	\begin{enumerate}
		\item[(I)] Equi-coercivity and Compactness: Any sequence of discrete states $(\mathbf{U}^{\varepsilon_j}) \subset \mathbf{X}^{\varepsilon_j}$ with uniformly bounded energy $\sup_{\varepsilon_j > 0} \Psi_n^{\varepsilon_j, \tau}(\mathbf{U}^{\varepsilon_j}) < \infty$ is relatively compact with respect to the $s$-$s$-$w$ topology. Furthermore, any limit $\mathbf{U} = (U_i, U_e, S)$ belongs to $\mathbf{X}= H^1(\Omega) \times H_\circ^1(\Omega) \times L^2(\Omega)$. 
		\item[(II)] $\Gamma$-convergence: The sequence of functionals $\widetilde\Psi_n^{\varepsilon_j, \tau}$ $\Gamma$-converges to the functional $\Psi_n^\tau$ with respect to the $s$-$s$-$w$ topology. Specifically, the following holds:
		\begin{enumerate}
			\item liminf inequality: For every sequence $(\mathbf{U}^{\varepsilon_j})$ such that $\mathbf{U}^{\varepsilon_j} \xrightarrow{s-s-w} \mathbf{U}$,
			\begin{equation*}
				\liminf_{j \to +\infty} \widetilde\Psi_n^{\varepsilon_j, \tau}(\mathbf{U}^{\varepsilon_j}) \ge \Psi_n^\tau(\mathbf{U}).
			\end{equation*}
			\item limsup inequality: For every $\mathbf{U} \in \mathbf{X}$, there exists a recovery sequence $(\widetilde{\mathbf{U}}_n^{\varepsilon_j,\tau})$ converging $s$-$s$-$w$ to $\mathbf{U}$ such that
			\begin{equation*}
				\limsup_{j \to +\infty} \widetilde\Psi_n^{\varepsilon_j, \tau}(\widetilde{\mathbf{U}}_n^{\varepsilon_j,\tau}) \le \Psi_n^\tau(\mathbf{U}).
			\end{equation*}
		\end{enumerate}
	\end{enumerate}

<!-- proof-10: body_md -->
To prove (I), let $\mathbf{U}^\varepsilon=(U_i^\varepsilon,U_e^\varepsilon,S^\varepsilon)\in \mathbf{X}^\varepsilon$ be a sequence satisfying the uniform energy bound 
	\begin{align*}
		\sup_{\varepsilon>0} \Psi_n^{\varepsilon,\tau}(\mathbf{U}^\varepsilon) 
		&= \sup_{\varepsilon>0} \left(\frac{1}{2\tau} \mathbf{b}^\varepsilon(\mathbf{U}^\varepsilon - \mathbf{U}_{n-1}^{\varepsilon,\tau}) 
		+ \frac{1}{2} \mathbf{a}^\varepsilon(\mathbf{U}^\varepsilon)\right.\\
		&\qquad \left.+ \mathbf{\phi}^\varepsilon(\mathbf{U}^\varepsilon) 
		- \mathbf{g}^\varepsilon(\mathbf{U}_{n-1}^{\varepsilon,\tau}, \mathbf{U}^\varepsilon)\right) < \infty.
	\end{align*}	 
	Since the dissipation term $\mathbf{b}^\varepsilon$ is non-negative and the linear perturbation $\mathbf{g}^\varepsilon$ is sub-quadratic with respect to the energy, this hypothesis implies that the energy $E^\varepsilon(\mathbf{U}^\varepsilon)= \frac{1}{2} \mathbf{a}^\varepsilon(\mathbf{U}^\varepsilon) + \phi^\varepsilon(\mathbf{U}^\varepsilon)$ remains uniformly bounded independently of $\varepsilon$.  Proposition \ref{proposition:compactness} and Theorem \ref{th:ac04} thus ensure the existence of a subsequence $(\varepsilon_j)_{j\in \mathbb{N}} \to 0$ and a limit state $\mathbf{U} = (U_{i}, U_{e}, S) \in \mathbf{X}$ such that $\mathbf{U}^{\varepsilon_j} \xrightarrow{s-s-w} \mathbf{U}$ and $\mathcal H_{\varepsilon_j}$ $\Gamma$-converges to $\mathcal H$. For sake of notation, in the rest of the proof, we still denote this subsequence by $(\varepsilon)$ instead of $(\varepsilon_j)$.
	
	Regarding statement (II), we first establish the liminf inequality. Let $\mathbf{U}^\varepsilon \xrightarrow{s-s-w} \mathbf{U}$. We analyze the terms of $\Psi_n^{\varepsilon, \tau}$ as follows:
	\begin{itemize}
		\item \textit{Diffusive and Recovery parts:} By Theorem \ref{th:ac04}, and Lemma \ref{lemma:Gamma_norm},                                                      
		\begin{equation*}
			\liminf_{\varepsilon \to 0} \frac{1}{2} \mathbf{a}^\varepsilon(\mathbf{U}^\varepsilon) \ge \frac{1}{2} \mathbf{a}(\mathbf{U}).
		\end{equation*}
		\item \textit{Dissipative and Linear parts:} Since $\mathbf{U}^\varepsilon \xrightarrow{s-s-w} \mathbf{U}$ and $\mathbf{U}_{n-1}^{\varepsilon,\tau} \xrightarrow{s-s-w} \mathbf{U}_{n-1}$, Lemmas \ref{lemma:Gamma_norm} and \ref{lemma:gamma_g} yield
		\begin{align*}
			\liminf_{\varepsilon \to 0} \frac{1}{2\tau} \mathbf{b}^\varepsilon(\mathbf{U}^\varepsilon - \mathbf{U}_{n-1}^{\varepsilon,\tau}) &\ge \frac{1}{2\tau} \mathbf{b}(\mathbf{U} - \mathbf{U}_{n-1}), \\
			\lim_{\varepsilon \to 0} \mathbf{g}^\varepsilon(\mathbf{U}^\varepsilon, \mathbf{U}_{n-1}^{\varepsilon,\tau}) &= \mathbf{g}(\mathbf{U}, \mathbf{U}_{n-1}).
		\end{align*}
		\item \textit{Nonlinear potential:}                                                                                                                           
		By the strong $L^2$-convergence of $U_{i}^{\varepsilon}-U_{e}^{\varepsilon}$, the lower semicontinuity of $\varphi$ and the standard lower semicontinuity of integral functionals, we obtain: 
		\begin{equation*}
			\liminf_{\varepsilon \to 0} \phi^\varepsilon(\Pi_\varepsilon\mathbf{U}^\varepsilon) = \liminf_{\varepsilon \to 0} \int_\Omega \varphi(\Pi_\varepsilon(U_{i}^{\varepsilon}-U_{e}^{\varepsilon}))d x \ge \int_\Omega \varphi(U_{i} - U_{e}) \, dx.
		\end{equation*}
	\end{itemize}
	Summing these contributions gives the required liminf inequality. To complete the proof of the $\Gamma$-convergence (II), it remains to establish the limsup inequality. We invoke \cite[Proposition 6.5]{PSCF05} for the set 
	\[
	D:= \left\{ (u_i,u_e,s)\in H^1(\Omega)\times H^1_\circ(\Omega)\times L^2(\Omega) : u_i-u_e \in L^\infty(\Omega)\right\},
	\]
	which satisfies a suitable density condition thanks to \cite[Lemma 4.19]{PSCF05}.
	Given a limit state $\mathbf{U}=(U_i, U_e, S) \in D$, owing to Theorem \ref{th:ac04}, it is possible to construct a recovery sequence $\widetilde{\mathbf{U}}^{\varepsilon} = (\widetilde{U}_{i}^{\varepsilon}, \widetilde{U}_{e}^{\varepsilon}, \widetilde{S}^{\varepsilon}) \in \mathbf{X}^\varepsilon$ such that $\widetilde{\mathbf{U}}^{\varepsilon} \xrightarrow{s-s-w} \mathbf{U}$ as $\varepsilon \to 0$,
	\begin{equation}
		\label{eq:linfty}
		\lim_{\varepsilon \to 0}\mathcal{H}_\varepsilon(\widetilde{U}_{i}^{\varepsilon}, \widetilde{U}_{e}^{\varepsilon}) = \mathcal{H}(U_i,U_e),\quad \text{and}\quad \sup \| \Pi_\varepsilon (\widetilde{U}_{i}^{\varepsilon}- \widetilde{U}_{e}^{\varepsilon})\|_{L^{\infty}(\Omega)} <+\infty.
	\end{equation}
	Moreover, $\widetilde{S}^{\varepsilon}$ defined as the cell-average interpolation of the limit function $S$, satisfies $\Pi_\varepsilon \widetilde{S}^\varepsilon \to S$ strongly in $L^2(\Omega)$ and thus, by \eqref{eq:linfty} and Lemma \ref{lemma:Gamma_norm}
	\begin{equation*}
		\lim_{\varepsilon \to 0}  \mathbf{a}^\varepsilon(\widetilde{\mathbf{U}}^{\varepsilon}) = \lim_{\varepsilon \to 0} \mathcal{H}_\varepsilon(\widetilde{U}_{i}^{\varepsilon}, \widetilde{U}_{e}^{\varepsilon}) + \gamma|\widetilde{S}^\varepsilon|_\varepsilon^2= \mathcal{H}(U_{i}, U_{e}) + \gamma\|S\|_{L^2(\Omega)}^2=   \mathbf{a}(\mathbf{U}).
	\end{equation*}
	By \eqref{eq:linfty} and continuity of $\varphi$, the terms $v_\varepsilon:=\Pi_\varepsilon (\widetilde{U}_{i}^{\varepsilon}- \widetilde{U}_{e}^{\varepsilon})$ and $-\varphi(v_\varepsilon)$ are uniformly bounded in $L^\infty$ and arguing as in the liminf estimate we obtain
	                                                                                                                                                       
	\begin{equation*}
		\liminf_{\varepsilon \to 0} \int_\Omega -\varphi(v_\varepsilon(x))dx \geq \int_\Omega -\varphi(U_i - U_e) dx
	\end{equation*}
	and thus
	\[
	\limsup_{\varepsilon \to 0} \int_\Omega \varphi(v_\varepsilon(x))dx \leq \int_\Omega \varphi(U_i - U_e) dx.
	\]
	Finally, we consider the metric and linear terms. The state $\mathbf{U}_{n-1}^{\varepsilon,\tau}$ is fixed and converges to $\mathbf{U}_{n-1}$ in the $s$-$s$-$w$ topology. Since the recovery sequence $\widetilde{\mathbf{U}}^{\varepsilon}$ converges strongly in $L^2(\Omega)$ in all three components, the quadratic dissipation $\mathbf{b}^\varepsilon$ and the linear perturbation $\mathbf{g}^\varepsilon$ satisfy, by Lemmas \ref{lemma:Gamma_norm} and \ref{lemma:gamma_g}:
	\begin{align*}
		\lim_{\varepsilon \to 0} \frac{1}{2\tau} \mathbf{b}^\varepsilon(\widetilde{\mathbf{U}}^{\varepsilon} - \mathbf{U}_{n-1}^{\varepsilon,\tau}) &= \frac{1}{2\tau} \mathbf{b}(\mathbf{U} - \mathbf{U}_{n-1}), \\
		\lim_{\varepsilon \to 0} \mathbf{g}^\varepsilon(\widetilde{\mathbf{U}}^{\varepsilon}, \mathbf{U}_{n-1}^{\varepsilon,\tau}) &= \mathbf{g}(\mathbf{U}, \mathbf{U}_{n-1}).
	\end{align*}

Let $\mathbf{u}^{\varepsilon_j} = (u_i^{\varepsilon_j},u_e^{\varepsilon_j},s^{\varepsilon_j})$ be a sequence of  solutions of the KNM$-\varepsilon_j$ problem, such that $\mathcal H_{\varepsilon_j}$ $\Gamma$-converges to $\mathcal H$. Owing to estimates \eqref{apriori-estimate} and Corollary \ref{cor:minimizers}, we only have to show that $\Pi_{\varepsilon_j} v^{\varepsilon_j}(t)=\Pi_{\varepsilon_j}\left(u_i^{\varepsilon_j}(t) - u_e^{\varepsilon_j}(t)\right) \to v(t)$ and that $\Pi_{\varepsilon_j} s^{\varepsilon_j}(t) \to s(t)$, strongly in $L^2(\Omega)$. For any given time step size $\tau > 0$, we denote by $(V^{\varepsilon_j,\tau},S^{\varepsilon_j,\tau})$ and $(V^\tau,S^\tau)$ the time-dependent piecewise linear interpolants of the discrete solutions generated by the minimizing movements schemes \eqref{MM-eps} and \eqref{MM-eps-tt}, respectively. Dropping the variable $t$ for sake of notation, we have:
\begin{equation}
	\label{eq:error_decomp}
	\begin{aligned}
		& \|\Pi_{\varepsilon_j}(v^{\varepsilon_j},s^{\varepsilon_j}) - (v,s)\|_{L^2(\Omega)^2}\leq \, \| \Pi_{\varepsilon_j}(v^{\varepsilon_j},s^{\varepsilon_j}) - \Pi_{\varepsilon_j}(V^{\varepsilon_j,\tau},S^{\varepsilon_j,\tau})\|_{L^2(\Omega)^2} \\
		&\quad \quad + \|\Pi_{\varepsilon_j}(V^{\varepsilon_j,\tau},S^{\varepsilon_j,\tau}) - (V^{\tau},S^{\tau})\|_{L^2(\Omega)^2} 
		+ \|(V^{\tau},S^{\tau}) - (v,s)\|_{L^2(\Omega)^2}.
	\end{aligned}
\end{equation}
We begin by estimating the first and third terms on the right-hand side of \eqref{eq:error_decomp}. By hypothesis, the sequence of well-prepared microscopic initial data satisfies the uniform energy bound $\limsup_{j \to \infty} E^{\varepsilon_j}(\mathbf{u}_0^{\varepsilon_j}) = M < \infty$. Consequently, by \eqref{final-error-estimate}, we have
\begin{equation}
	\label{eq:c1tau}
	\begin{aligned}
		\| \Pi_{\varepsilon_j}(v^{\varepsilon_j},s^{\varepsilon_j}) - \Pi_{\varepsilon_j}(V^{\varepsilon_j,\tau},S^{\varepsilon_j,\tau})\|_{L^2(\Omega)^2}^2  &= | v^{\varepsilon_j} -V^{\varepsilon_j,\tau}|_{\varepsilon_j}^2 +| s^{\varepsilon_j}-S^{\varepsilon_j,\tau}|_{\varepsilon_j}^2, \\
		&\leq \max\left\{\frac{1}{C_m},1\right\}\mathbf{b}^{\varepsilon_j}(\mathbf{u}^{\varepsilon_j}-{\mathbf{U}}^{\varepsilon_j, \tau}),\\
		& \leq C_1^2 \tau,
	\end{aligned}
\end{equation}
where the positive constant $C_1$ depends exclusively on $M$, $C_m$, and $C$. Analogously, applying the corresponding continuous time-discretization error estimate \eqref{final-error-estimate-bidomain} we obtain
\begin{equation}
	\label{eq:c2tau}
	\|(V^{\tau},S^{\tau}) - (v,s)\|_{L^2(\Omega)^2}
	\leq C_2 \sqrt\tau.
\end{equation}
Next, we address the spatial limit $j \to \infty$ for the remaining central term of \eqref{eq:error_decomp} at the fixed time-step level $\tau > 0$. Let $n \in \{1, \dots, N\}$ denote the specific time interval containing the instant $t$. We carry out an inductive argument on the time step index $n$. For $n=0$, the strong convergence holds by the hypothesis on the initial data. Assume now that $\Pi_{\varepsilon_j} \mathbf{U}_{n-1}^{\varepsilon_j,\tau} \xrightarrow{s-s-w} \mathbf{U}_{n-1}^\tau$ as $j \to \infty$. Under this inductive step, Corollary 1 guarantees that the sequence of unique discrete minimizers $\mathbf{U}_n^{\varepsilon_j,\tau} = (U_{i,n}^{\varepsilon_j,\tau}, U_{e,n}^{\varepsilon_j,\tau}, S_n^{\varepsilon_j,\tau})$ satisfies
\begin{equation*}
	\Pi_{\varepsilon_j} \mathbf{U}_n^{\varepsilon_j,\tau} \xrightarrow{s-s-w} \mathbf{U}_n^\tau = (U_{i,n}^\tau, U_{e,n}^\tau, S_n^\tau) \quad \text{as } j \to \infty,
\end{equation*}
where $\mathbf{U}_n^\tau$ is the unique minimizer of the continuous functional $\Psi_n^\tau$. To upgrade the weak convergence $\Pi_{\varepsilon_j} S_n^{\varepsilon_j,\tau} \rightharpoonup S_n^\tau$ in $L^2(\Omega)$ to the strong topology, we exploit the convergence of $\mathbf{b}^{\varepsilon_j}(\mathbf{U}_n^{\varepsilon_j,\tau})$ guaranteed by Corollary 1 (iii), which implies that $\lim_{j \to \infty} \|\Pi_{\varepsilon_j} S^{\varepsilon_j,\tau}_n\|_{L^2(\Omega)} = \|S_n^\tau\|_{L^2(\Omega)}.$ 
Since the linear interpolant $\mathbf{U}^{\varepsilon_j, \tau}(t)$ is a convex combination of the discrete vectors $\mathbf{U}_{n-1}^{\varepsilon_j,\tau}$ and $\mathbf{U}_n^{\varepsilon_j,\tau}$ on each time subinterval, this strong convergence naturally extends pointwise in time to the instant $t$, yielding
\begin{equation}
	\label{eq:pescayogurt}
	\lim_{j \to \infty}\|\Pi_{\varepsilon_j}(V^{\varepsilon_j,\tau}(t),S^{\varepsilon_j,\tau}(t)) - (V^{\tau}(t),S^{\tau}(t))\|_{L^2(\Omega)^2} = 0.
\end{equation}
Passing to the limit superior as $j \to \infty$ on both sides of inequality \eqref{eq:error_decomp}, by \eqref{eq:c1tau}, \eqref{eq:c2tau}, and \eqref{eq:pescayogurt}, we obtain
\begin{equation*}
	\limsup_{j \to \infty}  \| \Pi_{\varepsilon_j} (v^{\varepsilon_j}(t),s^{\varepsilon_j}(t)) - (v(t),s(t))\|_{L^2(\Omega)^2}  \leq (C_1 + C_2) \sqrt{\tau}.
\end{equation*}
Taking the limit as the time step size $\tau \to 0^+$ concludes the proof.
