<!-- assumption-1: body_md -->
\label{ass:coeff}
The flux satisfies \(f\in C^1(\mathbb{R};\mathbb{R}^d)\). For every \(\ell\geq1\), the
coefficient
\[
b_\ell:[0,T]\times\mathbb{R}^d\times\mathbb{R}\longrightarrow\mathbb{R}
\]
is jointly Borel measurable. For every \(t\in[0,T]\), the map
\[
(x,\lambda)\longmapsto b_\ell(t,x,\lambda)
\]
is of class \(C^1\), and the maps
\[
(t,x,\lambda)\longmapsto
\partial_\lambda b_\ell(t,x,\lambda),
\qquad
(t,x,\lambda)\longmapsto
\nabla_x b_\ell(t,x,\lambda)
\]
are jointly Borel measurable. Moreover,
for every compact set $K\Subset\mathbb{R}^d$ and every $R>0$,
\begin{equation}\label{eq:noise-local-bound}
	\sup_{(t,x,\lambda)\in[0,T]\times K\times[-R,R]} \sum_{\ell\geq1}
		\left(
	\left\lvertb_\ell(t,x,\lambda)\right\rvert^2
	+
	\left\lvert\partial_\lambda b_\ell(t,x,\lambda)\right\rvert^2
	+
	\left\lvert\nabla_x b_\ell(t,x,\lambda)\right\rvert^2
	\right)
	<\infty.
\end{equation}

<!-- definition-1: body_md -->
\label{def:kinetic-measure}
A random nonnegative Radon measure $m$ on
$(0,T)\times\mathbb{R}^d\times\mathbb{R}$ is called a local kinetic measure up to $t=0$
if the following properties hold:
\begin{enumerate}[label=\textup{(\roman*)}]
	\item for every
	$\psi\in C_c\bigl((0,T)\times\mathbb{R}^d\times\mathbb{R}\bigr)$, the map
	$$
	\omega
	\longmapsto
	\int_{(0,T)\times\mathbb{R}^d\times\mathbb{R}}
	\psi(t,x,\lambda)\,dm(\omega;t,x,\lambda)
	$$
	is measurable;

	\item for every compact set $K\Subset\mathbb{R}^d$ and every $R>0$,
	\begin{equation}\label{eq:expected-measure-bound}
		\operatorname{\mathbb{E}}
		m\bigl(
		(0,T)\times K\times[-R,R]
		\bigr)
		<\infty;
	\end{equation}

	\item for every
	$\psi\in C_c(\mathbb{R}^d\times\mathbb{R})$, the process
	$$
	t\longmapsto
	\int_{(0,t]\times\mathbb{R}^d\times\mathbb{R}}
	\psi(x,\lambda)\,dm(s,x,\lambda)
	$$
	admits a predictable version.
\end{enumerate}
As part of the bounded-solution framework, we assume that
$\operatorname{supp} m\subset(0,T)\times\mathbb{R}^d\times[-M,M]$ almost surely.

\label{def:kinetic-solution}
A function $u:\Omega\times(0,T)\times\mathbb{R}^d\to\mathbb{R}$ is called a bounded
kinetic solution of \eqref{eq:spde} on $(0,T)\times\mathbb{R}^d$ if:

\begin{enumerate}[label=\textup{(\roman*)}]
	\item $u$ is $\mathcal P\otimes\mathcal{B}(\mathbb{R}^d)$-measurable;

	\item $\left\lvertu(\omega,t,x)\right\rvert\leq M$ 
	for almost every $(\omega,t,x)$;

	\item there exists a local kinetic measure $m$
	in the sense of Definition~\ref{def:kinetic-measure}
	such that, with
	$h(\omega,t,x,\lambda):=\1_{\{\lambda<u(\omega,t,x)\}}$,
	the identity
	\begin{equation}\label{eq:kinetic-weak}
	\begin{aligned}
		&\int_0^T
		\int_{\mathbb{R}^d}
		\int_{\mathbb{R}}
		h(\omega,t,x,\lambda)
		\bigl(
		\partial_t\varphi(t,x,\lambda)
		+
		a(\lambda)\cdot\nabla_x\varphi(t,x,\lambda)
		\bigr)
		\,d\lambda\,dx\,dt\\
		&\quad=
		-\sum_{\ell\geq1}
		\int_0^T
		\left(
		\int_{\mathbb{R}^d}
		b_\ell(t,x,u(\omega,t,x))
		\varphi(t,x,u(\omega,t,x))
		\,dx
		\right)
		dW_\ell(t)\\
		&\qquad
		-\frac12
		\int_0^T
		\int_{\mathbb{R}^d}
		G^2(t,x,u(\omega,t,x))
		\partial_\lambda\varphi(t,x,u(\omega,t,x))
		\,dx\,dt\\
		&\qquad
		+
		\int_{(0,T)\times\mathbb{R}^d\times\mathbb{R}}
		\partial_\lambda\varphi(t,x,\lambda)
		\,dm(\omega;t,x,\lambda)
	\end{aligned}
	\end{equation}	holds almost surely for every
	$\varphi
	\in
	C_c^1\bigl(
	(0,T)\times\mathbb{R}^d\times(-L,L)
	\bigr)$.
\end{enumerate}

<!-- theorem-1: body_md -->
\label{thm:main}
Let Assumption~\ref{ass:coeff} hold, and let $u$ be a bounded kinetic
solution in the sense of Definition~\ref{def:kinetic-solution}.  
Then there exists a unique
function
$$
u_0
\in
L^\infty\bigl(
\Omega\times\mathbb{R}^d,
\mathcal{F}_0\otimes\mathcal{B}(\mathbb{R}^d),
\mathbb{P}\otimes\mathcal L^d
\bigr),
\qquad
\left\lvertu_0\right\rvert\leq M,
$$
and a set
$$
\Omega_0\subset\Omega,
\qquad
\mathbb{P}(\Omega_0)=1,
$$
such that, for every $\omega\in\Omega_0$ and every compact set
$K\Subset\mathbb{R}^d$,
\begin{equation*}
	\operatorname*{ess\,lim}_{t\downarrow0}
	\int_K
	\left\lvertu(\omega,t,x)-u_0(\omega,x)\right\rvert
	\,dx=0.
\end{equation*}
Here $\mathcal{F}_0=\bigcap_{t>0}\mathcal{F}_t$ is the initial sigma-algebra of the
right-continuous filtration $(\mathcal{F}_t)_{t\geq0}${, and 
$\mathcal L^d$ represents the Lebesgue measure on $\mathbb{R}^d$.}

Consequently, for every compact $K\Subset\mathbb{R}^d$,
\begin{equation*}
	\operatorname*{ess\,lim}_{t\downarrow0}
	\operatorname{\mathbb{E}}\int_K
	\left\lvertu(t,x)-u_0(x)\right\rvert
	\,dx
	=
	0.
\end{equation*}
Thus $u(t)\to u_0$ strongly in $L^1(\Omega\times K)$, in the
essential-time sense.

\label{lem:time-integrated}
For every
\[
\phi\in C_c^1(\mathbb{R}^d\times(-L,L)),
\]
there exists a set
\[
\Omega_\phi\subset\Omega_m,
\qquad
\mathbb{P}(\Omega_\phi)=1,
\]
such that, for every $\omega\in\Omega_\phi$,
$\mathscr H_\phi(\cdot,\omega)$ has a c\`adl\`ag representative on
$(0,T)$. For every $0<s<t<T$, this representative satisfies
\begin{equation}\label{eq:time-integrated}
\begin{aligned}
	\mathscr H_\phi(t,\omega)-\mathscr H_\phi(s,\omega)
	&=
	\int_s^t
	\int_{\mathbb{R}^d}\int_{\mathbb{R}}
	h(\omega,r,x,\lambda)
	a(\lambda)\cdot\nabla_x\phi(x,\lambda)
	\,d\lambda\,dx\,dr
	\\
	&\quad+
	M_\phi(t,\omega)-M_\phi(s,\omega)
	\\
	&\quad+
	\frac12
	\int_s^t
	\int_{\mathbb{R}^d}
	G^2(r,x,u(\omega,r,x))
	\partial_\lambda\phi(x,u(\omega,r,x))
	\,dx\,dr
	\\
	&\quad-
	\int_{(s,t]\times\mathbb{R}^d\times\mathbb{R}}
	\partial_\lambda\phi(x,\lambda)
	\,dm(\omega;r,x,\lambda).
\end{aligned}
\end{equation}
Moreover, this representative has a finite right limit at $t=0$.
Consequently,
\[
\operatorname*{ess\,lim}_{t\downarrow0}
\mathscr H_\phi(t,\omega)
\]
exists and is finite.

<!-- proof-1: body_md -->
Fix $\phi\in C_c^1(\mathbb{R}^d\times(-L,L))$. Choose
$\Omega_\phi\subset\Omega_m$ of probability one such that the chosen
continuous version of $M_\phi$ is defined and
\eqref{eq:kinetic-weak} holds for
\[
\varphi(t,x,\lambda)=\theta(t)\phi(x,\lambda)
\]
simultaneously for $\theta$ in a fixed countable determining family in
$C_c^1((0,T))$. Since
\[
\int_0^T\theta(t)\,dM_\phi(t)
=
-\int_0^T\theta'(t)M_\phi(t)\,dt,
\]
and all the resulting terms depend continuously on $\theta$, the identity
extends to every $\theta\in C_c^1((0,T))$.

Fix $\omega\in\Omega_\phi$. Define
\[
\begin{aligned}
A_\phi(r,\omega)
&:=
\int_{\mathbb{R}^d}\int_{\mathbb{R}}
h(\omega,r,x,\lambda)
a(\lambda)\cdot\nabla_x\phi(x,\lambda)
\,d\lambda\,dx
\\
&\quad+
\frac12
\int_{\mathbb{R}^d}
G^2(r,x,u(\omega,r,x))
\partial_\lambda\phi(x,u(\omega,r,x))
\,dx
\end{aligned}
\]
and define the finite signed measure $\nu_\phi^\omega$ on $(0,T)$ by
\[
\nu_\phi^\omega(B)
:=
\int_{B\times\mathbb{R}^d\times\mathbb{R}}
\partial_\lambda\phi(x,\lambda)
\,dm(\omega;r,x,\lambda)
\]
for every Borel set $B\subset(0,T)$. Its cumulative function
\[
V_\phi(t,\omega)
:=
\nu_\phi^\omega((0,t])
=
\int_{(0,t]\times\mathbb{R}^d\times\mathbb{R}}
\partial_\lambda\phi(x,\lambda)
\,dm(\omega;r,x,\lambda)
\]
is c\`adl\`ag and has bounded variation.

The identity obtained above can be written in the sense of distributions
on $(0,T)$ as
\[
D_t\bigl(\mathscr H_\phi-M_\phi\bigr)
=
A_\phi(t,\omega)\,dt-\nu_\phi^\omega.
\]
Consequently, there exists a constant $c_\phi(\omega)$ such that
\[
\widetilde{\mathscr H}_\phi(t,\omega)
:=
c_\phi(\omega)
+
M_\phi(t,\omega)
+
\int_0^t A_\phi(r,\omega)\,dr
-
V_\phi(t,\omega)
\]
is a representative of $\mathscr H_\phi(\cdot,\omega)$. Since
$M_\phi$ is continuous, $A_\phi(\cdot,\omega)\in L^1(0,T)$, and
$V_\phi$ is c\`adl\`ag, this representative is c\`adl\`ag.

Moreover, for every $0<s<t<T$,
\[
V_\phi(t,\omega)-V_\phi(s,\omega)
=
\int_{(s,t]\times\mathbb{R}^d\times\mathbb{R}}
\partial_\lambda\phi(x,\lambda)
\,dm(\omega;r,x,\lambda).
\]
Therefore, the preceding representation gives exactly
\eqref{eq:time-integrated}. We henceforth denote this representative again
by $\mathscr H_\phi$.

Finally, since $\omega\in\Omega_m$,
\[
m\bigl(
\omega;
(0,T)\times\operatorname{supp}\phi
\bigr)
<\infty.
\]
It follows that
\[
V_\phi(t,\omega)\longrightarrow0
\qquad\text{as }t\downarrow0.
\]
Furthermore,
\[
M_\phi(t,\omega)\longrightarrow0
\]
by continuity at zero, while
\[
\int_0^t A_\phi(r,\omega)\,dr\longrightarrow0
\]
because $A_\phi(\cdot,\omega)\in L^1(0,T)$. Hence
\[
\lim_{t\downarrow0}
\mathscr H_\phi(t,\omega)
=
c_\phi(\omega),
\]
which proves the asserted finite right limit and therefore the essential
limit.

\label{lem:weak-trace}
Recall that $M$ is the uniform bound for $u$ and that $L>M+1$ is the fixed
kinetic cutoff chosen before Definition~\ref{def:kinetic-measure}.
There exists a set
$$
\Omega_{\mathrm w}\subset\Omega,
\qquad
\mathbb{P}(\Omega_{\mathrm w})=1,
$$
such that, for every $\omega\in\Omega_{\mathrm w}$, there is a unique
function
$$
h_0^\omega\in
L^\infty(\mathbb{R}^d\times\mathbb{R}),
\qquad
0\leq h_0^\omega\leq1,
$$
with
$$
h_0^\omega(x,\lambda)=1
\quad\text{for }\lambda\leq-L,
\qquad
h_0^\omega(x,\lambda)=0
\quad\text{for }\lambda\geq L,
$$
and such that, for every
$$
\varphi\in C_c^1(\mathbb{R}^d\times\mathbb{R}),
$$
\begin{equation}\label{eq:pathwise-weak-trace}

	\int_{\mathbb{R}^d}\int_{\mathbb{R}}
	\bigl(
	h(\omega,t,x,\lambda)-h_0^\omega(x,\lambda)
	\bigr)
	\varphi(x,\lambda)
	\,d\lambda\,dx
	=
	0.
\end{equation}
No measurability of the map
$\omega\mapsto h_0^\omega$ is asserted or used.

<!-- proof-2: body_md -->
Choose an exhaustion
$$
U_1\Subset U_2\Subset\cdots\Subset\mathbb{R}^d,
\qquad
\bigcup_{j\geq1}U_j=\mathbb{R}^d,
$$
and, for every $j$, choose a countable vector space
$$
\mathcal{D}_j
\subset
C_c^1(U_j\times(-L,L))
$$
over $\mathbb Q$, dense in
$
L^1(U_j\times(-L,L)).
$
With the events $\Omega_\varphi$ from
Lemma~\ref{lem:time-integrated}, define
\begin{equation}\label{eq:weak-trace-event}
	\Omega_{\mathrm w}
	:=\Omega_m
	\cap
	\bigcap_{\varphi\in\bigcup_{j\geq1}\mathcal{D}_j}
	\Omega_\varphi .
\end{equation}
The determining family is countable, so
$\mathbb{P}(\Omega_{\mathrm w})=1$.  

For every
$\omega\in\Omega_{\mathrm w}$, all scalar pairings
$$
\mathscr H_\varphi(t,\omega)
=
\int_{\mathbb{R}^d}\int_{\mathbb{R}}
h(\omega,t,x,\lambda)\varphi(x,\lambda)
\,d\lambda\,dx,
\qquad
\varphi\in\bigcup_{j\geq1}\mathcal{D}_j,
$$
have finite essential limits at $t=0$.  The time-integrated identity also
holds on this set for the same determining family; that is,
\eqref{eq:time-integrated} holds on $\Omega_{\mathrm w}$ 
{simultaneously for
every $\varphi\in\bigcup_{j\geq1}\mathcal{D}_j$.}

Fix $\omega\in\Omega_{\mathrm w}$ and $j\geq1$, and write
$Q_j:=U_j\times(-L,L)$. Choose a full-measure set
$E_\omega\subset(0,T)$ such that $h(\omega,t,\cdot,\cdot)$ is defined
for $t\in E_\omega$ and, for
every $\varphi\in\bigcup_{j\geq1}\mathcal{D}_j$,
$\mathscr H_\varphi(t,\omega)$ equals the c\`adl\`ag representative supplied by
Lemma~\ref{lem:time-integrated}.

Let $t_n\in E_\omega$ with $t_n\downarrow0$. Since
$0\leq h(\omega,t_n,x,\lambda)\leq1$,
weak-$*$ sequential compactness of the unit ball of $L^\infty(Q_j)$ gives a
subsequence, not relabeled, and a function
$h_{0,j}^\omega\in L^\infty(Q_j)$ such that
$$
h(\omega,t_n,\cdot,\cdot)
\overset{*}{\rightharpoonup}
h_{0,j}^\omega
\quad\text{in }L^\infty(Q_j).
$$
Moreover, $0\leq h_{0,j}^\omega\leq1$ almost everywhere on $Q_j$.
For every $\varphi\in\mathcal{D}_j$,
$$
\int_{Q_j}
h_{0,j}^\omega(x,\lambda)\varphi(x,\lambda)
\,d\lambda\,dx
=
\operatorname*{ess\,lim}_{t\downarrow0}
\mathscr H_\varphi(t,\omega).
$$
The right-hand side {exists by Lemma~\ref{lem:time-integrated} and it} 
is independent of both the sequence $(t_n)$ and the
extracted subsequence.  Hence, every weak-$*$ accumulation point has the same
pairings against $\mathcal{D}_j$.  Since $\mathcal{D}_j$ is dense in
$L^1(Q_j)$, the weak-$*$ accumulation point is unique.

This uniqueness implies the essential weak-$*$ convergence of the whole family
on $Q_j$.  Indeed, if the convergence failed, then, using the uniform
$L^\infty$-bound and the density of $\mathcal{D}_j$, one could find
$\varphi\in\mathcal{D}_j$, $\varepsilon>0$, and a sequence
$t_n\in E_\omega$, $t_n\downarrow0$, such that
$$
\left\lvert
\int_{Q_j}
\bigl(
h(\omega,t_n,x,\lambda)-h_{0,j}^\omega(x,\lambda)
\bigr)
\varphi(x,\lambda)
\,d\lambda\,dx
\right\rvert
\geq\varepsilon.
$$
Weak-$*$ compactness would then produce a different accumulation point, which
is impossible.

The uniform bound
$0\leq h(\omega,t,\cdot,\cdot),h_{0,j}^\omega\leq1$ and the
$L^1(Q_j)$-density of $\mathcal{D}_j$ extend the convergence from $\mathcal{D}_j$ to
every test function in $L^1(Q_j)$.

If $j<k$, then the restriction of $h_{0,k}^\omega$ to $Q_j$ and
$h_{0,j}^\omega$ have the same pairings against $\mathcal{D}_j$.  Hence
they agree almost everywhere on $Q_j$.  After choosing representatives with
this restriction property, define
$$
h_0^\omega(x,\lambda)
:=
h_{0,j}^\omega(x,\lambda)
\quad
\text{whenever }(x,\lambda)\in Q_j.
$$
The restriction identity makes this definition independent of $j$ outside
one null set.  It gives
$
h_0^\omega\in L^\infty(\mathbb{R}^d\times(-L,L))
$
with $0\leq h_0^\omega\leq1$.
Extend it by setting $h_0^\omega(x,\lambda)=1$ for $\lambda\leq-L$
and $h_0^\omega(x,\lambda)=0$ for $\lambda\geq L$.
Because $\left\lvertu\right\rvert\leq M<L$, the same identities hold for
$h(\omega,t,x,\lambda)$ outside $(-L,L)$.  The convergence already proved
on each $Q_j$, applied to the restriction of an arbitrary compactly
supported test function to $Q_j$, therefore proves
\eqref{eq:pathwise-weak-trace} on the full kinetic line.  If
$\widehat h_0^\omega$ also satisfies \eqref{eq:pathwise-weak-trace}, then
$h_0^\omega-\widehat h_0^\omega$ has zero pairing with every
$\varphi\in\mathcal{D}_j$.  Density gives equality almost everywhere on every
$Q_j$; testing below $-L$ and above $L$ gives the same constant values
there.  Hence the trace is unique on $\mathbb{R}^d\times\mathbb{R}$.

Fix
\[
\varphi
\in
C_c^1\bigl([0,T)\times\mathbb{R}^d\times(-L,L)\bigr)
\]
and define the associated scalar martingale by
\begin{equation}\label{eq:time-dependent-scalar-martingale}
M_\varphi(t,\omega)
:=
\sum_{\ell\geq1}
\int_0^t
\left(
\int_{\mathbb{R}^d}
b_\ell(r,x,u(\omega,r,x))
\varphi(r,x,u(\omega,r,x))
\,dx
\right)
dW_\ell(r).
\end{equation}
By \eqref{eq:noise-local-bound}, $M_\varphi$ admits a continuous
version on $[0,T]$ starting from zero. If
\[
\varphi(t,x,\lambda)=\phi(x,\lambda)
\]
is independent of time, then $M_\varphi=M_\phi$, where $M_\phi$ is
defined in \eqref{eq:scalar-martingale-pairing}.

Fix a deterministic sequence $\delta_n\downarrow0$ with
$0<\delta_n<T$. For every $n$, choose a nondecreasing function
\[

\qquad\text{for every }p\in P.

The proof of \cite[Lemma~4.2]{Panov:2005lr}, applied on a countable
relatively compact exhaustion of $Q$, gives the first assertion:
{there exists a set $\widetilde S_\gamma \subset \mathbb{R}$ of full 
measure and compatible local measures $\gamma_\zeta$ defined 
for every $\zeta \in \widetilde S_\gamma$.}

We verify the finite-mass addition.  Fix $E_n$ and a compact interval
$J_0\Subset\mathbb{R}$, and choose a compact interval $J_1$ such that
$J_0\Subset\operatorname{int}J_1$.  Since $E_n$ is open, there are functions
$\psi_k\in C_c(Q)$ such that $0\leq\psi_k\leq1$ and
$\psi_k\uparrow\1_{E_n}$.  For each $k$, the function
\begin{equation*}
	\zeta
	\longmapsto
	\1_{{\widetilde S_\gamma}}(\zeta)
	\int_Q\psi_k(q)\,d\gamma_\zeta(q)
\end{equation*} {is Lebesgue measurable: indeed, on $\widetilde S_\gamma$ 
it is the pointwise limit of continuous functions of $\zeta$ 
(the mollified integrals in the statement), and it trivially 
vanishes on the null set $\mathbb{R}\setminus \widetilde S_\gamma$.}  
Fatou's lemma and Tonelli's theorem give
\begin{align*}
	\int_{J_0\cap {\widetilde S_\gamma}}
	\int_Q\psi_k(q)\,d\gamma_\zeta(q)\,d\zeta
	&\leq
	\liminf_{j\to\infty}
	\int_{Q\times\mathbb{R}}
	\psi_k(q)
	\left(
	\int_{J_0}\theta_j(r-\zeta)\,d\zeta
	\right)
	\,d\gamma(q,r)
	\\
	&\leq
	\gamma(E_n\times J_1).
\end{align*}
For the last inequality, use $\operatorname{supp}\theta_j\subset(-1/j,1/j)$: for all
sufficiently large $j$, the inner integral is bounded by $1$ and vanishes
unless $r\in J_1$.  Letting $k\to\infty$ yields
\begin{equation*}
	\int_{J_0\cap {\widetilde S_\gamma}}
	\gamma_\zeta(E_n)\,d\zeta
	\leq
	\gamma(E_n\times J_1)
	<
	\infty.
\end{equation*}
This implies that
\[
\gamma_\zeta(E_n)<\infty
\]
for almost every
\(
\zeta\in J_0\cap\widetilde S_\gamma.
\)

Let \((J_m)_{m\geq1}\) be a compact exhaustion of \(\mathbb{R}\), and, for
\(n,m\geq1\), define
\[
N_{n,m}
:=
\left\{
\zeta\in J_m\cap\widetilde S_\gamma:
\gamma_\zeta(E_n)=\infty
\right\}.

\[
\mathcal L^1(N_{n,m})=0
\qquad

\widetilde S_\gamma

Then

\(\zeta\in\widetilde S_\gamma\setminus N_{n,m}\), it follows that
\[

\]
Thus \(\gamma_\zeta(E_n)<\infty\) for every \(n\geq1\) and every
\(\zeta\in S_\gamma\). This construction uses only the measurability of
the scalar evaluations
\(\zeta\mapsto\gamma_\zeta(E_n)\), and does not require measurability of
the measure-valued map \(\zeta\mapsto\gamma_\zeta\).

	\int_0^T
	\int_{J_\alpha}
	\int_J
	\left\lvertF_{\ell,\phi}(t,\sigma,\zeta)\right\rvert^2
	\,d\zeta\,d\sigma\,dt.
	\label{eq:stochastic-fubini-BDG}
\end{align}

The process in \eqref{eq:scalar-stochastic-fubini} is a jointly measurable
family of real-valued martingales, with one continuous path for almost every
fixed $(\sigma,\zeta)$.  To integrate these sections in the parameter
variables, we also regard the same family
$(F_{\ell,\phi})_{\ell\geq1}$ as a single
$L^2(J_\alpha\times J)$-valued integrand.  Accordingly,
\eqref{eq:sliced-integrand-bound} defines the
$L^2(J_\alpha\times J)$-valued It\^o integral
\begin{equation}\label{eq:sliced-L2-martingale}
\mathcal Z_{\phi,J_\alpha,J}(t)
:= \sum_{\ell\geq1}
\int_0^t
F_{\ell,\phi}(r,\cdot)
\,dW_\ell(r)
\end{equation}
as a continuous $L^2(J_\alpha\times J)$-valued martingale.  For a countable
dense set of
$
g\in L^2(J_\alpha\times J),
$
stochastic Fubini gives, at every rational time,
\begin{align}
	\int_{J_\alpha}
	\int_J
	g(\sigma,\zeta)
	z_{\phi,J_\alpha,J}(t,\sigma,\zeta)
	\,d\zeta\,d\sigma
	&=
	\sum_{\ell\geq1}
	\int_0^t
	\left(
	\int_{J_\alpha}
	\int_J
	g(\sigma,\zeta)
	F_{\ell,\phi}(r,\sigma,\zeta)
	\,d\zeta\,d\sigma
	\right)
	dW_\ell(r)
	\notag\\
	&=
	\left\langle
	\mathcal Z_{\phi,J_\alpha,J}(t),g
	\right\rangle_{L^2(J_\alpha\times J)}.
	\label{eq:stochastic-fubini-identification}
\end{align}
The parameterized construction makes
$z_{\phi,J_\alpha,J}$ jointly measurable in
$(\omega',t,\sigma,\zeta)$.  By
\eqref{eq:stochastic-fubini-BDG}, outside one $\mathbb{P}$-null set,
$$
\int_{J_\alpha}\int_J
\sup_{0\leq t\leq T}
\left\lvertz_{\phi,J_\alpha,J}(t,\sigma,\zeta)\right\rvert^2
\,d\zeta\,d\sigma
<\infty.
$$
Together with the scalar path continuity in
\eqref{eq:scalar-stochastic-fubini}, dominated convergence shows that
$t\mapsto z_{\phi,J_\alpha,J}(t,\cdot,\cdot)$ belongs to
$C([0,T];L^2(J_\alpha\times J))$.  Consequently,
\eqref{eq:stochastic-fubini-identification}, first at rational times, and
density in the dual variable identify
$z_{\phi,J_\alpha,J}(t,\cdot,\cdot)$ with the continuous martingale
$\mathcal Z_{\phi,J_\alpha,J}(t)$ from
\eqref{eq:sliced-L2-martingale}.  Continuity extends the identity to every
$t\in[0,T]$, so, outside one $\mathbb{P}$-null set,
\begin{equation*}
	z_{\phi,J_\alpha,J}(t,\cdot,\cdot)
	=
	\mathcal Z_{\phi,J_\alpha,J}(t)
	\quad\text{in }L^2(J_\alpha\times J)
\end{equation*}
for every $t\in[0,T]$.
Tonelli applied to \eqref{eq:stochastic-fubini-BDG} gives, for
$(\mathbb{P}\otimes\mathcal L^{r_\alpha})$-almost every $(\omega',\sigma)$,
\begin{equation}\label{eq:fixed-parameter-stochastic-fubini-bound}

	\,d\zeta
	<\infty,
	\qquad

	\in C([0,T];L^2(J)).
\end{equation}

$\omega$.  Rename $\widetilde u_0$ as $u_0$ and replace it by

T_{-M,M}(u_0).
$

<!-- corollary-2: body_md -->
\label{cor:strong-trace-expectation}
For every compact set $K\Subset\mathbb{R}^d$,
$$
\operatorname*{ess\,lim}_{t\downarrow0}
\operatorname{\mathbb{E}}\int_K
\left\lvertu(\omega,t,x)-u_0(\omega,x)\right\rvert
\,dx=0.
$$
Equivalently,
$u(t)\to u_0$
strongly in $L^1(\Omega\times K)$, in the essential-time sense.
<!-- end corollary-2 -->
<!-- proof-17: body_md -->
For a compact set $K\Subset\mathbb{R}^d$, define
$F_K(t,\omega):=\int_K\left\lvertu(\omega,t,x)-u_0(\omega,x)\right\rvert\,dx$.
The joint measurability of $u$ and $u_0$ implies that
$(t,\omega)\mapsto F_K(t,\omega)$ is measurable. Moreover,
$0\leq F_K(t,\omega)\leq2M\left\lvertK\right\rvert$.

For $\delta>0$, set
$$
G_{K,\delta}(\omega)
:=
\operatorname*{ess\,sup}_{0<t<\delta}
F_K(t,\omega).
$$
The essential supremum is measurable because its superlevel sets can be
expressed in terms of the Lebesgue measure of time-superlevel sets.  Indeed,
for every $a\in\mathbb{R}$,
$$
\{G_{K,\delta}>a\}
=
\left\{
\omega:
\int_0^\delta
\1_{\{F_K(t,\omega)>a\}}
\,dt
>0
\right\}.
$$
The indicator in the last integral is jointly measurable in $(t,\omega)$
because $F_K$ is jointly measurable.  Tonelli's theorem therefore makes the
integral a measurable function of $\omega$, so the displayed set belongs to
$\mathcal{F}$.  Hence all superlevel sets of $G_{K,\delta}$ are measurable, and
$G_{K,\delta}$ is a random variable.  This permits the expectation and the
dominated-convergence argument below.
Since
$u_0(\omega,\cdot)=u_0^\omega$ in $L^1_{\mathrm{loc}}(\mathbb{R}^d)$
for almost every $\omega$, the pathwise trace
\eqref{eq:pathwise-final-strong-trace} gives
$G_{K,\delta}(\omega)\to0$ as $\delta\downarrow0$
for almost every $\omega$.  Also,
$0\leq G_{K,\delta}\leq2M\left\lvertK\right\rvert$. Dominated convergence therefore
yields $\operatorname{\mathbb{E}} G_{K,\delta}\to0$.

For almost every $t\in(0,\delta)$,
$F_K(t,\omega)\leq G_{K,\delta}(\omega)$
for almost every $\omega$.  Consequently,
$$
\operatorname*{ess\,sup}_{0<t<\delta}
\operatorname{\mathbb{E}} F_K(t)
\leq
\operatorname{\mathbb{E}} G_{K,\delta}.
$$
Letting $\delta\downarrow0$, we obtain
$$
\operatorname*{ess\,lim}_{t\downarrow0}
\operatorname{\mathbb{E}} F_K(t)
=
0,
$$
which is the asserted strong trace in expectation.

Lemma~\ref{lem:final-measurability} gives an
$\mathcal{F}_0\otimes\mathcal{B}(\mathbb{R}^d)$-measurable representative $u_0$ and a
full-probability set $\Omega_0\subset\Omega_*$ on which
$u_0(\omega,\cdot)=u_0^\omega$ in $L^1_{\mathrm{loc}}(\mathbb{R}^d)$.  Equation
\eqref{eq:pathwise-final-strong-trace} gives the pathwise strong trace for
every $\omega\in\Omega_0$, and
Corollary~\ref{cor:strong-trace-expectation} gives convergence in expectation.

It remains to prove uniqueness. Let $v_0$ be another
$\mathcal{F}_0\otimes\mathcal{B}(\mathbb{R}^d)$-measurable function satisfying the strong-trace
conclusion of Corollary~\ref{cor:strong-trace-expectation}, with $v_0$ in
place of $u_0$. For every compact $K\Subset\mathbb{R}^d$ and almost every $t>0$,
\begin{align*}
	\operatorname{\mathbb{E}}\int_K
	\left\lvertu_0(\omega,x)-v_0(\omega,x)\right\rvert
	\,dx
	&\leq
	\operatorname{\mathbb{E}}\int_K
	\left\lvertu(\omega,t,x)-u_0(\omega,x)\right\rvert
	\,dx
	\\
	&\quad+
	\operatorname{\mathbb{E}}\int_K
	\left\lvertu(\omega,t,x)-v_0(\omega,x)\right\rvert
	\,dx.
\end{align*}
Taking the essential limit as $t\downarrow0$ gives
$\operatorname{\mathbb{E}}\int_K\left\lvertu_0-v_0\right\rvert\,dx=0$. Hence $u_0=v_0$ almost everywhere on
$\Omega\times K$. An exhaustion of $\mathbb{R}^d$ proves
uniqueness on $\Omega\times\mathbb{R}^d$.
