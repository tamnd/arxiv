<!-- definition-1: body_md -->
\label{def:smp}
The vector $Y$ satisfies $\operatorname{SMP}(\kappa)$ if, for every $p\ge1$, every finite
$T\subset\mathbb R^d$ with $|T|\ge e^p$, and every $a>0$, the separation
condition
\begin{equation}\label{eq:separation}
 \lVert \langle t-u,Y\rangle\rVert_p\ge a
 \qquad\text{for all distinct }t,u\in T
\end{equation}
implies $W_Y(T)\ge\kappa a$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Under \eqref{eq:assumptions}, the vector $X=X_E$ satisfies
$\operatorname{SMP}(\kappa)$ for a universal $\kappa>0$. The constant is independent
of $d$, $E$, $\lambda$, and all the functions $\phi_i$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:comparison}
There is a universal $M\ge1$ and a parameter $\tau>0$ such that the vector
$V$ with independent coordinates and marginal densities
\begin{equation}\label{eq:product}
 f_{i,\tau}(v)=c_{i,\tau}
              \exp\{-\lambda|v|-\tau\phi_i(|v|)\},
 \qquad v\in\mathbb R,
\end{equation}
satisfies
\begin{equation}\label{eq:comparison}
 X\preceq_{\mathrm{cx}} 2MV,
 \qquad
 \mathbb E q(V)\le2\mathbb E q(X)
 \quad\text{for every seminorm }q\text{ on }\mathbb R^d.
\end{equation}
The parameter $\tau$ depends on the distribution of $X$, but neither
$\tau$ nor $V$ depends on $T$, $p$, or $q$.
<!-- end theorem-2 -->
<!-- lemma-1: body_md -->
\label{lem:transfer}
Let $X$ and $V$ be integrable symmetric vectors, and suppose $V$ satisfies
$\operatorname{SMP}(\kappa_0)$. Assume that, for constants $A,B>0$,
\begin{align}
 \lVert \langle h,X\rangle\rVert_p&\le A\lVert \langle h,V\rangle\rVert_p
 &&(h\in\mathbb R^d,\ p\ge1),\label{eq:transfermoment}\\
 \mathbb E q(V)&\le B\mathbb E q(X)
 &&(q\text{ a seminorm}).\label{eq:transfernorm}
\end{align}
Then $X$ satisfies $\operatorname{SMP}(\kappa_0/(AB))$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Suppose $T,p,a$ satisfy Definition~\ref{def:smp} for $X$.
By \eqref{eq:transfermoment}, distinct points of $T$ are separated for
$V$ at level $a/A$. Thus
\[
 W_V(T)\ge\frac{\kappa_0a}{A}.
\]
To use \eqref{eq:transfernorm}, introduce the range seminorm
\begin{equation}\label{eq:qT}
 q_T(x)=\max_{t,u\in T}\langle t-u,x\rangle
       =\max_{t\in T}\langle t,x\rangle-\min_{u\in T}\langle u,x\rangle.
\end{equation}
The set $T-T$ is symmetric and contains zero, so $q_T$ is nonnegative,
subadditive, and absolutely homogeneous. It is therefore a seminorm,
even when $T-T$ does not span the entire space.

For any integrable symmetric $Y$,
\begin{equation}\label{eq:widthseminorm}
 \mathbb E q_T(Y)
 =\mathbb E\max_{t\in T}\langle t,Y\rangle
  +\mathbb E\max_{u\in T}\langle u,-Y\rangle
 =2W_Y(T).
\end{equation}
Using \eqref{eq:transfernorm} with $q_T$ yields
$W_V(T)\le B W_X(T)$, and the result follows.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:sign}
If $0\le a_i\le b_i$ for $i=1,\ldots,d$, then, for every convex
$f:\mathbb R^d\to\mathbb R$,
\begin{equation}\label{eq:sign}
 \mathbb E_\varepsilon f(\varepsilon\odot a)\le\mathbb E_\varepsilon f(\varepsilon\odot b).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Define the sign average
\[
 \bar f(x)=2^{-d}\sum_{\varepsilon\in\{-1,1\}^d}f(\varepsilon\odot x).
\]
It is convex and unconditional. With all other coordinates fixed,
it is an even convex function of the remaining coordinate. An even
convex function is nondecreasing on $[0,\infty)$: for $0\le a\le b$
with $b>0$, express $a$ as a convex combination of $b$ and $-b$.
Applying this argument one coordinate at a time proves
$\bar f(a)\le\bar f(b)$.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:one}
For $b>0$ and $\Delta\ge0$,
\begin{equation}\label{eq:stochsandwich}
 Y_b\preceq_{\mathrm{st}} Y_{b+\Delta}\preceq_{\mathrm{st}} Y_b+\Delta.
\end{equation}
Consequently, the common-quantile coupling satisfies
\begin{equation}\label{eq:quantile}
 0\le Y_{b+\Delta}-Y_b\le\Delta\qquad\text{almost surely}.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Write $W_-=W_{J\setminus\{i\}}$.

\par\smallskip\noindent\textbf{First inequality: a larger budget increases the energy.}\ 
On the common support $0<y<b$,
\begin{equation}\label{eq:likelihoodbudget}
 \frac{g_{b+\Delta}(y)}{g_b(y)}
 =\frac{W_J(b)}{W_J(b+\Delta)}
  \frac{W_-(b+\Delta-y)}{W_-(b-y)}.
\end{equation}
For a concave function $\ell$, the increment
$\ell(z+\Delta)-\ell(z)$ is nonincreasing in $z$.
Take $\ell=\log W_-$ and $z=b-y$. The last ratio in
\eqref{eq:likelihoodbudget} is therefore nondecreasing in $y$.
Moreover, the extra support of $g_{b+\Delta}$ lies to the right of $b$.
The difference of the two densities can thus change sign only from
negative to positive. Since both have integral one, their distribution
functions are ordered, proving $Y_b\preceq_{\mathrm{st}} Y_{b+\Delta}$.
When $J=\{i\}$, the same argument works with $W_-=1$.

\par\smallskip\noindent\textbf{Second inequality: the energy cannot increase by more than the extra budget.}\ 
For $0\le y\le b$, substitute $u=v+\Delta$ in the upper-tail integral:
\begin{align*}
 \mathbb P(Y_{b+\Delta}>y+\Delta)
 &=\frac{\int_y^b w_i(v+\Delta)W_-(b-v)\,dv}{W_J(b+\Delta)}\\
 &\le\frac{\int_y^b w_i(v)W_-(b-v)\,dv}{W_J(b)}
 =\mathbb P(Y_b>y).
\end{align*}
Here $w_i$ is nonincreasing and $W_J$ is nondecreasing.
Outside the displayed range, the desired tail inequality is immediate
from the supports. This proves $Y_{b+\Delta}\preceq_{\mathrm{st}} Y_b+\Delta$.

\par\smallskip\noindent\textbf{Putting the two comparisons in one coupling.}\ 
If $F_b^{-1}$ denotes the generalized inverse distribution function, then
\eqref{eq:stochsandwich} gives
\[
 F_b^{-1}(u)\le F_{b+\Delta}^{-1}(u)
            \le F_b^{-1}(u)+\Delta,\qquad 0<u<1.
\]
Apply all three functions to the same uniform random variable.
<!-- end proof-3 -->
<!-- proposition-1: body_md -->
\label{prop:budget}
For $0\le s\le t$, there is a coupling of
$U_s\sim\mathcal L(|X_s|)$ and $U_t\sim\mathcal L(|X_t|)$ such that
\begin{equation}\label{eq:budget}
 U_s\le U_t\quad\text{coordinatewise},\qquad
 0\le\Phi(U_t)-\Phi(U_s)\le t-s.
\end{equation}
The construction is measurable in $(s,t)$ and may be chosen to be the
identity when $s=t$.
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
Sample coordinates in the order $1,\ldots,d$, using the same independent
uniform variables for the two vectors. Initially the remaining budgets
are $s$ and $t$. Suppose at some stage they are $b$ and $b+\Delta$.
Conditioning on the previously chosen coordinates leaves exactly the
positive-orthant law for the remaining coordinate set and the remaining
budget, because both the weight and the constraint are additive.

Use Lemma~\ref{lem:one} to select the next two energies. Their difference
$\delta$ satisfies $0\le\delta\le\Delta$. After those energies are
subtracted, the remaining budgets are still ordered, and their difference
is $\Delta-\delta$. Induction proves coordinatewise order of every energy,
and hence of every magnitude.

Writing $\Delta_k$ for the difference of remaining budgets before
coordinate $k$, and $\delta_k$ for the energy difference selected at that
coordinate, we have
\[
 \Delta_{k+1}=\Delta_k-\delta_k,\qquad
 0\le\delta_k\le\Delta_k,\qquad \Delta_1=t-s.
\]
Therefore
\[
 \Phi(U_t)-\Phi(U_s)=\sum_{k=1}^d\delta_k
                   =t-s-\Delta_{d+1}\le t-s.
\]
For $s=0$, take $U_s=0$.

The conditional distribution functions in \eqref{eq:conditional} are
measurable in their budget and coordinate arguments. Their generalized
inverses are measurable as well: the event that an inverse is at most
$y$ is expressed through the corresponding distribution function at
$y$. Thus the finite recursion is measurable in the budgets and the
uniform variables. Equal budgets use the same quantiles and give the
identity coupling.

<!-- lemma-4: body_md -->
\label{lem:slack}
Let $W=W_{\{1,\ldots,d\}}$. Then a version of the conditional law is
\begin{equation}\label{eq:disintegration}
 \mathcal L(V_\tau\mid S_\tau=s)=\mu_s,
 \qquad
 f_\tau(s)=\frac{W(s)e^{-\tau s}}{L(\tau)},\quad s>0,
\end{equation}
where $L(\tau)=\int_0^\infty W(u)e^{-\tau u}\,du$.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
In the joint density of $(V_\tau,H_\tau)$, make the substitution
$s=\Phi(v)+h$. The Jacobian in the slack variable is one. The result is
\begin{align*}
 f_{V_\tau,S_\tau}(v,s)
 &=\frac{\tau}{\mathcal Z_\tau}
   e^{-\lambda\sum_i|v_i|-\tau\Phi(v)}
   e^{-\tau(s-\Phi(v))}\mathbf 1_{\{\Phi(v)\le s\}}\\
 &=\frac{\tau}{\mathcal Z_\tau}
   e^{-\tau s}e^{-\lambda\sum_i|v_i|}
   \mathbf 1_{\{\Phi(v)\le s\}}.
\end{align*}
For fixed $s$, the dependence on $v$ is exactly the unnormalized density
of $\mu_s$. Integrating in $v$ gives $Z_s=2^dW(s)$ and hence a density
of $S_\tau$ proportional to $W(s)e^{-\tau s}$.
Normalization gives \eqref{eq:disintegration}.

<!-- lemma-5: body_md -->
\label{lem:rare}
For every $M\ge1$,
\begin{equation}\label{eq:rare}
 \mathbb P(S_{M\tau_0}>E)\le\eta_M,
 \qquad \eta_M=2(2/3)^{M-1}.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Write $f_0=f_{\tau_0}$ and
$\overline F_0(s)=\mathbb P(S_{\tau_0}>s)$.
We first show that a fixed amount of probability lies a definite distance
below the median, where the distance is measured in units of $1/\tau_0$.

\par\smallskip\noindent\textbf{A bound on the hazard.}\ 
Because $W$ is nondecreasing, for $u\ge s>0$,
\[
 f_0(u)\ge f_0(s)e^{-\tau_0(u-s)}.
\]
Integrating gives
\[
 \overline F_0(s)\ge\frac{f_0(s)}{\tau_0}.
\]
The hazard, namely the density divided by the remaining upper-tail
probability, therefore satisfies
\begin{equation}\label{eq:hazard}
 \frac{f_0(s)}{\overline F_0(s)}\le\tau_0.
\end{equation}

\par\smallskip\noindent\textbf{A quarter of the mass is separated from the median.}\ 
Let $h=\log(3/2)/\tau_0$.
The inequalities $S_{\tau_0}\ge H_{\tau_0}$ and
$\mathbb P(S_{\tau_0}>E)=1/2$ imply $\tau_0E\ge\log2$, so $h<E$.
Since the derivative of $-\log\overline F_0$ is the hazard, integrating
\eqref{eq:hazard} from $E-h$ to $E$ yields
\[
 \overline F_0(E-h)
 \le e^{\tau_0h}\overline F_0(E)
 =\frac34.
\]
Consequently,
\begin{equation}\label{eq:quarter}
 \mathbb P(S_{\tau_0}\le E-h)\ge\frac14.
\end{equation}

\par\smallskip\noindent\textbf{Exponential tilting separates the two regions.}\ 
Put $a=(M-1)\tau_0$. By \eqref{eq:disintegration}, the law at
$M\tau_0$ is the exponential tilt of the law at $\tau_0$:
\begin{align*}
 \mathbb P(S_{M\tau_0}>E)
 &=\frac{\mathbb E[e^{-aS_{\tau_0}}\mathbf 1_{\{S_{\tau_0}>E\}}]}
         {\mathbb E e^{-aS_{\tau_0}}}\\
 &\le\frac{\frac12 e^{-aE}}
              {\frac14 e^{-a(E-h)}}
 =2e^{-ah}
 =2(2/3)^{M-1}.
\end{align*}
The numerator uses the median identity, and the denominator uses
\eqref{eq:quarter}.

<!-- proposition-3: body_md -->
\label{prop:residual}
For every seminorm $q$ on $\mathbb R^d$,
\begin{equation}\label{eq:residual}
 \mathbb E q(D)\le\theta_M\mathbb E q(V),
 \qquad
 \theta_M=C_B\sqrt2\,(2/3)^{(M-1)/2}.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-9: body_md -->
Let $B=\{S>E\}$. Conditional on all magnitudes, the slack, and any
randomness used in the correction, write
\[
 V=\varepsilon\odot v,\qquad Q=\varepsilon\odot u,\qquad
 D=\varepsilon\odot(v-u),
\]
where $0\le u\le v$ coordinatewise. The signs are still independent and
symmetric. Moreover, $v-u=0$ off $B$. Thus Lemma~\ref{lem:sign} gives,
conditionally,
\[
 \mathbb E_\varepsilon q\bigl(\varepsilon\odot(v-u)\bigr)
 \le\mathbf 1_B\,\mathbb E_\varepsilon q(\varepsilon\odot v).
\]
After averaging the remaining variables, this becomes
\begin{equation}\label{eq:signresidual}
 \mathbb E q(D)\le\mathbb E[\mathbf 1_Bq(V)].
\end{equation}
Now use Cauchy--Schwarz, the log-concavity of $V$, and
Lemma~\ref{lem:rare}:
\begin{align*}
 \mathbb E q(D)
 &\le\sqrt{\mathbb P(B)}\,\bigl(\mathbb E q(V)^2\bigr)^{1/2}\\
 &\le C_B\sqrt{\eta_M}\,\mathbb E q(V)
 =C_B\sqrt2\,(2/3)^{(M-1)/2}\mathbb E q(V).
\end{align*}

<!-- remark-1: body_md -->
The pointwise implication $|r_i|\le|v_i|\Rightarrow q(r)\le q(v)$ is
false for an arbitrary seminorm. For example, take
$q(x)=|x_1-x_2|$, $v=(1,1)$, and $r=(1,0)$. Then $q(v)=0$ but
$q(r)=1$. The valid statement used above is the contraction after
averaging independent signs. Also, no independence between $B$ and $V$
is assumed in \eqref{eq:signresidual} or in Cauchy--Schwarz.

<!-- lemma-6: body_md -->
\label{lem:reverse}
For every $M\ge1$,
\begin{equation}\label{eq:reverse}
 X\preceq_{\mathrm{cx}}2V_{\tau_0}\preceq_{\mathrm{cx}}2MV_{M\tau_0}.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-10: body_md -->
\par\smallskip\noindent\textbf{The median mixture dominates half of the desired convex expectation.}\ 
Let $f:\mathbb R^d\to\mathbb R$ be convex, and let $\bar f$ be its sign average.
All vectors in the assertion are unconditional, so their expectations
are unchanged by replacing $f$ with $\bar f$. This function is convex,
unconditional, and minimized at zero.

When $s\ge E$, \eqref{eq:budgetcx} gives
$\mathbb E\bar f(X_s)\ge\mathbb E\bar f(X)$. For $s<E$, the lower bound
$\bar f(0)$ is sufficient. Disintegration and the median identity therefore
yield
\begin{equation}\label{eq:half}
 \mathbb E\bar f(V_{\tau_0})
 \ge\frac12\mathbb E\bar f(X)+\frac12\bar f(0).
\end{equation}
Convexity gives
$\bar f(x)\le\tfrac12\bar f(2x)+\tfrac12\bar f(0)$.
Combining this with \eqref{eq:half}, we obtain
\[
 \mathbb E\bar f(2V_{\tau_0})\ge\mathbb E\bar f(X),
\]
which proves $X\preceq_{\mathrm{cx}}2V_{\tau_0}$.

\par\smallskip\noindent\textbf{A universal dilation compensates for increasing the parameter.}\ 
Put $\tau=M\tau_0$ and consider one positive coordinate. The density of
$M|V_{\tau,i}|$, divided by the density of $|V_{\tau_0,i}|$, is a positive
constant times
\begin{equation}\label{eq:scaledratio}
 \exp\left\{\lambda(1-1/M)x
       +\tau_0\phi_i(x)-M\tau_0\phi_i(x/M)\right\},
 \qquad x>0.
\end{equation}
Its logarithm is locally absolutely continuous. Its almost-everywhere
derivative equals
\[
 \lambda(1-1/M)
   +\tau_0\bigl(\phi_i'(x)-\phi_i'(x/M)\bigr)\ge0,
\]
because a convex function has a nondecreasing derivative.
The ratio is therefore nondecreasing. The one-crossing argument used in
Lemma~\ref{lem:one} implies
\begin{equation}\label{eq:coordinatecooling}
 |V_{\tau_0,i}|\preceq_{\mathrm{st}} M|V_{\tau,i}|.
\end{equation}
Use independent common-quantile couplings in the different coordinates;
their product preserves the independent-coordinate marginal laws.
After adding common independent signs, Lemma~\ref{lem:sign} gives
$V_{\tau_0}\preceq_{\mathrm{cx}} MV_\tau$. Dilation by two completes
\eqref{eq:reverse}.
