<!-- theorem-1: body_md -->
\label{THM:ASGWP} 
Let $\alpha>\frac 14$ and  $0<\varepsilon \ll 1$. Then, the BBM equation \eqref{BBM} is almost surely globally well posed in $H^{\alpha-\frac 12-\varepsilon}(\mathbb{T})$, with respect to random initial data of the form 
\begin{align}
u_{0}^{\omega}=\sum_{n\in \mathbb{Z}}\frac{g_{n}(\omega)}{\langle n \rangle^{\alpha}}e^{inx}. \label{globaldata}
\end{align}
 More precisely, for almost every $\omega \in \Omega$, there exists a unique solution $u$ of \eqref{BBM} in \begin{align*}
u_{0}+C(\mathbb{R}; H^{\min(3\alpha-\frac 12 ,2\alpha)-\varepsilon}(\mathbb{T}))\subset C(\mathbb{R}; H^{\alpha-\frac 12-\varepsilon}(\mathbb{T}))
\end{align*}
with initial condition $u_{0}^{\omega}$ of the form \eqref{globaldata} and which satisfies 
\begin{align}
\| u(t) \|_{H^{\alpha-\frac 12-\varepsilon}} \leq C(\omega) (1+|t|)^{a} \label{polygrowth}
\end{align}
for some $a=a(\alpha)>0$ and $C(\omega)>0$.
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
\rm \label{RMK:BBM}
Using the translation \eqref{translate}, Theorem~\ref{THM:ASGWP} implies almost sure global well-posedness for \eqref{BBM0} with initial data of the form $u_{0}^{\omega}-1$. Then, we obtain almost sure global well-posedness for \eqref{BBM0} with initial data exactly of the form $u_0^{\omega}$ as a consequence of the Cameron-Martin Theorem; see \cite{OQ} for similar details in the context of the cubic nonlinear Schr\"{o}dinger equation.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
\rm \label{remark: gaussianity}
 Theorem~\ref{THM:ASGWP} also extends to random initial data \eqref{initialdata} where the random variables $\{g_{n}\}_{n\in \mathbb{Z}}$ are not necessarily Gaussian. For precise assumptions, see those in \cite[Appendix A]{FBBM}. 
In fact, as the random object $\mathcal{N}(u_0)$ is no longer time dependent, we only need second-moment estimates on $\|\mathcal{N}(\mathbf{P}_{>M}u_0)\|_{H^{\sigma}}$ which is technically simpler than those in \cite[Appendix A]{FBBM}.

<!-- lemma-3: body_md -->
\label{Lemma:wienerchaos}  
For any $2\leq p<\infty$ and $a\in \ell^2(\mathbb{Z})$ and $b\in \ell^2(\mathbb{Z}^2)$, it holds that 
\begin{align}
 \bigg\| \sum_{n\in \mathbb{Z}}a(n)g_{n}(\omega)  \bigg\|_{L^{p}(\Omega)} &\lesssim p^{\frac{1}{2}}\|a(n)\|_{\ell^{2}_{n}}. \label{wienerchaos} \\
\bigg\|  \sum_{n_1,n_2} b(n_1,n_2) g_{n_1}g_{n_2}  \bigg\|_{L^{p}(\Omega)} &\lesssim p  \bigg\|  \sum_{n_1,n_2} b(n_1,n_2) g_{n_1}g_{n_2}  \bigg\|_{L^{2}(\Omega)}
\end{align}
where the family $\{g_n\}_{n\in \mathbb{Z}}$ are as in \eqref{initialdata}.

<!-- proposition-1: body_md -->
\label{prop:stochobjects} 
Let $\frac 14 < \alpha \leq \frac 12$,
\begin{align*}
s_1 <\alpha-\tfrac 12 \quad \text{and} \quad s_2<2\alpha.
\end{align*}
 Let $\{ \rho_k \}_{k\in \mathbb{N}}$ be a family of mollifiers on $\mathbb{T}$ and let $u_{0,k} =u_0^{\omega}\ast \rho_k$, where $u_0^{\omega}$ is as in \eqref{initialdata}. 
Then, 
\begin{align*}
(u_{0,k}, \mathcal{N}(u_{0,k})) \longrightarrow (u_0, \mathcal{N}(u_0)),
\end{align*}
as $k \to \infty$ in $L^{q}(\Omega;  W^{s_1,\infty}(\mathbb{T})\times W^{s_2,\infty}(\mathbb{T}))$ for any $q\geq 1$ and almost surely in $ W^{s_1,\infty}(\mathbb{T})\times W^{s_2,\infty}(\mathbb{T})$. Moreover, the limit $(u_0,\mathcal{N}(u_0))$ is independent of the choice of mollification kernel $\rho$, including the regularisation by the (non-smooth) Dirichlet projection $\mathbf{P}_{\leq N}$.
Furthermore, there exist $C,C'>0$ such that 
\begin{align}
\begin{split}
\mathbb{P} \bigg( \|u_0\|_{H^{\alpha-\frac 12-}}+ \sup_{M\in 2^{\mathbb{N}}} &\big[ M^{\alpha-\frac 12-s_1-} \| \mathbf{P}_{> M}u_0\|_{W^{s_1,\infty}(\mathbb{T})}  \\
&+  M^{2\alpha-s_2-} \|\mathcal{N}( \mathbf{P}_{> M}u_0)\|_{H^{s_2}(\mathbb{T})} \big]  >L \bigg) \leq C'e^{-CL}
\end{split}
\label{probM}
\end{align}
 for any $L >0$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
The entire statement regarding convergence of mollified sequences and independence of the limits is proved in \cite[Proposition 2.6]{FBBM}. The only new part is the estimate on the event in \eqref{probM}. The statements for $\mathbf{P}_{> M}u_0$ follow from Sobolev embedding and the estimate 
\begin{align*}
\sum_{|n|>M} \frac{\langle n \rangle^{2s_1}}{\langle n \rangle^{2\alpha}} \lesssim M^{2s_1-2\alpha+1-}.
\end{align*}
 For $\mathcal{N}(\mathbf{P}_{> M}u_0)$, we provide details only for the second moment, as higher moments, and thence, the exponential decay follow from Lemma~\ref{Lemma:wienerchaos}. We have 
\begin{align*}
 \mathbb{E} \big[ \| \mathcal{N}(\mathbf{P}_{> M}u_0)\|_{H^{s_2}}^{2}    \big] &= \sum_{n\neq 0} \langle n \rangle^{2s_2} |\varphi(n)|^2 \mathbb{E} \bigg[ \bigg| \sum_{ \substack{n=n_1+n_2 \\ |n_1|, |n_2|> M}} \frac{g_{n_1}g_{n_2}}{\langle n_1 \rangle^{\alpha} \langle n_2 \rangle^{\alpha}} \bigg|^{2} \bigg] \\
 & \sim \sum_{n\neq 0} \langle n \rangle^{2s_2-2}\sum_{ \substack{n=n_1+n_2 \\ |n_1|, |n_2|> M}}  \frac{1}{\langle n_1 \rangle^{2\alpha} \langle n_2 \rangle^{2\alpha}},
\end{align*}
where in evaluating the expectation, we crucially used that fact that $n_1+n_2 \neq 0$ since  $\varphi(0)=0$. We consider two sub-cases in the above summation (assuming by symmetry that $|n_1|\geq |n_2|$). If $|n_1|\sim |n_2|$, then since $\alpha>\frac 14$, we get 
\begin{align*}
\sum_{n\neq 0} \langle n \rangle^{2s_2-2}\sum_{ \substack{n=n_1+n_2 \\ |n_1|\sim |n_2|> M}}  \frac{1}{\langle n_1 \rangle^{2\alpha} \langle n_2 \rangle^{2\alpha}} \lesssim \sum_{n} \frac{\langle n \rangle^{2s_2-2}}{ \max(|n|, M)^{4\alpha-1}}  \lesssim M^{2s_2-4\alpha}.
\end{align*}
On the the other hand, if $|n_1|\gg |n_2|$, then we have 
\begin{align*}
\sum_{n\neq 0} \langle n \rangle^{2s_2-2}\sum_{ \substack{n=n_1+n_2 \\ |n_1|\gg |n_2|> M}}  \frac{1}{\langle n_1 \rangle^{2\alpha} \langle n_2 \rangle^{2\alpha}} &\lesssim \sum_{|n|\gtrsim M} \langle n \rangle^{2s_2-2-2\alpha} \sum_{|n_2|\ll |n|} \frac{1}{\langle n_2 \rangle^{2\alpha}}  \\ 
&\lesssim  \sum_{|n|\gtrsim M} \langle n \rangle^{2s_2-1-4\alpha} \lesssim M^{2s_2 -4\alpha}.
\end{align*}
Then, in order to obtain the supremum over $M$, we use that for $M\in 2^{\mathbb{N}}$, it is enough to convert the supremum to a summation at the cost of a logarithm which explains the worse rates in \eqref{probM}.
This completes the proof.

<!-- proposition-2: body_md -->
\label{prop:detHIGH} 
Fix $T>0$, $0<\sigma<1$, $L\geq 1$, $a_1>0$ and $0<a_3<a_2$. Then, there exists $M_0=M_0(L,T,a_1,a_2,a_3)\geq 1$ such that for every $M\geq M_0$ the following holds:
For every pair of forcings $(z_1,z_2)$ satisfying 
 \begin{align}
\|z_{1}\|_{W^{-s_1,\frac{1}{s_1}}( \mathbb{T})} \leq L M^{-a_1}    \quad \text{and} \quad \|z_{2}\|_{H^{\sigma}( \mathbb{T})}\leq LM^{-a_2}, \label{Kcond}
\end{align} 
where $s_1:= \min(1-\sigma,\sigma)$, 
 there exists a unique solution $v\in C([0,T]; H^{\sigma}(\mathbb{T}))$ to the initial value problem:
 \begin{align}
\begin{cases}
\partial_{t}v =  -\tfrac{1}{2}\varphi(\partial_x) (  v^2 +2z_1 v  )  -\tfrac{1}{2}z_2 , \\
v\vert_{t = 0} =0,
\end{cases} 
\label{BBMz1z2}
\end{align}
which satisfies 
\begin{align}
\| v\|_{C([0,T];H^{\sigma})} \leq M^{-a_3}.
\end{align}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
We apply the contraction mapping argument to the integral formulation:
\begin{align*}
v(t) =\Gamma_{t}[v]:= -\frac 12 \int_{0}^{t}   \varphi(\partial_x) (  v^2 +2z_1 v  ) dt' + tz_2
\end{align*}
in the small ball 
\begin{align*}
B_{R} : = \{  v\in C([0,T];H^{\sigma}(\mathbb{T})) \, : \, \|v\|_{C_{T}H^{\sigma}} \leq R\}
\end{align*}
where $R=M^{-a_3}$.
The main term to consider is the linear in $v$ term $\varphi(\partial_x)[ z_1 v]$. When $0<\sigma\leq \frac 12$,  Lemma~\ref{LEM:negprod} implies
\begin{align}

<!-- lemma-6: body_md -->
\label{LEM:comm}
 Let $0\leq s\leq 1$ and $w\in H^{s}(\mathbb{T})$. Then, we have
 \begin{align*}
\bigg \vert\int_{\mathbb{T}} (\partial_{x}Iw)[I(w^{2})-(Iw)^{2}]\,dx \bigg\vert\lesssim N^{-\frac{3}{2}}\|Iw\|_{H^{1}(\mathbb{T})}^{3}.

<!-- proof-6: body_md -->
We recall the setup from the proof of \cite[Lemma 4.3]{FBBM}.
By Plancherel, we have 
\begin{multline*}
 \int_{\mathbb{T}} (\partial_{x}Iw)[I(w^{2})-(Iw)^{2}]\,dx  =\sum_{n_{1}+n_{2}+n_{3}=0}in_{3}m(n_{3})(m(n_{1}+n_{2})-m(n_{1})m(n_{2})) \prod_{j=1}^{3} \widehat w(n_{j}).
\end{multline*}
We symmetrise the right-hand side to obtain $$ \sum_{n_{1}+n_{2}+n_{3}=0} M(n_{1},n_{2},n_{3}) \widehat w (n_{1})\widehat w (n_{2})\widehat w(n_{3}),$$ where $M$ is the symmetric multiplier 
\begin{align*}
 M(n_{1},n_{2},n_{3})=\frac{i}{3} [ & n_{1}m(n_{1})(m(n_{2}+n_{3})-m(n_{2})m(n_{3})) \\ &+n_{2}m(n_{2})(m(n_{1}+n_{3})-m(n_{1})m(n_{3}))\\ &+n_{3}m(n_{3})(m(n_{1}+n_{2})-m(n_{1})m(n_{2}))].
\end{align*} 
By symmetry, we assume $|n_{3}|\leq |n_{2}|\leq |n_{1}|$. Furthermore, we assume $|n_{1}|>N$, since otherwise $m(n_{j})=1$ for all $j=1,2,3$, which implies $M(n_{1},n_{2},n_{3})=0$ on $n_{1}+n_{2}+n_{3}=0$. In addition, we also assume $|n_{2}|\gtrsim N$ since if $|n_{2}|\ll N$ we obtain a contradiction to the conditions $n_{1}+n_{2}+n_{3}=0$, $|n_{1}|>N$ and $|n_{3}|\leq |n_{2}|$. For shorthand, let us define 
\begin{multline*} 
\Lambda_{N}:=\{ (n_{1},n_{2},n_{3})\in \mathbb{Z}^{3}  :  n_{1}+n_{2}+n_{3}=0,\, |n_{3}|\leq |n_{2}|\leq |n_{1}|,\, |n_{1}|>N, \, |n_{2}|\gtrsim N \,\}.
\end{multline*}
Using the condition $n_{1}+n_{2}+n_{3}=0$, we see that
$$ M(n_{1},n_{2},n_{3})=\frac{i}{3}\left[n_{1}m^{2}(n_{1})+n_{2}m^{2}(n_{2})+n_{3}m^{2}(n_{3}) \right],$$ and hence on $\Lambda_{N}$, we have
\begin{equation}
|M(n_{1},n_{2},n_{3})|\lesssim |n_{3}|m^{2}(n_{3}). \label{Mbd}
\end{equation} 
Setting $ y(n)=\langle n \rangle m(n)|\widehat w(n)|$ and using \eqref{Mbd}, we have thus reduced to showing 
\begin{equation} \label{v2com1}
\sum_{\Lambda_{N}}\frac{ m(n_3) }{\langle n_{1} \rangle\langle n_{2} \rangle m(n_{1}) m(n_{2})} \prod_{j=1}^{3} y(n_j) \lesssim N^{-\frac{3}{2}}\|y(n)\|_{\ell_{n}^{2}}^{3}.
\end{equation} 
At this point, we diverge from the analysis in \cite[Lemma 4.3]{FBBM}.
As $|n_1|\geq |n_2| \gtrsim N$, we have 
\begin{align*}
\text{LHS} \eqref{v2com1} \lesssim N^{-2(1-s)} \sum_{\Lambda_{N}}  \frac{ |m(n_3)|}{\langle n_1 \rangle^{s}\langle n_2 \rangle^{s}} \prod_{j=1}^{3} y(n_j).
\end{align*}
We split the above summation into two cases. First, if $|n_3|<N$, then, by Cauchy-Schwarz, this contribution is bounded by
\begin{align*}
N^{-2(1-s)} \sum_{ |n_3|<N} y(n_3)  \sum_{|n_2|>N} \frac{y(n_2)y(-n_2-n_3)}{\langle n_2 \rangle^{2s}} \lesssim N^{-2} \bigg(  \sum_{ |n_3|<N} y(n_3) \bigg) \|y\|_{\ell^2}^{2} \lesssim N^{-\frac 32} \|y\|_{\ell^2}^{3}.
\end{align*}
Next, if $|n_3|>N$, this contribution is bounded by 
\begin{align*}
N^{-1+s} \sum_{\substack{|n_3|>N\\ \Lambda_{N}}} \frac{1}{ \langle n_1 \rangle^{s}\langle n_2 \rangle^{s} \langle n_3 \rangle^{1-s}} \prod_{j=1}^{3} y(n_j) \lesssim N^{-1+s} \sum_{|n_3|>N} \frac{y(n_3)}{\langle n_3 \rangle^{1+s}} \| y\|_{\ell^2}^{2} \lesssim N^{-\frac 32} \|y\|_{\ell^2}^{3},
\end{align*}
which completes the proof.

<!-- lemma-7: body_md -->
\label{LEM:comm2} 
Let $0\leq \gamma <\frac 12$ and $\gamma\leq s \leq 1-\gamma$. Then, 
\begin{equation}
\| I( fg)\|_{L^2} \lesssim   \big( \|If \|_{L^2} + N^{-\frac 12+\gamma} \|f\|_{H^{-\gamma}}\big) \|I g\|_{H^{1}}  , \label{vzcom}
\end{equation}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
By Plancherel and Cauchy-Schwarz, we have 
\begin{align*}
\| I( fg)\|_{L^2} &= \bigg\| \sum_{k}  \frac{m(n)}{m(k)\langle k \rangle} \widehat f(n-k)  \langle k \rangle\widehat{ Ig}(k) \bigg\|_{\ell^2_n} \\
& \lesssim  \bigg( \sum_{n,k}  \frac{m(n)^2}{m(k)^2\langle k \rangle^2}| \widehat f(n-k)|^2 \bigg)^{\frac 12} \|Ig\|_{H^1} = :M_{N}(f)^{\frac 12} \|Ig\|_{H^1}.
\end{align*}
It remains to show that, under the conditions on $(\gamma,s)$ in the statement, that
\begin{align}
M_{N}(f) \lesssim \|I f\|_{L^2}^2 + (N^{-\frac 12+\gamma} \|f\|_{H^{-\gamma}})^2. 
\label{MNest}
\end{align}
We consider a number of cases.

Suppose first that $|k|\ll |n|$. In this case, $|n|\sim |n-k|$ and so $m(n)^{2} \lesssim m(n-k)^2$.
If additionally $|k|\lesssim N$, then we have 
\begin{align*}
\sum_{|k|\lesssim N } \frac{1}{m(k)^{2} \langle k \rangle^2} \sum_{n} | \widehat{ I f}(n-k)|^{2} \lesssim  \|If\|_{L^2}^2,
\end{align*}
uniformly in $N$. If instead $|k|\gg N$, then we have 
\begin{align*}
\sum_{ N\ll |k| \ll |n|}  \frac{m(n)^2}{m(k)^2\langle k \rangle^2}| \widehat f(n-k)|^2 & \lesssim \sum_{ N\ll |k| \ll |n|} \frac{ \langle n-k \rangle^{2\gamma}}{\langle n \rangle^{2-2s} \langle k \rangle^{2s}}| \langle n-k \rangle^{-\gamma}\widehat f(n-k)|^2  \\
& \lesssim \sum_{|k|\gg N} \frac{1}{ \langle k \rangle^{2-2\gamma}} \| \langle n \rangle^{-\gamma} \widehat f (n)  \|_{\ell^2_n} \lesssim N^{-1+2\gamma} \|f\|_{H^{-\gamma}},
\end{align*}
where in the second inequality we used that $\gamma \leq 1-s$ and in the last inequality, we used $\gamma<\frac 12$.

Suppose now that $|k|\sim |n|$. Then, $ (\frac{m(n)}{m(k)\langle k \rangle})^2  \lesssim \langle k \rangle^{-2}$ which is $\ell^1$-summable. If $|n|\lesssim N$, then we use $|k-n|\lesssim N$ so $m(n-k)\sim 1$ which can be added to $f$ for free. Otherwise, $|n| \gg N$ and we bound instead by 
\begin{align*}
\sum_{|k|\gg N} \frac{1}{ \langle k \rangle^{2-2\gamma}} \sum_{n} | \langle n-k \rangle^{-\gamma}\widehat f(n-k)|^2 \lesssim N^{-1+2\gamma} \|f\|_{H^{-\gamma}}
\end{align*}
 for $\gamma<\frac 12$.
 
 Lastly, we suppose that $|k|\gg |n|$ and so $|k|\sim |n-k|$. If $|n|\lesssim N$, then we have the multiplier 
 \begin{align*}
\frac{\langle n-k \rangle^{2\gamma}}{ m(k)^{2} \langle k \rangle^{2}} \lesssim N^{-2(1-s)} \frac{\langle k \rangle^{2\gamma}}{\langle k \rangle^{2s}} \lesssim N^{-2(1-s)} \frac{1}{\langle k \rangle^{2s-2\gamma}} \lesssim N^{-2(1-s)} \frac{1}{\langle n \rangle^{2s-2\gamma}} 
\end{align*}
where we used $s\geq \gamma$. Then, we sum $|\langle n-k \rangle^{-\gamma}\widehat f(n-k)|^2$ over $k$ and are left with 
\begin{align*}
\sum_{|n|\lesssim N} N^{-2(1-s)} \frac{1}{\langle n \rangle^{2s-2\gamma}} \lesssim N^{-1+2\gamma}.
\end{align*}
Lastly, if $|n|\gg N$, then 
\begin{align*}
\frac{ m(n)^2 \langle n-k \rangle^{2\gamma} }{ m(k)^{2} \langle k \rangle^{2}} \lesssim \frac{1}{\langle n \rangle^{2-2s} \langle k \rangle^{2s-2\gamma} } \lesssim \frac{1}{\langle n \rangle^{2-2\gamma} } 
\end{align*}
and summing over $|n|\gg N$ provides a factor $N^{-1+2\gamma}$. 
This completes the proof of \eqref{MNest}.

Consider first \eqref{Cond1}. We have
\begin{align*}
N^{1-s}M^{s+\frac 12-\alpha+} \ll N^{\frac 52-s}M^{-\frac 12-\alpha+s+}  \quad \Longleftarrow \quad M\ll N^{\frac 32+}.
\end{align*}
Then, under $M\ll N^{\frac 32+}$, we find 
\begin{align*}
N^{\frac 32-\alpha}  \ll N^{\frac 52-s}M^{-\frac 12-\alpha+s+}
\end{align*}
since $s>0$ and $\alpha\leq \frac 12$. Therefore, \eqref{Cond1} is satisfied provided that $M\ll N^{\frac 32+}$. We move onto \eqref{Cond2}. As $\sigma<2\alpha$, we may choose $M$ large enough so that $M^{2\alpha-\sigma-} \gg LT$.
Moreover, if we take $M=N^{k}$ for some $1<k<\frac 32$ to be determined, then 
\begin{align}
N^{1-s}M^{-\frac 12-\alpha+s} = N^{1-s-k(\frac 12 +\alpha-s)} \ll L^{-1}T^{-1} \quad \Longleftarrow \quad k>\frac{1-s}{\frac 12+\alpha-s}. \label{Cond3}
\end{align}
For the condition on $k$ in \eqref{Cond3} to be consistent with $k <\frac 32$ we need
\begin{align*}
\frac{1-s}{\frac 12+\alpha-s}< \frac 32 \quad \Longleftarrow \quad s<3\alpha -\frac 12.
\end{align*}
This condition on $s$ is consistent with $\frac 12-\alpha <s<\alpha+\frac 12$ provided that $\tfrac 12 -\alpha <3\alpha -\tfrac 12$
which forces $\alpha>\frac 14$.


 Now given any $\varepsilon>0$ sufficiently small, we may fix $\frac 14 <\alpha \leq \frac 12$, $L=\varepsilon^{-1}$, $M=N^{\max(\frac{1-s}{\frac 12+\alpha-s},1)+}$ and  choose $N=N(\varepsilon,T) \gg 1$ so that \eqref{Cond3} holds. 
 Note that $N\sim (1+\varepsilon T)^{\theta_1}$ for some $\theta_1=\theta_1(\alpha,s,\sigma)>0$, namely, it depends polynomially on $T$. Then, by \eqref{smallset}, $\mathbb{P}( \Omega^{c}_{\varepsilon^{-1},T,\alpha,\varepsilon}) <C' e^{-C\varepsilon^{-1}} <\varepsilon$, we have the bound
 \begin{align*}
\sup_{t\in [0,T]} E(Iw_{M})(t) \leq C(\varepsilon)\langle T \rangle^{\theta_2}
\end{align*}
for some $\theta_2 >0$.
By \eqref{hstoIh1} and the definition of the modified energy $E(Iw_M)$ we obtain \eqref{bdvk}. 
Setting $\widetilde{\Omega}_{T,\varepsilon} =\Omega_{\varepsilon^{-1},T,\alpha,\varepsilon}$ completes the proof of Proposition~\ref{prop:bdk}.

We point out that the event $\widetilde{\Omega}_{T,\varepsilon}$ controls the forcings $\mathbf{P}_{>M}u_0$ and $v_{M}$ on all of $[0,T]$ and already has small complemental probability. As we then obtain a uniform bound on the growth of $w_{M}$ on $[0,T]$, we conclude that $T_{\max}^{\omega}$ from Lemma~\ref{LEM:wM} satisfies $T_{\max}^{\omega}\geq T$ for each fixed $\omega\in \widetilde{\Omega}_{T,\varepsilon}$. 
To establish the almost sure existence statement is a standard argument and one can see for instance \cite[Section 4.3]{FBBM} for details.
