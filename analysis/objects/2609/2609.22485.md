<!-- theorem-1: body_md -->
\label{thm:padic}
Let $n,d\geq1$, and let $V\subset\mathbb P^{n-1}(\mathbb Q_p)$ be nonempty.
Let $E\subset\mathbb Q_p^n$. Suppose that for every $\omega\in V$ there is a
polynomial map
\begin{equation*}
 \gamma_\omega:\mathbb Q_p\longrightarrow\mathbb Q_p^n,\qquad
 \gamma_\omega(t)\mathrel{:=} \sum_{j=0}^{d-1}a_{\omega,j}t^j+v_\omega t^d,
\end{equation*}
where $a_{\omega,j}\in\mathbb Q_p^n$, $v_\omega\neq0$, and
$[v_\omega]=\omega$, such that
\begin{equation*}
 \gamma_\omega(\mathbb Z_p)\subset E.
\end{equation*}
Then
\begin{equation*}
 \dim_{\mathrm H} E\geq\dim_{\mathrm H} V+1.
\end{equation*}
In particular, if $V=\mathbb P^{n-1}(\mathbb Q_p)$, then $\dim_{\mathrm H} E=n$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:polynomial-projections}
For the family \eqref{eq:projection-family}, let
$\mathcal E\subset(\mathbb Q_p^n)^{d+1}$ be nonempty and analytic, and set
$S\mathrel{:=} \dim_{\mathrm H} P_0(\mathcal E)$. Then
\begin{equation}\label{eq:projection-ae}
 \dim_{\mathrm H} P_r(\mathcal E)\geq S
 \quad\text{for Haar-almost every }r\in\mathbb Q_p.
\end{equation}
For every $0\leq\sigma<S$,
\begin{equation}\label{eq:projection-exceptional}
 \dim_{\mathrm H}\{r\in\mathbb Q_p:\dim_{\mathrm H} P_r(\mathcal E)\leq\sigma\}
 \leq 1-\frac{S-\sigma}{dn}.
\end{equation}

<!-- corollary-1: body_md -->
\label{cor:maximal-projection-dimension}
Under the hypotheses of Theorem~\ref{thm:polynomial-projections}, put
\begin{equation*}
 S_*\mathrel{:=} \sup_{s\in\mathbb Q_p}\dim_{\mathrm H} P_s(\mathcal E).
\end{equation*}
Then
\begin{equation}\label{eq:projection-maximal-ae}
 \dim_{\mathrm H} P_r(\mathcal E)=S_*
 \quad\text{for Haar-almost every }r\in\mathbb Q_p.
\end{equation}
For every $0\leq\sigma<S_*$,
\begin{equation}\label{eq:projection-maximal-exceptional}
 \dim_{\mathrm H}\{r\in\mathbb Q_p:\dim_{\mathrm H} P_r(\mathcal E)\leq\sigma\}
 \leq 1-\frac{S_*-\sigma}{dn}.
\end{equation}
In particular, a lower bound $\dim_{\mathrm H} P_{r_0}(\mathcal E)\geq S$ at any
one parameter implies $\dim_{\mathrm H} P_r(\mathcal E)\geq S$ for Haar-almost
every $r$.

\label{thm:intro-anisotropic-maximality}

<!-- theorem-4: body_md -->
\label{thm:dhar}
Let $m,\ell\geq1$, $1\leq h\leq p^\ell$, and
$0<\varepsilon\leq1$. Suppose that $A\subset\mathcal R_\ell^m$
meets a line in at least $h$ points in each of at least
$\varepsilon\,\#\mathbb P^{m-1}(\mathcal R_\ell)$ projective directions. Then
\begin{equation}\label{eq:dhar-set}
 \#A\geq
 \frac{\varepsilon\,h^m}
 {[2(\ell+\lceil\log_p m\rceil)]^m}.
\end{equation}
This is \cite[Theorem~1.8]{DharSet}, with the residue exponent denoted
by $\ell$.

<!-- theorem-5: body_md -->
\label{thm:Qp-lines}
Let $m\geq1$ and let $D\subset\mathbb P^{m-1}(\mathbb Q_p)$ be nonempty.
Suppose that for every $\theta\in D$ there are $a_\theta\in\mathbb Q_p^m$ and
$v_\theta\in\mathbb Q_p^m\setminus\{0\}$ with $[v_\theta]=\theta$ such that
\begin{equation*}
 a_\theta+\mathbb Z_pv_\theta\subset A\subset\mathbb Q_p^m.
\end{equation*}
Then
\begin{equation}\label{eq:Qp-lines}
 \dim_{\mathrm H} A\geq\dim_{\mathrm H} D+1.
\end{equation}
<!-- end theorem-5 -->
<!-- proof-2: body_md -->
We first prove the theorem when $A\subset\mathbb Z_p^m$ and the
prescribed segments have unit length. Write these segments as
\begin{equation*}
 I_\theta\mathrel{:=} \{a_\theta+tu_\theta:t\in\mathbb Z_p\}\subset A,
 \quad a_\theta\in\mathbb Z_p^m,\quad\norm{u_\theta}=1,\quad[u_\theta]=\theta, \quad \theta\in D.
\end{equation*}
Fix $\alpha>\dim_{\mathrm H} A$, an integer $\ell_0\geq1$, and $\varepsilon_0>0$.
Choose a cover of $A$ by balls of radii $p^{-\ell}$, $\ell\geq \ell_0$, satisfying
\begin{equation*}
 \sum_{\ell\geq \ell_0}N_\ell p^{-\ell\alpha}<\varepsilon_0,
\end{equation*}
where $N_\ell$ is the number of distinct balls at scale $p^{-\ell}$. Let $B_\ell$ be their
union, so $A\subset\bigcup_{\ell\geq \ell_0}B_\ell$.

For each $\theta\in D$, define
\begin{equation*}
 T_{\theta,\ell}\mathrel{:=} \{t\in\mathbb Z_p:a_\theta+tu_\theta\in B_\ell\},\qquad
 \beta_\ell\mathrel{:=} \frac{1}{2\ell(\ell+1)}.
\end{equation*}
Observe that $T_{\theta,\ell}$ is a finite union of parameter
residue classes. Since the sets $B_\ell$ cover $I_\theta$, we have
\begin{equation*}
 1\leq\sum_{\ell\geq \ell_0}\mathcal L^1(T_{\theta,\ell}),\qquad
 \sum_{\ell\geq1}\beta_\ell=\frac12.
\end{equation*}
Consequently, every direction belongs to at least one of the sets
\begin{equation*}
 D_\ell\mathrel{:=} \{\theta\in D:\mathcal L^1(T_{\theta,\ell})\geq\beta_\ell\},\qquad
 D=\bigcup_{\ell\geq \ell_0}D_\ell.
\end{equation*}

Write $\bar x$ for the reduction modulo $p^\ell$ of an integral
vector $x\in\mathbb Z_p^m$. For projective directions, write
$\operatorname{red}_\ell$ for reduction using a primitive integral representative.
This is independent of the representative, since any two such
representatives differ by a unit. Put
\begin{equation*}
 S_\ell\mathrel{:=} B_\ell\bmod p^\ell\subset \mathcal R_\ell^m,
 \qquad J_\ell\mathrel{:=} \#\operatorname{red}_\ell(D_\ell).
\end{equation*}
By definition, $\#S_\ell=N_\ell$. For each $\theta\in D_\ell$,
consider the line over $\mathcal R_\ell$ given by
\begin{equation*}
 L_{\theta,\ell}\mathrel{:=} \{\overline{a_\theta}+\bar t\,\overline{u_\theta}:\bar t\in \mathcal R_\ell\}.
\end{equation*}
The parameter map is injective because $u_\theta$ has a unit
coordinate. Moreover, membership in $B_\ell$ depends only on
$t\bmod p^\ell$. Each such parameter class has measure
$p^{-\ell}$, and hence
\begin{equation*}
 \#(S_\ell\cap L_{\theta,\ell})=p^\ell\mathcal L^1(T_{\theta,\ell})\geq\beta_\ell p^\ell.
\end{equation*}
Choose one direction in $D_\ell$ for each distinct reduction.
Applying \eqref{eq:dhar-J} with
$h=\lceil\beta_\ell p^\ell\rceil$, we obtain
\begin{align*}
 N_\ell
 &\geq c_{p,m}(\ell+1)^{-m}
 \frac{J_\ell(\beta_\ell p^\ell)^m}{p^{\ell(m-1)}}\notag\\
 &\geq c'_{p,m}(\ell+1)^{-3m}J_\ell p^\ell.
\end{align*}
Therefore
\begin{equation}\label{eq:direction-cover-count}
 J_\ell\leq C_{p,m}(\ell+1)^{3m}N_\ell p^{-\ell}.
\end{equation}

We next turn this count into a cover of the direction set.
Recall that the reduction map
\begin{equation*}
 \operatorname{red}_\ell:\mathbb P^{m-1}(\mathbb Q_p)\longrightarrow\mathbb P^{m-1}(\mathcal R_\ell)
\end{equation*}
is defined using any primitive integral representative. Its fibers are
projective balls of radius $p^{-\ell}$. Indeed, in a fiber one can choose a common
unit coordinate and normalize it to $1$. Equality of reductions then means
that all remaining coordinates agree modulo $p^\ell$. Thus $D_\ell$ is covered by
$J_\ell$ such balls, and their union over $\ell$ covers $D$.

Fix $\eta>0$, and write $\xi\mathrel{:=} \alpha-1+\eta$. Because $A$ contains a unit segment,
$\dim_{\mathrm H} A\geq1$, so $\xi>0$. By \eqref{eq:direction-cover-count}, the
$\xi$-dimensional cost of the direction cover is at most
\begin{align*}
 \sum_{\ell\geq \ell_0}J_\ell p^{-\ell\xi}
 &\leq C_{p,m}\sum_{\ell\geq \ell_0}
 (\ell+1)^{3m}N_\ell p^{-\ell(\xi+1)}\notag\\
 &=C_{p,m}\sum_{\ell\geq \ell_0}
 (\ell+1)^{3m}p^{-\ell\eta}N_\ell p^{-\ell\alpha}\notag\\
 &\leq C_{p,m,\eta}\sum_{\ell\geq \ell_0}N_\ell p^{-\ell\alpha}
 <C_{p,m,\eta}\varepsilon_0.
\end{align*}
The last inequality uses
$\sup_{\ell\geq1}(\ell+1)^{3m}p^{-\ell\eta}<\infty$.
Since $\varepsilon_0$ and $p^{-\ell_0}$ may be arbitrarily small,
it follows that $\mathcal H^{\alpha-1+\eta}(D)=0$.
Letting $\alpha\downarrow\dim_{\mathrm H} A$ and then $\eta\downarrow0$
proves the assertion for unit segments in $\mathbb Z_p^m$.

It remains to remove the restrictions on the lengths and locations
of the segments. Partition $\mathbb Q_p^m$ into its countably many
unit-ball cosets $c_\nu+\mathbb Z_p^m$, and define
\begin{equation*}
 A_*\mathrel{:=}
 \bigcup_{h\in\mathbb Z}\ \bigcup_\nu
 \bigl((p^hA\cap(c_\nu+\mathbb Z_p^m))-c_\nu\bigr)\subset\mathbb Z_p^m.
\end{equation*}
Each member of this countable union is a translated subset of a similarity
image of $A$, so $\dim_{\mathrm H} A_*\leq\dim_{\mathrm H} A$.
For a prescribed segment $a_\theta+\mathbb Z_pv_\theta$, choose $h\in\mathbb Z$ such that
$\norm{v_\theta}=p^h$. Then $\norm{p^hv_\theta}=1$, and the segment
$p^ha_\theta+\mathbb Z_p(p^hv_\theta)$ lies in one unit-ball coset.
After translation, this is a unit segment in $A_*$ with direction
$\theta$. Applying the result already proved, we obtain
\begin{equation*}
 \dim_{\mathrm H} A\geq\dim_{\mathrm H} A_*\geq\dim_{\mathrm H} D+1.
\end{equation*}
This completes the proof.

<!-- lemma-1: body_md -->
\label{lem:products}
For every nonempty $X\subset \mathbb Q_p^a$ and every integer $b\geq0$,
\begin{equation}\label{eq:product-dimension}
 \dim_{\mathrm H}(X\times \mathbb Q_p^b)=\dim_{\mathrm H} X+b.
\end{equation}
The same equality holds with the second factor replaced by any nonempty open
subset of $\mathbb Q_p^b$.
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
The case $b=0$ is immediate, so we assume $b>0$. By countable stability and bi-Lipschitz invariance of
Hausdorff dimension, it suffices to treat the factor $\mathbb Z_p^b$, because 
every nonempty open subset of $\mathbb Q_p^b$ is a countable union of
balls, and $X\times B$ is bi-Lipschitz equivalent to
$X\times\mathbb Z_p^b$ for every such ball $B$.

For $s\geq0$ and $0<\rho\leq1$, let $\mathcal H_\rho^s$ denote
Hausdorff content computed using balls of positive radii
$r\in p^{\mathbb Z}$, $r\leq\rho$, with cost $r^s$.
Given a ball cover $\{B_i\}$ of $X$, with radii $r_i\leq\rho$,
partition $\mathbb Z_p^b$ into $r_i^{-b}$ balls of radius $r_i$.
The resulting product cover gives
\begin{equation*}
 \mathcal H_\rho^{s+b}(X\times\mathbb Z_p^b)
 \leq \inf_{\{B_i\}}\sum_i r_i^{-b}r_i^{s+b}
 =\mathcal H_\rho^s(X).
\end{equation*}

Conversely, let $\{A_i\times C_i\}$ be a ball cover of
$X\times\mathbb Z_p^b$, where both factors have radius $r_i\leq\rho$.
For each $y\in\mathbb Z_p^b$, the balls $A_i$ with $y\in C_i$ cover $X$.
Integrating against $\mathcal{L}^b$ yields
\begin{equation*}
 \mathcal H_\rho^s(X)
 \leq \int_{\mathbb Z_p^b}\sum_i r_i^s\mathbf 1_{C_i}(y)\,d\mathcal{L}^b(y)
 \leq \sum_i r_i^{s+b}.
\end{equation*}
Taking the infimum over these covers proves the reverse inequality.
Hence
\begin{equation*}
 \mathcal H_\rho^{s+b}(X\times\mathbb Z_p^b)
 =\mathcal H_\rho^s(X).
\end{equation*}
Letting $\rho\downarrow0$ and varying $s$ gives
$\dim_{\mathrm H}(X\times\mathbb Z_p^b)=\dim_{\mathrm H} X+b$; when $\dim_{\mathrm H} X=0$,
the lower bound follows by taking $s=0$, since $X\neq\varnothing$.

\label{lem:lifting}
Suppose that for every $w$ in a set $W\subset\mathbb Q_p^n\setminus\{0\}$
there is a polynomial map
\begin{equation*}
 \gamma_w(t)\mathrel{:=} a_0(w)+a_1(w)t+\cdots+a_{d-1}(w)t^{d-1}+wt^d,
 \qquad \gamma_w(\mathbb Z_p)\subset B.
\end{equation*}
Then $K_B$ contains a nondegenerate affine line segment in every direction in
\begin{equation}\label{eq:lift-directions}
 D(W)\mathrel{:=} \bigl\{
 [u_0:\cdots:u_{d-2}:w:1]:
 u_j\in \mathbb Q_p^n,\ w\in W
 \bigr\}\subset\mathbb P^{dn}(\mathbb Q_p).
\end{equation}
When $d=1$, the list $u_0,\ldots,u_{d-2}$ is empty.
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
Fix $w\in W$, choose arbitrary $u_0,\ldots,u_{d-2}\in \mathbb Q_p^n$, and put
$u_{d-1}\mathrel{:=} w$. Define
\begin{equation*}
 b_0\mathrel{:=} a_0(w),\qquad b_j\mathrel{:=} a_j(w)-u_{j-1}\quad(1\leq j\leq d-1).
\end{equation*}
Consider the line
\begin{equation*}
 \mathcal L(t)\mathrel{:=} (b_0+tu_0,\ldots,b_{d-1}+tu_{d-1},t),\qquad t\in \mathbb Q_p.
\end{equation*}
We claim that it has the required property. Its last coordinate is $t$, so it is
nondegenerate. Substituting into the definition of $\Phi$, we have
\begin{align*}
 \Phi(\mathcal L(t))
 &=\sum_{j=0}^{d-1}t^j(b_j+tu_j)\notag\\
 &=b_0+\sum_{j=1}^{d-1}(b_j+u_{j-1})t^j+u_{d-1}t^d\notag\\
 &=\gamma_w(t).
\end{align*}
Since $\gamma_w(\mathbb Z_p)\subset B$, this implies
\begin{equation*}
 \mathcal L(\mathbb Z_p)\subset K_B,
 \qquad \mathrm{direction}\,\mathcal L=[u_0:\cdots:u_{d-2}:w:1].
\end{equation*}
The direction vector has norm at least $1$. Thus, by restricting
$t$ to a suitable ball $p^h\mathbb Z_p\subset\mathbb Z_p$, the segment
contains a unit segment. The vectors $u_0,\ldots,u_{d-2}$ were
arbitrary, so every stated direction occurs. This proves the lemma.

<!-- lemma-4: body_md -->
\label{lem:lift-dimension}
For nonempty $B\subset \mathbb Q_p^n$,
\begin{equation}\label{eq:lift-dimension}
 \dim_{\mathrm H} K_B=\dim_{\mathrm H} B+(d-1)n+1.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
By Lemma~\ref{lem:automorphism}, the map \eqref{eq:psi} sends
$B\times \mathbb Q_p^{(d-1)n+1}$ onto $K_B$. Suppose that the
coordinates of two inputs have norm at most $R\geq1$, and their
supremum distance is $\delta$. For $j\geq1$, we have
\begin{equation*}
 z^j-w^j=(z-w)\sum_{j'=0}^{j-1}z^{j-1-j'}w^{j'},
 \qquad |z^j-w^j|_p\leq R^{j-1}|z-w|_p.
\end{equation*}
Adding and subtracting $z^jy_j$, then using the ultrametric inequality, yields
\begin{align*}
 \norm{z^jx_j-w^jy_j}
 &=\norm{z^j(x_j-y_j)+(z^j-w^j)y_j}\notag\\
 &\leq\max\bigl\{|z|_p^j\norm{x_j-y_j},
 |z^j-w^j|_p\norm{y_j}\bigr\}
 \leq R^j\delta.
\end{align*}
It follows that $\norm{\Psi(P)-\Psi(Q)}\leq R^{d-1}\norm{P-Q}$ on that box.
The inverse satisfies the same kind of estimate. Taking the countable union of bounded boxes covering $\mathbb Q_p^{dn+1}$, we obtain
\begin{equation*}
 \dim_{\mathrm H} K_B=\dim_{\mathrm H}(B\times \mathbb Q_p^{(d-1)n+1}).
\end{equation*}
By \eqref{eq:product-dimension}, the right-hand side is
$\dim_{\mathrm H} B+(d-1)n+1$. This proves \eqref{eq:lift-dimension}.

\label{rem:finite-extensions}
The curved Kakeya theorem also holds over a finite extension
$F/\mathbb Q_p$, with ring of integers $\mathcal O_F$, when the hypothesis
is $\gamma_\omega(\mathcal O_F)\subset E$. To adapt the proof,

$F$-metric. Write $\dim_{\mathrm H}^{\mathbb Q_p}$ for Hausdorff dimension in
$\mathbb Q_p$ coordinates and $\dim_{\mathrm H}^F$ for Hausdorff dimension in the

each $F$-direction form $\mathbb P^{\kappa-1}(\mathbb Q_p)$. Local projective
product charts and Lemma~\ref{lem:products} therefore give
\begin{equation*}
 \dim_{\mathrm H}^{\mathbb Q_p}\widetilde D
 =\kappa\dim_{\mathrm H}^F D+(\kappa-1).
\end{equation*}

$\mathbb Q_p$-segment in each such direction: multiply a given scalar by
a sufficiently large power of $p$ to place its $\mathbb Z_p$-multiples
inside the parameter ball. The segment form of \eqref{eq:Qp-lines}
now yields
\begin{equation*}
 \kappa\dim_{\mathrm H}^F A=\dim_{\mathrm H}^{\mathbb Q_p}A
 \geq\dim_{\mathrm H}^{\mathbb Q_p}\widetilde D+1
 =\kappa\dim_{\mathrm H}^F D+\kappa.
\end{equation*}
Thus $\dim_{\mathrm H}^F A\geq\dim_{\mathrm H}^F D+1$, as required for the line estimate
over $F$.

Unit reparametrizations preserve $\mathcal O_F$, and the same Hensel argument
shows that $\mathcal O_F^\times/(\mathcal O_F^\times)^d$ is finite.
Using powers of a uniformizer in the countable cone decomposition, the
leading-vector lemma and the lifting proof are unchanged. They give
$\dim_{\mathrm H} E\geq\dim_{\mathrm H} V+1$ under $\gamma_\omega(\mathcal O_F)\subset E$.

bounded ball $B_0$ such that $\dim_{\mathrm H}(A\cap B_0)>\tau$.

 d_p=d_2^{\log_2 p}.

<!-- proof-11: body_md -->
Put $q=p^\ell$, and let $\eta_k$ denote normalized counting
measure on $\mathcal R_\ell^k$. For a nonnegative function
$g:\mathcal R_\ell^m\to[0,\infty)$, define
\begin{equation*}
 \mathcal M_\ell g(u)
 \mathrel{:=}\max_{w\in\mathcal R_\ell^{m-1}}
       \frac1q\sum_{t\in\mathcal R_\ell}g(t,w+tu),
 \qquad u\in\mathcal R_\ell^{m-1}.
\end{equation*}
We first establish the finite-ring bound with all norms taken
with respect to these probability measures.

Let $A\subset\mathcal R_\ell^m$ and $0<\alpha<1$.
If $\mathcal M_\ell\mathbf 1_A(u)>\alpha$, then $A$ meets a line
of direction $[1:u]$ in at least $\lceil q\alpha\rceil$ points.
Distinct slopes give distinct projective directions. Applying
Corollary~\ref{cor:dhar-directions} to these directions and
rearranging gives
\begin{equation*}
 \eta_{m-1}\{u:\mathcal M_\ell\mathbf 1_A(u)>\alpha\}
 \leq C_{p,m}(\ell+1)^m\alpha^{-m}\eta_m(A).
\end{equation*}
This is the restricted weak-type estimate obtained from the set
bound. Since $0\leq\mathcal M_\ell\mathbf 1_A\leq1$, integration of
its distribution function gives, for nonempty $A$,
\begin{equation*}
    \begin{aligned}
 \|\mathcal M_\ell\mathbf 1_A\|_{L^m(\eta_{m-1})}^m
 &=m\int_0^1\alpha^{m-1}
       \eta_{m-1}\{\mathcal M_\ell\mathbf 1_A>\alpha\}\,d\alpha\\
 &\leq q^{-m}
       +C_{p,m}(\ell+1)^m\eta_m(A)
                         \int_{q^{-1}}^1\frac{d\alpha}{\alpha}\\
 &\leq C_{p,m}(\ell+1)^{m+1}\eta_m(A).
\end{aligned}
\end{equation*}
Here we used that $\eta_{m-1}$ is a probability measure,
$\log q=\ell\log p$, and $q^{-m}\leq\eta_m(A)$ for nonempty $A$.
The estimate is immediate when $A$ is empty. Thus
\begin{equation*}
 \|\mathcal M_\ell\mathbf 1_A\|_{L^m(\eta_{m-1})}
 \leq C_{p,m}(\ell+1)^{1+1/m}\eta_m(A)^{1/m}.
\end{equation*}

For a general nonzero $g$, normalize
$\|g\|_{L^m(\eta_m)}=1$. Every point has $\eta_m$-mass
$q^{-m}$, so $\|g\|_\infty\leq q$.
Put $L=\lceil\log_2q\rceil$ and
\begin{equation*}
 A_j\mathrel{:=}\{x:2^j<g(x)\leq2^{j+1}\},
 \qquad 0\leq j<L.
\end{equation*}
These sets are disjoint and satisfy
\begin{equation*}
 g\leq1+2\sum_{j=0}^{L-1}2^j\mathbf 1_{A_j},
 \qquad
 \sum_{j=0}^{L-1}2^{jm}\eta_m(A_j)\leq1.
\end{equation*}
Sublinearity, the indicator-function estimate, and H\"older's
inequality in the finite sum imply
\begin{equation*}
\begin{aligned}
 \|\mathcal M_\ell g\|_{L^m(\eta_{m-1})}
 &\leq1+C_{p,m}(\ell+1)^{1+1/m}
                    \sum_{j=0}^{L-1}2^j\eta_m(A_j)^{1/m}\\
 &\leq1+C_{p,m}(\ell+1)^{1+1/m}L^{1-1/m}
       \left(\sum_{j=0}^{L-1}2^{jm}\eta_m(A_j)\right)^{1/m}\\
 &\leq C_{p,m}(\ell+1)^2,
\end{aligned}
\end{equation*}
since $L\leq C_p(\ell+1)$. By homogeneity, including the case
$g=0$, this proves
\begin{equation*}
 \|\mathcal M_\ell g\|_{L^m(\eta_{m-1})}
 \leq C_{p,m}(\ell+1)^2\|g\|_{L^m(\eta_m)}.
\end{equation*}

To pass to $\mathbb Z_p^m$, average $|f|$ over residue classes:
\begin{equation*}
 g(\bar x)\mathrel{:=} q^m
       \int_{x+p^\ell\mathbb Z_p^m}|f(y)|\,dy,
 \qquad \bar x\in\mathcal R_\ell^m.
\end{equation*}
Jensen's inequality gives
$\|g\|_{L^m(\eta_m)}\leq\|f\|_{L^m(\mathbb Z_p^m)}$.
The tube in direction $[1:u]$ with intercept $w$ is the union
of the $q$ residue classes corresponding to
$(\bar t,\bar w+\bar t\bar u)$, with $\bar t\in\mathcal R_\ell$.
Its normalized integral is therefore
$q^{-1}\sum_{\bar t}g(\bar t,\bar w+\bar t\bar u)$.
Consequently,
\begin{equation*}
 \mathcal K_{p^{-\ell}}f(u)=\mathcal M_\ell g(\bar u),
 \qquad
 \|\mathcal K_{p^{-\ell}}f\|_{L^m(\mathbb Z_p^{m-1})}
   =\|\mathcal M_\ell g\|_{L^m(\eta_{m-1})}.
\end{equation*}
The finite-ring estimate proves
\eqref{eq:projection-tube-maximal}. Finally,
$(\ell+1)^2\leq C_{p,\varepsilon}p^{\varepsilon\ell}$ gives
the stated $\delta^{-\varepsilon}$ form.

\label{thm:quantitative-line-slicing}
Let $Y\subset\mathbb Q_p^{m-1}\times\mathbb Q_p^{m-1}$ be nonempty and analytic, where $m\geq2$.
Define the slope set and the slices by
\begin{equation*}
 D\mathrel{:=} \{u:\exists w,\ (u,w)\in Y\},\qquad
 H_t\mathrel{:=}\{w+tu:(u,w)\in Y\},\qquad s\mathrel{:=}\dim_{\mathrm H} D.
\end{equation*}
The incidence set
\begin{equation*}
 \mathcal I(Y)\mathrel{:=} \{(t,w+tu):t\in\mathbb Q_p,\ (u,w)\in Y\}\subset\mathbb Q_p^m
\end{equation*}
satisfies
\begin{equation*}
 \dim_{\mathrm H}\mathcal I(Y)\geq s+1.
\end{equation*}
Moreover,
\begin{equation*}
 \dim_{\mathrm H} H_t\geq s
 \quad\text{for Haar-almost every }t\in\mathbb Q_p.
\end{equation*}
For every $0\leq\sigma<s$,
\begin{equation}\label{eq:quantitative-line-slicing}
 \dim_{\mathrm H}\{t\in\mathbb Q_p:\dim_{\mathrm H} H_t\leq\sigma\}
 \leq1-\frac{s-\sigma}{m-1}.
\end{equation}

We first prove the incidence bound. For every $u\in D$, choose
$w$ such that $(u,w)\in Y$. The line
$\{(t,w+tu):t\in\mathbb Q_p\}$ lies in $\mathcal I(Y)$ and has direction
$[1:u]$. On bounded slope sets, $u\mapsto[1:u]$ is bi-Lipschitz onto its
image. Countable localization therefore gives
$\dim_{\mathrm H}\{[1:u]:u\in D\}=s$, and Theorem~\ref{thm:Qp-lines} implies
$\dim_{\mathrm H}\mathcal I(Y)\geq s+1$.
It remains to prove the assertions for individual slices.
We first work with an integral family and integral parameters,
and then remove these restrictions.

\emph{Slice measures in the integral model.}
Suppose that $Y\subset\mathbb Z_p^{m-1}\times\mathbb Z_p^{m-1}$ and
$t\in\mathbb Z_p$. If $s=0$, every $H_t$ is nonempty and the
conclusion is immediate. There are also no exceptional sets
to consider. We may therefore assume that $s>0$.
Fix $0<\tau<s$.
The set $D$ is analytic, being a coordinate projection of $Y$.
Lemma~\ref{lem:projection-frostman} supplies a probability measure $\nu$
carried by a compact subset of $D$ such that
\begin{equation*}
 \nu(B(u,\delta))\leq C_\nu\delta^\tau,
 \qquad \delta\mathrel{:=} p^{-\ell},\quad\ell\geq1.
\end{equation*}
By Theorem~\ref{thm:projection-selection}, choose a Borel probability
measure $\lambda$ carried by $Y$ whose slope marginal is $\nu$.
Thus the slope distribution of $\lambda$ satisfies the preceding
Frostman bound. Define the slice measures and their averaged
densities by
\begin{equation*}
 \mu_t\mathrel{:=} ((u,w)\mapsto w+tu)_*\lambda,
 \qquad
 G_\delta(t,z)\mathrel{:=} \delta^{-(m-1)}\mu_t(B(z,\delta)).
\end{equation*}
For every $t$, the measure $\mu_t$ is a probability measure
carried by $H_t$. The density $G_\delta(t,\cdot)$ distributes
the mass of each residue class uniformly over that class.
We also average the slope measure at scale $\delta$ by setting
\begin{equation*}
 \nu_\delta(u)\mathrel{:=} \delta^{-(m-1)}\nu(B(u,\delta)).
\end{equation*}
Every ball of radius $\delta$ in $\mathbb Z_p^{m-1}$ has Haar measure
$\delta^{m-1}$, so Tonelli's theorem gives
\begin{equation*}
 \int\nu_\delta(u)\,du
 =\delta^{-(m-1)}\int\!\int
       \mathbf 1_{\{\|u-v\|\leq\delta\}}\,du\,d\nu(v)=1.
\end{equation*}
The Frostman estimate gives
$\|\nu_\delta\|_\infty\leq C_\nu\delta^{-(m-1-\tau)}$.
For $m'\mathrel{:=} m/(m-1)$, the conjugate exponent to $m$, it follows that
\begin{align*}
 \|\nu_\delta\|_{L^{m'}}^{m'}
 &\leq\|\nu_\delta\|_\infty^{m'-1}\|\nu_\delta\|_1\notag\\
 &\leq C_\nu^{m'-1}
          \delta^{-(m-1-\tau)(m'-1)}.
\end{align*}
Since $(m'-1)/m'=1/m$, taking the $m'$th root yields
\begin{equation}\label{eq:projection-marginal-norm}
 \|\nu_\delta\|_{L^{m'}(\mathbb Z_p^{m-1})}
 \leq C\delta^{-(m-1-\tau)/m}.
\end{equation}

\emph{Duality and the averaged mass estimate.}
We now estimate the norm of $G_\delta$. Let $f\geq0$ belong
to $L^m(\mathbb Z_p^m)$. By the definition of $\mu_t$ and Tonelli's
theorem, we have
\begin{align*}
 \int_{\mathbb Z_p}\int_{\mathbb Z_p^{m-1}}f(t,z)G_\delta(t,z)\,dz\,dt
 &=\int_Y\int_{\mathbb Z_p}\delta^{-(m-1)}
       \int_{B(w+tu,\delta)}f(t,z)\,dz\,dt\,d\lambda(u,w)\notag\\
 &\leq\int_{\mathbb Z_p^{m-1}}\mathcal K_\delta f(u)\,d\nu(u).
\end{align*}
If $\|u-u'\|\leq\delta$, then
$\|tu-tu'\|\leq\delta$ for every $t\in\mathbb Z_p$. Equal-radius balls
with these centers coincide. Hence $\mathcal K_\delta f$ is constant on every slope ball
of radius $\delta$. It follows from the definition of
$\nu_\delta$ that
\begin{equation*}
 \int\mathcal K_\delta f\,d\nu
 =\int\mathcal K_\delta f(u)\nu_\delta(u)\,du.
\end{equation*}
H\"older's inequality, Lemma~\ref{lem:projection-tube-maximal}, and
\eqref{eq:projection-marginal-norm} now give
\begin{align*}
 \int fG_\delta
 &\leq\|\mathcal K_\delta f\|_{L^m}\|\nu_\delta\|_{L^{m'}}\notag\\
 &\leq C(\ell+1)^2\delta^{-(m-1-\tau)/m}\|f\|_{L^m}.
\end{align*}
Taking the supremum over all nonnegative $f$ with $L^m$ norm
at most one, and applying $L^m$--$L^{m'}$ duality, we obtain
\begin{equation}\label{eq:projection-dual-density}
 \|G_\delta\|_{L^{m'}(\mathbb Z_p^m)}
 \leq C(\ell+1)^2\delta^{-(m-1-\tau)/m}.
\end{equation}
All constants here may depend on $\nu$, but they are independent of
$\ell$.

To use this norm bound for the slice measures, define
\begin{equation}\label{eq:projection-fractional-mass}
 F_\delta(t)\mathrel{:=} \int\mu_t(B(z,\delta))^{1/(m-1)}\,d\mu_t(z).
\end{equation}
Let $\mathcal B_\delta$ be the finite partition of $\mathbb Z_p^{m-1}$ into
balls of radius $\delta$, and write $m_B(t)\mathrel{:=} \mu_t(B)$.
For $z\in B$, the ball $B(z,\delta)$ is exactly $B$, so
\begin{equation*}
 F_\delta(t)
 =\sum_{B\in\mathcal B_\delta}m_B(t)^{1+1/(m-1)}
 =\sum_{B\in\mathcal B_\delta}m_B(t)^{m'}.
\end{equation*}
This is the order-$m'$ moment of the masses of the partition
cells. Related projection results for dimension spectra were
studied by Hunt and Kaloshin \cite{HK}. In the present argument,
we can estimate this moment directly from the preceding norm bound.
Indeed,
\begin{align*}
 \int_{\mathbb Z_p^{m-1}}G_\delta(t,z)^{m'}\,dz
 &=\sum_{B\in\mathcal B_\delta}
       \delta^{m-1}\bigl(\delta^{-(m-1)}m_B(t)\bigr)^{m'}\notag\\
 &=\delta^{(m-1)(1-m')}\sum_{B\in\mathcal B_\delta}m_B(t)^{m'}
 =\delta^{-1}F_\delta(t).
\end{align*}
Thus \eqref{eq:projection-dual-density} implies
\begin{equation}\label{eq:projection-averaged-mass}
\begin{aligned}
 \int_{\mathbb Z_p}F_\delta(t)\,dt
 &=\delta\|G_\delta\|_{L^{m'}}^{m'}\\
 &\leq C(\ell+1)^{2m'}
       \delta^{\,1-(m-1-\tau)m'/m}\\
 &=C(\ell+1)^{2m'}\delta^{\tau/(m-1)}.
\end{aligned}
\end{equation}

\emph{Covers of the exceptional parameters.}
Fix $0\leq\sigma<\beta<\tau$, and put $\delta_\ell\mathrel{:=} p^{-\ell}$.
Let the set of parameters where the moment exceeds the desired
bound be
\begin{equation*}
 A_\ell(\beta)\mathrel{:=} \{t\in\mathbb Z_p:
       F_{\delta_\ell}(t)>\delta_\ell^{\beta/(m-1)}\}.
\end{equation*}
By Markov's inequality and \eqref{eq:projection-averaged-mass},
\begin{align*}
 \mathcal L^1(A_\ell(\beta))
 &\leq\delta_\ell^{-\beta/(m-1)}
                        \int_{\mathbb Z_p}F_{\delta_\ell}(t)\,dt\notag\\
 &\leq C(\ell+1)^{2m'}
                  \delta_\ell^{(\tau-\beta)/(m-1)}.
\end{align*}
We claim that $A_\ell(\beta)$ is a union of parameter residue
classes of radius $\delta_\ell$. To see this, suppose that
$|t-t'|_p\leq\delta_\ell$. Then
\begin{equation*}
 \|(w+tu)-(w+t'u)\|
 =|t-t'|_p\|u\|\leq\delta_\ell
 \qquad ((u,w)\in Y).
\end{equation*}
Therefore the two evaluations lie in the same spatial residue class,
$\mu_t(B)=\mu_{t'}(B)$ for every $B\in\mathcal B_{\delta_\ell}$,
and $F_{\delta_\ell}(t)=F_{\delta_\ell}(t')$.
This proves the claim. Each parameter ball has Haar measure
$\delta_\ell$, so the measure estimate above shows that the
number of balls in $A_\ell(\beta)$ is at most
\begin{equation}\label{eq:projection-bad-cover}
 C(\ell+1)^{2m'}
       \delta_\ell^{-1+(\tau-\beta)/(m-1)}.
\end{equation}

Let
\begin{equation*}
 A(\beta)\mathrel{:=} \limsup_{\ell\to\infty}A_\ell(\beta)
 =\bigcap_{\ell_0\geq1}\bigcup_{\ell\geq\ell_0}A_\ell(\beta).
\end{equation*}
For $\eta>0$, set $\alpha\mathrel{:=} 1-(\tau-\beta)/(m-1)+\eta$.
The covers in \eqref{eq:projection-bad-cover}, with $\ell\geq\ell_0$,
cover $A(\beta)$ and have total $\alpha$-dimensional cost at most
\begin{align*}
 C\sum_{\ell\geq\ell_0}(\ell+1)^{2m'}
       \delta_\ell^{-1+(\tau-\beta)/(m-1)}\delta_\ell^\alpha
 &=C\sum_{\ell\geq\ell_0}(\ell+1)^{2m'}p^{-\ell\eta}.
\end{align*}
The series converges. As $\ell_0\to\infty$, both the total
cost and the largest covering radius tend to zero. Therefore
$\mathcal H^\alpha(A(\beta))=0$. Letting $\eta\downarrow0$, we obtain
\begin{equation}\label{eq:projection-limsup-bound}
 \dim_{\mathrm H} A(\beta)\leq1-\frac{\tau-\beta}{m-1}.
\end{equation}

\emph{From mass decay to slice dimension.}
We now show that $\dim_{\mathrm H} H_t\geq\beta$ whenever
$t\notin A(\beta)$. Fix such a parameter. By the definition of
the limsup set, for every sufficiently large $\ell$ we have
\begin{equation*}
 F_{\delta_\ell}(t)\leq\delta_\ell^{\beta/(m-1)}.
\end{equation*}
Fix $0<\gamma<\beta$, and define
\begin{equation*}
 E_\ell\mathrel{:=} \{z:\mu_t(B(z,\delta_\ell))>\delta_\ell^\gamma\}.
\end{equation*}
On $E_\ell$, the integrand in \eqref{eq:projection-fractional-mass}
is larger than $\delta_\ell^{\gamma/(m-1)}$. Hence
\begin{align*}
 \mu_t(E_\ell)
 &\leq\delta_\ell^{-\gamma/(m-1)}F_{\delta_\ell}(t)\notag\\
 &\leq\delta_\ell^{(\beta-\gamma)/(m-1)}.
\end{align*}
Since the right-hand side is summable in $\ell$, Borel--Cantelli
implies that $\mu_t$-almost every $z$ belongs to only finitely
many sets $E_\ell$. Thus
\begin{equation*}
 \mu_t(B(z,p^{-\ell}))\leq p^{-\ell\gamma}
 \quad\text{for all sufficiently large $\ell$, for $\mu_t$-almost every $z$}.
\end{equation*}

To obtain the Hausdorff dimension bound, we restrict to a set
where this estimate holds uniformly. For $J\geq1$, define
\begin{equation*}
 K_J\mathrel{:=} \bigcap_{\ell\geq J}
       \{z\in\mathbb Z_p^{m-1}:\mu_t(B(z,p^{-\ell}))\leq p^{-\ell\gamma}\}.
\end{equation*}
These sets are Borel, and $\mu_t(\bigcup_JK_J)=1$.
Choose $J$ with $\mu_t(K_J)>0$, and define a Borel probability measure
\begin{equation*}
 \widetilde\mu(A)\mathrel{:=} \frac{\mu_t(A\cap K_J)}{\mu_t(K_J)}.
\end{equation*}
It is still carried by $H_t$. If a ball $B(x,p^{-\ell})$, $\ell\geq J$,
meets $K_J$, choose $z$ in the intersection. The ultrametric property
gives $B(x,p^{-\ell})=B(z,p^{-\ell})$, and therefore
\begin{equation*}
 \widetilde\mu(B(x,p^{-\ell}))
 \leq\frac{p^{-\ell\gamma}}{\mu_t(K_J)}.
\end{equation*}
The same bound is trivial when the ball misses $K_J$. For an arbitrary
small radius $\rho$, choose $\ell\geq J$ with
$p^{-\ell}\leq\rho<p^{-\ell+1}$. The discrete set of distances gives
$B(x,\rho)=B(x,p^{-\ell})$, and hence
\begin{equation*}
 \widetilde\mu(B(x,\rho))
 \leq\frac{p^{-\ell\gamma}}{\mu_t(K_J)}
 \leq\frac{\rho^\gamma}{\mu_t(K_J)}.
\end{equation*}
We have therefore obtained a Frostman bound for every sufficiently
small radius. If $\{B_i\}$ is any small-ball cover of $H_t$, then
\begin{equation*}
 1\leq\sum_i\widetilde\mu(B_i)
 \leq\frac1{\mu_t(K_J)}\sum_i(\operatorname{rad}B_i)^\gamma.
\end{equation*}
It follows that $\mathcal H^\gamma(H_t)>0$ and hence
$\dim_{\mathrm H} H_t\geq\gamma$. Letting $\gamma\uparrow\beta$
proves $\dim_{\mathrm H} H_t\geq\beta$, as claimed. Consequently,
\begin{equation*}
 \{t\in\mathbb Z_p:\dim_{\mathrm H} H_t\leq\sigma\}\subset A(\beta),
\end{equation*}
because $\sigma<\beta$. By \eqref{eq:projection-limsup-bound},
\begin{equation*}
 \dim_{\mathrm H}\{t\in\mathbb Z_p:\dim_{\mathrm H} H_t\leq\sigma\}
 \leq1-\frac{\tau-\beta}{m-1}.
\end{equation*}
Letting first $\beta\downarrow\sigma$ and then $\tau\uparrow s$,
we obtain \eqref{eq:quantitative-line-slicing} for the integral
family with parameters in $\mathbb Z_p$.

\emph{Localization and the almost-everywhere lower bound.}
It remains to treat an arbitrary analytic family and all
parameters in $\mathbb Q_p$. For each integer $A_0\geq0$, define
\begin{equation*}
 Y_{A_0}\mathrel{:=} Y\cap\{(u,w):\|u\|,\|w\|\leq p^{A_0}\},
 \qquad D_{A_0}\mathrel{:=} \{u:\exists w,\ (u,w)\in Y_{A_0}\}.
\end{equation*}
Both $Y_{A_0}$ and $D_{A_0}$ are analytic. Since $D=\bigcup_{A_0\geq0}D_{A_0}$,
countable stability gives $s=\sup_{A_0}\dim_{\mathrm H} D_{A_0}$. Given
$0\leq\sigma<\tau<s$, choose $A_0$ with $s_{A_0}\mathrel{:=} \dim_{\mathrm H} D_{A_0}>\tau$.
The slices of $Y_{A_0}$ are contained in the corresponding
slices of $Y$. Thus, if an original slice has dimension at most
$\sigma$, the same is true for its $Y_{A_0}$-slice. It is
therefore enough to estimate the exceptional parameters of
$Y_{A_0}$.

Fix a parameter ball $p^{-A}\mathbb Z_p$, where $A\geq0$ is an integer, and write
$t_0\mathrel{:=} p^At\in\mathbb Z_p$, so that $t=p^{-A}t_0$.
Choose an integer $\kappa\geq A+A_0$, and
make the invertible linear change
\begin{equation*}
 u'\mathrel{:=} p^{\kappa-A}u,\qquad w'\mathrel{:=} p^{\kappa}w.
\end{equation*}
For $(u,w)\in Y_{A_0}$,
\begin{equation*}
 \|u'\|\leq p^{-\kappa+A+A_0}\leq1,\qquad
 \|w'\|\leq p^{-\kappa+A_0}\leq1.
\end{equation*}
The transformed family is integral, and the invertible linear
change preserves the slope dimension $s_{A_0}$. At parameter
$t_0$, its slice is
\begin{equation*}
 \{w'+t_0u':(u,w)\in Y_{A_0}\}
 =p^{\kappa} H^{Y_{A_0}}_{p^{-A}t_0},
\end{equation*}
where $H^{Y_{A_0}}$ denotes the slices formed using only $Y_{A_0}$.
Applying the integral result, and using that scalar similarities
preserve Hausdorff dimension, gives
\begin{align*}
 \dim_{\mathrm H}\{t\in p^{-A}\mathbb Z_p:\dim_{\mathrm H} H_t\leq\sigma\}
 &\leq1-\frac{s_{A_0}-\sigma}{m-1}\notag\\
 &\leq1-\frac{\tau-\sigma}{m-1}.
\end{align*}
Now let $\tau\uparrow s$ and use
$\mathbb Q_p=\bigcup_{A\geq0}p^{-A}\mathbb Z_p$ to obtain the global bound.

Finally, if $s>0$, choose $0\leq\sigma_i<s$ with $\sigma_i\uparrow s$.
The set of parameters with $\dim_{\mathrm H} H_t<s$ is the countable union of
$\{t:\dim_{\mathrm H} H_t\leq\sigma_i\}$. Each has dimension strictly less than
one by \eqref{eq:quantitative-line-slicing}, and hence Haar outer
measure zero: on $\mathbb Q_p$, a ball of radius $p^{-\ell}$ has Haar
measure $p^{-\ell}$. Their countable union is Haar-null, which proves the
almost-everywhere assertion. The case $s=0$ is immediate because
the slices are nonempty. This completes the proof.

We begin with the lower bound. Let
$S_0\mathrel{:=} \dim_{\mathrm H} P_0(\mathcal E)$. For each
$v=(v_0,\ldots,v_d)\in\mathcal E$, define
\begin{equation*}
 \Gamma_v(t)
 \mathrel{:=} \left(\sum_{j=0}^d a_jv_jt^{d+1-j},\ t^{d+1}\right)
 \in\mathbb Q_p^{n+1},\qquad t\in\mathbb Q_p.
\end{equation*}
The last coordinate is $t^{d+1}$, so the map has degree exactly
$d+1$. Its leading vector is $(a_0v_0,1)$ and is nonzero for
every tuple. Thus the set of leading directions is
\begin{equation*}
 V_0\mathrel{:=} \{[x:1]:x\in P_0(\mathcal E)\}\subset\mathbb P^n(\mathbb Q_p),
 \qquad\dim_{\mathrm H} V_0=S_0.
\end{equation*}

Let $B\mathrel{:=} \bigcup_{v\in\mathcal E}\Gamma_v(\mathbb Q_p)$.
For every direction in $V_0$, the set $B$ contains the image
$\Gamma_v(\mathbb Z_p)$ of a corresponding polynomial. By
Theorem~\ref{thm:padic}, applied in dimension $n+1$ and degree
$d+1$, we have
\begin{equation}\label{eq:augmented-incidence-lower}
 \dim_{\mathrm H} B\geq S_0+1.
\end{equation}

For $t\neq0$, we see that
\begin{equation*}
 \Gamma_v(t)=\bigl(t^{d+1}P_{1/t}(v),t^{d+1}\bigr),
 \qquad\Gamma_v(0)=0.
\end{equation*}
Consequently, for
\begin{equation*}
 \Theta:\mathbb Q_p^\times\times\mathbb Q_p^n\longrightarrow\mathbb Q_p^{n+1},
 \qquad\Theta(r,y)\mathrel{:=} (r^{-(d+1)}y,r^{-(d+1)}),
\end{equation*}
we have the exact set identity
\begin{equation}\label{eq:augmented-incidence-image}
 B=\{0\}\cup\Theta\bigl(\mathcal I^\times(\mathcal E)\bigr).
\end{equation}
On each set where
$|r|_p$ is fixed and $\|y\|$ is bounded, the map $\Theta$ is
Lipschitz. These sets give a countable cover of the domain. Therefore
\eqref{eq:augmented-incidence-image} gives
$\dim_{\mathrm H} B\leq\dim_{\mathrm H}\mathcal I^\times(\mathcal E)$.
Combining this with \eqref{eq:augmented-incidence-lower} proves
the lower bound.

For the upper bound, the map
\begin{equation*}
 F:\mathcal E\times\mathbb Q_p^\times\longrightarrow\mathcal I^\times(\mathcal E),
 \qquad F(v,r)\mathrel{:=} (r,P_r(v))
\end{equation*}
is surjective and Lipschitz on every bounded box. The full-factor lemma \eqref{eq:product-dimension} therefore give
\begin{equation*}
 \dim_{\mathrm H}\mathcal I^\times(\mathcal E)
 \leq\dim_{\mathrm H}(\mathcal E\times\mathbb Q_p^\times)
 =\dim_{\mathrm H}\mathcal E+1.
\end{equation*}
The incidence set is contained in $\mathbb Q_p^{n+1}$, so its dimension
is also at most $n+1$. Combining these two upper bounds completes
the proof.

\label{rem:coefficient-dimension-counterexample}
The condition on the reference image in
\eqref{eq:polynomial-incidence-threshold} cannot be replaced by
$\dim_{\mathrm H}\mathcal E\geq S$. This fails even when $\mathcal E$
is compact, all scalar coefficients are nonzero, and $S\leq n$.
To see this, take $n=2$, $d=1$, and $a_0=a_1=1$.
Let $e\mathrel{:=} (1,0)\in\mathbb Q_p^2$, and define
\begin{equation*}
 \mathcal E\mathrel{:=} \{(xe,ye):x,y\in\mathbb Z_p\}\subset(\mathbb Q_p^2)^2.
\end{equation*}
The map $(x,y)\mapsto(xe,ye)$ is an isometry, so
$\dim_{\mathrm H}\mathcal E=2$. If $\mathcal L\mathrel{:=}\mathbb Q_pe$, then
\begin{equation*}
 P_r(\mathcal E)=(\mathbb Z_p+r\mathbb Z_p)e\subset\mathcal L
 \quad\text{for every }r\in\mathbb Q_p.
\end{equation*}
In fact, each of these images has dimension one, because it contains
$\mathbb Z_pe$ and is contained in $\mathcal L$. Moreover,
\begin{equation*}
 \mathbb Z_p^\times\times(\mathbb Z_pe)
 \subset\mathcal I^\times(\mathcal E)
 \subset\mathbb Q_p^\times\times\mathcal L.
\end{equation*}

\emph{The full inverse-image identity.}

$H_t\mathrel{:=} \{w+tu:(u,w)\in Y\}$. For a vector
$z=(z_0,\ldots,z_{d-1})\in(\mathbb Q_p^n)^d$, put
\begin{equation*}
 R_t(z)\mathrel{:=} \sum_{j=0}^{d-1}t^{d-1-j}z_j.
\end{equation*}
The coefficient of $z_{d-1}$ is $1$, so $R_t$ is surjective for every
$t$. Moreover,
\begin{equation*}
 R_t(z)=\Phi(z_{d-1},z_{d-2},\ldots,z_0,t).
\end{equation*}
Thus $R_t$ is the slice map of the polynomial lift, with the
order of the vector blocks reversed.

By \eqref{eq:projection-lifted-line}, the $j$th block of $w+tu$
is $b_{j+1}-c_{j+1}+tc_j$. Substitution gives
\begin{equation*}
\begin{aligned}
 R_t(w+tu)
 &=\sum_{j=0}^{d-1}t^{d-1-j}
                  (b_{j+1}-c_{j+1}+tc_j)\\

 &=b_0t^d+b_1t^{d-1}+\cdots+b_d=q_b(t).
\end{aligned}
\end{equation*}

$R_t^{-1}(q_b(t))$.

\end{equation*}

\begin{equation*}

\end{equation*}

 =\dim_{\mathrm H} H_t\geq S+(d-1)n.
\end{equation*}

\begin{equation*}

\begin{equation*}
 P_r(\mathcal E_{r_*})=(r-r_*)B,
 \qquad
 \dim_{\mathrm H} P_r(\mathcal E_{r_*})=
 \begin{cases}
  0,&r=r_*,\\
  S,&r\neq r_*.
 \end{cases}
\end{equation*}

image collapses at $r_*$. For every $0\leq\sigma<S$, the exceptional
set is exactly $\{r_*\}$.

exceptional set is optimal. In some scalar cases, stronger bounds are known. For the scalar linear family
$P_r(x,y)\mathrel{:=} x+ry$ and a nonempty Borel set

$\dim_{\mathrm H} P_r(\mathcal E)<u$. Letting $u\downarrow\sigma$ gives
$\max\{2\sigma-T,0\}$, which is often strictly smaller than the
bound $1-S+\sigma$ given by the present theorem.

 z_{i,h_i-1}
 =y_i-\sum_{j=0}^{h_i-2}t^{h_i-1-j}z_{i,j}.
\end{equation*}

 \dim_{\mathrm H} H_t=L+\dim_{\mathrm H}\widetilde P_{1/t}(\mathcal E),
 \qquad t\neq0.
\end{equation}

$m=\Lambda+1$. The slope dimension satisfies
$S+L\leq N+L=\Lambda$.

 L+\dim_{\mathrm H}\widetilde P_{1/t}(\mathcal E)
 =\dim_{\mathrm H} H_t\geq S+L.
\end{equation*}

            \dim_{\mathrm H}\widetilde P_{1/t}(\mathcal E)\leq\sigma\}\notag\\
 &\qquad\leq1-\frac{(S+L)-(\sigma+L)}{\Lambda}

not exceptional when $\sigma<S$. This proves
\eqref{eq:block-projection-ae} and
\eqref{eq:block-projection-exceptional}, completing the proof.

<!-- definition-1: body_md -->
\label{def:block-rectangles}
Fix $\boldsymbol\alpha=(\alpha_1,\ldots,\alpha_k)$ with
$\alpha_i\geq0$. For $0<\delta<1$, put
\begin{equation}\label{eq:block-rectangle-lattice}
 m_i(\delta)\mathrel{:=}
       \left\lceil\alpha_i\log_p(1/\delta)\right\rceil,
 \qquad
 \mathcal Q_\delta^{\boldsymbol\alpha}
       \mathrel{:=}\prod_{i=1}^k p^{m_i(\delta)}\mathbb Z_p^{n_i}.
\end{equation}
A $\delta$-rectangle is a translate of
$\mathcal Q_\delta^{\boldsymbol\alpha}$. Its $i$th block is a ball
of radius $\delta^{\alpha_i}$. For $A\subset\prod_i\mathbb Q_p^{n_i}$, let
$N_\delta^{\boldsymbol\alpha}(A)$ denote the number of these cosets
that meet $A$, equivalently this is the least number of such rectangles
covering $A$. This number is finite for bounded $A$. It is $0$ for
the empty set and $\infty$ for an unbounded set. At a discrete scale
$\delta=p^{-\ell}$, one has $m_i(\delta)=\lceil\alpha_i\ell\rceil$.

<!-- theorem-9: body_md -->
\label{thm:block-anisotropic-covering}
For the family \eqref{eq:block-evaluation-family}, fix
$\boldsymbol\alpha\in[0,\infty)^k$, and let
$\mathcal E\subset\mathcal V$ be nonempty with bounded joint absorbed
coefficient set $\mathcal A$. For a compact parameter ball
$B\subset\mathbb Q_p$ of positive radius, put
\begin{equation*}
 M_{\delta,B}\mathrel{:=}
          \max_{s\in B}N_\delta^{\boldsymbol\alpha}
                              (\widetilde P_s(\mathcal E)).
\end{equation*}
This maximum exists and is finite. There is a constant $C$ such that,
for every $\ell\geq1$, $\delta=p^{-\ell}$, and $0<\eta<1$,
\begin{equation}\label{eq:block-anisotropic-maximal-tail}
 \mu_B\left\{r\in B:
       N_\delta^{\boldsymbol\alpha}(\widetilde P_r(\mathcal E))
                      <\eta M_{\delta,B}\right\}
 \leq C(1+\ell)^{(\Lambda+1)/\Lambda}\eta^{1/\Lambda}.
\end{equation}
In particular, the same bound holds with $M_{\delta,B}$ replaced by
$N_\delta^{\boldsymbol\alpha}(\widetilde P_{r_0}(\mathcal E))$,
uniformly for all $r_0\in B$. The constant $C$ depends only on
$p,(n_i),(d_i),\boldsymbol\alpha,B$, and the fixed bound for
$\mathcal A$, not on $\ell,\eta,r_0$, or the particular coefficient
set within that bound.

<!-- lemma-9: body_md -->
\label{lem:block-finite-ring-comparison}
Let $\nu\geq1$, $\mathcal R\mathrel{:=}\mathcal R_\nu$, and $q\mathrel{:=} p^\nu$.
Fix integers $k\geq1$, output dimensions $n_i\geq1$, and degree
bounds $h_i\geq1$. Set
$\Lambda=\sum_i n_i h_i$ and $L=\sum_i n_i(h_i-1)$.
For a nonempty set
\begin{equation*}
 F\subset\prod_{i=1}^k(\mathcal R^{n_i})^{h_i+1},
 \qquad
 \mathsf P_r(b)\mathrel{:=}
       \left(\sum_{j=0}^{h_i}b_{i,j}r^j\right)_{i=1}^k,
\end{equation*}
one has, for every $r_0\in\mathcal R$ and $0<\eta<1$,
\begin{equation}\label{eq:block-finite-ring-comparison}
 \frac1q\#\{r\in\mathcal R:
          \#\mathsf P_r(F)<\eta\,\#\mathsf P_{r_0}(F)\}
 \leq C_{p,\Lambda}(1+\nu)^{(\Lambda+1)/\Lambda}
                   \eta^{1/\Lambda}.
\end{equation}
The same bound holds with $\#\mathsf P_{r_0}(F)$ replaced by
$\max_{u\in\mathcal R}\#\mathsf P_u(F)$. No nonvanishing
assumption is imposed on the coefficient vectors. The constant $C$ is independent of
$\nu,F,r_0$, and $\eta$.

We first prove the lemma for $r_0=0$ and unit parameters.
Define
\begin{equation*}
 K_0\mathrel{:=}\#\mathsf P_0(F),\qquad
 T\mathrel{:=}\{r\in\mathcal R^\times:
                         \#\mathsf P_r(F)<\eta K_0\},
 \qquad U\mathrel{:=} T^{-1}.
\end{equation*}
If $T$ is empty, the unit-parameter bound is immediate.
Assume that $T$ is nonempty. For $t\in U$, define
\begin{equation*}
 A_t\mathrel{:=}
    \operatorname{diag}(t^{h_i}I_{n_i})_{i=1}^k\,
                  \mathsf P_{t^{-1}}(F).
\end{equation*}
Every $t\in U$ is a unit, so $\#A_t=\#\mathsf P_{t^{-1}}(F)<\eta K_0$.
Use the finite-ring version of \eqref{eq:block-slice-map}, namely
\begin{equation*}
 (R_tz)_i\mathrel{:=}
       \sum_{j=0}^{h_i-1}t^{h_i-1-j}z_{i,j},
 \qquad z\in\mathcal R^\Lambda,
\end{equation*}
and form the incidence set
\begin{equation*}
 X\mathrel{:=}\{(t,z)\in U\times\mathcal R^\Lambda:R_tz\in A_t\}.
\end{equation*}
We estimate $\#X$ in two ways. First, the coefficient of
$z_{i,h_i-1}$ in each block is $1$. Hence $R_t$ is surjective
and every fiber contains exactly $q^L$ points. It follows that
\begin{equation}\label{eq:block-finite-incidence-upper}
 \#X=q^L\sum_{t\in U}\#A_t
       \leq\eta K_0q^L\#T.
\end{equation}

For the lower bound, we construct lines meeting $X$ in many
points. For each $a=(b_{i,0})_i\in\mathsf P_0(F)$, exists $b\in F$ such that $P_0(b)=a$.
For every choice of auxiliary vectors
$c_{i,j}\in\mathcal R^{n_i}$, $1\leq j<h_i$, set
$c_{i,0}\mathrel{:=} b_{i,0}$ and $c_{i,h_i}\mathrel{:=}0$, and consider the line
\begin{equation*}
 \mathcal L_{b,c}(t)\mathrel{:=}
 \left(t,
       (b_{i,j+1}-c_{i,j+1}+tc_{i,j})_{i,\,0\leq j<h_i}\right),
 \qquad t\in\mathcal R.
\end{equation*}
Write $z(t)$ for the second component. The same calculation
as in \eqref{eq:block-telescoping}, now over $\mathcal R$, gives
\begin{align*}
 (R_tz(t))_i
 &=\sum_{j=0}^{h_i-1}t^{h_i-1-j}
             (b_{i,j+1}-c_{i,j+1}+tc_{i,j})\notag\\
 &=\sum_{j=0}^{h_i}t^{h_i-j}b_{i,j}
  =t^{h_i}\mathsf P^i_{t^{-1}}(b),\qquad t\in U.
\end{align*}
Hence each such line meets $X$ in at least $\#T$ points. Its direction
has the primitive representative
\[
 (1,(c_{i,j})_{i,\,0\leq j<h_i}).

The constant slope blocks range over $\mathsf P_0(F)$ and
the free slope blocks range over $\mathcal R^L$. Since the
first coordinate is $1$, distinct such tuples represent distinct
projective directions. Thus the number of directions is exactly
$K_0q^L$.

Dhar's set estimate \cite[Theorem~1.8]{DharSet}, in ambient dimension
$\Lambda+1$. It gives
\begin{equation}\label{eq:block-finite-incidence-lower}
 \#X\geq c_{p,\Lambda}(1+\nu)^{-(\Lambda+1)}
        \frac{K_0q^L(\#T)^{\Lambda+1}}{q^\Lambda}.
\end{equation}

\eqref{eq:block-finite-incidence-lower}, and cancelling
$K_0q^L\#T>0$, yields
\begin{equation*}
 \left(\frac{\#T}{q}\right)^\Lambda
       \leq C_{p,\Lambda}(1+\nu)^{\Lambda+1}\eta.
\end{equation*}
Taking the $\Lambda$th root proves the unit-parameter estimate.

\begin{equation*}
 F^{(a)}\mathrel{:=}

\end{equation*}

\begin{equation*}
 T_a\mathrel{:=}\{u\in\mathcal R^\times:

\end{equation*}

$p^a\mathcal R^\times$, and each fiber has size $p^a$.

\begin{equation*}
 \mathcal B\mathrel{:=}
 \{r\in\mathcal R:\#\mathsf P_r(F)<\eta K_0\}.
\end{equation*}
Then
\begin{equation*}
    \begin{aligned}
 \frac{\#\mathcal B}{q}
 &=\sum_{a=0}^{\nu-1}p^{-a}\frac{\#T_a}{q}\\
 &\leq C_{p,\Lambda}(1+\nu)^{(\Lambda+1)/\Lambda}
       \eta^{1/\Lambda}\sum_{a=0}^{\nu-1}p^{-a}\\
 &\leq \frac{C_{p,\Lambda}}{1-p^{-1}}
       (1+\nu)^{(\Lambda+1)/\Lambda}\eta^{1/\Lambda}.
\end{aligned}
\end{equation*}

$r_0\in\mathcal R$ exists. Apply the estimate at that reference
to obtain the maximal assertion. This completes the proof.

<!-- corollary-4: body_md -->
\label{cor:block-anisotropic-common-reference}
Under the hypotheses of Theorem~\ref{thm:block-anisotropic-covering},
there is a Haar-full set $G\subset\mathbb Q_p$ such that, for every $r\in G$,
every compact parameter ball $B\subset\mathbb Q_p$ of positive radius, and
every $\varepsilon>0$, there is
$\delta_0=\delta_0(r,B,\varepsilon)>0$ for which
\begin{equation}\label{eq:block-anisotropic-compact-uniform}
 N_\delta^{\boldsymbol\alpha}(\widetilde P_r(\mathcal E))
       \geq\delta^\varepsilon
          \max_{s\in B}N_\delta^{\boldsymbol\alpha}
                              (\widetilde P_s(\mathcal E)),
 \qquad 0<\delta<\delta_0.
\end{equation}
Here $\delta$ may be any real scale, and $r$ need not belong to $B$.
In particular, the same set $G$ works simultaneously for every
$r_0\in\mathbb Q_p$ and every $\varepsilon>0$ in the comparison
\begin{equation}\label{eq:block-anisotropic-arbitrary-reference}
 N_\delta^{\boldsymbol\alpha}(\widetilde P_r(\mathcal E))
       \geq\delta^\varepsilon
             N_\delta^{\boldsymbol\alpha}
                              (\widetilde P_{r_0}(\mathcal E))
 \quad\text{for all sufficiently small }\delta.
\end{equation}
The full-measure set may depend on the fixed coefficient set and
$\boldsymbol\alpha$, but not on the reference parameter or
$\varepsilon$. On each fixed compact reference ball, the scale threshold
is uniform in the reference parameter.

We first prove the conclusion at the discrete scales. For each
integer $A\geq0$, let $B_A\mathrel{:=} p^{-A}\mathbb Z_p$. For fixed $A\geq0$ and $\varepsilon>0$, define the bad sets
\begin{equation*}
 \mathcal B_\ell\mathrel{:=}
 \left\{r\in B_A:
 N_{p^{-\ell}}^{\boldsymbol\alpha}
       (\widetilde P_r(\mathcal E))
 <p^{-\ell\varepsilon}M_{p^{-\ell},B_A}\right\}.
\end{equation*}
By \eqref{eq:block-anisotropic-power-tail},
\begin{equation*}
 \sum_{\ell=1}^{\infty}\mu_{B_A}(\mathcal B_\ell)
 \leq C\sum_{\ell=1}^{\infty}
 (1+\ell)^{(\Lambda+1)/\Lambda}
 p^{-\ell\varepsilon/\Lambda}<\infty.
\end{equation*}
The Borel--Cantelli lemma therefore gives
$\mu_{B_A}(\limsup_{\ell\to\infty}\mathcal B_\ell)=0$.
Thus almost every $r\in B_A$ belongs to only finitely many
$\mathcal B_\ell$. Equivalently, for such $r$ there exists
$\ell_0=\ell_0(r,A,\varepsilon)$ such that
\begin{equation*}
 N_{p^{-\ell}}^{\boldsymbol\alpha}
       (\widetilde P_r(\mathcal E))
 \geq p^{-\ell\varepsilon}M_{p^{-\ell},B_A}
 \qquad\text{for every }\ell\geq\ell_0.
\end{equation*}
Remove the exceptional null sets for all
$A\geq0$ and all $\varepsilon=1/j$, $j\geq1$, and denote the resulting
Haar-full subset of $\mathbb Q_p$ by $G$. For $0<\delta<1$, a comparison
with exponent $1/j<\varepsilon$ implies the one with exponent
$\varepsilon$. Thus $G$ works for every positive exponent.

Given $r\in G$ and a compact parameter ball $B$, choose $A$ so large
that $\{r\}\cup B\subset B_A$. The maximal comparison on $B_A$
implies the one on $B$, with a scale threshold independent of the
reference point in $B$. This proves
\eqref{eq:block-anisotropic-compact-uniform} at discrete scales.
Observe that we have removed only countably many null sets.
The maximum over references was taken before applying
Borel--Cantelli, so no uncountable intersection of full-measure
sets is involved.

It remains to pass from the discrete scales to all real scales.

\begin{equation*}
 C_{\boldsymbol\alpha}\mathrel{:=}
             p^{\sum_i n_i\lceil\alpha_i\rceil}.
\end{equation*}

from \eqref{eq:block-rectangle-lattice} and their indices give,

\begin{equation*}
 N_{p^{-\ell}}^{\boldsymbol\alpha}(Y)
 \leq N_\delta^{\boldsymbol\alpha}(Y)
 \leq N_{p^{-(\ell+1)}}^{\boldsymbol\alpha}(Y)
 \leq C_{\boldsymbol\alpha}
                         N_{p^{-\ell}}^{\boldsymbol\alpha}(Y).
\end{equation*}

       \geq\delta^\varepsilon M_{\delta,B}
\end{equation*}
for all sufficiently large $\ell$.

The maximal comparison on $B$ implies
\eqref{eq:block-anisotropic-arbitrary-reference}. This completes
the proof.

<!-- remark-6: body_md -->
\label{rem:block-reference-threshold}
The exceptional null set is independent of the reference.
However, the scale threshold need not be uniform over all of
$\mathbb Q_p$. To see this, take
$P_s(a,b)=a+sb$ and $\mathcal E=\{(0,b):b\in\mathbb Z_p\}$.
The isotropic covering numbers satisfy
\begin{equation*}
 P_s(\mathcal E)=s\mathbb Z_p,
 \qquad N_{p^{-\ell}}(P_{p^{-a}}(\mathcal E))=p^{a+\ell}
                       \quad(a\geq0,\ \ell\geq1).
\end{equation*}
At each fixed scale the reference covering numbers are unbounded as
$a\to\infty$, whereas each fixed evaluation image has a finite
covering number. Thus the compact reference range in
\eqref{eq:block-anisotropic-compact-uniform} cannot generally be
replaced by all of $\mathbb Q_p$.
