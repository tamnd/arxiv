<!-- remark-1: body_md -->
\label{Ainftystr}
		If $\mu^d_{\cal{A}}=0$ for $d \geq 3$ then ${\cal{A}}$ can be equipped with a canonical DG-category structure (in the sense of \cite{KelDG}). Let $f:Y \to Z$ and $g:X \to Y$ be homogeneous morphisms, the differential and composition are defined as follows.
 		\[\partial(f) = (-1)^{|f|} \mu^1_{\cal{A}}(f)\]
 		\[f \circ g = (-1)^{|g|} \mu^2_{\cal{A}}(f,g)\]
 		With this construction, we can also equip any $k$-linear category or DG-category with an $A_\infty$-structure such that $\mu^d = 0$ for $d \geq 3$.
<!-- end remark-1 -->
<!-- definition-1: body_md -->
Let ${\cal{A}}$ be an $A_\infty$-category. The $A_\infty$-category $\mathbb{Z} {\cal{A}}$ is defined as follows.
		\begin{itemize}
			\item Objects of $\mathbb{Z} {\cal{A}}$ are formal finite direct sums $\displaystyle\bigoplus_{i \in I} X_i[p_i]$ where for all $i \in I$, $p_i$ is an integer.
			\item If $X$, $Y$ are objects of ${\cal{A}}$ and $p$, $q$ are integers then 
			\[\textup{Hom}_{\mathbb{Z} {\cal{A}}}(X[p], Y[q]):= \textup{Hom}_{\textup{gr} \, k}(k[p],k[q]) \otimes_k \textup{Hom}_{\cal{A}}(X,Y)\]
			where $\textup{gr} \, k$ is the $\mathbb{Z}$-graded category of $\mathbb{Z}$-graded $k$-modules.
			The degree $i$ component $\textup{Hom}^i(X[p],Y[q])$ is generated by morphisms of the form $s^p_q \otimes a$ where $s^p_q$ is the identity map from $k[p]$ to $k[q]$ (it has degree $p-q$). The morphism $s^p_q \otimes a$ has degree $p-q + |a|$.
			\item The identity map of $X[p]$ is $s^p_p \otimes \textup{id}_X$.
			\item Consider $\mu^d_{\mathbb{Z} {\cal{A}}}$ defined by 
			\[\mu_{\mathbb{Z} {\cal{A}}}^d(s^{p_{d-1}}_{p_d} \otimes a_d, \dots, s^{p_1}_{p_2} \otimes a_1):= (-1)^{\displaystyle\sum_{i < j} (p_{i-1} -p_i).\|a_j\|} s^{p_1}_{p_d} \otimes \mu_{\cal{A}}^d(a_d,\dots, a_1)\]
			on composable homogeneous morphisms. This extends to an $A_\infty$-structure on $\mathbb{Z} {\cal{A}}$ by linearity.
		\end{itemize}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Let ${\cal{A}}$ be an $A_\infty$-category. The $A_\infty$-category $\textup{Tw}({\cal{A}})$ is defined as follows.
		\begin{itemize}
			\item Objects of $\textup{Tw}({\cal{A}})$ are pairs $(X,\delta_X)$ where $X$ is an object of $\mathbb{Z} {\cal{A}}$ and $\delta_X: X \to X$ is a degree 1 endomorphism of $X$ in $\mathbb{Z} {\cal{A}}$ that is strictly lower triangular (for a detailled definition of strictly lower triangular, see \cite[Section 3l]{Sei}) and such that
			\begin{equation} 
			\sum_{d>0} \mu_{\mathbb{Z} {\cal{A}}}^d(\delta_X,\dots, \delta_X) = 0
			\end{equation}
			This equation is called the generalized Maurer-Cartan equation. We call objects of $\textup{Tw}({\cal{A}})$ twisted complexes over ${\cal{A}}$.
			\item If $(X,\delta_X)$ and $(Y,\delta_Y)$ are twisted complexes over ${\cal{A}}$, set
			\[\textup{Hom}_{\textup{Tw}({\cal{A}})}((X,\delta_X),(Y,\delta_Y)):= \textup{Hom}_{\mathbb{Z} {\cal{A}}}(X,Y)\]
			\item Consider the structure maps $\mu^d_{\textup{Tw}({\cal{A}})}$ given by
			\[\forall d >0, \;\mu_{\textup{Tw}({\cal{A}})}^d(f_d,\dots,f_1):= \sum_{i_0,\dots,i_d \geq 0} \mu_{\mathbb{Z} {\cal{A}}}^{d+i_0 + \dots + i_d}(\delta_{X_d}^{\otimes i_d},f_d,\delta_{X_{d-1}}^{\otimes i_{d-1}},a_{d-1},\dots,a_1,\delta_{X_0}^{\otimes i_0})\]
			on composable homogeneous morphisms. This defines an $A_\infty$-structure on $\textup{Tw}({\cal{A}})$ by linearity.
		\end{itemize}

<!-- proposition-2: body_md -->
\label{sameorbit}
		Let ${\cal{A}}$ be a DG-category and $F: {\cal{A}} \to {\cal{A}}$ an invertible DG-functor. Then $F$ defines a strict $\mathbb{Z}$ action on ${\cal{A}}$ (with the $A_\infty$-structure given in Remark \ref{Ainftystr}). There is a isomorphism of $A_\infty$-categories (which is also an isomorphism of DG-categories)
		\[{\cal{F}}: {\cal{A}} *_F \mathbb{Z} \overset{\sim}{\to} {\cal{A}} / F.\]
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
As ${\cal{A}} *_F \mathbb{Z}$ and ${\cal{A}}/F$ have the same objects, define ${\cal{F}}$ to be the identity on objects. Let $X$ and $Y$ be objects of ${\cal{A}}$, a morphism $f: \{X\} \to \{Y\}$ in ${\cal{A}} *_F \mathbb{Z}$ decomposes as $f = \displaystyle\sum_{\ell \in \mathbb{Z}} a_\ell \otimes \sigma^\ell$ with $a_\ell: F^\ell X \to Y$. We then set
		\[{\cal{F}}(f):= \sum_{\ell \in \mathbb{Z}} F^{\ell}(a_{-\ell}) \in \bigoplus_{\ell \in \mathbb{Z}} \textup{Hom}_{\cal{A}}(X,F^\ell Y)\]
		To show that it is a strict $A_\infty$-functor, it is sufficient to show that it is compatible with $\mu^1_{{\cal{A}} * \mathbb{Z}}$ and $\mu^2_{{\cal{A}} * \mathbb{Z}}$ as $\mu^d_{{\cal{A}} * \mathbb{Z}}=0$ for $d > 2$. Let $\partial$ be the differential in ${\cal{A}}$ (from its DG-category structure).
		\begin{align*}
			{\cal{F}}(\mu^1_{{\cal{A}} * \mathbb{Z}}(a \otimes \sigma^\ell)) &= {\cal{F}}(\mu^1_{\cal{A}}(a) \otimes \sigma^\ell) \\
			&= F^{-\ell}(\mu^1_{\cal{A}}(a)) \\
			&= F^{-\ell}((-1)^{|a|}\partial(a)) \\
			&= (-1)^{|a|}\partial(F^{-\ell}(a))\\
			&= (-1)^{|a| + |F^{-\ell}(a)|} \mu^1_{{\cal{A}}/F}({\cal{F}}(a \otimes \sigma^\ell)) \\
			&= \mu^1_{{\cal{A}}/F}({\cal{F}}(a \otimes \sigma^\ell))
		\end{align*}
		
		\noindent Similarly, we have
		
		\begin{align*}
			{\cal{F}}(\mu^2_{{\cal{A}} * \mathbb{Z}}(a \otimes \sigma^\ell, b \otimes \sigma^k)) &= {\cal{F}}(\mu^2_{\cal{A}}(a,F^\ell(b)) \otimes \sigma^{\ell+k}) \\
			&= F^{-\ell-k}(\mu^2_{\cal{A}}(a,F^{\ell}(b))) \\
			&= (-1)^{|F^\ell(b)|} F^{-\ell-k}(a \circ F^{\ell}(b)) \\
			&= (-1)^{|b|} F^{-\ell-k}(a) \circ F^{-k}(b)) \\
			&= (-1)^{|b|+|F^{-k}(b)|}\mu^2_{{\cal{A}}/F}(F^{-\ell}(a),F^{-k}(b)) \\
			&= \mu^2_{{\cal{A}}/F}({\cal{F}}(a \otimes \sigma^\ell),{\cal{F}}(b \otimes \sigma^k))
		\end{align*}
		
		\noindent Thus ${\cal{F}}$ is a strict $A_\infty$-functor. 
		
		\noindent Furthermore, we claim that it is an isomorphism. Its inverse is the identity on the objects and sends $a: X \to F^\ell(Y)$ in $\textup{Hom}_{{\cal{A}}/F}(X,Y)$ to $F^{-\ell}(a) \otimes \sigma^{-\ell}$. It is straightforward to check that this also defines a strict $A_\infty$-functor.

		\[\textup{Hom}_{\textup{Tw}({\cal{A}}) * \mathbb{Z}}((X,\delta_X),(Y,\delta_Y)) = \bigoplus_{\ell \in \mathbb{Z}} \textup{Hom}_{\mathbb{Z} {\cal{A}}}(X[\ell],Y)\]

			\item If $s^{p+\ell}_q \otimes a \otimes \sigma^\ell: \{X[p]\} \to \{Y[q]\}$ is a morphism in $\mathbb{Z} {\cal{A}} * \mathbb{Z}$, set

			&= (-1)^{-p_j(p_i-p_j)-\ell + |a^\ell_{i,j}|} s^{p_i}_0 \otimes t^{\ell-p_j} \otimes a^\ell_{i,j}
		\end{align*}
		
		\noindent Computing the second term of the sum we have
		
		\begin{align*}
			& \mu^2_{\mathbb{Z}(k[t,t^{-1}] \otimes_k {\cal{A}})}((-1)^{(p_i+1)\|a^\ell_{i,j}\| -p_j(p_i-p_j)-\ell}s^0_0 \otimes t^{- \|a^\ell_{i,j}\|} \otimes a^\ell_{i,j},s^{p_i}_0 \otimes t^{-p_i} \otimes \textup{id}_{X_i}) \\
			&= (-1)^{(p_i+1)\|a^\ell_{i,j}\| -p_j(p_i-p_j)-\ell}s^{p_i}_0 \otimes \mu^2_{k[t,t^{-1}] \otimes_k {\cal{A}}}(t^{- \|a^\ell_{i,j}\|} \otimes a^\ell_{i,j},t^{-p_i} \otimes \textup{id}_{X_i}) \\
			&= (-1)^{\|a^\ell_{i,j}\|  -p_j(p_i-p_j)-\ell} s^{p_i}_0 \otimes t^{-\|a^\ell_{i,j}\| - p_i} \otimes a^\ell_{i,j}
		\end{align*}
		As $p_i - p_j +\ell + \|a^\ell_{i,j}\| = 0$, we have the wanted relation, and $\varphi_X$ is an morphism in $Z^0(\textup{Tw}(k[t,t^{-1}] \otimes_k {\cal{A}}))$. It is an isomorphism with inverse $\varphi_X^{-1}:= \displaystyle\sum_{i \in I} s^0_{p_i} \otimes t^{p_i} \otimes \textup{id}_{X_i}$.
		
		Now define the retraction $r$ of $Z^0(\iota)$ as follows:
		\begin{itemize}
			\item if $X$ is an object of $Z^0(\textup{Tw}(k[t,t^{-1}] \otimes_k {\cal{A}}))$ then $rX$ is defined as above;
			\item if $f:X \to Y$ is a morphism in $Z^0(\textup{Tw}(k[t,t^{-1}] \otimes_k {\cal{A}}))$ then $r(f):= \varphi_Y \circ f\circ\varphi_X^{-1}$
		\end{itemize}
		By construction, $r:Z^0(\textup{Tw}(k[t,t^{-1}] \otimes_k {\cal{A}})) \to Z^0({\cal{S}})$ is a functor. As $\varphi_X = \textup{id}_X$ for $X$ in ${\cal{S}}$, $r$ is a retraction of $Z^0(\iota)$. 
		
		\noindent As $Z^0(\iota)$ is an equivalence of categories, so is $H^0(\iota)$.

\label{difmod}
		Let $A$ be a $\mathbb{Z}$-graded algebra.
		We define the category of differential $A$-modules as follows.
		\begin{itemize}
			\item Objects are pairs of the form $(M,d_M)$ where $M$ is an (ungraded) $A$-module and $d_M: M \to M$ is a $k$-linear map such that $d_M^2 = 0$ and
			\[\forall m \in M, \; \forall a \in A, \; d_M(m.a) = (-1)^{|a|} d_M(m).a\]
			\item A morphism $f: (M,d_M) \to (N,d_N)$ is the data of a morphism $f:M \to N$ in $\textup{mod} \, A$ such that $f \circ d_M = d_N \circ f$ in $\textup{mod} \, k$.
		\end{itemize}

\label{equivDif}

			\item If $M$ is an $A^\ltimes$-module then $\Gamma(M) = (M_{|A},d_M)$ where $M_{|A}$ is the restriction of $M$ to $\textup{mod} \, A$ and $d_M$ is given by the action of $1\otimes \varepsilon$ on $M$. 
			\item If $f: M \to N$ is a morphism in $\textup{mod} \, A^\ltimes$, define $\Gamma(f): (M_{|A},d_M) \to (N_{|A},d_N)$ to be the restriction $f_{|A}$ of $f$ to $\textup{mod} \, A$.
		\end{itemize}

We first check that for $M$ in $\textup{mod} \, A^\ltimes$, $\Gamma(M)$ is indeed a differential $A$-module. It is sufficient to check that for all $m$ in $M$, $a$ in $A$ we have
		\[(m.1\otimes \varepsilon).a = (-1)^{|a|} (m.a).1 \otimes \varepsilon\] 
		This is true from the Koszul sign rule on $A^\ltimes$. 
		Furthermore, if $f: M \to N$ is a morphism in $\textup{mod} \, A^\ltimes$, then $f_{|A} \circ d_M = d_N \circ f_{|A}$ follows from the definition of morphisms in $\textup{mod} \, A^\ltimes$. 	
		
		\noindent We now claim that $\Gamma$ is an equivalence of categories, with inverse functor defined as follows.

			\item A differential module $(M,d_M)$ is sent to the $A^\ltimes$-module with underlying vector space $M$ (seen as a $k$-vector space) and $A^\ltimes$ action given by
			\[\forall m \in M, \; \forall a \in A, \; m.(a \otimes \varepsilon^\ell) = (m.(a \otimes 1)).(1 \otimes \varepsilon^\ell) = d_M^\ell(m.a)\]
			\item If $f:(M,d_M) \to (N,d_N)$ is a morphism in $\textup{Dif}(A)$ then the underlying morphism of $A$-modules $f: M \to N$ defines a morphism of $A^\ltimes$ modules thanks to the relation $f \circ d_M = d_N \circ f$.
		\end{itemize}
		It is straightforward to see that this defines a functor that is inverse to $\Gamma$.
		
		The fact that $\Gamma$ restricts to an equivalence between $\textup{CM}(A^\ltimes)$ and ${\cal{P}}(A)$ follows from Proposition~\ref{CMproj}.

<!-- proposition-9: body_md -->
\label{exactequi}
		There is an equivalence of exact categories $Z^0({\cal{S}}) \simeq {\cal{P}}(A)$.
<!-- end proposition-9 -->
<!-- proof-9: body_md -->
We construct an equivalence ${\cal{G}}: Z^0({\cal{S}}) \to {\cal{P}}(A)$ as follows:
		\begin{itemize}
			\item If $X = (\displaystyle\bigoplus_{i \in I} X_i,\delta_X = \sum_{i,j \in I} t^{- \|a_{i,j}\|} \otimes a_{i,j})$ is an object of $Z^0({\cal{S}})$, define ${\cal{G}}(X)$ to be the pair $(\displaystyle\bigoplus_{i \in I} X_i,d_X = \sum_{i,j \in I} (-1)^{\mathfrak{s}(\|a_{i,j}\|)} \overline{a_{i,j}})$ where $\mathfrak{s}(x) = \frac{x(x+1)}{2}$ and $\overline{a_{i,j}}: X_i \to X_j$ is defined by $\overline{a_{i,j}}(m) = (-1)^{|m|} a_{i,j}(m)$. Remark that $\mathfrak{s}(x+y) = \mathfrak{s}(x) + \mathfrak{s}(y) + xy$ and that if $a_{i,j}$ and $a_{j,k}$ are homogeneous then $\overline{a_{j,k}} \circ \overline{a_{i,j}} = (-1)^{|a_{i,j}|} a_{j,k} \circ a_{i,j}$ (this defines an involution).
			\item If $f = \displaystyle\sum_{i \in I} \sum_{j \in J} t^{-|f_{i,j}|} \otimes f_{i,j}$ is a morphism between $X$ and $Y$ (using the notations above), set ${\cal{G}}(f) = \displaystyle\sum_{i \in I} \sum_{j \in J} (-1)^{\mathfrak{s}(|f_{i,j}|)}f_{i,j}$.
		\end{itemize}	
		We now check that ${\cal{G}}$ is a functor.
		
		First we check that if $X$ an object of $Z^0({\cal{S}})$ with notations above, ${\cal{G}}(X)$ is indeed an object of ${\cal{P}}(A)$. To do so, it is sufficient to check that $d_X^2 = 0$. The generalized Maurer-Cartan equations gives us $\mu^2_{k[t,t^{-1}] \otimes_k {\cal{A}}}(\delta_X,\delta_X)= 0$. Thus we have
		\[\sum_{i,j,k \in I} \mu^2_{k[t,t^{-1}] \otimes_k {\cal{A}}}(t^{- \|a_{j,k}\|} \otimes a_{j,k},t^{- \|a_{i,j}\|} \otimes a_{i,j}) = 0\]
		i.e.
		\[\sum_{i,j,k \in I} (-1)^{-\|a_{i,j}\|.\|a_{j,k}\| + |a_{i,j}|} t^{-\|a_{i,j}\|-\|a_{j,k}\|} \otimes a_{j,k} \circ a_{i,j} = 0\] 
		By grouping terms in the sum, we have
		
		\[\sum_{\ell \in \mathbb{Z}} t^{-\ell} \otimes \sum_{\substack{i,j,k \in I \\ \|a_{i,j}\| + \|a_{j,k}\| = \ell}} (-1)^{-\|a_{i,j}\|.\|a_{j,k}\| + |a_{i,j}|} a_{j,k} \circ a_{i,j} = 0\] 
		Thus for all integers $l$, we have
		\[\sum_{\substack{i,j,k \in I \\ \|a_{i,j}\| + \|a_{j,k}\| = \ell}} (-1)^{-\|a_{i,j}\|.\|a_{j,k}\| + |a_{i,j}|} a_{j,k} \circ a_{i,j} = 0\]
		
		\noindent However, we have 
		\begin{align*}
			d_X^2 &= \sum_{i,j,k \in I} (-1)^{\mathfrak{s}(\|a_{i,j}\|) + \mathfrak{s}(\|a_{j,k}\|) + |a_{i,j}|} a_{j,k} \circ a_{i,j} \\
			&= \sum_{\ell \in \mathbb{Z}} (-1)^{\mathfrak{s}(\ell)} \sum_{\substack{i,j,k \in I \\ \|a_{i,j}\| + \|a_{j,k}\| = \ell}} (-1)^{-\|a_{i,j}\|.\|a_{j,k}\| + |a_{i,j}|}  a_{j,k} \circ a_{i,j} \\
			&= 0
		\end{align*}
		\noindent Thus ${\cal{G}}(X)$ is indeed an object of ${\cal{P}}(A)$.
		
		
		\noindent We now check that ${\cal{G}}$ respects composition. Consider morphisms of $Z^{0}({\cal{S}})$
		\[t^{-|f|} \otimes f: (Y = \bigoplus_{i \in J} Y_i,\delta_Y = \sum_{i,j \in J} t^{- \|b_{i,j}\|} \otimes b_{i,j}) \to (Z = \bigoplus_{i \in K} Z_i,\delta_Z = \sum_{i,j \in K} t^{- \|c_{i,j}\|} \otimes c_{i,j})\]
		 and
		\[t^{-|g|} \otimes g: (X = \bigoplus_{i \in I} X_i,\delta_X = \sum_{i,j \in I} t^{- \|a_{i,j}\|} \otimes a_{i,j}) \to (Y = \bigoplus_{i \in J} Y_i,\delta_Y = \sum_{i,j \in J} t^{- \|b_{i,j}\|} \otimes b_{i,j})\]
		We have
		
		\begin{align*}
			{\cal{G}}(t^{-|f|} \otimes f \circ t^{-|g|} \otimes g) &= {\cal{G}}(\mu^2_{k[t,t^{-1}]\otimes_k {\cal{A}}}(t^{-|f|} \otimes f,t^{-|g|} \otimes g)) \\
			&= {\cal{G}}((-1)^{-|g|.\|f\|}t^{-|f| - |g|} \otimes \mu^2_{\cal{A}}(f,g)) \\
			&= (-1)^{-|g|.\|f\| + |g|} {\cal{G}}(t^{-|f| - |g|} \otimes f \circ g) \\
			&= (-1)^{|g|.|f| + \mathfrak{s}(-|f|-|g|)} f \circ g \\
			&= (-1)^{|g|.|f|+ \mathfrak{s}(-|f|-|g|) - \mathfrak{s}(-|f|) - \mathfrak{s}(-|g|)} {\cal{G}}(t^{-|f|} \otimes f) \circ {\cal{G}}(t^{-|g|} \otimes g) \\
			&= {\cal{G}}(t^{-|f|} \otimes f) \circ {\cal{G}}(t^{-|g|} \otimes g)
		\end{align*} 
		
		\noindent Consider a morphism $t^{-|f|} \otimes f: (X, \delta_X) \to (Y,\delta_Y)$ where $(X,\delta_X)$ and $(Y,\delta_Y)$ are as above and $f: X_k \to Y_\ell$ is a homogeneous morphism in ${\cal{A}}$. Remark also that if $a$ and $b$ are homogeneous morphisms of ${\cal{A}}$ then $a \circ \overline{b} = \overline{a \circ b}$ and $\overline{a} \circ b = (-1)^{|b|} \overline{a \circ b}$. We now have
		\begin{align*}
			{\cal{G}}(f) &\circ d_X - d_Y \circ {\cal{G}}(f) = \sum_{\substack{i \in I \\ j \in J}} (-1)^{\mathfrak{s}(|f|) + \mathfrak{s}(\|a_{i,k}\|)} f \circ \overline{a_{i,k}} - (-1)^{\mathfrak{s}(|b_{\ell,j}|) + \mathfrak{s}(|f|)} \overline{b_{\ell,j}} \circ f \\
			&= \sum_{\substack{i \in I \\ j \in J}} (-1)^{\mathfrak{s}(|f|) + \mathfrak{s}(\|a_{i,k}\|)} \overline{f \circ a_{i,k}} - (-1)^{\mathfrak{s}(\|b_{\ell,j}\|) + \mathfrak{s}(|f|) + |f|} \overline{b_{\ell,j} \circ f} \\
			&= \sum_{r \in \mathbb{Z}} \left( \sum_{\substack{i \in I \\ j \in J \\ \|a_{i,k}\|+|f| = r}} (-1)^{\mathfrak{s}(|f|) + \mathfrak{s}(\|a_{i,k}\|)} \overline{f \circ a_{i,k}} - \sum_{\substack{i \in I \\ j \in J \\ \|b_{\ell,j}\|+|f| = r}} (-1)^{\mathfrak{s}(\|b_{\ell,j}\|) + \mathfrak{s}(|f|) + |f|} \overline{b_{\ell,j} \circ f} \right)
		\end{align*}
		However, $\mu^1_{{\cal{S}}}(t^{-|f|} \otimes f) = 0$. Therefore
		
		\begin{align*}
			\mu^1_{{\cal{S}}}(t^{-|f|} \otimes f) &= \mu^2_{k[t,t^{-1}] \otimes_k {\cal{A}}}(t^{-|f|} \otimes f,\delta_X) + \mu^2_{k[t,t^{-1}] \otimes_k {\cal{A}}}(\delta_Y,t^{-|f|} \otimes f) \\
			&= \sum_{\substack{i \in I \\ j \in J}} \mu^2_{k[t,t^{-1}] \otimes_k {\cal{A}}}(t^{-|f|} \otimes f,t^{- \|a_{i,k}\|} \otimes a_{i,k}) +  \mu^2_{k[t,t^{-1}] \otimes_k {\cal{A}}}(t^{- \|b_{\ell,j}\|} \otimes b_{\ell,j},t^{-|f|} \otimes f) \\
			&= \sum_{\substack{i \in I \\ j \in J}} (-1)^{\|a_{i,k}\|.\|f\| + |a_{i,k}|} t^{-|f|-\|a_{i,k}\|} \otimes f \circ a_{i,k} +(-1)^{|f|.\|b_{\ell,j}\| + |f|} t^{-\|b_{\ell,j}\|-|f|} \otimes b_{\ell,j} \circ f \\
			&= \sum_{r \in \mathbb{Z}} t^{-r} \otimes \left(\sum_{\substack{i \in I \\ j \in J \\ \|a_{i,k}\|+|f| = r}} (-1)^{\|a_{i,k}\|.|f| + 1} f \circ a_{i,k} + \sum_{\substack{i \in I \\ j \in J \\ \|b_{\ell,j}\|+|f| = r}} (-1)^{|f|.\|b_{\ell,j}\| + |f|}  b_{\ell,j} \circ f \right) \\
			&= \sum_{r \in \mathbb{Z}} (-1)^{\mathfrak{s}(r)} t^{-r} \otimes \left( \sum_{\substack{i \in I \\ j \in J \\ \|a_{i,k}\|+|f| = r}} (-1)^{\mathfrak{s}(\|a_{i,k}\|)+\mathfrak{s}(|f|) + 1} f \circ a_{i,k} \right) \\
			& + (-1)^{\mathfrak{s}(r)} t^{-r} \otimes \left(\sum_{\substack{i \in I \\ j \in J \\ \|b_{\ell,j}\|+|f| = r}} (-1)^{\mathfrak{s}(|f|)+\mathfrak{s}(\|b_{\ell,j}\|) + |f|}  b_{\ell,j} \circ f \right)  \\
			&= 0
		\end{align*}
		Thus for all integers $r$, we have 
		\[\left(\sum_{\substack{i \in I \\ j \in J \\ \|a_{i,k}\|+|f| = r}} (-1)^{\mathfrak{s}(\|a_{i,k}\|)+\mathfrak{s}(|f|) + 1} f \circ a_{i,k} + \sum_{\substack{i \in I \\ j \in J \\ \|b_{\ell,j}\|+|f| = r}} (-1)^{\mathfrak{s}(|f|)+\mathfrak{s}(\|b_{\ell,j}\|) + |f|}  b_{\ell,j} \circ f \right) = 0\]
		This gives us the wanted relation ${\cal{G}}(f) \circ d_X - d_Y \circ {\cal{G}}(f) = 0$. 
		
		We have now shown that ${\cal{G}}$ is indeed a functor. From its definition on objects, we deduce that it is essentially surjective: in fact if $X = \left(\displaystyle\bigoplus_{i \in I} P_i, d_X = \sum_{i,j \in I} a_{i,j}\right)$ is an object of ${\cal{P}}(A)$, we can construct an essential preimage $\left(\displaystyle\bigoplus_{i \in I} P_i, \delta_X = \sum_{i,j \in I} (-1)^{\mathfrak{s}(\|\overline{a_{i,j}}\|)}t^{-\|\overline{a_{i,j}}\|} \otimes \overline{a_{i,j}}\right)$. Similarly, we can construct an inverse map on morphism spaces to show that ${\cal{G}}$ is fully faithful.
		Using Remark \ref{exactstr} it is straightforward to see that ${\cal{G}}$ is exact.

<!-- proof-12: body_md -->
Let $d >0$ be an integer and $(s^{p_k + l_k}_{p_{k+1}} \otimes a_k \otimes \sigma^{l_k})_{1 \leq k \leq d}$ be a family of composable morphisms in $\textup{Tw}({\cal{A}}) * \mathbb{Z}$ where $s^{p_k + l_k}_{p_{k+1}} \otimes a_k \otimes \sigma^{l_k}: (X_{k-1},\delta_{k-1}) \to (X_k,\delta_k)$. As each $\delta_k$ is an endomorphism of $X_k$ in $\mathbb{Z} {\cal{A}}$, we can decompose them as $\delta_k = \displaystyle\sum_{j \in J_k} s^{q_{k,j}}_{r_{k,j}} \otimes a_{k,j}$. Let $l:= \displaystyle\sum_{i=1}^d l_i$, we now have
	
	\begin{align*}
		& \tilde{\nu}(\mu_{\textup{Tw}({\cal{A}}) * \mathbb{Z}}^d(s^{p_d + l_d}_{p_{d+1}} \otimes a_d \otimes \sigma^{l_d},\dots,s^{p_1 + l_1}_{p_{2}} \otimes a_1 \otimes \sigma^{l_1})) \\
		&= \tilde{\nu}((-1)^{\Delta_1} \mu_{\textup{Tw}({\cal{A}})}^d(s^{p_d + l_d}_{p_{d+1}} \otimes a_d,s^{p_{d-1} + l_d + l_{d-1}}_{p_d + l_d} \otimes a_{d-1},\dots,s^{p_1 + l}_{p_2 + l - l_1} \otimes a_1) \otimes \sigma^{l}) \\
		&= \tilde{\nu}((-1)^{\Delta_1} \sum_{k_0,\dots,k_d} \mu_{\mathbb{Z} {\cal{A}}}^{d + k_0 + \dots + k_d}(\delta_d,\dots,\delta_d,s^{p_d + l_d}_{p_{d+1}} \otimes a_d,\delta_{d-1}[l_d],\dots,s^{p_1+l}_{p_2 + l - l_1} \otimes a_1, \delta_1[l],\dots,\delta_1[l])\otimes \sigma^l) \\
		&= \tilde{\nu}((-1)^{\Delta_1} \sum_{k_0,\dots,k_d} \sum_{(j(x))_{1 \leq x \leq k} \in \displaystyle\prod_{i=0}^d J_i^{k_i}} (-1)^{\Delta_2(j)} \mu_{\mathbb{Z} {\cal{A}}}^{d + k_0 + \dots + k_d}(s^{q_{d,j(\sum_{i=0}^{d} k_i)}}_{r_{d,j(\sum_{i=0}^{d} k_i)}}\otimes a_{d,j(\sum_{i=0}^{d} k_i)},\dots,s^{q_{d,j(1 +\sum_{i=0}^{d-1} k_i)}}_{r_{d,j(1+\sum_{i=0}^{d-1} k_i)}} \\
		&\otimes a_{d,j(1+\sum_{i=0}^{d-1} k_i)},s^{p_d + l_d}_{p_{d+1}} \otimes a_d,\dots,s^{p_1+l}_{p_2 + l - l_1} \otimes a_1, s^{q_{1,j(k_0)}+l}_{r_{1,j(k_0)}+l} \otimes a_{1,j(k_0)},\dots,s^{q_{1,j(1)}+l}_{r_{1,j(1)}+l}\otimes a_{1,j(1)})\otimes \sigma^l) \\
		&= \tilde{\nu}((-1)^{\Delta_1} \sum_{k_0,\dots,k_d} \sum_{(j(x))_{1 \leq x \leq k} \in \displaystyle\prod_{i=0}^d J_i^{k_i}} (-1)^{\Delta_2(j) + \Delta_3(j)} s^{q_{1,j(1)}+l}_{r_{d,j(\sum_{i=0}^{d}k_i)}} \otimes \mu_{\cal{A}}^{d + k_0 + \dots + k_d}(a_{d,j(\sum_{i=0}^{d} k_i)},\dots, \\
		&a_{d,j(1+\sum_{i=0}^{d-1} k_i)},a_d,\dots, a_1, a_{1,j(k_0)},\dots,a_{1,j(1)})\otimes \sigma^l) \\
		&= (-1)^{\Delta_1} \sum_{k_0,\dots,k_d} \sum_{(j(x))_{1 \leq x \leq k} \in \displaystyle\prod_{i=0}^d J_i^{k_i}} (-1)^{\Delta_2(j) + \Delta_3(j)} s^{q_{1,j(1)}}_{r_{d,j(\sum_{i=0}^{d}k_i)}} \otimes s^l_0 \otimes \mu_{\cal{A}}^{d + k_0 + \dots + k_d}(a_{d,j(\sum_{i=0}^{d} k_i)},\dots, \\
		&a_{d,j(1+\sum_{i=0}^{d-1} k_i)},a_d,\dots, a_1, a_{1,j(k_0)},\dots,a_{1,j(1)})\otimes \sigma^l 
	\end{align*} 
	where 
	\[k = \sum_{i=0}^{d} k_i\]
	\[\textup{ind}(j(x)) = i \;\textup{if}\; j(x) \in J_i\]
	\[\Delta_1 = \sum_{i=1}^{d-1} (\sum_{k=i+1}^{d} l_k) (p_i + l_i - p_{i+1}) \]
	\[\Delta_2(j) = \sum_{x=1}^{k} \left( \sum_{m=\textup{ind}(j(x))+1}^{d} l_m\right) (q_{\textup{ind}(j(x)),j(x)}-r_{\textup{ind}(j(x)),j(x)})\]
	\begin{align*}\Delta_3(j) =& \sum_{x<y} (q_{\textup{ind}(j(x)),j(x)} - r_{\textup{ind}(j(x)),j(x)}) \|a_{\textup{ind}(j(y)),j(y)}\| \\
		&+ \sum_{x=1}^{k} (q_{\textup{ind}(j(x)),j(x)} - r_{\textup{ind}(j(x)),j(x)}) \left(\sum_{m=\textup{ind}(j(x))+1}^{d} \|a_m\| \right) \\
		& + \|a_{\textup{ind}(j(x)),j(x)}\|\left(\sum_{m=1}^{\textup{ind}(j(x))} (p_m +l_m-p_{m+1})\right) + \sum_{i<j} (p_i+l_i-p_{i+1})\|a_j\| 
	\end{align*}
	
	Conversely, we have
	
	\begin{align*}
		& \mu_{\textup{Tw}(\mathbb{Z} {\cal{A}} * \mathbb{Z})}^d(\tilde{\nu}(s^{p_d + l_d}_{p_{d+1}} \otimes a_d \otimes \sigma^{l_d}),\dots,\tilde{\nu}(s^{p_1 + l_1}_{p_2} \otimes a_1 \otimes \sigma^{l_1})) \\
		&= \mu_{\textup{Tw}(\mathbb{Z} {\cal{A}} * \mathbb{Z})}^d (s^{p_d}_{p_{d+1}} \otimes s^{l_d}_0 \otimes a_d \otimes \sigma^{l_d},\dots,s^{p_1}_{p_2} \otimes s^{l_1}_0 \otimes a_1 \otimes \sigma^{l_1}) \\
		&= \sum_{k_0,\dots,k_d} \sum_{(j(x))_{1 \leq x \leq k} \in \displaystyle\prod_{i=0}^d J_i^{k_i}} \mu_{\mathbb{Z}(\mathbb{Z} {\cal{A}} * \mathbb{Z})}^{d+k_0+\dots+k_d}(s^{q_{d,j(\sum_{i=0}^{d} k_i)}}_{r_{d,j(\sum_{i=0}^{d} k_i)}} \otimes s^0_0 \otimes a_{d,j(\sum_{i=0}^{d}k_i)}\otimes \sigma^0,\dots, \\
		&s^{q_{d,j(1+\sum_{i=0}^{d-1} k_i)}}_{r_{d,j(1+\sum_{i=0}^{d-1} k_i)}} \otimes s^0_0 \otimes a_{d,j(1+\sum_{i=0}^{d-1}k_i)}\otimes \sigma^0,s^{p_d}_{p_{d+1}} \otimes s^{l_d}_0 \otimes a_d \otimes \sigma^{l_d},\dots,s^{p_1}_{p_2} \otimes s^{l_1}_0 \otimes a_1 \otimes \sigma^{l_1}, \\
		&s^{q_{1,j(k_0)}}_{r_{1,j(k_0)}} \otimes s^0_0 \otimes a_{1,j(k_0)} \otimes \sigma^0,\dots,s^{q_{1,j(1)}}_{r_{1,j(1)}} \otimes s^0_0 \otimes a_{1,j(1)} \otimes \sigma^0) \\
		&= \sum_{k_0,\dots,k_d} \sum_{(j(x))_{1 \leq x \leq k} \in \displaystyle\prod_{i=0}^d J_i^{k_i}} (-1)^{\Sigma_1(j)} s^{q_{1,j(1)}}_{r_{d,j(\sum_{i=0}^{d} k_i)}} \otimes \mu_{\mathbb{Z} {\cal{A}} * \mathbb{Z}}^{d+k_0+\dots+k_d}(s^0_0 \otimes a_{d,j(\sum_{i=0}^{d}k_i)}\otimes \sigma^0,\dots, \\
		&s^0_0 \otimes a_{d,j(1+\sum_{i=0}^{d-1}k_i)}\otimes \sigma^0,s^{l_d}_0 \otimes a_d \otimes \sigma^{l_d},\dots,s^{l_1}_0 \otimes a_1 \otimes \sigma^{l_1},s^0_0 \otimes a_{1,j(k_0)} \otimes \sigma^0,\dots,s^0_0 \otimes a_{1,j(1)} \otimes \sigma^0) \\
		&= \sum_{k_0,\dots,k_d} \sum_{(j(x))_{1 \leq x \leq k} \in \displaystyle\prod_{i=0}^d J_i^{k_i}} (-1)^{\Sigma_1(j) + \Sigma_2(j)} s^{q_{1,j(1)}}_{r_{d,j(\sum_{i=0}^{d} k_i)}} \otimes \mu_{\mathbb{Z} {\cal{A}}}^{d+k_0+\dots+k_d}(s^0_0 \otimes a_{d,j(\sum_{i=0}^{d}k_i)},\dots, \\
		&s^0_0 \otimes a_{d,j(1+\sum_{i=0}^{d-1}k_i)},s^{l_d}_0 \otimes a_d,\dots,s^{l}_{l-l_1} \otimes a_1,s^l_l \otimes a_{1,j(k_0)},\dots,s^l_l \otimes a_{1,j(1)})\otimes \sigma^l \\
		&= \sum_{k_0,\dots,k_d} \sum_{(j(x))_{1 \leq x \leq k} \in \displaystyle\prod_{i=0}^d J_i^{k_i}} (-1)^{\Sigma_1(j) + \Sigma_2(j) + \Sigma_3(j)} s^{q_{1,j(1)}}_{r_{d,j(\sum_{i=0}^{d} k_i)}} \otimes s^l_0 \otimes \mu_{{\cal{A}}}^{d+k_0+\dots+k_d}(a_{d,j(\sum_{i=0}^{d}k_i)},\dots, \\
		&a_{d,j(1+\sum_{i=0}^{d-1}k_i)},a_d,\dots,a_1,a_{1,j(k_0)},\dots,a_{1,j(1)})\otimes \sigma^l \\
	\end{align*}
	where
	\begin{align*} \Sigma_1(j) =& \sum_{x<y} (q_{\textup{ind}(j(x)),j(x)} - r_{\textup{ind}(j(x)),j(x)}) \|a_{\textup{ind}(j(y)),j(y)}\| \\
		&+ \sum_{x=1}^{k} (q_{\textup{ind}(j(x)),j(x)} - r_{\textup{ind}(j(x)),j(x)}) \left(\sum_{m=\textup{ind}(j(x))+1}^{d} l_m + \|a_m\|\right) \\
		& + \sum_{x=1}^{k} \left(\sum_{i=1}^{\textup{ind}(j(x))}(p_i - p_{i+1})\right) \| a_{\textup{ind}(j(x)),j(x)} \| + \sum_{i<j} (p_i-p_{i+1})(l_j + \|a_j\|) \end{align*}
	\[\Sigma_2(j) = \sum_{i=1}^{d-1} l_i \left(\sum_{j=i+1}^{d} l_j\right) \]
	\[\Sigma_3(j) = \sum_{x=1}^{k} \|a_{\textup{ind}(j(x)),j(x)}\| \left(\sum_{m=1}^{\textup{ind}(j(x))} l_m \right) + \sum_{i=1}^{d} \|a_i\| \left(\sum_{m=1}^{i-1} l_m\right)\]
	
	A straightforward computation shows that for all $(j(x))_{1\leq x \leq k}$ in $\prod_{i=0}^d J_i^k$, $\Delta_1 + \Delta_2(j) + \Delta_3(j) = \Sigma_1(j)+\Sigma_2(j) + \Sigma_3(j)$. Thus $\tilde{\nu}$ is a strict $A_\infty$-functor.
