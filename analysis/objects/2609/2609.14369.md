<!-- head-1: body_md -->
\emph{What is the Morse index of the standard $G_2$-instanton over $S^7$?}
<!-- end head-1 -->
<!-- theorem-1: body_md -->
\label{thm1} 
		Let $\nabla'$ be the standard anti-self-dual BPST instanton on $P_4\to S^4$, and let $\nabla=\phi^*\nabla'$ be its pullback under the quaternionic Hopf fibration. Then the only negative eigenvalue of $\mathscr S^\nabla|_{\ker\delta^\nabla}$ is $-3$, with multiplicity $8$. In particular, 
		\begin{align*} 
			i(\nabla)=8. 
		\end{align*} 
		Therefore, $\nabla$ attains the sharp lower bound for the Morse index of nonflat Yang--Mills connections on $S^7$.
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
The negative eigenspace in Theorem \ref{thm1} has a natural geometric description. For any vector $v\in\mathbb{R}^8$, let $F_v(x)=v\cdot x$ be the linear map on $\mathbb{R}^8$ and $f_v=F_v\mid_{S^7}$ be the restriction on $S^7$. Define the conformal gradient vector field 
		\begin{align*} 
			V_v(x)=\operatorname{grad}_{S^7}f_v(x). 
		\end{align*} 
		By \cite[Lemma 3.1 and Propositions 3.3--3.4]{Ni}, we have $i_{V_v}F\in\operatorname{ker}\delta^\nabla$ and $\mathscr S^\nabla(i_{V_v}F)=-3i_{V_v}F$. It is easily to see that $\{i_{V_v}F_\nabla\mid v\in\mathbb{R}^8\}$ is an 8-dimensional space. Applying Theorem \ref{thm1}, we obtain 
		\begin{align*} 
			\ker(\mathscr S^\nabla\mid_{\ker\delta^\nabla}+3I)=\{i_{V_v}F\mid v\in\mathbb{R}^8\}. 
		\end{align*}
<!-- end remark-1 -->
<!-- theorem-2: body_md -->
\label{thm2} 
		Let $\nabla$ be the $G_2$ instanton defined in Theorem \ref{thm1}. Then nullity of $\nabla$ is 
		\begin{align*} 
			n(\nabla)=20. 
		\end{align*} 
		Moreover, we have 
		\begin{align*} 
			\ker\mathscr S^\nabla\mid_{\operatorname{ker}\delta^\nabla}=\{i_VF\mid V\in\mathcal V\}, 
		\end{align*} 
		where 
		\begin{align*} 
			\mathcal V=\{&x\mapsto(I-xx^\dagger)(M_0x+M_1xi+M_2xj+M_3xk)\mid\\ 
			&M_r\in M_2(\mathbb H),\ M_r^\dagger=M_r,\ \operatorname{tr}(M_r)=0,\ 0\le r\le3\} 
		\end{align*} 
		is a 20-dimensional vector space. Here $x\in S^7\subset\mathbb H^2$, the matrices $M_r$ are constant, and $\dagger$ denotes quaternionic conjugate transpose.
<!-- end theorem-2 -->
<!-- remark-2: body_md -->
By \cite[Proposition 4.4 and Theorem 5.11]{AW}, the $15$-dimensional infinitesimal instanton deformation space is precisely the subspace in Theorem \ref{thm2} obtained by setting $M_3=0$. Thus the terms $M_0x$, $M_1xi$, and $M_2xj$ account for Waldron's deformations, while the remaining term $M_3xk$ gives a complementary 5-dimensional subspace.

<!-- lemma-1: body_md -->
Under the identification $\mathbb R^3\cong\operatorname{Im}\mathbb H\cong\mathfrak k$, the associated bundle $E=P\times_{\operatorname{SO}(3)}\mathbb R^3$ has the homogeneous realization 
		\begin{align} 
			E\cong G\times_K\mathfrak k, 
		\end{align} 
		with $[g\operatorname{diag}(1,q),Y]=[g,\operatorname{Ad}(q)Y]$. There is also a natural $G$-equivariant vector-bundle isomorphism $E'\cong E$ over $G/K$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $\lambda$ is a group homomorphism, its differential satisfies 
		\begin{align} 
			\lambda_*\circ\operatorname{Ad}(q)=\operatorname{Ad}(\lambda(q))\circ\lambda_* 
		\end{align} 
		for every $q\in \operatorname{Sp}(1)$. Define 
		\begin{align*} 
			\Phi:E'\to E,\qquad\Phi([[g,h],X])=[g,\lambda_*^{-1}(\operatorname{Ad}(h)X)]. 
		\end{align*} 
		For $h_1,h_2\in \operatorname{SO}(3)$, we have 
		\begin{align*} 
			\Phi([[g,h_1h_2],X])=[g,\lambda_*^{-1}(\operatorname{Ad}(h_1)\operatorname{Ad}(h_2)X)]=\Phi([[g,h_1],\operatorname{Ad}(h_2)X]). 
		\end{align*} 
		Moreover, for $q\in \operatorname{Sp}(1)$, the equivariance of $\lambda_*$ gives 
		\begin{align*} 
			\Phi([[g\operatorname{diag}(1,q),h],X])=[g,\operatorname{Ad}(q)\lambda_*^{-1}(\operatorname{Ad}(h)X)]=[g,\lambda_*^{-1}(\operatorname{Ad}(\lambda(q)h)X)]=\Phi([[g,\lambda(q)h],X]). 
		\end{align*} 
		Thus $\Phi$ is well defined. 
		 
		Define 
		\begin{align*} 
			\Psi:E\to E',\qquad\Psi([g,Y])=[[g,I_3],\lambda_*(Y)]. 
		\end{align*} 
		The same equivariance identity shows that $\Psi$ is well defined. Moreover, 
		\begin{align*} 
			\Phi\circ\Psi=\operatorname{id}_E,\qquad \Psi\circ\Phi=\operatorname{id}_{E'}. 
		\end{align*} 
		Therefore, $\Phi$ is a $G$-equivariant vector-bundle isomorphism.
