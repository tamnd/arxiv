<!-- definition-1: body_md -->
\label{DefPW}
    Let \( \omega \in [\lambda_{min},\lambda_{max}]\). A function \( f \in \ell^2(G) \) is said to belong to the \emph{Graph Paley-Wiener space} \( GPW_\omega(G) \) if its ``graph Fourier transform'' \( \mathcal{U} f \) is supported in the interval \( [0, \omega] \); that is,
    \[
    \operatorname{supp}(\mathcal{U} f) \subseteq [0, \omega].
    \]
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
Let \( f \in L^2(\mathbb{R}) \). The following statements are equivalent:
            \begin{enumerate}
                \item $f\in PW_u(\mathbb{R}).$

                \item The function \( f \) extends to an entire function on \( \mathbb{C} \) and satisfies the growth condition
                \[
                |f(z)| \leq C e^{2\pi u |\operatorname{Im}(z)|}, \quad \text{for all } z \in \mathbb{C},
                \]
                for some constant \( C > 0 \).
            \end{enumerate}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{ShannonT}
        Let \(f\in PW_u(\mathbb{R}) \).
        Then \( f \) is completely determined by its samples at the points \( \left\{ \frac{j}{2u} \right\}_{j \in \mathbb{Z}} \), and can be reconstructed by the formula
        \begin{equation}\label{shannon}
        f(x) = \sum_{j \in \mathbb{Z}} f\left( \frac{j}{2u} \right) \cdot \frac{\sin\left(2\pi
         u\left(x - \frac{j}{2u}\right)\right)}{2\pi u\left(x - \frac{j}{2u}\right)}= \sum_{j \in \mathbb{Z}} f\left( \frac{j}{2u} \right) \cdot \operatorname{sinc}\left(2u\left(x - \frac{j}{2u}\right)\right)
                                                                      
                             
        \end{equation}
        where the \emph{sinc function} is defined by
        \begin{equation}\label{eq:sinc}
        \operatorname{sinc}(x) = \frac{\sin(\pi x)}{\pi x},\quad x\not=0,\,\,\mbox{    and }\,\,\operatorname{sinc}(0)=1.
        \end{equation}
 The series converges in the \(L^2(\mathbb{R})\)-sense and uniformly.
<!-- end theorem-2 -->
<!-- definition-2: body_md -->
A sequence \(\{\lambda_n\}_{n\ge1}\subset\mathbb{D}\) is called \emph{exponential} if there exists a constant
\(0<c<1\) such that
\[
1-|\lambda_{n+1}|
\le
c\bigl(1-|\lambda_n|\bigr),
\qquad n\ge1.
\]
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
Every exponential sequence is uniformly separated.
<!-- end proposition-1 -->
<!-- example-1: body_md -->
\label{ex:unif}
        An example of uniformly separated sequence is given by $(\lambda_n)_{n\geq 1}$, with $\lambda_n=1-e^{-n}$. Then $\lambda_{n+1}=1-e^{-n-1}$, and
        $$1-\lambda_{n+1}=e^{-n-1}=\frac{1}{e}e^{-n}=\frac{1}{e}(1-\lambda_n),$$
        as desired.
<!-- end example-1 -->
<!-- theorem-3: body_md -->
\label{TeorCarl}
        Let \( T \in \mathcal{B}(\mathcal{H}) \) be a normal operator and let \( f_0 \in \mathcal{H} \). Then \( (T^n f_0)_{n \in \mathbb{N}} \) is a frame for \( \mathcal{H} \) if and only if
        \begin{equation}\label{diag}
        T = \sum_{j=0}^\infty \lambda_j \langle \cdot, e_j \rangle e_j,
        \end{equation}
        where \( (\lambda_j)_{j \in \mathbb{N}} \subset \mathbb{D} \) is uniformly separated, \( (e_j)_{j \in \mathbb{N}} \) is an orthonormal basis for \( \mathcal{H} \), and
        \begin{equation}\label{coeff}
        |\langle f_0, e_j \rangle|^2 \sim 1 - |\lambda_j|^2.
        \end{equation}
        In this case, the frame \( (T^n f_0)_{n \in \mathbb{N}} \) has frame bounds \( \alpha \Delta^{-1} \) and \( \beta \Delta \), where
        \[
        \alpha := \inf_{j \in \mathbb{N}} \frac{|\langle f_0, e_j \rangle|^2}{1 - |\lambda_j|^2}, \qquad
        \beta := \sup_{j \in \mathbb{N}} \frac{|\langle f_0, e_j \rangle|^2}{1 - |\lambda_j|^2}, \qquad
        \Delta := \frac{2}{\delta_\Lambda^4} \left(1 - 2 \log \delta_\Lambda \right)
        \]
        and $\delta_\Lambda$ is the uniform bound defined in \eqref{carleson-cond}.

<!-- lemma-1: body_md -->
\label{lemma0}
    The normalized Laplace operator \( \mathcal{L} \) in \eqref{eq:normlap} is a convolution operator. Thus, on the  Fourier transform side, it is a multiplication operator:
    \begin{equation}\label{L-multip}
    \mathcal{F}\mathcal{L} \mathcal{F}^{-1} = M_{\varphi},
    \end{equation}
    where  $M_{\varphi} : L^2(\mbox{${\mathbb{T}}$}) \rightarrow L^2(\mbox{${\mathbb{T}}$})$ is the multiplication operator by the function
     \begin{equation}\label{fi}
        \varphi(\xi)=(1-\cos(2\pi \xi))=2\, \sin^2({\pi\xi}),\quad\xi\in\ \mbox{${\mathbb{T}}$}.
    \end{equation}
  Namely, $ M_{\varphi} u=\varphi \, u$, for every $u\in L^2(\mbox{${\mathbb{T}}$})$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Using \eqref{eq:normlap} and \eqref{FT}, we compute
    \begin{align*}
        \mathcal{F}(\mathcal{L} f)(\xi)&=\frac{1}{{2}}\left(2\sum_{n\in\mathbb{Z}} f(n)e^{2\pi i n\xi}-\sum_{n\in\mathbb{Z}}f(n+1)e^{2\pi i n\xi}-\sum_{n\in\mathbb{Z}}f(n-1)e^{2\pi i n\xi}\right)\\
        &= \mathcal{F}{f}(\xi)-\frac12 e^{-2\pi i \xi} \mathcal{F}{f}(\xi)-\frac12 e^{2\pi i \xi} \mathcal{F}{f}(\xi)\\
        &=\mathcal{F}{f}(\xi)\left(1-\frac12\left( e^{-2\pi i \xi}+e^{2\pi i \xi}\right)\right)\\
        &=(1-\cos(2\pi \xi))\mathcal{F}{f}(\xi).               
    \end{align*}
   This is the desired equality.

\label{teospectr}
Consider the normalized Laplace operator \( \mathcal{L} \) in \eqref{eq:normlap} and  the operator  $W$ defined in \eqref{W}. Then,  operator $ \mathcal{U}:   \ell^2(\mathbb{Z})\to X,$
  defined by
  \begin{equation} \quad \mathcal{U}= W\mathcal F,
      \end{equation}
      is a unitary operator.  Furthermore,
  \begin{equation}\label{spettU}
       \mathcal{L}\,  =\mathcal{U}^{-1} M_z \,\mathcal{U},
    \end{equation}
where $M_z: X \rightarrow X$ is the  multiplication operator by the independent variable defined in \eqref{eq:Multip}, i.e., $(M_z h)(z)=z h(z)$, $z\in [0,2).$

<!-- proof-2: body_md -->
We will first prove that the operator $W :  L^2(\mbox{${\mathbb{T}}$}) \longrightarrow X$ is unitary, which will imply that  $ \mathcal{U}:   \ell^2(\mathbb{Z}) \to X$ is unitary.


Using Lemma \ref{lemma0} we have
    \begin{equation}\label{intertw}
        \mathcal{L} =\mathcal{F}^{-1}M_\varphi \mathcal{F}.
    \end{equation}
    Since $\mathcal{F}$ is unitary, the spectrum $\sigma(\mathcal{L})$ of $\mathcal{L}$ is given by
    $$\sigma(\mathcal{L})=\sigma(M_\varphi)$$
    and, by the spectral mapping theorem,
    \[
    \sigma(M_\varphi)=\varphi(\sigma(M_\xi)),
    \]
    where $M_\xi$ is the multiplication by the independent variable, namely, $M_\xi g(\xi)=\xi g(\xi)$, for $\xi\in \mbox{${\mathbb{T}}$}$ and $g\in L^2(\mbox{${\mathbb{T}}$})$. Since  $\sigma(M_\xi)=[-1/2,1/2]$,  we obtain $\varphi(\sigma(M_\xi))=[0,2]$.

Now take $g\in L^2(\mbox{${\mathbb{T}}$})$ and compute
    \begin{align*}
        \|g\|_{2}^2&=\int_{-1/2}^{1/2} |g(\xi)|^2 d\xi=\int_0^{1/2}|g(\xi)|^2 d\xi+\int_{-1/2}^0 |g(\xi)|^2 d\xi\\
        &=  \int_0^{1/2}\left(|g(\xi)|^2 +|g(1-\xi)|^2\right) d\xi.
    \end{align*}
    We make the change of variable $z=\varphi(\xi)$, where $\varphi$ is defined in \eqref{fi}, so that
    \begin{equation*}
        \xi=\frac1{2\pi}\arccos(1-z),\quad d\xi=\frac1{2\pi}\frac{dz}{\sqrt{(1-(1-z)^2)}}=\frac1{2\pi}\frac{dz}{\sqrt{z(2-z)}},
    \end{equation*}
    and
    $$    \|g\|_{2}^2=\int_0^2\left( \left|g\left(\frac1{2\pi}\arccos(1-z)\right)\right|^2+\left|g\left(1-\frac1{2\pi}\arccos(1-z)\right)\right|^2\right)\frac1{2\pi}\frac{dz}{\sqrt{z(2-z)}} =\|Wg\|^2_X$$
                                    
                                
                   
    Hence $W$ is isometric, $$    \|W g\|_{X} = \|g\|_{2},$$
    where
    \begin{equation*}\label{tilde u-bis}
       Wg(z)=\left(g\left(\frac1{2\pi}\arccos(1-z)\right),g\left(1-\frac1{2\pi}\arccos(1-z)
        \right)\right).
    \end{equation*}
                                                                                                                                             
    
  The range of $W$ is the whole space $X$; indeed, given $(u_1,u_2)\in X$, the function $g$ defined by 
$$g(\xi)=\left\{\begin{array}{ll}
        u_1(\varphi(\xi)), &\xi\in[0,1/2), \\
       u_2(\varphi(\xi)), & \xi\in[-1/2,0),
   \end{array}\right.$$
    and extended $1$-periodically, belongs to $L^2(\mbox{${\mathbb{T}}$})$ and satisfies $Wg=(u_1,u_2).$ In particular, by periodicity, for $\xi=\frac1{2\pi}\arccos(1-z)\in[0,\frac12),$ 
    $$g(1-\xi)=g(-\xi)= u_2(\varphi(-\xi))=u_2(\varphi(\xi))=u_2(z).$$
 Hence, $W$ is unitary.

    $\mathcal{U}=W\mathcal{F}, \quad \mathcal{U}:\ell^2(\mathbb{Z})\to X$ is then unitary.
    

Furthermore, it is easy to see that $M_{\varphi}=W^{-1} M_z W$ and since $\mathcal{L} =\mathcal{F}^{-1} M_\varphi \mathcal{F},$ we have that
$\mathcal{U}^{-1} M_z\,\mathcal{U} = \mathcal{L}$.

<!-- remark-5: body_md -->
The Graph Paley--Wiener space is well defined for any
$\omega \in (0,2]$, or equivalently, for any symmetric interval in the group
$\mbox{${\mathbb{T}}$}$. This construction yields a closed subspace of
$\ell^2(\mbox{${\mathbb{Z}}$})$. However, the Classical Sampling Theorem in
$\ell^2(\mbox{${\mathbb{Z}}$})$ makes sense only when the corresponding interval in
$\mbox{${\mathbb{T}}$}$ has length $1/\ell$ for some positive integer $\ell$.

The reason for this is that we will  evaluate the functions in the
Paley--Wiener space on a sub-lattice of $\mbox{${\mathbb{Z}}$}$, namely $\ell\mbox{${\mathbb{Z}}$}$.
Consequently, the interval defining the Paley--Wiener space must
be a fundamental domain of the quotient group $\mbox{${\mathbb{T}}$}$ with the
dual lattice $\{k/\ell\}_{k \in \mbox{${\mathbb{Z}}$}_{\ell}}$. See \cite{K1965}.

This obstacle does not pose a problem in applications if we are willing
to increase the sampling rate.
Let us denote
\[\label{wl}
\omega_\ell = \varphi(1/2\ell), \qquad \ell \in \mbox{${\mathbb{N}}$}.
\]
Then every function $f \in \ell^2(\mbox{${\mathbb{Z}}$})$ belongs to some
Graph Paley--Wiener space $GPW_{\omega}$ for a certain
$\omega \in (0,2]$.
Since there exists an  $\ell \in \mbox{${\mathbb{N}}$}$ such that $\omega_{\ell+1} < \omega \leq \omega_{\ell}$ and
$GPW_{\omega} \subseteq GPW_{\omega_{\ell}}$,
it follows that $f$ can be sampled
on the sub-lattice $\ell\mbox{${\mathbb{Z}}$}.$

\label{ShannotheoremZ}

& \langle f,\,g_{j}\rangle_{\ell^2(\mathbb{Z})}=\frac{1}{\sqrt{\ell}}\,\sum_{m\in\mathbb{Z}}f(m)\,\operatorname{sinc}(m/\ell - j)=
\frac{1}{\sqrt{\ell}}\,\sum_{m\in\mathbb{Z}}\sum_{p \in \mathbb{Z}} f(\ell p)\operatorname{sinc}(m/\ell - p)\,\operatorname{sinc}(m/\ell - j)\\ \\
&=
\frac{1}{\sqrt{\ell}}\,\sum_{p \in \mathbb{Z}} f(\ell p)\sum_{m\in\mathbb{Z}}\operatorname{sinc}(m/\ell - p)\,\operatorname{sinc}(m/\ell - j)=
\sqrt{\ell}\,\sum_{p \in \mathbb{Z}} f(\ell p)\langle g_p,g_j\rangle_{\rule{0pt}{1.5ex} \ell^2(\mathbb{Z})}\\ \\
&=
\sqrt{\ell}\,\sum_{p \in \mathbb{Z}} f(\ell p)\,\delta_{p,j}
=
\sqrt{\ell}\,f(\ell j).
\end{align*}
