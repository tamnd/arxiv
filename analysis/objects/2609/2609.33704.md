For a finite set $X$, the collection of its unordered two-element subsets is denoted by $\binom X2$. The binomial distribution with parameters $m$ and $p'$ is written $\operatorname{Bin}(m,p')$, and the uniform distribution on $X$ is written $\operatorname{Unif}(X)$. Expectation is denoted by $\operatorname{E}[X]$, with square brackets enclosing its argument throughout; subscripts indicate the randomness being averaged whenever this is not clear from the context, and in particular $\operatorname{E}_t[X]$ is defined in~\cref{sec:global}. <!-- definition-1: body_md -->
\label{def:defective}
At a level with $g$ blocks, a block $u$ is \emph{defective} if $\mathcal G_u$ contains an edge of $G$. An unordered pair $\{u,v\}\in\binom{[g]}2$ is \emph{defective} if $\mathcal G_u\cup\mathcal G_v$ contains an edge of $G$. A block or pair that is not defective is called \emph{non-defective}. The \emph{level graph} is the simple graph $G_g:=([g],E_g)$, where
\[
E_g:=\left\{\{u,v\}\in\binom{[g]}2:
\mathcal G_u\cup\mathcal G_v\text{ contains an edge of }G\right\}.
\]

<!-- fact-1: body_md -->
\label{lem:Dg-det}
For every graph and every level with $g$ blocks, we have
\begin{equation}
D_g\le |E|+gH_g.
\label{eq:Dg-bound}
\end{equation}

<!-- definition-2: body_md -->
\label{def:typical}
Let $(\epsilon_n)$ be a sequence of positive numbers with $\epsilon_n\to0$. We define $\mathcal T(\epsilon_n)$ as the set of graphs $G=([n],E)$ satisfying the following conditions:
\begin{enumerate}[label=(\roman*),itemsep=2pt]
\item The number of edges satisfies
\begin{equation}
(1-\epsilon_n)\bar k\le |E|\le(1+\epsilon_n)\bar k.
\label{eq:struct-edges}
\end{equation}
\item The number of internal edges at the base level satisfies
\begin{equation}
H_{g_0}\le3\sqrt{\bar k}.
\label{eq:struct-internal}
\end{equation}
\item The crossing-edge counts across all levels satisfy
\begin{equation}
\max_{0\le j\le L}\Delta_{g_j}^{\times}\le24\sqrt{\bar k}.
\label{eq:struct-cross}
\end{equation}
\item The weighted sum of internal-edge counts satisfies
\begin{equation}
\sum_{j=0}^{L-1}g_jH_{g_j}\le 3\bar k\log n.
\label{eq:struct-weighted}
\end{equation}
\end{enumerate}
A graph in $\mathcal T(\epsilon_n)$ is called \emph{$\epsilon_n$-typical}.

<!-- proposition-1: body_md -->
\label{prop:one-point}
For every fixed $r\in\mathbb F_p$, the variables $(P_u(r))_{u\in[g]}$ are independent and uniform on $\mathbb F_p$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For every $s\in\mathbb F_p$ and every $a\in\mathbb F_p$, exactly one value $b=s-ar$ satisfies $ar+b=s$. Each value therefore has exactly $p$ preimages under $(a,b)\mapsto ar+b$, which proves uniformity. Independence follows since each image depends only on the coefficient pair of its own block.
<!-- end proof-1 -->
<!-- proposition-2: body_md -->
\label{prop:two-point}
For every block $w$ and distinct $r,r'\in\mathbb F_p$, the vector $(P_w(r),P_w(r'))$ is uniform on $\mathbb F_p^2$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
The linear map $(a,b)\mapsto(ar+b,ar'+b)$ has matrix $\begin{pmatrix}r&1\\r'&1\end{pmatrix}$
    
                                       
    
with nonzero determinant $r-r'$. It is therefore a bijection of $\mathbb F_p^2$ and preserves the uniform law.
<!-- end proof-2 -->
<!-- proposition-3: body_md -->
\label{prop:composition}
Let $u,v$ be distinct blocks. Condition on $\mathcal C_{uv}$ and on the collision point $T_{uv}=(r,s)$ from~\eqref{eq:collision-point}, assuming that this event has positive probability. Then:
\begin{enumerate}[label=(\roman*),itemsep=2pt]
\item both $u$ and $v$ belong to the test $(r,s)$;
\item each block outside $\{u,v\}$ belongs to this test independently with probability $1/p$;
\item the conditional joint distribution of these membership indicators does not depend on $(r,s)$.
\end{enumerate}
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
On the conditioning event, $P_u(r)=P_v(r)=s$, which proves~(i). This event depends only on the coefficients of $u$ and $v$. The coefficient pairs of all other blocks therefore remain independent and uniform on $\mathbb F_p^2$. By~\cref{prop:one-point}, their values at $r$ are independent and uniform on $\mathbb F_p$. Each value equals $s$ with probability $1/p$, proving~(ii). Thus, the membership indicators are independent Bernoulli variables with parameter $1/p$. Their joint distribution is the same for every $(r,s)$, which proves~(iii).
<!-- end proof-3 -->
<!-- proposition-4: body_md -->
\label{prop:first_cordinate}
The first coordinates $r_{uv}^*$ defined in~\eqref{eq:collision-point} satisfy the following properties.
\begin{enumerate}[label=(\roman*),itemsep=2pt]
\item For distinct blocks $u,v$, the first coordinate $r_{uv}^*$ is uniform on $\mathbb F_p$ conditional on $\mathcal C_{uv}$.
\item For disjoint pairs $\{u,v\}$ and $\{u',v'\}$, the first coordinates $r_{uv}^*$ and $r_{u'v'}^*$ are independent and uniform on $\mathbb F_p$ conditional on $\mathcal C_{uv}\cap\mathcal C_{u'v'}$.
\item For pairs $\{u,v\}$ and $\{u,v'\}$ with $v\ne v'$, the first coordinates $r_{uv}^*$ and $r_{uv'}^*$ are independent and uniform on $\mathbb F_p$ conditional on $\mathcal C_{uv}\cap\mathcal C_{uv'}$ and the coefficients $(a_u,b_u)$ of the shared block.
\end{enumerate}
In both~(ii) and~(iii), the probability that both collision events hold and the two first coordinates are equal is at most $1/p$.
<!-- end proposition-4 -->
<!-- proof-4: body_md -->
For~(i), fix $a_u,b_u,a_v$ with $a_u\ne a_v$. The coefficient $b_v$ remains uniform on $\mathbb F_p$, and
\[
r_{uv}^*=(b_v-b_u)(a_u-a_v)^{-1}.
\]
Since $a_u-a_v\ne0$, each value of $r_{uv}^*$ corresponds to exactly one value of $b_v$. Thus, $r_{uv}^*$ is uniform on $\mathbb F_p$. This holds for every choice of the fixed coefficients, so averaging over them proves~(i).

For~(ii), the two pairs involve disjoint sets of independent coefficients. Conditioning on $\mathcal C_{uv}\cap\mathcal C_{u'v'}$ preserves their independence because each event depends only on the coefficients of its own pair. Applying~(i) to each pair gives the claim.

For~(iii), fix $(a_u,b_u)$. The coefficient pairs $(a_v,b_v)$ and $(a_{v'},b_{v'})$ remain independent. The conditions $a_v\ne a_u$ and $a_{v'}\ne a_u$ restrict these pairs separately, so they remain independent after conditioning. The argument in~(i) shows that each first coordinate is uniform on $\mathbb F_p$, proving~(iii).

Two independent uniform elements of $\mathbb F_p$ are equal with probability $1/p$. In case~(iii), this probability is the same for every choice of the shared coefficients, so it remains $1/p$ after averaging over them. In either case, multiplying by the probability that both collision events hold gives the final bound.

<!-- theorem-1: body_md -->
\label{thm:main}
Let $G\sim \mathrm{ER}(n,q)$, where $\bar k=q\binom n2$ satisfies $\bar k\to\infty$ and $\bar k=o(n^2)$. Consider the affine test design in Algorithm~\ref{alg:test} and the splitting decoder in Algorithm~\ref{alg:decoder} with $C_1=2000$ and $R=\log n$. Then the following guarantees hold as $n\to\infty$:
\begin{enumerate}[label=(\roman*),itemsep=3pt]
\item the test design uses $\mathcal O(\bar k\log n)$ non-adaptive queries;
\item the decoder recovers $E$ exactly with probability $1-o(1)$;
\item the decoding time is $\mathcal O(\bar k\log n)$ with probability $1-o(1)$.
\end{enumerate}

<!-- lemma-2: body_md -->
\label{lem:survival}
For every fixed $G\in\mathcal T(\epsilon_n)$ and all sufficiently large $n$, every non-defective pair $e$ at every non-final level and every final round satisfies
\begin{equation}
q_e:=\mathbb P[\mathcal E_e]\le \frac1{24}.
\label{eq:alpha}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-5: body_md -->
The slopes of the two blocks are independent and uniform, whence $\mathbb P[\mathcal{C}_e^c]=1/p$ and
\begin{equation}
q_e=\frac1p+\left(1-\frac1p\right)\rho_e,
\qquad
\rho_e:=\mathbb P[Y_{T_e}=1\mid\mathcal{C}_e].
\label{eq:q-decomp}
\end{equation}
By~\cref{prop:composition}, the conditional positivity probability given $\mathcal{C}_e$ and $T_e=(r,s)$ is the same for every admissible $(r,s)$ and hence equals $\rho_e$. Fix one such point and let
\[
\mathcal{B}:=\{w\in[g]\setminus e:\ w\in T_e\}
\]
denote the set of non-special blocks belonging to the test, so that the queried vertex set is $\bigcup_{x\in e\cup\mathcal{B}}\mathcal{G}_x$. By~\cref{prop:composition}(ii), the blocks of $[g]\setminus e$ lie in $\mathcal{B}$ independently with probability $1/p$.

Since $e$ is non-defective, neither block of $e$ contains an internal edge and no edge joins the two blocks. Every edge witnessing a positive outcome therefore meets at least one block of $\mathcal{B}$, and its two endpoint blocks determine which of the following three events occurs:
\begin{align}
\mathcal{F}_1&:=\{\exists\,w\in\mathcal{B}:\ \mathcal{G}_w\text{ contains an edge}\},
\label{eq:event-A1}\\
\mathcal{F}_2&:=\{\exists\,w\in\mathcal{B},\ x\in e:\ \text{some edge joins }\mathcal{G}_w\text{ and }\mathcal{G}_x\},
\label{eq:event-A2}\\
\mathcal{F}_3&:=\{\exists\,w,w'\in\mathcal{B},\ w\ne w':\ \text{some edge joins }\mathcal{G}_w\text{ and }\mathcal{G}_{w'}\}.
\label{eq:event-A3}
\end{align}
These cases exhaust the possibilities, an edge whose endpoint blocks both lie in $e$ being excluded by non-defectiveness. Consequently,
\begin{equation}
\{Y_{T_e}=1\}\subseteq\mathcal{F}_1\cup\mathcal{F}_2\cup\mathcal{F}_3 .
\label{eq:event-cover}
\end{equation}
Each event is now bounded separately, all three estimates being uniform over the admissible collision points.

\emph{The event $\mathcal{F}_1$.}
Distinct internally defective blocks contain distinct internal edges, so that the number of blocks $w$ with $\mathcal{G}_w$ containing an edge is at most $H_g$; by nestedness and~\eqref{eq:struct-internal}, this is at most $H_{g_0}\le3\sqrt{\bar k}$, and at the singleton level it is zero. Each such block lies in $\mathcal{B}$ with probability $1/p$, whence
\begin{equation}
\mathbb P[\mathcal{F}_1]\le\frac{3\sqrt{\bar k}}p .
\label{eq:bound-A1}
\end{equation}

\emph{The event $\mathcal{F}_2$.}
A block $w$ joined to $\mathcal{G}_u$ or to $\mathcal{G}_v$ requires an edge crossing the boundary of $\mathcal{G}_u$ or of $\mathcal{G}_v$, and distinct such blocks require distinct crossing edges; there are therefore at most $2\Delta_g^{\times}\le48\sqrt{\bar k}$ of them by~\eqref{eq:struct-cross}. The same union bound gives
\begin{equation}
\mathbb P[\mathcal{F}_2]\le\frac{48\sqrt{\bar k}}p .
\label{eq:bound-A2}
\end{equation}

\emph{The event $\mathcal{F}_3$.}
An edge witnesses $\mathcal{F}_3$ only if its two endpoint blocks are distinct, lie outside $e$, and both belong to $\mathcal{B}$, which by independence has probability $1/p^2$. Since at most $|E|\le2\bar k$ edges qualify, by~\eqref{eq:struct-edges},
\begin{equation}
\mathbb P[\mathcal{F}_3]\le\frac{2\bar k}{p^2}.
\label{eq:bound-A3}
\end{equation}

Combining~\eqref{eq:event-cover} with~\eqref{eq:bound-A1}--\eqref{eq:bound-A3} yields
\[

q_e\le\frac1p+\rho_e
\le\frac{52}{C_1}+\frac2{C_1^2}
\le\frac1{24}.
\qedhere
\]

<!-- lemma-5: body_md -->
\label{lem:L2}
There is an absolute constant $C_2$ such that, for every fixed typical graph and all sufficiently large $n$,
\begin{equation}
\sum_{t=0}^{M}\operatorname{E}[Z_t^2]
\le C_2(Z_0^2+A^2+M).
\label{eq:L2-sum}
\end{equation}
<!-- end lemma-5 -->
<!-- proof-9: body_md -->
Fix the affine coefficients of all stages before $t$. Both sides of~\eqref{eq:one-step-global} are nonnegative. Squaring this inequality and taking expectation over $\mathcal A_t$ gives
\begin{align*}
\operatorname{E}_t[Z_{t+1}^2]
&\le36\operatorname{E}_t[(\beta_t+X_t)^2]\\
&=36(\beta_t+m_t)^2+36\operatorname{Var}_t(X_t)\\
&\le36\left(\beta_t+\frac{Z_t}{24}\right)^2
  +36Z_t+\frac{252}{p}Z_t^2\\
&\le72\beta_t^2+
  \left(\frac18+\frac{252}{p}\right)Z_t^2+36Z_t.
\end{align*}
The third line follows from~\eqref{eq:stagewise-mean-global}--\eqref{eq:stagewise-var-global}, and the last line uses $(x+y)^2\le2x^2+2y^2$. Since $p\to\infty$, we have $252/p\le1/8$ for all sufficiently large $n$. Also,
\[
36z\le\frac18z^2+2592\qquad(z\ge0),
\]
which follows from $(z-144)^2\ge0$. Combining these bounds, we obtain
\[
\operatorname{E}_t[Z_{t+1}^2]\le\frac12Z_t^2+C(\beta_t^2+1)
\]
for an absolute constant $C$. Averaging over the earlier families gives
\begin{equation}
\operatorname{E}[Z_{t+1}^2]\le\frac12\operatorname{E}[Z_t^2]+C(\beta_t^2+1).
\label{eq:L2-step-full}
\end{equation}
Let $B_2:=\sum_{t=0}^{M}\operatorname{E}[Z_t^2]$. Summing~\eqref{eq:L2-step-full} over $0\le t<M$ yields
\[
B_2-Z_0^2\le\frac12(B_2-\operatorname{E}[Z_M^2])
+C\left(\sum_{t=0}^{M-1}\beta_t^2+M\right).
\]
Using $\operatorname{E}[Z_M^2]\ge0$ and
\[
\sum_{t=0}^{M-1}\beta_t^2
\le\left(\sum_{t=0}^{M-1}\beta_t\right)^2=A^2,
\]
we obtain
\[
B_2\le2Z_0^2+2C(A^2+M).
\]
Choosing $C_2:=\max\{2,2C\}$ proves the result.

<!-- lemma-6: body_md -->
\label{lem:orthogonality}
Let $\Delta_t=X_t-m_t$ be the centered stage error, where $m_t=\operatorname{E}_t[X_t]$. For $0\le s<t<M$, we have
\begin{equation}
\operatorname{E}[\Delta_s\Delta_t]=0.
\label{eq:orthogonality}
\end{equation}
Consequently, with $S:=\sum_{t=0}^{M-1}\Delta_t$,
\begin{equation}
\operatorname{E}[S]=0,
\qquad
\operatorname{E}[S^2]=\sum_{t=0}^{M-1}\operatorname{E}[\Delta_t^2].
\label{eq:sum-errors}
\end{equation}
<!-- end lemma-6 -->
<!-- proof-10: body_md -->
Fix $0\le s<t<M$ and the affine coefficients of all stages before $t$. Since $X_s$ and $m_s$ depend only on earlier families, $\Delta_s$ is fixed. Therefore,
\[
\operatorname{E}_t[\Delta_s\Delta_t]=\Delta_s\operatorname{E}_t[\Delta_t]=0.
\]
The product $\Delta_s\Delta_t$ depends only on $\mathcal A_0,\ldots,\mathcal A_t$, so averaging over the earlier families using~\eqref{eq:successive-averaging} gives~\eqref{eq:orthogonality}. Similarly,
\[
\operatorname{E}[\Delta_t]=\operatorname{E}[\operatorname{E}_t[\Delta_t]]=0.
\]
Summing these mean-zero identities gives $\operatorname{E}[S]=0$. Expanding $S^2$ and using~\eqref{eq:orthogonality} gives the second identity in~\eqref{eq:sum-errors}.

\label{thm:workload}

\le C_4\left(
\frac1{\bar k\log n}+\frac1p+

\end{equation}

W-Z_0\le6A+\frac14(W-Z_M)+6S.

Since $Z_M\ge0$, we obtain
\begin{equation}
\frac34W\le Z_0+6A+6S.
\label{eq:W-decomp}
\end{equation}

<!-- lemma-7: body_md -->
\label{lem:no-fp}
Let $\widehat E$ be the output of~\cref{alg:decoder} after $R=\log n$ final rounds. For every fixed typical graph and all sufficiently large $n$, we have
\begin{equation}
\mathbb P[\widehat E\setminus E\ne\varnothing]
\le n^{2-\log 24}=o(1).
\label{eq:false-positive-tail}
\end{equation}
<!-- end lemma-7 -->
<!-- proof-12: body_md -->
Fix a non-edge $e=\{x,y\}$ and define
\[
\mathcal B_\tau:=\{e\in\mathcal{PD}_{L+\tau}\},
\qquad 0\le\tau\le R.
\]
For $1\le\tau\le R$, fix the affine coefficients of all stages before final round $\tau$. If $e\notin\mathcal{PD}_{L+\tau-1}$, it cannot appear in any later candidate set. Otherwise, $e$ is a non-defective pair of singleton blocks, and the round uses an independent affine family. By~\cref{lem:survival}, its retention probability is at most $1/24$. Since this round is global stage $L+\tau-1$, we have
\[
\operatorname{E}_{L+\tau-1}[\mathbf 1\{\mathcal B_\tau\}]
\le\frac1{24}\mathbf 1\{\mathcal B_{\tau-1}\}.
\]
Averaging over the earlier families using~\eqref{eq:successive-averaging} gives
\[
\mathbb P[\mathcal B_\tau]\le\frac1{24}\mathbb P[\mathcal B_{\tau-1}].
\]
Applying this inequality repeatedly and using $\mathbb P[\mathcal B_0]\le1$, we obtain
\[
\mathbb P[e\in\widehat E]\le24^{-R}=n^{-\log24},
\]
where $R=\log n$. A union bound over fewer than $n^2$ non-edges proves~\eqref{eq:false-positive-tail}. This bound tends to zero because $\log 24>2$.
<!-- end proof-12 -->
<!-- proof-13: body_md -->
Each of the $M=L+R$ affine families uses $p^2$ queries, all chosen before any outcomes are observed. Since $p^2=\Theta(\bar k)$ and $M=\mathcal O(\log n)$, the design uses $\mathcal O(\bar k\log n)$ non-adaptive queries. This proves~(i).

To prove edge preservation, fix any graph and any choice of affine coefficients, and let $\{x,y\}\in E$. For all sufficiently large $n$, we have $g_0\ge2$. Thus $\mathcal{PD}_0$ contains a pair of distinct blocks whose union contains both endpoints. If the endpoints lie in different blocks, take those two blocks. Otherwise, pair their common block with any other base block.

Every candidate pair whose union contains both endpoints is retained. If its slopes are equal, retention is automatic. Otherwise, its collision test contains the edge $\{x,y\}$ and therefore has positive outcome. At a non-final level, the retained pair generates all six pairs among its four child blocks. At least one generated pair has both endpoints in its union: if the endpoints lie in different child blocks, take those two blocks; otherwise, pair their common child block with another child.

By induction, a candidate pair containing both endpoints in its union exists at every level. At the singleton level, this pair is $\{x,y\}$ itself, and it is retained in every final round. Hence
\[
E\subseteq\widehat E
\]
for every graph and every choice of affine coefficients.

Recovery can therefore fail only if a non-edge survives. The bound in~\cref{lem:no-fp} holds uniformly over $G\in\mathcal T(\epsilon_n)$. Combining it with~\cref{lem:structure} gives
\[
\mathbb P[\widehat E\ne E]
\le\mathbb P[G\notin\mathcal T(\epsilon_n)]
+n^{2-\log 24}
=o(1).
\]
This proves~(ii).

Similarly, the uniform bound in~\cref{thm:workload} and~\cref{lem:structure} give
\begin{align*}
\mathbb P[W>C_3\bar k\log n]
&\le
\mathbb P[G\notin\mathcal T(\epsilon_n)]
+\sup_{G\in\mathcal T(\epsilon_n)}
\mathbb P[W>C_3\bar k\log n\mid G]
=o(1).
\end{align*}
The decoder examines $\sum_{t=0}^{M-1}Z_t\le W$ candidate pairs and attempts at most $6\sum_{t=0}^{L-1}Z_t\le6W$ child-pair insertions. Each examination uses a constant number of field operations and array accesses. Under the computational assumptions in~\cref{sec:problem}, initializing, iterating over, updating, and copying the candidate sets, together with producing the output, takes $\mathcal O(W+M)$ operations. The $M$ term accounts for stage overhead, and $W$ already includes the output size $Z_M$. Since $M=\mathcal O(\log n)$ and $\bar k\to\infty$, the decoding time is $\mathcal O(\bar k\log n)$ with probability $1-o(1)$. This proves~(iii).

<!-- proof-14: body_md -->
We use the binomial upper-tail bound
\begin{equation}
\mathbb P[X\ge t]\le\left(\frac{e\lambda}{t}\right)^t,
\qquad
X\sim\operatorname{Bin}(m,p'),\quad
\lambda=mp',\quad t>0,\quad t\ge\lambda.
\label{eq:bintail}
\end{equation}
The same bound holds with $\lambda$ replaced by any $\lambda'$ satisfying $\lambda\le\lambda'\le t$.

\paragraph{Edge count.}
Set $\epsilon_n:=\bar k^{-1/3}$. Since $|E|$ is binomial with mean $\bar k$, a two-sided Chernoff bound gives, for all sufficiently large $n$,
\[
\mathbb P\!\left[\bigl||E|-\bar k\bigr|>\epsilon_n\bar k\right]
\le2\exp\!\left(-\frac{\epsilon_n^2\bar k}{3}\right)
=2\exp\!\left(-\frac{\bar k^{1/3}}3\right)=o(1).
\]
Thus~\eqref{eq:struct-edges} holds with probability $1-o(1)$.

\paragraph{Internal edges at the base level.}
There are $M_0:=g_0\binom{n/g_0}{2}$ potential edges with both endpoints in the same base block. Hence $H_{g_0}\sim\operatorname{Bin}(M_0,q)$ and
\[
\operatorname{E}[H_{g_0}]
=\bar k\frac{n/g_0-1}{n-1}
\le\frac{\bar k}{g_0}\le\sqrt{\bar k}.
\]
Applying~\eqref{eq:bintail} with $t=3\sqrt{\bar k}$ gives
\[
\mathbb P[H_{g_0}>3\sqrt{\bar k}]
\le\left(\frac e3\right)^{3\sqrt{\bar k}}=o(1),
\]
which proves~\eqref{eq:struct-internal}. Since each level refines the previous one, $H_{g_j}\le H_{g_0}$ for every $j$. Thus, on the same event, every level has at most $3\sqrt{\bar k}$ internally defective blocks.

\paragraph{Uniform crossing degree.}
Fix a level $j$ and a block $B$ of size $n/g_j$. Let $X_B$ count the edges with exactly one endpoint in $B$. Then
\[
X_B\sim\operatorname{Bin}\!\left(
\frac n{g_j}\left(n-\frac n{g_j}\right),q
\right).
\]
For $n\ge3$, its mean satisfies
\[
\lambda_j:=\operatorname{E}[X_B]
\le\frac{qn^2}{g_j}
\le\frac{3\bar k}{g_j}
\le\frac{3\sqrt{\bar k}}{2^j}.
\]
Set $d_*:=24\sqrt{\bar k}$. By~\eqref{eq:bintail}, we have
\[
\mathbb P[X_B\ge d_*]
\le\left(\frac{e\lambda_j}{d_*}\right)^{d_*}
\le\left(\frac{e}{8\cdot2^j}\right)^{d_*}.
\]
Since $g_j<2^{j+1}\sqrt{\bar k}$, a union bound over all blocks and levels gives
\begin{align*}
\mathbb P\!\left[\max_{0\le j\le L}\Delta_{g_j}^{\times}\ge d_*\right]
&\le\sum_{j=0}^L g_j\left(\frac{e}{8\cdot2^j}\right)^{d_*}\\
&\le2\sqrt{\bar k}\left(\frac e8\right)^{d_*}
  \sum_{j=0}^{\infty}2^{-j(d_*-1)}\\
&\le4\sqrt{\bar k}\left(\frac e8\right)^{24\sqrt{\bar k}}
=o(1).
\end{align*}
The geometric sum is at most $2$ once $d_*\ge2$. This proves~\eqref{eq:struct-cross}.

\paragraph{Weighted internal edges over the hierarchy.} The idea of the proof is to choose a deterministic cutoff level so that, with high probability, no internal edges remain at this or any later level. We then apply Bernstein's inequality to the weighted sum over earlier levels, expressed as a sum of independent edge indicators with bounded weights. Define
\begin{equation}
\omega_n:=\sqrt{\log n},
\qquad
b_n:=\min\{\bar k\omega_n,n\},
\label{eq:cutoff-scale}
\end{equation}
and let $h$ be the smallest index in $\{0,\ldots,L\}$ such that $g_h\ge b_n$. These quantities depend only on $n$ and $\bar k$. For all sufficiently large $n$, both $n$ and $\bar k\omega_n$ exceed $g_0$, so $h\ge1$. By the choice of $h$,
\begin{equation}
g_h=2g_{h-1}<2b_n\le2\bar k\omega_n.
\label{eq:cutoff-upper}
\end{equation}
If $h=L$, then $H_{g_h}=0$ because the blocks are singletons. If $h<L$, then $b_n=\bar k\omega_n$ and
\[
\operatorname{E}[H_{g_h}]
=\bar k\frac{n/g_h-1}{n-1}
\le\frac{\bar k}{g_h}\le\frac1{\omega_n}.
\]
Markov's inequality therefore gives, in both cases,
\begin{equation}
\mathbb P[H_{g_h}>0]\le\frac1{\omega_n}=o(1).
\label{eq:cutoff-empty}
\end{equation}

For each potential edge $e=\{x,y\}\in\binom{[n]}2$, define
\begin{equation}
w_h(e):=\sum_{j=0}^{h-1}g_j
\mathbf 1\{x,y\text{ lie in the same level-}j\text{ block}\},
\qquad
\xi_e:=\mathbf 1\{e\in E\}.
\label{eq:cutoff-weight}
\end{equation}
Changing the order of summation gives
\begin{equation}
K_h:=\sum_{e\in\binom{[n]}2}\xi_e w_h(e)
=\sum_{j=0}^{h-1}g_jH_{g_j}.
\label{eq:cutoff-sum}
\end{equation}
The weights are deterministic and satisfy
\begin{equation}
0\le w_h(e)\le\sum_{j=0}^{h-1}g_j=g_h-g_0<g_h.
\label{eq:cutoff-weight-bound}
\end{equation}

To compute the average weight, let $U$ be a uniformly chosen element of $\binom{[n]}2$, and write $\mathbb P_U$ and $\operatorname{E}_U[\cdot]$ for probability and expectation over $U$ alone. At a level with $g$ blocks,
\begin{equation}
\mathbb P_U[U\text{ is internal at this level}]
=\frac{g\binom{n/g}{2}}{\binom n2}
=\frac{n/g-1}{n-1}\le\frac1g.
\label{eq:internal-prob}
\end{equation}
It follows that $\operatorname{E}_U[w_h(U)]\le h\le\log n$. Hence
\begin{equation}
\mu_h:=\operatorname{E}[K_h]=q\sum_e w_h(e)
=\bar k\,\operatorname{E}_U[w_h(U)]\le\bar k\log n.
\label{eq:cutoff-mean}
\end{equation}

Since the cutoff and the weights are deterministic, the variables $\xi_e w_h(e)$ are independent. Their total variance satisfies
\begin{align}
V_h&:=\sum_e\operatorname{Var}(\xi_e w_h(e))
\le q\sum_e w_h(e)^2
\le qg_h\sum_e w_h(e)
=g_h\mu_h
\le\bar k g_h\log n.
\label{eq:cutoff-var}
\end{align}
Each summand differs from its mean by at most $g_h$. Bernstein's inequality therefore gives, for $t>0$,
\begin{equation}
\mathbb P[K_h-\mu_h\ge t]
\le\exp\!\left(-\frac{t^2}{2V_h+\frac23g_ht}\right).
\label{eq:cutoff-bernstein-general}
\end{equation}
Taking $t=2\bar k\log n$ and using~\eqref{eq:cutoff-mean}, \eqref{eq:cutoff-var}, and~\eqref{eq:cutoff-upper}, we obtain
\begin{align}
\mathbb P[K_h>3\bar k\log n]
\le\exp\!\left(-c\frac{\bar k\log n}{g_h}\right)
\le\exp\!\left(-c'\frac{\log n}{\omega_n}\right)
=\exp(-c'\sqrt{\log n})=o(1)
\label{eq:cutoff-bernstein}
\end{align}
for absolute constants $c,c'>0$.

The Bernstein bound above is unconditional. On the event $H_{g_h}=0$, every finer level also has no internal edges. Thus
\[
\sum_{j=0}^{L-1}g_jH_{g_j}=K_h
\]
on this event. Combining~\eqref{eq:cutoff-empty} and~\eqref{eq:cutoff-bernstein} gives
\begin{align*}
\mathbb P\!\left[\sum_{j=0}^{L-1}g_jH_{g_j}>3\bar k\log n\right]
\le\mathbb P[H_{g_h}>0]+\mathbb P[K_h>3\bar k\log n]
\le\frac1{\sqrt{\log n}}+\exp(-c'\sqrt{\log n})
=o(1).
\end{align*}
This proves~\eqref{eq:struct-weighted}.

\paragraph{Combining the estimates.}
A union bound over the four failure events shows that $G\in\mathcal T(\epsilon_n)$ with probability $1-o(1)$. On this event, $|E|\le2\bar k$ for all sufficiently large $n$. By~\cref{lem:Dg-det},
\[
\sum_{j=0}^{L-1}D_{g_j}
\le L|E|+\sum_{j=0}^{L-1}g_jH_{g_j}
\le5\bar k\log n.
\]
This also proves~\eqref{eq:sum-Dg} and completes the proof.

By the definition of $\Omega_{ef}$,
\begin{align}
\Omega_{ef}^c\subseteq{}&
\mathcal C_e^c\cup\mathcal C_f^c
\cup\bigl(\mathcal C_e\cap\mathcal C_f\cap\{r_e^*=r_f^*\}\bigr)
{}\cup\bigcup_{w\in D}\{w\in T_f\}
\cup\bigcup_{w\in f\setminus e}\{w\in T_e\}.
\label{eq:covering}
\end{align}
Indeed, if both pairs have distinct slopes, separation fails only if the collision points have the same first coordinate or one of the required block exclusions fails.

Each equal-slope event has probability $1/p$. For $w\in D$, we have $w\notin f$, so~\cref{prop:composition} gives
\[
\mathbb P[w\in T_f\mid\mathcal C_f,\ T_f=(r',s')]=\frac1p
\]
for every possible collision point. By the convention $\{w\in\dagger\}=\varnothing$, the event $\{w\in T_f\}$ implies $\mathcal C_f$. The law of total probability therefore gives
\begin{align*}
\mathbb P[w\in T_f]
&=\sum_{(r',s')\in\mathbb F_p^2}
\mathbb P[\mathcal C_f,\ T_f=(r',s')]
\mathbb P[w\in T_f\mid\mathcal C_f,\ T_f=(r',s')]
=\frac1p\mathbb P[\mathcal C_f]\le\frac1p.
\end{align*}
Terms with zero-probability conditioning events are omitted. Exchanging $e$ and $f$ gives the same bound for each $w\in f\setminus e$.

For disjoint pairs,~\cref{prop:first_cordinate}(ii) gives
\[
\mathbb P[r_e^*=r_f^*\mid\mathcal C_e\cap\mathcal C_f]=\frac1p.
\]
If the pairs share a block $u$,~\cref{prop:first_cordinate}(iii) gives the same probability when $(a_u,b_u)$ is also fixed. Averaging over these coefficients conditional on $\mathcal C_e\cap\mathcal C_f$ gives the same equality. Thus, in both cases,
\[
\mathbb P[\mathcal C_e\cap\mathcal C_f\cap\{r_e^*=r_f^*\}]
=\frac1p\mathbb P[\mathcal C_e\cap\mathcal C_f]\le\frac1p.
\]
There are $3+|D|+|f\setminus e|=3+|e\triangle f|$ events on the right-hand side of~\eqref{eq:covering}. A union bound now gives
\[
\mathbb P[\Omega_{ef}^c]\le\frac{3+|e\triangle f|}{p}\le\frac7p.
\qedhere

For $Q\subseteq[g]\setminus f$, let 
\[
\Phi_f(Q):=Y\!\left(\bigcup_{x\in f\cup Q}\mathcal G_x\right)
\]
be the outcome of the query formed by the blocks in $f\cup Q$. Since $G$ is fixed, this outcome is deterministic. Adding blocks cannot remove an edge, so
\begin{equation}
Q\subseteq Q'\quad\Longrightarrow\quad
\Phi_f(Q)\le\Phi_f(Q').
\label{eq:Phi-monotone}
\end{equation}
For a finite set $X$ and $B\subseteq X$, define
\[
\mu_X(B):=\left(\frac1p\right)^{|B|}
\left(1-\frac1p\right)^{|X|-|B|}.
\]
This is the probability of selecting exactly $B$ when each element of $X$ is included independently with probability $1/p$. In particular, $\sum_{B\subseteq X}\mu_X(B)=1$. By~\cref{prop:composition} and the disjoint decomposition $[g]\setminus f=D\cup R_0$,
\begin{equation}
\rho_f=\sum_{B\subseteq R_0}\mu_{R_0}(B)
\sum_{J\subseteq D}\mu_D(J)\Phi_f(B\cup J).
\label{eq:rho-expanded}
\end{equation}

\paragraph{Partition of the separation event.}
Let $\mathbf C$ be the vector of coefficient pairs of the special blocks. Every condition defining $\Omega_{ef}$ depends only on $\mathbf C$. Let $\mathcal S$ be the set of possible values of $\mathbf C$ that satisfy these conditions. Then
\[
\Omega_{ef}=\bigcup_{\mathbf c\in\mathcal S}\{\mathbf C=\mathbf c\},
\]
where the events in the union are disjoint. Fix $\mathbf c\in\mathcal S$ and write the corresponding collision points as
\[
T_e(\mathbf c)=(r_{\mathbf c},s_{\mathbf c}),
\qquad
T_f(\mathbf c)=(r'_{\mathbf c},s'_{\mathbf c}).
\]
By the definition of $\mathcal S$,
\begin{equation}
r_{\mathbf c}\ne r'_{\mathbf c},
\qquad
w\notin T_f(\mathbf c)\ (w\in D),
\qquad
w\notin T_e(\mathbf c)\ (w\in f\setminus e).
\label{eq:special-separation}
\end{equation}
For $w\in R_0$, define the membership indicators
\[
U_w^{\mathbf c}:=\mathbf 1\{P_w(r_{\mathbf c})=s_{\mathbf c}\},
\qquad
V_w^{\mathbf c}:=\mathbf 1\{P_w(r'_{\mathbf c})=s'_{\mathbf c}\}.
\]
For each $\mathbf z=(z_w)_{w\in R_0}\in\{0,1\}^{R_0}$, define
\[
A_{\mathbf c,\mathbf z}:=
\{\mathbf C=\mathbf c\}\cap
\bigcap_{w\in R_0}\{U_w^{\mathbf c}=z_w\}.
\]
As $\mathbf c$ and $\mathbf z$ vary, these events form a disjoint partition of $\Omega_{ef}$. Each event has positive probability: after the special coefficients are fixed, the remaining coefficients are still independent and uniform, and every prescribed membership pattern has positive probability.

On $A_{\mathbf c,\mathbf z}$, the blocks included in the first test are exactly
\[
e\cup\{w\in R_0:z_w=1\}.
\]
The blocks in $e$ belong to their collision test, the blocks in $f\setminus e$ are excluded by~\eqref{eq:special-separation}, and $\mathbf z$ determines all remaining memberships. Since $G$ is fixed and $\mathcal C_e$ holds, both $Y_{T_e}$ and $I_e$ are fixed on this event.

\paragraph{Conditional distribution of the second test.}
The coefficient pairs of the blocks in $R_0$ are independent of one another and of $\mathbf C$. For fixed $\mathbf c$, the variables $U_w^{\mathbf c}$ and $V_w^{\mathbf c}$ depend only on the coefficients of block $w$. Therefore, for every $\mathbf y\in\{0,1\}^{R_0}$,
\begin{align}
&\mathbb P[(V_w^{\mathbf c})_{w\in R_0}=\mathbf y\mid A_{\mathbf c,\mathbf z}]
=
\prod_{w\in R_0}
\frac{\mathbb P[U_w^{\mathbf c}=z_w,\ V_w^{\mathbf c}=y_w]}
{\mathbb P[U_w^{\mathbf c}=z_w]}
=\prod_{w\in R_0}
\mathbb P[V_w^{\mathbf c}=y_w\mid U_w^{\mathbf c}=z_w].
\label{eq:cell-factorization}
\end{align}
All denominators are positive. Since $r_{\mathbf c}\ne r'_{\mathbf c}$,~\cref{prop:two-point} implies that $U_w^{\mathbf c}$ and $V_w^{\mathbf c}$ are independent Bernoulli variables with parameter $1/p$. Hence~\eqref{eq:cell-factorization} gives
\begin{equation}
\mathbb P[(V_w^{\mathbf c})_{w\in R_0}=\mathbf y\mid A_{\mathbf c,\mathbf z}]
=\mu_{R_0}(\{w:y_w=1\}).
\label{eq:cell-law}
\end{equation}
Thus, after fixing the special coefficients and all memberships in the first test, the blocks in $R_0$ are still included independently in the second test with probability $1/p$.

\paragraph{Bounding the second retention probability.}
On $A_{\mathbf c,\mathbf z}$, both collision events hold. The second test contains exactly the blocks in $f\cup B$, where
\[
B:=\{w\in R_0:V_w^{\mathbf c}=1\}.
\]
The blocks in $D$ are excluded by~\eqref{eq:special-separation}. Since $\mathcal C_f$ holds, the pair $f$ is retained exactly when this test is positive. By~\eqref{eq:cell-law},
\[
\mathbb P[\mathcal E_f\mid A_{\mathbf c,\mathbf z}]
=\sum_{B\subseteq R_0}\mu_{R_0}(B)\Phi_f(B).
\]
For every $B\subseteq R_0$ and $J\subseteq D$, monotonicity gives $\Phi_f(B)\le\Phi_f(B\cup J)$. Averaging first over $J$ and then over $B$, we obtain
\begin{align}
\mathbb P[\mathcal E_f\mid A_{\mathbf c,\mathbf z}]
&\le\sum_{B\subseteq R_0}\mu_{R_0}(B)
\sum_{J\subseteq D}\mu_D(J)\Phi_f(B\cup J)
=\rho_f.
\label{eq:cell-domination}
\end{align}

Let $\Gamma$ be the set of indices $(\mathbf c,\mathbf z)$ for which $A_{\mathbf c,\mathbf z}\subseteq\mathcal E_e$. Since $I_e$ is constant on each event in the partition,
\[
\mathcal E_e\cap\Omega_{ef}
=\bigcup_{(\mathbf c,\mathbf z)\in\Gamma}A_{\mathbf c,\mathbf z}
\]
is a disjoint union. Using~\eqref{eq:cell-domination}, the law of total probability, and~\eqref{eq:rho-le-q}, we get
\begin{align*}
\mathbb P[\mathcal E_e\cap\mathcal E_f\cap\Omega_{ef}]
=\sum_{(\mathbf c,\mathbf z)\in\Gamma}
\mathbb P[A_{\mathbf c,\mathbf z}]
\mathbb P[\mathcal E_f\mid A_{\mathbf c,\mathbf z}]
\le\rho_f\sum_{(\mathbf c,\mathbf z)\in\Gamma}
\mathbb P[A_{\mathbf c,\mathbf z}]
=\rho_f\,\mathbb P[\mathcal E_e\cap\Omega_{ef}] \le q_eq_f.
\end{align*}
The argument applies to both disjoint pairs and pairs sharing a block, proving~\eqref{eq:product-bound}.
