<!-- proposition-1: body_md -->
\label{optimal_transport_dro_dual}
Let $h_{\boldsymbol{\beta}}: \mathcal{Z} \rightarrow \mathbb{R}$ be a measurable function with $\mathbb{E}_{\widehat\mathbb{P}_{n}}[h_{\boldsymbol{\beta}}] > -\infty$. Then the robust risk in \eqref{eq_normalised_robust_risk} equals:
\begin{equation}\label{eq:robust_risk_intermediate_form}
{V}_{\delta}(\boldsymbol{\beta})\ = \min_{\lambda\ge 0} \left(\lambda \delta^p + \mathbb{E}_{\boldsymbol{z} \sim \widehat\mathbb{P}_{n}}\Big[\sup_{\boldsymbol{z}'\in \mathcal{Z}} \big(h_{\boldsymbol{\beta}}(\boldsymbol{z}') - \lambda \|\boldsymbol{z}'-\boldsymbol{z}\|^p\big)\Big] \right).
\end{equation}
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
\label{remark_p_less_2_blows_up}
We exclude $1 \leq p <2$ since in this case the robust risk $V_{\delta}(\boldsymbol{\beta})$ becomes infinite. The reason is that the squared loss increases faster than the penalty for moving probability mass, resulting in an unbounded robust risk. See the appendix for a detailed derivation.
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{th_supremum_robust_risk}
The robust risk of Wasserstein DRO linear regression ($2 \leq p \leq \infty$) equals
\begin{equation}\label{supremum_form_linear_regression}
V_{\delta}(\boldsymbol{\beta}) = \sup_{t_i \geq 0, \; \|\boldsymbol{t}\|_p \leq n^{1/p} \delta} \frac{1}{n} \sum_{i=1}^n (|\boldsymbol{x}_i^{\top}\boldsymbol{\beta}-y_i|+t_i \|\boldsymbol{\beta}\|_*)^2.
\end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{th_special_cases}
Wasserstein DRO linear regression with $p=2$ has robust risk $V_{2,\delta}(\boldsymbol{\beta}) = \left ( \sqrt{\frac{1}{n}\sum_{i=1}^n (\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i)^2}+\delta \|\boldsymbol{\beta}\|_* \right)^2$, and thus corresponds to square-root Lasso by taking $\| \cdot \|=\| \cdot \|_{\infty}$, 
while $p = \infty$ corresponds to adversarial linear regression with the continuous limit
$V_{\infty,\delta}(\boldsymbol{\beta}) = \lim_{p \rightarrow \infty} V_{p,\delta}(\boldsymbol{\beta}) =
    \sum_{i=1}^n \sup_{\|\boldsymbol{v}_i\| \leq \delta} ( (\boldsymbol{x}_i+\boldsymbol{v}_i)^T \boldsymbol{\beta} - y_i )^2$.
<!-- end corollary-1 -->
<!-- remark-2: body_md -->
Weaker versions of \Cref{th_supremum_robust_risk} are described in other papers.
The right-hand side of \eqref{supremum_form_linear_regression} is shown to lower-bound the robust risk~\cite[Theorem 6]{kuhn_wasserstein_2019}, but that work does not explicitly show when equality holds. For $p=\infty$, \citet[Theorem 2]{zhang_short_2024} proves this result in general, while \citet[Proposition 6.16]{kuhn_distributionally_2025} shows that equality holds and a saddle-point exists, but only for compact sets. We show that equality in fact holds for all $2 \leq p \leq \infty$ for linear regression. See also \citet[Corollary 2(iii)]{gao2023distributionally} and \citet[Proposition 2]{an2021generalization} for results that only hold~approximately.
<!-- end remark-2 -->

<!-- theorem-2: body_md -->
\label{improved_in_sample_error_decay}
                                                                                                                       
    Assume $\delta > \bar{\delta} := 2\frac{\|X^{\top} \boldsymbol{\varepsilon}\|}{n^{1/p}\|\boldsymbol{\varepsilon}\|_q}$. Then
    \begin{equation}\label{improved_in_sample_error_decay_eq}
        \frac{1}{n}\|X \widehat{\Delta}\|_2^2 \leq 6 \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}}\delta\|\boldsymbol{\beta}^*\|_*+2\delta^2 \|\boldsymbol{\beta}^*\|_*^2.
    \end{equation}
    In particular, let $\boldsymbol{\varepsilon} \sim N(0,\sigma^2 I_n)$ and assume each entry of $X$ is bounded by $M>0$. Then, with high probability (independent of the Wasserstein parameter $p$):
    \begin{equation*}
        \frac{1}{n}\|X \widehat{\Delta}\|_2^2 \in O(\delta).
    \end{equation*}
    That is, the error rate coincides with $\delta$ (as long as $\delta> \bar{\delta}$). In particular, let $c>0$ be the norm-equivalence constant between $\|\cdot \|_\infty$ and $\| \cdot \|$, that is, $\|\cdot \|_\infty \leq c\| \cdot \|$. Then, by setting $\delta = K M \sqrt{ \frac{\log(d/\gamma)}{n}}$ with $K > 2c \sqrt{\pi}$, we get with probability greater than $1-4\gamma$:
                                                                                                                                                                       
    \begin{equation}\label{eq_improved_in_sample_error_decay_rate}
    \frac{1}{n}\|X \widehat{\Delta}\|_2^2 \in O(n^{-1/2}).
    \end{equation}

\label{def_restricted_eigenvalue_condition_v2}
The matrix $X \in \mathbb{R}^{n \times d}$ satisfies the restricted eigenvalue (RE) condition if there exists a positive constant $\kappa = \kappa(s,l)>0$ such that 
\begin{equation*}

\end{equation*}
where $S$ is some subset of $\{1,2,\dots,d\}$.

<!-- theorem-3: body_md -->
\label{thm_fast_rate_main_theorem_v2}
Let $\boldsymbol{\beta}^*$ have $s$ nonzero entries indexed by $S \subseteq \{1,\dots,d\}$ with $s = |S|$, assume $X$ is in $\mathrm{RE}(s,3)$ with positive constant $\kappa = \kappa(s,3)$ (same for all $n$), and consider $\| \cdot \| = \| \cdot \|_\infty$. Then with $\delta > \bar{\delta} := 2\frac{\|X^{\top} \boldsymbol{\varepsilon}\|}{n^{1/p} \|\boldsymbol{\varepsilon}\|_q}$, we have
\begin{equation}\label{eq_fast_rate_main_theorem_v2_bound}
    \frac{1}{n}  \| X \widehat{\Delta} \|_2^2 \leq \delta^2 \cdot \max \left \{ 4 C^2 \|\boldsymbol{\beta}^*\|_1^2, \frac{16 B^2 s^2}{\kappa^2} \right \}
\end{equation}
where $B = 3 \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} + (2MC+\delta C+\delta) \|\boldsymbol{\beta}^*\|_*$ and $C = (  3+\frac{n^{1/q} \delta}{\| \boldsymbol{\varepsilon}\|_q} \|\boldsymbol{\beta}^*\|_* )$ and each entry of $X$ is bounded by $M>0$. In particular, for $\boldsymbol{\varepsilon} \sim N(0,\sigma^2 I_n)$, we have with high probability that $B$ and $C$ are bounded by constants and therefore (independent of the Wasserstein parameter $p$)
\begin{equation}
    \frac{1}{n} \|X \widehat{\Delta} \|_2^2 \in O \left (\delta^2 \right).
\end{equation}
Thus, setting $\delta$ as in \Cref{improved_in_sample_error_decay}, we get the fast rate with probability greater than $1-5\gamma$:
\begin{equation}
    \frac{1}{n} \|X \widehat{\Delta} \|_2^2 \in O\left ( 1/n \right ).
\end{equation}

<!-- lemma-3: body_md -->
\label{lemma_robust_risk_1D_formulation}
The robust risk of Wasserstein DRO linear regression for $1 \leq p<\infty$~equals
\begin{equation}\label{eq_lemma_robust_risk_1D_formulation}
  n V_\delta({\boldsymbol{\beta}})
  = \min_{\lambda \ge 0}
    \left[
      n\delta^p\lambda
      + \sum_{i=1}^{n} \sup_{t_i \ge 0}
        \Big\{ \big( \lvert r_i \rvert + t_i \lVert {\boldsymbol{\beta}} \rVert_* \big)^2 - \lambda t_i^p \Big\}
    \right].
\end{equation}
with $r_i := \boldsymbol{x}_i^\top\boldsymbol{\beta}-y_i$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
We start from \Cref{optimal_transport_dro_dual}, where we note that the expectation in \eqref{eq:robust_risk_intermediate_form} turns into a sum due to the empirical distribution $\widehat\mathbb{P}_{n} := \frac{1}{n} \sum_{i=1}^n \delta_{\boldsymbol{z}_i}$, yielding:
\begin{equation}\label{eq_towards_scalar_problem_start}
  n V_\delta(\boldsymbol{\beta}) = 
  \min_{\lambda \ge 0}
    \left[
      n\delta^p \lambda
      + \sum_{i=1}^{n} \sup_{\boldsymbol{z}' \in \mathcal{Z}}
        \Big\{ h_{\boldsymbol{\beta}}(\boldsymbol{z}') - \lambda \|\boldsymbol{z}'-\boldsymbol{z}_i\| \Big\}
    \right].
\end{equation}
In our case, $\mathcal{Z} = \mathbb{R}^{d+1}$ and $h_{\boldsymbol{\beta}} : \mathbb{R}^{d+1} \to \mathbb{R}$ is the square loss combined with the linear predictor model, i.e.,
for $\boldsymbol{z} = (\boldsymbol{x},y)$, we have $h_{\boldsymbol{\beta}}(\boldsymbol{z}) = \big({\boldsymbol{\beta}}^{\top} \boldsymbol{x} - y\big)^2$, and $\|\boldsymbol{z}'-\boldsymbol{z}\|$ with $z=(\boldsymbol{x},y)$ and $z' = (\boldsymbol{x}',y')$ equals
\begin{equation*}
      \|\boldsymbol{z}'-\boldsymbol{z}\| =
  \begin{cases}
    \lVert \boldsymbol{x}' - \boldsymbol{x} \rVert ^p & \text{if } y = y', \\[2pt]
    +\infty & \text{otherwise.}
  \end{cases}
\end{equation*}
Since $\|\boldsymbol{z}'-\boldsymbol{z}\| = \infty$ whenever $y \neq y'$, the supremum in \eqref{eq_towards_scalar_problem_start} never moves $\boldsymbol{z}'$ away from $\boldsymbol{z}_i$ in the $y$-coordinate.\footnote{Indeed, such a deviation would let the outer minimization in \eqref{eq_towards_scalar_problem_start} pick any $\lambda>0$, resulting in an $-\infty$ cost due to $\|\boldsymbol{z}'-\boldsymbol{z}\| = \infty$, which is strictly suboptimal for the inner supremum (since the zero deviation $\boldsymbol{z}' = \boldsymbol{z}$ results in a bounded cost).} In other words, we have that
\begin{equation*}
  n V_\delta({\boldsymbol{\beta}})
  = \min_{\lambda \ge 0}
    \left[
      n\delta^p \lambda
      + \sum_{i=1}^{n} \sup_{\boldsymbol{x}' \in \mathbb{R}^d}
        \Big\{ \big({\boldsymbol{\beta}}^{\top} \boldsymbol{x}' - y_i\big)^2 - \lambda \lVert \boldsymbol{x}' - \boldsymbol{x}_i \rVert ^p \Big\}
    \right].
\end{equation*}
Now center the problem by letting $\boldsymbol{u}_i = \boldsymbol{x}' - \boldsymbol{x}_i$. Then
\[
  {\boldsymbol{\beta}}^{\top} \boldsymbol{x}' - y_i
  = {\boldsymbol{\beta}}^{\top}(\boldsymbol{u}_i + \boldsymbol{x}_i) - y_i
  = {\boldsymbol{\beta}}^{\top} \boldsymbol{x}_i - y_i + {\boldsymbol{\beta}}^{\top} \boldsymbol{u}_i
  = r_i + {\boldsymbol{\beta}}^{\top} \boldsymbol{u}_i ,
\]
and therefore
\[
  \big({\boldsymbol{\beta}}^{\top} \boldsymbol{x}' - y_i\big)^2 - \lambda \lVert \boldsymbol{x}' - \boldsymbol{x}_i \rVert ^p
  = \big(r_i + {\boldsymbol{\beta}}^{\top} \boldsymbol{u}_i \big)^2 - \lambda \lVert \boldsymbol{u}_i \rVert ^p .
\]
Thus, we have
\begin{equation*}
  n V_\delta({\boldsymbol{\beta}})
  = \min_{\lambda \ge 0}
    \left[
      n\delta^p \lambda
      + \sum_{i=1}^{n} \sup_{\boldsymbol{u} \in \mathbb{R}^d}
        \Big\{ \big(r_i + {\boldsymbol{\beta}}^{\top} \boldsymbol{u}\big)^2 - \lambda \lVert {\boldsymbol{u}} \rVert ^p \Big\}
    \right],
\end{equation*}
where we drop the subscript on $\boldsymbol{u}$ for convenience. We note that the penalty $\lambda \lVert {\boldsymbol{u}} \rVert ^p$ depends only on the magnitude of $\boldsymbol{u}$, and thus its
direction is irrelevant to the cost. To decompose direction and magnitude, let $\boldsymbol{u} = t \boldsymbol{v}$ with $t \ge 0$ and
$\lVert \boldsymbol{v} \rVert = 1$. Then, the supremum can be split into two parts:
\[
  \sup_{t \ge 0} \; \sup_{\lVert \boldsymbol{v} \rVert  = 1}
    \Big\{ \big(r_i + t{\boldsymbol{\beta}}^{\top} \boldsymbol{v}\big)^2 - \lambda t^p \Big\}
  = \sup_{t \ge 0}
    \left[ \sup_{\lVert \boldsymbol{v} \rVert  = 1} \Big\{ \big(r_i + t{\boldsymbol{\beta}}^{\top} \boldsymbol{v}\big)^2 \Big\} - \lambda t^p \right].
\]
The remaining inner problem $\sup_{\lVert \boldsymbol{v} \rVert  = 1} (r_i + t{\boldsymbol{\beta}}^{\top} \boldsymbol{v})^2$ can be
solved in closed form:
\begin{equation*}
    \sup_{\lVert \boldsymbol{v} \rVert  = 1} \Big\{ \big(r_i + t{\boldsymbol{\beta}}^{\top} \boldsymbol{v}\big)^2 \Big\} = 
    \sup_{\lVert \boldsymbol{v} \rVert  = 1} \Big\{ \big(|r_i| + t{\boldsymbol{\beta}}^{\top} \boldsymbol{v}\big)^2 \Big\} =
     \big(|r_i| + t \sup_{\lVert \boldsymbol{v} \rVert  = 1} \{ {\boldsymbol{\beta}}^{\top} \boldsymbol{v} \} \big)^2 =
     \big( \lvert r_i \rvert + t \lVert {\boldsymbol{\beta}} \rVert_* \big)^2,
\end{equation*}
where the first equality is due to symmetry, the second holds since $w \mapsto (a+tw)^2$ is increasing for $a,t \geq 0$, and the third holds since $\lVert {\boldsymbol{\beta}} \rVert_* = \sup_{\lVert \boldsymbol{v} \rVert  = 1} \{ {\boldsymbol{\beta}}^{\top} \boldsymbol{v} \}$. This leaves us with the scalar~problem
\begin{equation*}
  n V_\delta({\boldsymbol{\beta}})
  = \min_{\lambda \ge 0}
    \left[
      n\delta^p\lambda
      + \sum_{i=1}^{n} \sup_{t \ge 0}
        \Big\{ \big( \lvert r_i \rvert + t \lVert {\boldsymbol{\beta}} \rVert_* \big)^2 - \lambda t^p \Big\}
    \right],
\end{equation*}
completing the proof.

We first consider the case $p < \infty$. We want to show that
\begin{equation}
F(\boldsymbol{\beta}) := \sup_{t_i \geq 0, \sum_{i=1}^n t_i^p \leq n \delta^p} \sum_{i=1}^n \big(|r_i|+t_i \|\boldsymbol{\beta}\|_*\big)^2.                                                                                     
\label{eq:theorem_2_dual_cleaned}
\end{equation}
is equal to $nV_{\delta}(\boldsymbol{\beta})$. For $\delta=0$, we trivially have $F(\boldsymbol{\beta}) = nV_{\delta}(\boldsymbol{\beta}) = \sum_{i=1}^n r_i^2$ using \eqref{eq_lemma_robust_risk_1D_formulation}. 
        
                                                                                                              
              
                      
                        
          
                      
                                       
                                                    
                
                                       
                         
                                     
                                 
                                          
           
           
                                       
                         
                                     
                                 
                                          
             
                                                                            
                                                 
            
                                                                                                                                                                                                                                                                                    
   
For the case $\delta>0$, we show equality using standard strong duality theory \cite[Chapter 5]{Boyd_Vandenberghe_2004_1}. More precisely, consider \eqref{eq:theorem_2_dual_cleaned}, and apply the change of variable $\nu_i = t_i^p$, leading to
\begin{align*}
    F(\boldsymbol{\beta}) = 
    \sup_{\nu_i \geq 0, \sum_{i=1}^n \nu_i \leq n \delta^p} \sum_{i=1}^n \big(|r_i|+\nu_i^{1/p} \|\boldsymbol{\beta}\|_*\big)^2 = \\
    \sup_{\nu_i \geq 0, \sum_{i=1}^n \nu_i \leq n \delta^p} \sum_{i=1}^n
    \underbrace{r_i^2 + 2 \lvert r_i \rvert \|\boldsymbol{\beta}\|_* \nu_i^{1/p} + \|\boldsymbol{\beta}\|_*^2 \nu_i^{2/p}}_{\phi_i(\nu_i) :=} =
    \sup_{\nu_i \geq 0, \sum_{i=1}^n \nu_i \leq n \delta^p} \sum_{i=1}^n \phi_i(\nu_i).
\end{align*}
Here, $\phi_i(\nu_i) = r_i^2 + 2 \lvert r_i \rvert \|\boldsymbol{\beta}\|_* \nu_i^{1/p} + \|\boldsymbol{\beta}\|_*^2 \nu_i^{2/p}$ is concave in $\nu_i \geq 0$ since $p \geq 2$ implies $1/p \in [0,1/2]$ and $2/p \in [0,1]$. In other words, $F(\boldsymbol{\beta})$ is a concave maximization over an affine domain. Equivalently, to work with the more standard convex setting \cite[Chapter 5]{Boyd_Vandenberghe_2004_1}, we have that
\begin{equation}\label{eq_th2_dual_convex_form}
    \tilde{F}(\boldsymbol{\beta}) := -F(\boldsymbol{\beta}) = \inf_{\nu_i \geq 0, \sum_{i=1}^n \nu_i \leq n \delta^p} \sum_{i=1}^n -\phi_i(\nu_i)
\end{equation}
is a convex minimization over an affine domain. The dual problem of \eqref{eq_th2_dual_convex_form} with respect to the constraint $\sum_{i=1}^n \nu_i \leq n \delta^p$ (leaving $\nu_i \geq 0$ as part of the domain) is\footnote{We compactly write $\boldsymbol{\nu} \geq 0$ meaning $\nu_i \geq 0$ for each $i$.}
\begin{align}\label{eq_th2_dual2_convex_form}
    \sup_{\lambda \geq 0} \inf_{\boldsymbol{\nu} \geq 0} L(\nu,\lambda) := \sup_{\lambda \geq 0} \inf_{\boldsymbol{\nu} \geq 0} \left [ \sum_{i=1}^n (-\phi_i(\nu_i)) + \lambda \left ( \sum_{i=1}^n (\nu_i)-n\delta^{p} \right ) \right]
\end{align}
Moreover, $\nu_i = 0$ yields $\sum_{i=1}^n \nu_i =0 < n \delta^p$ for $\delta>0$, so Slater's condition holds. Therefore, strong duality holds \cite[Chapter 5]{Boyd_Vandenberghe_2004_1} and hence \eqref{eq_th2_dual_convex_form} is equal to \eqref{eq_th2_dual2_convex_form}. Equivalently,
\begin{align*}
    F(\boldsymbol{\beta}) = -\sup_{\lambda \geq 0} \inf_{\boldsymbol{\nu} \geq 0} \left [ \sum_{i=1}^n (-\phi_i(\nu_i)) + \lambda \left ( \sum_{i=1}^n (\nu_i)-n\delta^{p} \right ) \right] = \\
    \inf_{\lambda \geq 0} \sup_{\boldsymbol{\nu} \geq 0} \left [ \sum_{i=1}^n (\phi_i(\nu_i)) + \lambda \left ( n\delta^{p} - \sum_{i=1}^n \nu_i \right ) \right] =
    \inf_{\lambda \geq 0} \left [ n\delta^{p} \lambda +  \sum_{i=1}^n \sup_{\nu_i \geq 0} (\phi_i(\nu_i)-\lambda \nu_i) \right] = \\
    \inf_{\lambda \geq 0} \left [ n\delta^{p} \lambda +  \sum_{i=1}^n \sup_{t_i \geq 0} (\phi_i(t_i^p)-\lambda t_i^p) \right] =
    \inf_{\lambda \geq 0} \left [ n\delta^{p} \lambda +  \sum_{i=1}^n \sup_{t_i \geq 0} \left \{ \big( \lvert r_i \rvert + t_i \lVert {\boldsymbol{\beta}} \rVert_* \big)^2-\lambda t_i^p) \right \} \right] =\\  
    n V_\delta(\boldsymbol{\beta})
\end{align*}
where the last equality is due to \eqref{eq_lemma_robust_risk_1D_formulation}. This completes the proof for $p<\infty$.

For $p=\infty$, we have by Theorem 2 in \citet{zhang_short_2024} that
\begin{equation*}
    n V_\delta(\boldsymbol{\beta}) = \sum_{i=1}^n \sup_{\|\boldsymbol{v}_i\| \leq \delta} ( (\boldsymbol{x}_i+\boldsymbol{v}_i)^T \boldsymbol{\beta} - y_i )^2
\end{equation*}
which can be rewritten to the desired form using a dual norm argument, see, e.g., Proposition 1 in~\citet{ribeiro_regularization_2023}. For completeness, we provide this argument here too:

\begin{align*}
\sup_{\|v_i\| \leq \delta} ( (\boldsymbol{x}_i+\boldsymbol{v}_i)^T \boldsymbol{\beta} - y_i )^2 =
\sup_{\|v_i\| \leq \delta} ( \boldsymbol{v}_i^T \boldsymbol{\beta}+ \boldsymbol{x}_i^T \boldsymbol{\beta} - y_i )^2 = 
\sup_{\|v_i\| \leq \delta} ( \boldsymbol{v}_i^T \boldsymbol{\beta}+ |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |)^2 = \\
\sup_{\|v_i\| \leq 1} (\delta \boldsymbol{v}_i^T \boldsymbol{\beta}+ |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |)^2 = 
(\delta \sup_{\|v_i\| \leq 1} \boldsymbol{v}_i^T \boldsymbol{\beta}+ |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |)^2 =
(\delta \|\boldsymbol{\beta}\|_*+ |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |)^2
\end{align*}
and thus
\begin{align*}
    V_{\delta}(\boldsymbol{\beta}) =
    \sum_{i=1}^n (|\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |+\delta \|\boldsymbol{\beta}\|_*)^2  = \sup_{t_i \geq 0, \|\boldsymbol{t}\| \leq \delta} \sum_{i=1}^n  (|\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |+t_i \|\boldsymbol{\beta}\|_*)^2
\end{align*}
completing the proof for $p = \infty$ and hence of \Cref{th_supremum_robust_risk}.

<!-- proof-4: body_md -->
We will use \Cref{th:saddle_point_form} (that we prove latter in the appendix). This result state that 
\[nV_{\delta}(\boldsymbol{\beta}) = \sup_{\boldsymbol{\alpha} \geq 0} K(\boldsymbol{\beta},\boldsymbol{\alpha}) =  \sup_{\boldsymbol{\alpha} \geq 0} \left [ n^{1/2}\delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha} \|_2+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | - \alpha_i^2/4 \right ]\]
 
We start with $p=2$ and compute \eqref{eq:saddle_point_form} noting that $q=2$. For brevity, let $D$ be the vector with entries $D_i =|\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |$. Then:
\begin{align*}
    nV_{\delta}(\boldsymbol{\beta}) 
    &=\sup_{\boldsymbol{\alpha} \geq 0} \left [ n^{1/2}\delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha} \|_2+\langle \boldsymbol{\alpha}, \boldsymbol{D} \rangle - \|\alpha\|_2^2/4 \right ]  \\
    &= \sup_{\boldsymbol{\alpha} \geq 0} \left [ n^{1/2}\delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha} \|_2+\|\boldsymbol{\alpha}\|_2 \|\boldsymbol{D}\|_2  - \|\boldsymbol{\alpha}\|_2^2/4 \right ] \\
     &= \sup_{t \geq 0} \left [ n^{1/2}\delta \|\boldsymbol{\beta}\|_* t+t \|\boldsymbol{D}\|_2  - t^2/4 \right ],
\end{align*}
where we used the Cauchy–Schwarz inequality $\langle \boldsymbol{\alpha}, \boldsymbol{D} \rangle \leq \|\boldsymbol{\alpha}\|_2 \|\boldsymbol{D}\|_2$ with equality if $\boldsymbol{\alpha} = l\cdot \boldsymbol{D}$ for some scalar $l \geq 0$ (note that such an $\boldsymbol{\alpha}$ is indeed feasible since $\boldsymbol{D}\geq0$), and then transformed the supremum to a one-dimensional optimization by setting $t=\|\boldsymbol{\alpha}\|_2$. This one-dimensional optimization is quadratic, with solution
\begin{equation*}
( \|\boldsymbol{D} \|_2+n^{1/2}\delta \|\boldsymbol{\beta}\|_*)^2.
\end{equation*}
Therefore
\begin{equation*}
    V_{\delta}(\boldsymbol{\beta}) = \left ( \frac{1}{n^{1/2}}  \|\boldsymbol{D} \|_2 +\delta \|\boldsymbol{\beta}\|_* \right)^2 = \left ( \sqrt{\frac{1}{n}\sum_{i=1}^n (\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i)^2}+\delta \|\boldsymbol{\beta}\|_* \right)^2,
\end{equation*}
which coincides with square-root Lasso \citep{belloni_squareroot_2011} for $\| \cdot \|_* = \| \cdot \|_1$, i.e., $\| \cdot \| = \| \cdot \|_\infty$. 

To show the adversarial linear regression case, we have by \Cref{th_supremum_robust_risk} that
\begin{align*}
    V_{\infty,\delta}(\boldsymbol{\beta}) = 
    \sup_{t_i \geq 0, \|t_i\| \leq \delta} \sum_{i=1}^n  (|\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i |+t_i \|\boldsymbol{\beta}\|_*)^2 = 
    \sum_{i=1}^n (|\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i |+\delta \|\boldsymbol{\beta}\|_*)^2 = \\
    \sum_{i=1}^n \sup_{\|\boldsymbol{v}_i\| \leq \delta} ( (\boldsymbol{x}_i+\boldsymbol{v}_i)^{\top} \boldsymbol{\beta} - y_i )^2
\end{align*}
where the last step follows by the dual norm argument in the proof of \Cref{th_supremum_robust_risk}. Here, we identify $v_i$ as the adversarial noise on the input $\boldsymbol{x}_i$. In other words, $p$-Wasserstein linear regression with $p = \infty$ coincides with adversarial linear regression. It remains to show that $\lim_{p \rightarrow \infty} V_{p,\delta}(\boldsymbol{\beta}) = V_{\infty,\delta}(\boldsymbol{\beta})$. Again, using \Cref{th:saddle_point_form}, we have that:
\begin{equation*}
    \lim_{p \rightarrow \infty} V_{p,\delta}(\boldsymbol{\beta}) = \lim_{p \rightarrow \infty} \sup_{\boldsymbol{\alpha} \geq 0} K(\boldsymbol{\beta},\boldsymbol{\alpha}) =  \lim_{p \rightarrow \infty} \sup_{\boldsymbol{\alpha} \geq 0} \left [ n^{1/p} \delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha} \|_q+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | - \alpha_i^2/4 \right ].
\end{equation*}
We want to swap the limit and the supremum. To this end, fix $\boldsymbol{\beta}$ and let $f_p(\boldsymbol{\alpha}) := K(\boldsymbol{\beta},\boldsymbol{\alpha})$, where we stress the dependence on $p$. We have that $\|\boldsymbol{\alpha}\|_q \leq n^{1/q} \| \boldsymbol{\alpha}  \|_2$ by a norm inequality and $\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | \leq \|\boldsymbol{\alpha}\|_2 \|\boldsymbol{D} \|_2$ by Cauchy–Schwarz inequality. Thus,
\begin{equation*}
    f_p(\boldsymbol{\alpha}) \leq n^{1/p} \delta \|\boldsymbol{\beta}\|_* n^{1/q} \| \boldsymbol{\alpha}  \|_2 + \|\boldsymbol{\alpha}\|_2 \|\boldsymbol{D} \|_2 - \|\boldsymbol{\alpha}\|_2^2/4 = (n \delta \|\boldsymbol{\beta}\|_* + \|\boldsymbol{D} \|_2)\|\boldsymbol{\alpha}\|_2 - \|\boldsymbol{\alpha}\|_2^2/4
    \rightarrow -\infty
\end{equation*}
when $\|\boldsymbol{\alpha}\|_2 \rightarrow \infty$ independent of $p$. Thus, there exists a radius $R>0$ such that the maximum of $f_p(\boldsymbol{\alpha})$ for $\boldsymbol{\alpha} \geq 0$ is attained in $\Lambda := \{\boldsymbol{\alpha} \geq 0: \|\boldsymbol{\alpha}\|_2 \leq R\}$. Let 
\begin{equation*}
    f_\infty(\boldsymbol{\alpha}) := \lim_{p \rightarrow  \infty} f_p(\boldsymbol{\alpha}) = \delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha} \|_1+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i | - \alpha_i^2/4
\end{equation*}
and note that $f_\infty(\boldsymbol{\alpha})$ also attains its maximum on $\Lambda$. Moreover, since $\Lambda$ is compact, $f_p(\boldsymbol{\alpha})$ converges uniformly to $f_\infty(\boldsymbol{\alpha})$ on $\Lambda$. Due to this uniform convergence, we can swap the limit and the~supremum:
\begin{equation*}
    \lim_{p \rightarrow \infty} \sup_{\boldsymbol{\alpha} \geq 0} f_p(\boldsymbol{\alpha}) = \lim_{p \rightarrow \infty} \sup_{\boldsymbol{\alpha} \in \Lambda} f_p(\boldsymbol{\alpha}) = \sup_{\boldsymbol{\alpha} \in \Lambda} \lim_{p \rightarrow \infty} f_p(\boldsymbol{\alpha}) = \sup_{\boldsymbol{\alpha} \in \Lambda} f_\infty(\boldsymbol{\alpha}) = \sup_{\boldsymbol{\alpha} \geq 0} f_\infty(\boldsymbol{\alpha})
\end{equation*}
Therefore,
\begin{align*}
    \lim_{p \rightarrow \infty} V_{p,\delta}(\boldsymbol{\beta}) = \lim_{p \rightarrow \infty} \sup_{\boldsymbol{\alpha} \geq 0} f_p(\boldsymbol{\alpha}) =  \sup_{\boldsymbol{\alpha} \geq 0} f_\infty(\boldsymbol{\alpha}) = \\
    \sup_{\boldsymbol{\alpha} \geq 0} \left [ \delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha} \|_1+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | - \alpha_i^2/4 \right ] =

    \sum_{i=1}^n (\delta \|\boldsymbol{\beta}\|_*+ |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i |)^2,
\end{align*}
The last expression coincides with $V_{\infty,\delta}(\boldsymbol{\beta})$, completing the proof.

\label{lemma_in_sample_error_prestage}
                                                                                                                   
We have the following bounds:
\begin{align}
n V_{\delta}(\boldsymbol{\beta}^*) &\leq n \delta^2 \|\boldsymbol{\beta}^*\|_*^2+2n^{1/p} \delta \|\boldsymbol{\beta}^*\|_* \|\boldsymbol{\varepsilon}\|_q+\|\boldsymbol{\varepsilon}\|_2^2 \\
n V_{\delta}(\widehat{\boldsymbol{\beta}}) &\geq n \delta^2 \|\widehat{\boldsymbol{\beta}}\|_*^2+2\delta \|\widehat{\boldsymbol{\beta}}\|_* \|r(\widehat{\boldsymbol{\beta}})\|_1+\|r(\widehat{\boldsymbol{\beta}})\|_2^2,
\end{align}
where $r_i(\boldsymbol{\beta}) = |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |$.

<!-- proof-5: body_md -->
We obtain these bounds using the representation from \Cref{th_supremum_robust_risk}:
\begin{align*}
    n V_{\delta}(\boldsymbol{\beta}) = \max_{\boldsymbol{t} \geq 0, \|\boldsymbol{t}\|_p \leq 1} \sum_{i=1}^n \Big (n^{1/p}\delta \|\boldsymbol{\beta}\|_* t_i+ |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | \Big)^2 = \\
    \max_{\boldsymbol{t} \geq 0, \|\boldsymbol{t}\|_p \leq 1} \Big ( n^{2/p} \delta^2 \|\boldsymbol{\beta}\|_*^2 \|\boldsymbol{t}\|_2^2+2n^{1/p} \delta \|\boldsymbol{\beta}\|_* \sum_{i=1}^n t_i r_i(\boldsymbol{\beta})+\|r(\boldsymbol{\beta})\|_2^2 \Big ).
\end{align*}
To upper bound $V_{\delta}(\boldsymbol{\beta}^*)$, we note that $r(\boldsymbol{\beta}^*) = |\boldsymbol{\varepsilon}|$, resulting in
\begin{equation*}
    n V_{\delta}(\boldsymbol{\beta}^*) = \max_{\boldsymbol{t} \geq 0, \|\boldsymbol{t}\|_p \leq 1} \Big ( n^{2/p}\delta^2 \|\boldsymbol{\beta}^*\|_*^2 \|\boldsymbol{t}\|_2^2+2n^{1/p}\delta \|\boldsymbol{\beta}^*\|_* \sum_{i=1}^n t_i |\varepsilon_i|+\|\boldsymbol{\varepsilon}\|_2^2 \Big ).
\end{equation*}
Then we use Hölder's inequality $\sum_{i=1}^n t_i |\varepsilon_i| \leq \|\boldsymbol{t}\|_p \|\boldsymbol{\varepsilon}\|_q \leq \|\boldsymbol{\varepsilon}\|_q$ and the norm inequality $\|\boldsymbol{t}\|_2^2 \leq n^{1-2/p} \|\boldsymbol{t}\|_p^2 \leq n^{1-2/p}$ to get
\begin{equation*}
    n V_{\delta}(\boldsymbol{\beta}^*) \leq n \delta^2 \|\boldsymbol{\beta}^*\|_*^2+2n^{1/p} \delta \|\boldsymbol{\beta}^*\|_* \|\boldsymbol{\varepsilon}\|_q+\|\boldsymbol{\varepsilon}\|_2^2.
\end{equation*}
We next lower bound $n V_{\delta}(\widehat{\boldsymbol{\beta}})$. For this, we simply pick the point $s_i = n^{-1/p}$ (which satisfies $\|s\|_p =1$ and $s \geq 0$), yielding
\begin{equation*}
    n V_{\delta}(\widehat{\boldsymbol{\beta}}) \geq n \delta^2 \|\widehat{\boldsymbol{\beta}}\|_*^2+2\delta \|\widehat{\boldsymbol{\beta}}\|_* \|r(\widehat{\boldsymbol{\beta}})\|_1+\|r(\widehat{\boldsymbol{\beta}})\|_2^2.
\end{equation*}
This completes the proof.

\end{proof}

    4 \delta \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} \|\boldsymbol{\beta}^*\|_*+\delta^2 \|\boldsymbol{\beta}^*\|_*^2,
\end{align*}
    from which we get
\begin{align*}
\frac{1}{n}\|X \widehat{\Delta}\|_2^2 \leq 6 \delta \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} \|\boldsymbol{\beta}^*\|_*+2\delta^2 \|\boldsymbol{\beta}^*\|_*^2.
\end{align*}
We have thus shown \eqref{improved_in_sample_error_decay_eq}. 

To show the high-probability rate in \eqref{eq_improved_in_sample_error_decay_rate}, note first that by a norm inequality (since $1\leq q \leq 2$):
\begin{equation*}
    6 \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} \leq 
    6 \frac{n^{1/q-1/2}\|\boldsymbol{\varepsilon}\|_2}{n^{1/q}} = 
    6 \frac{\|\boldsymbol{\varepsilon}\|_2}{\sqrt{n}}.
\end{equation*}
We bound $ f(\boldsymbol{\varepsilon}) = 6 \frac{\|\boldsymbol{\varepsilon}\|_2}{\sqrt{n}}$ with high probability. Since $f$ is Lipschitz with respect to the 2-norm with constant $L=\frac{6}{\sqrt{n}}$, a concentration bound for Lipschitz functions with Gaussian input \citep[Theorem 5.6]{10.1093/acprof:oso/9780199535255.001.0001} yields:
\begin{equation*}
    \mathbb{P}(f(\boldsymbol{\varepsilon})-\mathbb{E}[f(\boldsymbol{\varepsilon})] \geq t) \leq \exp \left ( -\frac{t^2}{2\sigma^2L^2} \right ).
\end{equation*}
By setting
                  
    $\gamma := \exp \left ( -\frac{t^2}{2\sigma^2L^2} \right )$
                
we can solve for $t$ to get 
                  
    $t = \sigma L \sqrt{2 \log(1/\gamma)} = 6\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}$.
                
Thus with probability $\geq 1-\gamma$, we have that
                  
    $f(\boldsymbol{\varepsilon}) < \mathbb{E}[f(\boldsymbol{\varepsilon})]+6\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}},$
                
and since $\mathbb{E}[f(\boldsymbol{\varepsilon})] = 6\sigma$, we get that the bound 
\begin{equation*}
    6 \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} \leq 6\sigma + 6\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}
\end{equation*}
holds with probability greater than $1-\gamma$. Therefore, due to \eqref{improved_in_sample_error_decay_eq}, the rate of $\frac{1}{n}\|X \widehat{\Delta}\|_2^2$ is with high probability determined by the rate of $\delta$. That is, $\frac{1}{n}\|X \widehat{\Delta}\|_2^2 \in O(\delta)$ with high probability ($\geq 1- \gamma$). 

We now obtain the rate for the particular choice $\delta = K M \sqrt{ \frac{\log(d/\gamma)}{n}}$. To this end, we first upper bound $\bar{\delta}$. By a norm inequality
\begin{equation*}
    \bar{\delta} = \frac{2}{\sqrt{n}} \frac{\frac{\|X^{\top} \boldsymbol{\varepsilon}\|}{\sqrt{n}}}{\frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}}} \leq \frac{2}{\sqrt{n}} \frac{\frac{\|X^{\top} \boldsymbol{\varepsilon}\|}{\sqrt{n}}}{\frac{\|\boldsymbol{\varepsilon}\|_1}{n}}.
\end{equation*}
We upper bound $\|X^{\top} \boldsymbol{\varepsilon}\|$ and lower bound $\frac{\|\boldsymbol{\varepsilon}\|_1}{n}$. To lower bound $\frac{\|\boldsymbol{\varepsilon}\|_1}{n}$, note that $f(\boldsymbol{\varepsilon}) = \frac{\|\boldsymbol{\varepsilon}\|_1}{n} \leq \frac{\|\boldsymbol{\varepsilon}\|_2}{\sqrt{n}}$ is 2-norm Lipschitz with constant $L=1/\sqrt{n}$. Hence, a concentration bound for Gaussians \citep[Theorem 5.6]{10.1093/acprof:oso/9780199535255.001.0001} yield
\begin{equation*}
    \mathbb{P}(\mathbb{E}[f(\boldsymbol{\varepsilon})]-f(\boldsymbol{\varepsilon}) \geq t) \leq \exp \left ( -\frac{t^2}{2\sigma^2L^2} \right ).
\end{equation*}
Setting $\gamma = \exp \left ( -\frac{t^2}{2\sigma^2L^2} \right )$ so that
                  
    $t = \sigma L \sqrt{2 \log(1/\gamma)} = \sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}$,
                
and since $\mathbb{E}[f(\boldsymbol{\varepsilon})] = \sigma \sqrt{\frac{2}{\pi}}$, we get with probability $\geq 1-\gamma$:
                  
                                                                                                                                                                                                                                            
                
                                                                           
\begin{equation*}
    \frac{\|\boldsymbol{\varepsilon}\|_1}{n} \geq \sigma \sqrt{\frac{2}{\pi}}-\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}.
\end{equation*}
Moreover, we have with probability $\geq 1-2 \gamma$ that $\frac{\|X^{\top} \boldsymbol{\varepsilon}\|_\infty}{\sqrt{n}} \leq M \sigma \sqrt{2 \log(d/\gamma)}$ (see, e.g., \citep[Section C.3]{ribeiro_regularization_2023}), which implies that with probability $\geq 1-2 \gamma$:
\begin{equation*}
    \frac{\|X^{\top} \boldsymbol{\varepsilon}\|}{\sqrt{n}} \leq \frac{c\|X^{\top} \boldsymbol{\varepsilon}\|_\infty}{\sqrt{n}} \leq c M \sigma \sqrt{2 \log(d/\gamma)},
\end{equation*}
where we used that $\|\cdot \|_\infty \leq c\| \cdot \|$. Finally, we combine these two bounds to get a bound for $\bar{\delta}$. Concretely, with probability greater than $1-3\gamma$ (due to a union bound):
\begin{equation}\label{eq_upper_bound_bar_delta}
    \bar{\delta} = \frac{2}{\sqrt{n}} \frac{\frac{\|X^{\top} \boldsymbol{\varepsilon}\|}{\sqrt{n}}}{\frac{\|\boldsymbol{\varepsilon}\|_1}{n}} \leq \frac{2}{\sqrt{n}} \frac{cM \sigma \sqrt{2 \log(d/\gamma)}}{\sigma \sqrt{\frac{2}{\pi}}-\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}} = \frac{2}{\sqrt{n}} \frac{cM \sqrt{ \log(d/\gamma)}}{\sqrt{\frac{1}{\pi}}- \sqrt{\frac{ \log(1/\gamma)}{n}}} =: \frac{C_1(n)}{\sqrt{n}},
\end{equation}
valid for $n > \pi \log(1/\gamma)$ since the denominator is then positive. Note that the function $C_1(n)$ is decreasing with $n$ (given that $n > \pi \log(1/\gamma)$). In other words, $\bar{\delta} \in O(1/\sqrt{n})$ with probability $\geq 1-3\gamma$.\footnote{That we do not have an upper bound for $\bar{\delta}$ over the transition period $n \leq \pi \log(1/\gamma)$ is immaterial since $\bar{\delta}$ is almost surely finite over these finitely many value of $n$.} 

Finally, we tune $K$ in $\delta = K M \sqrt{ \frac{\log(d/\gamma)}{n}}$ to satisfy $\delta > \bar{\delta}$ for large enough $n$, which is true if
\begin{equation}\label{eq:tuning_K_slow_rate}
    K > \frac{2c}{\sqrt{\frac{1}{\pi}}- \sqrt{\frac{ \log(1/\gamma)}{n}}}
\end{equation}
for large enough $n$. Since $\frac{2c}{\sqrt{\frac{1}{\pi}}- \sqrt{\frac{ \log(1/\gamma)}{n}}} \rightarrow 2c\sqrt{\pi}$ as $n \rightarrow \infty$, a sufficient condition for satisfying $\delta > \bar{\delta}$ for large enough $n$ (with probability $\geq 1-3\gamma$) is that $K>2c\sqrt{\pi}$, where a larger $K$ results in $\delta > \bar{\delta}$ being satisfied sooner (smaller $n$). Thus, for $K>2c\sqrt{\pi}$, we have $\frac{1}{n}\|X \widehat{\Delta}\|_2^2 \in O(\delta) = O(1/\sqrt{n})$ (with probability $\geq 1- 4 \gamma$ due to a union bound), achieving the slow rate. This completes the proof. 
\end{proof}

\begin{equation*}

\begin{equation*}

\begin{equation*}
    \|\boldsymbol{\beta}^*\|_*-\|\widehat{\boldsymbol{\beta}}\|_* \leq \|\boldsymbol{\beta}^*-\widehat{\boldsymbol{\beta}}\|_*  = \|\widehat{\Delta}\|_*,

\begin{equation*}

        2\|\widehat{\boldsymbol{\beta}}\|_1 \langle |X\widehat{\Delta}|,\boldsymbol{t}^*\rangle.
\end{align*}
where we used the assumption $\|\widehat{\boldsymbol{\beta}}-\boldsymbol{\beta}^* \|_1+2(\|\boldsymbol{\beta}^*\|_1-\|\widehat{\boldsymbol{\beta}}\|_1) \leq 0$ and that $\|\boldsymbol{\beta}^*\|_1-\|\widehat{\boldsymbol{\beta}}\|_1 \leq 0$ in the last step. Moreover, by a norm inequality,
\begin{equation*}
\| \boldsymbol{t}^* \|_2 \leq n^{1/2-1/p} \|\boldsymbol{t}^*\|_p  \leq n^{1/2-1/p} n^{1/p}\delta = n^{1/2} \delta,
\end{equation*}
we get
\begin{equation*}
    \langle |X\widehat{\Delta}|,\boldsymbol{t}^*\rangle \leq \|X\widehat{\Delta}\|_2 \|\boldsymbol{t}^*\|_2 \leq \|X\widehat{\Delta}\|_2 n^{1/2} \delta,
\end{equation*}
using the Cauchy–Schwarz inequality. Hence, $\|X\widehat{\Delta}\|_2^2 \leq 2\|\widehat{\boldsymbol{\beta}}\|_1 \|X\widehat{\Delta}\|_2 n^{1/2} \delta$ which implies $\frac{1}{\sqrt{n}} \|X \widehat{\Delta}\|_2 \leq 2\delta\|\widehat{\boldsymbol{\beta}}\|_1$ and so
\begin{align*}
        \frac{1}{n} \|X \widehat{\Delta}\|_2^2 \leq 4\delta^2\|\widehat{\boldsymbol{\beta}}\|_1^2.
\end{align*}
We finally bound $\|\widehat{\boldsymbol{\beta}}\|_1^2$ using \Cref{th_controlled_dual_norm_improved_v3} yielding $\|\widehat{\boldsymbol{\beta}}\|_1^2 \leq C^2 \|\boldsymbol{\beta}^*\|_1^2$ and
\begin{align*}
        \frac{1}{n} \|X \widehat{\Delta}\|_2^2 \leq 4\delta^2C^2 \|\boldsymbol{\beta}^*\|_1^2.
\end{align*}

\textbf{Case 2: Assume $\|\widehat{\boldsymbol{\beta}}-\boldsymbol{\beta}^* \|_1+2(\|\boldsymbol{\beta}^*\|_1-\|\widehat{\boldsymbol{\beta}}\|_1) \geq 0$.} In this case, we will use the RE condition. More precisely, we have
\begin{equation*}
    \|\widehat{\Delta} \|_1 \leq \|\widehat{\Delta}_S \|_1 + \|\widehat{\Delta}_{S^c} \|_1
\end{equation*}
and
\begin{align*}
    \|\boldsymbol{\beta}^*\|_1-\|\widehat{\boldsymbol{\beta}}\|_1 = \|\boldsymbol{\beta}^*_S\|_1 - \|\boldsymbol{\beta}^*_S+\widehat{\Delta}_S+\widehat{\Delta}_{S^c} \|_1 = \\ \|\boldsymbol{\beta}^*_S\|_1 - \|\boldsymbol{\beta}^*_S+\widehat{\Delta}_S \|_1 - \| \widehat{\Delta}_{S^c} \|_1 \leq \|\widehat{\Delta}_S\|_1 - \| \widehat{\Delta}_{S^c} \|_1.
\end{align*}
Hence,
\begin{equation*}
    0 \leq \|\widehat{\Delta} \|_1+2(\|\boldsymbol{\beta}^*\|_1-\|\widehat{\boldsymbol{\beta}}\|_1) \leq \|\widehat{\Delta}_S \|_1 + \|\widehat{\Delta}_{S^c} \|_1 + 2(\|\widehat{\Delta}_S\|_1 - \| \widehat{\Delta}_{S^c} \|_1)
\end{equation*}
which implies that $\|\widehat{\Delta}_{S^c} \|_1 \leq 3 \|\widehat{\Delta}_S\|_1$. In other words, we can use $\mathrm{RE}(s,3)$ to get
\begin{equation*}
\|\widehat{\Delta}\|_1 = \|\widehat{\Delta}_S\|_1+\|\widehat{\Delta}_{S^c}\|_1 \leq 4 \|\widehat{\Delta}_S\|_1 \leq 4s \|\widehat{\Delta}_S\|_2 \leq 4s \frac{\|X\widehat{\Delta}\|_2}{\kappa \sqrt{n}},
\end{equation*}
where we also used the norm inequality $\|\widehat{\Delta}_S\|_1 \leq s \|\widehat{\Delta}_S\|_2$. This bound combined with \Cref{theorem_in_sample_controlled_by_Delta}~yields
\begin{align*}
    \frac{1}{n} \|X\widehat{\Delta}\|_2^2 \leq \delta B \|\widehat{\Delta}\|_1 \leq 4\delta B s \frac{\|X\widehat{\Delta}\|_2}{\kappa \sqrt{n}} \quad \Rightarrow \quad \frac{1}{\sqrt{n}} \|X\widehat{\Delta}\|_2 \leq \frac{4\delta B s}{\kappa}.
\end{align*}
Therefore, $\frac{1}{n} \|X\widehat{\Delta}\|_2^2 \leq \frac{16\delta^2 B^2 s^2}{\kappa^2}$, completing the second case.

Combining Cases 1 and 2, we conclude that
\begin{align*}
\frac{1}{n} \|X\widehat{\Delta}\|_2^2 \leq \delta^2 \cdot \max \left \{ 4C^2 \|\boldsymbol{\beta}^*\|_1^2, \frac{16 B^2 s^2}{\kappa^2} \right \}.
\end{align*}
That is, we have proved \eqref{eq_fast_rate_main_theorem_v2_bound}. It remains to show the rates. To this end, assume $\boldsymbol{\varepsilon}  \sim N(0, \sigma  I)$. We show that $B$ and $C$ are bounded by constants with high probability. We start with $C$. By a norm inequality and the proof of \Cref{improved_in_sample_error_decay}, we have with probability $\geq 1-\gamma$:
\begin{equation*}
    \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} \geq \frac{\|\boldsymbol{\varepsilon}\|_1}{n} \geq \sigma \sqrt{\frac{2}{\pi}}-\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}.
\end{equation*}
Therefore, with probability $\geq 1-\gamma$:
\begin{equation*}
    C =  \left (  3+\frac{n^{1/q} \delta}{\| \boldsymbol{\varepsilon}\|_q} \|\boldsymbol{\beta}^*\|_* \right ) \leq \left (  3+\frac{\delta}{\sigma \sqrt{\frac{2}{\pi}}-\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}} \|\boldsymbol{\beta}^*\|_* \right ),
\end{equation*}
valid for $n > \pi \log(1/\gamma)$ (to ensure the denominator is positive). In other words, $C$ is bounded by a constant with high probability.\footnote{For the finitely many $n$ such that $n \leq \pi \log(1/\gamma)$, we have that $C$ is almost surely bounded.} Moreover, by the proof of \Cref{improved_in_sample_error_decay}, we have with probability $\geq 1-\gamma$:
\begin{equation*}
    \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} \leq \sigma + \sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}.
\end{equation*}
Hence, applying a union bound, we have with probability $\geq 1-2\gamma$:
\begin{align*}
    B = 3 \frac{\|\boldsymbol{\varepsilon}\|_q}{n^{1/q}} + (2M+\delta )\|\boldsymbol{\beta}^*\|_*C +\delta\|\boldsymbol{\beta}^*\|_* \leq \\ 
    3 \left ( \sigma + \sigma \sqrt{\frac{2 \log(1/\gamma)}{n}} \right ) + (2M+\delta )\|\boldsymbol{\beta}^*\|_* \left (  3+\frac{\delta}{\sigma \sqrt{\frac{2}{\pi}}-\sigma \sqrt{\frac{2 \log(1/\gamma)}{n}}} \|\boldsymbol{\beta}^*\|_* \right ) +\delta\|\boldsymbol{\beta}^*\|_*  
\end{align*}
valid for $n > \pi \log(1/\gamma)$. We conclude that $B$ is also bounded by a constant with high probability. As a result, we have that $\frac{1}{n} \|X\widehat{\Delta}\|_2^2 \in O(\delta^2)$ with high probability ($\geq 1-2\gamma$), independent of $p$ since the upper bounds on $B$ and $C$ are independent of $p$. Finally, by setting $\delta$ as in \Cref{improved_in_sample_error_decay} and following its proof (where $\delta>\bar{\delta}$ with probability $\geq 1-3\gamma$), we analogously obtain that $\frac{1}{n} \|X\widehat{\Delta}\|_2^2 \in O(\delta^2) = O(1/n)$ for $K>2c\sqrt{\pi}$ with probability $\geq 1-5\gamma$ due to a union bound. This completes the proof. 
                                                                               
\end{proof}

\label{th_general_danskin}
Let $\phi: \mathbb{R}^d \times \mathbb{R}^n \rightarrow \mathbb{R}$ be a continuous function such that $\phi(\cdot,\boldsymbol{t})$ is convex for each $\boldsymbol{t} \in T$ in a compact set $T$, and consider $f(\boldsymbol{\beta}):= \sup_{\boldsymbol{t} \in T} \phi(\boldsymbol{\beta},\boldsymbol{t})$. Then, for each $\boldsymbol{\beta} \in R^d$, the subdifferential of $f$ equals
\begin{equation}
\partial f(\boldsymbol{\beta}) = \mathrm{conv}\{ \partial_{\boldsymbol{\beta}} \phi(\boldsymbol{\beta},\boldsymbol{t}): \boldsymbol{t} \in T(\boldsymbol{\beta}) \}, \quad T(\boldsymbol{\beta}) := \{\boldsymbol{t} \in T: \phi(\boldsymbol{\beta},\boldsymbol{t}) = \max_{\boldsymbol{t} \in T} \phi(\boldsymbol{\beta},\boldsymbol{t}) \},
\end{equation}
where $\partial_{\boldsymbol{\beta}} \phi(\boldsymbol{\beta},\boldsymbol{t})$ is the subdifferential of $\phi(\cdot,\boldsymbol{t})$ at $\boldsymbol{\beta}$.
<!-- end theorem-6 -->
<!-- proof-13: body_md -->
Note that $\widehat{\boldsymbol{\beta}} = \boldsymbol{0}$ minimizes the robust risk $V_\delta(\boldsymbol{\beta})$ if and only if $\boldsymbol{0} \in  \partial V_\delta(\boldsymbol{0})$, since $V_\delta(\boldsymbol{\beta})$ is convex in $\boldsymbol{\beta}$. We compute $\partial V_\delta(\boldsymbol{0})$ using \Cref{th_general_danskin}. More precisely, note that $nV_\delta(\boldsymbol{\beta}) = \sup_{\boldsymbol{t} \in T} \phi(\boldsymbol{\beta},\boldsymbol{t})$ with
\begin{align*}
\phi(\boldsymbol{\beta},\boldsymbol{t}) &= \sum_{i=1}^n \left ( |\boldsymbol{x}_i^{\top} \boldsymbol{\beta}-y_i|+t_i \|\boldsymbol{\beta} \|_* \right )^2 \\
T &=  \left \{ \boldsymbol{t} \in \mathbb{R}^n:  t_i \geq 0, \|\boldsymbol{t}\|_p \leq n^{1/p} \delta \right \}
                                                                                      
\end{align*}
It is easy to verify that the conditions of \Cref{th_general_danskin} hold, yielding:
\begin{equation}\label{eq_subgrad_robust_risk_at_zero}
    \partial_{\boldsymbol{\beta}} (nV_{\delta}(\boldsymbol{0})) = \mathrm{conv}\{ \partial_{\boldsymbol{\beta}} \phi(\boldsymbol{0},\boldsymbol{t}): \boldsymbol{t} \in T(\boldsymbol{0}) \}.
\end{equation}
We compute $\partial_{\boldsymbol{\beta}} \phi(\boldsymbol{0},\boldsymbol{t})$ and $T(\boldsymbol{0})$. In general,
\begin{align}\label{eq_subgrad_f_s_general}
\partial_{\boldsymbol{\beta}} \phi(\boldsymbol{\beta},\boldsymbol{t}) = 2\sum_{i=1}^n \left ( |\boldsymbol{x}_i^{\top} \boldsymbol{\beta}-y_i|+t_i \|\boldsymbol{\beta} \|_* \right ) \left ( \partial_{\boldsymbol{\beta}} |\boldsymbol{x}_i^{\top} \boldsymbol{\beta}-y_i|+t_i \partial_{\boldsymbol{\beta}} \|\boldsymbol{\beta} \|_* \right ), 
\end{align}
where
\begin{align*}
\partial_{\boldsymbol{\beta}} |\boldsymbol{x}_i^{\top} \boldsymbol{\beta}-y_i| &= \boldsymbol{x}_i \mathrm{sign}( \boldsymbol{x}_i^{\top} \boldsymbol{\beta}-y_i) \quad \text{with} \quad 
\mathrm{sign}(a) :=
\begin{cases}
    \{1\} &\text{ if } a>0  \\
    \{-1\} &\text{ if } a<0 \\
    [-1,1] &\text{ if } a=0 
\end{cases}
\\
\partial_{\boldsymbol{\beta}} \| \boldsymbol{\beta} \|_* &= \{ \boldsymbol{z} \in \mathbb{R}^d: \|\boldsymbol{z}\| \leq 1, \langle \boldsymbol{z},\boldsymbol{\beta} \rangle = \|\boldsymbol{\beta} \|_* \}
\end{align*}
by standard derivations.\footnote{Note that we are working with sets where addition is the Minkowski addition, standard for~subdifferentials.} In particular, for $\boldsymbol{\beta} = \boldsymbol{0}$, we have $\partial_{\boldsymbol{\beta}} \| \boldsymbol{\beta} \|_* = \mathbb{B} := \{\boldsymbol{z} \in \mathbb{R}^d: \|\boldsymbol{z}\|\leq 1\}$ and $\partial_{\boldsymbol{\beta}} |\boldsymbol{x}_i^{\top} \boldsymbol{\beta}-y_i|  = -\boldsymbol{x}_i \mathrm{sign}(y_i)$. Hence,
\begin{align*}
\partial_{\boldsymbol{\beta}} \phi(\boldsymbol{0},\boldsymbol{t}) = 
2\sum_{i=1}^n |y_i| \left ( -\boldsymbol{x}_i \mathrm{sign}(y_i) +t_i \mathbb{B} \right ) = \\ 
2\sum_{i=1}^n \left ( -\boldsymbol{x}_i y_i \right ) + 2 \sum_{i=1}^n \left ( t_i |y_i| \mathbb{B} \right ) = 2\big ( -X^T \boldsymbol{y}+\langle \boldsymbol{t},|\boldsymbol{y}| \rangle \mathbb{B} \big )
\end{align*}
where we used that $|y_i|\mathrm{sign}(y_i)=y_i$, and that the Minkowski sum of balls with the same center (the origin) equals the ball with their radii added. Next, we note that $\phi(\boldsymbol{0},\boldsymbol{t}) = \sum_{i=1}^n y_i^2$ for each $\boldsymbol{t} \in T$, so $T(\boldsymbol{0}) = T$. Therefore, \eqref{eq_subgrad_robust_risk_at_zero} yields
\begin{align*}
    \partial_{\boldsymbol{\beta}} (nV_{\delta}(\boldsymbol{0}))  = 
    \mathrm{conv} \left [ \bigcup_{\boldsymbol{t} \in T} 2\big ( -X^{\top} \boldsymbol{y}+\langle \boldsymbol{t},|\boldsymbol{y}| \rangle \mathbb{B} \big ) \right ]  = \\
     -X^{\top} \boldsymbol{y}+\big (\max_{\boldsymbol{t} \in T} \langle \boldsymbol{t},|\boldsymbol{y}| \rangle \big ) \mathbb{B} = -X^{\top} \boldsymbol{y} + n^{1/p} \delta \|\boldsymbol{y}\|_q \mathbb{B}
\end{align*}
since the convex hull of a union of balls with the same center equals the ball with largest radius (i.e., $\max_{\boldsymbol{t} \in T} \langle \boldsymbol{t},|\boldsymbol{y}| \rangle$ in our case) and
\begin{align*}
\max_{\boldsymbol{t} \in T} \langle \boldsymbol{t},|\boldsymbol{y}| \rangle = n^{1/p} \delta \max_{\|\boldsymbol{t}\|_p \leq 1} \langle \boldsymbol{t},|\boldsymbol{y}| \rangle = n^{1/p} \delta \|\boldsymbol{y}\|_q.
\end{align*}
Therefore, $\boldsymbol{0} \in \partial_{\boldsymbol{\beta}} V_{\delta}(\boldsymbol{0})$ if and only if
\begin{align*}
\boldsymbol{0} \in -X^{\top} \boldsymbol{y} + n^{1/p} \delta \|\boldsymbol{y}\|_q \mathbb{B} \quad \Leftrightarrow \quad \|X^{\top} \boldsymbol{y} \| \leq n^{1/p} \delta \|\boldsymbol{y}\|_q \quad \Leftrightarrow \quad \frac{\|X^{\top} \boldsymbol{y} \|}{n^{1/p}\|\boldsymbol{y}\|_q} \leq  \delta
\end{align*}
completing the proof.

\label{lemma_when_beta_minimum_norm}
Let $\bar{\boldsymbol{\beta}}$ be the minimum $\|\cdot\|_*$-norm interpolator. Then $X^T \widehat{\boldsymbol{\alpha}} \in \partial \|\bar{\boldsymbol{\beta}} \|_*$ if and only if $\widehat{\boldsymbol{\alpha}}$ solves $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$.

Analogously to the proof of Lemma 1 in \citet{ribeiro_regularization_2023}, we have by strong duality that
\begin{align*}
    \| \bar{\boldsymbol{\beta}} \|_* = \min_{X \boldsymbol{\beta} = \boldsymbol{y}} \|\boldsymbol{\beta}  \|_* = \max_{\boldsymbol{\alpha}} \min_{\boldsymbol{\beta}} \left ( \|\boldsymbol{\beta}  \|_*+ \boldsymbol{\alpha}^{\top}(\boldsymbol{y}-X \boldsymbol{\beta})  \right ) = \\
    \max_{\boldsymbol{\alpha}} \left ( \boldsymbol{\alpha}^{\top} \boldsymbol{y} + \min_{\boldsymbol{\beta}} (\|\boldsymbol{\beta}  \|_*- \boldsymbol{\alpha}^{\top} X \boldsymbol{\beta}) \right) = \max_{\|X^{\top}\boldsymbol{\alpha}\|\leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y},
\end{align*}
where the last step used that the Fenchel conjugate of $\|\boldsymbol{\beta}  \|_*$ is the indicator function on the unit ball with norm~$\|\cdot \|$. Moreover, 
\begin{align*}
\partial \| \bar{\boldsymbol{\beta}} \|_* &= \{ \boldsymbol{z} \in \mathbb{R}^d: \|\boldsymbol{z}\| \leq 1, \boldsymbol{z}^{\top} \bar{\boldsymbol{\beta}} = \| \bar{\boldsymbol{\beta}} \|_* \}
\end{align*}
so $X^{\top} \widehat{\boldsymbol{\alpha}} \in \partial \| \bar{\boldsymbol{\beta}} \|_*$ implies that $\| X^{\top} \widehat{\boldsymbol{\alpha}} \| \leq 1$ and 
\begin{align*}
\widehat{\boldsymbol{\alpha}}^{\top} \boldsymbol{y} = \widehat{\boldsymbol{\alpha}}^{\top} X\bar{\boldsymbol{\beta}} = (X^{\top} \widehat{\boldsymbol{\alpha}})^{\top} \bar{\boldsymbol{\beta}}  =   \| \bar{\boldsymbol{\beta}} \|_* = \max_{\|X^{\top} \boldsymbol{\alpha}\|\leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}.
\end{align*}
That is, $\widehat{\boldsymbol{\alpha}}$ solves $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$. Conversely, assume $\widehat{\boldsymbol{\alpha}}$ solves $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$. Then $\| X^{\top} \widehat{\boldsymbol{\alpha}} \| \leq 1$ and
\begin{equation*}
    (X^{\top} \widehat{\boldsymbol{\alpha}})^{\top} \bar{\boldsymbol{\beta}} = \widehat{\boldsymbol{\alpha}}^{\top} \boldsymbol{y} = \max_{\|X^{\top} \boldsymbol{\alpha}\|\leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y} = \| \bar{\boldsymbol{\beta}} \|_*
\end{equation*}
so $X^{\top} \widehat{\boldsymbol{\alpha}} \in \partial \| \bar{\boldsymbol{\beta}} \|_*$, completing the proof.
<!-- end proof-14 -->
<!-- proof-15: body_md -->
Note that $\bar{\boldsymbol{\beta}}$ minimizes $V_{\delta}(\boldsymbol{\beta})$ if and only if $\boldsymbol{0} \in \partial_{\boldsymbol{\beta}} V_{\delta}(\bar{\boldsymbol{\beta}})$. We divide the proof into two subcases: $2 <p \leq \infty$ and $p=2$.

Assume first that $2 <p \leq \infty$. We compute $\partial_{\boldsymbol{\beta}} V_{\delta}(\bar{\boldsymbol{\beta}})$ using \Cref{th_general_danskin}. To this end, let $\phi$ and $T$ be as in the proof of \Cref{th_when_beta_zero}, so that $nV_\delta(\boldsymbol{\beta}) = \sup_{\boldsymbol{t} \in T} \phi(\boldsymbol{\beta},\boldsymbol{t})$. \Cref{th_general_danskin} implies that
\begin{equation}\label{eq_subgrad_robust_risk_small_delta}
    \partial (nV_\delta(\bar{\boldsymbol{\beta}})) = \mathrm{conv}\{ \partial_{\boldsymbol{\beta}} \phi(\bar{\boldsymbol{\beta}},\boldsymbol{t}): \boldsymbol{t} \in T(\bar{\boldsymbol{\beta}}) \}.
\end{equation}
We compute $\partial_{\boldsymbol{\beta}} \phi(\bar{\boldsymbol{\beta}},\boldsymbol{t})$ and $T(\bar{\boldsymbol{\beta}})$. By \eqref{eq_subgrad_f_s_general}, we have
\begin{align*}
\partial_{\boldsymbol{\beta}} \phi(\bar{\boldsymbol{\beta}},\boldsymbol{t}) = 2 \sum_{i=1}^n t_i \|\bar{\boldsymbol{\beta}}\|_* (\boldsymbol{x}_i \mathbb{I}+t_i \partial_{\boldsymbol{\beta}} \|\bar{\boldsymbol{\beta}}\|_*)
\end{align*}
since $X \bar{\boldsymbol{\beta}} = \boldsymbol{y}$, where $\mathbb{I} = [-1,1]$. Moreover,
\begin{align*}
\phi(\bar{\boldsymbol{\beta}},\boldsymbol{t}) = \sum_{i=1}^n t_i^2 \|\bar{\boldsymbol{\beta}}\|_*^2 = \|\bar{\boldsymbol{\beta}}\|_*^2 \|\boldsymbol{t} \|_2^2
\end{align*}
and therefore
\begin{align*}
    \sup_{\boldsymbol{t} \in T} \phi(\bar{\boldsymbol{\beta}},\boldsymbol{t}) = \|\bar{\boldsymbol{\beta}}\|_*^2 \sup_{\boldsymbol{t} \in T} \|\boldsymbol{t} \|_2^2 = \|\bar{\boldsymbol{\beta}}\|_*^2 n \delta^2
\end{align*}
with the supremum attained only at $t_i = \delta$ due to a norm inequality (recall that $t_i \geq 0$ in $T$). That is, $T(\bar{\boldsymbol{\beta}})$ equals the singleton $\bar{\boldsymbol{t}}$ with entries $\bar{t}_i = \delta$.\footnote{This argument holds provided $p>2$. For $p=2$, the set $T(\bar{\boldsymbol{\beta}})$ has more than one element since the supremum $\sup_{t \in T} \|t \|_2^2$ has multiple solutions, which is why we consider the case $p=2$ separately.} Hence, \eqref{eq_subgrad_robust_risk_small_delta} yields
\begin{align*}
    \partial_{\boldsymbol{\beta}} n V_{\delta}(\bar{\boldsymbol{\beta}}) =  \partial_{\boldsymbol{\beta}} \phi(\bar{\boldsymbol{\beta}},\bar{\boldsymbol{t}}) =  
    2 \sum_{i=1}^n \delta \|\bar{\boldsymbol{\beta}}\|_* (\boldsymbol{x}_i \mathbb{I}+\delta \partial_{\boldsymbol{\beta}} \|\bar{\boldsymbol{\beta}}\|_*).
\end{align*}
Pick an arbitrary point in this set:
\begin{equation}\label{eq_min_norm_subgrad_parameter_form}
    2 \sum_{i=1}^n \delta \|\bar{\boldsymbol{\beta}}\|_* (\boldsymbol{x}_i \rho_i+\delta \boldsymbol{g_i}),
\end{equation}
where $\rho_i \in \mathbb{I} = [-1,1]$ and $\boldsymbol{g_i} \in \partial_{\boldsymbol{\beta}} \|\bar{\boldsymbol{\beta}}\|_*$. Since $\partial_{\boldsymbol{\beta}} \|\bar{\boldsymbol{\beta}}\|_*$ is a convex set, we have $\sum_{i=1}^n \boldsymbol{g_i} = n\boldsymbol{g}$ for some $\boldsymbol{g} \in \partial_{\boldsymbol{\beta}} \|\bar{\boldsymbol{\beta}}\|_*$, which simplifies \eqref{eq_min_norm_subgrad_parameter_form} to
\begin{align*}
    2 \delta \|\bar{\boldsymbol{\beta}}\|_* \left [ \sum_{i=1}^n (\boldsymbol{x}_i \rho_i)+\delta n \boldsymbol{g}  \right ] = 
    2 \delta \|\bar{\boldsymbol{\beta}}\|_* \left [ X^{\top} \boldsymbol{\rho}+\delta n \boldsymbol{g}  \right ].
\end{align*}
Thus, $\boldsymbol{0} \in \partial_{\boldsymbol{\beta}} nV_{\delta}(\bar{\boldsymbol{\beta}})$ if and only if 
\begin{equation*}
    \boldsymbol{g} = -\frac{X^{\top} \boldsymbol{\rho}}{n \delta}
\end{equation*}
for some $\boldsymbol{g} \in \partial_{\boldsymbol{\beta}} \|\bar{\boldsymbol{\beta}}\|_*$ and $\|\boldsymbol{\rho}\|_\infty \leq 1$. By \Cref{lemma_when_beta_minimum_norm}, this is if and only if there exists some $\|\boldsymbol{\rho}\|_\infty \leq 1$ such that
\begin{equation*}
    \widehat{\boldsymbol{\alpha}} := -\frac{\boldsymbol{\rho}}{n \delta}
\end{equation*}
solves $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$. Equivalently, there exists $\widehat{\boldsymbol{\alpha}}$ solving $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$ such that
\begin{equation*}
    \|\widehat{\boldsymbol{\alpha}} \|_\infty \leq \frac{1}{n \delta}.
\end{equation*}
Crucially, this condition holds if and only if 
\begin{equation*}
    \|\boldsymbol{\bar{\alpha}} \|_\infty \leq \frac{1}{n \delta}
\end{equation*}
where $\boldsymbol{\bar{\alpha}}$ solves $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$ and has the \emph{minimum} $\infty$-norm among all such solutions. In other words, $\boldsymbol{\bar{\alpha}}$ solves $\min_{\boldsymbol{\alpha}  \in Q} \|\boldsymbol{\alpha}\|_\infty$ with $Q = \operatorname*{arg\,max}_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$, completing the proof for $2  < p \leq \infty$.

We now continue with the case $p=2$. By \Cref{th_special_cases}, we have that 
\begin{equation*}
    n V_\delta(\boldsymbol{\beta}) = \left ( \sqrt{\frac{1}{n}\sum_{i=1}^n (\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i)^2}+\delta \|\boldsymbol{\beta}\|_* \right)^2 = \left ( \frac{1}{\sqrt{n}} \|X \boldsymbol{\beta}-\boldsymbol{y}  \|_2 + \delta \|\boldsymbol{\beta}\|_* \right)^2
\end{equation*}
To ease the computation of subgradients, we consider instead the equivalent objective
\begin{equation*}
    \sqrt{n V_\delta(\boldsymbol{\beta})} = \frac{1}{\sqrt{n}} \|X \boldsymbol{\beta}-\boldsymbol{y}  \|_2 + \delta \|\boldsymbol{\beta}\|_*.
\end{equation*}
Indeed, $\bar{\boldsymbol{\beta}}$ minimizes $n V_\delta(\boldsymbol{\beta})$ if and only if $\bar{\boldsymbol{\beta}}$ minimizes $\sqrt{n V_\delta(\boldsymbol{\beta})}$, which in turn is if and only if $\boldsymbol{0} \in \partial_{\boldsymbol{\beta}} \sqrt{n V_\delta(\boldsymbol{\bar{\boldsymbol{\beta}}})}$. We compute $ \partial_{\boldsymbol{\beta}} \sqrt{n V_\delta(\boldsymbol{\bar{\boldsymbol{\beta}}})}$:
\begin{equation}\label{eq_subgrad_p2_interpolation_root}
     \partial_{\boldsymbol{\beta}} \sqrt{n V_\delta(\bar{\boldsymbol{\beta}})} = \frac{1}{\sqrt{n}} \partial_{\boldsymbol{\beta}}\|X \bar{\boldsymbol{\beta}}-\boldsymbol{y}  \|_2 + \delta \partial_{\boldsymbol{\beta}}\|\bar{\boldsymbol{\beta}}\|_*.
\end{equation}
By the chain rule for affine transformations (see, e.g., \cite[Proposition 5.4.5]{bertsekas2009convex}), we have that
\begin{equation*}
    \partial_{\boldsymbol{\beta}}\|X \bar{\boldsymbol{\beta}}-\boldsymbol{y}  \|_2 = X^T \partial_{\boldsymbol{r}} \|\boldsymbol{r}\|_2, 
\end{equation*}
where $\partial_{\boldsymbol{r}} \|\boldsymbol{r}\|_2$ is the subgradient of the norm $\| \cdot \|_2$ at $\boldsymbol{r} = X \bar{\boldsymbol{\beta}}-y$. Since $\boldsymbol{r} = X \bar{\boldsymbol{\beta}}-\boldsymbol{y} = \boldsymbol{0}$, we get $\partial_{\boldsymbol{r}} \|\boldsymbol{r}\|_2 = \{\boldsymbol{u}: \|\boldsymbol{u} \|_2 \leq 1\}$ and therefore
\begin{equation*}
    \partial_{\boldsymbol{\beta}}\|X \bar{\boldsymbol{\beta}}-\boldsymbol{y}  \|_2 = \{X^{\top} \boldsymbol{u}: \|\boldsymbol{u} \|_2 \leq 1\}.
\end{equation*}
Thus, the subgradient in \eqref{eq_subgrad_p2_interpolation_root} contains zero  if and only if
\begin{equation*}
    \frac{1}{\sqrt{n}} X^{\top} \boldsymbol{u} + \delta \boldsymbol{g} = \boldsymbol{0} \quad \Leftrightarrow \quad \boldsymbol{g} = \frac{1}{\delta \sqrt{n}} X^{\top} \boldsymbol{u}
\end{equation*}
for some $\|\boldsymbol{u}\|_2 \leq 1$ and $\boldsymbol{g} \in \partial_{\boldsymbol{\beta}}\|\bar{\boldsymbol{\beta}}\|_*$. By \Cref{lemma_when_beta_minimum_norm}, this is if and only if there exists $\boldsymbol{u} \in \mathbb{R}^n$ such that $\|\boldsymbol{u} \|_2 \leq 1$ and
\begin{equation*}
    \widehat{\boldsymbol{\alpha}} := -\frac{\boldsymbol{u}}{\delta \sqrt{n}}
\end{equation*}
solves $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$. Equivalently, there exists $\widehat{\boldsymbol{\alpha}}$ solving $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$ such that
\begin{equation*}
    \|\widehat{\boldsymbol{\alpha}}\|_2 \leq \frac{1}{\delta \sqrt{n}}.
\end{equation*}
Similarly to the case $2 <p \leq \infty$, this holds if and only if  
\begin{equation*}
    \|\boldsymbol{\bar{\alpha}}\|_2 \leq \frac{1}{\delta \sqrt{n}},
\end{equation*}
where $\boldsymbol{\bar{\alpha}}$ solves $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$ and has the \emph{minimum} $2$-norm among all such solutions. In other words, $\boldsymbol{\bar{\alpha}}$ solves $\min_{\boldsymbol{\alpha}  \in Q} \|\boldsymbol{\alpha}\|_2$ with $Q = \operatorname*{arg\,max}_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$, completing the proof for $p = 2$. Combining both cases completes the proof in general.

<!-- remark-3: body_md -->
\label{remark_computing_delta_S}
For brevity, let $\bar{\boldsymbol{\alpha}}$ denote any solution of $\min_{\boldsymbol{\alpha}  \in Q}\|\boldsymbol{\alpha}\|_{\infty}$ for $2<p\leq \infty$ and $\min_{\boldsymbol{\alpha}  \in Q}\|\boldsymbol{\alpha}\|_{2}$ for $p=2$. To compute the bound $\delta_S$ in \Cref{th_when_beta_minimum_norm}, we need to compute $\bar{\boldsymbol{\alpha}}$ (any will do). Fortunately, $\bar{\boldsymbol{\alpha}}$ can be easily obtained from a simple convex program. Indeed, solve the inner maximization $v^{\star}:= \max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$ first, which is a convex program with linear objective and a compact convex constraint $\|X^{\top} \boldsymbol{\alpha}\| \leq 1$ since $X^{\top}$ is injective ($X$ has full rank). Then solve the outer minimization $\min_{\boldsymbol{\alpha}  \in Q} \|\boldsymbol{\alpha}\|_\infty$ or $\min_{\boldsymbol{\alpha}  \in Q} \|\boldsymbol{\alpha}\|_2$, with convex and affine constraints $Q = \{\boldsymbol{\alpha}: \|X^{\top} \boldsymbol{\alpha}\| \leq 1, \boldsymbol{\alpha}^{\top} \boldsymbol{y} = v^{\star} \}$ to obtain $\bar{\boldsymbol{\alpha}}$. Furthermore, for $\boldsymbol{y}\neq0$, we have a unique solution for the inner maximization provided the set $\{\boldsymbol{\alpha}:\|X^{\top} \boldsymbol{\alpha}\| \leq 1\}$ is strictly convex.\footnote{Indeed, the set $\{ \boldsymbol{\alpha}: \|X^{\top} \boldsymbol{\alpha}\| \leq 1\}$ contains a ball around the origin and thus spans the directions where $\boldsymbol{\alpha}^{\top} \boldsymbol{y}$ increases. Thus, strict convexity of $\{ \boldsymbol{\alpha}: \|X^{\top} \boldsymbol{\alpha}\| \leq 1\}$ guarantees uniqueness of $\max_{\|X^{\top} \boldsymbol{\alpha}\| \leq 1} \boldsymbol{\alpha}^{\top} \boldsymbol{y}$.} In this case, $\bar{\boldsymbol{\alpha}}$ equals this unique solution (for any $p$). An example is the Euclidean norm $\| \cdot \| = \| \cdot \|_2$ where an easy derivation yields $\bar{\boldsymbol{\alpha}} = (XX^T)^{-1}\boldsymbol{y}/\sqrt{\boldsymbol{y}^{\top} (XX^T)^{-1}\boldsymbol{y}}$ (with $XX^T$ invertible since $X$ has full rank).\footnote{To see this, let $M := XX^T$. By Cauchy–Schwarz $\boldsymbol{\alpha}^\top \boldsymbol{y} = (X^{\top} \boldsymbol{\alpha})^{\top} (X^{\top} M^{-1} \boldsymbol{y}) \leq \|X^{\top} \boldsymbol{\alpha}\|_2 \|X^{\top} M^{-1} \boldsymbol{y}\|_2$ with equality iff $X^{\top} \boldsymbol{\alpha} =c X^{\top} M^{-1} \boldsymbol{y}$ for some $c>0$. The constraint $\|X^{\top} \boldsymbol{\alpha}\|_2 = 1$ (attained at optimality) yields $c = 1/\|X^{\top} M^{-1} \boldsymbol{y}\|_2$ and thus $X^{\top} \boldsymbol{\alpha} =c X^{\top} M^{-1} \boldsymbol{y}$ iff $\boldsymbol{\alpha} =cM^{-1} \boldsymbol{y} =M^{-1} \boldsymbol{y}/\|X^{\top} M^{-1} \boldsymbol{y}\|_2$ (since $X^{\top}$ is injective), completing the derivation.}

<!-- theorem-7: body_md -->
\label{th:saddle_point_form}
The robust risk of Wasserstein DRO linear regression ($2 \leq p < \infty$) with square loss equals
\begin{equation}\label{eq:saddle_point_form}
    n V_{\delta}(\boldsymbol{\beta}) = \sup_{\boldsymbol{\alpha} \geq 0} K(\boldsymbol{\beta},\boldsymbol{\alpha}) :=  \sup_{\boldsymbol{\alpha} \geq 0} \left [ n^{1/p} \delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha} \|_q+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i | - \alpha_i^2/4 \right ].
\end{equation}
                                                                              
Moreover, by applying the bijection $\gamma_i = \alpha_i^q$, where $1/p+1/q=1$, we get the following form:
\begin{equation}\label{eq:saddle_point_form_alternative}
n V_{\delta}(\boldsymbol{\beta}) = \sup_{\boldsymbol{\gamma} \geq 0} K(\boldsymbol{\beta},\boldsymbol{\gamma}^{1/q}) =
\sup_{\boldsymbol{\gamma} \geq 0} \left [ n^{1/p} \delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\gamma}\|_1^{1/q}+\sum_{i=1}^n \gamma^{1/q}_i |\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i | - \gamma_i^{2/q}/4 \right ]
\end{equation}
Crucially, $K(\boldsymbol{\beta},\boldsymbol{\gamma}^{1/q})$ is a convex-concave function with respect to $(\boldsymbol{\beta},\gamma)$. Thus, minimizing the robust risk is equivalent to a convex-concave optimization problem $\min_{\boldsymbol{\beta}} n V_{\delta}(\boldsymbol{\beta}) = \min_{\boldsymbol{\beta}} \sup_{\boldsymbol{\gamma} \geq 0} H(\boldsymbol{\beta},\boldsymbol{\gamma})$, where $H(\boldsymbol{\beta},\boldsymbol{\gamma}) := K(\boldsymbol{\beta},\boldsymbol{\gamma}^{1/q})$.

We first note that we do not need to prove \eqref{eq:saddle_point_form_alternative}. Indeed, \eqref{eq:saddle_point_form_alternative} follows from \eqref{eq:saddle_point_form} by using the bijection $\gamma_i = \alpha_i^q$. Moreover, $\|\boldsymbol{\gamma}\|_1^{1/q}$, $\gamma^{1/q}_i$ and $- \gamma_i^{2/q}$ are all concave in $\boldsymbol{\gamma}$ since $1 < q \leq 2$, so $K(\boldsymbol{\beta},\boldsymbol{\gamma}^{1/q})$ is concave in $\boldsymbol{\gamma} \geq 0$. Finally, $K(\boldsymbol{\beta},\boldsymbol{\alpha})$ is convex in $\boldsymbol{\beta}$ (since $\| \boldsymbol{\beta} \|_*$ and $|\boldsymbol{x}_i^T \boldsymbol{\beta} - y_i |$ are convex), so $K(\boldsymbol{\beta},\boldsymbol{\gamma}^{1/q})$ is also convex in $\boldsymbol{\beta}$. Thus, $K(\boldsymbol{\beta},\boldsymbol{\gamma}^{1/q})$ is a convex-concave function with respect to $(\boldsymbol{\beta},\boldsymbol{\gamma})$. Therefore, what remains to prove is \eqref{eq:saddle_point_form}. \Cref{th_supremum_robust_risk}~yields
    \begin{align*}
        n V_{\delta}(\boldsymbol{\beta}) = \max_{\boldsymbol{t} \geq 0, \|t\|_p^p \leq n\delta^p} \sum_{i=1}^n \Big (\|\boldsymbol{\beta}\|_* t_i+ |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | \Big)^2 = \\
        \max_{\boldsymbol{t} \geq 0, \|t\|_p \leq 1} \sum_{i=1}^n \Big (n^{1/p} \delta \|\boldsymbol{\beta}\|_* t_i+ |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | \Big)^2 = \\
        \max_{\boldsymbol{t} \geq 0, \|t\|_p \leq 1} \left [ \sum_{i=1}^n \sup_{\alpha_i \geq 0} \ \Big (n^{1/p} \delta \|\boldsymbol{\beta}\|_* t_i+ |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | \Big) \alpha_i - \alpha_i^2/4 \right ] =  \\
        \max_{\boldsymbol{t} \geq 0, \|\boldsymbol{t}\|_p \leq 1} \sup_{\alpha \geq 0} \left [ n^{1/p} \delta \|\boldsymbol{\beta}\|_* \boldsymbol{t}^{\top}\alpha+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | - \alpha_i^2/4 \right ] = \\
        \sup_{\alpha \geq 0} \left [ n^{1/p} \delta \|\boldsymbol{\beta}\|_*  \max_{t \geq 0, \|\boldsymbol{t}\|_p \leq 1} \{\boldsymbol{t}^{\top}\alpha \}+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | - \alpha_i^2/4 \right ] = \\
        \sup_{\boldsymbol{\alpha} \geq 0} \left [ n^{1/p} \delta \|\boldsymbol{\beta}\|_* \|\boldsymbol{\alpha}\|_q+\sum_{i=1}^n \alpha_i |\boldsymbol{x}_i^{\top} \boldsymbol{\beta} - y_i | - \alpha_i^2/4 \right ] = \sup_{\boldsymbol{\alpha} \geq 0} K(\boldsymbol{\beta},\boldsymbol{\alpha})
    \end{align*}
    where we used that $\|\boldsymbol{\alpha}\|_q = \max_{\|\boldsymbol{t}\|_p \leq 1} \boldsymbol{t}^{\top} \boldsymbol{\alpha} = \max_{\boldsymbol{t} \geq 0, \|\boldsymbol{t}\|_p \leq 1} \boldsymbol{t}^{\top} \boldsymbol{\alpha}$ since $\boldsymbol{\alpha} \geq 0$. This proves \eqref{eq:saddle_point_form} and completes the proof.
