<!-- theorem-1: body_md -->
\label{thm:main-intro}
Let $\Lambda \subset \mathbb R^2$ be a lattice with density $D(\Lambda) \in \mathbb Q$. Then the following two statements are equivalent:
\begin{enumerate}
    \item $\mathcal G(h_1,\Lambda)$ is a frame for $L^2(\mathbb R)$;
    \item $D(\Lambda) = \frac qp$, where $p,q \in \mathbb N$ are coprime with $q\ge p+2$.
\end{enumerate}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:ZZ_crit_cited}
    Let $g\in L^2(\mathbb R)$ and let $p,q\in \mathbb N$ be coprime with $p<q$. 
    Then $\mathcal G(g,\tfrac{p}{q}\mathbb Z\times \mathbb Z)$ is a Gabor frame if and only if the singular values of the matrix $\mathcal{Q}(x,\omega)\in \mathbb C^{q\times p}$ with entries 
    \begin{equation}
        \mathcal{Q}(x,\omega)_{rs} = \mathcal{Z} g( x+\tfrac{pr}{q},  \omega +\tfrac{s}{p}) ,\quad\, 0\leq r\leq q-1, 0\leq s\leq p-1,
    \end{equation}
    are contained in a compact interval in $(0,\infty)$ for almost all $x,\omega\in \mathbb R$.
<!-- end theorem-2 -->
<!-- proposition-1: body_md -->
\label{prop:elem_properties_calTp}
Let $\tau\in\mathbb H$,  $p\in\mathbb N$, and  $a,b\in\mathbb C$. The following statements hold:
\begin{enumerate}[(i)]
    \item 
    The space $\mathcal T_{p,a,b}(\tau)$ is a $p$-dimensional $\mathbb C$-vector space.
    \item The entire functions
    \begin{equation}
    \begin{split}
    z\mapsto
        {\vartheta\left[\begin{smallmatrix}
        (a+s)/p \\ b
    \end{smallmatrix}\right]}(pz,p\tau),\quad 0\leq s\leq p-1,
    \end{split}
    \end{equation}
    form a canonical basis of $\mathcal T_{p,a,b}(\tau)$. 
    \item Any nonzero $f\in\mathcal T_{p,a,b}(\tau)$ has exactly $p$ zeros
$z_1,\dots,z_p$, counted with multiplicity, in the half-open fundamental
period parallelogram
\[
\mathcal P_\tau:=[0,1)+\tau[0,1)
=\{x+y\tau:0\le x<1,\ 0\le y<1\},
\]
and their sum satisfies
\begin{equation}\label{eq:p_sum_zeros}
    \sum_{l=1}^p z_l
    \in \tfrac{p-2b}{2}+\tfrac{p-2a}{2}\tau+\mathbb Z+\tau\mathbb Z.
\end{equation}
Conversely, any multiset $z_1,\dots,z_p\in\mathcal P_\tau$ satisfying
\eqref{eq:p_sum_zeros} is the zero multiset of a nonzero function in
$\mathcal T_{p,a,b}(\tau)$.
\end{enumerate}
<!-- end proposition-1 -->
<!-- lemma-1: body_md -->
\label{lem:torsion-jet}
Let $\tau\in\mathbb H$, $q\in\mathbb N, \, q\geq 2$ and let 
$\Theta(z) = {\vartheta\left[\begin{smallmatrix}
        a \\ b
    \end{smallmatrix}\right]}(z,\tau)$ for some $a,b\in\mathbb C$.
Let $P$ be a
nonzero polynomial with
$$
\deg P\leq q-2.
$$
Then, for every $w\in\mathbb C$, the system of equations
\begin{equation}\label{eq:impossible_system}
    P(\delta_z)\Theta\left(w+\frac{j}{q}\right)=0,
\quad j=0,\dots,q-1,
\end{equation}
cannot be satisfied.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Suppose that \eqref{eq:impossible_system} holds for some $P\neq 0$ of degree $m\leq q-2$, i.e., 
$$

\label{lem:frobenius}
Let $p\in\mathbb N$, $b\in\mathbb C$ and $\tau\in\mathbb H$.
Let $\phi_0,\dots,\phi_{p-1}$ be a basis of $\mathcal T_{p,0,b}(\tau)$. Define

:=
\det[\phi_s(z_l)]_{l,s=0}^{p-1},
\quad
S(z):=\sum_{l=0}^{p-1}z_{l}.
$$
Then there exists a nonzero constant $C$ such that

\Delta(z_0,\dots,z_{p-1})
=
C\, {\vartheta\left[\begin{smallmatrix}

    \end{smallmatrix}\right]}(S(z), \tau) \prod\limits_{0\leq l <l'< p}{\vartheta\left[\begin{smallmatrix}
        1/2 \\ 1/2
    \end{smallmatrix}\right]}(z_{l'}-z_l,\tau).

\label{thm:primitive-derivative}

{
Since 
\begin{equation}
    \mathcal T_{p,a,b}\to \mathcal T_{p,0,b+a\tau}, \quad f\mapsto e^{-2\pi i a\cdot} f
\end{equation}
is an isomorphism and 
\begin{equation}
    \delta_z f(z) = \delta_z \left(e^{2\pi i az}g(z)\right)= e^{2\pi i a z}(a g(z) + \delta_z g(z)),\quad g(z) = e^{-2\pi i az} f(z),
\end{equation}
by Leibniz' rule there exists a polynomial $Q$ of degree at most $n$ such that
\begin{equation}\label{eq:P_vs_Q_for_a}
    P(\delta_z) f (z) = e^{2\pi i a z} Q(\delta_z) g(z),\quad g(z) = e^{-2\pi i az} f(z). 
\end{equation}
The polynomial $Q$ has degree $n$ because its leading coefficient equals the leading coefficient of $P$. Thus, we can assume without loss of generality that $a=0$ and that $P$ has a leading coefficient equal to one.

{Let $\phi_0,\dots,\phi_{p-1}$ be a basis of $\mathcal T_{p,0,b}(\tau)$.}  Suppose, to
the contrary, that the stated map is not injective.  Then the
$q\times p$ matrix
$$
R_{r,s}:={{P(\delta_z)}}\phi_s\left(w+\tfrac{r}{q}\right),
\quad
r=0,\dots,q-1,
\quad
s=0,\dots,p-1,
$$
has rank strictly smaller than $p$.  Hence, all of its $p\times p$ minors
vanish.

For $j\in\mathbb Z$, define the consecutive minor
$$
M_j:=
\det\left[
{P(\delta_z)}\phi_s\left(w+\tfrac{j+l}{q}\right)
\right]_{l,s=0}^{p-1}.
$$
Each $P(\delta_z)\phi_s$ is $1$-periodic, so reducing the indices
$j+l$ modulo $q$ only permutes the rows. 
Rank failure gives
$$
M_j=0,
\quad j=0,\dots,q-1.
$$
Let
$$
\Delta(z_0,\dots,z_{p-1})
:=
\det[\phi_s(z_l)]_{l,s=0}^{p-1}.
$$
By Lemma~\ref{lem:frobenius}, we have
$$
\Delta(z_0,\dots,z_{p-1})
=
C\, {\vartheta\left[\begin{smallmatrix}
        1/2+p/2 \\ 1/2+b+p/2
    \end{smallmatrix}\right]}(S(z), \tau)V(z_0,\dots,z_{p-1}) ,

where
$$
S(z_0,\dots,z_{p-1}):=\sum_{l=0}^{p-1} z_l,
\quad
V(z_0,\dots,z_{p-1})
:=\prod\limits_{0\leq l<l'< p}{\vartheta\left[\begin{smallmatrix}
        1/2 \\ 1/2
    \end{smallmatrix}\right]}(z_{l'}-z_l,\tau).
$$
We apply $P(\delta_z)$ to each row of $[\phi_s(z_l)]_{l,s=0}^{p-1}$ with respect to $z_l$ and denote this by 
$$
P(\delta_{z_0})\cdots P(\delta_{z_{p-1}}) [\phi_s(z_l)]_{l,s=0}^{p-1}.
$$
We then restrict the argument to 
$$
z_l=w+\tfrac{l}{q},
\quad l=0,\dots,p-1.
$$
On the left, multilinearity of the determinant gives
$$
\left.
P(\delta_{z_0})\cdots P(\delta_{z_{p-1}})\Delta
\right|_{z_l=w+l/q}
=
\det\left[
P(\delta_z)\phi_s\left(w+\frac{l}{q}\right)
\right]_{l,s=0}^{p-1}

$$
At the restricted points, we have
$$
z_{l'}-z_l=\tfrac{l'-l}{q}.
$$
As $1\leq l'-l\leq p-1<q$, none of these differences lies in
$\mathbb Z+\tau\mathbb Z$.  Since ${\vartheta\left[\begin{smallmatrix}
        1/2 \\ 1/2
    \end{smallmatrix}\right]}(\cdot,\tau)$ has its only zero at the
origin modulo $\mathbb Z+\tau\mathbb Z$, it follows that
$$
V_0:=
V\left(w,w+\tfrac1q,\dots,w+\tfrac{p-1}{q}\right)
=
\prod\limits_{0\leq l <l'< p}{\vartheta\left[\begin{smallmatrix}
        1/2 \\ 1/2
    \end{smallmatrix}\right]}(\tfrac{l'-l}{q},\tau)
\neq0.
$$
The above number is independent of $w$.

\begin{equation}
    \Theta(z)={\vartheta\left[\begin{smallmatrix}
        1/2+p/2 \\ 1/2+b+p/2
    \end{smallmatrix}\right]}(z,\tau).
\end{equation}
By Leibniz' rule, after the restriction $z_l=w+l/q$, every derivative of
$V$ is evaluated at the fixed tuple of differences
$\{(l'-l)/q\}_{l<l'}$ and is therefore a constant.  Hence, there are
constants $c_0,\dots,c_{np}$ such that
$$
M(w)
=
\sum_{d=0}^{np}c_d
\delta_z^d\Theta \left(pw+\tfrac{p(p-1)}{2q}\right).
$$
Equivalently,
$$
M(w)=Q(\delta_z)\Theta \left(pw+\tfrac{p(p-1)}{2q}\right),
$$
for a polynomial $Q$ of degree at most $np$.

The coefficient $c_{np}$ of the derivative of the highest order  comes only from the term in which the highest order
derivative part of every operator $P(\delta_{z_l})$
falls on the factor $\Theta(S(z_0,\dots,z_{p-1}))$ and no derivative falls on $V$. Therefore,
$$
c_{np}
=
C\prod_{0\leq l<l'< p}
{\vartheta\left[\begin{smallmatrix}
        1/2 \\ 1/2
    \end{smallmatrix}\right]}\left(\tfrac{l'-l}{q},\tau\right)
=
CV_0
\neq0.
$$
Thus, $Q\neq0$ and $\deg Q={{n}}p$.

Since $M_j=M(w+j/q)$, the vanishing of the consecutive minors gives
$$
Q(\delta_z)\Theta\left(
pw+\tfrac{pj}{q}+\tfrac{p(p-1)}{2q}
\right)=0,
\quad j=0,\dots,q-1.
$$
Since $(p,q)=1$, for each $j$ there exists a residue $j'\in\{0,\dots,q-1\}$
and an integer $n_j$ such that
$$
\frac{pj}{q}=\frac{j'}{q}+n_j.
$$
Let
$$
w':=pw+\frac{p(p-1)}{2q}.
$$
The preceding vanishing can be expressed as
$$
Q(\delta_z) \Theta\left(w'+\tfrac{j'}{q}+n_j\right)=0,
\quad j'=0,\dots,q-1,
$$
after reordering the residues. 
By \eqref{eq:thetac_quasip}, $\Theta(z+n_j)=e^{2\pi i(1/2+p/2)n_j}\Theta(z)$ for all $z\in\mathbb C$, and hence $Q(\delta_z)\Theta(z+n_j)=e^{2\pi i(1/2+p/2)n_j}Q(\delta_z)\Theta(z)$. Since this factor is nonzero, it follows that
$$
Q(\delta_z)\Theta\left(w'+\tfrac{j}{q}\right)=0,
\quad j=0,\dots,q-1.
$$
Finally,
$$
\deg Q={{n}}p\leq q-2.
$$
Since $\Theta={\vartheta\left[\begin{smallmatrix}
        1/2+p/2 \\ 1/2+b+p/2
    \end{smallmatrix}\right]}(\cdot,\tau)$ and $q\geq np+2\geq 2$, this contradicts Lemma~\ref{lem:torsion-jet}. Hence, the original derivative evaluation map is injective.

          e^{2\pi i x\omega }\mathcal{Z} (\widehat{V}_{\eta} \widehat{D}_{\nu}\Tilde{h}_d)(\omega, -x)
     &= \nu^{-(d+1/2)}e^{2\pi i x\omega }\sum\limits_{k\in\mathbb Z} (k+\omega)^d e^{\pi i (\eta+i/\nu^2) (k+\omega)^2} e^{2\pi i k x} \\
     &=\nu^{-(d+1/2)} \sum\limits_{k\in\mathbb Z} (k+\omega)^d e^{\pi i (\eta+i/\nu^2) (k+\omega)^2} e^{2\pi i (k+\omega) x}\\
     & = \nu^{-(d+1/2)}\delta_z^d {\vartheta\left[\begin{smallmatrix}
        \omega \\ 0
    \end{smallmatrix}\right]}(x,\tau),\quad \tau = \eta+i/\nu^2.
\end{split}
\end{equation}
Let $P(X) = \sum_{d=0}^n \nu^{-(d+1/2)} c_d'X^d$. Then 
\begin{equation}
    e^{2\pi i x\omega }\mathcal{Z} (\widehat{V}_{\eta} \widehat{D}_{\nu}\Tilde{g})(\omega, -x) = \sum\limits_{d=0}^n \nu^{-(d+1/2)} c_d'\delta_z^d{\vartheta\left[\begin{smallmatrix}
        \omega \\ 0
    \end{smallmatrix}\right]}(x,\tau) = P(\delta_z){\vartheta\left[\begin{smallmatrix}
        \omega \\ 0
    \end{smallmatrix}\right]}(x,\tau)
\end{equation}
and 
\begin{equation}
    \mathcal{Q}(x,\omega)_{rs} = P(\delta_z){\vartheta\left[\begin{smallmatrix}
        \tfrac{p\omega+ s}{p} \\ 0
    \end{smallmatrix}\right]}(p(\tfrac{x}{p}+\tfrac{r}{q}),\tau).
\end{equation}
\textbf{Step 4.}
We apply 
Theorem~\ref{thm:primitive-derivative} for the space $\mathcal T_{p,p\omega,0}(\tau/p)$. 
Define
$$
\widetilde P(X):=P(X/p).
$$
We have $\deg \widetilde P=n$. By Proposition \ref{prop:elem_properties_calTp}~(ii)
and the chain rule, 
the matrix $\mathcal{Q}(x,\omega)$ is the matrix representation of the linear map
\begin{equation}
    \mathcal T_{p,p\omega ,0}(\tfrac{\tau}{p})\to \mathbb C^{q},\quad f \mapsto
\left(\widetilde P(\delta_z)f\left(\tfrac{x}{p}+\tfrac{r}{q}\right)\right)_{r=0}^{q-1}.
\end{equation}
Since $(p,q)=1$ and
$q\geq np+2$, the matrix $\mathcal{Q}(x,\omega)$ has full rank for every $x$ and every
$\omega$.  

With the same quasi-periodicity and Wiener-amalgam arguments, this implies that $\mathcal G(g,A\mathbb Z^2)$ is a frame.
