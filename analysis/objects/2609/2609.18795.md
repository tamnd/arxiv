For a solution $u$ of \eqref{main}-\eqref{initial} and for a given unit vector $\xi \in\mathbb{S}^{N-1}$ $($the unit Euclidean sphere of $\mathbb{R}^N$$)$, a quantity $\omega(\xi)\in [0,+\infty]$ is called spreading speed $($in the direction~$\xi$$)$ if
\begin{eqnarray}\label{conv:ss}
\left\{\begin{array}{lll}
u(t,ct\xi)\rightarrow 1 &\hbox{as $t\rightarrow +\infty$, $\ $for every $0\le c<\omega(\xi)$},\vspace{3pt}\\
u(t,ct \xi)\rightarrow 0 &\hbox{as $t\rightarrow +\infty$, $\ $for every $c>\omega(\xi)$}.
\end{array}
\right.
\end{eqnarray}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:W}
We say that a solution $u$ to~\eqref{main} admits a spreading set
 $\mathcal{W}\subset\mathbb{R}^N$ if~$\mathcal{W}$ coincides  with the interior of its closure and satisfies 
\begin{equation}\label{subset}
\lim_{t\to+\infty}\,\Big(\min_{x\in C}u(t, tx)\Big)=1\, \text{ for every non-empty compact set }C\subset \mathcal{W},
\end{equation}
and
\begin{equation}\label{superset}
\lim_{t\to+\infty}\,\Big(\max_{x\in C}u(t,tx)\Big)=0\, \text{ for every non-empty compact set }C\subset \hbox{\rm int}(\mathbb{R}^N\!\setminus\!\mathcal{W}).
\end{equation}
If only \eqref{subset} $($resp. \eqref{superset}$)$ holds, we say that $\mathcal{W}$ is a spreading subset $($resp. superset$)$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:pf}
A pulsating traveling front\ connecting~$1$ to~$0$ in the direction $e$ with a speed $c^*(e)$ is an entire $($defined for all $t\in\mathbb{R}$$)$ classical solution $\phi_e:\mathbb{R}\times\mathbb{R}^N\to(0,1)$ of~\eqref{main} of the type
$$\phi_e(t,x)=U_e(x,x\cdot e-c^*(e)t),$$
where the function $(x,z)\in\mathbb{R}^N\times\mathbb{R} \mapsto U_e(x,z)$ is periodic in the $x$-variable and satisfies
$$\lim_{z\to-\infty}U_e(x,z)=1,\ \ \lim_{z\to+\infty}U_e(x,z)=0,\ \hbox{uniformly with respect to $x\in\mathbb{R}^N$}.$$

<!-- hypothesis-1: body_md -->
\label{hyp:KPP}
For every $(x,s)\in\mathbb{R}^N\times(0,1)$, $0<f(x,s)\le f_s(x,0) s$.
<!-- end hypothesis-1 -->
<!-- hypothesis-2: body_md -->
\label{hyp:ignition}
There exist $0<\sigma<\varsigma<1$ such that $f=0$ in $\mathbb{R}^N\times [0,\sigma]$, $f\ge0$ in~$\mathbb{R}^N\times[\sigma,1]$, $\max_{\mathbb{R}^N} f(\cdot,s)>0$ for every $s\in (\sigma,1)$, $f>0$ in $\mathbb{R}^N\times(\varsigma,1)$, and $f(x,\cdot)$ is non-increasing in $[\varsigma,1]$ for every $x\in\mathbb{R}^N$.
<!-- end hypothesis-2 -->
<!-- hypothesis-3: body_md -->
\label{hyp:bistable}
There exists $\sigma\in(0,1/2)$ such that $f(x,\cdot)$ is non-increasing in $[0, \sigma]$ and $[1-\sigma, 1]$ for every $x\in\mathbb{R}^N$, and $\min_{\mathbb{R}^N} f(\cdot,s)< 0$ for every $s\in(0,\sigma]$. Moreover, for every $e\in\mathbb{S}^{N-1}$, there is a pulsating front $\phi_e(t,x)=U_e(x,x\cdot e-c^*(e)t)$ connecting~$1$ to~$0$ with $c^*(e)>0$.
<!-- end hypothesis-3 -->
<!-- proposition-1: body_md -->
\label{pro:rho}
Assume that one of Hypotheses~$\ref{hyp:KPP}$-$\ref{hyp:bistable}$ holds. Then there exist $\theta\in (0,1)$ and $\rho>0$ such that  for any $x_0\in\mathbb{R}^N$, if 
$$\theta\,\1_{B_{\rho}(x_0)}\le u_0\le 1\ \hbox{ in $\mathbb{R}^N$},$$
where $B_{\rho}(x_0)$ denotes the open Euclidean ball of center $x_0$ and radius $\rho$, then the solution $u(t,x)$ of \eqref{main} with initial condition $u_0$ satisfies $u(t,x)\rightarrow 1$ as $t\rightarrow +\infty$ locally uniformly with respect to $x\in\mathbb{R}^N$. Such a solution is then said to be {\rm invading}.
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{pro:suf-conditions}
Assume one of Hypotheses~\ref{hyp:KPP}-\ref{hyp:bistable}. Any of the following conditions ensures~\eqref{supt-plane}:
\begin{itemize}
\item[(i)] the spreading set $\mathcal{W}_0$ defined by~\eqref{w0}-\eqref{W0-wulff} is differentiable at any boundary point;
\item[(ii)] for any $\xi\in{\mathbb{S}}^{N-1}$, there exists a unique $e\in{\mathbb{S}}^{N-1}$ such that $e\cdot\xi>0$ and
$$\frac{c^*(e)}{e\.\xi}=\min_{\genfrac{}{}{0pt}{}{e'\in{\mathbb{S}}^{N-1}}{e'\.\xi>0}}\frac{c^*(e')}{e'\.\xi}=\omega_0(\xi).$$
\end{itemize}

\label{Th:Ass}
Assume one of Hypotheses~\ref{hyp:KPP}-\ref{hyp:bistable} and  that~\eqref{supt-plane} holds. 
If $U_{\rho}\neq \emptyset$, with $\rho>0$ given in Proposition~\ref{pro:rho}, and if
\begin{equation}\label{BUS}

\label{theo:ass}
Assume that all conditions of Theorem~\ref{Th:Ass} hold.
\begin{enumerate}
\item[(i)] If the initial support $U$ is V-shaped, then the spreading set for $u$ is given by
$$\mathcal{W}_V:=\Big\{x\in\mathbb{R}^N:\!\displaystyle\inf_{e\in\mathcal{L}(\mathcal{B}(U))}(x\cdot e- c^*(e))<0\Big\}=\big\{r\xi:\xi\in\mathbb{S}^{N-1},\, 0\le r< \omega_V(\xi)\big\},$$
where 
\begin{eqnarray}\label{ASS-V}
\omega_V(\xi):=\left\{\begin{array}{lll}
\displaystyle\sup_{e\in\mathcal{L}(\mathcal{B}(U))} \frac{c^*(e)}{e\cdot \xi}\in (0,+\infty) & \hbox{if $\xi\in \mathcal{B}(U)$},\\
+\infty & \hbox{if $\xi\in\mathcal{U}(U)=\mathbb{S}^{N-1}\!\setminus\!\mathcal{B}(U)$}.
\end{array}
\right.
\end{eqnarray}
\item[(ii)] If the initial support $U$ is $\Lambda$-shaped, then the spreading set for $u$ is given by
$$\mathcal{W}_{\Lambda}:=\Big\{x\in\mathbb{R}^N:\!\displaystyle\inf_{e\in\mathcal{L}(\mathcal{U}(U))}(x\cdot e+ c^*(-e))> 0\Big\}=\big\{r\xi:\xi\in\mathbb{S}^{N-1},\, 0\le r<\omega_{\Lambda}(\xi)\big\},$$
where
\begin{eqnarray}\label{omega-Lambda}
\omega_{\Lambda}(\xi):=\left\{\begin{array}{ll}
\displaystyle\inf_{\genfrac{}{}{0pt}{}{e\in\mathcal{L}( \mathcal{U}(U))}{e\.\xi<0}}\frac{c^*(-e)}{-e\cdot \xi}\in (0,+\infty) & \hbox{if $\xi\in\mathcal{B}(U)$},\\
+\infty &  \hbox{if $\xi\in\mathcal{U}(U)=\mathbb{S}^{N-1}\!\setminus\!\mathcal{B}(U)$}.
\end{array}
\right.
\end{eqnarray}
\end{enumerate}

<!-- proof-1: body_md -->
We will show that (i) $\Rightarrow$ (ii) $\Rightarrow$ \eqref{supt-plane}.

For (i) $\Rightarrow$ (ii), take any $\xi\in \mathbb{S}^{N-1}$. Since $\partial \mathcal{W}_0$ is differentiable at the boundary point $\omega_0(\xi)\xi$, there is a unique exterior normal $\nu\in \mathbb{S}^{N-1}$ at $\omega_0(\xi)\xi$. In other words, the hyperplane $\{x\in\mathbb{R}^N:\, (x-\omega_0(\xi)\xi)\cdot \nu=0\}$ is the unique supporting hyperplane at $\omega_0(\xi)\xi$. On the other hand, by~\eqref{w0} and~\eqref{W0-wulff}, there is $e\in\mathbb{S}^{N-1}$ such that $\omega_0(\xi)\xi\cdot e=c^*(e)$ and $x\cdot e<c^*(e)$ for all $x\in\mathcal{W}_0$. Then, by the uniqueness of the supporting hyperplane at $\omega_0(\xi)\xi$, one has $\nu=e$ and $\omega_0(\xi)\xi\cdot \nu=c^*(\nu)$. Therefore, the minimizer $e$ in \eqref{w0} is unique, showing (ii).

For  (ii) $\Rightarrow$ \eqref{supt-plane}, assume that \eqref{supt-plane} does not hold. Then, there exists $e\in{\mathbb{S}}^{N-1}$ such that
\begin{equation}\label{under-supt-plane}
\forall\,x\in \partial\mathcal{W}_0,\quad x\. e<c^*(e),
\end{equation}
Notice that \eqref{under-supt-plane} also holds  for all $x\in\mathcal{W}_0$, by~\eqref{W0-wulff}. We then set 
$$c:=\max_{x\in \overline{\mathcal{W}_0}}\,x\.e<c^*(e).$$
Let $z\in \overline{\mathcal{W}_0}$ be such that $z\.e=c$. One has that $z\in\partial \mathcal{W}_0$. By \eqref{w0}, there exists $e'\in{\mathbb{S}}^{N-1}$ such that $z\.e'=c^*(e')>0$, that is, $e'$ is a minimizer in~\eqref{w0} for the direction $\widehat{z}=z/|z|$. Recalling that $z\.e=c<c^*(e)$, we see that $e'\neq e$, that is, $e'\. e<1$. One can also observe that the hyperplanes $\{x\in\mathbb{R}^N:x\cdot e=c\}$ and $\{x\in\mathbb{R}^N:x\cdot e'=c^*(e')\}$ are two different supporting hyperplanes of $\mathcal{W}_0$ containing the point $z\in\partial\mathcal{W}_0$, which implies in particular that $\partial\mathcal{W}_0$ is not differentiable at $z$, as seen above in the proof of (i)~$\Rightarrow$~(ii).

We then consider, for $n\in\mathbb{N}\setminus\{0\}$, $x_n:=z+\frac1n(e-e')$. We find that
$$x_n\. e=c+\frac{1-e'\.e}n>c,$$
which shows that $x_n\notin\overline{\mathcal{W}_0}$, hence there exists $e_n\in{\mathbb{S}}^{N-1}$ such that $x_n\. e_n>c^*(e_n)$. As a consequence, we get
$$c^*(e_n)<x_n\. e_n=z\.e_n+\frac{e-e'}n\.e_n\leq c^*(e_n)+\frac{e-e'}n\.e_n,$$
which means that $e\. e_n>e'\. e_n$ for all $n\in\mathbb{N}\setminus\{0\}$. From this, we deduce that the sequence $(e_n)_{n\in\mathbb{N}}$ has to converge (up to subsequences) towards a limit $e''\neq e'$ (indeed, otherwise, if $e''=e'$, then $e\cdot e'\ge e'\cdot e'=1$, a contradiction). Using again the continuity of $c^*(e)$ and passing to the limit as $n\to+\infty$ in $c^*(e_n)<x_n\cdot e_n$, we finally infer that
$$c^*(e'')\leq z\.e'',$$
and finally $0<c^*(e'')=z\cdot e''$, since $z\in \overline{\mathcal{W}_0}$. Recalling that $\widehat{z}=z/|z|$, we have thereby found that both $e'$ and $e''$ satisfy $\widehat{z}\cdot e'>0$, $\widehat{z}\cdot e''>0$,
$$\frac{c^*(e')}{\widehat{z}\.e'}=\frac{c^*(e'')}{\widehat{z}\.e''}=|z|=\min_{\genfrac{}{}{0pt}{}{\widetilde e\in S^{N-1}}{\widehat{z}\cdot\widetilde e>0}}\frac{c^*(\widetilde e)}{\widehat{z}\cdot\widetilde e},$$
that is, (ii) fails. The proof of Proposition~\ref{pro:suf-conditions} is thereby complete.

<!-- lemma-1: body_md -->
\label{lemma:P}
For any $\mathcal{A}\subset{\mathbb{S}}^{N-1}$ such that $\mathcal{C}(\mathcal{A})$ is convex, there holds
$$\mathcal{C}(\overline{\mathcal{A}})=\big\{x\in\mathbb{R}^N:\ x\cdot e\ge 0\text{ \ for all \ } e\in\mathcal{L}(\mathcal{A})\big\}.$$
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
When $\mathcal{A}={\mathbb{S}}^{N-1}$, the desired conclusion holds at once, since $\mathcal{C}({\mathbb{S}}^{N-1})=\mathbb{R}^N$ and $\mathcal{L}({\mathbb{S}}^{N-1})=\emptyset$. One can then assume in the sequel that $\mathcal{A}\subsetneqq{\mathbb{S}}^{N-1}$, and the convexity of~$\mathcal{C}(\mathcal{A})$ then implies that $\mathcal{L}(\mathcal{A})\neq\emptyset$. 

It is obvious that 
$$\mathcal{C}(\overline{\mathcal{A}})\subset \big\{x\in\mathbb{R}^N:\, \forall\,e\in\mathcal{L}(\mathcal{A}),\ x\cdot e\ge 0\big\}.$$

For the reversed inclusion, assume by contradiction that there is $x_0\in\mathbb{R}^N$ such that
\begin{equation}\label{x0}
\inf_{e\in\mathcal{L}(\mathcal{A})} \{x_0\cdot e\}\ge 0\ \ \hbox{and}\ \ x_0\not\in\mathcal{C}(\overline{\mathcal{A}}).
\end{equation}
Notice that $\mathcal{C}(\overline{\mathcal{A}})=\overline{\mathcal{C}(\mathcal{A})}$ is closed and non-empty (it always contains the origin~$0$), whence there is $y_0\in\mathcal{C}(\overline{\mathcal{A}})$ such that $|x_0-y_0|=d(x_0,\mathcal{C}(\overline{\mathcal{A}}))>0$. Since $\mathcal{C}(\overline{\mathcal{A}})=\overline{\mathcal{C}(\mathcal{A})}$ is closed and convex, the point $y_0$ is the projection of $x_0$ onto $\mathcal{C}(\overline{\mathcal{A}})$, whence
\begin{equation}\label{PbP}
\mathcal{C}(\overline{\mathcal{A}})\subset \big\{x\in\mathbb{R}^N:(y_0-x_0)\cdot (x-y_0)\ge 0\big\}.
\end{equation}
Then, it follows from the definition of $\mathcal{C}(\overline{\mathcal{A}})$ and \eqref{PbP} that 
$$ry_0\in \big\{x\in\mathbb{R}^N:(y_0-x_0)\cdot (x-y_0)\ge 0\big\} \ \ \hbox{for all $r\ge 0$},$$
that is, $(r-1)(y_0-x_0)\cdot y_0\ge 0$ for all $r\ge 0$. As a consequence,
\begin{equation}\label{y0-x0}
(y_0-x_0)\cdot y_0=0.
\end{equation}
Therefore, by following \eqref{PbP} and \eqref{y0-x0},
$$\mathcal{C}(\mathcal{A})\subset\mathcal{C}(\overline{\mathcal{A}})\subset \big\{x\in\mathbb{R}^N:(y_0-x_0)\cdot x\ge 0\big\},$$
which means that $\widehat{y_0-x_0}=(y_0-x_0)/|y_0-x_0|\in\mathcal{L}(\mathcal{A})$ and $(y_0-x_0)\cdot x_0\ge 0$ by \eqref{x0}. It follows from \eqref{y0-x0} again that  $0\le (y_0-x_0)\cdot x_0=(y_0-x_0)\cdot (x_0-y_0)$, and hence $x_0=y_0$, a contradiction with $x_0\neq y_0$. The proof of Lemma~\ref{lemma:P} is thereby complete.

<!-- lemma-2: body_md -->
\label{lem:distW}
Let $K$ be a compact convex set containing a ball $B_\delta$ centered at the origin, with $\delta>0$. Then
$$\forall\,h\geq 1,\quad\min_{x\in K}\ d\big(x,\mathbb{R}^N\!\setminus\!(hK)\big)\geq (h-1)\delta.$$
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
For any $e\in\mathbb{S}^{N-1}$, let $\tau(e):=\min_{x\in K}x\cdot e$, which is a well defined real number since $K\neq \emptyset$ is compact. Define
$$K'=\Big\{x\in\mathbb{R}^N:\inf_{e\in\mathbb{S}^{N-1}}(x\cdot e-\tau(e))\ge 0\Big\}.$$
Obviously, $K\subset K'$. Assume now by way of contradiction that $K'\setminus K\neq \emptyset$. Then, for $x_0\in K'\setminus K$, let $y_0\neq x_0$ be the projection of $x_0$ onto the closed convex set $K$. One has
$$\widehat{y_0-x_0}\cdot(x-y_0)=\frac{y_0-x_0}{|y_0-x_0|}\cdot (x-y_0)\ge 0\ \hbox{ for all $x\in K$}.$$
Since $x_0\in K'$ and $x_0\neq y_0$, it follows that
$$\widehat{y_0-x_0}\cdot x_0-\tau(\widehat{y_0-x_0})\ge 0>-|y_0-x_0|=\widehat{y_0-x_0}\cdot (x_0-y_0).$$
Hence, for all $x\in K$,
$$\tau(\widehat{y_0-x_0})<\widehat{y_0-x_0}\cdot y_0\le\widehat{y_0-x_0}\cdot x,$$
which contradicts the definition of $\tau(\widehat{y_0-x_0})$. Thus, $K=K'$.

Then, for any $h\ge1$,
$$\mathbb{R}^N\setminus(hK)=\Big\{x\in\mathbb{R}^N:\, \inf_{e\in\mathbb{S}^{N-1}}(x\cdot e-h\tau(e))< 0\Big\}.$$
We deduce from $B_\delta\subset K$ that $\tau(e)\leq -\delta$ for all $e\in{\mathbb{S}}^{N-1}$. Then, we get that, for all $x\in K$,
$$\inf_{e\in\mathbb{S}^{N-1}}(x\cdot e-h\tau(e))\ge \inf_{e\in\mathbb{S}^{N-1}}(x\cdot e-\tau(e))+ (h-1)\delta\ge (h-1)\delta.$$
The result follows.

<!-- lemma-3: body_md -->
\label{lemma:approximation}
Let $K$ be a non-empty convex compact set, and let $\varepsilon>0$.
\begin{itemize}
\item[(i)] The set $K^\varepsilon:=K+\overline{B_{\varepsilon}}\supset K$ is a convex compact set with boundary of class~$C^1$; moreover, $K^\varepsilon \rightarrow K$ as $\varepsilon\rightarrow 0$, in the sense that $d_H(K^\varepsilon,K)\to0$ as $\varepsilon\to0$;
\item[(ii)]
There exists $l\in\mathbb{N}$, depending on $K,\varepsilon,N$, and a compact polyhedron $P^\varepsilon$ with at most $l$ faces such that
\begin{equation}\label{KPK}
K^{\varepsilon/2}\subset P^\varepsilon\subset K^\varepsilon;
\end{equation}
there are then some $e_i\in\mathbb{S}^{N-1}$ and $a_i\in\mathbb{R}$ such that~$P^\varepsilon$ can then be written as
\begin{equation}\label{Pe}
P^\varepsilon=\bigcap_{i=1}^l\big\{x\in\mathbb{R}^N:\, x\cdot e_i\le a_i\big\};
\end{equation}
\item[(iii)] There exists $m\in\mathbb{N}$, depending on $K,\varepsilon,N$, 
such that, for any $z\in\partial K^\varepsilon$, there is a compact polyhedron $P^{\varepsilon,z}$ with at most $m$ faces such that
\begin{equation}\label{Pez1}
B_{\varepsilon/2}(z-\varepsilon \nu)\subset K^{\varepsilon/2}\subset P^{\varepsilon,z} \subset K^\varepsilon\ \hbox{ and }\ z\in P^{\varepsilon,z},
\end{equation}
where $\nu$ is the exterior unit normal of $\partial K^\varepsilon$ at $z$; the polyhedron $P^{\varepsilon,z}$ can then be written as
\begin{equation}\label{Pez2}
P^{\varepsilon,z}=\bigcap_{i=1}^m\big\{x\in\mathbb{R}^N:\, x\cdot f_i\le b_i\big\},
\end{equation}
for some $f_i\in\mathbb{S}^{N-1}$ and $b_i\in\mathbb{R}$ depending on $z$.
\end{itemize}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
(i) Clearly, $K\subset K^\varepsilon$, $K^\varepsilon \rightarrow K$ as $\varepsilon\rightarrow 0$, and it is also readily seen that $K_{\varepsilon}$ is convex and compact. Let $z\in\partial K^\varepsilon$. Then $d(z,K)=\varepsilon$ and there is $y\in \partial K$ such that $|z-y|=\varepsilon$. As for \eqref{PbP}, $y$ is the projection of $z$ onto the closed convex set $K$ and
$$K\subset\big\{x\in\mathbb{R}^N:x\.\nu\leq y\.\nu\big\}, \hbox{ where }\nu:=\widehat{z-y}=\frac{z-y}{|z-y|}.$$
From this, it follows that 
$$\big\{x\in\mathbb{R}^N:x\.\nu> z\.\nu\big\}\subset \mathbb{R}^N\setminus K^\varepsilon.$$
On the other hand, $B_\varepsilon(y)\subset K^\varepsilon$. Thus, $K^\varepsilon$ fulfills both an interior and an exterior ball condition at $z$ (with balls with uniform radius $\varepsilon$ independent of $z\in\partial K^\varepsilon$). Therefore,~$\partial K^\varepsilon$~is of class $C^1$. Furthermore, at $z$, $\nu$ is the outward unit normal to $\partial K^{\varepsilon}$.

(ii) Fix $\varepsilon>0$. Notice that the conclusions~(ii) and~(iii) hold at once when $N=1$, in which case the sets $K$, $K^{\varepsilon/2}$, $K^\varepsilon$, $P^\varepsilon$ and $P^{\varepsilon,z}$ are all non-empty segments, and $l=m=2$ (faces are just points in dimension $N=1$). We can assume in the sequel that $N\ge2$.

The non-empty compact convex set $K^{\varepsilon/2}$ can be written as
$$K^{\varepsilon/2}=\bigcap_{i\in I}\mathcal{F}_i,$$
where each $\mathcal{F}_i$ is a closed affine half-space and $(\partial\mathcal{F}_i)_{i\in I}$ is the family of all supporting hyperplanes of $K^{\varepsilon/2}$. Since $K^{\varepsilon/2}\subset{\rm{int}}(K^\varepsilon)$ and the non-empty compact set
$$K^{2\varepsilon}\cap\overline{\mathbb{R}^N\!\setminus\!K^{\varepsilon}}=K^{2\varepsilon}\cap(\underbrace{\mathbb{R}^N\!\setminus\!{\rm{int}}(K^\varepsilon)}_{\subset\,\mathbb{R}^N\setminus K^{\varepsilon/2}})$$
is then covered by the union $\cup_{i\in I}(\mathbb{R}^N\!\setminus\!\mathcal{F}_i)$ of open sets $\mathbb{R}^N\!\setminus\!\mathcal{F}_i$, there is a (positive) integer $l=l(K,\varepsilon,N)$ depending in general on $(K,\varepsilon,N)$ and a family $\{i_1,\cdots,i_l\}\subset I$ such that
$$K^{2\varepsilon}\cap\overline{\mathbb{R}^N\!\setminus\!K^\varepsilon}\ \subset\ \bigcup_{j=1}^l\,(\mathbb{R}^N\!\setminus\!\mathcal{F}_{i_j}).$$
Therefore,
$$K^{\varepsilon/2}=\bigcap_{i\in I}\mathcal{F}_i\ \subset\ \underbrace{\bigcap_{j=1}^l\mathcal{F}_{i_j}}_{=:P^\varepsilon}\ \subset\ (\mathbb{R}^N\!\setminus\!K^{2\varepsilon})\cup{\rm{int}}(K^\varepsilon).$$
The set $P^\varepsilon$ is a closed polyhedron of at most $l$ faces, that is,~\eqref{Pe} holds. If $P^\varepsilon$ were included in $\mathbb{R}^N\!\setminus\!K^{2\varepsilon}$, then so would be $K^{\varepsilon/2}$, which is impossible since $K^{\varepsilon/2}\subset K^{2\varepsilon}$. Therefore, $P^\varepsilon\subset{\rm{int}}(K^\varepsilon)\subset K^\varepsilon$, and~\eqref{KPK} is proved.

(iii) Fix $\varepsilon>0$, and consider the closed polyhedron $P^\varepsilon$ given in part~(ii). Since $P^\varepsilon$ has at most $l=l(K,\varepsilon,N)$ faces, there is an integer $k=k(K,\varepsilon,N)$ depending on~$l$ and~$N$, and therefore depending on $(K,\varepsilon,N)$, such that $P^\varepsilon$ has at most~$k$ vertices, whence there are some pairwise distinct points $q_1,\cdots,q_{k'}$ (with $k'\le k$) of $P^\varepsilon$ such that~$P^\varepsilon$ is equal to the convex hull of $\{q_1,\cdots,q_{k'}\}$.

Consider now any point $z\in\partial K^\varepsilon$, and let $P^{\varepsilon,z}$ be the convex hull of $\{q_1,\cdots,q_{k'},z\}$, that is, the convex hull of $P^\varepsilon\cup\{z\}$. The set $P^{\varepsilon,z}$ is a closed polyhedron containing~$z$. Furthermore,~$P^{\varepsilon,z}$ has at most $m=m(K,\varepsilon,N):=2^{k(K,\varepsilon,N)+1}$ faces, that is,~\eqref{Pez2} holds. 

To show~\eqref{Pez1}, observe first that $K^{\varepsilon/2}\subset P^\varepsilon\subset P^{\varepsilon,z}$. Since $P^\varepsilon\cup\{z\}\subset K^\varepsilon$ and $K^\varepsilon$ is convex, one has $P^{\varepsilon,z}\subset K^\varepsilon$. Lastly, from the proof of~(i) and the notations used there, one knows that $z-\varepsilon\nu=y\in\partial K$, where $\nu$ is the exterior unit normal to~$\partial K^\varepsilon$ at~$z$. Thus, $B_{\varepsilon/2}(z-\varepsilon\nu)=B_{\varepsilon/2}(y)\subset K^{\varepsilon/2}\subset P^{\varepsilon,z}$. The proof of Lemma~\ref{lemma:approximation} is thereby complete.

<!-- lemma-5: body_md -->
\label{lemma:bistable}
Assume that Hypothesis~\ref{hyp:bistable} holds. Let $v:\mathbb{R}\times\mathbb{R}^N\to[0,1]$ be an entire $($defined in $\mathbb{R}\times\mathbb{R}^N$$)$ solution of~\eqref{main}, for which there exist $\nu\in{\mathbb{S}}^{N-1}$ and $c>c^*(\nu)$ such that 
\begin{equation}\label{con:v} 
v(t,x)\le \sigma\ \ \hbox{for all $t\le 0$ and $x\cdot \nu \ge c t$}.
\end{equation}
Then, $v(t,x)\equiv 0$ in $\mathbb{R}\times\mathbb{R}^N$.

<!-- proof-6: body_md -->
We first show that 
\begin{equation}\label{lim0}
\lim_{\rho\rightarrow +\infty}\Big(\sup_{\genfrac{}{}{0pt}{}{t\le0}{x\cdot \nu-ct\ge \rho}}v(t,x)\Big)=0.
\end{equation}
Assume by contradiction that there exist $\delta_1>0$ and sequences $\{t_n\}_{n\in\mathbb{N}}$ of $\mathbb{R}$, $\{x_n\}_{n\in\mathbb{N}}$ of $\mathbb{R}^N$ such that $t_n\le 0$, $v(t_n,x_n)\ge \delta_1$, and $x_n\cdot \nu-ct_n\rightarrow +\infty$ as $n\rightarrow+\infty$. Define $v_{n}(t,x):=v(t+t_n,x+x_n)$ for $(t,x)\in\mathbb{R}\times\mathbb{R}^N$. Then, 
\begin{equation}\label{vn}
v_n(0,0)\ge \delta_1\hbox{ and } v_n(t,x)\le \sigma \hbox{ for all $t\le -t_n$ and $x\cdot \nu\ge ct-(x_n\cdot \nu -c t_n)$}.
\end{equation} 
Denote $x_n=y_n+z_n$ where $y_n\in [0,1]^N$ and $z_n\in \mathbb{Z}^N$. Then, there is $y_0\in [0,1]^N$ such that $y_n\rightarrow  y_0$ as $n\rightarrow +\infty$ up to extraction of a subsequence. By parabolic estimates, the sequence $\{v_{n}\}_{n\in\mathbb{N}}$ converges as $n\rightarrow +\infty$, up to extraction of a subsequence, locally uniformly to an entire solution $v_{\infty}:\mathbb{R}\times\mathbb{R}^N\to[0,1]$ of a translation of \eqref{main}, that is, 
$$  \partial_t v_{\infty}=\operatorname{div}(A(x+y_0)\nabla v_{\infty}) +q(x+y_0)\cdot\nabla v_{\infty} + f(x+y_0,v_{\infty}).$$
By \eqref{vn} and $\lim_{n\to+\infty}(x_n\cdot\nu-ct_n)=+\infty$, one also has that $v_{\infty}(0,0)\ge\delta_1$ and $0\le v_{\infty}(t,x)\le \sigma$ for all $t\le 0$ and $x\in\mathbb{R}^N$. Since $\min_{x\in\mathbb{R}^N} f(x,s)<0$ for every $s\in (0,\sigma]$, it then follows easily from the maximum principle, as in~\cite[Lemma~4.2]{GHR1}, that $v_{\infty}(t,x)\equiv 0$, which contradicts $v_{\infty}(0,0)\ge \delta_1$. Thus,~\eqref{lim0} is proved.

Let $\overline{v}(t,x):=\phi_{\nu}(t ,x+c^*(\nu)t\nu)$ and $\underline{v}(t,x):=v(t,x+c^*(\nu)t\nu)$, where $\phi_{\nu}$ is given in Hypothesis~\ref{hyp:bistable}. They satisfy
$$\partial_t u=\operatorname{div}(A(x+c^*(\nu)t\nu)\nabla u) +c^*(\nu) \nu \cdot\nabla u +q(x+c^*(\nu)t\nu)\cdot \nabla u+ f(x+c^*(\nu)t\nu,u)$$
in $\mathbb{R}\times\mathbb{R}^N$, and all conditions in \cite[Lemma~2.2]{R1} can be verified.\footnote{The functions $\underline{v}$ and $\overline{v}$ are of class $C^{1+\delta/2,2+\delta}(\mathbb{R}\times\mathbb{R}^N)$ for every $0<\delta<1$ by standard parabolic estimates. This regularity is sufficient to apply \cite[Lemma~2.2]{R1} even if in our paper the function $f$ (H\"older continuous in $x$, Lipschitz continuous in $u$) is slightly less regular than the function $f$ (Lipschitz continuous in all variables) used in \cite[Lemma~2.2]{R1}.} Then, it follows from \cite[Lemma~2.2]{R1} that $0\le\underline{v}(t,x)\le \overline{v}(t,x)$ for $(t,x)\in(-\infty,0]\times\mathbb{R}^N$. The comparison principle yields that
$$v(t,x)\le \phi_{\nu}(t,x)=U_{\nu}(x,x\cdot\nu -c^*(\nu) t)\ \hbox{ for all $(t,x)\in \mathbb{R}\times\mathbb{R}^N$}.$$

For any $t_0<0$, there is $y_0=y_0(t_0)\in [-1,1]^N$ such that $[ct_0\nu]:=ct_0\nu-y_0\in \mathbb{Z}^N$ and $y_0\cdot\nu\le0$. 
Since the new function $V:(t,x)\mapsto V(t,x):=v(t+t_0,x+[ct_0 \nu])$ still satisfies \eqref{main} and~\eqref{con:v}, it follows from the arguments of the previous paragraph that $0\le v(t+t_0,x+[ct_0 \nu])\le \phi_{\nu}(t,x)=U_{\nu}(x,x\cdot\nu -c^*(\nu) t)$ for all $(t,x)\in \mathbb{R}\times\mathbb{R}^N$, that is,
$$\begin{array}{rcl}
0\le v(t,x) & \le & U_{\nu}(x-[ct_0 \nu],x\cdot\nu -c^*(\nu) t+(c^*(\nu) -c)t_0+y_0\cdot \nu)\vspace{3pt}\\
& = & U_{\nu}(x,x\cdot\nu -c^*(\nu) t+(c^*(\nu) -c)t_0+y_0\cdot \nu),\end{array}$$
for all $(t,x)\in\mathbb{R}\times\mathbb{R}^N$. By taking $t_0\rightarrow -\infty$ and using $c>c^*(\nu)$ and $U_\nu(\cdot,+\infty)=0$, one concludes that $v(t,x)=0$ for all $(t,x)\in\mathbb{R}\times\mathbb{R}^N$.

Assume that $U_{\rho}\neq \emptyset$, where $\rho>0$ is given in Proposition~\ref{pro:rho}. For any $0<\tau<1$, it holds

$$[0,1]^N\subset\bigcup_{i=1}^kB_{\varepsilon_{y_i}}(y_i).$$

$$T':=\max(T_{y_1},\cdots,T_{y_k})\in(0,+\infty).$$

\begin{equation}\label{T'y}
\forall\,y\in[0,1]^N,\ \exists\,T'_y\in(0,T'],\ \ v_y(T'_y,\cdot)\ge\theta\,\1_{B_\rho}\hbox{ in $\mathbb{R}^N$}.
\end{equation}

Since $\tau'\mathcal{W}_0\Subset\mathcal{W}_0$ (i.e. $\tau'\mathcal{W}_0$ has compact closure included in the open set $\mathcal{W}_0$), there is then a time $\overline{T}>0$ such that
$$\forall\,s\ge\overline{T},\ \forall\,z\in\mathcal{W}_0,\ \ w(s,s\tau'z)\ge\lambda.$$
Since $\tau\mathcal{W}_0\Subset\tau'\mathcal{W}_0$, there is then a time $T\ge T'$ such that
\begin{equation}\label{Tw}
\forall\,t\ge T,\, \forall\,s'\in(0,T'],\ \forall\,z\in\mathcal{W}_0,\ \forall\,y\in[0,1]^N,\ \ w(t-s',t\tau z+y)\ge\lambda.
\end{equation}

Since $u_0=\1_U\ge\1_{B_\rho(x_0)}=\1_{B_\rho(y_0)}(\cdot+y_0-x_0)=v_{y_0}(0,\cdot+y_0-x_0)$ and the coefficients of~\eqref{main} are $\mathbb{Z}^N$-periodic, the comparison principle implies that $u(t,x)\ge v_{y_0}(t,x+y_0-x_0)$ for all $(t,x)\in[0,+\infty)\times\mathbb{R}^N$. In particular, by using~\eqref{T'y}-\eqref{Tw} and the comparison principle again, one gets that
$$\forall\,t\ge T,\ \forall\,z\in\mathcal{W}_0,\ \ u(t,x_0+t\tau z)\ge v_{y_0}(t,t\tau z+y_0)\ge w(t-T'_{y_0},t\tau z+y_0)\ge\lambda.$$
Since $\lambda\in(0,1)$ was arbitrary and $u\le1$ in $[0,+\infty)\times\mathbb{R}^N$, the proof is~complete.

$$L_{e_i,\lambda_i} \varphi_i=k(e_i,\lambda_i) \varphi_i\ \hbox{ in $\mathbb{R}^N$}$$
with $\varphi_i\in E$, $\|\varphi_i\|_{L^{\infty}(\mathbb{R}^N)}=1$, and $\varphi_i>0$ in $\mathbb{R}^N$. Then, by Hypothesis~\ref{hyp:KPP}, for every $1\le i\le l$ and every $\varrho\in\mathbb{R}$, the function $u(t,x):=e^{-\lambda_i(x\cdot e_i-a_it-\varrho)}  \varphi_i(x)$ satisfies
\begin{equation}\label{linEq}\begin{array}{l}
\partial_t u-\hbox{div}(A(x)\nabla u)-q(x)\cdot \nabla u -f(x,u)\vspace{3pt}\\
\qquad\qquad\qquad\qquad\ge \partial_t u-\hbox{div}(A(x)\nabla u)-q(x)\cdot \nabla u-f_s(x,0)u=0\end{array}
\end{equation}
in $\mathbb{R}\times\mathbb{R}^N$. Let $\tau>0$ be large enough such that
\begin{equation}\label{Re}
e^{\lambda_i\tau a_i}  \varphi_i(x)\ge 1\ \hbox{ and }\ e^{-\lambda_i \tau a_i}  \varphi_i(x)\le \frac{\eta}{l}\ \hbox{ for all $i=1,\cdots,l$, and for all $x\in\mathbb{R}^N$}.
\end{equation}

Define, for $\kappa>0$, $t\ge0$ and $x\in\mathbb{R}^N$,
$$u^+(t,x):=\min\left\{ \sum_{i=1}^{l} e^{-\lambda_i(x\cdot e_i+a_i(\kappa-\tau-t))} \varphi_i(x),1\right\}.$$
By \eqref{linEq} and Hypothesis~\ref{hyp:KPP}, one can easily check that 
$$\partial_t u^+-\hbox{div}(A(x)\nabla u^+)-q(x)\cdot \nabla u^+-f(x,u^+)\ge 0,$$
for all $t\ge 0$ and $x\in\mathbb{R}^N$ such that $u^+(t,x)<1$. Furthermore, for every $x\in \mathbb{R}^N\!\setminus\!(-\kappa P^\varepsilon)$, there is $i\in \{1,\cdots,l\}$ such that $-x\cdot e_i>\kappa a_i$, which implies by~\eqref{Re} that
$$e^{-\lambda_i(x\cdot e_i+a_i(\kappa-\tau))} \varphi_i(x)\ge e^{\lambda_i\tau a_i} \varphi_i(x)\ge 1.$$
Hence, $u^+(0,x)=1=v_{-\kappa P^\varepsilon}(0,x)$ for $x\in \mathbb{R}^N\!\setminus\!(-\kappa P^\varepsilon)$, while $u^+(0,x)>0=v_{-\kappa P^\varepsilon}(0,x)$ for $x\in-\kappa P^\varepsilon$. Thus, $u^+(0,\cdot)\ge v_{-\kappa P^\varepsilon}(0,\cdot)$ in $\mathbb{R}^N$, and the comparison principle yields
$$0\le v_{-\kappa P^\varepsilon}(t,x)\le u^+(t,x)\ \hbox{ for all $t\ge 0$ and $x\in\mathbb{R}^N$}.$$
Especially, by taking $R:=2\tau$, and then any $\kappa\ge 2R$, one has by~\eqref{-Pe} and~\eqref{Re} that 
$$0\le v_{-\kappa P^\varepsilon}(t,x)\le u^+(t,x)\le \sum_{i=1}^{l} e^{-\lambda_i \tau a_i} \varphi_i(x)\le \eta, $$
for all $t\in[0,\kappa-2R]$ and $x\in (-\kappa+R+t)P^\varepsilon=-|-\kappa+R+t|P^\varepsilon$.

\label{lem:vk<e}
Assume that  one of Hypotheses~\ref{hyp:ignition}-\ref{hyp:bistable} holds, as well as~\eqref{supt-plane}, with $\mathcal{W}_0$ given in~\eqref{w0}-\eqref{W0-wulff}. Let $\varepsilon>0$ and $\eta>0$ be arbitrary, and let $K^\varepsilon:=\overline{\mathcal{W}_0}+\overline{B_\varepsilon}$ be the $C^1$ exterior approximation of $K:=\overline{\mathcal{W}_0}$ given in Lemma~\ref{lemma:approximation}-(i). Then there exists $R>0$ such that~\eqref{kW} holds for all $\kappa\geq2R$ with $K^\varepsilon$ instead of $P^\varepsilon$.

We argue by contradiction, supposing that the statement fails for some $\varepsilon>0$ and~$\eta>0$. We can then assume without loss of generality that $0<\varepsilon\le1$ and $0<\eta\le \delta_{\varepsilon}\le \sigma$, where $\sigma$ is given in Hypotheses~\ref{hyp:ignition} or~\ref{hyp:bistable}, and $\delta_\varepsilon$ is given by Lemma~\ref{lemma:combu} (in case of Hypothesis~\ref{hyp:ignition}). For any $\kappa>0$, denote the function $v_{-\kappa K^\varepsilon}$ by $v_{\kappa}$ for short.

Then, from our assumption, there exist some sequences $(\kappa_n)_{n\in\mathbb{N}}$ in $(0,+\infty)$, $(t_n)_{n\in\mathbb{N}}$ in $\mathbb{R}$ and~$(x_n)_{n\in\mathbb{N}}$ in $\mathbb{R}^N$ satisfying:
\begin{formula}{tnxn}
\kappa_n\geq 2n,\ \ 0\le t_n\leq \kappa_n-2n,\ \ x_n\in (-\kappa_n+n+t_n)K^\varepsilon,\ \ \eta<v_{\kappa_n}(t_n,x_n)\le1,
\end{formula}\noindent
for all $n\in\mathbb{N}$. We have in particular
$$-\kappa_n+n\leq -\kappa_n+n+t_n\leq -n\le -1$$
for all $n\ge 1$, and
\begin{formula}{<xn<}
x_n\in (-\kappa_n+n)K^\varepsilon,
\end{formula}\noindent
because $K^\varepsilon$ is in particular star-shaped with respect to the origin.
	
Since $0\in\mathcal{W}_0$, one has $B_\varepsilon\subset K^\varepsilon$, whence $(-\kappa_n+n)K^\varepsilon\supset B_{(\kappa_n-n)\varepsilon}$. Applying Lemma~\ref{lem:distW} to the set $(-\kappa_n+n)K^\varepsilon$ (instead of $K$), we deduce that, for each $n\ge1$,
\begin{formula}{k+n}
\min_{x\in (-\kappa_n+n)K^\varepsilon}\ d\big(x,\mathbb{R}^N\!\setminus\!(-\kappa_nK^\varepsilon)\big)\geq \Big(\frac{-\kappa_n}{-\kappa_n+n}-1\Big)(\kappa_n-n)\varepsilon=n\varepsilon>0.
\end{formula}\noindent
Since $v_{\kappa_n}(0,\.)=0$ in ${\rm{int}}(-\kappa_nK^\varepsilon)\supset(-\kappa_n+n)K^\varepsilon$, the function $v_{\kappa_n}$ is continuous in~$[0,+\infty)\times(-\kappa_n+n)K^\varepsilon$ and thus there exist a time and a point, that we still call~$t_n$ and~$x_n$ without loss of generality, fulfilling~\eqref{tnxn} together with $t_n>0$ and
\begin{formula}{tangency}
\forall\,t\in[0,t_n), \ \ \forall\,x\in (-\kappa_n+n+t)K^\varepsilon,\ \ v_{\kappa_n}(t,x)<\eta\ \hbox{ and }\ v_{\kappa_n}(t_n,x_n)=\eta.
\end{formula}\noindent

We now claim the following:
\begin{formula}{xnDW}
\forall\,n\in\mathbb{N},\ \ x_n\in(-\kappa_n+n+t_n)\partial K^\varepsilon,
\end{formula}\noindent
and
\begin{formula}{tninfty}
t_n\to+\infty\ \hbox{ as }n\to+\infty.
\end{formula}\noindent	
Property \eqref{xnDW} immediately follows from the strong parabolic maximum principle, because the constant $\eta$ is a supersolution to~\eqref{main} and $v_{\kappa_n}(0,\.)=0<\eta$ in $(-\kappa_n+n)K^\varepsilon$. Property \eqref{tninfty} follows instead from parabolic estimates. Indeed,~\eqref{<xn<} and \eqref{k+n} entail that $d(x_n,\mathbb{R}^N\!\setminus\!(-\kappa_nK^\varepsilon))\to+\infty$ as $n\to+\infty$ and thus, since $v_{\kappa_n}(0,\.)=0$ in~$-\kappa_nK^\varepsilon$, the parabolic estimates imply that the functions $(t,x)\mapsto v_{\kappa_n}(t,x+x_n)$ converge locally uniformly in $[0,+\infty)\times\mathbb{R}^N$, up to a subsequence, to a solution $\widetilde v$ of some translation of the equation~\eqref{main} with initial datum $\widetilde v(0,\cdot)=0$ in $\mathbb{R}^N$, that is, $\widetilde v\equiv0$ in $[0,+\infty)\times\mathbb{R}^N$. However, since $v_{\kappa_n}(t_n,x_n)=\eta$, this shows that $(t_n)_{n\in\mathbb{N}}$ cannot have a bounded subsequence, i.e.,~\eqref{tninfty} holds too.

Let us then define, for $(t,x)\in[-t_n,+\infty)\times\mathbb{R}^N$, 
$$ v^n(t,x):=v_{\kappa_n}(t+t_n,x+x_n).$$ 
Then, \eqref{tangency} can be rewritten as
\begin{formula}{vn<e}
\forall\,t\in[-t_n,0),\ \ \forall\,x\in (-\kappa_n+n+t_n+t)K^\varepsilon-\{x_n\},\ \ v^n(t,x)<\eta\ \hbox{ and }\ v^n(0,0)=\eta.
\end{formula}\noindent
Call
\begin{equation}\label{defbarxn}
\bar x_n:=\frac{x_n}{-\kappa_n+n+t_n},
\end{equation}
which belongs to $\partial K^\varepsilon$ owing to~\eqref{xnDW}. Up to extraction of a subsequence, the sequence $(\bar x_n)_{n\in\mathbb{N}}$ converges to some $\bar x\in\partial K^\varepsilon$. Let $\nu_n$ and $\nu$ denote the exterior normals to $K^\varepsilon$ at $\bar x_n$ and $\bar x$ respectively. The regularity of $\partial K^\varepsilon$ implies that $(\nu_n)_{n\in\mathbb{N}}$ converges to $\nu$ and, roughly speaking, the dilations of $K^\varepsilon$ will converge to half-spaces locally uniformly around boundary points, as we see below.

More precisely, we introduce the family of open half-spaces $(H_t)_{t<0}$ defined by
$$H_t:=\big\{x\in\mathbb{R}^N\ :\ x\.\nu>(c^*(\nu)+\varepsilon)t\big\}.$$
Fix in this paragraph $t<0$ and $x\in H_t$. We claim that
\begin{equation}\label{vneta}
v^n(t,x)<\eta\ \hbox{ for all $n$ sufficiently large}.
\end{equation}
For all $n$ large enough, we have $t\in[-t_n,0)$, while $-\kappa_n+n+t_n+t<-\kappa_n+n+t_n\le -n<0$. Thus, owing to~\eqref{vn<e}, in order to have~\eqref{vneta}, it is sufficient to show that $x+x_n\in (-\kappa_n+n+t_n+t)K^\varepsilon$, i.e.
$$z_n:=\frac{x+x_n}{-\kappa_n+n+t_n+t}\in K^\varepsilon.$$
Using that $x_n=(-\kappa_n+n+t_n)\bar x_n$  we find
$$|z_n-\bar x_n|=\frac{|x-t\bar x_n|}{\kappa_n-n-t_n-t},$$
which converges to $0$ as $n\to+\infty$ because $(\bar x_n)_{n\in\mathbb{N}}$ is bounded while $\kappa_n-n-t_n-t\ge n-t$, by~\eqref{tnxn}. Next, using $\bar x\.\nu=c^*(\nu)+\varepsilon$ from Lemma~\ref{lemma:app-W0}-(i), we have $t\overline{x}\cdot \nu=(c^*(\nu)+\varepsilon)t<x\cdot \nu$, whence $t\overline{x}\neq x$. Therefore, $|x-t\bar x_n|>0$ and $|z_n-\bar x_n|>0$ for all $n$ large enough and, using $\bar x_n\.\nu_n=c^*(\nu_n)+\varepsilon$ from Lemma~\ref{lemma:app-W0}-(i) again,		
$$\frac{z_n-\bar x_n}{|z_n-\bar x_n|}\.\nu_n=-\frac{x\.\nu_n-t(c^*(\nu_n)+\varepsilon)}{|x-t\bar x_n|}\to-\frac{x\.\nu-t(c^*(\nu)+\varepsilon)}{|x-t\bar x|}\ \hbox{ as } n\to+\infty.$$
We notice that the above limit is strictly negative because $x\in H_t$. By the $C^1$ regularity of $K^\varepsilon$ and the fact that $|z_n-\bar x_n|\to0$ as $n\rightarrow +\infty$, we deduce that $z_n\in K^\varepsilon$ for all $n$ sufficiently large, whence $v^n(t,x)<\eta$ thanks to~\eqref{vn<e}. Therefore,~\eqref{vneta} is proved.

Now, the sequence $(v^n)_{n\in\mathbb{N}}$ converges locally uniformly in $\mathbb{R}\times\mathbb{R}^N$, up to a subsequence, to an entire solution $v^*:\mathbb{R}\times\mathbb{R}^N\to[0,1]$ of some translation of the equation~\eqref{main}, satisfying the following properties:
\begin{equation}\label{vstar}
v^*(0,0)=\eta,\ \hbox{ and }\ v^*(t,x)\leq\eta\hbox{ for all }t\leq0\hbox{ and }x\in H_t.
 \end{equation}

{\it Case 1: Hypothesis~\ref{hyp:bistable} holds}. Here, one gets a contradiction from \eqref{vstar} by applying Lemma~\ref{lemma:bistable} to (a time-space translation of) $v^*$.

{\it Case 2: Hypothesis~\ref{hyp:ignition} holds}. Here, one gets from \eqref{vstar} and Lemma~\ref{lemma:combu} that $v^*(t,x)\le \eta$ for all $(t,x)\in\mathbb{R}\times\mathbb{R}^N$. Then, from the strong maximum principle,
\begin{equation}\label{v*eta}
v^*(t,x)=\eta\ \hbox{ for all $(t,x)\in\mathbb{R}\times\mathbb{R}^N$}.
\end{equation}
To get a contradiction, we need further arguments.

For each $e\in\mathbb{S}^{N-1}$ and $\lambda\in\mathbb{R}$, call $k(e,\lambda)$ the principal eigenvalue of the operator
$$\mathscr{L}_{e,\lambda} \varphi:=-\nabla\cdot (A\nabla\varphi) +2\lambda eA\nabla \varphi -q\cdot \nabla\varphi  +[\lambda \nabla\cdot (A e) +\lambda q\cdot e -\lambda^2 eAe]\varphi,$$
which is actually the same as the operator $L_{e,\lambda}$ in~\eqref{L-FK} since here $f_s(x,0)\equiv0$, acting on the same set~$E$ as in~\eqref{Eelambda}. By evaluating the equality $\mathscr{L}_{e,\lambda} \varphi=k(e,\lambda)\varphi$ at a minimum point $x\in[0,1]^N$ of a principal (that is, positive) eigenfunction $\varphi$, one gets that $k(e,\lambda)\le\lambda\nabla\cdot(Ae)(x)+\lambda q(x)\cdot e -\lambda^2 eA(x)e$, whence
\begin{equation}\label{kelambda1}
\limsup_{\lambda\rightarrow+\infty}\Big(\sup_{e\in\mathbb{S}^{N-1}}\frac{k(e,\lambda)}{\lambda^2}\Big)<0.
\end{equation}
Notice that $k(e,0)=0$ for all $e\in{\mathbb{S}}^{N-1}$, with any positive constant function as principal eigenfunction. Furthermore, we claim that
\begin{equation}\label{kelambda2}
\sup_{e\in\mathbb{S}^{N-1}}|k(e,\lambda)|=o(\lambda)\ \hbox{ as }\lambda\to0.
\end{equation}
To see it, consider any sequence $(e_n)_{n\in\mathbb{N}}$ in ${\mathbb{S}}^{N-1}$ and $(\lambda_n)_{n\in\mathbb{N}}$ in $\mathbb{R}^*$ such that $\lambda_n\to0$ as $n\to+\infty$, and let us show that $k(e_n,\lambda_n)/\lambda_n\to0$ as $n\to+\infty$. By evaluating the equality $\mathscr{L}_{e_n,\lambda_n} \varphi_n=k(e_n,\lambda_n)\varphi_n$ at a minimum and a maximum point of a principal (that is, positive) eigenfunction $\varphi_n$, one gets that
\begin{equation}\label{ken}
|k(e_n,\lambda_n)|\le|\lambda_n|\,\|\nabla\cdot(Ae_n)\|_{L^\infty(\mathbb{R}^N)}+|\lambda_n|\|q\|_{L^\infty(\mathbb{R}^N)}+\lambda_n^2\|e_nAe_n\|_{L^\infty(\mathbb{R}^N)},
\end{equation}
whence $k(e_n,\lambda_n)\to0$ as $n\to+\infty$. By normalizing the principal positive $\mathbb{Z}^N$-periodic eigenfunction $\varphi_n$ with $\max_{\mathbb{R}^N}\varphi_n=1$, it follows from standard elliptic estimates that, up to extraction of a subsequence, the sequence $(\varphi_n)_{n\in\mathbb{N}}$ converges in $C^2(\mathbb{R}^N)$ to a non-negative $\mathbb{Z}^N$-periodic solution $\varphi_\infty$ of $-\nabla\cdot (A\nabla\varphi_\infty)-q\cdot \nabla\varphi_\infty=0$ in $\mathbb{R}^N$, with $\max_{\mathbb{R}^N}\varphi_\infty=1$. Thus, $\varphi_\infty=1$ in $\mathbb{R}^N$, from the strong maximum principle. Next, by integrating the equality $\mathscr{L}_{e_n,\lambda_n} \varphi_n=k(e_n,\lambda_n)\varphi_n$ over $(0,1)^N$ and using the $\mathbb{Z}^N$-periodicity of the coefficients $A$ and $q$, together with~\eqref{hypq}, one infers that
$$\frac{k(e_n,\lambda_n)}{\lambda_n}\int_{(0,1)^N}\varphi_n=\int_{(0,1)^N}e_nA\nabla\varphi_n+\int_{(0,1)^N}(q\cdot e_n)\varphi_n-\int_{(0,1)^N}\lambda_ne_nAe_n\varphi_n.$$
Using again~\eqref{hypq} and the convergence of $(\varphi_n)_{n\in\mathbb{N}}$ to $1$ in $C^2(\mathbb{R}^N)$, one concludes that $k(e_n,\lambda_n)/\lambda_n\to0$ as $n\to+\infty$. As a consequence,~\eqref{kelambda2} is proved.

Now, it follows from \cite[Proposition~5.7]{BH1} that, for each $e\in{\mathbb{S}}^{N-1}$, the function $\lambda\mapsto k(e,\lambda)$ is concave in $\mathbb{R}$. Together with~\eqref{kelambda1}-\eqref{kelambda2}, one infers that, for each $e\in{\mathbb{S}}^{N-1}$ and $\gamma>0$, there exists a unique $\lambda_{e,\gamma}>0$ such that
$$k(e,\lambda_{e,\gamma})+\gamma\lambda_{e,\gamma}=0.$$
Call $\varphi_{e,\gamma}$ a principal eigenfunction of the operator $\mathscr{L}_{e,\lambda_{e,\gamma}}$. Let $\varrho>0$ be such that $c^*(e)\le\varrho$ for all $e\in{\mathbb{S}}^{N-1}$, and remember also that the map $e\mapsto c^*(e)$ is continuous and positive in ${\mathbb{S}}^{N-1}$. One then infers from~\eqref{kelambda1}-\eqref{kelambda2} that
\begin{equation}\label{lowerlambda}
0<\inf_{\genfrac{}{}{0pt}{}{e\in{\mathbb{S}}^{N-1}}{\gamma\in[c^*(e),\varrho+1]}}\lambda_{e,\gamma}\le\sup_{\genfrac{}{}{0pt}{}{e\in{\mathbb{S}}^{N-1}}{\gamma\in[c^*(e),\varrho+1]}}\lambda_{e,\gamma}<+\infty,
\end{equation}
and then, from upper bounds similar to those in~\eqref{ken}, one gets that
$$\sup_{\genfrac{}{}{0pt}{}{e\in{\mathbb{S}}^{N-1}}{\gamma\in[c^*(e),\varrho+1]}}|k(e,\lambda_{e,\gamma})|<+\infty.$$
From the Harnack inequality applied to the $\mathbb{Z}^N$-periodic positive $C^2(\mathbb{R}^N)$ functions $\varphi_{e,\gamma}$, there is finally a positive real number $M$ such that
\begin{equation}\label{harnack}
\sup_{\genfrac{}{}{0pt}{}{e\in{\mathbb{S}}^{N-1}}{\gamma\in[c^*(e),\varrho+1]}}\frac{\max_{\mathbb{R}^N}\varphi_{e,\gamma}}{\min_{\mathbb{R}^N}\varphi_{e,\gamma}}\le M.
\end{equation}
Even if it means decreasing $\eta>0$, one can assume without loss of generality that
$$0<\eta\le\min\Big(1,\delta_{\varepsilon},\frac{\sigma}{Mm}\Big),$$
where we recall that $\sigma\in(0,1)$ is given in Hypothesis~\ref{hyp:ignition}, $\delta_\varepsilon\in(0,\sigma)$ is given in Lemma~\ref{lemma:combu}, and $m=m(\overline{\mathcal{W}_0},\varepsilon,N)\in\mathbb{N}$ is given in Lemma~\ref{lemma:approximation}-(iii) with $K=\overline{\mathcal{W}_0}$.

Coming back to the sequence $(t_n,x_n)_{n\in\mathbb{N}}$ in $(0,+\infty)\times\mathbb{R}^N$ satisfying~\eqref{tangency}-\eqref{tninfty} and to the points $\bar{x}_n\in\partial K^\varepsilon$ defined in~\eqref{defbarxn}, let now $P^{\varepsilon,\bar{x}_n}$ be a closed polyhedral exterior approximation of $K=\overline{\mathcal{W}_0}$ given in Lemma~\ref{lemma:approximation}-(iii) and used in Lemma~\ref{lemma:app-W0}-(ii), such that $\bar{x}_n\in P^{\varepsilon,\bar{x}_n}\subset K^\varepsilon$ and
$$-P^{\varepsilon,\bar{x}_n}=\bigcap_{i=1}^m\big\{x\in\mathbb{R}^N:-x\cdot f^n_i\le b^n_i\big\},$$
where $f^n_i\in{\mathbb{S}}^{N-1}$ and $b^n_i\ge c^*(f^n_i)>0$ for every $1\le i\le m$. Recall that $m$ is independent of $n$, but the directions $f^n_i$ and the real numbers $b^n_i$ depend on $n$ in general. Furthermore, since $P^{\varepsilon,\bar{x}_n}\subset K^\varepsilon=\overline{\mathcal{W}_0}+\overline{B_\varepsilon}\subset\overline{B_{\varrho+1}}$ (remember that $0<\omega_0(e)\le c^*(e)\le\varrho$ for every $e\in{\mathbb{S}}^{N-1}$ and that $0<\varepsilon\le1$), one can assume without loss of generality that $b^n_i\le\varrho+1$ for all $n\in\mathbb{N}$ and $1\le i\le m$. Then, for each $n\in\mathbb{N}$ and $1\le i\le m$, let $\lambda_i^n:=\lambda_{f^n_i,b^n_i}$ be the positive root of
$$k(f^n_i,\lambda^n_i)+b^n_i\lambda^n_i=0$$
and let $\varphi_i^n:=\varphi_{f^n_i,b^n_i}\in E$ be the related normalized principal eigenfunction, that is, satis\-fying $\mathscr{L}_{f^n_i,\lambda^n_i}\varphi^n_i=k(f^n_i,\lambda^n_i) \varphi^n_i$ in $\mathbb{R}^N$, $\varphi_i^n>0$ in $\mathbb{R}^N$ and $\|\varphi^n_i\|_{L^\infty(\mathbb{R}^N)}=1$. From~\eqref{harnack}, the functions $\varphi^n_i$ also satisfy
$$\min_{\mathbb{R}^N}\varphi^n_i\ge\frac{\displaystyle\max_{\mathbb{R}^N}\varphi^n_i}{M}=\frac{1}{M},$$
with $M>0$ independent of $n$ and $i$.
 
Define, for $n\in\mathbb{N}$ and $(t,x)\in \mathbb{R}\times\mathbb{R}^N$,
$$u^+_n(t,x):= \sum_{i=1}^{m} \frac{\eta }{\displaystyle\min_{\mathbb{R}^N} \varphi_i^n}e^{-\lambda_i^n(x\cdot f^n_i +b^n_i(\kappa_n-n-t))} \varphi^n_i(x).$$
Then
$$0<u^+_n(t,x)\le \eta M m\le \sigma\ \hbox{ for all $t\in [0,t_n]$ and $x\in (-\kappa_n+n+t)P^{\varepsilon,\bar{x}_n}$},$$
and
$$u^+_n(t,x)\ge \eta\ \hbox{ for all $t\in [0,t_n]$ and $x\in (-\kappa_n+n+t)\partial P^{\varepsilon,\bar{x}_n}$}.$$
Owing to the definition of $\lambda^n_i$ and $\varphi^n_i$, one can easily check that 
\begin{equation}\label{eq:u+}
\partial_t u^+_n-\hbox{div}(A(x)\nabla u^+_n)-q(x)\cdot \nabla u^+_n= 0\ \hbox{ in }[0,t_n]\times(-\kappa_n+n+t)P^{\varepsilon,\bar{x}_n}
\end{equation}
(and even in $\mathbb{R}\times\mathbb{R}^N$).

Regarding $v_{\kappa_n}$, we have that $v_{\kappa_n}(0,x)=0\le u^+_n(0,x)$ for $x\in (-\kappa_n +n)P^{\varepsilon,\bar{x}_n}\subset(-\kappa_n+n)K^\varepsilon$ and
$$v_{\kappa_n}(t,x)\le\eta\le u^+_n(t,x)\ \hbox{ for all $t\in [0,t_n]$ and $x\in (-\kappa_n+n+t)\partial P^{\varepsilon,\bar{x}_n}$},$$ 
by \eqref{tangency} and $P^{\varepsilon,\bar{x}_n}\subset K^\varepsilon$. Since $f(x,s)=0$ for $(x,s)\in\mathbb{R}^N\times[0,\sigma]$, it follows from \eqref{tangency} and $P^{\varepsilon,\bar{x}_n}\subset K^\varepsilon$ that  $v_{\kappa_n}$ also satisfies \eqref{eq:u+} in $(0,t_n]\times(-\kappa_n+n+t)P^{\varepsilon,\bar{x}_n}$. Therefore, by the comparison principle,
$$v_{\kappa_n}(t,x)\le u^+_n(t,x)\ \hbox{ for all $t\in [0,t_n]$ and $x\in (-\kappa_n+n+t)P^{\varepsilon,\bar{x}_n}$}.$$

Finally, choose $\overline{R}>0$ such that, for all $n\in\mathbb{N}$,
$$\sum_{i=1}^{m}\eta M e^{-\lambda_i^n\overline{R}/2} \le\frac{\eta}{2},$$
which is possible because of~\eqref{lowerlambda} and $\lambda_i^n=\lambda_{f^n_i,b^n_i}$ with $c^*(f^n_i)\le b^n_i\le\varrho+1$ for all $n\in\mathbb{N}$ and $1\le i\le m$. For every $n\in\mathbb{N}$, since $B_{\varepsilon/2}(\bar{x}_n-\varepsilon\nu_n)\cup\{\bar{x}_n\}\subset P^{\varepsilon,\bar{x}_n}$ and $P^{\varepsilon,\bar{x}_n}$ is convex by Lemma~\ref{lemma:approximation}-(iii), one has that $B_{r\varepsilon/2}(\overline{x}_n- r\varepsilon\nu_n) \subset P^{\varepsilon,\bar{x}_n}$ for all $r\in [0,1]$. Then, one can take any $n$ large enough such that 
$$B_{\overline{R}/(2(\kappa_n-n-t_n))}\Big(\overline{x}_n-\frac{\overline{R}}{\kappa_n-n-t_n} \nu_n\Big)\subset P^{\varepsilon,\bar{x}_n},$$ 
whence
$$(x_n+\overline{R}\nu_n)\cdot f^n_i +(\kappa_n-n-t_n)b^n_i\ge \frac{\overline{R}}{2}\ \hbox{ for all $1\le i\le m$}.$$
Thus, $x_n+\overline{R}\nu_n\in (-\kappa_n+n+t_n)P_{\varepsilon,\bar{x}_n}$ for all $n$ large enough, and
$$\begin{array}{rcl}
v^n(0,\overline{R}\nu_n) & = & v_{\kappa_n}(t_n,x_n+\overline{R}\nu_n)\vspace{3pt}\\
& \le & \displaystyle u^+_n(t_n,x_n+\overline{R}\nu_n)\\
& \le & \displaystyle\sum_{i=1}^m \frac{\eta}{\displaystyle\min_{\mathbb{R}^N}\varphi^n_i} e^{-\lambda_i^n\overline{R}/2}\varphi_i^n(x_n+\overline{R}\nu_n)\le\sum_{i=1}^{m}\eta M e^{-\lambda_i^n\overline{R}/2}\le\frac\eta2.
\end{array}$$
This implies that $v^*(0,\overline{R}\nu)\le \eta/2$. One has finally reached a contradiction with $v^*(t,x)\equiv \eta$, see~\eqref{v*eta}. The proof of Lemma~\ref{lem:vk<e} is thereby complete.

Assume that \eqref{supt-plane} holds. For any $\alpha>1$, one has
$$\sup_{x\in \mathbb{R}^N\setminus (U + t\alpha \mathcal{W}_0)} u(t,x)\rightarrow 0\ \hbox{ as $t\rightarrow +\infty$}.$$

Let $\alpha>1$ be fixed. Assume by way of contradiction that the conclusion does not hold. Then there exist $\eta>0$, a sequence $(t_n)_{n\in\mathbb{N}}$ of positive real numbers diverging to $+\infty$, and a sequence $(x_n)_{n\in\mathbb{N}}$ in $\mathbb{R}^N$ such that $x_n\in \mathbb{R}^N\setminus (U + t_n\alpha \mathcal{W}_0)$ and
\begin{equation}\label{utnxneta}
u(t_n,x_n)> \eta>0
\end{equation}
for all $n\in\mathbb{N}$. One then has 
\begin{equation}\label{xnW}
U-x_n\subset\mathbb{R}^N\setminus(-t_n\alpha\mathcal{W}_0).
\end{equation}
For each $n\in\mathbb{N}$, let $\bar{x}_n\in\mathbb{Z}^N$ be such that $x_n-\bar{x}_n\in[0,1]^N$.

For $\varepsilon>0$, let $K^\varepsilon:=\overline{\mathcal{W}_0}+\overline{B_\varepsilon}$ and $P^\varepsilon$ be the exterior $C^1$ and polyhedral approximations of $K:=\overline{\mathcal{W}_0}$ given in Lemma~\ref{lemma:approximation}. Remember that $\overline{\mathcal{W}_0}\subset P^\varepsilon\subset K^\varepsilon$. Since the compact set $\overline{W_0}$ is included into the open set $\alpha\mathcal{W}_0$, one can then fix $\varepsilon>0$ small enough so that
$$\overline{\mathcal{W}_0}\subset P^\varepsilon\subset K^\varepsilon\subset\alpha\mathcal{W}_0.$$
For these fixed values of $\varepsilon>0$ and $\eta>0$, let then $R>0$ be given by Lemma~\ref{extract-kpp} (in case of Hypothesis~\ref{hyp:KPP}) or by Lemma~\ref{lem:vk<e} (in case of Hypothesis~\ref{hyp:ignition} or~\ref{hyp:bistable}). Call also $C_\varepsilon:=P^\varepsilon$ in case of Hypothesis~\ref{hyp:KPP}, resp. $C_\varepsilon:=K^\varepsilon$ in case of Hypothesis~\ref{hyp:ignition} or~\ref{hyp:bistable}. Since $C_\varepsilon\supset\overline{\mathcal{W}_0}$ and $\mathcal{W}_0$ contains a ball centered at the origin, one can take $\beta\ge2$ large enough so that $(1-\beta)RC_\varepsilon\supset[0,1]^N$.

Since the compact set $C_\varepsilon$ is included into the open set $\alpha\mathcal{W}_0$, and since $t_n\to+\infty$ as $n\to+\infty$, there holds
\begin{equation}\label{tn2Re}
\frac{(t_n+\beta R)C_\varepsilon-[0,1]^N}{t_n}\subset \alpha \mathcal{W}_0\ \hbox{ for all large $n$}.
\end{equation}
For each $n\in\mathbb{N}$, call $\kappa_n:=t_n+\beta R$, whence $\kappa_n\ge t_n+2R\ge 2R$ and
$$(-\kappa_n+R+t_n)C_\varepsilon=(1-\beta)RC_{\varepsilon}\supset[0,1]^N.$$
It follows then from Lemma~\ref{extract-kpp} or~\ref{lem:vk<e} that, for each $n\in\mathbb{N}$,
\begin{equation}\label{vkappan}
v_{-\kappa_nC_\varepsilon}(t_n,y)\le \eta\ \hbox{ for all }y\in[0,1]^N,
\end{equation}
where we recall that the functions $v_{-\kappa_nC_\varepsilon}$ are the solutions of~\eqref{main} with initial conditions $\1_{\mathbb{R}^N\setminus(-\kappa_nC_\varepsilon)}$. On the other hand,~\eqref{tn2Re} implies that
$$-\big(\kappa_nC_\varepsilon-[0,1]^N\big)=-\big((t_n+\beta R)C_\varepsilon-[0,1]^N\big)\,\subset\,-t_n\alpha\mathcal{W}_0\ \hbox{ for all large $n$},$$
whence
$$\big(\mathbb{R}^N\setminus(-t_n\alpha\mathcal{W}_0)\big)+[0,1]^N\ \subset\ \mathbb{R}^N\setminus(-\kappa_nC_\varepsilon)\ \hbox{ for all large $n$}.$$
Therefore, there exists $n_1\in\mathbb{N}$ such that, for all $n\ge n_1$ and for all $x\in U$, one has $x-x_n\in\mathbb{R}^N\setminus(-t_n\alpha\mathcal{W}_0)$ by~\eqref{xnW}, and then $x-\bar{x}_n\in\mathbb{R}^N\setminus(-\kappa_nC_e)$ since $x_n-\bar{x}_n\in[0,1]^N$. As a consequence, $u_0=\1_U\le\1_{\mathbb{R}^N\setminus(-\kappa_nC_\varepsilon)}(\cdot-\bar{x}_n)=v_{-\kappa_nC_\varepsilon}(0,\cdot-\bar{x}_n)$ in $\mathbb{R}^N$ for all $n\ge n_1$. Since the coefficients of~\eqref{main} are $\mathbb{Z}^N$-periodic and $\bar{x}_n\in\mathbb{Z}^N$, it follows from the comparison principle that, for all $n\ge n_1$,
$$u(t,x)\le v_{-\kappa_nC_\varepsilon}(t,x-\bar{x}_n)\ \hbox{ for all $t\ge 0$ and $x\in\mathbb{R}^N$}.$$
Thus, for all $n\ge n_1$, there holds $u(t_n,x_n)\le v_{-\kappa_nC_\varepsilon}(t_n,x_n-\bar{x}_n)\le\eta$ by~\eqref{vkappan}. One has then reached a contradiction with~\eqref{utnxneta}, and the proof of Lemma~\ref{lem:upLS} is thereby complete.

Next, we show in this paragraph that
\begin{equation}\label{hatW-L}
\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0=\underbrace{\Big\{x\in\mathbb{R}^N:\, \inf_{e\in\mathcal{L}(\mathcal{U}(U))}(x\cdot e+\widehat{c}(-e))> 0\Big\}}_{=:\mathcal{W}_4},
\end{equation}
where we recall from~\eqref{defhatc} that $\widehat{c}(e)=\sup_{\xi\in\mathbb{S}^{N-1}} \omega_0(\xi)\xi\cdot e\ge\omega_0(e)>0$. Take first any $x\in\mathbb{R}^N\setminus\mathcal{W}_4$, that is,
\begin{equation}\label{infhatc}
\inf_{e\in\mathcal{L}(\mathcal{U}(U))}(x\cdot e+\widehat{c}(-e))\le 0,
\end{equation}
and take any $y\in\mathcal{C}(\mathcal{U}(U))$. Lemma~\ref{lemma:P} implies that $y\cdot e\ge0$ for all $e\in\mathcal{L}(\mathcal{U}(U))$ and, together with $\widehat{c}(-e)\ge\omega_0(-e)$ and the uniform positivity of the $\omega_0(-e)$'s, one gets that $\inf_{e\in\mathcal{L}(\mathcal{U}(U))}\{y\cdot e+\widehat{c}(-e)\}>0$. Therefore, $x\neq y$. Call $\xi:=\widehat{x-y}=(x-y)/|x-y|$ and consider any $\eta\in(0,1)$. Then, $x=y +|x-y|\xi$ and, since
$$\inf_{e\in\mathcal{L}(\mathcal{U}(U))}(x\cdot e+\widehat{c}(-e)-\eta\,\widehat{c}(-e))<0$$
from~\eqref{infhatc} and the uniform positivity of the $\widehat{c}(-e)$'s, there is $e\in\mathcal{L}(\mathcal{U}(U))$ such that
$$\eta\,\widehat{c}(-e)\ge x\cdot e+\widehat{c}(-e)=y\cdot e +|x-y|\xi\cdot e+\widehat{c}(-e),$$
whence 
$$0\ge y\cdot e +|x-y|\xi\cdot e+(1-\eta)\,\widehat{c}(-e).$$
Since $y\in\mathcal{C}(\mathcal{U}(U))$ and $(1-\eta)\,\widehat{c}(-e)>0$, one has $y\cdot e\ge 0$ and then $\xi\cdot e <0$, whence
$$|x-y|\ge\frac{-(1-\eta)\,\widehat{c}(-e)}{\xi\cdot e}\ge(1-\eta)\,\omega_0(\xi).$$
Since this is true for any $\eta\in(0,1)$, one infers that $|x-y|\ge\omega_0(\xi)$, i.e. $x-y\not\in\mathcal{W}_0$. Therefore, $x\not\in\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0$. One has then shown that $\mathbb{R}^N\setminus\mathcal{W}_4\subset\mathbb{R}^N\setminus(\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0)$, that is,
\begin{equation}\label{subset1}
\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0\subset\mathcal{W}_4.
\end{equation}
Take now any $x\in\mathbb{R}^N\setminus\overline{\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0}$, let $y$ be the projection of $x$ onto the closed convex set $\overline{\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0}$, and call
$$e:=\widehat{y-x}=\frac{y-x}{|y-x|}.$$
One has that $e\cdot z\ge e\cdot y$ for all $z\in\overline{\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0}$. In particular, for any $\xi\in\mathcal{U}(U)$, since $0\in\mathcal{W}_0$, one has $e\cdot(t\xi)\ge e\cdot y$ for all $t\ge0$, whence $e\cdot\xi\ge0$. Therefore, $e\in\mathcal{L}(\mathcal{U}(U))$. Since $0\in\mathcal{C}(\mathcal{U}(U))$, one also gets that $\overline{\mathcal{W}_0}\subset\overline{\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0}$ and then $e\cdot(\omega_0(\xi)\xi)\ge e\cdot y$ for all $\xi\in{\mathbb{S}}^{N-1}$, whence $\widehat{c}(-e)+e\cdot y\le0$. As a consequence,
$$\widehat{c}(-e)+e\cdot x\le e\cdot (x-y)=-|x-y|<0$$
and $x\not\in\mathcal{W}_4$. One has then shown that $\mathbb{R}^N\setminus\overline{\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0}\subset\mathbb{R}^N\setminus\mathcal{W}_4$, that is, $\mathcal{W}_4\subset\overline{\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0}$. Since $\mathcal{W}_4$ is open and $\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0$ is open (because $\mathcal{W}_0$ is open) and convex (as the sum of two convex sets), one gets that $\mathcal{W}_4\subset{\rm{int}}(\overline{\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0})=\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0$. Together with~\eqref{subset1}, this completes the proof of~\eqref{hatW-L}.

Finally, assuming condition~\eqref{supt-plane}, one has $\widehat{c}(e)=c^*(e)$ for all $e\in\mathbb{S}^{N-1}$, and it follows from \eqref{EqF-L} and \eqref{hatW-L} that $\mathcal{W}_{\Lambda}=\mathcal{C}(\mathcal{U}(U))+\mathcal{W}_0$. By Theorem~\ref{Th:Ass}, the set $\mathcal{W}_{\Lambda}$ is then the spreading set. The proof of Corollary~\ref{theo:ass} is thereby complete.
