<!-- lemma-1: body_md -->
\label{lem:auxiliary-est}
    Let $f\in W^{1,2}_{\sigma}$, $g\in L^2_{\sigma,h}$, $a\in L^\infty_z$. Then the following hold:
    \begin{align}
        \| f \|_{L^2_z L^4_h}^2 &\le c \|f\| \|\nabla^hf\| \label{eq:ladyzh}\\
        \left| \int_\mathcal{O} a(z)g^l(x,y)f^j f^k\right| &\le c \|a\|_{L^\infty_z}\|g\|_{L^2_h}\|f\|\|\nabla^h f\|. \label{aux-est2}
    \end{align}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let us start with the first inequality. By $2$-d Ladyzhenskaya's inequality, for each $z$ we have
\begin{align*}
     \| f(\cdot,z) \|_{L^4_h}^2 &\le c \|f(\cdot,z)\|_{L^2_h}\|\nabla^h f(\cdot,z)\|_{L^2_h}.
\end{align*}
Now applying the Cauchy-Schwarz inequality gives the first statement \eqref{eq:ladyzh}.

For the second statement, we start by bounding $a$ uniformly and applying a horizontal H\"older with exponents $2,4,4$, followed 
by an application of \eqref{eq:ladyzh}:
\begin{align*}
    \left| \int_\mathcal{O} a(z)g(x,y)f^j f^k\right| &\le c\|a\|_{L^\infty_z}\|g\|_{L^2_h}  \int_{[0,1]} \|f(\cdot, z)\|_{L^4_h}^2 dz
    \le c \|a\|_{L^\infty_z}\|g\|_{L^2_h}\|f\|\|\nabla^h f\|.
\end{align*}

\label{unnamed}
    Let $\phi \in W^{1,\infty}$ satisfy that $\nabla \cdot \phi = 0$. Then for all $f\in H^1$ and $g \in H^1$ such that either $f = 0$ on $\partial \mathcal{O}$ or $g = 0$ on $\partial \mathcal{O}$, we have that
    \begin{align*}
        \left\langle \mathcal{T}_{\phi}f, g \right\rangle &= -\left\langle f, \mathcal{T}_{\phi}g \right\rangle,\\
        \left\langle \mathcal{S}_{\phi}f, g \right\rangle &= \left\langle f, \mathcal{T}_g\phi \right\rangle.
    \end{align*}
Therefore, we define the operators $\mathcal{T}_{\phi}^*$, $\mathcal{S}_{\phi}^*$ and $\mathcal{G}_{\phi}^*$ on $H^1$ by
\begin{align*}
    \mathcal{T}_{\phi}^*f &= -\mathcal{T}_{\phi}f,\\
    \mathcal{S}_{\phi}^*f &= \mathcal{T}_{f}\phi,\\
    \mathcal{G}_{\phi}^*f &= \mathcal{T}_{\phi}^*f + \mathcal{S}_{\phi}^*f.
\end{align*}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
The first result is classical, using a compactly supported approximation of either $f$ or $g$ in $H^1$ to conduct the integration by parts. For the second result, we simply observe that
    \begin{align*}
        \left\langle \mathcal{S}_{\phi}f, g \right\rangle = \sum_{j=1}^3\sum_{l=1}^3\left\langle f^j\partial_l\phi^j, g^l \right\rangle = \sum_{j=1}^3\sum_{l=1}^3\left\langle f^j, g^l\partial_l\phi^j \right\rangle = \sum_{j=1}^3\left\langle f^j, \mathcal{T}_g\phi^j \right\rangle = \left\langle f, \mathcal{T}_g\phi \right\rangle.
    \end{align*}

<!-- definition-1: body_md -->
\label{definitionofspacetimeweakmartingale}
Let $\mathcal{S}$ be a filtered probability space and $u_0 \in L^2_{\sigma}$ be deterministic. A pair $(u, \mathcal{W})$, where $\mathcal{W}$ is a Cylindrical Brownian Motion over $\mathfrak{U}$ with respect to $\mathcal{S}$, and $u$ is a progressively measurable process in $W^{1,2}_{\sigma}$ such that  $u \in  L^{\infty}\left([0,T];L^2_{\sigma}\right) \cap L^2\left([0,T];W^{1,2}_{\sigma}\right)$ $\mathbbm{P}-a.s.$, is said to be a martingale weak solution of (\ref{expanded ito form}) with respect to $\mathcal{S}$ and $u_0$ if the identity
\begin{align*}
     \left\langle u_t, \phi \right\rangle = \left\langle u_0, \phi \right\rangle &- \int_0^t\left\langle B(u_s, u_s), \phi \right\rangle ds  - \int_0^t \sum_{j=1}^2\left\langle \partial_j u_s, \partial_j \phi \right\rangle ds - \nu\int_0^t\left\langle \partial_3 u_s, \partial_3\phi \right\rangle ds\\ & -\int_0^t\left\langle  \frac{e_3 \wedge u_s}{\varepsilon}, \phi \right\rangle ds +\frac{1}{2} \int_0^t\sum_{i=1}^\infty\left\langle \mathcal{P}\tilde{\mathcal{G}}_iu_s, \tilde{\mathcal{G}}_i^*\phi \right\rangle ds -  \int_0^t\left\langle  \tilde{\mathcal{G}} u_s, \phi \right\rangle d\mathcal{W}_s
\end{align*}
holds for every $\phi \in W^{1,2}_{\sigma}$, $\mathbbm{P}-a.s.$ in $\mathbb{R}$, for all $t \in [0,T]$.

\label{definitionofstrongsolution2d}
Let $\mathcal{S}$ be a filtered probability space, $w_0 \in W^{1,2}_{\sigma,h}$ be deterministic,  and $\mathcal{W}$ a Cylindrical Brownian Motion over $\mathfrak{U}$ with respect to $\mathcal{S}$ . A process $w$ is said to be a strong solution of (\ref{Ito form for w}) with respect to $\mathcal{S}$, $w_0$ and $\mathcal{W}$ if $w$ is progressively measurable in $W^{2,2}_{\sigma,h}$, $w \in  C\left([0,T];W^{1,2}_{\sigma,h}\right) \cap L^2\left([0,T];W^{2,2}_{\sigma,h}\right)$ $\mathbbm{P}-a.s.$, and satisfies the identity

$\mathbbm{P}-a.s.$ in $L^2_{\sigma,h}$, for all $t \in [0,T]$. The solution is said to be unique if for any other solution $v$, $w$ and $v$ are indistinguishable.

<!-- theorem-1: body_md -->
\label{main result}
    Let $u_0 \in W^{1,2}_{\sigma,h}$ be deterministic, $\frac{1}{2} < \alpha$, and $(\nu^k)$, $(\varepsilon^k)$ be any two sequences of positive constants converging to zero such that $\left(\frac{\nu^k}{\varepsilon^k} \right)$ converges to $0 \leq \beta < \infty$. Then there exists a filtered probability space $\mathcal{S}$ and a Cylindrical Brownian Motion $\mathcal{W}$ over $\mathfrak{U}$ with respect to $\mathcal{S}$, such that:
    \begin{enumerate}
        \item For every $k \in \mathbb{N}$, there exists a process $u^k$ such that $(u^k,\mathcal{W})$ is a martingale weak solution of (\ref{expanded ito form}) with respect to $\mathcal{S}$ and $u_0$, for $\nu = \nu^k$ and $\varepsilon = \varepsilon^k$;
        \item There exists a unique strong solution $w$ of (\ref{Ito form for w}) with respect to $\mathcal{S}$, $u_0$ and $\mathcal{W}$;
        \item $(u^k) \longrightarrow w$ in $L^2\left(\Omega;L^\infty\left([0,T];L^2_{\sigma}\right)\right)$ as $k \longrightarrow \infty$.
    \end{enumerate}

Then there exists a unique strong solution of (\ref{Ito form for w}) with respect to $\mathcal{S}$, $u_0$ and $\mathcal{W}$. <!-- lemma-7: body_md -->
\label{uniform bounds on w}
    There exists a constant $C$ such that for any filtered probability space $\mathcal{S}$, Cylindrical Brownian Motion $\mathcal{W}$ and any $k$, the unique strong solution $w^k$ of (\ref{Ito form for w k}) with respect to $\mathcal{S}$, $u_0$ and $\mathcal{W}$ satisfies
    \begin{equation}\label{Estimate1}\sup_{t\in[0,T]}\left\Vert w^k_t \right\Vert_{H^1}^2 + \int_0^T\left\Vert w^k_s \right\Vert_{H^2}^2ds \leq C\left\Vert u_0 \right\Vert_{H^1}^2.\end{equation}
Furthermore for any $m$, the unique strong solution $w^{k,m}$ of (\ref{Ito form for w k}) with respect to $\mathcal{S}$, $w^m_0$ and $\mathcal{W}$ satisfies the bound
\begin{equation}
    \label{Estimate2}
    \mathbbm{E}\left[\sup_{t\in[0,T]}\left\Vert w^{k,m}_t \right\Vert_{H^2}^2 + \int_0^T\left\Vert w^{k,m}_s \right\Vert_{H^3}^2ds \right] \leq C\left\Vert w^m_0 \right\Vert_{H^2}^2.
\end{equation}
<!-- end lemma-7 -->
<!-- proof-5: body_md -->
In the absence of damping, the bound (\ref{Estimate1}) was shown in Subsection 3.1 of [\cite{goodair2025navier}] and we shall defer some of the details to there. Nevertheless, we present the core idea here. Setting $\eta^k \coloneqq \textnormal{curl}w^k$, then the scalar $\eta^k$ satisfies
$$d\eta^k_t = -(w^k_t \cdot \nabla^h)\eta^k_t \, dt + \Delta^h \eta^k_t dt - \sqrt{\frac{2\nu^k}{\varepsilon^k}} \eta^k_t \, dt - \sum_i \mathcal{T}_{\xi_i^h}\eta^k_t dW^i_t + \frac{1}{2}\sum_{i=1}^\infty \mathcal{T}_{\xi_i^h}^2 \eta^k_t \, dt$$
weakly. The pure transport noise provides a perfect cancellation in the $L^2$ energy computation, whilst the damping term can simply be dropped, leading to the inequality
$$\left\Vert \eta^k_t \right\Vert^2 + 2\int_0^t\left\Vert \nabla^h\eta^k_s \right\Vert^2ds \leq \left\Vert \textnormal{curl}u_0 \right\Vert^2$$
where inequality, rather than equality, is due to the damping. By taking the supremum in time, and using equivalence of the $H^1$ norm and $L^2$ norm of the curl, the estimate is proven.

The $H^2$ energy bound is somewhat more involved, so we provide a proof. We will drop the $k,m$ superscripts for the calculations in the proof for notational convenience. Using the energy identity, we have
\begin{align*}
    d\|w_t\|_{H^2}^2 &= -2\|\nabla^h w_t\|_{H^2}^2 dt -2 \left\langle \mathcal P (w_t \cdot\nabla )w_t, w_t \right\rangle_{H^2} dt-2\left\langle \sqrt{\frac{2\nu}{\varepsilon}}w_t, w_t \right\rangle_{H^2}dt \\
    &-2\left\langle \mathcal P \mathcal G^hw_t, w_t \right\rangle_{H^2}d\mathcal{W}_t\\
    &+\sum_i \left\langle \mathcal P(\mathcal{G}_{i}^h)^2w_t, w_t \right\rangle_{H^2}dt + \sum_i \left\Vert \mathcal P \mathcal{G}_{i}^hw_t \right\Vert^2_{H^2}dt.
\end{align*}
Let us now treat all the terms on the right hand side. In the first line, we keep the negative diffusion term, drop the negative damping term, and estimate the nonlinear term as follows
\begin{align*}
    |\left\langle \mathcal P (w_t \cdot\nabla )w_t, w_t \right\rangle_{H^2}| &\le \left\Vert (w_t \cdot\nabla )w_t \right\Vert_{H^1} \left\Vert w_t \right\Vert_{H^3}.
\end{align*}
We now note 
\begin{align*}
    \left\Vert (w_t \cdot\nabla )w_t \right\Vert_{H^1} &\lesssim \left\Vert  w_t \right\Vert^2_{W^{1,4}} + \left\Vert w _t\nabla^2w_t \right\Vert \\ &\lesssim \left\Vert w_t \right\Vert_{H^1}\left\Vert w_t \right\Vert_{H^2} + \left\Vert w_t \right\Vert^{\frac{1}{2}}\left\Vert w_t \right\Vert_{H^1}^{\frac{1}{2}} \left\Vert w_t \right\Vert_{H^2}^{\frac{1}{2}}\left\Vert w_t \right\Vert_{H^3}^{\frac{1}{2}} \lesssim \left\Vert w_t \right\Vert_{H^1} \left\Vert w_t \right\Vert_{H^2}^{\frac{1}{2}}\left\Vert w_t \right\Vert_{H^3}^{\frac{1}{2}},
\end{align*}
which gives 
\begin{align*}
    |\left\langle \mathcal P (w_t \cdot\nabla )w_t, w_t \right\rangle_{H^2}| &\le c \left\Vert w_t \right\Vert_{H^1} \left\Vert w_t \right\Vert_{H^2}^{\frac{1}{2}}\left\Vert w_t \right\Vert_{H^3}^{\frac{3}{2}} \le c\left\Vert w_t \right\Vert_{H^2}^{\frac{1}{2}}\left\Vert w_t \right\Vert_{H^3}^{\frac{3}{2}} \\
    \le \frac{1}{2}\left\Vert w_t \right\Vert_{H^3}^2 + c\left\Vert w_t \right\Vert_{H^2}^2,
\end{align*}
using that the $H^1$ norm of $w$ is bounded uniformly in time by a deterministic constant, and a Young's inequality in the last line.

Next we treat the martingale term using the BDG inequality, and the It\^o-Stratonovich correction term along with the quadratic variation term. For this purpose we recall the bounds:
\begin{align}
    \left\langle \mathcal P\mathcal G_i^2f, f \right\rangle_{H^2} +  \left\Vert \mathcal P\mathcal G_if \right\Vert_{H^2}^2  &\leq c\left\Vert \xi_i \right\Vert_{W^{4,\infty}}^2\left\Vert f \right\Vert_{H^2}^2, \label{bigbound1} \\
    \left\langle \mathcal P\mathcal G_i f, f \right\rangle_{H^2}^2 &\leq c\left\Vert \xi_i \right\Vert^2_{W^{3,\infty}}\left\Vert f \right\Vert^4_{H^2}, \label{bigbound2} 
\end{align}
which were proved in [\cite{goodair2025closed}].
For the martingale term, we have:
\begin{align*}
    \mathbb E \left[ \sup_{s\in[0,t]}\left| \int_0^s\left\langle \mathcal P \mathcal G^hw_r, w_r \right\rangle_{H^2}d\mathcal{W}_r \right|\right] &\le c \mathbb E \left[ \left( \int_0^t \left|\left\langle \mathcal P \mathcal G^hw_s, w_s \right\rangle_{H^2}\right|^2ds \right)^{\frac{1}{2}} \right] \\
    &\le c \mathbb E \left[ \left( \int_0^t \left\Vert w_s \right\Vert^4_{H^2} ds \right)^{\frac{1}{2}} \right] \\
    &\le c \mathbb E \left[\sup_{s\in [0,t]}\left\Vert w_s \right\Vert_{H^2} \left( \int_0^t \left\Vert w_s \right\Vert^2_{H^2} ds \right)^{\frac{1}{2}} \right] \\
    &\le \frac{1}{2} \mathbb E \left[ \sup_{s\in [0,t]}\left\Vert w_s \right\Vert_{H^2}^2 \right] + c \mathbb E \left[\int_0^t \left\Vert w_s \right\Vert^2_{H^2} ds \right].
\end{align*}
For the last line, we simply apply \eqref{bigbound1}
\begin{align*}
    \sum_i \left(\left\langle \mathcal P(\mathcal{G}_{i}^h)^2w_t, w_t \right\rangle_{H^2} +  \left\Vert \mathcal P \mathcal{G}_{i}^hw_t \right\Vert^2_{H^2} \right) \le c \left\Vert w_t \right\Vert_{H^2}^2.
\end{align*}
Collecting all the estimates, taking the supremum and expectations in the energy identity, we have:
\begin{align*}
    \mathbb E \left[ \sup_{t\in[0,t']}\left\Vert w_t \right\Vert_{H^2}^2 \right] + \mathbb E \left[ \int_0^{t'}\left\Vert w_t \right\Vert_{H^3}^2 dt \right] &\le c \mathbb E\left[\left\Vert w_0 \right\Vert_{H^2}^2\right] + c\int_0^{t'} \mathbb E\left[ \sup_{s\in[0,t]}\left\Vert w_s \right\Vert_{H^2}^2 \right]dt.
\end{align*}
Now an application of the standard Gr\"onwall inequality gives
\begin{equation*}
    \mathbb E \left[ \sup_{t\in[0,T]}\left\Vert w_t \right\Vert_{H^2}^2 \right] + \mathbb E \left[ \int_0^{T}\left\Vert w_t \right\Vert_{H^3}^2 dt \right] \le c \left\Vert w_0 \right\Vert_{H^2}^2,
\end{equation*}
which is exactly the desired result.

<!-- proof-6: body_md -->
The proof follows by establishing continuity in the damping parameter and initial data through a Gr\"onwall argument. To ease the notations, consider $w$ to be the solution to
    \begin{equation*}
        dw_t= -A_hw_t dt - B(w_t,w_t)dt - \beta w_tdt - \mathcal P\mathcal G^h w_t d\mathcal{W}_t + \frac{1}{2}\sum_i \mathcal P(\mathcal{G}_{i} ^h)^2w_t dt,
    \end{equation*}
    with initial data $w_0\in W^{1,2}_{\sigma, h}$, and let $\tilde w$ solve the same equation with parameter $\tilde \beta$ and initial data $\tilde w_0$. Let further $v:= w-\tilde w$. Then the difference process satisfies the equation:
    \begin{equation*}
        dv_t = -A_h v_t dt - B(w_t,v_t)dt - B(v_t,\tilde w_t)dt -\beta v dt-(\beta - \tilde \beta)\tilde w dt -\mathcal P\mathcal G^h v_t d\mathcal{W}_t + \frac{1}{2}\sum_i \mathcal P(\mathcal{G}_{i} ^h)^2v_t dt,
    \end{equation*}
    with $v_0=w_0-\tilde w_0$. The energy identity now yields:
    \begin{align*}
        d\left\Vert v_t \right\Vert^2 = &-2\left\Vert \nabla^h v_t \right\Vert^2 dt -2 \left\langle B(v_t,\tilde w_t), v_t \right\rangle dt-2\beta \left\Vert v_t \right\Vert^2dt -2(\beta - \tilde \beta)\left\langle \tilde w_t, v_t \right\rangle dt\\
        & -2 \left\langle \mathcal P \mathcal G^hv_t, v_t \right\rangle d\mathcal{W}_t\\
        &+ \sum_i \left( \left\langle \mathcal P(\mathcal{G}_{i} ^h)^2v_t, v_t \right\rangle + \left\Vert \mathcal P\mathcal{G}_{i}^hv_t \right\Vert^2\right)dt.
    \end{align*}
    For the terms in the first line, we drop the $v$ damping term, and estimate the rest as follows:
    \begin{equation*}
        \left|\left\langle B(v_t,\tilde w_t), v_t \right\rangle \right| \le c\left\Vert \nabla^h \tilde w_t \right\Vert\left\Vert v_t \right\Vert_{L^4}^2 \le c \left\Vert v_t \right\Vert\left\Vert v_t \right\Vert_{H^1}\le \frac{1}{2}\left\Vert v_t \right\Vert_{H^1}^2 + c\left\Vert v_t \right\Vert^2,
    \end{equation*}
    where we have exploited the deterministic, uniform in time bound on $\|\tilde w\|_{H^1}$ from \ref{uniform bounds on w}, and a $2$-d Ladyzhenskaya inequality. For the remaining term we have
    \begin{equation*}
        \left|(\beta - \tilde \beta)\left\langle \tilde w_t, v_t \right\rangle \right|\le |(\beta - \tilde \beta)|^2 + c\|v_t\|^2,
    \end{equation*}
    by Cauchy-Schwarz followed by a Young's inequality.\\

    The martingale term, and the remaining noise terms are dealt with exactly the same as the ones in the proof of lemma \ref{uniform bounds on w}, using the corresponding bounds:
    \begin{align*}
    \left\langle \mathcal P\mathcal G_i^2f, f \right\rangle +  \left\Vert \mathcal P\mathcal G_if \right\Vert_{}^2  &\leq c\left\Vert \xi_i \right\Vert_{W^{2,\infty}}^2\left\Vert f \right\Vert_{}^2,\\
    \left\langle \mathcal P\mathcal G_i f, f \right\rangle_{}^2 &\leq c\left\Vert \xi_i \right\Vert^2_{W^{1,\infty}}\left\Vert f \right\Vert^4_{},
    \end{align*}
    which were also proved in [\cite{goodair2025closed}]. For the martingale term, we have:
    \begin{align*}
        \mathbb E \left[ \sup_{s\in[0,t]}\left| \int_0^s\left\langle \mathcal P \mathcal G^hv_r, v_r \right\rangle d\mathcal{W}_r \right|\right] &\le c \mathbb E \left[ \left( \int_0^t \left|\left\langle \mathcal P \mathcal G^hv_s, v_s \right\rangle\right|^2ds \right)^{\frac{1}{2}} \right] \\
        &\le c \mathbb E \left[ \left( \int_0^t \left\Vert v_s \right\Vert^4 ds \right)^{\frac{1}{2}} \right] \\
        &\le c \mathbb E \left[\sup_{s\in [0,t]}\left\Vert v_s \right\Vert \left( \int_0^t \left\Vert v_s \right\Vert^2 ds \right)^{\frac{1}{2}} \right] \\
        &\le \frac{1}{2} \mathbb E \left[ \sup_{s\in [0,t]}\left\Vert v_s \right\Vert^2 \right] + c \mathbb E \left[\int_0^t \left\Vert v_s \right\Vert^2 ds \right],
    \end{align*}
    and for the remaining ones
    \begin{align*}
        \sum_i \left(\left\langle \mathcal P(\mathcal{G}_{i}^h)^2v_t, v_t \right\rangle +  \left\Vert \mathcal P \mathcal{G}_{i}^hv_t \right\Vert^2 \right) \le c \left\Vert v_t \right\Vert^2.
    \end{align*}
    Collecting all the estimates, we have:
    \begin{align*}
        \mathbb E \left[ \sup_{t\in[0,t']}\left\Vert v_t \right\Vert^2 \right]  &\le c \mathbb E\left[\left\Vert v_0 \right\Vert^2\right] + c|\beta - \tilde \beta|^2 + c\int_0^{t'} \mathbb E\left[ \sup_{s\in[0,t]}\left\Vert v_s \right\Vert^2 \right]dt.
    \end{align*}
    Applying Gr\"onwall's lemma, we obtain
    \begin{equation*}
        \mathbb E \left[ \sup_{t\in[0,T]}\left\Vert v_t \right\Vert^2 \right] \le c\left(\mathbb E\left[\left\Vert v_0 \right\Vert^2\right] + |\beta - \tilde \beta|^2\right),
    \end{equation*}
    which proves both lemmas.

    \item \label{item2} $u^k$ is obtained as the $n \rightarrow \infty$ limit of Galerkin Approximations $u^{k,n}$ satisfying

\end{align}

The solutions $(u^{k,n})$ above, enjoy an additional uniform (in $n$) bound on \\ $L^4\left(\Omega; L^\infty\left([0,T];L^2_{\sigma}\right) \cap L^2\left([0,T];W^{1,2}_{\sigma} \right) \right)$;

\end{enumerate}

The existence of a martingale weak solution to (\ref{some main equation 2 Ito k}), as a limit of Galerkin Approximations, was given in the isotropic case and without rotation on a bounded domain in [\cite{goodair2024weak}] Theorem 5.1. Anisotropy and alteration of the domain does not disturb the proof in a significant way, as one only suffers the worse bound (\ref{bigbound1NEW}) which is entirely sufficient for the proof in [\cite{goodair2024weak}] by combining with the viscous term. Likewise, the rotational term is a bounded linear function of the solution and does not affect the proof. \\

Let us now address the uniform (in $n$) bound in $L^4\left(\Omega; L^\infty\left([0,T];L^2_{\sigma}\right) \cap L^2\left([0,T];W^{1,2}_{\sigma} \right) \right)$. This is fairly standard, but we sketch the argument for completeness. Applying the energy identity, we deduce
\begin{align*}
    \left\Vert u^{k,n}_t \right\Vert^2 + \int_0^t\left\Vert \nabla^hu^{k,n}_s \right\Vert^2 + \nu_k\left\Vert \partial_3^2u^{k,n}_s \right\Vert^2 ds \le &c\left\Vert u^{k,n}_0 \right\Vert^2 + c\left| \int_0^t \left\langle \mathcal{P}_n\mathcal{P} \tilde{\mathcal{G}} u^{k,n}_s, u^{k,n}_s \right\rangle d\mathcal{W}^n_s \right| +\\ &+ c \int_0^t \sum_i \left\langle \mathcal{P}_n\mathcal{P}\tilde{\mathcal{G}}_i\mathcal{P}\tilde{\mathcal{G}}_iu^{k,n}_s, u^{k,n}_s \right\rangle + \left\Vert \mathcal{P}_n\mathcal{P} \tilde{\mathcal{G}}_i u^{k,n}_s \right\Vert^2 ds.
\end{align*}
Using the bound, \eqref{bigbound1NEW} and taking $\delta$ small enough, we see
\begin{align*}
    \left\Vert u^{k,n}_t \right\Vert^2 + \int_0^t\left\Vert \nabla^hu^{k,n}_s \right\Vert^2 + \nu_k \left\Vert \partial_3^2u^{k,n}_s \right\Vert^2 ds \le &c\left\Vert u^{k,n}_0 \right\Vert^2 + c\left| \int_0^t \left\langle \mathcal{P}_n\mathcal{P} \tilde{\mathcal{G}} u^{k,n}_s, u^{k,n}_s \right\rangle d\mathcal{W}^n_s \right| +\\ &+ c_\delta \int_0^t \left\Vert u^{k,n}_s \right\Vert^2 ds.
\end{align*}
We now take supremums in time, square the inequality and take expectations to obtain (with some elementary bounds and an application of the BDG inequality at the last step)
\begin{align*}
    \mathbb E \left[ \left(\sup_{t'\le t}\left\Vert u^{k,n}_{t'} \right\Vert^2 +  \int_0^t\left\Vert \nabla^hu^{k,n}_s \right\Vert^2 + \nu_k \left\Vert \partial_3^2u^{k,n}_s \right\Vert^2 ds\right)^2 \right] \le &c\mathbb E \left[ \left\Vert u^{k,n}_0 \right\Vert^4 \right] \\ &+c \mathbb E \left[ \sup_{t' \le t}\left| \int_0^{t'} \left\langle \mathcal{P}_n\mathcal{P} \tilde{\mathcal{G}} u^{k,n}_s, u^{k,n}_s \right\rangle d\mathcal{W}^n_s \right|^2 \right] \\ &+ c_\delta \mathbb E \left[ \int_0^t \left\Vert u^{k,n}_s \right\Vert^4 ds \right] \\
    &\le c\mathbb E \left[ \left\Vert u^{k,n}_0 \right\Vert^4 \right] + c  \int_0^t \mathbb E \left[ \sup_{t'\le s}\left\Vert u^{k,n}_{t'} \right\Vert^4 \right] ds.
\end{align*}
Gr\"onwall's lemma now implies
\begin{equation*}
    \mathbb E \left[ \left( \sup_{t\le T}\left\Vert u^{k,n}_t \right\Vert^2  + \int_0^T\left\Vert \nabla^hu^{k,n}_s \right\Vert^2 + \nu_k \left\Vert \partial_3^2u^{k,n}_s \right\Vert^2 ds\right)^2 \right] \le c\mathbb E \left[ \left\Vert u^{k,n}_0 \right\Vert^4 \right],
\end{equation*}
where the constant $c$ is uniform both in $k,n$, and due to the uniform boundedness of the initial data for the Galerkin equations, this establishes the desired estimate.\\

The remaining point of this proposition is that the martingale weak solution $(u^k, \mathcal{W})$ is constructed so that $\mathcal{W}$ is independent of $k$, as well as the convergence $(w^{k,m,n}) \longrightarrow w^{k,m}$.
The strategy allowing to achieve this, is presented in [\cite{breit-hofmanova2016}].
The first step is to fix a probability space with a Cylindrical Brownian Motion $\mathcal{W}$. Then, we construct the probabilistically strong solution to the equations for $u^{k,n},w^{k,m}$ driven by this $\mathcal{W}$. Tightness is a consequence of the standard uniform estimates used in the proof referenced above. Now the key idea is to look at the following collection of variables indexed by $n$:
\begin{equation*}
    \left\{ \left( u^{k,n},w^{k,m}, \mathcal{W} \right)_{k,m\in \mathbb{N}}\,|\, n\in \mathbb{N} \right\}.
\end{equation*}
Passing to a subsequence, which we don't relabel, we get that this family converges in law to the family $(u^k,w^{k,m}, \mathcal{W})_{k,m \in \mathbb{N}}$. Applying the Skorokhod embedding theorem, we now get the family
\begin{equation*}
    \left\{ \left( \tilde{u}^{k,n},\tilde{w}^{k,m,n}, \tilde{\mathcal{W}}^n \right)_{k,m\in \mathbb{N}}\,|\, n\in \mathbb{N} \right\}
\end{equation*}
converging almost surely to $(\tilde u^k, \tilde w^{k,m},\tilde{ \mathcal{W}})_{k,m \in \mathbb{N}}$. The topologies in which this happens are exactly the ones implied by the available uniform bounds leading to tightness for $u^{k,n}$. For $\tilde w^{k,m,n}$, we get almost sure convergence in $C\left([0,T];W^{2,2}_{\sigma,h}\right) \cap L^2\left([0,T];W^{3,2}_{\sigma,h}\right)$ due to the regularity of the initial condition, as well as  $L^2\left(\Omega; C\left([0,T];W^{1,2}_{\sigma,h}\right) \cap L^2\left([0,T];W^{2,2}_{\sigma,h}\right) \right)$ due to the deterministic bound from Lemma \ref{uniform bounds on w} allowing to extend to the probabilistic $L^2$ mode of convergence. The identification of the equations solved by the variables is completely standard, so we do not reproduce it. From now on, we work on this common probability space, and drop the tildes.

<!-- proof-8: body_md -->
We choose the filtered probability space $\mathcal{S}$ and Cylindrical Brownian Motions $\mathcal{W}$, $(\mathcal{W}^n)$ as specified in Proposition \ref{prop choosing spaces}. Furthermore, we introduce a sequence of more regular initial conditions $(w^m_0)$, $w^m_0 \in W^{2,2}_{\sigma,h}$ which converges to $u_0$ in $W^{1,2}_{\sigma,h}$. Let us now fix some notation for the solutions and their approximations, as required in the proof:
    \begin{itemize}
        \item $(u^k)$ and $(u^{k,n})$ are as in Proposition \ref{prop choosing spaces};
        \item $w$ is the unique strong solution of (\ref{Ito form for w}) with respect to $\mathcal{S}$, $u_0$ and $\mathcal{W}$;
        \item $(w^k)$ are the unique strong solutions of (\ref{Ito form for w k}) with respect to $\mathcal{S}$, $u_0$ and $\mathcal{W}$;
        \item $(w^{k,m})$ are the unique strong solutions of (\ref{Ito form for w k}) with respect to $\mathcal{S}$, $(w^m_0)$ and $\mathcal{W}$;
        \item $(w^{k,m,n})$ are the unique strong solutions of (\ref{Ito form for w k}) with respect to $\mathcal{S}$, $(w^m_0)$ and $(\mathcal{W}^n)$.
    \end{itemize}
We begin the proof by establishing sufficiency of working at the level of the  sequences. We have that
    \begin{align*}
        \mathbbm{E}\left[\left\Vert u^k - w \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2 \right] &\leq 2 \mathbbm{E}\left[\left\Vert u^k - w^k  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right] + 2\mathbbm{E}\left[\left\Vert w^k - w  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right]
    \end{align*}
so by Lemma \ref{damping convergence}, 
\begin{align*}
       \lim_{k 
       \rightarrow \infty} \mathbbm{E}\left[\left\Vert u^k - w \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2 \right] &\leq 2\lim_{k \rightarrow \infty} \mathbbm{E}\left[\left\Vert u^k - w^k  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right].
    \end{align*}
Moreover,
$$\mathbbm{E}\left[\left\Vert u^k - w^k  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right] \leq 2\mathbbm{E}\left[\left\Vert u^k - w^{k,m}  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right] + 2\mathbbm{E}\left[\left\Vert w^{k,m} - w^k  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right]$$
and due to the weak* convergence specified in item \ref{item2} and the strong convergence in \ref{item3} of Proposition \ref{prop choosing spaces},
$$\mathbbm{E}\left[\left\Vert u^k - w^{k,m}  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right] \leq \liminf_{n \rightarrow \infty}\mathbbm{E}\left[\left\Vert u^{k,n} - w^{k,m,n}  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right].$$
Therefore, we have that
\begin{align*}
    &\mathbbm{E}\left[\left\Vert u^k - w^k  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right]\\
    &\leq 2\lim_{m \rightarrow \infty} \liminf_{n \rightarrow \infty}\mathbbm{E}\left[\left\Vert u^{k,n} - w^{k,m,n}  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right] + 2\lim_{m \rightarrow \infty}\mathbbm{E}\left[\left\Vert w^{k,m} - w^k  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right]\\
    &= 2\lim_{m \rightarrow \infty} \liminf_{n \rightarrow \infty}\mathbbm{E}\left[\left\Vert u^{k,n} - w^{k,m,n}  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right]
\end{align*}
having applied Lemma \ref{ic convergence}. In total,
\begin{equation} \label{in total}
   \lim_{k 
       \rightarrow \infty} \mathbbm{E}\left[\left\Vert u^k - w \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2 \right] \leq 4\lim_{k 
       \rightarrow \infty}\lim_{m \rightarrow \infty}\liminf_{n \rightarrow \infty}\mathbbm{E}\left[\left\Vert u^{k,n} - w^{k,m,n}  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right]
\end{equation}
so the computation boils down to treating $\left\Vert u^{k,n} - w^{k,m,n}  \right\Vert^2$. At this point we will drop the superscripts, thus renaming the variables:
\begin{equation*}
    u^{k,n} \mapsto u, \quad w^{k,m,n} \mapsto w, \quad \mathcal{W}^n \mapsto \mathcal{W},
\end{equation*}
and the dependence on parameters will be tracked carefully in the constants, to ensure the correct dependencies. We would like to carry out an energy estimate for the difference, but in order to do this, we need to look at the difference corrected by the boundary corrector. We thus define
$$v:= u-w-\mathcal{B}[w].$$
To be able to write down the energy identity for $v$, we need to first understand the semimartingale decomposition of $\mathcal{B}[w]$. Due to the specific form \eqref{eq:corrector-def}, the deterministic linear map $\mathcal{B}[\cdot]$ commutes with the horizontal derivatives. In combination with $w$ only depending on the horizontal variables, we have
\begin{equation}\label{eq:dB}
\begin{aligned}
d\mathcal B={}&-\mathcal B[B(w,w)]\,dt-A_h\mathcal B\,dt
-\sqrt{\frac{2\nu}{\varepsilon}}\mathcal B\,dt\\
&+\frac12\sum_i\mathcal B[\mathcal P(\mathcal G_i^h)^2w]\,dt
-\sum_i\mathcal B[\mathcal P\mathcal G_i^hw]\,dW^{i}.
\end{aligned}
\end{equation}
With this decomposition in place, we can apply the energy identity to $v$, understood as $u-w- \mathcal{B}[w]$:
\begin{align}
d\|v\|^2=&
-2\langle\mathcal P_nB(u,u)-B(w,w)-\mathcal B[B(w,w)],v\rangle\,dt\nonumber\\
&-2\langle\mathcal P_nA_hu-A_hw-A_h\mathcal B,v\rangle\,dt\nonumber\\
&-2\nu\langle\mathcal P_nA_zu,v\rangle\,dt
-\frac2\varepsilon\langle\mathcal P_n\mathcal P(e_3\wedge u),v\rangle\,dt\nonumber\\
&+2\sqrt{\frac{2\nu}{\varepsilon}}\langle w+\mathcal B,v\rangle\,dt\nonumber\\
&+\sum_i\langle
\mathcal P_n\mathcal P\widetilde{\mathcal G}_i\mathcal P\widetilde{\mathcal G}_iu-\mathcal P(\mathcal G_i^h)^2w-\mathcal B[\mathcal P(\mathcal G_i^h)^2w],v\rangle\,dt
\nonumber\\
&+\sum_i\|\mathcal P_n\mathcal P\widetilde{\mathcal G}_iu-\mathcal P\mathcal G_i^hw-\mathcal B[\mathcal P\mathcal G_i^hw]\|^2\,dt
\nonumber\\
&-2\sum_i\langle
\mathcal P_n\mathcal P\widetilde{\mathcal G}_iu-\mathcal P\mathcal G_i^hw-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle\,dW^{i}.
\label{eq:exact}
\end{align}
We now wish to bound the right hand side by terms that can be treated by an application of a (stochastic) Gr\"onwall lemma \ref{gronny}. This is somewhat delicate, as one has to exploit various exact cancellations permitted by the specific boundary layer corrector, as well as top order derivative cancellation between the It\^o-Stratonovich correction and the quadratic variation term.\\

\noindent In order to see the above features, we need to isolate the error coming from the Galerkin projections. To this end we note
$$\mathcal P_nv=v+(I-\mathcal P_n)(w+\mathcal B),$$ and consequently
$$\langle\mathcal P_n f,v\rangle
=\langle f,v\rangle+
\langle f,(I-\mathcal P_n)(w+\mathcal B)\rangle.$$
For the nonlinear term, expand \(u=v+w+\mathcal B\):
$$ B(u,u)-B(w,w)
=B(u,v)+B(v,w)+B(\mathcal B,w)+B(u,\mathcal B),$$
which (exploiting the skew-symmetry of $B$) gives
\begin{equation} \label{eq:galerkin-res-1}
    \langle\mathcal P_nB(u,u)-B(w,w),v\rangle
=\langle B(v,w)+B(\mathcal B,w)+B(u,\mathcal B),v\rangle\\
+\langle B(u,u),(I-\mathcal P_n)(w+\mathcal B)\rangle.
\end{equation}
In both the horizontal and vertical viscous terms we can drop the projection, as we can choose the basis for the Stokes operator, that also diagonalises the horizontal Laplacian. In this case:
$$\langle\mathcal P_nA_hu-A_hw-A_h\mathcal B,v\rangle
=\|\nabla^hv\|^2,$$as well as
$$\langle\mathcal P_nA_zu,v\rangle
=\|\partial_3v\|^2+\langle A_z\mathcal B,v\rangle.$$
As for the Coriolis term, we get
\begin{equation}\label{eq:rotation-exact}
\langle\mathcal P_n\mathcal P(e_3\wedge u),v\rangle
=\langle e_3\wedge\mathcal B,v\rangle
+\langle e_3\wedge u,(I-\mathcal P_n)(w+\mathcal B)\rangle,
\end{equation}
as $\langle e_3\wedge v,v\rangle=0$ due to skew symmetry, and $\langle e_3\wedge w,v\rangle=0$ due to a Taylor-Proudman theorem asserting that $ e_3 \wedge f$ is a gradient if and only if the (mean free) $f$ depends only on the horizontal variables.\\
\noindent
In the It\^o-Stratonovich correction term, we note that $ \mathcal P\widetilde{\mathcal G}_i\mathcal P\widetilde{\mathcal G}_iu = \mathcal P\widetilde{\mathcal G}_i\mathcal P\widetilde{\mathcal G}_i v + \mathcal P (\mathcal G^h_i)^2 w + \mathcal P\widetilde{\mathcal G}_i\mathcal P\widetilde{\mathcal G}_i \mathcal{B} $, so that we have
\begin{equation}\label{eq:ito-exact}
\begin{aligned}
&\langle
\mathcal P_n\mathcal P\widetilde{\mathcal G}_i\mathcal P\widetilde{\mathcal G}_iu-\mathcal P(\mathcal G_i^h)^2w-\mathcal B[\mathcal P(\mathcal G_i^h)^2w],v\rangle=\\
&\quad=
\langle\mathcal P\widetilde{\mathcal G}_iv,\widetilde{\mathcal G}_i^*v\rangle+\langle\mathcal P\widetilde{\mathcal G}_i\mathcal B,\widetilde{\mathcal G}_i^*v\rangle
-\langle\mathcal B[\mathcal P(\mathcal G_i^h)^2w],v\rangle +
\langle\mathcal P\widetilde{\mathcal G}_iu,\widetilde{\mathcal G}_i^*(I-\mathcal P_n)(w+\mathcal B)\rangle.
\end{aligned}
\end{equation}
For the quadratic variation term, we note that
\begin{equation*}\label{eq:diffusion-split}
\begin{aligned}
&\mathcal P_n\mathcal P\widetilde{\mathcal G}_iu-\mathcal P\mathcal G_i^hw-\mathcal B[\mathcal P\mathcal G_i^hw]\\
&\quad=
\mathcal P_n\mathcal P\bigl(\widetilde{\mathcal G}_iv+\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw]\bigr)-
(I-\mathcal P_n)\bigl(\mathcal P\mathcal G_i^hw+\mathcal B[\mathcal P\mathcal G_i^hw]\bigr),
\end{aligned}
\end{equation*}
where the two terms are orthogonal. This along with bounding the norm of $\mathcal P_n$ by one, gives
\begin{equation}\label{eq:qv}
\begin{aligned}
&\|\mathcal P_n\mathcal P\widetilde{\mathcal G}_iu-\mathcal P\mathcal G_i^hw-\mathcal B[\mathcal P\mathcal G_i^hw]\|^2\\
&\quad\le
\|\mathcal P\widetilde{\mathcal G}_iv+\mathcal P\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw]\|^2+
\|(I-\mathcal P_n)(\mathcal P\mathcal G_i^hw+\mathcal B[\mathcal P\mathcal G_i^hw])\|^2.
\end{aligned}
\end{equation}
Finally, for the integrand of the martingale term we use that 
\begin{equation}\label{eq:mart-split}
\begin{aligned}
&\langle\mathcal P_n\mathcal P\widetilde{\mathcal G}_iu-\mathcal P\mathcal G_i^hw-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle\\
&\quad=
\langle\widetilde{\mathcal G}_iv+\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle+\langle\widetilde{\mathcal G}_iu,(I-\mathcal P_n)(w+\mathcal B)\rangle.
\end{aligned}
\end{equation}
Let us now collect all the Galerkin projection remainder terms, that is the terms from equations \eqref{eq:galerkin-res-1}, \eqref{eq:rotation-exact}, \eqref{eq:ito-exact}, \eqref{eq:qv} and \eqref{eq:mart-split} into $R_i$ for $i \in \{1,2,3,4,5\}$, where $R_5$ is an It\^o integrand.
We can now update the energy balance to
\begin{align}\label{eq:EE_to_est}
    d\|v\|^2\le& -2\langle B(v,w)+B(\mathcal B,w)+B(u,\mathcal B) -\mathcal B[B(w,w)],v\rangle dt\nonumber\\
    &-2 \|\nabla^hv\|^2 dt\nonumber\\
    &-2\nu \|\partial_3v\|^2dt -2\nu\langle A_z\mathcal B,v\rangle dt -\frac{2}{\varepsilon} \langle e_3\wedge\mathcal B,v\rangle dt\nonumber\\
    &+2\sqrt{\frac{2\nu}{\varepsilon}}\langle w+\mathcal B,v\rangle\,dt\nonumber\\ 
    &+\sum_i \left( \langle\mathcal P\widetilde{\mathcal G}_iv,\widetilde{\mathcal G}_i^*v\rangle dt+\langle\mathcal P\widetilde{\mathcal G}_i\mathcal B,\widetilde{\mathcal G}_i^*v\rangle dt-\langle\mathcal B[\mathcal P(\mathcal G_i^h)^2w],v\rangle dt \right) \nonumber \\
    &+\sum_i\|\mathcal P\widetilde{\mathcal G}_iv+\mathcal P\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw]\|^2 dt \nonumber \\
    &-2 \sum_i\langle\widetilde{\mathcal G}_iv+\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle dW^i \nonumber\\
    &+ \sum_{i=1}^4 R_i \, dt + R_5 d\mathcal{W}_t.
\end{align}
We are now in a good setup to start estimating terms on the right hand side "group by group". We start with the nonlinear terms  i.e. the first line of \eqref{eq:EE_to_est}, and treat each term separately.\\

\noindent For the first term, we use \eqref{aux-est2} componentwise, with $f=v$ and $g=\partial w$, followed by Young's inequality ($\delta$ to be chosen later, and $\|w\|_{H^1}$ absorbed into the absolute constant due to the deterministic bound \eqref{Estimate1}):
\begin{equation}\label{eq:Bvw}
|\langle B(v,w),v\rangle| \le c \|w\|_{H^1}\|v\| \|\nabla^hv\|
\le\delta\|\nabla^hv\|^2+c_{\delta}\|v\|^2.
\end{equation}
The second term is treated by a horizontal H\"older with exponents $4,2,4$ followed by a vertical Cauchy-Schwarz and an application of \eqref{eq:ladyzh}:
\begin{equation}\label{eq:BBw}
\begin{aligned}
|\langle B(\mathcal B,w),v\rangle|
&\le c\|\mathcal M^h\|_{L_z^2}
\|w\|_{L_h^4}\|\nabla^hw\|_{L_h^2}\|v\|_{L_z^2L_h^4}\\
&\le c(\varepsilon\nu)^{1/4}(\|v\|\|\nabla^hv\|)^{\frac{1}{2}}\le c(\varepsilon\nu)^{1/4}(\|v\| + \|\nabla^hv\|)\\
&\le \delta \|\nabla^h v\|^2 + \|v\|^2 + c_\delta (\varepsilon \nu)^{\frac{1}{2}}.
\end{aligned}
\end{equation}
For the next term, we expand $B(u,\mathcal B)=B(v,\mathcal B)+B(w,\mathcal B)+B(\mathcal B,\mathcal B)$ and estimate each separately. Starting with $B(v,\mathcal B)$, we decompose it further to look at different horizontal and vertical parts separately: 
\begin{align*}
\langle B(v,\mathcal B),v\rangle
=&\langle v^h\cdot\nabla^h\mathcal B^h,v^h\rangle
+\langle v^h\cdot\nabla^h\mathcal B^3,v^3\rangle\\
&+\langle v^3\partial_3\mathcal B^h,v^h\rangle
+\langle v^3\partial_3\mathcal B^3,v^3\rangle
\end{align*}
For the first term, we apply \eqref{aux-est2}, as $\| \mathcal{M}^h\|_{L^\infty_z} \le c$, followed by Young's inequality:
\begin{align*}
    \left|\langle v^h\cdot\nabla^h\mathcal B^h,v^h\rangle \right| \le c \|w\|_{H^1} \| v \| \| \nabla^h v\| \le \delta \| \nabla^h v\|^2 + c_\delta \|v\|^2.
\end{align*}
We do the same for the next term, obtaining
\begin{align*}
    \left| \langle v^h\cdot\nabla^h\mathcal B^3,v^3\rangle\right| &\le c(\varepsilon\nu)^{1/2}\|w\|_{H^2}\|v\| \|\nabla^hv\| \le \delta\|\nabla^hv\|^2+
 c_\delta \varepsilon \nu \|w\|_{H^2}^2\|v\|^2.
\end{align*}
For the third term, we start again with a horizontal H\"older
\begin{equation*}
    \left| \langle v^3\partial_3\mathcal B^h,v^h\rangle \right| \le c \int_{[0,1]} \|v^3(\cdot,z)\|_{L^2_h} |\partial_3 \mathcal{M}^h| \|v^h(\cdot,z)\|_{L^4_h} dz.
\end{equation*}
Now, we inspect $\|v^3(\cdot,z)\|_{L^2_h}$ for $z\leq \frac{1}{2}$, the case $z\geq \frac{1}{2}$ is symmetric. Note that, due to the divergence free condition we have:
\begin{align*}
    \|v^3(\cdot,z)\|_{L^2_h} &= \left\Vert  \int_0^z \partial_3 v^3(\cdot, z') dz'  \right\Vert_{L^2_h} = \left\Vert  \int_0^z \operatorname{div_h} v^h(\cdot, z') dz'  \right\Vert_{L^2_h} \\
    &\le \int_0^z \left\Vert \nabla^h v(\cdot, z') \right\Vert_{L^2_h} dz'\le \sqrt{d(z)}\|\nabla^hv\|
\end{align*}
where in the second line we have used the Minkowski integral inequality, followed by Cauchy-Schwarz. Returning to the term at hand, we have applying Cauchy-Schwarz in $z$ twice:
\begin{align*}
    \left| \langle v^3\partial_3\mathcal B^h,v^h\rangle \right| &\le c \|\nabla^hv\|\int_{[0,1]} \sqrt{d(z)} |\partial_3 \mathcal{M}^h| \|v^h(\cdot,z)\|_{L^4_h} dz \\
    &\le c \|\nabla^hv\| \|v\|_{L^2_zL^4_h} \left( \int_0^1 d(z) |\partial_3\mathcal{M}^h|^2 dz \right)^{\frac{1}{2}}\\
    &\le c \|\nabla^hv\| \|v\|_{L^2_zL^4_h} (\|\partial_3\mathcal{M}^h\|_{L^2_z}\|d\partial_3\mathcal{M}^h\|_{L^2_z} )^{\frac{1}{2}} \\
    &\le c \|\nabla^hv\| \|v\|_{L^2_zL^4_h} \le\  c \|\nabla^hv\|^{\frac{3}{2}} \|v\|^{\frac{1}{2}} \le \delta \| \nabla^h v\|^2 + c_\delta \|v\|^2,
\end{align*}
where the last estimate follows by Young's inequality with exponents $4, \frac{4}{3}$.
\noindent
For the last term, we have from \eqref{aux-est2}:
\begin{equation*}
    \left| \langle v^3\partial_3\mathcal B^3,v^3\rangle \right| \le c \|v\|\|\nabla^h v\| \le \delta \| \nabla^h v\|^2 + c_\delta \|v\|^2,
\end{equation*}
as $\|\partial_3 \mathcal{M}^z\|_{L^\infty_z}\le c$ and $\|\operatorname{curl} w \|_{L^2_h}\le c$.
Collecting the above, we have obtained
\begin{equation}\label{eq:BvB}
    |\langle B(v,\mathcal B),v\rangle| \le 4\delta\|\nabla^h v\|^2 + c_\delta\|v\|^2 + c_\delta \varepsilon\nu \|w\|_{H^2}^2\|v\|^2.
\end{equation}
For \(B(w,\mathcal B)\), horizontal H\"older with exponents \(4,2,4\) followed by vertical Cauchy-Schwarz gives along with the corrector bounds \eqref{eq:matrix}:
\begin{equation*}
|\langle B(w,\mathcal B),v\rangle|
\le c(\varepsilon\nu)^{1/4}\|w\|_{H^2}
 \|v\|^\frac{1}{2}\|\nabla^hv\|^\frac{1}{2}.
\end{equation*}
Two applications of Young's inequality give
\begin{equation}\label{eq:BwB}
    |\langle B(w,\mathcal B),v\rangle| \le \delta \|\nabla^h v\|^2 + c_\delta(\varepsilon \nu)^{\frac{1}{2}}\|w\|_{H^2}^2 + c\|v\|^2.
\end{equation}
We finally move to the $B(\mathcal B,\mathcal B)$ term, which we decompose into $$\langle B(\mathcal B,\mathcal B), v\rangle = \langle \mathcal B^h \cdot \nabla^h \mathcal B, v\rangle + \langle \mathcal B^3 \partial_3 \mathcal B^h, v^h \rangle + \langle \mathcal B^3 \partial_3 \mathcal B^3, v^3 \rangle.$$
The first term of the three is bounded in exactly the same way as $\langle B(w,\mathcal B),v\rangle$ bounding $\mathcal{M}^h$ uniformly, so
$$|\langle \mathcal B^h \cdot \nabla^h \mathcal B, v\rangle| \le \delta \|\nabla^h v\|^2 + c_\delta(\varepsilon \nu)^{\frac{1}{2}}\|w\|_{H^2}^2 + c\|v\|^2.$$
For the remaining two terms, we have:
$$
\begin{aligned}
|\langle\mathcal B^3\partial_3\mathcal B^h,v^h\rangle|
&\le c\|\mathcal M^z\|_{L_z^\infty}
\|\partial_3\mathcal M^h\|_{L_z^2}
\|\operatorname{curl} w\|_{L_h^2}\|w\|_{L_h^4}\|v\|_{L_z^2L_h^4}\\
&\le c(\varepsilon\nu)^{1/4}(\|v\|+\|\nabla^hv\|),\\
|\langle\mathcal B^3\partial_3\mathcal B^3,v^3\rangle|
&\le\|\mathcal M^z\|_{L_z^2}
\|\partial_3\mathcal M^z\|_{L_z^\infty}
\|\operatorname{curl} w\|_{L_h^4}^2\|v\|\\
&\le c(\varepsilon\nu)^{1/4}\|w\|_{H^2}\|v\|,
\end{aligned}
$$
where we have used the $2$-d Ladyzhenskaya inequality in the last line. Applying Young's inequality as before and collecting the above we have:
\begin{equation}\label{eq:BBB}
    |\langle B(\mathcal B,\mathcal B),v\rangle| \le \delta \|\nabla^h v\|^2 + c_\delta(\varepsilon \nu)^{\frac{1}{2}}(\|w\|_{H^2}^2 +1)+ c\|v\|^2.
\end{equation}
Next we treat the last of the nonlinear terms, along with the $\mathcal B$ damping term. For this purpose, we will appeal to the crude bound $\|B(w,w)\|_{L^2_h} \le c \|w\|_{L^4_h} \|\nabla^hw\|_{L^4_h} \le c\|w\|_{H^2},$ and the estimate \eqref{eq:Bdual}:
\begin{align}\label{eq:Bnonlinear}
\left|\left\langle\mathcal B[B(w,w)]+
\sqrt{2\nu/\varepsilon}\,\mathcal B,v\right\rangle\right|
&\le c(\varepsilon\nu)^{1/4}(1+\|w\|_{H^2})
 (\|v\|+\|\nabla^hv\|) \nonumber\\
 & \le \delta \|\nabla^h v\|^2 + c_\delta(\varepsilon \nu)^{\frac{1}{2}}(\|w\|_{H^2}^2 +1)+ c\|v\|^2,
\end{align}
where the last line is just an application of Young's inequality.\\

\noindent We now deal with the remaining deterministic terms, in which we have to exploit the corrector cancellations \eqref{eq:profiles}.\\

\noindent By \eqref{eq:profiles}, the horizontal contributions of \(\mathcal B_1\) and
\(\mathcal B_2\) cancel exactly against rotation and
\(\sqrt{2\nu/\varepsilon}\,w\). The remaining horizontal terms are
\begin{align*}
    \left|\frac{2}{\varepsilon}\langle e_3 \wedge \mathcal B_3^h, v\rangle \right| + \left|\frac{2}{\varepsilon}\langle e_3 \wedge \mathcal B_4^h, v\rangle \right| &+ \left|\nu\langle A_z \mathcal B_4^h, v\rangle \right| 
    \le c\left( \varepsilon^{-1}\| \mathcal B_3[w]^h\| + \varepsilon^{-1}\| \mathcal B_4[w]^h\| + \nu \| \partial^2_3 \mathcal B_4[w]^h\| \right)\|v\|\\
    &\le c\left[
    \varepsilon^{-1}e^{-1/\sqrt{2\varepsilon\nu}}
    +\nu(\varepsilon\nu)^{-1/4}
    +\varepsilon^{-1}(\varepsilon\nu)^{3/4}
    +\nu \right]\|w\| \|v\|.
\end{align*}
We further note that due to $\frac{\nu}{\varepsilon}$ being bounded, we have $\varepsilon^{-1}e^{-1/\sqrt{2\varepsilon\nu}}
\le c(\varepsilon\nu)^{-1/2}
e^{-1/\sqrt{2\varepsilon\nu}}
\le c(\varepsilon\nu)^{1/4},$ as well as $\nu\le c(\varepsilon\nu)^{1/2}
\le c(\varepsilon\nu)^{1/4}$. Combining with the above, we have:
$$\left|\frac{2}{\varepsilon}\langle e_3 \wedge \mathcal B_3^h, v\rangle \right| + \left|\frac{2}{\varepsilon}\langle e_3 \wedge \mathcal B_4^h, v\rangle \right| + \left|\nu\langle A_z \mathcal B_4^h, v\rangle \right| \le c (\varepsilon \nu)^{\frac{1}{4}}\|v\|.$$
For the vertical component, integration by parts,
\eqref{eq:B3derivative}, and \(\partial_3v^3=-\operatorname{div_h} v^h\) give
\[
|\nu\langle\partial_3^2\mathcal B^3,v^3\rangle|
\le\nu\|\partial_3\mathcal B^3\|\|\partial_3v^3\|
\le c\nu(\varepsilon\nu)^{1/4}\|w\|_{H^1}\|\nabla^hv\|.
\]
We have thus proved
\begin{equation}\label{eq:Ekman-residual}
\begin{aligned}
&\left|\left\langle
\nu\mathcal P\partial_3^2\mathcal B
-\frac1\varepsilon\mathcal P(e_3\wedge\mathcal B)
+\sqrt{\frac{2\nu}{\varepsilon}}w,v
\right\rangle\right|\\
&\quad\le c(\varepsilon\nu)^{1/4}\|v\|
+c\nu(\varepsilon\nu)^{1/4}\|\nabla^hv\|\\
&\quad\le\delta\|\nabla^hv\|^2
+c\|v\|^2+c_{\delta}(\varepsilon\nu)^{1/2}.
\end{aligned}
\end{equation}
Collecting all the bounds on the deterministic terms (and making $\delta$ smaller by a factor of $100$), we see that they're bounded by an expression:
\begin{equation*}
\delta\|\nabla^hv\|^2+
 c_{\delta}(1+\|w\|_{H^2}^2)\|v\|^2+
 c_{\delta}(\varepsilon\nu)^{1/2}(1+\|w\|_{H^2}^2).
\end{equation*}
Thus, our energy inequality now takes the form:
\begin{align*}
    d\|v\|^2\le
    &-2 \|\nabla^hv\|^2 dt -2\nu \|\partial_3v\|^2 dt \\
    &+\sum_i \left( \langle\mathcal P\widetilde{\mathcal G}_iv,\widetilde{\mathcal G}_i^*v\rangle dt+\langle\mathcal P\widetilde{\mathcal G}_i\mathcal B,\widetilde{\mathcal G}_i^*v\rangle dt-\langle\mathcal B[\mathcal P(\mathcal G_i^h)^2w],v\rangle dt  \right) \\
    &+\sum_i\|\mathcal P\widetilde{\mathcal G}_iv+\mathcal P\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw]\|^2 dt  \\
    &-2 \sum_i\langle\widetilde{\mathcal G}_iv+\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle dW^i \nonumber\\
     &+(\delta\|\nabla^hv\|^2+
 c_{\delta}(1+\|w\|_{H^2}^2)\|v\|^2+
 c_{\delta}(\varepsilon\nu)^{1/2}(1+\|w\|_{H^2}^2)) dt\\
    &+ \sum_{i=1}^4 R_i \, dt + R_5 d\mathcal{W}_t.
\end{align*}
We now move to estimating the terms coming from the It\^o-Stratonovich corrector along with the quadratic variation terms.
Expanding the square and grouping terms, we have to estimate the following:
\begin{align*}
    &\sum_i\bigl(\langle\mathcal P\widetilde{\mathcal G}_iv,\widetilde{\mathcal G}_i^*v\rangle+
    \|\mathcal P\widetilde{\mathcal G}_iv\|^2\bigr)\nonumber \\
    &-\sum_i\langle\mathcal B[\mathcal P(\mathcal G_i^h)^2w],v\rangle\nonumber\\
    &+\sum_i\langle\mathcal P\widetilde{\mathcal G}_i\mathcal B,\widetilde{\mathcal G}_i^*v\rangle
    +2\sum_i\langle\mathcal P\widetilde{\mathcal G}_iv,
     \mathcal P\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw]\rangle
    +\sum_i\|\mathcal P\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw]\|^2.
\end{align*}
For the first term, we use \eqref{bigbound1NEW}:
$$\sum_i\bigl(\langle\mathcal P\widetilde{\mathcal G}_iv,\widetilde{\mathcal G}_i^*v\rangle+
    \|\mathcal P\widetilde{\mathcal G}_iv\|^2\bigr) \le \delta(\|\nabla^hv\|^2+\nu\|\partial_3v\|^2)+c_\delta\|v\|^2,$$ as $\nu^{2\alpha}\le \nu$.

\noindent For the second term, we use the bound \eqref{eq:Bdual}, along with the standard estimates on $\widetilde{\mathcal G}$, to get
$$\begin{aligned}
\sum_i|\langle\mathcal B[\mathcal P(\mathcal G_i^h)^2w],v\rangle|
&\le \sum_i c(\varepsilon\nu)^{1/4}\|\mathcal P(\mathcal G_i^h)^2w\|_{L^2_h}
 (\|v\|+\|\nabla^hv\|)\\
&\le c(\varepsilon\nu)^{1/4}\|w\|_{H^2}
 (\|v\|+\|\nabla^hv\|)\\
&\le\delta\|\nabla^hv\|^2+c_\delta\|v\|^2
 +c_\delta(\varepsilon\nu)^{1/2}\|w\|_{H^2}^2.
\end{aligned}$$
Finally for the last line, we use the bounds \eqref{eq:Bnorm} along with Lemma \ref{bounds on split noise} to get:
\begin{align}
   \label{halfstuff} \|\mathcal P\widetilde{\mathcal G}_i\mathcal B\| 
    &\le c\|\xi_i\|_{W^{1,\infty}}
    \bigl((\varepsilon\nu)^{1/4}\|w\|_{H^2}
    +\nu^\alpha(\varepsilon\nu)^{-1/4}\|w\|_{H^1}\bigr)\\ \nonumber
    &\le c\|\xi_i\|_{W^{1,\infty}}
    \bigl((\varepsilon\nu)^{1/4}\|w\|_{H^2}
    +\nu^{\alpha-\frac{1}{2}}\bigr)\\ \nonumber
                                    
                                             
                 
    \|\mathcal B[\mathcal P\mathcal G_i^hw]\| 
    &\le c\|\xi_i\|_{W^{1,\infty}}(\varepsilon\nu)^{1/4}\|w\|_{H^2}.
\end{align}
For the first two terms in the last line, we apply Cauchy-Schwarz to bound those terms by
$$c\bigl((\varepsilon\nu)^{1/4}\|w\|_{H^2}
+\nu^{\alpha-\frac{1}{2}}\bigr)
\bigl(\|\nabla^hv\|+\nu^\alpha\|\partial_3v\|+\|v\|\bigr) \le \delta(\|\nabla^hv\|^2 +\nu\|\partial_3v\|^2) +c_\delta(\|v\|^2
 +(\varepsilon\nu)^{1/2}\|w\|_{H^2}^2 + \nu^{2\alpha-1}).$$
Finally, the last term is handled similarly, and bounded by
\begin{equation*}
\begin{aligned}
\sum_i\bigl(\|\widetilde{\mathcal G}_i\mathcal B\|^2+\|\mathcal B[\mathcal P\mathcal G_i^hw]\|^2\bigr)
                                                 
                                                            
&\le c\bigl((\varepsilon\nu)^{1/2}\|w\|_{H^2}^2
+\nu^{2\alpha-1}\bigr).
\end{aligned}
\end{equation*}
Collecting all the terms, we get that the It\^o-Stratonovich correction terms and quadratic variation terms are bounded by 
\begin{equation}\label{eq:noise-total}
\begin{aligned}
&\delta(\|\nabla^hv\|^2+\nu\|\partial_3v\|^2)+c_\delta\|v\|^2+c_\delta\bigl((\varepsilon\nu)^{1/2}\|w\|_{H^2}^2
+\nu^{2\alpha-1}\bigr)
\end{aligned}
\end{equation}
up to again changing $\delta$ by a factor. Updating our energy inequality, we now have
\begin{align}\label{eq:EE-last}
    d\|v\|^2\le
    &-2 \|\nabla^hv\|^2 dt\nonumber -2\nu \|\partial_3v\|^2 dt\\
    &-2 \sum_i\langle\widetilde{\mathcal G}_iv+\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle dW^i \nonumber\\
     &+(\delta(\|\nabla^hv\|^2 + \nu\|\partial_3 v\|^2)+
 c_{\delta}(1+\|w\|_{H^2}^2)\|v\|^2+
 c_{\delta}(\varepsilon\nu)^{1/2}(1+\|w\|_{H^2}^2) + c_\delta \nu^{2\alpha -1})dt \nonumber\\
    &+ \sum_{i=1}^4 R_i \, dt + R_5 d\mathcal{W}_t.
\end{align}
We therefore move to bounding the stochastic integral using the BDG inequality. Anticipating the use of stochastic Gr\"onwall, we carry out the estimate between two arbitrary stopping times $0\le \theta \le \theta' \le T.$ Using the estimates appearing in the step above to control the quadratic variation term, along with \eqref{bigbound2NEW}, we have
\begin{equation*}
    \sum_i|\langle\widetilde{\mathcal G}_iv+\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle|^2
    \le c\|v\|^4+
    c\bigl((\varepsilon\nu)^{1/2}\|w\|_{H^2}^2
    +\nu^{2\alpha - 1}\bigr)\|v\|^2.
\end{equation*}
The BDG inequality therefore gives
\begin{align*}
&\mathbb E\left[ \sup_{\theta\le r\le\theta'}\left|
2\sum_i\int_\theta^r
\langle\widetilde{\mathcal G}_iv+\widetilde{\mathcal G}_i\mathcal B-\mathcal B[\mathcal P\mathcal G_i^hw],v\rangle\,dW_s^{i}
\right| \right]\nonumber\\
&\quad\le c\mathbb E\left[\left(\sup_{\theta\le r\le\theta'}\|v_r\|^2\right)^{1/2}
 \left\{\int_\theta^{\theta'}\|v_s\|^2\,ds
 +c\bigl((\varepsilon\nu)^{1/2}
 +\nu^{2\alpha -1}\bigr)\right\}^{1/2}\right]
\nonumber\\
&\quad\le \delta \mathbb E\left[ \sup_{\theta\le r\le\theta'}\|v_r\|^2 \right]
 +c_\delta\mathbb E\left[ \int_\theta^{\theta'}\|v_s\|^2\,ds \right]
 +c_{\delta}\bigl((\varepsilon\nu)^{1/2}
 +\nu^{2\alpha-1}\bigr).
\end{align*}
Taking the supremum over $[\theta,\theta']$ and expectations in the inequality \eqref{eq:EE-last}, and taking $\delta\le \frac{1}{2}$ we have
\begin{align} \label{eq:EE-pre-G}
     \mathbb E\left[ \sup_{\theta\le r\le\theta'}\|v_r\|^2\right] &+ \mathbb E \left[ \int_\theta^{\theta'} (\|\nabla^hv_s\|^2 + \nu\|\partial_3 v_s\|^2) \, ds \right] \le \nonumber\\ 
     &\le c\mathbb E \left[ \|v_\theta\|^2 \right] + c\mathbb E \left[\sum_{i=1}^4 \int_0^T |R_i|\, ds \right]+ c\mathbb E \left[ \sup_{s \in [0,T]} \left| \int_0^sR_5 d \mathcal
     {W}_s\right|\right] + c_\delta((\varepsilon \nu)^{\frac{1}{2}} + \nu^{2\alpha-1})\nonumber\\
     &+ c_\delta \mathbb E\left[ \int_\theta^{\theta'}(1+\|w_s\|^2_{H^2})\|v_s\|^2\,ds\right],
\end{align}
where we have bounded $\mathbb E \int_\theta^{\theta'} \|w_s\|^2_{H^2} \, ds \le \mathbb E \int_0^T  \|w_s\|^2_{H^2} \, ds \le c.$
Let us now deal with the $R_i$ terms. We keep in mind, that we want those contributions to vanish as $n\to \infty$ while keeping $k,m$ fixed. 
\begin{align*}
    |R_1| \le c |\langle B(u,u),(I-\mathcal P_n)(w+\mathcal B[w])\rangle| &\le c \|u\|_{L^3}\|\nabla u\|
             \|(I-\mathcal P_n)(w+\mathcal B[w])\|_{L^6}\\
    &\le c\|u\|^{1/2}\|u\|_{H^1}^{3/2}
             \|(I-\mathcal P_n)(w+\mathcal B[w])\|_{H^1},
\end{align*}
where we used interpolation of \(L^3\) between \(L^2\) and \(L^6\), and a Sobolev embedding. We further note that
\begin{align*}
    \int_0^{T}\|u_n\|^{1/2}\|u_n\|_{H^1}^{3/2}\,ds
    &\le T^{1/4}(\sup_{s\le T}\|u_n(s)\|^2)^{1/4}
    \left(\int_0^T\|u_n\|_{H^1}^2\,ds\right)^{3/4}\\
    &\le c_T\left(\sup_{s\le T}\|u_n(s)\|^2+
    \int_0^T\|u_n\|_{H^1}^2\,ds\right),
\end{align*}
where we applied H\"older's inequality in the first line, and Young's in the second. Altogether, applying Cauchy-Schwarz and exploiting the additional probabilistic $L^4$ bound from \ref{item2.5}, we have 
\begin{equation}\label{eq:R1}
    \mathbb E \left[ \int_0^T |R_1|\,ds \right] \le c_k \left(\mathbb E\left[ \sup_{s\le T}
\|(I-\mathcal P_n)(w_s+\mathcal B[w_s])\|_{H^1}^2 \right]\right)^{1/2}.
\end{equation}
The terms $R_2$ and $R_3$ are bounded using simpler combination of the above tools, so we skip the details and simply give
\begin{equation}\label{eq:R2}
     \mathbb E \left[ \int_0^T |R_2|\,ds \right]  \le c \varepsilon^{-1}  \left(\mathbb E \left[ \sup_{s\le T}
\|(I-\mathcal P_n)(w_s+\mathcal B[w_s])\|_{L^2}^2\right]\right)^{1/2},
\end{equation}
as well as
\begin{equation}\label{eq:R3}
     \mathbb E \left[ \int_0^T |R_3|\,ds \right]  \le c \left(\mathbb E\left[ \sup_{s\le T}
\|(I-\mathcal P_n)(w_s+\mathcal B[w_s])\|_{H^1}^2 \right] \right)^{1/2}.
\end{equation}
We keep 
\begin{equation}
     \mathbb E \left[ \int_0^T |R_4|\,ds \right]= \mathbb E \left[ \int_0^T \sum_i \|(I-\mathcal P_n)(\mathcal P\mathcal G_i^hw_s+\mathcal B[\mathcal P\mathcal G_i^hw_s])\|^2 ds\right] .
\end{equation}
For the final term we apply the BDG inequality using the bound $$\sum_i|\langle\widetilde{\mathcal G}_iu,(I-\mathcal P_n)(w+\mathcal B[w])\rangle|^2
\le c\|u\|^2\|(I-\mathcal P_n)(w+\mathcal B[w])\|_{H^1}^2.$$
We have:
\begin{align} \label{eq:R5}
    \mathbb E \left[ \sup_{s\in [0,T]} \left| \int_0^s R_5 d\mathcal{W}_s \right| \right] &\le c \mathbb E \left[ \left( \int_0^T\|u_s\|^2\|(I-\mathcal P_n)(w_s+\mathcal B[w_s])\|_{H^1}^2ds \right)^{\frac{1}{2}}\right] \nonumber\\
    & \le c \left(\mathbb E\left[ \sup_{s\le T}
    \|(I-\mathcal P_n)(w_s+\mathcal B[w_s])\|_{H^1}^2 \right]\right)^{1/2}.
\end{align}
Collecting \eqref{eq:R1}-\eqref{eq:R5}, we arrive at
\begin{align*}
    \mathbb E \left[ \sum_{i=1}^4 \int_0^T |R_i|\, ds \right]+ \mathbb E \left[ \sup_{s \in [0,T]} \left| \int_0^sR_5 d \mathcal
     {W}_s\right|\right] &\le c_k \left(\mathbb E \left[ \sup_{s\le T}
    \|(I-\mathcal P_n)(w_s+\mathcal B[w_s])\|_{H^1}^2\right]\right)^{1/2} +\\
    &+ \mathbb E \left[ \int_0^T \sum_i \|(I-\mathcal P_n)(\mathcal P\mathcal G_i^hw_s+\mathcal B[\mathcal P\mathcal G_i^hw_s])\|^2 ds\right].
\end{align*}
We will now argue that the right hand side goes to zero, as $n \to \infty$, for $k,m$ fixed. To this end, we note that $I-\mathcal P_n$ converges strongly (in the operator sense) to $0$ over both $L^2_\sigma$ and $H^1$. We now note, that for a fixed function, $f\in C([0,T]; H^2)$, we have that the image of $[0,T]$ under $t\to f_t + \mathcal B[f_t]$ is compact in $H^1$, and its image under $t \to  \mathcal P\mathcal G_i^hf_t+\mathcal B[\mathcal P\mathcal G_i^hf_t]$ is compact in $ L^2_\sigma$ for each $i$. Thus, as pointwise convergence is uniform on compact sets, we have 
$$\sup_{t\le T}\|(I-\mathcal P_n)(f_t+\mathcal B[f_t])\|_{H^1}^2
\to 0, \text{ as $n \to \infty$},$$
and 
$$\sup_{t\le T} \sum_i \|(I-\mathcal P_n)( \mathcal P\mathcal G_i^hf_t+\mathcal B[\mathcal P\mathcal G_i^hf_t])\|_{L^2}^2
\to 0, \text{ as $n \to \infty$},$$
where taking the sum over $i$ is allowed, thanks to the assumptions on the $\xi_i$'s.
We further note, that using the corrector bounds \eqref{eq:Bnorm}
we have 
\begin{align*}
    \sup_{t\le T}\|(I-\mathcal P_n)(f_t+\mathcal B[f_t])\|_{H^1}^2 & \le c_{\varepsilon,\nu} \sup_{t\in[0,T]}\| f_t\|_{H^2}^2, \\
    \sup_{t\le T}\sum_i \|(I-\mathcal P_n)( \mathcal P\mathcal G_i^hf_t+\mathcal B[\mathcal P\mathcal G_i^hf_t])\|_{L^2}^2& \le c_{\varepsilon,\nu} \sup_{t\in[0,T]}\| f_t\|_{H^2}^2.
\end{align*}
We finally note, that the law of $w=w^{k,m,n}$ does not depend on $n$, and hence we can take the expectation with respect to that law (in particular by replacing $f$ by $w^{k,m}$), and pass the $n \to \infty$ limit through the expectation by the DCT. This application of the DCT is permitted, due to the additional regularity in the initial data, in particular for a fixed $k,m$, lemma \ref{uniform bounds on w} shows that $\mathbb E\left[ \sup_{t\in[0,T]}\| w^{k,m}_t\|_{H^2}^2\right] < \infty$. We summarise the above in writing: 
\begin{equation}
    \mathbb E\left[ \sum_{i=1}^4 \int_0^T |R_i|\, ds \right]+ \mathbb E \left[ \sup_{s \in [0,T]} \left| \int_0^sR_5 d \mathcal
     {W}_s\right| \right]\le o_n^{k,m},
\end{equation}
where $o_n^{k,m} \to 0$ as $n\to \infty$ for fixed $k,m$.

\noindent Coming back to \eqref{eq:EE-pre-G}, we have:
\begin{align} \label{eq:EE-Gready}
     \mathbb E\left[ \sup_{\theta\le r\le\theta'}\|v_r\|^2\right] + \mathbb E \left[ \int_\theta^{\theta'} (\|\nabla^hv_s\|^2 + \nu\|\partial_3 v_s\|^2) \, ds\right] &\le \nonumber c\mathbb E \left[ \|v_\theta\|^2\right] + o_n^{k,m} + c_\delta((\varepsilon \nu)^{\frac{1}{2}} + \nu^{2\alpha-1})\nonumber\\
     &+ c_\delta \mathbb E \left[ \int_\theta^{\theta'}(1+\|w_s\|^2_{H^2})\|v_s\|^2\,ds\right] .
\end{align}
Seeing that $$\int_0^T(1+\|w_s\|_{H^2}^2)\,ds\le c$$ with a deterministic constant, we apply the Stochastic Gr\"onwall lemma to obtain
\begin{equation}\label{eq:gronny-bound}
     \mathbb E\left[ \sup_{r\in [0,T]}\|v_r\|^2\right] + \mathbb E \left[ \int_0^{T} (\|\nabla^hv_s\|^2 + \nu\|\partial_3 v_s\|^2) \, ds \right]\le c_{\delta}\left( \mathbb E \left[ \|v_0\|^2\right] + o_n^{k,m} + c_\delta((\varepsilon \nu)^{\frac{1}{2}} + \nu^{2\alpha-1}) \right).
\end{equation}
Coming back to the beginning of the proof, denoting $v^{k,m,n} := u^{k,n} - w^{k,m,n} - \mathcal B[w^{k,m,n}]$, we have
\begin{align*}
    \lim_{k 
       \rightarrow \infty} &\mathbbm{E}\left[\left\Vert u^k - w \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2 \right] 
       \leq 4\lim_{k 
       \rightarrow \infty}\lim_{m \rightarrow \infty}\liminf_{n \rightarrow \infty}\mathbbm{E}\left[\left\Vert u^{k,n} - w^{k,m,n}  \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right] \\
       &\le c\lim_{k 
       \rightarrow \infty}\lim_{m \rightarrow \infty}\liminf_{n \rightarrow \infty}\mathbbm{E}\left[\left\Vert v^{k,m,n} \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2 + \left\Vert \mathcal B[w^{k,m,n}] \right\Vert_{L^\infty([0,T];L^2_{\sigma})}^2\right] \\
       &\le c_\delta \lim_{k 
       \rightarrow \infty}\lim_{m \rightarrow \infty}\liminf_{n \rightarrow \infty} \left( \mathbb E \left[ \|u_0^{n}-w_0^{m} - \mathcal B[w_0^{m}]\|^2 \right] + o_n^{k,m} +(\varepsilon_k \nu_k)^{\frac{1}{2}} + \nu_k^{2\alpha-1} \right) \\
       &+  \lim_{k 
       \rightarrow \infty}\lim_{m \rightarrow \infty}\liminf_{n \rightarrow \infty} (\varepsilon_k \nu_k)^{\frac{1}{2}}\mathbb E\left[  \|w^{k,m,n}\|_{L^\infty([0,T];H^1)}^2\right]\\
       &\le c_\delta \lim_{k 
       \rightarrow \infty}\lim_{m \rightarrow \infty}\left( \mathbb E \left[ \|u_0-w_0^{m} - \mathcal B[w_0^{m}]\|^2 \right] +(\varepsilon_k \nu_k)^{\frac{1}{2}} + \nu_k^{2\alpha-1} \right) \\
       &\le  c_\delta \lim_{k 
       \rightarrow \infty}\lim_{m \rightarrow \infty}\left( \mathbb E \left[ \|u_0-w_0^{m}\|^2 \right] +(\varepsilon_k \nu_k)^{\frac{1}{2}} + \nu_k^{2\alpha-1} \right) \\
       &\le  c_\delta \lim_{k 
       \rightarrow \infty}\left((\varepsilon_k \nu_k)^{\frac{1}{2}} + \nu_k^{2\alpha-1} \right) =0,
\end{align*}
where we have used the energy estimate \eqref{eq:gronny-bound} in line three and the bound \eqref{eq:Bnorm} in line four and six.
