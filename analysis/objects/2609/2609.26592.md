\label{thm:main-curvature}
    Let $n\geq 2, m\geq 1$ be integers and let $\Lambda>0$. 
    Suppose $M$ is an $n$-dimensional area-minimizing current ${\rm mod}(2)$ in $B_8\subset {\mathbb R}^{n+m}$ with $\partial M\mathbin{\vrule height 1.6ex depth 0pt width
0.13ex\vrule height 0.13ex depth 0pt width 1.3ex} B_8 =0$ and $|M|(B_8) \leq \Lambda$. Then there exists a constant $C=C(m,n,\Lambda)>0$ such that 
    \begin{equation}\label{eq:main-L2-bound}
        \int_{B_1} |A_M|^2 d|M| \leq C(n,m,\Lambda).
    \end{equation}
    Here $A_M$ is the second fundamental form of $\operatorname{reg}(M)$ extended by zero across the singular set.

<!-- prop-1: body_md -->
\label{prop:monotonicity}
    Let $M$ be an integral stationary $n$-varifold in $B_4$. Then, for every $B_r(x)\subset B_4$ we have 
    \begin{equation}
        r\frac{d}{dr}\theta_r(x) = \frac{2}{\omega_n r^{n+2}}\int_{B_r(x)} \left| \pi_{T_yM}^\perp(y-x)\right|^2d|M|(y) 
    \end{equation}
    In particular, we have for every $1\leq \lambda<2$,
    \begin{equation} \label{eq:monotonicity-pinching}
        \theta_{2r}(x)-\theta_r(x) \geq \frac{C(n,\lambda)}{ r^{n+2}} \int_{B_{\lambda r}(x) }|\pi_{T_yM}^\perp(y-x)|^2d|M|(y)
    \end{equation}
<!-- end prop-1 -->
<!-- proof-1: body_md -->
The proof is a straightforward modification of the usual monotonicity formula by testing with the vector field $X_{x,r}(y) = \left( 1- \frac{|y-x|^2}{r^2}\right)_+ (y-x)$.
    We may then integrate over $s\in [\lambda r, 2r]$, restrict the inner integral to $B_{\lambda r}(x)$, and integrate in $s$ to obtain \ref{eq:monotonicity-pinching}.

<!-- prop-2: body_md -->
\label{prop:symmetry-eps-regularity}
    There exists $\epsilon_{reg}=\epsilon_{reg}(n,m,\Lambda)>0$ such that the following holds. Suppose $M$ is a ${\rm mod}(2)$ $n$-dimensional area-minimizing current in $B_4$ with $\partial M\mathbin{\vrule height 1.6ex depth 0pt width
0.13ex\vrule height 0.13ex depth 0pt width 1.3ex} B_4 =0$ and $|M|(B_4)\leq \Lambda$. If $M$ is $(n-1,\epsilon_{reg})$-symmetric in $B_4$, then  $M$ is smooth in $B_{63/64}$ and 
    \begin{equation}
        |A_M|\leq c(m,n,\Lambda) \quad \text{in }B_{63/64}
    \end{equation}
    for some $c(m,n,\Lambda)>0$.
<!-- end prop-2 -->
<!-- proof-2: body_md -->
The proof is by contradiction. 
    If no such $\epsilon_{reg}$ existed, we could find a sequence of $n$-dimensional ${\rm mod}(2)$ area-minimizing currents $M_i$ which are $(n-1,\epsilon_i)$-symmetric with respect to $(n-1)$-planes $L_i$ in $B_4$ with $\epsilon_i\downarrow 0$. 
    Taking a subsequence if necessary and passing to the limit, we see that there exists a ${\rm mod}(2)$ area-minimizing current $M$ in $B_4$ with $\partial M \mathbin{\vrule height 1.6ex depth 0pt width
0.13ex\vrule height 0.13ex depth 0pt width 1.3ex} B_4=0$ such that $M_i\to M$ as currents and as varifolds. Moreover by compactness of the Grassmannian we may assume $L_i\to L$ for some $(n-1)$-plane $L$.
    Passing the $(n-1,\epsilon_i)$-symmetry to the limit shows that $M$ satisfies the hypotheses of Lemma~\ref{lem:(k,0)-symmetry-cone} and hence $M$ is a cone with spine containing $L$. Thus $M$ splits as a product $M = \llbracket L\rrbracket \times M_0$ with $M_0$ a $1$-dimensional ${\rm mod}(2)$ area-minimizing cone in $L^\perp$. 
    But any such $M_0$ must be a multiplicity 1 line, so that $M$ is  an $n$-plane with multiplicity 1. 
    Since $M_i\to M$ we may apply Allard's regularity theorem and elliptic regularity to reach a contradiction for large enough $i$.

\label{def:annular-region}
    Let $M$ be an area-minimizing current ${\rm mod}(2)$ in $B_{4r_a}(x_a)$ with $\partial M \mathbin{\vrule height 1.6ex depth 0pt width
0.13ex\vrule height 0.13ex depth 0pt width 1.3ex} B_{4r_a}(x_a)=0$ and $|M|(B_{4r_a}(x_a))\leq \Lambda$. Let $T^{n-2}\subset B_{4r_a}(x_a)$ be an $(n-2)$-dimensional topological submanifold of ${\mathbb R}^{n+m}$, called the approximate singular set.
    Let $r_x:T^{n-2}\to {\mathbb R}_{\geq 0}$ be a nonnegative function satisfying $\operatorname{Lip}(r_x)\leq 2$, called the symmetry scale.
    We call $\mathcal A:=B_{r_a}(x_a)\setminus \overline{B}_{r_x}(T^{n-2})$ a $\delta$-annular region if the following hold:
    \begin{itemize}
        \item [(a1)] $T^{n-2}$ is $\delta$-graphical on $B_{r_x}(x)$ for all $x\in T^{n-2}$;
        \item [(a2)] For all $x\in T^{n-2}$ and $r>r_x$ such that $B_{r}(x)\subset B_{r_a}(x_a)$, there exists a $(n-2)$-dimensional affine subspace $L^{n-2}_{x,r}$ such that
        \begin{equation}
            d_{H}(L_{x,r}^{n-2}\cap B_r(x),T^{n-2}\cap B_r(x))<\delta r.
        \end{equation}
        \item [(a3)] For all $x\in T^{n-2}$ and $r>r_x$ such that $B_r(x)\subset B_{r_a}(x_a)$, we have that $M$ is $(n-2,\delta)$-symmetric on $B_r(x)$, but not $(n-1,\epsilon')$-symmetric on $B_r(x)$, where $\epsilon' = \epsilon'(n,m,\epsilon_{reg}) = \epsilon'(n,m,\Lambda)$ is the constant in Theorem~\ref{A:thm:annular-decomp} when we take $\epsilon=\epsilon_{reg}$.
        \item [(a4)] For each $x\in T^{n-2}$, we have 
        \begin{equation}
            |\theta_{2r_a}(x)-\theta_{r_x}(x)|<\delta.
        \end{equation}
    \end{itemize}

\label{lem:pointwise-curvature}
 Let $P^n$ be an $n$-dimensional linear subspace of ${\mathbb R}^{n+m}$ and $P^\perp$ its orthogonal complement. 
   Let $w_0,w_1,\ldots, w_{n-2}\in {\mathbb R}^{n+m}$ be in $\frac{1}{2}$-general position in $B_{r}\subset {\mathbb R}^{n+m}$
   and $L:= \operatorname{span}\{w_{n-2}-w_0,\ldots, w_1-w_0\}$. 
   Let $L^\perp$ be the orthogonal complement for $L$ in ${\mathbb R}^{n+m}$. 
   Let $H= \pi_{P}(L)$ and $V = L^\perp \cap P = H^{\perp_P}$ be the orthogonal complement of $H$ in $P$. 
   Finally, let $A\in \mathrm{Sym}^2(P^*)\otimes P^\perp$ be a symmetric bilinear form. 
   Suppose $\|\pi_{P^\perp}|_L\| <\frac{1}{2}$. Then, 
   \begin{itemize}
       \item [(i)] $H$ is $(n-2)$-dimensional, $V$ is $2$-dimensional, and $H\oplus V  = P$.  Moreover, $\{\pi_{P}(w_i-w_0)\}_{i=1}^{n-2}$ form a basis for $H$.
       \item [(ii)] For any orthonormal basis $\{h_i\}_{i=1}^{n-2}$ for $H$, we have 
       \begin{equation}
           \sum_{i=1}^{n-2} |A(\cdot, h_i)|^2  \leq \frac{C(n)}{r^{2}} \left( \sum_{i=1}^{n-2} |A(\cdot, \pi_P(w_i-w_0))|^2 \right)
       \end{equation}
        
       
       \item [(iii)] If moreover $\Tr_{P} A =0$, then for any orthonormal basis $\{h_i\}_{i=1}^{n-2}$ for $H$ and any unit vector $e\in V$,
       \begin{equation}
           |A|^2 \leq n \left(\sum_{i=1}^{n-2} |A(\cdot, h_i)|^2 + |A(\cdot, e)|^2\right).
       \end{equation}
       Consequently,
       \begin{equation}
           |A|^2 \leq C(n) \left(r^{-2} \sum_{i=1}^{n-2} |A(\cdot, \pi_P(w_i-w_0))|^2 + |A(\cdot, e)|^2 \right).
       \end{equation}
   \end{itemize}
<!-- end lem-2 -->
<!-- proof-4: body_md -->
We verify (i) and (ii) as  an elementary application of linear algebra. For $n=2$ there is nothing to prove so we assume $n\geq 3$.
 Set $v_i:=w_{i}-w_0$ for $i=1,\ldots, n-2$. Since  $\|\pi_{P^\perp}|_L\|<\frac{1}{2}$, for every $v\in L$ we have 
 \begin{equation}
     |\pi_P v|^2 = |v|^2 - |\pi_{P^\perp} v|^2 \geq (1-\frac{1}{4})|v|^2 .
 \end{equation}
 Thus the restriction $\pi_P|_{L}$ is injective and $\dim H = \dim \pi_P(L) = \dim L = n-2$ and $\pi_P v_1,\ldots, \pi_P v_{n-2}$ form a basis for $H$. 
 For $z\in P$, we know that $z$ is orthogonal to $H$ if and only if $\langle z,\pi_P v\rangle =0$ for all $v\in L$, which is equivalent to $\langle z,v\rangle =0$ for all $v\in L$. Thus $V= H^{\perp_P} = L^\perp \cap P$, which proves (i).

For (ii),  let $W$ be the matrix with columns equal to the vectors $v_1,\ldots, v_{n-2}$. 
Since $|v_i|<2r$ and $\{w_i\}$ are in $\frac{1}{2}$-general position, the smallest singular value $\sigma(W)$ satisfies $\sigma(W) >c(n) r$ for some constant $c(n)$.
 Thus for any vector $v\in {\mathbb R}^{n-2}$,
\begin{equation*}
    | \pi_P W v|\geq \frac{3}{4}|Wv|\geq c(n) r |v|
\end{equation*}
and we have the estimate $\sigma(\pi_PW)\geq c(n) r$ for the smallest singular value.
For any orthonormal basis $h_1,\ldots, h_{n-2}$ of $H$, we may view $A$ as a linear map $ \overline{A}:H\to \operatorname{Hom}(P,P^\perp)$ defined by $H\ni h \mapsto A(\cdot, h)$. Viewed in this way, $|\overline{A}|^2 = \sum_{i=1}^{n-2} |A(\cdot, h_i)|^2$. Since $| \overline{A} \pi_P W|^2 = \sum_{i=1}^{n-2} |A(\cdot, \pi_P v_i)|^2$, we estimate 
\begin{equation*}
    \sum_{i=1}^{n-2} |A(\cdot, \pi_P v_i)|^2\geq \sigma(\pi_P W)^2 |\overline{A}|^2 \geq c(n) r^2\sum_{i=1}^{n-2}|A(\cdot, h_i)|^2
\end{equation*}
as desired. 

We now prove (iii). Let $f\in V$ be a unit vector perpendicular to $e$, so that $h_1,\ldots, h_{n-2}, e, f$ form an orthonormal basis of $P$. For simplicity set $h_{n-1} := e$. By trace-freeness,
\begin{equation*}
    A(f,f) = - \sum_{i=1}^{n-1} A(h_i, h_i).
\end{equation*}
Together with symmetry we obtain
\begin{equation*}
    |A(\cdot ,f)|^2 \leq (n-1) \sum_{i=1}^{n-1}\left( |A(f, h_i)|^2 + |A(h_i,h_i)|^2\right) \leq (n-1)\sum_{i=1}^{n-1} |A(\cdot, h_i)|^2.
\end{equation*}
Thus,
\begin{equation*}
    |A|^2 \leq \sum_{i=1}^{n-1} |A(\cdot, h_i)|^2 + |A(\cdot, f)|^2 \leq n\sum_{i=1}^{n-1} |A(\cdot, h_i)|^2.
\end{equation*}
This finishes the proof.

\label{lem:jacobi-radial-trans}
      Let $M^n\subset {\mathbb R}^{n+m}$ be an $n$-dimensional minimal immersion with no boundary in an open set $U\subset {\mathbb R}^{n+m}$. 
     Let $Y$ be a vector field tangent to $M$.
     Fix $x_0\in {\mathbb R}^{n+m}$ and consider the vector field $X(x)=x-x_0$ and let $Z(x):= \pi_{T_xM}^\perp (x-x_0)$ be the projection onto the normal space of $M$. Then $\mathcal{J}_M Z=0$ and
      \begin{equation}
          \nabla^\perp_{M, Y} Z = -A_M(Y,\pi_{T_xM}(x-x_0)) 
      \end{equation}
<!-- end lem-3 -->
<!-- proof-5: body_md -->
The flow generated by $X$ is a dilation centered at $x_0$, which preserves minimality. It follows that the normal variation field $Z$ is Jacobi. 
    Taking normal components in $\nabla_{{\mathbb R}^{n+m}, Y} X = Y$ gives
    \begin{equation*}
        0 = A_M(Y, \pi_{T_xM}(x-x_0)) + \nabla_{M,Y}^\perp Z
    \end{equation*}
    as desired.

<!-- lem-4: body_md -->
\label{lem:jacobi-cacc}
    Let $Z$ be a normal Jacobi field on a smooth minimal immersion $M$ and $\phi$ a compactly supported Lipschitz function with $0\leq \phi\leq 1$. Suppose $M$ has no boundary in an open set containing $\operatorname{spt} \phi$.
    Then,
    \begin{equation}
        \int_M \phi^2 |\nabla_M^\perp Z|^2 \leq 4\int_M |\nabla_M \phi|^2 |Z|^2 +2 \int_M \phi^2 |A_M|^2 |Z|^2.
    \end{equation}
    In particular if $|A_M|\leq Ks^{-1}$ on $\operatorname{spt} \phi$ and $|\nabla \phi|\leq C s^{-1}$, then 
    \begin{equation}
        \int_M \phi^2 |\nabla_M^\perp Z|^2 \leq C(1+K^2)s^{-2}\int_{M\cap \operatorname{spt} \phi}|Z|^2.
    \end{equation}
<!-- end lem-4 -->
<!-- proof-6: body_md -->
Testing the Jacobi equation with $\phi^2 Z$ and integrating by parts, we obtain
    \begin{equation*}
        \int_M \phi^2 |\nabla_M^\perp Z|^2  = -2\int_M \phi  \langle \nabla_{M,\nabla_M \phi}^\perp Z,Z\rangle  + \int_M \phi^2 \langle \Tilde{A}_M(Z),  Z\rangle.
    \end{equation*}
    Since $|\langle \Tilde{A}_M(Z),  Z\rangle|\leq |A_M|^2 |Z|^2$, this yields
    \[
    \int_M \phi^2 |\nabla_M^\perp Z|^2 \leq 2 \int_M |\phi| |\nabla_M \phi| |\nabla_M^\perp Z| |Z| + \int_M \phi^2 |A_M|^2 |Z|^2.
    \]
    Applying Young's inequality and rearranging, we obtain the desired inequality.

\label{lem:local-annular-geometry}
    Let $M,\mathcal A,T^{n-2}, y,z,r$ be as in Proposition \ref{prop:local-annular-curvature}.
    There exists  $C=C(m,n,\Lambda)>0$ such that for every $\eta\in (0,1)$, if $\delta\leq \delta(n,m,\Lambda,\eta)$ then there exist points $w_0,\ldots, w_{n-2}\in T^{n-2}\cap B_{r/4}(z)$ such that
    \begin{itemize}
        \item [(i)]$\{w_0,\ldots, w_{n-2}\}$ are in $1/2$-general position in $B_{r/4}(z)$;

        \item [(ii)] We have 
        \begin{equation}
            |\theta_{2r}-\theta_r|(w_j)\leq C r^{2-n}\int_{B_{r/4}(z)} |\theta_{2r}-\theta_r|(w) d{\mathcal H}_T^{n-2}(w) \quad \forall 0\leq j\leq n-2.
        \end{equation}

                \item [(iii)] Set $L:= \operatorname{span}\{w_{n-2}-w_0,\ldots, w_1-w_0\}$. Then $M$ is smooth in $B_{3r/4}(y)$. Moreover, for every $x\in \operatorname{spt}(M)\cap B_{3r/4}(y)$, define 
                \begin{equation}
                    H_x := \pi_{T_xM}(L), \quad  V_x:=H_x^{\perp_{T_xM}}=T_xM\cap L^\perp.
                \end{equation} 
                Then, $H_x$ is $(n-2)$-dimensional, $V_x$ is $2$-dimensional,  and for all $x\in \operatorname{spt}(M)\cap B_{3r/4}(y)$ we have the estimates
        \begin{equation}
            | \pi_{T_xM}^\perp |_L|<\eta, \quad  |\pi_{V_x}(x-w_0)|\geq \frac{1}{8} r, \quad |A_M|(x)\leq \eta r^{-1}.
        \end{equation}

    \end{itemize}

    Recalling that $M$ is smooth in $B_{3r/4}(y)$, for each $x\in \operatorname{spt}(M)\cap B_{3r/4}(y)$ we let $V_x = L^\perp\cap T_xM = \pi_{T_xM}(L)^{\perp_{T_xM}}$ be the orthogonal complement of $H_x:= \pi_{T_xM}(L)$ in $T_xM$.  
    For $i=0,\ldots, n-2$, set $Z_i(x) = \pi_{T_xM}^\perp (x-w_i)$ and write for each $x\in B_{3r/4}(y)\cap \operatorname{spt}(M)$,
    \begin{equation}
        \pi_{T_xM} (x-w_0) = h(x) + v(x), \quad h(x)\in H_x, \,v(x)\in V_x
    \end{equation}
    and let $e(x) = \frac{v(x)}{|v(x)|}$.
    By Lemma~\ref{lem:pointwise-curvature} we have for $x\in \operatorname{spt}(M)\cap B_{3r/4}(y)$,
    \begin{equation} \label{eq:sff-two-terms}
        |A_M|^2(x) \leq C \left( r^{-2}\sum_{i=1}^{n-2}|A_M(\cdot, \pi_{T_xM}v_i)|^2 + |A_M(\cdot,e(x))|^2\right).
    \end{equation}
    Let us begin by controlling the second  term in \ref{eq:sff-two-terms}. 
    By Lemma~\ref{lem:jacobi-radial-trans},
    \begin{equation}
        \nabla^\perp_M Z_0(x) = -A_M(\cdot, \pi_{T_{x}M}(x-w_0)) = - A_M(\cdot, h(x)) - |v(x)| A_M(\cdot, e(x)).
    \end{equation}
    so that 
    \begin{equation}
        |v(x)|^2 |A_M(\cdot, e(x))|^2 \leq C\left(  |\nabla_M^\perp Z_0|^2 + |A_M(\cdot,h(x))|^2\right)
    \end{equation}
    By Lemma~\ref{lem:local-annular-geometry}, $|v(x)|\geq \frac{1}{8}r$ and thus
    \begin{equation} \label{eq:sff-e-bound}
    \begin{aligned}
        |A_M(\cdot, e(x))|^2 &\leq Cr^{-2} \left( |\nabla_M^\perp Z_0|^2 + |A_M(\cdot,h(x))|^2 \right) \\
        &\leq 
        Cr^{-2} \left(|\nabla_M^\perp Z_0|^2+  \sum_{i=1}^{n-2}|A_M(\cdot, \pi_{T_xM}v_i)|^2 \right)
    \end{aligned}
    \end{equation}
    where in the second line we have used that $|h(x)|\leq Cr$, $\|\pi_{T_xM}^\perp|_L\|<\frac{1}{8}$, and $w_0,\ldots, w_{n-2}$ are in $1/2$-general position at scale $r/4$.

        &\leq C \left( |\nabla^\perp_M Z_i|^2 + |\nabla^\perp_M Z_0|^2 \right)
    \end{aligned}
    \end{equation}
    by Lemma~\ref{lem:jacobi-radial-trans}. Thus we only need to estimate $\int_{B_{r/2}(y)}|\nabla_M^\perp Z_i|^2$ for $i=0,\ldots, n-2$.

We take a Whitney decomposition $\mathcal{W}$ of ${\mathbb R}^{n+m}\setminus T^{n-2}$ into dyadic closed cubes $Q\in \mathcal{W}$ satisfying
\begin{equation} \label{eq:whitney-diam}
    3\operatorname{diam} Q \leq \operatorname{dist}(Q,T^{n-2})\leq 9\operatorname{diam}(Q).
\end{equation}
Indeed, we may take $\mathcal{W}$ to be the maximal  (under inclusion) subcollection of dyadic cubes $Q$ in ${\mathbb R}^{n+m}$ which satisfy
\begin{equation}
    3\operatorname{diam} (Q) \leq \operatorname{dist}(Q,T^{n-2}).
\end{equation}
It is now straightforward to verify that $\mathcal{W}$ is a Whitney decomposition satisfying \ref{eq:whitney-diam}.





Set
\begin{equation}
    \mathcal{W}_{{\mathcal A}}:= \{Q\in \mathcal{W} \operatorname{\,\mid\,} Q\cap B_1\cap{\mathcal A}\cap \operatorname{spt}(M)\neq \emptyset\}
\end{equation}
and for every $Q\in \mathcal{W}_{{\mathcal A}}$ choose $y_Q\in Q\cap B_1\cap {\mathcal A}\cap \operatorname{spt}(M)$ and set 
\begin{equation}
    d_Q= \operatorname{dist}(y_Q,T^{n-2})  = |y_Q-z_Q|
\end{equation}
with $z_Q\in T^{n-2}$.
By the lower bound in \ref{eq:whitney-diam}, $Q\subset B_{\operatorname{diam}(Q)}(y_Q)\subset B_{d_Q/2}(y_Q)$ and thus the collection $\{B_{d_Q/2}(y_Q)\}_{Q\in \mathcal{W}_{\mathcal A}}$ covers $B_1\cap {\mathcal A} \cap \operatorname{spt}(M)$. 

Set, for each $Q\in \mathcal{W}_{{\mathcal A}}$, $R_Q :=T^{n-2}\cap B_{d_Q/4}(z_Q)$. We claim that for each $w\in T^{n-2}$ and $t>0$, we have 
\begin{equation} \label{eq:whitney-packing}
     \# \mathcal{W}_{w,t} := \#\left\{Q\in \mathcal{W}_{\mathcal A} \operatorname{\,\mid\,} w\in R_Q , \,t\in (d_Q, 2d_Q] \right\} \leq C(n,m).
\end{equation}
Indeed, if $Q\in \mathcal{W}_{w,t}$ then 
\begin{equation}
    \operatorname{dist}(w,Q)\leq |w-y_Q|\leq |w-z_Q| + |z_Q-y_Q|\leq 2d_Q \leq 2t.
\end{equation}
Since $\operatorname{diam}(Q) \leq C(n,m)d_Q \leq C(n,m)t$, it follows that every $Q\in \mathcal{W}_{w,t}$ is contained in $B_{C(m,n)t}(w)$. 
 Since the $Q$'s have pairwise disjoint interiors with diameter comparable to $t$, a packing estimate proves the claim. 

We have 
\begin{equation} \label{eq:sff-neck-estimate-split}
    \begin{aligned}
        \int_{{\mathcal A}\cap B_1} |A_M|^2 d|M| &\leq \sum_{Q\in \mathcal{W}_{\mathcal{A}}} \int_{B_{d_Q/2}(y_Q)} |A_M|^2 d|M| \\
        &\leq \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>4r_{z_Q}} }\int_{B_{d_Q/2}(y_Q)} |A_M|^2 d|M| + \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>r_{z_Q}\geq d_Q/4}} \int_{B_{d_Q/2}(y_Q)} |A_M|^2 d|M|. 
    \end{aligned}
\end{equation}
For the first sum in \ref{eq:sff-neck-estimate-split}, we note that for every $Q\in \mathcal{W}_{{\mathcal A}}$ with $d_Q>4r_{z_Q}$ and $w\in R_Q= T^{n-2}\cap B_{d_Q/4}(z_Q)$, we have 
\(
    r_w\leq r_{z_Q} + \frac{d_Q}{2}<d_Q 
\)
since $\operatorname{Lip}(r_z)\leq 2$. 
On the other hand for fixed $w\in T^{n-2}$ consider the monotone function $\theta_{r}(w)$ and let $\nu_w$ be its associated  Lebesgue–Stieltjes measure obeying $\nu_w((s,r]) = \theta_r(w)-\theta_s(w)$. By \ref{eq:whitney-packing},
\begin{equation} \label{eq:whitney-packing-indicator}
    m_w(t):= \sum_{\substack{Q\in \mathcal{W}_{\mathcal A} \\ w\in R_Q}} 1_{(d_Q, 2d_Q]}(t) \leq C(n,m) 
\end{equation}
for all $w\in T^{n-2}\cap B_1$ and $t>0$.
Thus we apply Lemma~\ref{lem:pointwise-curvature}, Proposition~\ref{prop:local-annular-curvature} and \ref{eq:whitney-packing-indicator} to estimate
\begin{equation} \label{eq:neck-est-first-term}
    \begin{aligned}
        \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>4r_{z_Q}} }\int_{B_{d_Q/2}(y_Q)} |A_M|^2 d|M|& \leq C  \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>4r_{z_Q}} } \int_{R_Q} |\theta_{2d_Q}-\theta_{d_Q}|(w)d{\mathcal H}^{n-2}_T(w)\\
        &\leq C \int_{T^{n-2}\cap B_3} \int_{r_w}^{4} m_w(t) d\nu_w d{\mathcal H}_T^{n-2}(w) 
        \leq C \delta
    \end{aligned}
\end{equation}
where we have applied Definition~\ref{def:annular-region}(a4) and the Ahlfors regularity Theorem~\ref{thm:annular-structure} in the last line.

To estimate the second sum in \ref{eq:sff-neck-estimate-split}, we use Lemma~\ref{lem:local-annular-geometry} to estimate, with $\delta<\delta(n,m,\Lambda,\epsilon)$ small enough so that $\eta^2 <\epsilon$,
\begin{equation} \label{eq:inner-neck-est}
    \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>r_{z_Q}\geq d_Q/4}} \int_{B_{d_Q/2}(y_Q)} |A_M|^2 d|M| \leq C \epsilon  \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>r_{z_Q}\geq d_Q/4}} d_Q^{n-2}.
\end{equation}
Define for each $w\in T^{n-2}$,
\begin{equation}
    N(w) := \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>r_{z_Q}\geq d_Q/4}} 1_{T^{n-2}\cap B_{r_{z_Q}/8}(z_Q)}(w).
\end{equation}
For every $Q\in \mathcal{W}_{\mathcal A}$ with $d_Q>r_{z_Q}\geq d_Q/4$, any $w\in T^{n-2}\cap B_{r_{z_Q}/8}(z_Q)$ satisfies 
$
    \frac{4}{5}r_{w}<d_Q \leq \frac{16}{3}r_w.
$
Therefore for any $w\in T^{n-2}$,
\begin{equation}
\begin{aligned}
    N(w) & \leq C  \int_0^\infty \sum_{\substack{ Q\in \mathcal{W}_{\mathcal{A}},\, w\in T^{n-2}\cap B_{r_{z_Q}/8}(z_Q) \\ d_Q>r_{z_Q}\geq d_Q/4}} 1_{(d_Q, 2d_Q]}(t) \frac{dt}{t} 
    \leq \int_{\frac{4}{5}r_w}^{\frac{32}{3}r_w} m_w(t) \frac{dt}{t} \leq C(n,m,\Lambda)
\end{aligned}
\end{equation}
since $T^{n-2}\cap B_{r_{z_Q}}(z_Q)\subset R_Q$ and \ref{eq:whitney-packing-indicator} holds.

Finally, by the Ahlfors regularity in Theorem~\ref{thm:annular-structure} 
\begin{equation}
    \begin{aligned}
        \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>r_{z_Q}\geq d_Q/4}} d_Q^{n-2} &\leq C \sum_{\substack{Q\in \mathcal{W}_{\mathcal{A}}\\ d_Q>r_{z_Q}\geq d_Q/4}}  {\mathcal H}^{n-2}_T(B_{r_{z_Q}/8}(z_Q) )
        \leq C \int N(w) d{\mathcal H}^{n-2}_T(w) \leq C.
    \end{aligned}
\end{equation}
Recalling \ref{eq:neck-est-first-term} and \ref{eq:inner-neck-est}, we finish the proof of the theorem.

    and so by Lemma~\ref{A:lem:radial-comparison} we can choose $\delta=\delta(n,m,\Lambda,\epsilon, r)$ sufficiently small so that 
    \begin{equation*}
        |\theta_{at}(tx_j) - \theta_{as/4}(sx_j/4)|<\epsilon
    \end{equation*}
    for all $a\in [1,2]$. Note that $\delta$ and $r(\delta;n,m,\Lambda,\epsilon)$ must be chosen so that $\delta\leq c(n,m) \frac{\epsilon^2}{1+\Lambda} r(\delta;n,m,\Lambda,\epsilon)^{n+2}$, which  shows the dependency on $\delta$. Summing the  estimates for $a=1,2$ gives
    \begin{equation*}
        \theta_{2t}(tx_j) - \theta_t(tx_j)\leq C(n)\epsilon.
    \end{equation*}
    From this pinching it is now easy to see the $(k,C(n,m)\epsilon)$-symmetry on $B_t$.
