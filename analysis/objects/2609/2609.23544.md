<!-- theorem-1: body_md -->
\label{thm:counterexample}
Let $n\ge2$ and let $0<\alpha_-<\alpha_+$ be prescribed. There exists
a bounded connected Lipschitz polyhedral domain $\Omega\subset\mathbb{R}^n$
such that $\mathcal{R}_\Omega$ is continuously differentiable near
$[\alpha_-,\alpha_+]$ and satisfies $\mathcal{R}_\Omega'(\alpha)>0$
throughout this interval. In particular,
\begin{equation}\label{eq:reversal}
 \mathcal{R}_\Omega(\alpha_+)>\mathcal{R}_\Omega(\alpha_-).
\end{equation}
When $n=2$, $\Omega$ can be chosen to be a simply connected polygon.

<!-- theorem-2: body_md -->
\label{thm:boxes}
Let $n\ge1$ and $B=\prod_{j=1}^n(-t_j,t_j)$ with $t_j>0$.
Then $\mathcal{R}_B$ is differentiable on $(0,+\infty)$ and $\mathcal{R}_B'(\alpha)<0$.
Writing $T=\max_{1\le j\le n}t_j$, its endpoint limits are
\begin{equation}\label{eq:boxlimits}
 \lim_{\alpha\downarrow0}\mathcal{R}_B(\alpha)=+\infty,
 \,\,\,
 \lim_{\alpha\to+\infty}\mathcal{R}_B(\alpha)
 =1+\frac{3/T^2}{\sum_{j=1}^n t_j^{-2}}.
\end{equation}
The conclusion includes boxes whose longest edge is not unique.

<!-- lemma-1: body_md -->
\label{lem:derivatives}
For $a>0$,
\begin{equation}\label{eq:derivatives}
 p'(a)=\frac{2p(a)}{p(a)+c(a)},\,\,\,
 q'(a)=\frac{2q(a)}{q(a)+c(a)}.
\end{equation}
The function
\begin{equation}\label{eq:elasticity}
 E(a)=\frac{ap'(a)}{p(a)}
 =\frac{2}{1+2x(a)/\sin\bigl(2x(a)\bigr)}
\end{equation}
is strictly decreasing and satisfies $0<E(a)<1$. Moreover,
\begin{equation}\label{eq:gapelasticity}
 \frac{ad'(a)}{d(a)}
 =E(a)\frac{c(a)}{q(a)+c(a)}<E(a),
\end{equation}
and $q(a)/p(a)$ is strictly decreasing, with
\begin{equation}\label{eq:qpbound}
 \frac{q(a)}{p(a)}>4,
 \,\,\, \lim_{a\to+\infty}\frac{q(a)}{p(a)}=4.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
In the following calculations, $x=x(a)$, $y=y(a)$, $p=p(a)$,
$q=q(a)$ and $c=c(a)$.
Using $\tan x=a/x$ and $\sec^2x=1+\tan^2x$, we obtain
\[
 \frac{\mathrm da}{\mathrm dx}
 =\tan x+x\sec^2x
 =\frac{a}{x}+x+\frac{a^2}{x}
 =\frac{p+c}{x}.
\]
Similarly, $\cot y=-a/y$ gives
\[
 \frac{\mathrm da}{\mathrm dy}
 =-\cot y+y\csc^2y
 =\frac{a}{y}+y+\frac{a^2}{y}
 =\frac{q+c}{y}.
\]
Since $p'=2x x'$ and $q'=2y y'$, these identities prove
\eqref{eq:derivatives}. Substituting $a=x\tan x$ then yields
\[
 E(a)=\frac{2a}{p+c}
 =\frac{2}{1+a+p/a}
 =\frac{2}{1+x\tan x+x\cot x}
 =\frac{2}{1+2x/\sin(2x)}.
\]
For $0<u<\pi$,
\[
 \frac{\mathrm d}{\mathrm du}\left(\frac{u}{\sin u}\right)
 =\frac{\sin u-u\cos u}{\sin^2u},
 \,\,\,
 \sin u-u\cos u=\int_0^u v\sin v\,\mathrm{d} v>0.
\]
Thus $u/\sin u$ is strictly increasing. Since $x(a)$ is increasing,
\eqref{eq:elasticity} shows that $E$ is strictly decreasing.
The inequalities $0<\sin(2x)<2x$ give $0<E<1$.
In particular, for $0<t<T$ and fixed $\alpha>0$,
\[
 \alpha\frac{\partial}{\partial\alpha}
       \log\lambda_1(I_t;\alpha)
 =E(\alpha t)>E(\alpha T)
 =\alpha\frac{\partial}{\partial\alpha}
       \log\lambda_1(I_T;\alpha).
\]
This is the comparison between different scales.

Subtracting the two identities in \eqref{eq:derivatives}, we obtain
\[
 \begin{aligned}
 d'
 &=\frac{2q}{q+c}-\frac{2p}{p+c}\\
 &=\frac{2q(p+c)-2p(q+c)}{(p+c)(q+c)}\\
 &=\frac{2c(q-p)}{(p+c)(q+c)}.
 \end{aligned}
\]
Dividing by $d=q-p>0$ and multiplying by $a$ gives
\[
 \frac{ad'}{d}
 =\frac{2a}{p+c}\,\frac{c}{q+c}
 =E(a)\frac{c}{q+c}<E(a),
\]
which proves \eqref{eq:gapelasticity}. The factor $c/(q+c)$ lies
strictly between zero and one. It is precisely the strict loss in
relative growth that will make the box argument work. Also,
\[
 \frac{\mathrm d}{\mathrm da}\log\left(\frac{q}{p}\right)
 =\frac{q'}q-\frac{p'}p
 =\frac{2}{q+c}-\frac{2}{p+c}
 =-\frac{2(q-p)}{(p+c)(q+c)}<0.
\]
Finally, \eqref{eq:roots} implies $x(a)\to\pi/2$ and
$y(a)\to\pi$ as $a\to+\infty$. Hence $q(a)/p(a)\to4$.
A strictly decreasing function with this limit is strictly greater
than four at every finite positive $a$, proving \eqref{eq:qpbound}.

<!-- lemma-2: body_md -->
\label{lem:longest}
Fix $\alpha>0$. The interval gap
\[
 \delta_\alpha(t)=t^{-2}d(\alpha t)
\]
is strictly decreasing for $t>0$. Consequently, for the box in
Theorem~\ref{thm:boxes},
\begin{equation}\label{eq:boxgap}
 \begin{aligned}
 \lambda_1(B;\alpha)=\sum_{j=1}^n t_j^{-2}p(\alpha t_j),\,\,\,
 \lambda_2(B;\alpha)-\lambda_1(B;\alpha)=T^{-2}d(\alpha T).
 \end{aligned}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Differentiation and \eqref{eq:gapelasticity} give
\[
 \frac{t\delta_\alpha'(t)}{\delta_\alpha(t)}
 =-2+\frac{\alpha t\,d'(\alpha t)}{d(\alpha t)}
 <-2+E(\alpha t)<-1.
\]
In particular, $\delta_\alpha'(t)<0$.
By separation of variables, the eigenvalues of $B$ are all sums
\[
 \sum_{j=1}^n\lambda_{m_j}(I_{t_j};\alpha),
 \,\,\, m_j\ge1.
\]
Taking every $m_j=1$ gives the first eigenvalue. In any other sum,
at least one coordinate contributes at least its second eigenvalue.
The smallest possible increase is attained by changing only one
coordinate from its first to its second eigenvalue. Therefore
\[
 \lambda_2(B;\alpha)-\lambda_1(B;\alpha)
 =\min_{1\le j\le n}\delta_\alpha(t_j)
 =\delta_\alpha(T),
\]
where the final equality uses the strict decrease just proved.
This establishes \eqref{eq:boxgap}.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
By Lemma~\ref{lem:longest}, the gap has a single expression determined
by $T$, independently of which longest edge is selected. Write
\[
 L_j(\alpha)=t_j^{-2}p(\alpha t_j),\,\,\,
 L(\alpha)=\sum_{j=1}^n L_j(\alpha),\,\,\,
 G(\alpha)=T^{-2}d(\alpha T).
\]
All these functions are positive and smooth for $\alpha>0$.
Since $L_j'=t_j^{-1}p'(\alpha t_j)$, we have
\[
 \alpha\frac{L_j'(\alpha)}{L_j(\alpha)}
 =\frac{\alpha t_j p'(\alpha t_j)}{p(\alpha t_j)}
 =E(\alpha t_j).
\]
The numbers $L_j/L$ are positive and sum to one. The decrease of
$E$, together with $t_j\le T$, therefore gives
\begin{equation}\label{eq:weighted}
 \alpha\frac{L'(\alpha)}{L(\alpha)}
 =\sum_{j=1}^n\frac{L_j(\alpha)}{L(\alpha)}E(\alpha t_j)
 \ge E(\alpha T).
\end{equation}
On the other hand, \eqref{eq:gapelasticity} gives
\begin{equation}\label{eq:strictgap}
 \alpha\frac{G'(\alpha)}{G(\alpha)}
 =E(\alpha T)\frac{c(\alpha T)}{q(\alpha T)+c(\alpha T)}
 <E(\alpha T).
\end{equation}
Subtracting \eqref{eq:weighted} from \eqref{eq:strictgap}, and using
$\alpha>0$, yields
\[
 \frac{\mathrm d}{\mathrm d\alpha}
 \log\left(\frac{G(\alpha)}{L(\alpha)}\right)
 =\frac{G'(\alpha)}{G(\alpha)}
  -\frac{L'(\alpha)}{L(\alpha)}<0.
\]
Using \eqref{eq:relativecriterion}, it follows that
\[
 \mathcal{R}_B'(\alpha)
 =\frac{G(\alpha)}{L(\alpha)}
  \left(\frac{G'(\alpha)}{G(\alpha)}
       -\frac{L'(\alpha)}{L(\alpha)}\right)<0.
\]
If several edges have length $2T$, each gives the same smooth
expression $T^{-2}d(\alpha T)$ for the gap. Thus a possible
multiplicity of $\lambda_2$ causes no loss of differentiability.
The strict inequality \eqref{eq:strictgap} also applies when every
edge has the same length.

As $\alpha\downarrow0$, \eqref{eq:plimits} gives
\[
 L(\alpha)\sim\alpha\sum_{j=1}^n t_j^{-1},\,\,\,
 G(\alpha)\longrightarrow\frac{\pi^2}{4T^2}.
\]
Hence $G/L\to+\infty$. As $\alpha\to+\infty$,
\[
 L(\alpha)\longrightarrow\frac{\pi^2}{4}\sum_{j=1}^n t_j^{-2},
 \,\,\, G(\alpha)\longrightarrow\frac{3\pi^2}{4T^2}.
\]
Substitution into $\mathcal{R}_B=1+G/L$ proves \eqref{eq:boxlimits}.

<!-- proposition-1: body_md -->
\label{prop:negative-boxes}
For the box $B$ in Theorem~\ref{thm:boxes}, the function
$\mathcal{R}_B$ is differentiable on $(-\infty,0)$ and satisfies
\[
 \mathcal{R}_B'(\alpha)<0\,\,\,(\alpha<0),
 \,\,\,
 \lim_{\alpha\to-\infty}\mathcal{R}_B(\alpha)=1,
 \,\,\,
 \lim_{\alpha\uparrow0}\mathcal{R}_B(\alpha)=-\infty.
\]
There is a unique $\alpha_0<0$ such that
$\lambda_2(B;\alpha_0)=0$. The ratio belongs to $(0,1)$ for
$\alpha<\alpha_0$ and is negative for $\alpha_0<\alpha<0$.
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
Write $L(\alpha)=\lambda_1(B;\alpha)$ and
$G(\alpha)=\lambda_2(B;\alpha)-\lambda_1(B;\alpha)$.
The longest-edge formula holds for all real parameters
\cite[Corollary~6.3]{Laugesen}, so the smooth dependence of interval
eigenvalues gives smooth functions $L$ and $G$, including at
parameters where $\lambda_2(B;\alpha)=0$. The gap theorem
\cite[Theorem~2.1]{Laugesen} gives $G'\ge0$ and
$G(\alpha)\to0$ as $\alpha\to-\infty$.

The constant trial function gives
\[
 L(\alpha)\le
 \alpha\frac{\abs{\partial B}_{n-1}}{\abs{B}}<0.
\]
Let $u_1(\alpha)$ be a real, $L^2(B)$-normalized first eigenfunction.
Differentiating the form along $u_1$, and using its eigenfunction
equation and the derivative of its normalization, yields
\[
 \begin{aligned}
 L'(\alpha)
 &=\int_{\partial B}\abs[\big]{\operatorname{Tr} u_1}^2\,\mathrm{d} S
   +2\mathfrak a_{B,\alpha}(u_1,\partial_\alpha u_1)\\
 &=\int_{\partial B}\abs[\big]{\operatorname{Tr} u_1}^2\,\mathrm{d} S
   +2L\langle u_1,\partial_\alpha u_1\rangle_{L^2(B)}\\
 &=\int_{\partial B}\abs[\big]{\operatorname{Tr} u_1}^2\,\mathrm{d} S>0.
 \end{aligned}
\]
{Indeed, a zero trace would imply
$L=\int_B\abs{\nabla u_1}^2\,\mathrm{d} x\ge0$, contrary to $L<0$.}
As $L<0$ and $G>0$, we now obtain
\[
 \mathcal{R}_B'(\alpha)
 =\frac{G'L-GL'}{L^2}<0.
\]
Here $G'L\le0$ and $-GL'<0$; no relative-growth estimate is needed.

The constant-function bound implies $L\to-\infty$ as
$\alpha\to-\infty$. Together with $G\to0$, it gives
$\mathcal{R}_B=1+G/L\to1$. At $\alpha\uparrow0$, continuity at the
Neumann problem gives $L\to0^-$ and $G\to\pi^2/(4T^2)>0$, so
$\mathcal{R}_B\to-\infty$. The strict decrease and these two limits give
a unique zero of the ratio. Since $L<0$ throughout this range, it
is exactly the unique zero of $\lambda_2$. The remaining signs
follow as well.

<!-- lemma-3: body_md -->
\label{lem:disconnected}
{The function $R_s$ satisfies $R_s'(\alpha)>0$ for every $\alpha>0$, and}
\begin{equation}\label{eq:Rrange}
 \frac1s<R_s(\alpha)<\frac1{s^2}.
\end{equation}
For every $\alpha>0$,
\begin{equation}\label{eq:ordering}
 \lambda_1(D;\alpha)=\lambda_1(Q_1;\alpha),\,\,\,
 \lambda_2(D;\alpha)=\lambda_1(Q_2;\alpha).
\end{equation}
In particular, $\mathcal{R}_D=R_s$ is strictly increasing.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
For a single box we compared the gap with its first eigenvalue.
Here both quantities being compared are first eigenvalues, but at
different scales. Taking the logarithmic derivative of
\eqref{eq:Rlimit} and using the chain rule, we find
\[
 \begin{aligned}
 \alpha\frac{R_s'(\alpha)}{R_s(\alpha)}
 &=\alpha\left(\frac{s p'(s\alpha)}{p(s\alpha)}
              -\frac{p'(\alpha)}{p(\alpha)}\right)\\
 &=\frac{s\alpha p'(s\alpha)}{p(s\alpha)}
   -\frac{\alpha p'(\alpha)}{p(\alpha)}\\
 &=E(s\alpha)-E(\alpha)\\
 &>0.
 \end{aligned}
\]
The last inequality follows from $s\alpha<\alpha$ and the strict
decrease of $E$ in Lemma~\ref{lem:derivatives}. Thus $R_s$ is strictly
increasing. To compute its limit at zero, write
\[
 R_s(\alpha)
 =\frac1s\,
   \frac{p(s\alpha)/(s\alpha)}{p(\alpha)/\alpha}.
\]
Since both fractions in the final quotient tend to one,
\[
 \lim_{\alpha\downarrow0}R_s(\alpha)=\frac1s.
\]
As $\alpha\to+\infty$, both $p(s\alpha)$ and $p(\alpha)$ tend to
$\pi^2/4$, and hence
\[
 \lim_{\alpha\to+\infty}R_s(\alpha)
 =\frac1{s^2}\,\frac{\pi^2/4}{\pi^2/4}
 =\frac1{s^2}.
\]
The strict increase gives the strict bounds in \eqref{eq:Rrange}.

It remains to identify the first two eigenvalues of the union.
For $Q_1$, the second eigenvalue is obtained by choosing the second
interval eigenvalue in one coordinate, so
\[
 \lambda_2(Q_1;\alpha)=(n-1)p(\alpha)+q(\alpha).
\]
Consequently,
\[
 \begin{aligned}
 \frac{\lambda_2(Q_1;\alpha)}{\lambda_1(Q_1;\alpha)}
 &=\frac{(n-1)p(\alpha)+q(\alpha)}{n p(\alpha)}\\
 &=1+\frac1n\left(\frac{q(\alpha)}{p(\alpha)}-1\right)\\
 &>1+\frac3n,
 \end{aligned}
\]
where \eqref{eq:qpbound} was used. The choice \eqref{eq:srange}
means that $s^2>n/(n+3)$, or equivalently
\[
 \frac1{s^2}<\frac{n+3}{n}=1+\frac3n.
\]
Combining these inequalities gives the full ordering
\[
 1<\frac1s<R_s(\alpha)<\frac1{s^2}<1+\frac3n
 <\frac{\lambda_2(Q_1;\alpha)}{\lambda_1(Q_1;\alpha)}.
\]
Multiplication by $\lambda_1(Q_1;\alpha)>0$ yields
\[
 \lambda_1(Q_1;\alpha)<\lambda_1(Q_2;\alpha)
 <\lambda_2(Q_1;\alpha).
\]
Every other eigenvalue of $Q_1$ is at least $\lambda_2(Q_1;\alpha)$,
and every other eigenvalue of $Q_2$ is strictly greater than
$\lambda_1(Q_2;\alpha)$, because its first eigenvalue is simple.
Thus the first two entries in the merged spectrum are exactly those
in \eqref{eq:ordering}. If $s=1$, the two first
eigenvalues would coincide and their ratio would be the constant
one, so the unequal sizes are essential.

<!-- lemma-4: body_md -->
\label{lem:window}
Let $Q\subset\mathbb{R}^n$ be a fixed bounded Lipschitz domain, let $\alpha>0$,
and let $\Gamma_\varepsilon\subset\partial Q$ be measurable sets such that
$\abs{\Gamma_\varepsilon}_{n-1}\to0$ as $\varepsilon\downarrow0$.
Let $\nu_{k,\varepsilon}$ be the eigenvalues of the form
\[
 \mathfrak a_\varepsilon[u]=\int_Q\abs{\nabla u}^2\,\mathrm{d} x
 +\alpha\int_{\partial Q\setminus\Gamma_\varepsilon}
       \abs[\big]{\operatorname{Tr} u}^2\,\mathrm{d} S,
 \,\,\, u\in H^1(Q).
\]
Then $\nu_{k,\varepsilon}\to\lambda_k(Q;\alpha)$ for each fixed $k$.
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
The compactness of the trace map from $H^1(Q)$ to $L^2(\partial Q)$
implies
\begin{equation}\label{eq:eta}
 \eta_\varepsilon:=\sup_{\norm{u}_{H^1(Q)}\le1}
       \int_{\Gamma_\varepsilon}\abs[\big]{\operatorname{Tr} u}^2\,\mathrm{d} S
       \longrightarrow0.
\end{equation}
To see this, suppose otherwise. There would be a sequence
$\varepsilon_j\downarrow0$, a number $b>0$, and functions $u_j$ with
$\norm{u_j}_{H^1(Q)}\le1$ such that
\[
 \int_{\Gamma_{\varepsilon_j}}\abs[\big]{\operatorname{Tr} u_j}^2\,\mathrm{d} S\ge b.
\]
After passage to a subsequence, the traces converge strongly in
$L^2(\partial Q)$ to some $v$. But
\[
 \begin{aligned}
 \int_{\Gamma_{\varepsilon_j}}\abs[\big]{\operatorname{Tr} u_j}^2\,\mathrm{d} S
 &\le 2\int_{\Gamma_{\varepsilon_j}}\abs[\big]{\operatorname{Tr} u_j-v}^2\,\mathrm{d} S
       +2\int_{\Gamma_{\varepsilon_j}}\abs{v}^2\,\mathrm{d} S\\
 &\le2\norm[\big]{\operatorname{Tr} u_j-v}_{L^2(\partial Q)}^2
       +2\int_{\Gamma_{\varepsilon_j}}\abs{v}^2\,\mathrm{d} S
 \longrightarrow0.
 \end{aligned}
\]
Here the first term tends to zero by strong convergence, and the
second does so by absolute continuity of the integral of
$\abs{v}^2$, since $\abs{\Gamma_{\varepsilon_j}}_{n-1}\to0$.
This contradiction proves \eqref{eq:eta}.

By homogeneity, \eqref{eq:eta} gives, for every $u\in H^1(Q)$,
\[
 \int_{\Gamma_\varepsilon}\abs[\big]{\operatorname{Tr} u}^2\,\mathrm{d} S
 \le\eta_\varepsilon\norm{u}_{H^1(Q)}^2.
\]
The boundary term in $\mathfrak a_\varepsilon$ is nonnegative because
$\alpha>0$, so $\norm{\nabla u}_{L^2(Q)}^2\le\mathfrak a_\varepsilon[u]$.
We can therefore compare the two forms step by step:
\[
 \begin{aligned}
 \mathfrak a_\varepsilon[u]
 &\le\mathfrak a_{Q,\alpha}[u]\\
 &=\mathfrak a_\varepsilon[u]
   +\alpha\int_{\Gamma_\varepsilon}\abs[\big]{\operatorname{Tr} u}^2\,\mathrm{d} S\\
 &\le\mathfrak a_\varepsilon[u]
    +\alpha\eta_\varepsilon\left(
       \norm{\nabla u}_{L^2(Q)}^2+\norm{u}_{L^2(Q)}^2\right)\\
 &\le(1+\alpha\eta_\varepsilon)\mathfrak a_\varepsilon[u]
       +\alpha\eta_\varepsilon\norm{u}_{L^2(Q)}^2.
 \end{aligned}
\]
For $u\ne0$, division by $\norm{u}_{L^2(Q)}^2$ gives the same
comparison for the Rayleigh quotients. The min--max formula is
\[
 \nu_{k,\varepsilon}
 =\min_{\substack{V\subset H^1(Q)\\ \dim V=k}}
   \;\max_{u\in V\setminus\{0\}}
      \frac{\mathfrak a_\varepsilon[u]}{\norm{u}_{L^2(Q)}^2},
\]
and the corresponding formula for $\lambda_k(Q;\alpha)$ has the
same class of subspaces. Taking these maxima and minima in the
quotient comparison yields
\[
 \nu_{k,\varepsilon}\le\lambda_k(Q;\alpha)
 \le(1+\alpha\eta_\varepsilon)\nu_{k,\varepsilon}+\alpha\eta_\varepsilon.
\]
Equivalently,
\[
 \frac{\lambda_k(Q;\alpha)-\alpha\eta_\varepsilon}
      {1+\alpha\eta_\varepsilon}
 \le\nu_{k,\varepsilon}\le\lambda_k(Q;\alpha).
\]
Both bounds tend to $\lambda_k(Q;\alpha)$, proving the claim.

{In Steps 1--3, $\alpha>0$ and $k\ge1$ are fixed.}

\emph{Step 1. A lower bound obtained by separating the three pieces.}
The interfaces are
\[
 \Gamma_{1,\varepsilon}=\{1\}\times(-\varepsilon,\varepsilon)^{n-1},\,\,\,
 \Gamma_{2,\varepsilon}=\{2\}\times(-\varepsilon,\varepsilon)^{n-1}.
\]
On $Q_i$, let $\mathfrak a_{i,\varepsilon}$ be the form in
Lemma~\ref{lem:window}, with $\Gamma_\varepsilon=\Gamma_{i,\varepsilon}$.
On the passage, let
\[
 \mathfrak t_\varepsilon[w]
 =\int_{T_\varepsilon}\abs{\nabla w}^2\,\mathrm{d} x
  +\alpha\int_{\partial_{\mathrm{lat}}T_\varepsilon}
        \abs[\big]{\operatorname{Tr} w}^2\,\mathrm{d} S,
 \,\,\, w\in H^1(T_\varepsilon),
\]
where $\partial_{\mathrm{lat}}T_\varepsilon$ consists of the lateral faces,
not the two ends. Consider the form
\[
 \mathfrak b_\varepsilon[u_1,w,u_2]
 =\mathfrak a_{1,\varepsilon}[u_1]+\mathfrak t_\varepsilon[w]
  +\mathfrak a_{2,\varepsilon}[u_2]
\]
on the larger space
\[
 \mathcal H_\varepsilon
 =H^1(Q_1)\oplus H^1(T_\varepsilon)\oplus H^1(Q_2).
\]
This means that the functions on the three pieces are allowed to
be independent. In boundary-condition language, the cut faces have
Neumann conditions on both sides, while the exterior faces keep
their Robin conditions.

If $u\in H^1(\Omega_\varepsilon)$, its restrictions define an element
$Ju=(u|_{Q_1},u|_{T_\varepsilon},u|_{Q_2})$ of $\mathcal H_\varepsilon$.
The restrictions have matching traces at the interfaces, and
\[
 \begin{aligned}
 \norm{Ju}_{L^2(Q_1)\oplus L^2(T_\varepsilon)\oplus L^2(Q_2)}^2
 &=\norm{u}_{L^2(\Omega_\varepsilon)}^2,\\
 \mathfrak b_\varepsilon[Ju]&=\mathfrak a_{\Omega_\varepsilon,\alpha}[u].
 \end{aligned}
\]
The second equality holds because the interfaces are interior to
$\Omega_\varepsilon$ and neither form includes a boundary integral there.
Thus the Rayleigh quotient is unchanged on the original space,
while allowing all of $\mathcal H_\varepsilon$ enlarges the set of admissible
subspaces in the min--max principle. If $\mu_{k,\varepsilon}$ denotes the
$k$th eigenvalue of $\mathfrak b_\varepsilon$, then
\[
 \mu_{k,\varepsilon}\le\lambda_k(\Omega_\varepsilon;\alpha).
\]

Let $\nu_{j,i,\varepsilon}$ be the eigenvalues on $Q_i$ with the boundary
term removed on $\Gamma_{i,\varepsilon}$. Since
$\abs{\Gamma_{i,\varepsilon}}_{n-1}=(2\varepsilon)^{n-1}\to0$,
Lemma~\ref{lem:window} gives
\[
 \nu_{j,i,\varepsilon}\longrightarrow\lambda_j(Q_i;\alpha)
 \,\,\,(i=1,2)
\]
for each fixed $j$. On $T_\varepsilon$, the longitudinal Neumann eigenvalues
are $(m\pi)^2$, $m=0,1,2,\ldots$, because its length is one. Each
transverse interval has first Robin eigenvalue
$\varepsilon^{-2}p(\alpha\varepsilon)$. Separation of variables therefore gives
the lowest passage eigenvalue
\begin{equation}\label{eq:tube}
 \begin{aligned}
 \tau_{1,\varepsilon}
 &=0+\sum_{j=1}^{n-1}\varepsilon^{-2}p(\alpha\varepsilon)\\
 &=(n-1)\varepsilon^{-2}p(\alpha\varepsilon)
 \sim\frac{(n-1)\alpha}{\varepsilon}
 \longrightarrow+\infty.
 \end{aligned}
\end{equation}
Both $n\ge2$ and $\alpha>0$ are used in the last limit.
The factor $1/\varepsilon$ expresses the growth of lateral boundary area
relative to passage volume. Thus the positive Robin boundary term,
not the small volume alone, excludes bounded passage eigenvalues.

Let $\widehat\nu_{k,\varepsilon}$ be the $k$th eigenvalue of the two cube
problems together. It is the $k$th smallest entry of
$\{\nu_{j,i,\varepsilon}:1\le j\le k,\ i=1,2\}$, counted with
multiplicity. Changing each entry by at most $\delta$ changes the
$k$th smallest entry by at most $\delta$. Hence
\[
 \abs[\big]{\widehat\nu_{k,\varepsilon}-\lambda_k(D;\alpha)}
 \le\max_{\substack{1\le j\le k\\ i=1,2}}
       \abs[\big]{\nu_{j,i,\varepsilon}-\lambda_j(Q_i;\alpha)}
 \longrightarrow0.
\]
By \eqref{eq:tube}, all passage eigenvalues exceed
$\widehat\nu_{k,\varepsilon}$ for small $\varepsilon$, so
$\mu_{k,\varepsilon}=\widehat\nu_{k,\varepsilon}$. The variational bound
$\lambda_k(\Omega_\varepsilon;\alpha)\ge\mu_{k,\varepsilon}$ now gives
\begin{equation}\label{eq:liminf}
 \liminf_{\varepsilon\downarrow0}\lambda_k(\Omega_\varepsilon;\alpha)
 \ge\lim_{\varepsilon\downarrow0}\mu_{k,\varepsilon}
 =\lambda_k(D;\alpha).
\end{equation}

\emph{Step 2. Cutoff functions near the attachment points.}
Let $P_1=(1,0,\ldots,0)$ and $P_2=(2,0,\ldots,0)$.
We construct Lipschitz functions $\chi_{i,\varepsilon}:\mathbb{R}^n\to[0,1]$
which vanish in a neighborhood of the whole interface
$\Gamma_{i,\varepsilon}$ and satisfy
\begin{equation}\label{eq:cutoffprops}
 \abs[\big]{\operatorname{supp}(1-\chi_{i,\varepsilon})}\longrightarrow0,
 \,\,\,
 \int_{\mathbb{R}^n}\abs{\nabla\chi_{i,\varepsilon}}^2\,\mathrm{d} x\longrightarrow0.
\end{equation}
Here and below, $\abs{\cdot}$ applied to a subset of $\mathbb{R}^n$ denotes
its $n$-dimensional Lebesgue measure.

For $n\ge3$, put $\rho_\varepsilon=2\sqrt n\,\varepsilon$ and define, with
$r=\abs{x-P_i}$,
\[
 \chi_{i,\varepsilon}(x)=
 \begin{cases}
 0,&r\le\rho_\varepsilon,\\[2pt]
 \displaystyle\frac{r-\rho_\varepsilon}{\rho_\varepsilon},
    &\rho_\varepsilon<r<2\rho_\varepsilon,\\[6pt]
 1,&r\ge2\rho_\varepsilon.
 \end{cases}
\]
Every point of $\overline{\Gamma_{i,\varepsilon}}$ lies at distance at most
$\sqrt{n-1}\,\varepsilon<\rho_\varepsilon$ from $P_i$, so the required vanishing
holds. If $\omega_n$ denotes the volume of the unit ball, then
\[
 \abs[\big]{\operatorname{supp}(1-\chi_{i,\varepsilon})}
 \le\omega_n(2\rho_\varepsilon)^n\longrightarrow0.
\]
The gradient has magnitude $1/\rho_\varepsilon$ in the annulus and is
zero elsewhere, apart from sets of measure zero. Hence
\[
 \begin{aligned}
 \int_{\mathbb{R}^n}\abs{\nabla\chi_{i,\varepsilon}}^2\,\mathrm{d} x
 &=\frac{n\omega_n}{\rho_\varepsilon^2}
      \int_{\rho_\varepsilon}^{2\rho_\varepsilon}r^{n-1}\,\mathrm{d} r\\
 &=\omega_n(2^n-1)\rho_\varepsilon^{n-2}
 \longrightarrow0.
 \end{aligned}
\]
A cutoff changing linearly between these two radii would have
nonvanishing energy in dimension two, so a different choice is needed
there.

For $n=2$, let $r_\varepsilon=2\varepsilon$ and $\rho_\varepsilon=\sqrt\varepsilon$.
For $\varepsilon<1/4$, $r_\varepsilon<\rho_\varepsilon$. Define
\[
 \chi_{i,\varepsilon}(x)=
 \begin{cases}
 0,&\abs{x-P_i}\le r_\varepsilon,\\[2pt]
 \displaystyle\frac{\log\bigl(\abs{x-P_i}/r_\varepsilon\bigr)}
 {\log(\rho_\varepsilon/r_\varepsilon)},
   &r_\varepsilon<\abs{x-P_i}<\rho_\varepsilon,\\[6pt]
 1,&\abs{x-P_i}\ge\rho_\varepsilon.
 \end{cases}
\]
The whole interface lies inside the disk of radius $\varepsilon<r_\varepsilon$.
Also,
\[
 \abs[\big]{\operatorname{supp}(1-\chi_{i,\varepsilon})}
 \le\pi\rho_\varepsilon^2=\pi\varepsilon\longrightarrow0.
\]
In the annulus, the radial derivative is
$\bigl(r\log(\rho_\varepsilon/r_\varepsilon)\bigr)^{-1}$. Polar coordinates give
\[
 \begin{aligned}
 \int_{\mathbb{R}^2}\abs{\nabla\chi_{i,\varepsilon}}^2\,\mathrm{d} x
 &=\frac{2\pi}{\bigl[\log(\rho_\varepsilon/r_\varepsilon)\bigr]^2}
      \int_{r_\varepsilon}^{\rho_\varepsilon}\frac{\,\mathrm{d} r}{r}\\
 &=\frac{2\pi}{\log(\rho_\varepsilon/r_\varepsilon)}\longrightarrow0,
 \end{aligned}
\]
since
\[
 \log(\rho_\varepsilon/r_\varepsilon)
 =\frac12\log(1/\varepsilon)-\log2\longrightarrow+\infty.
\]
This proves both assertions in \eqref{eq:cutoffprops} in every
dimension under consideration.

\emph{Step 3. Trial functions and the upper bound.}
Choose real orthonormal eigenfunctions $\phi_1,\ldots,\phi_k$ for
the first $k$ eigenvalues on $D$, each supported on one cube.
Such a choice is possible even if the two cube spectra have common
eigenvalues. On each cube, the eigenfunctions may be chosen as
products of interval eigenfunctions. Thus $\phi_\ell$ and
$\nabla\phi_\ell$ are bounded on their supporting cube.

If $\phi_\ell$ is supported on $Q_i$, define
\[
 v_{\ell,\varepsilon}(x)=
 \begin{cases}
 \chi_{i,\varepsilon}(x)\phi_\ell(x),&x\in Q_i,\\
 0,&x\in\Omega_\varepsilon\setminus Q_i.
 \end{cases}
\]
The cutoff vanishes in a neighborhood of the attachment face.
The pieces therefore have matching zero traces there, and their
union belongs to $H^1(\Omega_\varepsilon)$.

We spell out the convergence on the fixed cube. Put
$S_{i,\varepsilon}=Q_i\cap\operatorname{supp}(1-\chi_{i,\varepsilon})$ and choose a constant
$M_\ell$ such that
$\abs{\phi_\ell}+\abs{\nabla\phi_\ell}\le M_\ell$ on $Q_i$.
Since $0\le\chi_{i,\varepsilon}\le1$,
\[
 \norm[\big]{(\chi_{i,\varepsilon}-1)\phi_\ell}_{L^2(Q_i)}^2
 \le M_\ell^2\abs{S_{i,\varepsilon}}\longrightarrow0.
\]
The product rule gives
\[
 \nabla\bigl((\chi_{i,\varepsilon}-1)\phi_\ell\bigr)
 =(\chi_{i,\varepsilon}-1)\nabla\phi_\ell
   +\phi_\ell\nabla\chi_{i,\varepsilon}.
\]
Using $\abs{a+b}^2\le2\abs{a}^2+2\abs{b}^2$, we obtain
\[
 \begin{aligned}
 \norm[\Big]{\nabla\bigl((\chi_{i,\varepsilon}-1)\phi_\ell\bigr)}_{L^2(Q_i)}^2
 &\le2M_\ell^2\abs{S_{i,\varepsilon}}
   +2M_\ell^2\int_{Q_i}\abs{\nabla\chi_{i,\varepsilon}}^2\,\mathrm{d} x\\
 &\longrightarrow0
 \end{aligned}
\]
by \eqref{eq:cutoffprops}. Hence
$\chi_{i,\varepsilon}\phi_\ell\to\phi_\ell$ strongly in $H^1(Q_i)$.
The continuity of the trace on this fixed cube also yields
\[
 \norm[\Big]{\operatorname{Tr}(\chi_{i,\varepsilon}\phi_\ell)-\operatorname{Tr}\phi_\ell}_{L^2(\partial Q_i)}
 \le C_i\norm[\big]{(\chi_{i,\varepsilon}-1)\phi_\ell}_{H^1(Q_i)}
 \longrightarrow0.
\]
In particular, the boundary integrals converge. For example, the
absolute difference of the two squared trace integrals is at most
\[
 \begin{aligned}
 \norm[\Big]{\operatorname{Tr}(\chi_{i,\varepsilon}\phi_\ell)-\operatorname{Tr}\phi_\ell}_{L^2(\partial Q_i)}
\times\left(
   \norm[\big]{\operatorname{Tr}(\chi_{i,\varepsilon}\phi_\ell)}_{L^2(\partial Q_i)}
   +\norm[\big]{\operatorname{Tr}\phi_\ell}_{L^2(\partial Q_i)}\right),
 \end{aligned}
\]
which tends to zero. All trace estimates here are on the fixed
cubes; no uniform trace estimate on $\Omega_\varepsilon$ is needed.

Let $w_{\ell,\varepsilon}=v_{\ell,\varepsilon}|_D$. The preceding estimates say
that $w_{\ell,\varepsilon}\to\phi_\ell$ in $H^1(D)$, where the squared norm is the
sum of the squared $H^1$ norms on the two components.
Because $v_{\ell,\varepsilon}$ is zero in the passage and its cube trace
is zero on the attachment face, we have the exact identities
\[
 \begin{aligned}
 \langle v_{\ell,\varepsilon},v_{m,\varepsilon}\rangle_{L^2(\Omega_\varepsilon)}
 &=\langle w_{\ell,\varepsilon},w_{m,\varepsilon}\rangle_{L^2(D)},\\
 \mathfrak a_{\Omega_\varepsilon,\alpha}
       (v_{\ell,\varepsilon},v_{m,\varepsilon})
 &=\mathfrak a_{D,\alpha}(w_{\ell,\varepsilon},w_{m,\varepsilon}).
 \end{aligned}
\]
Here $\mathfrak a(\cdot,\cdot)$ denotes the symmetric bilinear form
associated with the quadratic form. In the second identity the
Robin boundary term on each attachment face is zero on both sides;
thus replacing the exterior cube faces by the full cube boundaries
introduces no extra term. Strong $H^1$ convergence and continuity
of the trace now give
\[
 \begin{aligned}
 M_{\ell m,\varepsilon}
 &:=\langle v_{\ell,\varepsilon},v_{m,\varepsilon}\rangle_{L^2(\Omega_\varepsilon)}
   \longrightarrow\delta_{\ell m},\\
 A_{\ell m,\varepsilon}
 &:=\mathfrak a_{\Omega_\varepsilon,\alpha}
       (v_{\ell,\varepsilon},v_{m,\varepsilon})
   \longrightarrow\lambda_\ell(D;\alpha)\delta_{\ell m}.
 \end{aligned}
\]
The second limit also uses the eigenfunction identity
$\mathfrak a_{D,\alpha}(\phi_\ell,\phi_m)
=\lambda_\ell(D;\alpha)\langle\phi_\ell,\phi_m\rangle$.

To obtain a bound for all trial functions at once, write
$M_\varepsilon=(M_{\ell m,\varepsilon})$, $A_\varepsilon=(A_{\ell m,\varepsilon})$, and
\[
 \Lambda=\operatorname{diag}\bigl(\lambda_1(D;\alpha),\ldots,
                       \lambda_k(D;\alpha)\bigr).
\]
Set
\[
 \theta_\varepsilon=k\max_{1\le\ell,m\le k}
 \max\left\{
   \abs{M_{\ell m,\varepsilon}-\delta_{\ell m}},
   \abs{A_{\ell m,\varepsilon}-\Lambda_{\ell m}}\right\}
 \longrightarrow0.
\]
For $b\in\mathbb{R}^k$, the inequality
$\bigl(\sum_{\ell=1}^k\abs{b_\ell}\bigr)^2\le k\abs{b}^2$
gives
\[
 \begin{aligned}
 \norm*{\sum_{\ell=1}^k b_\ell v_{\ell,\varepsilon}}_{L^2(\Omega_\varepsilon)}^2
 &=b^{\mathsf T}M_\varepsilon b
 \ge(1-\theta_\varepsilon)\abs{b}^2,\\
 \mathfrak a_{\Omega_\varepsilon,\alpha}
   \left[\sum_{\ell=1}^k b_\ell v_{\ell,\varepsilon}\right]
 &=b^{\mathsf T}A_\varepsilon b
 \le\bigl(\lambda_k(D;\alpha)+\theta_\varepsilon\bigr)\abs{b}^2.
 \end{aligned}
\]
For small $\varepsilon$, $\theta_\varepsilon<1$, so the trial functions are
linearly independent. Applying the min--max principle on
$V_\varepsilon=\operatorname{span}\{v_{1,\varepsilon},\ldots,v_{k,\varepsilon}\}$ gives
\[
 \lambda_k(\Omega_\varepsilon;\alpha)
 \le\max_{v\in V_\varepsilon\setminus\{0\}}
       \frac{\mathfrak a_{\Omega_\varepsilon,\alpha}[v]}
            {\norm{v}_{L^2(\Omega_\varepsilon)}^2}
 \le\frac{\lambda_k(D;\alpha)+\theta_\varepsilon}{1-\theta_\varepsilon}.
\]
Taking the upper limit and using \eqref{eq:liminf} proves
\eqref{eq:convergence}.

\emph{Step 4. Uniform convergence and parameter derivatives.}
Fix a compact interval $I\subset(0,+\infty)$ and choose a larger
compact interval $J\subset(0,+\infty)$ with $I\subset\operatorname{int} J$.
Write
\[
 \lambda_{j,\varepsilon}(\alpha)=\lambda_j(\Omega_\varepsilon;\alpha),
 \qquad \ell_j(\alpha)=\lambda_j(D;\alpha).
\]
For $j=1,2,3$, the convergence in \eqref{eq:convergence} is uniform
on $J$. Indeed, the min--max principle makes $\lambda_{j,\varepsilon}$
nondecreasing in $\alpha$, and the ordered cube eigenvalues
$\ell_j$ are continuous. Given $\delta>0$, partition $J$ into
finitely many intervals on each of which $\ell_j$ changes by less
than $\delta$. Once the errors at all partition points are less
than $\delta$, monotonicity bounds the error at every intermediate
point by $2\delta$.

By Lemma~\ref{lem:disconnected}, $\ell_1<\ell_2<\ell_3$ on $J$.
Continuity and compactness give
\[
 \gamma:=\min_{\alpha\in J}
 \min\bigl\{\ell_2(\alpha)-\ell_1(\alpha),
            \ell_3(\alpha)-\ell_2(\alpha)\bigr\}>0.
\]
Uniform convergence therefore preserves both gaps, with lower
bound $\gamma/2$, for sufficiently small $\varepsilon$. In particular,
$\lambda_{1,\varepsilon}$ and $\lambda_{2,\varepsilon}$ are simple throughout $J$.
For each fixed such $\varepsilon$, the Robin form has the fixed domain
$H^1(\Omega_\varepsilon)$ and depends affinely on $\alpha$; its boundary
term is bounded with respect to the $H^1$ norm by the trace theorem.
Analytic perturbation theory for forms thus makes these simple
eigenvalues analytic on $\operatorname{int} J$; see
\cite[Chapter~VII, Sections~3--4]{Kato}.

Set $F_\varepsilon=\lambda_{1,\varepsilon}$ and
$S_\varepsilon=\lambda_{1,\varepsilon}+\lambda_{2,\varepsilon}$.
The Rayleigh principle shows that $F_\varepsilon$ is concave in $\alpha$.
The variational formula for the sum of the first two eigenvalues is
\[
 S_\varepsilon(\alpha)
 =\inf_{\substack{u_1,u_2\in H^1(\Omega_\varepsilon)\\
       \langle u_i,u_j\rangle_{L^2(\Omega_\varepsilon)}=\delta_{ij}
       \ (i,j=1,2)}}
 \bigl(\mathfrak a_{\Omega_\varepsilon,\alpha}[u_1]
       +\mathfrak a_{\Omega_\varepsilon,\alpha}[u_2]\bigr).
\]
Indeed, fix $\varepsilon$ and $\alpha$, and let
$(e_m)_{m\ge1}$ be an $L^2(\Omega_\varepsilon)$-orthonormal eigenbasis with
corresponding eigenvalues $\mu_m=\lambda_m(\Omega_\varepsilon;\alpha)$.
For any admissible pair $(u_1,u_2)$, write
\[
 c_{im}=\langle u_i,e_m\rangle_{L^2(\Omega_\varepsilon)},
 \qquad p_m=\abs{c_{1m}}^2+\abs{c_{2m}}^2.
\]
Parseval's identity gives $\sum_{m\ge1}p_m=2$. Bessel's inequality,
applied to $e_m$ and the orthonormal pair $(u_1,u_2)$, gives
$0\le p_m\le\norm{e_m}_{L^2(\Omega_\varepsilon)}^2=1$.
The spectral representation of the quadratic form on its domain
$H^1(\Omega_\varepsilon)$ therefore yields
\[
 \begin{aligned}
 \mathfrak a_{\Omega_\varepsilon,\alpha}[u_1]
 +\mathfrak a_{\Omega_\varepsilon,\alpha}[u_2]
 &=\sum_{m\ge1}\mu_m p_m\\
 &\ge\mu_1p_1+\mu_2\sum_{m\ge2}p_m\\
 &=\mu_1p_1+\mu_2(2-p_1)\\
 &=\mu_1+\mu_2+(\mu_2-\mu_1)(1-p_1)\\
 &\ge\mu_1+\mu_2.
 \end{aligned}
\]
Taking $u_1=e_1$ and $u_2=e_2$ gives equality, proving the formula.
Its
right-hand side is an infimum of affine functions of $\alpha$,
so $S_\varepsilon$ is concave as well. Their uniform limits on $J$ are
$F=\ell_1$ and $S=\ell_1+\ell_2$, which are smooth by
\eqref{eq:ordering}.

We use an elementary consequence of concavity. If a differentiable
concave function $f_\varepsilon$ converges uniformly on $J$ to a function
$f\in C^1(J)$, then, for $\alpha\in I$ and sufficiently small $h>0$,
\[
 \frac{f_\varepsilon(\alpha+h)-f_\varepsilon(\alpha)}h
 \le f_\varepsilon'(\alpha)
 \le\frac{f_\varepsilon(\alpha)-f_\varepsilon(\alpha-h)}h.
\]
More explicitly, take $0<h<\operatorname{dist}(I,\partial J)$,
so that $\alpha\pm h\in J$ for every $\alpha\in I$, and put
$E_\varepsilon=\sup_{\beta\in J}\abs{f_\varepsilon(\beta)-f(\beta)}$.
Each difference quotient contains two function values, so
\[
 \begin{aligned}
 \frac{f_\varepsilon(\alpha+h)-f_\varepsilon(\alpha)}h
 &\ge\frac{f(\alpha+h)-f(\alpha)}h-\frac{2E_\varepsilon}h,\\
 \frac{f_\varepsilon(\alpha)-f_\varepsilon(\alpha-h)}h
 &\le\frac{f(\alpha)-f(\alpha-h)}h+\frac{2E_\varepsilon}h.
 \end{aligned}
\]
The fundamental theorem of calculus also gives
\[
 \begin{aligned}
 \frac{f(\alpha+h)-f(\alpha)}h
 &=\frac1h\int_0^h f'(\alpha+t)\,\mathrm{d} t
 \ge f'(\alpha)-\omega_{f'}(h),\\
 \frac{f(\alpha)-f(\alpha-h)}h
 &=\frac1h\int_0^h f'(\alpha-t)\,\mathrm{d} t
 \le f'(\alpha)+\omega_{f'}(h).
 \end{aligned}
\]
Here $\omega_{f'}$ is defined below; it tends to zero because $f'$
is uniformly continuous on the compact interval $J$.
Combining these estimates with the concavity bounds yields
\[
 -\omega_{f'}(h)-\frac{2E_\varepsilon}h
 \le f_\varepsilon'(\alpha)-f'(\alpha)
 \le\omega_{f'}(h)+\frac{2E_\varepsilon}h.
\]
Comparing these slopes with those of $f$ gives
\[
 \sup_{\alpha\in I}\abs{f_\varepsilon'(\alpha)-f'(\alpha)}
 \le \omega_{f'}(h)
    +\frac2h\sup_{\alpha\in J}\abs{f_\varepsilon(\alpha)-f(\alpha)},
\]
where
\[
 \omega_{f'}(h)
 =\sup\bigl\{\abs{f'(x)-f'(y)}:
                x,y\in J,\ \abs{x-y}\le h\bigr\}
 \longrightarrow0\quad(h\downarrow0).
\]
First letting $\varepsilon\downarrow0$ and then $h\downarrow0$ proves
uniform convergence of the derivatives on $I$. Apply this argument
to $F_\varepsilon$ and $S_\varepsilon$. Since
$\lambda_{2,\varepsilon}=S_\varepsilon-F_\varepsilon$, both eigenvalues converge in
$C^1(I)$, as claimed.

\label{rem:nonmonotone}
For each fixed $\varepsilon>0$, connectedness gives
$\mathcal{R}_{\Omega_\varepsilon}(\alpha)\to+\infty$ as
$\alpha\downarrow0$~\cite{Laugesen}. Together with
\eqref{eq:reversal}, this rules out monotonicity in either direction
on $(0,+\infty)$ for the domain chosen in
Theorem~\ref{thm:counterexample}. By Proposition~\ref{prop:convergence} and
Lemma~\ref{lem:disconnected}, the two limits do not commute:
\[
 \lim_{\alpha\downarrow0}\,
      \lim_{\varepsilon\downarrow0}\mathcal{R}_{\Omega_\varepsilon}(\alpha)=\frac1s,
 \,\,\,
 \lim_{\varepsilon\downarrow0}\,
      \lim_{\alpha\downarrow0}\mathcal{R}_{\Omega_\varepsilon}(\alpha)=+\infty.
\]
The passage width is chosen for the prescribed compact interval;
no fixed width can give strict increase on all of $(0,+\infty)$.
<!-- end remark-2 -->
<!-- remark-3: body_md -->
In the planar construction, one may take $s=9/10$,
$\alpha_-=1$ and $\alpha_+=2$.
The proof gives a positive threshold for the passage width below
which $\mathcal{R}_{\Omega_\varepsilon}'(\alpha)>0$ throughout $[1,2]$, and hence
\eqref{eq:reversal} holds.
