<!-- definition-1: body_md -->
\label{def:classical-ot}
A probability measure $\gamma\in\mathcal{P}(K\times K)$ is called a
\emph{transport plan}, or equivalently a \emph{coupling}, from $\mu$ to $\nu$
if
\[
(\pi_1)_\#\gamma=\mu,
\qquad
(\pi_2)_\#\gamma=\nu,
\]
where $T_\#\rho$ denotes the push-forward of a measure $\rho$ by a Borel map
$T$.  The set of all transport plans from $\mu$ to $\nu$ is denoted by
\begin{equation}\label{eq:transport-plans}
\Pi(\mu,\nu)
:=
\left\{
\gamma\in\mathcal{P}(K\times K):
(\pi_1)_\#\gamma=\mu,
\ (\pi_2)_\#\gamma=\nu
\right\}.
\end{equation}
The two equalities in \eqref{eq:transport-plans} are called the
\emph{marginal conditions}.  Equivalently,
\begin{equation}\label{eq:marginal-conditions}
\gamma(A\times K)=\mu(A),
\qquad
\gamma(K\times B)=\nu(B),
\end{equation}
for every pair of Borel sets $A,B\subset K$.

For a continuous cost $c:K\times K\to\mathbb{R}$, the classical Kantorovich optimal
transport value is
\begin{equation}\label{eq:classical-ot}
\mathsf{OT}_c(\mu,\nu)
:=
\inf_{\gamma\in\Pi(\mu,\nu)}
\int_{K\times K}c(x,y)\,\mathrm{d}\gamma(x,y).
\end{equation}
Any minimizer is called an \emph{optimal transport plan}.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:classical-ot-duality}
For every continuous cost $c:K\times K\to\mathbb{R}$,
\begin{equation}\label{eq:classical-ot-strong-duality}
\mathsf{OT}_c(\mu,\nu)=\mathsf{OT}_c^*(\mu,\nu).
\end{equation}
Moreover, the supremum in \eqref{eq:classical-ot-dual} is attained.  A
maximizing pair is called a pair of \emph{Kantorovich potentials}.

<!-- definition-2: body_md -->
\label{def:capacity-couplings}
The set of transport plans from $\mu$ to $\nu$ bounded by
$\widetilde\gamma$ is
\begin{equation}\label{eq:capacity-couplings}
\Pi^{\widetilde\gamma}(\mu,\nu)
:=\bigl\{\gamma\in\Pi(\mu,\nu):\gamma\le\widetilde\gamma\bigr\}.
\end{equation}
For a continuous cost $c:K\times K\to\mathbb{R}$, the capacity-constrained transport
value is
\begin{equation}\label{eq:ccot-primal}
\mathsf{CCOT}_c^{\widetilde\gamma}(\mu,\nu)
:=\inf_{\gamma\in\Pi^{\widetilde\gamma}(\mu,\nu)}
\int_{K\times K}c(x,y)\,\mathrm{d}\gamma(x,y),
\end{equation}
with the convention that the value is $+\infty$ if the admissible set is
empty.

<!-- theorem-2: body_md -->
\label{thm:capacity-feasibility}
Let $\mu,\nu\in\mathcal{P}(K)$ and
$\widetilde\gamma\in\mathcal{M}_+(K\times K)$.  Then
$\Pi^{\widetilde\gamma}(\mu,\nu)\neq\varnothing$ if and only if
\begin{equation}\label{eq:capacity-feasibility}
\mu(A)+\nu(B)
\le \widetilde\gamma(A\times B)+1
\qquad
\text{for all Borel sets }A,B\subset K.
\end{equation}

<!-- theorem-3: body_md -->
\label{thm:ccot-duality}
For every continuous cost $c:K\times K\to\mathbb{R}$,
\begin{equation}\label{eq:ccot-strong-duality}
\mathsf{CCOT}_c^{\widetilde\gamma}(\mu,\nu)
=\mathsf{CCOT}_c^{\widetilde\gamma,*}(\mu,\nu).
\end{equation}

<!-- remark-1: body_md -->
\label{rem:capacity-sign}
The condition $w\le0$ is forced by the upper-bound constraint.  Indeed, if
$\gamma\le\widetilde\gamma$, then
$\int w\,\mathrm{d}\widetilde\gamma\le\int w\,\mathrm{d}\gamma$ for every nonpositive $w$.
Thus $w$ lowers the dual value precisely on regions in which the prescribed
capacity may become active.

<!-- definition-3: body_md -->
\label{def:classical-barycenter}
The classical Wasserstein barycenter value associated with the measures
$\nu_1,\ldots,\nu_p$ and weights $\lambda_1,\ldots,\lambda_p$ is
\begin{equation}\label{eq:classical-barycenter}
\mathsf{BC}
:=\inf_{\nu\in\mathcal{P}(K)}J_0(\nu),
\qquad
J_0(\nu)
:=\sum_{i=1}^p\lambda_i\mathsf{OT}_{c_i}(\nu_i,\nu).
\end{equation}
Equivalently,
\begin{equation}\label{eq:classical-barycenter-expanded}
\mathsf{BC}
=
\inf_{\nu\in\mathcal{P}(K)}
\sum_{i=1}^p\lambda_i
\inf_{\gamma_i\in\Pi(\nu_i,\nu)}
\int_{K\times K}\frac12\lvert x-y\rvert^2\,\mathrm{d}\gamma_i(x,y).
\end{equation}
Any minimizer is called a \emph{Wasserstein barycenter} of
$\nu_1,\ldots,\nu_p$ with weights $\lambda_1,\ldots,\lambda_p$.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:ccbc}
The feasible barycenter set is
\begin{equation}\label{eq:feasible-barycenters}
\mathcal{P}_{\widetilde\gamma}
:=\left\{\nu\in\mathcal{P}(K):
\Pi^{\widetilde\gamma_i}(\nu_i,\nu)\neq\varnothing
\text{ for every }i=1,\ldots,p\right\}.
\end{equation}
For $\nu\in\mathcal{P}(K)$, define
\begin{equation}\label{eq:pairwise-cc-cost}
\widetilde W_i^2(\nu_i,\nu)
:=\mathsf{CCOT}_{c_i}^{\widetilde\gamma_i}(\nu_i,\nu).
\end{equation}
The capacity-constrained barycenter value is
\begin{equation}\label{eq:ccbc-primal}
\mathsf{CCBC}
:=\inf_{\nu\in\mathcal{P}_{\widetilde\gamma}}
J(\nu),
\qquad
J(\nu):=\sum_{i=1}^p\lambda_i\widetilde W_i^2(\nu_i,\nu).
\end{equation}
Any minimizer is called a capacity-constrained barycenter of
$\nu_1,\ldots,\nu_p$ associated with the capacities
$\widetilde\gamma_1,\ldots,\widetilde\gamma_p$.

<!-- remark-2: body_md -->
\label{rem:simple-feasible-family}
The nonempty condition is not restrictive in principle.  For example, if $\xi\in\mathcal{P}(K)$ and
$\widetilde\gamma_i=\nu_i\otimes\xi$, then $\xi\in\mathcal{P}_{\widetilde\gamma}$ because

<!-- proposition-1: body_md -->
\label{prop:basic-properties}
The capacity-constrained barycenter problem has the following properties.
\begin{enumerate}[label=\textup{(\roman*)}]
\item Every $\nu\in\mathcal{P}_{\widetilde\gamma}$ satisfies
\begin{equation}\label{eq:support-control}
\operatorname{spt}\nu\subset
\bigcap_{i=1}^p
\overline{\operatorname{Proj}_y(\operatorname{spt}\widetilde\gamma_i)}.
\end{equation}
\item The set $\mathcal{P}_{\widetilde\gamma}$ is convex.
\item For each $i$, the map
$\nu\mapsto\widetilde W_i^2(\nu_i,\nu)$ is convex on
$\mathcal{P}_{\widetilde\gamma}$; hence $J$ is convex.
\item If $\widetilde\gamma_i\le\widehat\gamma_i$ for every $i$, then
\begin{equation}\label{eq:capacity-monotonicity}
\mathcal{P}_{\widetilde\gamma}\subset\mathcal{P}_{\widehat\gamma}
\qquad\text{and}\qquad
\mathsf{CCBC}(\widehat\gamma_1,\ldots,\widehat\gamma_p)
\le
\mathsf{CCBC}(\widetilde\gamma_1,\ldots,\widetilde\gamma_p).
\end{equation}
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
If $\gamma_i\in\Pi^{\widetilde\gamma_i}(\nu_i,\nu)$, then the second marginal of $\gamma_i$ is supported on the closure of the second projection of $\operatorname{spt}\widetilde\gamma_i$, proving \eqref{eq:support-control}.

Let $\nu^0,\nu^1\in\mathcal{P}_{\widetilde\gamma}$ and $t\in[0,1]$.  Choose
$\gamma_i^k\in\Pi^{\widetilde\gamma_i}(\nu_i,\nu^k)$ for $k=0,1$.  Then
$t\gamma_i^0+(1-t)\gamma_i^1$ has first marginal $\nu_i$, second marginal
$t\nu^0+(1-t)\nu^1$, and is still dominated by $\widetilde\gamma_i$.  This proves convexity of the feasible set.  Taking $\gamma_i^0$ and $\gamma_i^1$ optimal, or arbitrarily close to optimal, and using linearity of the cost gives convexity of each constrained cost and of $J$.

Finally, increasing a capacity can only enlarge the corresponding admissible set of plans.  The inclusion of feasible barycenter sets and the inequality of optimal values follow immediately.

<!-- lemma-1: body_md -->
\label{lem:cc-lsc}
For each $i$, the map
$\nu\mapsto\widetilde W_i^2(\nu_i,\nu)$ is weakly lower semicontinuous on $\mathcal{P}(K)$, with the value $+\infty$ outside its feasible domain.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Let $\nu_n\rightharpoonup\nu$ and suppose that
$\liminf_n\widetilde W_i^2(\nu_i,\nu_n)<\infty$.  Passing to a subsequence, we may assume that the liminf is a limit.  Choose optimal plans
$\gamma_{i,n}\in\Pi^{\widetilde\gamma_i}(\nu_i,\nu_n)$.  Since $K\times K$ is compact, a further subsequence converges weakly to some
$\gamma_i\in\mathcal{P}(K\times K)$.  The marginal constraints pass to the limit, so
$\gamma_i\in\Pi(\nu_i,\nu)$.

For every nonnegative $f\in C(K\times K)$,
\[
\int f\,\mathrm{d}\gamma_i
=\lim_n\int f\,\mathrm{d}\gamma_{i,n}
\le\int f\,\mathrm{d}\widetilde\gamma_i.
\]
Hence $\gamma_i\le\widetilde\gamma_i$.  Since $c_i$ is continuous,
\[
\widetilde W_i^2(\nu_i,\nu)
\le\int c_i\,\mathrm{d}\gamma_i
=\lim_n\int c_i\,\mathrm{d}\gamma_{i,n}
=\liminf_n\widetilde W_i^2(\nu_i,\nu_n).
\]
<!-- end proof-2 -->
<!-- proposition-2: body_md -->
\label{prop:feasible-compact}
The set $\mathcal{P}_{\widetilde\gamma}$ is weakly compact in $\mathcal{P}(K)$.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
Since $K$ is compact, it is enough to prove that the feasible set is weakly closed.  Let
$\nu_n\in\mathcal{P}_{\widetilde\gamma}$ and $\nu_n\rightharpoonup\nu$.  For each $i$ and $n$, choose
$\gamma_{i,n}\in\Pi^{\widetilde\gamma_i}(\nu_i,\nu_n)$.  By compactness and a diagonal extraction, assume that
$\gamma_{i,n}\rightharpoonup\gamma_i$ for every $i$.  As in the proof of
\Cref{lem:cc-lsc}, the limiting plan belongs to
$\Pi^{\widetilde\gamma_i}(\nu_i,\nu)$.  Therefore $\nu\in\mathcal{P}_{\widetilde\gamma}$.
<!-- end proof-3 -->
<!-- theorem-4: body_md -->
\label{thm:ccbc-existence}
If $\mathcal{P}_{\widetilde\gamma}\neq\varnothing$, then the problem \eqref{eq:ccbc-primal} has a minimizer.
<!-- end theorem-4 -->
<!-- proof-4: body_md -->
Let $\{\nu_n\}\subset\mathcal{P}_{\widetilde\gamma}$ be a minimizing sequence.  By

Thus $\nu^*$ is a minimizer.

<!-- theorem-5: body_md -->
\label{thm:classical-barycenter-duality}
The classical barycenter value in \eqref{eq:classical-barycenter} satisfies
\begin{equation}\label{eq:classical-barycenter-dual}
\mathsf{BC}
=
\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_{\mathrm{cl}}}
\sum_{i=1}^p\lambda_i\int_K\phi_i\,\mathrm{d}\nu_i.
\end{equation}
For the quadratic costs \eqref{eq:quadratic-cost}, the supremum is attained.
<!-- end theorem-5 -->
<!-- theorem-6: body_md -->
\label{thm:ccbc-duality}
If \(\mathcal{P}_{\widetilde\gamma}\neq\varnothing\), then
\begin{align}\label{eq:ccbc-dual-positive-part}
\mathsf{CCBC}
=\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\left[
\int_K\phi_i\,\mathrm{d}\nu_i
-\int_{K\times K}
\left[\phi_i(x)+\psi_i(y)-c_i(x,y)\right]_{+}
\,\mathrm{d}\widetilde\gamma_i(x,y)
\right].
\end{align}
<!-- end theorem-6 -->
<!-- proof-5: body_md -->
Every measure \(\gamma_i\le\widetilde\gamma_i\) has the form
\(\gamma_i=q_i\widetilde\gamma_i\) for some
\[
q_i\in\mathcal D_i
:=\{q\in L^\infty(K\times K,\widetilde\gamma_i):0\le q\le1\}.
\]
Endow \(\mathcal D_i\) with the weak-* topology of \(L^\infty\).  It is convex and weak-* compact.  A tuple \(\boldsymbol q=(q_1,\ldots,q_p)\) represents an admissible barycenter tuple precisely when the first marginal of \(q_i\widetilde\gamma_i\) is \(\nu_i\) for every \(i\), and the second marginals of all the measures \(q_i\widetilde\gamma_i\) coincide.

These linear constraints have the indicator representation
\begin{align}\label{eq:barycentric-constraint-indicator}
\mathcal I(\boldsymbol q)
=\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\left[
\int_K\phi_i\,\mathrm{d}\nu_i
-\int_{K\times K}
\bigl(\phi_i(x)+\psi_i(y)\bigr)q_i(x,y)
\,\mathrm{d}\widetilde\gamma_i(x,y)
\right],
\end{align}
where \(\mathcal I(\boldsymbol q)=0\) for an admissible tuple and
\(\mathcal I(\boldsymbol q)=+\infty\) otherwise.  Indeed, the expression vanishes on every admissible tuple.  If a first marginal is incorrect, a source potential \(\phi_i\) separates the two measures; if the second marginals do not coincide, continuous functions \(\psi_i\) satisfying the weighted zero-sum condition separate them.  Scaling the separating functions gives \(+\infty\).

Consequently,
\begin{align*}
\mathsf{CCBC}
&=\min_{\boldsymbol q\in\prod_i\mathcal D_i}
\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\left[
\int_K\phi_i\,\mathrm{d}\nu_i
+\int_{K\times K}
\bigl(c_i-\phi_i-\psi_i\bigr)q_i
\,\mathrm{d}\widetilde\gamma_i
\right].
\end{align*}
The set \(\prod_i\mathcal D_i\) is compact and convex, the displayed functional is affine and weak-* continuous in \(\boldsymbol q\), and it is affine in the potentials.  Sion's minimax theorem \cite{sion1958general} therefore permits the interchange of minimum and supremum.  Finally, the minimization in \(q_i\) is pointwise, and
\[
\min_{0\le q\le1}(c_i-\phi_i-\psi_i)q
=-\left[\phi_i+\psi_i-c_i\right]_{+}.
\]
Substitution gives \eqref{eq:ccbc-dual-positive-part}.
<!-- end proof-5 -->
<!-- corollary-1: body_md -->
\label{cor:ccbc-capacity-multiplier}
Let \(\widehat\mathcal{A}_0\) be the set of
\(\phi_i,\psi_i\in C(K)\) and
\(w_i\in C(K\times K)\) satisfying
\begin{equation}\label{eq:unreg-dual-constraints}
w_i\le0,
\qquad
\phi_i(x)+\psi_i(y)+w_i(x,y)\le c_i(x,y),
\qquad
\sum_{i=1}^p\lambda_i\psi_i=0.
\end{equation}
Then
\begin{equation}\label{eq:ccbc-dual-symmetric}
\mathsf{CCBC}
=\sup_{(\boldsymbol\phi,\boldsymbol\psi,\boldsymbol w)\in\widehat\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\left(
\int_K\phi_i\,\mathrm{d}\nu_i
+\int_{K\times K}w_i\,\mathrm{d}\widetilde\gamma_i
\right).
\end{equation}
<!-- end corollary-1 -->
<!-- proof-6: body_md -->
For fixed \(\phi_i\) and \(\psi_i\), the largest admissible capacity potential is
\[
w_i(x,y)=-\left[\phi_i(x)+\psi_i(y)-c_i(x,y)\right]_{+}.
\]
It is continuous, nonpositive, and satisfies the pointwise constraint in
\eqref{eq:unreg-dual-constraints}.  Substituting it into
\eqref{eq:ccbc-dual-symmetric} yields
\eqref{eq:ccbc-dual-positive-part}; every other admissible \(w_i\) gives a no larger value.
<!-- end proof-6 -->
<!-- corollary-2: body_md -->
\label{cor:ccbc-inf-transform}
Let \(f_i:=\lambda_i\psi_i\) and \(\omega_i:=\lambda_i w_i\).  Then
\begin{align}\label{eq:ccbc-dual-inf-transform}
\mathsf{CCBC}
=\sup\Bigg\{&\sum_{i=1}^p
\int_K
\inf_{z\in K}
\left\{\lambda_i c_i(x,z)-f_i(z)-\omega_i(x,z)\right\}
\,\mathrm{d}\nu_i(x)\nonumber\\
&\quad+
\sum_{i=1}^p\int_{K\times K}\omega_i\,\mathrm{d}\widetilde\gamma_i:
\quad
f_i\in C(K),\quad
\omega_i\in C(K\times K),\quad
\omega_i\le0,
\quad
\sum_{i=1}^pf_i=0
\Bigg\}.
\end{align}
<!-- end corollary-2 -->
<!-- proof-7: body_md -->
For fixed \(\psi_i\) and \(w_i\), the largest admissible source potential is
\[
\phi_i(x)
=\inf_{z\in K}\{c_i(x,z)-\psi_i(z)-w_i(x,z)\}.
\]
Substituting this expression into \eqref{eq:ccbc-dual-symmetric} and multiplying the \(i\)th potentials by \(\lambda_i\) gives \eqref{eq:ccbc-dual-inf-transform}.

<!-- remark-3: body_md -->
\label{rem:classical-special-case}
Setting $w_i\equiv0$ in the capacity-multiplier constraints
\eqref{eq:unreg-dual-constraints} recovers the classical admissible potential
class \eqref{eq:classical-barycenter-potential-class}.  The objective in
\eqref{eq:ccbc-dual-symmetric} then reduces to the classical barycenter dual
objective \eqref{eq:classical-barycenter-dual}.  Thus the functions $w_i$
measure precisely the departure from the unconstrained barycenter geometry.

Moreover, if a classical optimal barycenter admits optimal transport plans
satisfying all prescribed capacity bounds, then it is feasible for the
capacity-constrained problem and the two optimal values coincide.

<!-- definition-5: body_md -->
\label{def:relative-entropy}
For $\gamma,\xi\in\mathcal{P}(K\times K)$, the relative entropy of $\gamma$ with
respect to $\xi$ is
\begin{equation}\label{eq:kl-definition}
\operatorname{KL}(\gamma\mid\xi)
:=
\begin{cases}
\displaystyle
\int_{K\times K}
\log\!\left(\frac{\,\mathrm{d}\gamma}{\,\mathrm{d}\xi}\right)\,\mathrm{d}\gamma,
&\gamma\ll\xi,\\[0.8em]
+\infty,&\text{otherwise}.
\end{cases}
\end{equation}
If $\gamma=q\xi$, then
\begin{equation}\label{eq:entropy-shift}
\operatorname{KL}(\gamma\mid\xi)
=1+\int_{K\times K}\rho(q)\,\mathrm{d}\xi,
\qquad
\rho(q):=q\log q-q,
\end{equation}
where $0\log0:=0$.
<!-- end definition-5 -->
<!-- definition-6: body_md -->
\label{def:classical-entropic-ot}
Let $\mu,\nu\in\mathcal{P}(K)$, let $c\in C(K\times K)$, and let
$\epsilon>0$.  The entropy-regularized optimal transport value is
\begin{equation}\label{eq:classical-entropic-ot}
\mathsf{EOT}_{\epsilon,c}(\mu,\nu)
:=
\inf_{\gamma\in\Pi(\mu,\nu)}
\left\{
\int_{K\times K}c\,\mathrm{d}\gamma
+\epsilon\operatorname{KL}(\gamma\mid\mu\otimes\nu)
\right\}.
\end{equation}
<!-- end definition-6 -->
<!-- theorem-7: body_md -->
\label{thm:classical-entropic-ot-duality}
The problem \eqref{eq:classical-entropic-ot} has a unique minimizer and
\begin{align}\label{eq:classical-entropic-ot-dual}
\mathsf{EOT}_{\epsilon,c}(\mu,\nu)
=\epsilon+
\sup_{u,v\in C(K)}
\Bigg\{
&\int_Ku\,\mathrm{d}\mu+\int_Kv\,\mathrm{d}\nu\nonumber\\
&-\epsilon\int_{K\times K}
\exp\!\left(\frac{u(x)+v(y)-c(x,y)}{\epsilon}\right)
\,\mathrm{d}(\mu\otimes\nu)(x,y)
\Bigg\}.
\end{align}

\label{def:classical-entropic-barycenter}
For $\epsilon>0$, define

\mathsf{EBC}_{\epsilon,\eta}

<!-- theorem-8: body_md -->
\label{thm:classical-entropic-barycenter-duality}
The problem \eqref{eq:classical-entropic-barycenter} has a unique optimal
tuple of plans and hence a unique barycenter.  Moreover,
\begin{align}\label{eq:classical-entropic-barycenter-dual}
\mathsf{EBC}_{\epsilon,\eta}
=\epsilon+
\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\Bigg[
\int_K\phi_i\,\mathrm{d}\nu_i
-\epsilon\int_{K\times K}
\exp\!\left(
\frac{\phi_i(x)+\psi_i(y)-c_i(x,y)}{\epsilon}
\right)
\,\mathrm{d}\xi_i(x,y)
\Bigg].
\end{align}
Whenever the dual supremum is attained, the optimal plans satisfy
\begin{equation}\label{eq:classical-barycenter-gibbs-relation}
\frac{\,\mathrm{d}\gamma_{i,\epsilon}}{\,\mathrm{d}\xi_i}(x,y)
=
\exp\!\left(
\frac{\phi_i^*(x)+\psi_i^*(y)-c_i(x,y)}{\epsilon}
\right)
\qquad \xi_i\text{-a.e.}
\end{equation}

<!-- proof-8: body_md -->

fixed-support regularized barycenter theorem in \cite{li2020continuous}.  The
objective used there contains $\epsilon\rho$; the present objective differs by
the constant $\epsilon$ because of \eqref{eq:entropy-shift}.  The feasible set of plan tuples is convex, and the
sum of the fixed-reference relative entropies is strictly convex.  Hence the
optimal tuple, and therefore its common second marginal, is unique.  The
relation \eqref{eq:classical-barycenter-gibbs-relation} is the corresponding
Fenchel equality whenever dual optimizers exist.

\mathsf{ECCOT}_{\epsilon,c}^{h,\xi}(\mu,\nu)

<!-- definition-9: body_md -->
\label{def:capped-conjugate}
For $\epsilon>0$, $h\ge0$, and $s\in\mathbb{R}$, define
\begin{equation}\label{eq:capped-conjugate-definition}
\Phi_{\epsilon,h}(s)
:=
\sup_{0\le q\le h}
\bigl\{sq-\epsilon\rho(q)\bigr\},
\qquad
\rho(q)=q\log q-q.
\end{equation}
Equivalently,
$\Phi_{\epsilon,h}=(\epsilon\rho+I_{[0,h]})^*$, where
$I_{[0,h]}$ is $0$ on $[0,h]$ and $+\infty$ outside this interval.
<!-- end definition-9 -->
<!-- proposition-3: body_md -->
\label{prop:capped-conjugate}
If $h=0$, then $\Phi_{\epsilon,0}(s)=0$.  If $h>0$, then
\begin{equation}\label{eq:capped-conjugate-explicit}
\Phi_{\epsilon,h}(s)
=
\begin{cases}
\epsilon\exp(s/\epsilon),
& s\le\epsilon\log h,\\[0.4em]
hs-\epsilon h\log h+\epsilon h,
& s>\epsilon\log h.
\end{cases}
\end{equation}
Moreover, $\Phi_{\epsilon,h}$ is convex and continuously differentiable,
with
\begin{equation}\label{eq:capped-conjugate-derivative}
\frac{\,\mathrm{d}}{\,\mathrm{d} s}\Phi_{\epsilon,h}(s)
=\min\bigl\{h,\exp(s/\epsilon)\bigr\}.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-9: body_md -->
For $q>0$, the derivative of
$q\mapsto sq-\epsilon(q\log q-q)$ is $s-\epsilon\log q$.
The unconstrained maximizer is $q=\exp(s/\epsilon)$.  If this value does not
exceed $h$, substitution gives the first branch in
\eqref{eq:capped-conjugate-explicit}; otherwise the maximum is attained at
$q=h$.  The values and first derivatives agree at $s=\epsilon\log h$.
<!-- end proof-9 -->
<!-- theorem-9: body_md -->
\label{thm:eccot-duality}
Assume that the admissible set in \eqref{eq:eccot-primal} is nonempty.  Then
the infimum is attained at a unique plan and
\begin{align}\label{eq:eccot-dual}
\mathsf{ECCOT}_{\epsilon,c}^{h,\xi}(\mu,\nu)
=\epsilon+
\sup_{u,v\in C(K)}
\Bigg\{
&\int_Ku\,\mathrm{d}\mu+\int_Kv\,\mathrm{d}\nu\nonumber\\
&-\int_{K\times K}
\Phi_{\epsilon,h(x,y)}
\bigl(u(x)+v(y)-c(x,y)\bigr)
\,\mathrm{d}\xi(x,y)
\Bigg\}.
\end{align}
<!-- end theorem-9 -->
<!-- proof-10: body_md -->
Write $\gamma=q\xi$.  The feasible density set is weak-* compact, the cost is
weak-* continuous, and the entropy is weak-* lower semicontinuous and strictly
convex.  Thus the primal problem has a unique minimizer.  To derive the dual,
the marginal constraints have the indicator
representation
\[
\sup_{u,v\in C(K)}
\left\{
\int_Ku\,\mathrm{d}\mu+\int_Kv\,\mathrm{d}\nu
-\int_{K\times K}(u(x)+v(y))q(x,y)\,\mathrm{d}\xi(x,y)
\right\}.
\]
The order interval $\{q\in L^\infty(\xi):0\le q\le h\}$ is weak-* compact.
The map $q\mapsto\int\rho(q)\,\mathrm{d}\xi$ is weak-* lower semicontinuous on this
interval, since
\[
\int\rho(q)\,\mathrm{d}\xi
=
\sup_{a\in L^\infty(\xi)}
\left\{
\int aq\,\mathrm{d}\xi-\int e^a\,\mathrm{d}\xi
\right\}.
\]
Sion's minimax theorem therefore permits interchange of the minimum and the
supremum.  The pointwise minimization in $q$ is exactly the negative of
\eqref{eq:capped-conjugate-definition}.  The minimizing density
$\min\{h,\exp((u+v-c)/\epsilon)\}$ is measurable, so the pointwise minimum may
be integrated.  Restoring the constant in
\eqref{eq:entropy-shift} gives \eqref{eq:eccot-dual}.

<!-- definition-10: body_md -->
\label{def:eccbc}
For $\epsilon>0$, define
\begin{align}\label{eq:eccbc-primal}
\mathsf{ECCBC}_{\epsilon,\eta}
&:=
\inf_{\nu\in\mathcal{P}(K)}
\sum_{i=1}^p\lambda_i
\mathsf{ECCOT}_{\epsilon,c_i}^{h_i,\xi_i}(\nu_i,\nu)\nonumber\\
&=
\inf_{\substack{\nu\in\mathcal{P}(K)\\
\gamma_i\in\Pi^{\widetilde\gamma_i}(\nu_i,\nu)}}
\sum_{i=1}^p\lambda_i
\left[
\int_{K\times K}c_i\,\mathrm{d}\gamma_i
+\epsilon\operatorname{KL}(\gamma_i\mid\xi_i)
\right].
\end{align}
A minimizer $\nu_\epsilon$ is called an entropy-regularized
capacity-constrained barycenter.

<!-- proof-11: body_md -->
The feasible set of plan tuples is weakly compact: the domination
$0\le\gamma_i\le\widetilde\gamma_i$ gives tightness, and the domination and
marginal conditions are closed under weak convergence.  The transport costs
are weakly continuous because the $c_i$ are continuous on the compact set
$K\times K$, while relative entropy with respect to a fixed reference is
weakly lower semicontinuous \cite[Lemma~1.4.3]{dupuis2011weak}.  Hence a
minimizer exists.  The feasible set is
convex and the weighted sum of the fixed-reference relative entropies is
strictly convex, so the optimal tuple is unique.  Its common second marginal
is therefore unique.

<!-- theorem-11: body_md -->
\label{thm:eccbc-duality}
Suppose \Cref{ass:capacity-densities} holds and the admissible set in
\eqref{eq:eccbc-primal} is nonempty.  Then
\begin{align}\label{eq:eccbc-dual-capped}
\mathsf{ECCBC}_{\epsilon,\eta}
=\epsilon+
\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\Bigg[
\int_K\phi_i\,\mathrm{d}\nu_i
-\int_{K\times K}
\Phi_{\epsilon,h_i(x,y)}
\bigl(\phi_i(x)+\psi_i(y)-c_i(x,y)\bigr)
\,\mathrm{d}\xi_i(x,y)
\Bigg].
\end{align}
<!-- end theorem-11 -->
<!-- proof-12: body_md -->
For each $i$, set
\[
\mathcal D_i^h
:=
\{q\in L^\infty(K\times K,\xi_i):0\le q\le h_i\}.
\]
This is a convex weak-* compact subset of $L^\infty(\xi_i)$.  For
$\boldsymbol q\in\prod_i\mathcal D_i^h$, the marginal and common-target
constraints have the indicator representation
\begin{align}\label{eq:entropic-barycentric-constraint-indicator}
\mathcal I_\xi(\boldsymbol q)
:=\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\left[
\int_K\phi_i\,\mathrm{d}\nu_i
-\int_{K\times K}
(\phi_i(x)+\psi_i(y))q_i(x,y)\,\mathrm{d}\xi_i(x,y)
\right].
\end{align}
The same separation argument as in
\eqref{eq:barycentric-constraint-indicator} shows that
$\mathcal I_\xi$ is zero precisely for tuples with the prescribed source
marginals and a common second marginal, and is $+\infty$ otherwise.  Applying
\eqref{eq:entropy-shift} on the feasible tuples therefore gives
\begin{align*}
\mathsf{ECCBC}_{\epsilon,\eta}-\epsilon
=
\min_{\boldsymbol q\in\prod_i\mathcal D_i^h}
\sup_{(\boldsymbol\phi,\boldsymbol\psi)\in\mathcal{A}_0}
\sum_{i=1}^p\lambda_i
\Bigg[
\int_K\phi_i\,\mathrm{d}\nu_i
+\int_{K\times K}
\Bigl((c_i-\phi_i-\psi_i)q_i+\epsilon\rho(q_i)\Bigr)
\,\mathrm{d}\xi_i
\Bigg].
\end{align*}
For fixed potentials, the functional is convex and weak-* lower
semicontinuous in $\boldsymbol q$.  The lower semicontinuity follows, for
example, from the variational representation
\[
\int\rho(q_i)\,\mathrm{d}\xi_i
=
\sup_{a\in L^\infty(\xi_i)}
\left\{
\int aq_i\,\mathrm{d}\xi_i-\int e^a\,\mathrm{d}\xi_i
\right\}.
\]
For fixed $\boldsymbol q$, the functional is affine and continuous in the
potentials.  Sion's minimax theorem therefore gives
\[
\min_{\boldsymbol q}\sup_{(\boldsymbol\phi,\boldsymbol\psi)}
=
\sup_{(\boldsymbol\phi,\boldsymbol\psi)}\min_{\boldsymbol q}.
\]
The inner minimization is pointwise, and
\[
\min_{0\le q\le h_i}
\left\{(c_i-\phi_i-\psi_i)q+\epsilon\rho(q)\right\}
=
-\Phi_{\epsilon,h_i}(\phi_i+\psi_i-c_i).
\]
The measurable minimizer is
$\min\{h_i,\exp((\phi_i+\psi_i-c_i)/\epsilon)\}$, which justifies integrating
the pointwise minimum.  Restoring the additive constant proves \eqref{eq:eccbc-dual-capped}.

<!-- remark-6: body_md -->
\label{rem:continuous-dual-attainment}
\Cref{thm:eccbc-duality} proves equality of the primal and dual values.  It
does not by itself prove that the supremum is attained by continuous
potentials, and no such continuous-attainment claim is made here.  In a
finite-dimensional discrete problem, standard convex duality gives dual
attainment under a relative-interior feasibility condition for the box
$0\le q_i\le h_i$.

<!-- theorem-12: body_md -->
\label{thm:eccbc-kkt}
Assume the hypotheses of \Cref{thm:eccbc-duality}, and suppose that
$(\phi_i^*,\psi_i^*)_{i=1}^p$ is a dual maximizer in
\eqref{eq:eccbc-dual-capped}.  Let
$(\gamma_{i,\epsilon})_{i=1}^p$ be the unique primal optimizer and write
$\gamma_{i,\epsilon}=q_{i,\epsilon}\xi_i$ and
$\nu_\epsilon=r_\epsilon\eta$.  Then
\begin{equation}\label{eq:clipped-gibbs-density}
q_{i,\epsilon}(x,y)
=
\min\left\{
 h_i(x,y),
 \exp\!\left(
 \frac{\phi_i^*(x)+\psi_i^*(y)-c_i(x,y)}{\epsilon}
 \right)
\right\}
\qquad\xi_i\text{-a.e.}
\end{equation}
and
\begin{align}
\int_Kq_{i,\epsilon}(x,y)\,\mathrm{d}\eta(y)&=1
&&\nu_i\text{-a.e.},
\label{eq:clipped-schrodinger-row}\\
\int_Kq_{i,\epsilon}(x,y)\,\mathrm{d}\nu_i(x)&=r_\epsilon(y)
&&\eta\text{-a.e.},
\label{eq:clipped-schrodinger-column}\\
\sum_{i=1}^p\lambda_i\psi_i^*(y)&=0
&&\text{for every }y\in K.
\label{eq:clipped-schrodinger-barycenter}
\end{align}
On the set $E_i:=\{h_i>0\}$, define the measurable capacity field
\begin{equation}\label{eq:optimal-capacity-field}
w_i^*(x,y)
:=
\min\left\{
0,
\epsilon\log h_i(x,y)
-\phi_i^*(x)-\psi_i^*(y)+c_i(x,y)
\right\}.
\end{equation}
Then
\begin{equation}\label{eq:capacity-complementarity}
w_i^*\le0,
\qquad
q_{i,\epsilon}\le h_i,
\qquad
w_i^*(h_i-q_{i,\epsilon})=0
\quad\xi_i\text{-a.e. on }E_i.
\end{equation}
On $\{h_i=0\}$, one has $q_{i,\epsilon}=0$.

<!-- remark-7: body_md -->
\label{rem:capacity-saturation}
On $E_i=\{h_i>0\}$, where
$\phi_i^*(x)+\psi_i^*(y)-c_i(x,y)<\epsilon\log h_i(x,y)$,
the capacity is inactive and the density has the usual Gibbs form.  At or
above the threshold, the density equals $h_i$ and the capacity is saturated.
On $\{h_i=0\}$, the density vanishes.  This clipped relation suggests a
capacity-aware scaling procedure, but no iterative algorithm or convergence
theorem is proved in this paper.

<!-- proposition-4: body_md -->
\label{prop:dual-penalty-limit}
For every $h\ge0$ and $s\in\mathbb{R}$,
\begin{equation}\label{eq:dual-penalty-limit}
\lim_{\epsilon\downarrow0}\Phi_{\epsilon,h}(s)
=h\left[s\right]_{+}.
\end{equation}
Thus the capped-exponential penalty is a continuously differentiable
approximation of the positive-part capacity penalty.

<!-- theorem-13: body_md -->
\label{thm:eccbc-zero-limit}
Assume \Cref{ass:capacity-densities} and that the common feasible set is
nonempty.  There exists a unique tuple
$\overline{\boldsymbol\gamma}
=(\overline\gamma_1,\ldots,\overline\gamma_p)\in\mathcal S_0$
minimizing $\mathcal H$ over $\mathcal S_0$.  If $\overline\nu$ is its
common second marginal, then
\begin{equation}\label{eq:value-rate}
0
\le
\mathsf{ECCBC}_{\epsilon,\eta}-\mathsf{CCBC}
\le
\epsilon\mathcal H(\overline{\boldsymbol\gamma})
\qquad\text{for every }\epsilon>0,
\end{equation}
and, as $\epsilon\downarrow0$,
\begin{equation}\label{eq:entropy-selected-convergence}
\gamma_{i,\epsilon}\rightharpoonup\overline\gamma_i
\quad\text{for every }i,
\qquad
\nu_\epsilon\rightharpoonup\overline\nu.
\end{equation}
In particular, the whole regularized family converges to the unique
entropy-minimizing unregularized optimal tuple.
<!-- end theorem-13 -->
<!-- proof-16: body_md -->
The feasible set of plan tuples is weakly compact and convex.  Since each
$c_i$ is continuous, $\mathcal S_0$ is a nonempty compact convex subset of
that feasible set.  Moreover, $0\le\gamma_i\le h_i\xi_i$ with
$h_i\in L^\infty(\xi_i)$, so $\mathcal H$ is finite on the feasible set.
It is weakly lower semicontinuous and strictly convex on plan tuples.
Therefore it has a unique minimizer $\overline{\boldsymbol\gamma}$ on $\mathcal S_0$.

Let
$\boldsymbol\gamma_\epsilon
=(\gamma_{1,\epsilon},\ldots,\gamma_{p,\epsilon})$
be the regularized optimizer and write
\[
\mathcal C(\boldsymbol\gamma)
:=
\sum_{i=1}^p\lambda_i\int_{K\times K}c_i\,\mathrm{d}\gamma_i.
\]
Optimality gives
\[
\mathcal C(\boldsymbol\gamma_\epsilon)
+\epsilon\mathcal H(\boldsymbol\gamma_\epsilon)
\le
\mathcal C(\overline{\boldsymbol\gamma})
+\epsilon\mathcal H(\overline{\boldsymbol\gamma})
=
\mathsf{CCBC}+\epsilon\mathcal H(\overline{\boldsymbol\gamma}).
\]
Because relative entropy between probability measures is nonnegative,
$\mathsf{ECCBC}_{\epsilon,\eta}\ge\mathsf{CCBC}$.  In addition,
$\mathcal C(\boldsymbol\gamma_\epsilon)\ge\mathsf{CCBC}$ in the preceding
optimality inequality, which gives the upper bound in \eqref{eq:value-rate}
and also
\begin{equation}\label{eq:entropy-bound-regularized-plans}
\mathcal H(\boldsymbol\gamma_\epsilon)
\le
\mathcal H(\overline{\boldsymbol\gamma}).
\end{equation}

Let $\epsilon_n\downarrow0$.  Compactness gives a subsequence, not relabeled,
with $\boldsymbol\gamma_{\epsilon_n}\rightharpoonup\boldsymbol\gamma^*$.
The limiting tuple is feasible.  By \eqref{eq:value-rate} and continuity of
the transport costs,
$\mathcal C(\boldsymbol\gamma^*)=\mathsf{CCBC}$, so
$\boldsymbol\gamma^*\in\mathcal S_0$.  Lower semicontinuity of
$\mathcal H$ and \eqref{eq:entropy-bound-regularized-plans} yield
\[
\mathcal H(\boldsymbol\gamma^*)
\le
\liminf_{n\to\infty}
\mathcal H(\boldsymbol\gamma_{\epsilon_n})
\le
\mathcal H(\overline{\boldsymbol\gamma}).
\]
By uniqueness of the entropy minimizer,
$\boldsymbol\gamma^*=\overline{\boldsymbol\gamma}$.  Every weak cluster
point is therefore the same, which proves convergence of the entire family.
The convergence of the common second marginals follows by continuity of the
push-forward under the coordinate projection.

\label{rem:selection-reference-dependence}
The entropy-selected tuple
$\overline{\boldsymbol\gamma}$ generally depends on the prescribed
support measure $\eta$, since the selecting functional is

=

<!-- proposition-5: body_md -->
\label{prop:moment-bound}
Let $\mu\in\mathcal{P}_2(\mathbb{R}^d)$, let
$\{\nu_n\}_{n\in\mathbb{N}}\subset\mathcal{P}_2(\mathbb{R}^d)$, and let
$\widetilde\gamma\in\mathcal{M}_+(\mathbb{R}^d\times\mathbb{R}^d)$ be such that
$\Pi^{\widetilde\gamma}(\mu,\nu_n)\neq\varnothing$ for every $n$.  If
\[
\mathsf{CCOT}_{\frac12|x-y|^2}^{\widetilde\gamma}(\mu,\nu_n)
\le M
\qquad\text{for every }n,
\]
then
\[
\sup_n\int_{\mathbb{R}^d}|y|^2\,\mathrm{d}\nu_n(y)<\infty.
\]
<!-- end proposition-5 -->
<!-- proof-17: body_md -->
The pointwise inequality
\[
-\frac12|x|^2+\frac14|y|^2
\le\frac12|x-y|^2
\]
follows from
$\frac12|x-y|^2+\frac12|x|^2-\frac14|y|^2
=|x-y/2|^2$.  Integrating it against any
$\gamma_n\in\Pi^{\widetilde\gamma}(\mu,\nu_n)$ gives
\[
\frac14\int_{\mathbb{R}^d}|y|^2\,\mathrm{d}\nu_n(y)
\le
\int_{\mathbb{R}^d\times\mathbb{R}^d}\frac12|x-y|^2\,\mathrm{d}\gamma_n(x,y)
+\frac12\int_{\mathbb{R}^d}|x|^2\,\mathrm{d}\mu(x).
\]
Taking the infimum over admissible $\gamma_n$ yields
\[
\frac14\int_{\mathbb{R}^d}|y|^2\,\mathrm{d}\nu_n(y)
\le
\mathsf{CCOT}_{\frac12|x-y|^2}^{\widetilde\gamma}(\mu,\nu_n)
+\frac12\int_{\mathbb{R}^d}|x|^2\,\mathrm{d}\mu(x),
\]
and the right-hand side is uniformly bounded.
