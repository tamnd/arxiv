<!-- lemma-1: body_md -->
(\cite{[Duff],[WW]})\label{hofflemma} 
Let $F(t)\geq0$  and $F(t)\in L^{\frac{1}{p}}(0,T)$. Suppose that  $G(t)\geq0$  and 
\begin{equation}\label{2.1} 
\frac{d}{dt}F(t)+G(t)\leq CF(t)^{\frac{q}{p}}. 
\end{equation} 
Then for $q>p+1$, there holds $G(t)\in L^{\frac{1}{q}}(0,T)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Rearranging the inequality \eqref{2.1}, we have 
\begin{equation} 
\frac{\frac{d}{dt}F(t)}{(F(t)+1)^{\frac{q-1}{p }}}+\frac{G(t)}{(F(t)+1)^{\frac{q-1}{p }}}\leq C[F(t)+1]^{\frac{1}{p}}. 
\end{equation} 
A simple  calculation  leads to $$\int_{0}^{T}\frac{G (t)}{(1+F(t))^{\frac{q-1}{p }}}dt<\infty.$$ 
Then we deduce from the H\"older inequality that 
$$\begin{aligned} 
\int_{0}^{T}G^{\frac{1}{q}}(t)dt=&\int_{0}^{T}(1+F(t))^{\frac{q-1}{pq}} 
\frac{G^{\frac{1}{q}}(t)}{(1+F(t))^{\frac{q-1}{pq}}}dt\\ 
\leq& \Big[\int_{0}^{T}(1+F(t))^{\frac1p}dt \Big]^{\frac{q-1}{ q}}\Big[\int_{0}^{T}\frac{G (t)}{(1+F(t))^{\frac{q-1}{p }}}dt\Big]^{\frac1q}. 
\end{aligned}$$ 
This lemma is  thus proved.

(1) Multiplying  the velocity field  and the magnetic field equations in the MHD equations \eqref{MHD} 
  by $ D ^{2}u$ and $ D ^{2}B$ respectively, we infer that 
 \begin{equation}\label{uH1} 
\frac{1}{2}\frac{d}{dt}\| D  u (t)\|^{2}_{L^{2}} 
+ \| D ^{2}u (t)\|^{2}_{L^{2}}= -\int_{\mathbb{R}^{3}} 
u (t)\cdot\nabla u (t)\cdot D ^{2}u (t)dx+\int_{\mathbb{R}^{3}} 
B (t)\cdot\nabla B (t)\cdot D ^{2}u (t)dx, \end{equation} 
 \begin{equation}\label{BH1} 
\frac{1}{2}\frac{d}{dt}\| D  B (t)\|^{2}_{L^{2}} 
+ \| D ^{2}B (t)\|^{2}_{L^{2}}= -\int_{\mathbb{R}^{3}} 
u (t)\cdot\nabla B (t)\cdot D ^{2}B (t)dx+\int_{\mathbb{R}^{3}} 
B (t)\cdot\nabla u (t)\cdot D ^{2}B (t)dx. \end{equation} 
 Summarizing \eqref{uH1} and \eqref{BH1}, we 
directly obtain 
 \begin{equation}\label{H1}\begin{aligned} 
&\frac{1}{2}\frac{d}{dt}(\| D  u (t)\|^{2}_{L^{2}}+\| D  B (t)\|^{2}_{L^{2}}) 
+ \| D ^{2}u (t)\|^{2}_{L^{2}}+ \| D ^{2}B (t)\|^{2}_{L^{2}} 
\\=& -\int_{\mathbb{R}^{3}} 
u (t)\cdot\nabla u (t)\cdot D ^{2}u (t)dx+\int_{\mathbb{R}^{3}} 
B (t)\cdot\nabla B (t)\cdot D ^{2}u (t)dx\\ 
 &-\int_{\mathbb{R}^{3}} 
u (t)\cdot\nabla B (t)\cdot D ^{2}B (t)dx+\int_{\mathbb{R}^{3}} 
B (t)\cdot\nabla u (t)\cdot D ^{2}B (t)dx. 
\end{aligned}\end{equation} 
In view of  the H\"older inequality, one gets 
$$\begin{aligned} 
\Big|\int_{\mathbb{R}^{3}}u (t) \cdot\nabla u (t)\cdot D ^{2}u (t)dx\Big| 
&\leq\|u (t)\|_{L^{6}}\|\nabla u (t)\| 
_{L^{3}} 
\| D ^{2}u (t)\|_{L^{2}}. 
\end{aligned}$$ 
 Thanks to the Sobolev embedding and the 
Gagliardo-Nirenberg inequality \eqref{keyinequality}, we infer that 
$$ 
\|u (t)\|_{L^{6} }\leq 
C\| D  u (t)\|_{L^{2}}, 
$$ 
and 
$$ 
\|\nabla u (t)\|_{L^{3} }\leq 
C\| D  u (t)\|_{L^{2}}^{\frac{1}{2}} 
\| D ^{2}u (t)\|^{\frac{1}{2}}_{L^{2}}. 
$$ 
As a consequence, we obtain 
$$\begin{aligned} 
\Big|\int_{\mathbb{R}^{3}}u (t) \cdot\nabla u (t)\cdot D ^{2}u (t)dx\Big| 
 &\leq 
C\| D  u (t)\|_{L^{2}} 
\| D  u (t)\|_{L^{2}}^{\frac{1}{2}} 
\| D ^{2}u (t)\|^{\frac{3}{2}}_{L^{2}} 
\\ 
&\leq 
\frac{1}{32} \| D ^{2}u (t)\|_{L^{2}}^{2} 
+C 
\| D  u (t)\|_{L^{2}}^{6},\end{aligned}$$ 
 where we  used    Young's inequality.\\ 
Proceeding as above, we discover that 
$$\begin{aligned} 
\Big|\int_{\mathbb{R}^{3}} 
B (t)\cdot\nabla B (t)\cdot D ^{2}u (t)dx \Big| \leq&\|B (t)\|_{L^{6}}\|\nabla B (t)\| 
_{L^{3}} 
\| D ^{2}u (t)\|_{L^{2}}\\ 
\leq&\| D  B (t)\|_{L^{2}}^{\frac{3}{2}} 
\| D ^{2}B (t)\|^{\frac{1}{2}}_{L^{2}}\| D ^{2}u (t)\|_{L^{2}}\\ 
\leq& 
\frac{1}{32} (\| D ^{2}u (t)\|_{L^{2}}^{2}+\| D ^{2}B (t)\|_{L^{2}}^{2}) 
+C 
\| D  B (t)\|_{L^{2}}^{6}, 
\end{aligned}$$ 
and 
$$\begin{aligned} 
&\Big|-\int_{\mathbb{R}^{3}} 
u (t)\cdot\nabla B (t)\cdot D ^{2}B (t)dx+\int 
B (t)\cdot\nabla u (t)\cdot D ^{2}B (t)dx\Big|\\ 
\leq& 
\frac{1}{32} (\| D ^{2}u (t)\|_{L^{2}}^{2}+\| D ^{2}B (t)\|_{L^{2}}^{2}) 
+C 
(\| D  B (t)\|_{L^{2}}^{6}+\| D  u (t)\|_{L^{2}}^{6}). 
\end{aligned}$$ 
Substituting  this   into estimate 
\eqref{H1}, we observe that 
\begin{equation}\label{3.2}

(2) Testing the MHD equations 
$\eqref{MHD}_{1}$  with $u _{t}$ and $\eqref{MHD}_{2}$  with $B _{t}$, we 
  integrate 
  in the space variable and sum to obtain

\end{aligned}\end{equation} 
With the help of the H\"older inequality, Sobolev inequality   and interpolation inequality \eqref{keyinequality}, we know that 
$$\begin{aligned} 
 \Big|\int_{\mathbb{R}^{3}}u\cdot\nabla u\cdot u _{t}dx\Big|\leq&\|u _{t}\|_{L^{2}}\|u  \|_{L^{6}}\|\nabla u \|_{L^{3}}\\\leq& C \|u _{t}\|_{L^{2}} \| D  u  \|_{L^{2}} \| D  u (t)\|_{L^{2}}^{\frac{1}{2}}

 \Big|\int_{\mathbb{R}^{3}}B\cdot\nabla B\cdot u _{t}dx \Big|\leq  \varepsilon\|u _{t}\|_{L^{2}}^{2} + C\| D  B (t)\|_{L^{2}}^{6}+

 \| D ^{2}u (t)\|^{2} _{L^{2}}+ 
  \| D ^{2}B (t)\|^{2} _{L^{2}}), 
\end{aligned}\end{equation}

\begin{equation}\begin{aligned}

\leq &2\varepsilon(\|u _{t}\|_{L^{2}}^{2} +\|B _{t}\|_{L^{2}}^{2} ) + C(\| D  u (t)\|_{L^{2}}^{2}+\| D  B (t)\|_{L^{2}}^{2})^{3}+2\delta( 
 \| D ^{2}u (t)\|^{2} _{L^{2}}+ 
  \| D ^{2}B (t)\|^{2} _{L^{2}}). 
\end{aligned} 
\end{equation} 
Taking $\varepsilon$ and $\delta$ small, we get 
\begin{equation}\begin{aligned}\label{3.6}

\leq&  C(\| D  u (t)\|_{L^{2}}^{2}+\| D  B (t)\|_{L^{2}}^{2})^{3}. 
\end{aligned} 
\end{equation} 
 
 
(3) 
Differentiating  the MHD equations 
\eqref{MHD} with respect to the time variable, we compute 
\begin{equation}\begin{aligned}\label{3.7} u^{(2) }_{t}-\Delta u_{t}+u_{t}\cdot\nabla u+u\cdot\nabla u_{t}-B_{t}\cdot\nabla B-B\cdot\nabla B_{t}+\nabla \Pi_{t}=0, \\

Taking the inner product of   $\eqref{3.7}_{1}$ with $u_{t}$  and $\eqref{3.7}_{2}$ with $B_{t}$,  integrating over the spatial variable,

$$\begin{aligned} 
&\frac{1}{2}\frac{d}{dt}\| u _{t}\|^{2}_{L^{2}}+\| D  u _{t}\|_{L^{2}}^{2}=-\int_{\mathbb{R}^{3}}u_{t}\cdot\nabla u\cdot u_{t}dx+\int_{\mathbb{R}^{3}}B_{t}\cdot\nabla B\cdot u_{t}dx+\int_{\mathbb{R}^{3}}B\cdot\nabla B_{t}\cdot u_{t}dx,\\ 
&\frac{1}{2}\frac{d}{dt}\| B _{t}\|^{2}_{L^{2}}+\| D  B _{t}\|_{L^{2}}^{2}=-\int_{\mathbb{R}^{3}}u_{t}\cdot\nabla B\cdot B_{t}dx+\int_{\mathbb{R}^{3}}B_{t}\cdot\nabla u\cdot B_{t}dx+\int_{\mathbb{R}^{3}}B\cdot\nabla u_{t}\cdot B_{t}dx.

$$ 
By means of   of integration by parts, we get

First, we assert that, for $k\in \mathbb{N}$, 
\begin{equation}\label{while} 
 D ^{k}u,  D ^{k}B\in L^{\frac{1}{k +1}}(0,T;L^{\infty}(\mathbb{R}^{3})). \end{equation} 
Indeed, we deduce from the Gagliardo-Nirenberg inequality and Sobolev embedding theorem that 
$$\| D ^{k}u\| 
^{ }_{L^{\infty}}\leq C\| u\| 
^{\frac{1}{2(k+1)} }_{L^{6}} \| D ^{k+2}u\| 
^{ \frac{2k+1}{2(k+1)} }_{L^{2}}\leq  C\| \nabla u\| 
^{\frac{1}{2(k+1)} }_{L^{2}} \| D ^{k+2}u\| 
^{ \frac{2k+1}{2(k+1)} }_{L^{2}}.$$ 
By integrating it with respect to time, we observe that 
$$\begin{aligned} 
\int_{0}^{T}\| D ^{k}u\| 
^{\frac{1}{k+1} }_{L^{\infty}}dt\leq&   C\int_{0}^{T}\| \nabla u\| 
^{\frac{1}{2(k+1)^{2}} }_{L^{2}} \| D ^{k+2}u\| 
^{ \frac{2k+1}{2(k+1)^{2}} }_{L^{2}}dt\\ 
\leq&   C\Big(\int_{0}^{T}\| \nabla u\| 
^{2 }_{L^{2}}dt\Big)^{\frac{1}{4(k+1)^{2}}} \Big(\int_{0}^{T}\| D ^{k+2}u\| 
^{ \frac{2 }{2 k+3} }_{L^{2}}dt\Big)^{1-\frac{1}{4(k+1)^{2}}}. 
\end{aligned}$$ 
Hence, we have $ D ^{k}u \in L^{\frac{1}{k +1}}(0,T;L^{\infty}(\mathbb{R}^{3})).$ 
Likewise,  $ D ^{k}B \in L^{\frac{1}{k +1}}(0,T;L^{\infty}(\mathbb{R}^{3})).$ 
\\ 
(1) According to the  interpolation inequality and Sobolev embedding theorem, we remark that 
$$\begin{aligned} 
\| u\|_{L^{q}}\leq \| u\|_{L^{6}}^{\frac6q}\| u\|^{1-\frac6q}_{L^{\infty}}\leq C\| \nabla u\|_{L^{2}}^{\frac6q}\| u\|^{1-\frac6q}_{L^{\infty}}, 6\leq q \leq\infty. 
\end{aligned}$$ 
Integrating the above inequality with respect to time variable and using the H\"older inequality, we get 
 \begin{equation}\begin{aligned} 
\int_{0}^{T}\| u\|_{L^{q}}^{\frac{q}{q-3}}dt \leq C \Big(\int_{0}^{T}\|\nabla u\|_{L^{2}}^{2}dt\Big)^{\frac{3}{q-3}}\Big( \int_{0}^{T} \|u\|_{L^{\infty}}dt\Big)^{1-\frac{3}{q-3}}. 
\end{aligned}\end{equation} 
We derive from \eqref{while} that $  u   \in L^{\frac{q}{q-3}}(0,T;L^{q} (\mathbb{R}^{3})),  6\leq q\leq\infty.$ Repeating the same deduction, we find $  B   \in L^{\frac{q}{q-3}}(0,T;L^{q} (\mathbb{R}^{3})),  6\leq q\leq\infty.$ \\ 
(2) 
We derive from the interpolation inequality that 
$$\begin{aligned} 
\| D ^{k}u\|_{L^{q}}\leq \| D ^{k}u\|_{L^{2}}^{\frac2q}\| D ^{k}u\|^{1-\frac2q}_{L^{\infty}}, 2\leq q \leq\infty. 
\end{aligned}$$

\begin{equation}\begin{aligned}

\\ 
\leq& \Big(\int_{0}^{T}\| D ^{k}u\|_{L^{2}}^{\frac{2}{2k -1}}dt\Big)^{\frac{2k -1}{kq+q-3}}\Big(\int_{0}^{T}\| D ^{k}u\| 
^{\frac{1}{k +1}}_{L^{\infty}}dt 
\Big) ^{1-\frac{2k -1}{kq+q-3}}. 
\end{aligned}\end{equation} 
Consequently, we find 
$ D ^{k}u   \in L^{\frac{q}{k(q+1)-3}}(0,T;L^{q} (\mathbb{R}^{3})), 2\leq q\leq\infty.$ Along the same line, we have $ D ^{k}B   \in L^{\frac{q}{k(q+1)-3}}(0,T;L^{q} (\mathbb{R}^{3})), 2\leq q\leq\infty.$ The proof of this corollary is completed.
