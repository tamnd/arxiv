<!-- lemma-1: body_md -->
\label{Lem:CalS1}
Let $n \geq 2$ and ${\varepsilon} > 0$. Then \eqref{Eq:Cal3} holds for $u \in u_{\varepsilon} + C_c^\infty(B\setminus \{0\},{\mathbb R}^n)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We compute as in \cite{INSZ_CRAS18}. Let $w = u - u_{\varepsilon} \in  C_c^\infty(B\setminus \{0\},{\mathbb R}^n)$. First, we have 
\begin{align*}
{\mycal E}_{\varepsilon}(u) - {\mycal E}_{\varepsilon}(u_{\varepsilon})
	&= \int_B \Big[\nabla u_{\varepsilon} : \nabla w - \frac{1}{{\varepsilon}^2} (1 - |u_{\varepsilon}|^2) u_{\varepsilon} \cdot w\Big]\,dx\\
		&\qquad + \int_B \Big[\frac{1}{2}|\nabla w|^2 - \frac{1}{2{\varepsilon}^2} (1 - |u_{\varepsilon}|^2) |w|^2 + \frac{1}{4{\varepsilon}^2}(2 u_{\varepsilon} \cdot w + |w|^2)^2\Big]\,dx.
\end{align*}
The first integral vanishes due to the criticality of $u_{\varepsilon}$ yielding
\begin{align*}
{\mycal E}_{\varepsilon}(u) - {\mycal E}_{\varepsilon}(u_{\varepsilon})
	&= \frac{1}{2}\int_B \Big[ |\nabla w|^2 - \frac{1}{ {\varepsilon}^2} (1 - f_{\varepsilon}^2) |w|^2\\
		&\qquad + \frac{2}{{\varepsilon}^2} f_{\varepsilon}^2 (n \cdot w)^2 + \frac{2}{ {\varepsilon}^2}  f_{\varepsilon} |w|^2(n \cdot w ) +  \frac{1}{2{\varepsilon}^2} |w|^4\Big]\,dx,
\end{align*}
where $n(x) = \frac{x}{|x|}$. Similarly,  as $\frac{F_{\varepsilon}}{f_{\varepsilon}}u-U_{\varepsilon}=\frac{F_{\varepsilon}}{f_{\varepsilon}}w$,  
\begin{align*}
{\mycal E}_{\varepsilon}\Big(\frac{F_{\varepsilon}}{f_{\varepsilon}}u\Big) - {\mycal E}_{\varepsilon}(U_{\varepsilon})
	&= \frac{1}{2}\int_B \Big[|\nabla \Big(\frac{F_{\varepsilon}}{f_{\varepsilon}}w\Big)|^2 - \frac{1}{{\varepsilon}^2}\frac{F_{\varepsilon}^2}{f_{\varepsilon}^2} (1 - F_{\varepsilon}^2) |w|^2\\
		&\qquad + \frac{2}{{\varepsilon}^2} \frac{F_{\varepsilon}^4}{f_{\varepsilon}^2} (n \cdot w)^2 + \frac{2}{ {\varepsilon}^2}  \frac{F_{\varepsilon}^4}{f_{\varepsilon}^3}  |w|^2(n \cdot w ) +  \frac{1}{2{\varepsilon}^2}\frac{F_{\varepsilon}^4}{f_{\varepsilon}^4} |w|^4  \Big]\,dx.
\end{align*}
Therefore,  since $f_{\varepsilon}>F_{\varepsilon}$ in $(0,1)$, 
\begin{align}
\nonumber
\Delta_{\varepsilon}(u)  
	&=\Big[{\mycal E}_{\varepsilon}(u) - {\mycal E}_{\varepsilon}(u_{\varepsilon})\Big] - \Big[{\mycal E}_{\varepsilon}\Big(\frac{F_{\varepsilon}}{f_{\varepsilon}} u\Big) - {\mycal E}_{\varepsilon}(U_{\varepsilon})\Big]
\nonumber	\\ & = \frac{1}{2} \int_B \Big[|\nabla w|^2 - \Big|\nabla\Big(\frac{F_{\varepsilon}}{f_{\varepsilon}}w\Big)\Big|^2 - \frac{1}{{\varepsilon}^2} \frac{f_{\varepsilon}^2 - F_{\varepsilon}^2}{f_{\varepsilon}^2} (1 - f_{\varepsilon}^2 - F_{\varepsilon}^2)|w|^2\Big]\,dx \\\nonumber	
		&\qquad  + \frac{2}{{\varepsilon}^2} \frac{1}{f_{\varepsilon}^4}(f_{\varepsilon}^4 - F_{\varepsilon}^4) \big(f_{\varepsilon}  n \cdot w  + \frac{1}{2}|w|^2\big)^2\Big]\,dx\\
\label{eq:ftilde}	&  \geq   \frac{1}{2} \int_B \Big[|\nabla w|^2 - \Big|\nabla\Big(\frac{F_{\varepsilon}}{f_{\varepsilon}}w\Big)\Big|^2 - \frac{1}{{\varepsilon}^2} \frac{f_{\varepsilon}^2 - F_{\varepsilon}^2 - f_{\varepsilon}^4 + F_{\varepsilon}^4}{f_{\varepsilon}^2}  |w|^2\Big]\,dx = \frac{1}{2} \widetilde{{\mycal F}}_{\varepsilon}(w).
\end{align}

It remains to prove for $\tilde w = \frac{w}{f_{\varepsilon}}$ that
\begin{equation}
\widetilde{{\mycal F}}_{\varepsilon}(w) = \int_B (f_{\varepsilon}^2 - F_{\varepsilon}^2)\Big[|\nabla \tilde w|^2 - \frac{n-1}{r^2} |\tilde w|^2\Big]\,dx.
	\label{Eq:Cal4}
\end{equation}
Indeed, we compute using \eqref{Eq:feps} and the  Hardy decomposition $w=f_{\varepsilon} \tilde w$ for the operator $-\Delta$ (see e.g. \cite[Lemma A.1]{INSZ3}),
\begin{align*}
\int_B |\nabla w|^2\,dx 
	&		= \int_B \Big[f_{\varepsilon}^2 |\nabla \tilde w|^2 +  |\tilde w|^2 f_{\varepsilon} \cdot (-\Delta f_{\varepsilon})   \Big]\,dx \\
	&= \int_B \Big[f_{\varepsilon}^2 |\nabla \tilde w|^2 - \frac{n-1}{r^2} f_{\varepsilon}^2|\tilde w|^2    + \frac{1}{{\varepsilon}^2} f_{\varepsilon}^2(1 - f_{\varepsilon}^2) |\tilde w|^2  \Big]\,dx.
\end{align*}
Similarly, by \eqref{Eq:Feps},
\begin{align*}
\int_B \Big|\nabla \Big(\frac{F_{\varepsilon}}{f_{\varepsilon}} w\Big)|^2\,dx 
	&= \int_B |\nabla (F_{\varepsilon} \tilde w)|^2\,dx\\
		 &= \int_B \Big[F_{\varepsilon}^2 |\nabla \tilde w|^2 - \frac{n-1}{r^2} F_{\varepsilon}^2|\tilde w|^2 + \frac{1}{{\varepsilon}^2} F_{\varepsilon}^2(1 - F_{\varepsilon}^2) |\tilde w|^2  \Big]\,dx.
\end{align*}
Putting these two identities into the definition of $\widetilde{{\mycal F}}_{\varepsilon}(w)$ we obtain \eqref{Eq:Cal4}.

<!-- lemma-2: body_md -->
Let $n \geq 2$, ${\varepsilon} > 0$ and $\phi_{\varepsilon}$ and $S_{\varepsilon}$ be given by \eqref{Eq:phi} and \eqref{Eq:S}. Then
$$ -\nabla \cdot [(f_{\varepsilon}^2 - F_{\varepsilon}^2)\nabla \phi_{\varepsilon}]-\frac{n-1}{r^2}  (f_{\varepsilon}^2 - F_{\varepsilon}^2)\phi_{\varepsilon}= (f_{\varepsilon}^2 - F_{\varepsilon}^2)S_{\varepsilon} \phi_{\varepsilon}  \quad \textrm{ in } B \setminus \{0\},$$
that is, 
\begin{equation}
- \frac{[r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}']'}{r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2)}  - \frac{n-1}{r^2} \phi_{\varepsilon}
= 
S_{\varepsilon} \phi_{\varepsilon} \quad  \textrm{ in } (0, 1) .
	\label{Eq:phi''}
\end{equation}

<!-- proof-2: body_md -->
Let 
\begin{align}
b_{\varepsilon}
	&= \frac{f_{\varepsilon} f_{\varepsilon}' - F_{\varepsilon} F_{\varepsilon}'}{f_{\varepsilon}^2 - F_{\varepsilon}^2} \quad \textrm{ in } (0, 1),\nonumber\\
\eta_{\varepsilon} 
	&= \frac{\phi_{\varepsilon}'}{\phi_{\varepsilon}}
	= \frac{2}{r}q_{\varepsilon}-\frac{2}{r}   -  b_{\varepsilon} \quad \textrm{ in } (0, 1).
	\label{Eq:Sc-1}
\end{align}
We compute
\begin{align}
\frac{[r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}']'}{r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2)\phi_{\varepsilon}} 
	&= \frac{[r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon} \eta_{\varepsilon}]'}{r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}}\nonumber\\
	&= \frac{[r^{n-3}(f_{\varepsilon}^2 - F_{\varepsilon}^2)^{1/2} F_{\varepsilon}^2 \eta_{\varepsilon}]'}{r^{n-3}(f_{\varepsilon}^2 - F_{\varepsilon}^2)^{1/2} F_{\varepsilon}^2}
		= \eta_{\varepsilon}' + \Big(\frac{2}{r} q_{\varepsilon} + \frac{n-3}{r}  +  b_{\varepsilon}\Big) \eta_{\varepsilon}.
	\label{Eq:Sc-2}
\end{align}


We proceed to compute $\eta_{\varepsilon}'$. First, since
\[
\frac{F_{\varepsilon}''}{F_{\varepsilon}} = \Big(\frac{F_{\varepsilon}'}{F_{\varepsilon}}\Big)' + \Big(\frac{F_{\varepsilon}'}{F_{\varepsilon}}\Big)^2 = \frac{q_{\varepsilon}'}{r} + \frac{q_{\varepsilon}^2 - q_{\varepsilon}}{r^2},
\]
the ODE \eqref{Eq:Feps} for $F_{\varepsilon}$ can be rewritten as
\begin{equation}
q_{\varepsilon}' = - \frac{1}{r} (q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1)  -\frac{r}{{\varepsilon}^2}(1 - F_{\varepsilon}^2).
	\label{Eq:q'}
\end{equation}
Next,
\[
b_{\varepsilon}'
	= \frac{f_{\varepsilon} f_{\varepsilon}'' - F_{\varepsilon} F_{\varepsilon}''}{f_{\varepsilon}^2 - F_{\varepsilon}^2} + \frac{(f_{\varepsilon}')^2 - (F_{\varepsilon}')^2}{f_{\varepsilon}^2 - F_{\varepsilon}^2}- 2b_{\varepsilon}^2.
\]
Using \eqref{Eq:feps} and \eqref{Eq:Feps}, we get
\[
b_{\varepsilon}'
	= \frac{n-1}{r^2} - \frac{n-1}{r} b_{\varepsilon} - \frac{1}{{\varepsilon}^2}(1 - f_{\varepsilon}^2 - F_{\varepsilon}^2) + \frac{(f_{\varepsilon}')^2 - (F_{\varepsilon}')^2}{f_{\varepsilon}^2 - F_{\varepsilon}^2}- 2b_{\varepsilon}^2.
\]
Therefore,  by \eqref{Eq:Sc-1},
\begin{align}
\eta_{\varepsilon}' 
	&= - \frac{1}{r^2} (2q_{\varepsilon}^2 + 2(n-1)q_{\varepsilon} - (n+1))  
	+ \frac{n-1}{r} b_{\varepsilon}  + 2b_{\varepsilon}^2\nonumber\\
	&\qquad - \frac{1}{{\varepsilon}^2}(1 + f_{\varepsilon}^2 - F_{\varepsilon}^2) 
	 - \frac{(f_{\varepsilon}')^2 - (F_{\varepsilon}')^2}{f_{\varepsilon}^2 - F_{\varepsilon}^2}.
	 \label{Eq:Sc-3}
\end{align}
Inserting \eqref{Eq:Sc-1} and \eqref{Eq:Sc-3} into \eqref{Eq:Sc-2} gives
\begin{align*}
\frac{[r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}']'}{r^{n-1}(f_{\varepsilon}^2 - F_{\varepsilon}^2)\phi_{\varepsilon}} 
	&=    \frac{1}{r^2} ( 2q_{\varepsilon}^2 - 8q_{\varepsilon} - (n-7))  
	- \frac{1}{{\varepsilon}^2}(1 + f_{\varepsilon}^2 - F_{\varepsilon}^2) 
	 \\
	&\qquad 
	+  b_{\varepsilon}^2  - \frac{(f_{\varepsilon}')^2 - (F_{\varepsilon}')^2}{f_{\varepsilon}^2 - F_{\varepsilon}^2}.
\end{align*}
Using the identity
\[
b_{\varepsilon}^2  - \frac{(f_{\varepsilon}')^2 - (F_{\varepsilon}')^2}{f_{\varepsilon}^2 - F_{\varepsilon}^2} = \Big( \frac{f_{\varepsilon} f_{\varepsilon}' - F_{\varepsilon} F_{\varepsilon}'}{f_{\varepsilon}^2 - F_{\varepsilon}^2}\Big)^2  - \frac{(f_{\varepsilon}')^2 - (F_{\varepsilon}')^2}{f_{\varepsilon}^2 - F_{\varepsilon}^2} =  \frac{(f_{\varepsilon}' F_{\varepsilon}  - F_{\varepsilon}' f_{\varepsilon})^2}{(f_{\varepsilon}^2 - F_{\varepsilon}^2)^2},
\]
we arrive at \eqref{Eq:phi''} after a simple rearrangement of terms.

<!-- lemma-3: body_md -->
\label{Lem:CalS2}
Let $n \geq 2$ and ${\varepsilon} > 0$. Then \eqref{Eq:tmcFS} holds.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
By \eqref{eq:ftilde}, $\widetilde{{\mycal F}}_{\varepsilon}$ is continuous in $H_0^1(B,{\mathbb R}^n)$
because $F_{\varepsilon}/f_{\varepsilon}$ is continuous and positive in $[0,1]$. Consider first the case $w \in C_c^\infty(B\setminus \{0\},{\mathbb R}^n)$.
Recalling that $w = f_{\varepsilon} \tilde w$, we use the Hardy decomposition $\tilde w= \phi_{\varepsilon} \hat w$ for the operator $-\nabla \cdot [(f_{\varepsilon}^2 - F_{\varepsilon}^2)\nabla]$ (see e.g. \cite[Lemma A.1]{INSZ3}) and \eqref{Eq:phi''} to find  
\begin{align*}
\int_B (f_{\varepsilon}^2 - F_{\varepsilon}^2) |\nabla \tilde w|^2\,dx
	&=  \int_B \Big\{(f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}^2 |\nabla \hat w|^2  +|\hat w|^2  \phi_{\varepsilon} \cdot \big[-\nabla \cdot \big((f_{\varepsilon}^2 - F_{\varepsilon}^2)  \nabla \phi_{\varepsilon}\big)\big]\Big\}\,dx \\
&	= \int_B (f_{\varepsilon}^2 - F_{\varepsilon}^2) \Big[\phi_{\varepsilon}^2 |\nabla \hat w|^2 + \Big(\frac{n-1}{r^2} + S_{\varepsilon}\Big)\phi_{\varepsilon}^2 |\hat w|^2  \Big]\,dx.
\end{align*}
Inserting this into \eqref{Eq:Cal4} we obtain \eqref{Eq:tmcFS} for $w\in C_c^\infty(B\setminus \{0\},{\mathbb R}^n)$. 

Consider now the general case $w \in H_0^1(B,{\mathbb R}^n)$. By density, we may select $\{w_j\} \subset C_c^\infty(B\setminus \{0\},{\mathbb R}^n)$ such that $w_j \rightarrow w$ in $H_0^1(B,{\mathbb R}^n)$. We have by the continuity of $\widetilde{{\mycal F}}_{\varepsilon}$ in $H_0^1(B,{\mathbb R}^n)$ that
\begin{equation}
\widetilde{{\mycal F}}_{\varepsilon}(w) = \lim_{j \rightarrow \infty} \widetilde{{\mycal F}}_{\varepsilon}(w_j) = \lim_{j \rightarrow \infty}  \int_{B} (f_{\varepsilon}^2 - F_{\varepsilon}^2) \Big[ \phi_{\varepsilon}^2\Big|\nabla \Big(\frac{w_j }{f_{\varepsilon} \phi_{\varepsilon}}\Big)\Big|^2 + \frac{S_{\varepsilon} |w_j|^2}{f_{\varepsilon}^2}\Big]  \,dx.
	\label{Eq:Fa0}
\end{equation}
Noting that $S_{\varepsilon} = O_{\varepsilon}(1)$ as $r \rightarrow 0$ (see \eqref{Eq:FAs}, \eqref{Eq:qFAs} and \eqref{Eq:feaAs}), we have that
\begin{align}
\lim_{j\rightarrow \infty}  \int_{B}\frac{ (f_{\varepsilon}^2 - F_{\varepsilon}^2) S_{\varepsilon} |w_j|^2}{f_{\varepsilon}^2}  \,dx 
	&=  \int_{B}\frac{ (f_{\varepsilon}^2 - F_{\varepsilon}^2) S_{\varepsilon} |w|^2}{f_{\varepsilon}^2}  \,dx,\label{Eq:Fa1}\\
\lim_{j,k\rightarrow \infty}  \int_{B}\frac{ (f_{\varepsilon}^2 - F_{\varepsilon}^2) S_{\varepsilon} |w_j - w_k|^2}{f_{\varepsilon}^2}  \,dx 
	&= 0.\label{Eq:Fa2}
\end{align}
By \eqref{Eq:Fa2} and the continuity of $\widetilde{{\mycal F}}_{\varepsilon}$, we have that
\begin{multline*}
\lim_{j,k\rightarrow \infty}  \int_{B} (f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}^2 \Big|\nabla \Big(\frac{w_j - w_k}{f_{\varepsilon} \phi_{\varepsilon}}\Big)\Big|^2  \,dx\\
	 = \lim_{j,k\rightarrow \infty} \Big[\widetilde{{\mycal F}}_{{\varepsilon}}(w_j - w_k) -  \int_{B}\frac{ (f_{\varepsilon}^2 - F_{\varepsilon}^2) S_{\varepsilon} |w_j - w_k|^2}{f_{\varepsilon}^2}  \,dx \Big]
	= 0.
\end{multline*}
Hence the sequence $\Big\{(f_{\varepsilon}^2 - F_{\varepsilon}^2)^{1/2} \phi_{\varepsilon}  \nabla \big(\frac{w_j}{f_{\varepsilon} \phi_{\varepsilon}}\big)\Big\}$ is Cauchy in $L^2(B, {\mathbb R}^{n\times n})$ and hence converges therein to $(f_{\varepsilon}^2 - F_{\varepsilon}^2)^{1/2} \phi_{\varepsilon}  \nabla \big(\frac{w}{f_{\varepsilon} \phi_{\varepsilon}}\big)$. In particular
\begin{equation}
\lim_{j \rightarrow \infty}  \int_{B} (f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}^2 \Big|\nabla \Big(\frac{w_j  }{f_{\varepsilon} \phi_{\varepsilon}}\Big)\Big|^2  \,dx
	= \int_{B} (f_{\varepsilon}^2 - F_{\varepsilon}^2) \phi_{\varepsilon}^2 \Big|\nabla \Big(\frac{w   }{f_{\varepsilon} \phi_{\varepsilon}}\Big)\Big|^2  \,dx.
	\label{Eq:Fa3}
\end{equation}
Using \eqref{Eq:Fa1} and \eqref{Eq:Fa3} in \eqref{Eq:Fa0}, we obtain \eqref{Eq:tmcFS}.

<!-- lemma-4: body_md -->
\label{Lem:qProp}
Let $n \geq 2$ and ${\varepsilon} > 0$. Then $q_{\varepsilon}(0) = 1$, $q_{\varepsilon}'(0) = q_{\varepsilon}(\infty) = q_{\varepsilon}'(\infty) = 0$, $0 < q_{\varepsilon} < 1$ and $q_{\varepsilon}' < 0$ in $(0,\infty)$.
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
A routine argument as in \cite{Hervex2, ODE_INSZ} shows that there exists a constant $d_0 > 0$ such that
\begin{equation}
\begin{cases}
F_{\varepsilon}(r) = d_0 {\varepsilon}^{-1} r - \frac{d_0}{2(n+2)} {\varepsilon}^{-3}r^3 + \frac{d_0(2(n+2)d_0^2 + 1)}{8(n+2)(n+4)}{\varepsilon}^{-5} r^5 + O({\varepsilon}^{-7} r^7) \text{ as } r \rightarrow 0,\\
F_{\varepsilon}'(r) = d_0 {\varepsilon}^{-1}  - \frac{3d_0}{2( n+2 )} {\varepsilon}^{-3}r^2 +   \frac{5d_0(2(n+2)d_0^2 + 1)}{8(n+2)(n+4)}{\varepsilon}^{-5} r^4 + O({\varepsilon}^{-7}r^6) \text{ as } r \rightarrow 0,\\
F_{\varepsilon}(r) = 1 - \frac{n-1}{2} {\varepsilon}^2 r^{-2} + \frac{3(n-1)(n-5)}{8} {\varepsilon}^{4} r^{-4} + O({\varepsilon}^6 r^{-6}) \text{ as } r \rightarrow \infty, \\ 
F_{\varepsilon}'(r) =   (n-1) {\varepsilon}^2 r^{-3} - \frac{3(n-1)(n-5)}{2} {\varepsilon}^{4} r^{-5} + O({\varepsilon}^6 r^{-7}) \text{ as } r \rightarrow \infty.
\end{cases}
\label{Eq:FAs}
\end{equation}
It follows that
\begin{equation}
\begin{cases}
q_{\varepsilon}(r) = 1 - \frac{1}{n+2} {\varepsilon}^{-2}r^2 + \frac{(n+2)^2 d_0^2 - 1}{(n+2)^2(n+4)} {\varepsilon}^{-4} r^4 + O({\varepsilon}^{-6}r^6) \text{ as } r \rightarrow 0,\\
q_{\varepsilon}'(r) = - \frac{2}{n+2} {\varepsilon}^{-2}r  + \frac{4(n+2)^2 d_0^2 - 4}{(n+2)^2(n+4)} {\varepsilon}^{-4} r^3 + O({\varepsilon}^{-6}r^5) \text{ as } r \rightarrow 0,\\
q_{\varepsilon}(r) = (n-1) {\varepsilon}^2 r^{-2} + O({\varepsilon}^{4} r^{-4}) \text{ as } r \rightarrow \infty, \\ 
q_{\varepsilon}'(r) = -2(n-1) {\varepsilon}^2 r^{-3} + O({\varepsilon}^{4} r^{-5}) \text{ as } r \rightarrow \infty.
\end{cases}
\label{Eq:qFAs}
\end{equation}
Hence $q_{\varepsilon}(0) = 1$, $q_{\varepsilon}'(0) = q_{\varepsilon}(\infty) = q_{\varepsilon}'(\infty) = 0$.

From the monotonicity of $F_{\varepsilon}$, it follows by definition that $q_{\varepsilon} > 0$. To see that $q_{\varepsilon} < 1$, consider the function $\check F_{\varepsilon} = F_{\varepsilon}/r$ which satisfies
\[
\check F_{\varepsilon}'' + \frac{n+1}{r} \check F_{\varepsilon}' = - \frac{1}{{\varepsilon}^2}(1 - F_{\varepsilon}^2)\check F_{\varepsilon} < 0.
\]
In particular, $r^{n+1}\check F_{\varepsilon}'$ is a decreasing function. Since its limit as $r \rightarrow 0$ is zero, it follows that $\check F_{\varepsilon}' < 0$ in $(0,\infty)$. This implies $\frac{F_{\varepsilon}'}{F_{\varepsilon}} - \frac{1}{r} = \frac{\check F_{\varepsilon}'}{\check F_{\varepsilon}} < 0$ and $q_{\varepsilon} = \frac{rF_{\varepsilon}'}{F_{\varepsilon}} < 1$ in $(0,\infty)$.


It remains to show $q_{\varepsilon}' < 0$ in $(0,\infty)$. Differentiating \eqref{Eq:q'} and using $r(F_{\varepsilon}^2)' = 2 F_{\varepsilon}^2 q_{\varepsilon}$ we find
\begin{equation}
q_{\varepsilon}'' = - \frac{1}{r} (2 q_{\varepsilon} + n - 2)q_{\varepsilon}' + \frac{1}{r^2} (q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1) + \frac{1}{{\varepsilon}^2} (2q_{\varepsilon} + 1)F_{\varepsilon}^2 - \frac{1}{{\varepsilon}^2}.
	\label{Eq:q''}
\end{equation}
In particular,
\[
- r^2 q_{\varepsilon}'' - r (2 q_{\varepsilon} + n - 1)q_{\varepsilon}' + \frac{2r^2F_{\varepsilon}^2 }{{\varepsilon}^2} q_{\varepsilon}= \frac{2r^2}{{\varepsilon}^2}(1 - F_{\varepsilon}^2) > 0.
\]
Equation \eqref{Eq:q''} can be written without reference to $F_{\varepsilon}$ as follows. First, by \eqref{Eq:q'},
\begin{equation}
\frac{F_{\varepsilon}^2}{{\varepsilon}^2} = \frac{q_{\varepsilon}'}{r} + \frac{1}{r^2} (q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1) + \frac{1}{{\varepsilon}^2}.
	\label{Eq:q'X}
\end{equation}
Using \eqref{Eq:q'X} in \eqref{Eq:q''}, we get
\begin{equation}
q_{\varepsilon}'' = - \frac{1}{r} (n - 3)q_{\varepsilon}' +  \frac{2}{r^2} (q_{\varepsilon} + 1)(q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1) + \frac{2q_{\varepsilon}}{{\varepsilon}^2}.
	\label{Eq:q''X}
\end{equation}
Differentiating \eqref{Eq:q''X}, we get
\begin{align}
q_{\varepsilon}''' &= - \frac{1}{r} (n - 3)q_{\varepsilon}'' + \frac{1}{r^2} (6q_{\varepsilon}^2 + 4(n-1)q_{\varepsilon} + n-5) q_{\varepsilon}' + \frac{2q_{\varepsilon}'}{{\varepsilon}^2}  \nonumber\\
	&\qquad -  \frac{4}{r^3} (q_{\varepsilon} + 1)(q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1)  .
	\label{Eq:q'''}
\end{align}
By \eqref{Eq:q'},
\[
- \frac{4}{r^3} (q_{\varepsilon}+ 1) (q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1) = \frac{4}{r^2} (q_{\varepsilon} + 1) q_{\varepsilon}'  + \frac{4}{{\varepsilon}^2 r}(q_{\varepsilon} + 1)(1 - F_{\varepsilon}^2).
\]
Using this for the last term in \eqref{Eq:q'''} we get
\[
-q_{\varepsilon}''' - \frac{1}{r} (n - 3)q_{\varepsilon}'' + \Big[\frac{1}{r^2} (6q_{\varepsilon}^2 + 4n q_{\varepsilon} + n-1) + \frac{2}{{\varepsilon}^2} \Big]q_{\varepsilon}'
	= -\frac{4}{{\varepsilon}^2 r}(q_{\varepsilon} + 1)(1 - F_{\varepsilon}^2) > 0.
\]
In particular, $q_{\varepsilon}'$ cannot have a non-negative local maximum in $(0,\infty)$. As $q_{\varepsilon}'(0) = q_{\varepsilon}'(\infty) = 0$, we thus have $q_{\varepsilon}' < 0$ in $(0,\infty)$.

<!-- lemma-5: body_md -->
\label{Lem:psiLB}
Let $n \geq 2$ and ${\varepsilon} > 0$. Then $\psi_{\varepsilon}(0) = \psi_{\varepsilon}(\infty) = 2$ and $\psi_{\varepsilon} > 2$ in $(0,\infty)$.
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
By \eqref{Eq:qFAs}, $\psi_{\varepsilon}(0) = \psi_{\varepsilon}(\infty) = 2$. Differentiating $\psi_{\varepsilon}$, we find
\[
\psi_{\varepsilon}' = \frac{q_{\varepsilon}''}{q_{\varepsilon}'} \psi_{\varepsilon}  - \frac{1}{r}(2q_{\varepsilon} - 1) \psi_{\varepsilon}^2+ \frac{1}{r} \psi_{\varepsilon}.
\]
Using \eqref{Eq:q''X}, we get
\begin{equation}
\psi_{\varepsilon}' = - \frac{1}{r} (n - 4)  \psi_{\varepsilon} - \frac{1}{r}(2q_{\varepsilon} - 1) \psi_{\varepsilon}^2 + \frac{2}{r q_{\varepsilon}} (q_{\varepsilon} + 1)(q_{\varepsilon} + n - 1)  - \frac{2r}{{\varepsilon}^2(1 - q_{\varepsilon})} .
	\label{Eq:psi'}
\end{equation}
We rewrite this as
\[
\frac{1}{r} (1 - q_{\varepsilon}) \psi_{\varepsilon}' = \frac{1}{r^2} A(q_{\varepsilon},\psi_{\varepsilon}) - \frac{2}{{\varepsilon}^2}
\]
with
\[
A(x,y) = (1-x)\Big[- (n - 4)  y - (2x - 1) y^2 + \frac{2}{x} (x + 1)(x + n - 1)\Big].
\]
Differentiating and using $q_{\varepsilon}' = \frac{1}{r} \psi_{\varepsilon} q_{\varepsilon} (q_{\varepsilon} - 1)$, we get 
\begin{equation}
(1 - q_{\varepsilon}) \psi_{\varepsilon}''
	- \frac{1}{r} [(1 - q_{\varepsilon}) (1 - \psi_{\varepsilon} q_{\varepsilon} ) + \partial_y A(q_{\varepsilon},\psi_{\varepsilon})]\psi_{\varepsilon}' 
	= \frac{1}{r^2} \frac{1-q_{\varepsilon}}{q_{\varepsilon}} B(q_{\varepsilon}, \psi_{\varepsilon})
	\label{Eq:psi''}
\end{equation}
with
\begin{align*}
B(x,y) 
	&= - x^2y \partial_x A(x,y) - \frac{2x}{1-x} A(x,y)\\
	&= - x^2 y^3(4x - 3) - xy^2[(n-8)x + 2] \\
		&\qquad + 2y[2x^3 + (n-1)x^2 + (n-4)x + (n-1)] \\
		&\qquad- 4(x+1)(x+n-1).
\end{align*}

By \eqref{Eq:psi''}, at an interior minimum point of $\psi_{\varepsilon}$, it holds that
\[
B(q_{\varepsilon}, \psi_{\varepsilon}) \geq 0.
\]
Therefore, as $\psi_{\varepsilon}(0) = \psi_{\varepsilon}(\infty) = 2$, to show that $\psi_{\varepsilon} > 2$, we only need to show that
\begin{equation}
B(x,y) < 0 \text{ for } x \in (0,1), y \in (0,2].
	\label{Eq:Bneg}
\end{equation}
Indeed, note that $B$ is non-increasing with respect to dimension $n$ for $x \in [0,1], y \in (0,2]$:
\[
\frac{\partial B}{\partial n} = - (2-y)(2 + 2x - x^2y) \leq 0.
\]
Thus, we may take $n = 2$ when proving \eqref{Eq:Bneg}. Let $B_2$ denote $B$ when $n = 2$, i.e.
\[
B_2(x,y) 
	= x^2 y^3(3 - 4x) - 2xy^2(1-3x) 
		+ 2y(2x^3 + x^2 - 2x + 1)
			- 4(x+1)^2.
\]
Note that
\[
B_2(x,2) 
	= -24x(x-1)^2 < 0 \text{ for } x \in (0,1).
\]
Therefore, \eqref{Eq:Bneg} is proved once we show that $B_2$ is increasing in $y$ for $x \in [0,1], y \in (0,2]$. We compute
\[
\partial_y B_2(x,y) = 3x^2 y^2(3 - 4x) - 4xy(1-3x) + 2\big(2x^3 + (x-1)^2\big).
\]
It is clear that $\partial_y B_2(x,y) > 0$ for $x \in (1/3,3/4)$, $y \in [0,2]$ (as all three summands are positive). When $0 \leq x \leq 1/3$, we have
\[
3 - 4x \geq \frac{5}{3} \text{ and } 2(2x^3 + x^2 - 2x + 1) \geq 2(x-1)^2 \geq \frac{8}{9} > \frac{4}{5}, 
\]
and so
\[
\partial_y B_2(x,y) > 5x^2y^2 - 4xy + \frac{4}{5} = 5(xy - \frac{2}{5})^2 \geq 0.
\]
When $x \geq 3/4$, $\partial_y B_2(x,\cdot)$ is concave in $y$, so its minimum in $[0,2]$ is attained at the endpoints. For $x\in [0,1]$,
\begin{align*}
\partial_y B_2(x,0) &= 4x^3 + 2(x-1)^2 > 0,\\
\partial_y B_2(x,2) &= -44x^3 + 62 x^2 - 12 x + 2  =44(x^2-x^3)+  2(3x - 1)^2 > 0 ,
\end{align*}
yielding $\partial_y B_2(x,y) > 0$ for $x \in [3/4,1]$, $y \in [0,2]$. 
In any case, we have shown that $B_2$ is increasing in $y$ for $x \in [0,1], y \in (0,2]$. As explained before, this proves \eqref{Eq:Bneg} and the inequality $\psi_{\varepsilon} > 2$ in $(0,\infty)$.

<!-- lemma-6: body_md -->
\label{Lem:psiUB}
Let $n \geq 2$ and ${\varepsilon} > 0$. Then 
\[
\psi_{\varepsilon} <  \begin{cases}
\frac{2(7+4q_{\varepsilon})}{4+ 7 q_{\varepsilon}} & \text{ if } n = 2,\\

<!-- proof-8: body_md -->
Define
\[
g(x) = \begin{cases}
\frac{2x(1 - x) (7 + 4 x)}{4 + 7 x} & \text{ if } n = 2,\\
3x(1-x) & \text{ if } n \geq 3.
\end{cases}
\]
We need to show that
\[
\chi_{\varepsilon}= rq_{\varepsilon}' + g(q_{\varepsilon}) > 0 \text{ in } (0,\infty).
\]
Suppose by contradiction that this does not hold. As $\chi_{\varepsilon}(0) = \chi_{\varepsilon}(\infty) = 0$, this implies that $\chi_{\varepsilon}$ has a non-positive interior minimum.


We compute
\[
\chi_{\varepsilon}'
	= r q_{\varepsilon}'' + (1 + g'(q_{\varepsilon})) q_{\varepsilon}'.
\]
Using \eqref{Eq:q''X}, we get
\begin{align}
\chi_{\varepsilon}'
	&=   (g'(q_{\varepsilon}) - n + 4) q_{\varepsilon}'
	+ \frac{2}{r} (q_{\varepsilon} + 1)(q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1) + \frac{2rq_{\varepsilon}}{{\varepsilon}^2} 	\nonumber\\
	&=   \frac{1}{r}(g'(q_{\varepsilon}) - n + 4) \chi_{\varepsilon}\nonumber\\
		&\qquad
		+ \frac{1}{r}[2(q_{\varepsilon} + 1)(q_{\varepsilon} - 1)(q_{\varepsilon} + n - 1) - (g'(q_{\varepsilon}) - n + 4)g(q_{\varepsilon}) ]
	  + \frac{2rq_{\varepsilon}}{{\varepsilon}^2}.
\label{Eq:chi'}
\end{align}
We rewrite this as
\[
\frac{1}{rq_{\varepsilon}} \chi_{\varepsilon}'
	= \frac{1}{r^2} M(q_{\varepsilon},  \chi_{\varepsilon}) + \frac{2}{{\varepsilon}^2} 
\]
where
\begin{align*}
M(x,z) 
	&= c_1(x) z + c_0(x),\\
c_1(x)
	&= \frac{1}{x}(g'(x) - n + 4),\\
c_0(x)
	&= \frac{1}{x}\Big[2(x + 1)(x - 1)(x + n - 1) - (g'(x) - n + 4)g(x)\Big].
\end{align*}
Differentiating and using $rq_{\varepsilon}' = \chi_{\varepsilon} - g(q_{\varepsilon}) $, we get
\begin{equation}
\frac{1}{q_{\varepsilon}} \chi_{\varepsilon}'' = \frac{1}{rq_{\varepsilon}^2}[q_{\varepsilon} + rq_{\varepsilon}'+ q_{\varepsilon}^2 \partial_z M (q_{\varepsilon}, \chi_{\varepsilon})]\chi_{\varepsilon}' + \frac{1}{r^2} P(q_{\varepsilon},\chi_{\varepsilon})
	\label{Eq:chi''}
\end{equation}
where
\begin{align*}
P(x,z) 
	&= \partial_x M(x,z)(z - g(x))  - 2M(x,z)\\
	&= [c_1'(x) z + c_0'(x)][z - g(x)] - 2[c_1(x) z + c_0(x)]\\
	&= c_1'(x) z^2 + [c_0'(x) - c_1'(x) g(x) - 2 c_1(x)]z - c_0'(x) g(x) - 2 c_0(x).
\end{align*}
In particular, 
\begin{equation}
P(q_{\varepsilon},\chi_{\varepsilon}) \geq 0 \text{ at an interior minimum of } \chi_{\varepsilon}.
	\label{Eq:Psign}
\end{equation}

We continue the proof with the case $3 \leq n \leq 7$, followed by the case $n \geq 8$, and then $n = 2$.

\medskip
\noindent\underline{Case 1:} $3 \leq n \leq 7$.

In this case, $g(x) = 3x (1 - x)$ and 
\begin{align}
P(x,z) 
	&= \frac{1}{x^2} \Big[(n - 7) z^2  + a_1(x) z + x(x-1) a_0(x)\Big],\label{Eq:P3-1}\\
a_1(x)
	&= -32x^3 + 2(n + 14) x^2   - (n-7)x + 2(n-1),\label{Eq:P3-2}\\
a_0(x)
	&=- 96x^3
			- (3n - 143)x^2  + 2(n -21)x
			+ 2 (n-1).\label{Eq:P3-3}
\end{align}

In light of \eqref{Eq:Psign} and the statement that $\chi_{\varepsilon}$ has a non-positive interior minimum, we only need to show that
\begin{equation}
P(x,z) < 0 \text{ for all } x \in (0,1), z \leq 0.
	\label{Eq:Pcond}
\end{equation}
To this end, it is enough to show that
\begin{equation}
a_1(x) > 0 \text{ and } a_0(x) > 0 \text{ for } x \in [0,1].
	\label{Eq:a01cond}
\end{equation}

\begin{align*}
a_1(x)

\end{align*}
For $a_0$, note that $a_0$ is increasing with respect to $n$ for $x \in (0,1)$:
\[

<!-- corollary-1: body_md -->
\label{Cor:F<>q}

<!-- corollary-3: body_md -->

<!-- lemma-8: body_md -->
\label{Lem:Gronwall}
Let $\theta \in C^1((0,r_0))$ and $a \in C^0([0,r_0])$  for some $r_0 > 0$, and assume that

	&= \frac{2r}{{\varepsilon}^2} + \frac{n+2}{r} (2- q_{\varepsilon})(1-q_{\varepsilon})(q_{\varepsilon} + n - 1) - \frac{(n+2)r}{{\varepsilon}^2}(2-q_{\varepsilon})(1 - F_{\varepsilon}^2)\\

\end{align*}

\begin{align*}

	&= \frac{4(26 - 9\sqrt{3}) y^3 +  57(2 - \sqrt{3}) y  }{8 + 32y^2 -  49y^6} \quad \text{ for } y \in [0,\sqrt{3}/2].
\end{align*}

\begin{equation}

	\label{Eq:AE-3}
\end{equation}
We make three observations.
\begin{enumerate}[(i)]
\item The function $y$ is strictly decreasing in $[0,3/4]$.
\item The function $c$ is strictly decreasing in $(0,1]$, as
\[
\frac{c'}{c} = -\frac{1 + 6x + 11 x^2 - 6x^3}{2x(1-x)(2x+1)(3x+1)} < 0.
\]

\[

\[

By the arithmetic-geometric mean inequality, the numerator is positive:
\[

\]
Thus $H_1' > 0$. Hence $H$ is strictly increasing.
\end{enumerate}
