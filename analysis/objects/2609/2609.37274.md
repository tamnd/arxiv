<!-- theorem-1: body_md -->
\label{3pc1}
		Let $X$ be a reflexive real Banach space, \mbox{$\Phi :X\rightarrow\mathbb{R}$} be a coercive, continuously G\^{a}teaux differentiable and sequentially weakly lower semicontinuous functional whose G\^{a}teaux derivative admits a continuous inverse on $X^*$, $\Psi:X\rightarrow \mathbb{R}$ be a continuously G\^{a}teaux differentiable functional whose G\^{a}teaux derivative is compact, such that
		\begin{enumerate}[label=\textnormal{(b$_\arabic*$)},ref=\textnormal{b$_\arabic*$}] 
		\item \label{b1} $\inf\limits_X \Phi=\Phi(0)=\Psi(0)=0$.
		\end{enumerate}
		Assume that there exist a positive constant $r$ and $\overline{v}\in X$, with $r<\Phi(\overline{v})$, such that
		\begin{enumerate}[label=\textnormal{(b$_\arabic*$)},ref=\textnormal{b$_\arabic*$}] \setcounter{enumi}{1}
			\item \label{b2} $\dfrac{\sup\limits_{u\in \Phi^{-1}((-\infty,r))}\Psi(u)}{r} < \dfrac{\Psi(\overline{v})}{\Phi(\overline{v})}$,
			\item \label{b3} $\Phi-\lambda\Psi$ is coercive for all $\lambda\in \Lambda_{r} = \left( \frac{\Phi(\overline{v})}{\Psi(\overline{v})},\frac{r}{\sup\limits_{u\in \Phi^{-1}((-\infty,r))}\Psi(u)} \right)$.
		\end{enumerate}
		Then, for each $\lambda\in \Lambda_{r}$, the functional $\Phi-\lambda\Psi$ admits at least three distinct critical points.

<!-- theorem-2: body_md -->
\label{3pc2}
		Let $X$ be a reflexive real Banach space, \mbox{$\Phi:X\rightarrow \mathbb{R}$} be a convex, coercive and continuously G\^{a}teaux differentiable functional whose derivative admits a continuous inverse on $X^*$, $\Psi:X\rightarrow \mathbb{R}$ be a continuously G\^{a}teaux differentiable functional whose derivative is compact, such that
		\begin{enumerate}[label=\textnormal{(e$_\arabic*$)},ref=\textnormal{e$_\arabic*$}]
			\item \label{e1} $\inf\limits_X \Phi=\Phi(0)=\Psi(0)=0;$
			\item \label{e2} for each $\lambda>0$ and for every $u_1,u_2$ which are local minima of the functional $\Phi-\lambda\Psi$, such that $\Psi(u_1)\geq 0$ and $\Psi(u_2)\geq 0$, one has
			\[\inf\limits_{s\in[0,1]}\Psi(su_1+(1-s)u_2)\geq 0.\]
		\end{enumerate}
		Assume that there are two positive constants $r_1,r_2$ and
		$\overline{v}\in X$, with \mbox{$2r_1<\Phi(\overline{v})<\dfrac{r_2}{2}$}, such that
		\begin{enumerate}[label=\textnormal{(e$_\arabic*$)},ref=\textnormal{e$_\arabic*$}] \setcounter{enumi}{2}
			\item \label{e3} $\dfrac{\sup\limits_{u\in \Phi^{-1}((-\infty,r_1))}\Psi(u)}{r_1}<\dfrac{2}{3}\dfrac{\Psi(\overline{v})}{\Phi(\overline{v})}$,
			
			\item \label{e4} $\dfrac{\sup\limits_{u\in \Phi^{-1}((-\infty,r_2))}\Psi(u)}{r_2}<\dfrac{1}{3}\dfrac{\Psi(\overline{v})}{\Phi(\overline{v})}$.
		\end{enumerate}
		Then, for each $\lambda\in \Lambda_{r_1,r_2} =
		\left( \frac{3}{2}\frac{\Phi(\overline{v})}{\Psi(\overline{v})} \, , \, \min\left\{\frac{r_1}{\sup\limits_{u\in \Phi^{-1}((-\infty,r_1))}\Psi(u)} , \frac{r_2/2}{\sup\limits_{u\in \Phi^{-1}((-\infty,r_2))} \Psi(u)}\right\} 	\right)$ the functional $\Phi-\lambda\Psi$ admits at least three distinct critical points which lie in $\Phi^{-1}((-\infty,r_2))$.

<!-- theorem-3: body_md -->
\label{3sol_1}
		Let \eqref{hp-problem} be satisfied and assume that there exist $x_0 \in \mathbb{R}^N$, $R>0$ and two constants $c, d$ such that
		\begin{enumerate}[itemsep=3pt, label=\textnormal{($h_\arabic*$)},ref=\textnormal{$h_\arabic*$}]
			\item\label{hp:cost}  $0<c<d$,
			
			\item\label{hp:segno} $F(x,t) \ge 0$ for a.e. $x \in S$ and for all $t \in [0,d]$,
			
			\item\label{hp:cond1} $ \dfrac{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c} F(x,t) \,\mathrm{d} x}{c^p}  < K_R \dfrac{\displaystyle\int_{B\left(x_0, \frac{R}{2}\right)} F(x, d) \,\mathrm{d} x}{d^p}$,
			
			\item\label{hp:sublin} $\limsup\limits_{|t|\to+\infty} \left( \mathop {\rm ess\,sup}\limits_{x \in \mathbb{R}^N} \dfrac{F(x,t)}{|t|^p} \right) \le 0$.
		\end{enumerate}
		Then, for each $\lambda\in (\lambda_1, \lambda_2)$, where
		\begin{equation} \label{Lambda}
			\lambda_1 = \frac{1}{p k_a^p K_R} \dfrac{d^p}{\displaystyle\int_{B\left(x_0, \frac{R}{2}\right)} F(x, d) \,\mathrm{d} x} \qquad \lambda_2= 	\frac{1}{p k_a^p} \dfrac{c^p}{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c} F(x,t) \,\mathrm{d} x},
		\end{equation}
		problem \eqref{problem} admits at least three distinct weak solutions.

<!-- proof-1: body_md -->
We apply Theorem \ref{3pc1} to show that, for every $\lambda \in (\lambda_1, \lambda_2)$, the functional $I_\lambda$ admits at least three critical points, that is, problem \eqref{problem} admits at least three weak solutions. As observed in Section \ref{Sec:prel}, the functionals $\Phi$ and $\Psi$ satisfy the regularity assumptions required in Theorem \ref{3pc1}, and hypothesis \eqref{b1} clearly holds true.
	In order to verify \eqref{b2}, set
	\[
	r = \dfrac{1}{p} \left( \dfrac{c}{k_a} \right)^p.
	\]
	For any $u \in \Phi^{-1}((-\infty,r))$, by using \eqref{embedding} we have
	\begin{align*}
		\|u\|_\infty \le k_a \|u\| \le c,
	\end{align*}
	so that
	\begin{align*}
		\Phi^{-1}((-\infty,r)) \subseteq \left\{ u \in W^{1,p}(\mathbb{R}^N) \, \colon \, \|u\|_\infty \le c  \right\}.
	\end{align*}
	Hence, one has
	\begin{align*}
		\sup\limits_{u\in \Phi^{-1}((-\infty,r))} \Psi(u) \le \sup\limits_{\|u\|_\infty \le c} \int_{\mathbb{R}^N} F(x,u(x)) \,\mathrm{d} x \le \int_{\mathbb{R}^N} \max\limits_{|t|\le c} F(x,t) \,\mathrm{d} x
	\end{align*}
	and
	\begin{align} \label{dim1-cond1}
		\frac{\sup\limits_{u\in \Phi^{-1}((-\infty,r))} \Psi(u) }{r} \le p k_a^p \frac{\int_{\mathbb{R}^N} \max\limits_{|t|\le c} F(x,t) \,\mathrm{d} x}{c^p}.
	\end{align}
	Now, we introduce $\bar{v} \colon \mathbb{R}^N \to \mathbb{R}$ defined by
	\begin{equation} \label{barv}
	\bar{v}(x) = \begin{cases}
		0 &\text{if} \; x \in \mathbb{R}^N \setminus B(x_0,R),\\
		\dfrac{2d}{R}(R-|x-x_0|) \quad & \text{if} \; x \in S,\\
		d & \text{if} \; x \in B\left(x_0, \frac{R}{2}\right).
	\end{cases}
	\end{equation}
	From Brezis \cite[Remark 4(ii), p. 265]{Brezis-book-2011} it follows that $\bar{v} \in W^{1,p}(\mathbb{R}^N)$. By the definition of $\bar{v}$, we have that
	\begin{align*}
		\Phi(\bar{v}) =& \dfrac{1}{p} \left( \int_S |\nabla \bar{v}|^p \,\mathrm{d} x + \int_S a(x) |\bar{v}|^p \,\mathrm{d} x + \int_{B\left(x_0, \frac{R}{2}\right)} a(x) d^p \,\mathrm{d} x  \right)\\
		\le&  \frac{1}{p} \left[ \left(\frac{2d}{R}\right)^p \operatorname*{meas}(S) \left(1 +  \|a\|_\infty \left(\frac{R}{2}\right)^p \right) + d^p \|a\|_\infty \operatorname*{meas}\left( B\left(x_0, \frac{R}{2}\right) \right) \right]\\
		=& \frac{d^p}{p} \, \frac{\pi^\frac{N}{2}}{\Gamma\left( 1 + \frac{N}{2} \right)} \, R^N \, \left[ \frac{2^p}{R^p} \, \frac{2^N-1}{2^N} + \|a\|_\infty \, \frac{2^N-1}{2^N} + \|a\|_\infty \, \frac{1}{2^N} \right]\\
		=& \frac{d^p}{p} \, \frac{\pi^\frac{N}{2}}{\Gamma\left( 1 + \frac{N}{2} \right)} \, \frac{2^p-2^{p-N}+ R^p\|a\|_\infty}{R^{p-N}},
	\end{align*}
	namely
	\begin{align} \label{dim1-cond2}
		\Phi(\bar{v}) \le \frac{d^p}{p} \, \frac{1}{k_a^p \, K_R}.
	\end{align}
	Moreover, it is easy to verify that $\Phi(\bar{v}) > r$. Indeed, $\|\bar{v}\|_\infty=d>c$ by \eqref{hp:cost}, so \eqref{embedding} gives
	\begin{align} \label{dim1-cost}
		\Phi(\bar{v}) = \frac{1}{p} \|\bar{v}\|^p \ge \frac{1}{p} \frac{1}{k_a^p} \|\bar{v}\|_\infty^p = \frac{1}{p} \frac{1}{k_a^p} d^p > \frac{1}{p} \frac{1}{k_a^p} c^p =r.
	\end{align}
	Furthermore, by \eqref{hp:segno},
	\begin{align} 
		\Psi(\bar{v}) &= \int_S F\left( x,  \dfrac{2d}{R}(R-|x-x_0|)\right) \,\mathrm{d} x + \int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x  \notag \\
		&\ge \int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x. \label{dim1-cond3}
	\end{align}
	Putting together \eqref{dim1-cond1}, \eqref{hp:cond1}, \eqref{dim1-cond2} and \eqref{dim1-cond3}, we obtain
	\begin{align*}
			\frac{\sup\limits_{u\in \Phi^{-1}((-\infty,r))} \Psi(u) }{r} &\le p k_a^p \frac{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c} F(x,t) \,\mathrm{d} x}{c^p}\\
			& < p k_a^p K_R \frac{ \displaystyle\int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x}{d^p}\\
			& \le \frac{\Psi(\bar{v})}{\Phi(\bar{v})},
	\end{align*}
	and assumption \eqref{b2} of Theorem \ref{3pc1} holds true, with $(\lambda_1, \lambda_2) \subseteq \Lambda_{r}$. Finally, we show that $I_\lambda$ is coercive for every $\lambda>0$, so that \eqref{b3} is verified. Fix $\lambda>0$ and $\varepsilon \in \left(0, \frac{a_-}{p\lambda}\right)$. By \eqref{hp:sublin} there exists $\delta>0$ such that $F(x,t) \le \varepsilon |t|^p$ for a.e. $x \in \mathbb{R}^N$ and all $|t|>\delta$. On the other hand, $F(x,t) \le \delta\, m_\delta(x)$ for a.e. $x\in\mathbb{R}^N$ and all $|t| \le \delta$, where $m_\delta = \sup_{|t|\le\delta}|f(\cdot,t)| \in L^{1}(\mathbb{R}^N)$ by \eqref{hp-problem}. Hence, since $a_- \|u\|_p^p \le \|u\|^p$, we get
	\[
	I_\lambda(u) \ge \frac{1}{p}\|u\|^p - \lambda \varepsilon \|u\|_p^p - \lambda \delta \|m_\delta\|_1 \ge \left( \frac{1}{p} - \frac{\lambda\varepsilon}{a_-} \right) \|u\|^p - \lambda \delta \|m_\delta\|_1
	\]
	for all $u \in W^{1,p}(\mathbb{R}^N)$, and the coercivity of $I_\lambda$ follows. Then, Theorem \ref{3pc1} ensures that for any $\lambda \in (\lambda_1, \lambda_2)$ the energy functional $I_\lambda$ admits at least three distinct critical points, that is, problem \eqref{problem} admits at least three distinct weak solutions and the proof is complete.

<!-- lemma-1: body_md -->
\label{solnonneg}
	Let \eqref{hp-problem} be satisfied and assume that
	\begin{equation} \label{trunc}
		f(x,t)=f(x,0) \qquad \text{for a.e. } x \in \mathbb{R}^N \text{ and all } t \le 0.
	\end{equation}
	If $f(x,0)\ge0$ for a.e. $x \in \mathbb{R}^N$, then any weak solution of \eqref{problem} is nonnegative.

<!-- remark-2: body_md -->
\label{rem:trunc}
Condition \eqref{trunc} cannot be removed in Lemma \ref{solnonneg}: for instance, if $f(x,\cdot)$ is odd, then $-u$ is a weak solution of \eqref{problem} whenever $u$ is. However, \eqref{trunc} is not restrictive for our purposes. Indeed, consider the truncated function
\[
\hat{f}(x,t) = \begin{cases} f(x,t) & \text{if } t > 0,\\ f(x,0) & \text{if } t \le 0, \end{cases}
\]
which satisfies \eqref{hp-problem} and \eqref{trunc}, and denote by $(\hat{P}_\lambda)$ problem \eqref{problem} with $\hat{f}$ in place of $f$. Assume that $f(x,0)\ge 0$ for a.e. $x\in\mathbb{R}^N$. Then, by Lemma \ref{solnonneg}, every weak solution $u$ of $(\hat{P}_\lambda)$ is nonnegative, so that $\hat{f}(x,u(x)) = f(x,u(x))$ for a.e. $x\in\mathbb{R}^N$ and $u$ is a nonnegative weak solution of \eqref{problem}. Moreover, setting $\hat{F}(x,t)=\int_0^t \hat{f}(x,\xi)\,\mathrm{d}\xi$, one has $\hat{F}(x,t)=F(x,t)$ for $t\ge0$ and $\hat{F}(x,t)=f(x,0)\,t\le0$ for $t\le0$, hence, for a.e. $x\in\mathbb{R}^N$,
\begin{equation} \label{trunc-max}
	\hat{F}(x,t) \le \max\{F(x,t),0\} \;\; \forall\, t\in\mathbb{R}, \qquad \max_{|t|\le c}\hat{F}(x,t) \le \max_{|t|\le c} F(x,t) \;\; \forall\, c>0.
\end{equation}

<!-- corollary-1: body_md -->
\label{th1:nonneg}
		Let \eqref{hp-problem} be satisfied and assume that there exist $x_0 \in \mathbb{R}^N$, $R>0$ and two constants $c, d$ such that \eqref{hp:cost}--\eqref{hp:sublin} hold true. Moreover, suppose that
				\begin{enumerate}[itemsep=3pt, label=\textnormal{($h_\arabic*$)},ref=\textnormal{$h_\arabic*$}]
					\setcounter{enumi}{4}
					\item\label{hp:f0} $f(x,0)\ge0$ for a.e. $x \in \mathbb{R}^N$.
				\end{enumerate}
		Then, for every $\lambda\in (\lambda_1, \lambda_2)$, defined in \eqref{Lambda}, problem \eqref{problem} admits at least three distinct nonnegative weak solutions.

<!-- remark-3: body_md -->
\label{rem:nontrivial}
	In Theorem \ref{3sol_1} and Corollary \ref{th1:nonneg} (as well as in Theorem \ref{3sol_2} below), if $f(\cdot,0) \not\equiv 0$, then $u \equiv 0$ is not a solution of \eqref{problem} and, therefore, all three solutions obtained are nontrivial. If $f(\cdot,0) \equiv 0$, then $u\equiv 0$ is a solution of \eqref{problem} and the results ensure the existence of at least two nontrivial solutions.

<!-- theorem-4: body_md -->
\label{3sol_2}
	Let \eqref{hp-problem} be satisfied and assume that there exist $x_0 \in \mathbb{R}^N$, $R>0$ and three positive constants $c_1, c_2, d$ such that
	\begin{enumerate}[itemsep=4pt, label=\textnormal{($h'_\arabic*$)},ref=\textnormal{$h'_\arabic*$}]
		\item\label{hp2:costanti} $2^\frac{1}{p} c_1 < d < c_2$,
				
		\item\label{hp2:segno} $f(x,t) \ge 0$ for a.e. $x \in \mathbb{R}^N$ and for all $t \ge 0$,
		
		\item\label{hp2:cond1} $ \dfrac{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c_1} F(x,t) \,\mathrm{d} x}{c_1^p}  < \dfrac{2}{3} K_R \dfrac{\displaystyle\int_{B\left(x_0, \frac{R}{2}\right)} F(x, d) \,\mathrm{d} x}{d^p}$,
		
		\item\label{hp2:cond2} $ \dfrac{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c_2} F(x,t) \,\mathrm{d} x}{c_2^p}  < \dfrac{1}{3} K_R \dfrac{\displaystyle\int_{B\left(x_0, \frac{R}{2}\right)} F(x, d) \,\mathrm{d} x}{d^p}$.
	\end{enumerate}
	Then, for each $\lambda\in \left(\lambda'_1, \lambda'_2\right)$, where
	\begin{align*}
		\lambda'_1 &= \frac{3}{2}\frac{1}{p k_a^p K_R} \dfrac{d^p}{\displaystyle\int_{B\left(x_0, \frac{R}{2}\right)} F(x, d) \,\mathrm{d} x}, \\
		\lambda'_2 &= \min\left\{ \frac{1}{p k_a^p} \dfrac{c_1^p}{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c_1} F(x,t) \,\mathrm{d} x} \, , \, \frac{1}{2 p k_a^p} \dfrac{c_2^p}{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c_2} F(x,t) \,\mathrm{d} x}\right\},
	\end{align*}
	problem \eqref{problem} admits at least three distinct nonnegative weak solutions $u_1, u_2, u_3 \in W^{1,p}(\mathbb{R}^N)$ such that $\|u_i\| < \dfrac{c_2}{k_a}$ and $\|u_i\|_\infty < c_2$ for every $i=1,2,3$.

<!-- proof-2: body_md -->
First, note that $f(x,0)\ge0$ for a.e. $x\in\mathbb{R}^N$ by \eqref{hp2:segno}. Hence, by Remark \ref{rem:trunc}, we may assume that $f$ satisfies \eqref{trunc}: indeed, $\hat{f}$ satisfies \eqref{hp-problem}, \eqref{hp2:costanti} and \eqref{hp2:segno}; by \eqref{trunc-max} it also satisfies \eqref{hp2:cond1} and \eqref{hp2:cond2}, with a value of $\lambda'_2$ not smaller than the one of $f$ and the same $\lambda'_1$; finally, the weak solutions of $(\hat{P}_\lambda)$ are nonnegative weak solutions of \eqref{problem}.

	The proof is based on applying Theorem \ref{3pc2} to the functionals $\Phi, \Psi$ defined in Section \ref{Sec:prel}, which, as observed there, satisfy the required regularity assumptions and \eqref{e1}. Also, \eqref{e2} holds true by the definition of $\Psi$ and assumption \eqref{hp2:segno}. Indeed, fix $\lambda>0$ and let $u_1, u_2 \in W^{1,p}(\mathbb{R}^N)$ be local minima of $I_\lambda = \Phi-\lambda\Psi$; then they are critical points of $I_\lambda$, that is, weak solutions of \eqref{problem}, and since \eqref{hp2:segno} holds, by Lemma \ref{solnonneg} it follows that $u_1 \ge0, u_2 \ge 0$. Again from \eqref{hp2:segno}, we have that $F(x,t) \ge 0$ for a.e. $x \in \mathbb{R}^N$ and all $t \ge 0$, which leads to $\Psi(u_1)\ge0$ and $\Psi(u_2)\ge0$. Moreover,
	\[
	s u_1 + (1-s) u_2 \ge 0 \qquad \text{for any} \; s \in [0,1],
	\]
	which yields
	\[
	F(x,s u_1 + (1-s) u_2) \ge 0 \qquad \text{for a.e.} \; x \in \mathbb{R}^N,
	\]
	and
	\[
	\Psi(s u_1 + (1-s) u_2) \ge 0 \qquad \text{for any} \; s \in [0,1].
	\]
	Now, in order to prove \eqref{e3} and \eqref{e4} we proceed as in the proof of Theorem \ref{3sol_1}. Note that the left-hand side of \eqref{hp2:cond1} is nonnegative, since $F(x,0)=0$. Hence, \eqref{hp2:cond1} implies $\int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x > 0$, so that $\lambda'_1$ is well defined, and the interval $(\lambda'_1, \lambda'_2)$ is nonempty by \eqref{hp2:cond1}--\eqref{hp2:cond2}. Set $r_1 = \dfrac{1}{p} \left( \dfrac{c_1}{k_a} \right)^p$ and $r_2 = \dfrac{1}{p} \left( \dfrac{c_2}{k_a} \right)^p$. Then, by \eqref{embedding},
	\begin{align*}
		&\|u\|_\infty \le k_a \|u\| \le c_1 \qquad \text{for all} \; u \in \Phi^{-1}((-\infty, r_1)),\\
		&\|u\|_\infty \le k_a \|u\| \le c_2 \qquad \text{for all} \; u \in \Phi^{-1}((-\infty, r_2)).
	\end{align*}
	Then, taking $\bar{v} \in W^{1,p}(\mathbb{R}^N)$ defined as in \eqref{barv}, from assumptions \eqref{hp2:cond1} and \eqref{hp2:segno} we obtain
	\begin{align*}
		\frac{\sup\limits_{u\in \Phi^{-1}((-\infty,r_1))} \Psi(u) }{r_1} &\le p k_a^p \frac{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c_1} F(x,t) \,\mathrm{d} x}{c_1^p}\\
		& < \frac{2}{3} p k_a^p K_R \frac{\displaystyle \int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x}{d^p}\\
		& \le\frac{2}{3} \frac{\Psi(\bar{v})}{\Phi(\bar{v})}.
	\end{align*}
	Similarly, by using assumptions  \eqref{hp2:cond2} and \eqref{hp2:segno} we have
	\begin{align*}
		\frac{\sup\limits_{u\in \Phi^{-1}((-\infty,r_2))} \Psi(u) }{r_2} &\le p k_a^p \frac{\displaystyle\int_{\mathbb{R}^N} \max\limits_{|t|\le c_2} F(x,t) \,\mathrm{d} x}{c_2^p}\\
		& < \frac{1}{3} p k_a^p K_R \frac{\displaystyle \int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x}{d^p}\\
		& \le\frac{1}{3} \frac{\Psi(\bar{v})}{\Phi(\bar{v})}.
	\end{align*}
	Finally, we show that $2r_1 < \Phi(\bar{v}) < \frac{r_2}{2}$. On the one hand, from \eqref{hp2:costanti} and \eqref{dim1-cost} one has
	\[
	2 r_1 = \frac{2}{p} \frac{c_1^p}{k_a^p} < \frac{1}{p}\frac{d^p}{k_a^p} \le \Phi(\bar{v}).
	\]
	On the other hand, by \eqref{hp2:segno}, for a.e. $x \in \mathbb{R}^N$ the function $F(x,\cdot)$ is nonnegative and nondecreasing in $[0,+\infty)$. Hence, since $d<c_2$, we get
	\[
	\int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x \le \int_{\mathbb{R}^N} F(x,c_2) \,\mathrm{d} x \le \int_{\mathbb{R}^N} \max\limits_{|t|\le c_2} F(x,t) \,\mathrm{d} x,
	\]
	which, together with \eqref{hp2:cond2}, gives
	\[
	\frac{1}{c_2^p} \int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x < \frac{K_R}{3} \, \frac{1}{d^p} \int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x.
	\]
	Since $\int_{B\left(x_0, \frac{R}{2}\right)} F(x,d) \,\mathrm{d} x > 0$, we get $d^p < \frac{K_R}{3} c_2^p$. Therefore, by \eqref{dim1-cond2},
	\[
	\Phi(\bar{v}) \le \frac{1}{p} \frac{d^p}{k_a^p K_R} < \frac{1}{3p} \frac{c_2^p}{k_a^p} < \frac{r_2}{2}.
	\]
	Hence, all the assumptions of Theorem \ref{3pc2} are satisfied. Moreover, by the first inequality in each of the two chains above and by \eqref{dim1-cond2}, \eqref{dim1-cond3}, one has $(\lambda'_1, \lambda'_2) \subseteq \Lambda_{r_1,r_2}$. Therefore, by applying Theorem \ref{3pc2} we get that for any $\lambda \in (\lambda'_1, \lambda'_2)$ problem \eqref{problem} admits at least three distinct weak solutions such that $\|u_i\| < \dfrac{c_2}{k_a}$ and $\|u_i\|_\infty < c_2$ for every $i=1,2,3$. Also, the solutions are nonnegative thanks to \eqref{hp2:segno} and Lemma \ref{solnonneg}.

				\setcounter{enumi}{1}
		\item[\textnormal{(${h}_1$)}] $0<c<d$,
		
		\item\label{hp:cond1-SV} 
		$ \dfrac{G(c)}{c^p}  < \tilde{K}_R \dfrac{G(d)}{d^p}$,
		
		\item\label{hp:sublin-SV}
		$\lim\limits_{t \to+\infty} \dfrac{G(t)}{t^p}=0$.
	\end{enumerate}

		 \tilde{\lambda}_2= 	\frac{1}{p k_a^p \|h\|_1} \dfrac{c^p}{G(c)},
	\end{equation*}
	problem \eqref{problem-SV} admits at least three distinct nonnegative weak solutions.

<!-- proof-3: body_md -->
It suffices to apply Theorem \ref{3sol_1} to $\hat{f}$, with $f(x,t)=h(x)g(t)$, and to take Remark \ref{rem:trunc} into account. Note that $f(x,0)=h(x)g(0)\ge0$ and that $\hat{f}$ satisfies \eqref{hp:cost}--\eqref{hp:cond1}, with $\lambda_1=\tilde{\lambda}_1$ and $\lambda_2=\tilde{\lambda}_2$, since, by \eqref{max-SV}, $\int_{\mathbb{R}^N} \max_{|t|\le c} \hat{F}(x,t) \,\mathrm{d} x = \|h\|_1 G(c)$, and $\int_{B\left(x_0, \frac{R}{2}\right)} \hat{F}(x,d) \,\mathrm{d} x = G(d) \int_{B\left(x_0, \frac{R}{2}\right)} h(x) \,\mathrm{d} x$. We only point out that, since $h$ may be unbounded, hypothesis \eqref{hp:sublin} is not directly implied by \eqref{hp:sublin-SV}. However, \eqref{hp:sublin} is used in the proof of Theorem \ref{3sol_1} only to obtain the coercivity of the energy functional, which still holds. Indeed, fix $\lambda>0$ and $\varepsilon \in \left(0, \frac{1}{p\lambda k_a^p \|h\|_1}\right)$; by \eqref{hp:sublin-SV} there is $\delta>0$ such that $G(t) \le \varepsilon t^p$ for all $t>\delta$. Since $G(t) \le G(\delta)$ for $t \in [0,\delta]$ and $\hat{F}(x,t)=h(x)g(0)\,t \le 0$ for $t \le 0$, we get $\hat{F}(x,t) \le h(x)\left(G(\delta)+\varepsilon|t|^p\right)$ for a.e. $x\in\mathbb{R}^N$ and all $t\in\mathbb{R}$. Hence, by \eqref{embedding},
	\[
	\int_{\mathbb{R}^N} \hat{F}(x,u(x)) \,\mathrm{d} x \le \|h\|_1 G(\delta) + \varepsilon \int_{\mathbb{R}^N} h(x) |u(x)|^p \,\mathrm{d} x \le \|h\|_1 G(\delta) + \varepsilon \|h\|_1 k_a^p \|u\|^p,
	\]
	so that $\Phi(u) - \lambda \int_{\mathbb{R}^N} \hat{F}(x,u(x)) \,\mathrm{d} x \ge \left( \frac{1}{p} - \lambda \varepsilon k_a^p \|h\|_1 \right) \|u\|^p - \lambda \|h\|_1 G(\delta)$ for all $u \in W^{1,p}(\mathbb{R}^N)$.

\label{th3-SV}
	Let \eqref{hp-problem-SV} be satisfied and let $x_0 \in \mathbb{R}^N$ and $R>0$ be such that $\int_{B\left(x_0, \frac{R}{2}\right)} h(x) \,\mathrm{d} x >0$. Moreover, assume that
	\begin{enumerate}[itemsep=4pt, label=\textnormal{($\tilde{h}'_\arabic*$)},ref=\textnormal{$\tilde{h}'_\arabic*$}]
		\setcounter{enumi}{3}
		\item\label{hp:limit-SV}
		$ \lim\limits_{t \to 0^+}\dfrac{G(t)}{t^p}  = \lim\limits_{t \to + \infty }\dfrac{G(t)}{t^p} =0 $.
	\end{enumerate}
	Then, for each $\lambda > \lambda^*$, where
	\begin{align*}
		\lambda^* = \frac{3}{2}\frac{1}{p k_a^p \|h\|_1 \tilde{K}_R} \inf_{d>0} \dfrac{d^p}{G(d)},
	\end{align*}
	problem \eqref{problem-SV} admits at least three distinct nonnegative weak solutions, at least two of which are nontrivial.
<!-- end theorem-5 -->
<!-- proof-4: body_md -->
First of all, note that by \eqref{hp-problem-SV} one has $G(d)>0$ for $d$ large enough, so that $\lambda^*<+\infty$ (with the convention $\frac{d^p}{G(d)}=+\infty$ if $G(d)=0$). Now, fix $\lambda > \lambda^*$, so there exists $d>0$ such that
	\[
		\lambda> \frac{3}{2}\frac{1}{p k_a^p \|h\|_1 \tilde{K}_R} \dfrac{d^p}{G(d)}.
	\]
	From the first part of hypothesis \eqref{hp:limit-SV} it follows that there is $c_1 \in \left(0, \frac{d}{2^\frac{1}{p}}\right)$ such that
	\[
		\dfrac{G(c_1)}{c_1^p} < \dfrac{1}{\lambda} \dfrac{1}{p k_a^p \|h\|_1} < \dfrac{2}{3} \tilde{K}_R \dfrac{G(d)}{d^p},
	\]
	where the last inequality follows from the choice of $d$; namely, assumption \eqref{hp2:cond1-SV} is verified. On the other hand, from the second part of \eqref{hp:limit-SV} we get that there exists $c_2 > d$ such that
	\[
		\dfrac{G(c_2)}{c_2^p} < \dfrac{1}{2} \dfrac{1}{\lambda} \dfrac{1}{p k_a^p \|h\|_1} < \dfrac{1}{3} \tilde{K}_R \dfrac{G(d)}{d^p},
	\]
	which implies that assumption \eqref{hp2:cond2-SV} holds true. Finally, by the choice of $c_1$ and $c_2$, assumption \eqref{hp2:costanti} is satisfied. Moreover, $\lambda \in (\tilde{\lambda}'_1, \tilde{\lambda}'_2)$: indeed, $\lambda > \tilde{\lambda}'_1$ by the choice of $d$, while the inequalities satisfied by $\frac{G(c_1)}{c_1^p}$ and $\frac{G(c_2)}{c_2^p}$ give $\lambda < \tilde{\lambda}'_2$. Hence, Corollary \ref{th2-SV} guarantees the existence of at least three distinct nonnegative weak solutions for any $\lambda > \lambda^*$. In addition, the first limit in \eqref{hp:limit-SV} implies $g(0)=0$: indeed, $\frac{G(t)}{t}\to g(0)$ as $t\to0^+$, while $\frac{G(t)}{t}=\frac{G(t)}{t^p}\,t^{p-1}\to0$. Hence, $u\equiv0$ is a solution of \eqref{problem-SV} and, by Remark \ref{rem:nontrivial}, at least two of the three solutions are nontrivial.

\label{rem:BOV}
	Let $N=1$, $a\equiv B$ and $h=\alpha$, as in \cite{Bonanno-O'Regan-Vetro-2017}.
