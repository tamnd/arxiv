\label{def:spatial-mixing}
    Fix a graph $G = (V, E)$ and a Gibbs distribution on $G$.
    We say $\mu$ exhibits \emph{strong spatial mixing} of rate $\rho < 1$ if for two different partial assignments $\sigma, \tau$ on a vertex subset $\Lambda \subseteq V$ and $v \in V \setminus \Lambda$,
    \begin{align*}
        \ab\|\mu(v \gets \cdot \;|\; \Lambda \gets \sigma) - \mu(v \gets \cdot \;|\; \Lambda \gets \tau)\|_{\mathrm{TV}} \le \ensuremath{\mathrm{poly}}\left(\ab|V|\right) \rho^{\ensuremath{\mathrm{dist}}_G(v, \Delta(\sigma, \tau))}
    \end{align*}
    where $\Delta(\sigma, \tau) \coloneqq \left\{v \in \Lambda \;\middle\vert\; \sigma(v) \neq \tau(v)\right\}$ and $\ensuremath{\mathrm{dist}}_G(\cdot, \cdot)$ is the distance metric on $G$.

<!-- lemma-1: body_md -->
\label{lem:TV-distance-by-Hilbert-distance}
    For two positive probability vectors $p, p'$ supported on a common state space $\Omega$, it holds that
    \begin{align*}
        \ab\|p - p'\|_{\mathrm{TV}} \le \tanh\ab(\frac{d_H(p, p')}{4}) \le \frac{d_H(p, p')}{4}.
    \end{align*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
When $p = p'$, the lemma is trivial.
    Otherwise, define the ratio vector of $p$ and $p'$ as $R(c) = p(c)/p'(c)$, and set $m = \min_{c \in \Omega} R(c)$ and $M = \max_{c \in \Omega} R(c)$.
    It is clear that $m \le 1 \le M$.
    We compute the total variation distance as
    \begin{align*}
        \ab\|p - p'\|_{\mathrm{TV}} = \sum_{c \in \Omega} \max\left\{p(c) - p'(c), 0\right\} = \sum_{c \in \Omega} p'(c) \max\left\{R(c) - 1, 0\right\}.
    \end{align*}
    By the secant-line bound
    \begin{align*}
        \max \left\{0, x - 1\right\} \le \frac{M - x}{M - m} \max\left\{0, m - 1\right\} + \frac{x - m}{M - m} \max\left\{0, M - 1\right\} = \frac{(M - 1)(x - m)}{M - m},
    \end{align*}
    it holds that
    \begin{align*}
        \ab\|p - p'\|_{\mathrm{TV}} \le \sum_{c \in \Omega} p'(c) \frac{(M - 1)(R(c) - m)}{M - m} = \frac{M - 1}{M - m} (1 - m).
    \end{align*}
    Observe that $M/m = {\mathrm e}^{t}$ where $t = d_H(p, p')$.
    By the AM-GM inequality ${\mathrm e}^t m + \frac{1}{m} \ge 2 {\mathrm e}^{t/2}$, we have
    \begin{align*}
        \ab\|p - p'\|_{\mathrm{TV}} \le \frac{{\mathrm e}^t + 1 - ({\mathrm e}^tm + 1/m)}{{\mathrm e}^t - 1} \le \frac{{\mathrm e}^t - 2{\mathrm e}^{t/2} + 1}{{\mathrm e}^t - 1} = \frac{{\mathrm e}^{t/2} - 1}{{\mathrm e}^{t/2} + 1} = \tanh(t/4).
    \end{align*}
    This proves the first inequality.
    For the second inequality, just observe that $\tanh(x) \le x$ on $[0, \infty)$ and we conclude it.

    For every $d \ge 1$ and $q > \max\;\left\{d, 3\right\}$, with probability $1 - o_n(1)$ over $G \sim \mathcal G(n, d/n)$, $G$ is $q$-colorable and a proper $q$-coloring on $G$ can be output in $\ensuremath{\mathrm{poly}}\left(n, q\right)$ bit operations.

<!-- proof-2: body_md -->
By \cite{PSW96}, with probability $1 - o_n(1)$, $G$ has no $q$-core, \emph{i.e.}\xspace, an induced subgraph of minimal degree at least $q$.
    Consider the following algorithm:
    \begin{enumerate}
        \item initialize $H = G$, $S = \emptyset$ and $Q = \left\{v \in V(H) \;:\; \deg_H(v) \le q - 1\right\}$;
        \item repeat the following process until $Q = \emptyset$:
        \begin{enumerate}
            \item pick a smallest vertex $v \in Q$;
            \item append $v$ to the tail of $S$ and remove $v$ from $Q$;
            \item remove $v$ from $H$ and delete all incident edges to $v$;
            \item $Q \gets Q \cup \left\{u \in V(H) \;: \; \deg_H(u) \le q - 1\right\}$;
        \end{enumerate}
        \item from the tail of $S$ to the head, color the vertex with the smallest color which is not used by assigned neighbors.
    \end{enumerate}
    When $G$ has no $q$-core, it holds that when $Q = \emptyset$, $H$ is empty and thus $S$ contains all vertices in $V(G)$.
    The feasibility of the coloring comes directly from the order to construct $S$ and the total bit operations is at most $O(\ab|V(G)| + \ab|E(G)| + q(\ab|V(G)| + \ab|E(G)|) \ln q) = \ensuremath{\mathrm{poly}}\left(n, q\right)$.

\label{def:pinnings}

    \end{itemize}

<!-- lemma-4: body_md -->
\label{lem:spin-ratio-recursion}
    Given an instance $\Phi = (G = (V, E), [q], \boldsymbol A, \boldsymbol\lambda)$, for every vertex $v \in V$ and $a, b \in [q]$, it holds that
    \begin{align*}
        \frac{\mu_\Phi(v \gets a)}{\mu_\Phi(v \gets b)} = \frac{\lambda_v(a)}{\lambda_v(b)} \prod_{i = 1}^d \frac{\sum_{c \in [q]} A_{(v, v_i)}(a, c) \mu_{\Phi_i}(v_i \gets c)}{\sum_{c \in [q]} A_{(v, v_i)}(b, c) \mu_{\Phi_i}(v_i \gets c)}.
    \end{align*}
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
The proof is similar to \cite[Proposition 1]{LY13}.
    For simplicity, we use $A_i$ to denote $A_{(v, v_i)}$.
    For a partial assignment $\pi$ on $V \setminus \left\{v\right\}$, define
    \begin{align*}
        W(\pi) \coloneqq \prod_{u \in V \setminus \left\{v\right\}} \lambda_u(\pi(u)) \prod_{(u, w) \in E \setminus \left\{(v, v_1), \ldots, (v, v_d)\right\}} A_{(u, w)}(\pi(u), \pi(w)).
    \end{align*}
    For every $c \in [q]$, let
    \begin{align*}
        F(c) \coloneqq \sum_{\pi : V \setminus \left\{v\right\} \to [q]} W(\pi) \prod_{i = 1}^d A_{(v, v_i)}(c, \pi(v_i)).
    \end{align*}
    By \eqref{eq:spin-Gibbs-distribution},
    \begin{align*}
        \frac{\mu_\Phi(v \gets a)}{\mu_\Phi(v \gets b)} = \frac{\lambda_v(a) F(a)}{\lambda_v(b) F(b)}.
    \end{align*}
    To telescope $F(a)/F(b)$, for $i = 0, \ldots, d$, define
    \begin{align} \label{eq:spin-telescoping}
        F_i \coloneqq \sum_{\pi : V \setminus \left\{v\right\} \to [q]} W(\pi) \prod_{j = 1}^i A_{(v, v_j)}(a, \pi(v_j)) \prod_{j = i + 1}^d A_{(v, v_j)}(b, \pi(v_j)).
    \end{align}
    Then it holds that $F(a)/F(b) = \prod_{i = 1}^d F_i/F_{i - 1}$ since $F(a) = F_d$ and $F(b) = F_0$.
    For a fixed $i = 1, \ldots, d$, write
    \begin{align*}
        W_i(\pi) = W(\pi) \prod_{j = 1}^{i - 1} A_{(v, v_j)}(a, \pi(v_j)) \prod_{j = i + 1}^d A_{(v, v_j)}(b, \pi(v_j)).
    \end{align*}
    Observe that the partition function of $\Phi_i$ is equal to
    \begin{align*}
        \mathcal{Z}(\Phi_i) = \sum_{\pi : V \setminus \left\{v\right\} \to [q]} W_i(\pi)
    \end{align*}
    and for the marginal vector $p_i$ on $v_i$,
    \begin{align*}
        \forall c \in [q], \quad p_i(c) = \frac1{\mathcal{Z}(\Phi_i)} \sum_{\pi : V \setminus \left\{v\right\} \to [q], \pi(v_i) = c} W_i(\pi).
    \end{align*}
    Then we have
    \begin{align*}
        F_i &= \sum_{\pi : V \setminus \left\{v\right\} \to [q]} A_{(v, v_i)}(a, \pi(v_i)) W_i(\pi) \\
        &= \mathcal{Z}(\Phi_i) \sum_{c \in [q]} A_{(v, v_i)}(a, c) p_i(c)
    \end{align*}
    and similarly $F_{i - 1} = \mathcal{Z}(\Phi_i) \sum_{c \in [q]} A_{(v, v_i)}(b, c) p_i(c)$.
    Therefore
    \begin{align*}
        \frac{F_i}{F_{i - 1}} = \frac{\sum_{c \in [q]} A_{(v, v_i)}(a, c) p_i(c)}{\sum_{c \in [q]} A_{(v, v_i)}(b, c) p_i(c)}.
    \end{align*}
    Plugging it into \eqref{eq:spin-telescoping}, we conclude the lemma.

Fix the boundary set $\Lambda$ and two different partial assignments $\sigma, \tau$ on $\Lambda$.
    Let $\Gamma = \Delta(\sigma, \tau)$, \emph{i.e.}\xspace, the discrepancy of $\sigma$ and $\tau$.
    Set $\overline \Delta = 4\operatorname*{arctanh} c$ and $\rho = D \, c$.
    For convenience, denote by $H_\Phi(v, \Lambda, \sigma, \tau)$ the Hilbert distance $d_H(\mu_\Phi(v \gets \cdot \;|\; \Lambda \gets \sigma), \mu_\Phi(v \gets \cdot \;|\; \Lambda \gets \tau))$.

    We will show that
    \begin{align} \label{eq:spatial-mixing-Hilbert-decay}
        H_\Phi(v, \Lambda, \sigma, \tau) \le \frac{K n^\alpha D \overline{\Delta}}{1 - \rho} \rho^{\ensuremath{\mathrm{dist}}_G(v, \Gamma) - 1}
    \end{align}
    and therefore by \Cref{lem:TV-distance-by-Hilbert-distance}, we conclude the theorem.

    We first introduce \emph{local activities} on $V \setminus \Lambda$ after a pinning $\pi$ on $\Lambda$.
    Let
    \begin{align*}
        \forall x \in V \setminus \Lambda, c \in [q], \quad \lambda_x^\pi(c) = \lambda_x(c) \prod_{y \in \Lambda \cap N_G(x)} A_{(x, y)}(c, \pi(y)).
    \end{align*}
    The local Hilbert discrepancy is induced by local activities conditional on different pinnings:
    \begin{align*}
        H^{\mathsf{local}}(x) \coloneqq d_H(\lambda_x^\sigma, \lambda_x^\tau).
    \end{align*}
    By direct calculation on $d_H(\lambda_x^\sigma, \lambda_x^\tau)$, it holds that
    \begin{align*}
        H^{\mathsf{local}}(x) \le \sum_{y \in \Gamma \cap N_G(x)} \Delta_B\ab(A_{(x, y)}).
    \end{align*}
    We remark here $H^{\mathsf{local}}(x) = 0$ unless $x$ is adjacent to $\Gamma$.

    To show \eqref{eq:spatial-mixing-Hilbert-decay}, we specify a sub-instance $\mathcal I$ by a vertex subset $S = S_{\mathcal I}$ and an external field $\boldsymbol h = (h_x : [q] \to \mathbb R_{> 0})_{x \in S}$ with the original edge interactions restricted on $G[S]$.
    We say a pair of instances $(\mathcal I = (S, \boldsymbol h), \mathcal I' = (S, \boldsymbol h'))$ are \emph{admissible} if $S \subseteq V \setminus \Lambda$ and for every $x \in S$,
    \begin{align*}
        \forall c \in [q], \quad \frac{h_x(c)}{h_x'(c)} = \frac{\lambda_x^\sigma(c)}{\lambda_x^\tau(c)}.
    \end{align*}
    Consider the Hilbert distance $d_H(\mu_{\mathcal I}(x \gets \cdot), \mu_{\mathcal I'}(x \gets \cdot))$.
    Assume that $x_1, \ldots, x_d$ be the neighbors of $x$ in $G[S]$.
    For two positive probability $p, p'$ and $a, b \in [q]$, by \Cref{lem:Birknoff-contraction},
    \begin{align*}
        \ab|\ln\frac{\sum_{c \in [q]} A_{(x, x_i)}(a, c) p(c)}{\sum_{c \in [q]} A_{(x, x_i)}(b, c) p(c)} - \ln\frac{\sum_{c \in [q]} A_{(x, x_i)}(a, c) p'(c)}{\sum_{c \in [q]} A_{(x, x_i)}(b, c) p'(c)}| &\le d_H\ab(A_{(x, x_i)}p, A_{(x, x_i)}p') \\
        &\le c_B\ab(A_{(x, x_i)}) d_H(p, p').
    \end{align*}
    Apply \Cref{lem:spin-ratio-recursion}, and we obtain that
    \begin{align*}
        d_H(\mu_{\mathcal I}(x \gets \cdot), \mu_{\mathcal I'}(x \gets \cdot)) \le H^{\mathsf{local}}(x) + \max_{a, b \in [q]} \sum_{i = 1}^d c_B\ab(A_{(x, x_i)}) d_H(\mu_{\mathcal I_i}(x_i \gets \cdot), \mu_{\mathcal I'_i}(x_i \gets \cdot))
    \end{align*}
    where $\mathcal I_i, \mathcal I'_i$ are generated by \Cref{def:pinnings} from $\mathcal I$, $\mathcal I'$ respectively.

        \textsc{Inf}(S, x) = H^{\mathsf{local}}(x) + \sum_{i = 1}^d c_B\ab(A_{(x, x_i)}) \textsc{Inf}(S \setminus \left\{x\right\}, x_i)
    \end{align}
    with boundary condition $\textsc{Inf}(S, x) = H^{\mathsf{local}}(x)$ for $S = \left\{x\right\}$ or $x$ is isolated in the graph $G[S]$.
    By induction, $d_H(\mu_{\mathcal I}(x \gets \cdot), \mu_{\mathcal I'}(x \gets \cdot))$ is upper bounded by $\textsc{Inf}(S, x)$ for every pair of admissible $(\mathcal I, \mathcal I')$.
    In addition, note that the instance $\Phi$ with pinning $\pi$ on $\Lambda$ is specified by $V \setminus \Lambda$ and $\boldsymbol{\lambda}^\pi = (\lambda_x^\pi)_{x \in V \setminus \Lambda}$.
    Therefore, it holds that
    \begin{align*}
        H_\Phi(v, \Lambda, \sigma, \tau) \le \textsc{Inf}(V \setminus \Lambda, v).
    \end{align*}
    Unfold \eqref{eq:spatial-mixing-recursive-bound} and substitute the bound of $H^{\mathsf{local}}(x)$.
    Every non-zero term is encoded by a self-avoiding walk $P = (v = x_0, \ldots, x_{\ell - 1})$ in $G[V \setminus \Lambda]$ and a vertex $y \in \Gamma$ incident to $x_{\ell - 1}$.
    Append $y$ to the tail of $P$.
    Therefore $P$ is a self-avoiding walk from $v$ to $\Gamma$ of length $\ell$ with contribution at most $c^{\ell - 1} \cdot \overline{\Delta}$.
    Hence,
    \begin{align*}
        \textsc{Inf}(V \setminus \Lambda, v) &\le \sum_{\ell = \ensuremath{\mathrm{dist}}_G(v, \Gamma)}^{n - 1} c^{\ell - 1} \cdot \overline{\Delta} \cdot \ab|{\mathrm{SAW}}_G(v, \ell)| \\
        &\le K n^\alpha D \cdot \overline{\Delta} \sum_{\ell \ge \ensuremath{\mathrm{dist}}_G(v, \Gamma) - 1} \rho^\ell \\
        &= \frac{K n^\alpha D \overline \Delta}{1 - \rho} \rho^{\ensuremath{\mathrm{dist}}_G(v, \Gamma) - 1}.
    \end{align*}
    Combining all things together, we conclude the theorem.

<!-- definition-3: body_md -->
\label{def:spin-polytope}
    The variables of $\mathscr{P}_t(\Phi, v)$ is $(z_c)_{c \in [q]}$.
    When $v$ is an isolated vertex in $G$, $\mathscr{P}_t(\Phi, v)$ is the exact polytope
    \begin{align*}
        \forall c \in [q], \quad z_c = \frac{\lambda_v(c)}{\sum_{k \in [q]} \lambda_v(k)}.
    \end{align*}
    
    When $v$ is not isolated, assume that its neighbors in $G$ are $v_1, \ldots, v_d$ where $d = \deg_G(v)$.
    The polytope $\mathscr{P}_t(\Phi, v)$ at first contains:
    \begin{itemize}
        \item \textbf{Non-negativity.}
        \begin{align} \label{eq:spin-LFP-non-negativity}
            \forall c \in [q], \quad z_c \ge 0.
        \end{align}
        \item \textbf{Normalization.}
        \begin{align} \label{eq:spin-LFP-normalization}
            \sum_{c \in [q]} z_c = 1.
        \end{align}
    \end{itemize}
    
    We separate cases based on the value of the depth $t$.
    \begin{itemize}
        \item $t = 0$.
        For every distinct spins $a, b \in [q]$, the polytope contains
        \begin{align} \label{eq:spin-LFP-depth-zero}
            L_{a, b}^0 \cdot z_b \le z_a \le U_{a, b}^0 \cdot z_b
        \end{align}
        where
        \begin{align*}
            L_{a, b}^0 \coloneqq \frac{\lambda_v(a)}{\lambda_v(b)} \prod_{i = 1}^d \min_{c \in [q]}\; \frac{A_{(v, v_i)}(a, c)}{A_{(v, v_i)}(b, c)}, \quad
            U_{a, b}^0 \coloneqq \frac{\lambda_v(a)}{\lambda_v(b)} \prod_{i = 1}^d \max_{c \in [q]}\; \frac{A_{(v, v_i)}(a, c)}{A_{(v, v_i)}(b, c)}.
        \end{align*}

        \item $t > 0$.
        For every distinct spins $a, b \in [q]$ and $i = 1, \ldots, d$, construct an instance $\Phi_i = \Phi_{i, a, b}^v$.
        Recursively obtain the polytope $\mathscr{P}_i = \mathscr{P}_{t - 1}(\Phi_i, v_i)$.
        Solve the following two LFPs:
        \begin{align*}
            \ell_{i, a, b}^* \coloneqq \min_{z \in \mathscr{P}_i}\; \frac{\sum_{c \in [q]} A_{(v, v_i)}(a, c) z_c}{\sum_{c \in [q]} A_{(v, v_i)}(b, c) z_c}, \quad
            u_{i, a, b}^* \coloneqq \max_{z \in \mathscr{P}_i}\; \frac{\sum_{c \in [q]} A_{(v, v_i)}(a, c) z_c}{\sum_{c \in [q]} A_{(v, v_i)}(b, c) z_c}.
        \end{align*}
        and set $\underline\ell_{i, a, b} = \mathsf{round}_P^-(\ell_{i, a, b}^*), \overline u_{i, a, b} = \mathsf{round}_P^+(u_{i, a, b}^*)$.
        Setting $L_{a, b}^t = \frac{\lambda_v(a)}{\lambda_v(b)} \prod_{i = 1}^d \underline\ell_{i, a, b}$ and $U_{a, b}^t = \frac{\lambda_v(a)}{\lambda_v(b)} \prod_{i = 1}^d \overline u_{i, a, b}$, the polytope then contains
        \begin{align} \label{eq:spin-LFP-ratio-recursion}
            L_{a, b}^t \cdot z_b \le z_a \le U_{a, b}^t \cdot z_b.
        \end{align}
    \end{itemize}

<!-- fact-1: body_md -->
\label{fact:spin-polytope-soundness}
    At every depth the recursive construction is well-defined, the true marginal vector at $v$ belongs to $\mathscr{P}_t(\Phi, v)$ and the bit complexity to encode $\mathscr{P}_t(\Phi, v)$ is at most $O(q^3(\ab|\Phi| + nP))$.
<!-- end fact-1 -->
<!-- proof-5: body_md -->
We prove the fact by induction.
    When the polytope contains only an exact point (the case $v$ is isolated), the fact is trivial.
    Otherwise, at depth $0$, it is direct to see the true marginal vector $p_v$ at $v$ satisfies constraints \eqref{eq:spin-LFP-non-negativity} and \eqref{eq:spin-LFP-normalization}.
    By \Cref{lem:spin-ratio-recursion}, it is easy to verify $p_v$ satisfies \eqref{eq:spin-LFP-depth-zero}.
    The bit complexity to encode $\mathscr{P}_0(\Phi, v)$ is at most $O(q^3 \ab|\Phi|)$ since we have $q$ variables and $O(q^2)$ constraints with each constraint encoded with at most $O(\ab|\Phi|)$ bits.

    Suppose that the assertion holds through depth $t - 1$.
    It is trivial to see the true marginal vector $p_v$ satisfies \eqref{eq:spin-LFP-non-negativity} and \eqref{eq:spin-LFP-normalization}.
    Since the true marginal vector $p_i$ belongs to $\mathscr{P}_i$ by induction, by \Cref{lem:spin-ratio-recursion}, $p_v$ also satisfies the constraint \eqref{eq:spin-LFP-ratio-recursion} even after applying rounding oracles.
    Therefore we conclude that $p_v \in \mathscr{P}_t(\Phi, v)$.
    To compute the number of bits to encode $\mathscr{P}_t(\Phi, v)$, note that we have $q$ variables and $O(q^2)$ linear constraints.
    For a constraint \eqref{eq:spin-LFP-ratio-recursion}, note that for each $i = 1, \ldots, d$, it holds that
    \begin{align*}
        \min_{c \in [q]} \frac{A_{(v, v_i)}(a, c)}{A_{(v, v_i)}(b, c)} \le \ell_{i, a, b}^* \le u_{i, a, b}^* \le \max_{c \in [q]} \frac{A_{(v, v_i)}(a, c)}{A_{(v, v_i)}(b, c)}.
    \end{align*}
    Hence, it takes $O(\mathrm{bit}(A_{(v, v_i)}) + P)$ bits to encode the rounded rational number $\underline \ell_{i, a, b}, \overline u_{i, a, b}$ and thus $O(\mathrm{bit}(\lambda_v) + dP + \sum_{i = 1}^d \mathrm{bit}(A_{(v, v_i)}))$ bits to encode $L_{a, b}^t$ and $U_{a, b}^t$.
    Therefore, the bit complexity to encode $\mathscr{P}_t(\Phi, v)$ is at most $O(q^3(\ab|\Phi| + nP))$.

\label{prop:spin-accuracy-recursion}
    Fix an admissible feasible query $(\Phi, v, t)$.
    When $v$ is isolated, $\mathrm{diam}_H(\mathscr{P}_t(\Phi, v)) = 0$.
    Otherwise, assume that its neighbors are $v_1, \ldots, v_d$.

            \sum_{i = 1}^d \Delta_B(A_{(v, v_i)}), & t = 0 \\
            \max_{a, b \in [q]} \sum_{i = 1}^d \ab(c_B(A_{v, v_i}) \mathrm{diam}_H(\mathscr{P}_{t - 1}(\Phi_{i, a, b}^v, v_i)) + 2\xi_P), & t > 0
        \end{cases}\;.
    \end{align*}

<!-- proof-6: body_md -->
When $v$ is isolated, $\mathscr{P}_t(\Phi, v)$ contains only a single point and thus $\mathrm{diam}_H(\mathscr{P}_t(\Phi, v)) = 0$.
    Suppose that $v$ is not isolated and its neighbors are $v_1, \ldots, v_d$.
    For $t = 0$, it holds that
    \begin{align*}
        \ln \frac{U_{a, b}^0}{L_{a, b}^0} = \sum_{i = 1}^d \ab(\ln \max_{c, d \in [q]} \frac{A_{(v, v_i)}(a, c)A_{(v, v_i)}(b, d)}{A_{(v, v_i)}(a, d)A_{(v, v_i)}(b, c)}) \le \sum_{i = 1}^d \Delta_B(A_{(v, v_i)}).
    \end{align*}
    Then for every $x, y \in \mathscr{P}_0(\Phi, v)$, it holds that
    \begin{align*}
        d_H(x, y) = \max_{a, b \in [q]} \ln \frac{x(a)y(b)}{x(b)y(a)} \le \max_{a, b \in [q]} \ln\frac{U_{a, b}^0}{L_{a, b}^0} \le \sum_{i = 1}^d \Delta_B(A_{(v, v_i)}).
    \end{align*}
    For $t > 0$, for every $i = 1, \ldots, d$, by \Cref{lem:Birknoff-contraction},
    \begin{align*}
        \ln\frac{u_{i, a, b}^*}{\ell_{i, a, b}^*} &\le \max_{x, y \in \mathscr{P}_{t - 1}(\Phi_i, v_i)} d_H(A_{(v, v_i)}x, A_{(v, v_i)}y) \\
        &\le c_B(A_{(v, v_i)}) \mathrm{diam}_H(\mathscr{P}_{t - 1}(\Phi_i, v_i)).
    \end{align*}
    Together with \eqref{eq:rounding-accuracy}, it holds that
    \begin{align*}
        \ln\frac{U_{a, b}^t}{L_{a, b}^t} \le \sum_{i = 1}^d \ab(c_B(A_{(v, v_i)}) \mathrm{diam}_H(\mathscr{P}_{t - 1}(\Phi_{i, a, b}^v, v_i)) + 2\xi_P).
    \end{align*}
    A similar argument to the case $t = 0$ shows
    \begin{align*}
        \mathrm{diam}_H(\mathscr{P}_t(\Phi, v)) \le \max_{a, b \in [q]} \ln \frac{U_{a, b}^t}{L_{a, b}^t}
    \end{align*}
    and thus we conclude the full proposition.

<!-- corollary-2: body_md -->
\label{cor:spin-accuracy}
    Given an instance $\Phi = (G = (V, E), [q], \boldsymbol A, \boldsymbol\lambda)$ such that $G$ satisfies \eqref{eq:polynomial-connective-constant} and $D \cdot c_B(A_e) < 1$, for every integer $R \ge 0$ and $v \in V$, it holds that
    \begin{align} \label{eq:spin-accuracy-bound}
        \mathrm{diam}_H(\mathscr{P}_R(\Phi, v)) \le K n^\alpha D \ab(\overline\Delta \rho^R + \frac{2\xi_P}{1 - \rho})
    \end{align}
    where $c = \max_{e \in E} c_B(A_e)$, $\overline\Delta = 4\operatorname*{arctanh} c$ and $\rho = D \cdot c$.
    Moreover, set
    \begin{align} \label{eq:spin-depth-accuracy-parameters}
        R = R_\varepsilon = \left\lceil \frac{\ln(4Kn^\alpha D\max\left\{1, \overline\Delta\right\}/\varepsilon)}{\ln(1/\rho)}\right\rceil, \quad
        P = P_\varepsilon = \max \; \left\{3, \left\lceil \ln_2 \frac{32Kn^\alpha D}{\varepsilon(1 - \rho)}\right\rceil\right\}.
    \end{align}
    For every feasible point $z \in \mathscr{P}_R(\Phi, v)$, it holds that for every $c \in [q]$,
    \begin{align*}
        {\mathrm e}^{-\varepsilon} \mu_\Phi(v \gets c) \le z_c \le {\mathrm e}^\varepsilon \mu_\Phi(v \gets c).
    \end{align*}

    First we have
    \begin{align*}
        \Delta_B(A_e) = 4\operatorname*{arctanh} c_B(A_e) \le 4\operatorname*{arctanh} c = \overline\Delta.

    Consider the recursion in \Cref{prop:spin-accuracy-recursion}.
    We unfold the recursion into a rooted tree $\mathcal T$ with root $\varrho$.
    A node $\omega$ in $\mathcal T$ at level $0 \le r \le R$ is labelled by a derived instance $\Phi_\omega$ on the underlying graph $G_\omega$, a vertex $v_\omega$ and a remaining depth $R - r$.
    At every node $\omega$ at level $r < R$, we choose any pair of $(a_\omega, b_\omega)$ to attain the maximum in the recursion.

    Construct intermediate instances $\Psi_0, \ldots, \Psi_{\ab|\partial_E B|}$ as: $\Psi_0$ is the instance derived by removing $B$ from $G_\Phi$, $L_\Phi(v)$ from color lists for every $v \in B$ and keeping all remaining color lists unchanged; for every $j = 1, \ldots, \ab|\partial_E B|$, generate $\Psi_j$ from $\Psi_{j - 1}$ by removing $\sigma(u_j)$ from the color list on $v_j$.
    For any intermediate instance $\Psi$, suppose that we need to delete color $c$ from the list of $v$ to obtain the instance $\Psi'$.
    Since $B$ is a permissive block, by \Cref{lem:coloring-marginal-upper-bound},
    \begin{align*}
        \mu_\Psi(v \gets c) \le \frac12.
    \end{align*}
    Therefore,
    \begin{align*}
        \mathcal{Z}(\Psi') &= \mathcal{Z}(\Psi) - \ab|\left\{\sigma~\text{is a proper list-coloring in $\Psi$} \;:\; \sigma(v) = c\right\}| \\
        &= \mathcal{Z}(\Psi) - \mu_\Psi(v \gets c) \cdot \mathcal{Z}(\Psi) \\
        &\ge \mathcal{Z}(\Psi)/2.
    \end{align*}
    Observe that $\mathcal{Z}(\Phi) \le \ab|\Omega_B| \cdot \mathcal{Z}(\Psi_0)$ and $\mathcal{Z}(\Psi_{\partial_E B}) = \mathcal{Z}(\Phi \;|\; B \gets \sigma)$.
    Therefore,
    \begin{align*}
        \mathcal{Z}(\Phi \;|\; B \gets \sigma) \ge 2^{-\ab|\partial_E B|} \cdot \mathcal{Z}(\Psi_0) \ge 2^{-\ab|\partial_E B|} q^{-\ab|B|} \mathcal{Z}(\Phi).
    \end{align*}
    Then we conclude the marginal lower bound.

<!-- definition-4: body_md -->
\label{def:coloring-polytope}
    The variables of $\mathscr{P}_t(\Phi, B)$ is $(z_\sigma)_{\sigma \in \Omega_B}$.
    When $\partial_E B = \emptyset$, $\mathscr{P}_t(\Phi, B)$ is the exact uniform polytope
    \begin{align*}
        \forall \sigma \in \Omega_B, \quad z_\sigma = \frac{1}{\ab|\Omega_B|}.
    \end{align*}
    
    When $\partial_E B \neq \emptyset$, list $\partial_E B$ in order $(u_1, v_1), \ldots, (u_m, v_m)$.
    The polytope $\mathscr{P}_t(\Phi, B)$ at first contains:
    \begin{itemize}
        \item \textbf{Marginal lower bound.}
        \begin{align} \label{eq:coloring-LFP-marginal-lower-bound}
            \forall \sigma \in \Omega_B, \quad z_\sigma \ge b_B = 2^{-m} q^{-\ab|B|}.
        \end{align}
        \item \textbf{Normalization.}
        \begin{align} \label{eq:coloring-LFP-normalization}
            \sum_{\sigma \in \Omega_B} z_\sigma = 1.
        \end{align}
        \item \textbf{Marginal upper bound.}
        \begin{align} \label{eq:coloring-LFP-marginal-upper-bound}
            \forall v \in B \setminus X, c \in L_\Phi(v), \quad \sum_{\sigma \in \Omega_B : \sigma(v) = c} z_\sigma \le \frac{1}{s_v}.
        \end{align}
    \end{itemize}
    
    We separate cases based on the value of the depth $t$.
    \begin{itemize}
        \item $t = 0$.
        For every distinct partial colorings $\sigma, \tau \in \Omega_B$, the polytope contains
        \begin{align} \label{eq:coloring-LFP-depth-zero}
            2^{-m} z_\tau \le z_\sigma \le 2^m z_\tau.
        \end{align}

        \item $t > 0$.
        For every distinct partial colorings $\sigma, \tau \in \Omega_B$ and $i = 1, \ldots, m$, construct an instance $\Phi_i = \Phi_{i, \sigma, \tau}$.
        Recursively obtain the polytope $\mathscr{P}_i = \mathscr{P}_{t - 1}(\Phi_i, B_{\Phi_i}(v_i))$.
        For a probability vector $y \in \mathscr{P}_i$, we define
        \begin{align*}
            p_c(y) \coloneqq \sum_{\pi : \pi(v_i) = c} y_\pi
        \end{align*}
        with $p_c(y) = 0$ if $c$ is absent.
        Solve the following two LFPs:
        \begin{align*}
            \ell_{i, \sigma, \tau}^* \coloneqq \min_{y \in \mathscr{P}_i}\; \frac{1 - p_{\sigma(u_i)}(y)}{1 - p_{\tau(u_i)}(y)}, \quad
            u_{i, \sigma, \tau}^* \coloneqq \max_{y \in \mathscr{P}_i}\; \frac{1 - p_{\sigma(u_i)}(y)}{1 - p_{\tau(u_i)}(y)}.
        \end{align*}
        and set $\underline\ell_{i, \sigma, \tau} = \mathsf{round}_P^-(\ell_{i, \sigma, \tau}^*), \overline u_{i, \sigma, \tau} = \mathsf{round}_P^+(u_{i, \sigma, \tau}^*)$.
        Setting $L_{\sigma, \tau}^t = \prod_{i = 1}^m \underline\ell_{i, \sigma, \tau}$ and $U_{\sigma, \tau}^t = \prod_{i = 1}^m \overline u_{i, \sigma, \tau}$, the polytope then contains
        \begin{align} \label{eq:coloring-LFP-ratio-recursion}
            L_{\sigma, \tau}^t \cdot z_\tau \le z_\sigma \le U_{\sigma, \tau}^t \cdot z_\tau.
        \end{align}
    \end{itemize}

<!-- proof-11: body_md -->
The proof is similar to \Cref{fact:spin-polytope-soundness}.
    When the polytope contains only an exact point (the case $\partial_E B = \emptyset$), the fact is trivial.
    Otherwise, at depth $0$, it is direct to see the true marginal vector $p_B$ on $B$ satisfies constraints \eqref{eq:coloring-LFP-marginal-lower-bound}, \eqref{eq:coloring-LFP-normalization} and \eqref{eq:coloring-LFP-marginal-upper-bound}.
    By \Cref{lem:coloring-ratio-recursion}, it is easy to verify $p_B$ satisfies \eqref{eq:coloring-LFP-depth-zero} since every denominator is at least $1/2$ and every enumerator is at most $1$.
    The bit complexity to encode $\mathscr{P}_0(\Phi, B)$ is at most $O(q^{4\ab|B|}(\ab|B| + \ab|\partial_E B|))$ since we have $q^{\ab|B|}$ variables and $O(q^{2\ab|B|})$ constraints with each constraint encoded with at most $O(q^{2\ab|B|} (\ab|B| + \ab|\partial_E B|))$ bits.

    It is trivial to see the true marginal vector $p_B$ satisfies \eqref{eq:coloring-LFP-marginal-lower-bound}, \eqref{eq:coloring-LFP-normalization} and \eqref{eq:coloring-LFP-marginal-upper-bound}.
    By induction and \Cref{lem:coloring-ratio-recursion}, it is not hard to see $p_B$ also satisfies the constraint \eqref{eq:coloring-LFP-ratio-recursion} even after applying rounding oracles.
    Therefore we conclude that $p_B \in \mathscr{P}_t(\Phi, B)$.
    To compute the number of bits to encode $\mathscr{P}_t(\Phi, B)$, note that we have $q^{\ab|B|}$ variables and $O(q^{2\ab|B|})$ linear constraints.
    For a constraint \eqref{eq:coloring-LFP-ratio-recursion}, note that for each $i = 1, \ldots, m$, it holds that
    \begin{align*}
        \frac12 \le \ell_{i, \sigma, \tau}^* \le u_{i, \sigma, \tau}^* \le 2.
    \end{align*}
    Hence, it takes $O(P)$ bits to encode the rounded rational number $\underline \ell_{i, \sigma, \tau}, \overline u_{i, \sigma, \tau}$ and thus $O(\ab|\partial_E B| \cdot P)$ bits to encode $L_{\sigma, \tau}^t$ and $U_{\sigma, \tau}^t$.
    Therefore, the bit complexity to encode $\mathscr{P}_t(\Phi, B)$ is at most $O(q^{4\ab|B|}(\ab|B| + \ab|\partial_E B| \cdot P))$.

\label{prop:coloring-accuracy-recursion}

<!-- proof-12: body_md -->
When $\partial_E B = \emptyset$, the rational polytope $\mathscr{P}_t(\Phi, B)$ contains exactly one feasible point representing the uniform distribution on $\Omega_B$.
    Therefore, $\mathrm{diam}_H(\mathscr{P}_t(\Phi, B)) = 0$.

    Suppose that $\partial_E B = \left\{(u_1, v_1), \ldots, (u_m, v_m)\right\}$ for some $m \ge 1$.
    For $t = 0$, by \eqref{eq:coloring-LFP-depth-zero}, for every $\sigma, \tau \in \Omega_B$,
    \begin{align*}
        -m\ln2 \le \ln\frac{z_\sigma}{z_\tau} \le m\ln2.
    \end{align*}
    Thus $\mathrm{diam}_H(\mathscr{P}_0(\Phi, B)) \le 2m\ln2$.
    
    At positive depth $t > 0$, fix a pair of $\sigma, \tau \in \Omega_B$.
    For $i = 1, \ldots, m$, consider the child rational polytope $\mathscr{P}_i = \mathscr{P}_{t - 1}(\Phi_{i, \sigma, \tau}, B_{\Phi_{i, \sigma, \tau}}(v_i))$.
    Since $v_i \notin X$ by the permissiveness of $B$, for every $y \in \mathscr{P}_i$, by \eqref{eq:coloring-LFP-marginal-upper-bound},
    \begin{align*}
        p_c(y) \le \frac{1}{s_{v_i}}.
    \end{align*}

        \ab|\ln\frac{1 - p_{\sigma(u_i)}(y)}{1 - p_{\tau(u_i)}(y)} - \ln\frac{1 - p_{\sigma(u_i)}(y')}{1 - p_{\tau(u_i)}(y')} | \le \frac{d_H(p(y), p(y'))}{s_{v_i} - 1} \le \kappa_{v_i} d_H(y, y')

    where the last inequality holds since $p(\cdot)$ is a weighted average operator.
    Together with the rounding accuracy \eqref{eq:rounding-accuracy},
    \begin{align*}
        \ln\frac{U_{\sigma, \tau}^t}{L_{\sigma, \tau}^t} \le \sum_{i = 1}^m \ab(\kappa_{v_i} \mathrm{diam}_H(\mathscr{P}_i) + 2\xi_P).

    \end{align*}
    which concludes the proposition.

\label{lem:block-walk-weight}
    For all sufficiently large positive integer $n$, with probability at least $1 - n^{-4}$ over $G \sim \mathcal G(n, d/n)$, for every $0 \le \ell < n$,

<!-- proof-16: body_md -->
At first, we define the witness of a permissive block walk.
    For a permissive block walk $B_0, \ldots, B_\ell$ of length $\ell$ with roots $v_0, \ldots, v_\ell$,
    assume that the union of it contains $t$ vertices in $X$ and set $t_j = \ab|B_j \cap X|$.
    The witness is defined as: for every $j = 0, \ldots, \ell$,
    \begin{itemize}
        \item pick a spanning tree every connected component in $B_j \cap X$;
        \item for every spanning tree in $B_j \cap X$, pick an edge connecting $v_j$ and it.
    \end{itemize}
    Note that the witness has exactly $t$ edges.

    For every $j < \ell$, there are at most $t_j + 1$ possible transitions between $B_j$ and $v_{j + 1}$.
    Hence there are at most $\prod_{j = 0}^{\ell - 1} (t_j + 1) \le 2^t$ possible recorded transitions.
    Meanwhile, by the Cayley's formula, for a fixed collection containing $\ell + 1$ roots and additional $t$ vertices in blocks, the number of witness is at most
    \begin{align*}
        (\ell + 1)(\ell + 1 + t)^{t - 1}.
    \end{align*}
    Then for every $t = 0, \ldots, n$, the total number of witnesses with recorded transitions containing $\ell + 1$ roots and additional $t$ vertices in blocks is at most
    \begin{align*}
        n(n - 1) \ldots (n - \ell) \binom{n - \ell - 1}{t} (\ell + 1)(\ell + 1 + t)^{t - 1} 2^t \le \frac{n^{\ell + 1 + t}}{t!} (\ell + 1)(\ell + 1 + t)^{t - 1} 2^t.
    \end{align*}

    Fix a fixed witness with recorded transitions containing $\ell + 1$ roots and additional $t$ vertices in blocks (denoted by $\mathsf W$).
    For a vertex $v \in \mathsf W$ such that it is prescribed to be in $X$, set $f_v = \deg_{\mathsf W}(v)$.
    Conditional on the event $\mathcal E_{\mathsf W}$ that $\mathsf W$ is in $G$, it holds that
    \begin{align*}
        \deg_G(v) = f_v + Y_v, \quad \text{where}~Y_v \sim \mathrm{Bin}(n - 1 - f_v, d/n).
    \end{align*}
    By the Markov inequality,
    \begin{align*}
         
    \ensuremath{\mathbb{P}\ifblank{v \in X \;\middle\vert\; \mathcal E_{\mathsf W}}{}{_{v \in X \;\middle\vert\; \mathcal E_{\mathsf W}}}\left(#2\right)} 
 \le 2^{-(q - 1 - f_v)}  
    \ensuremath{\mathbf{E}\ifblank{2^{Y_v}}{}{_{2^{Y_v}}}\left[#2\right]} 
 \le {\mathrm e}^{-J + f_v\ln 2}.
    \end{align*}
    Finner's inequality for a read-two family tells us that
    \begin{align*}
         
    \ensuremath{\mathbb{P}\ifblank{\text{All $t$ prescribed vertices are in $X$} \;\middle\vert\; \mathcal E_{\mathsf W}}{}{_{\text{All $t$ prescribed vertices are in $X$} \;\middle\vert\; \mathcal E_{\mathsf W}}}\left(#2\right)} 
 &\le \prod_v  
    \ensuremath{\mathbb{P}\ifblank{v \in X \;\middle\vert\; \mathcal E_{\mathsf W}}{}{_{v \in X \;\middle\vert\; \mathcal E_{\mathsf W}}}\left(#2\right)} 
^{1/2} \\
        &\le \exp\ab(-\frac{Jt}{2} + \frac{\ln2}{2} \sum_v f_v) \\
        &\le {\mathrm e}^{-Jt/2} 2^{\ell + t}.
    \end{align*}
    since $\sum_v f_v \le 2(\ell + t)$.
    Therefore,
    \begin{align*}
         
    \ensuremath{\mathbf{E}\ifblank{\mathcal W_\ell(G)}{}{_{\mathcal W_\ell(G)}}\left[#2\right]} 
 &\le \sum_{t \ge 0} \frac{n^{\ell + 1 + t}}{t!} (\ell + 1)(\ell + 1 + t)^{t - 1} 2^t \frac{d^{\ell + t}}{n^{\ell + t}} q^{K(\ell + 1 + t)} {\mathrm e}^{-Jt/2} 2^{\ell + t} \\
        &= nq^K\ab(2dq^K)^\ell \sum_{t \ge 0} \frac{(\ell + 1)(\ell + 1 + t)^{t - 1}}{t!} \ab(4dq^K {\mathrm e}^{-J/2})^t.
    \end{align*}
    To compute the series, we consider the exponential rooted-tree generating function $T$:
    \begin{align*}
        T(x) = \sum_{m \ge 1} m^{m - 1} \frac{x^m}{m!}.
    \end{align*}
    Then $T(x) = x{\mathrm e}^{T(x)}$ and $T(x) \in (0, 1)$.
    Apply the following form of Lagrange inversion (see \cite{Gessel16,SW23} for more details)
    \begin{align*}
        [x^t] F(u) = \frac{1}{t} [u^{t - 1}] F'(u) \phi(u)^t
    \end{align*}
    for $u = x \phi(u)$ with choice $\phi(u) = {\mathrm e}^u, F(u) = {\mathrm e}^{(\ell + 1)u}$ and $u = T(x)$, and we obtain that
    \begin{align*}
        \sum_{t \ge 0} \frac{(\ell + 1)(\ell + 1 + t)^{t - 1}}{t!} x^t = {\mathrm e}^{(\ell + 1)T(x)}.
    \end{align*}
    Hence,
    \begin{align*}
         
    \ensuremath{\mathbf{E}\ifblank{\mathcal W_\ell(G)}{}{_{\mathcal W_\ell(G)}}\left[#2\right]} 
 \le nq^K\ab(2dq^K)^\ell {\mathrm e}^{(\ell + 1) T(y)}.
    \end{align*}
    Consider the fix-point iteration
    \begin{align*}
        T_0 = 0, \quad \forall k \ge 1,\; T_k = y {\mathrm e}^{T_{k - 1}}.
    \end{align*}
    Since $T_1 = y \ge T_0$ and $x \mapsto y {\mathrm e}^x$ is increasing, by induction it holds that $T_0 \le T_1 \le T_2 \le \cdots$.
    For all $0 \le x \le 2y \le (1/2{\mathrm e})$, it holds that $y{\mathrm e}^x \le y{\mathrm e}^{2y} \le 2y$.
    Therefore $\lim_{k \to \infty} T_k = T(y)$ and $0 \le T_k \le 2y$ for every $k \ge 0$.
    This means $T(y) \le 2y$.
    Then by the Markov inequality at the threshold $n^7\ab(4dq^K{\mathrm e}^{2y})^\ell$, as long as $n \ge q^K {\mathrm e}^{2y}$, it holds that
    \begin{align*}
         
    \ensuremath{\mathbb{P}\ifblank{\exists \ell < n, \mathcal W_\ell(G) \ge n^7 \ab(4dq^K{\mathrm e}^{2y})^\ell}{}{_{\exists \ell < n, \mathcal W_\ell(G) \ge n^7 \ab(4dq^K{\mathrm e}^{2y})^\ell}}\left(#2\right)} 
 \le \sum_{\ell = 0}^{n - 1} n^2 n^{-7} \cdot 2^{-\ell} \le n^{-4}
    \end{align*}
    concluding the lemma.

Assume the joint event of \Cref{lem:random-graph-colorability,lem:path-weight-concentration,lem:exceptional-component-size,lem:exceptional-incident-volume,lem:block-walk-weight}.
    The event occurs with probability at least $1 - o_n(1) - 5n^{-4} = 1 - o_n(1)$.

    Set $R = R_{\varepsilon/(32n)}$ and $P = P_{\varepsilon/(32n)}$ like \eqref{eq:coloring-depth-accuracy-parameters}.
    It is clear to verify that $R = O(\ln{(n/\varepsilon)})$ and $P = O(\ln{(n/\varepsilon)})$ where the hidden constants depend only on $\eta, d$ and $q$.
    If $R \ge n$, it must hold that $\varepsilon = {\mathrm e}^{-\Omega(n)}$ and we enumerate all $q^n$ assignments and check whether it is a proper coloring.
    This is a deterministic algorithm to exactly compute the number of proper colorings in $\ensuremath{\mathrm{poly}}\left(n, \varepsilon^{-1}\right)$ bit operations.

    Otherwise, using the algorithm in \Cref{lem:random-graph-colorability}, we obtain a proper $q$-coloring $\sigma$ of $G$.
    The approximation scheme $\mathsf A_{\eta, d, q}$ runs as following:
    \begin{enumerate}
        \item initialize $\Phi = (G_\Phi = G, (L_\Phi(v) = [q])_{v \in V})$;
        \item process the component of $G[X]$ in a preset order:
        \begin{enumerate}
            \item for the current component $B$, construct the rational polytope $\mathscr{P} = \mathscr{P}_R(\Phi, B)$ and record $\widehat p_{\sigma(B)} = \min_{z \in \mathscr{P}} z_{\sigma(B)}$;
            \item pin $B$ to the partial coloring $\sigma(B)$ in $\Phi$;
        \end{enumerate}
        \item process the remaining vertex of $G$ in a preset order:
        \begin{enumerate}
            \item for the current vertex $v$, construct the rational polytope $\mathscr{P} = \mathscr{P}_R(\Phi, v)$ and pick $\widehat p_{\sigma(v)} = \min_{z \in \mathscr{P}} z_{\sigma(v)}$;
            \item pin $v$ to color $\sigma(v)$ in $\Phi$;
        \end{enumerate}
        \item output the rational product of reciprocals of recorded rational numbers.
    \end{enumerate}

    We first show the accuracy of the algorithm.
    Suppose that the instances in the algorithm are $\Phi = \Phi_0, \ldots, \Phi_M = \varnothing$, the true values of the marginal probabilities are $p_0, \ldots, p_{M - 1}$ and the estimations by the algorithm are $\widehat p_0, \ldots, \widehat p_{M - 1}$.
    Note that all needed analysis holds through the running of $\mathsf A_{\eta, d, q}$.
    By \Cref{lem:coloring-accuracy-bound}, it holds that ${\mathrm e}^{-\varepsilon/(16n)} \le \widehat p_j/p_j \le 1$.
    Hence
    \begin{align*}
        1 \le \frac{\prod_{j = 0}^{M - 1} \widehat p_j^{-1}}{\ab|\Omega(G, q)|} = \frac{\prod_{j = 0}^{M - 1} \widehat p_j^{-1}}{\prod_{j = 0}^{M - 1} p_j^{-1}} \le {\mathrm e}^{\varepsilon/16} \le 1 + \varepsilon.
    \end{align*}

    Now we show the bit complexity.
    For a linear-fractional programming with rounding process in the recursive construction of rational polytopes, it contains $O(q^{\ab|B|})$ variables and $O(q^{2\ab|B|})$ linear constraints.
    Using a crude bound for solving the linear-fractional programming, it cost at most
    \begin{align*}
        q^{50\ab|B|} (nP)^{40}
    \end{align*}
    bit operations to finish the linear-fractional programming task in one call of a child polytope.

    We turn to the total bit complexity to estimate $p_i$.
    Note that a recursive branch of the computation is encoded by a starting block $B$, a block permissive block walk of length at most $R$ and pairs of partial colorings on blocks.
    Therefore, the cost to solve all linear-fractional programmings is at most
    \begin{align}\label{eq:coloring-block-cost}
        q^{40\ab|B|} \cdot \ab(\sum_{v \in B} \deg_G(v)) (nP)^{40} \cdot (nP)^{40} \sum_{\text{a permissive block walk} \atop B_0, \ldots, B_\ell : \ell \le R} \prod_{j = 0}^{\ell} q^{2\ab|B_j|} \cdot q^{50\ab|B_j|}.
    \end{align}
    By \Cref{lem:exceptional-component-size,lem:exceptional-incident-volume,lem:block-walk-weight}, $\ab|B|$ is of size at most $4\ln n$ and there are at most $C_{\mathrm{vol}} \ln n$ roots to start permissive block walks.

    \end{align*}
    where $C$ depends only on $\eta, d$ and $q$.
    Then we conclude the desired bit complexity of $\mathsf A_{\eta, d, q}$

\label{lem:coloring-parameter-choice}
For every fixed $\eta\in(0,1)$, there exists $d_0=d_0(\eta)\ge8$
such that, for every $d\ge d_0$ and every integer
$q\ge(2+\eta)d$, the parameters defined in
\eqref{eq:coloring-degree-parameters},
\eqref{eq:coloring-tail-parameters}, and
\eqref{eq:coloring-block-parameters} satisfy
\eqref{eq:coloring-threshold-estimates},
\eqref{eq:coloring-path-parameter-estimate},
\eqref{eq:coloring-component-parameter-estimate}, and
\eqref{eq:coloring-block-parameter-estimate}.

<!-- proof-19: body_md -->
Recall that $K=200$ is an absolute constant.
Set $j_\eta=(2+\eta)\ln2-1>0$.
Choose $d_0\ge8$ so that, for every $d\ge d_0$,
\begin{equation}\label{eq:coloring-d-setting}
\begin{aligned}
    \eta d &\ge24,\\
    d{\mathrm e}^{-dh(\eta/6)/2} &\le \frac{\eta}{2(3+\eta)},\\
    4{\mathrm e} d{\mathrm e}^{-dh(\eta/6)/4} &\le1,\\
    d &\ge \frac{2K}{(2+\eta)\ln2},\\
    j_\eta d-\ln2
    &\ge 2\ln(16{\mathrm e})+2(K+1)\ln d+2K\ln(2+\eta).
\end{aligned}
\end{equation}
Such a choice exists because $h(\eta/6)>0$ and $j_\eta>0$:
the exponential terms dominate the polynomial factors,
and the linear term in the last inequality dominates $\ln d$.
All these conditions depend only on $\eta$ and the fixed constant $K$.

First, $\delta\ge\eta/6$ and $\delta d\ge4$.
Since $(d+q)/3=(1+2\delta)d$, we have
\[
    D-3>(1+2\delta)d-4\ge(1+\delta)d.
\]
Moreover,
\[
    q-D-1
    \ge \frac{2q-d}{3}-1
    \ge \frac{(3+2\eta)d}{3}-1
    \ge \frac{(3+\eta)d}{3}.
\]
In particular, $D<q-1$ and
\[
    d\gamma\le\frac{3}{3+\eta}.
\]
This proves \eqref{eq:coloring-threshold-estimates}.

Next, $h'(x)=\ln(1+x)>0$ for $x>0$, so
$r\le{\mathrm e}^{-dh(\eta/6)}$.
Consequently,
\[
    d(\gamma+\sqrt r)
    \le \frac{3}{3+\eta}+\frac{\eta}{2(3+\eta)}
    =\lambda_0<\rho<1,
\]
and
\[
    {\mathrm e} d r^{1/4}
    \le {\mathrm e} d{\mathrm e}^{-dh(\eta/6)/4}
    \le\frac14.
\]
These are \eqref{eq:coloring-path-parameter-estimate}
and \eqref{eq:coloring-component-parameter-estimate}.

Finally, put $q_0=(2+\eta)d$.
By \eqref{eq:coloring-d-setting}, $q_0\ge2K/\ln2$.
The function $u\mapsto u^K{\mathrm e}^{-u\ln2/2}$ is nonincreasing
for $u\ge q_0$, since its logarithmic derivative is
$K/u-\ln2/2\le0$.
It follows that
\begin{align*}
    {\mathrm e} y
    &=4{\mathrm e} d\,{\mathrm e}^{(d+\ln2)/2}q^K{\mathrm e}^{-q\ln2/2}\\
    &\le4{\mathrm e} d\,((2+\eta)d)^K
          {\mathrm e}^{-(j_\eta d-\ln2)/2}\\
    &\le\frac14,
\end{align*}
where the last step uses the final inequality in
\eqref{eq:coloring-d-setting}.
This proves \eqref{eq:coloring-block-parameter-estimate}
with a threshold $d_0$ independent of $q$.
