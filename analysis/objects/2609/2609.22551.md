<!-- xdefinition-1: body_md -->
A \emph{finitely supported permutation} of $\mathbb{A}$ is a permutation
  $\pi:\mathbb{A}\to\mathbb{A}$ such that the set $\{a\in \mathbb{A}\mid \: \pi(a) \not\nobreak\mkern -2mu =
    a \}$ is finite. Let $\Pi_{\mathbb{A}}$ be the group of these finitely
  supported permutations. We drop the subscript when it is clear from
  the context. The group operation is the composition of permutations,
  which we denote by $\pi_1\circ\pi_2$ or simply $\pi_1\pi_2$. The
  identity permutation is denoted by $\text{\textrm{id}}$.
<!-- end xdefinition-1 -->
<!-- xdefinition-2: body_md -->
A $\Pi$-set is a pair $(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})$, where $X$ is a set and $\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}}$ is an action of $\Pi$ on $X$. We often just write $X$ for a $\Pi$-set when the action is clear from the context.
<!-- end xdefinition-2 -->
<!-- xdefinition-3: body_md -->
Let $X$ and $Y$ be $\Pi$-sets. A function $f:X\to Y$ is \emph{equivariant} if for all  $\pi\in \Pi$ and for all $x\in X, \pi\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} \left( f(x) \right)=f\left( \pi\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x \right)$.
<!-- end xdefinition-3 -->
<!-- xdefinition-4: body_md -->
Let $X$ be a $\Pi$-set. Then the action of $\Pi$ on $\mathcal{P}(X)$,
  the powerset of $X$, is defined by $\pi \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} S := \{ \pi a
    \mid a\in S \}$. This means that $\mathcal{P}(X)$ is a $\Pi$-set.
<!-- end xdefinition-4 -->
<!-- xdefinition-5: body_md -->
\label{def:function-pi-action}
Let $X$ and $Y$ be $\Pi$-sets and let $\pi \in \Pi$. The action of
$\pi$ on a function $f:X\to Y$ is defined by $(\pi \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} f)(x) = \pi
\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} \left(f \left( \pi^{-1} \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x\right) \right)$.
<!-- end xdefinition-5 -->
<!-- xdefinition-6: body_md -->
Let $X$ be a $\Pi$-set and let $x\in X$. We say that the set
  $S\subseteq \mathbb{A}$ is a \emph{support} for $x$ if for all $a,b\in
  \mathbb{A}- S$, $(a\,b) \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x=x$. Given a $\Pi$-set $X$,
  $x\in X$ is \emph{finitely supported} if there exists some finite
  support $S$ for it. One can show that for any finitely supported
  $x$, there exists a smallest finite support, and we denote it by
  $\mathop{\text{\relax supp}}(x)\in P(\mathbb{A})$ \cite{Pitts2013}.
<!-- end xdefinition-6 -->
<!-- xdefinition-7: body_md -->
A $\Pi$-set $X$ is called a \emph{nominal set} if all $x\in X$ are
  finitely supported.
<!-- end xdefinition-7 -->
<!-- xlem-1: body_md -->
\label{LemSuppAct}
  Let $X$ be a nominal set and let $x\in X$. Let $\pi' ,\pi'' \in \Pi$
  be two finitely supported permutations. If
  $\pi'|_{\mathop{\text{\relax supp}}(x)}=\pi''|_{\mathop{\text{\relax supp}}(x)}$ then $\pi' \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x = \pi''
  \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x$.
<!-- end xlem-1 -->
<!-- proof-1: body_md -->
Recall that if $\mathop{\text{\relax supp}}(\pi)$ is disjoint from $\mathop{\text{\relax supp}}(x)$, then
  $\pi\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x = x$.  Now assume
  $\pi'|_{\mathop{\text{\relax supp}}(x)}=\pi''|_{\mathop{\text{\relax supp}}(x)}$ and consider
  $\pi=\left( \pi''\right)^{-1} \pi'\in \Pi$. Then
  $\pi|_{\mathop{\text{\relax supp}}(x)}=\text{\textrm{id}}|_{\mathop{\text{\relax supp}}(x)}$, since for all $a\in
  \mathop{\text{\relax supp}}(x)$, we have $\pi'(a)=\pi''(a)$. So we have $\mathop{\text{\relax supp}}(\pi)\cap
  \mathop{\text{\relax supp}}(x)=\emptyset$, which implies $\pi\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x=x$, and hence
  $\pi'\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x=\pi''\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x$, as claimed.
<!-- end proof-1 -->

<!-- xdefinition-8: body_md -->
\label{def:PIA-iota}
  Let $\mathbb{A}$, $\mathbb{B}$ be atom sets and let $\iota:\mathbb{A}\to\mathbb{B}$ be an injective
  function. We define $\Pi_\iota:\Pi_{\mathbb{A}} \to \Pi_{\mathbb{B}}$ to be the unique
  group homomorphism such that $\Pi_\iota \left( (a\,a') \right)=(b\,b')$
  whenever $b=\iota(a)$ and $b'=\iota(a')$. Equivalently, for any permutation
  $\pi\in \Pi_{\mathbb{A}}$ and any atom $b$, we have
  \[
  \Pi_\iota(\pi) (b)=
  \begin{cases} \iota\left( \pi\left( \iota^{-1} (b) \right)\right) &
    \text{if } b\in \mathop{\text{\relax img}}(\iota)\\ b & \text{otherwise.}
  \end{cases}
  \relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                     
  \else \@badmath \fi
<!-- end xdefinition-8 -->
<!-- xdefinition-9: body_md -->
Given atom sets $\mathbb{A}$, $\mathbb{B}$ and an injective function
  $\iota:\mathbb{A}\to\mathbb{B}$, the functor $\overline{\iota}:\textrm{\bf Nom}_{\mathbb{B}}\to\textrm{\bf Nom}_{\mathbb{A}}$ is defined
  as follows.

\begin{itemize}
\item{} On objects: For $(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})\in\textrm{\bf Nom}_{\mathbb{B}}$, define
  $\overline{\iota}(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})=(X,\diamond)$ where $\pi\diamond x=
  \Pi_\iota(\pi)\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x$ for all $\pi\in \Pi_{\mathbb{A}}$.
\item{} On morphisms: Given an equivariant map $f:(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}}) \to
  (Y,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})$ in $\textrm{\bf Nom}_{\mathbb{B}}$, we define $\overline{\iota}(f):(X,\diamond)\to
  (Y,\diamond)$ in $\textrm{\bf Nom}_{\mathbb{A}}$ by $\overline{\iota}(f)(x)=f(x)$. In other
  words, $\overline{\iota}(f)$ and $f$ have the same underlying function
  $X\to Y$.
\end{itemize}

To show that this is a well-defined functor, we first need to show
that $(X,\diamond)$ is a nominal set. First, it is clear that
$\diamond$ is an action since $\Pi_{\iota}$ is a group homomorphism. We
must show that every $x\in X$ is finitely supported. We claim that
$S=\{a\in \mathbb{A} \mid \iota(a)\in \mathop{\text{\relax supp}}_{\mathbb{B}}(x)\}$ is a support of $x$.
Indeed, for $a',a''\not\in S$, we have $(a'a'')\diamond x =
(\iota(a'),\iota(a''))\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x =x$, since $\iota(a'),\iota(a'')\not\in \mathop{\text{\relax supp}}_\mathbb{B}(x)$.
Therefore, $x$ is finitely supported, and $(X,\diamond)$ is a
well-defined nominal set.  Note that we have proved that
$\mathop{\text{\relax supp}}_{\mathbb{A}}(x)\subseteq S$, but it is not hard to show that actually
$\mathop{\text{\relax supp}}_{\mathbb{A}}(x)=S$.

Next, we need to show that $\overline{\iota}$ is well-defined on
morphisms. So given $f:(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}}) \to (Y,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})$ in $\textrm{\bf Nom}_{\mathbb{B}}$, we
must show that $\overline{\iota}(f)$ is equivariant. Therefore, consider
any $\pi \in \Pi_{\mathbb{A}}$. We must show that
$\overline{\iota}(f)(\pi\diamond x) = \pi \diamond \overline{\iota}(f)(x)$.
Indeed, we have:
\[
\pi \diamond \overline{\iota}(f)(x) = \pi \diamond f(x)
 =  \Pi_\iota(\pi) \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} f(x)
 =  f(\Pi_\iota(\pi) \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x)
 =  f(\pi \diamond x)
 =  \overline{\iota} (f)(\pi \diamond x)
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                     
  \else \@badmath \fi
So, $\overline{\iota}(f)$ is equivariant. The functoriality of
$\overline{\iota}$ is then trivial, since the composition in both
categories is just composition of the underlying functions, which are
the same.

<!-- xdefinition-10: body_md -->
\label{def:muhat}
  For a bijection $\mu:\mathbb{A}\to\mathbb{B}$, the functor $\widehat{\mu}:\textrm{\bf Nom}_{\mathbb{A}}\to\textrm{\bf Nom}_{\mathbb{B}}$
  is given as follows:
  \begin{itemize}
  \item On objects: For every $(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})\in\textrm{\bf Nom}_{\mathbb{A}}$, we have
    $\widehat{\mu}(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})=(X,\diamond)$, where
    \[
    \pi\diamond x = (\mu^{-1} \pi \mu) \mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}} x.
    \relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                     
  \else \@badmath \fi
    
  \item On morphisms: For every equivariant function $f:(X,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})\to
    (Y,\mathbin{\text{\raisebox{0.15ex}{\scalebox{0.8}{$\bullet$}}}})$, the morphism $\widehat{\mu}(f):(X,\diamond)\to (Y,\diamond)$
    is defined by $\widehat{\mu}(f)=f$. 
  \end{itemize}

  \relax\ifmmode \egroup $\hfil \egroup
