<!-- proposition-1: body_md -->
\label{prop:intersection}
    $\mathcal{C}^{\underline{r}}$ is an embedded submanifold of $X_N$. For all $u \in \mathcal{C}^{\underline{r}}$,
    \[
        \mathrm{T}_u (\mathcal{M}^{\underline{r}} \cap \mathcal{S}) =
        \mathrm{T}_u \mathcal{M}^{\underline{r}} \cap \mathrm{T}_u \mathcal{S}.
    \]
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
The manifolds $\mathcal{M}^{\underline{r}}$ and $\mathcal{S}$ intersect transversally.
    Let $u\in\mathcal{M}^{\underline{r}} \cap \mathcal{S}$.
    Then $\mathrm{dim}(\mathrm{T}_u\mathcal{S}) = N-1$, with
    \[
        \mathrm{T}_u \mathcal{S} + \{\alpha u, \alpha\in\mathbb{R}\} = X_N.
    \]
    For any $\alpha \in\mathbb{R}$, $\alpha \mathbb{A}^{\Phi}_N u \in \mathrm{T}_{\mathbb{A}^{\Phi}_N u} \mathfrak{M}^{\underline{r}}$ (choose $\delta G_k = 0$ for $k=1,\dots,d-1$ and $\delta G_d = \alpha G_d$).
    Therefore $\alpha u\in \mathrm{T}_u \mathcal{M}^{\underline{r}}$ and we obtain that
    \[
        \mathrm{T}_u \mathcal{S} + \mathrm{T}_u \mathcal{M}^{\underline{r}} = X_N,
    \]
    which implies transversality and hence concludes the proof,
    by~\cite[Corollary 3.5.13]{Abraham1988}.
<!-- end proof-1 -->
<!-- lemma-1: body_md -->
\label{lem:proj-commute-single}
    For all $u \in \mathcal{C}^{\underline{r}}$,
    \[
        \mathrm{Proj}^{\mathcal{C}^{\underline{r}},L^2}_u = \mathrm{Proj}^{\mathcal{S},L^2}_u \mathrm{Proj}^{\mathcal{M}^{\underline{r}},L^2}_u = \mathrm{Proj}^{\mathcal{M}^{\underline{r}},L^2}_u \mathrm{Proj}^{\mathcal{S},L^2}_u,
    \]
    where $\mathrm{Proj}^{\mathcal{M}^{\underline{r}},L^2}_u$ and $\mathrm{Proj}^{\mathcal{S},L^2}_u$ denote the $L^2$-orthogonal projections onto
    $\mathrm{T}_u \mathcal{M}^{\underline{r}}$ and $\mathrm{T}_u \mathcal{S}$, respectively.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Since $\left\|{u}\right\|_{L^2(\Omega)} = 1$, the $L^2$-orthogonal projection onto
    $\mathrm{T}_u\mathcal{S} = \{v \in X_N : (v,u)_{L^2(\Omega)} = 0\}$ is $\mathrm{Proj}^{\mathcal{S},L^2}_u v = v - (u,v)_{L^2(\Omega)}\,u$.
    By the same arguments as in the proof of \Cref{prop:intersection}, $u \in \mathrm{T}_u\mathcal{M}^{\underline{r}}$, so that
    $\mathrm{Proj}^{\mathcal{M}^{\underline{r}},L^2}_u u = u$ and, by self-adjointness of $\mathrm{Proj}^{\mathcal{M}^{\underline{r}},L^2}_u$,

<!-- lemma-3: body_md -->
\label{lem:H-formula}
    Let $\mathcal{V}$ be a finite-dimensional real inner product space, let $\mathrm{T}
        \subseteq \mathcal{V}$ be a subspace, and let $\mathsf{P}\colon \mathcal{V} \to \mathcal{V}$ be
    the orthogonal projector onto $\mathrm{T}$, i.e.,
    \[
        \mathrm{range}(\mathsf{P}) = \mathrm{T}, \qquad \mathsf{P}^2 = \mathsf{P} = \mathsf{P}^*,
    \]
    where $\mathsf{P}^*$ is the adjoint of $\mathsf{P}$ with respect to the inner product of $\mathcal{V}$.
    Let $\mathsf{A} \colon \mathcal{V} \to \mathcal{V}$ be a self-adjoint positive definite linear
    operator and let $\mathsf{B} \colon \mathcal{V} \to \mathcal{V}$ be an arbitrary linear
    operator. The unique linear operator $\mathsf{H} \colon \mathcal{V} \to \mathcal{V}$
    satisfying
    \begin{enumerate}[(i)]
        \item $\mathrm{range}(\mathsf{H}) \subseteq \mathrm{T}$,
        \item $\ifthenelse{\equal{}{}}{\left\langle{\mathsf{A} \mathsf{H} \mathbf{z}},\,{\mathbf{w}}\right\rangle}{\left\langle{\mathsf{A} \mathsf{H} \mathbf{z}},\,{\mathbf{w}}\right\rangle_{}} = \ifthenelse{\equal{}{}}{\left\langle{\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle}{\left\langle{\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle_{}}$ for all $\mathbf{w}
                  \in \mathrm{T}$, $\mathbf{z} \in \mathcal{V}$,
    \end{enumerate}
    is
    \begin{equation}
        \label{eq:H-matrix-formula}
        \mathsf{H} =  \mathsf{P} (\mathsf{P}\mathsf{A}\mathsf{P})^\dagger \mathsf{P}\mathsf{B} = (\mathsf{P}\mathsf{A}\mathsf{P})^\dagger \mathsf{P}\mathsf{B},
    \end{equation}
    where $(\cdot)^\dagger$ denotes the Moore--Penrose pseudo-inverse.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
The existence and uniqueness of $\mathsf{H}$ follow from the Riesz representation
    theorem since $\mathsf{A}$ is positive definite. Indeed, for fixed $\mathbf{z}$, $\mathsf{H}\mathbf{z}
        \in \mathrm{T}$ is the unique vector representing the functional $\mathbf{w} \mapsto
        \ifthenelse{\equal{}{}}{\left\langle{\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle}{\left\langle{\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle_{}}$ on $\mathrm{T}$ in the $\mathsf{A}$-inner product.

    For the explicit formula, we check that $\mathsf{H} = \mathsf{P}(\mathsf{P}\mathsf{A}\mathsf{P})^\dagger
        \mathsf{P}\mathsf{B}$ satisfies (i) and (ii). Note first that $\mathrm{range}(\mathsf{H})
        \subseteq \mathrm{range}(\mathsf{P}) = \mathrm{T}$. For (ii), if $\mathbf{w} \in \mathrm{T}$, then $\mathbf{w}
        = \mathsf{P}\mathbf{w}$, and, using that $\mathsf{P}$ is self-adjoint,
    \begin{align*}
        \ifthenelse{\equal{}{}}{\left\langle{\mathsf{A} \mathsf{H} \mathbf{z}},\,{\mathbf{w}}\right\rangle}{\left\langle{\mathsf{A} \mathsf{H} \mathbf{z}},\,{\mathbf{w}}\right\rangle_{}} - \ifthenelse{\equal{}{}}{\left\langle{\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle}{\left\langle{\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle_{}}
         & = \ifthenelse{\equal{}{}}{\left\langle{(\mathsf{A} \mathsf{H} - \mathsf{B}) \mathbf{z}},\,{\mathsf{P}\mathbf{w}}\right\rangle}{\left\langle{(\mathsf{A} \mathsf{H} - \mathsf{B}) \mathbf{z}},\,{\mathsf{P}\mathbf{w}}\right\rangle_{}}
        = \ifthenelse{\equal{}{}}{\left\langle{\bigl(\mathsf{P}\mathsf{A}\mathsf{P} (\mathsf{P}\mathsf{A}\mathsf{P})^\dagger \mathsf{P}\mathsf{B} - \mathsf{P}\mathsf{B}\bigr) \mathbf{z}},\,{\mathbf{w}}\right\rangle}{\left\langle{\bigl(\mathsf{P}\mathsf{A}\mathsf{P} (\mathsf{P}\mathsf{A}\mathsf{P})^\dagger \mathsf{P}\mathsf{B} - \mathsf{P}\mathsf{B}\bigr) \mathbf{z}},\,{\mathbf{w}}\right\rangle_{}}    \\
         & = \ifthenelse{\equal{}{}}{\left\langle{\bigl(\mathsf{P}\mathsf{A}\mathsf{P} (\mathsf{P}\mathsf{A}\mathsf{P})^\dagger - \operatorname{Id}\bigr) \mathsf{P}\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle}{\left\langle{\bigl(\mathsf{P}\mathsf{A}\mathsf{P} (\mathsf{P}\mathsf{A}\mathsf{P})^\dagger - \operatorname{Id}\bigr) \mathsf{P}\mathsf{B} \mathbf{z}},\,{\mathbf{w}}\right\rangle_{}} = 0,
    \end{align*}
    since $\mathsf{P}\mathsf{A}\mathsf{P} (\mathsf{P}\mathsf{A}\mathsf{P})^\dagger$ is the orthogonal projection
    onto $\mathrm{range}(\mathsf{P}\mathsf{A}\mathsf{P}) = \mathrm{range}(\mathsf{P}) = \mathrm{T}$ and $\mathsf{P}\mathsf{B}\mathbf{z} \in
        \mathrm{T}$.

<!-- remark-1: body_md -->
\label{rmk:highorder_quadrature}
    Employing the quadrature formula that uses the GLL points introduces an
    additional source of error, justified by the computational savings discussed
    above. At the end of the optimization process, however, the energy can be
    evaluated with a higher-order quadrature at moderate additional cost.
    For example, exact integration of the
    nonlinearity requires increasing the quadrature order from $2k-1$ to $4k$. This
    is achieved by introducing the extension operator mapping $\mathbf{U}$, the tensor of
    coefficients in the rescaled basis \eqref{eq:rescaled-basis}, to $\mathbf{U}_Q$, the
    tensor of values at quadrature nodes (for order $4k$). By
    \eqref{eq:coeff-scaling}, this operator first undoes the rescaling, mapping
    $\mathbf{U}$ to the tensor of nodal values by a Hadamard product with
    $\hat{\mathbf{M}}_{\mathrm{rec}}^{1/2} = \otimes_{i=1}^d [m_1^{-1/2}, \cdots, m_n^{-1/2}]^\top$, and
    then evaluates at the quadrature nodes. The energy is then computed using
    \eqref{eq:E_discrete} with $\mathbf{U}_Q$ and the updated quadrature points and
    weights, the latter now multiplying the nonlinear terms rather than appearing as
    reciprocals. Since both factors of the extension operator are tensor products of
    their one-dimensional counterparts, it is a rank-$1$ TT matrix, so $\mathbf{U}_Q$ has
    the same TT rank as $\mathbf{U}$ and the extension costs $\mathcal{O}(dn^2r^2)$, still polynomial in $n$ but quadratic instead of linear.

<!-- lemma-4: body_md -->
\label{lem:multi-G-invertible}

<!-- end lemma-4 -->
<!-- proof-8: body_md -->
Let $\boldsymbol{\alpha} \in \mathbb{R}^p$ be such that $\mathbf{G}\boldsymbol{\alpha} = 0$. Then, for all $i = 1,
        \dots, p$,
    \[
        0 = \sum_{j=1}^p \alpha_j \bigl( (\mathrm{H}_\varphi \widehat{\varphi}_j)_i, \varphi_i \bigr)_{L^2(\Omega)}
        = \bigl( (\mathrm{H}_\varphi(\varphi \mathrm{diag} \boldsymbol{\alpha}))_i, \varphi_i \bigr)_{L^2(\Omega)},
    \]
    that is, $\langle\!\langle \mathrm{H}_\varphi(\varphi\mathrm{diag}\boldsymbol{\alpha}), \varphi \rangle\!\rangle = \mathbf{0}$, so that
    $\mathrm{H}_\varphi(\varphi\mathrm{diag}\boldsymbol{\alpha}) \in \mathrm{T}_\varphi\mathcal{OB}_{\mathbf{N}}(p, X_N)$. On the other hand, for every
    $w \in \mathrm{T}_\varphi\mathcal{M}^{\underline{r}} \cap \mathcal{OB}_{\mathbf{N}}(p, X_N)$,
    \[
        a_\varphi\bigl(\mathrm{H}_\varphi(\varphi\mathrm{diag}\boldsymbol{\alpha}), w\bigr)
        = (\varphi\mathrm{diag}\boldsymbol{\alpha}, w)_{\mathbf{L}}
        = \sum_{i=1}^p \alpha_i (\varphi_i, w_i)_{L^2(\Omega)} = 0 .

    taking $w = \mathrm{H}_\varphi(\varphi\mathrm{diag}\boldsymbol{\alpha})$ and using the coercivity of $a_\varphi$
    gives $\mathrm{H}_\varphi(\varphi\mathrm{diag}\boldsymbol{\alpha}) = 0$. Finally, $\varphi\mathrm{diag}\boldsymbol{\alpha} \in
        \mathrm{T}_\varphi\mathcal{M}^{\underline{r}}$ by the proof of \Cref{prop:multi-intersection}, hence
    \[
        0 = a_\varphi\bigl(\mathrm{H}_\varphi(\varphi\mathrm{diag}\boldsymbol{\alpha}), \varphi\mathrm{diag}\boldsymbol{\alpha}\bigr)
        = (\varphi\mathrm{diag}\boldsymbol{\alpha}, \varphi\mathrm{diag}\boldsymbol{\alpha})_{\mathbf{L}}

        = \sum_{i=1}^p \alpha_i^2 N_i,
    \]
    which forces $\boldsymbol{\alpha} = 0$.

\label{lem:proj-commute}
    For all $\boldsymbol{\Phi} \in \mathfrak{C}^{\underline{r}}$,
    $\mathrm{Proj}^{\mathfrak{C}^{\underline{r}}}_{\boldsymbol{\Phi}} = \mathrm{Proj}^{\mathfrak{OB}}_{\boldsymbol{\Phi}}\mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}
        = \mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}\mathrm{Proj}^{\mathfrak{OB}}_{\boldsymbol{\Phi}}$.
<!-- end lemma-5 -->
<!-- proof-9: body_md -->
By \eqref{eq:proj-L2-disc} and \eqref{eq:frame}, $\mathrm{Proj}^{\mathfrak{OB}}_{\boldsymbol{\Phi}}$ subtracts a
    linear combination of the frames $\widehat{\varphi}_1, \dots, \widehat{\varphi}_p$ of $\boldsymbol{\Phi}$, which
    are Frobenius-orthogonal with $\ifthenelse{\equal{}{}}{\langle{\widehat{\varphi}_i},\,{\widehat{\varphi}_j}\rangle}{\langle{\widehat{\varphi}_i},\,{\widehat{\varphi}_j}\rangle_{}} = N_i
        \delta_{ij}$. Hence $\mathrm{Proj}^{\mathfrak{OB}}_{\boldsymbol{\Phi}}\mathbf{V} = \mathbf{V} - \sum_j
        N_j^{-1}\ifthenelse{\equal{}{}}{\langle{\widehat{\varphi}_j},\,{\mathbf{V}}\rangle}{\langle{\widehat{\varphi}_j},\,{\mathbf{V}}\rangle_{}}\,\widehat{\varphi}_j$. By the proof of
    \Cref{prop:multi-intersection}, $\widehat{\varphi}_j \in \mathrm{T}_{\boldsymbol{\Phi}}\mathfrak{M}^{\underline{r}}$, so
    $\mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}\widehat{\varphi}_j = \widehat{\varphi}_j$ and
    $\ifthenelse{\equal{}{}}{\langle{\widehat{\varphi}_j},\,{\mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}\mathbf{V}}\rangle}{\langle{\widehat{\varphi}_j},\,{\mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}\mathbf{V}}\rangle_{}} =
        \ifthenelse{\equal{}{}}{\langle{\mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}\widehat{\varphi}_j},\,{\mathbf{V}}\rangle}{\langle{\mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}\widehat{\varphi}_j},\,{\mathbf{V}}\rangle_{}} = \ifthenelse{\equal{}{}}{\langle{\widehat{\varphi}_j},\,{\mathbf{V}}\rangle}{\langle{\widehat{\varphi}_j},\,{\mathbf{V}}\rangle_{}}$.
    Both compositions therefore equal $\mathbf{V} \mapsto \mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}\mathbf{V} - \sum_j
        N_j^{-1}\ifthenelse{\equal{}{}}{\langle{\widehat{\varphi}_j},\,{\mathbf{V}}\rangle}{\langle{\widehat{\varphi}_j},\,{\mathbf{V}}\rangle_{}}\widehat{\varphi}_j$, so
    $\mathrm{Proj}^{\mathfrak{M}^{\underline{r}}}_{\boldsymbol{\Phi}}$ and $\mathrm{Proj}^{\mathfrak{OB}}_{\boldsymbol{\Phi}}$ commute. Two commuting orthogonal
    projections compose to the orthogonal projection onto the intersection of
    their ranges, which by \Cref{prop:multi-intersection} is
    $\mathrm{T}_{\boldsymbol{\Phi}}\mathfrak{M}^{\underline{r}} \cap \mathrm{T}_{\boldsymbol{\Phi}}\mathfrak{OB}_{\mathbf{N}}(p) = \mathrm{T}_{\boldsymbol{\Phi}}\mathfrak{C}^{\underline{r}}$.

<!-- remark-4: body_md -->
\label{rem:L2-soundness}
    The two constructions differ in their robustness with respect to the
    discretization. Route~I is the discretization of a well-posed continuous
    formula: \eqref{eq:multi-grad-Hu} makes sense on $\mathcal{C}^{\underline{r}}$ for every $\varphi$,
    because $a_\varphi$ is coercive on $\mathbf{H}$. Route~II is not: what
    \eqref{eq:change-of-metric-disc} computes is the \emph{discrete} $L^2$
    Riemannian gradient, which exists for every $\boldsymbol{\Phi} \in \mathfrak{C}^{\underline{r}}$ but whose
    continuous counterpart \eqref{eq:change-of-metric-L2} requires the additional
    assumption $\mathrm{A}_\varphi\varphi \in \mathbf{L}$. One should therefore not expect
    route~II to be robust under mesh refinement in the sense of
    \cite{altmann_riemannian_2025}. We nonetheless use it in all experiments,
    since it is roughly $p+1$ times cheaper.

<!-- lemma-6: body_md -->
\label{lem:last-core}
    Let $\boldsymbol{\Phi} \in \mathfrak{M}^{\underline{r}}$ with cores $G_1, \dots, G_{d+1}$, and assume
    $\boldsymbol{\Phi}$ left-orthogonal in the sense of \eqref{eq:left-orth}, i.e., that
    $(\mathsf{G}_k^{\mathsf{L}})^\top\mathsf{G}_k^{\mathsf{L}} = \mathsf{I}_{r_k}$ for $k = 1,
        \dots, d$.
    Let $\mathbf{W} \in \mathrm{T}_{\boldsymbol{\Phi}}\mathfrak{M}^{\underline{r}}$ be parametrized by the $\delta$-cores
    $(\delta W_1, \dots, \delta W_{d+1})$. Then
    \begin{equation}
        \label{eq:last-core}
        \langle\!\langle \mathbf{W}, \boldsymbol{\Phi} \rangle\!\rangle_{jj} = \delta W_{d+1}(j)^\top G_{d+1}(j),
        \qquad j = 1, \dots, p .
    \end{equation}
<!-- end lemma-6 -->
<!-- proof-10: body_md -->
By the proof of \Cref{prop:multi-intersection}, the gauged representation of
    $\boldsymbol{\Phi}$ as an element of $\mathrm{T}_{\boldsymbol{\Phi}}\mathfrak{M}^{\underline{r}}$ is given by the
    $\delta$-cores $(\delta G_1, \dots, \delta G_{d+1}) = (0, \dots, 0,
        G_{d+1})$. We write, as in \eqref{eq:TT-interface},
    \[
        \mathbf{W}(i_1, \dots, i_d, j) = \sum_{k=1}^{d+1}\boldsymbol{\Phi}_{\leq k-1}(i_1, \dots, i_{k-1})\,
        \delta W_k(i_k)\, \boldsymbol{\Phi}_{\geq k+1}(i_{k+1}, \dots, i_d, j)^\top,
    \]
    where the interface matrices have been introduced in \eqref{eq:interface} and $i_{d+1} = j$. Contracting over $i_1, \dots, i_d$, the term $k = d+1$ gives
    \[
        \sum_{i_1, \dots, i_d} \delta W_{d+1}(j)^\top \boldsymbol{\Phi}_{\leq d}(i_1, \dots, i_d)^\top \boldsymbol{\Phi}_{\leq d}(i_1, \dots, i_d)\, G_{d+1}(j)
        = \delta W_{d+1}(j)^\top G_{d+1}(j),
    \]
    since $\boldsymbol{\Phi}_{\leq d}^\top \boldsymbol{\Phi}_{\leq d} = \mathsf{I}_{r_d}$ by
    left-orthogonality. For $k \leq d$, the factors $\boldsymbol{\Phi}_{\geq k+1}$ do not depend on
    $i_k$, so the sum over $i_k$ can be carried out first and produces $\sum_{i_k}
        \delta W_k(i_k)^\top G_k(i_k) = [\delta W_k, G_k] = 0$ by the gauge
    conditions. Hence all terms with $k \leq d$ vanish.

<!-- remark-5: body_md -->
\label{rem:rank-component}
    The rank $r_d$, which separates the spatial modes from the component mode, is
    bounded by the number of components: since $r_{d+1} = 1$ and the last mode has
    size $p$,
    \[
        r_d \leq r_{d+1}\, p = p .
    \]
    Equivalently, the $d$-th unfolding of $\boldsymbol{\Phi}$ has only $p$ columns. In our
    numerical experiments we always take $r_d = p$, its maximal value.
