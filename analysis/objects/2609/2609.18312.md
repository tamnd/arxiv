<!-- theorem-1: body_md -->
\label{thm:thm1} 
Consider large dimensional proportional regime with $n,m,k_1,k_2\in{\mathbb N}$ as in (\ref{eq:amat0a0}). Let each element of ${\bf h}\in{\mathbb R}^m$ and ${\bf g}\in{\mathbb R}^n$ be standard normal random variable independent of any other randomness. Set 
\begin{eqnarray} 
   \label{eq:thm1eq1} 
 L & = &   
\frac{1}{\sqrt{k_1 k_2}} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}  \left ( {\bf h}^T{\bf y} +{\bf g}^T{\bf x} \right )  . 
\end{eqnarray} 
For $A\in{\mathbb R}^{m\times n}$  comprised of independent standard normal entries, let $\xi$ be as in (\ref{eq:algb1}).  One then has   
\begin{eqnarray} 
   \label{eq:thm1eq2} 
\lim_{n\rightarrow \infty}    {\mathbb E} \xi\sqrt{n}   \leq  \lim_{n\rightarrow \infty}   {\mathbb E} L \sqrt{n}, 
\end{eqnarray} 
with the righthand side being the random dual.

<!-- proof-1: body_md -->
For a standard normal $g$ (independent of all other randomness), we consider two centered Gaussian processes indexed by an array ${\mathcal X} = \{{\bf x},{\bf y}\}$ 
  \begin{eqnarray} 
\label{eq:mr1} 
 {\mathcal G} ({\mathcal X}) & \triangleq &  {\mathcal G} ({\bf x},{\bf y})  \triangleq   {\bf y}^TA{\bf x} +  g  \nonumber   \\ 
 {\mathcal G}_u ({\mathcal X}) & \triangleq &  {\mathcal G}_u ({\bf x},{\bf y})  \triangleq    {\bf h}^T{\bf y} +{\bf g}^T{\bf x}. 
  \end{eqnarray} 
 For two arrays ${\mathcal X}^{(1)}=\{ {\bf x}^{(1)},{\bf y}^{(1)}\}$ and ${\mathcal X}^{(2)}=\{ {\bf x}^{(2)},{\bf y}^{(2)}\}$ with $ {\bf x}^{(i)} \in {\mathcal S}_{{\bf x}}$ and $ {\bf y}^{(i)} \in {\mathcal S}_{{\bf y}}$, $i=1,2$,  we further write 
  \begin{eqnarray} 
\label{eq:mr2} 
{\mathbb E} {\mathcal G} ({\mathcal X}^{(1)}){\mathcal G} ({\mathcal X}^{(2)})   & =  &   \left ({\bf y}^{(1)}\right )^T{\bf y}^{(2)} \left ({\bf x}^{(2)}\right )^T{\bf x}^{(1)} + 1 
\nonumber   \\ 
{\mathbb E} {\mathcal G}_u ({\mathcal X}^{(1)}){\mathcal G}_u ({\mathcal X}^{(2)})  & =  & \left ({\bf y}^{(2)}\right )^T{\bf y}^{(1)} + \left ({\bf x}^{(2)}\right )^T{\bf x}^{(1)} . 
  \end{eqnarray} 
From (\ref{eq:mr2}) one then finds 
  \begin{align} 
\label{eq:mr5} 
  {\mathbb E} & {\mathcal G} ({\mathcal X}^{(1)}){\mathcal G} ({\mathcal X}^{(2)}) 
 - 
{\mathbb E} {\mathcal G}_u ({\mathcal X}^{(1)}){\mathcal G}_u ({\mathcal X}^{(2)} ) = 
\nonumber 
\\ 
 & =        
 \left ({\bf y}^{(1)}\right )^T{\bf y}^{(2)} \left ({\bf x}^{(2)}\right )^T{\bf x}^{(1)} + 1 
 -\left ({\bf y}^{(2)}\right )^T{\bf y}^{(1)} - \left ({\bf x}^{(2)}\right )^T{\bf x}^{(1)} 
 \nonumber \\ 
  & =        
\left ( 1 -  \left ({\bf y}^{(1)}\right )^T{\bf y}^{(2)}  \right ) \left ( 1 - \left ({\bf x}^{(2)}\right )^T{\bf x}^{(1)}\right ) 
 \nonumber \\ 
&  \geq   0.   
 \end{align} 
One also notes  
  \begin{align} 
\label{eq:mr5a0} 
  {\mathbb E} & {\mathcal G} ({\mathcal X}^{(1)}){\mathcal G} ({\mathcal X}^{(1)}) 
 - 
{\mathbb E} {\mathcal G}_u ({\mathcal X}^{(1)}){\mathcal G}_u ({\mathcal X}^{(1)} ) 
   =  
\left ( 1 -  \left ({\bf y}^{(1)}\right )^T{\bf y}^{(1)}  \right ) \left ( 1 - \left ({\bf x}^{(1)}\right )^T{\bf x}^{(1)}\right ) 
  =   0, 
  \end{align} 
  where the last equality follows since  $\|{\bf x}^{(1)}\|_2 = \left ( {\bf x}^{(1)}\right )^T {\bf x}^{(1)}=\left ( {\bf y}^{(1)}\right )^T {\bf y}^{(1)}=\|{\bf x}^{(1)}\|_2 = 1$. 
   
The proof is completed after one recalls Theorem 1.1 from \cite{Gordon85} (the part of the theorem that is utilized below is known as Slepian lemma and was introduced in \cite{Slep62}; both Slepian lemma and the theorem below can be deduced as special cases of results obtained in Corollary 3 in \cite{Stojnicgscompyx16}  and in Corollary 4 in \cite{Stojnicgscomp16}). 
 
\begin{theorem}(\cite{Gordon85,Slep62}) 
\label{thm:Gordonpos1} Let $X_{i}$ and $Y_{i}$, $1\leq i\leq n$, be two centered Gaussian processes which satisfy the following inequalities for all choices of indices 
\begin{enumerate} 
\item ${\mathbb E}(X_{i}^2)={\mathbb E}(Y_{i}^2)$ 
\item ${\mathbb E}(X_{i}X_{l})\leq {\mathbb E}(Y_{i}Y_{l}), i\neq l$. 
\end{enumerate} 
 Then 
\begin{equation*} 
{\mathbb E}(\min_{i} X_{i})\leq {\mathbb E}(\min_i Y_{i}) \quad  \Longleftrightarrow \quad {\mathbb E}(\max_{i} X_{i})\geq {\mathbb E}(\max_i Y_{i}). 
\end{equation*} 
\end{theorem} 
 
After noting correspondence $Y\leftrightarrow{\mathcal G}$ and $X\leftrightarrow{\mathcal G}_u$ and 
applying  Theorem \ref{thm:Gordonpos1} to processes ${\mathcal G}(\cdot)$ and  ${\mathcal G}_u(\cdot)$, we obtain    
\begin{align}\label{eq:mt5a1a0} 
&  &{\mathbb E} \max_{{\mathcal X}^{(1)}} {\mathcal G}({\mathcal X})  & \leq {\mathbb E} \max_{{\mathcal X}^{(1)}} {\mathcal G}_u({\mathcal X}) 
\nonumber \\ 
\Longleftrightarrow & & {\mathbb E} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}  
 \left ( {\bf y}^TA{\bf x} +  {\bf g} \right ) 
 & \leq {\mathbb E}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y} + {\bf g}^T{\bf x}  \right ) 
\nonumber \\ 
\Longleftrightarrow & & {\mathbb E} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}  
  {\bf y}^TA{\bf x}  
 & \leq {\mathbb E}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y} + {\bf g}^T{\bf x}  \right ) 
 . 
 \end{align} 
Connecting  (\ref{eq:algb1}) and (\ref{eq:mt5a1a0}), one then has 
\begin{eqnarray}\label{eq:mt5a1a1} 
 \lim_{n\rightarrow \infty}   \sqrt{n}  {\mathbb E}  \xi  & \leq &   \lim_{n\rightarrow \infty} \frac{\sqrt{n}}{\sqrt{k_1k_2}} {\mathbb E}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y} + {\bf g}^T{\bf x}  \right ) 
 , 
 \end{eqnarray} 
which, together with (\ref{eq:thm1eq1}), gives  (\ref{eq:thm1eq2}) and completes the proof.

<!-- proof-5: body_md -->
We start by recalling Corollary 1.3 from \cite{Gordon85}.    
 
\begin{theorem}(\cite{Gordon85}) 
\label{thm:Gordoncor1} Assume the setup of Theorem \ref{thm:Gordonpos1}. Let $\psi_q(\cdot)$ be an increasing function on the real axis. Then 
\begin{equation}\label{eq:thm5eq1} 
E(\min_{i}\psi_q(X_{i}))\leq E(\min_i \psi_q(Y_{i})) \quad \Longleftrightarrow \quad  E(\max_{i}\psi_q(X_{i}))\geq E(\max_i\psi_q(Y_{i})). 
\end{equation} 
\end{theorem} 
 
For $\psi_q(x)=x$, one has that Theorem \ref{thm:Gordoncor1} matches Theorem \ref{thm:Gordonpos1}. The key question is whether a better than linear choice can be found for $\psi_q(\cdot)$. We discover that $\psi_q(x)=e^{c_3 x}$, with $c_3>0$, is an excellent choice. Along the same lines, we also note that Theorems \ref{thm:Gordonpos1} and  \ref{thm:Gordoncor1} are special cases of concepts presented in Corollary 3 in \cite{Stojnicgscompyx16}  and in Corollary 4 in \cite{Stojnicgscomp16}. 
 
Combining (\ref{eq:mr5}), (\ref{eq:mr5a0}), and (\ref{eq:thm5eq1})  with correspondence $Y\leftrightarrow{\mathcal G}$ and $X\leftrightarrow{\mathcal G}_u$ and the above mentioned choice $\psi_q(x)=e^{c_3 x},c_3>0$, gives the following 
\begin{eqnarray} 
\label{eq:mr6} 
  {\mathbb E} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}} e^{c_{3,u}{\mathcal G}({\bf x},{\bf y})} &\leq &    {\mathbb E} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}} e^{c_{3,u}{\mathcal G}_u({\bf x},{\bf y})} . 
\end{eqnarray} 
From (\ref{eq:mr1}) and (\ref{eq:mr6}) we then find 
\begin{eqnarray} 
\label{eq:mr7} 
  {\mathbb E}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}} e^{c_{3}\left (  {\bf y}^TA{\bf x} +  g \right ) } &\leq &   {\mathbb E} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}  e^{c_{3} \left (  {\bf h}^T{\bf y} +{\bf g}^T{\bf x} \right )  }, 
  \end{eqnarray} 
which is equivalent to 
\begin{eqnarray} 
\label{eq:mr8} 
  {\mathbb E}  e^{c_{3} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf y}^TA{\bf x} \right ) }  
  \leq   
   e^{-\frac{c_{3}^2}{2}} {\mathbb E} e^{c_{3}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y} +{\bf g}^T{\bf x} \right )    } , 
\end{eqnarray} 
and 
\begin{eqnarray} 
\label{eq:mr9} 
     \log \left ( {\mathbb E}  e^{c_{3} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf y}^TA{\bf x} \right ) }  
   \right ) 
      \leq  
   -\frac{c_{3}^2}{2} + \log \left ( 
    {\mathbb E} e^{c_{3}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y} +{\bf g}^T{\bf x} \right )    } 
        \right ). 
\end{eqnarray} 
Applying Jensen's inequality to the left-hand side and dividing both sides by $c_3$, we obtain 
\begin{eqnarray} 
\label{eq:mr10} 
{\mathbb E} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}    {\bf y}^TA{\bf x}    
     & \leq & 
   -\frac{c_{3}}{2} + \frac{1}{c_3}\log \left ( 
    {\mathbb E} e^{c_{3}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y} +{\bf g}^T{\bf x} \right )    } 
        \right ) 
\nonumber \\ 
 & \leq & 
  -\frac{c_{3}}{2} + \frac{1}{c_3}\log \left ( 
    {\mathbb E} e^{c_{3}  \max_{{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y}  \right )    } 
        \right ) 
        + 
  \frac{1}{c_3}\log \left ( 
    {\mathbb E} e^{c_{3}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}}}   \left (  {\bf g}^T{\bf x} \right )    } 
        \right ). 
\end{eqnarray} 
Since the above holds for any $c_3>0$, it also holds for the ``worst-case'' one, i.e., one has 
\begin{eqnarray} 
\label{eq:mr10} 
{\mathbb E} \max_{{\bf x}\in{\mathcal S}_{{\bf x}},{\bf y}\in{\mathcal S}_{{\bf y}}}    {\bf y}^TA{\bf x}    
   \leq  
   \min_{c_3>0}  
   \left ( 
  -\frac{c_{3}}{2} + \frac{1}{c_3}\log \left ( 
    {\mathbb E} e^{c_{3}  \max_{{\bf y}\in{\mathcal S}_{{\bf y}}}   \left (  {\bf h}^T{\bf y}  \right )    } 
        \right ) 
        +

    {\mathbb E} e^{c_{3}  \max_{{\bf x}\in{\mathcal S}_{{\bf x}}}   \left (  {\bf g}^T{\bf x} \right )    } 
        \right ) 
        \right ) , 
\end{eqnarray} 
which matches the first inequality in (\ref{eq:thm4eq1}). After observing that the second inequality in (\ref{eq:thm4eq1}) follows automatically for $c_3\rightarrow 0$, the theorem's proof is completed.
