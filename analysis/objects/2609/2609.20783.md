<!-- theorem-1: body_md -->
\label{thm:main}
Let $r\geq 2$, $d\geq 2$, and $e\geq 1$. Suppose $K$ has characteristic zero or greater than $d$. Let $X\subset \mathbb P_K^{n-1}$ be a smooth hypersurface of degree $d$. If
\[
 n>2^d(d-1)\binom{de+r-1}{r-1},
\]
then $\operatorname{Mor}_e(\mathbb P^r,X)$ is irreducible and has dimension $\mu_r(e)$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:Q_e}
    Under the hypothesis of Theorem \ref{thm:main}, the projective coefficient space $Q_e(X)$ is irreducible of dimension $\mu_r(e)$.
<!-- end theorem-2 -->
<!-- lemma-1: body_md -->
\label{lem:shrinking}
Let $L_1,\ldots,L_N\in \mathbb K_\infty[t_1,\ldots,t_N]$ be linear forms
\[
 L_i(\mathbf t)=\sum_{j=1}^N\gamma_{ij}t_j
\]
whose coefficient matrix is symmetric. Let $A\in\mathbb Q_{\geq 0}$ and $Z\in\mathbb Q_{\leq 0}$, and put
\[
 \mathcal N_A(Z)=\#\left\{\mathbf t\in\mathbb{F}_q[u]^N:
 |t_i|<q^{A+Z},\quad \|L_i(\mathbf t)\|<q^{-A+Z}
 \ \text{for every }i\right\}.
\]
If $Z_1\leq Z_2\leq 0$, $A-Z_2\geq 0$, and $A\pm Z_i$ are integers, then
\[
 \mathcal N_A(Z_2)\leq q^{N(Z_2-Z_1)}\mathcal N_A(Z_1).
\]
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:poly-points}
Let $V\subset \mathbb A^N_{\mathbb{F}_q}$ be an affine variety of dimension $m$. For every integer $J\geq 1$,
\[
 \#\{\mathbf x\in V(\mathbb{F}_q[u]):|\mathbf x|<q^J\} \leq \delta(V)\cdot q^{Jm},
\]
where $\delta(V)= \sum_i\deg V_i$ with $V=\cup_i V_i$ is the decomposition of $V$ into its irreducible components.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:dirichlet}
Let $L\geq 0$ and let $\alpha\in\mathbb T$. There exist polynomials $a,g\in\mathbb{F}_q[u]$, with $g$ monic and $\gcd(a,g)=1$, such that
\[
 0<|g|\leq q^L,\qquad |g\alpha-a|<q^{-L}.
\]
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
Consider the $q^{L+1}$ fractional parts
\[
\{\{h\alpha\}: \deg h\leq L\}
\]
of $h\alpha$. Partition $\mathbb T$ according to the coefficients of $u^{-1},\ldots,u^{-L}$. Since two fractional parts lie in the same part, their difference gives nonzero $g$ with $\deg g\leq L$ and $\|g\alpha\|<q^{-L}$. We take $a$ to be the polynomial part of $g\alpha$
and canceling the common divisor of $a$ and $g$ proves the result.
<!-- end proof-1 -->
<!-- definition-1: body_md -->
\label{def:deficiency}
A monomial $\mathfrak m$ in the scalar coordinates of $\textbf{x}_1,\ldots,\textbf{x}_s$ is called \emph{$(m_1,\ldots,m_s)$ deficient} if there exists at least one block index $j$ such that
\[
 \deg_{\textbf{x}_j}\mathfrak m<m_j,
\]
where $\deg_{\textbf{x}_j}$ is the total degree in the $n$ scalar coordinates of $\textbf{x}_j$.

\label{prop:uniform}
Under the assumptions above, for every $\alpha\in\mathbb T$ and every
$\omega\in\Omega$ one has

|T_\omega(\alpha)|^{2^{d-1}} \leq M^{2^{d-1}}\frac{E(\alpha)}{E(0)}.

<!-- lemma-4: body_md -->
\label{lem:one-step}
Let $H:B_1\times\cdots\times B_s\to \mathbb K_\infty$ be any function, and fix a block index $j$. Then
\[
 \left|\mathbb E_{\textbf{x}_1,\ldots,\textbf{x}_s}\psi(H(\textbf{x}_1,\ldots,\textbf{x}_s))\right|^2 \leq \mathbb E_{\textbf{h}\in B_j} \left| \mathbb E_{\textbf{x}_1,\ldots,\textbf{x}_s} \psi\bigl(\Delta_{\textbf{h}}^{(j)}H(\textbf{x}_1,\ldots,\textbf{x}_s)\bigr) \right|.
\]
<!-- end lemma-4 -->
<!-- proof-2: body_md -->
Let $\textbf{x}'$ denote all blocks other than $\textbf{x}_j$. By Cauchy--Schwarz in the $\textbf{x}'$-average,
\begin{align}
 \left|\mathbb E_{\textbf{x}'}\mathbb E_{\textbf{x}_j}\psi(H(\textbf{x}_j,\textbf{x}'))\right|^2 &\leq \mathbb E_{\textbf{x}'} \left|\mathbb E_{\textbf{x}_j}\psi(H(\textbf{x}_j,\textbf{x}'))\right|^2.
 \label{eq:CS-outer}
\end{align}
For fixed $\textbf{x}'$, expanding we get
\[
 \left|\mathbb E_{\textbf{x}_j}\psi(H(\textbf{x}_j,\textbf{x}'))\right|^2 =\frac{1}{M_j^2} \sum_{\textbf{x}_j,\textbf{y}_j\in B_j} \psi\bigl(H(\textbf{x}_j,\textbf{x}')-H(\textbf{y}_j,\textbf{x}')\bigr).
\]
Note that the change of variables
\[
 (\textbf{x}_j,\textbf{y}_j)=(\textbf{y}_j+\textbf{h},\textbf{y}_j)
\]
is a bijection from $B_j\times B_j$ to itself. Hence
\[
 \left|\mathbb E_{\textbf{x}_j}\psi(H(\textbf{x}_j,\textbf{x}'))\right|^2 =\mathbb E_{\textbf{h}\in B_j}\mathbb E_{\textbf{y}_j\in B_j} \psi\bigl(\Delta_{\textbf{h}}^{(j)}H(\textbf{y}_j,\textbf{x}')\bigr).
\]
Substituting this into \eqref{eq:CS-outer} we obtain
\[
\left|\mathbb E_{\textbf{x}}\psi(H(\textbf{x}))\right|^2\leq \mathbb E_{\textbf{h}\in B_j} \mathbb E_{\textbf{x}} \psi\bigl(\Delta_{\textbf{h}}^{(j)}H(\textbf{x})\bigr).
\]
The right side as a whole is real and nonnegative, but the inner average for a fixed $\textbf{h}$ need not be. Applying the triangle inequality to the $\textbf{h}$-average proves the lemma.
<!-- end proof-2 -->
<!-- lemma-5: body_md -->
\label{lem:iterated}
Let $i_1,\ldots,i_k\in\{1,\ldots,s\}$ be any sequence of block indices. For $1\leq r\leq k$, let $\textbf{h}_r\in B_{i_r}$ and put
\[
 \mathcal D_r=\Delta_{\textbf{h}_r}^{(i_r)}\cdots\Delta_{\textbf{h}_1}^{(i_1)}.
\]
Then, for any function $H:B_1\times\cdots\times B_s\to \mathbb K_\infty$,
\[
 \left|\mathbb E_{\textbf{x}}\psi(H(\textbf{x}))\right|^{2^k}\leq \mathbb E_{\textbf{h}_1\in B_{i_1}}\cdots \mathbb E_{\textbf{h}_k\in B_{i_k}} \left|\mathbb E_{\textbf{x}}\psi(\mathcal D_kH(\textbf{x}))\right|.
\]
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
We induct on $k$. The case $k=1$ is Lemma \ref{lem:one-step}. Suppose the result is known for $k$. Write
\[
 A_k(\textbf{h}_1,\ldots,\textbf{h}_k) =\mathbb E_{\textbf{x}}\psi(\mathcal D_kH(\textbf{x})).
\]
By the induction hypothesis,
\[
 \left|\mathbb E_{\textbf{x}}\psi(H(\textbf{x}))\right|^{2^k} \leq \mathbb E_{\textbf{h}_1,\ldots,\textbf{h}_k}|A_k(\textbf{h}_1,\ldots,\textbf{h}_k)|.
\]
Squaring both sides, since the increment averages are probability averages, Cauchy--Schwarz gives
\begin{align}
 \left|\mathbb E_{\textbf{x}}\psi(H(\textbf{x}))\right|^{2^{k+1}}
 &\leq
 \mathbb E_{\textbf{h}_1,\ldots,\textbf{h}_k}|A_k(\textbf{h}_1,\ldots,\textbf{h}_k)|^2.
 \label{eq:induction-CS}
\end{align}
For each fixed tuple $(\textbf{h}_1,\ldots,\textbf{h}_k)$, apply Lemma \ref{lem:one-step} to the phase $\mathcal D_kH$ and to block $i_{k+1}$. This gives
\begin{align*}
 |A_k(\textbf{h}_1,\ldots,\textbf{h}_k)|^2 &\leq \mathbb E_{\textbf{h}_{k+1}\in B_{i_{k+1}}}\left|\mathbb E_{\textbf{x}}\psi\bigl(\Delta_{\textbf{h}_{k+1}}^{(i_{k+1})}\mathcal D_kH(\textbf{x})\bigr)\right|.
\end{align*}
Substituting into \eqref{eq:induction-CS} proves the statement for $k+1$, completing the inductive step.
<!-- end proof-3 -->
<!-- lemma-6: body_md -->
\label{lem:multilinear-difference}
Let $V$ be a vector space over a field, let $W$ be another vector space, and let $\Lambda:V^m\to W$ be symmetric and $m$-linear. Set
\[
 Q(\textbf{x})=\Lambda(\textbf{x},\ldots,\textbf{x}).
\]
Then
\begin{align}
 \Delta_{\textbf{h}_m}\cdots\Delta_{\textbf{h}_1}Q(\textbf{x}) &=m!\,\Lambda(\textbf{h}_1,\ldots,\textbf{h}_m),
 \label{eq:complete-diff}
\end{align}
and
\begin{align}
 \Delta_{\textbf{h}_{m-1}}\cdots\Delta_{\textbf{h}_1}Q(\textbf{x}) &=m!\,\Lambda(\textbf{h}_1,\ldots,\textbf{h}_{m-1},\textbf{x}) +C(\textbf{h}_1,\ldots,\textbf{h}_{m-1}),
 \label{eq:almost-complete-diff}
\end{align}
where $C$ is independent of $\textbf{x}$.
<!-- end lemma-6 -->
<!-- proof-4: body_md -->
The inclusion-exclusion formula for iterated differences is
\begin{align}
 \Delta_{\textbf{h}_k}\cdots\Delta_{\textbf{h}_1}Q(\textbf{x}) &=\sum_{I\subseteq\{1,\ldots,k\}} (-1)^{k-|I|} Q\left(\textbf{x}+\sum_{i\in I}\textbf{h}_i\right).
 \label{eq:inc-exc}
\end{align}
We expand each value of $Q$ by multilinearity. Consider a term in the expansion and fix an increment $\textbf{h}_r$. If the term does not contain $\textbf{h}_r$, then its total coefficient in \eqref{eq:inc-exc} is zero, because the contributions from subsets $I$ containing $r$ cancel those from subsets not containing $r$. Hence a term can survive only if it contains every increment $\textbf{h}_1,\ldots,\textbf{h}_k$ at least once.

If $k=m$, the total number of multilinear slots is $m$, so every surviving term contains each of the $m$ increments exactly once and contains no $\textbf{x}$. The symmetry of $\Lambda$ shows that all such terms are equal, and there are $m!$ permutations of the increments among the slots. This proves \eqref{eq:complete-diff}.

If $k=m-1$, every surviving term that still depends on $\textbf{x}$ contains each of the $m-1$ increments exactly once and contains $\textbf{x}$ in the remaining slot. There are $m!$ ways to place these $m$ entries among the slots, and all resulting values are equal by symmetry. This gives the first term on the right of \eqref{eq:almost-complete-diff}. Every other surviving term contains no $\textbf{x}$ and can therefore be absorbed into the function $C$.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
We have that
 \[
 \mathbb E_{\textbf{x}_1,\ldots,\textbf{x}_s} \psi(\Phi_{\alpha,\omega}(\textbf{x}_1,\ldots,\textbf{x}_s)) =\frac{T_\omega(\alpha)}{M}.
 \]
 We distinguish the last block corresponding to $s$. For blocks with $j<s$, we choose exactly $m_j$ increments
\[
 \textbf{h}_{j,1},\ldots,\textbf{h}_{j,m_j}\in B_j.
\]
In block $s$, choose exactly $m_s-1$ increments
\[
 \textbf{h}_{s,1},\ldots,\textbf{h}_{s,m_s-1}\in B_s.
\]
Here we use the convention that this last list is empty if $m_s=1$. The total number of increments is
\[
 \sum_{j=1}^{s-1}m_j+(m_s-1)=d-1.
\]
We consider the combined difference operator
\begin{align}
 \mathcal D &=\left(\prod_{\ell=1}^{m_s-1} \Delta_{\textbf{h}_{s,\ell}}^{(s)}\right) \prod_{j=1}^{s-1} \left(\prod_{\ell=1}^{m_j}\Delta_{\textbf{h}_{j,\ell}}^{(j)}\right).
 \label{eq:D}
\end{align}
Note that the difference operators commute, so their written order is irrelevant.

The number of possible increment tuples is
\begin{equation}\label{eq:D-size}
 D_0=\left(\prod_{j=1}^{s-1}M_j^{m_j}\right)M_s^{m_s-1}.
\end{equation}
Thus the normalised increment average is
\[
 \mathbb E_{\textbf{h}}[-]=\frac{1}{D_0}\sum_{\substack{\textbf{h}_{j,\ell}\in B_j\\ j<s,\ 1\leq\ell\leq m_j\\ j=s,\ 1\leq\ell\leq m_s-1}}-.
\]

Applying Lemma \ref{lem:iterated} to the phase $\Phi_{\alpha,\omega}$ and to this sequence of $d-1$ block differences. We obtain
\begin{align}
 \left|\frac{T_\omega(\alpha)}{M}\right|^{2^{d-1}} &\leq \mathbb E_{\textbf{h}} \left| \mathbb E_{\textbf{x}_1,\ldots,\textbf{x}_s} \psi\bigl(\mathcal D\Phi_{\alpha,\omega}(\textbf{x}_1,\ldots,\textbf{x}_s)\bigr) \right|.
 \label{eq:after-weyl}
\end{align}


Next we compute the differenced leading term $\mathcal DG$. In every block $j<s$, the form $G$ has degree exactly $m_j$, and we apply exactly $m_j$ differences. In block $s$, it has degree exactly $m_s$, and we apply $m_s-1$ differences. Applying Lemma \ref{lem:multilinear-difference} successively in the different blocks gives
\begin{align}
 \mathcal DG(\textbf{x}_1,\ldots,\textbf{x}_s) &=c_*\,\Gamma_f(\textbf{h}_{1,1},\ldots,\textbf{h}_{1,m_1},\ldots, \textbf{h}_{s,1},\ldots,\textbf{h}_{s,m_s-1},\textbf{x}_s)+C_G(\textbf{h}),
 \label{eq:DG}
\end{align}
where $C_G$ is independent of every original variable and $c_*$ is given by \eqref{eq:E}.

We now estimate the differenced remainder. Fix a monomial $\mathfrak m$ occurring in $R_{\alpha,\omega}$. By the deficiency hypothesis, there exists $j$ such that
\[
 \deg_{\textbf{x}_j}\mathfrak m<m_j.
\]
If this happens for some $j<s$, then the $m_j$ differences taken in block $j$ annihilate the monomial. Therefore
\[
 \mathcal D\mathfrak m=0.
\]
Otherwise, the monomial is not deficient in any block $j<s$. Its required deficiency must then occur in block $s$, so
\[
 \deg_{\textbf{x}_s}\mathfrak m<m_s.
\]
After the $m_s-1$ differences in block $s$, the resulting degree in $\textbf{x}_s$ is at most
\[
 \deg_{\textbf{x}_s}\mathfrak m-(m_s-1)\leq 0.
\]
Thus the differenced monomial is independent of $\textbf{x}_s$. It may still depend on $\textbf{x}_1,\ldots,\textbf{x}_{s-1}$ and on all increments; that dependence is harmless.

Summing over the monomials of $R_{\alpha,\omega}$, we conclude that
\begin{equation}\label{eq:DR}
 \mathcal DR_{\alpha,\omega}= C_{R,\alpha,\omega}(\textbf{x}_1,\ldots,\textbf{x}_{s-1};\textbf{h}),
\end{equation}
where the right side is independent of $\textbf{x}_s$. Combining the leading and remainder terms we deduce that
\begin{align}
 \mathcal D\Phi_{\alpha,\omega}(\textbf{x}_1,\ldots,\textbf{x}_s) &=\alpha c_*\,\Gamma_f(\textbf{h}_{1,1},\ldots,\textbf{h}_{s,m_s-1},\textbf{x}_s) +C_{\alpha,\omega}(\textbf{x}_1,\ldots,\textbf{x}_{s-1};\textbf{h}),
 \label{eq:Dphi}
\end{align}
where $C_{\alpha,\omega}$ is independent of $\textbf{x}_s$.

Note that fixing an increment $\textbf{h}$ we get
\begin{align}
 &\mathbb E_{\textbf{x}_1,\ldots,\textbf{x}_s} \psi\bigl(\mathcal D\Phi_{\alpha,\omega}(\textbf{x}_1,\ldots,\textbf{x}_s)\bigr) \notag\\ &\quad= \mathbb E_{\textbf{x}_1,\ldots,\textbf{x}_{s-1}} \left[ \psi\bigl(C_{\alpha,\omega}(\textbf{x}_1,\ldots,\textbf{x}_{s-1};\textbf{h})\bigr) \mathbb E_{\textbf{x}_s}\psi\bigl(\alpha c_*\,\Gamma_f(
 \textbf{h}_{1,1},\ldots,\textbf{h}_{s,m_s-1},\textbf{x}_s)\bigr) \right].
 \label{eq:factor-inner}
\end{align}
The $\textbf{x}_s$-average is independent of $\textbf{x}_1,\ldots,\textbf{x}_{s-1}$. Taking absolute values and using $|\psi(C_{\alpha,\omega})|=1$, we get
\begin{align}
 \left| \mathbb E_{\textbf{x}_1,\ldots,\textbf{x}_s} \psi\bigl(\mathcal D\Phi_{\alpha,\omega}\bigr) \right|\leq \left|\mathbb E_{\textbf{x}_s\in B_s} \psi\bigl(\alpha c_*\,\Gamma_f( \textbf{h}_{1,1},\ldots,\textbf{h}_{s,m_s-1},\textbf{x}_s)\bigr)\right|.
 \label{eq:discard-remainder}
\end{align}

Using linearity and orthogonality \eqref{sum-orth-id} we deduce that
\begin{align}
 \mathbb E_{\textbf{x}_s\in B_s} \psi\bigl(\alpha c_*\,\Gamma_f( \textbf{h}_{1,1},\ldots,\textbf{h}_{s,m_s-1},\textbf{x}_s)\bigr) &=I_\alpha(\textbf{h}),
 \label{eq:I}
\end{align}
where $I_\alpha(\textbf{h})$ is the indicator of the $n$ simultaneous conditions
\begin{equation}\label{eq:dual-conditions}
 \left\|\alpha c_*\Gamma_f(\textbf{h}_{1,1},\ldots,\textbf{h}_{s,m_s-1},\textbf{e}_i)\right\|<q^{-P_s} \qquad (1\leq i\leq n).
\end{equation}
Therefore, we obtain
\begin{equation}\label{eq:ratio-N}
 \left|\frac{T_\omega(\alpha)}{M}\right|^{2^{d-1}} \leq \mathbb E_{\textbf{h}}I_\alpha(\textbf{h}) =\frac{N(\alpha)}{D_0},
\end{equation}
where $N(\alpha)$ is the number of increment tuples satisfying \eqref{eq:dual-conditions}. We now identify $E(\alpha)$ in the above bound. Regarding the last variable $\textbf{z}_{s,m_s}$ as the last summation variable and fixing all the other $d-1$ variables. By definition \eqref{eq:E}, the inner sum is
\[
 \sum_{\textbf{z}_{s,m_s}\in B_s} \psi\Bigl(\alpha c_*\Gamma_f(\textbf{z}_{1,1},\ldots,\textbf{z}_{s,m_s-1},\textbf{z}_{s,m_s})\Bigr).
\]
By orthogonality again, this sum is $M_s$ if the fixed $d-1$ variables satisfy the conditions \eqref{eq:dual-conditions}, and is zero otherwise. Consequently,
\begin{equation}\label{eq:E=N}
 E(\alpha)=M_sN(\alpha).
\end{equation}
When $\alpha=0$, every increment tuple satisfies the dual conditions, so $N(0)=D_0$. Combining \eqref{eq:ratio-N} and \eqref{eq:E=N} we get
\[
|T_\omega(\alpha)|^{2^{d-1}} \leq M^{2^{d-1}}\frac{E(\alpha)}{E(0)}.
\]
This completes the proof.

<!-- lemma-8: body_md -->
\label{lem:maj-min}
Let $1\leq J\leq P_{\min}$. Then one of the following alternatives holds:
\begin{enumerate}
\item the following bound
\[
 |\tilde E(\alpha)|\leq (d-1)^n \tilde E(0)q^{-\sigma J};
\]
\item there exist coprime $a,g\in\mathbb{F}_q[u]$, with $g$ monic, such that
\[
 0<|g|\leq q^{(d-1)(J-1)},
 \qquad
 |g\alpha-a|<q^{-H+(d-1)J}.
\]
\end{enumerate}
<!-- end lemma-8 -->
<!-- proof-7: body_md -->
Reorder the variables so that $P_d=\max_i P_i$; this is harmless because $\Lambda$ is symmetric. We leave the last variable $\mathbf z_d$ undifferenced. Exactly as in the proof of Proposition \ref{prop:uniform}, using orthogonality,
\[
 \tilde E(\alpha)=q^{nP_d}\tilde N(\alpha),
\]
where $\tilde N(\alpha)$ counts
$(\mathbf z_1,\ldots,\mathbf z_{d-1})$ satisfying
\[
 |\mathbf z_i|<q^{P_i} \quad(1\leq i\leq d-1),
\]
and
\[
 \|\alpha\Lambda(\mathbf z_1,\ldots,\mathbf z_{d-1},\mathbf e_j)\| <q^{-P_d} \quad(1\leq j\leq n).
\]

Define
\[
B_0=P_d
\]
and, for $1\leq t\leq d-1$,
\[
B_t = P_d+\sum_{i=1}^{t}(P_i-J).
\]
Thus
\[
B_t-B_{t-1}=P_t-J.
\]

For $0\leq t\leq d-1$, let $\tilde N^{(t)}(\alpha)$ denote the number of tuples
\[
(\mathbf z_1,\ldots,\mathbf z_{d-1})
\]
satisfying
\[
|{\mathbf z_i}|<q^J,
\qquad 1\leq i\leq t,
\]
\[
|{\mathbf z_i}|<q^{P_i},
\qquad t<i\leq d-1,
\]
and
\[
\| \alpha\Lambda(\mathbf z_1,\ldots,\mathbf z_{d-1},\mathbf e_j) \|< q^{-B_t}, \qquad 1\leq j\leq n.
\]
We claim that for every $1\leq t\leq d-1$,
\[
\tilde N^{(t-1)}(\alpha)\leq q^{n(P_t-J)}\tilde N^{(t)}(\alpha).
\]
To see this, fix $t$, and freeze all variables except $\mathbf z_t$. Write
\[
\mathbf z_t=(z_{t,1},\ldots,z_{t,n}).
\]
For each $1\leq j\leq n$, define a linear form in the coordinates of $\mathbf z_t$ by
\[
L_j(\mathbf z_t) = \alpha \Lambda( \textbf{z}_1,\ldots,\textbf{z}_{t-1}, \textbf{z}_t, \textbf{z}_{t+1},\ldots,\textbf{z}_{d-1}, \textbf{e}_j
).
\]
Since $\Lambda$ is linear in the $t$-th slot, we may write
\[
L_j(\mathbf z_t) = \sum_{k=1}^n \gamma_{jk}z_{t,k},
\]
where
\[
\gamma_{jk} = \alpha \Lambda( \textbf{z}_1,\ldots,\textbf{z}_{t-1}, \textbf{e}_k, \textbf{z}_{t+1},\ldots,\textbf{z}_{d-1}, \mathbf e_j ).
\]
Note that $L_1,\ldots,L_n$ form a symmetric system of $n$ linear forms in $n$ variables.

For the fiber occurring in $\tilde N^{(t-1)}(\alpha)$, the variable $\mathbf z_t$ satisfies
\[
|{\textbf{z}_t}|<q^{P_t}
\]
and
\[
\|L_j(\textbf{z}_t)\|<q^{-B_{t-1}}, \qquad 1\leq j\leq n.
\]
Using the notation in Lemma \ref{lem:shrinking}, we set $A_t=\frac{P_t+B_{t-1}}{2}, Z_2=\frac{P_t-B_{t-1}}{2}$. Then $A_t+Z_2=P_t$ and $-A_t+Z_2=-B_{t-1}$. Therefore the original fiber is exactly $\mathcal N_{A_t}(Z_2)$. For the shrunken fiber, set $Z_1=J-A_t$, then $A_t+Z_1=J$ while $-A_t+Z_1= -B_t$. Hence $\mathcal N_{A_t}(Z_1)$ counts precisely those
$\mathbf z_t$ satisfying
\[
|{\mathbf z_t}|<q^J
\]
and
\[
\|{L_j(\mathbf z_t)}\|<q^{-B_t},
\qquad 1\leq j\leq n.
\]
Note that
\[
 B_{t-1}\geq P_d\geq P_t,\qquad J\leq P_t.
\]
Consequently $Z_2\leq0$, $Z_1\leq Z_2$, and
$A_t-Z_2=B_{t-1}\geq0$; moreover $A_t\pm Z_i$ are integers. Shrinking, Lemma \ref{lem:shrinking}, therefore gives
\[
\mathcal N_{A_t}(Z_2) \leq q^{n(Z_2-Z_1)} \mathcal N_{A_t}(Z_1) = q^{n(P_t-J)} \mathcal N_{A_t}(Z_1).
\]
This estimate holds for every fixed choice of the remaining variables. Summing it over all allowed values of those frozen variables yields
\[
\tilde N^{(t-1)}(\alpha)
\leq
q^{n(P_t-J)}\tilde N^{(t)}(\alpha).
\]
The above application of Lemma \ref{lem:shrinking} yields
\begin{equation}\label{eq:multi-shrink}
 \tilde N(\alpha)
 \leq q^{n\sum_{i=1}^{d-1}(P_i-J)}\tilde N_J(\alpha),
\end{equation}
where $\tilde N_J(\alpha)$ counts tuples with
\[
 |\mathbf z_i|<q^J\quad(1\leq i\leq d-1)
\]
and
\begin{equation}\label{eq:NJ-dual}
 \|\alpha\Lambda(\mathbf z_1,\ldots,\mathbf z_{d-1},\mathbf e_j)\| <q^{-H+(d-1)J} \quad(1\leq j\leq n).
\end{equation}
Suppose first that every tuple counted by $\tilde N_J(\alpha)$ belongs to $Z$. The variety $Z$ is cut out by $n$ equations of degree $d-1$, so $\delta(Z)\leq(d-1)^n$. By Lemma \ref{lem:poly-points},
\[
 \tilde N_J(\alpha)\leq(d-1)^n q^{J\dim Z}.
\]
Combining this with the bound for $\tilde E(\alpha)$ and \eqref{eq:multi-shrink}, and using
$\tilde E(0)=q^{nH}$, gives
\[
 |\tilde E(\alpha)| \leq(d-1)^n q^{np_d+n\sum_{i<d}(P_i-J)+J\dim Z} =(d-1)^n \tilde E(0)q^{-\sigma J}.
\]
This is the first alternative.

Otherwise, choose a tuple counted by $\tilde N_J(\alpha)$ that is not in $Z$. For some $j$ the polynomial
\[
 g_0=\Lambda(\textbf{z}_1,\ldots,\textbf{z}_{d-1},\textbf{e}_j)
\]
is nonzero. Since every $\textbf{z}_i$ has degree strictly less than $J$,
\[
 0<|g_0|\leq q^{(d-1)(J-1)}.
\]
By \eqref{eq:NJ-dual}, there is $a_0\in\mathbb{F}_q[u]$ such that
\[
 |g_0\alpha-a_0|<q^{-H+(d-1)J}.
\]
Divide $g_0$ and $a_0$ by their greatest common divisor and make the new
denominator monic. This proves the second alternative.

<!-- proposition-2: body_md -->
\label{prop:mean-value}
Let $\rho>0$. Suppose
\[
\sigma\rho>2(d-1)
\]
and
\[
 P_{\min}\geq \frac{H-d+1}{2(d-1)}.
\]
Put $\delta=\sigma\rho-2(d-1)>0$. Then
\begin{equation}\label{eq:mean-value}
 \int_\mathbb T |\tilde E(\alpha)|^\rho\,d\alpha \leq \tilde E(0)^\rho q^{-H+d-1} \left(1+(d-1)^{n\rho}\frac{q^{-\delta}}{1-q^{-\delta}}\right).
\end{equation}
<!-- end proposition-2 -->
<!-- proof-8: body_md -->
We bound contribution from $\mathfrak M(1)$ trivially. Only possible denominator in \eqref{eq:major-arcs} for $J=1$ has degree zero, using the normalization
\[
 d\alpha(\mathfrak M(1))\leq q^{-H+d-1},
\]
and therefore using triangle inequality
\begin{equation}\label{eq:M1}
 \int_{\mathfrak M(1)}|\tilde E(\alpha)|^\rho\,d\alpha \leq \tilde E(0)^\rho q^{-H+d-1}.
\end{equation}
For $J\geq 1$, put
\[
 I_J=\int_{\mathfrak M(J+1)\setminus\mathfrak M(J)}|\tilde E(\alpha)|^\rho\,d\alpha.
\]
If $J> P_{\min}$, then
\[
 2(d-1)J\geq 2(d-1)(P_{\min}+1)\geq H+d-1.
\]
Thus $\mathfrak M(J)=\mathbb T$ by Lemma \ref{lem:dirichlet}, and hence $I_J=0$. Now let $1\leq J\leq P_{\min}$. On $\mathfrak M(J+1)\setminus\mathfrak M(J)$, the second alternative of Lemma \ref{lem:maj-min} with parameter $J$ is impossible, so
\[
 |\tilde E(\alpha)|^\rho \leq(d-1)^{n\rho}\tilde E(0)^\rho q^{-\sigma\rho J}.
\]
For a fixed denominator $g$, summing the measures of the arcs over all reduced numerators $a$ gives the bound
\[
d\alpha(\mathfrak M(J+1)) \leq \sum_{\substack{\deg g\leq (d-1)J,\\ \mathrm{monic}}} \phi(g)\cdot \frac{q^{-H+(d-1)(J+1)}}{|g|},
\]
where $\phi$ is the usual Euler totient function. Using the fact (see, e.g. \cite{Rosen2002})
\[
\sum_{\substack{\deg g\leq (d-1)J,\\ \mathrm{monic}}} \frac{\phi(g)}{|g|}= q^{(d-1)J},
\]
we obtain
\[
d\alpha(\mathfrak M(J+1)) \leq q^{-H+(d-1)(J+1)+(d-1)J}.
\]
It follows that
\begin{align*}
 I_J &\leq(d-1)^{n\rho}\tilde E(0)^\rho q^{-H+(d-1)(J+1)+(d-1)J-\sigma\rho J}\\
 &= (d-1)^{n\rho}\tilde E(0)^\rho q^{-H+d-1-\delta J}.
\end{align*}
Summing this estimate over $J\geq1$ and adding \eqref{eq:M1} proves \eqref{eq:mean-value}.

\label{lem:balanced}
For every $\boldsymbol\beta\in\Delta_{r-1}(de)$, there are $\textbf{a}_1,\ldots,\textbf{a}_d\in\Delta_{r-1}(e)$ such that
\[
 \textbf{a}_1+\cdots+\textbf{a}_d=\boldsymbol\beta

Let us write
\[
 |\boldsymbol\beta|=dw+s, \qquad 0\leq s<d.
\]
Regarding $\boldsymbol\beta$ as a multiset containing $\beta_j$ copies of the $\textbf{e}_j$, standard basis vector of $\mathbb Z^{r-1}$, we partition this multiset into $d$ piles, $s$ of cardinality $w+1$ and $d-s$ of cardinality $w$. <!-- lemma-12: body_md -->
\label{lem:isolation}
Let
\[
 \Phi_{\boldsymbol\alpha}(\mathbf t)= \sum_{\boldsymbol\eta\in\Delta_{r-1}(de)} \alpha_{\boldsymbol\eta}F_{\boldsymbol\eta}(\mathbf t),
 \qquad \boldsymbol\alpha\in\mathbb T^{\Delta_{r-1}(de)}.
\]
Fix $\boldsymbol\beta$, and regard this phase as a polynomial in the selected blocks $\mathbf t_{\boldsymbol\gamma_1},\ldots,
\mathbf t_{\boldsymbol\gamma_s}$, keeping every other block fixed. Then
\begin{equation}\label{eq:isolation}
 \Phi_{\boldsymbol\alpha}(\mathbf t)= \alpha_{\boldsymbol\beta}G_{\boldsymbol\beta} (\mathbf t_{\boldsymbol\gamma_1},\ldots, \mathbf t_{\boldsymbol\gamma_s})+R_{\boldsymbol \alpha, \boldsymbol\beta}(\mathbf t),
\end{equation}
where
\begin{equation}\label{eq:G-beta}
 G_{\boldsymbol\beta}(\textbf{x}_1,\ldots,\textbf{x}_s)= \frac{d!}{m_1!\cdots m_s!} \Gamma_f( \underbrace{\mathbf x_1,\ldots,\textbf{x}_1}_{m_1},\ldots,
 \underbrace{\textbf{x}_s,\ldots,\textbf{x}_s}_{m_s}),
\end{equation}
and every monomial of $R_{\boldsymbol{\alpha, \beta}}$ has degree strictly less than $m_j$ in at least one selected block $j$.
<!-- end lemma-12 -->
<!-- proof-12: body_md -->
Every monomial in every $F_{\boldsymbol\eta}$ has total degree $d$ in all coefficient blocks. A monomial having degree at least $m_j$ in every selected block has selected degree at least $m_1+\cdots+m_s=d$. Hence it has degree exactly $m_j$ in each selected
block and contains no unselected block. Its index is necessarily
\[
 m_1\boldsymbol\gamma_1+\cdots+m_s\boldsymbol\gamma_s=\boldsymbol\beta.
\]
Thus it comes only from $F_{\boldsymbol\beta}$. In the ordered sum \eqref{eq:F-beta}, the corresponding multiset occurs $d!/(m_1!\cdots m_s!)$ times. All other monomials are deficient in at least one selected block.

<!-- theorem-3: body_md -->
\label{thm:finite-field}
Suppose that $f$ is a nonsingular degree $d$ form over $\mathbb{F}_q$ and
\[
 n>2^d(d-1)C.
\]
Then there is $\delta>0$, depending only on $n,d,e,r$, such that
\begin{equation}\label{eq:finite-estimate}
 N_{r, q}(e)
 \leq
 q^{n\binom{e+r}{r}-\binom{de+r}{r}}
 \bigl(1+O(q^{-\delta})\bigr),
\end{equation}
where the implied constant is independent of $q$.

<!-- lemma-13: body_md -->
\label{lem:linear-plane}
Under the assumption
\[
 n> 2^d(d-1)\binom{de+r-1}{r-1},
\]
the hypersurface $X$ contains a linear $\mathbb P^r$.
<!-- end lemma-13 -->
<!-- proof-13: body_md -->
We work over the algebraic closure and use an inductive approach. Start with a point of $X$ and suppose inductively that $L=\mathbb P(W)\simeq \mathbb P^k$ is contained in $X$, where $0\leq k<r$ and $\dim W=k+1$. We want a vector $y\notin W$ such that $\mathbb P(W+Ky)\subset X$.
Because $f|_W=0$, the required condition is that all coefficients of positive powers of $\lambda$ in $f(w+\lambda y)$ vanish identically as functions of $w\in W$. For $1\leq j\leq d$, the coefficient of $\lambda^j$ is a form
of degree $d-j$ in $w$. Thus the number of homogeneous equations imposed on the class $[y]\in\mathbb P(K^n/W)\simeq \mathbb P^{n-k-2}$ is
\begin{equation}\label{eq:extension-equations}
 \sum_{j=1}^d\binom{d-j+k}{k}=\binom{d+k}{k+1}.
\end{equation}
If
\begin{equation}\label{eq:extension-condition}
 n-k-2\geq\binom{d+k}{k+1},
\end{equation}
then the projective dimension theorem ensures that these equations have a
common zero in $\mathbb P^{n-k-2}$, and this zero gives the required extension.

It remains to check \eqref{eq:extension-condition}. Since $e\geq1$ and
$k\leq r-1$,
\[
 \binom{de+r-1}{r-1}\geq\binom{d+r-1}{r-1}.
\]
Moreover
\[
 \binom{d+r-1}{r-1}=\frac r d\binom{d+r-1}{r} \geq\frac r d\binom{d+k}{k+1}.
\]
For $r\geq2$ and $d\geq2$, $2^d(d-1)r/d\geq4$. Hence the bound on degree $d$ implies,
\[
 n>\binom{d+k}{k+1}+k+2.
\]
Thus \eqref{eq:extension-condition} holds. Induction constructs a linear $\mathbb P^r\subset X$.

<!-- proof-14: body_md -->
The above discussion completes the proof when the base field is algebraic closure of a finite field.

We treat the general case by spreading out, see \cite[Section 2]{Browning2017}, \cite[Section 3]{browning2025rationalsurfaceslowdegree} for details. Suppose the base field has characteristic zero, or it is an arbitrary algebraically closed field of characteristic $p>d$. Spreading out $f$, $X$, and the coefficient scheme over an integral finitely generated subring of the base field, the preceding argument then applies to every closed fiber after shrinking the base. If the geometric generic fiber were reducible or had dimension different from $\mu_r(e)$, then after a finite extension of the generic point its components and their dimensions would spread out over a nonempty open subset, contradicting the finite field fibers. Thus the geometric generic fiber is irreducible of the required dimension. Base change to the given algebraically closed field completes the proof of Theorem \ref{thm:Q_e}.

<!-- remark-1: body_md -->
\label{rem:singular-targets}
The smoothness hypothesis is used in Lemma \ref{lem:sigma} to bound the codimension of the multilinear singular locus. More generally, let
\[
\mathfrak B(f):=\operatorname{codim}_{\mathbb A^n} \{x\in\mathbb A^n:\nabla f(x)=0\}.
\]
The same diagonal argument gives $\sigma\geq \mathfrak B(f)$. Consequently, the analytic estimate underlying Theorem \ref{thm:finite-field} continues to apply under the condition
\[
\mathfrak B(f)> 2^d(d-1)\binom{de+r-1}{r-1}.
\]
We restrict the statements of the main theorems to smooth hypersurfaces in order to keep their geometric formulation transparent.
