<!-- conj-1: body_md -->
\label{ca}
		\emph{"Is it possible for the singular set of an area-minimizing integer (real) rectifiable current to be a Cantor type set with possibly non-integer Hausdorff dimension?"}
<!-- end conj-1 -->
<!-- thm-1: body_md -->
\label{thmm}
The hypersurface $M\subset\mathbb{R}^{n+m+l}$ in \cite[Theorem 3.7]{LSfr} is area-minimizing both as an integral current and as a mod $2$ current provided the construction parameters $\tau$ and $\delta$ are small, the Lawson cone $M_{(n,m)}$ is area-minimizing, and $(n,m)\not=(3,5)$.
<!-- end thm-1 -->
<!-- defn-1: body_md -->
\label{defnlc}
	The Lawson cone $(n,m)$, denoted by $M_{n,m}$, is defined in the standard flat $\mathbb{R}^n\times\mathbb{R}^m$ by
	\begin{align*}
		M_{n,m}=\left\{(x,y)\in\mathbb{R}^n\times\mathbb{R}^m:|\xi|=\sqrt{\frac{{m-1}}{{n-1}}}|x|\right\}.
	\end{align*}
<!-- end defn-1 -->
<!-- fact-1: body_md -->
\cite{BLep,PSmc}
	The Lawson cone $M_{n,m}$ is an area-minimizing integral current if and only if $n,m\ge 2,n+m\ge 9,$ or $(n,m)=(3,5),(4,4),(5,3).$
<!-- end fact-1 -->
<!-- defn-2: body_md -->
\label{defnh}
	Define
\begin{align*}
		E(x,\xi,y)=\begin{cases}
		\left(u(x,y)^2-|\xi|^2\right)u(x,y)^d,&\textnormal{if }|\xi|\le u(x,y),\\
		\left(u(x,y)^2-|\xi|^2\right)|\xi|^d,&\textnormal{if }|\xi|\ge u(x,y),\\
	\end{cases}
\end{align*}with $d\in(0,\infty)$.
<!-- end defn-2 -->
<!-- lem-1: body_md -->
\label{lemsub}
If $d=\frac{9}{5},(n,m)=(4,4),(5,3)$ or $d=3,n+m\ge 9$	then we have in the metric $g$\begin{align*}
		\operatorname{div}_g \frac{\nabla^g E}{\left\lVert\nabla^g E\right\rVert_g}\begin{cases}
			\ge 0,&\textnormal{if }u(x,y)-|\xi|\ge 0,\\
			\le0,&\textnormal{if } u(x,y)-|\xi|\le 0,\\
		\end{cases}
	\end{align*}
<!-- end lem-1 -->
<!-- fact-2: body_md -->
\label{fctl}
	For a quadratic polynomial $L:\mathbb{R}\to\mathbb{R}$ let $L_2,L_1,L_0$ be the coefficients of second, first and zeroth order terms respectively. If 
	\begin{align}\label{eqlll}
		L_2+L_1+L_0,L_1+2L_0,L_0\ge0,
	\end{align}then we have
	\begin{align}\label{eqlllt}
		\inf_{t\in[0,1]}L(t)\ge0.
	\end{align}Furthermore, if strict inequality holds in (\ref{eqlll}), then strict inequality holds in (\ref{eqlllt})
<!-- end fact-2 -->
<!-- proof-1: body_md -->
We have\begin{align*}
		L(t)=L_2t^2+L_1t+L_0=(L_2+L_1+L_0)t^2+t(1-t)(L_1+2L_0)+(1-t)^2L_0.
	\end{align*}The claimed fact follows directly.
<!-- end proof-1 -->
<!-- fact-3: body_md -->
\label{fctpoly}
	If $P_0:\mathbb{R}\to\mathbb{R} $ is a quadratic polynomial with strictly positive infimum on $[0,1]$ and $Q_0$ is another quadratic polynomial, then $P_0+\tau Q_0$ has strictly positive infimum on $[0,1]$ provided $\tau$ is small.
<!-- end fact-3 -->
<!-- proof-2: body_md -->
By positive infimum of $P_0,$ $\frac{|Q_0|}{P_0}$ is a bounded continuous function on $[0,1]$. We have
	\begin{align*}
		P_0+\tau Q_0=P_0\left(1+\tau\frac{Q_0}{P_0}\right)\ge P_0\left(1-\tau\frac{|Q_0|}{P_0}\right)>0.
	\end{align*}
<!-- end proof-2 -->
<!-- fact-4: body_md -->
\label{fctsubc}
Let $O$ be an oriented open set in $\mathbb{R}^{N+1}$ and assume that $O$ represents an integral current and let $M=\partial O$ both in the sense of sets and currents. If $E$ is a  function on $\mathbb{R}^{N+1}$ satisfying,
	\begin{itemize}
		\item $E^{-1}(0)=M,E^{-1}(-\infty,0)=O,$
		\item $\frac{\nabla^g E}{\left\lVert\nabla^g E\right\rVert_g}$ is bounded, continuous outside a set ${\operatorname{Sing}} \frac{\nabla^g E}{\left\lVert\nabla^g E\right\rVert_g}$ of Hausdorff $N$-dimensional measure $0$,
		\item $\frac{\nabla^g E}{\left\lVert\nabla^g E\right\rVert_g}$ is locally Lipschitz outside a set ${\operatorname{Sing}}^{Lip} \frac{\nabla^g E}{\left\lVert\nabla^g E\right\rVert_g}$ where the restriction of Hausdorff $N$-dimensional measure is $\sigma$-finite,
		\item $\operatorname{div}_g \frac{\nabla^g E}{\left\lVert\nabla^g E\right\rVert_g}$ has the same signs with $E$ off $M,$ i.e.,.
		\begin{align*}
			\operatorname{div}_g \frac{\nabla^g E}{\left\lVert\nabla^g E\right\rVert_g}
			\begin{cases}
				\ge 0, &\textnormal{ on }O^\complement\\
				\le 0,& \textnormal{ on }O.
		\end{cases}	\end{align*}
	\end{itemize}
	Then $M$ is area-minimizing both as an integral current and as a mod $2$ current.
