<!-- theorem-1: body_md -->
\label{thm:main}
  Let $\mathrm{M}$ be a loopless matroid and $\mathcal{L}_P$ a zonotopal class in $K(\mathrm{M})$.
  Then $\chi_{\mathrm{M},P}(q)$ is magic positive. In particular,
  $h^{*}_{\mathrm{M},P}(t)$ only has real zeros.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:main-EC-independence}
  If $\mathcal{L}_{P}$ is a zonotopal class in $K(\mathrm{M})$, then 
  \[
    \chi_{\mathrm{M},P}(q)=\sum_{I\in\mathrm{T}(\mathrm{M})}\omega_{\mathrm{M}, P}(I)q^{|I|}.
  \]
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:main-m0n}
  For $n\geqslant3$ and nonnegative integers $y = \begin{pmatrix}y_S \end{pmatrix}$ for $S \in \binom{[n-1]}{3}$,
  \[
    \chi\big(\overline{\mathcal M}_{0,n},\mathcal{L}_y^{\otimes q}\big)
    =\sum_{I\in\mathrm{T}_n}\left(\prod_{S\in I}y_S\right)q^{|I|}.
  \]
  This polynomial is magic positive, and its $h^{*}$-polynomial is real-rooted.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:main-psi}
  Let $n\geqslant3$, $d=n-3$, and $i\ne j$ in $[n]$. Then
  \[
    h^{*}\big(\overline{\mathcal M}_{0,n},\mathcal{L}_i\otimes\mathcal{L}_j;t\big)
    =\sum_{k=0}^d\binom dk^2t^k.
  \]
  This polynomial is real-rooted. The Snapper polynomial
  $\chi\big(\overline{\mathcal M}_{0,n},(\mathcal{L}_i\otimes\mathcal{L}_j)^{\otimes q}\big)$ is magic

  $\omega_{\mathrm{M}, P}(F)\geqslant\kappa_F$ for every supported flat, and

For pairs with nonnegative weights,

    \text{zonotopal}\ \Longrightarrow\ \text{saturated}\
    \Longrightarrow\ \text{magic positive}\
    \Longrightarrow\ \text{weakly saturated}.

<!-- definition-2: body_md -->
\label{def:matroid-independent-sets}
A \emph{matroid} $\mathrm{M}$ is a pair $(E, \mathcal{I})$ consisting of a finite ground set $E$ and a collection $\mathcal{I}$ of subsets of $E$, called \emph{independent}, satisfying the following properties:
\begin{enumerate}[leftmargin=20pt, font=\normalfont]
  \item \emph{Nonempty:} $\varnothing \in \mathcal{I}$.
  \item \emph{Hereditary:} If $X \in \mathcal{I}$, and $Y \subseteq X$, then $Y \in \mathcal{I}$.
  \item \emph{Independence augmentation property:} If $X, Y \in \mathcal{I}$ with $\left\lvertX\right\rvert < \left\lvertY\right\rvert$, then there exists an element $e \in Y \setminus X$ such that $X \cup \{e\} \in \mathcal{I}$.
\end{enumerate}
<!-- end definition-2 -->
<!-- theorem-6: body_md -->
\label{thm:edmonds}
  Let $E$ be a finite set and $f \colon 2^E \setminus \{\varnothing\} \to \mathbb{Z}$ a
  nonnegative, submodular function that is \emph{weakly increasing}, meaning $f(S) \leqslant f(T)$
  whenever $\varnothing \neq S \subseteq T$. 
  Then the set
  \[
    \mathcal{I}_f \coloneqq \big\{ I \subseteq E :\ |J| \leqslant f(J) \text{ for every nonempty } J \subseteq I \big\}
  \]
  is the collection of independent sets of a matroid on $E$. In this case, we call this matroid $\mathrm{M}_f$.
<!-- end theorem-6 -->
<!-- lemma-1: body_md -->
\label{lem:dilworth-matroid}
  Let $\mathcal{F} \subseteq \mathcal{L}(\mathrm{M}) \setminus \{\varnothing\}$ be a collection of
  nonempty flats of $\mathrm{M}$.  Then the function
  \[
    f(J) \;=\; \operatorname{rk}_{\mathrm{M}}\Big(\bigcup_{F \in J} F\Big) - 1,
    \qquad \varnothing \neq J \subseteq \mathcal{F},
  \]
  is nonnegative, weakly increasing and submodular.
<!-- end lemma-1 -->
<!-- definition-3: body_md -->
\label{def:dilworth}
  Let $\mathcal{F} \subseteq \mathcal{L}(\mathrm{M}) \setminus \{\varnothing\}$ and let $f$ be the
  function of \Cref{lem:dilworth-matroid}.  The \emph{Dilworth truncation} of
  $\mathrm{M}$ along $\mathcal{F}$ is the matroid $\mathrm{T}_{\mathcal{F}}(\mathrm{M}) \coloneqq \mathrm{M}_f$ on the
  ground set $\mathcal{F}$ given by \Cref{thm:edmonds}. Explicitly, its independent
  sets are
  \begin{equation}\label{eq:dilworth}
    \mathrm{T}_{\mathcal{F}}(\mathrm{M}) \;=\; \Big\{ I \subseteq \mathcal{F} \;:\;
      \operatorname{rk}_{\mathrm{M}}\Big(\bigcup_{F \in J} F\Big) \;\geqslant\; |J| + 1
      \ \text{ for every nonempty } J \subseteq I \Big\}.
  \end{equation}
<!-- end definition-3 -->
<!-- example-1: body_md -->
\label{ex:dilworth-boolean}
  Let $\mathrm{M} = \textrm{B}_n$ be the Boolean matroid on $[n]$, whose flats are all
  subsets of $[n]$ and whose rank function is the cardinality of sets.  Its lines are the
  $2$-subsets of $[n]$, and so $\mathrm{T}(\textrm{B}_n)$ has ground set equal to the edge set of the
  complete graph $K_n$. For any set $I$ of edges, $I$ is independent in $\mathrm{T}(\textrm{B}_n)$ if and only if for every nonempty $J \subseteq I$,
  \[
  \operatorname{rk}_{\mathrm{M}}\Big(\bigcup_{F \in J} F \Big) = \Big| \bigcup_{F \in J } F \Big| \;\geqslant\; |J| + 1
  \]
  where the union on the left is a union of subsets in the original matroid $\mathrm{M}$ with ground set $[n]$ thought of as the vertex set of $K_n$.
  This requires that no subset of $I$ is a circuit. Thus $\mathrm{T}(\textrm{B}_n) = \mathrm{M}(K_n)$
  is the graphic matroid of $K_n$ of rank $n-1$, and the lattice of
  flats of $\mathrm{T}(\textrm{B}_n)$ is the partition lattice of $[n]$.
<!-- end example-1 -->
<!-- proposition-1: body_md -->
\label{thm:vd-shellability}
  The independence complex of a matroid is vertex decomposable, and therefore, shellable.
<!-- end proposition-1 -->
<!-- lemma-2: body_md -->
\label{lem:weighted-h}
  Let $\Delta$ be a pure, $(d-1)$-dimensional, shellable complex, and let
  $w \colon \Delta(1) \to \mathbb{Z}_{\geqslant 1}$ be a weight function on the vertices of $\Delta$.  Then the weighted $h$-polynomial
  \[
  h(\Delta, w; y) \;=\; \sum_{B \in \Delta(d)} w(\rho(B))\, y^{|\rho(B)|} \prod_{v \in B \setminus \rho(B)} \big(1 + (w(v) - 1) y\big),
  \]  In particular, the weighted
  $h$-vector $h(\Delta, w)$ is nonnegative.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Fix a shelling with restriction map $\rho$. For a facet $B$ and
  $R=\rho(B)$, the contribution of the interval $[R,B]$ to
  \eqref{eq:weighted-h} is
  \[
    \sum_{R\subseteq I\subseteq B}w(I)y^{|I|}(1-y)^{d-|I|}
    =w(R)y^{|R|}\prod_{v\in B\setminus R}\big((1-y)+w(v)y\big).
  \]
  The identity follows by summing over the disjoint intervals in \eqref{eq:shelling-partition}.
  Since each factor has nonnegative coefficients with $w(v) \geqslant 1$, the coefficients of monomials in $y$ are nonnegative.
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
\label{lem:support-general}
  Let $\mathrm{M}$ be a loopless matroid, and $P = \sum_{S} y_S\,\Delta_S$ with $y_S \in \mathbb{Z}$. For every flat $F$ of rank at least $2$,
  \[
    \omega_{\mathrm{M}, P}(F) \;=\; \sum_{\textrm{cl}_{\mathrm{M}}(S) = F} y_S .
  \]
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Since $\sigma_{P}$ and $\omega_{\mathrm{M}, P}$ are additive functions on $P$, and so
  is the desired identity, it suffices to consider $P = \Delta_S$.  The minimum of
  $x(T)$ over $\Delta_S$ is attained at a vertex $e_i$ with $i \in S$, and
  $e_i(T) = 1$ precisely when $i \in T$; so that minimum is $1$ if
  $S \subseteq T$ and $0$ otherwise, whence $\sigma_{\Delta_S}(T) = -1$ if
  $S \subseteq T$ and $\sigma_{\Delta_S}(T) = 0$ otherwise.

  For a flat $G$ of $\mathrm{M}$, we have $S \subseteq G$ if and only if
  $\textrm{cl}_{\mathrm{M}}(S) \leqslant G$ in the lattice of flats. Thus, in the M\"{o}bius
  transformation \eqref{eq:moebius}, only the flats above $\textrm{cl}_{\mathrm{M}}(S)$
  contribute, and
  \[
    \omega_{\Delta_S}(F) \;=\;
    \sum_{\textrm{cl}_{\mathrm{M}}(S) \,\leqslant\, G \,\leqslant\, F} \mu_{\mathrm{M}}(G,F).
  \]
  The defining identity for the M\"obius function makes this sum equal to
  $1$ when $F=\textrm{cl}_{\mathrm{M}}(S)$ and $0$ otherwise. Additivity, including signed
  Minkowski sums, implies the result.
<!-- end proof-2 -->

<!-- lemma-4: body_md -->
\label{lem:magic-transform}
  The coefficients of the magic transform $m(f; y)$ in the $y^i$ basis form the magic vector of $f$; that is,
  \[
    f(x) = \sum_{i=0}^{d} m_i\, x^i (1+x)^{d-i}, \qquad
    m(f; y) = \sum_{i=0}^{d} m_i\, y^i.
  \]
  In particular, $f$ is magic positive of degree $d$ if and only if $m(f; y)$ has
  nonnegative coefficients.  In that case
  \[
  m_0 = c_0 = f(0), \qquad m_1 = c_1 - d\,c_0\quad(d\geqslant1), \qquad m_d = (-1)^d f(-1) = h_d ,
  \]
  where $h_d$ is the top coefficient of the $h^{*}$-vector of $f$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
It suffices to prove that the linear transformation $m \colon f \mapsto m(f; y)$ sends $x^{i}(1+x)^{d-i}$ to $y^i$.
  Indeed, substituting $x = y/(1-y)$, so that $1 + x = (1-y)^{-1}$,
  \[
  m (x^i (1+x)^{d-i}; y) = (1-y)^{d}\, \frac{y^i}{(1-y)^i}\, \frac{1}{(1-y)^{d-i}} = y^i.
  \]
  The identity of the coefficients follows from the identities above.

\label{def:zonotopal-class}
  A line bundle $\mathcal{L}_{P}$ in $K(\mathrm{M})$ is \emph{zonotopal}, if the weights $\omega_{\mathrm{M}, P}$ are nonnegative and supported on flats of rank $2$, or the \emph{lines} of $\mathrm{M}$. The corresponding $K$-class is a \emph{zonotopal class}.

<!-- lemma-5: body_md -->
\label{lem:zonotope-is-zonotopal}
  For any loopless matroid $\mathrm{M}$, if a lattice generalized permutohedron $P$ in $\mathbb{R}^{E}$ is a zonotope, then $\mathcal{L}_P$ is zonotopal in $K(\mathrm{M})$.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
Let $P = \sum_{1 \leqslant i<j \leqslant n} y_{ij}\Delta_{ij}$ with $y_{ij}\in\mathbb{Z}_{\geqslant0}$. For every flat $F$ in $\mathrm{M}$ of rank at least $2$,
  \begin{equation}\label{eq:wF}
    \omega_{\mathrm{M}, P}(F) \;=\; \begin{cases}
    \sum\limits_{\textrm{cl}_{\mathrm{M}}(i, j) = F} y_{ij} & \operatorname{rk}_{\mathrm{M}}(F) = 2, \\
    0 & \text{otherwise}.
    \end{cases}
  \end{equation}
  In particular, $(\mathrm{M},P)$ is a zonotopal pair.
  By \Cref{lem:support-general},
  \[
  \omega_{\mathrm{M}, P}(F) = \sum_{\textrm{cl}_{\mathrm{M}}(i, j) = F} y_{ij}.
  \]
  Since the closure of any $2$-subset has rank at most $2$, $\omega_{\mathrm{M}, P}(F) = 0$ for any flat $F$ of rank at least $3$.

<!-- remark-1: body_md -->
Let us clarify the relationship between these zonotopal $K$-classes
and other related zonotopal objects in the recent literature. Holtz--Ron
\cite{holtz2011zonotopal} and Ardila--Postnikov \cite{ardila2010combinatorics}
introduced, for a vector configuration, a family of zonotopal algebras $Z_k$,
named after its zonotope, whose lattice point count, volume, and interior
lattice point count compute $\dim Z_k$ for $k=0$ (external), $k=-1$ (central),
and $k=-2$ (internal) respectively. Crowley--Larson \cite[Corollary~1.2]{crowley2026cohomological}
showed that the graded dual of $Z_k$ is the space of sections of
$\mathcal{O}(1,\ldots,1)(k\alpha)$ on the augmented wonderful variety of the
configuration, where $\alpha$ is pulled back from the hyperplane class on the
projective completion. In the external case $k=0$, the class $\mathcal{O}(1,\ldots,1)$, is the
augmented analogue of the classes, which can be considered as a special case of the classes that we consider here. For $k\neq 0$, $\mathcal{O}(1, \ldots, 1)(k\alpha)$ produces classes
outside our family, as do the hierarchical zonotopal algebras of
Holtz--Ron--Xu \cite{holtz2012hierarchical} and Lenz \cite{lenz2012hierarchical}.

<!-- theorem-10: body_md -->
\label{thm:ec-zonotopal}
  Let $(\mathrm{M}, P)$ be a zonotopal pair.  Then
  \[
  \chi_{\mathrm{M},P}(q) \;=\; \sum_{I \in \mathrm{T}(\mathrm{M})} \omega_{\mathrm{M},P}(I)\, q^{\left\lvertI\right\rvert} .
  \]
<!-- end theorem-10 -->
<!-- proof-5: body_md -->
By \Cref{thm:dhr},
\[
 \chi_{\mathrm{M},P}(q)=\sum_{k\in\mathcal{D}(\mathrm{M})}
 \prod_F\binom{q\omega_{\mathrm{M},P}(F)+k_F-1}{k_F}.
\]
If $k_F>0$ and $\omega_{\mathrm{M},P}(F)=0$, the corresponding factor vanishes.
Every nonzero summand is therefore supported on $\operatorname{supp}_2(\omega)$.
For a line $F$, the dragon Hall--Rado condition gives
$k_F\leqslant\operatorname{rk}_{\mathrm{M}}F-1=1$. Hence every contributing tuple is the
indicator vector of a subset $I\subseteq\operatorname{supp}_2(\omega)$.
For such a tuple, the dragon Hall--Rado condition implies that, for every nonempty subset $J\subseteq I$,
\[
 \operatorname{rk}_{\mathrm{M}}\Big(\bigcup_{F\in J}F\Big)\geqslant |J|+1,
\]
which is precisely the independence condition in $\mathrm{T}(\mathrm{M},P)$.
The corresponding summand is
\[
\prod_{F\in I}q\omega_{\mathrm{M},P}(F)=\omega_{\mathrm{M},P}(I)q^{|I|}.
\]
The result follows by summing the formula over $\mathrm{T}(\mathrm{M},P)$ and noting that extending to $\mathrm{T}(\mathrm{M})$ only adds zero summands.
<!-- end proof-5 -->
<!-- theorem-11: body_md -->
\label{thm:magic-zonotopal}
  If $(\mathrm{M}, P)$ is zonotopal, then $\chi_{\mathrm{M},P}(q)$ is magic positive, and
  its magic vector is the $\omega_{\mathrm{M}, P}$-weighted $h$-vector of the independence complex of $\mathrm{T}(\mathrm{M},P)$.  In
  particular, $h^{*}_{\mathrm{M},P}(t)$ is real-rooted.
<!-- end theorem-11 -->
<!-- proof-6: body_md -->
Let $\Delta$ denote the independence complex of $\mathrm{T}(\mathrm{M},P)$, and set
  $d = \operatorname{rk}\big(\mathrm{T}(\mathrm{M},P)\big)$.  By
  \Cref{thm:ec-zonotopal}, $d = \deg \chi_{\mathrm{M},P}(q)$ and 
  \[
    \chi_{\mathrm{M},P}(q) \;=\; \sum_{I \in \Delta} \omega(I)\, q^{\left\lvertI\right\rvert},
  \]
  which is the weighted $f$-polynomial $f(\Delta, \omega_{\mathrm{M},P}; q)$.
  Its magic transform \eqref{eq:magic} is
  \[
    m(\chi_{\mathrm{M},P}; y) \;=\; (1-y)^d\, \chi_{\mathrm{M},P}\Big(\frac{y}{1-y}\Big)
      \;=\; \sum_{I \in \Delta} \omega(I)\, y^{\left\lvertI\right\rvert} (1-y)^{d - \left\lvertI\right\rvert}.
  \]
  By \Cref{cor:magic-f-to-h}, the magic vector of $\chi_{\mathrm{M},P}$ is the
  $\omega$-weighted $h$-vector of $\Delta$.
  Since $\Delta$ is the independence complex of a matroid, it is pure of dimension $d-1$ and shellable
  by \Cref{thm:vd-shellability}, and its vertex weights satisfy $\omega_{\mathrm{M}, P}(F) \geqslant 1$ by the
  definition of $\operatorname{supp}_2(\omega)$.  Therefore, \Cref{lem:weighted-h} implies that $m(\chi_{\mathrm{M},P}; y)$
  has nonnegative integer coefficients, which by \Cref{lem:magic-transform} is magic
  positivity of $\chi_{\mathrm{M},P}$.  Real-rootedness of $h^{*}_{\mathrm{M},P}$ then follows
  from \Cref{thm:magic-rr}.

\label{cor:magic-extremes}

  complex of $\mathrm{T}(\mathrm{M},P)$, which is pure of dimension $d - 1$ for

\label{ex:specializations}
As a corollary of the main theorem, $h^{*}$-polynomials of zonotopal pairs are real-rooted.
Some classical Eulerian polynomials are recovered as $h^{\ast}(\mathrm{M}, P)$ for a zonotopal pair.
  \begin{enumerate}[leftmargin=20pt]
    \item For $d\geqslant1$ and $k\geqslant1$, if $\mathrm{M} = \textrm{B}_{2d}$ and $P = k\big(\Delta_{12} + \cdots + \Delta_{2d-1,2d}\big)$,
      then $h^{*}(\mathrm{M},P)$ is the Eulerian polynomial of the
      $k$-colored permutations $\mathbb{Z}_k \wr S_d$
      \cite{steingrmsson1992permutation}, that is $A_d(t)$ for $k = 1$ and the
      type $B$ polynomial $B_d(t)$ for $k = 2$ \cite[Theorem 3.4]{brenti1994q}.
      It is also the $\mathbf{s}$-Eulerian polynomial for
      $\mathbf s = (k, 2k, \dots, dk)$ of \cite[Corollary 3.6]{savage2015s}.

    \item For $n\geqslant3$, if $\mathrm{M} = \textrm{B}_n$ and $P$ is the cycle zonotope $\Delta_{12} + \cdots + \Delta_{n-1,n} + \Delta_{1n}$,
      then the Dilworth truncation $\mathrm{T}(\mathrm{M},P) \cong \mathrm{U}_{n-1,n}$ and
      $h^{*}(\mathrm{M},P) = A_n(t)$, the classical Eulerian polynomial of $S_n$.
  \end{enumerate}

<!-- remark-2: body_md -->
Not every real-rooted polynomial in $\mathbb{Z}[t]$ arises as $h^{\ast}(\mathrm{M}, P)$ for a zonotopal pair.
\Cref{thm:magic-zonotopal} implies that the magic vector of a zonotopal pair is a
weighted $h$-vector, so its entries are nonnegative integers. Consider the
third Narayana polynomial $1 + 3t + t^2$, the $h$-polynomial of the graph
associahedron of the path graph on $3$ vertices, which is well known to be real-rooted.  For a zonotopal class $\mathcal{L}_P$ in $K(\mathrm{M})$ which has $\chi_{\mathrm{M}, P}$ of degree $d$, nonnegativity of the magic vector and
$m_0=1$ implies that
\[
  h^{*}_1=\chi_{\mathrm{M},P}(1)-(d+1)\geqslant2^d-d-1.
\]
Thus this polynomial could occur only for $d\leqslant2$, and its nonzero
quadratic coefficient forces $d=2$. In that degree, by
\eqref{eq:hstar-first-letter}, it would be $\sum_i m_i A_{2,i}$. Since the
$A_{2,i}$ form a basis, the expansion is unique, and solving the equation gives
$m = \big(1, \tfrac12, 1\big)$, which is not integral.

<!-- proposition-3: body_md -->
\label{prop:pullback}
  Let $\mathrm{M}$ be loopless and realized by an arrangement $\mathcal{A} \subseteq L$.  For
  every nonempty $S \subseteq E$,
  \[
    \iota^{*} \mathcal{L}_{\Delta_S} \;\cong\; \mathcal{L}_{\textrm{cl}_{\mathrm{M}}(S)} .
  \]
  As a consequence, for every lattice generalized permutohedron $P$,
  \[
    \iota^{*}\, \mathcal{O}_{X_{n-1}}(P) \;\cong\; \mathcal{L}_P .
  \]
<!-- end proposition-3 -->
<!-- proof-8: body_md -->
For the first statement, the polytope $\Delta_S$ is the polytope of the toric variety $\mathbb{P}^{|S|-1}$,
  and, since $\Sigma_{n-1}$ refines its normal fan, it defines a toric morphism
  \[
    p_{S} \colon X_{n-1} \longrightarrow \mathbb{P}^{|S|-1}
  \]
  such that $\mathcal{L}_{\Delta_S} \cong p_S^{\ast}\mathcal{O}(1)$.  Let $T^{n-1}$ denote the dense torus
  of $X_{n-1}$, which can be identified with the dense torus of $\mathbb{P}^{n-1}$.  On the dense torus $T^{n-1}$, the map $p_S$ is the coordinate projection
  \[
    [x_e]_{e \in E} \longmapsto [x_e]_{e \in S} .
  \]

  By the definition of the closure of $S$ in the matroid $\mathrm{M}$,
  $\bigcap_{e \in S} H_e = \bigcap_{e \in \textrm{cl}_{\mathrm{M}}(S)} H_e$, and
  $L^{S} = L^{\textrm{cl}_{\mathrm{M}}(S)}$.  The coordinates $x_e$ with $e \in S$ embed $L^{S}$
  into $\mathbb{C}^{S}$, realizing $\mathbb{P}(L^{S})$ as a linear subspace of $\mathbb{P}^{|S|-1}$.
  Therefore, the aforementioned maps form a commutative square
  \[
    \begin{tikzcd}[row sep=2.2em, column sep=3em]
      W_{\mathcal{A}} \arrow[r, hook, "\iota"] \arrow[d, "\pi_{\textrm{cl}_{\mathrm{M}}(S)}"'] & X_{n-1} \arrow[d, "p_S"] \\
      \mathbb{P}(L^{\textrm{cl}_{\mathrm{M}}(S)}) \arrow[r, hook] & \mathbb{P}^{|S|-1}.
    \end{tikzcd}
  \]
  Pulling back $\mathcal{O}(1)$ on $\mathbb{P}^{\left\lvertS\right\rvert-1}$ along the two paths
  now gives the first statement, and since $\mathcal{O}(1)$ on $\mathbb{P}^{\left\lvertS\right\rvert-1}$ restricts to $\mathcal{O}(1)$ on $\mathbb{P}(L^{\textrm{cl}_{\mathrm{M}}(S)})$, we have
  \[
    \iota^{*}\mathcal{L}_{\Delta_S} \;\cong\; (p_S \circ \iota)^{*}\,\mathcal{O}(1)
    \;=\; \pi_{\textrm{cl}_{\mathrm{M}}(S)}^{*}\,\mathcal{O}(1) \;=\; \mathcal{L}_{\textrm{cl}_{\mathrm{M}}(S)} .
  \]

  For the second statement, expand $\mathcal{O}_{X_{n-1}}(P)$ by \eqref{eq:OP}, apply the
  previous statement to each factor, and group the subsets by their closure:
  \[
    \iota^{*}\, \mathcal{O}_{X_{n-1}}(P)
    \;\cong\;
    \bigotimes_{\varnothing \ne S \subseteq E} \mathcal{L}_{\textrm{cl}_{\mathrm{M}}(S)}^{\otimes y_S}
    \;=\;
    \bigotimes_{\varnothing \ne F \in \mathcal{L}(\mathrm{M})}
      \mathcal{L}_{F}^{\otimes\, \sum_{\textrm{cl}_{\mathrm{M}}(S) = F}\, y_S} .
  \]
  For flats of rank at least two, \Cref{lem:support-general} identifies the
  exponent with $\omega_{\mathrm{M},P}(F)$; the remaining factors are trivial. Thus
  the product is $\mathcal{L}_P$ by \eqref{eq:LP-flats}.

<!-- theorem-12: body_md -->
\label{thm:li}
  Let $\mathrm{M}$ be a loopless matroid of rank $r \geqslant 2$ realized by a central and essential arrangement
  $\mathcal{A} \subseteq L$.  The multigraded Hilbert polynomial of
  \[
  W_{\mathcal{A}} \subseteq \prod_{\mathcal{L}_2(\mathrm{M})}\mathbb{P}^1
  \] is the independence
  polynomial, or equivalently, the $f$-polynomial of the independence complex, of the Dilworth truncation $\mathrm{T}(\mathrm{M})$,
  \begin{equation}\label{eq:hilb}
    H_{W_{\mathcal{A}}}(k) \;=\; \sum_{I \in \mathrm{T}(\mathrm{M})}\ \prod_{F \in I} k_F.
  \end{equation}
  In particular, the multidegree of $W_{\mathcal{A}}$ is the indicator function of the bases of
  $\mathrm{T}(\mathrm{M})$.
<!-- end theorem-12 -->
<!-- proof-10: body_md -->
Since $\operatorname{rk}(\mathrm{M}) \geqslant 2$ and $\mathrm{M}$ is loopless, every $e \in E$ has
  an element $f$ not parallel to it, and then $e$ is contained in
  $\textrm{cl}_{\mathrm{M}}(\{e, f\})$, and so the union of lines of $\mathrm{M}$ contains $E$ and
  \[
  \bigcap_{F \in \mathcal{L}_2(\mathrm{M})} L_F = \bigcap_{e \in E} H_e = \{0\}.
  \]
  Since $L$ has dimension $r$, for a nonempty set $I$ of lines
  \[
    \dim \bigcap_{F \in I} L_F \;=\; r - \operatorname{rk}_{\mathrm{M}}\Big(\bigcup_{F \in I} F \Big) .
  \]
  \cite[Theorem 1.1]{li2018images} states that, for a nonempty set $S$ of bases of $\mathrm{T}(\mathrm{M})$, each of which is a set of lines of $\mathrm{M}$,
  and if we denote $I_S = \bigcap_{B \in S} B$, then the multidegree of $W_{\mathcal{A}}$ is the indicator
  function of the bases of $\mathrm{T}(\mathrm{M})$, and
  \[
    H_{W_{\mathcal{A}}}(k) \;=\; \sum_{\varnothing \ne S \subseteq \mathcal B(\mathrm{T}(\mathrm{M}))}
      (-1)^{|S|-1} \prod_{F \in I_S} (k_F + 1) ,
  \]
  Each term expands as
  \[
    \prod_{F \in I_S} (k_F + 1) \;=\; \sum_{I \subseteq I_S}\ \prod_{F \in I} k_F .
  \]
  A set $I$ of lines satisfies $I \subseteq I_S$ for some nonempty $S \subseteq \mathcal{B}(\mathrm{T}(\mathrm{M}))$ if and only
  if $I \in \mathrm{T}(\mathrm{M})$, and for such $I$
  \[
    \sum_{\substack{S \ne \varnothing \\ I \subseteq I_S}} (-1)^{|S|-1} \;=\; 1 ,
  \]
  by the M\"{o}bius function on the Boolean poset of subsets of the set of bases
  containing $I$.  The desired equality
  \eqref{eq:hilb} follows.

<!-- remark-3: body_md -->
\label{rem:degeneration}
  The multidegrees of $W_{\mathcal{A}}$ are zero or one. Brion's theorem
  \cite[Theorem 1]{brion2003multiplicity} therefore gives a flat degeneration

  multihomogeneous coordinates. It has the same multigraded Hilbert
  polynomial as $W_{\mathcal{A}}$, giving a geometric interpretation of
  \eqref{eq:hilb}.

<!-- proposition-5: body_md -->
\label{prop:m0n-single}
  Let $S \subseteq [n-1]$ with $|S| \geqslant 3$, and let $y\in\mathbb{Z}_{>0}$. Then
  \[
    \chi\big(\overline{\mathcal M}_{0,n}, \mathcal{L}_S^{\otimes yq}\big)
    \;=\; \binom{yq + |S| - 2}{|S| - 2} ,
  \]
  and $\mathcal{L}_S^{\otimes y}$ is magic positive if and only if $y \geqslant |S| - 2$.
  In particular, the Euler characteristic of $\mathcal{L}_{[n-1]}^{\otimes q}$ is not magic
  positive for $n \geqslant 5$.
<!-- end proposition-5 -->
<!-- proof-13: body_md -->
By \eqref{eq:cerberus}, $c = c_S$ is Cerberus if and only if $0 \leqslant c_S \leqslant \left\lvertS\right\rvert - 2$.  
  By \Cref{prop:LLPP-cerberus},
  \[
  \chi\big(\overline{\mathcal M}_{0,n},\mathcal{L}_S^{\otimes yq}\big) = \sum_{0 \leqslant c_S \leqslant \left\lvertS\right\rvert - 2} \binom{y_S \, q + c_S - 1}{c_S} = \binom{y_S \,q + \left\lvertS\right\rvert -2}{\left\lvertS\right\rvert -2}, 
  \] 
  where the last equality is given by the hockey-stick identity. 
  For $y>0$, its magic transform is
  \[
  m(\chi\big(\overline{\mathcal M}_{0,n},\mathcal{L}_S^{\otimes yq}\big); t) = \frac1{(|S|-2)!}\prod_{j=1}^{|S|-2}(j+(y-j)t).
  \] 
  Since the coefficient of $t$ is nonnegative if and only if $y \geqslant \left\lvertS\right\rvert - 2$, we have that 
  $\chi\big(\overline{\mathcal M}_{0,n},\mathcal{L}_S^{\otimes yq}\big)$ is magic positive if and only if $y \geqslant \left\lvertS\right\rvert - 2$.

<!-- theorem-14: body_md -->
\label{thm:psi-two}
  For distinct $i, j \in [n]$,
  \begin{align*}
    \chi\big(\overline{\mathcal M}_{0,n}, (\mathcal{L}_i \otimes \mathcal{L}_j)^{\otimes q}\big)
    &\;=\; \sum_{k=0}^{d} \binom{d}{k}^{2} \binom{q + d - k}{d}, \\
    h^{*}\big(\overline{\mathcal M}_{0,n}, \mathcal{L}_i \otimes \mathcal{L}_j\big)
    &\;=\; \sum_{k=0}^{d} \binom{d}{k}^{2} t^{k} .
  \end{align*}
  The $h^{*}$-polynomial is real-rooted.
<!-- end theorem-14 -->
<!-- proof-14: body_md -->
The case $d=0$ is immediate. Suppose $d \geqslant1$ and without loss of generality let $i = 1, j = 2$. By 
  \cite[Theorem 9.2]{larson2024k}, and also Lee, and Pandharipande
  \cite{lee1997formula,pandharipande1997symmetric},
  \begin{equation}\label{eq:llpp-psi}
    \chi\Big(\overline{\mathcal M}_{0,n},\bigotimes_{k=1}^n\mathcal{L}_k^{\otimes w_k q}\Big)
    =\sum_{|a|\leqslant d}\binom{d}{a_1,\ldots,a_n,d-|a|}
      \prod_{k=1}^n\binom{w_k q}{a_k}.
  \end{equation}
  Setting $w=e_1+e_2$, summing over the second coordinate by Vandermonde's
  identity, 
  \begin{align*}
    \chi\Big(\overline{\mathcal M}_{0,n},\bigotimes_{k=1}^n\mathcal{L}_k^{\otimes w_k q}\Big) &=\sum_{a=0}^d\binom da\binom qa\binom{q+d-a}{d-a}\\
        &=\sum_{a=0}^d\binom da^2\binom{q+d-a}{d}.
  \end{align*}
  The definition of the $h^{*}$-vector in \eqref{eq:Wf-change-of-basis} yields the second formula.

  The formula for the $h^{\ast}$-polynomial identifies it with the $d$th type-B Narayana polynomial, whose real-rootedness follows from a result of orthogonal polynomials \cite{szego1975orthogonal}. If $P_d$ is the Legendre polynomial of degree $d$, then
  \[
    \sum_{k=0}^d\binom dk^2t^k
      =(1-t)^d P_d\!\left(\frac{1+t}{1-t}\right).
  \]
  The $d$ zeros of $P_d$ lie in $(-1,1)$, and so
  their inverse images under this change of variable are real and negative.
<!-- end proof-14 -->
<!-- proposition-6: body_md -->
Let $\mathcal{L}_i$ be the line bundle with $c_1(\mathcal{L}_i) = \psi_i$, and let $i \ne j \in [n]$.  
  The Snapper polynomial  
  $\chi\big(\overline{\mathcal M}_{0,n}, (\mathcal{L}_i \otimes \mathcal{L}_j)^{\otimes q}\big)$ is magic positive
  if and only if $n \leqslant 7$.
<!-- end proposition-6 -->
<!-- proof-15: body_md -->
Let $f_n(q) $ denote $\chi\big(\overline{\mathcal M}_{0,n}, (\mathcal{L}_i \otimes \mathcal{L}_j)^{\otimes q}\big)$. 
  By \Cref{thm:psi-two}, $f_n(0)=1$. By \Cref{cor:magic-extremes}, the first magic coefficient is $m_1=f_n'(0)-d$.
  Differentiating the binomial expression yields
  \[
    m_1=H_d+\sum_{k=1}^d\frac{(-1)^{k-1}}k\binom dk-d=2H_d-d,
    \qquad H_d=\sum_{j=1}^d\frac1j.
  \]
  The function $2H_d - d$ is already negative when $d = 5$, since $2H_5-5=\frac{-13}{30}$. 
  Furthermore, it is monotonously decreasing, since for any $d \geqslant2$,
  \[
  2H_{d+1}-(d+1)-(2H_d-d)=\frac{2}{d+1}-1<0,
  \]
  which implies that magic positivity of $f_n(q)$ fails for $d\geqslant5$.
  For $d=0,1,2,3,4$, respectively, the magic vectors are
  \[
    (1),\quad(1,1),\quad(1,1,1),\quad
    (1,\tfrac23,\tfrac23,1),\quad
    (1,\tfrac16,\tfrac7{12},\tfrac16,1).
  \]
  This proves our assertion.

<!-- lemma-7: body_md -->
\label{lem:single-flat}
  If $\omega_{\mathrm{M}, P}$ is supported on a single flat $F$, with capacity
  $\kappa_F = \operatorname{rk} F - 1 \geqslant 1$ and $\omega = \omega_{\mathrm{M}, P}(F) > 0$, then $d = \kappa_F$,
  \[
  \chi_{\mathrm{M},P}(q) = \binom{\omega q + \kappa_F}{\kappa_F}, 
  \qquad 
  m(\chi_{\mathrm{M},P}; y) \;=\; \frac{1}{\kappa_F!}\ \prod_{j=1}^{\kappa_F}\big(\, j + (\omega - j)\, y \,\big).
  \]
  In particular, $\chi_{M, P}(q)$ is magic positive, if and only if $\omega \geqslant \kappa_F$.
<!-- end lemma-7 -->
<!-- proof-17: body_md -->
By \Cref{thm:dhr}, and the hockey-stick identity,
  \[
    \chi_{\mathrm{M},P}(q) \;=\; \sum_{k = 0}^{\kappa_F} \binom{\omega q + k - 1}{k}
    \;=\; \binom{\omega q + \kappa_F}{\kappa_F}
    \;=\; \frac{1}{\kappa_F!}\prod_{j=1}^{\kappa_F} (\omega q + j) .
  \]
  Since this is a product of $\kappa_F$ linear factors in $q$, $d = \kappa_F$.

  The second and the third identities follow from applying the magic transform.
<!-- end proof-17 -->
<!-- example-10: body_md -->
If $\mathrm{M}$ is a Boolean matroid and $P=k\Delta_F$ with $|F|=d+1$ and $k\geqslant1$,
  then \Cref{lem:single-flat} implies that $P$ is magic positive if and only if
  $k\geqslant d$. Therefore, a simplex of dimension at least $2$ need not be
  magic positive, for instance $P = 2\Delta_{\{1,2,3,4\}}$.
  Since $\chi_{\mathrm{M},k\Delta_F}(q)=\chi_{\mathrm{M},\Delta_F}(kq)$, this is an instance of a
  theorem of Konoike \cite{konoike2025magic}: for any polynomial $f$ with positive
  real coefficients there is a threshold $k_0$ such that $f(kq)$ is magic positive
  for all $k\geqslant k_0$.  \Cref{lem:single-flat} identifies the threshold
  $k_0=d$ for $f(q)=\binom{q+d}{d}$.  Konoike also shows that no uniform bound is
  possible: for every $d\geqslant3$ and every $k$, there is a $d$-dimensional
  polytope whose $k$th dilation has Ehrhart polynomial that is not magic positive.

<!-- lemma-8: body_md -->
\label{lem:lex-shelling}
  Let the bases of $\mathcal{D}(\mathrm{M}, P)$ be in increasing lexicographic order
  $B_1 < \cdots < B_N$, and for each $i$, let
  \[
    C_i \coloneqq \big\{ F : (B_i)_F > 0 \text{ and } B_i - e_F \leqslant B_j \text{ for some } j < i \big\},
  \]
  together with
  \[
    (R_i)_F \coloneqq \begin{cases} (B_i)_F & F \in C_i, \\ 0 & F \notin C_i. \end{cases}
  \]
  Then $\mathcal{D}(\mathrm{M}, P)$ is the disjoint union of
  $[R_i, B_i] = \{k : R_i \leqslant k \leqslant B_i\}$ for $1 \leqslant i \leqslant N$.
<!-- end lemma-8 -->
<!-- proof-18: body_md -->
By the hereditary property, it suffices to prove that
  \[
    [0, B_i] \cap \bigcup_{j < i} [0, B_j] \;=\; \bigcup_{F \in C_i} [0, B_i - e_F],
  \]
  The right-hand side is contained in the left-hand side by the definition of $C_i$.  For
  the other direction, let $k \leqslant B_i$ and $k \leqslant B_j$ with $j < i$, and let $F$ be the
  first coordinate in which $B_j$ and $B_i$ differ, and so $(B_j)_F < (B_i)_F$.
  The exchange property applied to $B_i$, $B_j$ and $F$ implies $F'$ with
  $(B_i)_{F'} < (B_j)_{F'}$ such that $B' = B_i - e_F + e_{F'}$ is a basis.  As
  $B_i$ and $B_j$ agree in all coordinates before $F$, we have $F' > F$, hence $B' < B_i$,
  i.e., $B' = B_{j'}$ for some $j' < i$. Since $B_i - e_F \leqslant B'$, this implies
  $F \in C_i$. Finally, $k_F \leqslant (B_j)_F \leqslant (B_i)_F - 1$, so $k \in [0, B_i - e_F]$.
<!-- end proof-18 -->
<!-- lemma-9: body_md -->
\label{lem:box-factors}
  For every $b \geqslant 0$,
  \[
  \sum_{k=0}^{b} (1-y)^{b-k}\,\phi_{\omega,k}(y) = \Phi_{\omega, b}(y).
  \]
  Furthermore, if $\omega \geqslant b$, then $\Phi_{\omega,b}(y)$ and $\phi_{\omega,b}$ have nonnegative coefficients.
<!-- end lemma-9 -->
<!-- theorem-15: body_md -->
\label{thm:sufficient}
  If $\omega_{\mathrm{M}, P} \geqslant 0$ and $\omega_{\mathrm{M}, P}(F) \geqslant \kappa_F$ for every flat $F$ with $\omega_{\mathrm{M}, P}(F) > 0$, then $\chi_{\mathrm{M}, P}(q)$ is magic
  positive.  Furthermore, the magic transform of $\chi_{\mathrm{M},P}$ is 
  \begin{equation}\label{eq:box-formula}
    m(\chi_{\mathrm{M},P}; y) \;=\; \sum_{i=1}^{N}\ \prod_{F \notin C_i}\Phi_{\omega_F, (B_i)_F}(y)\
    \prod_{F \in C_i}\phi_{\omega_F,(B_i)_F}(y),
  \end{equation}
  and has nonnegative coefficients.
<!-- end theorem-15 -->
<!-- proof-19: body_md -->
Let $d = \deg \chi_{\mathrm{M}, P}(q)$. By \eqref{eq:dhr} and the substitution $q = y/(1-y)$,
  \[
    m(\chi_{\mathrm{M},P}; y) = (1-y)^d\,\chi_{\mathrm{M},P}\Big(\frac{y}{1-y}\Big)
         = \sum_{k \in \mathcal{D}(\mathrm{M}, P)} (1-y)^{d - |k|}\prod_F \phi_{\omega_F, k_F}(y).
  \]
  We split the sum according to the boxes of \Cref{lem:lex-shelling}. On any box $[R_i, B_i]$, one
  has
  \[d - |k| = \sum_F\big((B_i)_F - k_F\big). 
  \] Summing over the box factors yields
  \[
  \prod_F \sum_{k_F = (R_i)_F}^{(B_i)_F}(1-y)^{(B_i)_F - k_F} \, \phi_{\omega_F,k_F}(y).
  \]
  For $F \in C_i$, the inner sum is the single term $\phi_{\omega_F,(B_i)_F}(y)$. For
  $F \notin C_i$, the inner sum is $\Phi_{\omega_F, (B_i)_F}(y)$ by \Cref{lem:box-factors}.
  Then \Cref{lem:box-factors} together with $(B_i)_F \leqslant \kappa_F \leqslant \omega_F$ implies that each factor has nonnegative coefficients.
<!-- end proof-19 -->

<!-- theorem-16: body_md -->
\label{thm:necessary}
  If $\chi_{\mathrm{M},P}(q)$ is magic positive, then $\omega_F \geqslant \delta_F$, for every flat $F$ with
  $\omega_F > 0$, that is, $P$ is a weakly saturated class in $K(\mathrm{M})$.
<!-- end theorem-16 -->
<!-- proof-21: body_md -->
Suppose for contradiction that $\omega_F < \delta_F$. Then the factor $j + (\omega_F - j)y$ of $\Phi_{\omega_F, \delta_F}(y)$
  with $j = \delta_F$ vanishes for $y = \delta_F/(\delta_F - \omega_F) > 0$.  By
  \Cref{lem:forced}, $\Phi_{\omega_F, \delta_F}(y)$ divides $m(\chi_{\mathrm{M},P}; y)$, the
  magic transform has a positive real root.  But a polynomial with nonnegative
  coefficients and constant term $m_0 = 1$ is at least $1$ on $[0, \infty)$.

<!-- corollary-5: body_md -->
\label{cor:box}
  The following are equivalent:
  \begin{enumerate}[leftmargin=20pt]
    \item For every nonempty set $J$ of flats in $\operatorname{supp}(\omega_{\mathrm{M},P})$,
      \[
        \sum_{F \in J} \kappa_F \;\leqslant\; \operatorname{rk}_\mathrm{M}\Big(\bigcup_{F \in J} F\Big) - 1 ;
      \]
    \item 
    \[
    \mathcal{D}(\mathrm{M},P) = \prod_{F \in \operatorname{supp}(\omega_{\mathrm{M},P})} \big([0, \kappa_F] \cap \mathbb{Z}\big).
    \]
  \end{enumerate}
  In this case, $\delta_F = \kappa_F$ for every $F$, the saturated and the weakly
  saturated conditions coincide, and $\chi_{\mathrm{M},P}(q)$ is magic positive if and only
  if $\omega_F \geqslant \kappa_F$ for every $F \in \operatorname{supp}(\omega_{\mathrm{M},P})$.
<!-- end corollary-5 -->
