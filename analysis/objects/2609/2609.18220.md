<!-- theorem-1: body_md -->
\label{thm:main}
Let \(K\) be an imaginary quadratic field. As \( \mathrm{N}(\mathfrak{f}) \rightarrow \infty \) through integral ideals \( \mathfrak{f} \), at least \( (\frac{1}{3} - \mathsf{o}_K(1))h_K^*(\mathfrak f)\) primitive ray class characters modulo \(\mathfrak{f} \) satisfy \( L(\frac{1}{2}, \chi) \neq 0 \).
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
\label{prop:first mollified moment}
    Let \(K\) be an imaginary quadratic field. For \( 0 < \theta < 1/2 \) we have as \( \mathrm{N}(\mathfrak{f}) \rightarrow \infty \) that
    \[
    \sideset{}{^*}\sum_{\chi\bmod\mathfrak f} \mathcal M(\chi) L\left(\frac{1}{2},\chi\right)
    =(1+\mathsf{o}_{K,\theta}(1)) h_K^*(\mathfrak{f}).
    \]
    The asterisk restricts the sum to primitive ray class characters modulo \(\mathfrak f\).
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{prop:second mollified moment}
With the same setup as in Proposition~\ref{prop:first mollified moment}, we have
\[
    \sideset{}{^*}\sum_{\chi\bmod\mathfrak f}
    \left|\mathcal M(\chi)L\!\left(\frac12,\chi\right)\right|^2
    =\left(1+\frac1\theta+\mathsf{o}_{K,\theta}(1)\right)
    h_K^*(\mathfrak f).
\]
<!-- end proposition-2 -->
<!-- lemma-1: body_md -->
\label{lem:ray-class-exact-sequence}
There is an exact sequence
\[
    \begin{tikzcd}
    1 \ar[r] & \mathcal{O}_K^\times \cap K^{\mathfrak{f},1} \ar[r] & \mathcal{O}_K^\times \ar[r] & (\mathcal{O}_K/\mathfrak{f})^\times \ar[r] & \operatorname{Cl}_K(\mathfrak{f}) \ar[r] & I_K/P_K \ar[r] & 1.
    \end{tikzcd}
\]
Consequently,
\[
h_K(\mathfrak f)
=h_K\frac{\phi(\mathfrak f)}
{[\mathcal O_K^\times:\mathcal O_K^\times\cap K^{\mathfrak f,1}]}.
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
See \cite[Section~3.2.1]{Cohen} or \cite[Theorem~V.1.7]{MilneCFT}, taking the infinite part of the modulus to be trivial.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:primitive-ray-orthogonality}
For every nonzero fractional ideal \(\mathfrak m\) with \((\mathfrak m,\mathfrak f)=1\),
\[
\sideset{}{^*}\sum_{\chi \bmod \mathfrak f}\chi(\mathfrak m)
=

	\IfValueTF{#1} {
		\sum_{\substack{#1}}
	}{
		\sum
	}
_{\substack{\mathfrak m\in P_K(\mathfrak t)\\
\mathfrak s\mathfrak t=\mathfrak f}}
\mu(\mathfrak s)h_K(\mathfrak t).
\]
In particular, the number of primitive characters modulo \( \mathfrak{f} \) is
\begin{equation} \label{eq:}
h_K^*(\mathfrak f)
=
\sum_{\mathfrak s\mathfrak t=\mathfrak f}
\mu(\mathfrak s)h_K(\mathfrak t).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Every character modulo \(\mathfrak f\) is induced from a unique primitive
character whose conductor divides \(\mathfrak f\). Then the first result follows from the orthogonality relation and M\"obius inversion. The second result follows by taking \(\mathfrak m=\mathcal O_K\).
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{thm:functional-equation}
Let \(\chi\) be a nontrivial primitive ray class character of conductor
\(\mathfrak f \).
The completed \(L\)-function associated to \( \chi \) is entire and satisfies the functional equation
\[
\varLambda(s,\chi)
=
W(\chi)\,
\varLambda(1-s,\overline{\chi}),
\]
for a complex number \( W(\chi) \) of modulus \( 1\) (the root number).
With the above normalization,
\(W(\chi)=\tau(\chi)/\mathrm{N}(\mathfrak f)^{1/2}\).
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
\label{approximate functional equation}
Fix a weight function
\[
V(y)
=
\frac{1}{2\pi i}
\int_{(3)}
G(s)
\left(\frac{\sqrt{|d_K|}}{2\pi}\right)^s
\frac{\Gamma(\frac{1}{2}+s)}{\Gamma(\frac{1}{2})}
y^{-s}\frac{ds}{s}
\]
where \( G \) is an even entire function, decaying faster than any polynomial in vertical strips with \( G(0) = 1 \). For a nontrivial primitive character \(\chi\) of conductor \(\mathfrak f\),
\[
L\!\left(\frac12,\chi\right)
= \sum_{\substack{\mathfrak a\subseteq\mathcal O_K\\ \mathfrak a\ne(0)}}
\frac{\chi(\mathfrak a)}{\mathrm{N}(\mathfrak a)^{1/2}}
V\!\left(
\frac{\mathrm{N}(\mathfrak a)}
{\mathrm{N}(\mathfrak f)^{1/2}}
\right) +
\frac{\tau(\chi)}{\mathrm{N}(\mathfrak f)^{1/2}}
\sum_{\substack{\mathfrak a\subseteq\mathcal O_K\\
\mathfrak a\ne(0)}}
\frac{\overline{\chi}(\mathfrak a)}{\mathrm{N}(\mathfrak a)^{1/2}}
V\!\left(
\frac{\mathrm{N}(\mathfrak a)}
{\mathrm{N}(\mathfrak f)^{1/2}}
\right).
\]

<!-- lemma-5: body_md -->
\label{approximate functional equation II}
Let \(\chi\) be a nontrivial primitive character of conductor \(\mathfrak f\). Let \(G(s)\)
be an even entire function, decaying faster than any polynomial in every vertical strip, and
normalized by \(G(0)\Gamma(1/2)=1\). Define
\[
\lambda_\chi(n)
:=
\sum_{\substack{\mathfrak a,\mathfrak b\subseteq\mathcal O_K\\
\mathfrak a,\mathfrak b\ne(0)\\
\mathrm{N}(\mathfrak a)\mathrm{N}(\mathfrak b)=n}}
\chi(\mathfrak a)\overline{\chi}(\mathfrak b),
\]
so that
\(L(s,\chi)L(s,\overline\chi)=\sum_{n\ge1}\lambda_\chi(n)n^{-s}\)
for \(\Re s>1\). Then
   \[
   	\left|L\left(\frac{1}{2},\chi\right)\right|^2
	=
	2\sum_{n=1}^{\infty} \frac{\lambda_{\chi}(n)}{n^{1/2}}
	W\left(\frac{4n\pi^2}{|d_{K}|\mathrm{N}(\mathfrak{f})}\right)
   \]
   where
   \[
   	W(y)
	=
	\frac{1}{2\pi i}\int_{(3)}
	\Gamma^2\left(s+\frac{1}{2}\right)G^2(s)s^{-1}y^{-s}\,\mathrm{d}s.
   \]
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
Consider the integral
	\[
		I(\chi)
		=

\label{lem:uniform Mertens}
	    Fix \( B, C > 0 \). As \( x \rightarrow \infty \), the following two
	    estimates hold:
	    \begin{align}

<!-- proof-4: body_md -->
Both expressions decrease as \(\sigma\) increases, so it suffices to take
\(\sigma=1-1/(C\log_2x)\). For sufficiently large \(x\), this value is
at least \(3/4\). Split the prime divisors of \(\mathfrak n\) at
\(\mathrm{N}(\mathfrak p)=\log x\). Below this cutoff, Taylor's formula gives
\(\mathrm{N}(\mathfrak p)^{-\sigma}
=\mathrm{N}(\mathfrak p)^{-1}
(1+\mathsf{O}_C(\log\mathrm{N}(\mathfrak p)/\log_2x))\).
Consequently, \eqref{eq:mertens log over norm} implies that
\(\sum_{\mathrm{N}(\mathfrak p)\le\log x}
(\mathrm{N}(\mathfrak p)^{-\sigma}-\mathrm{N}(\mathfrak p)^{-1})\ll_{K,C}1\).
The terms of degree at least two in the logarithms of both Euler products
have bounded sum, since \(2\sigma\ge3/2\). Therefore
\[
\prod_{\mathrm{N}(\mathfrak p)\le\log x}
(1-\mathrm{N}(\mathfrak p)^{-\sigma})^{-1}
\ll_{K,C}
\prod_{\mathrm{N}(\mathfrak p)\le\log x}
(1-\mathrm{N}(\mathfrak p)^{-1})^{-1}
\ll_K\log_2x
\]
by \eqref{eq:mertens product}.
For the remaining prime divisors, the bound
\(\sum_{\mathfrak p\mid\mathfrak n}\log\mathrm{N}(\mathfrak p)
\le\log\mathrm{N}(\mathfrak n)\le B\log x\) gives
\[
\sum_{\substack{\mathfrak p\mid\mathfrak n\\\mathrm{N}(\mathfrak p)>\log x}}
\mathrm{N}(\mathfrak p)^{-\sigma}
\le\frac{B\log x}{(\log x)^\sigma\log_2x}
=\frac{Be^{1/C}}{\log_2x}.
\]
Its Euler product is bounded, which proves
\eqref{eq:uniform mertens product}. This argument also covers ideals of
bounded norm, including \(\mathfrak n=\mathcal O_K\), without a
separate assumption on \(\log_2\mathrm{N}(\mathfrak n)\).

For \eqref{eq:uniform mertens log over norm}, the primes below the cutoff contribute
\(\ll_C\)
\(\sum_{\mathrm{N}(\mathfrak p)\le\log x}
\log\mathrm{N}(\mathfrak p)/\mathrm{N}(\mathfrak p)\ll_K\log_2x\).
Above the cutoff it is
\(\ll_C(\log x)^{-\sigma}
\sum_{\mathfrak p\mid\mathfrak n}\log\mathrm{N}(\mathfrak p)
\ll_{B,C}1\). This proves the second estimate.

For any number field \(K\), nonzero integral ideal \(\mathfrak f\), and \(x\ge1\),
\[
\begin{aligned}

		\sum_{\substack{#1}}
	}{
		\sum
	}
_{\mathrm{N}(\mathfrak a)\le x\\(\mathfrak a,\mathfrak f)=1}
\frac1{\mathrm{N}(\mathfrak a)}
&=\frac{\phi(\mathfrak f)}{\mathrm{N}(\mathfrak f)}
\left\{\gamma_{-1}(K)\log x+\gamma_0(K)
+\gamma_{-1}(K)\eta(\mathfrak f)\right\}\\
&\quad+\mathsf{O}_K\!\left(\mathcal E_2(K,\mathfrak f)
 x^{-1/[K:\mathbb Q]}\right),
\end{aligned}
\]
where
\begin{equation}\label{eq:definition of E2}
\mathcal E_2(K,\mathfrak f)
=\prod_{\mathfrak p\mid\mathfrak f}
\left(1+\mathrm{N}(\mathfrak p)^{-1+1/[K:\mathbb Q]}\right).
\end{equation}
If \([K:\mathbb Q]\ge2\), then
\(\log\mathcal E_2(K,\mathfrak f)
\ll_K \log^{[1/[K:\mathbb{Q}]}\mathrm{N}(\mathfrak f)/
\log_2 \mathrm{N}(\mathfrak f) \).

<!-- proof-7: body_md -->
The ideal density asymptotic \eqref{eq:ideal-density} reads
\(\sum_{\mathrm{N}(\mathfrak a)\le x}1=\gamma_{-1}(K)x+R_K(x)\), where
\(R_K(x)=\mathsf{O}_K(x^{1-1/[K:\mathbb Q]})\). Partial summation gives
\(\zeta_K(s)-\gamma_{-1}(K)/(s-1)
=\gamma_{-1}(K)+s\int_1^\infty R_K(t)t^{-s-1}\,\mathrm dt\).
Letting \(s\rightarrow1\), we obtain
\begin{equation}\label{eq:Stieltjes coefficients identity}
\gamma_0(K)=\gamma_{-1}(K)+\int_1^\infty\frac{R_K(t)}{t^2}\,\mathrm dt.
\end{equation}
A second partial summation therefore yields
\begin{equation}\label{eq:reciprocal sum}
\sum_{\mathrm{N}(\mathfrak a)\le x}\frac1{\mathrm{N}(\mathfrak a)}
=\gamma_{-1}(K)\log x+\gamma_0(K)
+\mathsf{O}_K(x^{-1/[K:\mathbb Q]}).
\end{equation}
This estimate also holds for \(0<x<1\), since the sum is empty and
\(1+|\log x|\ll_K x^{-1/[K:\mathbb Q]}\).

	\[
		\begin{aligned}

		\sum_{\substack{#1}}
	}{
		\sum
	}
_{\mathrm{N}(\mathfrak{a}) \le x \\ (\mathfrak{a}, \mathfrak{f}) = 1} \frac{1}{\mathrm{N}(\mathfrak{a})}
		&= 
	\IfValueTF{#1} {
		\sum_{\substack{#1}}
	}{
		\sum
	}
_{\mathrm{N}(\mathfrak{a}) \le x} \frac{1}{\mathrm{N}(\mathfrak{a})} \sum_{\mathfrak{d} \mid (\mathfrak{a}, \mathfrak{f})} \mu(\mathfrak{d}) \\
		&= \sum_{\mathfrak{d} \mid \mathfrak{f}} \frac{\mu(\mathfrak{d})}{\mathrm{N}(\mathfrak{d})} \sum_{\mathrm{N}(\mathfrak{a}) \le x/\mathrm{N}(\mathfrak{d})} \frac{1}{\mathrm{N}(\mathfrak{a})}.
		\end{aligned}

		\sum

        &= (\gamma_{-1}(K) \log x + \gamma_0(K))
        \sum_{\mathfrak{d} \mid \mathfrak{f}} \frac{\mu(\mathfrak{d})}{\mathrm{N}(\mathfrak{d})}
        - \gamma_{-1}(K)\sum_{\mathfrak{d} \mid \mathfrak{f}}
        \frac{\mu(\mathfrak{d})}{\mathrm{N}(\mathfrak{d})} \log \mathrm{N}(\mathfrak{d}) \\

\label{lem:dirichlet-factorization}
	Let \( \mathfrak{f} \) be an integral ideal of \( K \). Let \( g \) be a multiplicative function on integral ideals, supported on squarefree ideals coprime to \( \mathfrak{f} \). 	\[
		H(s;g) \coloneqq \zeta_{K,\mathfrak{f}}(s)^{-1} \prod_{\mathfrak{p} \nmid \mathfrak{f}} \biggl( 1 + \frac{g(\mathfrak{p})}{\mathrm{N}(\mathfrak{p})^s}\biggr),
	\]
	initially defined on \( \Re s > 1 \), has Dirichlet series \[ H(s;g) = \sum_\mathfrak{m} h(\mathfrak{m})/\mathrm{N}(\mathfrak{m})^s \] which converges absolutely for \( \Re s > \max(1-\delta,1/2) \).
    Moreover, \( g \) is equal to the Dirichlet convolution
    \[
    g(\mathfrak{l}) = 
	\IfValueTF{#1} {
		\sum_{\substack{#1}}
	}{
		\sum
	}
_{\mathfrak{am} = \mathfrak{l} \\ (\mathfrak{a}, \mathfrak{f}) = 1}h(\mathfrak{m}). \]

The local factor in the Euler product of \( H(s;g) \) expands as
	\[
		\left(1 + \frac{g(\mathfrak{p})}{\mathrm{N}(\mathfrak{p})^s}\right)\!\left(1 - \frac{1}{\mathrm{N}(\mathfrak{p})^s}\right) = 1 + \frac{g(\mathfrak{p}) - 1}{\mathrm{N}(\mathfrak{p})^s} - \frac{g(\mathfrak{p})}{\mathrm{N}(\mathfrak{p})^{2s}}.
	\]
	The deviation of this local factor
	from \( 1 \) is \( \ll \mathrm{N}(\mathfrak{p})^{-\Re s -\delta} + \mathrm{N}(\mathfrak{p})^{-2\Re s} \) , giving the domain of absolute convergence \( \Re s >
	\max(1-\delta, 1/2) \) as claimed. The formula for \( g \) is immediate.

\label{lem:weighted reciprocal mean value}

for some \(\delta>0\). For every fixed
\(0<\kappa<\min(1/[K:\mathbb Q],\delta,1/2)\) and \(x\ge1\),
\[

=C_{K,\mathfrak f,\mathfrak g}\log x+D_{K,\mathfrak f,\mathfrak g}
+\mathsf{O}\!\left(\mathcal E_2(K,\mathfrak f)x^{-\kappa}

where
\[
\begin{aligned}
C_{K,\mathfrak f,\mathfrak g}
&=\frac{\phi(\mathfrak f)}{\mathrm{N}(\mathfrak f)}\gamma_{-1}(K)H(1;g),\\
D_{K,\mathfrak f,\mathfrak g}
&=\frac{\phi(\mathfrak f)}{\mathrm{N}(\mathfrak f)}
\left[\gamma_{-1}(K)H'(1;g)
+(\gamma_0(K)+\gamma_{-1}(K)\eta(\mathfrak f))H(1;g)\right].
\end{aligned}
\]
Here \(\mathcal E_2\) is defined in \eqref{eq:definition of E2}.
The implied constant depends only on \(K,\delta,\kappa\) and the uniform
constant in \(|g(\mathfrak p)-1|\ll\mathrm{N}(\mathfrak p)^{-\delta}\).

\(H(1;g),H'(1;g)\ll1\). Hence
\(C_{K,\mathfrak f,\mathfrak g}\ll1\).
We also have \(D_{K,\mathfrak f,\mathfrak g}\ll1+\eta(\mathfrak f)\),

\label{lem:lattice-disk-count}

&\le C_1 \biggl( 1+\frac{X}{\lambda_1^2} \biggr), \label{eq:lattice bound with origin} \\
\#\{0\ne w\in\Lambda:|w|^2\le X\}
&\le C_2 \frac{X}{\lambda_1^2}. \label{eq:lattice bound without origin}
\end{align}

condition, so we may assume \(X\ge Y\).
For each pair counted, choose a generator
\(\mathfrak J_1\mathfrak J_2^{-1}=(\alpha)\) with
\(\alpha\equiv1\pmod{\mathfrak t}\).
The element \(\alpha\) may be fractional, and
\(\alpha-1\ne0\) because \(\mathfrak J_1\mathfrak J_2^{-1}\ne\mathcal O_K\).

\mathfrak J_1+\mathfrak J_2\subseteq\mathcal O_K\).

\mathrm{N}(\alpha-1)
\le\left(\sqrt{\frac{\mathrm{N}(\mathfrak J_1)}{\mathrm{N}(\mathfrak J_2)}}+1\right)^2
\le\frac{4X}{\mathrm{N}(\mathfrak J_2)}.
\]
For each fixed \(\mathfrak J_2\), distinct \(\mathfrak J_1\) give distinct
chosen elements \(\alpha-1\). The nonzero lattice-point estimate in
Corollary~\ref{lem:ideal-coset-lattice-count} therefore bounds their number by
\[
\#\left\{0\ne\alpha-1\in\mathfrak t\mathfrak J_2^{-1}:
\mathrm{N}(\alpha-1)\le\frac{4X}{\mathrm{N}(\mathfrak J_2)}\right\}
\ll_K\frac{X/\mathrm{N}(\mathfrak J_2)}{\mathrm{N}(\mathfrak t\mathfrak J_2^{-1})}
=\frac{X}{\mathrm{N}(\mathfrak t)}.
\]
There are \(\ll_KY\) choices of \(\mathfrak J_2\) by
Lemma~\ref{lem:ideal-density-mertens}, proving
\eqref{eq:ideal-pair-lattice-count}.

For \eqref{eq:ideal-pair-lattice-weighted}, divide both ideal norms into
dyadic intervals starting at \(1\).
The block \(2^i\le\mathrm{N}(\mathfrak J_1)<2^{i+1}\),
\(2^j\le\mathrm{N}(\mathfrak J_2)<2^{j+1}\), contributes at most
\[
\ll_{W,A,\eta}\frac{2^{(1/2+\eta)(i+j)}}{\mathrm{N}(\mathfrak t)}
\min\left(1,\left(\frac{2^{i+j}}{X^2}\right)^{-A}\right)
\]
by \eqref{eq:ideal-pair-lattice-count} and the decay of \(W\).
For each \(k=i+j\) there are \(k+1\) blocks, so their total is bounded by
\[
\frac1{\mathrm{N}(\mathfrak t)}
\sum_{k\ge0}(k+1)2^{(1/2+\eta)k}
\min\left(1,\left(\frac{2^k}{X^2}\right)^{-A}\right).
\]
We obtain the desired bound by by splitting at \(2^k=X^2\) and choosing \(A>1/2+\eta\).

<!-- lemma-17: body_md -->
\label{lem:gauss-orthogonality}
Let \(\tau(\chi)\) be the Gauss sum occurring in the functional
equation of Lemma~\ref{thm:functional-equation}.
Then, for every nonzero integral ideal \(\mathfrak m\) satisfying
\((\mathfrak m,\mathfrak f)=(1)\),
\[
\begin{aligned}
\sum_{\chi\bmod\mathfrak f}^{*}
\chi(\mathfrak m)\tau(\chi)
&=
\sum_{\mathfrak s\mathfrak t=\mathfrak f}
\mu(\mathfrak s)h_K(\mathfrak t)
\sum_{\substack{\alpha\bmod\mathfrak f\\
(\alpha,\mathfrak f)=(1)\\
\mathfrak m\mathfrak e(\alpha)\in P_K(\mathfrak t)}}
\exp\!\left(
2\pi i\operatorname{Tr}(\alpha\beta)
\right).
\end{aligned}
\]
In particular, if \(\mathfrak m\mathfrak e\) is not principal, then
the right-hand side vanishes.
<!-- end lemma-17 -->
<!-- proof-15: body_md -->
Expanding \(\tau(\chi)\) and interchanging the finite sums, the left-hand
side becomes
\[
\sum_{\substack{\alpha\bmod\mathfrak f\\
(\alpha,\mathfrak f)=(1)}}
e^{2\pi i\operatorname{Tr}(\alpha\beta)}
\sum_{\chi\bmod\mathfrak f}^{*}
\chi\!\left(\mathfrak m\mathfrak e(\alpha)\right).
\]

Fix \(\alpha\).  The fractional ideal
\(\mathfrak m\mathfrak e(\alpha)\) is coprime to \(\mathfrak f\).
By clearing its denominator away from \(\mathfrak f\) and applying the
Chinese remainder theorem, we may choose
\(\gamma\in\mathcal O_K\) such that
\(\gamma\equiv1\pmod{\mathfrak f}\) and
\((\gamma)\mathfrak m\mathfrak e(\alpha)\) is an integral ideal coprime
to \(\mathfrak f\).

Since
\((\gamma)\in P_K(\mathfrak f)\subseteq P_K(\mathfrak t)\) for every
\(\mathfrak t\mid\mathfrak f\), multiplication by \((\gamma)\) changes
neither the value of a character modulo \(\mathfrak f\) nor membership
in \(P_K(\mathfrak t)\).  Hence
Lemma~\ref{lem:primitive-ray-orthogonality}, applied to
\((\gamma)\mathfrak m\mathfrak e(\alpha)\), gives
\[
\sum_{\chi\bmod\mathfrak f}^{*}
\chi\!\left(\mathfrak m\mathfrak e(\alpha)\right)
=
\sum_{\substack{\mathfrak s\mathfrak t=\mathfrak f\\
\mathfrak m\mathfrak e(\alpha)\in P_K(\mathfrak t)}}
\mu(\mathfrak s)h_K(\mathfrak t).
\]
Substitution proves the asserted identity.

Finally, if
\(\mathfrak m\mathfrak e(\alpha)\in P_K(\mathfrak t)\), then
\(\mathfrak m\mathfrak e(\alpha)\) is principal, and therefore so is
\(\mathfrak m\mathfrak e\), since \((\alpha)\) is principal.  Thus, if
\(\mathfrak m\mathfrak e\) is not principal, no such \(\alpha\) exists
and the right-hand side vanishes.

<!-- lemma-18: body_md -->
\label{lem:gauss-sum-bound}
Under the assumptions of Lemma~\ref{lem:gauss-orthogonality}, for every
nonzero integral ideal \(\mathfrak m\) satisfying
\((\mathfrak m,\mathfrak f)=(1)\),
\[
\left|
\sum_{\chi\bmod\mathfrak f}^{*}
\chi(\mathfrak m)\tau(\chi)
\right|
\le
|\mathcal O_K^\times|
\sum_{\mathfrak t\mid\mathfrak f}
h_K(\mathfrak t)
\frac{\mathrm{N}(\mathfrak f)}{\mathrm{N}(\mathfrak t)}
\ll_K
d(\mathfrak f)\mathrm{N}(\mathfrak f),
\]
where
 \(d(\mathfrak f)\) denotes the number of integral ideal divisors
of \(\mathfrak f\).
<!-- end lemma-18 -->
<!-- proof-16: body_md -->
By Lemma~\ref{lem:gauss-orthogonality}, the triangle inequality, and
\(|\mu(\mathfrak s)|\le1\), the left-hand side is bounded by
\[
\sum_{\mathfrak t\mid\mathfrak f}
h_K(\mathfrak t)
\#
\left\{
\alpha\bmod\mathfrak f:
(\alpha,\mathfrak f)=(1),\
\mathfrak m\mathfrak e(\alpha)\in P_K(\mathfrak t)
\right\}.
\]

If \(\mathfrak m\mathfrak e\) is not principal, all the sets in
the preceding display are empty by
Lemma~\ref{lem:gauss-orthogonality}.  We may therefore assume that
\(\mathfrak m\mathfrak e=(\xi)\) for some \(\xi\in K^\times\).

Since
\(
(\mathfrak m\mathfrak e,\mathfrak f)=(1),
\)
we have
\(
v_{\mathfrak p}(\xi)=0,
(\mathfrak p\mid\mathfrak f).
\)
Thus \(\xi\) is invertible modulo every
\(\mathfrak t\mid\mathfrak f\). The condition
\(
\mathfrak m\mathfrak e(\alpha)
=
(\xi\alpha)
\in P_K(\mathfrak t)
\)
is equivalent to the existence of a unit
\(u\in\mathcal O_K^\times\) such that
\(\xi\alpha\equiv u\pmod{\mathfrak t}\).
Indeed, if \((\xi\alpha)\in P_K(\mathfrak t)\), then there exists
\(\lambda\equiv1\pmod{\mathfrak t}\) satisfying
\(
(\lambda)=(\xi\alpha).
\)
Hence
\(
\lambda=u\xi\alpha
\)
for some \(u\in\mathcal O_K^\times\), and therefore
\(
\xi\alpha\equiv u^{-1}\pmod{\mathfrak t}.
\)
Since inverses of units are again units, this is equivalent to
the preceding congruence.  The converse follows in the same way.

For each fixed unit \(u\), since \(\xi\) is invertible modulo
\(\mathfrak t\), this congruence determines the single residue class
\(\alpha\equiv\xi^{-1}u\pmod{\mathfrak t}\).
The natural reduction map
\(\mathcal O_K/\mathfrak f\longrightarrow\mathcal O_K/\mathfrak t\)
has fibers of cardinality
\(
\frac{\mathrm{N}(\mathfrak f)}{\mathrm{N}(\mathfrak t)}.
\)
Restricting to residue classes coprime to \(\mathfrak f\) can only
decrease the number of lifts.  Hence
\[
\#
\left\{
\alpha\bmod\mathfrak f:
(\alpha,\mathfrak f)=(1),\
\mathfrak m\mathfrak e(\alpha)\in P_K(\mathfrak t)
\right\}
\le
|\mathcal{O}_{K}^{\times}|
\frac{\mathrm{N}(\mathfrak f)}{\mathrm{N}(\mathfrak t)}.
\]
Substituting this into the earlier absolute-value bound gives
\[
\left|
\sum_{\chi\bmod\mathfrak f}^{*}
\chi(\mathfrak m)\tau(\chi)
\right|
\le
|\mathcal O_K^\times|
\sum_{\mathfrak t\mid\mathfrak f}
h_K(\mathfrak t)
\frac{\mathrm{N}(\mathfrak f)}{\mathrm{N}(\mathfrak t)}.
\]

Finally, the cardinality formula in
Lemma~\ref{lem:ray-class-exact-sequence} gives
\[
h_K(\mathfrak t)
=
h_K
\frac{\phi(\mathfrak t)}
{[\mathcal O_K^\times:
  \mathcal O_K^\times\cap K^{\mathfrak t,1}]}
\le
h_K\phi(\mathfrak t)
\le
h_K\mathrm{N}(\mathfrak t).
\]
Therefore
\[
\begin{aligned}
|\mathcal O_K^\times|
\sum_{\mathfrak t\mid\mathfrak f}
h_K(\mathfrak t)
\frac{\mathrm{N}(\mathfrak f)}{\mathrm{N}(\mathfrak t)}
&\le
|\mathcal O_K^\times|h_K\mathrm{N}(\mathfrak f)
\sum_{\mathfrak t\mid\mathfrak f}1 \\
&=
|\mathcal O_K^\times|h_Kd(\mathfrak f)\mathrm{N}(\mathfrak f) \\
&\ll_K
d(\mathfrak f)\mathrm{N}(\mathfrak f).
\end{aligned}
\]
This proves the lemma.

<!-- numberedcorollary-3: body_md -->
\label{cor:gauss-sum-bound-fractional}
For every fractional ideal
\(\mathfrak b\in I_K(\mathfrak f)\),
\[
\left|
\sum_{\chi\bmod\mathfrak f}^{*}
\chi(\mathfrak b)\tau(\chi)
\right|
\le
|\mathcal O_K^\times|
\sum_{\mathfrak t\mid\mathfrak f}
h_K(\mathfrak t)
\frac{\mathrm{N}(\mathfrak f)}{\mathrm{N}(\mathfrak t)}
\ll_K
d(\mathfrak f)\mathrm{N}(\mathfrak f).
\]
<!-- end numberedcorollary-3 -->
<!-- proof-17: body_md -->
Choose an integral ideal \(\mathfrak d\), coprime to
\(\mathfrak f\), such that \(\mathfrak d\mathfrak b\) is integral.
By the Chinese remainder theorem, choose
\(\gamma\in\mathfrak d\) with
\(\gamma\equiv1\pmod{\mathfrak f}\). Then
\(\mathfrak n:=(\gamma)\mathfrak b\) is an integral ideal coprime to
\(\mathfrak f\). Since \((\gamma)\in P_K(\mathfrak f)\), one has
\(\chi(\mathfrak n)=\chi(\mathfrak b)\)
for every character modulo \(\mathfrak f\). The result now follows by
applying Lemma~\ref{lem:gauss-sum-bound} to \(\mathfrak n\).
