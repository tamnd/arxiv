<!-- lemma-1: body_md -->
\label{DirichletLaplace}
It holds true that
$
\mathcal{E}_\Lambda(u,v)=\langle u,\Delta_wv\rangle_{L^2}
$
for $u,v\in L^2(X,\mu_\Lambda)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The proof extends from the discrete setting
in \cite[Proposition 1.12]{JP2013} as follows:
Let $u,v\in\operatorname{dom}(\mathcal{E}_\Lambda)$. Then
\begin{align*}
\langle u,\Delta_w v\rangle_{L^2}&=\frac12\int_X\overline{u(x)}\Delta_wv(x)\,d\mu_\Lambda(x)+\frac12\int_X\overline{u(y)}\Delta_wv(y)\,d\mu_\Lambda(y)
\\
&=\frac12\int_X\int_X\overline{u(x)}w(x,y)(v(x)-v(y))\,d\mu_\Lambda(y)\,d\mu_\Lambda(x)
\\
&-\frac12\int_X\int_X\overline{u(y)}w(x,y)(v(x)-v(y))\,d\mu_\Lambda(x)\,d\mu_\Lambda(y)
\\
&=\frac12\int_X\int_Xw(x,y)\left(\overline{u(x)}-\overline{u(y)}\right)
(v(x)-v(y))\,d\mu_\Lambda(y)\,d\mu_\Lambda(x)
\\
&=\mathcal{E}_\Lambda(u,v)\,,

<!-- definition-1: body_md -->
Let $\phi\in L^2(\delta_w\Omega,\mu_\Lambda)$. A function 
$u\in\operatorname{dom}(\mathcal{E}_{\operatorname{cl}_w\Omega})$ solves
the \emph{Neumann Boundary Value Problem} for $(\phi,\Omega)$, if
\begin{align}\label{NBVP}
\Delta_\Omega u(x) = 0\quad\text{and}\quad N_{\delta_w\Omega} u(y) = \phi(y)
\end{align}
for $x\in\Omega$ and $y\in\delta_w\Omega$. It is a \emph{weak solution} to the Neumann Boundary Value
Problem for $(\phi,\Omega)$, if
\begin{align}\label{NBVP_w}
\langle\Delta_\Omega u, v\rangle_{L^2} = 0\quad\text{and}\quad
\langle N_{\delta_w\Omega}u,v\rangle_{L^2} = \langle\phi,v\rangle_{L^2}
\end{align}
for all $v\in\operatorname{dom}(\mathcal{E}_{\operatorname{cl}_w\Omega})$ having support in $\Omega$ and $\delta_w\Omega$, respectively.

Let $u\in \operatorname{dom}(\mathcal{E}_{\operatorname{cl}_w\Omega})$ be a solution (weak or not) of the Neumann boundary value problem for $(\phi,\Omega)$.
Using the $p$-adic Gauss-Green Identity (Lemma \ref{GaussGreen_p}), it follows that
\begin{align*}
\mathcal{E}_{\Omega,\operatorname{cl}_w\Omega}(1_{\operatorname{cl}_w\Omega},u)
&=\langle 1_{\operatorname{cl}_w\Omega}, \Delta_\Omega u\rangle_{L^2(\Omega,\mu_\Lambda)}
+\langle 1_{\operatorname{cl}_w\Omega},N_{\delta_w\Omega} u\rangle_{L^2(\delta_w\Omega,\mu_\Lambda)}
\\
&=\langle 1_{\operatorname{cl}_w\Omega},N_{\delta_w\Omega} u\rangle_{L^2(\delta_w\Omega,\mu_\Lambda)}
\\
&=\langle 1_{\operatorname{cl}_w\Omega},\phi\rangle_{L^2(\delta_w\Omega,\mu_\Lambda)}
\\
&=\int_{\delta_w\Omega}\phi(x)\,d\mu_\Lambda(x)\,.

\[

the assertion now follows.

<!-- cor-1: body_md -->
\label{Existence_NBVP}
Let $\phi\in L^2(\delta_w\Omega)$ such that
\[
\int_{\delta_w\Omega}\phi(x)\,d\mu_\Lambda(x)= 0\,.
\]
Then $u\in L^\infty(\operatorname{cl}_w\Omega)$ is a weak solution of the Neumann Boundary Value Problem for $(\phi,\Omega)$, i.e.\ a solution of (\ref{NBVP_w}),  if
\[
\langle u,\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}=
\int_{\delta_w\Omega}\phi(x)v(x)\,d\mu_\Lambda(x)
\]
for every $v \in\mathcal{N}$.
<!-- end cor-1 -->
<!-- proof-7: body_md -->
We just follow the proof of \cite[Lemma 3.8]{HS2022}: First, $u\in\operatorname{dom}(\mathcal{E}_{\operatorname{cl}_w\Omega})$.
Namely, for every $\alpha>0$, we have that
\[
\Delta_\Omega\alpha(\Delta_\Omega + \alpha)^{-1} u 
=\alpha u - \alpha^2(\Delta_\Omega + \alpha)^{-1} u\in L^\infty(\operatorname{cl}_w\Omega)\,,
\]
which shows that $\alpha(\Delta_\Omega+\alpha)^{-1}u \in\mathcal{N}$. Hence, from the hypothesis on $u$, we
have
\begin{align*}
\int_{\delta_W\Omega}\phi\alpha(\Delta_\Omega+\alpha)^{-1}u\,d\mu_\Lambda
&=\langle u,\Delta_\Omega\alpha(\Delta_\Omega+\alpha)^{-1}u\rangle_{L^2}
\\
&=\alpha\langle u,u-\alpha(\Delta_\Omega+\alpha)^{-1}u\rangle_{L^2}
\end{align*}
for $\alpha>0$. Since $\mathcal{E}_{\operatorname{cl}_w\Omega}$ being Markov is equivalent with $\alpha(\Delta_\Omega+\alpha)^{-1}$ being
a contraction, cf.\ \cite[Theorem 1.4.1]{Fukushima1980}, this means that
\[
\int_{\delta_w\Omega}\phi\alpha(\Delta_\Omega+\alpha)^{-1}u\,d\mu_\Lambda
\le\mu_\Lambda(\delta_w\Omega)\left\|\phi\right\|_\infty\left\|u\right\|_\infty\,,
\]
and  thus
\[
\lim\limits_{\alpha\to\infty}\alpha
\langle u,u-\alpha(\Delta_\Omega+\alpha)^{-1}u\rangle_{L^2}<\infty\,.
\]
From the equality
\[
\operatorname{dom}(\mathcal{E}) =
{\left\{u\in L^2(\operatorname{cl}_w\Omega\mid \lim\limits_\alpha\langle u, u\rangle_{\mathcal{E}} < \infty\right\}}
\]
in \cite[Lemma 1.3.4]{Fukushima1980}, it now follows that $u\in\operatorname{dom}(\mathcal{E}_{\operatorname{cl}_w\Omega})$. By the assumption,
one has
\begin{align}\label{Dirichlet_E_int}
\mathcal{E}_{\operatorname{cl}_w\Omega}(u,v)&=
\langle u,\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}=\int_{\delta_w\Omega}\phi(x)v(x)\,d\mu_\Lambda(x)
\end{align}
for $v\in\mathcal{N}$. For general $v\in\operatorname{dom}(\mathcal{E}_{\operatorname{cl}_w\Omega})$ let $(v_n)_n\in\mathcal{N}$ such that
\[
\lim\limits_n\left\|v_n -v\right\|_{\mathcal{E}_{\operatorname{cl}_w\Omega}} = 0\,.
\]
Then
\[
\lim\limits_{n}\mathcal{E}_{\operatorname{cl}_w\Omega}(u,v_n) 
=\mathcal{E}_{\operatorname{cl}_w\Omega}(u, v)\,,
\]
and since $\phi\in L^\infty(\operatorname{cl}_w\Omega,\mu_\Lambda)$, also
\[
\lim\limits_n\int_{\delta_w\Omega} \left\lvert\phi v_n-\phi v\right\rvert\,d\mu_\Lambda=0\,,
\]
which shows that (\ref{Dirichlet_E_int}) holds for all $v\in\operatorname{dom}(\mathcal{E}_{\operatorname{cl}_w\Omega})$. This now means that if $\operatorname{supp}(v)\subseteq\Omega$, then
\[
\langle \Delta_\Omega u,v\rangle_{L^2}=\langle u,\Delta_\Omega v\rangle_{L^2} = 0\,,
\]
and if $\operatorname{supp}(v)\subseteq\delta_w\Omega$, then
\[
\langle N_{\delta_w\Omega} u,v\rangle_{L^2}=\langle u,\Delta_\Omega v\rangle_{L^2} = \langle\phi,v\rangle_{L^2},
\]
i.e.\ $u$ is a weak solution of the Neumann Boundary Value Problem.

\label{ultracontractive}
Let $E$ be a locally compact separable metric space, and $m$ a
non-negative Radon measure on $E$ with full support. A semigroup $e^{-tL}$ is \emph{ultracontractive}, if there exists a decreasing function 
$\gamma\colon \mathds{R}_{>0}\to\mathds{R}_{>0}$ such that for
every $t > 0$ and $u\in L^2(E,m)$ we have
\[
\left\|e^{-tL} u\right\|_{L^\infty(E,m)}
\le\gamma(t)\left\|u\right\|_{L^2(E,m)}
\]
where the function $\gamma$ is called the \emph{rate function} of the semigroup.
<!-- end definition-2 -->

The process $L = (L_t)_{t\ge0}$ with
\[
L_t =\int_0^1 1_{{\left\{Y_s\in\delta_w\Omega\right\}}}\, ds\,,
\]
where $t\ge 0$, is called the \emph{local time} of the process $Y$ on the boundary $\delta_w\Omega$.

<!-- lemma-6: body_md -->
\label{solutionFromheatKernel} 
Assume that the heat kernel $p_t(x,y)$ associated with the semigroup $e^{-t\Delta_\Omega}$ exists. Let $\phi\in L^\infty(\delta_w\Omega)$. For every $t\ge 0$, the function
\[
u_t(x) = \mathds{E}_x\int_0^t
\phi(Y_s)\, dL_s
\]
is bounded, and
\[
u_t(x) = \int_{\delta_w\Omega}\int_0^t
\phi(y)p_s(x,)\,ds\, d\mu_\Lambda(y)
\]
for $x\in\operatorname{cl}_w\Omega$ holds true.
<!-- end lemma-6 -->
<!-- proof-8: body_md -->
Inspired by the proof of \cite[Lemma 3.10]{HS2022}, we see that
\begin{align*}
u_t(x)&=\mathds{E}_x\int_0^t\phi(Y_s)(L_s)'\,ds=\mathds{E}_x\int_0^t\phi(Y_s)1_{Y_s\in\delta_w\Omega}\,ds
\\
&=\int_{\delta_w\Omega}\mathds{E}_x\phi(y)\delta_y(Y_s)\,ds\,d\mu_\Lambda(y)
\\
&=\int_{\delta_w\Omega}\phi(y)\mathds{E}_x\int_0^t\delta_y(Y_s)\,ds\,d\mu_\Lambda(y)
\\
&\stackrel{(*)}{=}\int_{\delta_w\Omega}\phi(y)\int_0^t\mathds{E}_x\delta_y(Y_s)\,ds\,d\mu_\Lambda(y)
\\
&=\int_{\delta_w\Omega}\phi(y)\int_0^t\delta_y(z)p_s(x,d\mu_\Lambda(z))\,ds\,d\mu_\Lambda(y)
\\
&=\int_{\delta_w\Omega}\phi(y)\int_0^t p_s(x,y)\,ds\,d\mu_\Lambda(y)
\\
&=\int_{\delta_w\Omega}\int_0^t\phi(y)p_s(x,y)\,ds\,d\mu_\Lambda(y)
\end{align*}
for $x \in\operatorname{cl}_w\Omega$, which admits the bound
\[
\left\lvertu_t\right\rvert\le t\left\|\phi\right\|_\infty
\]
for $t\ge0$. The first equality holds true because $L_s$ is almost everywhere
differentiable on the half-line $s\ge 0$, cf.\ (\ref{localTimeDer}). Equality $(*)$ follows from
Fubini-Tonelli.

From \cite[Appendix B]{HS2022}, we take the following information:
Let $E$ be a locally compact separable metric space, and $m$ a non-negative
Radon measure on $E$ with full support.
\begin{enumerate}
\item If $(e^{-tL})_{t>0}$ is ultracontractive with rate function $\gamma$ and ${\left\{p_t(\cdot,\cdot)\right\}}_{t>0}$ is a heat

\begin{align}\label{ultracontractive_ineq}
\left\lvertp_t(x,y) -\frac{1}{m(E)}
\right\rvert
\le c_1 e^{-c_2 t}
\end{align}
holds for every $t > t_0$ and $m$-a.e.\ $x,y\in E$, cf.\ \cite[Eq. (18)]{HS2022}.
\end{enumerate}

<!-- thm-1: body_md -->
\label{SolutionOfNBVP_p}
 Let $X$ be a compact $p$-adic analytic $n$-manifold, and $\Delta_w$ an
integral operator with kernel function $w(x,y)$, which is the generator of an ultracontractive semigroup on a connected subdomain $\operatorname{cl}_w\Omega X$ with $\Omega\subseteq X$
open. 
Let $\phi\in L^\infty(\delta_w\Omega)$ such that
\[
\int_{\delta_w\Omega}
\phi\, d\mu_\Lambda = 0\,.
\]
Then the following statements hold true:
\begin{enumerate}
\item There is a unique function $u \in L^\infty(\operatorname{cl}_w\Omega)$ that weakly solves the Neumann Boundary Value Problem for $(\phi,\Omega)$, and satisfies
\[
\int_{\operatorname{cl}_w\Omega}u(x)\,d\mu_\Lambda(x) = 0 .
\]
\item This function satisfies
\begin{align*}
u(x)&=\lim\limits_{t\to\infty}\mathds{E}_x\int_0^t(Y_s)\,dL_s
\\
&=\int_0^\infty\int_{\delta_w\Omega}\phi(y)p_s(x,y)\,d\mu_\Lambda(y)\,ds
\\
&=\int_{\delta_w\Omega}\phi(y)G_\Omega(x,y)\,d\mu_\Lambda(y)
\end{align*}
for all $x\in\operatorname{cl}_w\Omega$.
\end{enumerate}

<!-- proof-9: body_md -->
We just follow the lines of the proof of \cite[Theorem 3.12]{HS2022}.

\smallskip
We first show that $u(x) := \lim\limits_{t\to\infty} u_t (x)$ exists. By Lemma \ref{solutionFromheatKernel}, and since
$\phi$ is centered,
\begin{align*}
u_t(x)&=\int_{\delta_w\Omega}\int_0^t\phi(y)p_s(x,y)\,ds\,d\mu_\Lambda(y)
\\
&=\int_{\delta_w\Omega}\int_0^t\phi(y)\left(p_s(x,y)-\frac{1}{\mu_\Lambda(\operatorname{cl}_w\Omega)}\right)\,ds\,d\mu_\Lambda(y)\,,
\end{align*}
and given $0 < s < t$, we can see that
\begin{align*}
\left\lvertu_t(x)-u_s(x)\right\rvert&\le\int_{\delta_w\Omega}\int_s^t\left\lvert\phi(y)\right\rvert\left\lvertp_r(x,y)-\frac{1}{\mu_\Lambda(\operatorname{cl}_w\Omega)}\right\rvert\,dr\,d\mu_\Lambda(y)
\\
&\le c_1\int_{\delta_w\Omega}\left\lvert\phi(y)\right\rvert\,d\mu_\Lambda(y)\int_s^te^{-c_2r}\,dr\,,
\end{align*}
using (\ref{ultracontractive_ineq}).
This expression becomes arbitrarily small for $s, t$ are large. Hence,
the limit function $u$ exists, is bounded, and satisfies
\[
u(t)=\int_0^\infty\int_{\delta_w\Omega}\phi(y)p_s(x,y)\,d\mu_\Lambda(y)\,ds
\]
for $x\in\operatorname{cl}_w\Omega$. By the expression (\ref{GreenFunctionOmega}) for the Green function and since $\phi$ is
centred, it follows that
\[
u(x)=\int_{\delta_w\Omega}\phi(y)G_\Omega(x,y)\,d\mu_\Lambda(y)
\]
or every $x\in\operatorname{cl}_w\Omega$. Integrating this yields
\[
\int_{\operatorname{cl}_w\Omega}u(x)\,d\mu_\Lambda(x)=\int_0^\infty\int_{\delta_w\Omega}\phi(y)\int_{\operatorname{cl}_w\Omega}p_s(x,y)\,d\mu_\Lambda(x)\,d\mu_\Lambda(y)\,ds\,,
\]
and so
\[
\int_{\operatorname{cl}_w\Omega}u(x)\,d\mu_\Lambda(x)=0\,,
\]
since $\phi$ is centred, and
\[
\int_{\operatorname{cl}_w\Omega}p_s(x,y)\,d\mu_\Lambda(y)=1\,,
\]
which can be used since $p_s(x, y)$ is symmetric in $x,y\in\operatorname{cl}\Omega$, cf.\ (\ref{hk_symmetric}). We show that
$u$ weakly solves the Neumann problem by an application of Corollary \ref{Existence_NBVP}.
Let $v\in\mathcal{N}$ be arbitrary. We have to show
\[
\langle u,\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}=\int_{\delta_w\Omega}\phi(x)v(x)\,d\mu_\Lambda(x)\,.
\]
The uniform convergence of $u_t$ to $u$ yields
\[
\langle u,\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega)}=\lim\limits_{t\to\infty}\langle u_t,\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}\,.
\]
On the other hand,
\begin{align*}
\langle u_t,\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}&=
\int_{\delta_w\Omega}\int_0^t\phi(y)\int_{\operatorname{cl}_w\Omega}p_s(x,y)\Delta_\Omega v(x)\,d\mu_\Lambda(x)\,d\mu_\Lambda(y)\,ds
\\
&=\int_{\delta_w\Omega}\int_0^t\phi(y)\langle p_s(\cdot,y),\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}\,d\mu_\Lambda(y)\,ds\,.
\end{align*}
Using of the self-adjointness of $\Delta_\Omega$ and the heat equation (\ref{heatEquation}), obtain
\[
\langle p_s(\cdot,y),\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}=-\left\langle
\frac{d}{dt}p_t(\cdot,y)|_{t=s}
\right\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}\,,
\]
and  therefore
\begin{align*}
\langle p_s&(\cdot,y),\Delta_\Omega v\rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}
\\
&=-\int_{\delta_w\Omega}\int_0^t\int_{\operatorname{cl}_w\Omega}\phi(y)\left(\frac{d}{dt}p_t(\cdot,y)|_{t=s}\right)(x)v(x)\,d\mu_\Lambda(x)\,d\mu_\Lambda(y)\,ds
\\
&=\int_{\delta_w\Omega}\int_{\operatorname{cl}_w\Omega}\phi(y)(p_0(x,y)-p_t(x,y))v(x)\,d\mu_\Lambda(x)\,d\mu_\Lambda(y)\,.
\end{align*}
By Lebesgue’s theorem and (\ref{ultracontractive_ineq}), it now follows that
\begin{align*}
\lim\limits_{t\to\infty}&\int_{\delta_w\Omega}\int_{\operatorname{cl}_w\Omega}\phi(y)p_t(x,y)v(x)\,d\mu_\Lambda(x)\,d\mu_\Lambda(y)
\\
&=\int_{\delta_w\Omega}\phi(y)\,d\mu_\Lambda(y)\frac{1}{\mu_\Lambda(\operatorname{cl}_w\Omega)}\int_{\operatorname{cl}_w\Omega}v(x)\,d\mu_\Lambda(x)\,,
\end{align*}
so that
\begin{align*}
\langle u,\Delta_\Omega v \rangle_{L^2(\operatorname{cl}_w\Omega,\mu_\Lambda)}&=\int_{\delta_w\Omega}\int_{\operatorname{cl}_w\Omega}\phi(y)p_0(x,y)v(x)\,d\mu_\Lambda(x)\,d\mu_\Lambda(x)
\\
&=\int_{\delta_w\Omega}\phi(y)v(y)\,d\mu_\Lambda(y)\,.
\end{align*}
The result now follows from Corollary \ref{Existence_NBVP}.

<!-- proposition-2: body_md -->
\label{smallWaveletEigenvalue}
The wavelets $\psi$ on $\operatorname{cl}_w\Omega$ supported in some sufficiently small
$B(a)\subset X\setminus V(f)$ are eigenfunctions of $\Delta_{f,\Omega}^\alpha$ with eigenvalue
\[
\lambda_\psi=\int_{\operatorname{cl}_w\Omega\setminus B(a)}d_\Lambda(x,y)^{-\alpha}\left\lvertf(y)\right\rvert\,d\mu_\Lambda(y)
+\nu_f(B(a))^{-n\alpha}\left(1-q^{-n}(1+(-1)^n\right)
\]
and is independent of $x\in B(a)$.
<!-- end proposition-2 -->
<!-- proof-10: body_md -->
Since $B(a)\cap V(f)=\emptyset$ and $B(a)$ is small, the important vanishing integral property
\[
\int_{\operatorname{cl}_w\Omega}\psi(x)\,d\nu(x)=0
\]
holds true, which can be seen as follows: locally, on a chart $U\subset \operatorname{cl}_w\Omega$ containing $B(a)$, we have
\begin{align}\label{polynomialMeasure}
d\nu(x)=\left\lvertg_U(x)\right\rvert\left\lvertf(x)\right\rvert\left\lvertdx\right\rvert=\left\lvertg_U(x)f(x)\right\rvert\left\lvertdx\right\rvert
=\left\lvertP_U(x)\right\rvert\left\lvertdx\right\rvert\,,
\end{align}
where $P_U(x)$ is a polynomial with coefficients in $K$ in $n$ variables, according to the Weierstrass preparation theorem. 
Notice that, by the
strict triangle inequality, this implies that
\[
z\mapsto \left\lvertP(z)\right\rvert
\]
defines a locally constant function on $U$ outside the zeros of $f$. Hence, if the the $p$-adic ball
$B(a)\subset U$ is sufficiently small, then $\left\lvertP(z)\right\rvert$ is constant on $B(a)$. In this
case,
\begin{align}\nonumber
\int_{\operatorname{cl}_w\Omega}\psi(x)\,d\nu_f(y)&=
\int_{\operatorname{cl}_w\Omega}\psi(x)\left\lvertf(x)\right\rvert{d\mu_\Lambda(x)}
\\\label{waveletZeroMean}
&=\int_U\psi(x)\left\lvertf(x)g_U(x)\right\rvert\left\lvertdx\right\rvert
=\tilde{C}\int_U\psi(x)\left\lvertdx\right\rvert=0
\end{align}
for some $\tilde{C}>0$, where the latter equality holds true, as
the vanishing of the Kozyrev wavelet on any measurable
set containing its support is well known, for example, cf.\ \cite[Theorem 2.3.9]{XKZ2018}. Notice that, in the context of the sloppy notation, the set $U$ is identified with its image in $K^n$ under the chart map, and $\psi$ is there identified with a Kozyrev wavelet.

\smallskip
For the eigenvalue formula, first observe that the independence of the
choice of $x\in B(a)$ is immediate. Next, observe that this formula is an extension to higher dimension of the one in \cite[Theorem 3]{Kozyrev2004}, and the proof of that uses only ultrametricity within the support of the wavelet. So, in this case, if $x \in \operatorname{cl}_w\Omega\setminus B(a)$, then
\begin{align}\nonumber
\Delta^\alpha_{f,\Omega}\psi(x)&=\int_{\operatorname{cl}_w\Omega}d_\Lambda(x,y)^{-n\alpha}(\psi(x)-\psi(y))\,d\nu_f(y)
\\\label{useVanishInt}
&=\psi(x)\int_{\operatorname{cl}_w\Omega} d_\Lambda(x,y)^{-n\alpha}\,d\nu_f(y)
-d_\Lambda(x,B(a))^{-n\alpha}\int_{\operatorname{cl}_w\Omega}\psi(x)\,d\nu_f(y)
\\\nonumber
&=0\,,
\end{align}
because  the latter integral vanishes by (\ref{waveletZeroMean}), and as $\psi(x)=0$. If $x\in B(a)$, then
\begin{align*}
\Delta^\alpha_{f,\Omega}\psi(x)&=\psi(x)\int_{X\setminus B(a)}d\Lambda(x,y)^{-n\alpha}\,d\nu_f(y)
+\int_{B(a)}d_\Lambda(x,y)^{-n\alpha}(\psi(x)-\psi(y))\,d\nu_f(y)\,.
\end{align*}
The latter integral equals
\begin{align*}
I&=\int_{\partial B(a)}d_\Lambda(x,y)^{-n\alpha}(\psi(x)-\psi(y))\left\lvertf(y)\right\rvert\,d\mu_\Lambda(y)\,.
\end{align*}
Again, since $B(a)$ is small, on a local chart $(U,\phi)$ containing $B(a)$, equality (\ref{polynomialMeasure}) shows that
\[
d\nu_f(y)|_{B(a)}=C\left\lvertdy\right\rvert
\]
for some $C>0$. Hence,
\begin{align*}
I&=C\int_{\partial \phi(B(a))}\left\|\phi(x)-z\right\|_{K^n}(\psi_{\phi(B(a)),j}(x)-\psi_{\phi(B(a)),j}(y))\left\lvertdy\right\rvert\,,
\end{align*}
where $\psi_{\phi(B(a)),j}$ is a Kozyrev wavelet supported in $\phi(B(a))$, and depending on $j\in\mathds{F}_q^\times$. Assume that the radius of $B(a)$ is $q^{-k}$. Then points $z\in\partial\phi(B(x))$ have the form
\[
z=\phi(x)+u+\epsilon
\]
with $u=(u_1,\dots,u_n)\in\pi^kO_K^n$ such that for $i\in{\left\{1,\dots,n\right\}}$:
\[
u_i=\pi^k\tau_i(\ell_i)\,,\quad\ell_i\in\mathds{F}_q
\]
and $\ell_i\in\mathds{F}_q^\times$
 for at least one $i\in{\left\{1,\dots,n\right\}}$, and with $\epsilon\in\pi^{k+1}O_K^n$. 
 The maps
 \[
\tau_i\colon \mathds{F}_q\to O_K
 \]
 are lifts of the canonical map $O_K\to\mathds{F}_q$ in the coordinates $i=1,\dots,n$, forming the map $\tau\colon \mathds{F}_q^n\to O_K^n$. Hence,
 \begin{align*}
I&=C\sum\limits_{i=1}^n\sum\limits_{\ell=(\ell_1,\dots,\ell_n)\in\atop\mathds{F}_q\times\dots\times\mathds{F}_q^\times\times\dots\times\mathds{F}_q}
\int_{\pi^{k+1}O_K^n}\left\|\tau(\ell)+\epsilon\right\|^{-n\alpha}
\left[\psi_{\phi(B(a)),j}(\phi(x))\right.
\\
&\quad\left.-\psi_{\phi(B(a)),j}(\phi(x)+\tau(\ell)+\epsilon)\right]\,\left\lvertd\epsilon\right\rvert
\\
&=Cq^{\alpha nk-n(k+1)}\sum\limits_{\ell\in\mathds{F}_q^n\setminus{\left\{\bar{0}\right\}}}\left[\psi_{\phi(B(a)),j}(\phi(x))-\psi_{\phi(B(a)),j}(\phi(x)+\tau(\ell))\right]
\\
&\stackrel{(*)}{=}
Cq^{\alpha nk-n(k+1)}\sum\limits_{\ell\in\mathds{F}_q^n\setminus{\left\{\bar{0}\right\}}}
\left[1-\chi(\pi^{-1}\tau(j)\tau(\ell))\psi_{\phi(B(a)),j}(\phi(x))\right]
\\
&\stackrel{(**)}{=}
Cq^{\alpha nk-n(k+1)}\left(q^n-(1+(-1)^n)\right)\psi_{\phi(B(a)),j}(\phi(x))
\\
&=C\mu(\phi(B(a)))^{n-\alpha}\left(1-q^{-n}(1+(-1)^n)\right)\psi_{\phi(B(a)),j}(\phi(x))
\\
&=\nu_f(B(a))^{n-\alpha}\left(1-q^{-n}(1+(-1)^n)\right)\psi(x)\,,
 \end{align*}
 where $(*)$ follows by inspecting the Kozyrev wavelets, and $(**)$ from
 \[
\sum\limits_{\ell\in\mathds{F}_q^n\setminus{\left\{\bar{0}\right\}}}\left(1-\chi(\pi^{-1}\tau(j)\tau(\ell))\right)
=q^n-(1+(-1)^n)\,,
 \]
 as can be seen by induction. This now implies the asserted eigenvalue.

<!-- example-2: body_md -->
Let $n=1$, $K=\mathds{Q}_p$ with $p\ge3$. Let $X=\mathds{Z}_p\cup (p^{-1}+\mathds{Z}_p)$, on which the differential $1$-form
\[
\omega(x)=\begin{cases}
(x-p^2)(x+p^2)\,dx,&x\notin B_3(p^2)\cup B_3(-p^2)
\\
dx,&\text{otherwise}
\end{cases}
\]
is defined,
and let 
\[
\Delta_\omega^\alpha u(x)
=\int_{\mathds{Z}_p}\left\lvertx-y\right\rvert^{-\alpha}(u(x)-u(y))\,\left\lvert\omega(x)\right\rvert
\] 
be, up to a constant factor, the Vladimirov operator on $\mathds{Z}_p$.
The Kozyrev wavelet
\[
\psi(x)=e^{\{i p^{-1}x\}_p}1_{\mathds{Z}_p}(x)\,,
\]
where 
\[
\{ \alpha_{-m}p^{-m}+\alpha_{-m+1}p^{-m+1}+\dots\}_p=\sum\limits_{j=-m}^{-1}\alpha_jp^j\in\mathds{Q}\,,\quad\alpha_j\in{\left\{0,\dots,p-1\right\}}\,,
\]
is the principal part of a $p$-adic number,
is not an eigenfunction of $\Delta_\omega^\alpha$.

<!-- proof-11: body_md -->
We check  that
\begin{align*}
\int_{\mathds{Z}_p}\psi(x)\left\lvert\omega(x)\right\rvert
&=\int_{B_3(p^2)}1\left\lvertdx\right\rvert
+\int_{B_3(-p^2)}1\left\lvertdx\right\rvert
\\
&+\int_{\left\lvertx\right\rvert=1}\psi(x)\left\lvertdx\right\rvert
+\int_{\left\lvertx\right\rvert=p^{-1}}1\left\lvertdx\right\rvert
+\underbrace{\int_{H}1\left\lvertx-p^2\right\rvert_p\left\lvertx+p^2\right\rvert_p\left\lvertdx\right\rvert}_{=I}
\\
&=p^{-2}+p^{-2}-p^{-1}+p^{-1}(1-p^{-1})+I
\\
&=I+p^{-2}\,,
\end{align*}
where
\[
H={\left\{\left\lvertx\right\rvert=p^{-2}\right\}}\setminus (B_3(p^2)\cup B_3(-p^2))\,.
\]
Since $I>0$, it follows that
\[
\int_{\mathds{Z}_p}\psi(x)\left\lvert\omega(x)\right\rvert\neq0\,.
\]
Hence, $\psi$ is not an eigenfunction of $\Delta_\omega^\alpha$, because
\[
\Delta_\omega^\alpha\psi(x)=-I\neq0
\]
for $x\in
X\setminus\mathds{Z}_p$
by the calculation in (\ref{useVanishInt}).

<!-- thm-2: body_md -->
The norm $\left\|u\right\|_\infty$ of the solution $u\in\operatorname{dom}(\mathcal{E}_{f,\operatorname{cl}_w\Omega})$ of the Neumann boundary value problem (\ref{NBVP_f})
informs for varying $a\in X\setminus V(f)$ about whether or not $a$ is approaching some $z_0\in V(f)$ of order $r>0$.
<!-- end thm-2 -->
<!-- proof-12: body_md -->
Let $z_0\in V(f)$ be of order $r>0$, and assume that $a\in X$ is closer to $z_0$ than to any other point in $V(f)$. Then
\[
\nu_f(B(a))=C(a)\left\lverta-z_0\right\rvert^r q^{-k}
=C(a)q^{-(k+rn_a)}
\]
for some $C(a)>0$ bounded from below, where
$\left\lverta-z_0\right\rvert=q^{-n_a}$.
According to Proposition \ref{smallWaveletEigenvalue},
\[
\lambda_{\psi_{B_k(a)},j}\in O(\nu_f(B_k(a))^{-n\alpha})\,.
\]
Hence, $\left\|u\right\|_\infty=\frac{1}{\lambda_\psi}$ is of the order
\[
Cq^{-(k+rn_a)n\alpha}=Cq^{-kn\alpha}(q^{-n\alpha})^{rn_a}
\]
for some $C>0$.
This means that for fixed radius of a small ball $B_k(a)$, the (existent or non-existent) rate of the exponential decay of the solution $\left\|u\right\|_\infty$ informs about an approximate location of $z_0\in V(f)$, as well as its order $r>0$, as $a$ approaches $z_0$ (or not).
