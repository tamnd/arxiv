<!-- problem-1: body_md -->
Given an LRS $u$, decide whether there exists $n \in \mathbb{N}$ such that $u_n = 0$.
<!-- end problem-1 -->
<!-- conjecture-1: body_md -->
\label{conj:skolem-conj}
Suppose that $u$ is a simple LRS such that for some integer $b \geq 1$, $u_n \in \mathbb{Z}[1/b]$ for all $n \in \mathbb{Z}$. Then $u$ has no integer zero if and only if there exists an integer $m \geq 1$ with $\gcd(m,b) = 1$ such that $u_n \not\equiv 0 \bmod m$ for all $n \in \mathbb{Z}$.
<!-- end conjecture-1 -->
<!-- conjecture-2: body_md -->
\label{conj:failed-conj}
If $u$ is a simple LRS which has a $p$-adic zero for every regular prime $p$, then there exists $n \in \mathbb{Z}$ with $u_n = 0$.
<!-- end conjecture-2 -->
<!-- example-1: body_md -->
\label{ex:intro-ex}
Let $u_n = (1+2^n)(1+(-4)^n) = 1 + 2^n + (-4)^n + (-8)^n$, which satisfies
\begin{align*}
    u_{n+4} = -9 u_{n+3} + 2 u_{n+2} + 72 u_{n+1} -64 u_n \, .
\end{align*}
It is clear that $1+2^n \neq 0$ and $1+(-4)^n \neq 0$ for all $n \in \mathbb{Z}$, and so $u_n \neq 0$ for all $n \in \mathbb{Z}$. However, 0 is a $p$-adic zero for every prime $p>2$. For a proof, see \cref{ex:bad-TRZ-example}.
<!-- end example-1 -->
<!-- theorem-1: body_md -->
\label{thm:sim-local-global-intro}
Suppose that $u,v$ are coprime LRS that are not weakly degenerate, taking values in $\mathbb{Z}[1/b]$, for some integer $b \geq 1$. Then, subject to the $p$-adic Schanuel conjecture, there does not exist $n \in \mathbb{Z}$ with $u_n = v_n = 0$ if and only if there exists an integer $m \geq 1$ with $\gcd(m,b) =1 $ such that $(u_n,v_n) \not\equiv (0,0) \bmod m$ for all $n \in \mathbb{Z}$.

Moreover, we may take $m = (pq)^k$, for some integer $k \geq 1$ and primes $p,q$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:avoiding-TRZs-intro}
Let $x \in \mathbb{Q}$ be a non-trivial TRZ, and $u$ be a non-degenerate LRS.
\begin{enumerate}
    \item If $x \not\in \mathbb{Z}$, then there are infinitely many primes $p$ for which $x$ is not a $p$-adic zero.
    \item If $u$ is also not weakly degenerate, $x \in \mathbb{Z}$ and $u_x \neq 0$, then there are infinitely many primes $p$ for which $x$ is not a $p$-adic zero.
\end{enumerate}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:detecting-TRZ-intro}
Any non-trivial TRZ of a non-weakly-degenerate LRS is a $p$-adic zero for infinitely many primes $p$ if and only if it is compatible.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:perucca}
Let $\mathbb{K}$ be a number field and for every $i=1, \dots, n$, let $R_i \in \mathbb{K}^\times$ be an algebraic number that is not a root of unity. Then for each integer $m > 0$ there exists a positive Dirichlet density of primes $\mathfrak{p} \subseteq \mathcal{O}_{\mathbb{K}}$ such that for each $i = 1 , \dots, n$, we have $\mathrm{ord}_\mathfrak{p}(R_i)$ is a multiple of $m$.
<!-- end theorem-4 -->
<!-- theorem-5: body_md -->
\label{thm:sgobba}
Let $\mathbb{K}$ be a number field and let $G$ be a finitely generated and torsion-free subgroup of $\mathbb{K}^\times$ of positive rank. Let $\mathbb{F}/\mathbb{K}$ be a finite Galois extension. Consider a finite set $L$ of prime numbers $\ell$, and for each of them fix a non-negative integer $a_\ell$. Consider the set of primes of $\mathbb{K}$ given by
\begin{align*}
    \mathcal V = \left \{ \mathfrak{p} \subseteq \mathcal{O}_\mathbb{K} : v_\mathfrak{p}(g) = 0 \ \forall g \in G \, , \, \mathfrak{p} \textnormal{ is unramified in } \mathbb{F}\, , \textnormal{ and } v_\ell(\mathrm{ord}_\mathfrak{p}(G)) = a_\ell \ \forall \ell \in L \right\} \, .
\end{align*}
Then $\mathcal V$ has positive natural density.
<!-- end theorem-5 -->
<!-- definition-1: body_md -->
We say that an LRS $u$ is degenerate if $\lambda_i/\lambda_j$ is a root of unity for some $i \neq j$. We say $u$ is weakly degenerate if $G  = \langle \lambda_1, \dots, \lambda_s \rangle \subseteq \mathbb{L}^\times$ is not torsion-free.
<!-- end definition-1 -->
<!-- theorem-6: body_md -->
\label{thm:finitely-many-primitive-tuples}
Let $u$ be a non-degenerate LRS. Then
\begin{enumerate}
    \item $u$ admits finitely many primitive vanishing tuples, and
    \item $u$ admits finitely many TRZs.
\end{enumerate}
<!-- end theorem-6 -->
<!-- definition-2: body_md -->
\label{def:TRZ-p-adic}
Given a number field $\mathbb{K}$, a $\mathbb{K}$-LRS $u$ and an admissible prime ideal $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ lying above rational prime $p$, we call $x \in \mathbb{Z}_p$ a $\mathfrak{p}$-adic zero of $u$ if there is a sequence of integers $n_j \in \mathbb{Z}$ such that $v_\mathfrak{p}(n_j-x) \to \infty$ and $v_{\mathfrak{p}}(u_{n_j}) \to \infty$ as $j \to \infty$.\@gobble{This differs from the definition of a $\mathfrak{p}$-adic zero in \cite{bacik_p-adic_2026} by a linear transformation of each $\mathfrak{p}$-adic zero.}
<!-- end definition-2 -->
<!-- lemma-1: body_md -->
\label{lem:p-adic-zero-def-equal}
Let $u$ be a $\mathbb{K}$-LRS and let $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ be an admissible prime ideal lying above rational prime $p \in \mathbb{Z}$. We have that $x \in \mathbb{Z}_p$ is a $\mathfrak{p}$-adic zero of $u$ according to \cref{def:TRZ-p-adic} if and only if there is $y \in \mathbb{Z}_p$ and $0 \leq \ell \leq N-1$ such that $F_\ell(y) = 0$, and $Ny+\ell =x $.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
If $x \in \mathbb{Z}_p$ is a $\mathfrak{p}$-adic zero of $u$, then there is a sequence of integers $n_j \in \mathbb{N}$ such that $v_\mathfrak{p}(n_j-x) \to \infty$ and $v_\mathfrak{p}(u_{n_j}) \to \infty$ as $j \to \infty$. By the pigeonhole principle there must be $0 \leq \ell \leq N-1$ such that there are infinitely many integers $n_j$ of the form $Nm_j + \ell$. Then $v_\mathfrak{p}(u_{Nm_j + \ell}) = v_\mathfrak{p}(F_\ell(m_j)) \to \infty$ as $j \to \infty$, and $m_j \to y \coloneq \frac{x-\ell}{N}$ in $p$-adic absolute value as $j \to \infty$, so by continuity of $F_\ell$, we have $F_\ell(y) = 0$ and $Ny + \ell =x$. 

For the other direction, if $F_\ell(y) = 0$ and $Ny+\ell = x$, then take any sequence of integers $m_j$ such that $v_\mathfrak{p}(m_j - y) \to \infty$, then we have $v_\mathfrak{p}(Nm_j+\ell - x) \to \infty$ as $j\to \infty$, and $v_\mathfrak{p}(F_\ell(m_j)) = v_\mathfrak{p}(u_{Nm_j+\ell}) \to \infty$ as $j \to \infty$ by continuity and the fact that $F_\ell(y) = 0$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:rat-zero-is-TRZ}
If $u$ is a $\mathbb{K}$-LRS, $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ is an admissible prime ideal, and $x \in \mathbb{Q}$ is a $\mathfrak{p}$-adic zero, then $x$ is a TRZ.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:pp-adic-to-PP-adic}
Let $u$ be a $\mathbb{K}$-LRS, let $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ be an admissible prime ideal and let $x \in \mathbb{Z}_p$. Let $\mathbb{L} \supseteq \mathbb{K}$ be some finite extension of $\mathbb{K}$. The following are equivalent
\begin{enumerate}
    \item $x$ is a $\mathfrak{p}$-adic zero of $u$;
    \item $x$ is a $\mathfrak{P}$-adic zero of $u$ for some prime ideal $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$ lying above $\mathfrak{p}$;
    \item $x$ is a $\mathfrak{P}$-adic zero of $u$ for all prime ideals $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$ lying above $\mathfrak{p}$.
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
According to \cref{def:TRZ-p-adic}, $x \in \mathbb{Z}_p$ is a $\mathfrak{p}$-adic zero if and only if there is a sequence of integers $n_j \in \mathbb{Z}$ such that $v_\mathfrak{p}(x- n_j) \to \infty$ and $v_\mathfrak{p}(u_{n_j}) \to \infty$ as $j \to \infty$. For every $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$ lying above $\mathfrak{p}$, and every $a \in \mathbb{K}_\mathfrak{p}$, we have $v_\mathfrak{P}(a) = e_{\mathfrak{P}/\mathfrak{p}} v_\mathfrak{p}(a)$, so $v_\mathfrak{p}(x-n_j) \to \infty$ and $v_\mathfrak{p}(u_{n_j}) \to \infty$ if and only if $v_\mathfrak{P}(x-n_j) \to \infty$ and $v_\mathfrak{P}(u_{n_j}) \to \infty$. This is exactly the definition of $x$ being a $\mathfrak{P}$-adic zero.
<!-- end proof-2 -->
<!-- lemma-4: body_md -->
\label{lem:root-of-unity-unique}
Let $a/b \in \mathbb{Q}$ with $\gcd(a,b) = 1$. Let $\lambda \in \mathbb{K}$ be an algebraic number and $\theta \in \mathbb{K}$ be such that $\theta^b = \lambda$. Suppose $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ is a prime ideal lying above rational prime $p \in \mathbb{Z}$ such that $v_\mathfrak{p}(\lambda-1) > \frac{e_{\mathfrak{p}/p}}{p-1}$ and $v_\mathfrak{p}(b) = 0$. Then $\exp(\frac{a}{b} \log \lambda) = \xi \theta^a$ where $\xi$ is some $b$-th root of unity. Furthermore, if $p$ is unramified in $\mathbb{K}$ and $p>2$, then $\xi$ is the unique root of unity in $\mathbb{K}_\mathfrak{p}$ with $\xi \theta^a \equiv 1 \bmod \mathfrak{p}$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
We have $\exp(\frac{a}{b} \log \lambda)^b = \exp(a \log \lambda) = \lambda^a$ so $\exp(\frac{a}{b} \log \lambda)$ is some $b$-th root of $\lambda^a$, meaning $\exp(\frac{a}{b} \log \lambda) = \xi \theta^a$ for some $b$-th root of unity $\xi$.

Furthermore, if $p$ is unramified in $\mathbb{K}$, then $\mathbb{K}_\mathfrak{p}/\mathbb{Q}_p$ is an unramified extension. Suppose there are two roots of unity $\xi, \xi' \in \mathbb{K}_\mathfrak{p}$ such that $\xi \theta^a \equiv \xi' \theta^a \equiv 1 \bmod \mathfrak{p}$. Then $\xi/\xi' \equiv 1 \bmod \mathfrak{p}$. Suppose that $\zeta \coloneq \xi/\xi' \neq 1$, then it is a primitive $p^\ell m$-th root of unity, for some integers $\ell \geq 0$ and $m\geq 1$ with $\gcd(m,p) = 1$. Note that $\zeta^{p^\ell}$ satisfies the polynomial $X^m-1$, which is separable over $\faktor{\mathbb{Z}}{p\mathbb{Z}}$, so 1 is the unique $m$-th root of unity which reduces to 1 in $\faktor{\mathcal{O}_\mathfrak{p}}{\mathfrak{p}}$. Thus, since $\zeta^{p^\ell} \equiv 1 \bmod \mathfrak{p}$, we have $\zeta^{p^\ell} = 1$, so $m=1$. But then $\mathbb{Q}_p(\zeta)/\mathbb{Q}_p$ is ramified \cite[Proposition 7.13]{neukirch_algebraic_1999}, since $\zeta$ is a $p$-power root of unity, so $\mathbb{K}_\mathfrak{p}/\mathbb{Q}_p$ is ramified, which contradicts our assumption. We conclude that $\xi$ is the unique root of unity in $\mathbb{K}_\mathfrak{p}$ with $\xi \theta^a \equiv 1 \bmod \mathfrak{p}$.
<!-- end proof-3 -->
<!-- theorem-7: body_md -->
\label{thm:avoiding_TRZ}
Let $U$ be a finite set of non-degenerate $\mathbb{K}$-LRS, and for each $u \in U$, let $Y_u$ be the set of non-trivial TRZs of $u$ that are not integer zeros of $u$. Let $Y = \bigcup_{u \in U} Y_u$. Let $Y_{u,1}= Y_u \setminus \mathbb{Z}$, and $Y_{u,2} = Y_u \cap \mathbb{Z} = Y\setminus Y_{u,1}$. Define also $Y_j = \bigcup_{u \in U} Y_{u,j}$ for $j=1,2$. Then:
\begin{enumerate}
    \item The set of prime ideals $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ such that, for each $u \in U$, no $y \in Y_{u,1}$ is a $\mathfrak{p}$-adic zero of $u$, has positive Dirichlet density.
    \item If moreover, the multiplicative group generated by the characteristic roots of all $u \in U$ is torsion-free, then the set of prime ideals $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ such that, for each $u \in U$, no $y \in Y_{u,2}$ is a $\mathfrak{p}$-adic zero of $u$, has positive Dirichlet density.
\end{enumerate}
<!-- end theorem-7 -->
<!-- proof-4: body_md -->
Let $u \in U$ take the form\@gobble{Note that since $U$ is finite, we can write the LRS in this way where $\lambda_i$ do not depend on $u$, by allowing the $P_{i,u}$ to be identically zero.}
\begin{align} \label{eqn:u-def-2}
    u_n = \sum_{i=1}^s P_{u,i}(n) \lambda_i^n \, .
\end{align}
Any prime ideals considered in this proof will be assumed to be admissible for every LRS $u \in U$. Note that only finitely many primes are inadmissible, and discarding finitely many primes does not affect the truth of the result, so we may freely discard the inadmissible primes.

Note that by \cref{thm:finitely-many-primitive-tuples}, $Y$ must be finite. For each denominator $b$ of each TRZ $y \in Y_1$, we fix in advance a definition of $b$-th roots of each $\lambda_i$; let $\theta_{i,b} \in \smash{\overline{\mathbb{Q}}}$ be such that $\theta_{i,b}^b = \lambda_i$. Let $\Theta$ be the set of all $\theta_{i,b}$, for each $1 \leq i \leq s$ and each denominator $b$.

Consider all LRS formed by sub-sums of $u \in U$, that is, of the form
\begin{align*}
    u_n^{(I)} = \sum_{i \in I} P_{u,i}(n) \lambda_i^n \, ,
\end{align*}
where $\varnothing \neq I \subseteq \{1, \dots, s\}$. By \cref{thm:finitely-many-primitive-tuples}, each $u_n^{(I)}$ has finitely many primitive vanishing tuples associated to each TRZ $y \in Y$. Let $\Xi$ be the set of all roots of unity appearing in a primitive vanishing tuple of some $u_n^{(I)}$, ranging over $u \in U$ and $\varnothing \neq I \subseteq \{1, \dots, s\}$. Note that $\Xi$ is finite, since there are finitely many subsets $I \subseteq \{1,\dots, s\}$. 

Define $\mathbb{L} = \mathbb{K}(\Theta , \Xi)$. By \cref{lem:pp-adic-to-PP-adic}, it suffices to prove $x$ is not a $\mathfrak{P}$-adic zero of any $u \in U$ for a positive density of prime ideals $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$. Finally, let $N > 0$ be any integer for which $v_\mathfrak{P}(\lambda_i^N -1 ) > \frac{e_{\mathfrak{P}/p}}{p-1}$ for all $1 \leq i \leq s$. We denote the $\ell$-th $\mathfrak{P}$-adic interpolant of $u$ by $F_{u,\ell}$ and the $\ell$-th $\mathfrak{P}$-adic interpolant of $u^{(I)}$ by $F^{(I)}_{u,\ell}$. 

Given a vanishing tuple $(y, \xi_1, \dots, \xi_s)$ for $u$, if we have disjoint subsets $I_0,I_1, \dots, I_t \subseteq \{1, \dots, s\}$ such that $I_0 \cup I_1 \cup \dots \cup I_t = \{1, \dots, s\}$, we have $P_{u,i}(y) = 0$ for all $i \in I_0$, and each tuple $(y, (\xi_i)_{i \in I_j})$ is a primitive vanishing tuple of $u_n^{(I_j)}$ for $j=1 ,\dots, t$, then we say $(y, \xi_1, \dots, \xi_s)$ has a primitive decomposition given by $I_0, I_1, \dots, I_t$. We denote the largest element of a subset $I_j$ by $[I_j]$. 

We need the following lemma.
\begin{lemma} \label{lem:divisibility-condition}
Write $y = a/b$, where $a \in \mathbb{Z}$, $b \in \mathbb{N}$, and $\gcd(a,b) = 1$. If $y \in Y_u$ is a $\mathfrak{P}$-adic zero of some $u \in U$, then there is a vanishing tuple $(y, \xi_1, \dots, \xi_s)$ with a primitive decomposition given by subsets $I_0, I_1, \dots I_t$, and there exists $0 \leq \ell \leq N-1$ such that for each $j=1, \dots, t$ and each $i \in I_j$, we have
\begin{align*}
    \mathrm{ord}_\mathfrak{P} \left( \frac{\theta_{i,b}}{\theta_{[I_j],b}} \right) \mid \mathrm{ord}(\xi_{i})(a-b\ell) \, .
\end{align*}
Furthermore, if $b = 1$ and
\begin{align*}
    \mathrm{ord}_\mathfrak{P} \left( \frac{\theta_{i}}{\theta_{[I_j]}} \right) \mid (a-\ell) \, .
\end{align*}
for all $j=1, \dots, t$ and $i \in I_j$, then $u_y = u_a = 0$.
\end{lemma}
\begin{proof}
If $y$ is a $\mathfrak{P}$-adic zero of $u$ then by \cref{lem:p-adic-zero-def-equal} we have $0 \leq \ell \leq N-1$ such that $\frac{a-b\ell}{bN} \in \mathbb{Z}_p$, where $p \in \mathbb{Z}$ is the rational prime lying below $\mathfrak{P}$, and
\begin{align*}
    F_{u,\ell} \left(\frac{a-b\ell}{bN} \right) &= \sum_{i=1}^s P_{u,i}(y) \lambda_i^{\ell} \exp \left( \frac{a-b\ell}{bN} \log \lambda_i^N \right) = 0 \, .
\end{align*}
We split the sum into primitive sub-sums; there are disjoint $I_0,I_1, \dots, I_t \subseteq \{1, \dots, s\}$ with $I_0 \cup I_1 \cup \dots \cup I_t = \{1, \dots, s\}$ such that $P_i(y) = 0$ for all $i \in I_0$, and for each $j=1 , \dots, t$ and any proper subset $J \subsetneq I_j$ we have
\begin{align*}
    \sum_{i \in I_j} P_{u,i}(x) \lambda_i^{\ell} \exp \left( \frac{a-b\ell}{bN} \log \lambda_i^N \right) =  0 \, , \qquad \sum_{i \in J} P_{u,i}(x) \lambda_i^{\ell} \exp \left( \frac{a-b\ell}{bN} \log \lambda_i^N \right) \neq 0 \, .
\end{align*}

Since $\frac{a-b\ell}{bN} \in \mathbb{Z}_p$, when written in lowest terms, the denominator has valuation zero with respect to $p$. Therefore, we may apply \cref{lem:root-of-unity-unique} to get that
\begin{align*}
    \exp \left(\frac{a-b\ell}{bN} \log \left(  \lambda_{i}^N \right) \right) = \zeta_{i} \theta_{i,b}^{a-b\ell} \, ,
\end{align*}
where $\zeta_{i}$ is a root of unity with $\zeta_{i}^{bN} = 1$. Therefore, this gives rise to a primitive vanishing tuple $\left( y, \left( \zeta_i/\zeta_{[I_j]} \right)_{i \in I_j} \right)$ of $u^{(I_j)}$. Let $\xi_i = \zeta_i/\zeta_{[I_j]}$. 

Now note that since 
\begin{align*}
\exp \left(\frac{a-b\ell}{bN} \log \left(  \lambda_{i}^N \right) \right) \equiv 1 \mod \mathfrak{P} \, ,
\end{align*}
we have $\zeta_i \theta_{i,b}^{a-b\ell} \equiv 1 \bmod \mathfrak{P}$, and therefore
\begin{align*}
    \xi_{i} \left( \frac{\theta_{i,b}}{\theta_{[I_j],b}} \right)^{a-b\ell} \equiv 1 \mod \mathfrak{P} \, ,
\end{align*}
for all $i \in I_j$, and so $\left( \theta_{i,b}/\theta_{[I_j],b} \right)^{(a-b\ell) \mathrm{ord}(\xi_{i})} \equiv 1 \mod \mathfrak{P}$, which implies that 
\begin{align} \label{eqn:divisibility-condition}
    \mathrm{ord}_\mathfrak{P} \left( \frac{\theta_{i,b}}{\theta_{[I_j],b}} \right) \mid \mathrm{ord}(\xi_{i})(a-b\ell) \, ,
\end{align}
and this is what was required to show, for the first part.

For the second part, if $b=1$ and $\mathrm{ord}_\mathfrak{P} \left( \lambda_{i}/\lambda_{[I_j]} \right) \mid (a-\ell)$ for all $j = 1, \dots, t$ and $i \in I_j$, then we have  
\begin{align*}
    &F^{(I_j)}_{u,\ell} \left( \frac{a- \ell}{N} \right) = \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^\ell \exp \left(\frac{a-\ell}{N} \log \left(  \left(\frac{\lambda_i}{\lambda_{[I_j]}} \right)^N \right) \right) \\
   &=   \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^\ell \exp \left(\frac{a-\ell}{\mathrm{ord}_\mathfrak{P}(\lambda_i/\lambda_{[I_j]})} \log \left(  \left(\frac{\lambda_i}{\lambda_{[I_j]}} \right)^{\mathrm{ord}_\mathfrak{P}(\lambda_i/\lambda_{[I_j]})} \right) \right) \\
   &=   \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^\ell \exp \left( \log \left(  \left(\frac{\lambda_i}{\lambda_{[I_j]}} \right)^{a-\ell} \right) \right) \\
   &=  \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^a \\
   &= \frac{\lambda_{[I_j]}^\ell \exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right)}{\lambda_{[I_j]}^a} u^{(I_j)}_a  \\
   &= 0 \, ,
\end{align*}
so $u^{(I_j)}_a = 0$ for all $j = 1 , \dots, t$, and therefore $u_a = \sum_{j=1}^t u^{(I_j)}_a = 0$.
\end{proof}
We now show how the divisibility condition in \cref{lem:divisibility-condition} may be avoided for a positive density of primes $\mathfrak{P}$, for TRZs in $Y_1$ and $Y_2$.

\textbf{Case 1:} For each $y \in Y_1$, write $y = a/b$ where $a \in \mathbb{Z}$, $b \in \mathbb{N}$ and $\gcd(a,b) = 1$. Let $B$ be the set of all denominators of all $y \in Y_1$.

Define the integer
\begin{align*}
    m = 1+ \max\{v_q(\mathrm{ord}(\xi)) \mid \xi \in \Xi \text{ and } q \text{ is a prime dividing some }  b \in B \} \, .
\end{align*}
Consider the multiplicative group $\mathbb{K}^\times$, and consider the finite set of elements 
\begin{align*}
    A = \left\{ \frac{\theta_{i,b}}{\theta_{j,b}} \, \middle| \, \lambda_i,\lambda_j \text{ are both characteristic roots of some $u \in U$,} \text{ and } i\neq j\, , \text{ and } b \in B \right\} \, .
\end{align*}
Since each $u \in U$ is non-degenerate, no element of $A$ is a root of unity. Therefore, by \cref{thm:perucca} applied to the elements of $A$, there is a positive Dirichlet density of primes $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$ such that for every $i \neq j$, we have $\prod_{\tilde b \in B} \tilde{b}^m \mid \mathrm{ord}_\mathfrak{P}(\theta_{i,b}/\theta_{j,b})$ for every $b \in B$. Now, given any $u \in U$, $y = a/b \in Y_{u,1}$, and any prime $q \mid b$, \cref{lem:divisibility-condition} implies that for such primes $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$, if $y$ is a $\mathfrak{P}$-adic zero of $u$, then we have $q^m \mid \mathrm{ord}(\xi)(a-b\ell)$ for some $\xi \in \Xi$ and some integer $\ell$. But since $\gcd(a,b) = 1$, $q \nmid (a-b \ell)$; indeed otherwise $q \mid a$ which contradicts $\gcd(a,b)=1$. Therefore, $q^m \mid \mathrm{ord}(\xi)$. This contradicts the definition of $m$, so $y$ is not a $\mathfrak{P}$-adic zero of $u$. 

\textbf{Case 2:} Let $Q$ be the (finite) set of rational primes that divide any $\mathrm{ord}(\xi)$ for $\xi \in \Xi$. Let $G = \langle \lambda_1, \dots, \lambda_s \rangle \subseteq \mathbb{L}^\times$, then $G$ is torsion-free, so we may apply \cref{thm:sgobba} to conclude that there is a positive density of prime ideals $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$ such that $v_{q}(\mathrm{ord}_\mathfrak{P}(G)) = 0$ for all $q \in Q$. In particular, for each $1 \leq i, j \leq s$ with $i \neq j$, we have $q \nmid \mathrm{ord}_\mathfrak{P}(\lambda_i/\lambda_j)$ for each $q \in Q$, and hence $\gcd(\mathrm{ord}_\mathfrak{P}(\lambda_i/\lambda_j) , \mathrm{ord}(\xi_i) ) = 1$. 

Suppose that for such a prime $\mathfrak{P}$, there is $u \in U$ for which some $y \in Y_{u,2}$ is a $\mathfrak{P}$-adic zero of $u$. Then by \cref{lem:p-adic-zero-def-equal}, we have a vanishing tuple $(y, \xi_1, \dots, \xi_s)$ with primitive decomposition $I_0,I_1, \dots, I_t$ such that for each $j=1, \dots, t$ and all $i \in I_j$ we have
\begin{align*}
    \mathrm{ord}_\mathfrak{P} \left( \frac{\lambda_{i}}{\lambda_{[I_j]}} \right) \mid \mathrm{ord}(\xi_{i})(a-\ell) \, .
\end{align*}
Furthermore, since $u_y \neq 0$ by assumption, there is some $i,j$ for which $\mathrm{ord}_\mathfrak{P} \left( \lambda_{i}/\lambda_{[I_j]} \right) \nmid (a-\ell) $, so therefore $\gcd \left(\mathrm{ord}_\mathfrak{P} \left( \lambda_{i}/\lambda_{[I_j]} \right) , \mathrm{ord}(\xi_i) \right) > 1$. But this contradicts the choice of $\mathfrak{P}$. So no $y \in Y_{u,2}$ is a $\mathfrak{P}$-adic zero of $u$ for any such prime $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$.

<!-- lemma-5: body_md -->
\label{lem:divisibility-condition}
Write $y = a/b$, where $a \in \mathbb{Z}$, $b \in \mathbb{N}$, and $\gcd(a,b) = 1$. If $y \in Y_u$ is a $\mathfrak{P}$-adic zero of some $u \in U$, then there is a vanishing tuple $(y, \xi_1, \dots, \xi_s)$ with a primitive decomposition given by subsets $I_0, I_1, \dots I_t$, and there exists $0 \leq \ell \leq N-1$ such that for each $j=1, \dots, t$ and each $i \in I_j$, we have
\begin{align*}
    \mathrm{ord}_\mathfrak{P} \left( \frac{\theta_{i,b}}{\theta_{[I_j],b}} \right) \mid \mathrm{ord}(\xi_{i})(a-b\ell) \, .
\end{align*}
Furthermore, if $b = 1$ and
\begin{align*}
    \mathrm{ord}_\mathfrak{P} \left( \frac{\theta_{i}}{\theta_{[I_j]}} \right) \mid (a-\ell) \, .
\end{align*}
for all $j=1, \dots, t$ and $i \in I_j$, then $u_y = u_a = 0$.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
If $y$ is a $\mathfrak{P}$-adic zero of $u$ then by \cref{lem:p-adic-zero-def-equal} we have $0 \leq \ell \leq N-1$ such that $\frac{a-b\ell}{bN} \in \mathbb{Z}_p$, where $p \in \mathbb{Z}$ is the rational prime lying below $\mathfrak{P}$, and
\begin{align*}
    F_{u,\ell} \left(\frac{a-b\ell}{bN} \right) &= \sum_{i=1}^s P_{u,i}(y) \lambda_i^{\ell} \exp \left( \frac{a-b\ell}{bN} \log \lambda_i^N \right) = 0 \, .
\end{align*}
We split the sum into primitive sub-sums; there are disjoint $I_0,I_1, \dots, I_t \subseteq \{1, \dots, s\}$ with $I_0 \cup I_1 \cup \dots \cup I_t = \{1, \dots, s\}$ such that $P_i(y) = 0$ for all $i \in I_0$, and for each $j=1 , \dots, t$ and any proper subset $J \subsetneq I_j$ we have
\begin{align*}
    \sum_{i \in I_j} P_{u,i}(x) \lambda_i^{\ell} \exp \left( \frac{a-b\ell}{bN} \log \lambda_i^N \right) =  0 \, , \qquad \sum_{i \in J} P_{u,i}(x) \lambda_i^{\ell} \exp \left( \frac{a-b\ell}{bN} \log \lambda_i^N \right) \neq 0 \, .
\end{align*}

Since $\frac{a-b\ell}{bN} \in \mathbb{Z}_p$, when written in lowest terms, the denominator has valuation zero with respect to $p$. Therefore, we may apply \cref{lem:root-of-unity-unique} to get that
\begin{align*}
    \exp \left(\frac{a-b\ell}{bN} \log \left(  \lambda_{i}^N \right) \right) = \zeta_{i} \theta_{i,b}^{a-b\ell} \, ,
\end{align*}
where $\zeta_{i}$ is a root of unity with $\zeta_{i}^{bN} = 1$. Therefore, this gives rise to a primitive vanishing tuple $\left( y, \left( \zeta_i/\zeta_{[I_j]} \right)_{i \in I_j} \right)$ of $u^{(I_j)}$. Let $\xi_i = \zeta_i/\zeta_{[I_j]}$. 

Now note that since 
\begin{align*}
\exp \left(\frac{a-b\ell}{bN} \log \left(  \lambda_{i}^N \right) \right) \equiv 1 \mod \mathfrak{P} \, ,
\end{align*}
we have $\zeta_i \theta_{i,b}^{a-b\ell} \equiv 1 \bmod \mathfrak{P}$, and therefore
\begin{align*}
    \xi_{i} \left( \frac{\theta_{i,b}}{\theta_{[I_j],b}} \right)^{a-b\ell} \equiv 1 \mod \mathfrak{P} \, ,
\end{align*}
for all $i \in I_j$, and so $\left( \theta_{i,b}/\theta_{[I_j],b} \right)^{(a-b\ell) \mathrm{ord}(\xi_{i})} \equiv 1 \mod \mathfrak{P}$, which implies that 
\begin{align} \label{eqn:divisibility-condition}
    \mathrm{ord}_\mathfrak{P} \left( \frac{\theta_{i,b}}{\theta_{[I_j],b}} \right) \mid \mathrm{ord}(\xi_{i})(a-b\ell) \, ,
\end{align}
and this is what was required to show, for the first part.

For the second part, if $b=1$ and $\mathrm{ord}_\mathfrak{P} \left( \lambda_{i}/\lambda_{[I_j]} \right) \mid (a-\ell)$ for all $j = 1, \dots, t$ and $i \in I_j$, then we have  
\begin{align*}
    &F^{(I_j)}_{u,\ell} \left( \frac{a- \ell}{N} \right) = \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^\ell \exp \left(\frac{a-\ell}{N} \log \left(  \left(\frac{\lambda_i}{\lambda_{[I_j]}} \right)^N \right) \right) \\
   &=   \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^\ell \exp \left(\frac{a-\ell}{\mathrm{ord}_\mathfrak{P}(\lambda_i/\lambda_{[I_j]})} \log \left(  \left(\frac{\lambda_i}{\lambda_{[I_j]}} \right)^{\mathrm{ord}_\mathfrak{P}(\lambda_i/\lambda_{[I_j]})} \right) \right) \\
   &=   \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^\ell \exp \left( \log \left(  \left(\frac{\lambda_i}{\lambda_{[I_j]}} \right)^{a-\ell} \right) \right) \\
   &=  \lambda_{[I_j]}^\ell\exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right) \sum_{i \in I_j} P_{u,i}(a) \left(\frac{\lambda_i}{\lambda_{[I_j]}}\right)^a \\
   &= \frac{\lambda_{[I_j]}^\ell \exp \left(\frac{a-\ell}{N} \log \left(  \lambda_{[I_j]}^N \right) \right)}{\lambda_{[I_j]}^a} u^{(I_j)}_a  \\
   &= 0 \, ,
\end{align*}
so $u^{(I_j)}_a = 0$ for all $j = 1 , \dots, t$, and therefore $u_a = \sum_{j=1}^t u^{(I_j)}_a = 0$.

<!-- corollary-1: body_md -->
\label{cor:avoiding-cor}
Let $u$ be a non-degenerate $\mathbb{K}$-LRS and $x \in \mathbb{Q}$. Suppose for all prime ideals $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ except a set of zero density, there is a sequence of integers $n_j \in \mathbb{Z}$ (depending on $\mathfrak{p}$) such that
\begin{align*}
    v_\mathfrak{p}(n_j - x) \to \infty \, \qquad v_\mathfrak{p}(u_{n_j}) \to \infty \, .
\end{align*}
Then
\begin{enumerate}
    \item if $x \not\in \mathbb{Z}$, then $x$ is a trivial TRZ.
    \item if $u$ is also non-weakly-degenerate, then either $x$ is a trivial TRZ, or $x \in \mathbb{Z}$ and $u_x = 0$. 
\end{enumerate}

Item 1 simply follows from the contrapositive of item 1 of \cref{thm:avoiding_TRZ}, while item 2 follows from the contrapositive of item 2 of \cref{thm:avoiding_TRZ}, using $U = \{u\}$ and $Y = \{x\}$.

<!-- theorem-8: body_md -->
\label{thm:detecting-TRZ}
Any non-trivial TRZ of a non-weakly-degenerate $\mathbb{K}$-LRS $u$ is a $\mathfrak{p}$-adic zero for a positive density of primes $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ if and only if it is compatible.
<!-- end theorem-8 -->
<!-- proof-7: body_md -->
We once again assume any prime ideal considered is admissible for $u$, as this discards only finitely many primes. Let $u$ have characteristic roots $\lambda_1, \dots, \lambda_s$, whose multiplicative group is generated by $\mu_1, \dots, \mu_r$, with
\begin{align*}
    \lambda_i = \mu_1^{m_{i,1}} \cdots \mu_r^{m_{i,r}} \, .
\end{align*}
Let $y \in \mathbb{Q}$ be a TRZ of $u$, and write $y=a/b$, where $\gcd(a,b) = 1$. First we prove that if $y$ is a $\mathfrak{p}$-adic zero for some prime ideal $\mathfrak{p}$, it is compatible with $u$. Indeed, let $\mathfrak{P}$ be a prime of $\mathbb{K}(\mu_1, \dots, \mu_r)$, and let $N >0$ be an integer such that $v_\mathfrak{P}(\mu_i^{N}-1) > \frac{e_{\mathfrak{P}/p}}{p-1}$. Let $\theta_i$ be a $b$-th root of $\mu_i$; we have $\theta_i^b = \mu_i$. If $a/b$ is a $\mathfrak{P}$-adic zero, then by \cref{lem:p-adic-zero-def-equal} there is integer $0 \leq \ell \leq N-1$ such that $\frac{a-b\ell}{bN} \in \mathbb{Z}_p$, and we have
\begin{align*}
    F_\ell \left(\frac{a-b\ell}{bN} \right) &= \sum_{i=1}^s P_i(a/b) \lambda_i^\ell \exp \left(\frac{a-b\ell}{bN} \log(\lambda_i^N) \right) = 0 \\
    &= \sum_{i=1}^s P_i(a/b) \lambda_i^\ell \prod_{j=1}^r \exp \left(\frac{a-b\ell}{bN} \log(\mu_j^{N}) \right)^{m_{i,j}} \\
    &= \sum_{i=1}^s P_i(a/b) \lambda_i^\ell \prod_{j=1}^r (\xi_j \theta_j^{a-b\ell})^{m_{i,j}} \\
    &= \sum_{i=1}^s P_i(a/b) \prod_{j=1}^r (\xi_j \theta_j^{a})^{m_{i,j}} \, ,
\end{align*}


where $\xi_j$ are some $bN$-th roots of unity, by \cref{lem:root-of-unity-unique}. This is exactly the definition of a compatible TRZ.

Conversely, suppose that $a/b$ is a compatible TRZ. Then there exist roots of unity $\xi_1, \dots, \xi_r$ such that
\begin{align*}
\sum_{i=1}^s P_i(a/b) (\xi_1 \theta_1^a)^{m_{i,1}} \cdots (\xi_r \theta_r^a)^{m_{i,r}} = 0 \, ,
\end{align*}
where $\theta_i^b = \mu_i$. Let $\mathbb{L} = \mathbb{K}(\theta_1,\dots,\theta_r, \xi_1,\dots,\xi_r)$, and consider prime ideals $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$ lying above $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ and rational prime $p \in \mathbb{Z}$, such that $p>2$ and $p$ is unramified in $\mathbb{L}$. Note that restricting to $\mathfrak{P}$ lying above unramified $p>2$ excludes finitely many primes. Let $N$ be the smallest positive integer such that $v_\mathfrak{P}(\mu_i^N-1) > \frac{e_{\mathfrak{P}}}{p-1}$ for all $i = 1 , \dots, r$. By \cref{lem:root-of-unity-unique}, for any integer $0 \leq \ell \leq N-1$, and for each $i=1,\dots,r$, there is a root of unity $\zeta_i \in \mathbb{L}_\mathfrak{P}$ such that
\begin{align} \label{eqn:exp-log-eq}
    \exp \left( \frac{a-b\ell}{bN} \log(\mu_i^N)\right) = \zeta_i \theta_i^{a-b\ell} \, ,
\end{align}
and furthermore $\zeta_i$ is the unique root of unity in $\mathbb{L}_\mathfrak{P}$ with $\zeta_i \theta_i^{a-b\ell} \equiv 1 \bmod \mathfrak{P}$. Now, $a/b$ is certainly identified as a $\mathfrak{P}$-adic zero for such prime ideal $\mathfrak{P}$ if $\zeta_i = \xi_i$, which by uniqueness of $\zeta_i$ is the case if and only if $\xi_i \theta_i^{a-b\ell} \equiv 1 \bmod \mathfrak{P}$. It suffices therefore, to show there is a positive density of such prime ideals $\mathfrak{P}$ for which there is integer $0 \leq \ell \leq N-1$ with $\xi_i \theta_i^{a-b\ell} \equiv 1 \bmod \mathfrak{P}$ for all $i = 1, \dots, r$. 

Now, let $M = \prod_i \mathrm{ord}(\xi_i)$. If $b=1$, define $c=1$, otherwise if $b>1$, pick $c$ to be an integer such that $\gcd(c,M) =1$ and $c \equiv -a \bmod b$. Indeed, use the Chinese Remainder Theorem to find $c$ satisfying $c \equiv -a \bmod b$ and $c \equiv 1 \bmod q$ for all primes $q \mid M$ such that $q \nmid b$. Then for any prime $q' \mid M$ satisfying $q' \mid b$, the fact that $\gcd(a,b) = 1$ ensures that $q' \nmid c$, so indeed $\gcd(c,M) = 1$ as required.

Now pick $\eta_i$ to be a root of unity such that $\eta_i^c = \xi_i$ for all $i$. Note that $\eta_i \in \mathbb{L}$, as it is just some power of $\xi_i$, since $\gcd(c,M) =1$. Let $G = \langle 
\theta_1/\eta_1, \dots, \theta_r/\eta_r\rangle \leq \mathbb{L}^\times$. We prove $G$ is torsion-free; suppose there are integers $n_1, \dots, n_r$ such that $(\theta_1/\eta_1)^{n_1}\cdots (\theta_r/\eta_r)^{n_r} = \zeta$ for some root of unity $\zeta$. This implies that
\begin{align*}
((\theta_1/\eta_1)^{n_1}\cdots (\theta_r/\eta_r)^{n_r})^{bM\mathrm{ord}(\zeta)} = \mu_1^{n_1 M\mathrm{ord}(\zeta)} \cdots \mu_r^{n_r M \mathrm{ord}(\zeta)} = 1 \, ,
\end{align*}
and by multiplicative independence of the $\mu_1, \dots, \mu_r$ this means $n_1 M \mathrm{ord}(\zeta) = \dots = n_r M \mathrm{ord}(\zeta) = 0$ so $n_1 = \dots = n_r = 0$ and $\zeta =1$, so $G$ is indeed torsion-free (and clearly of positive rank, as e.g. $\theta_1/\eta_1$ is not a root of unity).

Therefore, we may apply \cref{thm:sgobba} to conclude that there is a positive density of prime ideals $\mathfrak{P} \subseteq \mathcal{O}_\mathbb{L}$ such that $\mathrm{ord}_\mathfrak{P}(G)$ is coprime to $bM$, and so $\mathrm{ord}_\mathfrak{P}(\theta_i/\eta_i)$ is coprime to $bM$ for all $i$. Then by Chinese Remainder Theorem, we may find a solution to the congruences
\begin{align*}
    e \equiv 0 \mod  \prod_i \mathrm{ord}_\mathfrak{P}(\theta_i/\eta_i)\, , \quad e \equiv c \mod M\, , \quad e \equiv c \mod b
\end{align*}
and for such an integer $e$, we have $e = -a + b \ell$ for some integer $\ell$, and
\begin{align*}
    \theta_i^{-a+b\ell} &\equiv \theta_i^{e} \mod \mathfrak{P} \\
    &\equiv \eta_i^e \left( \frac{\theta_i}{\eta_i} \right)^e \mod \mathfrak{P} \\
    &\equiv \xi_i \mod \mathfrak{P} \, ,
\end{align*}
and hence $\xi_i \theta_i^{a-b\ell} \equiv 1 \bmod \mathfrak{P}$. Note that if $\ell' \equiv \ell \bmod N$, then $\theta_i^{b\ell} \equiv \theta_i^{b \ell'} \bmod \mathfrak{P}$, so we may pick $0 \leq \ell' \leq N-1$ such that $\ell' \equiv \ell \bmod N$ and so $\xi_i \theta_i^{a-b\ell'} \equiv 1 \bmod \mathfrak{P}$ as required.

<!-- lemma-6: body_md -->
\label{lem:even-ord}
Let $a \in \mathbb{Z}$, and let $p \in \mathbb{Z}$ be an odd prime not dividing $a$. If $\mathrm{ord}_p(a)$ is even, then $\exp(\frac{1}{2} \log a^{\mathrm{ord}_p(a)}) = -a^{\mathrm{ord}_p(a)/2}$.
<!-- end lemma-6 -->
<!-- proof-8: body_md -->
By \cref{lem:root-of-unity-unique} we have $\exp(\frac{1}{2} \log a^{\mathrm{ord}_p(a)}) = \pm a^{\mathrm{ord}_p(a)/2}$. If $\exp(\frac{1}{2} \log a^{\mathrm{ord}_p(a)}) = a^{\mathrm{ord}_p(a)/2}$, then $a^{\mathrm{ord}_p(a)/2} \equiv 1 \bmod p$, contradicting the definition of $\mathrm{ord}_p(a)$ as the minimal integer for which $a^{\mathrm{ord}_p(a)} \equiv 1 \bmod p$.

<!-- example-2: body_md -->
\label{ex:bad-TRZ-example}
Let 
\begin{align*}
    u_n = (1+2^n)(1 + (-4)^n) = 1 + 2^n +(-4)^n + (-8)^n \, .
\end{align*}
Clearly, $u$ is non-degenerate, though it is weakly degenerate as $(-4) \cdot 2^{-2} = -1$, so the multiplicative group $\langle 2, -4, -8\rangle \subseteq \mathbb{Q}^\times$ has torsion. Furthermore, we observe that $u_0 = 4 \neq 0$.

We show that 0 is a $p$-adic zero of $u$ for every $p>2$. We split into 2 cases.

\textbf{Case 1:} $\mathrm{ord}_p(2)$ is even. Let $N = \mathrm{ord}_p(2)$. Then $(-4)^{N} \equiv (2^{N})^2 \equiv 1 \bmod p$. Also, by \cref{lem:even-ord}, we have $\exp(\frac{1}{2} \log 2^N) = -2^{N/2}$. Consider the $N/2$-th $p$-adic interpolant
\begin{align*}
    F_{N/2}(x) = (1 + 2^{N/2} \exp(x \log 2^N)) ( 1 + (-4)^{N/2} \exp(x \log (-4)^N)) \, ,
\end{align*}
and compute that
\begin{align*}
    F_{N/2}(-1/2) &= (1 + 2^{N/2} \exp(-\frac{1}{2} \log 2^N)) ( 1 + (-4)^{N/2} \exp(-\frac{1}{2} \log (-4)^N)) \\
    &= (1 + 2^{N/2} (-2^{-N/2}))( 1 + (-4)^{N/2} \exp(-\frac{1}{2} \log (-4)^N)) \\
    &= 0 \, .
\end{align*}
Therefore, by \cref{lem:p-adic-zero-def-equal}, this identifies 0 as a $p$-adic zero. 

\textbf{Case 2:} $\mathrm{ord}_p(2)$ is odd. Then $(-4)^{\mathrm{ord}_p(2)} \equiv -(2^{\mathrm{ord}_p(2)})^2 \equiv -1 \bmod p$, so $\mathrm{ord}_p(-4) = 2\mathrm{ord}_p(2)$, so is even. Let $N = \mathrm{ord}_p(-4)$, then by \cref{lem:even-ord}, we have $\exp(\frac{1}{2} \log (-4)^N) = -(-4)^{N/2}$, and therefore we again have $F_{N/2}(-1/2)$, once again proving that 0 is a $p$-adic zero.

Consider the non-weakly-degenerate LRS
\begin{align*}
    u_n = -(4^n+1)(4^n-2)(4^n+2) = 4 + 4\cdot 4^n -16^n -64^n \, .
\end{align*}
There are at least two TRZs of $u$; we have vanishing tuples $(0,1,-1,1,-1)$ for the TRZ 0, and $(1/2,1,1,1,1)$ for the TRZ $1/2$, where we take $4^{1/2} \coloneq 2$. We show that for any prime $p > 2$, either $0$ or $\frac{1}{2}$ arises as a $p$-adic zero of $u$. 

Given prime number $p$, let $N = \mathrm{ord}_p(4)$. Then we have
\begin{align*}
    F_\ell(n) = -(4^\ell\exp(n\log4^N) +1)(4^\ell\exp(n\log 4^N)-2)) (4^\ell\exp(n \log 4^N) +2) \, .
\end{align*}
\textbf{Case 1:} $N$ is even. Then by \cref{lem:even-ord}, we have $\exp(\frac{1}{2} \log(4^N)) = -4^{N/2}$. Therefore we may compute
\begin{align*}
F_{N/2}(- 1/2) = -(-4^{N/2} \cdot 4^{-N/2} +1)(4^{N/2}\cdot 4^{-N/2} -2) (4^\ell \cdot 4^{-N/2} +2) = 0 \, .
\end{align*}
Since $N(-\frac{1}{2}) + \frac{N}{2} = 0$, this identifies $0$ as a $p$-adic zero.

\textbf{Case 2:} $N$ is odd. Then we have
\begin{align*}
    F_{\frac{N+1}{2}}(- 1/2) = -(4^{\frac{N+1}{2}} \exp(-\frac{1}{2} \log4^N) -1)(4^{\frac{N+1}{2}} \exp(-\frac{1}{2} \log 4^N)-2)) (4^{\frac{N+1}{2}} \exp(-\frac{1}{2} \log 4^N) +2) \, .
\end{align*}
Now, $\exp(\frac{1}{2} \log 4^N) = \pm 2^N$, so $4^{\frac{N+1}{2}} \exp(-\frac{1}{2} \log 4^N) = \pm 2$. Either choice of sign makes one of the factors $(4^{\frac{N+1}{2}} \exp(-\frac{1}{2} \log 4^N) - 2)$ or $(4^{\frac{N+1}{2}} \exp(-\frac{1}{2} \log 4^N) +2)$ vanish, so $F_{\frac{N+1}{2}}(-\frac{1}{2}) = 0$, and since $N(-\frac{1}{2}) + \frac{N+1}{2} = \frac{1}{2}$, this identifies $\frac{1}{2}$ as a $p$-adic zero. 


This proves the claim.

<!-- theorem-9: body_md -->
\label{thm:simultaneous}
Suppose $u,v$ are coprime non-weakly-degenerate $\mathbb{K}$-LRS. Then, assuming the $p$-adic Schanuel Conjecture, for any prime ideal $\mathfrak{p}$, the only common $\mathfrak{p}$-adic zeros of $u,v$ are rational.
<!-- end theorem-9 -->
<!-- theorem-10: body_md -->
\label{thm:sim-local-global}
Suppose $u,v$ are coprime non-weakly-degenerate $\mathbb{K}$-LRS as above. Then, subject to the $p$-adic Schanuel Conjecture, there exists $n \in \mathbb{Z}$ with $u_n = v_n = 0$ if and only if for all non-zero ideals $\mathfrak{a}$ of the ring $\mathcal{O}_S$, there exists $n \in \mathbb{Z}$ with $u_n \equiv v_n \equiv 0 \mod \mathfrak{a}$.

Furthermore, we may take $\mathfrak a = (\mathfrak{p} \mathfrak{q})^m$ for some integer $m \geq 1$ and prime ideals $\mathfrak{p},\mathfrak{q} \subseteq \mathcal{O}_\mathbb{K}$.
<!-- end theorem-10 -->
<!-- proof-9: body_md -->
One direction is trivial; if $u_n = v_n = 0$ then clearly for any ideal $\mathfrak{a}$ of $\mathcal{O}_S$, we have $u_n \equiv v_n \equiv 0 \mod \mathfrak{a}$.  

For the other direction, suppose there is no $n \in \mathbb{Z}$ with $u_n = v_n = 0$.


By \cref{thm:simultaneous}, for any prime ideal $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$, any $\mathfrak{p}$-adic zero shared between $u$ and $v$ is rational, so is in particular, a TRZ, by \cref{lem:rat-zero-is-TRZ}. Any shared TRZ must be a non-trivial TRZ of at least one of $u$ or $v$. Indeed, if $y$ is a trivial TRZ of both $u$ and $v$, then $(n-y)$ is a common factor of $u_n, v_n$, contradicting the hypothesis that $u,v$ are coprime. 

Let $Y_u$ be the set of all non-trivial TRZs of $u$ which are also TRZs of $v$ and not genuine integer zeros of $u$, and $Y_v$ be the analogous set for $v$. Define $Y_{u,1} = Y_u \setminus \mathbb{Z}$, $Y_{u,2} = Y_u \cap \mathbb{Z}$ and $Y_{v,1} = Y_v \setminus \mathbb{Z}$, $Y_{v,2} = Y_v \cap \mathbb{Z}$. Finally, define $Y_1 = Y_{u,1} \cup Y_{v,1}$, $Y_2 = Y_{u,2} \cup Y_{v,2}$ and $Y = Y_1 \cup Y_2$. Note that $Y$ is finite by \cref{thm:finitely-many-primitive-tuples}, and contains all the common TRZs of $u$ and $v$.

By \cref{thm:avoiding_TRZ}, there exist infinitely primes $\mathfrak{p} \subseteq \mathcal{O}_\mathbb{K}$ not lying in $S$ (as $S$ is finite) such that no element of $Y_{u,2}$ is a $\mathfrak{p}$-adic zero of $u$, and no element of $Y_{v,2}$ is a $\mathfrak{p}$-adic zero of $v$. 

Now, since each $y \in Y_2$ is not a $\mathfrak{p}$-adic zero of at least one of $u$ or $v$, this implies that there is a neighbourhood of $y$ on which the minimum valuation of $u$ or $v$ are bounded. Indeed, consider the case that $y \in Y_{u,2}$ (the case $y \in Y_{u,1}$ is symmetric). Then since $y$ is not a $\mathfrak{p}$-adic zero of $u$, there exist integers $r_y,m_y \geq 0$ such that for every $n \in \mathbb{Z}$, if $v_\mathfrak{p}(n-y) \geq r_y$, we have $v_\mathfrak{p}(u_n) < m_y$. Indeed, suppose to the contrary that for all $j \in \mathbb{N}$, there exists integer $n_j \in \mathbb{Z}$ with $v_\mathfrak{p}(n_j-y) \geq j$ and $v_\mathfrak{p}(u_{n_j}) \geq j$. Then $v_\mathfrak{p}(n_j-y) \to \infty$ and $v_\mathfrak{p}(u_{n_j}) \to \infty$ as $j \to \infty$; this is exactly the definition of $y$ being a $\mathfrak{p}$-adic zero of $u$, so this gives a contradiction.

Repeat this for every $y \in Y_2$, and let $m = \max\{m_y \mid y \in Y_2\}$ and $r = \max\{ r_y \mid y \in Y_2\}$. Then we conclude that whenever there is some $y \in Y_2$ for which $v_\mathfrak{p}(n-y) \geq r$, we have $\min \{ v_\mathfrak{p}(u_n), v_\mathfrak{p}(v_n)\} < m$.

Now, let $N = p^r$. Then for any $y \in Y_2$, there is a unique $0 \leq \ell_y \leq N-1$ for which there is $n_y \in \mathbb{Z}$ with $Nn_y + \ell_y = y$. Now, for each $y \in Y_2$, for every $n \in \mathbb{Z}$ we have $N n + \ell_y \equiv y \mod p^r$, so $v_\mathfrak{p}(Nn+\ell_y - y) \geq r$, and therefore $\min\{v_\mathfrak{p}(u_{Nn+\ell_y}) , v_\mathfrak{p}(v_{Nn+\ell_y})\} < m$. 

Thus, we only need to deal with the arithmetic progressions $\langle Nn+\ell \rangle_{n= -\infty}^\infty$ that do not contain any $y \in Y_2$. Let $L$ be the set of integers $0 \leq \ell \leq N-1$ such that the arithmetic progression $\langle Nn+\ell \rangle_{n= -\infty}^\infty$ does not contain any $y \in Y_2$. 

Crucially, for each such $\ell \in L$, each pair of subsequences $\langle u_{Nn+\ell} \rangle_{n=0}^\infty, \langle v_{Nn+\ell} \rangle_{n=0}^\infty$ for $\ell \in L$ has \emph{only} non-integer common TRZs, by definition of $L$. Therefore, by \cref{thm:avoiding_TRZ}, there exists infinitely many prime ideals $\mathfrak{q} \subseteq \mathcal{O}_\mathbb{K}$ such that every such subsequence $\langle u_{Nn+\ell} \rangle_{n=0}^\infty , \langle v_{Nn+\ell} \rangle_{n=0}^\infty$ for every $\ell \in L$ has no TRZ that is also a $\mathfrak{q}$-adic zero of the respective sequence -- choose such a $\mathfrak{q}$ distinct from $\mathfrak{p}$.

In the case that $n$ is $\mathfrak{q}$-adically close to a TRZ, we use the definition of $\mathfrak{q}$ to bound $\min\{v_\mathfrak{q}(u_n), v_\mathfrak{q}(v_n)\}$. By the same argument as earlier, there are integers $r',m' \geq 0$ such that for all $\ell \in L$, if $y'$ is a TRZ of the subsequence $ \langle u_{Nn+\ell} \rangle_{n=0}^\infty$, then for every $n \in \mathbb{Z}$, if $v_\mathfrak{q}(n - y') \geq r'$, then $\min\{v_\mathfrak{q}(u_{Nn+\ell}), v_\mathfrak{q}(v_{Nn+\ell})\} < m'$. Note that for any $y \in Y_u$, we have $\frac{y-\ell}{N}$ is a TRZ of the subsequence $\langle u_{Nn+\ell} \rangle_{n=0}^\infty$ (and analogously for $y \in Y_v$ and $\langle v_{Nn+\ell} \rangle_{n=0}^\infty$). Therefore, we conclude that if $v_\mathfrak{q}(n-y) \geq r'$, then $v_\mathfrak{q}(\frac{n-\ell}{N} - \frac{y-\ell}{N}) \geq r'$ and so $\min\{v_\mathfrak{q}(u_n),v_\mathfrak{q}(v_n)\} < m'$.

Now, in the case where $n$ is $\mathfrak{q}$-adically far from any TRZ $y \in Y$, we use the fact that by \cref{thm:simultaneous}, the only common $\mathfrak{q}$-adic zeros of $u,v$ are TRZs, and hence in $Y$. We claim that there is an integer $\tilde m$ such that for all $n \in \mathbb{Z}$, if $v_\mathfrak{q}(n - y) \leq r'$ for all $y \in Y$, we have $\min\{v_\mathfrak{q}(u_n),v_\mathfrak{q}(v_n) \} < \tilde m$. Indeed, suppose no such $\tilde m$ existed, then let $q$ be the rational prime lying below $\mathfrak{q}$ and define the set
\begin{align*}
    B = \bigcap_{y \in Y} \{x \in \mathbb{Z}_q : v_\mathfrak{q}(x-y) \leq r' \} \, .
\end{align*}
Since $B$ is a finite intersection of compact sets, $B$ is compact. Now, by assumption there must exist a sequence of integers $n_j \in \mathbb{Z}$ such that $n_j \in B$ and $\min\{v_\mathfrak{q}(u_{n_j}),v_\mathfrak{q}(v_{n_j})\} \geq j$ for each $j \in \mathbb{N}$. By compactness of $B$, there exists a convergent subsequence $\langle n_j\rangle_{j \in J}$, where $J \subseteq \mathbb{N}$, converging to some $x \in B$. Therefore, we have $v_\mathfrak{q}(n_j - x) \to \infty$ and $\min\{v_\mathfrak{q}(u_{n_j}),v_\mathfrak{q}(v_{n_j})\} \to \infty$, as $j \to \infty$ for $j \in J$. This means $x$ is a $\mathfrak{q}$-adic zero of both $u$ and $v$, yet does not lie in $Y$ by definition of $B$. This contradicts the fact that $Y$ contains all possible common $\mathfrak{q}$-adic zeros of $u$ and $v$, and this proves the claim.

This covers all cases. Indeed, for all $\ell \not\in L$, we have $\min\{v_\mathfrak{p}(u_{Nn+\ell}),v_\mathfrak{p}(v_{Nn+\ell})\} < m $. For all $\ell \in L$, we have $\min\{ v_\mathfrak{q}(u_{Nn+\ell}) , v_\mathfrak{q}(v_{Nn +\ell}) \} < \max\{m', \tilde m\}$. Letting $M = \max\{m,m',\tilde m\}$, we conclude that $(u_n, v_n) \not\equiv (0,0) \mod (\mathfrak{p} \mathfrak{q})^M$ for all $n \in \mathbb{Z}$.
