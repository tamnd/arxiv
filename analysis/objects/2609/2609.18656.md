<!-- assumption-1: body_md -->
\label{ass:oracle}
The objective satisfies $f\ge f_*>-\infty$. For every worker and all $x,y$, we have
\begin{align}
\mathbb E_\xi g_j(x;\xi)&=\nabla f_j(x),\\
\mathbb E_\xi\lVert g_j(x;\xi)\rVert_2^2&\le H^2,\label{eq:oracle-moment}\\
\mathbb E_\xi\lVert g_j(x;\xi)-g_j(y;\xi)\rVert_2^2
&\le L^2\lVert x-y\rVert_2^2.\label{eq:ms-smooth}
\end{align}
<!-- end assumption-1 -->
<!-- head-1: body_md -->
Here, Jensen's inequality implies that $f_j$ and $f$ are also $L$-smooth.
Besides, inequality~(\ref{eq:oracle-moment}) can be replaced with bounded oracle variance $\sigma^2$ and a true-gradient bound
$\lVert \nabla f_j(x)\rVert_2\le G_2$, which imply inequality~(\ref{eq:oracle-moment}) with
$H^2=\sigma^2+G_2^2$.
<!-- end head-1 -->
<!-- assumption-2: body_md -->
\label{ass:compressor}
For every input $v$, the output $Q(v)$ satisfies
\begin{equation}
\mathbb E[Q(v)\mid v]=v,\qquad
\mathbb E[\lVert Q(v)-v\rVert_2^2\mid v]\le\omega\lVert v\rVert_2^2.
\label{eq:relative}
\end{equation}
<!-- end assumption-2 -->
<!-- head-2: body_md -->
The general compression assumption includes the usual scaled stochastic
sign. For $v\ne0$, let $\rho(v)=\lVert v\rVert_\infty$ and draw independent signs with
$\Pr(S_k=1\mid v)=(1+v_k/\rho(v))/2$. Then
\begin{equation}
Q(v)=\rho(v)S,\qquad Q(0)=0,
\label{eq:scaled-sign}
\end{equation}
is unbiased and satisfies $1+\omega=d$ since $\mathbb E[\lVert Q(v)-v\rVert_2^2\mid v]
=d\lVert v\rVert_\infty^2-\lVert v\rVert_2^2\le(d-1)\lVert v\rVert_2^2$.
<!-- end head-2 -->
<!-- assumption-3: body_md -->
\label{ass:finite-sum}
The average objective is lower bounded and each component satisfies
\begin{equation}
\lVert \nabla f_{j,i}(x)-\nabla f_{j,i}(y)\rVert_2\le L\lVert x-y\rVert_2
\quad\text{for all }x,y,j,i.
\label{eq:fs-smooth}
\end{equation}
<!-- end assumption-3 -->
<!-- proposition-1: body_md -->
\label{prop:majority-floor}
There exist three lower-bounded smooth one-dimensional local objectives
with gradients bounded by $G=1$ such that SSVR-MV Option~1,
initialized at a global minimizer and using exact gradients, satisfies
\begin{equation}
\liminf_{K\to\infty}\mathbb E|f'(x_\tau)|\ge\frac1{1542}>0
\label{eq:majority-floor-main}
\end{equation}
for $R=4$ and step size
$\eta=\Theta(K^{-1/2})$, where $\tau$ is independent and uniform on $\{1,\ldots,K\}$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Take
\[
f_1(x)=f_2(x)=\tfrac12\log\cosh x+\tfrac14x,
\qquad f_3(x)=\tfrac12\log\cosh x-\tfrac12x.
\]
Their average is $f(x)=\tfrac12\log\cosh x$, minimized at $x=0$.
At this point the worker sign means are $(u,u,-2u)$ with $u=1/16$,
so \eqref{eq:majority-identity} gives a nonzero expected vote despite
$f'(0)=0$. Appendix~\ref{app:majority} proves that this bias produces
the stated gradient floor.
<!-- end proof-1 -->
<!-- theorem-1: body_md -->
\label{thm:master}
Under Assumptions~\ref{ass:oracle} and~\ref{ass:compressor}, for any $\eta>0$, $\beta\in(0,1]$, and integers
$K,B_0\ge1$, the DVR-Sign variant of Algorithm~\ref{alg:ucsvr} satisfies
\begin{align}
E_1&\le cH^2/B_0,\qquad
E_t\le(1-\beta)E_{t-1}+2c(\beta^2H^2+L^2\eta^2d)\quad(t\ge2),\label{eq:tracking}\\
\overline E_K&\le
\frac{cH^2}{B_0\beta K}
+2c\left(H^2\beta+\frac{L^2\eta^2d}{\beta}\right),\label{eq:average-error}\\
\mathbb E\lVert \nabla f(x_\tau)\rVert_1
&\le\frac{\Delta_f}{\eta K}+\frac{L\eta d}{2}
+2\sqrt{d\,\overline E_K}.
\label{eq:master}
\end{align}

<!-- corollary-1: body_md -->
\label{cor:rate}
Under the assumptions of Theorem~\ref{thm:master}, choose
\begin{align}
u_1=\frac1{\sqrt K+c^{1/3}K^{2/3}},\qquad
\beta=u_1,\quad \eta=\frac{Hu_1}{L\sqrt d},\quad
B_0=\left\lceil\frac1{u_1^2K}\right\rceil.
\label{eq:rate-parameters0}
\end{align}
Then DVR-Sign satisfies
\begin{align}\label{eq:rate-l1}
\begin{split}
    \mathbb E\lVert \nabla f(x_\tau)\rVert_1
\le & \sqrt d\left[
\left(\frac{L\Delta_f}{H}+\frac H2\right)K^{-1/2}
+\left(\frac{L\Delta_f}{H}+2\sqrt5H\right)
  \left(\frac cK\right)^{1/3}\right]\\
  = & O\!\left(\sqrt{\frac dK}

\end{split}
\end{align}

\label{eq:fs-master}

<!-- lemma-1: body_md -->
\label{lemma:fs-tracking}
For the estimator~\eqref{eq:fs-recursion}, the error is zero at each
refresh $t_0=1+kq$. For $t_0\le t<t_0+q$, we have
\begin{equation}
\mathbb E\lVert z_t-\nabla f(x_t)\rVert_2^2
\le\frac{aL^2}{n}\sum_{s=t_0+1}^{t}
                             \mathbb E\lVert x_s-x_{s-1}\rVert_2^2.
\label{eq:fs-local-error}
\end{equation}
Deterministic sign updates therefore give the bound
$aL^2\eta^2d(q-1)/n$. For updates $x_{t+1}=x_t-\eta Q(z_t)$,
with a fresh common downlink satisfying the same compressor assumption,
\begin{equation}
\frac1K\sum_{t=1}^K\mathbb E\lVert z_t-\nabla f(x_t)\rVert_2^2
\le\frac{a^2L^2\eta^2(q-1)}{n}
                         \frac1K\sum_{t=1}^K\mathbb E\lVert z_t\rVert_2^2.
\label{eq:fs-q-average-error}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
\leavevmode
At an ordinary iteration, condition on the history through $x_t$ and
before the current component indices and compression calls. Denote this
conditional expectation by $\mathbb E_t$; the points $x_t,x_{t-1}$ and the
previous estimate $z_{t-1}$ are fixed under this conditioning.
Write
\begin{align*}
X_j&=Q\bigl(\nabla f_{j,i_t^j}(x_t)
                   -\nabla f_{j,i_t^j}(x_{t-1})\bigr),\\
\mu_j&=\mathbb E_tX_j=\nabla f_j(x_t)-\nabla f_j(x_{t-1}).
\end{align*}
Uniform component sampling and unbiased compression give the expression
for $\mu_j$. The same component is evaluated at both points. Thus
\begin{align}
\mathbb E_t\lVert X_j\rVert_2^2
\le \frac a m\sum_{i=1}^m
       \lVert \nabla f_{j,i}(x_t)-\nabla f_{j,i}(x_{t-1})\rVert_2^2
\nonumber\le aL^2\lVert x_t-x_{t-1}\rVert_2^2.
\label{eq:fs-candidate-moment}
\end{align}
The first inequality uses~\eqref{eq:compressor-second}; the second is
component smoothness. Since $X_j-\mu_j$ is centered,
\[
\mathbb E_t\lVert X_j-\mu_j\rVert_2^2
=\mathbb E_t\lVert X_j\rVert_2^2-\lVert \mu_j\rVert_2^2
\le aL^2\lVert x_t-x_{t-1}\rVert_2^2.
\]
Note that the component indices and compressor calls are conditionally independent
across all $n$ workers. For $j\ne k$, this independence and centering imply
\[
\mathbb E_t\langle X_j-\mu_j,X_k-\mu_k\rangle
=\langle\mathbb E_t(X_j-\mu_j),\mathbb E_t(X_k-\mu_k)\rangle=0.
\]
Consequently,
\begin{align}
\mathbb E_t\left\|\frac1n\sum_{j=1}^n(X_j-\mu_j)\right\|^2
=\frac1{n^2}\sum_{j=1}^n\mathbb E_t\lVert X_j-\mu_j\rVert_2^2
\nonumber \le\frac{aL^2}{n}\lVert x_t-x_{t-1}\rVert_2^2.
\label{eq:fs-worker-variance}
\end{align}
The averaging is over all workers and
$n^{-1}\sum_j\mu_j=\nabla f(x_t)-\nabla f(x_{t-1})$ exactly.

Next, let $e_t=z_t-\nabla f(x_t)$. The estimator recursion gives
\[
e_t=e_{t-1}+\frac1n\sum_{j=1}^n(X_j-\mu_j).
\]
The new sum has zero conditional mean, and $e_{t-1}$ is fixed under
$\mathbb E_t$. Expanding the square yields
\begin{align*}
\mathbb E_t\lVert e_t\rVert_2^2
&=\lVert e_{t-1}\rVert_2^2
 +2\left\langle e_{t-1},\frac1n\sum_{j=1}^n
                                     \mathbb E_t(X_j-\mu_j)\right\rangle
 +\mathbb E_t\left\|\frac1n\sum_{j=1}^n(X_j-\mu_j)\right\|^2\\
&\le\lVert e_{t-1}\rVert_2^2+\frac{aL^2}{n}\lVert x_t-x_{t-1}\rVert_2^2.
\end{align*}
At a refresh, $z_{t_0}=\nabla f(x_{t_0})$, so $e_{t_0}=0$ on every
sample path. Taking total expectations and applying the recursion
successively at $t_0+1,\ldots,t$ gives
\[
\mathbb E\lVert e_t\rVert_2^2
\le\mathbb E\lVert e_{t_0}\rVert_2^2
       +\frac{aL^2}{n}\sum_{s=t_0+1}^t
                                     \mathbb E\lVert x_s-x_{s-1}\rVert_2^2.
\]
This proves~\eqref{eq:fs-local-error}. For DVR-Sign-FS,
each summand is $\eta^2d$, and there are at most $q-1$ summands.

For DVR-Q-FS, condition on the history $\mathcal G_s$ after all uplinks
in iteration $s$ and before the server's new compression call. Then
\[
\mathbb E[\lVert x_{s+1}-x_s\rVert_2^2\mid\mathcal G_s]
=\eta^2\mathbb E[\lVert Q(z_s)\rVert_2^2\mid\mathcal G_s]
\le a\eta^2\lVert z_s\rVert_2^2.
\]
Substituting its total expectation in~\eqref{eq:fs-local-error} yields
\[
\mathbb E\lVert e_t\rVert_2^2
\le\frac{a^2L^2\eta^2}{n}
                   \sum_{s=t_0}^{t-1}\mathbb E\lVert z_s\rVert_2^2.
\]
Let $t_1=\min\{t_0+q-1,K\}$ be the last iteration in this interval
between refreshes. Reversing the order of the finite sums gives
\begin{align*}
\sum_{t=t_0}^{t_1}\mathbb E\lVert e_t\rVert_2^2
&\le\frac{a^2L^2\eta^2}{n}
           \sum_{s=t_0}^{t_1-1}(t_1-s)\mathbb E\lVert z_s\rVert_2^2\\
&\le\frac{a^2L^2\eta^2(q-1)}{n}
           \sum_{s=t_0}^{t_1}\mathbb E\lVert z_s\rVert_2^2.
\end{align*}
These intervals are disjoint. Summing over them and dividing by $K$
proves~\eqref{eq:fs-q-average-error}.

The deterministic sign $s_t=\operatorname{Sign}(z_t)$ has squared norm $d$.
Thus the deterministic sign case of Lemma~\ref{lemma:fs-tracking}
proves~\eqref{eq:fs-error}. To translate this into stationarity,
smoothness gives
\[
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),\operatorname{Sign}(z_t)\rangle
                                  +\frac{L\eta^2d}{2}.
\]
By~\eqref{eq:alignment},
\[
\langle\nabla f(x_t),\operatorname{Sign}(z_t)\rangle
\ge\lVert \nabla f(x_t)\rVert_1
                    -2\sqrt d\lVert z_t-\nabla f(x_t)\rVert_2.
\]
Substituting and rearranging before taking expectations gives
\[
\eta\mathbb E\lVert \nabla f(x_t)\rVert_1
\le\mathbb E f(x_t)-\mathbb E f(x_{t+1})
 +2\eta\sqrt d\mathbb E\lVert z_t-\nabla f(x_t)\rVert_2+\frac{L\eta^2d}{2}.
\]
Sum this inequality over $t=1,\ldots,K$. The objective terms telescope
to $f(x_1)-\mathbb E f(x_{K+1})\le f(x_1)-f_*=\Delta_f\le\Delta_0$.
After division by $\eta K$, we obtain
\begin{align*}
\frac1K\sum_{t=1}^K\mathbb E\lVert \nabla f(x_t)\rVert_1
&\le\frac{\Delta_0}{\eta K}+\frac{L\eta d}{2}
       +\frac{2\sqrt d}{K}\sum_{t=1}^K
                              \mathbb E\lVert z_t-\nabla f(x_t)\rVert_2\\
&\le\frac{\Delta_0}{\eta K}+\frac{L\eta d}{2}
       +2\sqrt d\sqrt{\frac{aL^2\eta^2d(q-1)}n}\\
&=\frac{\Delta_0}{\eta K}
       +L\eta\left[\frac d2+2d\sqrt{\frac{a(q-1)}n}\right].
\end{align*}
In the second line, $\mathbb E\lVert z_t-\nabla f(x_t)\rVert_2
\le\sqrt{\mathbb E\lVert z_t-\nabla f(x_t)\rVert_2^2}$ and
Lemma~\ref{lemma:fs-tracking} bound each summand. Uniformity of $\tau$ implies that the average on the left equals
$\mathbb E\lVert \nabla f(x_\tau)\rVert_1$. This proves~\eqref{eq:fs-master}.

$L\eta J_1=\epsilon/2$. If $\Delta_0>0$, the choice of $K$ gives

\frac{\Delta_0}{\eta K}
=\frac{2L\Delta_0J_1}{\epsilon K}\le\frac\epsilon2.

Condition on the post-uplink history $\mathcal G_t$, which includes
$x_t$ and $z_t$ and precedes the server's new call to $Q$.
Unbiasedness and the second-moment bound give
\[
\mathbb E[Q(z_t)\mid\mathcal G_t]=z_t,\qquad
\mathbb E[\lVert Q(z_t)\rVert_2^2\mid\mathcal G_t]\le a\lVert z_t\rVert_2^2.
\]
Since $x_{t+1}=x_t-\eta Q(z_t)$, smoothness implies
\[
\mathbb E[f(x_{t+1})\mid\mathcal G_t]
\le f(x_t)-\eta\langle\nabla f(x_t),z_t\rangle
                                      +\frac{La\eta^2}{2}\lVert z_t\rVert_2^2.
\]
The inner product has the exact decomposition
\[
2\langle\nabla f(x_t),z_t\rangle
=\lVert \nabla f(x_t)\rVert_2^2+\lVert z_t\rVert_2^2
                            -\lVert z_t-\nabla f(x_t)\rVert_2^2.
\]
Substitute this identity, take total expectations, and rearrange:
\begin{align*}
\frac\eta2\mathbb E\lVert \nabla f(x_t)\rVert_2^2
+\frac\eta2(1-La\eta)\mathbb E\lVert z_t\rVert_2^2
\le \mathbb E f(x_t)-\mathbb E f(x_{t+1})
                +\frac\eta2\mathbb E\lVert z_t-\nabla f(x_t)\rVert_2^2.
\end{align*}
Sum the last inequality over $t=1,\ldots,K$ and divide by $\eta K/2$.
The objective values telescope, and $f(x_1)-\mathbb E f(x_{K+1})\le\Delta_0$.
Applying~\eqref{eq:fs-q-average-error} to the right side gives
\begin{align}
\frac1K\sum_{t=1}^K\mathbb E\lVert \nabla f(x_t)\rVert_2^2
&+\left[1-La\eta-\frac{a^2L^2\eta^2(q-1)}n\right]
                \frac1K\sum_{t=1}^K\mathbb E\lVert z_t\rVert_2^2
\nonumber\\
&\le\frac{2\Delta_0}{\eta K}.
\label{eq:fs-q-descent}
\end{align}
Write $s=\sqrt{(q-1)/n}\ge0$.
The choice in~\eqref{eq:fs-geometry-l2} gives
\[
La\eta+\frac{a^2L^2\eta^2(q-1)}n
=\frac1{2(1+s)}+\frac{s^2}{4(1+s)^2}
=\frac14+\frac1{4(1+s)^2}\le\frac12.
\]
Thus the bracket in~\eqref{eq:fs-q-descent} is at least $1/2$,
and we may discard its nonnegative contribution. Since
$2\Delta_0/(\eta K)=4L\Delta_0J_Q/K$, we obtain the squared-norm
bound in~\eqref{eq:fs-master-l2}. The independent uniform choice of
$\tau$ identifies its expectation with the averaged left side.
Cauchy--Schwarz then gives
\[
\mathbb E\lVert \nabla f(x_\tau)\rVert_2
\le\sqrt{\mathbb E\lVert \nabla f(x_\tau)\rVert_2^2}
\le2\sqrt{\frac{L\Delta_0J_Q}{K}}.
\]
