\label[lemma]{lem:macaulay-numerics}
\leavevmode\par
\begin{lemmaparts}
\item
\label{part:macaulay-balance}
Let $k, p, b$ be integers with $k\geq2$, $p\geq 1$, and $0\leq b<\partial_k(p)$. Then
\begin{equation*}
 \partial_{k-1}(b)+\partial_k(p-b)\geq \partial_k(p).
\end{equation*}

\item 
\label{part:macaulay-sum}
Let $k, p, b_1, \ldots, b_p$ be integers with $k\geq1$, $p\geq1$, and $b_i \geq 1$ for all $i$. Then
\begin{equation*}
 \sum_{i=1}^p \partial_k(b_i)
 \geq
 \partial_k\!\left(\sum_{i=1}^p b_i-p+1\right)+p-1.
\end{equation*}

\end{lemmaparts}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
\Cref{part:macaulay-balance} is \cite[Theorem~1]{AFML}. For \cref{part:macaulay-sum}, the case $p=1$ is an equality, so consider the case $p=2$. Let $b,c>0$. Set $\mathcal B$ and $\mathcal C$ to be the lexicographically smallest $b$ and $c$ degree-$k$ monomials in two sets of variables intersecting only in $z$, with $z$ the smallest variable in both orders. Their shadows have cardinalities $\partial_k(b)$ and $\partial_k(c)$, respectively. 

Then $\mathcal B\cap\mathcal C=\{z^k\}$, $\partial\mathcal B\cap\partial\mathcal C=\{z^{k-1}\}$, and $|\mathcal B\cup\mathcal C|=b+c-1$. Furthermore, we have $|\partial(\mathcal B\cup\mathcal C)| = \partial_k(b)+\partial_k(c)-1$. As $\partial_k$ is the minimal size of the shadow of a set of degree-$k$ monomials, we have
\begin{equation}
 \partial_k(b+c-1)
 \leq
 \partial_k(b)+\partial_k(c)-1.
 \label{eq:macaulay-pair}
\end{equation}
The $p\ge 3$ case follows by induction: set $c:=\sum_{i=1}^{p-1}b_i-p+2$, so that inductively we have
\[ \partial_k(c)
 \leq
 \sum_{i=1}^{p-1}\partial_k(b_i)-(p-2).
\]
Applying \cref{eq:macaulay-pair} to $c$ and $b_p$, we obtain
\[
 \begin{aligned}
 \partial_k\!\left(\sum_{i=1}^p b_i-p+1\right)
 &=
 \partial_k(c+b_p-1)\\
 &\leq
 \partial_k(c)+\partial_k(b_p)-1\\
 &\leq
 \sum_{i=1}^p \partial_k(b_i)-(p-1),
 \end{aligned}
\]
proving part (ii).

\label[lemma]{lem:separated-shadow}
Let $\Delta$ be a simplicial complex, and let $\mathcal{F}$ be a finite set of degree-$k$ $\Delta$-monomials for $k\geq2$. Suppose that there are $v_1,\ldots,v_p\in V(\Delta)$ satisfying
\[
 \operatorname{Nbd}_\Delta[v_i]\cap\operatorname{Nbd}_\Delta[v_j]=\varnothing
\]
for $i \neq j$, and that for every $v_i$, some monomial of $\mathcal{F}$ is divisible by $x_{v_i}$. Then
\begin{equation}
 |\partial\mathcal{F}|-\partial_k(|\mathcal{F}|)
 \geq
 p-\partial_k(p).
 \label{eq:separated-shadow-defect}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The $p=0$ case is Macaulay's theorem, so assume that $p\geq1$. As no face of $\Delta$ contains any two $v_i$, no monomial is divisible by two of the $v_i$, so $|\mathcal{F}|\geq p$. Set $N:=|\mathcal{F}|-p+1$. We prove the lower bound
\begin{equation*}
 |\partial\mathcal{F}|
 \geq
 \partial_k(N)+p-1
\end{equation*}
for the shadow. Once this is known, \cref{part:macaulay-sum} of \Cref{lem:macaulay-numerics} gives
\[
 \partial_k(|\mathcal{F}|)
 =
 \partial_k(N+p-1)
 \leq
 \partial_k(N)+\partial_k(p)-1.
\]
Combining these bounds gives
\begin{align*}
    |\partial\mathcal{F}|-\partial_k(|\mathcal{F}|) & \geq \partial_k(N)+p-1 - (\partial_k(N)+\partial_k(p)-1) = p - \partial_k (p).
\end{align*}
We now prove $|\partial\mathcal{F}| \geq \partial_k(N)+p-1$. For $1\leq i\leq p$, define $\mathcal{F}_i:=\{m\in\mathcal{F}:x_{v_i}\mid m\}$ and $\mathcal{L}_i:=\{m/x_{v_i}:m\in\mathcal{F}_i\}$, and let $\mathcal{F}_0 := \mathcal{F}\setminus\bigcup_{i=1}^p\mathcal{F}_i$.
The assumption $\operatorname{Nbd}_\Delta[v_i]\cap\operatorname{Nbd}_\Delta[v_j]=\varnothing$ implies that we have a disjoint decomposition $\mathcal{F} = \bigsqcup_{i=0}^p \mathcal{F}_i$.
Write $a_i:=|\mathcal{F}_i|$ for all $0 \le i \le p$, so that $a_i=|\mathcal{F}_i|=|\mathcal{L}_i|$ for all $i \geq 1$. Set $A:=\sum_{i=1}^p a_i$, so that $|\mathcal{F}|=a_0+A$. 

Next, we obtain two lower bounds for $|\partial\mathcal{F}|$. First, $\mathcal{L}_i\subseteq\partial\mathcal{F}$ for every $i$, and the assumption $\operatorname{Nbd}_\Delta[v_i]\cap\operatorname{Nbd}_\Delta[v_j]=\varnothing$ implies that $\mathcal{L}_1,\ldots,\mathcal{L}_p$ are pairwise disjoint. Every monomial in $\mathcal{L}_i$ has degree $k-1\geq 1$, and thus
\begin{equation*}
 |\partial\mathcal{F}|\geq A.
\end{equation*}

Second, the sets $\partial\mathcal{F}_0, x_{v_1}\partial\mathcal{L}_1,\ldots, x_{v_p}\partial\mathcal{L}_p$ are pairwise disjoint subsets of $\partial\mathcal{F}$: no monomial in $\partial \mathcal{F}_0$ is divisible by any $x_{v_i}$, and the sets $x_{v_1}\partial\mathcal{L}_1,\ldots, x_{v_p}\partial\mathcal{L}_p$ are pairwise disjoint by the assumption on the neighborhoods of $v_i$. Macaulay's theorem then gives
\begin{align*}
 |\partial\mathcal{F}|
 &\geq
 |\partial\mathcal{F}_0|
 +
 \sum_{i=1}^p|\partial\mathcal{L}_i|\\
 &\geq
 \partial_k(a_0)
 +
 \sum_{i=1}^p\partial_{k-1}(a_i).
\end{align*}

Set $a:=A-p+1$, so that $N=a_0+a$. By \cref{part:macaulay-sum} of
\Cref{lem:macaulay-numerics}, we have
\[
 \sum_{i=1}^p\partial_{k-1}(a_i)
 \geq
 \partial_{k-1}(a)+p-1.
\]
These two lower bounds for $|\partial \mathcal{F}|$ give
\begin{equation}
 |\partial\mathcal{F}|
 \geq
 \max\bigl\{
   a,\,
   \partial_{k-1}(a)+\partial_k(a_0)
 \bigr\} + p-1.
 \label{eq:shadow-max-bound}
\end{equation}
Now, observe that the maximum in \cref{eq:shadow-max-bound} is at least $\partial_k(N)$: if $a\geq \partial_k(N)$, this is immediate, and if $a<\partial_k(N)$, then \cref{part:macaulay-balance} of \Cref{lem:macaulay-numerics} gives
\[
 \partial_{k-1}(a)+\partial_k(N-a)
 \geq
 \partial_k(N).
\]
Since $N-a=a_0$, the second expression inside the maximum in
\cref{eq:shadow-max-bound} is at least $\partial_k(N)$. Therefore
\[
 |\partial\mathcal{F}|
 \geq
 \partial_k(N)+p-1
\]
as desired.

\label[lemma]{mainbound}
    Let $W \subseteq R_k(\Delta)$ be a vector subspace, with $k\geq 2$. Let $f_1, \ldots, f_t$ be nonzero polynomials in $D(W)\subseteq R_{k-1}(\Delta)$, and let $\mu_i := \operatorname{in}_{\prec} (f_i)$ for $1 \le i \le t$. Assume that
    \begin{equation*}
        \operatorname{Nbd}_\Delta [\operatorname{supp} (\mu_i)] \cap \operatorname{Nbd}_\Delta [\operatorname{supp}(\mu_j)] = \varnothing
    \end{equation*}
    for $i \neq j$. Then
    \begin{equation*}
        \dim D(W) - \partial_{k} (\dim W) \geq t - \partial_k (t). 
    \end{equation*}
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Set $\mathcal{F}:= \operatorname{in}_{\prec} (W)$ and $ \mathcal{H}:=\operatorname{in}_{\prec} \bigl ( D(W)\bigr)$. By \Cref{lem:init}, we have $|\mathcal{F}|=\dim W$, $|\mathcal{H}|=\dim D(W)$ and $\partial\mathcal{F}\subseteq\mathcal{H}$, so that $\dim D(W) \geq |\partial\mathcal{F}| \geq \partial_k (|\mathcal{F}|) $. Thus
    \begin{equation}
    \label{secondsplit}
        \dim D(W) - \partial_{k} (\dim W) = \bigl [|\mathcal{H}| - |\partial \mathcal{F}| \bigr ] + \bigl [|\partial \mathcal{F}| - \partial_{k} (|\mathcal{F}|)\bigr ].
    \end{equation}

    As $f_i \in D(W)$, we have $\mu_i \in \mathcal{H}$ for all $i$. The monomials $\mu_i$ have positive degree and are pairwise distinct.
    After reordering, we may assume that $\{\mu_1, \ldots, \mu_p\} \subseteq \partial \mathcal{F}$, and $\{\mu_{p+1}, \ldots, \mu_t\} \subseteq \mathcal{H} \setminus \partial \mathcal{F}$.  For each $\mu_1, \ldots, \mu_p$, choose a vertex $v_i$ with $x_{v_i} \mid \mu_i$. Since $\mu_i\in\partial\mathcal{F}$, it divides some monomial of $\mathcal{F}$, which is consequently divisible by $x_{v_i}$. Moreover,
    \[
     \operatorname{Nbd}_\Delta[v_i]
     \subseteq\operatorname{Nbd}_\Delta[\operatorname{supp}(\mu_i)].
    \]
    By the neighborhood assumption, we may apply \Cref{lem:separated-shadow} with these $v_1,\ldots, v_p$ to get the lower bound
    \[
        |\partial \mathcal{F}| - \partial_k (|\mathcal{F}|) \geq p - \partial_{k} (p)
    \]
    for the second term in \cref{secondsplit}. As $\{\mu_{p+1}, \ldots, \mu_t\} \subseteq \mathcal{H} \setminus \partial \mathcal{F}$, we have $|\mathcal{H}| - |\partial \mathcal{F}| \geq t - p$, so that
    \begin{align*}
        \dim D(W) - \partial_{k} (\dim W) & \geq t - p + p - \partial_{k} (p) \\
        & \geq t - \partial_k (t),
    \end{align*}
    where we used $p\leq t$ and the monotonicity of $\partial_k$.

\label[lemma]{lem:distant-spheres}
Let $K\subseteq\mathbb R^d$ be a $C^1$ convex body, and let
$P_n\to K$ be a sequence of simplicial $d$-polytopes converging
in the Hausdorff metric. Let $\Delta_n:=\partial P_n$ and $1\leq k\leq\left\lfloor\frac d2\right\rfloor-1$. For all positive integers $s$ and $R$, and all sufficiently large $n$, there exist $s$ homologically full simplicial $(k-1)$-spheres $\gamma_{1,n},\ldots,\gamma_{s,n}\subseteq\Delta_n$ such that
\[
 \operatorname{dist}_{\Delta_n^{(1)}}(V(\gamma_{i,n}),V(\gamma_{j,n}))>R
\]
for $i \neq j$.
<!-- end lemma-6 -->
<!-- proof-4: body_md -->
Using the geometric construction in the proof of \cite[Theorem~4.5]{ANS} with sufficiently generic projections $\pi_i$ for $\ 1 \le i \le s$ onto $k$-dimensional subspaces, we get $s$ pairwise disjoint spheres $\{\eta_i\}_{i = 1}^s \subseteq\partial K$ defined as
\[
 \eta_i := K \cap \pi_i^{-1}(\partial \pi_i (K)).
\]
These spheres satisfy
\[
 K\cap H_y\subseteq \eta_i
\]
for all $y\in\eta_i$, where $H_y$ is the supporting hyperplane of $K$ at $y$. As $K$ is $C^1$, this hyperplane is unique.

Now, \cite[Lemmas~4.1--4.4]{ANS} imply that, for each $\eta_i$ and all large enough $n$, there exists a simplicial $(k-1)$-sphere $\gamma_{i,n}\subseteq\Delta_n$ close to $\eta_i$ in the Hausdorff metric that is homologically full in $\Delta_n$.

It remains to prove the assertion on the pairwise distance between the vertex sets $V(\gamma_{i,n})$. For a contradiction, suppose that this fails for infinitely many $n$. After passing to a subsequence, there are indices $i\neq j$, a fixed integer $1\leq m\leq R$, and paths $v_{0,n},\ldots,v_{m,n}$ of adjacent vertices with $v_{0,n}\in V(\gamma_{i,n})$ and $v_{m,n}\in V(\gamma_{j,n})$. Furthermore, $v_{r,n}\longrightarrow v_r$ for $0\leq r\leq m$, $v_0\in\eta_i$, and $v_m\in\eta_j$.

Now, for each $1\leq r\leq m$, let $H_{r,n}$ be a supporting hyperplane of $P_n$ containing the edge $[v_{r-1,n},v_{r,n}]$. Hausdorff convergence to $K$ then gives a supporting hyperplane $H_r$ of $K$ containing both $v_{r-1}$ and $v_r$. Since $K$ is $C^1$, the supporting hyperplane at each boundary point is unique, so $H_1=\cdots=H_m=H_{v_m}$. Thus $v_0\in K\cap H_{v_m}\subseteq\eta_j$, contradicting $v_0\in\eta_i$ and $\eta_i\cap\eta_j=\varnothing$.

Let $\mathcal{S}_k^a$ and $\mathcal{S}_{k+1}^a$ be the spaces of affine $k$- and $(k+1)$-stresses on $P$ respectively. Let $D_k:= D(\mathcal{S}_{k+1}^a) \subseteq \mathcal{S}_k^a$ be the span of the coordinate derivatives of $\mathcal{S}_{k+1}^a$ in $\mathcal{S}_k^a$. We then have $g_k = \dim \mathcal{S}_k^a \geq \dim D_k \geq \partial_{k+1}(\dim \mathcal{S}_{k+1}^a) = \partial_{k+1} (g_{k+1})$, where the second of the inequalities is Macaulay's theorem. We split the defect as
    \begin{equation}\label{deficitsplit}
        \delta_k = \bigl [ \dim \mathcal{S}_k^a - \dim D_k\bigr ] +\bigl [ \dim D_k - \partial_{k+1} (\dim \mathcal{S}_{k+1}^a) \bigr].
    \end{equation}

    By \Cref{lem:local-stress}, for each homologically full $(k-1)$-sphere $\gamma_i \subseteq \partial P, \ 1 \leq i \leq s$, we have a corresponding affine $k$-stress $\lambda_i \in \mathcal{S}_k^a$ that satisfies $\operatorname{vsupp} (\lambda_i) \subseteq \operatorname{Nbd}_\Delta [\gamma_i]$, so that $\operatorname{Nbd}_\Delta[\operatorname{vsupp}(\lambda_i)] \subseteq \operatorname{Nbd}_\Delta^2[\gamma_i]$. The separation assumption
    \[
        \operatorname{Nbd}_\Delta^2[\gamma_i] \cap \operatorname{Nbd}_\Delta^2[\gamma_j] = \varnothing
    \]
    for $i \neq j$ implies the $\lambda_i$ are linearly independent. 
    
    Let $L$ denote the span of $\{\lambda_1, \ldots, \lambda_s\}$ in $\mathcal{S}_k^a$. Set $t := \dim (L \cap D_k)$. We have an injection $L / (L \cap D_k) \hookrightarrow \mathcal{S}_k^a / D_k$, so we get the lower bound 
    \[ 
        \dim \mathcal{S}_k^a - \dim D_k \geq s - t
    \]
    for the first term in \cref{deficitsplit}.
    
    For the second term, let $\mu_i = \operatorname{in}_{\prec} (\lambda_i)$ for $1 \le i \le s$. As the $\lambda_i$ have disjoint supports, the initial monomial of every nonzero $f \in L$ is some $\mu_i$, so $\operatorname{in}_{\prec} (L \cap D_k) \subseteq \{\mu_1, \ldots, \mu_s\}$. By \Cref{lem:init}, we have $|\operatorname{in}_{\prec} (L \cap D_k)| = t$. The separation assumption on the spheres $\{\gamma_i\}_{i=1}^s$ and $\operatorname{supp}(\mu_i) \subseteq \operatorname{vsupp}(\lambda_i)$ then give
    \[
    \operatorname{Nbd}_\Delta [\operatorname{supp} (\mu_i)] \cap \operatorname{Nbd}_\Delta [\operatorname{supp} (\mu_j)] = \varnothing
    \]
    for $i \neq j$. Thus, we can apply \Cref{mainbound} with $W = \mathcal{S}_{k+1}^a$ and $f_1,\ldots, f_t \in L \cap D_k$ polynomials whose initial monomials are the $t$ distinct elements of $\operatorname{in}_{\prec} (L \cap D_k)$ to get 
    \[ 
        \dim D_k - \partial_{k+1} (\dim \mathcal{S}_{k+1}^a) \geq t - \partial_{k+1}(t),
    \]
    so that
    \begin{align*}
        \delta_k & \geq \bigl [ s - t \bigr ]+ \bigl [ t - \partial_{k+1}(t)\bigr ] \\
            & \geq s - \partial_{k+1}(s),
    \end{align*}
    where we used $t \le s$ and the monotonicity of $\partial_{k+1}$.

As $P_n$ converges to $K$ in the Hausdorff distance, \Cref{lem:distant-spheres} with $R=4$ supplies us with increasingly many homologically full $(k-1)$-spheres $\{\gamma_i\}_{i=1}^s$, all satisfying the hypothesis of \Cref{mainthm}. <!-- proposition-1: body_md -->
\label[proposition]{prop:matroid-chow}
Let $M$ be a rank-$r$ loopless matroid, and suppose that $H_1,\ldots,H_p$ are hyperplanes of $M$ such that
\[
 \operatorname{rk}(H_i\cap H_j)\leq 1
\]
for all $i \neq j$. Then, for every $2\leq k\leq r-2$,
\[
 a_{k-1}(M)-\partial_k\bigl(a_k(M)\bigr)
 \geq
 p-\partial_k(p).
\]
<!-- end proposition-1 -->
<!-- proof-7: body_md -->
Let $\mathcal{F}_k$ denote the degree-$k$ Feichtner--Yuzvinsky monomials, so that $|\mathcal{F}_k|=a_k(M)$ and $\partial\mathcal{F}_k\subseteq\mathcal{F}_{k-1}$. Let $\Delta$ be the simplicial complex whose faces are the supports of FY monomials. As a divisor of an FY-monomial is an FY-monomial, this is a simplicial complex, and by construction every monomial in $\mathcal{F}_k$ is a $\Delta$-monomial. Every face of $\Delta$ corresponds to a chain $F_1\subsetneq\cdots\subsetneq F_\ell$ of flats. 

Observe that as $k\leq r-2=\operatorname{rk}(H_i)-1$, $x_{H_i}^k$ belongs to $\mathcal{F}_k$ for every $i$, so for every $H_i$ there is a corresponding vertex $v_i$ of $\Delta$. We will apply \Cref{lem:separated-shadow} to this complex and the vertices $v_i$. 

First, we claim that the neighborhoods $\operatorname{Nbd}_\Delta [v_i]$ are pairwise disjoint. Suppose for a contradiction that there is a vertex $v$ in $\operatorname{Nbd}_\Delta [v_i] \cap \operatorname{Nbd}_\Delta [v_j]$. This corresponds to a flat $F$ that is comparable to both $H_i$ and $H_j$, with $i \neq j$. 

As $v$ is a vertex of $\Delta$, the variable $x_F$ occurs with positive exponent in an FY-monomial, and hence $\operatorname{rk}(F)\geq2$. Distinct hyperplanes are incomparable, so $F$ is distinct from $H_i$ and $H_j$. Furthermore, as $\operatorname{rk} (E) - \operatorname{rk} (H_i) = 1$, the rank condition for FY monomials does not allow $F = E$, so $F$ is a proper subset of both $H_i$ and $H_j$. Thus $F \subseteq H_i \cap H_j$, which implies $\operatorname{rk} (H_i \cap H_j) \geq 2$, contrary to the assumption $\operatorname{rk} (H_i \cap H_j) \leq 1$ for $i \neq j$.

Now, we can apply \Cref{lem:separated-shadow} to get
\[
 |\partial\mathcal{F}_k|-\partial_k(|\mathcal{F}_k|)
 \geq p-\partial_k(p).
\]
As $|\partial\mathcal{F}_k| \leq|\mathcal{F}_{k-1}|=a_{k-1}(M)$ and $|\mathcal{F}_k|=a_k(M)$, we get
\[
 a_{k-1}(M)-\partial_k\bigl(a_k(M)\bigr)
 \geq p-\partial_k(p).
\]
