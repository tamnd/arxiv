<!-- definition-1: body_md -->
\label{def:ai}
                         
For a metrizable space
$X$,
we say that a metric
$d\in\operatorname{Met}(X)$
has algebraically independent distances if,
for every positive integer
$r$,
every choice of distinct
$\{x_1,y_1\},\ldots,\{x_r,y_r\}\in[X]^2$,
and every nonzero polynomial
$P\in\mathbb Q[T_1,\ldots,T_r]$,
we have
\[
P(d(x_1,y_1),\ldots,d(x_r,y_r))\ne0.
\]
We denote by
$\mathrm{A}(X)$
the set of all such metrics.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:density}
                         
Let
$X$
be a strongly zero-dimensional metrizable space with
$|X|\le\mathfrak c$.
Then, for every
$d\in\operatorname{Met}(X)$
and every
$\varepsilon>0$,
there exists
$e\in\mathrm{A}(X)$
such that
$\mathcal{D}_X(d,e)<\varepsilon$.
If
$X$
is completely metrizable, then
$e$
can be chosen complete.
<!-- end theorem-1 -->

<!-- lemma-1: body_md -->
\label{lem:digits}
                         
Let
$b\ge 2$
and
$r\ge 1$
be integers.
Let
$\{k_j\}_{j\in\mathbb Z_{\ge 0}}$
be a strictly increasing sequence of positive integers satisfying
\begin{equation}\label{eq:gap}
\lim_{j\to\infty}\frac{k_{j+1}}{k_j}=\infty.
\end{equation}
Let
$\{\boldsymbol v_j\}_{j\in\mathbb Z_{\ge 0}}$
be a sequence in
$\{0,1\}^r$,
and define
\[
\mathcal V=\{\boldsymbol v\in\{0,1\}^r\mid \boldsymbol v_j=\boldsymbol v\text{ for infinitely many }j\}.
\]
If
$\operatorname{span}_{\mathbb R}\mathcal V=\mathbb R^r$,
then the coordinates of the vector
\[
\boldsymbol\xi=\sum_{j=0}^{\infty}b^{-k_j}\boldsymbol v_j
\]
are algebraically independent over
$\mathbb Q$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since the exponents are strictly increasing positive integers,
the series defining
$\boldsymbol\xi$
converges absolutely.
For the sake of contradiction,
suppose that a nonzero polynomial with rational coefficients vanishes at
$\boldsymbol\xi$.
Choose a nonzero polynomial
$P\in\mathbb Z[T_1,\ldots,T_r]$
of the least total degree among all such polynomials.
Write its degree as
$d$.
A nonzero constant cannot vanish,
and hence
$d\ge 1$.

                         
For every
$N\in\mathbb Z_{\ge 0}$,
define
\[
\boldsymbol s_N=\sum_{j=0}^{N}b^{-k_j}\boldsymbol v_j.
\]
Each coordinate of
$\boldsymbol s_N$
belongs to
$b^{-k_N}\mathbb Z$.
Consequently,
for every
$N\in\mathbb Z_{\ge 0}$,
we have
\begin{equation}\label{eq:denominator}
P(\boldsymbol s_N)\in b^{-dk_N}\mathbb Z.
\end{equation}
All the vectors
$\boldsymbol s_N$
and
$\boldsymbol\xi$
belong to a fixed compact box in
$\mathbb R^r$.
The gradient of
$P$
is bounded on that box.
Thus there exists a constant
$C>0$
such that,
for every
$N\in\mathbb Z_{\ge 0}$,
\begin{equation}\label{eq:tail}
\begin{split}
|P(\boldsymbol s_N)|
&=|P(\boldsymbol s_N)-P(\boldsymbol\xi)|
\le C\|\boldsymbol s_N-\boldsymbol\xi\|_{\infty}\\
&\le C\sum_{j>N}b^{-k_j}
\le \frac{Cb}{b-1}b^{-k_{N+1}}.
\end{split}
\end{equation}
The last inequality uses strict increase of the integer exponents.
By
\eqref{eq:gap},
we have
$k_{N+1}-dk_N\to\infty$.
Combining
\eqref{eq:denominator}
and
\eqref{eq:tail},
we obtain
$N_*\in\mathbb Z_{\ge 0}$
such that
\begin{equation}\label{eq:eventual-zero}
P(\boldsymbol s_N)=0\qquad(N\ge N_*).
\end{equation}
Indeed,
the absolute value of
$P(\boldsymbol s_N)$
is eventually smaller than the least possible positive value
$b^{-dk_N}$.

                         
Fix
$\boldsymbol v\in\mathcal V$.
Choose an increasing sequence
$\{N_\ell\}_{\ell\in\mathbb Z_{\ge 0}}$
such that
$N_\ell>N_*$
and
$\boldsymbol v_{N_\ell}=\boldsymbol v$
for every
$\ell\in\mathbb Z_{\ge 0}$.
For every
$\ell\in\mathbb Z_{\ge 0}$,
put
$\varepsilon_\ell=b^{-k_{N_\ell}}$.
Applying the chain rule and the fundamental theorem of calculus to the function
$t\mapsto P(\boldsymbol s_{N_\ell-1}+t\varepsilon_\ell\boldsymbol v)$
on
$[0,1]$,
and using
$\boldsymbol s_{N_\ell}=\boldsymbol s_{N_\ell-1}+\varepsilon_\ell \boldsymbol v$
and
\eqref{eq:eventual-zero},
we obtain
\begin{equation}\label{eq:derivative}
0=\frac{P(\boldsymbol s_{N_\ell})-P(\boldsymbol s_{N_\ell-1})}{\varepsilon_\ell}
=\int_0^1\nabla P(\boldsymbol s_{N_\ell-1}+t\varepsilon_\ell \boldsymbol v)\cdot \boldsymbol v\,dt.
\end{equation}
The vectors in the integrand converge to
$\boldsymbol\xi$
uniformly for
$t\in[0,1]$,
since
\[
\sup_{t\in[0,1]}\|\boldsymbol s_{N_\ell-1}+t\varepsilon_\ell \boldsymbol v-\boldsymbol\xi\|_\infty
\le \|\boldsymbol s_{N_\ell-1}-\boldsymbol\xi\|_\infty+\varepsilon_\ell\longrightarrow 0.
\]
Continuity of the polynomial gradient and
\eqref{eq:derivative}
therefore imply
$\nabla P(\boldsymbol\xi)\cdot \boldsymbol v=0$.
This holds for every
$\boldsymbol v\in\mathcal V$.
Since
$\mathcal V$
spans
$\mathbb R^r$,
we have
\[
\nabla P(\boldsymbol\xi)=0.
\]
Since
$P$
is nonconstant,
there exists
$i\in\{1,\ldots,r\}$
such that the polynomial
$Q=\partial P/\partial T_i$
satisfies
\[
Q\in\mathbb Z[T_1,\ldots,T_r]\setminus\{0\},
\qquad \deg Q<d,
\qquad Q(\boldsymbol\xi)=0.
\]
This contradicts the minimality of
$d$.
<!-- end proof-1 -->
<!-- remark-1: body_md -->
\label{rem:shiokawa}
                         
To deduce Lemma~\ref{lem:digits} from \cite[Corollary~1, p.~440]{Shiokawa1982},
choose a basis from the coefficient vectors that occur infinitely often.
An invertible rational change of coordinates sends these basis vectors to integer multiples of the standard basis vectors while keeping all coefficients integral.
Thus each transformed coordinate series has infinitely many exponents that occur in no other coordinate series.
Together with the gap condition \eqref{eq:gap},
this allows Shiokawa's criterion to be applied at
$z=1/b$,
yielding the conclusion of the lemma.
See also \cite[Example~1, p.~441]{Shiokawa1982} for an application to series with factorial exponents.

<!-- lemma-2: body_md -->
\label{lem:cuts}
                         
Let
$X$
be a strongly zero-dimensional metrizable space with
$2\le|X|\le\mathfrak c$.
Then there exists a countable Boolean algebra
$\mathcal B$
of clopen subsets of
$X$
such that,
for every finite set
$V\subseteq X$
and every
$H\subseteq V$,
there exists
$B\in\mathcal B$

Fix
$x\in X$.
Since both sequences converge on
$A$,
for every
$a\in A$
we obtain
\[
\limsup_{k\to\infty}\max\{d(f_k(x),f(x)),d(g_k(x),g(x))\}\le2d(x,a).
\]
The density of
$A$
therefore implies
\[
f_k(x)\longrightarrow f(x),\qquad g_k(x)\longrightarrow g(x).
\]
It remains to prove that
$f$
and
$g$
are inverse maps.
For each
$x\in X$,
applying \eqref{eq:approx-isometry} to
$g_k$
yields
\[
d(g_k(f_k(x)),g_k(f(x)))\le d(f_k(x),f(x))+2\eta_k\longrightarrow0.
\]
Since
$g_k(f(x))\to g(f(x))$
and
$g_k(f_k(x))=x$,
we obtain
$gf=\operatorname{id}_X$.
Interchanging
$f_k$
and
$g_k$
proves
$fg=\operatorname{id}_X$.
Thus
$f\in\operatorname{Isom}(X,d)$.
Condition~\ref{cond:rigid-basepoint} passes to the limit since
$f_k(o)\to f(o)$,
$g_k(o)\to g(o)=f^{-1}(o)$,
and
$K_n$
is closed.
For condition~\ref{cond:rigid-displacement},
uniform convergence and the reverse triangle inequality yield
\[
\begin{aligned}
&|d_k(a_j,f_k(a_j))-d(a_j,f(a_j))|\\
&\quad\le |d_k(a_j,f_k(a_j))-d(a_j,f_k(a_j))|
+|d(a_j,f_k(a_j))-d(a_j,f(a_j))|\\
&\quad\le\eta_k+d(f_k(a_j),f(a_j))\longrightarrow0.
\end{aligned}
\]
Since
$d_k(a_j,f_k(a_j))\ge2^{-m}$
for every
$k$,
we obtain
$d(a_j,f(a_j))\ge2^{-m}$.
Thus condition~\ref{cond:rigid-displacement} also holds.
Hence
$d\in \mathrm{E}_{j,m,n}$.
Equation~\eqref{eq:rigid-complement} proves the proposition.

<!-- proof-7: body_md -->
Every compatible metric on
$X$
is proper, so Proposition~\ref{prop:proper-gdelta} proves the
$G_\delta$
assertion.

$d\in\operatorname{Met}(X)$
and

$t>0$

$t\operatorname{diam}_d(X)<\varepsilon$.

$e\in\mathrm{R}(X)$

\]

<!-- proposition-3: body_md -->
\label{prop:tb-rigid}
                         
Let
$X$
be a metrizable space with
$|X|>2$.
Then, for every
$d\in\operatorname{Met}_{\mathrm{tb}}(X)$
and every
$\varepsilon>0$,
there exists
$e\in\operatorname{Met}_{\mathrm{tb}}(X)$
with
$\mathcal{D}_X(d,e)<\varepsilon$
such that the compact completion of
$(X,e)$
has trivial isometry group.
In particular,
$\mathrm{R}(X)\cap\operatorname{Met}_{\mathrm{tb}}(X)$
is dense in
$\operatorname{Met}_{\mathrm{tb}}(X)$.
<!-- end proposition-3 -->
<!-- proof-8: body_md -->
Let
$(K,\widehat d)$
be the compact completion of
$(X,d)$,
identifying
$X$
with its dense image.
By Theorem~\ref{thm:compact-rigid}, choose a compatible rigid metric
$\rho$
on
$K$
with
$\mathcal{D}_K(\widehat d,\rho)<\varepsilon$.
Put
$e=\rho|_{X^2}$.
Then
$e$
is compatible, totally bounded, and satisfies
$\mathcal{D}_X(d,e)<\varepsilon$.
Since
$\rho$
induces the topology of
$K$,
the set
$X$
is dense in
$(K,\rho)$,
which is the completion of
$(X,e)$.
Every bijective isometry of
$(X,e)$
extends to a bijective isometry of
$(K,\rho)$
by extending both the map and its inverse.
Rigidity of
$(K,\rho)$
implies rigidity of
$(X,e)$.
