<!-- definition-1: body_md -->
\label{DefGeneralGibbs}
Given a Borel measurable function~$h : \mathcal X\times\mathcal Y\to\mathbb{R}$; a~$\sigma$-finite measure~$Q$ on the measurable space~$(\mathcal Y, \mathscr{B}(\mathcal Y))$; and
a real~$\lambda \in \mathbb{R}\setminus\lbrace 0 \rbrace$, the probability measure~$P^{(h, Q, \lambda)}_{Y \mid X}\in \triangle\left(\mathcal Y \mid \mathcal X \right)$ is said to be an~$(h, Q, \lambda)$-Gibbs conditional probability measure if
\vspace{-3ex}
\begin{IEEEeqnarray}{rCl}
\label{EqGeneralGibbsIntegrability}
\relax{$\forall x \in \mathcal X$, } 0 < \int \exp\left( -\frac{1}{\lambda}h\left(x,y\right)\right)\mathrm{d}Q\left( y \right) < + \infty;
\end{IEEEeqnarray}
                                           
and for all~$\left(x, y \right) \in \mathcal X \times \operatorname{supp} Q$,
\vspace{-1ex}
\begin{IEEEeqnarray}{rcl}
\label{EqGeneralGibbsRN}
\frac{\mathrm{d}P_{Y\mid X = x}^{(h, Q, \lambda)}}{\mathrm{d}Q } \left(y\right) & = &\frac{\exp\left(- \frac{1}{\lambda} h\left(x,y\right)\right)}{ \int \exp\left( -\frac{1}{\lambda}h\left(x,y\right)\right)\mathrm{d}Q\left( y \right)}.
\end{IEEEeqnarray}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{DefRenormalization}
Consider $\alpha \in \mathbb{R}\setminus\lbrace 0 \rbrace$, two $\sigma$-finite measures $Q_1$ and $Q_2$, and a probability measure $P$, all on the same measurable space with $P \ll Q_1$ and $Q_2 \ll Q_1$. The $\left(\alpha,Q_1, Q_2\right)$-renormalization of $P$ is a probability measure,  denoted by $R$, such that for all $y \in \operatorname{supp} Q_2$,
\begin{IEEEeqnarray}{rCl}
\label{EqRenormalization}
\displaystyle \frac{\mathrm{d}{R}}{\mathrm{d}{Q_2}}\left(y\right)
&=&
\frac{\left(\displaystyle \frac{\mathrm{d}{P}}{\mathrm{d}{Q_1}}\left(y\right)\right)^{\alpha}}
{\displaystyle\int \left(\displaystyle \frac{\mathrm{d}{P}}{\mathrm{d}{Q_1}}\left(\tilde{y}\right)\right)^{\alpha}\mathrm{d}Q_2\left(\tilde{y}\right)},
\end{IEEEeqnarray}
provided that 
                          
$0
<
\int \left(\displaystyle \frac{\mathrm{d}{P}}{\mathrm{d}{Q_1}}\left(y\right)\right)^{\alpha}\mathrm{d}Q_2\left(y\right)
< + \infty.
$

\label{TheoremApril16at13h13in2026Antibes}
The $(\alpha, Q, Q_1)$-renormalization of the Gibbs probability measure $P^{(h,Q,\lambda)}_{Y\mid X=x}$ in \eqref{EqGeneralGibbsRN}, namely the measure $R_{1}$ in \eqref{EqApril15at17h43in2026Sophia}, satisfies for all $y \in \operatorname{supp} Q_1$, 
\begin{IEEEeqnarray}{rCl}
\label{EqApril16at13h03in2026Antibes}
\displaystyle \frac{\mathrm{d}{R_1}}{\mathrm{d}{Q_1}}\left(y\right) &=& \frac{\exp\left(-~\frac{\alpha}{\lambda} h(x,y)\right)}{\int \exp\left(-~\frac{\alpha}{\lambda} h(x,\widehat{y})\right) \mathrm{d}Q_1(\widehat{y})} =  \displaystyle \frac{\mathrm{d}{P^{(h,Q_1,\frac{\lambda}{\alpha})}_{Y\mid X=x}}}{\mathrm{d}{Q_1}}\left(y\right). \quad
\end{IEEEeqnarray}

\label{DefApril14at23h55in2026Antibes}
Consider $\alpha_1 \in \mathbb{R}\setminus\lbrace 0 \rbrace$, $\alpha_2 \in \mathbb{R}\setminus\lbrace 0 \rbrace$, two probability measures $P_1$ and $P_2$; and three $\sigma$-finite measures $Q$, $Q_1$ and $Q_2$, all on the same measurable space, such that $P_1 \ll Q_1$, $Q\ll Q_1$, $P_2 \ll Q_2$ and $Q\ll Q_2$. The $(\alpha_1, \alpha_2, Q_1, Q_2, Q)$-normalized log-linear combination of $P_1$ and $P_2$ is a probability measure, denoted by $S$, such that for all $y \in \operatorname{supp} Q$,
\begin{IEEEeqnarray}{rCl}
\label{EqApril15at18h35in2026Sophia}
\displaystyle \frac{\mathrm{d}{S}}{\mathrm{d}{Q}}\left(y\right)
