<!-- dfn-1: body_md -->
A location-scale family is a family of probability distributions parametrized by a location parameter $\boldsymbol{\mu}\in{\mathbb R}^d$ and a positive scale parameter $\sigma>0$. Suppose $Z$ is a random variable following distribution in this family corresponding to $\boldsymbol{\mu}={\bf 0}$ and $\sigma=1$ with cumulative distribution function (CDF) $F_0(z)$. Then $X=\boldsymbol{\mu}+\sigma Z$ has CDF $F(X) = F_0((X-\boldsymbol{\mu})/\sigma)$.
<!-- end dfn-1 -->
<!-- dfn-2: body_md -->
The \emph{exponential power distribution}, denoted as ${\mathcal P}_q(\boldsymbol{\mu}, \boldsymbol{\varSigma})$, has the following probability density
\begin{equation}
p({\bf x})= \frac{q\Gamma(\frac{d}{2})}{2\Gamma(\frac{d}{q})} 2^{-\frac{d}{q}}\pi^{-\frac{d}{2}} |\boldsymbol{\varSigma}|^{-\frac12} \exp\left\{-\frac{r^\frac{q}{2}}{2}\right\}, \quad r({\bf x}) = {({\bf x}-\boldsymbol{\mu})}^{\mathsf T} \boldsymbol{\varSigma}^{-1} ({\bf x}-\boldsymbol{\mu}) .
\end{equation}
<!-- end dfn-2 -->
<!-- rk-1: body_md -->
When $q=2$, the exponential power distribution reduces to multivariate normal distribution, i.e. ${\mathcal P}_2(\boldsymbol{\mu}, \boldsymbol{\varSigma}) = {\mathcal N}(\boldsymbol{\mu}, \boldsymbol{\varSigma})$.
Generally, the parameter $q$ controls the regularization around center and tail behavior: the smaller $q$, the sharper regularization and the heavier tail it induces, as illustrated in Figure \ref{fig:exponential_power}.
<!-- end rk-1 -->
<!-- prop-1: body_md -->
\label{prop:epd_prop}
For a $d$-dimensional random variable $X\sim {\mathcal P}_q(\boldsymbol{\mu}, \sigma^2{\bf I})$, we have
\begin{enumerate}[nosep]
\item $\mathrm E(X)=\boldsymbol{\mu}$, $\mathrm{Cov}(X) = s_{q,d}\boldsymbol{\varSigma}$ with a scaling factor $s_{q,d} = \frac{2^{\frac{2}{q}}\Gamma(\frac{d+2}{q})}{d\Gamma(\frac{d}{q})}$.
\item $X \overset{L}{=} \boldsymbol{\mu} + \sqrt{r} {\bf L} U_S$, where $r = {(X-\boldsymbol{\mu})}^{\mathsf T} \boldsymbol{\varSigma}^{-1} (X-\boldsymbol{\mu})$ and $r^{\frac{q}{2}} \sim \Gamma(\alpha=\frac{d}{q}, \beta=\frac12)$, ${\bf L}{{\bf L}}^{\mathsf T}=\boldsymbol{\varSigma}$, and $\sqrt{r}\perp U_S$ with $U_S\sim \mathcal{U}({\mathcal S}^{d-1})$ uniformly distributed on the sphere.
                                                                                                                                                             
                   
                 
                                                                                                                                        
                                                                                                                   
               
                 
                                                                                                                                                                                                     
\end{enumerate}

The geodesic equation under the hyperbolic metric $ds^2 = \frac{c_{\mu}\Vert d\boldsymbol{\mu}\Vert^2+c_{\sigma} d\sigma^2}{\sigma^2}$ \eqref{eq:fisher_metric} can be solved in either semi-circle parametrization or hyperbolic parametrization, which are detailed as follows.

\paragraph{Poincar\'e semi-circle parametrization}

The metric \eqref{eq:fisher_metric} can be rescaled to a standard Poincar\'e metric which leads to a semi-circle in the Poincar\'e upper half-plane.
Denote $\lambda = \sqrt{c_\sigma/c_\mu}$ and ${\bf e}= (\boldsymbol{\mu}_1-\boldsymbol{\mu}_0)/\Vert \boldsymbol{\mu}_1-\boldsymbol{\mu}_0\Vert_2$.
Use the scalar projection $x(t) = \langle \boldsymbol{\mu}(t)-\boldsymbol{\mu}_0, {\bf e}\rangle$ and rescale $y(t) = \lambda \sigma(t)$. 
The metric \eqref{eq:fisher_metric} reduces to
\begin{equation*}
    ds^2 = \frac{c_{\sigma}}{y^2} (dx^2 + dy^2).
\end{equation*}
By calculus of variation ($L(x, y, \dot{x}, \dot{y}) = \frac{c_{\sigma}(\dot{x}^2+\dot{y}^2)}{y^2}$), we can get
\begin{equation*}
    (x-\mu_c)^2 + y^2 = R^2, \quad \mu_c = \frac{\Vert \boldsymbol{\mu}_1-\boldsymbol{\mu}_0\Vert^2+\lambda^2(\sigma_1^2-\sigma_0^2)}{2\Vert \boldsymbol{\mu}_1-\boldsymbol{\mu}_0\Vert}, \quad R = \sqrt{\mu_c^2+\lambda^2\sigma_0^2} .
\end{equation*}
where $\mu_c$ and $R$ are determined by the boundary conditions $(x_0, y_0) = (0, \lambda\sigma_0)$ and $(x_1, y_1)=(\Vert \boldsymbol{\mu}_1-\boldsymbol{\mu}_0\Vert, \lambda\sigma_1)$.
Then the semi-circle can be parametrized by angle $\theta(t)\in(0,\pi)$: $x(t) = \mu_c+ R\cos\theta(t), y(t) = R\sin\theta(t)$ with two end points $\theta(0)=\theta_0$ and $\theta(1)=\theta_1$ corresponding to $(x_0, y_0) = (0, \lambda\sigma_0)$ and $(x_1, y_1)=(\Vert \boldsymbol{\mu}_1-\boldsymbol{\mu}_0\Vert, \lambda\sigma_1)$ respectively:
\begin{equation*}
    \theta_0 = \atan2(\lambda\sigma_0, -\mu_c), \quad \theta_1 = \atan2(\lambda\sigma_1, \Vert \boldsymbol{\mu}_1-\boldsymbol{\mu}_0\Vert-\mu_c), 
\end{equation*}
where $\atan2(y, x) = \arg(x+ i y) \in(-\pi, \pi)$.

Note the arc-length along the Poincar\'e semi-circle has constant speed $L$: $ds = \frac{\sqrt{dx^2+dy^2}}{y} = \frac{R|d\theta|}{R\sin\theta} = L dt$.
Therefore, we have
\begin{equation*}
    \theta(t) = 2\arctan\left(e^{Lt}\tan\frac{\theta_0}{2}\right), \quad L = \log\frac{\tan(\theta_1/2)}{\tan(\theta_0/2)} .
\end{equation*}

Therefore, the final geodesic equation in the semi-circle parameterization of Poincar\'e becomes
                                               
                   
\begin{align*}
\boldsymbol{\mu}(t) &= \boldsymbol{\mu}_0 + x(t) {\bf e}, \quad x(t) = \mu_c + R\cos\theta(t)   ;\\                             
\sigma(t) &= \frac{R}{\lambda} \sin\theta(t).                            
\end{align*}

\begin{equation*}
e({\bf v}) = \int_{\gamma} ds^2 = c_{\sigma}\int_0^1 \frac{\dot{x}^2 + \dot{y}^2}{y^2} dt = c_{\sigma} \int_0^1 \frac{\dot{\theta}^2(t)}{\sin^2\theta(t)}dt = c_{\sigma} L^2 ,

\paragraph{Hyperbolic parametrization}

With Einstein's notation, $g_{\mu_i\mu_j} = \frac{c_{\mu}}{\sigma^2}\delta_{ij}$, $g_{\sigma\sigma} = \frac{c_{\sigma}}{\sigma^2}$, we compute the Christoffel symbols $\Gamma_{\mu\nu}^{\lambda} = \frac12 g^{\lambda\rho}[\partial_{\mu}g_{\nu\rho}+\partial_{\nu}g_{\mu\rho}-\partial_{\rho}g_{\mu\nu}]$ with inverse metric $g^{\mu_i\mu_j} = \frac{\sigma^2}{c_{\mu}}\delta_{ij}$, $g^{\sigma\sigma} = \frac{\sigma^2}{c_{\sigma}}$.
The only nonzero partial derivatives come from $\partial_{\sigma} g= -\frac{2}{\sigma}g$, giving
\begin{equation*}
    \Gamma_{\mu_j\sigma}^{\mu_i} = -\frac{1}{\sigma}\delta_{ij}, \quad \Gamma_{\mu_i\mu_j}^{\sigma} = \frac{c_{\mu}}{c_{\sigma}\sigma}\delta_{ij}, \quad \Gamma_{\sigma\sigma}^{\sigma} = -\frac{1}{\sigma} .
\end{equation*}

Therefore, the geodesic equations $\ddot{x}^{\lambda}+\Gamma_{\mu\nu}^{\lambda} \dot{x}^{\mu} \dot{x}^{\nu} = 0$ become
\begin{subequations}\label{eq:geod_qep}
\begin{align}
    \ddot{\mu}_i - \frac{2}{\sigma} \dot{\mu}_i \dot{\sigma}
 &=0 , \label{eq:geod_qep-1}\\

 \end{align}
\end{subequations}
Solving \eqref{eq:geod_qep-1} yields $\dot{\boldsymbol{\mu}} = {\bf c} \sigma^2$. Substituting it into \eqref{eq:geod_qep-2} gives $\ddot{\sigma} + \frac{c_{\mu}}{c_{\sigma}}\Vert {\bf c}\Vert^2 \sigma^3 - \frac{1}{\sigma} \dot{\sigma}^2 =0$ .
By changing the variable $\sigma=e^\tau$, \eqref{eq:geod_qep-2} becomes a standard autonomous second-order ODE
\begin{equation*}
    \ddot\tau + \frac{c_{\mu}}{c_{\sigma}}\Vert {\bf c}\Vert^2 e^{2\tau} = 0 ,
\end{equation*}

\begin{equation}\label{eq:geod_energy}

\end{equation}

and the conditional flow 
\begin{equation*}
    \psi_t({\bf x}) = {\bf x}_1 \frac{\tanh(\omega t+\alpha) - \tanh(\alpha)}{\tanh(\omega +\alpha) - \tanh(\alpha)} + \frac{\omega}{\kappa}\operatorname{sech}(\omega t +\alpha) {\bf x} .
\end{equation*}

<!-- thm-4: body_md -->
\label{thm:CFMbound}
For every parameter $\theta$ in the neural network ${\bf v}_t(\cdot;\theta)$, we have
\begin{align*}
{\mathcal L}_\textrm{CFM}(\theta) &= {\mathcal L}_\textrm{FM}(\theta) + C, \quad
C:= \mathbb E_{t, p_t({\bf x})}\mathrm{Var}_{{\bf x}_1|{\bf x}_t={\bf x}}[{\bf u}_t({\bf x}|{\bf x}_1)] , \\
C &= \inf_{\theta} {\mathcal L}_\textrm{CFM}(\theta) \leq \sigma^2_{\max} \kappa_{q,d} \mathbb E_{q({\bf x}_1)} L^2({\mathcal P}_q({\bf 0}, {\bf I}), {\mathcal P}_q({\bf x}_1,\sigma^2_{\min}{\bf I})) ,
\end{align*}
where $C$ does not depend on the neural network ${\bf v}_t(\cdot;\theta)$, 
$\sigma_{\max}$, $\kappa_{q,d}$ are constants depending on the geodesic,
and $L^2({\mathcal P}_q({\bf 0}, {\bf I}), {\mathcal P}_q({\bf x}_1,\sigma^2_{\min}{\bf I}))$ is the scaled square Fisher-Rao distance between ${\mathcal P}_q({\bf 0}, {\bf I})$ and ${\mathcal P}_q({\bf x}_1,\sigma^2_{\min}{\bf I})$.
<!-- end thm-4 -->
<!-- proof-10: body_md -->
\label{apx:CFMbound}
Denote ${\bf x}_t:=\psi_t({\bf x}_0)$ as the conditional flow \eqref{eq:affine_path}.
With $q({\bf x}_1)p_t({\bf x}_t|{\bf x}_1) = p_t({\bf x}_t)p({\bf x}_1|{\bf x}_t)$, the CFM loss \eqref{eq:loss_epd} can be rewritten
\begin{align*}
{\mathcal L}_\textrm{CFM}(\theta) &= \mathbb E_t\mathbb E_{q({\bf x}_1)}\mathbb E_{p_t({\bf x}_t|{\bf x}_1)}\Vert {\bf v}_t({\bf x}_t;\theta)-{\bf u}_t({\bf x}_t|{\bf x}_1)\Vert_2^2 \\
&= \mathbb E_t\mathbb E_{p_t({\bf x}_t)}\mathbb E_{p({\bf x}_1|{\bf x}_t)}\Vert {\bf v}_t({\bf x}_t;\theta)-{\bf u}_t({\bf x}_t|{\bf x}_1)\Vert_2^2 \\
&= \underbrace{\mathbb E_t\mathbb E_{p_t({\bf x}_t)} \Vert {\bf v}_t({\bf x}_t;\theta)-{\bf u}_t({\bf x}_t)\Vert_2^2}_{{\mathcal L}_\textrm{FM}(\theta)} + \underbrace{\mathbb E_t\mathbb E_{p_t({\bf x}_t)} \mathbb E_{p({\bf x}_1|{\bf x}_t)}\Vert {\bf u}_t({\bf x}_t|{\bf x}_1) - {\bf u}_t({\bf x}_t)\Vert_2^2}_{C} .
\end{align*}
where $C=\mathbb E_{t, p_t({\bf x})}\mathrm{Var}_{{\bf x}_1|{\bf x}_t={\bf x}}[{\bf u}_t({\bf x}|{\bf x}_1)]$ based on the definition of marginal vector field ${\bf u}_t({\bf x})$.
Since ${\mathcal L}_\textrm{FM}(\theta)\geq 0$, we get ${\mathcal L}_\textrm{CFM}(\theta)\geq C$, with equality achieved iff $\Vert{\bf v}_t({\bf x}_t;\theta)-{\bf u}_t({\bf x}_t)\Vert=0$ for $p_t$-a.e $({\bf x}, t)$.
Therefore, $C$ is the lower bound of the CFM loss \eqref{eq:loss_epd}, regardless of the network ${\bf v}_t(\cdot; \theta)$.

To get the upper bound of $C$ for the geodesic path, we notice that with conditional vector field \eqref{eq:conditional_vf}
\begin{align*}
C &\leq \mathbb E_t\mathbb E_{q({\bf x}_1)}\mathbb E_{p_t({\bf x}|{\bf x}_1)} \Vert{\bf u}_t({\bf x}|{\bf x}_1)\Vert^2 \\
&= \mathbb E_t\mathbb E_{q({\bf x}_1)}[\Vert\dot\boldsymbol{\mu}_t({\bf x}_1)\Vert^2 + (\dot\sigma_t({\bf x}_1)/\sigma_t({\bf x}_1))^2\mathbb E_{p_t({\bf x}|{\bf x}_1)}\Vert{\bf x}-\boldsymbol{\mu}_t({\bf x}_1)\Vert^2]\\
&\overset{prop\ref{prop:epd_prop}}{=} \mathbb E_{q({\bf x}_1)}\int_0^1[\Vert\dot\boldsymbol{\mu}_t({\bf x}_1)\Vert^2+s_{q,d}d\dot\sigma_t^2({\bf x}_1)] dt \\
&\leq \mathbb E_{q({\bf x}_1)} \int_0^1 \sigma_t^2\max\left\{\frac{1}{c_{\mu}}, \frac{s_{q,d}d}{c_{\sigma}}\right\} \frac{c_{\mu}\Vert\dot\boldsymbol{\mu}_t\Vert^2+c_\sigma \dot\sigma_t^2}{\sigma_t^2} dt\\
&\leq \sigma^2_{\max} \kappa_{q,d} \mathbb E_{q({\bf x}_1)} L^2({\mathcal P}_q(0, {\bf I}), {\mathcal P}_q({\bf x}_1,\sigma^2_{\min}{\bf I})) ,
\end{align*}
where $\sigma_{\max}:=\sup_{t\in[0,1], {\bf x}_1\in\mathrm{supp}(q({\bf x}_1))}\sigma_t^2({\bf x}_1)$, and $\kappa_{q,d}=\max\left\{\frac{1}{c_{\mu}}, \frac{s_{q,d}d}{c_{\sigma}}\right\}$.
<!-- end proof-10 -->
<!-- rk-6: body_md -->
Note that the CFM loss bound $C$ depends only on the probability path $p_t(\cdot|{\bf x})$ and the data distribution $q(\cdot)$, not the neural network ${\bf v}(\cdot, \theta)$.

<!-- rk-7: body_md -->
\label{rk:energy_hump}
With the OT-interpolant path \eqref{eq:OT_interp}, the CFM loss bound collapses to
\begin{equation*}
C=\mathbb E_{t, p_t({\bf x})}\mathrm{Var}_{{\bf x}_1|{\bf x}_t}[{\bf x}_1]
\left(1+\frac{(1-\sigma_{\min})t}{\sigma_t}\right)^2.
\end{equation*}
This quantity peaks around $t\approx \frac12$, where the posterior $p({\bf x}_1|{\bf x}_t)$ is most uncertain. Correspondingly, an ``energy hump" may appear, as illustrated in Figure \ref{fig:ED_traj}.

The proof suggests a more natural choice of loss based on the energy \eqref{eq:energy_fisher}, i.e. replacing the $L_2$ norm with the Fisher-Rao norm in \eqref{eq:loss_epd}. That would yield a tighter bound for $C$.
