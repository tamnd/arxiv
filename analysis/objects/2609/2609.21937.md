\]

$c,C>0$, depending only on $\hat{\beta}_{\text{min}}$ and $\hat{\beta}_{\text{max}}$, such that the

\begin{equation}\label{ub:eq-upper-two-scale}
 {\ensuremath{\mathcal M}} ^h_{N,\boldsymbol{\hat{\beta}}}

 {\ensuremath{\mathcal M}} ^h_{N,\boldsymbol{\hat{\beta}}}

<!-- remark-1: body_md -->
\label{ub:intuition}
  We give an intuitive explanation to the threshold at $h\asymp (\log N)^{\frac{1}{3}}$. 
  The major correction term $\exp\big(\frac{h^3}{L}+\frac{h^6}{L^2}\big)$ in \eqref{ub:eq-upper-two-scale} arises from two pairwise collisions occurring at different times and involving a common
particle, such as $\{\mathsf i,\mathsf j\}$ and $\{\mathsf j,\mathsf k\}$. We illustrate this with an overly simplified example. For some fixed $\hat{\beta}\in (0,1)$, consider the following sums:
\begin{align*}
  A_{\mathrm{cor}}:= \big(\frac{\pi\hat{\beta}}{\log N}\big)^2 \hspace{-.3cm}
  \sum_{\substack{1\le n+m\le N, n,m\ge 1\\ x,y\in {\ensuremath{\mathbb Z}} ^2}} \hspace{-.5cm} p_{n}(x)^2 p_m(y-x) p_{n+m}(y),
  \quad  \quad
  A_{\mathrm{ind}}:= \big(\frac{\pi\hat{\beta}}{\log N}\big)^2 \hspace{-.3cm}
  \sum_{\substack{1\le n+m\le N, n,m\ge 1\\ x,y\in {\ensuremath{\mathbb Z}} ^2}} \hspace{-.5cm} p_{n}(x)^2 p_{n+m}(y)^2.
\end{align*}
Here $A_{\mathrm{cor}}$ is the simplest term involving $3$ walks in \eqref{eq: ub-after ck} assuming all $\hat{\beta}_\mathsf{e}\equiv \hat{\beta}$ for some $\hat{\beta}\in (0,1)$, whereas $A_{\mathrm{ind}}$ is the corresponding term obtained under the hypothetical assumption that the local times are independent. They have graphical representation Figure \ref{fig:three-particle-collision-diagrams}. We wish to show that $A_{\mathrm{cor}}$ can be replaced by $A_{\mathrm{ind}}$. By completing the sum, one could compare the two expression in the following way:
\begin{align*}
  \frac{A_{\mathrm{cor}}}{A_{\mathrm{ind}}}
  =
  1+O\big(\frac{1}{\log N}\big).
\end{align*}
Consequently, the first-order correction, associated with a fixed pair of
collision events $\{\mathsf i,\mathsf j\}$ and $\{\mathsf j,\mathsf k\}$, is of order
$1/\log N$.  Since the number of configurations consisting of two pairwise
collisions involving a common particle is $\binom{h}{3}$, the total
first-order correction is expected to be
$O\bigl(h^3/\log N\bigr)$.

Adding the same constant to all the shifts does not change the shifted
ordering.  Hence we may assume
$$
 0=\min_{1\le r\le m}s_r\le s_r\le R:= R(\mathbf s).
$$
For $1\le r\le m$, define the change of variables:
\begin{align*}
  w_r = w_r(y_r):= \frac{M_N}{M_N+R} \big( y_r-s_r+R \big).
\end{align*}
Then the map $y_r\mapsto w_r: [0,M_N] \to [0,M_N]$ is injective, but not necessarily surjective. Hence if we define the reverse map of $w_r\mapsto y_r$ on the full range of $[0,M_N]$, we might map outside $[0,M_N]$. Therefore, we introduce the following clipping function.  For $x\in{\ensuremath{\mathbb R}} $, define its clipping to $[0,M_N]$ by
\[
 [x]_{[0,M_N]}
 :=\min\{M_N,\max\{0,x\}\}
 =
 \begin{cases}
  0, & x<0,\\
  x, & 0\le x\le M_N,\\
  M_N, & x>M_N.
 \end{cases}
\]
The clipping function $x\mapsto[x]_{[0,M_N]}$ is increasing and one-Lipschitz.

The original domain 
\begin{align}\label{ub:original-domain}
\big\{y_1-s_1 \le \cdots \le y_m- s_m, y_r\in [0,M_N]\big\}, 
\end{align}
after applying the change of variable, is a subset of the simplex
\begin{align}\label{ub:new-domain}
\big\{ 0\le w_1 \le \cdots \le w_m \le M_N \big\},
\end{align}
and the Jacobian is $\mathrm{d} \mathbf y = \Big( \frac{M_N+R}{M_N} \Big)^m \mathrm{d} \mathbf w$. 


For $0\le w\le M_N$, set the formal and genuine reverse of $y_r\mapsto w_r$ to be, respectively,
\begin{equation*}
 \tilde{y}_r(w):=\frac{M_N+R}{M_N}w-R+s_r,
 \qquad
 \widehat{y}_r(w):=[\tilde{y}_r(w)]_{[0,M_N]}.
\end{equation*}
On the image of the original domain \eqref{ub:original-domain}, $\tilde{y}_r(w_r)=y_r\in[0,M_N]$;
hence $\widehat{y}_r(w_r)=y_r$.

Since $[w]_{[0,M_N]}=w$ and clipping is one-Lipschitz,
\begin{align}
 |\widehat{y}_r(w)-w|
 &=\big|[\tilde{y}_r(w)]_{[0,M_N]}-[w]_{[0,M_N]}\big|
 \le |\tilde{y}_r(w)-w|
 =\left|\frac{R}{M_N}w-R+s_r\right|
 \le R.
 \label{ub:new-eq-theta-distance1}
\end{align}
Moreover, by monotonicity of clipping and $\tilde{y}_r(w)-(w-R) = \frac{Rw}{M_N}+s_r\ge 0$, 
\begin{align}
 \widehat{y}_r(w)
 &=[\tilde{y}_r(w)]_{[0,M_N]}
 \ge[w-R]_{[0,M_N]}
 =(w-R)_+.
 \label{ub:new-eq-theta-distance2}
\end{align}

Now, by changing variables $y_r\mapsto w_r$ and extending integration domain, we have that
\begin{align}\label{ub:y-to-w}
  & \int_{\substack{0\le y_r\le M_N\\
 y_1-s_1\le\cdots\le y_m-s_m}}
 \prod_{r=1}^m e^{C_0e^{-y_r}}
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}y_r}
 \,\mathrm{d}\mathbf y \notag \\
 & \le
 \Big(1+ \frac{R}{M_N}\Big)^m
 \int_{0\le w_1\le \cdots \le w_m\le M_N}
 \prod_{r=1}^m e^{C_0e^{-\widehat{y}_r(w_r)}}
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}\widehat{y}_r(w_r)}
 \,\mathrm{d}\mathbf w.
\end{align}

For an arbitrary $\mathsf{e}\in \Pi_h$ and $1\le r\le m$, by the mean value theorem,
\begin{align*}
  \log \frac{\hat{\beta}_{\mathsf{e}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}}}{L}\widehat{y}_r(w)}
  -
  \log \frac{\hat{\beta}_{\mathsf{e}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}}}{L}w}
  & \le 
  \bigg|\frac{\mathrm{d}}{\mathrm{d} y} \log \frac{\hat{\beta}_{\mathsf{e}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}}}{L}y} \bigg|\cdot
  \big|\widehat{y}_r(w) - w\big|\notag \\
  & =
  \bigg| \frac{\hat{\beta}_{\mathsf{e}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}}}{L}y} \bigg| 
  \cdot
  \big|\widehat{y}_r(w) - w\big|
  \le \frac{CR}{L},
\end{align*}
for some $C>0$, where in the last inequality we have used \eqref{ub:new-eq-theta-distance1}. Thus
\begin{align}\label{ub:no-shift1}
  \frac{\hat{\beta}_{\mathsf{e}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}}}{L}\widehat{y}_r(w)}
  \le 
  e^{\frac{CR}{L}}
  \frac{\hat{\beta}_{\mathsf{e}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}}}{L}w}.
\end{align}
Also we have
$
 \left(1+\frac{R}{M_N}\right)^m
 \le\exp\left\{\frac{mR}{M_N}\right\}
 \le\exp\left\{\frac{mCR}{L}\right\}
$. Together with \eqref{ub:no-shift1}, we upper bound \eqref{ub:y-to-w} by
\begin{align}\label{ub:drop-modif1}
  \exp\Big\{\frac{mCR}{L}\Big\}
 \int_{0\le w_1\le \cdots \le w_m\le M_N}
 \prod_{r=1}^m e^{C_0e^{-\widehat{y}_r(w_r)}}
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}w_r}
 \,\mathrm{d}\mathbf w.
\end{align}

By \eqref{ub:new-eq-theta-distance2}, this can be further bounded by
\begin{align}\label{ub:drop-modif2}
 \exp\Big\{\frac{mCR}{L}\Big\}
 \int_{0\le w_1\le \cdots \le w_m\le M_N}
 \prod_{r=1}^m e^{C_0e^{-(w_r-R)_+}}
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}w_r}
 \,\mathrm{d}\mathbf w.
\end{align}
Suppose that $\mathbf W = \{W_1<...<W_m\}$ is a family of ordered uniform random variables on $[0,M_N]$. Then the above can be written in the following expectation form
\begin{align*}
 \frac{M_N^m}{m!}
  \exp\Big\{\frac{mCR}{L}\Big\}
 {\ensuremath{\mathbb E}} _{\mathbf W}\bigg[
 \prod_{r=1}^m e^{C_0e^{-(W_r-R)_+}}
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}W_r}
 \bigg].
\end{align*}   
The first product in the above expectation is decreasing with respect to each $W_r$, and the second product is increasing with respect to each $W_r$. Hence the two products are negatively correlated, and the above is bounded above by
\begin{align}
 & \frac{M_N^m}{m!}
  \exp\Big\{\frac{mCR}{L}\Big\}
 {\ensuremath{\mathbb E}} _{\mathbf W}\bigg[
 \prod_{r=1}^m e^{C_0e^{-(W_r-R)_+}}\bigg] \,
 {\ensuremath{\mathbb E}} _{\mathbf W}\bigg[
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}W_r}
 \bigg] \notag \\ 
 & = 
 \exp\Big\{\frac{mCR}{L}\Big\}
 \frac{m!}{M_N^m} \Bigg( \int_{0\le w_1\le\cdots \le w_m\le M_N}
 \prod_{r=1}^m
 e^{C_0e^{-(w_r-R)_+}}
 \, \mathrm{d} \mathbf w \Bigg)
 \Bigg( \int_{0\le w_1\le \cdots \le w_m\le M_N}
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}w_r}
 \,\mathrm{d}\mathbf w\Bigg)\notag \\ 
 & =
 \exp\Big\{\frac{mCR}{L}\Big\}
 \Bigg( \frac{1}{M_N} \int_{0}^{ M_N}
 e^{C_0e^{-(w-R)_+}}
 \, \mathrm{d}  w \Bigg)^m
 \Bigg( \int_{0\le w_1\le \cdots \le w_m\le M_N}
 \prod_{r=2}^m
 \frac{\hat{\beta}_{\mathsf{e}_{r-1}}/L}{1-\frac{\hat{\beta}_{\mathsf{e}_{r-1}}}{L}w_r}
 \,\mathrm{d}\mathbf w\Bigg).\label{ub:drop-modif3}
\end{align}
The last task is to estimate the integral under the $m$'th power. If $R\ge M_N$, then $(w-R)_+=0$ for $0\le w\le M_N$, and
\begin{align}\label{ub:int11}
\frac1{M_N}\int_0^{M_N}
 e^{C_0 e^{-(w-R)_+}} \,\mathrm{d} w
=e^{C_0}
 \le\exp\left\{C_0\frac{R}{M_N}\right\}.
\end{align}
If $0\le R<M_N$, then
\begin{align}\label{ub:int22}
\frac1{M_N}\int_0^{M_N}
 e^{C_0 e^{-(w-R)_+}} \,\mathrm{d} w
 &=\frac{R}{M_N}e^{C_0}
   +\frac1{M_N}\int_R^{M_N}e^{C_0e^{R-w}}\,\mathrm{d} w\notag \\
 &=1+\frac{R}{M_N}(e^{C_0}-1)
   +\frac1{M_N}\int_R^{M_N}
      \bigl(e^{C_0e^{R-w}}-1\bigr)\,\mathrm{d} w\notag \\
 &\le1+\frac{R}{M_N}(e^{C_0}-1)
   +\frac{C_0e^{C_0}}{M_N}
      \int_R^{M_N}e^{R-w}\,\mathrm{d} w\notag \\
 &\le1+C \frac{R+1}{M_N}
 \le\exp\Big\{C \frac{R+1}{M_N}\Big\}.
\end{align}
Combining \eqref{ub:int11} and \eqref{ub:int22} and plugging into \eqref{ub:drop-modif3}, we finish the proof of \eqref{ub:new-eq-shifted-simplex-bound}.

We start with a tail estimate: there exists $c>0$ such that
\begin{align}
 {\ensuremath{\mathbb P}} (\mathcal W_m^*\ge x)
 &\le
 2\exp\Big\{-c\min\Big(\frac{x^2}{m},x\Big)\Big\}.
 \label{ub:new-eq-max-tail}
 \end{align}
To see this, we apply a martingale maximal inequality. Fix $t\in(0,t_0]$ and let
$
\mathcal F_r=\sigma\big(\tilde{W}_{2,N},\ldots,\tilde{W}_{r,N}\big)
$.  
By \eqref{ub:new-eq-centered-local-mgf},
\[
 \exp\Big\{t \sum_{i=2}^r \tilde{W}_{i,N}-C_\star(r-1)t^2\Big\},
 \qquad 1\le r\le m,
\]
is a nonnegative supermartingale.  The maximal inequality for a nonnegative
supermartingale gives
\[
 {\ensuremath{\mathbb P}} \Big(\max_{r\le m}\sum_{i=2}^r \tilde{W}_{i,N} \ge x\Big)
 \le\exp\{-tx+C_\star  mt^2\}.
\]
The unconstrained minimiser of the above is \(t=x/(2C_\star m)\). However \eqref{ub:new-eq-centered-local-mgf} is only proved for $|t|\le t_0$, so if $x/(2C_\star m)\le t_0$, we take $t$ to be the unconstrained minimiser and we get a Gaussian tail:
\begin{align}\label{ub:oneside1}
  {\ensuremath{\mathbb P}} \Big(\max_{r\le m} \sum_{i=2}^r \tilde{W}_{i,N} \ge x\Big)
 \le\exp\{-\frac{x^2}{4C_\star m}\}.
\end{align}
If $x/(2C_\star m)> t_0$, we simply take $t = t_0$. This gives:
\begin{align}\label{ub:oneside2}
  {\ensuremath{\mathbb P}} \left(\max_{r\le m} \sum_{i=2}^r \tilde{W}_{i,N} 
  \ge x\right)
 \le \exp\{-tx+C_\star mt^2\}
 \le \exp \{ -\frac{t_0 x}{2}   \}.
\end{align}
Similarly, 
$$
 \exp\left\{-t \, \sum_{i=2}^r \tilde{W}_{i,N} -C_\star(r-1)t^2\right\},
 \qquad 1\le r\le m,
$$
is also a supermartingale. Hence again by the maximal inequality for a nonnegative supermartingale,
\begin{align}\label{ub:otherside}
  {\ensuremath{\mathbb P}} \Big(\min_{r\le m}\sum_{i=2}^r \tilde{W}_{i,N}\le  -x\Big)
  =
  {\ensuremath{\mathbb P}} \Big(\max_{r\le m} \big( -\sum_{i=2}^r \tilde{W}_{i,N}\big) \ge x\Big)
  \le
  \exp\big\{ - \min\{ \frac{x^2}{4C_\star m}, \frac{t_0 x}{2} \} \big\}.
\end{align}
Gathering \eqref{ub:oneside1}, \eqref{ub:oneside2}, \eqref{ub:otherside} and by a union bound, we finish the proof of \eqref{ub:new-eq-max-tail}.

Now we prove \eqref{ub:new-eq-max-mgf}. As $e^{\theta {\ensuremath{\mathcal W}} ^*_m}$ is a nonnegative random variable, by \eqref{ub:new-eq-max-tail}, we have:

 {\ensuremath{\mathbb E}}  e^{\theta {\ensuremath{\mathcal W}} ^*_m}
 & =
 1+\theta\int_0^\infty e^{\theta x}{\ensuremath{\mathbb P}} ({\ensuremath{\mathcal W}} ^*_m\ge x)\,\mathrm{d} x\notag \\
 & \le
 1+2 \theta \int_0^\infty e^{\theta x} \exp\big\{ -c \min\big( \frac{x^2}{m},x \big) \big\}
 \, \mathrm{d} x\notag \\
&  =
1+2 \theta 
\bigg( \int_0^m e^{\theta x} \exp\big\{ -c \frac{x^2}{m} \big\}\, \mathrm{d} x
+ \int_m^\infty e^{\theta x} \exp\big\{ -c x  \big\} \, \mathrm{d} x \bigg).\label{ub:int0}
\end{align}
For the first integral above, we complete the square and extend the integral to $(-\infty,+\infty)$ to obtain:
\begin{align}\label{ub:int1}
  \int_0^m e^{\theta x} \exp\Big\{ -c  \frac{x^2}{m} \Big\}\, \mathrm{d} x
  & \le 
  \sqrt{\frac{\pi m}{c }}e^{\frac{m\theta^2}{4c }}.
\end{align}
For the second integral, for $\theta\in [0,\frac{c}{2}]$, 
  \begin{align}\label{ub:int2}
  \int_m^\infty e^{ (\theta  -c ) x }  \, \mathrm{d} x 
  = \frac{e^{-(c -\theta)m}}{c -\theta}
  \le 
  \frac{2e^{-c m/2}}{ c  }.
\end{align} 
  Plugging  \eqref{ub:int1} and \eqref{ub:int2} into \eqref{ub:int0} finishes the proof. Indeed,
\begin{align*}
 {\ensuremath{\mathbb E}}  e^{\theta{\ensuremath{\mathcal W}} _m^*}
 &\le
 1+C \theta\sqrt m\,
       e^{C m\theta^2}
   +C \theta e^{-c m/2}
   \le
 1+C \theta\sqrt m\,
       e^{C  m\theta^2}
  \le 
  e^{C\big(\theta\sqrt{m} + \theta^2 m\big)}.
\end{align*}

 \label{ub:eq-A-recursion}

$

By integrating \eqref{ub:int-A}, we have
\begin{equation*}

\end{equation*}
Hence there exists $C>0$ such that for all $T\in [0,M_N]$, we have
\begin{equation}\label{ub:eq-A-comparison-short}

 \le C \min_{\mathsf{e}\in\Pi_h}A_{\mathsf{e}}(T),
 \qquad 0\le T\le M_N.
\end{equation}
For $r\ge |\vec E|$, we always have ${\ensuremath{\mathbb E}} _{{\ensuremath{\mathbb Q}} _T}\big[ I_r |{\ensuremath{\mathcal G}} _r \big] = 0$, so ${\ensuremath{\mathbb E}} _{{\ensuremath{\mathbb Q}} _T}\big[ e^{\theta I_r}|{\ensuremath{\mathcal G}} _r \big] = 1$ for any $\theta\in {\ensuremath{\mathbb R}} $.
For $1\le r \le |\vec E|-1$, to study probability conditioned on ${\ensuremath{\mathcal G}} _r$, it suffices to study conditional probability on 
$\big\{ \tilde{E}_1 = \mathsf{e}_1,\cdots, \tilde{E}_r = \mathsf{e}_r \big\}$ for all $(\mathsf{e}_1,...,\mathsf{e}_r)\in \Pi_h^{\otimes r}$, as they form the atoms in ${\ensuremath{\mathcal G}} _r$. So for any $\mathsf{e}\neq \mathsf{e}_r$, we consider
\begin{align}

    1\le r\le |\vec E|-1,\, 
    \tilde{E}_{r+1} = \mathsf{e} \,
    \Big| \, \tilde{E}_1 = \mathsf{e}_1,\cdots ,\tilde{E}_r = \mathsf{e}_r
    \Big]\notag \\

  \frac{
    \int^T_0 A_{\mathsf{e}}(t)
    \Big( \lambda_{\mathsf{e}_1}(T) \int_{t<y_2<\cdots <y_r<T} 
    \prod_{i=2}^r \lambda_{\mathsf{e}_{r-i+2}}(y_i) \, \mathrm{d} y_2\cdots \mathrm{d} y_r \Big)
    \, \mathrm{d} t
  }{
    \int^T_0 
    \big( 1+\sum_{\mathsf f\neq \mathsf{e}_r} A_{\mathsf f}(t)\big)
    \Big( \lambda_{\mathsf{e}_1}(T) \int_{t<y_2<\cdots <y_r<T} 
    \prod_{i=2}^r \lambda_{\mathsf{e}_{r-i+2}}(y_i) \, \mathrm{d} y_2\cdots \mathrm{d} y_r \Big)
    \, \mathrm{d} t
  }.
\end{align}
To get the above equation, refer to \eqref{ub:def-Q} and recall that we define $(\tilde{E}_r)$ to be the reverse sequence. Therefore,
\begin{align}
  & {\ensuremath{\mathbb Q}} _T\Big[
    1\le r\le |\vec E|-1,\, 
    \tilde{E}_{r+1} \cap \mathsf{e}_r \neq \emptyset \,
    \Big| \, \tilde{E}_1 = \mathsf{e}_1,\cdots ,\tilde{E}_r = \mathsf{e}_r
    \Big]\notag \\

  \frac{
    \sum_{\mathsf{e}:|\mathsf{e} \cap \mathsf{e}_r| =1}
    \int^T_0 A_{\mathsf{e}}(t)
    \Big( \lambda_{\mathsf{e}_1}(T) \int_{t<y_2<\cdots <y_r<T} 
    \prod_{i=2}^r \lambda_{\mathsf{e}_{r-i+2}}(y_i) \, \mathrm{d} y_2\cdots \mathrm{d} y_r \Big)
    \, \mathrm{d} t

    \int^T_0 
    \big( 1+\sum_{\mathsf f\neq \mathsf{e}_r} A_{\mathsf f}(t)\big)
    \Big( \lambda_{\mathsf{e}_1}(T) \int_{t<y_2<\cdots <y_r<T} 
    \prod_{i=2}^r \lambda_{\mathsf{e}_{r-i+2}}(y_i) \, \mathrm{d} y_2\cdots \mathrm{d} y_r \Big)
    \, \mathrm{d} t
  }.
\label{ub:cond-prob}\end{align}
Notice that in the numerator in \eqref{ub:cond-prob}, the summation contains $2(h-2)$ terms, while in the denominator, the summation contains $\binom{h}{2}-1$ terms. By \eqref{ub:eq-A-comparison-short}, we see that the summands are all comparable, so \eqref{ub:cond-prob} is upper bounded by $C \frac{ 2 (h-2)}{\binom{h}{2}-1} \le \frac{C'}{h}$, for some $C,C'>0$. Since this upper bound is uniform for all $(\mathsf{e}_1,\cdots \mathsf{e}_r)\in \Pi_h^{\otimes r}$, we achieve: 
\begin{align*}
  {\ensuremath{\mathbb Q}} _T\big[ 
    1\le r\le |\vec E|-1,\, 
    \tilde{E}_{r+1} \cap \tilde{E}_r \neq \emptyset \,
    \Big| \, {\ensuremath{\mathcal G}} _r
     \big]
  \le \frac{C}{h}. 
\end{align*}
To finish, just notice that
\begin{align*}
  {\ensuremath{\mathbb E}} _{{\ensuremath{\mathbb Q}} _T} \big[
      e^{\theta I_r}\big| {\ensuremath{\mathcal G}} _r \big]
      =
      1+
      (e^{\theta}-1) \, {\ensuremath{\mathbb Q}} _T\big[ 
                    1\le r\le |\vec E|-1,\, 
                    \tilde{E}_{r+1} \cap \tilde{E}_r \neq \emptyset \,
                    \Big| \, {\ensuremath{\mathcal G}} _r
                    \big]
      \le 1+\frac{C(e^{\theta}-1)}{h}
\end{align*}
and we get \eqref{ub:1step-R}.

\begin{equation}\label{lb:eq:lambda-good}
 \Delta \ge L^4,
 \qquad
 \lambda\ge\frac{\Delta}{4}\ge\frac{L^4}{4},
 \qquad
 m\le C_{\ref*{lb:lem:count}} h^2\log L\le C L.
\end{equation}
In particular, the assumption $\lambda \ge c^{(2)}_{\ref*{lb:prop:LLT}} m^3$ of Proposition \ref{lb:prop:LLT} holds for all sufficiently large $N$, since for some $c>0$,
\begin{align}\label{lb:gl-m}

 \ge cL\longrightarrow\infty.
\end{align}
So we apply Proposition \ref{lb:prop:LLT} on $\mathsf P^{\otimes h}(\bigcap_{\mathsf{e}}\mathcal E_{I_{\mathsf{e}}}^{\mathsf{e}})$. The first term in \eqref{lb:eq:joint-LLT-error} satisfies
\begin{equation}\label{lb:cor-err1}
 {C_{\ref*{lb:prop:LLT}}}\sqrt{\frac{m^3}{\lambda}}
 \le C\frac{(h^2\log L)^{3/2}}{L^2}.
\end{equation}
For a standard $d$-dimensional Gaussian vector $G_d$,
\begin{equation}\label{lb:eq:gauss-tail}
 \mathsf P(|G_d|>x)\le e^{-x^2/8},
 \qquad \text{ for } x\ge2\sqrt d.
\end{equation}
Notice that \eqref{lb:gl-m} implies
$
 {c_{\ref*{lb:prop:LLT}}^{(1)}}\left(\frac{\lambda}{m}\right)^{1/4}
 \ge2\sqrt{2m}
$ for sufficiently large $N$. 
So \eqref{lb:eq:gauss-tail} gives
\begin{align}\label{lb:cor-err2}
 &\mathsf P\Big(
   |G_{2m}|>{c_{\ref*{lb:prop:LLT}}^{(1)}}\big(\frac{\lambda}{m}\big)^{1/4}
 \Big)
 \le
 \exp\Big\{-c\sqrt{\frac{\lambda}{m}}\Big\}
 \le
 \exp\Big\{-\frac{cL^2}{\sqrt{h^2\log L}}\Big\}.
\end{align}
For the third term in \eqref{lb:eq:joint-LLT-error}, by \eqref{lb:eq:lambda-good},
$
 \frac{{c_{\ref*{lb:prop:LLT}}^{(1)}}\sqrt{\Delta/m}}{2\sqrt{2m}}
 \ge\frac{cL^2}{h^2\log L}\longrightarrow\infty

\eqref{lb:eq:gauss-tail}.  Together with \eqref{lb:eq:eig-det},
\begin{align}

 \mathsf P\Big(|G_{2m}|>{c_{\ref*{lb:prop:LLT}}^{(1)}}\sqrt{\frac{\Delta}{m}}\Big)

 \exp\big\{-c\frac{\Delta}{m}\big\}

 \exp\Big\{
   mL-m\log\frac{\Delta}{{C_{\ref*{lb:prop:LLT}}}}
   -c\frac{\Delta}{m}
 \Big\}\notag \\
 & \le

\eqref{lb:eq:Gamma-edge-explicit} gives
\begin{equation}\label{lb:eq:edge-det-gaps}
 \det\Gamma_{\mathsf{e}}
 =\prod_{r=1}^{m_{\mathsf{e}}}d_r^{\mathsf{e}}.
\end{equation}

\begin{align}
 \prod_{\mathsf{e}\in\Pi_h}
 \mathsf P^{\otimes h}(\mathcal E_{I_{\mathsf{e}}}^{\mathsf{e}})
 &=\prod_{\mathsf{e}\in\Pi_h}\prod_{r=1}^{m_{\mathsf{e}}}
   p_{2d_r^{\mathsf{e}}}(0)
  =\frac{\pi^{-m}}
 {\prod_{\mathsf{e}\in\Pi_h}\det\Gamma_{\mathsf{e}}}
 \Big( \prod_{\mathsf{e}\in\Pi_h}\prod_{r=1}^{m_{\mathsf{e}}}
 \big(\pi d_r^{\mathsf{e}}p_{2d_r^{\mathsf{e}}}(0)\big)\Big).
 \label{lb:eq:pair-LLT-product}
\end{align}
By Stirling's approximation, $p_{2d}(0) = 4^{-2d}\binom{2d}{d}^2 = \frac{1}{\pi d} -\frac{1}{4\pi d^2} + O(\frac{1}{d^3})$. So $\pi d p_{2d}(0)<1$, and we have
\begin{align}

\end{align}
where we have used the Stirling's expansion of $p_{2d}(0)$, definition of $\Delta$ and  \eqref{lb:eq:lambda-good}. Assembling \eqref{lb:cor-err1}, \eqref{lb:cor-err2}, \eqref{lb:eq:LLT-tail-two-a} and \eqref{lb:cor-err4} we finish the proof.

\[

 K_u:=(t_{u-1},t_u].

all times in $I_{\mathsf{e}}\cup I_{\mathsf f}$.  For any $q\in {\ensuremath{\mathbb N}} $, let $L_q$ be the $q\times q$
lower triangular matrix:
\begin{align*}
L_q:=
  \begin{pmatrix}
    1 & 0 & \cdots & \cdots & 0 \\ 
    -1 & 1 & 0 & \cdots  & 0 \\ 
    0 & -1 & 1 & \cdots & 0 \\ 
    \vdots & \vdots & \vdots & \ddots & \vdots \\ 
    0  & 0 & \hdots & -1 &  1
  \end{pmatrix}.
\end{align*}
We have $\det L_q=1$. By considering ${\ensuremath{\mathcal L}}  = \begin{pmatrix} L_{m_\mathsf{e}} & 0 \\ 0 & L_{m_\mathsf f} \end{pmatrix}$, we have 
$
 \det(\mathcal L\Gamma_{\mathsf{e},\mathsf f}\mathcal L^{\mathsf T})
 =\det\Gamma_{\mathsf{e},\mathsf f}.

and
\begin{equation*}
 \mathcal L\Gamma_{\mathsf{e},\mathsf f}\mathcal L^{\mathsf T}
 =
 \begin{pmatrix}
  D_{\mathsf{e}}&B\\
  B^{\mathsf T}&D_{\mathsf f}
 \end{pmatrix},
 \quad \text{ with }
 D_{\mathsf{e}}=\operatorname{diag}(|J_1|,\ldots,|J_{m_{\mathsf{e}}}|),
 \text{ and }
 D_{\mathsf f}=\operatorname{diag}(|K_1|,\ldots,|K_{m_{\mathsf f}}|).
\end{equation*}
The off-diagonal blocks are given by
\begin{align}
 B_{r,u}
 &=\frac12\langle\alpha_{\mathsf{e}},\alpha_{\mathsf f}\rangle
 \bigl[
   s_r\wedge t_u-s_{r-1}\wedge t_u
   -s_r\wedge t_{u-1}+s_{r-1}\wedge t_{u-1}
 \bigr]
 =\frac12\langle\alpha_{\mathsf{e}},\alpha_{\mathsf f}\rangle
 |J_r\cap K_u|.
 \label{lb:eq:B-overlap}
\end{align}
By the operation above and considering $I_\mathsf{e}$ or $I_\mathsf f$ to be empty, one also observes that
\begin{equation}\label{lb:eq:single-edge-increment-dets}
 \det\Gamma_{\mathsf{e}}=\det D_{\mathsf{e}},
 \qquad

\end{equation}

\begin{align}\label{lb:shur}

    D_\mathsf f - B^\mathsf T D_\mathsf{e}^{-1} B

\end{align}

\end{align*}

  \le e^{-1/8} \prod_{\mathsf g\in \Pi_h} \det \Gamma_\mathsf g.
\end{align*}
