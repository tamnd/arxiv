\label{th:1}
We have the following correspondence of operators
in $C^\infty(S^2)$ and $\EuScript W^\infty$:
\begin{align}
\frac{\overline z}{1+z \overline z}\quad&\longleftrightarrow\quad
-\frac1{2(n+1)(2n+1)}\, T_+\, \frac{\partial^2}{\partial u^2}
+\frac{n}{2(2n+1)}\,T_-
;
\label{eq:drob1}
\\
\frac{1- z\overline z}{1+z \overline z}\quad&\longleftrightarrow\quad
\frac1{(n+1)(2n+1)} \, T_+\, u \frac{\partial^2}{\partial u^2}
+\frac 1{n+1}\, T_+ \frac{\partial}{\partial u}
-\frac{n}{2n+1}\, T_- \,u
;
\label{eq:drob2}
\\
\frac{z}{1+z \overline z} \quad&\longleftrightarrow\quad
-\frac{1}{2(n+1)(2n+1)}\, T_+\, u^2 \frac{\partial^2}{\partial u^2}
+\frac{2}{(n+1)}\, T_+\, u \frac{\partial}{\partial u}-
\label{eq:drob3}
\\
\quad&\phantom{\longleftrightarrow}\quad
\qquad\qquad\qquad\qquad\qquad\qquad\qquad
-2T_+ 
- \frac n{2(2n+1)} \,T_- \, u^2
\notag
.
\end{align}

<!-- corollary-1: body_md -->
\label{cor:}
{\rm a)} For any  $0\leqslant k\leqslant n$ and nonnegative integers
$p$, $q$, $r$, $p'$, $q'$, $r'$ 
consider the operator
\begin{equation}
                            
\Theta_{n,k}^\pm \, \Bigl(\frac{\partial}{\partial z}\Bigr)^p\,
\Bigl(z\frac{\partial}{\partial z}\Bigr)^q \,
\Bigl(z^2\frac{\partial}{\partial z}\Bigr)^r\,
\Bigl(\frac{\partial}{\partial \overline z}\Bigr)^{p'}\,
\Bigl(\overline z\frac{\partial}{\partial \overline z}\Bigr)^{q'} \,
\Bigl(\overline z^2\frac{\partial}{\partial \overline z}\Bigr)^{r'}.
\label{eq:NEBASIS}
\end{equation}
Then the corresponding operator in $\EuScript W^\infty$
is a finite sum
of the form
\begin{equation}
\sum_{M,L\geqslant 0, K\in {\mathbb Z}} \Xi_{M,L,K}(n) \, T_+^K\, u^M\, \frac{\partial^L}{\partial u^L},
\label{eq:diff-diff}
\end{equation}
where $\Xi_{M,L,K}(n)$ are rational functions in $n$ with poles at
integer and half-integer points.

\smallskip

{\rm b)} The linear span $\EuScript U$ of operators \eqref{eq:NEBASIS}
is an algebra, i.e., it
is closed with respect to the product.

For each $\lambda \in{\mathbb C }$ we  define a representation
of $\mathrm  {SO}_0(1,3)$ in the space $C^\infty(S^2)$ by the formula
$$
T_\lambda\begin{pmatrix}
\alpha&\beta\\\gamma&\delta
\end{pmatrix} f(x)=
f\bigl((\alpha+x\gamma)^{-1}(\beta+x\delta)\bigr)\cdot (\alpha+x\gamma)^{-2\lambda}.
$$
By the chain rule,
$$
T_\lambda(g_1)\,T_\lambda(g_2)=T_\lambda(g_1g_2).
$$
The representations $T_\lambda$ are called {\it representations
of spherical principal series}.

are unitary in the Hilbert space $L^2(S^2)$.

<!-- proposition-1: body_md -->
\label{pr:intertwiner}
For a fixed $n$ the operator
\begin{equation}
A_n f(u):= \frac1\pi\int_{\mathbb C } K(z, \overline z; u,n)\, 
f(z,\overline z)\frac{\widehat d z}{(1+z\overline z)^2}
\label{eq:A-n}
\end{equation}
is an $\mathrm {SU}(2)$-intertwining operator
$$
L^2\bigl({\mathbb C },(1+z\overline z)^{-2} \widehat d z\bigr)\to \EuScript B_{2n}
.$$
<!-- end proposition-1 -->
<!-- head-6: body_md -->
Fix $n$.
First, let us check that the integral is well defined. 
 The expression 
$$
K(\cdot)=(\dots)^n=\sum_{j=0}^{2n}
c_j(z,\overline z)  u^j
$$
is a polynomial in $u$ of degree $2n$
with coefficients of the form
$$
c_j(z,\overline z)= \frac{\varkappa_j(z,\overline z)}
{(1+z\overline z)^n}
,$$
where $\varkappa_j(z,\overline z)$ are polynomials
of degree $2n$. So, the coefficients $c_j(z,\overline z)$
are bounded functions, they are contained in our $L^2$.
Hence, for $f\in L^2$,
$$
A_n f(u)=\frac1\pi\sum_{j=0}^{2n} u^j\int_{\mathbb C } c_j(z,\overline z)\, f(z,\overline z)
\frac{\widehat d z}{(1+z\overline z)^2}=\frac1\pi\sum_{j=0}^{2n} u^j\cdot
\bigl\langle\, c_j,\overline {\phantom{.}f \phantom{.}}\, \bigr\rangle_{L^2(\dots)}.
$$
The inner products are well-defined,  therefore we get a well-defined operator $L^2\to \EuScript B_{2n}$.



\smallskip

Keeping in  mind \eqref{eq:1+zu}--\eqref{eq:z-u}, we get
that for $h\in\mathrm {SU}(2)$
\begin{equation}
K\bigl(z^{[h]}, \overline {z^{[h]}};u^{[h]},n\bigr)
=\frac{ K(z,\overline z; u,n)}{(a-\overline b u)^{2n}} 
.
\label{eq:KK}
\end{equation}
Consider the composition of the transformation
$f(z,\overline z)\mapsto f\bigl(z^{[h]}, \overline{z^{[h]}}\bigr)$
and the operator $A_n$.
Keeping in the mind   \eqref{eq:whdzh} and \eqref{eq:KK},
we get
\begin{multline*}
\frac1\pi\int_{\mathbb C } K(z,\overline z;u,n)\, f\bigl(z^{[h]}, \overline{z^{[h]}}\bigr)\,
\frac{\widehat d z}{(1+z \overline z)^2}
=\\=
\frac1\pi\int_{\mathbb C } K(z,\overline z;u,n)\,f\bigl(z^{[h]}, \overline{z^{[h]}}\bigr)
\,
\frac{\widehat d z^{[h]}}{(1+z^{[h]} \overline {z^{[h]}})^2}
=\\=
\frac1\pi\int_{\mathbb C } 
K\bigl(z^{[h]} ,\overline {z^{[h]}} ;u^{[h]} ,n\bigr)\,
(a-\overline b u)^{2n}\,
f\bigl(z^{[h]}, \overline{z^{[h]}}\bigr)\,
\frac{\widehat d z^{[h]}}{(1+z^{[h]} \overline {z^{[h]}})^2}.
\end{multline*}
We substitute $\xi=z^{[h]}$ and come to
$$
\qquad
\biggl(\frac1\pi\int_{\mathbb C } 
K\bigl(\xi ,\overline \xi ;u^{[h]} ,n\bigr)\,
f(\xi, \overline \xi)
\frac{\widehat d \xi}{(1+\xi \overline \xi)^2}\biggr)
\cdot (a-\overline b u)^{2n}=
\tau_{2n} \, A_n f(u).
\qquad

<!-- lemma-1: body_md -->
\label{l:norms}
Let $\psi\in \EuScript H_n$.
Then 
$$
\|A_n\psi\|^2_{\EuScript B_{2n}}=\frac{n!\,n!}{(2n+1)!}\, \|\psi\|^2_{\EuScript H_n}.
$$
<!-- end lemma-1 -->
<!-- head-7: body_md -->
The operator $A_n:\EuScript H_n \to\EuScript B_{2n}$   intertwines
two irreducible unitary representations; by the Schur lemma,
the ratio $\|A_n\psi\|^2/\|\psi\|^2$ does not depend on $\psi$.
Let us evaluate it for  the function
$\psi_0^{(n)}(z,\overline z)=\overline z^n/(1+z \overline z)^n\in \EuScript H_n$.

We have
\begin{multline*}
\|\psi_0^{(n)}\|^2_{L^2({\mathbb C },(1+z\overline z)^{-2}\widehat d z)}=
\frac 1\pi \int_{\mathbb C } \frac{\overline z^n}{(1+z\overline z)^n}
\cdot \frac{z^n}{(1+z\overline z)^n}\cdot \frac{\widehat d z}{(1+z\overline z)^2} 
=\\=
\frac{2\pi}{\pi}\int_0^{\infty}
\frac{r^{2n}\cdot rdr}{(1+r^2)^{2n+2}}=\int_0^\infty\frac{\rho^n d\rho}{(1+\rho)^{2n+2}}=B(n+1,n+1)=\frac{n!\,n!}{(2n+1)!}.
\end{multline*}
Next,
\begin{multline*}
A_n \psi_0^{(n)}(u)
=\frac 1\pi \int_{\mathbb C } \Bigl(\frac{(1+\overline z u)(z-u}{1+z\overline z}\Bigr)^n
\cdot
  \frac{\overline z^n}{(1+z\overline z)^n}\cdot \frac{\widehat d z}{(1+z\overline z)^2} 
  =\\=
 \frac 1\pi\int_0^\infty \int_0^{2\pi}
 \frac{\bigl[1+u r e^{-i\varphi} \bigr]^n \, \bigl[r e^{i\varphi}-u\bigr]^n\, r^n e^{-i\varphi n}\,r\,d\varphi\,dr}{(1+r^2)^{2n+2}} 
 =\\=
  \frac 1\pi\int_0^\infty \int_0^{2\pi}
 \frac{\bigl[1+u r e^{-i\varphi} \bigr]^n \, \bigl[r -u e^{-i\varphi}\bigr]^n\, r^n \,r\,d\varphi\,dr}{(1+r^2)^{2n+2}} .
\end{multline*}
We open brackets
$[\dots]^n [\dots]^n$
and get
$$
[\dots]^n [\dots]^n=r^n+\sum_{k=1}^{2n} p_k(u,r) e^{-ik\varphi}.
$$
After integration $\int_0^{2\pi}$ all summands except the first one vanish
and we get
$$
2\int_0^\infty\frac{r^n\cdot r^n\,r\,dr}{(1+r^2)^{2n+2}}=
\frac{n!\,n!}{(2n+1)!}= \frac{n!\,n!}{(2n+1)!}\cdot u^0.
$$
By \eqref{eq:norms-monomials}, we have
$$
\|u^0\|^2_{\EuScript B_{2n}}=1.
$$
Therefore,
$$
\|A_m \psi_0^{(n)}\|_{\EuScript B_{2n}}^2=\Bigl(\frac{n!\,n!}{(2n+1)!}\Bigr)^2,
$$
and we come do the desired statement.
\hfill $\square$

\begin{observation}
\label{obs:1}
The space  ${\mathbb C }[\boldsymbol{\EuScript S}]$
is closed with respect to the product.
\end{observation}


             
Clearly, $\Theta^+_{m,k}\, \Theta^+_{n,l}=\Theta^+_{m+n,k+l}$.
Let us examine products
$\Theta^+_{m,k}\cdot \Theta^-_{n,l}$.
 To be definite, assume
that $k\geqslant l$. Then
\begin{multline*}
\Theta^+_{k,m}\cdot \Theta^-_{l,n}=
\frac{(z\overline z)^l z^{k-l}}{(1+z \overline z)^{m+n}}=
\frac{\bigl[(1+z\overline z)-1\bigr]^l z^{k-l}}{(1+z \overline z)^{m+n}}
=\\=
 \frac{\sum_{j=0}^l C_l^j(-1)^{j}(1+z\overline z)^j\cdot z^{k-l}}
{(1+z \overline z)^{m+n}}=
\sum_{j=0}^l C_l^j(-1)^{j} \Theta^+_{m+n-j,k-l}.
\end{multline*}



\begin{observation}
\label{obs:2}
 The space ${\mathbb C }[\boldsymbol{\EuScript S}]$ is invariant with respect to  the operators
$\frac{\partial}{\partial z}$, $z\frac{\partial}{\partial z}$, 
$z^2\frac{\partial}{\partial z}$, $\frac{\partial}{\partial \overline z}$, $\overline z\frac{\partial}{\partial \overline z}$, 
$\overline z^2\frac{\partial}{\partial \overline z}$.
\end{observation}

\smallskip

This follows from the following trivial calculations 
(but their result is not a priory obvious). Let $0\leqslant m \leqslant n$. We apply our operators
to $\Theta^+_{n,n-m}$:
$$
z^2 \frac{\partial}{\partial z} \frac{z^{n-m}} {(1+z \overline z)^n}
= \frac{n z^{n-m+1}}{(1+z\overline z)^{n+1}}-
\frac{m z^{n-m+1}}{(1+z\overline z)^{n}}\,\,
=\,\, n\Theta^+_{n+1,n-m+1}-m\Theta^+_{n,n-m+1}.
$$
If $m=0$, we can not write $\Theta^+_{n,n-m+1}$,
but in this case the term with $\Theta^+_{n,n-m+1}$
is absent.
 Next,
$$
z \frac{\partial}{\partial z} \frac{z^{n-m}} {(1+z \overline z)^n}
\!=\! \frac{n z^{n-m}}{(1+z\overline z)^{n+1}}-
\frac{m z^{n-m}}{(1+z\overline z)^{n}},
\,\, 
 \frac{\partial}{\partial z} \frac{z^{n-m}} {(1+z \overline z)^n}
\!=\! \frac{n z^{n-m-1}}{(1+z\overline z)^{n+1}}-
\frac{m z^{n-m-1}}{(1+z\overline z)^{n}}.
$$
The right hand side of the second equality does not have the desired form if $m=n$. But in that case $\frac{\partial}{\partial z} (1+z \overline z)^{-n}=-n \overline z (1+z \overline z)^{-n-1}$. Further, 
$$
\frac{\partial}{\partial \overline z}\,
\frac{z^{n-m}}
{(1+z \overline z)^{n}}=\frac{n\,z^{n-m+1}}
{(1+z \overline z)^{n+1}},\quad 
\overline z\frac{\partial}{\partial \overline z}\,
\frac{z^{n-m}}
{(1+z \overline z)^{n}}=\frac{n\,z^{n-m}}{(1+z \overline z)^{n+1}}-\frac{n\,z^{n-m}}{(1+z \overline z)^{n}};
$$
$$
\overline z^2\frac{\partial}{\partial \overline z}\,
\frac{z^{n-m}}
{(1+z \overline z)^{n}}=-\frac{n z^{n-m-1}}{(1+z \overline z)^{n+1}}+
\frac{2n z^{n-m-1}}{(1+z \overline z)^{n}}-
\frac{n z^{n-m-1}}{(1+z \overline z)^{n-1}}.
$$
The last transformation is not appropriate for us
if $m=n$. In this case,
$$
\overline z^2\frac{\partial}{\partial \overline z}\,
\frac{1}
{(1+z \overline z)^{n}}=-\frac{n z \overline z^2}{(1+z \overline z)^{n+1}}
=\frac{n \overline z}{(1+z \overline z)^{n+1}}- \frac{n \overline z}{(1+z \overline z)^{n}}.
$$
