<!-- remark-1: body_md -->
\label{rem:dictionary}
The curve \eqref{SWcurve} and the curve \eqref{eq:SU2-elliptic-curve} of Section \ref{Witten---Seiberg-Theory} are related by an isogeny. In the units $\Lambda=1$
used there, the two descriptions are matched by $u=2\tilde u$, where $\tilde u$ denotes the modulus of \eqref{eq:SU2-elliptic-curve}; the period $a$ is the same
function, given by \eqref{eq:a-hypergeometric}, while $a_D$ and $\tau$ of this chapter are twice those of \eqref{eq:aD-hypergeometric} and \eqref{eq:tau-elliptic}. For
instance, at $u=6\Lambda^2$ the modulus in \eqref{eq:tau-elliptic} is $k^2=1/2$, so that $\tau=2i$ exactly.
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{thm:prepotential}
Let $\Sigma_{u_0}$ be a smooth Seiberg-Witten curve, $a_0=a(u_0)$, and let $\omega_{0,n}$ be the genus-zero
multi-differentials produced by the topological recursion on the spectral curve \eqref{spectral} at $u=u_0$. Then, in a neighbourhood of $a_0$,
\begin{equation}
\mathcal{F}(a)=\mathcal{F}(a_0)+2\pi i\,a_D(a_0)\,(a-a_0)+\sum_{n=2}^{\infty}\frac{(-1)^n}{n!}\left[\oint_{B}\!\!\cdots\!\oint_{B}\omega_{0,n}\right](a-a_0)^n
\label{Taylor}
\end{equation}
that is,
\begin{equation}
\frac{\partial^n\mathcal{F}}{\partial a^n}=(-1)^n\oint_{B}\!\!\cdots\!\oint_{B}\omega_{0,n}\qquad(n\geq2)
\label{derivs}
\end{equation}
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
\label{prop:genus0}
For the spectral curve \eqref{spectral} of pure $SU(2)$:
\begin{align}
\oint_{B}\!\oint_{B}\omega_{0,3}(p,\cdot,\cdot)&=\frac{1}{2{a'}^2}\left(\frac{1}{z^2}-\frac{a}{2a'\Delta}\right)\frac{\mathrm{d} z}{y}\label{om03p}\\[4pt]
\oint_{B}\!\oint_{B}\!\oint_{B}\omega_{0,3}&=\frac{1}{\Delta\,{a'}^3}\label{om03}\\[4pt]
\oint_{B}\!\oint_{B}\!\oint_{B}\!\oint_{B}\omega_{0,4}&=\frac{1}{\Delta^2{a'}^4}\left(2u-\frac{3a}{4a'}\right)\label{om04}
\end{align}
where $a'=\mathrm{d} a/\mathrm{d} u$ and $\Delta=u^2-4\Lambda^4$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Let $\mathcal{K}(p):=\oint_{p_1\in B}K(p_1,p)$. By \eqref{Bergper}, \eqref{holo} and \eqref{localdata}, in the coordinate $z$ near $r_\pm$,
\begin{equation}
\mathcal{K}=-\pi i\,\frac{\int_{\sigma(p)}^{p}\omega}{\omega_{0,1}-\sigma^*\omega_{0,1}}=-\frac{I(z)\,y(z)}{8a'z^2\,\mathrm{d} z}\quad,\quad I(z):=\int_{-z}^{z}\frac{\mathrm{d}\zeta}{y(\zeta)}
\label{Kint}
\end{equation}
with $I(z)\,y(z)=2z\,(1-\frac{2u}{3\Delta}z^2+\dots)$ and $I(z)/y(z)=\frac{2z}{\Delta}(1+\frac{4u}{3\Delta}z^2+\dots)$.

For $n=3$ the bracket in \eqref{TR} is $\omega_{0,2}(p,p_2)\omega_{0,2}(\sigma(p),p_3)+\omega_{0,2}(p,p_3)\omega_{0,2}(\sigma(p),p_2)$. Integrating $p_2$
and $p_3$ over $B$ turns it into $2(2\pi i)^2\omega\,\sigma^*\omega=-2(2\pi i)^2\omega^2$. If $p_1$ is kept free, the kernel \eqref{kernel} is needed with
\eqref{Bergman}; since $\int_{-z'}^{z'}\omega_{0,2}(p,\cdot)=2z'\,\omega_{0,2}(p,r_\pm)/\mathrm{d} z'+\mathcal{O}({z'}^3)$, each residue is elementary and
\begin{equation}
\begin{split}
\oint_{B}\!\oint_{B}\omega_{0,3}(p,\cdot,\cdot)&=\frac{1}{2{a'}^2}\sum_{r_\pm}\frac{1}{y(r_\pm)}\frac{\omega_{0,2}(p,r_\pm)}{\mathrm{d} z'}\\
&=\frac{1}{2{a'}^2}\sum_{\pm}\left[\pm\frac{1}{2\sqrt\Delta\,z^2}+\frac{1}{2yz^2}-\frac{a}{4a'\Delta\,y}\right]\mathrm{d} z
\end{split}
\end{equation}
which is \eqref{om03p}. If $p_1$ is integrated as well, $\mathcal{K}$ replaces $K$ and
\begin{equation}
\oint_{B}\!\oint_{B}\!\oint_{B}\omega_{0,3}=-2(2\pi i)^2\sum_{r_\pm}\mathop{\mathrm{Res}}_{z=0}\mathcal{K}\,\omega^2=\sum_{r_\pm}\mathop{\mathrm{Res}}_{z=0}\frac{I(z)}{4{a'}^3z^2y(z)}\,\mathrm{d} z=\frac{2}{2{a'}^3\Delta}
\end{equation}

For $n=4$ the bracket in \eqref{TR} contains, for each of the three ways of singling out one of the points
$p_2$, $p_3$, $p_4$, the two terms $\omega_{0,3}(p,\cdot,\cdot)\omega_{0,2}(\sigma(p),\cdot)+\omega_{0,3}(\sigma(p),\cdot,\cdot)\omega_{0,2}(p,\cdot)$. Let $\Phi$ denote the
differential \eqref{om03p}. It is odd under $\sigma$, like $\omega$, so that after the three integrations the bracket becomes
$3\cdot2\pi i\,(\Phi\,\sigma^*\omega+\sigma^*\Phi\,\omega)=-12\pi i\,\Phi\,\omega$, and
\begin{equation}
\oint_{B}\!\oint_{B}\!\oint_{B}\!\oint_{B}\omega_{0,4}=-12\pi i\sum_{r_\pm}\mathop{\mathrm{Res}}_{z=0}\mathcal{K}\,\Phi\,\omega=\sum_{r_\pm}\mathop{\mathrm{Res}}_{z=0}\frac{3}{8{a'}^4}\,\frac{I(z)}{z^2y(z)}\left(\frac1{z^2}-\frac{a}{2a'\Delta}\right)\mathrm{d} z
\end{equation}
The residue is $\frac{3}{8{a'}^4}\cdot\frac{2}{\Delta^2}\left(\frac{4u}{3}-\frac{a}{2a'}\right)$ at each of the two points, which gives \eqref{om04}.

<!-- proposition-2: body_md -->
\label{prop:genus1}
For the spectral curve \eqref{spectral} of pure $SU(2)$,
\begin{equation}
\oint_{B}\omega_{1,1}=\frac{1}{\Delta\,a'}\left(\frac{u}{6}-\frac{a}{8a'}\right)=-\frac{\partial F^{(1)}}{\partial a}
\label{om11}
\end{equation}
with $F^{(1)}$ as in \eqref{genusexp}.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Evaluating \eqref{Bergman} at $p_2=\sigma(p_1)$, i.e. $z_2=-z_1=-z$, $y_2=y_1=y$, and using
$(z^2+u)^2-4\Lambda^4=y^2+4uz^2$,
\begin{equation}
\omega_{0,2}(p,\sigma(p))=-\left[\frac{1}{4z^2}+\frac{1}{y^2}\left(\frac u2-\frac{a}{4a'}\right)\right]\mathrm{d} z^2
\end{equation}
With $\mathcal{K}$ of \eqref{Kint},
\begin{equation}
\begin{split}
\oint_{B}\omega_{1,1}&=\sum_{r_\pm}\mathop{\mathrm{Res}}_{z=0}\frac{I(z)y(z)}{8a'z^2}\left[\frac{1}{4z^2}+\frac{1}{y^2}\left(\frac u2-\frac{a}{4a'}\right)\right]\mathrm{d} z\\
&=\frac{2}{4a'}\left[-\frac{u}{6\Delta}+\frac1\Delta\left(\frac u2-\frac{a}{4a'}\right)\right]
\end{split}
\end{equation}
which is the first equality. For the second, $\partial_aF^{(1)}=\frac{1}{a'}\left(-\frac{a''}{2a'}-\frac{u}{6\Delta}\right)$ and $a''=-a/4\Delta$ by \eqref{PF}.

<!-- conjecture-1: body_md -->
\label{conj}
Let $F^{(g)}(a)$ be the coefficients of the expansion \eqref{genusexp} of the pure $SU(2)$ Nekrasov partition
function on the self-dual $\Omega$-background, and let $\omega_{g,n}$ be the multi-differentials produced by the
topological recursion on the Seiberg-Witten spectral curve \eqref{spectral}. Then, for all $g\geq0$ and
$n\geq1$ with $2g-2+n>0$,
\begin{equation}
\oint_{B}\!\!\cdots\!\oint_{B}\omega_{g,n}=(-1)^n\frac{\partial^nF^{(g)}}{\partial a^n}
\label{conjeq}
\end{equation}
