<!-- ass-1: body_md -->
\label{asm:0} 
We suppose $F\left(\mathbf{x}_{1}\right)-\min_{\mathbf{x} \in \mathcal{X}} F(\mathbf{x}) \leq \Delta_{F}$ for the initial point $\mathbf{x}_{1}$. The feasible set $\mathcal{X}$ is closed and convex with bounded diameter, i.e., $\max _{\mathbf{x}, \mathbf{y} \in \mathcal{X}}\left\|\mathbf{x}-\mathbf{y}\right\| \leq D$.
<!-- end ass-1 -->
<!-- ass-2: body_md -->
\label{asm:1} 
All functions $f_1, \dotsc, f_K$ are $L_f$-Lipschitz continuous, and their Jacobians $\nabla f_1, \dotsc, \nabla f_K$ are $L_J$-Lipschitz continuous on the bounded domain $\mathcal{X}$.
<!-- end ass-2 -->
<!-- head-2: body_md -->
Under Assumption~\ref{asm:1}, the objective $F$ is $L_F$-smooth, where
$L_F \coloneqq L_f^{2K-1}L_J\sum_{i=1}^K\frac{1}{L_f^i}$ \citep{jiang2022optimal}.
<!-- end head-2 -->
<!-- ass-3: body_md -->
\label{asm:stochastic2} For $1\leq i\leq K$, we assume that:
\begin{gather*}
       \mathbb{E}_{\xi_t^i}\left[f_i(\mathbf{x};\xi_t^i)\right] = f_i(\mathbf{x}), \quad
	  \mathbb{E}_{\xi_t^i}\left[\nabla f_i(\mathbf{x};\xi_t^i)\right] = \nabla f_i(\mathbf{x}),\\
 \mathbb{E}_{\xi_t^i}\left[\left\|f_i(\mathbf{x};\xi_t^i) - f_i(\mathbf{x})\right\|^2 \right]\leq \sigma^2, \quad
	\mathbb{E}_{\xi_t^i}\left[\left\|\nabla f_i(\mathbf{x};\xi_t^i) - \nabla f_i(\mathbf{x}) \right\|^2 \right] \leq  \sigma_J^2,
\end{gather*}
where $\{\xi_t^i\}_{i=1}^K$ are mutually independent.
<!-- end ass-3 -->
<!-- ass-4: body_md -->
\label{asm:stoc_smooth3} (Average smoothness and Lipschitz)
For $1\leq i\leq K$, we assume that:
	\begin{align*}
	\mathbb{E}_{\xi_t^i}\left[\left\|f_i(\mathbf{x};\xi_t^i)-  f_i(\mathbf{y};\xi_t^i)\right\|^2\right] &\leq L_f^2 \left\|\mathbf{x} - \mathbf{y}\right\|^2,\\
		\mathbb{E}_{\xi_t^i}\left[\left\|\nabla f_i(\mathbf{x};\xi_t^i)- \nabla f_i(\mathbf{y};\xi_t^i)\right\|^2\right] &\leq  L_J^2 \left\|\mathbf{x} - \mathbf{y}\right\|^2.
	\end{align*}

<!-- theorem-1: body_md -->
\label{thm:main}
Under Assumptions~\ref{asm:0},\ref{asm:1},\ref{asm:stochastic2},\ref{asm:stoc_smooth3}, PMVR-v1 enjoys the following guarantees:
\begin{compactenum}[(i)]
\item By setting $B_1 = \Omega(1)$, $\eta = \Theta(\epsilon^2)$, and $\alpha = \Theta(\epsilon^2)$, we can ensure that  $\mathbb{E}\left[\mathcal{F}(\mathbf{x}_\tau)\right] \leq \epsilon$ within $\mathcal{O}(\epsilon^{-3})$ iterations.
\item By setting $B_1 = \Omega(\epsilon^{-1})$, $\eta = \Theta(\epsilon)$, and $\alpha = \Theta(\epsilon)$, we can ensure that $\mathbb{E}\left[\mathcal{F}(\mathbf{x}_\tau)\right] \leq \epsilon$ within $\mathcal{O}(\epsilon^{-2})$ iterations.
\end{compactenum}
<!-- end theorem-1 -->
<!-- head-3: body_md -->
With constant batch size, the theorem implies $\mathcal{O}(\epsilon^{-3})$ complexity for both SFO and LMO calls, matching the rates for projection-free single-level problems~\citep{Zhang2019OneSS}.
(ii) With a large batch size $B_1=\Omega(\epsilon^{-1})$, we obtain SFO complexity of $\mathcal{O}(\epsilon^{-3})$ and LMO complexity of $\mathcal{O}(\epsilon^{-2})$, consistent with the existing single-level projection-free results~\citep{pmlr-v97-yurtsever19b}.

<!-- theorem-2: body_md -->
\label{thm:main+}
Under Assumptions~\ref{asm:0},\ref{asm:1},\ref{asm:stochastic2},\ref{asm:stoc_smooth3}, Stage-wise PMVR-v1 guarantees the following:
\begin{compactenum}[(i)]
    \item  By setting $T_s=2^{s-1}$, $B_1^s = 1$, and $\eta_s = \alpha_s = T_s^{-2/3}$, we can ensure that  $\mathbb{E}\left[\mathcal{F}(\mathbf{x}_\tau)\right] \leq \epsilon$ within $\mathcal{O}(\epsilon^{-3})$ iterations.
    \item By setting $T_s=2^{s-1}$, $B_1^s = \sqrt{T_s}$, and $\eta_s = \alpha_s = T_s^{-1/2}$, we can ensure $\mathbb{E}\left[\mathcal{F}(\mathbf{x}_\tau)\right] \leq \epsilon$ within $\mathcal{O}(\epsilon^{-2})$ iterations.
\end{compactenum}

The stage-wise schedule matches the SFO and LMO complexities in Theorem~\ref{thm:main} while avoiding problem-dependent constants to set hyper-parameters.

<!-- theorem-3: body_md -->
\label{thm:2} Under Assumptions~\ref{asm:0},\ref{asm:1},\ref{asm:stochastic2},\ref{asm:stoc_smooth3}, our PMVR-v2 enjoys the following guarantees:
\begin{compactenum}[(i)]
    \item By setting $B_1 = \Omega(1)$, $N = \Omega(\epsilon^{-1})$, $\eta = \Theta(\sqrt{\epsilon})$, and $\alpha = \Theta(\epsilon)$, we can ensure $ \mathbb{E}\left[\mathcal{G} (\mathbf{x}_\tau) \right]\leq \epsilon$ in $\mathcal{O}(\epsilon^{-1.5})$ iterations.

\end{compactenum}

A function $F:\mathcal{X} \mapsto \mathbb{R} $ is $\lambda$-strongly convex if
\begin{align*}
  F(\mathbf{y}) \geq F(\mathbf{x})+\langle\nabla F(\mathbf{x}), \mathbf{y}-\mathbf{x}\rangle+\frac{\lambda}{2}\|\mathbf{y}-\mathbf{x}\|^{2}, \quad \forall \mathbf{x}, \mathbf{y} \in \mathcal{X}.
\end{align*}

According to the Lemma 4 of the literature~\citep{jiang2022optimal}, we know that
\begin{align*}
\mathbb{E}\left[\left\|\nabla F(\mathbf{x}_t)- \prod_{i=1}^K\nabla f_i(\mathbf{u}_t^{i-1}) \right\|^2\right] \leq K\sum_{i=1}^{K-1} C_i^2\mathbb{E}\left[\left\|f_i(\mathbf{u}_t^{i-1})-\mathbf{u}_t^i\right\|^2\right],

<!-- lemma-2: body_md -->
\label{lem:3} The gradient estimator $\mathbf{v}_t$ enjoys the following guarantee:
\begin{align*}
&\frac{1}{T}\sum_{t=1}^T \mathbb{E}\left[\left\|\mathbf{v}_t -\prod_{i=1}^K \nabla f_i(\mathbf{u}_t^{i-1})\right\|^2\right] \leq  \frac{1}{\alpha T}\mathbb{E}\left[\left\| \mathbf{v}_{1} - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{1}^{i-1})\right\|^2\right] \\
&\qquad+ 2 K^2 \sigma_J^2 L_f^{2K-2}\frac{\alpha}{B_1}+2K L_J^2L_f^{2K-2} \frac{1}{B_1\alpha T}\sum_{t=1}^T\sum_{i=1}^{K} \mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-1} - \mathbf{u}_{t}^{i-1}\right\|^2\right].
\end{align*}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
According to the definition of $\mathbf{v}_t$, we know that:
\begin{align*}
&\mathbb{E}\left[\left\|\mathbf{v}_t -\prod_{i=1}^K \nabla f_i(\mathbf{u}_t^{i-1})\right\|^2\right]\\
=&\mathbb{E}\left[\left \| \frac{1}{B_1}\sum_{j=1}^{B_1}\prod_{i=1}^K \nabla f_i(\mathbf{u}_t^{i-1};\xi_t^{i,j})  + (1-\alpha)\left(\mathbf{v}_{t-1} - \frac{1}{B_1} \sum_{j=1}^{B_1} \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j})\right) - \prod_{i=1}^K \nabla f_i(\mathbf{u}_t^{i-1})\right \|^2 \right]\\
=& \mathbb{E}\left[\left \| (1-\alpha) \left( \mathbf{v}_{t-1} - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1}) \right) + \alpha \left( \frac{1}{B_1}\sum_{j=1}^{B_1}  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right) \right.\right.\\
& \left.\left. + \left( \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1}) -\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1})   -  \frac{1}{B_1}\sum_{j=1}^{B_1}\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) + \frac{1}{B_1}\sum_{j=1}^{B_1}\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1};\xi_t^{i,j})\right) \right\|^2\right] \\
\leq & (1-\alpha)^2 \mathbb{E}\left[\left\| \mathbf{v}_{t-1} - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right\|^2\right] + 2\alpha^2\mathbb{E}\left[\left\|\frac{1}{B_1}\sum_{j=1}^{B_1}  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right\|^2\right]\\
& +2\mathbb{E}\left[\left\|\frac{1}{B_1}\sum_{j=1}^{B_1}  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) -  \frac{1}{B_1}\sum_{j=1}^{B_1}  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1};\xi_t^{i,j}) -  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1}) + \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1})\right\|^2\right],
\end{align*}
where the last equality is due to $(a+b)^2 \leq 2a^2 +2b^2$ and the fact that 
\begin{align*}
    \mathbb{E} \left[ \alpha\left( \frac{1}{B_1}\sum_{j=1}^{B_1} \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right) +\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1}) -\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1})\right. \\
\left. -  \frac{1}{B_1}\sum_{j=1}^{B_1} \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) +  \frac{1}{B_1}\sum_{j=1}^{B_1} \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1};\xi_t^{i,j}) \right] =0.
\end{align*}
Then, we would bound the two terms, respectively. First, we have that:
\begin{align*}
    &\mathbb{E}\left[\left\| \frac{1}{B_1}\sum_{j=1}^{B_1}  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right\|^2\right]\\
    = & \frac{1}{B_1^2}\sum_{j=1}^{B_1}\mathbb{E}\left[\left\|\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right\|^2\right]\\
    =&\frac{1}{B_1^2}\sum_{j=1}^{B_1}\mathbb{E}\left[\left\|\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \nabla f_1(\mathbf{u}_{t-1}^0) \prod_{i=2}^K \nabla  f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) \right. \right. \\
    & \quad \left.\left. + \nabla f_1(\mathbf{u}_{t-1}^0) \prod_{i=2}^K \nabla  f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \nabla f_1(\mathbf{u}_{t-1}^0) \nabla f_2(\mathbf{u}_{t-1}^1) \prod_{i=3}^K \nabla  f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) \right. \right.\\ 
    &  \qquad \cdots \\
    & \quad \left.\left. +  \left(\prod_{i=1}^{K-1} \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right) \nabla f_K(\mathbf{u}_{t-1}^{K-1};\xi_t^{i,j}) - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1}) \right\|^2\right] \\
    \leq & \frac{K}{B_1^2}\sum_{j=1}^{B_1} \mathbb{E}\left[\left\|\prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) - \nabla f_1(\mathbf{u}_{t-1}^0) \prod_{i=2}^K \nabla  f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j})\right\|^2 \right] \\
    & + \frac{K}{B_1^2}\sum_{j=1}^{B_1}\mathbb{E}\left[\left\|\nabla f_1(\mathbf{u}_{t-1}^0) \prod_{i=2}^K \nabla  f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) -\nabla f_1(\mathbf{u}_{t-1}^0) \nabla f_2(\mathbf{u}_{t-1}^1) \prod_{i=3}^K \nabla  f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j})\right\|^2 \right] \\
    & + ... \\
    & + \frac{K}{B_1^2}\sum_{j=1}^{B_1}\mathbb{E}\left[\left\|\left(\prod_{i=1}^{K-1} \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right) \nabla f_K(\mathbf{u}_{t-1}^{K-1};\xi_t^{i,j})  - \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1})\right\|^2 \right]  \\
   \leq &  \frac{K}{B_1^2}\sum_{j=1}^{B_1} \sigma_J^2 L_f^{2K-2} \\

\end{align*}

\begin{equation}\label{diff}
    \begin{split}
        &\mathbb{E}\left[\left\|\frac{1}{B_1}\sum_{j=1}^{B_1}  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) -  \frac{1}{B_1}\sum_{j=1}^{B_1}  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1};\xi_t^{i,j}) -  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1}) + \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1})\right\|^2\right]\\

    \leq & \frac{1}{B_1^2}\sum_{j=1}^{B_1}\mathbb{E}\left[\left\|  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t-1}^{i-1};\xi_t^{i,j}) -  \prod_{i=1}^K \nabla f_i(\mathbf{u}_{t}^{i-1};\xi_t^{i,j}) \right\|^2\right]\\

<!-- proof-10: body_md -->
(1) For the first level, i.e., $i=1$, we have: 
    \begin{align*}
	    \sum_{t=1}^T \mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-1} - \mathbf{u}_{t}^{i-1}\right\|^2\right] = \sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{x}_{t+1} - \mathbf{x}_{t}\right\|^2\right] = \sum_{t=1}^T\mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right].
	\end{align*}
(2) For other levels, i.e., $2\leq i\leq K$, we have:
	\begin{align*}
		& \sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-1} - \mathbf{u}_{t}^{i-1}\right\|^2\right]\\
		=&\sum_{t=1}^T\mathbb{E}\left[\left\|\alpha \left(\mathbf{h}_{t+1}^{i-1} - \mathbf{u}_{t}^{i-1}\right) + \frac{(1-\alpha)}{B_1}\sum_{j=1}^{B_1}(f_{i-1}(\mathbf{u}_{t+1}^{i-2};\xi_{t+1}^{i-1,j}) - f_{i-1}(\mathbf{u}_{t}^{i-2};\xi_{t+1}^{i-1,j})) \right\|^2 \right]\\
\leq& 2\alpha^2\sum_{t=1}^T\mathbb{E}\left[\left\|f_{i-1}(\mathbf{u}_{t}^{i-2}) - \mathbf{u}_{t}^{i-1} + \mathbf{h}_{t+1}^{i-1}-f_{i-1}(\mathbf{u}_{t}^{i-2})\right\|^2\right]+ 2L_f^2\sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-2} - \mathbf{u}_{t}^{i-2}\right\|^2 \right]\\
	\leq&  2\alpha^2\sum_{t=1}^T \mathbb{E}\left[\left\|f_{i-1}(\mathbf{u}_{t}^{i-2}) - \mathbf{u}_{t}^{i-1}\right\|^2\right] + \frac{2\alpha^2 L_f^2}{B_1}\sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-2} - \mathbf{u}_{\tau}^{i-2}\right\|^2 \right] \\
 &\quad + 2L_f^2\sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-2} - \mathbf{u}_{t}^{i-2}\right\|^2 \right]\\
	\leq&  2\alpha^2\sum_{t=1}^T \mathbb{E}\left[\left\|f_{i-1}(\mathbf{u}_{t}^{i-2}) - \mathbf{u}_{t}^{i-1}\right\|^2\right]  + 4L_f^2\sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-2} - \mathbf{u}_{t}^{i-2}\right\|^2 \right].
	\end{align*} 
Denote that $\Upsilon^i =\sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{u}_t^i - f_i(\mathbf{u}_t^{i-1})\right\|^2\right]$ and  $Q^{i} = \sum_{t=1}^T\mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-1} - \mathbf{u}_{t}^{i-1}\right\|^2\right]$, we have $Q^{i} \leq 4L_f^2 Q^{i-1} + 2\alpha^2 \Upsilon^{i-1} $ for $i\geq 2$. Then we can get:
\begin{align*}
Q^{1} &\leq \mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right] \\
Q^{2} &\leq \left(4L_f^2\right) \mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right] +2\alpha^2 \Upsilon_t^{1} \\
Q^{3} &\leq \left(4L_f^2\right)^2 \mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right]  + 2\alpha^2\left( 4L_f^2  \Upsilon_t^{1}+ \Upsilon_t^{2}\right) \\
& \cdots\\
Q^{i} &\leq \left(4L_f^2\right)^{i-1} \mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right] + 2\alpha^2 \sum_{j=1}^{i-1} \left(4L_f^2\right)^{i-1-j} \Upsilon_t^{j}\\
&\leq \left(4L_f^2\right)^{i-1} \mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right] + 2\alpha^2 \sum_{j=1}^{K} \sum_{l=1}^{K} \left(4L_f^2\right)^{K-l} \Upsilon_t^{j}.
\end{align*}
When summing up, we have:
\begin{align*}
    &\sum_{t=1}^T\sum_{i=1}^K \mathbb{E}\left[\left\|\mathbf{u}_{t+1}^{i-1} - \mathbf{u}_{t}^{i-1}\right\|^2 \right] 
    = \sum_{t=1}^T\sum_{i=1}^{K} Q^{i}_t \\
    \leq &\sum_{t=1}^T\sum_{i=1}^{K}\left(4L_f^2\right)^{i-1} \mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right] + 2\alpha^2 K \sum_{t=1}^T\sum_{j=1}^{K} \sum_{l=1}^{K} \left(4L_f^2\right)^{K-l} \Upsilon_t^{j}  \\
                                                                                                                                                                                                             
    \leq &\left(\sum_{i=1}^{K}\left(4L_f^2\right)^{i-1}\right) \left(\sum_{t=1}^T\mathbb{E} \left[\eta^2 \left\|\mathbf{z}_t - \mathbf{x}_t\right\|^2\right] + 2\alpha^2K \sum_{t=1}^T\sum_{i=1}^{K} \mathbb{E}\left[\left\|\mathbf{u}_t^i - f_i(\mathbf{u}_t^{i-1})\right\|^2\right]  \right)
\end{align*}
Thus, we complete the proof of this lemma.

\end{align*}
