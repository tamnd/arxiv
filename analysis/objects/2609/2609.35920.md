<!-- theorem-1: body_md -->
\label{thm:fishburn}
Suppose that $n\ge1$ and $\mathcal F\subseteq2^{[n]}$ is dual intersecting.
Then there is an $i\in[n]$ such that
\[
|\mathcal F^L(i)|\ge|\mathcal F(i)|.
\]
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:maximal}
A family $\mathcal B\subseteq2^{[n]}$ is maximal intersecting if and only if
it is an upset and antipodal. In particular, every maximal intersecting family has $2^{n-1}$ members.
<!-- end lemma-1 -->
<!-- proposition-1: body_md -->
\label{prop:CLL}
Let $\mathcal B\subseteq2^{[n]}$ be maximal intersecting, and set $f=\mathbf{1}_{\mathcal B}$.
For $i\in[n]$, define
\[
 \lambda_i
 =4\sum_{\substack{\varnothing\ne S\subseteq[n]\\
\max_\prec S=i}}
\widehat f(S)^2.
\]
Then $\lambda_i\ge0$ for every $i\in[n]$ and $\sum_{i=1}^n\lambda_i=1$. Moreover, if $\omega:2^{[n]}\to\mathbb R_{\ge0}$ satisfies
\[
A\subseteq B\quad\Longrightarrow\quad
\omega(A)\ge\omega(B),
\] 
then
\begin{equation}\label{eq:weighted-star}
\sum_{S\in\mathcal B}\omega(S)
\le
\sum_{i=1}^n\lambda_i
\sum_{S\ni i}\omega(S)
\le
\max_{i\in[n]}
\sum_{S\ni i}\omega(S).
\end{equation}
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
The indicator function of any downset is nonnegative and decreasing, so it is an admissible weight in~\eqref{eq:weighted-star}.
If $z$ is last in the order $\prec$, then $\max_\prec S=z$ if and only if $z\in S$. 
Hence, by
\eqref{eq:fourier-influence},
\begin{equation}\label{eq:last-coefficient}
 \lambda_z
 =4\sum_{S\ni z}\widehat f(S)^2
 =\operatorname{Inf}_z(f).
\end{equation}
<!-- end remark-1 -->
<!-- lemma-2: body_md -->
\label{basiclemma}
Suppose that $\mathcal A$ is an upset.
Then the Boolean dual $\mathcal A^\dagger$ is an upset.
If $\mathcal A$ is intersecting, then $\mathcal A\subseteq\mathcal A^\dagger$.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:normalized}
Let $n\ge1$. 
Suppose that $\mathcal H\subseteq2^{[n]}$ is a nonempty intersecting upset and $\mathcal E\subseteq2^{[n]}$ is a downset. Then
\begin{equation}\label{eq:normalized}
|\mathcal H\cap\mathcal E|
\le \frac{|\mathcal H|}{2^{n-1}}
\max_{i\in[n]}|\mathcal E(i)|.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
Define
\[
\mathcal K
=
\mathcal H
\cup
\bigl\{T\cup\{n+1\}:T\in\mathcal H^\dagger\bigr\}
\subseteq 2^{[n+1]}.
\]
By Lemma~\ref{basiclemma}, both $\mathcal H$ and $\mathcal H^\dagger$ are
upsets, and $\mathcal H\subseteq\mathcal H^\dagger$.
We will apply Proposition~\ref{prop:CLL} to $\mathcal K$.
We show that $\mathcal K$ is an upset. 

Let $A\in\mathcal K$
and let $A\subseteq B\subseteq[n+1]$.
If $n+1\notin B$, then neither $A$ nor $B$ contains $n+1$.
Thus $A\in\mathcal H$, and the upward closure of $\mathcal H$
gives $B\in\mathcal H\subseteq\mathcal K$.
If $n+1\in A$, then
\[
A\setminus\{n+1\}\in\mathcal H^\dagger,
\qquad
A\setminus\{n+1\}\subseteq B\setminus\{n+1\}.
\]
Since $\mathcal H^\dagger$ is an upset, we obtain
$B\setminus\{n+1\}\in\mathcal H^\dagger$, and hence
$B\in\mathcal K$.
Suppose that $n+1\notin A$ but $n+1\in B$.
Then
\[
A\in\mathcal H\subseteq\mathcal H^\dagger,
\qquad
A\subseteq B\setminus\{n+1\}.
\]
Again, the upward closure of $\mathcal H^\dagger$ gives
$B\setminus\{n+1\}\in\mathcal H^\dagger$, so $B\in\mathcal K$.
Therefore, $\mathcal K$ is an upset.
For every $T\subseteq[n]$, the definition of $\mathcal H^\dagger$ gives
\[
T\in\mathcal K
\iff T\in\mathcal H  \iff [n]\setminus T\notin\mathcal H^\dagger
 \iff ([n]\setminus T)\cup\{n+1\}\notin\mathcal K.
\]
Since $T$ and $([n]\setminus T)\cup\{n+1\}$ are complements in $[n+1]$, $\mathcal K$ is antipodal. 
Thus Lemma~\ref{lem:maximal} implies that $\mathcal K$ is maximal intersecting.

Apply Proposition~\ref{prop:CLL} to  $\mathcal K\subseteq 2^{[n+1]}$.
Choose a total order $\prec$ on $[n+1]$ in which $n+1$
comes last, and put $f=\mathbf{1}_{\mathcal K}$.
Let $\lambda_1,\ldots,\lambda_{n+1}$ be the coefficients
associated with $f$ and this order.
In particular, Proposition~\ref{prop:CLL} gives
\[
\lambda_i\ge 0
\quad\text{and}\quad
\sum_{i=1}^{n+1}\lambda_i=1.
\]

Since $n+1$ is last in the order, the condition
$\max_{\prec}S=n+1$ is equivalent to $n+1\in S$.
Thus,
\[
\lambda_{n+1}
=
4\sum_{\substack{S\subseteq[n+1]\\ n+1\in S}}
\widehat f(S)^2
=
\operatorname{Inf}_{n+1}(f).
\]
To compute $\operatorname{Inf}_{n+1}(f)$, the definition
of $\mathcal K$ gives, for every $T\subseteq[n]$,
\[
f(T)=\mathbf{1}_{\mathcal H}(T),
\qquad
f(T\cup\{n+1\})=\mathbf{1}_{\mathcal H^\dagger}(T).
\]
Since $\mathcal H\subseteq\mathcal H^\dagger$, these
two values differ precisely when
$T\in\mathcal H^\dagger\setminus\mathcal H$.
Then we have 
\[
\begin{aligned}
\lambda_{n+1}
=\operatorname{Inf}_{n+1}(f)
=\frac{1}{2^n}\sum_{T\subseteq[n]}
  \bigl|f(T\cup\{n+1\})-f(T)\bigr|=\frac{|\mathcal H^\dagger\setminus\mathcal H|}{2^n}.
\end{aligned}
\]


Since $\mathcal H^\dagger=2^{[n]}\setminus\check{\mathcal H}$ and $|\check{\mathcal H}|=|\mathcal H|$,
we have 
\[
 |\mathcal H^\dagger|
 =|2^{[n]}|-|\check{\mathcal H}|
 =2^n-|\mathcal H|,\qquad |\mathcal H^\dagger\setminus\mathcal H|
 =|\mathcal H^\dagger|-|\mathcal H|
 =2^n-2|\mathcal H|.
\]
Therefore
\[
 \lambda_{n+1}
 =\frac{|\mathcal H^\dagger\setminus\mathcal H|}{2^n}
 =1-\frac{|\mathcal H|}{2^{n-1}}.
\]
By Proposition~\ref{prop:CLL}, $\sum_{i=1}^{n+1}\lambda_i=1$. Then
\begin{equation}\label{eq:lambda-base}
 \sum_{i=1}^{n}\lambda_i
 =\frac{|\mathcal H|}{2^{n-1}}.
\end{equation}

For the downset $\mathcal E$,
define $\omega:2^{[n+1]}\to\mathbb R_{\ge0}$ by
\[
 \omega(T)=
 \begin{cases}
  \mathbf{1}_{\mathcal E}(T),&n+1\notin T,\\
  0,&n+1\in T.
 \end{cases}
\]
In particular, every set containing $n+1$ has weight zero.
This weight is nonnegative and decreasing because $\mathcal E$ is a downset.
Indeed, let
$A\subseteq B\subseteq[n+1]$. If $\omega(B)=0$, then
$\omega(A)\ge\omega(B)$ automatically. If $\omega(B)=1$,
then $B\in\mathcal E$, and the downward closure of $\mathcal E$
implies $A\in\mathcal E$. Hence $\omega(A)=\omega(B)=1$.

                                                
                           
Since $\omega$ is the indicator
of $\mathcal E$, the total weight of $\mathcal K$ is
$|\mathcal K\cap\mathcal E|$. Moreover, every member of $\mathcal E$
lies in $2^{[n]}$, where $\mathcal K$ agrees with $\mathcal H$.
Therefore,
\[
\sum_{T\in\mathcal K}\omega(T)
=
|\mathcal K\cap\mathcal E|
=
|\mathcal H\cap\mathcal E|.
\]
For each $i\in[n]$, the total weight of the $i$-star
counts precisely the members of $\mathcal E$ that contain $i$.
Thus,
\[
\sum_{\substack{T\subseteq[n+1]\\i\in T}}\omega(T)
=
|\{T\in\mathcal E:i\in T\}|
=
|\mathcal E(i)|.
\]
The $(n+1)$-star has total weight zero, since no member
of $\mathcal E$ contains $n+1$.

Substituting these quantities into
\eqref{eq:weighted-star}, we obtain
\[
|\mathcal H\cap\mathcal E|
\le
\sum_{i=1}^{n}\lambda_i|\mathcal E(i)|
+\lambda_{n+1}\cdot0
=
\sum_{i=1}^{n}\lambda_i|\mathcal E(i)|.
\]

Finally, since each $\lambda_i$ is nonnegative, replacing
every $|\mathcal E(i)|$ by their maximum can only increase
the right-hand side. Using \eqref{eq:lambda-base}, we get
\[
\begin{aligned}
|\mathcal H\cap\mathcal E|
&\le
\left(\sum_{i=1}^{n}\lambda_i\right)
\max_{i\in[n]}|\mathcal E(i)|=
\frac{|\mathcal H|}{2^{n-1}}
\max_{i\in[n]}|\mathcal E(i)|.
\end{aligned}
\]
This completes the proof.

\label{thm:witness}
Let $n\ge2$, and let $\mathcal D\subseteq2^{[n]}$ be a downset such that $A\cup A'\ne[n]$ for all $A,A'\in\mathcal D$.
Let $\mathcal B\subseteq2^{[n]}$ be maximal intersecting, and choose $i\in[n]$ with $\operatorname{Inf}_i(\mathbf{1}_{\mathcal B})=\min_{j\in[n]}\operatorname{Inf}_j(\mathbf{1}_{\mathcal B})$.
Then
\[
 2\,|\mathcal B\cap\mathcal D(i)|\le|\mathcal D(i)|.
\]

<!-- proof-2: body_md -->
By relabelling the ground set, we may  assume that $i=n$, and
\begin{equation}\label{mini-condition}
\operatorname{Inf}_n(\mathbf{1}_{\mathcal B})
\le \operatorname{Inf}_j(\mathbf{1}_{\mathcal B}),
\qquad j\in[n-1].
\end{equation}

\noindent\textbf{Step 1. Transfer the counting problem to $[n-1]$.}
Define
\[
 \mathcal E=\{T\subseteq[n-1]:T\cup\{n\}\in\mathcal D\},
 \qquad
 \mathcal H=\mathcal B\cap2^{[n-1]},
 \qquad
 \mathcal C=2^{[n-1]}\setminus\mathcal H.
\]
Let $\check{\mathcal E}$ denote the family obtained by
taking complements of the members of $\mathcal E$
relative to $[n-1]$:
\[
\check{\mathcal E}
=\{[n-1]\setminus T:T\in\mathcal E\}.
\]
By Lemma~\ref{lem:maximal}, $\mathcal B$ contains exactly
one member of each complementary pair in $2^{[n]}$.
Thus, for every $T\subseteq[n-1]$,
\begin{equation}\label{eq:B-upper-section}
T\cup\{n\}\in\mathcal B
\iff [n-1]\setminus T\notin\mathcal B\iff [n-1]\setminus T\notin\mathcal H\iff [n-1]\setminus T\in\mathcal C.  
\end{equation}
Also,
\[
 T\cup\{n\}\in\mathcal D(n)\iff T\in\mathcal E\iff [n-1]\setminus T\in\check{\mathcal E}.
\]
Under the bijection $T\cup\{n\}\mapsto[n-1]\setminus T$, these equivalences yield
\begin{equation}\label{eq:witness-reduction}
|\mathcal B\cap\mathcal D(n)|=|\check{\mathcal E}\cap\mathcal C|,
 \qquad
 |\mathcal D(n)|=|\check{\mathcal E}|.
\end{equation}
It suffices to prove that 
\begin{equation}
2\,|\check{\mathcal E}\cap\mathcal C|\le |\check{\mathcal E}|.
\end{equation}


\medskip
\noindent\textbf{Step 2. Verify the hypotheses and apply
Lemma~\ref{lem:normalized}.}
If $\mathcal E=\varnothing$, then $\mathcal D(n)=\varnothing$, and the conclusion follows. 
Assume that $\mathcal E\ne\varnothing$.

Since $\mathcal D$ is a downset, $\mathcal E$ is a downset in $2^{[n-1]}$, and therefore $\check{\mathcal E}$ is an upset.
For $S,T\in\mathcal E$, the hypothesis on $\mathcal D$ gives
\[
(S\cup\{n\})\cup(T\cup\{n\})\ne[n].
\]
Consequently,
\[
([n-1]\setminus S)\cap([n-1]\setminus T)
=[n-1]\setminus(S\cup T)\ne\varnothing.
\]
Thus $\check{\mathcal E}$ is a nonempty intersecting upset.

By Lemma~\ref{lem:maximal}, $\mathcal B$ is an upset.
Hence $\mathcal H$ is an upset in $2^{[n-1]}$, and $\mathcal C$ is a downset.
Applying Lemma~\ref{lem:normalized} to $\check{\mathcal E}$ and $\mathcal C$ on the ground set $[n-1]$, we obtain
\begin{equation}\label{eq:witness-normalized}
 |\check{\mathcal E}\cap\mathcal C|
 \le\frac{|\check{\mathcal E}|}{2^{n-2}}
       \max_{j\in[n-1]}|\mathcal C(j)|.
\end{equation}
\medskip
\noindent\textbf{Step 3. Bound the sizes of $\mathcal C(j)$.}
We will show that $|\mathcal C(j)|\le 2^{n-3}$
for every $j\in[n-1]$.

Put $f=\mathbf{1}_{\mathcal B}$. 
Since $\mathcal B$ is an upset, $f(T\cup\{k\})-f(T)\ge0$ for every $T\subseteq[n]\setminus\{k\}$. 
Hence
\[
\begin{aligned}
 \operatorname{Inf}_k(f)
 =\frac{1}{2^{n-1}}
   \sum_{T\subseteq[n]\setminus\{k\}}
       \bigl(f(T\cup\{k\})-f(T)\bigr)
 =\frac{|\mathcal B(k)|
         -|\{A\in\mathcal B:k\notin A\}|}{2^{n-1}}.
\end{aligned}
\]
Since $\mathcal B$ contains exactly one set from each complementary pair, $|\mathcal B|=2^{n-1}$. 
Therefore
\begin{equation}\label{eq:influence-star}
 \operatorname{Inf}_k(f)
 =\frac{2|\mathcal B(k)|-2^{n-1}}{2^{n-1}}
 =\frac{|\mathcal B(k)|}{2^{n-2}}-1,
 \qquad k\in[n].
\end{equation}
By \eqref{eq:B-upper-section}, the sets in $\mathcal B(n)$ correspond bijectively to $\mathcal C$. 
Hence
\[
 |\mathcal B(n)|=|\mathcal C|=2^{n-1}-|\mathcal H|,
 \qquad
 \operatorname{Inf}_n(f)=1-\frac{|\mathcal H|}{2^{n-2}}.
\]

Now fix $j\in[n-1]$. 
The members of $\mathcal B(j)$ that do not contain $n$ are counted by $|\mathcal H(j)|$.
By \eqref{eq:B-upper-section}, the members of $\mathcal B(j)$ that contain $n$ correspond to the sets in $\mathcal C$ that do not contain $j$. 
Therefore
\[
\begin{aligned}
 |\mathcal B(j)|
 &=|\mathcal H(j)|+|\mathcal C|-|\mathcal C(j)|\\
 &=|\mathcal H(j)|
   +(2^{n-1}-|\mathcal H|)
   -(2^{n-2}-|\mathcal H(j)|)\\
 &=2^{n-2}-|\mathcal H|+2|\mathcal H(j)|.
\end{aligned}
\]
Using \eqref{eq:influence-star} again gives
\[
 \operatorname{Inf}_j(f)
 =\frac{2|\mathcal H(j)|-|\mathcal H|}{2^{n-2}}.
\]

By \eqref{mini-condition}, $\operatorname{Inf}_n(f)\le\operatorname{Inf}_j(f)$.
The formulas above give
\[
 1-\frac{|\mathcal H|}{2^{n-2}}
 \le
 \frac{2|\mathcal H(j)|-|\mathcal H|}{2^{n-2}},
\]
so $|\mathcal H(j)|\ge2^{n-3}$. 
Since $\mathcal H(j)$ and $\mathcal C(j)$ partition the subsets of $[n-1]$
containing $j$,
\[
 |\mathcal C(j)|
 =2^{n-2}-|\mathcal H(j)|
 \le2^{n-3},
 \qquad j\in[n-1].
\]
Thus \eqref{eq:witness-normalized} yields
\[
 |\check{\mathcal E}\cap\mathcal C|
 \le\frac{|\check{\mathcal E}|}{2^{n-2}}
       \max_{j\in[n-1]}|\mathcal C(j)|
 \le\frac{|\check{\mathcal E}|}{2}.
\]
Finally, \eqref{eq:witness-reduction} gives
\[
 2|\mathcal B\cap\mathcal D(n)|
 =2|\check{\mathcal E}\cap\mathcal C|
 \le|\check{\mathcal E}|
 =|\mathcal D(n)|,
\]
as required.
