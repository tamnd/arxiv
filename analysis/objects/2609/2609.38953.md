<!-- theorem-1: body_md -->
\label{thm:main-algebraic}
For every nondegenerate planar triangle $T$,
\[
 H\left(\mu_2,\mu_3\right)L^2\leq16\pi^2,\qquad
 G\left(\mu_2,\mu_3\right)A\leq\frac{4\pi^2}{3\sqrt3}.
\]
Equality in either inequality holds if and only if $T$ is equilateral.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:stability-proof}
For every nondegenerate planar triangle,
\[
 1-\frac{H\left(\mu_2,\mu_3\right)L^2}{16\pi^2}
 \geq\frac{\mathcal A\left(T\right)}{4800},\qquad
 1-\frac{27\mu_2\mu_3A^2}{16\pi^4}
 \geq\frac{2}{1251}\mathcal A\left(T\right).
\]
<!-- end corollary-1 -->
<!-- lemma-1: body_md -->
\label{lem:ritz-proof}
Let $f_1,f_2$ be linearly independent real functions in
$H^1_*\left(T\right)$, and define their mass and energy matrices by
\[
 M_{ij}=\int_Tf_if_j\,\mathrm{d}\boldsymbol{x},\qquad
 K_{ij}=\int_T\nabla f_i\cdot\nabla f_j\,\mathrm{d}\boldsymbol{x},
 \qquad i,j\in\left\{1,2\right\}.
\]
Both matrices are positive definite. The generalized eigenvalues,
called the Ritz values, are the numbers $\nu$ for which
$Kc=\nu Mc$ has a solution $c\in\mathbb R^2\setminus\left\{0\right\}$.
Their ordered values satisfy
\[
 0<\nu_1\leq\nu_2,\qquad
 \mu_2\leq\nu_1,\qquad \mu_3\leq\nu_2,
\]
and
\[
 H\left(\nu_1,\nu_2\right)
 =\frac{2\det K}{\operatorname{tr}\left(\operatorname{adj}\left(K\right)M\right)},
 \qquad
 \nu_1\nu_2=\frac{\det K}{\det M}.
\]

Set $V=\operatorname{span}\left\{f_1,f_2\right\}$. A coefficient vector
$c=\left(c_1,c_2\right)^{\mathsf T}$ represents the function
$f_c=c_1f_1+c_2f_2$, and
\[
 \int_T f_c^2\,\mathrm{d}\boldsymbol{x}=c^{\mathsf T}Mc,
 \qquad
 \int_T\left|\nabla f_c\right|^2\,\mathrm{d}\boldsymbol{x}
 =c^{\mathsf T}Kc.
\]
The first expression is positive for $c\ne0$ by independence.
If the second vanished, $f_c$ would be constant on the connected
triangle. Its zero mean would then give $f_c=0$, contradicting
independence. Thus both quadratic forms are positive definite.

The finite-dimensional variational characterization of generalized
eigenvalues \cite[Chapter~7]{HornJohnson2013} gives
\[
 \nu_1=\min_{0\ne f\in V}\mathscr R_T\left(f\right),
 \qquad
 \nu_2=\max_{0\ne f\in V}\mathscr R_T\left(f\right).
\]
Restricting the minimization defining $\mu_2$ to $V$ gives
$\mu_2\leq\nu_1$. In the characterization of $\mu_3$, the space $V$
is one admissible two-dimensional competitor, so
$\mu_3\leq\nu_2$. These are the two applications of the Neumann
min--max principle needed here.

It remains to express the symmetric combinations without solving
for the Ritz values separately. Taking determinants in the
generalized characteristic equation gives
$\nu_1\nu_2=\det K/\det M$. For the harmonic mean it is more
convenient to use reciprocals. The eigenvalues of $K^{-1}M$ are
$\nu_1^{-1},\nu_2^{-1}$; hence
\[
 \frac1{\nu_1}+\frac1{\nu_2}
 =\operatorname{tr}\left(K^{-1}M\right)
 =\frac{\operatorname{tr}\left(\operatorname{adj}\left(K\right)M\right)}{\det K}.
\]
The definition of $H$ now gives the asserted formula.

We compare the angular changes of the two sides of $W\leq Y$.
It suffices to show that the angular derivative of $Y$ is larger than that of $W$.
At fixed $\eta$, the identity $v=\eta^3c$ gives
\[
 \left(F_H\right)_c=\eta^3Y_v-W_c.
\]
We will prove $Y_v>-9/32$ and $-W_c\geq\left(9/32\right)\eta^3$.
Their matching constants imply strict positivity of this derivative.
The geometric estimate comes from the elementary symmetric functions
of the three normalized side lengths. For the spectral estimate,
differentiation separates the full determinant contribution from the
complementary determinant contribution: the former is positive, and
the possible loss from the latter is bounded by the same constant.
Combining the two estimates leaves only the isosceles endpoint $c=-1$
to be checked.

\paragraph{The geometric part.}
Put $t_j=\sqrt{1-\eta\cos\left(\theta+2j\pi/3\right)}$,
$S_{\rm geom}=\sum_jt_j$, $Q_{\rm geom}=\sum_{i<j}t_it_j$, and

\[
 p_{\rm geom}^2=1-\frac34u-\frac14v,\qquad
 \left(W-3\right)^2=12-3u+8p_{\rm geom}S_{\rm geom}.
\]
Subscripts $c,v$ below denote partial derivatives with $\eta,u$,

\[
 \left(p_{\rm geom}\right)_c=-\frac{\eta^3}{8p_{\rm geom}},\qquad
 \left(S_{\rm geom}\right)_c=\frac{W_c}{2S_{\rm geom}}.

\paragraph{The isosceles boundary.}
Define the degree-ten polynomial $p_H$ by the exact identity
\begin{equation}
 8\pi^2\left(1-t^2\right)H_D\left(t^2,-t^3\right)
 -\left(9-t^2\right)H_U\left(t^2,-t^3\right)=t^2p_H\left(t\right).                \label{eq:iso-target}
\end{equation}
Its coefficient vector, in ascending powers of $t$, differs by less
than $10^{-6}$ in every entry from
\begin{equation}
\begin{array}{rrrrrr}
11.360043&6.718733&-35.527940&-18.530405&35.266653&13.807750\\
-13.711071&-1.881133&2.377526&-.314987&-.102749&
\end{array}                                                    \label{eq:iso-coefficients}
\end{equation}
Let $t_*=31/40>\eta_0$. These bounds imply, for $0\leq t\leq t_*$,
\begin{align*}
 p_H'\left(t\right)&\leq6.719-71.055t-55.591t^2+141.067t^3
               +69.039t^4-82.266t^5+19.021t^7\\
 &\leq6.719-16t-8.8t^2.
\end{align*}
For the last step, the function $141.067x-82.266x^2$ increases on
$0\leq x\leq t_*^2$, and
\[
 141.067t_*^2-82.266t_*^4<55.050876,\qquad
 69.039t_*^2+19.021t_*^5<46.784471.
\]
Thus $p_H$ decreases on $\left[2/5,t_*\right]$. On $\left[0,2/5\right]$, discarding
positive terms in \eqref{eq:iso-coefficients} gives $p_H>4.43$. Substitution at the single
rational endpoint gives
\[
 p_H\left(31/40\right)>.16819>\frac16.
\]
Hence $p_H>1/6$ throughout $\left[0,t_*\right]$. Equations \eqref{eq:perimeter-upper} and \eqref{eq:iso-target} prove
$F_H>0$ at $c=-1$ for $0<\eta\leq\eta_0$.

Since $\left(F_H\right)_c>0$, the same conclusion holds for every
$c\in\left[-1,1\right]$ at fixed $0<\eta\leq\eta_0=1397/1803$.
