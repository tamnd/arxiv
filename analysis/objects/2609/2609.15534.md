\label{main}

a bounded open set $\Omega$ satisfying
$Q_0\cup Q_+\cup Q_-\Subset\Omega$, exponents

\[
 C_0=C_0(\Theta),\quad

\item[(i)](Weak Harnack inequality) For every nonnegative strong supersolution $w$ of $\mathscr L_{a,b,c}$ in

\label{lem:leverage}
For any $s<t$, the matrix $\Sigma_{s,t}$ is invertible and
\begin{equation}
 \mathbf L_s^\top \Sigma_{s,t}^{-1}\mathbf L_s=\frac{4}{t-s}I_d,
 \qquad

 \label{eq:leverage}
\end{equation}
Moreover, for $\kappa{\geqslant}1$ and every $s,r>0$,
\begin{equation}
 \kappa\Sigma_{s,s+r}\preceq
\Sigma_{s,s+4\kappa r}.
 \label{eq:gram-compare}
\end{equation}

<!-- proof-1: body_md -->
Let $r=t-s$.  {By a change of variables,} we have
\begin{align}\label{As8}
\Sigma_{s,t}=\Sigma_{s,s+r}=\int_0^r\mathbf L_{s+u}\mathbf L_{s+u}^\top\mathrm d u
=\mathcal S_s\left(\int_0^r\mathbf L_{u}\mathbf L_{u}^\top\mathrm d u\right) \mathcal S^\top_s,
\end{align}
where
\[
\mathcal S_s=
 \begin{pmatrix}
  I_d&-sI_d\\[1mm]
  0&I_d
 \end{pmatrix},\quad
 \mathcal S_s^{-1}=
 \begin{pmatrix}
  I_d&sI_d\\[1mm]
 0&I_d
 \end{pmatrix}.
\]
Hence, it suffices to consider $s=0$.  In this case
\[
 \Sigma_{0,r}^{-1}
 =\begin{pmatrix}
  12r^{-3}I_d&6r^{-2}I_d\\
  6r^{-2}I_d&4r^{-1}I_d
 \end{pmatrix},

  0_d&I_d
 \end{pmatrix}
 \preceq 4
 \begin{pmatrix}
  \frac{r^2}{3}I_d&-\frac{r}{2}I_d\\[1mm]

 \end{pmatrix}=
 \frac{4}{r}\int_0^r\mathbf L_{u}\mathbf L_{u}^\top\mathrm d u.

$\Sigma_2=\Sigma_1+\Sigma$.  By the Gaussian semigroup property,
\[
 q_s^{ij}*g_{\Sigma_2}
 =q_s^{ij}*g_{\Sigma_1}*g_\Sigma=:u_s^{ij}*g_\Sigma,
 \qquad
 \mu_s*g_{\Sigma_2}
 =\mu_s*g_{\Sigma_1}*g_\Sigma=:p_s*g_\Sigma.

|u_s^{ij}*g_\Sigma|^2
 \leqslant
((|u_s^{ij}|^2/p_s^{2-\digamma})*g_\Sigma)\,(p_s^{2-\digamma}*g_\Sigma)

 (\nabla\overline{\mathfrak a}_{s,t}^{\sigma,ij})^\top
 \Sigma_{s,t}\nabla\overline{\mathfrak a}_{s,t}^{\sigma,ij}\mathrm d z.

{Fix $s<t$, write $\Sigma:=\Sigma_{s,t}$, and set }

 \partial_\sigma q^{ij}=\Sigma:\nabla^2q^{ij}.

 &-4(\digamma-1)\int_{\mathbb R^{2d}}p^\digamma \overline{\mathfrak a}^{ij}

        (\nabla\log p)^\top\Sigma\nabla\log p\mathrm d z.
\end{split}

 +4(\digamma-1)\overline{\mathfrak a}^{ij}(\nabla \overline{\mathfrak a}^{ij})^\top\Sigma\nabla\log p +\digamma(\digamma-1)|\overline{\mathfrak a}^{ij}|^2

The classical de Bruijn equality states that the derivative
of Shannon entropy along Gaussian convolution is the Fisher
information (see \cite{Stam1959}). More precisely, for $f\geqslant0$,

\int_{\mathbb R^{2d}}(f*g_{\sigma\Sigma})
\left|\Sigma^{1/2}\nabla\log (f*g_{\sigma\Sigma})\right|^2\,\mathrm d z,

The above \eqref{Aw3} is a variant of the classical de Bruijn equality
(see \cite[Proposition 1]{Chafai2004}).

\label{thm:entropy}
For Lebesgue-a.e. $t>0$, the measure $\mu_t$ is absolutely continuous with
respect to Lebesgue measure.  Moreover, there exists $\digamma=\digamma(d,\lambda,\Lambda)\in(1,1+\frac{1}{2d})$
such that, for every $T>0$,
\begin{align}\label{Dual1}

 \leqslant C_{d,\lambda,\Lambda}\,T^{1-2d(\digamma-1)}.
\end{align}

<!-- proof-5: body_md -->
Note that
\begin{align*}
 \sum_{i,j}|\mathscr D_{s,j}
       \overline{\mathfrak a}_{s,t}^{\sigma,ij}|^2
 \stackrel{\eqref{DD5}}{\leqslant}\sum_{i,j}(\nabla\overline{\mathfrak a}_{s,t}^{\sigma,ij})^\top
       \mathbf L_s\mathbf L_s^\top
       \nabla\overline{\mathfrak a}_{s,t}^{\sigma,ij}
\stackrel{\eqref{eq:leverage}}{\leqslant}\frac4{t-s}
       \sum_{i,j}(\nabla\overline{\mathfrak a}_{s,t}^{\sigma,ij})^\top
       \Sigma_{s,t}\nabla\overline{\mathfrak a}_{s,t}^{\sigma,ij}.
\end{align*}
By \eqref{eq:entropy-upper} and \eqref{eq:debruijn}, for
$\sigma\in(0,\lambda)$ and Lebesgue-a.e. $s\in(0,t)$,
\[
 \frac{\mathrm d}{\mathrm d s}\int_{\mathbb R^{2d}}(p_{s,t}^\sigma)^\digamma\mathrm d z
 \leqslant
 -\frac{d\digamma^2(\digamma-1)}
 {2(2-\digamma)(\lambda-\sigma)(t-s)}
 \partial_\sigma\mathcal V_s^\digamma(\sigma\Sigma_{s,t}).
\]
Integrating both sides in $\sigma$ from $\sigma_0:=\lambda/4$ to $\sigma_1:=\lambda/2$, we have
\begin{align*}
 &\frac{\mathrm d}{\mathrm d s}\int_{\sigma_0}^{\sigma_1}\!\!\!
       \int_{\mathbb R^{2d}}(p_{s,t}^\sigma)^\digamma\mathrm d z\,\mathrm d\sigma\leqslant
 \frac{d\digamma^2(\digamma-1)}{(2-\digamma)\lambda}
 \frac{\mathcal V_s^\digamma(\sigma_0\Sigma_{s,t})
       -\mathcal V_s^\digamma(\sigma_1\Sigma_{s,t})}{t-s}.
\end{align*}
Furthermore, for $\varepsilon\in(0,t)$,  integrating {in} $s$ from $0$ to $t-\varepsilon$, we get
\begin{align*}
 &\int_{\sigma_0}^{\sigma_1}\!\!\!
       \int_{\mathbb R^{2d}}(p_{t-\varepsilon,t}^\sigma)^\digamma\mathrm d z\,\mathrm d\sigma
       -\int_{\sigma_0}^{\sigma_1}\!\!\!
       \int_{\mathbb R^{2d}}(p_{0,t}^\sigma)^\digamma\mathrm d z\,\mathrm d\sigma
       \leqslant
 \frac{d\digamma^2(\digamma-1)}{(2-\digamma)\lambda}
 \int^{t-\varepsilon}_0\frac{\mathcal V_s^\digamma(\sigma_0\Sigma_{s,t})
       -\mathcal V_s^\digamma(\sigma_1\Sigma_{s,t})}{t-s}\mathrm d s.
\end{align*}
Finally, for $\varepsilon\in(0,T)$, 
integrating in $t$ from
$\varepsilon$ to $T$ and applying Fubini's theorem together with the change of variables $r=t-s$, 
we obtain
\begin{equation}\label{eq:terminal-entropy}
\begin{aligned}
 &\int_\varepsilon^T\!\!\!\int_{\sigma_0}^{\sigma_1}\!\!\!
       \int_{\mathbb R^{2d}}|p_{t-\varepsilon,t}^\sigma|^\digamma
       \mathrm d z\,\mathrm d\sigma\,\mathrm d t
       -\int_\varepsilon^T\!\!\!\int_{\sigma_0}^{\sigma_1}\!\!\!
       \int_{\mathbb R^{2d}}|p_{0,t}^\sigma|^\digamma
       \mathrm d z\,\mathrm d\sigma\,\mathrm d t\\
 &\quad\leqslant
 \frac{d\digamma^2(\digamma-1)}{(2-\digamma)\lambda}
 \int_0^{T-\varepsilon}\!\!\!\int_\varepsilon^{T-s}
 \frac{\mathcal V_s^\digamma(\sigma_0\Sigma_{s,s+r})
       -\mathcal V_s^\digamma(\sigma_1\Sigma_{s,s+r})}{r}
       \mathrm d r\,\mathrm d s.
\end{aligned}
\end{equation}
For notational simplicity, we write
\[
 F^{\sigma_0}_s(r):=\mathcal V_s^\digamma(\sigma_0\Sigma_{s,s+r}),\qquad r>0.
\]
Since $\sigma_1/\sigma_0=2$, by \eqref{eq:gram-compare} we have
\[
 \sigma_0\Sigma_{s,s+r}\preceq\sigma_1\Sigma_{s,s+r}
 \preceq\sigma_0\Sigma_{s,s+8r}.
\]
Thus, by Lemma \ref{lem:data-processing},
\[
F^{\sigma_0}_s(r)= \mathcal V_s^\digamma(\sigma_0\Sigma_{s,s+r})
 \geqslant\mathcal V_s^\digamma(\sigma_1\Sigma_{s,s+r})
 \geqslant\mathcal V_s^\digamma(\sigma_0\Sigma_{s,s+8r})=F^{\sigma_0}_s(8r)\geqslant 0.
\]
Consequently, for $R>\varepsilon$,
\begin{equation}\label{eq:log-telescope}
\begin{aligned}
 0&\leqslant\int_\varepsilon^R
 \frac{\mathcal V_s^\digamma(\sigma_0\Sigma_{s,s+r})
       -\mathcal V_s^\digamma(\sigma_1\Sigma_{s,s+r})}{r}\mathrm d r
       \leqslant\int_\varepsilon^R\frac{F^{\sigma_0}_s(r)-F^{\sigma_0}_s(8r)}r\mathrm d r\\
&=\int_\varepsilon^{8\varepsilon}\frac{F^{\sigma_0}_s(r)}r\mathrm d r
   -\int_R^{8R}\frac{F^{\sigma_0}_s(r)}r\mathrm d r\leqslant F^{\sigma_0}_s(\varepsilon)
   \int_\varepsilon^{8\varepsilon}\frac{\mathrm d r}r
\leqslant F^{\sigma_0}_s(\varepsilon)\log 8.
\end{aligned}
\end{equation}
Hence,
\begin{align}\label{Ag1}

 \frac{\mathcal V_s^\digamma(\sigma_0\Sigma_{s,s+r})
       -\mathcal V_s^\digamma(\sigma_1\Sigma_{s,s+r})}{r}
       \mathrm d r\,\mathrm d s\leqslant\log 8\int_0^{T-\varepsilon}F^{\sigma_0}_s(\varepsilon)\mathrm d s.
\end{align}
By \eqref{eq:variance-budget}, for $\sigma\in[\sigma_0,\sigma_1]$, we have
\[

 \leqslant d\Lambda^2\int_{\mathbb R^{2d}}
             |p_{s,s+\varepsilon}^{\sigma_0}|^\digamma\mathrm d z
             \leqslant d\Lambda^2 2^{d\digamma}\int_{\mathbb R^{2d}}
             |p_{s,s+\varepsilon}^{\sigma}|^\digamma\mathrm d z,

 \leqslant\left(\frac{\sigma}{\sigma_0}\right)^d g_{\sigma\Sigma}
 \leqslant2^d g_{\sigma\Sigma}.

 {(2-\digamma)\lambda^2}
 \leqslant\frac12.

       \mathrm d z\,\mathrm d\sigma\,\mathrm d t\leqslant

       \mathrm d z\,\mathrm d\sigma\,\mathrm d t\\
&\quad       +\frac{4d^2\Lambda^2\digamma^2(\digamma-1)\,2^{d\digamma}\log8}

       \mathrm d z\,\mathrm d\sigma\,\mathrm d t\\
 &\qquad\leqslant C_{d,\lambda,\digamma}\int^T_\varepsilon t^{-2d(\digamma-1)}\mathrm d t+

       \mathrm d z\,\mathrm d\sigma\,\mathrm d t.
\end{align*}

 \mathrm d z\,\mathrm d\sigma\,\mathrm d t
 \leqslant \frac{2C_{d,\lambda,\digamma}T^{1-2d(\digamma-1)}}{1-2d(\digamma-1)}.

&\leqslant\varliminf_{\varepsilon\to 0}\frac{1}{\sigma_1-\sigma_0} \int_\varepsilon^T\!\!\!\int_{\sigma_0}^{\sigma_1}\!\!\!

and that \eqref{Dual1} holds.

Let $\digamma\in(1,1+\frac1{2d})$ be as in Theorem \ref{thm:entropy},

For $\boldsymbol\beta\equiv0$, {it follows from the change of variables, Theorem \ref{thm:entropy},

&\quad\leqslant \|f\|_{L^{\digamma^*}([0,T]\times\mathbb R^{2d})}(C_{d,\lambda,\Lambda}\,T^{1+2d-2d\digamma})^{1/\digamma}.
\label{DD8}
\end{align}
For general $\boldsymbol\beta$, {we use Girsanov's theorem}.

$\theta_t:=\sigma_t^{-1}\boldsymbol\beta_t$.  Since
$|\theta_t|^2\leqslant \hbar^2/(2\lambda)$, by Novikov's condition, we may
define a probability measure $\mathbb Q$ by
\[
 \frac{\mathrm d\mathbb Q}{\mathrm d\mathbb P}
 =\exp\left(
 -\int_0^T\theta_t\cdot\mathrm d W_t
 -\frac12\int_0^T|\theta_t|^2\mathrm d t
 \right).
\]
Under $\mathbb Q$,
\[
 W_t^{\mathbb Q}:=W_t+\int_0^t\theta_s\mathrm d s,\qquad 0\leqslant t\leqslant T,
\]
is a Brownian motion, and therefore
\[
 \mathrm d\mathsf X_t=\mathsf V_t\mathrm d t,\qquad
 \mathrm d\mathsf V_t=\sqrt{2\mathbf A_t}\mathrm d W_t^{\mathbb Q}.
\]
For $q>1$ and $q^*:=q/(q-1)$, by H\"older's inequality, we have
\begin{align}
 \mathbb E\left(\int_0^T f(t,\mathsf X_t,\mathsf V_t)\mathrm d t\right)
{\leqslant}
 \left(\mathbb E_{\mathbb Q}
 \left(\frac{\mathrm d\mathbb P}{\mathrm d\mathbb Q}\right)^q\right)^{1/q}
 \left(\mathbb E_{\mathbb Q}\left|\int_0^T f(t,\mathsf X_t,\mathsf V_t)\mathrm d t\right|^{q_*}\right)^{1/q_*}.\label{DD9}
\end{align}
By the exponential martingale and the bound
$|\theta_t|^2\leqslant \hbar^2/(2\lambda)$, we have
\begin{equation}\label{DD7}
 \left(\mathbb E_{\mathbb Q}
 \left(\frac{\mathrm d\mathbb P}{\mathrm d\mathbb Q}\right)^q\right)^{1/q}
 \leqslant
 \exp\left(\frac{(q-1)\hbar^2T}{4\lambda}\right).
\end{equation}
Combining \eqref{DD8}, \eqref{DD9} and \eqref{DD7}, we obtain the desired estimate with $p=q^*\digamma^*$.

 \frac{\mathrm d\langle M^\phi \rangle_s}{\mathrm d s}
 \preceq
 2\Lambda I_d.
\]
By Proposition \ref{prop:sb}, we have
\begin{align}\label{Aw1}
 \mathbb Q^\phi\bigg(
 {\sup_{0\leqslant s\leqslant h}}|M^\phi_s|<R
 \bigg)\geqslant {\mathrm e^{-\mu_1h/R^2}}.
\end{align}
For $p>1$, the exponential-martingale identity and
\eqref{eq:girsanov-energy} yield
\[
 \mathbb E_{\mathbb P}\left(\frac{\mathrm d\mathbb Q^\phi}{\mathrm d\mathbb P}\right)^p
 \leqslant\exp\left(\frac{p(p-1)(\mathcal A_\phi+\hbar^2h)}{2\lambda}\right).
\]
Now, by H\"older's inequality, we get
\[
\mathbb Q^\phi\bigg(
 {\sup_{0\leqslant s\leqslant h}}|M^\phi_s|<R
 \bigg)\leqslant \mathbb P\bigg(
 {\sup_{0\leqslant s\leqslant h}}|M^\phi_s|<R
 \bigg)^{1-1/p}
 \exp\left(\frac{(p-1)(\mathcal A_\phi+\hbar^2h)}{2\lambda}\right),
\]
which together with \eqref{Aw1} yields
\[
 \mathbb P\bigg(
 {\sup_{0\leqslant s\leqslant h}}|M^\phi_s|<R
 \bigg)
 \geqslant\exp\left(-\frac{p\mu_1h}{(p-1)R^2}
 -\frac{p(\mathcal A_\phi+\hbar^2h)}{2\lambda}\right).
\]
With $a:=\mu_1h/R^2$ and
$c:=(\mathcal A_\phi+\hbar^2h)/(2\lambda)$,
{if $c>0$, the exponent is minimized by
$p-1=\sqrt{a/c}$.  If $c=0$, we let $p\to\infty$.  In either case, its
infimum over $p>1$ is $(\sqrt a+\sqrt c)^2$.}
This proves \eqref{eq:tube}.

\label{thm:ink-spots}
Let $E, F$ be two measurable {subsets} of $\mathbb R^{1+2d}$ with
$$
E \subset F\cap Q_+. 
$$
Let $\mu\in(0,1)$ be a given constant. Define a set
$$
\mathcal E_{\mu}:=\Big\{(z,r)\in\mathbb R^{1+2d}\times(0,1): Q_{z,r} \subset Q_+,\ |Q_{z,r}\setminus E| \leqslant \mu |Q_{z,r}| \Big\}.
$$
Suppose that there are $r_0\in(0,1)$ and $m\in\mathbb N$ such that for any $(z,r)\in \mathcal E_{\mu}$,
\begin{align}\label{Af9}
{\bar{Q}_{z,r}}^m \subset F\mbox{ and }r<r_0.
\end{align}
Then there {exist} two constants $c_0\in(0,1)$ and $C_0\geqslant 1${,} depending only on $d${,} such that 
\begin{equation}
|E| \leqslant \frac{m+1}{m}(1-c_0\mu)\left(|F \cap Q_+| + C_0mr_0^{2}\right).
\end{equation}
