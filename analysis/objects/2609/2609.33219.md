<!-- head-1: body_md -->
The infinity issue of \cref{minimax} originates from \cref{prob:nuclear-norm-SDP-formulation}, where every point on the unit sphere corresponds to a constraint.
For the sake of tractable computation, we need to find a way to render the ``infinity" of the problem to ``finity".
The question is whether finitely many sphere-indexed constraints
can preserve the optimal value for a given tensor, or approximate
it with a controlled error. We first show that exact preservation
of the optimal value is possible.
In particular, for every $\boldsymbol{\mathcal{T}}\in\mathbb{R}^{n_1\times n_2\times n_3}$, there exists a finite support set $\Theta\subsetneq \mathbb{B}^{n_1}$ with $|\Theta|\leq n_1n_2n_3$ such that the following SDP with finite number of constraints has the same optimal value as \cref{prob:nuclear-norm-SDP-formulation}:
 \begin{align*}
        \|\boldsymbol{\mathcal{T}}\|_* = \max_{\boldsymbol{\mathcal{Z}}}\left\{
        \langle \boldsymbol{\mathcal{T}},\boldsymbol{\mathcal{Z}}\rangle
        :\begin{bmatrix}
        \mathbf{I}_{n_2} & \sum_{i=1}^{n_1}x_i\mathbf{Z}_i \\
        \sum_{i=1}^{n_1}x_i\mathbf{Z}_i^\top &\mathbf{I}_{n_3}
    \end{bmatrix}\succeq 0\;\forall\,\mathbf{x}\in\Theta
        \right\}.
    \end{align*}
This result {\it theoretically} gives a finite SDP representation to tensor nuclear norm. It is worth mentioning that such result is 
derived from another SDP formulation of the tensor nuclear norm and we call it ``P-SDP" (see \cref{sec:nuc-norm}), as it is from the definition of the nuclear norm in \cref{tensor-nuclear-norm}, i.e., the primal side. We show that the strong duality holds for \cref{prob:nuclear-norm-SDP-formulation,vD-eq2}.

The crux of the above problem thus reduces to identifying the set \(\Theta\), which nevertheless remains a difficult task.
If this were not the case, the original problem would be solvable in polynomial time.
In this paper, we bypass the difficulty of finding the exact $\Theta$, 
instead, we try to find good approximations. Two approaches are developed. 
The first (\cref{sec:compute}) adopts the sphere
discretization idea in \cite{he2023approximation} and uses the so-called $\tau$-hitting set to replace the set $\Theta$. We show that the resulting approximation problem has worst-case theoretical guarantee of the original problem. The second approach (\cref{sec:dual}) selects sphere points adaptively, a similar idea to column generation for linear programming adopted in \cite{he2026nuclear} to compute the tensor nuclear norm. We prove that the adaptive procedure converges asymptotically to the optimal value of \cref{prob:tensor-comp-linear-constr}.

<!-- definition-1: body_md -->
The nuclear norm of a tensor $\mathcal{T} \in \mathbb{R}^{n_1 \times n_2 \times \cdots \times n_d}$ is 
\begin{align}\label{tensor-nuclear-norm}
    \|\boldsymbol{\mathcal{T}}\|_*:=\min_{\lambda_r,\mathbf{x}_r, R} \left\{\sum_{r=1}^R|\lambda_r|: \boldsymbol{\mathcal{T}} = \sum_{r=1}^R \lambda_r \mathbf{x}_1^{(r)} \otimes \mathbf{x}_2^{(r)} \otimes\cdots\otimes \mathbf{x}_d^{(r)}, R \in \mathbb{N},~\|\mathbf{x}_{\ell}^{(r)}\|=1 \forall\,\ell\in[d], r\in[R] \right\},
\end{align}
where  
$\otimes$ denotes the vector outer product.
\hfill $\triangleleft$
<!-- end definition-1 -->
<!-- definition-2: body_md -->
If a tensor $\mathcal{T} \in \mathbb{R}^{n_1 \times n_2 \times \cdots \times n_d}$, written as a rank-one decomposition
    \begin{align}
     &\boldsymbol{\mathcal{T}} \;=\; \sum_{r=1}^R \lambda_r \mathbf{x}_1^{(r)} \otimes \mathbf{x}_2^{(r)} \otimes \cdots \otimes\mathbf{x}_d^{(r)},~\label{def:nuclear-decomp}
\end{align}
satisfies that $\|\boldsymbol{\mathcal{T}}\|_* = \sum_{r=1}^R |\lambda_r|$, then \cref{def:nuclear-decomp} is called a nuclear decomposition of $\boldsymbol{\mathcal{T}}$.
\hfill $\triangleleft$
<!-- end definition-2 -->
<!-- definition-3: body_md -->
The nuclear rank of a tensor $\boldsymbol{\mathcal{T}} \in \mathbb{R}^{n_1 \times n_2 \times \cdots \times n_d}$ is defined by
        \begin{align}
        \label{def:nuclear-rank}
        \operatorname{rank}_*(\boldsymbol{\mathcal{T}}) := \min\left\{R\in\mathbb{N}: \boldsymbol{\mathcal{T}} = \sum_{r=1}^R \lambda_r \mathbf{x}_1^{(r)} \otimes\cdots\otimes \mathbf{x}_d^{(r)}, \|\boldsymbol{\mathcal{T}}\|_*= \sum_{r=1}^R|\lambda_r|, ~\|\mathbf{x}_{\ell}^{(r)}\|=1 \forall\,\ell\in[d], r\in[R]\right\}.
    \end{align}
    \hfill $\triangleleft$
<!-- end definition-3 -->
<!-- lemma-1: body_md -->
\label{lemma:nuclear-rank-ub}
    $\operatorname{rank}_*(\boldsymbol{\mathcal{T}}) \leq n_1n_2n_3$ for any $\boldsymbol{\mathcal{T}} \in \mathbb{R}^{n_1\times n_2\times n_3}$.

\label{thm:nuc-norm-finite-rep}

        \|\boldsymbol{\mathcal{T}}\|_* = \min_{\bm\Lambda} \left\{\frac{1}{2}\sum_{\mathbf{x}\in\Theta} \mathrm{tr}(\bm\Lambda(\mathbf{x})): \boldsymbol{\mathcal{T}}=\sum_{\mathbf{x}\in\Theta}\mathbf{x}\otimes\bm\Gamma(\mathbf{x}),\bm\Lambda(\mathbf{x}) = \begin{bmatrix}
        \bm\Lambda^{(1)}(\mathbf{x}) &\bm\Gamma(\mathbf{x})\\
        \bm\Gamma^\top(\mathbf{x}) &\bm\Lambda^{(2)}(\mathbf{x})
    \end{bmatrix}\succeq \bm 0~~\text{for all $\mathbf{x}\in \Theta$}
    \right\}\label{vD-eq1}
    \end{align}

<!-- lemma-2: body_md -->
\label{lemma:matrix-nuclear-norm} Given any $\bm\Gamma\in\mathbb{R}^{n_1\times n_2}$, let $\bm\Gamma=\mathbf{P}^\top \mathbf{D} \mathbf{Q}$ be its SVD and $\omega_1,\dots,\omega_{n_1}$ be its singular values, i.e., $\mathbf{D}=[\mathrm{diag}(\omega_1,\dots,\omega_{n_1},0,\dots,0)]$. Then the optimization problem parameterized by $\bm\Gamma$
\begin{equation}\label{eq:inf-trace-sum}
    \min_{\mathbf{U},\mathbf{V} }\left\{\frac{1}{2}\left(\mathrm{tr}(\mathbf{U})+\mathrm{tr}(\mathbf{V} )\right):\begin{bmatrix}
        \mathbf{U} &\bm\Gamma\\
        \bm\Gamma^\top &\mathbf{V} 
    \end{bmatrix}\succeq \bm 0,~\mathbf{U}\in\mathbb{R}^{n_1\times n_1}, \mathbf{V} \in\mathbb{R}^{n_2\times n_2}\right\}
\end{equation}
is solved by
\begin{equation}\label{UV-def}
      \mathbf{U}^\star(\bm\Gamma) = \mathbf{P}^\top\mathrm{diag}(\omega_1,\dots,\omega_{n_1})\mathbf{P}, ~\mathbf{V} ^\star(\bm\Gamma) = \mathbf{Q}^\top\mathrm{diag}(\omega_1,\dots,\omega_{n_1},0,\dots,0)\mathbf{Q}
\end{equation}
and $\text{val}\cref{eq:inf-trace-sum}= \|\bm\Gamma\|_*$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
The proof idea is as follows.
    For any given $\Theta=\{\tilde{\mathbf{x}}^{(1)},\dots,\tilde{\mathbf{x}}^{(R)}\}$, we use the following problem as an intermediary:
    \begin{align}
        \min_{W^{(r)}}\left\{\sum_{r=1}^R\|\bm W^{(r)}\|_*: \boldsymbol{\mathcal{T}} = \sum_{r=1}^R \tilde{\mathbf{x}}^{(r)} \otimes {\bm W}^{(r)},~\bm W^{(r)}\in \mathbb{R}^{n_2\times n_3}\right\}.\label{eq:nuclear-norm-no-rank}
    \end{align}
    We first prove 
    that  $\text{val}\cref{vD-eq1}=\text{val}\cref{eq:nuclear-norm-no-rank}$ for any given $\Theta$. Then we prove that there exists $\Theta$ such that $\|\boldsymbol{\mathcal{T}}\|_*=\text{val}\cref{eq:nuclear-norm-no-rank}$ and the result follows.
    
    Now we prove $\text{val}\cref{vD-eq1}=\text{val}\cref{eq:nuclear-norm-no-rank}$ for any given $\Theta$.
    To see this, we note that by \cref{lemma:matrix-nuclear-norm}, for any matrix $\bm W\in \mathbb{R}^{p\times q}$, it holds that 
    \[
        \|\bm W\|_* = \min_{\mathbf{U},\mathbf{V} }\left\{\frac{1}{2}\left(\mathrm{tr}(\mathbf{U})+\mathrm{tr}(\mathbf{V} )\right):\begin{bmatrix}
        \mathbf{U} &\bm W\\
        \bm W^\top &\mathbf{V} 
    \end{bmatrix}\succeq \bm 0,~\mathbf{U}\in\mathbb{R}^{p\times p}, \mathbf{V} \in\mathbb{R}^{q\times q}\right\}.
    \]
    Then \cref{eq:nuclear-norm-no-rank} is equivalent to
    \begin{align}
        \min_{\mathbf{U},\mathbf{V},\bm W^{(r)}}\left\{\frac{1}{2}\sum_{r=1}^R\left(\mathrm{tr}(\mathbf{U}^{(r)}) + \mathrm{tr}(\mathbf{V}^{(r)})\right): \boldsymbol{\mathcal{T}} = \sum_{r=1}^R \tilde{\mathbf{x}}^{(r)} \otimes {\bm W}^{(r)},\begin{bmatrix}
        \mathbf{U}^{(r)} &\bm W^{(r)}\\
        {\bm W^{(r)}}^\top &\mathbf{V} ^{(r)}
    \end{bmatrix}\succeq \bm 0\right\}.\label{eq:nuclear-norm-no-rank-sdp}
    \end{align}
    Note that \cref{eq:nuclear-norm-no-rank-sdp} is equivalent to \cref{vD-eq1}. Thus $\text{val}\cref{vD-eq1}=\text{val}\cref{eq:nuclear-norm-no-rank-sdp}=\text{val}\cref{eq:nuclear-norm-no-rank}$.
    
    Now we prove that there exists $\Theta$ such that $\|\boldsymbol{\mathcal{T}}\|_*=\text{val}\cref{eq:nuclear-norm-no-rank}$. By \cref{lemma:nuclear-rank-ub} and the definition of nuclear norm, it holds that 
    \begingroup\small
\begin{align}
        \|\boldsymbol{\mathcal{T}}\|_* \;=\; &\min_{\bm\lambda_r,\mathbf{x}_1^{(r)},\mathbf{x}_2^{(r)},\mathbf{x}_3^{(r)},R}\left\{\sum_{r=1}^R|\lambda_r|: \boldsymbol{\mathcal{T}} = \sum_{r=1}^R \lambda_r \mathbf{x}_1^{(r)} \otimes \mathbf{x}_2^{(r)} \otimes \mathbf{x}_3^{(r)},~ 
    \|\mathbf{x}_{d}^{(r)}\|=1,~d\in\{1,2,3\}, R \in \mathbb{N},R\leq n_1n_2n_3\right\}\nonumber\\
    \;=\; & \min_{\mathbf{x}_1^{(r)},\bm W^{(r)},R}\left\{\sum_{r=1}^R\|\bm W^{(r)}\|_*: \boldsymbol{\mathcal{T}} = \sum_{r=1}^R \mathbf{x}_1^{(r)} \otimes {\bm W}^{(r)},~ 
    \|\mathbf{x}_{1}^{(r)}\|=1, \operatorname{rank}(\bm W^{(r)})=1, R \in \mathbb{N},R\leq n_1n_2n_3\right\},\label{eq:nuclear-norm-rank-1}
    \end{align}
\endgroup
    where in the second equality, we use the fact that $\|\lambda_r \mathbf{x}_2^{(r)}\otimes \mathbf{x}_3^{(r)}\|_* = |\lambda_r|$. 
    Let $\{{\mathbf{x}_1^*}^{(r)},r=1,\dots,R\}$ be a set of optimal solution to \cref{eq:nuclear-norm-rank-1}. Let $\Theta=\{{\mathbf{x}_1^*}^{(1)},\dots,{\mathbf{x}_1^*}^{(R)}\}$.
    In the following, we show that under $\Theta$ constructed in this way, $\text{val}\cref{eq:nuclear-norm-rank-1}=\text{val}\cref{eq:nuclear-norm-no-rank}$.
    It is straightforward that $\text{val}\cref{eq:nuclear-norm-no-rank}\leq \text{val}\cref{eq:nuclear-norm-rank-1}$, since the feasible region of \cref{eq:nuclear-norm-rank-1} is a subset of that of \cref{eq:nuclear-norm-no-rank}. Now we show that $\text{val}\cref{eq:nuclear-norm-no-rank}\geq \text{val}\cref{eq:nuclear-norm-rank-1}$. First note that $\text{val}\cref{eq:nuclear-norm-rank-1}=\text{val}\cref{tensor-nuclear-norm}$ according to the definition of tensor nuclear norm. Next, we will show that for every feasible solution of \cref{eq:nuclear-norm-no-rank}, we can construct a feasible solution of \cref{tensor-nuclear-norm}, and they share the same objective value. 
    
    Let $\bm W^{(r)}, r=1,\dots,R$ be a feasible solution to \cref{eq:nuclear-norm-no-rank}. Then it holds that 
    $\boldsymbol{\mathcal{T}} = \sum_{r=1}^R {\mathbf{x}_1^*}^{(r)} \otimes {\bm W}^{(r)}$. 
    For each $r$, take a SVD decomposition of $\bm W^{(r)}$, i.e., 
    \[
        \bm W^{(r)} = \sum_{j=1}^{n_2} \sigma_{r,j} \mathbf{u}_{r,j} \otimes \mathbf{v}_{r,j},~\sigma_{r,j} \geq 0,~ \|\mathbf{u}_{r,j}\| = \|\mathbf{v}_{r,j}\|=1,
    \]
    where $\|\bm W^{(r)}\|_*=\sum_{j=1}^{n_2} \sigma_{r,j}$. Then it holds that 
    \[
        \boldsymbol{\mathcal{T}} = \sum_{r=1}^R \sum_{j=1}^{n_2} \sigma_{r,j} {\mathbf{x}_1^*}^{(r)} \otimes \mathbf{u}_{r,j} \otimes \mathbf{v}_{r,j}.
    \]
    Note that $\{{\mathbf{x}_1^*}^{(r)}, \mathbf{u}_{r,j}, \mathbf{v}_{r,j}\}$ form a feasible solution to the minimization problem in \cref{tensor-nuclear-norm} and the objective value of \cref{tensor-nuclear-norm} equals $\sum_{r=1}^R \sum_{j=1}^{n_2} \sigma_{r,j}$ under this solution. 
    Because $\sum_{r=1}^R \|\bm W^{(r)}\|_* = \sum_{r=1}^R \sum_{j=1}^{n_2} \sigma_{r,j}$, we prove that for every feasible solution of \cref{eq:nuclear-norm-no-rank}, we can construct a feasible solution of \cref{tensor-nuclear-norm}, and they share the same objective value.

<!-- remark-2: body_md -->
\label{rem:approx-nuclear-decomposition}
Recall that a nuclear decomposition of $\boldsymbol{\mathcal{T}}$, as in
\cref{def:nuclear-decomp}, is an exact rank-one decomposition
with unit factors whose sum of absolute coefficients equals
$\|\boldsymbol{\mathcal{T}}\|_*$.
For $\tau\in(0,1]$, we call an exact rank-one decomposition
with unit factors a \emph{$\tau$-approximate nuclear decomposition}
if its coefficients satisfy
$  \|\boldsymbol{\mathcal{T}}\|_*
    \leq \sum_{r=1}^{R}|\lambda_r|
    \leq \frac{1}{\tau}\|\boldsymbol{\mathcal{T}}\|_*$.

The optimal SDP solution in \cref{thm:approximation-ratio}
provides such a decomposition of $\widehat{\boldsymbol{\mathcal{T}}}$.
Specifically, let $\widehat{\bm\Gamma}_i$ be the off-diagonal
block of $\widehat{\bm\Lambda}_i$, and let
$    \widehat{\bm\Gamma}_i
    = \sum_{r=1}^{R_i}\sigma_{ir}\mathbf{y}_{ir}{\mathbf{z}}_{ir}^{\top},
    ~\text{where}~
    R_i:=\operatorname{rank}(\widehat{\bm\Gamma}_i)$
be the singular value decomposition of $\widehat{\bm\Gamma}_i$,
where $\sigma_{ir}>0$, $\mathbf{y}_{ir}\in\mathbb{R}^{n_2}$,
${\mathbf{z}}_{ir}\in\mathbb{R}^{n_3}$, and $\|\mathbf{y}_{ir}\|=\|{\mathbf{z}}_{ir}\|=1$.
Then the second constraint in
\cref{prob:primal-linear-constr-discret} can be written as
\begin{equation}\label{eq:approx-nuclear-decomposition}
    \widehat{\boldsymbol{\mathcal{T}}}
    = \sum_{i=1}^{m}\mathbf{x}_i\otimes\widehat{\bm\Gamma}_i
    = \sum_{i=1}^{m}\sum_{r=1}^{R_i}
      \sigma_{ir}\mathbf{x}_i\otimes\mathbf{y}_{ir}\otimes{\mathbf{z}}_{ir}.
\end{equation}

By \cref{lemma:matrix-nuclear-norm} and optimality of the
diagonal blocks, the sum of its coefficients satisfies
\[
    \sum_{i=1}^{m}\sum_{r=1}^{R_i}\sigma_{ir}
    = \sum_{i=1}^{m}\|\widehat{\bm\Gamma}_i\|_*
    = \frac{1}{2}\sum_{i=1}^{m}\mathrm{tr}(\widehat{\bm\Lambda}_i)
    = \widehat{V}.
\]
Consequently, \cref{ineq:app-1} yields
$  \|\widehat{\boldsymbol{\mathcal{T}}}\|_*
    \leq \sum_{i=1}^{m}\sum_{r=1}^{R_i}\sigma_{ir}
    \leq \frac{1}{\tau}\|\widehat{\boldsymbol{\mathcal{T}}}\|_*$.
Hence, \cref{eq:approx-nuclear-decomposition} is a
$\tau$-approximate nuclear decomposition of $\widehat{\boldsymbol{\mathcal{T}}}$.
It is a nuclear decomposition whenever
$\widehat{V}=\|\widehat{\boldsymbol{\mathcal{T}}}\|_*$.

<!-- proof-2: body_md -->
Consider the set 
    $A := \{\mathbf{x}_1\otimes \mathbf{x}_2 \otimes \mathbf{x}_3: \|\mathbf{x}_1\|=\|\mathbf{x}_2\|=\|\mathbf{x}_3\|=1, \mathbf{x}_1\in \mathbb{R}^{n_1},\mathbf{x}_2\in \mathbb{R}^{n_2},\mathbf{x}_3\in \mathbb{R}^{n_3} \}$
    
    and define $K:= \operatorname{conv}(A)$. Then we have 
    \begin{align*}
        \|\boldsymbol{\mathcal{T}}\|_* \;=\; &\min\left\{\sum_{r=1}^R c_r: \boldsymbol{\mathcal{T}} = \sum_{r=1}^R c_r\mathbf{\mathcal{A}}_r, c_r \geq 0, \mathbf{\mathcal{A}}_r \in A\right\}\\
        \;=\; &\min\left\{\Bar{c}: \boldsymbol{\mathcal{T}} = \Bar{c} \sum_{r=1}^R \frac{c_r}{\Bar{c}} \mathbf{\mathcal{A}}_r, c_r\geq 0, \mathbf{\mathcal{A}}_r\in A, \Bar{c}=\sum_{r=1}^R c_r\right\}\\
        \;=\; &\min\left\{\Bar{c}\geq 0: \boldsymbol{\mathcal{T}} = \Bar{c} \boldsymbol{\mathcal{K}}, \boldsymbol{\mathcal{K}}\in K\right\}.
    \end{align*}
    where the first line is from the definition of tensor nuclear norm, and the third line is from the definition of $K$.
    \citet[Proposition 3.1]{ref:friedland2018nuclear} show that the minimizer exist for the above problem.
    Thus, for any tensor $\boldsymbol{\mathcal{T}}\in \mathbb{R}^{n_1\times n_2\times n_3}$, 
    there exists $(t, \boldsymbol{\mathcal{K}})$ such that $t=\|\boldsymbol{\mathcal{T}}\|_*$ and $\boldsymbol{\mathcal{T}}=t\boldsymbol{\mathcal{K}}$, i.e., 
    $\frac{\boldsymbol{\mathcal{T}}}{\|\boldsymbol{\mathcal{T}}\|_*} \in K$.
    Then by an extension of Carath\'{e}odory's theorem to connected sets \citep[Theorem 18 (ii)]{ref:Eggleston_1958}, there exist $R\leq n_1n_2n_3, c_r\geq 0, \sum_{r=1}^R c_r = 1$ and $\mathbf{\mathcal{A}}_r\in A$, such that 
    $\frac{\boldsymbol{\mathcal{T}}}{\|\boldsymbol{\mathcal{T}}\|_*} = \sum_{r=1}^R c_r \mathbf{\mathcal{A}}_r$.
    
    That is, 
    \begin{align}
        \label{eq:nuclear-rank-decomp}

\emph{(i)} We derive the dual of~\cref{prob:tensor-comp-dual-app} by a standard Lagrangian argument.
Recall that $\Upsilon=\{\mathbf{x}_1,\dots,\mathbf{x}_m\}$, and we introduce multipliers $\bm\Lambda_i=\begin{bmatrix}\bm\Lambda_i^{(1)}&\bm\Gamma_i\\ \bm\Gamma_i^\top&\bm\Lambda_i^{(2)}\end{bmatrix}\succeq\bm 0$ for the $m$ semi-definite constraints.
Dualizing these constraints, problem~\cref{prob:tensor-comp-dual-app} can be written as
\begin{align*}
    \max_{\bm\lambda,\boldsymbol{\mathcal{Z}}}\min_{\bm\Lambda_i\succeq\bm 0}~\bm\lambda^\top\mathbf{b}+\sum_{i=1}^m\left\langle\bm\Lambda_i,\begin{bmatrix}
        \mathbf{I}_{n_2} &\boldsymbol{\mathcal{Z}}(\mathbf{x}_i)\\
        \boldsymbol{\mathcal{Z}}(\mathbf{x}_i)^\top &\mathbf{I}_{n_3}
    \end{bmatrix}\right\rangle
    \quad\text{s.t.}~~\boldsymbol{\mathcal{Z}}=\sum_{\ell=1}^d\bm\lambda_\ell\boldsymbol{\mathcal{A}}_\ell.
\end{align*}
Exchanging the order of the max and min, the dual problem is
\begin{align}
    \min_{\bm\Lambda_i\succeq\bm 0}\left\{\max_{\bm\lambda,\boldsymbol{\mathcal{Z}}}~\bm\lambda^\top\mathbf{b}+\sum_{i=1}^m\left\langle\bm\Lambda_i,\begin{bmatrix}
        \mathbf{I}_{n_2} &\boldsymbol{\mathcal{Z}}(\mathbf{x}_i)\\
        \boldsymbol{\mathcal{Z}}(\mathbf{x}_i)^\top &\mathbf{I}_{n_3}
    \end{bmatrix}\right\rangle
    \quad\text{s.t.}~~\boldsymbol{\mathcal{Z}}=\sum_{\ell=1}^d\bm\lambda_\ell\boldsymbol{\mathcal{A}}_\ell\right\}.\label{prob:lg-dual}
\end{align}
Now we prove that \cref{prob:lg-dual} is equivalent to \cref{prob:primal-linear-constr-discret}. We do this by first analyzing the inner maximization problem.
To facilitate the analysis of the inner maximization problem, let $\boldsymbol{\mathcal{T}}'$ be any tensor satisfying $\mathscr{A}(\boldsymbol{\mathcal{T}}') = \mathbf{b}$ but fixed. Then we have
\begin{align} 
&\max_{\bm\lambda,\boldsymbol{\mathcal{Z}}}~\left\{\bm\lambda^\top\mathbf{b}+\sum_{i=1}^m\left\langle\bm\Lambda_i,\begin{bmatrix}
        \mathbf{I}_{n_2} &\boldsymbol{\mathcal{Z}}(\mathbf{x}_i)\\
        \boldsymbol{\mathcal{Z}}(\mathbf{x}_i)^\top &\mathbf{I}_{n_3}
    \end{bmatrix}\right\rangle
    \quad\text{s.t.}~~\boldsymbol{\mathcal{Z}}=\sum_{\ell=1}^d\bm\lambda_\ell\boldsymbol{\mathcal{A}}_\ell\right\}\nonumber\\
    \overset{\cref{eq:ZT}}{=}~&\max_{\bm\lambda,\boldsymbol{\mathcal{Z}}}~\left\{\langle \boldsymbol{\mathcal{Z}}, \boldsymbol{\mathcal{T}}'\rangle + \sum_{i=1}^m \mathrm{tr}(\bm\Lambda_i) + \sum_{i=1}^m 2\langle \mathbf{x}_i\otimes \bm\Gamma_i,\boldsymbol{\mathcal{Z}}\rangle\quad\text{s.t.}~~\boldsymbol{\mathcal{Z}}=\sum_{\ell=1}^d\bm\lambda_\ell\boldsymbol{\mathcal{A}}_\ell\right\}\nonumber\\
    =~&\max_{\bm\lambda,\boldsymbol{\mathcal{Z}}}~ \left\{\sum_{i=1}^m \mathrm{tr}(\bm\Lambda_i) + \left\langle \boldsymbol{\mathcal{Z}}, \boldsymbol{\mathcal{T}}' + 2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i\right\rangle\quad\text{s.t.}~~ \boldsymbol{\mathcal{Z}}=\sum_{\ell=1}^d\bm\lambda_\ell\boldsymbol{\mathcal{A}}_\ell\right\}\nonumber\\
    =~&\max_{\bm\lambda}~ \left\{\sum_{i=1}^m \mathrm{tr}(\bm\Lambda_i) + \left\langle \sum_{\ell=1}^d\bm\lambda_\ell\boldsymbol{\mathcal{A}}_\ell, \boldsymbol{\mathcal{T}}' + 2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i\right\rangle\right\}\nonumber\\
    =~&\max_{\bm\lambda}~ \left\{\sum_{i=1}^m \mathrm{tr}(\bm\Lambda_i) +\sum_{\ell=1}^d\bm\lambda_\ell \left\langle \boldsymbol{\mathcal{A}}_\ell, \boldsymbol{\mathcal{T}}' + 2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i\right\rangle\right\}\label{prob:inner-2}
\end{align}
The first equality is by the choice of $\boldsymbol{\mathcal{T}}'$ and \cref{lemma:ZT}. 
Since $\bm\lambda\in \mathbb{R}^d$ is unrestricted, the optimal value of \cref{prob:inner-2} is finite if and only if 
\[
    \left\langle \mathcal{A}_\ell, \boldsymbol{\mathcal{T}}' +2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i\right\rangle = 0, \quad \text{for all}~\ell=1,\dots,d.
\]
By the definition of operator $\mathscr{A}(\cdot)$, we can rewrite the relation as
\[
    \mathscr{A}\left(\boldsymbol{\mathcal{T}}' +2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i\right) = \bm 0.
\]
Because $\mathscr{A}(\boldsymbol{\mathcal{T}}') = \mathbf{b}$, for the above to hold, we must have
\[
    \mathscr{A}\left(2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i\right) =-  \mathscr{A}(\boldsymbol{\mathcal{T}}')=-\mathbf{b}.
\]
Hence, we conclude that 
\begin{align*}
    \text{val}\cref{prob:inner-2} = \left\{\begin{matrix}
        \sum_{i=1}^m \mathrm{tr}(\bm\Lambda_i), &\mathscr{A}\left(2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i\right) = -\mathbf{b}\\
        +\infty,&\text{o.w.}
        \end{matrix}\right.
\end{align*}
Now we look at the minimax problem in \cref{prob:lg-dual}. 
Letting $\boldsymbol{\mathcal{T}}=-2\sum_{i=1}^m \mathbf{x}_i \otimes \bm\Gamma_i$, we can rewrite \cref{prob:lg-dual} as
\begin{align*}
    \min_{\bm\Lambda_i,\boldsymbol{\mathcal{T}}}~\sum_{i=1}^m\mathrm{tr}(\bm\Lambda_i)\quad\text{s.t.}~~\bm\Lambda_i\succeq\bm 0,~~\boldsymbol{\mathcal{T}}+2\sum_{i=1}^m\mathbf{x}_i\otimes\bm\Gamma_i=\bm 0,~~\mathscr{A}(\boldsymbol{\mathcal{T}})=\mathbf{b},
\end{align*}
which is exactly~\cref{prob:primal-linear-constr-discret} by rescaling the decision variables $\bm\Lambda_1,\dots,\bm\Lambda_m$.
Because $\bm\lambda=\bm 0, \boldsymbol{\mathcal{Z}}=\bm 0$ is strictly feasible for~\cref{prob:tensor-comp-dual-app}, Slater's condition holds and strong duality follows.

We assume that $\Upsilon$ spans $\mathbb{R}^{n_1}$ without loss of generality: if $\Upsilon$ does not span $\mathbb{R}^{n_1}$, we can add elements to obtain a spanning set $\mathcal{H}''$ with $\widehat{\Phi}(\mathcal{H}'')\subseteq\widehat{\Phi}(\Upsilon)$, so proving strictness for $\mathcal{H}''$ implies it for $\Upsilon$.

\begin{align}

    \|\boldsymbol{\mathcal{Z}}(\mathbf{u})\|_\sigma> 1,\quad\text{where }\mathbf{u}\in\mathbb{B}^{n_1}.\label{eq:ZnotinS}
\end{align}
By the Schur complement,~\cref{eq:ZinhatS} is equivalent to $\boldsymbol{\mathcal{Z}}\in\widehat{\Phi}(\Upsilon)$ and~\cref{eq:ZnotinS} is equivalent to $\boldsymbol{\mathcal{Z}}\notin\Phi$.
Since $\Upsilon$ contains only finitely many elements, we can find $\mathbf{u}\in\mathbb{B}^{n_1}$ such that $\mathbf{u}\notin\Upsilon$ and $-\mathbf{u}\notin\Upsilon$.
Let $c:=\max_{\mathbf{x}\in\Upsilon}|\langle\mathbf{u},\mathbf{x}\rangle|$; then $0<c<1$ because $\Upsilon$ spans $\mathbb{R}^{n_1}$ while $\mathbf{u}$ is parallel to no element of $\Upsilon$.
Let $\mathbf{M}\in\mathbb{R}^{n_2\times n_3}$ satisfy $\|\mathbf{M}\|_\sigma=1/c$.
Then for any $\mathbf{x}\in\Upsilon$,
\begin{align*}
    \|\boldsymbol{\mathcal{Z}}(\mathbf{x})\|_\sigma=\|(\mathbf{u}\otimes\mathbf{M})(\mathbf{x})\|_\sigma=\|\langle\mathbf{u},\mathbf{x}\rangle\mathbf{M}\|_\sigma=|\langle\mathbf{u},\mathbf{x}\rangle|\,\|\mathbf{M}\|_\sigma\leq c\cdot\frac{1}{c}=1,
\end{align*}
while
\begin{align*}
    \|\boldsymbol{\mathcal{Z}}(\mathbf{u})\|_\sigma=\|\langle\mathbf{u},\mathbf{u}\rangle\mathbf{M}\|_\sigma=\|\mathbf{M}\|_\sigma=\frac{1}{c}>1.
\end{align*}
This proves~\cref{eq:ZinhatS} and~\cref{eq:ZnotinS}, and hence $\Phi\subsetneq\widehat{\Phi}(\Upsilon)$.
