<!-- theorem-1: body_md -->
\label{thm:linear_variance_bound}
For each $d \geq 1$, there are constants $c_d,C_d \in (0,\infty)$ such that for every $t \geq 1, r \geq 0,$
\begin{equation}
\label{eq:bernstein_type_bound_on_height}
    \mathbb{P}\left(|f(t,0) - \mathbb{E} f(t,0)| \geq r \right) \leq C_d \exp\left\{ -c_d \frac{r^2}{t + r} \right\}.
\end{equation}
As a result,
\begin{equation}
\label{eq:bd_linear_variance}
    \operatorname{Var} f(t,0) \leq C_d t, \qquad t \geq 1.
\end{equation}

<!-- proposition-1: body_md -->
\label{prop:next_arrival_markov}
The next-arrival height process $\eta(t,x)$ for a ballistic deposition process started at any initial configuration evolves according to Equation \eqref{eq:next_arrival_height_autonomous_dynamics}.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Upon a clock ring $(t,x),$ the surface height $\xi(t,x)$ is set to $\eta(t^-,x)$, by definition. Heights at other sites remain the same. Thus,
\[
\eta(t,x) = \max\left\lbrace \eta(t^-,x)+1, \max_{y \sim x} \xi(t^-,y) \right\rbrace.
\]
But $\eta(t^-,x) \geq \max_{y \sim x} \xi(t^-,y)$, which shows $\eta(t,x) = \eta(t^-,x) + 1.$

Upon the same clock ring $(t,x)$, we now calculate the effect on surface height at an adjacent site $y \sim x$. By \eqref{eq:next_arrival_surface} again,
\begin{align*}
\eta(t,y) = \max\left\lbrace \xi(t^-,y)+1, \ \max_{\substack{z \sim y \\ z \neq x}} \xi(t^-,z), \ \eta(t^-,x)  \right\rbrace.
\end{align*}
Because $\eta(t^-,x) \geq \xi(t^-,x)$, we also have
\begin{align*}
\eta(t,y) & = \max\left\lbrace \xi(t^-,y)+1, \ \max_{z \sim y} \xi(t^-,z), \ \eta(t^-,x)  \right\rbrace \\
          & = \max\left\lbrace \eta(t^-,y), \eta(t^-,x) \right\rbrace.
\end{align*}
<!-- end proof-1 -->
<!-- lemma-1: body_md -->
\label{lemma:delayed_seed_identity}
Let $E$ be exponential with rate $1$, independently of $\eta^0$. Let $\widehat H^0_t$ be the delayed process
\[
\widehat H^0_t =
\begin{cases}
    0, & \quad t < E \\
    H^0_{t - E}, & \quad t \geq E,
\end{cases}
\]
Then $(H_t)_{t \geq 0} \stackrel{(d)}{=} (\widehat H^0_t)_{t \geq 0}$ as processes.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Let $E$ be the first clock ring at the origin for $\eta$. Before $E$, every ring is at a site of height $-\infty$ and leaves the profile unchanged. At time $E$, the update rule gives
\[
\eta(E,x) =
\begin{cases}
1, & x=0,\\
0, & x\sim 0,\\
-\infty, & \text{otherwise}.
\end{cases}
\]
This is exactly $\eta^0(0,\cdot)$, as follows from \eqref{eq:seed_initial_configuration} and \eqref{eq:next_arrival_surface}. By the strong Markov property of the independent Poisson clocks, the clocks after $E$, shifted by $E$, are independent of $E$ and have their original law. The autonomous dynamics therefore identify $(\eta_{E+s})_{s \geq 0}$ with an independent copy of $(\eta^0(s,\cdot))_{s\geq0}$ in distribution. Taking maxima proves the claim.
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lemma:path_bounds}
For every $t \geq 0$ and integers $k \geq 1$,
\begin{align}
    \max\left\lbrace \mathbb{P}(R_t \geq k), \mathbb{P}(H_t \geq k) \right\rbrace & \leq \exp(e(2d+1)t - k)
\end{align}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
By the pathwise characterization \eqref{eq:LPP_characterization}, $R_t \geq k$ implies the existence of a path ending at $(t,x)$ for some $\norm{x}_1 \geq k$, with any weight. Similarly, the event $H_t \geq k$ implies the existence of a path $\gamma$ ending at $(t,x)$ for some $x$, with weight $W(\gamma)$ at least $k$. Both of these events imply the existence of distinct updates $\left\lbrace (s_i,x_i) \right\rbrace_{i=1}^k$ where $x_1 = 0, x_{i+1} \in \left\lbrace x_i \right\rbrace\cup\mathcal{N}_{x_i}$ and $s_1 \leq \dots \leq s_k \leq t$. There are at most $(2d+1)^{k-1}$ sequences of sites $(x_1,\dots,x_k)$ and the probability that all these ring times fall in $[0,t]$ is $\mathbb{P}(N \geq k),$ with $N \sim \textsf{Poi}(t)$.
Therefore,
\begin{align*}
    \max\left\lbrace \mathbb{P}(R_t \geq k), \mathbb{P}(H_t \geq k) \right\rbrace & \leq (2d+1)^{k-1}\mathbb{P}(N \geq k) \\
    & \leq (2d+1)^{k-1}\mathbb{E}\left[\binom{N}{k}\right] \\
    & \leq \frac{((2d+1)t)^k}{k!} \\
    & \leq e^{-k}\frac{(e(2d+1)t)^k}{k!} \\
    & \leq \exp(e(2d+1)t - k).
\end{align*}
The third line follows from the Poisson factorial moment identity.
<!-- end proof-3 -->
<!-- proposition-2: body_md -->
\label{prop:exponential_concentration_hitting_times}
For all $r \geq 0$,
\begin{equation}
\label{eq:hitting_time_concentration_upper_tail}
\mathbb{P}(T_n \geq \mu_n + r) \leq \exp\left\lbrace -\frac{r^2}{2(\mu_n + r)} \right\rbrace.
\end{equation}
For $0 \leq r \leq \mu_n$, we also obtain the lower tail estimate
\begin{equation}
\label{eq:hitting_time_concentration_lower_tail}
\mathbb{P}(T_n \leq \mu_n - r) \leq \exp\left\lbrace -\frac{r^2}{4\mu_n} \right\rbrace. 
\end{equation}

<!-- proof-4: body_md -->
For any functional of surface heights $F:(\left\lbrace -\infty \right\rbrace \cup \mathbb{N}_{\geq 0})^\Lambda \rightarrow \mathbb{R}$, the generator $\mathcal{L}_t$ of the next-arrival process $\eta_{t}$ is given by
\begin{equation}
(\mathcal{L}_t F)(\eta) = \sum_{x \in \Lambda} \left\lbrace F(U_x \eta) - F(\eta) \right\rbrace
\end{equation}
It is a standard fact that if $X_t$ is a continuous-time Markov jump process with generator $\mathcal{L}_t$ and state space $S$, for any $F: \mathbb{R}_{t \geq 0} \times S$ which is $C^1$ in time and bounded on finite time intervals,
\begin{equation}
    F(t,X_t) - \int_0^t \left( \partial_s F + \mathcal{L}_s F\right)(s,X_s) ds, \qquad t \geq 0
\end{equation}
is a martingale. See \cite[\S5.1]{eberle2020markov} for a concrete reference. For any $\lambda \geq 0$, apply this to $\eta_{t}$ with 
\[
F(t,\eta) := \exp\left\lbrace \lambda(\mu_n(\eta) - \mu_n) + \left(\lambda - \frac{\lambda^2}{2}\right)t \right\rbrace.
\]
Then, $\partial_s F(s,\eta_s) = (\lambda - \lambda^2/2) F$ and 
\begin{align*}
(\mathcal{L}_s F_s)(\eta) & = \sum_{x \in \Lambda} \left\lbrace \exp(\lambda (\mu_n(U_x \eta) - \mu_n) + (\lambda - \lambda^2/2)s) - F(s,\eta) \right\rbrace \\
& = F(s,\eta)\sum_{x \in \Lambda} \left\lbrace \exp(-\lambda \delta_x(\eta)) - 1 \right\rbrace.
\end{align*}
For $s \leq t \wedge T_n$, by \eqref{eq:jumps_simplex} and the elementary inequality $e^{-x} - 1 + x \leq x^2/2, \ x \geq 0$, 
\begin{align*}
    (\partial_s F + \mathcal{L}_s F)(s,\eta_s) & = F(s,\eta_s)\left(\lambda - \lambda^2/2 + \sum_{x \in \Lambda} \left\lbrace \exp(-\lambda \delta_x(\eta_s)) - 1 \right\rbrace \right) \\
    & = F(s,\eta_s)\left( - \lambda^2/2 + \sum_{x \in \Lambda} \left\lbrace \exp(-\lambda \delta_x(\eta_s)) + \lambda \delta_x(\eta_s) - 1 \right\rbrace \right) \\
    & \leq F(s,\eta_s)\left(-\lambda^2/2 + \frac{\lambda^2}{2}\sum_{x \in \Lambda} \delta_x(\eta_s)^2  \right) \\
    & \leq 0.
\end{align*}
So $\int_0^{t \wedge T_n} \left( \partial_s F + \mathcal{L}_s F\right)(s,\eta_s) ds$ is a non-increasing process adapted to $\mathcal{F}_t$. Therefore, the stopped process $F(t \wedge T_n,\eta_{t \wedge T_n})$ is a supermartingale. Expressing this in terms of $M_t$ and rescaling shows that 
\begin{equation}
\exp\left\lbrace \lambda M_t - \frac{\lambda^2}{2}(t \wedge T_n) \right\rbrace
\end{equation}
is also a supermartingale starting at $1$. Therefore,
$\mathbb{E}[e^{\lambda M_t}] \leq e^{\frac{\lambda^2t}{2}}.$ Applying the standard Chernoff bound,
\begin{align*}
    \mathbb{P}(T_n \geq \mu_n + r) & \leq \mathbb{P}(M_{\mu_n + r} \geq r) \\
    & \leq e^{-\lambda r} \mathbb{E}[e^{\lambda M_{\mu_n + r}}] \\
    & \leq e^{-\lambda r} e^{\lambda^2 (\mu_n + r)/2}.
\end{align*}
Optimizing $\lambda$ gives \eqref{eq:hitting_time_concentration_upper_tail}. For the lower tail, apply the same argument with
\[
G(t,\eta) := \exp\left\lbrace -\theta(\mu_n(\eta) - \mu_n) + \left(1 - e^\theta\right)t \right\rbrace
\]
for any $\theta \geq 0.$ We compute
\begin{align*}
    (\partial_s G + \mathcal{L}_s G)(s,\eta_s) & = G(s,\eta_s)\left( 1 - e^\theta + \sum_{x \in \Lambda}\left\lbrace \exp(\theta \delta_x(\eta_s)) - 1 \right\rbrace \right) \\
    & = G(s,\eta_s)\left( 1 + \theta - e^\theta + \sum_{x \in \Lambda} \left\lbrace \exp(\theta \delta_x(\eta_s)) -\delta_x(\eta_s) \theta - 1 \right\rbrace \right).
\end{align*}
By the convexity of $f(x) = e^x - x - 1$ and the fact that $f(0) = 0$, we have $f(\alpha x) \leq \alpha f(x)$ for any $\alpha \in [0,1]$. Then for $s \leq t \wedge T_n$,
\begin{align*}
\sum_{x \in \Lambda} \left\lbrace \exp(\theta \delta_x(\eta_s)) -\delta_x(\eta_s) \theta - 1 \right\rbrace  & \leq (e^{\theta} - \theta - 1) \sum_{x \in \Lambda} \delta_x(\eta_s) \\
& = (e^{\theta} - \theta - 1).
\end{align*}
By the same argument as in the first part of the proof, $G(t \wedge T_n,\eta_{t \wedge T_n})$ is a supermartingale. In particular, so is
\[
\exp\left\lbrace -\theta M_t - (e^\theta - 1 - \theta)(t \wedge T_n) \right\rbrace.
\]
Since $t \wedge T_n \leq t, \mathbb{E}[e^{-\theta M_t}] \leq e^{t(e^\theta - 1 - \theta)} \leq e^{t \theta^2},$ for $\theta \in [0,1]$. Applying the Chernoff bound and setting $\theta = r/(2\mu_n)$ yields \eqref{eq:hitting_time_concentration_lower_tail}.

<!-- lemma-3: body_md -->
\label{lemma:mean_spacings}
There are constants $C_d \geq 1, c_d > 0$ independent of $\Lambda$ such that with
$L_n := \log(2C_d) + d\log(n+1)$, 
\begin{equation}
\label{eq:mu_spacing_lower_bound}
    \mu_{n+k} - \mu_n \geq c_d \left(k - L_n \right).
\end{equation}
for all integers $n \geq 0,k\geq 1$. Also, $0 \leq \mu_{n+1} - \mu_n \leq 1$.

<!-- proof-5: body_md -->
Throughout this proof, let $C,c$ be constants depending on $d$, but independent of $\Lambda$, that may change line by line. We will first establish the following bound on the size of the support at $T_n$:
\begin{equation}
\label{eq:expected_support_size}
    \mathbb{E}[|S_{T_n}|] \leq C(n+1)^d.
\end{equation}
Let $\kappa := \frac{1}{2(2d+1)e}$. Monotonicity of $R_t$ shows
\begin{align*}
    \mathbb{P}(R_{T_n} \geq r) & \leq \mathbb{P}(T_n >  \kappa r) + \mathbb{P}(R_{\kappa r} \geq r).
\end{align*}
To bound the first term on the right hand side, let $\Gamma_n$ denote the time of the $n$th clock ring at the origin. Note $\Gamma_n \sim \textsf{Gamma}(n,1)$ and $T_n \leq \Gamma_n$. Then $\mathbb{P}(T_n > \kappa r) \leq \mathbb{P}(\Gamma_n > \kappa r) \leq \mathbb{E}[e^{\Gamma_n/2}]e^{-\kappa r/2} = 2^ne^{-\kappa r / 2}$, where the last equality follows by the formula for the moment generating function of $\textsf{Gamma}(n,1)$. For the second term, apply Lemma \ref{lemma:path_bounds}. Then
\[
\mathbb{P}(R_{T_n} \geq r) \leq 2^n e^{-\kappa r / 2} + e^{-r/2}.
\]
For $r \geq C(n+1)$ for $C$ large enough, the right hand side is less than $2e^{-cr}$, for another constant $c.$ Then
\begin{align*}
    \mathbb{E}|S_{T_n}| & \leq \mathbb{E}(2R_{T_n} + 1)^d \\
    & \leq C \left[1 + \sum_{r = 1}^\infty r^{d-1}\mathbb{P}(R_{T_n} \geq r) \right] \\
    & \leq C(n+1)^d,
\end{align*}
by splitting the sum into the terms $r \geq C(n+1)$ and $r \leq C(n+1)$. \\

Next, we prove \eqref{eq:mu_spacing_lower_bound}. Consider a process $\tilde \eta_t$ as follows: $\tilde \eta_t$ is equal to $\eta_t$ for all $t < T_n$. At $T_n$, immediately set all heights to $n$ for all sites $x \in S_{T_n}$ in the support. Continue evolving $\tilde \eta_t$ according to the next-arrival rule. Let $\tilde T_{n+k}$ denote the first time $s \geq T_n$ such that $\sup_{x \in \Lambda } \tilde \eta(s,x) = n + k.$ Both processes $\tilde \eta, \eta$ evolve according to the same clocks. Because all sites in $\eta_{T_n}$ have height at most $n$, this coupling forces $\tilde{T}_{n+k} - T_n \leq T_{n+k} - T_n$, so 
\[
\mathbb{P}(T_{n+k} - T_n \leq s) \leq \mathbb{P}(\tilde T_{n+k} - T_n \leq s)
\]
for all $s \geq 0.$ 

By the strong Markov property, $\tilde{T}_{n+k} - T_n$ has the same law, conditional on $\mathcal{F}_{T_n}$, as the first time a next-arrival ballistic deposition process has maximum height $k$, started from the initial profile 
\[
\begin{cases}
0 & \ x \in S_{T_n} \\
-\infty & \ \text{otherwise}.
\end{cases}
\]
Note that $\left\lbrace \tilde T_{n+k} - T_n \leq s \right\rbrace$ implies the existence of a forwards path starting at some site $x$ in the support $S_{T_n}$, ending at time $s$, with at least $k$ many clock rings.  Following the proof of Lemma \ref{lemma:path_bounds} and taking a union bound,

\[
\mathbb{P}(\tilde T_{n+k} - T_n \leq s \mid \mathcal{F}_{T_n}) \leq |S_{T_n}| e^{(2d+1)es - k}.
\]

By \eqref{eq:expected_support_size},
\begin{align*}
\mathbb{P}(\tilde T_{n+k} - T_n \leq s) & \leq \mathbb{E}[|S_{T_n}|] e^{(2d+1)es - k} \\   
& \leq C(n+1)^d e^{(2d+1)es - k}.
\end{align*}

Take $k > L_n$. If not, then monotonicity $\mu_{n+k} \geq \mu_n$ gives \eqref{eq:mu_spacing_lower_bound}. With $L_n = \log(2C) + d\log(n+1)$, taking $s = (k - L_n)/(e(2d+1))$ gives $\mathbb{P}(\tilde T_{n+k} - T_n \leq s) \leq 1/2.$ Thus
\[
\mu_{n+k} - \mu_n \geq s\mathbb{P}(\tilde T_{n+k} - T_n \geq s) \geq \frac{k - L_n}{2e(2d+1)}.
\]
This establishes \eqref{eq:mu_spacing_lower_bound}. For the last claim, $\mu_{n+1} \geq \mu_n$ holds by monotonicity and $\mu_{n+1} \leq \mu_n + 1$ since it takes expected time $1$ for a clock to ring over a height-maximizing site of $\eta_{T_n}.$

<!-- proof-6: body_md -->
Throughout this proof, again let $C,c$ be constants that may change line by line, which are independent of $\Lambda$. It suffices to prove Equation \eqref{eq:bernstein_type_bound_on_height}, as integrating yields the linear variance bound \eqref{eq:bd_linear_variance}. We claim that for every $t \geq 1$ and $r \geq 0$,
\begin{equation}
\label{eq:concentration_on_H_t}
    \mathbb{P}\left(|H_t - \mathbb{E} H_t| \geq r\right) \leq C \exp\left\lbrace -c\frac{r^2}{t+r} \right\rbrace.
\end{equation}
The constants do not depend on the box $\Lambda$. For $t \geq 1$, let
\begin{equation}
m := \min\left\lbrace n \geq 1 : \mu_n \geq t \right\rbrace.
\end{equation}
We record the following facts. First, $t \leq \mu_m$ by definition and $\mu_m - \mu_{m-1} \leq 1 \Rightarrow \mu_m \leq t + 1$, so $t \leq \mu_m < t+1$. Second, $m \leq Ct$ because of the spacing lower bound in Lemma \ref{lemma:mean_spacings}. Third, $L_m \leq C\log(t+2)$ because of the second fact.

For any integer $r \geq C \log(t+2),$ Lemma \ref{lemma:mean_spacings} shows $\mu_{m+r} - t \geq \mu_{m+r} -\mu_m \geq c(r - L_m) \geq cr/2$. Applying Proposition \ref{prop:exponential_concentration_hitting_times},
\begin{align*}
\mathbb{P}(H_t \geq m + r) & = \mathbb{P}(T_{m+r} \leq t) \\
                          & \leq \mathbb{P}(T_{m+r} - \mu_{m+r} \leq -cr/2) \\
                          & \leq \exp\left\lbrace -c\frac{r^2}{\mu_{m+r}} \right\rbrace.
\end{align*}
With $\mu_{m+r} \leq m+r \leq Ct+r,$ we obtain
\[
\mathbb{P}(H_t \geq m + r) \leq \exp\left\lbrace -c\frac{r^2}{t + r} \right\rbrace.
\]
For the other direction, let $r \leq m$. Using Lemma \ref{lemma:mean_spacings}, $\mu_m - \mu_{m - r + 1} \geq c_d(r-1-L_{m-r+1})$. Since $L_{m - r + 1} \leq L_m$, 
\begin{align*}
t - \mu_{m - r + 1} & = t - \mu_m + \mu_m - \mu_{m - r + 1} \\
                    & \geq -1 + c(r - 1 - L_m) \\
                    & \geq cr/2,
\end{align*}
for $r \geq C\log(t+2)$. Applying \eqref{eq:hitting_time_concentration_upper_tail}, we obtain
\begin{align*}
\mathbb{P}(H_t \leq m - r) & = \mathbb{P}(T_{m-r+1} > t) \\
                          & \leq \mathbb{P}(T_{m-r+1} - \mu_{m-r+1} > cr/2) \\
                          & \leq \exp\left\lbrace -c\frac{r^2}{r + \mu_{m-r+1}} \right\rbrace \\
                          & \leq \exp\left\lbrace -c\frac{r^2}{r + \mu_{m}} \right\rbrace.
\end{align*}


Combining these two estimates, we obtain
\begin{equation}
    \mathbb{P}(|H_t - m| \geq r) \leq C\exp\left\lbrace -c\frac{r^2}{t+r} \right\rbrace,
\end{equation}
for all integer $r \geq C \log (t+2).$ The same bound for all $r \geq 0$ holds upon enlarging $C$. Integration by parts shows $\mathbb{E}|H_t - m| \leq C\sqrt{t}$. Combining this with the triangle inequality gives \eqref{eq:concentration_on_H_t}, upon potentially changing $C,c$. \\

Next, we pass to infinite volume. Let $\Lambda_L=[-L,L]^d\cap\mathbb{Z}^d$, and couple the processes $\eta_{t}^{\mathbb{Z}^d}$ and $\left\lbrace \eta^{\Lambda_L} \right\rbrace_{L \geq 1}$ using the same clocks, extending their profiles by $-\infty$ outside $\Lambda_L$. Clearly, the surface heights $\eta_t^{\Lambda_L}$, supports $S_t^{\Lambda_L}$, and maxima $H_t^{\Lambda_L}$ increase with $L$. By Lemma \ref{lemma:path_bounds}, 
\[
\mathbb{P}(R_t^{\mathbb{Z}^d} \geq r)\leq\exp\left\lbrace e(2d+1)t-r \right\rbrace.
\]
On $\left\lbrace R_t^{\mathbb{Z}^d} < L \right\rbrace$, all clock rings up to time $t$ which update the surface height occur in $\Lambda_L$. Therefore,
\begin{equation}
\label{eq:infinite_volume_coupling}
\mathbb{P}\left(\eta^{\Lambda_L}_s \neq \eta_s^{\mathbb{Z}^d} \text{ for some }s\leq t\right)
\leq\exp\left\lbrace e(2d+1)t-L \right\rbrace.
\end{equation}

In particular, $H_t^{\Lambda_L} \rightarrow H_t^{\mathbb{Z}^d}$ almost surely. Monotone convergence implies $\mathbb{E} H_t^{\Lambda_L} \rightarrow \mathbb{E} H_t^{\mathbb{Z}^d}$. Let $B := \left\lbrace \exists s \leq t : \eta^{\Lambda_L}_s \neq \eta_s^{\mathbb{Z}^d} \right\rbrace$. Take $L$ large enough so that $\mathbb{P}(B) \leq \epsilon, \mathbb{E} H_t^{\mathbb{Z}^d} - \mathbb{E} H_t^{\Lambda_L} \leq \epsilon$. Then 
\begin{align*}
    \mathbb{P}\left( \left| H_t^{\mathbb{Z}^d} - \mathbb{E} H_t^{\mathbb{Z}^d}\right| \geq r \right) & \leq \mathbb{P}\left( \left| H_t^{\mathbb{Z}^d} - \mathbb{E} H_t^{\Lambda_L} \right| \geq r - \epsilon \right) \\
    & \leq \epsilon + C\exp\left\lbrace -c\frac{(r-\epsilon)^2}{t + r - \epsilon} \right\rbrace.
\end{align*}
Taking $\epsilon \downarrow 0$ establishes \eqref{eq:concentration_on_H_t} when $\Lambda = \mathbb{Z}^d$. Finally, using the duality \eqref{eq:duality_reexpressed} gives the result.
