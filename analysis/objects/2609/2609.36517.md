\label{thm:mar29ago1553}
For $0<\theta<\pi$, define
\begin{align}
 g^*(\theta)
 :=\frac{1}{2\pi}\sum_{\varepsilon\in\{-1,1\}}
 \int_0^1\!\int_0^1
 &g^*_{1,\varepsilon}(\rho_{AB},\rho_{CD},\theta)\nonumber\\
 &\times
 \mathds{1}_{\left\{
 \sqrt{1-\rho_{AB}^2}\,|\sin\theta|
 \geq |\rho_{AB}\cos\theta+\varepsilon\rho_{CD}|
 \right\}}
 \,\textnormal{d}\rho_{AB}\,\textnormal{d}\rho_{CD},
 \label{eq:corrected-gstar}
\end{align}
where
\begin{align*}
 g^*_{1,\varepsilon}(\rho_{AB},\rho_{CD},\theta)
 &:=\left(\frac{4}{\pi}\right)^2
 \sqrt{1-\rho_{AB}^2}\sqrt{1-\rho_{CD}^2}\\
 &\quad\times
 \left[
 1-\frac{\rho_{AB}^2+\rho_{CD}^2
 +2\varepsilon\rho_{AB}\rho_{CD}\cos\theta}
 {\sin^2\theta}
 \right]^2
 \mathds{1}_{\{\rho_{AB},\rho_{CD}\in[0,1]\}}.
\end{align*}
Set $g^*(0)=g^*(\pi)=0$ and
\[
 c:=\int_0^\pi g^*(s)\,\textnormal{d} s.
\]
If $S_{AB},S_{CD}$ are the random segments which extremes are independent and identically distributed random variables with uniform distribution on the unit disc, then
\[
 c=\mathbb{P}\left(S_{AB}\cap S_{CD}\neq\varnothing\right)
 =\frac13\left(1-\frac{35}{12\pi^2}\right)
 =0.234826627014\ldots,
\]
and the conditional density of the small angle between $S_{AB},S_{CD}$, $\Theta$, given intersection is
\[
 g(\theta)=\frac{g^*(\theta)}{c}\,
 \mathds{1}_{\{\theta\in(0,\pi)\}}.
\]
Equivalently,
\[
\mathbb{P}\left(\Theta\leq\theta\mid S_{AB}\cap S_{CD}\neq\varnothing\right)
 =\int_0^\theta g(s)\,\textnormal{d} s,
 \qquad 0\leq\theta\leq\pi.
\]

\label{thm:main.thm}
If $A,B,C,D$ are independent random points uniformly distributed on $\Omega$, then
\begin{align*}
& \mathbb{P}\left(\Theta\leq \theta | S_{AB} \cap S_{CD} \neq \varnothing\right)  = \\
& \quad \frac{1}{\mathbb{P}\left(S_{AB} \cap S_{CD} \neq \varnothing\right)\times \textnormal{Area}(\Omega)^4 } \int_{0}^{2\pi}  \int_{0}^{2\pi} \mathds{1}_{\left\{\left|\cos(\gamma_{AB}-\gamma_{CD})\right| \leq \sin \theta\right\}} \\
& \quad\quad \times \int_{0}^{h_{\Omega}(\gamma_{AB})} \int_{0}^{h_{\Omega}(\gamma_{CD})} K_{\Omega}\left(r_{AB},\gamma_{AB}; \frac{r_{AB}\cos(\gamma_{AB}-\gamma_{CD}) - r_{CD}}{\sin(\gamma_{AB}-\gamma_{CD})}\right) \\
& \quad\quad\quad \times K_{\Omega}\left(r_{CD},\gamma_{CD}; \frac{r_{AB} - r_{CD}\cos(\gamma_{AB}-\gamma_{CD})}{\sin(\gamma_{AB}-\gamma_{CD})}\right) \textnormal{d} r_{CD} \textnormal{d} r_{AB} \textnormal{d}\gamma_{CD}\textnormal{d}\gamma_{AB}.
\end{align*}
