<!-- proof-1: body_md -->
In order to prove that two probability measures $\mu,\nu$ on a finite space $\Omega$ are far apart in total variation, it is classically enough to find a  test function $f\in\mathbb {R}^\Omega$ -- often referred to as a \emph{distinguishing statistic} -- whose expectations are very different under $\mu$ and $\nu$, compared to the corresponding standard deviations. More precisely, we  have
\begin{eqnarray}
\label{distinguishing}
\mathrm{d}_{\textsc{tv}}(\mu,\nu) & \ge & \left(1+2\frac{{\mathrm{Var}}_\mu [f]+{\mathrm{Var}}_\nu[f]}{\left|{\mathbb{E}}_\mu[f]-{\mathbb{E}}_\nu[f]\right|^2}\right)^{-1},
\end{eqnarray}
see, e.g., \cite{MR3726904}. Following a celebrated idea due to Wilson \cite{MR2023023}, we here use a leading eigenfunction of the generator as our distinguishing statistic.  Specifically, we let $\phi\in\mathbb {R}^N$ be an eigenvector of ${\mathrm{P}}$ corresponding to the second largest eigenvalue $\lambda_2$, normalized so that $\|\phi\|_\infty=1$, and we define a test function $f\in\mathbb {R}^{\Omega_k}$ as follows:
\begin{eqnarray}
\forall A\in\Omega_k,\qquad f(A) & := & \sum_{i\in A}\phi(i).
\end{eqnarray}
It then easily follows from the definition of $\mathcal {L}^\textsc{ex}_{{\mathrm{P}},k}$ that
$
\mathcal {L}^\textsc{ex}_{{\mathrm{P}},k} f =  -\gamma f, 
$
where $\gamma=1-\lambda_2$ is the spectral gap of ${\mathrm{P}}$. Consequently, letting $(\mathcal {S}_t)_{t\ge 0}$ denote a $k-$particle Exclusion Process with rates ${\mathrm{P}}$ starting from a non-random initial state $\mathcal {S}_0\in\Omega_k$, we have
\begin{eqnarray}
\label{expectation}
{\mathbb{E}}\left[f(\mathcal {S}_t)\right] & = & e^{-\gamma t}f(\mathcal {S}_0),
\end{eqnarray}
for all $t\ge 0$. On the other hand,  the events $\left(\{i\in \mathcal {S}_t\}\right)_{1\le i \le N}$ are  well known to be pairwise negatively correlated (see \cite{NegCorr} for a proof,  and \cite{Rayleigh} for a much stronger property), so that
\begin{eqnarray*}
\label{variance}
{\mathrm{Var}}\left[f(\mathcal {S}_t)\right] & \le & 2{\mathrm{Var}}\left[\sum_{i\in\mathcal {S}_t}\phi_+(i)\right]+2{\mathrm{Var}}\left[\sum_{i\in\mathcal {S}_t}\phi_-(i)\right]\\
& \le & 2\sum_{i=1}^N\phi^2_+(i){\mathbb{P}}(i\in \mathcal {S}_t){\mathbb{P}}(i\notin \mathcal {S}_t)+2\sum_{i=1}^N\phi^2_-(i){\mathbb{P}}(i\in \mathcal {S}_t){\mathbb{P}}(i\notin \mathcal {S}_t) \\
& \le & 2\|\phi\|_\infty^2 \sum_{i=1}^N{\mathbb{P}}(i\in \mathcal {S}_t) \ = \ 2k.
\end{eqnarray*}
In view of those estimates and their $t\to\infty$ counterparts, the bound (\ref{distinguishing}) yields
\begin{eqnarray}
\mathrm{d}(t) & \ge & \left(1+\frac{8ke^{2\gamma t}}{f^2(\mathcal {S}_0)}\right)^{-1}.\end{eqnarray}
Choosing $t=\mathrm{t}_{\textsc{mix}}(\varepsilon)$, and recalling that ${\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})=1/\gamma$, we arrive at 
\begin{eqnarray}
\label{LB1}
\mathrm{t}_{\textsc{mix}}(\varepsilon) & \ge & \frac{1}{2}{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}}) \log\left[\frac{(1-\varepsilon)f^2(\mathcal {S}_0)}{8\varepsilon k}\right].
\end{eqnarray}
It now remains to optimize on the choice of the initial set $\mathcal {S}_0\in\Omega_k$. Because of (\ref{particle-hole}) and upon replacing $\phi$ by $-\phi$ if necessary, we may assume that $\phi$ has at least $k$ non-negative entries, and the set $\mathcal {S}_0$ corresponding to the $k$ largest entries then clearly satisfies 
\begin{eqnarray}
\label{LB2}
f(\mathcal {S}_0) & \ge & \frac{k}{2N}\sum_{j=1}^N|\phi(j)|. 
\end{eqnarray}
On the other hand,  for any $i\in[N]$ and $t\ge 0$, we can use ${\mathrm{P}}_t \phi=e^{-\gamma t}\phi$ and (\ref{diagonal}) to write
\begin{eqnarray*}
e^{-\gamma t}|\phi(i)| & = & \left|\sum_{j=1}^N{\mathrm{P}}_t(i,j)\phi(j)\right|
\ \le \  \left(\max_{i\in[N]}{\mathrm{P}}_t(i,i)\right)\sum_{j=1}^N|\phi(j)|.
\end{eqnarray*}
Setting $t:=u{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})$ and choosing $i\in[N]$ such that $|\phi(i)|=\|\phi\|_\infty=1$, we deduce that
 \begin{eqnarray}
\frac{e^{-u}}{M_u({\mathrm{P}})}  & \le & \frac{1}{N}\sum_{j=1}^N|\phi(j)|.
\end{eqnarray}
Re-inserting this back into (\ref{LB2})  and then into (\ref{LB1}) concludes the proof.

\label{pr:MFJO}
\begin{eqnarray}
\label{dir}
\mathcal {E}_{{\mathrm{K}},k}^{\textsc{jo}} & \le & {\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\,\mathcal {E}_{{\mathrm{P}},k}^\textsc{jo}.
\end{eqnarray}

<!-- proof-2: body_md -->
An elementary but crucial property of the induced chain is that it satisfies ${\mathrm{t}_{\textsc{rel}}}({\mathrm{P}}^A)\le {\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})$, for any  non-empty $A\subseteq[N]$ (see, e.g., \cite[Theorem 13.16]{MR3726904}). In other words, 
\begin{eqnarray}
\label{trace}
\frac{1}{|A|}\sum_{i,j\in A}\left[g(i)-g(j)\right]^2 & \le & {\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\sum_{i,j\in A}{\mathrm{P}}^A(i,j) \left[g(i)-g(j)\right]^2,
\end{eqnarray}
for any $g\in\mathbb {R}^A$. In particular, for any $f\in L^2(\Omega_k)$, we can write
\begin{eqnarray*}
\mathcal {E}_{{\mathrm{P}},k}^\textsc{jo}(f,f) & = & \frac{1}{2}\sum_{A\in\Omega_k}\sum_{i\in A}\sum_{j\in A^c}{\mathrm{P}}^{(A^i)^c}(i,j)\left[f(A^{ij})-f(A)\right]^2\\
& = & \frac{1}{2}\sum_{B\in\Omega_{k-1}}\sum_{i,j\in B^c}{\mathrm{P}}^{B^c}(i,j)\left[f(B^j)-f(B^i)\right]^2\\
& \ge & \frac{1}{2{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})}\sum_{B\in\Omega_{k-1}}\frac{1}{|B^c|}\sum_{i,j\in B^c}\left[f(B^j)-f(B^i)\right]^2\\
& = & \frac{1}{{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})}\mathcal {E}_{{\mathrm{K}},k}^\textsc{jo}(f,f),
\end{eqnarray*}
 where the second line just uses the change of variable $B:=A^i$, while the third uses (\ref{trace})  with $A:=B^c$ and $g(i):= f(B^i)$.

<!-- corollary-3: body_md -->
\label{co:MFEX}
\begin{eqnarray}
{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\,\mathcal {E}^{\textsc{ex}}_{{\mathrm{P}},k} & \ge & \left(1-\delta({\mathrm{P}},k)\right)\,\mathcal {E}^{\textsc{ex}}_{{\mathrm{K}},k},
\end{eqnarray}
where the error term $\delta({\mathrm{P}},k)$ is defined as follows:
\begin{eqnarray}
\label{def:delta}
\delta({\mathrm{P}},k) & := & \frac{2k}{N}\left(N\max_{1\le i,j\le N}{\mathrm{P}}(i,j)\right)^2{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}}).
\end{eqnarray}
<!-- end corollary-3 -->
<!-- proof-3: body_md -->
We may assume that $\delta({\mathrm{P}},k)\le 1$, otherwise there is nothing to prove. For any set $A\in\Omega_k$ and any $(i,j)\in A\times A^c$, we have by definition
\begin{eqnarray*}
{\mathrm{P}}^{(A^i)^c}(i,j) - {\mathrm{P}}(i,j)  & = &  \sum_{m=1}^\infty\sum_{i_1,\ldots,i_m\in A\setminus i}{\mathrm{P}}(i,i_1){\mathrm{P}}(i_1,i_2)\cdots {\mathrm{P}}(i_{m-1},i_m){\mathrm{P}}(i_m,j),
\end{eqnarray*}
In particular, crudely bounding each entry of ${\mathrm{P}}$ in the sum by $p:=\max_{i,j\in[N]}{\mathrm{P}}(i,j)$ yields
\begin{eqnarray}
0 \ \le &  {\mathrm{P}}^{(A^i)^c}(i,j)-{\mathrm{P}}(i,j) & \le \ 2kp^2,
\end{eqnarray}
where we have used $2kp\le \delta({\mathrm{P}},k)\le 1$. Now, for any $f\in L^2(\Omega_k)$, we may multiply both sides by $\frac 12[f(A^{ij})-f(A)]^2$ and sum over all $(i,j)\in A\times A^c$ and  $A\in\Omega_k$ to conclude that
\begin{eqnarray}
0 \ \le  & \mathcal {E}^{\textsc{jo}}_{{\mathrm{P}},k}-\mathcal {E}^{\textsc{ex}}_{{\mathrm{P}},k} & \le \   \frac{\delta({\mathrm{P}},k)}{{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})}\mathcal {E}^\textsc{ex}_{{\mathrm{K}},k},
\end{eqnarray}
where we have used the fact that $2Nkp^2=\delta({\mathrm{P}},k)/{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})$. 
Of course, the first inequality also applies to ${\mathrm{P}}={\mathrm{K}}$, and the desired result now readily follows from Proposition \ref{pr:MFJO}.

<!-- proposition-3: body_md -->
\label{pr:octopus}For all $t\ge 0$, we have
\begin{eqnarray}
 \mathcal {E}_{{\mathrm{P}}_t}^\textsc{ip} &\le & t\mathcal {E}_{\mathrm{P}}^\textsc{ip}.
\end{eqnarray}
In particular, the same is true for the $k-$particle Exclusion Process.
<!-- end proposition-3 -->
<!-- proof-4: body_md -->
The Dirichlet form associated with $\mathcal {L}^{\textsc{ip}}_{\mathrm{P}}$ is $\mathcal {E}^\textsc{ip}_{\mathrm{P}} =  \frac{1}{2}\sum_{ i,j\in[N]}{\mathrm{P}}(i,j)\,\mathcal {E}_{i,j}^\textsc{ip}$, where
\begin{eqnarray*}
\mathcal {E}_{i,j}^\textsc{ip}(f,f) & := & \frac{1}{2}\sum_{\sigma\in\mathfrak {S}_N}[f(\tau_{ij}\sigma)-f(\sigma)]^2,
\end{eqnarray*}
represents the contribution from the edge $\{i,j\}$. Note that this quantity is symmetric in $i$ and $j$ and is zero when $i=j$. With this notation at hand, the Octopus Inequality \cite{MR2629990} reads
\begin{eqnarray}
\left(1-{\mathrm{P}}(o,o)\right)\sum_{i=1}^N{\mathrm{P}}(o,i)\mathcal {E}_{o,i}^\textsc{ip} & \ge & \frac{1}{2}\sum_{i,j\in[N]\setminus \{o\}}{\mathrm{P}}(o,i){\mathrm{P}}(o,j)\mathcal {E}_{i,j}^\textsc{ip},
\end{eqnarray}
for any $o\in[N]$. Moving the negative term to the right-hand side and re-arranging, we obtain
\begin{eqnarray}
\sum_{i=1}^N{\mathrm{P}}(o,i)\mathcal {E}_{o,i}^\textsc{ip} & \ge & \frac{1}{2}\sum_{i,j\in[N]}{\mathrm{P}}(o,i){\mathrm{P}}(o,j)\mathcal {E}_{i,j}^\textsc{ip},
\end{eqnarray}
and summing this  over  $o\in[N]$ yields a discrete-time version of our target inequality:
\begin{eqnarray}
\label{octopus:discrete}
2\mathcal {E}_{\mathrm{P}}^\textsc{ip} & \ge & \mathcal {E}^\textsc{ip}_{{\mathrm{P}}^2}.
\end{eqnarray}
 The same inequality was obtained in \cite{MR4164852,MR4254474}, but under additional assumptions on ${\mathrm{P}}$. Now, fix a time $t\ge 0$ and an integer $n\ge \log_2 t$, and consider the symmetric stochastic matrix 
\begin{eqnarray}
\mathrm{Q} & := & \left(1-\frac{t}{2^{n}}\right)\mathrm{Id}+\frac{t}{2^{n}}{\mathrm{P}}.
\end{eqnarray}
Applying (\ref{octopus:discrete}) to ${\mathrm{Q}}$ and iterating $n$ times, we arrive at
\begin{eqnarray}
2^n\mathcal {E}_{\mathrm{Q}}^\textsc{ip} & \ge & \mathcal {E}^\textsc{ip}_{{\mathrm{Q}}^{2^n}}.
\end{eqnarray}
But the left-hand side is exactly $t\mathcal {E}^\textsc{ip}_{{\mathrm{P}}}$, while the right-hand side tends to $\mathcal {E}^\textsc{ip}_{{\mathrm{P}}_t}$ as $n\to\infty$, establishing the first claim. Finally, recall that the $k-$particle Exclusion Process is the image of the Interchange Process under the projection map $\Phi_k\colon \mathfrak {S}_N\to\Omega_k$ defined by $\Phi_k(\sigma):= \sigma(\{1,2,\ldots,k\})$. More precisely, for any $f\in L^2(\Omega_k)$, we have
\begin{eqnarray}
(\mathcal {L}^\textsc{ex}_{{\mathrm{P}},k} f)\circ\Phi_k & = &  \mathcal {L}^\textsc{ip}_{\mathrm{P}} (f\circ \Phi_k).
\end{eqnarray}
Since each $A\in\Omega_k$ has exactly $k!(N-k)!$ pre-images under $\Phi_k$, this implies
\begin{eqnarray}
k!(N-k)!\,\mathcal {E}^\textsc{ex}_{{\mathrm{P}},k}(f,f)   & = & \mathcal {E}^\textsc{ip}_{\mathrm{P}}(f\circ \Phi_k,f\circ\Phi_k).
\end{eqnarray}
Thus, the conclusion automatically extends to the Exclusion Process.

<!-- corollary-4: body_md -->
\label{co:refined}Let $(N,k,{\mathrm{P}})$ depend on  $n\in\mathbb {N}$ in such a way that
\begin{eqnarray}

\end{eqnarray}

\end{eqnarray}

\begin{eqnarray}

The quantity on the right-hand side of (\ref{diagonal}) is the $\chi^2-$divergence w.r.t. equilibrium of the continuous-time dynamics (\ref{def:Pt}), at time $t/2$.  This function of $t$ is well known to decay exponentially at rate $\gamma=1/{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})$. Moreover, its value at $t={\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})$ is $M_1({\mathrm{P}})-1$, which is assumed to be bounded  independently of $n$. \forall i,j\in[N], \qquad {\mathrm{P}}_{t}(i,j) & \ge & \frac{1}{2N},
\end{eqnarray}

To prove the second claim, we first apply Corollary \ref{co:MFEX}  to ${\mathrm{P}}_t$ instead of ${\mathrm{P}}$ and then use Proposition \ref{pr:octopus} to deduce that for any $t\ge 0$,

\left[{1-\delta({\mathrm{P}}_t,k)}\right] \mathcal {E}_{{\mathrm{K}},k}^\textsc{ex} & \le & {{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}}_t)t}\,\mathcal {E}_{{\mathrm{P}},k}^\textsc{ex}.

\end{eqnarray*}
Finally, note that the constant on the left-hand side can be made arbitrarily close to $1$ by choosing $u$ small enough.

<!-- theorem-2: body_md -->
\label{th:fourier}The space $L^2(\Omega_k) $ admits a canonical (meaning, depending only on $N$ and $k$) orthogonal decomposition of the form
\begin{eqnarray}
\label{canonical}
L^2(\Omega_k) & = & \bigoplus_{\ell=0}^k\mathcal{H}_{k, \ell}
\end{eqnarray}
where $\mathcal{H}_{k,0}$ is the space of constant functions and for any  $\ell\in\{1,\ldots,k\}$, 
\begin{enumerate}
 \item the dimension of  $\mathcal{H}_{k,\ell}$ is  
$
d_\ell :=  {N\choose \ell}-{N\choose \ell-1};
$
  \item $\mathcal{H}_{k,\ell}$ is invariant under the natural action of $\mathfrak {S}_N$, hence in particular under  $\mathcal {L}^{\textsc{ex}}_{{\mathrm{P}},k}$;
\item the spectrum of the restriction of $\mathcal {L}^{\textsc{ex}}_{{\mathrm{P}},k}$ to $\mathcal{H}_{k,\ell}$ is independent of $k$. 
\end{enumerate}
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
Any $f\in L^2(\Omega_{k-1})$ can be  \emph{lifted} to produce a function $U_k f\in L^2(\Omega_{k})$ via
\begin{eqnarray}
U_k f(A) & := & \sum_{i\in A}f(A\setminus\{i\}).
\end{eqnarray}
Moreover, this transformation is injective as long as $k\le N/2$, since one can check that
\begin{eqnarray*}
f(A) & = & \sum_{B\in\Omega_{k}}c_{|A\cap B|}\, U_k f(B), \quad\textrm{ where }\quad c_r\ :=\ \frac{(-1)^{k-1-r}}{k-r}{N+1-k\choose k-r}^{-1}.
\end{eqnarray*}
Dually, we may \emph{downgrade} any function $g\in L^2(\Omega_{k})$ into one in $L^2(\Omega_{k-1})$ via
\begin{eqnarray}
U_k^\star g(A) & := & \sum_{i\in A^c}g(A\cup\{i\}).
\end{eqnarray}
The notation is justified by the fact that $U_k^\star\colon L^2(\Omega_{k})\to L^2(\Omega_{k-1}) $ is the adjoint of $U_k$, i.e.
\begin{eqnarray}
\forall (f,g)\in L^2(\Omega_{k-1})\times L^2(\Omega_{k}),\qquad \langle U_k f,g\rangle_{L^2(\Omega_{k})} & = & \langle  f,U_k^\star g\rangle_{L^2(\Omega_{k-1})}.
\end{eqnarray}
as is readily checked. This yields the key  orthogonal decomposition
\begin{eqnarray}
 L^2(\Omega_k) & = &  {\mathrm{Im}}(U_k)\oplus \ker(U_k^\star).
 \end{eqnarray} 
Thus, the desired decomposition of $L^2(\Omega_k)$ can be simply defined recursively by lifting the decomposition   of $L^2(\Omega_{k-1})$ through $U_k$, and then adding  its orthogonal complement:
\begin{eqnarray}
\label{rule}
\left[\mathcal{H}_{k,0} ,\ldots,\mathcal{H}_{k,k} \right] & := & \left[U_k(\mathcal{H}_{k-1,0}),\ldots,U_k(\mathcal{H}_{k-1,k-1}),\ker(U_k^\star)\right].
\end{eqnarray}
The orthogonality of $\left[\mathcal{H}_{k-1,0},\ldots,\mathcal{H}_{k-1,k-1}\right]$ is preserved under $U_k$, thanks to the identity
\begin{eqnarray}
\forall f\in\mathcal{H}_{k-1,\ell},\qquad  U_k^\star U_{k} f & = & (k-\ell)(N-k-\ell+1)f,
 \end{eqnarray} 
which can be checked by an easy induction over $k>\ell$. Finally, let us verify the three claims in the theorem by induction. First, for any $\ell\in\{0,\ldots,k-1\}$ we have
\begin{eqnarray}
\dim(\mathcal{H}_{k,\ell}) & = & \dim(\mathcal{H}_{k-1,\ell}) \ = \ d_\ell,
\end{eqnarray}
by the  injectivity of $U_k$  and the induction hypothesis, and it follows that  
\begin{eqnarray}
\dim(\mathcal{H}_{k,k}) & = & {N \choose k}- \left(d_0+\cdots+d_{k-1}\right) \ = \ d_k.
\end{eqnarray}
Second, if $T^\sigma_k$ denotes the natural action of a permutation $\sigma\in\mathfrak {S}_N$ on $L^2(\Omega_k)$ given by
\begin{eqnarray}
(T^\sigma_k f)(A) & := & f(\sigma(A)),
\end{eqnarray}
then the stability of $\mathcal{H}_{k,\ell}$ under  $T^\sigma_k$ follows inductively from the  intertwining relation
\begin{eqnarray}
 T^\sigma_k  U_{k} & = & U_{k}  T^\sigma_{k-1}.
\end{eqnarray}
Finally, in view of the expression 
\begin{eqnarray}
\mathcal {L}_{{\mathrm{P}},k}^\textsc{ex} & = & \frac{1}{2}\sum_{i,j\in[N]}{\mathrm{P}}(i,j)(T^{(\tau_{ij})}_k -\mathrm{Id}_k),
\end{eqnarray}
the operator $\mathcal {L}^\textsc{ex}_{{\mathrm{P}},k}$ inherits the same properties: it leaves $\mathcal{H}_{k,\ell}$ invariant and 
\begin{eqnarray}
 \mathcal {L}_{{\mathrm{P}},k}^\textsc{ex}  U_{k} & = & U_{k}  \mathcal {L}_{{\mathrm{P}},k-1}^\textsc{ex}.
\end{eqnarray}
Thus, any eigenfunction of $\mathcal {L}_{{\mathrm{P}},k-1}^\textsc{ex}$ in $\mathcal{H}_{k-1,\ell}$  can be lifted through $U_{k}$ to yield an eigenfunction of  $\mathcal {L}_{{\mathrm{P}},k}^\textsc{ex}$ in $\mathcal{H}_{k,\ell}$ with the same eigenvalue, establishing the third claim.

<!-- corollary-5: body_md -->
\label{co:Fourier}Let $(\mathcal {S}_t)_{t\ge 0}$ denote a $k-$particle Exclusion Process with rates ${\mathrm{P}}$, started from any initial state $\mathcal {S}_0\in\Omega_k$. Then, 
\begin{eqnarray*}
\forall t\ge 0,\qquad \chi^2(\mathrm{Law}(\mathcal {S}_t)\,|\,\pi) & \le & \sum_{\ell=1}^k d_\ell e^{-2\gamma_\ell({\mathrm{P}})t}.
\end{eqnarray*}
<!-- end corollary-5 -->
<!-- proof-7: body_md -->
Letting $f_t\colon A\mapsto {\mathbb{P}}(\mathcal {S}_t=A)$ be the probability mass function of $\mathcal {S}_t$, we have
\begin{eqnarray}
\chi^2(\mathrm{Law}(\mathcal {S}_t)\,|\,\pi) & = & |\Omega_k|\left\|f_t\right\|^2-1,
\end{eqnarray}
For each $\ell\in\{0,\ldots,k\}$, consider an orthonormal basis $(\phi_{\ell,1},\ldots,\phi_{\ell,d_\ell})$ of $\mathcal{H}_{k,\ell}$ consisting of eigenvectors of $-\mathcal {L}^{\textsc{ex}}_{{\mathrm{P}},k}$ with eigenvalues $(\gamma_{\ell,1},\ldots,\gamma_{\ell,d_\ell})$.  Since  $f_t=e^{t\mathcal {L}^\textsc{ex}_{{\mathrm{P}},k}} f_0$, we have  
\begin{eqnarray}
\left\|f_t\right\|^2 & = &  \sum_{\ell=0}^k\sum_{i=1}^{d_\ell}e^{-2t\gamma_{\ell,i}}\langle f_0,\phi_{\ell,i}\rangle^2.                                                                                                      
\end{eqnarray}
Now, the spectrum $\{\gamma_{\ell,1},\ldots,\gamma_{\ell,d_\ell}\}$ does not depend on $k$ by the above theorem, and its minimum is precisely $\gamma_\ell({\mathrm{P}})$. Moreover, the term $\ell=0$ in the above sum is simply $1/|\Omega_k|$, because $\mathcal{H}_{k,0}$ is the space of constant functions.   Thus, the last two displays imply
\begin{eqnarray}
\label{CS2}
\chi^2(\mathrm{Law}(\mathcal {S}_t)\,|\,\pi) & \le & |\Omega_k|\sum_{\ell=1}^ke^{-2t\gamma_{\ell}({\mathrm{P}})}\sum_{i=1}^{d_\ell}\langle f_0,\phi_{\ell,i}\rangle^2,
\end{eqnarray}
Finally, recall that $f_0(A)$ is $1$ if $A=\mathcal {S}_0$ and $0$ else, so that
\begin{eqnarray}
\sum_{i=1}^{d_\ell}\langle f_0,\phi_{\ell,i}\rangle^2 & = & \sum_{i=1}^{d_\ell}\phi_{\ell,i}^2(\mathcal {S}_0).
\end{eqnarray}
On the other hand, the left-hand side is the squared-norm of the projection of $f_0$ onto the space $\mathcal{H}_{k,\ell}$, and the latter is invariant under the action of $\mathfrak {S}_N$. Therefore, the norm is independent of the choice of $\mathcal {S}_0\in\Omega_k$, and summing over all such choices yields
\begin{eqnarray}
|\Omega_k|\sum_{i=1}^{d_\ell}\langle f_0,\phi_{\ell,i}\rangle^2 & = & \sum_{\mathcal {S}_0\in\Omega_k}\sum_{i=1}^{d_\ell}\phi_{\ell,i}^2(\mathcal {S}_0) \ = \ \sum_{i=1}^{d_\ell}\|\phi_{\ell,i}\|^2 \ = \ d_\ell.
\end{eqnarray}
Inserting this back into (\ref{CS2}) concludes the proof.

<!-- proof-8: body_md -->
The estimate in Corollary \ref{co:Fourier} is increasing in $k$, so we may assume that $k=\lfloor N/2\rfloor$.  
Using the definition of $\gamma_\ell({\mathrm{P}})$ at (\ref{def:gl}) and the first part of Corollary \ref{co:refined}, we know that
\begin{eqnarray}
 \gamma_\ell({\mathrm{K}}) & \le & C{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\gamma_\ell({\mathrm{P}}),
\end{eqnarray}
for each $\ell\le N/2$, where $C<\infty$ is independent of $n,\ell$. Moreover, we classically have
\begin{eqnarray}
\label{MFspectrum}
\gamma_\ell({\mathrm{K}}) & = & \frac{\ell(N-\ell+1)}{N},
\end{eqnarray}
as computed in the seminal work \cite{MR626813} and  recently rediscovered in \cite{MR3647074}. Thus,
\begin{eqnarray*}
\sum_{\ell= L}^{\lfloor N/2\rfloor }d_\ell e^{-2\gamma_\ell({\mathrm{P}}) t} & \le & \sum_{\ell=L}^{\lfloor N/2\rfloor} d_\ell\exp\left(-\frac{\ell(N-\ell+1)}{CN}\log N\right)\\
& \le & 2^N\exp\left(-\frac {L\log N}{2C}\right),
\end{eqnarray*}
for any $L\le N/2$. Choosing $L=\lfloor N/\sqrt{\log N}\rfloor$ is more than enough to make this $o(1)$.  On the other hand, when $\ell\le L$, we may use the second part of Corollary \ref{co:refined} with $k=\ell$ to obtain 
\begin{eqnarray*}
{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\gamma_\ell({\mathrm{P}}) & \ge & (1-o(1))\,\gamma_\ell({\mathrm{K}}),
\end{eqnarray*}
as $n\to\infty$. Using again the explicit formulae for $ \gamma_\ell({\mathrm{K}})$ and $d_\ell$, we obtain
 \begin{eqnarray*}
\sum_{\ell= 1}^{L}d_\ell e^{-2\gamma_\ell({\mathrm{P}}) t} & \le & \sum_{\ell=1}^{L}d_\ell\exp\left(-(1+\delta-o(1))\frac{\ell(N-\ell+1)}{N}\log N\right)\\
&\le &  \sum_{\ell=1}^{L}\frac{N^{-(\delta-o(1))\ell}}{\ell!}\\
& \le & \exp\left(N^{-\delta+o(1)}\right)-1,
\end{eqnarray*}
which is also $o(1)$ as $n\to\infty$. This concludes the proof.

<!-- proposition-5: body_md -->
\label{pr:window}Let $(N,k,{\mathrm{P}})$ depend on $n$ in such a way that
\begin{enumerate}[(i)]
\item $N\to\infty$ as $n\to\infty$;
\item $\displaystyle{0<\liminf_{n\to\infty }\frac{k}{N}\le \limsup_{n\to\infty}\frac{k}{N}<1}$;
\item $\displaystyle{\limsup_{n\to\infty}M_u({\mathrm{P}})<\infty}$ for each $u>0$. 
\item For each $\eta>0$, there is $c>0$ independent of $n$ such that $M_{\frac{c}{\log N}}({\mathrm{P}})=o(N^\eta)$.
\end{enumerate}
Then,  for any fixed
$\varepsilon\in(0,1)$,
\begin{eqnarray}
\left|\mathrm{t}_{\textsc{mix}}(\varepsilon) -\frac12 {\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\log N\right| &= & O_\varepsilon({\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})).
\end{eqnarray}

Set $t:=\frac12{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\log N$ and $L:=\lceil N^{1-\frac{1}{4C}}\rceil$, with $C$ as in  Corollary~\ref{co:refined}. Then, for $\ell\ge L$, we have ${\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\gamma_\ell({\mathrm{P}})\ge\ell/(2C)$ and $d_\ell\le \left(\frac{Ne}{\ell}\right)^\ell\le (N^\frac{1}{4C} e)^\ell$, so 
\begin{eqnarray}
\sum_{\ell=L}^{\lfloor N/2\rfloor}
d_\ell e^{-2t\gamma_\ell({\mathrm{P}})}
& \le & \sum_{\ell=L}^{\lfloor N/2\rfloor}
e^{\ell(1+\frac{1}{4C}\log N-\frac{1}{2C}\log N)} \ = \ o(1).
\end{eqnarray}
On the other hand, for $\ell\le L$, we have by (\ref{MFt}) and (\ref{def:delta}),
\begin{eqnarray}\label{eq:quantitative-spectrum}
{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})\gamma_\ell({\mathrm{P}}) & \ge & \gamma_\ell({\mathrm{K}})
\left[\frac{1-e^{-u}}{u}-\frac{2\ell M_u^2({\mathrm{P}})}{Nu}\right],
\end{eqnarray}
for  $u>0$. In view of  assumption (iv) and (\ref{MFspectrum}),  we can choose $u=c/\log N$ with $c$ independent of $n$ to make the right-hand side at least $\left(1-\frac{2c}{\log N}\right)\ell$ for  large enough  $n$, so that
\begin{eqnarray}
\sum_{\ell=1}^{L}
d_\ell e^{-2t\gamma_\ell({\mathrm{P}})} & \le & \sum_{\ell=1}^{L}
d_\ell N^{-\ell} e^{2c \ell} \ \le \ e^{e^{2c}}.
\end{eqnarray}
Consequently, Corollary~\ref{co:Fourier} yields
$
\limsup_{n\to\infty}\max_{\mathcal {S}_0\in\Omega_k}
\chi^2(\mathrm{Law}(\mathcal {S}_{t})\,|\,\pi)
 \le  e^{e^{2c}}$,
 and this is enough to prove the upper-bound, since $t\mapsto \chi(\mathrm{Law}(\mathcal {S}_{t})\,|\,\pi)$ classically decays exponentially at rate $1/{\mathrm{t}_{\textsc{rel}}}({\mathrm{P}})$ \cite{MR2629990}. 
The lower bound already follows from
Proposition~\ref{pr:LB}.
