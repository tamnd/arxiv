Let $\Psi_{\tau}$ be $L_\tau$-smooth and $m_\tau$-convex and assume $\gamma_\tau\sim \sqrt{L_\tau}$. Denote $\inf_\tau m_\tau = m_\infty$, $\sup_\tau L_\tau = L_\infty$. If $m_\infty=0$,~\cref{eq:an_under_sde} satisfies
    \begin{equation}
        D_{\mathrm{KL}}(\pi|Z_T) \lesssim \frac{\sqrt{L_\infty}}{T}\big(\mathcal{W}_2^2(\pi_x(1),\operatorname{law}(X_0)) + \mathcal{A}(\pi)\big),
    \end{equation}
    and if $m_\infty>0$ and $T\gtrsim \frac{\sqrt{L_\infty}}{m_\infty}$
    \begin{equation}
        D_{\mathrm{KL}}(\pi_{\tau}|Z_T)\lesssim \frac{L_\infty^{3/2}}{T^3 m_\infty^2}\mathcal{A}(\pi) + m_\infty\mathrm{e}^{-c\frac{m_\infty}{3\sqrt{L_\infty}} T} \; \mathcal{W}_2^2(\pi_x(1),\operatorname{law}(X_0)).
    \end{equation}

<!-- theorem-2: body_md -->
If $m_\infty>0$ we have for the exponential Euler discretization of~\cref{eq:an_under_sde} with step-sizes $h_k$ denoted as $(\mathbb{X}_k, \mathbb{V}_k)_k$ and with $T=Nh$
    \begin{equation}
        D_{\mathrm{KL}}(\operatorname{law}(X_T)|\operatorname{law}(\mathbb{X}_N))
            \lesssim\sum_{k=0}^{N-1}h_{k}^3 \frac{1}{\sqrt{L_{\tau(t_k+1)}}}\left\{\frac{\beta_{\tau(t_{k+1})}^2}{T^2} (1+\frac{d}{m_{\tau(t_{k+1})}}) +\frac{L_{\tau(t_{k+1})}^3 d}{m_{\tau(t_{k+1})}}\right\} + \frac{\varepsilon_{\mathrm{score}}^2}{\sqrt{L_1}}.
    \end{equation}
                                                                                                        
    In particular, to reach $\varepsilon$ accuracy in total variation distance, the discretization of~\cref{eq:an_under_sde} requires 
    \begin{equation}
        \tilde{\mathcal{O}}\bigg(\frac{\mathcal{A}(\pi)^{1/6}\beta_\infty d^{1/2}}{\varepsilon^{4/3}m_\infty^{5/6}} 
        \vee \frac{\mathcal{A}(\pi)^{1/2}L_\infty^2d^{1/2}}{m_\infty^{3/2}\varepsilon^{2}}
        \vee
        \frac{\beta_\infty\xi^{1/2}d^{1/4}}{m_\infty^{3/4}\varepsilon}
        \vee
        \frac{L_\infty^2\xi^{3/2}}{d^{1/4}m_\infty^{5/4}\varepsilon}\bigg)
    \end{equation}
    iterations.

<!-- head-1: body_md -->
Throughout this article $x$ will be used for the position variable, $v$ for the velocity, and $z=(x,v)$ for the position-velocity pair. Depending on context, these might appear as a capital or lower-case letter or in other variations ($x$, $X$, $\mathbb{X}$, $\bar X$ etc.).
We use Greek letters to denote probability measures with $\pi(\tau,\mathrm{d} z)\propto\mathrm{e}^{-\Psi_\tau(x)-\frac{1}{2}|v|^2}\mathrm{d} z$ the (moving) target, $\mu(t,z)$ the distribution of~\cref{eq:an_under_sde}, and $\hat\mu(t,z)$ the distribution of the discretization of~\cref{eq:an_under_sde} introduced in~\cref{sec:discrete_time_analysis}. Note here the difference in that $\pi$ depends on the parameter $\tau$ (which itself will be time-dependent), whereas $\mu$, $\hat \mu$ depend on the time $t$. Moreover, $p(\tau,z)$, $q(t,z)$, and $\hat q(t,z)$ denote the densities with respect to the Lebesgue measure of $\pi(\tau,z)$, $\mu(t,z)$, and $\hat\mu(t,z)$, respectively. We will use subscripts to denote marginal distributions, e.g., $p_x(\tau,x) = \int p(\tau,x,v)\mathrm{d} v$ and analogously for the measures. Lastly, for ease of notation, we will sometimes write $\mu(t)=\mu(t,\mathrm{d} z)$, $q(t)=q(t,z)$, etc.
The set of probability measures on \(\mathbb{R}^d\) is denoted as \(\mathcal{P}(\mathbb{R}^d)\). We denote with \(\mu \ll \nu\) that the probability measure \(\mu\) is absolutely continuous w.r.t. \(\nu\), and with \(\frac{\mathrm{d} \mu}{\mathrm{d} \nu}\) the Radon-Nikodym derivative of \(\mu\) w.r.t. \(\nu\). For a probability measure \(\mu \in \operatorname{\mathbb{P}}(\mathbb{R}^d)\) and function \(f: \mathbb{R}^d \rightarrow \mathbb{R}^d\), \(\left\lVertf\right\rVert_{L^2(\mu)} = (\int \left\lVertf\right\rVert^2 \mathrm{d} \mu)^\frac12\) and the \(n\)-th moment is \(\mathbb{E}_\mu[|\cdot|^n]\) for \(n > 0\). We define the the Wasserstein-2 distance between \(\mu, \nu \in \mathcal{P}(\mathbb{R}^d)\) as \(W_2(\mu, \nu) = \inf_{\gamma \in \Pi(\mu, \nu)} (\int |x - y|^2 \gamma(\mathrm{d} x, \mathrm{d} y))^{\frac12}\) where \(\Pi(\mu, \nu)\) is the set of probability measures on \(\mathbb{R}^d \times \mathbb{R}^d\) with marginals \((\mu)\) and \(\nu\), the total-variation distance as \(\|\mu - \nu\|_{\operatorname*{TV}} = \sup_{A \subset \mathbb{R}^d}|\mu(A) - \nu(A)|\) and the Kullback--Leibler divergence as \(D_{\mathrm{KL}}(\mu \mid \nu)  = \int \log \frac{\mathrm{d} \mu}{\mathrm{d} \nu} \mathrm{d} \mu\) for \(\mu \ll \nu\) and $\infty$ otherwise.
For $\mathcal{W}_2$, $\|\cdot\|_{\operatorname*{TV}}$, and $D_{\mathrm{KL}}$ we will often write, \textit{e.g.}, $\mathcal{W}_2(X,Y)$ to denote $\mathcal{W}(\operatorname{law}(X),\operatorname{law}(Y))$ for two random variables $X,Y$.
The symbol $\tilde{\mathcal{O}}$ is used to denote complexity results omitting logarithmic factors.

<!-- assumption-1: body_md -->
\label{ass}\
    The potential family \(\Psi_\tau \in \mathcal{C}^2(\mathbb{R}^d)\) satisfies
    \begin{enumerate}
    \item There exist \(0 \leq m_\tau \leq L_\tau\) such that $m_\tau I\preceq \nabla^2 \Psi_\tau(x) \preceq L_\tau I$ for all \(x \in \mathbb{R}^d\) and \(\tau \in [0,1]\). Furthermore $L_\infty\coloneqq\sup_{\tau}L_\tau<\infty$ and $m_\infty \coloneqq \inf_\tau m_\tau\geq 0$.\label{ass:pot1} 
    We assume without loss of generality that $L_\tau$ and $m_\tau$ are decreasing, respectively increasing in $\tau$.\footnote{This can always be achieved by replacing $L_\tau$ by $\sup_{\tau'\geq \tau}L_{\tau'}$ and similarly for $m_\tau$.}
    \item There exists an increasing function $\tau\mapsto\beta_\tau>0$ such that $|\partial_\tau \nabla\Psi_\tau(x)|\leq \beta_\tau (1+|x|)$. We denote $\beta_\infty = \sup_\tau \beta_\tau$.\label{ass:pot2}
    \item Let $x_\tau^*\in \operatorname*{arg\,min}_{x}\Psi_\tau(x)$. There exists $\xi>0$ such that $|\partial_\tau x^*_\tau| \leq \xi$.\label{ass:pot3}
    \end{enumerate}

<!-- assumption-2: body_md -->
\label{ass:schedule}
The annealing schedule $\tau:[0,T]\to [0,1]$ is of the form $\tau(t) = \chi(\frac{t}{T})$ such that the following hold: 
\begin{enumerate}
    \item $\chi: [0,1]\rightarrow [0,1]$ is strictly decreasing with $\chi(0)=1$, $\chi(1)=0$\label{ass:schedule1}
    \item There exists $\alpha>0$ such that $\sup_{t\in[0,1]}\frac{|\dot \chi(t)|}{(1-t)^2}\leq \alpha$.\label{ass:schedule2}
\end{enumerate}
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:friction}
    There exists $1= \tau_0>\tau_1>\dots >\tau_n=0$ such that $\gamma_{\tau} = \gamma_k$ for $\tau\in (\tau_k,\tau_{k+1}]$ and there exists $\nu>0$ such that $\sup_k\frac{|\log \gamma_{k+1} - \log\gamma_k|}{\tau_{k+1}-\tau_k}\leq \nu$. We denote $\gamma_\infty = \sup_k \gamma_k$.

<!-- remark-1: body_md -->
Note that we will later fix the choice $\gamma_k = 2\sqrt{L_{\tau_k}}$. In this case we have for the variance preserving diffusion model where $L_\tau = \frac{L_0}{1+\tau(L_0-1)}$ (\textit{cf.}\,\cref{lemma:tau_dep_hessian_bounds})
    \begin{equation}
        \partial_\tau \log\sqrt{L_\tau} = \frac{\tau (L_0 - 1)}{1+ \tau (L_0 - 1)}.
    \end{equation}
    and by monotonicity of this derivative $\nu_k \lesssim L_0$.

<!-- example-1: body_md -->
\label{example:conv}
    Let $\Psi_0$ be $m_0$-strongly convex and $L_0$-smooth and assume $\|\nabla^3\log p(0,x)\|_{op}\leq M$ for some $M>0$ and all $x$. Then all assumptions are satisfied for the variance preserving convolutional path defined for $\kappa>0$ and $\tau\in [0,1-\kappa]$
    \begin{equation}
        \pi_\tau \coloneqq \operatorname{law}(\sqrt{1-\tau}X + \sqrt{\tau}Z), \quad X \perp Z,\; X\sim \pi_0,\; Z\sim \mathcal{N}(0,I_d)
    \end{equation}
    with the schedule defined by $\chi(t) = \big(\frac{1+\cos(\pi t)}{2}\big)^2$ or simply $\chi(t) = (1-t)^3$. See~\cref{sec:convolutional} for a proof.

\label{ass:shifts}

<!-- lemma-1: body_md -->
\label{lemma:contraction}
    Denote the errors between auxiliary and reference process in transformed coordinates as
    \begin{equation}\label{eq:shifted coordinates}
        T_t \coloneqq \begin{bmatrix}
                \hat X_t - X_t^{{}^{\text{aux}}}\\
                \hat P_t - P_t^{{}^{\text{aux}}}
    \end{bmatrix}.
    \end{equation}
    Then it holds true that 
                                                         
                         
                               
                                            
                                                     
                          
                       
                    
    \begin{equation}\label{eq:shift_error_differential}
        \begin{aligned}
            \mathrm{d} \left\lVertT_t\right\rVert^2 
            = -2 \left\langleT_t,Q_tT_t\right\rangle \mathrm{d} t
            + 2\left\langleT_t,\begin{bmatrix} b_t\\b_t
    \end{bmatrix}\right\rangle \mathrm{d} t
        \end{aligned}
    \end{equation}
    where 
    \begin{equation*}
    Q_t = \begin{bmatrix}
        \frac{\hat{\gamma}_t}{2}I & (\frac{\eta_t^x}{\hat{\gamma}_t} - \frac{\hat{\gamma}_t}{2} - \frac{\dot \hat{\gamma}_t}{2\hat{\gamma}_t})I + \frac{1}{\hat{\gamma}_t}H_t \\
        (\frac{\eta_t^x}{\hat{\gamma}_t} - \frac{\hat{\gamma}_t}{2} - \frac{\dot \hat{\gamma}_t}{2\hat{\gamma}_t})I + \frac{1}{\hat{\gamma}_t}H_t & (\frac{\hat{\gamma}_t}{2} + \frac{\dot \hat{\gamma}_t}{\hat{\gamma}_t})I
    \end{bmatrix}.
\end{equation*}
and a random process $H_t\in \mathbb{R}^{d\times d}\succeq m$ given in the proof.

<!-- lemma-2: body_md -->
\label{thm:continuous}
    Assume $m_\infty\geq 0$ and the shift parameters are chosen according to~\cref{ass:shifts}, \cref{item_shifts_non-strongly}. Recall that $\hat Z_t\sim\pi(\tau(t))$ denotes the reference process~\ref{eq:reference_underdamped} and $Z_t$ the annealed underdamped Langevin process~\ref{eq:an_under_sde}. Then it holds true that
                      
                                                                                                                           
                    
    \begin{equation}
        D_{\mathrm{KL}}(\hat Z_T|Z_T) \lesssim \bigg(\frac{\gamma_\infty}{T} + \frac{1}{ T^3}\bigg)\big(\mathcal{A}(\pi) + \mathcal{W}^2_2(\hat Z_0, Z_0)\big).
    \end{equation}
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{thm:continuous_strongly}
    Let $m_\infty>0$, that is, $\Psi_\tau$ is uniformly strongly-convex and let $\eta^x_t, \eta^v_t$ be chosen according to~\cref{ass:shifts}, \cref{item_shifts_strongly}. Then,
    \begin{equation}
        D_{\mathrm{KL}}(\hat Z_T|Z_T)\lesssim \frac{(\gamma_\infty+\omega^2)}{\omega^2 T^3}\mathcal{A}(\pi) + \frac{\gamma_\infty\omega + \omega^3}{(1-\mathrm{e}^{-\omega T})^{4}}\mathrm{e}^{-c'\omega T} \; \mathcal{W}^2_2(\hat Z_0, Z_0).
    \end{equation}
<!-- end lemma-3 -->
<!-- remark-4: body_md -->
The gains in the strongly convex case are two-fold: (i) The initialization error caused by $T_0$ decays now exponentially in $T$ instead of $T^{-1}$. (ii) The annealing error decays as $T^{-3}$ instead of $T^{-1}$.

<!-- remark-5: body_md -->
\label{rem:finite_action}

<!-- proof-1: body_md -->
Denote the discrepancy in original coordinates as $D_t = \hat Z_t - Z^{{}^{\text{aux}}}_t$.
By the synchronous coupling we have
\begin{align} \label{eq:ito_sq_T_t}
    \mathrm{d} \left\lVertT_t\right\rVert^2 = 2\langle T_t,  \mathrm{d} T_t \rangle
\end{align}
and 
\begin{align} \label{eq:difference_sde_transformed}
    \mathrm{d} T_t = \mathrm{d} (A_t D_t) 
    &= \dot A_t D_t \mathrm{d} t  + A_t \mathrm{d} D_t.
\end{align}
For the second term we find,
\begin{align}
    \mathrm{d} D_t &= 
    \begin{bmatrix}
        0 & I \\
        -\eta_t^x I - H_t & - \hat{\gamma}_t I  
    \end{bmatrix}
    D_t \mathrm{d} t 
    + \begin{bmatrix}
        b(t,\hat X_t) \\
        0
    \end{bmatrix} \mathrm{d} t\\
    &= M_t D_t \mathrm{d} t + \begin{bmatrix}
        b(t,\hat X_t) \\
        0
    \end{bmatrix} \mathrm{d} t
\end{align}
where we used that by the fundamental theorem of calculus
\begin{equation}
    \nabla_x \Psi_\tau(t)(\hat X_t) - \nabla_x \Psi_\tau(t)(X_t^{{}^{\text{aux}}}) = \underbrace{\int_0^1\nabla^2 \Psi_{\tau(t)}(X_t^{{}^{\text{aux}}} + \lambda (\hat X_t-X^{{}^{\text{aux}}}_t) \mathrm{d} \lambda}_{\eqqcolon H_t}\, (\hat X_t - X_t^{{}^{\text{aux}}}).
\end{equation}
Plugging into \cref{eq:ito_sq_T_t} yields
\begin{align*}
    \mathrm{d} \left\lVertT_t\right\rVert^2 
    =& 2 \left\langleT_t,(\dot A_t + A_t M_t)A_t^{-1} T_t\right\rangle \mathrm{d} t
    + 2\left\langleT_t,\begin{bmatrix} b_t\\b_t
    \end{bmatrix}\right\rangle \mathrm{d} t
                                                                                                                                    
                                                                                  
\end{align*}
where we recognize $Q = -\mathrm{sym}((\dot A_t + A_t M_t)A_t^{-1})$ with the symmetrization $\mathrm{sym}(M) \coloneqq (M+M^\top)/2$ for any quadratic matrix $M$.

<!-- lemma-6: body_md -->
\label{lemma:T_bound_varying_fric}
    Assume $T\gtrsim \nu$. It holds for some absolute constant $c'>0$
    \begin{equation}
        \begin{aligned}
            \mathbb{E}[|T_t|^2]
            \lesssim& \bigg(\frac{T-t}{T}\bigg)^{c_0c'}
            \mathbb{E}[|T_0|^2] + \int_0^t \frac{(T-t)^{c'c_0}}{(T-s)^{c_0c'-1}}\mathbb{E}[|b_s|^2]\mathrm{d} s\\
        \end{aligned}
    \end{equation}
<!-- end lemma-6 -->
<!-- proof-2: body_md -->
By~\cref{lemma_eigenvalue_bound} we have $\lambda_{\min}(Q_t)\geq \frac{\eta^v_t}{48}$. Applying Young's inequality to~\ref{eq:shift_error_differential} then yields
    \begin{equation}
        \begin{aligned}
            \mathrm{d} |T_t|^2 
            =& - \lambda_{\min}(Q_t)|T_t|^2 + 2\frac{|b_t|^2}{\lambda_{\min}(Q_t)} \mathrm{d} t\\
            \leq& - \frac{\eta^v_t}{48}|T_t|^2 + 92\frac{|b_t|^2}{\eta^v_t} \mathrm{d} t.
        \end{aligned}
    \end{equation}
    Taking expectations and using Grönwall's inequality and the definition of $\eta^v_t$ leads to
    \begin{equation}\label{eq:KL_boundproof3}
        \begin{aligned}
            \mathbb{E}[|T_{t_{k+1}}|^2]
            \lesssim& \mathrm{e}^{-c'\int_{t_k}^t\lambda_{\min}(Q_s)\mathrm{d} s}
            \mathbb{E}[|T_{t_k}^+|^2] + \int_{t_k}^{t_{k+1}} \mathrm{e}^{-c'\int_{s}^t\lambda_{\min}(Q_r)\mathrm{d} r}\mathbb{E}[|b_s|^2]\mathrm{d} s\\
                                                             
                                                                                                                                
        \end{aligned}
    \end{equation}
    for some $c'>0$. Note that in the grid edges $(t_k)_k$, the transformation matrix $A_t$ has a discontinuity and, thus, we define $A_{t_k}^+ = \lim_{t\downarrow t_k}A_t$, $z^+_{t_k} = A_{t_k}^+ z_{t_k}$ and analogously $T_{t_k}^+$. Then one can check that 
    \begin{equation}
        A_{t_k}^+ A_{t_k}^{-1} = I_d + (1-r_k)\begin{bmatrix}
            0&0\\
            1&-1
        \end{bmatrix}
    \end{equation}
    where $r_k = \frac{\gamma_{k} + \eta^v_{t_k}}{\gamma_{k+1} + \eta^v_{t_{k}}}$. Then we have for the spectral norm 
    \begin{equation}
        \|A_{t_k}^+ A_{t_k}^{-1}\|
        \leq 1+\sqrt{2}(1-r_k)
        \leq \mathrm{e}^{\log(1+\sqrt{2}(1-r_k))}
        \leq \mathrm{e}^{\sqrt{2}(1-r_k)}.
    \end{equation}
    Moreover, since for $0<a<b$, $x>0$ we have $\frac{a+x}{b+x}\geq \frac{a}{b}$ it holds that $r_k\geq \frac{\gamma_k}{\gamma_{k+1}}$ and we have
    \begin{equation}
        1-r_k
        = 1-\mathrm{e}^{\log \frac{\gamma_k}{\gamma_{k+1}}}
                                                                                     
        \leq 1-\mathrm{e}^{-\int_{t_k}^{t_{k+1}}\nu |\dot \tau(t)|\mathrm{d} t}
        \leq \int_{t_k}^{t_{k+1}}\nu |\dot \tau(t)|\mathrm{d} t
    \end{equation}
    with $\nu$ from~\cref{ass:friction}. Inserting $|T_k^+|\leq \|A_{t_k}^+ A_{t_k}^{-1}\||T_k|$ into~\ref{eq:KL_boundproof3} gives
    \begin{equation}
        \begin{aligned}
            \mathbb{E}[|T_{t_{k+1}}|^2]
            \lesssim\mathrm{e}^{-\int_{t_k}^tc'\lambda_{\min}(Q_s) - 2\sqrt{2}\nu |\dot \tau(s)|\mathrm{d} s}
            \mathbb{E}[|T_{t_k}|^2] + \int_{t_k}^{t_{k+1}} \mathrm{e}^{-c'\int_{s}^t\lambda_{\min}(Q_r)\mathrm{d} r}\mathbb{E}[|b_s|^2]\mathrm{d} s.
        \end{aligned}
    \end{equation}
    We have
    \begin{equation}

\label{lemma:optimal_velocity_independent}
    Let $\mu_x(t),\mu_v(t)$, $t\in[0,T]$ be absolutely continuous curves in $\mathcal{P}_2(\mathbb{R}^{d})$. Let $\hat b_x(t)\in L^2(\mu_x(t))$, $\hat b_v(t)\in L^2(\mu_v(t))$ be the uniquely determined smallest vector fields generating $\mu_x(t),\mu_v(v)$ according to~\cref{thm:optimal_velocity}. 
              
                      
                                                                                                 
                    
                                                                                                      
                                                                                                                                                                                                 
    Then it holds true that $(\hat b_x(t),\hat b_v(t))\in L^2(\mu_x(t)\otimes\mu_v(t))$ is the smallest velocity field generating the path $\mu = \mu_x\otimes\mu_v$.
<!-- end lemma-8 -->
<!-- proof-7: body_md -->
First of all, it is clear that $(\hat b_x(t),\hat b_v(t))\in L^2(\mu_x(t)\otimes\mu_v(t))$ for a.e. $t$ as
    \begin{equation}
        \iint |(\hat b_x(t,x),\hat b_v(t,v))|^2\mathrm{d} \mu_x(t,x)\mathrm{d} \mu_v(t,v)
        = \|b_x(t)\|_{L^2(\mu_x(t))}^2+\|b_v(t)\|_{L^2(\mu_v(t))}^2.
    \end{equation}
    Moreover, one can easily see that $(\hat b_x(t),\hat b_v(t))$ in fact generates the path $\mu$. 
    In order to prove that $(\hat b_x(t),\hat b_v(t))$ is the minimal velocity field, let $(b_x(t,x,v),b_v(t,x,v))$ be an arbitrary velocity field generating $\mu$. We will show that its $L^2$ norm is a.s. lower-bounded by that of $\hat b$.
                                                                          
                      
                                                                                           
                    
    Note that the continuity equation with $(b_x,b_v)$ reads as
    \begin{equation}
        \begin{aligned}
            \partial_t \mu
            = (\partial_t\mu_x)\mu_v + \mu_x(\partial_t\mu_v)
            =&\nabla\cdot \begin{bmatrix}
                \begin{pmatrix}
                    b_x\\
                    b_v
                \end{pmatrix}
                \mu
            \end{bmatrix}\\
            =& \mu_v \nabla_x\cdot (b_x \mu_x) + \mu_x \nabla_v\cdot (b_v\mu_v)\\
        \end{aligned}
    \end{equation}
    Integrating over $v$ yields
    \begin{equation}
        \begin{aligned}
            \partial_t\mu_x
            =& \nabla_x\cdot \bigg(\int b_x(t,x,v)\mathrm{d} \mu_v(t,v) \mu_x\bigg).
        \end{aligned}
    \end{equation}
    and similarly for integrating over $x$. That is, the $x$ and $v$ marginals satisfy transport equations with the $v$- and $x$-averaged velocity fields, $\bar b_x(t,x) := \int b_x(t,x,v)\mathrm{d} \mu_v(t,v)$ and $\bar b_v(t,v):= \int b_v(t,x,v)\mathrm{d} \mu_x(t,x)$, respectively. But then, by minimality of $\hat b_x$ and $\hat b_v$ it follows for a.e. $t$
    \begin{equation}
        \begin{aligned}
            \|\hat b_x(t)\|_{L^2(\mu_x(t))}&\leq \|\bar b_x(t)\|_{L^2(\mu_x(t))},\text{ and}\\
            \|\hat b_v(t)\|_{L^2(\mu_v(t))}&\leq \|\bar b_v(t)\|_{L^2(\mu_v(t))}
        \end{aligned}
    \end{equation}
    and, thus, by Hölder's inequality for a.e. $t$ we have
    \begin{equation}
        \begin{aligned}
            \|(\hat b_x(t),&\hat b_v(t))\|_{L^2(\mu(t))}^2 
            \leq \int |\bar b_x(t)|^2\mathrm{d} \mu_x + \int |\bar b_v(t)|^2\mathrm{d} \mu_v\\
            =& \int \bigg| \int b_x(t,x,v)\mathrm{d}\mu_v(t,v)\bigg|^2\mathrm{d} \mu_x(t,x) + \int \bigg| \int b_v(t,x,v)\mathrm{d}\mu_x(t,x)\bigg|^2\mathrm{d} \mu_v(t,v)\\
            \leq& \iint | b_x(t,x,v)|^2\mathrm{d}\mu_v(t,v)\mathrm{d} \mu_x(t,x) + \iint |b_v(t,x,v)|^2\mathrm{d}\mu_x(t,x)\mathrm{d} \mu_v(t,v)\\
            =& \|(b_x(t),b_v(t))\|_{L^2(\mu(t))}^2
        \end{aligned}
    \end{equation}
    concluding the proof.

\label{lemma:bound_minimizer}

<!-- proof-8: body_md -->
Let $(X^\tau_t)_t$ follow
    \begin{equation}
        \mathrm{d} X^\tau_t = -\nabla \Psi_\tau (X^\tau_t)\mathrm{d} t + \sqrt{2} \mathrm{d} W_t
    \end{equation}
    that is, the Langevin diffusion targeting $\pi(\tau)$ for \emph{frozen} $\tau$. By It\^{o}{}'s lemma
    \begin{equation}
        \begin{aligned}
            \mathrm{d} |X^\tau_t - x^*_\tau|^2 
            &= \{ -2\left\langleX^\tau_t - x^*_\tau,\nabla \Psi_\tau (X^\tau_t) - \nabla \Psi_\tau (x^*_\tau)\right\rangle + d\}\mathrm{d} t + M_t\\
            &\leq \{ -2m_\tau|X^\tau_t - x^*_\tau|^2 + d\}\mathrm{d} t + M_t,\\
        \end{aligned}
    \end{equation}
    where \(M_t\) is a martingale.
    Taking the expectation and applying Grönwall's lemma yields
    \begin{equation}
        \mathbb{E}[|X^\tau_t - x^*_\tau|^2]
        \leq \mathrm{e}^{-2m_\tau t}\left (\mathbb{E}[|X^\tau_0 - x^*_\tau|^2] - \frac{d}{2m_\tau}\right) + \frac{d}{2m_\tau}.
    \end{equation}
    Taking the limit $t\rightarrow 0$ yields
    \begin{equation}
        \lim_{t\rightarrow \infty}\mathbb{E}[|X^\tau_t - x^*_\tau|^2] = \int |x-x^*_\tau|^2 \pi(\tau,\mathrm{d} x) \leq \frac{d}{m_\tau}.
    \end{equation}
    Lastly, denoting $\bar x_\tau = \int x \pi(\tau,\mathrm{d} x)$ we have
    \begin{equation}
        |\bar x_\tau - x^*_\tau|^2 
        = |\mathbb{E}_{\pi(\tau)}[X-x^*_\tau]|^2 
        \leq \mathbb{E}_{\pi(\tau)}[|X-x^*_\tau|^2]
        \leq \frac{d}{m_\tau}
    \end{equation}
    and consequently, $|x^*_\tau|\leq |\bar x_\tau| + |x^*_\tau - \bar x_\tau| \lesssim \sqrt{d/m_\tau}$.

<!-- lemma-10: body_md -->
\label{lemma:moment_bound_moving}
                                                       
                      
                                                                    
                    
    Let $\gamma = 2\sqrt{L_\infty}$. Define 
    \begin{equation}
        \begin{aligned}
            \omega_\tau(x,v) 
            \coloneqq& 2|x-x_\tau^*|^2 + \frac{4}{\gamma}\left\langlex-x_\tau^*,v\right\rangle + \frac{4}{\gamma^2}|v|^2\\
        \end{aligned}
    \end{equation}
    where $x_\tau^*\in\operatorname*{arg\,min}_x \Psi_\tau(x)$. Denote the generator at time $t$ of~\ref{eq:an_under_sde} as $\mathcal{L}_{\tau(t)}$. Then it holds true that 
    \begin{equation}
        \mathcal{L}_{\tau(t)} \omega_{\tau(t)}(x,v)
            \leq -\frac{m_\infty}{8\sqrt{L_\infty}}\omega_{\tau(t)}(x,v)+\frac{4d}{\sqrt{L_\infty}} + 
            \frac{\xi^2}{T^2}\bigg(\frac{4(m_\infty + L_\infty)}{m_\infty \sqrt{L_\infty}} + \frac{2}{\sqrt{L_\infty}} \bigg)
    \end{equation}
    and consequently, 
    \begin{equation}
        \begin{aligned}
            \mathbb{E}[\omega_{\tau(t_k)}(X_{t_k},V_{t_k})]
            \lesssim \mathbb{E}[\omega_{\tau(0)}(X_{0},V_{0})] 
            + \frac{4d}{m_{\tau_{k-1}}} + 
            \frac{8 \xi^2L_{\tau_{k-1}}}{T^2 m_{\tau_{k-1}}^2 }.
        \end{aligned}
    \end{equation}
    In particular, if $\mathbb{E}[\omega_{\tau(0)}(X_0,V_0)]\lesssim \frac{d}{m_0} + \frac{\xi^2L_0}{T^2m_0^2}$, than $\mathbb{E}[\omega_{\tau(t)}(X_t,V_t)]\lesssim \frac{d}{m_{\tau(t)}} + \frac{\xi^2L_{\tau(t)}}{T^2m_{\tau(t)}^2}$ for all $t\geq 0$.

<!-- proof-9: body_md -->
In the following we drop the time dependence from the notation and write $\tau$ instead of $\tau(t)$ for simplicity.
    By It\^{o}{}'s lemma within $(\tau_k,\tau_{k+1})$ where $\gamma_\tau=\gamma_k$ we find
    \begin{equation}\label{eq:lyapunov_cont1}
        \begin{aligned}
            \mathcal{L}_\tau \omega_\tau(x,v)
            \leq& -\frac{4}{\gamma_\tau}\left\langlex-x^*_\tau,\nabla\Psi_\tau(x)\right\rangle
            -\frac{4}{\gamma_\tau}|v|^2
            -\frac{8}{\gamma_\tau^2}\left\langlev,\nabla\Psi_\tau(x)\right\rangle
            +\frac{8d}{\gamma_\tau} \\
            &+ 4\left\langlex^*_\tau - x + \frac{1}{\gamma_\tau} v,\dot \tau  \partial_\tau x^*_{\tau(t)}\right\rangle
        \end{aligned}
\end{equation}
    Using~\citep[Theorem 2.1.12]{nesterov2013introductory} we find
    \begin{equation}
        \begin{aligned}
            \left\langlex-x^*_\tau,\nabla \Psi_\tau(x)\right\rangle
            \geq& \frac{m_\tau L_\tau}{m_\tau + L_\tau}|x-x^*_\tau|^2 + \frac{1}{m_\tau + L_\tau}|\nabla \Psi_\tau(x)|^2
        \end{aligned}
    \end{equation}
    Moreover, for any $\delta>0$ which (will be chosen in a moment) again by Young
    \begin{equation}
        \left\langlev,\nabla\Psi_\tau(x)\right\rangle
        \leq \frac{1}{2\delta}|v|^2 + \frac{\delta}{2}|\nabla\Psi_\tau(x)|^2.
    \end{equation}
    Inserting these estimates in~\ref{eq:lyapunov_cont1} we obtain
    \begin{equation}
        \begin{aligned}
            \mathcal{L}_\tau \omega_\tau(x,v)
            \leq& -\frac{4}{\gamma_\tau}\frac{m_\tau L_\tau}{m_\tau + L_\tau}|x-x^*_\tau|^2 -|v|^2\big(\frac{4}{\gamma_\tau} - \frac{4}{\delta\gamma_\tau^2}\big)\\
            &+|\nabla\Psi_\tau(x)|^2 \{\frac{4\delta}{\gamma_\tau^2} - \frac{4}{\gamma_\tau(m_\tau + L_\tau)}\}
            +\frac{8d}{\gamma_\tau} + 4\left\langlex^*_\tau - x + \frac{1}{\gamma_\tau} v,\dot x^*_\tau\right\rangle.
        \end{aligned}
    \end{equation}
    Now we pick $\delta = \frac{\gamma_\tau}{m_\tau+L_\tau}$ so that the term involving $\nabla\Psi_\tau$ vanishes.
    Moreover, in order for the coefficient of $|v|^2$ to be negative, we require
    \begin{equation}
        1<\delta\gamma_\tau = \frac{\gamma_\tau^2}{m_\tau + L_\tau}
    \end{equation}
    which is true for the choice $\gamma_\tau = 2\sqrt{L_\tau}$. Using that by~\cref{ass,ass:schedule} $|\dot \tau \partial_\tau x^*_\tau|\lesssim \frac{\xi}{T}$ and applying Young we find
    \begin{equation}\label{eq:lyapunov_cont2}
        \begin{aligned}
            \mathcal{L}_\tau \omega_\tau(x,v)
                                                                                                                                                                    
                                                                                 
            \leq& -\frac{2}{\gamma_\tau}\frac{m_\tau L_\tau}{m_\tau + L_\tau}|x-x^*_\tau|^2 -\frac{1}{\gamma_\tau}|v|^2
            +\frac{8d}{\gamma_\tau} + 
            \frac{\xi^2}{T^2}\big(\frac{2\gamma_\tau(m_\tau + L_\tau)}{m_\tau L_\tau} + \frac{4}{\gamma_\tau} \big)
        \end{aligned}
    \end{equation}
    Note that we can write
    \begin{equation}
        \omega_\tau(x,v) 
        = \begin{bmatrix}
            x-x^*_\tau\\
            v
        \end{bmatrix}^\top
        \begin{bmatrix}
            2& \frac{2}{\gamma_\tau}\\
            *& \frac{4}{\gamma_\tau^2}
        \end{bmatrix}
        \begin{bmatrix}
            x-x^*_\tau\\
            v
        \end{bmatrix}.
    \end{equation}
    Thus, in view of~\ref{eq:lyapunov_cont2}, to ensure a drift condition of the form $\mathcal{L}_\tau \omega_\tau\leq -\lambda \omega_\tau + b$ with $\lambda,b>0$ we have to ensure that
    \begin{equation}
        \begin{bmatrix}
            -\frac{2}{\gamma_\tau}\frac{m_\tau L_\tau}{m_\tau + L_\tau} + 2\lambda& \lambda\frac{2}{\gamma_\tau}\\
            *& -\frac{1}{\gamma_\tau}+ \lambda \frac{4}{\gamma_\tau^2}
        \end{bmatrix}\preceq 0
    \end{equation}
    which is the case if the above matrix admits negative trace and positive determinant, which is true for $\lambda = \frac{m_\tau}{8\sqrt{L_\tau}}$.
    Indeed, with this choice the trace reads as
    \begin{equation}
        \begin{aligned}
            -\frac{2}{\gamma_\tau}\frac{m_\tau L_\tau}{m_\tau + L_\tau} + 2\lambda -\frac{1}{\gamma_\tau}+ \lambda \frac{4}{\gamma_\tau^2}
            =&-\frac{m_\tau \sqrt{L_\tau}}{m_\tau + L_\tau} + \frac{m_\tau}{4\sqrt{L_\tau}} -\frac{1}{2\sqrt{L_\tau}}+ \frac{m_\tau}{8 L_\tau^{3/2}}\\
            \leq&-\frac{m_\tau}{\sqrt{L_\tau}} + \frac{m_\tau}{4\sqrt{L_\tau}} -\frac{1}{2\sqrt{L_\tau}}+ \frac{1}{8 \sqrt{L_\tau}}\\
            <&0
        \end{aligned}
    \end{equation}
    where we used that $m_\tau\leq L_\tau$, respectively $m_\tau/L_\tau\leq 1$. For the determinant, on the other hand, we obtain
    \begin{equation}
        \begin{aligned}
            (-\frac{2}{\gamma_\tau}&\frac{m_\tau L_\tau}{m_\tau + L_\tau} + 2\lambda)(-\frac{1}{\gamma_\tau}+ \lambda \frac{4}{\gamma_\tau^2}) -  \lambda^2\frac{4}{\gamma_\tau^2}\\

            \geq& \frac{m_\tau}{2(m_\tau+L_\tau)} - \frac{ m_\tau^2}{8L_\tau(m_\tau+L_\tau)} - \frac{m_\tau}{8L_\tau} + \frac{m_\tau^2}{64L_\tau^2}\\
            \geq& \frac{m_\tau}{L_\tau}\big(\frac{1}{4} - \frac{1}{16} - \frac{1}{8}\big) + \frac{m_\tau^2}{128L_\tau^2}>0.
        \end{aligned}
    \end{equation}
    It follows,
    \begin{equation}\label{eq:moment_bound_omega}
        \begin{aligned}
            \mathcal{L}_\tau \omega_\tau(x,v)
            \leq& -\frac{m_\tau}{8\sqrt{L_\tau}}\omega_\tau(x,v)+\frac{8d}{\gamma_\tau} + \frac{4}{\gamma_\tau}(m_\tau + 
            \frac{\xi^2}{T^2}\big(\frac{2\gamma_\tau(m_\tau + L_\tau)}{m_\tau L_\tau} + \frac{4}{\gamma_\tau} \big)\\
            \lesssim& -\frac{m_\tau}{8\sqrt{L_\tau}}\omega_\tau(x,v) + \frac{4d}{\sqrt{L_\tau}} +

        \end{aligned}
    \end{equation}
    Now we note that with the notation 
    \begin{equation}
        A=\begin{bmatrix}
            1&0\\
            1&\frac{2}{\gamma_k}
        \end{bmatrix},\quad 
        A_+=\begin{bmatrix}
            1&0\\

        \end{bmatrix}
    \end{equation}
    we have that

        \omega_{\tau_k}(x,v) 
        = |A_+(x,v)|
        = |A_+A^{-1}A(x,v)|
        \leq \|A_+A^{-1}\|\omega_{\tau_{k+1}}(x,v)

As in the proof of~\cref{lemma:T_bound_varying_fric} we have $\|A_+A^{-1}\|\leq \mathrm{e}^{\sqrt{2}\int_{t_k}^{t_{k+1}} \nu |\dot \tau(s)|\mathrm{d} s}\leq \mathrm{e}^{\sqrt{2}\int_{t_k}^{t_{k+1}} \frac{\nu\alpha}{T} \mathrm{d} s}$ with $\alpha$ from \cref{ass:schedule}. Then, integrating~\ref{eq:moment_bound_omega} over $(t_k,t_{k+1})$ yields
    \begin{equation}
        \begin{aligned}
            \mathbb{E}[\omega_{\tau(t_{k+1})}(X_{t_{k+1}},V_{t_{k+1}})]
            \lesssim& \mathrm{e}^{-\frac{m_\tau}{8\sqrt{L_\tau}}h_k}\mathrm{e}^{\sqrt{2}\int_{t_k}^{t_{k+1}} \frac{\nu\alpha}{T} \mathrm{d} s}\mathbb{E}[\omega_{\tau(t_k)}(X_{t_{k+1}},V_{t_{k+1}})] 
            \\

Let $u,v,w\in\mathbb{R}^d$ normalized vectors. One can easily verify that 
    \begin{equation}\label{eq:third_cumulant}
        \nabla^3 \Psi_\tau(z)[u,v,w] = \frac{1}{\tau^3}\int f_uf_vf_w \mathrm{d} \mu_z(x)
    \end{equation}
    where we define the tilted distribution
    \begin{equation}
        \mu_z(\mathrm{d} x) \propto \mathrm{e}^{-\Psi_0(\frac{x}{\sqrt{1-\tau}}) - \frac{|x-z|^2}{2\tau}}
    \end{equation}
    and $f_h(x) = (x-\mathbb{E}_{\mu_z}[X])\cdot h$ for $h\in \{u,v,w\}$.
    Denote moreover $U_z(x) = \Psi_0(\frac{x}{\sqrt{1-\tau}}) + \frac{|x-z|^2}{2\tau}$ and the differential operator $\mathcal{L} \coloneqq -\nabla U_z\cdot + \Delta$. By strong log-concavity of $\mu_z$ and Lax-Milgram, there exists a unique, smooth, zero-mean $\phi_u$ such that $-\mathcal{L}\phi_u = f_u$ and by a standard bootstrapping argument $\phi_u\in \mathcal{C}^\infty$.
    We will in the following make frequent use of the fact that for any $f,g$, $\int -\mathcal{L}(f) g\mathrm{d} \mu_z = \int \nabla f\cdot \nabla g\mathrm{d} \mu_z$.
    We find
    \begin{equation}
        \begin{aligned}
            \int f_uf_vf_w \mathrm{d} \mu_z(x) 
            = \int (-\mathcal{L}\phi_u)f_vf_w \mathrm{d} \mu_z(x)
            =& \int \nabla \phi_u\cdot \nabla(f_vf_w) \mathrm{d} \mu_z(x)\\
            =& \int \nabla \phi_u\cdot (w f_v + v f_w) \mathrm{d} \mu_z(x).
        \end{aligned}
    \end{equation}
    Repeating this trick with $-\mathcal{L}\phi_v = f_v$, $-\mathcal{L}\phi_w = f_w$ we obtain
    \begin{equation}\label{eq:cum_est1}
        \begin{aligned}
            \int f_uf_vf_w \mathrm{d} \mu_z(x) 
                                                    
                                                                      
                                                                                    
            =& \int \nabla \phi_v\cdot \nabla^2 \phi_u w + \nabla \phi_w\cdot \nabla^2 \phi_u v\, \mathrm{d} \mu_z(x).
        \end{aligned}
    \end{equation}
    We proceed by bounding the $L^2(\mu_z)$-norms of $\nabla \phi_h$, $h\in\{v,w\}$, and $\nabla^2 \phi_u$. 
    We begin with $\nabla \phi_h$. 
    By definition of $\phi_h$ we have for $\psi_h\coloneqq \nabla\phi_h$
    \begin{equation}\label{eq:psi_pde}
        -\mathcal{L}(\psi_h) = h - \nabla^2 U_z\psi_h
    \end{equation}
    where we use the same notation $\mathcal{L}$ for the corresponding differential operator now acting on vector-valued functions.
    By~\ref{eq:psi_pde} and $(\frac{m_0}{1-\tau}+\frac{1}{\tau})$-strong convexity of $U_z$
    \begin{equation}
        \begin{aligned}
            0\leq \int |\nabla\psi|^2\mathrm{d} \mu_z
            = \int -\mathcal{L}(\psi_h)\psi_h\mathrm{d} \mu_z
            =& \int h\cdot \psi_h - \psi_h^\top \nabla^2 U\psi_h\mathrm{d} \mu_z\\
            \leq& \int h\cdot \psi_h - (\frac{m_0}{1-\tau}+\frac{1}{\tau})|\psi_h|^2\mathrm{d} \mu_z.
        \end{aligned}
    \end{equation}
    Since $h = \nabla f_h$ and $\psi_h = \nabla \phi_h$ we, therefore, have
    \begin{equation}
        \begin{aligned}
            (\frac{m_0}{1-\tau}+\frac{1}{\tau})\int|\psi_h|^2\mathrm{d} \mu_z\leq \int h\cdot \psi_h\mathrm{d} \mu_z = \int \nabla f_h \cdot \nabla \phi_h \mathrm{d} \mu_z 
            =& \int f_h(-\mathcal{L} \phi_h)\mathrm{d} \mu_z \\
            =& \int |f_h|^2\mathrm{d} \mu_z\\
            \overset{(*)}{\leq}& \frac{1}{\frac{m_0}{1-\tau}+\frac{1}{\tau}}\int |\nabla f_h|^2\mathrm{d} \mu_z\\
            \overset{(**)}{=}& \frac{\tau(1-\tau)}{\tau m_0+(1-\tau)}
        \end{aligned}
    \end{equation}
    where we used the Poincar\'e inequality for strongly log-concave densities in $(*)$ and $|\nabla f_h| = |h|=1$ in $(**)$.
    Next we bound $\nabla^2\phi_h$. Differentiating~\ref{eq:psi_pde} once more yields for $\Phi_h\coloneqq \nabla^2 \phi_h$ the equation
    \begin{equation}\label{eq:LPhi}
        -\mathcal{L}(\Phi_h)= -\nabla^3 U_z [\nabla \phi_h,\cdot,\cdot] -\nabla^2U_z\Phi_h - \Phi_h\nabla^2 U_z.
    \end{equation}
    where $\mathcal{L}$ denotes the appropriate action on matrix-valued functions and $\nabla^3 U_z$ is a 3d tensor. We have again by integration by parts
    \begin{equation}
        \int -\mathcal{L}(\Phi)\cdot \Phi\mathrm{d} \mu_z = \int \|\nabla \Phi\|_F^2\mathrm{d} \mu_z\geq 0.
    \end{equation}
    Combining with~\ref{eq:LPhi}, $\nabla^2 U_z \succeq \frac{m_0}{1-\tau}+\frac{1}{\tau}$ and 
    \begin{equation}
        \begin{aligned}
            2(\frac{m_0}{1-\tau}+\frac{1}{\tau}) \int \|\Phi\|_F^2\mathrm{d} \mu 
            \leq& \int (\nabla^2U_z\Phi + \Phi\nabla^2 U_z)\cdot\Phi\mathrm{d} \mu_z \\
            \leq& \int -\nabla^3 U_z [\nabla \phi,\cdot,\cdot]\cdot \Phi\mathrm{d} \mu_z\\
            \leq& \frac{M}{(1-\tau)^{3/2}} \bigg(\int |\nabla\phi|^2\mathrm{d}\mu_z\bigg)^{1/2}\bigg(\int \|\Phi\|_F^2\mathrm{d}\mu_z\bigg)^{1/2}
        \end{aligned}
    \end{equation}
    Thus,
    \begin{equation}
        \begin{aligned}
            \int \|\Phi\|_F^2\mathrm{d} \mu 
            \leq& \big(\frac{\tau(1-\tau)M}{2(1-\tau)^{3/2}(m_0\tau + (1-\tau))}\big)^2 \int |\nabla\phi|^2\mathrm{d}\mu_z\\
            \leq& \big(\frac{\tau(1-\tau)M}{2(1-\tau)^{3/2}(m_0\tau + (1-\tau))}\big)^2\big(\frac{\tau(1-\tau)}{m_0\tau + (1-\tau)}\big)^2
        \end{aligned}
    \end{equation}
    Inserting into~\ref{eq:cum_est1} then leads to
    \begin{equation}
        \begin{aligned}
            \big|\int f_uf_vf_w \mathrm{d} \mu_z(x) \big|
            \leq& \int |\nabla \phi_v| |\nabla^2 \phi_u| + |\nabla \phi_w| |\nabla^2 \phi_u|\, \mathrm{d} \mu_z(x)\\
            \leq& \int |\nabla \phi_v| |\nabla^2 \phi_u| + |\nabla \phi_w| |\nabla^2 \phi_u|\, \mathrm{d} \mu_z(x)\\
            \leq& \frac{M \tau^3(1-\tau)^3}{(1-\tau)^{3/2}(m_0\tau + (1-\tau))^3}\\
            \leq& \frac{M \tau^3(1-\tau)^{3/2}}{(m_0\tau + (1-\tau))^3}.
        \end{aligned}
    \end{equation}
    Inserting into~\ref{eq:third_cumulant} then yields
    \begin{equation}
        |\nabla^3 \Psi_\tau(z)[u,v,w]| \leq \frac{M(1-\tau)^{3/2}}{(m_0\tau + (1-\tau))^3}.
    \end{equation}
