<!-- prop-1: body_md -->
\label{defi:productmeasure}
  Let $n \in \mathbb{N}$ and let $p_i \in \mathcal{P} X_i$ be probability measures on measurable spaces $(X_i,\Sigma_{X_i})$ for $i \in [n]$.
  Then there is a unique probability measure $\bigotimes_{i=1}^n p_i$ on $\prod_{i=1}^n X_i$ such that
  \begin{equation}
    \label{eq:productmeasure}
    (p_1 \otimes \cdots \otimes p_n)(S_1 \times \cdots \times S_n)
    = p_1(S_1) \cdots p_n(S_n)
  \end{equation}
  for all $S_i \in \Sigma_{X_i}$.
<!-- end prop-1 -->
<!-- defi-1: body_md -->
For measurable spaces $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$, a \emph{Markov kernel} from $X$ to $Y$ is a map
	\begin{align*}
		\kappa \: : \: \Sigma_Y \times X & \longrightarrow [0,1] \\
		(S, x) & \longmapsto \kappa(S|x)
	\end{align*}
	such that:
	\begin{enumerate}[label=(\roman*)]
		\item $\kappa(-|x)$ is a probability measure on $(Y, \Sigma_Y)$ for all $x \in X$.
		\item $\kappa(S|\cdot)$ is measurable for all $S \in \Sigma_Y$.
	\end{enumerate}
<!-- end defi-1 -->
<!-- defi-2: body_md -->
Let $\rho_X : X \to B$ be a measurable map between arbitrary measurable spaces.
	Then a \emph{regular conditional probability} of a probability measure $p \in \mathcal{P} X$ with respect to $\rho_X$ is a Markov kernel $p(-|\cdot) : \Sigma_X \times B \to [0,1]$ such that
  \begin{equation}
		\label{eq:disintegration}
		p(S \cap \rho_X^{-1}(T)) = \int_{b \in T} p(S|b) \  \omega(\mathrm{d} b)
		\qquad
		\forall S \in \Sigma_X, T \in \Sigma_B,
	\end{equation}
	where $\omega \coloneq \rho_{X,\ast} p$.
<!-- end defi-2 -->
<!-- prop-2: body_md -->
\label{prop:regularconditionalprobability}
	If $X$ and $B$ are standard Borel, then such a regular conditional probability exists for every $p \in \mathcal{P} X$,
	and it is unique up to $\omega$-a.e.~equality.
<!-- end prop-2 -->
<!-- defi-3: body_md -->
\label{defi:relativeproductmeasure}
  Let $X_1,\dots,X_n$ and $B$ be standard Borel spaces with measurable maps $\rho_{X_i} : X_i \to B$, and fix $\omega \in \mathcal{P} B$.
	Suppose that measures $p_i \in \mathcal{P} X_i$ satisfy $\rho_{X_i,\ast} p_i = \omega$ for every $i$.
  Then the \emph{relative product} of $p_1,\dots,p_n$ over $\omega$ is the probability measure on $\prod_{i=1}^n X_i$ defined by
  \begin{equation}
		\label{eq:relativeproductmeasure}
    (p_1 \otimes_\omega \cdots \otimes_\omega p_n)(S)
    \coloneqq \int_{b \in B} \bigl(p_1(-|b) \otimes \cdots \otimes p_n(-|b)\bigr)(S)\,\omega(\mathrm{d} b)
    \qquad \forall S \in \Sigma_{X_1 \times \cdots \times X_n}.
  \end{equation}
<!-- end defi-3 -->
<!-- lem-1: body_md -->
\label{lem:relativeproductsupport}
  The relative product measure $p_1 \otimes_\omega \cdots \otimes_\omega p_n$ is supported on the fiber product $X_1 \times_B \cdots \times_B X_n$,
  considered as a measurable subset of $\prod_{i=1}^n X_i$.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
Since $B$ is standard Borel, the diagonal $\Delta_B \coloneqq \{(b,\dots,b):b\in B\}$ in $B^n$ is measurable. Hence
  \[
    X_1 \times_B \cdots \times_B X_n
    = (\rho_{X_1} \times \cdots \times \rho_{X_n})^{-1}(\Delta_B)
  \]
  is a measurable subset of $\prod_{i=1}^n X_i$.
  Since there are only finitely many factors, the fiber concentration property~\eqref{eq:fiberconcentration} gives
  \[
    p_i(\rho_{X_i}^{-1}(b)|b)=1 \qquad \forall i \in [n]
  \]
  away from an $\omega$-null set.
  For each such $b$, the product measure $\bigotimes_{i=1}^n p_i(-|b)$ is concentrated on
  \[
    \rho_{X_1}^{-1}(b) \times \cdots \times \rho_{X_n}^{-1}(b) \:\subseteq\: X_1 \times_B \cdots \times_B X_n.
  \]
  Thus, with $S=X_1 \times_B \cdots \times_B X_n$ in~\eqref{eq:relativeproductmeasure}, the integrand is a.e.~equal to $1$,
  and hence the integral is $1$ as well.
<!-- end proof-1 -->
<!-- lem-2: body_md -->
\label{lem:pushforwarddisintegration}
  Let $f : X \to Y$ be a morphism in $\mathsf{BorelMeas}/B$.
	Then for every $p \in \mathcal{P}^\omega X$, the Markov kernel
	\begin{align*}
		B & \longrightarrow \mathcal{P} Y \\
		b & \longmapsto f_\ast p(-|b)
	\end{align*}
	is a regular conditional probability of $f_\ast p$ with respect to $\rho_Y$.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
The displayed map is a Markov kernel because it is the composite of the Markov kernel
	$p(-|\cdot) : B \to \mathcal{P} X$ with the measurable pushforward map $f_\ast : \mathcal{P} X \to \mathcal{P} Y$.
	Since $f$ is a morphism over $B$, we have
	\[
		\rho_{Y,\ast}(f_\ast p)
		= (\rho_Y \circ f)_\ast p
		= \rho_{X,\ast} p
		= \omega.
	\]
	Consequently, for $S \in \Sigma_Y$ and $T \in \Sigma_B$,
	\begin{align*}
		(f_\ast p)(S \cap \rho_Y^{-1}(T))
		&= p\bigl(f^{-1}(S) \cap \rho_X^{-1}(T)\bigr) \\
		&= \int_{b \in T} p(f^{-1}(S)|b) \, \omega(\mathrm{d} b) \\
		&= \int_{b \in T} \bigl(f_\ast p(-|b)\bigr)(S) \, \omega(\mathrm{d} b),
	\end{align*}
	which is precisely the disintegration identity~\eqref{eq:disintegration}.
<!-- end proof-2 -->
<!-- lem-3: body_md -->
\label{lem:relativeproductnaturality}
	For every $n \in \mathbb{N}$, the formation of relative product measures defines maps
	\begin{equation}
		\label{eq:relativeproductnaturality}
    \nabla^\omega \: : \: \mathcal{P}^\omega X_1 \times \cdots \times \mathcal{P}^\omega X_n \longrightarrow \mathcal{P}^\omega(X_1 \times_B \cdots \times_B X_n)
	\end{equation}
	that are natural in $X_1,\dots,X_n \in \mathsf{BorelMeas}/B$.
<!-- end lem-3 -->
<!-- proof-3: body_md -->
Let $p_i \in \mathcal{P}^\omega X_i$ for every $i$.
  By \cref{lem:relativeproductsupport}, we can regard
  $r \coloneqq p_1 \otimes_\omega \cdots \otimes_\omega p_n$ as a probability measure on $X_1 \times_B \cdots \times_B X_n$.
  For every coordinate projection $\pi_i$ and $S \in \Sigma_{X_i}$,
  \[
		(\pi_{i,\ast} r)(S)
    = \int_{b \in B} p_i(S|b)\,\omega(\mathrm{d} b)
    = p_i(S).
  \]
  Here we have used $p_j(X_j|b)=1$ for all $j \ne i$. Thus $r$ has the prescribed marginals, and
  \[
		(\rho_{X_i} \circ \pi_i)_\ast r = \rho_{X_i,\ast} p_i = \omega.
	\]
  Hence $r \in \mathcal{P}^\omega(X_1 \times_B \cdots \times_B X_n)$, making $\nabla^\omega$ well-defined.

  For naturality, let $f_i : X_i \to Y_i$ be morphisms over $B$, so that the diagrams
  \[
    \begin{tikzcd}[column sep=large]
      X_i \arrow[rr, "f_i"] \arrow[dr, "\rho_{X_i}"'] & & Y_i \arrow[dl, "\rho_{Y_i}"] \\
      & B &
    \end{tikzcd}
  \]
	commute.
	Write $f \coloneqq \prod_{i=1}^n f_i$ and $r' \coloneqq f_{1,\ast} p_1 \otimes_\omega \cdots \otimes_\omega f_{n,\ast} p_n$.
  By \cref{lem:pushforwarddisintegration}, regular conditional probabilities of $f_{i,\ast} p_i$ are given by $b \mapsto f_{i,\ast} p_i(-|b)$.
  Therefore, for every measurable $S \subseteq \prod_{i=1}^n Y_i$, naturality of ordinary product measures gives
  \begin{align*}
    r'(S)
		&= \int_{b \in B} \bigl(f_{1,\ast} p_1(-|b) \otimes \cdots \otimes f_{n,\ast} p_n(-|b)\bigr)(S)\,\omega(\mathrm{d} b) \\
    &= \int_{b \in B} \bigl(p_1(-|b) \otimes \cdots \otimes p_n(-|b)\bigr)(f^{-1}(S))\,\omega(\mathrm{d} b) \\
		&= r(f^{-1}(S)) \\[2pt]
    &= (f_\ast r)(S).
  \end{align*}
  Since each $f_i$ is a morphism over $B$, the map $F$ restricts to
  \[
		f_1 \times_B \cdots \times_B f_n \: : \: X_1 \times_B \cdots \times_B X_n \longrightarrow X_1' \times_B \cdots \times_B X_n'.
	\]
  The support property of $r$ and $r'$ therefore gives
  \[
    (f_1 \times_B \cdots \times_B f_n)_\ast (p_1 \otimes_\omega \cdots \otimes_\omega p_n)
		= f_{1,\ast} p_1 \otimes_\omega \cdots \otimes_\omega f_{n,\ast} p_n,
  \]
  which is the desired naturality.

<!-- lem-4: body_md -->
\label{rem:measurableproducts}
	With these $\sigma$-algebras, the relative product maps~\eqref{eq:relativeproductnaturality} are measurable.
<!-- end lem-4 -->
<!-- proof-4: body_md -->
We need to show that the map
	\begin{align*}
		\mathcal{P}^\omega X_1 \times \cdots \times \mathcal{P}^\omega X_n & \longrightarrow [0,1] \\
		(p_1,\dots,p_n) & \longmapsto (p_1 \otimes_\omega \cdots \otimes_\omega p_n)(S)
	\end{align*}
	is measurable for every measurable $S \subseteq X_1 \times \cdots \times X_n$.
	By a standard application of the $\pi$-$\lambda$ theorem,
	it is enough to check this for rectangles $S = S_1 \times \cdots \times S_n$.

	To see this, recall that if $q : A \to \mathcal{P}^\omega X$ is measurable with $A$ standard Borel too,
	then there is a jointly measurable map $A \times B \to \mathcal{P} X$ that is a version of the regular conditional probability for every $a \in A$.\footnote{This is arguably understood most cleanly in terms of conditionals in the Markov category $\mathsf{BorelMeas}$, see~\cite[Proposition~11.7]{Fritz2020}. In measure-theoretic terms, the relevant result is~\cite[Theorem~1.25]{Kallenberg2017}.}
	Taking $A = \mathcal{P}^\omega X$ itself and $q$ the identity map shows that the regular conditional probabilities $p(-|b)$ can be chosen jointly measurably in $b \in B$ and $p \in \mathcal{P}^\omega X$.
	Hence the integrand in the construction of the relative product measure
	\[
		(p_1 \otimes_\omega \cdots \otimes_\omega p_n)(S_1 \times \cdots \times S_n) = \int_{b \in B} p_1(S_1|b) \cdots p_n(S_n|b)\,\omega(\mathrm{d} b)
	\]
	depends measurably on $(p_1,\dots,p_n)$,
	being the product of the corresponding measurable functions $p_i \mapsto p_i(S_i|\cdot)$.
	But then by measurability of integration~\cite[Lemma~1.15(i)]{Kallenberg2017},
	also the integral depends measurably on $(p_1,\dots,p_n)$,
	as was to be shown.

\label{theo:onenaturaltransformation}
  \begin{enumerate}
  \item 
    With $\mathcal{P} : \mathsf{Meas} \to \mathsf{Set}$ the Giry functor and any $n \in \mathbb{N}$, the formation of product measures is the only family of maps
    \[
      \mathcal{P} X_1
      \times \dots \times \mathcal{P} X_n \longrightarrow \mathcal{P}(X_1 \times \dots \times X_n)
    \]
    which is natural in $X_1,\dots,X_n \in \mathsf{Meas}$.
  \item 
    The analogous statement for the distribution functor $\mathcal{D} : \mathsf{Set} \to \mathsf{Set}$ (resp.~$\mathcal{D}_{\mathrm{fin}} : \mathsf{FinSet} \to \mathsf{Set}$) also holds.
  \end{enumerate}

<!-- theo-2: body_md -->
\label{theo:onenaturaltransformation_relative}
  \begin{enumerate}
  \item For all $n \in \mathbb{N}$, any standard Borel space $B$, and $\omega \in \mathcal{P} B$, the formation of relative product measures with respect to $\omega$ is the only family of maps
    \[
      \mathcal{P}^\omega X_1 \times  \cdots \times \mathcal{P}^\omega X_n \longrightarrow \mathcal{P}^\omega(X_1 \times_B \cdots \times_B X_n)
    \]
    which is natural in $X_1,\dots,X_n \in \mathsf{BorelMeas}/B$.
  \item With $B$ a (finite) set, the analogous statement for $\mathcal{D}^\omega : \mathsf{Set}/B \to \mathsf{Set}$ (resp.~$\mathcal{D}_{\mathrm{fin}}^\omega : \mathsf{FinSet}/B \to \mathsf{Set}$) also holds.
  \end{enumerate}

<!-- rem-2: body_md -->
\begin{enumerate}
  \item 
    The proofs of both results will first consider the case of finite sets and then extend from there.
    This is designed such that the proofs go through for all of $\mathcal{P}$, $\mathcal{D}$ and $\mathcal{D}_{\mathrm{fin}}$ simultaneously.
    If one is interested in $\mathcal{P}$ only,
    then simpler proofs are possible using the Lebesgue measure instead together with the fact that every other probability measure on a standard Borel space is a pushforward of it.
    This is essentially the approach taken in the existing copula theory result~\cite[Theorem~10]{Durante2009}.
  \item 
    In \cref{theo:onenaturaltransformation,theo:onenaturaltransformation_relative}, we have considered the functors $\mathcal{P}$ and $\mathcal{P}^\omega$ to take values in $\mathsf{Set}$.
    With the corresponding $\mathcal{P} : \mathsf{Meas} \to \mathsf{Meas}$ and $\mathcal{P}^\omega : \mathsf{BorelMeas}/B \to \mathsf{BorelMeas}$ versions of these functors,
    the same statements hold as corollaries:
    the product maps have measurable components by \cref{rem:measurableproducts},
		and uniqueness follows from the corresponding set-valued statements.
  \end{enumerate}

<!-- cor-1: body_md -->
\label{cor:relativeproductlaxmonoidal}
  For any standard Borel space $B$ and $\omega \in \mathcal{P} B$, the functor
  \[
    \mathcal{P}^\omega\colon (\mathsf{BorelMeas}/B,\times_B,B) \longrightarrow (\mathsf{Set},\times,\{\ast\})
  \]
  carries a unique lax symmetric monoidal structure. Its structure maps are the relative product maps
  \begin{align*}
    \nabla^\omega \: \colon \: \mathcal{P}^\omega X \times \mathcal{P}^\omega Y &\longrightarrow \mathcal{P}^\omega(X \times_B Y) \\
                                                             (p,q) &\longmapsto p \otimes_\omega q
  \end{align*}
  and the trivial map
  \begin{align*}
    \iota \: \colon \: \{\ast\} & \longrightarrow \mathcal{P}^\omega B = \{\omega\} \\
		\ast & \longmapsto \omega.
  \end{align*}
<!-- end cor-1 -->
<!-- proof-5: body_md -->
The maps $\nabla^\omega$ are natural by \cref{lem:relativeproductnaturality},
	and it is clear from \cref{theo:onenaturaltransformation_relative} that they are the only natural maps of the correct type.
	Therefore at most one lax monoidal structure on $\mathcal{P}^\omega$ can exist.

	To see that these structure maps indeed do the job, we first check associativity.
	With
  \[
    \alpha \: \colon \: (X \times_B Y) \times_B Z \longrightarrow X \times_B (Y \times_B Z)
  \]
	being the associator in $\mathsf{BorelMeas}/B$,
	the two assignments 
  \begin{align*}
    (p,q,r) &\longmapsto \alpha_*((p \otimes_\omega q) \otimes_\omega r) \\
    (p,q,r) &\longmapsto p \otimes_\omega (q \otimes_\omega r)
  \end{align*}
  define maps
  \[
    \mathcal{P}^\omega X \times \mathcal{P}^\omega Y \times \mathcal{P}^\omega Z \longrightarrow \mathcal{P}^\omega(X \times_B (Y \times_B Z))
  \]
  which are natural in $X,Y,Z \in \mathsf{BorelMeas}/B$.
	By the $n=3$ case of \cref{theo:onenaturaltransformation_relative}, these assignments agree. 
	This is the associativity condition for a lax monoidal functor.

	The unitality and symmetry conditions follow similarly from the $n=1$ and $n=2$ cases of \cref{theo:onenaturaltransformation_relative}, respectively.

  Let $X$ be a finite set and $G$ a group acting transitively on $X$. Then, there is exactly one $G$-invariant probability measure on $X$, namely the uniform measure $u_X$.

\label{lem:rationalmeasure}
  Let $X$ be a finite set and $p \in \mathcal{D} X$ rational. Then there is $N \in \mathbb{N}$ and $f:[N] \rightarrow X$ such that $p = f_\ast u_{[N]}$.
<!-- end lem-6 -->
<!-- proof-7: body_md -->
Let $X = \{x_1, \ldots, x_{k}\}$ and choose a common denominator $N$ for $p(x_1), \ldots, p(x_{k})$.
	This means that $p(x_j) = \frac{q_j}{N}$ for suitable $q_j \in \mathbb{N}$ with $q_1 + \ldots + q_{k} = N$.
  Define $f : [N]  \rightarrow X$ by $f(a) \coloneqq x_{\ell(a)}$ for every $a \in [N]$,
  where $\ell(a) \in [k]$ is the unique index with $\sum_{j=1}^{\ell(a)-1} q_j < a \leq \sum_{j=1}^{\ell(a)} q_j$.
  Then we have for every $\ell \in [k]$,
  \[
    f_\ast u_{[N]}(x_\ell) = u_{[N]}(f^{-1}(x_\ell)) 
		= u_{[N]} \left(\left\{ \sum_{j=1}^{\ell-1} q_j + 1,\ldots, \sum_{j=1}^\ell q_j \right\}\right)
		= \frac{q_\ell}{N} = p(x_\ell).
  \]
  Thus, $p = f_\ast u_{[N]}$.

<!-- prop-3: body_md -->
\label{prop:rationalcase}
  Let $\mathcal{D}_{\mathrm{fin}}$ be the distribution functor on $\mathsf{FinSet}$. Let $n \in \mathbb{N}$ and let
  \begin{equation}
    \label{eq:alpha}
    \alpha \: \colon \: \mathcal{D}_{\mathrm{fin}} X_1 \times \cdots \times \mathcal{D}_{\mathrm{fin}} X_n \Longrightarrow \mathcal{D}_{\mathrm{fin}}(X_1 \times \cdots \times X_n)
  \end{equation}
  be natural in $X_1, \dots, X_n$. Then,
  \[
    \alpha(p_1,\dots,p_n) = p_1 \otimes \cdots \otimes p_n
  \]
  for all rational $p_i \in \mathcal{D}_{\mathrm{fin}} X_i$.
<!-- end prop-3 -->
<!-- proof-8: body_md -->
Without loss of generality we assume that $n \geq 1$ and all $X_i$ are nonempty. For all permutations $\sigma_i \in \operatorname{Sym}(X_i)$, naturality of $\alpha$ gives
  \begin{align*}
    (\sigma_1 \times \cdots \times \sigma_n)_\ast\alpha(u_{X_1},\dots,u_{X_n}) &= \alpha(\sigma_{1,\ast} u_{X_1},\dots,\sigma_{n,\ast} u_{X_n}) \\[2pt]
                                                                          &= \alpha(u_{X_1},\dots,u_{X_n}),
  \end{align*}
  since uniform measures are invariant under permutations. Thus, $\alpha(u_{X_1},\dots,u_{X_n})$ is invariant under $\prod_{i=1}^n\operatorname{Sym}(X_i)$. This group acts transitively on $\prod_{i=1}^n X_i$. Hence, by \cref{lem:uniquepermutationinvariantmeasure},
  \[
    \alpha(u_{X_1},\dots,u_{X_n}) = u_{X_1 \times \cdots \times X_n}.
  \]

  Now let $p_i \in \mathcal{D}_{\mathrm{fin}} X_i$ be rational for every $i$. By \cref{lem:rationalmeasure}, there are  $N_1,\dots,N_n \in \mathbb{N}$ and maps
  \[
    f_i \: \colon \: [N_i] \longrightarrow X_i
  \]
	such that $p_i = f_{i,\ast} u_{[N_i]}$. Naturality of $\alpha$ and the uniform case give
  \begin{align*}
    \alpha(p_1,\dots,p_n) &= (f_1 \times \cdots \times f_n)_\ast\alpha(u_{[N_1]},\dots,u_{[N_n]}) \\[2pt]
                          &= (f_1 \times \cdots \times f_n)_\ast u_{[N_1] \times \cdots \times [N_n]}.
  \end{align*}
  By naturality of the product measure construction $\nabla$, hence,
  \begin{align*}
    \alpha(p_1,\dots,p_n) &= (f_1 \times \cdots \times f_n)_\ast (u_{[N_1]} \otimes \cdots \otimes u_{[N_n]}) \\[2pt]
													& = f_{1,\ast} u_{[N_1]} \otimes \cdots \otimes f_{n,\ast} u_{[N_n]} \\[2pt]
													& = p_1 \otimes \cdots \otimes p_n. \qedhere
  \end{align*}

\label{prop:order}
  Define $\theta^-,\theta^+: [3] \rightarrow [2]$ by
  \begin{align*}
	  \theta^-(1) = \theta^-(2) = 1, &&  \theta^+(1) = 1,  \\
	  \theta^-(3) = 2, && \theta^+(2) = \theta^+(3) = 2.
  \end{align*}
  Then for all $p, q \in \mathcal{D}[2]$, we have $p(1) \leq q(1)$ if and only if there is $\eta \in \mathcal{D}[3]$ such that
  \begin{equation}\label{eq:theta}
    \theta^+_\ast\eta = p \qquad \text{and} \qquad \theta^-_\ast\eta = q.
  \end{equation}
  In this case,
	there is only one such $\eta$, namely
  \begin{equation}
	  \label{eq:theta2}
	  \eta(1) = p(1), \qquad \eta(2) = q(1)-p(1), \qquad \eta(3) = q(2).
  \end{equation}

  It is nonnegative if and only if $p(1) \leq q(1)$.

\label{lem:bernoulliirrational}
  For all $p_1,\dots,p_n \in \mathcal{D}[2]$, every natural transformation $\alpha$ as in~\eqref{eq:alpha} satisfies
  \[
    \alpha(p_1,\dots,p_n) = p_1 \otimes \cdots \otimes p_n.
  \]
<!-- end lem-8 -->
<!-- proof-10: body_md -->
Define
  \begin{align*}
    \phi\colon \mathcal{D}[2] \times \cdots \times \mathcal{D}[2] &\longrightarrow [0,1] \\
                                       (p_1,\dots,p_n) &\longmapsto \alpha(p_1,\dots,p_n)(\{1\}^n).
  \end{align*}
  We first show that $\phi$ is monotonically nondecreasing in the product order. Let $p_i \leq p_i'$ for every $i$. As per \cref{prop:order}, these inequalities are witnessed by $\eta_i \in \mathcal{D} [3]$ with
  \[
    \theta^+_\ast\eta_i = p_i \qquad \text{and} \qquad \theta^-_\ast\eta_i = p_i'.
  \]
  Naturality of $\alpha$ gives
  \begin{align*}
    \phi(p_1',\dots,p_n')-\phi(p_1,\dots,p_n) &= \alpha(\theta^-_\ast\eta_1,\dots,\theta^-_\ast\eta_n)(\{1\}^n) - \alpha(\theta^+_\ast\eta_1,\dots,\theta^+_\ast\eta_n)(\{1\}^n)\\
                                              &= \alpha(\eta_1,\dots,\eta_n)(\{1,2\}^n) - \alpha(\eta_1,\dots,\eta_n)(\{1\}^n) \\
                                              &= \alpha(\eta_1,\dots,\eta_n)(\{1,2\}^n \setminus \{1\}^n) \geq 0.
  \end{align*}
  Hence, $\phi$ is monotonically nondecreasing.

  Now let $p_1,\dots,p_n \in \mathcal{D}[2]$ be arbitrary. By density of $\mathbb{Q}$ in $\mathbb{R}$, for every $i \leq n$, we can find rational $(p_{i,m}^-)_{m \in \mathbb{N}}$ and $(p_{i,m}^+)_{m \in \mathbb{N}}$ in $\mathcal{D}[2]$ with $p_{i,m}^-\nearrow p_i$ and $p_{i,m}^+ \searrow p_i$ as $m \to \infty$.
  By monotonicity and \cref{prop:rationalcase}, we obtain
  \[
    p_{1,m}^-(1) \cdots p_{n,m}^-(1) \leq \phi(p_1,\dots,p_n) \leq p_{1,m}^+(1) \cdots p_{n,m}^+(1).
  \]
  Taking $m \rightarrow \infty$ shows
  \[
    \phi(p_1,\dots,p_n) = p_1(1) \cdots p_n(1).
  \]

  Finally, let $(x_1,\dots,x_n) \in [2]^n$. For every $i$, choose $\sigma_i \in \operatorname{Sym}([2])$ such that $\sigma_i(x_i)=1$. By naturality of $\alpha$ and the equality just proved,
\begin{align*}
  \alpha(p_1,\dots,p_n)(x_1,\dots,x_n) &= \phi(\sigma_{1,\ast} p_1,\dots,\sigma_{n,\ast} p_n) \\
                                       &= (\sigma_{1,\ast} p_1)(1) \cdots (\sigma_{n,\ast} p_n)(1) \\
                                       &= p_1(x_1) \cdots p_n(x_n).
\end{align*}
Thus, $\alpha(p_1,\dots,p_n)$ and $\bigotimes_{i=1}^n p_i$ agree on every atom of $[2]^n$, which proves the claim.

\label{prop:relative_rationalcase}
  Let $X_1,\dots,X_n$ be nonempty finite sets and $p_i\colon  B \to \mathcal{D} X_i$ Markov kernels.
	Suppose that there are $N_1, \dots, N_n \in \mathbb{N}$ such that
  \[
    N_ip_i(x|b) \in \mathbb{N}
		\qquad
		\forall x \in X_i, b \in B, i \in [n].
  \]
  Then
  \[
    \alpha(\overline{p_1},\dots,\overline{p_n}) = \overline{p_1} \otimes_\omega \cdots \otimes_\omega \overline{p_n}.
  \]
<!-- end prop-4 -->
<!-- proof-12: body_md -->
We first consider the uniform measures $u_{X_1}, \dots, u_{X_n}$. Naturality under $\sigma_i \times \operatorname{id}_B$, for permutations $\sigma_i \in \operatorname{Sym}(X_i)$, implies that
  \[
    r \coloneq \alpha(\overline{u_{X_1}}, \dots, \overline{u_{X_n}})
  \]
  is invariant under $(\prod_{i=1}^n \sigma_i) \times \operatorname{id}_B$. For every $S \in \Sigma_B$, transitivity of the action on $\prod_{i=1}^n X_i$ therefore makes $r(\{(x_1,\dots,x_n)\} \times S)$ independent of $(x_1,\dots,x_n)$. Since $r$ has pushforward $\omega$ on $B$, the only possible value is
  \[
    r(\{(x_1,\dots,x_n)\} \times S) = \frac{\omega(S)}{|X_1| \cdots |X_n|}.
  \]
  Thus $r = u_{\prod_{i=1}^n X_i} \otimes \omega = \overline{u_{X_1}} \otimes_\omega \cdots \otimes_\omega \overline{u_{X_n}}$.

  Now let $p_1,\dots,p_n$ satisfy the stated denominator conditions. As in \cref{lem:rationalmeasure},
	write $X_i = \{x_{i,1},\ldots,x_{i,k_i}\}$ with $k_i = |X_i|$ and define maps over $B$ by
  \begin{align*}
    f_i \: \colon \: [N_i] \times B & \longrightarrow X_i \times B \\
                                         (a,b) & \longmapsto (x_{i,\ell(a,b)},b),
  \end{align*}
  where $\ell(a,b)$ is the unique index with
  \[
    \sum_{j=1}^{\ell(a,b)-1} N_i p_i(x_{i,j}|b) < a \leq \sum_{j=1}^{\ell(a,b)} N_i p_i(x_{i,j}|b).
  \]
  The defining inequalities are measurable in $b$, and therefore each $f_i$ is measurable.
	For each $b$ and $\ell$, exactly $N_i p_i(x_{i,\ell}|b)$ elements map to $(x_{i,\ell},b)$, and hence
  \[
		f_{i,\ast} \overline{u_{[N_i]}} = \overline{p_i}.
  \]
  By naturality, the uniform case, and \cref{lem:relativeproductnaturality}, we have
  \begin{align*}
    \alpha(\overline{p_1},\dots,\overline{p_n}) &= (f_1 \times_B \cdots \times_B f_n)_\ast \alpha(\overline{u_{[N_1]}},\dots,\overline{u_{[N_n]}}) \\[2pt]
                                                &= (f_1 \times_B \cdots \times_B f_n)_\ast (\overline{u_{[N_1]}} \otimes_\omega \cdots \otimes_\omega \overline{u_{[N_n]}}) \\[2pt]
                                                &= \overline{p_1} \otimes_\omega \cdots \otimes_\omega \overline{p_n},
  \end{align*}
	as was to be shown.

\label{lem:relative_bernoulliirrational}
For all Markov kernels $p_1,\dots,p_n\colon B \to \mathcal{D}[2]$, we have
  \[
    \alpha(\overline{p_1},\dots,\overline{p_n}) = \overline{p_1} \otimes_\omega \cdots \otimes_\omega \overline{p_n}.
  \]
<!-- end lem-9 -->
<!-- proof-13: body_md -->
For $S \in \Sigma_B$, define
  \[
    \phi_S(p_1,\dots,p_n) \coloneq \alpha(\overline{p_1},\dots,\overline{p_n})(\{1\}^n \times S).
  \]
  We first show that $\phi_S$ is monotonically nondecreasing in the product order. Let $p_i \leq p_i'$ for every $i$. As above, these inequalities are witnessed by Markov kernels $\eta_i\colon B \to \mathcal{D} [3]$ with
  \[
    \theta_{B,\ast}^+\overline{\eta_i} = \overline{p_i} \qquad \text{and} \qquad \theta_{B,\ast}^-\overline{\eta_i} = \overline{p_i}'.
  \]
  Naturality of $\alpha$ gives
  \begin{align*}
    \phi_S(p_1',\dots,p_n')-\phi_S(p_1,\dots,p_n) &= \alpha(\overline{\eta_1},\dots,\overline{\eta_n})(\{1,2\}^n \times S) - \alpha(\overline{\eta_1},\dots,\overline{\eta_n})(\{1\}^n \times S) \\[2pt]
                                                  &= \alpha(\overline{\eta_1},\dots,\overline{\eta_n})\bigl((\{1,2\}^n \setminus \{1\}^n) \times S\bigr) \geq 0.
  \end{align*}
  Hence $\phi_S$ is monotonically nondecreasing.

  Now let $p_1,\dots,p_n$ be arbitrary.
	For every $i$, define measurable dyadic approximations by
  \[
    p_{i,m}^-(1|b) \coloneq 2^{-m}\lfloor 2^m p_i(1|b)\rfloor, \qquad p_{i,m}^+(1|b) \coloneq 2^{-m}\lceil 2^m p_i(1|b)\rceil,
  \]
  and $p_{i,m}^\pm(2|b) \coloneq 1-p_{i,m}^\pm(1|b)$. These satisfy $p_{i,m}^- \nearrow p_i$ and $p_{i,m}^+ \searrow p_i$  and all have common denominator~$2^m$. By monotonicity and \cref{prop:relative_rationalcase}, we obtain
  \[
		\int_{b \in S} p_{1,m}^-(1|b) \cdots p_{n,m}^-(1|b)\,\omega(\mathrm{d} b) \leq \phi_S(p_1,\dots,p_n) \leq \int_{b \in S} p_{1,m}^+(1|b) \cdots p_{n,m}^+(1|b)\,\omega(\mathrm{d} b).
  \]
  Taking $m \to \infty$ and using dominated convergence gives
  \[
    \phi_S(p_1,\dots,p_n) = \int_{b \in S} p_1(1|b) \cdots p_n(1|b)\,\omega(\mathrm{d} b).
  \]

  Finally, let $(x_1,\dots,x_n) \in [2]^n$. For every $i$, choose $\sigma_i \in \operatorname{Sym}([2])$ such that $\sigma_i(x_i)=1$. By naturality under $\sigma_i \times \operatorname{id}_B$ and the equality just proved,
  \begin{align*}
    \alpha(\overline{p_1},\dots,\overline{p_n})(\{(x_1,\dots,x_n)\} \times S) &= \phi_S(\sigma_{1,\ast} p_1,\dots,\sigma_{n,\ast} p_n) \\[2pt]
                                                                              &= \int_{b \in S} (\sigma_{1,\ast} p_1)(1|b) \cdots (\sigma_{n,\ast} p_n)(1|b)\,\omega(\mathrm{d} b) \\[2pt]
                                                                              &= \int_{b \in S} p_1(x_1|b) \cdots p_n(x_n|b)\,\omega(\mathrm{d} b).
  \end{align*}
  Since evaluation on sets of the form $\{(x_1,\dots,x_n)\} \times S$ distinguishes measures on $[2]^n \times B$, the claim follows.

Given arbitrary $X_1,\dots,X_n \in \mathsf{BorelMeas}/B$, we show that
  \[
    \alpha(p_1,\dots,p_n) = p_1 \otimes_\omega \cdots \otimes_\omega p_n
  \]
  for all $p_i \in \mathcal{P}^\omega X_i$. Choose regular conditional probabilities $p_i(-|b)$ using \cref{prop:regularconditionalprobability}. For $T_i \in \Sigma_{X_i}$, define measurable maps $f_{T_i}\colon X_i \to [2] \times B$ over $B$ by
  \[
    f_{T_i}(x_i) \coloneq (\chi_{T_i}(x_i),\rho_{X_i}(x_i)),
  \]
  where $\chi_{T_i}$ is the indicator function.
	Also consider the Markov kernels $p_{i,T_i}\colon B \to \mathcal{D}[2]$ characterized by
  \[
    p_{i,T_i}(\{1\}|b) \coloneq p_i(T_i|b).
  \]
  Then the disintegration identity~\eqref{eq:disintegration} gives
  \[
    f_{T_i,\ast} p_i = \overline{p_{i,T_i}}.
  \]
  Naturality and \cref{lem:relative_bernoulliirrational} therefore imply
  \begin{align*}
		\alpha(p_1,\dots,p_n)\bigl(T_1 & \times \cdots \times T_n \bigr) \\[2pt]
		&\quad= \bigl((f_{T_1} \times_B \cdots \times_B f_{T_n})_\ast\alpha(p_1,\dots,p_n)\bigr)(\{1\}^n \times B) \\[2pt]
		&\quad= \alpha(\overline{p_{1,T_1}},\dots,\overline{p_{n,T_n}})(\{1\}^n \times B) \\[2pt]
		&\quad= \int_{b \in B} p_1(T_1|b) \cdots p_n(T_n|b) \, \omega(\mathrm{d} b) \\[2pt]
    &\quad= (p_1 \otimes_\omega \cdots \otimes_\omega p_n)\bigl(T_1 \times \cdots \times T_n\bigr).
  \end{align*}
  Since (the restrictions of) measurable rectangles form a generating $\pi$-system for $X_1 \times_B \cdots \times_B X_n$,
	we conclude the required $\alpha(p_1,\dots,p_n) = p_1 \otimes_\omega \cdots \otimes_\omega p_n$ from another application of the $\pi$-$\lambda$ theorem.

  Again, the restriction of the whole proof to $\mathcal{D}^\omega$ and $\mathcal{D}_{\mathrm{fin}}^\omega$ is straightforward.
