<!-- thm-1: body_md -->
\label{thm:estimate}
For all $\varepsilon,\delta\in(0,1)$, there exists an LOCC tomography protocol
that, given
\begin{equation}\begin{aligned}
    N
    =
    O\left(
        \frac{d^6+d^2\log(1/\delta)}{\varepsilon^2}
    \right)
\end{aligned}\end{equation}
copies of an unknown bipartite state
$\rho\in\mathcal{D}(\mathbb{C}^{d}\otimes\mathbb{C}^{d})$, outputs an estimator
$\widehat{\rho}\in\mathcal{D}(\mathbb{C}^{d}\otimes\mathbb{C}^{d})$ such that $\frac12\left\|\widehat{\rho}-\rho\right\|_1\leq\varepsilon$ with probability at least $1-\delta$.
<!-- end thm-1 -->
<!-- lemma-1: body_md -->
\label{lemma:robustness}
For any distillable state $\rho$, and for any \(0<R<E_d(\rho)\), there exist a radius \(r>0\) and a sequence of LOCC maps \(\Lambda_n^{(\rho,R)}\) such that
\begin{equation}
    \lim_{n\to\infty}
    \sup_{\sigma\in B_r(\rho)}
    \frac12
    \left\|\Lambda_n^{(\rho,R)}(\sigma^{\otimes n})-
    \Phi_2^{\otimes\ceil{Rn}}\right\|_1
    =0 \; .
\label{eq:local_robustness}
\end{equation}
<!-- end lemma-1 -->
<!-- thm-2: body_md -->
\label{thm:univ_rate}
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                               
There exists a state-agnostic sequence of LOCC protocols with the following property: for any bipartite quantum state $\rho \in \mathcal{D}(\mathbb C^d \otimes \mathbb C^d)$, and any rate $0<R<E_d(\rho)$, the universal protocol applied to $\rho^{\otimes n}$ asymptotically distills $\ceil{Rn}$ ebits with probability tending to one and with vanishing error in trace distance.
<!-- end thm-2 -->
<!-- thm-3: body_md -->
\label{thm:fament}
For any nonempty family of states $\mathcal{X}\subseteq\mathcal{D}(\mathbb{C}^d\otimes\mathbb{C}^d)$ the universal distillable entanglement is
\begin{equation}\begin{aligned}
    \hat{E}_d(\mathcal{X}) = \min_{\rho\in\bar{\mathcal{X}}}E_d(\rho) \; ,
\end{aligned}\end{equation}
where $\bar{\mathcal{X}}$ is the closure of $\mathcal{X}$.
<!-- end thm-3 -->
