<!-- theorem-1: body_md -->
\label{thm:conv-riem-metric}
For any $i, j \in [m]$, we have
\begin{equation}
\label{eq:quantitative-bound}
\left| g_{ij}^\varepsilon - g_{ij}^0 \right| \leq 2 \alpha^\varepsilon B^2 + \frac{6\alpha^\varepsilon(BD)^2}{\varepsilon}.
\end{equation}
                                                                   
In particular, since $\alpha^\varepsilon \leq N e^{-c/\varepsilon}$,
\begin{equation}

\lim_{\varepsilon \to 0} g_{ij}^\varepsilon(p) = g_{ij}^0(p)
\end{equation}
for each $p \in U$.

<!-- proposition-1: body_md -->
\label{prop: OfDiagMass}
Recall the definitions $\alpha^\varepsilon(p) \vcentcolon = \sum_{k \neq l} \pi_{kl}^\varepsilon(p)$ and $c(p) \vcentcolon = \min_{k \neq l} \|\iota_k(p) - \iota_l(p)\|^2$.
For each $p \in U$, it holds that

\alpha^\varepsilon(p) \le N e^{- c(p)/\varepsilon }.

<!-- proof-1: body_md -->
As the cost is zero on the diagonal and we consider self-transport, 
\[ \pi_{kk}^\varepsilon = e^{2 \mathsf{f}^{\varepsilon}_k /\varepsilon} \]
and thus
\[
\pi_{kl}^\varepsilon = \sqrt{\pi_{kk}^\varepsilon\pi_{ll}^\varepsilon} \ e^{-\|\iota_k - \iota_l\|^2/\varepsilon} \le \sqrt{a_k a_l}\ e^{-\|\iota_k - \iota_l\|^2/\varepsilon} \leq  \sqrt{a_k a_l}\ e^{-c(p)/\varepsilon},
\]
where the second inequality follows from the fact that an entry of the coupling cannot be larger than the corresponding row sum.
The claim follows by applying Cauchy-Schwarz to $\sum_{k \neq l} \sqrt{a_k a_l}$.

<!-- lemma-1: body_md -->
\label{lemma:decomp-general}

Let ${\mathcal X} \subset \mathbb{R}^d$ be the closure of a bounded open set. Suppose that $\{\mu_t\}_{t \in (-\delta, \delta)} \subset \mathcal{P}({\mathcal X})$ is continuously differentiable in $C^1({\mathcal X})^*$ with the weak-* topology, and that its derivative at $t = 0$ is given by $\dot{\mu} = - \mathrm{div}(v \mu)$ for $\mu = \mu_0$ and a vector field $v \colon {\mathcal X} \to \mathbb{R}^d$ in $L^1({\mathcal X}, \mu; \mathbb{R}^d).$ Then the following formulae hold
\begin{equation}
\label{eq:h-mu}
\frac{\varepsilon}{2}H^\varepsilon_\mu[\dot{\mu}](y) =  \int k^\varepsilon_\mu(x, y) (y - T^\varepsilon_{\mu}(x)) \cdot v(x) d\mu(x)
\end{equation}
\begin{equation}
    \label{eq:h-mu-norm}
    \frac{\varepsilon}{2}\|H^\varepsilon_\mu[\dot{\mu}]\|_{\mathcal{H}_\mu^\varepsilon}^2 = \int v(x)^\top v(y)  + \frac{2}{\varepsilon} \left(v(x)^\top \left( y - T^\varepsilon_{\mu}(x)\right)\left(x - T^\varepsilon_{\mu}(y) \right)^\top v(y) \right) d\pi_{\mu, \mu}^\varepsilon(x, y),
\end{equation}
and we have the following decomposition for $g^\varepsilon$:
\begin{equation}
\label{eq:decomp-general}
g^\varepsilon_\mu (\dot{\mu}, \dot{\mu}) = \frac{\varepsilon}{2}\left( \|H^\varepsilon_\mu[\dot{\mu}]\|_{\mathcal{H}_\mu^\varepsilon}^2 +  \langle H_\mu^\varepsilon[\dot{\mu}], K_\mu^\varepsilon (I- (K_\mu^\varepsilon)^2)^{-1} H_\mu^\varepsilon[\dot{\mu}] \rangle_{L^2(\mu)/\mathbb{R}} \right).
\end{equation}

<!-- proof-2: body_md -->
(\ref{eq:h-mu}) and (\ref{eq:h-mu-norm}) are similar to computations already done in \cite{lavenant2025riemannian}. For example, for (\ref{eq:h-mu}), use
\[H^\varepsilon_\mu[\dot{\mu}](y) = \langle \dot{\mu}, k_\mu^\varepsilon(\cdot, y) \rangle = \int \nabla_xk_\mu^\varepsilon (x, y) v(x) d\mu(x) \]
\[ \nabla_xk_\mu^\varepsilon (x, y) = k_\mu^\varepsilon(x, y) \left( \frac{\nabla f_{\mu, \mu}^\varepsilon(x) - 2 (x-y)}{\varepsilon} \right) \]
with the identity $2x - \nabla f^\varepsilon_{\mu, \mu}(x) = 2 T^\varepsilon_{\mu}(x)$; this is similar to the computation done in \cite[Proposition~5.14]{lavenant2025riemannian}. 
For
(\ref{eq:h-mu-norm}), 
                                                                                                            
use 
\[ \|H^\varepsilon_\mu[\dot{\mu}]\|_{\mathcal{H}_\mu^\varepsilon}^2 = \langle \dot{\mu}, H^\varepsilon_\mu[\dot{\mu}] \rangle, \]
then the computations in the proof of \cite[Theorem 4.17]{lavenant2025riemannian}.
To prove (\ref{eq:decomp-general}), we expand
\begin{align*}
   g^\varepsilon_\mu(\dot{\mu}, \dot{\mu}) &= \frac{\varepsilon}{2} \langle \dot{\mu}, (\mathrm{id} - (K_\mu^\varepsilon)^2)^{-1} H_\mu^\varepsilon [\dot{\mu}] \rangle \\
    &= \frac{\varepsilon}{2} \langle H_\mu^\varepsilon[\dot{\mu}], (\mathrm{id} - (K_\mu^\varepsilon)^2)^{-1} H_\mu^\varepsilon [\dot{\mu}] \rangle_{\mathcal{H}_\mu^\varepsilon/ \mathbb{R}} \\

<!-- remark-2: body_md -->
\label{rmk:spec-gap}
Fix any $p \in U$.
From \cite{lavenant2025riemannian}, $K^\varepsilon(p)$ has eigenvalues satisfying $1 = \lambda_1^\varepsilon > \lambda_2^\varepsilon \geq \cdots  \geq \lambda_N^\varepsilon \geq 0$, so that $1- \lambda_2^\varepsilon$ corresponds to the spectral gap of the Markov chain given by $K^\varepsilon(p).$
By (\ref{eq:h-i-formula}),
\[ \left\| \frac{\varepsilon}{2} h_i^\varepsilon(p) \right\|_a \lesssim \alpha^\varepsilon(p) = O \left( e^{-c(p)/\varepsilon} \right), \]
where the multiplicative constants can depend on $\iota(p)$ and $v_{k, i}(p)$ for $i\in [m], k \in [N]$.
As we treat $N$ and $a_1, \ldots, a_N$ as multiplicative constants as well, there exists $C > 0$, independent of $\varepsilon$, such that
\[ \frac{1}{C}\alpha^\varepsilon(p) \leq \left\| I - K^\varepsilon(p) \right\|_{op} \leq C\alpha^\varepsilon(p)\]
by the equivalence of norms on a finite-dimensional vector space.
Therefore a crude bound of $s^\varepsilon_{ij}(p)$ is given by
\[ \left| s_{ij}^\varepsilon (p) \right| \leq \frac{\lambda_2^\varepsilon}{1- (\lambda_2^\varepsilon)^2} \left\| \frac{\varepsilon}{2} h_i^\varepsilon(p) \right\|_a\left\|  h_j^\varepsilon(p) \right\|_a  \lesssim \frac{\left\| I - K^\varepsilon(p) \right\|_{op}^2}{\varepsilon (1-\lambda_2^\varepsilon)}. \]
If there exists $C' > 0$ such that $\left\| I - K^\varepsilon(p) \right\|_{op} \leq C' (1-\lambda_2^\varepsilon)$ for all $\varepsilon > 0$ (e.g., if we only have $N = 2$ particles), then we obtain the same rate of convergence as (\ref{eq:s-eps-rate}), up to a multiplicative constant. More generally, if
\begin{equation}
\label{eq:spectral-gap-assump}
\lim_{\varepsilon \to 0} \frac{e^{-2c(p)/\varepsilon}}{\varepsilon(1 - \lambda_2^\varepsilon)} = 0
\end{equation}
then one can show that $s_{ij}^\varepsilon(p) \to 0$ as $\varepsilon \to 0$.
However, from \cite{lavenant2025riemannian} we only have the bound
\begin{equation}
\label{eq:spec-gap-lower-bound}
1-\lambda_2^\varepsilon \geq e^{-4D(p)^2/\varepsilon}, 
\end{equation}
i.e., a lower bound on the spectral gap which is controlled by the \emph{maximum} distance between two particles, whereas the upper bound on $\left\| I - K^\varepsilon(p) \right\|_{op}$ is controlled by the \emph{minimum} distance between two particles. We will also see this behavior in the following example.

    \end{enumerate}

<!-- lemma-4: body_md -->
\label{lemma:grad-u-upper-bound}
Let $G = (V, E, W)$ be a graph with $|V|$ finite and symmetric, non-negative edge weights. Consider any two edge flows $X, Y \in L^2_\wedge (E)$, and define $u \vcentcolon = L_G^\dagger \mathrm{div}_G Y,$ where $L_G^\dagger$ is the generalized inverse of $L_G$ as an operator from $L^2(V)$ to $L^2(V)$. Then
\begin{equation}
    \label{eq:upper-bound-gradu}
    \left\| \mathrm{grad}_G u \right\|_{L^2_\wedge(E)} \leq \left\| Y \right\|_{L^2_\wedge(E)} 
\end{equation}
    and we also have
\begin{equation}
\label{eq:neg-sob-norm-bound}
\left | \left \langle  \mathrm{div}_G X, L_G^\dagger \mathrm{div}_G Y  \right \rangle_{L^2(V)} \right| \leq \left\| X \right\|_{L^2_\wedge(E)} \left \| Y \right\|_{L^2_\wedge(E)}. 
\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
We claim that $\mathrm{div}_G Y \in (\ker L_G)^\perp$, the orthogonal complement of $\ker L_G$ in $L^2(V)$. Indeed, if $v \in \ker L_G$, then
\[ 
\langle v, L_G v \rangle_{L^2(V)} = \left\langle \mathrm{grad}_G v, \mathrm{grad}_G v \right\rangle_{L^2_\wedge(E)} = 0, 
\]
so that $(\mathrm{grad}_G v)_{kl}$ is zero whenever the edge weight $W_{kl}$ is non-zero. Thus
\[
\langle v, \mathrm{div}_G Y \rangle_{L^2(V)} = -\langle \mathrm{grad}_G v, Y \rangle_{L^2_\wedge(E)} = 0. 
\]
As $u  = L_G^\dagger \mathrm{div}_G Y$, this implies that 
$L_G u = \mathrm{div}_G Y$,
    so
    $Y + \mathrm{grad}_G u$ must belong to $\ker \mathrm{div}_G$, which is orthogonal to $\operatorname{Im} \mathrm{grad}_G$ as $\mathrm{div}_G$ is the negative adjoint of $\mathrm{grad}_G$.
    Therefore (\ref{eq:upper-bound-gradu}) holds
    and we obtain
    \[ \left | \left \langle  \mathrm{div}_G X, L_G^\dagger \mathrm{div}_G Y  \right \rangle_{L^2(V)} \right| = \left| \langle X, \mathrm{grad}_G u \rangle_{L^2_\wedge(E)} \right| \leq \left\| X \right\|_{L^2_\wedge(E)} \left \| Y \right\|_{L^2_\wedge(E)} \]
    by Cauchy-Schwarz and (\ref{eq:upper-bound-gradu}).

<!-- remark-3: body_md -->
We remark on some analogues of the objects in \cref{lemma:grad-u-upper-bound}.
\begin{itemize}[label = {--}]
    \item For $f \in (\ker L_G)^\perp$, solving $L_G u = f$ is a graph version of solving a Poisson equation, and 
    \[ \left\| f \right\|_{\dot{H}^{-1}(G)} \vcentcolon = \left( \langle f, L_G^\dagger f \rangle_{L^2(V)} \right)^{1/2}\] can be viewed as a graph version of a homogeneous negative Sobolev norm. 
    Other works have studied similar objects, although sometimes with the unnormalized graph Laplacian, i.e.,  $L_G^{un} \vcentcolon = D_G-W$.
    (Recall that the graph Laplacian $L_G$ as we have defined it corresponds to the graph Laplacian with a random walk normalization (\ref{eq:gl-rw}).) For example, see \cite{calder2020poisson} for a discussion of the graph Poisson equation with $L_G^{un}$; \cite{bandeira2026mathematics} for the relationship between the pseudo-inverse of $L_G^{un}$, the commute time distance and effective resistance; and \cite{robertson2026resistance} for the connection with linearized optimal transport on a graph.
    \item One way to view (\ref{eq:upper-bound-gradu}) is via a graph version of the Helmholtz decomposition; see \cite{lim2020hodge}. (\ref{eq:upper-bound-gradu}) is also similar to why the tangent space of Wasserstein-2 space at $\mu \in \mathcal{P}_2(\mathbb{R}^d)$ is defined as
    \[T_\mu \mathcal{P}_2(\mathbb{R}^d) \vcentcolon = \overline{ \{ \nabla \varphi \colon \varphi \in C_c^\infty(\mathbb{R}^d) \}}^{L^2(\mu)}.\]
    Indeed, as noted in the introduction, $v \in T_\mu \mathcal{P}_2(\mathbb{R}^d)$ if and only if
\[
     \|v\|_{L^2(\mu)} \leq \left\| v+ w \right\|_{L^2(\mu)}
\]
for all vector fields $w \in L^2(\mu)$ such that $\mathrm{div}(w\mu) = 0$ \cite[Lemma 8.4.2]{ambrosio2008gradient}.
\end{itemize}

\label{lemma:mult-by-Keps-div}
Fix any $\varepsilon > 0$ and $p \in U$. Consider the graph $G$ with $N$ vertices and edge weights $W = \pi^\varepsilon(p)$. Then for any edge flow $X \in L^2_\wedge(E)$,
\[ K^\varepsilon(p) \mathrm{div}_G X = \mathrm{div}_G Y, \quad (I+ K^\varepsilon(p))^{-1} \mathrm{div}_G X = \mathrm{div}_G Z, \]
where edge flows $Y, Z \in L^2_\wedge(E)$ are given by
\begin{align*}
    Y &= (I + \mathrm{grad}_G \mathrm{div}_G) X \\
    Z &= (2I + \mathrm{grad}_G \mathrm{div}_G )^{-1}X.
\end{align*}
In particular, $\mathrm{grad}_G \mathrm{div}_G$ is a self-adjoint operator from $L^2_\wedge(E)$ to itself with eigenvalues lying in $[-1, 0]$, so that $2I + \mathrm{grad}_G \mathrm{div}_G$ is indeed invertible, and the operator norm of \[(2I + \mathrm{grad}_G \mathrm{div}_G)^{-1} \colon L^2_\wedge(E) \to L^2_\wedge(E)\] is at most 1.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
To see that $K^\varepsilon(p) \mathrm{div}_G X = \mathrm{div}_G Y$, recall that $K^\varepsilon(p) = I - L_G = I + \mathrm{div}_G \mathrm{grad}_G$. 
    The self-adjointness of $\mathrm{grad}_G \mathrm{div}_G$ is clear from $\mathrm{grad}_G = - \mathrm{div}_G^*$. Additionally, $\mathrm{grad}_G = - \mathrm{div}_G^*$ implies that the non-zero eigenvalues of $\mathrm{grad}_G \mathrm{div}_G$ agree with the non-zero eigenvalues of \[\mathrm{div}_G \mathrm{grad}_G = - L_G = K^\varepsilon(p) - I.\] As shown in \cite{lavenant2025riemannian}, the eigenvalues of $K^\varepsilon(p)$ lie in $[0, 1]$, so the eigenvalues of $\mathrm{grad}_G \mathrm{div}_G$ lie in $[-1, 0]$. Therefore $2I + \mathrm{grad}_G \mathrm{div}_G$ is invertible, the eigenvalues of $(2I + \mathrm{grad}_G \mathrm{div}_G)^{-1}$ lie in $\left[\frac{1}{2}, 1\right]$ and
    \[ \mathrm{div}_G X = \mathrm{div}_G (2I + \mathrm{grad}_G \mathrm{div}_G)Z = (2I - L_G) \mathrm{div}_G Z = (I + K^\varepsilon(p))\mathrm{div}_G Z. \qedhere
    \]

<!-- proposition-3: body_md -->
\label{prop:r-eps-cont-setting}
Suppose $\Omega \subset \mathbb{R}^d$ is a bounded open set with $C^1$ boundary, and $\mu\in \mathcal{P}(\overline{\Omega})$ has density $\rho$ (with respect to the Lebesgue measure) satisfying $\rho(x) > 0$ for $x \in \Omega$, $\rho(x) = 0$ for $x \in \partial \Omega$ and $\widetilde{\rho} \vcentcolon= \sqrt{\rho} \in C^2(\overline{\Omega})$.
Then for $\dot{\mu} = -\mathrm{div}(v \mu)$ with any fixed $v \in \mathbb{R}^d\setminus\{0\}$,
\begin{equation}
\label{eq:r-eps-covariance-err}
r^\varepsilon_\mu(\dot{\mu}, \dot{\mu})  = -\frac{2}{\varepsilon} v^\top \left(\int \left( y - T_\mu^\varepsilon(x)\right)\left( y - T_\mu^\varepsilon(x)\right)^\top d\pi_{\mu, \mu}^\varepsilon(x, y) \right) v + O(\varepsilon^{1/2}) 
\end{equation}
\begin{equation}
\label{eq:covar-asymp}
\frac{1}{\varepsilon} v^\top \left(\int \left( y - T_\mu^\varepsilon(x)\right)\left( y - T_\mu^\varepsilon(x)\right)^\top d\pi_{\mu, \mu}^\varepsilon(x, y) \right) v = \Theta(1) 
\end{equation}
as $\varepsilon \to 0$.
<!-- end proposition-3 -->
<!-- proof-7: body_md -->
Let us prove the $O(1)$ upper bound for (\ref{eq:covar-asymp}) first.
We use \cite[Lemma 1]{pmlr-v162-pooladian22a}, which states that
\[
\int \left\| \nabla f^\varepsilon_{\mu, \mu} \right\|^2d\mu \leq \frac{\varepsilon^2}{4} I(\mu), 
\]
where $I(\mu) \vcentcolon = 4 \int \|\nabla \sqrt{\rho}\|^2 $ is the Fisher information of $\mu$, which is well-defined and finite under our assumptions (see Appendix \ref{app:cramer-rao}). Recalling that $2x - \nabla f_{\mu, \mu}^\varepsilon(x) = 2 T_\mu^\varepsilon(x)$, this implies that
\begin{equation}
\label{eq:diff-with-T-eps}
\int \left\| x- T_\mu^\varepsilon(x) \right\|^2 d\mu(x) \leq \frac{\varepsilon^2}{16} I(\mu) 
\end{equation}
and so by Cauchy-Schwarz
\begin{align*}
   \left| \int \int y \cdot (y-x) k^\varepsilon_\mu(x, y) d\mu(x) d\mu(y) \right |&=\left| \int y \cdot (y - T_\mu^\varepsilon(y)) d\mu(y) \right| \\
    &\leq \left( \int \|y\|^2 d\mu(y) \right)^{1/2} \left( \int \|y- T_\mu^\varepsilon(y)\|^2 d\mu(y) \right)^{1/2} \\
    &= O(\varepsilon).
\end{align*}
Therefore, using a symmetry argument, 
$\int \|y-x\|^2 d\pi_{\mu, \mu}^\varepsilon(x, y) = O(\varepsilon)$ and also
\begin{equation}
\label{eq:y-t-eps-x-bound}
\int \|y- T_\mu^\varepsilon(x)\|^2 d\pi_{\mu, \mu}^\varepsilon(x, y) \leq 2 \left(\int \|y-x\|^2 + \left\| x- T_\mu^\varepsilon(x) \right\|^2 d\pi_{\mu, \mu}^\varepsilon(x, y)  \right) = O(\varepsilon), \end{equation}
which implies the $O(1)$ upper bound in (\ref{eq:covar-asymp}). 
As
\[ r^\varepsilon_\mu(\dot{\mu}, \dot{\mu})  = \frac{2}{\varepsilon} v^\top \left(\int  \left( y - T^\varepsilon_{\mu}(x)\right)\left(x - T^\varepsilon_{\mu}(y) \right)^\top d\pi_{\mu, \mu}^\varepsilon(x, y)\right)v \]
under the constant velocity assumption, (\ref{eq:r-eps-covariance-err}) follows from the previous bounds and another application of Cauchy-Schwarz:
\begin{align*}
    \bigg| r_\mu^\varepsilon(\dot{\mu}, \dot{\mu} ) &+ \frac{2}{\varepsilon} v^\top \left(\int \left( y - T_\mu^\varepsilon(x)\right)\left( y - T_\mu^\varepsilon(x)\right)^\top d\pi_{\mu, \mu}^\varepsilon(x, y) \right) v \bigg| \\
    &= \frac{2}{\varepsilon} \left| v^\top \left(\int \left( y - T_\mu^\varepsilon(x)\right)\left( y - T_\mu^\varepsilon(y) + x- T_\mu^\varepsilon(x)\right)^\top d\pi_{\mu, \mu}^\varepsilon(x, y) \right) v  \right| \\
    &\leq \frac{4}{\varepsilon} \left( \int \left| v^\top  \left( y - T_\mu^\varepsilon(x)\right)\right|^2 d\pi_{\mu, \mu}^\varepsilon(x, y) \right)^{1/2} \left( \int \left| \left( x - T_\mu^\varepsilon(x)\right)^\top v  \right|^2 d\pi_{\mu, \mu}^\varepsilon(x, y)\right)^{1/2}
\end{align*}
which is $O(\varepsilon^{1/2})$ by (\ref{eq:diff-with-T-eps}) and (\ref{eq:y-t-eps-x-bound}).

It remains to prove the lower bound of $\Omega(1)$ for (\ref{eq:covar-asymp}). This is inspired by \cite{chewi2023entropic}, where they prove Caffarelli's contraction theorem by using the Hessian of the potentials and covariance inequalities and then taking $\varepsilon\to 0$. Let us define $W(x) \vcentcolon= -\log \rho(x) = - 2 \log \widetilde{\rho}(x) $ for $x \in \Omega$. One of the observations in \cite{chewi2023entropic} (and earlier in \cite[Appendix~B.2]{daniels2021score}), specialized to the self-transport setting, is that for $\varphi^\varepsilon(x) \vcentcolon = \|x\|^2 - f^\varepsilon_{\mu, \mu}(x)$,
\begin{equation}
\label{eq:hess-potential}
\nabla^2 \varphi^\varepsilon(x) = \frac{4}{\varepsilon} \int (y - T_\mu^\varepsilon(x)) (y- T_\mu^\varepsilon(x))^\top k^\varepsilon_\mu(x, y) d\mu(y) = \frac{4}{\varepsilon} \mathrm{Cov}_{Y \sim \pi_x^\varepsilon}(Y) 
\end{equation}
\begin{equation}
-\nabla^2 \log \pi_x^\varepsilon (y) = - \left( \frac{\nabla^2 f_{\mu, \mu}^\varepsilon(y) - 2I }{\varepsilon} \right) + \nabla^2 W(y) = \frac{1}{\varepsilon} \nabla^2 \varphi^\varepsilon(y) + \nabla^2 W(y) 
\end{equation}
for $x, y \in \Omega$,
where the measure $\pi_x^\varepsilon$ is given by $d\pi_x^\varepsilon(y) = k_\mu^\varepsilon(x, y) d\mu(y).$ (The factor of $4$ arises from our use of the cost $x, y \mapsto \|x-y\|^2$, compared to the convention in \cite{chewi2023entropic} of $x, y \mapsto \frac{1}{2}\|x-y\|^2$.)
As in \cite{chewi2023entropic}, using the Cram\'er-Rao inequality, we bound
\begin{align*}
\frac{1}{\varepsilon} \mathrm{Cov}_{Y \sim \pi_x^\varepsilon}(Y) &\succeq \frac{1}{\varepsilon}\left( \mathbb{E}_{Y \sim \pi_x^\varepsilon} \left[ \frac{1}{\varepsilon} \nabla^2 \varphi^\varepsilon(Y) + \nabla^2 W(Y) \right] \right)^{-1} \\
&= \left( \mathbb{E}_{Y \sim \pi_x^\varepsilon} \left[\nabla^2 \varphi^\varepsilon(Y)  + \varepsilon \nabla^2 W(Y)  \right] \right)^{-1}.
\end{align*}
See Appendix \ref{app:cramer-rao} for a more detailed justification.
Since \cite{chewi2023entropic} would like bounds on $\nabla^2\varphi^\varepsilon(x)$ for all $x$, they then use a uniform upper bound on $\nabla^2 W$ and the Brascamp--Lieb inequality (which requires a strictly log-concave measure) to bound the last line. However, as $W(x)$ goes to infinity as $x$ approaches $\partial \Omega$, we do not have $\beta > 0$ such that $\nabla^2 W(x) \preceq \beta I$ for all $x \in \Omega$.
Instead, as we are only interested in
\[  \int \left( y - T_\mu^\varepsilon(x)\right)\left( y - T_\mu^\varepsilon(x)\right)^\top d\pi_{\mu, \mu}^\varepsilon(x, y) = \mathbb{E}_{X \sim \mu}\left[\mathrm{Cov}_{Y \sim \pi_X^\varepsilon}(Y) \right], \]
we can use the convexity of the matrix inverse on positive definite matrices to bound
\begin{align*}
\frac{1}{\varepsilon} \mathbb{E}_{X \sim \mu} \left[w^\top \mathrm{Cov}_{Y \sim \pi_X^\varepsilon}(Y) w\right] &\geq \mathbb{E}_{X \sim \mu} \left[ w^\top\left(\mathbb{E}_{Y \sim \pi_X^\varepsilon} \left[\nabla^2 \varphi^\varepsilon(Y)  + \varepsilon  \nabla^2 W(Y)  \right]\right)^{-1}w\right] \\ 
&\geq   w^\top \left(\mathbb{E}_{X \sim \mu} \left[\mathbb{E}_{Y \sim \pi_X^\varepsilon} \left[\nabla^2 \varphi^\varepsilon(Y)  + \varepsilon  \nabla^2 W(Y)  \right]\right]\right)^{-1} w \\
&= w^\top \left(\mathbb{E}_{ Y \sim \mu} \left[ \nabla^2 \varphi^\varepsilon(Y)  + \varepsilon  \nabla^2 W(Y)  \right]\right)^{-1}w
\end{align*}
for any $w \in \mathbb{R}^d$, where we use Jensen's inequality in the second line and Fubini's theorem in the last line. One can check that each expectation (integral) above is well-defined, and Fubini's theorem can be applied in the last line: by the integrability of $\nabla^2 W$ with respect to $\mu$ (\cref{lemma:cramer-rao}), $x, y \mapsto \left( \nabla^2 \varphi^\varepsilon(y) + \varepsilon \nabla^2 W(y) \right) k_\mu^\varepsilon(x, y)$ is integrable with respect to $\mu \otimes \mu$, and
\[ x \mapsto \mathbb{E}_{Y \sim \pi_x^\varepsilon} \left[\nabla^2 \varphi^\varepsilon(Y)  + \varepsilon  \nabla^2 W(Y)  \right] = \int \left( \nabla^2 \varphi^\varepsilon(y) + \varepsilon \nabla^2 W(y) \right) k_\mu^\varepsilon(x, y) d\mu(y) \]
is continuous on $\Omega$ and also integrable with respect to $\mu$.
                
                                                                                                                                                                                                                                    
                                                                                                                                                    
                                                                                                                              
                                                                                                        
              
                                                          
By (\ref{eq:hess-potential}),
\[ \mathbb{E}_{X \sim \mu} \left[ \nabla^2 \varphi^\varepsilon(X) \right] = \frac{4}{\varepsilon}\mathbb{E}_{X \sim \mu} \left[ \mathrm{Cov}_{Y \sim \pi_X^\varepsilon}(Y) \right]. \]
Moreover, under our assumptions on $\mu$,
$\beta \vcentcolon= \|\mathbb{E}_{\mu}\left[  \nabla^2 W \right]\|_{op}$
is finite.
Therefore 
\[ \frac{1}{\varepsilon} \mathbb{E}_{X \sim \mu} \left[ \mathrm{Cov}_{Y \sim \pi_X^\varepsilon}(Y) \right] \succeq \left(\frac{4}{\varepsilon} \mathbb{E}_{X \sim \mu} \left[ \mathrm{Cov}_{Y \sim \pi_X^\varepsilon}(Y) \right] + \varepsilon  \beta  \right)^{-1}, \]
and any eigenvalue $\lambda$ of $\frac{1}{\varepsilon} \mathbb{E}_{X \sim \mu} \left[\mathrm{Cov}_{Y \sim \pi_X^\varepsilon}(Y)\right]$ satisfies
\[ \lambda (4\lambda + \varepsilon \beta) \geq 1, \]
which gives us
\[ \lambda \geq \sqrt{\frac{1}{4} + \left( \frac{\varepsilon \beta}{8} \right)^2} - \frac{\varepsilon \beta}{8} \]
since $\lambda$ must be non-negative.
As a result,
\[ \frac{1}{\varepsilon} v^\top \mathbb{E}_{X \sim \mu} \left[\mathrm{Cov}_{Y \sim \pi_X^\varepsilon}(Y)\right] v = \Omega(1) \]
as $\varepsilon \to 0$, proving (\ref{eq:covar-asymp}).

Then there exists $C_{n+1} \colon U \to \mathbb{R}_{> 0}$, which can be written as a continuous function of $\iota$ and its partial derivatives of order $\leq n+1$, such that

<!-- proposition-4: body_md -->
\label{prop:pi-derivs}
    The partial derivatives of $\pi^\varepsilon$ with respect to $p$ can be written in the form
    \begin{equation}
    \label{eq:pi-derivs}
        \frac{\partial^n \pi_{kl}^\varepsilon}{\partial p_{i_1} \cdots \partial p_{i_n}} = \frac{1}{\varepsilon^n} \pi_{kl}^\varepsilon \psi^\varepsilon_{kl, i_1, \ldots, i_n}
    \end{equation}
    for some smooth $\psi^\varepsilon_{kl, i_1, \ldots, i_n} \colon U \to \mathbb{R}$ with $\psi^\varepsilon_{kl, i_1, \ldots, i_n} = \psi^\varepsilon_{lk, i_1, \ldots, i_n}$. Moreover, there exists  a function $C_n \colon U \to \mathbb{R}_{> 0}$, which can be written as a continuous function of $\iota$ and its partial derivatives of order $\leq n$, such that
    \begin{equation}
    \label{eq:psi-bound}
        \left|\psi^\varepsilon_{kl, i_1, \ldots, i_n}\right| \leq C_n
    \end{equation}
    for all $\varepsilon \in (0, \varepsilon_0)$.
    Since $K^\varepsilon_{kl} = \frac{1}{a_k} \pi^\varepsilon_{kl}$, we also have
    \begin{equation}
    \label{eq:k-derivs}
    \frac{\partial^n K_{kl}^\varepsilon}{\partial p_{i_1} \cdots \partial p_{i_n}} = \frac{1}{\varepsilon^n} K_{kl}^\varepsilon \psi^\varepsilon_{kl, i_1, \ldots, i_n}.
    \end{equation}

\label{cor:deriv-of-graph-div}
    Suppose that $X \colon U \to \mathbb{R}^{N \times N}$ is smooth with $X_{kl} = - X_{lk}$, and fix any $\varepsilon > 0$. Consider
    $\eta_k \vcentcolon = \sum_l K^\varepsilon_{kl}X_{kl}$ and any $i_1, \ldots, i_n \in [m]$.  Then
    \[ \frac{\partial^n \eta_k}{\partial p_{i_1} \cdots \partial p_{i_n}} = \sum_l \sum_{S \subseteq [n]} \frac{\partial^{|S|} K_{kl}^\varepsilon}{\partial p_{i_S}} \frac{\partial^{|[n] \setminus S|}X_{kl}}{\partial p_{i_{[n] \setminus S}}} =  \sum_l K_{kl}^\varepsilon \sum_{S \subseteq [n]} \varepsilon^{ - |S|}\psi^\varepsilon_{kl, i_S} \frac{\partial^{|[n] \setminus S|}X_{kl}}{\partial p_{i_{[n] \setminus S}}}. \]
    In other words, if $\eta_k(p) = (\mathrm{div}_G X(p))_k$ for each $p \in U$ and graph $G$ with edge weights $W = \pi^\varepsilon(p)$, then for each $p \in U$
    \[ \frac{\partial^n \eta_k}{\partial p_{i_1} \cdots \partial p_{i_n}}(p) =  (\mathrm{div}_G Z(p))_k, \]
    where again $G$ is the graph with edge weights $W = \pi^\varepsilon(p)$ and $Z$ is given by
    \[Z_{kl}(p) = \sum_{S \subseteq [n]} \varepsilon^{ - |S|}\psi^\varepsilon_{kl, i_S}(p) \frac{\partial^{|[n] \setminus S|}X_{kl}}{\partial p_{i_{[n] \setminus S}}}(p).\]

\label{assump:mu-density}
    Suppose $\Omega \subset \mathbb{R}^d$ is a bounded open set with $C^1$ boundary, and $\mu\in \mathcal{P}(\overline{\Omega})$ has density $\rho$ (with respect to the Lebesgue measure) satisfying:
\begin{itemize}[label = {--}]
    \item $\rho(x) > 0$ for $x \in \Omega$,
    \item $\rho(x) = 0$ for $x \in \partial \Omega$,
    \item $\widetilde{\rho} \vcentcolon= \sqrt{\rho} \in C^2(\overline{\Omega})$.
\end{itemize}
We also define $W(x) \vcentcolon= -\log \rho(x) = - 2 \log \widetilde{\rho}(x) $ for $x \in \Omega$, i.e., $\rho(x) = e^{-W(x)}$ on $\Omega$.
<!-- end assumption-2 -->
<!-- lemma-7: body_md -->
\label{lemma:cramer-rao}
    Under Assumption \ref{assump:mu-density}, $\nabla^2W$ is integrable with respect to $\mu$, $\mathbb{E}_\mu\left[\nabla^2W \right]$ is positive definite, and
\begin{equation}
\label{eq:cramer-rao}
\mathrm{Cov}_{X \sim \mu}(X) \succeq \left( \mathbb{E}_\mu \left[\nabla^2W \right] \right)^{-1}. 
\end{equation}

<!-- remark-5: body_md -->
Recall that we would like to apply Cram\'er-Rao to $d\pi_x^\varepsilon = k_\mu^\varepsilon(x, y) d\mu(y)$. If $\mu$ satisfies Assumption \ref{assump:mu-density}, then $\pi_x^\varepsilon$ has density 
\[y \mapsto k_\mu^\varepsilon(x, y) \rho(y) = \exp \left( \frac{f_{\mu, \mu}^\varepsilon(x) + f_{\mu, \mu}^\varepsilon(y) - \|x-y\|^2}{\varepsilon} \right) \rho(y).\]
Since $f_{\mu, \mu}^\varepsilon$ is smooth (recall the representation \eqref{eq:schrodinger-1-1}), each $\pi_x^\varepsilon$ satisfies Assumption \ref{assump:mu-density} whenever $\mu$ does, so we can apply \cref{lemma:cramer-rao} to $\pi_x^\varepsilon$.

<!-- proof-13: body_md -->
First we check that $\nabla^2 W$ is integrable with respect to $\mu$. For this, it is convenient to write the derivatives of $W = - 2 \log \widetilde{\rho}$ in terms of $\widetilde{\rho}$:
\[ \frac{\partial W}{\partial x_k} = -\frac{2}{\widetilde{\rho}} \frac{\partial \widetilde{\rho}}{\partial x_k} \]
\[
\frac{\partial^2 W}{\partial x_k \partial x_l} = -2 \left( -\frac{1}{\widetilde{\rho}^2} \frac{\partial \widetilde{\rho}}{\partial x_l} \frac{\partial \widetilde{\rho}}{\partial x_k} + \frac{1}{\widetilde{\rho}} \frac{\partial^2 \widetilde{\rho}}{\partial x_k \partial x_l} \right) 
\]
on $\Omega$.
As $\widetilde{\rho}^2 = \rho$ and $\widetilde{\rho} \in C^2(\overline{\Omega})$, we have that each $\frac{\partial^2 W}{\partial x_k \partial x_l}$ is integrable with respect to $\mu$ and
\begin{equation}
\label{eq:psd}
\int_\Omega \frac{\partial^2 W}{\partial x_k \partial x_l} \rho(x) dx = 2\int_\Omega \frac{\partial \widetilde{\rho}}{\partial x_l} \frac{\partial \widetilde{\rho}}{\partial x_k} - \widetilde{\rho} \frac{\partial^2 \widetilde{\rho}}{\partial x_k \partial x_l} dx = 4 \int_\Omega \frac{\partial \widetilde{\rho}}{\partial x_l} \frac{\partial \widetilde{\rho}}{\partial x_k} dx,
\end{equation}
where the last equality holds by Green's identity with the boundary condition $\widetilde{\rho}(x) = 0$ for $x \in \partial\Omega$.
                                                                                 
Modifying the proof from \cite{chewi2023entropic},
let us fix any non-zero vector $w \in \mathbb{R}^d$, and take $h \vcentcolon= \langle w, \cdot\rangle$.
Then
\[ w = \mathbb{E}_\mu[\nabla h] = \int_\Omega \rho(x) \nabla h(x) dx = - \int_\Omega h(x) \nabla \rho(x) dx \]
\[ \int_\Omega \nabla \rho(x) dx = 0 \]
again by Green's identity and the boundary condition $\rho(x) = 0$ for $x \in \partial \Omega$. Therefore
\[ w = - \int_\Omega (h(x) - \mathbb{E}_\mu[h]) \nabla \rho(x) dx = - 2\int_\Omega (h(x) - \mathbb{E}_\mu[h]) \widetilde{\rho}(x) \nabla \widetilde{\rho}(x) dx, \]
so that
\[ \|w\|^2 = - 2\int_\Omega (h(x) - \mathbb{E}_\mu[h]) \widetilde{\rho}(x) w^\top \nabla \widetilde{\rho}(x) dx.\]
Applying Cauchy-Schwarz to the right-hand side gives us
\begin{align*}
\|w\|^2 &\leq \left( \int_\Omega (h(x) - \mathbb{E}_\mu[h])^2 \widetilde{\rho}^2(x) dx \right)^{1/2} \left( 4 \int_\Omega (w^\top \nabla \widetilde{\rho}(x))^2 dx \right)^{1/2} \\
&= \left( \mathrm{Var}_\mu(h) \right)^{1/2} \left( w^\top \mathbb{E}_\mu[\nabla^2 W] w \right)^{1/2}
\end{align*}
since $\mathbb{E}_\mu[\nabla^2 W]$ is given by (\ref{eq:psd}).
As $\mathrm{Var}_\mu(h) = \int_\Omega (h(x) - \mathbb{E}_\mu[h])^2 \rho(x) dx > 0$, we have that $\mathbb{E}_\mu[\nabla^2 W]$ is strictly positive definite, so that
\begin{align*}
    \left\langle w, \mathbb{E}_\mu[\nabla^2 W]^{-1}w \right\rangle &=  - 2\int_\Omega (h(x) - \mathbb{E}_\mu[h]) \widetilde{\rho}(x) \left(\mathbb{E}_\mu[\nabla^2 W]^{-1}w \right)^\top \nabla \widetilde{\rho}(x) dx \\
    &\leq \left( \mathrm{Var}_\mu(h) \right)^{1/2} \left( 4 \int_\Omega \left(\left(\mathbb{E}_\mu[\nabla^2 W]^{-1}w \right)^\top \nabla \widetilde{\rho}(x)\right)^2 dx \right)^{1/2} \\
    &= \left( \mathrm{Var}_\mu(h) \right)^{1/2} \left(  \left\langle w, \mathbb{E}_\mu[\nabla^2 W]^{-1}w \right \rangle \right)^{1/2}
\end{align*}
by another application of Cauchy-Schwarz. (\ref{eq:cramer-rao}) follows from rearranging.
