\label{thm: théorème principal}
Let $n \ge 1$ and $\omega \in A_2(\mathbb{R}^n)$. The following assertions hold. 
\begin{enumerate}[label=(\arabic*), leftmargin=*, itemsep=1pt]
    \item \label{Point (1)} \textbf{Range of exponents:} We have $p_-(\mathcal{H})\in [1,2)$ with an upper bound depending only on $[\omega]_{A_2}$ and $n$. In addition, for every $q \in (2_\star,2)$, with $2_\star=2-\frac{4}{n+4}\in (1,2)$,
    \begin{equation*}
    \omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n)\quad \Longrightarrow \quad p_-(\mathcal{H}) < q .
    \end{equation*}
    In the endpoint case $q = 2_\star$, if $\omega \in RH_{1+\frac{2}{n}}(\mathbb{R}^n)$, then $p_-(\mathcal{H}) \le 2_\star$.
    \item \label{Point (2)} \textbf{Boundedness of the parabolic Riesz transform:} For every $p \in (p_-(\mathcal{H}),2]$, the parabolic Riesz transform
    \begin{equation*}
    \mathcal{R}_{\mathcal{H}} = (\nabla_x \mathcal{H}^{-1/2}, D_t^{1/2}\mathcal{H}^{-1/2})
    \end{equation*}
    extends to a bounded operator on $\mathrm{L}^p_{\mu_p}(\mathbb{R}^{n+1})$.
    \item \label{Point (3)} \textbf{Case of real coefficients:} If $A$ has real coefficients, then for every $q \in (1,2]$,
    \begin{equation*}
        p_-(\mathcal{H}) < q \quad \Longleftrightarrow \quad \omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n).
    \end{equation*}
    In particular, $p_-(\mathcal{H}) = 1$ if and only if
    \begin{equation*}
         \omega \in \bigcap_{q \in (1,2]} RH_{\frac{q'}{2}}(\mathbb{R}^n)=: RH_\star(\mathbb{R}^n).
    \end{equation*}
    Moreover, if $\omega \in RH_{\infty}(\mathbb{R}^n)\subsetneq RH_\star(\mathbb{R}^n)$, then $p_-(\mathcal{H}) = 1$, and the spatial gradient component $\omega^{\frac{1}{2}} \nabla_x \mathcal{H}^{-1/2}$ extends to a bounded operator from $\mathrm{L}^1_{\mu_1}(\mathbb{R}^{n+1})$ into $\mathrm{L}^{1,\infty}(\mathbb{R}^{n+1})$.
\end{enumerate}

<!-- defn-1: body_md -->
\label{def: toolbox}
Let $1\le p \le q \le \infty$. The family $(T_\lambda)_{\lambda \in \mathcal{U}}$ is said to 
\begin{enumerate}
    \item be  $\mathrm{L}^p-\mathrm{L}^q$ bounded if there exists a constant $C>0$ such that 
    \begin{equation*}
        \| T_\lambda u\|_q \le C |\lambda|^{\gamma_{p,q}} \|  u\|_p,
    \end{equation*}
    for all $\lambda\in \mathcal{U}$ and all $u \in \mathrm{L}^p(\mathbb{R}^{n+1}) \cap \mathrm{L}^2(\mathbb{R}^{n+1})$.
    \item satisfy the  $\mathrm{L}^p-\mathrm{L}^q$ off-diagonal estimates if there exist constants $C,c$ such that
    \begin{equation*}
        \| \boldsymbol{1}_F T_\lambda (\boldsymbol{1}_E u)\|_q \le C |\lambda|^{\gamma_{p,q}} e^{-c \frac{\,\mathrm{d}(E,F)}{|\lambda|}} \| \boldsymbol{1}_E u\|_p,
    \end{equation*}
    for all $\lambda\in \mathcal{U}$, $u \in \mathrm{L}^p(\mathbb{R}^{n+1}) \cap \mathrm{L}^2(\mathbb{R}^{n+1})$ and all measurable sets $E,F\subset \mathbb{R}^{n+1}$, where $\,\mathrm{d}(E,F)$ is the parabolic distance between $E$ and $F$.
\end{enumerate}
When $p=q$, we speak of  $\mathrm{L}^p$-boundedness and  $\mathrm{L}^p$ off-diagonal estimates, respectively.

<!-- lem-1: body_md -->
\label{lem: Toolbox}
Let $(T_\lambda)_{\lambda \in \mathcal{U}}$ and $(S_\lambda)_{\lambda \in \mathcal{U}}$ be operator families as above. Let $1\le p \le q \le r \le \infty$.
\begin{enumerate}
    \item \emph{(Duality)} $(T_\lambda)_{\lambda \in \mathcal{U}}$ is  $\mathrm{L}^p-\mathrm{L}^q$ bounded if and only if $(T^\star_\lambda)_{\lambda \in \mathcal{U}}$ is  $\mathrm{L}^{q'}-\mathrm{L}^{p'}$ bounded.
    \item \emph{(Composition)} If $(T_\lambda)_{\lambda \in \mathcal{U}}$ is  $\mathrm{L}^p-\mathrm{L}^q$ bounded and $(S_\lambda)_{\lambda \in \mathcal{U}}$ is  $\mathrm{L}^q-\mathrm{L}^r$ bounded, then $(S_\lambda T_\lambda)_{\lambda \in \mathcal{U}}$ is  $\mathrm{L}^p-\mathrm{L}^r$ bounded. 
    \item \emph{(Extrapolation)} If $(T_\lambda)_{\lambda \in \mathcal{U}}$ satisfies  $\mathrm{L}^p-\mathrm{L}^q$ off-diagonal estimates, then it is  $\mathrm{L}^p$-bounded and  $\mathrm{L}^q$-bounded.
\end{enumerate}
Points (1) and (2) remain valid if boundedness is replaced by off-diagonal estimates.

\label{prop: time supports}
Let $p \in (1,2]$ be such that the families $(\omega^{\frac{1}{2}} \, \mathcal{E}_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ and $(\omega^{\frac{1}{2}} \, \lambda D_t^{1/2}\mathcal{E}_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ are $\mathrm{L}^p$-bounded, and that $\mu_{p'}$ is l.f. on $\mathbb{R}^{n+1}$. For every $N>1$, there exists a constant $C = C(N,p)$  such that the following off-diagonal estimates hold:
\begin{equation*}
    \|\boldsymbol{1}_{\mathbb{R}^n\times F} \, \omega^{\frac{1}{2}} \lambda D_t^{1/2}\mathcal{E}_\lambda \left ( \omega^{-\frac{1}{2}}  \boldsymbol{1}_{\mathbb{R}^n\times E} u \right ) \|_{p}
        \lesssim_p C\left(\frac{\lambda}{r} + \left(\frac{\lambda}{r}\right)^2\right)
        N^{-j\varepsilon}
        \|\boldsymbol{1}_{\mathbb{R}^n\times E} u\|_{p},
\end{equation*}
for all $\lambda, r > 0$, all $j \ge 1$ and all $u \in \mathrm{L}^p(\mathbb{R}^{n+1})\cap \mathrm{L}^2(\mathbb{R}^{n+1})$ in the following two scenarios:
\begin{enumerate}[label=(\roman*), ref=\roman*]
    \item\label{item: cube to annulus time} \emph{(From the cube to the annulus)} For $E= N^{j\delta}I_r$, $F= N^{j+1}I_{r} \setminus N^{j} I_{r}$ with $\varepsilon = 1+\frac{1}{1+p'}$. Here, the parameter $\delta \in [0,\gamma)$, where \begin{equation}\label{eq: epsilon*}
    \gamma= \frac{2+p'}{2+2p'} \in \left ( \frac{1}{2} , \frac{3}{4} \right ),
    \end{equation}
    
    \item\label{item: annulus to cube time} \emph{(From the annulus to the cube)} 
    For $E= N^{j+1}I_r \setminus N^j I_r$, $F= I_r$ with $\varepsilon = 2$.
\end{enumerate}
<!-- end prop-4 -->
<!-- proof-2: body_md -->
For $\eta \in \mathrm{C}^\infty_b(\mathbb{R})$ and $u\in \mathrm{L}_\mu^2(\mathbb{R}^{n+1})$, we have for all $\lambda>0$:
\begin{align*}
    \eta \lambda \mathcal{E}^\star_\lambda D_t^{1/2}u=\lambda [\eta, \mathcal{E}_\lambda^\star]D_t^{1/2}u+\lambda \mathcal{E}^\star_\lambda \eta D_t^{1/2}u
    &= \lambda \mathcal{E}^\star_\lambda [1+\lambda^2 \mathcal{H}^\star, \eta]\mathcal{E}_\lambda^\star D_t^{1/2}u+\lambda \mathcal{E}^\star_\lambda \eta D_t^{1/2}u
    \\&= -\lambda^3 \mathcal{E}^\star_\lambda (\partial_t \eta) \mathcal{E}^\star_\lambda D_t^{1/2}u+\lambda \mathcal{E}^\star_\lambda \eta D_t^{1/2}u
    \\&=: \mathrm{I}^\eta_u+\mathrm{II}^\eta_u,
\end{align*}
where we used that $[A,B^{-1}]=B^{-1}[B,A]B^{-1}$ in the second equality, and that 
\begin{equation}\label{eq: commutator}
    [1+\lambda^2 \mathcal{H}^\star, \eta]=\lambda^2[\mathcal{H}^\star, \eta]=-\lambda^2\partial_t \eta \quad \text{on} \ \mathrm{E}_\mu \ (\text{valued in} \ \mathrm{E}^\star_\mu)
\end{equation}
in the third equality, since $\mathcal{H}^\star = -\partial_t -  \omega^{-1} \mathrm{div}_x\!\left(A^\star \nabla_x\right)$ and $\mathcal{H}^\star$ is local in its variables. For \eqref{eq: commutator}, one first proves the identity on $\mathrm{C}_0^\infty(\mathbb{R}^{n+1})$ and then extends it to $\mathrm{E}_\mu$ by density. Now, by duality, the families $(\omega^{\frac{1}{2}}\mathcal{E}^\star_\lambda)_{\lambda>0}$ and $(\omega^{\frac{1}{2}}\lambda \mathcal{E}^\star_\lambda D_t^{1/2})_{\lambda>0}$ are uniformly bounded from $\mathrm{L}^{p'}_{\mu_{p'}}(\mathbb{R}^{n+1}) \cap \mathrm{L}^{2}_{\mu}(\mathbb{R}^{n+1})$ to $\mathrm{L}^{p'}(\mathbb{R}^{n+1})$. Thus, for $u\in \mathrm{L}^{p'}_{\mu_{p'}}(\mathbb{R}^{n+1}) \cap \mathrm{E}_\mu$, we have
\begin{equation*}
    \Vert\omega^{\frac{1}{2}}\eta \lambda \mathcal{E}^\star_\lambda D_t^{1/2}u\Vert_{p'} \le \Vert\omega^{\frac{1}{2}}\mathrm{I}^\eta_{u}\Vert_{p'}+\Vert\omega^{\frac{1}{2}}\mathrm{II}^\eta_{u}\Vert_{p'} \lesssim_p  \lambda^2 \Vert\partial_t \eta\Vert_{\infty} \Vert\omega^{\frac{1}{2}}u\Vert_{p'}+\lambda \Vert\omega^{\frac{1}{2}}\eta D_t^{1/2}u\Vert_{p'}.
\end{equation*}
Now, one proceeds exactly as in \cite[Prop.~4.4 and Rem.~4.9]{BEK26}: one first takes $u \in \mathrm{C}_0^\infty(\mathbb{R}^{n+1})$, uses the representation formula \eqref{eq: Dt representation} for $D_t^{1/2}u$, chooses the appropriate $\eta$ for each scenario, and concludes by density. We stress that we assumed that $\mu_{p'}$ is l.f. on $\mathbb{R}^{n+1}$, so that $\mathrm{C}_0^\infty(\mathbb{R}^{n+1}) \subset \mathrm{L}^{p'}_{\mu_{p'}}(\mathbb{R}^{n+1}) \cap \mathrm{L}^{2}_{\mu}(\mathbb{R}^{n+1})$ and is a dense subspace of $\mathrm{L}^{p'}_{\mu_{p'}}(\mathbb{R}^{n+1})$. This is also used in the duality argument mentioned earlier.

Let $\varrho \in [1,2]$ such that the families $(\omega^{\frac{1}{2}} \,\mathcal{E}_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ and $(\omega^{\frac{1}{2}} \, \lambda \mathbb{D}\mathcal{E}_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ are $\mathrm{L}^\varrho$-bounded, and that $\mu_{\varrho'}$ is l.f. on $\mathbb{R}^{n+1}$. <!-- prop-5: body_md -->
\label{lem: Sobolev}
Let $q \in [2_\star,2)$ and assume that $\omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n)$. Then
\begin{align*}
    \mathrm{E}_\mu \subset \mathrm{L}_{\mu_{q'}}^{q'}(\mathbb{R}^{n+1}).
\end{align*}
Moreover, there exists a constant $C = C([\omega]_{A_2}, [\omega]_{RH_{\frac{q'}{2}}},q, n)$ such that, for every $u \in \mathrm{E}_\mu$,
\begin{equation}\label{eq: Gagliardo-Nirenberg}
    \Vertu\Vert_{\mathrm{L}_{\mu_{q'}}^{q'}(\mathbb{R}^{n+1})}
    \le C \|\mathbb{D} u\|_{2,\mu}^{\kappa}\|u\|_{2,\mu}^{1-\kappa},
\end{equation}
where $\kappa=\nicefrac{(n+2)}{2}-\nicefrac{(n+2)}{q'}=-\gamma_{2,q'}$. Observe that when $q=2_\star$, or equivalently $q'=2^\star$, one has $\kappa=1$. In this case, the embedding is homogeneous and therefore optimal.
<!-- end prop-5 -->
<!-- proof-3: body_md -->
Fix $u \in \mathrm{E}_\mu$. Set $\alpha = n\left(\frac{1}{2} - \frac{1}{q'}\right)$. We define the degenerate Laplacian $-\Delta_\omega$ via the method of forms. We refer to \cite{auscherbaadi2025hardy} for details. By \cite[Theorem 1.1]{auscherbaadi2025hardy}, there exists a constant $C_1 = C_1([\omega]_{A_2}, [\omega]_{RH_{\frac{q'}{2}}}, q, n)$ such that we have
\begin{equation}\label{eq: HLS}
    \Vertu(t)\Vert_{\mathrm{L}^{q'}_{\omega^{q'/2}}(\mathbb{R}^n)} \le C_1 \Vert((-\Delta_\omega)^{\frac{\alpha}{2}} u)(t)\Vert_{2,\omega}
    \quad \text{for all } t \in \mathbb{R}.
\end{equation}
Since $q' \le 2^\star$, and distinguishing between the cases $n \in \{1,2\}$ and $n \ge 3$, it is easy to see that $\alpha \in (0,1]$. Set $r = \frac{2}{\alpha} \in [2,\infty)$. This is equivalent to
\begin{equation}\label{eq: compatibilité}
\frac{1}{r} + \frac{n}{2q'} = \frac{n}{4}.
\end{equation}
By taking the $\mathrm{L}^r$-norm of \eqref{eq: HLS} and using \cite[Proposition 5.4]{auscherbaadi2024fundamental} with $S = (-\Delta_\omega)^{1/2}$ therein, there exists a constant $C_2 = C_2(r)$ such that
\begin{align*}
\| u \|_{\mathrm{L}^r(\mathbb{R};\mathrm{L}^{q'}_{\omega^{q'/2}}(\mathbb{R}^n))}
\leq C_1 \left\| (-\Delta_\omega)^{\frac{\alpha}{2}} u \right\|_{\mathrm{L}^r(\mathbb{R};\mathrm{L}^2_\omega(\mathbb{R}^n))}
\leq C_1 C_2 \|\nabla_x u \|_{2,\mu}^\alpha \|D_t^{1/2}u \|_{2,\mu}^{1-\alpha}  \le C_1 C_2 \Vert\mathbb{D} u\Vert_{2,\mu},
\end{align*}
where we used the trivial identity $\Vert\nabla_x \cdot\Vert_{2,\omega} = \Vert(-\Delta_\omega)^{\frac{1}{2}} \cdot\Vert_{2,\omega}$ in the penultimate inequality. The case $q=2_\star$ follows directly, since solving for $r=q'$ in \eqref{eq: compatibilité} yields $r=q'=2^\star$. The case $q\in(2_\star,2)$ follows by interpolation. First, taking the $\mathrm{L}^2$-norm of \eqref{eq: HLS} and using an interpolation inequality (the moment inequality) yields
\begin{align*}
    \| u \|_{\mathrm{L}^2(\mathbb{R};\mathrm{L}^{q'}_{\omega^{q'/2}}(\mathbb{R}^n))}
\leq C_1 \| (-\Delta_\omega)^{\frac{\alpha}{2}} u \|_{\mathrm{L}^2(\mathbb{R};\mathrm{L}^2_\omega(\mathbb{R}^n))} &\le C_1 \left\|  u \right\|^{1-\alpha}_{\mathrm{L}^2(\mathbb{R};\mathrm{L}^2_\omega(\mathbb{R}^n))} \| (-\Delta_\omega)^{\frac{1}{2}} u \|^\alpha_{\mathrm{L}^2(\mathbb{R};\mathrm{L}^2_\omega(\mathbb{R}^n))} \\&= C_1 \left\|  u \right\|^{1-\alpha}_{2,\mu} \left\| \nabla_x u \right\|^\alpha_{2,\mu}
\\& \le C_1 \left\|  u \right\|^{1-\alpha}_{2,\mu} \left\| \mathbb{D} u \right\|^\alpha_{2,\mu},
\end{align*}
To conclude, note that since $q' \le 2^\star$, we have $q' \in (2,r]$, and we write $\frac{1}{q'} = \frac{1-\theta}{2} + \frac{\theta}{r}$ with $\theta \in (0,1]$. Therefore, by H\"older's inequality and the previous inequalities, 
\begin{align*}
    \Vertu\Vert_{\mathrm{L}^{q'}_{\mu_{q'}}(\mathbb{R}^{n+1})} \le \Vertu\Vert^\theta_{\mathrm{L}^r(\mathbb{R};\mathrm{L}^{q'}_{\omega^{q'/2}}(\mathbb{R}^n))}  \| u \|^{1-\theta}_{\mathrm{L}^2(\mathbb{R};\mathrm{L}^{q'}_{\omega^{q'/2}}(\mathbb{R}^n))} \le C \Vert\mathbb{D} u\Vert^{\theta+\alpha(1-\theta)}_{2,\mu} \Vertu\Vert^{(1-\alpha)(1-\theta)}_{2,\mu},
\end{align*}
where $C = C([\omega]_{A_2}, [\omega]_{RH_{\frac{q'}{2}}}, q, n)$ is a constant. As $\frac{1}{q'} = \frac{1-\theta}{2} + \frac{\theta}{2}\alpha$, we have $\theta(1-\alpha)= 1-\frac{2}{q'}$. The result follows since
\begin{equation*}
    \theta+\alpha(1-\theta)=\alpha+\theta(1-\alpha)
    = n\left(\frac{1}{2}-\frac{1}{q'}\right)
      +2\left(\frac{1}{2}-\frac{1}{q'}\right)=\kappa .
\end{equation*}

\label{lem: bornes gaussiennes} 
Fix $\lambda\in\mathbb{C}$ with $\mathrm{Re}(\lambda^{-2})>0$ and $m\geq1$. Then the resolvent $\mathcal{E}_\lambda^m$ is an integral operator on $\mathrm{L}^2_\mu(\mathbb{R}^{n+1})$ with kernel $K_\lambda^m$ satisfying pointwise estimates in the following cases.
\begin{enumerate}[label=(\arabic*), leftmargin=*, itemsep=0pt]
    \item \textbf{Upper and lower Gaussian bounds on $\mathbb{R}_+\setminus\{0\}$:} If $\lambda>0$, then
    \begin{equation*}
    \frac{1}{C}\mathbf{1}_{t>s} \frac{e^{-\frac{t-s}{\lambda^2}}}{\lambda^{2m}}(t-s)^{(m-1)}\frac{e^{-c\frac{|x-y|^2}{t-s}}}{\omega_{t-s}(y)}\le K^m_\lambda(x,t;y,s)\le C \, \mathbf{1}_{t>s} \frac{e^{-\frac{t-s}{\lambda^2}}}{\lambda^{2m}}(t-s)^{(m-1)}\frac{e^{-\frac{1}{c}\frac{|x-y|^2}{t-s}}}{\omega_{t-s}(y)},
    \end{equation*}
    where $C>1$ and $c>1$ are constants depending only on $M$, $\nu$, $[\omega]_{A_2}$, $n$, and $m$.
    \item \textbf{Upper Gaussian bounds on sectors:} Fix $\mu\in(0,\nicefrac{\pi}{4})$ and set $S^+_{\mu}:=\{z\in\mathbb{C}\setminus\{0\}:|\arg(z)|<\mu\}$. If $\lambda\in S^+_{\mu}$, then
    \begin{equation*}
    |K^m_\lambda(x,t;y,s)|\le C \, \mathbf{1}_{t>s} \frac{e^{-c_\mu\frac{t-s}{|\lambda|^2}}}{|\lambda|^{2m}}(t-s)^{(m-1)}\frac{e^{-\frac{1}{c}\frac{|x-y|^2}{t-s}}}{\omega_{t-s}(y)},
    \end{equation*}
    where $C$ and $c$ are as in (1), and $c_\mu\in(0,1]$ depends only on $\mu$.
\end{enumerate}

<!-- lem-7: body_md -->
\label{lem2: sans m}
Let $1\le p \le q <\infty $ with $\frac{n+2}{p}-\frac{n+2}{q}<1$. Assume that there exists an integer $m\ge1$ such that the family $(\omega^{\frac{1}{2}} \mathcal{E}_\lambda^{m+1} \omega^{-\frac{1}{2}})_{\lambda>0}$ is $\mathrm{L}^p-\mathrm{L}^q$ bounded. <!-- proof-6: body_md -->
We divide the proof into several steps.
\newline
\paragraph{\textit{\textbf{Step 1:} For all $q \in (1,2]$, $\omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n) \, \Rightarrow \, p_-(\mathcal{H}) < q$.}} We fix $q\in (1,2]$ and assume that $\omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n)$. For $m \ge 1$, $u \in \mathrm{L}^2_\mu(\mathbb{R}^n)$, we set, for all $(x,t)\in \mathbb{R}^{n+1}$,
\begin{equation*}
    v(x,t):= \lambda^{-2m} \iint_{\mathbb{R}^{n+1}} \boldsymbol{1}_{t>s}\, e^{-\frac{t-s}{\lambda^2}} (t-s)^{(m-1)}\frac{e^{-\frac{1}{c}\frac{|x-y|^2}{t-s}}}{\omega_{t-s}(y)} |u(s,y)| \,\mathrm{d} \omega(y) \,\mathrm{d} s.
\end{equation*}
Using Fubini's theorem and then Minkowski's inequality, we have for all $t\in \mathbb{R}$,
\begin{align*}
    \Vertv(\cdot,t)\Vert_{\mathrm{L}^{q'}_{\omega^{q'/2}}(\mathbb{R}^n)} &\le\lambda^{-2m} \int_{\mathbb{R}} \boldsymbol{1}_{t>s}\, e^{-\frac{t-s}{\lambda^2}} (t-s)^{(m-1)}  \left\| \int_{\mathbb{R}^n}\frac{e^{-\frac{1}{c}\frac{|\cdot-y|^2}{t-s}}}{\omega_{t-s}(y)} |u(s,y)| \,\mathrm{d} \omega(y) \right\|_{\mathrm{L}^{q'}_{\omega^{q'/2}}(\mathbb{R}^n)} \!\!\!\!\!\!\!\! \,\mathrm{d} s
    \\& \le C \lambda^{-2m} \int_{\mathbb{R}} \boldsymbol{1}_{t>s}\, e^{-\frac{t-s}{\lambda^2}} (t-s)^{(m-1)} (t-s)^{\frac{n}{2}(\frac{1}{q'}-\frac{1}{2})}\Vertu(\cdot,s)\Vert_{2,\omega} \,\mathrm{d} s,
\end{align*}
where $C = C([ \omega ]_{A_2},[ \omega ]_{RH_{\frac{q'}{2}}},n,q)$, and we have used \cite[Lemma 3.7]{auscherbaadi2025hardy} in the last line, since $\omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n)$. Using Young's convolution inequality on $\mathbb{R}$ with $\ell>1$ such that $\frac{1}{2}+\frac{1}{\ell}=1+\frac{1}{q'}$, we deduce that 
\begin{align*}
    \Vertv\Vert_{\mathrm{L}^{q'}_{\mu_{q'}}(\mathbb{R}^{n+1})} &\le C \lambda^{-2m} \left ( \int_{0}^{+\infty} e^{-\ell\frac{t}{\lambda^2}} \, t^{\ell(m-1)} \, t^{\frac{\ell n}{2}(\frac{1}{q'}-\frac{1}{2})} \,\mathrm{d} t  \right )^{1/\ell} \Vertu\Vert_{2,\mu} \\&= C \left ( \int_{0}^{+\infty} e^{-\ell t} \, t^{\ell(m-1)} \, t^{\frac{\ell n}{2}(\frac{1}{q'}-\frac{1}{2})} \,\mathrm{d} t  \right )^{1/\ell} \lambda^{(n+2)\left( \frac{1}{q'}-\frac{1}{2} \right)} \, \Vertu\Vert_{2,\mu}.
\end{align*}
We then choose $m\ge 1$ so that the above integral is finite (integrability near $0$). By the upper bound in (1) of Lemma~\ref{lem: bornes gaussiennes} and the above computations, we now know that the family $\bigl( \omega^{\frac{1}{2}} \mathcal{E}^m_\lambda \, \omega^{-\frac{1}{2}} \bigr)_{\lambda>0}$ is $\mathrm{L}^2-\mathrm{L}^{q'}$ bounded. The same is true for the family $\bigl( \omega^{\frac{1}{2}} (\mathcal{E}^\star_\lambda)^m \, \omega^{-\frac{1}{2}} \bigr)_{\lambda>0}$. By adjointess, the family $\bigl( \omega^{\frac{1}{2}} \mathcal{E}^m_\lambda \, \omega^{-\frac{1}{2}} \bigr)_{\lambda>0}$ is then $\mathrm{L}^q-\mathrm{L}^2$ bounded. Fix $p\in (q,2]$. By interpolating with the $\mathrm{L}^2$ off-diagonal estimates for the family $(\omega^{\frac{1}{2}} \,\mathcal{E}^{m}_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ (see Proposition~\ref{prop: classical ODEs} and Lemma~\ref{lem: Toolbox}, Point~(2)), we deduce that this family satisfies $\mathrm{L}^p-\mathrm{L}^2$ off-diagonal estimates. Consequently, it is $\mathrm{L}^p$-bounded by Lemma~\ref{lem: Toolbox}, Point~(3). By Lemma~\ref{lem2: sans m}, the family $(\omega^{\frac{1}{2}} \,\mathcal{E}_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ is $\mathrm{L}^p$-bounded. As this true for all $p\in (q,2]$, then $p_-(\mathcal{H}) \le q$. Finally, to prove that $p_-(\mathcal{H}) < q$, we use the open-ended property of reverse H\"older classes (Proposition~\ref{prop:weights}, Point~(2)) to pick $\tilde{q}\in (1,q)$ such that $\omega \in RH_{\frac{\tilde{q}'}{2}}(\mathbb{R}^n)$. Then, by the same reasoning, $p_-(\mathcal{H}) \le \tilde{q}<q$.
\newline
\paragraph{\textit{\textbf{Step 2:} For all $q \in (1,2]$, $ p_-(\mathcal{H}) < q \, \Rightarrow \, \omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n)$.}} 
We fix $q \in (1,2]$ and assume that $p_-(\mathcal{H}) < q$. Let $\tilde{q} \in (p_-(\mathcal{H}),q)$. By Section~\ref{s42}, there exists $\varrho_\omega=\varrho_\omega([\omega]_{A_2},n) \in (1,2)$ such that the family $\bigl(\omega^{\frac{1}{2}} \mathcal{E}_\lambda \,\omega^{-\frac{1}{2}}\bigr)_{\lambda>0}$ is $\mathrm{L}^{\varrho_\omega}-\mathrm{L}^2$ bounded. Since it is also $\mathrm{L}^{\tilde{q}}$- and $\mathrm{L}^2$-bounded, Lemma~\ref{lem: gives m} yields the existence of $m \ge 1$ such that the family $\bigl( \omega^{\frac{1}{2}} \mathcal{E}^m_\lambda \, \omega^{-\frac{1}{2}} \bigr)_{\lambda>0}$ is $\mathrm{L}^q-\mathrm{L}^2$ bounded. By adjointess, $\bigl( \omega^{\frac{1}{2}} (\mathcal{E}^\star_\lambda)^m \, \omega^{-\frac{1}{2}} \bigr)_{\lambda>0}$ is $\mathrm{L}^2-\mathrm{L}^{q'}$ bounded. Since the conclusion will be the same, we assume for simplicity that $\bigl(\omega^{\frac{1}{2}} \mathcal{E}_\lambda^m \,\omega^{-\frac{1}{2}}\bigr)_{\lambda>0}$ is $\mathrm{L}^2-\mathrm{L}^{q'}$ bounded. Using the lower bound in (1) of Lemma~\ref{lem: bornes gaussiennes}, we have
\begin{align*}
    \lambda^{-2q'm} \iint_{\mathbb{R}^{n+1}} &\left ( \iint_{\mathbb{R}^{n+1}} \mathbf{1}_{t>s}
e^{-\frac{t-s}{\lambda^2}}
(t-s)^{(m-1)}\frac{e^{-c\frac{|x-y|^2}{t-s}}}{\omega_{t-s}(y)}  u(y,s) \,\mathrm{d} \omega(y) \,\mathrm{d} s \right)^{q'} \omega^{\frac{q'}{2}}(x) \,\mathrm{d} x \,\mathrm{d} t \\&\hspace{9cm}\le C^{q'} \lambda^{(n+2)\left( 1-\frac{q'}{2} \right)} \Vertu\Vert^{q'}_{2,\mu},
\end{align*}
for all nonnegative $u \in \mathrm{L}^2_\mu(\mathbb{R}^{n+1}) \cap \mathrm{L}^{q'}_{\mu_{q'}}(\mathbb{R}^{n+1})$. Let $Q \subset \mathbb{R}^n$ be a cube with sides parallel to the coordinate axes and side length $\lambda >0$. Taking $u=\boldsymbol{1}_{Q\times [0,\lambda^2]}$ yields
\begin{align*}
&\lambda ^{-2q'm}  \int_{2\lambda^2}^{3\lambda^2}  \int_Q  \left ( \int_{0}^{\lambda^2}  \int_Q e^{-\frac{t-s}{\lambda^2}}(t-s)^{(m-1)}\frac{e^{-c\frac{|x-y|^2}{t-s}}}{\omega_{t-s}(y)} \,\mathrm{d} \omega(y) \,\mathrm{d} s  \right)^{q'}  \omega^{\frac{q'}{2}}(x) \,\mathrm{d} x \,\mathrm{d} t \\&\hspace{9cm}\le C^{q'} \lambda^{(n+2)\left( 1-\frac{q'}{2} \right)}(\lambda^2 \omega(Q))^{\frac{q'}{2}}.
\end{align*}
Since $\lambda^2 \le t-s \le 3\lambda^2$, we obtain
\begin{align*}
  \int_Q \left (  \int_Q \frac{e^{-c\frac{|x-y|^2}{\lambda^2}}}{\omega_{3\lambda^2}(y)}  \,\mathrm{d} \omega(y)\right)^{q'} \omega^{\frac{q'}{2}}(x) \,\mathrm{d} x\le e^{3q'} C^{q'} \lambda^{n\left( 1-\frac{q'}{2} \right)} \omega(Q)^{\frac{q'}{2}}.
\end{align*}
Using the doubling and reverse doubling properties of $\omega$, together with norm equivalence on $\mathbb{R}^n$, we easily see that there exists a constant $\gamma=\gamma(n,[\omega]_{A_2},c)>0$ such that, for all $\lambda>0$,
\begin{equation*}
    \gamma \leq \inf_{x\in Q} \int_Q \frac{e^{-\frac{c|x-y|^2}{\lambda^2}}}{\omega_{3\lambda^2}(y)}  \,\mathrm{d} \omega(y).
\end{equation*}
Thus,
\begin{equation*}
    \int_{Q} \omega^{q'/2}(x) \, \mathrm d x \leq e^{3q'} \gamma^{-q'} C^{q'} |Q|^{1-\frac{q}{p}} \omega(Q)^{\frac{q'}{2}}. 
\end{equation*}
Therefore, $\omega \in RH_{\frac{q'}{2}}(\mathbb{R}^n)$. 
\newline
\paragraph{\textit{\textbf{Step 3:} $\omega \in RH_{\star}(\mathbb{R}^n)$ if and only if $p_-(\mathcal{H})=1$.}} 
If $\omega \in RH_{\star}(\mathbb{R}^n)$, then by \textbf{Step~1} we have that $p_-(\mathcal{H})<q$ for all $q\in(1,2]$. Therefore, $p_-(\mathcal{H})=1$. Conversely, if $p_-(\mathcal{H})=1$, then $\omega \in RH_{\star}(\mathbb{R}^n)$ by \textbf{Step~3}.
\newline
\paragraph{\textit{\textbf{Step 4:} The case $\omega \in RH_{\infty}(\mathbb{R}^n)$.}} We know that $RH_{\infty}(\mathbb{R}^n) \subset RH_{\star}(\mathbb{R}^n)$, but let us prove the desired results in a direct manner. Using the upper bounds in (1) of Lemma \ref{lem: bornes gaussiennes} with $m=1$, for every $\lambda>0$ and every $u \in \mathrm{L}^1(\mathbb{R}^{n+1}) \cap \mathrm{L}^2(\mathbb{R}^{n+1})$, Fubini's theorem yields
\begin{align*}
    &\Vert\omega^{\frac{1}{2}}\mathcal{E}_\lambda \, \omega^{-\frac{1}{2}} u\Vert_1 \leq \iint_{\mathbb{R}^{n+1}} \left( \iint_{R^{n+1}} |K^1_\lambda(x,t;y,s)| |u(y,s)| \omega^{\frac{1}{2}}(y) \, \mathrm{d}y \, \mathrm{d}s \right) \omega^{\frac{1}{2}}(x) \, \mathrm{d}x \, \mathrm{d}t
    \\& \le C \iint_{\mathbb{R}^{n+1}} \left( \iint_{R^{n+1}} \boldsymbol{1}_{t>s}\, \frac{e^{-\frac{t-s}{\lambda^2}}}{\lambda^2} \frac{e^{-\frac{1}{c}\frac{|x-y|^2}{t-s}}}{\sqrt{\omega_{t-s}(x)}\sqrt{\omega_{t-s}(y)}} |u(y,s)| \omega^{\frac{1}{2}}(y) \, \mathrm{d}y \, \mathrm{d}s \right) \omega^{\frac{1}{2}}(x) \, \mathrm{d}x \, \mathrm{d}t \\
    &=C \iint_{\mathbb{R}^{n+1}} |u(y,s)|  \left( \iint_{R^{n+1}} \boldsymbol{1}_{t>s} \, \frac{e^{-\frac{t-s}{\lambda^2}}}{\lambda^2} e^{-\frac{1}{c} \frac{|x-y|^2}{t-s}}  \left ( \frac{\omega(x)}{\omega_{t-s}(x)} \right )^{\frac{1}{2}}  \left ( \frac{\omega(y)}{\omega_{t-s}(y)} \right )^{\frac{1}{2}} \mathrm{d}x \, \mathrm{d}t \right) \, \mathrm{d}y \, \mathrm{d}s.
\end{align*}
Now, since $\omega \in RH_{\infty}(\mathbb{R}^n)$, for all $s<t$ and for a.e.\ $x \in \mathbb{R}^n$ and for a.e.\ $y \in \mathbb{R}^n$, we have
\begin{equation}\label{eq: RHinfini}
    \left ( \frac{\omega(x)}{\omega_{t-s}(x)} \right )^{\frac{1}{2}}  \left ( \frac{\omega(y)}{\omega_{t-s}(y)} \right )^{\frac{1}{2}} \le \frac{[\omega]_{RH_\infty}}{(t-s)^{n/2}}. 
\end{equation}
By using this and a change of variables in the above computations, we obtain
\begin{equation*}
    \Vert\omega^{\frac{1}{2}}\mathcal{E}_\lambda \, \omega^{-\frac{1}{2}} u\Vert_1 \leq C \, [\omega]_{RH_\infty} \left( {\pi}{c} \right)^{n/2} \Vertu\Vert_1. 
\end{equation*}
It follows that $p_-(\mathcal{H}) = 1$. More than that, we may reuse the upper bounds in (1) of Lemma \ref{lem: bornes gaussiennes} to show that there exists $m \ge 1$ such that the family $(\omega^{\frac{1}{2}}\mathcal{E}^m_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ satisfies $\mathrm{L}^1-\mathrm{L}^2$ off-diagonal estimates. Indeed, if $E,F \subset \mathbb{R}^{n+1}$ are measurable sets, $\,\mathrm{d} := \,\mathrm{d}(E,F)$ is the parabolic distance between $E$ and $F$, and $u \in \mathrm{L}^1(\mathbb{R}^{n+1}) \cap \mathrm{L}^2(\mathbb{R}^{n+1})$, then for all $m \ge 1$:
\begin{align*}
    &\Vert\boldsymbol{1}_F \, \omega^{\frac{1}{2}} \mathcal{E}_\lambda^m \, \omega^{-\frac{1}{2}} (\boldsymbol{1}_E u)\Vert^2_2 
    \le \iint_{\mathbb{R}^{n+1}} \boldsymbol{1}_F \left( \iint_{R^{n+1}}|K^m_\lambda(x,t;y,s)| |\boldsymbol{1}_E u|(y,s) \,  \omega^{\frac{1}{2}}(y) \, \mathrm{d}y \, \mathrm{d}s \right)^2 \mathrm{d}\omega(x) \, \mathrm{d}t
    \\& \le C^2 \iint_{\mathbb{R}^{n+1}}\left( \iint_{R^{n+1}} \mathbf{1}_{t>s} \, \boldsymbol{1}_{|x-y|^2+|t-s| - \,\mathrm{d}^2 \ge 0} \, \frac{e^{-\frac{t-s}{\lambda^2}}}{\lambda^{2m}}
(t-s)^{(m-1)} e^{-\frac{1}{c}\frac{|x-y|^2}{t-s}} |\boldsymbol{1}_E u|(y,s)
\right. \\
    &\hspace{7cm} \left. \left ( \frac{\omega(x)}{\omega_{t-s}(x)} \right )^{\frac{1}{2}}  \left ( \frac{\omega(y)}{\omega_{t-s}(y)} \right )^{\frac{1}{2}}   \, \mathrm{d}y \, \mathrm{d}s  \right)^2  \, \mathrm{d}x \, \mathrm{d}t.
\end{align*}
Using \eqref{eq: RHinfini} and Young's convolution inequality on $\mathbb{R}^{n+1}$ yields:
\begin{align*}
   \Vert\boldsymbol{1}_F \, \omega^{\frac{1}{2}} \mathcal{E}_\lambda^m \, \omega^{-\frac{1}{2}} (\boldsymbol{1}_E u)\Vert_2 &\le C \, [\omega]_{RH_\infty} \left( \iint_{R^{n+1}} \boldsymbol{1}_{t>0} \, \frac{e^{-2\frac{t}{\lambda^2}}}{\lambda^{4m}} \frac{e^{-\frac{2}{c} \frac{|x|^2}{t}}}{t^{n-2(m-1)}} \boldsymbol{1}_{|x|^2+t - \,\mathrm{d}^2\ge 0} \, \mathrm{d}x \, \mathrm{d}t \right)^{1/2} \Vert\boldsymbol{1}_E u\Vert_1
    \\& =  \frac{C \, [\omega]_{RH_\infty}}{\lambda^{2m}} \left( \int^{\infty}_{0}  \frac{e^{-2\frac{t}{\lambda^2}}}{t^{n/2-2(m-1)}}  \left ( \int_{|x|^2\ge \frac{\,\mathrm{d}^2}{t}-1} e^{-\frac{2}{c} |x|^2} \, \mathrm{d}x \right ) \, \mathrm{d}t \right)^{1/2} \Vert\boldsymbol{1}_E u\Vert_1
    \\& \leq e^{\frac{1}{2c}} \left( {\pi}{c} \right)^{n/4} \frac{C \, [\omega]_{RH_\infty}}{\lambda^{2m}} \left( \int^{\infty}_{0}  \frac{e^{-2\frac{t}{\lambda^2}} e^{-\frac{1}{c} \frac{\,\mathrm{d}^2}{t}}}{t^{n/2-2(m-1)}}  \, \mathrm{d}t \right)^{1/2} \Vert\boldsymbol{1}_E u\Vert_1
    \\&= C \, [\omega]_{RH_\infty} e^{\frac{1}{2c}} \left( {\pi}{c} \right)^{n/4} \lambda^{\frac{n+2}{2}-\frac{n+2}{1}} \left( \int^{\infty}_{0}  \frac{e^{-2t} e^{-\frac{1}{c} \frac{\,\mathrm{d}^2}{\lambda^2} \frac{1}{t}}}{t^{n/2-2(m-1)}}  \, \mathrm{d}t \right)^{1/2} \Vert\boldsymbol{1}_E u\Vert_1,
\end{align*}
where we have used a change of variables in $x$ in the second line and in $t$ in the last line, and wrote $e^{-\frac{2}{c}|x|^2}=e^{-\frac{1}{c}|x|^2}e^{-\frac{1}{c}|x|^2}$ in the second line to bound the expression by a Gaussian integral. We then choose $m \ge 1$ such that $n/2 - 2(m-1) < 1$, so that the above integral is finite (integrability near $0$ when $\,\mathrm{d} = 0$). The result then follows immediately by writing
\begin{align*}
    \int^{\infty}_{0}  \frac{e^{-2t} e^{-\frac{1}{c} \frac{\,\mathrm{d}^2}{\lambda^2} \frac{1}{t}}}{t^{n/2-2(m-1)}}  \, \mathrm{d}t &= \int^{\,\mathrm{d} /\lambda}_{0}  \frac{e^{-2t} e^{-\frac{1}{c} \frac{\,\mathrm{d}^2}{\lambda^2} \frac{1}{t}}}{t^{n/2-2(m-1)}}  \, \mathrm{d}t + \int^{\infty}_{\,\mathrm{d} /\lambda}  \frac{e^{-2t} e^{-\frac{1}{c} \frac{\,\mathrm{d}^2}{\lambda^2} \frac{1}{t}}}{t^{n/2-2(m-1)}}  \, \mathrm{d}t \\&\le  \left( \int^{\infty}_{0}  \frac{e^{-t} }{t^{n/2-2(m-1)}}  \, \mathrm{d}t \right) e^{-\frac{1}{c} \frac{\,\mathrm{d}}{\lambda}},
\end{align*}
where the last estimate follows from elementary inequalities.


Furthermore, using the same estimates as above, we show that the family $(\omega^{\frac{1}{2}}\mathcal{E}_\lambda \omega^{-\frac{1}{2}})_{\lambda>0}$ satisfies $\mathrm{L}^1$ off-diagonal estimates. By the same computations and using (2) of Lemma \ref{lem: bornes gaussiennes}, the family $(\omega^{\frac{1}{2}}\mathcal{E}_\lambda \omega^{-\frac{1}{2}})_{\lambda \in S^+_{\mu}}$, for any fixed $\mu \in (0,\nicefrac{\pi}{4})$, satisfies $\mathrm{L}^1$ off-diagonal estimates, with constants depending additionally on $\mu$. In particular, Lemma~\ref{lem: funct calculus} holds with $q=1$.

Finally, proving that the spatial gradient component $\omega^{\frac{1}{2}} \nabla_x \mathcal{H}^{-1/2}$ defines a bounded operator from $\mathrm{L}^1_{\mu_1}(\mathbb{R}^{n+1})$ to $\mathrm{L}^{1,\infty}(\mathbb{R}^{n+1})$ is equivalent to the weak-type $(1,1)$ boundedness of $\omega^{\frac{1}{2}}\nabla_x \mathcal{H}^{-1/2} \omega^{-\frac{1}{2}}$ from $\mathrm{L}^1(\mathbb{R}^{n+1})$ to $\mathrm{L}^{1,\infty}(\mathbb{R}^{n+1})$. To prove the latter, we use the above $\mathrm{L}^1-\mathrm{L}^2$ bounds for $(\omega^{\frac{1}{2}}\mathcal{E}^m_\lambda \, \omega^{-\frac{1}{2}})_{\lambda>0}$ and the $\mathrm{L}^1$ off-diagonal estimates of arbitrarily large order for the family $(\omega^{\frac{1}{2}}\mathcal{E}_\lambda \omega^{-\frac{1}{2}})_{\lambda \in S^+_{\mu}}$ for a fixed $\mu \in (0,\nicefrac{\pi}{4})$ (more precisely, Lemma~\ref{lem: funct calculus} with $q=1$ therein), together with the fact that the family $( \omega^{\frac{1}{2}} \lambda\nabla_x \mathcal{E}_\lambda \omega^{-\frac{1}{2}})_{\lambda>0}$ satisfies $\mathrm{L}^2$ off-diagonal estimates. These ingredients allow for the extrapolation of the weak-type $(1,1)$ boundedness of $\omega^{\frac{1}{2}}\nabla_x \mathcal{H}^{-1/2} \omega^{-\frac{1}{2}}$ from $\mathrm{L}^1(\mathbb{R}^{n+1})$ to $\mathrm{L}^{1,\infty}(\mathbb{R}^{n+1})$, exactly as in the classical elliptic case \cite[Sect.~7.1]{AEbook2023}. Indeed, one uses the extrapolation criterion of Theorem~\ref{thm: BK} in the classical homogeneous-scale case $N=2$, yielding the result directly, without resorting to any iterative argument. We omit the details.
