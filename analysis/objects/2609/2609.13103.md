<!-- theorem-1: body_md -->
\label{thm:main}
Let $\mathcal{A}$ be a $\mathbb{C}$-elliptic first-order homogeneous differential operator.
Then there exists $C_{d,\mathcal{A}}>0$ such that, for every open set
$\Omega\subset\mathbb{R}^d$ and every $u\in\mathrm{BV}^\mathcal{A}(\Omega)$ satisfying
\[
 \nabla_{\mathrm{ap}}u\in L^1(\Omega;V\otimes\mathbb{R}^d),
\]
one has $u\in\mathrm{BV}(\Omega;V)$ and
\[
 |Du|(\Omega)
 \le C_{d,\mathcal{A}}\left(
 |\mathcal{A} u|(\Omega)+\int_\Omega|\nabla_{\mathrm{ap}}u|\,dx
 \right).
\]
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
The operator $\mathcal{A}$ is said to be \textbf{$\mathbb{R}$-elliptic} if
\(
    \mathbb{A}[\xi]:V\to W
\)
is injective for every $\xi\in\mathbb{R}^d\setminus\{0\}$. The operator $\mathcal{A}$ is said to be \textbf{$\mathbb{C}$-elliptic} if its complexified symbol
\(
    \mathbb{A}[\xi]:V+iV\to W+iW
\)
is injective for every $\xi\in\mathbb{C}^d\setminus\{0\}$.
<!-- end definition-1 -->
<!-- remark-1: body_md -->
\label{rmk:CimpliesR}
{\rm Clearly $\mathbb{C}$-ellipticity implies $\mathbb{R}$-ellipticity.}
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lem:product}
Let $U\subset\mathbb{R}^d$ be open, $v\in \mathrm{BV}^{\mathcal{A}}(U)$, and $\eta\in C_c^1(U)$. Extend $\eta v$ by zero outside $U$. Then $\eta v\in \mathrm{BV}^{\mathcal{A}}(\mathbb{R}^d)$ and
\begin{equation}\label{eq:product}
 \mathcal{A}(\eta v)
 =\eta\,\mathcal{A} v+\mathbb{A}[\nabla\eta]v\,\mathcal{L}^{d}
 \qquad\text{in }\mathbb{R}^d.
\end{equation}
Consequently,
\begin{equation}\label{eq:productbound}
 |\mathcal{A}(\eta v)|(\mathbb{R}^d)
 \le \|\eta\|_{L^\infty(U)}|\mathcal{A} v|(U)
 +C_{\mathcal{A}}\|\nabla\eta\|_{L^\infty(U)}\|v\|_{\mathrm{L}^1(U)}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $\varphi\in C_c^1(\mathbb{R}^d;W)$, using the distributional adjoint $\mathcal{A}^*$ in \eqref{eqn:adjoint} and the Leibniz rule for $\mathcal{A}^*(\eta\varphi)$, a simple computation immediately gives
\[
 \langle \mathcal{A}(\eta v),\varphi\rangle
 =\langle \eta\mathcal{A} v,\varphi\rangle
 +\int_U \varphi\cdot\mathbb{A}[\nabla\eta]v\,dx.
\]
This is \eqref{eq:product}. Since the finite-dimensional bilinear map $(\xi,v)\mapsto\mathbb{A}[\xi]v$ satisfies $|\mathbb{A}[\xi]v|\le C_{\mathcal{A}}|\xi||v|$, by $\mathbb{C}$-ellipticity, estimate \eqref{eq:productbound} follows.
<!-- end proof-1 -->
<!-- theorem-2: body_md -->
\label{thm:criticaldiff}
Let $\Omega\subset\mathbb{R}^d$ be open, let $\mathcal{A}$ be an $\mathbb{R}$-elliptic first-order homogeneous differential operator, and let $u\in \mathrm{BV}^{\mathcal{A}}_{\mathrm{loc}}(\Omega)$. Then $u$ is
$L^1$-differentiable at $\mathcal{L}^{d}$-almost every point of $\Omega$.
In particular, there exists
$\nabla_{\mathrm{ap}}u(x)\in V\otimes\mathbb{R}^d$ such that
\[
    \fint_{B_r(x)}
    |u(y)-u(x)-\nabla_{\mathrm{ap}}u(x)(y-x)|\,dy
    =o(r)
\]
for $\mathcal{L}^{d}$-almost every $x\in\Omega$.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:PoincareA}
Assume that $\mathcal{A}$ is $\mathbb{C}$-elliptic. There exists $C_{d,\mathcal{A}}>0$ such that for every ball $B_r(x)\subset\mathbb{R}^d$ and every $u\in \mathrm{BV}^{\mathcal{A}}(B_r(x))$,
\begin{equation}\label{eq:PoincareA}
 \inf_{q\in\mathrm{Ker}(\mathcal{A})}\|u-q\|_{\mathrm{L}^1(B_r(x))}
 \le \|u-\Pi_{B_r(x)} u\|_{\mathrm{L}^1(B_r(x))}
 \le C_{d,\mathcal{A}}\,r\,|\mathcal{A} u|(B_r(x)).
\end{equation}
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:FDN}
Let $\mathcal{A}$ be a first-order homogeneous differential operator with constant coefficients. The following are equivalent:
\begin{enumerate}[label=\textup{(\alph*)}]
 \item $\mathcal{A}$ has finite-dimensional null-space $\mathrm{Ker}(\mathcal{A})$;
 \item $\mathcal{A}$ is $\mathbb{C}$-elliptic;
 \item there exists $\ell\in\mathbb N$ such that $\mathrm{Ker}(\mathcal{A})$ is contained in the space of $V$-valued polynomials of degree at most $\ell$.
\end{enumerate}

<!-- lemma-2: body_md -->
\label{lem:mollgrad}
Let $v\in \mathrm{L}^1_{\mathrm{loc}}(\mathbb{R}^d;V)$. Suppose that $v$ is $\mathrm{L}^1$-differentiable at $x$, namely
\[
 \fint_{B_r(x)}|v(y)-v(x)-\nabla_{\mathrm{ap}}v(x)(y-x)|\mathrm{d} y=o(r).
\]
Then
\begin{equation}\label{eq:mollgrad}
 \nabla(v*\varrho_\delta)(x)\longrightarrow \nabla_{\mathrm{ap}}v(x)
 \qquad\text{as }\delta\downarrow0.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Set
\[
 R(y):=v(y)-v(x)-\nabla_{\mathrm{ap}}v(x)(y-x).
\]
Then
\begin{align*}
\nabla(v*\varrho_\delta)(x)
&=
\int_{\mathbb{R}^d} v(y)\otimes \nabla\varrho_\delta(x-y)\,dy
\\
&=
\int_{\mathbb{R}^d}
\Bigl(
v(x)+\nabla_{\rm ap}v(x)(y-x)+R(y)
\Bigr)
\otimes \nabla\varrho_\delta(x-y)\,dy
\\
&=
v(x)\otimes
\int_{\mathbb{R}^d}\nabla\varrho_\delta(x-y)\,dy
\\
&\quad
+
\nabla_{\rm ap}v(x)
\int_{\mathbb{R}^d}
(y-x)\otimes\nabla\varrho_\delta(x-y)\,dy
\\
&\quad
+
\int_{\mathbb{R}^d}
R(y)\otimes\nabla\varrho_\delta(x-y)\,dy
\\
&=
\nabla_{\rm ap}v(x)
+
\int_{B_\delta(x)}
R(y)\otimes\nabla\varrho_\delta(x-y)\,dy.
\end{align*}
The last equality follows from integration by parts and the properties of the mollifier. Therefore
    
                                                       
                                                                       
    

\[
 |\nabla(v*\varrho_\delta)(x)-\nabla_{\mathrm{ap}}v(x)|
 \le C_\varrho\delta^{-d-1}
 \int_{B_\delta(x)}|R(y)|\,dy
 =o(1).
\]

<!-- proposition-1: body_md -->
\label{prop:globalweak}
Assume that $\mathcal{A}$ is elliptic. Then there exists $C_{d,\mathcal{A}}>0$
such that every compactly supported
$v\in BV^{\mathcal{A}}(\mathbb{R}^d)$ satisfies
\begin{equation}\label{eq:globalweak}
    \|\nabla_{\mathrm{ap}}v\|_{L^{1,\infty}(\mathbb{R}^d)}
    \le C_{d,\mathcal{A}}|\mathcal{A} v|(\mathbb{R}^d).
\end{equation}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
By the potential representation for elliptic operators
\cite[Lemma~2.1]{GR2019diff}, one can write
$v=K_{\mathcal{A}}*\mathcal{A} v$, where $K_{\mathcal{A}}$ is smooth away from the origin
and $(1-d)$-homogeneous. By \cite[Section~3.3, Theorem~3.4]{alberti2014p},
$\nabla_{\mathrm{ap}}v$ is the sum of the singular integral associated with
$\nabla K_{\mathcal{A}}$ and a bounded linear image of
$\frac{d\mathcal{A} v}{d\mathcal{L}^{d}}$. The former satisfies the classical weak-$(1,1)$
estimate, while the latter belongs to $L^1$ with norm bounded by
$C_{\mathcal{A}}|\mathcal{A} v|(\mathbb{R}^d)$. Hence
\[
\|\nabla_{\mathrm{ap}}v\|_{L^{1,\infty}(\mathbb{R}^d)}
\le C_{d,\mathcal{A}}|\mathcal{A} v|(\mathbb{R}^d).
\]
<!-- end proof-3 -->
<!-- proposition-2: body_md -->
\label{prop:local}
Assume that $\mathcal{A}$ is $\mathbb{C}$-elliptic. Let $u\in \mathrm{BV}^{\mathcal{A}}(B_r(x))$. Set
\[
 a_{r,x}:=\Pi_{B_r(x)} u\in\mathrm{Ker}(\mathcal{A}).
\]
Then
\begin{equation}\label{eq:localPoincare}
 \int_{B_r(x)}|u- a_{r,x}|\,dx
 \le C_{d,\mathcal{A}}r|\mathcal{A} u|(B_r(x)),
\end{equation}
and
\begin{equation}\label{eq:localweak}
 \|\nabla_{\mathrm{ap}}u-\nabla  a_{r,x}\|_{\mathrm{L}^{1,\infty}(B_{r/2}(x))}
 \le C_{d,\mathcal{A}}|\mathcal{A} u|(B_r(x)).
\end{equation}
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Estimate \eqref{eq:localPoincare} follows directly from \cref{thm:PoincareA}. Now choose a cutoff function $\eta\in C_c^\infty(B_r(x))$ such that
\[
 0\le\eta\le1,
 \qquad
 \eta\equiv1\text{ on }B_{r/2}(x),
 \qquad
 |\nabla\eta|\le C_dr^{-1}.
\]
Set
\begin{equation}
 w:=\mathbbmss{1}_{B_r(x)}\eta(u-a_{r,x}) \in \mathrm{BV}^{\mathcal{A}}(\mathbb{R}^d).
\end{equation}
Since $a_{r,x}\in\mathrm{Ker}(\mathcal{A})$, one has $\mathcal{A} a_{r,x}=0$. By \cref{lem:product} and \eqref{eq:localPoincare},
\begin{align}
 |\mathcal{A} w|(\mathbb{R}^d)
 &\le |\mathcal{A} u|(B_r(x))
 +C_{\mathcal{A}}r^{-1}\int_{B_r(x)}|u-a_{r,x}|\,dx \le C_{d,\mathcal{A}}|\mathcal{A} u|(B_r(x)).
 \label{eq:Awlocal}
\end{align}
Applying \cref{prop:globalweak} to $w$ gives
\[
 \|\nabla_{\mathrm{ap}}w\|_{\mathrm{L}^{1,\infty}(\mathbb{R}^d)}
 \le C_{d,\mathcal{A}}|\mathcal{A} u|(B_r(x)).
\]
On $B_{r/2}(x)$, $w=u-a_{r,x}$ and hence
\[
 \nabla_{\mathrm{ap}}w=\nabla_{\mathrm{ap}}u-\nabla a_{r,x}
 \qquad\text{a.e.}
\]
Restricting the weak estimate proves \eqref{eq:localweak}.

<!-- lemma-3: body_md -->
\label{lem:finitedim}
Assume that $\mathcal{A}$ is $\mathbb{C}$-elliptic. There exists $C_{d,\mathcal{A}}>0$ such that for every ball $B_r(x_0)$ and every $a\in\mathrm{Ker}(\mathcal{A})$,
\begin{equation}\label{eq:finitedim}
 \int_{B_r(x_0)}|\nabla a|\,dx
 \le C_{d,\mathcal{A}}\|\nabla a\|_{\mathrm{L}^{1,\infty}(B_r(x_0))}.
\end{equation}
The constant is independent of $x_0$ and $r$.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
By translation and scaling it is enough to prove the estimate on the unit ball. Indeed, if
\(
    \widetilde a(y):=a(x_0+ry),
\)
then $\widetilde a\in\mathrm{Ker}(\mathcal{A})$ (since $\mathcal{A}$ is a first-order homogeneous differential operator with constant coefficients) and
\(
    \nabla\widetilde a(y)=r\nabla a(x_0+ry).
\)
By \cref{thm:FDN}, the space
\[
 X:=\{\nabla a|_{B} :a\in\mathrm{Ker}(\mathcal{A})\}
\]
is finite-dimensional and consists of polynomial tensor fields. Since $\|\cdot \|_{\mathrm{L}^{1,\infty}(B)}$ is a quasi-norm and $\|\cdot\|_{\mathrm{L}^1(B)}$ is a norm (hence, in particular, a quasi-norm), and since all quasi-norms are equivalent on finite-dimensional spaces, we conclude.
<!-- end proof-5 -->
<!-- lemma-4: body_md -->
\label{lem:anchor}
Under the assumptions of \cref{prop:local},
\begin{equation}\label{eq:anchor}
 \int_{B_{r/2}(x)}|\nabla a_{r,x}|\,dx
 \le C_{d,\mathcal{A}}\left(
 \int_{B_{r/2}(x)}|\nabla_{\mathrm{ap}}u|\,dx
 +|\mathcal{A} u|(B_r(x))
 \right).
\end{equation}
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
Observe that, for two arbitrary measurable functions $f,g$, we trivially have
\[
 \{|f+g|>s\}\subset\{|f|>s/2\}\cup\{|g|>s/2\}.
\]
Thus
\begin{equation}\label{eq:weaktriangle}
 \|f+g\|_{\mathrm{L}^{1,\infty}(U)}
 \le2\bigl(\|f\|_{\mathrm{L}^{1,\infty}(U)}+\|g\|_{\mathrm{L}^{1,\infty}(U)}\bigr),
\end{equation}
Using \cref{lem:finitedim} on $B_{r/2}(x)$, then \eqref{eq:weaktriangle}, \eqref{eq:L1toWeak}, and \eqref{eq:localweak}, we obtain
\begin{align*}
 \int_{B_{r/2}(x)}|\nabla a_{r,x}|\,dx
 &\le C_{d,\mathcal{A}}\|\nabla a_{r,x}\|_{\mathrm{L}^{1,\infty}(B_{r/2}(x))}\\
 &\le C_{d,\mathcal{A}}\left(
 \|\nabla_{\mathrm{ap}}u\|_{\mathrm{L}^{1,\infty}(B_{r/2}(x))}
 +\|\nabla_{\mathrm{ap}}u-\nabla a_{r,x}\|_{\mathrm{L}^{1,\infty}(B_{r/2}(x))}
 \right)\\
 &\le C_{d,\mathcal{A}}\left(
 \int_{B_{r/2}(x)}|\nabla_{\mathrm{ap}}u|\,dx
 +|\mathcal{A} u|(B_r(x))
 \right).
\end{align*}
<!-- end proof-6 -->
<!-- proof-7: body_md -->
Fix $\Omega'\subset\subset\Omega$ and set
\[
 \delta:=\operatorname{dist}(\Omega',\partial\Omega)>0.
\]
Let $\varepsilon>0$ be small and consider the standard grid of pairwise disjoint cubes $Q_{4\varepsilon}(x_Q)$. Let $\mathcal Q_\varepsilon(\Omega')$ be the family of grid cubes meeting $\Omega'$. For each $Q_{4\varepsilon}(x_Q)\in\mathcal Q_\varepsilon(\Omega')$ consider the concentric balls $ B_{6\varepsilon\sqrt d\, }(x_Q)\supset B_{3\varepsilon\sqrt d\, }(x_Q)\supset Q_{6\varepsilon}(x_Q)$ . Note that for $\varepsilon<\varepsilon_0(\delta)$ small enough we can guarantee that $B_{6\varepsilon\sqrt d\,}(x_Q)\subset\subset \Omega$ for every $Q_{4\varepsilon}(x_Q)\in\mathcal Q_\varepsilon(\Omega')$.\\

Now, for each $Q_{4\varepsilon}(x_Q)$, apply \cref{prop:local} on $B_{6\varepsilon\sqrt d\,}(x_Q)$. Thus, setting
\[
 a_{\varepsilon,x_Q}:=\Pi_{B_{6\varepsilon\sqrt d}(x_Q)}u\in\mathrm{Ker}(\mathcal{A}),
\]
we have
\begin{equation}\label{eq:localPoincareQ}
 \int_{B_{6\varepsilon\sqrt d}(x_Q)}
 |u-a_{\varepsilon,x_Q}|\,dx
 \le
 C_{d,\mathcal{A}}\,\varepsilon\,
 |\mathcal{A} u|\bigl(B_{6\varepsilon\sqrt d}(x_Q)\bigr),
\end{equation}
and
\begin{equation}\label{eq:localweakQ}
 \|\nabla_{\mathrm{ap}}u-\nabla a_{\varepsilon,x_Q}\|_
 {\mathrm{L}^{1,\infty}(B_{3\varepsilon\sqrt d}(x_Q))}
 \le
 C_{d,\mathcal{A}}\,
 |\mathcal{A} u|\bigl(B_{6\varepsilon\sqrt d}(x_Q)\bigr).
\end{equation}
Take a mollifier $\{\varrho_{\varepsilon}\}_{\varepsilon>0}$ as in Subsection \ref{sbsct:Molly}. For $x\in\Omega_\varepsilon:=\{x\in\Omega:\operatorname{dist}(x,\partial\Omega)>\varepsilon\}$, set
\[
u_\varepsilon(x):=\int_{B_\varepsilon}u(x-z)\varrho_\varepsilon(z)\,dz.
\]
On $Q_{4\varepsilon}(x_Q)$ we decompose
\begin{equation}\label{eq:griddecomp}
 \nabla u_\varepsilon
 =\nabla\bigl((u-a_{\varepsilon,x_Q})*\varrho_\varepsilon\bigr)
 +\nabla(a_{\varepsilon,x_Q}*\varrho_\varepsilon).
\end{equation}
For the oscillation term, Fubini's theorem gives
\begin{align}
 \int_{Q_{4\varepsilon}(x_Q)}\left|\nabla\bigl((u-a_{ \varepsilon,x_Q})*\varrho_\varepsilon\bigr)(x)\right|\,dx
 &\le \int_{B_\varepsilon}|\nabla\varrho_\varepsilon(z)|
 \int_{Q_{4\varepsilon}(x_Q)}|u(x-z)-a_{ \varepsilon,x_Q}(x-z)|\,dx\,dz\notag\\
 &\le \|\nabla\varrho_\varepsilon\|_{\mathrm{L}^1}
 \int_{Q_{6\varepsilon}(x_Q)}|u-a_{\varepsilon,x_Q}|\,dx\notag\\
 &\le \frac{C_\varrho}{\varepsilon}
 \int_{B_{6\varepsilon\sqrt d\,}(x_Q)}|u-a_{\varepsilon,x_Q}|\,dx\notag\\
 &\le C_{d,\mathcal{A}}|\mathcal{A} u|(B_{6\varepsilon\sqrt d\,}(x_Q)).
 \label{eq:gridosc}
\end{align}
The second inequality follows from the fact that, since $\operatorname{spt}\varrho_\varepsilon\subset B_\varepsilon$, for every $x\in Q_{4\varepsilon}(x_Q)$ and $z\in\operatorname{spt}\varrho_\varepsilon$ one has $x-z\in Q_{6\varepsilon}(x_Q) $. In the last inequality, we applied \eqref{eq:localPoincareQ}.\\

For the null-space term we now use that $a_{\varepsilon,x_Q}$ is smooth. Hence
\[
 \nabla(a_{\varepsilon,x_Q}*\varrho_\varepsilon)=(\nabla a_{\varepsilon,x_Q})*\varrho_\varepsilon,
\]
and, again by Fubini,
\begin{align}
 \int_{Q_{4\varepsilon}(x_Q)}|\nabla(a_{\varepsilon,x_Q}*\varrho_\varepsilon)(x)|\,dx
 &\le \int_{B_\varepsilon}\varrho_\varepsilon(z)
 \int_{Q_{4\varepsilon}(x_Q)}|\nabla a_{\varepsilon,x_Q}(x-z)|\,dx\,dz\notag\\
 &\le \int_{Q_{6\varepsilon}(x_Q)}|\nabla a_{\varepsilon,x_Q}|\,dx
 \le \int_{B_{3\varepsilon\sqrt d\,}(x_Q)}|\nabla a_{\varepsilon,x_Q}|\,dx.
 \label{eq:gridpoly1}
\end{align}

Applying \cref{lem:anchor} on $B_{6\varepsilon\sqrt d\,}(x_Q)$ and using \eqref{eq:gridpoly1} gives
\begin{equation}\label{eq:gridpoly}
 \int_{Q_{4\varepsilon}(x_Q)}|\nabla(a_{\varepsilon ,x_Q}*\varrho_\varepsilon)|\,dx
 \le C_{d,\mathcal{A}}\left(
 \int_{B_{3\varepsilon\sqrt d\,}(x_Q)}|\nabla_{\mathrm{ap}}u|\,dx
 +|\mathcal{A} u|(B_{6\varepsilon\sqrt d\,}(x_Q))
 \right).
\end{equation}
Combining \eqref{eq:griddecomp}, \eqref{eq:gridosc}, and \eqref{eq:gridpoly} yields
\begin{equation}\label{eq:onecubefinal}
 \int_{Q_{4\varepsilon}(x_Q)}|\nabla u_\varepsilon|\,dx
 \le C_{d,\mathcal{A}}\left(
 \int_{B_{3\varepsilon\sqrt d\,}(x_Q)}|\nabla_{\mathrm{ap}}u|\,dx
 +|\mathcal{A} u|(B_{6\varepsilon\sqrt d\,}(x_Q))
 \right).
\end{equation}

The family $\{B_{6\varepsilon\sqrt d\,}(x_Q):Q_{4\varepsilon}(x_Q)\in\mathcal Q_\varepsilon(\Omega')\}$ has overlap bounded by a constant depending only on $d$. The same is true for the half-balls, while the cubes are essentially disjoint. Summing \eqref{eq:onecubefinal} over $Q_{4\varepsilon}(x_Q)\in\mathcal Q_\varepsilon(\Omega')$ yields
\begin{equation}\label{eq:gridglobal}
 \int_{\Omega'}|\nabla u_\varepsilon|\,dx
 \le C_{d,\mathcal{A}}\left(
 \int_\Omega|\nabla_{\mathrm{ap}}u|\,dx
 +|\mathcal{A} u|(\Omega)
 \right)
\end{equation}
for every $\varepsilon<\varepsilon_0(\delta)$ small enough. \\

By the standard properties of mollifiers, $u_\varepsilon\to u$ in $\mathrm{L}^1(\Omega';V)$. At the same time, the compactness theorem in $\mathrm{BV}(\Omega';V)$, together with
\eqref{eq:gridglobal}, yields, up to a subsequence,
$u_\varepsilon\to v$ in $\mathrm{L}^1(\Omega';V)$ for some
$v\in\mathrm{BV}(\Omega';V)$. Hence $u=v$ and therefore
$u\in\mathrm{BV}(\Omega';V)$. Moreover, by lower semicontinuity,
\[
 |Du|(\Omega')
 \le C_{d,\mathcal{A}}\left(
 \int_\Omega|\nabla_{\mathrm{ap}}u|\,dx
 +|\mathcal{A} u|(\Omega)
 \right).
\]
Since $\Omega'\subset\subset\Omega$ is arbitrary and the right-hand side is independent of $\Omega'$, an exhaustion of $\Omega$ by relatively compact open subsets yields $u\in\mathrm{BV}(\Omega;V)$ and
\[
 |Du|(\Omega)
 \le C_{d,\mathcal{A}}\left(
 \int_\Omega|\nabla_{\mathrm{ap}}u|\,dx
 +|\mathcal{A} u|(\Omega)
 \right).
\]
