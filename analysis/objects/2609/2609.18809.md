\label{lq-linf}

\label{prop:expansion_positivity}
Let $q>1$ and  $Q:= B_{R}(x_0) \times (t_1, t_2] \Subset \Omega_T$. Let $u$ be a local, nonnegative in $Q$, weak  super-solution to  \eqref{eq:PDE} in $\Omega_T$.
 Suppose that, for some $\alpha\in(0,1)$ and $k>0$, there holds
\[
    \bigl|
        \{u(\cdot,t_0)\geq k\}
        \cap B_\varrho(x_0)
    \bigr|
    \geq
    \alpha|B_\varrho|.
\]
Then, there exist constants $\Theta \equiv \Theta (N,s,q,\texttt{L}) >1$ and $\delta,\eta\in(0,1)$, depending only on $N,q,s,\textnormal{\texttt{L}}$ and on $\alpha$, such that either
$$       \Theta 
        \operatorname{Tail}^{\frac{1}{q}}\bigl(u_{-};Q\bigr)
    >
    \eta k,
    $$
or
\[
    u\geq\eta k
    \quad\mbox{a.e. in }\;
    B_{2\varrho}(x_0)\times
    \left(
        t_0+\frac12\delta k^{q-1}\varrho^{2s},
        t_0+\delta k^{q-1}\varrho^{2s}
    \right],
\]
provided
\[
    B_{8\varrho}(x_0)\times
    \left(
        t_0,
        t_0+\delta k^{q-1}(4\varrho)^{2s}
    \right]
    \subset Q.
\]
Moreover,
$ \delta\approx\alpha^{N+2s+1}$, $ \eta\approx\alpha^\beta$
for some $\beta>1$ depending only on the data.

<!-- definition-1: body_md -->
\label{def}
A measurable function $u: {\mathbb R}^N \times (0,T] \to {\mathbb R}$ satisfying
$$ u \in C_{\mathrm{loc}} (0, T; L^{q+1}_{\mathrm{loc}}( \Omega) ) \cap L^2_{\mathrm{loc}} (0, T; W^{s,2}_{\mathrm{loc}}( \Omega) ), $$
is a local, {weak super(sub)-solution} to the equation  \eqref{eq:PDE} in $\Omega_T$, if for every sub-interval $[t_1,t_2] \subset (0,T]$, every bounded open set $\tilde{\Omega} \Subset \Omega$, it holds
\begin{align} \label{ineq:L_1_tail_condition}
    \int_{t_1}^{t_2} \int_{{\mathbb R}^N} \frac{|u(x,t)|}{1+|x|^{N+2s}}\,\mathrm{d}x\,\mathrm{d}t < \infty
\end{align}
and 
\begin{align}  \label{eq}
& \int_{{\tilde{\Omega}}} |u|^{q-1}u \phi\,\mathrm{d}x \big |_{t_1}^{t_2}-
\int_{t_1}^{t_2}\int_{ {\tilde{\Omega}}}  |u|^{q-1}u \, \partial_t\phi\,\mathrm{d}x\,\mathrm{d}t \notag \\ & \qquad \qquad + \int_{t_1}^{t_2} \iint_{{\mathbb R}^N \times {\mathbb R}^N}(u(x,t)-u(y,t))(\phi(x,t)-\phi(y,t)) K(x,y,t)\,\mathrm{d}x\, \mathrm{d}y \,\mathrm{d}t \geq (\leq) \, 0
\end{align}
is satisfied for all nonnegative test functions
\[
\phi \in  L^2_{\mathrm{loc}} (0, T; W^{s,2}_0({\tilde{\Omega}}) ) \cap  W^{1,q+1}_{\mathrm{loc}} (0, T; L^{q+1}({\tilde{\Omega}})) .
\]
A function that is both a weak super- and sub-solution is called a weak solution.

Assume for simplicity $(x_0,\bar{\tau})=(0,0)$ and set $P_{-} = P_{-}(\varrho,x_0;\bar{\tau},\tau)$.  Consider $0<\sigma<\sigma'<1$, and set for brevity 
\begin{equation} \label{BB}
    B = B_\varrho(0), \ \check B = B_{\sigma\varrho}(0), \ \hat B = B_{\sigma'\varrho}(0),
\end{equation}
then $\check B\subset\hat B\subset B$. Let $\xi\in C^1_c(\hat{B})$ be a function such that \
\begin{align}\label{property_of_tst_fct} 
\xi = 1 \ \text{in $\check B$,} \ 0 \leq \xi \leq 1, \ |\nabla\xi| \lesssim \frac{1}{(\sigma'-\sigma)\varrho} \ \text{in ${\mathbb R}^N$}.
\end{align}
                                                                          
                             
                                                                         
                
                                                                                                                                   
In what follows, the function
\(
(u+\nu)^{\frac{1-q}{2}}\xi^2
\)
is understood to be extended by zero outside $B$.

Now, we apply Lemma \ref{mol} and obtain the following inequality:
\begin{equation}\label{tst1}
\begin{aligned}
    0 &\leq \tau^{\frac{1}{2}} \int_{\hat{B}}\xi^2(x)  \biggl(\int_0^{u(x,\tau)} \theta^{q-1}(\theta+\nu)^{\frac{1-q}{2}} \, \mathrm{d}\theta \biggl) \,\mathrm{d}x - \int_0^\tau  \frac{t^{-\frac{1}{2}}}{2} \int_{\hat{B}} \xi^2 \biggl(\int_0^{u(x,t)} \theta^{q-1} (\theta+\nu)^{\frac{1-q}{2}} \, \mathrm{d}\theta \biggl) \,\mathrm{d}x\,\mathrm{d}t \\
    &\quad 
    +  \gamma \int_0^\tau t^{\frac{1}{2}} \iint_{{\mathbb R}^N \times {\mathbb R}^N} {(u(x,t)-u(y,t))}  [(u(x,t)+\nu)^{\frac{1-q}{2}}\xi^{2}(x)-(u(y,t)+\nu)^{\frac{1-q}{2}}\xi^{2}(y)]K(x,y,t) \,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t \\
    &=: I_{1} + I_{2},
\end{aligned}
\end{equation}
where we denote by $I_1$ the sum of the first two terms on the right-hand side and by $I_2$ the remaining nonlocal term.
We first estimate \(I_1\) from above. Since its second term is nonpositive, we may discard it. Using the fact that $q>1$ and $\nu >0$, we obtain that
\begin{align} \label{t1}
     \int_{0}^{u} \theta^{q-1} (\theta+\nu)^{\frac{1-q}{2}} \,\mathrm{d}\theta \leq \gamma u^{\frac{q+1}{2}}.
\end{align}
Then, by  \eqref{t1}, Hölder's inequality with exponents $\left (\frac{2q}{q+1}, \frac{2q}{q-1} \right )$, the fact that $\xi \leq 1$ and \eqref{lambda}, we estimate $I_1$ as
\begin{align} \label{i1}
    I_{1} &\leq \gamma  \tau^{\frac{1}{2}} \int_{B} (u(x,\tau)+\nu)^{\frac{q+1}{2}} \xi^{2}(x) \,\mathrm{d}x \notag \\
    &\leq \gamma  \tau^{\frac{1}{2}} \Bigg[ \sup_{t \in [0,\tau]} \int_{B} (u(x,t)+\nu)^{q} \,\mathrm{d}x \Bigg]^{\frac{q+1}{2q}} \vert B \vert^{\frac{q-1}{2q}} \notag \\
    &\leq \gamma \varrho^{s} \Bigl( \frac{\tau}{\varrho^{\lambda}} \Bigl)^{\frac{1}{2}} \Bigg[ \sup_{t \in [0,\tau]} \int_{B} u^{q}(x,t) \,\mathrm{d}x + \Bigl( \frac{\tau}{\varrho^{\lambda}} \Bigl)^{\frac{q}{q-1}} \Bigg]^{\frac{q+1}{2q}}.
\end{align}
Now, we proceed with the diffusion term $I_{2}$. By symmetry, we are able to rewrite this term into the following two parts
\begin{align} \label{i2}
    I_{2} &= 2 \gamma \int_0^\tau \iint_{A_{t}^{+}} {(u(x,t)-u(y,t))} t^{\frac{1}{2}} [(u(x,t)+\nu)^{\frac{1-q}{2}}\xi^{2}(x)-(u(y,t)+\nu)^{\frac{1-q}{2}}\xi^{2}(y)]K(x,y,t) \,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t  \notag \\
    &\qquad 
    +2 \gamma \int_0^\tau \iint_{A_{t}^{-}} {(u(x,t)-u(y,t))} t^{\frac{1}{2}} [(u(x,t)+\nu)^{\frac{1-q}{2}}\xi^{2}(x)-(u(y,t)+\nu)^{\frac{1-q}{2}}\xi^{2}(y)]K(x,y,t) \,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t \notag \\
    &=: I_{3} + I_{4},
\end{align}
where
\begin{align*}
    A_{t}^{+} &:= \big\{(x,y)\in {\mathbb R}^N \times {\mathbb R}^N :\,u(x,t)>u(y,t),\, u(y,t) \geq 0 \big\}, \\
    A_{t}^{-} &:= \big\{(x,y)\in {\mathbb R}^N \times {\mathbb R}^N :\,u(x,t)>u(y,t) ,\, u(y,t) < 0 \big\}.
\end{align*}
Therefore, for $I_3$ we obtain
\begin{align} \label{i3}
    I_{3} &\leq  +2\gamma \int_0^\tau \iint_{A_{t}^{+} \cap \{ \xi(x) \leq \xi(y) \}}
    \frac{u(x,t)-u(y,t)}{|x-y|^{N+2s}} t^{\frac{1}{2}}

    +2\gamma \int_0^\tau \iint_{A_{t}^{+} \cap \{ \xi(x) > \xi(y) \}}
    \frac{u(x,t)-u(y,t)}{|x-y|^{N+2s}} t^{\frac{1}{2}}
    [(u(x,t)+\nu)^{\frac{1-q}{2}} -(u(y,t)+\nu)^{\frac{1-q}{2}}]
    \xi^{2}(x) \,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t \notag \\

\begin{align*}

In order to estimate $J_{5}$ further we apply \eqref{tst} with 
\begin{equation} \label{isigma}
    (\varrho, \sigma)=\left (\varrho_{n+1}, \frac{{\hat{\varrho}}_{n}}{\varrho_{n+1}} \right ).
\end{equation}
This yields $\sigma \in (0,1)$ and 
\begin{align*}
    1- \sigma \gtrsim \frac{ \varrho_{n+1} -  \hat{\varrho}_{n}}{ \varrho_{n+1}} \gtrsim \frac{1}{2^{n}}.
\end{align*}
Therefore, using also the fact that $\varrho \leq \varrho_{n+1} \leq 2\varrho$, we obtain
\begin{align}\label{ineq:estimate_of_J_6}
    J_{5} &\leq \gamma \varrho_{n+1}^{s} 2^{n(N+2s)}   \left[ \sup_{t \in [0,\tau]} \int_{B_{\varrho_{n+1}}} u^{q}(x,t) \,\mathrm{d}x + \left( \frac{\tau}{\varrho_{n+1}^{\lambda}} \right)^{\frac{q}{q-1}} \right]^{\frac{q+1}{2q}} \left( \frac{\tau}{\varrho_{n+1}^{\lambda}} \right)^{\frac{1}{2}} P_{-} \nonumber \\
    &\leq \gamma b^{n} \varrho^{s} \left[ S_{n+1} + \left( \frac{\tau}{\varrho^{\lambda}} \right)^{\frac{q}{q-1}} \right]^{\frac{q+1}{2q}} \left( \frac{\tau}{\varrho^{\lambda}} \right)^{\frac{1}{2}} P_{-},
\end{align}
where $P_{-}$ is defined according to \eqref{PP}, and $b>1$ depending only on the data. Next, we estimate $J_{6}$ by
\begin{align}\label{ineq:estimate_of_J_7}
    J_{6} &\leq \Bigg[ \int_{0}^{\tau} t^{-\frac{1}{2}} \,\mathrm{d}t \Bigg] \Bigg[ \sup_{t \in [0,\tau]} \int_{B_{n+1}} ( u(x,t) + \nu )^{\frac{q+1}{2}} \,\mathrm{d}x \Bigg] \Bigg[ \sup_{x \in B_{n+1}} \int_{B_{n+1}} \frac{1}{\vert x-y \vert^{N+2s-2}} \,\mathrm{d}y \Bigg] \nonumber\\
    &\leq \gamma \tau^{\frac{1}{2}} \Bigg[ \sup_{t \in [0,\tau]} \int_{B_{n+1}} ( u^{q}(x,t) + \nu^{q} ) \,\mathrm{d}x \Bigg]^{\frac{q+1}{2q}} \vert B_{n+1} \vert^{\frac{q-1}{2q}} \Bigg[ \int_{B_{4 \varrho}  } \frac{1}{\vert z \vert^{N+2s-2}} \,\mathrm{d}z \Bigg] \nonumber\\

    &\leq \gamma \varrho^{2-s} \Bigg[ S_{n+1} + \Bigl( \frac{\tau}{\varrho^{\lambda}} \Bigl)^{\frac{q}{q-1}} \Bigg]^{\frac{q+1}{2q}} \Bigl( \frac{\tau}{\varrho^{\lambda}} \Bigl)^{\frac{1}{2}},

where we used H\"older's inequality with exponents $\left (\frac{2q}{q+1}, \frac{2q}{q-1} \right )$.

\begin{align*}

\end{align*}

\begin{align}\label{ineq:estimate_of_J_3}

\end{align}
where $\gamma_{\varepsilon} \to \infty$ as $\varepsilon \to 0$ and $b>1$ depending only on the data. For the estimate of $J_{3}$ in  \eqref{jj1}, using \eqref{xi}, we easily observe that
\begin{align*}

\begin{align} \label{th}

\end{align}
Therefore, by  \eqref{th}, Hölder's inequality and scaled Young's inequality with exponents $\left ( q, \frac{q}{q-1}\right )$, $\varrho_{n+1} \approx \varrho$, we obtain 
\begin{align}\label{ineq:estimate_of_J_4}

    &\leq \gamma\frac{ b^{n} \tau}{\varrho^{\lambda}} S_{n+1}^{\frac{1}{q}} \leq \varepsilon S_{n+1} + \gamma_{\varepsilon} b^{n} \Bigl( \frac{\tau}{\varrho^{\lambda}} \Bigl)^{\frac{q}{q-1}},
\end{align}
where $\varepsilon>0$ will be determined later.
Here, we also have that $\gamma_{\varepsilon} \to \infty$ as $\varepsilon \to 0$ and $b>1$ depending only on the data. The term $J_{4}$ can be estimated similarly, recalling the definition in \eqref{PP}, using \eqref{xi}, \eqref{th} and $\check{\varrho}_n \approx \varrho$, we get

\qquad\text{almost everywhere in }Q_{\frac{\varrho}{2}}(\theta).

\frac{
\left|
\bigl(u_{\bar h}(x,t)+\nu\bigr)^{\frac{1-q}{2}}
-
\bigl(u_{\bar h}(y,t)+\nu\bigr)^{\frac{1-q}{2}}
\right|^2
}{
|x-y|^{N+2s}
}
\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t \notag
\\
& \qquad \quad+
\gamma\int_0^\tau t \psi_{\varepsilon}(t) \iint_{B\times B}
\left(u_{\bar h}(y,t)+\nu\right)^{1-q}
\frac{
|\xi^2(x)-\xi^2(y)|^2
}{
|x-y|^{N+2s}
}
\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t \notag
\\
& \qquad =:H_8+H_9.
\end{align}
To estimate $H_8$, we observe that the map
\[
r\mapsto (r+\nu)^{\frac{1-q}{2}}
\]
is Lipschitz continuous in $[0,+\infty)$. Indeed, since $u_{\bar h}\geq0$, the mean value theorem yields
\[
\left|
\bigl(u_{\bar h}(x,t)+\nu\bigr)^{\frac{1-q}{2}}
-
\bigl(u_{\bar h}(y,t)+\nu\bigr)^{\frac{1-q}{2}}
\right|
\leq
\frac{q-1}{2}\nu^{-\frac{q+1}{2}}
\left|u_{\bar h}(x,t)-u_{\bar h}(y,t)\right|.
\]
Therefore, using $t \leq \tau$, $\psi_{\varepsilon}(t) \leq 1$, the inequality above and Proposition \ref{con} \ref{con4}, we obtain
\begin{align*}
H_8
\leq
\frac{\gamma \tau}{\nu^{q+1}}
\int_0^\tau \iint_{B\times B}
\frac{
\left|u_{\bar h}(x,t)-u_{\bar h}(y,t)\right|^2
}{
|x-y|^{N+2s}
}
\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t
\leq
\frac{\gamma \tau}{\nu^{q+1}}
\|u\|_{L^2(0,T_*;W^{s,2}(B))}^2.
\end{align*}
To estimate $H_9$, we use the inequalities
\[
|\xi^2(x)-\xi^2(y)|
\leq 2|\xi(x)-\xi(y)|
\leq
\frac{4}{(\sigma'-\sigma)\varrho}|x-y|
\]
and
\[
\bigl(u_{\bar h}(y,t)+\nu\bigr)^{1-q}
\leq
\frac{\bigl(u_{\bar h}(y,t)+\nu\bigr)^2}{\nu^{q+1}}.
\]
Therefore, recalling that $t \leq \tau$, $\psi_{\varepsilon}(t) \leq 1$ and using
Proposition \ref{con} \ref{con2} with $r=2$, we obtain
\begin{align*}
H_9
&\leq
\frac{\gamma \tau}{(\sigma'-\sigma)^2\varrho^2\nu^{q+1}}
\int_0^\tau \iint_{B\times B}
\bigl(u_{\bar h}(y,t)+\nu\bigr)^2
\frac{1}{|x-y|^{N+2s-2}}
\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t
\\
&\leq
\frac{\gamma \tau}{(\sigma'-\sigma)^2\varrho^2\nu^{q+1}}
\left[
\int_0^\tau \int_B
\bigl(u_{\bar h}(y,t)+\nu\bigr)^2
\,\mathrm{d}y\,\mathrm{d}t
\right]
\left[
\sup_{y\in B}
\int_B
\frac{\mathrm{d}x}{|x-y|^{N+2s-2}}
\right]
\\
&\leq
\frac{\gamma \tau}{(\sigma'-\sigma)^2\varrho^2\nu^{q+1}}
\left[
\int_0^\tau \int_B
u_{\bar h}^2(y,t)\,\mathrm{d}y\,\mathrm{d}t
+
\nu^2\varrho^N \tau
\right]
\left[
\int_{B_{2\varrho}(0)}
\frac{dz}{|z|^{N+2s-2}}
\right]
\\
&\leq
\frac{\gamma \tau}{(\sigma'-\sigma)^2\varrho^{2s}\nu^{q+1}}
\left[
\|u\|_{L^2(B\times(0,T_*))}^2
+
\nu^2\varrho^N \tau
\right].
\end{align*}
Combining the estimates for $H_8$ and $H_9$, we obtain from \eqref{ig} that $(F_{h,\varepsilon})_h$ is bounded in
\[
L^2\bigl(B\times B\times(0,\tau);K(x,y,t)\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t\bigr).
\]
By \eqref{ae}, up to a subsequence,
\[
F_{h, \varepsilon} \rightharpoonup F_{\varepsilon}
\quad\text{weakly in}\quad
L^2\bigl(B\times B\times(0,\tau);K(x,y,t)\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t\bigr).
\]
Moreover, by \eqref{mol3} and \eqref{k}, $G$ belongs to the same space. Therefore, up to a subsequence, the term $H_6$ in \eqref{h6h7} satisfies
\begin{equation}\label{mol5}
H_6\longrightarrow0
\qquad\text{as }h\to0^+.
\end{equation}
Next, we focus on $H_7$ in \eqref{h6h7}. First note that, for every
$x\in\hat B$ and $y\in B^c$, we have
\begin{equation}\label{ineq:distance_H7}
|x-y|
\geq |y|-|x|
\geq (1-\sigma')|y|.
\end{equation}
Recalling that $0\leq\xi\leq1$, $t \leq \tau$, $\psi_{\varepsilon}(t) \leq 1$ using \eqref{ineq:distance_H7}, the upper bound in \eqref{k}, and the inequality
\[
|G(x,y,t)|
\leq u(x,t)+|u(y,t)|,
\]
we obtain
\begin{align}\label{ineq:estimate_H7}
|H_7|
\leq{}&
\frac{\gamma \tau^{\frac12}}{(1-\sigma')^{N+2s}}
\int_0^\tau \iint_{\hat B\times B^c}
\frac{u(x,t)}{|y|^{N+2s}}
\left|
\bigl(u_{\bar h}(x,t)+\nu\bigr)^{\frac{1-q}{2}}
-
\bigl(u(x,t)+\nu\bigr)^{\frac{1-q}{2}}
\right|
\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t
\nonumber\\
&+
\frac{\gamma \tau^{\frac12}}{(1-\sigma')^{N+2s}}
\int_0^\tau \iint_{\hat B\times B^c}
\frac{|u(y,t)|}{|y|^{N+2s}}
\left|
\bigl(u_{\bar h}(x,t)+\nu\bigr)^{\frac{1-q}{2}}
-
\bigl(u(x,t)+\nu\bigr)^{\frac{1-q}{2}}
\right|
\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t
\nonumber\\
=:{}&
\frac{\gamma \tau^{\frac12}}{(1-\sigma')^{N+2s}}
\bigl(H_{10}+H_{11}\bigr).
\end{align}
We now estimate $H_{10}$. By the mean value theorem, we have
\begin{equation}\label{ineq:lagrange_H10}
\left|
\bigl(u_{\bar h}(x,t)+\nu\bigr)^{\frac{1-q}{2}}
-
\bigl(u(x,t)+\nu\bigr)^{\frac{1-q}{2}}
\right|
\leq
\frac{q-1}{2\nu^{\frac{q+1}{2}}}
\left|u_{\bar h}(x,t)-u(x,t)\right|.
\end{equation}
Therefore, applying H\"older's inequality with exponents $(2,2)$ and using \eqref{ineq:lagrange_H10}, we obtain
\begin{align}\label{ineq:estimate_H10}
H_{10}
&\leq
\left[
\int_0^\tau \int_{\hat B}u^2(x,t)\,\mathrm{d}x\,\mathrm{d}t
\right]^{\frac12}
\left[
\int_0^\tau \int_{\hat B}
\left|
\bigl(u_{\bar h}(x,t)+\nu\bigr)^{\frac{1-q}{2}}
-
\bigl(u(x,t)+\nu\bigr)^{\frac{1-q}{2}}
\right|^2
\,\mathrm{d}x\,\mathrm{d}t
\right]^{\frac12}
\int_{B^c}\frac{\mathrm{d}y}{|y|^{N+2s}}
\nonumber\\
&\leq
\frac{\gamma}{\nu^{\frac{q+1}{2}}\varrho^{2s}}
\|u\|_{L^2(\hat B\times(0,\tau))}
\|u_{\bar h}-u\|_{L^2(\hat B\times(0,\tau))}.
\end{align}
Consequently, by Proposition \ref{con} \ref{con2} with $r=2$, we have
\[
H_{10}\longrightarrow0
\qquad\text{as }h\to0^+.
\]
As regards $H_{11}$, by \eqref{ineq:lagrange_H10}, we obtain
\begin{align}\label{ineq:estimate_H11}
H_{11}
&\leq
\frac{\gamma}{\nu^{\frac{q+1}{2}}}
\int_0^\tau
\left[
\int_{\hat B}
|u_{\bar h}(x,t)-u(x,t)|\,\mathrm{d}x 
\right]
\left[
\int_{B^c}
\frac{|u(y,t)|}{|y|^{N+2s}}\,\mathrm{d}y 
\right]
\,\mathrm{d}t
\nonumber\\
&\leq
\frac{\gamma}{\nu^{\frac{q+1}{2}}}
\left[
\sup_{t\in[0,\tau]}
\|u_{\bar h}(\cdot,t)-u(\cdot,t)\|_{L^1(\hat B)}
\right]
\textnormal{Tail}\bigl(|u|;B_\varrho\times(0,\tau]\bigr).
\end{align}
By \cite[Lemma 2.6]{FDF} and \eqref{ineq:L_1_tail_condition}, it follows from
\eqref{ineq:estimate_H11} that
\[
H_{11}\longrightarrow0
\qquad\text{as }h\to0^+.
\]
Therefore, $H_7$ defined in \eqref{h6h7} satisfies
\begin{equation}\label{mol6}
H_7\longrightarrow0
\qquad\text{as }h\to0^+.
\end{equation}
Combining \eqref{mol5} and \eqref{mol6}, we obtain in \eqref{h6h7}, up to a subsequence,
\begin{equation} \label{h2h2h2}
    \int_0^\tau \iint_{{\mathbb R}^N\times{\mathbb R}^N}
G(x,y,t)\bigl[F_{h, \varepsilon}(x,y,t)-F_{\varepsilon}(x,y,t)\bigr]
K(x,y,t)\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t
\longrightarrow0,
\end{equation}
as $h\to0^+$. Hence, recalling $H_2$ in \eqref{mol1}, from \eqref{h2h2h2}, we get
\begin{equation}\label{mol7}
\lim_{h\to0^+}H_2
=
\int_0^\tau \iint_{{\mathbb R}^N\times{\mathbb R}^N}
G(x,y,t)F_{\varepsilon}(x,y,t)K(x,y,t)\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t,
\end{equation}
up to a subsequence. Letting $\varepsilon\to0^+$ and using the dominated convergence theorem, we obtain
\begin{align} \label{h2h2}
\lim_{\varepsilon\to0^+} \lim_{h\to0^+} H_2
={}&
\int_{0}^{\tau}
\iint_{{\mathbb R}^N\times{\mathbb R}^N}
G(x,y,t)F(x,y,t)K(x,y,t)
\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}t.
\end{align}
Finally, combining \eqref{mol2} and \eqref{h2h2} and passing to the limit in \eqref{mol1}, we conclude the proof.
