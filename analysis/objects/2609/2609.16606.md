<!-- definition-1: body_md -->
\label{def:tsk}
    For each  $k\in [d]$, let $\kappa_k$ be a univariate kernel. A total sensitivity kernel (TSK) is defined by
    \begin{equation}\label{eq:tsk}
        \kappa_{\boldsymbol\Sigma}(\boldsymbol x,\boldsymbol y) = \prod_{k=1}^d\big(1-\Sigma_k + \Sigma_k\kappa_k(x_k,y_k)\big).
    \end{equation}
    There, we denote the TSK factors $\boldsymbol\Sigma=\left(
   \Sigma_1,  \dots , \Sigma_d 
    \right)$, with $\Sigma_k\in(0,1).$ Let $\mathcal{H}_{\boldsymbol \Sigma}$ be the RKHS associated to $\kappa_{\boldsymbol \Sigma}$, with norm $\|\cdot\|_{\boldsymbol\Sigma}$.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:norm_sum}
    Let $\kappa_{{\boldsymbol  \Sigma}}$ be a TSK. For $f\in\mathcal{H}_{\boldsymbol\Sigma}$, with corresponding RKHS norm $\|\cdot\|_{\boldsymbol  \Sigma}$, then
    \begin{equation}\label{equ:ker_sum}
    \|f\|_{{{\boldsymbol  \Sigma}}}^2=\min_{f=\sum_{\boldsymbol u} f_{\boldsymbol u}} \sum_{\boldsymbol u\subseteq [d]}\frac{1}{\gamma_{\boldsymbol u}}\|f_{\boldsymbol u}\|_{\mathcal{H}_{\boldsymbol u}}^2, \quad f_{\boldsymbol u}\in\mathcal H_{\boldsymbol u}.
    \end{equation}
    Each RKHS $\mathcal H_{\boldsymbol u}$ corresponds to the kernel $\kappa_{{\boldsymbol u}}=\prod_{k\in\boldsymbol u}\kappa_k$ with attached norm $\|\cdot\|_{\mathcal H_{\boldsymbol u}}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Each $\gamma_{\boldsymbol u}\kappa_{\boldsymbol u}$ is a kernel with an associated RKHS, which we denote $\mathcal H_{\gamma,\boldsymbol u}$ with corresponding norm $\|\cdot\|_{\gamma,\boldsymbol u}$. According to~\cite[Part I.6]{AronszajnReview} about the sum of reproducing kernels, the norm of $f\in\mathcal H_{\boldsymbol \Sigma}$ is 
    \begin{equation*}
    \|f\|_{{\boldsymbol \Sigma}}^2=\min_{f=\sum_{\boldsymbol u} f_{\boldsymbol u}} \sum_{\boldsymbol u\subseteq [d]}\|f_{\boldsymbol u}\|^2_{\mathcal H_{\gamma,\boldsymbol u}}, \quad f_{\boldsymbol u}\in\mathcal H_{\gamma,\boldsymbol u}.
    \end{equation*}
    Yet, note that the class of functions comprising $\mathcal H_{\boldsymbol u}$ is the same as that of $\mathcal H_{\gamma,\boldsymbol u}$. If $f\in\mathcal H_{\gamma,\boldsymbol u}$, then we can express it as
    \begin{equation*}
    f = \sum_{i=1}^M\alpha_i(\gamma_{\boldsymbol u}\kappa_{\boldsymbol u}(\boldsymbol x^i,\cdot)).
    \end{equation*}
    It is clear that $f=\sum_{i=1}^M\beta_i\kappa_{\boldsymbol u}(\boldsymbol x^i,\cdot)$, with $\beta_i=\gamma_{\boldsymbol u}\alpha_i$ so that $f\in\mathcal H_{\boldsymbol u}$. The norm of $f$ in $\mathcal H_{\gamma,\boldsymbol u}$ is 
    \begin{equation*}
    \|f\|^2_{\mathcal H_{\gamma,\boldsymbol u}}=\sum_{i,\ell=1}^M\alpha_i\alpha_\ell(\gamma_{\boldsymbol u}\kappa_{\boldsymbol u}(\boldsymbol{x}^i,\boldsymbol{x}^{\ell})) = \frac{1}{\gamma_{\boldsymbol u}} \sum_{i,\ell=1}^M\beta_i\beta_\ell\kappa_{\boldsymbol u}(\boldsymbol{x}^i,\boldsymbol{x}^\ell) = \frac{1}{\gamma_{\boldsymbol u}}\|f\|^2_{\mathcal H_{\boldsymbol u}}.
    \end{equation*}
    Summing all terms gives
    \begin{equation*}
        \min_{f=\sum_{\boldsymbol u} f_{\boldsymbol u}} \sum_{\boldsymbol u\subseteq [d]}\|f_{\boldsymbol u}\|^2_{\mathcal H_{\gamma,\boldsymbol u}}=\min_{f=\sum_{\boldsymbol u} f_{\boldsymbol u}} \sum_{\boldsymbol u\subseteq [d]}\frac{1}{\gamma_{\boldsymbol u}}\|f_{\boldsymbol u}\|^2_{\mathcal H_{\boldsymbol u}}.
    \end{equation*}
    This finishes the proof.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{prop:rnderiv}
    Consider a family of TSKs parameterized by $\boldsymbol\Sigma$.
     Let ${\boldsymbol  \Sigma}\in(0,1)^d$ and $\widetilde{{\boldsymbol  \Sigma}}\in[0,1]^d$. Then any $f\in\mathcal{H}_{\widetilde{{\boldsymbol  \Sigma}}}$ is also a member of $\mathcal{H}_{{\boldsymbol  \Sigma}}$.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
For each $f\in\mathcal{H}_{\widetilde{{\boldsymbol  \Sigma}}}$, there exists a spectral factor $\widetilde{\beta}$ in $L_2(\tau_{\widetilde{{\boldsymbol  \Sigma}}})$ such that
    \begin{equation*}
f(\boldsymbol x) = \int \widetilde{\beta}(\boldsymbol\omega)\mathrm{e}^{\mathrm{i} \boldsymbol\omega^\top\boldsymbol x}\,\mathrm{d}\tau_{\widetilde{{\boldsymbol  \Sigma}}}(\boldsymbol\omega) 
= \sum_{\boldsymbol u\subseteq[d]}\widetilde{\gamma}_{\boldsymbol u}\int \widetilde{\beta}_{\boldsymbol u}^0\mathrm{e}^{\mathrm{i}\boldsymbol\omega_{\boldsymbol u}^\top\boldsymbol x_{\boldsymbol u}}\,\mathrm{d}\tilde \tau_{\boldsymbol u}(\boldsymbol\omega_{\boldsymbol u}),
\end{equation*}
where $\tilde{\tau}_{\boldsymbol u} \coloneqq \bigotimes_{k\in\boldsymbol u}\tau_{\tilde \Sigma_k}$ and $\widetilde{\gamma}_{\boldsymbol u}=\prod_{k\in\boldsymbol u}\tilde\Sigma_k\prod_{l\notin\boldsymbol u}(1-\tilde\Sigma_l)$.
Now, define the function
\begin{equation*}
\frac{\,\mathrm{d}\tau_{\widetilde{{\boldsymbol  \Sigma}}}}{\,\mathrm{d}\tau_{{\boldsymbol  \Sigma}}}(\boldsymbol\omega)= \left\{\begin{array}{ccc}
\frac{\widetilde{\gamma}_{\emptyset}}{\gamma_{\emptyset}},\quad \boldsymbol\omega\in\Omega_{\emptyset} \\
\frac{\widetilde{\gamma}_{\{1\}}}{\gamma_{\{1\}}},\quad \boldsymbol\omega\in\Omega_{\{1\}} \\
\vdots \\
\frac{\widetilde{\gamma}_{\boldsymbol u}}{\gamma_{\boldsymbol u}},\quad \boldsymbol\omega\in\Omega_{\boldsymbol u} \\
\vdots \\
\frac{\widetilde{\gamma}_{[d]}}{\gamma_{[d]}},\quad \boldsymbol\omega\in\Omega_{[d]} \end{array}  \right..
\end{equation*}
This function is piecewise constant on the $2^d$ different sets $\Omega_{\boldsymbol u}\subset\mathbb{R}^d$, with $\Omega_{\boldsymbol u}=\{\boldsymbol{w}\in\mathbb{R}^d\colon \operatorname{supp} \boldsymbol{w} = \boldsymbol{u}\}$. For example, $\Omega_\emptyset = \{0\}$, $\Omega_{\{1\}}=\{\boldsymbol\omega\in\mathbb{R}^d\colon \omega_k=0,\ k\neq 1\}$, and $\Omega_{[d]}=\{\boldsymbol\omega\in\mathbb{R}^d\colon\omega_k\neq 0\}$.  The function $\frac{d\tau_{\widetilde{{\boldsymbol  \Sigma}}}}{d\tau_{{\boldsymbol  \Sigma}}}(\boldsymbol\omega)$ is the Radon-Nikodym derivative. This exists when $\tau_{\widetilde{{\boldsymbol  \Sigma}}}$ is absolutely continuous with respect to $\tau_{\boldsymbol \Sigma}$, which is the case whenever ${\boldsymbol  \Sigma}\in(0,1)^d$. With these we can write
    \begin{align*}
    f(\boldsymbol x)  =& \sum_{\boldsymbol u\subseteq[d]}\widetilde{\gamma}_{\boldsymbol u}\int_{\Omega^{|\boldsymbol u|}} \widetilde{\beta}_{\boldsymbol u}^0\mathrm{e}^{\mathrm{i}\boldsymbol\omega_{\boldsymbol u}^\top\boldsymbol x_{\boldsymbol u}}\,\mathrm{d}\tilde\tau_{\boldsymbol u}(\boldsymbol\omega_{\boldsymbol u}) 
    = \sum_{\boldsymbol u\subseteq[d]}\gamma_{\boldsymbol u}\int_{\Omega^{|\boldsymbol u|}} \frac{\widetilde{\gamma}_{\boldsymbol u}}{\gamma_{\boldsymbol u}}\widetilde{\beta}_{\boldsymbol u}^0\mathrm{e}^{\mathrm{i}\boldsymbol\omega_{\boldsymbol u}^\top\boldsymbol x_{\boldsymbol u}}\,\mathrm{d}\tilde \tau_{\boldsymbol u}(\boldsymbol\omega_{\boldsymbol u}) \\
    = & \int \widetilde{\beta}(\boldsymbol\omega)\mathrm{e}^{\mathrm{i}\boldsymbol\omega^\top\boldsymbol\boldsymbol{x}}\frac{d\tau_{\widetilde{{\boldsymbol  \Sigma}}}}{d\tau_{{\boldsymbol  \Sigma}}}(\boldsymbol\omega)\,\mathrm{d}\tau_{\boldsymbol u}(\boldsymbol\omega_{\boldsymbol u}).
    \end{align*}
    This shows that $f$ can be expressed through the new spectral factor $\beta = \widetilde{\beta}\frac{\,\mathrm{d}\tau_{\widetilde{{\boldsymbol  \Sigma}}}}{\,\mathrm{d}\tau_{{\boldsymbol  \Sigma}}}$ in $\mathcal{H}_{{\boldsymbol  \Sigma}}$. To ensure $f\in \mathcal{H}_{{\boldsymbol  \Sigma}}$, we verify that $\beta \in L_2(\Omega,\tau_{{\boldsymbol  \Sigma}})$. Let $m \coloneqq \max_{\boldsymbol u\subseteq[d]} \frac{\widetilde{\gamma}_{\boldsymbol u}}{\gamma_{\boldsymbol u}} < \infty$. Then,
    \begin{equation*}
    \|\beta\|_{L_2(\tau_{{\boldsymbol  \Sigma}})}^2 = \int |\widetilde{\beta}|^2 \left(\frac{d\tau_{\widetilde{{\boldsymbol  \Sigma}}}}{d\tau_{{\boldsymbol  \Sigma}}}\right)^2 d\tau_{{\boldsymbol  \Sigma}} \le m \int |\widetilde{\beta}|^2 \frac{d\tau_{\widetilde{{\boldsymbol  \Sigma}}}}{d\tau_{{\boldsymbol  \Sigma}}} d\tau_{{\boldsymbol  \Sigma}} = m\|\widetilde{\beta}\|_{L_2(\tau_{\widetilde{{\boldsymbol  \Sigma}}})}^2 < \infty.
    \end{equation*}
    Since $\beta$ has a finite RKHS norm under $\tau_{{\boldsymbol  \Sigma}}$, we conclude $f\in \mathcal{H}_{{\boldsymbol  \Sigma}}$.

<!-- theorem-1: body_md -->
\label{thm:convexity}
Let $ 0\neq f\in \mathcal H_{\boldsymbol\Sigma}$ for some, and hence every,
$\boldsymbol\Sigma\in(0,1)^d$. Assume that each kernel $\kappa_k$ is a strictly positive definite kernel satisfying
$\kappa_k(x_k,x_k)=1.$
Then the function
\[
    g(\boldsymbol\Sigma)=\|f\|_{\boldsymbol\Sigma}^2
\]
is strongly convex on $(0,1)^d$.
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
We first construct a decomposition of each one-dimensional kernel $\kappa_k$ into a
constant part and a part whose RKHS contains no nonzero constant
functions. Define  $r_k\coloneqq \|1\|_{\mathcal H_k}^2$. For every $k\in [d]$, there are two cases and we define the constants
\begin{equation}\label{equ:Ck}
C_k\coloneqq \begin{cases}
    1 &\text{ if } 1\notin\mathcal H_k,\\
     1-\frac{1}{r_k} &\text{ if } 1\in\mathcal H_k.
\end{cases}\end{equation}
Then we define the RKHS, which contains no constant by
\[\widetilde{\kappa}_k(x_k,y_k)\coloneqq \begin{cases}
    \kappa_k(x_k,y_k) &\text{ if } 1\notin\mathcal H_k,\\
    \kappa_k(x_k,y_k)-\frac{1}{r_k} &\text{ if } 1\in\mathcal H_k.
\end{cases}\]
To see that $\widetilde{\kappa}_k$ is positive definite in the second case,
let $\{x_k^1,\ldots,x_k^M\}\subset\mathbb{D}$ and $a_1,\ldots,a_M\in\mathbb{R}$.
By the reproducing property,
\begin{align*}
\sum_{i,j=1}^M a_i a_j
\widetilde{\kappa}_k(x_k^i,x_k^j)
&=
\Big\|\sum_{i=1}^M a_i\kappa_k(x_k^i,\cdot)\Big\|_{\mathcal H_k}^2
-\frac{1}{r_k}
\Big(\sum_{i=1}^M a_i\Big)^2\\
&=
\Big\|\sum_{i=1}^M a_i\kappa_k(x_k^i,\cdot)\Big\|_{\mathcal H_k}^2
-\frac{
\Big|
\Big\langle
1,\sum_{i=1}^M a_i\kappa_k(x_k^i,\cdot)
\Big\rangle_{\mathcal H_k}
\Big|^2
}{
\|1\|_{\mathcal H_k}^2
}
\geq 0,
\end{align*}
where the final inequality follows from the Cauchy--Schwarz inequality.
Now, if $1\in\mathcal H_k$, $\mathcal H_k=\mathcal H^1_k\oplus\mathcal H_{\widetilde{\kappa}_k}$
where $\mathcal H_k^1=\operatorname{span}\{1\}$ with norm inherited from $\mathcal H_k$. The reproducing kernel of this
one-dimensional subspace is the constant kernel $1/r_k$.

We next show that 
    $0< C_k\leq 1.$
The result is immediate if $1\notin\mathcal H_k$. Suppose instead that
$1\in\mathcal H_k$. Since $\kappa_k(x_k,x_k)=1$, the reproducing property
gives
\[
    |h(x_k)|
    \leq
    \|h\|_{\mathcal H_k}
    \|\kappa_k(x_k,\cdot)\|_{\mathcal H_k}
    =
    \|h\|_{\mathcal H_k}
\]
for every $h\in\mathcal H_k$. Taking $h=1$ gives $1\leq\|1\|_{\mathcal H_k}.$
Equality would imply equality in the Cauchy--Schwarz inequality for
$1$ and $\kappa_k(x_k,\cdot)$ for every $x_k\in\mathbb{D}$. This would mean
    $\kappa_k(x_k,\cdot)=1$
for every $x_k$, and hence $\kappa_k\equiv1$, contradicting strict
positive definiteness on a domain containing at least two distinct
points. Then, 
    $r_k=\|1\|_{\mathcal H_k}^2>1$
and 
\[
    0<C_k=1-\frac{1}{r_k}<1.
\]
In either case,
\[
    0< C_k\leq 1.
\]

The $k$th factor of the TSK can be written as
\begin{align*}
    1-\Sigma_k+\Sigma_k\kappa_k(x_k,y_k) = 1+\Sigma_k\big(\kappa_k(x_k,y_k)-1\big)
    &=
1+\Sigma_k\big(\widetilde{\kappa}_k(x_k,y_k)-C_k\big),
\end{align*}
since in both cases $\widetilde{\kappa}_k(x_k,y_k) - C_k = \kappa_k(x_k,y_k)-1$. 
Therefore,
\begin{equation}\label{eq:tsk_orthogonal_decomposition}
    \kappa_{\boldsymbol\Sigma}(\boldsymbol{x},\boldsymbol{y})
    =
    \prod_{k=1}^d
    \big(
        1-C_k\Sigma_k
        +\Sigma_k\widetilde{\kappa}_k(x_k,y_k)
    \big).
\end{equation}

\textbf{Decomposition of the RKHS norm:}\\
Expanding~\eqref{eq:tsk_orthogonal_decomposition} gives
\[
    \kappa_{\boldsymbol\Sigma}(\boldsymbol{x},\boldsymbol{y})
    =
    \sum_{\boldsymbol u\subseteq[d]}
    \theta_{\boldsymbol u}(\boldsymbol\Sigma)
    \widetilde{\kappa}_{\boldsymbol u}(\boldsymbol{x}_{\boldsymbol u},\boldsymbol{y}_{\boldsymbol u}),\quad 
    \theta_{\boldsymbol u}(\boldsymbol\Sigma)
    \coloneqq
    \prod_{k\in\boldsymbol u}\Sigma_k
    \prod_{\ell\notin\boldsymbol u}(1-C_\ell\Sigma_\ell),
\]
where
\[
    \widetilde{\kappa}_{\boldsymbol u}
    \coloneqq
    \prod_{k\in\boldsymbol u}\widetilde{\kappa}_k,
    \qquad
    \widetilde{\kappa}_{\emptyset}=1.
\]
Since $0< C_k\leq 1$ and $0<\Sigma_k<1$, we have
    $1-C_k\Sigma_k>0.$
So, $\theta_{\boldsymbol u}(\boldsymbol\Sigma)>0$ for every $\boldsymbol u\subseteq[d]$.

By construction, the constant space and
$\mathcal H_{\widetilde{\kappa}_k}$ have trivial intersection for every $k\in [d]$.
Then, the tensor-product spaces corresponding to distinct subsets
$\boldsymbol u$ decompose $\mathcal H_{\boldsymbol\Sigma}$ by direct sum,
\begin{equation}\label{equ:tensor}
\mathcal H_{\boldsymbol\Sigma} = \bigoplus_{\boldsymbol u\subseteq[d]}\Big(\bigotimes_{k\in\boldsymbol u}\Sigma_k\mathcal H_{\widetilde{\kappa}_k}\otimes\bigotimes_{\ell\notin\boldsymbol u} (1-C_k\Sigma_k)\mathcal H_k^1\big) .
\end{equation}
Every $f\in\mathcal H_{\boldsymbol\Sigma}$ therefore
has a unique decomposition
\[
    f=\sum_{\boldsymbol u\subseteq[d]}f_{\boldsymbol u},
    \qquad
    f_{\boldsymbol u}\in
    \mathcal H_{\widetilde{\kappa}_{\boldsymbol u}},
\]
where the constant factors are the inactive coordinates $[d]\backslash \boldsymbol u$. Note that these component spaces, and therefore the functions
$f_{\boldsymbol u}$, do not depend on $\boldsymbol\Sigma$.

By the scaling property of RKHS norms,
\begin{align*}
    g(\boldsymbol\Sigma)
    =
    \|f\|_{\boldsymbol\Sigma}^2
    &=
    \sum_{\boldsymbol u\subseteq[d]}
    \frac{
        \|f_{\boldsymbol u}\|_{\mathcal H_{\widetilde{\kappa}_{\boldsymbol u}}}^2
    }{
        \theta_{\boldsymbol u}(\boldsymbol\Sigma)
    }
    =
    \sum_{\boldsymbol u\subseteq[d]}
    \frac{\beta_{\boldsymbol u}}
    {\theta_{\boldsymbol u}(\boldsymbol\Sigma)}, \quad   \beta_{\boldsymbol u}
    \coloneqq
    \|f_{\boldsymbol u}\|_{\mathcal H_{\widetilde{\kappa}_{\boldsymbol u}}}^2.
\end{align*}
The coefficients $\beta_{\boldsymbol u}\geq0$ depend only on $f$ and the
chosen kernels and not on $\boldsymbol\Sigma$. Since $f\neq0$, there exists
at least one $\boldsymbol u\subseteq[d]$ such that $\beta_{\boldsymbol u}>0$.\\


\textbf{The Hessian matrix:}\\
For $\boldsymbol u\subseteq[d]$, define
\[
    \xi_k^{\boldsymbol u}
    \coloneqq
    \begin{cases}
      -  \dfrac{1}{\Sigma_k},
        & k\in\boldsymbol u,\\[3mm]
        \dfrac{C_k}{1-C_k\Sigma_k},
        & k\notin\boldsymbol u.
    \end{cases}
\]
Then $\xi_k^{\boldsymbol u}>0$ for $k\in\boldsymbol u$ and
$\xi_k^{\boldsymbol u}<0$ for $k\notin\boldsymbol u$. Direct differentiation gives
\[
    \boldsymbol H
    =
    \nabla^2g(\boldsymbol\Sigma)
    =
    \sum_{\boldsymbol u\subseteq[d]}
    \frac{\beta_{\boldsymbol u}}{\theta_{\boldsymbol u}}
    \boldsymbol H^{\boldsymbol u},
\]
with diagonal entries
$H^{\boldsymbol u}_{k,k}=
 2(\xi_k^{\boldsymbol u})^2$
and, non-diagonal entries for $k\neq j$,
$H^{\boldsymbol u}_{k,j}
    =
    \xi_k^{\boldsymbol u}\xi_j^{\boldsymbol u}.$
Consequently, for any $\boldsymbol v\in\mathbb{R}^d$,
\begin{align*}
    \boldsymbol v^\top \boldsymbol H^{\boldsymbol u}\boldsymbol v
    &=
    \sum_{k=1}^d
    v_k^2(\xi_k^{\boldsymbol u})^2
    +
    \left(
        \sum_{k=1}^d v_k\xi_k^{\boldsymbol u}
    \right)^2.
\end{align*}
In particular, $H^{\boldsymbol u}$ is positive definite because
$\xi_k^{\boldsymbol u}\neq0$ for every $k$.

It remains to show that the smallest eigenvalue of $\boldsymbol H$ is uniformly
bounded below on $(0,1)^d$. Choose $\boldsymbol u$ such that
$\beta_{\boldsymbol u}>0$. 
Define
\[
    \widehat C\coloneqq
    \min_{k\in[d]}C_k>0
\]
If $k\in\boldsymbol u$, then
    $|\xi_k^{\boldsymbol u}|=\frac{1}{\Sigma_k}>1>\widehat C.$ 
If $k\notin \boldsymbol u$, then 
\[
    |\xi_k^{\boldsymbol u}| = \frac{C_k}{1-C_k\Sigma_k} >C_k>\widehat C.\]
Moreover,
$
    0<\theta_{\boldsymbol u}(\boldsymbol\Sigma)\leq1
$
because $0<\Sigma_k<1$ and
$0<1-C_k\Sigma_k\leq1$. Hence, for any $\boldsymbol v$ with
$\|\boldsymbol v\|=1$,
\begin{align*}
    \boldsymbol v^\top H\boldsymbol v
    &\geq
    \frac{\beta_{\boldsymbol u}}{\theta_{\boldsymbol u}}
    \boldsymbol v^\top H^{\boldsymbol u}\boldsymbol v
    \geq
    \frac{\beta_{\boldsymbol u}}{\theta_{\boldsymbol u}}
    \sum_{k=1}^d
    v_k^2(\xi_k^{\boldsymbol u})^2
    \geq
    \beta_{\boldsymbol u}\widehat C^2
    \sum_{k=1}^d v_k^2
    =
    \beta_{\boldsymbol u}\widehat C^2>0.
\end{align*}
The lower bound is independent of $\boldsymbol v$ and $\boldsymbol\Sigma$. Therefore,
\[
    \inf_{\boldsymbol\Sigma\in(0,1)^d}
    \lambda_{\min}\big(\nabla^2g(\boldsymbol\Sigma)\big)
    \geq
    \beta_{\boldsymbol u}\widehat C^2>0,
\]
which proves that $g$ is strongly convex on $(0,1)^d$.

<!-- theorem-2: body_md -->
\label{thm:unif_conv}
Let $\{(\boldsymbol x^i,f^i)\}_{i\in\mathbb{N}}$ be a sequence where each $\boldsymbol x^i\in\mathbb{D}^d$ is sampled i.i.d. from a distribution $\mu$, whose support is $\mathbb{D}^d$. Here, $f^i=f(\boldsymbol x^i)$. Let $g_M({\boldsymbol  \Sigma})$ be defined as in~\eqref{eq:g_M}. Define $\boldsymbol f^M$ to be the vector of length $M$ containing the first $M$ values $f^i=f(\boldsymbol x^i)$. The matrix $\boldsymbol K_{{\boldsymbol  \Sigma},M}$ is the kernel matrix from the first $M$ points $\boldsymbol x^i$ for the kernel $\kappa_{\boldsymbol  \Sigma}$. Let $g$ be as defined in Theorem~\ref{thm:convexity} as $g({\boldsymbol  \Sigma})=\|f\|_{\boldsymbol  \Sigma}^2$. Then $g_M$ converges uniformly to $g$ on the domain $[\eta,1-\eta]^d$ for any $\eta\in(0,1)$.
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
The structure of $g_M$ and $g$ implies they are continuous.
    For any ${\boldsymbol  \Sigma}$, the evaluation of $g_M({\boldsymbol  \Sigma})$ is equal to 
    \[
    \min_{\boldsymbol \alpha\in\mathbb{R}^M}\boldsymbol\alpha^\top \boldsymbol K_{{\boldsymbol  \Sigma},M}\boldsymbol\alpha\quad \text{subject to}\ \boldsymbol K_{{\boldsymbol  \Sigma},M}\boldsymbol\alpha=\boldsymbol f^M.
    \]
    This is the smallest norm possible for a function in $\mathcal{H}_{\boldsymbol  \Sigma}$ that interpolates $\{(\boldsymbol x^i,f^i)\}_{i=1}^M$. In other words, it is equal to 
    \[
    \min_{\hat f\in\mathcal H_{\boldsymbol  \Sigma}}\|\hat f\|_{\boldsymbol  \Sigma}^2\quad \text{subject to}\ \hat f(\boldsymbol x^i)=f^i,\ i=1,\dots,M.
    \]
    Since $f$ also interpolates the data, $\|\hat f_{{\boldsymbol  \Sigma},M}\|_{\boldsymbol  \Sigma}\leq \|f\|_{\boldsymbol  \Sigma}$, where $\hat f_{{\boldsymbol  \Sigma},M}$ is the norm-minimizing interpolating function. Then, $g_M({\boldsymbol  \Sigma})\leq g({\boldsymbol  \Sigma})$ for all ${\boldsymbol  \Sigma}.$ In fact, the subset of interpolating functions in $\mathcal{H}_{\boldsymbol  \Sigma}$ shrinks as we increase the number of points to interpolate. Then, $g_M({\boldsymbol  \Sigma})\leq g_{M+1}({\boldsymbol  \Sigma})$ and the sequence $\{g_M({\boldsymbol  \Sigma})\}_{M\in\mathbb{N}}$ is monotone increasing. 
    
    We show that $g_M({\boldsymbol  \Sigma})\to g({\boldsymbol  \Sigma})$ pointwise. 
    Each pointwise sequence $g_M({\boldsymbol  \Sigma})$ is monotone increasing and bounded above by $g({\boldsymbol  \Sigma})$. Then, $g_M({\boldsymbol  \Sigma})\to g_{\boldsymbol  \Sigma}^*$, with $g_{\boldsymbol  \Sigma}^*\leq g({\boldsymbol  \Sigma})$. 
    Since $\{\hat f_{{\boldsymbol  \Sigma},M}\}_{M\in\mathbb{N}}$ is uniformly bounded in $\mathcal{H}_{\boldsymbol  \Sigma}$, there is a  subsequence $\{\hat f_{{\boldsymbol  \Sigma},M_i}\}_{i\in\mathbb{N}}$ that weakly converges to some $f^*_{\boldsymbol  \Sigma}$ in $\mathcal{H}_{\boldsymbol  \Sigma}$. As $\mathcal{H}_{\boldsymbol  \Sigma}$ is an RKHS,  $\langle h,\kappa_{\boldsymbol  \Sigma}(\cdot,\boldsymbol x)\rangle_{\mathcal{H}_{\boldsymbol  \Sigma}}=h(\boldsymbol x)$ for all $h\in\mathcal{H}_{\boldsymbol  \Sigma}$ and $\boldsymbol x\in\mathbb{D}^d$. Weak convergence implies pointwise convergence of the subsequence, 
    $\lim_{i \to \infty} \langle \hat{f}_{{\boldsymbol \Sigma},M_i}, \kappa_{\boldsymbol \Sigma}(\cdot, \boldsymbol x)\rangle = \langle f^*_{\boldsymbol \Sigma}, \kappa_{\boldsymbol \Sigma}(\cdot, \boldsymbol x)\rangle$.
                                                                      
    This implies \[\|f^*_{\boldsymbol  \Sigma}\|_{\boldsymbol  \Sigma}^2 \leq \liminf_{i\to\infty}\|\hat f_{{\boldsymbol  \Sigma},M_i}\|_{\boldsymbol  \Sigma}^2\leq g_{\boldsymbol  \Sigma}^*.\]
 Suppose $\{\boldsymbol x^i\}_{i=1}^\infty$ is dense in $\mathbb{D}^d,$ which is true almost surely. Because $f^*_{\boldsymbol  \Sigma}(\boldsymbol x^i) = f^i$ for all $i$, $f^*_{\boldsymbol  \Sigma}$ and $f$ are equal to each other on a dense subset of $\mathbb{D}^d$. 
Since the kernel $\kappa_{\boldsymbol \Sigma}$ is continuous, all functions in the RKHS $\mathcal{H}_{\boldsymbol \Sigma}$ are continuous. Continuity of both $f$ and $f^*_{\boldsymbol \Sigma}$ implies $f^*_{\boldsymbol  \Sigma}=f$. So, $g({\boldsymbol  \Sigma})\leq g_{\boldsymbol  \Sigma}^*$, meaning $g({\boldsymbol  \Sigma})= g_{\boldsymbol  \Sigma}^*$.

    Because all $g_M$ and $g$ are continuous and the sequence is monotone increasing and converges pointwise, Dini's theorem~\cite{Rudin} yields uniform convergence $g_M\to g$ on any compact subset of $(0,1)^d$.

<!-- corollary-1: body_md -->
By Theorem~\ref{thm:unif_conv}, for every $M\in\mathbb{N}$ there exists $\epsilon_M > 0$ such that
    \begin{equation*}                       
        \sup_{\boldsymbol \Sigma\in [\eta,1-\eta]^d} \vert g(\boldsymbol \Sigma) - g_M(\boldsymbol \Sigma) \vert \leq \epsilon_M,
    \end{equation*}
   where $\eta\in(0,1)$. Suppose $g$ is $m$-strongly convex, as given by Theorem~\ref{thm:convexity}. 
    Then, the error between the TSK factors $\hat{\boldsymbol \Sigma}_M$ derived by the empirical minimization~\eqref{eq:Sigma_hat} and the true minimizer ${\boldsymbol \Sigma}^*$ of the function $g$ is bounded by
    \[
    \left\lVert \smash{\hat{{\boldsymbol \Sigma}}_M - {\boldsymbol \Sigma}^*} \right\rVert \leq 2\sqrt{\frac{\epsilon_M}{m}}.
    \]
<!-- end corollary-1 -->
<!-- proof-5: body_md -->
Using Theorem~\ref{thm:convexity} and the definition of strong convexity evaluated at the minimum ${\boldsymbol \Sigma}^*$ (see \cite{Ne04}), there exists a constant $m > 0$ such that for any ${\boldsymbol \Sigma} \in [\eta,1-\eta]^d$,
\[
\frac m2 \left\lVert \smash{{\boldsymbol \Sigma} - {\boldsymbol \Sigma}^*} \right\rVert^2 \leq g({\boldsymbol \Sigma}) - g({\boldsymbol \Sigma}^*).
\]
Evaluating this at the empirical minimizer $\hat{{\boldsymbol \Sigma}}_M$ and utilizing the uniform bound yields
\begin{align*}
\frac{m}{2} \Vert\hat{{\boldsymbol \Sigma}}_M - {\boldsymbol \Sigma}^*\Vert^2 &\leq g(\hat{{\boldsymbol \Sigma}}_M) - g({\boldsymbol \Sigma}^*) 
= g(\hat{{\boldsymbol \Sigma}}_M) - g_M(\hat{{\boldsymbol \Sigma}}_M) + g_M(\hat{{\boldsymbol \Sigma}}_M) - g({\boldsymbol \Sigma}^*) \\
&\leq g(\hat{{\boldsymbol \Sigma}}_M) - g_M(\hat{{\boldsymbol \Sigma}}_M) + g_M({\boldsymbol \Sigma}^*) - g({\boldsymbol \Sigma}^*) 
\leq \vert g(\hat{{\boldsymbol \Sigma}}_M) - g_M(\hat{{\boldsymbol \Sigma}}_M) \vert + \vert g_M({\boldsymbol \Sigma}^*) - g({\boldsymbol \Sigma}^*) \vert \\
&\leq 2 \sup_{{\boldsymbol \Sigma} \in [\eta,1-\eta]^d} \vert g_M({\boldsymbol \Sigma}) - g({\boldsymbol \Sigma})\vert \leq 2\epsilon_M.
\end{align*}
Here, the first inequality relies on the definition of the empirical minimizer, which guarantees $g_M(\hat{{\boldsymbol \Sigma}}) \leq g_M({\boldsymbol \Sigma}^*)$. Rearranging this inequality directly provides a bound on the parameter estimation error,
\[
\left\lVert \smash{\hat{{\boldsymbol \Sigma}}_M - {\boldsymbol \Sigma}^*} \right\rVert \leq 2\sqrt{\frac{\epsilon_M}{m}}.
\]
This finishes the proof.
