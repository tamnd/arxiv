<!-- lemma-1: body_md -->
Let $\lambda >0$ and consider a positive definite matrix $B>0$ with $\rm{tr} (B)=1$
acting in $\mathbb{R}^n$.  Let $M$ be a map from $\mathbb{R}^n \to \mathbb{R}^n$.  Then 
\begin{equation} 
\rm{tr} \left(M \left(M^{\top} B^{-1}\, M+\lambda I\right)^{-1}\, M^{\top}\right) \leqslant \rm{tr}(B) = 1. 
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Write
\begin{equation} 
\begin{aligned}
&\rm{tr} \left(M \left(M^{\top} B^{-1} M + \lambda I\right)^{-1} M^{\top}\right)    \\
& \quad = \rm{tr} \big(B K (LK + \lambda I)^{-1} L\big), \, 
\text{ where } \, K=B^{-\frac{1}{2}} M \, \text{ and } \, L = \big(B^{-\frac{1}{2}} M\big)^{\top}.
\end{aligned}
\end{equation} 
But by \eqref{1.2}, $\lambda (KL + \lambda I)^{-1} + K (LK+\lambda I)^{-1} L = I$, and so 
\begin{align} 
\begin{split}
& \rm{tr} \Big( M \left( M^{\top} B^{-1} \, M+\lambda I\right)^{-1} M^{\top} \Big)   \\
& \quad = \rm{tr} (B) - \rm{tr} \big(B \lambda (KL + \lambda I)^{-1}\big)    \\
& \quad = \rm{tr} (B) - \rm{tr}\Big(\lambda B^{1/2} \big(B^{1/2} M M^{\top}\ B^{1/2} + \lambda I\big)^{-1} B^{1/2}\Big). 
\end{split}
\end{align} 
But the second matrix is positive definite and so we obtain
\begin{equation} 
\rm{tr} \Big(M\left( M^{\top}\, B^{-1} M + \lambda I \right)^{-1} M^{\top}\Big) \leqslant \rm{tr}(B) = 1,
\end{equation} 
which proves the Lemma.

<!-- theorem-2: body_md -->
Let $A, B \in {\mathcal B}(\mathcal{H})$ with $A\geqslant 0$.  Suppose that
$\|AB\| \leqslant 1$ and $\|B A\| \leqslant 1$. Then
\begin{equation}\label{eq25}
\|A^\alpha \,B\, A^{1-\alpha}\| \leqslant 1 \ , \quad 0 \leqslant \alpha \leqslant 1. 
\end{equation}
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
We recall that since $\|T \| = \|T^*\|$ for any operator $T$, $\|A^\alpha B\,A^{1-\alpha}\| \leqslant 1$ 
for $0 \leqslant \alpha \leqslant 1$ implies 
\begin{equation}\label{eq26}
\|A^\alpha B^*\ A^{1-\alpha}\| \leqslant 1, \quad 0\leqslant \alpha \leqslant 1.
\end{equation}

By taking limits, it is enough to prove \eqref{eq25} for $\alpha =j/2^n$, $0 \leqslant j
\leqslant 2^n$.  This is clearly true for $n=0$, that is, $\alpha=0$ or $1$.  Assuming
\eqref{eq25} is true for $n$, we show that it is true for $n+1$, that is, for  
$\alpha = j/2^{n+1}$, $j=0, 1, \dots, 2^{n+1}$. 
For $j\in \{0, 1, \dots, 2^n\} \subset \{ 0, 1, \dots, 2^{n+1}\}$, one infers, using
$\|T\|^2= \|T^*\, T\|$ and the fact that $\|T\| =\sup \{ |\lambda| = \lambda \in \sigma (T)\}$ if $T=T^*$,
\begin{equation} 
\big\|A^{j/2^{n+1}} B A^{1-j/2^{n+1}} \big\|^2 = \sup \Big\{|\lambda| \, \Big| \,  \lambda \in \sigma
\big(A^{1-j/2^n} B^* A^{j/2^n} BA\big) \Big\} .
\end{equation} 
But for \textit{any} $T$, $\:\sup \{ |\lambda| \leqslant \sigma(T)\} \leqslant \|T\|$ and hence
\begin{equation} 
\big\|A^{j/2^{n+1}} B\;A^{1-j/2^{n+1}}\big\|^2 \leqslant \big\|A^{1-j/2^n} B^*\;A^{j/2^n}\big\|
\;\|BA\| \leqslant 1,
\end{equation} 
by the induction hypothesis as $0\leqslant j \leqslant 2^n$.

If $2^n <j \leqslant 2^{n+1}$, 
\begin{equation} 
\big\|A^{j/2^{n+1}} B\:A^{1-(j/2^{n+1})}\big\| = \big\|A^{1-j/2^{n+1}} B^*\:A^{j/2^{n+1}}\big\|
= \big\|A^{j'/2^{n+1}} B^*\:A^{1-(j'/2^{n+1})}\big\|, 
\end{equation} 
where $\jmath'=2^{n+1}-j$.  As $2^n<\jmath\leqslant 2^{n+1}$, $0\leqslant \jmath'\leqslant 2^{n+1}-2^n = 2^n$, 
the induction is true for $n+1$.  This proves \eqref{eq25}.

Let  $\chi_s(x) = 1_{x \geqslant s}$ and let 
\begin{equation} 
L_s: \begin{cases} L^2 ((s,\infty)) \to L^2 ((-\infty, \infty))  \\
f \mapsto (L_s f)(x) = \int^\infty_s\, A_i (x+y) f(y) \, dy. 
\end{cases} 
\end{equation} 
Then,  
\begin{align}
\chi_s(x) \:L_{-\infty} \, \sigma \;L_s\, f &= \chi_s(x) \int^\infty_{-\infty}\;
A_i(x+u)\, \sigma(u) \, \left(L_s\, f\right)(u)\, du   \notag \\
&= \chi_s(x) \int^\infty_{-\infty} A_i (x+u) \, \sigma (u) \, \bigg(\int^\infty_s A_i

&= \chi_s(x) \int^\infty_s \left(\int^\infty_{-\infty} A_i (x+u) \,
\sigma (u)\, A_i(y+u)\, du\right) f(y)\, dy    \notag \\
&= \chi_s(x) \int^\infty_s \, K_\sigma (x,y)\, f(y)\, dy     \notag \\
&= K_\sigma \, f(x), 
\end{align}
with $K_\sigma: L^2((s, \infty)) \to L^2((s, \infty))$, that is, $K_\sigma= \chi_s L_{-\infty} \sigma L_s = A B$, where 
$A=\chi_s\, L_{-\infty} \sqrt{\sigma}$, $B=\sqrt{\sigma} L_s$.
On the other hand, 
\begin{equation} 
\begin{aligned}
B\:A f &= \left(\sqrt{\sigma} L_s \chi_s L_{-\infty} \sqrt{\sigma} f\right)(x)   \\
&= \sqrt{\sigma(x)}\int^\infty_s A_i (x+u) \chi_s(u)\left(L_{-\infty} \sqrt{\sigma} f \right)(u)\,du    \\
&=\sqrt{\sigma(x)} \int^\infty_s A_i (x+u) \left(\int^\infty_{-\infty} A_i(u+y) \sqrt{\sigma(y)} f(y)\,dy\right) du \\
&= \sqrt{\sigma(x)} \int^\infty_{-\infty} \left(\int^\infty_s \:A_i (x+u) A_i(y+u) \,du \right) \sqrt{\sigma(y)} f(y)\, dy\\
&= \int^\infty_{-\infty} \left[ \sqrt{\sigma(x)} \left( \int^\infty_0 A_i (x+u+s) A_i (y+u+s)\,du\right) \sqrt{\sigma(y)} \right] f(y)\,dy. 
\end{aligned}
\end{equation} 
By commutation, the operator 
\begin{equation} 
D \equiv \sqrt{\sigma(x)} \left(\int^\infty_0
A_i (x+u+s) A_i(y+u+s)\,du\right) \sqrt{\sigma(y)} \, dy\, \text{ on $L^2((-\infty,\infty))$,} 
\end{equation}

the Lemma follows.

<!-- proof-5: body_md -->
Since algebraic multiplicities were not addressed in \cite{De78}, we now provide a proof that algebraic multiplicities of nonzero discrete eigenvalues of $AB$ and $BA$ coincide. (The case of geometric multiplicities is already treated in item $(i)$.) Let $P(\lambda_0,AB)$ denote the Riesz projection associated with $\lambda_0 \in \sigma_d(AB)\backslash \{0\}$ and suppose that $f \in \operatorname{ran}(P(\lambda_0,AB))$, that is, $P(\lambda_0,AB) f = f$. Then 
\begin{equation}
P(\lambda_0,AB) f = \cfrac{-1}{2\pi i} \ointctrclockwise_{C(\lambda_0;r_0)} (AB- \zeta I_{X_1})^{-1} f \, d\zeta = f,
\end{equation}
and hence 
\begin{align} 
\begin{split} 
B f &= \cfrac{-1}{2\pi i} \ointctrclockwise_{C(\lambda_0;r_0)} B (AB- \zeta I_{X_1})^{-1} f \, d\zeta   \\
&= \cfrac{-1}{2\pi i} \ointctrclockwise_{C(\lambda_0;r_0)} (BA- \zeta I_{X_1})^{-1} B f \, d\zeta,   
\end{split} 
\end{align}
since by \eqref{2.17}, $B (AB- \zeta I_{X_1})^{-1} = (BA- \zeta I_{X_2})^{-1} B$. Thus,
\begin{equation}
P(\lambda_0,BA) g = g, \quad g = B f. 
\end{equation}
Consequently, 
\begin{equation}
B \operatorname{ran}(P(\lambda_0,AB)) \subseteq \operatorname{ran}(P(\lambda_0,BA)).
\end{equation}
It remains to show injectivity of the map $B \colon \operatorname{ran}(P(\lambda_0,AB)) \to \operatorname{ran}(P(\lambda_0,BA))$. To this end suppose that $Bf = 0$, we will show this implies $f=0$.

For $\zeta \in C(\lambda_0;r_0)$ one obtains that
\begin{equation}
f = (AB - \zeta I_{X_1})^{-1} (AB -  \zeta I_{X_1}) f = - \zeta (AB - \zeta I_{X_1})^{-1} f,
\end{equation}
equivalently,
\begin{equation}
(AB - \zeta I_{X_1})^{-1} f = - \zeta^{-1} f.
\end{equation}
But then
\begin{equation}
f = \cfrac{-1}{2\pi i} \ointctrclockwise_{C(\lambda_0;r_0)} (AB- \zeta I_{X_1})^{-1} f \, d\zeta 
= \cfrac{1}{2\pi i} \ointctrclockwise_{C(\lambda_0;r_0)} \zeta^{-1} \, d\zeta \, f = 0,
\end{equation}
since $\zeta^{-1}$ is analytic in a sufficiently small open neighborhood of $\overline{D(\lambda_0;r_0)}$. Thus,
\begin{equation}
\dim(\operatorname{ran}(P(\lambda_0,AB))) \leqslant \dim(\operatorname{ran}(P(\lambda_0,BA))), 
\end{equation}
and interchanging the role of $A$ and $B$ yields 
\begin{equation}
\dim(\operatorname{ran}(P(\lambda_0,AB))) = \dim(\operatorname{ran}(P(\lambda_0,BA))), 
\end{equation}
as was to be proven.

\operatorname{mKdV}_n(\phi) = 0 \, \text{ implies } \, \operatorname{KdV}_n(V_j) = 0, \quad j=1,2.     \label{6.28} 
\end{equation}
$(ii)$ Suppose $V_1 \in C^1({\mathbb{R}}^2)$ is real-valued, $\partial_x^m V_1 \in L^{\infty}({\mathbb{R}}^2)$, $m \in {\mathbb{N}}_0$, and for $t \in {\mathbb{R}}$, let $0 < \psi_1(\,\cdot\,,t)$ be a solution of $L_1(t) \psi_1(t) = 0$, where $L_1(t) = - (d^2/dx^2) + V_1(x,t)$, $t \in {\mathbb{R}}$. Introducing $\phi$ via 
\begin{equation}
\phi(x,t) = - \psi_{1,x} (x,t)/\psi_1(x,t), \quad (x,t) \in {\mathbb{R}}^2,
\end{equation}
then $\phi$ satisfies 
\begin{align} 
& \phi \in C^1({\mathbb{R}}^2), \quad \partial_x^m \phi \in L^{\infty}({\mathbb{R}}^2), \quad m \in {\mathbb{N}}_0,    \label{6.30} \\
& V_1(x,t) = \phi^2(x,t) - \phi_x(x,t), \quad (x,t) \in {\mathbb{R}}^2.
\end{align} 
Moreover, for each $n \in {\mathbb{N}}_0$, 
\begin{equation}
\operatorname{KdV}_n(V_1) = 0 \, \text{ implies } \, \operatorname{mKdV}_n(\phi) = 0 \, \text{ and } \, \operatorname{KdV}_n(V_2) = 0,     \label{6.32} 
\end{equation}
where
\begin{equation}
V_2(x,t) = \phi^2(x,t) + \phi_x(x,t), \quad (x,t) \in {\mathbb{R}}^2.
\end{equation}
The same applies with $V_1$ and $V_2$ interchanged.

$(i)$ Assertions \eqref{6.27} for $V_j$, $j=1,2$, are clear from the hypotheses on $\phi$. Similarly, the claim \eqref{6.28} is an instant consequence of Miura's identity \eqref{6.26}. \\[1mm] 
$(ii)$ Given $C \in (0,\infty)$ such that $V_1(x,t) \leqslant C$, $(x,t) \in {\mathbb{R}}^2$, one applies \cite[p.~358--360]{Ha02} to conclude that
\begin{equation}
- C^{1/2} \leqslant \psi_{1,x}(x,t)/\psi_1(x,t) = - \phi(x,t) \leqslant C^{1/2}, \quad (x,t) \in {\mathbb{R}}^2, 
\end{equation}
that is, $\phi \in L^{\infty}({\mathbb{R}}^2)$ and hence also $\phi_x \in L^{\infty}({\mathbb{R}}^2)$ since by hypothesis, $V_1 = \phi^2 - \phi_x \in L^{\infty}({\mathbb{R}}^2)$. Differentiating $V_1$ then yields $\phi_{xx} \in L^{\infty}({\mathbb{R}}^2)$, and continuing this process shows that $\partial_x^m \phi \in L^{\infty}({\mathbb{R}}^2)$, $m \in {\mathbb{N}}_0$. Clearly, $L_1 \psi_1 = 0$ and $\phi = - \psi_{1,x}/\psi_1$ yields $V_1 = \phi^2 - \phi_x$. 

If $\operatorname{KdV}_n(V_1) = 0$, then Miura's identity \eqref{6.26} yields 
\begin{equation}
0 = [2 \phi - \partial_x] \operatorname{mKdV}_n(\phi), \quad n \in {\mathbb{N}}_0.     \label{6.35}
\end{equation}
Since the first-order differential equation $[\partial_x - 2 \phi] f = 0$ has the solution 
\begin{equation}
f(x,t) = C(t) \exp\bigg(-2 \int^x_0 dx' \, \phi(x',t)\bigg), \quad (x,t) \in {\mathbb{R}}^2,
\end{equation}
with $C(t) \in {\mathbb{C}}$ independent of $\phi$, one concludes that
\begin{equation}
\operatorname{mKdV}_n(\phi) = C(t) \exp\bigg(-2 \int^x_0 dx' \, \phi(x',t)\bigg), \quad (x,t) \in {\mathbb{R}}^2.   \label{6.37}
\end{equation}
Choosing a particular $\phi$ such that $\phi_t$ and $\partial_x^m \phi$, $m \in {\mathbb{N}}$, all vanish as $x \to +\infty$ (or, as $x \to - \infty$)\footnote{For a concrete example of $\phi$ one can choose the $\operatorname{mKdV}$ soliton solutions $\phi_{2N-1}$, or $\phi_{2N}$, where $\lim_{x \to \pm\infty} \phi_{2N-1}(x) = \phi_{2N-1,\pm}$, with $\phi_{2N-1,+} = - \phi_{2N-1,-} \in {\mathbb{R}} \backslash \{0\}$, or $\lim_{x \to \pm\infty} \phi_{2N}(x) = \phi_{2N,\pm}$, with $\phi_{2N,+} = \phi_{2N,-} \in {\mathbb{R}} \backslash \{0\}$, see \cite[Sect.~8]{GSS91}.} yields the vanishing of the left-hand side in \eqref{6.37} and hence implies the contradiction of a nonvanishing right-hand side of \eqref{6.37} unless $C(t) = 0$. \label{r.3.29}
We comment a bit on the spectral properties of $0 \leqslant H_j(t)$, $t \in {\mathbb{R}}$, $j=1,2$, under the conditions imposed in Theorem~\ref{t3.28}. For this purpose we first recall two facts that will repeatedly be used below: First, if $H=T^*T \geqslant 0$ in the complex, separable Hilbert space ${\mathcal H}$, with $T$ a densely defined and closed operator in ${\mathcal H}$, then, for $f_0 \in \operatorname{dom}(H)$, 
\begin{equation}
\big\|H^{1/2} f_0\big\|^2_{{\mathcal H}} = (f_0, T^*T f_0)_{{\mathcal H}} = \|T f_0\|^2_{{\mathcal H}}, 
\end{equation}
implies that 
\begin{equation}
0 \in \sigma_p(H) \, \text{ if and only if } \, 0 \in \sigma_p(T).
\end{equation}
Second, if a Schr\"odinger differential expression $L = - (d^2/dx^2) + V(x)$ has a generalized (i.e., distributional) solution $\psi_0$ of $L \psi_0 = 0$ that is positive on an interval $(a,b) \subseteq {\mathbb{R}}$, then a second linearly independent solution $\widetilde \psi_0$ of $L \psi = 0$ on that interval is given by
\begin{equation} 
\widetilde \psi_0 (x) = \psi_0(x) \bigg[C + D \int_{x_0}^x dx' \, [\psi_0(x')]^{-2}\bigg], \quad x_0, x \in (a,b),      \label{3.315} 
\end{equation}
where $C \in {\mathbb{C}}$, $D \in {\mathbb{C}}\backslash \{0\}$ are constants. (In our present context $C, D$ will always be chosen to be real-valued.)   \\[1mm] 
Given these facts, we now suppose that 
\begin{align} 
\begin{split} 
& \phi(x,t) \underset{x \to \pm \infty}{\longrightarrow} \phi_{\pm} \, \text{ in the sense of \eqref{3.200}, viewing $t \in {\mathbb{R}}$ as a parameter,}    \\
& \quad \text{with $\phi_{\pm}$ independent of $t \in {\mathbb{R}}$.} 
\end{split} 
\end{align} 
Then,  
\begin{equation}
\sigma_{ess}(H_j(t)) = \sigma_{ac}(H_j(t)) = \big[\min\big(\phi_-^2,\phi_+^2\big), \infty\big), \quad t \in {\mathbb{R}}, \; j=1,2, 
\end{equation}
and one observes that the spectral multiplicity of $H_j(t)$ equals one on the interval $\big(\min\big(\phi_-^2,\phi_+^2\big), \max\big(\phi_-^2,\phi_+^2\big)\big)$, assuming the latter is nonempty, and the spectral multiplicity of $H_j(t)$ is two on the half-line $\big(\min\big(\phi_-^2,\phi_+^2\big), \infty\big)$, $j=1,2$. In particular, the interval $\big[0, \min\big(\phi_-^2,\phi_+^2\big)\big)$ contains discrete eigenvalues of $H_j(t)$ only (if any), $j=1,2$. The half-line $\big(\min\big(\phi_-^2,\phi_+^2\big), \infty\big)$ contains no embedded eigenvalues of $H_j(t)$, $t \in {\mathbb{R}}$, $j=1,2$. \\[1mm] 
\noindent 
$(i)$. Suppose that $\phi_{\pm} \neq 0$: Then, for each $t \in {\mathbb{R}}$,  
\begin{equation}
\begin{rcases}
0 \in \sigma_p(H_1(t)) \\
0 \notin \sigma_p(H_2(t))
\end{rcases} \Longleftrightarrow \phi_- < 0 < \phi_+.     \label{3.318}
\end{equation}
In this case $H_1(t)$ and $H_2(t)$ are obviously not isospectral, however, when restricting $H_1(t)$ to the orthogonal complement of its one-dimensional kernel generated by 
\begin{equation}
\psi_1(x,t) = C_1(t) \exp\bigg(-2 \int_0^x dx' \, \phi(x',t)\bigg), \quad (x,t) \in {\mathbb{R}}^2,     \label{3.319}
\end{equation}
then it becomes unitarily equivalent to $H_2(t)$ for each $t \in {\mathbb{R}}$, see \eqref{3.179}. 

Similarly, for each $t \in {\mathbb{R}}$, 
\begin{equation}
\begin{rcases}
0 \notin \sigma_p(H_1(t)) \\
0 \in \sigma_p(H_2(t))
\end{rcases} \Longleftrightarrow \phi_+ < 0 < \phi_-.     \label{3.320} 
\end{equation}
In this case $H_1(t)$ and $H_2(t)$ are obviously once again not isospectral, however, when restricting $H_2(t)$ to the orthogonal complement of its one-dimensional kernel generated by 
\begin{equation}
\psi_2(x,t) = C_2(t) \exp\bigg(2 \int_0^x dx' \, \phi(x',t)\bigg), \quad (x,t) \in {\mathbb{R}}^2,    \label{3.321}
\end{equation}
then it becomes unitarily equivalent to $H_1(t)$ for each $t \in {\mathbb{R}}$, see again \eqref{3.179}. 

Employing the analog of \eqref{3.315} one verifies that neither $H_1(t)$ nor $H_2(t)$ supports a zero-energy resonance that would be associated with a nontrivial $L^{\infty}({\mathbb{R}};dx)$-solution at zero energy (which has to be linearly independent of the zero-energy eigenfunction $\psi_j$ of $H_j$, $j=1,2$, in connection with \eqref{3.318} and \eqref{3.320}). 

Next, suppose that either $\phi_{\pm} > 0$ or $\phi_{\pm} < 0$: In these cases one shows with the help of \eqref{3.315}, \eqref{3.319}, and \eqref{3.321} that  for each $t \in {\mathbb{R}}$, $H_j(t)$, $j=1,2$, have neither a zero-energy eigenvalue nor a zero-energy resonance. Moreover, in both cases $\phi_{\pm} > 0$ and $\phi_{\pm} < 0$, $H_1(t)$ and $H_2(t)$ are unitarily equivalent for each $t \in {\mathbb{R}}$. \\[1mm]
\noindent 
$(ii)$. Suppose that $\phi_- = 0$, $\phi_+ > 0$: Then, once more employing \eqref{3.315}, \eqref{3.319}, and \eqref{3.321} one concludes that for each $t \in {\mathbb{R}}$, $H_1(t)$ and $H_2(t)$ have no zero-energy eigenvalue; in addition, $H_1(t)$ has a zero-energy resonance, but $H_2(t)$ does not. Moreover, $H_1(t)$ and $H_2(t)$ are unitarily equivalent for each $t \in {\mathbb{R}}$. \\[1mm] 
$(iii)$. Suppose that $\phi_- = 0$, $\phi_+ < 0$: Then item $(ii)$ with $H_1(t)$ and $H_2(t)$ interchanged, $t \in {\mathbb{R}}$, holds. \\[1mm] 
$(iv)$. Suppose that $\phi_- = 0 = \phi_+$: Then once more employing \eqref{3.315}, \eqref{3.319}, and \eqref{3.321} one concludes that for each $t \in {\mathbb{R}}$, neither $H_1(t)$ nor $H_2(t)$ has a zero-energy eigenvalue. However, both, $H_1(t)$ and $H_2(t)$ now have a zero-energy resonance for each $t \in {\mathbb{R}}$. Once more,  in this case $H_1(t)$ and $H_2(t)$ are unitarily equivalent for each $t \in {\mathbb{R}}$. 
\hfill $\diamond$
