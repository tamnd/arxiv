<!-- definition-1: body_md -->
\label{def_2_1} 
    Let $M$ be a closed surface of nonpositive curvature and genus at least $2$. Define 
    \begin{displaymath} 
        F_{\infty}\coloneq\left\{x\in M \mid \nabla^{j}K(x)=0, \forall j\geq 0\right\}. 
    \end{displaymath} 
 
    Here $\nabla^{j} K$ is the $j$-th covariant derivative of the Gaussian curvature function $K$ with the convention $\nabla^{0}K=K$.

<!-- theorem-4: body_md -->
\label{thm_2_4} 
    On a closed, connected, smooth, rank-one manifold $M$ of nonpositive sectional curvature, let $\mu_L$ be the normalized Liouville measure. Then 
    \begin{displaymath} 
        \mu_{L}(\operatorname{Sing})\leq \dfrac{\operatorname{Vol}(F^{J}_{\infty})}{\operatorname{Vol}(M)}. 
    \end{displaymath}

<!-- theorem-5: body_md -->
\label{thm_2_5} 
    On a closed, connected, smooth, rank-one manifold $M$ of nonpositive sectional curvature, let $\Omega_1,\ldots,\Omega_N$ be pairwise disjoint compact smooth regions in it, where $N\geq 1$. Assume that: 
    \begin{enumerate} 
        \item All sectional curvatures at each point of $M\setminus\bigcup_{i=1}^N\Omega_i$ are strictly negative.

<!-- lemma-1: body_md -->
\label{lem_3_1} 
    The set $F_{\infty}=\left\{x\in M\mid \nabla^{j}K(x)=0,~j\geq 0\right\}$ is a closed set.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For every $j\ge0$, the function $x\mapsto \nabla^{j}K(x)$ is continuous. Hence, 
    \begin{displaymath} 
        F_{\infty}=\bigcap_{j=0}^{\infty}\{x\in M:\nabla^{j}K(x)=0\} 
    \end{displaymath} 
    is closed.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem_3_2} 
    For each $x\in M$, let 
    \begin{displaymath} 
        \Lambda_{x} = \left\{\bm{v}\in T^{1}_{x}M \mid K(c_{\bm{v}}(t))=0,~\forall t\in\mathbb{R}\right\}. 
    \end{displaymath} 
 
    If $x\notin F_{\infty}$, then $\Lambda_x$ is a finite set, thus has zero angular measure on $T_x^1M\simeq S^1$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Take any point $x\notin F_{\infty}$. If $K(x)<0$, $\Lambda_{x}=\emptyset$ by definition. If $K(x)=0$, then $K$ attains a local maximum at $x$ because $K\leq 0$. Hence $\mathrm{d} K(x)=0$. Since $x\notin F_{\infty}$, we can find the minimal $m\ge 2$, such that $\nabla^{m}K(x)\neq 0$. 
 
    By definition, for any $\bm{v}\in \Lambda_{x}$, $K(c_{\bm{v}}(t))=0$ for any $t\in\mathbb{R}$. Thus, 
    \begin{displaymath} 
        \dfrac{\mathrm{d}^n}{\mathrm{d}t^{n}} K(c_{\bm{v}}(t))=0,\quad\forall n\in\mathbb{N}. 
    \end{displaymath} 
 
    In particular, $\dfrac{\mathrm{d}^{m}}{\mathrm{d}t^{m}} K(c_{\bm{v}}(t))=0$. 
 
    We now claim that 
    \begin{displaymath} 
        \nabla^{n}K(x)(\overbrace{\bm{v},\ldots,\bm{v}}^{n})=\left.\dfrac{\mathrm{d}^{n}}{\mathrm{d}t^{n}}\right\rvert_{t=0}K(c_{\bm{v}}(t)). 
    \end{displaymath} 
 
    We prove this claim by induction. When $n=1$, by the chain rule 
    \begin{displaymath} 
        \left.\dfrac{\mathrm{d}}{\mathrm{d}t}\right\rvert_{t=0}K(c_{\bm{v}}(t))=\langle \operatorname{grad} K(c_{\bm{v}}(t)),c'_{\bm{v}}(t)\rangle\rvert_{t=0}=\langle \operatorname{grad} K(x),\bm{v}\rangle=\mathrm{d} K(x)(\bm{v}). 
    \end{displaymath} 
 
    Suppose the claim is true for $n=k$, then we can compute that 
    \begin{displaymath} 
        \begin{aligned} 
            \dfrac{\mathrm{d}^{k+1}}{\mathrm{d}t^{k+1}}K(c_{\bm{v}}(t)) & = \dfrac{\mathrm{d}}{\mathrm{d}t}[\nabla^{k} K(c_{\bm{v}}(t))(\overbrace{c'_{\bm{v}}(t),\ldots,c'_{\bm{v}}(t)}^{k})]\\ 
                                                                        & =(\nabla_{c'_{\bm{v}}(t)}\nabla^{k}K(c_{\bm{v}}(t)))(\overbrace{c'_{\bm{v}}(t),\ldots,c'_{\bm{v}}(t)}^{k})\\ 
                                                                        & +\sum_{j=1}^k \nabla^{k}K(c_{\bm{v}}(t))(c'_{\bm{v}}(t),\ldots,\overbrace{\nabla_{c'_{\bm{v}}(t)}c'_{\bm{v}}(t)}^{j},\ldots, c'_{\bm{v}}(t)) \\ 
        \end{aligned} 
    \end{displaymath} 
 
    Given $c_{\bm{v}}(t)$ is a geodesic, $\nabla_{c'_{\bm{v}}(t)}c'_{\bm{v}}(t)=0$, terms in the summation all vanish. Therefore, we obtain the required formula: 
    \begin{displaymath} 
        \begin{aligned} 
            \left.\dfrac{\mathrm{d}^{k+1}}{\mathrm{d}t^{k+1}}\right\rvert_{t=0}K(c_{\bm{v}}(t)) & =\nabla_{c'_{\bm{v}}(t)}\nabla^{k}K(c_{\bm{v}}(t))(\overbrace{c'_{\bm{v}}(t),\ldots,c'_{\bm{v}}(t)}^{k})\rvert_{t=0} \\ 
                                                                                                & =\nabla^{k+1}K(c_{\bm{v}}(t))(\overbrace{c'_{\bm{v}}(t),\ldots,c'_{\bm{v}}(t)}^{k+1})\rvert_{t=0}                    \\ 
                                                                                                & =\nabla^{k+1} K(x)(\overbrace{\bm{v},\ldots,\bm{v}}^{k+1}). 
        \end{aligned} 
    \end{displaymath} 
 
    Now let $P_{m,x}(\bm{u})\coloneq(\nabla^{m}K(x))(\bm{u},\ldots,\bm{u})$, we can compute the coefficients of $P_{m,x}$ in local coordinates $(x_1,x_2)$ near the point $x$. Start with the first order derivative, we have 
    \begin{displaymath} 
        \nabla K(x)=\dfrac{\partial K(x)}{\partial x_{1}}\mathrm{d}x_{1} + \dfrac{\partial K(x)}{\partial x_{2}}\mathrm{d}x_2=0. 
    \end{displaymath} 
 
    Now the second order covariant derivative $\nabla^{2}K(x)$ is given by 
    \begin{displaymath} 
        \nabla_{j}\nabla_{i}K(x)=\dfrac{\partial^2 K(x)}{\partial x_{j}\partial x_{i}}-\Gamma^{k}_{ij}\dfrac{\partial K}{\partial x_{k}}=\dfrac{\partial^2 K(x)}{\partial x_{j}\partial x_{i}}. 
    \end{displaymath} 
 
    By induction on $l<m$, we obtain 
    \begin{displaymath} 
        0={(\nabla^{l} K(x))}_{i_{1}\cdots i_{l}}=\dfrac{\partial^{l} K(x)}{\partial x_{i_1}\cdots\partial x_{i_l}},\qquad i_1,\ldots,i_l=1,2, 
    \end{displaymath} 
    provided that all lower order derivatives vanish at $x$. In particular, when $l=m$ 
    \begin{displaymath} 
        {\left(\nabla^{m} K(x)\right)}_{i_{1} \cdots i_{m}}=\dfrac{\partial^{m} K(x)}{\partial x_{i_{1}}\cdots\partial x_{i_{m}}}+\sum_{0\leq l\leq m-1} A\cdot B,\qquad i_1,\ldots,i_m=1,2. 
    \end{displaymath} 
 
    In the summation, $A$ stands for lower orders of partial derivatives of $K(x)$, which all vanish at $x$, and $B$ stands for the Christoffel symbols and their derivatives. Therefore, we get 
    \begin{equation}\label{eq_3_1} 
        {(\nabla^{m} K(x))}_{i_1\cdots i_m}=\dfrac{\partial^m K(x)}{\partial x_{i_1}\cdots\partial x_{i_m}}. 
    \end{equation} 
 
    Choose $\bm{u}\in T^{1}_x M$. In the local geodesic normal coordinate with the orthonormal frame $\{\bm{e}_{1}(y),\bm{e}_{2}(y)\}$ for $y$ near $x$, set $\bm{u}=u_{1}\,\bm{e}_{1}(x)+u_{2}\,\bm{e}_{2}(x)=u_{1}\,\partial_{x_1}+u_{2}\,\partial_{x_2}$, we can write 
    \begin{equation}\label{eq_3_2} 
        P_{m,x}(\bm{u})=\nabla^{m} K(x)(\overbrace{\bm{u},\ldots,\bm{u}}^{m})=\sum_{i_{1},\ldots,i_{m}=1}^{2}\dfrac{\partial^{m}K(x)}{\partial x_{i_1}\cdots\partial x_{i_m}}u_{i_1}\ldots u_{i_m}. 
    \end{equation} 
 
    Equation~\eqref{eq_3_2} shows that $P_{m,x}(\bm{u})$ is a nonzero homogeneous polynomial of degree $m$ in $u_1$ and $u_2$. 
 
    Now, for any $\bm{v}\in \Lambda_x$, by definition, the geodesic $c_{\bm{v}}$ is flat, hence 
    \begin{displaymath} 
        P_{m,x}(\bm{v})=\dfrac{\mathrm{d}^{m}}{\mathrm{d}t^{m}}K(c_{\bm{v}}(t))=0. 
    \end{displaymath} 
 
    On $T^1_x M\simeq S^1$, let $\bm{v}=\cos\theta\bm{e}_1(x)+\sin\theta\bm{e}_2(x)$. By homogeneity, the restriction of $P_{m,x}(\bm{v})$ to the unit circle now becomes a  nonzero trigonometric polynomial of $\theta$ of degree at most $m$: 
 
    \begin{equation}\label{eq_3_3} 
        P_{m,x}(\bm{v}(x,\theta))=\sum_{i=0}^{m} a_{i}(x)\cos^{m-i}\theta\sin^{i}\theta, 
    \end{equation} 
    where $a_i(x)$ are smooth functions defined near $x$. For fixed $x$, there are only finitely many $\theta\in S^{1}$, such that $P_{m,x}=0$. Thus, $\Lambda_x$ is a finite set with $0$ angular measure.
<!-- end proof-2 -->
<!-- proposition-1: body_md -->
\label{prop_3_1} 
    For $\mu_{L}$-almost every $\bm{v}\in\Lambda$, $c_{\bm{v}}(\mathbb{R})\subset F_{\infty}$.
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
In local coordinates $\bm{v}=(x,\theta)$, the Liouville measure $\mu_L$ can be written as 
    \begin{equation} 
        \mathrm{d}\mu_L(\bm{v})=c\mathrm{d}A(x)\mathrm{d}\theta, 
    \end{equation} 
    where $\mathrm{d}A$ is the Riemannian area element on $M$, $\mathrm{d}\theta$ is the angular measure on the fiber $T^{1}_x M\simeq S^1$, and $c=\dfrac{1}{2\pi \operatorname{Area}(M)}$ is the normalization constant. From Lemma~\ref{lem_3_2}, we know that 
    \begin{displaymath} 
        \theta_{x}(\Lambda_{x})=0,\quad \forall x\notin F_{\infty}. 
    \end{displaymath} 
 
    Thus, we can conclude that 
    \begin{displaymath} 
        \mu_{L}(\left\{\bm{v}\in\Lambda\mid c_{\bm{v}}(0)\notin F_{\infty}\right\})=c\int_{\pi(\Lambda)\setminus F_{\infty}}\theta_{x}(\Lambda_{x})\,\mathrm{d}A(x)=0. 
    \end{displaymath} 
 
    Here $\Lambda$ is compact, then $\pi(\Lambda)$ is also compact and hence measurable. Note that both $\mu_L$ and $\Lambda$ are invariant under the geodesic flow $\phi_t$, for any $q\in\mathbb{Q}$, we obtain that: 
    \begin{displaymath} 
        \begin{aligned} 
            \mu_L\left(\left\{\bm{v}\in\Lambda\mid c_{\bm{v}}(q)\notin F_{\infty}\right\}\right) & =\mu_L\left(\left\{\bm{v}\in\Lambda\mid c_{\phi_{q}\bm{v}}(0)\notin F_{\infty}\right\}\right)    \\ 
                                                                                                 & =\mu_{L}\left(\phi_{-q}\left\{\bm{v}\in\Lambda\mid c_{\bm{v}}(0)\notin F_{\infty}\right\}\right) \\ 
                                                                                                 & =0. 
        \end{aligned} 
    \end{displaymath} 
 
    By the countable subadditivity of measure, we have 
    \begin{displaymath} 
        \mu_L\left(\bigcup_{q\in\mathbb{Q}}\left\{\bm{v}\in\Lambda\mid c_{\bm{v}}(q)\notin F_{\infty}\right\}\right)=0. 
    \end{displaymath}

    From Lemma~\ref{lem_3_1}, $F_{\infty}$ is a closed set. Therefore, $c_{\bm{u}}(\mathbb{R})\subset F_{\infty}$ as required.

    \begin{displaymath} 
        \begin{aligned} 
            a_{ij}(x,\bm{v}) & =\langle J_{\bm{v}}(\bm{e}_j(x)),\bm{e}_{i}(x) \rangle                                                        \\

<!-- lemma-9: body_md -->
\label{lem_6_3}

This proves the equality of $H$ with the stabilizer. 
 
    The restriction $p|_{\widetilde{\Omega}}$ is a covering of $\Omega$: its local covering charts are obtained by restricting evenly covered neighborhoods in $M$, and its surjectivity follows by lifting paths in $\Omega$. If two points of $\widetilde{\Omega}$ project to the same point, a unique deck transformation carries one to the other. That transformation preserves the component and hence belongs to $H$. Thus the fibers are exactly the $H$-orbits, proving $\widetilde{\Omega}/H\simeq\Omega$. 
 
    Finally, passing from $H$ to $H_0$ gives a covering of degree $[H:H_0]$. Since the actions are isometric, it is a Riemannian covering. A finite covering of the compact manifold with boundary $\Omega$ is compact.

<!-- proposition-4: body_md -->
\label{prop_6_1}

If $k=0$, take any point for $F$, which is invariant under the trivial group $H_{0}$. 
 
    Since $F$ is $H_{0}$-invariant, $x\mapsto d(x,F)$ descends to a continuous function on $\widetilde{\Omega}/H_{0}$. This quotient is compact by Lemma~\ref{lem_6_3}, so this function is bounded.

<!-- lemma-11: body_md -->
\label{lem_6_5} 
    The function $r(x)=d(x,Y)$ is convex along every geodesic in $\widetilde{M}$.
<!-- end lemma-11 -->
<!-- proof-18: body_md -->
Let $c:[0,1]\to\widetilde M$ be an affinely parametrized geodesic, and $a:[0,1]\to Y$ be the affinely parametrized geodesic joining $q(c(0))$ to $q(c(1))$. 
 
    The convexity of the distance between two geodesics in a Hadamard manifold gives that for any $t\in [0,1]$, 
    \begin{displaymath} 
        r(c(t))=d(c(t),Y)\leq d(c(t),a(t))\leq (1-t)r(c(0))+t\cdot r(c(1)). 
    \end{displaymath} 
 
    This is the required convexity.

<!-- lemma-12: body_md -->
\label{lem_6_6} 
    The nearest point projection $q:\widetilde{M}\to Y$ is smooth, and the normal exponential map 
    \begin{displaymath} 
        \exp^{\perp}:NY\longrightarrow\widetilde{M},\qquad (y,\bm{v})\to\exp_{y}(\bm{v}) 
    \end{displaymath} 
    is a diffeomorphism. 
 
    In these coordinates, $q(\exp^{\perp}(y,\bm{v}))=y$, and $f(\exp^{\perp}(y,\bm{v}))=\dfrac{1}{2}{\lVert \bm{v}\rVert}^2$. Thus, $f$ is smooth on $\widetilde{M}$ and $\mathcal{E}$ is a smooth subbundle of $T\widetilde{M}$ of rank $k$.
<!-- end lemma-12 -->
<!-- proof-19: body_md -->
Since $\widetilde{M}$ is a Hadamard manifold and $Y$ is complete, connected, and totally geodesic, $Y$ is closed and totally convex. By \cite[Lemma~3.1]{BishopONeill1969}, the normal exponential map $\exp^\perp:NY\to \widetilde{M}$ is a diffeomorphism. 
 
    Let $\pi_{NY}:NY\to Y$ denote the bundle projection. First variation shows that a minimizing geodesic from $Y$ to $x\in\widetilde{M}$ is normal to $Y$. Uniqueness under $\exp^{\perp}$ gives 
    \begin{displaymath} 
        q\left(\exp^{\perp}(y,\bm{v})\right)=y, \qquad d\left(\exp^{\perp}(y,\bm{v}),Y\right)=\|\bm{v}\|. 
    \end{displaymath} 
    Hence, 
    \begin{displaymath} 
        q=\pi_{NY}\circ{(\exp^{\perp})}^{-1},\qquad f\left(\exp^\perp(y,\bm{v})\right)=\frac12{\|\bm{v}\|}^2. 
    \end{displaymath} 
    These formulas show that $q$ and $f$ are smooth. Finally, smooth dependence of parallel transport along the normal geodesics implies that $\mathcal{E}$ is a smooth rank $k$ subbundle of $T\widetilde{M}$.

<!-- lemma-13: body_md -->
\label{lem_6_7} 
    For each $x\in\widetilde{M}$ and $\bm{w}\in T_{x}\widetilde{M}$, write $\bm{w}=\bm{w}_{\parallel}+\bm{w}_{\perp}$ according to the orthogonal decomposition 
    $T_x{\widetilde{M}}=\mathcal{E}_{x}\oplus\mathcal{E}_{x}^{\perp}$. Then 
    \begin{equation} 
        \operatorname{Hess}_{x} f(\bm{w},\bm{w}) \geq {\|\bm{w}_{\perp}\|}^2. 
    \end{equation} 
    In particular, $\operatorname{Hess} f$ is positive semidefinite and 
    \begin{displaymath} 
        \ker\operatorname{Hess}_{x} f\subset\mathcal{E}_x. 
    \end{displaymath}
<!-- end lemma-13 -->
<!-- proof-20: body_md -->
First assume $x\notin Y$. Let $x(s)$ be a geodesic with $x(0)=x$ and $x'(0)=\bm{w}$, and put $y(s)=q(x(s))$. 
 
    Denote $(y(s),\bm{v}(s))={(\exp^{\perp})}^{-1}(x(s))$. By Lemma~\ref{lem_6_6}, the normal geodesics 
    \begin{displaymath} 
        \sigma_s(t)=\exp_{y(s)}(t\bm{v}(s)), \qquad 0\leq t\leq 1, 
    \end{displaymath} 
    from $y(s)$ to $x(s)$ form a smooth geodesic variation. 
 
    Set $\sigma=\sigma_0$, $T=\partial_t\sigma_s|_{s=0}$, and $J=\partial_s\sigma_s|_{s=0}$. Since the parameter interval has length one, the energy of $\sigma_s$ equals $f(x(s))$. 
 
    The second variation formula gives 
    \begin{displaymath} 
        \begin{aligned} 
            \operatorname{Hess}_{x} f(\bm{w},\bm{w})&=\int_0^1\left({\|D_tJ\|}^2-\langle R(J,T)T,J\rangle\right)\,\mathrm{d}t+{[\langle D_s\partial_s\sigma_s,T\rangle]}_{t=0}^{1}\\ 
                                      &=\int_0^1\left({\|D_tJ\|}^2-\langle R(J,T)T,J\rangle\right)\,\mathrm{d}t. 
        \end{aligned} 
    \end{displaymath} 
 
    Indeed, the acceleration at $t=1$ is zero because $x(s)$ is a geodesic. At $t=0$, the acceleration of $y(s)$ is tangent to $Y$ because $Y$ is totally geodesic, whereas $T(0)$ is normal to $Y$. Thus both endpoint terms vanish. Then, nonpositive curvature implies 
    \begin{displaymath} 
        \operatorname{Hess}_{x} f(\bm{w},\bm{w})\geq\int_0^1{\|D_tJ\|}^2\,\mathrm{d}t. 
    \end{displaymath} 
 
    Let $P_{t\to 1}$ denote parallel transport along $\sigma$ from $\sigma(t)$ to $x$. All vectors in the following identity lie in $T_{x}\widetilde{M}$: 
    \begin{displaymath} 
        \bm{w}-P_{0\to 1}J(0)=\int_0^1 P_{t\to 1}(D_{t}J(t))\,\mathrm{d}t. 
    \end{displaymath} 
    Cauchy--Schwarz inequality therefore yields 
    \begin{displaymath} 
        \int_0^1 {\|D_tJ\|}^2\,\mathrm{d}t\geq{\|\bm{w}-P_{0\to 1}J(0)\|}^2\geq{\|\bm{w}_{\perp}\|}^2, 
    \end{displaymath} 
    since $J(0)=y'(0)\in T_{q(x)}Y$ and $P_{0\to 1}J(0)\in\mathcal{E}_x$. 
 
    If $x\in Y$, use the normal bundle coordinates of Lemma~\ref{lem_6_6}. At $(x,0)$ we have $df_x=0$ and $f(y,\bm{v})=\frac12\|\bm{v}\|^2$. The differential of $\exp^\perp$ at $(x,0)$ identifies the tangent and normal factors with $T_{x}Y$ and ${(T_{x}Y)}^\perp$. Consequently, 
    \begin{displaymath} 
        \operatorname{Hess}_{x} f(\bm{w},\bm{w})={\|\bm{w}_{\perp}\|}^2. 
    \end{displaymath} 
 
    This proves the estimate everywhere. For a positive semidefinite symmetric form, its kernel consists of the vectors on which its quadratic form vanishes. Thus, $\ker\operatorname{Hess}_{x}f\subset\mathcal{E}_{x}$.
<!-- end proof-20 -->
<!-- theorem-7: body_md -->
\label{thm_6_1} 
    Let $\Omega\subset M$ be a compact smooth region. Suppose 
    \begin{displaymath} 
        H=\operatorname{im}\left(\pi_1(\Omega)\longrightarrow\pi_1(M)\right) 
    \end{displaymath} 
    is virtually Abelian of rank $k<n$. Then $\mu_L(\operatorname{Trap}(\Omega))=0$. 
 
    If $k\geq 1$, then 
    \begin{displaymath} 
        \dim_H\operatorname{Trap}(\Omega)\leq n+k-1. 
    \end{displaymath} 
 
    If $k=0$, then $\operatorname{Trap}(\Omega)=\emptyset$.
<!-- end theorem-7 -->
<!-- proof-21: body_md -->
Choose a component $\widetilde{\Omega}$ of $p^{-1}(\Omega)$ and the flat $F\simeq\mathbb{R}^k$ given by Proposition~\ref{prop_6_1}. Write $f(x)=\frac{1}{2} {d(x,F)}^2$ and $\mathcal{E}_{x}=P_{q(x)\to x}(T_{q(x)}F)$. 
 
    Let $\widetilde{c}:\mathbb{R}\to \widetilde{M}$ be a unit-speed geodesic contained in $\widetilde{\Omega}$. By Lemma~\ref{lem_6_5}, the function $t\mapsto d(\widetilde{c}(t),F)$ is convex. It is bounded above by Proposition~\ref{prop_6_1}, and a convex function on the whole real line that is bounded above is therefore constant. 
 
    Thus, $f\circ\widetilde{c}$ is constant, and 
    \begin{displaymath} 
        0=(f\circ\widetilde{c})''(t)=\operatorname{Hess}_{\widetilde{c}(t)}f(\widetilde{c}'(t),\widetilde{c}'(t)). 
    \end{displaymath} 
    Lemma~\ref{lem_6_7} gives 
    \begin{equation}\label{eq_6_1} 
        \widetilde{c}'(t)\in\mathcal{E}_{\widetilde{c}(t)}\qquad\forall t\in\mathbb{R}. 
    \end{equation} 
 
    If $k=0$, the bundle $\mathcal{E}$ has zero-dimensional fibers, which contain no unit vectors. Therefore there is no such geodesic. 
 
    When $1\leq k<n$, the unit sphere bundle 
    \begin{displaymath} 
        T^{1}(\mathcal{E})=\{\bm{w}\in T^{1}\widetilde{M}\mid\bm{w}\in\mathcal{E}_{\pi_{\widetilde{M}}(\bm{w})}\} 
    \end{displaymath} 
    is a smooth submanifold of $T^{1}\widetilde{M}$ of dimension $n+k-1$, where $\pi_{\widetilde{M}}:T^1\widetilde{M}\to \widetilde{M}$ is the projection. Equation~\eqref{eq_6_1} shows that $\operatorname{Trap}_{\widetilde{M}}(\widetilde{\Omega})\subset T^{1}(\mathcal{E})$. 
 
    At each $x\in \widetilde{M}$, the fiber of $T^{1}(\mathcal{E})$ is a sphere of dimension $k-1$ in $T_{x}^1\widetilde{M}\simeq S^{n-1}$. Since $k<n$, it has zero angular measure. The Liouville measure on $T^1\widetilde{M}$ disintegrates as Riemannian volume on $\widetilde{M}$ times angular measure on the fibers, up to a fixed positive normalization. Thus $T^{1}(\mathcal{E})$ has zero Liouville measure in $T^{1}\widetilde{M}$. 
 
    The induced map $Dp:T^1\widetilde{M}\to T^1M$ is a covering and a local isometry. Every geodesic trapped in $\Omega$ has a lift trapped in $\widetilde{\Omega}$. Conversely, projecting a geodesic trapped in $\widetilde{\Omega}$ gives a geodesic trapped in $\Omega$. Hence 
    \begin{displaymath} 
        Dp\left(\operatorname{Trap}_{\widetilde{M}}(\widetilde{\Omega})\right)=\operatorname{Trap}(\Omega). 
    \end{displaymath} 
 
    Cover $T^{1}\widetilde{M}$ by countably many coordinate neighborhoods on which $Dp$ is a diffeomorphism. On each such neighborhood it preserves Liouville null sets and does not increase Hausdorff dimension, by restriction to countably many relatively compact charts. Countable unions therefore give 
    \begin{displaymath} 
        \mu_L(\operatorname{Trap}(\Omega))=0,\qquad\dim_{H}\operatorname{Trap}(\Omega)\leq n+k-1. 
    \end{displaymath}
<!-- end proof-21 -->
<!-- remark-3: body_md -->
The trapped-set estimate requires no additional curvature assumptions on the exceptional region $\Omega$ or its complement. Thus, we expect this estimate to have further applications to study the singular set.

<!-- proof-22: body_md -->
If $\bm{v}\in\operatorname{Sing}$, then the geodesic along $\bm{v}$ cannot pass through a point at which every sectional curvature is strictly negative. It follows that $c_{\bm{v}}(\mathbb{R})\subset\bigcup_{i=1}^N\Omega_i$. Since the finitely many compact sets $\Omega_{i}$ are pairwise disjoint and $c_{\bm v}(\mathbb R)$ is connected, the entire geodesic lies in a single $\Omega_{i}$. Therefore, 
    \begin{displaymath} 
        \operatorname{Sing}\subset\bigcup_{i=1}^{N}\operatorname{Trap}(\Omega_{i}). 
    \end{displaymath} 
 
    By Theorem~\ref{thm_6_1}, each $\operatorname{Trap}(\Omega_i)$ has zero Liouville measure, and hence, the singular set has zero Liouville measure. 
 
    We also have that 
    \begin{displaymath} 
        \dim_{H}(\operatorname{Sing})\leq\max_{i}(n+k_{i}-1)\leq 2n-2. 
    \end{displaymath} 
 
    If all $k_{i}$ vanish, the singular set is empty, which yields that the geodesic flow is Anosov.

<!-- definition-9: body_md -->
\label{def_7_1} 
    On $E_{\bm{v}}$ we define the parallel-transport pullback of the curvature endomorphism $A(\bm{v},t):E_{\bm{v}}\to E_{\bm{v}}$ by 
    \begin{equation} 
        A(\bm{v},t)(\bm{w})=P^{-1}_{\bm{v},t}(R(P_{\bm{v},t}(\bm{w}),c'_{\bm{v}}(t))c'_{\bm{v}}(t)). 
    \end{equation} 
 
    Then, $A(\bm{v},t)$ is a self-adjoint linear operator.
<!-- end definition-9 -->
<!-- proof-23: body_md -->
We first show that this operator is well-defined. Fix $\bm{v}$ and choose $\bm{w}\in E_{\bm{v}}$, we need to show $A(\bm{v},t)(\bm{w})\in E_{\bm{v}}$ as well. Denote $\bm{w}_t=P_{\bm{v},t}\bm{w}$, the curvature symmetries imply 
    \begin{displaymath} 
        \langle R(\bm{w}_t,c'_{\bm{v}}(t))c'_{\bm{v}}(t),c'_{\bm{v}}(t)\rangle=0. 
    \end{displaymath} 
    Thus, the curvature vector lies in $E_{\phi_{t}(\bm{v})}$, so $A(\bm{v},t)$ is well-defined. 
 
    Linearity follows immediately from the fact that $P_{\bm{v},t}$, its inverse, and the curvature endomorphism are all linear maps. To verify that it is self-adjoint, choose $\bm{u},\bm{w}\in E_{\bm{v}}$ and denote $\bm{u}_t=P_{\bm{v},t}(\bm{u})$ and $\bm{w}_t=P_{\bm{v},t}(\bm{w})$. 
 
    \begin{displaymath} 
        \begin{aligned} 
            \langle A(\bm{v},t)(\bm{u}),\bm{w}\rangle & =\langle P^{-1}_{\bm{v},t}(R(\bm{u}_t,c'_{\bm{v}}(t))c'_{\bm{v}}(t)),\bm{w}\rangle \\ 
                                                      & =\langle R(\bm{u}_t,c'_{\bm{v}}(t))c'_{\bm{v}}(t),\bm{w}_{t}\rangle                    \\ 
                                                      & =R(\bm{u}_t, c'_{\bm{v}}(t),c'_{\bm{v}}(t),\bm{w}_t)                                   \\ 
                                                      & =R(\bm{w}_t,c'_{\bm{v}}(t),c'_{\bm{v}}(t),\bm{u}_t)                                    \\ 
                                                      & =\langle A(\bm{v},t)\bm{w},\bm{u}\rangle. 
        \end{aligned} 
    \end{displaymath} 
 
    Therefore, $A^{\ast}(\bm{v},t)=A(\bm{v},t)$ is self-adjoint.
<!-- end proof-23 -->
<!-- definition-10: body_md -->
\label{def_7_2} 
    For any $\bm{v}\in T^{1}M$, define the operator $Q(\bm{v}):E_{\bm{v}}\to E_{\bm{v}}$ by 
    \begin{displaymath} 
        Q(\bm{v})(\bm{w})=\int_{-1}^{1} A^{\ast}(\bm{v},t)\circ A(\bm{v},t)(\bm{w})\,\mathrm{d}t. 
    \end{displaymath} 
 
    Then, $Q(\bm{v})$ is self-adjoint and positive semidefinite.
<!-- end definition-10 -->
<!-- proof-24: body_md -->
Choose any $\bm{w}\in E_{\bm{v}}$. We can see that $Q(\bm{v})$ is well-defined from $E_{\bm{v}}$ to itself. This is because $A(\bm{v},t)$ preserves $E_{\bm{v}}$, and so does $A^{\ast}(\bm{v},t)\circ A(\bm{v},t)$. This holds for any $t\in [-1,1]$ and hence, the integral also lies in $E_{\bm{v}}$. $A^{\ast}(\bm{v},t)A(\bm{v},t)$ is self-adjoint for all $t$, and so is $Q(\bm{v})$. To verify positive semidefiniteness, we compute  
 
    \begin{displaymath} 
        \begin{aligned} 
            \langle Q(\bm{v})(\bm{w}),\bm{w}\rangle & =\langle \int_{-1}^{1} A^{\ast}(\bm{v},t)\circ A(\bm{v},t)(\bm{w})\,\mathrm{d}t,\bm{w}\rangle \\ 
                                                    & =\int_{-1}^{1}\langle A^{\ast}(\bm{v},t)\circ A(\bm{v},t)(\bm{w}),\bm{w}\rangle\,\mathrm{d}t  \\ 
                                                    & =\int_{-1}^{1}\langle A(\bm{v},t)(\bm{w}),A(\bm{v},t)(\bm{w})\rangle\,\mathrm{d}t             \\ 
                                                    & =\int_{-1}^{1}{\lVert A(\bm{v},t)(\bm{w})\rVert}^2\,\mathrm{d}t\geq 0. 
        \end{aligned} 
    \end{displaymath}
<!-- end proof-24 -->
<!-- definition-11: body_md -->
\label{def_7_3} 
    Define the function $D:T^{1}M\to\mathbb{R}$ by 
    \begin{equation} 
        D(\bm{v})\coloneq \det Q(\bm{v}). 
    \end{equation} 
 
    From Definition~\ref{def_7_2}, we have $Q(\bm{v})$ positive semidefinite, thus, $D(\bm{v})\ge 0$.

<!-- theorem-8: body_md -->
\label{thm_7_1} 
    A vector $\bm v\in T^1M$ is singular if and only if $D(\bm{v})=0$.
<!-- end theorem-8 -->
<!-- proof-25: body_md -->
Suppose that $\bm{v}$ lies in the singular set, $\operatorname{rank}(\bm{v})\ge 2$. By the definition of rank, we can find a nonzero parallel normal Jacobi field $Y$ along $c_{\bm{v}}$. Denote $\bm{w}=Y(0)\in E_{\bm{v}}$ which is a nonzero vector and $Y(t)=P_{\bm{v},t}(\bm{w})$. 
 
    $Y(t)$ satisfies the Jacobi equation, that is 
    \begin{displaymath} 
        \dfrac{\mathrm{D}^2 Y}{\mathrm{d}t^{2}}+R(Y(t),c'_{\bm{v}}(t))c'_{\bm{v}}(t)=0. 
    \end{displaymath} 
 
    Since $Y(t)$ is parallel, $\dfrac{\mathrm{D}Y}{\mathrm{d}t}(t)=0$, thus, $\dfrac{\mathrm{D}^{2}Y}{\mathrm{d}t^2}(t)=0$. Therefore, we get 
    \begin{displaymath} 
        R(Y(t),c'_{\bm{v}}(t))c'_{\bm{v}}(t)=0. 
    \end{displaymath} 
 
    Hence, 
    \begin{displaymath} 
        A(\bm{v},t)(\bm{w})=P^{-1}_{\bm{v},t}(R(P_{\bm{v},t}(\bm{w}),c'_{\bm{v}}(t))c'_{\bm{v}}(t))=0, 
    \end{displaymath} 
    $Q(\bm{v})(\bm{w})=0$ and $D(\bm{v})=0$. 
 
    Conversely, suppose that $D(\bm{v})=0$, then $\ker Q(\bm{v})$ is non-trivial. Choose a nonzero vector $\bm{w}\in \ker Q(\bm{v})\subset E_{\bm{v}}$, we have 
    \begin{displaymath} 
        0=\langle Q(\bm{v})(\bm{w}),\bm{w}\rangle=\int_{-1}^{1} {\lVert A(\bm{v},t)(\bm{w})\rVert}^{2}\,\mathrm{d}t. 
    \end{displaymath} 
 
    $A(\bm{v},t)(\bm{w})$ is continuous, this implies $A(\bm{v},t)(\bm{w})=0$ for $t\in [-1,1]$. Since the metric is real analytic, geodesics and parallel transport depend real analytically on time. Thus, $t\mapsto A(\bm{v},t)\bm{w}$ is real analytic on $\mathbb{R}$. By the unique continuation of the analytic function, we get 
    \begin{displaymath} 
        A(\bm{v},t)(\bm{w})=0,\qquad t\in\mathbb{R}. 
    \end{displaymath} 
 
    Now let $Y(t)=P_{\bm{v},t}(\bm{w})$, from the above argument, we have 
    \begin{displaymath} 
        R(Y(t),c'_{\bm{v}}(t))c'_{\bm{v}}(t)=0,\qquad t\in\mathbb{R}. 
    \end{displaymath} 
 
    Since $Y$ is parallel, $D_{t}^{2}Y=0$. Together with the preceding curvature identity, this shows that $Y$ is a nonzero parallel normal Jacobi field along $c_{\bm{v}}$. Thus, $\bm{v}\in\operatorname{Sing}$.
<!-- end proof-25 -->
<!-- corollary-2: body_md -->
For a real-analytic metric, 
    \begin{displaymath} 
        \operatorname{rank}(\bm{v})=1+\dim\ker Q(\bm{v}). 
    \end{displaymath}
<!-- end corollary-2 -->
<!-- proof-26: body_md -->
In local real-analytic frames of the normal bundle, the matrix entries of $A(\bm{v},t)$ are jointly real analytic in $(\bm{v},t)$. Integration over the fixed interval $[-1,1]$ therefore yields a real-analytic endomorphism $Q$. Consequently, $D=\det Q$ is a globally defined real analytic function on $T^{1}M$. 
 
    Given $\operatorname{rank}(M)=1$, we can find some $\bm{v}\in T^{1}M$ with rank $1$. By Theorem~\ref{thm_7_1}, $D(\bm{v})>0$, so $D$ is not identically zero. 
 
    Since $M$ is connected and $\dim M\geq2$, $T^1M$ is connected. The nontrivial real-analytic function $D$ therefore has a zero set of measure zero with respect to the Liouville measure. By Theorem~\ref{thm_7_1},    
 
    \begin{displaymath} 
        \mu_{L}(\operatorname{Sing})=\mu_{L}(D^{-1}(0))=0, 
    \end{displaymath} 
    ergodicity of the geodesic flow follows.
<!-- end proof-26 -->
