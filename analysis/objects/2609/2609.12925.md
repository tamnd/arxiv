\label{lema:prop_K}
    For $P\geq 0$ the following statements hold:
    \begin{enumerate}
        \item $\mathcal{K}(P) = P \mathcal{K}(1)$, in particular $P\mapsto \mathcal{K}(P)$ is monotone, i.e.\ from $0\leq P' \leq P$ follows $\mathcal{K}(P') \subseteq \mathcal{K}(P)$, and $0\in \mathcal{K}(P)$ for all $P\geq 0$.
        \item Boundedness: $|\boldsymbol{\tau} |_F \leq c_\mathcal{K} P$ for all $\boldsymbol{\tau} \in \mathcal{K}(P)$ with $c_\mathcal{K} = \sqrt{2+1/(2e^2)}$.
        \item For $P>0$, $\mathcal{K}(P)$, understood in the three-dimensional space $(\tau_I, \operatorname{dev}\boldsymbol{\tau}) \in \mathbb R \times \mathbb R^2$, is an ellipsoid with centre $-\frac{P}{2} \boldsymbol{\mathrm{I}}$, hence strictly convex with $C^\infty$ boundary, and the outer unit normal $\partial \mathcal{K}(P) \rightarrow S^2$ is bijective.
    \end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For \textit{1.}, the case $P=0$ follows directly from $\mathcal{K}(0)=\{0\}$.
    For $P>0$ we use the scaling of the invariants,
    \begin{align*}
        (P\boldsymbol{\tau})_I = P\tau_I \quad (P\boldsymbol{\tau})_{II} = P\tau_{II},
    \end{align*}
    which leads directly to $F(P\boldsymbol{\tau}, P) = F(\boldsymbol{\tau}, 1)$ and consequently $P\boldsymbol{\tau} \in \mathcal{K}(P) \Leftrightarrow \boldsymbol{\tau} \in \mathcal{K}(1)$. The monotonicity
    follows from $\mathcal{K}(P') = \frac{P'}{P}\mathcal{K}(P) \subseteq \mathcal{K}(P)$, since $\mathcal{K}(P)$ is convex and $0\in\mathcal{K}(P)$. Finally $F(0, P) = 0$ for all $P>0$, so $0$ lies on the boundary of $\mathcal{K}(P)$, and $0\in \mathcal{K}(0)$ by definition.

    The definition of $F$ in \eqref{eq:def_F} shows $\tau_I \in [-P,0]$ and $|\tau_{II}|\leq P/(2e)$ for $\boldsymbol{\tau} \in \mathcal{K}(P)$. Then the boundedness follows from the estimate
    \begin{align*}
        |\boldsymbol{\tau}|_F^2 = 2|\tau_I|^2 + 2|\tau_{II}|^2 \leq 2 P^2 + \frac{P^2}{2e^2} \ .
    \end{align*}
    The last claim follows from the fact that $F(\cdot, P)$ is a quadratic polynomial with positive definite principal part in the coordinates $(\tau_I, \operatorname{dev}\boldsymbol{\tau})$. Then the set $\{F\leq 0\}$ is a non-degenerate ellipsoid and the Gauss map is a diffeomorphism onto $S^2$.

\label{lem:prop_D}
    For $P\geq 0$ and $\boldsymbol{\xi} \in \mathbb S$ we have
    \begin{align}
        D(P,\boldsymbol{\xi}) := \sup_{\boldsymbol{\tau}\in\mathcal{K}(P)} \boldsymbol{\tau} : \boldsymbol{\xi} = \frac{P}{2}(\Delta(\boldsymbol{\xi})-\xi_I) \ .
        \label{eq:def_D}
    \end{align}
    In particular $D(P,\cdot)$ is convex, homogeneous of degree 1, non-negative,
    Lipschitz continuous with constant $c_\mathcal{K} P$, and we have
    \begin{align}
        D(P,\boldsymbol{\xi}) = 0 \Leftrightarrow \boldsymbol{\xi} = \alpha \boldsymbol{\mathrm{I}} \ \text{for some}\ \alpha \geq 0 \label{eq:dissipationfree}
    \end{align}
    for all $P>0$. Thus isotropic expansion is dissipation-free. Positive
divergence alone does not imply zero dissipation. This property does not
exclude positive principal stresses elsewhere on the yield ellipse.
Additionally the pointwise estimate
    \begin{align}
        D(P, \boldsymbol{\xi}) \geq P(\xi_I)^- \label{eq:negativ_part_D}
    \end{align}
    holds, where $(\cdot)^-$ is the negative part.

<!-- proof-2: body_md -->
First we show \eqref{eq:def_D}. For $P=0$ it follows immediately. For $P>0$ we estimate with Cauchy--Schwarz and
    the ellipse condition \eqref{eq:def_F}
    \begin{align*}
        \boldsymbol{\tau} : \boldsymbol{\xi} \leq \frac{P}{2}(\Delta(\boldsymbol{\xi})-\xi_I)
    \end{align*}
    for all $\boldsymbol{\tau} \in \mathcal{K}(P)$. For $\boldsymbol{\xi}\ne0$, the maximum is attained by
    \begin{align*}
        \boldsymbol{\tau}^* = \frac{P}{2}\left(\frac{\xi_I}{\Delta(\boldsymbol{\xi})}-1\right) \boldsymbol{\mathrm{I}} + \frac{P}{e^2\Delta(\boldsymbol{\xi})}\operatorname{dev}\boldsymbol{\xi} \in \mathcal{K}(P) \ .
    \end{align*}
    For $\boldsymbol{\xi}=0$, every $\boldsymbol{\tau}\in\mathcal{K}(P)$ is a maximiser. For $P=0$ the
    only admissible stress is zero.
    Convexity, positive homogeneity and non-negativity (by \eqref{eq:xi_estimate}) follow from the properties of the support function of bounded sets with $0 \in \mathcal{K}(P)$. Lemma \ref{lema:prop_K} leads to the Lipschitz constant
    \begin{align*}
        \sup_{\boldsymbol{\tau}\in\mathcal{K} (P)} |\boldsymbol{\tau}|_F \leq c_\mathcal{K} P \ .
    \end{align*}
    The characterisation of zero dissipation holds since $D=0$ if and only if $\Delta(\boldsymbol{\xi}) = \xi_I$, and from \eqref{eq:xi_estimate} follows $\operatorname{dev} \boldsymbol{\xi} = 0$ and $\xi_I \geq 0$.
    The last estimate follows directly from
    \begin{align*}
        \Delta(\boldsymbol{\xi}) - \xi_I \geq |\xi_I| - \xi_I = 2 (\xi_I)^- \ .
    \end{align*}

<!-- lemma-3: body_md -->
\label{lem:hubler}
    For $\mu >0$, $P\geq 0$ and $\boldsymbol{\xi} \in \mathbb S$, let $\Delta_\mu := \max (\Delta(\boldsymbol{\xi}),\mu)$ and
    \begin{align*}
        \boldsymbol{\tau}_\mu (P,\boldsymbol{\xi}) := \frac{P}{2} \left(\frac{\xi_I}{\Delta_\mu}-1\right) \boldsymbol{\mathrm{I}} + \frac{P}{e^2\Delta_\mu}\operatorname{dev}\boldsymbol{\xi} \ .
    \end{align*}
    Then the following statements hold:
    \begin{enumerate}
        \item $\boldsymbol{\tau}_\mu(P,\boldsymbol{\xi}) \in \mathcal{K}(P)$ for all $\boldsymbol{\xi}$. On $\{ \Delta (\boldsymbol{\xi})\geq \mu\}$,
        $\boldsymbol{\tau}_\mu$ lies on $\partial\mathcal{K}(P)$ and coincides with the
        maximiser of the support function.
        \item
        \begin{align}
            \boldsymbol{\tau}_\mu (P, \boldsymbol{\xi}) : \boldsymbol{\xi} = \frac{P}{2}\Big(\frac{\Delta(\boldsymbol{\xi})^2}{\Delta_\mu} - \xi_I \Big) \geq D(P, \boldsymbol{\xi}) - \frac{P\mu}{8}
            \label{eq:hubler}
        \end{align}
        \item The function $\boldsymbol{\tau}_\mu(P,\cdot)$ is globally Lipschitz continuous on $\mathbb S$ with constant $C_e P/\mu$, where $C_e$ only depends on $e$.
        \item $\boldsymbol{\tau}_\mu$ is linear in $P$: we have $\boldsymbol{\tau}_\mu(P,\boldsymbol{\xi}) = P\boldsymbol{\tau}_\mu (1,\boldsymbol{\xi})$, in particular $|\boldsymbol{\tau}_\mu(P, \boldsymbol{\xi})- \boldsymbol{\tau}_\mu(P',\boldsymbol{\xi})|_F \leq c_\mathcal{K} |P-P'|$ and $\boldsymbol{\tau}_\mu(0,\cdot) \equiv 0$.
    \end{enumerate}

<!-- proof-3: body_md -->
We calculate the invariants of $\boldsymbol{\tau}_\mu$
    \begin{align*}
        \tau_I = \frac{P}{2}\Big(\frac{\xi_I}{\Delta_\mu}-1\Big) \quad \text{and} \quad \tau_{II} = \frac{P}{2e^2\Delta_\mu}\xi_{II} \ .
    \end{align*}
    \textit{1.}
    For $P=0$ the stress is zero and the claim is immediate. For $P>0$
    we insert these invariants into \eqref{eq:def_F} and get
    \begin{align*}
        1\geq F(\boldsymbol{\tau}_\mu, P) +1 =\frac{\xi_I^2+e^{-2}\xi_{II}^2}{\Delta_\mu^2} = \frac{\Delta(\boldsymbol{\xi})^2}{\Delta_\mu^2} \ .
    \end{align*}
    For $\Delta \geq \mu$ we have $\Delta_\mu = \Delta$, the quotient
    is $1$ and $\boldsymbol{\tau}_\mu$ is the maximiser from the proof of Lemma \ref{lem:prop_D}.

    \textit{2.}
    A direct computation yields
    \begin{align*}
        \boldsymbol{\tau}_\mu : \boldsymbol{\xi} = \frac{P}{2\Delta_\mu}(\xi_I^2+e^{-2}\xi_{II}^2) - \frac{P}{2}\xi_I \ .
    \end{align*}
    For $\Delta \geq \mu$ this coincides with $D(P,\boldsymbol{\xi})$.
    For $\Delta < \mu$, the gap satisfies
    \begin{align*}
        D(P, \boldsymbol{\xi}) - \boldsymbol{\tau}_\mu : \boldsymbol{\xi} = \frac{P}{2} \Big(\Delta -\frac{\Delta^2}{\mu}\Big) = \frac{P}{2}\Delta \Big(1-\frac{\Delta}{\mu}\Big) \leq \frac{P}{2} \frac{\mu}{4} = \frac{P\mu}{8} \ .
    \end{align*}
    \textit{3.}
    On the open set $\{\Delta < \mu\}$ we have $\Delta_\mu = \mu$ and $\boldsymbol{\tau}_\mu(P,\cdot)$ is affine with constant $C_e P/\mu$.
    On $\{\Delta >\mu\}$ we have $\boldsymbol{\tau}_\mu(P,\boldsymbol{\xi}) = \frac{P}{2}\big(\Delta(\boldsymbol{\xi})^{-1}(\xi_I\boldsymbol{\mathrm{I}}+2e^{-2}\operatorname{dev}\boldsymbol{\xi})-\boldsymbol{\mathrm{I}} \big)$ and the
    quotient rule leads to $|\nabla \boldsymbol{\tau}_\mu| \leq C_e P/\Delta \leq C_e P/\mu$. Since $\boldsymbol{\tau}_\mu(P,\cdot)$ is continuous, $\{\Delta \leq \mu\}$ is convex and every segment $[\boldsymbol{\xi}_1, \boldsymbol{\xi}_2]$ cuts the interface in at most 2 points, a piecewise application of the fundamental theorem of calculus gives the global Lipschitz constant as the maximum of both constants.

    \textit{4.} $\boldsymbol{\tau}_\mu$ is linear in $P$ and the constant follows from \textit{1.}\ and Lemma \ref{lema:prop_K}: $|\boldsymbol{\tau}_\mu(1,\boldsymbol{\xi})|_F \leq c_\mathcal{K}$.

<!-- lemma-4: body_md -->
\label{lem:bdd}
    Under Assumption \ref{ass}, the following statements hold:
    \begin{enumerate}
        \item $\mathcal{D} : V \rightarrow [0,\infty]$ is convex, positively homogeneous of degree 1, and weakly lower semicontinuous on $V$. Also $\mathcal{D} (0)=0$, so $\operatorname{dom} \mathcal{D} \neq \emptyset$.
        \item
        For $\boldsymbol{v} \in W^{1,3}(\Omega)^2$ with arbitrary trace, there is a relaxed
        boundary term
        \begin{align}
            \mathcal{D} (\boldsymbol{v}) = \int_\Omega D(P(x),\varepsilon(\boldsymbol{v})(x)) \:\mathrm{d} x + \int_{\Gamma_D} D(P(x), -(\boldsymbol{v}\odot \nu) ) \:\mathrm{d} \mathcal{H}^1 \ , \label{eq:int_darstellung}
        \end{align}
        with $(\boldsymbol{v} \odot \nu) := \frac{1}{2}(\boldsymbol{v} \otimes \nu + \nu \otimes \boldsymbol{v})$.
    \end{enumerate}

<!-- proof-4: body_md -->
\textit{1.}
    For any $\boldsymbol{\tau} \in \Sigma$ the map $\boldsymbol{v} \mapsto \int_\Omega \boldsymbol{v} \cdot \operatorname{\mathrm{div}} \boldsymbol{\tau} \:\mathrm{d} x$ is linear and (weakly) $L^3$-continuous, since $\operatorname{\mathrm{div}}\boldsymbol{\tau}\in V^*$. Then $\mathcal{D}$, as the pointwise
    supremum over such functionals, is convex and weakly lower semicontinuous. For the homogeneity we notice that
    $\Sigma$ is independent of $\boldsymbol{v}$ and the functional is linear in $\boldsymbol{v}$. Thus homogeneity of degree 1 holds, since
    the supremum scales accordingly. Since $0 \in \Sigma$ and $\operatorname{\mathrm{div}} 0 =0$, we have $\mathcal{D} \geq 0$ and $\mathcal{D} (0) =0$.

    \textit{2.}
    The inequality "$\leq$" is proved in Lemma \ref{lem:prove_leq} in the appendix
    by a slicing argument \cite{A83}. We prove the reverse inequality.
    Let $\mu, \rho, h, \eta >0$ and $k,j \in \mathbb N$.

    Step 1: Let $I := \bar \Gamma_D \cap \bar \Gamma_N$. From the disjointness of $\Gamma_D$ and $\Gamma_N$
    follows $\Gamma_D \cap I = \emptyset$. Choose $\zeta_\eta \in \mathrm{Lip}(\mathbb R^2)$ with
    $\zeta_\eta=0$ in $U_{\eta/2}(I):=\{x\in \mathbb R^2 :\operatorname{dist}(x,I)<\eta/2\}$ and $\zeta_\eta =1$
    in $\mathbb R^2 \setminus U_{\eta}(I)$, $0\leq \zeta_\eta \leq 1$. Additionally let
    $\vartheta_h(x):= \psi (\operatorname{dist}(x,\bar\Gamma_D)/h)$ with $\psi \in \mathrm{Lip}(\mathbb R)$ and
    $\psi = 0$ in $[1,\infty)$, $\psi =1$  in $(-\infty,1/2]$ and $0\leq \psi \leq 1$.
    Set $K_\eta := \bar\Gamma_D \setminus U_{\eta/4}(I)$, which is compact, and
    since $K_\eta \cap \bar\Gamma_N \subseteq I \setminus U_{\eta/4}(I) = \emptyset$ we have
    $d_\eta := \operatorname{dist}(K_\eta,\bar\Gamma_N)>0$, with $d_\eta = \infty$ for $K_\eta = \emptyset$.

    For $h<\min (\eta/4,d_\eta)$, $\vartheta_h\zeta_\eta$ vanishes on a neighbourhood of $\bar\Gamma_N$.
    Indeed, if $\vartheta_h(x)\zeta_\eta(x) \neq 0$, then $\operatorname{dist}(x,\bar\Gamma_D)<h$ and $\operatorname{dist}(x,I)\geq \eta/2$,
    and for the nearest point $y\in \bar\Gamma_D$ we have
    \begin{align*}
        \operatorname{dist}(y,I) \geq \eta/2 - h \geq \eta/4 \ ,
    \end{align*}
    hence $y\in K_\eta$ and $\operatorname{dist}(x,\bar\Gamma_N) \geq d_\eta-h >0$.
    Let $\chi_h \in C^\infty_c(\Omega)$ with $0\leq \chi_h\leq 1$, $\chi_h =1$ on $\{x:\operatorname{dist}(x,\partial\Omega)\geq 3h\}$,
    $\operatorname{supp} \chi_h \subset \{x : \operatorname{dist}(x,\partial\Omega)\geq 2h \} $.
    The supports of $\chi_h$ and $\vartheta_h$ are disjoint.

    Step 2: Let $g := - (\boldsymbol{v} \odot \nu) \in L^3(\Gamma_D; \mathbb S)$. On the compact metric space
    $(\bar\Gamma_D,|\cdot|)$ with the finite measure $\mathcal{H}^1$, Lipschitz functions are dense in $L^3$.
    Choose Lipschitz functions $G_j$ with $\Vert G_j - g\Vert_{L^3(\Gamma_D)}\rightarrow 0$ and
    extend them componentwise (McShane \cite{M34}, in a basis of $\mathbb S$) Lipschitz continuously to $\mathbb R^2$. Additionally let
    $\boldsymbol{v}_k \in C^\infty(\mathbb R^2; \mathbb R^2)$ with $\boldsymbol{v}_k \rightarrow \boldsymbol{v}$ in $W^{1,3}(\Omega)$.
    By the Tietze extension theorem and multiplication with a compactly supported continuous cutoff
    equal to $1$ on $\bar\Omega$, choose $\widetilde P\in C_c(\mathbb R^2)$ with
    $\widetilde P=P$ on $\bar\Omega$ and $0\leq\widetilde P\leq\Vert P\Vert_{C(\bar\Omega)}$.
    Let
    \begin{align*}
        \omega_{\widetilde P}(r):=\sup_{|x-y|\leq r}|\widetilde P(x)-\widetilde P(y)|,
        \qquad x,y\in\mathbb R^2,
    \end{align*}
    so that $\omega_{\widetilde P}(r)\rightarrow0$ as $r\downarrow0$ by uniform continuity.
    For the standard non-negative mollifier $\varphi_\rho$ supported in $B_\rho(0)$, define
    \begin{align*}
        P_\rho := (\widetilde P * \varphi_\rho - \omega_{\widetilde P}(\rho))^+ \ .
    \end{align*}
    Then $P_\rho$ is Lipschitz continuous and
    \begin{align}
        0\leq P_\rho\leq P\quad\text{on}\ \bar\Omega,
        \qquad \Vert P-P_\rho\Vert_{C(\bar\Omega)}\leq2\omega_{\widetilde P}(\rho) \ .
        \label{eq:Prho}
    \end{align}
    Indeed, $|\widetilde P*\varphi_\rho-P|\leq\omega_{\widetilde P}(\rho)$ on $\bar\Omega$.
    Subtracting this modulus and taking the positive part gives
    $(P-2\omega_{\widetilde P}(\rho))^+\leq P_\rho\leq P$ there.

Step 3:
Define
\begin{align*}
    \boldsymbol{\tau} &:= \chi_h \boldsymbol{\tau}_\mu (P_\rho, \varepsilon(\boldsymbol{v}_k)) + \vartheta_h\zeta_\eta\boldsymbol{\tau}_\mu(P_\rho,G_j) \\
    &=: \boldsymbol{\tau}^i + \boldsymbol{\tau}^b \ .
\end{align*}
Both terms are Lipschitz continuous by Lemma \ref{lem:hubler}, pointwise admissible,
\begin{align*}
    \boldsymbol{\tau}_\mu (P_\rho,\cdot) \in \mathcal{K}(P_\rho) \subseteq \mathcal{K}(P) \ ,
\end{align*}
by Lemma \ref{lem:hubler} and Lemma \ref{lema:prop_K}. Scaling with a factor in $[0,1]$ preserves admissibility via
convexity and $0\in \mathcal{K}$. Additionally their supports are disjoint, so that the sum is pointwise
admissible and $\operatorname{\mathrm{div}} \boldsymbol{\tau} \in L^\infty\subset L^{3/2}$. For $\boldsymbol{w} \in C^1(\bar \Omega)$ with
$\boldsymbol{w}_{|\Gamma_D} =0$, by integration by parts we have
\begin{align}
    \int_\Omega \boldsymbol{\tau} : \varepsilon(\boldsymbol{w}) \:\mathrm{d} x + \int_\Omega \boldsymbol{w} \cdot \operatorname{\mathrm{div}} \boldsymbol{\tau}\:\mathrm{d} x
    = \int_{\partial\Omega} \boldsymbol{w} \cdot (\boldsymbol{\tau} \nu) \:\mathrm{d} \mathcal{H}^1 = 0 \label{eq:ip}
\end{align}
since $\boldsymbol{w}$ vanishes on $\Gamma_D$ and $\boldsymbol{\tau}$ vanishes on $\bar\Gamma_N$ ($\boldsymbol{\tau}^i$ is compactly
supported and $\boldsymbol{\tau}^b$ because of Step 1 for $h<\min(\eta/4,d_\eta)$), and the remaining boundary set has measure zero
with $\boldsymbol{\tau}\nu \in L^\infty$. Thus $\boldsymbol{\tau} \in \Sigma$.

Step 4:
    Since $\boldsymbol{\tau}$ is Lipschitz continuous, approximation by $C^1(\bar\Omega)$ functions gives
    the Green formula for every $\boldsymbol{v}\in W^{1,3}(\Omega)$ with arbitrary trace:
    \begin{align*}
        -\int_\Omega\boldsymbol{v}\cdot\operatorname{\mathrm{div}}\boldsymbol{\tau}\:\mathrm{d} x
        =\int_\Omega\boldsymbol{\tau}:\varepsilon(\boldsymbol{v})\:\mathrm{d} x
        -\int_{\partial\Omega}\boldsymbol{v}\cdot(\boldsymbol{\tau}\nu)\:\mathrm{d}\mathcal{H}^1 \ .
    \end{align*}
    With $\boldsymbol{v}\cdot(\boldsymbol{\tau}\nu)=\boldsymbol{\tau}:(\boldsymbol{v}\odot\nu)$, $\vartheta_h=1$ on $\bar\Gamma_D$
    and $\boldsymbol{\tau}=0$ on $\bar\Gamma_N$, we therefore obtain
    \begin{align*}
        \mathcal{D}(\boldsymbol{v})&\geq-\int_\Omega\boldsymbol{v}\cdot\operatorname{\mathrm{div}}\boldsymbol{\tau}\:\mathrm{d} x\\
        &=\int_\Omega\chi_h\boldsymbol{\tau}_\mu(P_\rho,\varepsilon(\boldsymbol{v}_k)):\varepsilon(\boldsymbol{v})\:\mathrm{d} x
        +R_h+\int_{\Gamma_D}\zeta_\eta\boldsymbol{\tau}_\mu(P_\rho,G_j):g\:\mathrm{d}\mathcal{H}^1,
    \end{align*}
    where $R_h:=\int_\Omega\boldsymbol{\tau}^b:\varepsilon(\boldsymbol{v})\:\mathrm{d} x$ is the volume contribution of the boundary field.
    Since $|\boldsymbol{\tau}^b|_F\leq c_\mathcal{K}\Vert P\Vert_{L^\infty}$ and $\boldsymbol{\tau}^b=0$ whenever
    $\operatorname{dist}(x,\bar\Gamma_D)\geq h$, it satisfies
    \begin{align*}
        |R_h|\leq r(h):=c_\mathcal{K}\Vert P\Vert_{L^\infty}
        \int_{\Omega\cap U_h(\bar\Gamma_D)}|\varepsilon(\boldsymbol{v})|_F\:\mathrm{d} x\rightarrow0
        \quad\text{as }h\downarrow0 \ .
    \end{align*}
    This follows by dominated convergence from $\varepsilon(\boldsymbol{v})\in L^1(\Omega)$ and holds
    uniformly in the other approximation parameters. In the lower bound we retain the error $-r(h)$.

Step 5:
    Lemma \ref{lem:hubler}, H\"older and $P_\rho \leq \Vert P\Vert_{L^\infty}$ lead to
    \begin{align*}
        &\left|\int_\Omega \chi_h[\boldsymbol{\tau}_\mu(P_\rho, \varepsilon(\boldsymbol{v}_k)) - \boldsymbol{\tau}_\mu(P_\rho,\varepsilon(\boldsymbol{v}))]:\varepsilon(\boldsymbol{v})\:\mathrm{d} x\right|\\
        &\qquad\leq \frac{C_e \Vert P\Vert_{L^\infty(\Omega)}}{\mu} \Vert \varepsilon(\boldsymbol{v}_k)-\varepsilon(\boldsymbol{v})\Vert_{L^{3/2}}
        \Vert \varepsilon(\boldsymbol{v})\Vert_{L^3} \rightarrow 0
    \end{align*}
    as $k\rightarrow \infty$ and
    \begin{align*}
        \int_\Omega \chi_h\boldsymbol{\tau}_\mu (P_\rho, \varepsilon(\boldsymbol{v})):\varepsilon(\boldsymbol{v})\:\mathrm{d} x \geq
        \int_\Omega \chi_h D(P_\rho, \varepsilon(\boldsymbol{v}))\:\mathrm{d} x - \frac{\mu}{8} \Vert P \Vert_{L^1} \ .
    \end{align*}
    We apply the linearity of $D(\cdot,\boldsymbol{\xi})$ in the strength from Lemma \ref{lem:prop_D}, \eqref{eq:Prho} and
    $\Delta (\boldsymbol{\xi}) - \xi_I \leq 2\Delta(\boldsymbol{\xi}) \leq 2 \sqrt{2}|\boldsymbol{\xi}|_F$ and conclude
    \begin{align*}
        \int_\Omega \chi_h|D(P,\varepsilon(\boldsymbol{v}))-D(P_\rho, \varepsilon(\boldsymbol{v}))|\:\mathrm{d} x
        \leq 2 \sqrt{2}\omega_{\widetilde P}(\rho) \Vert\varepsilon(\boldsymbol{v})\Vert_{L^{1}} \rightarrow 0
    \end{align*}
    as $\rho \rightarrow 0$.

Step 6:
    Pointwise a.e.\ on $\Gamma_D$ we have \eqref{eq:hubler}.  Lemma \ref{lem:hubler}
    and Lemma \ref{lema:prop_K}  lead to the $L^{\infty}$ bound
    $|\boldsymbol{\tau}_\mu (P_\rho, \cdot)|_F \leq c_\mathcal{K} \Vert P \Vert_{L^\infty} $. Applying these
    estimates together with the Lipschitz continuity from Lemma \ref{lem:prop_D} and the $P$-linearity
    we get the following chain of estimates
    \begin{align*}
        \boldsymbol{\tau}_\mu(P_\rho, G_j) :g &=  \boldsymbol{\tau}_\mu(P_\rho, G_j) : G_j + \boldsymbol{\tau}_\mu (P_\rho, G_j) :(g-G_j) \\
        &\geq D(P_\rho, G_j) - \frac{P_\rho\mu}{8} - c_\mathcal{K} \Vert P \Vert_{L^\infty} |g-G_j|_F \\
        &\geq D(P_\rho, g) - \frac{P_\rho\mu}{8} - 2c_\mathcal{K} \Vert P \Vert_{L^\infty} |g-G_j|_F \\
        &\geq D(P, g) - \frac{P_\rho\mu}{8} - 2c_\mathcal{K} \Vert P \Vert_{L^\infty} |g-G_j|_F - 2 \sqrt{2} \omega_{\widetilde P}(\rho)|g|_F \ .
    \end{align*}
    Multiply with $\zeta_\eta \in [0,1]$ (keep it in the main term, estimate it by $1$ in the error terms), integrate
    and estimate:
    \begin{align}
        \begin{split}
        \int_{\Gamma_D} \zeta_\eta \boldsymbol{\tau}_\mu (P_\rho, G_j) : g \:\mathrm{d} \mathcal{H}^{1}
        &\geq \int_{\Gamma_D} \zeta_\eta D(P, g) \:\mathrm{d} \mathcal{H}^{1}
        -\frac{\mu}{8} \Vert P \Vert_{L^{1}(\Gamma_D)}\\
        &\quad - C \Vert g-G_j\Vert_{L^{3}(\Gamma_D)} - C' \omega_{\widetilde P}(\rho) \ .
        \end{split}
    \end{align}

Step 7:

All limits are taken with the left-hand side $\mathcal{D}(\boldsymbol{v})$ fixed. First we take the limits $k\rightarrow \infty$,
$j\rightarrow \infty$ and $\rho\rightarrow 0$. Combining Steps 4--6 yields
\begin{align*}
    \mathcal{D}(\boldsymbol{v})\geq{}&\int_\Omega\chi_hD(P,\varepsilon(\boldsymbol{v}))\:\mathrm{d} x
    +\int_{\Gamma_D}\zeta_\eta D(P,g)\:\mathrm{d}\mathcal{H}^1\\
    &-\frac{\mu}{8}\left(\Vert P\Vert_{L^1(\Omega)}+\Vert P\Vert_{L^1(\Gamma_D)}\right)-r(h) \ .
\end{align*}
Then let $h\rightarrow0$ with fixed $\eta$. Since $0\leq\chi_h\leq1$ and $\chi_h\rightarrow1$ pointwise in $\Omega$,
dominated convergence applies to the volume term, while $r(h)\rightarrow0$ by Step 4.
The restriction $h<\min(\eta/4,d_\eta)$ is preserved for all sufficiently small $h$.
Then $\eta \rightarrow 0$: since $\Gamma_D \cap I = \emptyset$ (Step 1), we have $\zeta_\eta \rightarrow 1$ pointwise
on $\Gamma_D$, with the majorant $c_\mathcal{K} \Vert P \Vert_{L^{\infty}}|g|\in L^{1}(\Gamma_D)$. Finally $\mu \rightarrow 0$,
and both remaining error terms $\frac{\mu}{8}(\Vert P\Vert_{L^{1}(\Omega)}+\Vert P\Vert_{L^{1}(\Gamma_D)})$ vanish.

<!-- lemma-5: body_md -->
\label{lem:stress_reconstruction}
    Under Assumption \ref{ass}, let $\boldsymbol{u}\in V$ with $\mathcal{D}(\boldsymbol{u})<\infty$ and
    $\boldsymbol{\xi}\in V^*$. Then $\boldsymbol{\xi}\in\partial\mathcal{D}(\boldsymbol{u})$ if and only if there is
    a $\boldsymbol{\sigma}\in\Sigma$ such that
    \begin{align}
        \boldsymbol{\xi}=-\operatorname{\mathrm{div}}\boldsymbol{\sigma},\qquad
        \mathcal{D}(\boldsymbol{u})=-\int_\Omega\boldsymbol{u}\cdot\operatorname{\mathrm{div}}\boldsymbol{\sigma}\:\mathrm{d} x \ .
        \label{eq:stress_saturation}
    \end{align}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Set $C:=\{-\operatorname{\mathrm{div}}\boldsymbol{\tau}:\boldsymbol{\tau}\in\Sigma\}\subset V^*$. This set is
    nonempty and convex. It is also norm closed. Indeed, if
    $-\operatorname{\mathrm{div}}\boldsymbol{\tau}_j\rightarrow\boldsymbol{\xi}$ in $V^*$, then
    $|\boldsymbol{\tau}_j|_F\leq c_\mathcal{K} P$ gives a subsequence converging weakly-$\ast$
    in $L^\infty$ to a symmetric field $\boldsymbol{\sigma}$ by weak-$\ast$ compactness
    \cite[Chapter 3]{Br11}. The pointwise constraint
    $\boldsymbol{\sigma}(x)\in\mathcal{K}(P(x))$ is preserved because it is closed and convex.
    Distributional testing gives $-\operatorname{\mathrm{div}}\boldsymbol{\sigma}=\boldsymbol{\xi}$. Passing to the limit
    in the Green identity against $C^1(\bar\Omega)$ fields with zero
    Dirichlet trace preserves the weak Neumann condition. Thus
    $\boldsymbol{\sigma}\in\Sigma$ and $\boldsymbol{\xi}\in C$.

    By definition, $\mathcal{D}$ is the support function of $C$. Since $V$ is
    reflexive and $C$ is closed and convex, its Fenchel conjugate is the
    indicator function of $C$. By the conjugacy and subdifferential relations
    in \cite[Chapter I]{ET99}, Fenchel equality characterises
    $\boldsymbol{\xi}\in\partial\mathcal{D}(\boldsymbol{u})$ by $\boldsymbol{\xi}\in C$ and
    $\langle\boldsymbol{\xi},\boldsymbol{u}\rangle=\mathcal{D}(\boldsymbol{u})$, which is precisely
    \eqref{eq:stress_saturation}.
<!-- end proof-5 -->
<!-- remark-1: body_md -->
Equation \eqref{eq:stress_saturation} is the global saturation condition
    for velocities in $L^3$. If additionally
    $\boldsymbol{u}\in W^{1,3}(\Omega)^2$ has zero Dirichlet trace, the Green
    identity and \eqref{eq:int_darstellung} imply
    $\boldsymbol{\sigma}:\varepsilon(\boldsymbol{u})=D(P,\varepsilon(\boldsymbol{u}))$ a.e.\ Hence the
    pointwise normal flow rule holds in that regularity class. For general
    $L^3$ velocities, \eqref{eq:stress_saturation} is the relaxed formulation.
    The stress need not be unique: for full Dirichlet boundary, constant
    $P>0$ and zero velocity, both $0$ and $-P\boldsymbol{\mathrm{I}}/2$ are admissible,
    divergence-free stresses satisfying the saturation condition.

<!-- theorem-1: body_md -->
\label{th:measure_structure}
    Let $P\in C^{0,1}(\bar\Omega)$, $P\geq0$, and let $\boldsymbol{v}\in V$ with
    $\mathcal{D}(\boldsymbol{v})<\infty$. Define the distributions
    \begin{align*}
        S&:=\operatorname{\mathrm{div}}(P\boldsymbol{v})-\boldsymbol{v}\cdot\nabla P,\\
        T_M&:=\operatorname{\mathrm{div}}(PM\boldsymbol{v})-(M\boldsymbol{v})\cdot\nabla P
        \quad\text{for constant }M\in\mathbb S \ .
    \end{align*}
    Then the following statements hold.
    \begin{enumerate}
        \item[(a)] $S$ is a locally finite signed Radon measure in $\Omega$,
        and its negative part satisfies the sharp global bound
        \begin{align}
            S^-(\Omega)\leq\mathcal{D}(\boldsymbol{v}) \ .\label{eq:S-}
        \end{align}
        \item[(b)] For every deviatoric $M$, $T_M$ is a locally finite signed
        Radon measure. Set $c_M:=|M|_F/\sqrt{2}$, the stress II-invariant of
        $M$. If $\omega\Subset\Omega$ is open and
        $\chi\in C_c^\infty(\Omega)$ satisfies $0\leq\chi\leq1$ and
        $\chi=1$ on a neighbourhood of $\bar\omega$, then
        \begin{align}
            |T_M|(\omega)
            &\leq e c_M\bigl(2\mathcal{D}(\boldsymbol{v})+\langle S,\chi\rangle\bigr)
            \notag\\
            &\leq e c_M\bigl(2\mathcal{D}(\boldsymbol{v})+S^+(\operatorname{supp}\chi)\bigr) \ .
            \label{eq:TM}
        \end{align}
        For $M=0$, both sides are zero.
        \item[(c)] The symmetric tensor-valued distribution
        \begin{align}
            A:=\varepsilon(P\boldsymbol{v})-\boldsymbol{v}\odot\nabla P\,\mathcal{L}^2
            \label{eq:weighted_deformation}
        \end{align}
        is a locally finite Radon measure. We use $P\varepsilon(\boldsymbol{v})$ as
        shorthand for this distribution. Moreover,
        \begin{align*}
            P\boldsymbol{v}\in BD_{\mathrm{loc}}(\Omega),\qquad
            \boldsymbol{v}\in BD_{\mathrm{loc}}(\{P>0\}),
        \end{align*}
        and hence $\boldsymbol{v}\in BD_{\mathrm{loc}}(\{P>\gamma\})$ for every
        $\gamma>0$. Here $BD$ denotes the space of integrable vector fields
        whose distributional symmetric gradient is a finite Radon measure.
    \end{enumerate}

<!-- proof-6: body_md -->
The distributions are well defined because $P\boldsymbol{v}$ and
    $\boldsymbol{v}\cdot\nabla P$ belong to $L^3$. For scalar
    $\mu\in C_c^\infty(\Omega)$,
    \begin{align}
        \langle S,\mu\rangle
        =-\int_\Omega\boldsymbol{v}\cdot\nabla(P\mu)\:\mathrm{d} x \ .
        \label{eq:ip_measure}
    \end{align}

    Step 1: If $0\leq\mu\leq1$, the compactly supported Lipschitz field
    $\boldsymbol{\tau}=-P\mu\boldsymbol{\mathrm{I}}$ lies in $\Sigma$. Indeed, for $P>0$ its normalised
    yield expression is $(1-2\mu)^2\leq1$, and for $P=0$ it is zero.
    Consequently,
    \begin{align}
        -\langle S,\mu\rangle
        =-\int_\Omega\boldsymbol{v}\cdot\operatorname{\mathrm{div}}\boldsymbol{\tau}\:\mathrm{d} x\leq\mathcal{D}(\boldsymbol{v}) \ .
        \label{eq:S}
    \end{align}
    Fix a compact $K\Subset\Omega$ and choose
    $\chi\in C_c^\infty(\Omega)$ with $0\leq\chi\leq1$ and $\chi=1$
    near $K$. For $\varphi\in C_c^\infty(\Omega)$ with
    $\operatorname{supp}\varphi\subset K$ and $\Vert\varphi\Vert_{L^\infty}\leq1$, both
    $(\chi+\varphi)/2$ and $(\chi-\varphi)/2$ are admissible in
    \eqref{eq:S}. Therefore
    \begin{align*}
        |\langle S,\varphi\rangle|
        \leq2\mathcal{D}(\boldsymbol{v})+\langle S,\chi\rangle
        \leq2\mathcal{D}(\boldsymbol{v})+|\langle S,\chi\rangle| \ .
    \end{align*}
    Thus $S$ is a distribution of order zero on every compact subset and
    hence a locally finite signed Radon measure. The characterisation
    \begin{align*}
        S^-(\Omega)=\sup\{-\langle S,\mu\rangle:
        \mu\in C_c^\infty(\Omega),\ 0\leq\mu\leq1\}
    \end{align*}
    proves \eqref{eq:S-}. To see sharpness, choose nonzero
    $0\leq P\in C_c^\infty(\Omega)$ and $\boldsymbol{v}(x)=-x$. Then
    $\varepsilon(\boldsymbol{v})=-\boldsymbol{\mathrm{I}}$, $S^-=(P\operatorname{\mathrm{div}}\boldsymbol{v})^-\mathcal{L}^2
    =2P\mathcal{L}^2$, and \eqref{eq:int_darstellung}, whose boundary term
    vanishes here, gives $\mathcal{D}(\boldsymbol{v})=2\int_\Omega P\:\mathrm{d} x$.

    Step 2: Let $M\ne0$ be deviatoric and put $\widehat M=M/c_M$. For
    $\varphi\in C_c^\infty(\omega)$ with $|\varphi|\leq1$, define
    \begin{align*}
        \boldsymbol{\tau}_\pm=-\frac{P\chi}{2}\boldsymbol{\mathrm{I}}
                    \ \pm\frac{P\varphi}{2e}\widehat M \ .
    \end{align*}
    We have $|\varphi|\leq\chi$, so for $P>0$
    \begin{align*}
        F(\boldsymbol{\tau}_\pm,P)+1=(1-\chi)^2+\varphi^2
        \leq(1-\chi)^2+\chi^2\leq1 \ .
    \end{align*}
    For $P=0$, both stresses are zero. They are Lipschitz and compactly
    supported, hence belong to $\Sigma$. Testing the definition of $\mathcal{D}$
    with the two signs yields
    \begin{align*}
        -\frac{1}{2}\langle S,\chi\rangle
        \ \pm\frac{1}{2e c_M}\langle T_M,\varphi\rangle
        \leq\mathcal{D}(\boldsymbol{v}) \ .
    \end{align*}
    Therefore
    \begin{align*}
        |\langle T_M,\varphi\rangle|
        \leq e c_M\bigl(2\mathcal{D}(\boldsymbol{v})+\langle S,\chi\rangle\bigr) \ .
    \end{align*}
    This uniform estimate proves the local measure structure of $T_M$ and,
    on taking the supremum over $\varphi$, the first inequality in
    \eqref{eq:TM}. The second follows from $0\leq\chi\leq1$.
    The case $M=0$ is immediate.

    Step 3: By the distributional definitions, $\operatorname{tr} A=S$ and $M:A=T_M$
    for every constant deviatoric $M$. Two linearly independent deviatoric
    matrices together with $\boldsymbol{\mathrm{I}}$ determine all components of $A$, so
    $A$ is a locally finite matrix-valued measure. As
    $\boldsymbol{v}\odot\nabla P\in L^3$, \eqref{eq:weighted_deformation} gives
    $P\boldsymbol{v}\in BD_{\mathrm{loc}}(\Omega)$. On every compact subset of
    $\{P>0\}$, the factor $P^{-1}$ is bounded and Lipschitz. The product
    rule for a Lipschitz function and a $BD$ field gives
    \begin{align*}
        \varepsilon(\boldsymbol{v})
        =\varepsilon\bigl(P^{-1}(P\boldsymbol{v})\bigr)=P^{-1}A,
    \end{align*}
    since the terms involving $\nabla P$ cancel. Thus
    $\boldsymbol{v}\in BD_{\mathrm{loc}}(\{P>0\})$. See also the $BD$ framework
    in \cite{RG80,KT83}.

<!-- lemma-6: body_md -->
\label{lem:drag}
    For any $a,b\in\mathbb R^{n}$ we have
    \begin{align}
        (|a|a-|b|b) \cdot (a-b) \geq \frac{1}{2} (|a|+|b|)|a-b|^{2} \geq \frac 1 2 |a-b|^{3} \ . \label{eq:ocean_drag}
    \end{align}
    Both inequalities are sharp.
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
Write $a\cdot b = |a||b|\cos(\theta)$, then we have
 \begin{align*}
      (|a|a-|b|b) \cdot (a-b) &= |a|^{3} + |b|^{3} -(|a|+|b|)|a||b|\cos(\theta) \\
      &= (|a| + |b|) (|a| - |b|)^{2} +(|a|+|b|)|a||b|(1-\cos(\theta)) \ .
 \end{align*}
 Additionally $|a-b|^2 = (|a|-|b|)^2+2|a||b|(1-\cos\theta)$, so
 \begin{align*}
    &(|a|+|b|)(|a|-|b|)^2 + (|a|+|b|)|a||b|(1-\cos\theta)\\
    &\qquad\geq \frac 1 2
    (|a|+|b|)\left((|a|-|b|)^2+2|a||b|(1-\cos\theta)\right)\\
    &\qquad= \frac 1 2 (|a|+|b|)|a-b|^{2} \ .
 \end{align*}
 The second inequality in \eqref{eq:ocean_drag} follows by the triangle inequality. The first inequality is sharp for $|a|=|b|$ and the second one
 for $b=-a$.

\label{th:td}
    Under Assumption \ref{ass}, the variational inequality \eqref{eq:vi_td} has a unique solution $\boldsymbol{u} \in V$. The following statements hold:
    \begin{enumerate}
        \item Let $\boldsymbol{u}_1, \boldsymbol{u}_2$ be the solutions corresponding to $\boldsymbol{f}_1,\boldsymbol{f}_2\in V^*$. Then
        \begin{align*}
            \Vert \boldsymbol{u}_1 -\boldsymbol{u}_2\Vert_{L^3}^2 \leq \frac{2}{c_o} \Vert \boldsymbol{f}_1-\boldsymbol{f}_2\Vert_{L^{3/2}} \ .
        \end{align*}
        Thus the solution map $\boldsymbol{f} \mapsto \boldsymbol{u}$ is H\"older continuous with exponent $\frac 1 2$.
    \item In the interior of $\{P=0\}$, \eqref{eq:sub_vi_td} reduces to the pointwise algebraic
    equation
    \begin{align*}
        \frac m k (\boldsymbol{u} - \boldsymbol{u}^{n-1}) + m f_c \boldsymbol{e}_3 \times \boldsymbol{u} + c_o |\boldsymbol{u}-\boldsymbol{u}_o|(\boldsymbol{u}-\boldsymbol{u}_o) = \boldsymbol{f}
    \end{align*}
    which is uniquely solvable.
    \item Let $\boldsymbol{u}_1,\boldsymbol{u}_2$ be the solutions corresponding to the ocean
    velocities $\boldsymbol{u}_o^{(1)},\boldsymbol{u}_o^{(2)}$, with all other data fixed. Set
    $\boldsymbol{\eta}:=\boldsymbol{u}_o^{(1)}-\boldsymbol{u}_o^{(2)}$ and $\boldsymbol{z}_i:=\boldsymbol{u}_i-\boldsymbol{u}_o^{(i)}$.
    Then
    \begin{align}
        \Vert\boldsymbol{z}_1-\boldsymbol{z}_2\Vert_{L^{3}}^3 &\leq
        4 (\Vert\boldsymbol{z}_1\Vert_{L^3}+\Vert\boldsymbol{z}_2\Vert_{L^3})\Vert\boldsymbol{\eta}\Vert_{L^3}^2\label{eq:first_uo}\\
        \frac{1}{k}\Vert\sqrt{m}(\boldsymbol{u}_1-\boldsymbol{u}_2)\Vert^2_{L^2}
        &\leq c_o(\Vert\boldsymbol{z}_1\Vert_{L^3}+\Vert\boldsymbol{z}_2\Vert_{L^3})\Vert\boldsymbol{\eta}\Vert^2_{L^3}\label{eq:second_uo}\\
        \Vert\boldsymbol{u}_1-\boldsymbol{u}_2\Vert_{L^3}&\leq \Vert\boldsymbol{\eta}\Vert_{L^3} +(4(\Vert\boldsymbol{z}_1\Vert_{L^3}+\Vert\boldsymbol{z}_2\Vert_{L^3}))^{1/3}\Vert\boldsymbol{\eta}\Vert_{L^3}^{2/3} \ . \label{eq:third_uo}
    \end{align}
    On bounded subsets of the ocean-velocity data space $L^3$, the solution
    map $\boldsymbol{u}_o\mapsto\boldsymbol{u}$ is H\"older continuous with exponent $2/3$ in
    $L^3$ and Lipschitz continuous with respect to the mass seminorm
    $\Vert\sqrt m\,\cdot\Vert_{L^2}$, with the other data fixed.

    \begin{align*}
        |\boldsymbol{u}_1-\boldsymbol{u}_2|\leq \sqrt{1+(kf_c)^2}|\boldsymbol{u}_o^{(1)}-\boldsymbol{u}_o^{(2)}| \ .
    \end{align*}
    \end{enumerate}

<!-- proof-8: body_md -->
We verify the assumptions of the existence theorem for elliptic variational inequalities
    of the second kind with monotone main term \cite[Theorem 54.A]{Z85}. We recall that $V$ is reflexive and separable, and according to Lemma \ref{lem:bdd} the functional $\mathcal{D}:V\rightarrow [0,\infty]$ is convex, lower semicontinuous and proper. We show that $\mathcal{A} : V \rightarrow V^{*}$
    is bounded, hemicontinuous, monotone and coercive.

    Step 1:
    Pointwise we can estimate
    \begin{align*}
        |\mathcal{A}(\boldsymbol{v})| \leq \frac{\Vert m\Vert_{L^{\infty}}}{k} |\boldsymbol{v}| + \frac{1}{k} \sqrt{\Vert m\Vert_{L^{\infty}}} \sqrt{m} |\boldsymbol{u}^{n-1}| + \Vert m f_c\Vert_{L^{\infty}} |\boldsymbol{v}| +c_o (|\boldsymbol{v}|+|\boldsymbol{u}_o|)^{2} \ .
    \end{align*}
    Since $\Omega$ is a bounded domain, we have the embeddings $L^{3} \hookrightarrow L^{3/2}$ and $L^{2} \hookrightarrow L^{3/2}$. Together with Assumption \ref{ass} (A4) we have
    $\mathcal{A}(\boldsymbol{v})\in L^{3/2}(\Omega)^2=V^*$ with $\Vert \mathcal{A}(\boldsymbol{v})\Vert_{V^{*}} \leq C (1+\Vert \boldsymbol{v}\Vert^{2}_V)$, so $\mathcal{A}$ is bounded.

    Step 2: We show for $\boldsymbol{u}, \boldsymbol{v}, \boldsymbol{w} \in V$ that the function $t \mapsto \langle \mathcal{A}(\boldsymbol{u} +t \boldsymbol{w}),\boldsymbol{v}\rangle$ is continuous on $[0,1]$. The integrands are continuous in $t$ and we have an integrable majorant independent of $t$:
    \begin{align*}
        C\big((|\boldsymbol{u}| + |\boldsymbol{w}| +|\boldsymbol{u}_o|+1)^{2} + \sqrt{m}|\boldsymbol{u}^{n-1}|/k \big) |\boldsymbol{v}| \ .
    \end{align*}
    By Lebesgue's theorem we conclude the hemicontinuity.

    Step 3:
    Let $\boldsymbol{v}, \boldsymbol{w}\in V$, $\delta := \boldsymbol{v}-\boldsymbol{w}$. The mass term gives
    \begin{align*}
        \int_\Omega \frac m k |\delta|^{2} \:\mathrm{d} x \geq 0 \ .
    \end{align*}
    The Coriolis term is pointwise skew-symmetric, $(mf_c\boldsymbol{e}_3 \times \delta) \cdot \delta =0$, and
    Lemma \ref{lem:drag} leads to the estimate
    \begin{align*}
        \langle \mathcal{A}(\boldsymbol{v})-\mathcal{A}(\boldsymbol{w}), \delta\rangle \geq \frac{c_o}{2} \int_\Omega
        (|\boldsymbol{v}-\boldsymbol{u}_o|+|\boldsymbol{w}-\boldsymbol{u}_o|)|\delta|^2 \:\mathrm{d} x \geq \frac{c_o}{2} \Vert\delta\Vert_{L^{3}}^3
      \end{align*}
    for the ocean drag. Therefore $\mathcal{A}$ is strictly monotone.

    Step 4: To prove the coercivity we estimate with Young's inequality for $\boldsymbol{v}\in V$ and
    $d := \boldsymbol{v} -\boldsymbol{u}_o$:
    \begin{align*}
        &\langle \mathcal{A} (\boldsymbol{v}) - \boldsymbol{f} , \boldsymbol{v} \rangle\\
        &\quad= \int_\Omega \left(\frac m k (\boldsymbol{v}-\boldsymbol{u}^{n-1})\cdot \boldsymbol{v} + c_o |d| d\cdot(d+\boldsymbol{u}_o)-\boldsymbol{f} \cdot \boldsymbol{v}\right) \:\mathrm{d} x\\
        &\quad\geq \int_\Omega \left(\frac{m}{2k} |\boldsymbol{v}|^{2} - \frac{m}{2k}|\boldsymbol{u}^{n-1}|^{2} + c_o |d|^3 - c_o |d|^2|\boldsymbol{u}_o|
        -|\boldsymbol{f}|(|d|+|\boldsymbol{u}_o|)\right) \:\mathrm{d} x\\
        &\quad\geq \frac{c_o}{3} \Vert d \Vert^3_{L^3} - \frac{1}{2k} \Vert \sqrt{m}\boldsymbol{u}^{n-1}\Vert^2_{L^2}
        - \frac{4c_o}{3} \Vert \boldsymbol{u}_o\Vert_{L^3}^3\\
        &\qquad - \frac{2}{3\sqrt{c_o}} \Vert \boldsymbol{f} \Vert_{L^{3/2}}^{3/2}
        -\Vert\boldsymbol{f}\Vert_{L^{3/2}} \Vert \boldsymbol{u}_o\Vert_{L^3} \ ,
    \end{align*}
    where we applied the Young inequalities $c_o|d|^2|\boldsymbol{u}_o| \leq \frac{c_o}{3}|d|^3 + \frac{4c_o}{3}|\boldsymbol{u}_o|^3$ and
    $|\boldsymbol{f}||d| \leq \frac{c_o}{3}|d|^3 + \frac{2}{3\sqrt{c_o}}|\boldsymbol{f}|^{3/2}$.
    The triangle inequality $\Vert \boldsymbol{v}\Vert^3_{L^3} \leq 4(\Vert d\Vert_{L^3}^3 + \Vert\boldsymbol{u}_o\Vert^3_{L^3})$
    leads to
    \begin{align*}
        \langle \mathcal{A}(\boldsymbol{v}) - \boldsymbol{f}, \boldsymbol{v} - 0\rangle \geq \frac{c_o}{12}\Vert\boldsymbol{v}\Vert^3_{L^3} - C_0
    \end{align*}
    with $C_0 = C_0 (\Vert \sqrt{m}\boldsymbol{u}^{n-1}\Vert_{L^2}, \Vert\boldsymbol{u}_o\Vert_{L^3}, \Vert\boldsymbol{f}\Vert_{L^{3/2}}, k, c_o)$.
    This shows the coercivity of $\mathcal{A}$.

    Step 5: The result in \cite{Z85} gives existence, and strict monotonicity gives uniqueness.

    Step 6: For the stability, we take $\boldsymbol{f}_1, \boldsymbol{f}_2$ and their solutions $\boldsymbol{u}_1, \boldsymbol{u}_2$.
    We test the variational inequality for $\boldsymbol{u}_1$ with $\boldsymbol{v}=\boldsymbol{u}_2$ and conversely, and add the two inequalities.
    This gives the stability estimate:
    \begin{align*}
        \frac{c_o}{2} \Vert \boldsymbol{u}_1-\boldsymbol{u}_2\Vert_{L^{3}}^3 \leq \langle\boldsymbol{f}_1-\boldsymbol{f}_2,\boldsymbol{u}_1-\boldsymbol{u}_2\rangle
        \leq \Vert \boldsymbol{f}_1-\boldsymbol{f}_2\Vert_{L^{3/2}}\Vert\boldsymbol{u}_1-\boldsymbol{u}_2\Vert_{L^3} \ .
    \end{align*}

    Step 7: For $\boldsymbol{\tau}\in\Sigma$ we have $|\boldsymbol{\tau}|\leq c_\mathcal{K} P=0$ a.e.\ in $\{P=0\}$ by Lemma \ref{lema:prop_K}.
    By Lemma \ref{lem:stress_reconstruction}, the subgradient
    $\boldsymbol{f}-\mathcal{A}(\boldsymbol{u})$ equals $-\operatorname{\mathrm{div}}\boldsymbol{\sigma}$ for some $\boldsymbol{\sigma}\in\Sigma$.
    On the open set $\operatorname{int}\{P=0\}$ this stress vanishes, so its
    divergence is zero there in distributions and hence a.e.\ This gives
    the stated pointwise algebraic equation. For fixed $x$ the map $\boldsymbol{v} \mapsto \frac m k \boldsymbol{v} + m f_c\boldsymbol{e}_3\times \boldsymbol{v}
    +c_o |\boldsymbol{v}-\boldsymbol{u}_o|(\boldsymbol{v}-\boldsymbol{u}_o)$ is by Lemma \ref{lem:drag} strictly monotone, continuous and coercive on
    $\mathbb R^2$, hence bijective.

    Proof of \textit{3.}: Set $\delta:=\boldsymbol{u}_1-\boldsymbol{u}_2$. Testing as in Step 6 eliminates
    $\mathcal{D}$ and $\boldsymbol{f}$. The Coriolis term vanishes and we have
    \begin{align*}
        \frac{1}{k}\Vert\sqrt{m}\delta\Vert^2_{L^2} +
         \langle c_o|\boldsymbol{z}_1|\boldsymbol{z}_1-c_o|\boldsymbol{z}_2|\boldsymbol{z}_2,\delta\rangle\leq 0 \ .
    \end{align*}
    We notice $\delta = (\boldsymbol{z}_1-\boldsymbol{z}_2)+\boldsymbol{\eta}$. With $\omega := |\boldsymbol{z}_1|+|\boldsymbol{z}_2|$ we apply the first estimate in \eqref{eq:ocean_drag}
    \begin{align*}
        \langle c_o|\boldsymbol{z}_1|\boldsymbol{z}_1-c_o|\boldsymbol{z}_2|\boldsymbol{z}_2,\boldsymbol{z}_1-\boldsymbol{z}_2\rangle \geq \frac{c_o}{2}\int_\Omega \omega |\boldsymbol{z}_1-\boldsymbol{z}_2|^2\:\mathrm{d} x
    \end{align*}
    and from the identity $|a|a-|b|b = |a|(a-b)+(|a|-|b|)b$ we conclude the pointwise estimate
    \begin{align*}
        \left|c_o|\boldsymbol{z}_1|\boldsymbol{z}_1-c_o|\boldsymbol{z}_2|\boldsymbol{z}_2\right|\leq c_o \omega |\boldsymbol{z}_1-\boldsymbol{z}_2| \ .
    \end{align*}
    Young's inequality, $c_o \omega |\boldsymbol{z}_1-\boldsymbol{z}_2||\boldsymbol{\eta}|\leq \frac{c_o}{4}\omega |\boldsymbol{z}_1-\boldsymbol{z}_2|^2+c_o\omega|\boldsymbol{\eta}|^2$ and H\"older lead to
    \begin{align*}
        \frac{1}{k}\Vert\sqrt{m}\delta\Vert^2_{L^2} + \frac{c_o}{4} \int_\Omega\omega |\boldsymbol{z}_1-\boldsymbol{z}_2|^2\:\mathrm{d} x
        &\leq c_o \int_\Omega \omega|\boldsymbol{\eta}|^2 \:\mathrm{d} x\\
        &\leq c_o(\Vert\boldsymbol{z}_1\Vert_{L^3}+\Vert\boldsymbol{z}_2\Vert_{L^3})\Vert\boldsymbol{\eta}\Vert^2_{L^3} \ .
    \end{align*}
    Since $|\boldsymbol{z}_1-\boldsymbol{z}_2|\leq \omega$ we have $\int_\Omega\omega|\boldsymbol{z}_1-\boldsymbol{z}_2|^2\:\mathrm{d} x \geq \Vert\boldsymbol{z}_1-\boldsymbol{z}_2\Vert^3_{L^3}$, so we conclude \eqref{eq:first_uo} and \eqref{eq:second_uo}. The estimate \eqref{eq:third_uo} follows from $\delta = \boldsymbol{z}_1-\boldsymbol{z}_2 + \boldsymbol{\eta}$.

    The coercivity estimate, tested at the solutions, bounds
    $\Vert\boldsymbol{z}_i\Vert_{L^3}$ uniformly when the ocean velocities range over
    a bounded subset of $L^3$ and the other data are fixed. This justifies
    the local continuity assertions in the theorem.

    In the interior of $\{P=0\}$ both solutions satisfy the pointwise equation in \textit{2.}. Subtracting the two equations gives
    \begin{align*}
        \frac{m}{k}\delta +mf_c\boldsymbol{e}_3\times \delta + c_o|\boldsymbol{z}_1|\boldsymbol{z}_1-c_o|\boldsymbol{z}_2|\boldsymbol{z}_2 =0 \ .
    \end{align*}
    Taking the scalar product with $\boldsymbol{z}_1-\boldsymbol{z}_2=\delta-\boldsymbol{\eta}$, using monotonicity $(c_o|\boldsymbol{z}_1|\boldsymbol{z}_1-c_o|\boldsymbol{z}_2|\boldsymbol{z}_2)\cdot(\boldsymbol{z}_1-\boldsymbol{z}_2)\geq 0$ and $(\boldsymbol{e}_3\times\delta)\cdot\delta=0$, we obtain
    \begin{align*}

\label{lem:DQT}

\textit{(a)} As in Lemma \ref{lem:bdd}, $\mathcal{D}_{Q_T}[q]$ is the pointwise supremum of the linear, weakly $L^{3}$-continuous

    $\boldsymbol{\tau}(\cdot,t) \in \Sigma[q_n]$. Definition \ref{def:D} gives

        \langle -\operatorname{\mathrm{div}}\boldsymbol{\tau}(\cdot, t), \boldsymbol{w}_n\rangle \leq \mathcal{D} [q_n](\boldsymbol{w}_n) \ .

        \langle - \operatorname{\mathrm{div}} \boldsymbol{\tau}_j, \boldsymbol{v}(t_j)\rangle \geq \mathcal{D}[(q(\cdot,t_j)-\delta)^+](\boldsymbol{v}(t_j)) - \eta  \ .

<!-- remark-2: body_md -->
For sufficiently regular functions, set $\boldsymbol{w}:=\boldsymbol{v}-\boldsymbol{u}$. Then
\begin{align*}
    \int_{0}^{T}\rho \langle m \partial_t \boldsymbol{u},\boldsymbol{w}\rangle \:\mathrm{d} t
    = \int_{0}^{T} \rho \langle m \partial_t \boldsymbol{v} , \boldsymbol{w}\rangle \:\mathrm{d} t +
    \frac 1 2 \Vert \sqrt{m(\cdot,0)} \boldsymbol{w}(0)\Vert_{L^{2}}^2 \\
    -\frac{\rho(T)}{2} \Vert \sqrt{m(\cdot, T)} \boldsymbol{w} (T)\Vert^{2}_{L^2}
    + \frac 1 2 \int_0^T \rho\int_\Omega (\partial_t m-\lambda m)|\boldsymbol{w}|^2 \:\mathrm{d} x \:\mathrm{d} t
\end{align*}
where the last two terms are non-positive, the first trivially and the second by \eqref{eq:wachstumschranke}.

shift $\mathcal{B}$ by monotonicity to the test function, the two non-positive terms can be discarded.
Thus every sufficiently regular solution of \eqref{eq:time_evolution} is a weak solution, for every
admissible rate $\lambda$. The definition depends on the chosen admissible rate.

variational inequality with $\mathcal{B}$ evaluated at the solution. For time-independent $m$
the canonical choice is $\lambda \equiv 0$, with $\beta \equiv 0$, $\hat P =P$, and the weak solution
becomes the classical unweighted form. To see that the terms are well defined, note that $t\mapsto \rho(t)\mathcal{D}_t(\boldsymbol{v}(t))$
is continuous (Lemma \ref{lem:prop_time_D}(c), $\beta$ is absolutely continuous), therefore we have a classical time integral.
On the solution side, $\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u})$ avoids requiring a separate
measurability and integral-representation result for
$\int_0^T\rho\mathcal{D}_t(\boldsymbol{u})\:\mathrm{d} t$.

<!-- proof-12: body_md -->
We use the time-discretisation strategy of Rothe's method
    \cite[Section 8.2]{R13}, with weighted estimates for the present coefficients.

    Step 1:

    Jensen's inequality leads to $\sum_n k \Vert \boldsymbol{f}^{n}\Vert_{L^{3/2}}^{3/2} \leq \Vert\boldsymbol{f}\Vert_{L^{3/2}}^{3/2}$
    and analogously for $\boldsymbol{u}_o$. Applying H\"older to the mixed term leads to $\sum_n k g^{n}\leq C(\boldsymbol{f}, \boldsymbol{u}_o)$.
    The discrete energy estimate \eqref{eq:DES} for the data $(m^{n},P^{n},\boldsymbol{f}^{n},\boldsymbol{u}^{n}_o)$ is multiplied
    by $\rho_n \leq 1$ and summed up. Due to \eqref{eq:ass_m} we have $\rho_n \Vert \sqrt{m^{n}}\boldsymbol{u}^{n-1}\Vert^{2}_{L^{2}}
    \leq \rho_{n-1}\Vert\sqrt{m^{n-1}}\boldsymbol{u}^{n-1}\Vert^{2}_{L^{2}}$ for $n\geq 2$ and $\rho_1 \Vert \sqrt{m^{1}}\boldsymbol{u}^{0}
    \Vert^{2}\leq \Vert \sqrt{m(\cdot,0)}\boldsymbol{u}_0\Vert^{2}$. The weighted sum telescopes, and for any
    $N'\leq N$ it follows
    \begin{align}

        \frac{\rho_{N'}}{2}\Vert\sqrt{m^{N'}}\boldsymbol{u}^{N'}\Vert^{2}_{L^{2}} &+ \sum_{n=1}^{N'}k \rho_n
        \left(\frac{c_o}{3}\Vert\boldsymbol{u}^{n}-\boldsymbol{u}^{n}_o\Vert^{3}_{L^{3}}+\mathcal{D}[P^{n}](\boldsymbol{u}^{n})\right)\\
        &\leq \frac{1}{2} \Vert \sqrt{m(\cdot,0)}\boldsymbol{u}_0\Vert^{2}_{L^{2}} + \sum_{n=1}^{N}k g^{n} \leq C \ .
                \end{split}
    \end{align}
    With $\rho_n \geq \exp(-\Vert\lambda\Vert_{L^{1}})$ and $\sum_n k\Vert\boldsymbol{u}^{n}_o\Vert_{L^{3}}^{3}
    \leq \Vert\boldsymbol{u}_o\Vert^{3}_{L^{3}}$, in particular $(\bar\boldsymbol{u}_k)$ is bounded in $L^{3}(Q_T)$,
    $(\sqrt{\bar m_k}\bar\boldsymbol{u}_k)$ bounded in $L^{\infty}(0,T;L^{2})$ and $\sum_n k \mathcal{D}[P^{n}](\boldsymbol{u}^{n})\leq C$.

    Step 2:

    We take a subsequence, with the same notation, $\bar\boldsymbol{u}_k \rightharpoonup \boldsymbol{u}$ in $L^{3}(Q_T)$. The averaged data converge strongly, $\bar \boldsymbol{f}_k \rightarrow \boldsymbol{f}$ in $L^{3/2}(Q_T)$, $\bar\boldsymbol{u}_{o,k}\rightarrow\boldsymbol{u}_o$ in
    $L^{3}(Q_T)$. From the uniform continuity of $m: [0,T] \rightarrow L^{3}(\Omega)$ it follows
    \begin{align*}
        \Vert\bar m_k-m\Vert_{L^{\infty}(0,T;L^{3})} \leq
    \sup_{|t-s|\leq k} \Vert m(t)-m(s)\Vert_{L^{3}} \rightarrow 0
    \end{align*}
    and with $|\sqrt{a}-\sqrt{b}|\leq \sqrt{|a-b|}$ therefore
    \begin{align*}
        \Vert \sqrt{\bar m_k}-\sqrt{m}\Vert_{L^{\infty}(0,T;L^{6})} \leq \Vert\bar m_k-m\Vert^{1/2}_{L^{\infty}(0,T;L^{3})}\rightarrow 0 \ .
    \end{align*}
    Assumption (B) shows that $\bar\rho_k \rightarrow \rho$ uniformly.
    From the boundedness of Step 1 and identification of the limit via test functions it follows that
     $\sqrt{\bar m_k}\bar \boldsymbol{u}_k \overset{\ast}{\rightharpoonup} \sqrt{m}\boldsymbol{u}$ in $L^{\infty}(0,T;L^{2})$,
     in particular $\sqrt{m}\boldsymbol{u} \in L^{\infty}(0,T;L^{2})$.

     Step 3: We use monotonicity to evaluate the nonlinear operator at the
     test function, as in the Minty formulation \cite[Section 2.3]{R13}.

    Let $\boldsymbol{v} \in C^1([0,T];X)$, $\boldsymbol{v}^{n}:= \boldsymbol{v}(t_n)$. Testing the discrete VI \eqref{eq:vi_td} in time step $n$ with $\boldsymbol{v}^{n}$,
    the monotonicity
    \begin{align*}
        &\langle \mathcal{B}^{n}(\boldsymbol{u}^{n})-\mathcal{B}^n(\boldsymbol{v}^n), \boldsymbol{v}^n-\boldsymbol{u}^n\rangle \leq 0 \\
        &\mathcal{B}^n(\boldsymbol{w}) := m^nf_c\boldsymbol{e}_3 \times \boldsymbol{w} + c_o |\boldsymbol{w}-\boldsymbol{u}^n_o|(\boldsymbol{w}-\boldsymbol{u}_o^n)
    \end{align*}
    and the fact that the Coriolis term can be shifted due to pointwise skew symmetry, independently of $m^n$, we have
    \begin{align}\label{eq:minty}
        \begin{split}
        \left\langle \frac{m^n}{k}(\boldsymbol{u}^n-\boldsymbol{u}^{n-1}),\boldsymbol{v}^n-\boldsymbol{u}^n \right\rangle
        + \langle\mathcal{B}^n(\boldsymbol{v}^n)-\boldsymbol{f}^n,\boldsymbol{v}^n-\boldsymbol{u}^n\rangle \\+ \mathcal{D}[P^n](\boldsymbol{v}^n)-\mathcal{D}[P^n](\boldsymbol{u}^n)\geq 0 \ .
        \end{split}
    \end{align}

    Step 4:

    Set $\boldsymbol{w}^n := \boldsymbol{v}^n -\boldsymbol{u}^n$ and calculate
    \begin{align*}
        \langle m^n(\boldsymbol{u}^n-\boldsymbol{u}^{n-1}),\boldsymbol{w}^n\rangle = \langle m^n (\boldsymbol{v}^n-\boldsymbol{v}^{n-1}),\boldsymbol{w}^n\rangle - \langle m^n(\boldsymbol{w}^n-\boldsymbol{w}^{n-1}),\boldsymbol{w}^n\rangle \ .
    \end{align*}
    The estimate $\langle m^n (a-b),a\rangle \geq \frac 1 2 (\Vert\sqrt{m^n}a\Vert^2-\Vert\sqrt{m^n}b\Vert^2)$, the rearrangement
    \begin{align*}
        \sum_{n=1}^{N} \rho_n (a_n-b_n) = \rho_Na_N-\rho_1b_1 + \sum_{n=2}^{N}(\rho_{n-1}a_{n-1}-\rho_nb_n)
    \end{align*}
    for $a_n := \Vert \sqrt{m^n}\boldsymbol{w}^n\Vert^2_{L^2}$ and $b_n:=\Vert\sqrt{m^n}\boldsymbol{w}^{n-1}\Vert^2_{L^2}$, \eqref{eq:ass_m} and
    \begin{align*}
        \rho_{n-1}a_{n-1}-\rho_nb_n &= \int_\Omega (\rho_{n-1}m^{n-1}-\rho_nm^n)
        |\boldsymbol{w}^{n-1}|^2\:\mathrm{d} x \geq 0,\\
        \rho_1b_1 &\leq \Vert\sqrt{m(\cdot,0)}\boldsymbol{w}^0\Vert^2_{L^2}\ ,
    \end{align*}
    lead, for any $k$, to
    \begin{align*}
        &\sum^N_{n=1} \rho_n \langle m^n (\boldsymbol{u}^n-\boldsymbol{u}^{n-1}),\boldsymbol{w}^n\rangle\\
        &\quad\leq \sum_{n=1}^{N}\rho_n\langle m^n(\boldsymbol{v}^n-\boldsymbol{v}^{n-1}),\boldsymbol{w}^n\rangle
        +\frac 1 2 \Vert\sqrt{m(\cdot,0)}(\boldsymbol{v}(0)-\boldsymbol{u}_0)\Vert^2_{L^2} \ .
    \end{align*}
    The two terms $\rho_N a_N \geq 0$ and the sum possess the right sign thanks
    to the growth rate condition \eqref{eq:wachstumschranke}. This summation-by-parts step uses the same mass monotonicity as the energy estimate. For the first term
    on the right-hand side we have the difference quotient $\:\mathrm{d}_k \boldsymbol{v}^n =
    (\boldsymbol{v}^n-\boldsymbol{v}^{n-1})/k = \frac{1}{k}\int_{I_n} \partial_t \boldsymbol{v} \:\mathrm{d} s$.
    The step function $t\mapsto \rho_n m^n \:\mathrm{d}_k \boldsymbol{v}^n$ for $t\in I_n$ converges
    strongly in $L^{3/2}(Q_T)$ to $\rho m\partial_t\boldsymbol{v}$, together with $\bar \boldsymbol{v}_k -\bar \boldsymbol{u}_k \rightharpoonup \boldsymbol{v}-\boldsymbol{u}$ in $L^3(Q_T)$. So
    \begin{align}
        \sum_{n=1}^{N}k\rho_n\langle m^n \:\mathrm{d}_k\boldsymbol{v}^n, \boldsymbol{v}^n-\boldsymbol{u}^n
        \rangle \rightarrow \int_{0}^{T}\rho \langle m\partial_t \boldsymbol{v},
        \boldsymbol{v}-\boldsymbol{u}\rangle \:\mathrm{d} t \ . \label{eq:limit_time}
    \end{align}

    Step 5:
    We multiply \eqref{eq:minty} by $k\rho_n$ and sum up, then $k\rightarrow 0$
    along a subsequence:
    \begin{itemize}
        \item The step function $t\mapsto \rho_n (\mathcal{B}^n(\boldsymbol{v}^n)-\boldsymbol{f}^n)$ converges
        strongly in $L^{3/2}(Q_T)$. For the Coriolis term, this follows by H\"older's inequality from $\bar m_k\rightarrow m$ in $L^\infty(0,T;L^3)$ and $\sup_{t\in[0,T]}\Vert\bar\boldsymbol{v}_k(t)-\boldsymbol{v}(t)\Vert_X\rightarrow0$. For the ocean drag, we use continuity of $\boldsymbol{w}\mapsto|\boldsymbol{w}|\boldsymbol{w}$ from $L^3(Q_T)$ to $L^{3/2}(Q_T)$, convergence of the averaged data as in Step 2 and uniform convergence of the weights. Together with $\bar \boldsymbol{v}_k-\bar \boldsymbol{u}_k \rightharpoonup \boldsymbol{v}-\boldsymbol{u}$ we conclude
        \begin{align*}
            \sum_n k\rho_n\langle\mathcal{B}^n(\boldsymbol{v}^n)-\boldsymbol{f}^n,\boldsymbol{v}^n-\boldsymbol{u}^n\rangle
            \rightarrow \int_{0}^{T}\rho \langle \mathcal{B}(t;\boldsymbol{v})-\boldsymbol{f},\boldsymbol{v}-\boldsymbol{u}\rangle \:\mathrm{d} t \ .
        \end{align*}
        \item For $t\in I_n$ we have
        \begin{align*}
            |\rho_n\mathcal{D}[P^n](\boldsymbol{v}^n)-\rho(t)\mathcal{D}_t(\boldsymbol{v}(t))|
            \leq \mu_k\mathcal{D}[P^n](\boldsymbol{v}^n) + |\mathcal{D}[P^n](\boldsymbol{v}^n)-\mathcal{D}_t(\boldsymbol{v}(t))|
        \end{align*}
        and both terms vanish uniformly in $n$. The first one since
        \begin{align*}
            \mathcal{D}[P^n](\boldsymbol{v}^n) = \int_\Omega P^nD(1,\varepsilon(\boldsymbol{v}^n)) \:\mathrm{d} x
            \leq c_\mathcal{K} \sup_n\Vert P^n\Vert_{L^\infty} \sup_t\Vert\varepsilon(\boldsymbol{v}(t))\Vert_{L^1} \leq C
        \end{align*}
        and the second one with Lemma \ref{lem:prop_time_D}(b),(c) and Assumption (B). Therefore
        \begin{align*}
            \sum_n k \rho_n \mathcal{D}[P^n](\boldsymbol{v}^n) \rightarrow \int_0^T\rho(t)\mathcal{D}_t(\boldsymbol{v}(t))\:\mathrm{d} t \ .
        \end{align*}
        \item For the scaling we have $\rho_n \mathcal{D}[P^n]=\mathcal{D}[\rho_nP^n]$ (Lemma \ref{lem:prop_time_D}(a)), and $\bar q_{k|I_n}:= \rho_n P^n = \max_{\bar I_n} \hat P$ satisfies $\bar q_k\geq \hat P$ in $Q_T$ (Assumption (B)).
        So with Lemma \ref{lem:DQT}(a),(b) and monotonicity we conclude
        \begin{align*}
            \sum_n k \mathcal{D}[\rho_nP^n](\boldsymbol{u}^n) &\geq \mathcal{D}_{Q_T}[\bar q_k](\bar \boldsymbol{u}_k)
            \geq \mathcal{D}_{Q_T}[\hat P](\bar\boldsymbol{u}_k),\\
            \liminf_k \mathcal{D}_{Q_T}[\hat P](\bar\boldsymbol{u}_k)&\geq \mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}) \ ,
        \end{align*}
        and with this we have
        \begin{align*}
            \limsup_k \Big(-\sum_n k\rho_n\mathcal{D}[P^n](\boldsymbol{u}^n)\Big) \leq -\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}) \ .
        \end{align*}
        \item The time term is the limit \eqref{eq:limit_time} from Step 4.
    \end{itemize}
    Thus $\boldsymbol{u}$ satisfies the inequality \eqref{eq:weak_solution} and
    \begin{align*}
        \mathcal{D}_{Q_T}(\boldsymbol{u}) \leq \exp(\Vert\lambda\Vert_{L^1})\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u})
        \leq \exp(\Vert\lambda\Vert_{L^1})\liminf_k\sum_n k \rho_n\mathcal{D}[P^n](\boldsymbol{u}^n)
        <\infty
    \end{align*}
    by Lemma \ref{lem:DQT}(a) and \eqref{eq:weighted_discrete_energie}.
    Thus $\boldsymbol{u}$ is a solution in the sense of Definition \ref{def:solution_evolution}.

    For the energy estimate set
    \begin{align*}
        C_k := \exp(\Vert\lambda\Vert_{L^1})
        \left(\frac{1}{2}\Vert\sqrt{m(\cdot,0)}\boldsymbol{u}_0\Vert^2_{L^2}+\sum_{n=1}^{N}kg^n\right) \ .
    \end{align*}
    Multiplying \eqref{eq:weighted_discrete_energie} by $\exp(\Vert\lambda\Vert_{L^1})$
    and using $\exp(\Vert\lambda\Vert_{L^1})\rho_n\geq1$, we obtain for every $N'\leq N$
    \begin{align*}
        \frac{1}{2} \Vert\sqrt{m^{N'}}\boldsymbol{u}^{N'}\Vert^2_{L^2}
        +\sum_{n=1}^{N'}k\left(\frac{c_o}{3}\Vert\boldsymbol{u}^n-\boldsymbol{u}^n_o\Vert^3_{L^3} +\mathcal{D}[P^n](\boldsymbol{u}^n)\right) \leq C_k \ .
    \end{align*}
    All terms on the left are non-negative. Taking the maximum of the kinetic energy over $N'$
    gives a bound by $C_k$, while choosing $N'=N$ bounds the total dissipation by $C_k$.
    Adding these two bounds yields
    \begin{align*}
        \frac{1}{2}\Vert\sqrt{\bar m_k}\bar\boldsymbol{u}_k\Vert_{L^\infty(0,T;L^2)}^2
        +\int_0^T\frac{c_o}{3}\Vert\bar\boldsymbol{u}_k-\bar\boldsymbol{u}_{o,k}\Vert_{L^3}^3\:\mathrm{d} t
        +\sum_{n=1}^{N}k\mathcal{D}[P^n](\boldsymbol{u}^n)\leq2C_k \ .
    \end{align*}
    By the weak-$\ast$ convergence established in Step 2 and lower semicontinuity of the norm,
    \begin{align*}
        \operatorname{ess sup}_{t\in(0,T)}\frac{1}{2}\Vert\sqrt{m(\cdot,t)}\boldsymbol{u}(t)\Vert_{L^2}^2
        \leq\liminf_k\frac{1}{2}\Vert\sqrt{\bar m_k}\bar\boldsymbol{u}_k\Vert_{L^\infty(0,T;L^2)}^2 \ .
    \end{align*}
    The ocean drag term is weakly lower semicontinuous (convex integral functional and $\bar\boldsymbol{u}_{o,k}\rightarrow \boldsymbol{u}_o$ strongly in $L^3(Q_T)$). For
    the dissipation term we have $P^n\geq P(\cdot,t)$ in $I_n$ (Assumption (B)), so
    as in Step 5, $\sum_n k\mathcal{D}[P^n](\boldsymbol{u}^n)\geq\mathcal{D}_{Q_T}(\bar\boldsymbol{u}_k)$ and thus
    \begin{align*}
        \liminf_k\sum_n k\mathcal{D}[P^n](\boldsymbol{u}^n) \geq \mathcal{D}_{Q_T}(\boldsymbol{u}) \ .
    \end{align*}
    Finally,
    \begin{align*}
        \limsup_k\sum_n kg^n \leq \int_{0}^{T}g \:\mathrm{d} t
    \end{align*}
    by Jensen's inequality. The mixed terms converge since $t \mapsto \Vert \bar\boldsymbol{f}_k(t)\Vert_{L^{3/2}}$ converges in $L^{3/2}(0,T)$ and $t \mapsto \Vert\bar\boldsymbol{u}_{o,k}(t)\Vert_{L^3}$ in $L^3(0,T)$, hence the product in $L^1(0,T)$.
    Applying these lower semicontinuity estimates to the sum and using the bound on $\limsup_k C_k$
    proves the asserted energy estimate with the factor $2$.

Define the convex, positively homogeneous functional on $L^3(Q_T)^2$
\begin{align*}
     F(\boldsymbol{v}):=
 \begin{cases}
  \displaystyle\int_{Q_T} D(\hat P,\varepsilon(\boldsymbol{v}))\:\mathrm{d} x\:\mathrm{d} t,&\text{if }\boldsymbol{v}\in C^1([0,T];X),\\
  +\infty,&\text{if }\boldsymbol{v}\notin C^1([0,T];X) \ .
 \end{cases}
\end{align*}
We show that the lower semicontinuous relaxation of $F$ is exactly
$\mathcal{D}_{Q_T}[\hat P]$.
Its convex conjugate is the indicator function of
\begin{align*}
 C:=\{\boldsymbol{\xi}\in L^{3/2}(Q_T)^2: \langle\boldsymbol{\xi},\boldsymbol{v}\rangle\leq F(\boldsymbol{v})
                         \text{ for all }\boldsymbol{v}\in C^1([0,T];X)\} \ .
\end{align*}
Indeed, positive homogeneity and $F(0)=0$ give $F^*(\boldsymbol{\xi})=0$
for $\boldsymbol{\xi}\in C$ and $F^*(\boldsymbol{\xi})=+\infty$ otherwise. We claim that
\begin{align}\label{eq:wu-polar}
 C=\{-\operatorname{\mathrm{div}}\boldsymbol{\tau}:\boldsymbol{\tau}\in\Sigma_{Q_T}[\hat P]\} \ .
\end{align}
The first inclusion follows from the Green identity and
$\boldsymbol{\tau}:\varepsilon(\boldsymbol{v})\leq D(\hat P,\varepsilon(\boldsymbol{v}))$.
For the reverse inclusion, choose $\boldsymbol{\xi}\in C$. We define a linear functional
on the subspace $\varepsilon(C^1([0,T];X))$ of $L^1(Q_T;\mathbb S)$ by
\begin{align*}
    \ell(\varepsilon(\boldsymbol{v})):=\langle\boldsymbol{\xi},\boldsymbol{v}\rangle \ .
\end{align*}
This is well defined. If $\varepsilon(\boldsymbol{v})=0$, applying the inequality for
$C$ to $\boldsymbol{v}$ and $-\boldsymbol{v}$ gives $\langle\boldsymbol{\xi},\boldsymbol{v}\rangle=0$. Moreover, $\ell$ is
dominated by the continuous sublinear functional
\begin{align*}
     H(E):=\int_{Q_T} D(\hat P,E)\:\mathrm{d} x\:\mathrm{d} t,
 \qquad 0\leq H(E)\leq c_\mathcal{K}\Vert\hat P\Vert_{L^\infty(Q_T)}\Vert E\Vert_{L^1} \ .
\end{align*}
The Hahn--Banach theorem in its dominated extension form
\cite[Theorem 1.1]{Br11} extends $\ell$ to all of
$L^1(Q_T;\mathbb S)$, still with $\ell(E)\leq H(E)$. Applying this bound to
$E$ and $-E$ shows that the extension is continuous. Hence there is
$\boldsymbol{\tau}\in L^\infty(Q_T;\mathbb S)$ representing it. Testing the estimate
with constant symmetric matrices multiplied by characteristic functions,
first for matrices with rational entries and then by continuity for all
matrices, gives
\begin{align*}
 \boldsymbol{\tau}(x,t):E\leq D(\hat P(x,t),E)\quad\text{for all }E\in\mathbb S
 \quad\text{a.e.},
\end{align*}
and therefore $\boldsymbol{\tau}(x,t)\in\mathcal{K}(\hat P(x,t))$ a.e.
A related use of Hahn--Banach to construct a space-time flux is given in
\cite[Section 5.4]{KS22} for the total variation flow.
The identity
\begin{align*}
 \int_{Q_T}\boldsymbol{\xi}\cdot \boldsymbol{v}\:\mathrm{d} x\:\mathrm{d} t
   =\int_{Q_T}\boldsymbol{\tau}:\varepsilon(\boldsymbol{v})\:\mathrm{d} x\:\mathrm{d} t\quad\text{for all }\boldsymbol{v}\in C^1([0,T];X)
\end{align*}
first gives $-\operatorname{\mathrm{div}}\boldsymbol{\tau}=\boldsymbol{\xi}$ in the distributional sense. Applying it to
$\boldsymbol{v}(x,t)=\eta(t)\phi(x)$, and then using a countable dense subset of $X$,
gives the spatial Green identity for all $\phi\in X$ for almost every
$t$. This is the weak zero-traction condition on $\Gamma_N$. Thus
$\boldsymbol{\tau}\in\Sigma_{Q_T}[\hat P]$, proving \eqref{eq:wu-polar}.

By \eqref{eq:wu-polar} and the definition of $\mathcal{D}_{Q_T}[\hat P]$,
\begin{align*}
 F^{**}(\boldsymbol{w})=\sup_{\boldsymbol{\xi}\in C}\langle\boldsymbol{\xi},\boldsymbol{w}\rangle=\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w}) \ .
\end{align*}
The Fenchel--Moreau theorem \cite[Chapter I]{ET99} identifies $F^{**}$ with the lower
semicontinuous relaxation of the proper convex functional $F$.
The norm closure of its epigraph therefore yields $\boldsymbol{w}_j\in C^1([0,T];X)$ with
$\boldsymbol{w}_j\rightarrow \boldsymbol{w}$ in $L^3(Q_T)^2$ and $F(\boldsymbol{w}_j)\rightarrow\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w})$. By
\eqref{eq:wu-F-smooth} we conclude $F(\boldsymbol{w}_j)=\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w}_j)$, as claimed.

\label{lem:wu-parabolic}
For every $\boldsymbol{w}\in L^3(Q_T)^2$ with $\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w})<\infty$, there is
a sequence $(\boldsymbol{z}_n)_n\subset C^1([0,T];X)$ such that

Choose $\boldsymbol{b}_n\in C_c^\infty(\Omega)^2\subset X$ with
\begin{align*}
 \Vert\boldsymbol{b}_n-\boldsymbol{u}_0\Vert_{a(0)}\rightarrow0 \ .
\end{align*}
Such a sequence is obtained by truncation in the weighted $L^2$ norm and
smooth approximation of the bounded truncations in $L^2(\Omega)^2$, since
$a(0)\in L^\infty(\Omega)$.
In particular, $\mathcal{D}(\boldsymbol{b}_n)<\infty$.
Choose $0<h_n\leq 1/n$ small enough that
\begin{align}\label{eq:wu-scale}
 h_n\bigl(\Vert\boldsymbol{b}_n\Vert_{L^3}^3+\mathcal{D}(\boldsymbol{b}_n)\bigr)\leq1/n \ .
\end{align}
Lemma \ref{lem:wu-relaxation} allows
us to choose $\boldsymbol{w}_n\in C^1([0,T];X)$ such that
\begin{align}\label{eq:wu-input-recovery}
    \begin{split}
    \Vert\boldsymbol{w}_n-\boldsymbol{w}\Vert_{L^3(Q_T)}&\leq h_n^2,\\
    |\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w}_n)-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w})|&\leq1/n\ .
    \end{split}
\end{align}
Following the exponential time mollification in \cite[Section 3.2]{E25}, set
\begin{align}\label{eq:wu-filter}
 \boldsymbol{z}_n(t)=\exp(-t/h_n)\boldsymbol{b}_n
       +\frac{1}{h_n}\int_0^t \exp(-(t-s)/h_n)\boldsymbol{w}_n(s)\:\mathrm{d} s \ .
\end{align}
This is an $X$-valued integral. We have $\boldsymbol{z}_n\in C^1([0,T];X)$, $\boldsymbol{z}_n(0)=\boldsymbol{b}_n$ and
\begin{align}\label{eq:wu-filter-derivative}
 \partial_t \boldsymbol{z}_n=(\boldsymbol{w}_n-\boldsymbol{z}_n)/h_n \ .
\end{align}
The convolution term in \eqref{eq:wu-filter} defines an operator on
$L^3(Q_T)^2$ with norm at most $1-\exp(-T/h_n)<1$ by Young's convolution
inequality \cite[Chapter 4]{Br11}.
After extension by zero to negative times, continuity of translations and
concentration of the kernel at zero show strong convergence to the identity
as $h_n\rightarrow0$. Direct integration and \eqref{eq:wu-scale} give
\begin{align*}
 \Vert \exp(-t/h_n)\boldsymbol{b}_n\Vert_{L^3(Q_T)}^3\leq\frac{h_n}{3}\Vert\boldsymbol{b}_n\Vert_{L^3}^3\leq\frac{1}{3n}\rightarrow0 \ .
\end{align*}
Together with \eqref{eq:wu-input-recovery}, this proves $\boldsymbol{z}_n\rightarrow \boldsymbol{w}$ in $L^3(Q_T)^2$.

We next control the dissipation by convexity and an exchange of the time
integrals, as in \cite[Lemma 3.3]{E25}. Convexity of $\mathcal{D}$ and the fact that
\begin{align*}
    \exp(-t/h_n)
       +\frac{1}{h_n}\int_0^t \exp(-(t-s)/h_n)\:\mathrm{d} s  = 1
\end{align*}
imply by Jensen's inequality that
\begin{align*}
 \mathcal{D}(\boldsymbol{z}_n(t))\leq \exp(-t/h_n)\mathcal{D}(\boldsymbol{b}_n)
       +\frac{1}{h_n}\int_0^t \exp(-(t-s)/h_n)\mathcal{D}(\boldsymbol{w}_n(s))\:\mathrm{d} s \ .
\end{align*}
Since $P$ is independent of time, the same spatial functional $\mathcal{D}$
occurs at $t$ and $s$. Multiplying by $\rho(t)$, using
$\rho(t)\leq\rho(s)$ for $s\leq t$, and applying Fubini gives
\begin{align}
 \mathcal{D}_{Q_T}[\hat P](\boldsymbol{z}_n)
 &\leq h_n\mathcal{D}(\boldsymbol{b}_n)\notag\\
 &\quad+\int_0^T\mathcal{D}(\boldsymbol{w}_n(s))
       \left(\int_s^T\frac{\rho(t)}{h_n}\exp(-(t-s)/h_n)\:\mathrm{d} t\right)\:\mathrm{d} s
       \notag\\
 &\leq h_n\mathcal{D}(\boldsymbol{b}_n)+\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w}_n) \ .\label{eq:wu-filter-energy}
\end{align}
Thus $\limsup_n\mathcal{D}_{Q_T}[\hat P](\boldsymbol{z}_n)\leq\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w})$. Lower semicontinuity and $\boldsymbol{z}_n\rightarrow \boldsymbol{w}$
give the reverse bound, proving \eqref{eq:wu-recovery-energy}.

Finally, \eqref{eq:wu-filter-derivative} yields the exact identity
\begin{align}\label{eq:wu-time-defect}
 \int_{Q_T} a\,\partial_t \boldsymbol{z}_n\cdot(\boldsymbol{z}_n-\boldsymbol{w})\:\mathrm{d} x\:\mathrm{d} t
 ={}&-h_n\int_{Q_T} a|\partial_t \boldsymbol{z}_n|^2\:\mathrm{d} x\:\mathrm{d} t\notag\\
   &+\int_{Q_T} a\,\partial_t \boldsymbol{z}_n\cdot(\boldsymbol{w}_n-\boldsymbol{w})\:\mathrm{d} x\:\mathrm{d} t\ .
\end{align}
Both $(\boldsymbol{w}_n)$ and $(\boldsymbol{z}_n)$ are bounded in $L^3(Q_T)^2$, so
$\Vert\partial_t \boldsymbol{z}_n\Vert_{L^3(Q_T)}\leq C/h_n$. Since $|Q_T|<\infty$, H\"older's
inequality and \eqref{eq:wu-input-recovery} give
\begin{align*}
 &\left|\int_{Q_T} a\,\partial_t \boldsymbol{z}_n\cdot(\boldsymbol{w}_n-\boldsymbol{w})\:\mathrm{d} x\:\mathrm{d} t\right|\\
 &\quad\leq\Vert a\Vert_{L^\infty(Q_T)} |Q_T|^{1/3}
       \Vert\partial_t \boldsymbol{z}_n\Vert_{L^3(Q_T)}\Vert\boldsymbol{w}_n-\boldsymbol{w}\Vert_{L^3(Q_T)}\\
 &\quad\leq C h_n\rightarrow0 \ .
\end{align*}
The first term on the right of \eqref{eq:wu-time-defect} is non-positive.
This proves \eqref{eq:wu-recovery-time}, including its initial-value claim.

<!-- lemma-12: body_md -->
\label{lem:wu-unshift}
Every weak solution in the sense of Definition \ref{def:solution_evolution}
satisfies, for all $\boldsymbol{v}\in C^1([0,T];X)$,
\begin{align}\label{eq:wu-unshifted}
    \begin{split}
    \mathcal{H}_{\boldsymbol{u}}(\boldsymbol{v})&+\mathcal{D}_{Q_T}[\hat P](\boldsymbol{v})-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u})\\
    &+\int_0^T\rho(t)
       \langle\mathcal{B}(t;\boldsymbol{u})-\boldsymbol{f},\boldsymbol{v}-\boldsymbol{u}\rangle\:\mathrm{d} t\geq0\ .
    \end{split}
\end{align}
<!-- end lemma-12 -->
<!-- proof-15: body_md -->
We combine the recovery sequence with a Minty perturbation argument
\cite[Section 2.3]{R13}.
For fixed $\boldsymbol{u}$, the functional $\boldsymbol{v}\mapsto\mathcal{H}_{\boldsymbol{u}}(\boldsymbol{v})$ is convex on $C^1([0,T];X)$.
Integration by parts involving only the smooth test function gives
\begin{align}
 \mathcal{H}_{\boldsymbol{u}}(\boldsymbol{v})
 ={}&\frac{1}{2}\Vert\boldsymbol{v}(T)\Vert_{a(T)}^2
     +\frac{1}{2}\int_{Q_T}(-\partial_ta)|\boldsymbol{v}|^2\:\mathrm{d} x\:\mathrm{d} t
     -\int_{Q_T} a\,\partial_t \boldsymbol{v}\cdot \boldsymbol{u}\:\mathrm{d} x\:\mathrm{d} t\notag\\
    &-\int_\Omega a(0)\boldsymbol{v}(0)\cdot \boldsymbol{u}_0\:\mathrm{d} x
     +\frac{1}{2}\Vert\boldsymbol{u}_0\Vert_{a(0)}^2 \ .\label{eq:wu-H-convex}
\end{align}
All terms are finite. The quadratic terms are non-negative by
\eqref{eq:wu-a-monotone}, and the remaining terms are affine in $\boldsymbol{v}$,
which proves convexity.
Fix $\boldsymbol{v}\in C^1([0,T];X)$ and $0<\theta<1$. Apply Lemma \ref{lem:wu-parabolic} to $\boldsymbol{w}=\boldsymbol{u}$
and use the admissible test functions
\begin{align*}
 \boldsymbol{v}_{n,\theta}=(1-\theta)\boldsymbol{z}_n+\theta \boldsymbol{v} \ .
\end{align*}
Convexity of $\mathcal{H}_{\boldsymbol{u}}$ and $\mathcal{D}_{Q_T}[\hat P]$, followed by \eqref{eq:weak_solution}, gives
\begin{align*}
 0\leq{}&(1-\theta)
       \bigl[\mathcal{H}_{\boldsymbol{u}}(\boldsymbol{z}_n)+\mathcal{D}_{Q_T}[\hat P](\boldsymbol{z}_n)-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u})\bigr]\\
 &+\theta\bigl[\mathcal{H}_{\boldsymbol{u}}(\boldsymbol{v})+\mathcal{D}_{Q_T}[\hat P](\boldsymbol{v})-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u})\bigr]\\
 &+\int_0^T\rho(t)
       \langle\mathcal{B}(t;\boldsymbol{v}_{n,\theta})-\boldsymbol{f},\boldsymbol{v}_{n,\theta}-\boldsymbol{u}\rangle\:\mathrm{d} t \ .
\end{align*}
Now $\boldsymbol{v}_{n,\theta}\rightarrow \boldsymbol{u}+\theta(\boldsymbol{v}-\boldsymbol{u})$ in $L^3(Q_T)^2$. The continuity
$\mathcal{B}:L^3(Q_T)^2\rightarrow L^{3/2}(Q_T)^2$ and the recovery properties imply, on taking the upper
limit as $n\rightarrow\infty$,
\begin{align*}
 0\leq{}&\theta\bigl[\mathcal{H}_{\boldsymbol{u}}(\boldsymbol{v})+\mathcal{D}_{Q_T}[\hat P](\boldsymbol{v})-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u})\bigr]\\
 &+\theta\int_0^T\rho(t)
       \langle\mathcal{B}(t;\boldsymbol{u}+\theta(\boldsymbol{v}-\boldsymbol{u}))-\boldsymbol{f},\boldsymbol{v}-\boldsymbol{u}\rangle\:\mathrm{d} t \ .
\end{align*}
Divide by $\theta$ and let $\theta\downarrow0$, again using continuity
of $\mathcal{B}$. This proves \eqref{eq:wu-unshifted}.

<!-- proof-16: body_md -->
Let $\boldsymbol{u}_1,\boldsymbol{u}_2$ be two weak solutions for the same data. Put
\begin{align*}
 \boldsymbol{w}=\frac{\boldsymbol{u}_1+\boldsymbol{u}_2}{2},\qquad\delta=\boldsymbol{u}_1-\boldsymbol{u}_2 \ .
\end{align*}
Convexity gives $\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w})\leq\frac{1}{2}(\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}_1)+\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}_2))<\infty$.
Apply Lemma \ref{lem:wu-parabolic} to $\boldsymbol{w}$, with the common initial datum
$\boldsymbol{u}_0$, and denote the resulting sequence by $\boldsymbol{z}_n$.

Use $\boldsymbol{v}=\boldsymbol{z}_n$ in \eqref{eq:wu-unshifted} for both $\boldsymbol{u}_1$ and $\boldsymbol{u}_2$ and add.
The time and initial terms add to $2\mathcal{H}_{\boldsymbol{w}}(\boldsymbol{z}_n)$, hence
\begin{align*}
 0\leq{}&2\mathcal{H}_{\boldsymbol{w}}(\boldsymbol{z}_n)+2\mathcal{D}_{Q_T}[\hat P](\boldsymbol{z}_n)\\
 &-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}_1)-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}_2)\\
 &+\int_0^T\rho(t)\Big[
       \langle\mathcal{B}(t;\boldsymbol{u}_1)-\boldsymbol{f},\boldsymbol{z}_n-\boldsymbol{u}_1\rangle
       +\langle\mathcal{B}(t;\boldsymbol{u}_2)-\boldsymbol{f},\boldsymbol{z}_n-\boldsymbol{u}_2\rangle\Big]\:\mathrm{d} t \ .
\end{align*}
By strong convergence $\boldsymbol{z}_n\rightarrow \boldsymbol{w}$ in $L^3(Q_T)^2$, the last line converges to
\begin{align*}
 -\frac{1}{2}\int_0^T\rho(t)
       \langle\mathcal{B}(t;\boldsymbol{u}_1)-\mathcal{B}(t;\boldsymbol{u}_2),\delta\rangle\:\mathrm{d} t \ .
\end{align*}
The recovery properties and convexity of $\mathcal{D}_{Q_T}[\hat P]$ therefore give
\begin{align*}
 0
 &\leq 2\mathcal{D}_{Q_T}[\hat P](\boldsymbol{w})-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}_1)-\mathcal{D}_{Q_T}[\hat P](\boldsymbol{u}_2)\\
 &\qquad-\frac{1}{2}\int_0^T\rho(t)
           \langle\mathcal{B}(t;\boldsymbol{u}_1)-\mathcal{B}(t;\boldsymbol{u}_2),\delta\rangle\:\mathrm{d} t\\
 &\leq-\frac{c_o}{4}\int_0^T\rho(t)\Vert\delta(t)\Vert_{L^3}^3\:\mathrm{d} t
 \leq0 \ .
\end{align*}
Since $c_o>0$ and $\rho(t)\geq \exp(-\Vert\lambda\Vert_{L^1(0,T)})>0$, it follows that
$\delta=0$ in $L^3(Q_T)^2$. This proves uniqueness on all of $Q_T$. Existence follows from part (a).

<!-- remark-3: body_md -->
Time independence of $P$ is used in the dissipation estimate

Korn or Poincar\'e inequality, and no time derivative of either solution
are needed. The approximation order is essential: first choose $\boldsymbol{b}_n$,

    $\langle\boldsymbol{\tau}\nu,\boldsymbol{w}\rangle =0$.

<!-- lemma-14: body_md -->
\label{lem:a3}
    Under Assumption \ref{ass}, for every $\boldsymbol{\tau}\in\Sigma$ and $\boldsymbol{v} \in W^{1,3}(\Omega)^2$
    \begin{align*}
        - \langle\boldsymbol{\tau}\nu,\boldsymbol{v}\rangle \leq \int_{\Gamma_D} D(P, -(\boldsymbol{v}\odot \nu)) \:\mathrm{d} \mathcal{H}^1

<!-- proof-18: body_md -->
Step 1:
    Let $\eta>0$ and $\zeta_\eta(x) := \psi (\operatorname{dist}(x,\bar\Gamma_D)/\eta)$ with $\psi \in \mathrm{Lip}(\mathbb R)$
    and $\psi = 1$ in $(-\infty,1/2]$, $\psi=0$ in $[1,\infty)$ and $0\leq \psi \leq 1$.
    Then $(1-\zeta_\eta)\boldsymbol{v} =0$ in $U_{\eta/2}(\bar\Gamma_D)\cap \Omega$. By Lemma \ref{lem:a2} we have
    $\langle \boldsymbol{\tau} \nu, \boldsymbol{v}\rangle = \langle \boldsymbol{\tau}\nu,\zeta_\eta \boldsymbol{v}\rangle$. We will prove that
    \begin{align}
        -\langle\boldsymbol{\tau}\nu,\boldsymbol{w}\rangle \leq \int_{\partial\Omega} D(P,-(\boldsymbol{w}\odot \nu)) \:\mathrm{d} \mathcal{H}^1 \label{eq:step1}
    \end{align}
    for $\boldsymbol{w}:=\zeta_\eta\boldsymbol{v} \in W^{1,3}(\Omega)^2$, and we take the limit as $\eta\rightarrow0$
    in Step 6.

    Step 2:
    Assumption \ref{ass} (A1) implies that there exist finitely many open cylinders $Z_1,\dots,Z_N \subset \mathbb R^2$
    which cover $\partial\Omega$, such that (after a rigid motion)
    \begin{align*}
        \Omega\cap Z_j = \{y=(y_1,y_2)\in Z_j : y_2>\gamma_j(y_1)\}, \quad \partial\Omega\cap Z_j
        =\{y:y_2=\gamma_j(y_1)\}\cap Z_j
    \end{align*}
    with $L$-Lipschitz continuous functions $\gamma_j$. The outer normal and the surface element
    in these coordinates are, for $\mathcal{L}^1$ a.e.\ $y_1$,
    \begin{align*}
        \nu(y_1) = \frac{(\gamma_j'(y_1),-1)}{\sqrt{1+\gamma'_j(y_1)^2}}, \quad \:\mathrm{d} \mathcal{H}^1 = \sqrt{1+\gamma'_j(y_1)^2} \:\mathrm{d} y_1 \ .
    \end{align*}
    Both are invariant under vertical translations. Choose $Z_0\Subset \Omega$ with
    $\bar\Omega \subset \cup_{j=0}^N Z_j$ and a subordinate smooth partition of unity $(\theta_j)_{j=0}^N$,
    $\theta_j\in C^\infty_c(Z_j)$ and $\sum_{j=0}^{N}\theta_j =1$ on $\bar\Omega$.

    Since \eqref{eq:normaltrace} is linear in $\boldsymbol{w}$ and
    $\theta_j\boldsymbol{w} \in W^{1,3}(\Omega)$, we notice that \eqref{eq:step1} can be shown term by term.
    The term for $j=0$ vanishes, since $\theta_0\boldsymbol{w}\in W^{1,3}$ has compact support
    in $\Omega$. As in the proof of Lemma \ref{lem:a2}, the mollified functions have zero trace on $\partial\Omega$ and the limit leads to
    $\langle \boldsymbol{\tau}\nu, \theta_0\boldsymbol{w}\rangle =0$.

    From now on let $j\geq 1$ be fixed, $\gamma:= \gamma_j$, and redefine $\boldsymbol{w}:= \theta_j\zeta_\eta\boldsymbol{v}$. Choose an open interval $B' \Subset \{y_1:(y_1,y_2)\in Z_j\ \text{for a}\ y_2\}$, a height $M$ and $t_0>0$ with
    \begin{align*}
        \operatorname{supp} \boldsymbol{w} \cap \bar\Omega \subset \{y : y_1\in B', \gamma(y_1)\leq y_2 < M-t_0\} \Subset Z_j \ .
    \end{align*}
    Step 3: For $t \in (0,t_0)$ we define
    \begin{align*}
        D_t := \{y : y_1\in B', \gamma(y_1)+t <y_2<M\}, \quad S_t =\{y : y_1\in B', y_2 = \gamma(y_1)+t\} \ .
    \end{align*}
    $D_t$ is a Lipschitz domain and $S_t$ its lower boundary. We notice that $\boldsymbol{w}$ vanishes for $y_1\in\partial B'$ and $y_2=M$. For $y\in D_t$ a simple calculation shows that $\operatorname{dist}(y,\partial\Omega)\geq c_L t$ for $c_L := (1+L^2)^{-1/2}$ and
    $L$ the Lipschitz constant of $\gamma$.

    Let $\boldsymbol{\tau}_\rho := \boldsymbol{\tau} * \varphi_\rho$ with $\varphi_\rho$ the standard mollifier and $\rho < c_L t/2$. On a neighbourhood of $\bar D_t$, $\boldsymbol{\tau}_\rho$ is
    smooth and we have
     \begin{align*}
        \operatorname{\mathrm{div}} \boldsymbol{\tau}_\rho = (\operatorname{\mathrm{div}}\boldsymbol{\tau}) * \varphi_\rho \ ,
     \end{align*}
     and with the outer normal $\nu(y_1)$ of $D_t$ on $S_t$ (identical to the one of $\Omega$, by
     translation invariance)
     \begin{align}
        \int_{D_t} \boldsymbol{\tau}_\rho : \varepsilon(\boldsymbol{w}) + \boldsymbol{w} \cdot \operatorname{\mathrm{div}}\boldsymbol{\tau}_\rho \:\mathrm{d} y =
        \int_{S_t} \boldsymbol{w} \cdot(\boldsymbol{\tau}_\rho \nu) \:\mathrm{d} \mathcal{H}^1 \ . \label{eq:trans}
     \end{align}
     Step 4:
     The function $T(y_1, s):= (y_1, \gamma(y_1)+s)$ is bi-Lipschitz from $B'\times (0,t_0)$ to
     \begin{align*}
        C := T(B'\times (0,t_0)) = \{y : y_1\in B', 0<y_2-\gamma(y_1)<t_0\} = \cup_{0<t<t_0} S_t
     \end{align*}
     and measure preserving ($\det \mathrm{D}T=1$). In the following we will apply
     a Fubini argument to select admissible slices.
     Let $N_1 \subset \Omega$ be the $\mathcal{L}^2$ null set of non-Lebesgue points of $\boldsymbol{\tau}$,
     $N_2$ the null set where $\boldsymbol{\tau}(x)\in \mathcal{K}(P(x))$ fails and $N_3$ the null set where $\boldsymbol{w} \circ T$ is not absolutely continuous on lines (ACL) in $s$-direction. With Fubini we have for $\mathcal{L}^1$ a.e.\ $t\in(0,t_0)$ that $\mathcal{H}^1$ a.e.\ points of $S_t$ are Lebesgue points of $\boldsymbol{\tau}$ and admissible.

    For such a $t$ we take the limit as $\rho \rightarrow 0$ in \eqref{eq:trans}.
    For the right-hand side, $\boldsymbol{\tau}_\rho \rightarrow \boldsymbol{\tau}$ pointwise $\mathcal{H}^1$ a.e.\ on $S_t$ with majorant $\Vert \boldsymbol{\tau}\Vert_{L^\infty}|\boldsymbol{w}_{|S_t}| \in L^1(S_t)$.
    With the pointwise bound of the support function (Lemma \ref{lem:prop_D}) we have
    \begin{align*}
        \boldsymbol{w} \cdot (\boldsymbol{\tau}\nu) = \boldsymbol{\tau} : (\boldsymbol{w}\odot \nu) \geq - D(P, -(\boldsymbol{w}\odot \nu))
    \end{align*}
    $\mathcal{H}^1$ a.e.\ on $S_t$. For a.e.\ $t\in(0,t_0)$ we therefore have
    \begin{align}
        -\int_{D_t} \boldsymbol{\tau} : \varepsilon(\boldsymbol{w}) + \boldsymbol{w}\cdot\operatorname{\mathrm{div}}\boldsymbol{\tau} \:\mathrm{d} y
        \leq \int_{S_t} D(P, -(\boldsymbol{w}\odot \nu)) \:\mathrm{d} \mathcal{H}^1 \ . \label{eq:slice}
    \end{align}
    Step 5:
    For the left-hand side, the monotone exhaustion $D_t \nearrow D_0$ with
    $L^1$ integrand leads to convergence to $-\langle \boldsymbol{\tau}\nu, \boldsymbol{w}\rangle$, since
    the support of $\boldsymbol{w}$ makes the $D_0$ integral equal to the $\Omega$ integral.
    For the right-hand side we use the translation invariance of $\nu$ and $\:\mathrm{d} \mathcal{H}^1$:
    \begin{align*}
        &\int_{S_t} D(P, -(\boldsymbol{w}\odot \nu)) \:\mathrm{d} \mathcal{H}^1\\
        &\quad=\int_{B'} D(P(T(y_1,t)), -(\boldsymbol{w} (T(y_1,t))\odot \nu(y_1))) \sqrt{1+\gamma'(y_1)^2} \:\mathrm{d} y_1 \ .
    \end{align*}
    By ACL we have for a.e.\ $y_1$
    \begin{align*}
        |\boldsymbol{w} (T(y_1,t)) - \boldsymbol{w} (T(y_1,0^+))| \leq \int_{0}^{t} |\partial_s(\boldsymbol{w}\circ T)(y_1,s)| \:\mathrm{d} s,
    \end{align*}
    and integration over $y_1$ and H\"older lead to
    \begin{align*}
        \Vert \boldsymbol{w} \circ T(\cdot, t) - \boldsymbol{w} \circ T(\cdot, 0^+)\Vert_{L^1(B')}
        \leq  C t^{2/3} \Vert \nabla(\boldsymbol{w} \circ T)\Vert_{L^3} \rightarrow 0 \ .
    \end{align*}
    The ACL boundary values $\boldsymbol{w} \circ T(\cdot, 0^{+})$ coincide $\mathcal{L}^{1}$ a.e.\ with the trace $\boldsymbol{w}_{|\partial\Omega}$.
    Also $|P(T(y_1,t))-P(T(y_1,0))| \leq \omega_P(t)$, where $\omega_P$ is the modulus of continuity of $P$.
    The Lipschitz continuity from Lemma \ref{lema:prop_K} and Lemma \ref{lem:prop_D} lead to
    \begin{align*}
        |D(P_1,\boldsymbol{\xi}_1) - D(P_2,\boldsymbol{\xi}_2)| \leq c_\mathcal{K} |P_1-P_2| |\boldsymbol{\xi}_1|_F + c_\mathcal{K} \Vert P\Vert_{L^\infty}
        |\boldsymbol{\xi}_1-\boldsymbol{\xi}_2|_F \ .
    \end{align*}
    This and $|a\odot b|_F \leq |a||b|$ lead to the convergence of the right-hand side of \eqref{eq:slice}
    to $\int_{\partial\Omega} D(P,-(\boldsymbol{w}\odot \nu)) \:\mathrm{d} \mathcal{H}^1$ along a sequence of admissible $t$. This proves \eqref{eq:step1}, after summation over $j$.

    Step 6:
    Returning to Step 1 and \eqref{eq:step1}, we have with $\boldsymbol{w} = \zeta_\eta\boldsymbol{v}$ and the positive homogeneity
    of degree 1 of $D(P,\cdot)$ (Lemma \ref{lem:prop_D}) and $0\leq\zeta_\eta\in\mathrm{Lip}(\bar\Omega)$
    \begin{align*}
        -\langle \boldsymbol{\tau}\nu, \boldsymbol{v}\rangle = - \langle \boldsymbol{\tau}\nu, \zeta_\eta \boldsymbol{v}\rangle
        \leq \int_{\partial\Omega} \zeta_\eta D(P, -(\boldsymbol{v}\odot \nu)) \:\mathrm{d} \mathcal{H}^1 \ .
    \end{align*}
    As $\eta\rightarrow 0$, $\zeta_\eta \rightarrow \mathbf 1_{\bar\Gamma_D}$ pointwise on $\partial\Omega$.
    The integrands are bounded by the $\mathcal{H}^{1}$-integrable function
    $c_\mathcal{K} \Vert P\Vert_{L^{\infty}}|\boldsymbol{v}_{|\partial\Omega}|$.
    Dominated convergence and $\mathcal{H}^{1}(\bar\Gamma_D\setminus\Gamma_D) =0$ conclude the proof.
