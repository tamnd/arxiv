\label{thm:entire}
If $f\in\mathcal B$ and $S(f)'\subset\mathcal F(f)$, then $f$ has no wandering Fatou components.
<!-- end theorem-2 -->
<!-- definition-1: body_md -->
\label{def:area}
An open set $G\subset\widehat{\mathbb C}$ is \emph{hyperbolic} if each component has $\mathbb D$ as its universal covering surface. For such a set, we use the following notation:
\begin{enumerate}
\item The \emph{Poincar\'e metric} $\rho_G(z)|dz|$ is the complete conformal metric of curvature $-4$, normalized by $\rho_{\mathbb D}(z)=(1-|z|^2)^{-1}$;
\item The \emph{hyperbolic distance} is $d_G(z,w)=\inf_\gamma\int_\gamma\rho_G|dz|$, where the infimum is taken over piecewise smooth paths in $G$ joining $z$ to $w$, with $\inf\varnothing=\infty$;
\item The \emph{normalized hyperbolic area measure} is given in local coordinates by $\,\mathrm d\alpha_G=(2/\pi)\rho_G^2\,\,\mathrm d x\,\mathrm d y$;
\item For a measurable set $A\subset G$, its area is $\alpha_G(A)=\int_A\,\mathrm d\alpha_G$, and $\alpha_G(G)$ is the \emph{total intrinsic area} of $G$.
\end{enumerate}
<!-- end definition-1 -->
<!-- head-1: body_md -->
\begin{enumerate}[ref=(\roman*)]
\item\label{fact:punctured-area} For a finite set $F\subset\widehat{\mathbb C}$ with $|F|\ge3$, Gauss--Bonnet and the Euler characteristic $2-|F|$ give $\alpha_{\widehat{\mathbb C}\setminus F}(\widehat{\mathbb C}\setminus F)=|F|-2$.
\item\label{fact:infinite-area} Every simply connected hyperbolic domain has infinite intrinsic area, since it is conformally equivalent to $\mathbb D$ and $\alpha_{\mathbb D}(\mathbb D)=4\int_0^1 r(1-r^2)^{-2}\,\mathrm d r=\infty$.
\item\label{fact:metric-properties} For a holomorphic map $\psi:G\to H$ between hyperbolic open sets, the Schwarz--Pick theorem \cite[Theorem~2.11]{Milnor} gives $\rho_H(\psi(z))|\psi'(z)|\le\rho_G(z)$, equivalently $\psi^*\,\mathrm d\alpha_H\le\,\mathrm d\alpha_G$, with equality if $\psi$ is a covering. In particular, an inclusion $U\subset V$ gives \emph{domain monotonicity}: $\rho_U\ge\rho_V$ on $U$, $d_U(z,w)\ge d_V(z,w)$ for $z,w\in U$, and $\alpha_U(A)\ge\alpha_V(A)$ for every measurable $A\subset U$.
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:recovery}
Let $F_1\subseteq F_2\subseteq\cdots\subset\widehat{\mathbb C}$ be finite sets with $|F_1|\ge3$. If $D$ is a component of $\widehat{\mathbb C}\setminus\overline{\bigcup_jF_j}$, then $\rho_{\widehat{\mathbb C}\setminus F_j}$ increases locally uniformly to $\rho_D$ on $D$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Write $G_j=\widehat{\mathbb C}\setminus F_j$ and fix $a\in D$. Domain monotonicity in Fact~\ref{fact:metric-properties} gives $\rho_{G_j}(a)\nearrow\ell$ with $0<\ell\le\rho_D(a)$. Choose universal covering maps $\pi_j:\mathbb D\to G_j$ with $\pi_j(0)=a$. Covering invariance gives $|\pi_j'(0)|=1/\rho_{G_j}(a)$, with derivatives and densities taken in a local coordinate at $a$.

The maps $\pi_j$ omit the three or more points of $F_1$, so Montel's theorem gives a subsequence converging locally uniformly in the spherical metric to a limit $\pi$. Since $\pi(0)=a$ and $|\pi'(0)|=1/\ell>0$, the limit is nonconstant. Every point of $\bigcup_jF_j$ is omitted by all sufficiently late maps $\pi_j$. Hurwitz's theorem \cite{Ahlfors}, applied in local coordinates, shows that $\pi$ omits this union. As $\pi$ is open, its image also avoids the closure of the union. Hence $\pi(\mathbb D)\subset D$, and Schwarz--Pick (Fact~\ref{fact:metric-properties}) gives
\[
 \rho_D(a)\le\frac{1}{|\pi'(0)|}=\ell\le\rho_D(a).
\]
Thus $\rho_{G_j}(a)\to\rho_D(a)$ for every $a\in D$. The densities are continuous and increase pointwise, so Dini's theorem gives uniform convergence on every compact subset of $D$.

<!-- lemma-2: body_md -->
\label{lem:punctures}
If $G\subset\widehat{\mathbb C}$ is hyperbolic and $E\subset G$ is finite, then
\begin{equation}\label{eq:puncture-cost}
 0\le\int_{G\setminus E}(\,\mathrm d\alpha_{G\setminus E}-\,\mathrm d\alpha_G)\le |E|.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Choose increasing finite sets $F_j\subset\widehat{\mathbb C}\setminus G$, containing a fixed triple, with union dense in $\widehat{\mathbb C}\setminus G$. Set $G_j=\widehat{\mathbb C}\setminus F_j$ and $H_j=G_j\setminus E$. Fact~\ref{fact:punctured-area} gives
\[
 \int_{H_j}(\,\mathrm d\alpha_{H_j}-\,\mathrm d\alpha_{G_j})=|E|.
\]
Lemma~\ref{lem:recovery} gives $\rho_{G_j}\to\rho_G$ and $\rho_{H_j}\to\rho_{G\setminus E}$ pointwise on $G\setminus E$. The area differences are nonnegative by Fact~\ref{fact:metric-properties}. Restricting the preceding integral to $G\setminus E$ and applying Fatou's lemma proves \eqref{eq:puncture-cost}.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
Suppose that $R$ has a wandering Fatou component. By Baker's reduction \cite[Lemma~5.34, p.~90]{McMullen}, its forward images are simply connected for all sufficiently large iterates. Since $\operatorname{CV}(R)$ is finite, we may discard finitely many initial images and denote the remaining components by $(V_n)_{n\ge0}$, so that they are pairwise distinct, simply connected, and disjoint from $\operatorname{CV}(R)$. Each $V_n$ is a component of $R^{-1}(V_{n+1})$: complete invariance of the Fatou set gives one inclusion, and connectedness gives the other. Properness makes $R:V_n\to V_{n+1}$ a covering, and simple connectivity of $V_{n+1}$ makes it a conformal isomorphism. Hence
\[
 W=\bigsqcup_{n\ge0}V_n,
 \qquad R:W\longrightarrow W\setminus V_0
\]
is a conformal isomorphism.

By density of repelling periodic points \cite[Theorem~4]{Bergweiler}, choose increasing finite unions $P_j\subset\mathcal J(R)$ of complete repelling cycles such that $|P_1|\ge3$ and $\overline{\bigcup_jP_j}=\mathcal J(R)$. In particular, $R(P_j)=P_j$. Put
\[
 X_j=\widehat{\mathbb C}\setminus P_j,\qquad
 Y_j=X_j\setminus\operatorname{CV}(R),
\]
\nopagebreak[4]
and define $\,\mathrm d\mu_j=\,\mathrm d\alpha_{X_j}$ on $X_j$ and $\,\mathrm d\nu_j=\,\mathrm d\alpha_{Y_j}-\,\mathrm d\alpha_{X_j}\ge0$ on $Y_j$. Fact~\ref{fact:punctured-area} and Riemann--Hurwitz give
\begin{equation}\label{eq:rational-charge}
 \nu_j(Y_j)=|\operatorname{CV}(R)\setminus P_j|
        \le 2\deg R-2.
\end{equation}

As $R(P_j)=P_j$, we have $R^{-1}(Y_j)\subset X_j$. The map $R:R^{-1}(Y_j)\to Y_j$ is an unramified covering. Fact~\ref{fact:metric-properties} gives
\begin{equation}\label{eq:rational-pullback}
 R^*(\,\mathrm d\mu_j+\,\mathrm d\nu_j)
 =\,\mathrm d\alpha_{R^{-1}(Y_j)}\ge\,\mathrm d\mu_j
 \quad\text{on }R^{-1}(Y_j).
\end{equation}
Since $W\subset\mathcal F(R)\setminus\operatorname{CV}(R)$, we have $R(W)=W\setminus V_0\subset W\subset Y_j$, so $W\subset R^{-1}(Y_j)$. Integrating \eqref{eq:rational-pullback} over $W$ and using the conformal isomorphism above gives
\[
 \mu_j(V_0)+\mu_j(W\setminus V_0)=\mu_j(W)
 \le\mu_j(W\setminus V_0)+\nu_j(W\setminus V_0).
\]
The common term is finite because $\mu_j(X_j)=|P_j|-2$ by Fact~\ref{fact:punctured-area}. We may therefore subtract it and obtain from \eqref{eq:rational-charge}
\begin{equation}\label{eq:rational-bound}
 \mu_j(V_0)\le\nu_j(W\setminus V_0)\le 2\deg R-2.
\end{equation}
But Lemma~\ref{lem:recovery}, monotone convergence and Fact~\ref{fact:infinite-area} give
\[
 \mu_j(V_0)\longrightarrow\alpha_{V_0}(V_0)=\infty,
\]
contrary to \eqref{eq:rational-bound}.

<!-- lemma-3: body_md -->
\label{lem:holes}
Let $K,N\subset\widehat{\mathbb C}$ be compact sets with $K\Subset\operatorname{Int}N$. There is a constant $C=C(K,N)<\infty$ such that, for every finite set $F\subset\widehat{\mathbb C}\setminus N$ with $|F|\ge3$, the open sets $G=\widehat{\mathbb C}\setminus F$ and $H=G\setminus K$ satisfy
\begin{equation}\label{eq:holes}
 \int_{G\setminus N}(\,\mathrm d\alpha_H-\,\mathrm d\alpha_G)\le C.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
If $K=\varnothing$ or $N=\widehat{\mathbb C}$, take $C=0$. Otherwise, fix a three-point set $Q\subset\widehat{\mathbb C}\setminus N$ and put $G_0=\widehat{\mathbb C}\setminus Q$.

First suppose that $Q\subset F$. Compact containment gives $\varepsilon=d_{G_0}(K,G_0\setminus\operatorname{Int}N)>0$. Since $G\subset G_0$, Fact~\ref{fact:metric-properties} implies $d_G(z,K)\ge\varepsilon$ for $z\in G\setminus N$. A universal cover $\phi:\mathbb D\to G$ with $\phi(0)=z$ therefore maps $\{|w|<r\}$ into $H$, where $r=\tanh\varepsilon$. Applying Fact~\ref{fact:metric-properties} to $w\mapsto\phi(rw)$ gives
\[
 r\rho_H(z)|\phi'(0)|\le1=\rho_G(z)|\phi'(0)|.
\]
Thus $u=\log(\rho_H/\rho_G)$ satisfies $0\le u\le\log(1/r)$ on $G\setminus N$. The curvature equation gives
\begin{equation}\label{eq:curvature}
 \Delta u=4(\rho_H^2-\rho_G^2)\ge0
 \quad\text{on }H.
\end{equation}
Here $\Delta=\partial_x^2+\partial_y^2$ in a local coordinate. Hence $u$ is subharmonic on $H$. The bound above holds near every point of $F$, since $F\cap N=\varnothing$. Defining its value at each such point by the upper limit gives a subharmonic extension to $\widehat{\mathbb C}\setminus K$.

Let $\sigma$ be the spherical metric and $L=\operatorname{Int}N\setminus K$. Choose $\chi\in C_c^\infty(\widehat{\mathbb C}\setminus K)$ with $0\le\chi\le1$, $\chi=1$ near $\widehat{\mathbb C}\setminus N$, and $\operatorname{supp}\Delta_\sigma\chi\Subset L$. On $L$, Fact~\ref{fact:metric-properties} and the inclusions $L\subset H$ and $G\subset G_0$ give $0\le u\le\log(\rho_L/\rho_{G_0})$. The subharmonic extension has a positive distributional Laplacian. Thus \eqref{eq:curvature} and integration by parts yield
\begin{align*}
 \int_{G\setminus N}(\,\mathrm d\alpha_H-\,\mathrm d\alpha_G)
 &\le\frac1{2\pi}\langle\Delta_\sigma u,\chi\rangle
 =\frac1{2\pi}\int_L u\Delta_\sigma\chi\,\,\mathrm d A_\sigma\\
 &\le\frac1{2\pi}
  \int_L\log\frac{\rho_L}{\rho_{G_0}}\,|\Delta_\sigma\chi|\,\,\mathrm d A_\sigma=:C_0.
\end{align*}
The integrand is supported in a compact subset of $L$, so $C_0$ is finite and independent of $F$.

For an arbitrary $F$, apply this bound to $F\cup Q$. Finite sets have zero area, so Fact~\ref{fact:metric-properties} and Lemma~\ref{lem:punctures} give
\begin{align*}
 \int_{G\setminus N}(\,\mathrm d\alpha_H-\,\mathrm d\alpha_G)
 &\le\int_{G\setminus(N\cup Q)}
       (\,\mathrm d\alpha_{H\setminus Q}-\,\mathrm d\alpha_{G\setminus Q})\\
 &\quad+\int_{G\setminus Q}
       (\,\mathrm d\alpha_{G\setminus Q}-\,\mathrm d\alpha_G)
 \le C_0+|Q\setminus F|\le C_0+3.
\end{align*}
Since $Q$ and $\chi$ were chosen using only $K,N$, the constant $C=C_0+3$ depends only on $K,N$.

<!-- proof-5: body_md -->
Since $f\in\mathcal B$, every Fatou component is simply connected \cite[Proposition~3, p.~993]{EL}. As $S(f)'$ is a compact subset of $\mathcal F(f)$, choose compact neighborhoods $K,N\subset\mathcal F(f)$ satisfying
\begin{equation}\label{eq:KN}
 S(f)'\subset\operatorname{Int}K,
 \qquad K\Subset\operatorname{Int}N\Subset\mathcal F(f).
\end{equation}
If $S(f)'=\varnothing$, take $K=N=\varnothing$. The set $E=S(f)\setminus\operatorname{Int}K$ is finite: otherwise compactness of $S(f)$ would give an accumulation point outside $\operatorname{Int}K$.

Suppose that $f$ has a wandering component. Since $N\cup E$ meets only finitely many Fatou components, after discarding finitely many initial forward components and reindexing, we obtain a sequence $(U_n)_{n\ge0}$ with $U_n\cap(N\cup E)=\varnothing$ for every $n$. Each $U_n$ is a component of $f^{-1}(U_{n+1})$ by complete invariance of the Fatou set. As $U_{n+1}$ avoids $S(f)\subset K\cup E$, the restriction $f:U_n\to U_{n+1}$ is a covering \cite[Section~3.2]{Sixsmith}; simple connectivity makes it a conformal isomorphism. Thus, with $W=\bigsqcup_{n\ge0}U_n$, the map $f:W\to W\setminus U_0$ is a conformal isomorphism.

Choose increasing finite unions $P_j\subset\mathcal J(f)$ of complete repelling cycles with $|P_1|\ge3$ and $\overline{\bigcup_jP_j}=\mathcal J(f)$ \cite[Theorem~4]{Bergweiler}. Set
\[
 X_j=\mathbb C\setminus P_j,\quad Y_j=X_j\setminus S(f),\qquad
 \,\mathrm d\mu_j=\,\mathrm d\alpha_{X_j},\quad
 \,\mathrm d\nu_j=\,\mathrm d\alpha_{Y_j}-\,\mathrm d\alpha_{X_j}\ge0.
\]
Because $W$ avoids $N\cup E$ and $S(f)\subset K\cup E$, Fact~\ref{fact:metric-properties} and Lemmas~\ref{lem:holes} and \ref{lem:punctures} give, with a constant independent of $j$,
\begin{align}\label{eq:entire-cost}
 \nu_j(W)
 &\le\int_W(\,\mathrm d\alpha_{X_j\setminus K}-\,\mathrm d\alpha_{X_j})\notag\\
 &\quad+\int_W(\,\mathrm d\alpha_{X_j\setminus(K\cup E)}
                   -\,\mathrm d\alpha_{X_j\setminus K})
 \le C+|E|.
\end{align}
For the first term, apply Lemma~\ref{lem:holes} with $F=P_j\cup\{\infty\}$; these sets lie in $\widehat{\mathbb C}\setminus N$ by \eqref{eq:KN} and have at least three points. The constant $C=C(K,N)$ is therefore independent of $j$. The second term is bounded by Lemma~\ref{lem:punctures} applied to $E\cap(X_j\setminus K)$.

Since each $P_j$ is a union of complete cycles, $f(P_j)=P_j$ and $f^{-1}(Y_j)\subset X_j$. As $Y_j$ avoids $S(f)$, the inverse-singularity covering theorem gives a covering $f:f^{-1}(Y_j)\to Y_j$ \cite[Section~3.2]{Sixsmith}. By Fact~\ref{fact:metric-properties},
\begin{equation}\label{eq:entire-pullback}
 f^*(\,\mathrm d\mu_j+\,\mathrm d\nu_j)
 =\,\mathrm d\alpha_{f^{-1}(Y_j)}\ge\,\mathrm d\mu_j
 \quad\text{on }f^{-1}(Y_j).
\end{equation}
We have $W\subset Y_j$ and $f(W)=W\setminus U_0$, so $W\subset f^{-1}(Y_j)$. Integrating \eqref{eq:entire-pullback} over $W$ and changing variables under $f:W\to W\setminus U_0$ yields
\[
 \mu_j(U_0)+\mu_j(W\setminus U_0)
 \le\mu_j(W\setminus U_0)+\nu_j(W\setminus U_0).
\]
The common term is finite, since $\mu_j(X_j)=|P_j|-1$ by Fact~\ref{fact:punctured-area}. Subtraction and \eqref{eq:entire-cost} give
\begin{equation}\label{eq:entire-bound}
 \mu_j(U_0)\le\nu_j(W\setminus U_0)\le C+|E|.
\end{equation}
\nopagebreak[4]
Finally, Lemma~\ref{lem:recovery} applied to $P_j\cup\{\infty\}$, monotone convergence and Fact~\ref{fact:infinite-area} imply $\mu_j(U_0)\to\alpha_{U_0}(U_0)=\infty$, contrary to \eqref{eq:entire-bound}.

<!-- corollary-1: body_md -->
No function in $\mathcal S$ has a wandering Fatou component.
<!-- end corollary-1 -->
<!-- proof-6: body_md -->
For $f\in\mathcal S$, the set $S(f)$ is finite, hence $S(f)'=\varnothing$. The conclusion follows from Theorem~\ref{thm:entire}.

<!-- example-1: body_md -->
\label{ex:sine}
Let $\lambda=e^{2\pi i\theta}$, where $\theta$ is Diophantine irrational, and set
\[
 h_\lambda(z)=\lambda\frac{\sin^2z}{z},\qquad h_\lambda(0)=0.
\]
Then $h_\lambda\in\mathcal B\setminus\mathcal S$ and $S(h_\lambda)'=\{0\}$; see \cite[Example~4.25]{Y} for the singular-value calculation. Since $h_\lambda'(0)=\lambda$, the origin lies in a Siegel disk \cite{Siegel}, so Theorem~\ref{thm:entire} excludes wandering domains. This family is covered by neither the finite-type theorem nor the no-wandering criteria of \cite{ARS,MBRG,PS}.
