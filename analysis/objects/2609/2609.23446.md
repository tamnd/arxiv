\label{thm:main}
Assume \eqref{eq:pq}, \eqref{eq:V}, and \eqref{eq:gap}. Then there exist constants
$$
\lambda_\infty>0,
\qquad C>0,
$$
depending only on $N,p,q,V$, such that for every $\lambda\geq\lambda_\infty$ equation \eqref{eq:original} possesses a nontrivial weak solution
$$
u_\lambda\in H^1(\mathbb{R}^N)\cap L^\infty(\mathbb{R}^N).
$$
Moreover,
$$
\left\lVert u_\lambda\right\rVert_{H^1(\mathbb{R}^N)}
+\left\lVert u_\lambda\right\rVert_{L^\infty(\mathbb{R}^N)}
\leq C\lambda^{-1/(q-2)}.
$$

\qquad

<!-- lemma-1: body_md -->
\label{lem:truncation}
For every $\varepsilon>0$ and every $t\in\mathbb{R}$ the following hold:
\begin{align}
(1-\delta)|t|^q
&\leq f_\varepsilon(t)t\leq |t|^q,\label{eq:ft-bounds}\\
\frac{1-\delta}{q}|t|^q
&\leq F_\varepsilon(t)\le\frac1q|t|^q,\label{eq:F-bounds}\\
|f_\varepsilon(t)|&\leq |t|^{q-1}.\label{eq:f-growth}
\end{align}
Moreover
\begin{enumerate}[label=\textup{(\roman*)}]
\item $f_\varepsilon$ is odd, continuous, locally Lipschitz, and $f_\varepsilon(t)t>0$ for $t\ne0$;
\item $f_\varepsilon(t)=o(t)$ as $t\to0$;
\item $F_\varepsilon(t)/t^2\to\infty$ as $|t|\to\infty$;
\item the map
$$
t\mapsto\frac{f_\varepsilon(t)}{|t|}
$$
is strictly increasing on $(-\infty,0)$ and on $(0,\infty)$;
\item with
$$
\theta:=q(1-\delta)>2,
$$
one has the uniform Ambrosetti-Rabinowitz condition (cf. \cite{AmbrosettiRabinowitz1973})
\begin{equation}\label{eq:AR}
0<\theta F_\varepsilon(t)\leq f_\varepsilon(t)t,
\qquad t\ne0.
\end{equation}
\end{enumerate}

<!-- proof-1: body_md -->
By oddness it suffices to consider $t\geq0$. If $0\leq t\leq M_\varepsilon$, then
\begin{equation}\label{eq:f-inside}
f_\varepsilon(t)=t^{q-1}-\varepsilon t^{p-1}
=t^{q-1}\bigl(1-\varepsilon t^{p-q}\bigr).
\end{equation}
Since $t\leq M_\varepsilon$, \eqref{eq:epsM} gives
$$
0\le\varepsilon t^{p-q}\le\varepsilon M_\varepsilon^{p-q}=\delta.
$$
Hence
$$
(1-\delta)t^{q-1}\leq f_\varepsilon(t)\leq t^{q-1}.
$$
If $t\geq M_\varepsilon$, then by \eqref{eq:aM} and \eqref{eq:epsM},
\begin{equation}\label{eq:f-outside}
f_\varepsilon(t)
=t^{q-1}-\varepsilon M_\varepsilon^{p-q}t^{q-1}
=(1-\delta)t^{q-1}.
\end{equation}
This proves \eqref{eq:ft-bounds}, \eqref{eq:f-growth}, and the positivity statement.

We compute the primitive explicitly. For $0\leq t\leq M_\varepsilon$,
\begin{equation}\label{eq:F-inside}
F_\varepsilon(t)=\frac{t^q}{q}-\varepsilon\frac{t^p}{p}.
\end{equation}
Thus $F_\varepsilon(t)\leq t^q/q$, and, using $\varepsilon t^{p-q}\le\delta$,
$$
F_\varepsilon(t)
\geq t^q\left(\frac1q-\frac{\delta}{p}\right)
\geq \frac{1-\delta}{q}t^q,
$$
because
$$
\frac1q-\frac{\delta}{p}-\frac{1-\delta}{q}
=\delta\left(\frac1q-\frac1p\right)>0.
$$
For $t\geq M_\varepsilon$, let $A_M(t)=\int_0^t a_M(s)\,ds$. Then
\begin{align*}
A_{M_\varepsilon}(t)
&=\frac{M_\varepsilon^p}{p}
+M_\varepsilon^{p-q}\frac{t^q-M_\varepsilon^q}{q}.
\end{align*}
Using $\varepsilon M_\varepsilon^{p-q}=\delta$ and $\varepsilon M_\varepsilon^p=\delta M_\varepsilon^q$, we obtain
\begin{align}
F_\varepsilon(t)
&=\frac{t^q}{q}-\varepsilon A_{M_\varepsilon}(t)\notag\\
&=\frac{1-\delta}{q}t^q
+\delta M_\varepsilon^q\left(\frac1q-\frac1p\right).
\label{eq:F-outside}
\end{align}
The second term in \eqref{eq:F-outside} is nonnegative, proving the lower bound in \eqref{eq:F-bounds}. The upper bound follows either from \eqref{eq:F-outside} directly or by integrating $f_\varepsilon(s)\leq s^{q-1}$.

The behavior at zero and infinity is now immediate from \eqref{eq:f-inside}, \eqref{eq:F-bounds}, and $q>2$.

It remains to prove the strict monotonicity of $f_\varepsilon(t)/t$ on $(0,\infty)$. For $0<t<M_\varepsilon$,
$$
\frac{f_\varepsilon(t)}{t}=t^{q-2}-\varepsilon t^{p-2},
$$
and therefore
\begin{align*}
\frac{d}{dt}\left(\frac{f_\varepsilon(t)}{t}\right)
&=t^{q-3}\left((q-2)-\varepsilon(p-2)t^{p-q}\right)\\
&\geq t^{q-3}\left((q-2)-(p-2)\delta\right)>0
\end{align*}
by \eqref{eq:delta-choice}. For $t>M_\varepsilon$, \eqref{eq:f-outside} gives
$$
\frac{f_\varepsilon(t)}{t}=(1-\delta)t^{q-2},
$$
which is strictly increasing. At $t=M_\varepsilon$ the two expressions coincide
$$
M_\varepsilon^{q-2}-\varepsilon M_\varepsilon^{p-2}
=(1-\delta)M_\varepsilon^{q-2}.
$$
Oddness gives the corresponding monotonicity on $(-\infty,0)$.

Note that, for $0<t<M_\varepsilon$,
$$
f_\varepsilon'(t)
=(q-1)t^{q-2}-\varepsilon(p-1)t^{p-2}
\geq\bigl((q-1)-(p-1)\delta\bigr)t^{q-2}>0,
$$
while for $t>M_\varepsilon$,
$$
f_\varepsilon'(t)=(q-1)(1-\delta)t^{q-2}.
$$
Thus $f_\varepsilon$ is locally Lipschitz (in fact, piecewise $C^1$) and nondecreasing. Its a.e. derivative satisfies a bound of the form
$$
|f_\varepsilon'(t)|\leq C|t|^{q-2},
$$
where $C$ depends on $p,q,\delta$, but not on $\varepsilon$.

Finally, \eqref{eq:ft-bounds} and the upper estimate in \eqref{eq:F-bounds} yield
$$
f_\varepsilon(t)t
\geq(1-\delta)|t|^q
=q(1-\delta)\frac{|t|^q}{q}
\geq q(1-\delta)F_\varepsilon(t).
$$
Since $q(1-\delta)>2$ by \eqref{eq:delta-choice}, this is \eqref{eq:AR}.

<!-- lemma-2: body_md -->
\label{lem:scalar}
Let $f:\mathbb{R}\to\mathbb{R}$ be odd, continuous, $f(t)t>0$ for $t\ne0$, and assume that $t\mapsto f(t)/|t|$ is strictly increasing on each of $(-\infty,0)$ and $(0,\infty)$. Let $F(t)=\int_0^t f(s)\,ds$. Then, for all $a,b\in\mathbb{R}$ and all $t\geq0$,
\begin{equation}\label{eq:scalar-ineq}
f(a)\left(\frac{t^2-1}{2}a+tb\right)
+F(a)-F(ta+b)
\le0.
\end{equation}
If $a\ne0$, equality in \eqref{eq:scalar-ineq} is possible only when $t=1$ and $b=0$.

\noindent \textbf{Step 1: the geometry.}
Fix $y\in Y\setminus\{0\}$. Since $f_\varepsilon(t)=o(t)$ at zero,
$$
\Phi_\varepsilon(ty)
=\frac{t^2}{2}\left\lVert y\right\rVert^2+o(t^2)>0
$$
for all sufficiently small $t>0$. On the other hand, $\Phi_\varepsilon(z)\le0$ for $z\in Z$.

The functional $\Phi_\varepsilon$ tends to $-\infty$ along unbounded sequences in the fiber $\widehat E(y)$. To see this, let
$$
w_n=t_ny+z_n\in\widehat E(y),
\qquad
\left\lVert w_n\right\rVert\to\infty,
$$
and put $r_n=\left\lVert w_n\right\rVert$ and $\widetilde w_n=w_n/r_n$. Passing to a subsequence,
$$
\frac{t_n}{r_n}\to t_0\geq0,
\qquad
\frac{z_n}{r_n}\rightharpoonup z_0\in Z.
$$
If $t_0=0$, then, by orthogonality of $Y$ and $Z$,
$$
\frac{\left\lVert z_n\right\rVert^2}{r_n^2}\to1.
$$
Since $F_\varepsilon\geq0$,
$$
\frac{\Phi_\varepsilon(w_n)}{r_n^2}
\leq \frac12\frac{t_n^2\left\lVert y\right\rVert^2}{r_n^2}
-\frac12\frac{\left\lVert z_n\right\rVert^2}{r_n^2}
\longrightarrow-\frac12.
$$
If $t_0>0$, then
$$
\widetilde w_n\rightharpoonup t_0y+z_0\ne0,
$$
because $t_0y\in Y\setminus\{0\}$ and $z_0\in Z$. By local compactness, after passing to a subsequence $\widetilde w_n(x)\to\widetilde w(x)$ a.e., where $\widetilde w=t_0y+z_0\ne0$. On a set of positive measure, $\widetilde w(x)\ne0$; there
$$
|w_n(x)|=r_n|\widetilde w_n(x)|\to\infty.
$$
Since $F_\varepsilon(t)/t^2\to\infty$, Fatou's lemma gives
$$
\frac1{r_n^2}\int_{\mathbb{R}^N}F_\varepsilon(w_n)\,dx\to\infty.
$$
Hence $\Phi_\varepsilon(w_n)\to-\infty$.

The function $F_\varepsilon$ is convex because $f_\varepsilon$ is nondecreasing (see the proof of Lemma \ref{lem:truncation}). Therefore $w\mapsto\int_{\mathbb{R}^N} F_\varepsilon(w) \, dx$ is weakly lower semicontinuous. The negative quadratic term on $Z$ is weakly upper semicontinuous. It follows that $\Phi_\varepsilon|_{\widehat E(y)}$ attains a positive global maximum at some point
$$
m_\varepsilon(y)=t_yy+z_y,
\qquad t_y>0,
\quad z_y\in Z.
$$
At such a maximum the derivative vanishes in the $y$ direction and in all $Z$ directions, hence $m_\varepsilon(y)\in\mathcal{M}_\varepsilon$.

Conversely, let $v\in\mathcal{M}_\varepsilon$. Write $v=y+z$ with $y\in Y\setminus\{0\}$ and $z\in Z$. For any $t\geq0$ and $\zeta\in Z$, the identities
$$
\Phi_\varepsilon'(v)(v)=0,
\qquad
\Phi_\varepsilon'(v)(\zeta)=0
$$
yield, after expanding the quadratic part,
\begin{align*}
\Phi_\varepsilon(tv+\zeta)-\Phi_\varepsilon(v)
&=-\frac12\left\lVert \zeta\right\rVert^2\\
&\quad+\int_{\mathbb{R}^N}
f_\varepsilon(v)\left(\frac{t^2-1}{2}v+t\zeta\right)
+F_\varepsilon(v)-F_\varepsilon(tv+\zeta)
 \, dx.
\end{align*}
Lemma \ref{lem:scalar}, applied pointwise with $a=v(x)$ and $b=\zeta(x)$, implies
$$
\Phi_\varepsilon(tv+\zeta)\le\Phi_\varepsilon(v).
$$
Strict monotonicity implies uniqueness of the maximizer on each fiber. Consequently the map
$$
S_Y\ni y\mapsto m_\varepsilon(y)\in\mathcal{M}_\varepsilon
$$
is a well-defined bijection (in fact, it is an homeomorphism), and
$$
\inf_{v\in\mathcal{M}_\varepsilon}\Phi_\varepsilon(v)
=\inf_{y\in S_Y}\max_{w\in\widehat E(y)}\Phi_\varepsilon(w).
$$

Then, one can show (see \cite{SzulkinWeth2009}) that $\Psi$ defined as
$$
\Psi_\varepsilon(y):=\Phi_\varepsilon(m_\varepsilon(y)),
\qquad y\in S_Y,
$$
is of class $C^1$, and for $\eta\in T_yS_Y$,
$$
\Psi_\varepsilon'(y)(\eta)
=\left\lVert P_Ym_\varepsilon(y)\right\rVert\,
\Phi_\varepsilon'(m_\varepsilon(y))(\eta).
$$
Ekeland's variational principle applied to $\Psi_\varepsilon$ therefore yields a sequence $(w_n)\subset\mathcal{M}_\varepsilon$ such that
\begin{equation}\label{eq:PSM}
\Phi_\varepsilon(w_n)\to c_\varepsilon,
\qquad
\Phi_\varepsilon'(w_n)\to0.
\end{equation}


\noindent \textbf{Step 2: positivity and boundedness on $\mathcal{M}_\varepsilon$.}
Let $w=y+z\in\mathcal{M}_\varepsilon$, with $y\in Y=X^-$ and $z\in Z=X^+$. Since $\Phi_\varepsilon'(w)(z)=0$ and $\Phi_\varepsilon'(w)(w)=0$, we also have $\Phi_\varepsilon'(w)(y)=0$. Thus
\begin{align*}
\left\lVert y\right\rVert^2&=\int_{\mathbb{R}^N}f_\varepsilon(w)y\,dx,\\
\left\lVert z\right\rVert^2&=-\int_{\mathbb{R}^N}f_\varepsilon(w)z\,dx.
\end{align*}
Using \eqref{eq:f-growth}, H\"older inequality, and \eqref{eq:sobolev-spectral},
\begin{align}
\left\lVert y\right\rVert
&\leq C\left\lVert w\right\rVert_{L^q(\mathbb{R}^N)}^{q-1},
\qquad
\left\lVert z\right\rVert
\leq C\left\lVert w\right\rVert_{L^q(\mathbb{R}^N)}^{q-1}.
\label{eq:component-bound-M}
\end{align}
Hence
\begin{equation}\label{eq:norm-vs-Lq-M}
\left\lVert w\right\rVert\leq C\left\lVert w\right\rVert_{L^q(\mathbb{R}^N)}^{q-1}
\leq C\left\lVert w\right\rVert^{q-1}.
\end{equation}
Because $w\ne0$ and $q>2$, \eqref{eq:norm-vs-Lq-M} gives a number $\rho_0>0$, independent of $\varepsilon$, such that
\begin{equation}\label{eq:M-away-zero}
\left\lVert w\right\rVert\geq\rho_0,
\qquad \mbox{for all }  w\in\mathcal{M}_\varepsilon.
\end{equation}
Combining the first inequality in \eqref{eq:norm-vs-Lq-M} with \eqref{eq:M-away-zero}, we also obtain
\begin{equation}\label{eq:Lq-away-zero}
\left\lVert w\right\rVert_{L^q(\mathbb{R}^N)}\geq\eta_0>0,
\qquad \mbox{for all } w\in\mathcal{M}_\varepsilon,
\end{equation}
with $\eta_0$ independent of $\varepsilon$.

Moreover, since $\Phi_\varepsilon'(w)(w)=0$,
\begin{align}
\Phi_\varepsilon(w)
&=\Phi_\varepsilon(w)-\frac12\Phi_\varepsilon'(w)(w)=\int_{\mathbb{R}^N}
\frac12 f_\varepsilon(w)w-F_\varepsilon(w) \, dx.
\label{eq:energy-identity}
\end{align}
By \eqref{eq:ft-bounds} and \eqref{eq:F-bounds},
\begin{equation}\label{eq:kappa}
\frac12 f_\varepsilon(t)t-F_\varepsilon(t)
\geq\left(\frac{1-\delta}{2}-\frac1q\right)|t|^q
=:\kappa_\delta |t|^q, \qquad t \in \mathbb{R},
\end{equation}
where $\kappa_\delta>0$ by \eqref{eq:delta-choice}. Hence, using \eqref{eq:Lq-away-zero},
\begin{equation}\label{eq:c-positive-uniform}
\Phi_\varepsilon(w)\geq\kappa_\delta\left\lVert w\right\rVert_{L^q(\mathbb{R}^N)}^q
\geq\kappa_\delta\eta_0^q>0.
\end{equation}
In particular $c_\varepsilon>0$.

If $\Phi_\varepsilon(w)$ is bounded on a subset of $\mathcal{M}_\varepsilon$, then \eqref{eq:c-positive-uniform} bounds $\left\lVert w\right\rVert_{L^q}$; \eqref{eq:component-bound-M} then bounds $\left\lVert w\right\rVert$. Thus $\Phi_\varepsilon|_{\mathcal{M}_\varepsilon}$ is coercive. In particular, the Palais-Smale sequence in \eqref{eq:PSM} is bounded.


\noindent \textbf{Step 3: concentration-compactness.}
Let $(w_n)\subset\mathcal{M}_\varepsilon$ be the bounded Palais-Smale sequence from \eqref{eq:PSM}. Suppose it vanishes in the sense of Lions, i.e. for some $R>0$,
$$
\sup_{\xi \in\mathbb{R}^N}\int_{B(\xi, R)}|w_n|^2\,dx\to0.
$$
Then the Lions concentration-compactness lemma \cite{Lions1984} gives
\begin{equation}\label{eq:Lions-vanish}
\left\lVert w_n\right\rVert_{L^q(\mathbb{R}^N)}\to0
\end{equation}
because $2<q<2^*$. But \eqref{eq:component-bound-M} would imply $\left\lVert w_n\right\rVert\to0$, contradicting \eqref{eq:M-away-zero}. Therefore the sequence does not vanish.

Consequently, after replacing arbitrary concentration centers by nearby lattice points, there exist $R>0$, $\eta>0$, and $k_n\in\mathbb{Z}^N$ such that
\begin{equation}\label{eq:nonvanish-ball}
\int_{B(0,R)}|w_n(x+k_n)|^2\,dx\geq\eta.
\end{equation}
Set
$$
\widetilde w_n(x)=w_n(x+k_n).
$$
By periodicity, $(\widetilde w_n)$ is again a bounded Palais-Smale sequence for $\Phi_\varepsilon$. After passing to a subsequence,
\begin{equation*}
\widetilde w_n\rightharpoonup v_\varepsilon
\quad\mbox{in }H^1(\mathbb{R}^N),
\end{equation*}
and
$$
\widetilde w_n\to v_\varepsilon
\quad\text{in }L^r_{\mathrm{loc}}(\mathbb{R}^N),
\qquad 2\leq r<2^*,
$$
and a.e. Since $q<2^*$ and $|f_\varepsilon(t)|\le|t|^{q-1}$, the Nemytskii operator is locally continuous from $L^q$ to $L^{q'}$. Hence, for every $\varphi\in C_c^\infty(\mathbb{R}^N)$,
$$
\Phi_\varepsilon'(v_\varepsilon)(\varphi)
=\lim_{n\to\infty}\Phi_\varepsilon'(\widetilde w_n)(\varphi)=0.
$$
By density, $\Phi_\varepsilon'(v_\varepsilon)=0$. The strong convergence in $L^2_{\mathrm{loc}}$ together with \eqref{eq:nonvanish-ball} gives $v_\varepsilon\ne0$.

By \eqref{eq:kappa}
$$
\frac12f_\varepsilon(t)t-F_\varepsilon(t) \geq 0, \qquad t \in \mathbb{R}.
$$
Since $v_\varepsilon$ is a nontrivial critical point, $v_\varepsilon\in\mathcal{M}_\varepsilon$, and therefore
$$
\Phi_\varepsilon(v_\varepsilon)
=\int_{\mathbb{R}^N} \frac12f_\varepsilon(v_\varepsilon)v_\varepsilon-F_\varepsilon(v_\varepsilon)  \, dx.
$$
Fatou's lemma and \eqref{eq:energy-identity} yield
\begin{align*}
\Phi_\varepsilon(v_\varepsilon)
&\leq \liminf_{n\to\infty}
\int_{\mathbb{R}^N} \frac12f_\varepsilon(\widetilde w_n)\widetilde w_n-F_\varepsilon(\widetilde w_n)   \, dx\\
&=\liminf_{n\to\infty}\Phi_\varepsilon(w_n)=c_\varepsilon.
\end{align*}
On the other hand, $v_\varepsilon\in\mathcal{M}_\varepsilon$ implies $\Phi_\varepsilon(v_\varepsilon)\geq c_\varepsilon$. Thus equality holds and \eqref{eq:c-eps} follows.

\label{lem:ca-scaling}
For every $a>0$,
\begin{equation}\label{eq:ca-scaling}
c_a=a^{-2/(q-2)}c_1.
\end{equation}

For $v\in H^1(\mathbb{R}^N)$, put $w=a^{1/(q-2)}v$, i.e.

v=a^{-1/(q-2)}w.

Then
\begin{align*}
\Phi_a(v)
&=\frac12a^{-2/(q-2)}

&=a^{-2/(q-2)}\Phi_1(w),
\end{align*}

$$

<!-- proposition-2: body_md -->
\label{prop:H1-uniform}
There exist $c_H, C_H>0$, independent of $\varepsilon>0$, such that the ground state $v_\varepsilon$ in Proposition \ref{prop:ground-truncated} satisfies
\begin{equation}\label{eq:H1-uniform}
c_H \leq \left\lVert v_\varepsilon\right\rVert_{H^1(\mathbb{R}^N)}\leq C_H.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Since $v_\varepsilon$ is a critical point of $\Phi_\varepsilon$,
$$
\Phi_\varepsilon(v_\varepsilon)
=\int_{\mathbb{R}^N}
\frac12 f_\varepsilon(v_\varepsilon)v_\varepsilon-F_\varepsilon(v_\varepsilon) \, dx.
$$
By \eqref{eq:kappa} and Lemma \ref{lem:c-uniform},
\begin{equation*}
\kappa_\delta\left\lVert v_\varepsilon\right\rVert_{L^q(\mathbb{R}^N)}^q
\leq c_\varepsilon\leq C_c.
\end{equation*}
Thus
\begin{equation}\label{eq:Lq-C}
\left\lVert v_\varepsilon\right\rVert_{L^q}\leq C_q
:=\left(\frac{C_c}{\kappa_\delta}\right)^{1/q}.
\end{equation}

Write
$$
v_\varepsilon=v_\varepsilon^-+v_\varepsilon^+,
\qquad v_\varepsilon^-\in X^-,\quad v_\varepsilon^+\in X^+.
$$
Testing \eqref{eq:truncated} by $v_\varepsilon^-$ gives
$$
-\left\lVert v_\varepsilon^-\right\rVert^2
=-\int_{\mathbb{R}^N}f_\varepsilon(v_\varepsilon)v_\varepsilon^-\,dx,
$$
so
\begin{equation}\label{eq:minus-test}
\left\lVert v_\varepsilon^-\right\rVert^2
=\int_{\mathbb{R}^N}f_\varepsilon(v_\varepsilon)v_\varepsilon^-\,dx.
\end{equation}
Similarly, testing by $v_\varepsilon^+$ gives
\begin{equation}\label{eq:plus-test}
\left\lVert v_\varepsilon^+\right\rVert^2
=-\int_{\mathbb{R}^N}f_\varepsilon(v_\varepsilon)v_\varepsilon^+\,dx.
\end{equation}
Using \eqref{eq:f-growth}, H\"older's inequality, and \eqref{eq:sobolev-spectral},
\begin{align*}
\left\lVert v_\varepsilon^-\right\rVert^2
&\le\left\lVert v_\varepsilon\right\rVert_{L^q(\mathbb{R}^N)}^{q-1}\left\lVert v_\varepsilon^-\right\rVert_{L^q(\mathbb{R}^N)}
\leq C_q^{q-1}C_S\left\lVert v_\varepsilon^-\right\rVert,\\
\left\lVert v_\varepsilon^+\right\rVert^2
&\le\left\lVert v_\varepsilon\right\rVert_{L^q(\mathbb{R}^N)}^{q-1}\left\lVert v_\varepsilon^+\right\rVert_{L^q(\mathbb{R}^N)}
\leq C_q^{q-1}C_S\left\lVert v_\varepsilon^+\right\rVert.
\end{align*}
Hence
$$
\left\lVert v_\varepsilon^-\right\rVert+\left\lVert v_\varepsilon^+\right\rVert\leq C
$$
with $C$ independent of $\varepsilon$. Norm equivalence gives the right inequality in \eqref{eq:H1-uniform}.

For the lower bound, the same two component estimates, without using \eqref{eq:Lq-C}, give
$$
\left\lVert v_\varepsilon\right\rVert
\leq C\left\lVert v_\varepsilon\right\rVert_{L^q(\mathbb{R}^N)}^{q-1}
\leq C\left\lVert v_\varepsilon\right\rVert^{q-1}.
$$
Since $v_\varepsilon\ne0$ and $q>2$, division by $\left\lVert v_\varepsilon\right\rVert$ yields
$$
\left\lVert v_\varepsilon\right\rVert^{q-2}\geq C^{-1}.
$$
Thus $\left\lVert v_\varepsilon\right\rVert\geq c>0$, and norm equivalence gives the left part of \eqref{eq:H1-uniform}.

<!-- lemma-5: body_md -->
\label{lem:Linfty}
Let $(h_n)$ be a sequence of continuous functions satisfying
\begin{equation}\label{eq:h-growth}
|h_n(t)|\leq |t|^{q-1},
\qquad t\in\mathbb{R},
\end{equation}
where $2<q<2^*$. Suppose $u_n\in H^1(\mathbb{R}^N)$ are weak solutions of
$$
-\Delta u_n+V(x)u_n=-h_n(u_n)
$$
and
$$
\sup_n\left\lVert u_n\right\rVert_{H^1(\mathbb{R}^N)}\leq C_H.
$$
Then there exists $C_\infty>0$, depending only on $N,q,\left\lVert V\right\rVert_{L^\infty(\mathbb{R}^N)},C_H$, such that
\begin{equation}\label{eq:Linfty-unif}
\sup_n\left\lVert u_n\right\rVert_{L^\infty(\mathbb{R}^N)}\leq C_\infty.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
The argument is a finite local $W^{2,s}$ bootstrap, made uniform with respect to translations in $\mathbb{R}^N$.

By Sobolev embeddings,
$$
\sup_n\left\lVert u_n\right\rVert_{L^{r_0}(\mathbb{R}^N)}\leq C,
\qquad r_0:=2^*=\frac{2N}{N-2}.
$$
Fix an arbitrary $x_0\in\mathbb{R}^N$. We shall use nested balls centered at $x_0$; the constants in the interior estimates do not depend on $x_0$ because the principal part is $-\Delta$ and $\left\lVert V\right\rVert_{L^\infty(\mathbb{R}^N)}$ is fixed.

Assume at some stage that, on a ball $B(x_0,R)$,
$$
\left\lVert u_n\right\rVert_{L^{r}(B(x_0,R))}\leq C_R
$$
for some $r\geq r_0$, uniformly in $n,x_0$. Put $s:=\frac{r}{q-1}$. Since $r\geq r_0=2^*>q-1$, we have $s>1$. By \eqref{eq:h-growth},
$$
\left\lVert h_n(u_n)\right\rVert_{L^s(B(x_0,R))}
\le\left\lVert u_n\right\rVert_{L^r(B(x_0,R))}^{q-1}\leq C.
$$
Also $s\leq r$, hence on the bounded ball
$$
\left\lVert V u_n\right\rVert_{L^s(B(x_0,R))}
\leq C\left\lVert V\right\rVert_{L^\infty(\mathbb{R}^N)}\left\lVert u_n\right\rVert_{L^r(B(x_0,R))}.
$$
The Calder\'on-Zygmund inequality for
$$
-\Delta u_n=-V(x) u_n-h_n(u_n)
$$
gives, for some $0<r_1<R$,
\begin{equation}\label{eq:W2s}
\left\lVert u_n\right\rVert_{W^{2,s}(B(x_0,r_1))}
\leq C\left(
\left\lVert Vu_n+h_n(u_n)\right\rVert_{L^s(B(x_0,R))}
+\left\lVert u_n\right\rVert_{L^s(B(x_0,R))}
\right)
\leq C.
\end{equation}
The constant is uniform in $n,x_0$.

If $2s>N$, the embedding $W^{2,s}\hookrightarrow L^\infty$ gives the desired local $L^\infty$ bound. Since $x_0$ is arbitrary, the global bound follows.

If $2s=N$, then $W^{2,N/2}$ embeds into $L^r$ for every finite $r$. We choose a sufficiently large finite exponent $r$ so that, at the next step, $r/(q-1)>N/2$; then the preceding case applies.

Suppose therefore that $2s<N$. Sobolev embedding for $W^{2,s}$ gives the new exponent
$$
r_{\mathrm{new}}
=\frac{Ns}{N-2s},
\qquad
\frac1{r_{\mathrm{new}}}
=\frac1s-\frac2N
=\frac{q-1}{r}-\frac2N.
$$
We iterate this construction on finitely many nested balls. To see that the iteration reaches the $L^\infty$ regime after finitely many steps, define $r_j$ by $r_0=2^*$ and, as long as $2r_j/(q-1)<N$,
$$
\frac1{r_{j+1}}
=(q-1)\frac1{r_j}-\frac2N.
$$
Let
$$
a_j:=\frac1{r_j},
\qquad
a_*:=\frac{2}{N(q-2)}.
$$
Then
$$
a_{j+1}-a_*
=(q-1)(a_j-a_*),
$$
so
\begin{equation}\label{eq:affine-solution}
a_j-a_*=(q-1)^j(a_0-a_*).
\end{equation}
Now
$$
a_0=\frac{N-2}{2N},
$$
and the subcritical condition $q<2^*=2N/(N-2)$ is exactly
$$
\frac{N-2}{2N}<\frac{2}{N(q-2)}=a_*.
$$
Hence $a_0-a_*<0$. Since $q-1>1$, the right-hand side of \eqref{eq:affine-solution} tends to $-\infty$. Therefore, after finitely many iterations, we must reach
$$
\frac{r_j}{q-1}>\frac N2,
$$
i.e. $2s_j>N$, or the borderline case $2s_j=N$, which was already handled.

Choose in advance a finite nested chain of balls, for example
$$
B(x_0, 2)\supset B(x_0, 2-1/m)\supset\cdots\supset B(x_0, 1),
$$
with $m$ larger than the finite number of bootstrap steps. Applying \eqref{eq:W2s} successively yields
$$
\left\lVert u_n\right\rVert_{L^\infty(B(x_0,1))}\leq C_\infty
$$
with $C_\infty$ independent of $n$ and $x_0$. Taking the supremum over $x_0\in\mathbb{R}^N$ proves \eqref{eq:Linfty-unif}.

<!-- corollary-1: body_md -->
\label{cor:Linfty-veps}
There exists $C_\infty>0$, independent of $\varepsilon>0$, such that
\begin{equation}\label{eq:veps-Linfty}
\left\lVert v_\varepsilon\right\rVert_{L^\infty(\mathbb{R}^N)}\leq C_\infty.
\end{equation}
<!-- end corollary-1 -->
<!-- proof-7: body_md -->
Choose
$$
\varepsilon_0
:=\min\left\{1,\ \delta C_\infty^{-(p-q)}\right\}>0.
$$
If $0<\varepsilon\le\varepsilon_0$, then
$$
M_\varepsilon
=\left(\frac\delta\varepsilon\right)^{1/(p-q)}
\geq C_\infty.
$$
Together with \eqref{eq:veps-Linfty}, this gives
$$
|v_\varepsilon(x)|\leq M_\varepsilon
\quad\text{for a.e. }x\in\mathbb{R}^N.
$$
Therefore, by the first branch in \eqref{eq:aM},
$$
a_{M_\varepsilon}(v_\varepsilon)
=|v_\varepsilon|^{p-2}v_\varepsilon
\quad\text{a.e. in }\mathbb{R}^N.
$$
Equation \eqref{eq:truncated} hence becomes
\begin{align*}
Lv_\varepsilon
&=-\left(|v_\varepsilon|^{q-2}v_\varepsilon
-\varepsilon |v_\varepsilon|^{p-2}v_\varepsilon\right)=\varepsilon |v_\varepsilon|^{p-2}v_\varepsilon-|v_\varepsilon|^{q-2}v_\varepsilon.
\end{align*}
Thus $v_\varepsilon$ is a nontrivial solution of the \emph{untruncated} equation \eqref{eq:scaled} whenever $0<\varepsilon\le\varepsilon_0$.

Define
$$
\lambda_\infty
:=\varepsilon_0^{-(q-2)/(p-2)}.
$$
For $\lambda\geq\lambda_\infty$, put
$$
\varepsilon(\lambda)=\lambda^{-(p-2)/(q-2)}\le\varepsilon_0
$$
and
\begin{equation}\label{eq:u-from-v}
u_\lambda
:=\lambda^{-1/(q-2)}v_{\varepsilon(\lambda)}.
\end{equation}
By the calculation in Section 2, $u_\lambda$ solves \eqref{eq:original}. It is nontrivial because $v_{\varepsilon(\lambda)}\ne0$.

Finally, Propositions \ref{prop:H1-uniform} and Corollary \ref{cor:Linfty-veps} yield
\begin{align*}
\left\lVert u_\lambda\right\rVert_{H^1(\mathbb{R}^N)}
&=\lambda^{-1/(q-2)}\left\lVert v_{\varepsilon(\lambda)}\right\rVert_{H^1(\mathbb{R}^N)}
\leq C_H\lambda^{-1/(q-2)},\\
\left\lVert u_\lambda\right\rVert_{L^\infty(\mathbb{R}^N)}
&=\lambda^{-1/(q-2)}\left\lVert v_{\varepsilon(\lambda)}\right\rVert_{L^\infty(\mathbb{R}^N)}
\leq C_\infty\lambda^{-1/(q-2)}.
\end{align*}
This proves Theorem \ref{thm:main}.

<!-- remark-1: body_md -->
\label{rem:not-ground-original}
The solution $v_\varepsilon$ is selected as a ground state of the truncated defocusing functional $\Phi_\varepsilon$. Once the truncation is shown to be inactive on this solution, $v_\varepsilon$ solves the original equation. We do not claim that it is a ground state among all solutions of the untruncated equation and the question about existence of a ground state remains open.
<!-- end remark-1 -->
<!-- proof-8: body_md -->
For all sufficiently large $n$, recall that $v_n:=v_{\varepsilon_n}$ satisfies the untruncated equation
\begin{equation}\label{eq:vn-exact}
Lv_n
=\varepsilon_n|v_n|^{p-2}v_n-|v_n|^{q-2}v_n.
\end{equation}
Proposition \ref{prop:H1-uniform} gives
$$
\sup_n\left\lVert v_n\right\rVert_{H^1(\mathbb{R}^N)}<\infty,
\qquad
\inf_n\left\lVert v_n\right\rVert_{H^1(\mathbb{R}^N)}>0.
$$
We claim that $(v_n)$ cannot vanish in the sense of Lions. Otherwise
$$
\left\lVert v_n\right\rVert_{L^q(\mathbb{R}^N)}\to0.
$$
Testing \eqref{eq:truncated} (or, equivalently, using \eqref{eq:minus-test}-\eqref{eq:plus-test}) and $|f_{\varepsilon_n}(t)|\le|t|^{q-1}$ gives
$$
\left\lVert v_n\right\rVert
\leq C\left\lVert v_n\right\rVert_{L^q(\mathbb{R}^N)}^{q-1}\to0,
$$
contradicting the uniform lower bound. Hence, after lattice translations, there exist $R,\eta>0$ such that
$$
\int_{B(0,R)}|v_n(x+k_n)|^2\,dx\geq\eta.
$$
Set $\widetilde v_n=v_n(\cdot+k_n)$. Periodicity of $V$ preserves \eqref{eq:vn-exact}. Boundedness gives, after a subsequence,
$$
\widetilde v_n\rightharpoonup v_0\quad\text{in }H^1(\mathbb{R}^N)
$$
and strong convergence in every $L^r_{\mathrm{loc}} (\mathbb{R}^N)$, $2 \leq r<2^*$. Moreover, $v_0\ne0$.

For the $p$-term,
\begin{equation}\label{eq:pterm-zero}
\left\lVert \varepsilon_n|\widetilde v_n|^{p-2}\widetilde v_n\right\rVert_{L^{p'}(\mathbb{R}^N)}
=\varepsilon_n\left\lVert \widetilde v_n\right\rVert_{L^p(\mathbb{R}^N)}^{p-1}
\leq C\varepsilon_n\to0.
\end{equation}
For the $q$-term, strong convergence in $L^q_{\mathrm{loc}} (\mathbb{R}^N)$ implies
$$
|\widetilde v_n|^{q-2}\widetilde v_n
\to |v_0|^{q-2}v_0
\quad\text{in }L^{q'}_{\mathrm{loc}} (\mathbb{R}^N).
$$
Passing to the limit in the weak formulation of \eqref{eq:vn-exact} yields \eqref{eq:profile-eq}.
