<!-- definition-5: body_md -->
Let $({\cal C},\langle\langle-\rangle\rangle,\textbf{T})$ denote a 2-category with shadow and adjoints for 1-morphisms, and let $F:\mathtt{U}\to \mathtt{V}$, $G\in\operatorname{End}_{\cal C}(\mathtt{V}),\, H\in\operatorname{End}_{\cal C}(\mathtt{U})$. The \textbf{shadow trace} of a 2-cell  $\eta: G\circ F\Rightarrow F\circ H$  is given by the composite in $\textbf{T}$:
    \begin{align}
        \operatorname{tr}_\textbf{T}(\eta)&: \langle\langle G\rangle\rangle_\mathtt{V} \xrightarrow{\langle\langle \operatorname{id}_G\circ \iota_F\rangle\rangle} \langle\langle G\circ (F\circ F^\dagger)\rangle\rangle_\mathtt{V} \cong \langle\langle (G\circ F)\circ F^\dagger\rangle\rangle_\mathtt{V} \nonumber\\
        &\qquad\qquad \quad\xrightarrow{\langle\langle\eta\circ F^\dagger\rangle\rangle}\langle\langle  (F\circ H)\circ F^\dagger\rangle\rangle_\mathtt{V} \xrightarrow{\theta_{F\circ H,F^\dagger}} \langle\langle F^\dagger\circ (F\circ H)\rangle\rangle_\mathtt{U} \nonumber\\
        &\qquad \qquad \quad \cong  \langle\langle (F^\dagger\circ  F)\circ H\rangle\rangle_\mathtt{U}\xrightarrow{\langle\langle e_F\circ H\rangle\rangle} \langle\langle H\rangle\rangle_\mathtt{U}\,,\label{shadowtrace}
    \end{align}
    where we have suppressed the morphisms induced by the composition associators $\alpha$. This morphism $ \langle\langle G\rangle\rangle_\mathtt{V}\to  \langle\langle H\rangle\rangle_\mathtt{U}$ in $\textbf{T}$ does not depend on the choice of the adjunction data $F^\dagger,\iota_F,e_F$.

<!-- theorem-1: body_md -->
\label{shadowproperties}
Let $({\cal C},\langle\langle-\rangle\rangle,\textbf{T})$ denote a bicategory with rigid adjunctions and a $\textbf{T}$-shadow. We shall suppress the composition associators $\alpha$ in the following.
    \begin{enumerate}
        \item \textbf{Tightening:} For 2-cells $\eta: G\circ F\Rightarrow F\circ H$, and $\nu: G'\Rightarrow G\,, \mu: H\Rightarrow H'$, we have 
        \begin{equation*}
            \langle\langle \mu\rangle\rangle\circ \operatorname{tr}_\textbf{T}(\eta)\circ \langle\langle\nu\rangle\rangle = \operatorname{tr}_\textbf{T}\left( (\nu\circ F)\ast \eta\ast (F\circ \mu) \right)
        \end{equation*}
        as a morphism $\langle\langle G'\rangle\rangle_\mathtt{V} \to \langle\langle H'\rangle\rangle_\mathtt{U}$ in $\textbf{T}$.
        \item \textbf{Cyclicity:} for 2-cells $\eta_1: G_1\circ F\Rightarrow F'\circ H_1$ and $\eta_2: G_2\circ F'\Rightarrow F\circ H_2$, we form the  composites 
        \begin{align*}
            \eta_1\ast_A \eta_2:\,& (G_1\circ G_2)\circ F'\cong G_1\circ( G_2\circ F')\xRightarrow{G_1\circ \eta_2}G_1\circ(F\circ H_2) \cong (G_1\circ F)\circ H_2\\
            &\qquad \xRightarrow{\eta_1\circ H_2} (F'\circ H_1)\circ H_2 \cong F'\circ (H_1\circ H_2)\\
            \eta_2\ast_A\eta_1 :\,& (G_2\circ G_1)\circ F\cong  G_2\circ (G_1\circ F)\xRightarrow{G_2\circ \eta_1} G_2\circ(F'\circ H_1) \cong (G_2\circ G')\circ H_1 \\
            &\qquad \xRightarrow{\eta_2\circ H_1} (F\circ H_2)\circ H_1\cong F\circ(H_2\circ H_1).
        \end{align*}
        The following diagram commutes,
                                                                                                                                                                                                                                                                                                                                                                                                                                                                         
\[\begin{tikzcd}
	{\langle\langle G_1\circ G_2\rangle\rangle} && {\langle\langle H_1\circ H_2\rangle\rangle} \\
	{\langle\langle G_2\circ G_1\rangle\rangle} && {\langle\langle H_1\circ H_2\rangle\rangle}
	\arrow["{\operatorname{tr}_{\bf T}(\eta_1\ast_A\eta_2)}", from=1-1, to=1-3]
	\arrow["{\theta_{G_1,G_2}}"', from=1-1, to=2-1]
	\arrow["{\theta_{H_1,H_2}}", from=1-3, to=2-3]
	\arrow["{\operatorname{tr}_{\bf T}(\eta_2\ast_A\eta_1)}"', from=2-1, to=2-3]
\end{tikzcd}\]
In particular if $\eta_1: F_1\Rightarrow F_2$ and $\eta_2: F_2\Rightarrow F_1$, then $$\operatorname{tr}_\textbf{T}(\eta_1\ast \eta_2)= \operatorname{tr}_\textbf{T}(\eta_2\ast\eta_1)\,.$$
        \item If $\eta: G\circ 1_\mathtt{V}\Rightarrow 1_\mathtt{V}\circ H$, then $\operatorname{tr}_\textbf{T}(\eta) = \langle\langle \eta\rangle\rangle_\mathtt{V}$.
        \item \textbf{1-functoriality:} for 2-cells $\eta_1: G\circ F_1\Rightarrow F_1\circ H$ and $\eta_2: H\circ F_2\Rightarrow F_2\circ K$, we form the composite
        \begin{align*}
            \eta_2\ast_B\eta_1:\,& G\circ (F_1\circ F_2)\cong (G\circ F_1)\circ F_2\xRightarrow{\eta_1\circ F_2} (F_1\circ H)\circ F_2\cong F_1\circ (H\circ F_2)\\
            &\qquad \xRightarrow{F_1\circ \eta_2} F_1\circ(F_2\circ K)\cong (F_1\circ F_2)\circ K\,.
        \end{align*}
        Then we have $$\operatorname{tr}_\textbf{T}(\eta_2\ast_B\eta_1) = \operatorname{tr}_\textbf{T}(\eta_2)\circ \operatorname{tr}_\textbf{T}(\eta_1): \langle\langle G\rangle\rangle \to \langle\langle H\rangle\rangle \to\langle\langle K\rangle\rangle\,.$$
        \item \textbf{Conjugation invariance:} The dagger adjunctions and the composition associators $\alpha $ in ${\cal C}$  give rise to the linear natural isomorphism
\begin{align}
    \bar\cdot &:\operatorname{2Hom}_{\cal C}(G\circ F,F'\circ G')\cong \operatorname{2Hom}_{\cal C}(F,G^\dagger\circ (F'\circ G'))\nonumber\\
    &\qquad\qquad\stackrel{\alpha}{\cong} \operatorname{2Hom}_{\cal C}(F,(G^\dagger\circ F')\circ G')\cong \operatorname{2Hom}_{\cal C}( F\circ G'^\dagger,G^\dagger\circ F')
\label{conjugation}
\end{align}
such that, for a cubical 2-cell $\eta: G\circ F\Rightarrow F\circ H$ we have
        $$\operatorname{tr}_\textbf{T}(\eta) = \operatorname{tr}_\textbf{T}(\bar\eta): \langle\langle G\rangle\rangle_\mathtt{V} \to \langle\langle H\rangle\rangle_\mathtt{U},$$
        where $ \operatorname{tr}_\textbf{T}(\bar\eta)$ is evaluated using the left-dualizability of $F^\dagger$.
    \end{enumerate}

<!-- example-1: body_md -->
\label{2charexample}
    Let us return to the example ${\cal C}=\operatorname{2Rep}(H)$ of interest. For each $\mathtt{U}\in{\cal C}$, consider the $\mathsf{Hilb}$-shadow given by the \textit{end}
    \begin{equation*}
        \langle\langle F\rangle\rangle_\mathtt{U} = \int_{x\in \mathtt{U}} \operatorname{Hom}_\mathtt{U}(x,F(x))\cong \operatorname{Nat}(1_\mathtt{U},F)
    \end{equation*}
    for each $F\in\operatorname{End}_{\cal C}(\mathtt{U})$.\footnote{In fact, for  2-representations ${\cal C}=\operatorname{2Rep}(G)$ (in $\mathsf{2Vect}$) of a finite (2-)group $G$, the end is exactly the \textit{2-character} $$\chi:\operatorname{2Rep}(G)\to \mathsf{Vect},\qquad \chi_{(M,\rho)}(g) = \operatorname{Nat}(1_M,\rho(g))$$ introduced in \cite{Ganter:2006,Ganter:2014,Bartlett:2009PhD,Huang:2024}, and the toroidal trace of a 2-cell $\eta_{g,h}: \rho(g)\rho(h)\Rightarrow\rho(h)\rho(g)$ coming from mutually commuting elements $g,h\in G$ is precisely the \textit{joint character}. The cyclicity operator $\theta$ furnishes $\chi_{(M,\rho)}$ as a representation of the double $D(G)\simeq G//G$.} Following Prop. 3.8 of \cite{Ganter:2006}, we have a linear map $ \tilde\theta_{F,G}:\langle\langle F'\rangle\rangle_\mathtt{U} \rightarrow \langle\langle G\circ F\circ G^\dagger\rangle\rangle_\mathtt{V}$ for each $G: \mathtt{U}\to \mathtt{V}$, which precomposes a natural transformation $1_\mathtt{U}\Rightarrow F$ by the counit $\iota_G: 1_\mathtt{V}\Rightarrow G\circ G^\dagger$. By dagger adjunctions \eqref{unitaryadjoint}, this map $\tilde\theta$ is invertible and induces the requisite cyclicity operator $\theta$ (cf. Remark 4.7 of \cite{Huang:2024}). 
    
    Recall that, for simple $\mathtt{U}\in{\cal C}$, its identity is simple; this is Prop. 1.2.14 in \cite{Douglas:2018}. Then, $\langle\langle 1_\mathtt{U}\rangle\rangle_\mathtt{U} = \operatorname{Nat}(1_\mathtt{U},1_\mathtt{U})\cong \mathbb{C}$ is the simple unit in $\textbf{T}=\mathsf{Hilb}$. Now specifically for finite groups $H=\mathsf{Hilb}_G$, we have $\Omega{\cal C} = \operatorname{End}_{\operatorname{2Rep}(G)}(\mathtt{I})\simeq \operatorname{Rep}(G)$ \cite{Bartsch:2022mpm} whence 
    \begin{equation*}
        \langle\langle F\rangle\rangle_\mathtt{I} \cong \operatorname{Nat}(1_\mathtt{I},F)\cong\operatorname{Hom}_G(\mathbb{C},V_F) \cong V^G_F
    \end{equation*}
    is the $G$-coinvariants of the representation determined by $F: \mathtt{I}\to \mathtt{I}$. The shadow trace of a 2-cells $ \eta: F\circ H'\Rightarrow H\circ F$ for $H',H\in\Omega{\cal C}$ can therefore be viewed as a linear map
    \begin{equation*}
        \operatorname{tr}_\mathsf{Vect}(\eta): V^{G}_{H'}\to V^{G}_H\,\,\quad \underline{\text{in } \mathsf{Hilb}}\,,
    \end{equation*}
    and the toroidal trace $\mathbb{T}\mathrm{r}(\eta)$ can then be seen as the usual trace of this matrix. The same computation can be performed for generic endomorphism categories $\operatorname{End}_H(\mathtt{U})$ in ${\cal C}=\operatorname{2Rep}(H).$

\label{2decorate}
    A \textbf{${\cal C}$-decoration on $\Delta$} is an assignment
    \begin{equation}
K([i]) = \mathtt{U}_{[i]},\qquad K([ij]) = F_{[ij]}: \mathtt{U}_{[i]}\to \mathtt{U}_{[j]},\qquad K([ijk]) = \eta_{[ijk]}: F_{[jk]}\circ F_{[ij]}\Rightarrow F_{[ik]}
\nonumber
\end{equation}
 such that, for each $i,j$,
   \begin{enumerate}
       \item $F_{[ji]} = F^\dagger_{[ij]}$ is the adjoint 1-morphism, 
       \item the transpose (2-morphism adjoint) of $ \eta_{[ijk]}$ is given by $\eta_{[ijk]}^T: F_{[ik]}\Rightarrow F_{[jk]}\circ F_{[ij]}$,
       \item the adjoint-mate of the 2-morphism $\eta_{[ijk]}$ is $\eta^\dagger_{[ijk]}=\eta_{[kji]}^T: F_{[ki]}\to F_{[ji]}\circ F_{[kj]}$,
       \item the identity 1-morphism $F_{[ii]} = 1_{\mathtt{U}_{[i]}}$ is (1-)self-adjoint,

<!-- definition-14: body_md -->
A \textbf{${\cal C}$-decoration on $\Gamma$} consists of the following assignments:
\begin{itemize}
    \item the 3-cells                                  
    are assigned with objects $\mathtt{U}\in{\cal C}$,
    \item a 2-cell interfacing two 3-cells with decorations $\mathtt{U},\mathtt{V}\in{\cal C}$ is assigned a 1-morphism $F:\mathtt{U}\to \mathtt{V}$,
    \item a 1-cell interfacing two 2-cells with decorations $F,G\in\operatorname{Hom}_{\cal C}(\mathtt{U},\mathtt{V})$ is assigned a 2-morphism $\eta:F\Rightarrow G$;
    \item 2-cells with the opposite orientation is assigned the 1-morphism adjoint $F^\dagger: {\mathtt{V}}\to {\mathtt{U}}$, and 
    \item 1-cells with the opposite orientation is assigned the 2-morphism adjoint $\eta^T: G\Rightarrow F$.
\end{itemize}
We denote such ${\cal C}$-decorated cells by $\gamma\in {\cal C}^\Gamma$, and call it a \textbf{membrane-net}.

 \end{equation}

Consider first the graph $\Gamma$ dual to a (positively-oriented) tetrahedron $\tau^+=\tau\subset D^3$ inscribed within a 3-ball $D^3$; see Fig. \ref{fig:lwhilbspace}. For an edge $e\in\Gamma$ dual to an ordered 2-simplex $\Delta=[012]$, its attached three surfaces $f_1,f_2,f_3$ are dual to respectively the 1-simplices $[01],[12],[20]$ respectively, and the 3-cells in $\pi_0(D^3\setminus\Gamma)$ are dual to the 0-simplices $[0],[1],[2]$. These correspondences are understood implicitly in the following.
    
    Consider a ${\cal C}$-decoration on $\Gamma$. We pick a "starting" 3-cell corresponding to the vertex decoration $\mathtt{U}_1=\mathtt{U}_{[0]}$. For each edge $e$, define a linear map which sends a 2-morphism $\eta: F_{f_3}\circ (F_{f_2}\circ F_{f_1})\Rightarrow 1_{\mathtt{U}_1}$ to a ${\cal C}$-decorated 2-simplex 
    \begin{equation*}
        \eta_{[012]}=(e_{F_{f_3}}\circ (F_{f_2}\circ F_{f_1}))\ast(F_{f_3}^\dagger\circ \eta_\Delta)
    \end{equation*}
     given by \eqref{Deltastates}. By dagger adjunction \eqref{unitaryadjoint}, this define bijections which give
    \begin{align}
        {\cal H}_{e^+}(\gamma)&=\operatorname{2Hom}(F_{3}\circ (F_{2}\circ F_{1}),1_{\mathtt{U}_1})\to \operatorname{2Hom}(F_{[12]}\circ F_{[01]},F_{[02]}), \nonumber\\
         {\cal H}_{e^-}(\gamma) &=\operatorname{2Hom}((F_{2}\circ F_{1})\circ F_{3},1_{\mathtt{U}_1})\to  \operatorname{2Hom}(F_{[02]},F_{[12]}\circ F_{[01]})\label{dualmap}
    \end{align}
    for the two different orientations on the edge $e$.

    Now consider the vertex space $\mathcal{H}_v$. As can be seen in Fig. \ref{fig:lwhilbspace}, the constituent edge Hilbert spaces (with specified surface ordering) are given by 
    \begin{align*}
        {\cal H}_{e_1^+}(\gamma) = \operatorname{2Hom}(F_3\circ (F_2\circ F_{1}),1_{\mathtt{U}_1}),\qquad {\cal H}_{e_4^-}(\gamma)=\operatorname{2Hom}((F_5\circ F_6)\circ F_3,1_{\mathtt{U}_3}),\\
        {\cal H}_{e_2^+}(\gamma) = \operatorname{2Hom}(F_2\circ (F_5\circ F_4),1_{\mathtt{U}_2}),\qquad {\cal H}_{e_3^-}(\gamma) = \operatorname{2Hom}((F_4\circ F_1)\circ F_6,1_{\mathtt{U}_4}).
    \end{align*}
    The above maps \eqref{dualmap} then induce linear isomorphisms 
    \begin{align*}
        {\cal H}_{e_1^+}(\gamma)\otimes{\cal H}_{e_4^-}(\gamma)\xrightarrow{\sim} \operatorname{2Hom}_{\cal C}(F_{[02]},F_{[32]}\circ F_{[03]} )\otimes  \operatorname{2Hom}_{\cal C}(F_{[12]}\circ F_{[01]},F_{[02]}) \\

    \end{align*}

<!-- proposition-6: body_md -->
\label{Fmove}
    Let $\mathtt{U},\mathtt{V},\mathtt{X},\mathtt{W}\in{\cal C}$ and $F:\mathtt{U}\to \mathtt{V},\, G:\mathtt{V}\to \mathtt{W} $ and $G':\mathtt{U}\to \mathtt{X},\,F': \mathtt{X}\to \mathtt{W}$. There are linear isomorphisms
$$\bigoplus_i \operatorname{2Hom}_{\cal C}(G\circ F,F_i)\otimes\operatorname{2Hom}_{\cal C}(F_i,F'\circ G') \cong \bigoplus_j \operatorname{2Hom}_{\cal C}(F,G_j\circ G')\otimes\operatorname{2Hom}_{\cal C}(G\circ G_j,F')\,,$$
where $F_i\in \operatorname{Hom}_{\cal C}(\mathtt{U}, \mathtt{W}),\, G_j\in \operatorname{Hom}_{\cal C}(\mathtt{X},\mathtt{V})$ are simple. This is the \textbf{F-move}, and is graphically represented by Fig. \ref{fig:Fmove}.

<!-- proof-11: body_md -->
By local semisimplicity, there is a natural isomorphisms
\begin{align*}
\operatorname{2Hom}_{\cal C}(G\circ F,F'\circ G')&\cong \bigoplus_i \operatorname{2Hom}_{\cal C}(G\circ F,F_i)\otimes\operatorname{2Hom}_{\cal C}(F_i,F'\circ G'),\\
\operatorname{2Hom}_{\cal C}(F\circ G'^\dagger,  G^\dagger\circ F')&\cong \bigoplus_j \operatorname{2Hom}_{\cal C}(F\circ G'^\dagger,G_j)\otimes\operatorname{2Hom}_{\cal C}(G_j,G^\dagger\circ F')
\end{align*}
 for simple 1-morphisms $F_i\in \operatorname{Hom}_{\cal C}(\mathtt{U}, \mathtt{W}),\, G_j\in \operatorname{Hom}_{\cal C}(\mathtt{X},\mathtt{V})$. Together with \eqref{conjugation}, we then have
\begin{align*}

    &\cong  \bigoplus_j \operatorname{2Hom}_{\cal C}(F,G_j\circ G')\otimes\operatorname{2Hom}_{\cal C}(G \circ G_j,F')
\end{align*}
as desired.
