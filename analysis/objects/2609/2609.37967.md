<!-- lemma-1: body_md -->
\label{rem_2}
	Let $(Q, \fn)$ be a local Cohen-Macaulay ring, and let $J \subseteq Q$ be an ideal. Then for any $f \in \fn$, we have
$$\operatorname{ht}({J}+(f)) \leq \operatorname{ht}({J}) + 1.$$
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $Q$ is a local Cohen-Macaulay ring, it follows by the dimension equality that
	$$\operatorname{ht}(J) + \dim(Q/J) =\operatorname{ht}(J + (f)) + \dim(Q/(J+(f))).$$
Thus, it suffices to show that $
	\dim(Q/J) \le \dim(Q/(J + (f))) + 1.$ But this inequality clearly holds as $f \in \fn$; see  \cite[Appendix~A.5]{BH}.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lemma_serre_gen_equiv} Assume $Q$ is Cohen-Macaulay and let $J$ be an ideal. Set ${I} = {J} + (f)$. Then the following are equivalent: 
	\begin{enumerate}[label=\normalfont(\alph*)]
		\item  \label{lemma_serre_gen_equiv_a}  The $Q$-module $Q/J$ is a Serre lift of the $R$-module $Q/I$. 
		\item  \label{lemma_serre_gen_equiv_b}  $\operatorname{ht}({I}) = \operatorname{ht}({J}) + 1$. 
		\item \label{lemma_serre_gen_equiv_b'}  $grade({I}) = grade({J}) + 1$. 
		\item  \label{lemma_serre_gen_equiv_c}  $\min_0({I}) \cap \min_0({J}) = \emptyset$.
		\item  \label{lemma_serre_gen_equiv_d} $f \notin \bigcup_{\fp\in \min_0({J})} \fp$. 
	\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since $Q$ is local Cohen-Macaulay, we have \begin{align*}
		\operatorname{ht}({I}) - \operatorname{ht}({J}) & =\left( \dim(Q) -\dim(Q/{I})\right) - \left(\dim(Q) - \dim(Q/{J}) \right)\\ & =  1+\left(\dim(R) -\dim(Q/{I})\right) - \left(\dim(Q) - \dim(Q/{J}) \right).
	\end{align*} This proves $\ref{lemma_serre_gen_equiv_a} \iff \ref{lemma_serre_gen_equiv_b}$. The equivalence $\ref{lemma_serre_gen_equiv_b} \iff \ref{lemma_serre_gen_equiv_b'}$ follows from \cite[Corollary~2.1.4]{BH}.  
	
	If $\fq \in \min_0({I}) \cap \min_0({J}),$ then clearly $\operatorname{ht}({I}) = \operatorname{ht}(\fq) = \operatorname{ht}({J}),$ a contradiction.  So $\ref{lemma_serre_gen_equiv_b} \implies \ref{lemma_serre_gen_equiv_c}$. To see $\ref{lemma_serre_gen_equiv_c} \implies \ref{lemma_serre_gen_equiv_b},$ let $\fp \in \min_0({I})$. Then ${J} \subseteq \fp$ and $\fp \notin \min_0({J})$. Thus, $\operatorname{ht}({I}) = \operatorname{ht}(\fp) > \operatorname{ht}({J})$. Since ${I} = {J} + (f),$ height can increase at most by $1$ by Lemma \ref{rem_2}, so we get $\ref{lemma_serre_gen_equiv_b}$. 
	
	It is clear that $\ref{lemma_serre_gen_equiv_d} \implies \ref{lemma_serre_gen_equiv_c}$. To complete the proof, we show that $\ref{lemma_serre_gen_equiv_b} \implies \ref{lemma_serre_gen_equiv_d}$. If $f \in \fp$ for some $\fp \in \min_0({J}),$ then ${I} \subseteq \fp$. Thus, $\operatorname{ht}({I}) = \operatorname{ht}({J}),$ which contradicts $\ref{lemma_serre_gen_equiv_b}$.
<!-- end proof-2 -->
<!-- corollary-1: body_md -->
\label{lemma_char_serre} 
	Assume that $Q$ is Cohen-Macaulay. Let $I$ be an ideal and assume that $f$ is a minimal generator of ${I}$. Let $h$ be the height of $I$. Then the following are equivalent
	\begin{enumerate}[label=(\alph*)]
		\item The cyclic $R$-module $Q/{I}$ is Serre liftable.
		\item There is a prime ideal $\fq$ of height $h-1$ such that ${I} \subseteq \fq + (f)$. 
	\end{enumerate} If any of these conditions hold, then there is an ideal $J \subset \fq$ of height $\operatorname{ht}(I) - 1$ such that $Q/J$ is a Serre lift of $Q/I$ to $Q$.
<!-- end corollary-1 -->
<!-- proof-3: body_md -->
Let $Q/J$ be a Serre lift of $Q/I$ and let $\fq \in min_0({J})$. Then by Lemma~\ref{lemma_serre_gen_equiv}, $\operatorname{ht}(\fq) =  \operatorname{ht}(J) = h-1$ and ${I} = J+(f) \subseteq \fq + (f)$.  
	
	Conversely, suppose $\fq$ is as given. Let $(f, f_2,\cdots, f_m)$ be a minimal generating set of ${I}$. Let $g_i \in \fq$ and $a_i \in Q$ be such that $f_i  = g_i + a_if$ for $i = 2,\cdots, m$. Set ${J} = (f_2 - a_2f, \cdots, f_m - a_mf)$. Then ${J} \subseteq \fq$ and ${J} + (f) = {I}$. By Lemma \ref{rem_2}, $\operatorname{ht}({J}) \geq h-1$ but since ${J}$ is contained in a height $h-1$ prime ideal, we must have $\operatorname{ht}({J}) = h-1$. Thus, by Lemma~\ref{lemma_serre_gen_equiv}, $Q/{J}$ is a Serre lift of $Q/{I}$. The last assertion is clear.
<!-- end proof-3 -->
<!-- corollary-2: body_md -->
\label{cor_small_height}
	Let $Q$ be a local Cohen-Macaulay domain and  $R = Q/(f)$ where $f$ is a nonzerodivisor. Let $I$ be an ideal of $Q$ such that $f$ is a minimal generator of ${I}$.
	\begin{enumerate}[label=\normalfont(\alph*)]
		\item \label{item_cor_sm_a}   If $\operatorname{ht}(I) = 1,$ then the cyclic $R$-module $Q/I$ is Serre liftable to $Q$ if and only if $I$ is the principal ideal $(f)$. 
		\item \label{item_cor_sm_b}    If $\operatorname{ht}(I) = 2,$ $I$ is prime and $Q$ is, in addition, a unique factorization domain, then $Q/I$ is Serre liftable to $Q$ if and only if $I$ is a complete intersection ideal. 
	\end{enumerate}
<!-- end corollary-2 -->
<!-- proof-4: body_md -->
For \ref{item_cor_sm_a}, Corollary \ref{lemma_char_serre} and the domain hypothesis give \(J=0\), hence \(I=(f)\).
	
	To see \ref{item_cor_sm_b}, suppose $I \subseteq \fq + (f)$. Since  $\operatorname{ht}(\fq+(f)) = \operatorname{ht}(I),$ and $I$ is prime, we must have $I = \fq+(f)$. Since $\fq$ is a height $1$ prime in a unique factorization domain, it is principal, thus $I$ is a complete intersection ideal. 
	
	The converse implication holds more generally in both cases, as $Q/I$ clearly lifts when $I$ is a complete intersection ideal.

<!-- corollary-3: body_md -->

<!-- lemma-3: body_md -->
\label{lemma_serre_inequality} 
Assume the setup above. Then $M_1$ and $M_2$ satisfy the Serre inequality, i.e., $$\dim(M_1) + \dim(M_2) \leq \dim(R) + \dim(M_1 \otimes_R M_2),$$ with equality only if $\dim(N_1 \otimes N_2) - \dim(M_1 \otimes M_2) = t$.
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
By Serre liftability, we know that $\dim(M_i)  = \dim(N_i) - t$ for $i = 1,2$.  Using Serre's inequality \cite{Serre}, we have that $\dim(N_1) + \dim(N_2) \leq \dim(Q) + \dim(N_1 \otimes N_2)$.  Combining these facts, along with the inequality \eqref{eqn_ineq}, we obtain $$\dim(M_1) + \dim(M_2) \leq \dim(R) +  \dim(M_1 \otimes M_2).$$ It is clear that equality implies that $\dim(N_1 \otimes N_2) - \dim(M_1 \otimes M_2) = t$.
<!-- end proof-6 -->
<!-- remark-3: body_md -->
Let $Q$ be a regular local ring and let $R$ be a locally complete intersection $Q/({\bf f})$ and assume  that $R/\overline{I}_1$ and $R/\overline{I}_2$ are Serre liftable to $Q$. It follows by Lemma~\ref{lemma_serre_inequality} that $\dim(R/\overline{I}_1) + \dim(R/\overline{I}_2) \leq \dim(R) + \dim(R/(\overline{I}_1+\overline{I}_2))$. Since $R$ is local CM,  we get $$\operatorname{ht}(\overline{I}_1 + \overline{I}_2) \leq \operatorname{ht}(\overline{I}_1) + \operatorname{ht}(\overline{I}_2).$$

<!-- lemma-4: body_md -->
\label{lemma_serre_tensor_prod} 
Let $Q$ be a regular local ring and let ${\bf f} = \{f_1,\cdots, f_t\} \subset Q$ be a regular sequence. Set $R = Q/({\bf f})$. Let $M_1$ and $M_2$ be two $R$-modules that are Serre liftable to $Q$-modules $N_1$ and $N_2$. Then $N_1 \otimes N_2$  is a Serre lift of $M_1 \otimes M_2$ to $Q$ if any of the following holds. 
	\begin{enumerate}[label=\normalfont{(\alph*)}]
		\item \label{item_1a} Either $M_1$ or $M_2$ is supported everywhere on $\operatorname{Spec}(R)$. 
		\item \label{item_2a} $\ell_R(M_1 \otimes M_2) < \infty,$ and $\dim(M_1) + \dim(M_2) = \dim(R)$. 
		\item \label{item_3a} $M_1 \otimes M_2$ is {CM} and $\operatorname{Tor}_i^R(M_1,M_2) = 0$ for $i = 1,\cdots, t+1$. 
	\end{enumerate}  Furthermore, if \ref{item_3a} holds, then we also conclude that $M_1$ and $M_2$ are CM.
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Since the map $Q \rar R$ is a codimension $t$ deformation, to show Serre liftability of $M_1 \otimes_R M_2,$ it is sufficient to show that $\dim(N_1\otimes_Q N_2) - \dim(M_1 \otimes_R M_2) = t$.
	
	Without loss of generality, we assume that $Supp(M_1) = Spec(R),$ which implies that $\dim(M_1) = \dim(R)$.  Thus,  by Lemma~\ref{lemma_serre_inequality}, $\dim(M_2) \leq \dim(M_1 \otimes_R M_2)$. This is clearly an equality, hence, all the above inequalities are equalities. In particular, $\dim(N_1\otimes_Q N_2) - \dim(M_1 \otimes_R M_2) = t$. This proves \ref{item_1a}. 
	
	The proof of \ref{item_2a} is similar - the hypothesis implies that the inequality in Lemma~\ref{lemma_serre_inequality} is an equality. Thus,  we must have $ \dim(N_1 \otimes N_2)  = t$.  

	
	
	To see \ref{item_3a}, note that it follows by \cite{Murthy} that $M_1$ and $M_2$ are Tor-independent. Thus, by Auslander's depth formula \cite[Theorem~2.5]{HW} \begin{align*}
		\operatorname{depth}(M_1 \otimes M_2) &= \operatorname{depth}(M_1) + \operatorname{depth}(M_2) - \operatorname{depth}(R) \leq \dim(M_1) + \dim(M_2) - \dim(R) \\ & = \dim(N_1) + \dim(N_2) -\dim(Q) - t \leq \dim(N_1 \otimes N_2) - t \\ & \leq \dim(M_1 \otimes M_2).

		\item \label{item_b}  The induced map $\beta: Q/{I} \rar {I}/{I}^2$ is split.

E=(\widetilde I_1+\widetilde I_2)/

		0 \rar \operatorname{Hom}_R(I_1 + I_2,E ) & \xrightarrow{h'} \operatorname{Hom}_R(I_1 \oplus I_2,E )  \xrightarrow{h} \operatorname{Hom}_R(I_1 \cap I_2,E ).

	\end{align*}

\label{lemma_Auslander_transpose_lift} \label{torsionfree}
	The following are equivalent:
	\begin{enumerate}[label=\normalfont(\alph*)]
		\item \label{item_1} $f$ is regular on $\operatorname{Tr}_Q(N)$.
		\item \label{item_2} $\operatorname{Tr}_Q(N)$ is a lift of $\operatorname{Tr}_R(M)$.
		\item \label{item_3} $\operatorname{Hom}_Q(N,Q)$ is a lift of $\operatorname{Hom}_R(M,R)$ via the natural map 
		$$\theta: \operatorname{Hom}_Q(N,Q) \otimes_Q R \rar \operatorname{Hom}_R(M,R)$$ 
		\item \label{item_4} Either $\operatorname{Ext}^1_Q(N,Q) = 0$ or $f$ is regular on $\operatorname{Ext}^1_Q(N,Q)$.
	\end{enumerate}

Choose a minimal free presentation $F_1\xrightarrow d F_0\to N\to0$. Assume that $f$ is regular on $\operatorname{Tr}_Q(N)$. This proves	\ref{item_1} $\implies$ \ref{item_2}. 
	
The implication	\ref{item_2} $\implies$ \ref{item_3} follows by applying $\otimes_Q R$ to the following exact sequence
\begin{align} \label{seq_1} 
	0\rar N^* \rar F_0^* \xrightarrow{d^*}F_1^* \rar \operatorname{Tr}_Q(N) \rar 0.
\end{align}  
Set $K=\operatorname{Im}(d^*)$. Since $K\subseteq F_1^*$, the element $f$ is regular on $K$; by \ref{item_2}, it is also regular on $\operatorname{Tr}_Q(N)$. Breaking the exact sequence \ref{seq_1} into two short exact sequences with intermediate module $K$, and tensoring one of them with $R$ yields the following exact sequence
$$0 \rar N^*\otimes_QR \rar F_0^*\otimes_QR \xrightarrow{d^*\otimes R}F_1^*\otimes_QR.$$ 
                                                                                   
The kernel of the last map in this sequence is $Hom_R(M,R)$. The induced identification $$Hom_Q(N,Q) \otimes_Q R \rar Hom_R(M,R)$$ 
is precisely the canonical map $\theta$, hence it is an isomorphism. Since $f$ is regular on $Hom_Q(N,Q)$, therefore \ref{item_3} follows.
	
Assuming \ref{item_3}, we consider the long exact sequence $$0 \rar \operatorname{Hom}_Q(N,Q) \xrightarrow{f} \operatorname{Hom}_Q(N,Q) \rar \operatorname{Hom}_Q(N,R) \xrightarrow{\partial} \operatorname{Ext}^1_Q(N,Q) \xrightarrow{f}  \operatorname{Ext}^1_Q(N,Q).$$ Since $\operatorname{Hom}_Q(N,R) \cong \operatorname{Hom}_R(M,R),$ the hypothesis implies that the boundary map $\partial$ is zero, thus proving \ref{item_4}.

Thus, $\operatorname{ht}(I_{\alpha}) - \operatorname{ht}(J_{\alpha}) = 1.$ This shows, by the characterization obtained in Lemma~\ref{lemma_serre_gen_equiv}, that the $Q$-module $Q/J_{\alpha}$ is a Serre lift of the cyclic $R$-module $Q/I_{\alpha}$. 
	
	It can be verified that $m^4 \subset I_{\alpha} \subset m^2$. We note that $Q/I_{\alpha}$ is Gorenstein, and the socle $(\fm^3 + I_{\alpha})/I_{\alpha}$ is generated by the image of $g = X_1X_2X_5$; see \cite{GP}. It follows from this that $g \notin I_{\alpha}$. By considering $\alpha$ as an independent variable, a Macaulay2 computation shows the following inclusion of ideals \begin{align} \label{eqn_containment} 
		(\alpha^2 - 1) \subseteq (I_{\alpha}^2: fg)
	\end{align}Thus, we conclude that when $\alpha \neq \pm 1$, there is a containment $g \in  (I_{\alpha}^2: f) \setminus I_{\alpha}$. By Dao's criterion for weak lifting \ref{sec_WL}\ref{Dao_criterion}, it follows that $Q/I_{\alpha}$ is not weakly liftable.
		
Specializing to $\alpha = 2$, a computation in Macaulay2 verifies that $Q/I_2$ is a perfect $R$-module. Direct verifications of the two computational steps used here are given in the following remark.

			X_1u^2h	\equiv

\label{ex_1} 	Let $Q= k[[X_1,X_2,X_3]]/(X_1X_3 ,X_2X_3)$ and let $x_i$ be the image of $X_i$ in $Q$ for $i=1,2,3$. Consider the $Q$-module $N= Q/(x_2 ^2)$ and set $\fp = (x_2, x_3)$. On the other hand, $\fp$ is a nonsingular point of the variety defined by $Q$ so $ \operatorname{depth}(Q_{\fp}) = \dim(Q_{\fp}) = 1$. So $N$ does not satisfy $\wtil{S}_1$. 
	
We set $f = x_1 + x_3$ and $R = Q/(f).$ It is clear that $f$ is not in the associated primes of $Q$ or $N$. It is therefore a nonzerodivisor on $Q \oplus N$. Let $M = N/fN$ be the quotient $R$-module.  Since $\operatorname{Tr}_Q(N)\cong N$, the equivalent conditions of Lemma \ref{lemma_Auslander_transpose_lift} are also satisfied. 

The computations in the Macaulay2 code included in the supplementary material verifies that 
\[
\operatorname{depth}R=\operatorname{depth}M
=\operatorname{depth}M^*=0,
\qquad
\operatorname{Ext}_R^1(M,R)\neq0,
\quad
\operatorname{Ext}_R^1(M^*,R)\neq0.
\]
The Auslander–Bridger formula therefore implies that both \(M\) and \(M^*\) have infinite Gorenstein dimension.
