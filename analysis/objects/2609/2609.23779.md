<!-- proof-1: body_md -->
By the definition~\eqref{eq:main}, we have 
	\begin{equation}\label{eq:F-recur}
		F_{r,n}(q)
		=F_{r,n-1}(q)\sum_{\ell=0}^{r-1}q^{\ell n}.
	\end{equation}
	Comparing the coefficients of $q^j$ on both sides of~\eqref{eq:F-recur} yields
	\begin{equation*}
		A_{r,n}(j)
		=\sum_{\ell=0}^{r-1}A_{r,n-1}(j-\ell n).
	\end{equation*}
	Hence,
	\begin{align}\label{fd:recurrence}
		B_{r,n}(j)
		&=A_{r,n}(j)-A_{r,n}(j-1)\notag\\
		&=\sum_{\ell=0}^{r-1}
		\bigl(A_{r,n-1}(j-\ell n)
		-A_{r,n-1}(j-\ell n-1)\bigr)\notag\\
		&=\sum_{\ell=0}^{r-1}B_{r,n-1}(j-\ell n).
	\end{align}
	
	For $n=1,$ $F_{r,1}(q)=1+q+\cdots+q^{r-1}$.
	So $B_{r,n}(j)=1$ for $j=0$ and $B_{r,n}(j)=0$ for $1\le j\le r-1$.
	For $n\ge2$, suppose that $F_{r,n-1}(q)$ is unimodal.
	By Proposition~\ref{fd:window},
	$$
	B_{r,n-1}(j)\ge0
	\quad\text{for  }j\le
	\left\lfloor\mu_{r,n-1}\right\rfloor.
	$$
	Then $B_{r,n-1}(j-\ell n)\ge 0$ since $j-\ell n\le j$. 
	It follows from~\eqref{fd:recurrence}
	that $B_{r,n}(j)\ge0$ for
	$1\le j\le\lfloor\mu_{r,n-1}\rfloor$.
	This completes the proof.

<!-- lemma-1: body_md -->
\label{fd:initial}
	Let $r\ge2$ and $n\ge3$ be integers. Then we have $B_{r,n}(j)\ge0$ for $0\le j< 2r.$
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
By \eqref{GF-B}, we have
	$B_{r,n}(j)=[q^j](1-q)F_{r,n}(q).$
	Since $n\ge 3$, we have
	$$
	\begin{aligned}
		(1-q)F_{r,n}(q) =\frac{\prod_{k=1}^{n}(1-q^{rk})} {\prod_{k=2}^{n}(1-q^k)} =\frac{1-q^r}{(1-q^2)(1-q^3)}
		\left(\prod_{k=4}^{n}\frac{1}{1-q^k}\right) \left(\prod_{k=2}^{n}(1-q^{rk})\right),
	\end{aligned}
	$$
	where an empty product is interpreted as 1.
	
	Write $r=2u+3v$, where $u\ge 0$ and $v\in\{0,1\}$;
	take $v=0$ when $r$ is even and $v=1$  when $r$ is odd.
	Since $v\in\{0,1\}$,
	we have $1-q^{3v}=v(1-q^3)$ and
	\begin{align}\label{fd:num-decom}
		1-q^r
		=1-q^{2u+3v}
		=(1-q^{2u})+q^{2u}(1-q^{3v})
		=(1-q^2)\sum_{\ell=0}^{u-1}q^{2\ell}
		+vq^{2u}(1-q^3).
	\end{align}
	Dividing both sides of~\eqref{fd:num-decom}
	by $(1-q^2)(1-q^3)$, we obtain
	\begin{equation}\label{eq:non-expansion}
		\frac{1-q^r}{(1-q^2)(1-q^3)}
		=\frac{\displaystyle\sum_{\ell=0}^{u-1}q^{2\ell}}{1-q^3}
		+\frac{vq^{2u}}{1-q^2},
	\end{equation}
	where an empty sum is interpreted as 0.
	All coefficients on the right-hand side of~\eqref{eq:non-expansion}
	are nonnegative. Consequently, all coefficients of
	$$
	\frac{1-q^r}{(1-q^2)(1-q^3)}
	\prod_{k=4}^{n}\frac{1}{1-q^k}
	$$
	are nonnegative.
	
	Finally, every nonconstant term in
	$\prod_{k=2}^{n}(1-q^{rk})$
	has degree at least $2r$, and its constant term is $1$.  Hence, for $0\le j<2r$, we have
	$$
	B_{r,n}(j) = [q^j]\left( \frac{1-q^r}{(1-q^2)(1-q^3)} \prod_{k=4}^{n}\frac{1}{1-q^k} \right) \ge 0.
	$$
	This completes the proof.

<!-- proposition-3: body_md -->
\label{sc:three}
	For every integer $r\ge 2$, the polynomial $F_{r,3}(q)$ is
	unimodal.
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
For $r\ge2$, since $D_{r,3}=6r-6$,
	Proposition~\ref{fd:window} shows that $F_{r,3}(q)$ is unimodal
	if and only if $B_{r,3}(j)\ge0$ for $1\le j\le3r-3$.
	The case $j<2r$ follows from Lemma~\ref{fd:initial}.
	It remains to prove $B_{r,3}(j)\ge0$ for $2r\le j\le3r-3$.
	
	For $m\ge0$, let
	\begin{equation}\label{c3m}
		c(m) =[q^m]\frac{1}{(1-q^2)(1-q^3)} =\#\{(x,y)\in\mathbb Z_{\ge0}^2:2x+3y=m\}.
	\end{equation}
	\begin{claim}\label{cla:n=3}
		We have
		\begin{equation}\label{eq:cm}
			c(m)=\left\lfloor\frac m2\right\rfloor -\left\lfloor\frac{m-1}{3}\right\rfloor.
		\end{equation}
	\end{claim}
	\noindent\textbf{Proof of Claim~\ref{cla:n=3}.}
	Let $m=2x+3y$ and $k=x+y$. Then $x=3k-m$ and $y=m-2k$.
	Since $(x,y)\in\mathbb Z_{\ge0}^2$,
	we have $m/3\le k\le m/2$ and $k\in \mathbb Z_{\ge0}.$
	Every $k$ corresponds to a pair $(x,y)$. Thus we prove~\eqref{eq:cm}.
	
	Fix $2r\le j\le3r-3$. Then
	\begin{align}\label{eq:Br3}
		B_{r,3}(j)&=[q^j](1-q)F_{r,3}(q)\notag\\
		&=[q^j]\frac{(1-q^r)(1-q^{2r})(1-q^{3r})} {(1-q^2)(1-q^3)}\notag\\
		&=[q^j]\frac{1-q^r-q^{2r}+q^{4r}+q^{5r}-q^{6r}} {(1-q^2)(1-q^3)}.
	\end{align}
	Since $j<3r$, it follows from~\eqref{eq:cm}
	and~\eqref{eq:Br3} that
	\begin{align}\label{eq:Br3-floor}
		B_{r,3}(j) =&c(j)-c(j-r)-c(j-2r)\notag\\
		=&\left(\left\lfloor\frac j2\right\rfloor
		-\left\lfloor\frac{j-1}{3}\right\rfloor\right)-\left(\left\lfloor\frac{j-r}{2}\right\rfloor
		-\left\lfloor\frac{j-r-1}{3}\right\rfloor\right)\notag\\
        &-\left(\left\lfloor\frac{j-2r}{2}\right\rfloor
		-\left\lfloor\frac{j-2r-1}{3}\right\rfloor\right)\notag\\
		=&r	-\left\lfloor\frac{j-1}{3}\right\rfloor-\left\lfloor\frac{j-r}{2}\right\rfloor
		+\left\lfloor\frac{j-r-1}{3}\right\rfloor +\left\lfloor\frac{j-2r-1}{3}\right\rfloor,
	\end{align}
	where we use $\left\lfloor j/2\right\rfloor-\left\lfloor{(j-2r)}/{2}\right\rfloor=r$
	in the last equality.
	
	Let $t \equiv r \pmod{3}$ and \(t\in\{0,1,2\}\).
	Then \(r=3s+t\).
	Set
	\(h=3r-j\). Since \(j\le 3r-3\), we have \(h\ge 3\).
	Note that 
	\begin{equation}\label{a-x}
		\left\lfloor a-x \right\rfloor
		= a-\left\lceil x \right\rceil, \qquad \textrm{for}\quad a\in\mathbb{Z}.
	\end{equation}
	Substituting \(r=3s+t\) and \(j=3r-h\) into~\eqref{eq:Br3-floor}, and using~\eqref{a-x},
	we obtain
	\begin{align}\label{br3-j}
		B_{r,3}(j)
		=&r-\left\lfloor\frac{3r-h-1}{3}\right\rfloor
		-\left\lfloor\frac{2r-h}{2}\right\rfloor
		+\left\lfloor\frac{2r-h-1}{3}\right\rfloor
		+\left\lfloor\frac{r-h-1}{3}\right\rfloor\nonumber\\

\label{sc:closure}
	Suppose that $P(q)$ is a polynomial of odd degree $2d+1$.
	If $P(q)$ is symmetric and unimodal with all coefficients
	nonnegative, then so is
	$$
	Q(q)=(1+q)(1+q^3)P(q^2).
	$$

<!-- proof-4: body_md -->
Clearly, all coefficients of $Q(q)$ are nonnegative. 
	The symmetry of $P(q)$ gives $q^{4d+6}Q(q^{-1})=Q(q)$, 
	so the coefficients of $Q(q)$ are symmetric.
	Write
	$$
	P(q)=\sum_{k=0}^{2d+1}a_kq^k, \qquad Q(q)=\sum_{k=0}^{4d+6}b_k q^k,
	$$
	and set $a_k=0$ outside $0\le k\le2d+1$.
	Since $P(q)$ is symmetric and unimodal with nonnegative coefficients, we have 
	$$
	0\le a_0\le a_1\le\cdots\le a_d=a_{d+1}.
	$$
	
	
	Note that
	$Q(q)=(1+q+q^3+q^4)P(q^2).$
	Taking the coefficients of $q^{2k}$ and $q^{2k+1}$ on both sides yields
	$b_{2k}=a_k+a_{k-2}$ and $b_{2k+1}=a_k+a_{k-1}$.
	Thus
	\begin{align*}
		b_{2k}-b_{2k-1} &=a_k-a_{k-1}\ge0, \quad1\le k\le d+1,\\
		b_{2k+1}-b_{2k} &=a_{k-1}-a_{k-2}\ge0, \quad0\le k\le d+1.
	\end{align*}
	This gives $0\le b_0\le b_1\le\cdots\le b_{2d+3}.$ 
	By symmetry of $Q(q)$, we obtain the unimodality of $Q(q).$
	This completes the proof.

<!-- proof-7: body_md -->
For every integer $m\in \mathbb{Z}$, let 	$0\le s <M_n$ and  $m\equiv s\pmod{M_n}$.  Define
	\begin{equation}\label{low:extension-definition}
		\widetilde c_n(m)
		=R_s\!\left(m+\frac{n(n+1)-2}{4}\right),
	\end{equation}
	where $\widetilde c_n(m)$ is a polynomial in $m$ on $s+M_n\,\mathbb{Z}$ with degree $n-2$.
	Thus,  for $m\ge 0$, we have 
	\begin{align}\label{low:extension-agreement}
		\widetilde c_n(m)=c_n(m).
	\end{align}
	Let
	$$
	d=\frac{n(n+1)-2}{2},\qquad \prod_{k=2}^{n}(1-q^k)=\sum_{j=0}^{d}\gamma_jq^j, \qquad
	\gamma_{d}=(-1)^{n-1}.
	$$
	By~\eqref{def:c}, we have 
	\begin{equation}\label{low:generating-identity}
		\prod_{k=2}^{n}(1-q^k)\sum_{m\ge0}c_n(m)q^m
		=\prod_{k=2}^{n}(1-q^k)\prod_{k=2}^{n}\frac{1}{1-q^k}
		=1.
	\end{equation}
	Comparing the coefficients of $q^m$ on both sides
	of~\eqref{low:generating-identity} for $m\ge1$, we obtain
	\begin{align}\label{low:coefficient-recurrence}
		\sum_{j=0}^{d}\gamma_j c_n(m-j)=0.
	\end{align}
	For $m\in \mathbb{Z}$, define
	$$
	H(m):=\sum_{j=0}^{d} \gamma_j \widetilde c_n(m-j).
	$$
	It follows immediately from~\eqref{low:extension-definition}
	that $H(m)$ is a polynomial in $m$ of degree at most $n-2$
	on each residue class $s+M_n\mathbb{Z}$. 
	For every $m\ge d,$ since $c_n(m-j)=\widetilde c_n(m-j)$, we have
	$$
	H(m)=\sum_{j=0}^{d}\gamma_j c_n(m-j)=0.
	$$
	For each fixed $s$ with $0\le s<M_n$, $H(s+M_nt)$
	is a polynomial in $t$.
	Since $H(m)=0$ for $m\ge d$, this polynomial vanishes
	whenever $s+M_nt\ge d$.
	It therefore has infinitely many integer zeros and is
	identically zero.
	Every integer $m$ can be written as $s+M_nt$ with
	$0\le s<M_n$ and $t\in\mathbb{Z}$.
	Hence, for every integer $m$, we have
	\begin{equation}\label{low:H-zero}
		H(m)=\sum_{j=0}^{d}\gamma_j\widetilde c_n(m-j)=0.
	\end{equation}
	By~\eqref{low:extension-agreement},
	\eqref{low:coefficient-recurrence}, and~\eqref{low:H-zero},
	we have
	\begin{align*}
		0=H(d-1)
		=\sum_{j=0}^{d}\gamma_j\widetilde c_n(d-1-j)
		=\sum_{j=0}^{d-1}\gamma_j c_n(d-1-j)
		+\gamma_d\widetilde c_n(-1)
		=\gamma_d\widetilde c_n(-1).
	\end{align*}
	Since $\gamma_d\ne0$, we obtain $\widetilde c_n(-1)=0$.
	Repeating this argument with $m=d-2,\ldots,1$ successively gives
	$$
	\widetilde c_n(-1)=\widetilde c_n(-2)
	=\cdots=\widetilde c_n(1-d)=0.
	$$
	By~\eqref{low:extension-definition} and
	$d=(n(n+1)-2)/2$, we obtain
	$$
	R_s\!\left(m+\frac{n(n+1)-2}{4}\right)=0
	$$
	for every integer $m$ with $-(n(n+1)-2)/2<m<0$,
	where $0\le s<M_n$ and $m\equiv s\pmod{M_n}$.
	This completes the proof.

	A_{r,n}(j)=\#\left\{(x_1,\ldots,x_n)\in\{0,\ldots,r-1\}^n: \sum_{k=1}^{n}kx_k=j\right\}.

	each uniformly distributed on $\{0,\ldots,r-1\}$,

	\Pr(Y=j)=\frac{A_{r,n}(j)}{r^n}.

	=\frac{r-1}{2}

		\operatorname{Var}(X_k)
		=&\mathbb E[X_k^2]-(\mathbb E[X_k])^2\\
		=&\frac{1}{r}\sum_{\ell=0}^{r-1}\ell^2

		=&\frac{(r-1)(2r-1)}{6}-\frac{(r-1)^2}{4}\\
		=&\frac{r^2-1}{12}.

\label{lem:sine-square}

	$$

<!-- proof-14: body_md -->
By the trigonometric summation identity
	\cite[Eq.~(2.11)]{DJ23},
	\begin{equation}\label{DJ}
		\sum_{k=1}^{n}\sin^2(k\theta/2) =\frac n2+\frac14 -\frac{\sin((2n+1)\theta/2)}{4\sin(\theta/2)}.
	\end{equation}
	For $\pi/n\le\theta\le2\pi/n$,  we have
	$\sin(n\theta/2)\ge0$ and
	$\cos((n+1)\theta/2)\le0$, and
	\begin{align*}
		\sum_{k=1}^{n}\sin^2(k\theta/2)&=\frac n2+ \frac{\sin(\theta/2)-\sin((2n+1)\theta/2)}{4\sin(\theta/2)}\\
		&=\frac n2 -\frac{\sin((2n+1)\theta/2)-\sin(\theta/2)}{4\sin(\theta/2)}\\
		&=\frac n2- \frac{\sin(n\theta/2)\cos((n+1)\theta/2)} {2\sin(\theta/2)} \ge\frac n2>\frac{2n}{5}.
	\end{align*}
	For $2\pi/n\le\theta\le\pi$,
	we have $\sin (\theta/2)\ge \sin(\pi/n)$.
	Taking $a=\pi/3$ in~\eqref{eq:sine-chord},
	we have
	$$
	\sin x\ge \frac{3\sqrt3}{2\pi}x \quad \textrm{and} \quad 0\le x\le \pi/3.
	$$
	Thus,
	$$
	\sin(\theta/2)\ge\sin(\pi/n) \ge\frac{3\sqrt3}{2n}>\frac{5}{2n}.
	$$
	Consequently, by~\eqref{DJ},
	$$
	\sum_{k=1}^{n}\sin^2(k\theta/2) \ge\frac n2+\frac14-\frac{1}{4\sin(\theta/2)}
	>\frac{n}{2}+\frac14-\frac{n}{10}>\frac{2n}{5}.
	$$
	This completes the proof.

Recall that by~\eqref{eq:unl},
	\begin{equation}\label{A1}
		\prod_{k=1}^{n}(1-q^k)=\sum_{\ell=0}^{2b}u_{n,\ell}q^\ell.
	\end{equation}
	By~\eqref{eq:Ly-sum},
	$$L(y)=\frac1{n!(n-2)!}
	\sum_{\ell\in\mathcal J_I}u_{n,\ell}(y-\ell)^{n-2}.$$
	
	We first prove that for every $x,$
	\begin{equation}\label{app:center-full-sum}
		\sum_{\ell=0}^{2b}u_{n,\ell}(x-\ell)^{n-2}=0.
	\end{equation}
	For a polynomial $f(x)$ and $1\le k\le n$,
	define the difference operator by
	\begin{align}\label{eq:diff-def}
		\nabla_k f(x)=f(x)-f(x-k).
	\end{align}
	If $\deg f(x)\ge 1$, then
	$\deg \nabla_k f(x)=\deg f(x)-1$.
	If $f(x)$ is constant, then $\nabla_k f(x)=0$.
	Thus, for $f(x)=x^{n-2}$, we have
	$$
	\nabla_1\cdots\nabla_n f(x)=0.
	$$
	It follows from~\eqref{eq:diff-def} that
	\begin{equation}\label{app:difference-expansion}
		\nabla_1\cdots\nabla_n f(x)
		=\sum_{S\subseteq\{1,\ldots,n\}}(-1)^{|S|}
		f\left(x-\sum_{k\in S}k\right).
	\end{equation}
	See~\cite[Proposition~1, p.~429]{SWMP17}.
	
	
	By definition, 
	\begin{align}\label{eq:Uq-S}
		\prod_{k=1}^{n}(1-q^k)
		=\sum_{S\subseteq\{1,\ldots,n\}}
		(-1)^{|S|}q^{\sum_{k\in S}k}.
	\end{align}
	Comparing the coefficients of $q^\ell$ in~\eqref{A1}
	and~\eqref{eq:Uq-S}, we obtain
	\begin{equation}\label{app:subset-coefficient}
		u_{n,\ell}
		=\sum_{\substack{S\subseteq\{1,\ldots,n\}\\
				\ell=\sum_{k\in S}k}}
		(-1)^{|S|}.
	\end{equation}
	By~\eqref{app:difference-expansion} and~\eqref{app:subset-coefficient}, 
	we get
	\begin{align*}
		0
		=\nabla_1\cdots\nabla_n f(x)
		=\sum_{\ell=0}^{2b}
		\left(
		\sum_{\substack{S\subseteq\{1,\ldots,n\}\\
				\sum_{k\in S}k=\ell}}
		(-1)^{|S|}
		\right)f(x-\ell)
		=\sum_{\ell=0}^{2b}u_{n,\ell}(x-\ell)^{n-2}.
	\end{align*}
	This proves~\eqref{app:center-full-sum}.
	Setting $x=b$ in~\eqref{app:center-full-sum} yields
	\begin{equation}\label{x=b}
		\sum_{\ell=0}^{2b}u_{n,\ell}(b-\ell)^{n-2}=0.
	\end{equation}
	
	
	Clearly
	\begin{align}\label{eq:U-sym}
		q^{2b} \prod_{k=1}^{n}(1-q^{-k})
		=(-1)^n\prod_{k=1}^{n}(1-q^k),
	\end{align}
	Comparing the coefficients of $q^\ell$ on both sides
	of~\eqref{eq:U-sym}, we obtain
	\begin{equation*}
		u_{n,2b-\ell}=(-1)^nu_{n,\ell}
	\end{equation*}
	for $0\le\ell\le 2b$.
	Thus, 
	\begin{align*}
		u_{n,2b-\ell}\bigl(\ell-b\bigr)^{n-2}
		=(-1)^nu_{n,\ell}(\ell-b)^{n-2}
		=u_{n,\ell}(b-\ell)^{n-2}.
	\end{align*}
	Hence, 
	\begin{align*}
		\sum_{\ell=0}^{2b}u_{n,\ell}(b-\ell)^{n-2}&= \sum_{\ell< b}
		u_{n,\ell}(b-\ell)^{n-2}+\sum_{\ell>b}
		u_{n,\ell}(b-\ell)^{n-2}\\
		&=\sum_{\ell<b}
		u_{n,\ell}(b-\ell)^{n-2}+\sum_{\ell>b}
		u_{n,2b-\ell}\bigl(\ell-b\bigr)^{n-2}\\
		&=2\sum_{\ell<b}
		u_{n,\ell}(b-\ell)^{n-2}.
	\end{align*}
	Since $\sum_{\ell=0}^{2b}u_{n,\ell}(b-\ell)^{n-2}=0$ by~\eqref{x=b},
	we obtain
	\begin{equation}\label{app:center-half-sum}
		\sum_{\ell<b}
		u_{n,\ell}(b-\ell)^{n-2}=0.
	\end{equation}
	
	Finally, since the interval $(a,b)$ contains no integer,
	the definition of $\mathcal J_I$ gives
	$$
	\mathcal J_I
	=\left\{\ell\in\mathbb Z_{\ge0}:
	\ell<\frac{a+b}{2}\right\}
	=\{\ell\in\mathbb Z_{\ge0}:\ell<b\}.
	$$
	Therefore, by~\eqref{eq:Ly-sum}
	and~\eqref{app:center-half-sum}, we conclude that
	$$
	L\left(\frac{n(n+1)}4\right)
	=L(b)
	=\frac{1}{n!(n-2)!}
	\sum_{\ell<b}
	u_{n,\ell}(b-\ell)^{n-2}
	=0.
	$$
	This completes the proof.
