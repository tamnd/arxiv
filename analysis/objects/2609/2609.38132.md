<!-- assumption-1: body_md -->
\label{assump:wcmdp:aperiodic-unichain}
    The transition matrix $P_{{{\bar{\pi}}^*}}$ induced by ${{\bar{\pi}}^*}$ has a simple eigenvalue $1$; all other eigenvalues have modulus strictly less than $1$; i.e., $P_{{{\bar{\pi}}^*}}$ defines an aperiodic unichain on $\mathbb{S}$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{assump:wcmdp:non-degeneracy}
    The matrix $C^*$ has full column rank.
<!-- end assumption-2 -->
<!-- remark-1: body_md -->
To gain intuition for \eqref{eq:wcmdp:active-system} and \Cref{assump:wcmdp:non-degeneracy}, we consider the RB specialization $|\mathbb{A}|=2$, $\mathcal{K}^*=\{1\}$, $c_1(s,a)=a$, with $S^{\emptyset}=\emptyset$. The state space partition as $\mathbb{S} = S^+ \cup S^- \cup S^0$, where $S^+ = \{s\colon y^*(s,1)>0,\,y^*(s,0)=0\}$, $S^- = \{s\colon y^*(s,0)>0,\,y^*(s,1)=0\}$, and the neutral states $S^0 = \{s\colon y^*(s,0)>0,\,y^*(s,1)>0\}$. Then $\mathcal{U}^* = \{(s,0)\colon s\in S^+\}\cup\{(s,1)\colon s\in S^-\}$, and the linear system \eqref{eq:wcmdp:active-system} specializes into
\[
\begin{aligned}
y(s,0)&=0\;(s\in S^+), & y(s,1)&=0\;(s\in S^-), \\
\textstyle\sum_s y(s,1)&=\alpha, & y(s,0)+y(s,1)&=x(s)\;(s\in\mathbb{S}).
\end{aligned}
\]
The first two set of equtions pin down $y(s,1) = x(s)$ for $s\in S^+$ and $y(s,0) = x(s)$ for $s\in S^-$. To find $y(s,a)$ for $s\in S^0$, we discuss based on the number of neutral states. 
\begin{itemize}
    \item When there is exactly one neutral state $\tilde{s}$ (the usual non-degenerate RB case), we can use the budget constraint to obtain $y(\tilde{s},1) = \alpha - \sum_{s\in S^+} x(s)$ and use the marginal constraint to obtain $y(\tilde{s}, 0) = x(\tilde{s}) - y(\tilde{s}, 1)$. In this case, the solution $y$ is uniquely determined for any $x$, implying that $C^*$ is an invertible matrix and thus has full column rank (i.e., \Cref{assump:wcmdp:non-degeneracy} holds). This is consistent with the dimensionality of $C^*$: one can verify that $d = 2|\mathbb{S}| = |\mathbb{S}||\mathbb{A}|$, so $C^*$ is a square matrix.
    \item When there is no neutral state, the linear system is overdetermined with $|S^+| + |S^-| + 1 + |\mathbb{S}| = 2|\mathbb{S}|+1 > |\mathbb{S}||\mathbb{A}|$ equations, so it does not have solutions for all $x$; $C^*$ is wide and cannot have full column rank.
    \item When there is more than one neutral state for RBs, the solution $y$ always exists but is not unique, so $C^*$ still has full column rank and \Cref{assump:wcmdp:non-degeneracy} still holds. In this case, we will select a fixed solution $y$, as we discuss immediately below while setting up for the next assumption.
\end{itemize}
When there is exactly one neutral state, the affine target allocates action $1$ to the states in $S^+$ and uses the neutral state to meet the budget constraint.

\label{def:wcmdp:maximal-feasible-set}
    Given the current system state $x$ and $\epsilon_N^{\text{rd}}\geq0$,
    a set of arms $D\subseteq[N]$ is $\epsilon_N^{\text{rd}}$-maximal feasible if the two conditions hold:

<!-- proposition-1: body_md -->
\label{prop:wcmdp:cmat-dim}
    Suppose \Cref{assump:wcmdp:aperiodic-unichain} holds, $|S^{\emptyset}|=0$, and $y^*$ is a non-degenerate basic feasible solution of \eqref{eq:wcmdp-lp}. Then $d = |\mathbb{S}||\mathbb{A}|$, so $C^*$ is square.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
We write the flow-balance equations \eqref{eq:wcmdp:flow} as $yB=0$, where $B\in\mathbb{R}^{|\mathbb{S}||\mathbb{A}|\times|\mathbb{S}|}$ has entries
\[
    B_{(s,a),s'}=\mathds{1}\!\left\{s=s'\right\}-P(s,a,s').
\]
Since $B\mathds{1}^\top=0$, its rank is at most $|\mathbb{S}|-1$. Conversely, if a column vector $v$ satisfies $Bv=0$, then
\[
    v(s)=\sum_{s'\in\mathbb{S}}P(s,a,s')v(s')
    \quad\forall (s,a)\in\mathbb{S}\times\mathbb{A}.
\]
Averaging over ${{\bar{\pi}}^*}(a\mid s)$ gives $P_{{\bar{\pi}}^*} v=v$. By \Cref{assump:wcmdp:aperiodic-unichain}, the eigenspace at $1$ is spanned by $\mathds{1}^\top$. Thus $\ker B=\operatorname{span}\{\mathds{1}^\top\}$ and $\operatorname{rank}B=|\mathbb{S}|-1$.

The equation $\sum_{s\in\mathbb{S}, a\in\mathbb{A}} y(s,a)=1$ in \eqref{eq:wcmdp:simplex} adds one independent equality: its coefficient vector $\mathds{1}^\top_{|\mathbb{S}||\mathbb{A}|}$ cannot lie in the column space of $B$, because $y^*B=0$ whereas $y^*\mathds{1}^\top_{|\mathbb{S}||\mathbb{A}|}=1$. Thus the LP has $|\mathbb{S}|$ independent equality constraints.

By the definition of non-degenerate BFS, the number of tight inequality constraints at $y^*$ equals $|\mathbb{S}||\mathbb{A}|$ minus the rank of the equality constraints, $|\mathbb{S}|$, so there are $|\mathbb{S}||\mathbb{A}| - |\mathbb{S}|$ tight inequality constraints. 
With $|S^{\emptyset}|=0$, the binding inequalities at $y^*$ are the $|\mathcal{K}^*|$ tight budget constraints and the $|\mathcal{U}^*|$ non-negativity constraints on $\mathcal{U}^*$, giving $|\mathcal{K}^*| + |\mathcal{U}^*| = |\mathbb{S}||\mathbb{A}| - |\mathbb{S}|$. Consequently, we have $|\mathcal{U}^*| = |\mathbb{S}|(|\mathbb{A}|-1) - |\mathcal{K}^*|$. Substituting into the definition of $d$, we get
\[
\begin{aligned}
d &= |\mathcal{U}^*| + |\mathcal{K}^*| + |\mathbb{S}| \\
    &= \bigl(|\mathbb{S}|(|\mathbb{A}|-1) - |\mathcal{K}^*|\bigr) + |\mathcal{K}^*| + |\mathbb{S}| \\
    &= |\mathbb{S}||\mathbb{A}|.
\end{aligned}\qedhere
\]
<!-- end proof-1 -->
<!-- remark-2: body_md -->
\label{rem:wcmdp:transient-degeneracy}
Under \Cref{assump:wcmdp:aperiodic-unichain}, a non-degenerate BFS cannot have transient states. To prove this claim, we consider a BFS $y^*$ with $S^{\emptyset}\ne\emptyset$ and define the column vector $u$ by $u(s)=1$ for $s\inS^{\emptyset}$ and $u(s)=0$ otherwise. Summing flow balance over $S^{\emptyset}$ gives $yBu=0$, with $B$ as defined in the preceding proof. Since $S^{\emptyset}$ is a nonempty proper subset of $\mathbb{S}$, $u$ is nonconstant, so the identity $\ker B=\operatorname{span}\{\mathds{1}^\top\}$ from that proof implies $Bu\ne0$.

The equation $yBu=0$ involves only variables $y(s,a)$ such that $y^*(s,a)=0$. Indeed, zero stationary mass in $S^{\emptyset}$ and the flow-balance equations \eqref{eq:wcmdp:flow} evaluated at $y^*$ give
\[
\begin{aligned}
0 &= \sum_{s'\inS^{\emptyset}}\sum_{a\in\mathbb{A}}y^*(s',a) \\
  &= \sum_{s\in\mathbb{S}, a\in\mathbb{A}} y^*(s,a)\sum_{s'\inS^{\emptyset}}P(s,a,s').
\end{aligned}
\]
Nonnegativity of the terms in the last sum implies $\sum_{s'\inS^{\emptyset}}P(s,a,s')=0$ whenever $y^*(s,a)>0$. For such a pair $(s,a)$, we also have $u(s)=0$, so $(Bu)_{(s,a)}=0$.

To conclude degeneracy, we replace the LP equalities by an independent spanning set of $|\mathbb{S}|$ equations, using the equality rank established in the preceding proof. The retained equalities have the same linear consequences as the original LP equalities, including $yBu=0$. The equation $yBu=0$ is also a linear combination of the tight non-negativity constraints, written as $y(s,a)=0$ at pairs $(s,a)$ where $y^*(s,a)=0$. The two expressions for the nonzero vector $Bu$, one using the retained equalities and the other using the tight non-negativity constraints, give a nontrivial linear dependence among the constraint coefficient vectors. Since the retained LP equalities are independent and $y^*$ is a BFS, this linear dependence proves that $y^*$ is degenerate.
<!-- end remark-2 -->
<!-- lemma-6: body_md -->
\label{lem:wcmdp:Cinvs-ones}
    The matrix $C^+_\mathbb{S}$ has row sums equal to $1$, and the square matrix $C^+_\mathbb{S} \mathcal{P}$ has eigenvalue $1$ with right eigenvector $\mathds{1}^\top$:
    \begin{align}
        \label{eq:Cinvs-eig}
        C^+_\mathbb{S}\,\mathds{1}^\top_{|\mathbb{S}||\mathbb{A}|} &= \mathds{1}^\top \\
        \label{eq:Cinvs-P-eig}
        C^+_\mathbb{S}\,\mathcal{P}\,\mathds{1}^\top &= \mathds{1}^\top.
    \end{align}
<!-- end lemma-6 -->
<!-- proof-2: body_md -->
We first show \eqref{eq:Cinvs-eig}.
    We recall that
    \begin{equation}\tag{\ref{eq:wcmdp:Cinvs-Cmat}}
        C^+_\mathbb{S}\,C^* = \bigl(0,\;0,\;0,\;I_{|\mathbb{S}|}\bigr).
    \end{equation}
    We let $M\in\mathbb{R}^{(|\mathbb{S}||\mathbb{A}|)\times|\mathbb{S}|}$ be the last $\left\lvert\mathbb{S}\right\rvert$ columns of $C^*$. Then
    \[
        C^+_\mathbb{S} M = I_{|\mathbb{S}|}.
    \]
    Since $C^*$ is the coefficient matrix of the linear system \eqref{eq:wcmdp:active-system}, one can read off from the linear system that $M_{(s,a),s'} = \mathds{1}\!\left\{s=s'\right\}$ for $s,s'\in\mathbb{S}$ and $a\in\mathbb{A}$.
    Since $M\,\mathds{1}^\top = \mathds{1}^\top_{|\mathbb{S}||\mathbb{A}|}$, we have $C^+_\mathbb{S}\,\mathds{1}^\top_{|\mathbb{S}||\mathbb{A}|} = C^+_\mathbb{S}\,M\,\mathds{1}^\top = \mathds{1}^\top$.

    Next, we show \eqref{eq:Cinvs-P-eig}. Since $\mathcal{P}_{(s,a),s'} = P(s,a,s')$, we have $\sum_{s'}\mathcal{P}_{(s,a),s'} = 1$ for each row $(s,a)\in\mathbb{S}\times\mathbb{A}$. Consequently, we have $\mathcal{P}\,\mathds{1}^\top = \mathds{1}^\top_{|\mathbb{S}||\mathbb{A}|}$ and thus $C^+_\mathbb{S}\,\mathcal{P}\,\mathds{1}^\top = C^+_\mathbb{S}\,\mathds{1}^\top_{|\mathbb{S}||\mathbb{A}|} = \mathds{1}^\top$.
<!-- end proof-2 -->
<!-- proposition-2: body_md -->
\label{prop:wcmdp:phi-vs-CinvsP}
    Let $\Phi$ be defined as in \eqref{eq:wcmdp:phi-def}. Then:
    \begin{enumerate}[leftmargin=2em, label=(\roman*)]
        \item $\Phi$ and $C^+_\mathbb{S}\mathcal{P}$ agree on the hyperplane $H \triangleq \{v\in\mathbb{R}^{|\mathbb{S}|} : v\mathds{1}^\top = 0\}$, i.e., $v\Phi = v\,C^+_\mathbb{S}\mathcal{P}$ for all $v\in H$.
        \item The eigenvalues of $\Phi$ are exactly the eigenvalues of $C^+_\mathbb{S}\mathcal{P}$ with one occurrence of the eigenvalue $1$ (associated with the right eigenvector $\mathds{1}^\top$) replaced by $0$, leaving all other eigenvalues unchanged.
    \end{enumerate}
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
For (i): For any $v\in H$, $v(\Phi - C^+_\mathbb{S}\mathcal{P}) = -v\,\mathds{1}^\top\,\mu^*\,C^+_\mathbb{S}\mathcal{P} = -(v\,\mathds{1}^\top)\,\mu^*\,C^+_\mathbb{S}\mathcal{P} = 0$.

For (ii), we apply Brauer\textquotesingle s rank-one perturbation theorem \cite{Bra_52}. Since $C^+_\mathbb{S} \mathcal{P}$ has an eigenvalue $1$ with right eigenvector $\mathds{1}^\top$,
Brauer's theorem implies that
the spectrum of $\Phi =  C^+_\mathbb{S} \mathcal{P} - \mathds{1}^\top\mu^*\,C^+_\mathbb{S}\mathcal{P}$ is identical to the spectrum of $C^+_\mathbb{S} \mathcal{P}$, with one occurrence of $1$ replaced by $1 - \mu^*\,C^+_\mathbb{S}\mathcal{P}\mathds{1}^\top = 0$.
<!-- end proof-3 -->

The well-definedness and eigenvalue lower bound for $W$ has been proved in \cite{HonXieCheWan_24}. The main fact used in the proof is that all eigenvalues of $P_{{\bar{\pi}}^*} - \mathds{1}^\top \mu^*$ have moduli strictly less than $1$ under \Cref{assump:wcmdp:aperiodic-unichain}.
    Since we have assumed the same things for $\Phi$ in \Cref{assump:wcmdp:local-stability}, the proof for $U$ is analogous with $P_{{\bar{\pi}}^*} - \mathds{1}^\top \mu^*$ replaced by  $\Phi$.

<!-- lemma-8: body_md -->
\label{lem:wcmdp:one-step-contraction-W-U}
    Assume \Cref{assump:wcmdp:aperiodic-unichain,assump:wcmdp:non-degeneracy,assump:wcmdp:local-stability}. For any distribution $v\in\Delta(\mathbb{S})$,
    \begin{align}
        \label{eq:wcmdp:pibar-contraction}
        \left\lVert(v - \mu^*)P_{{\bar{\pi}}^*}\right\rVert_W &\leq \rho_{w} \left\lVertv - \mu^*\right\rVert_W, \\
        \label{eq:wcmdp:lp-contraction}
        \left\lVert(v - \mu^*)\Phi\right\rVert_U &\leq \rho_{u} \left\lVertv - \mu^*\right\rVert_U,
    \end{align}
    where $\rho_{w} = 1 - 1/(2\lambda_W)$ and $\rho_{u} = 1 - 1/(2\lambda_U)$.
<!-- end lemma-8 -->

The inequality \eqref{eq:wcmdp:pibar-contraction} follows from the weighted-norm argument in \cite{HonXieCheWan_24}; the same argument applies to \eqref{eq:wcmdp:lp-contraction} with $P_{{\bar{\pi}}^*}-\mathds{1}^\top\mu^*$ replaced by $\Phi$.

<!-- lemma-9: body_md -->
\label{lem:wcmdp:linear-feasibility}
    With $\epsilon$ as in \eqref{eq:wcmdp:eps-eta} and $\eta$ as in \eqref{eq:wcmdp:eta-def}, for every $x\in\Delta(\mathbb{S})$ with $\left\lVertx - \mu^*\right\rVert_U \leq \eta$, the vector $y_t = y^* + (x-\mu^*)\,C^+_\mathbb{S}$ defined in \eqref{eq:wcmdp:y-linear} satisfies

<!-- proof-6: body_md -->
We first invoke Cauchy--Schwarz to get the following bound that is used for proving both (i) and (ii): for every $(s,a)\in\mathbb{S}\times\mathbb{A}$, we have
    \begin{equation}\label{eq:wcmdp:y_t-entrywise}
\begin{aligned}
& \left\lverty_t(s,a)-y^*(s,a)\right\rvert \\
    & \quad = \left\lvert(x-\mu^*)\,C^+_\mathbb{S}\,e_{(s,a)}\right\rvert \\
    & \quad \leq \left\lVertC^+_\mathbb{S}\right\rVert_{U^{-1}}\,\left\lVertx-\mu^*\right\rVert_U \\
    & \quad \leq \left\lVertC^+_\mathbb{S}\right\rVert_{U^{-1}}\,\eta = \epsilon.
\end{aligned}
\end{equation}

    Now we prove (i) and (ii) separately.

    \emph{(i) Non-negativity.} For $(s,a)\notin\mathcal{U}^*\cup(S^{\emptyset}\times\mathbb{A})$, $y^*(s,a) \geq \epsilon$ by \eqref{eq:wcmdp:eps-eta}, so \eqref{eq:wcmdp:y_t-entrywise} gives $y_t(s,a)\geq 0$. For $(s,a)\in\mathcal{U}^*$, $y_t(s,a) = 0$ by the $\mathcal{U}^*$-block of \eqref{eq:wcmdp:Cinvs-Cmat}. For $s\inS^{\emptyset}$, the $S^{\emptyset}$-block of \eqref{eq:wcmdp:active-system} forces $y_t(s,0) = \cdots = y_t(s,|\mathbb{A}|-1) = x(s)/|\mathbb{A}| \geq 0$.

    \emph{(ii) Budget constraints.}
    For $k\in\mathcal{K}^*$, the linear equations \eqref{eq:wcmdp:active-system} force $\sum_{s\in\mathbb{S}, a\in\mathbb{A}} c_k(s,a)\,y_t(s,a) = \alpha_k$.
    For $k\notin\mathcal{K}^*$, by the definition of $\epsilon$ in \eqref{eq:wcmdp:eps-eta}, we have
    \[
        \sum_{s\in\mathbb{S}, a\in\mathbb{A}} c_k(s,a)\,y^*(s,a) \leq \alpha_k - |\mathbb{S}||\mathbb{A}|\,\epsilon
    \]
    Combining this bound with \eqref{eq:wcmdp:y_t-entrywise} and using the fact that $c_k(s,a)\leq 1$ for all $(s,a)\in\mathbb{S}\times\mathbb{A}$, we have
    \[
\begin{aligned}
& \sum_{s\in\mathbb{S}, a\in\mathbb{A}} c_k(s,a)\,y_t(s,a) \\
    & \quad \leq \sum_{s\in\mathbb{S}, a\in\mathbb{A}} c_k(s,a)\,y^*(s,a) + |\mathbb{S}||\mathbb{A}|\,\epsilon \\
    & \quad \leq \alpha_k.
\end{aligned}\qedhere
\]

\label{lem:wcmdp:subroutine-conform}
    Assume \Cref{assump:wcmdp:aperiodic-unichain,assump:wcmdp:non-degeneracy,assump:wcmdp:local-stability} hold.

    \begin{enumerate}[leftmargin=2em, label=(\roman*)]
        \item The arms in $D^{\textup{LP}}_t$ can follow \textup{LP fixed-point}\xspace.
        \item The budget constraints are satisfied, i.e., for every $k\in[K]$, $\sum_{i\in [N]} c_k(S_t(i), A_t(i)) \leq \alpha_k N$.

We first consider the selection of $D^{\textup{LP}}_t$. If the full set is feasible, it is $\epsilon_N^{\text{rd}}$-maximal. Otherwise, when $D^{\textup{LP}}_{t-1}$ is feasible, the family of feasible supersets of $D^{\textup{LP}}_{t-1}$ is nonempty; when it is infeasible, the family of all feasible sets contains $\emptyset$. In either case, a set of maximum cardinality within that family is $\epsilon_N^{\text{rd}}$-maximal, since $\epsilon_N^{\text{rd}}\geq0$ and every superset with slack at least $\epsilon_N^{\text{rd}}$ is also feasible. Thus an admissible $D^{\textup{LP}}_t$ exists.

    For the selection of $D^{{{\bar{\pi}}^*}}_t$, we let $B_t=D^{{{\bar{\pi}}^*}}_{t-1}\setminusD^{\textup{LP}}_t$ and $q_t=\lfloor\alpha_{\min}(N-|D^{\textup{LP}}_t|)\rfloor$. Both $B_t\subseteq[N]\setminusD^{\textup{LP}}_t$ and $0\leq q_t\leq N-|D^{\textup{LP}}_t|$ hold. If $|B_t|\geq q_t$, we take a subset of $B_t$ of size $q_t$; otherwise, we extend $B_t$ to that size within $[N]\setminusD^{\textup{LP}}_t$. These choices satisfy the required nesting condition. Each set-selection step therefore samples from a finite, nonempty family of admissible choices.

    For (i), if $D^{\textup{LP}}_t = \emptyset$, \textup{LP fixed-point}\xspace returns without assigning actions. If $D^{\textup{LP}}_t \neq \emptyset$, because the two-set policy chooses $D^{\textup{LP}}_t$ to satisfy $\delta(X_t, D^{\textup{LP}}_t) \geq 0$, we have
    \[
        \left\lVertX_t(D^{\textup{LP}}_t)/m(D^{\textup{LP}}_t) - \mu^*\right\rVert_U \leq \eta.
    \]
    Since $X_t(D^{\textup{LP}}_t)/m(D^{\textup{LP}}_t)$ is the empirical distribution of arms in $D^{\textup{LP}}_t$, by \Cref{lem:wcmdp:linear-feasibility}, the arms in $D^{\textup{LP}}_t$ can follow \textup{LP fixed-point}\xspace.

    For (ii), the arms in $D^{\textup{LP}}_t$ satisfy the scaled budget constraints
    \[
        \sum_{i\in D^{\textup{LP}}_t} c_k(S_t(i), A_t(i))
        \leq N m(D^{\textup{LP}}_t)\,\alpha_k.
    \]
    This follows from \Cref{lem:wcmdp:linear-feasibility} when $D^{\textup{LP}}_t\ne\emptyset$; for an empty set, both sides are zero.
    The cost in $D^{{{\bar{\pi}}^*}}_t$ is at most $|D^{{{\bar{\pi}}^*}}_t| \leq N \alpha_{\min}(1 - m(D^{\textup{LP}}_t)) \leq N \alpha_k(1 - m(D^{\textup{LP}}_t))$ since $c_k(s,a) \in [0,1]$ for all $s\in\mathbb{S}$, $a\in\mathbb{A}$, and $k\in[K]$.

    \[
\begin{aligned}
& \sum_{i\in[N]}c_k(S_t(i),A_t(i)) \\
    & \quad \leq N m(D^{\textup{LP}}_t)\,\alpha_k + N\alpha_k(1-m(D^{\textup{LP}}_t)) + 0 \\
    & \quad = N\alpha_k.
\end{aligned}\qedhere
\]

    By the pseudo-contraction property of $P_{{\bar{\pi}}^*}$ under the $W$-weighted norm given in \Cref{lem:wcmdp:one-step-contraction-W-U},

    & \quad = \rho_{w}\left\lVertX_t(D)-m(D)\mu^*\right\rVert_W \\
    & \quad \geq \left\lVert(X_t(D)-m(D)\mu^*)\,P_{{\bar{\pi}}^*}\right\rVert_W \\
    & \quad = \left\lVertX_t(D)\,P_{{\bar{\pi}}^*}-m(D)\mu^*\right\rVert_W,
\end{align*}
    so by the triangle inequality and the relation between the $W$-weighted norm and the $L_1$ norm,
    \[
\begin{aligned}
& h_W(X_{t+1}',D)-\rho_{w}\,h_W(X_t,D) \\
    & \quad \leq \left\lVertX_{t+1}'(D)-X_t(D)\,P_{{\bar{\pi}}^*}\right\rVert_W \\
    & \quad \leq \lambda_W^{1/2}\left\lVertX_{t+1}'(D)-X_t(D)\,P_{{\bar{\pi}}^*}\right\rVert_1.
\end{aligned}
\]
    Therefore, it suffices to upper bound
\[\left\lVertX_{t+1}'(D) - X_t(D)\,P_{{\bar{\pi}}^*}\right\rVert_1\] in expectation and in probability.

    Under \Cref{alg:wcmdp:uoc}, each arm $i\in D$ independently samples an action $A_t(i) \sim {{\bar{\pi}}^*}(\cdot \,|\, S_t(i))$ and then transitions according to $P$. Hence, given $X_t$, the indicators $\mathds{1}\!\left\{S_{t+1}'(i) = s\right\}$ for $i\in D$ are independent Bernoulli random variables with means $P_{{\bar{\pi}}^*}(S_t(i), s)$, where $S_{t+1}'(i)$ denotes the state of arm $i$ at time $t+1$ if the arms in $D$ follow \Cref{alg:wcmdp:uoc}. In particular, by \Cref{lem:wcmdp:fr-transition}, $\mathbb{E}\left[X_{t+1}'(D) \,\middle|\, X_t\right] = X_t(D)\,P_{{\bar{\pi}}^*}$. 

    For each $s\in\mathbb{S}$, \[X_{t+1}'(D, s) = (1/N)\sum_{i\in D} \mathds{1}\!\left\{S_{t+1}'(i) = s\right\}\] is a sum of independent random variables taking values in $[0, 1/N]$. By Cauchy--Schwarz,
    \[
\begin{aligned}
& \mathbb{E}\Big[\Big\lvertX_{t+1}'(D,s)-\mathbb{E}\left[X_{t+1}'(D,s)\,\middle|\, X_t\right]\Big\rvert\,\Big|\, X_t\Big] \\
    & \quad \leq \text{Var}{\left[X_{t+1}'(D,s)\,\Big|\, X_t\right]}^{1/2} \\
    & \quad \leq \frac{|D|^{1/2}}{N}\leq\frac{1}{\sqrt{N}},
\end{aligned}
\]
    and by Hoeffding's inequality, for all $r\geq 0$,
    \[
\begin{aligned}
& \mathbb{P}\Big[\Big\lvertX_{t+1}'(D,s)-\mathbb{E}\left[X_{t+1}'(D,s)\,|\, X_t\right]\Big\rvert>r\,\Big|\, X_t\Big] \\
    & \quad \leq 2\exp\big(-2N r^2\big).
\end{aligned}
\]
    Summing the expectation bound over $s\in\mathbb{S}$, and applying the union bound over $s\in\mathbb{S}$ (with $r$ replaced by $r/|\mathbb{S}|$) to the probability bound, we get
    \begin{align*}
    & \mathbb{E}\big[\left\lVertX_{t+1}'(D)-X_t(D)\,P_{{\bar{\pi}}^*}\right\rVert_1\,\big|\, X_t\big] \\
    & \quad \leq \frac{|\mathbb{S}|}{\sqrt{N}} \quad a.s., \\
    & \mathbb{P}\big[\left\lVertX_{t+1}'(D)-X_t(D)\,P_{{\bar{\pi}}^*}\right\rVert_1>r\,\big|\, X_t\big] \\
    & \quad \leq 2|\mathbb{S}|\exp\Big(-\frac{2N r^2}{|\mathbb{S}|^2}\Big) \quad a.s.,\;\forall r\geq0.
\end{align*}
    Combining the two displays with the bound $h_W(X_{t+1}', D) - \rho_{w}\,h_W(X_t, D) \leq \lambda_W^{1/2}\left\lVertX_{t+1}'(D) - X_t(D)\,P_{{\bar{\pi}}^*}\right\rVert_1$ established earlier, we obtain \eqref{eq:wcmdp:hw:drift} and \eqref{eq:wcmdp:hw:high-prob} with $K_{W,1} = \lambda_W^{1/2}|\mathbb{S}|$, $K_{W,2} = 2|\mathbb{S}|$, and $C_{W} = 2/(\lambda_W|\mathbb{S}|^2)$.

    \textbf{Proving the analogs of \eqref{eq:wcmdp:hw:drift} and \eqref{eq:wcmdp:hw:high-prob} for $h_U$.}
    For $h_U$, we must account for the rounding residual in \Cref{lem:wcmdp:olc-transition}. We write $K_{\mathrm{rnd}}\triangleq 2|\mathbb{S}|(|\mathbb{A}|-1)$ for the constant in the rounding bound of \Cref{lem:wcmdp:olc-rounding}.

    By the pseudo-contraction property of $\Phi$ under the $U$-weighted norm given in \Cref{lem:wcmdp:one-step-contraction-W-U},
    \begin{align*}
    \rho_{u}h_U(X_t,D) & = \rho_{u}\left\lVertX_t(D)-m(D)\mu^*\right\rVert_U \\
    & \quad \geq \left\lVert(X_t(D)-m(D)\mu^*)\,\Phi\right\rVert_U.
\end{align*}
    Consequently,
    \begin{align*}
    & h_U(X_{t+1}',D)-\rho_{u}h_U(X_t,D) \\
    & \quad \leq \left\lVertX_{t+1}'(D)-m(D)\mu^*\right\rVert_U \\
    & \qquad - \left\lVert(X_t(D)-m(D)\mu^*)\,\Phi\right\rVert_U \\
    & \quad \leq \left\lVertX_{t+1}'(D)-m(D)\mu^*-(X_t(D)-m(D)\mu^*)\,\Phi\right\rVert_U \\
    & \quad \leq \lambda_U^{1/2}\Bigl\lVert X_{t+1}'(D)-m(D)\mu^* \\
    & \qquad - (X_t(D)-m(D)\mu^*)\,\Phi\Bigr\rVert_1.
\end{align*}
    Therefore, it suffices to show that, with $C'=1/(2|\mathbb{S}|^2)$, we have
    \begin{align}
    & \mathbb{E}\Bigl[\left\lVertX_{t+1}'(D)-m(D)\mu^*-(X_t(D)-m(D)\mu^*)\,\Phi\right\rVert_1 \nonumber \\
    & \qquad \,|\, X_t\Bigr] \nonumber \\
    & \quad = O(1/\sqrt{N}) \quad a.s., \label{eq:wcmdp:hu:interm-goal-1} \\
    & \mathbb{P}\Bigl[\Bigl\lVert X_{t+1}'(D)-m(D)\mu^* \nonumber \\
    & \qquad - (X_t(D)-m(D)\mu^*)\,\Phi\Bigr\rVert_1>r\,|\, X_t\Bigr] \nonumber \\
    & \quad = O(\exp(-C'N r^2)) \quad a.s.,\;\forall r\geq0. \label{eq:wcmdp:hu:interm-goal-2}
\end{align}

    To control the rounding residual, we define the LP target for the arms in $D$ by
    \[
        y_t^D\triangleq y^*+\bigl(X_t(D)/m(D)-\mu^*\bigr)C^+_\mathbb{S}.
    \]
    Applying \Cref{lem:wcmdp:olc-transition} to the arms in $D$ and multiplying by $m(D)$ gives
    \begin{equation}\label{eq:wcmdp:hu:cond-mean}
\begin{aligned}
& \mathbb{E}\left[X_{t+1}'(D)-m(D)\mu^*\,\middle|\, X_t,(A_t(i))_{i\in D}\right] \\
    & \quad = (X_t(D)-m(D)\mu^*)\,\Phi \\
    & \qquad + (Y_t(D)-m(D)\,y_t^D)\,\mathcal{P}.
\end{aligned}
\end{equation}
    We decompose $X_{t+1}'(D) - m(D)\mu^* - (X_t(D) - m(D)\mu^*)\,\Phi$ by adding and subtracting the conditional mean $\mathbb{E}\left[X_{t+1}'(D) \,|\, X_t, (A_t(i))_{i\in D}\right]$:
    \begin{align}
    & X_{t+1}'(D)-m(D)\mu^*-(X_t(D)-m(D)\mu^*)\,\Phi \nonumber \\
    & \quad = \Bigl(X_{t+1}'(D)-\mathbb{E}\left[X_{t+1}'(D)\,|\, X_t,(A_t(i))_{i\in D}\right]\Bigr) \nonumber \\
    & \qquad + \Bigl(\mathbb{E}\left[X_{t+1}'(D)\,|\, X_t,(A_t(i))_{i\in D}\right] \nonumber \\
    & \qquad - m(D)\mu^*-(X_t(D)-m(D)\mu^*)\,\Phi\Bigr) \nonumber \\
    & \quad = \bigl(X_{t+1}'(D)-\mathbb{E}\left[X_{t+1}'(D)\,|\, X_t,(A_t(i))_{i\in D}\right]\bigr) \nonumber \\
    & \qquad + (Y_t(D)-m(D)\,y_t^D)\,\mathcal{P}. \nonumber
\end{align}
    The rounding bound for $|D|$ arms, scaled by $m(D)=|D|/N$, is $K_{\mathrm{rnd}}/N$. Since $\mathcal{P}$ is row-stochastic,
    \[
    \begin{aligned}
    & \left\lVert(Y_t(D)-m(D)y_t^D)\mathcal{P}\right\rVert_1 \\
    & \quad \leq \left\lVertY_t(D)-m(D)y_t^D\right\rVert_1
        \leq \frac{K_{\mathrm{rnd}}}{N}.
    \end{aligned}
    \]
    The triangle inequality therefore gives
    \begin{equation}\label{eq:wcmdp:hu:decomp}
\begin{aligned}
& \left\lVertX_{t+1}'(D)-m(D)\mu^*-(X_t(D)-m(D)\mu^*)\,\Phi\right\rVert_1 \\
    & \quad \leq \left\lVertX_{t+1}'(D)-\mathbb{E}\left[X_{t+1}'(D)\,|\, X_t,(A_t(i))_{i\in D}\right]\right\rVert_1 \\
    & \qquad + \frac{K_{\mathrm{rnd}}}{N}.
\end{aligned}
\end{equation}
    Given $X_t$ and $(A_t(i))_{i\in D}$, the next-state indicators are independent across arms. For each state, the centered coordinate on the right-hand side of \eqref{eq:wcmdp:hu:decomp} is a sum of $|D|$ centered Bernoulli variables divided by $N$. The same variance estimate used for $h_W$ bounds the expected $L_1$ norm by $|\mathbb{S}|\sqrt{|D|}/N\leq |\mathbb{S}|/\sqrt{N}$. Averaging over the actions and adding the rounding bound gives an upper bound of $(|\mathbb{S}|+K_{\mathrm{rnd}})/\sqrt{N}$ in \eqref{eq:wcmdp:hu:interm-goal-1}.

    To prove the tail bound, first suppose $r\geq 2K_{\mathrm{rnd}}/N$. By \eqref{eq:wcmdp:hu:decomp}, a deviation larger than $r$ requires the centered $L_1$ norm to exceed $r-K_{\mathrm{rnd}}/N\geq r/2$. Hoeffding's inequality and a union bound over states give
    \begin{align*}
    & \mathbb{P}\Bigl[\Bigl\lVert X_{t+1}'(D)-m(D)\mu^* - (X_t(D)-m(D)\mu^*)\,\Phi\Bigr\rVert_1>r\\
    & \qquad\qquad   \,\Big|\, X_t,(A_t(i))_{i\in D}\Bigr] \\
    & \quad \leq \mathbb{P}\Bigl[\Bigl\lVert X_{t+1}'(D) - \mathbb{E}\left[X_{t+1}'(D)\,|\, X_t,(A_t(i))_{i\in D}\right]\Bigr\rVert_1 > r/2 \\
    & \qquad\qquad \,\Big|\, X_t,(A_t(i))_{i\in D}\Bigr] \\
    & \quad \leq 2|\mathbb{S}|\exp\!\left(-\frac{N^2r^2}{2|D||\mathbb{S}|^2}\right) \\
    & \quad \leq 2|\mathbb{S}|\exp(-C'Nr^2).
\end{align*}
    For $0\leq r<2K_{\mathrm{rnd}}/N$, we instead use the probability bound $1$. Since $Nr^2\leq 4K_{\mathrm{rnd}}^2$ for $N\geq1$, this bound is at most $\exp(4C'K_{\mathrm{rnd}}^2)\exp(-C'Nr^2)$. Thus, setting
    \[
        K_{U,2}=\max\bigl\{2|\mathbb{S}|,\exp(4C'K_{\mathrm{rnd}}^2)\bigr\}
    \]
    and averaging over the actions gives, for every $r\geq0$,
    \begin{equation}\label{eq:pf-hu-drift-tail-1}
\begin{aligned}
& \mathbb{P}\Bigl[\Bigl\lVert X_{t+1}'(D)-m(D)\mu^* \\
    & \qquad - (X_t(D)-m(D)\mu^*)\,\Phi\Bigr\rVert_1>r\,\Big|\, X_t\Bigr] \\
    & \quad \leq K_{U,2}\exp(-C'Nr^2) \quad a.s.
\end{aligned}
\end{equation}
    This proves \eqref{eq:wcmdp:hu:interm-goal-2}. Using the factor $\lambda_U^{1/2}$ in the earlier bound for $h_U(X_{t+1}',D)-\rho_{u}h_U(X_t,D)$ establishes the claimed drift and tail bounds with
    \[
        K_{U,1}=\lambda_U^{1/2}(|\mathbb{S}|+K_{\mathrm{rnd}}),
        \qquad C_{U}=\frac{C'}{\lambda_U}.
    \]
    All constants are independent of $N$, $D$, and the current state.

    The inequalities \eqref{eq:wcmdp:hw:strength}--\eqref{eq:wcmdp:hw:lipschitz} and their $h_U$ analogs follow from the definitions of the weighted norms; the corresponding norm estimates are also discussed in \cite{HonXieCheWan_24}. \qedhere

The second claim of \Cref{lem:wcmdp:sufficient-coverage} follows directly from the definition of the policy, so we only need to prove its first claim.
    We claim that either $m(D^{{{\bar{\pi}}^*}}_t) \leq \epsilon_N^{\text{rd}}$, or

        \label{eq:two-set:sufficient-coverage:interm-goal-1}
        \lambda_U^{1/2} h_W(X_t, D^{{{\bar{\pi}}^*}}_t) >  \eta m(D^{{{\bar{\pi}}^*}}_t) - \epsilon_N^{\text{rd}} - 0.
    \end{equation}
    We first show \Cref{lem:wcmdp:sufficient-coverage} assuming this claim.
    We recall that $m(D^{{{\bar{\pi}}^*}}_{t})= \big\lfloor \beta (N -|D^{\textup{LP}}_{t}|)\big\rfloor \big/ N$. If $m(D^{{{\bar{\pi}}^*}}_t) \leq \epsilon_N^{\text{rd}}$,
    \[
        \beta(1-m(D^{\textup{LP}}_t)) - \frac{1}{N} \leq m(D^{{{\bar{\pi}}^*}}_t) \leq \epsilon_N^{\text{rd}},
    \]
    so $1-m(D^{\textup{LP}}_t) \leq 1/(\beta N) + \epsilon_N^{\text{rd}}/\beta$, which implies \Cref{lem:wcmdp:sufficient-coverage} by the non-negativity of $h_W(X_t, D^{{{\bar{\pi}}^*}}_t)$.
    If $m(D^{{{\bar{\pi}}^*}}_t) > \epsilon_N^{\text{rd}}$, \eqref{eq:two-set:sufficient-coverage:interm-goal-1} holds. Because $m(D^{{{\bar{\pi}}^*}}_t) = \left\lfloor\beta (N - |D^{\textup{LP}}_t|)\right\rfloor /N \geq \big(\beta (N - |D^{\textup{LP}}_t|) - 1\big) / N$, we have
    \begin{align*}
    & \lambda_U^{1/2}h_W(X_t,D^{{{\bar{\pi}}^*}}_t) \\
    & \quad > \eta\big(\beta(N-|D^{\textup{LP}}_t|)-1\big)\frac{1}{N}-\epsilon_N^{\text{rd}}-0 \\

\end{align*}
    Rearranging gives

        1 - m(D^{\textup{LP}}_t) < \frac{\lambda_U^{1/2}}{\eta\beta} h_W(X_t, D^{{{\bar{\pi}}^*}}_t) + \frac{1}{\beta N} + \frac{\epsilon_N^{\text{rd}}+0}{\eta\beta}.

1-m(D^{\textup{LP}}_t) &\leq \frac{\lambda_U^{1/2}}{\eta\beta}h_W(X_t,D^{{{\bar{\pi}}^*}}_t) + \frac{1}{\beta N} \\

    &\leq 2r_{\max}\Bigl(1+\frac{\sqrt{2}\,\lambda_U^{1/2}}{\eta}\Bigr) = K_{g},
\end{aligned}

    where in the last inequality, we use the facts that $\left\lVerte_s - \mu^*\right\rVert_U \leq \lambda_U^{1/2}\,\left\lVerte_s - \mu^*\right\rVert_2$ and $\left\lVerte_s - \mu^*\right\rVert_2^2 \leq \left\lVerte_s - \mu^*\right\rVert_1 \left\lVerte_s - \mu^*\right\rVert_\infty \leq 2$. \qedhere

We use the abbreviation that $a \triangleq V_1(\sigma)$, $b \triangleq V_1(\sigma')$, and $r_0 \triangleq (1-\rho_1)\overline{\eta}/2$.
The key algebraic identity
\begin{equation}\label{eq:trunc-key-identity}
    b - \frac{\overline{\eta}}{2} = \Bigl(b - \rho_1 a - r_0\Bigr) + \rho_1\Bigl(a - \frac{\overline{\eta}}{2}\Bigr),
\end{equation}
together with subadditivity $(x+y)^+ \leq x^+ + y^+$, yields
\begin{equation}\label{eq:trunc-unified}
\begin{aligned}
& \Bigl(b-\frac{\overline{\eta}}{2}\Bigr)^{\!+}-\Bigl(a-\frac{\overline{\eta}}{2}\Bigr)^{\!+} \\
    & \quad \leq \Bigl(b-\rho_1 a-r_0\Bigr)^{\!+} \\
    & \qquad - (1-\rho_1)\Bigl(a-\frac{\overline{\eta}}{2}\Bigr)^{\!+}.
\end{aligned}
\end{equation}
We derive the claim from \eqref{eq:trunc-unified} by cases on $a$.

\medskip
\noindent\textbf{Case 1: $a \leq \overline{\eta}$.}
Since $(1-\rho_1)(a-\overline{\eta}/2)^+ \geq 0$, \eqref{eq:trunc-unified} gives
\[
    \Bigl(b-\frac{\overline{\eta}}{2}\Bigr)^{\!+} - \Bigl(a-\frac{\overline{\eta}}{2}\Bigr)^{\!+}
    \leq  \Bigl(b - \rho_1 a - r_0\Bigr)^{\!+},
\]
which matches the RHS of \Cref{lem:wcmdp:drift-Vds-truncate-term} because $\mathds{1}\!\left\{a>\overline{\eta}\right\}=0$.

\medskip
\noindent\textbf{Case 2: $a > \overline{\eta}$.}
Then $(a-\overline{\eta}/2)^+ = a - \overline{\eta}/2 > \overline{\eta}/2$, so $(1-\rho_1)(a-\overline{\eta}/2)^+ > (1-\rho_1)\overline{\eta}/2 = r_0$.
Hence \eqref{eq:trunc-unified} gives
\[
    \Bigl(b-\frac{\overline{\eta}}{2}\Bigr)^{\!+} - \Bigl(a-\frac{\overline{\eta}}{2}\Bigr)^{\!+}
    \leq  \Bigl(b - \rho_1 a - r_0\Bigr)^{\!+} - r_0,
\]
which matches the RHS of \Cref{lem:wcmdp:drift-Vds-truncate-term} because $\mathds{1}\!\left\{a>\overline{\eta}\right\}=1$.

    \begin{equation}\label{eq:wcmdp:drift-locapprox-claim}

Here the term $2\lambda_{Q}K_{g}$ comes from bounding the left-hand side by $2\lambda_{Q}\left\lVert\bm{g}\right\rVert_\infty$ on the event $D^{\textup{LP}}_t \neq [N]$, and the term $\tilde{K}/N$ comes from bounding the rounding error term in \Cref{lem:wcmdp:olc-transition} by $\left\lVertY_t - y_t\right\rVert_1\,\left\lVertQ\right\rVert_\infty\left\lVert\bm{g}\right\rVert_\infty$ on the event $D^{\textup{LP}}_t = [N]$; both bounds use the property $\left\lVert\bm{g}\right\rVert_\infty \leq K_{g}$ from \Cref{lem:wcmdp:inst-reward}.

\end{aligned}

By \Cref{lem:wcmdp:hw-hu-properties}, $h_U(x, D)$ is Lipschitz continuous in $D$, so $h_U(x, D^{\textup{LP}})$ changes by at most $2\lambda_U^{1/2}(1-m(D^{\textup{LP}}))$ when we replace $D^{\textup{LP}}$ with $[N]$. Consequently, 
\[
\begin{aligned}
V_1(\sigma) &\geq h_U(x,D^{\textup{LP}}) + 2\lambda_U^{1/2}(1-m(D^{\textup{LP}})) \\
    &\geq h_U(x,[N]) \\
    &= \left\lVertx([N])-\mu^*\right\rVert_U.
\end{aligned}
\]

\medskip
\textbf{A preliminary bound for proving \eqref{eq:wcmdp:Vds-mean} and \eqref{eq:wcmdp:Vds-tail}.}
We prove the following bound for constants $\rho_1\in(0,1)$ and $K_{\mathrm{drift}}>0$, both independent of $N$ and specified below:
\begin{align}
    & V_1(\Sigma_{t+1}) - \rho_1V_1(\Sigma_t) \nonumber \\
    & \quad \leq \big(h_U(X_{t+1},D^{\textup{LP}}_t) - \rho_{u}h_U(X_t,D^{\textup{LP}}_t)\big) \nonumber \\
    & \qquad + \big(h_W(X_{t+1},D^{{{\bar{\pi}}^*}}_t) - \rho_{w}h_W(X_t,D^{{{\bar{\pi}}^*}}_t)\big) \label{eq:wcmdp:calc:v-diff-final-1} \\
    & \qquad + 4\big(\lambda_U^{1/2}+\lambda_W^{1/2}\big)m(D^{\textup{LP}}_t\backslashD^{\textup{LP}}_{t+1}) \nonumber \\
    & \qquad + \frac{K_{\mathrm{drift}}}{N}. \label{eq:wcmdp:calc:v-diff-final-2}
\end{align}
To show this, we start with the decomposition:
\begin{align}
    & V_1(\Sigma_{t+1})-V_1(\Sigma_t) \nonumber \\
    & \quad = \big(V_1(X_{t+1},D^{\textup{LP}}_t,D^{{{\bar{\pi}}^*}}_t)-V_1(X_t,D^{\textup{LP}}_t,D^{{{\bar{\pi}}^*}}_t)\big) \label{eq:two-set:thm:state-change-diff-term} \\
    & \qquad + \big(V_1(X_{t+1},D^{\textup{LP}}_{t+1},D^{{{\bar{\pi}}^*}}_{t+1})-V_1(X_{t+1},D^{\textup{LP}}_t,D^{{{\bar{\pi}}^*}}_t)\big). \label{eq:two-set:thm:set-change-diff-term}
\end{align}
We refer to the difference term in right-hand side of \eqref{eq:two-set:thm:set-change-diff-term} the \emph{set-update term}, and the difference term in \eqref{eq:two-set:thm:state-change-diff-term} the \emph{state-transition term}. 
We calculate these terms separately. 

For the state-transition term in \eqref{eq:two-set:thm:state-change-diff-term}, it follows directly from the definition of $V_1$ that 
\begin{align}
    & V_1(X_{t+1},D^{\textup{LP}}_t,D^{{{\bar{\pi}}^*}}_t) - V_1(X_t,D^{\textup{LP}}_t,D^{{{\bar{\pi}}^*}}_t) \nonumber \\
    & \quad \leq \big(h_U(X_{t+1},D^{\textup{LP}}_t) - h_U(X_t,D^{\textup{LP}}_t)\big) \nonumber \\
    & \qquad + \big(h_W(X_{t+1},D^{{{\bar{\pi}}^*}}_t) - h_W(X_t,D^{{{\bar{\pi}}^*}}_t)\big) \label{eq:two-set:thm:state-transition-term-1}
\end{align}

For the set-update term in \eqref{eq:two-set:thm:set-change-diff-term}, we can derive a bound fully in terms of the sizes of sets, using the Lipschitz continuity of $h_W(x, D)$ and $h_U(x, D)$ with respect to $D$ given in \eqref{eq:wcmdp:hw:lipschitz} of \Cref{lem:wcmdp:hw-hu-properties}:
\begin{align}
    & V_1(X_{t+1},D^{\textup{LP}}_{t+1},D^{{{\bar{\pi}}^*}}_{t+1}) - V_1(X_{t+1},D^{\textup{LP}}_t,D^{{{\bar{\pi}}^*}}_t) \nonumber \\
    & \quad = h_U(X_{t+1},D^{\textup{LP}}_{t+1}) - h_U(X_{t+1},D^{\textup{LP}}_t) \nonumber \\
    & \qquad + h_W(X_{t+1},D^{{{\bar{\pi}}^*}}_{t+1}) - h_W(X_{t+1},D^{{{\bar{\pi}}^*}}_t) \nonumber \\
    & \qquad - L_1\big(m(D^{\textup{LP}}_{t+1})-m(D^{\textup{LP}}_t)\big) \nonumber \\
    & \quad \leq 2\lambda_U^{1/2}\big(m(D^{\textup{LP}}_{t+1}\backslashD^{\textup{LP}}_t) + m(D^{\textup{LP}}_t\backslashD^{\textup{LP}}_{t+1})\big) \nonumber \\
    & \qquad + 2\lambda_W^{1/2}\big(m(D^{{{\bar{\pi}}^*}}_{t+1}\backslashD^{{{\bar{\pi}}^*}}_t) + m(D^{{{\bar{\pi}}^*}}_t\backslashD^{{{\bar{\pi}}^*}}_{t+1})\big) \label{eq:two-set:thm:set-update-term-1} \\
    & \qquad - L_1\big(m(D^{\textup{LP}}_{t+1}\backslashD^{\textup{LP}}_t)-m(D^{\textup{LP}}_t\backslashD^{\textup{LP}}_{t+1})\big).
\end{align}
Next, we show that 
\begin{equation}\label{eq:two-set:thm:da-diff-bound}
\begin{aligned}
& m(D^{{{\bar{\pi}}^*}}_{t+1}\backslashD^{{{\bar{\pi}}^*}}_t) + m(D^{{{\bar{\pi}}^*}}_t\backslashD^{{{\bar{\pi}}^*}}_{t+1}) \\
    & \quad \leq 2m(D^{\textup{LP}}_{t+1}\backslashD^{\textup{LP}}_t) \\
    & \qquad - \beta\big(m(D^{\textup{LP}}_{t+1})-m(D^{\textup{LP}}_t)\big) + \frac{1}{N}.
\end{aligned}
\end{equation}
We discuss based on whether $D^{{{\bar{\pi}}^*}}_{t+1} \supseteq D^{{{\bar{\pi}}^*}}_{t} \backslash D^{\textup{LP}}_{t+1}$ or $D^{{{\bar{\pi}}^*}}_{t+1} \subseteq D^{{{\bar{\pi}}^*}}_{t} \backslash D^{\textup{LP}}_{t+1}$. 
\begin{itemize}
    \item If $D^{{{\bar{\pi}}^*}}_{t+1} \supseteq D^{{{\bar{\pi}}^*}}_{t} \backslash D^{\textup{LP}}_{t+1}$, because $D^{{{\bar{\pi}}^*}}_t$ is disjoint from $D^{\textup{LP}}_t$, it is not hard to see that
    \begin{equation}\label{eq:two-set:thm:set-update-term-2}
\begin{aligned}
D^{{{\bar{\pi}}^*}}_t\backslashD^{{{\bar{\pi}}^*}}_{t+1} &\subseteq D^{\textup{LP}}_{t+1}\backslashD^{\textup{LP}}_t, \\
    m(D^{{{\bar{\pi}}^*}}_t\backslashD^{{{\bar{\pi}}^*}}_{t+1}) &\leq m(D^{\textup{LP}}_{t+1}\backslashD^{\textup{LP}}_t).
\end{aligned}
\end{equation}
    Moreover, by the definition of $D^{{{\bar{\pi}}^*}}_t$, $m(D^{{{\bar{\pi}}^*}}_t) \geq \beta (1-m(D^{\textup{LP}}_{t})) - 1/N$ and  $m(D^{{{\bar{\pi}}^*}}_{t+1}) \leq \beta (1-m(D^{\textup{LP}}_{t+1}))$, so 
    \begin{align}
    & m(D^{{{\bar{\pi}}^*}}_{t+1}\backslashD^{{{\bar{\pi}}^*}}_t) \nonumber \\
    & \quad = m(D^{{{\bar{\pi}}^*}}_t\backslashD^{{{\bar{\pi}}^*}}_{t+1}) + m(D^{{{\bar{\pi}}^*}}_{t+1}) - m(D^{{{\bar{\pi}}^*}}_t) \nonumber \\
    & \quad \leq m(D^{\textup{LP}}_{t+1}\backslashD^{\textup{LP}}_t) \nonumber \\
    & \qquad - \beta\big(m(D^{\textup{LP}}_{t+1})-m(D^{\textup{LP}}_t)\big) + \frac{1}{N}. \label{eq:two-set:thm:set-update-term-3}
\end{align}
    Combining \eqref{eq:two-set:thm:set-update-term-2} and \eqref{eq:two-set:thm:set-update-term-3}, we get \eqref{eq:two-set:thm:da-diff-bound}.
    \item If $D^{{{\bar{\pi}}^*}}_{t+1} \subseteq D^{{{\bar{\pi}}^*}}_{t} \backslash D^{\textup{LP}}_{t+1}$, we have $m(D^{{{\bar{\pi}}^*}}_{t+1} \backslash D^{{{\bar{\pi}}^*}}_{t}) = 0$ and
    \begin{align}
    m(D^{{{\bar{\pi}}^*}}_t\backslashD^{{{\bar{\pi}}^*}}_{t+1}) 
    &  = m(D^{{{\bar{\pi}}^*}}_t)-m(D^{{{\bar{\pi}}^*}}_{t+1}) \nonumber \\
    &  = \frac{1}{N}\Bigl(\lfloor\beta N(1-m(D^{\textup{LP}}_t))\rfloor \nonumber \\
    & \quad - \lfloor\beta N(1-m(D^{\textup{LP}}_{t+1}))\rfloor\Bigr) \nonumber \\
    &  \leq \beta\big(m(D^{\textup{LP}}_{t+1})-m(D^{\textup{LP}}_t)\big) + \frac{1}{N}, \nonumber
\end{align}
    which implies \eqref{eq:two-set:thm:da-diff-bound} because $m(D^{\textup{LP}}_{t+1}\backslashD^{\textup{LP}}_t) \geq m(D^{\textup{LP}}_{t+1}) - m(D^{\textup{LP}}_t)$ and $\beta < 1$. 
\end{itemize}
Plugging \eqref{eq:two-set:thm:da-diff-bound} into \eqref{eq:two-set:thm:set-update-term-1}
and rearranging the terms, we get an upper bound for the set-update term:
\begin{align}
    & V_1(X_{t+1},D^{\textup{LP}}_{t+1},D^{{{\bar{\pi}}^*}}_{t+1}) - V_1(X_{t+1},D^{\textup{LP}}_t,D^{{{\bar{\pi}}^*}}_t) \nonumber \\
    & \quad \leq 4\big(\lambda_U^{1/2}+\lambda_W^{1/2}\big)m(D^{\textup{LP}}_t\backslashD^{\textup{LP}}_{t+1}) \nonumber \\
    & \qquad + \frac{2\lambda_W^{1/2}}{N}, \label{eq:two-set:thm:set-update-term-4}
\end{align}
where the terms involving to $m(D^{\textup{LP}}_{t+1}\backslash D^{\textup{LP}}_t)$ have been canceled out. 

Substituting the above calculation into
\eqref{eq:two-set:thm:state-change-diff-term} and \eqref{eq:two-set:thm:set-change-diff-term}, we get
\begin{align}
    & V_1(\Sigma_{t+1}) - V_1(\Sigma_t) \nonumber \\
    & \quad \leq \big(h_U(X_{t+1},D^{\textup{LP}}_t)-h_U(X_t,D^{\textup{LP}}_t)\big) \nonumber \\
    & \qquad + \big(h_W(X_{t+1},D^{{{\bar{\pi}}^*}}_t)-h_W(X_t,D^{{{\bar{\pi}}^*}}_t)\big) \label{eq:two-set:thm:v-diff-term-1} \\
    & \qquad + 4\big(\lambda_U^{1/2}+\lambda_W^{1/2}\big)m(D^{\textup{LP}}_t\backslashD^{\textup{LP}}_{t+1}) \nonumber \\
    & \qquad + \frac{2\lambda_W^{1/2}}{N}. \label{eq:two-set:thm:v-diff-term-2}
\end{align}
By \Cref{lem:wcmdp:sufficient-coverage}, we have
\begin{align}
 V_1(\Sigma_t)
    &\leq h_U(X_t,D^{\textup{LP}}_t) + h_W(X_t,D^{{{\bar{\pi}}^*}}_t) \nonumber \\
    & \qquad + L_1\bigg(K_{C,1}h_W(X_t,D^{{{\bar{\pi}}^*}}_t)+\frac{K_{C,2}}{N}\bigg) \nonumber \\
    & \quad \leq K_{Vh}\cdot\big((1-\rho_{u})h_U(X_t,D^{\textup{LP}}_t) \nonumber \\
    & \qquad + (1-\rho_{w})h_W(X_t,D^{{{\bar{\pi}}^*}}_t)\big) + \frac{L_1K_{C,2}}{N}, \nonumber
\end{align}
where we choose $K_{Vh}>1$, independent of $N$, so that $K_{Vh}(1-\rho_{u})\geq1$ and $K_{Vh}(1-\rho_{w})\geq1+L_1K_{C,1}$. We set $\rho_1=1-1/K_{Vh}\in(0,1)$. Multiplying the preceding bound by $1-\rho_1=1/K_{Vh}$ gives
\begin{equation}\label{eq:v-h-bound}
\begin{aligned}
& (1-\rho_1)V_1(\Sigma_t) \\
    & \quad \leq (1-\rho_{u})h_U(X_t,D^{\textup{LP}}_t) + (1-\rho_{w})h_W(X_t,D^{{{\bar{\pi}}^*}}_t) \\
    & \qquad + \frac{(1-\rho_1)L_1K_{C,2}}{N}.
\end{aligned}
\end{equation}
Adding \eqref{eq:v-h-bound} to \eqref{eq:two-set:thm:v-diff-term-1}--\eqref{eq:two-set:thm:v-diff-term-2} proves \eqref{eq:wcmdp:calc:v-diff-final-1}--\eqref{eq:wcmdp:calc:v-diff-final-2} with
\[
    K_{\mathrm{drift}}\triangleq 2\lambda_W^{1/2}+(1-\rho_1)L_1K_{C,2}.
\]
