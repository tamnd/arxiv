<!-- theorem-1: body_md -->
\label{theorem:invariant_measure}
For every $\kappa\in\mathbb R$, the unique solution of \eqref{eq:SPDE_skew} admits

\begin{equation}\label{eq:invariant_measure_ABLM}
	\mathbb Q^{\kappa}_{0,0}(\mathrm{d} w)=Z_{\kappa}^{-1}\exp\left(-2\kappa\int_0^1\1_{\mathbb R_-}(w_x)\,\mathrm{d} x\right)\mathbb W_{0,0}(\mathrm{d} w)
\end{equation}
as a reversible probability measure, where $\mathbb W_{0,0}$ is the law of the Brownian bridge and $0<Z_{\kappa}<\infty$ is a normalisation constant.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{theorem:static_convergence}
	One has the following convergence of measures
	\begin{equation}\label{eq:convergence_stationary_meas}
		\mathbb Q_{0,0}^{\kappa}\underset{\kappa\rightarrow+\infty}{\Longrightarrow}\mathbb P_{0,0}^{3}\,,
	\end{equation}
	where $\mathbb P_{0,0}^{3}$ is the law of the 3-Bessel bridge (and of the normalised Brownian excursion).
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{theorem:dynamical_convergence}
	Let $u^{\kappa}$ be the solution of \eqref{eq:SPDE_skew} started from $u_0^{\kappa}\overset{\mathrm{law}}{=}\mathbb Q_{0,0}^{\kappa}$ and let $u^+$ be the solution of \eqref{eq:SPDE_NP} started from $u^+_0\overset{\mathrm{law}}{=}\mathbb P_{0,0}^{3}$. One has the following convergence in law

	\end{equation*}

<!-- proof-1: body_md -->
By~\cref{lemma:equality_measures} we know that $\mathbb P_{0,0}^{f_\varepsilon}$ coincides with $\mathbb Q_{0,0}^{\kappa}$ provided $\kappa = \varepsilon^{-2}/ 4$. We thus need to prove that $\mathbb P_{0,0}^{f_\varepsilon}$ converges to $\mathbb P^3_{0,0}$ as $\varepsilon \downarrow 0$. This is carried out below.\\
	\textbf{Convergence of the finite dimensional distributions.} Fix $n\geqslant1$, $t_1<\dots<t_n\in(0,1)$ and $a_1<b_1,\dots,a_n<b_n\in(0,\infty)$. Then
	\begin{equation*}
	\begin{split}
		&\mathbb P_{0,0}^{f_\varepsilon}\left(w_{t_1}\in [a_1,b_1],\dots,w_{t_n}\in [a_n,b_n]\right)\\
		=&\int_{a_1}^{b_1}\dots\int_{a_n}^{b_n}p^{\varepsilon}_{t_1}(0,x_1)\dots p^{\varepsilon}_{t_n-t_{n-1}}(x_{n-1},x_n)\,\frac{p^{\varepsilon}_{1-t_n}(x_n,0)}{p^{\varepsilon}_1(0,0)}\, \mathrm{d} x_1\dots\mathrm{d} x_n\\
		\underset{\varepsilon\downarrow0}{\longrightarrow}&\int_{a_1}^{b_1}\dots\int_{a_n}^{b_n}p^{3-\mathrm{Bessel}}_{t_1}(0,x_1)\dots p^{3-\mathrm{Bessel}}_{t_n-t_{n-1}}(x_{n-1},x_n)\,\frac{p^{3-\mathrm{Bessel}}_{1-t_n}(x_n,0)}{p^{3-\mathrm{Bessel}}_1(0,0)}\, \mathrm{d} x_1\dots\mathrm{d} x_n\\
		=&\mathbb P^3_{0,0}\left(w_{t_1}\in [a_1,b_1],\dots,w_{t_n}\in [a_n,b_n]\right)
	\end{split}
	\end{equation*}
	by the Dominated Convergence Theorem and~\cref{lemma:transition_densities_convergence}. This establishes the convergence of the marginals at times $0<t_1 < \cdots < t_n<1$ on any product of closed intervals of $(0,\infty)$. By a monotone class argument, this extends to any Borel set of $(0,\infty)^n$. Since for all $t\in (0,1)$, $\mathbb P^3_{0,0}$-a.s.~$w_t > 0$, the convergence remains true for any Borel set of $\mathbb{R}^n$. We have thus proven the convergence of the marginals at times $t_1 < \cdots <t_n$, provided $t_1 > 0$ and $t_n <1$. The extension to $t_1= 0$ and/or $t_n = 1$ is straightforward since under either $\mathbb P_{0,0}^{f_\varepsilon}$ or $\mathbb P^3_{0,0}$, $w_0=w_1=0$ a.s.
	
	\smallskip
	
	\noindent \textbf{Tightness.} Tightness is a consequence of~\cref{lemma:tightness}. By the Markov inequality, for every $K>0$, for all $p\geqslant1$, for every $\varepsilon\in(0,1]$,
	\[
		\mathbb P_{0,0}^{f_\varepsilon}[\|w\|_{\mathcal C^q}> K]\leqslant K^{-p}\,\mathbb P_{0,0}^{f_\varepsilon}[\|w\|_{\mathcal C^q}^p]
	\]
	for some $q>0$ given by~\cref{lemma:tightness} such that $\sup_{\varepsilon\in(0,1]}\mathbb P_{0,0}^{f_\varepsilon}[\|w\|_{\mathcal C^q}^p]<\infty$. We conclude using the Arzelà-Ascoli theorem, taking $K$ large enough.

<!-- lemma-1: body_md -->
\label{lemma:law_of_X_eps}
	Fix $\varepsilon>0$. For any $t\in [0,1]$, the law of $X^{\varepsilon}$ satisfies
	\begin{equation}\label{eq:law_of_X_eps}
		\left.\frac{\mathrm{d} \mathbb P_x^{f_\varepsilon}}{\mathrm{d} \mathbb W_x}\right|_{\mathcal F_t}(w):=\exp\left(
		                                   
		F_\varepsilon(w_t)-F_\varepsilon(x)-\frac{\varepsilon^{-2}}{2}\int_0^t\1_{(w_s\leqslant0)}\,\mathrm{d} s\right)\,.
	\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Recall that under $\mathbb W_x$ the canonical process $w$ is a Brownian motion starting from $x$. We set $L_t=\int_0^tf_\varepsilon(w_s)\,\mathrm{d} w_s$.
	As $f_\varepsilon$ is bounded by $\varepsilon^{-1}$, one has that
	\begin{equation*}
		\mathbb E\left[\exp\left(\frac12\langle L\rangle_1\right)\right]=\mathbb E\left[\exp\left(\frac12\int_0^1f_\varepsilon^2(w_s)\,\mathrm{d} s\right)\right]\leqslant\exp\left(\frac{\varepsilon^{-2}}{2}\right)<\infty\,,
	\end{equation*}
	and by Novikov's criterion~\cite[Prop. VIII.1.15]{Revuz1999}, the exponential martingale $\mathcal E(L)_t:=\exp\left(L_t-\frac12\langle L\rangle_t\right)$, $t\in[0,1]$, is a true martingale starting from $1$ at time $0$. Consequently, by Girsanov's Theorem, the process $w'$ defined by 
	\begin{equation}\label{eq:SDE_girsanov}
		w'_t:=w_t-\int_0^tf_\varepsilon(w_s)\,\mathrm{d} s\,,\quad t\in[0,1]\,,
	\end{equation}
	is a Brownian motion under $\tilde{\mathbb W}_x(\mathrm{d} w) := \mathcal E(L)_1(w) \, \mathbb W_x(\mathrm{d} w)$. In particular under $\tilde{\mathbb W}_x$, $w$ has the same law as the solution $X^\varepsilon$ of~\eqref{eq:EDS_eps}. To conclude the proof, we only need to prove that almost surely for any $t\in [0,1]$
	\begin{equation}\label{eq:RN_density_after_ito_trick}
		\mathcal E(L)_t(w)=\exp\left(                                   
		F_\varepsilon(w_t)-F_\varepsilon(x)-\frac{\varepsilon^{-2}}{2}\int_0^t\1_{(w_s\leqslant0)}\,\mathrm{d} s\right)\,.
	\end{equation}
	Recall that
	\begin{equation}\label{eq:RN_density_before_ito_trick}
		\mathcal E(L)_t(w)=\exp\left(\int_0^t f_\varepsilon(w_s)\,\mathrm{d} w_s-\frac12\int_0^t f_\varepsilon^2(w_s)\,\mathrm{d} s\right)\,.
	\end{equation}
	The function $f_\varepsilon$ is $C^1$ except at $x=0$ where the derivative has a discontinuity:
	$$ f_\varepsilon'(x) = -\frac{\1_{(x>0)}}{(\varepsilon+x)^2}\;.$$
	Since $f_\varepsilon'\leqslant 0$, we deduce that $F_\varepsilon$ is concave. By the It\^o-Tanaka formula~\cite[Th VI.1.5]{Revuz1999}, we deduce that almost surely for any $t\in [0,1]$
	\begin{equation}\label{eq:ito_trick2}
	F_\varepsilon(w_t)-F_\varepsilon(x) = \int_0^t f_\varepsilon(w_s)\,\mathrm{d} w_s + \frac12\int_0^t f_\varepsilon'(w_s)\,\mathrm{d} s\,.
	\end{equation}
	Notice that for every $x\neq0$,
	\begin{equation}\label{eq:RN_identity}
		f_\varepsilon'(x)+f_\varepsilon^2(x)=-\frac{\1_{(x>0)}}{(\varepsilon+x^+)^2}+\frac{1}{(\varepsilon+x^+)^2}=\frac{\1_{(x\leqslant 0)}}{(\varepsilon+x^+)^2}=\varepsilon^{-2}\1_{(x\leqslant 0)}\,.
	\end{equation}
	Plugging~\eqref{eq:ito_trick2} and~\eqref{eq:RN_identity} into~\eqref{eq:RN_density_before_ito_trick},
	we obtain~\eqref{eq:RN_density_after_ito_trick}.

<!-- lemma-5: body_md -->
\label{lemma:holder_tightness_unconditionned}
	For any $\eta\in(0,1/2)$ and any $p\geqslant 2$
	$$\sup_{\varepsilon \in (0,1]} \mathbb P_{0}^{f_\varepsilon}\left[\left(\sup_{0 \leqslant s < t \leqslant 1} \frac{|w_t - w_s|}{|t-s|^\eta}\right)^{p}\right] < \infty\;.$$
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
We denote by $X^\varepsilon$ the solution of
	$$ \mathrm{d} X_t^\varepsilon=f_\varepsilon(X_t^\varepsilon)\,\mathrm{d} t+\mathrm{d} B_t\,\quad X_0^\varepsilon=0\,,$$
	where $B$ is a Brownian motion starting from $0$. It suffices to prove that for any $p\geqslant 2$, there exists $C_p>0$ such that for all $0 \leqslant s \leqslant t \leqslant 1$ and all $\varepsilon \in (0,1]$
	$$ \mathbb{E} [|X_t^\varepsilon - X_s^\varepsilon|^p] \leqslant C_p |t-s|^{p/2}\;.$$
	From now on, we fix $0\leqslant s \leqslant t \leqslant 1$ and $\varepsilon \in (0,1]$ and we establish bounds that hold uniformly in these parameters.
	
	We argue differently according to the sign of $X_s^\varepsilon$. Let $(Y_r,r\geqslant s)$ be the solution of
	$$ \mathrm{d} Y_r = \frac1{Y_r}\, \mathrm{d} r + \mathrm{d} B_r\;,\quad r\geqslant s\;,\qquad Y_s := (X_s^\varepsilon)^+\;.$$
	On the event $\{X_s^\varepsilon \geqslant 0\}$, we have
	$$ B_t-B_s \leqslant X_t^\varepsilon - X_s^\varepsilon \leqslant Y_t - Y_s\;,$$
	and consequently
	$$ |X_t^\varepsilon - X_s^\varepsilon|^p \leqslant |B_t-B_r|^p + |Y_t-Y_s|^p\;.$$
	Since the $3$d-Bessel process has the law of the Euclidean norm of a $3$d-Brownian motion, we deduce that there exists a constant $C>0$, independent of $s,t$ and $\varepsilon$, such that
	$$ \mathbb{E}\left[|X_t^\varepsilon - X_s^\varepsilon|^p \1_{\{X_s^\varepsilon \geqslant 0\}}\right] \leqslant C \, |t-s|^{p/2}\;.$$
	We have thus established the desired bound in the case where $X_s^\varepsilon \geqslant 0$.
	
	Next, we observe that, whatever the sign of $X_s^\varepsilon$, we have
	$$ B_t - B_s \leqslant X_t^\varepsilon - X_s^\varepsilon \leqslant B_t - B_s + \frac{t-s}{\varepsilon}\;,$$
	and therefore there exists a constant $C>0$, independent of $s,t$ and $\varepsilon$, such that
	$$ \mathbb{E}[|X_t^\varepsilon - X_s^\varepsilon|^p] \leqslant C \left(|t-s|^{p/2} + \left(\frac{|t-s|}{\varepsilon}\right)^p\right)\;.$$
	If $|t-s| \leqslant \varepsilon^2$, this provides the desired bound.
	
	It remains to cover the case where $X_s^\varepsilon < 0$ and $|t-s| > \varepsilon^2$. We let $\tau:=\inf\{r\geqslant s: X_r^\varepsilon = 0\}$. Then we write
	$$|X_t^\varepsilon - X_s^\varepsilon| = |X_t^\varepsilon - X_{t\wedge \tau}^\varepsilon| + |X_{t\wedge \tau}^\varepsilon - X_s^\varepsilon|\;.$$
	The strong Markov property and the arguments above show that
	$$\mathbb{E}[|X_t^\varepsilon - X_{t\wedge \tau}^\varepsilon|^p] \leqslant C|t-s|^{p/2}\;.$$
	We thus concentrate on bounding
	$$ \mathbb{E}\left[|X_{t\wedge \tau}^\varepsilon - X_s^\varepsilon|^p \1_{\{X_s^\varepsilon < 0\}}\right] = A_1 + A_2\;,$$
	where
	\[
		A_1:= \mathbb{E}\left[|X_{t\wedge \tau}^\varepsilon - X_s^\varepsilon|^p \1_{\{X_s^\varepsilon < 0; \tau \leqslant t\}}\right],\qquad
		A_2:= \mathbb{E}\left[|X_{t\wedge \tau}^\varepsilon - X_s^\varepsilon|^p \1_{\{X_s^\varepsilon < 0; \tau > t\}}\right]\;.
	\]
	On the event $\{\tau \leqslant t\}$, we have $|X_{t\wedge \tau}^\varepsilon - X_s^\varepsilon| = |X_s^\varepsilon|$. Consequently
	$$ A_1 \leqslant \mathbb{E}[|X_s^\varepsilon|^p \1_{\{X_s^\varepsilon < 0\}}] = \int_{-\infty}^0 |x|^p \, p_s^{\varepsilon}(0,x)\,\mathrm{d} x \,.$$
	Let us prove that
	\begin{equation}\label{eq:claimnow}
	\int_{-\infty}^0 |x|^p \, p_s^{\varepsilon}(0,x)\,\mathrm{d} x\leqslant C \, \varepsilon^p.
	\end{equation}
		By~\eqref{eq:transition_densities} and since $\Phi_t(\lambda,0,x) \leqslant 1$, we obtain
	\begin{align*}
		\int_{-\infty}^0 |x|^p \, p_s^{\varepsilon}(0,x)\,\mathrm{d} x &= \int_{-\infty}^0 |x|^p \, e^{\frac{x}{\varepsilon}} g_t(x) \Phi_t(\lambda;0,x)\,\mathrm{d} x\\
		&\leqslant \varepsilon^p \int_{-\infty}^0 \left(\frac{|x|}{\varepsilon}\right)^p \, e^{\frac{x}{\varepsilon}} g_t(x) \,\mathrm{d} x\\
		&\leqslant \varepsilon^p \left( \sup_{y > 0} y^p \, e^{-y}\right) \int_{-\infty}^0 g_t(x) \,\mathrm{d} x\\
		&\leqslant C \varepsilon^p\;.
	\end{align*}
	
	We turn to $A_2$. First consider the event $\{-\frac{t-s}{2\varepsilon} < X_s^\varepsilon < 0\}$ and note that if $\tau > t$ then necessarily
	$$ B_t-B_s + \frac{t-s}{\varepsilon} < - X_s^\varepsilon\;,$$
	and therefore
	$$ B_t-B_s < -\frac{t-s}{2\varepsilon}\;.$$
	Consequently, setting $\zeta:=\frac{\sqrt{t-s}}{2\varepsilon}$
	\begin{align*}
		\mathbb{E}\left[|X_{t\wedge \tau}^\varepsilon - X_s^\varepsilon|^p \1_{\{-\frac{t-s}{2\varepsilon} < X_s^\varepsilon < 0,\,\tau > t\}}\right] &= \mathbb{E}\left[|X_{t}^\varepsilon - X_s^\varepsilon|^p \1_{\{-\frac{t-s}{2\varepsilon} < X_s^\varepsilon < 0,\,\tau > t\}}\right]\\
		&\leqslant \mathbb{E}\left[\left|B_t-B_s + \frac{t-s}{\varepsilon}\right|^p \1_{\{B_t-B_s < -\frac{t-s}{2\varepsilon}\}}\right]\\
		& = |t-s|^{p/2} \, \mathbb{E}\left[\left|Z + 2\zeta\right|^p \1_{\{Z < -\zeta\}}\right],
	\end{align*}
	where $Z\sim\mathcal N(0,1)$.	Now it remains to estimate
	\[
	\begin{split}
	 \sup_{\zeta>\frac12}\mathbb{E}\left[\left|Z + 2\zeta\right|^p \1_{\{Z < -\zeta\}}\right] & \leqslant C \sup_{\zeta>\frac12}\mathbb{E}\left[(|Z|^p + \zeta^p) \1_{\{Z < -\zeta\}}\right]

	\end{enumerate}

\label{Def:sshe}

Let $u_0\in\mathcal C([0,1])$. A measurable adapted process $u:\,\Omega\times[0,\infty)\times[0,1]\rightarrow\mathbb{R}$ is called a \emph{mild solution} of \eqref{eq:SPDE_skew}
with initial condition $u_0$ if there exists a process $K:\,\Omega\times[0,\infty)\times[0,1]\rightarrow\mathbb{R}$ such that
\begin{enumerate}[(1)]
	\item For every $x\in[0,1]$ and every $t\geqslant0$,
	\begin{equation}\label{eq:sshe}
		u_t(x)=P_tu_0(x)+K_t(x)+\int_0^t\int_0^1p_{t-r}(x,y)W(\mathrm{d} r,\mathrm{d} y)\,,
	\end{equation}
   \item for any sequence of functions $(b^n)_{n\geqslant 0}$ in $\mathcal C_b^\infty$ such that $b^n\to b$ in $\mathcal C^{-1-}$, we have for any $T>0$,
		\begin{equation}\label{Kcurlydef}
		\sup_{t\in[0,T]}\sup_{x\in[0,1]}\left|\int_0^t\int_0^1p_{t-r}(x,y) b^n(u_r(y))\,\mathrm{d} y\,\mathrm{d} r- K_t(x)\right|\to0\quad  \text{in probability as $n\to\infty$.}
		\end{equation}
	\item a.s. the function $u$ is continuous on $[0,\infty)\times[0,1]$.
  \end{enumerate}

<!-- theorem-5: body_md -->
\label{theorem:properties_spde_dirac}
	Let $b\in\mathcal C^{-1}$ and $u_0\in\mathcal C([0,1])$. Then:
	\begin{enumerate}
		\item\label{item:properties_spde_dirac1} The SPDE~\eqref{eq:SHE} has a unique solution in law (in the sense of~\cref{Def:sshe}).
		\item\label{item:properties_spde_dirac2} The unique solution is a Markov process with state space $\mathcal C([0,1])$.
		\item\label{item:properties_spde_dirac3} (stability) If $u^n$ is a solution to~\eqref{eq:SHE} (in the sense of~\cref{theorem:existence_uniqueness_smooth}) with drift $b^n\in\mathcal C_b^{\infty}$ and starting from $u_0$, and if $b^n$ converges to $b$ in $\mathcal C^{-1-}$,
	then we have the convergence in distribution
	\begin{equation*}
		u^n\rightarrow u\,,\quad\textrm{as }\,n\rightarrow\infty\,,
	\end{equation*}
	in $\mathcal C([0,\infty)\times[0,1])$ endowed with the topology of uniform convergence on compact sets.
	\end{enumerate}

<!-- remark-1: body_md -->
\label{remark:approximation}
	The sequence $b^n$ is not explicit in the notation $\mathbb Q^{\kappa,n}_{0,0}$, but will always be given by~\eqref{eq:approximate_b}.
	We could have taken any sequence $b^n$ which converges to $\kappa\delta_0$ in $\mathcal C^{-1-}$ as $n\rightarrow\infty$, such that, uniformly in $n\geqslant0$, the antiderivatives given by~\eqref{eq:anditerivatives} are uniformly bounded above and below.
	Indeed, this implies that the Radon-Nikodym derivatives
	\begin{equation*}
		\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa,n}}{\mathrm{d} \mathbb W_{0,0}}
	\end{equation*}
	are uniformly bounded in $n$. In our case, they are uniformly bounded by $\exp(2\kappa)$.

Let $u^{\kappa,n}$ be the solution to~\eqref{eq:SHE} (in the sense of~\cref{theorem:existence_uniqueness_smooth}) with drift $b^n$ and started from $u^{\kappa,n}_0\overset{\mathrm{law}}{=}\mathbb Q_{0,0}^{\kappa,n}$ (given by~\eqref{eq:approx_reversible_measure}) taken independent from $W$. 	\begin{equation*}

	\end{equation*}
	where $u^\kappa$ is the unique solution to~\eqref{eq:SHE} (in the sense of~\cref{Def:sshe}) with drift $b=\kappa\delta_0$ started from $u_0^\kappa\overset{\mathrm{law}}{=}\mathbb Q_{0,0}^\kappa$.

In this proof, we take $(b^n)$ the sequence given by~\eqref{eq:approximate_b}. We need to prove that for every bounded measurable $f_1, f_2:\mathcal C([0,\infty)\times[0,1])\to\mathbb{R}$ and for every $t>0$,
	\begin{equation}\label{eq:reversible_measure}
		\int\mathbb Q^{\kappa}_{0,0}(\mathrm{d} u_0)\,f_1(u_0) \, \mathbb E[f_2(u_t^{\kappa}(u_0))]=\int\mathbb Q^{\kappa}_{0,0}(\mathrm{d} u_0)\,\mathbb E[f_1(u_t^{\kappa}(u_0))]
		 \, f_2(u_0)\,.
	\end{equation}
	It follows by~\cref{remark:reversible_measure_spde} that for all $t>0$, for all $n\geqslant0$
	\begin{equation}\label{eq:QKN_reversible}
		\int\mathbb Q^{\kappa,n}_{0,0}(\mathrm{d} u_0)\,f_1(u_0) \, \mathbb E[f_2(u_t^{\kappa,n}(u_0))]=\int\mathbb Q^{\kappa,n}_{0,0}(\mathrm{d} u_0)\,\mathbb E[f_1(u_t^{\kappa,n}(u_0))] \, f_2(u_0)\,.
	\end{equation}
	By~\cref{prop:stability}, see in particular~\eqref{eq:stability1} with $F(u)=f_1(u_0)f_2(u_t)$, we can pass to the limit in~\eqref{eq:QKN_reversible} and obtain~\eqref{eq:reversible_measure}.

<!-- proof-11: body_md -->
The result is a consequence of~\cref{theorem:properties_spde_dirac},~\cref{item:properties_spde_dirac3} and the Dominated Convergence Theorem.
	Let $F$ be any bounded continuous function on $\mathcal C([0,\infty)\times[0,1])$. For any $u_0\in \mathcal C([0,\infty)\times[0,1])$, as $n\rightarrow\infty$, $\mathbb E[F(u^{\kappa,n}(u_0))]$ goes to $\mathbb E[F(u^{\kappa}(u_0))]$ by~\cref{theorem:properties_spde_dirac}, and the sequence is uniformly bounded in $n$ ($F$ is bounded). We claim that for $\mathbb W_{0,0}$-almost every $u_0$, as $n\rightarrow\infty$,
	\begin{equation*}
		\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa,n}}{\mathrm{d} \mathbb W_{0,0}}(u_0)\longrightarrow\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa}}{\mathrm{d} \mathbb W_{0,0}}(u_0)
	\end{equation*}
	and that the L.H.S.~is bounded uniformly over all $u_0\in \mathcal C([0,\infty)\times[0,1])$ and all $n\geqslant 1$. Then, we deduce from the Dominated Convergence Theorem for $\mathbb W_{0,0}$ that
	\begin{equation}\label{eq:stability1}
		\int\mathbb Q_{0,0}^{\kappa,n}(\mathrm{d} u_0)\,\mathbb E[F(u^{\kappa,n}(u_0))]=\int\mathbb W_{0,0}(\mathrm{d} u_0)\,\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa,n}}{\mathrm{d} \mathbb W_{0,0}}(u_0)\,\mathbb E[F(u^{\kappa,n}(u_0))]\,,
	\end{equation}
	converges as $n\to\infty$ to
	\begin{equation*}
		\int\mathbb W_{0,0}(\mathrm{d} u_0)\,\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa}}{\mathrm{d} \mathbb W_{0,0}}(u_0)\,\mathbb E[F(u^{\kappa}(u_0))]=\int\mathbb Q_{0,0}^{\kappa,n}(\mathrm{d} u_0)\,\mathbb E[F(u^{\kappa}(u_0))]\,.
	\end{equation*}
	This establishes the asserted result.
	
	It remains to prove the claim. Recall that
	\begin{equation*}
		\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa,n}}{\mathrm{d} \mathbb W_{0,0}}(w) =\frac{\exp\left(2\int_0^1B^n(w_s)\,\mathrm{d} s\right)}{Z_{\kappa,n}}, \ \mbox{ with }\quad Z_{\kappa,n} = \mathbb W_{0,0}\left[\exp\left(2\int_0^1B^n(w_s)\,\mathrm{d} s\right)\right],
	\end{equation*}
	and similarly,
	\begin{equation*}
		\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa}}{\mathrm{d} \mathbb W_{0,0}}(w) =\frac{\exp\left(-2\kappa \int_0^1 \1_{\mathbb{R}_-}(w_s)\mathrm{d} s\right)}{Z_{\kappa}}, \ \mbox{ with }\ Z_{\kappa} = \mathbb W_{0,0}\left[\exp\left(-2\kappa \int_0^1 \1_{\mathbb{R}_-}(w_s)\mathrm{d} s\right)\right]\!.
	\end{equation*}	
	As $n\rightarrow\infty$, $B^n$ goes to $-\kappa\1_{\mathbb{R}_-}$ except at $0$. Indeed, for $x\neq0$, by a change of variables,
	\begin{equation*}
		\int_x^\infty b^n(y)\,\mathrm{d} y=\kappa\int_x^\infty g_{1/n}(y)\,\mathrm{d} y=\kappa\int_{\sqrt{n} x}^{\infty}g_1(y)\,\mathrm{d} y\,.
	\end{equation*}
	By the Dominated Convergence theorem, if $x<0$, the previous integral goes to $\kappa$, and if $x>0$, it goes to $0$.
	Consequently, as $B^n$ is uniformly bounded in $n$, and because for $\mathbb W_{0,0}$-almost every $w$, $\textrm{Leb}(\{t\in[0,1]:\,w_t=0\})=0$, by the Dominated Convergence Theorem, for $\mathbb W_{0,0}$-almost every $w$, as $n$ goes to $\infty$,
	\begin{equation*}
		\int_0^1B^n(w_s)\,\mathrm{d} s\longrightarrow-2\kappa\int_0^1\1_{\mathbb{R}_-}(w_s)\,\mathrm{d} s\,.
	\end{equation*}
	As, for every $n\geqslant0$, $B^n\leqslant0$, then for every $w\in\mathcal C([0,1])$ and $n\geqslant0$,
	\begin{equation*}
		\exp\left(2\int_0^1B^n(w_s)\,\mathrm{d} s\right)\leqslant 1\,.
	\end{equation*}
	As a consequence, by the Dominated Convergence Theorem for $\mathbb W_{0,0}$,
	\begin{equation*}
		Z_{\kappa,n} = \mathbb W_{0,0}\left[\exp\left(2\int_0^1B^n(w_s)\,\mathrm{d} s\right)\right]\longrightarrow\mathbb W_{0,0}\left[\exp\left(-2\kappa\int_0^1\1_{\mathbb{R}_-}(w_s)\,\mathrm{d} s\right)\right]=Z_\kappa\,.
	\end{equation*}
	Finally, as for every $n\geqslant0$, $-\kappa\leqslant B^n$, then
	\begin{equation*}
		\mathbb W_{0,0}\left[\exp\left(2\int_0^1B^n(w_s)\,\mathrm{d} s\right)\right]\geqslant\exp(-2\kappa)\,.
	\end{equation*}
	This implies that for $\mathbb W_{0,0}$-almost every $u_0$, as $n\rightarrow\infty$,
	\begin{equation*}
		\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa,n}}{\mathrm{d} \mathbb W_{0,0}}(u_0)\longrightarrow\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa}}{\mathrm{d} \mathbb W_{0,0}}(u_0)
	\end{equation*}
	and uniformly in $n\geqslant0$ and $u_0$,
	\begin{equation*}
		\frac{\mathrm{d} \mathbb Q_{0,0}^{\kappa,n}}{\mathrm{d} \mathbb W_{0,0}}(u_0)\leqslant\exp(2\kappa)\,,
	\end{equation*}
	and the claim is proved.

<!-- definition-3: body_md -->
\label{Def:NP}
	A pair $(u,\eta)$ is said to be a solution of \eqref{eq:SPDE_NP} if:
	\begin{enumerate}[(i)]
		\item \label{item:NP_1} $u$ is a nonnegative, continuous process with $u(t,0)=u(t,1)=0$, for all $t\geqslant0$ a.s.
		\item \label{item:NP_2} $\eta(\mathrm{d} t,\mathrm{d} x)$ is a random measure on $[0,\infty)\times(0,1)$ such that $\eta([0,T]\times(\varepsilon,1-\varepsilon))<\infty$ for all $\varepsilon>0$ and for all $T>0$ a.s.
		\item \label{item:NP_3} For all $t\geqslant0$ and all $\varphi\in\mathcal C^\infty([0,1])$ such that $\varphi(0)=\varphi(1)=0$,
			\begin{align*}
				\mathrm{a.s.},\quad(u_t,\varphi)_{L^2}=(u_0,\varphi)_{L^2}+\int_0^t(u_s,\frac12\varphi'')_{L^2}\,\mathrm{d} s+\int_0^t\int_0^1\varphi(x)\,\eta(\mathrm{d} s,\mathrm{d} x)+W_t(\varphi)\,.
			\end{align*}
		\item \label{item:NP_4} $\int u\,\mathrm{d}\eta=0$ a.s.
	\end{enumerate}

\emph{Control of the increments in time}.
	We prove that for all $T>0$, there exists $p>0$ such that
	\begin{equation}\label{eq:tightness_criterion}
		\lim_{h\downarrow0}\limsup_{\kappa\rightarrow+\infty}\mathbb E\left[\vphantom{\sup_{s,t\leqslant T}}\right.\sup_{\substack{s,t\leqslant T\\|t-s|\leqslant h}}\|u^{\kappa}_t-u^{\kappa}_s\|^p_{\mathcal C([0,1])}\left.\vphantom{\sup_{s,t\leqslant T}}\right]=0\,.
	\end{equation}

	Let $(b^n)$ be the sequence given by~\eqref{eq:approximate_b}. For every $\kappa>0$, $n\geqslant0$, we denote by $u^{\kappa,n}$ the solution of~\eqref{eq:SHE} in the sense of~\cref{theorem:existence_uniqueness_smooth}, started from $\mathbb Q_{0,0}^{\kappa,n}$ given by~\eqref{eq:approx_reversible_measure}.
	We fix $\kappa>0$, $n\geqslant0$, and $T>0$. The Lyons-Zheng decomposition \cite{LyonsZheng} writes for all $h\in H$, for all $0\leqslant t\leqslant T$,

	\begin{equation*}
		(h,u^{\kappa,n}_t-u_0^{\kappa,n})_H=\frac12 M_t+\frac12 (N_T-N_t)\,,
	\end{equation*}
	where $M$ (resp.~$N$) is a martingale in the natural filtration of $u^{\kappa,n}$ (resp.~that of $(u^{\kappa,n}_{T-t})_{t\in[0,T]}$).
	Moreover, the quadratic variations are both equal to

	\begin{equation*}
		\langle M\rangle_t=\langle N\rangle_t=t\|h\|_{H}^2\,.
	\end{equation*}

	\noindent By the Burkholder-Davis-Gundy inequality and the Minkowski inequality, there exists $c_p\in(0,\infty)$ such that

	\begin{equation*}
		\mathbb E\left[|(h,u^{\kappa,n}_t-u^{\kappa,n}_s)_H|^p\right]^{1/p}\leqslant c_p\|h\|_H(t-s)^{1/2}.
	\end{equation*}

	\noindent By Fatou's lemma and the convergence of~\cref{prop:stability}, this implies that
	\begin{equation}\label{eq:tightness_u_1}
		\mathbb E\left[|(h,u^{\kappa}_t-u^{\kappa}_s)_H|^p\right]^{1/p}\leqslant\liminf_{n\rightarrow\infty}\mathbb E\left[|(h,u^{\kappa,n}_t-u^{\kappa,n}_s)|^p\right]^{1/p}\leqslant c_p\|h\|_H(t-s)^{1/2}
	\end{equation}
	which implies, by the Minkowski inequality for $p\geqslant2$,
	\begin{equation}\label{eq:tightness_u_2}
	\begin{split}
		\mathbb E\left[\|u^{\kappa}_t-u^{\kappa}_s\|_{H^{-\beta}}^p\right]^{2/p}&=\mathbb E\left[\left(\sum_{k\geqslant1}k^{-2\beta}|(e_k,u_t^{\kappa}-u_s^{\kappa})_H|^2\right)^{p/2}\right]^{2/p}\\
		&\leqslant\sum_{k\geqslant1}k^{-2\beta}\mathbb E[|(e_k,u^\kappa_t-u^\kappa_s)_H|^p]^{2/p}\leqslant C(t-s)
	\end{split}
	\end{equation}
	since $\beta>1/2$. Note that $C$ does not depend on $\kappa > 0$, nor on $s,t$.

	\noindent Let $0\leqslant s\leqslant t$, $p\geqslant1$ and $\eta<1/2$. By the triangle inequality, the Minkowski inequality, and stationarity at the second line,
	\begin{equation*}
	\begin{split}
		\mathbb E[\|u_t^{\kappa}-u_s^{\kappa}\|_{\mathcal C^{\eta}}^p]^{1/p}&\leqslant\mathbb E[\|u_s^{\kappa}\|_{\mathcal C^{\eta}}^p]^{1/p}+\mathbb E[\|u_t^{\kappa}\|_{\mathcal C^{\eta}}^p]^{1/p}\\
		&\leqslant 2\mathbb E[\|u_0^{\kappa}\|_{\mathcal C^{\eta}}^p]^{1/p}=2\mathbb Q_{0,0}^{\kappa}\left[\|w\|_{\mathcal C^{\eta}}^p\right]^{1/p}
	\end{split}
	\end{equation*}
	which is finite uniformly over all $\kappa > 1$ by~\cref{lemma:tightness}. This leads to
	\begin{equation}\label{eq:tightness_u_3}
		\sup_{\kappa > 1}\sup_{0\leqslant s\leqslant t}\mathbb E\left[\left\|u^{\kappa}_t-u^{\kappa}_s\right\|^p_{\mathcal C^{\eta}}\right]<\infty\,.
	\end{equation}

	\noindent\cref{lemma:interpolation} implies via Hölder's inequality that there exist $\gamma,\theta\in(0,1)$ such that for all $p$ large enough,

	\begin{equation*}
		\mathbb E\left[\left\|u^{\kappa}_t-u^{\kappa}_s\right\|^p_{\mathcal C^{\gamma}}\right]\lesssim(t-s)^{\frac{p(1-\theta)}{2}}\,,
	\end{equation*}
	uniformly over all $\kappa > 1$ and $0\leqslant s\leqslant t\leqslant T$. Applying Kolmogorov's continuity theorem, this implies that for all $\nu\in(0,(1-\theta)/2)$ and all $p\geqslant 1$, we have
	\begin{equation*}
		\sup_{\kappa>1}\mathbb E\left[\sup_{s\neq t\in[0,T]}\frac{\left\|u^{\kappa}_t-u^{\kappa}_s\right\|^p_{\mathcal C^{\gamma}}}{|t-s|^{\nu p}}\right]<\infty\,.
	\end{equation*}
	This implies that~\eqref{eq:tightness_criterion} is fulfilled.

	\noindent\emph{Tightness of the one-dimensional marginals}.
	The processes are stationary, so it suffices to prove tightness of $(\mathbb Q_{0,0}^{\kappa})_{\kappa>1}$.
	The latter is a direct consequence of~\cref{lemma:tightness}.

\label{Lemma:TightMandphi}
	Let $(\eta^i)_{i\in I}$ be a family of random elements of $\mathbb M$. Assume that for every function $\varphi\in \mathcal C^\infty_0([0,1])$, $(X^i)_{i\in I}$ is tight in $\mathcal C([0,\infty),\mathbb{R})$ where
	$$ X_t^i := \int_{[0,t]\times[0,1]} \varphi(x) \, \eta^i(\mathrm{d} s, \mathrm{d} x)\;,\quad t\geqslant 0\;.$$
	Then $(\eta^i)_{i\in I}$ is tight in $\mathbb M$ and any converging subsequence $(\eta^{i_k})_{k\geqslant 1}$ with limit, say, $\eta$, satisfies the following property: for any $\varphi\in \mathcal C^\infty_0([0,1])$,  $(X^{i_k})_{k\geqslant 1}$ converges in law to the continuous process
	$$ X_t := \int_{[0,t]\times[0,1]} \varphi(x) \, \eta(\mathrm{d} s, \mathrm{d} x)\;,\quad t\geqslant 0\;.$$

<!-- proof-13: body_md -->
To prove that $(\eta^i)_{i\in I}$ is tight, it suffices to prove that for every $\psi \in \mathcal C_c^\infty([0,\infty)\times[0,1],\mathbb R)$, the family
	$$ \int_{[0,\infty)\times [0,1]} \psi(s,x) \, \eta^i(\mathrm{d} s,\mathrm{d} x)\;,\quad i\in I\;,$$
	is tight in $\mathbb{R}$. Given such a function $\psi$, one can find a non-negative $\varphi \in \mathcal C^\infty_0([0,1])$ such that $|\psi(t,x) | \leqslant \varphi(x)$ for all $t\geqslant 0 $ and $x\in[0,1]$, so the assertion of the statement ensures tightness of $(\eta^i)_{i\in I}$.
	
	For simplicity, we denote by $(\eta^n)_{n\geqslant 0}$ a converging subsequence and we let $\eta$ be its limit. Fix $\varphi$ as in the statement. Consider a further subsequence $(\eta^{n_k},X^{n_k})_k$ that converges in law to, say, $(\eta,X)$. If we show that
	$$ X_t = \int_{[0,t]\times[0,1]} \varphi(x) \, \eta(\mathrm{d} s, \mathrm{d} x)\;,\quad t\geqslant 0\;,$$
	then we will deduce that all converging subsequences of $(X^n)_n$ have the same limit, and therefore that $(\eta_n,X_n)$ converges to $(\eta,X)$.
	
	We use the Skorokhod representation theorem and suppose that the convergence in law of $(\eta^{n_k},X^{n_k})_k$ holds a.s.~on some other probability space. Let us suppose that $\varphi\geqslant0$, the general case is treated by writing $\varphi=\varphi^+-\varphi^-$, where $\varphi^+$ and $\varphi^-$ both remain in $\mathcal C^\infty_0([0,1])$.
	For every $t>0$, we set
	\begin{equation}\label{eq:Xtilde}
			\widetilde{X}_t=\int_{[0,t]\times[0,1]}\varphi(x)\,\eta(\mathrm{d} s,\mathrm{d} x)\,.
	\end{equation}
	We aim at proving that the process $\widetilde{X}$ is a.s. continuous and that, as processes, $\widetilde{X}=X$.
		
		\noindent \emph{Proof that $\widetilde{X}$ is continuous}. For $a>0,\varepsilon>0$, we let $\chi_{a,\varepsilon}:\mathbb{R}\to [0,1]$ be any continuous function such that $\1_{[a,a+\varepsilon]}\leqslant\chi_{a,a+\varepsilon}\leqslant\1_{[a-\varepsilon,a+2\varepsilon]}$.
		Let $\delta>0$, $t>0$,
		\begin{equation*}
			|\widetilde{X}_{t+\delta}-\widetilde{X}_{t}|\leqslant\int_{[0,\infty)\times(0,1)}\1_{[t,t+\delta]}(s)\varphi(x)\,\eta(\mathrm{d} s,\mathrm{d} x)\leqslant\int_{[0,\infty)\times(0,1)}\chi_{t,t+\delta}(s)\varphi(x)\,\eta(\mathrm{d} s,\mathrm{d} x)\,.
		\end{equation*}
		Because $\chi_{t,t+\delta}\otimes\varphi\in\mathcal C_c([0,\infty)\times(0,1))$, it follows by the definition of the topology on $\mathbb M$ that one has the a.s. limit
		\begin{equation*}
			\int_{[0,\infty)\times(0,1)}\chi_{t,t+\delta}(s)\varphi(x)\,\eta(\mathrm{d} s,\mathrm{d} x)=\lim_{k\rightarrow\infty}\int_{[0,\infty)\times(0,1)}\chi_{t,t+\delta}(s)\varphi(x)\,\eta^{n_k}(\mathrm{d} s,\mathrm{d} x)\,.
		\end{equation*}
		Furthermore, for every $k\geqslant0$,
		\begin{equation*}
			\begin{split}
				\int_{[0,\infty)\times(0,1)}\chi_{t,t+\delta}(s)\varphi(x)\,\eta^{n_k}(\mathrm{d} s,\mathrm{d} x)&\leqslant\int_{[0,\infty)\times(0,1)}\1_{t-\delta,t+2\delta}(s)\varphi(x)\,\eta^{n_k}(\mathrm{d} s,\mathrm{d} x)\\
				&=X^{n_k}_{t+2\delta}-X^{n_k}_{t-\delta}\,.
			\end{split}
		\end{equation*}
		Also, by continuity of the map $\mathcal C([0,\infty))\ni x\mapsto x_{t+2\delta}-x_{t-\delta}$,
		\begin{equation*}
			\lim_{k\rightarrow\infty}\,X^{n_k}_{t+2\delta}-X^{n_k}_{t-\delta}=X_{t+2\delta}-X_{t-\delta}\,.
		\end{equation*}
		Finally,
		\begin{equation*}
			\limsup_{\delta\downarrow0}|\widetilde{X}_{t+\delta}-\widetilde{X}_{t}|\leqslant \limsup_{\delta\downarrow0}|X_{t+2\delta}-X_{t-\delta}|=0\,.
		\end{equation*}
		This proves right continuity of $\widetilde{X}$, left continuity is proved similarly.
		
		\noindent\emph{Proof that $X=\widetilde X$}. For $a>0,\varepsilon>0$, we let $\lambda_{a,\varepsilon}:\mathbb{R}\to[0,1]$ be any continuous function such that $\1_{[0,a]}\leqslant\lambda_{a,\varepsilon}\leqslant\1_{[0,a+\varepsilon]}$.
		For any $t>0$
		\begin{equation*}
			|X_t-\widetilde X_t|=\lim_{k\rightarrow\infty}|X^{n_k}_t-\widetilde X_t|\,.
		\end{equation*}
		For $n\geqslant0$, for every $\varepsilon>0$,
		\begin{equation*}
			\begin{split}
				|X^{n_k}_t-\widetilde X_t|=&|\eta^{n_k}(\1_{[0,t]}\otimes\varphi)-\eta(\1_{[0,t]}\otimes\varphi)|\\

\label{Prop:MildLimitPoint}
	The collection $(u^{\kappa,n},\eta^{\kappa,n}, W)_n$ is tight in
	$$ \mathcal C([0,\infty)\times [0,1]) \times \mathbb M \times \mathcal C([0,\infty), \mathcal S'([0,1]))$$

Recall that $u^{\kappa,n}$ is a stationary solution of~\eqref{eq:SHE} with $b^n$ given by~\eqref{eq:approximate_b}. We define a random measure on $[0,\infty)\times(0,1)$
	\[
	\eta^{\kappa,n}(\mathrm{d} t,\mathrm{d} x):=b^n(u_t^{\kappa,n})\,\mathrm{d} t\,\mathrm{d} x\,.
	\]
	Since $b^n$ is a bounded function, this measure belongs to $\mathbb M$. \cref{prop:stability} implies that the family $(u^{\kappa,n})_n$ is tight. In addition, $W$ is a space-time white noise, so its law does not depend on $\kappa$ and $n$. The weak formulation~\eqref{eq:she_weak} of the SPDE satisfied by $u^{\kappa,n}$ yields for any $\varphi\in\mathcal C^\infty_0([0,1])$
	\begin{equation}\label{Eq:tildeukappan}
		(\tilde{u}^{\kappa,n}_t,\varphi)-(\tilde{u}^{\kappa,n}_0,\varphi)-\int_0^t(\tilde{u}^{\kappa,n}_s,\frac12\varphi'')\,\mathrm{d} s-X^{\kappa,n}_t-W_t(\varphi) = 0\;,
	\end{equation}
	where
	$$ X^{\kappa,n} := \int_0^t \varphi(x) \, \eta^{\kappa,n}(\mathrm{d} s, \mathrm{d} x)\;,\quad t\geqslant 0\,, n\geqslant 1\;.$$
	From this identity and the previous arguments, we deduce that $(X^{\kappa,n})_n$ is tight in $\mathcal C([0,\infty),\mathbb{R})$. In turn, by~\cref{Lemma:TightMandphi} we deduce that $(\eta^{\kappa,n})_{n\geqslant 1}$ is tight in $\mathbb M$.
	
	As the product of compact sets is compact in the product topology, $(u^{\kappa,n},\eta^{\kappa,n},W)_n$ is tight in the product topology. Let $(\tilde{u}^{\kappa},\tilde{\eta}^{\kappa},\tilde{W})$ be the limit of a converging subsequence, that we still denote $(u^{\kappa,n},\eta^{\kappa,n},W)_n$ for convenience.
	
	Let us now check that each of the three assertions of the statement holds true. The first assertion is a direct consequence of~\cref{prop:stability}. Regarding the second assertion, using~\cref{Lemma:TightMandphi} we can pass to the limit on~\eqref{Eq:tildeukappan} and obtain~\eqref{Eq:tildeukappa}.
	
	Let us finally prove the third assertion. For convenience, by Skorokhod representation theorem we can assume that $(u^{\kappa,n},\eta^{\kappa,n},W^n)_n$ converges almost surely to $(\tilde{u}^{\kappa},\tilde{\eta}^{\kappa},\tilde{W})$.	We use~\cref{lemma:continuity_u_eta} to write that for any $\psi\in\mathcal C_c([0,\infty)\times(0,1),\mathbb R_+)$,
	\begin{equation*}
		\int_{[0,\infty)\times(0,1)}\tilde{u}^{\kappa}\psi\,\mathrm{d}\tilde{\eta}^{\kappa}=\lim_n\int_{[0,\infty)\times(0,1)}u^{\kappa,n}\psi\,\mathrm{d}\eta^{\kappa,n}
	\end{equation*}
	Moreover, for any $A>0$, for every $n\geqslant0$,
	\begin{align*}
		&\int_{[0,\infty)\times(0,1)}|u^{\kappa,n}|\,\psi\,\mathrm{d}\eta^{\kappa,n}\\
		= &\int_{[0,\infty)\times(0,1)}\1_{(|u^{\kappa,n}|\leqslant A)}|u^{\kappa,n}|\,\psi\,\mathrm{d}\eta^{\kappa,n}+\int_{[0,\infty)\times(0,1)}\1_{(|u^{\kappa,n}|>A)}|u^{\kappa,n}|\,\psi\,\mathrm{d}\eta^{\kappa,n}\\
		\leqslant&A\eta^{\kappa,n}(\psi)+\kappa g_{1/n}(A)\int_{[0,\infty)\times(0,1)}|u^{\kappa,n}(t,x)|\,\psi(t,x)\,\mathrm{d} t\,\mathrm{d} x\,.
	\end{align*}
	Notice that $\eta^{\kappa,n}(\psi)$ converges to $\tilde{\eta}^\kappa(\psi)$ as $n\rightarrow\infty$, a quantity which is a.s. finite.
	Also, $g_{1/n}(A)$ goes to $0$ as $n\rightarrow\infty$, and by the Dominated Convergence theorem ($u^{\kappa,n}$ goes to $\tilde{u}^\kappa$ uniformly on compact sets, and $\psi$ has compact support),
	\begin{equation*}
		\lim_{n}\int_{[0,\infty)\times(0,1)}|u^{\kappa,n}(t,x)|\,\psi(t,x)\,\mathrm{d} t\,\mathrm{d} x=\int_{[0,\infty)\times(0,1)}|\tilde{u}^\kappa(t,x)|\,\psi(t,x)\,\mathrm{d} t\,\mathrm{d} x\,,
	\end{equation*}
	a quantity which is a.s. finite. Combining the previous convergences,
	\begin{equation*}
		\lim_n\int_{[0,\infty)\times(0,1)}|u^{\kappa,n}|\,\psi\,\mathrm{d}\eta^{\kappa,n} \leqslant A\tilde{\eta}^\kappa(\psi)\,.
	\end{equation*}
	As this is true for every $A>0$, we let $A\downarrow0$ to conclude that $\int_{[0,\infty)\times(0,1)}|\tilde{u}^{\kappa}|\,\psi\,\mathrm{d}\tilde{\eta}^{\kappa}=0$ a.s.
	Skorokhod representation theorem was harmless because being equal to $0$ a.s.~implies being equal in law to $0$.
	As a consequence, one has that for all $\psi\in\mathcal C_c([0,\infty)\times(0,1),\mathbb R_+)$,
	\begin{equation}\label{eq:reflection_measure}
		\int_{[0,\infty)\times(0,1)}|\tilde{u}^{\kappa}|\,\psi\,\mathrm{d}\tilde{\eta}^{\kappa}=0\,.
	\end{equation}
	The proof is complete.

By~\cref{prop:tightness}, $(u^\kappa)_{\kappa > 1}$ is tight. It remains to show that the limit of any converging subsequence coincides in law with $u^+$. Let $(u^{\kappa_k})_{k\geqslant 1}$ be a converging subsequence. Now for every $k\geqslant 1$, let $(\tilde{u}^{\kappa_k}, \tilde{\eta}^{\kappa_k},W)$ be an arbitrary limit point of a converging subsequence of $(u^{\kappa_k,n},\eta^{\kappa_k,n}, W)_n$, which is tight by~\cref{Prop:MildLimitPoint}. Still by~\cref{Prop:MildLimitPoint}, for every $k\geqslant 1$ and every $\varphi\in \mathcal C^\infty_0([0,1])$, a.s.~for all $t>0$
	\begin{equation}\label{Eq:tildeu}
		(\tilde{u}^{\kappa_k}_t,\varphi)-(\tilde{u}^{\kappa_k}_0,\varphi)-\int_0^t(\tilde{u}^{\kappa_k}_s,\frac12\varphi'')\,\mathrm{d} s-\int_{[0,t]\times[0,1]}\varphi(x)\,\tilde{\eta}^{\kappa_k}(\mathrm{d} s,\mathrm{d} x)-W_t(\varphi) = 0\;.
	\end{equation}
	Since $\tilde{u}^{\kappa_k}$ coincides in law with $u^{\kappa_k}$, we deduce that $(\tilde{u}^{\kappa_k})_{k\geqslant 1}$ converges in law, and is therefore tight. Moreover the law of $W$ does not depend on $k\geqslant1$. By~\eqref{Eq:tildeu} we deduce that the sequence of processes $(\int_{[0,t]\times[0,1]}\varphi(x)\,\tilde{\eta}^{\kappa_k}(\mathrm{d} x,\mathrm{d} s),t\geqslant 0)_{k\geqslant 1}$ is tight in $\mathcal C([0,\infty),\mathbb{R})$. Since this holds for any given $\varphi\in \mathcal C^\infty_0([0,1])$, \cref{Lemma:TightMandphi} ensures that $(\tilde{\eta}^{\kappa_k})_{k\geqslant 1}$ is tight.
	
	The sequence $(\tilde{u}^{\kappa_k}, \tilde{\eta}^{\kappa_k},W)_{k\geqslant 1}$ is therefore tight. Up to extracting a further subsequence, we can assume that it converges in law: denote by $(\tilde{u}^\infty,\tilde{\eta}^\infty,W)$ the limit. Note that $\tilde{u}^{\kappa_k}(0,\cdot)$ is independent of $W$ for each $k\geqslant 1$, so that the same property holds at the limit.\\
	Let us check that $(\tilde{u}^\infty,\tilde{\eta}^\infty,W)$ satisfies all the conditions listed in~\cref{Def:NP}.
	
	\noindent \cref{item:NP_1} We have chosen the sequence of initial conditions $u_0^{\kappa}\overset{\text{law}}{=}\mathbb Q_{0,0}^{\kappa}$ so by~\cref{theorem:static_convergence}, $\tilde{u}^\infty_0\overset{\text{law}}{=}\mathbb P_{0,0}^3$. Moreover $\tilde{u}^{\infty}$ is stationary as the limit of stationary processes, so for every $t>0$, $\tilde{u}^\infty_t\overset{\text{law}}{=}\mathbb P_{0,0}^3$. Since in addition, $\tilde{u}^\infty$ is continuous in time, we deduce that $\tilde{u}^\infty$ is a.s. nonnegative and vanishes at $x=0,1$.
	
	\noindent \cref{item:NP_2} For every $\varepsilon>0$ and $T>0$
	\begin{equation*}
		\tilde{\eta}^{\infty}([0,T]\times(\varepsilon,1-\varepsilon))\leqslant C_\varepsilon \int_{[0,T]\times [0,1]} x(1-x) \tilde{\eta}^\infty(\mathrm{d} s,\mathrm{d} x)<\infty\,,
	\end{equation*}
	where $C_\varepsilon > 0$ is a constant only depending on $\varepsilon$, and the fact that the latter quantity is finite follows from $\tilde{\eta}^\infty\in\mathbb M$.
	
	\noindent \cref{item:NP_3} Fix $\varphi \in \mathcal C^\infty_0([0,1])$. By~\cref{Lemma:TightMandphi}, we know that $(\tilde{u}^{\kappa_k}, X^k,W)_{k\geqslant 1}$ converges in law to $(\tilde{u}^\infty,X,W)$ where
	$$ X^k_t := \int_{[0,t]\times[0,1]}\varphi(x)\,\tilde{\eta}^{\kappa_k}(\mathrm{d} s,\mathrm{d} x)\;,$$
	and
	$$ X_t := \int_{[0,t]\times[0,1]}\varphi(x)\,\tilde{\eta}^{\infty}(\mathrm{d} s,\mathrm{d} x)\;.$$
	Passing to the limit in~\eqref{Eq:tildeu}, we deduce that a.s.~for all $t>0$
	\begin{equation}\label{Eq:tildeulimit}
		(\tilde{u}^{\infty}_t,\varphi)-(\tilde{u}^{\infty}_0,\varphi)-\int_0^t(\tilde{u}^{\infty}_s,\frac12\varphi'')\,\mathrm{d} s-\int_{[0,t]\times[0,1]}\varphi(x)\,\tilde{\eta}^{\infty}(\mathrm{d} s,\mathrm{d} x)-W_t(\varphi) = 0\;.
	\end{equation}
	\noindent \cref{item:NP_4} Fix $\psi \in\mathcal C_c([0,\infty)\times[0,1])$. By~\cref{Prop:MildLimitPoint} a.s.~for all $k\geqslant 1$
	\[
	\int_{[0,\infty)\times(0,1)}\psi\, |\tilde{u}^{\kappa_k}|\,\mathrm{d}\tilde{\eta}^{\kappa_k}=0\,.
	\]
	By~\cref{lemma:continuity_u_eta}
	\[
	\int_{[0,\infty)\times(0,1)}\psi\, |\tilde{u}^{\infty}|\,\mathrm{d}\tilde{\eta}^{\infty}=0\,.
	\]
	We claim that this implies that 
	\begin{equation}\label{eq:reflection_measure_infty}
		\int_{[0,\infty)\times(0,1)}\tilde{u}^{\infty}\,\mathrm{d}\tilde{\eta}^{\infty}=0\,.
	\end{equation}
	Indeed, we let $(a_j,b_j,T_j)_j$ be three sequences such that $\lim_j a_j=0$, $\lim_j b_j=1$, $\lim_j T_j=\infty$, and we let $(\psi_j)_{j\geqslant0}$ be a sequence of functions
	in $\mathcal C_c([0,\infty)\times[0,1],\mathbb R)$ such that for every $j$, $\psi_j$ has support in $[0,T_j]\times[a_j,b_j]$ and such that $\psi_j\longrightarrow1$ pointwise a.e. on $[0,\infty)\times[0,1]$.

	\begin{equation*}
		\int_{[0,\infty)\times(0,1)}|\tilde{u}^{\infty}|\,\mathrm{d}\tilde{\eta}^{\infty}\leqslant\liminf_{j\rightarrow\infty}\int_{[0,\infty)\times(0,1)}|\tilde{u}^{\infty}|\,\psi_j\,\mathrm{d}\tilde{\eta}^{\infty}=0\quad\text{a.s.}
	\end{equation*}
	We have thus proven that $(\tilde{u}^\infty,\tilde{\eta}^\infty)$ is a solution of~\eqref{eq:SPDE_NP}. In particular, this identifies uniquely all possible limits in law of $(u^{\kappa})_{\kappa > 1}$.
