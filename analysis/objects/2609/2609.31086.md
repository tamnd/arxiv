Let $f_{\textup{c}}$ denote the right-hand side of the $\xi$-dynamics in~\eqref{eq:controller_ineq}. To prove the statement we show that $\mathbb{R}_{\geq 0}^m$ is a forward invariant set for $f_{\textup{c}}$. By applying Nagumo's Theorem~\cite{MN:1942}, this holds if and only if
\begin{equation}
\label{eq:nagumo_pos}
f_{\textup{c,i}}(\xi) \geq 0 \quad \forall \xi \in \mathbb{R}_{\geq 0}^m \text{ such that } \xi_i =0.
\end{equation}
Consider~\eqref{eq:controller_ineq} in components $\dot{\xi}_i = - k_{\textup{i}}^{\textup{in}} \xi_i + k_{\textup{i}}^{\textup{in}} \operatorname{ReLU}\big(\xi_i + k_{\textup{p}}^{\textup{in}} g_i(x)\big) = f_{\textup{c,i}}(\xi)$, $i \in \{1,\dots, m\}$.
Writing condition~\eqref{eq:nagumo_pos} for $f_{\textup{c,i}}$, for all $\xi \in \mathbb{R}_{\geq 0}^m$ such that $\xi_i =0$ we have
$
f_{\textup{c,i}}(\xi) = k_{\textup{i}}^{\textup{in}}\operatorname{ReLU}\big(\xi_i + k_{\textup{p}}^{\textup{in}} g_i(x)\big) \geq 0,
$
since $\operatorname{ReLU}(\cdot) \geq 0$ for all arguments. This concludes the proof.

<!-- remark-1: body_md -->
\label{rem:kd_inequality}
In the equality-constrained case, introducing a derivative term through $J_g(x)\dot x$ allows for the explicit inversion of the matrix $M(x)$, inducing a state-dependent Riemannian metric. 
In the inequality case, however, the derivative term interacts non-smoothly with the $\operatorname{ReLU} (\cdot)$ operator enforcing dual feasibility, generating a non-smooth implicit algebraic loop. Resolving this algebraic loop and, in turn, designing a PID scheme for inequality-constrained problems, remains an open problem.

Let $z := (x, \nu, \xi)$ and let $z^\star$ be a KKT point of~\eqref{eq:eq_ineq} guaranteed to exist by Assumption~\ref{ass:1}. Consider the following Lyapunov function candidate
$$
V (x, \nu, \xi) = \frac{1}{2}\|x - x^\star\|^2_{M} + \frac{1}{2 k_{\textup{i}}^{\textup{eq}}}\|\nu - \nu^\star\|^2 + \frac{1}{2 k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}}\|\xi - \xi^\star\|^2.
$$
Clearly, $V(x, \nu, \xi)$ is positive definite, radially unbounded, and vanishes only at $(x^\star, \nu^\star, \xi^\star)$. Throughout, we write $r(x,\xi) := \operatorname{ReLU}\bigl(\xi + k_{\textup{p}}^{\textup{in}} g(x)\bigr)$ for brevity.

\textbf{Computing $\dot{V}$.}
Differentiating along trajectories of~\eqref{eq:sp_pid_eq_ineq_convex} gives
\begin{align*}
\dot{V}(x, \nu, \xi) &= (x - x^\star)^{\top} M \dot x + \frac{1}{k_{\textup{i}}^{\textup{eq}}} (\nu - \nu^\star)^{\top} \dot \nu + \frac{1}{k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \dot \xi \\
&= (x - x^\star)^{\top} \Bigl(- \nabla f(x) - J_g(x)^{\top} r(x, \xi) - A^{\top} \nu - k_{\textup{p}}^{\textup{eq}}A^{\top} h(x)\Bigr)\\ 
&\quad + \frac{1}{k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \Bigl( - k_{\textup{i}}^{\textup{in}} \xi + k_{\textup{i}}^{\textup{in}} r(x, \xi) \Bigr)+ (\nu - \nu^\star)^{\top} h(x)\\
&= - (x - x^\star)^{\top} \nabla f(x) - (x - x^\star)^{\top} J_g(x)^{\top} r(x, \xi) - (x - x^\star)^{\top} A^{\top} \bigl(\nu + k_{\textup{p}}^{\textup{eq}}h(x)\bigr)\\
&\quad + (\nu - \nu^\star)^{\top} h(x) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \xi +  \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} r(x, \xi).
\end{align*}
Using the stationarity condition~\eqref{kkt_stationarity}, $\nabla f(x^\star) = - J_g(x^\star)^{\top} \xi^\star  - A^{\top} \nu^\star $, and adding and subtracting $(x-x^\star)^{\top} \nabla f(x^\star)$ to $\dot{V}(x, \nu, \xi)$ yields
\begin{align}
\dot{V}(x, \nu, \xi) &= - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + (x - x^\star)^{\top} J_g(x^\star)^{\top} \xi^\star + (x - x^\star)^{\top} A^{\top} \nu^\star \nonumber\\
&\quad - (x - x^\star)^{\top} J_g(x)^{\top} r(x, \xi) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \xi +  \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} r(x, \xi) \nonumber \\
& \quad - (x - x^\star)^{\top} A^{\top} \bigl(\nu + k_{\textup{p}}^{\textup{eq}}h(x)\bigr) + (\nu - \nu^\star)^{\top} h(x) \nonumber\\
&= - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + (x - x^\star)^{\top} \bigl(J_g(x^\star)^{\top} \xi^\star - J_g(x)^{\top} r(x, \xi)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \bigl(\xi - r(x, \xi)\bigr) \nonumber \\
& \quad - (x - x^\star)^{\top} A^{\top} \bigl(\nu -\nu^\star + k_{\textup{p}}^{\textup{eq}}h(x)\bigr) + (\nu - \nu^\star)^{\top} h(x) \nonumber\\ 
&= - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + (x - x^\star)^{\top} \bigl(J_g(x^\star)^{\top} \xi^\star - J_g(x)^{\top} r(x, \xi)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \bigl(\xi - r(x, \xi)\bigr) \nonumber \\
& \quad - k_{\textup{p}}^{\textup{eq}}(x - x^\star)^{\top} A^{\top} A\bigl(x - x^\star\bigr) \nonumber \\
&= - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + (x - x^\star)^{\top} \bigl(J_g(x^\star)^{\top} \xi^\star - J_g(x)^{\top} r(x, \xi)\bigr) \nonumber\\
& \quad - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \bigl(\xi - r(x, \xi)\bigr) - k_{\textup{p}}^{\textup{eq}}\|h(x)\|^2, \label{v_ineq_1_}
\end{align}
where in the last equalities we used the fact that $h(x) = Ax -b = Ax - A x^\star = A(x - x^\star)$.

\textbf{Bounding the Jacobian cross-term.}
Since $g$ is convex and continuously differentiable, the first-order inequality $g(z) - g(y) \geq J_g(y)(z-y)$ holds for all $z,y$.
Evaluating the above inequality at $(z, y) = (x, x^\star)$ and $(z, y) = (x^\star, x)$, respectively, yields
\begin{align*}
g(x) - g(x^\star) \geq J_g(x^\star) (x - x^\star) \quad &\implies \quad (g(x) - g(x^\star))^{\top} \xi^\star \geq \big(J_g(x^\star)(x - x^\star)\big)^{\top} \xi^\star, \\
g(x^\star) - g(x) \geq J_g(x) (x^\star - x) \quad &\implies \quad -\big(g(x) - g(x^\star)\bigr)^{\top} r(x, \xi) \geq -\big(J_g(x)(x - x^\star)\big)^{\top} r(x, \xi),
\end{align*}
where the inequalities are preserved under multiplication since $\xi^\star \geq \0_m$ by KKT conditions and $r(x, \xi) \geq \0_m$ by the definition of the $\operatorname{ReLU}$ operator.
Substituting the above inequalities into~\eqref{v_ineq_1_} gives
\begin{align*}
\dot{V}(x, \xi) &\leq - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr)+ \bigl(g(x) - g(x^\star)\bigr)^{\top} \bigl(\xi^\star - r(x, \xi)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \bigl(\xi - r(x, \xi)\bigr)- k_{\textup{p}}^{\textup{eq}}\|h(x)\|^2.
\end{align*}

\textbf{Applying properties of $\operatorname{ReLU}$.}
The firmly non-expansive property of the $\operatorname{ReLU}$ operator allows us to establish the following bound on the constraint interaction terms:
\begin{align}
\bigl(g(x) - g(x^\star)\bigr)^{\top}& \bigl(\xi^\star - r(x, \xi)\bigr) \leq - \frac{1}{k_{\textup{p}}^{\textup{in}}} \|r(x, \xi) - \xi^\star\|^2 + \frac{1}{k_{\textup{p}}^{\textup{in}}} \bigl(\xi - \xi^\star \bigr)^{\top} \bigl(r(x, \xi) - \xi^\star \bigr). \label{eq:relu_nonexpansive_bound_}
\end{align}
To see this, recall that firmly non-expansiveness of the $\operatorname{ReLU}$ implies 
$\|\operatorname{ReLU}(z) - \operatorname{ReLU}(y)\|^2 \leq \bigl(z - y\bigr)^{\top}\bigl(\operatorname{ReLU}(z) - \operatorname{ReLU}(y) \bigr)$, for all $z,y \in \mathbb{R}^m$.
Inequality~\eqref{eq:relu_nonexpansive_bound_} follows directly by selecting $z = \xi + k_{\textup{p}}^{\textup{in}} g(x)$ and $y = \xi^\star + k_{\textup{p}}^{\textup{in}} g(x^\star)$.  

\textbf{Combining the bounds.}
Then, we can bound $\dot{V}(x, \nu, \xi)$ obtaining
\begin{align*}
- &(x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} \|r(x, \xi) - \xi^\star\|^2 + \frac{1}{k_{\textup{p}}^{\textup{in}}} \bigl(\xi - \xi^\star \bigr)^{\top} \bigl(r(x, \xi) - \xi^\star \bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^{\top} \bigl(\xi - r(x, \xi)\bigr)\\
&= - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + \frac{1}{k_{\textup{p}}^{\textup{in}}} \Bigl(- \|r(x, \xi)\|^2 - \|\xi^\star\|^2 + 2r(x, \xi)^{\top} \xi^\star + 
\xi^{\top} r(x, \xi) - r(x, \xi)^{\top} \xi^\star - \xi^{\top} \xi^\star + \|\xi^\star\|^2\\
&\quad \quad \quad \quad \quad\quad \quad \quad \quad \quad\quad \quad \quad \quad \quad \quad \quad - \|\xi\|^2 + \xi^{\top} \xi^\star + \xi^{\top} r(x, \xi) - r(x, \xi)^{\top} \xi^\star\Bigr)\\
&= - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + \frac{1}{k_{\textup{p}}^{\textup{in}}} \Bigl(- \|r(x, \xi)\|^2 - \|\xi\|^2 + 2 \xi^{\top} r(x, \xi) \Bigr) \\
&= - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} \|r(x, \xi) - \xi\|^2 \leq 0, 
\end{align*}
where the last inequality follows from $(x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) \geq 0$, by convexity of $f$.
In summary, we have shown that
\begin{equation}
\label{eq:Vdot_final_}
\dot{V}(x, \nu, \xi) \leq - (x - x^\star)^{\top} \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} \|r(x, \xi) - \xi\|^2 - k_{\textup{p}}^{\textup{eq}}\|h(x)\|^2 \leq 0.
\end{equation}
Hence, $V$ is non-increasing along trajectories of~\eqref{eq:sp_pid_eq_ineq_convex}, every trajectory is bounded, and $(x^\star,\nu^\star, \xi^\star)$ is stable.

\textbf{LaSalle's invariance principle.}
Let $\mathcal{E} := \{(x,\nu,\xi)\in\mathbb{R}^{n+p+m} \; | \; \dot{V}(x,\nu,\xi)=0\}$ and let $\Omega\subseteq\mathcal{E}$ denote the largest invariant subset of $\mathcal{E}$. Consider a trajectory $(x(t),\nu(t),\xi(t))\in\Omega$. From~\eqref{eq:Vdot_final_}, we have
\begin{align}
&(x-x^\star)^{\top} \bigl(\nabla f(x) -\nabla f(x^\star)\bigr) = 0, \label{eq:vdot_primal_}\\
&\operatorname{ReLU}\bigl(\xi+k_{\textup{p}}^{\textup{in}}g(x)\bigr) - \xi = \0_m, \label{eq:vdot_dual_1_}\\
&h(x) = \0_p. \label{eq:vdot_dual_2_}
\end{align}
Equations~\eqref{eq:vdot_dual_1_} and~\eqref{eq:vdot_dual_2_} imply $\dot{\xi} = \0_m$ and $\dot{\nu} = \0_p$, respectively. Therefore $\xi(t) = \bar{\xi}$ and $\nu(t) = \bar{\nu}$, for all $t \geq 0$.
Moreover, the fixed-point condition~\eqref{eq:vdot_dual_1_} and~\eqref{eq:vdot_dual_2_} imply primal feasibility, dual feasibility, and complementary slackness~\eqref{kkt_primal_feasibility}--\eqref{kkt_complementarity} in $(x, \bar{\nu}, \bar{\xi})$, namely, $h(x) = \0_p$, $g(x) \leq \0_m$, $\bar{\xi} \geq \0_m$, and $\bar{\xi}^{\top} g(x) = 0$.
 
Then, the primal dynamics reduce to $\dot{x} = - M^{-1}\Bigl(\nabla f(x) + J_h(x)^{\top} \bar{\nu} + J_g(x)^{\top} \bar{\xi}\Bigr)$. Define the Lagrangian $L(x,\nu,\xi) = f(x) + h(x)^{\top} \nu + g(x)^{\top} \xi$, which is convex due to $f$, $g$ and $h$ convex.
The primal dynamics read $\dot{x} = - M^{-1}\nabla L(x,\bar{\nu},\bar{\xi})$. To understand the invariant set for the condition~\eqref{eq:vdot_primal_}, consider the first term of the Lyapunov candidate $\phi(x) = 1/2 \|x - x^\star\|^2_M$. Along trajectories in $\Omega$, we have
\begin{align*}
\dot{\phi}(x) &= (x - x^\star)^{\top} M \dot{x} = -(x - x^\star)^{\top} \nabla L(x,\bar{\nu},\bar{\xi})  \leq L(x^\star,\bar{\nu},\bar{\xi}) - L(x,\bar{\nu},\bar{\xi}) \leq f(x^\star) - f(x) \leq 0,
\end{align*}
where the first inequality follows from convexity of $L(\cdot,\bar{\nu},\bar{\xi})$, the second uses $L(x^\star,\bar{\nu},\bar{\xi}) \leq f(x^\star)$ due to $h(x^\star)=\0_p$, $\bar{\xi}\geq \0_m$ and $g(x^\star)\leq \0_m$, and $L(x,\bar{\nu},\bar{\xi}) = f(x)$ due to complementary slackness and~\eqref{eq:vdot_dual_2_}. The last inequality follows from the optimality of $x^\star$ and the feasibility of $x$.
So $\phi(x)$ is nonincreasing along trajectories in $\Omega$, and the feasible set is forward invariant, implying all solutions converge to the largest invariant subset of $\{\dot{\phi}=0\}$.
On this invariant set, $0=\dot{\phi} \leq f(x^\star)-f(x) \leq 0$, which implies $f(x^\star) = f(x)$. Thus, $x$ minimizes $L(\cdot, \bar{\nu}, \bar\xi)$ by convex sufficiency, giving $\nabla f(x) + J_h(x)^{\top}\bar{\nu} + J_g(x)^{\top}\bar{\xi} = \0_n$, implying $\dot{x} = \0_n$.
Therefore, every point in $\Omega$ satisfies the KKT conditions and, by LaSalle's invariance principle, the $\omega$-limit set of every solution is contained in $\Omega$, and we conclude $\operatorname{dist}((x, \nu, \xi), \mathcal{K}) \to 0$.
This proves item~\ref{thm_conv_eq_ineq:item1}.

        
Item~\ref{thm_conv_eq_ineq:item2} follows by noticing that strict convexity of $f$ and LICQ together imply that the KKT set reduces to the singleton  $\{(x^\star, \nu^\star, \xi^\star)\}$. Global asymptotic stability then follows from LaSalle's invariance principle combined with radial unboundedness of $V$. This concludes the proof.

Since~\eqref{eq:eq_ineq} is convex under Assumption~\ref{ass:1}, and strict complementarity, LICQ, and SOSC hold at $z^\star$, $x^\star$ is a strict local minimizer of~\eqref{eq:eq_ineq}~\cite{DPB:97}. Moreover, since a strict local minimizer of a convex program is its unique global minimizer, and LICQ ensures unique multipliers, $z^\star$ is the unique KKT point of~\eqref{eq:eq_ineq}, i.e., $\mathcal{K} = \{z^\star\}$.
We show that SPPID is globally weakly contracting (non-expansive), and locally strongly contracting in a neighborhood of $z^\star$.
 
Define $P = \operatorname{diag}\bigl(M, \frac{1}{k_{\textup{i}}^{\textup{eq}}} I_p, \frac{1}{k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}} I_m\bigr) \succ 0$.
Moreover, let $y := \xi + k_{\textup{p}}^{\textup{in}} g(x)$ and define $G(y) := \nabla \operatorname{ReLU}(y)$ where it exists. By Rademacher's Theorem, the Jacobian $J(z)$ of the SPPID~\eqref{eq:sp_pid_eq_ineq_convex} exists almost everywhere and is given by
$$
J(z)= -
\begin{bmatrix}
M^{-1}\left( \nabla^2 f(x) + k_{\textup{p}}^{\textup{eq}}A^{\top} A + \sum_{j=1}^m \operatorname{ReLU}(y_j)\nabla^2 g_j(x) +k_{\textup{p}}^{\textup{in}}J_g(x)^{\top} G(y) J_g(x) \right) & M^{-1}A^{\top} & M^{-1}J_g(x)^{\top} G(y) \\
-k_{\textup{i}}^{\textup{eq}}A & 0 & 0 \\
-k_{\textup{i}}^{\textup{in}}k_{\textup{p}}^{\textup{in}}G(y)J_g(x) & 0 & k_{\textup{i}}^{\textup{in}}(I_m-G(y))
\end{bmatrix}.
$$
To prove globally weak contractivity, it suffices to show that $\mu_{P}(J(z)) \leq 0$ or equivalently $\frac{1}{2}\left(P J(z) + J(z)^{\top} P\right) \preceq 0$ for all $z$ for which $J(z)$ exists.
A direct calculation gives
\begin{align*}
\frac{1}{2}\left(P J(z) + J(z)^{\top} P\right) =
\begin{bmatrix} 
-\nabla^2 f(x) - k_{\textup{p}}^{\textup{eq}} A^{\top} A - \sum_{j=1}^m\operatorname{ReLU}(y_j)\nabla^2g_j(x) - k_{\textup{p}}^{\textup{in}}J_g(x)^{\top} G(y)J_g(x) & 0 & 0 \\
0 & 0 & 0 \\
0 & 0 & -\frac{1}{k_{\textup{p}}^{\textup{in}}}(I_m - G(y))
\end{bmatrix},
\end{align*}
 
which is block-diagonal. The LMI then follows by noticing that convexity of $f$ and $g_j$ implies $\nabla^2 f(x) \succeq 0$ and $\nabla^2 g_j(x) \succeq 0$, for all $x$, and $0 \preceq G(y) \preceq I_m$, for all $y$.
 

Next, we prove local strong contraction around $z^\star$. By strict complementarity, for $j \in \mathcal{A}$, $g_j(x^\star) = 0$ and $\xi^\star_j > 0$ imply $y^\star_j > 0$ and thus $G_j(y^\star) = 1$. For $j \notin \mathcal{A}$, $g_j(x^\star) < 0$ and $\xi^\star_j = 0$ imply $y^\star_j < 0$, hence $G_j(y^\star) = 0$.
 
In particular, $y^\star_j \neq 0$ for every $j$ and, by continuity, the activation pattern remains constant in a neighborhood of $z^\star$. The inactive multiplier dynamics satisfy $\dot{\xi}_{\mathcal{I}} = -k_{\textup{i}}^{\textup{in}} \xi_{\mathcal{I}}$, and therefore converge exponentially. The remaining active subsystem $(x, \nu, \xi_{\mathcal{A}})$ has Jacobian
$$
J_{\mathcal{A}}(z^\star)=
\begin{bmatrix}
-M^{-1}\left( \nabla^2 f(x^\star) + k_{\textup{p}}^{\textup{eq}}A^{\top} A + \sum_{j=1}^m \xi^\star_j\nabla^2 g_j(x^\star) +k_{\textup{p}}^{\textup{in}}J_g^{\mathcal{A}}(x^\star)^{\top} J_g^{\mathcal{A}}(x^\star) \right) & -M^{-1}A^{\top} & -M^{-1}J_g^{\mathcal{A}}(x^\star)^{\top} \\
k_{\textup{i}}^{\textup{eq}}A & 0 & 0 \\
k_{\textup{i}}^{\text{in}}k_{\textup{p}}^{\textup{in}}J_g^{\mathcal{A}}(x^\star) & 0 & 0
\end{bmatrix}.
$$
By LICQ, the matrix $C := \begin{bmatrix} A\\ J_g^{\mathcal A}(x^\star) \end{bmatrix}$ has full row rank. By SOSC at $(x^\star, \xi^\star)$, we have $H := \nabla^2 f(x^\star) + \sum_{j=1}^m \xi^\star_j\nabla^2 g_j(x^\star) \succ 0$. 
In fact, SOSC implies
$d^\top H d > 0$ for all $d \neq \0_n$ satisfying $Ad=0$, and $J_g^{\mathcal A}(x^\star)d=0$, that is, for all nonzero $d\in\ker(C) \setminus \{\0_n\}$. For any $d\neq \0_n$,
$
\begin{aligned}
d^\top J_1(z^\star)d &= d^\top H d + k_{\textup{p}}^{\textup{eq}}\|Ad\|^2 + k_{\textup{p}}^{\textup{in}} \|J_g^{\mathcal A}(x^\star)d\|^2.
\end{aligned}
$
If $d\in\ker(C)$, then the penalty terms vanish and the SOSC implies $d^\top J_1(z^\star)d = d^\top H d >0$.
Otherwise, $Cd\neq0$, and we have
$
k_{\textup{p}}^{\textup{eq}}\|Ad\|^2 + k_{\textup{p}}^{\textup{in}} \|J_g^{\mathcal A}(x^\star)d\|^2 >0.
$
Moreover, $H\succeq0$ by convexity and $\xi_j^\star\ge0$. Hence,
$
d^\top J_1(z^\star)d>0
$
for every $d\neq0$, and therefore
$
J_1(z^\star)\succ0 .
$
Then, $J_{\mathcal{A}}(z^\star)$ has the saddle form in Lemma~\ref{lemma:saddle-matrices_general}, which in turn guarantees that the $z^\star$ is locally exponentially stable with rate 
$$
c_{\mathcal{A}}= \frac{1}{4} \min \left\{ \frac{1}{\lambda_{\max}(J_1(z^\star))}, \tau \frac{\lambda_{\min}(J_1(z^\star))} {\sigma_{\max}^2(C)} \right\} \frac{\sigma_{\min}^2(C)}{1 + k_{\textup{d}}^{\textup{eq}} a_{\textup{max}}},
$$

where $\tau := \min \bigl(k_{\textup{i}}^{\textup{eq}}, k_{\textup{i}}^{\textup{in}}k_{\textup{p}}^{\textup{in}}\bigr)^{-1}$.
Then,~\eqref{eq:sp_pid_eq_ineq} is strongly infinitesimally contracting in a neighborhood of $z^\star$~\cite{TS:75}. The statement then follows by~\cite[Theorem 2]{VC-AD-AG-GR-FB:24a}. This concludes the proof.

Substituting these gradients back into~\eqref{eq:pd_gradient_flow_ineq} recovers the SPPI dynamics~\eqref{eq:pi_spf_ineq}. This concludes the proof.

Let $(x^\star, \xi^\star)$ be any KKT point of problem~\eqref{eq:ineq_constrained}, which exists by  Assumption~\ref{ass:1}~\cite{DPB:97}.
Throughout, we write $r(x,\xi) := \operatorname{ReLU}\bigl(\xi + k_{\textup{p}}^{\textup{in}} g(x)\bigr)$ for brevity.
Consider the following Lyapunov function candidate
$$
V (x, \xi) = \frac{1}{2}\|x - x^\star\|^2 + \frac{1}{2 k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}}\|\xi - \xi^\star\|^2.
$$
Clearly, $V(x, \xi)$ is positive definite, radially unbounded, and vanishes only at $(x^\star,\xi^\star)$.
 
\textbf{Computing $\dot{V}$.}
Differentiating along trajectories of~\eqref{eq:pi_spf_ineq} gives
\begin{align}
\dot{V}(x, \xi) &= (x - x^\star)^\top \dot x + \frac{1}{k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top \dot \xi \nonumber\\
&= (x - x^\star)^\top \Bigl(- \nabla f(x) - J_g(x)^\top r(x, \xi)\Bigr) + \frac{1}{k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top \Bigl( - k_{\textup{i}}^{\textup{in}} \xi + k_{\textup{i}}^{\textup{in}} r(x, \xi) \Bigr) \label{eq:vdot_lasalle}\\
&= - (x - x^\star)^\top \nabla f(x) - (x - x^\star)^\top J_g(x)^\top r(x, \xi) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top \xi +  \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top r(x, \xi). \nonumber
\end{align}
Using the stationarity condition~\eqref{kkt_stationarity}, $\nabla f(x^\star) = - J_g(x^\star)^\top \xi^\star$, and adding and subtracting $(x-x^\star)^\top \nabla f(x^\star)$ to $\dot{V}(x, \xi)$ yields
\begin{align}
\dot{V}(x, \xi) &= - (x - x^\star)^\top \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + (x - x^\star)^\top J_g(x^\star)^\top \xi^\star - (x - x^\star)^\top J_g(x)^\top r(x, \xi)\\
&\quad \quad - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top \xi +  \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top r(x, \xi) \nonumber \\
&= - (x - x^\star)^\top \bigl(\nabla f(x) - \nabla f(x^\star)\bigr) + (x - x^\star)^\top \bigl(J_g(x^\star)^\top \xi^\star - J_g(x)^\top r(x, \xi)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top \bigl(\xi - r(x, \xi)\bigr). \label{v_ineq_1}
\end{align}

\textbf{Bounding the Jacobian cross-term.}
Since $g$ is convex and continuously differentiable, the first-order inequality $g(z) - g(y) \geq J_g(y)(z-y)$ holds for all $z,y$~\cite{DPB:97}.
Evaluating the above inequality at $(z, y) = (x, x^\star)$ and $(z, y) = (x^\star, x)$, respectively, yields
\begin{align*}
g(x) - g(x^\star) \geq J_g(x^\star) (x - x^\star) \quad &\implies \quad (g(x) - g(x^\star))^\top \xi^\star \geq \big(J_g(x^\star)(x - x^\star)\big)^\top \xi^\star, \\
g(x^\star) - g(x) \geq J_g(x) (x^\star - x) \quad &\implies \quad -\big(g(x) - g(x^\star)\bigr)^\top r(x, \xi) \geq -\big(J_g(x)(x - x^\star)\big)^\top r(x, \xi),
\end{align*}
where the inequalities are preserved under multiplication since $\xi^\star \geq \0_m$ by KKT conditions and $r(x, \xi) \geq \0_m$ by the definition of the $\operatorname{ReLU}$ operator.
Substituting the above inequalities into~\eqref{v_ineq_1} gives
\begin{align*}
\dot{V}(x, \xi) &\leq - (x - x^\star)^\top \bigl(\nabla f(x) - \nabla f(x^\star)\bigr)+ \bigl(g(x) - g(x^\star)\bigr)^\top \bigl(\xi^\star - r(x, \xi)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}} (\xi - \xi^\star)^\top \bigl(\xi - r(x, \xi)\bigr).
\end{align*}

\textbf{Applying properties of $\operatorname{ReLU}$.}
The firmly non-expansive property of the $\operatorname{ReLU}$ operator allows us to establish the following bound on the constraint interaction terms:
\begin{align}
\bigl(g(x) - g(x^\star)\bigr)^\top& \bigl(\xi^\star - r(x, \xi)\bigr) \leq - \frac{1}{k_{\textup{p}}^{\textup{in}}} \|r(x, \xi) - \xi^\star\|^2 + \frac{1}{k_{\textup{p}}^{\textup{in}}} \bigl(\xi - \xi^\star \bigr)^\top \bigl(r(x, \xi) - \xi^\star \bigr).
\label{eq:relu_nonexpansive_bound}
\end{align}
To see this, recall that firmly non-expansiveness of the $\operatorname{ReLU}$ implies 
$\|\operatorname{ReLU}(z) - \operatorname{ReLU}(y)\|^2 \leq \bigl(z - y\bigr)^\top\bigl(\operatorname{ReLU}(z) - \operatorname{ReLU}(y) \bigr)$, for all $z,y \in \mathbb{R}^m$.
Inequality~\eqref{eq:relu_nonexpansive_bound} follows directly by selecting $z = \xi + k_{\textup{p}}^{\textup{in}} g(x)$ and $y = \xi^\star + k_{\textup{p}}^{\textup{in}} g(x^\star)$. 

\textbf{Combining the bounds.}
Substituting~\eqref{eq:relu_nonexpansive_bound} into~\eqref{v_ineq_1} and simplifying the dual terms, yields $-\tfrac{1}{k_{\textup{p}}^{\textup{in}}}\|r-\xi^\star\|^2 + \tfrac{1}{k_{\textup{p}}^{\textup{in}}}(\xi-\xi^\star)^\top(r-\xi^\star) - \tfrac{1}{k_{\textup{p}}^{\textup{in}}}(\xi-\xi^\star)^\top(\xi-r) = -\tfrac{1}{k_{\textup{p}}^{\textup{in}}}\|r-\xi\|^2$, where the last equality follows by expanding and cancelling all terms involving $\xi^\star$.
Therefore,
\begin{equation}
\label{eq:Vdot_final}
\dot{V}(x,\xi) \leq - (x-x^\star)^\top \bigl(\nabla f(x) -\nabla f(x^\star)\bigr) - \frac{1}{k_{\textup{p}}^{\textup{in}}}\|r(x,\xi)-\xi\|^2 \leq 0,
\end{equation}
where the last inequality uses convexity (hence monotonicity) of $\nabla f$. Hence, $V$ is nonincreasing along trajectories of~\eqref{eq:pi_spf_ineq}, every trajectory is bounded, and $(x^\star,\xi^\star)$ is stable.

\textbf{LaSalle's invariance principle.}
Let $\mathcal{E} := \{(x,\xi)\in\mathbb{R}^{n+m} \; | \; \dot{V}(x,\xi)=0\}$ and let $\Omega\subseteq\mathcal{E}$ denote the largest invariant subset of $\mathcal{E}$. Consider a trajectory $(x(t),\xi(t))\in\Omega$. From~\eqref{eq:Vdot_final}, we have
\begin{align}
&(x-x^\star)^\top \bigl(\nabla f(x) -\nabla f(x^\star)\bigr) = 0, \label{eq:vdot_primal}\\
&\operatorname{ReLU}\bigl(\xi+k_{\textup{p}}^{\textup{in}}g(x)\bigr) - \xi = \0_m \label{eq:vdot_dual}.
\end{align}
Equation~\eqref{eq:vdot_dual} implies $\dot{\xi} = -k_{\textup{i}}^{\textup{in}}\xi + k_{\textup{i}}^{\textup{in}}r(x,\xi) = \0_m$, and therefore $\xi(t) = \bar{\xi}$, for all $t \geq 0$.
Moreover, the fixed-point condition~\eqref{eq:vdot_dual} implies primal feasibility, dual feasibility, and complementary slackness~\eqref{kkt_primal_feasibility}--\eqref{kkt_complementarity} in $(x, \bar{\xi})$, namely, $g(x) \leq \0_m$, $\bar{\xi} \geq \0_m$, 
and $\bar{\xi}^\top g(x) = 0$.
 
Since $\bar{\xi}$ is constant on $\Omega$, the primal dynamics reduce to $\dot{x} = - \nabla f(x) - J_g(x)^\top \bar{\xi}$. Define the Lagrangian $L(x,\xi) = f(x) + g(x)^\top \xi$, which is convex due to $f$ and $g$ convex.
The primal dynamics read $\dot{x} = -\nabla L(x,\bar{\xi})$. To understand the invariant set for the condition~\eqref{eq:vdot_primal}, consider the first term of the Lyapunov candidate $\phi(x) = 1/2 \|x - x^\star\|^2$. Along trajectories in $\Omega$, we have
\begin{align*}
\dot{\phi}(x) &= (x - x^\star)^\top\dot{x} = -(x - x^\star)^\top \nabla L(x,\bar{\xi}) \leq L(x^\star,\bar{\xi}) - L(x,\bar{\xi}) \leq f(x^\star) - f(x) \leq 0,
\end{align*}
where the first inequality follows from convexity of $L(\cdot,\bar{\xi})$, the second uses $L(x^\star,\bar{\xi}) \leq f(x^\star)$ due to $\bar{\xi}\geq \0_m$ and $g(x^\star)\leq \0_m$, and $L(x,\bar{\xi}) = f(x)$ due to complementary slackness.
The last inequality follows from the optimality of $x^\star$ and the feasibility of $x$.
So $\phi(x)$ is nonincreasing along trajectories in $\Omega$, and the feasible set is forward invariant, implying all solutions converge to the largest invariant subset of $\{\dot{\phi}=0\}$.
On this invariant set, $0=\dot{\phi} \leq f(x^\star)-f(x) \leq 0$, which implies $f(x^\star) = f(x)$.
Thus, $x$ minimizes $L(\cdot,\bar\xi)$ by convex sufficiency, giving $\nabla f(x) + J_g(x)^\top\bar{\xi} = \0_n$, implying $\dot{x} = \0_n$.
Therefore, every point in $\Omega$ satisfies the KKT conditions and, by LaSalle's invariance principle, the $\omega$-limit set of every solution is contained in $\Omega$, and we conclude $\operatorname{dist}((x, \xi), \mathcal{K}) \to 0$.
This proves item~\ref{thm_conv_ineq:item1}.

Item~\ref{thm_conv_ineq:item2} follows by noticing that strict convexity  of $f$ and LICQ together imply that the KKT set reduces to the singleton  $\{(x^\star, \xi^\star)\}$. Global asymptotic stability then follows from LaSalle's invariance principle combined with radial unboundedness of $V$. This concludes the proof.

We show that SPPI is globally weakly contracting and locally strongly contracting around $z^\star$.
 
Let $z = (x, \xi) \in \mathbb{R}^{n+m}$ denote the full state vector, $\dot{z} = F(z)$ the SPPI~\eqref{eq:pi_spf_ineq}, and let $P = \begin{bmatrix} I_n & 0 \\ 0 & \frac{1}{k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}}} I_m \end{bmatrix} \succ 0$.
Moreover, let $y := \xi + k_{\textup{p}}^{\textup{in}} g(x)$ and define $G(y) := \nabla \operatorname{ReLU}(y)$ where it exists. By Rademacher's theorem, the Jacobian of the SPPI~\eqref{eq:pi_spf_ineq} exists almost everywhere and is given by
$$
J(z)=
\begin{bmatrix}
-\nabla^2 f(x) - \sum_{j=1}^m \operatorname{ReLU}\big(\xi_j + k_{\textup{p}}^{\textup{in}} g_j(x)\big) \nabla^2 g_j(x) - k_{\textup{p}}^{\textup{in}} J_g(x)^\top G(y) J_g(x) & -J_g(x)^\top G(y) \\ 
k_{\textup{i}}^{\textup{in}}k_{\textup{p}}^{\textup{in}}G(y) J_g(x) & - k_{\textup{i}}^{\textup{in}} (I_m - G(y))
\end{bmatrix}.
$$
To prove globally weak contractivity, it suffices to show that for all $z$ for which $J(z)$ exists, we have $\mu_{P}(J(z)) \leq 0$ or equivalently $\frac{1}{2}\left(P J(z) + J(z)^\top P\right) \preceq 0$ for all $z \in \mathbb{R}^{n+m}$.
We have 
\begin{align*}
\frac{1}{2}\left(P J(z) + J(z)^\top P\right) =
\begin{bmatrix} 
-\nabla^2 f(x) - \sum_{j=1}^m \operatorname{ReLU} \big(\xi_j + k_{\textup{p}}^{\textup{in}} g_j(x)\big) \nabla^2 g_j(x) - k_{\textup{p}}^{\textup{in}} J_g(x)^\top G(y) J_g(x) & 0 \\ 
0 & -\frac{1}{k_{\textup{p}}^{\textup{in}}}(I_m - G(y)) 
\end{bmatrix},
\end{align*}
 
which is block-diagonal. The LMI then follows by convexity of $f$ and $g_j$, and the condition $0 \preceq G(y) \preceq I_m$, for almost every $y$.

Next, we prove local strong contraction around $z^\star$. Write $\mathcal{A} := \mathcal{A}(x^\star)$ for the active set and $\mathcal{I} := \{1,\dots,m\}\setminus\mathcal{A}$ for its complement. By strict complementarity, for $j \in \mathcal{A}$, $g_j(x^\star) = 0$ and $\xi^\star_j > 0$ imply $y^\star_j > 0$ and thus $G_j(y^\star) = 1$. For $j \notin \mathcal{A}$, $g_j(x^\star) < 0$ and $\xi^\star_j = 0$ imply $y^\star_j < 0$, hence $G_j(y^\star) = 0$.
 
In particular, $y^\star_j \neq 0$ for all $j$ and, by continuity, in a sufficiently small local neighborhood $\mathcal{B}(z^\star)$ of $z^\star$, the activation pattern $G$ remains constant. The inactive multiplier dynamics decouple as a Hurwitz linear subsystem $\dot{\xi}_{\mathcal{I}} = -k_{\textup{i}}^{\textup{in}} \xi_{\mathcal{I}}$. The remaining active subsystem coordinates $(x, \xi_{\mathcal{A}})$ have the local Jacobian:
$$
J_{\textup{loc}} = \begin{bmatrix} - \nabla^2 f(x^\star) - \sum_{j \in \mathcal{A}} \xi^\star_j \nabla^2 g_j(x^\star) - k_{\textup{p}}^{\textup{in}} J_g^{\mathcal{A}}(x^\star)^\top J_g^{\mathcal{A}}(x^\star) & -J_g^{\mathcal{A}}(x^\star)^\top \\ k_{\textup{i}}^{\textup{in}} k_{\textup{p}}^{\textup{in}} J_g^{\mathcal{A}}(x^\star) & 0 \end{bmatrix}.
$$
By SOSC, $H := \nabla^2 f(x^\star) + \sum_{j \in \mathcal{A}} \xi^\star_j \nabla^2 g_j(x^\star) \succ 0$ on $\ker J_g^{\mathcal{A}}(x^\star)$, and by LICQ, $B := J_g^{\mathcal{A}}(x^\star)$ has full row rank. By the same argument as in the proof of Theorem~\ref{thm:lin_exp_eq_ineq} (with the equality-constraint terms absent), $H + k_{\textup{p}}^{\textup{in}} B^\top B \succ 0$. Hence $J_{\textup{loc}}$ has the saddle form of Lemma~\ref{lemma:saddle-matrices_general} and is Hurwitz for any $k_{\textup{p}}^{\textup{in}}, k_{\textup{i}}^{\textup{in}} > 0$.
 
Because the entire Jacobian $J(z^\star)$ is Hurwitz, the unique equilibrium is locally exponentially stable. Then, the system~\eqref{eq:pi_spf_ineq} is strongly infinitesimally contracting in a neighborhood of the equilibrium point $z^\star$~\cite{TS:75}. The statement then follows by applying~\cite[Theorem 2]{VC-AD-AG-GR-FB:24a}. This concludes the proof.

\textbf{Positivity of P.}\\
We start by verifying that $P\succ0$. Using the Schur complement of the $(2,2)$ entry, $P\succ0$ is equivalent to
$

$
{Since $\Theta \preceq \theta_{\textup{max}}I_p$, for every $x\in\mathbb{R}^n$ we have $x^\top A^\top \Theta A x = (Ax)^\top \Theta (Ax) \leq \theta_{\textup{max}}\|Ax\|^2 = \theta_{\textup{max}} x^\top A^\top A x$, i.e., $A^\top \Theta A \preceq \theta_{\textup{max}} A^\top A$. \begin{align*}

2\alpha A^\top {\Theta} A  + 2 c_{\textup{eq}} M &\preceq \alpha \left(2{\theta_{\textup{max}}} a_{\textup{max}} + 2 c_{\textup{eq}} m_{\textup{max}}\right)I_n \overset{\eqref{lem:def:c}}{\preceq} \alpha\left(2{\theta_{\textup{max}}}a_{\textup{max}} + {\theta_{\textup{min}}}a_{\textup{min}}\right)I_n \\
& \!\!\!\!\!\!\!\!\! \overset{\alpha\leq\frac{1}{2}b_{\textup{min}}/({\theta_{\textup{max}}}a_{\textup{max}})}{\preceq}
\frac{b_{\textup{min}}}{2{\theta_{\textup{max}}}a_{\textup{max}}}\left(2{\theta_{\textup{max}}}a_{\textup{max}} + {\theta_{\textup{min}}}a_{\textup{min}}\right) I_n = b_{\textup{min}}\Bigl(1+{\frac{\theta_{\textup{min}}a_{\textup{min}}}{2\theta_{\textup{max}}a_{\textup{max}}}}\Bigr)I_n \preceq \displaystyle \frac{3}{2}b_{\textup{min}} I_n,
\end{align*}
{where the last inequality uses $\theta_{\textup{min}}\leq\theta_{\textup{max}}$ and $a_{\textup{min}}\leqa_{\textup{max}}$.}
Next, we upper bound the left hand side of~\eqref{eq:ineq_1_lem} exactly as follows
\begin{align*}
2B - \alpha BM^{-1}B &\succeq 2B - \alpha\frac{1}{m_{\textup{min}}} B^2 \succeq 2B - \alpha\frac{b_{\textup{max}}}{m_{\textup{min}}} B \succeq \left(2 - \alpha\frac{b_{\textup{max}}}{m_{\textup{min}}}\right) B \succeq \left(2 - \frac{1}{2}\right) B \succeq \frac{3}{2}b_{\textup{min}} I_n.
\end{align*}

\textbf{Prove $\mathbf{4 \alpha c_{\textup{eq}} B \succeq 4 \alpha c_{\textup{eq}}^2 M}$.}\\
The LMI $4 \alpha c_{\textup{eq}} B \succeq 4 \alpha c_{\textup{eq}}^2 M$ follows from the inequality $\displaystyle c_{\textup{eq}} \leq \frac{b_{\textup{min}}}{m_{\textup{max}}}$. This follows from~\eqref{lem:def:c} and $\alpha \leq \frac{1}{2}b_{\textup{min}}/({\theta_{\textup{max}}}a_{\textup{max}})$:
\[
c_{\textup{eq}} = \frac{1}{2}\alpha{\theta_{\textup{min}}}\frac{a_{\textup{min}}}{m_{\textup{max}}} \leq \frac{1}{4}\frac{b_{\textup{min}}}{{\theta_{\textup{max}}}a_{\textup{max}}}\cdot{\theta_{\textup{min}}}\frac{a_{\textup{min}}}{m_{\textup{max}}} = \frac{1}{4}{\frac{\theta_{\textup{min}}}{\theta_{\textup{max}}}}\frac{a_{\textup{min}}b_{\textup{min}}}{a_{\textup{max}}m_{\textup{max}}} < \frac{b_{\textup{min}}}{m_{\textup{max}}}.
\]
This concludes the proof.

\textbf{Positivity of P.}\\
We verify $P \succ 0$ via the Schur complement of the $(2,2)$ block. Since $I_m \succ 0$, we need
$
\gamma I_n - \frac{\alpha\beta}{\gamma }C^\top C \succ 0.
$
This LMI holds if and only if $\gamma^2 > \alpha\beta\textup{c}_{\textup{max}}$, which is guaranteed by hypothesis, being $\gamma > \alpha\textup{c}_{\textup{max}}$ and $\gamma > \beta$.

\textbf{Expansion of the LMI.}\\
We aim to show that $-\bar{S}^\top P - P\bar{S} -2 c_{\textup{in}} P\succeq 0$. We compute
\begin{align*}
-\bar{S}^\top P - P\bar{S} -2 c_{\textup{in}} P
&=
\begin{bmatrix}
B + \alpha C^\top G C & -\beta\alpha C^\top G \\
G C & \beta (I_m - G)
\end{bmatrix}
\begin{bmatrix}
\gamma I_n & C^\top \\
C & \dfrac{\gamma}{\alpha\beta} I_m
\end{bmatrix}
+
\begin{bmatrix}
\gamma I_n & C^\top \\
C & \dfrac{\gamma}{\alpha\beta} I_m
\end{bmatrix}
\begin{bmatrix}
B + \alpha C^\top G C & C^\top G \\
-\beta\alpha G C & \beta (I_m - G)
\end{bmatrix}\\
&\quad - 2 c_{\textup{in}}
\begin{bmatrix}
\gamma I_n & C^\top \\
C & \dfrac{\gamma}{\alpha\beta} I_m
\end{bmatrix}
\\&=
\begin{bmatrix}
\gamma(B+\alpha C^\top GC) - \beta\alpha C^\top GC
&
(B+\alpha C^\top GC)C^\top - \gamma C^\top G \\
\gamma GC + \beta C - \beta GC
&
GCC^\top + \dfrac{\gamma}{\alpha}(I_m-G)
\end{bmatrix}
\\&\quad+
\begin{bmatrix}
\gamma(B+\alpha C^\top GC) - \beta\alpha C^\top GC
&
\gamma C^\top G + \beta C^\top - \beta C^\top G \\
CB + \alpha CC^\top GC - \gamma GC
&
CC^\top G + \dfrac{\gamma}{\alpha}(I_m-G)
\end{bmatrix}
-
\begin{bmatrix}
2c_{\textup{in}}\gamma I_n & 2c_{\textup{in}}C^\top \\
2c_{\textup{in}}C & \dfrac{2c_{\textup{in}}\gamma}{\alpha\beta}I_m
\end{bmatrix}
\\&=
\begin{bmatrix}
2\gamma B + 2\alpha(\gamma-\beta)C^\top G C - 2c_{\textup{in}}\gamma I_n
&
\bigl(B+\beta I_n - 2c_{\textup{in}}I_n\bigr)C^\top +
C^\top G\bigl(\alpha CC^\top - \beta I_m\bigr) \\
C\bigl(B+\beta I_n - 2c_{\textup{in}}I_n\bigr) +
\bigl(\alpha CC^\top - \beta I_m\bigr)GC
&
GCC^\top + CC^\top G + \frac{2\gamma}{\alpha}(I_m-G) -
\dfrac{2c_{\textup{in}}\gamma}{\alpha\beta}I_m
\end{bmatrix} \succeq 0.
\end{align*}
                                       
First, we find a lower bound for $Q_{22}$. Let $G=U[d]U^\top$ be an eigendecomposition, with $U \in \mathbb{R}^{m \times m}$ orthogonal and $d\in[0,1]^m$. Substituting this decomposition into $Q_{22}$ and multiplying on the left and on the right by $U$ and $U^\top$, respectively, we get
\begin{equation}
\label{eq:decomposition}
U^\top Q_{22}U = [d]U^\top CC^\top U + U^\top CC^\top U[d] + \frac{2\gamma}{\alpha}(I_m- [d]) - \dfrac{2c_{\textup{in}}\gamma}{\alpha\beta}I_m
\end{equation} 
Note that the eigenvalues of $U^\top CC^\top U$ and $CC^\top$ coincide. Then, applying Lemma~\cite[Lemma 6]{GQ-NL:19} to~\eqref{eq:decomposition} with $X:=U^\top CC^\top U$, and multiplying this on the left and on the right by $U$ and $U^\top$, for any $\gamma \geq \alpha \textup{c}_{\textup{max}}$ we get
$$
(C C^\top G + GC C^\top) + 2 \frac{\beta}{\alpha} (I_m - G) - \dfrac{2c_{\textup{in}}\gamma}{\alpha\beta}I_m \succeq \frac{3}{2} C C^\top - \dfrac{2c_{\textup{in}}\gamma}{\alpha\beta}I_m.
$$
Then, the (2,2) block satisfies the lower bound 
\begin{align*}
(C C^\top G + GC C^\top) + 2 \frac{\beta}{\alpha} (I_m - G) - \dfrac{2c_{\textup{in}}\gamma}{\alpha\beta} I_m  &\succeq \frac{3}{2} C C^\top - \dfrac{2c_{\textup{in}}\gamma}{\alpha\beta}I_m\\
&= 2 \Big(\frac{3}{8} C C^\top -\dfrac{c_{\textup{in}}\gamma}{\alpha\beta} I_m\Big) + \frac{3}{4} C C^\top \\
& \succeq 2\Big(\frac{3}{8} \textup{c}_{\textup{min}} - \dfrac{c_{\textup{in}}\gamma}{\alpha\beta} \Big)I_m + \frac{3}{4}C C^\top\\
&\overset{c_{\textup{in}}:=  \frac{3\alpha\beta\textup{c}_{\textup{min}}}{8\gamma}}{=} \frac{3}{4} C C^\top\succ 0.
\end{align*}
Given this lower bound, we compute
$$
-\bar{S}^\top P - P\bar{S} -2 c_{\textup{in}} P \succeq
\begin{bmatrix}
2\gamma B + 2\alpha(\gamma-\beta)C^\top G C - 2c_{\textup{in}}\gamma I_n
&
\bigl(B+\beta I_n - 2c_{\textup{in}}I_n\bigr)C^\top +
C^\top G\bigl(\alpha CC^\top - \beta I_m\bigr) \\
C\bigl(B+\beta I_n - 2c_{\textup{in}}I_n\bigr) +
\bigl(\alpha CC^\top - \beta I_m\bigr)GC
&
\frac{3}{4} C C^\top
\end{bmatrix}.
$$
It now suffices to show that the Schur complement of the (2,2) block of the above matrix is positive semidefinite. To this end, we first bound $Q_{12}Q_{22}^{-1}Q_{12}^\top$.

\textbf{Schur complement verification.}\\
First, we bound the matrix $Q_{12}Q_{22}^{-1}Q_{12}^\top$. Let $Q_{12} = \bigl(B+\beta I_n - 2c_{\textup{in}}I_n\bigr)C^\top + C^\top G\bigl(\alpha CC^\top - \beta I_m\bigr) := \bigl(B + Q_3\bigr) C^\top + \beta C^\top(I_m-G)$, with $Q_3 = \alpha C^\top GC - 2c_{\textup{in}}I_n$. We have
\begin{align*}
\frac{3}{4}Q_{12}&Q_{22}^{-1}Q_{12}^\top \preceq Q_{12}(CC^\top)^{-1}Q_{12}^\top \\
& = \left(\bigl(B + Q_3\bigr) C^\top + \beta C^\top(I_m-G)\right)(CC^\top)^{-1}\left(\bigl(B + Q_3\bigr) C^\top + \beta C^\top(I_m-G)\right)^\top\\
& = \left(\bigl(B + Q_3\bigr) C^\top + \beta C^\top(I_m-G)\right)(CC^\top)^{-1}\left(C \bigl(B + Q_3^\top\bigr) + \beta (I_m-G)C\right)\\
& = \underbrace{\bigl(B + Q_3\bigr) C^\top (CC^\top)^{-1}C \bigl(B + Q_3^\top\bigr)}_{\text{(i)}} + \underbrace{\beta \bigl(B + Q_3\bigr) C^\top (CC^\top)^{-1} (I_m-G)C + 
\beta C^\top(I_m-G)(CC^\top)^{-1}C \bigl(B + Q_3^\top\bigr)}_{\text{(ii)}} \\
& \quad + \underbrace{\beta^2 C^\top(I_m-G)(CC^\top)^{-1}(I_m-G)C}_{\text{(iii)}}.
\end{align*}
We bound each term separately. To do so, we use the bounds $\|I_m-G\| \leq 1$ and $\|C^\top\| = \|C\| = \sqrt{\textup{c}_{\textup{max}}}$. Moreover, we have 
$$
\|Q_3\| = \|\alpha C^\top GC - 2c_{\textup{in}}I_n\| \leq \alpha \|C^\top GC\| + 2c_{\textup{in}} =  \alpha \|C^\top GC\| + 2 \frac{3\alpha\beta\textup{c}_{\textup{min}}}{8\gamma} \leq 
\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}},
$$
where we used the inequality $\gamma \geq \alpha \textup{c}_{\textup{max}}$. We compute

\emph{Term (i):} Since $\textup{c}_{\textup{min}}>0$, $C$ has full row rank, so $C^\top(CC^\top)^{-1}C$ is the orthogonal projector onto $\mathrm{row}(C)$, giving $C^\top(CC^\top)^{-1}C \preceq I_n$. Then
\begin{equation}
\label{eq:bound_i}
\begin{aligned}
\bigl(B + Q_3\bigr) C^\top (CC^\top)^{-1}C \bigl(B + Q_3\bigr)^\top &\preceq \bigl(B + Q_3\bigr)^2 = B^2+BQ_3+Q_3B+Q_3^2 \preceq b_{\textup{max}}B +\bigl(2b_{\textup{max}}\|Q_3\|+\|Q_3\|^2\bigr)I_n\\
&\preceq b_{\textup{max}}B + \left(2b_{\textup{max}}(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}) + \Bigl(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)^2\right)I_n,
\end{aligned}
\end{equation}
where we used the notation $X^2 =: X^\top X$ as shorthand for the symmetric product.

\emph{Term (ii):} Noting that $C^\top(CC^\top)^{-1}$ is the Moore--Penrose pseudoinverse of $C$, with $\|C^\top(CC^\top)^{-1}\| = 1/\sqrt{\textup{c}_{\textup{min}}}$, we compute
\begin{equation}
\label{eq:bound_iii}
\begin{aligned}
\|\beta \bigl(B + Q_3\bigr) C^\top (CC^\top)^{-1} (I_m-G)C + \beta C^\top(I_m-G)(CC^\top)^{-1}C \bigl(B + Q_3^\top\bigr)\| &\leq 2 \beta \|B + Q_3\| \|C^\top (CC^\top)^{-1}\| \|(I_m-G)\| \|C^\top\| \\
&\leq 2 \beta \|B + Q_3\| \dfrac{\sqrt{\textup{c}_{\textup{max}}}}{\sqrt{\textup{c}_{\textup{min}}}}\\
&\leq 2 \beta \Bigl(b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr) \dfrac{\sqrt{\textup{c}_{\textup{max}}}}{\sqrt{\textup{c}_{\textup{min}}}}.
\end{aligned}
\end{equation}

\emph{Term (iii):} Since $\|(CC^\top)^{-1}\| = 1/\textup{c}_{\textup{min}}$, we have bound
\begin{equation}
\label{eq:bound_iii}
\|\beta^2 C^\top(I_m-G)(CC^\top)^{-1}(I_m-G)C\| \leq \beta^2 \|C^\top\|^2 \|I_m-G\|^2 \|(CC^\top)^{-1}\|\leq \dfrac{\beta^2 \textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}.
\end{equation}
Combining bounds~\eqref{eq:bound_i}--\eqref{eq:bound_iii} we have
\begin{align*}
\frac{3}{4}Q_{12}&Q_{22}^{-1}Q_{12}^\top \preceq
b_{\textup{max}}B + \left(\Bigl(2b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)\Bigl(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)
+ 
2 \beta \Bigl(b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr) \dfrac{\sqrt{\textup{c}_{\textup{max}}}}{\sqrt{\textup{c}_{\textup{min}}}}
+
\dfrac{\beta^2 \textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}\right) I_n.
\end{align*}

Next, we note that $\gamma \geq \beta$ and $C^\top GC \succeq 0$, then $\alpha(\gamma-\beta)C^\top GC \succeq 0$ which in turn implies
$$
Q_{11} = 2\gamma B + 2\alpha(\gamma-\beta)C^\top G C - 2c_{\textup{in}}\gamma I_n \succeq 2\gamma B - 2 \gamma c_{\textup{in}}I_n.
$$
Then
\begin{align*}
Q_{11} - Q_{12}&Q_{22}^{-1}Q_{12}^\top \succeq (2\gamma - \frac{4}{3}b_{\textup{max}})B - 2 \gamma c_{\textup{in}}I_n \\
&\quad - \frac{4}{3} \left(\Bigl(2b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)\Bigl(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)
+ 
2 \beta \Bigl(b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr) \dfrac{\sqrt{\textup{c}_{\textup{max}}}}{\sqrt{\textup{c}_{\textup{min}}}}
+
\dfrac{\beta^2 \textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}\right) I_n \\
&\overset{2\gammac_{\textup{in}}:=  \frac{3\alpha\beta\textup{c}_{\textup{min}}}{4}}{=}
\bigl(2\gamma - \frac{4}{3}b_{\textup{max}}\bigr)B - \frac{3\alpha\beta\textup{c}_{\textup{min}}}{4}I_n \\
&\quad - \frac{4}{3} \left(\Bigl(2b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)\Bigl(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)
+ 
2 \beta \Bigl(b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr) \dfrac{\sqrt{\textup{c}_{\textup{max}}}}{\sqrt{\textup{c}_{\textup{min}}}}
+
\dfrac{\beta^2 \textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}\right) I_n \\ 
& \succeq \Bigl(\bigl(2\gamma - \frac{4}{3}b_{\textup{max}}\bigr)b_{\textup{min}} - \frac{3\alpha\beta\textup{c}_{\textup{min}}}{4}\Bigr)I_n \\
&\quad - \frac{4}{3} \left(\Bigl(2b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)\Bigl(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr)
+ 
2 \beta \Bigl(b_{\textup{max}} + \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}}\Bigr) \dfrac{\sqrt{\textup{c}_{\textup{max}}}}{\sqrt{\textup{c}_{\textup{min}}}}
+
\dfrac{\beta^2 \textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}\right) I_n \\
& \succeq \Bigl(\bigl(2\gamma - \frac{4}{3}b_{\textup{max}}\bigr)b_{\textup{min}}
- \frac{3\alpha\beta\textup{c}_{\textup{min}}}{4}\Bigr)I_n \\
&\quad - \frac{4}{3} \Bigl(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}} + \beta\sqrt{\frac{\textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}}\Bigr)\Bigl(\alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}} + \beta\sqrt{\frac{\textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}} + 2b_{\textup{max}}\Bigr) I_n
,
\end{align*}
where in the last inequality we used the fact that $\gamma \geq \dfrac{2b_{\textup{max}}}{3}$.
Let $x := \alpha \textup{c}_{\textup{max}} + \frac{3\beta\textup{c}_{\textup{min}}}{4 \textup{c}_{\textup{max}}} + \beta\sqrt{\frac{\textup{c}_{\textup{max}}}{\textup{c}_{\textup{min}}}}$. 
Then, the condition $\gamma \geq \dfrac{2}{3} b_{\textup{max}} + \dfrac{3\alpha\beta\textup{c}_{\textup{min}}}{8 b_{\textup{min}}} +\dfrac{2}{3} \dfrac{x(x+2b_{\textup{max}})}{b_{\textup{min}}}$ implies $Q_{11} - Q_{12}Q_{22}^{-1}Q_{12}^\top \succeq 0$ and thus the thesis. This concludes the proof.
