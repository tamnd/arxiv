<!-- head-1: body_md -->
We introduce the notations that we will use throughout the paper.
$\left\|{\bm{v}} \right\|_2$ defines its $l_2$ norm for a given vector $\bm{v}$.
$\left\|{A} \right\|_{{\rm F}}$ defines the Frobenius norm of a given matrix $A$, and 
$\lambda_2(A)$ denotes its second largest eigenvalue when $A$ is a square matrix.  
${\mathbb{R}}^d$ defines a $d$-dimensional vector space.  
$[m] \triangleq \{1, \cdots, m\}$.  
$\mathbf{1}$ is an all-one vector.
${\mathbf{1}_{\left\{{{\mathcal{E}}}\right\}}}$ is an indicator function of event ${\mathcal{E}}$, i.e.\xspace, ${\mathbf{1}_{\left\{{{\mathcal{E}}}\right\}}}=1$ when the event ${\mathcal{E}}$ occurs; ${\mathbf{1}_{\left\{{{\mathcal{E}}}\right\}}}=0$ otherwise.
${\mathcal{F}}^t$ denotes the sigma-algebra generated by the randomness up to round $t$. 
$f(n) = {\mathcal{O}}( g(n))$ if there exist 
$c_0>0$ and $n_0 \in {\mathbb{N}}$ such that $f(n)\le c_0 g(n)$, $\forall n \ge n_0$.

<!-- assumption-1: body_md -->
\label{ass: threat model}
There exists a $c \in (0,1]$ such that $p_i^t \ge c$, where the activation events of uplink $i$ are independent across devices $i \in [m]$ and across rounds $t \in [T]$.
<!-- end assumption-1 -->
<!-- remark-1: body_md -->
\label{rmk: objective inconsistency}
    In a standard digital FedAvg implementation, each device $i$ computes a local update based on the most recent global model parameters and transmits it to the parameter server.
    When $p_i$'s are heterogeneous across devices but time-invariant across rounds, the learning process minimizes
    a biased objective \citep{wang2023lightweight,11475389}
    \begin{align}
    \label{eq:biased convergence aux}
    \widetilde F(\bm{x}) = \frac{\sum_{i=1}^{m} p_i F_i (\bm{x})}{\sum_{i=1}^m p_i},    
    \end{align}
    rather than the unbiased objective in \eqref{eq:global objective}.
    Depending on the heterogeneity 
    in $p_i$'s, the output of~FedAvg~can be significantly away from the true optimum of \eqref{eq:global objective}. 
    It is easy to see that time-varying $p_i^t$'s only exacerbate the bias.
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lmm: spectral norm main}
For any matrix $B \in {\mathbb{R}}^{d \times m}$ that is independent of $W$ matrices, it holds that
\begin{align*}
\mathbb{E}\left[ \Bigg\|B \left( \prod_{r=1}^{t} W^{(r)} - \mathbf J \right)\Bigg\|_{\mathrm{F}}^2 ~\Big|~ B \right] \le \rho^t \left\|{B} \right\|_{{\rm F}}^2, 
\end{align*}
where $\left\|{\cdot} \right\|_{{\rm F}}$ denotes the Frobenius norm of a matrix.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lmm: ergodicity}
Suppose that~\prettyref{ass: threat model} holds.
For all $t\ge 1$, it holds that 
$\rho = \max_{t} \rho(t) \le 1 - \frac{c^4\left[ 1-\left( 1-c \right)^m \right]^2}{8}$.
<!-- end lemma-2 -->
<!-- assumption-2: body_md -->
\label{ass: 2 smmothness}
For each device node $i\in [m]$, its local stochastic gradient function
$\nabla \ell_{i}(\bm{x}, \xi)$ is $L_i$-Lipschitz, i.e.\xspace, for any stochastic sample $\xi$,
$$
\left\|{\nabla \ell_{i}(\bm{x}_1; \xi)-\nabla \ell_{i}(\bm{x}_2; \xi)} \right\|_2
\le L_i \left\|{\bm{x}_1-\bm{x}_2} \right\|_2
\le L \left\|{\bm{x}_1-\bm{x}_2} \right\|_2
, ~~~~ \forall \bm{x}_1, \bm{x}_2,
$$ 
where $L \triangleq \max\limits_{i\in[m]} L_i$.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass: bounded variance client-wise}
Stochastic gradients at each device node $i\in[m]$ are unbiased estimates of the true gradient of the local objectives, i.e., 
\[
\mathbb{E}\left[ \nabla \ell_i(\bm{x}_i^t; \xi_{i}^{t}) \mid {\mathcal{F}}^{t} \right]=\nabla F_i(\bm{x}_i^t),  
\]
and the variance of stochastic gradients at each device node $i\in[m]$ is uniformly bounded, i.e., 
\[
\mathbb{E}\left[ \left\|{\nabla \ell_i(\bm{x}_{i}^{t}; \xi_{i}^{t})-\nabla F_i(\bm{x}_{i}^{t})} \right\|_2^2 \mid {\mathcal{F}}^t \right]\le\sigma^2, ~~~ \forall ~ \bm{x} \in {\mathbb{R}}^d. 
\]
<!-- end assumption-3 -->
<!-- lemma-3: body_md -->
\label{lmm: local step perturbation}
Suppose~\prettyref{ass: 2 smmothness} holds. When $s> 1$,   
\[
\|\sum_{k=0}^{s-1} 
[\nabla \ell_i (\bm{x}_i^{\left( t,k \right)};\xi_i^{t}) - \nabla \ell_i (\bm{x}_i^t; \xi_i^{t})]\|_2
\le \frac{\kappa \eta s(s-1) L_i}{2} \left\|{\nabla \ell_{i}(\bm{x}_i^t;\xi_i^{t})} \right\|_2,
\]
where
$
\kappa \triangleq \max_{i}\frac{2 \left( (1+\eta L_i)^s - 1- s \eta L_i \right)}{
s(s-1)
\left( \eta L_i \right)^2} 
$
and non-decreases with respect to $\eta>0$.
When $s=1$, $\|\nabla \ell_i (\bm{x}_i^{\left( t,0 \right)};\xi_i^{t}) - \nabla \ell_i (\bm{x}_i^t; \xi_i^{t})\|_2 =0$.
<!-- end lemma-3 -->
<!-- assumption-4: body_md -->
\label{ass: bounded similarity}
We say that local objective function $F_i$'s satisfy $(\beta,\zeta)$-bounded dissimilarity condition for $\beta,\zeta \ge 0$ if
\begin{align}
\label{eq: BG condition}
\frac{1}{m}\sum_{i=1}^m \left\|{\nabla F_i(\bm{x})- \nabla F(\bm{x})} \right\|_2^2 \le \beta^2 \left\|{\nabla F(\bm{x})} \right\|_2^2+ \zeta^2, ~~~ \forall ~ \bm{x} \in {\mathbb{R}}^d.
\end{align}
<!-- end assumption-4 -->
<!-- assumption-5: body_md -->
\label{ass: bounded true gradient}
The full-batch gradient is uniformly bounded, i.e.\xspace, 
$\left\|{\nabla F_i (\bm{x})} \right\|_2 \le G_{\max}$, $\forall \bm{x} \in {\mathbb{R}}^d$.
<!-- end assumption-5 -->
<!-- lemma-4: body_md -->
\label{lmm: descent lemma}
Suppose Assumptions \ref{ass: 2 smmothness}, \ref{ass: bounded variance client-wise}, and \ref{ass: bounded similarity} hold.
Choose a learning rate $\eta$ such that
$\eta \le \frac{1}{108s L (\kappa + 1) (\beta^2+1)}$.
It holds that
\begin{align*}
&\mathbb{E}\left[ F(\bar{\bm{x}}^{t+1})  -  F(\bar{\bm{x}}^{t}) \mid {\mathcal{F}}^{t} \right]
\le -\frac{\eta s }{3} \left\|{\nabla F(\bar{\bm{x}}^t)} \right\|_2^2 
+
\underbrace{\frac{\eta^2  L d \sigma_z^2}{\gamma^2 m^2}\mathbb{E}\left[ B_t^2 \mid {\mathcal{F}}^t \right]}_{\text{Receiver Noise}}
\\
&\qquad  \qquad
+ 6 \eta^2 s^2  L\left( \zeta^2+\sigma^2 \right)\left( 1+\kappa^2 L^2 \right)
+  \frac{\eta s L^2 }{m} \sum_{i=1}^m 
\underbrace{\left\|{\bm{x}_i^t - \bar{\bm{x}}^t} \right\|_2^2}_{\text{Consensus Error}}. 
\end{align*}
<!-- end lemma-4 -->
<!-- proof-1: body_md -->
Unroll one iteration of $\bar{\bm{x}}^{t+1}$, it holds that    
\begin{small}
\begin{align}
    \notag
    \bar{\bm{x}}^{t+1} - \bar{\bm{x}}^{t}
    &=
    \frac{1}{m} 
    \sum_{i \in {\mathcal{A}}^t}
    \left( 
    \frac{1}{\left | {\mathcal{A}}^t \right |} 
    \left( \frac{\eta B_t \bm{z}^t}{\gamma}
    +
    \sum_{i\in {\mathcal{A}}^t} \widetilde \bm{x}_i^{t} 
     \right)
    - \bm{x}_i^t \right) 
    +
    \frac{1}{m} 
    \sum_{i \not \in {\mathcal{A}}^t}
    \left( 
    \widetilde \bm{x}_i^{t}
    - \bm{x}_i^t
     \right) \\
    \label{eq: unroll one round}
    &=
    -
    \frac{\eta}{m} 
    \sum_{i=1}^m
    \sum_{r=0}^{s-1}
    \nabla \ell_{i}(\bm{x}_{i}^{(t,r)})
    +
    \frac{\eta }{m} \frac{B_t \bm{z}^t}{\gamma}
    .
\end{align}
\end{small}
Via~\prettyref{ass: 2 smmothness}, we have
\begin{small}
\begin{align*}
&F(\bar{\bm{x}}^{t+1})  -  F(\bar{\bm{x}}^{t}) 
\le 
\underbrace{\left \langle \nabla F(\bar{\bm{x}}^{t}), - \frac{\eta }{m} 
\sum_{i=1}^m 
\sum_{r=0}^{s-1}
\nabla \ell_i(\bm{x}_i^{(t,r)};\xi_{i}^{t})
 \right\rangle
}_{(A)} \\
&~~~+  
\underbrace{\left \langle \nabla F(\bar{\bm{x}}^{t}), \frac{\eta}{m} \frac{B_t}{\gamma} \bm{z}^{(t)} \right\rangle }_{(B)}
+
\frac{L}{2}
\underbrace{\left\|{\frac{\eta}{m} 
\sum_{i=1}^m 
\sum_{r=0}^{s-1}
\nabla \ell_i(\bm{x}_i^{(t,r)};\xi_{i}^{t})
- 
\frac{\eta B_t \bm{z}^t}{\gamma m}} \right\|_2^2
}_{(C)}
.  
\end{align*}
\end{small}
Term $(A)$ can be bounded by following the standard roadmap of stochastic gradient analysis with~\prettyref{lmm: local step perturbation} and Assumptions~\ref{ass: bounded variance client-wise} and~\ref{ass: bounded similarity}.
Recall that $B_t \triangleq \max_{i \in {\mathcal{A}}^t} \|\Delta_{i}^{t}\|_2^2$ is an online norm tracker to obtain the maximum norm of local updates in ${\mathcal{A}}^t$ in round $t$.
The key observation is that, conditional on ${\mathcal{F}}^t$, $B_t$ and receiver noise $\bm{z}^t$ are independent. So, we have 
\begin{small}
\begin{align*}
    \mathbb{E}\left[ (B) \mid {\mathcal{F}}^t \right]
    &=
    \frac{\eta}{m \gamma}
    \left \langle \nabla F(\bar{\bm{x}}^{t}), 
    \mathbb{E}\left[ B_t \mid {\mathcal{F}}^t \right]
    \mathbb{E}\left[ \bm{z}^{t} \mid {\mathcal{F}}^t \right]  \right\rangle= 0
\end{align*}
\end{small}
Next, we use Jensen's inequality to upper bound the term $(C)$:
\begin{small}
\begin{align*}
    \mathbb{E}\left[ (C) \mid {\mathcal{F}}^t \right]
    &\le
    \frac{2 \eta^2}{m^2}
    \underbrace{\left\|{
    \sum_{i=1}^m 
    \sum_{r=0}^{s-1}
    \nabla \ell_i(\bm{x}_i^{(t,r)};\xi_{i}^{t})
    } \right\|_2^2}_{(C.{\rm I})}
    +
    \frac{2 \eta^2}{\gamma^2 m^2 }
    \underbrace{\left\|{B_t \bm{z}^{t}} \right\|_2^2}_{(C.{\rm II})}
    .
\end{align*}
\end{small}
By the conditional independence of $B_t$ and $\bm{z}^t$ on ${\mathcal{F}}^t$, we have $(C.{\rm II}) = \mathbb{E} [\|B_t \bm{z}^t\|_2^2 | {\mathcal{F}}^t] \le d \sigma_z^2 \mathbb{E}[B_t^2 | {\mathcal{F}}^t]$.
Term $(C.{\rm I})$ can be upper bounded by invoking Assumptions~\ref{ass: bounded variance client-wise},~\ref{ass: bounded similarity}, and~\prettyref{lmm: local step perturbation}.
The rest of the proof combines the intermediate results of terms $(A)$, $(B)$, and $(C)$, and by applying our learning rate condition $\eta \le {1}/(108s L (\kappa + 1) (\beta^2+1))$. 
Unlike FedAvg algorithm,~\prettyref{lmm: descent lemma} highlights a consensus error term arising from the implicit gossiping and receiver noise term from OTA computation.
<!-- end proof-1 -->
<!-- lemma-5: body_md -->
\label{lmm:gradient Bt}
    Suppose Assumptions~\ref{ass: threat model},~\ref{ass: 2 smmothness},  
\ref{ass: bounded variance client-wise}, 
\ref{ass: bounded similarity},  
and \ref{ass: bounded true gradient} 
hold, and $\eta \le \frac{1}{s L (\kappa + 1)}$.
It holds that
\begin{small}
\begin{align*}
    \frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\left[ B_t^2 \right]
    &\le
    \frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\left[ C_t^2 \right] 
    \le
    \frac{12 m s^2 
    \left( \sigma^2 + G_{\max}^2 \right) }{c^2}.
\end{align*}
\end{small}
<!-- end lemma-5 -->
<!-- proof-2: body_md -->
Expand $C_t^2$, and by Jensen's inequality, we have
\begin{small}
\begin{align*}
    C_t^2 
    =
    \sum_{i=1}^m
    \left\|{\Delta_{i}^{t}} \right\|_2^2
    &=
    \sum_{i=1}^m
    \left\|{\sum_{k=\tau_i(t) + 1}^{t}
    \sum_{r=0}^{s-1} \nabla \ell_i (\bm{x}_i^{(k,r)}; \xi_i^{k})} \right\|_2^2 \\
    &\le
    \sum_{i=1}^m
    (t - \tau_i(t))
    \sum_{k=\tau_i(t) + 1}^{t}
    \left\|{\sum_{r=0}^{s-1} \nabla \ell_i (\bm{x}_i^{(k,r)};\xi_i^{k})} \right\|_2^2
    .
\end{align*}
\end{small}
By taking the expectation and averaging over $T$ rounds, we have
\begin{small}
\begin{align}
    \frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\left[ C_t^2 \right]
    &\le
    \frac{1}{T}
    \sum_{t=0}^{T-1}
    \sum_{i=1}^m
    \mathbb{E}\Bigg[
        (t - \tau_i(t))
        \sum_{k=\tau_i(t) + 1}^{t}
        \left\|{\sum_{r=0}^{s-1} \nabla \ell_i (\bm{x}_i^{(k,r)};\xi_i^{k})} \right\|_2^2
    \Bigg].
    \label{eq:bt_start main text}
\end{align}
\end{small}
The key step is to regroup the terms in \eqref{eq:bt_start main text}: 
\begin{small}
\begin{align*}
    \eqref{eq:bt_start main text}
    &=
    \frac{1}{T}
    \sum_{i=1}^m
    \sum_{k=0}^{T-1}
    \mathbb{E}
    \Bigg[
         \underbrace{\left\|{
       \sum_{r=0}^{s-1} \nabla \ell_i (\bm{x}_i^{(k,r)};\xi_i^{k})} \right\|_2^2}_{
        \triangleq\; g_{i,k}
        }
        \underbrace{
        \sum_{t=k}^{T-1}
        (t - \tau_i(t))
        {\mathbf{1}_{\left\{{\tau_i(t) + 1 \le k}\right\}}}
        }_{\triangleq\; S_{i,k}}
    \Bigg].
\end{align*}
\end{small}
Crucially, conditional on ${\mathcal{F}}^{k}$, 
the multi-step gradient sum $g_{i,k}$ and $S_{i,k}$ are {\em independent} because future channel randomness and the current round stochastic gradients are independent.
Therefore, we have
\begin{small}
\begin{align}
\label{eq: conditional independence}
\mathbb{E}\left[ g_{i,k} \cdot S_{i,k} ~\Big |~{\mathcal{F}}^k \right]
&=
\mathbb{E}\left[ g_{i,k} ~\Big |~{\mathcal{F}}^k \right]
\mathbb{E}\left[ S_{i,k} ~\Big |~{\mathcal{F}}^k \right].    
\end{align}
\end{small}
We first bound $S_{i, k}$. 
We define an auxiliary look-ahead variable $\tau_{i}^{+}(k) \triangleq \min\{t > k : i \in {\mathcal{A}}^t\}$ to denote the
first round after $k$ at which uplink $i$ becomes active.
Since ${\mathbf{1}_{\left\{{\tau_i(t) + 1 \le k}\right\}}} = 0$ for all $t \ge \tau_{i}^{+}(k) + 1$, the
sum $S_{i,k}$ accumulates only until device $i$'s next contribution:
\begin{small}
\begin{align*}
    &S_{i,k}
    =
    \sum_{t=k}^{\tau_{i}^{+}(k) }(t - \tau_i(t))
    \overset{(a)}{=}
    \sum_{j=0}^{\tau_{i}^{+}(k) - k} \left( j + k - \tau_i(k) \right) \\
    &=
    \frac{(\tau_{i}^{+}(k) - k)^2}{2}
    +
    \frac{\tau_{i}^{+}(k) - k}{2} 
    +(k - \tau_i(k)) \left( \tau_{i}^{+}(k) - k \right)
    +(k - \tau_i(k)),
\end{align*}
\end{small}
where equality $(a)$ follows from change of variable $j = t - k$ and the fact that $\tau_i (k) = \tau_i(j + k)$ for $k \le j + k  \le \tau_i^{+}(k)$ because the uplink is inactive during that period.
We can then invoke
Lemma 7 in~Appendix D
to bound the first moment of the staleness factor $(\tau_i^{+}(k) - k)$ by $1/c$ and its second moment by $2/c^2$.
\prettyref{lmm: local step perturbation} is used to bound $g_{i,k}$.
Combining them together, we have
\begin{small}
\begin{align*}
    &\mathbb{E}\left[ g_{i, k} S_{i, k} ~\big | ~ {\mathcal{F}}^k  \right] 
    =
    \mathbb{E}\left[  \frac{{(\tau_{i}^{+}(k) - k)^2}
    +{\tau_{i}^{+}(k) - k} }{2} 
      ~\big | ~ {\mathcal{F}}^k \right]
     \mathbb{E}\left[ g_{i, k} ~\big | ~{\mathcal{F}}^k \right]\\
    &~~~+
    \mathbb{E}\left[ (k - \tau_i(k)) 
    \left( \tau_{i}^{+}(k) - k \right) 
    +(k - \tau_i(k)) ~\big | ~ {\mathcal{F}}^k \right] 
    \mathbb{E}\left[ g_{i, k} ~\big | ~{\mathcal{F}}^k \right]\\
    &\le
    \left[ \frac{1}{c^2} + \frac{1}{2 c}
    + \left( \frac{1}{c} + 1 \right)
    (k - \tau_i(k))
     \right] s^2 \left( \kappa^2 \eta^2 s^2 L^2 + 2 \right)
    \left( \sigma^2 + \left\|{\nabla F_i(\bm{x}_i^k)} \right\|_2^2 \right).
\end{align*}
\end{small}
We use~\prettyref{ass: bounded true gradient} to bound $\left\|{\nabla F_i (\bm{x}_i^k)} \right\|_2 \le G_{\max}$.
Observe that the only remaining randomness comes from $(k -\tau_i(k))$.
Taking expectation over the remaining randomness and reusing~Lemma 7 to bound $(k - \tau_i(k))$ yields~\prettyref{lmm:gradient Bt} under $\eta \le 1/ (sL(\kappa + 1)$.

Recall that $\mathbf J = \frac{1}{m}\mathbf{1} \mathbf{1}^{\top}$ .
Unroll the recursion, we have
\begin{small}
\begin{align*}
    &\left\|{\bm{X}^{(t)} (\mathbf I - \mathbf J)} \right\|_{{\rm F}}^2
    \overset{(a)}{=} \eta^2
    \left\|{\sum_{q=0}^{t-1} 
    \left( \bm{G}^{(q)} - \bm{Z}^{(q)} \right) 
    \left( \prod_{l=q}^{t-1} W^{l} - \mathbf J \right)} \right\|_{{\rm F}}^2 \\

    \overset{(b)}{\le}
    2\left\|{\sum_{q=0}^{t-1} 
    \bm{G}^{(q)}
    \left( \prod_{l=q}^{t-1} W^{l} - \mathbf J \right)} \right\|_{{\rm F}}^2 
    +
    2\left\|{\sum_{q=0}^{t-1} 
    \bm{Z}^{(q)}
    \left( \prod_{l=q}^{t-1} W^{l} - \mathbf J \right)} \right\|_{{\rm F}}^2 
    ,
\end{align*}
\end{small}
where equality $(a)$ follows from the double-stochasticity, and inequality $(b)$ follows from Jensen's inequality.
By following a similar roadmap as in \citep{wang2022matcha,xiang2023towards,xiang2024efficient,xiang2025empowering},  Lemmas~\ref{lmm: local step perturbation} and~\ref{lmm: spectral norm main}, 
we can bound the term $\|\sum_{q=0}^{t-1} \bm{G}^{(q)} (\prod_{l=q}^{t-1} W^{l} - \mathbf J)\|_{\rm F}^2$.  


The key technical challenge for bounding $\|\sum_{q=0}^{t-1} \bm{Z}^{(q)} (\prod_{l=q}^{t-1}  W^{l} - \mathbf J)\|_{\rm F}^2$ comes from the fact that $\bm{Z}^{q}$ and $W^{q}$ are {\em not independent}.
Observe that, by construction, $\bm{Z}^{q}$ depends on ${\mathcal{A}}^q$ and $B_q$, which are both functions of ${\mathcal{A}}^q$.
We first upper bound $B_q$ with $C_q = \sum_{i=1}^{m} \|\Delta_{i}^{q}\|_2^2$ because $C_q$ and $({\mathcal{A}}^q, W^{(q)})$ are independent.
It is easy to see that $\prod_{l=q}^{t-1} W^{(l)} - \mathbf J = (W^{(q)} - \mathbf J)( \prod_{l=q+1}^{t-1} W^{(l)} - \mathbf J)$ due to double-stochasticity.
Let $\mathbb{I}_{{\mathcal{A}}^q} \triangleq \left[ {\mathbf{1}_{\left\{{1 \in {\mathcal{A}}^q}\right\}}}, \cdots, {\mathbf{1}_{\left\{{m \in {\mathcal{A}}^q}\right\}}} \right]$.
We then have $\bm{Z}^{q} =  \frac{B_q}{\gamma}  
\frac{{\mathbf{1}_{\left\{{\left | {\mathcal{A}}^q \right | > 0}\right\}}} \bm{z}^{q} \mathbb{I}_{{\mathcal{A}}^q}
}{\left | {\mathcal{A}}^q \right | + {\mathbf{1}_{\left\{{{\mathcal{A}}^{q} = 0}\right\}}}}$. 
We can show that 
\begin{small}
\begin{align*}
    &\Big \|\frac{{\mathbf{1}_{\left\{{\left | {\mathcal{A}}^q \right | > 0}\right\}}} \bm{z}^{q} \mathbb{I}_{{\mathcal{A}}^q} (W^{(q)} - \mathbf J)}{\left | {\mathcal{A}}^q \right | +   {\mathbf{1}_{\left\{{\left | {\mathcal{A}}^q \right | = 0}\right\}}}}\Big\|_{\mathrm{F}}^2 
    \le 
    \left\|{\bm{z}^{q}} \right\|_2^2
    \left\|{W^{(q)} - \mathbf J} \right\|_{{\rm F}}^2 
    .
\end{align*}
\end{small}

Using a similar technique to bounding $\|\sum_{q=0}^{t-1} \bm{G}^{(q)} (\prod_{l=q}^{t-1} W^{l} - \mathbf J)\|_{\rm F}^2$ and by~\prettyref{lmm:gradient Bt}, we complete 
bounding $\|\sum_{q=0}^{t-1} \bm{Z}^{(q)} (\prod_{l=q}^{t-1}  W^{l} - \mathbf J)\|_{\rm F}$.
Putting everything together, we obtain the desired consensus error bound in~\prettyref{lmm: consensus} with $\eta \le
(1 - \sqrt{\rho})/(632 s L (\kappa + 1) \sqrt{\beta^2 + 1})$.

<!-- remark-2: body_md -->
\label{rmk: convergence rate}
    \prettyref{thm: main} and \prettyref{cor: damped convergence} establish our final convergence bound.


    \noindent {\bf On the structures.} $(A)$ characterizes the initialization error, $(B)$ results from the consensus error, $(C)$ is due to noisy stochastic gradients (\prettyref{ass: bounded variance client-wise}) and inter-device gradient heterogeneity (\prettyref{ass: bounded similarity}), and $(D)$ stems from receiver noise.
    Notably, the receiver noise term in~\prettyref{thm: main} exhibits an ${\mathcal{O}} (1/m)$ reduction, where simultaneous transmission effectively averages out the receiver noise, consistent with the OTA literature \citep{sery2021over,abrar2025non}.

    
    \noindent {\bf On the convergence.} 
    \prettyref{cor: damped convergence} shows that $\bar{\bm{x}}^t$ converges to a
    stationary point of an {\em unbiased} non-convex objective $F$ at rate $1/\sqrt{T}$, which
    is optimal for any first-order method with access only to stochastic
    gradients~\citep{arjevani2023lower}, rather than to the biased auxiliary
    objective $\widetilde F$ in~\eqref{eq:biased convergence aux}. 
    Term~$(A)$ exhibits linear speedup in $m$ and $s$, and term~$(D)$ speeds up with $m$. 
    However, terms~$(B)$ and~$(C)$ ultimately dominate, consistent
    with the literature~\citep{Li2020}. 
    In future work, we will investigate how to achieve linear speedup in both the number of devices $m$ and the number of local steps $s$.

    
    
    \noindent {\bf On the role of uplink dynamics.}
    A larger $c$ implies a better uplink reliability and a smaller spectral norm $\rho$.
    Consequently, our bound on $\frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E} [\|\nabla F (\bar{\bm{x}}^t)\|_2^2]$ would become tighter.
    In the special case when $c=1$, i.e.\xspace, uplinks are always reliable,~{\tt FedOAG}~reduces to FedAvg with a receiver noise, and we also have $\rho = 0$.
    Without 
    receiver noise ($\sigma_z^2 = 0$),
    our bound reduces to ${\mathcal{O}}(\frac{1}{\sqrt{m s T}} + \sqrt{\frac{ms}{T}} (\sigma^2 + \zeta^2))$ and matches the FedAvg literature \citep{wang2020tackling}.
    When $p_i^t = c$ for all $i \in [m]$ and $t \in [T]$, i.e.\xspace, uniformly at random uplink availability. Setting $\eta = \sqrt{k/sT}$ in~\prettyref{thm: main}, our convergence rate becomes ${\mathcal{O}} (\frac{1}{\sqrt{k s T}} + \frac{\sigma^2 + \zeta^2}{(1 - \sqrt{\rho})^2}\sqrt{\frac{k}{sT}}) $, which introduces a larger variance compared to fully reliable, consistent with literature \citep{yang2021achieving}.
    Note that $\gamma$ also affects $c$.
    For example, we have $p_{i}^{t} = \exp(- \gamma^2/(d \Lambda_{i}^{t}E_s))$ in the special case of Rayleigh fading.
    When the parameters $d$, $\Lambda_{i}^{t}$, and $E_s$ are fixed, a smaller $\gamma$ leads to a greater $c$, yet at the expense of amplifying the receiver noise term $(D)$. 
    Similarly, a greater $\gamma$, which attenuates the receiver noise term $(D)$,  leads to a smaller $c$, i.e.\xspace, fewer active uplinks.
    In other words, the term $(D)$ reveals the fundamental tradeoff between uplink availability and signal quality.
    We demonstrate such a tradeoff empirically in~\prettyref{fig:impacts of gamma}. A principled choice of $\gamma$ would require explicit knowledge of the statistical CDIs; we defer the design of CDI-agnostic tuning frameworks for $\gamma$ to future work.
