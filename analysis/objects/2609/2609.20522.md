<!-- definition-1: body_md -->
Let $(X, \mu,T)$ be an ergodic system and $f \in L^{\infty}(\mu)$. The $k$-uniformity seminorms of $f$, $\left\lVertf\right\rVert_{U^{k}(X,\mu,T)}$, are defined inductively as follows:

<!-- proposition-2: body_md -->
\label{Prop G_tau}
    Let  $G_\tau = \langle G^\circ , \tau\rangle $, then $G_\tau$ is normal rational group in $G$. Moreover, if $X_\tau = G_\tau e_X \subset X$ then there is a finite set $D \subset {\mathbb Z}^d$ such that 
    \begin{equation} \label{eq partition Xtau}
        X = \bigsqcup_{h \in D} S^{h_1}_1 \cdots S^{h_d}_d X_\tau. 
    \end{equation}
    In particular, \eqref{eq partition Xtau} holds true for any translate of $D$ in ${\mathbb Z}^d$.
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
For the normality, since $G = \langle G^\circ , \tau, \sigma_1, \ldots, \sigma_d\rangle$ it is enough to prove that $\sigma_i G_\tau \sigma_i^{-1} = G_\tau$ for all $i =1, \ldots, d$. Fix $i\in [d]$. Using that $G^\circ$ is normal in $G$ and that $\tau$ commutes with $\sigma_i$ we get that
    \begin{align*}
        \sigma_i G_\tau \sigma_i^{-1} = \sigma_i \langle \tau, G^\circ \rangle \sigma_i^{-1} =  \langle \sigma_i \tau \sigma_i^{-1}, \sigma_i G^\circ \sigma_i^{-1} \rangle = \langle \tau, G^\circ \rangle = G_\tau.
    \end{align*}
    For the rationality, it is enough to notice
    $X_\tau$ is closed. Indeed, we  actually prove that if $X_0=G^\circ\Gamma$, then
\begin{equation}\label{eq-descriving-X-tau}
        X_\tau= \bigsqcup_{i=0}^r \tau^i X_0
    \end{equation}
    and since $X_0$ is closed (see \cite[Chapter 10]{Host_Kra_nilpotent_structures_ergodic_theory:2018}), then so is $X_\tau$. Moreover, $H$ can be taken to be generated by $\tau$ and $H^\circ$ the connected component of $H$ containing $e_G$, in particular $H$ is a closed subgroup in $G_\tau$. With this $Hx$ is a $T$-invariant subnilmanifold and $T \mu_x = \mu_{x}$ for all $x \in X_\tau$.  By the same proposition, $H$ is normal in $G_\tau$ (the original statement in \cite[Proposition 3.6]{hernandez2026nonergodicNil} assumes $G = G_\tau$, so adapting the proof to our context we get this version).

    \begin{equation} \label{eq disjoint part measure}
        \mu = \frac{1}{|D|} \sum_{h \in D} S_1^{h_1} \cdots S_d^{h_d} m_{X_\tau}. 
    \end{equation}

In particular, the map $x \mapsto S_1^{h_1} \cdots S_d^{h_d} \mu_{\psi_h (x)}$ from $S_1^{h_1} \cdots S_d^{h_d} X_\tau$ to ${\mathcal M}(X)$ is continuous.     \begin{align*}
        \overline{O_T(x)} = \overline{O_T(S_1^{h_1} \cdots S_d^{h_d} \psi_h (x))} = S_1^{h_1} \cdots S_d^{h_d}  \overline{O_T(\psi_h (x))}
    \end{align*}

    \begin{equation} \label{eq s1 s2 supp mu_x}
        S_1^{h_1} \cdots S_d^{h_d} H \psi_h(x) = S_1^{h_1} \cdots S_d^{h_d}  \operatorname{supp} \mu_{\psi_h(x)} = \operatorname{supp} \mu_x.
    \end{equation}
    Thus, for $S_1^{h_1} \cdots S_d^{h_d} m_{X_\tau}$-almost every $x$, $\mu_x$ is ergodic. 
    
    Taking the disjoint pieces we have defined a map $x \mapsto \mu_x$ from $X$ to ${\mathcal M}(X)$ that is continuous and such that by \eqref{eq disjoint part measure} fulfils that $\mu_x$ is ergodic for $\mu$ almost every $x \in X$. We conclude by noticing that for an arbitrary $f \in C(X)$

    \end{align*}
    concluding the proof.

<!-- proposition-6: body_md -->
\label{prop marginals}
    Let $(Z,m,T)$ be an ergodic pronilsystem.   For all $w \in  \operatorname{HP}_\ell(Z)$, $\rho_{w}^0 = \delta_{w_0}$ and $\rho_{w}^i \leq i m$ for all $i = 1, \ldots, \ell$.
<!-- end proposition-6 -->
<!-- proof-11: body_md -->
Notice that  $m$-almost every $w$, $\rho_w$ is ergodic, hence by properties of pronilsystems $(\operatorname{supp} \rho_w, \tilde T)$ is a minimal system (see \cref{preliminaries-2}). In particular, if $\tilde w \in \operatorname{supp} \rho_w$, then $\tilde w_0 = w_0$. This already ensures that $\rho_w^0 = \delta_{w_0}$ for $\lambda$-almost all $w \in \operatorname{HP}_\ell(Z) $ but, by continuity of the ergodic decomposition, we conclude it for all $ w \in \operatorname{HP}_\ell(Z)$. Similarly, since for $\lambda$-almost every $w \in \operatorname{HP}_\ell(Z)$, $(\operatorname{supp} \rho_w,\tilde{T})$ is uniquely ergodic, we get that for those $w \in \operatorname{HP}_\ell(Z)$ 
    \begin{equation} \label{eq casi def of sigma}
        \rho_w = \lim_{N \to \infty } \frac{1}{N} \sum_{n =1}^N \tilde T^n \delta_w  \implies \rho_w^i = \lim_{N \to \infty } \frac{1}{N} \sum_{n =1}^N T^{in} \delta_{w_i}
    \end{equation}
    for all $i =1, \ldots, \ell$, in particular $\rho_{w}^i \leq i m$. Again, by the continuity of the ergodic decomposition we derive that $\rho_{w}^i \leq i m$ for all $w \in \operatorname{HP}_\ell(Z)$.

<!-- theorem-8: body_md -->
\label{thrm new cont ergodic decomp} Let $(Z,m,T)$ be an ergodic pronilsystem and $\ell \in {\mathbb N}$,
    \begin{enumerate}[i)]
        \item \begin{equation} \label{eq second ergodic decomposition}
            m_{\operatorname{HP}_\ell(Z)} = \int_Z \rho_z \mathop{}\!\mathrm{d} m(z),
        \end{equation}
        \item \label{thrm point 2} for every $z \in Z$ and $w \in \operatorname{HP}_\ell(Z)$, $\rho_z = \rho_w$ if and only if $w_0 = z$, and
        \item \label{thrm point 3} for $m$-almost every $z \in Z$, $\rho_z = \sigma_z$. In particular,  $\operatorname{supp} \rho_z$ is uniquely ergodic with $\tilde T$-invariant measure $\rho_z$ for $m$-almost every $z \in Z$.  
    \end{enumerate}
<!-- end theorem-8 -->
<!-- proof-12: body_md -->
First, set $\nu = \int_Z \rho_z \mathop{}\!\mathrm{d} m(z)$. We have to prove that $\nu= m_{\operatorname{HP}_\ell(Z)} $. Clearly, $\operatorname{supp} \nu \subset \operatorname{HP}_\ell(Z)$ and since $\rho_z$ is $\tilde T$-invariant for all $z \in Z$, so is $\nu$. Using the $T$-invariance of $m$
    \begin{equation*}
        \nu = \int_Z \rho_z \mathop{}\!\mathrm{d} m (z) = \int_Z \rho_{Tz} \mathop{}\!\mathrm{d} m (z) = \int_Z T_\Delta \rho_{z} \mathop{}\!\mathrm{d} m (z) = T_\Delta \nu.
    \end{equation*}
    Thus $\nu$ is $\langle T_\Delta, \tilde T\rangle$-invariant and supported in $\operatorname{HP}_\ell(Z)$ which is $\langle T_\Delta, \tilde T\rangle$-uniquely ergodic, so we conclude the equality. 

 For \eqref{thrm point 2}, if $\rho_z=\rho_w$ for $z\in Z$, $w\in HP_\ell(Z)$, then by \cref{prop marginals} we get $w_0=z$. For the other implication, notice that, by \cref{rmk fully supp cont disintegration} and \cref{lemma ergodic fully supp decomposition}, for $w, w' \in \operatorname{HP}_\ell (Z)$, $w \in \operatorname{supp} \rho_{w '}$ if and only if $\rho_w = \rho_{w'}$. Thus, since by \cref{prop marginals}, for all $w \in \operatorname{HP}_\ell(Z)$, $$\operatorname{supp} \rho_w \subset \{ u \in \operatorname{HP}_\ell(Z) \colon u_0 = w_0\},$$ we get that $\rho_z$-almost every $w$ we have that $\rho_w=\rho_{P_0(w)}$, where $P_0 \colon HP_\ell(Z) \to Z$ is the projection to the $0$ coordinate. 
 Integrating and using \eqref{eq second ergodic decomposition}, we conclude that $m_{\operatorname{HP}_\ell(Z)}$-almost every $w$, we have $\rho_w=\rho_{P_0(w)}$ and by continuity this equality holds for every $w\in \operatorname{HP}_\ell(Z)$. 


                                                                                                                                                                                                                            
                       
                                                                                                                         
                     
                                                                                                                                                   
                                                                                                                                                                                                                                                                                 
                                                                       
                                                                                                                                                                                                                                                                                                      
                                                               
                                                                                                                                                                                                                                                                                                 

    For \eqref{thrm point 3}, first notice that with the previous point, $z \mapsto \rho_z$ defines a continuous ergodic decomposition. Let $W_z = \overline{{\mathcal O}_{\tilde T}(z,\ldots, z)} $ with its unique $\tilde T$-invariant measure $\sigma_z$. 
    By unique ergodicity in $(W_z,\tilde{T})$ and the dominated convergence theorem, we have that, for any $f_0, \ldots, f_\ell \in C(Z)$
    \begin{align*}
       \int_Z &\int_{\operatorname{HP}_\ell(Z)}  \bigotimes_{j=0}^\ell f_j \mathop{}\!\mathrm{d} \sigma_z \mathop{}\!\mathrm{d} m(z) = \int_Z \lim_{N \to \infty} \frac{1}{N} \sum_{n=0}^{N-1} \prod_{j=0}^\ell f_j(T^{jn}z)  \mathop{}\!\mathrm{d} m(z) \\
       &= \lim_{N \to \infty} \frac{1}{N} \sum_{n=0}^{N-1} \int_Z  \prod_{j=0}^\ell f_j(T^{jn}z) \mathop{}\!\mathrm{d} m(z) = \int_{\operatorname{HP}_\ell(Z)}  \bigotimes_{j=0}^\ell f_j \mathop{}\!\mathrm{d} m_{\operatorname{HP}_\ell(Z)}.
    \end{align*}
    Therefore, $z \mapsto \sigma_z$ defines a (non-continuous) ergodic decomposition of $m_{\operatorname{HP}_\ell(Z)}$ with this $\rho_z = \sigma_z$ for $m$-almost all $z \in Z$ concluding the proof.

(Good projection property)  \label{prop good projection properties}

        \int_{Z^k} f_1 \otimes \cdots \otimes f_k \mathop{}\!\mathrm{d} \rho_z^L = \int_{Z^k_{k-1}} \mathbb{E}(f_1\mid Z_{k-1}) \otimes \cdots \otimes \mathbb{E}(f_k \mid Z_{k-1}) \mathop{}\!\mathrm{d} \eta^L_{\pi_{k-1}(z)}

Then by unique ergodicity and by the main result of Host and Kra in \cite{Host_Kra_nilpotent_structures_ergodic_theory:2018}, for all $f_1, \ldots, f_k \in C(Z)$, we get that
    \begin{align*}
        \left\lVert \lim_{N \to \infty}  \frac{1}{N} \sum_{n=0}^{N-1} \prod_{i=1}^k T^{\ell_in} f_i(\bullet)   - \int_{Z^L} \bigotimes_{i=1}^k f_i  \mathop{}\!\mathrm{d} \sigma_{\bullet}^L\right\rVert_{L^2(m)} =0 \\

    \end{align*}
    where we represent the integration variable with $\bullet$ for clarity, and in the last equality we are using that $u \mapsto \mathbb{E}(f_i \mid Z_{k-1})(u)$ agrees with a continuous function in a set of full measure by Lemmas \ref{prop cont disintegration and conditional expectation} and \ref{prop cont k-step max pronilfactors}.         \int_{Z^L} \bigotimes_{i=1}^k f_i  \mathop{}\!\mathrm{d} \sigma_z^L = \int_{Z^L_{k-1}} \bigotimes_{i=1}^k \mathbb{E}(f_i \mid Z_{k-1})  \mathop{}\!\mathrm{d} \nu_{\pi_{k-1}(z)}^L.  
    \end{equation}
    For $u \in Z_{k-1}$ we define $\breve \nu_u, \breve \eta_u \in {\mathcal M}(Z^k)$ the measures defined by

        \int_{Z^L} \bigotimes_{i=1}^k f_i  \mathop{}\!\mathrm{d} \breve \nu_u = \int_{Z^L_{k-1}} \bigotimes_{i=1}^k \mathbb{E}(f_i \mid Z_{k-1})  \mathop{}\!\mathrm{d} \nu_{u}^L \\

We say that a  measure $\tau \in {\mathcal M}(X^{\ell+1})$ is uniformly progressive along $\Phi$ if for every $\eta >0$ there exists $\delta >0$ such that for every continuous and non-negative functions $f_1, \ldots, f_\ell \in C(X)$ bounded by $1$ whenever

\end{equation*}

\begin{equation}\label{conclusion-Uniform-progressiveness}
     \limsup_{N \to \infty }\frac{1}{|\Phi_N|} \sum_{n \in \Phi_N} \int_{X^{\ell+1}}(\1 \otimes f_1 \otimes \cdots \otimes f_\ell) \cdot T_{\Delta}^n (f_1 \otimes \cdots \otimes f_\ell \otimes \1) \mathop{}\!\mathrm{d} \tau \geq \delta.
\end{equation}

\label{theorem key}

For $(\operatorname{HP}_\ell(Z), m_{\operatorname{HP}_\ell(Z)}, \tilde T, T_\Delta)$ consider $z \mapsto \rho_z$ and $w \mapsto \lambda_w$ the continuous $\tilde T$-ergodic and $T_\Delta$-ergodic decompositions of $m_{\operatorname{HP}_\ell(Z)}$ respectively. <!-- proof-14: body_md -->
The case $Z = G/\Gamma$ being and $s$-step nilsystem is direct from \cref{theorem cross ergodic decompositions} and the fact that  $\operatorname{HP}_\ell(G)=\langle\operatorname{HP}_\ell(G)^\circ, \tilde \tau, \tau_\Delta \rangle$, (see \cite[proof of Theorem 5, Chapter 15]{Host_Kra_nilpotent_structures_ergodic_theory:2018}). 
    
    For the inverse limit case, let $(Z_i, m_i, T)$ be the sequence of $s$-step nilsystem such that $(Z, m, T) = \lim_{\leftarrow} (Z_i, m_i, T)$ and denote $\theta_i \colon Z \to Z_i$ the factor map. We also denote $\tilde \theta_i = \theta_i \times \cdots \times \theta_i$ that maps $\operatorname{HP}_\ell(Z)$ to $\operatorname{HP}_\ell(Z_i)$.
    Fix $z \in Z$. Since the span of $$\{ f \circ \tilde \theta_i \colon f \in C(\operatorname{HP}_\ell(Z_i)), i \in {\mathbb N}\}$$ is dense in $C(\operatorname{HP}_\ell(Z))$ and $C(\operatorname{HP}_\ell(Z_i))$ is separable, it is enough to prove that for $\rho_z$-almost every $w \in \operatorname{HP}_\ell(Z)$, if $f \in C(\operatorname{HP}_\ell(Z_i))$ then $\lim_{N \to \infty} \frac{1}{N} \sum_{n =1}^N f\circ \tilde \theta_i(T^n_\Delta w) = \int f \circ \tilde \theta_i \mathop{}\!\mathrm{d} \lambda_w$. 
    
    For $i \in {\mathbb N}$, denote $u \in Z_i \mapsto \rho^{Z_i}_u$ the $\tilde T$-continuous ergodic decomposition of $m_{\operatorname{HP}_\ell(Z_i)}$. Similarly for $\lambda_w^{Z_i}$.  Let $W_i \subset \operatorname{HP}_\ell(Z_i)$ be the measurable set with $\rho_{\theta_i(z)}^{Z_i}(W_i) =1$ such that $\lambda_w^{Z_i}$ is ergodic for all $w \in W_i$. Notice that $\rho_{\theta_i(z)}^{Z_i}(W_i) = \rho_{z}(\tilde{\theta}_i^{-1}(W_i))$ and therefore the set $$ W= \bigcap_{i \in {\mathbb N}} \tilde{\theta}_i^{-1}(W_i)$$ has full measure for $\rho_z$. Then for $w \in W$ and $f \in C(\operatorname{HP}_\ell(Z_i))$ for some $i\in {\mathbb N}$
    \begin{equation*}
        \lim_{N \to \infty} \frac{1}{N} \sum_{n \leq N} f\circ \tilde \theta_i(T^n_\Delta w) = \lim_{N \to \infty} \frac{1}{N} \sum_{n \leq N} f(T^n_\Delta \tilde\theta_i( w)) = \int_{\operatorname{HP}_\ell(Z_i)} f \mathop{}\!\mathrm{d} \lambda_{\tilde\theta_i(w)}^{Z_i} = \int_{\operatorname{HP}_\ell(Z)} f \circ \tilde \theta_i \mathop{}\!\mathrm{d} \lambda_{w}.
    \end{equation*}
    In particular, we have that $\lambda_{\tilde\theta_i(w)}^{Z_i}= (\tilde\theta_i)_*\lambda_{w}$ from which the ergodicity of $\lambda_w$ follows, concluding the proof.

    \end{equation*}
