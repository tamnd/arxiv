\label{thm:GR}
For each $k\geq 1$,
let $d = \left\lfloor (k+1)/2\right\rfloor - \delta_{2+4\mathbb Z}(k)$
where $\delta_{2+4\mathbb Z}$ is the characteristic function
supported on $2+4\mathbb Z \subset \mathbb Z$.
There exists a compatible family of continuous $\ell$-adic representations
for prime numbers $\ell$
\[ \rho_{k,\ell}\colon \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb Q) \longrightarrow \operatorname{GL}_d(\mathbb Q_\ell), \]
which is unramified and satisfying
\[ \operatorname{tr}\big(\rho_{k,\ell}(\mathrm{Frob}_p)\big) = -\mathsf{m}_k(p) - \delta_{2+4\mathbb Z}(k)p^{k/2} \]
at prime $p\neq\ell$
and $p > k$ (resp., $p>k/2$) if $k$ is odd (resp., $k$ is even).
Here $\mathrm{Frob}_p$ is a geometric Frobenius element at $p$
in $\operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb Q)$.
Moreover, $\rho_{k,\ell}$ is self-dual, up to a Tate twist,
is de Rham above $\ell$ and crystalline
if $\ell > \max\{k,2\}$ (resp., $\ell >k/2$) when $k$ is odd (resp., $k$ is even),
and is potentially automorphic.

\label{thm:EC}
For a cuspidal Hecke eigenform $f$ on some congruence subgroup,
let
$f(q)=\sum_{n=1}^\infty a_f(n)q^n$
be its $q$-expansion.
The following modularity holds.
\begin{enumerate}
\item
Let $f$ be the modular form $f_{\text{48.3.e.a}}$ (CM by $\mathbb Q(\sqrt{-3}))$
in {\texttt{LMFDB}}.
Then, for $p\geq 5$,
\begin{equation}\label{eq:Evans3}
\mathsf{m}_3(p) = -a_f(p)\cdot p.
\end{equation}
\item
Let $f$ be the modular form $f_{\text{8.4.a.a}}$ in {\texttt{LMFDB}}.
Then, for each odd prime $p$,
\[ \mathsf{m}_4(p) = -a_f(p)\cdot p \]
\item
Let $f$ be the modular form $f_{\text{60.3.b.a}}$ in {\texttt{LMFDB}}
(with $a_f(n)\in \mathbb Q(\sqrt 5, i)$).
Then, when $p\geq 7$,
\begin{equation}\label{eq:Evans5}
\mathsf{m}_5(p) =
- \Bigl(\frac{-1}{p}\Bigr)a_f(p)^2\cdot p + \Bigl(\frac{15}p\Bigr)p^3,
\end{equation}
where $\(\frac{\cdot}p\)$ is the Legendre symbol. 
\item
Let $f$ be the modular form $f_{\text{24.6.a.a}}$ in {\texttt{LMFDB}}.
Then, when $p\geq 5$,
\[ \mathsf{m}_6(p) = -a_f(p)\cdot p-p^3. \]
\end{enumerate}

\label{prop:FW}
As $\operatorname{Gal}(\overline{\mathbb{F}}_p/\mathbb F_p)$-modules,
there is the exact sequence
\[ 0 \longrightarrow (\mathscr{V}_k^+)^{I_0}\oplus (\mathscr{V}_k^+)^{I_\infty}
\longrightarrow \mathrm{H}_{\operatorname{\acute{e}t},\mathrm{c}}^1(\operatorname{\mathbb{G}_{\mathrm{m},\overline{\mathbb F}_\mathnormal{p}}},\mathscr{V}^+_k)
\longrightarrow \mathrm{H}_{\operatorname{\acute{e}t},\mathrm{mid}}^1(\operatorname{\mathbb{G}_{\mathrm{m},\overline{\mathbb F}_\mathnormal{p}}},\mathscr{V}^+_k)
\longrightarrow 0. \]
The space $(\mathscr{V}_k^+)^{I_0}$ is of dimension one,
upon which $\operatorname{Gal}(\overline{\mathbb{F}}_p/\mathbb F_p)$ acts trivially.
The following properties hold.
\begin{enumerate}
\item
Suppose $k$ is odd.
Then $(\mathscr{V}_k^+)^{I_\infty} = 0$
and
\begin{equation}\label{eq:FWdet_odd}
\det\big(F_p\mid\mathrm{H}_{\operatorname{\acute{e}t},\mathrm{mid}}^1(\operatorname{\mathbb{G}_{\mathrm{m},\overline{\mathbb F}_\mathnormal{p}}},\mathscr{V}^+_k)\big)
= p^{\frac{k+1}{2}\delta^+}
\left(\frac{-2}{p}\right)^{\left\lfloor \frac{k}{2p}+\frac{1}{2}\right\rfloor}

\[ \delta^+ = \dim \mathrm{H}_{\operatorname{\acute{e}t},\mathrm{mid}}^1(\operatorname{\mathbb{G}_{\mathrm{m},\overline{\mathbb F}_\mathnormal{p}}},\mathscr{V}^+_k)
= \frac{k-1}{2}-\left\lfloor \frac{k}{2p}+\frac{1}{2}\right\rfloor. \]
\item
Suppose $k$ is even.
\begin{enumerate}
\item
$\operatorname{Gal}(\overline{\mathbb{F}}_p/\mathbb F_p)$ acts on $(\mathscr{V}_k^+)^{I_\infty}$
semisimply and
\[ \det(1-F_pT \mid (\mathscr{V}_k^+)^{I_\infty})

(1-(-1)^{(p-1)/2}p^{k/2}T)^{n_k^-(p)}

\begin{equation}\label{eq:n_k}
n_k^+(p) = \left\lfloor \frac{k}{4p}\right\rfloor,
\quad

\end{equation}
\item

$\dim \mathrm{H}_{\operatorname{\acute{e}t},\mathrm{c}}^1(\operatorname{\mathbb{G}_{\mathrm{m},\overline{\mathbb F}_\mathnormal{p}}},\mathscr{V}^+_k) = \frac{k}{2}-\left\lfloor \frac{k}{2p}\right\rfloor$.

\[ \det\big(F_p\mid\mathrm{H}_{\operatorname{\acute{e}t},\mathrm{mid}}^1(\operatorname{\mathbb{G}_{\mathrm{m},\overline{\mathbb F}_\mathnormal{p}}},\mathscr{V}^+_k)\big)
= p^{\frac{k+1}{2}\delta^+} \]
where
\[ \delta^+ = \dim \mathrm{H}_{\operatorname{\acute{e}t},\mathrm{mid}}^1(\operatorname{\mathbb{G}_{\mathrm{m},\overline{\mathbb F}_\mathnormal{p}}},\mathscr{V}^+_k)
= \frac{k}{2}-2\left\lfloor \frac{k}{2p}\right\rfloor - \delta_{4\mathbb Z}(k). \]
\end{enumerate}
\end{enumerate}

\label{lem:sVt_infty}
We have the decomposition of
$\widetilde{\mathscr{V}}_k|_{\eta_{\widetilde{\infty}}}$
as follows.
\begin{enumerate}
\item
If $k=2r+1$ is odd,
\begin{equation}\label{eq:decompKF}
\widetilde{\mathscr{V}}_k|_{\eta_{{\widetilde{\infty}}}}
= \bigoplus_{i=0}^r \mathcal{V}_i^{\oplus 2},
\quad
\mathcal{V}_i = \mathcal{K}_i\otimes\mathcal{F}_{i+1}
\end{equation}
with Swan conductor
$\operatorname{sw}(\mathcal{V}_i) = 1-\delta_{\Xi_k(p)}(i)$.
In particular,
$\widetilde{\mathscr{V}}_k^{I_{\widetilde{\infty}}} = 0$.
\item
In case $k=2r$ is even,
\[ \widetilde{\mathscr{V}}_k|_{\eta_{{\widetilde{\infty}}}}
= \mathcal{F}_r
\oplus \bigoplus_{i=0}^{r-1} (\mathcal{V}_i^+)^{\oplus 2},
\quad
\mathcal{V}_i^+ = \mathcal{K}_i^+\otimes\mathcal{F}_i \]
with Swan conductor
$\operatorname{sw}(\mathcal{V}_i^+) = 1-\delta_{\Xi_k(p)}(i)$.
Moreover,
$\operatorname{Gal}(\overline{\mathbb{F}}_p/\mathbb F_p)$ acts on $\widetilde{\mathscr{V}}_k^{I_{\widetilde{\infty}}}$
semisimply and
\begin{equation}\label{eq:ch_poly_inftyt}
\det(1-F_pT \mid \widetilde{\mathscr{V}}_k^{I_{\widetilde{\infty}}})
= (1-p^{k/2}T)^{1+2n_k^+(p)}(1-(-1)^{(p-1)/2}p^{k/2}T)^{2n_k^-(p)}
\end{equation}
with $n_k^{\pm}(p)$ defined previously in \eqref{eq:n_k}.
\end{enumerate}
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Recall \cite[Lemma 2.4]{FW10} that
\[ {\mathscr{V}_k^+}|_{\eta_{\infty}}
= \begin{cases}\displaystyle
\bigoplus_{i=0}^r [2]_*\mathcal{K}_i
	\otimes\mathcal{F}_{i+1}
& \text{for $k=2r+1$ odd}, \\
\displaystyle
\kappa_2^r\otimes\mathcal{F}_r
\oplus \bigoplus_{i=0}^{r-1} [2]_*\mathcal{K}_i^+
	\otimes\mathcal{F}_i
& \text{for $k=2r$ even}.
\end{cases} \]
Pulling back to $\eta_{\widetilde{\infty}}$,
the statements follow,
except for the last equality \eqref{eq:ch_poly_inftyt}.
For $i \in \Xi_k(p)$,
$\mathcal{K}_i^+$ is trivial and $\mathcal{V}_i^+ = \mathcal{F}_i$,
so one has
\[ \widetilde{\mathscr{V}}_k^{I_{\widetilde{\infty}}}
= \mathcal{F}_r
\oplus \bigoplus_{i\in\Xi_k(p)} \mathcal{F}_i^{\oplus 2}. \]
By \eqref{eq:gauss_square},
we have $F_p|_{\mathcal{F}_r} = (\theta_0^2\theta_1)^r(F_p) = p^r$ and,
for $i\in\Xi_k(p)$,
\begin{align*}
F_p|_{\mathcal{F}_i} = (\theta_0^k\theta_1^i)(F_p)
&= \big((-1)^{(p-1)/2}\big)^{r-i}p^r \\
&= \begin{cases}
p^r, & (k-2i)/p \equiv 0 \bmod{4}, \\
(-1)^{(p-1)/2}p^r, & (k-2i)/p \equiv 2 \bmod{4}. \\
\end{cases}
\end{align*}
Noticing the counting
\begin{align*}
\#\{ i\in\Xi_k(p) \mid (k-2i)/p \equiv 0\bmod{4} \}
&= \left\lfloor \frac{k}{4p}\right\rfloor = n_k^+(p), \\
\#\{ i\in\Xi_k(p) \mid (k-2i)/p \equiv 2\bmod{4} \}
&= \left\lfloor \frac{k}{4p}+\frac{1}{2}\right\rfloor = n_k^-(p),
\end{align*}
the formula \eqref{eq:ch_poly_inftyt} then follows.

\begin{lem}\label{ControlFrob}

$I_p \subset \operatorname{Gal}(\overline{\mathbb{Q}}_p/\mathbb Q_p)$

$\operatorname{diag}(1,-1)$
(up to scaling).
Moreover,
the image $r'_\ell(\operatorname{Gal}(\overline{\mathbb{Q}}_p/\mathbb Q_p))\subset \operatorname{PGL}_2(\overline{\mathbb{Q}}_\ell)$
is conjugate to a subgroup of the diagonal matrices
(up to scaling).
\end{lem}

Since $r_\ell$ is tamely ramified at odd primes,
the same holds for $r_\ell'$.
For $p=3, 5$,

Since $\dim_{\mathbb Q_\ell}V_{5,\ell}^{I_p}=2$,
$\rho_\ell(\sigma)$ is similar to $\operatorname{diag}(1,1,-1)$.
As a result, $r_\ell(\sigma)$ is similar to $\operatorname{diag}(-1,-1,1)$
in $\operatorname{SO}(3)({\overline{\mathbb{Q}}_\ell})$,
and hence $r'_\ell(\sigma)$ is similar to
$\operatorname{diag}(1,-1)$ in $\operatorname{PGL}_2({\overline{\mathbb{Q}}_\ell})$.
In the following,

$r'_\ell(\sigma) = \operatorname{diag}(1,-1)$
(up to scaling).

be a lifting of the geometric Frobenius.
We have $\phi_p^{-1}\sigma\phi_p=\sigma^p$ in $I_{p}^t$,

$\bigl( \begin{smallmatrix}
a&0\\ 0&d
\end{smallmatrix} \bigr)$
or
$\bigl( \begin{smallmatrix}
0&b\\ c&0
\end{smallmatrix} \bigr)$.

$r_\ell(\phi_p)$ are then similar to
\[ \begin{pmatrix}
ad^{-1} \\
& a^{-1}d \\
&& 1 \end{pmatrix},
\quad
\begin{pmatrix}
& bc^{-1} \\
b^{-1}c \\
&& -1 \end{pmatrix}, \]

$(\overline{\mathbb{Q}}_\ell^3)^{\sigma +1}$ occor in the upper-left parts.

\begin{prop}[Cf.\,{\cite[Lemma 4.7.6]{Yun}}]
\label{lift}
For every prime $\ell > 5$, there exists a lifting
\[ \widetilde{r}_\ell \colon \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb Q) \longrightarrow \operatorname{GL}_2(\overline{\mathbb{Q}}_\ell) \]
of $r'_\ell\colon \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb Q) \to \operatorname{PGL}_2(\overline{\mathbb{Q}}_\ell)$
satisfying the following properties:
\begin{itemize}
    \item For prime $p\neq 2, 3, 5, \ell$, the representation $\widetilde{r}_\ell$ is unramified at $p$.
    \item For prime $p=3, 5$, the representation $\widetilde{r}_\ell$ is tamely ramified at $p$,
and it sends a generator of $I_p^t$
to a matrix conjugate to $\operatorname{diag}(1,-1)$.
\item
$\widetilde{r}_\ell$ is crystalline at $\ell$
and its Hodge-Tate weight equals $(2,0)$.
\item
$\det\widetilde{r}_\ell = \big(\frac{-15}{\bullet}\big)\otimes\chi_{\mathrm{cyc},\ell}^{-2}$.
\end{itemize}
We then have
$r_\ell = \operatorname{Sym}^2\widetilde{r}_\ell \otimes \det^{-1}\widetilde{r}_\ell$
over $\overline{\mathbb{Q}}_\ell$.
\end{prop}

\begin{proof}
By \cite[Prop.5.5]{Pat15},
there exists a lifting
$\Phi\colon \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb Q) \to \operatorname{GL}_2(\overline{\mathbb{Q}}_\ell)$ of $r'_\ell$,
which is unramified almost everywhere
and is de Rham at $\ell$.

For $p\neq 2, 3, 5, \ell$,
we have $\Phi|_{I_p} = c_p\mathrm{Id}$
for some character $c_p$ of $I_p$
since $r'_\ell|_{I_p}$ is trivial.
Note that $c_p$ are trivial for almost all $p$ since $\Phi$ is unramified almost everywhere.

For $p=3,5$,
the representation $\Phi|_{\operatorname{Gal}(\overline{\mathbb{Q}}_p/\mathbb Q_p)}$
is isomorphic to $d_p\oplus d_p^{\prime}$
by Lemma \ref{ControlFrob}.
The restrictions of $d_p$ and $d_p^{\prime}$ at the wild inertia group are the same
since $r_\ell'$ is tamely ramified.
We denote $c_p$ the restriction of $d_p$ to the inertia group.

Since $r_\ell'$ is crystalline
and the lifting $\Phi$ is de Rham at $\ell$,
by \cite[Cor.6.7]{Conrad},
there exists a crystalline lifting $\phi$ of
$r'_\ell|_{\operatorname{Gal}(\overline{\mathbb{Q}}_\ell/\mathbb Q_\ell)}$.
We may assume that $\phi$ has Hodge-Tate weight $(2,0)$
after a suitable Tate twist.
Then $\Phi|_{\operatorname{Gal}(\overline{\mathbb{Q}}_\ell/\mathbb Q_\ell)} = \phi\cdot d_\ell$
for a continuous character
$d_\ell\colon \operatorname{Gal}(\overline{\mathbb{Q}}_\ell/\mathbb Q_\ell) \to \overline{\mathbb{Q}}_\ell^\times$.
Take $c_\ell = d_\ell|_{I_\ell}$.

At $p=2$,

$\{\pm 1\}\times \mathbb Z/2^a$.

such that $\det\Phi|_{I_2}(1,1) = \alpha^2$.
Let $c_2: I_2 \to \overline{\mathbb{Q}}_\ell^\times$
given by $c_2|_{\{\pm 1\}} = 1$
and $c_2(1,1) = \alpha$

\begin{lem}\label{lem:abs_irred}

\begin{proof}
(Cf.\,\cite[(4.8.9)]{Serre-GQ},
\cite[Thm.1.2.1]{Kisin},
\cite[Lemma 4.7.2]{Yun}
and their proofs.)
Suppose, on the contrary, that there are infinitely many primes $\ell$
such that $\overline{\widetilde{r}}_\ell$
is not irreducible.
So, for these $\ell$,
there are characters
$\psi_1,\psi_2\colon \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb Q) \to \overline{\mathbb{F}}_\ell^\times$
such that
$\overline{\widetilde{r}}_\ell \simeq \psi_1\oplus\psi_2$
up to semisimplification.
Then,
for the reduction $\overline{r}_\ell$ of $r_\ell$ in \eqref{eq:repre_r},
we have
$\overline{r}_\ell \simeq \psi_1\psi_2^{-1}\oplus 1\oplus \psi_1^{-1}\psi_2$.
Let $N$ be the conductor of $r_\ell$.
Since $r_\ell$ is de Rham at $\ell$,
there is a Dirichlet character
$\chi_0\colon \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb Q) \to \overline{\mathbb{Q}}^\times\subset\overline{\mathbb{Q}}_\ell^\times$
of conductor dividing $N$
such that $\psi_1\psi_2^{-1} = \overline{\chi}$
where $\overline{\chi}$ is the reduction of $\chi = \chi_{\mathrm{cyc},\ell}^{-2}\chi_0$
(after exchanging $\psi_1,\psi_2$ if needed).
In particular,
by the evaluation of
$\rho_{5,\ell} = r_\ell\chi_{\mathrm{cyc},\ell}^{-3}\big(\frac{15}{\bullet}\big)$
at $\mathrm{Frob}_p$
for $p>5, p\neq\ell$,
one has
\begin{equation}\label{eq:m5_cong}
-\Bigl(\frac{15}{p}\Bigr)\mathsf{m}_5(p)
\equiv p^3\big(p^2\chi_0(p) + 1 + p^{-2}\chi_0^{-1}(p)\big) \bmod\ell.
\end{equation}
Since the congruence holds for arbitrarily large $\ell$
and the involved characters $\chi_0$ have bounded conductors,
there exists a fixed such $\chi_0$
making \eqref{eq:m5_cong}
an equality in the ring of integers of a fixed cyclotomic field
(adding values of $\chi_0$ to $\mathbb Q$).

On the other hand,
the integer $\mathsf{m}_5(p)$
is in the range $[-3p^3,3p^3]$
since $\rho_{5,\ell}$ is of rank 3 and weight 6.
However,
one has
\[ |p^5\chi_0(p) + p^3 + p\chi_0^{-1}(p)| \geq p^5-p^3-p, \]
and the right-hand side is greater than $3p^3$ for $p\geq 5$,
yielding a contradiction.

\begin{lem}\label{lemma:tm5}
For $p\equiv 5 \bmod 6$, we have
	$$
                                                       
\frac{-\mathsf{m}_5(p)}p  +\Bigl(\frac{15}{p}\Bigr) p^2 
 \quad	\in  5\mathbb Z.

	Furthermore, up to $p=2357$ (the $350$th prime), the values  $\mathsf{m}_5(p)$ are of the form
	\begin{align*}

		\begin{cases}
			m_p^2, & \mbox{ if } p \equiv 1 \bmod 6,\\
			-5n_p^2, & \mbox{ if } p \equiv 5 \bmod 6,

	where $m_p$ and $n_p \in \mathbb Z$.  
	
\end{lem}

Immediately from the definition, one has
$$
  \operatorname{Kl}_2^{\operatorname{Sym}^5}(a;p)                                                  
  =  \operatorname{Kl}_2^{\operatorname{Sym}^1}(p;a)^5 -4p  \operatorname{Kl}_2^{\operatorname{Sym}^3}(a;p) -5\operatorname{Kl}_2^{\operatorname{Sym}^1}(p;a)p^2.

By the direct evaluation \eqref{eq:small_moments} of $\mathsf{m}_1(p)$,

  \end{align*}
  for  $p\equiv 5 \bmod 6$.   	Observe that 
	   $$
	  \frac{-\mathsf{m}_5(p)}p \equiv \frac 1p  \sum_{a\in\mathbb F_p^\times}\Big(\frac ap\Big)\operatorname{Kl}_2(p;a)^5 \equiv \frac 1p \sum_{a\in\mathbb F_p^\times}\Big(\frac ap\Big)\operatorname{Kl}_2(p;a)  = \Bigl(\frac{-1}{p}\Bigr) \equiv   \Bigl(\frac{-5}{p}\Bigr) p^2\bmod 5,

as in Proposition \ref{lift}.

We may assume that $\bar{\widetilde{r}}_\ell$ is absolutely irreducible
by choosing $\ell\gg 0$, by Lemma \ref{lem:abs_irred}.  
The Artin conductor of $\widetilde{r}_\ell$ is $2^c\cdot3\cdot5$,
for some $c \in \mathbb Z_{\geq 0}$,
by Proposition \ref{lift}. By Proposition \ref{prop:HodgeNumbers},
we deduce that the Serre weight of  $\widetilde{r}_\ell$ is $3$.
The assertion of Serre's modularity conjecture
tells us that there exists a newform $f$ of level $2^c\cdot15$
for some $0\leq c\leq 8$,
weight $3$, character $\epsilon_f$ with $q$-expansion $\sum a_f(n)q^n$ and associated Galois representation $\rho_f$ such that the reduction mod $\ell$ representation of $\rho_f$ is isomorphic to $\bar{\widetilde{r}}_\ell$.

By Proposition \ref{lift}, the character $\epsilon_f$ is induced from $\(\frac{-15}{\bullet}\)$.
Therefore, the target Hecke eighenform $f$ satisfies that
\begin{gather*}
    a_f(p)^2-{\Big(\frac{-15}{p}\Big)}p^2
    = {\Big(\frac{-15}{p}\Big)} p^2\cdot \operatorname{tr} (r_\ell(\mathrm{Frob}_p))
    = {\Big(\frac{-1}{p}\Big)}  \frac{-\mathsf{m}_5(p)}{p},
\end{gather*} 
for prime $p \geq 7$. 
Further by Lemma \ref{lemma:tm5} for the case of $p\equiv 5\bmod 6$, we must have
$$
    \(\frac{a_f(p)}{\sqrt{-5} n_p}\)^2={\Big(\frac{-15}{p}\Big)\Big(\frac{-3}p\Big)
    =\Big(\frac{5}{p}\Big)}, \quad \mbox{for some } n_p\in \mathbb Z.
$$
Also, for $p\equiv 1\bmod{6}$ with  $p\leq 2357$, we have 
	\[ 
	\(\frac{a_f(p)}{m_p}\)^2=\Big(\frac{-15}{p}\Big)\Big(\frac{-3}p\Big)
    =\Big(\frac{5}{p}\Big), \quad \mbox{for some } m_p\in \mathbb Z.
    \]
	So
	$a_f(p) = \pm m_p, \pm m_pi, \pm\sqrt{5}n_p, \pm\sqrt{-5}n_p$. Since  the Sturm bound for the  space of modular form of level $2^8\cdot 15$ with a given character is $2304$,  we conclude that the coefficient field $K_f:=\mathbb Q(a_f(n) : n \geq  1)$ is $\mathbb Q(\sqrt{5},i)$. 
	
	Taking $f$ to be any of the normalized newform in the space $S_3\big(60,  \big(\frac{-15}{\bullet}\big)\big)$ with $K_f = \mathbb Q(\sqrt{5},i)$, one can check the validity of the identity numerically

	\quad
	\text{for primes $7\leq p \leq 2309$},
	$$
	and this confirms the claimed modularity for $\mathsf{m}_5$. 
	


\begin{rem}\label{rem:Evans5}
Suppose $f$ satisfies the equality \eqref{eq:Evans5}
in Theorem \ref{thm:EC}.
Let $g=f\otimes\psi$ be the twist of $f$ by a character $\psi$.
Then the coefficients $a_g(p) = \psi(p)a_f(p)$ satisfy
a similar equality,
like \eqref{eq:Evans5_2} below.
In this sense,
the modular form $f$ is the one with minimal level
that interprets $\mathsf{m}_5(p)$ in terms of
(the symmetric squares of) modular forms.
We refer the computation of the level of a twist to \cite[Lemma 11.2.1]{Computing}.

In particular,
let $g$ be one of the forms
$f_{\text{300.3.g.e}}$ and $f_{\text{300.3.g.h}}$ in {\texttt{LMFDB}},
which are the twists of $f_{\text{60.3.b.a}}$
by the characters
$\chi = \chi_{5.c}$ or $\chi_{15.e}$.
We have $\chi^2(n) = \big(\frac{5}{n}\big)$,
and hence one has
\begin{equation}\label{eq:Evans5_2}
\Big(\frac{-3}{p}\Big)a_g(p)^2
= p^2\Big[\Big(\frac{15}{p}\Big)\frac{-\mathsf{m}_5(p)}{p^3} + 1\Big]
\end{equation}
for prime  $p>5$.
On the other hand,
we have
$\big(\frac{15}{p}\big)\frac{-\mathsf{m}_5(p)}{p^3} = \operatorname{tr}(r_\ell(\mathrm{Frob}_p)) \in \mathbb Q$
for $r_\ell(\mathrm{Frob}_p) \in \operatorname{SO}(V_{5,\ell})$
in \eqref{eq:trace_r_ell}.
Now choosing a basis of $(V_{5,\ell})_{\overline{\mathbb{Q}}_\ell}$
and taking an isomorphism $\overline{\mathbb{Q}}_\ell \cong \mathbb C$
as abstract fields,
we regard $r_\ell(\mathrm{Frob}_p)$ as an element in $\operatorname{SO}(3,\mathbb C)$.
Regarding traces, we may assume that $r_\ell(\mathrm{Frob}_p)$
is semisimple.
Then $r_\ell(\mathrm{Frob}_p)$ lies in a compact subgroup of $\operatorname{SO}(3,\mathbb C)$
since the eigenvalues have absolute value one.
Therefore $r_\ell(\mathrm{Frob}_p)$ is conjugate to an element
in the compact form $\operatorname{SO}(3)$.
For $T\in\operatorname{SO}(3)$,
one has
$\operatorname{tr}(T) = 1+2\cos\theta \geq -1$
since every nontrivial rotation $T$ on $\mathbb R^3$ is determined by
its axis and angle of rotation $\theta \in \mathbb R/2\pi\mathbb Z$.
Consequently,
one can replace the term on the left-hand side of \eqref{eq:Evans5_2}
by $|a_g(p)|^2$,
which recovers Evans's original formulation
in \cite[p.524]{Evans-hyper}.
The positivity is consistent with the observation
that $a_n(g)\in \mathbb Z$ when $n\equiv 1 \bmod 6$ and $a_n(g)\in \mathbb Z \cdot \sqrt{-5}$ when $n\equiv 5 \bmod6$ in \texttt{LMFDB}.
\end{rem}

\end{gather*}
(The minus sign on the left hand side is due to the twist of the Legendre symbol.)

\begin{equation*}

       a_j=(-1)^{j+1} a_{k-j},\\

\end{gather*}
