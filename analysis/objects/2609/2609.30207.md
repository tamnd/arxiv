<!-- definition-1: body_md -->
Fix an integer $k\geqslant1$. Let $\chi_1,\chi_2,\ldots$ be primitive real
Dirichlet characters with strictly increasing conductors
$D_1<D_2<\cdots$. Suppose that $L(s,\chi_j)$ has a real zero
$\beta_j\in(0,1)$ for each $j$, and that
\[
1-\beta_j=o((\log D_j)^{-k})\qquad \text{as $j\to\infty$}.
\]
We call $(\beta_j)$ a sequence of \emph{Siegel zeros of strength $k$}.
When $k=1$, we simply call it a sequence of \emph{Siegel zeros}.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Assume the existence of infinitely many Siegel zeros. Let $\alpha>0$
be an irrational number of finite type and let $\beta\in\mathbb R$.
Then, for every $\varepsilon>0$, there exist infinitely many primes $p$ such that
\[
\|\alpha p+\beta\|\leqslant p^{-1/3+\varepsilon}.
\]
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:harman}
Assume the existence of infinitely many Siegel zeros of strength $7$.
Let $\alpha>0$ be an irrational number of finite type and let $\beta\in\mathbb R$.
Then, for every $0<\eta<\eta_0$, there exist infinitely many primes $p$
such that
\[
\|\alpha p+\beta\|\leqslant p^{-1/3-\eta},
\]
where
\[
\eta_0:=\frac{\sqrt{\mathrm e}-1}{3(1+2\sqrt{\mathrm e})}
=0.0503183978\ldots.
\]
<!-- end theorem-2 -->

<!-- theorem-3: body_md -->
\label{thm:asymp}
Fix $\varepsilon\in(0,1/12]$ and let $\chi$ be a primitive quadratic character of conductor $D\geqslant2$
such that its Dirichlet series $L(\cdot,\chi)$ vanishes at the real number
\[
\beta_\chi=1-\frac1{\xi\log D}\quad\text{for some}\quad\xi\geqslant10.
\]
Set $x=D^V$ with $V\geqslant20000/\varepsilon$, and $\delta=x^{-1/3+\varepsilon}$.
Let also $\alpha>0$ be irrational and
$\beta\in\mathbb R$, and

\label{lem:fourier}
Let $t$ and $\delta$ be such that $0<t\leqslant\delta/2$ and $\delta+t<1/2$. There exist smooth, one-periodic functions $\Phi^\pm$ such that
\[
0\leqslant\Phi^-\leqslant\mathbf 1_{\|\cdot\|\leqslant\delta}\leqslant\Phi^+\leqslant1,
\qquad \widehat\Phi^\pm(0)=2\delta+O(t).
\]
For $h\ne0$ and every fixed $A>0$,
\begin{equation}\label{bPhi}
|\widehat\Phi^\pm(h)|\ll_A
\min\{\delta,|h|^{-1}\}(1+|h|t)^{-A}.    
\end{equation}
Consequently, if $Ht\geqslant1$, then uniformly in $s$,
\begin{equation}\label{eq:fourierexpansion}
\Phi^\pm(s)=2\delta
+\sum_{0<|h|\leqslant H}\widehat\Phi^\pm(h)e(hs)
+O(t)+O_A((Ht)^{-A}).
\end{equation}

This lemma is in the spirit of
\cite[Lemma~2.1]{Har07}. For completeness, let us sketch a proof. Take a non-negative smooth
function $K$, supported on $[-1,1]$, with integral $1$ along the real line, and periodize the rescaled function $K_t(s):=t^{-1}K(s/t)$. On $\mathbb R/\mathbb Z$, set
\[
\Phi^+=\mathbf 1_{[-\delta-t,\delta+t]}*K_t,\qquad
\Phi^-=\mathbf 1_{[-\delta+t,\delta-t]}*K_t,
\]
where the convolutions here are on $\mathbb R/\mathbb Z$.
The inequalities $0\leqslant \Phi^-\leqslant \mathbf 1_{\|\cdot\|\leqslant\delta}\leqslant \Phi^+\leqslant1$ and the estimation of $\widehat\Phi^{\pm}(0)$ are immediate. The Fourier transforms

parts gives $\widehat K_t(h)\ll_A(1+|h|t)^{-A}$. Hence, the bounds on $\hat\Phi^{\pm}(h)$ follow by multiplying the above two bounds since $\hat\Phi^{\pm}=\hat\mathbf 1_{[-\delta\mp t,\delta\pm t]}\cdot \hat K_t$. Using the resulting bound on $\hat\Phi^{\pm}(h)$ to estimate the tail proves \eqref{eq:fourierexpansion}.

\sum_{h\sim J}\frac{|T_{j,h}|}{h}&\ll_{\varepsilon,j}\frac{Dx^{\varepsilon/4}}J\sum_{\substack{h\sim J\\r\leqslant Y^{1-1/j}}}\min\left\{\frac{2|b|DYJ}{|b|Dhr},\frac1{2\|\alpha bDhr\|}\right\}\\
&\leqslant\frac{Dx^{\varepsilon/4}}J\sum_{m\leqslant 2|b|DJY^{1-1/j}}\tau(m)\min\left\{\frac{2|b|DYJ}m,\frac1{2\|\alpha m\|}\right\}.
\end{align*}
We apply $\tau(m)\ll_{\varepsilon}m^{\varepsilon/100}$ and then
Lemma~\ref{lem:reciprocal} with
$M=2|b|DJY^{1-1/j}$ and $X=2|b|DJY$.
After division by $J$, its three terms are $\ll$
$D|b|Y^{1-1/j}$, $q$, and $D|b|Y/q$, respectively.
It follows that $\sum_{h\sim J}|T_{j,h}|/h\ll_{\varepsilon,j}D^2x^{3\varepsilon/4}(|b|Y^{1-1/j}+q+|b|Y/q)$. Summing the $O(\log H)$ dyadic ranges, we absorb the logarithm into $x^\varepsilon$.

For the second part of the lemma, partial summation handles the logarithmic factor; the preceding argument then applies.
