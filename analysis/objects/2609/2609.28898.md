<!-- thm-1: body_md -->
\label{Th: Szm}
For every integer $k\geqslant 1$ and every $\delta>0$, there exists $N=N(k,\delta)$ such that whenever $n\geqslant N$, every subset of $\{1,\ldots,n\}$ of size at least $\delta n$ contains a non-trivial $k$-term arithmetic progression, i.e. a sequence of the form
\[a,a+d,\ldots,a+(k-1)d
\qquad\text{for some integers }a,d\geqslant1.
\]
<!-- end thm-1 -->
<!-- defn-1: body_md -->
\label{Def: Replica of T_k}
A \textit{replica} of $T_k$ inside a subset $H\subset T_n$ is a copy of $T_k$ in which both the level structure and the splitting structure are preserved. More precisely, it is the image of a map $\phi: T_k \to H$ such that
\begin{enumerate}[(i)]
\item for every $x,y\in T_k$ with $\left\lvertx\right\rvert=\left\lverty\right\rvert$, we have that $\left\lvert\phi(x)\right\rvert = \left\lvert\phi(y)\right\rvert$, and
\item for every non-leaf vertex $x\in T_k$ with children $y$ and $z$, the vertices $\phi(y)$ and $\phi(z)$ are descendants of the two distinct children of $\phi(x)$, respectively.
\end{enumerate}
The sequence of levels occupied by the image $\phi(T_k)$ is called the \textit{signature} of the replica. The replica is \textit{arithmetic} if its signature is an arithmetic progression.
<!-- end defn-1 -->
<!-- thm-2: body_md -->
\label{Th:FW Density}
For every $k\geqslant1$ and $\delta>0$, there exists $N=N(k,\delta)$ such that, for every $n\geqslant N$, every $H\subset T_n$ with $d(H)\geqslant\delta$ contains an arithmetic replica of $T_k$.

<!-- defn-3: body_md -->
\label{Defn: Mixed arity replicas}
For $2\leqslant s \leqslant b$, a map $\phi: T_k^{(s)} \to T_n^{(b)}$ is called a \textit{regular embedding} if the following two conditions are satisfied.
\begin{enumerate}[(i)]
\item If $x,y\in T_k^{(s)}$ have $\left\lvertx\right\rvert = \left\lverty\right\rvert$ then $\left\lvert\phi(x)\right\rvert = \left\lvert\phi(y)\right\rvert$.
\item If $x \in \mathrm{int}\, T_k^{(s)}$ then $\phi$ maps each of the $s$ distinct children of $x$ to a descendant of a distinct child of $\phi(x)$.
\end{enumerate}
The image of $\phi$ is called a replica of $T_k^{(s)}$.
Together conditions (i) and (ii) ensure that there exists a sequence of levels
\[ 0 \leqslant l_0 < \cdots < l_k \leqslant n\]
called the signature of $\phi$ such that $\phi$ maps level $i$ of $T_k^{(s)}$ into level $l_i$ of $T_n^{(b)}$ for each $i=0,\ldots,k$. If the signature forms an arithmetic progression inside $\{0,1,\ldots,n\}$ the map and the replica are called arithmetic.

\label{Thm: Mixed arity FW}
For $2\leqslant s \leqslant b$, $k\geqslant 1$ and $\alpha > \log_b(s-1)$, there exists some $N=N(b,s,k,\alpha)$ such that, for $n\geqslant N$, the ancestor closure of every $A\subset \partial T_n^{(b)}$ with $\left\lvertA\right\rvert \geqslant b^{\alpha n}$ contains an arithmetic replica of $T_k^{(s)}$.

<!-- lem-1: body_md -->
\label{Lemma: Sig counting lemma}
For any $2\leqslant s\leqslant b$, any $n\geqslant 0$, and any $A\subset \partial T_n^{(b)}$, we have that
\begin{equation}\label{eq: branching sig inequ}
\left\lvertA\right\rvert \leqslant \sum_{\sigma \in \mathrm{BrSig}_s(A)} (b-s+1)^{\left\lvert\sigma\right\rvert} (s-1)^{n-\left\lvert\sigma\right\rvert}.
\end{equation}
<!-- end lem-1 -->
<!-- proof-1: body_md -->
We use induction on $n$. For $n=0$ the two possibilities are $A=\emptyset$ or $A=\{\emptyset\}$, and the assertions of equation~\eqref{eq: branching sig inequ} are $0 \leqslant 0$ and $1\leqslant 1$ respectively.

Now assume the lemma has been shown for $n-1$ and let $A\subset \partial T_n^{(b)}$. For each $\lambda \in \Lambda_b$ write
\[ A_\lambda: = \{ x\in \Lambda_b^{n-1} \, : \lambda x \in A\} \subset \partial T_{n-1}^{(b)}.\]
For each $\tau \subset \{0,1,\ldots,n-2\}$ let
\[ t(\tau)= \left\lvert\{ \lambda \in \Lambda_b \, : \, \tau \in \mathrm{BrSig}_s(A_\lambda)\}\right\rvert\]
be the number of branches of $T_A$ such that $T_{A_\lambda}$ realises the branching signature $\tau$. By the inductive hypothesis applied to each $A_\lambda$ we get that
\[ \left\lvertA_\lambda\right\rvert \leqslant \sum_{\tau \in \mathrm{BrSig}_s(A_\lambda)} (b-s+1)^{\left\lvert\tau\right\rvert} (s-1)^{n-1-\left\lvert\tau\right\rvert}.\]
Summing over $\lambda \in \Lambda_b$ yields
\[ \left\lvertA\right\rvert \leqslant \sum_{\tau\subset \{0,1,\ldots,n-2\}} t(\tau) (b-s+1)^{\left\lvert\tau\right\rvert} (s-1)^{n-1-\left\lvert\tau\right\rvert}.\] 
For each $\tau \subset \{0,1,\ldots,n-2\}$, let $\tau^+ \subset \{1,\ldots,n-1\}$ denote the set obtained by adding $1$ to each element of $\tau$. If $t(\tau)\geqslant 1$ then clearly $\tau^+ \in \mathrm{BrSig}_s(A)$. If $t(\tau)\geqslant s$ then again $\tau^+\in\mathrm{BrSig}_s(A)$, but we also have that $\tau^+\cup\{0\} \in \mathrm{BrSig}_s(A)$ since we can join $s$-many replicas through the root. Notice there is no issue with the compatibility of the terminal vertices of these replicas since we may always assume that they lie at level $n$.

We now compare each summand with the weight assigned to the corresponding branching signature in $\mathrm{BrSig}_s(A)$ by equation~\eqref{eq: branching sig inequ}. Put
\[W_\tau:=(b-s+1)^{\left\lvert\tau\right\rvert}(s-1)^{n-1-\left\lvert\tau\right\rvert}.\]
Each $\tau$ with $1 \leqslant t(\tau) \leqslant s-1$ contributes at most $(s-1)W_\tau$, which is exactly the weight assigned to $\tau^+ \in \mathrm{BrSig}_s(A)$. Each $\tau$ with $s\leqslant t(\tau) \leqslant b$ contributes at most
\[bW_\tau=(b-s+1)W_\tau+(s-1)W_\tau,\]
where the two terms on the right are exactly the weights assigned to $\tau^+\cup \{0\}$ and $\tau^+$, respectively. Equation~\eqref{eq: branching sig inequ} then follows.
<!-- end proof-1 -->
<!-- pro-1: body_md -->
\label{Prop: Linear length 1d replicas}
For $2\leqslant s \leqslant b$, and $\alpha > \log_b(s-1)$, there exists some $c=c(s,b,\alpha)>0$ such that, for sufficiently large $n$, the ancestor closure of every $A\subset \partial T_n^{(b)}$ with $\left\lvertA\right\rvert \geqslant b^{\alpha n}$ contains a replica of $T_{\floor{cn}}^{(s)}$.
<!-- end pro-1 -->
<!-- proof-2: body_md -->
Let $A\subset \partial T_n^{(b)}$ be as in the statement of the proposition. If $T_A$ contains no replica of $T_k^{(s)}$, then every $\sigma \in \mathrm{BrSig}_s(A)$ has $\left\lvert\sigma\right\rvert\leqslant k-1$. By Lemma~\ref{Lemma: Sig counting lemma} we must then have that
\begin{equation}\label{eq: bound for leaf set missing height k rep}
 \left\lvertA\right\rvert \leqslant \sum_{r=0}^{\min(k-1,n)}\binom{n}{r} (b-s+1)^r (s-1)^{n-r}.
\end{equation}
We will show that this contradicts that $\left\lvertA\right\rvert\geqslant b^{\alpha n}$ when $k = \floor{cn}$ for some appropriate choice of $c>0$.

For $x\in [0,1]$ define
\[ f(x) := h_2(x) + x \ln (b-s+1) + (1-x) \ln (s-1)\]
where $h_2(x) =-x \ln( x) - (1-x) \ln (1-x)$ is the binary entropy function, with $h_2(0):=0$ and $h_2(1):=0$. Since $\log_b(s-1)<\alpha$, we have $f(0) = \ln(s-1) < \alpha \ln (b)$. By continuity of $f$, there exist $c=c(s,b,\alpha)>0$ and $\eta>0$ such that
\begin{equation}\label{eq: f is small enough near 0}
f(x) \leqslant \alpha \ln(b) -\eta \qquad \text{for every } x\in [0,c].
\end{equation}
Set $k=\floor{cn}$. For $0\leqslant r \leqslant k$ the standard entropy estimate\footnote{Indeed, for $x=r/n$ the binomial theorem implies that
\[ 1= (x+(1-x))^n \geqslant \binom{n}{r}x^r(1-x)^{n-r}\] from which the desired bound follows.}
\[ \binom{n}{r} \leqslant e^{n h_2\left( r/n\right)}\]
and equation~\eqref{eq: f is small enough near 0} imply that
\begin{align*}
\binom{n}{r} (b-s+1)^r (s-1)^{n-r} &\leqslant e^{n h_2\left( r/n\right)}(b-s+1)^r (s-1)^{n-r}\\
&=e^{n f\left(r/n\right)} \leqslant e^{n(\alpha \ln(b) - \eta)} = b^{n\alpha} e^{-\eta n}.
\end{align*}
Applying this to each term in equation~\eqref{eq: bound for leaf set missing height k rep} implies that
\[ \left\lvertA\right\rvert \leqslant (n+1) b^{n \alpha} e^{-\eta n},\]
but then $\left\lvertA\right\rvert < b^{n\alpha}$ for large enough $n$ which is a contradiction.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
Proposition~\ref{Prop: Linear length 1d replicas} ensures that $T_A$ contains a replica of $T_{\floor{cn}}^{(s)}$ for $n$ large enough. The signature of this replica is then a subset of $\{0,1,\ldots,n\}$ of density at least $c/2$ for large enough $n$, and so it follows from Szemer\'{e}di's theorem that it must contain an arithmetic progression of length $k+1$ provided $n$ is large enough. Pruning our replica of $T_{\floor{cn}}^{(s)}$ in the obvious way then provides an arithmetic replica of $T_k^{(s)}$ inside $T_A$.
<!-- end proof-3 -->
<!-- rema-1: body_md -->
Let $W\subset \Lambda_b$ have $\left\lvertW\right\rvert = s-1$ and put $A = W^n \subset \partial T_n^{(b)}$. Then $\left\lvertA\right\rvert =(s-1)^n = b^{n\log_b(s-1)}$, but every non-leaf vertex in $T_A$ has exactly $s-1$ children, and so $T_A$ does not even contain a replica of $T_1^{(s)}$. This shows that the condition $\alpha > \log_b(s-1)$ in Proposition~\ref{Prop: Linear length 1d replicas} and Theorem~\ref{Thm: Mixed arity FW} is sharp.

<!-- defn-4: body_md -->
Let $A\subset \partial \Gamma_{n,m}$. We say that a vertex $(u,v) \in \Gamma_A$ \textit{splits} if

\end{equation}
and denote by $\mathrm{Split}_2^{(2)}(A)$ the set of all vertices in $\Gamma_A$ which split. The four vertices in equation~\eqref{eq: diag children} are called the \textit{diagonal children} of $(u,v)$. Notice also that $\mathrm{Split}_2^{(2)}(A) \subset \mathrm{int}\, \Gamma_{n,m}$.

<!-- pro-2: body_md -->
\label{Prop: 2d binary splitting count}
For all $n,m\geqslant 0$ and $A \subset \partial \Gamma_{n,m}$ we have that
\[ \left\lvert\mathrm{Split}_2^{(2)}(A)\right\rvert \geqslant \left\lvertA\right\rvert -(2^n + 2^m -1).\]
<!-- end pro-2 -->
<!-- lem-2: body_md -->
\label{Lemma: Baby split lem}
For all $n\geqslant 0$ and $C\subset \partial T_n$ we have that
\[ \left\lvert\mathrm{Split}_1^{(2)}(C)\right\rvert = \left\lvertC\right\rvert - \mathbf{1}_{C\neq \emptyset}.\]
In particular, if $C\neq \emptyset$, then $T_C$ has exactly $\left\lvertC\right\rvert-1$ vertices with $2$ children.
<!-- end lem-2 -->
<!-- proof-5: body_md -->
For $C=\emptyset$ the statement is obvious. For non-empty $C$, let $t_0$, $t_1$, and $t_2$ be the number of vertices in $T_C$ with $0$, $1$, and $2$ children respectively. By counting edges as the number of non-root vertices and as the sum of the child counts we have that
\[ t_0 + t_1 +t_2 -1 = t_1 + 2 t_2.\] 
Since $t_0 = \left\lvertC\right\rvert$ and $t_2 = \left\lvert\mathrm{Split}_1^{(2)}(C)\right\rvert$, the result follows.
<!-- end proof-5 -->
<!-- proof-6: body_md -->
Let $A\subset \partial \Gamma_{n,m}$. For each $x \in \{0,1\}^n$ let
\[ A_x:= \{y \in \{0,1\}^m \, : \, (x,y) \in A\}\]
be the vertical fibre above $x$. For each $v \in \mathrm{int}\, T_m$ consider
\[ E_v:= \{ x \in \{0,1\}^n \, : \, v \in \mathrm{Split}_1^{(2)}(A_x)\} \subset \partial T_n\]
as a one-dimensional leaf set. By counting in two orders and using Lemma~\ref{Lemma: Baby split lem} we have that
\begin{align} 
\sum_{v\in \mathrm{int}\, T_m} \left\lvertE_v\right\rvert &= \sum_{x\in \{0,1\}^n} \left\lvert\mathrm{Split}_1^{(2)}(A_x)\right\rvert \nonumber \\
&= \sum_{x\in \{0,1\}^n} \left( \left\lvertA_x\right\rvert - \mathbf{1}_{A_x \neq \emptyset}\right)\label{eq: first use of baby split}
\geqslant \left\lvertA\right\rvert - 2^n.
\end{align}
Let $u \in \mathrm{Split}_1^{(2)}(E_v)$ for some $v\in \mathrm{int}\, T_m$. We claim that $(u,v) \in \mathrm{Split}_2^{(2)}(A)$.
Indeed, by definition of $\mathrm{Split}_1^{(2)}(E_v)$, for each $\varepsilon \in \{0,1\}$ there exists some $x_\varepsilon \in E_v$ such that $u\varepsilon \preceq x_\varepsilon$. By definition of $E_v$, for each $\varepsilon \in \{0,1\}$, $v \in \mathrm{Split}_1^{(2)}(A_{x_\varepsilon})$, and so for each $\delta \in \{0,1\}$ there exists some $y_{\delta}^{\varepsilon} \in A_{x_\varepsilon}$ such that $v\delta \preceq y_{\delta}^{\varepsilon}$. It then follows that $(u\varepsilon,v\delta) \preceq_2 (x_\varepsilon,y_{\delta}^\varepsilon) \in A$ for each $(\varepsilon,\delta )\in \{0,1\}^2$, which proves the claim.

It follows that
\begin{align*}
\left\lvert\mathrm{Split}_2^{(2)}(A)\right\rvert &\geqslant \sum_{v\in  \mathrm{int}\, T_m} \left\lvert\mathrm{Split}_1^{(2)}(E_v)\right\rvert\\
&\geqslant\sum_{v\in \mathrm{int}\, T_m} \left(\left\lvertE_v\right\rvert-1\right)\\
&\geqslant \left\lvertA\right\rvert-2^n - (1+2+\cdots+2^{m-1}) = \left\lvertA\right\rvert - (2^n + 2^m-1).
\end{align*}
where in the second line we use Lemma~\ref{Lemma: Baby split lem} again, noting that the bound $\left\lvert\mathrm{Split}_1^{(2)}(E_v) \right\rvert\geqslant \left\lvertE_v\right\rvert-1$ holds irrespective of whether $E_v$ is empty or not, and in the last line we use equation~\eqref{eq: first use of baby split}.
<!-- end proof-6 -->
<!-- lem-3: body_md -->
\label{Lemma: Block branching gives local splitting}
Let $\omega \in T_{\tilde{A}}$ and write $\Theta_q(\omega)=(U,V) \in \Gamma_A$. For any set 
\[ C\subset \{\lambda \in \Omega_q \, : \, \omega \lambda \in T_{\tilde{A}}\}\] 
with $\left\lvertC\right\rvert\geqslant 2^{q+1}$ there exists some $(u,v) \in \{0,1\}^{\leqslant q-1}\times \{0,1\}^{\leqslant q-1}$ such that $(Uu,Vv) \in \mathrm{Split}_2^{(2)}(A)$. Moreover, $(u,v)$ has the additional property that for each $(\varepsilon,\delta) \in \{0,1\}^2$ there exists some $\lambda_{\varepsilon,\delta} \in C$ with
\[ (Uu\varepsilon,V v \delta) \preceq_2 \Theta_q(\omega \lambda_{\varepsilon,\delta}).\]
<!-- end lem-3 -->

If $n=0$ the result is immediate so suppose that $n\geqslant 1$. Let $a_i$ be the number of vertices in $T_A$ at level $i$, and let $f_i$ be the number of splitting vertices at level $i$. \[ a_{i+1} \leqslant b f_i + (b-1)(a_i-f_i) = (b-1)a_i + f_i,\]

\[ f_i \geqslant a_{i+1} - (b-1) a_i.\] 
Multiplying by $(b-1)^{n-i-1}$ and summing over levels gives
\begin{align*}
\sum_{u\in \mathrm{Split}_1^{(b)}(A)} (b-1)^{n-\left\lvertu\right\rvert-1} & = \sum_{i=0}^{n-1}f_i (b-1)^{n-i-1}\\
& \geqslant \sum_{i=0}^{n-1} \left[ (b-1)^{n-i-1}a_{i+1} - (b-1)^{n-i}a_i \right]\\
&= a_n - (b-1)^n a_0 \geqslant \left\lvertA\right\rvert - (b-1)^n
\end{align*}
since the final sum telescopes.

\end{equation}

C_q = b^{dq} z_q \sum_{i=0}^{d-1}(1-z_q)^i = b^{q \alpha_{\text{crit}}(d,b)} \sum_{i=0}^{d-1}(1-z_q)^i.
\]
Since $z_q \in (0,1)$ the sum lies between $1$ and $d$ so
\[ b^{q \alpha_{\text{crit}}(d,b)} \leqslant C_q \leqslant d b^{q \alpha_{\text{crit}}(d,b)}.\] 
Taking logarithms base $b$ and dividing by $q$ yields

of all $d$-tuples of words of length $n$ in the alphabet $\Lambda_b$ such that the word in the first component does not contain the symbol $b-1$. The set $A$ has cardinality
\[\left\lvertA\right\rvert = (b-1)^{n} b^{(d-1)n} = b^{\alpha_{\text{crit}}(d,b)n}\]
yet
\[ \mathrm{Split}_d^{(b)}(A) = \emptyset.\]
Indeed, for any $\mathbf{u}=(u_1,\ldots,u_d) \in \Gamma_{A}$ with $\left\lvertu_i\right\rvert< n$ for each $i=1,\ldots,d$, every diagonal child of $\mathbf{u}$ whose first coordinate is $u_1 (b-1)$ is not in $\Gamma_A$, since no leaf in $A$ contains the digit $(b-1)$ in its first coordinate.

\label{Thm: Refined product density result}
For all $b\geqslant 2$, $d,k\geqslant 1$, and $\delta>0$, there exists $N=N(b,d,k,\delta)$ such that for every $\mathbf{n} = (n_1,\ldots,n_d) \in \mathbb{Z}_{\geqslant 0}^{d}$ with $\min_{i} n_i \geqslant N$, every $H \subset \Gamma_{\mathbf{n}}^{(b)}$ with
\[ w_b^d(H) \geqslant \delta \prod_{i=1}^{d} (n_i +1)\]
contains an arithmetic replica of $T_k^{\left(b^d\right)}$. Moreover, the replica can be chosen so that its signature has the form
\[ \mathbf{v} + tQ \mathbf{1} \qquad \text{for each }t=0,1,\ldots,k\]
for some $Q\geqslant 1$.

Partition the product levels of $\Gamma_{\mathbf{n}}^{(b)}$ into maximal diagonal chains
\[ C_\mathbf{a} = \{ \mathbf{a} + t \mathbf{1} \, : \, 0\leqslant t <l_{\mathbf{a}}\}\]
where
\[ 0\leqslant \mathbf{a} \leqslant \mathbf{n}, \qquad \min_{i} a_i = 0, \qquad \text{ and } l_{\mathbf{a}} = 1+ \min_{i}(n_i-a_i).\]
Each product level belongs to exactly one such chain. By a union bound over the coordinate faces, the number of such chains $R$ satisfies
\[ R \leqslant \sum_{i=1}^d \prod_{j \neq i} (n_j+1) \leqslant \frac{d \prod_{i=1}^d (n_i+1)}{m}\]
where $ m: = 1+ \min_{i}n_i$. For each chain let
\[ H_{\mathbf{a}}= \{\mathbf{x} \in H \,: \, \left\lvert\mathbf{x}\right\rvert \in C_{\mathbf{a}}\}.\]
Since these sets partition $H$, we have that $w_b^d(H) = \sum_{\mathbf{a}} w_b^d(H_{\mathbf{a}})$, and so there must be one chain $C_{\mathbf{a}}$ with length $l:= l_\mathbf{a}$ such that
\[ w_b^d(H_{\mathbf{a}}) \geqslant \frac{w_b^d(H)}{R} \geqslant \frac{\delta m\prod_{i=1}^d (n_i+1)}{d \prod_{i=1}^d(n_i+1)} = \frac{m\delta}{d}.\]
The total weight of each product level in $\Gamma_{\mathbf{n}}^{(b)}$ is $1$, and each $C_{\mathbf{a}}$ has length at most $m$, so $w_b^d(H_{\mathbf{a}}) \leqslant l \leqslant m$. Hence
\[ l\geqslant \frac{\delta m}{d} \qquad \text{and} \qquad w_b^d (H_{\mathbf{a}}) \geqslant \frac{\delta l}{d}.\]
In words, the chain is long and carries a positive proportion of its maximum possible weight.

Now write
\[ \mathcal{L}_{\mathbf{a}} = \Lambda_b^{a_1} \times \cdots \times \Lambda_b^{a_d}\]
for level $\mathbf{a}$ of $\Gamma_{\mathbf{n}}^{(b)}$. Let $\Omega = \Lambda_b^d$ be a block alphabet.\footnote{$\Omega$ is exactly $\Omega_q$ with $q=1$ from Section~\ref{Sec: general case}.} For each $\mathbf{u} = (u_1,\ldots,u_d) \in \mathcal{L}_{\mathbf{a}}$, define a map
\[\phi_{\mathbf{u}}: \Omega^{\leqslant l-1} \to \Gamma_{\mathbf{n}}^{(b)}\]
by concatenating blocks separately in each coordinate and then prefixing by $\mathbf{u}$. Explicitly, if
\[ \omega = \bm{\alpha}^{(1)}\ldots \bm{\alpha}^{(t)}, \qquad \text{where each} \qquad \bm{\alpha}^{(i)} = \left(\alpha_1^{(i)},\ldots,\alpha_d^{(i)}\right) \in \Omega\] 
then
\[ \phi_{\mathbf{u}}(\omega) = \left(u_1 \alpha_1^{(1)}\ldots \alpha_1^{(t)},\ldots,u_d \alpha_d^{(1)}\ldots \alpha_d^{(t)}\right).\]
The map $\phi_{\mathbf{u}}$ sends level $t$ of $\Omega^{\leqslant l-1}$ to level $\mathbf{a} + t \mathbf{1}$, and the $b^d$ children of any non-leaf $x\in \Omega^{\leqslant l-1}$ are mapped to the diagonal children of $\phi_{\mathbf{u}}(x)$.

Let
\[ K_{\mathbf{u}}:= \phi_{\mathbf{u}}^{-1}(H). \]
As $\mathbf{u}$ varies over $\mathcal{L}_{\mathbf{a}}$, the images of $\phi_{\mathbf{u}}$ partition all product vertices on $C_{\mathbf{a}}$ and so
\begin{equation}\label{eq: computing weights through phi_u}
w_b^d(H_{\mathbf{a}}) = \sum_{\mathbf{u}\in \mathcal{L}_{\mathbf{a}}} w_b^d(\phi_{\mathbf{u}}(K_{\mathbf{u}})) = b^{-(a_1 + \cdots + a_d)} \sum_{\mathbf{u} \in \mathcal{L}_{\mathbf{a}}} w_{b^d}(K_{\mathbf{u}})
\end{equation}
where the second equality follows since any $x \in \Omega^{\leqslant l-1}$ with $\left\lvertx\right\rvert = t$ satisfies
\[ w_b^d(\{\phi_{\mathbf{u}}(x)\}) = b^{-\left(a_1+\cdots+a_d + dt\right)} = b^{-(a_1+\cdots+a_d)} w_{b^d}(\{x\}).\]
Since $\left\lvert\mathcal{L}_{\mathbf{a}}\right\rvert=b^{a_1+\cdots+a_d}$, the right-hand side of equation~\eqref{eq: computing weights through phi_u} is exactly the average of the weights $w_{b^d}(K_{\mathbf{u}})$. Hence some $\mathbf{u}\in \mathcal{L}_{\mathbf{a}}$ must satisfy
\[ w_{b^d}(K_{\mathbf{u}}) \geqslant w_b^d(H_{\mathbf{a}}) \geqslant \frac{\delta l}{d}.\]
Since $l \geqslant (\delta/d)(1+\min_i n_i)$, Theorem~\ref{Thm: Equal arity density} ensures that $K_{\mathbf{u}}$ contains an arithmetic replica of $T_k^{\left(b^d\right)}$ provided that $\min_i n_i$ is large enough. The image of this replica under $\phi_{\mathbf{u}}$ is then an arithmetic replica of $T_k^{\left(b^d\right)}$ inside $H$ whose signature has common difference proportional to $\mathbf{1}$, as required.
