\label{prop:relative-cone-lift}
Let $Q$ be a unital C*-algebra, let $J\lhd Q$ be a $\sigma$-ideal, and let
$p\colon Q\to Q/J$ be the quotient map. Let $A\subseteq Q$ and $C\subseteq Q/J$ be
separable and unital C*-subalgebras such that $A\cap J=\{0\}$  and 
$p(A)\subseteq C$.
Identifying $A$ with $p(A)\subseteq C$, there exists a unital
$*$-homomorphism $\Theta\colon\mathcal D(A,C)\to Q$
fixing $A$ and such that 
$p\circ\Theta=\mathrm{ev}_1$.
Equivalently, the following diagram commutes:
$$\label{liftdiagram}
\begin{tikzcd}
A \arrow[r,hook] \arrow[d,equal]
& \mathcal D(A,C) \arrow[r,"\mathrm{ev}_1"] \arrow[d,"\Theta"]
& C \arrow[d,hook] \\
A \arrow[r,hook]
& Q \arrow[r,"p"]
& Q/J .
\end{tikzcd}

Choose a unital separable C*-subalgebra $Q_0\subseteq Q$ containing $A$ and such that $p(Q_0)=C$. Set $I=Q_0\cap J$.
Since $J$ is a $\sigma$-ideal, there exists a positive contraction
$d\in Q_0'\cap J$ such that $db=b$ for all $b\in I$.
Set $h=1-d$. Then $h$ is a positive contraction such that $h\in Q_0'$, $h\perp I$, and $p(h)=1$.

Since $C^*(1,h)$ and $Q_0$ commute, there is a unital $*$-homomorphism
$$ 
\alpha\colon C([0,1])\otimes Q_0\to Q,\qquad
\alpha(f\otimes c)=f(h)c.
$$
Here we use the nuclearity of $C([0,1])$.

Pointwise application of $p|_{Q_0}\colon Q_0\to C$ induces a $*$-homomorphism
$$
\widehat p\colon\mathcal D(A,Q_0)\to\mathcal D(A,C).
$$
Writing $\mathcal D(A,C)=A+C_0((0,1])\otimes C$, we readily see
that $\widehat p$ is surjective. Moreover, if
$\widehat p(f)=0$, then $f$ is $I$-valued and $f(0)=0$, since
$A\cap I=\{0\}$. Thus
$$
\ker\widehat p=C_0((0,1])\otimes I.
$$

On the other hand, if $f\in C_0((0,1])$ and $b\in I$, then
$hb=0$ and hence $f(h)b=0$. It follows that
$\alpha$ vanishes on $C_0((0,1])\otimes I$. Hence, the
restriction of $\alpha$ to $\mathcal D(A,Q_0)$ factors through
$\widehat p$, giving a unital $*$-homomorphism
$\Theta\colon \mathcal D(A,C)\to Q$
 such that 
 $$
 \alpha|_{\mathcal D(A,Q_0)}=\Theta\circ\widehat p.
 $$

For $f\in\mathcal D(A,Q_0)$, since $p(h)=1$, we have $p(\alpha(f))=p(f(1))$.
It follows that $p\circ\Theta=\mathrm{ev}_1$.
Finally, for $a\in A$, viewed as a constant function, $\Theta(a)=a$, so $\Theta$ fixes $A$.

The proof of Theorem \ref{mainthm} can be reduced to the case that $A$ is separable by considering separable elementary submodels of a given $A$. We thus fix a separable, infinite-dimensional, simple, unital C*-algebra $A$ with unique trace $\tau$ and strict comparison with respect to $\tau$. We choose a free ultrafilter  $\mathcal U$ over $\mathbb N$. 

Let $M=\pi_{\tau}(A)''$, which is a $\mathrm{II}_1$ factor with separable predual. Let $M^{\mathcal U}$ denote the tracial ultrapower of $M$.

Recall that we write $\mathcal S_{\tau}=C^*(A,s_{\tau})$,
and that we denote by $\tau_{\Omega}$ the tracial state on $\mathcal S_{\tau}$ induced by the vacuum state. 
By the reduced-free-product description above and \cite[Theorem~2]{dykema}, $\mathcal S_{\tau}$ is a simple C*-algebra
with unique trace $\tau_{\Omega}$.

The following lemma is an immediate consequence of Popa's theorem \cite{Popa}.

\begin{lemma}
There is a unital trace-preserving embedding $\iota\colon \mathcal S_{\tau}\to M^{\mathcal U}$ fixing $A$.	
\end{lemma}	 

\begin{proof}
Choose a semicircular element $s\in M$ and, using Popa's theorem \cite{Popa}, a Haar unitary
$u\in M^{\mathcal U}$ that is freely independent from $M$. Set $s_1=u^{-1}su$. Then $A\subseteq M$ and $C^*(1,s_1)$ are freely independent in $M^{\mathcal U}$. Since the trace $\tau_{M^{\mathcal U}}$ is faithful on $M^{\mathcal U}$, 
$$
C^*(A,s_1)\cong A*C^*(1,s) 
\cong \mathcal S_{\tau},
$$
where the free products are reduced.
\end{proof}		

Let $\mathcal D=\mathcal D(A,\mathcal S_{\tau})$, and let $t\in \mathcal D$ denote the identity on $[0,1]$. 
We regard $\mathcal D$ endowed with the trace $\tau_\Omega\circ \mathrm{ev}_1$.



By \cite[Theorem 3.3]{KR}, we have an onto map $p\colon A^{\mathcal U}\to M^{\mathcal U}$ whose kernel is the trace-kernel ideal $J_\tau\lhd A^{\mathcal U}$ formed by all $x\in A^{\mathcal U}$ such that $\tau^{\mathcal U}(x^*x)=0$.
By \cite[Proposition~4.6 and Remark~4.7]{KR}, $J_\tau$ is a $\sigma$-ideal.

\begin{lemma}
There is a unital $*$-homomorphism $\Theta\colon \mathcal D\to A^{\mathcal U}$ fixing $A$ and such that $p\circ\Theta=\iota \circ \mathrm{ev}_1$ and $\tau^{\mathcal U}\circ\Theta=\tau_\Omega\circ \mathrm{ev}_1$. 
In other words, the following diagram is commutative and all the $*$-homomorphisms are trace preserving: 
\begin{equation}\label{diagram}
\begin{tikzcd}
A \arrow[r,hook] \arrow[d,equal]
& \mathcal D \arrow[r,"\mathrm{ev}_1"] \arrow[d,"\Theta"]
& \mathcal S_{\tau} \arrow[d,hook,"\iota"] \\
A \arrow[r,hook]
& A^{\mathcal U} \arrow[r,"p"]
& M^{\mathcal U}.
\end{tikzcd}
\end{equation}
\end{lemma}	



\begin{proof}
Identify $\mathcal S_{\tau}$ with $\iota(\mathcal S_{\tau})\subseteq M^{\mathcal U}=A^{\mathcal U}/J_\tau$ and apply Proposition \ref{prop:relative-cone-lift}.
The equality $\tau^{\mathcal U}\circ\Theta=\tau_\Omega\circ \mathrm{ev}_1$ follows by applying $\tau_{M^{\mathcal U}}$ in $p\circ\Theta=\iota \circ \mathrm{ev}_1$ and using that $\iota$ is trace preserving.
\end{proof}

\begin{lemma}
For every finite set $\Sigma\subset B$ and $\epsilon>0$ there exist positive contractions $c,c'\in B$  such that $cc'=c'$, $\tau^{\mathcal U}(c')>0$, and 
$$
\|cbc - \tau^{\mathcal U}(b)c^2\|<\epsilon,\qquad b\in \Sigma.
$$	
\end{lemma}	

\begin{proof}
Let $\Sigma=\{b_1,\ldots,b_m\}$. Choose lifts $f_k\in \mathcal D$ of each $b_k$. Since $\mathcal S_{\tau}$ is a simple C*-algebra, its trace $\tau_\Omega$ can be excised by positive contractions \cite[Proposition~2.3]{AAP}. Thus, there exists a positive $d\in \mathcal S_{\tau}$ of norm 1 such that 
$$
\|df_k(1)d-\tau_\Omega(f_k(1))d^2\|<\epsilon/8,\qquad k=1,\ldots,m.
$$
We modify $d$ via functional calculus: choose $\chi\in C_0((0,1])$ that is zero on $[0,1/2]$ and $1$ on $[3/4,1]$, and linear on $[1/2,3/4]$, and then set
$$
d_1=\chi(d),\qquad d_1'=(d-3/4)_+.
$$
Then $d_1d_1'=d_1'$ and $d_1'\neq 0$. Writing $\chi(t)=t\xi(t)$ with $\|\xi\|\leq 2$, the preceding estimate gives
$$
\|d_1f_k(1)d_1-\tau_\Omega(f_k(1))d_1^2\|<\epsilon/2.
$$
Now pick $0\leq t_0<1$ such that 
$\|f_k(1)-f_k(t)\|<\epsilon/2$
for all $t\in [t_0,1]$.
Pick $g,g'\in C_0((0,1])$ supported in $[t_0,1]$, such that $0\leq g,g'\leq 1$,  $gg'=g'$ and $g(1)=g'(1)=1$. Set 
$$
c=\Theta(g(t)d_1),\qquad c'=\Theta(g'(t)d_1').
$$
Then $cc'=c'$, $c$ satisfies the inequality of the lemma,  and $\tau^{\mathcal U}(c')=\tau_\Omega(d_1')>0$,
since $d_1'\neq 0$ and $\tau_\Omega$ is faithful
on $\mathcal S_{\tau}$.
\end{proof}	

\begin{lemma}
There exists $r\in A^{\mathcal U}$ such that
$$
r^*br=\tau^{\mathcal U}(b)e,\qquad b\in B.
$$
\end{lemma}

\begin{proof}
Let $\Sigma\subset B$ be a finite set with $1\in\Sigma$ and let $\epsilon>0$.
Let $\eta>0$. Let $c,c'\in B$ be supplied by the previous lemma such that 
$$
\|cbc - \tau^{\mathcal U}(b)c^2\|<\eta,\qquad b\in \Sigma.
$$
Since $A$ has strict comparison by $\tau$, its ultrapower $A^{\mathcal U}$ has strict comparison by $\tau^{\mathcal U}$ (\cite[Lemma 1.23]{bosaetal}, \cite[Theorem 8.2.2]{farahetal}). From
$$
d_{\tau^{\mathcal U}}(e)=0<\tau^{\mathcal U}(c')\leq d_{\tau^{\mathcal U}}(c'),
$$
we deduce that $e\precsim_{\mathrm{Cu}}c'$. Thus, for every $\delta>0$ there exists
$v\in A^{\mathcal U}$ such that
$$
\|e-v^*v\|<\delta,\qquad v\in\overline{c'A^{\mathcal U}}.
$$
Since $cc'=c'$, we have $cv=v$. Moreover,
$\|v\|^2<1+\delta$.

For $b\in\Sigma$,
$$
v^*bv-\tau^{\mathcal U}(b)e
=
v^*(cbc-\tau^{\mathcal U}(b)c^2)v
+(v^*v-e)\tau^{\mathcal U}(b).
$$
Thus,
$$
\|v^*bv-\tau^{\mathcal U}(b)e\|
\leq
\eta\|v\|^2
+\delta|\tau^{\mathcal U}(b)|.
$$
Choosing $\eta$ and $\delta$ sufficiently small, the right-hand side can be made arbitrarily small, uniformly for $b\in\Sigma$.

The conclusion now follows from Kirchberg's $\epsilon$-test \cite{kirchberg-abel}.
\end{proof}

\|df_k(1)d-\tau_\Omega(f_k(1))d^2\|<\epsilon/8,\qquad k=1,\ldots,m.
$$
We modify $d$ via functional calculus: choose $\chi\in C_0((0,1])$ that is zero on $[0,1/2]$ and $1$ on $[3/4,1]$, and linear on $[1/2,3/4]$, and then set
$$
d_1=\chi(d),\qquad d_1'=(d-3/4)_+.
$$
Then $d_1d_1'=d_1'$ and $d_1'\neq 0$. Writing $\chi(t)=t\xi(t)$ with $\|\xi\|\leq 2$, the preceding estimate gives
$$
\|d_1f_k(1)d_1-\tau_\Omega(f_k(1))d_1^2\|<\epsilon/2.
$$
Now pick $0\leq t_0<1$ such that 
$\|f_k(1)-f_k(t)\|<\epsilon/2$
for all $t\in [t_0,1]$.
Pick $g,g'\in C_0((0,1])$ supported in $[t_0,1]$, such that $0\leq g,g'\leq 1$,  $gg'=g'$ and $g(1)=g'(1)=1$. Set 
$$
c=\Theta(g(t)d_1),\qquad c'=\Theta(g'(t)d_1').
$$
Then $cc'=c'$, $c$ satisfies the inequality of the lemma,  and $\tau^{\mathcal U}(c')=\tau_\Omega(d_1')>0$,
since $d_1'\neq 0$ and $\tau_\Omega$ is faithful
on $\mathcal S_{\tau}$.

Let $\Sigma\subset B$ be a finite set with $1\in\Sigma$ and let $\epsilon>0$.
Let $\eta>0$. Let $c,c'\in B$ be supplied by the previous lemma such that 
$$
\|cbc - \tau^{\mathcal U}(b)c^2\|<\eta,\qquad b\in \Sigma.
$$
Since $A$ has strict comparison by $\tau$, its ultrapower $A^{\mathcal U}$ has strict comparison by $\tau^{\mathcal U}$ (\cite[Lemma 1.23]{bosaetal}, \cite[Theorem 8.2.2]{farahetal}). From
$$
d_{\tau^{\mathcal U}}(e)=0<\tau^{\mathcal U}(c')\leq d_{\tau^{\mathcal U}}(c'),
$$
we deduce that $e\precsim_{\mathrm{Cu}}c'$. Thus, for every $\delta>0$ there exists
$v\in A^{\mathcal U}$ such that

\|e-v^*v\|<\delta,\qquad v\in\overline{c'A^{\mathcal U}}.
$$
Since $cc'=c'$, we have $cv=v$. Moreover,
$\|v\|^2<1+\delta$.

For $b\in\Sigma$,

=

Recall that in Part I we have obtained a $*$-homomorphism $\Theta\colon \mathcal D\to A^{\mathcal U}$ fixing $A$ and such that 
$$
\tau^{\mathcal U}\circ \Theta=\tau_\Omega\circ\mathrm{ev}_1.
$$
Recall that in Part II we have obtained $r\in A^{\mathcal U}$ such that
\begin{equation}\label{rstarThetar}
r^*\Theta(f)r=\tau^{\mathcal U}(\Theta(f))e,\qquad f\in \mathcal D,
\end{equation}
where $e=\Theta(1-t^2)$.

Let us define a cp map  $\phi\colon \mathcal D\to \mathcal D$ by
$$
\phi(f)=\tau_\Omega(f(1))(1-t^2).
$$
Let $\mathcal X$ denote the C*-correspondence over $\mathcal D$ obtained from the KSGNS construction applied  to $\phi$ (see \cite{brown-ozawa}). Let $\xi\in \mathcal X$ be its canonical generator (i.e., the class of $1\otimes 1$).

We note that
$$
\mathcal X\otimes_{\mathcal D}\mathcal X=0.
$$
Indeed, this readily follows from the fact that inner products in $\mathcal X$ take values in the two-sided ideal $(1-t^2)\mathcal D$, which is contained in the kernel of $\phi$. Consequently, the Fock space of $\mathcal X$ is $\mathcal F(\mathcal X)=\mathcal D\oplus \mathcal X$. Let $\mathcal T(\mathcal X)$ be the Toeplitz-Pimsner
C*-algebra associated to $\mathcal X$ and $\ell_{\xi}\in \mathcal T(\mathcal X)$ the left creation operator associated to $\xi$.

\begin{lemma}
There exists $\Psi\colon \mathcal T(\mathcal X)\to A^{\mathcal U}$ extending $\Theta$ and such that $\Psi(\ell_{\xi})=r$.
\end{lemma}

\begin{proof}
This follows from the universal property of $\mathcal T(\mathcal X)$. Indeed, we can rewrite \eqref{rstarThetar} as
$$
r^*\Theta(f)r=\Theta(\phi(f)).
$$
Thus, mapping $\xi\mapsto r$ extends to a representation of $\mathcal X$ in $A^{\mathcal U}$ with underlying $*$-homomorphism  $\Theta$. By the universal property of $\mathcal T(\mathcal X)$, we get $\Psi$.
\end{proof}

We will show next that there exists a unital $*$-homomorphism 
$\mathcal S_{\tau}\to\mathcal T(\mathcal X)$ fixing $A$. Composing it with $\Psi$
yields the desired $*$-homomorphism of $\mathcal S_{\tau}$ into $A^{\mathcal U}$ witnessing selflessness of $A$.

Recall that $\mathcal T_{\tau}$ denotes the Toeplitz-Pimsner algebra associated to $\tau$. Define 
$$
\widetilde \mathcal D=\{f\in C([0,1],\mathcal T_{\tau}):f(0)\in A\}.
$$
Since $\mathcal S_{\tau}\subseteq \mathcal T_{\tau}$,  we have $\mathcal D\subseteq \widetilde \mathcal D$. Let $\widetilde \phi\colon \widetilde \mathcal D\to \widetilde \mathcal D$ be the cp map defined by
$$
\widetilde\phi(f)=\tau_{\Omega}(f(1))(1-t^2),\qquad f\in \widetilde\mathcal D.
$$
Here $\tau_{\Omega}$ denotes the vacuum state on $\mathcal T_{\tau}$, which extends the trace $\tau_{\Omega}$ on $\mathcal S_{\tau}$.
It is clear that $\widetilde \phi$ extends $\phi$. Let $\widetilde \mathcal X$ denote the C*-correspondence over $\widetilde \mathcal D$ 
obtained from the KSGNS construction applied to the cp map $\widetilde\phi$. Let $\widetilde\xi\in\widetilde\mathcal X$ be its canonical generator. As before, we have
 $$
\widetilde\mathcal X\otimes_{\widetilde \mathcal D}\widetilde\mathcal X = 0.
 $$
Thus, $\mathcal F(\widetilde\mathcal X)=\widetilde\mathcal D\oplus \widetilde \mathcal X$. 


\begin{lemma}
The assignment $f\xi g\mapsto f\widetilde \xi g$, for $f,g\in \mathcal D$, extends to a covariant embedding $\mathcal X\to \widetilde\mathcal X$ with underlying
$*$-homomorphism the inclusion $\mathcal D\to\widetilde\mathcal D$.
The assignment $f\mapsto f$, $\ell_{\xi}\mapsto \ell_{\widetilde\xi}$ extends to an embedding of $\mathcal T(\mathcal X)$
into $\mathcal T(\widetilde\mathcal X)$.
\end{lemma}

\begin{proof}
Since $\mathcal X$ is generated as a closed $\mathcal D$-bimodule by $\xi$, by \cite[Lemma~2.1]{gaojungeetal}
it suffices to check that the assignment
$$
\xi\longmapsto \widetilde\xi
$$
preserves the corresponding coefficient map. For $f\in\mathcal D$,
$$
\langle \widetilde\xi,f\widetilde\xi\rangle
=\widetilde\phi(f)
=\phi(f)
=\langle \xi,f\xi\rangle.
$$
Thus $\xi\mapsto\widetilde\xi$ extends to a covariant embedding
$\mathcal X\to\widetilde\mathcal X$ with underlying $*$-homomorphism
$\mathcal D\hookrightarrow\widetilde\mathcal D$.

The universal property of $\mathcal T(\mathcal X)$ now gives a $*$-homomorphism
$$
\mathcal T(\mathcal X)\longrightarrow\mathcal T(\widetilde\mathcal X),
\qquad
f\mapsto f,\quad
\ell_\xi\mapsto\ell_{\widetilde\xi}.
$$
To see that it is injective, note that $\mathcal D\oplus\mathcal X$, viewed as a submodule of
$\widetilde\mathcal D\oplus\widetilde\mathcal X$, is reducing for the left action of $\mathcal D$
and for $\ell_{\widetilde\xi}$. Indeed,
$\ell_{\widetilde\xi}$ maps $\mathcal D$ into $\mathcal X$, while it vanishes on
$\mathcal X$ because
$\widetilde\mathcal X\otimes_{\widetilde\mathcal D}\widetilde\mathcal X=0$; similarly,
$\ell_{\widetilde\xi}^*$ maps $\mathcal X$ into $\mathcal D$ and vanishes on
$\mathcal D$. Moreover, under the identification
$\mathcal D\oplus\mathcal X\subseteq\widetilde\mathcal D\oplus\widetilde\mathcal X$, the
restrictions of the left action of $\mathcal D$ and of
$\ell_{\widetilde\xi}$ are precisely the left action of $\mathcal D$ and
$\ell_\xi$ on $\mathcal F(\mathcal X)$, i.e., the Fock
representation of $\mathcal T(\mathcal X)$. Hence the $*$-homomorphism is injective.
\end{proof}



Identify $\mathcal T(\mathcal X)$ as a C*-subalgebra of $\mathcal T(\widetilde\mathcal X)$ by the previous embedding and regard $\ell_{\widetilde \xi}$ as an extension
of $\ell_{\xi}$.


\begin{lemma}
Let $W\in \mathcal T(\widetilde\mathcal X)$ be defined as
$$
W=t\ell_{\tau} + \ell_{\widetilde \xi}.
$$
\begin{enumerate}[(i)]
\item
We have 
$$
W^*aW=\tau(a)1,\qquad a\in A,
$$
and $W+W^*\in \mathcal T(\mathcal X)$.
\item
The assignment $A\ni a\mapsto a$ and $\ell_{\tau}\mapsto W$ extends to a $*$-homomorphism from $\mathcal T_{\tau}$
into $\mathcal T(\widetilde\mathcal X)$.
\item
The assignment $A\ni a\mapsto a$ and $s_{\tau}\mapsto W+W^*$ extends to a $*$-homomorphism from $\mathcal S_{\tau}$
into $\mathcal T(\mathcal X)$.

\end{enumerate}
\end{lemma}

\begin{proof}
(i) We have
\begin{align*}
W^*aW &=(t\ell_{\tau})^*a(t\ell_{\tau}) +
(t\ell_{\tau})^*a\ell_{\widetilde \xi}
+ \ell_{\widetilde\xi}^*a(t\ell_{\tau}) + \ell_{\widetilde\xi}^*a\ell_{\widetilde\xi}\\
&=t^2\tau(a) + 0 + 0 + (1-t^2)\tau(a)\\
&=\tau(a)1.
\end{align*}
To see that the cross terms are zero, we used that
$$
(t\ell_{\tau})^*a\ell_{\widetilde \xi}=
\ell_{(t\ell_{\tau})^*a\widetilde \xi}
$$
and that $(t\ell_{\tau})^*a\widetilde \xi=0$, which follows from
$$
\langle (t\ell_{\tau})^*a\widetilde \xi,(t\ell_{\tau})^*a\widetilde \xi\rangle 
=\widetilde\phi(t^2a^*\ell_{\tau}\ell_{\tau}^*a)=
\tau_{\Omega}(a^*\ell_{\tau}\ell_{\tau}^*a)(1-t^2)=0.
$$

Finally, to see that $W+W^*\in \mathcal T(\mathcal X)$, notice that $\ell_{\widetilde\xi}\in \mathcal T(\mathcal X)$, since we have identified $\ell_{\xi}\in \mathcal T(\mathcal X)$ with $\ell_{\widetilde\xi}$, and that
\begin{align*}
t\ell_{\tau}+t\ell_{\tau}^*=ts_{\tau}\in \mathcal D\subseteq\mathcal T(\mathcal X). 
\end{align*}

(ii) This follows from (i) and the universal property of $\mathcal T_{\tau}$.

(iii) Since $W+W^*\in \mathcal T(\mathcal X)$, the restriction of the $*$-homomorphism from (ii) to $\mathcal S_{\tau}$ ranges in $\mathcal T(\mathcal X)$.
\end{proof}


\begin{proof}[Proof of Theorem \ref{mainthm}]
Let $(A,\tau)$ be as in the theorem and assume additionally that $A$ is separable.
By the previous lemma, there exists a unital
$*$-homomorphism $\mathcal S_{\tau}\to \mathcal T(\mathcal X)$ fixing $A$, which sends $s_\tau$ to
$W+W^*=ts_\tau+\ell_\xi+\ell_\xi^*$. Combined with $\Psi\colon \mathcal T(\mathcal X)\to A^{\mathcal U}$,
we obtain $j\colon \mathcal S_{\tau}\to A^{\mathcal U}$ fixing $A$ and sending $s_\tau$ to
$\Theta(ts_\tau)+r+r^*$. Preservation of the respective traces is automatic since $\mathcal S_{\tau}$ has a unique trace (by \cite[Theorem~2]{dykema}). It follows that $A$ is selfless by the reduced-free-product description of $\mathcal S_{\tau}$ and \cite[Theorem~2.6(iii)]{selfless}. 

Let us now drop the assumption that $A$ is separable.  The properties of simplicity, infinite dimensionality,  uniqueness of trace, and strict comparison by the trace all
pass to any elementary submodel $(A',\tau|_{A'})\subseteq (A,\tau)$ (see \cite[Proposition 5.10.3, Theorem 8.2.2]{farahetal}). Thus, every separable elementary submodel of $A$ is selfless.
Passing to the direct limit over all its separable elementary submodels, we deduce that $(A,\tau)$ is also selfless \cite[Theorem 4.1]{selfless}. 
\end{proof}	

\begin{proof}[Proof of Corollary \ref{coro:HTO}]
It is well known that the Jiang--Su C*-algebra
is simple, has a unique trace, and has strict comparison
with respect to its trace \cite{jiang-su}.
By Theorem \ref{mainthm}, it is selfless.
By \cite[Theorem~2.6]{selfless}, there is an embedding
of $\mathcal Z*C_r^*(\mathbb F_\infty)$ in $\mathcal Z^{\mathcal U}$.
In particular, $C_r^*(\mathbb F_2)$ embeds in $\mathcal Z^{\mathcal U}$. Since $\mathcal Z$ embeds in $\mathcal Q$ (as the latter is unital and $\mathcal Z$-stable), $C_r^*(\mathbb F_2)$ embeds in $\mathcal Q^{\mathcal U}$.

\end{proof}

Let $(A,\tau)$ be as in the theorem and assume additionally that $A$ is separable.
By the previous lemma, there exists a unital
$*$-homomorphism $\mathcal S_{\tau}\to \mathcal T(\mathcal X)$ fixing $A$, which sends $s_\tau$ to
$W+W^*=ts_\tau+\ell_\xi+\ell_\xi^*$. Combined with $\Psi\colon \mathcal T(\mathcal X)\to A^{\mathcal U}$,
we obtain $j\colon \mathcal S_{\tau}\to A^{\mathcal U}$ fixing $A$ and sending $s_\tau$ to
$\Theta(ts_\tau)+r+r^*$. Preservation of the respective traces is automatic since $\mathcal S_{\tau}$ has a unique trace (by \cite[Theorem~2]{dykema}). It follows that $A$ is selfless by the reduced-free-product description of $\mathcal S_{\tau}$ and \cite[Theorem~2.6(iii)]{selfless}. 

Let us now drop the assumption that $A$ is separable.  The properties of simplicity, infinite dimensionality,  uniqueness of trace, and strict comparison by the trace all
pass to any elementary submodel $(A',\tau|_{A'})\subseteq (A,\tau)$ (see \cite[Proposition 5.10.3, Theorem 8.2.2]{farahetal}). Thus, every separable elementary submodel of $A$ is selfless.
Passing to the direct limit over all its separable elementary submodels, we deduce that $(A,\tau)$ is also selfless \cite[Theorem 4.1]{selfless}.
