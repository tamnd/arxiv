\label{lem:JJS-powers}
Let ${\mathscr T}=( V, E)$ be a directed tree and let $S_\lambda\in{\mathcal{B}}(\ell^2( V))$ be a weighted shift on ${\mathscr T}$. Then, for every $u\in V$ and $n\in\mathbb Z_{\geqslant0}$,

    \end{equation*}

<!-- lemma-3: body_md -->
\label{lem:compute-beta-m}
    Let ${\mathscr T} = ( V,  E)$ be a directed tree and let $S_{\lambda} \in {\mathcal{B}} (\ell ^2 ( V))$ be a weighted shift on ${\mathscr T}$. Then
    \begin{equation*}
        \beta_{m} (S_{\lambda}) e_u = \left( \sum_{j=0}^{m} (-1)^{m-j} \binom{m}{j} \left\lVert S_{\lambda}^{j}e_u  \right\rVert^{2} \right) e_u, \qquad u \in V, m \in \mathbb N.
    \end{equation*}
    Consequently, $S_{\lambda}$ is a $m$-isometry, $m \in \mathbb N$ if and only if $\beta_{m} (S_{\lambda}) e_u = 0$ for each $u \in  V$.

<!-- proof-1: body_md -->
We have by \Cref{lem:JJS-powers} that for every $j\in \mathbb Z_{\geqslant 0}$ and $u\in V$,
\begin{equation*}
S_\lambda^j e_u = \sum_{v\in\mathsf{Chi}^{\langle j\rangle}(u)} \lambda_{u|v}e_v.
\end{equation*}

If $u, v \in  V$ satisfy $u\neq v$, then $\mathsf{Chi}^{\langle j\rangle}(u)$ and $\mathsf{Chi}^{\langle j\rangle}(v)$ are disjoint. Hence, $\langle S_\lambda^j e_u,S_\lambda^j e_v\rangle=0$ for each $j \in \mathbb Z_{\geqslant 1}$ and each $u, v \in  V$ satisfying $u \ne v$. Therefore $S_\lambda^{*j}S_\lambda^j e_u = \left\lVert S_\lambda^j e_u \right\rVert^2e_u$ for every $j\in\mathbb Z_{\geqslant0}$ and $u\in V$.

It follows that
\begin{equation*}
\beta_m(S_\lambda)e_u = \left( \sum_{j=0}^{m}(-1)^{m-j}\binom{m}{j} \left\lVert S_\lambda^j e_u \right\rVert^2 \right)e_u, \qquad m \in \mathbb Z_{\geqslant 0},\, u \in  V.
\end{equation*}
Since $\{e_u\}_{u\in V}$ is an orthonormal basis of $\ell^2(V)$,
 $S_{\lambda}$ is a $m$-isometry if and only if $\beta_{m} (S_{\lambda}) e_u = 0$ for each $u \in  V$.

<!-- definition-1: body_md -->
Let $T\in\mathcal B(\mathcal H)$ be a left-invertible operator and let $L$ be its Moore--Penrose inverse. We say that $T$ \textit{possesses the weak subsequence property} if there exists a total subset ${\mathcal{F}} \subset {\mathcal{H}}$ such that for every $x \in {\mathcal{F}}$, $\{ T^n L^n x\}_{n \in \mathbb Z_{\geqslant 1}}$ has a weakly convergent subsequence.
<!-- end definition-1 -->
<!-- proposition-4: body_md -->
\label{prop:equivalences-of-WSP}
    Let $T\in\mathcal B(\mathcal H)$ be left-invertible, and let $L=(T^*T)^{-1}T^*$ be its Moore--Penrose inverse. If $T$ possesses the wandering subspace property, then it possesses the weak subsequence property. Moreover, if $T$ is analytic, then the converse holds. As a result, the following statements are equivalent whenever $T$ is analytic:
    \begin{enumerate}
        \item[\rm (i)] $T$ possesses the wandering subspace property.
        \item[\rm (ii)] $T$ possesses the weak subsequence property.
        \item[\rm (iii)] $T'$ is analytic.
    \end{enumerate}
<!-- end proposition-4 -->
<!-- proof-3: body_md -->
Define $ E:=\mathcal H\ominus T\mathcal H = \ker T^*$. We first show that if $T$ possesses the wandering subspace property then it possesses the weak subsequence property. To this end, define
    \begin{equation*}
        {\mathcal{F}} := \operatorname{span} \{ T^j  E : j \in \mathbb Z_{\geqslant 0} \}.
    \end{equation*}
    It is easy to see that for each $x \in {\mathcal{F}}$, there exists some $N \in \mathbb Z_{\geqslant 1}$ such that  $T^n L^n x = 0$ for all $n \in \mathbb Z_{\geqslant N}$. Moreover, since $T$ possesses the wandering subspace property, we have that $\bigvee {\mathcal{F}} = {\mathcal{H}}$. This shows that $T$ possesses the weak subsequence property.
    
    Conversely, assume that $T$ is analytic and possesses the weak subsequence property. Let $\mathcal F\subset\mathcal H$ be a total subset such that for every $x \in {\mathcal{F}}$, $\{ T^n L^n x\}_{n \in \mathbb Z_{\geqslant 1}}$ has a weakly convergent subsequence. We show that $T$ possesses the wandering subspace property. Let $P:=I-TL$ be the orthogonal projection onto $ E$. Hence, for every $n\in \mathbb Z_{\geqslant 1}$,
    \begin{equation*}
    I-T^nL^n=\sum_{k=0}^{n-1}T^kPL^k,
    \end{equation*}
    and therefore $(I-T^nL^n)x\in [ E]_T$ for every $x\in\mathcal H$. This follows by induction on $n \in \mathbb Z_{\geqslant 1}$. See \cite[Equations (2.1) and (2.2)]{zbMATH01572594}.
    
    First of all, we fix $x\in{\mathcal{F}}$ and choose a subsequence $(n_j)$ such that $T^{n_j}L^{n_j}x\to y$ weakly for some $y \in {\mathcal{H}}$. Let $k\in\mathbb Z_{\geqslant1}$. For every $j\in\mathbb Z_{\geqslant k}$, we have $n_j\geqslant j\geqslant k$, and hence $T^{n_j}L^{n_j}x\in T^k\mathcal H$. Since $T^k\mathcal H$ is closed, and hence weakly closed, it follows that
    \begin{equation*}
    y\in\bigcap_{k\in \mathbb Z_{\geqslant 1}}T^k \mathcal H=\{0\},
    \end{equation*}
    where the last equality follows from the analyticity of $T$. Thus $T^{n_j}L^{n_j}x\to 0$ weakly, and consequently $(I-T^{n_j}L^{n_j})x\to x$ weakly.
    
    Since $[ E]_T$ is weakly closed, we have $x\in[ E]_T$. Hence, $\mathcal F\subseteq[ E]_T$, and it follows that $\mathcal H=[ E]_T$ as ${\mathcal{F}}$ is a total set of ${\mathcal{H}}$. Therefore, $T$ possesses the wandering subspace property.

    The equivalence of \rm{(i)} and \rm{(iii)} is precisely item \rm{(ii)} of \Cref{prop:analytic-wsp-in-terms-of-cauchy-dual}.

<!-- proposition-5: body_md -->
Let ${\mathscr T} = ( V,  E)$ be a rootless directed tree, and let $S_{\lambda} \in {\mathcal{B}} (\ell ^2 ( V))$ be a norm-increasing $3$-concave weighted shift with positive weights. Then $S_{\lambda}$ is either analytic or an isometry.
    \label{prop:analytic-or-isometry}
<!-- end proposition-5 -->
<!-- proof-5: body_md -->
If $S_\lambda$ were analytic, there would be nothing to prove. Hence, assume that $S_\lambda$ is not analytic. Since $S_\lambda$ is norm-increasing, it is left-invertible. We have that ${\mathcal{H}}_\infty(S_\lambda)$ reduces $S_\lambda$ by \Cref{prop:Shimorin-hyper-range}. Consequently, $S_\lambda$ is balanced by \cite[Theorem 4.1]{MR5052237}.
    
    Since $S_\lambda$ is injective, ${\mathscr T}$ is leafless (see \cite[Proposition 3.1.7]{MR2919910}). Fix $v\in V$. Since ${\mathscr T}$ is rootless and leafless, we may choose a sequence $(v_n)_{n\in\mathbb Z}$ of vertices such that $v_0=v$ and $v_{n+1}\in\mathsf{Chi}(v_n)$ for every $n\in\mathbb Z$. Define $\gamma_n:=\left\lVert S_\lambda e_{v_n} \right\rVert$ for each $n\in\mathbb Z$, and let $S_\gamma$ denote the bilateral weighted shift on $\ell^2(\mathbb Z)$ defined by $S_\gamma e_n=\gamma_ne_{n+1}$ for each $n \in \mathbb Z$.

    Since $S_\lambda$ is balanced, it can be shown using induction and \Cref{lem:JJS-powers} that $\left\lVert S_\lambda^k e_u \right\rVert=\left\lVert S_\lambda^k e_w \right\rVert$ for every $k\in\mathbb Z_{\geqslant0}$ whenever $u$ and $w$ belong to the same generation.

    Consequently, we have
    \begin{equation*}
        \left\lVert S_\lambda ^k e_{v_n} \right\rVert^2 =\prod_{j=0}^{k-1}\gamma_{n+j}^2 =\left\lVert S_\gamma^ke_n \right\rVert^2, \qquad k\in\mathbb Z_{\geqslant0},\ n\in\mathbb Z.
    \end{equation*}
      It follows by \Cref{lem:compute-beta-m} that $S_\gamma$ is norm-increasing and $3$-concave.

      Since $1\leqslant\gamma_n\leqslant\left\lVert S_\lambda \right\rVert$ for every $n\in\mathbb Z$, the operator $S_\gamma$ is invertible. We have that $S_\gamma$ is unitary by using \Cref{prop:Shimorin-hyper-range} again. Thus $\gamma_n=1$ for every $n\in\mathbb Z$. Since $v\in V$ was arbitrary, $\left\lVert S_\lambda e_v \right\rVert=1$ for every $v\in V$, and using \Cref{lem:compute-beta-m} with $m=1$, we have that $S_\lambda$ is an isometry.
<!-- end proof-5 -->
<!-- lemma-5: body_md -->
Let $\left\{ a_n \right\}_{n \in \mathbb Z_{\geqslant 1}}$ be a nonnegative nondecreasing sequence which is not identically zero. Define $\sigma_n := \frac{1}{n} \sum_{k=1}^{n} a_k$ for each $n \in \mathbb Z_{\geqslant 1}$. Then we have
    \begin{equation*}
        \liminf_{n \to \infty} \frac{a_n}{\sigma_n ^{2}} < \infty.
    \end{equation*}
    \label{lem:Cesaro}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
First, we observe that since $\{ a_n \}_{n \in \mathbb Z_{\geqslant1}}$ is nondecreasing, $\{ \sigma_{n} \}_{n \in \mathbb Z_{\geqslant 1}}$ is also nondecreasing. Indeed, by monotonicity of $\{ a_n \}_{n \in \mathbb Z_{\geqslant1}}$, we have $\sigma_n \leqslant a_{n+1}$ for each $n \in \mathbb Z_{\geqslant 1}$, and consequently,
    \begin{equation*}
        \begin{aligned}
            \sigma_{n+1} - \sigma_n = \frac{1}{n+1} \sum_{k=1}^{n+1} a_k - \sigma_n 
                                    = \frac{n \sigma_n + a_{n+1}}{n+1} - \sigma_n 
                                    = \frac{a_{n+1} - \sigma_n}{n+1} \geqslant 0,
        \end{aligned}
    \end{equation*}
    for all $n \in \mathbb Z_{\geqslant 1}$.

    Moreover, since $\left\{ a_n \right\}_{n \in \mathbb Z_{\geqslant 1}}$ is not identically zero we have that there is some $N \in \mathbb Z_{\geqslant 1}$ such that $\sigma_n > 0$ for each $n \in \mathbb Z_{\geqslant N}$.

    Now, observe that $a_n = n \sigma_n - (n-1) \sigma_{n-1} = \sigma_n + (n-1) (\sigma_n - \sigma_{n-1})$ for each $n \in \mathbb Z_{\geqslant 2}$. Using the fact that $\sigma_{n-1} \leqslant \sigma_{n}$ for each $n \in \mathbb Z_{\geqslant 2}$, it follows that for each $n \in \mathbb Z_{\geqslant N+1}$,
    \begin{equation}
        \begin{aligned}
        \frac{a_n}{\sigma_n ^2} &= \frac{\sigma_n + (n-1) (\sigma_n - \sigma_{n-1})}{\sigma_n ^2} \\
        &= \frac{1}{\sigma_n} + \frac{(n-1) (\sigma_n - \sigma_{n-1})}{\sigma_{n} ^2} \\
        &\leqslant \frac{1}{\sigma_n} + (n-1) \left( \frac{1}{\sigma_{n-1}} - \frac{1}{\sigma_n} \right).
    \end{aligned}
    \label{eq:Cesaro-estimate}
    \end{equation}
    
    We define $c_n := \frac{1}{\sigma_{n-1}} - \frac{1}{\sigma_n}$ for each $n \in \mathbb Z_{\geqslant N+1}$. Observe that $c_n \geqslant 0$ for each $n \in \mathbb Z_{\geqslant N+1}$ and $\sum_{n = N+1}^{\infty} c_n < \infty$. Consequently, $\liminf_{n \to \infty} nc_n = 0$. Indeed, otherwise there would exist some $\varepsilon > 0$ and some $N_0 \in \mathbb Z_{\geqslant N+1}$ such that $c_n \geqslant \varepsilon/n$ for all $n \in \mathbb Z_{\geqslant N_0}$, but this would imply that $\sum_{n = N+1}^{\infty} c_n = \infty$.

    We now select a strictly increasing sequence $\{ n_j \}_{j \in \mathbb Z_{\geqslant 1}} \subset \mathbb Z_{\geqslant N+1}$ such that $n_j c_{n_j} \to 0$. From \eqref{eq:Cesaro-estimate}, we have that 
    \begin{equation*}
        \frac{a_{n_j}}{\sigma_{n_j} ^2} \leqslant \frac{1}{\sigma_N} + (n_j - 1) c_{n_j}, \qquad j \in \mathbb Z_{\geqslant 1}.
    \end{equation*}
    Since the right hand side of the previous inequality is bounded, we have $\liminf_{n \to \infty} \frac{a_{n}}{\sigma_{n} ^2} < \infty$. This completes the proof of the lemma.

<!-- theorem-1: body_md -->
Let ${\mathscr T} = ( V,  E)$ be a rootless directed tree, and let $S_{\lambda} \in {\mathcal{B}} (\ell ^2 ( V))$ be an analytic, norm-increasing $3$-concave weighted shift with positive weights. Then $S_{\lambda}$ possesses the wandering subspace property.
    \label{thm:result-for-positive-weights}
<!-- end theorem-1 -->
<!-- proof-7: body_md -->
We begin by defining $\beta_j (v) = \left\langle \beta_j (S_{\lambda})e_v, e_v \right\rangle$ for each $j \in \{1,2\}$ and each $v\in  V$. Since $S_{\lambda}$ is norm-increasing, we have $\beta_1 (S_{\lambda}) \geqslant 0$. We have by using \Cref{lem:m-concave} and $3$-concavity that $\beta_2 (S_{\lambda}) \geqslant 0$ and 
    \begin{equation}
        S_{\lambda}^{*n}S_{\lambda}^{n} \leqslant I + n \beta_1 (S_{\lambda}) + \binom{n}{2} \beta_2 (S_{\lambda}), \qquad n \in \mathbb Z_{\geqslant 0}.
        \label{eq:shift-3-concave}
    \end{equation}
    As $\beta_j (S_\lambda) \geqslant 0$ for each $j \in \{1,2\}$ and $S_{\lambda}$ is bounded, we choose $M > 0$ such that $0 \leqslant \beta_j (v) \leqslant M$ for every $v \in  V$ and every $j \in \{1,2\}$.

    Now, we fix $u \in  V$ and define $u_n := \mathsf{par}^{\left\langle n \right\rangle} (u)$ for each $n \in \mathbb Z_{\geqslant 0}$. In view of \Cref{prop:equivalences-of-WSP}, we show that $\{ S_{\lambda} ^{n} L_{\lambda} ^{n} e_u \}_{n \in \mathbb Z_{\geqslant 0}}$ has a bounded subsequence. To this end, we define for $n \in \mathbb Z_{\geqslant 0}$,
    \begin{equation*} 
    D_n := \frac{1}{\lambda_{u_n\mid u} ^{2}} \prod_{k=1}^{n}\bigl(1+\beta_1(u_k)\bigr)\quad \text{and} \quad
    H_n := \sum_{k=1}^{n} \frac{\beta_2(u_k)}{\lambda_{u_k\mid u} ^{2}}.
    \end{equation*}
    Using \Cref{lem:all-about-L-lambda} and equation \eqref{eq:shift-3-concave} to estimate $\left\lVert S_{\lambda} ^{n} L_{\lambda} ^{n} e_u \right\rVert$ for $n \in \mathbb Z_{\geqslant 0}$, we have
    \begin{equation} 
           \left\lVert S_{\lambda} ^{n} L_{\lambda}^{n} e_u \right\rVert ^2 \leqslant \mathbf{I}_n + \mathbf{II}_n + \mathbf{III}_n, \qquad n \in \mathbb Z_{\geqslant 0},
           \label{eq:summands-of-S-L}
    \end{equation}
    where we define for each $n \in \mathbb Z_{\geqslant 0}$,
    \begin{equation*}
        \mathbf{I}_n  := \frac{1}{\lambda_{u_n \mid u} ^{2} D_{n}^{2}}, \quad \mathbf{II}_n := \frac{n \beta_1 (u_n)}{\lambda_{u_n \mid u} ^{2} D_{n}^{2}}, \quad \mathbf{III}_n := \frac{\binom{n}{2} \beta_2 (u_n)}{\lambda_{u_n \mid u} ^{2} D_{n}^{2}}.
    \end{equation*}

    In view of inequality \eqref{eq:summands-of-S-L}, it suffices for us to show that $\{ \mathbf{I}_n \}_{n\in\mathbb Z_{\geqslant 0}}$ and $\{ \mathbf{II}_n \}_{n\in\mathbb Z_{\geqslant 0}}$  are bounded and that $\{ \mathbf{III}_n \}_{n\in\mathbb Z_{\geqslant 0}}$  has a bounded subsequence. We make the following claims in order to establish these assertions.

    \textit{Claim 1.} The following inequalities hold for all $n \in \mathbb Z_{\geqslant 0}$,
    \begin{equation*}
        \begin{gathered}
            D_n \geqslant 1, \qquad 
            D_n \geqslant \frac{1}{\lambda_{u_n \mid u} ^{2}} + \sum_{k=1}^{n} \frac{\beta_1 (u_k)}{\lambda_{u_k \mid u} ^{2}}, \qquad H_n \leqslant MD_n.
        \end{gathered}
    \end{equation*}
    Moreover, the sequence $\left\{ \frac{\beta_2 (u_n)}{\lambda_{u_n \mid u} ^2} \right\}_{n \in \mathbb Z_{\geqslant 0}}$ is nonnegative and nondecreasing. If the previous sequence is identically zero, then the sequence $\left\{ \frac{\beta_1 (u_n)}{\lambda_{u_n \mid u} ^2} \right\}_{n \in \mathbb Z_{\geqslant 0}}$ is nonnegative and nondecreasing.

    \begin{proof}[Proof of Claim 1.] The following equalities $S_{\lambda}^* S_{\lambda} =I+\beta_{1}(S_\lambda)$,  $S_{\lambda}^*\beta_1(S_{\lambda})S_{\lambda} = \beta_{1}(S_{\lambda})+\beta_{2}(S_{\lambda})$ and the inequality $S_{\lambda}^*\beta_{2}(S_{\lambda})S_{\lambda} = \beta_{2}(S_{\lambda})+\beta_{3}(S_{\lambda}) \leqslant \beta_{2}(S_{\lambda})$
    together with \Cref{lem:JJS-powers,lem:compute-beta-m} imply that the following holds for each $v \in  V$,
    \begin{equation}

    \end{equation}

            \end{equation*}
        where we have used $\delta_2(m-1)=\delta_3(m)$ and $\delta_3(m-1)=\delta_4(m)$ for all $m \in \mathbb Z$.
    
        Hence, the map $m\in\mathbb Z\mapsto a_{m-1}-a_m+b_{m-1}+b_m-\delta_2(m)+\delta_3(m)$ is constant. Therefore, there exists $d\in\mathbb R$ such that
            \begin{equation*}
                a_{m-1}-a_m+b_{m-1}+b_m = d+\delta_2(m)-\delta_3(m),\qquad m\in\mathbb Z.
            \end{equation*}
        This completes the proof of the equivalences of $(ii)$ and $(iii)$, and hence, of $(i)$, $(ii)$ and $(iii)$.
    
        Now, suppose that one of the equivalent conditions $(i)$, $(ii)$ or $(iii)$ holds. Then by item $(iii)$, we have for $m \in \mathbb Z_{\geqslant 4}$ that $d= a_{m-1} - a_{m} + b_{m-1} + b_{m}$. Consequently, for $n \in \mathbb Z_{\geqslant 4}$, we have 
        \begin{equation*}
            (n-3) d = a_3 - a_{n} + \sum_{k=4}^{n} (b_{k-1} + b_k).
        \end{equation*}
        Dividing both sides by $n-3$ and using the fact that $\{ a_m \}_{m \in \mathbb Z}$ is bounded, we have by letting $n \to \infty$ that  $2\inf_{m\in\mathbb Z}b_m \leqslant d \leqslant 2\sup_{m\in\mathbb Z}b_m$. Consequently, $d > 0$.
        
        We proceed to show that $S_\lambda$ is never $3$-concave. Again, it follows by \Cref{lem:compute-beta-m} and the earlier computations that 
            \begin{equation*}
                \beta_3(S_\lambda)e_{(0,m)} = \lambda_{(1,m)}^2 \left( a_{m-1}-a_m+b_{m-1}+b_m -\delta_2(m)+\delta_3(m) \right)e_{(0,m)}.
            \end{equation*}
        If $S_\lambda$ were $3$-concave, then $\beta_3(S_\lambda)\leqslant0$. Consequently, for each $m \in \mathbb Z_{\geqslant 4}$, we have $a_{m-1}-a_m+b_{m-1}+b_m\leqslant0$. Thus, $a_m-a_{m-1} \geqslant b_{m-1}+b_m \geqslant 2\inf_{k\in\mathbb Z}b_k$ for each $m \in \mathbb Z_{\geqslant 4}$. Taking sums, we have for $n \in \mathbb Z_{\geqslant 4}$, that $a_n \geqslant a_3+2(n-3)\inf_{k\in\mathbb Z}b_k$. This would contradict the fact that $\{ a_m \}_{m \in \mathbb Z}$ is bounded. Thus, $S_{\lambda}$ cannot be $3$-concave. In particular, it cannot be a $3$-isometry. Hence, whenever $(i), (ii)$ or $(iii)$ holds, $S_{\lambda}$ must be a strict $4$-isometry.

<!-- proposition-15: body_md -->
\label{prop:classify-3-isometries}
        Let $p (t) =  1 + a t + b t^2, \quad a, b >0$ and let $c,d >0$. Consider the weighted shift $S_{\lambda}$ on ${\mathscr T}_{qb}$ with the system of weights
            \begin{equation}
        \label{eq:CS-weights-3-isometry}
        \begin{gathered}
            \lambda_{(n,m)} = \begin{dcases}
                c, & n=0, \quad\quad m\in \mathbb Z, \\
                d, & n=1, \quad\quad m\in \mathbb Z, \\ 
                \sqrt{\frac{p(n-1)}{p(n-2)}}, & n \in \mathbb Z_{\geqslant 2},\quad m \in \mathbb Z.
            \end{dcases}
        \end{gathered}
    \end{equation}
    Then $S_{\lambda}$ is bounded and left-invertible. The following statements hold:
    \begin{enumerate}
        \item[\rm (i)] \begin{equation*}
            \inf_{v \in V} \left\lVert S_{\lambda} e_v \right\rVert^{2} = \min \{ c^2 + d^2, 1 \}.
        \end{equation*}
        \item[\rm (ii)] $S_{\lambda}$ is norm-increasing if and only if $c^2 + d^2 \geqslant 1$.
        \item[\rm (iii)] $S_\lambda$ is analytic if and only if $c \leqslant 1$.
        \item[\rm (iv)] $S_{\lambda}$ posseses the wandering subspace property if and only if $c^2 + d^2 > c$.
        \item[\rm (v)] $S_{\lambda}$ is a $3$-isometry if and only if
        \begin{equation*}
            d^2 ((1-c^2)a - (1+c^2)b) = (1-c^2)^{2} (c^2 + d^2 -1).
        \end{equation*}
        Moreover, whenever $S_\lambda$ is a $3$-isometry, it is strict $3$-isometry.
    \end{enumerate}

<!-- proof-19: body_md -->
Observe that 
        \begin{equation*}
            \left\lVert S_{\lambda} e_{(0,m)} \right\rVert^{2} = c^2 +d ^2, \quad \left\lVert S_{\lambda} e_{(n,m)} \right\rVert ^{2} = \frac{p(n)}{p(n-1)}, \qquad m \in \mathbb Z, n \in \mathbb Z_{\geqslant 1}.
        \end{equation*}
        Since $\frac{p(n)}{p(n-1)} \to 1$ as $n \to \infty$, $\sup_{(n,m) \in V} \left\lVert S_{\lambda} e_{(n,m)} \right\rVert^2 < \infty$, hence $S_{\lambda}$ is bounded. Moreover, since $S_{\lambda} ^{*} S_{\lambda}$ is a diagonal operator with entries $\left\lVert S_{\lambda} e_{v} \right\rVert^{2}, \quad v \in V$, we have that $\inf_{v \in V} \left\lVert S_{\lambda} e_v \right\rVert^{2} = \min \{ c^2 + d^2, 1\} > 0$. Thus, $S_{\lambda}$ is left-invertible and norm-increasing if and only if $c^2 + d^2 \geqslant 1$. This proves {\rm (i)} and {\rm (ii)}.

        We proceed to prove {\rm (iii)}. By \Cref{prop:analytic-criterion}, it is necessary and sufficient to prove $\alpha_{\lambda} ((0,0)) = \infty$ in order to prove that $S_{\lambda}$ is analytic. By definitions, we have that $\mathsf{par} ^{\langle n\rangle}((0,0))=(0,n)$ and $A((0,0),n)=\{(n,n)\}$ for every $n\in\mathbb Z_{\geqslant 1}$. Hence, $\lambda_{(0,n)|(n,n)}=d\sqrt{p(n-1)}$ and $\lambda_{(0,n)|(0,0)}=c^n$. Thus,
        \begin{equation*}
            \alpha_\lambda((0,0))=1+d^2\sum_{n=1}^{\infty}\frac{p(n-1)}{c^{2n}}.
        \end{equation*}
        It is easy to see that $\alpha_\lambda((0,0))=\infty$ if and only if $c\leqslant 1$. This completes the proof of item {\rm (iii)}.

        We proceed to prove item {\rm (iv)}. Using \Cref{prop:analytic-wsp-in-terms-of-cauchy-dual}, $S_{\lambda}$ possesses the wandering subspace property if and only if the Cauchy dual $S_{\lambda}'$ is analytic. Note that $S_{\lambda} '$ is a weighted shift on ${\mathscr T}_{qb}$ with positive weights whose weights are given by
        \begin{equation*}
            \lambda_{u} ' = \frac{\lambda_u}{\left\lVert S_{\lambda} e_{\mathsf{par} (u)} \right\rVert^{2}}, \qquad u \in V.
        \end{equation*}
        Therefore, we have that 
            \begin{equation*}
                \lambda'_{(n,m)} =\begin{dcases}
                    \frac{c}{c^2 + d^2}, & n=0, m\in \mathbb Z, \\
                    \frac{d}{c^2 + d^2}, & n=1, m \in \mathbb Z, \\
                    \sqrt{\frac{p(n-2)}{p(n-1)}}, & n \in \mathbb Z_{\geqslant 2}, m \in \mathbb Z.
                \end{dcases}
            \end{equation*}
        and, hence, 
        \begin{equation*}
            \lambda ' _{(0, n) | (0,0)} = \left( \frac{c}{c^2+d^2} \right)^{n}, \quad \lambda'_{(0,n) \mid (n,n)} = \frac{d}{(c^2 +d^2) \sqrt{p (n-1)}}, \qquad n \in \mathbb Z_{\geqslant 1}.
        \end{equation*}
        Observe that 
        \begin{equation*}
            \alpha_{\lambda'} ((0,0)) = 1 + \frac{d^2}{(c^2 + d^2)^2} \sum_{n=1}^{\infty} \left( \frac{(c^2 + d^2)^2}{c^2} \right)^{n} \frac{1}{p(n-1)}.
        \end{equation*}
        From here, it is easy to see that $\alpha_{\lambda'} ((0,0)) = \infty$ if and only if $c^2 + d^2 > c$. Thus, $S_{\lambda}$ possesses the wandering subspace property if and only if $c^2 + d^2 > c$. This proves item {\rm (iv)}.

        The proof of item {\rm (v)} is similar to that of \Cref{thm:classify-4-isometries} and follows from \Cref{lem:compute-beta-m}. We omit the proof.
