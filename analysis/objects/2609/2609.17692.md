<!-- prop-1: body_md -->
The elements
    \begin{equation*}
        \mathbf{e} = \frac{1 + \mathbf{i}\,\mathbf{j}}{2} \quad \text{and} \quad
        \mathbf{e}^\dagger = \frac{1 - \mathbf{i}\,\mathbf{j}}{2}
    \end{equation*}
    satisfy the following properties:
    \begin{enumerate}
        \item $\mathbf{e}\,\mathbf{e}^\dagger = \mathbf{e}^\dagger\,\mathbf{e} = 0$ (they are orthogonal zero divisors);
        \item $\mathbf{e}^2 = \mathbf{e}$ and $(\mathbf{e}^\dagger)^2 = \mathbf{e}^\dagger$ (idempotency);
        \item $\mathbf{e} + \mathbf{e}^\dagger = 1$ and $\mathbf{e} - \mathbf{e}^\dagger = \mathbf{i}\,\mathbf{j}$.
    \end{enumerate}
<!-- end prop-1 -->
<!-- defn-1: body_md -->
The \emph{null cone} of $\mathbb{BC}^n$ is the set
    \begin{equation*}
        \mathcal{N}_n = \{ v = v^1\mathbf{e} + v^2\mathbf{e}^\dagger \in \mathbb{BC}^n \mid v^1 = 0 \text{ or } v^2 = 0 \text{ in } \mathbb{C}^n \}.
    \end{equation*}
    A vector $v \notin \mathcal{N}_n$, i.e. with both $v^1$ and $v^2$ non-zero, is called a \emph{regular vector}.
<!-- end defn-1 -->
<!-- defn-2: body_md -->
\label{def:inner-product}
    For $u = u^1\mathbf{e} + u^2\mathbf{e}^\dagger$ and $v = v^1\mathbf{e} + v^2\mathbf{e}^\dagger$ in $\mathbb{BC}^n$, define
    \begin{equation*}
        \langle u, v \rangle_{\mathbb{BC}}
        = \langle u^1, v^1 \rangle_{\mathbb{C}} \mathbf{e}
          + \langle u^2, v^2 \rangle_{\mathbb{C}} \mathbf{e}^\dagger,
    \end{equation*}
    where $\langle \cdot, \cdot \rangle_{\mathbb{C}}$ denotes the standard complex Hermitian inner product.
<!-- end defn-2 -->
<!-- defn-3: body_md -->
The \emph{bicomplex general linear group} is
    \begin{equation*}
        \mathrm{GL}(n,\mathbb{BC})
        = \{ A \in \mathrm{M}_n(\mathbb{BC}) \mid \det(A) \text{ is a unit} \}.
    \end{equation*}
<!-- end defn-3 -->
<!-- defn-4: body_md -->
A \emph{bicomplex fiber bundle of rank $n$} over $B$ is a fiber bundle $\pi:E \to B$ with typical fiber $\mathbb{BC}^n$ and structure group $\operatorname{GL}(n,\mathbb{BC})$. Equivalently, there exists an open cover $\{U_\alpha\}$ of $B$ and trivializations
    \begin{equation*}
        \phi_\alpha : \pi^{-1}(U_\alpha) \to U_\alpha \times \mathbb{BC}^n
    \end{equation*}
    such that the transition functions
    \begin{equation*}
        g_{\alpha\beta} : U_\alpha \cap U_\beta \to \operatorname{GL}(n,\mathbb{BC})
    \end{equation*}
    are continuous and satisfy the usual cocycle conditions, and they also preserve the bicomplex structure.
<!-- end defn-4 -->
<!-- example-1: body_md -->
The \emph{trivial bundle} $E = B \times \mathbb{BC}^n$ with the projection onto the first factor is a bicomplex fiber bundle of rank $n$.
<!-- end example-1 -->
<!-- example-2: body_md -->
\label{ex: universal}
Let $\underline{\mathbb C}\mathbf{e} = B \times \mathbb C\mathbf{e}$ and $\underline{\mathbb C}\mathbf{e}^\dagger = B \times \mathbb C\mathbf{e}^\dagger$ denote the trivial complex line bundles whose fibers are the one-dimensional complex vector spaces $\mathbb C\mathbf{e}$ and $\mathbb C\mathbf{e}^\dagger$, respectively. Given a complex vector bundle $E^1 \to B$, we define
\begin{equation*}
    E^1 \mathbf{e} = E^1 \otimes_{\mathbb C} \underline{\mathbb C}\mathbf{e}.
\end{equation*}
Similarly, for a complex bundle \(E^2 \to B\),
\begin{equation*}
    E^2 \mathbf{e}^\dagger = E^2 \otimes_{\mathbb C} \underline{\mathbb C}\mathbf{e}^\dagger.
\end{equation*}
Since $\underline{\mathbb C}\mathbf{e}$ and $\underline{\mathbb C}\mathbf{e}^\dagger$ are trivial as complex line bundles, we have isomorphisms $E^1\mathbf{e} \cong E^1$ and $E^2\mathbf{e}^\dagger \cong E^2$ as complex vector bundles. Nevertheless, the notation $E^1\mathbf{e}$ and $E^2\mathbf{e}^\dagger$ is useful because it explicitly records the action of the idempotents: an element $v^1\mathbf{e} \in E^1\mathbf{e}$ is multiplied by \(\mathbf{e}\) as $(v^1\mathbf{e})\mathbf{e} = v^1\mathbf{e}^2 = v^1\mathbf{e}$, and similarly for \(\mathbf{e}^\dagger\). We shall use this shorthand throughout the paper.

Let $\pi^1:E^1 \to B$ and $\pi^2:E^2 \to B$ be complex vector bundles of rank $n$. Then
\begin{equation*}
    E = (E^1 \otimes_{\mathbb C} \underline{\mathbb C}\mathbf{e}) \oplus (E^2 \otimes_{\mathbb C} \underline{\mathbb C}\mathbf{e}^\dagger)
\end{equation*}
is a bicomplex fiber bundle of rank $n$, with the bicomplex structure defined by the idempotent action. In the previous notation, we write
\begin{equation*}
    E \cong E^1\mathbf{e} \oplus E^2\mathbf{e}^\dagger.

<!-- prop-2: body_md -->
\label{prop:idempotent_bundles}
    Let \(\pi:E \to B\) be a bicomplex fiber bundle of rank \(n\). Then there exist complex vector bundles \(\pi^1:E^1 \to B\) and \(\pi^2:E^2 \to B\), each of rank \(n\), and an isomorphism of bicomplex bundles
    \begin{equation*}
        E \cong E^1\mathbf{e} \oplus E^2\mathbf{e}^\dagger.
    \end{equation*}
    Moreover, this decomposition is canonical up to isomorphism.
<!-- end prop-2 -->
<!-- proof-1: body_md -->
We define the complex bundles
  \begin{equation*}
  E^1\mathbf{e}= E \otimes_{\mathbb{BC}} \underline{\mathbb{C}}\mathbf{e} \quad \text{and}\quad
  E^2\mathbf{e}^\dagger= E \otimes_{\mathbb{BC}} \underline{\mathbb{C}}\mathbf{e}^\dagger.
  \end{equation*}
  Since $\mathbb{BC} \cong \mathbb{C}\mathbf{e} \oplus \mathbb{C}\mathbf{e}^\dagger$ as $\mathbb{BC}$-modules and tensor products distribute over direct sums, the canonical isomorphism $E \cong E \otimes_{\mathbb{BC}} \mathbb{BC}$ induces
  \begin{equation*}
  E \cong (E \otimes_{\mathbb{BC}} \mathbb{C}\mathbf{e}) \oplus (E \otimes_{\mathbb{BC}} \mathbb{C}\mathbf{e}^\dagger) = E^1\mathbf{e} \oplus E^2\mathbf{e}^\dagger.
  \end{equation*}
  This decomposition is intrinsic, as it depends only on the algebraic splitting of the ring $\mathbb{BC}$ and not on any choice of cover or trivializations.

  To verify that $E^1\mathbf{e}$ and $E^2\mathbf{e}^\dagger$ are complex bundles of rank $n$, let $\{U_\alpha\}$ be a trivializing cover for $E$ with transition functions $g_{\alpha\beta}: U_{\alpha\beta} \to \mathrm{GL}(n,\mathbb{BC})$. The idempotent decomposition writes $g_{\alpha\beta}=g_{\alpha\beta}^1\mathbf{e}+g_{\alpha\beta}^2\mathbf{e}^\dagger$, with $g_{\alpha\beta}^1,g_{\alpha\beta}^2\in \mathrm{GL}(n,\mathbb{C})$. Under the local trivialization $\phi_\alpha: E|_{U_\alpha} \to U_\alpha \times \mathbb{BC}^n$, the induced bundle $(E^1\mathbf{e})|_{U_\alpha}$ is identified with $U_\alpha \times \mathbb{C}^n$ via
  \begin{gather*}
  \phi_\alpha^1: (E^1\mathbf{e})|_{U_\alpha} \to U_\alpha \times \mathbb{C}^n, \\ v^1\mathbf{e} \mapsto v^1.
  \end{gather*}
  A direct computation shows that the transition function of $E^1\mathbf{e}$ on $U_{\alpha\beta}$ is precisely $g_{\alpha\beta}^1$; similarly, that of $E^2\mathbf{e}^\dagger$ is $g_{\alpha\beta}^2$. Hence both are complex bundles of rank $n$.

  Finally, since the construction via tensor products with the global bundles $\underline{\mathbb{C}}\mathbf{e}$ and $\underline{\mathbb{C}}\mathbf{e}^\dagger$ involves no coordinate choices, the decomposition is canonical up to isomorphism.

\label{ex:regular_from_zero_divisors}
Let $B=\mathbb{C}\mathbb{P}^{1}\times\mathbb{C}\mathbb{P}^{1}$ and denote by 
$p_{1},p_{2}\colon B\to\mathbb{C}\mathbb{P}^{1}$ the projections onto the two factors. 
Let $\gamma_{\mathbb{C}}\to\mathbb{C}\mathbb{P}^{1}$ be the complex tautological line bundle.

Consider the two bicomplex line bundles over $B$ defined by
\begin{equation*}
L_{1}=\bigl(p_{1}^{*}\gamma_{\mathbb{C}}\bigr)\mathbf{e}\oplus\underline{\mathbb{C}}\mathbf{e}^\dagger
\quad\text{and}\quad
L_{2}=\underline{\mathbb{C}}\,\mathbf{e}\;\oplus\;\bigl(p_{2}^{*}\gamma_{\mathbb{C}}\bigr)\mathbf{e}^{\dagger},
\end{equation*}
and set $E=L_{1}\oplus L_{2}$, a bicomplex bundle of rank $2$.

Since $\gamma_{\mathbb{C}}$ has no non-zero global sections, every section of $L_{1}$ is of the form 
$\sigma^{1}\mathbf{e}^{\dagger}$ with $\sigma^{1}\in\Gamma(B,\underline{\mathbb{C}})$; likewise, every section of $L_{2}$ is of the form 
$\sigma^{2}\mathbf{e}$ with $\sigma^{2}\in\Gamma(B,\underline{\mathbb{C}})$. 
Take the constant sections
\begin{equation*}
s^{1}=0\cdot\mathbf{e}+1\cdot\mathbf{e}^\dagger\in\Gamma(B,L_{1})\quad\text{and}\quad
s^{2}=1\cdot\mathbf{e}+0\cdot\mathbf{e}^\dagger\in\Gamma(B,L_{2}).
\end{equation*}
Viewed as sections of the summands of $E$, both $s^{1}$ and $s^{2}$ are pointwise zero divisors: 
at each $x\in B$ we have $s^{1}(x)\in\mathbb{C}\mathbf{e}^{\dagger}$ and $s^{2}(x)\in\mathbb{C}\mathbf{e}$.

Now form the global section $s=(s^{1},s^{2})\in\Gamma(B,E)$. 
Using the canonical identification $E_{x}\cong\mathbb{B}\mathbb{C}^{2}$, its idempotent decomposition at any point is
\begin{equation*}
s(x)=\begin{pmatrix}0\\ 1\end{pmatrix}\mathbf{e}+\begin{pmatrix}1\\ 0\end{pmatrix}\mathbf{e}^{\dagger}.
\end{equation*}
Both complex components are non-zero vectors in $\mathbb{C}^{2}$, hence $s(x)\notin\mathcal{N}_{2}$ for every $x\in B$. Therefore $s$ is a regular section of $E$.

This shows that regularity is a property of the idempotent decomposition of the total vector in the fiber, not of the individual summands: two pointwise zero-divisor sections can combine to produce a regular section     of the direct sum.

<!-- prop-3: body_md -->
A bicomplex fiber bundle $\pi:E \to B$ of rank $n$ is trivial if and only if it admits $n$ linearly independent regular sections $s_1,\dots,s_n$.
<!-- end prop-3 -->
<!-- proof-2: body_md -->
Define a map
    \begin{gather*}
        \phi : B \times \mathbb{BC}^n \to E, \\ 
        \phi(x,(v_1,\dots,v_n)) = \sum_{i=1}^n v_i s_i(x).
    \end{gather*}
    Since each $s_i$ is regular and linearly independent, the fibers $\phi_x : \mathbb{BC}^n \to E_x$ are isomorphisms of $\mathbb{BC}$-modules for every $x \in B$. 
    
    To see that \(\phi\) is a fiberwise isomorphism, let \(A(x) \in \mathrm{M}_n(\mathbb{BC})\) be the matrix whose columns are the coordinates of \(s_1(x),\dots,s_n(x)\) in a bicomplex trivialization of \(E_x\). The idempotent decomposition gives
\begin{equation*}
A(x) = A^1(x)\mathbf{e} + A^2(x)\mathbf{e}^\dagger,
\end{equation*}
Since the sections $\{s_i\}$ are linearly independent, their idempotent
components $\{s_i^1\}$ and $\{s_i^2\}$ are linearly independent in
$\mathbb{C}^n$. Hence the matrices $A^1$ and $A^2$ are invertible.
Therefore, the bicomplex matrix $A$ is invertible.
Therefore, \(\phi_x: \mathbb{BC}^n \to E_x\) is an isomorphism of \(\mathbb{BC}\)-modules for every \(x \in B\).

<!-- end thm-1 -->
<!-- lem-2: body_md -->
\label{lem:classifying_map}
The map
\begin{gather*}
\iota_{\phi}:B\to\operatorname{Gr}_{m}(\mathbb{BC}^{N}),\\
x\mapsto\ker(\phi_{x})^{\perp},    
\end{gather*}
is well defined and continuous.  Moreover, there is a natural isomorphism
\[
\iota_{\phi}^{*}(\gamma_{\mathbb{B}\mathbb{C}}^{m,N})\cong E.
\]
<!-- end lem-2 -->
<!-- proof-4: body_md -->
By the bicomplex Gram--Schmidt process (Subsection~\ref{subsec:gram-schmidt}),
$\mathbb{BC}^{N}=\ker(\phi_{x})\oplus\ker(\phi_{x})^{\perp}$ and the bicomplex rank of $\ker(\phi_{x})^{\perp}$ is $m$ for every $x\in B$. 
By definition,
\begin{equation*}
\iota_{\phi}^{*}(\gamma_{\mathbb{B}\mathbb{C}}^{m,N})
=\bigl\{(x,v)\in B\times\mathbb{B}\mathbb{C}^{N}\mid v\in\ker(\phi_{x})^{\perp}\bigr\}.
\end{equation*}
Since $\phi_{x}$ is surjective and $\ker(\phi_{x})\cap\ker(\phi_{x})^{\perp}=\{0\}$,
the restriction $\phi_{x}|_{\ker(\phi_{x})^{\perp}}:\ker(\phi_{x})^{\perp}\to E_{x}$
is an isomorphism of $\mathbb{B}\mathbb{C}$-modules for each $x$.  These fiberwise
isomorphisms assemble into a global bundle isomorphism
$\iota_{\phi}^{*}(\gamma_{\mathbb{B}\mathbb{C}}^{m,N})\cong E$.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
We have already shown that the map $[f] \mapsto f^*(\gamma_{\mathbb{BC}}^m)$ is well-defined.

To prove surjectivity, let $E\to B$ be a bicomplex fiber bundle of rank $m$. Choose an
epimorphism $\phi:B\times\mathbb{BC}^N\to E$ as above, and consider the induced map
\begin{align*}
\iota_{\phi}:B&\to\operatorname{Gr}_m(\mathbb{BC}^{N})
\subset\operatorname{Gr}_m(\mathbb{BC}^{\infty}),\\
x&\mapsto\ker(\phi_{x})^{\perp}.
\end{align*}
By Lemma~\ref{lem:classifying_map}, $\iota_{\phi}^{*}(\gamma_{\mathbb{B}\mathbb{C}}^{m})\cong E$.

To prove injectivity, suppose $f, g: B \to \operatorname{Gr}_m(\mathbb{BC}^\infty)$ satisfy $f^*(\gamma_{\mathbb{BC}}^m) \cong g^*(\gamma_{\mathbb{BC}}^m)$. By compactness, we may assume their images lie in a finite Grassmannian $\operatorname{Gr}_m(\mathbb{BC}^N)$. The isomorphism of the pullback bundles gives an isomorphism of the associated epimorphisms from the trivial bundle to the pullback. Standard arguments (see \cite[Theorem~1.4.15]{Ati1967}) show that the maps $f$ and $g$ are homotopic. Alternatively, one can use the fact that the space of epimorphisms $B \times \mathbb{BC}^N \to E$ is path-connected, since $\operatorname{GL}(N,\mathbb{BC})$ is connected (indeed, $\operatorname{GL}(N,\mathbb{BC}) \cong \operatorname{GL}(N,\mathbb{C})\mathbf{e} \oplus \operatorname{GL}(N,\mathbb{C})\mathbf{e}^\dagger$, and $\operatorname{GL}(N,\mathbb{C})$ is connected). Hence the induced maps to the Grassmannian are homotopic.

<!-- remark-2: body_md -->
If $E=E^{1}\mathbf{e}\oplus E^{2}\mathbf{e}^\dagger$, then any epimorphism
$\phi:B\times\mathbb{BC}^{N}\to E$ decomposes as

$\phi^{1}:B\times\mathbb{C}^{N}\to E^{1}$ and

Because the bicomplex inner product is componentwise
(Definition~\textup{\ref{def:inner-product}}), the orthogonal complement
satisfies

=\ker(\phi^{1}_{x})^{\perp}\mathbf{e}\oplus\ker(\phi^{2}_{x})^{\perp}\mathbf{e}^{\dagger}.
\end{equation*}
Consequently, the classifying map decomposes as
\begin{equation*}
\iota_{\phi}
=(\iota_{\phi^{1}},\iota_{\phi^{2}})
:B\to\operatorname{Gr}_{m}(\mathbb{C}^{N})\times\operatorname{Gr}_{m}(\mathbb{C}^{N}),
\end{equation*}
which is precisely the idempotent decomposition of\/

<!-- prop-6: body_md -->
\label{prop:cohomology_bcp}
    The integral cohomology of $\mathbb{BCP}^{n-1}$ is given by
    \begin{equation*}
        H^*(\mathbb{BCP}^{n-1};\mathbb{Z}) \cong \mathbb{Z}[x,y] / \langle x^{n}, y^{n} \rangle.
    \end{equation*}
    Consequently,
    \begin{equation*}
        H^*(\mathbb{BCP}^\infty;\mathbb{Z}) \cong \mathbb{Z}[x,y].
    \end{equation*}
<!-- end prop-6 -->
<!-- proof-6: body_md -->
Since $\mathbb{BCP}^{n-1} \cong \mathbb{CP}^{n-1} \times \mathbb{CP}^{n-1}$, the Künneth formula gives
    \begin{equation*}
        H^*(\mathbb{BCP}^{n-1};\mathbb{Z}) \cong H^*(\mathbb{CP}^{n-1};\mathbb{Z}) \otimes H^*(\mathbb{CP}^{n-1};\mathbb{Z}).
    \end{equation*}
    The cohomology of complex projective space is $H^*(\mathbb{CP}^{n-1};\mathbb{Z}) \cong \mathbb{Z}[x]/\langle x^{n}\rangle$, and similarly for $y$. The tensor product yields the stated ring. The vanishing of the $\operatorname{Tor}$-terms follows from the fact that each $H^k(\mathbb{CP}^{n-1};\mathbb{Z})$ is a free $\mathbb{Z}$-module.

<!-- remark-3: body_md -->
\label{rem:projective_abuse}
    We are abusing notation with $\mathbb{P}$. Note that the bicomplex projective bundle differs from the complex projective bundle of the underlying complex vector bundle. Indeed, if $\pi:E \to B$ is viewed as a complex vector bundle of rank $2n$ (with the underlying complex structure) that we shall denote by $\pi:E_\mathbb{C}\to B$, then its complex projective bundle $\mathbb{P} E_{\mathbb{C}} \to B$ has fibers isomorphic to $\mathbb{CP}^{2n-1}$. In contrast, the bicomplex projective bundle $\mathbb{P} E$ has fibers isomorphic to
    \begin{equation*}
        \mathbb{BCP}^{n-1} \cong \mathbb{CP}^{n-1} \times \mathbb{CP}^{n-1},
    \end{equation*}
    which is a product of two projective spaces of dimension $n-1$, not a single projective space of dimension $2n-1$. This will be crucial in our definition of characteristic classes for the bicomplex case.

<!-- defn-10: body_md -->
The \emph{first bicomplex characteristic class} of a line bundle $L \to B$ is defined as
    \begin{equation}
        b_1(L) = f^*(x + y) = f^*(x) + f^*(y) \in H^2(B;\mathbb{Z}),
    \end{equation}
    where $f: B \to \mathbb{BCP}^\infty$ is a classifying map for $L$.
<!-- end defn-10 -->
<!-- remark-4: body_md -->
The class $x + y$ in $H^2(\mathbb{BCP}^\infty;\mathbb{Z})$ is characterized by the commutativity of the factors and the canonical normalization. In terms of complex Chern classes, if $L = L^{1}\mathbf{e} \oplus L^{2}\mathbf{e}^\dagger$ is viewed as a complex vector bundle of rank $2$ (via the underlying complex structure), then the complex first Chern class satisfies
    \begin{align*}
        c_1(L^{1}) + c_{1}(L^{2}) &=
        f^*(p_1^*(a)) + f^*(p_2^*(a))\\
        &=f^*(x) + f^*(y)\\
        &=b_1(L).\\
    \end{align*}
     However, the complex second Chern class $c_2(L)$ is not captured by any bicomplex Chern class.

<!-- thm-2: body_md -->
\label{thm:leray_hirsch}
    Let $\pi:E \to B$ be a bicomplex fiber bundle of rank $n$. The cohomology ring $H^*(\mathbb{P} E;\mathbb{Z})$ is a free $H^*(B;\mathbb{Z})$-module with basis
    \begin{equation}\label{eq:basis}
        \mathcal{B} = \{ \xi_x^i \xi_y^j \mid 0 \le i \le n-1,\ 0 \le j \le n-1 \}.
    \end{equation}
<!-- end thm-2 -->
<!-- proof-7: body_md -->
For the trivial bundle $E = B \times \mathbb{BC}^n$, we have $\mathbb{P}E \cong B \times \mathbb{BCP}^{n-1}$. By the Künneth formula and Proposition~\ref{prop:cohomology_bcp},
    \begin{equation*}
        H^*(\mathbb{P}E;\mathbb{Z}) \cong H^*(B;\mathbb{Z}) \otimes \mathbb{Z}[x,y]/\langle x^n, y^n\rangle,
    \end{equation*}
    which is exactly the statement with basis $\{x^i y^j \mid 0 \le i,j \le n-1\}$.

    For a general bicomplex bundle, the proof follows the classical argument: one covers $B$ by trivializing open sets, uses the Mayer--Vietoris sequence, and compares the cohomology rings. The idempotent decomposition ensures that the argument decomposes componentwise, reducing to the complex case.

<!-- prop-7: body_md -->
\label{prop:splitting}
    Let $\pi:E \to B$ be a bicomplex fiber bundle. Then there exists a continuous map $f: P \to B$ such that
    \begin{enumerate}
        \item $f^*E$ is a direct sum of bicomplex line bundles on $P$;
        \item $f^*: H^*(B;\mathbb{Z}) \to H^*(P;\mathbb{Z})$ is injective.
    \end{enumerate}
    The map $f$ is called a \emph{splitting map}.
<!-- end prop-7 -->
<!-- proof-8: body_md -->
The proof is identical to the complex case. We proceed by induction on the rank $n$. For $n=1$, take $P=B$ and $f=\mathrm{id}$. Assume the result holds for ranks $< n$. Let $\mathbb{P}\pi: \mathbb{P} E \to B$ be the bicomplex projective bundle. Over $\mathbb{P} E$, we have the tautological line subbundle $L(\pi) \subset \pi^*E$. Consider the idempotent decomposition
\begin{equation*}
\pi^*E=\pi^*E^1e\oplus\pi^*E^2e^\dagger.
\end{equation*}
The tautological line bundle decomposes accordingly as
\begin{equation*}
L(\pi)=L^1(\pi)e\oplus L^2(\pi)e^\dagger.
\end{equation*}
Taking the orthogonal complements of $L^1(\pi)$ in $\pi^*E^1$ and of $L^2(\pi)$ in $\pi^*E^2$, we obtain complex subbundles $Q^1$ and $Q^2$. Hence,
\begin{equation*}
Q=Q^1e\oplus Q^2e^\dagger
\end{equation*}
is a bicomplex subbundle, and we obtain the splitting sequence
\begin{equation*}
0\longrightarrow L(\pi)\longrightarrow \pi^*E\longrightarrow Q\longrightarrow 0,
\end{equation*}
Moreover,
\begin{equation*}
\pi^*E=L(\pi)\oplus Q.
\end{equation*}

The bundle $Q$ has rank $n-1$. By the induction hypothesis, there exists a map $g: P' \to \mathbb{P} E$ such that $g^*Q$ splits into line bundles and $g^*$ is injective on cohomology. Let $f = \mathbb{P}\pi \circ g: P' \to B$. Then $f^*E = g^*\pi^*E \cong g^*L(\pi) \oplus g^*Q$, where $g^*L(\pi)$ is a bicomplex line bundle and $g^*Q$ splits by induction. The injectivity of $f^*$ follows from the injectivity of $\mathbb{P}\pi^*$ (Leray--Hirsch) and of $g^*$.

<!-- defn-11: body_md -->
\label{def:bicomplex-chiern-classes}
    For $k = 1, \dots, n$, the \emph{$k$-th bicomplex Chern class} of $E$ is defined as
    \begin{equation*}
        b_k(E) = \sum_{i+j = n-k} \alpha_{i,j} \in H^{2k}(B;\mathbb{Z}).
    \end{equation*}
    We set $b_0=2$, then the \emph{total bicomplex characteristic class} to be
    \begin{equation*}
        b(E) = b_0 + b_1(E) + \cdots + b_n(E) \in H^*(B;\mathbb{Z}).
    \end{equation*}
<!-- end defn-11 -->
<!-- remark-5: body_md -->
One might alternatively try to define the bicomplex Chern classes by imposing the relation in the Leray--Hirsch theorem for $(\xi_x+\xi_y)^n$. Expanding this expression yields
\begin{equation*}
(\xi_x+\xi_y)^n \;=\; \sum_{i=0}^{n}\binom{n}{i}\,\xi_x^{\,n-i}\xi_y^{\,i}.
\end{equation*}
For $1\le i\le n-1$, the cross--terms $\xi_x^{\,n-i}\xi_y^{\,i}$ already belong to the basis 
$\mathcal{B}=\{\xi_x^{p}\xi_y^{q}\mid 0\le p,q\le n-1\}$ 
of $H^{*}(\mathbb{P}E;\mathbb{Z})$ as an $H^{*}(B;\mathbb{Z})$--module. Hence they contribute directly to the coefficients of total fibre--degree $n$ in the relation in Leray--Hirch theorem. The sum of all these coefficients is
\begin{equation*}
\sum_{i=0}^{n}\binom{n}{i}=2^{n},
\end{equation*}
which would force $b_0=2^{n}$, a value that depends on the rank $n$ and is incompatible with the idempotent decomposition. Indeed, for a bundle $E=E^{1}\mathbf{e}\oplus E^{2}\mathbf{e}^\dagger$ we must have
\begin{equation*}
b_0(E)=c_0(E^{1})+c_0(E^{2})=1+1=2.
\end{equation*}
Therefore, the relation $\xi_x^{n}+\xi_y^{n}$ is the correct choice: its degree--$n$ terms have coefficients $1+1=2$, yielding the topologically consistent normalization $b_0=2$ independent of the rank.

Since $E = E^1\mathbf{e}\oplus E^2\mathbf{e}^\dagger$ can be viewed as a complex vector bundle of rank $2n$, its total complex Chern class is
\begin{equation*}
c(E_{\mathbb{C}}) = c(E^1)c(E^2).
\end{equation*}
However, the bicomplex Chern classes $b_k(E)$ differ from the complex ones $c_k(E_{\mathbb{C}})$ for the underlying complex vector bundle $E_\mathbb{C}$. In particular, $b_{2n}(E) = 0$ by definition (since $b_k$ is defined only for $k \le n$), whereas $c_{2n}(E_{\mathbb{C}})$ may be nontrivial.

<!-- thm-3: body_md -->
\label{prop:bicomplex-class-descomposotion}
    If $E=E^1\mathbf{e}\oplus E^2\mathbf{e}^\dagger$, then 
    \begin{equation*}
        b_k(E)=c_k(E^1)+c_k(E^2) \in H^{2k}(B;\mathbb{Z})\quad \text{for $k=0,\cdots,n$},
    \end{equation*}
    where $c_k(E^1)$ and $c_k(E^2)$ are the complex Chern classes.
<!-- end thm-3 -->
<!-- proof-9: body_md -->
Note that $\xi(E^1)$ and $\xi(E^2)$ in Equation \eqref{eq:xi-complex} satisfy the complex Leray--Hirsch theorem:
\begin{gather*}
    \xi(E^1)^n+\sum_{k=1}^n (\mathbb{P}\pi^1)^*(c_k(E^1))\xi(E^1)^{n-k}=0,\\
    \xi(E^2)^n+\sum_{k=1}^n (\mathbb{P}\pi^2)^*(c_k(E^2))\xi(E^2)^{n-k}=0.
\end{gather*}
where $(\mathbb{P}\pi^i)^*:H^*(B;\mathbb{Z})\to H^*(PE^i;\mathbb{Z})$ and $c_k(E^1),c_k(E^2)\in H^{2i}(B;\mathbb{Z})$. We have the following commutative diagram:
\begin{equation*}
    \xymatrix{\mathbb{P} E\ar[r]^{p_1}\ar[d]_{\pi}&\mathbb{P} E^1\ar[d]^{\pi^1}\\ B\ar@{=}[r]&B.}
\end{equation*}
Then, $(p^1)^*\circ(\pi^1)^*=(\pi^1\circ p_1)^*=\pi^*$. The same for $(p^2)^*\circ(\pi^2)^*=(\pi^2\circ p_2)^*=\pi^*$. Substituting in Leray--Hirsch relation in Equation~\eqref{eq:linear-relation}, we get
\begin{align*}
    \xi_x^n + \xi_y^n &=p^*_1(\xi(E^1))^n+p^*_2(\xi(E^2))^n\\
    &=-\sum_{k=1}^n (\mathbb{P}\pi)^*(c_k(E^1))\xi(E^1)^{n-k}-\sum_{k=1}^n (\mathbb{P}\pi)^*(c_k(E^2))\xi(E^2)^{n-k}.
\end{align*}
Since relation in Equation~\eqref{eq:linear-relation} is unique, we have that $b_k=c_k(E^1)+c_k(E^2)\in H^{2k}(B;\mathbb{Z})$.

<!-- example-4: body_md -->
\label{ex:trivial-component}
Let $E^1 \to B$ be any complex vector bundle of rank $n$, and let $E^2 = \underline{\mathbb{C}}^n \to B$ be the trivial complex vector bundle of the same rank. Consider the bicomplex fiber bundle
\begin{equation*}
E = E^1\mathbf{e} \oplus \underline{\mathbb{C}}^n\mathbf{e}^\dagger.
\end{equation*}

Using $\xi_y^n = 0$, we now could compute the bicomplex characteristic classes of $E$ using the Leray-Hirsch Theorem~\ref{thm:leray_hirsch}, since every term containing a factor of $\xi_y^n$ vanishes. But by the previous Proposition~\ref{prop:bicomplex-class-descomposotion} 
\begin{equation*}
    b_k(E)=c_k(E^1)+c_k(\underline{\mathbb{C}})=c_k(E^1)
\end{equation*}

Thus, in this special case where one of the idempotent components is trivial, the bicomplex Chern classes recover exactly the classical Chern classes of the non-trivial complex component. In particular, the theory of complex characteristic classes is embedded in the bicomplex theory as the subfamily of bundles with a trivial $\mathbf{e}^\dagger$-component (or $\mathbf{e}$-component).

This example also illustrates that the non-triviality of the bicomplex classes arises precisely from the interaction between two non-trivial complex components. When one component is trivial, no such interaction occurs, and the theory reduces to the complex case.

<!-- remark-7: body_md -->
\label{rem:whitney-failure}
The Whitney sum formula $b(E \oplus F) = b(E)\cdot b(F)$ does not hold for the bicomplex characteristic classes. To see this, let $B = \mathbb{CP}^2$ and let $L \to \mathbb{CP}^2$ be the tautological complex line bundle with $a = c_1(L)$ and $a^2 \neq 0$. Define two bicomplex line bundles:
\begin{equation*}
L_1 = L\mathbf{e} \oplus \underline{\mathbb{C}}\mathbf{e}^\dagger \quad
\text{and}\quad L_2 = \underline{\mathbb{C}}\mathbf{e} \oplus L\mathbf{e}^\dagger.
\end{equation*}

Their first bicomplex Chern classes are
\begin{equation*}
b_1(L_1) = c_1(L) + c_1(\underline{\mathbb{C}}) = a + 0 = a,
\end{equation*}
\begin{equation*}
b_1(L_2) = c_1(\underline{\mathbb{C}}) + c_1(L) = 0 + a = a.
\end{equation*}
Hence their total classes are
\begin{equation*}
b(L_1) = 2 + a, \quad b(L_2) = 2 + a.
\end{equation*}

Now consider the sum $E = L_1 \oplus L_2$. Its complex components are
\begin{equation*}
E^1 = L_1^1 \oplus L_2^1 = L \oplus \underline{\mathbb{C}}, \quad\text{and}\quad
E^2 = L_1^2 \oplus L_2^2 = \underline{\mathbb{C}} \oplus L.
\end{equation*}
Both components are isomorphic to $L \oplus \underline{\mathbb{C}}$. Over the projective bundle $\mathbb{P} E \cong \mathbb{P} E^1\mathbf{e} \oplus \mathbb{P}E^2\mathbf{e}^\dagger$, the classes $\xi_x$ and $\xi_y$ satisfy the relations:
\begin{equation*}
\xi_x^2 + a\xi_x = 0 \quad \text{and}\quad \xi_y^2 + a\xi_y = 0.
\end{equation*}
Then
\begin{equation*}
\xi_x^2 + \xi_y^2 = -a(\xi_x + \xi_y).
\end{equation*}

Comparing with the general Leray-Hirsch relation
\begin{equation*}
\xi_x^2 + \xi_y^2 + \alpha_{0,0} + \alpha_{1,0}\xi_x + \alpha_{0,1}\xi_y + \alpha_{1,1}\xi_x\xi_y = 0,
\end{equation*}
we obtain
\begin{equation*}
\alpha_{1,0} = a, \quad \alpha_{0,1} = a, \quad \alpha_{1,1} = 0, \quad \alpha_{0,0} = 0.
\end{equation*}

The bicomplex characteristic classes of $E$ are
\begin{equation*}
b_1(E) = \alpha_{1,0} + \alpha_{0,1} = 2a,
\end{equation*}
\begin{equation*}
b_2(E) = \alpha_{0,0} = 0.
\end{equation*}
Therefore,
\begin{equation*}
b(E) = 2 + 2a.
\end{equation*}

However,
\begin{equation*}
b(L_1)\cdot b(L_2) = (2 + a)^2 = 4 + 4a + a^2.
\end{equation*}
Clearly,
\begin{equation*}
b(E) \neq b(L_1)\cdot b(L_2).
\end{equation*}
Thus the Whitney sum formula fails already for a direct sum of two bicomplex line bundles.
<!-- end remark-7 -->
<!-- remark-8: body_md -->
In the complex case the following holds: if $E\cong\bigoplus_{i=1}^{n}L_{i}$ is a sum of complex line bundles and $E$ admits a nowhere vanishing section, then the top Chern class vanishes; equivalently,
\begin{equation*}
c_{1}(L_{1})\smile\cdots\smile c_{1}(L_{n})=0.
\end{equation*}
No analogous result holds for regular sections of bicomplex bundles. Example~\ref{ex:regular_from_zero_divisors} provides a concrete illustration: the bicomplex bundle
\begin{equation*}
E=\bigl((p_{1}^{*}\gamma_{\mathbb{C}})\mathbf{e}\oplus\underline{\mathbb{C}}\,\mathbf{e}^{\dagger}\bigr)
   \oplus
   \bigl(\underline{\mathbb{C}}\,\mathbf{e}\oplus(p_{2}^{*}\gamma_{\mathbb{C}})\mathbf{e}^{\dagger}\bigr)
\end{equation*}
over $B=\mathbb{CP}^1\times\mathbb{CP}^1$ admits the regular section
\begin{equation*}
s=(0\mathbf{e}+1\mathbf{e}^\dagger,1\mathbf{e}+0\mathbf{e}^\dagger),
\end{equation*}
yet
\begin{equation*}
b_{1}(L_1)\smile b_1(L_2)=x\smile y\neq 0\in H^{4}(B;\mathbb{Z}),
\end{equation*}
where $x=p_{1}^{*}(a)$ and $y=p_{2}^{*}(a)$ are the degree-$2$ generators coming from the two factors. Thus the existence of a regular section does not force the vanishing of the product of the first bicomplex Chern classes of the summands.

\label{prop:deformation_retract}

\label{prop:universal-axioms}
The classes $\mathbf{b}_1,\dots,\mathbf{b}_n$ satisfy:
\begin{enumerate}
\item\label{it:nat} \textbf{Naturality.} For every bicomplex bundle $E\to B$ of rank $n$ with classifying map $f:B\to \operatorname{Gr}_n(\mathbb{BC}^\infty)$,
\begin{equation*}
b_k(E)=f^*\mathbf{b}_k.
\end{equation*}
\item\label{it:nor} \textbf{Normalisation.} For the universal bicomplex line bundle over $\mathbb{BCP}^\infty$,
\begin{equation*}
\mathbf{b}=2+x+y\in H^2(\mathbb{BCP}^\infty;\mathbb{Z})\cong\mathbb{Z}[x,y].
\end{equation*}
\item \label{it:inv}\textbf{$\tau$--invariance.} If $\tau$ is the involution map in Equation~\ref{eq:involution-esp}, then $\tau^*\mathbf{b}_k=\mathbf{b}_k$ for all $k$.
\item\label{it:red} \textbf{Reduction to one factor.} With $i_1,i_2:\operatorname{Gr}_n(\mathbb{C}^\infty)\hookrightarrow \operatorname{Gr}_n(\mathbb{BC}^\infty)$ the inclusions of the two factors,
\begin{equation*}
i_1^*\mathbf{b}_k=c_k^{\mathrm{univ}}=i_2^*\mathbf{b}_k.
\end{equation*}
\item\label{it:diag} \textbf{Diagonal restriction.} With $\Delta:\operatorname{Gr}_n(\mathbb{C}^\infty)\to \operatorname{Gr}_n(\mathbb{BC}^\infty)$ the diagonal embedding,
\begin{equation*}
\Delta^*\mathbf{b}_k=2\,c_k^{\mathrm{univ}}.
\end{equation*}
\end{enumerate}

\label{rem:non-mult}
In general,
\begin{equation*}
\operatorname{ch}_{\mathbb{BC}}(E\otimes_{\mathbb{BC}}F)\neq\operatorname{ch}_{\mathbb{BC}}(E)\cdot\operatorname{ch}_{\mathbb{BC}}(F).
\end{equation*}
Indeed, $(E\otimes_{\mathbb{BC}}F)\mathbf{e}\cong E^1\mathbf{e}\otimes_{\mathbb{C}} F
^1\mathbf{e}^\dagger$ and similarly for $\mathbf{e}^\dagger$, so the classical multiplicativity gives
\begin{equation*}
\operatorname{ch}_{\mathbb{BC}}(E\otimes_\mathbb{BC} F)
=\operatorname{ch}(E^1)\operatorname{ch}(F^1)+\operatorname{ch}(E^2)\operatorname{ch}(F^2),
\end{equation*}
whereas the product of the ordinary characters is
\begin{equation*}
\operatorname{ch}_{\mathbb{BC}}(E)\operatorname{ch}_{\mathbb{BC}}(F)
=\operatorname{ch}(E^1)\operatorname{ch}(F^1)
+\operatorname{ch}(E^1)\operatorname{ch}(F^2)
+\operatorname{ch}(E^2)\operatorname{ch}(F^1)
+\operatorname{ch}(E^2)\operatorname{ch}(F^2).
\end{equation*}
The cross terms $\operatorname{ch}(E^1)\operatorname{ch}(F^2)$ and $\operatorname{ch}(E^2)\operatorname{ch}(F^1)$ have no counterpart in the bicomplex tensor product.

<!-- defn-15: body_md -->
\label{def:almost_bicomplex}
Let $M$ be a smooth real manifold. An \emph{almost bicomplex structure} on $M$ is a pair $(I,J)$ of real endomorphisms of the tangent bundle $TM$ such that
\begin{equation*}
I^{2}=J^{2}=-\mathrm{Id},\quad IJ=JI\quad \text{and}\quad I\neq\pm J.
\end{equation*}
The triple $(M,I,J)$ is called an \emph{almost bicomplex manifold}.
<!-- end defn-15 -->
<!-- remark-12: body_md -->
Analogous results to \cite[Appendix~A]{hodge-theory} (to appear in a future article of the second author) allow us to affirm the following: If $(M,I,J)$ is an almost bicomplex manifold, then for each $p \in M$, the pair $(I_p, J_p)$ defines a $\mathbb{BC}$-vector space structure on $T_pM$ by
    \begin{equation*}
        (x + y\,\mathbf{i} + z\,\mathbf{j} + w\,\mathbf{k}) \cdot v
        = x v + y I_p(v) + z J_p(v) + w K_p(v),
    \end{equation*}
    where $K_p = I_p J_p$. Since $\mathbb{BC}$ is a real algebra of dimension $4$, the tangent space $T_pM$ must have real dimension divisible by $4$. Hence any almost bicomplex manifold has rank $4n$ for some integer $n$.

    Moreover, because the structure group of the tangent bundle reduces to $\operatorname{GL}(n,\mathbb{BC})$, which is connected (indeed, $\operatorname{GL}(n,\mathbb{BC}) \cong \operatorname{GL}(n,\mathbb{C}) \times \operatorname{GL}(n,\mathbb{C})$ and each factor is connected), the manifold $M$ is orientable. In fact, the first bicomplex Chern class $b_1(TM) \in H^2(M;\mathbb{Z})$ provides an obstruction to the existence of such a structure, as we shall see below.
<!-- end remark-12 -->
<!-- example-6: body_md -->
\label{ex:product_complex}
    Let $(M_1, J_1)$ and $(M_2, J_2)$ be almost complex manifolds. Then the product $M = M_1 \times M_2$ admits an almost bicomplex structure given by
    \begin{equation*}
        I = J_1 \oplus J_2 \quad \text{and}\quad J = J_1 \oplus (-J_2),
    \end{equation*}
    where we identify $TM = TM_1 \oplus TM_2$. More explicitly, on $M_1 \times M_2$,
    \begin{equation*}
        I(v_1, v_2) = (J_1 v_1, J_2 v_2), \quad\text{and}\quad J(v_1, v_2) = (J_1 v_1, -J_2 v_2).
    \end{equation*}
    A direct calculation shows that $I^2 = J^2 = -Id$ and $IJ = JI$. Thus every product of two almost complex manifolds carries a natural almost bicomplex structure. This structure is not degenerate if and only if the dimensions of $M_1$ and $M_2$ agree.

<!-- thm-5: body_md -->
\label{thm:4d_obstruction}
Let $(M,I,J)$ be a closed almost bicomplex $4$-manifold. Then
\begin{equation*}
3\sigma(M)-2\chi(M)=d^{2}[M]
\end{equation*}
for some class $d\in H^{2}(M;\mathbb{Z})$ satisfying $d\equiv w_{2}(M)\pmod{2}$.
<!-- end thm-5 -->
<!-- proof-16: body_md -->
Since the structure is non-degenerate, $K=IJ$ satisfies $K^{2}=\mathrm{Id}$ and $K\neq\pm\mathrm{Id}$. Hence $K$ has eigenvalues $+1$ and $-1$, and the eigenspaces decompose $TM$ into a sum of complex line bundles $L_{1}\oplus L_{2}$ with respect to the almost complex structure $I$. Write $a_{i}=c_{1}(L_{i})\in H^{2}(M;\mathbb{Z})$. Then
\begin{equation*}
c_{1}(TM,I)=a_{1}+a_{2}\quad \text{and}\quad c_{2}(TM,I)=a_{1}a_{2}.
\end{equation*}
For any almost complex structure on a closed $4$-manifold, the Hirzebruch signature theorem and the Gauss--Bonnet theorem give the characteristic number identities
\begin{equation*}
c_{1}^{2}[M]=2\chi(M)+3\sigma(M)\quad\text{and}\quad c_{2}[M]=\chi(M),
\end{equation*}
which depend only on the oriented smooth topology of $M$ and not on the chosen almost complex structure. Therefore
\begin{equation*}
(a_{1}-a_{2})^{2}[M]=c_{1}^{2}[M]-4c_{2}[M]=(2\chi+3\sigma)-4\chi=3\sigma-2\chi.
\end{equation*}
Setting $d=a_{1}-a_{2}$, we obtain $d^{2}[M]=3\sigma-2\chi$. Moreover, reducing modulo $2$ and using that $c_{1}(TM,I)\equiv w_{2}(M)\pmod{2}$ for any almost complex structure, we get
\begin{equation*}
d=a_{1}-a_{2}\equiv a_{1}+a_{2}=c_{1}(TM,I)\equiv w_{2}(M)\pmod{2}.\qedhere
\end{equation*}

The complex projective plane $\mathbb{C}\mathbb{P}^2$ admits no almost bicomplex structure.

<!-- proof-17: body_md -->
For $\mathbb{C}\mathbb{P}^2$ the characteristic numbers are $\chi(\mathbb{C}\mathbb{P}^{2})=3$ and $\sigma(\mathbb{C}\mathbb{P}^2)=1$. If $\mathbb{C}\mathbb{P}^2$ carried an almost bicomplex structure, Theorem~\ref{thm:4d_obstruction} would provide a class $d\in H^2(\mathbb{C}\mathbb{P}^2;\mathbb{Z})\cong\mathbb{Z} a$ with
\begin{equation*}
d^{2}[\mathbb{C}\mathbb{P}^2]=3\sigma-2\chi=-3.
\end{equation*}
Writing $d=ka$ for some $k\in\mathbb{Z}$ and using $a^{2}[\mathbb{C}\mathbb{P}^2]=1$, we obtain $k^2=-3$, which is impossible in $\mathbb{Z}$.

<!-- remark-13: body_md -->
\label{rem:chern_detection}
     Chern classes alone are not sufficient in general to detect bicomplex structure. For instance, consider a complex vector bundle that is the sum of four complex line bundles $K$, $L$, $M$, and $N$. We can form two different bicomplex bundles of rank $2$:
    \begin{equation*}
        E = (K \oplus L)\mathbf{e} \oplus (M \oplus N)\mathbf{e}^\dagger,
        \qquad
        F = (K \oplus M)\mathbf{e} \oplus (L \oplus N)\mathbf{e}^\dagger.
    \end{equation*}
    As complex vector bundles, $E$ and $F$ are isomorphic (both are $K \oplus L \oplus M \oplus N$). However, as bicomplex bundles, they are generally not isomorphic, and their second bicomplex Chern classes differ in general. Thus bicomplex Chern classes carry strictly more information than ordinary complex Chern classes. Nevertheless, just as in the complex case, they are not complete invariants: two non-isomorphic bicomplex structures on the same underlying complex bundle may still have the same bicomplex Chern classes. Further invariants, such as higher characteristic classes or connections, are needed for a full classification.
