<!-- definition-1: body_md -->
\label{def:embedding} 
            An \emph{embedding} of measure spaces $\iota \colon X \rightarrow Y$ is an injective map such that $A \subseteq X$ is measurable if and only if $\iota(A) \subseteq Y$ is measurable and such that $\mu_X(A) = \mu_Y \left( \iota(A) \right)$. 
            We denote by $\mathbf{Meas}_\mathrm{emb}$ the category whose objects are measure spaces with finite total measure and whose morphisms are embeddings.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:complementary embeddings} 
            Two embeddings of measure spaces 
            \begin{equation*} 
                Y \xrightarrow{~~\alpha~~} X \xleftarrow{~~\beta~~} Z 
            \end{equation*} 
            are called \emph{complementary} if $\alpha(Y) \cap \beta(Z) = \emptyset$ and $\alpha(Y) \cup \beta(Z) = X$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
A \emph{measure-preserving partial map} $(D,p) \colon X \rightarrow Y$ is a measurable subset $D \subseteq X$ together with a measure-preserving map $p \colon D \rightarrow Y$, where $D$ is given the unique measure space structure such that the inclusion $D \hookrightarrow X$ is an embedding. 
            Given two measure-preserving partial maps $(D,p) \colon X \rightarrow Y$ and $(E,q) \colon Y \rightarrow Z$, their composition is defined as $\left( p^{-1}(E),q \circ p \right) \colon X \rightarrow Z$. 
            We denote by $\mathbf{Meas}$ the category whose objects are measure spaces with finite total measure and whose morphisms are measure-preserving partial maps.
<!-- end definition-3 -->
<!-- remark-1: body_md -->
\label{rem:structure of morphisms in Meas} 
            As pointed out in \cite[\S3]{Leinster2023}, any embedding $\iota \colon X \rightarrow Y$ determines a morphism in the category $\mathbf{Meas}$ denoted by $\left( \iota(X),\iota^{-1} \right) \colon Y \rightarrow X$.  
            In particular, this implies that to every measurable subset $A \subseteq X$ we can associate a morphism $(A,\operatorname{id}_A) \colon X \rightarrow A$ in $\mathbf{Meas}$. 
            Furthermore, every measure-preserving map $m \colon X \rightarrow Y$, being a special instance of a measure-preserving partial map, determines a morphism $(X,m) \colon X \rightarrow Y$ in $\mathbf{Meas}$. \\ 
            These elementary facts correspond to the existence of two faithful and injective-on-objects functors (i.e.~embeddings in the categorical sense) as follows:  
            \begin{equation*} 
                \begin{tikzcd} 
                    \mathbf{Meas}_\mathrm{emb} \ar[dr, hook]      & \\ 
                            & \mathbf{Meas}^\mathrm{op} \\ 
                    \mathbf{Meas}_{\mathrm{pres}}^{\mathrm{op}} \ar[ur, hook]      &  
                \end{tikzcd} 
            \end{equation*} 
            where $\mathbf{Meas}_{\mathrm{pres}}$ denotes the category whose objects are measure spaces with finite total measure and whose morphisms are measure-preserving maps. \\ 
            Moreover, any measure-preserving partial map $(D,p) \colon X \rightarrow Y$ can be canonically expressed as the composition of the morphism $(D,\operatorname{id}_D) \colon X \rightarrow D$, associated to the embedding $D \hookrightarrow X$, with the measure-preserving map $p \colon D \rightarrow Y$. \\ 
            The peculiar structure of the morphisms of the category $\mathbf{Meas}^\mathrm{op}$ has two relevant consequences concerning functors and natural transformations: 
            \begin{itemize} 
                \item[(i)] any functor from $\mathbf{Meas}^\mathrm{op}$ to any category $\mathbf{C}$ is completely determined by its action on objects, embeddings and measure-preserving maps; 
                \item[(ii)] any transformation between two functors from $\mathbf{Meas}^\mathrm{op}$ to any category $\mathbf{C}$ is natural if and only if it is natural with respect to both embeddings and measure-preserving maps.  
            \end{itemize}

\label{def:NV_embp} 
            Let $p \in [1,\infty]$.  
            We denote by $\mathbb{NV}_{\mathrm{emb}}^p$ the category whose objects are pairs $(F,c)$, where $F \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}} \rightarrow \mathbf{NVect}_{\mathbb{K}}$ is a bifunctor and $c$ assigns a bounded linear (i.e.~continuous) map $c_{(X,V)} \colon V \rightarrow F(X,V)$ to every object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$.  
            In addition, we require the following axioms to hold: 
            \begin{equation} 
            \tag{\textbf{O}} 
            \label{ax:compatibility with linear contractive maps} 
                F_{(\operatorname{id}_X,T)} \circ c_{(X,V)} = c_{(X,W)} \circ T, 
            \end{equation}             
            for any object $X \in \mathbf{Meas}_\mathrm{emb}$ and any morphism $T \colon V \rightarrow W$ in $\mathbf{NVect}_{\mathbb{K}}$; 
            \begin{equation} 
            \tag{\textbf{I}} 
            \label{ax:compatibility with complementary embeddings in the normed case} 
                c_{(Y,V)}^{(X,V)} + c_{(Z,V)}^{(X,V)} = c_{(X,V)}, 
            \end{equation} 
            for any morphisms $(\alpha,\operatorname{id}_V) \colon (Y,V) \rightarrow (X,V)$ and $(\beta,\operatorname{id}_V) \colon (Z,V) \rightarrow (X,V)$ in $\mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, where $Y \xrightarrow{~\alpha~} X \xleftarrow{~\beta~} Z$ are a couple of complementary embeddings and the notation is explained in Remark \ref{rem:notation for the maps c} below; 
            \begin{equation}  
            \tag{\textbf{III}} 
            \label{ax:boundedness of c} 
                \left\lVert c_{(X,V)}\right\rVert_{\mathcal{L}(V,F(X,V))} \leq \mu_X(X)^{\frac{1}{p}}, \qquad \forall (X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}},  
            \end{equation} 
            where we recall that the operator norm of $c_{(X,V)}$ is defined as 
            \begin{equation*} 
                \left\lVert c_{(X,V)}\right\rVert_{\mathcal{L}(V,F(X,V))} = \inf \Set{\epsilon \geq 0 | \left\lVert c_{(X,V)}(v)\right\rVert_{F(X,V)} \leq \epsilon \left\lVert v\right\rVert_V, \quad \forall v \in V}, 
            \end{equation*} 
            and, in the case $p = \infty$, the expression $\mu_X(X)^{\frac{1}{p}}$ is intended as $0$ if $\mu_X(X) = 0$ and as $1$ otherwise; 
            \begin{equation} 
            \tag{\textbf{IV}} 
            \label{ax:norm inequality wrt complementary embeddings} 
                \left\lVert F_{(\alpha,\operatorname{id}_V)}u + F_{(\beta,\operatorname{id}_V)}w\right\rVert_{F(X,V)} \leq \left( \left\lVert u\right\rVert_{F(Y,V)}^p + \left\lVert w\right\rVert_{F(Z,V)}^p \right)^{\frac{1}{p}}, 
            \end{equation} 
            for any morphisms $(\alpha,\operatorname{id}_V) \colon (Y,V) \rightarrow (X,V)$ and $(\beta,\operatorname{id}_V) \colon (Z,V) \rightarrow (X,V)$ in $\mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, where $Y \xrightarrow{~\alpha~} X \xleftarrow{~\beta~} Z$ are a couple of complementary embeddings, for all $u \in F(Y,V)$ and $w \in F(Z,V)$ and where, in the case $p = \infty$, the right-hand side of the inequality is intended as $\max \Set{\left\lVert u\right\rVert_{F(Y,V)},\left\lVert w\right\rVert_{F(Z,V)}}$. \\ 
            A morphism $\psi \colon (F,c) \rightarrow (G,d)$ in the category $\mathbb{NV}_{\mathrm{emb}}^p$ is a natural transformation $\psi \colon F \Rightarrow G$ such that $\psi_{(X,V)} \circ c_{(X,V)} = d_{(X,V)}$, for all $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$.

\label{ex:simple functions with values in normed vector spaces 1} 
            The fundamental example of an object in the category $\mathbb{NV}_{\mathrm{emb}}^p$ is the pair $(S^p,\sigma)$.  
            Here, $S^p \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}} \rightarrow \mathbf{NVect}_{\mathbb{K}}$ is the bifunctor which associates, to any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, the normed vector space $S^p(X,V)$ of equivalence classes of simple functions on $X$ with values in $V$, under equality almost everywhere.  
            The norm on $S^p(X,V)$ is defined, for $p \in [1,\infty)$ and for any $f \in S^p(X,V)$, as 
            \begin{equation} 
            \label{eq:Lp norm for simple functions with values in normed vector spaces} 
                \left\lVert f\right\rVert_{S^p(X,V)} = \left( \sum_{v \in V} \left\lVert v\right\rVert_V^p \mu_{X}\left( f^{-1}(v) \right) \right)^{\frac{1}{p}}, 
            \end{equation} 
            where, by definition of (equivalence class of) simple functions, only finitely many summands are non-zero and, for $p = \infty$, as 
            \begin{equation} 
            \label{eq:Linfty norm for simple functions with values in normed vector spaces} 
                \left\lVert f\right\rVert_{S^\infty(X,V)} = \inf \Set{\epsilon \geq 0 | \left\lVert f(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X}. 
            \end{equation} 
            The action of the bifunctor $S^p$ on a morphism $(\iota,T) \colon (X,V) \rightarrow (Y,W)$ in the category $\mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$ is defined informally by extension to zero outside the image of the embedding $\iota \colon X \rightarrow Y$ and by post-composition with the linear contractive map $T \colon V \rightarrow W$. More precisely, $S^p_{(\iota,T)} \colon S^p(X,V) \rightarrow S^p(Y,W)$ is defined, for any $f \in S^p(X,V)$, by choosing a representative (denoted, by abuse of notation, by the same symbol $f$) and setting, for almost every $y \in Y$: 
            \begin{equation} 
            \label{eq:action of Sp on emb-morphisms} 
                \begin{split} 
                    S^p_{(\iota,T)}f \colon y \mapsto  
                        \begin{cases} 
                            (T \circ f \circ \iota^{-1})(y) & \text{if} ~ y \in \iota(X), \\ 
                            0 & \text{otherwise,} 
                        \end{cases} 
                \end{split} 
            \end{equation} 
            where the fact that embeddings map measure-zero subsets into measure-zero subsets (see Definition \ref{def:embedding}) guarantees that this definition does not depend on the choice of the representative for $f$. 
            That $S^p_{(\iota,T)}$ is a contractive map can be checked, for $p \in [1,\infty)$ and for any $f \in S^p(X,V)$, as follows: 
            \begin{align} 
                \left\lVert S^p_{(\iota,T)} f\right\rVert_{S^p(Y,W)} &= \left( \sum_{v \in V} \left\lVert T(v)\right\rVert_W^p \mu_Y\left( \iota\left( f^{-1}(v) \right) \right) \right)^{\frac{1}{p}} \notag \\ 
                &= \left( \sum_{v \in V} \left\lVert T(v)\right\rVert_W^p \mu_X\left( f^{-1}(v) \right) \right)^{\frac{1}{p}} \label{eq:using that iota is an embedding} \\ 
                & \leq \left( \sum_{v \in V} \left\lVert v\right\rVert_V^p \mu_X\left( f^{-1}(v) \right) \right)^{\frac{1}{p}} \label{eq:using that T is a contractive map} \\ 
                &= \left\lVert f\right\rVert_{S^p(X,V)}, \notag 
            \end{align} 
            where equation \eqref{eq:using that iota is an embedding} follows from the fact that $\iota \colon X \rightarrow Y$ is an embedding, while inequality \eqref{eq:using that T is a contractive map} follows from the fact that $T \colon V \rightarrow W$ is a contractive map. 
            In the case $p = \infty$, we have: 
            \begin{align} 
                \left\lVert S^\infty_{(\iota,T)} f\right\rVert_{S^\infty(Y,W)} &= \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f \circ \iota^{-1})(y)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ \iota(X)} \notag \\ 
                &= \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f)(x)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ X} \notag \\ 
                &\leq \inf \Set{\epsilon \geq 0 | \left\lVert f(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X} \label{eq:using again that T is a contraction} \\ 
                &= \left\lVert f\right\rVert_{S^\infty(X,V)}, \notag 
            \end{align} 
            where inequality \eqref{eq:using again that T is a contraction} follows from the fact that $T \colon V \rightarrow W$ is a contractive map. \\ 
            For any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, the map $\sigma_{(X,V)} \colon V \rightarrow S^p(X,V)$ is defined by: 
            \begin{equation} 
            \label{eq:definition of sigma} 
                \sigma_{(X,V)}(v) = v \cdot \chi_X, 
            \end{equation} 
            where $\chi_X$ denotes the (equivalence class, under equality almost everywhere, of the) constant function on $X$ with value $1$ (i.e.~the indicator function supported on $X$) and $\cdot$ denotes the natural extension of the scalar multiplication in $V$ to the ring of (equivalence classes, under equality almost everywhere, of) $\mathbb{K}$-valued simple functions on $X$.  
            Linearity of $\sigma_{(X,V)}$ is clear from the definition, while the fact that it is a bounded map is guaranteed, for the case $p \in [1,\infty)$, by the following basic equality:  
            \begin{equation*} 
                \left\lVert \sigma_{(X,V)}(v)\right\rVert_{S^p(X,V)} = \left\lVert v\right\rVert_V \mu_X(X)^{\frac{1}{p}}, \qquad \forall v \in V. 
            \end{equation*} 
            Analogously, for the case $p = \infty$, boundedness of $\sigma_{(X,V)}$ follows directly from: 
            \begin{equation*} 
                \begin{cases} 
                    \left\lVert \sigma_{(X,V)}(v)\right\rVert_{S^\infty(X,V)} = \left\lVert v\right\rVert_V   & \text{if} ~ \mu_X(X) \neq 0, \\ 
                    \left\lVert \sigma_{(X,V)}(v)\right\rVert_{S^\infty(X,V)} = 0   & \text{if} ~ \mu_X(X) = 0, 
                \end{cases} 
            \end{equation*}  
            for all $v \in V$. The last two equations imply, respectively, that  
            \begin{equation*} 
                \left\lVert \sigma_{(X,V)}\right\rVert_{\mathcal{L}\left( V,S^p(X,V) \right)} = \mu_X(X)^{\frac{1}{p}} 
            \end{equation*} 
            and 
            \begin{equation*} 
                \begin{cases} 
                    \left\lVert \sigma_{(X,V)}\right\rVert_{\mathcal{L}\left( V,S^\infty(X,V) \right)} = 1   & \text{if} ~ \mu_X(X) \neq 0, \\ 
                    \left\lVert \sigma_{(X,V)}\right\rVert_{\mathcal{L}\left( V,S^\infty(X,V) \right)} = 0   & \text{if} ~ \mu_X(X) = 0, 
                \end{cases} 
            \end{equation*} 
            which, incidentally, correspond to axiom \ref{ax:boundedness of c} being satisfied. \\ 
            In order to check the validity of axiom \ref{ax:compatibility with linear contractive maps}, it suffices to apply formula \eqref{eq:action of Sp on emb-morphisms} to a morphism of the type $(\operatorname{id}_X,T) \colon (X,V) \rightarrow (X,W)$ in $\mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$. As expected, for all $v \in V$, we obtain: 
            \begin{equation*} 
                \begin{split} 
                    \left( S^p_{(\operatorname{id}_X,T)} \circ \sigma_{(X,V)} \right)(v) &= S^p_{(\operatorname{id}_X,T)}(v \cdot \chi_X) \\ 
                    &= T(v) \cdot \chi_X \\ 
                    &= \left( \sigma_{(X,W)} \circ T \right)(v).  
                \end{split} 
            \end{equation*} 
            Axiom \ref{ax:compatibility with complementary embeddings in the normed case} follows from the elementary fact that the sum of indicator functions supported on disjoint subsets (up to measure-zero sets) gives the indicator function supported on the union of the respective supports (up to measure-zero sets). \\ 
            Finally, we have to check the validity of axiom \ref{ax:norm inequality wrt complementary embeddings}. 
            For $p \in [1,\infty)$ and for any simple functions $f \in S^p{(Y,V)}$ and $g \in S^p{(Z,V)}$, denoting  
            \begin{equation*} 
                h = S^p_{(\alpha,\operatorname{id}_V)}f + S^p_{(\beta,\operatorname{id}_V)}g \in S^p{(X,V)}, 
            \end{equation*} 
            with $Y \xrightarrow{~\alpha~} X \xleftarrow{~\beta~} Z$ any couple of complementary embeddings, we obtain indeed: 
            \begin{align} 
                \left\lVert h\right\rVert_{S^p(X,V)} &= \left( \sum_{v \in V} \left\lVert v\right\rVert_V^p \mu_{X}\left( h^{-1}(v) \right) \right)^{\frac{1}{p}} \notag \\ 
                &= \left( \sum_{v \in V} \left\lVert v\right\rVert_V^p \left( \mu_{X}\left( h^{-1}(v) \cap \alpha(Y) \right) + \mu_{X}\left( h^{-1}(v) \cap \beta(Z) \right) \right) \right)^{\frac{1}{p}} \label{eq:complementary embeddings and additivity of mu} \\ 
                &= \left( \sum_{v \in V} \left\lVert v\right\rVert_V^p \mu_{Y}\left( f^{-1}(v) \right) + \sum_{v \in V} \left\lVert v\right\rVert_V^p \mu_{Z}\left( g^{-1}(v) \right) \right)^{\frac{1}{p}} \label{eq:h coincides with f on Y and with g on Z} \\ 
                &= \left( \left\lVert f\right\rVert_{S^p(Y,V)}^p + \left\lVert g\right\rVert_{S^p(Z,V)}^p \right)^{\frac{1}{p}}, \notag 
            \end{align} 
            where equation \eqref{eq:complementary embeddings and additivity of mu} follows from the fact that $X = \alpha(Y) \cup \beta(Z)$ and $\alpha(Y) \cap \beta(Z) = \emptyset$ (see Definition \ref{def:complementary embeddings}) and from additivity of the measure $\mu_X$, while equation \eqref{eq:h coincides with f on Y and with g on Z} follows from the fact that $h$ coincides, respectively, with $f$ on $\alpha(Y)$ and with $g$ on $\beta(Z)$ and from the fact that embeddings preserve measures.  
            For $p = \infty$ and for any simple functions $f \in S^\infty{(Y,V)}$ and $g \in S^\infty{(Z,V)}$, denoting  
            \begin{equation*} 
                h = S^\infty_{(\alpha,\operatorname{id}_V)}f + S^\infty_{(\beta,\operatorname{id}_V)}g \in S^\infty{(X,V)}, 
            \end{equation*} 
            with $Y \xrightarrow{~\alpha~} X \xleftarrow{~\beta~} Z$ any couple of complementary embeddings, we obtain: 
            \begin{align} 
                \left\lVert h\right\rVert_{S^\infty(X,V)} &= \inf \Set{\epsilon \geq 0 | \left\lVert h(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X} \notag \\ 
                &= \max \big\{ \inf \Set{\epsilon_1 \geq 0 ~|~ \left\lVert h(x)\right\rVert_V \leq \epsilon_1 \quad \text{a.e.~on} ~ \alpha(Y)}, \notag \\ 
                &\qquad \qquad \inf \Set{\epsilon_2 \geq 0 ~|~ \left\lVert h(x)\right\rVert_V \leq \epsilon_2 \quad \text{a.e.~on} ~ \beta(Z)} \big\} \notag \\ 
                &= \max \big\{ \inf \Set{\epsilon_1 \geq 0 ~|~ \left\lVert f(y)\right\rVert_V \leq \epsilon_1 \quad \text{a.e.~on} ~ Y}, \label{eq:h coincides with f on Y and with g on Z and embeddings dont matter} \\ 
                &\qquad \qquad \inf \Set{\epsilon_2 \geq 0 ~|~ \left\lVert g(z)\right\rVert_V \leq \epsilon_2 \quad \text{a.e.~on} ~ Z} \big\} \notag \\ 
                &= \max \Set{\left\lVert f\right\rVert_{S^\infty(Y,V)},\left\lVert g\right\rVert_{S^\infty(Z,V)}}, \notag 
            \end{align} 
            where equation \eqref{eq:h coincides with f on Y and with g on Z and embeddings dont matter} follows from the fact that $h$ coincides, respectively, with $f$ on $\alpha(Y)$ and with $g$ on $\beta(Z)$.

\label{ex:simple functions with values in normed vector spaces 2} 
            The fundamental example of an object in the category $\mathbb{NV}^p$ is denoted, by abuse of notation, with the same symbol $(S^p,\sigma)$ as in Example \ref{ex:simple functions with values in normed vector spaces 1} and the bifunctor $S^p$ is defined in the same way on objects. \\

            The action of $S^p$ on morphisms of the type $\left( m^{\mathrm{op}},T \right) \colon (X,V) \rightarrow (Y,W)$ in $\mathbf{Meas}^\mathrm{op} \times \mathbf{NVect}_{\mathbb{K}}$, where $m \colon Y \rightarrow X$ is a measure-preserving map, is defined informally by pre-composition with the measure-preserving map $m$ and post-composition with the linear contractive map $T$.  
            More explicitly, we set: 
            \begin{equation} 
            \label{eq:action of Sp on measure-preserving maps} 
                \begin{split} 
                    S^p_{\left( m^{\mathrm{op}},T \right)} \colon S^p(X,V) &\rightarrow S^p(Y,W) \\ 
                    f &\mapsto S^p_{\left( m^{\mathrm{op}},T \right)}f = T \circ f \circ m, 
                \end{split} 
            \end{equation} 
            and well-posedness is guaranteed by the fact that $m$ is a measure-preserving map.  
            The fact that $S^p_{\left( m^{\mathrm{op}},T \right)}$ is a contractive map can be checked, for $p \in [1,\infty)$ and for any $f \in S^p(X,V)$, as follows: 
            \begin{align} 
                \left\lVert S^p_{\left( m^{\mathrm{op}},T \right)}f\right\rVert_{S^p(Y,W)} &= \left( \sum_{v \in V} \left\lVert T(v)\right\rVert_W^p \mu_Y\left( m^{-1}\left( f^{-1}(v) \right) \right) \right)^{\frac{1}{p}} \notag \\ 
                &= \left( \sum_{v \in V} \left\lVert T(v)\right\rVert_W^p \mu_X\left( f^{-1}(v) \right) \right)^{\frac{1}{p}} \label{eq:using that m is a measure-preserving map} \\ 
                & \leq \left( \sum_{v \in V} \left\lVert v\right\rVert_V^p \mu_X\left( f^{-1}(v) \right) \right)^{\frac{1}{p}} \label{eq:using contractiveness of T} \\ 
                &= \left\lVert f\right\rVert_{S^p(X,V)}, \notag 
            \end{align} 
            where equation \eqref{eq:using that m is a measure-preserving map} follows from the fact that $m \colon Y \rightarrow X$ is measure-preserving, while inequality \eqref{eq:using contractiveness of T} follows from the fact that $T \colon V \rightarrow W$ is a contractive map. 
            In the case $p = \infty$, we have: 
            \begin{align} 
                \left\lVert S^\infty_{\left( m^{\mathrm{op}},T \right)} f\right\rVert_{S^\infty(Y,W)} &= \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f \circ m)(y)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ Y} \notag \\ 
                &\leq \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f)(x)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ X} \notag \\ 
                &\leq \inf \Set{\epsilon \geq 0 | \left\lVert f(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X} \label{eq:using again contractiveness of T} \\ 
                &= \left\lVert f\right\rVert_{S^\infty(X,V)}, \notag 
            \end{align} 
            where inequality \eqref{eq:using again contractiveness of T} follows from the fact that $T \colon V \rightarrow W$ is a contractive map. \\ 
            For any object $(X,V) \in \mathbf{Meas}^\mathrm{op} \times \mathbf{NVect}_{\mathbb{K}}$, the linear map $\sigma_{(X,V)} \colon V \rightarrow S^p(X,V)$ is defined by formula \eqref{eq:definition of sigma} and the fact that it is a bounded map follows from the same arguments used in Example \ref{ex:simple functions with values in normed vector spaces 1}. \\ 
            Axioms \ref{ax:compatibility with linear contractive maps 1}, \ref{ax:compatibility with complementary embeddings in the normed case 1}, \ref{ax:boundedness of c 1} and \ref{ax:norm inequality wrt complementary embeddings 1} are satisfied for the same reasons, up to notational adaptations regarding the morphisms involved, as in Example \ref{ex:simple functions with values in normed vector spaces 1}. 
            Axiom \ref{ax:compatibility with measure-preserving maps in the normed case} is satisfied as a direct consequence of formula \eqref{eq:action of Sp on measure-preserving maps}.

            \begin{itemize} 
                \item[(i)] $(S^p,\sigma)$ is the initial object of the category $\mathbb{NV}_{\mathrm{emb}}^p$; 
                \item[(ii)] $(S^p,\sigma)$ is the initial object of the category $\mathbb{NV}^p$. 
            \end{itemize}

<!-- proof-5: body_md -->
First, we consider statement (i).  
            Let $(F,c) \in \mathbb{NV}_{\mathrm{emb}}^p$.  
            We want to show that there exists a unique morphism $\eta \colon (S^p,\sigma) \rightarrow (F,c)$ in the category $\mathbb{NV}_{\mathrm{emb}}^p$. 
 
            \emph{Existence.}\hspace{.5em} We observe that, given an object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, for any element $f \in S^p(X,V)$ we can choose a representative (denoted, by abuse of notation, with the same symbol $f$) and write: 
            \begin{equation} 
            \label{eq:expression of simple functions using sigma} 
                f = \sum_{v \in V} \sigma_{\left( f^{-1}(v),V \right)}^{(X,V)}(v), 
            \end{equation} 
            where the sum over $v \in V$ is finite because $f$ is a simple function and because, according to Lemma \ref{lemma:compatibility of c with embeddings in the normed case}, $\sigma_{\left( f^{-1}(v),V \right)}^{(X,V)} = 0$ whenever $f^{-1}(v) \subset X$ is a measure-zero subset. 
            We define, then: 
            \begin{equation} 
            \label{eq:definition of eta} 
                \eta_{(X,V)}(f) = \sum_{v \in V} c_{(f^{-1}(v),V)}^{(X,V)}(v) \in F(X,V).  
            \end{equation} 
            It is routine to check, combining Lemma \ref{lemma:functors and pairwise disjoint embeddings in the normed case} with Lemma \ref{lemma:compatibility of c with embeddings in the normed case}, that the definition of $\eta_{(X,V)}(f)$ does not depend on the choice of a representative for $f$. 
 
            To show that the map $\eta_{(X,V)} \colon S^p(X,V) \rightarrow F(X,V)$ just defined is linear, let $f,g \in S^p(X,V)$. By formula \eqref{eq:definition of eta}, we have: 
            \begin{equation} 
            \label{eq:linearity of the morphism from Sp to F} 
                \eta_{(X,V)}(f + g) = \sum_{v \in V} c_{\left( (f + g)^{-1}(v),V \right)}^{(X,V)}(v).  
            \end{equation} 
            Up to measure-zero subsets of $X$, we can write the following decomposition:  
            \begin{equation*} 
                (f + g)^{-1}(v) = \bigsqcup_{\substack{w,z \in V \\ w + z = v}} f^{-1}(w) \cap g^{-1}(z). 
            \end{equation*} 
            Noting that the family of pairwise disjoint measurable subsets $\left( f^{-1}(w) \cap g^{-1}(z) \right)$, indexed by $w,z \in V$ with $w + z = v$, is finite due to $f$ and $g$ being simple functions, we can apply Lemma \ref{lemma:functors and pairwise disjoint embeddings in the normed case} and obtain: 
            \begin{equation*} 
                c_{\left( (f + g)^{-1}(v),V \right)}^{(X,V)} = \sum_{\substack{w,z \in V \\ w + z = v}} c_{\left( f^{-1}(w) \cap g^{-1}(z),V \right)}^{(X,V)}. 
            \end{equation*} 
            Substituting the last expression into formula \eqref{eq:linearity of the morphism from Sp to F}, we get: 
            \begin{align} 
                \eta_{(X,V)}(f &+ g) = \sum_{v \in V} \sum_{\substack{w,z\in V \\ w + z = v}} c_{\left( f^{-1}(w)\cap g^{-1}(z),V \right)}^{(X,V)}(w + z) \notag \\ 
                &= \sum_{v \in V} \sum_{\substack{w,z \in V \\ w + z = v}} c_{\left( f^{-1}(w) \cap g^{-1}(z),V \right)}^{(X,V)}(w) + \sum_{v \in V} \sum_{\substack{w,z \in V \\ w + z = v}} c_{\left( f^{-1}(w)\cap g^{-1}(z),V \right)}^{(X,V)}(z) \label{eq:use linearity of the map c} \\ 
                &= \sum_{w \in V} \sum_{v \in V} c_{\left( f^{-1}(w) \cap g^{-1}(v - w),V \right)}^{(X,V)}(w) + \sum_{z \in V} \sum_{v \in V} c_{\left( f^{-1}(v-z) \cap g^{-1}(z),V \right)}^{(X,V)}(z) \label{eq:rename the indices in the summations} \\ 
                &= \sum_{w \in V} c_{\left( f^{-1}(w),V \right)}^{(X,V)}(w) + \sum_{z \in V} c_{\left( g^{-1}(z),V \right)}^{(X,V)}(z) \label{eq:sum up one of the summations} \\ 
                &= \eta_{(X,V)}(f) + \eta_{(X,V)}(g), \notag 
            \end{align} 
            where equation \eqref{eq:use linearity of the map c} follows from linearity of the map $c_{\left( f^{-1}(w)\cap g^{-1}(z),V \right)}^{(X,V)}$, equation \eqref{eq:rename the indices in the summations} is simply a rearrangement of the summation indices and equation \eqref{eq:sum up one of the summations} is a consequence of Lemma \ref{lemma:functors and pairwise disjoint embeddings in the normed case} applied to the following decompositions 
            \begin{equation*} 
                f^{-1}(w) = \bigsqcup_{v \in V} f^{-1}(w) \cap g^{-1}(v - w), \quad \qquad g^{-1}(z) = \bigsqcup_{v \in V} f^{-1}(v-z)\cap g^{-1}(z), 
            \end{equation*} 
            which hold up to measure-zero subsets of $X$. 
            Linearity of $\eta_{(X,V)}$ with respect to scalar multiplication follows from the basic observation that  
            \begin{equation*} 
                \lambda f = \sum_{v \in V} \sigma_{\left( (\lambda f)^{-1}(v),V \right)}^{(X,V)}(v) = \sum_{v \in V} \sigma_{\left( f^{-1}(v),V \right)}^{(X,V)}(\lambda v), 
            \end{equation*} 
            for any $\lambda \in \mathbb{K}$, and from linearity of the maps $c_{\left( f^{-1}(v),V \right)}^{(X,V)}$. \\ 
            In addition, we need to show that, for any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, the map $\eta_{(X,V)} \colon S^p(X,V) \rightarrow F(X,V)$ is contractive. 
            Indeed, for $p \in [1,\infty)$ and for any $f \in S^p(X,V)$, this can be seen as follows: 
            \begin{align} 
                \left\lVert \eta_{(X,V)}(f)\right\rVert_{F(X,V)} &= \left\lVert \sum_{v \in V} c_{\left( f^{-1}(v),V \right)}^{(X,V)}(v)\right\rVert_{F(X,V)} \notag \\ 
                &\leq \left( \sum_{v \in V} \left\lVert c_{\left( f^{-1}(v),V \right)}(v)\right\rVert^p_{F\left( f^{-1}(v),V \right)} \right)^{\frac{1}{p}} \label{eq:apply norm inequality wrt complementary embeddings in the normed case} \\ 
                &\leq \left( \sum_{v \in V} \left\lVert v\right\rVert_V^p \mu_{f^{-1}(v)}\left( f^{-1}(v) \right) \right)^{\frac{1}{p}} \label{eq:apply axiom III} \\ 
                &= \left( \sum_{v \in V} \left\lVert v\right\rVert^p \mu_X\left( f^{-1}(v) \right) \right)^{\frac{1}{p}} \label{eq:apply that embeddings preserve measure} \\ 
                &= \left\lVert f\right\rVert_{S^p(X,V)}, \notag 
            \end{align} 
            where the first equation corresponds to the definition of $\eta_{(X,V)}(f)$ according to formula \eqref{eq:definition of eta}, inequality \eqref{eq:apply norm inequality wrt complementary embeddings in the normed case} follows by repeatedly applying axiom \ref{ax:norm inequality wrt complementary embeddings}, inequality \eqref{eq:apply axiom III} follows from axiom \ref{ax:boundedness of c} and equality \eqref{eq:apply that embeddings preserve measure} is a consequence of the fact that the embedding $f^{-1}(v) \hookrightarrow X$ preserves measures, namely, $\mu_{f^{-1}(v)}\left( f^{-1}(v) \right) = \mu_X\left( f^{-1}(v) \right)$. \\ 
            For $p = \infty$, let us first assume that $\mu_X(X) \neq 0$. 
            We then have: 
            \begin{align} 
                \left\lVert \eta_{(X,V)}(f)\right\rVert_{F(X,V)} &= \left\lVert \sum_{v \in V} c_{\left( f^{-1}(v),V \right)}^{(X,V)}(v)\right\rVert_{F(X,V)} \notag \\ 
                &\leq \max_{v \in V}\Set{\left\lVert c_{\left( f^{-1}(v),V \right)}(v)\right\rVert_{F\left( f^{-1}(v),V \right)}} \label{eq:apply axiom IV repeatedly} \\ 
                &\leq \max_{v \in V}\Set{\left\lVert v\right\rVert_V \left\lVert c_{\left( f^{-1}(v),V \right)}\right\rVert_{\mathcal{L}\left( V,F\left( f^{-1}(v),V \right) \right)}} \label{eq:use the operator norm for c} \\ 
                &\leq \max_{\substack{v \in V \\ \mu_X\left( f^{-1}(v) \right) \neq 0}} \Set{\left\lVert v\right\rVert_V} \label{eq:use axiom III} \\ 
                &= \left\lVert f\right\rVert_{S^\infty(X,V)}, \notag 
            \end{align} 
            where inequality \eqref{eq:apply axiom IV repeatedly} follows by repeatedly applying axiom \ref{ax:norm inequality wrt complementary embeddings}, inequality \eqref{eq:use the operator norm for c} follows by using the operator norms of the maps $c_{\left( f^{-1}(v),V \right)}$ and, finally, inequality \eqref{eq:use axiom III} follows from axiom \ref{ax:boundedness of c}. 
            In the case $\mu_X(X) = 0$, we directly obtain: 
            \begin{align} 
                \left\lVert \eta_{(X,V)}(f)\right\rVert_{F(X,V)} &= \left\lVert \sum_{v \in V} c_{\left( f^{-1}(v),V \right)}^{(X,V)}(v)\right\rVert_{F(X,V)} \notag \\ 
                &\leq \sum_{v \in V} \left\lVert v\right\rVert_V \left\lVert c_{\left( f^{-1}(v),V \right)}^{(X,V)}\right\rVert_{\mathcal{L}(V,F(X,V))} \notag \\ 
                &= 0 \notag \\ 
                &= \left\lVert f\right\rVert_{S^\infty(X,V)}, \notag 
            \end{align}  
            where $\left\lVert c_{\left( f^{-1}(v),V \right)}^{(X,V)}\right\rVert_{\mathcal{L}(V,F(X,V))} = 0$, for all $v \in V$, due to Lemma \ref{lemma:compatibility of c with embeddings in the normed case}. 
 
            Next, we show that $\eta$ defines a natural transformation from the bifunctor $S^p$ to the bifunctor $F$.  
            In other words, we have to show that, for any morphism $(\iota,T) \colon (X,V) \rightarrow (Y,W)$ in $\mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, the diagram 
            \begin{equation*} 
                \begin{tikzcd} 
                    S^p(X,V) \ar[rr, "S^p_{(\iota,T)}"] \ar[dd, "\eta_{(X,V)}"']  &   & S^p(Y,W) \ar[dd, "\eta_{(Y,W)}"] \\ 
                        &   & \\ 
                    F(X,V) \ar[rr, "F_{(\iota,T)}"]   &   & F(Y,W) 
                \end{tikzcd} 
            \end{equation*} 
            commutes in the category $\mathbf{NVect}_{\mathbb{K}}$. 
            By formula \eqref{eq:expression of simple functions using sigma}, it suffices to check commutativity when composing the arrows of the diagram with the maps $\sigma_{(B,V)}^{(X,V)}$, where $B \subseteq X$ is any measurable subset.  
            On the one hand, we have: 
            \begin{align} 
                F_{(\iota,T)} \circ \eta_{(X,V)} \circ \sigma_{(B,V)}^{(X,V)} &= F_{(\iota,T)} \circ c_{(B,V)}^{(X,V)} \notag \\ 
                &= F_{(\iota,\operatorname{id}_W)} \circ F_{(\operatorname{id}_X,T)} \circ c_{(B,V)}^{(X,V)} \label{eq:separate the two components of the morphism} \\ 
                &= F_{(\iota,\operatorname{id}_W)} \circ c_{(B,V)}^{(X,V)} \circ T \label{eq:apply axiom O} \\ 
                &= c_{(B,V)}^{(Y,W)} \circ T, \notag 
            \end{align} 
            where the first equality follows from the definition of $\eta_{(X,V)}$, see formula \eqref{eq:definition of eta}, in equality \eqref{eq:separate the two components of the morphism} we used Remark \ref{rem:morphisms and functors in product categories in the normed case}, equation \eqref{eq:apply axiom O} follows from axiom \ref{ax:compatibility with linear contractive maps} and the last equality is just a matter of notation. 
            On the other hand, using the same arguments, we obtain: 
            \begin{align*} 
                \eta_{(Y,W)} \circ S^p_{(\iota,T)} \circ \sigma_{(B,V)}^{(X,V)} &= \eta_{(Y,W)} \circ S^p_{(\iota,\operatorname{id}_W)} \circ S^p_{(\operatorname{id}_X,T)} \circ \sigma_{(B,V)}^{(X,V)} \\ 
                &= \eta_{(Y,W)} \circ S^p_{(\iota,\operatorname{id}_W)} \circ \sigma_{(B,W)}^{(X,W)} \circ T \\ 
                &= \eta_{(Y,W)} \circ \sigma_{(B,W)}^{(Y,W)} \circ T \\ 
                &= c_{(B,W)}^{(Y,W)} \circ T. 
            \end{align*} 
            We have thus shown that $\eta$ defines a natural transformation $S^p \Rightarrow F$ and, moreover, for all $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, we have that $\eta_{(X,V)} \circ \sigma_{(X,V)} = c_{(X,V)}$ directly from formula \eqref{eq:definition of eta}.  
            Hence $\eta \colon (S^p,\sigma) \rightarrow (F,c)$ is a morphism in the category $\mathbb{NV}_{\mathrm{emb}}^p$. 
 
            \emph{Uniqueness.}\hspace{.5em} Assume the existence of a morphism $\eta \colon (S^p,\sigma) \rightarrow (F,c)$ in the category $\mathbb{NV}_{\mathrm{emb}}^p$.  
            For any couple of objects $(X,V),(Y,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, where $X \subseteq Y$ is any measurable subset, naturality of $\eta$ with respect to the morphism $(\iota,\operatorname{id}_V) \colon (X,V) \rightarrow (Y,V)$, with $\iota \colon X \rightarrow Y$ denoting the embedding of $X$ into $Y$, gives the following commutative diagram in $\mathbf{NVect}_{\mathbb{K}}$ 
            \begin{equation*} 
                \begin{tikzcd} 
                    S^p(X,V) \ar[rr, "S^p_{(\iota,\operatorname{id}_V)}"] \ar[dd, "\eta_{(X,V)}"']  &   & S^p(Y,V) \ar[dd, "\eta_{(Y,V)}"] \\ 
                        &   & \\ 
                    F(X,V) \ar[rr, "F_{(\iota,\operatorname{id}_V)}"]   &   & F(Y,V) 
                \end{tikzcd} 
            \end{equation*} 
            Composing the arrows of this diagram with the map $\sigma_{(X,V)} \colon V \rightarrow S^p(X,V)$ and exploiting the commutativity, we obtain: 
            \begin{align} 
                \eta_{(Y,V)} \circ \sigma_{(X,V)}^{(Y,V)} &= \eta_{(Y,V)} \circ S^p_{(\iota,\operatorname{id}_V)} \circ \sigma_{(X,V)} \notag \\ 
                &= F_{(\iota,\operatorname{id}_V)} \circ \eta_{(X,V)} \circ \sigma_{(X,V)} \notag \\ 
                &= F_{(\iota,\operatorname{id}_V)} \circ c_{(X,V)} \label{eq:apply the definition of morphism in NVembp} \\ 
                &= c_{(X,V)}^{(Y,V)}. \notag 
            \end{align} 
            This implies that $\eta_{(Y,V)}$ is uniquely determined on the image of the maps $\sigma_{(X,V)}^{(Y,V)}$, for any measurable subset $X \subseteq Y$.  
            Hence, by formula \eqref{eq:expression of simple functions using sigma} and linearity, $\eta_{(Y,V)}$ is uniquely determined on all of $S^p(Y,V)$.  
            This completes the proof of statement (i). 
 
            To prove statement (ii), let $(F,c) \in \mathbb{NV}^p$.  
            According to Remark \ref{rem:forgetful functor from NVp to NVembp}, we can regard $(F,c)$ as an object in the category $\mathbb{NV}_{\mathrm{emb}}^p$ and, from statement (i), we know that there exists a unique morphism $\eta \colon (S^p,\sigma) \rightarrow (F,c)$ in $\mathbb{NV}_{\mathrm{emb}}^p$.  
            To conclude, it suffices to show that $\eta$ naturally defines a morphism in the category $\mathbb{NV}^p$. \\  
            This amounts to showing that, for any morphism $\left( m^{\mathrm{op}},T \right) \colon (X,V) \rightarrow (Y,W) $ in $\mathbf{Meas}^\mathrm{op} \times \mathbf{NVect}_{\mathbb{K}}$, where $m \colon Y \rightarrow X$ is a measure-preserving map, the diagram 
            \begin{equation*} 
                \begin{tikzcd} 
                    S^p(X,V) \ar[rr, "S^p_{\left( m^{\mathrm{op}},T \right)}"] \ar[dd, "\eta_{(X,V)}"']  &   & S^p(Y,W) \ar[dd, "\eta_{(Y,W)}"] \\ 
                        &   & \\ 
                    F(X,V) \ar[rr, "F_{\left( m^{\mathrm{op}},T \right)}"]    &   & F(Y,W) 
                \end{tikzcd} 
            \end{equation*} 
            commutes in the category $\mathbf{NVect}_{\mathbb{K}}$. 
            By formula \eqref{eq:expression of simple functions using sigma}, it suffices to check commutativity when composing the arrows of the diagram with the maps $\sigma_{(B,V)}^{(X,V)}$, where $B \subseteq X$ is any measurable subset.  
            Using Lemma \ref{lemma:functors and measure-preserving maps in the normed case}, we have, on the one hand: 
            \begin{align*} 
                \eta_{(Y,W)} \circ S^p_{\left( m^{\mathrm{op}},T \right) }\circ \sigma_{(B,V)}^{(X,V)} &= \eta_{(Y,W)} \circ \sigma_{\left( m^{-1}(B),W \right)}^{(Y,W)} \circ T \\ 
                &= c_{\left( m^{-1}(B),W \right)}^{(Y,W)} \circ T. 
            \end{align*} 
            Using again Lemma \ref{lemma:functors and measure-preserving maps in the normed case}, we obtain, on the other hand: 
            \begin{align*} 
                F_{\left( m^{\mathrm{op}},T \right)} \circ \eta_{(X,V)}\circ \sigma_{(B,V)}^{(X,V)} &= F_{\left( m^{\mathrm{op}},T \right)} \circ c_{(B,V)}^{(X,V)} \\ 
                &= c_{\left( m^{-1}(B),W \right)}^{(Y,W)} \circ T. 
            \end{align*} 
            This completes the proof of the theorem.

\label{rem:relating Ban and NVect} 
            We point out some basic facts relating the category $\mathbf{BAN}_{\mathbb{K}}$ to the category $\mathbf{NVECT}_{\mathbb{K}}$, which will be useful in the sequel. 
            First, there is an obvious faithful and injective-on-objects functor $\mathcal{I} \colon \mathbf{BAN}_{\mathbb{K}} \hookrightarrow \mathbf{NVECT}_{\mathbb{K}}$ which forgets the completeness of Banach spaces. 
            The restriction of this functor to the subcategory $\mathbf{Ban}_{\mathbb{K}} \subset \mathbf{BAN}_{\mathbb{K}}$ naturally yields a faithful and injective-on-objects functor denoted, by abuse of notation, with the same symbol $\mathcal{I} \colon \mathbf{Ban}_{\mathbb{K}} \hookrightarrow \mathbf{NVect}_{\mathbb{K}}$. \\ 
            The functor $\mathcal{I}$ has a left adjoint given by the functor $\overline{\cdot} \colon \mathbf{NVECT}_{\mathbb{K}} \rightarrow \mathbf{BAN}_{\mathbb{K}}$, which acts on normed vector spaces by taking their completion and assigns to every bounded linear map between normed vector spaces the extension by continuity to the completion of its domain and codomain. 
            Since extension by continuity preserves the operator norm of bounded linear maps, the restriction of the completion functor to the subcategory $\mathbf{NVect}_{\mathbb{K}} \subset \mathbf{NVECT}_{\mathbb{K}}$ yields a functor denoted, by abuse of notation, with the same symbol $\overline{\cdot} \colon \mathbf{NVect}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$, which is the left adjoint to the functor $\mathcal{I} \colon \mathbf{Ban}_{\mathbb{K}} \hookrightarrow \mathbf{NVect}_{\mathbb{K}}$. \\ 
            Secondly, using the functor $\mathcal{I}$, we can define two faithful and injective-on-objects bifunctors which, by abuse of notation, we denote with the same symbol:  
            \begin{equation*} 
                \begin{split} 
                    (\operatorname{id},\mathcal{I}) &\colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{BAN}_{\mathbb{K}} \hookrightarrow \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVECT}_{\mathbb{K}}, \\ 
                    (\operatorname{id},\mathcal{I}) &\colon \mathbf{Meas}^\mathrm{op} \times \mathbf{BAN}_{\mathbb{K}} \hookrightarrow \mathbf{Meas}^\mathrm{op} \times \mathbf{NVECT}_{\mathbb{K}}. 
                \end{split} 
            \end{equation*} 
            Clearly, these two bifunctors admit left adjoints given, respectively, by: 
            \begin{equation*} 
                \begin{split} 
                    (\operatorname{id},\overline{\cdot}) &\colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVECT}_{\mathbb{K}} \rightarrow \mathbf{Meas}_\mathrm{emb} \times \mathbf{BAN}_{\mathbb{K}}, \\ 
                    (\operatorname{id},\overline{\cdot}) &\colon \mathbf{Meas}^\mathrm{op} \times \mathbf{NVECT}_{\mathbb{K}} \rightarrow \mathbf{Meas}^\mathrm{op} \times \mathbf{BAN}_{\mathbb{K}}. 
                \end{split} 
            \end{equation*} 
            Finally, all the bifunctors above can be restricted, respectively, to the subcategories $\mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, $\mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$, $\mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$ and $\mathbf{Meas}^\mathrm{op} \times \mathbf{NVect}_{\mathbb{K}}$ to give corresponding pairs of adjoint bifunctors.

<!-- example-3: body_md -->
\label{ex:Lpemb functor} 
            The fundamental example of an object in the category $\mathbb{B}_{\mathrm{emb}}^p$ is the pair $(L^p,\sigma)$. 
            Here, $L^p \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$ is the bifunctor which associates, to any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, the Banach space $L^p(X,V)$ of equivalence classes of $L^p$ Bochner integrable functions on $X$ with values in $V$, under equality almost everywhere.  
            We recall that the norm on $L^p(X,V)$ is defined, for any $p \in [1,\infty)$ and for any $f \in L^p(X,V)$, as 
            \begin{equation} 
            \label{eq:Lp norm} 
                \left\lVert f\right\rVert_{L^p(X,V)} = \left( \int_X \left\lVert f(x)\right\rVert_V^p d\mu_{X}(x) \right)^{\frac{1}{p}}, 
            \end{equation} 
            while, for $p = \infty$ and for any $f \in L^p(X,V)$, it is defined as 
            \begin{equation} 
            \label{eq:Linfty norm} 
                \left\lVert f\right\rVert_{L^\infty(X,V)} = \inf \Set{\epsilon \geq 0 | \left\lVert f(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X}. 
            \end{equation} 
            The action of the bifunctor $L^p$ on a morphism $(\iota,T) \colon (X,V) \rightarrow (Y,W)$ in the category $\mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$ is defined, in an analogous way as in Example \ref{ex:simple functions with values in normed vector spaces 1}, by formula \eqref{eq:action of Sp on emb-morphisms}. 
            That $L^p_{(\iota,T)} \colon L^p(X,V) \rightarrow L^p(Y,W)$ is a contractive map can be checked, for any $p \in [1,\infty)$ and for any $f \in L^p(X,V)$, as follows: 
            \begin{align} 
                \left\lVert L^p_{(\iota,T)} f\right\rVert_{L^p(Y,W)} &= \left( \int_{\iota(X)} \left\lVert (T \circ f \circ \iota^{-1})(y)\right\rVert_W^p d\mu_{Y}(y) \right)^{\frac{1}{p}} \notag \\ 
                &= \left( \int_{X} \left\lVert (T \circ f)(x)\right\rVert_W^p d\mu_{X}(x) \right)^{\frac{1}{p}} \label{eq:embeddings preserve measures} \\ 
                &\leq \left( \int_{X} \left\lVert f(x)\right\rVert_V^p d\mu_{X}(x) \right)^{\frac{1}{p}} \label{eq:T is a contractive map} \\ 
                &= \left\lVert f\right\rVert_{L^p(X,V)}, \notag 
            \end{align} 
            where equation \eqref{eq:embeddings preserve measures} follows from the fact that embeddings preserve measures, while inequality \eqref{eq:T is a contractive map} follows from the fact that $T \colon V \rightarrow W$ is a contractive map. 
            In the case $p = \infty$, for any $f \in L^\infty(X,V)$, we have: 
            \begin{align} 
                \left\lVert L^\infty_{(\iota,T)} f\right\rVert_{L^\infty(Y,W)} &= \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f \circ \iota^{-1})(y)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ \iota(X)} \notag \\ 
                &= \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f)(x)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ X} \notag \\ 
                &\leq \inf \Set{\epsilon \geq 0 | \left\lVert f(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X} \label{eq:use again that T is a contractive map} \\ 
                &= \left\lVert f\right\rVert_{L^\infty(X,V)}, \notag 
            \end{align} 
            where, again, inequality \eqref{eq:use again that T is a contractive map} follows from the fact that $T$ is a contractive map. \\ 
            For any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, the linear map $\sigma_{(X,V)} \colon V \rightarrow L^p(X,V)$ is defined by formula \eqref{eq:definition of sigma}.  
            That $\sigma_{(X,V)}$ is a bounded linear map follows from the same arguments used in Example \ref{ex:simple functions with values in normed vector spaces 1}. \\ 
            Axioms \eqref{ax:compatibility with linear contractive maps}, \eqref{ax:compatibility with complementary embeddings in the normed case} and \eqref{ax:boundedness of c}, which involve only the assignment $\sigma$, are satisfied for the same reasons as in Example \ref{ex:simple functions with values in normed vector spaces 1}. 
            Regarding the validity of axiom \eqref{ax:norm inequality wrt complementary embeddings}, for any $p \in [1,\infty)$ and for any elements $f \in L^p{(Y,V)}$ and $g \in L^p{(Z,V)}$, denoting  
            \begin{equation*} 
                h = L^p_{(\alpha,\operatorname{id}_V)}f + L^p_{(\beta,\operatorname{id}_V)}g \in L^p{(X,V)}, 
            \end{equation*} 
            with $Y \xrightarrow{~\alpha~} X \xleftarrow{~\beta~} Z$ any couple of complementary embeddings, we obtain indeed: 
            \begin{align} 
                \left\lVert h\right\rVert_{L^p(X,V)} &= \left( \int_{X} \left\lVert h(x)\right\rVert_V^p d\mu_{X}(x) \right)^{\frac{1}{p}} \notag \\ 
                &= \left( \int_{\alpha(Y)} \left\lVert h(x)\right\rVert_V^p d\mu_{X}(x) + \int_{\beta(Z)} \left\lVert h(x)\right\rVert_V^p d\mu_{X}(x) \right)^{\frac{1}{p}} \label{eq:use additivity of the Bochner integral} \\ 
                &= \left( \int_{Y} \left\lVert f(y)\right\rVert_V^p d\mu_{Y}(y) + \int_{Z} \left\lVert g(z)\right\rVert_V^p d\mu_{Z}(z) \right)^{\frac{1}{p}} \label{eq:use that h corresponds to f on Y and to g on Z and that embeddings preserve measures} \\ 
                &= \left( \left\lVert f\right\rVert_{L^p(Y,V)}^p + \left\lVert g\right\rVert_{L^p(Z,V)}^p \right)^{\frac{1}{p}}, \notag 
            \end{align} 
            where equation \eqref{eq:use additivity of the Bochner integral} follows from the fact that $X = \alpha(Y) \cup \beta(Z)$ and $\alpha(Y) \cap \beta(Z) = \emptyset$ (see Definition \ref{def:complementary embeddings}) and from additivity of the Lebesgue integral, while equation \eqref{eq:use that h corresponds to f on Y and to g on Z and that embeddings preserve measures} follows from the fact that $h$ coincides, respectively, with $f$ on $\alpha(Y)$ and with $g$ on $\beta(Z)$ and from the fact that embeddings preserve measures. 
            For $p = \infty$ and for any elements $f \in L^\infty{(Y,V)}$ and $g \in L^\infty{(Z,V)}$, denoting  
            \begin{equation*} 
                h = L^\infty_{(\alpha,\operatorname{id}_V)}f + L^\infty_{(\beta,\operatorname{id}_V)}g \in L^\infty{(X,V)}, 
            \end{equation*} 
            with $Y \xrightarrow{~\alpha~} X \xleftarrow{~\beta~} Z$ any couple of complementary embeddings, we obtain: 
            \begin{align} 
                \left\lVert h\right\rVert_{S^\infty(X,V)} &= \inf \Set{\epsilon \geq 0 | \left\lVert h(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X} \notag \\ 
                &= \max \big\{ \inf \Set{\epsilon_1 \geq 0 ~|~ \left\lVert h(x)\right\rVert_V \leq \epsilon_1 \quad \text{a.e.~on} ~ \alpha(Y)}, \notag \\ 
                &\qquad \qquad \inf \Set{\epsilon_2 \geq 0 ~|~ \left\lVert h(x)\right\rVert_V \leq \epsilon_2 \quad \text{a.e.~on} ~ \beta(Z)} \big\} \notag \\ 
                &= \max \big\{ \inf \Set{\epsilon_1 \geq 0 ~|~ \left\lVert f(y)\right\rVert_V \leq \epsilon_1 \quad \text{a.e.~on} ~ Y}, \label{eq:use again that h corresponds to f on Y and to g on Z and that embeddings preserve measures} \\ 
                &\qquad \qquad \inf \Set{\epsilon_2 \geq 0 ~|~ \left\lVert g(z)\right\rVert_V \leq \epsilon_2 \quad \text{a.e.~on} ~ Z} \big\} \notag \\ 
                &= \max \Set{\left\lVert f\right\rVert_{S^\infty(Y,V)},\left\lVert g\right\rVert_{S^\infty(Z,V)}}, \notag 
            \end{align} 
            where equation \eqref{eq:use again that h corresponds to f on Y and to g on Z and that embeddings preserve measures} follows from the fact that $h$ coincides, respectively, with $f$ on $\alpha(Y)$ and with $g$ on $\beta(Z)$.

\label{def:Bp}

            We denote by $\mathbb{B}^p$ the category whose objects are pairs $(F,c)$, where $F \colon \mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$ is a bifunctor and $c$ denotes the assignment of a bounded linear (i.e.~continuous) map $c_{(X,V)} \colon V \rightarrow F(X,V)$ to every object $(X,V) \in \mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$.

\label{ex:Lp functor} 
            The fundamental example of an object in the category $\mathbb{B}^p$ is the pair $(L^p,\sigma)$. 
            Here, $L^p \colon \mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$ is the bifunctor which associates, to any object $(X,V) \in \mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$, the Banach space $L^p(X,V)$ of equivalence classes of $L^p$ Bochner integrable functions on $X$ with values in $V$, under equality almost everywhere.  
            The norm on the space $L^p(X,V)$ is defined, for any $p \in [1,\infty)$, by formula \eqref{eq:Lp norm} and, for $p = \infty$, by formula \eqref{eq:Linfty norm}. \\ 
            The action of the bifunctor $L^p$ on a morphism $\left( \left( \iota(X),\iota^{-1} \right)^{\mathrm{op}},T \right) \colon (X,V) \rightarrow (Y,W)$ in $\mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$, where $\iota \colon X \rightarrow Y$ is an embedding, is defined by formula \eqref{eq:action of Sp on emb-morphisms}. 
            The fact that $L^p_{\left( \left( \iota(X),\iota^{-1} \right)^{\mathrm{op}},T \right)} \colon L^p(X,V) \rightarrow L^p(Y,W)$ is a contractive map is shown using the same arguments as in Example \ref{ex:Lpemb functor}. 
            The action of $L^p$ on morphisms of the type $\left( m^{\mathrm{op}},T \right) \colon (X,V) \rightarrow (Y,W)$ in $\mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$, where $m \colon Y \rightarrow X$ is a measure-preserving map, is defined by formula \eqref{eq:action of Sp on measure-preserving maps}. 
            The fact that $L^p_{\left( m^{\mathrm{op}},T \right)} \colon L^p(X,V) \rightarrow L^p(Y,W)$ is a contractive map can be verified, for any $p \in [1,\infty)$ and for any $f \in L^p(X,V)$, in the following way: 
            \begin{align} 
                \left\lVert L^p_{\left( m^{\mathrm{op}},T \right)} f\right\rVert_{L^p(Y,W)} &= \left( \int_{Y} \left\lVert (T \circ f \circ m)(y)\right\rVert_W^p d\mu_{Y}(y) \right)^{\frac{1}{p}} \notag \\ 
                &= \left( \int_{X} \left\lVert (T \circ f)(x)\right\rVert_W^p d\mu_{X}(x) \right)^{\frac{1}{p}} \label{eq:use that m is a measure-preserving map} \\ 
                &\leq \left( \int_{X} \left\lVert f(x)\right\rVert_V^p d\mu_{X}(x) \right)^{\frac{1}{p}} \label{eq:use once more that T is a contractive map} \\ 
                &= \left\lVert f\right\rVert_{L^p(X,V)}, \notag 
            \end{align} 
            where equality \eqref{eq:use that m is a measure-preserving map} follows from the fact that $m$ is a measure-preserving map, while inequality \eqref{eq:use once more that T is a contractive map} follows from the fact that $T \colon V \rightarrow W$ is a contractive map. 
            In the case $p = \infty$, for any $f \in L^\infty(X,V)$, we have: 
            \begin{align} 
                \left\lVert L^\infty_{\left( m^{\mathrm{op}},T \right)} f\right\rVert_{L^\infty(Y,W)} &= \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f \circ m)(y)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ Y} \notag \\ 
                &\leq \inf \Set{\epsilon \geq 0 | \left\lVert (T \circ f)(x)\right\rVert_W \leq \epsilon \quad \text{a.e.~on} ~ X} \notag \\ 
                &\leq \inf \Set{\epsilon \geq 0 | \left\lVert f(x)\right\rVert_V \leq \epsilon \quad \text{a.e.~on} ~ X} \label{eq:use once more again that T is a contractive map} \\ 
                &= \left\lVert f\right\rVert_{L^\infty(X,V)}, \notag 
            \end{align} 
            where inequality \eqref{eq:use once more again that T is a contractive map} follows from the fact that $T$ is contractive. \\ 
            For any object $(X,V) \in \mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$, the linear map $\sigma_{(X,V)} \colon V \rightarrow L^p(X,V)$ is defined by formula \eqref{eq:definition of sigma} and its boundedness follows from the same arguments used in Example \ref{ex:simple functions with values in normed vector spaces 1}. \\ 
            Finally, axioms \eqref{ax:compatibility with linear contractive maps 1}, \eqref{ax:compatibility with complementary embeddings in the normed case 1}, \ref{ax:compatibility with measure-preserving maps in the normed case}, \eqref{ax:boundedness of c 1} and \eqref{ax:norm inequality wrt complementary embeddings 1} are satisfied for the same reasons as in Example \ref{ex:simple functions with values in normed vector spaces 2}.

\label{rem:on all the forgetful functors} 
            Expanding on Remark \ref{rem:relating Ban and NVect}, we point out the existence of two faithful and injective-on-objects functors $\mathcal{V}_{\mathrm{emb}}^p \colon \mathbb{B}_{\mathrm{emb}}^p \hookrightarrow \mathbb{NV}_{\mathrm{emb}}^p$ and $\mathcal{V}^p \colon \mathbb{B}^p \hookrightarrow \mathbb{NV}^p$. \\ 
            For any object $(F,c) \in \mathbb{B}_{\mathrm{emb}}^p$ -- where we recall that $F \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$ is a bifunctor and $c$ assigns, to every object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, a bounded linear map $c_{(X,V)} \colon V \rightarrow F(X,V)$ -- we define $\mathcal{V}_{\mathrm{emb}}^p(F,c) = \left( \mathcal{V}_{\mathrm{emb}}^pF,\mathcal{V}_{\mathrm{emb}}^pc \right) \in \mathbb{NV}_{\mathrm{emb}}^p$ as follows. 
            The bifunctor $\mathcal{V}_{\mathrm{emb}}^pF$ is defined by pre-composing the bifunctor $F$ with the bifunctor $(\operatorname{id},\overline{\cdot})$ and post-composing it with the functor $\mathcal{I}$ (both were introduced in Remark \ref{rem:relating Ban and NVect}). More explicitly, we set:   
            \begin{equation} 
            \label{eq:definition of the functor Vembp} 
                \mathcal{V}_{\mathrm{emb}}^pF = \mathcal{I} \circ F \circ (\operatorname{id},\overline{\cdot}) \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}} \rightarrow \mathbf{NVect}_{\mathbb{K}}. 
            \end{equation} 
            The bounded linear map $\left( \mathcal{V}_{\mathrm{emb}}^pc \right)_{(Y,W)} \colon W \rightarrow \left( \mathcal{V}_{\mathrm{emb}}^pF \right)(Y,W)$ is defined, for any object $(Y,W) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, as:  
            \begin{equation*} 
                \left( \mathcal{V}_{\mathrm{emb}}^pc \right)_{(Y,W)} = \mathcal{I}_{c_{(\operatorname{id},\overline{\cdot})(Y,W)}},  
            \end{equation*} 
            where the last formula is understood as the functor $\mathcal{I} \colon \mathbf{BAN}_{\mathbb{K}} \hookrightarrow \mathbf{NVECT}_{\mathbb{K}}$ acting on the bounded linear map of Banach spaces $c_{(Y,\overline{W})} \colon \overline{W} \rightarrow F\left( Y,\overline{W} \right)$.   
            The validity of axioms \ref{ax:compatibility with linear contractive maps}, \ref{ax:compatibility with complementary embeddings in the normed case}, \ref{ax:boundedness of c} and \ref{ax:norm inequality wrt complementary embeddings} for $\mathcal{V}_{\mathrm{emb}}^p(F,c) \in \mathbb{NV}_{\mathrm{emb}}^p$ is ensured by the fact that $(F,c) \in \mathbb{B}_{\mathrm{emb}}^p$ satisfies the same axioms. \\ 
            The action of the functor $\mathcal{V}_{\mathrm{emb}}^p$ on any morphism $\psi \colon (F,c) \rightarrow (G,d)$ in $\mathbb{B}_{\mathrm{emb}}^p$ is defined by:  
            \begin{equation*} 
                \left( \mathcal{V}_{\mathrm{emb}}^p \right)_{\psi} = \mathcal{I} \bullet \psi \bullet (\operatorname{id},\overline{\cdot}) \colon \mathcal{V}_{\mathrm{emb}}^pF \Rightarrow \mathcal{V}_{\mathrm{emb}}^pG, 
            \end{equation*} 
            where $\mathcal{I} \bullet \psi \bullet (\operatorname{id},\overline{\cdot})$ denotes the \emph{whiskered composite} of the natural transformation $\psi$ with the functors $\mathcal{I}$ and $(\operatorname{id},\overline{\cdot})$ (see \cite[\S 1.7]{Riehl2016} for more details about whiskering). 
            The requirement, for any morphism in the category $\mathbb{NV}_{\mathrm{emb}}^p$, that the equality $\left( \left( \mathcal{V}_{\mathrm{emb}}^p \right)_{\psi} \right)_{(Y,W)} \circ \left( \mathcal{V}_{\mathrm{emb}}^pc \right)_{(Y,W)} = \left( \mathcal{V}_{\mathrm{emb}}^pd \right)_{(Y,W)}$ is satisfied for any object $(Y,W) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, can be checked as follows: 
            \begin{align} 
                \left( \left( \mathcal{V}_{\mathrm{emb}}^p \right)_{\psi} \right)_{(Y,W)} \circ \left( \mathcal{V}_{\mathrm{emb}}^pc \right)_{(Y,W)} &= \mathcal{I}_{\psi_{(\operatorname{id},\overline{\cdot})(Y,W)}} \circ \mathcal{I}_{c_{(\operatorname{id},\overline{\cdot})(Y,W)}} \notag \\ 
                &= \mathcal{I}_{\psi_{(\operatorname{id},\overline{\cdot})(Y,W)} \circ c_{(\operatorname{id},\overline{\cdot})(Y,W)}} \notag \\ 
                &= \mathcal{I}_{d_{(\operatorname{id},\overline{\cdot})(Y,W)}} \label{eq:use that psi is a morphism in Bembp} \\ 
                &= \left( \mathcal{V}_{\mathrm{emb}}^pd \right)_{(Y,W)}, \notag 
            \end{align} 
            where equation \eqref{eq:use that psi is a morphism in Bembp} follows from the fact that $\psi \colon (F,c) \rightarrow (G,d)$ is a morphism in the category $\mathbb{B}_{\mathrm{emb}}^p$. \\  
            The analogous faithful and injective-on-objects functor $\mathcal{V}^p \colon \mathbb{B}^p \hookrightarrow \mathbb{NV}^p$ is defined in the exact same way as the functor $\mathcal{V}_{\mathrm{emb}}^p$ except that the category $\mathbf{Meas}_\mathrm{emb}$ has to be substituted everywhere with the category $\mathbf{Meas}^\mathrm{op}$. \\  
            Combining $\mathcal{V}_{\mathrm{emb}}^p$ and $\mathcal{V}^p$ with the functor $\mathcal{U}^p \colon \mathbb{NV}^p \rightarrow \mathbb{NV}_{\mathrm{emb}}^p$ from Remark \ref{rem:forgetful functor from NVp to NVembp}, we obtain a faithful functor $\tilde{\mathcal{U}}^p \colon \mathbb{B}^p \rightarrow \mathbb{B}_{\mathrm{emb}}^p$ defined as the restriction of $\mathcal{U}^p$ to the subcategories $\mathbb{B}^p$ and $\mathbb{B}_{\mathrm{emb}}^p$. 
            The overall situation is summarised in the following commutative diagram of functors: 
            \begin{equation*} 
                \begin{tikzcd} 
                    \mathbb{B}^p \ar[rr, "\tilde{\mathcal{U}}^p"] \ar[dd, hook, "\mathcal{V}^p"']    &   & \mathbb{B}_{\mathrm{emb}}^p \ar[dd, hook, "\mathcal{V}_{\mathrm{emb}}^p"] \\ 
                        &   & \\ 
                    \mathbb{NV}^p \ar[rr, "\mathcal{U}^p"]     &   & \mathbb{NV}_{\mathrm{emb}}^p 
                \end{tikzcd} 
            \end{equation*} 
            As a special instance of this feature, we have that the object $(L^p,\sigma) \in \mathbb{B}_{\mathrm{emb}}^p$, discussed in Example \ref{ex:Lpemb functor}, is the image of the object $(L^p,\sigma) \in \mathbb{B}^p$, discussed in Example \ref{ex:Lp functor}, via the functor $\tilde{\mathcal{U}}^p$.  
            Incidentally, this also motivates why we use the same symbol to denote them.

\label{thm:universal properties of Bochner integrable functions} 
            Let $p \in [1,\infty)$. The bifunctors of the Banach spaces of equivalence classes of $L^p$ Bochner integrable functions, under equality almost everywhere, have the following universal properties: 
            \begin{itemize} 
                \item[(i)] $(L^p,\sigma)$ is the initial object of the category $\mathbb{B}_{\mathrm{emb}}^p$; 
                \item[(ii)] $(L^p,\sigma)$ is the initial object of the category $\mathbb{B}^p$. 
            \end{itemize}

Let us consider statement (i). 
            We define a functor $\mathcal{C}_{\mathrm{emb}}^p \colon \mathbb{NV}_{\mathrm{emb}}^p \rightarrow \mathbb{B}_{\mathrm{emb}}^p$ as follows.  
            The action of $\mathcal{C}_{\mathrm{emb}}^p$ on any object $(F,c) \in \mathbb{NV}_{\mathrm{emb}}^p$ -- where we recall that $F \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}} \rightarrow \mathbf{NVect}_{\mathbb{K}}$ is a bifunctor and $c$ assigns, to every object $(Y,W) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{NVect}_{\mathbb{K}}$, a bounded linear map of normed vector spaces $c_{(Y,W)} \colon W \rightarrow F(Y,W)$ -- is denoted by $\mathcal{C}_{\mathrm{emb}}^p(F,c) = \left( \mathcal{C}_{\mathrm{emb}}^pF,\mathcal{C}_{\mathrm{emb}}^pc \right) \in \mathbb{B}_{\mathrm{emb}}^p$ and is defined informally by pre-composing the bifunctor $F$ with the bifunctor $(\operatorname{id},\mathcal{I})$ and post-composing it with the functor $\overline{\cdot}$ (both were introduced in Remark \ref{rem:relating Ban and NVect}). More explicitly, we set: 
            \begin{equation} 
            \label{eq:definition of the functor Cembp} 
                \begin{split} 
                    \mathcal{C}_{\mathrm{emb}}^pF = \overline{\cdot} \circ F \circ (\operatorname{id},\mathcal{I}) \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}. 
                \end{split} 
            \end{equation} 
            The bounded linear map $\left( \mathcal{C}_{\mathrm{emb}}^pc \right)_{(X,V)} \colon V \rightarrow \left( \mathcal{C}_{\mathrm{emb}}^pF \right)(X,V)$ is defined, for any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, as:  
            \begin{equation} 
            \label{eq:action of Cembp on c} 
                \left( \mathcal{C}_{\mathrm{emb}}^pc \right)_{(X,V)} = \overline{\cdot}_{c_{(\operatorname{id},\mathcal{I})(X,V)}},  
            \end{equation} 
            where the completion functor $\overline{\cdot} \colon \mathbf{NVECT}_{\mathbb{K}} \rightarrow \mathbf{BAN}_{\mathbb{K}}$ acts on the bounded linear map of normed vector spaces $c_{(X,\mathcal{I}V)} \colon \mathcal{I}V \rightarrow F\left( Y,\mathcal{I}V \right)$. 
            The validity of axioms \ref{ax:compatibility with linear contractive maps}, \ref{ax:compatibility with complementary embeddings in the normed case}, \ref{ax:boundedness of c} and \ref{ax:norm inequality wrt complementary embeddings} for $\mathcal{C}_{\mathrm{emb}}^p(F,c) \in \mathbb{B}_{\mathrm{emb}}^p$ is ensured by the fact that the axioms do not rely on any completeness assumption for the vector spaces and by the fact that the object $(F,c) \in \mathbb{NV}_{\mathrm{emb}}^p$ satisfies the same axioms. \\ 
            The action of the functor $\mathcal{C}_{\mathrm{emb}}^p$ on any morphism $\psi \colon (F,c) \rightarrow (G,d)$ in the category $\mathbb{NV}_{\mathrm{emb}}^p$ is defined by: 
            \begin{equation*} 
                \left( \mathcal{C}_{\mathrm{emb}}^p \right)_{\psi} = \overline{\cdot} \bullet \psi \bullet (\operatorname{id},\mathcal{I}) \colon \mathcal{C}_{\mathrm{emb}}^pF \Rightarrow \mathcal{C}_{\mathrm{emb}}^pG, 
            \end{equation*} 
            where $\overline{\cdot} \bullet \psi \bullet (\operatorname{id},\mathcal{I})$ denotes the whiskered composite of the natural transformation $\psi$ with the functors $\overline{\cdot}$ and $(\operatorname{id},\mathcal{I})$ (see \cite[\S 1.7]{Riehl2016} about whiskering). 
            The requirement that the equality $\left( \left( \mathcal{C}_{\mathrm{emb}}^p \right)_{\psi} \right)_{(X,V)} \circ \left( \mathcal{C}_{\mathrm{emb}}^pc \right)_{(X,V)} = \left( \mathcal{C}_{\mathrm{emb}}^pd \right)_{(X,V)}$ is satisfied for any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$ can be checked as follows: 
            \begin{align} 
                \left( \left( \mathcal{C}_{\mathrm{emb}}^p \right)_{\psi} \right)_{(X,V)} \circ \left( \mathcal{C}_{\mathrm{emb}}^pc \right)_{(X,V)} &= \overline{\cdot}_{\psi_{(\operatorname{id},\mathcal{I})(X,V)}} \circ \overline{\cdot}_{c_{(\operatorname{id},\mathcal{I})(X,V)}} \notag \\ 
                &= \overline{\cdot}_{\psi_{(\operatorname{id},\mathcal{I})(X,V)} \circ c_{(\operatorname{id},\mathcal{I})(X,V)}} \notag \\ 
                &= \overline{\cdot}_{d_{(\operatorname{id},\mathcal{I})(X,V)}} \label{eq:use that psi is a morphism in NVembp} \\ 
                &= \left( \mathcal{C}_{\mathrm{emb}}^pd \right)_{(X,V)}, \notag 
            \end{align} 
            where equation \eqref{eq:use that psi is a morphism in NVembp} follows from the fact that $\psi \colon (F,c) \rightarrow (G,d)$ is a morphism in the category $\mathbb{NV}_{\mathrm{emb}}^p$. 
 
            We claim that the functor $\mathcal{C}_{\mathrm{emb}}^p \colon \mathbb{NV}_{\mathrm{emb}}^p \rightarrow \mathbb{B}_{\mathrm{emb}}^p$ is left adjoint to the functor $\mathcal{V}_{\mathrm{emb}}^p \colon \mathbb{B}_{\mathrm{emb}}^p \hookrightarrow \mathbb{NV}_{\mathrm{emb}}^p$ described in Remark \ref{rem:on all the forgetful functors}.                  
            To verify this, it suffices to notice that, while the functor $\mathcal{V}_{\mathrm{emb}}^p$ is defined, in formula \eqref{eq:definition of the functor Vembp}, by pre-composition and post-composition with certain functors, the functor $\mathcal{C}_{\mathrm{emb}}^p$ is defined, in formula \eqref{eq:definition of the functor Cembp}, by pre-composition and post-composition with the left adjoints of the corresponding functors (cf.~Remark \ref{rem:relating Ban and NVect}). 
            The claim then follows from standard results in category theory, see for example \cite[Proposition 4.4.6]{Riehl2016}. \\ 
            Using the well-known fact that left adjoint functors preserve initial objects, in combination with Theorem \ref{thm:universal properties of simple functions with values in normed vector spaces}, we thus obtain that the initial object of the category $\mathbb{B}_{\mathrm{emb}}^p$ is $\mathcal{C}_{\mathrm{emb}}^p(S^p,\sigma) = \left( \mathcal{C}_{\mathrm{emb}}^p S^p,\mathcal{C}_{\mathrm{emb}}^p \sigma \right)$, for any $p \in [1,\infty)$. \\ 
            According to formula \eqref{eq:definition of the functor Cembp}, the bifunctor $\mathcal{C}_{\mathrm{emb}}^p S^p \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$ associates, to any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, the Banach space 
            \begin{equation*} 
                \begin{split} 
                    \left( \mathcal{C}_{\mathrm{emb}}^p S^p \right)(X,V) &= \left( \overline{\cdot} \circ S^p \circ (\operatorname{id},\mathcal{I}) \right)(X,V) \\ 
                    &= \left( \overline{\cdot} \circ S^p \right)(X,\mathcal{I}V) \\ 
                    &= \overline{S^p(X,\mathcal{I}V)} \\ 
                    &= L^p(X,V),  
                \end{split} 
            \end{equation*} 
            where the last equality corresponds to the definition of the Banach spaces of equivalence classes of $L^p$ Bochner integrable functions, under equality almost everywhere, as the completion, with respect to the $L^p$ norm \eqref{eq:Lp norm}, of the normed vector spaces of equivalence classes of simple functions, under equality almost everywhere (we refer to \cite{Cohn2013} and \cite{Yosida1995} for more details). \\ 
            Regarding the bounded linear map $\left( \mathcal{C}_{\mathrm{emb}}^p \sigma \right)_{(X,V)} \colon V \rightarrow \mathcal{C}_{\mathrm{emb}}^p S^p(X,V)$, for any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, it is obtained, according to formula \eqref{eq:action of Cembp on c}, as the extension by continuity of the bounded linear map $\sigma_{(X,\mathcal{I}V)} \colon \mathcal{I}V \rightarrow S^p(X,\mathcal{I}V)$ between normed vector spaces to a bounded linear map from the completion of $\mathcal{I}V$ to the completion of $S^p(X,\mathcal{I}V)$. 
            Since $\mathcal{I}V$ is already complete, the extension is in fact trivial and the resulting map $\left( \mathcal{C}_{\mathrm{emb}}^p \sigma \right)_{(X,V)}$ coincides with the map $\sigma_{(X,V)}$, except that its values are regarded as elements of $\overline{S^p(X,\mathcal{I}V)} = L^p(X,V)$. \\ 
            We have thus identified $\mathcal{C}_{\mathrm{emb}}^p(S^p,\sigma) = (L^p,\sigma)$ as the initial object of the category $\mathbb{B}_{\mathrm{emb}}^p$, for any $p \in [1,\infty)$, and this concludes the proof of statement (i). 
             
            The same argument, with the category $\mathbf{Meas}_\mathrm{emb}$ substituted everywhere by the category $\mathbf{Meas}^\mathrm{op}$, the functors $\mathcal{C}_{\mathrm{emb}}^p$ and $\mathcal{V}_{\mathrm{emb}}^p$ substituted by the functors $\mathcal{C}^p$ and $\mathcal{V}^p$, the categories $\mathbb{NV}_{\mathrm{emb}}^p$ and $\mathbb{B}_{\mathrm{emb}}^p$ substituted by the categories $\mathbb{NV}^p$ and $\mathbb{B}^p$, yields the proof of statement (ii).

<!-- remark-11: body_md -->
\label{rem:issue with the infinity case} 
            The case $p = \infty$ is not covered by Teorem \ref{thm:universal properties of Bochner integrable functions} because the normed vector spaces $S^\infty(X,V)$ of equivalence classes of simple functions, under equality almost everywhere, endowed with the $L^\infty$ norm \eqref{eq:Linfty norm} are in general not dense inside the Banach spaces $L^\infty(X,V)$, for any measure space $X$ and any Banach space $V$. \\ 
            Density of $S^\infty(X,V)$ inside $L^\infty(X,V)$ is in fact guaranteed only when $V$ is a separable Banach space. \\   
            The following elementary example illustrates the situation.  
            Consider the measure space $\mathbb{N}$ endowed with the discrete measure $\mu_{\mathbb{N}}(n) = 2^{-n}$, for all $n \in \mathbb{N}$. 
            Then, $(\mathbb{N},\mu_{\mathbb{N}})$ has finite total measure and we can regard it as an object of the category $\mathbf{Meas}$. 
            Consider the Hilbert space $l^2(\mathbb{N})$ and define a function $f \colon \mathbb{N} \rightarrow l^2(\mathbb{N})$ by setting $f(n) = e_n$, for any $n \in \mathbb{N}$, where $\{e_n\}_{n \in \mathbb{N}}$ is the standard ortonormal basis of $l^2(\mathbb{N})$. 
            It is easy to check that $f$ is strongly measurable, Bochner integrable and satisfies $\left\lVert f\right\rVert_{L^\infty\left( \mathbb{N},l^2(\mathbb{N}) \right)} = 1$. 
            Suppose $g \in L^\infty\left( \mathbb{N},l^2(\mathbb{N}) \right)$ is any simple function. 
            Since $g$ has finite range, at least one of its level-sets, say $A_x = g^{-1}(x)$ for some $x \in l^2(\mathbb{N})$, must be infinite inside $\mathbb{N}$. 
            For any elements $n_1,n_2 \in A_x$, we then have: 
            \begin{equation*} 
                \begin{split} 
                    \sqrt{2} &= \left\lVert e_{n_1} - e_{n_2}\right\rVert_{l^2(\mathbb{N})} \\ 
                    &= \left\lVert f(n_1) - f(n_2)\right\rVert_{l^2(\mathbb{N})} \\ 
                    &= \left\lVert \left( f(n_1) - x \right) - \left( f(n_2) - x \right)\right\rVert_{l^2(\mathbb{N})} \\ 
                    &= \left\lVert \left( f(n_1) - g(n_1) \right) - \left( f(n_2) - g(n_2) \right)\right\rVert_{l^2(\mathbb{N})} \\ 
                    &\leq \left\lVert f(n_1) - g(n_1)\right\rVert_{l^2(\mathbb{N})} + \left\lVert f(n_2) - g(n_2)\right\rVert_{l^2(\mathbb{N})}. 
                \end{split} 
            \end{equation*} 
            Taking the supremum of the last inequality over all $n_1,n_2 \in \mathbb{N}$, we finally obtain 
            \begin{equation*} 
                \frac{\sqrt{2}}{2} \leq \left\lVert f - g\right\rVert_{L^\infty\left( \mathbb{N},l^2(\mathbb{N}) \right)}, 
            \end{equation*} 
            hence simple functions cannot be dense in $L^\infty\left( \mathbb{N},l^2(\mathbb{N}) \right)$.

\label{prop:uniqueness of the Bochner integral} 
            The unique morphism $(L^1,\sigma) \rightarrow (\mathrm{proj}_{2},\tau)$ in the category $\mathbb{B}_{\mathrm{emb}}^1$ is given, component-wise, by the Bochner integral: 
            \begin{equation*} 
                \int_{(X,V)} \colon L^1(X,V) \rightarrow V, 
            \end{equation*} 
            for any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$.

<!-- proof-7: body_md -->
By statement (i) of Theorem \ref{thm:universal properties of Bochner integrable functions}, it suffices to show that $\int_{-}$ is a morphism in the category $\mathbb{B}_{\mathrm{emb}}^1$. 
            For any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, the fact that the component $\int_{(X,V)} \colon L^1(X,V) \rightarrow V$ is a linear contractive map follows directly from linearity of the Bochner integral and from the basic inequality (see, for example, \cite[Theorem II.2.4]{Diestel/Uhl1977})  
            \begin{equation*} 
                \left\lVert \int_X f d\mu_X\right\rVert_V \leq \int_X \left\lVert f\right\rVert_V d\mu_X, \qquad \forall f \in L^1(X,V). 
            \end{equation*} 
            In order to check naturality of the transformation $\int_{-} \colon L^1 \Rightarrow \mathrm{proj}_{2}$, we need to show that, for any morphism $(\iota,T) \colon (X,V) \rightarrow (Y,W)$ in $\mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$, the diagram 
            \begin{equation*} 
                \begin{tikzcd} 
                    L^1(X,V) \ar[rr, "L^1_{(\iota,T)}"] \ar[dd, "\int_{(X,V)}"']  &   & L^1(Y,W) \ar[dd, "\int_{(Y,W)}"] \\ 
                        &   & \\ 
                    \mathrm{proj}_{2}(X,V) \ar[rr, "\left( \mathrm{proj}_{2} \right)_{(\iota,T)}"]   &   & \mathrm{proj}_{2}(Y,W) 
                \end{tikzcd} 
            \end{equation*} 
            commutes in the category $\mathbf{Ban}_{\mathbb{K}}$. 
            Combining formula \eqref{eq:action of Sp on emb-morphisms} with formula \eqref{eq:action of proj on morphisms}, for any element $f \in L^1(X,V)$, we obtain:  
            \begin{align} 
                \left( \int_{(Y,W)} \circ ~ L^1_{(\iota,T)} \right) (f) &= \int_Y L^1_{(\iota,T)} f d\mu_Y \notag \\ 
                &= \int_{\iota(X)} T \circ f \circ \iota^{-1} d\mu_{\iota(X)} \label{eq:use definition of L1_iotaT} \\ 
                &= \int_X T \circ f d\mu_X \label{eq:use that iota is measure-preserving} \\ 
                &= T \left( \int_X f d\mu_X \right) \label{eq:use properties of the Bochner integral} \\ 
                &= \left( \left( \mathrm{proj}_{2} \right)_{(\iota,T)} \circ \int_{(X,V)} \right) (f), \notag 
            \end{align} 
            where equation \eqref{eq:use definition of L1_iotaT} follows from the definition of $L^1_{(\iota,T)} f$, equation \eqref{eq:use that iota is measure-preserving} follows from the fact that embeddings preserve measures and equation \eqref{eq:use properties of the Bochner integral} is one of the basic properties of the Bochner integral (cf.~\cite[\S V.5, Corollary 2]{Yosida1995}). \\ 
            Finally, we need to check the requirement that $\int_{(X,V)} \circ ~ \sigma_{(X,V)} = \tau_{(X,V)}$, for any object $(X,V) \in \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}}$. 
            Using formulas \eqref{eq:definition of sigma} and \eqref{eq:definition of tau}, for any vector $v \in V$, we obtain indeed:  
            \begin{align*} 
                \left( \int_{(X,V)} \circ ~ \sigma_{(X,V)} \right) (v) &= \int_X \sigma_{(X,V)}(v) d\mu_X \\ 
                &= \int_X v \cdot \chi_X d\mu_X \\ 
                &= \mu_X(X) \cdot v \\ 
                &= \tau_{(X,V)} (v). 
            \end{align*} 
            This concludes the proof of the proposition.

<!-- remark-13: body_md -->
In Proposition \ref{prop:uniqueness of the Bochner integral}, we have regarded the pair $(\mathrm{proj}_{2},\tau)$ as an object of the category $\mathbb{B}_{\mathrm{emb}}^1$. 
            But the pair $(\mathrm{proj}_{2},\tau)$ can also be regarded as an object of the category $\mathbb{B}^1$, where the bifunctor $\mathrm{proj}_{2} \colon \mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$ is the obvious extension of the bifunctor $\mathrm{proj}_{2} \colon \mathbf{Meas}_\mathrm{emb} \times \mathbf{Ban}_{\mathbb{K}} \rightarrow \mathbf{Ban}_{\mathbb{K}}$ and the assignment $\tau$ is left unchanged. 
            In particular, the validity of axiom \ref{ax:compatibility with measure-preserving maps in the normed case}, for any morphism $(m^{\mathrm{op}},\operatorname{id}_V) \colon (X,V) \rightarrow (Y,V)$ in the category $\mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$, where $m \colon Y \rightarrow X$ is a measure-preserving map, can be checked as follows: 
            \begin{align} 
                \left( \mathrm{proj}_{2} \right)_{(m^{\mathrm{op}},\operatorname{id}_V)} \circ \tau_{(X,V)} &= \operatorname{id}_V \circ \, \mu_X(X) \operatorname{id}_V \notag \\ 
                &= \mu_X(X) \operatorname{id}_V \notag \\ 
                &= \mu_Y(Y) \operatorname{id}_V \label{eq:using that m is measure-preserving} \\ 
                &= \tau_{(Y,V)} \notag, 
            \end{align} 
            where eqation \eqref{eq:using that m is measure-preserving} follows from the fact that $m$ is a measure-preserving map. \\ 
            From statement (ii) of Theorem \ref{thm:universal properties of Bochner integrable functions} we then obtain that the unique morphism $\int_{-} \colon (L^1,\sigma) \rightarrow (\mathrm{proj}_{2},\tau)$ in the category $\mathbb{B}_{\mathrm{emb}}^1$ appearing in Proposition \ref{prop:uniqueness of the Bochner integral} is in fact a morphism in the category $\mathbb{B}^1$. 
            The additional naturality of $\int_{-}$ with respect to morphisms of the type $\left( m^{\mathrm{op}},T \right) \colon (X,V) \rightarrow (Y,W)$ in the category $\mathbf{Meas}^\mathrm{op} \times \mathbf{Ban}_{\mathbb{K}}$, where $m \colon Y \rightarrow X$ is a measure-preserving map, corresponds, for any $f \in L^1(X,V)$, to the formula for integration under a change of variables: 
            \begin{align} 
                \left( \int_{(Y,W)} \circ ~ L^1_{\left( m^{\mathrm{op}},T \right)} \right) (f) &= \int_Y L^1_{\left( m^{\mathrm{op}},T \right)} f d\mu_Y \notag \\ 
                &= \int_Y T \circ f \circ m d\mu_Y \label{eq:use the action of Sp on measure-preserving maps} \\ 
                &= \int_X T \circ f d\mu_X \notag \\ 
                &= T \left( \int_X f d\mu_X \right) \notag \\ 
                &= \left( \left( \mathrm{proj}_{2} \right)_{\left( m^{\mathrm{op}},T \right)} \circ \int_{(X,V)} \right) (f), \notag 
            \end{align}  
            where, in equality \eqref{eq:use the action of Sp on measure-preserving maps}, we used formula \eqref{eq:action of Sp on measure-preserving maps}.
