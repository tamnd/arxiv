<!-- lem-1: body_md -->
\label{lem:li-normalisation} 
For all $t \in \mathcal{T}_{BMI}$ and $N \in \mathcal{N}_{BMI}$, 
\[ 
t \mathrel{\rightarrow_{w}}^{\ast} N \iff  t \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} N. 
\] 
In particular, the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of a $BMI$--term terminates if and only if the term has a weak normal form.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
(See also Klop~\cite[Section 2.1]{Klo92}.) 
Since $\mathrel{\rightarrow_{\mathrm{li}}}$ implies $\mathrel{\rightarrow_{w}}$, the right-to-left implication is immediate. 
To prove the forward direction, we first observe that $BMI$--combinatory logic is a non-erasing, orthogonal term rewriting system. 
 
Now, suppose that $t \mathrel{\rightarrow_{w}}^{\ast} N$. 
By Church's uniform normalisation theorem~\cite{KOO01}, $t$ is strongly normalisable, that is, every $\mathrel{\rightarrow_{w}}$--sequence from $t$ terminates. 
In particular, the maximal $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence from $t$ terminates at some $P \in \mathcal{N}_{BMI}$. 
Moreover, orthogonal term rewriting systems are confluent~\cite{Ros73}, so $t \mathrel{\rightarrow_{w}}^{\ast} N$ and $t \mathrel{\rightarrow_{w}}^{\ast} P$ imply $P=N$. 
Therefore $t \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} N$.
<!-- end proof-1 -->
<!-- lem-2: body_md -->
\label{lem:infinite-events} 
Let $u_{0} \mathrel{\rightarrow_{\mathrm{li}}} u_{1} \mathrel{\rightarrow_{\mathrm{li}}} \cdots$ be an infinite $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence in $\mathcal{T}_{BMI}$. 
The $M$--contraction $\Gamma\left[Mv\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[vv\right]$ appears in the sequence infinitely many times.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Otherwise, there would exist an infinite reduction sequence consisting solely of contractions other than $M$. 
However, since contractions other than $M$ strictly reduce the total number of occurrences of constants and variables, this would contradict the assumption.
<!-- end proof-2 -->
<!-- defi-1: body_md -->
Let $u_{0} \mathrel{\rightarrow_{\mathrm{li}}} u_{1} \mathrel{\rightarrow_{\mathrm{li}}} \cdots$ be an infinite $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence in $\mathcal{T}_{BMI}$ and let $x \in \mathcal{V}$. 
For $i \in \mathbb{N}$, let 
\[ 
u_{m_{i}} = \Gamma_{i}\left[M v_{i}\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma_{i}\left[v_{i}v_{i}\right] = u_{m_{i}+1} 
\] 
be the $i$--th contraction of $M$ in the sequence. 
Define the \emph{$x$--trace} $\tau_{x}\left(u_{0}\right)$ of $u_{0}$ by 
\[ 
\tau_{x}\left( u_{0} \right) := \left(\left(p_{i}, q_{i}\right)\right)_{i\in \mathbb{N}} \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}, 
\] 
where 
\begin{align*} 
p_{i} &:= \text{the number of occurrences of $x$ in $u_{m_{i}}$ strictly before the redex $M v_{i}$}, \\ 
q_{i} &:= \left|v_{i}\right|_{x}. 
\end{align*}
<!-- end defi-1 -->
<!-- defi-2: body_md -->
For $\alpha,\beta \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}$, define \emph{tail equivalence} by 
\[ 
\alpha \mathrel{\sim_{\mathrm{tail}}} \beta :\iff \exists i, j \in \mathbb{N}.\  \forall k \in \mathbb{N}.\ \alpha\left(i + k\right) = \beta\left(j + k\right). 
\]
<!-- end defi-2 -->
<!-- defi-3: body_md -->
Define the set $\mathcal{A}_{BMI}$ as the disjoint union: 
\[ 
\mathcal{A}_{BMI} := \mathcal{N}_{BMI} \sqcup \left(\mathbb{N}^{2}\right)^{\mathbb{N}} / \mathrel{\sim_{\mathrm{tail}}}. 
\] 
For a term $u \in \mathcal{T}_{BMI}$ and a variable $x$, define the \emph{trace invariant} $\mathrm{Tr}_{x}\left(u\right) \in \mathcal{A}_{BMI}$ as follows: 
\[ 
\mathrm{Tr}_{x}\left(u\right) := \begin{cases} 
    N, & \text{$u$ has a normal form $N$}, \\ 
    \left[\tau_{x}\left(u\right)\right], & \text{otherwise}. 
\end{cases} 
\]
<!-- end defi-3 -->
<!-- lem-3: body_md -->
\label{lem:prefix} 
If $u \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} v$, then $\mathrm{Tr}_{x}\left(u\right)=\mathrm{Tr}_{x}\left(v\right)$.
<!-- end lem-3 -->
<!-- proof-3: body_md -->
If $v$ has a normal form $N$, then so does $u$. 
Hence $\mathrm{Tr}_{x}\left(u\right)=\mathrm{Tr}_{x}\left(v\right) = N$. 
 
Suppose that $v$ has no normal form. 
Let $u \mathrel{\rightarrow_{\mathrm{li}}} w_{1} \mathrel{\rightarrow_{\mathrm{li}}} \cdots \mathrel{\rightarrow_{\mathrm{li}}} w_{n} = v$ be a $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence from $u$ to $v$. 
Let $c$ be the number of $M$--contractions in that sequence. 
Then $\tau_{x}\left(u\right)_{i+c} = \tau_{x}\left(v\right)_{i}$ for all $i\in \mathbb{N}$, so $\tau_{x}\left(u\right) \mathrel{\sim_{\mathrm{tail}}} \tau_{x}\left(v\right)$. 
Therefore $\mathrm{Tr}_{x} \left(u\right) = \left[\tau_{x}\left(u\right)\right] = \left[\tau_{x}\left(v\right)\right]=\mathrm{Tr}_{x}\left(v\right)$.
<!-- end proof-3 -->
<!-- defi-4: body_md -->
For $c \in \mathbb{N}$, define the map 
\[ 
S_{c}\colon \left(\mathbb{N}^{2}\right)^{\mathbb{N}} \to \left(\mathbb{N}^{2}\right)^{\mathbb{N}} 
\] 
by  
\[ 
S_{c}\alpha :=\left(\left(p_{i} + c, q_{i}\right)\right)_{i\in\mathbb{N}}, 
\] 
where 
\[ 
\alpha = \left(\left(p_{i}, q_{i}\right)\right)_{i\in\mathbb{N}}. 
\] 
Since $S_{c}$ preserves tail equivalence, it induces a map 
\[ 
\mathcal{S}_{c} \colon \left(\mathbb{N}^{2}\right)^{\mathbb{N}}/\mathrel{\sim_{\mathrm{tail}}} \to \left(\mathbb{N}^{2}\right)^{\mathbb{N}}/\mathrel{\sim_{\mathrm{tail}}},\qquad \left[\alpha\right] \mapsto \left[S_{c}\alpha\right]. 
\]
<!-- end defi-4 -->
<!-- defi-5: body_md -->
Let $x \in \mathcal{V}$.  
Define $\star_{x} \colon \mathcal{A}_{BMI} \times \mathcal{A}_{BMI} \to \mathcal{A}_{BMI}$ by 
\[ 
a \star_{x} b := \begin{cases} 
    a, & a \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}} / \mathrel{\sim_{\mathrm{tail}}}, \\ 
    \mathcal{S}_{\left|a\right|_{x}} b & a \in \mathcal{N}_{BMI},\ b \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}} / \mathrel{\sim_{\mathrm{tail}}}, \\ 
    \mathrm{Tr}_{x}\left(ab\right), & a,b \in \mathcal{N}_{BMI}. 
\end{cases} 
\]
<!-- end defi-5 -->
<!-- lem-4: body_md -->
Let $x \in \mathcal{V}$. 
The operator $\star_{x} \colon \mathcal{A}_{BMI} \times \mathcal{A}_{BMI} \to \mathcal{A}_{BMI}$ is well-defined.
<!-- end lem-4 -->
<!-- proof-4: body_md -->
We consider three cases, corresponding to the three clauses in the definition of $\star_{x}$. 
 
\begin{enumerate} 
\item $a \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}} / \mathrel{\sim_{\mathrm{tail}}}$. 
The value $a \star_{x} b = a$ is an element of $\mathcal{A}_{BMI}$ and involves no choice of representatives. 
 
\item $a \in \mathcal{N}_{BMI}$ and $b \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}} / \mathrel{\sim_{\mathrm{tail}}}$. 
Suppose that $b = \left[\alpha\right] = \left[\beta\right]$. 
Put $c := \left|a\right|_{x}$. 
Since $\alpha \mathrel{\sim_{\mathrm{tail}}} \beta$, there exist $i,j \in \mathbb{N}$ such that $\alpha\left(i+k\right) = \beta\left(j+k\right)$ for every $k \in \mathbb{N}$. 
Adding $c$ to the first coordinate gives 
\[ 
\left(S_{c}\alpha\right)\left(i+k\right) = \left(S_{c}\beta\right)\left(j+k\right) \qquad \left(k \in \mathbb{N}\right). 
\] 
Thus $S_{c}\alpha \mathrel{\sim_{\mathrm{tail}}} S_{c}\beta$, so $a \star_{x} b = \mathcal{S}_{c}b$ is independent of the representative of $b$. 
 
\item $a, b \in \mathcal{N}_{BMI}$. 
This case involves no choice of representatives. \qedhere 
\end{enumerate}

<!-- thm-1: body_md -->
\label{thm:homomorphism} 
For all $u,v \in \mathcal{T}_{BMI}$ and $x\in \mathcal{V}$, 
\[ 
\mathrm{Tr}_{x}\left(uv\right) = \mathrm{Tr}_{x}\left(u\right) \star_{x} \mathrm{Tr}_{x}\left(v\right). 
\]
<!-- end thm-1 -->
<!-- proof-5: body_md -->
We consider three cases according to whether $u$ and $v$ have normal forms. 
 
\begin{enumerate} 
\item The $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $u$ is infinite. 
Write this sequence as $u = u_{0} \mathrel{\rightarrow_{\mathrm{li}}} u_{1} \mathrel{\rightarrow_{\mathrm{li}}} \cdots$. 
By the definition of $\mathrel{\rightarrow_{\mathrm{li}}}$, the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $uv$ is 
\[ 
uv = u_{0}v \mathrel{\rightarrow_{\mathrm{li}}} u_{1}v \mathrel{\rightarrow_{\mathrm{li}}} \cdots. 
\] 
Each $M$--contraction $\Gamma\left[Mw\right]\mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[ww\right]$ in $u$'s sequence corresponds precisely to the $M$--contraction $\Gamma\left[Mw\right]v\mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[ww\right]v$ in $uv$'s sequence, and vice versa. 
So 
\[ 
\mathrm{Tr}_{x}\left(uv\right) = \left[\tau_{x}\left(u\right)\right] = \mathrm{Tr}_{x}\left(u\right) = \mathrm{Tr}_{x}\left(u\right) \star_{x} \mathrm{Tr}_{x}\left(v\right). 
\] 
 
\item $u \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} U$ for some $U \in \mathcal{N}_{BMI}$, and the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $v$ is infinite. 
Write the latter sequence as $v = v_{0} \mathrel{\rightarrow_{\mathrm{li}}} v_{1} \mathrel{\rightarrow_{\mathrm{li}}} \cdots$. 
Since $U$ is normal, and each $v_{i}$ is non-normal, the leftmost-innermost strategy selects a redex inside $v_{i}$, rather than at the root of $Uv_{i}$. 
Thus, the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $uv$ has the form 
\[ 
uv \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} Uv = Uv_{0} \mathrel{\rightarrow_{\mathrm{li}}} Uv_{1} \mathrel{\rightarrow_{\mathrm{li}}} \cdots. 
\] 
Each $M$--contraction $\Gamma\left[Mw\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[ww\right]$ in $v$'s sequence corresponds precisely to the $M$--contraction 
\[ 
U\left(\Gamma\left[Mw\right]\right) \mathrel{\rightarrow_{\mathrm{li}}} U\left(\Gamma\left[ww\right]\right) 
\] 
in $Uv$'s sequence, and vice versa. 
The duplicated argument is $w$ in both contractions, while the latter has exactly $\left|U\right|_{x}$ additional occurrences of $x$ before the redex. 
Thus a trace pair $\left(p,q\right)$ in the former becomes $\left(p+\left|U\right|_{x},q\right)$ in the latter, so 
\[ 
\tau_{x}\left(Uv\right) = S_{\left|U\right|_{x}}\tau_{x}\left(v\right). 
\] 
Since $\mathrm{Tr}_{x}\left(u\right) = U$, \prettyref{lem:prefix} gives 
\begin{align*} 
\mathrm{Tr}_{x}\left(uv\right) 
&= \mathrm{Tr}_{x}\left(Uv\right) \\ 
&= \left[S_{\left|U\right|_{x}}\tau_{x}\left(v\right)\right] \\ 
&= \mathcal{S}_{\left|U\right|_{x}}\left(\mathrm{Tr}_{x}\left(v\right)\right) \\ 
&= \mathrm{Tr}_{x}\left(u\right) \star_{x} \mathrm{Tr}_{x}\left(v\right). 
\end{align*} 
 
\item $u \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} U$ and $v \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} V$ for some $U,V \in \mathcal{N}_{BMI}$. 
Write these finite sequences as 
\[ 
u = u_{0} \mathrel{\rightarrow_{\mathrm{li}}} \cdots \mathrel{\rightarrow_{\mathrm{li}}} u_{m} = U, 
\qquad 
v = v_{0} \mathrel{\rightarrow_{\mathrm{li}}} \cdots \mathrel{\rightarrow_{\mathrm{li}}} v_{n} = V. 
\] 
Each contraction $\Gamma\left[R\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[R'\right]$ in the first sequence occurs in $uv$'s sequence as 
\[ 
\Gamma\left[R\right]v \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[R'\right]v. 
\] 
Once $U$ is reached, its normality ensures that each contraction $\Delta\left[R\right] \mathrel{\rightarrow_{\mathrm{li}}} \Delta\left[R'\right]$ in the second sequence occurs as 
\[ 
U\left(\Delta\left[R\right]\right) \mathrel{\rightarrow_{\mathrm{li}}} U\left(\Delta\left[R'\right]\right). 
\] 
Consequently, the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $uv$ begins with 
\begin{align*} 
uv = u_{0}v 
&\mathrel{\rightarrow_{\mathrm{li}}} \cdots \mathrel{\rightarrow_{\mathrm{li}}} u_{m}v = Uv_{0} \\ 
&\mathrel{\rightarrow_{\mathrm{li}}} \cdots \mathrel{\rightarrow_{\mathrm{li}}} Uv_{n} = UV. 
\end{align*} 
Therefore, by \prettyref{lem:prefix} and the definition of $\star_{x}$, 
\[ 
\mathrm{Tr}_{x}\left(uv\right) 
= \mathrm{Tr}_{x}\left(UV\right) 
= U \star_{x} V 
= \mathrm{Tr}_{x}\left(u\right) \star_{x} \mathrm{Tr}_{x}\left(v\right). 
\] 
This applies whether or not $UV$ has a normal form. \qedhere 
\end{enumerate}

<!-- thm-2: body_md -->
\label{thm:identities} 
Let $x \in \mathcal{V}$. 
Regard the constant terms $B,M,I$ as elements of $\mathcal{A}_{BMI}$. 
Then, for all $a,b,c \in \mathcal{A}_{BMI}$, 
\begin{itemize} 
\item $\left(\left(B\star_{x} a\right) \star_{x} b\right) \star_{x} c = a \star_{x} \left(b \star_{x} c\right)$; 
\item $M \star_{x} a = a \star_{x} a$; 
\item $I \star_{x} a = a$. 
\end{itemize}
<!-- end thm-2 -->
<!-- proof-6: body_md -->
To prove the $B$--identity, we consider four cases. 
\begin{enumerate} 
 
\item $a \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}/\mathrel{\sim_{\mathrm{tail}}}$. 
Since $\left|B\right|_{x}=0$, we have $B\star_{x}a=\mathcal{S}_{0}a=a$. 
The first clause in the definition of $\star_{x}$ therefore gives 
\[ 
\left(\left(B\star_{x}a\right)\star_{x}b\right)\star_{x}c 
= a 
= a\star_{x}\left(b\star_{x}c\right). 
\] 
 
\item $a \in \mathcal{N}_{BMI}$ and $b \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}/\mathrel{\sim_{\mathrm{tail}}}$. 
The term $Ba$ is normal, so $B\star_{x}a=Ba$ and $\left|Ba\right|_{x}=\left|a\right|_{x}$. 
Hence 
\begin{align*} 
\left(\left(B\star_{x}a\right)\star_{x}b\right)\star_{x}c 
&= \left(\mathcal{S}_{\left|a\right|_{x}}b\right)\star_{x}c \\ 
&= \mathcal{S}_{\left|a\right|_{x}}b \\ 
&= a\star_{x}\left(b\star_{x}c\right), 
\end{align*} 
where the last two equalities use the fact that a tail-equivalence class is unchanged by application on the right. 
 
\item $a,b \in \mathcal{N}_{BMI}$ and $c \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}/\mathrel{\sim_{\mathrm{tail}}}$. 
The terms $Ba$ and $Bab$ are normal, and 
$\left|Bab\right|_{x}=\left|a\right|_{x}+\left|b\right|_{x}$. 
Consequently, 
\begin{align*} 
\left(\left(B\star_{x}a\right)\star_{x}b\right)\star_{x}c 
&= \mathcal{S}_{\left|a\right|_{x}+\left|b\right|_{x}}c \\ 
&= \mathcal{S}_{\left|a\right|_{x}}\left(\mathcal{S}_{\left|b\right|_{x}}c\right) \\ 
&= a\star_{x}\left(b\star_{x}c\right). 
\end{align*} 
Here we used $\mathcal{S}_{r+s}=\mathcal{S}_{r}\circ\mathcal{S}_{s}$, which follows directly from the definition of the shift operators. 
 
\item $a,b,c \in \mathcal{N}_{BMI}$. 
Since all proper subterms of $Babc$ are normal, its selected contraction is 
\[ 
Babc \mathrel{\rightarrow_{\mathrm{li}}} a\left(bc\right). 
\] 
By \prettyref{lem:prefix} and \prettyref{thm:homomorphism}, 
\begin{align*} 
\left(\left(B\star_{x}a\right)\star_{x}b\right)\star_{x}c 
&= \mathrm{Tr}_{x}\left(Babc\right) \\ 
&= \mathrm{Tr}_{x}\left(a\left(bc\right)\right) \\ 
&= a\star_{x}\left(b\star_{x}c\right). 
\end{align*} 
\end{enumerate} 
 
For the $M$-- and $I$--identities, first suppose that 
$a \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}/\mathrel{\sim_{\mathrm{tail}}}$. 
Since $\left|M\right|_{x}=\left|I\right|_{x}=0$, the definition gives 
\[ 
M\star_{x}a=\mathcal{S}_{0}a=a=a\star_{x}a, 
\qquad 
I\star_{x}a=\mathcal{S}_{0}a=a. 
\] 
If $a \in \mathcal{N}_{BMI}$, the selected contractions are 
\[ 
Ma \mathrel{\rightarrow_{\mathrm{li}}} aa, 
\qquad 
Ia \mathrel{\rightarrow_{\mathrm{li}}} a. 
\] 
Thus \prettyref{lem:prefix} and the definition of $\star_{x}$ give 
\begin{align*} 
M\star_{x}a 
&= \mathrm{Tr}_{x}\left(Ma\right) 
= \mathrm{Tr}_{x}\left(aa\right) 
= a\star_{x}a, \\ 
I\star_{x}a 
&= \mathrm{Tr}_{x}\left(Ia\right) 
= \mathrm{Tr}_{x}\left(a\right) 
= a. \qedhere 
\end{align*}

Let $x \in \mathcal{V}$.

\label{thm:main} 
Let $x \in \mathcal{V}$ and $r \geq 1$. 
For every finite $x$--free term $Y \in \mathcal{T}_{BMI}$, 
\[ 
Yx \mathrel{\neq_{w}} x^{r}\left(Yx\right). 
\] 
In particular, $BMI$--combinatory logic has no fixed-point combinator under weak equivalence $\mathrel{=_{w}}$.

<!-- lem-5: body_md -->
\label{lem:rightmost} 
For every $u$ such that $Px \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} u$, the rightmost leaf of $u$ is labelled $x$, and every redex occurrence in $u$ contains that leaf.
<!-- end lem-5 -->
<!-- proof-8: body_md -->
We argue by induction on the number of $\mathrel{\rightarrow_{\mathrm{li}}}$--steps. 
The assertion holds for $Px$: its rightmost leaf is $x$, and every subterm not containing this leaf is a subterm of the normal term $P$. 
 
Suppose that the assertion holds for $u$ and that 
\[ 
u = \Gamma\left[R\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[R'\right]=u'. 
\] 
Since $R$ is a redex and therefore, not normal, it contains the rightmost leaf of $u$ by the induction hypothesis. 
Moreover, every proper subterm of $R$ is normal because the contraction is innermost. 
There are three cases. 
\begin{enumerate} 
 
\item The contraction is 
\[ 
\Gamma\left[Babc\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[a\left(bc\right)\right]. 
\] 
The rightmost leaf lies in $c$, which is retained unchanged. 
The only new subterms appearing on the contractum are $bc$ and $a(bc)$, both of which contain this leaf. 
Every other subterm within the contractum is a subterm of one of the normal terms $a,b,c$. 
 
\item The contraction is 
\[ 
\Gamma\left[Ma\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[a_{L}a_{R}\right], 
\] 
where $a_{L}$ and $a_{R}$ denote the two occurrences of the same normal term $a$. 
The rightmost leaf of $a$ is labelled $x$, so the new rightmost leaf is the final $x$ in $a_{R}$. 
Both copies of $a$ are normal, and the new subterm $a_{L}a_{R}$ contains the new rightmost leaf. 
 
\item The contraction is 
\[ 
\Gamma\left[Ia\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[a\right]. 
\] 
The normal term $a$ is retained unchanged and contains the rightmost leaf. 
\end{enumerate} 
In each case, subterms disjoint from the contracted occurrence are unchanged, while every ancestor of that occurrence still contains the rightmost leaf. 
Thus the assertion also holds for $u'$.

<!-- lem-6: body_md -->
\label{lem:increase} 
If $MA, MA'$ are the redexes at two consecutive $M$--contractions in the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $Px$, then 
\[ 
\left|A'\right|_{x} \geq \left|A\right|_{x}+1. 
\]
<!-- end lem-6 -->
<!-- proof-9: body_md -->
Write the first of the two $M$--contractions as 
\[ 
\Gamma\left[MA\right] \mathrel{\rightarrow_{\mathrm{li}}} \Gamma\left[AA\right]. 
\] 
Since the contraction is innermost, $A$ is a normal form. 
By \prettyref{lem:rightmost}, the redex $MA$ contains the rightmost leaf of the whole term, which is labelled $x$. 
Thus the rightmost leaf of $A$ is labelled $x$. 
 
Only $B$-- and $I$--contractions occur between the two consecutive $M$--contractions. 
We prove, by induction on the number of these intervening contractions, that every term between the consecutive $M$--contractions has the form 
\[ 
U_{1}\left(U_{2}\left(\cdots\left(U_{k}A\right)\cdots\right)\right), 
\] 
where $k\geq 1$, all the $U_{i}$ are normal, and the rightmost leaf of $U_{k}$ is labelled $x$. 
 
For the initial term $\Gamma\left[AA\right]$, observe that the path from the root of $\Gamma\left[MA\right]$ to the redex $MA$ takes only right-child edges, since $MA$ contains the rightmost leaf of $\Gamma\left[MA\right]$. 
Each left subtree along this path is normal, because $MA$ is the leftmost-innermost redex of $\Gamma\left[MA\right]$. 
Taking these left subterms in order from the root to $MA$ as $U_{1},\ldots,U_{m}$, we have 
\[ 
\Gamma\left[-\right] = U_{1}\left(U_{2}\left(\cdots\left(U_{m} {-}\right)\cdots\right)\right), 
\] 
where $m=0$ means $\Gamma\left[-\right] = {-}$. 
Therefore, 
\[ 
\Gamma\left[AA\right] = U_{1}\left(U_{2}\left(\cdots\left(U_{m}\left(AA\right)\right)\cdots\right)\right). 
\] 
Setting $k := m+1$ and $U_{k} := A$ gives the required form. 
 
For the induction step, suppose that the current term has the form: 
\[ 
U_{1}\left(U_{2}\left(\cdots\left(U_{k}A\right)\cdots\right)\right). 
\] 
Every subterm of the current term is either a subterm of one of $U_{1},\ldots,U_{k},A$ or has the form 
\[ 
U_{i}\left(\cdots\left(U_{k}A\right)\cdots\right) \qquad \left(1\leq i\leq k\right). 
\] 
Since $U_{1},\ldots,U_{k},A$ are all normal, the redex that is contracted at this step must have the latter form. 
Write this redex as $U_{j}T$. 
 
For a $B$--contraction, we have $U_{j} = Bab$, and the contraction replaces $BabT$ with $a\left(bT\right)$. 
Thus the factor $U_{j}$ is replaced with the two factors $a,b$. 
The new factors $a$ and $b$ are normal, since $U_{j}$ is normal. 
If $j < k$, the last factor $U_{k}$ is unchanged. 
If $j = k$, the new last factor is $b$, whose rightmost leaf 
is the same as the rightmost leaf of $Bab=U_{k}$ and is therefore labelled $x$. 
Hence the required form is preserved. 
 
For an $I$--contraction, we have $U_{j} = I$, and the contraction replaces $IT$ with $T$, just deleting the factor $U_{j}$. 
Since the rightmost leaf of $U_{k}$ is labelled $x$, we have 
$U_{k} \neq I$ and hence $j < k$. 
The last factor is therefore unchanged, and at least one factor remains. 
This completes the induction. 
 
At the second $M$--contraction, the selected redex has the form 
\[ 
MA' = U_{j}\left(\cdots\left(U_{k}A\right)\cdots\right) 
\] 
for some $1\leq j\leq k$. 
Thus $U_{j} = M$. 
Since the rightmost leaf of $U_{k}$ is labelled $x$, we have $U_{k}\neq M$ and hence $j < k$. 
Consequently, 
\[ 
A' = U_{j+1}\left(\cdots\left(U_{k}A\right)\cdots\right), 
\] 
and therefore, 
\begin{align*} 
\left|A'\right|_{x} 
&= \left|A\right|_{x} + \sum_{i=j+1}^{k}\left|U_{i}\right|_{x} \\ 
&\geq \left|A\right|_{x} + \left|U_k\right|_{x} \\ 
&\geq \left|A\right|_{x} + 1. \qedhere 
\end{align*}
<!-- end proof-9 -->
<!-- cor-1: body_md -->
\label{cor:increasing} 
If $Px$ has an infinite $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence and 
\[ 
\tau_{x}\left(Px\right) = \left(\left(p_{i}, q_{i}\right)\right)_{i \in \mathbb{N}}, 
\] 
then $q_{0} < q_{1} < q_{2} < \cdots$.
<!-- end cor-1 -->
<!-- proof-10: body_md -->
By \prettyref{lem:infinite-events}, there are infinitely many $M$--contractions. 
Let $A_i$ be the argument duplicated at the $i$--th $M$--contraction. 
By definition, $q_{i} = \left|A_{i}\right|_{x}$. 
Applying \prettyref{lem:increase} to each consecutive pair gives 
\[ 
q_{i+1} = \left|A_{i+1}\right|_{x} \geq \left|A_{i}\right|_{x} + 1 = q_{i} + 1 \qquad \left(i\in\mathbb{N}\right).\qedhere 
\]
<!-- end proof-10 -->
<!-- lem-7: body_md -->
\label{lem:separation} 
Let $\alpha := \left(\left(p_{i}, q_{i}\right)\right)_{i\in\mathbb{N}} \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}$ and suppose the second coordinates of $\alpha$ are strictly increasing. 
Then 
\[ 
\alpha \not\mathrel{\sim_{\mathrm{tail}}} S_{r}\alpha 
\] 
for every $r\in\mathbb{N}$ with $r\geq1$.
<!-- end lem-7 -->
<!-- proof-11: body_md -->
Suppose that $\alpha\mathrel{\sim_{\mathrm{tail}}} S_{r}\alpha$. 
By the definition of tail equivalence, there exist $i,j\in\mathbb{N}$ such that 
\[ 
\left(p_{i+k}, q_{i+k}\right) = \left(p_{j+k} + r, q_{j+k}\right) \qquad \left(k\in\mathbb{N}\right). 
\] 
Taking $k=0$ in the second coordinate gives $q_{i} = q_{j}$. 
Since the $q$--coordinate is strictly increasing, $i = j$, so the first coordinate gives $p_{i} = p_{i} + r$. 
This contradicts $r\geq 1$.
<!-- end proof-11 -->
<!-- lem-8: body_md -->
\label{lem:head-xs-shift-trace} 
For every $u\in\mathcal{T}_{BMI}$ and $r\in\mathbb{N}$, 
\[ 
\mathrm{Tr}_{x}\left(x^{r}u\right) = \begin{cases} 
x^{r}N, & \mathrm{Tr}_{x}\left(u\right)=N\in\mathcal{N}_{BMI},\\ 
\left[S_r\alpha\right] = \mathcal{S}_{r}\left[\alpha\right], & \mathrm{Tr}_{x}\left(u\right)=\left[\alpha\right] \in \left(\mathbb{N}^{2}\right)^{\mathbb{N}}/\mathrel{\sim_{\mathrm{tail}}}. 
\end{cases} 
\]
<!-- end lem-8 -->
<!-- proof-12: body_md -->
Note that the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $x^{r}u$ is obtained from that of $u$ by placing each term in the context $x^{r}\left(-\right)$. 
 
Suppose first that $u\mathrel{\rightarrow_{\mathrm{li}}}^{\ast}N$ with $N$ normal. 
Then $x^{r}u\mathrel{\rightarrow_{\mathrm{li}}}^{\ast}x^{r}N$, and $x^{r}N$ is normal. 
 
Now suppose that the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $u$ is infinite, and write it as $u=u_{0}\mathrel{\rightarrow_{\mathrm{li}}} u_{1}\mathrel{\rightarrow_{\mathrm{li}}}\cdots$. 
The $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence starting from $x^{r}u$ is 
\[ 
x^{r}u = x^{r}u_{0} \mathrel{\rightarrow_{\mathrm{li}}} x^{r}u_{1} \mathrel{\rightarrow_{\mathrm{li}}} \cdots. 
\] 
Each contraction $\Gamma\left[MA\right]\mathrel{\rightarrow_{\mathrm{li}}}\Gamma\left[AA\right]$ corresponds to 
\[ 
x^{r}\left(\Gamma\left[MA\right]\right) \mathrel{\rightarrow_{\mathrm{li}}} x^{r}\left(\Gamma\left[AA\right]\right). 
\] 
The duplicated argument is unchanged and exactly $r$ additional occurrences of $x$ precede the redex. 
Thus 
\[ 
\tau_{x}\left(x^{r}u\right) = S_{r}\tau_{x}\left(u\right), 
\] 
which proves the second clause.
<!-- end proof-12 -->
<!-- proof-13: body_md -->
Fix an $x$--free term $Y\in\mathcal{T}_{BMI}$ and $r\geq 1$. 
By \prettyref{thm:invariance}, it suffices to show that 
\[ 
\mathrm{Tr}_{x}\left(Yx\right) \neq \mathrm{Tr}_{x}\left(x^{r}\left(Yx\right)\right). 
\] 
Every reduct of an $x$--free term is $x$--free, since the $BMI$--contraction rules introduce no new variables. 
There are three cases: 
\begin{enumerate} 
 
\item The $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $Y$ is infinite. 
At each $M$--contraction in this sequence, both the duplicated argument and the whole term are $x$--free. 
Hence 
\[ 
\tau_{x}\left(Y\right) = \zeta_{0}, 
\] 
where 
\[ 
\zeta_{s} := \left(\left(s,0\right)\right)_{i\in\mathbb{N}}. 
\] 
By \prettyref{thm:homomorphism} and the first clause in the definition of $\star_x$, 
\[ 
\mathrm{Tr}_{x}\left(Yx\right) =\left[\zeta_{0}\right]\star_{x} x =\left[\zeta_{0}\right]. 
\] 
On the other hand, \prettyref{lem:head-xs-shift-trace} gives 
\[ 
\mathrm{Tr}_{x}\left(x^{r}\left(Yx\right)\right) = \left[S_{r}\zeta_{0}\right] =\left[\zeta_{r}\right]. 
\] 
Since $r\geq 1$, no entry of $\zeta_{0}$ equals an entry of $\zeta_{r}$, so $\zeta_{0} \not\mathrel{\sim_{\mathrm{tail}}} \zeta_{r}$. 
Therefore, $\mathrm{Tr}_{x}\left(Yx\right) \neq \mathrm{Tr}_{x}\left(x^{r}\left(Yx\right)\right)$. 
 
\item $Y\mathrel{\rightarrow_{\mathrm{li}}}^{\ast}P$ and $Px\mathrel{\rightarrow_{\mathrm{li}}}^{\ast}N$ for some $P,N\in\mathcal{N}_{BMI}$. 
The term $P$ is $x$--free, and 
\[ 
Yx \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} Px \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} N. 
\] 
By \prettyref{lem:prefix} and \prettyref{lem:head-xs-shift-trace}, 
\[ 
\mathrm{Tr}_{x}\left(Yx\right)=N, \qquad 
\mathrm{Tr}_{x}\left(x^{r}\left(Yx\right)\right)=x^{r}N. 
\] 
These are distinct normal-form terms, because 
\[ 
\left|x^{r}N\right|_{x} = \left|N\right|_{x} + r > \left|N\right|_{x}. 
\] 
 
\item $Y\mathrel{\rightarrow_{\mathrm{li}}}^{\ast}P$ for some $P\in\mathcal{N}_{BMI}$, and the $\mathrel{\rightarrow_{\mathrm{li}}}$--sequence of $Px$ is infinite. 
The term $P$ is $x$--free. 
Put $\alpha := \tau_{x}\left(Px\right)$. 
Since $Yx \mathrel{\rightarrow_{\mathrm{li}}}^{\ast} Px$, \prettyref{lem:prefix} and \prettyref{lem:head-xs-shift-trace} give 
\[ 
\mathrm{Tr}_{x}\left(Yx\right)=\left[\alpha\right], \qquad 
\mathrm{Tr}_{x}\left(x^{r}\left(Yx\right)\right)=\left[S_{r}\alpha\right]. 
\] 
The second coordinates of $\alpha$ are strictly increasing by \prettyref{cor:increasing}. 
Therefore $\alpha\not\mathrel{\sim_{\mathrm{tail}}} S_{r}\alpha$ by \prettyref{lem:separation}, so the two invariants are distinct. 
\end{enumerate} 
Thus the required inequality holds in every case.

<!-- cor-2: body_md -->
\label{cor:BM} 
Let $x \in \mathcal{V}$ and $r \geq 1$. 
For every finite $x$--free term $Y \in \mathcal{T}_{BM}$, 
\[ 
Yx \mathrel{\neq_{w}} x^{r}\left(Yx\right). 
\] 
In particular, $BM$--combinatory logic has no fixed-point combinator under weak equivalence $\mathrel{=_{w}}$.
