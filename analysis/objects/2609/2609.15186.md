<!-- remark-1: body_md -->
\label{rmk:symmetry-and-homogeneous-BC}
    Symmetry propagates through the system: the symmetry of $\nabla^{\mathrm{s}} \bm{u}$ implies $\widetilde{\bm{\sigma}}^{\mathsf{T}} = \widetilde{\bm{\sigma}}$ via \eqref{eq:first-order-SGE-1}, which in turn yields $\bm{\chi}_{ij\bullet} = \bm{\chi}_{ji\bullet}$ and $\bm{\sigma}^{\mathsf{T}} = \bm{\sigma}$ through \eqref{eq:first-order-SGE-2} and \eqref{eq:first-order-SGE-3}, respectively. Moreover, the homogeneous boundary conditions \eqref{eq:homogeneous-BC} imply $\bm{\varepsilon} = \bm{0}$ and hence $\widetilde{\bm{\sigma}} = \bm{0}$ on $\partial \Omega$.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
The auxiliary variables admit a direct mechanical interpretation. In the general strain gradient elasticity theory, the material response is described by two stress measures, the Cauchy stress $\bm{\sigma}^c \triangleq \bm{C} 
  \mathbin{\vcenter{\offinterlineskip
    \hbox{$\cdot$}\kern-0.2ex
    \hbox{$\cdot$}}}
 \bm{\varepsilon}$ and the hyper stress $\bm{\tau} \triangleq \iota^2 \nabla \bm{\sigma}^c$. By \eqref{eq:auxiliary-variables}, $\widetilde{\bm{\sigma}} = \mathcal{A}^{1/2} \bm{\sigma}^c$ and $\bm{\chi} = \iota^{-1} \mathcal{A}^{1/2} \bm{\tau}$, so that $\widetilde{\bm{\sigma}}$ and $\bm{\chi}$ are scaled versions of the Cauchy stress and the hyper stress, where the factor $\iota^{-1}$ balances the magnitude of the hyper stress. As $\lambda \to \infty$, the energy norm induced by $\mathcal{A}$ degenerates to its deviatoric part alone; the scaling by $\mathcal{A}^{1/2}$ confines this degeneration to the total stress $\bm{\sigma}$, where a discrete Korn inequality recovers full $L^2$ control, as detailed in \cref{sec:4}.
<!-- end remark-2 -->
<!-- lemma-1: body_md -->
\label{lem:SV}
    Let
    \begin{align*}
        W_h &\triangleq \{ \bm{w}_h \in H^1 (\Omega; \mathbb{R}^2) : \bm{w}_h|_K \in \mathcal{P}_{k+1} (K; \mathbb{R}^2),\ \forall K \in \mathcal{T}_h \}, \\
        Q_h &\triangleq \{ q_h \in L^2 (\Omega) : q_h|_K \in \mathcal{P}_k (K),\ \forall K \in \mathcal{T}_h \}.
    \end{align*}
    If $k \geq 1$, then $\operatorname{div} : W_h \to Q_h$ is surjective and admits a bounded right inverse: for every $q_h \in Q_h$ there exists $\bm{w}_h \in W_h$ with $\operatorname{div} \bm{w}_h = q_h$ and $\lVert \bm{w}_h \rVert_{H^1} \lesssim \lVert q_h \rVert$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Set $W_{h, 0} \triangleq \{ \bm{w}_h \in W_h : \bm{w}_h|_{\partial \Omega} = \bm{0} \}$ and $Q_{h, 0} \triangleq \{ q_h \in Q_h : \int_\Omega q_h = 0 \}$. Since $\mathcal{T}_h$ is the Alfeld split of a shape-regular triangulation, the theory of the Scott--Vogelius pair \cite{guzman2018scottvogelius} shows that for every $q^0_h \in Q_{h, 0}$ there exists $\bm{w}^0_h \in W_{h, 0}$ with $\operatorname{div} \bm{w}^0_h = q^0_h$ and $\lVert \bm{w}^0_h \rVert_{H^1} \lesssim \lVert q^0_h \rVert$.

    To remove the zero-mean restriction, given $q_h \in Q_h$ set $c \triangleq |\Omega|^{-1} \int_\Omega q_h$ and $\bm{w}^1_h \triangleq \frac{c}{2} \bm{x} \in W_h$, so that $\operatorname{div} \bm{w}^1_h = c$ and $\lVert \bm{w}^1_h \rVert_{H^1} \lesssim \lVert q_h \rVert$. Since $q_h - c \in Q_{h, 0}$, the above provides $\bm{w}^0_h \in W_{h, 0}$ with $\operatorname{div} \bm{w}^0_h = q_h - c$ and $\lVert \bm{w}^0_h \rVert_{H^1} \lesssim \lVert q_h - c \rVert \lesssim \lVert q_h \rVert$. The field $\bm{w}_h \triangleq \bm{w}^0_h + \bm{w}^1_h \in W_h$ then satisfies $\operatorname{div} \bm{w}_h = q_h$ and $\lVert \bm{w}_h \rVert_{H^1} \lesssim \lVert q_h \rVert$.

<!-- lemma-2: body_md -->
\label{lem:symmetrization}
    There exists an operator $\mathcal{S} : \Sigma_h \to \Sigma^s_h$ such that
    \begin{align}
        \lVert \mathcal{S} \bm{\tau}_h \rVert &\lesssim \lVert \bm{\tau}_h \rVert, & &\forall \bm{\tau}_h \in \Sigma_h, \label{eq:symmetrization-1} \\
        b^*_h (\bm{v}_h, \mathcal{S} \bm{\tau}_h) &= b^*_h (\bm{v}_h, \bm{\tau}_h), & &\forall (\bm{v}_h, \bm{\tau}_h) \in U_{h, 0} \times \Sigma_h. \label{eq:symmetrization-2}
    \end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Fix $\bm{\tau}_h \in \Sigma_h$, set $q_h \triangleq (\bm{\tau}_h)_{21} - (\bm{\tau}_h)_{12}$, and let $\bm{w}_h \in W_h$ be as in \cref{lem:SV}, so that $\operatorname{div} \bm{w}_h = q_h$ and $\lVert \bm{w}_h \rVert_{H^1} \lesssim \lVert q_h \rVert$. Define $\mathcal{S} \bm{\tau}_h \triangleq \bm{\tau}_h + \operatorname{curl} \bm{w}_h$.

    Each entry of $\operatorname{curl} \bm{w}_h$ is piecewise $\mathcal{P}_k$, and $\bm{w}_h \in H^1 (\Omega; \mathbb{R}^2)$ implies $\operatorname{curl} \bm{w}_h \in H(\operatorname{div}; \Omega)$; hence $\mathcal{S} \bm{\tau}_h \in \Sigma_h$. Since $(\operatorname{curl} \bm{w}_h)_{12} - (\operatorname{curl} \bm{w}_h)_{21} = \operatorname{div} \bm{w}_h$,
    \begin{align*}
        (\mathcal{S} \bm{\tau}_h)_{12} - (\mathcal{S} \bm{\tau}_h)_{21} = \left[ (\bm{\tau}_h)_{12} - (\bm{\tau}_h)_{21} \right] + \operatorname{div} \bm{w}_h = -q_h + q_h = 0,
    \end{align*}
    and therefore $\mathcal{S} \bm{\tau}_h \in \Sigma^s_h$. Property \eqref{eq:symmetrization-1} follows from
    \begin{align*}
        \lVert \mathcal{S} \bm{\tau}_h \rVert \leq \lVert \bm{\tau}_h \rVert + \lVert \operatorname{curl} \bm{w}_h \rVert \lesssim \lVert \bm{\tau}_h \rVert + \lVert \bm{w}_h \rVert_{H^1} \lesssim \lVert \bm{\tau}_h \rVert + \lVert q_h \rVert \lesssim \lVert \bm{\tau}_h \rVert.
    \end{align*}
    For \eqref{eq:symmetrization-2}, the conformity $\operatorname{curl} \bm{w}_h \in H(\operatorname{div}; \Omega)$ gives $\llbracket \operatorname{curl} \bm{w}_h \cdot \bm{n} \rrbracket_{|_e} = \bm{0}$ across every interior edge, while $\operatorname{div} (\operatorname{curl} \bm{w}_h) = \bm{0}$ pointwise. Consequently, for every $\bm{v}_h \in U_{h, 0}$,
    \begin{align*}
        b^*_h (\bm{v}_h, \operatorname{curl} \bm{w}_h) = - (\bm{v}_h, \operatorname{div} (\operatorname{curl} \bm{w}_h))_{\mathcal{T}_h} + (\bm{v}_h, \llbracket \operatorname{curl} \bm{w}_h \cdot \bm{n} \rrbracket)_{\mathcal{F}^{pr}_h} = 0,
    \end{align*}
    which follows \eqref{eq:symmetrization-2}.
<!-- end proof-2 -->
<!-- theorem-1: body_md -->
\label{thm:inf-sup-b*_h-sym}
    There exists a constant $\beta > 0$ independent of the mesh size $h$ such that
    \begin{align}
        \sup_{\bm{\tau}_h \in \Sigma^s_h \setminus \{ \bm{0} \}} \frac{b^*_h (\bm{v}_h, \bm{\tau}_h)}{\lVert \bm{\tau}_h \rVert} &\geq \beta \lVert \bm{v}_h \rVert_{H^1, h}, & &\forall \bm{v}_h \in U_{h, 0}. \label{eq:inf-sup-b*_h-sym}
    \end{align}
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
It suffices to show that every $\bm{v}_h \in U_{h, 0}$ admits some $\bm{\tau}^s_h \in \Sigma^s_h$ with $b^*_h (\bm{v}_h, \bm{\tau}^s_h) \gtrsim \lVert \bm{v}_h \rVert^2_{H^1, h}$ and $\lVert \bm{\tau}^s_h \rVert \lesssim \lVert \bm{v}_h \rVert_{H^1, h}$. By \eqref{eq:inf-sup-b*_h} there exists $\bm{\tau}_h \in \Sigma_h$ with
    \begin{align*}
        b^*_h (\bm{v}_h, \bm{\tau}_h) \gtrsim \lVert \bm{v}_h \rVert^2_{H^1, h}, \qquad
        \lVert \bm{\tau}_h \rVert \lesssim \lVert \bm{v}_h \rVert_{H^1, h}.
    \end{align*}
    Set $\bm{\tau}^s_h \triangleq \mathcal{S} \bm{\tau}_h \in \Sigma^s_h$ by \cref{lem:symmetrization}. The invariance \eqref{eq:symmetrization-2} preserves the first estimate, and the boundedness \eqref{eq:symmetrization-1} preserves the second.
<!-- end proof-3 -->
<!-- lemma-3: body_md -->
\label{lem:korn-ineq-Sigma_h}
    For all $\bm{\tau}_h \in \Sigma_h$,
    \begin{align}
        \lVert \bm{\tau}_h \rVert_{L^2_*} &\lesssim \lVert \operatorname{dev} \bm{\tau}_h \rVert + \lVert \bm{\tau}_h \rVert_{\operatorname{div}, h}. \label{eq:korn-ineq-Sigma_h}
    \end{align}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Fix $\bm{\tau}_h \in \Sigma_h$. We set $c \triangleq \frac{1}{2 \lvert \Omega \rvert} \int_\Omega \operatorname{tr} \bm{\tau}_h$ and $q_h \triangleq \operatorname{tr} \bm{\tau}_h - 2 c$, so that $\bm{\tau}_h - c \bm{I} = \operatorname{dev} \bm{\tau}_h + \tfrac{1}{2} q_h \bm{I}$. By the definition of the quotient norm,
    \begin{align}
        \lVert \bm{\tau}_h \rVert_{L^2_*} = \lVert \bm{\tau}_h - c \bm{I} \rVert \lesssim \lVert \operatorname{dev} \bm{\tau}_h \rVert + \lVert q_h \rVert. \label{prf:korn-ineq-Sigma_h-1}
    \end{align}
    It remains to bound $\lVert q_h \rVert$. Clearly $q_h \in L^2_0 (\Omega)$, and since $\Omega$ is a bounded convex polygon, the divergence operator maps $H^1_0 (\Omega; \mathbb{R}^2)$ onto $L^2_0 (\Omega)$ with a bounded right inverse \cite{girault1986finite}: there exists $\bm{w} \in H^1_0 (\Omega; \mathbb{R}^2)$ with $\operatorname{div} \bm{w} = q_h$ and $\lVert \bm{w} \rVert_{H^1} \lesssim \lVert q_h \rVert$. Using $\int_\Omega \operatorname{div} \bm{w} = \int_{\partial \Omega} \bm{w} \cdot \bm{n} = 0$,
    \begin{align*}
        \lVert q_h \rVert^2 = (q_h, \operatorname{div} \bm{w}) = \left( (\operatorname{tr} \bm{\tau}_h) \bm{I}, \nabla \bm{w} \right) = 2 (\bm{\tau}_h, \nabla \bm{w}) - 2 (\operatorname{dev} \bm{\tau}_h, \nabla \bm{w}).
    \end{align*}
    Integrating by parts elementwise and using $\llbracket \bm{\tau}_h \cdot \bm{n} \rrbracket_{|_e} = \bm{0}$ for $e \in \mathcal{F}^{dl}_h$ together with $\bm{w} = \bm{0}$ on $\partial \Omega$, we obtain
    \begin{align*}
        (\bm{\tau}_h, \nabla \bm{w}) = - (\operatorname{div} \bm{\tau}_h, \bm{w})_{\mathcal{T}_h} + (\llbracket \bm{\tau}_h \cdot \bm{n} \rrbracket, \bm{w})_{\mathcal{F}^{pr, o}_h} \lesssim \lVert \bm{\tau}_h \rVert_{\operatorname{div}, h} \lVert \bm{w} \rVert_{L^2, h},
    \end{align*}
    while the trace inequality \cite{ern2004theory} gives $\lVert \bm{w} \rVert_{L^2, h} \lesssim \lVert \bm{w} \rVert_{H^1}$. Combining the last two displays yields $\lVert q_h \rVert^2 \lesssim ( \lVert \bm{\tau}_h \rVert_{\operatorname{div}, h} + \lVert \operatorname{dev} \bm{\tau}_h \rVert ) \lVert q_h \rVert$; dividing by $\lVert q_h \rVert$ and inserting the result into \eqref{prf:korn-ineq-Sigma_h-1} gives \eqref{eq:korn-ineq-Sigma_h}.
<!-- end proof-4 -->
<!-- corollary-1: body_md -->
\label{cor:L2-control-Sigma_h}
    For all $\bm{\tau}_h \in \Sigma_h$,
    \begin{align}
        \lVert \bm{\tau}_h \rVert^2_{L^2_*} \lesssim (\mathcal{A}^{\frac{1}{2}} \bm{\tau}_h, \bm{\tau}_h - c \bm{I}) + \lVert \bm{\tau}_h \rVert^2_{\operatorname{div}, h}, \label{eq:L2-control-Sigma_h}
    \end{align}
    where $c \triangleq \frac{1}{2 \lvert \Omega \rvert} \int_\Omega \operatorname{tr} \bm{\tau}_h$.
<!-- end corollary-1 -->
<!-- proof-5: body_md -->
With $q_h \triangleq \operatorname{tr} \bm{\tau}_h - 2 c \in L^2_0 (\Omega)$ as in the proof of \cref{lem:korn-ineq-Sigma_h}, we have $\bm{\tau}_h - c \bm{I} = \operatorname{dev} \bm{\tau}_h + \tfrac{1}{2} q_h \bm{I}$ and $(\operatorname{tr} \bm{\tau}_h, q_h) = (2 c + q_h, q_h) = \lVert q_h \rVert^2$. Recalling the definition \eqref{eq:A^1/2} of $\mathcal{A}^{1/2}$ and using the $L^2$-orthogonality of $\operatorname{dev} \bm{\tau}_h$ and $\bm{I}$,
    \begin{align*}
        (\mathcal{A}^{\frac{1}{2}} \bm{\tau}_h, \bm{\tau}_h - c \bm{I}) = \frac{1}{\sqrt{2\mu}} \lVert \operatorname{dev} \bm{\tau}_h \rVert^2 + \frac{1}{2 \sqrt{2\lambda + 2\mu}} \lVert q_h \rVert^2 \geq \frac{1}{\sqrt{2\mu}} \lVert \operatorname{dev} \bm{\tau}_h \rVert^2,
    \end{align*}
    which, together with \eqref{eq:korn-ineq-Sigma_h}, yields \eqref{eq:L2-control-Sigma_h}.
<!-- end proof-5 -->
<!-- theorem-2: body_md -->
\label{thm:stability}
    Let $(\bm{u}_h, \bm{\sigma}_h, \widetilde{\bm{\sigma}}_h, \bm{\chi}_h) \in U_{h, 0} \times \Sigma^s_h \times \widetilde{\Sigma}^s_h \times X^s_h$ solve the discrete scheme \eqref{eq:discrete-scheme-1}--\eqref{eq:discrete-scheme-4}. Then
    \begin{align}
        \lVert \bm{u}_h \rVert + \lVert \bm{u}_h \rVert_{H^1, h} + \lVert \widetilde{\bm{\sigma}}_h \rVert + \lVert \bm{\chi}_h \rVert &\lesssim \lVert \bm{f} \rVert, \label{eq:stability-1} \\
        \lVert \bm{\sigma}_h \rVert_{L^2_*} &\lesssim \left( 1 + \tfrac{\iota}{h} \right) \lVert \bm{f} \rVert. \label{eq:stability-2}
    \end{align}
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
Taking $\bm{\tau}_h = \bm{\sigma}_h$, $\bm{\psi}_h = \bm{\chi}_h$, $\widetilde{\bm{\tau}}_h = \widetilde{\bm{\sigma}}_h$ and $\bm{v}_h = \bm{u}_h$ in \eqref{eq:discrete-scheme-1}--\eqref{eq:discrete-scheme-4} and summing gives
    \begin{align}
        \lVert \widetilde{\bm{\sigma}}_h \rVert^2 + \lVert \bm{\chi}_h \rVert^2 = (\bm{f}, \bm{u}_h) \leq \lVert \bm{f} \rVert \lVert \bm{u}_h \rVert, \label{prf:stability-1}
    \end{align}
    while the inf-sup condition \eqref{eq:inf-sup-b*_h-sym} and \eqref{eq:discrete-scheme-1} give
    \begin{align}
        \lVert \bm{u}_h \rVert_{H^1, h} \lesssim \sup_{\bm{\tau}_h \in \Sigma^s_h \setminus \{ \bm{0} \}} \frac{b^*_h (\bm{u}_h, \bm{\tau}_h)}{\lVert \bm{\tau}_h \rVert} = \sup_{\bm{\tau}_h \in \Sigma^s_h \setminus \{ \bm{0} \}} \frac{(\mathcal{A}^{\frac{1}{2}} \widetilde{\bm{\sigma}}_h, \bm{\tau}_h)}{\lVert \bm{\tau}_h \rVert} \lesssim \lVert \widetilde{\bm{\sigma}}_h \rVert. \label{prf:stability-2}
    \end{align}
    Combining \eqref{prf:stability-1} and \eqref{prf:stability-2} with the discrete Poincar\'e inequality \eqref{eq:poincare-ineq-U_h} yields \eqref{eq:stability-1}.

    For \eqref{eq:stability-2}, let $c \triangleq \frac{1}{2 \lvert \Omega \rvert} \int_\Omega \operatorname{tr} \bm{\sigma}_h$, so that \cref{cor:L2-control-Sigma_h} gives
    \begin{align}
        \lVert \bm{\sigma}_h \rVert^2_{L^2_*} \lesssim (\mathcal{A}^{\frac{1}{2}} \bm{\sigma}_h, \bm{\sigma}_h - c \bm{I}) + \lVert \bm{\sigma}_h \rVert^2_{\operatorname{div}, h}. \label{prf:stability-3}
    \end{align}
    Since $\Sigma^s_h \subset \widetilde{\Sigma}^s_h$, we may take $\widetilde{\bm{\tau}}_h = \bm{\sigma}_h - c \bm{I}$ in \eqref{eq:discrete-scheme-3}; together with the inverse inequality \eqref{eq:inverse-ineq-X_h} this bounds the first term,
    \begin{align}
        (\mathcal{A}^{\frac{1}{2}} \bm{\sigma}_h, \bm{\sigma}_h - c \bm{I}) = (\widetilde{\bm{\sigma}}_h + \iota \operatorname{div} \bm{\chi}_h, \bm{\sigma}_h - c \bm{I}) \lesssim \left( \lVert \widetilde{\bm{\sigma}}_h \rVert + \tfrac{\iota}{h} \lVert \bm{\chi}_h \rVert \right) \lVert \bm{\sigma}_h \rVert_{L^2_*}, \label{prf:stability-4}
    \end{align}
    while the inf-sup condition \eqref{eq:inf-sup-b_h} and \eqref{eq:discrete-scheme-4} bound the second,
    \begin{align}
        \lVert \bm{\sigma}_h \rVert_{\operatorname{div}, h} \lesssim \sup_{\bm{v}_h \in U_{h, 0} \setminus \{ \bm{0} \}} \frac{b_h (\bm{\sigma}_h, \bm{v}_h)}{\lVert \bm{v}_h \rVert} = \sup_{\bm{v}_h \in U_{h, 0} \setminus \{ \bm{0} \}} \frac{(\bm{f}, \bm{v}_h)}{\lVert \bm{v}_h \rVert} \lesssim \lVert \bm{f} \rVert. \label{prf:stability-5}
    \end{align}
    Substituting \eqref{prf:stability-4} and \eqref{prf:stability-5} into \eqref{prf:stability-3} and invoking \eqref{eq:stability-1} gives \eqref{eq:stability-2}.
<!-- end proof-6 -->
<!-- remark-3: body_md -->
Compared with \eqref{eq:stability-1}, the stability estimate \eqref{eq:stability-2} for the total stress carries the extra factor $\iota/h$; since $\iota$ is typically a small parameter in applications, this factor remains under control and the estimate is still meaningful in practice.

<!-- remark-4: body_md -->
\label{rmk:incompressible-limit}
    If $\lambda$ is bounded above, i.e., $\lambda \lesssim 1$, then taking $\widetilde{\bm{\tau}}_h = \bm{\sigma}_h$ in \eqref{eq:discrete-scheme-3} and arguing as in the proof of \cref{thm:stability} improves the stability estimate \eqref{eq:stability-2} for the total stress to the $L^2$ norm. If $\lambda$ is not bounded above, however, the stability of the total stress degenerates in the direction of the constant spherical modes $c \bm{I}$, $c \in \mathbb{R}$, as it does in the Hellinger--Reissner formulation of linear elasticity. Indeed, as $\lambda \to \infty$ the material approaches the incompressible limit and the total stress is determined only up to such a mode, so that an additional normalization, for instance $\int_\Omega \operatorname{tr} \bm{\sigma}_h = 0$, is required to fix it.

<!-- lemma-4: body_md -->
\label{lem:proj-est-Pi-Sigma-s}
    For any symmetric $\bm{\sigma} \in H^r (\Omega; \mathbb{R}^{2 \times 2})$ with $1 \leq r \leq k + 1$,
    \begin{align}
        \lVert \bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma} \rVert &\lesssim h^r \lVert \bm{\sigma} \rVert_r, \label{eq:proj-est-Pi-Sigma-s} \\
        b_h (\bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma}, \bm{v}_h) &= 0, & &\forall \bm{v}_h \in U_{h, 0}. \label{eq:orthogonality-Pi-Sigma-s}
    \end{align}
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
By the construction of $\mathcal{S}$ in the proof of \cref{lem:symmetrization},
    \begin{align}
        \Pi^{\Sigma, s}_h \bm{\sigma} = \Pi^{\Sigma}_h \bm{\sigma} + \operatorname{curl} \bm{w}_h \label{prf:consistency-Pi-Sigma-s-1}
    \end{align}
    for some $\bm{w}_h \in H^1 (\Omega; \mathbb{R}^2)$ with $\lVert \bm{w}_h \rVert_{H^1} \lesssim \lVert (\Pi^{\Sigma}_h \bm{\sigma})_{21} - (\Pi^{\Sigma}_h \bm{\sigma})_{12} \rVert$ and $b^*_h (\bm{v}_h, \operatorname{curl} \bm{w}_h) = 0$ for all $\bm{v}_h \in U_{h, 0}$.

    For \eqref{eq:proj-est-Pi-Sigma-s}, the symmetry of $\bm{\sigma}$ gives
    \begin{align*}
        \lVert \bm{w}_h \rVert_{H^1} \lesssim \lVert (\Pi^{\Sigma}_h \bm{\sigma} - \bm{\sigma})_{21} - (\Pi^{\Sigma}_h \bm{\sigma} - \bm{\sigma})_{12} \rVert \lesssim \lVert \bm{\sigma} - \Pi^{\Sigma}_h \bm{\sigma} \rVert,
    \end{align*}
    so that, by \eqref{prf:consistency-Pi-Sigma-s-1},
    \begin{align*}
        \lVert \bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma} \rVert \leq \lVert \bm{\sigma} - \Pi^{\Sigma}_h \bm{\sigma} \rVert + \lVert \operatorname{curl} \bm{w}_h \rVert \lesssim \lVert \bm{\sigma} - \Pi^{\Sigma}_h \bm{\sigma} \rVert,
    \end{align*}
    which combined with \eqref{eq:proj-est-Pi-Sigma} yields \eqref{eq:proj-est-Pi-Sigma-s}. For \eqref{eq:orthogonality-Pi-Sigma-s}, \eqref{prf:consistency-Pi-Sigma-s-1} gives $b_h (\bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma}, \bm{v}_h) = b_h (\bm{\sigma} - \Pi^{\Sigma}_h \bm{\sigma}, \bm{v}_h) - b_h (\operatorname{curl} \bm{w}_h, \bm{v}_h)$; the first term vanishes by \eqref{eq:orthogonality-Pi-Sigma}, and the second equals $b^*_h (\bm{v}_h, \operatorname{curl} \bm{w}_h) = 0$ by the adjointness \eqref{eq:adjointness}.

<!-- theorem-3: body_md -->
\label{thm:optimal-convergence}
    Let $(\bm{u}, \bm{\sigma}, \widetilde{\bm{\sigma}}, \bm{\chi})$ solve the SGE problem \eqref{eq:first-order-SGE-1}--\eqref{eq:first-order-SGE-4} and let $(\bm{u}_h, \bm{\sigma}_h, \widetilde{\bm{\sigma}}_h, \bm{\chi}_h)$ solve the discrete scheme \eqref{eq:discrete-scheme-1}--\eqref{eq:discrete-scheme-4}. Under sufficient regularity, the following estimates hold for $1 \leq r \leq k + 1$:
    \begin{align}
        \lVert \bm{u} - \bm{u}_h \rVert + h \lVert \bm{u} - \bm{u}_h \rVert_{H^1, h} + \lVert \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert + \lVert \bm{\chi} - \bm{\chi}_h \rVert &\lesssim h^r \mathcal{E}_r (\bm{u}), \label{eq:optimal-convergence-1} \\
        \lVert \bm{\sigma} - \bm{\sigma}_h \rVert_{L^2_*} &\lesssim (h^r + \iota h^{r - 1}) \mathcal{E}_r (\bm{u}). \label{eq:optimal-convergence-2}
    \end{align}
<!-- end theorem-3 -->
<!-- proof-8: body_md -->
Subtracting the discrete scheme \eqref{eq:discrete-scheme-1}--\eqref{eq:discrete-scheme-4} from the variational formulation \eqref{eq:variation-formulation-1}--\eqref{eq:variation-formulation-4} and using the orthogonality properties \eqref{eq:orthogonality-Pi-U}, \eqref{eq:orthogonality-Pi-Sigma-s}, \eqref{eq:orthogonality-pi-tilde-Sigma-s} and \eqref{eq:orthogonality-Pi-X-s} of $\Pi^U_h$, $\Pi^{\Sigma, s}_h$, $\pi^{\widetilde{\Sigma}, s}_h$ and $\Pi^{X, s}_h$, we obtain the error equations
    \begin{align}
        &(\mathcal{A}^{\frac{1}{2}} (\pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h), \bm{\tau}_h) - b^*_h (\Pi^U_h \bm{u} - \bm{u}_h, \bm{\tau}_h) = 0, \quad \forall \bm{\tau}_h \in \Sigma^s_h, \label{eq:error-eq-1} \\
        &\begin{aligned} \label{eq:error-eq-2}
            &(\Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h, \bm{\psi}_h) + \iota (\pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h, \operatorname{div} \bm{\psi}_h) \\ &\qquad = -(\bm{\chi} - \Pi^{X, s}_h \bm{\chi}, \bm{\psi}_h), \quad \forall \bm{\psi}_h \in X^s_h,
        \end{aligned} \\
        &\begin{aligned} \label{eq:error-eq-3}
            &(\pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h, \widetilde{\bm{\tau}}_h) - (\mathcal{A}^{\frac{1}{2}} (\Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h), \widetilde{\bm{\tau}}_h) - \iota (\operatorname{div} (\Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h), \widetilde{\bm{\tau}}_h) \\ &\qquad = (\mathcal{A}^{\frac{1}{2}} (\bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma}), \widetilde{\bm{\tau}}_h), \quad \forall \widetilde{\bm{\tau}}_h \in \widetilde{\Sigma}^s_h,
        \end{aligned} \\
        &b_h (\Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h, \bm{v}_h) = 0, \quad \forall \bm{v}_h \in U_{h, 0}. \label{eq:error-eq-4}
    \end{align}

    \emph{Proof of \eqref{eq:optimal-convergence-1}.} Taking $\bm{\tau}_h = \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h$, $\bm{\psi}_h = \Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h$, $\widetilde{\bm{\tau}}_h = \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h$ and $\bm{v}_h = \Pi^U_h \bm{u} - \bm{u}_h$ in \eqref{eq:error-eq-1}--\eqref{eq:error-eq-4} and summing, we obtain
    \begin{align*}
        &\lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert^2 + \lVert \Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h \rVert^2 \\
        &= - (\bm{\chi} - \Pi^{X, s}_h \bm{\chi}, \Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h) + (\mathcal{A}^{\frac{1}{2}} (\bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma}), \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h) \\
        &\lesssim \lVert \bm{\chi} - \Pi^{X, s}_h \bm{\chi} \rVert \lVert \Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h \rVert + \lVert \bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma} \rVert \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert,
    \end{align*}
    which, with the projection estimates \eqref{eq:proj-err-Pi-X-s} and \eqref{eq:proj-err-Pi-Sigma-s}, gives
    \begin{align}
        \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert + \lVert \Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h \rVert \lesssim h^r \mathcal{E}_r (\bm{u}). \label{prf:optimal-convergence-1}
    \end{align}
    By the inf-sup condition \eqref{eq:inf-sup-b*_h-sym} and the error equation \eqref{eq:error-eq-1},
    \begin{align*}
        \lVert \Pi^U_h \bm{u} - \bm{u}_h \rVert_{H^1, h}
        \lesssim \sup_{\bm{\tau}_h \in \Sigma^s_h \setminus \{ \bm{0} \}} \frac{ (\mathcal{A}^{\frac{1}{2}} (\pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h), \bm{\tau}_h) }{\lVert \bm{\tau}_h \rVert} \lesssim \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert,
    \end{align*}
    which, with the discrete Poincar\'e inequality \eqref{eq:poincare-ineq-U_h}, gives
    \begin{align}
        \lVert \Pi^U_h \bm{u} - \bm{u}_h \rVert + \lVert \Pi^U_h \bm{u} - \bm{u}_h \rVert_{H^1, h} \lesssim \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert. \label{prf:optimal-convergence-2}
    \end{align}
    Combining \eqref{prf:optimal-convergence-1} and \eqref{prf:optimal-convergence-2} with the projection estimates \eqref{eq:proj-err-Pi-U}, \eqref{eq:proj-err-pi-tilde-Sigma-s} and \eqref{eq:proj-err-Pi-X-s} yields \eqref{eq:optimal-convergence-1}.

    \emph{Proof of \eqref{eq:optimal-convergence-2}.} The inf-sup condition \eqref{eq:inf-sup-b_h} and the error equation \eqref{eq:error-eq-4} give $\lVert \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h \rVert_{\operatorname{div}, h} = 0$, so that \cref{cor:L2-control-Sigma_h} reduces to
    \begin{align}
        \lVert \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h \rVert_{L^2_*}^2 \lesssim (\mathcal{A}^{\frac{1}{2}} (\Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h), \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h - c \bm{I}), \label{prf:optimal-convergence-3}
    \end{align}
    where $c \triangleq \frac{1}{2\lvert \Omega \rvert} \int_{\Omega} \operatorname{tr} (\Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h)$. Taking $\widetilde{\bm{\tau}}_h = \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h - c \bm{I}$ in \eqref{eq:error-eq-3} bounds the right-hand side by
    \begin{align*}
        \big( \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert + \iota \lVert \operatorname{div} (\Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h) \rVert + \lVert \bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma} \rVert \big) \lVert \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h \rVert_{L^2_*},
    \end{align*}
    so that, by the inverse inequality \eqref{eq:inverse-ineq-X_h},
    \begin{align*}
        \lVert \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h \rVert_{L^2_*} \lesssim \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert + \tfrac{\iota}{h} \lVert \Pi^{X, s}_h \bm{\chi} - \bm{\chi}_h \rVert + \lVert \bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma} \rVert.
    \end{align*}
    Together with \eqref{prf:optimal-convergence-1} and the projection estimate \eqref{eq:proj-err-Pi-Sigma-s}, this implies \eqref{eq:optimal-convergence-2}.

<!-- remark-5: body_md -->
It is instructive to compare these estimates with those of the displacement-based methods, in particular the $H^2$-nonconforming elements of \cite{chen2023robust, liao2023taylorhood, chen2025nonconforming}. For a displacement space of degree~$k$, those methods satisfy
    \begin{align}
        \lVert \bm{u} - \bm{u}_h \rVert_{1, h} + \iota \lVert \bm{u} - \bm{u}_h \rVert_{2, h} \lesssim (h^r + \iota h^{r - 1}) {\left\vert\kern-0.25ex\left\vert\kern-0.25ex\left\vert \bm{u} \right\vert\kern-0.25ex\right\vert\kern-0.25ex\right\vert}_{r + 1}, \qquad 1 \leq r \leq k, \label{eq:nonconforming-optimal-convergence}
    \end{align}
    where $\lVert \cdot \rVert_{1, h}$ and $\lVert \cdot \rVert_{2, h}$ denote discrete $H^1$ and $H^2$ norms. In the discrete $H^1$ norm they are optimal only when $\iota \lesssim h$ and lose one order otherwise, whereas our method attains the optimal rate for the displacement uniformly in $\iota$. Our method also approximates the total stress directly, with the same convergence pattern as \eqref{eq:nonconforming-optimal-convergence}; the displacement-based theory does not cover such an approximation, since the total stress involves third-order derivatives of the displacement. The price is a higher regularity requirement, a trade-off inherent in the mixed formulation.

<!-- remark-6: body_md -->
\cref{thm:optimal-convergence} is developed for the regime in which $\iota$ is not too small. It tracks explicitly the dependence of the error on the regularity of $\bm{u}$ and on the two parameters $\lambda$ and $\iota$; since $\lambda$ enters the norm ${\left\vert\kern-0.25ex\left\vert\kern-0.25ex\left\vert \cdot \right\vert\kern-0.25ex\right\vert\kern-0.25ex\right\vert}$ only in combination with $\operatorname{div} \bm{u}$, the estimates are already robust with respect to $\lambda$. As $\iota \to 0$, however, the solution develops a boundary layer and its regularity degenerates correspondingly, so that the estimates cease to be informative. This motivates the parameter-robust analysis of the next subsection, which bounds the error directly in terms of $\bm{f}$.

<!-- proof-10: body_md -->
Proceeding as in the proof of \cref{thm:optimal-convergence}, but with $\Pi^{X, s}_h$ replaced by $\pi^{X, s}_h$ and using its $L^2$-orthogonality, we obtain the error equations
    \begin{align}

        \lVert \bm{\sigma} - \bm{\sigma}_h \rVert_{L^2_*} \lesssim h^{\epsilon_1} \iota^{1/2 - \epsilon_1} \lVert \bm{f} \rVert_{\epsilon_1} + h^{r_1 + 1} \lVert \bm{f} \rVert_{r_1} + h^{\epsilon_2 - 1} \iota^{3/2 - \epsilon_2} \lVert \bm{f} \rVert_{\epsilon_2} + h^{r_2} \iota \lVert \bm{f} \rVert_{r_2}.
    \end{align*}
    Choosing $\epsilon_1 = \epsilon$, $\epsilon_2 = 1$ and $r_1 = r_2 = r$ gives \eqref{eq:parameter-robust-3}.

Taking $\bm{\tau}_h = \Pi^{\Sigma, s}_h \bm{\sigma} - \bm{\sigma}_h$, $\bm{\psi}_h = \pi^{X, s}_h \bm{\chi} - \bm{\chi}_h$, $\widetilde{\bm{\tau}}_h = \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h$ and $\bm{v}_h = \Pi^U_h \bm{u} - \bm{u}_h$ in \eqref{eq:error-eq-new-1}--\eqref{eq:error-eq-new-4} and summing, we obtain
    \begin{align*}
        &\lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert^2 + \lVert \pi^{X, s}_h \bm{\chi} - \bm{\chi}_h \rVert^2 \\
        &= (\mathcal{A}^{\frac{1}{2}} (\bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma}), \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h) + \iota (\operatorname{div} (\bm{\chi} - \pi^{X, s}_h \bm{\chi}), \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h) \\
        &\lesssim \big( \lVert \bm{\sigma} - \Pi^{\Sigma, s}_h \bm{\sigma} \rVert + \iota \lVert \operatorname{div} (\bm{\chi} - \pi^{X, s}_h \bm{\chi}) \rVert \big) \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert,
    \end{align*}
    which, with \eqref{eq:proj-err-Pi-Sigma-s-low-reg} and \eqref{eq:proj-err-pi-X-s-low-reg-div}, gives
    \begin{align}
        \lVert \pi^{\widetilde{\Sigma}, s}_h \widetilde{\bm{\sigma}} - \widetilde{\bm{\sigma}}_h \rVert + \lVert \pi^{X, s}_h \bm{\chi} - \bm{\chi}_h \rVert \lesssim h^\epsilon \iota^{1/2 - \epsilon} \lVert \bm{f} \rVert_{\epsilon} + h^{r + 1} \lVert \bm{f} \rVert_r. \label{prf:parameter-robust-1}
    \end{align}

    \end{align*}

\label{cor:convergence-to-elasticity}

        \lVert \bm{\sigma}_h - \bm{\sigma}_0 \rVert_{L^2_*} &\lesssim \big( \iota^{1/2} + h^\epsilon \iota^{1/2 - \epsilon} \big) \lVert \bm{f} \rVert_1 + (h^r + \iota h^{r - 1}) \lVert \bm{f} \rVert_r, & &1 \leq r \leq k.
    \end{align*}

<!-- remark-7: body_md -->
The $H^2$-nonconforming finite element methods \cite{chen2023robust, liao2023taylorhood} achieve the parameter-uniform estimate
    \begin{align*}
        \lVert \bm{u} - \bm{u}_h \rVert_{1, h} + \lVert \bm{u}_h - \bm{u}_0 \rVert_{1, h} + \iota \lVert \bm{u} - \bm{u}_h \rVert_{2, h} + \iota \lVert \bm{u}_h - \bm{u}_0 \rVert_{2, h} \lesssim (\iota^{1/2} + h^{1/2}) \lVert \bm{f} \rVert.
    \end{align*}
    As discussed in the introduction, when $\iota \ll 1$ these displacement-based methods suffer from a numerical boundary layer and are therefore only half-order uniformly convergent. To recover the optimal rate in this regime, \cite{chen2025nonconforming} employs Nitsche's method, obtaining an improved bound $\iota^{1/2} \lVert \bm{f} \rVert + h^{r + 1} \lVert \bm{f} \rVert_r$ for $0 \leq r \leq k - 1$. Our method attains the optimal rate as $\iota \ll 1$ without any additional treatment. Again the price is regularity: the asymptotic term $h^\epsilon \iota^{1/2 - \epsilon} \lVert \bm{f} \rVert_\epsilon$ in \cref{thm:parameter-robust} cannot be reduced to the sharp bound $\iota^{1/2} \lVert \bm{f} \rVert$ of the $H^2$-nonconforming methods, although $\epsilon$ may be taken arbitrarily small.

<!-- lemma-6: body_md -->
\label{lem:inf-sup-b*_h-hybrid}
    There exists a constant $\beta > 0$ independent of the mesh size $h$ such that
    \begin{align}
        \sup_{\widetilde{\bm{\tau}}_h \in \widetilde{\Sigma}^s_h \setminus \{ \bm{0} \}} \frac{b^*_h (\bm{v}_h, \widetilde{\bm{\tau}}_h) + (\widehat{\bm{v}}_h, \llbracket \widetilde{\bm{\tau}}_h \cdot \bm{n} \rrbracket)_{\mathcal{F}^{dl}_h}}{\lVert \widetilde{\bm{\tau}}_h \rVert} &\geq \beta \lVert (\bm{v}_h, \widehat{\bm{v}}_h) \rVert_{H^1, h} \label{eq:inf-sup-b*_h-hybrid}
    \end{align}
    for all $(\bm{v}_h, \widehat{\bm{v}}_h) \in U_{h, 0} \times \widehat{U}_h$.
<!-- end lemma-6 -->
<!-- proof-13: body_md -->
It suffices to construct, for each $(\bm{v}_h, \widehat{\bm{v}}_h) \in U_{h, 0} \times \widehat{U}_h$, some $\widetilde{\bm{\tau}}^s_h \in \widetilde{\Sigma}^s_h$ with
    \begin{align}
        b^*_h (\bm{v}_h, \widetilde{\bm{\tau}}^s_h) + (\widehat{\bm{v}}_h, \llbracket \widetilde{\bm{\tau}}^s_h \cdot \bm{n} \rrbracket)_{\mathcal{F}^{dl}_h} \gtrsim \lVert (\bm{v}_h, \widehat{\bm{v}}_h) \rVert^2_{H^1, h}, \qquad
        \lVert \widetilde{\bm{\tau}}^s_h \rVert \lesssim \lVert (\bm{v}_h, \widehat{\bm{v}}_h) \rVert_{H^1, h}. \label{prf:inf-sup-b*_h-hybrid-0}
    \end{align}
    We first construct such a field in the discontinuous space without symmetry constraint,
    \begin{align*}
        \widetilde{\Sigma}_h &\triangleq \{ \widetilde{\bm{\tau}} \in L^2 (\Omega; \mathbb{R}^{2 \times 2}) : \widetilde{\bm{\tau}}|_K \in \mathcal{P}_k (K; \mathbb{R}^{2 \times 2}),\ \forall K \in \mathcal{T}_h \},
    \end{align*}
    and then symmetrize it.

    \emph{Step 1: construction in $\widetilde{\Sigma}_h$.} Integrating by parts in the definition \eqref{eq:b*_h} of $b^*_h$ gives, for every $\widetilde{\bm{\tau}}_h \in \widetilde{\Sigma}_h$,
    \begin{align}
        b^*_h (\bm{v}_h, \widetilde{\bm{\tau}}_h) + (\widehat{\bm{v}}_h, \llbracket \widetilde{\bm{\tau}}_h \cdot \bm{n} \rrbracket)_{\mathcal{F}^{dl}_h} = (\widetilde{\bm{\tau}}_h, \nabla \bm{v}_h)_{\mathcal{T}_h} - \sum_{K \in \mathcal{T}_h} (\bm{v}_h - \widehat{\bm{v}}_h, \widetilde{\bm{\tau}}_h \cdot \vec{\bm{n}})_{\partial K \cap \mathcal{F}^{dl}_h}. \label{prf:inf-sup-b*_h-hybrid-1}
    \end{align}
    By the degrees of freedom of $\Sigma_h$, on each $K \in \mathcal{T}_h$ there exists $\widetilde{\bm{\tau}}_K \in \mathcal{P}_k (K; \mathbb{R}^{2 \times 2})$ such that
    \begin{align*}
        (\widetilde{\bm{\tau}}_K, \bm{\phi})_K &= (\nabla \bm{v}_h, \bm{\phi})_K, & &\forall \bm{\phi} \in \mathcal{P}_{k - 1} (K; \mathbb{R}^{2 \times 2}), \\
        (\widetilde{\bm{\tau}}_K \cdot \vec{\bm{n}}, \bm{\phi})_e &= - h^{-1}_e (\bm{v}_h - \widehat{\bm{v}}_h, \bm{\phi})_e, & &\forall \bm{\phi} \in \mathcal{P}_k (e; \mathbb{R}^2),\ \forall e \in \partial K \cap \mathcal{F}^{dl}_h.
    \end{align*}
    Writing $S_K \triangleq \lVert \nabla \bm{v}_h \rVert^2_K + \sum_{e \in \partial K \cap \mathcal{F}^{dl}_h} h^{-1}_e \lVert \bm{v}_h - \widehat{\bm{v}}_h \rVert^2_e$, these two conditions give
    \begin{align*}
        (\widetilde{\bm{\tau}}_K, \nabla \bm{v}_h)_K - (\bm{v}_h - \widehat{\bm{v}}_h, \widetilde{\bm{\tau}}_K \cdot \vec{\bm{n}})_{\partial K \cap \mathcal{F}^{dl}_h} = S_K,
    \end{align*}
    while a standard scaling argument (see \cite{chung2024stabilizationfree} for reference) gives $\lVert \widetilde{\bm{\tau}}_K \rVert^2_K \lesssim S_K$. Assembling over all $K \in \mathcal{T}_h$ and recalling \eqref{prf:inf-sup-b*_h-hybrid-1}, we obtain $\widetilde{\bm{\tau}}_h \in \widetilde{\Sigma}_h$ with
    \begin{align}
        b^*_h (\bm{v}_h, \widetilde{\bm{\tau}}_h) + (\widehat{\bm{v}}_h, \llbracket \widetilde{\bm{\tau}}_h \cdot \bm{n} \rrbracket)_{\mathcal{F}^{dl}_h} = \lVert (\bm{v}_h, \widehat{\bm{v}}_h) \rVert^2_{H^1, h}, \qquad
        \lVert \widetilde{\bm{\tau}}_h \rVert \lesssim \lVert (\bm{v}_h, \widehat{\bm{v}}_h) \rVert_{H^1, h}. \label{prf:inf-sup-b*_h-hybrid-2}
    \end{align}

    \emph{Step 2: symmetrization.} As in \cref{lem:symmetrization}, there is an operator $\mathcal{S} : \widetilde{\Sigma}_h \to \widetilde{\Sigma}^s_h$ of the form $\mathcal{S} \widetilde{\bm{\tau}}_h = \widetilde{\bm{\tau}}_h + \operatorname{curl} \bm{w}_h$ with $\bm{w}_h \in H^1 (\Omega; \mathbb{R}^2)$, satisfying $b^*_h (\bm{v}_h, \mathcal{S} \widetilde{\bm{\tau}}_h) = b^*_h (\bm{v}_h, \widetilde{\bm{\tau}}_h)$ for all $\bm{v}_h \in U_{h, 0}$ and $\lVert \mathcal{S} \widetilde{\bm{\tau}}_h \rVert \lesssim \lVert \widetilde{\bm{\tau}}_h \rVert$. The conformity $\operatorname{curl} \bm{w}_h \in H (\operatorname{div}; \Omega)$ gives $\llbracket \operatorname{curl} \bm{w}_h \cdot \bm{n} \rrbracket_{|_e} = \bm{0}$ for every $e \in \mathcal{F}^{dl}_h$, so the multiplier term is unchanged as well. Hence $\widetilde{\bm{\tau}}^s_h \triangleq \mathcal{S} \widetilde{\bm{\tau}}_h$ inherits \eqref{prf:inf-sup-b*_h-hybrid-2} and satisfies \eqref{prf:inf-sup-b*_h-hybrid-0}.

<!-- remark-8: body_md -->
\label{rmk:incompressible-hybrid}
    In the incompressible limit $\lambda \to \infty$, \cref{rmk:incompressible-limit} requires the normalization $\int_{\Omega} \operatorname{tr} \bm{\sigma}_h = 0$, which is global and therefore obstructs the local elimination of $\bm{\sigma}_h$. The remedy is to split $\bm{\sigma}_h = \bm{\sigma}^0_h + \overline{\sigma}_h \bm{I}$, where $\bm{\sigma}^0_h$ has elementwise zero-mean trace and $\overline{\sigma}_h$ is piecewise constant with zero mean over $\Omega$. The first part retains the local structure of \eqref{eq:hybrid-scheme-1}--\eqref{eq:hybrid-scheme-6}, while $\overline{\sigma}_h$ joins $(\widehat{\bm{u}}_h, \widehat{\bm{\sigma}}_h)$ as a third globally coupled unknown. The resulting incompressible hybridized scheme and its local and global problems are given in \Cref{sm:incompressible}.
