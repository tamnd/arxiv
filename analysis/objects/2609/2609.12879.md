<!-- lemma-1: body_md -->
\label{lemma:cone_divergence_theorem_sector_like_domains}
    Let $\Omega \subset {\Sigma_D}$ be a sector-like domain and let $F: \overline{\Omega} \to {\mathbb{R}}^N$ be a vector field such that
    \begin{align}
        & F \in C^1((\Omega \cup {\Gamma_{0, \Omega}} \cup {\Gamma_{\Omega}}) \setminus \{O\}) \cap L^2(\Omega), \nonumber \\ 
        & \mathrm{div } F \in L^1(\Omega). \nonumber
    \end{align}
    Then
    \begin{equation*}
        \int_\Omega \mathrm{div } F \ dx = \int_{\Gamma_{0, \Omega}} F \cdot \nu \ d\sigma_{{\Gamma_{0, \Omega}}} + \int_{{\Gamma_{\Omega}} \setminus \{O\}} F \cdot \nu \ d\sigma_{{\partial{\Sigma_D}}}.
    \end{equation*}

<!-- definition-1: body_md -->
The capacity of a set $E \subseteq {\mathbb{R}}^N$ is defined as
    \begin{equation*}
        {\text{cap}}(E) \coloneqq \inf\{\|v\|^2_{H^1({\mathbb{R}}^N)} \ : \ v \geq 1 \text{ in a neighbourhood of } E\}.
    \end{equation*}

<!-- definition-2: body_md -->
A set $U \subset {\mathbb{R}}^N$ is said to be quasi-open if for every $\varepsilon > 0$ there exists an open set $V_\varepsilon$ such that $U \cup V_\varepsilon$ is open and ${\text{cap}}(V_\varepsilon) \leq \varepsilon$.

<!-- prop-1: body_md -->
\label{prop:cone_first_derivative}
    Let $\varphi \in {C^{2, \alpha}(\overline{D})}$. For any $\eta \in {C^{2, \alpha}(\overline{D})}$, there exists $\delta > 0$ such that the map
    \begin{equation*}
        t \in (- \delta, \delta) \mapsto (u_{\varphi + t\eta} \circ \xi_t, \lambda_1(\varphi + t\eta)) \in L^2({\Sigma_D}) \times {\mathbb{R}}
    \end{equation*}
    is of class $C^\infty$ in $(- \delta, \delta)$ and ${\widetilde{u}}_\eta = \left.\mfrac{d}{dt} (u_{\varphi + t\eta}) \right|_{t = 0} \in H^1(\Omega_\varphi)$ is the unique solution of
    \begin{equation}
        \label{eq:cone_pde_u_prime_general}
        \left\{
        \begin{array}{rcll}
            - \Delta {\widetilde{u}}_\eta & = & \lambda_1(\varphi) {\widetilde{u}}_\eta + \lambda' u_\varphi & \quad \text{ in } \Omega_\varphi \\
            {\widetilde{u}}_\eta & = & \displaystyle - \mfrac{\partial u_\varphi}{\partial \nu} (V_\eta \cdot \nu) & \quad \text{ on } {\Gamma_{0, \varphi}} \\
            \displaystyle \mfrac{\partial {\widetilde{u}}_\eta}{\partial \nu} & = & 0 & \quad \text{ on } {\Gamma_\varphi} \setminus \{O\} \\
            \displaystyle \int_{\Omega_\varphi} {\widetilde{u}}_\eta u_\varphi \ dx & = & 0 &
        \end{array}
        \right.
        ,
    \end{equation}    
    where $\lambda' \coloneqq \left. \frac{d \lambda_t}{dt}\right|_{t = 0}$ is given by
    \begin{equation}
        \label{eq:cone_lambda_prime}
        \lambda'(\varphi)[\eta] = \left. \frac{d \lambda_t}{dt}\right|_{t = 0} = - \int_D e^{N\varphi(q)} \eta(q) \left(\frac{\partial u_\varphi}{\partial \nu}(e^{\varphi(q)}q)\right)^2 \ d\sigma.
    \end{equation}
<!-- end prop-1 -->
<!-- proof-1: body_md -->
Let $\delta > 0$ be a positive number to be chosen later. For every $t \in (- \delta, \delta)$ sufficiently small, the eigenvalue $\lambda_t \coloneqq \lambda_1(\varphi + t\eta)$ and the corresponding eigenfunction $u_t \coloneqq u_{\Omega_{\varphi + t\eta}}$ are well-defined in the domain $\Omega_t \coloneqq \Omega_{\varphi + t\eta}$. Moreover, the map
    \begin{equation*}
        t \in (- \delta, \delta) \mapsto (u_t, \lambda_t) \in H_0^1(\Omega_t \cup \Gamma_t) \times {\mathbb{R}}
    \end{equation*}
    where for brevity we have set $\Gamma_t \coloneqq \Gamma_{\varphi + t\eta}$, is continuous (\cite[Chapter 4]{HenrotPierre2018}). The variational formulation of the eigenvalue problem in $\Omega_t$ is
    \begin{align}
        & u_t \in H_0^1(\Omega_t \cup \Gamma_t), \nonumber \\
        & \int_{\Omega_t} u_t^2\ dx = 1, \nonumber \\
        & \int_{\Omega_t} \nabla u_t \nabla v \ dx = \lambda_t \int_{\Omega_t} u_t v \ dx \quad \forall v \in H_0^1(\Omega_t \cup \Gamma_t). \nonumber
    \end{align}

    As usual in the theory of shape derivatives, the idea is to transport this equation to the fixed domain $\Omega_\varphi$ and then to apply the implicit function theorem (see \cite{HenrotPierre2018, SokolowskiZolesio1992}). This can be done by virtue of the fact that
    \begin{equation*}
        H_0^1(\Omega_\varphi \cup {\Gamma_\varphi}) = \{v \circ \xi_t \ : \ v \in H_0^1(\Omega_t \cup \Gamma_t)\}. 
    \end{equation*}
    Thus, under the change of variables given by $\xi_t$, the transported function
    \begin{equation*}
        {\widehat{u}}_t \coloneqq u_t \circ \xi_t
    \end{equation*}
    satisfies
    \begin{align}
        & {\widehat{u}}_t \in H_0^1(\Omega_\varphi \cup {\Gamma_\varphi}), \nonumber \\
        & \int_{\Omega_\varphi} {\widehat{u}}_t^2 J_t \ dx = 1, \nonumber \\ 
        & \int_{\Omega_\varphi} (A_t \nabla {\widehat{u}}_t) \nabla w \ dx = \lambda_t \int_{\Omega_\varphi} {\widehat{u}}_t w J_t \ dx \quad \forall w \in H_0^1(\Omega_\varphi \cup {\Gamma_\varphi}), \nonumber
    \end{align}
    where, denoting by $D\xi_t$ the Jacobian matrix of $\xi_t$,  
    \begin{equation*}
        J_t = \det D\xi_t,
    \end{equation*}
    which is positive for $t$ small because $\xi_t$ is close to the identity, and 
    \begin{equation*}
        A_t = J_t(D\xi_t^{-1}) (D \xi_t^{-1})^T.
    \end{equation*}
    Note that $J_t$ and $A_t$ are in $L^\infty$, even if $\xi_t$ and $V$ are not differentiable at the vertex of the cone.
    

    Next, we define the operator
    \begin{equation*}
        {\mathcal{F}} : (- \delta, \delta) \times H_0^1(\Omega_\varphi \cup {\Gamma_\varphi}) \times {\mathbb{R}} \to H^{-1}(\Omega_\varphi \cup {\Gamma_\varphi}) \times {\mathbb{R}}
    \end{equation*}
    where $H^{-1}(\Omega_\varphi \cup {\Gamma_\varphi})$ denotes the dual space of $H_0^1(\Omega_\varphi \cup {\Gamma_\varphi})$, by
    \begin{equation*}
        {\mathcal{F}}(t, v, \lambda) = \left(- \mathrm{div }(A_t \nabla v) - \lambda v J_t, \int_{\Omega_\varphi} v^2 J_t \ dx - 1\right).
    \end{equation*}
    It is a standard fact from the theory of shape derivatives that the map ${\mathcal{F}}$ is of class $C^\infty$ (see, e.g., \cite[Page 207]{HenrotPierre2018}).
    
    Note that 
    \begin{equation*}
        {\mathcal{F}}(t, {\widehat{u}}_t, \lambda_t) = 0 \quad \forall t \in (-\delta, \delta).
    \end{equation*}
    We aim to apply the implicit function theorem to obtain the differentiability of ${\widehat{u}}_t$, and hence of $u_t$, at $t = 0$. To this end, let us note that
    \begin{equation*}
        \partial_{v, \lambda}{\mathcal{F}}(0, u_\varphi, \lambda_1(\varphi)) [v, \lambda] = \left(- \Delta v - \lambda u_\varphi - \lambda_1(\varphi) v, 2 \int_{\Omega_\varphi} u_\varphi v \ dx \right), \quad (v, \lambda) \in H_0^1(\Omega_\varphi \cup {\Gamma_\varphi}) \times {\mathbb{R}}.
    \end{equation*}
    It is also standard to show that 
    \begin{equation*}
        \partial_{v, \lambda}{\mathcal{F}}(0, u_\varphi, \lambda_1(\varphi)) : H_0^1(\Omega_\varphi \cup {\Gamma_\varphi}) \times {\mathbb{R}} \to H^{-1}(\Omega_\varphi \cup {\Gamma_\varphi}) \times {\mathbb{R}}
    \end{equation*}
    is an isomorphism; the proof goes as in \cite[Lemma 5.7.3]{HenrotPierre2018}, with small modifications regarding the functional spaces (see also \cite{LamboleySicbaldi2014}). Hence, by the implicit function theorem, up to taking a smaller $\delta > 0$, it holds that the map
    \begin{equation*}
        t \in (- \delta, \delta) \mapsto ({\widehat{u}}_t, \lambda_t)
    \end{equation*}
    is of class $C^\infty$. In addition, since $u_t = {\widehat{u}}_t \circ \xi_t^{-1}$, then small modifications of \cite[Lemma 5.3.3] {HenrotPierre2018} yield that 
    \begin{equation*} 
    t \in (- \delta, \delta) \mapsto u_t \in L^2({\Sigma_D})
    \end{equation*}
    is differentiable at $t = 0$ and
    \begin{equation}
        \label{eq:cone_u_prime}
        {\widetilde{u}}_\eta = \left.\mfrac{d}{dt}{\widehat{u}}_t\right|_{t = 0} - \nabla u_\varphi \cdot V_\eta \in H^1(\Omega_\varphi)
    \end{equation}
    where ${\widetilde{u}}_\eta = \left.\mfrac{d}{dt}\right|_{t = 0} u_t$ and $V_\eta$ is the vector field corresponding to the deformations $\xi_t$ (see \eqref{eq:cone_vector_field}).

    Let us now characterize ${\widetilde{u}}_\eta$ as the solution of a differential problem. Since $\nabla u_\varphi = \mfrac{\partial u_\varphi}{\partial \nu} \nu$ on ${\Gamma_{0, \varphi}}$ and $\left.\mfrac{d}{dt}{\widehat{u}}_t\right|_{t = 0} \in H_0^1(\Omega_\varphi \cup {\Gamma_\varphi})$, then
    \begin{equation*}
        {\widetilde{u}}_\eta = - \frac{\partial u_\varphi}{\partial \nu} (V \cdot \nu) \quad \text{ on } {\Gamma_{0, \varphi}}.
    \end{equation*}
    Next, arguing as in the proof of \cite[Theorem 5.3.1]{HenrotPierre2018}, namely taking $v \in C_c^\infty(\Omega_\varphi)$, integrating by parts and applying Hadamard's formula (\cite[Section 5.2]{HenrotPierre2018}), it follows that
    \begin{equation*}
        \int_{\Omega_\varphi} (- \Delta {\widetilde{u}}_\eta - \lambda_1(\varphi){\widetilde{u}}_\eta - \lambda' u_\varphi) v \ dx = 0. 
    \end{equation*}
    Since $v \in C_c^\infty(\Omega_\varphi)$ is arbitrary, we conclude that ${\widetilde{u}}_\eta$ satisfies, in the weak sense (and therefore, by standard elliptic regularity theory, also in the classical sense)
    \begin{equation*}
        - \Delta {\widetilde{u}}_\eta = \lambda_1(\varphi) {\widetilde{u}}_\eta + \lambda' u_\varphi \quad \text{ in } \Omega_\varphi.
    \end{equation*}
    Also, by Hadamard's formula, differentiating the relation
    \begin{equation*}
        \int_{\Omega_t} u_t^2 \ dx = 1 \quad \forall t \in (- \delta, \delta)
    \end{equation*}
    at $t = 0$ yields
    \begin{equation*}
        2 \int_{\Omega_\varphi} u_\varphi {\widetilde{u}}_\eta \ dx + \int_{{\Gamma_\varphi}} u_\varphi^2 (V \cdot \nu) \ d\sigma = 0.
    \end{equation*}
    Since $V \perp \nu$ on ${\partial{\Sigma_D}}$, then we obtain the orthogonality constraint
    \begin{equation*}
        \int_{\Omega_\varphi} u_\varphi {\widetilde{u}}_\eta \ dx = 0.
    \end{equation*}
    Finally, differentiating the relation 
    \begin{equation*}
        \nabla u_t(\xi_t(x)) \cdot \nu_{\xi_t(x)} = 0 \quad \forall t \in (- \delta, \delta),
    \end{equation*}
    which holds because $V$ is tangent to ${\partial{\Sigma_D}}$ on ${\partial{\Sigma_D}}$, and arguing as in \cite[Section 5.5]{HenrotPierre2018}, we obtain
    \begin{equation*}
        \frac{\partial {\widetilde{u}}_\eta}{\partial \nu} = 0 \quad \text{ on } {\Gamma_\varphi} \setminus \{O\}.
    \end{equation*}
    Summarizing, ${\widetilde{u}}_\eta \in H^1(\Omega_\varphi)$ satisfies \eqref{eq:cone_pde_u_prime_general}.

    The differential problem \eqref{eq:cone_pde_u_prime_general} allows us to obtain a general expression for $\lambda'$ as follows. Multiplying the differential equation by $u_\varphi$, taking into account that $\int_{\Omega_\varphi} {\widetilde{u}}_\eta u_\varphi \ dx = 0$, and integrating by parts (making use of Lemma \ref{lemma:cone_divergence_theorem_sector_like_domains}), we obtain:
    \begin{align}
        \lambda' \underbrace{\int_{\Omega_\varphi} u_\varphi^2 \ dx}_{= 1}
        & = \int_{\Omega_\varphi} \nabla {\widetilde{u}}_\eta \nabla u_\varphi \ dx - \underbrace{\int_{\Omega_\varphi} \lambda_1(\varphi) {\widetilde{u}}_\eta u_\varphi \ dx}_{= 0} \nonumber \\
        & = \underbrace{\int_{\Omega_\varphi} \lambda_1(\varphi) u_\varphi {\widetilde{u}}_\eta \ dx}_{= 0} + \int_{{\Gamma_{0, \varphi}}} {\widetilde{u}}_\eta \frac{\partial u_\varphi}{\partial \nu} \ d\sigma \nonumber \\
        & = - \int_{{\Gamma_{0, \varphi}}} \left(\frac{\partial u_\varphi}{\partial \nu} \right)^2 (V \cdot \nu) \ d\sigma. \nonumber 
    \end{align}
    Formula \eqref{eq:cone_lambda_prime} then follows immediately, in view of Remark \ref{rem:cone_normal_dsigma} and since
    \begin{equation*}
        V \cdot \nu = \frac{|x|}{\displaystyle \sqrt{1 + \left|\nabla_{\mathbb{S}^{N - 1}} \varphi \left(\frac{x}{|x|} \right)\right|^2}} \eta\left(\frac{x}{|x|} \right) \quad \text{ on } {\Gamma_{0, \varphi}}.
    \end{equation*}
    The proof is complete.

By the theory of traces (see, e.g., \cite{Wloka1987, Leoni2017}), we can choose a lift $\phi \in H^1({\Omega_D})$ such that $\phi = - u_D'(1)\eta$ on $D$ (in the sense of traces) and

        \label{eq:cone_proof_estimate_a}
        \|\phi\|_{H^1({\Omega_D})} \leq C_1 \|\eta\|_{H^{1/2}(D)}.
    \end{equation}
    By composing with the orthogonal projection operator onto the space $\langle u_D\rangle^\perp$ if necessary, we can further assume that
    \begin{equation*}
        \int_{\Omega_D} \phi u_D \ dx = 0.
    \end{equation*}

    With these ingredients at hand, finding a weak solution ${\widetilde{u}}_\eta \in H^1({\Omega_D})$ to \eqref{eq:cone_pde_u_eta_prime_spherical_sector} is equivalent to finding a weak solution ${\widetilde{v}} \in (H_0^1({\Omega_D} \cup \Gamma_D) \cap \langle u_D \rangle^\perp)$ to
    \begin{equation}
        \label{eq:cone_variational_pde_v}
        \int_{\Omega_D} \nabla {\widetilde{v}} \nabla v \ dx - \int_{\Omega_D} \lambda_1({\Omega_D}) {\widetilde{v}} v \ dx = - \Phi(v) \quad \forall v \in H_0^1({\Omega_D} \cup \Gamma_D),
    \end{equation}
    where $\Phi \in H^{-1}({\Omega_D} \cup \Gamma_D)$ is the continuous linear functional
    \begin{equation*}
        \Phi(v) = \int_{\Omega_D} \nabla \phi \nabla v \ dx - \lambda_1({\Omega_D}) \int_{\Omega_D} \phi v \ dx.
    \end{equation*}
    To this end, we may use the Lax--Milgram theorem. Indeed, solutions are the critical points in the subspace $\langle u_D \rangle^\perp$ of the functional
    \begin{equation*}
        J(v) = \frac{1}{2} \int_{\Omega_D} |\nabla v|^2 - \frac{1}{2}\lambda_1({\Omega_D}) \int_{\Omega_D} v^2 \ dx + \Phi(v).
    \end{equation*}
    Note that $J$ is coercive in $\langle u_D \rangle^\perp$, since for every $v$ in this subspace it holds that
    \begin{equation*}
        \int_{\Omega_D} (|\nabla v|^2 - \lambda_1({\Omega_D}) v^2) \ dx \geq \left(1 - \frac{\lambda_1({\Omega_D})}{\lambda_2({\Omega_D})} \right) \int_{{\Omega_D}} |\nabla v|^2 \ dx.
    \end{equation*}
    Hence, by Lax-Milgram theorem, for each $\eta$ there exists a unique solution ${\widetilde{v}}$ to \eqref{eq:cone_variational_pde_v}, which moreover depends continuously on the datum $\phi$, that is, there exists some constant $C_2 > 0$ independent of $\phi$ such that
    \begin{equation}
        \label{eq:cone_proof_estimate_b}
        \|{\widetilde{v}}\|_{H_0^1({\Omega_D} \cup \Gamma_D)} \leq C_2 \|\phi\|_{H^1({\Omega_D})},
    \end{equation}
    see \cite[Section 3.2.1]{Kesavan1989}. Then, writing ${\widetilde{u}}_\eta = {\widetilde{v}} + \phi$ and combining \eqref{eq:cone_proof_estimate_a} with \eqref{eq:cone_proof_estimate_b}, we readily obtain \eqref{eq:cone_estimate_u_eta}.

        u_k = - u_D'(1)\left(\sum_{j = 1}^k(\eta, \psi_j)\psi_j\right) \quad \text{ on } D.
    \end{equation*}
    Then, by making use of the regularity of $\eta$ and \eqref{eq:cone_estimate_u_eta}, we obtain
    \begin{align}
        0
        & \leq \limsup_{k \to \infty} \|{\widetilde{u}}_\eta - u_k\|_{H^1({\Omega_D})} \nonumber \\
        & \leq \limsup_{k \to \infty} C \left\|\sum_{j = k + 1}^\infty (\eta, \psi_j) \psi_j\right\|_{H^{1/2}(D)} \nonumber \\
        & = 0. \nonumber
    \end{align}
    Now, by definition, 
    \begin{equation*}
        u_k \to \sum_{j = 1}^\infty (\eta, \psi_j) {\widetilde{u}}_j
    \end{equation*}
    strongly in $H^1({\Omega_D})$ as $k \to \infty$ (see \cite[Lemma 5.1]{Brezis2010}), which completes the proof.

<!-- theorem-1: body_md -->
\label{thm:cone_stability}
    Let ${\Sigma_D}$ be the cone spanned by the smooth domain $D \subset {\mathbb{S}^{N - 1}}$, with $N \geq 3$, and let $\mu_1$ be the first non-zero Neumann eigenvalue of the Laplace--Beltrami operator on $D$.
    \begin{enumerate}[label=(\roman*)]
        \item If $\mu_1 < N - 1$, then ${\Omega_D}$ is an unstable critical domain.

        \item If $\mu_1 > N - 1$, then $\Omega_D$ is a stable critical domain.
    \end{enumerate}
<!-- end theorem-1 -->
<!-- proof-12: body_md -->
We argue as in \cite[Theorem 1.1]{AfonsoIacopettiPacella2024Energypublished}.

    We first prove \textit{(i)}. Let ${\widetilde{u}}_1(r, q) = h_1(r) \psi_1(q)$ in ${\Omega_D}$ be the solution of \eqref{eq:cone_pde_u_eta_prime_spherical_sector}. By Lemma \ref{lem:cone_simplified_second_derivative}, we have
    \begin{equation}
        \label{eq:cone_proof_main_lambda_primeprime}
        \lambda_1|_{\mathcal{M}}''(0)[\psi_1, \psi_1] = - 2 u_D'(1)(h_1'(1) + u_D''(1)).
    \end{equation}

    Writing the equation for $h_1$ in Sturm--Liouville form, we obtain
    \begin{equation}
        \label{eq:cone_proof_main_equation_h_1_SL}
        - (r^{N - 1} h_1')' - r^{N - 1} \lambda_1({\Omega_D}) h_1 = - r^{N - 3} \mu_1 h_1.
    \end{equation}
    On the other hand, writing $- \Delta u_D = \lambda_1({\Omega_D}) u_D$ in polar coordinates and taking the derivative with respect to $r$, we obtain
    \begin{equation*}
        - (u_D')'' - \frac{N - 1}{r} (u_D')' - \lambda_1({\Omega_D}) u_D' = - \frac{N - 1}{r^2} u_D', 
    \end{equation*}
    which in Sturm--Liouville form is
    \begin{equation}
        \label{eq:cone_proof_main_equation_u_D_prime_SL}
        - (r^{N - 1} u_D'')' - r^{N - 1} \lambda_1({\Omega_D}) u_D' = - r^{N - 3} (N - 1) u_D'.
    \end{equation}

    Let $\bar r \in (0, 1)$. Multiplying \eqref{eq:cone_proof_main_equation_h_1_SL} by $u_D'$ and integrating by parts in $(\bar r, 1)$ we obtain
    \begin{align}
        \int_{\bar r}^1 r^{N - 1} h_1' u_D'' \ dr - (r^{N - 1} h_1' u_D')|_{\bar r}^1 - \lambda_1({\Omega_D}) \int_{\bar r}^1 r^{N - 1}
        u_D' h_1 \ dr = - \mu_1 \int_{\bar r}^1 r^{N - 3} h_1 u_D' \ dr. \label{eq:cone_proof_main_3.43}
    \end{align}
    Similarly, multiplying \eqref{eq:cone_proof_main_equation_u_D_prime_SL} by $h_1$ and integrating by parts we deduce that
    \begin{align}
        \int_{\bar r}^1 r^{N - 1} h_1' u_D'' \ dr - (r^{N - 1} h_1 u_D'')|_{\bar r}^1 - \lambda_1({\Omega_D}) \int_{\bar r}^1 r^{N - 1} u_D' h_1 \ dr = - (N - 1) \int_{\bar r}^1 r^{N - 3} u_D' h_1 \ dr. \label{eq:cone_proof_main_3.44}
    \end{align}

    Combining Lemma \ref{lem:cone_reg_h_j} with Hölder's inequality, and since $u_D'$ is bounded in $[0, 1]$, the right-hand sides of \eqref{eq:cone_proof_main_3.43} and \eqref{eq:cone_proof_main_3.44} remain finite in the limit as $\bar r \to 0$.

    Next, we claim that
    \begin{equation}
        \label{eq:cone_proof_main_claim}
        \lim_{\bar r \to 0^+} \bar r^{N - 1} h_1'(\bar r) u_D'(\bar r) = 0. 
    \end{equation}
    To prove the claim, we integrate \eqref{eq:cone_proof_main_equation_h_1_SL} to obtain
    \begin{align}
        \left|\int_{\bar r}^1 -(r^{N - 1} h_1')' \ dr \right|
        & = \left|\bar r^{N - 1} h_1'(\bar r) - h_1'(1) \right| \nonumber \\
        & \leq \lambda_1({\Omega_D}) \int_{\bar r}^1 r^{N - 1} h_1 \ dr + \int_{\bar r}^1 r^{N - 3} \mu_1 h_1 \ dr \nonumber \\
        & \leq C_1 \nonumber
    \end{align}
    for some $C_1 > 0$. As a consequence, there exists $C_2 > 0$ such that
    \begin{equation*}
        \limsup_{\bar r \to 0^+} \bar r^{N - 1} |h_1'(\bar r)| \leq C_2.
    \end{equation*}
    Then \eqref{eq:cone_proof_main_claim} follows from the fact that
    \begin{equation*}
        \lim_{\bar r \to 0^+} u_D'(\bar r) = u_D'(0) = 0,
    \end{equation*}
    because $u_D = u_{B_1(O)}|_{\Sigma_D}$ and $\nabla u_{B_1(O)}(O) = 0$. 

    Now, subtracting \eqref{eq:cone_proof_main_3.44} from \eqref{eq:cone_proof_main_3.43} and taking the limit as $\bar r \to 0^+$ it follows that
    \begin{equation}
        \label{eq:cone_proof_main_main_eq}
        - u_D'(1)(h_1'(1) + u_D''(1)) = (N - 1 - \mu_1) \int_0^1 r^{N - 3} h_1 u_D' \ dr < 0,
    \end{equation}
    because $\mu_1 < N - 1$, $h_1 > 0$ and $u_D' < 0$ in $(0, 1]$. Hence, in view of \eqref{eq:cone_proof_main_lambda_primeprime}, it holds that
    \begin{equation*}
        \lambda_1|_{\mathcal{M}}''(0)[\psi_1, \psi_1] < 0,
    \end{equation*}
    which concludes the proof of \textit{(i)}.

    To prove \textit{(ii)}, we recall that the family $\{\psi_j\}_{j \in {\mathbb{N}} \cup \{0\}}$ is an orthonormal basis for $L^2(D)$. Then any $\eta \in T_0{\mathcal{M}}$ can be written as
    \begin{equation}
        \label{eq:cone_proof_main_eta}
        \eta = \sum_{j = 1}^\infty (\eta, \psi_j) \psi_j,
    \end{equation}
    where $(\cdot, \cdot)$ denotes the inner product in $L^2(D)$. Without loss of generality, we may assume that $\int_D \eta^2 \ d\sigma = 1$. By Lemma \ref{lem:cone_decomposition_u_eta}, 
    \begin{equation*}
        {\widetilde{u}}_\eta = \sum_{j = 1}^\infty (\eta, \psi_j) {\widetilde{u}}_j.
    \end{equation*}

    Now we show that if $k > j$, then $h_k'(1) \geq h_j'(1)$, with strict inequality if $\mu_k > \mu_j$. To this end, we argue as in the proof of \textit{(i)}, namely, we write the equations for $h_k$ and $h_j$ in Sturm--Liouville form, multiply by $h_j$ and $ h_k$, respectively, integrate by parts, and subtract, to obtain
    \begin{equation*}
        - u_D'(1) (h_k'(1) - h_j'(1)) = (- \mu_j + \mu_k) \int_0^1 r^{N - 3} h_k h_j \ dr \geq 0.
    \end{equation*}
    This fact will be useful together with the following remark: for every $j \in {\mathbb{N}}$, it holds that
    \begin{equation*}
        \frac{\partial {\widetilde{u}}_j}{\partial \nu} (1, q) = h_j'(1) \psi_j(q).
    \end{equation*}

    We now compute
    \begin{align}
        \lambda_1|_{\mathcal{M}}''(0)[\eta, \eta]
        & = - 2u_D'(1) \left(\int_D \left(\sum_{j = 1}^\infty (\eta, \psi_j) \psi_j \right)\left(\sum_{k = 1}^\infty (\eta, \psi_k) h_k'(1) \psi_k \right) \ d\sigma + u_D''(1) \int_D \eta^2 \ d\sigma\right) \nonumber \\
        & = - 2u_D'(1) \left(\sum_{j = 1}^\infty (\eta, \psi_j)^2 h_j'(1) + u_D''(1)\right) \nonumber \\
        & \geq - 2u_D'(1) (h_1'(1) + u_D''(1)) \nonumber \\
        &  = 2 (N - 1 - \mu_1) \int_0^1 r^{N - 3} h_1 u_D' \ dr \nonumber \\
        & > 0, \nonumber
    \end{align}
    in view of \eqref{eq:cone_proof_main_main_eq}, and since $\mu_1 > N - 1$ (by assumption), $h_1 > 0$ and $u_D' < 0$ in $(0, 1]$. Since $\eta \in T_0{\mathcal{M}}$ is arbitrary, this shows that ${\Omega_D}$ is a stable domain for the first eigenvalue. The proof is complete.

<!-- proof-14: body_md -->
We begin with some geometric considerations regarding the cone ${\Sigma_D}$.
    
    Fix $q \in \partial D \subset \partial {\Sigma_D} \setminus \{O\}$. Since $D$ is smooth, $\partial {\Sigma_D}$ is smooth near $q$, and therefore the tangent plane $T_q\partial {\Sigma_D}$ is well-defined. Let $(x', x_N)$ be the coordinates of points in ${\mathbb{R}}^N$ with respect to the coordinate system $\{v_1, \ldots, v_{N - 1}, - \nu(q)\}$, where $\{v_i\}_{i = 1, \ldots, N - 1}$ is an orthonormal basis for $T_q {\partial{\Sigma_D}}$ and $- \nu(q)$ is the inner unit normal vector to ${\partial{\Sigma_D}}$ at $q$.

    Observe that there exists an open neighborhood $V$ of $q$ in ${\partial{\Sigma_D}} \setminus \{O\}$ and an open neighborhood $U$ of the origin $0$ in $T_q{\partial{\Sigma_D}}$ such that $V - q$ is the graph of a smooth function $g: U \to {\mathbb{R}}$ (see \cite[Page 1034]{IacopettiPacellaWeth2022}). In particular, $\nabla_{x'} g(0) = 0$, where $\nabla_{x'}$ denotes the gradient with respect to the variables $x_1', \ldots, x_{N - 1}'$.

    Since ${\Sigma_D}$ is a cone, we have that for any $t > 0$ it holds
    \begin{equation*}
        T_{tq} {\partial{\Sigma_D}} = T_q {\partial{\Sigma_D}}, \quad \nu(tq) = \nu(q),
    \end{equation*}
    and moreover $tV - tq$ is the graph on $T_q {\partial{\Sigma_D}}$ of the function $g_t: tU \to {\mathbb{R}}$ given by
    \begin{equation*}
        g_t(x') = t g\left(\frac{x'}{t} \right), \quad x' \in tU.
    \end{equation*}
    By the chain rule and the mean value theorem, for any $x' \in tU$ and $i = 1, \ldots, N - 1$ it holds
    \begin{align}
        \frac{\partial g_t}{\partial x_i'} (x')
        & = \frac{\partial g}{\partial x_i'} \left( \frac{x'}{t} \right) \nonumber \\
        & = \frac{\partial g}{\partial x_i'}(0) + \left(\nabla_{x'} \frac{\partial g}{\partial x_i'}\right) (\eta_i) \cdot \frac{x'}{t} \nonumber \\
        & = \left(\nabla_{x'} \frac{\partial g}{\partial x_i'}\right) (\eta_i) \cdot \frac{x'}{t}, \label{eq:cone_mean_value_theorem}
    \end{align}
    where $\eta$ belongs to the segment joining $0$ and  $\mfrac{x'}{t}$. It follows that for any fixed ball $B_R = B_R(0) \subset T_q {\partial{\Sigma_D}}$ and all $t$ sufficiently large such that $B_R \subset tU$ it holds
    \begin{equation*}
        \max_{x' \in \overline{B_R}} |\nabla_{x'} g_t(x')| \leq \frac{1}{t} \max_{x' \in \overline U} \sqrt{\sum_{i = 1}^{N - 1} \left|\left(\nabla_{x'} \frac{\partial g}{\partial x_i'}\right)(x')\right|^2} R \leq \frac{C}{t},
    \end{equation*}
    where $C$ does not depend on $t$. In particular, it holds
    \begin{equation}
        \label{eq:cone_nabla_xprime_g_t_vanishes}
        \lim_{t \to + \infty} \max_{x' \in \overline{B_R}} |\nabla_{x'} g_t(x')| = 0.
    \end{equation}

    Next we consider the upper cylinder $C_R^+$ generated by $B_R$ and the epigraph $E_t^+$ of $g_t|_{B_R}$:
    \begin{align}
        & C_R^+ \coloneqq \{(x', x_N) \in {\mathbb{R}}^N \ : \ x' \in B_R, \ x_N > 0\}, \nonumber \\
        & E_t^+ \coloneqq \{(x', x_N) \in {\mathbb{R}}^N \ : \ x' \in B_R, \ x_N > g_t(x')\}. \nonumber 
    \end{align}
    We observe that the map $F_t: \overline{C_R^+} \to \overline{E_t^+}$ given by the formula
    \begin{equation*}
        F_t(x', x_N) = (x', x_N + g_t(x')), \quad (x', x_N) \in C_R^+
    \end{equation*}
    is a diffeomorphism, whose Jacobian matrix is given by
    \begin{equation*}
        J F_t(x', x_N) = 
        \begin{bmatrix}
            I_{N - 1} & 0_{N - 1}^T \\
            \nabla_{x'} g_t(x') & 1
        \end{bmatrix}
        ,
    \end{equation*}
    where $I_{N - 1}$ is the identity matrix of order $N - 1$ and $0_{N - 1}^T$ is the transpose of the null vector in ${\mathbb{R}}^{N - 1}$. In addition, it easily follows that 
    \begin{equation*}
        \det J F_t \equiv 1
    \end{equation*}
    and that the inverse map is given by
    \begin{equation*}
        F_t^{-1}(x', x_N) = (x', x_N - g_t(x')), \quad (x', x_N) \in E_t^+
    \end{equation*}
    
    
    We remark that $J F_t$ and $JF_t^{-1}$ do not depend on $x_N$. Moreover, both linear operators $J F_t$ and $JF_t^{-1}$ converge to the identity as $t \to + \infty$: in view of \eqref{eq:cone_nabla_xprime_g_t_vanishes}, it holds that
    \begin{equation}
        \label{eq:cone_JFt_to_id}
        \lim_{t \to + \infty} \|J F_t - I_N\| = \lim_{t \to + \infty} \|J F_t^{-1} - I_N\| = 0.
    \end{equation}

    Now we proceed to prove \eqref{eq:cone_bound_lambda_hemisphere}.

    Let $B^+ \subset {\mathbb{R}}^N_+$ be the $N$-dimensional half-ball such that $|B^+| = m$, thought of as contained in the upper half-space delimited by $T_q{\partial{\Sigma_D}} \simeq {\mathbb{R}}^N_+$. Let $u_{B^+}$ be the first $L^2$-normalized eigenfunction of $B^+$, so that
    \begin{equation*}
        \lambda_1(B^+) = \int_{B^+} |\nabla u_{B^+}|^2 \ dx = \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m).
    \end{equation*}

    Let $B_R \subset T_q {\partial{\Sigma_D}}$ be an $(N - 1)$-dimensional ball with a radius $R > 0$ large enough so that
    \begin{equation*}
        B^+ \subset B_R \times [0, + \infty).
    \end{equation*}
    For $t > 0$, we set 
    \begin{equation*}
        v_t = u_{B^+} \circ F_t^{-1}.
    \end{equation*}
    Henceforth we consider $t > 0$ large enough so that, by construction, $v_t \in H_0^1(F_t(B^+); {\Sigma_D} - tq)$.

    Since, as already noted, $\det JF_t \equiv 1$, it easily follows by the theorem of change of variables in the integral that 
    \begin{equation*}
        |F_t(B^+)| = |B^+| = m
    \end{equation*}
    and
    \begin{equation*}
        \int_{F_t(B^+)} v_t^2 \ dx = \int_{B^+} u_{B^+}^2 \ dx = 1.
    \end{equation*}
    Moreover, setting
    \begin{equation*}
        M_t \coloneqq \max_{(x', x_N) \in \overline{E_t^+}} \max_{|\zeta| \leq 1} |JF_t(x', x_N)^{-1} \zeta| = \max_{(x', x_N) \in \overline{E_t^+}} \|J F_t(x', x_N)^{-1}\|
    \end{equation*}
    and observing that 
    \begin{equation*}
        \nabla v_t(x', x_N) = [JF_t(x', x_N)^{-1}]^T \nabla u_{B^+}(F_t^{-1}(x', x_N)) \quad \text{ a.e. in } F_t(B^+),
    \end{equation*}
    we obtain
    \begin{equation*}
        \int_{F_t(B^+)} |\nabla v_t|^2 \ dx \leq M_t^2 \int_{B^+} |\nabla u_{B^+}|^2 \ dx.
    \end{equation*}
    Moreover, since \eqref{eq:cone_JFt_to_id} holds, then 
    \begin{equation*}
        \lim_{t \to + \infty} M_t = 1.
    \end{equation*}

    To conclude, we note that
    \begin{align}
        \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m) 
        & = \int_{B^+} |\nabla u_{B^+}|^2 \ dx \nonumber \\
        & \geq \frac{1}{M_t^2} \int_{F_t(B^+)} |\nabla v_t|^2 \ dx \nonumber \\
        & \geq \frac{1}{M_t^2} \lambda_1(F_t(B^+)) \nonumber \\
        & \geq \frac{1}{M_t^2} \lambda({\Sigma_D}, m). \nonumber
    \end{align}
    Passing to the limit as $t \to \infty$ yields \eqref{eq:cone_bound_lambda_hemisphere} and concludes the proof.

Let $\{\Omega_n\}_{n \in \mathbb N}$ be a minimizing sequence for \eqref{eq:cone_min_A}, that is, a sequence of quasi-open sets such that $|\Omega_n| = m$ for every $n \in \mathbb{N}$ and

        \lambda_1(\Omega_n) = \int_{\Omega_n} |\nabla u_n|^2 \ dx \to \lambda({\Sigma_D}, m) 
    \end{equation*}
    as $n \to \infty$, where $u_n \in H_0^1(\Omega_n; {\Sigma_D})$ is the corresponding positive $L^2$-normalized eigenfunction (that is, $\|u_n\|_2 = 1$ for every $n \in {\mathbb{N}}$).

    By definition, the sequence $\{u_n\}_{n \in {\mathbb{N}}}$ is bounded in $H^1({\Sigma_D})$. Since $\|u_n\|_2 = 1$ for every $n \in {\mathbb{N}}$, we can apply a slight modification of the concentration-compactness lemma (\cite[Lemma III.1]{Lions1984}) to account for working in the Sobolev space $H^1({\Sigma_D})$ instead of $H^1({\mathbb{R}}^N)$. More precisely, it holds that there exists a subsequence $\{u_{n_k}\}_{k \in {\mathbb{N}}}$ satisfying one of the following possibilities:
    \begin{enumerate}[label=(\roman*)]
        \item Compactness: there exists a sequence of points $y_k \in {\Sigma_D}$ such that 
        \begin{equation}
            \label{eq:cone_compactness}
            \forall \varepsilon > 0 \ \exists R > 0 \ \text{such that} \ \int_{B_R(y_k) \cap {\Sigma_D}} u_{n_k}^2 \ dx \geq 1 - \varepsilon \ \forall k \in {\mathbb{N}};
        \end{equation}

        \item Vanishing:
        \begin{equation}
            \label{eq:cone_vanishing}
            \lim_{k \to \infty} \sup_{y \in {\Sigma_D}} \int_{B_R(y) \cap {\Sigma_D}} u_{n_k}^2 \ dx = 0 \quad \forall R > 0;
        \end{equation}

        \item Dichotomy: there exists $\alpha \in (0, 1)$ such that for all $\varepsilon > 0$ there exist $k_0 \geq 1$ and two sequences $\{u_{1, k}\}_{k \in {\mathbb{N}}} \subset H^1({\Sigma_D})$, $\{u_{2, k}\}_{k \in {\mathbb{N}}} \subset H^1({\Sigma_D})$, bounded in $H^1({\Sigma_D})$, such that for all $k \geq k_0$ it holds
        \begin{align}
            & \|u_{n_k} - u_{1, k} - u_{2, k}\|_{L^2({\Sigma_D})} \leq 4\varepsilon, \nonumber \\
            & \left|\int_{\Sigma_D} u_{1, k}^2 \ dx - \alpha\right| \leq \varepsilon, \quad \left|\int_{\Sigma_D} u_{2, k}^2 \ dx - (1 - \alpha) \right| \leq \varepsilon, \nonumber \\
            & \text{dist} ({\text{supp }} u_{1, k}, {\text{supp }} u_{2, k}) \to + \infty \quad \text{ as } k \to \infty, \nonumber \\
            & \liminf_{k \to \infty} \int_{\Sigma_D} (|\nabla u_{n_k}|^2 - |\nabla u_{1, k}|^2 - |\nabla u_{2, k}|^2) \ dx \geq 0. \nonumber
        \end{align}
    \end{enumerate}

    We now show that, under our assumptions, only the compactness situation may occur. For simplicity, we denote the subsequence $\{u_{n_k}\}_{k \in {\mathbb{N}}}$ by $\{u_k\}_{k \in {\mathbb{N}}}$, as well as $\Omega_k = \{u_k > 0\}$.

    To rule out the vanishing possibility, we proceed by contradiction. Hence we assume that \eqref{eq:cone_vanishing} holds. Then it follows by \cite[Lemma I.1]{Lions1984CCb} (see also \cite[Lemma 1.21]{Willem1996}), whose proof can be easily adapted to the case of the Sobolev space $H^1({\Sigma_D})$ (since ${\Sigma_D}$ satisfies the cone condition), that $u_k \to 0$ strongly in $L^p({\Sigma_D})$ as $k \to \infty$ for every $p \in (2, 2^*)$. Since $|\Omega_k| = m$, then Hölder's inequality would imply
    \begin{equation*}
        1 = \|u_k\|_2 \leq |\{u_k > 0\}|^{\frac{1}{2} - \frac{1}{p}} \|u_k\|_p \to 0 \quad \text{ as } k \to \infty,
    \end{equation*}
    absurd. Hence vanishing does not occur.

    With the aim of ruling out dichotomy, we observe that a diagonal argument together with the construction performed in the proof of \cite[Lemma III.1]{Lions1984} allows us to assume that the sequences $\{u_{1, k}\}_{k \in {\mathbb{N}}}$ and $\{u_{2, k}\}_{k \in {\mathbb{N}}}$ satisfy 
    \begin{align}
        & {\text{supp }} u_{1, k} \cup {\text{supp }} u_{2, k} \subseteq {\text{supp }} u_{k} \quad \forall k \in {\mathbb{N}} \nonumber \\
        & \|u_{k} - u_{1, k} - u_{2, k}\|_{L^2({\Sigma_D})} \to 0 \quad \text{ as } k \to \infty, \nonumber \\
        & \int_{\Sigma_D} u_{1, k}^2 \ dx \to \alpha, \quad \int_{\Sigma_D} u_{2, k}^2 \ dx \to (1 - \alpha), \quad \text{ as } k \to \infty, \label{eq:cone_dichotomy} \\
        & \text{dist} ({\text{supp }} u_{1, k}, {\text{supp }} u_{2, k}) \to + \infty \quad \text{ as } k \to \infty, \nonumber \\
        & \liminf_{k \to \infty} \int_{\Sigma_D} (|\nabla u_{k}|^2 - |\nabla u_{1, k}|^2 - |\nabla u_{2, k}|^2) \ dx \geq 0. \nonumber
    \end{align}
    Moreover, the functions $u_{1, k}$ and $u_{2, k}$ can be assumed to be nonnegative for every $k \in {\mathbb{N}}$. Let us set $\Omega_{i, k} = \{u_{i, k} > 0\}$, for $i = 1, 2$, and define
    \begin{equation*}
        c_i \coloneqq \liminf_{k \to \infty} |\Omega_{i, k}|, \quad i = 1, 2.
    \end{equation*}
    We claim that $c_i > 0$ for $i = 1, 2$. Indeed, combining Hölder's and Sobolev inequalities, we obtain:
    \begin{align}
        \int_{{\Sigma_D}} u_{i, k}^2 \ dx 
        & \leq \left(\int_{\Sigma_D} u_{i, k}^{2^*} \right)^{\frac{2}{2^*}} |\Omega_{i, k}|^{\frac{2}{N}} \nonumber \\
        & \leq C(N, {\Sigma_D}) |\Omega_{i, k}|^{\frac{2}{N}} \int_{\Sigma_D} |\nabla u_{i, k}|^2 \ dx. \nonumber
    \end{align}
    Should we have $c_i = 0$, we would be in contradiction with \eqref{eq:cone_dichotomy}. Next, arguing as in \cite[Section 3.3]{Bucur2000}, we obtain that $u_{1, k} + u_{2, k}$ is a minimizing sequence for \eqref{eq:cone_min_A}: since $u_{i, k} \in H_0^1(\Omega_k; {\Sigma_D})$, $i = 1, 2$, for every $k \in {\mathbb{N}}$, we have
    \begin{align}
        \int_{\Sigma_D} |\nabla u_k - \nabla u_{1, k} - \nabla u_{2, k}|^2 \ dx 
        & = \int_{\Sigma_D} |\nabla u_k|^2 - 2 \int_{\Sigma_D} \nabla u_k \nabla (u_{1, k} + u_{2, k}) \ dx \nonumber \\
        & \quad + \int_{\Sigma_D} |\nabla (u_{1, k} + u_{2, k})|^2 \ dx \nonumber \\
        & = 2 \int_{\Sigma_D} \lambda_1(\Omega_k) u_k (u_k - u_{1, k} - u_{2, k}) \ dx \nonumber \\
        & \quad - \int_{\Sigma_D} \left(|\nabla u_k|^2 - |\nabla (u_{1, k} + u_{2, k})|^2\right) \ dx \nonumber \\
        & \leq C \|u_k\|_{L^2({\Sigma_D})}\|u_k - u_{1, k} - u_{2, k}\|_{L^2({\Sigma_D})} \nonumber \\
        & \quad - \int_{\Sigma_D} \left(|\nabla u_k|^2 - |\nabla (u_{1, k} + u_{2, k})|^2\right) \ dx, \nonumber 
    \end{align}    
    where $C > 0$ is a fixed constant that does not depend on $k$ (since the sequence $\{\lambda_1(\Omega_k)\}_{k \in {\mathbb{N}}}$ is bounded). Since, by \eqref{eq:cone_dichotomy},
    \begin{equation*}
        0 \leq \lim_{k \to \infty} C \|u_k\|_{L^2({\Sigma_D})} \|u_k - u_{1, k} - u_{2, k}\|_{L^2({\Sigma_D})} = 0
    \end{equation*}
    and
    \begin{equation*}
        \limsup_{k \to \infty} \left(- \int_{\Sigma_D} \left(|\nabla u_k|^2 - |\nabla (u_{1, k} + u_{2, k})|^2\right) \right) \leq 0,
    \end{equation*}
    it follows that
    \begin{equation*}
        \frac{\displaystyle \int_{\Sigma_D} |\nabla (u_{1, k} + u_{2, k})|^2 \ dx}{\displaystyle \int_{\Sigma_D} (u_{1, k}^2 + u_{2, k}^2)} \to \lambda({\Sigma_D}, m) \quad \text{ as } k \to \infty.
    \end{equation*}
    Now, recalling the inequality
    \begin{equation*}
        \frac{a + b}{c + d} \geq \min\left\{\frac{a}{c}, \frac{b}{d}\right\}, \quad \forall a, b, c, d > 0,
    \end{equation*}
    it follows that one of the sequences $\Omega_{i, k}$, $i = 1, 2$, is minimizing for $\eqref{eq:cone_min_A}$. Since $|\Omega_{i, k}| < m - \varepsilon$ for some $\varepsilon > 0$, the strict monotonicity \eqref{eq:cone_monotonicity} would allow us to find a quasi-open set $\Omega'$ of measure $m$ such that $\lambda_1(\Omega') < \lambda({\Sigma_D}, m)$, a contradiction.

    We have thus proved that the vanishing and dichotomy cases cannot occur, so that compactness holds for our sequence $\{u_k\}_{k \in {\mathbb{N}}}$. Roughly speaking, \eqref{eq:cone_compactness} means that the mass of $u_k$ concentrates in a ball around $y_k$. As the next step shows, the same is true for the eigenvalue as well (in other words, the tails of $\Omega_k$ do not play an important role). 

    Precisely, we claim that there exists a constant $\Lambda > 0$ such that for every fixed $\varepsilon > 0$ there exist $R_\varepsilon > 1$ and $k_\varepsilon \in {\mathbb{N}}$, both depending only on $\varepsilon$, such that
    \begin{equation}
        \label{eq:cone_concentration_eigenvalue}
        \lambda_1(\Omega_k) \geq \lambda_1(B_{2R}(y_k) \cap \Omega_k) - \Lambda \varepsilon \quad \forall k \geq k_\varepsilon, \ \forall R \geq R_\varepsilon.
    \end{equation}
    To prove the claim, let us fix $\varepsilon > 0$ and let $R > 0$ be the corresponding radius given by the concentration-compactness theorem such that \eqref{eq:cone_compactness} holds. Let $\varphi$ be a fixed cut-off function with the following properties:
    \begin{align}
        & \varphi \in C_c^\infty({\mathbb{R}}^N) \nonumber \\
        & 0 \leq \varphi \leq 1 \text{ in } {\mathbb{R}}^N, \nonumber \\
        & \varphi \equiv 1 \text{ in } B_R(O), \nonumber \\
        & \varphi \equiv 0 \text{ in } {\mathbb{R}}^N \setminus B_{2R}(O), \nonumber \\
        & |\nabla \varphi| \leq \frac{C_0}{R} \text{ in } {\mathbb{R}}^N, \nonumber 
    \end{align}
    where $C_0$ is a constant independent of $R$. Let $\varphi_k$ be the translation of $\varphi$ with center at $y_k$, that is, 
    \begin{equation*}
    \varphi_k(x) = \varphi(x - y_k), \quad x \in {\mathbb{R}}^N.
    \end{equation*}
    Now we note that
    \begin{align}
        \int_{\Sigma_D} |\nabla u_k|^2 \ dx
        & \geq \int_{\Sigma_D} |\nabla u_k|^2 \varphi_k^2 \ dx \nonumber \\
        & = \int_{\Sigma_D} |\nabla (u_k \varphi_k)|^2 \ dx - 2 \int_{\Sigma_D} (u_k \varphi_k) (\nabla u_k \cdot \nabla \varphi_k) \ dx - \int_{\Sigma_D} u_k^2 |\nabla \varphi_k|^2 \ dx. \nonumber
    \end{align}
    By Hölder's inequality, we have
    \begin{align}
        \int_{\Sigma_D} (u_k \varphi_k) (\nabla u_k \cdot \nabla \varphi_k) \ dx
        & \leq \|\varphi_k\|_\infty \|\nabla \varphi_k\|_\infty \int_{\Sigma_D} |u_k| |\nabla u_k| \ dx \nonumber \\
        & \leq \frac{C_0}{R} \|u_k\|_{L^2({\Sigma_D})} \|\nabla u_k\|_{L^2({\Sigma_D})} \nonumber \\
        & \leq \frac{C_0 C_1}{R}, \nonumber
    \end{align}
    where also $C_1$ is independent of $k$ and $R$. Similarly,
    \begin{equation*}
        \int_{\Sigma_D} u_k^2 |\nabla \varphi_k|^2 \ dx \leq \frac{C_0^2}{R^2}.
    \end{equation*}
    Without loss of generality, we can assume that $R > 1$. Hence
    \begin{equation*}
        \int_{\Sigma_D} |\nabla u_k|^2 \ dx \geq \int_{\Sigma_D} |\nabla (u_k \varphi_k)|^2 \ dx - \frac{C_2}{R},
    \end{equation*}
    where $C_2 > 0$ is independent of $k$ and $R$.
    Next, since $\varphi_k^2 \equiv 1$ in $B_R(y_k)$ and in view of \eqref{eq:cone_compactness}, we have:
    \begin{align}
        \int_{\Sigma_D} u_k^2 \ dx 
        & = \int_{\Sigma_D} u_k^2 \varphi_k^2 \ dx + \underbrace{\int_{{\Sigma_D} \cap B_R(y_k)} u_k^2 (1 - \varphi_k^2) \ dx}_{ = 0} + \int_{{\Sigma_D} \setminus B_R(y_k)} u_k^2 (1- \varphi_k^2) \ dx \nonumber \\
        & \leq \int_{\Sigma_D} (u_k \varphi_k)^2 \ dx + \varepsilon. \nonumber
    \end{align}
    Since $u_k \varphi_k \in H_0^1(B_{2R}(y_k) \cap \Omega_k; {\Sigma_D})$, it is an admissible test function for the Rayleigh quotient. We have:
    \begin{align}
        \lambda_1(B_{2R}(y_k) \cap \Omega_k)
        & \leq \frac{\displaystyle \int_{\Sigma_D} |\nabla (u_k \varphi_k)|^2 \ dx}{\displaystyle \int_{\Sigma_D} (u_k \varphi_k)^2 \ dx} \nonumber \\
        & \leq \frac{\displaystyle \int_{\Sigma_D} |\nabla u_k|^2 \ dx + \frac{C_2}{R}}{1 - \varepsilon}. \nonumber 
    \end{align}
    Since $C_2$ is independent of $k$ and $R$, we can assume that $\mfrac{C_2}{R} < \varepsilon$ (up to taking a larger $R$). Then, by a simple Taylor series expansion, we obtain
    \begin{align}
        \lambda_1(B_{2R}(y_k) \cap \Omega_k) \leq \lambda_1(\Omega_k) + (\lambda_1(\Omega_k) + 1) \varepsilon + o(\varepsilon). \nonumber
    \end{align}
    Choosing $\Lambda = \lambda({\Sigma_D}, m) + K$, for a large constant $K > 0$, we readily obtain \eqref{eq:cone_concentration_eigenvalue}.

    Our next step is to show that the sequence of points $\{y_k\}_{k \in {\mathbb{N}}} \subset {\Sigma_D}$ is bounded. The idea is that if we assume that the sequence is unbounded, using the previous step and the fact that the cone ``flattens out" at infinity, we converge in some sense to the eigenvalue of a half-ball, contradicting the assumption \eqref{eq:cone_ineq}.

    For the sake of contradiction, let us assume that there exists a subsequence (still indexed by $k$) such that
    \begin{equation*}
        |y_k| \to + \infty \quad \text{ as } k \to \infty.
    \end{equation*}
    By our key assumption \eqref{eq:cone_ineq}, there exists $\varepsilon > 0$ sufficiently small such that
    \begin{equation}
        \label{eq:cone_contradiction_assumption_y_k}
        \lambda({\Sigma_D}, m) + \Lambda \varepsilon < \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m),
    \end{equation}
    where $\Lambda$ is the constant from \eqref{eq:cone_concentration_eigenvalue}. Moreover, by \eqref{eq:cone_concentration_eigenvalue} there exists $R_\varepsilon > 0$ sufficiently large, depending only on $\varepsilon$, such that for all sufficiently large $k$ it holds
    \begin{equation}
        \label{eq:cone_ineq_y_k}
        \lambda_1(\Omega_k) \geq \lambda_1({B_{2R}(y_k)} \cap \Omega_k) - \Lambda \varepsilon. 
    \end{equation}
    Note that there exists $\bar k \in {\mathbb{N}}$ such that $d(y_k, {\partial{\Sigma_D}}) \leq 2R$ for all $k \geq \bar k$, because otherwise we would have
    \begin{equation*}
        \lambda_1(B_{2R}(y_k) \cap \Omega_k) \geq \lambda_1(B^m) > \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m), 
    \end{equation*}
    where $B^m$ is the ball of measure $m$, contradicting \eqref{eq:cone_ineq_y_k}-\eqref{eq:cone_concentration_eigenvalue}. Therefore there exists a sequence $\{z_k\}_{k \in {\mathbb{N}}} \subset {\partial{\Sigma_D}}$ such that for all $k \geq \bar k$ it holds
    \begin{align}
        & z_k \in \left(\overline{{B_{2R}(y_k)} \cap \Omega_k} \cap {\partial{\Sigma_D}}\right), \nonumber \\
        & {B_{2R}(y_k)} \subset {B_{4R}(z_k)}. \nonumber
    \end{align}
    Since the functional $\Omega \mapsto \lambda_1(\Omega)$ is monotone with respect to inclusion and $H_0^1({B_{2R}(y_k)} \cap \Omega_k; {\Sigma_D}) \subset H_0^1({B_{4R}(z_k)} \cap \Omega_k; {\Sigma_D})$, we have that
    \begin{equation}
        \label{eq:cone_ineq_monot}
        \lambda_1({B_{2R}(y_k)} \cap \Omega_k) \geq \lambda_1({B_{4R}(z_k)} \cap \Omega_k) \quad \forall k \geq \bar k.
    \end{equation}
    We now show that, up to a further subsequence still indexed by $k$, it holds
    \begin{equation}
        \label{eq:cone_claim}
        \lambda_1({B_{4R}(z_k)} \cap \Omega_k) \geq \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m) +  o(1) \quad \text{ as } k \to \infty. 
    \end{equation}
    To prove the claim \eqref{eq:cone_claim}, we use the geometric construction and considerations presented in the proof of Proposition \ref{prop:cone_bound_lambda_hemisphere}. We set
    \begin{equation*}
        q_k \coloneqq \frac{z_k}{|z_k|} \in \partial D, \quad k \in {\mathbb{N}}.
    \end{equation*}
    By the compactness of $\partial D$ in ${\mathbb{S}^{N - 1}}$, there exists $q \in \partial D$ such that $q_k \to q$ (up to a subsequence still indexed by $k$), where the convergence is to be understood with respect to the geodesic distance on ${\mathbb{S}^{N - 1}}$. As in the proof of Proposition \ref{prop:cone_bound_lambda_hemisphere}, we take $U \subset T_q{\partial{\Sigma_D}}$ to be a convex neighborhood of the origin in the tangent plane and a map $g: \overline{U} \to {\mathbb{R}}$ such that $g(0) = 0$, $\nabla_{x'} g(0) = 0$, and the graph of $g$ on $U$ is a neighborhood $V$ of $q$ on ${\partial{\Sigma_D}}$. As before, we use the coordinates $(x', x_N)$ with respect to the orthonormal frame $\{v_1, \ldots, v_{N - 1}, - \nu(q)\}$, where $\{v_i\}_{i = 1, \ldots, N - 1}$ is an orthonormal frame of $T_q{\partial{\Sigma_D}}$. Let $\Pi: {\mathbb{R}}^N \to T_q{\partial{\Sigma_D}}$ be the orthogonal projection operator. Since $q_k \to q$, then for all sufficiently large $k$ we have
    \begin{align}
        & q_k \in V, \nonumber \\
        & \Pi(q_k - q) \in U, \nonumber
    \end{align}
    and moreover
    \begin{equation*}
        \Pi(q_k - q) \to 0 \quad \text{ as } k \to \infty.
    \end{equation*}
    For each $k \in {\mathbb{N}}$, let $\bar x_k'$ be the coordinates of $\Pi(q_k - q)$:
    \begin{equation*}
        \bar x_k' = (\bar x_{1, k}', \ldots, \bar x_{N - 1, k}') = \Pi(q_k - q) \in T_q{\partial{\Sigma_D}}. 
    \end{equation*}
    In addition, let us set, for each $k \in {\mathbb{N}}$, the translated domain
    \begin{equation*}
        U_k \coloneqq U - \bar x_k' \nonumber
    \end{equation*}
    and the function $g_k: U_k \to {\mathbb{R}}$ given by
    \begin{equation*}
        g_k(x') = g(x' + \bar x_k') - g(\bar x_k'), \quad x' \in U_k.
    \end{equation*}
    It is clear that $V - q_k$ is the graph of $g_k$. Now, by construction, $\bar x_k' \to 0$ in $T_q{\partial{\Sigma_D}}$ as $k \to \infty$, and therefore there exists a ball $B_{\bar R} = B_{\bar R}(0)$ in $T_q{\partial{\Sigma_D}}$ such that $B_{\bar R} \subset U_k$ for all sufficiently large $k$. Let us set $t_k = |z_k|$ for $k \in {\mathbb{N}}$. By definition, $t_k \to + \infty$ as $k \to \infty$, and therefore $t_k U_k$ covers $T_q{\partial{\Sigma_D}}$ as $k \to \infty$. Next, for $k \in {\mathbb{N}}$ we define the scaled map
    \begin{align}
        h_k(x') = t_k g_k \left(\frac{x'}{t_k}\right) = t_k \left(g\left(\frac{x'}{t_k} + \bar x_k' \right) - g(\bar x_k') \right), \quad x' \in t_k U_k. \nonumber
    \end{align}
    Arguing as in \eqref{eq:cone_mean_value_theorem}, we obtain
    \begin{align}
        \frac{\partial h_k}{\partial x_i'}(x')
        & = \frac{\partial g}{\partial x_i'}\left(\frac{x'}{t_k} + \bar x_k' \right) \nonumber \\
        & = \frac{\partial g}{\partial x_i'}(\bar x_k') +  \left(\nabla_{x'} \frac{\partial g}{\partial x_i'} \right) (\eta_k + \bar x_k') \cdot \frac{x'}{t_k}, \nonumber
    \end{align}
    where $\eta_k$ belongs to the segment joining $\bar x_k'$ and $\bar x_k' + \mfrac{x'}{t_k}$, for every $k \in {\mathbb{N}}$. Consider a ball ${B_{\widetilde R}(0)}$ in $T_q{\partial{\Sigma_D}}$, where $\widetilde R$ is to be chosen later and independently of $k$. Since $t_k U_k$ covers $T_q{\partial{\Sigma_D}}$ as $k \to \infty$, then ${B_{\widetilde R}(0)} \subset t_k U_k$ for all sufficiently large $k$. Now, since $g$ is smooth, $\bar x_k' \to 0$ in $T_q{\partial{\Sigma_D}}$ as $k \to \infty$ and $\nabla_{x'} g(0) = 0$, arguing as in \eqref{eq:cone_nabla_xprime_g_t_vanishes} we obtain that
    \begin{equation*}
        \lim_{k \to \infty} \max_{x' \in \overline{{B_{\widetilde R}(0)}}} |\nabla_{x'} h_k(x')| = 0.
    \end{equation*}
    Recalling the notations
    \begin{align}
        & C_{\widetilde R}^+ \coloneqq \{(x', x_N) \in {\mathbb{R}}^N \ : \ x' \in {B_{\widetilde R}(0)}, \ x_N > 0\}, \nonumber \\
        & E_k^+ \coloneqq \{(x', x_N) \in {\mathbb{R}}^N \ : \ x' \in {B_{\widetilde R}(0)}, \ x_N > h_k(x')\}, \nonumber 
    \end{align}
    we consider the diffeomorphism 
    \begin{equation*}
        F_k: \overline{C_{\widetilde R}^+} \to \overline{E_k^+}
    \end{equation*}
    given by
    \begin{equation*}
        F_k(x', x_N) = (x', x_N + h_k(x')), \quad (x', x_N) \in \overline{C_{\widetilde R}^+}.
    \end{equation*}
    Note that for all sufficiently large $k$ it holds
    \begin{equation*}
        \left(\overline{{B_{4R}(z_k)} \cap \Omega_k} \cap {\partial{\Sigma_D}}\right) - z_k \subset t_k(V - q_k).
    \end{equation*}
    For simplicity, we set
    \begin{equation*}
        {\Omega_{R, k}} \coloneqq ({B_{4R}(z_k)} \cap \Omega_k) - z_k.
    \end{equation*}
    Since ${\Omega_{R, k}}$ is uniformly bounded and $t_kU_k$ covers $T_q{\partial{\Sigma_D}}$ as $k \to \infty$, then there exists $\widetilde R > 0$ independent of $k$ such that
    \begin{equation*}
        F_k^{-1}(\overline{{\Omega_{R, k}}}) \subset {B_{\widetilde R}(0)} \times [0, + \infty).
    \end{equation*}
    We denote by ${\widetilde{u}_k} \in H_0^1({\Omega_{R, k}}; {\Sigma_D} - z_k)$ the first positive $L^2$-normalized eigenfunction of ${\Omega_{R, k}}$, and set ${v_k} \coloneqq {\widetilde{u}_k} \circ F_k$. It then readily follows that ${v_k} \in H_0^1(F_k^{-1}({\Omega_{R, k}}); {\mathbb{R}}^N_+)$. As in the proof of Proposition \ref{prop:cone_bound_lambda_hemisphere}, we have that
    \begin{align}
        & |F_k^{-1}({\Omega_{R, k}})| = |{\Omega_{R, k}}| \leq m, \nonumber \\
        & \int_{{\Omega_{R, k}}} {\widetilde{u}_k}^2 \ dx = \int_{F_k^{-1}({\Omega_{R, k}})} {v_k}^2 \ dx = 1, \nonumber \\
        & \int_{F_k^{-1}({\Omega_{R, k}})} |\nabla {v_k}|^2 \ dx \leq M_k^2 \int_{{\Omega_{R, k}}} |\nabla {\widetilde{u}_k}|^2 \ dx, \nonumber
    \end{align}
    where
    \begin{equation*}
        M_k = \max_{(x', x_N) \in \overline{C_{\widetilde R}^+}} \max_{|\zeta| \leq 1} |JF_k(x', x_N) \zeta| \to 1 \quad \text{ as } k \to \infty.
    \end{equation*}
    Now, in view of Remark \ref{rem:cone_sphere} and the monotonicity of the eigenvalue with respect to a scaling of the domain, we have that
    \begin{align}
        \lambda_1({\Omega_{R, k}})
        & \geq \frac{1}{M_k^2} \int_{F_k^{-1}({\Omega_{R, k}})} |\nabla {v_k}|^2 \ dx \nonumber \\
        & \geq \frac{1}{1 + o(1)} \lambda_1(F_k^{-1}({\Omega_{R, k}})) \nonumber \\
        & \geq \frac{1}{1 + o(1)} \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, |{\Omega_{R, k}}|) \nonumber \\
        & \geq \frac{1}{1 + o(1)} \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m) \nonumber \\
        & = \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m) + o(1) \nonumber
    \end{align}    
    as $k \to \infty$, which is precisely \eqref{eq:cone_claim}. Now, combining \eqref{eq:cone_ineq_y_k}, \eqref{eq:cone_ineq_monot}, \eqref{eq:cone_claim} we obtain
    \begin{equation*}
        \lambda_1(\Omega_k) \geq \lambda(\Sigma_{\mathbb{S}_+^{N - 1}}, m) + o(1) - \Lambda \varepsilon \quad \text{ as } k \to \infty,
    \end{equation*}
    contradicting \eqref{eq:cone_contradiction_assumption_y_k}. It therefore follows that the sequence $\{y_k\}_{k \in {\mathbb{N}}} \subset {\Sigma_D}$ is bounded.

    The boundedness of the sequence $\{y_k\}_{k \in {\mathbb{N}}}$ allows us to prove that the sequence of eigenfunctions $\{u_k\}_{k \in {\mathbb{N}}}$ admits a strongly convergent subsequence in $L^2({\Sigma_D})$.

    To the end of proving this claim, we begin by showing that 
    \begin{equation}
        \label{eq:cone_exterior_vanishing}
        \lim_{R \to \infty} \sup_{k \in {\mathbb{N}}} \int_{{\Sigma_D} \setminus B_R(O)} u_k^2 \ dx = 0.
    \end{equation}
    We proceed by contradiction. Should \eqref{eq:cone_exterior_vanishing} not hold, there would exist $\varepsilon' > 0$ and a sequence $\{R_j\}_{j \in {\mathbb{N}}} \subset {\mathbb{R}}^+$, with $R_j \to + \infty$ as $j \to \infty$, and a subsequence $\{u_j\}_{j \in {\mathbb{N}}}$ of eigenfunctions such that
    \begin{equation*}
        \int_{{\Sigma_D} \setminus B_{R_j}(O)} u_j^2 \ dx \geq \frac{\varepsilon'}{2} \quad \forall j \in {\mathbb{N}}.
    \end{equation*}
    On the other hand, recalling \eqref{eq:cone_compactness}, if we take $\varepsilon = \mfrac{\varepsilon'}{4}$ and $R_\varepsilon$ be the corresponding radius given by \eqref{eq:cone_compactness}, then for all $k \in {\mathbb{N}}$ we have
    \begin{equation*}
        \int_{B_{R_\varepsilon}(y_k) \cap {\Sigma_D}} u_k^2 \ dx \geq 1 - \frac{\varepsilon'}{4}.
    \end{equation*}
    Now we make use of the boundedness of the sequence $\{y_k\}_{k \in {\mathbb{N}}}$. It implies that there exists $\bar R > 0$ independent of $k$ such that $B_{R_\varepsilon}(y_k) \subset B_{\bar R}(O)$ for all $k \in {\mathbb{N}}$. Hence
    \begin{equation*}
        \int_{B_{\bar R}(O) \cap {\Sigma_D}} u_k^2 \ dx \geq \int_{B_{R_\varepsilon}(y_k) \cap {\Sigma_D}} u_k^2 \ dx \geq 1 - \frac{\varepsilon'}{4}
    \end{equation*}
    for all $k$ sufficiently large. However, since $R_j \to + \infty$, there exists $j_0 \in {\mathbb{N}}$ such that $R_j \geq \bar R$ for all $j \geq j_0$, and the corresponding eigenfunctions satisfy
    \begin{align}
        \int_{\Sigma_D} u_j^2 \ dx 
        & = \int_{B_{R_j}(O) \cap {\Sigma_D}} u_j^2 \ dx + \int_{{\Sigma_D} \setminus B_{R_j}(O)} u_j^2 \ dx \nonumber \\
        & \geq 1 - \frac{\varepsilon'}{4} + \frac{\varepsilon'}{2} \nonumber \\
        & > 1, \nonumber
    \end{align}
    a contradiction with the constraint
    \begin{equation*}
        \int_{\Sigma_D} u_k^2 \ dx = 1 \quad \forall k \in {\mathbb{N}}. 
    \end{equation*}
    This proves \eqref{eq:cone_exterior_vanishing}, which in turn implies that for any $\varepsilon > 0$ there exists $R > 0$ such that
    \begin{equation*}
        \int_{{\Sigma_D} \setminus B_R(O)} u_k^2 \ dx < \varepsilon \quad \forall k \in {\mathbb{N}}.
    \end{equation*}
    Consider the truncated functions
    \begin{equation*}
        f_k \coloneqq \chi_{B_R(O)} u_k, \quad k \in {\mathbb{N}},
    \end{equation*}
    where $\chi_E$ denotes, as usual, the characteristic function of the set $E$. We readily obtain that the sequence $\{f_k\}_{k \in {\mathbb{N}}}$ is bounded in $H^1(B_R(O) \cap {\Sigma_D})$, because $\{u_k\}_{k \in {\mathbb{N}}}$ is bounded in $H^1({\Sigma_D})$, and therefore there exists $f \in H^1(B_R(O) \cap {\Sigma_D})$ such that
    \begin{equation*}
        f_k \rightharpoonup f \quad \text{ weakly in } H^1(B_R(O) \cap {\Sigma_D}) \quad \text{ as } k \to \infty.
    \end{equation*} 
    In addition, it is also immediate from \eqref{eq:cone_exterior_vanishing} that
    \begin{equation}
        \label{eq:cone_uk_vk}
        \|u_k - f_k\|_{L^2({\Sigma_D})} \leq \varepsilon \quad \forall k \in {\mathbb{N}}.
    \end{equation}
    Moreover, since $B_R(O) \cap {\Sigma_D}$ is bounded and satisfies the cone condition, the embedding
    \begin{equation*}
        H^1(B_R(O) \cap {\Sigma_D}) \hookrightarrow L^2(B_R(O) \cap {\Sigma_D})
    \end{equation*}
    is compact, and therefore $f_k \to f$ strongly in $L^2(B_R(O) \cap {\Sigma_D})$, and hence in $L^2({\Sigma_D})$ (up to a subsequence). This relative compactness of $\{f_k\}_{k \in {\mathbb{N}}}$, together with \eqref{eq:cone_uk_vk}, imply that the sequence $\{u_k\}_{k \in {\mathbb{N}}}$ can be covered with finitely many balls of radius $\varepsilon$. Since $\varepsilon$ is arbitrary, it follows (by definition) that $\{u_k\}_{k \in {\mathbb{N}}}$ is totally bounded in $L^2({\Sigma_D})$. Since $L^2({\Sigma_D})$ is a Banach space, then it follows that $\{u_k\}_{k \in {\mathbb{N}}}$ is relatively compact (\cite[Chapter 7]{Munkres2000}), i.e., there exists $u \in L^2({\Sigma_D})$ such that
    \begin{equation*}
        u_k \to u \quad \text{ strongly in } L^2({\Sigma_D}) \quad \text{ as } k \to \infty. 
    \end{equation*}

    We now come to the conclusion of the proof. Since the sequence $\{u_k\}_{k \in {\mathbb{N}}}$ is bounded in $H^1({\Sigma_D})$, it converges weakly in this space (up to a subsequence). By the previous step, $u_k \to u$ strongly in $L^2({\Sigma_D})$, hence $u$ is precisely the weak limit of $u_k$ in $H^1({\Sigma_D})$ as $k \to \infty$, since the embedding $H^1({\Sigma_D}) \hookrightarrow L^2({\Sigma_D})$ is a continuous linear map. It is clear that $u \geq 0$ in ${\Sigma_D}$. We set
    \begin{equation*}
        \Omega \coloneqq \{u > 0\}.
    \end{equation*}
    This set is a quasi-open subset of ${\Sigma_D}$ and, by Fatou's Lemma,
    \begin{equation*}
        |\Omega| = \int_{\Sigma_D} \chi_{\{u > 0\}} \leq \liminf_{k \to \infty} \int_{\Sigma_D} \chi_{\Omega_k} \ dx = m.
    \end{equation*}
    In addition,
    \begin{equation*}
        \|u\|_{H^1({\Sigma_D})} \leq \liminf_{k \to \infty} \|u_k\|_{H^1({\Sigma_D})}, 
    \end{equation*}
    which, combined with $\|u_k\|_{L^2({\Sigma_D})} \to \|u\|_{L^2({\Sigma_D})}$ as $k \to \infty$ leads to
    \begin{equation*}
        \lambda_1(\Omega) \leq \frac{\displaystyle \int_\Omega |\nabla u|^2 \ dx}{\displaystyle \int_\Omega u^2 \ dx} \leq \liminf_{k \to \infty} \frac{\displaystyle \int_{\Sigma_D} |\nabla u_k|^2 \ dx}{\displaystyle \int_{\Sigma_D} u_k^2 \ dx} = \liminf_{k \to \infty} \lambda_1(\Omega_k) = \lambda({\Sigma_D}, m).
    \end{equation*}
    Should $|\Omega| < m$, we would be able to consider a suitable scaling $t\Omega$ such that $|t\Omega| = m$, in which case we would have $\lambda_1(t\Omega) < \lambda({\Sigma_D}, m)$, absurd. Hence $|\Omega| = m$ and
    \begin{equation*}
        \lambda_1(\Omega) = \lambda({\Sigma_D}, m),
    \end{equation*}
    which completes the proof.

<!-- lemma-10: body_md -->
\label{lem:cone_equivalent_problems}
    For every $\tau > 0$, problem \eqref{eq:cone_min_B} is equivalent to
    \begin{equation}
        \label{eq:cone_min_equiv}
        \inf\{\lambda_1(\Omega) + \tau |\Omega| \ : \ \Omega \subset {\Sigma_D} \text{ is quasi-open}\},
    \end{equation}
    in the sense that if $\Omega$ is a solution to \eqref{eq:cone_min_B} (respectively, \eqref{eq:cone_min_equiv}), then there exists a scaling of $\Omega$ which is a solution for \eqref{eq:cone_min_equiv} (respectively, \eqref{eq:cone_min_B}).
<!-- end lemma-10 -->
<!-- proof-17: body_md -->
For any fixed quasi-open set $\Omega \subset {\Sigma_D}$ and any given $\tau > 0$, there exists an optimal scaling factor $\bar t$ that minimizes the function
    \begin{equation*}
        t \in (0, + \infty) \mapsto G(t\Omega) = t^{-2}\lambda_1(\Omega) + t^N \tau |\Omega|.
    \end{equation*}
    Indeed, $G(t\Omega)$ is smooth, $G(t\Omega) \to + \infty$ as $t \to 0^+$ and $G(t\Omega) \to + \infty$ as $t \to + \infty$. Since
    \begin{equation*}
        \frac{d}{dt} G(t \Omega) = - 2 t^{-3} \lambda_1(\Omega) + N t^{N - 1} \tau |\Omega|,
    \end{equation*}
    then
    \begin{equation*}
        \frac{d}{dt}G(t\Omega) = 0
    \end{equation*}
    has the unique root
    \begin{equation*}
        \bar t_\Omega = \left(\frac{2 \lambda_1(\Omega)}{N \tau |\Omega|}\right)^{\frac{1}{N + 2}}.
    \end{equation*}

    Now, note that
    \begin{align}
        G(\bar t \Omega) 
        & = \underbrace{\left(\left(\frac{N \tau}{2}\right)^{\frac{2}{N + 2}} + \tau^{\frac{2}{N + 2}}\left(\frac{2}{N}\right)^{\frac{N}{N + 2}}\right)}_{\coloneqq C_{N, \tau}} \lambda_1(\Omega)^{\frac{N}{N + 2}} |\Omega|^{\frac{2}{N + 2}} \nonumber \\
        & = C_{N, \tau} \left(\lambda_1(\Omega) |\Omega|^{\frac{2}{N}}\right)^{\frac{N}{N + 2}}. \nonumber
    \end{align}
    
    Since the map
    \begin{equation*}
        s \in (0, + \infty) \mapsto C_{N, \tau} s^{\frac{N}{N + 2}}
    \end{equation*}
    is strictly increasing, then minimizing $G(\bar t_\Omega \Omega)$ among all quasi-open sets $\Omega \subset {\Sigma_D}$ is equivalent to \eqref{eq:cone_min_B}. Consequently, if a quasi-open set $\Omega$ minimizes \eqref{eq:cone_min_B}, then a suitable rescaling of $\Omega$ minimizes \eqref{eq:cone_min_equiv}. On the other hand, if $\Omega$ is a minimizer for \eqref{eq:cone_min_equiv}, then $\bar t_\Omega = 1$. In particular, $\Omega$ is a minimizer for \eqref{eq:cone_min_B} with $m = |\Omega|$. Then, for $m \neq |\Omega|$, a simple scaling of $\Omega$ yields a minimizer for \eqref{eq:cone_min_B}.

\label{thm:cone_overdet_minimizer}

<!-- proof-24: body_md -->
We may argue as in the proof of \cite[Proposition 7.4]{IacopettiPacellaWeth2022}.

    Fix $x_0 \in {\Gamma_{0, \Omega}^*}$ and let $B_r(x_0)$ be a small ball such that
    \begin{equation*}
        B_{2r}(x_0) \subset {\Sigma_D}, \quad (B_{2r}(x_0) \cap {\Gamma_{0, \Omega}}) \subset {\Gamma_{0, \Omega}^*}.
    \end{equation*}
    Let $\varphi \in C_c^\infty(B_r(x_0))$ and $\bar \nu$ be an extension of the normal vector $\nu$ on ${\Gamma_{0, \Omega}}$ to a smooth vector field defined in $\overline{B_r(x_0)}$ (see, e.g., \cite[Section 5.4]{HenrotPierre2018} for the construction). Then we consider the smooth vector field
    \begin{equation*}
        V = \varphi \bar \nu.
    \end{equation*}
    By construction, $V: {\mathbb{R}}^N \to {\mathbb{R}}^N$ is a smooth vector field with support in $B_r(x_0)$, $V(O) = 0$ and $V(x) = 0 \in T_x{\partial{\Sigma_D}}$ for $x \in {\partial{\Sigma_D}} \setminus\{O\}$. Then we consider the induced deformations
    \begin{equation*}
        \Omega_t = \xi_t(\Omega), \quad t \in (-\delta, \delta)
    \end{equation*}
    where $\delta > 0$ is small enough such that the flow $\xi_t$ corresponding to $V$ preserves ${\partial{\Sigma_D}}$. In addition, we note that, up to choosing a smaller $\delta$, it holds that
    \begin{equation*}
        \xi_t(x) = x \quad \forall x \in {\Sigma_D} \setminus B_{\frac{3}{2}r}(x_0), \ t \in (-\delta, \delta).
    \end{equation*}

    For $t \in (-\delta, \delta)$, let $u_t \in H_0^1(\Omega_t, {\Sigma_D})$ be the positive $L^2$-normalized first eigenfunction of $\Omega_t$. The proof of Proposition \ref{prop:cone_first_derivative} can be easily adapted to this general perturbation, with even more regularity (cf. \cite[Proposition 5.3.10]{HenrotPierre2018}). We obtain that the map
    \begin{equation*}
        t \in (-\delta, \delta) \mapsto u_t \in H^1({\Sigma_D})
    \end{equation*}
    is differentiable, where we abuse notation to denote an $H^1({\Sigma_D})$ extension of $u_t$ (see \cite[Section 5.3.6]{HenrotPierre2018}). We also obtain that
    \begin{equation*}
        t \in (-\delta, \delta) \mapsto |\nabla u_t|^2 \in L^1({\Sigma_D})
    \end{equation*}
    is differentiable at $t = 0$. Let us point out that the derivative
    \begin{equation*}
        {\widetilde{u}} = \left. \frac{d}{dt}u_t \right|_{t = 0} \in H^1(\Omega)
    \end{equation*}
    is the solution of
    \begin{equation*}
        \left\{ 
            \begin{array}{rcll}
                - \Delta {\widetilde{u}} & = & \lambda_1(\Omega) {\widetilde{u}} + \lambda' u & \quad \text{ in } \Omega \\
                {\widetilde{u}} & = & -\mfrac{\partial u}{\partial \nu} (V \cdot \nu) & \quad \text{ on } {\Gamma_{0, \Omega}} \\
                \mfrac{\partial {\widetilde{u}}}{\partial \nu} & = & 0 & \quad \text{ on } {\Gamma_{\Omega}} \\
                \displaystyle \int_\Omega {\widetilde{u}} u \ dx & = & 0 &
            \end{array}
        \right.
        ,
    \end{equation*} 
    where
    \begin{equation*}
        \lambda' \coloneqq \left. \frac{d}{dt} \lambda_t \right|_{t = 0}
    \end{equation*}
    By standard regularity theory, ${\widetilde{u}} \in W^{2, 2}(\Omega \cap B_{\frac{3}{2}r}(x_0))$ and is smooth in $\Omega$.
    
    By standard elliptic regularity results, the first eigenfunction $u$ of $\Omega$ is smooth in $\Omega \cap B_r(x_0)$. In particular,
    \begin{equation*}
        u \in W^{2, 2}(\Omega \cap B_r(x_0)),
    \end{equation*}
    and therefore
    \begin{equation*}
        |\nabla u|^2 \varphi \bar \nu \in W^{1, 1}({\Sigma_D}, {\mathbb{R}}^N).
    \end{equation*}
    
    Then, we are able to apply Hadamard's formula (\cite[Theorem 5.2.2]{HenrotPierre2018}) to the function
    \begin{equation*}
        t \in (-\delta, \delta) \mapsto \lambda_t = \int_{\Omega_t}|\nabla u_t|^2 \in {\mathbb{R}}, 
    \end{equation*}
    which yields, since all objects are smooth enough for the application of Green's formula
    \begin{align}
        \lambda'
        & = 2 \int_\Omega \nabla u \nabla {\widetilde{u}} \ dx + \int_\Omega \mathrm{div } (|\nabla u|^2 \varphi \bar \nu) \ dx \nonumber \\
        & = 2 \underbrace{\int_\Omega \lambda_1 {\widetilde{u}} u \ dx}_{= 0} + 2\lambda' \underbrace{\int_\Omega u^2 \ dx}_{= 1} + \int_{{\Gamma_{0, \Omega}} \cap B_r(x_0)} |\nabla u|^2 \varphi \ d\sigma,
    \end{align}
    where $d\sigma$ is the area element of ${\Gamma_{0, \Omega}} \cap B_r(x_0)$, and the first integral is zero because of the $L^2$-orthogonality between ${\widetilde{u}}$ and $u$ (which comes from differentiating the normalization condition $\|u_t\|_2^2 = 1$), whence
    \begin{equation*}
        \lambda' = - \int_{{\Gamma_{0, \Omega}} \cap B_r(x_0)} |\nabla u|^2 \varphi \ d\sigma
    \end{equation*}

    On the other hand, for the function
    \begin{equation*}
        t \in (-\delta, \delta) \mapsto |\Omega_t| \in {\mathbb{R}},
    \end{equation*}
    the standard application of Hadamard's formula yields
    \begin{equation*}
        \left. \frac{d}{dt} |\Omega_t| \right|_{t = 0} = \int_{\Gamma_{0, \Omega}} V \cdot \nu \ d\sigma = \int_{{\Gamma_{0, \Omega}} \cap B_r(x_0)} \varphi \ d\sigma.
    \end{equation*}

    Now, since $\Omega$ is a minimizer for \eqref{eq:cone_min_equiv}, it is also a minimizer for the scaling-invariant shape functional
    \begin{equation*}
        \Omega \mapsto \lambda_1(\Omega)|\Omega|^{\frac{2}{N}}.
    \end{equation*}
    Then, exploiting the optimality, we have
    \begin{align}
        0 
        & = \left. \frac{d}{dt} (\lambda_1(\Omega_t)|\Omega_t|^{\frac{2}{N}}) \right|_{t = 0} \nonumber \\
        & = |\Omega|^{\frac{2}{N}} \lambda' + \lambda_1(\Omega) \frac{2}{N} |\Omega|^{\frac{2}{N} - 1} \left(\left. \frac{d}{dt} |\Omega_t| \right|_{t = 0}\right), \nonumber
    \end{align}
    which leads to
    \begin{equation*}
        \int_{{\Gamma_{0, \Omega}} \cap B_r(x_0)} \left(|\nabla u|^2 - \frac{2 \lambda_1(\Omega)}{N |\Omega|} \right) \varphi \ d\sigma = 0.
    \end{equation*}
    Since $x_0$ and $\varphi$ are arbitrary, and $\mfrac{\partial u}{\partial \nu} < 0$ on ${\Gamma_{0, \Omega}}$ (by Hopf's Lemma), then \eqref{eq:cone_overdet_reg} follows.

As expected, the value $\sqrt{\frac{2\lambda_1(\Omega)}{N|\Omega|}}$ agrees with the Pohozaev identity (if ${\Gamma_{0, \Omega}}$ is regular).

<!-- remark-13: body_md -->
\label{rem:cone_non-radial}
    Let $D \subset {\mathbb{S}^{N - 1}}$ be a smooth domain such that ${\mathcal{H}^{N - 1}}(D) < {\mathcal{H}^{N - 1}}({\mathbb{S}_+^{N - 1}})$ and $\mu_1 < N - 1$. Note that the minimizer $\Omega$ and the corresponding first eigenfunction $u_\Omega$ cannot be radial. Indeed: the only possible radial domains in a non-convex cone for which ${\mathcal{H}^{N - 1}}({\partial \Omega} \cap {\partial{\Sigma_D}}) > 0$ are the intersection with an annulus or a half-sphere centred on a flat part of ${\partial{\Sigma_D}}$. However, in the former case, the eigenfunction could be extended to the full annulus, contradicting Serrin's Theorem (\cite{Serrin1971}); in the latter case, it would contradict \eqref{eq:cone_ineq}.
<!-- end remark-13 -->
<!-- theorem-4: body_md -->
\label{thm:cone_main}
    Let ${\Sigma_D}$ be the cone spanned by the smooth domain $D \subset {\mathbb{S}^{N - 1}}$, with $N \geq 3$. If ${\mathcal{H}^{N - 1}}(D) < {\mathcal{H}^{N - 1}}({\mathbb{S}_+^{N - 1}})$ and $\mu_1 < N - 1$, then there exists a non-radial open bounded domain $\Omega \subset {\Sigma_D}$, with an ${\mathcal{H}^{N - 1}}$-a.e. analytic boundary, such that 
    \eqref{eq:cone_overdet_pde} admits a non-radial solution $u_\Omega$ (with the boundary conditions satisfied ${\mathcal{H}^{N - 1}}$-a.e.).

Follows by choosing $m = |\Omega_D|$ and combining Theorem \ref{thm:cone_stability}, Proposition \ref{prop:cone_measure_condition}, the results of Section \ref{sec:cone_bound_conn_reg}, and Theorem \ref{thm:cone_overdet_minimizer}, and taking Remark \ref{rem:cone_non-radial} into account.

We note that on dimension $N = 2$, there are two conflicting possibilities for the opening angle $\theta$ (which plays the role of ${\mathcal{H}^{N - 1}}(D)$): either $\theta < \pi$ (in which case the cone is convex and therefore the radial sector is stable) or $\theta > \pi$, in which case the cone is non-convex but the existence of a minimizer is not guaranteed.

<!-- remark-15: body_md -->
Let us comment on some interesting directions for further study.
    \begin{enumerate}[label=\roman*)]
        \item Uniqueness: It would be interesting to understand whether the minimizer $\Omega$ is unique or not, as well as whether it (they?) coincides with the minimizing sets for the torsion and/or isoperimetric problems studied in \cite{IacopettiPacellaWeth2022}. In my opinion, a multiplicity result for minimizing sets would be surprising (and fascinating), but it is reasonable to expect that a critical shape for one of these problems is a critical shape for the others as well.

        \item Nonlinear problems: in \cite{AfonsoIacopettiPacella2024Energypublished}, only the stability is studied, and the existence of minimizers for nonlinear problems is left open. It seems likely that our arguments could be adapted to the study of the best constant for the subcritical Sobolev inequality:
        \begin{equation*}
            C_p(\Omega) = \inf_{\substack{u \in H_0^1(\Omega \cup \Gamma) \\ \|u\|_p = 1}} \int_\Omega |\nabla u|^2 \ dx, \quad 2 < p < 2^*.
        \end{equation*}
        The stability/instability and existence of a minimizer should be very similar, while the regularity and other qualitative properties may be more challenging. It would be interesting to understand if the scheme here presented could be followed in this case, giving rise to a nonlinear, non-radial, positive solution for the overdetermined problem.

        \item Bifurcation: When $\mu_1 = N - 1$, the problem becomes degenerate, and one expects some bifurcation to occur. This has been done in cylinders (\cite{LianPacellaSicbaldi2025}) and is expected to hold also in the cone, but the geometry is more challenging.
    \end{enumerate}
