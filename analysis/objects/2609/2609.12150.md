\label{def:Z}
		The \emph{partition function} and the \emph{free energy} of $\mu$
		are
		\begin{equation}\label{eq:defZ}
			Z_\mu(s)\mathrel{\vcentcolon=}\int_{\mathbb{R}_{>0}}x^{s}\,\mathrm{d}\mu(x),
			\qquad
			K_\mu\mathrel{\vcentcolon=}\log Z_\mu .
		\end{equation}
		The \emph{Lehmer transform} of $\mu$ and its logarithm are
		\begin{equation}\label{eq:defL}
			\mathscr{L}_\mu(s)\mathrel{\vcentcolon=}\frac{Z_\mu(s)}{Z_\mu(s-1)},
			\qquad
			\Lambda_\mu\mathrel{\vcentcolon=}\log\mathscr{L}_\mu .
		\end{equation}
		For every $u$ with $Z_\mu(u)<\infty$ the \emph{escort measure} of
		order $u$ is the probability measure
		\begin{equation}\label{eq:escort}
			\,\mathrm{d}\mu_u\mathrel{\vcentcolon=}\frac{x^{u}\,\mathrm{d}\mu(x)}{Z_\mu(u)} .
		\end{equation}
		For a signal the escort weights are $p_i(u)=x_i^{u}/\sum_jx_j^{u}$.

\label{thm:master}
		Let $\mu$ be compactly supported in $\mathbb{R}_{>0}$ and not a point mass.
		For all real $s$,
		\begin{equation}\label{eq:master}
			\mathscr{L}_\mu(s)=\mathbb{E}_{\mu_{s-1}}[X]=\exp\bigl(K_\mu(s)-K_\mu(s-1)\bigr).
		\end{equation}
		The derivatives of the free energy are escort moments,
		\begin{equation}\label{eq:Kderiv}
			K_\mu'(u)=\mathbb{E}_{\mu_u}[\log X],
		\end{equation}
		\begin{equation}\label{eq:Kderiv2}
			K_\mu''(u)=\operatorname{Var}_{\mu_u}(\log X).
		\end{equation}
		The slope of the transform is an escort covariance,
		\begin{equation}\label{eq:covid}
			\mathscr{L}_\mu'(s)=\operatorname{Cov}_{\mu_{s-1}}\bigl(X,\log X\bigr)
			=\mathscr{L}_\mu(s)\bigl[K_\mu'(s)-K_\mu'(s-1)\bigr]>0 .

			\mathscr{L}_\mu^{-1}(t)=1+\frac{t-\mathscr{L}_\mu(1)}{\operatorname{Cov}_{\mu_0}(X,\log X)}

				F_T(u)=\frac{K_\mu'(u)-\log m}{\ell_\mu} .

				f_S(s)=F_T(s)-F_T(s-1)\le1,
