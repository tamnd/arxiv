<!-- definition-1: body_md -->
A controller $K$ is stable if $\rho(A-BK)<1$, where $\rho(\cdot)$ denotes the spectral radius.
<!-- end definition-1 -->
<!-- condition-1: body_md -->
\label{condition_noise}
For any $\Delta > 0$, and $\delta \in (0,1)$, there exist $b \geq 0$, $V_\Delta > 0$ and $M_{\Delta,\delta} >0$, such that 
\begin{align}
 \lVert\mathbb{E}[G_t| \mathcal{F}_t]-\nabla f(K_t)\rVert & \leq b, \label{condition_bound_bias} \\
 \mathbb{E}[\lVertG_t\rVert^2| \mathcal{F}_t] &\leq V_\Delta,  \label{condition_bound_moment}\\
 \mathbb{P}\left(\lVertG_t\rVert \leq M_{\Delta,\delta}| \mathcal{F}_t\right) & \geq 1- \delta , \label{condition_bound_grad}
\end{align}
for any $K_t$ with $\Delta(K_t) \leq \Delta$.
<!-- end condition-1 -->

\label{lemma_lqr_landscape}
There exists $\mu>0$ such that for any $\Delta>0$, one can choose $L_{\Delta},r_{\Delta}>0$ so that for every policy $K$ with $\Delta(K) < \Delta$, we have that 
$K'$ is stable for any $K'$ satisfying $\lVertK' - K\rVert \leq r_{\Delta}$, and 
\begin{align}
 f(K') & \leq f(K)+\langle \nabla f(K),K' - K \rangle
                  +\tfrac{L_{\Delta}}{2}\lVertK - K'\rVert^2,     \label{ineq_local_smoothness}\\
 \mu\Delta(K) & \leq \lVert\nabla f(K)\rVert^2
                    \leq 2L_{\Delta}\Delta(K).    \label{ineq_gradient_gap_bounds}
\end{align}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
From  \cite[Lemmas 5 and 6, and discussion on page 14]{malik2020derivative},  we obtain \eqref{ineq_local_smoothness},  the stability of $K'$, and the first inequality in \eqref{ineq_gradient_gap_bounds}.
Note that without loss of generality we can assume $L_{\Delta} \geq \tfrac{2\Delta}{r_{\Delta}^2}$.
By taking $K' = K - r_{\Delta} \tfrac{\nabla f(K)}{\lVert\nabla f(K)\rVert}$, we have 
from \eqref{ineq_local_smoothness} that
\begin{align*}
f^* \leq f(K') \leq f(K) - r_{\Delta} \lVert\nabla f(K)\rVert + \tfrac{L_{\Delta}r_{\Delta}^2}{2},
\end{align*}
which implies $ \lVert\nabla f(K)\rVert \leq \tfrac{1}{r_{\Delta}} \left(\Delta + \tfrac{L_{\Delta} r_{\Delta}^2}{2} \right) \leq L_{\Delta}r_{\Delta}$. 
Hence one can apply \eqref{ineq_local_smoothness} again with $K' = K - \nabla f(K) / L_{\Delta}$, which yields the second inequality of \eqref{ineq_gradient_gap_bounds}.

<!-- lemma-2: body_md -->
\label{lemma_aux_seq}
Suppose $0<\eta_t\leq 1/\mu$ and $\eta_tM\leq r_{\Delta}$ for some $M > 0$.
Then we have 
\begin{align}
 Y_{t+1}
 &\leq (1-\mu\eta_t)Y_t
       +\eta_t\langle \nabla f(K_t),\delta_t\mathbbm{1}_{E_t\cap\left\{\lVertG_t\rVert\leq M\right\}} \rangle
       +\tfrac{L_{\Delta}}{2}\eta_t^2M^2.
       \label{ineq_single_step_gap_recursion}
\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
On the event of $E_t\cap\left\{\lVertG_t\rVert\leq M\right\}$, 
since $\Delta(K_t) \leq \Delta_t \leq \Delta$ and $\lVertK_{t+1} - K_t\rVert = \eta_t \lVertG_t\rVert \leq \eta_tM \leq r_{\Delta}$,
one can invoke \eqref{ineq_local_smoothness} in Lemma \ref{lemma_lqr_landscape} and obtain 
\begin{align*}
 f(K_{t+1})
 &\leq f(K_t)+\langle \nabla f(K_t),K_{t+1}-K_t \rangle
              +\tfrac{L_{\Delta}}{2}\lVertK_{t+1}-K_t\rVert^2\\
 &=f(K_t)-\eta_t\langle \nabla f(K_t),G_t \rangle
              +\tfrac{L_{\Delta}}{2}\eta_t^2\lVertG_t\rVert^2\\
 &=f(K_t)-\eta_t\lVert\nabla f(K_t)\rVert^2
              +\eta_t\langle \nabla f(K_t),\delta_t \rangle
              +\tfrac{L_{\Delta}}{2}\eta_t^2\lVertG_t\rVert^2.
\end{align*}
Applying the first inequality in \eqref{ineq_gradient_gap_bounds}, we have 
\begin{align*}
 \Delta(K_{t+1})
 \leq (1-\mu\eta_t)\Delta(K_t)
       +\eta_t\langle \nabla f(K_t),\delta_t \rangle
       +\tfrac{L_{\Delta}}{2}\eta_t^2\lVertG_t\rVert^2.
\end{align*}
Consequently, for any $t \geq 1$, multiplying both sides by $\mathbbm{1}_{E_t\cap\left\{\lVertG_t\rVert\leq M\right\}}$ and using the definition of $Y_t$ yields  
\begin{align}
 Y_{t+1}
 &\leq (1-\mu\eta_t)\Delta(K_t)
       \mathbbm{1}_{E_t\cap\left\{\lVertG_t\rVert\leq M\right\}}  +\eta_t\langle \nabla f(K_t),\delta_t \mathbbm{1}_{E_t\cap\left\{\lVertG_t\rVert\leq M\right\}} \rangle
       +\tfrac{L_{\Delta}}{2}\eta_t^2\lVertG_t\rVert^2 \mathbbm{1}_{E_t\cap\left\{\lVertG_t\rVert\leq M\right\}} \label{ineq_recusion_y_raw}  \\
  & \overset{(a)}{\leq} (1-\mu\eta_t)\Delta(K_t)
       \mathbbm{1}_{E_{t-1} \cap\left\{\lVertG_{t-1}\rVert\leq M\right\}}
        +\eta_t\langle \nabla f(K_t),\delta_t\mathbbm{1}_{E_t\cap\left\{\lVertG_t\rVert\leq M\right\}} \rangle
       +\tfrac{L_{\Delta}}{2}\eta_t^2 M^2  \nonumber \\
 & \leq (1-\mu\eta_t)Y_{t}
        +\eta_t\langle \nabla f(K_t),\delta_t\mathbbm{1}_{E_t\cap\left\{\lVertG_t\rVert\leq M\right\}} \rangle
       +\tfrac{L_{\Delta}}{2}\eta_t^2 M^2,  \nonumber 
\end{align}
where $(a)$ follows from the fact that 
$
 E_t\cap\left\{\lVertG_t\rVert\leq M\right\}
 \subseteq E_{t-1}\cap\left\{\lVertG_{t-1}\rVert\leq M\right\}
$
and $\Delta(K_t) \geq 0$.
Hence \eqref{ineq_single_step_gap_recursion} holds for any $t \geq 1$. 
On the other hand, for $t = 0$, 
it is clear that 
\eqref{ineq_recusion_y_raw} still holds.
Combining this observation with the fact that 
$
(1-\mu\eta_0)\Delta(K_0)
       \mathbbm{1}_{E_0 \cap\left\{\lVertG_0 \rVert\leq M\right\}}  \leq (1-\mu\eta_0)\Delta(K_0) = (1-\mu \eta_0) Y_0,
$
we conclude that \eqref{ineq_single_step_gap_recursion} holds at $t = 0$.

<!-- lemma-3: body_md -->
\label{lemma_recovery_of_actual_gap}
Let $\Omega$ denote the sample space associated with $T$ iterations of SPG.
Suppose the random outcome $\omega$ satisfies\footnote{
For notational simplicity, we omit the explicit dependence of relevant random variables evaluated at $\omega$ within the statement here. 
} 
\begin{align}\label{restriction_omega}
Y_t \leq \Delta_t, ~ 0 \leq t < T; ~~~ \lVertG_t\rVert \leq M ~ \text{if}~ \omega \in E_t, ~ 0 \leq t < T.
\end{align}
  Then for such $\omega$, we have
\begin{align*}
 \Delta(K_t) = Y_t, ~ \lVertG_{t-1}\rVert \leq M, ~ 0\leq t\leq T,
\end{align*}
where we define $G_{-1} = 0$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Clearly the desired claim holds at $t=0$ given the definition of $Y_0 = \Delta(K_0)$. Suppose the claim holds at $t$.
That is, suppose for such $\omega$ we have $ \Delta(K_i) = Y_i$ for any $i \leq t$
and $\lVertG_i\rVert \leq M$ for $i \leq t-1$. 
 Then for the considered $\omega$, we have 
\begin{align*}
\Delta(K_{t+1}) & \overset{(a)}{=} \Delta(K_{t+1}) \mathbbm{1}_{\left\{Y_i \leq \Delta_i, i \leq t, \lVertG_i\rVert \leq M, i \leq t-1\right\}}  \\
& \overset{(b)}{=} \Delta(K_{t+1}) \mathbbm{1}_{\left\{\Delta(K_i) \leq \Delta_i, i \leq t, \lVertG_i\rVert \leq M, i \leq t-1 \right\}} \\
& \overset{(c)}{=}  \Delta(K_{t+1}) \mathbbm{1}_{E_t}  \overset{(d)}{=}  \Delta(K_{t+1}) \mathbbm{1}_{E_t \cap \left\{\lVertG_t\rVert \leq M\right\}} \overset{(e)}{=} Y_{t+1},
                                                                       
                                                                                                                                    
                                                                                                                                           
                                    
\end{align*}
where $(a)$ follows from the induction hypothesis that $\lVertG_i\rVert\leq M$ for $i \leq t-1$,  and the requirement of $Y_i \leq \Delta_i$ for $i \leq t$ in \eqref{restriction_omega}, 
$(b)$ applies the induction hypothesis that $Y_i = \Delta(K_i)$ for $i \leq t$, 
$(c)$ applies the definition of $E_t$, 
and $(d)$ follows from the requirement of $\lVertG_t\rVert \leq M$ provided $\omega \in E_t$, as indicated in \eqref{restriction_omega}.
Finally, $(e)$ follows directly from the definition of $Y_{t+1}$.

\label{lemma_weighted_gap_recursion}
Suppose the stepsize $\left\{\eta_t\right\}$ satisfies 
\begin{align}\label{cond_stepsize_sqrt}
\eta_t \leq \tfrac{1}{\mu}, ~~  \eta_t M \leq r_{\Delta}.
\end{align}
Define $\Lambda_0 = 1$ and 
$
 \Lambda_{t+1}=\tfrac{\Lambda_t}{1-\mu\eta_t}.
$
Then we have
 \begin{align}
 Y_t  \leq \tfrac{1}{\Lambda_t} \left( \Delta(K_0)+
       \textstyle\sum\nolimits_{i<t}\Lambda_{i+1}\eta_i

we obtain

The desired claim then follows from taking the telescopic sum of the above relation.

\label{lemma_azuma_fixed_gap_threshold}
Suppose the stepsize $\left\{\eta_t\right\}$ satisfies \eqref{cond_stepsize_sqrt} with $M = M_{\Delta, \delta}$ therein, 
and Condition \ref{condition_noise} holds.

\Delta_t' =   \tfrac{\Delta(K_0)}{\Lambda_t}
   +\tfrac{G_\Delta\left(b+\sqrt{V_\Delta\delta}\right)}{\mu}
 +\tfrac{4G_\Delta(M_{\Delta,\delta}+G_\Delta)}{\Lambda_t}
      \sqrt{\log(1/\delta)\textstyle\sum\nolimits_{i<t}\Lambda_{i+1}^2\eta_i^2}
   +\tfrac{L_{\Delta}M_{\Delta,\delta}^2}{2\Lambda_t}\textstyle\sum\nolimits_{i<t}\Lambda_{i+1}\eta_i^2 , ~  1 \leq t \leq T. 
   \label{def_azuma_gap_envelope_sqrt}
\end{align}

at least $1-2T\delta$, we have

\end{align*}

Consider choosing $\Delta_t = \Delta$ in \eqref{basic_requirement_delta_seq} and $M = M_{\Delta, \delta}$ in  \eqref{def_good_event}. 
Suppose $\Delta (K_i) \leq \Delta_i = \Delta$. Then in view of \eqref{bd_gd_via_opt_gap}, we have 
 $\lVert\nabla f(K_i)\rVert \leq G_\Delta$, and consequently 
\begin{align}\label{ineq_sqrt_azuma_abs_bd}
 \left\lvert \langle \nabla f(K_i),\delta_i\mathbbm{1}_{E_i\cap\left\{\lVertG_i\rVert\leq M_{\Delta,\delta}\right\}} \rangle \right\rvert
 &\leq G_\Delta(M_{\Delta,\delta}+G_\Delta).
\end{align}
In addition, we have 
\begin{align}
 &\mathbb{E}\left[\langle \nabla f(K_i),\delta_i\mathbbm{1}_{E_i\cap\left\{\lVertG_i\rVert\leq M_{\Delta,\delta}\right\}} \rangle| \mathcal{F}_i\right]
       \nonumber\\
 = & \mathbb{E}\left[\langle \nabla f(K_i),\delta_i\mathbbm{1}_{E_i} \rangle
       | \mathcal{F}_i\right]  -\mathbb{E}\left[\langle \nabla f(K_i),\delta_i\mathbbm{1}_{E_i\cap\left\{\lVertG_i\rVert>M_{\Delta,\delta}\right\}} \rangle| \mathcal{F}_i\right]
       \nonumber\\
 \overset{(a)}{\leq} &  bG_\Delta+G_\Delta
       \mathbb{E}[\lVertG_i\rVert\mathbbm{1}_{E_i\cap\left\{\lVertG_i\rVert>M_{\Delta,\delta}\right\}}| \mathcal{F}_i]
       \nonumber\\
 \overset{(b)}{\leq} &  bG_\Delta+G_\Delta
       \sqrt{\mathbb{E}[\lVertG_i\rVert^2\mathbbm{1}_{E_i}| \mathcal{F}_i] 
             \mathbb{P}(E_i\cap\left\{\lVertG_i\rVert>M_{\Delta,\delta}\right\}| \mathcal{F}_i)}
       \nonumber\\
 \overset{(c)}{\leq} &  \left(b+\sqrt{V_\Delta\delta}\right)G_\Delta,
 \label{ineq_truncated_inner_product_bias}
\end{align}
where $(a)$ follows from \eqref{condition_bound_bias} in Condition \ref{condition_noise} and the Cauchy-Schwarz inequality, 
and $(b)$ follows again from the Cauchy-Schwarz inequality, 
and $(c)$ follows from \eqref{condition_bound_moment} and \eqref{condition_bound_grad} in Condition \ref{condition_noise}.
Consequently, by combining  Lemma \ref{lemma_biased_azuma} with observations \eqref{ineq_sqrt_azuma_abs_bd} and \eqref{ineq_truncated_inner_product_bias} above,
 we obtain 
that with probability at least $1-T\delta$,
\begin{align*}
 &\textstyle\sum\nolimits_{i<t}\Lambda_{i+1}\eta_i
       \langle \nabla f(K_i),\delta_i\mathbbm{1}_{E_i\cap\left\{\lVertG_i\rVert\leq M_{\Delta,\delta}\right\}} \rangle\\
\leq &  \left(b+\sqrt{V_\Delta\delta}\right)G_\Delta\textstyle\sum\nolimits_{i<t}\Lambda_{i+1}\eta_i
       +4G_\Delta(M_{\Delta,\delta}+G_\Delta)\sqrt{\log(1/\delta)\textstyle\sum\nolimits_{i<t}\Lambda_{i+1}^2\eta_i^2}  \\ 
        \leq  &
   \tfrac{\Lambda_t G_\Delta\left(b+\sqrt{V_\Delta\delta}\right)}{\mu}
       +4G_\Delta(M_{\Delta,\delta}+G_\Delta)\sqrt{\log(1/\delta)\textstyle\sum\nolimits_{i<t}\Lambda_{i+1}^2\eta_i^2}  
\end{align*}
for any $1\leq t\leq T$,
where the last inequality follows from $ \textstyle\sum\nolimits_{i<t}\Lambda_{i+1}\eta_i
   =\tfrac{1}{\mu}\textstyle\sum\nolimits_{i<t}(\Lambda_{i+1}-\Lambda_i)
 \leq \tfrac{\Lambda_t}{\mu}.
$
Combining the above observation with the definition of $\Delta_t'$ in \eqref{def_azuma_gap_envelope_sqrt}, and \eqref{ineq_weighted_gap_recursion} in Lemma \ref{lemma_weighted_gap_recursion} implies $Y_t \leq \Delta_t'$ for all $0 \leq t \leq T$ with probability $1-T\delta$.
Now since $\Delta_t' \leq \Delta$,
we have $Y_t \leq \Delta$.
 In view of Lemma \ref{lemma_recovery_of_actual_gap},
and the fact that  $\lVertG_i\rVert \leq M_{\Delta, \delta}$ with probability at least $1- \delta$ conditioned on $E_i$, 
we obtain 
$
\Delta (K_t) = Y_t \leq \Delta_t' 
$
with probability at least $1-2T\delta$ for $0 \leq t \leq T$.

Let $\beta=G_\Delta\left(b+\sqrt{V_\Delta\delta}\right)/\mu$.
Clearly we have $\beta \leq \Delta(K_0)$.  
Consider choosing $\left\{\Delta_t\right\}$ as
\begin{align*}
 \Delta_t=2\beta+\tfrac{C}{N+t},~ 0\leq t\leq T.
\end{align*}
We first proceed to show that the defined $\left\{\Delta_t\right\}$ and $\Delta$ satisfy \eqref{basic_requirement_delta_seq}. 
It can be readily verified that we have  $\Delta(K_0)\leq \Delta_0$, $\Delta(K_0) \leq \Delta$, and that 
\begin{align*}
 \Delta_i
 \leq 2\beta+\tfrac{C}{N+i}
 \leq 2\beta+2\Delta(K_0)+\tfrac{2D}{N}  \leq 6\Delta(K_0) = \Delta. 
\end{align*}
Clearly, the choice of stepsize $\left\{\eta_t\right\}$ satisfies the condition in Lemma \ref{lemma_azuma_varying_gap_thresholds}, and hence it remains to verify that $\Delta_t' \leq \Delta_t$. 
As in the proof of Theorem \ref{thrm_sqrt_rate}, 
one can readily verify that
$
 \Lambda_t=\tfrac{(N+t-2)(N+t-1)}{(N-2)(N-1)}
$, and
\begin{align*}
 \textstyle\sum\nolimits_{i<t}\left(\tfrac{\Lambda_{i+1}\eta_i}{\Lambda_t}\right)^2\Delta_i
                                                
                                         
 \leq \tfrac{128\beta}{3\mu^2(N+t)}
       +\tfrac{32C}{\mu^2(N+t)^2}.
\end{align*}
Consequently, we obtain 
\begin{align*}
 &\tfrac{8(M_{\Delta,\delta}+G_\Delta)}{\Lambda_t}
       \sqrt{L_{\Delta}\log(1/\delta)\textstyle\sum\nolimits_{i<t}\Lambda_{i+1}^2\eta_i^2\Delta_i}\\
\overset{(a)}{\leq} &  \tfrac{64(M_{\Delta,\delta}+G_\Delta)}{\mu}
       \sqrt{\tfrac{L_{\Delta}\beta\log(1/\delta)}{N+t}}
       +\tfrac{64(M_{\Delta,\delta}+G_\Delta)
                     \sqrt{L_{\Delta}C\log(1/\delta)}}{\mu(N+t)}\\
\overset{(b)}{\leq} & \tfrac{\beta}{2}+\tfrac{C}{4(N+t)}
       +\tfrac{6144L_{\Delta}(M_{\Delta,\delta}+G_\Delta)^2\log(1/\delta)}{\mu^2(N+t)}, 
\end{align*}
where $(a)$ uses the inequality $\sqrt{a + b} \leq \sqrt{a} + \sqrt{b}$, and $(b)$ applies $ab \leq (a^2 + b^2 )/2$. 
Note that \eqref{sum_second_moment_grad} still holds.
Hence by combining the above observation with \eqref{sum_second_moment_grad}, 
together with the definition of $\Delta_t'$ in \eqref{def_azuma_gap_envelope_fast_rate}, 
we obtain 
\begin{align}
 \Delta_t' & \leq   \tfrac{N\Delta(K_0)}{N+t}+ \tfrac{3\beta}{2} +\tfrac{8L_{\Delta}M_{\Delta,\delta}^2}{\mu^2(N+t)} + \tfrac{C}{4(N+t)}
       +\tfrac{6144L_{\Delta}(M_{\Delta,\delta}+G_\Delta)^2\log(1/\delta)}{\mu^2(N+t)}  \label{fast_rate_opt_gap_raw} \\
& \leq 
\tfrac{3\beta}{2}+\tfrac{N\Delta(K_0)+C/4+D}{N+t}
 =\tfrac{3\beta}{2}+\tfrac{3C}{4(N+t)}
 \leq 2\beta+\tfrac{C}{N+t}=\Delta_t \nonumber
\end{align}
for $1 \leq t \leq T$. 
Hence applying Lemma \ref{lemma_azuma_varying_gap_thresholds} yields that $\Delta(K_t) \leq \Delta_t'$ with probability at least $1- 2 T\delta$. 
The desired claim then follows from  
\eqref{fast_rate_opt_gap_raw} with the choice of $C$ and $\beta$.

<!-- lemma-9: body_md -->
\label{lemma_lqr_truncation}
For any stable policy $K$ and any $x \in \mathbb{R}^n$, 
define  
$
 F_h (K; x) = \textstyle\sum\nolimits_{i < h} x_i^\top (Q + K^\top R K) x_i, 
$
where $x_i = (A - BK)^i x$.
For any $\Delta > 0$, let  $f_\Delta = f^* + \Delta$, and suppose $\Delta(K) \leq \Delta$. Then 
\begin{align}\label{ineq_two_point_return_tail}
0 \leq F(K; x) - F_h(K; x) \leq 
\left(
1 - \tfrac{\lambda_{\min} (Q) }{ f_\Delta }
\right)^h  f_\Delta \lVertx\rVert^2.
\end{align}
<!-- end lemma-9 -->
<!-- proof-10: body_md -->
Since $K$ is stable, there exists  $P_K \succ 0$ being the unique solution satisfying 
\begin{align}\label{lqr_dp_eval}
 P_K=Q+K^\top RK+(A-BK)^\top P_K(A-BK).
\end{align}
In addition, we have 
$F(K; x) = x^\top P_K x$, and $f(K) = \mathbb{E}_{x_0} F(K; x_0) = \mathrm{tr}(P_K)$,
where the last equality follows from \eqref{eq_dynamics}. 
From this we obtain 
\begin{align*}
F(K; x) \leq \lVertP_K\rVert \lVertx x^\top\rVert \leq \mathrm{tr} (P_K) \lVertx\rVert^2 \leq f(K) \lVertx\rVert^2.
\end{align*}
Combining the above observation with \eqref{lqr_dp_eval}, we have 
\begin{align*}
F(K; (A-BK) x) = F(K; x) - x^\top (Q + K^\top R K) x 
\leq F(K; x) - \lambda_{\min} (Q) \lVertx\rVert^2 
\leq \left(
1 - \tfrac{\lambda_{\min} (Q) }{ f(K) }
\right) F(K; x).
\end{align*}
Recursive application of the above inequality then yields
\begin{align*}
F(K; x) - F_h(K;x) = F(K; (A-BK)^h x) \leq \left(
1 - \tfrac{\lambda_{\min} (Q) }{ f(K) }
\right)^h F(K; x)
\leq 
\left(
1 - \tfrac{\lambda_{\min} (Q) }{ f(K) }
\right)^h  f(K) \lVertx\rVert^2 
\end{align*}
from which the desired claim follows.
