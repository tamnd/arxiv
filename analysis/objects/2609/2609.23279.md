<!-- definition-1: body_md -->
(Moreau envelopes and proximal mappings). For a proper, lsc function $f: \mathbb{R}^n \to \overline \mathbb{R}$ and $\lambda > 0$, the Moreau envelope function $e_\lambda f$ and proximal mapping $P_\lambda f$ are defined by
        \begin{align*}
        e_\lambda f(x) &:= \inf_w \left\{ f(w) + \frac{1}{2\lambda} \|w - x\|^2 \right \} \le f(x), \\
        P_\lambda f(x) &:= \arg \min_w \left\{ f(w) + \frac{1}{2\lambda} \|w-x\|^2\right\}.
        \end{align*}

<!-- fact-1: body_md -->
\label{_rockafellarProp}
    Suppose that $f : \mathbb{R}^n \to \overline \mathbb{R}$ is prox-regular at $\overline x$ for $\overline v = 0$, and that $f$ is prox-bounded. Then for all $\lambda > 0$ sufficiently small, there is a neighborhood of $\overline x$ on which
    \begin{enumerate}[(i)]
        \item\label{i:mor1}
        $P_\lambda f$ is monotone, single-valued and Lipschitz continuous; $P_\lambda f(\overline x) = \overline x$,
        \item\label{i:mor2}
         $e_\lambda f$ is differentiable with $\nabla (e_\lambda f)(\overline x) = 0$, in fact of class ${\cal C}^{1+}$ with
        \[
        \nabla e_\lambda f = \lambda^{-1}[I - P_\lambda f] = [\lambda I + T^{-1}]^{-1}
        \]
        for an $f$-attentive localization $T$ of $\partial f$ at $(\overline x, 0)$.
    \end{enumerate}

A function $f: \mathbb{R}^n \to \overline \mathbb{R}$ is lower semicontinuous (lsc) at $\overline x$ if
        $
        \liminf_{x \to \overline x} f(x) \ge f(\overline x),$ and lsc on $\mathbb{R}^n$ if this holds for every $\overline x\in\mathbb{R}^n$.

<!-- definition-3: body_md -->
(regular and general subgradients).
         For a function $f: \mathbb{R}^n \to \overline \mathbb{R}$ at a point $\overline x$ with $f(\overline x)$ finite, a vector $v \in \mathbb{R}^n$ is a regular subgradient of $f$ at $\overline x$, written $v\in \hat\partial f(\overline x)$, if
        \[
        \liminf_{\substack{x \to \overline x \\ x \ne \overline x}} \frac{f(x) - f(\overline x) - \langle v, x - \overline x\rangle}{\|x - \overline x \|} \ge 0;
        \]
        $v$ is a general subgradient (Mordukhovich limiting subgradient) of $f$ at $\overline x$, written $v \in \partial f(\overline x)$, if there exists a sequence $x^\nu \xrightarrow[f]{} x$ and $v^\nu \to v$ with $v^\nu \in \hat \partial f(x^\nu)$. An $f$-attentive localization of $\partial f$ around $(\overline x, \overline v)$ is some set-valued mapping $T$ which has a graph containing all points $(x,v) \in \operatorname{gph} \partial f$ with $\| v- \overline v\| < \varepsilon$, $\|x - \overline x\| < \varepsilon$ and $f(x) < f(\overline x) + \varepsilon$ for some $\varepsilon > 0$.

<!-- definition-4: body_md -->
(prox-regularity of functions).
        A function $f: \mathbb{R}^n \to \overline \mathbb{R}$ is prox-regular at $\overline x$ for $\overline v$ if $f$ is finite and locally lsc at $\overline x$ with $\overline v \in \partial f(\overline x)$, and there exists $\varepsilon > 0$ and $\rho \ge 0$ such that
        \begin{align*}
            f(x') \ge f(x) + \langle v, x' - x\rangle - \frac{\rho}{2} \|x' - x \|^2 \text{ for all } x' \in \mathbb{B}_\varepsilon(\overline x) \\
            \text{when } v \in \partial f(x), \|v - \overline v\| < \varepsilon, \|x - \overline x \|< \varepsilon,
            f(x) < f(\overline x) + \varepsilon.
        \end{align*}
        When this holds for all $\overline v \in \partial f(\overline x)$, $f$ is said to be prox-regular at $\overline x$.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
(prox-bounded).
        A function $f:\mathbb{R}^n \to \overline \mathbb{R}$ is prox-bounded if there exists $\lambda > 0$ such that $e_\lambda f(x) > -\infty$ for some $x \in \mathbb{R}^n$. The supremum of all such $\lambda$ is called the threshold $\lambda_f$ of prox-boundedness for $f$.

<!-- fact-3: body_md -->
\label{fact:proxMapNonemptyCompact} \cite[Theorem 1.25]{Rockafellar}.
        Let $f: \mathbb{R}^n \to \overline \mathbb{R}$ be proper, lsc, and prox-bounded with threshold $\lambda_f > 0$. Then for any $\lambda \in (0, \lambda_f)$ and $x \in \mathbb{R}^n$ the set $P_\lambda f(x)$ is nonempty and compact, and $e_\lambda f(x)$ is finite and depends continuously on $(\lambda, x)$ with $e_{\lambda}f(x)\uparrow f(x)$ for all $x$ as $\lambda\downarrow 0$. In particular, $P_\lambda f$ has a full domain, i.e., $\operatorname{dom} P_\lambda f = \mathbb{R}^n$.

<!-- definition-8: body_md -->
(semicontinuity of set-valued mappings).
   For a set-valued mapping $S:\mathbb{R}^n\rightrightarrows\mathbb{R}^m$,  define its outer and inner limits at $\overline x$ as follows:
    \begin{align*}
        \limsup_{x \to \overline x} S(x)& := \left\{ u \mid \exists x^\nu \to \overline x, \exists u^\nu \to u \text{ with } u^\nu \in S(x^\nu) \right \}, \\
        \liminf_{x \to \overline x} S(x)& := \left\{ u \mid \forall x^\nu \to \overline x, \exists N \in {\mathcal N}_\infty ,\exists u^\nu \xrightarrow[]{} u \text{ with } \nu \in N, u^\nu \in S(x^\nu) \right \}.
    \end{align*}
    We say that $S$ is outer semicontinuous (osc) at $\overline x$ if
    $
    \limsup_{x \to \overline x} S(x) \subseteq S(\overline x),
    $
    inner semicontinuous (isc) at $\overline x$ if
    $
    \liminf_{x \to \overline x} S(x) \supseteq S(\overline x),
    $
    and  is continuous at $\overline x$ if both conditions hold.

<!-- definition-9: body_md -->
(Lipschitz set-valued mappings). \label{d:lipset}
        A mapping $S : \mathbb{R}^n \rightrightarrows \mathbb{R}^n$ is Lipschitz continuous on $X \subseteq \mathbb{R}^n$ if it is nonempty-closed-valued on $X$ and there exists $\kappa > 0$ such that
        \[
        S(x') \subseteq S(x) + \kappa \|x ' - x\| \mathbb{B}
        \]
        for all $x,x' \in X$.

<!-- lemma-1: body_md -->
\label{l:singlevalue}
     Suppose that $S:\mathbb{R}^n\rightrightarrows\mathbb{R}^n$ is monotone and $\overline x\in\operatorname{int}\operatorname{dom} S$.
    If there exists $\varepsilon>0$ such that $S$ is Lipschitz on $\mathbb{B}_{\varepsilon}(\overline x)\subset\operatorname{int}\operatorname{dom} S$
    in the sense of Definition~\ref{d:lipset},
    then $S$ is single-valued on $\mathbb{B}_{\varepsilon}(\overline x)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $S$ is monotone, by \cite[Theorem 21.27]{bauschke} there exists a dense set $C$ of $\mathbb{B}_{\varepsilon}(\overline x)$ such that
  $S(x)$ is a singleton for every $x\in C$. Let $x\in \mathbb{B}_{\varepsilon}(\overline x)$.
  Then there exists a sequence $\{x^{\nu}\}_{\nu\in\mathbb{N}}$ from $C$ such that $S(x^{\nu})$ is a singleton
  and $\{x^{\nu}\}_{\nu\in\mathbb{N}}$ converges to $x$. By the Lipschitz continuity of $S$, there exists $\kappa>0$
  such that
  $S(x)\subset S(x^{\nu})+\kappa\|x-x^{\nu}\|\mathbb{B}$, which implies
  $\mbox{diam}(S(x))\leq 2\kappa\|x-x^{\nu}\|$. Taking limit as $\nu\rightarrow\infty$ gives $\mbox{diam}(S(x))=0$, so
  $S(x)$ is a singleton. Since $x \in \mathbb{B}_{\varepsilon}(\overline x)$ was arbitrary, we conclude that
  $S$ is single-valued on $\mathbb{B}_{\varepsilon}(\overline x)$.
<!-- end proof-1 -->

<!-- theorem-1: body_md -->
\label{_NecessaryConContGen}
        Let $S : \mathbb{R}^n \rightrightarrows \mathbb{R}^n$ be osc, locally bounded, and have a convex domain. Suppose there exist nonempty subsets of $\operatorname{gph} S$, $A$ and $B$, which are disjoint from each other's closure, that is $A \cap \overline B = \overline A \cap B = \varnothing$, and a point, $x_0 \in \operatorname{dom} S$, such that $\operatorname{gph} S = A \cup B$ and $\{(x_0,v) \mid v \in S(x_0)\} \subseteq A$. Then $S$ is not isc. Moreover, there exists a point where $S$ is multivalued.
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
Observe that, by Fact~\ref{fact:oscIffGphClosed}, $\operatorname{gph} S$ is closed because $S$ is osc. If $x^\nu \xrightarrow[A]{} x$ then $x \in \operatorname{gph} S$ so $x \in A$ or $x \in B$ and since $\overline A \cap B = \varnothing$ we must have $x \in A$ so $A$ is closed. Similarly, $B$ is closed.

        Take $x_1$ such that $\{(x_1,v) \mid v \in S(x_1)\} \cap B \ne \varnothing$. $x_1$ exists because $B$ is a nonempty subset of $\operatorname{gph} S$. Define
        \[
        x_\gamma := (1-\gamma)x_0 + \gamma x_1 \quad \gamma \in [0,1].
        \]
        Observe that $x_\gamma \in \operatorname{dom} S$ for all $\lambda \in [0,1]$ because $\operatorname{dom} S$ is convex. Take
        \[
        \overline \lambda = \sup\Big\{\lambda\in [0,1]  \mid \forall \gamma \in [0, \lambda], \{ (x_\gamma, v) \mid v \in S(x_\gamma) \} \subseteq A \Big\}.
        \]
        Such a $\overline \lambda$ is finite because $\lambda = 0$ is in the set. Take the sequence $\lambda^\nu_- \to \overline \lambda$ with $\lambda^\nu_- \le \overline \lambda$. Then $x_{\lambda^\nu_-}$ converges to $x_{\overline \lambda}$ since $x_\lambda$ is continuous relative to $\lambda$. Since $x_{\lambda^\nu_-}$ converges it is bounded. So, by local boundedness, any sequence $y^\nu \in S(x_{\lambda^\nu_-})$ is bounded. Take a subsequence if necessary so that $y^\nu$ converges to some $\overline y$. Now, $(x_{\lambda^\nu_-}, y^\nu) \xrightarrow[{A}]{} (x_{\overline \lambda}, \overline y)$ and because $A$ is closed we have $(x_{\overline \lambda}, \overline y) \in A$. Now, by the way $\overline \lambda$ is defined, there exists a sequence $\lambda^\nu_+ \to \overline \lambda$ with $\lambda^\nu_+ \ge \overline \lambda$ such that
        \[
        \{ (x_{\lambda^\nu_+}, v) \mid v \in S(x_{\lambda_+^\nu}) \} \not \subseteq A.
        \]
        But, because $A$ and $B$ cover $\operatorname{gph} S$, we must have
        \[
        \{ (x_{\lambda^\nu_+}, v) \mid v \in S(x_{\lambda_+^\nu}) \} \cap B \ne \varnothing.
        \]
        Take $u^\nu \in S(x_{\lambda^\nu_+})$ such that $(x_{\lambda^\nu_+}, u^\nu) \in B$. Now, by similar argument as for $x_{\lambda^\nu_-}$, we have $(x_{\lambda^\nu_+}, u^\nu) \xrightarrow[B]{} (x_{\overline \lambda}, \overline u)$ and because $B$ is closed, $(x_{\overline \lambda}, \overline u) \in B$. Recall $S$ is isc at $\overline x$ if and only if
        \[
            \liminf_{x \to \overline x} S(x) \supseteq S(\overline x)
        \]
        with
        \[
        \liminf_{x \to \overline x} S(x) = \left \{ u \mid \forall x^\nu \to \overline x, \exists N \in {\cal N_\infty}, u^\nu \xrightarrow[N]{} u, \text{ with } u^\nu \in S(x^\nu) \right\}.
        \]
        Let $\overline x = x_{\overline \lambda}$. Now, for any choice of $y^\nu \to \overline y$ with $y^\nu \in S(x_{\lambda^\nu_-})$, we have $(x_{\overline \lambda}, \overline y) \in A$ and there exists $u^\nu \to \overline u$ with $u^\nu \in S(x_{\lambda^\nu_+})$ such that $(x_{\overline \lambda}, \overline u) \in B$. $A$ and $B$ are disjoint so we must have $\overline y \ne \overline u$. So $\overline u \not \in \liminf_{x \to x_{\overline \lambda}} S(x_{\overline \lambda})$ but $(x_{\overline \lambda}, \overline u) \in B \subseteq\operatorname{gph} S$ giving $\overline u \in S(x_{\overline \lambda})$. Therefore, $S$ is not isc at $x_{\overline \lambda}$. Moreover, for any given choice of $y^\nu \in S(x_{\lambda^\nu_-})$ with $y^\nu \to \overline y$, we have $\{\overline y, \overline u\} \subseteq S(x_{\overline \lambda})$ with $\overline y \ne \overline u$ so $S$ is multivalued at $x_{\overline \lambda}$.

\label{_NecessaryConContRR}
        Let $S : \mathbb{R}^n \rightrightarrows \mathbb{R}^n$ be osc, locally bounded, and have convex domain. Suppose there exist nonempty subsets of $\operatorname{gph} S$, $A$ and $B$, which are disjoint from each other's closure, and points $x_1, x_0 \in \operatorname{dom} S$, such that $\operatorname{gph} S = A \cup B$, $\{(x_0, v) \mid v \in S(x_0)\} \subseteq A$, and $\{(x_1, v) \mid v \in S(x_1)\} \cap B \ne \varnothing$. Then, there exists a point $\overline x$ on the line segment connecting $x_0$ and $x_1$ such that $S$ is multivalued and not isc at $\overline x$.

            & \iff w = \ell_n.

<!-- lemma-6: body_md -->
Let $\lambda>0$. Then
    \label{_gapExistance}
        there exists $N \in \mathbb{N}$ such that for all $n > N$ we have $m_n \not \in P_\lambda f(\mathbb{R})$.
<!-- end lemma-6 -->
<!-- proof-8: body_md -->
Recall
        \[
        P_\lambda f(x) = \arg \min_w \left \{ f(w) + \frac{(x-w)^2}{2\lambda} \right \}
        \]
        and for all $y \in P_\lambda f(x)$ we have $e_\lambda f(x) = f(y) + \frac{(x-y)^2}{2\lambda}.$
         Take $N$ large enough so that for any $n > N$ and any $x \in (0, \ell_n)$ we have $n > 5$ and $\frac{1}{2\lambda}x^2 < \frac 4 {5}x$. We claim that $m_{k}\not\in P_{\lambda}f(\mathbb{R})$ for $k > N$.
         We have three cases to consider.

        First, suppose $x \le \ell_k$. Then, since $P_\lambda f$ is monotone by Fact~\ref{fact:proxMonotone} and since $P_\lambda f(\ell_n) = \{\ell_n\}$, $y \in P_\lambda f(x)$ has $y \le \ell_k < m_k$.

        Suppose $x \ge \ell_{k-1}$. Then, by monotonicity of $P_\lambda f$ and since $P_\lambda f(\ell_n) = \{\ell_n\}$, $y \in P_\lambda f(x)$ has $y \ge \ell_{k-1} > m_k$.

        Now, suppose $x \in (\ell_k, \ell_{k-1})$. By Lemma \ref{_sequenceInTermsOfx},
        \[
        k - 1 = \left \lfloor \frac{2-3\pi x}{4 \pi x} \right \rfloor.
        \]
        By our assumption on $N$ we have $k-1 \ge 5$. So we have,
        \begin{align*}
            m_k = \frac{2}{\pi(4k+1)} &= \frac{2}{\pi(4(k-1)+5)}\ge \frac{2}{5\pi (k-1)} = \frac{2}{5\pi \left \lfloor \frac{2-3\pi x}{4 \pi x} \right \rfloor} \\
            &\geq \frac{2}{5\pi\left ( \frac{2-3\pi x}{4 \pi x} \right )}
            =\frac{8 x }{5(2 - 3 \pi x)} > \frac{4}{5}x
        \end{align*}
        where the last inequality follows because $x < \ell_N < \ell_0 = \frac{2}{3 \pi}$. Now,
        \begin{align*}
            e_\lambda f(x) \le f(0) + \frac{1}{2\lambda}x^2 = \frac{1}{2\lambda}x^2 < \frac{4}{5}x < m_k < 2 m_k + \frac{(m_k - x)^2}{2 \lambda}
        \end{align*}
        so $m_k \not \in P_\lambda f(x)$. All three cases together give $m_k \not \in P_\lambda f(\mathbb{R})$.

<!-- theorem-3: body_md -->
\label{NotProxRegular_Theorem}
    Let $f : \mathbb{R} \to \mathbb{R}$ satisfy
\begin{enumerate} [(i)]
    \item\label{i:i1} $f(0) = 0$,
    \item\label{i:i2} there exists $C > 0$ such that $f(x) \geq C|x|$ for every $x\in\mathbb{R}$, and
    \item\label{i:i3} $f'(x)$ exists on the open set $(0,a)$ for some positive $a$ and oscillates between $-\infty$ and $+\infty$, say
    \[
    \limsup_{x \to 0^+} f'(x) = +\infty \text{ and } \liminf_{x \to 0^+} f'(x) = -\infty.
    \]
\end{enumerate}
Then the following hold:
\begin{enumerate}[(a)]
    \item\label{i:a1} For $\lambda > 0$ and $|x| < C\lambda$ we have $P_\lambda f(x) = \{0\}$ and $e_\lambda f(x) = \frac{1}{2\lambda} x^2$.
    \item\label{i:a2} $[-C,C] \subseteq \hat \partial f(0)$, and $\partial f(0) = \mathbb{R}$.
    \item\label{i:a3} $f$ is not prox-regular for $0 \in \partial f(0)$ at $x = 0$.
\end{enumerate}
<!-- end theorem-3 -->
<!-- proof-10: body_md -->
To prove
    \ref{i:a1} it suffices to show
    \[
    f(0) + \frac{1}{2\lambda}x^2 < f(w) + \frac{1}{2 \lambda} (w-x)^2
    \]
    for all $|x| < C\lambda$ and all $w \ne 0$. Suppose $xw \ge 0$. Since $f(w) \geq C|w|$ we have $\frac{f(w)}{|w|}\geq C$. Thus,
    \begin{align*}
        |x| &< C\lambda \leq \frac{\lambda }{|w|}f(w) < \frac{\lambda }{|w|}f(w) + \frac 1 2 |w| \\
        \implies & 0 < \frac{1}{|w|}f(w) + \frac{1}{2 \lambda} (|w| - 2|x|) \\
        \implies & 0 < f(w) + \frac{1}{2 \lambda} (w^2 - 2|wx|) \\
        \implies & \frac{1}{2\lambda}x^2 < f(w) + \frac{1}{2 \lambda} (w^2 - 2wx  + x^2) \\
        \implies & f(0) + \frac{1}{2\lambda}x^2 < f(w) + \frac{1}{2 \lambda} (w-x)^2.
    \end{align*}
    Now, suppose $xw \le 0$,
    \begin{align*}
    f(w) + \frac{1}{2\lambda}(w-x)^2 & = f(w) + \frac{1}{2\lambda}(|x|+|w|)^2 > f(0) + \frac{1}{2\lambda}x^2.
\end{align*}
    In any case we have $P_\lambda f(x) = \{0\}$ for all $|x| < C\lambda$ and
\[
e_\lambda f(x) = f(0) + \frac{1}{2\lambda}x^2 = \frac{1}{2\lambda}x^2.
\]
We now prove result \ref{i:a2}. Recall that $v \in \hat \partial f(\overline x)$ if and only if
\begin{equation}\label{e:barx}
\liminf_{\substack{x \to \overline x \\ x \ne \overline x}} \frac{f(x) - f(\overline x) - \langle v,x - \overline x \rangle}{|x - \overline x|} \ge 0.
\end{equation}
Let $v$ be such that $|v| \le C$. Then, with $\overline x=0$ in \eqref{e:barx},
\begin{align*}
    \liminf_{\substack{x \to 0 \\ x \ne 0}} \frac{f(x) - f(0) - \langle v,x  \rangle}{|x|} &= \liminf_{\substack{x \to 0 \\ x \ne 0}} \frac{f(x) - vx}{|x|} \\
    & \ge \liminf_{\substack{x \to 0 \\ x \ne 0}} \frac{C|x| - vx}{|x|} \\
    &= \liminf_{\substack{x \to 0 \\ x \ne 0}} \left(C- \frac{x}{|x|}v\right) \ge 0.
\end{align*}
Thus, $[-C,C] \subseteq \hat \partial f(0)$. Now, recall $v \in \partial f(\overline x)$ if and only if there are sequences $x^\nu \xrightarrow[f]{} \overline x$ and $v^\nu \in \hat \partial f(x^\nu)$ with $v^\nu \to v$. Take some $v \in \mathbb{R}$ and use Darboux's property of $f'$ (see, e.g., \cite[Theorem 7.31]{thomson}) on $(0,a)$ to find $x^\nu \to 0^+$ such that $f'(x^\nu) =y$. Then, by Fact~\ref{fact:diffSubdifEq}, $\hat \partial f(x^\nu) = \{f'(x^\nu)\} = \{y\}$ so $y \in \partial f(0)$. Thus, $\partial f(0) = \mathbb{R}$.

We now prove result \ref{i:a3}. Recall that $f$ is prox-regular at $\overline x $ for $\overline v$ if $f$ is finite and locally lsc at $\overline x$ with $\overline v \in \partial f(\overline x)$ and there exists $\varepsilon > 0$ and $\rho \ge 0$ such that
\begin{align*}
    f(x') \ge f(x) + \langle v,x' - x \rangle - \frac{\rho}{2}|x'-x|^2 \text{ for all }x' \in \mathbb{B}_\varepsilon (\overline x) \\
    \text{ when } v \in \partial f(x), |v - \overline v| < \varepsilon, |x-\overline x| < \varepsilon, f(x) < f(\overline x) + \varepsilon.
\end{align*}
By way of contradiction assume that $f$ is prox-regular at $\overline x = 0$ for $\overline v = 0$. Let $\varepsilon > 0$, $\rho \ge 0$, and $x' = 0 \in \mathbb{B}_\varepsilon(0)$. Apply Darboux's property to find an $x^\nu \to 0^+$ such that $f'(x^\nu) = 0$. Then $\{f'(x^\nu)\} = \{0\} \subseteq \partial f(x^\nu)$ so we can choose $v = 0$. Choose $\nu$ sufficiently large so that $x^\nu < \min\{\frac{2C}{\rho}, \varepsilon\}$ and $f(x^\nu) < \varepsilon$. We also have $|v - \overline v| = 0 < \varepsilon$, $|x - \overline x| = |x^\nu| < \varepsilon,$ and $f(x) = f(x^\nu) < \varepsilon$. Thus, by our assumption that $f$ is prox-regular,
\begin{align*}
    & f(x') \ge f(x) + \langle v,x' - x \rangle - \frac{\rho}{2}|x'-x|^2 \text{ for all }x' \in \mathbb{B}_\varepsilon (\overline x) \\
    \implies & f(0) \ge f(x^\nu)  - \frac{\rho}{2}|x^\nu|^2 \\
    \implies & \frac{\rho}{2}|x^\nu|^2 \ge f(x^\nu) \ge C|x^\nu| \\
    \implies & \frac{\rho}{2}|x^\nu| > C
\end{align*}
for this to be true $\rho \ne 0$ and
\[
|x^\nu| = x^\nu > \frac{2C}{\rho}
\]
which contradicts our assumption on $x^\nu$. Thus, $f$ is not prox-regular at $\overline x = 0$ for $\overline v = 0$.
