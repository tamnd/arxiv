<!-- definition-1: body_md -->
\label{defn:beta_mixing}
Let $\widetilde X^{n+1} = (\widetilde X_1,\ldots,\widetilde X_{n+1}) \in \ensuremath{\mathcal{X}}^{n+1}$
denote an i.i.d.\ copy of $X^{n+1}$. For a given time lag $\tau \in [n]$, the
$\beta$-mixing coefficient of the sequence $X^{n+1}$ is defined as
\citep{doukhan1995mixing}
\begin{align*}
    \beta(\tau) \coloneqq \max_{i \in [n+1-\tau]} \ensuremath{\mathsf{d_{TV}}}\big( (X_1,\ldots,X_{i},X_{i+\tau},\ldots,X_{n+1}), (X_1,\ldots,X_{i},\widetilde X_{i+\tau},\ldots,\widetilde X_{n+1}) \big).
\end{align*}

<!-- definition-2: body_md -->
\label{defn:martonizable}
Let $X^n \coloneqq (X_1,\ldots,X_n)$ be a vector of random variables taking values in
$\ensuremath{\mathcal{X}}^n$, and let $\gamma \geq 1$ and $s \in \mathbb{N}$. We say that $X^n$ is
\emph{$(\gamma,s)$-Martonizable} if the \emph{uniform block partition}
\[
    \widehat{X}^{m}
    \coloneqq
    \Big(
    \underbrace{(X_1,\ldots,X_{s})}_{\widehat X_1},\;
    \underbrace{(X_{s+1},\ldots,X_{2s})}_{\widehat X_2},\;
    \ldots,\;
    \underbrace{(X_{(m-1)s+1},\ldots,X_n)}_{\widehat X_m}
    \Big),
    \qquad m = \lceil n/s\rceil,
\]
into $m$ contiguous blocks of equal length $s$ (with a possibly shorter final
block) admits a Marton coupling in the sense of
\citet[Definition~2.1]{paulin2015concentration} whose mixing matrix
$\Gamma \in \mathbb{R}^{m\times m}$ satisfies $\ensuremath{|\!|\!| \Gamma | \! | \!|_{{\mathrm{op}}}} \leq \gamma$.
<!-- end definition-2 -->
<!-- remark-1: body_md -->
Definition~\ref{defn:martonizable} is a special case of the partition-based
Marton-coupling framework of \citet[Definition~2.3]{paulin2015concentration},
which permits an arbitrary partition of $\{X_1,\ldots,X_n\}$ into disjoint,
non-empty subsets. We restrict attention to \emph{contiguous blocks of a common
size $s$}, so that partition size (in the notation of~\citet{paulin2015concentration}) is exactly $s$ and the number of blocks is exactly $m = \lceil n/s\rceil$.

\label{thm:paulin_mcdiarmid}
    Let $X^n = (X_1,\ldots,X_n)$ be a sequence of random variables, and let $\widehat{X}^m = (\widehat{X}_1,\ldots,\widehat{X}_m)$ be a partition of this sequence. Suppose we have a Marton coupling for $\widehat{X}^m$ with mixing matrix given by $\Gamma$. Let $h:\ensuremath{\mathcal{X}}^n \to \mathbb{R}$ be a function such that
    \begin{align*}
        h(x) - h(y) \leq \sum_{i=1}^n c_i \ensuremath{{\mathbb{I}\left\{ x_i \neq y_i \right\}}} \text{ for all }x,y\in \ensuremath{\mathcal{X}}^n,
    \end{align*}
    where we term $\{c_i\}_{i=1}^n$ ``bounded-differences constants".
    Now, define the cumulative bounded differences constant based on the partition $\widehat{X}^m$ as follows:
    \begin{align*}
        C_i(c) = \sum_{X_j \in \widehat{X}_i} c_j \text{ for } i \in [m].
    \end{align*}
    Define $C(c) \coloneqq [C_1(c), \ldots, C_m(c)]^T \in \mathbb{R}^m$. Then we have, for any $t\geq 0$,
    \begin{align*}
        \ensuremath{{\mathbb{P}}}(|h(X^n) - \ensuremath{{\mathbb{E}}}[h(X^n)]| \geq t) \leq 2 \exp\big( \frac{-2t^2}{\| \Gamma \cdot C(c)\|_{2}^2} \big).
    \end{align*}

Let us define the following quantity for some fixed sequence $X^n$, pair of symbols $a,b \in \ensuremath{\mathcal{X}}$ and index $k \in [n]$:
    \begin{align*}
        V^{(k)}(a,b) &:= \sum_{j =1}^{n} \ensuremath{{\mathbb{I}\left\{ \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(a))\neq \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(b)) \right\}}}, \text{ where we recall that } 
        \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^n) &= \ensuremath{{\mathbb{I}\left\{ X_{j} \notin \ensuremath{\bm{X}}_{\ensuremath{\mathcal{I}}_{j}} \right\}}}.
    \end{align*}
    We then have
    \begin{align*}
        \left|\ensuremath{\widehat{\mathcal{S}}}(X^{(k)}(a))-\ensuremath{\widehat{\mathcal{S}}}(X^{(k)}(b)) \right| &\leq \frac{1}{n} \sum_{j=1}^n \left|\ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(a)) - \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(b))\right| \\
        &= \frac{1}{n} \sum_{j=1}^n \ensuremath{{\mathbb{I}\left\{ \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(a))\neq \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(b)) \right\}}} =: \frac{V^{(k)}(a,b)}{n}.
    \end{align*}
     
   Thus, it suffices to prove that $V^{(k)}(a,b) \leq 3$ for any pair of symbols $a,b \in \ensuremath{\mathcal{X}}$ and any index $k \in [n]$.
   We will do this by enumerating the maximum possible number of indices $j \in [n]$ for which $\ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(a)) \neq \ensuremath{\widehat{\mathcal{S}}}^{(j)}(X^{(k)}(b))$. Observe that, for any sequence, $\ensuremath{\widehat{\mathcal{S}}}_{(j)} =1$ necessitates that the symbol $X_j$ occurs for the first time at position $j$. For the second occurrence and beyond (i.e. considering all indices $j' > j$ such that $X_{j'} = X_j$), we always have $\ensuremath{\widehat{\mathcal{S}}}_{(j')} = \ensuremath{{\mathbb{I}\left\{ X_{j'} \notin \ensuremath{\bm{X}}_{\ensuremath{\mathcal{I}}_{j'}} \right\}}} = 0$. This is because $\ensuremath{\mathcal{D}}_{j'} = \{j',\ldots,(j'+\tau-1)\wedge n\}$ implies that $j \in \ensuremath{\mathcal{I}}_{j'}$, so that $X_{j'} = X_j \in \ensuremath{\bm{X}}_{\ensuremath{\mathcal{I}}_{j'}}$.
    We claim that if we replace one occurrence of $a$ by $b$ (in going from the original sequence $X^{(k)}(a)$ to the modified sequence $X^{(k)}(b)$) then at most $3$ indicators among $\{\ensuremath{\widehat{\mathcal{S}}}_{(j)}\}_{j=1}^n$ can change: two corresponding to the first and second occurrence of $a$ and one corresponding to the first occurrence of $b$. 
    To prove this claim, we analyze the following four cases, which together encompass all possibilities for the index $k \in [n]$, the pair of symbols $(a,b)$ and the original sequence $X^n$.
    \begin{enumerate}
        \item

        Essentially, in this case we are replacing the first occurrence of $a$ in $X^{(k)}(a)$ by the first occurrence of $b$ in $X^{(k)}(b)$. In this case, there are only $3$ indicators that can change:

        \item The indicator for the first occurrence of $a$ in the original sequence can change, i.e. we can have $\ensuremath{\widehat{\mathcal{S}}}_{(k)}(X^{(k)}(a)) \neq \ensuremath{\widehat{\mathcal{S}}}_{(k)}(X^{(k)}(b))$.
        \item The indicator for the second occurrence of $a$ in the original sequence can change, since in the modified sequence $X^{(k)}(b)$ it is the first occurrence of $a$.

            \item The indicator for the first occurrence of $b$ can change. In particular, if $j_b$ was the only occurrence of $b$ in the original sequence, we could have $\ensuremath{\widehat{\mathcal{S}}}_{(j_b)}(X^{(k)}(a)) = 1$ but $\ensuremath{\widehat{\mathcal{S}}}_{(j_b)}(X^{(k)}(b)) = 0$.
            \end{enumerate}
        The indicators corresponding to the other indices will not change in this case, i.e.~we will have $\ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(a)) = \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(b))$.
        To argue this, we enumerate the three alternative possibilities for the indices below.
        \begin{enumerate}
            \item The index $j$ is such that $X_j \notin \{a,b\}$. Then, changing $X_k$ from $a$ to $b$ does not affect the indicator $\ensuremath{{\mathbb{I}\left\{ X_j \notin \ensuremath{\bm{X}}_{\ensuremath{\mathcal{I}}_j} \right\}}}$, implying that $\ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(a)) = \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(b))$.
            \item For the index $k$, we have $\ensuremath{\widehat{\mathcal{S}}}_{(k)}(X^{(k)}(a)) = \ensuremath{\widehat{\mathcal{S}}}_{(k)}(X^{(k)}(b)) = 0$.
            \item The index $j \neq k$ is a second or later occurrence of either $a$ or $b$ in the original sequence.
            In this case, we again have $\ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(a)) = \ensuremath{\widehat{\mathcal{S}}}_{(j)}(X^{(k)}(b)) = 0$.
        \end{enumerate}
        Thus, $V^{(k)}(a,b) \leq 2$ in this case.
    \end{enumerate}
    Combining the four cases, we have $V^{(k)}(a,b) \leq 3$ for any $a,b \in \ensuremath{\mathcal{X}}$. This completes the proof of the lemma, and therefore the verification of Assumption~\ref{assmp:bd}.

<!-- lemma-5: body_md -->
\label{lemma:bdd_lemma_surprise_n}
    For any sequence $X^n \in \ensuremath{\mathcal{X}}^n$ and any window size $\tau \in \{1,\ldots, n-1\}$ we have
    \begin{align*}
        \sup_{a,b \in \ensuremath{\mathcal{X}}}\sup_{k \in [n]}\left|\ensuremath{\widehat{\mathcal{S}}_{\leq \zeta}}(X^{(k)}(a))-\ensuremath{\widehat{\mathcal{S}}_{\leq \zeta}}(X^{(k)}(b)) \right| \leq \frac{(2\zeta+5)}{n}.
    \end{align*}

<!-- head-10: body_md -->
Fix the sequence $X^n$, pair of symbols $a,b \in \ensuremath{\mathcal{X}}$ and index $k \in [n]$. Recall that we defined $\ensuremath{\widehat{\mathcal{S}}_{\leq \zeta,(j)}}(X^n) = \ensuremath{{\mathbb{I}\left\{ N_{X_j}(X_{\ensuremath{\mathcal{I}}_{j}}) \leq \zeta \right\}}}$, where $N_x(\cdot)$
    denotes the number of occurrences of $x$ in the sequence argument. In parallel to the
    proof of Lemma~\ref{lemma:bdd_lemma_surprise}, we define
    \begin{align*}
        V^{(k)}(a,b) = \sum_{j=1}^n \ensuremath{{\mathbb{I}\left\{ \ensuremath{\widehat{\mathcal{S}}_{\leq \zeta,(j)}}(X^{(k)}(a)) \neq \ensuremath{\widehat{\mathcal{S}}_{\leq \zeta,(j)}}(X^{(k)}(b)) \right\}}}.
    \end{align*}
    An identical series of steps as in the proof of Lemma~\ref{lemma:bdd_lemma_surprise} yields
    \[
    \bigl|\ensuremath{\widehat{\mathcal{S}}_{\leq \zeta}}(X^{(k)}(a))-\ensuremath{\widehat{\mathcal{S}}_{\leq \zeta}}(X^{(k)}(b))\bigr| \le \frac{V^{(k)}(a,b)}{n}.
    \]
    Thus, it suffices to prove that $V^{(k)}(a,b) \le 2\zeta+5$ for any pair of symbols $a,b \in \ensuremath{\mathcal{X}}$ and any index $k \in [n]$. We will prove this via a looser\footnote{Observe that specializing this bound to $\zeta = 0$ yields $V^{(k)}(a,b) \leq 5$ for the case of the surprise probability functional, which is looser than the claimed bound $V^{(k)}(a,b) \leq 3$ in the proof of Lemma~\ref{lemma:bdd_lemma_surprise}.}, but simpler and more general argument as compared to the one presented in the proof of Lemma~\ref{lemma:bdd_lemma_surprise}.

    For an index $j \in [n]$, the local estimator $\ensuremath{\widehat{\mathcal{S}}_{\leq \zeta,(j)}}$ can change only if switching the symbol at index
    $k$ from $a$ to $b$ alters the value of $\ensuremath{{\mathbb{I}\left\{ N_{X_j}(X_{\ensuremath{\mathcal{I}}_j}) \le \zeta \right\}}}$.
    In the worst case, the indicator corresponding to the index $j = k$ could change.
    For $j \neq k$, the count $N_{X_j}(X_{\ensuremath{\mathcal{I}}_j})$ changes, and therefore the corresponding local estimator can change only when $k \in \ensuremath{\mathcal{I}}_j$ and $X_j \in \{a,b\}$. Writing $A$ (respectively, $B$)
    for the number of changed indicators at positions $j \neq k$ with $X_j = a$
    (respectively, $X_j = b$), we therefore have
    \begin{align}\label{eq:V-A-B-bound}
    V^{(k)}(a,b) \le A + B + 1.
    \end{align}
    We first show that $A \le \zeta+2$. Consider an index $j \neq k$ such that $X_j = a$ and $k \in \ensuremath{\mathcal{I}}_j$. Switching $X_k$
    from $a$ to $b$ removes exactly one occurrence of $a$ from $\ensuremath{\mathcal{I}}_j$, lowering
    $N_a(X_{\ensuremath{\mathcal{I}}_j})$ by exactly one.
    Hence, the local estimator $\ensuremath{\widehat{\mathcal{S}}_{\leq \zeta,(j)}}$ changes (from $0$ to $1$) only if this count
    dips below the threshold $\zeta$, which can only happen if $N_a\bigl(X_{\ensuremath{\mathcal{I}}_j}\bigr) = \zeta+1$ (here and in all uses below, $X_{\ensuremath{\mathcal{I}}_j}$ is used as shorthand to denote the original decorrelated sequence corresponding to index $j$).
    
    We will show that the number of such indices for which this can happen is at most $\zeta + 1$.
    Let $M$ be the total number of occurrences of $a$ in the original sequence $X^{(k)}(a)$.
    If $M \leq \zeta + 1$, we are done.
    If $M > \zeta + 1$, let $j_1 < \cdots < j_M$ be the occurrences of $a$ in $X^{(k)}(a)$. For $j = j_i \neq k$, we have $\ensuremath{\mathcal{I}}_{j_i} = [n]\setminus[j_i,j_i+\tau-1]$. This yields
    \[
    N_a\bigl(X_{\ensuremath{\mathcal{I}}_{j_i}}\bigr) \geq i - 1,
    \]
    meaning that $N_a\bigl(X_{\ensuremath{\mathcal{I}}_{j_i}}\bigr) = \zeta + 1$ necessitates $i - 1 \le \zeta + 1 \implies i \leq \zeta + 2$.
    Essentially, only the local estimators corresponding to the first $\zeta+2$ occurrences
    $j_1,\ldots,j_{\zeta+2}$ of $a$ can change, giving $A \le \zeta+2$.
    A similar argument applied to the symbol $b$ (and only considering indices for which the count $N_b\bigl(X_{\ensuremath{\mathcal{I}}_j}\bigr) = \zeta$) yields $B \le \zeta + 2$. Substituting the bounds on $A$ and $B$ into Eq.~\eqref{eq:V-A-B-bound} yields the desired statement $V^{(k)}(a,b) \leq 2 \zeta + 5$ and completes the proof of the lemma.
\qed

<!-- lemma-6: body_md -->
\label{lemma:bdd_lemma_ns}
Suppose \(\ensuremath{\mathcal{X}} \subset \mathbb{R}\). For any sequence \(X^n \in \ensuremath{\mathcal{X}}^n\) and any window size
\(\tau \in \{1,\ldots,n-1\}\), we have
\[
\sup_{a,b\in\ensuremath{\mathcal{X}}}\sup_{k\in[n]}
\left|
\ensuremath{{\mathsf{\widehat{NS}}}}(X^{(k)}(a))-\ensuremath{{\mathsf{\widehat{NS}}}}(X^{(k)}(b))
\right|
\le \frac{5}{n}.
\]
<!-- end lemma-6 -->
<!-- head-11: body_md -->
Let us define the following quantity for some fixed sequence $X^n$, pair of symbols $a,b \in \ensuremath{\mathcal{X}}$ and index $k \in [n]$:
\[
V^{(k)}(a,b) := \sum_{j=1}^n \ensuremath{{\mathbb{I}\left\{ \ensuremath{{\mathsf{\widehat{NS}}}}_{(j)}(X^{(k)}(a)) \neq \ensuremath{{\mathsf{\widehat{NS}}}}_{(j)}(X^{(k)}(b)) \right\}}}.
\]
An identical series of steps as in the proof of Lemma~\ref{lemma:bdd_lemma_surprise} yields $\bigl|\ensuremath{{\mathsf{\widehat{NS}}}}(X^{(k)}(a))-\ensuremath{{\mathsf{\widehat{NS}}}}(X^{(k)}(b))\bigr| \le V^{(k)}(a,b)/n$. Thus, it suffices to prove that $V^{(k)}(a,b)\le 5$ for any pair of symbols $a, b \in \ensuremath{\mathcal{X}}$ and any index $k \in [n]$.
We first consider all indices $j \neq k$, i.e. indices that remain unchanged between the original and modified sequence.
Here, the local estimator $\ensuremath{{\mathsf{\widehat{NS}}}}_{(j)}$ can change only when $k\in\ensuremath{\mathcal{I}}_j$.
In this case, denoting as shorthand
\[
M_j := \min_{i\in\ensuremath{\mathcal{I}}_j\setminus\{k\}} |X_j - X_i|,
\]
we can write
\[
\ensuremath{{\mathsf{\widehat{NS}}}}_{(j)}\bigl(X^{(k)}(c)\bigr) = \ensuremath{{\mathbb{I}\left\{ M_j > \delta \ \text{ and }\ |X_j - c| > \delta \right\}}},
\text{ for } c\in\{a,b\}.
\]
Thus, the local estimator $\ensuremath{{\mathsf{\widehat{NS}}}}_{(j)}$ changes \emph{iff} $M_j > \delta$ and exactly one of $a,b$ lies within distance $\delta$ of $X_j$; i.e. either $\{|X_j - a| \leq \delta \text{ and } |X_j - b| > \delta\}$ or $\{|X_j - a| > \delta \text{ and } |X_j - b| \leq \delta \}$. Accordingly, define the (possibly larger) sets of \emph{type-$a$} and \emph{type-$b$} indices as below:
\begin{align*}
\ensuremath{\mathcal{T}}_a &:= \bigl\{\, j \neq k \ :\ k\in\ensuremath{\mathcal{I}}_j,\ \ M_j > \delta,\ \ \text{ and } |X_j - a| \le \delta \,\bigr\}\\
\ensuremath{\mathcal{T}}_b &:= \bigl\{\, j \neq k \ :\ k\in\ensuremath{\mathcal{I}}_j,\ \ M_j > \delta,\ \ \text{ and } |X_j - b| \le \delta \,\bigr\}.
\end{align*}
Clearly, every changed index $j\neq k$ must lie in $\ensuremath{\mathcal{T}}_a$ or $\ensuremath{\mathcal{T}}_b$ or both.
Including the index $j = k$, we then have
\begin{align}\label{eq:nns_V_split}
V^{(k)}(a,b) \;\le\; 1 + |\ensuremath{\mathcal{T}}_a| + |\ensuremath{\mathcal{T}}_b|.
\end{align}
 
We will now show that $|\ensuremath{\mathcal{T}}_a| \le 2$; the bound $|\ensuremath{\mathcal{T}}_b| \le 2$ can similarly be shown by an identical argument with $b$ in place of $a$.
To show this, we first claim that any pair of distinct indices $j \neq j' \in \ensuremath{\mathcal{T}}_a$ must be far apart, meaning that $|X_j - X_{j'}| > \delta$.
Without loss of generality, let $j < j'$. Since $\ensuremath{\mathcal{D}}_{j'} = \{j',\ldots,(j'+\tau-1)\wedge n\}$, and $j < j'$, we get $j \in \ensuremath{\mathcal{I}}_{j'}$. Since we are considering indices $j \neq k$, we further have $j \in \ensuremath{\mathcal{I}}_{j'}\setminus\{k\}$.
Therefore, we obtain
                                                         
\[
|X_{j'} - X_j| \;\ge\; \min_{i\in\ensuremath{\mathcal{I}}_{j'}\setminus\{k\}} |X_{j'} - X_i| \;=\; M_{j'} \;>\; \delta,
\]
where the last inequality uses the fact that $j' \in \ensuremath{\mathcal{T}}_a$.
This proves the claim.

Consequently, the points $\{X_j : j \in \ensuremath{\mathcal{T}}_a\}$ have pairwise distance greater than $\delta$, i.e. $|X_j - X_{j'}| > \delta$ for all $j \neq j' \in \ensuremath{\mathcal{T}}_a$. 
On the other hand, the definition of $\ensuremath{\mathcal{T}}_a$ implies that $X_j \in [a - \delta, a + \delta]$ for all $j \in \ensuremath{\mathcal{T}}_a$, meaning that the points all lie in an interval of length $2\delta$.
Putting these statements together yields $|\ensuremath{\mathcal{T}}_a| \le 2$, and likewise $|\ensuremath{\mathcal{T}}_b| \le 2$. Substituting these bounds into Equation~\eqref{eq:nns_V_split}, we obtain $V^{(k)}(a,b) \le 5$, and so we have completed the proof of the lemma.
\qed
