\label{remarkmatrices}
        Recall that the constant $K(2,2,\alpha,\alpha,r)=K_{r,\alpha}$ is sharp for the Fourier transform.
        On the other hand, inequality \eqref{PittForMetaps3} for $p=q=2$ gives 
        \begin{equation}
            \int_{\mathbb{R}^n}|B^{+}\mathcal{P}_{R(B)}^{A(\ker(B))}\xi|^{-2\alpha}|\widehat Sf(\xi)|^2\mathrm{d} \xi\leq K_{r,\alpha}\int_{\mathbb{R}^n}|\mathcal{P}_{\ker(B)^\perp}^{D^\top A(\ker(B))}x|^{2\alpha} |f(x)|^2\mathrm{d} x,
        \end{equation}
        where $0\leq\alpha<r/2$. This is precisely Proposition \ref{intro.thm:Pittpq2}.
        This inequality is sharp because it can be obtained by a sequence of identities applied to a sharp inequality. Formally, let $(h_1^{(k)})_k\subset\mathcal{S}(\mathbb{R}^r)$ be a sequence of functions so that
        \begin{equation}
            \lim_{k\to\infty}\frac{\int_{\mathbb{R}^r}|\eta|^{-2\alpha}|\widehat h_1^{(k)}(\eta)|^2\mathrm{d}\eta}{\int_{\mathbb{R}^r}|u|^{2\alpha}|h_1^{(k)}(u)|^2\mathrm{d} u}=K_{r,\alpha}.
        \end{equation}
        Take a sequence $f^{(k)}\in\mathcal{S}(\mathbb{R}^n)$ so that the corresponding $g_{\xi_2}^{(k)}$ in \eqref{defgxi2} tensorises: $$g_{\xi_2}^{(k)}(u)=h_1^{(k)}(u)h_2(D^\top\xi_2), \qquad u\in\mathbb{R}^r,\; \xi_2\in A(\ker(B)).$$
        By \eqref{Sfghat}, the modulus of $\widehat Sf$ also tensorises as
        $|\widehat Sf(BV\eta+\xi_2)|=\mu_S|\widehat{h_1^{(k)}}(\eta)h_2(D^\top\xi_2)|$. Consequently, integrating over $A(\ker(B))$ and using \eqref{presharp2} and \eqref{geds} with
        $p=2$, for which $[q_{R(B)^\perp}(A^\top)\sigma(B)]^{\frac p2-1}=1$, we obtain the identities
        \begin{align}\label{sharpnum}
            \int_{A(\ker(B))}\int_{\mathbb{R}^r}|\eta|^{-2\alpha}|\widehat{g_{\xi_2}^{(k)}}(\eta)|^2\mathrm{d}\eta\mathrm{d}\xi_2
            &=q_{\ker(B)}(A)\int_{\mathbb{R}^n}|B^+\mathcal{P}_{R(B)}^{A(\ker(B))}\xi|^{-2\alpha}|\widehat Sf^{(k)}(\xi)|^2\mathrm{d}\xi,\\
            \label{sharpden}
            \int_{A(\ker(B))}\int_{\mathbb{R}^r}|u|^{2\alpha}|g_{\xi_2}^{(k)}(u)|^2\mathrm{d} u\mathrm{d}\xi_2
            &=q_{\ker(B)}(A)\int_{\mathbb{R}^n}|\mathcal{P}_{\ker(B)^\perp}^{D^\top A(\ker(B))}x|^{2\alpha}|f^{(k)}(x)|^2\mathrm{d} x.
        \end{align}
        On the other hand, the tensorisation of $|g_{\xi_2}^{(k)}|$ and of $|\widehat{g_{\xi_2}^{(k)}}|$
        makes both left-hand sides factor, with the same factor
        $\Vert h_2(D^\top\cdot)\Vert_{L^2(A(\ker(B)))}^2>0$:
        \begin{align}
            \int_{A(\ker(B))}\int_{\mathbb{R}^r}|\eta|^{-2\alpha}|\widehat{g_{\xi_2}^{(k)}}(\eta)|^2\mathrm{d}\eta\mathrm{d}\xi_2
            &=\Vert h_2(D^\top\cdot)\Vert_{L^2(A(\ker(B)))}^2\int_{\mathbb{R}^r}|\eta|^{-2\alpha}|\widehat{h_1^{(k)}}(\eta)|^2\mathrm{d}\eta,\\
            \int_{A(\ker(B))}\int_{\mathbb{R}^r}|u|^{2\alpha}|g_{\xi_2}^{(k)}(u)|^2\mathrm{d} u\mathrm{d}\xi_2
            &=\Vert h_2(D^\top\cdot)\Vert_{L^2(A(\ker(B)))}^2\int_{\mathbb{R}^r}|u|^{2\alpha}|h_1^{(k)}(u)|^2\mathrm{d} u.

        singular contribution of $\Vert h_2(D^\top\cdot)\Vert_{L^2(A(\ker(B)))}^2$ cancel, and we are left with

            \frac{\displaystyle\int_{\mathbb{R}^n}|B^+\mathcal{P}_{R(B)}^{A(\ker(B))}\xi|^{-2\alpha}|\widehat Sf^{(k)}(\xi)|^2\mathrm{d}\xi}
            {\displaystyle\int_{\mathbb{R}^n}|{\mathcal{P}_{\ker(B)^\perp}^{D^\top A(\ker(B))}x}|^{2\alpha}|f^{(k)}(x)|^2\mathrm{d} x}
            =\frac{\displaystyle\int_{\mathbb{R}^r}|\eta|^{-2\alpha}|\widehat{h_1^{(k)}}(\eta)|^2\mathrm{d}\eta}

        \end{equation}
        Therefore no constant strictly smaller than $K_{r,\alpha}$ can replace this constant in \eqref{PittwithmatricesBinv}, which proves the sharpness. 
        Observe that such a sequence $f^{(k)}$ does exist: by \eqref{symplDecom1} and Proposition \ref{PropIsomorphisms} $(iii)$, every $x\in\mathbb{R}^n$ is uniquely written as $x=Vu+D^\top\xi_2$ with $u\in\mathbb{R}^r$ and $\xi_2\in A(\ker(B))$. Therefore, setting $$f^{(k)}(Vu+D^\top\xi_2):=h_1^{(k)}(u)h_2(D^\top\xi_2)e^{-i\pi(V^\top B^+AVu\cdot u-2V^\top C^\top\xi_2\cdot u)},$$ with $h_2(D^\top\cdot)\in\mathcal{S}(A(\ker(B)))\setminus\{0\}$, defines $f^{(k)}\in\mathcal{S}(\mathbb{R}^n)$, and  \eqref{defgxi2} gives $|g_{\xi_2}^{(k)}(u)|=|h_1^{(k)}(u)h_2(D^\top\xi_2)|$, with the phase factor being unimodular. 
        It is worth stressing that the extremal ratio depends on the metaplectic operator only through $r=\mathrm{rank}(B)$: the singular directions contribute the same factor to both sides and disappear from the sharp constant.

<!-- theorem-7: body_md -->
\label{thm:metaplecticSobolev}
Let $\widehat S\in\operatorname*{Mp}(n,\mathbb{R})$.
\begin{enumerate}[(i)]
    \item If $C=O$, then $\widehat S$ extends to an isomorphism of $\dot H^\alpha(\mathbb{R}^n)$ for every $-\infty<\alpha<n/2$. More precisely,
    \begin{equation}\label{sob.CzeroIdentity}
        \|\widehat Sf\|_{\dot H^\alpha}^2
        =\int_{\mathbb{R}^n}|D\xi|^{2\alpha}|\widehat f(\xi)|^2\mathrm{d}\xi,
    \end{equation}
    and
    \begin{equation}\label{sob.CzeroNorm}
        \|\widehat S\|_{\dot H^\alpha\to\dot H^\alpha}
        \leq\max\big\{\sigma_{\min}(D)^\alpha,\sigma_{\max}(D)^\alpha\big\}.
    \end{equation}
    In particular, if furthermore $D=I$, then $ \|\widehat Sf\|_{\dot H^\alpha}^2
        =1$. 
    
    \item If $C\neq O$, set $r=\mathrm{rank}(C)$. For every
    $0\leq\alpha<r/2$, the operator $\widehat S$ extends boundedly from
    $\dot H^\alpha(\mathbb{R}^n)$ to $\dot H^{-\alpha}(\mathbb{R}^n)$, and
    \begin{equation}\label{sob.PittSobolev}
        \|\widehat Sf\|_{\dot H^{-\alpha}}
        \leq
        K_{r,\alpha}^{1/2}\,
        \Vert\mathcal{P}_{\ker(C)^\perp}^{A^\top D(\ker(C))}\Vert^\alpha
        \Vert C^+\mathcal{P}_{R(C)}^{D(\ker(C))}\Vert^\alpha
        \|f\|_{\dot H^\alpha}.
    \end{equation}
\end{enumerate}

Theorem \ref{intro.thm:mainmetap} ($p=q=2$) implies the boundedness of metaplectic operators within inhomogeneous (or Lebesgue-)Sobolev spaces $H^\alpha(\mathbb{R}^n)$.
    Recall that $f\in H^\alpha(\mathbb{R}^n)$ for $\alpha\in\mathbb{R}$ if 
    \begin{equation}
        \|f\|_{H^\alpha}^2=\int_{\mathbb{R}^n}(1+|\xi|^2)^\alpha|\widehat f(\xi)|^2\mathrm{d}\xi<\infty.
    \end{equation}
    Indeed, it follows by the trivial upper bound $|\cdot|^2\leq1+|\cdot|^2$ that if $C=O$, then $\widehat S:H^\alpha(\mathbb{R}^n)\to H^\alpha(\mathbb{R}^n)$ for every $\alpha\in\mathbb{R}$, and $\widehat S:H^\alpha(\mathbb{R}^n)\to H^{-\alpha}(\mathbb{R}^n)$ for every $0\leq\alpha<\mathrm{rank}(C)/2$ otherwise. 
    In time-frequency analysis, boundedness of metaplectic operators within inhomogeneous Sobolev spaces is widely studied, because $H^\alpha(\mathbb{R}^n)$ is a particular instance of modulation space. Instead, our Pitt's inequality is stronger and is outside the realm of time-frequency analysis.

<!-- lemma-3: body_md -->
\label{Lemma-COVar}
    Let $\mathcal{L}\subseteq\mathbb{R}^n$ be a subspace of $\mathbb{R}^n$ of dimension $r>0$. Let $E=(E_1|E_2)$, be such that the columns of $E_1$ form an orthonormal basis of $\mathcal{L}^\perp$ and those of $E_2$ form an orthonormal basis of $\mathcal{L}$. Assume that $\mathbb{R}^n=\mathcal{L}^\perp\oplus M(\mathcal{L})$. Then, the modulus of the determinant of the Jacobian of 
    \[\Psi:(u,v)\in\mathbb{R}^{n-r}\times \mathbb{R}^{r}\longmapsto E_1u+ME_2v\in\mathcal{L}^\perp\oplus M(\mathcal{L})
    \]
    is $|\det\nabla_{(u,v)}\Psi|=|\det(E_2^\top ME_2)|$.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Since
    \begin{equation}
        \Psi(u,v)=E_1u+ME_2v=(E_1|ME_2)(u,v)^\top,
    \end{equation}
    the matrix of $\Psi$ is $N=(E_1|ME_2)$.
               
                      
                                                                                                                                                                                                       
                    
    Therefore, using that $E=(E_1|E_2)$ is an orthogonal linear parametrisation of $\mathcal{L}$,
    \begin{align}
        |\det\nabla_{(u,v)}\Psi|= |\det(N)|=|\det(E^\top N)|=\left|\det{
        \begin{pmatrix}
            I_{n-r} & E_1^\top ME_2\\
            O_{r\times (n-r)} & E_2^\top ME_2
        \end{pmatrix}
        }\right|=|\det(E_2^\top ME_2)|.
    \end{align}
    This concludes the proof.
