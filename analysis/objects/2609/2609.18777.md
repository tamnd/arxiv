The $\mathbb{Q}$-algebra $\mathfrak{Z}$ is a graded $\mathbb{Q}$-algebra, i.e., $\mathfrak{Z}=\bigoplus_{w\geq 0}\mathfrak{Z}_{w}$.

<!-- thm-1: body_md -->
\label{Thm:BK26}
    Let $\mathcal{E}_{\mathbb{C}}$ be the $\mathbb{C}$-vector space spanned by all multiple Eisenstein series. Then the following results hold.
    \begin{enumerate}
        \item $\mathcal{E}=\bigoplus_{w\geq 0}\mathcal{E}_{w}$.
        \item The natural map $\mathcal{E}\otimes_{\mathbb{Q}}\mathbb{C} \twoheadrightarrow \mathcal{E}_{\mathbb{C}}$ is an isomorphism.
    \end{enumerate}
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\label{thm.Goncharov-Zagier conjecture for function field}
    We keep the notation as above. Then the following results hold.
    \begin{enumerate}  
        \item $\mathcal{Z}_{\overline{K}}$ is a graded algebra, i.e.,  $\mathcal{Z}_{\overline{K}} = \bigoplus_{w=0}^{\infty} \mathcal{Z}_{\overline{K}, w}$.
        \item The natural map $\mathcal{Z}_K\otimes_K \overline{K}\to \mathcal{Z}_{\overline{K}}$ is an isomorphism.
    \end{enumerate}
<!-- end thm-2 -->
<!-- df-1: body_md -->
\label{df.multiple Eisenstein series}
    Fix a positive integer $r$.
    For any $\mathbf{z}\in \Omega^r$ and nonempty index $\mathfrak{s} = (s_1,\ldots,s_m)$, we define the multiple Eisenstein series of rank $r$ by
    \[
    E_r(\mathfrak{s}; \mathbf{z}) = \sum_{\substack{\bm{f}_1,\ldots,\bm{f}_m \in A^r \\  \bm{f}_1\succ \cdots \succ  \bm{f}_m \succ 0}}\frac{1}{\left\langle \bm{f}_1,\mathbf{z}\right\rangle^{s_1}\cdots \left\langle \bm{f}_m, \mathbf{z}\right\rangle^{s_m}}.
    \]
    We adopt the convention that $E_r(\varnothing; \mathbf{z}): = 1$.
<!-- end df-1 -->
<!-- thm-3: body_md -->
\label{thm.main-thm}
    For any integer $r\geq 1$, the following results hold.
    \begin{itemize}
        \item[(i)]  $\mathcal{Z}^{(r)}_{\overline{K}}$ is a graded algebra, i.e.,  \[\mathcal{Z}^{(r)}_{\overline{K}} = \bigoplus_{w=0}^{\infty}\mathcal{Z}^{(r)}_{\overline{K},w}.\]
        \item[(ii)] The natural map $\mathcal{Z}^{(r)}_{K}\otimes_K \overline{K}\twoheadrightarrow  \mathcal{Z}^{(r)}_{\overline{K}}$ is an isomorphism of $\overline{K}$-algebras. 
    \end{itemize}
<!-- end thm-3 -->
<!-- rmk-1: body_md -->

\label{rmk:rank 1}
We fix a $(q-1)$-st root of $(-\theta)$ and let
\[
\widetilde{\pi}
=
(-\theta)^{\frac{q}{q-1}}
\prod_{i=1}^{\infty}
\left(1-\frac{\theta}{\theta^{q^i}}\right)^{-1}
\in
K_{\infty}((-\theta)^{\frac{1}{q-1}})
\]
be a fixed fundamental period of the Carlitz $A$-module, namely,
$\phi_{\theta}^{\Lambda_{\widetilde{\pi}A}}=\theta+\tau$
(see \cite[\S3.2]{goss1996basic}). When $r=1$, we have the following.
\begin{enumerate}
    \item The ``rank one discriminant form'' $\Delta_1$ is given by
    $\Delta_1=\widetilde{\pi}^{q-1}$ since the Drinfeld $A$-module $\phi^A$
    associated with the point $1\in\Omega^1$ is determined by
    \[
    \phi^A_\theta=\theta+\widetilde{\pi}^{q-1}\tau.
    \]
    By the Euler--Carlitz formula \cite{Car35} (see also \cite[Theorem~5.2.1]{thakur2004function}),
    $\Delta_1$ is a $K$-multiple of $\zeta_A(q-1)$. Hence
    \[
    \Delta_1^m\in\mathcal{Z}^{(1)}_{K,m(q-1)}
    \qquad (m\geq0),\qquad \text{and}\qquad
    L[\Delta_1]\subseteq\mathcal{Z}^{(1)}_L
    \]
    for any subfield $L$ of $\mathbb{C}_\infty$ containing $K$.

    \item For every $n\geq0$, the Carlitz exponential formula
    (see \cite[\S3.2]{goss1996basic}) gives
    \[
    e_n(1)
    =
    \frac{\widetilde{\pi}^{q^n-1}}{D_n}
    =
    \frac{1}{D_n}\Delta_1^{(q^n-1)/(q-1)}
    \in
    \mathcal{Z}^{(1)}_{K,q^n-1},
    \]
    where $D_0=1$ and $D_n = \prod_{j=0}^{n-1} (\theta^{q^n} - \theta^{q^j})$.
\end{enumerate}

\label{lem:comparison between >_N and >}
    Let $\bm{f},\bm{g}\in A^r$ be nonzero with
    $|\bm{f}|,|\bm{g}|<q^N$. Then
    \begin{enumerate}
        \item $\bm{f}\succ_N0$ if and only if $\bm{f}\succ0$.
        \item $\bm{f}\succ_N\bm{g}$ if and only if $\bm{f}\succ\bm{g}$.
    \end{enumerate}

<!-- proof-9: body_md -->
Write $\bm{f}=(f_1,\ldots,f_r)$. We have
    \[
    \bm{f}\gamma_N
    =
    (f_1,\ldots,f_{r-2},
    \theta^Nf_{r-1}+f_r,-f_{r-1}).
    \]
    If $\ell(\bm{f})\leq r-2$, the first nonzero coordinate of
    $\bm{f}\gamma_N$ is the same as that of $\bm{f}$, so the first
    assertion is immediate. If $\ell(\bm{f})=r-1$, then
