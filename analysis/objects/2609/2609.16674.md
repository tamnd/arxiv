<!-- question-1: body_md -->
\label{ques:disk-triangle}
Let $D\subset\mathbb{R}^2$ be a disk of positive radius, and let $\Delta\subset\mathbb{R}^2$
be a nondegenerate triangle.
\begin{enumerate}
    \item[(i)] Does $L^2(D)$ admit a Riesz basis of exponentials?
    \item[(ii)] Does $L^2(\Delta)$ admit a Riesz basis of exponentials?
\end{enumerate}
<!-- end question-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Let $d\ge2$, and let $S\subset\mathbb{R}^d$ be one of the following sets:
\begin{enumerate}
    \item[(i)] A Euclidean ball $B(c,R)$, where $R>0$.

    \item[(ii)] A bounded convex polytope with nonempty interior having a facet
    whose supporting hyperplane is parallel to that of no other facet.
    In particular, this includes every nondegenerate $d$-simplex.

    \item[(iii)] A spherical shell $\{x\in\mathbb{R}^d:r<|x-c|<R\}$,
    where $0<r<R$.

    \item[(iv)] A finite union of Euclidean balls,
    \begin{equation*}
        S=\bigcup_{j=1}^{m}B(c_j,R_j),
        \qquad m\ge1,\quad R_j>0.
    \end{equation*}
    The balls are allowed to overlap.

    \item[(v)] A finite union $S=\bigcup_{j=1}^{m}S_j$ of Euclidean balls or
    spherical shells, where
    \begin{equation*}
        S_j=\{x\in\mathbb{R}^d:a_j<|x-c_j|<b_j\},
        \qquad 0\le a_j<b_j,
    \end{equation*}
    and
    \begin{equation*}
        \operatorname{dist}(\overline{S_i},\overline{S_j})>0
        \qquad (i\ne j).
    \end{equation*}
    When $a_j=0$, $S_j$ is understood to be the ball $B(c_j,b_j)$.
\end{enumerate}
Then $L^2(S)$ admits no Riesz basis of exponentials. The same conclusion holds
for every set $LS+v$, where $L$ is an invertible real $d\times d$ matrix and
$v\in\mathbb{R}^d$. It also holds for $S\times K$ and its invertible affine images,
where $k\ge1$ and $K\subset\mathbb{R}^k$ is any Lebesgue measurable set with
$0<|K|<\infty$.
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
\label{rem:more-general-sets}
In fact, our method also yields the nonexistence of Riesz bases of exponentials
for many more general sets.

<!-- theorem-2: body_md -->
\label{thm:abstract-windows}
There do not exist constants $\varepsilon_0>0$, $0<A\le B<\infty$ and $0\le J<\infty$,
and measurable sets $V_{\varepsilon,t},W_{\varepsilon,t}\subset U$ for
$0<\varepsilon<\varepsilon_0$ and $t\in I$, satisfying all the following conditions.
For each fixed $\varepsilon$, the indicators of the two windows are jointly measurable
in $(t,x)$.
\begin{enumerate}
    \item[(i)] The operator $T_{V_{\varepsilon,t}}$ is onto, and
    \begin{equation}\label{eq:main-window-bounds}
        \begin{aligned}
            AI\le X_{\varepsilon,t}:=T^*P_{V_{\varepsilon,t}}T\le BI,\\
            AI\le Y_{\varepsilon,t}:=T^*P_{W_{\varepsilon,t}}T\le BI
        \end{aligned}
    \end{equation}
    for every $\varepsilon,t$.

    \item[(ii)] For each $\varepsilon$ there is a measurable set $Q_\varepsilon\subset U$
    of positive measure such that
    \begin{equation*}
        \bigl|Q_\varepsilon\setminus(V_{\varepsilon,t}\setminus W_{\varepsilon,t})\bigr|=0
        \qquad(t\in I).
    \end{equation*}

    \item[(iii)] Writing $L_{\varepsilon,t}=W_{\varepsilon,t}\setminus V_{\varepsilon,t}$, we have
    \begin{equation}\label{eq:main-loss-overlap}
        \omega_\varepsilon:=\operatorname*{ess\,sup}_{x\in U}
        \int_I\1_{L_{\varepsilon,t}}(x)\,\mathrm{d}\nu(t)
        \longrightarrow0\qquad{\rm{as}}\ \varepsilon\to0.
    \end{equation}

    \item[(iv)] For each $\varepsilon$, the indicator representatives satisfy
    \begin{equation}\label{eq:main-turnover}
        \sum_{j=1}^{n}
        \left|\1_{V_{\varepsilon,t_j}}(x)-\1_{V_{\varepsilon,t_{j-1}}}(x)\right|
        \le J
    \end{equation}
    for almost every $x\in U$ and every finite sequence
    $t_0<\cdots<t_n$ in $I$. The exceptional null set is independent of
    the chosen sequence.
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:main-direct-cover}
Suppose that the windows satisfy \eqref{eq:main-turnover}. For every $v\in H$
and $\eta>0$, the set $\{X_{\varepsilon,t}v:t\in I\}$ has an $\eta$-net whose
centres belong to this set and whose cardinality is at most
\begin{equation}\label{eq:main-direct-number}
    \left\lceil1+\frac{JC^2\|v\|^2}{\eta^2}\right\rceil.
\end{equation}
The bound is independent of $\varepsilon$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $\varepsilon$ and omit it from the notation. For any $t_0<\cdots<t_n$, we have
\begin{equation}\label{eq:main-square-variation}
    \begin{aligned}
    \sum_{j=1}^{n}\|(X_{t_j}-X_{t_{j-1}})v\|^2
    &\le C\sum_{j=1}^{n}
       \left\|(P_{V_{t_j}}-P_{V_{t_{j-1}}})Tv\right\|_{L^2(U)}^2\\
    &=C\int_U|Tv(x)|^2\sum_{j=1}^{n}
       \left|\1_{V_{t_j}}(x)-\1_{V_{t_{j-1}}}(x)\right|^2\mathrm{d} x\\
    &\le JC\left\|Tv\right\|_{L^2(U)}^2
     \le JC^2\|v\|^2.
    \end{aligned}
\end{equation}
Here the square of each indicator difference equals its absolute value.

If $m$ orbit points have pairwise distances greater than $\eta$, order their
parameters increasingly. Consecutive points still have distance greater than
$\eta$, so \eqref{eq:main-square-variation} gives
$(m-1)\eta^2\le JC^2\|v\|^2$.
Starting with any orbit point, keep adding a point at distance greater than
$\eta$ from all previously chosen points whenever one exists.
The procedure terminates after finitely many steps, with the number of chosen
points bounded by \eqref{eq:main-direct-number}. The resulting set is the required net.

<!-- lemma-2: body_md -->
\label{lem:main-inverse-cover}
Suppose that $AI\le X_{\varepsilon,t}\le BI$ and that
\eqref{eq:main-square-variation} holds. For every $M>0$ and $\delta>0$,
there is a finite number $N_\delta$, depending only on $A,B,C,J,M,\delta$,
with the following property. For every $a\in H$ with $\|a\|\le M$ and
every $\varepsilon$, the set
\begin{equation*}
    \{X_{\varepsilon,t}^{-1}a:t\in I\}
\end{equation*}
has a $\delta$-net of at most $N_\delta$ centres, each of norm at most $M/A$.

\begin{equation*}
    q=1-\frac AB,\qquad R_t=I-\frac{X_t}{B},

\end{equation*}

\begin{equation}\label{eq:main-iteration}
    y_{0,t}=0,\qquad y_{n+1,t}=\frac aB+R_ty_{n,t}.
\end{equation}
Since $c_t=a/B+R_tc_t$ and $M/B+q\rho=\rho$, induction gives
\begin{equation}\label{eq:main-iteration-error}
    \|y_{n,t}\|\le\rho,
    \qquad\|c_t-y_{n,t}\|\le\rho q^n.
\end{equation}
We next construct finite approximations to these iterates with an error bound
independent of $n$ and $\varepsilon$.

Choose
\begin{equation}\label{eq:main-net-precision}
    h=\frac{(1-q)\delta}{2}=\frac{A\delta}{2B},
    \qquad
    L_\delta=\left\lceil1+\frac{4JC^2M^2}{A^4\delta^2}\right\rceil.
\end{equation}
For each fixed $z$ with $\|z\|\le\rho$, the identity
$(R_t-R_s)z=-(X_t-X_s)z/B$ and the packing argument in the proof of
Lemma~\ref{lem:main-direct-cover} show that
$\{R_tz:t\in I\}$ has an $h$-net $\mathcal F_z$ with centres in that orbit and
\begin{equation*}
    \#\mathcal F_z
    \le\left\lceil1+\frac{JC^2\rho^2}{B^2h^2}\right\rceil
    =L_\delta.
\end{equation*}
Starting from $\mathcal E_0=\{0\}$, define
\begin{equation*}
    \mathcal E_{n+1}
    =\bigcup_{z\in\mathcal E_n}\left(\frac aB+\mathcal F_z\right).
\end{equation*}
Each new centre has norm at most $M/B+q\rho=\rho$. Thus,
\begin{equation}\label{eq:main-net-recursion}
    \#\mathcal E_n\le L_\delta^n,
    \qquad\|z\|\le\rho\quad(z\in\mathcal E_n).
\end{equation}
Set $e_0=0$ and $e_{n+1}=qe_n+h$. We prove by induction that $\mathcal E_n$
covers all $y_{n,t}$ with error $e_n$. Suppose this holds at step $n$.
Choose $z\in\mathcal E_n$ within $e_n$ of $y_{n,t}$, and then choose a point

    e_n=h\sum_{j=0}^{n-1}q^j\le\frac\delta2.

Equations~\eqref{eq:main-iteration-error} and \eqref{eq:main-net-recursion}
show that $\mathcal E_m$ is a $\delta$-net for all $c_t$ with
$\#\mathcal E_m\le N_\delta:=L_\delta^m$ and centre norms at most $\rho$.
The choices of $m$ and $L_\delta$ depend only on the stated constants.

Suppose that all four conditions hold. For each $\varepsilon$, choose
\begin{equation*}
    f_\varepsilon=|Q_\varepsilon|^{-1/2}\1_{Q_\varepsilon}\in L^2(U),
    \qquad a_\varepsilon=T^*f_\varepsilon.
\end{equation*}
The function $f_\varepsilon$ has norm one and is fixed as $t$ varies.
By condition~(ii),
\begin{equation*}
    a_\varepsilon=T_{V_{\varepsilon,t}}^*(f_\varepsilon|_{V_{\varepsilon,t}}),
    \qquad\|a_\varepsilon\|\le\sqrt B.
\end{equation*}
Condition~(i) makes $T_{V_{\varepsilon,t}}$ invertible. Hence
\begin{equation}\label{eq:main-exact-reconstruction}
    c_{\varepsilon,t}:=X_{\varepsilon,t}^{-1}a_\varepsilon,
    \qquad T_{V_{\varepsilon,t}}c_{\varepsilon,t}=f_\varepsilon|_{V_{\varepsilon,t}}.
\end{equation}
Applying the two bounds on this window gives
\begin{equation}\label{eq:main-coefficient-size}
    \frac1B\le\|c_{\varepsilon,t}\|^2\le\frac1A.
\end{equation}

The function $Tc_{\varepsilon,t}$ vanishes almost everywhere on
$V_{\varepsilon,t}\cap W_{\varepsilon,t}$. Thus all its energy on the second window lies in
$L_{\varepsilon,t}$. More precisely, set
\begin{equation*}
    Z_{\varepsilon,t}=T^*P_{L_{\varepsilon,t}}T.
\end{equation*}
Then $0\le Z_{\varepsilon,t}\le Y_{\varepsilon,t}\le BI$, and
\begin{equation}\label{eq:main-forced-energy}
    \langle Z_{\varepsilon,t}c_{\varepsilon,t},c_{\varepsilon,t}\rangle
    =\left\|T_{W_{\varepsilon,t}}c_{\varepsilon,t}\right\|_{L^2(W_{\varepsilon,t})}^2
    \ge A\|c_{\varepsilon,t}\|^2\ge\frac AB.
\end{equation}

On the other hand, for every fixed $v\in H$, Tonelli's theorem gives
\begin{equation}\label{eq:main-fixed-energy}
    \begin{aligned}
    \int_I\langle Z_{\varepsilon,t}v,v\rangle\,\mathrm{d}\nu(t)
    &=\int_U|Tv(x)|^2
      \left(\int_I\1_{L_{\varepsilon,t}}(x)\,\mathrm{d}\nu(t)\right)\mathrm{d} x\\
    &\le\omega_\varepsilon\left\|Tv\right\|_{L^2(U)}^2
     \le C\omega_\varepsilon\|v\|^2.
    \end{aligned}
\end{equation}
To compare this with \eqref{eq:main-forced-energy}, fix $\delta>0$ and apply
Lemma~\ref{lem:main-inverse-cover} with $M=\sqrt B$. For each $\varepsilon$, it gives
a finite set
\begin{equation*}
    \{v_{\varepsilon,1},\ldots,v_{\varepsilon,N_\varepsilon}\},
    \qquad N_\varepsilon\le N_\delta,
    \qquad\|v_{\varepsilon,j}\|\le\frac{\sqrt B}{A},
\end{equation*}
within distance $\delta$ of every $c_{\varepsilon,t}$. The number $N_\delta$ is
independent of $\varepsilon$.

For each $t$, choose a centre $v_{\varepsilon,j}$ with
$\|c_{\varepsilon,t}-v_{\varepsilon,j}\|\le\delta$.
Using $\|Z_{\varepsilon,t}^{1/2}\|\le\sqrt B$, we obtain
\begin{equation*}
    \frac AB
    \le\|Z_{\varepsilon,t}^{1/2}c_{\varepsilon,t}\|^2
    \le2\|Z_{\varepsilon,t}^{1/2}v_{\varepsilon,j}\|^2+2B\delta^2
    \le2\sum_{j=1}^{N_\varepsilon}
       \langle Z_{\varepsilon,t}v_{\varepsilon,j},v_{\varepsilon,j}\rangle+2B\delta^2.
\end{equation*}
Each centre is fixed as $t$ varies, so the last expression is measurable.
Integrating and using \eqref{eq:main-fixed-energy} gives
\begin{equation}\label{eq:main-final-estimate}
    \frac AB
    \le\frac{2CB}{A^2}N_\delta\omega_\varepsilon+2B\delta^2.
\end{equation}
Choose $\delta^2=A/(8B^2)$, so the second term is $A/(4B)$.
With this choice fixed, condition~(iii) allows us to take $\varepsilon$ small enough
that the first term is less than $A/(4B)$.
This contradicts \eqref{eq:main-final-estimate}.

\label{lem:geom-moving-balls}
Let $R,a>0$ and $b\in\mathbb{R}^d$. Define
\begin{equation}\label{eq:geom-component-windows}
    V_{\varepsilon,\theta}^{a}=B\bigl(b-(R-\varepsilon)u_\theta,a\bigr),
    \qquad
    W_{\varepsilon,\theta}^{a}=B\bigl(b-(R+\varepsilon)u_\theta,a\bigr).
\end{equation}
There is a constant $C_{R,a}$, independent of $b,x,\varepsilon$, such that
\begin{equation}\label{eq:geom-outer-loss}
    \sup_{x\in\mathbb{R}^d}
    \bigl|\{\theta\in[0,2\pi]:
        x\in W_{\varepsilon,\theta}^{a}\setminus V_{\varepsilon,\theta}^{a}\}\bigr|
    \le C_{R,a}\sqrt{\varepsilon}
\end{equation}
whenever $0<\varepsilon<R/2$. If $0<a<R$, then also
\begin{equation}\label{eq:geom-inner-loss}
    \sup_{x\in\mathbb{R}^d}
    \bigl|\{\theta\in[0,2\pi]:
        x\in V_{\varepsilon,\theta}^{a}\setminus W_{\varepsilon,\theta}^{a}\}\bigr|
    \le C_{R,a}\sqrt{\varepsilon}
\end{equation}
for $0<\varepsilon<\min\{R/2,(R-a)/2\}$. Each of the two ball indicators has total
variation at most $2$ on any interval of length at most $2\pi$.

When $p>0$, there is $\phi\in\mathbb{R}$ such that

    H_\pm=\frac{a^2-\rho^2-(R\pm\varepsilon)^2}{2(R\pm\varepsilon)}.
\end{equation*}
Membership in $V_{\varepsilon,\theta}^{a}$ and $W_{\varepsilon,\theta}^{a}$ is given,
respectively, by $q_\theta<H_-$ and $q_\theta<H_+$.

Suppose that $x\in W_{\varepsilon,\theta}^{a}\setminus V_{\varepsilon,\theta}^{a}$
for some $\theta$. Subtracting the squared distances gives
\begin{equation*}
    4\varepsilon(R+q_\theta)
    =|z+(R+\varepsilon)u_\theta|^2-|z+(R-\varepsilon)u_\theta|^2<0.
\end{equation*}
Thus $p>R$. Also $\rho<a+R+\varepsilon$, and
\begin{equation}\label{eq:geom-threshold-width}
    0<H_+-H_-
    =\frac{\varepsilon(\rho^2-a^2-R^2+\varepsilon^2)}{R^2-\varepsilon^2}
    \le\left(\frac{4a}{R}+2\right)\varepsilon.
\end{equation}
The last bound uses $\varepsilon<R/2$ and $\rho<a+3R/2$.
The loss parameters satisfy $H_-\le q_\theta<H_+$, so the corresponding
interval for $\cos(\theta-\phi)$ has length at most
$(4a/R+2)\varepsilon/R$. Equation~\eqref{eq:geom-cosine} proves
\eqref{eq:geom-outer-loss}.

For the reverse difference, suppose that $a<R$ and
$x\in V_{\varepsilon,\theta}^{a}\setminus W_{\varepsilon,\theta}^{a}$.
Projection onto $E$ gives
\begin{equation*}
    p\ge R-\varepsilon-a>\frac{R-a}{2}.
\end{equation*}
In this case $H_+\le q_\theta<H_-$ and
\begin{equation*}
    0<H_--H_+
    =\frac{\varepsilon(a^2+R^2-\rho^2-\varepsilon^2)}{R^2-\varepsilon^2}
    \le\frac83\varepsilon.
\end{equation*}
After division by $p$, the cosine interval has length at most
$16\varepsilon/(3(R-a))$. Applying \eqref{eq:geom-cosine} proves
\eqref{eq:geom-inner-loss}. In both cases, a loss point must have $p>0$.

Finally, membership in either ball is one inequality in a cosine, or is
constant when $p=0$. Its indicator changes at most twice over one period.
The same variation bound holds on every shorter interval.

<!-- proposition-1: body_md -->
\label{prop:weighted-mean}
For each $\varepsilon$, let $P_\varepsilon$ be a finite-rank orthogonal projection in $H$,
fixed as $t$ varies.
Choose an orthonormal basis $e_{\varepsilon,1},\ldots,e_{\varepsilon,k_\varepsilon}$ of its range,
and define
\begin{equation}\label{eq:weighted-quantities}
    k_{P_\varepsilon}(x)=\sum_{j=1}^{k_\varepsilon}|Te_{\varepsilon,j}(x)|^2,\quad
    \ell_\varepsilon=\int_U q_\varepsilon(x)k_{P_\varepsilon}(x)\,\mathrm{d} x,\quad
    d_\varepsilon^2=\int_I\|(I-P_\varepsilon)c_{\varepsilon,t}\|^2\mathrm{d}\nu(t).
\end{equation}
Then
\begin{equation}\label{eq:weighted-mean-bound}
    \sqrt{\frac AB}
    \le\left(\int_I\langle Z_{\varepsilon,t}c_{\varepsilon,t},c_{\varepsilon,t}\rangle
       \,\mathrm{d}\nu(t)\right)^{1/2}
    \le\frac{\sqrt{\ell_\varepsilon}}{\sqrt A}+\sqrt B\,d_\varepsilon.
\end{equation}
In particular, conditions~(iii) and~(iv) may be replaced by the existence
of such projections with $\ell_\varepsilon\to0$ and $d_\varepsilon\to0$ as $\varepsilon\to0$.

<!-- proof-5: body_md -->
Write $c_{\varepsilon,t}=P_\varepsilon c_{\varepsilon,t}+(I-P_\varepsilon)c_{\varepsilon,t}$.
By Cauchy--Schwarz,
\begin{equation*}
    |TP_\varepsilon c_{\varepsilon,t}(x)|^2
    \le\|P_\varepsilon c_{\varepsilon,t}\|^2 k_{P_\varepsilon}(x)
    \le A^{-1}k_{P_\varepsilon}(x).
\end{equation*}
By Tonelli's theorem, the squared norm of
$\1_{L_{\varepsilon,t}}TP_\varepsilon c_{\varepsilon,t}$ in $L^2(I\times U,\nu\times\mathrm{d} x)$
is at most $\ell_\varepsilon/A$. The squared norm of the remaining part is at most
$Bd_\varepsilon^2$, since $Z_{\varepsilon,t}\le BI$.
The triangle inequality in this space proves the upper bound in
\eqref{eq:weighted-mean-bound}. Integrating the positive lower bound on
the loss energy proves the other inequality.

We first approximate $X_{\varepsilon,t}v$ for fixed $v$. Put
$\overline X_{\varepsilon,j}=T^*M_{p_{\varepsilon,j}}T$, where $M_p$ is multiplication by $p$.
These operators are the averages of $X_{\varepsilon,t}$ on the corresponding blocks,
so $AI\le\overline X_{\varepsilon,j}\le BI$.
For every fixed $v\in H$, Tonelli's theorem and
\eqref{eq:mean-window-partition} give
\begin{equation}\label{eq:mean-direct-estimate}
    \sum_{j=1}^{n}\int_{E_j}
    \|(X_{\varepsilon,t}-\overline X_{\varepsilon,j})v\|^2\mathrm{d}\nu(t)
    \le C\int_U|Tv(x)|^2
       \sum_{j=1}^{n}\int_{E_j}|\1_{V_{\varepsilon,t}}(x)-p_{\varepsilon,j}(x)|^2
       \mathrm{d}\nu(t)\,\mathrm{d} x
    \le C^2\eta^2\|v\|^2.
\end{equation}
Thus the family $\{X_{\varepsilon,t}v:t\in I\}$ has a mean approximation taking
finitely many values. Their number is bounded independently of $\varepsilon$
and uniformly for $v$ in a bounded ball.

We next pass from these fixed vectors to the coefficients involving $X_{\varepsilon,t}^{-1}$.
The vectors in the iteration depend on $t$, so this step needs a separate
argument. Omit $\varepsilon$, and set
$q=1-A/B$, $R_t=I-X_t/B$ and $\rho=\sqrt B/A$.
If $A=B$, then $X_t=AI$ and the coefficient family is constant.
Otherwise $0<q<1$. For $a=a_\varepsilon$, the iterates
$y_{0,t}=0$ and $y_{m+1,t}=a/B+R_ty_{m,t}$ satisfy
\begin{equation*}
    \|y_{m,t}\|\le\rho,\qquad
    \|X_t^{-1}a-y_{m,t}\|\le\rho q^m.
\end{equation*}
Given $h>0$ and fixed $z$ with $\|z\|\le\rho$,
\eqref{eq:mean-direct-estimate} gives a simple function $b_z(t)$ with
$\|R_tz-b_z(t)\|_{L^2(\nu)}\le h$.
The number of values is bounded in terms of $h$ and the fixed constants.
We take these values to be $(I-\overline X_{\varepsilon,j}/B)z$.
Their norms are at most $q\rho$.

Fix $\delta>0$. Starting with $z_{0,t}=0$, suppose that $z_{m,t}$ has
$L_m$ values, all of norm at most $\rho$. For each of these fixed values $z$,
choose $b_z$ with accuracy
\begin{equation*}
    h_m=\frac{(1-q)\delta}{2\sqrt{L_m}},
    \qquad z_{m+1,t}=a/B+b_{z_{m,t}}(t).
\end{equation*}
Each $z_{m+1,t}$ still has finitely many values of norm at most $\rho$.
At each fixed step, their number has a bound determined by $\delta$, the
fixed constants, and the bound from the preceding step. It is therefore
independent of $\varepsilon$.
If $e_m=\|y_m-z_m\|_{L^2(\nu)}$, then
\begin{equation*}
    e_{m+1}\le qe_m+\sqrt{L_m}\,h_m
    =qe_m+(1-q)\delta/2.
\end{equation*}
For each value of $z_m$, we bound the squared error on its level set by
the squared error for that fixed value over all of $I$.
Summing over the $L_m$ values gives the factor $\sqrt{L_m}$ above.
Thus $e_m\le\delta/2$.
Choosing $m$, independently of $\varepsilon$, with $\rho q^m\le\delta/2$ gives a mean $\delta$-approximation
to $c_{\varepsilon,t}$ by at most $L_\delta$ values, where $L_\delta$ is independent
of $\varepsilon$.

Let $P_\varepsilon$ project onto their span. Then $d_\varepsilon\le\delta$ and,
by condition~(iii), $\ell_\varepsilon\le CL_\delta\omega_\varepsilon$.
Proposition~\ref{prop:weighted-mean}, first with $\varepsilon\to0$ and then with
$\delta\to0$, gives a contradiction.

<!-- proposition-3: body_md -->
\label{prop:countable-balls}
Let $d\ge2$, $R>0$, and
\begin{equation}\label{eq:countable-ball-set}
    S=B(0,R)\cup\bigcup_{j=1}^{\infty}B(b_j,r_j),
    \qquad \sup_j|b_j|<\infty,\qquad
    0<r_j<R/10,\qquad \sum_{j=1}^{\infty}r_j<\infty.
\end{equation}
Suppose that, for some closed interval $I_0$ of positive length at most
$2\pi$ and some orthonormal $e_1,e_2$, the arc
$\Gamma=\{Ru_\theta:\theta\in I_0\}$ satisfies
\begin{equation}\label{eq:countable-exposed-arc}
    \gamma:=\operatorname{dist}\left(
    \Gamma,\overline{\bigcup_{j=1}^{\infty}B(b_j,r_j)}\right)>0,
    \qquad u_\theta=e_1\cos\theta+e_2\sin\theta.
\end{equation}
Then $L^2(S)$ admits no Riesz basis of exponentials.
The same conclusion holds for its invertible affine images, and for
$S\times K$ and their invertible affine images whenever
$K\subset\mathbb{R}^k$ is measurable and $0<|K|<\infty$.
<!-- end proposition-3 -->
<!-- proof-7: body_md -->
Choose $M$ such that $S\subset B(0,M)$, and set
$\varepsilon_0=\min\{R/10,\gamma/4\}$. For $0<\varepsilon<\varepsilon_0$, take
\begin{equation}\label{eq:countable-windows}
    \begin{aligned}
    V_{\varepsilon,\theta}&=S-(R-\varepsilon)u_\theta,&
    W_{\varepsilon,\theta}&=S-(R+\varepsilon)u_\theta,\\
    Q_\varepsilon&=B(0,\varepsilon/4),& U&=B(0,M+2R),
    \end{aligned}
    \qquad \mathrm{d}\nu(\theta)=\frac{\mathrm{d}\theta}{|I_0|}.
\end{equation}
These windows are jointly measurable and lie in $U$.
Assume that $S$ admits a Riesz basis of exponentials.
The synthesis operator from Section~\ref{sec:main-proof} gives condition~(i).
The estimates for $B(0,R)$ give
$Q_\varepsilon\subset B(-(R-\varepsilon)u_\theta,R)$ and exclude $Q_\varepsilon$ from
$B(-(R+\varepsilon)u_\theta,R)$.
For $x\in Q_\varepsilon$, the distance from $x+(R+\varepsilon)u_\theta$ to
$Ru_\theta$ is less than $5\varepsilon/4<\gamma$.
Thus it also lies outside every small ball. This proves condition~(ii).

We first bound the time spent in a small ball. If $0<r<R/10$ and
$9R/10\le\rho\le11R/10$, then, uniformly in $y\in\mathbb{R}^d$,
\begin{equation}\label{eq:small-ball-visitation}
    |\{\theta\in I_0:|\rho u_\theta-y|<r\}|
    \le 2\arcsin(r/\rho)\le\frac{10\pi r}{9R}.
\end{equation}
Indeed, put $p=|\operatorname{proj}_{\operatorname{span}\{e_1,e_2\}}y|$.
If $p=0$, the set is empty. Otherwise write the planar projection as $p u_\phi$.
The allowed directions satisfy
\begin{equation*}
    \cos(\theta-\phi)>
    \frac{\rho^2+|y|^2-r^2}{2\rho p}
    \ge\frac{\rho^2+p^2-r^2}{2\rho p}
    \ge\sqrt{1-r^2/\rho^2}.
\end{equation*}
This gives the first inequality in \eqref{eq:small-ball-visitation}.
The second follows from $\arcsin s\le\pi s/2$ for $0\le s\le1$.

Let $S_N=B(0,R)\cup\bigcup_{j=1}^{N}B(b_j,r_j)$, and let
$V^N_{\varepsilon,\theta},W^N_{\varepsilon,\theta}$ denote its corresponding translates.
Applying \eqref{eq:small-ball-visitation} to each remaining ball and summing gives
\begin{equation}\label{eq:countable-tail}
    \sup_{x\in U}\int_{I_0}
    \1_{V_{\varepsilon,\theta}\setminus V^N_{\varepsilon,\theta}}(x)\,\mathrm{d}\nu(\theta)
    \le\tau_N:=\frac{10\pi}{9R|I_0|}\sum_{j>N}r_j\longrightarrow0.
\end{equation}
The same bound holds with $W$ in place of $V$, uniformly in $\varepsilon$.
This one estimate controls both the loss and the mean approximation.
For the loss, use
\begin{equation*}
    W_{\varepsilon,\theta}\setminus V_{\varepsilon,\theta}
    \subset (W^N_{\varepsilon,\theta}\setminus V^N_{\varepsilon,\theta})
       \cup(W_{\varepsilon,\theta}\setminus W^N_{\varepsilon,\theta}),
\end{equation*}
Lemma~\ref{lem:geom-moving-balls} gives
$\omega_\varepsilon\le C_N\sqrt\varepsilon+\tau_N$, with $C_N$ independent of $\varepsilon$.
First letting $\varepsilon\to0$ and then $N\to\infty$ proves condition~(iii).

For the mean approximation, divide $I_0$ into
$m$ equal intervals and let $\mathcal A_m$ take the average on each interval.
For fixed $x$, write
$g(\theta)=\1_{V_{\varepsilon,\theta}}(x)$ and
$g_N(\theta)=\1_{V^N_{\varepsilon,\theta}}(x)$.
The finite union has variation at most $2(N+1)$, so
\begin{equation}\label{eq:countable-mean-approximation}
    \|g-\mathcal A_m g\|_{L^2(\nu)}
    \le\|g_N-\mathcal A_m g_N\|_{L^2(\nu)}
       +\|g-g_N\|_{L^2(\nu)}
    \le\sqrt{\frac{N+1}{2m}}+\sqrt{\tau_N}.
\end{equation}
Here $I-\mathcal A_m$ is an orthogonal projection, and the second term is
bounded by \eqref{eq:countable-tail}. Given $\eta>0$, first choose $N$ and
then $m$ so that the right-hand side is at most $\eta$.
Both choices are independent of $\varepsilon$ and $x$.
Proposition~\ref{prop:mean-window-criterion} now gives the contradiction.

The product and affine arguments from Section~\ref{sec:3} apply without change.
The product windows have indicators multiplied by $\1_K(y)$,
so \eqref{eq:mean-window-partition} also holds.

The small disks have pairwise disjoint closures and are separated from the
unit disk. Indeed, for $j<k$, the distance between their centres is at least
$\theta_j/4$, while $r_j+r_k\le3\theta_j/32$.
Their centres converge to $(5,0)$ and their radii have finite sum.
They stay a positive distance from the whole unit circle.
Thus Proposition~\ref{prop:countable-balls} applies.
This bounded open set has infinitely many connected components, so it is
not one of the finite unions in Theorem~\ref{thm:main}.

We show that the windows \eqref{eq:countable-windows}, with $I_0=[0,1]$,
have no variation bound independent of $\varepsilon$.
Put $g_j=3\cdot2^{-(j+2)}$ for $j\ge0$.
Given $M\ge1$, choose $0<\varepsilon<r_M/8$ and
$x\in B(x_0,r_M/8)$. For $1\le j\le M$,
\begin{equation*}
    |x+(1-\varepsilon)u_{\theta_j}-(x_0+u_{\theta_j})|
    \le |x-x_0|+\varepsilon<r_M/4<r_j.
\end{equation*}
Hence $x\in V_{\varepsilon,\theta_j}$. At $g_j$, where $0\le j\le M$,
write $\Delta=|g_j-\theta_k|$ for any $k\ge1$.
The definitions of $g_j$ and $\theta_k$ give
$\Delta\ge\theta_k/4$ and $\Delta\ge2^{-(M+2)}$.
Since $\Delta<1$, we have
$|u_{g_j}-u_{\theta_k}|\ge\Delta/2$ and
$r_k\le\theta_k/16\le\Delta/4$. It follows that
\begin{equation*}
    |x+(1-\varepsilon)u_{g_j}-(x_0+u_{\theta_k})|-r_k
    \ge\Delta/4-|x-x_0|-\varepsilon
    >2^{-(M+4)}-r_M/4>0.
\end{equation*}
Also $|x+(1-\varepsilon)u_{g_j}|\ge4-r_M/8-1>1$, so the unit disk does not
contribute at these times. Thus $x\notin V_{\varepsilon,g_j}$ for $0\le j\le M$.
Along the increasing sequence
\begin{equation*}
    g_M<\theta_M<g_{M-1}<\theta_{M-1}<\cdots<g_1<\theta_1<g_0,
\end{equation*}
the indicator of $V_{\varepsilon,\theta}$ alternates between zero and one.
The sum in \eqref{eq:main-turnover} is therefore $2M$.
This holds throughout the positive-measure disk
$B(x_0,r_M/8)$. Changing representatives on null sets cannot remove this failure.

Since $\sum_j r_j<\infty$, \eqref{eq:countable-mean-approximation} holds
uniformly in $\varepsilon$. Thus the mean condition is strictly weaker than
condition~(iv) for these windows.

<!-- remark-2: body_md -->
\label{rem:countable-original-criterion}
Example~\ref{ex:countable-disks} shows that the mean condition is weaker than
condition~(iv) for the specified windows on $[0,1]$.
The set itself can still be treated by Theorem~\ref{thm:abstract-windows}.
Indeed, restrict the parameter interval to $I_1=[3/4,1]$ and take
$\mathrm{d}\nu_1=4\,\mathrm{d}\theta$ there. For $j<k$ and $\theta,\phi\in I_1$,
projection onto $u_{(\theta+\phi)/2}$ gives
\begin{equation*}
    |u_{\theta_j}-u_{\theta_k}-(1-\varepsilon)(u_\theta-u_\phi)|
    \ge\sin(3/8)|u_{\theta_j}-u_{\theta_k}|>r_j+r_k.
\end{equation*}
Thus, for each fixed $\varepsilon$, different small disks cover disjoint regions
as the windows move over $I_1$. Each point meets at most one small disk.
Including the unit disk gives the variation bound $J=4$.
Conditions~(i) and~(ii) are unchanged, and the new overlap bound is at most
$4\omega_\varepsilon\to0$. A linear change of parameter from $I_1$ to $[0,1]$
then allows us to apply Theorem~\ref{thm:abstract-windows}.
