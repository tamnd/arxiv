\label{app-phase00p}
To calculate the phase, separate the evolution into the two
adiabatic intervals $(-t_0,-T)$ and $(T,t_0)$ and the nonadiabatic
interval $(-T,T)$ containing the crossing point of the $0$-th and
$n$-th adiabatic levels, where
\begin{equation}
1\ll T\ll\sqrt{x}.
\label{U00-T-condition}
\end{equation}
The positive adiabatic energy of the radial block is
\begin{equation}
E(t)
=
\left[
\left(4t^2+x-2R^2\right)^2
+(4tR)^2
+4(R')^2
\right]^{1/2}.
\label{U00-E}
\end{equation}
Averaging the rapidly oscillating terms gives
\begin{equation}
E(t)
=
E_{0;(0)}(t)
+
\frac{2\rho^2\sqrt{2x}}{E_{0;(0)}(t)}
+\ldots,
\label{U00-E-expansion}
\end{equation}
where
\begin{equation}
E_{0;(0)}(t)
\equiv
\sqrt{(4t^2)^2+8xt^2}
=
4|t|\sqrt{t^2+\frac{x}{2}}.
\label{U00-E0}
\end{equation}

The contribution of $E_{0;(0)}(t)$ to the phase accumulated in the two
adiabatic intervals is
\begin{eqnarray}
-2\int_T^{t_0}E_{0;(0)}(t)\,dt
=
-\frac{8t_0^3}{3}
-2xt_0
+\frac{2\sqrt{2}}{3}x^{3/2}
+2\sqrt{2x}\,T^2
+o(1).
\label{U00-E0-integral}
\end{eqnarray}
The correction proportional to $\rho^2$ gives
\begin{eqnarray}
-2\int_T^{\infty}
\frac{2\rho^2\sqrt{2x}}{E_{0;(0)}(t)}\,dt
=
-\rho^2
\left[
\ln2+\ln x-2\ln T
\right]+o(1).
\label{U00-rho-integral}
\end{eqnarray}

The asymptotic dynamical phases of the Landau--Zener solutions inside
the interval containing the avoideed crossing contribution
\begin{equation}
\phi_{\rm nad}
=
-2\sqrt{2x}\,T^2
-\rho^2
\left[
\ln(4\sqrt{2})
+\frac12\ln x
+2\ln T
\right].
\label{U00-nad-phase}
\end{equation}
With the phase convention of Eq.~(\ref{U00-LZ-H}), the phase of the
Landau--Zener transition amplitude is
\begin{equation}
\phi_{\rm LZ}
=
-\frac{3\pi}{4}
+{\rm arg}\Gamma(i\rho^2)
-\theta_1(x).
\label{U00-LZ-phase}
\end{equation}
Finally, the evolution along the two vertical parts of the deformed
integration path contributes the phase $4xt_0$. Hence,
\begin{eqnarray}
{\rm arg}\,U_{00}^{(n)}
=
4xt_0
-2\int_T^{t_0}E_{0;(0)}(t)\,dt
-2\int_T^\infty
\frac{2\rho^2\sqrt{2x}}{E_{0;(0)}(t)}\,dt
+\phi_{\rm nad}
+\phi_{\rm LZ}.
\label{U00-total-phase-before}
\end{eqnarray}
All terms that depend on $T$, $x^{3/2}$, and $\ln x$ cancel:
\begin{equation}
-\rho^2\ln x
-\frac{\rho^2}{2}\ln x
+\frac{3\rho^2}{2}\ln x
=0,
\label{U00-logx-cancel}
\end{equation}
where the last term follows from Eq.~(\ref{theta1p}). The remaining
logarithmic constant is
\begin{equation}
-\rho^2\ln2
-\rho^2\ln(4\sqrt{2})
=
-\frac{7\rho^2}{2}\ln2.
\label{U00-log2-result}
\end{equation}
Consequently,
\begin{eqnarray}
U_{00}^{(n)}
=
\sqrt{1-e^{-2\pi\rho^2}}
\times
\exp\left\{
i\left[
-\frac{8t_0^3}{3}
+2xt_0
-\frac{7\rho^2}{2}\ln2
-\frac{3\pi}{4}
+{\rm arg}\Gamma(i\rho^2)
-\phi_1
\right]
\right\}.
\label{U00-plus-final}
\end{eqnarray}
