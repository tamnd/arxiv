\label{rem:mass-conservation}
	The asymptotic solution constructed above respects mass conservation. To see this let $I_\mathrm{asy}$ and $\sigma_\mathrm{asy}$ solve \eqref{eq:nonlin_sys_0_1}. Integrating \eqref{eq:r_asy_eq_ode_0} over $-\infty<u<\infty$ and using \eqref{eq:nonlin_sys_0_1} for $\int_{-\infty}^0K(-u)r_\mathrm{asy}(u)du$ then yields {\small$R_\mathrm{asy} = \tfrac{I_\mathrm{asy}}{m}\left(\tfrac{m+wI_\mathrm{asy}}{m+\mathcal{R}_0 I_\mathrm{asy}} \mathcal{R}_0  - m - w\right)$}. With \eqref{eq:endemic-N-S} for $S_\mathrm{asy}$ it is then straightforward to show that $S_\mathrm{asy}+I_\mathrm{asy}+R_\mathrm{asy}=1$.

\label{rem:ode-for-rho}
    Since $K(z)=0$ for all $z\leq 0$ we can solve \eqref{eq:r_asy_eq_ode} explicitly for $u\geq 0 $ to get
    \begin{equation}\label{eq:r_asy_pos}
        r_\text{asy}(u) = \tfrac{1-m-w}{2 c_\mathrm{asy}}I_\text{asy}  e^{\frac{m}{c_\mathrm{asy}}u + \frac{m^2\sigma_\mathrm{asy}^2}{2c_\mathrm{asy}^2}}\mathrm{erfc}\left(\tfrac{u}{\sigma_\mathrm{asy}\sqrt{2}} + \tfrac{\sigma_\mathrm{asy} m}{c_\mathrm{asy}\sqrt{2}}   \right),\qquad u\geq 0.
    \end{equation}
    The remaining values of $r_\mathrm{asy}(u)$ for $u<0$ are then found by solving an initial-value-problem. Note that {\small$r_\text{asy}(0) = \tfrac{1-m-w}{2 \sqrt{\pi}c_\mathrm{asy}}I_\text{asy}  U\left(\tfrac{m^2\sigma^2}{2c_\mathrm{asy}^2} \right)$} where $U\left(z\right) := \sqrt{\pi}e^{z}\mathrm{erfc}(\sqrt{z})$. Thus
    \begin{subequations}\label{eq:r_asy_neg}
    \begin{equation}\label{eq:r_asy_rescale_0}
        r_\mathrm{asy}(u) = \frac{1-m-w}{2c_\mathrm{asy}\sqrt{\pi}}I_\mathrm{asy} U\left(\tfrac{m^2\sigma_\mathrm{asy}^2}{2c_\mathrm{asy}^2}\right) \rho(u),
    \end{equation}
    where $\rho(u)$ solves
    \begin{align}                     
        & \rho'(u) - \left(\tfrac{\mathcal{R}_0 I_\mathrm{asy}}{c_\mathrm{asy}}K(-u) + \tfrac{m}{c_\mathrm{asy}}\right)\rho(u) = -\tfrac{\sqrt{2}}{\sigma_\mathrm{asy}}\tfrac{\exp\left(-\tfrac{u^2}{2\sigma_\mathrm{asy}^2}\right)}{U\left(\tfrac{m^2\sigma_\mathrm{asy}^2}{2c_\mathrm{asy}^2}\right)}, & u< 0\label{eq:r_asy_rescale_1a}\\
        & \rho(0) = 1. \label{eq:r_asy_rescale_1b}
    \end{align}
    \end{subequations}
