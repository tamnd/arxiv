\label{thm:mishiba-coaction}
The following assertions hold.

\begin{enumerate}
\item
There exists an $R$-algebra homomorphism
\[
\widetilde{\Delta}_\infty^{\mathrm{Li}}:
\mathfrak{Z}_{\infty,R}
\longrightarrow
\mathfrak{Z}_{\infty,R}\otimes_R\overline{\mathfrak{Z}}_{\infty,R}
\]
such that, for every
$\mathfrak{s}\in\mathcal{I}$ with $\operatorname{dep}(\mathfrak{s})=r$,
\[
\widetilde{\Delta}_\infty^{\mathrm{Li}}
\left(
\mathrm{Li}_{\mathfrak{s}}(\mathbf{1})
\right)
=
\sum_{i=0}^r
\mathrm{Li}_{\mathfrak{s}[:i]}(\mathbf{1})
\otimes
\pi_\infty
(\mathrm{Li}_{\mathfrak{s}[i+1:]}(\mathbf{1})).
\]

\item The map $\widetilde{\Delta}_\infty^{\mathrm{Li}}$ is a coaction, and
$\overline{\mathfrak{Z}}_{\infty,R}$ carries a connected graded Hopf algebra
structure compatible with this coaction.

\item
The antipode
\[
S_\infty:
\overline{\mathfrak{Z}}_{\infty,R}
\longrightarrow
\overline{\mathfrak{Z}}_{\infty,R}
\]
is a non-trivial $R$-algebra involution satisfying
\[
S_\infty
\left(
\pi_\infty(\mathrm{Li}_{\mathfrak{s}}(\mathbf{1}))
\right)
=
\pi_\infty
\left(
\mathrm{Li}_{\mathfrak{s}}^{\star-\mathrm{inv}}(\mathbf{1})
\right)
\]
for every $\mathfrak{s}\in\mathcal{I}$. 
\end{enumerate}

from Condition \ref{cond:unital}. For non-empty $\mathfrak{s},\mathfrak{n}\in\mathcal{I}$, write
$
[\mathfrak{s}]=[s_1,\mathfrak{s}^-]$ and 
$[\mathfrak{n}]=[n_1,\mathfrak{n}^-]
$. 
Recall that
\begin{align*}
\mathfrak{s}\ast^\bullet\mathfrak{n}
={}&
[s_1,\mathfrak{s}^-\ast^\bullet\mathfrak{n}]+
[n_1,\mathfrak{s}\ast^\bullet\mathfrak{n}^-]+
[s_1+n_1,\mathfrak{s}^-\ast^\bullet\mathfrak{n}^-]
\label{eq:stuffle-third}\\&+
\varepsilon_\bullet
\sum_{1\leq j<s_1+n_1}
\Delta_{s_1,n_1}^{[j]}
[s_1+n_1-j,
[j]\ast^\zeta(\mathfrak{s}^-\ast^\zeta\mathfrak{n}^-)].
\end{align*}
Write
\[
\mathcal{R}(\mathfrak{s}^-)
=
\sum_{\mathfrak{s}^-=[A,B]}A\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B),
\qquad
\mathcal{R}(\mathfrak{n}^-)
=
\sum_{\mathfrak{n}^-=[C,D]}C\otimes\mathrm{ev}^{\star-\mathrm{inv}}(D).
\]
Then
\[
\mathcal{R}(\mathfrak{s})
=
1\otimes\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s})
+
\sum_{\mathfrak{s}^-=[A,B]}
[s_1,A]\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B),
\]
and similarly
\[
\mathcal{R}(\mathfrak{n})
=
1\otimes\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{n})
+
\sum_{\mathfrak{n}^-=[C,D]}
[n_1,C]\otimes\mathrm{ev}^{\star-\mathrm{inv}}(D).
\]
Hence,
$\mathcal{R}(\mathfrak{s})\star^\bullet\mathcal{R}(\mathfrak{n})$
consists of the term
\begin{equation}\label{eq:idconstant}
1\otimes \mathrm{ev}^{{\star-\mathrm{inv}}}(\mathfrak{s})\mathrm{ev}^{{\star-\mathrm{inv}}}(\mathfrak{n}),
\end{equation}
the two cross terms
\begin{equation}\label{eq:cross-s}
\sum_{\mathfrak{s}^-=[A,B]}
[s_1,A]\otimes
\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{n}),
\end{equation}
and
\begin{equation}\label{eq:cross-n}
\sum_{\mathfrak{n}^-=[C,D]}
[n_1,C]\otimes
\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s})\mathrm{ev}^{\star-\mathrm{inv}}(D),
\end{equation}
together with
\begin{equation}\label{eq:prefix-product}
\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
([s_1,A]\ast^\bullet[n_1,C])
\otimes
\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D).
\end{equation}

Applying the recursion to the first tensor factor in
\eqref{eq:prefix-product}, it becomes the sum of the following three terms:
\begin{align}
&\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
[s_1,A\ast^\bullet[n_1,C]]
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D),
\label{eq:prefix-first}\\
&\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
[n_1,[s_1,A]\ast^\bullet C]
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D),
\label{eq:prefix-second}\\
&\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
[s_1+n_1,A\ast^\bullet C]
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D),
\label{eq:prefix-third}
\end{align}
together with 
\begin{equation}\label{eq:prefix-correction}
\begin{aligned}
\varepsilon_\bullet
\sum_{1\leq j<s_1+n_1}
\Delta_{s_1,n_1}^{[j]}
\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
&[s_1+n_1-j,
[j]\ast^\zeta(A\ast^\zeta C)]\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D).
\end{aligned}
\end{equation}
Notice that the sum of \eqref{eq:cross-s} and \eqref{eq:prefix-first}
becomes
\[
\begin{aligned}
&\sum_{\mathfrak{s}^-=[A,B]}
[s_1,A]\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{n})+
\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
[s_1,A\ast^\bullet[n_1,C]]
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D)\\
&=
([s_1,-]\otimes\operatorname{id})
\left(
\mathcal{R}(\mathfrak{s}^-)\star^\bullet\mathcal{R}(\mathfrak{n})
\right)\\
&=
([s_1,-]\otimes\operatorname{id})
\mathcal{R}(\mathfrak{s}^-\ast^\bullet\mathfrak{n})\\
&=
\mathcal{R}
\left(
[s_1,\mathfrak{s}^-\ast^\bullet\mathfrak{n}]
\right)-1\otimes \mathrm{ev}^{{\star-\mathrm{inv}}}\left(
[s_1,\mathfrak{s}^-\ast^\bullet\mathfrak{n}]
\right),
\end{aligned}
\]
where the second equality follows from the induction hypothesis and
the last equality follows from \eqref{eq:R-left-letter}. 
Similarly, the sum of \eqref{eq:cross-n} and
\eqref{eq:prefix-second} gives
\[
\begin{aligned}
&\sum_{\mathfrak{n}^-=[C,D]}
[n_1,C]\otimes\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s})\mathrm{ev}^{\star-\mathrm{inv}}(D)+
\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
[n_1,[s_1,A]\ast^\bullet C]
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D)\\

([n_1,-]\otimes\operatorname{id})
\left(
\mathcal{R}(\mathfrak{s})\star^\bullet\mathcal{R}(\mathfrak{n}^-)
\right)\\
&=
([n_1,-]\otimes\operatorname{id})
\mathcal{R}(\mathfrak{s}\ast^\bullet\mathfrak{n}^-)\\
&=
\mathcal{R}
\left(
[n_1,\mathfrak{s}\ast^\bullet\mathfrak{n}^-]
\right)-1\otimes \mathrm{ev}^{{\star-\mathrm{inv}}}\left(
[n_1,\mathfrak{s}\ast^\bullet\mathfrak{n}^-]
\right).
\end{aligned}
\]
For \eqref{eq:prefix-third}, by \eqref{eq:R-left-letter} and the induction hypothesis, we have
\[
\begin{aligned}
&\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
[s_1+n_1,A\ast^\bullet C]
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D)\\

([s_1+n_1,-]\otimes\operatorname{id})
\left(
\mathcal{R}(\mathfrak{s}^-)\star^\bullet\mathcal{R}(\mathfrak{n}^-)
\right)\\

([s_1+n_1,-]\otimes\operatorname{id})

&=
\mathcal{R}
\left(
[s_1+n_1,\mathfrak{s}^-\ast^\bullet\mathfrak{n}^-]
\right)-1\otimes \mathrm{ev}^{{\star-\mathrm{inv}}}\left(
[s_1+n_1,\mathfrak{s}^-\ast^\bullet\mathfrak{n}^-]

\end{aligned}

=

[j]\otimes1.

\[
\begin{aligned}
&\mathcal{R}\left(

&=
\mathcal{R}([j])

\left(
[j]\ast^\zeta(A\ast^\zeta C)
\right)
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D).
\end{aligned}
\]

\eqref{eq:prefix-correction} becomes
\[
\begin{aligned}
&\sum_{1\leq j<s_1+n_1}
\Delta_{s_1,n_1}^{[j]}\sum_{\substack{\mathfrak{s}^-=[A,B]\\\mathfrak{n}^-=[C,D]}}
[s_1+n_1-j,
[j]\ast^\zeta(A\ast^\zeta C)]
\otimes\mathrm{ev}^{\star-\mathrm{inv}}(B)\mathrm{ev}^{\star-\mathrm{inv}}(D)\\
&=
\sum_{1\leq j<s_1+n_1}
\Delta_{s_1,n_1}^{[j]}([s_1+n_1-j,-]\otimes\operatorname{id})
\mathcal{R}\left(
[j]\ast^\zeta(\mathfrak{s}^-\ast^\zeta\mathfrak{n}^-)
\right)\\
&=

\Delta_{s_1,n_1}^{[j]}
[s_1+n_1-j,
[j]\ast^\zeta(\mathfrak{s}^-\ast^\zeta\mathfrak{n}^-)]
\right)\\
&\quad-1\otimes\mathrm{ev}^{{\star-\mathrm{inv}}}\left(\sum_{1\leq j<s_1+n_1}
\Delta_{s_1,n_1}^{[j]}
[s_1+n_1-j,
[j]\ast^\zeta(\mathfrak{s}^-\ast^\zeta\mathfrak{n}^-)]
\right),
\end{aligned}
\]
where the last equality follows from \eqref{eq:R-left-letter}.

\begin{align}\label{eq:difference}
\mathcal{R}(\mathfrak{s}\ast^\bullet\mathfrak{n})
-
\mathcal{R}(\mathfrak{s})\star^\bullet\mathcal{R}(\mathfrak{n})
=
1\otimes
\left(
\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s}\ast^\bullet\mathfrak{n})
-
\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s})\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{n})
\right).
\end{align}
It remains to show that the right-hand side of \eqref{eq:difference}
vanishes. Apply $m_B(\mathrm{ev}\otimes\operatorname{id})$ to both sides of
\eqref{eq:difference}. By Condition \ref{cond:convolution},
\[
m_B(\mathrm{ev}\otimes\operatorname{id})
\mathcal{R}(\mathfrak{s}\ast^\bullet\mathfrak{n})=0.
\]
On the other hand, by Condition \ref{cond:ev-product},
$m_B(\mathrm{ev}\otimes\operatorname{id})$ is multiplicative with respect to
$\star^\bullet$. Hence, Condition \ref{cond:convolution} implies that
\[
\begin{aligned}
m_B(\mathrm{ev}\otimes\operatorname{id})
\left(
\mathcal{R}(\mathfrak{s})\star^\bullet\mathcal{R}(\mathfrak{n})
\right)=
m_B(\mathrm{ev}\otimes\operatorname{id})\mathcal{R}(\mathfrak{s})\,
m_B(\mathrm{ev}\otimes\operatorname{id})\mathcal{R}(\mathfrak{n})
=0.
\end{aligned}
\]
Therefore, applying $m_B(\mathrm{ev}\otimes\operatorname{id})$ to \eqref{eq:difference}
gives Condition \ref{cond:ant-product}
\[
\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s}\ast^\bullet\mathfrak{n})
=
\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s})\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{n}).
\]
Substituting this into \eqref{eq:difference}, we obtain
\[
\mathcal{R}(\mathfrak{s}\ast^\bullet\mathfrak{n})
=
\mathcal{R}(\mathfrak{s})\star^\bullet\mathcal{R}(\mathfrak{n}),
\]
which completes the proof.

\label{lem:degree-convolution}
For every $s\in\mathbb{N}$ and $d\geq0$, fix $\mathscr{S}_d(s)\in B$, and fix $D\in\mathbb{N}\cup\{\infty\}$. Assume that the following sums are
well-defined in $B$. Define the $R$-linear maps $\mathrm{ev},\mathrm{ev}^{\star-\mathrm{inv}}$ by
$\mathrm{ev}(1)=\mathrm{ev}^{\star-\mathrm{inv}}(1)=1$ and, for every non-empty
$\mathfrak{s}=(s_1,\ldots,s_r)$, by
\begin{align*}
\mathrm{ev}([\mathfrak{s}])
&=\sum_{D>d_1>\cdots>d_r\geq0}
\prod_{k=1}^r\mathscr{S}_{d_k}(s_k),\\
\mathrm{ev}^{\star-\mathrm{inv}}([\mathfrak{s}])
&=(-1)^r\sum_{0\leq d_1\leq\cdots\leq d_r<D}
\prod_{k=1}^r\mathscr{S}_{d_k}(s_k).
\end{align*}
Then Conditions \ref{cond:unital} and \ref{cond:convolution} hold.

$\mathrm{ev}(1)=\mathrm{ev}^{\star-\mathrm{inv}}(1)=1$. Let $\mathfrak{s}=(s_1,\ldots,s_r)\in\mathcal{I}$ be non-empty. We first prove
\[
\sum_{i=0}^r

=0.

Set $H_0=H_{r+1}=0$,

\[

\prod_{k=1}^r\mathscr{S}_{d_k}(s_k).

\qquad

\begin{equation}\label{eq:degree-telescoping}
\mathrm{ev}(\mathfrak{s}[:i])\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s}[i+1:])

H_0-H_1
=
\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s})
=
\mathrm{ev}(\varnothing)\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s}),

H_r-H_{r+1}
=
\mathrm{ev}(\mathfrak{s})
=
\mathrm{ev}(\mathfrak{s})\mathrm{ev}^{\star-\mathrm{inv}}(\varnothing).
\]

\begin{aligned}
&\mathrm{ev}(\mathfrak{s}[:i])\mathrm{ev}^{\star-\mathrm{inv}}(\mathfrak{s}[i+1:])=
(-1)^{r-i}
\sum_{\substack{
D>d_1>\cdots>d_i\geq0\\

H_i-H_{i+1}.

\end{aligned}

\mathrm{ev}^{{\star-\mathrm{inv}}}(\mathfrak{s}[:i])\mathrm{ev}(\mathfrak{s}[i+1:])

\]
Thus, Condition \ref{cond:convolution} holds.

\cite[\S5.2]{Tha04}),
\[
\zeta_A(j)
=
\frac{\operatorname{BC}(j)}{\Gamma_{j+1}}
\widetilde{\pi}^{\,j},
\qquad
\zeta_A(q-1)
=
\frac{\widetilde{\pi}^{\,q-1}}{L_1},

$\Gamma_{j+1}$ is the $(j+1)$-st Carlitz factorial and $\widetilde{\pi}$ is a fixed Carlitz period (see \cite{Tha04}). 
Writing $j=m(q-1)$, we obtain
\[
\zeta_A(j)
=
\frac{\operatorname{BC}(j)L_1^m}{\Gamma_{j+1}}
\zeta_A(q-1)^m.
\]

\[
\frac{\operatorname{BC}(j)L_1^m}{\Gamma_{j+1}}
\in
\mathbb{F}_p(L_1)^\times.

Recall that
\[
\frac{z}{\exp_C(z)}
=
\sum_{n\geq0}
\frac{\operatorname{BC}(n)}{\Gamma_{n+1}}z^n
\]
and
\[
\exp_C(z)
=
\sum_{i\geq0}\frac{z^{q^i}}{D_i},

\label{rmk:F_p[L_1]doesnotwork}
The assumption that $R$ is an $\mathbb{F}_p(L_1)$-algebra in
\cref{thm:infty-coaction} cannot in general be replaced by the assumption

$R=\mathbb{F}_2[L_1]$. Expanding $\zeta_A(5)$ in the Thakur basis, we have
\[
\begin{aligned}
\zeta_A(5)
={}&
a\left(
\zeta_A(1,1,1,1,1)
+
\zeta_A(1,1,2,1)
\right)+
b\left(
\zeta_A(2,1,1,1)
+
\zeta_A(2,2,1)
\right),
\end{aligned}
\]
where
\[
a=L_1^8+L_1^5,
\qquad
b=L_1^8+L_1^6+L_1^5.
\]
If the conclusion of \cref{thm:infty-coaction} were valid over $R$, then,
since $(5)$ has depth one, the biweight $(2,3)$ component of
$\widetilde{\Delta}_\infty(\zeta_A(5))$ would be zero. On the other hand,
applying the coaction formula to the above expansion gives
\[
(a+L_1b)\,
\zeta_A(1,1)\otimes
\pi_\infty\left(
\zeta_A(1,1,1)+\zeta_A(2,1)
\right).
\]
Here, we use $\zeta_A(2)=L_1\zeta_A(1,1)$. Moreover, note that
\[
a+L_1b
=
L_1^5
\left(
L_1^4+L_1^3+L_1^2+L_1+1
\right),
\]
and
\[
\zeta_A(1)\zeta_A(1,1)
=
(L_1+1)
\left(
\zeta_A(1,1,1)+\zeta_A(2,1)
\right).
\]
Since $L_1+1\nmid a+L_1b$ in $\mathbb{F}_2[L_1]$, the above biweight $(2,3)$
component is nonzero, a contradiction. Thus,
\cref{thm:infty-coaction} does not in general hold over an arbitrary
$\mathbb{F}_p[L_1]$-subalgebra.

By the construction of $\Delta_\infty$ in
\cite[Theorem~1.3.4]{Mis26}, we have
\[
\Delta_\infty\circ\pi_\infty
=
(\pi_\infty\otimes\operatorname{id})\circ\widetilde{\Delta}_\infty.
\]
Hence \eqref{eq:quotient-zeta-coproduct} follows immediately from
\eqref{eq:infty-bullet-coaction} with $\bullet=\zeta$.

We prove \eqref{eq:antipode-zeta-to-ant} by induction on
$\operatorname{dep}(\mathfrak{s})$. The case $\mathfrak{s}=\varnothing$ is clear. Let
$\mathfrak{s}=(s_1,\ldots,s_r)\in\mathcal{I}$ be non-empty. By the antipode identity and

S_\infty

\pi_\infty(\zeta_A(\mathfrak{s}))

=
-\pi_\infty(\zeta_A(\mathfrak{s}))-

S_\infty
\left(
\pi_\infty(\zeta_A(\mathfrak{s}[i+1:]))
\right).
\end{aligned}

By the induction hypothesis, the right-hand side is
