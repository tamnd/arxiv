<!-- head-1: body_md -->
We use $\mathord{\color{black!33}\bullet}$ as a placeholder for the argument of a function.
We write $\|\mathord{\color{black!33}\bullet}\|$ for the Euclidean norm of a vector, and $\|\mathord{\color{black!33}\bullet}\|_{\mathrm{op}}$ for the matrix operator norm.
$\mathcal{P}(\mathbb{S}^{d-1})$ denotes the set of probability measures on $\mathbb{S}^{d-1}$, equipped with the Wasserstein-$1$ distance $W_1$ induced by the Euclidean distance $\|\mathord{\color{black!33}\bullet}\|$ on $\mathbb{R}^d$.
For a measurable map $h$, $h_{\sharp}\mu$ denotes the pushforward of a measure $\mu$ under $h$.
For a square matrix $G$, we denote its adjugate by $\mathrm{adj}(G)$.

<!-- assumption-1: body_md -->
\label{asm:well-posedness}
There exists a constant $L_{\mathcal{F}} \geq 0$ such that, for every $\varrho, \widetilde{\varrho} \in \mathcal{P}(\mathbb{S}^{d-1})$,
\begin{equation*}
    \left\| \mathcal{F}[\varrho] - \mathcal{F}[\widetilde{\varrho}] \right\|_{\mathrm{op}} \leq L_{\mathcal{F}} W_1 (\varrho, \widetilde{\varrho}) \, .
\end{equation*}
<!-- end assumption-1 -->
<!-- lemma-1: body_md -->
\label{lem:vf_property}
Suppose that Assumption \ref{asm:well-posedness} holds.
The vector field $b$ defined in \eqref{eq:vf} satisfies the following properties:
\begin{itemize}
    \item For every $x \in \mathbb{S}^{d-1}$ and $\varrho \in \mathcal{P}(\mathbb{S}^{d-1})$, 
    \[
    \|b(x, \varrho)\| \leq C_{\mathcal{F}} \, .
    \]

    \item For every $x, \widetilde{x} \in \mathbb{S}^{d-1}$ and every $\varrho, \widetilde{\varrho} \in \mathcal{P}(\mathbb{S}^{d-1})$, 
    \begin{equation*}
    \left\| b(x, \varrho) - b (\widetilde{x}, \widetilde{\varrho}) \right\| \leq 3 C_{\mathcal{F}} \|x - \widetilde{x}\| + L_{\mathcal{F}} W_1 (\varrho, \widetilde{\varrho}) \, .
    \end{equation*}
\end{itemize}
<!-- end lemma-1 -->
<!-- theorem-1: body_md -->
\label{thm:well-posedness_finite_sys}
Suppose that Assumption \ref{asm:well-posedness} holds.
Then, for every $n \in \mathbb{N}$, the system \eqref{eq:MP_oja_flow} admits a unique global solution $(x_1, x_2, \dots, x_n) \in C^1 ([0, \infty); (\mathbb{S}^{d-1})^n)$ for any initial configuration in $(\mathbb{S}^{d-1})^n$.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Fix $n \in \mathbb{N}$.
For a particle configuration $\mathbf{x} := \big(x_1, \dots, x_n\big)^{\top} \in (\mathbb{S}^{d-1})^{n}$, define the associated empirical distribution by $\rho_n^{\mathbf{x}} = \frac{1}{n} \sum_{k=1}^n \delta_{x_k}$.
For each $k=1,2,\dots, n$, set
\begin{equation*}
B_k (\mathbf{x}) := b \big(x_k, \rho_n^{\mathbf{x}} \big),
\end{equation*}
and define the full vector field
\begin{equation*}
    \mathbf{B} (\mathbf{x}) := \left( B_1 (\mathbf{x}), \dots, B_n (\mathbf{x})\right)^{\top} \in \mathbb{R}^{nd} \, .
\end{equation*}
The finite-particle system \eqref{eq:MP_oja_flow} can then be written as
\begin{equation}\label{eq:MP_oja_flow_rewrite}
\dot{\mathbf{x}}(t) = \mathbf{B} (\mathbf{x}(t) ) \, .
\end{equation}

We equip $(\mathbb{S}^{d-1})^n$ with the metric
\begin{equation*}
    d_{\infty} (\mathbf{x}, \widetilde{\mathbf{x}}) := \max_{1 \leq k \leq n} \|x_k - \widetilde{x}_k\| \, .
\end{equation*}
For any two particle configurations $\mathbf{x}, \widetilde{\mathbf{x}} \in (\mathbb{S}^{d-1})^n$, the coupling that pairs $x_k$ with $\widetilde{x}_k$ gives
\begin{equation*}
    W_1 (\rho_n^{\mathbf{x}}, \rho_n^{\widetilde{\mathbf{x}}}) \leq \frac{1}{n} \sum_{k=1}^n \|x_k - \widetilde{x}_k\| \leq d_{\infty} (\mathbf{x}, \widetilde{\mathbf{x}}) \, .
\end{equation*}
By Lemma \ref{lem:vf_property}, for every $k=1,\ldots, n$,
\begin{equation*}
    \|B_k(\mathbf{x})\| = \|b(x_k, \rho_n^{\mathbf{x}})\| \leq C_{\mathcal{F}},
\end{equation*}
and
\begin{equation*}
\|B_k (\mathbf{x}) - B_k (\widetilde{\mathbf{x}})\| = \|b(x_k, \rho_n^{\mathbf{x}}) - b(\widetilde{x}_k, \rho_n^{\widetilde{\mathbf{x}}})\| \leq 3C_{\mathcal{F}} \|x_k - \widetilde{x}_k\| + L_{\mathcal{F}} W_1 (\rho_n^{\mathbf{x}}, \rho_n^{\widetilde{\mathbf{x}}}) \leq \left(3C_{\mathcal{F}} + L_{\mathcal{F}} \right) d_{\infty} (\mathbf{x}, \widetilde{\mathbf{x}}) \, .
\end{equation*}
Moreover, a straightforward calculation shows that $x_k^{\top} B_k (\mathbf{x}) = 0$, and hence $\mathbf{B}(\mathbf{x})$ is tangent to $(\mathbb{S}^{d-1})^n$ at $\mathbf{x}$.
Therefore, the vector field $\mathbf{B}$ of the system \eqref{eq:MP_oja_flow_rewrite} is uniformly bounded, globally Lipschitz, and tangent to the product manifold $(\mathbb{S}^{d-1})^n$.

Then, standard existence and uniqueness results for ODEs on manifolds imply that, for every initial configuration in $(\mathbb{S}^{d-1})^n$, \eqref{eq:MP_oja_flow_rewrite} admits a unique maximal solution.
Finally, since $(\mathbb{S}^{d-1})^n$ is compact, the maximal solution cannot escape the state space in finite time.
It therefore extends to all $t \geq 0$, which completes the proof.

<!-- theorem-2: body_md -->
\label{thm:well-posedness_mf}
Suppose that Assumption \ref{asm:well-posedness} holds.
Then, for every initial distribution $\rho_0 \in \mathcal{P}(\mathbb{S}^{d-1})$, there exists a unique weak solution $\rho \in C ([0, \infty); \mathcal{P}(\mathbb{S}^{d-1}))$ to the mean-field equation \eqref{eq:mf_oja_flow}.
Moreover, the solution admits the characteristic representation
\begin{equation}\label{eq:pushforward_repre}
\rho(t) = (\Psi_t)_{\sharp} \rho_0,
\end{equation}
where $\Psi_t: \mathbb{S}^{d-1} \to \mathbb{S}^{d-1} $ is the characteristic flow associated with $\rho$, determined by
\begin{equation}\label{eq:charac_flow}
\frac{d}{dt}\Psi_t (x) = b (\Psi_t(x), \rho(t)), \qquad \Psi_0(x) = x \, .
\end{equation}

The solution is also stable with respect to the initial data in the Wasserstein-$1$ distance.
More specifically, let $\rho$ and $\widetilde{\rho}$ be the solutions corresponding to initial data $\rho_0$ and $\widetilde{\rho}_0$, respectively. Then, for every $t \geq 0$,
\begin{equation}\label{eq:ws_stability}
W_1 (\rho(t), \widetilde{\rho}(t)) \leq e^{(3C_{\mathcal{F}} + L_{\mathcal{F}})t} W_1 (\rho_0, \widetilde{\rho}_0) \, .
\end{equation}

<!-- proof-3: body_md -->
We first establish the existence and uniqueness of a weak solution.
The argument is divided into two steps.

\paragraph{Step 1: Construction of a solution by a fixed-point argument.} 
Fix $T > 0$.
Given a prescribed measure-valued curve $\mu \in C([0,T]; \mathcal{P}(\mathbb{S}^{d-1}))$ satisfying $\mu(0) = \rho_0$,
consider the characteristic equation
\begin{equation}\label{eq:chara_flow_in_proof}
    \dot{\Psi}^{\mu}_t (x) = b (\Psi^{\mu}_t (x), \mu(t)), \qquad \Psi^{\mu}_0(x) = x,
\end{equation}
where $b$ is given in \eqref{eq:vf}.
Since $t \mapsto \mu(t)$ is continuous with respect to $W_1$ and $\mathcal{F}$ is Lipschitz continuous under Assumption \ref{asm:well-posedness}, the map $t \mapsto \mathcal{F}[\mu(t)]$ is continuous.
Together with Lemma \ref{lem:vf_property}, this implies that the velocity field $(t, x) \mapsto b(x, \mu(t))$ is continuous in time, globally Lipschitz in space, and uniformly bounded.
Moreover, the velocity field is tangent to $\mathbb{S}^{d-1}$.
Standard existence and uniqueness theory for ODEs on manifolds therefore implies that \eqref{eq:chara_flow_in_proof} admits a unique solution $\Psi^{\mu}_t(x)$ on $[0,T]$; see, for example, \cite{hartman2002ordinary}.

Define the map $\mathcal{T} : C([0,T]; \mathcal{P}(\mathbb{S}^{d-1})) \to C ([0,T]; \mathcal{P}(\mathbb{S}^{d-1}))$ by
\begin{equation*}
    (\mathcal{T} \mu)(t) := (\Psi^{\mu}_t)_{\sharp} \rho_0 \, .
\end{equation*}
Then a fixed point $\mu = \mathcal{T} \mu$ satisfies
\begin{equation*}
    \mu(t) = (\Psi^{\mu}_t)_{\sharp} \rho_0,
\end{equation*}
and is therefore transported by the characteristic flow generated by itself.
We now show that $\mathcal{T}$ is a contraction with respect to a suitable metric.

Let $\mu, \widetilde{\mu} \in C([0,T]; \mathcal{P}(\mathbb{S}^{d-1}))$.
For a fixed $x \in \mathbb{S}^{d-1}$, set
\begin{equation*}
    X(t, x) := \Psi^{\mu}_t(x), \qquad \widetilde{X}(t, x) := \Psi^{\widetilde{\mu}}_t(x) \, .
\end{equation*}
Since $X(0, x) = \widetilde{X}(0, x) = x$, Lemma \ref{lem:vf_property} gives
\begin{equation*}
\big\|X(t, x) - \widetilde{X}(t, x) \big\| \leq 3 C_{\mathcal{F}} \int_0^t \big\|X(s, x) - \widetilde{X}(s, x) \big\| ds + L_{\mathcal{F}} \int_0^t W_1 (\mu(s), \widetilde{\mu}(s))ds \, .
\end{equation*}
By Grönwall's inequality,
\begin{equation*}
    \big\| X(t, x) - \widetilde{X}(t, x) \big\| \leq L_{\mathcal{F}} \int_0^t e^{3C_{\mathcal{F}}(t-s)} W_1 (\mu(s), \widetilde{\mu}(s)) ds \, .
\end{equation*}
Since $(\Psi^{\mu}_t, \Psi^{\widetilde{\mu}}_t)_{\sharp} \rho_0$ is a coupling between $(\mathcal{T} \mu)(t)$ and $(\mathcal{T} \widetilde{\mu})(t)$, it follows that
\begin{equation}\label{eq:contraction_proof_aux}
W_1 ( (\mathcal{T} \mu)(t), (\mathcal{T} \widetilde{\mu}) (t) ) \leq L_{\mathcal{F}} \int_0^t e^{3C_{\mathcal{F}} (t-s)} W_1 (\mu(s), \widetilde{\mu}(s)) ds \, .
\end{equation}
For $\lambda > 0$, introduce the weighted metric
\begin{equation*}
d_{\lambda} (\mu, \widetilde{\mu}) := \sup_{0\leq t \leq T} e^{-\lambda t} W_1 (\mu(t), \widetilde{\mu}(t)) \, .
\end{equation*}
Multiplying the preceding estimate \eqref{eq:contraction_proof_aux} by $e^{-\lambda t}$ and using the definition of $d_{\lambda}$, we obtain, for $\lambda > 3C_{\mathcal{F}}$,
\begin{equation*}
e^{-\lambda t} W_1 ( (\mathcal{T} \mu)(t), (\mathcal{T} \widetilde{\mu}) (t) ) \leq L_{\mathcal{F}} d_{\lambda} (\mu, \widetilde{\mu}) \int_0^t e^{-(\lambda - 3C_{\mathcal{F}})(t-s)} ds \leq \frac{L_{\mathcal{F}}}{\lambda - 3 C_{\mathcal{F}}} d_{\lambda} (\mu, \widetilde{\mu})\, .
\end{equation*}
Taking the supremum over $t \in [0,T]$ gives
\begin{equation*}
d_{\lambda} (\mathcal{T} \mu, \mathcal{T} \widetilde{\mu}) \leq \frac{L_{\mathcal{F}}}{\lambda - 3 C_{\mathcal{F}}} d_{\lambda} (\mu, \widetilde{\mu}) \, .
\end{equation*}
Choosing $\lambda > 3 C_{\mathcal{F}} + L_{\mathcal{F}}$ makes $\mathcal{T}$ a strict contraction.

Because $\mathcal{P}(\mathbb{S}^{d-1})$ is complete under $W_1$, the space $C([0,T]; \mathcal{P}(\mathbb{S}^{d-1}))$ is complete under the weighted metric $d_{\lambda}$.
The Banach fixed-point theorem \cite{banach1922sur} therefore gives a unique fixed point $\rho \in C([0,T]; \mathcal{P}(\mathbb{S}^{d-1}))$, satisfying 
\begin{equation*}
\rho(t) = (\Psi_t^{\rho})_{\sharp} \rho_0 \, .
\end{equation*}
Since $T > 0$ is arbitrary and, by uniqueness, solutions constructed on different time intervals agree on their overlaps, they define a global characteristic solution $\rho \in C([0,\infty); \mathcal{P}(\mathbb{S}^{d-1}))$.

\paragraph{Step 2: Verification of the weak formulation.}
Next, we verify that the fixed point $\rho$ constructed in Step $1$ is the unique weak solution to the mean-field equation \eqref{eq:mf_oja_flow}.

For every test function $\varphi \in C^1 (\mathbb{S}^{d-1})$ and every $t \geq 0$, the definition of the pushforward measure gives
\begin{equation*}
\int_{\mathbb{S}^{d-1}} \varphi (x) d\rho(t, x) = \int_{\mathbb{S}^{d-1}} \varphi (\Psi^{\rho}_t(x)) d\rho_0(x) \, .
\end{equation*}
Differentiating this identity with respect to $t$ and using the characteristic equation \eqref{eq:chara_flow_in_proof}, we obtain
\begin{equation}\label{eq:verify_weak_form}
\frac{d}{dt} \int_{\mathbb{S}^{d-1}} \varphi(x) d\rho(t,x) = \int_{\mathbb{S}^{d-1}} \nabla_{\mathbb{S}^{d-1}} \varphi( \Psi^{\rho}_t(x))^{\top} b (\Psi^{\rho}_t(x), \rho(t)) d\rho_0(x) = \int_{\mathbb{S}^{d-1}} \nabla_{\mathbb{S}^{d-1}} \varphi (x)^{\top} b(x, \rho(t)) d\rho(t, x) \, .  
\end{equation}
Integrating the above identity over $[0,t]$ shows that $\rho$ satisfies the weak formulation of  \eqref{eq:mf_oja_flow} for every $t \geq 0$.
Hence, $\rho$ is a weak solution of the mean-field equation.

It remains to prove uniqueness.
Let $\bar{\rho}$ be any weak solution to \eqref{eq:mf_oja_flow} with initial condition $\rho_0$, and let $\Psi^{\bar{\rho}}$ denote the characteristic flow \eqref{eq:chara_flow_in_proof}  generated by the curve $\bar{\rho}$.
Once $\bar{\rho}$ is fixed, the map $(t,x) \mapsto b(x, \bar{\rho}(t))$ can be regarded as a given time-dependent velocity field.
By Lemma \ref{lem:vf_property}, this velocity field is continuous in time and globally Lipschitz in space.
The classical characteristic representation for continuity equations with Lipschitz velocity fields therefore implies that $\bar{\rho}(t) = (\Psi^{\bar{\rho}}_t)_{\sharp} \rho_0$; see, for example, \cite{ambrosio2014continuity}.
Therefore, $\bar{\rho} = \mathcal{T} \bar{\rho}$, which shows that every weak solution of \eqref{eq:mf_oja_flow} is a fixed point of $\mathcal{T}$.
By the uniqueness of the fixed point established in Step $1$, we conclude that $\bar{\rho} = \rho$, which proves uniqueness of the weak solution.

\paragraph{Stability Estimate.}
Lastly, we perform the stability estimate of the solution with respect to the initial data.
Let $\rho$ and $\widetilde{\rho}$ be solutions corresponding to initial data $\rho_0$ and $\widetilde{\rho}_0$, respectively.
Let $\pi_0$ be an optimal coupling of $\rho_0$ and $\widetilde{\rho}_0$.
For $(x, y) \in \mathbb{S}^{d-1} \times \mathbb{S}^{d-1}$, define
\begin{equation*}
X(t, x) := \Psi^{\rho}_t (x), \qquad \widetilde{X} (t, y) := \Psi^{\widetilde{\rho}}_t (y),
\end{equation*}
and set
\begin{equation*}
    D(t) := \int_{\mathbb{S}^{d-1} \times \mathbb{S}^{d-1}} \left\| X(t,x) - \widetilde{X}(t, y) \right\| d\pi_0 (x, y) \, .
\end{equation*}
Since $(X(t, \mathord{\color{black!33}\bullet}), \widetilde{X}(t, \mathord{\color{black!33}\bullet}))_{\sharp} \pi_0$ is a coupling between $\rho(t)$ and $\widetilde{\rho}(t)$, we have $W_1 (\rho(t), \widetilde{\rho}(t)) \leq D(t)$.
Together with Lemma \ref{lem:vf_property}, this implies that
\begin{equation*}
D(t) \leq D(0) + 3C_{\mathcal{F}} \int_0^t D(s) ds + L_{\mathcal{F}} \int_0^t W_1 (\rho(s), \widetilde{\rho}(s))ds \leq D(0) + (3 C_{\mathcal{F}} + L_{\mathcal{F}}) \int_0^t D(s) ds \, .
\end{equation*}
Grönwall's inequality then gives
\begin{equation*}
    D(t) \leq e^{(3C_{\mathcal{F}} + L_{\mathcal{F}}) t} D(0) \, .
\end{equation*}
Since $\pi_0$ is an optimal coupling, we have $D(0) = W_1 (\rho_0, \widetilde{\rho}_0)$.
Combining this identity with the preceding estimate gives the Wasserstein stability bound \eqref{eq:ws_stability}.

\label{thm:PI_mf_sys}
Suppose that Assumption \ref{asm:well-posedness} and the tracelessness condition \eqref{eq:traceless_asm} hold.
Let $\rho(t)$ be the unique solution to \eqref{eq:mf_oja_flow} with initial condition $\rho(0) = \rho_0 \in \mathcal{P}(\mathbb{S}^{d-1})$.

    \rho(t) = (\Phi_{G(t)})_{\sharp} \rho_0,
\end{equation}
where $G: [0,\infty) \to \mathrm{SL}^{\pm}(d,\mathbb{R})$ solves the closed matrix-valued equation
\begin{equation}\label{eq:mf_govern_dyn}
\dot{G}(t) = \mathcal{F}[(\Phi_{G(t)})_{\sharp} \rho_0] G(t), \qquad G(0) = I_{d} \, .
\end{equation}

\label{rem:existing_finite_sys_result}
The normalized linear transformation representation of the finite-particle dynamics was established in \cite{lohe2020on,lohe2025exactA,lohe2025exactB}.
Theorem \ref{thm:PI_mf_sys} recovers the earlier results when specialized to empirical measures.
 
In particular, let $\rho_n(0) = \frac{1}{n} \sum_{k=1}^n \delta_{x_{k, 0}}$ be the initial empirical distribution, and let $\rho_n(t)$ denote the empirical distribution associated with the solution of the finite-particle system \eqref{eq:MP_oja_flow}.
By the consistency between the finite-particle and mean-field descriptions established in Section \ref{subsec:well-posedness_mf}, Theorem \ref{thm:PI_mf_sys} applies to $\rho_n$ and gives
\begin{equation*}
    \rho_n(t) = (\Phi_{G_n(t)})_{\sharp} \rho_n(0) = \frac{1}{n} \sum_{k=1}^n \delta_{\Phi_{G_n(t)} (x_{k, 0})},
\end{equation*}
where $G_n(t)$ satisfies the matrix-valued ODE
\begin{equation}\label{eq:finite_sys_gov_ode}
\dot{G}_n(t) = \mathcal{F} [(\Phi_{G_n(t)})_{\sharp} \rho_n(0)] G_n(t), \qquad G_n(0) = I_d \, .
\end{equation}
It follows that the unique solution of the finite-particle system \eqref{eq:MP_oja_flow} with initial configuration $x_k(0) = x_{k, 0}$ admits the representation
\begin{equation}\label{eq:PI_finite_sys}
    x_k(t) = \Phi_{G_n(t)} (x_{k, 0}), \qquad k=1,2, \dots, n \, .
\end{equation}
Thus, as shown in \cite{lohe2020on,lohe2025exactA,lohe2025exactB}, the finite-particle dynamics \eqref{eq:MP_oja_flow} are fully determined by the initial configuration and a single ODE whose dimension is independent of the number of particles $n$.
However, it is worth emphasizing that the representation \eqref{eq:PI_finite_sys}, together with equation \eqref{eq:finite_sys_gov_ode}, does not necessarily lead to a reduction in computational cost compared to the original finite-particle system \eqref{eq:MP_oja_flow}.
Indeed, evaluating the right-hand side of \eqref{eq:finite_sys_gov_ode} still requires aggregating information over all particles, so the computational cost generally remains dependent on $n$.

<!-- remark-3: body_md -->
\label{rem:connect_to_WS}
The representation of the multi-particle Oja flow presented in Theorem \ref{thm:PI_mf_sys} closely parallels the Watanabe--Strogatz description of the Kuramoto model \cite{watanabe1994constants,marvel2009identical,Lohe_2018,Lipton_2021}.
In that setting, the Kuramoto dynamics are shown to admit a representation through a time-dependent Möbius transformation acting on the initial data, with the transformation governed by finitely many variables.
In the present setting, the analogous representation is given by the normalized linear transformation $\Phi_{G(t)}$, with $G(t)$ evolving according to the matrix equation \eqref{eq:mf_govern_dyn}.

This type of representation is commonly referred to as \emph{partial integrability}.
More specifically, for fixed initial data, the evolution of the system is encoded by a single time-dependent transformation involving only finitely many collective variables \cite{watanabe1993integrability,watanabe1994constants,pikovsky2008partially,Lohe_2018,lohe2025exactA,park2022watanabe}.

<!-- proof-4: body_md -->
Let $\rho$ be the unique global weak solution to \eqref{eq:mf_oja_flow} with initial condition $\rho_0$, and consider the matrix-valued ODE
\begin{equation}\label{eq:Gt_ODE_in_proof}
\dot{G}(t) = \mathcal{F} \left[ \rho(t) \right] G(t), \qquad G(0) = I_d \, .
\end{equation}
Since $\rho \in C([0, \infty); \mathcal{P}(\mathbb{S}^{d-1}))$ and $\mathcal{F}$ is Lipschitz continuous under Assumption \ref{asm:well-posedness}, the map $t \mapsto \mathcal{F}[\rho(t)]$ is continuous.
The equation \eqref{eq:Gt_ODE_in_proof} for $G$ is therefore a linear matrix-valued ODE with continuous coefficients.
The standard existence and uniqueness theory for linear systems \cite{teschl2012ordinary} shows that it admits a unique global solution $G \in C^1 ([0, \infty); \mathbb{R}^{d\times d})$.

We next show that $G(t)$ remains in $\mathrm{SL}^{\pm}(d,\mathbb{R})$.
Suppressing the explicit dependence on $t$ for notational simplicity, Jacobi's formula in adjugate form and \eqref{eq:Gt_ODE_in_proof} give
\begin{equation*}
\frac{d}{dt} \det G = \mathrm{tr}(\mathrm{adj}(G) \dot{G}) = \mathrm{tr}(\mathrm{adj}(G) \mathcal{F}[\rho] G) = \mathrm{tr}(\mathcal{F}[\rho] G \, \mathrm{adj}(G)) = \mathrm{tr}(\mathcal{F}[\rho]) \det G = 0 \, .
\end{equation*}
Here, we used the identity $G\, \mathrm{adj}(G) = (\det G) I_d$ and the tracelessness condition \eqref{eq:traceless_asm}.
Consequently, $\det G(t) = \det G(0) = 1$ for every $t \geq 0$, and therefore $G(t) \in \mathrm{SL}^{\pm}(d,\mathbb{R})$ for all $t \geq 0$.

It remains to establish the representation \eqref{eq:rho_as_pushforward}.
Fix $x \in \mathbb{S}^{d-1}$.
Since $G(t)$ is invertible, the normalized linear transformation $\Phi_{G(t)}(x) = \frac{G(t)x}{\|G(t)x\|}$ is well-defined.
Suppressing again the explicit dependence on $t$, differentiating with respect to time gives
\begin{equation}\label{eq:infi_motion}
\begin{aligned}
\frac{d}{dt} \Phi_{G}(x) &= \frac{d}{dt} \frac{Gx}{\|Gx\|} = \frac{\dot{G} x}{\|G x\|} - \frac{G x}{\|G x\|^3} \left\langle \dot{G} x, G x \right\rangle = \frac{\mathcal{F}[\rho] G x}{\|G x\|} - \frac{G  x}{\|G x\|^3} \left\langle \mathcal{F}[\rho] G x, G x \right\rangle\\[4pt]
&= \mathcal{F}[\rho] \Phi_G(x) - \Phi_G(x)\left\langle \mathcal{F}[\rho] \Phi_G(x), \Phi_G(x) \right\rangle = \left( I_d - \Phi_G(x) \Phi_G(x)^{\top} \right) \mathcal{F}[\rho] \Phi_G(x) = b(\Phi_G(x), \rho) \, .
\end{aligned}
\end{equation}
Moreover, $\Phi_{G(0)} (x) = \Phi_{I_d} (x) = x$.
Therefore, the family of maps $\Phi_{G(t)}$ satisfies the characteristic equation associated with $\rho$.
By uniqueness of the characteristic flow, $\Phi_{G(t)} = \Psi_t$, where $\Psi_t$ denotes the characteristic flow from Theorem \ref{thm:well-posedness_mf}.
The characteristic representation of the mean-field solution therefore gives
\begin{equation*}
    \rho(t) = (\Psi_t)_{\sharp} \rho_0 = (\Phi_{G(t)})_{\sharp} \rho_0 \, .
\end{equation*}
This proves \eqref{eq:rho_as_pushforward}.
Substituting this representation into \eqref{eq:Gt_ODE_in_proof} gives the closed matrix-valued ODE \eqref{eq:mf_govern_dyn}.

<!-- definition-1: body_md -->
Let $h: \mathcal{X} \to \mathcal{X}$ be a measurable transformation, and let $P \subseteq \mathcal{P}(\mathcal{X})$ be a family of probability distributions.
We say that $P$ is \emph{closed under} $h$ if $h_{\sharp} p \in P$ for every $p \in P$.
Equivalently, whenever $X$ is a random variable with $\mathrm{Law}(X) \in P$, one has $\mathrm{Law}(h(X)) \in P$.

By Lemma \ref{lem:closure_implies_inv}, it suffices to show that $P_Q$ is closed under every normalized linear transformation $\Phi_G$.

Let $p = \Pi_{\sharp} q \in P_Q$ for some $q \in Q$, and fix $G \in \mathrm{GL}(d, \mathbb{R})$.
Using the composition rule for pushforward measures and the identity $\Phi_G \circ \Pi = \Pi \circ \Lambda_G$ on $\mathbb{R}^d \setminus \{0\}$, we obtain

\end{equation*}
Since $Q$ is closed under $\Lambda_G$, we have $(\Lambda_G)_{\sharp} q \in Q$.
It follows that $(\Phi_G)_{\sharp} p \in P_Q$.
This shows that $P_Q$ is closed under normalized linear transformations.

<!-- remark-4: body_md -->
The closure property under linear transformations extends naturally to finite mixtures of elliptical distributions.
As a simple illustration, consider a two-component Gaussian mixture 
\begin{equation*}
    Y \sim \alpha \mathcal{N} (\mu_1, \Sigma_1) + (1-\alpha) \mathcal{N} (\mu_2, \Sigma_2), \qquad \alpha \in [0,1] \, .
\end{equation*}
Then, for any $G \in \mathrm{GL}(d, \mathbb{R})$,
\begin{equation*}
   \Lambda_G(Y) \sim \alpha \mathcal{N} (G\mu_1, G\Sigma_1 G^{\top}) + (1-\alpha) \mathcal{N} (G\mu_2, G\Sigma_2 G^{\top}) \, .
\end{equation*}
Thus, the family of two-component Gaussian mixtures is closed under linear transformations, with both the number of mixture components and their associated weights preserved.
More generally, the same observation applies to finite mixtures of any family of distributions that is itself closed under linear transformations.

<!-- proposition-2: body_md -->
\label{prop:stationary_states}
Suppose that the assumptions of Theorem \ref{thm:ACG_invariant} hold and, in addition, that the interaction kernel $\mathcal{F}$ is symmetric. 
That is, $\mathcal{F}[\varrho] = \mathcal{F}[\varrho]^{\top}$ for every $\varrho \in \mathcal{P}(\mathbb{S}^{d-1})$. 
Then the set of stationary distributions of the original mean-field equation \eqref{eq:mf_oja_flow} is 
\begin{equation}\label{eq:stat_state_mf}
\mathcal{S} := \left\{ \rho_* \in \mathcal{P}(\mathbb{S}^{d-1}) : (I_d - x x^{\top}) \mathcal{F}[\rho_*] x = 0, \;\; \rho_*\text{-a.e.} \right\} \, .
\end{equation}

\label{rem:connect_to_OA}
The invariance of the angular central Gaussian family under the multi-particle Oja flow is analogous to the Ott--Antonsen ansatz for the Kuramoto model \cite{ott2008low,ott2009long}.
In particular, in dimension $d=2$, the wrapped Cauchy family plays the analogous role for the classical Kuramoto model \cite{ott2008low,ott2009long,marvel2009identical}, forming the well-known Ott--Antonsen invariant manifold.
For the higher-dimensional Kuramoto model on the unit sphere, the associated invariant family is the spherical Cauchy family \cite{chandra2019complexity,Lipton_2021,kato2020some}.

A further parallel arises from the transformation structure underlying these families. 
The spherical Cauchy family can be obtained by pushing forward the uniform distribution on the unit sphere through Möbius transformations \cite{chandra2019complexity,Lipton_2021,kato2020some}.
Analogously, the angular central Gaussian family can be obtained by pushing forward the uniform distribution through normalized linear transformations, as shown in Proposition \ref{prop:M_Unif_ACG_manifold}.

The present setting, however, reveals a broader structural picture.
In Section \ref{subsubsec:construction_inv_param_fam}, we develop a systematic approach for identifying invariant parametric families for the multi-particle Oja flow by exploiting closure under normalized linear transformations.
This leads to a broad class of invariant parametric families, of which the angular central Gaussian family is one example.

We note that, on the circle, other parametric families closed under Möbius transformations have also been identified, such as the Kato--Jones family \cite{kato2010family,jacimovic2025learning}. 
These examples, together with the parallel discussed above, naturally raise the question of whether there is a systematic approach for identifying parametric families that are closed under Möbius transformations and hence invariant under the corresponding Kuramoto dynamics.

We first characterize the stationary distributions of the original mean-field equation \eqref{eq:mf_oja_flow}.
Let $\rho_*$ be a stationary distribution of \eqref{eq:mf_oja_flow}.
Then, for every test function $\varphi \in C^1(\mathbb{S}^{d-1})$, 
\begin{equation*}
\int_{\mathbb{S}^{d-1}} \nabla_{\mathbb{S}^{d-1}} \varphi (x)^{\top} \left( I_d - x x^{\top} \right) \mathcal{F}[\rho_*] x d \rho_*(x) = 0 \, .
\end{equation*}
Choose $\varphi(x) = \frac{1}{2} x^{\top} \mathcal{F}[\rho_*] x$.
Since $\mathcal{F}[\rho_*]$ is symmetric, $\nabla_{\mathbb{S}^{d-1}} \varphi(x) = \left( I_d - x x^{\top} \right)\mathcal{F}[\rho_*] x$.
Therefore,
\begin{equation*}
\int_{\mathbb{S}^{d-1}} \left\| \left( I_d - x x^{\top} \right) \mathcal{F}[\rho_*] x  \right\|^2 d\rho_*(x) = 0 \, .
\end{equation*}
Since the integrand is nonnegative, it follows that
\begin{equation}\label{eq:stat_cond_mf}
    \left( I_d - x x^{\top} \right) \mathcal{F}[\rho_*] x = 0 \qquad \rho_*\text{-a.e.} \, .
\end{equation}
Conversely, if a probability measure $\rho_*$ satisfies \eqref{eq:stat_cond_mf}, then the velocity field $b$, as defined in \eqref{eq:vf}, vanishes $\rho_*$-almost everywhere. 
Hence $\rho_*$ is stationary.
Therefore, the set of stationary distributions of the mean-field equation \eqref{eq:mf_oja_flow} is precisely $\mathcal{S}$, as defined in \eqref{eq:stat_state_mf}.

We next characterize the stationary distributions within the angular central Gaussian family.
Consider the mean-field equation \eqref{eq:mf_oja_flow} with initial data $\rho_0 = p_{\Sigma_0}$ for some $\Sigma_0 \in \mathrm{SPD}_1(d)$.
By Theorem \ref{thm:ACG_invariant}, the corresponding solution remains within the ACG family and satisfies $\rho(t) = p_{\Sigma(t)}$, where $\Sigma(t)$ solves the matrix equation \eqref{eq:mf_goven_dyn_acg}.

Now suppose that $\rho_* = p_{\Sigma_*}$ is a stationary distribution within the ACG family.
For notational simplicity, set $\mathcal{A}_* := \mathcal{F}[p_{\Sigma_*}]$.
Since $\mathcal{F}$ is symmetric, the stationary condition for $\Sigma_*$ becomes 
\begin{equation}\label{eq:stat_cond_Sigma_origin}
0 = \dot{\Sigma}_* = \mathcal{A}_* \Sigma_* + \Sigma_* \mathcal{A}_* \, .
\end{equation}
Since $\Sigma_*$ is symmetric positive definite, it admits the eigendecomposition
\[
\Sigma_* = Q D Q^{\top},
\]
where $Q$ is orthogonal and $D = \mathrm{diag} (\lambda_1, \lambda_2, \dots, \lambda_d)$ with $\lambda_i > 0$.
Define
\[
\widetilde{\mathcal{A}}_* := Q^{\top} \mathcal{A}_* Q \, .
\]
Multiplying \eqref{eq:stat_cond_Sigma_origin} from the left by $Q^{\top}$ and from the right by $Q$, we obtain
\begin{equation*}
    \widetilde{\mathcal{A}}_* D + D \widetilde{\mathcal{A}}_* = 0 \, .
\end{equation*}
Hence, for every $i, j = 1,\dots, d$,
\begin{equation*}
    \left( \lambda_i + \lambda_j \right) (\widetilde{\mathcal{A}}_{*})_{ij} = 0 \, .
\end{equation*}
Since $\lambda_i + \lambda_j > 0$, we obtain $(\widetilde{\mathcal{A}}_*)_{ij} = 0$ for all $i, j$.
Therefore, $\widetilde{\mathcal{A}}_* = 0$, and consequently
\begin{equation*}
    \mathcal{A}_* = \mathcal{F}[p_{\Sigma_*}] = 0 \, .
\end{equation*}
Conversely, if $\mathcal{F}[p_{\Sigma_*}] = 0$, then the vector field $b$ vanishes $p_{\Sigma_*}$-almost everywhere.

Therefore, the set of stationary distributions of \eqref{eq:mf_oja_flow} within the ACG family is precisely $\mathcal{S}_{\mathrm{ACG}}$, as defined in \eqref{eq:stat_state_ACG}.

Finally, the inclusion $\mathcal{S}_{\mathrm{ACG}} \subseteq \mathcal{S}$ follows immediately from the definitions of the two sets. 
This completes the proof.

For any $i,j = 1,\dots,d$, 
\begin{equation*}
    m_i - m_j = \frac{1}{2} \int_0^{\infty} \frac{1}{\sqrt{\prod_{k=1}^d (1 + s\lambda_k)}} \frac{\lambda_i - \lambda_j}{(1 + s\lambda_i) (1 + s\lambda_j)} ds \, .
\end{equation*}
Since $\lambda_i, \lambda_j > 0$, we know that $m_i - m_j$ has the same sign as $\lambda_i - \lambda_j$.
Moreover, $m_i = m_j$ if and only if $\lambda_i = \lambda_j$.

Now suppose that $M(\Sigma_*) = \frac{1}{d} I_d$.
Then
\[
m_i = \frac{1}{d}, \qquad i = 1, \dots, d \, .
\]
Thus $m_i = m_j$ for all $i,j$, and the previous observation implies that 
\[
\lambda_1 = \lambda_2 = \cdots = \lambda_d =: \lambda_* \, .
\]
Since $\Sigma_* \in \mathrm{SPD}_1(d)$, we have
\[
1 = \det(\Sigma_*) = \prod_{i=1}^d \lambda_i = (\lambda_*)^d \, .
\]
Because $\lambda_* > 0$, it follows that $\lambda_* = 1$.
Therefore, $\Lambda_* = I_d$, and hence $\Sigma_* = Q I_d Q^{\top} = I_d$.
This proves that $M(\Sigma_*) = \frac{1}{d} I_d$ implies $\Sigma_* = I_d$.

Conversely, suppose that $\Sigma_* = I_d$.
Then, by the integral representation of $M$ in \eqref{eq:ACG_second_moment_main},
\[
M(I_d) = \left(\frac{1}{2} \int_0^{\infty} (1 + s)^{-\frac{d}{2} - 1} ds\right) I_d = \frac{1}{d} I_d \, .
\]
Therefore, 
\begin{equation*}
\left\{ \mathrm{ACG}(\Sigma_*): M(\Sigma_*) = \frac{1}{d} I_d, \Sigma_* \in \mathrm{SPD}_1(d) \right\} = \left\{ \mathrm{ACG}(I_d) \right\} \, .
\end{equation*}
This completes the proof.

<!-- proof-14: body_md -->
Fix $\varrho \in \mathcal{P}(\mathbb{S}^2)$.
For notational simplicity, we suppress the dependence on $\varrho$ and write $M := M_{\varrho}$ and $K:= K_{\varrho}$.

Since $M$ is real and symmetric, there exists an orthogonal matrix $Q \in \mathbb{R}^{3 \times 3}$ with $\det Q = 1$ such that
\begin{equation*}
    M = Q D Q^{\top}, \qquad D:= \mathrm{diag}(\lambda_1, \lambda_2, \lambda_3),
\end{equation*}
where $\lambda_1,\lambda_2, \lambda_3$ are the eigenvalues of $M$.

Define the rotated random vectors
\begin{equation*}
\widetilde{X} = Q^{\top} X, \qquad \widetilde{Y} = Q^{\top} Y \, .
\end{equation*}
Since $X$ and $Y$ are independent and identically distributed, so are $\widetilde{X}$ and $\widetilde{Y}$.
Moreover,
\begin{equation}\label{eq:WX_WY_second_moment}
    \mathbb{E} \left[ \widetilde{X} \widetilde{X}^{\top} \right] = \mathbb{E} \left[ \widetilde{Y} \widetilde{Y}^{\top} \right] = \mathbb{E} \left[ Q^{\top} X X^{\top} Q \right] = Q^{\top} M Q = D \, .
\end{equation}
Set 
\begin{equation*}
    \widetilde{K} := \mathbb{E} \left[ (\widetilde{X} \times \widetilde{Y}) (\widetilde{X} \times \widetilde{Y})^{\top} \right] \, .
\end{equation*}
We first compute $\widetilde{K} \in \mathbb{R}^{3 \times 3}$.
With a slight abuse of notation, we write
\begin{equation*}
    \widetilde{X} = \left( X_1, X_2, X_3 \right)^{\top}, \qquad \widetilde{Y} = (Y_1, Y_2, Y_3)^{\top} \, .
\end{equation*}
By \eqref{eq:WX_WY_second_moment}, we have
\begin{equation*}
\mathbb{E} [X_i X_j] = \mathbb{E} [Y_i Y_j] = \delta_{ij} \lambda_i, \qquad i, j = 1,2,3 \, .
\end{equation*}

Recall that
\begin{equation*}
 \widetilde{X} \times \widetilde{Y} = \left( \begin{array}{c}
    X_2 Y_3 - X_3 Y_2   \\
    X_3 Y_1 - X_1 Y_3 \\
    X_1 Y_2 - X_2 Y_1
 \end{array} \right) \, .
\end{equation*}
For the first diagonal entry of $\widetilde{K}$, independence of $\widetilde{X}$ and $\widetilde{Y}$ gives
\begin{equation*}
\widetilde{K}_{11} = \mathbb{E} \left[ (X_2 Y_3 - X_3 Y_2)^2 \right] = \mathbb{E} [X_2^2] \mathbb{E} [Y_3^2] + \mathbb{E} [X_3^2] \mathbb{E} [Y_2^2] - 2 \mathbb{E} [X_2 X_3] \mathbb{E} [Y_2 Y_3] = 2 \lambda_2 \lambda_3 \, .
\end{equation*}
Similarly,
\[
\widetilde{K}_{22} = 2\lambda_1 \lambda_3, \qquad \widetilde{K}_{33} = 2\lambda_1 \lambda_2 \, .
\]
We can further show that the off-diagonal entries vanish.
For example,
\[
\widetilde{K}_{12} = \mathbb{E} \left[ (X_2 Y_3 - X_3 Y_2) (X_3 Y_1 - X_1 Y_3) \right] \, .
\]
After expanding, each term contains an off-diagonal second moment of either $\widetilde{X}$ or $\widetilde{Y}$, which vanishes because their second moment matrix is diagonal.
The same argument applies to the remaining off-diagonal entries.
Therefore,
\begin{equation*}
    \widetilde{K} = 2 \left( \begin{array}{ccc}
        \lambda_2 \lambda_3 & 0 & 0  \\
        0 & \lambda_1 \lambda_3 & 0 \\
        0 & 0 & \lambda_1 \lambda_2
    \end{array} \right) = 2 \mathrm{adj} (D) \, .
\end{equation*}

Since $Q$ is an orthogonal matrix satisfying $\det (Q) = 1$, the cross product is equivariant under $Q$:
\begin{equation*}
    (Q^{\top} X) \times (Q^{\top} Y) = Q^{\top} (X \times Y) \, .
\end{equation*}
Therefore,
\begin{equation*}
\widetilde{K} = \mathbb{E} \left[ (\widetilde{X} \times \widetilde{Y}) (\widetilde{X} \times \widetilde{Y})^{\top} \right] = Q^{\top} \mathbb{E} \left[ (X \times Y) (X \times Y)^{\top} \right] Q = Q^{\top} K Q \, .
\end{equation*}
It follows that
\begin{equation*}
K = Q \widetilde{K} Q^{\top} = 2 Q \mathrm{adj} (D)  Q^{\top} \, .
\end{equation*}
For an arbitrary $3 \times 3$ matrix $A$, the Cayley--Hamilton theorem \cite{horn2012matrix} gives
\begin{equation*}
    \mathrm{adj} (A) = A^2 - \mathrm{tr}(A) A + \frac{1}{2} \left( (\mathrm{tr}(A))^2 - \mathrm{tr}(A^2) \right) I_3 \, .
\end{equation*}
Applying this identity to $Q D Q^{\top}$ gives
\begin{equation*}
\begin{aligned}
\mathrm{adj} (Q D Q^{\top})
&= Q D^2 Q^{\top} - Q \left( \mathrm{tr}(D) D \right) Q^{\top} + \frac{1}{2} \left( (\mathrm{tr}(D))^2 - \mathrm{tr}(D^2) \right) I_3\\
&= Q \left( D^2 - \mathrm{tr}(D) D + \frac{1}{2} \left( (\mathrm{tr}(D))^2 - \mathrm{tr}(D^2) \right) I_3 \right) Q^{\top} = Q \mathrm{adj} (D) Q^{\top} \, .
\end{aligned}
\end{equation*}
Since $M = Q D Q^{\top}$, we conclude that
\begin{equation*}
    K = 2 \mathrm{adj} (M) \, .
\end{equation*}
Finally, because $\varrho$ is supported on $\mathbb{S}^2$,
\begin{equation*}
    \mathrm{tr}(M) = \mathbb{E} \left[ \mathrm{tr}(X X^{\top}) \right] = \mathbb{E} [\|X\|^2] = 1 \, .
\end{equation*}
Hence,
\[
K = 2M^2 - 2 M + \left( 1 - \mathrm{tr}(M^2) \right) I_3 \, .
\]
Taking the trace gives
\begin{equation*}
\mathrm{tr}(K) = 2\mathrm{tr}(M^2) - 2\mathrm{tr}(M) + (1 - \mathrm{tr}(M^2)) \mathrm{tr}(I_3) = 1 - \mathrm{tr}(M^2) \, .
\end{equation*}
This completes the proof.
