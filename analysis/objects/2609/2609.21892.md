\label{thm:fg.equs}
The numbers $F_\gamma^\pm(u)$ satisfy the following system of algebraic equations:

<!-- proposition-1: body_md -->
\label{prop:reflection}
	Faddeev's modular quantum dilogarithm satisfies
	\[\Phi_{\gamma,m+1,n+1}(z;\tau)\Phi_{\gamma,-m,-n}(-z;\tau) = \mu_{\gamma}^{-2}(-1)^{m+n}\mathbf{e}(Q_{\gamma,m,n}(z,\tau))\,,\]

	\[Q_{\gamma,m,n}(z,\tau)=-\frac{1}{2}\Big[\frac{c z^2}{c\tau+d} + \Big(2m+1 - \frac{2n+1}{c\tau+d}\Big) z + B_2(m+1)\tau - B_2(n+1)\frac{a\tau+b}{c\tau+d}\Big]\]
	and $B_2(x)=x^2-x+1/6$ is the 2nd Bernoulli polynomial.

<!-- proposition-2: body_md -->
\label{prop:prod.id.mod,fad}
\emph{(i)}\cite[Thm. 2.1]{GKZ} For $m,n\in\mathbb{Z}$ and $\gamma=(\begin{smallmatrix}a&b\\c&d\end{smallmatrix})\in\operatorname{SL}_2(\mathbb{Z})$ with $c\ne 0$, we have
    \begin{equation} \label{eq:modulartofaddeevC}
    \Phi_{\gamma,m,n}(z;\tau) = \prod_{j=0}^{c-1}\Phi_{S,0,0}(z+(m+j)\tau+k_j;c\tau+d)\,,\qquad k_j=\Big\lfloor \frac{d(m+j)-n}{c}\Big\rfloor\,.
    \end{equation}
\emph{(ii)} For $m,n\in\mathbb{Z}$ and $\gamma=\prod_{j=1}^{r}(\begin{smallmatrix}b_j&-1\\1&0\end{smallmatrix})$, where $b_j\ge2$, define $m_r = m$, $m_0 = n$, and choose $m_1,\dots,m_{r-1}$ to be arbitrary integers. Let $\tau_r = \tau$, $z_r = z$ and define $\tau_{i},z_i$ for $i=1,\dots,r-1$ via $\tau_{j-1} = b_j - \frac{1}{\tau_j}$ and $z_{j-1} = \frac{z_j}{\tau_j}$. Then

    \Phi_{\gamma,m,n}(z;\tau) = \prod_{j=1}^{r}\Phi_{S,0,0}\big(z_j+m_j\tau_j-m_{j-1};\,\tau_j\big)\,.
    \end{equation}

<!-- theorem-4: body_md -->
\label{thm:fn.equs}
The numbers $F_N^{\pm}(u)$ satisfy the algebraic equations:
\begin{align}
    F_N^\pm(0)&=\sqrt{\frac{N+2\pm\sqrt{N^2+4N}}{2}}\,,\label{eq:Fpm.zero}\\
    F_N^{+}(u)F_N^{-}(-u)
    &=
    (-1)^u\mathbf{e}\Big(-\frac{u^2}{2N}\Big)\,,\label{eq:Fpm.reflection}\\
    \frac{1}{\sqrt{N}}\sum_{x\in\mathbb{Z}/N\mathbb{Z}}\mathbf{e}\Big(\frac{xu}{N}\Big)F_{N}^+(x)
    &=
    \mathbf{e}\Big(\frac{N-1}{24}\Big)\frac{1}{F_{N}^{-}(u)}\,,\label{eq:Fpm.fourier}\\
    \frac{1}{\sqrt{N}}\sum_{x\in\mathbb{Z}/N\mathbb{Z}}\mathbf{e}\Big(\frac{xu}{N}\Big)\frac{F_{N}^+(x)}{F_{N}^-(x+v)}
    &=
    \frac{F_N^+(u+v)}{F_{N}^-(u)F_{N}^-(v)}\,,\qquad (u,v)\ne (0,0)\,.\label{eq:Fpm.5term}
\end{align}
<!-- end theorem-4 -->
<!-- proof-3: body_md -->
Equation~\eqref{eq:Fpm.zero} follows from the modularity of the $\eta$-function.
Equation~\eqref{eq:Fpm.reflection} follows from the specialisation of Equation~\eqref{eq:PhiS.reflection} (essentially the modularity of the $\theta$-function).
To prove Equation~\eqref{eq:Fpm.fourier}, we use the residue theorem and Equation~\eqref{eq:PhiS.fourier}.
Note that $q=\widetilde{q}$ and suppose that $q\mathbf{e}(w/\varepsilon)=\mathbf{e}(w)$ and for simplicity that $0<\varepsilon<1$ (a similar argument works when $\varepsilon>1$).
Then $w-1=u(1-\varepsilon)/N$ for some $u\in\mathbb{Z}$ and we have the following:
\[
\begin{aligned}
    &\frac{\varepsilon}{1-\varepsilon}\sum_{x=-N}^{-1}\mathbf{e}\Big(\frac{xu}{N}\Big)\Phi_{S,0,0}\Big(\frac{x}{N}(1-\varepsilon);\varepsilon\Big)\\
    &=
    2\pi i\sum_{x=-N}^{-1}\underset{z=\frac{x}{N}(1-\varepsilon)}{\mathrm{Res}}\mathbf{e}\Big(\frac{z(w-1)}{\varepsilon}\Big)\frac{\Phi_{S,0,0}(z;\varepsilon)}{1-\mathbf{e}(z(\varepsilon-1)/\varepsilon)}\;dz\\
    &=
    \bigg(\int_{i\mathbb{R}-0}-\int_{\varepsilon-1+i\mathbb{R}-0}\bigg)\mathbf{e}\Big(\frac{zw}{\varepsilon}\Big)\frac{\Phi_{S,0,0}(z;\varepsilon)}{\mathbf{e}(z/\varepsilon)-\mathbf{e}(z)}\;dz\\
    &=
    \int_{i\mathbb{R}-0}\mathbf{e}\Big(\frac{zw}{\varepsilon}\Big)\frac{\Phi_{S,0,0}(z;\varepsilon)}{\mathbf{e}(z/\varepsilon)-\mathbf{e}(z)}\;dz
    -\int_{\varepsilon-1+i\mathbb{R}-0}\mathbf{e}\Big(\frac{zw}{\varepsilon}+\frac{w(1-\varepsilon)}{\varepsilon}\Big)\frac{\Phi_{S,0,0}(z;\varepsilon)}{\widetilde{q}^{-1}\mathbf{e}(z/\varepsilon)-q^{-1}\mathbf{e}(z)}\;dz\\
    &=
    \int_{i\mathbb{R}-0}\mathbf{e}\Big(\frac{zw}{\varepsilon}\Big)\frac{\Phi_{S,0,0}(z;\varepsilon)-\Phi_{S,0,0}(z+\varepsilon-1;\varepsilon)}{\mathbf{e}(z/\varepsilon)-\mathbf{e}(z)}\;dz\\
    &=
    \int_{i\mathbb{R}-0}\Phi_{S,1,0}(z;\varepsilon)\;\mathbf{e}\Big(\frac{zw}{\varepsilon}\Big)\frac{(1-\mathbf{e}(z))-(1-\mathbf{e}(z/\varepsilon))}{\mathbf{e}(z/\varepsilon)-\mathbf{e}(z)}\;dz\\
    &=\int_{i\mathbb{R}-0}\Phi_{S,1,0}(z;\varepsilon)\;\mathbf{e}\Big(\frac{zw}{\varepsilon}\Big)\;dz\\
    &=\mathbf{e}(1/8)\sqrt{\varepsilon}\mathbf{e}\Big(\frac{-N-2}{24}\Big)\Phi_{S,0,0}\Big(\frac{u}{N}(1-\varepsilon);\varepsilon\Big)^{-1}\,.
\end{aligned}
\]
(To compute the residue we need to assume not just that $0<w$ but that $0<w-\varepsilon$, which implies that $w-1>\varepsilon-1$.)
Noting that for $0<\varepsilon<1$ and $\ell\in\mathbb{Z}_{\geq0}$, we have $\Phi_{S,0,0}(\ell(1-\varepsilon);\varepsilon)\mathbf{e}(\frac{N-1}{24})=\sqrt{\varepsilon}=F_N^-(0)$ and $\Phi_{S,0,0}(\varepsilon-1;\varepsilon)\mathbf{e}(\frac{N-1}{24})=\sqrt{1/\varepsilon}=F_N^+(0)$.
This therefore implies Equation~\eqref{eq:Fpm.fourier}.

A similar trick also allows for a proof of the 5-term identity using Equation~\eqref{eq:PhiS.5term}. (Again we will assume $0<\varepsilon<1$ with the other case following from a similar argument.)
We suppose that $v\neq0$, since the identity for $v=0$ follows from elementary considerations.
Let $v,u$ be represented in $(-N,0]$ with $y=v(1-\varepsilon)/N$ and $w-1=u(1-\varepsilon)/N$.
This implies that $\varepsilon<w$, $y+w-1<0$, and $-y-1<-\varepsilon$.

\bigg(\int_\mathcal{C}-\int_{\mathcal{C}+\varepsilon-1}\bigg)

<!-- definition-1: body_md -->
A metric group is a finite abelian group~$G$ together with a choice of a function $\langle\cdot\rangle\colon G\to \mathbb{C}^{\times}$ such that $\langlex\rangle=\langle-x\rangle$ for all $x$ and 
    \[\langlex;y\rangle := \frac{\langlex+y\rangle}{\langlex\rangle\langley\rangle}\]
is a non-degenerate bicharacter on~$G$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
A u-operator is a group homomorphism $\mathbf{f}$ from $G$ to the group of unitary operators on $L^2(G)$. We denote it by $x\mapsto \mathbf{f}^x$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
Two u-operators $\mathbf{p}$ and $\mathbf{q}$ form a Heisenberg pair of level $\ell=\ell(\mathbf{p},\mathbf{q})$, if for all $x,y\in G$ they satisfy	the relation
	\[\mathbf{p}^x\mathbf{q}^y = \langlex;y\rangle{}^{\ell}\mathbf{q}^y\mathbf{p}^x\,.\]
<!-- end definition-3 -->
<!-- definition-4: body_md -->
Let $G$ be a metric group and assume $|G|>1$. A \emph{finite quantum dilogarithm} for a metric group~$G$ is a function $E\colon G\to\mathbb{C}$ satisfying
    \begin{equation} \label{eq:pentagonmatrix}
    \widehat{E}(\mathbf{p})\widehat{E}(\mathbf{q}) - \widehat{E}(\mathbf{q})\widehat{E}(\mathbf{p}+\mathbf{q})\widehat{E}(\mathbf{p}) = C \delta(\mathbf{q})\delta(\mathbf{p})\,,
	\end{equation}
	for some non-zero constant $C\in\mathbb{C}^{\times}$, where $\delta\colon G\to\mathbb{C}$ is the delta function at~$0$.
<!-- end definition-4 -->
<!-- theorem-5: body_md -->
\label{thm:fqdilogbasicproperties}
	Let $E\colon G\to \mathbb{C}$ be a finite quantum dilogarithm for a metric group $G$ with $|G|=N>4$. Then
	\begin{thmenum}
		\item $E(0)^2=\sqrt{N}E(0)+1$ and $C=-\widehat{E}(0)$;
		\item $E(u)E(-u) = \langleu\rangle^{-1}$ for all $u\ne 0$;
		\item $\widehat{E}(u) = \lambda\langleu\rangle E(u)$ for all $u\in G$, where $\lambda=\widehat{E}(0)/E(0)$ satisfies $\lambda^3=\int_{G}\langlex\rangle^{-1}dx$;
		\item the function $\widetilde{E}(u)=\langleu\rangle E(u)$ satisfies the dual pentagon equation
		\begin{equation} \label{eq:pentagonintegraldual}
		\langlex;y\rangle^{-1}\widetilde E(x)\widetilde E(y) = \widetilde{C} + \int_{G}\widetilde E(y-z)\widetilde E(z)\langlez\rangle^{-1}\widetilde E(x-z)dz
		\end{equation}
		for $\widetilde C=\lambda^{-2}C$.
	\end{thmenum}
<!-- end theorem-5 -->
<!-- proof-8: body_md -->
Let $p_x(z)=E(x-z)E(z)\langlez\rangle$. Then~\eqref{eq:pentagonintegral} can be rewritten using convolution as
	\[E(x)E(y)\langlex;y\rangle = C + (p_x\star E)(y)\,,\]
	which after Fourier inversion becomes
	\begin{equation} \label{eq:supportnew}
		E(x)\widehat{E}(w-x) = C\sqrt{N}\delta(w) + \widehat{E}(w)\widehat{p_x}(w)\,.
	\end{equation}
	This immediately implies that $0\in\mathrm{supp}(\widehat{E})$ and that $\mathrm{supp}(\widehat{E})$ is $H$-invariant, where $H$ is the span of $S=\mathrm{supp}(E)$. From~\eqref{eq:pentagonintegral} we also immediately get that $H=G$, since otherwise taking $x,y\not\in H$ would contradict $C\ne0$. Then we also get $\mathrm{supp}(\widehat{E})=G$. Taking $x=0$ in~\eqref{eq:supportnew} gives that $\widehat{p_0}(w)=E(0)$ for $w\ne 0$ and
		\[\widehat{p_0}(0) = E(0)-\frac{C\sqrt{N}}{\widehat{E}(0)}\,.\]
	This implies that 
	\begin{equation} \label{eq:reflection1}
    E(u)E(-u)\langleu\rangle=\sqrt{N}E(0)\delta(u)-\frac{C}{\widehat{E}(0)}\,.
    \end{equation}
	In particular, this implies that $E(u)\ne 0$ for all $u\in G$. Next, note that
		\[\widehat{p_x}(w) 
		= \int_{G}E(z)\langlez\rangle E(x-z)\langle-w;z\rangle dz 
		= \langlex\rangle\langle-w;x\rangle\int_{G}E(x-z)\langlez\rangle E(z)\langlew-x;z\rangle dz = \frac{\langlex-w\rangle}{\langlew\rangle}\widehat{p_{x}}(x-w) \,,\]	
	so
	\[\frac{E(x)\widehat{E}(w-x)-C\sqrt{N}\delta(w)}{\widehat{E}(w)\langlew\rangle^{-1}} = \frac{E(x)\widehat{E}(-w)-C\sqrt{N}\delta(x-w)}{\widehat{E}(x-w)\langlex-w\rangle^{-1}}\,.\]
	When $w\ne0$ and $x\ne w$, we get
	\[\widehat{E}(w-x)\widehat{E}(x-w)\langlex-w\rangle^{-1} = \widehat{E}(w)\langlew\rangle^{-1}\widehat{E}(-w)\,,\]
	so that $\widehat{E}(u)\widehat{E}(-u)\langleu\rangle^{-1} = d$ for some constant~$d$, for all $u\ne 0$. Then, setting $w=0$ and $x\ne 0$ gives
	\[E(x)\widehat{E}(-x)\widehat{E}(x)\langlex\rangle^{-1}-C\sqrt{N}\widehat{E}(x)\langlex\rangle^{-1} = E(x)\widehat{E}(0)^2\,,\]
	which can be rewritten as
	\begin{equation} \label{eq:weileigen}
	E(x)(d - \widehat{E}(0)^2) = C\sqrt{N}\widehat{E}(x)\langlex\rangle^{-1}\,,
	\end{equation}
	hence $\widehat{E}(x) = \lambda \langlex\rangle E(x)$ for all $x\ne 0$, for some constant $\lambda\in\mathbb{C}^{\times}$.
    
    Thus to prove (i) and (ii) it remains to show that $C=-\widehat{E}(0)$ and $\lambda = \widehat{E}(0)/E(0)$. Taking $x=w\ne 0$ in~\eqref{eq:supportnew} and using the symmetry $\widehat{p_x}(x)=\widehat{p_x}(0)\langlex\rangle^{-1}$ gives
	\[E(x)\widehat{E}(0) = \widehat{E}(x)\langlex\rangle^{-1}\widehat{p_x}(0)\,,\]
	so $\widehat{E}(0)=\lambda\widehat{p_x}(0)$ for $x\ne 0$. Since $\widehat{p_x}(0)=(\widetilde{E}\star E)(x)$, we see that $\widehat{\widetilde{E}}(y)\widehat{E}(y)$ is constant on $y\ne 0$, so $\widehat{\widetilde{E}}(y)$ is proportional to $E(-y)$ on $G\smallsetminus \{0\}$. Therefore, for some $c\in \mathbb{C}$, $\widetilde{E}(x)-c\widehat{E}(x)$ is constant, and since $\langlex\rangle E(x)$ is not constant on $G\smallsetminus\{0\}$\footnote{Since $|G|>4$, if $x\mapsto\langlex\rangle E(x)$ were constant on $G\smallsetminus\{0\}$, then by~\eqref{eq:reflection1} so would be~$\langle\cdot\rangle$, which is impossible for $N>4$.}, we see that $\widehat{E}(x)=\lambda\langlex\rangle E(x)$ for all $x$, so $\lambda=\widehat{E}(0)/E(0)$. This then also shows that $d=-\lambda^2C/\widehat{E}(0) = -C\widehat{E}(0)/E(0)^2$. Plugging this back into~\eqref{eq:weileigen} then gives
		\[-C\widehat{E}(0)/E(0)^2 - \widehat{E}(0)^2 = C\sqrt{N}\lambda\,.\]
	We can rewrite this as
        \[E(0)^2 = -\frac{C}{\widehat{E}(0)}(E(0)\sqrt{N}+1)\,,\]
    and using~\eqref{eq:reflection1} gives
        \[E(0)^2 = (E(0)^2-E(0)\sqrt{N})(E(0)\sqrt{N}+1)\,,\]
    which, since $E(0)\ne0$, is equivalent to $E(0)^2=\sqrt{N}E(0)+1$. Comparing this with~\eqref{eq:reflection1} again also gives us the identity $C=-\widehat{E}(0)$.

    To prove (iii), it remains to show that $\lambda^3$ is equal to the Gauss sum $\int_G\langlex\rangle^{-1}dx$. This is a general property of the Fourier--Weil operator $Wf(x)=\langlex\rangle^{-1}\widehat{f}(x)$: 
    \begin{align*}
    W^3f(x) &= \int_{G^3}\langlex\rangle^{-1}\langlex;r\rangle^{-1}\langler\rangle^{-1}\langler;s\rangle^{-1}\langles\rangle^{-1}\langles;t\rangle^{-1}f(t)drdsdt\\
            &= \int_{G^3}\langlex+r\rangle^{-1}\langler+t\rangle\langler+s+t\rangle^{-1}f(t)drdsdt\\
            &= \int_{G}\langles\rangle^{-1}ds\int_{G^2}\langlex+r\rangle^{-1}\langler+t\rangle f(t)drdt\\
            &= \int_{G}\langles\rangle^{-1}ds\int_{G^2}\langlet-x\rangle\langlex+r;t-x\rangle f(t)drdt = f(x)\int_{G}\langles\rangle^{-1}ds\,.
    \end{align*}
    Specialising this to $f=E\ne0$ gives $\lambda^3 = \int_{G}\langlex\rangle^{-1}dx$, as claimed.
    
	Finally, we show (iv). Multiplying~\eqref{eq:pentagonintegraldual} by $\langley;w\rangle$ and integrating in $y$ gives an equivalent relation
	\[\lambda^{-1}\langlex\rangle E(x)E(w-x) = \sqrt{N}\widetilde{C}\delta_0(w) 
	+ \lambda^{-1}E(w)\int_{G}E(x-z)E(z)\langlez;w\rangle\langlex-z\rangle dz\,,\]
	where we used $\widehat{E}(x)=\lambda \langlex\rangle E(x)=\lambda \widetilde{E}(x)$.
    Multiplying both sides by $\lambda^2\langlex\rangle^{-1}\langlew-x\rangle$ gives
		\[ E(x)\widehat{E}(w-x) = \lambda^2\sqrt{N}\widetilde{C}\delta(w) 
    	+ \langlew;-x\rangle\widehat{E}(w)\int_{G}E(x-z)E(z)\langlez;w\rangle\langlex-z\rangle dz\,,\]
	or equivalently
		\[ E(x)\widehat{E}(w-x) = \lambda^2\sqrt{N}\widetilde{C}\delta(w) 
    	+ \widehat{E}(w)\int_{G}E(x-z)E(z)\langlez-x;w\rangle\langlex-z\rangle dz\,.\]
    Changing $z\mapsto x-z$ in the integral gives an equivalent identity

\label{thm:ghostsic}
Let $\gamma=(\begin{smallmatrix}M-1&-1\\1&0\end{smallmatrix})^3$, $\varepsilon^2-(M-1)\varepsilon+1=0$, $\varepsilon>1$, and consider the operator

<!-- proof-11: body_md -->
Expanding~\eqref{eq:operatorquadratic} in the basis $\mathbf{q}^r\mathbf{p}^s$ gives an equivalent identity
	\[\frac{1}{M}\sum_{i,j=0}^{M-1}F_{\gamma}\Big(\frac{i\varepsilon+j}{M}\Big)F_{\gamma}\Big(\frac{(r-i)\varepsilon+(s-j)}{M}\Big)\Big\langle\frac{i\varepsilon+j}{M}\Big\rangle\mathbf{e}_M(i(-r) + j(r-s)) = \sqrt{M-3}F_{\gamma}\Big(\frac{r\varepsilon+s}{M}\Big)+M\delta(r)\delta(s)\,,\]
	and using the above formula for pairings, we see that it is equivalent to

    with $w=\tfrac{(r-s)\varepsilon+r}{\varepsilon^{-3}-1}$. Note that we have an exact sequence
		\[0\longrightarrow H\longrightarrow G \xlongrightarrow{\cdot M} A \longrightarrow 0\,,\]
	where $A\cong \mathbb{Z}/(M-3)\mathbb{Z}$ is generated by $\frac{\varepsilon-1}{M-3}$. Denoting $v=\frac{r\varepsilon+s}{M}$, the above identity can be rewritten as
	\[\frac{1}{M}\sum_{u\in H}F_{\gamma}(u)\langleu\rangle F_{\gamma}(v-u)\langlew;u\rangle = \sqrt{M-3}F_{\gamma}(v)+M\delta(v)\]
	For $v=0$, the identity is trivial. For $v\ne 0$, we can rewrite the left hand side as
	\begin{equation} \label{eq:sicmaintrick}
    \begin{aligned}
    \frac{1}{M}&\sum_{u\in H}F_{\gamma}(u)\langleu\rangle F_{\gamma}(v-u)\langlew;u\rangle \\
	&= \frac{1}{M(M-3)}\sum_{j=0}^{M-4}\sum_{u\in H}F_{\gamma}(u)\langleu\rangle F_{\gamma}(v-u)\langlew+j\tfrac{\varepsilon-1}{M-3};u\rangle \\
	&= \frac{1}{M(M-3)}\sum_{j=0}^{M-4}\sum_{u\in G}F_{\gamma}(u)\langleu\rangle F_{\gamma}(v-u)\langlew+j\tfrac{\varepsilon-1}{M-3};u\rangle\\
	&= \frac{1}{\sqrt{M-3}}\sum_{j=0}^{M-4}F_{\gamma}(v)F_{\gamma}(w+j\tfrac{\varepsilon-1}{M-3})\widetilde F_{\gamma}(-v-w-j\tfrac{\varepsilon-1}{M-3})\,,
	\end{aligned}
    \end{equation}
	where in the last step we used~\eqref{eq:pentagonproduct3} with the shorthand $\widetilde{F}_{\gamma}(x)=\langlex\rangle F_{\gamma}(x)$.
	We claim that
	\[F_{\gamma}(w+j\tfrac{\varepsilon-1}{M-3})= F_{\gamma}(v+w+j\tfrac{\varepsilon-1}{M-3})\,.\]
	Indeed, $(\varepsilon-1)^2=(M-3)\epsilon$, and note that
		\[(\varepsilon-1)w = (\varepsilon-1)\frac{(r-s)\varepsilon+r}{\varepsilon^{-3}-1} = \frac{r\varepsilon+s}{M}+((1-M)r+(M-2)s)\varepsilon+(r-s) \equiv v \ (\text{mod}\ \mathbb{Z}[\varepsilon])\,,\]
	so by~\eqref{eq:fgamma3sym}
	\[F_{\gamma}(w+j\tfrac{\varepsilon-1}{M-3})=F_{\gamma}((w+j\tfrac{\varepsilon-1}{M-3})\varepsilon)= F_{\gamma}(v+w+j\tfrac{\varepsilon-1}{M-3})\,.\]
	Thus, by the reflection property of~$F_{\gamma}$ (Theorem~\ref{thm:fqdilogbasicproperties} (ii)), in the last sum in~\eqref{eq:sicmaintrick} all terms are equal to $F_{\gamma}(v)$, and we get the claim.

<!-- theorem-9: body_md -->
Let $E\colon \mathbb{Z}/p\mathbb{Z}\to \mathbb{C}$ be a finite quantum dilogarithm on a cyclic group of prime order. Then $E(u)\in \overline{\mathbb{Q}}$ for all $u\in \mathbb{Z}/p\mathbb{Z}$.
<!-- end theorem-9 -->
<!-- proof-12: body_md -->
Let $X_u$ be a variable corresponding to $E(u)$ and consider the variety~$\mathcal{X}$ defined by equations (i)-(iii) in Theorem~\ref{thm:fqdilogbasicproperties}, namely:
	\begin{align*}
		X_0^2&=\sqrt{N}X_0+1\,,\\
		X_uX_{-u}&=\langleu\rangle^{-1}\,,\qquad u\ne 0\,,\\
		\frac{1}{\sqrt{N}}\sum_{v}X_v\langle-u;v\rangle &= \lambda \langleu\rangle X_u\,,\qquad u\in \mathbb{Z}/p\mathbb{Z}\,.
	\end{align*}
	Consider the projectivisation of this variety, with $X_u=Y_u/Z$, where $(Y_0\colon \cdots\colon Y_{p-1}\colon Z)\in\mathbb{P}^p(\mathbb{C})$, and look at its intersection with the hyperplane at infinity, given by $Z=0$. For any point of that intersection we have $Y_0=0$, $Y_uY_{-u}=0$, $u\in\mathbb{Z}/p\mathbb{Z}$, and $\widehat{Y}_u=\lambda \langleu\rangle Y_u$, where $\widehat{Y}_u=N^{-1/2}\sum_{v}Y_v\langle-u;v\rangle$. In particular, the supports of $u\mapsto Y_u$ and $u\mapsto \widehat{Y}_u$ are equal and have size at most $(p-1)/2$. By Tao's uncertainty principle~\cite{Tao05} for Fourier transform on cyclic groups of prime order, this implies that $Y_u=0$ for all $u$, and the intersection of the closure of $\mathcal{X}$ with the hyperplane at infinity is empty. Therefore, the variety~$\mathcal{X}$ is zero-dimensional, and since it is defined over a cyclotomic field, this shows that $E(u)\in\overline{\mathbb{Q}}$ for all~$u$.

    \sqrt{N}E(2u) =
    E(u)^4+

<!-- lemma-3: body_md -->
\label{lem:leavitt}
	Let $N\ge2$ and let $L_N$ be the Leavitt algebra defined by~\eqref{eq:leavitt}. Then
	\begin{thmenum}
	\item The set of reduced words $x_Ix_J'$ where $I=(i_1,\dots,i_k)$, $J=(j_1,\dots,j_l)$, and $(i_k,j_l)\ne (1,1)$ if $k,l\ge1$ forms a basis of~$L_N$ as a vector space;
	\item Centre of $L_N$ consists of scalars: $\mathcal{Z}(L_N)=\mathbb{C}\,1$;
	\item If $y\in L_N$ satisfies $x_iy=yx_j$ for some $i\ne j$, then $y=0$;
    \item The centraliser of an element $u=x_1+P(x_2,\dots,x_N)$, where $P$ is a non-commutative polynomial without a constant term, is $\mathbb{C}[u]$. The same conclusion holds for $u=x_1'+P(x_2',\dots,x_N')$.
	\end{thmenum}
<!-- end lemma-3 -->
<!-- proof-16: body_md -->
Parts (i) and (ii) are well known, see Leavitt's work such as~\cite{Lea65} and its citations. For (iii), note that for all $k\ge0$ we have, by induction,
		\[y=(x_i')^k y x_j^k\,.\]
	Let $y=\sum_{I,J}c_{I,J} x_Ix_J'$ be the decomposition of $y$ into the reduced words basis, and assume that $|I|,|J|<k$ for all pairs $(I,J)$ that occur. Then for all $I,J$ with $c_{I,J}\ne 0$, we have
	\[(x_i')^kx_Ix_J'x_j^k = (x_i')^{k-|I|}x_j^{k-|J|} = 0\,,\]
	by~\eqref{eq:leavitt}, so $y=(x_i')^k y x_j^k = 0$. For part (iv), when $P=0$, this is explained in~\cite[Proof of Proposition~1]{EG17}. For non-zero $P$, simply note that $x_1\mapsto x_1+P(x_2,\dots,x_N)$, $x_1'\mapsto x_1'$, $x_i\mapsto x_i$, $x_i'\mapsto x_i'-x_i'P(x_2,\dots,x_N)x_1'$, $i\ge2$, is an automorphism of $L_N$, so the statement reduces to the case $P=0$. For the last statement, applying the anti-involution reduces it to the statement we have just proved.

	\[-\lambda^{-1}E(0)\frac{\langler\rangle}{\langlek\rangle}+\frac{\langler\rangle}{\sqrt{N}}\sum_{h}\langleh\rangle^2E(-h)E(k-h)E(h-r)\langleh;k\rangle^{-1} = E(k-r)\langle-r\rangle E(-r)\]
	or equivalently
	\[-\lambda^{-1}E(0)+\frac{1}{\sqrt{N}}\sum_{h}\langley-h\rangle E(y-h)E(h)\langlex-h\rangle E(x-h) = \langlex-y\rangle E(x)E(y)\,,\]
	which is precisely~\eqref{eq:pentagonintegraldual}.
