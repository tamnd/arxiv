<!-- theorem-1: body_md -->
\label{thm:first-order-averaging}
Assume that \(\boldsymbol{X}_*\in\Omega\) is a simple zero of the averaged vector field
\(\bar{\boldsymbol{F}}\), that is,
\relax{equation}
\label{eq:simple-zero}
\bar{\boldsymbol{F}}(\boldsymbol{X}_*)=0,
\qquad
\det D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)\neq0.
\relax{equation}
Then, for all sufficiently small \(\varepsilon>0\), system
\eqref{eq:averaging-standard} possesses a \(T\)-periodic solution
\(\boldsymbol{X}(s,\varepsilon)\) such that
\[
\boldsymbol{X}(0,\varepsilon)\longrightarrow \boldsymbol{X}_*
\qquad
 
 \relax
 \ifmmode
  \mathchoice
   {\hbox{{\everymath{\displaystyle     }as}}} 
   {\hbox{{\everymath{\textstyle        }as}}} 
   {\hbox{{\everymath{\scriptstyle      }\%\ssf@size\sf@size\selectfontas}}} 
   {\hbox{{\everymath{\scriptscriptstyle}\%\ssf@size\ssf@size\selectfontas}}} 
  \glb@settings
 \else
  \mbox{as} 
 \fi

\qquad
\varepsilon\to0.
\]
Equivalently, every simple zero of the averaged vector field gives rise to a
periodic solution of the original perturbed system.

<!-- remark-1: body_md -->
\upshape
In the applications to the ABC flow, the independent variable \(s\) is
one of the angular variables \(x,y,z\). Hence, in all three cyclic regimes
considered below, the averaging period is
$
T=2\pi.
$
Consistently with the angular convention adopted in the Introduction, we
shall use the symmetric averaging interval
$
[-\pi,\pi].
$ 
Since the integrands are \(2\pi\)-periodic, this choice is equivalent to
the interval \([0,2\pi]\).
More precisely, near the \(a\)-axis we take \(s=y\), near the \(b\)-axis we take
\(s=z\), and near the \(c\)-axis we take \(s=x\).

The variable \(\boldsymbol{X}\) in
Theorem~\ref{thm:first-order-averaging} will be identified with the
corresponding reduced variables
\[
\boldsymbol{x}_a=(x,Z)^{\mathsf T},\qquad
\boldsymbol{x}_b=(y,X)^{\mathsf T},\qquad
\boldsymbol{x}_c=(z,Y)^{\mathsf T}
\]
in the three cyclic regimes, respectively. The detailed derivation of these
reduced systems is given in Section~\ref{sec:periodic-streamlines}.

\qquad
\boldsymbol{x}\in\Gamma.

<!-- theorem-3: body_md -->
\label{thm:LZ}
Consider system~\eqref{eq:averaging-standard} under the assumptions of
Theorem~\ref{thm:first-order-averaging}, and let
\(\boldsymbol{X}_*\in\Omega\) be a simple zero of the averaged vector field
\(\bar{\boldsymbol{F}}\). Assume, in addition, that \(\boldsymbol{F}\) is of class \(C^2\)
with respect to \(\boldsymbol{X}\) and that \(\boldsymbol{R}\) is of class \(C^1\) with
respect to \(\boldsymbol{X}\), uniformly on the compact subsets under
consideration.

Let \(\boldsymbol{X}(s,\varepsilon)\) be the \(T\)-periodic solution bifurcating from
\(\boldsymbol{X}_*\). Then
\relax{equation*}
\label{eq:periodic-solution-first-order}
\boldsymbol{X}(s,\varepsilon)
=
\boldsymbol{X}_*+{\mathcal O}(\varepsilon)
\relax{equation*}
uniformly for \(s\in[0,T]\).

Let \(\boldsymbol{\Phi}(s,\varepsilon)\) denote the fundamental matrix solution of the
variational system
\relax{equation}
\label{eq:var-averaged-system}
\relax{aligned}
&\frac{\mathrm{d}\boldsymbol{Y}}{\mathrm{d} s}
=
\varepsilon
\left.
D_{\boldsymbol{X}}
\bigl(
\boldsymbol{F}(s,\boldsymbol{X})
+
\varepsilon\boldsymbol{R}(s,\boldsymbol{X},\varepsilon)
\bigr)
\right|_{\boldsymbol{X}=\boldsymbol{X}(s,\varepsilon)}
\boldsymbol{Y},
\\
&\boldsymbol{\Phi}(0,\varepsilon)
=
\boldsymbol{I}_m.
\relax{aligned}
\relax{equation}
Then the corresponding monodromy matrix satisfies
\relax{equation}
\label{eq:LZ-monodromy}
\boldsymbol{\Phi}(T,\varepsilon)
=
\exp\!\left(
\varepsilon T D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)
\right)
+
{\mathcal O}(\varepsilon^2).
\relax{equation}

Let \(\lambda_1,\ldots,\lambda_m\) be the eigenvalues of
\(D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)\). If these eigenvalues are simple, then,
after a suitable labeling, the characteristic multipliers
\(\rho_1(\varepsilon),\ldots,\rho_m(\varepsilon)\) of the reduced periodic system
satisfy
\relax{equation}
\label{eq:multiplier-averaging}
\rho_j(\varepsilon)
=
\exp\!\left(
\varepsilon T\lambda_j
\right)
+
{\mathcal O}(\varepsilon^2),
\qquad
j=1,\ldots,m.
\relax{equation}

<!-- theorem-4: body_md -->
\label{thm:main-cyclic}
Consider the ABC flow~\eqref{eq:abc} on \(\REV@mathbb T^3\). Assume that one of the
following perturbative regimes holds:
\relax{align}
\mathrm{(A)}\qquad
&a\neq0,
&&b=\varepsilon^2 b_1,
&&c=\varepsilon^2 c_1,
&&b_1\neq0,
\notag\\[1mm]
\mathrm{(B)}\qquad
&b\neq0,
&&c=\varepsilon^2 c_1,
&&a=\varepsilon^2 a_1,
&&c_1\neq0,
\label{eq:cyclic-regimes}
\\[1mm]
\mathrm{(C)}\qquad
&c\neq0,
&&a=\varepsilon^2 a_1,
&&b=\varepsilon^2 b_1,
&&a_1\neq0,
\notag
\relax{align}
where \(0<\varepsilon\ll1\). Then, in each regime,  the ABC flow possesses two periodic solutions on \(\REV@mathbb T^3\)
bifurcating from the corresponding degenerate family on the integrable axis.

More precisely, let
\[
\theta_+=\frac{\pi}{2},
\qquad
\theta_-=-\frac{\pi}{2}.
\] 
In the three regimes, we introduce the transverse rescaled variables
\[
z=\varepsilon Z,\qquad
x=\varepsilon X,\qquad
y=\varepsilon Y,
\]
respectively.
The first averaged vector fields in the three regimes are as follows
\relax{equation}
\label{eq:averaged-fields-main}
\relax{aligned}
\bar{\boldsymbol{F}}_a(x,Z)
&=
\left(Z,\frac{b_1}{a}\cos x\right)^{\mathsf T},
\\
\bar{\boldsymbol{F}}_b(y,X)
&=
\left(X,\frac{c_1}{b}\cos y\right)^{\mathsf T},
\\
\bar{\boldsymbol{F}}_c(z,Y)
&=
\left(Y,\frac{a_1}{c}\cos z\right)^{\mathsf T}.
\relax{aligned}
\relax{equation}
In the corresponding   variables
$
(x,Z),\ (y,X)$, and $(z,Y)$, 
their simple zeros are
\[
P^a_\pm=(\theta_\pm,0),
\qquad
P^b_\pm=(\theta_\pm,0),
\qquad
P^c_\pm=(\theta_\pm,0).
\]

We give the proof for regime \(\mathrm{(A)}\). The Jacobian matrix of the
averaged vector field \(\bar{\boldsymbol{F}}_a\) is given by~\eqref{eq:A-Jac}, and its
characteristic polynomial is
\relax{equation}
\label{eq:A-charpoly}
p_a(\lambda;x)
=
\lambda^2+\frac{b_1}{a}\sin x.
\relax{equation}
At the two simple zeros~\eqref{eq:A-zeros}, this gives
\[
p_a(\lambda;\theta_+)
=
\lambda^2+\frac{b_1}{a},
\qquad
p_a(\lambda;\theta_-)
=
\lambda^2-\frac{b_1}{a}.
\]
Hence
\[
P^a_+:\quad
\lambda_{1,2}=\pm\sqrt{-\frac{b_1}{a}},
\qquad
P^a_-:\quad
\lambda_{1,2}=\pm\sqrt{\frac{b_1}{a}},
\]
as stated in~\eqref{eq:abc-eigenvalues}.

The averaged system
\[
x'=Z,
\qquad
Z'=\frac{b_1}{a}\cos x,
\]
is Hamiltonian in the canonical coordinates $(x,Z)$, with the Hamiltonian function
\relax{equation}
\label{eq:A-averaged-Hamiltonian}
\mathcal H_a(x,Z)
=
\frac{1}{2}Z^2-\frac{b_1}{a}\sin x.
\relax{equation}
Indeed,
\[
x'
=
\frac{\partial\mathcal H_a}{\partial Z},
\qquad
Z'
=
-\frac{\partial\mathcal H_a}{\partial x}.
\]
After the shift
\[
q=x-\frac{\pi}{2},
\]
the Hamiltonian becomes
\[
\mathcal H_a(q,Z)
=
\frac{1}{2}Z^2-\frac{b_1}{a}\cos q,
\]
which is the Hamiltonian of a mathematical pendulum. Consequently, if
\(b_1/a>0\), the equilibrium \(P^a_+\) is a nonlinear center, whereas
\(P^a_-\) is a hyperbolic saddle. If \(b_1/a<0\), the roles of the two
equilibria are reversed.

The formulas in regimes \(\mathrm{(B)}\) and \(\mathrm{(C)}\) are obtained by
applying the cyclic permutation~\eqref{eq:sym}. This gives the
remaining eigenvalue relations stated in~\eqref{eq:abc-eigenvalues}. The
classification into center and saddle type follows from the corresponding
Hamiltonian structure and the sign of the relevant parameter ratio.

\label{thm:abc-nonintegrability}
Assume that one of the perturbative regimes
\textup{(A)}, \textup{(B)}, or \textup{(C)} of
Theorem~\ref{thm:main-cyclic} holds. Then, for all sufficiently small
\(\varepsilon>0\), the characteristic multiplier \(1\) of each corresponding
periodic orbit has algebraic multiplicity one.

Consequently, the ABC flow~\eqref{eq:abc} admits no nonconstant first
integral \(H\in C^1\), defined in a neighborhood of any of these
periodic orbits, such that
\[
\nabla H(\boldsymbol{x})\neq0
\]
at every point \(\boldsymbol{x}\) of the corresponding periodic orbit.

\upshape
\label{rem:integrable-plane-critical-integral}
Theorem~\ref{thm:abc-nonintegrability} does not require both transverse
perturbation coefficients to be nonzero. For instance, in regime
\(\mathrm{(A)}\), the nondegeneracy condition requires \(ab_1\neq0\),
whereas \(c_1\) may vanish, since its contribution has zero average with
respect to the fast angle \(y\).

This does not contradict the integrability of the case \(c=0\). Indeed,
the ABC flow then admits the first integral
\[
H(x,y,z)=a\cos z+b\sin x.
\]
The two periodic orbits selected by averaging are, in this case, given
explicitly by
\[
x(t)\equiv\pm\frac{\pi}{2},
\qquad
z(t)\equiv0,
\qquad
y(t)=y_0+(a\pm b)t,
\]
with the signs chosen consistently. Since
\[
\nabla H(x,y,z)
=
\left(b\cos x,\,0,\,-a\sin z\right),
\]
we have
\[
\nabla H\left(\pm\frac{\pi}{2},y,0\right)=0.
\]
Thus, although a first integral exists, it is critical along the periodic
orbits selected by averaging. This is fully consistent with
Theorem~\ref{thm:abc-nonintegrability}, which excludes only \(C^1\) first
integrals that are regular along the corresponding periodic orbit.

By cyclic symmetry, the same observation applies to regimes
\(\mathrm{(B)}\) and \(\mathrm{(C)}\).

Let
\[
\mathcal P_\varepsilon(\boldsymbol{X}_0)
=
\boldsymbol{X}(T;0,\boldsymbol{X}_0,\varepsilon)
\]
denote the period map of
system~\eqref{eq:averaging-standard} based at \(s=0\).
Under the assumed regularity, integration of
system~\eqref{eq:averaging-standard} over one period gives
\relax{equation}
\label{eq:period-map-expansion}
\mathcal P_\varepsilon(\boldsymbol{X}_0)
=
\boldsymbol{X}_0
+
\varepsilon T\bar{\boldsymbol{F}}(\boldsymbol{X}_0)
+
{\mathcal O}(\varepsilon^2),
\relax{equation}
where the remainder is \({\mathcal O}(\varepsilon^2)\) in \(C^1\) on a
neighborhood of \(\boldsymbol{X}_*\).

For \(\varepsilon\neq0\), we define the normalized displacement map by
\[
\mathcal D(\boldsymbol{X}_0,\varepsilon)
=
\frac{\mathcal P_\varepsilon(\boldsymbol{X}_0)-\boldsymbol{X}_0}{\varepsilon}.
\]
Expansion~\eqref{eq:period-map-expansion} shows that this map extends
to \(\varepsilon=0\) by setting
\[
\mathcal D(\boldsymbol{X}_0,0)
=
T\bar{\boldsymbol{F}}(\boldsymbol{X}_0).
\]
Since
$
\mathcal D(\boldsymbol{X}_*,0)=0
$
and
\[
D_{\boldsymbol{X}_0}\mathcal D(\boldsymbol{X}_*,0)
=
T D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)
\]
is nonsingular, the Implicit-Function Theorem yields a fixed point
\(\boldsymbol{X}_0(\varepsilon)\) of \(\mathcal P_\varepsilon\) satisfying
\[
\boldsymbol{X}_0(\varepsilon)
=
\boldsymbol{X}_*+{\mathcal O}(\varepsilon).
\]
This fixed point determines the \(T\)-periodic solution
\(\boldsymbol{X}(s,\varepsilon)\). Since the right-hand side of
\eqref{eq:averaging-standard} is of order \(\varepsilon\), we also have
\[
\boldsymbol{X}(s,\varepsilon)
=
\boldsymbol{X}_*+{\mathcal O}(\varepsilon)
\]
uniformly for \(s\in[0,T]\).

The derivative of the period map at this fixed point is the monodromy
matrix of the corresponding reduced periodic solution. Therefore,
using the \(C^1\)-expansion~\eqref{eq:period-map-expansion}, we obtain
\relax{align*}
\boldsymbol{\Phi}(T,\varepsilon)
&=
D_{\boldsymbol{X}_0}\mathcal P_\varepsilon(\boldsymbol{X}_0(\varepsilon))
\\
&=
\boldsymbol{I}_m
+
\varepsilon T
D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_0(\varepsilon))
+
{\mathcal O}(\varepsilon^2)
\\
&=
\boldsymbol{I}_m
+
\varepsilon T
D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)
+
{\mathcal O}(\varepsilon^2)
\\
&=
\exp\!\left(
\varepsilon T D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)
\right)
+
{\mathcal O}(\varepsilon^2),
\relax{align*}
which proves~\eqref{eq:LZ-monodromy}.

Now set
\[
\boldsymbol{K}_\varepsilon
=
\frac{\boldsymbol{\Phi}(T,\varepsilon)-\boldsymbol{I}_m}{\varepsilon}.
\]
Then
\[
\boldsymbol{K}_\varepsilon
=
T D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)
+
{\mathcal O}(\varepsilon).
\]
Let \(\kappa_j(\varepsilon)\) be the eigenvalues of \(\boldsymbol{K}_\varepsilon\). Since the
eigenvalues \(\lambda_j\) of
\(D_{\boldsymbol{X}}\bar{\boldsymbol{F}}(\boldsymbol{X}_*)\) are simple, standard matrix perturbation

\[
\kappa_j(\varepsilon)
=
T\lambda_j+{\mathcal O}(\varepsilon),
\qquad
j=1,\ldots,m.
\]

\[
\boldsymbol{\Phi}(T,\varepsilon)
=
\boldsymbol{I}_m+\varepsilon\boldsymbol{K}_\varepsilon,

&=
1+\varepsilon\kappa_j(\varepsilon)
\\
&=
1+\varepsilon T\lambda_j+{\mathcal O}(\varepsilon^2)
\\
&=
\exp(\varepsilon T\lambda_j)+{\mathcal O}(\varepsilon^2),\qquad j=1,\ldots,m.
\relax{align*}
which proves~\eqref{eq:multiplier-averaging}.
