<!-- definition-1: body_md -->
\label{def: 1}
The functional ${\bf x}\mapsto\mathcal{Q}({\bf x},{\bf v}):\mathbb{R}^{n}\rightarrow \mathbb{R}$ is a 
quadratic tangent majorant for $\mathcal{J}_{\epsilon,\lambda}({\bf x})$ 
at ${\bf v}\in\mathbb{R}^n$, if for all ${\bf v} \in \mathbb{R}^n$,
\begin{enumerate}
\par\hangindent 40\p@ ${\bf x}\mapsto\mathcal{Q}({\bf x},{\bf v})$ is quadratic,
\par\hangindent 40\p@ $\mathcal{Q}({\bf v},{\bf v})=\mathcal{J}_{\epsilon,\lambda}({\bf v})$,
\par\hangindent 40\p@ $\bigtriangledown_{{\bf x}}\mathcal{Q}({\bf v},{\bf v})=\bigtriangledown_{{\bf x}}\mathcal{J}_{\epsilon,\lambda}({\bf v})$,
\par\hangindent 40\p@ $\mathcal{Q}({\bf x},{\bf v})\geq \mathcal{J}_{\epsilon,\lambda}({\bf x})\quad\forall {\bf x}\in \mathbb{R}^{n}$.
\end{enumerate}

<!-- head-1: body_md -->
\label{sec: MM_GKS}
In the MM approach, 
the next iterate, 
${\bf x}^{(k+1)}$, 
is 
the minimizer of 
\eqref{eq: QuadraticMajorantQ}.
Setting the gradient to zero
leads to the normal equations
\begin{equation}\label{eq: normaleqQuadMajorant}
({\bf A}^T{\bf A} + \lambda  \Psi^{T} ({\bf P}_{\epsilon}^{(k)})^2\Psi){\bf x} = {\bf A}^T{\bf d}.
\end{equation}
The system \eqref{eq: normaleqQuadMajorant} has a unique solution if
\begin{equation}\label{eq: NANL}
\mathcal{N}({\bf A}^T{\bf A})\cap \mathcal{N}(\Psi^{T} ({\bf P}_{\epsilon}^{(k)})^2\Psi)=\{0\},
\end{equation}
which typically holds in practice.
In that case, for given $\lambda$,
the solution of \eqref{eq: normaleqQuadMajorant} is the unique minimizer
of the quadratic tangent majorant $\mathcal{Q}({\bf x}, {\bf x}^{(k)})$.
In practice, $\lambda$ is usually not known in advance, and we discuss its computation in Section \ref{sec: RegParm}.

Unfortunately, solving \eqref{eq: normaleqQuadMajorant} for large ${\bf A}$, $\Psi$, and changing
${\bf P}_\epsilon^{(k)}$, may be computationally prohibitive.  Therefore, 
the 
MM-GKS method, proposed
in \cite{huang2017majorization}, approximates
the MM iterates 
by projection onto low dimensional subspaces. 
The method starts with a few steps of Golub-Kahan
bidiagonalization (GKB) 
with ${\bf A}$ and ${\bf u}_1 = {\bf d}/\| {\bf d} \|_2$ to determine an initial subspace ${\bf V}_\ell$ such that 
\begin{equation}\label{bdiag}
{\bf A}{\bf V}_{\ell}={\bf U}_{\ell+1}{\bf B}_{\ell},
\end{equation}
 
and ${\bf V}_\ell$ and
${\bf U}_{\ell+1}$ have orthonormal columns.
Given ${\bf x}^{(k)}$ and 
${\bf P}_{\epsilon}^{(k)}$ 
obtained from 
${\bf x}^{(k)}$ (with $k=0$ in the first step),
MM-GKS computes the thin QR factorizations 
\begin{align}\label{eq: QR}
{\bf A}{\bf V}_{\ell+k} = {\bf Q}_{{\bf A}}{\bf R}_{{\bf A}}, \quad
{\bf P}_{\epsilon}^{(k)}\Psi{\bf V}_{\ell+k} = {\bf Q}_{\Psi}{\bf R}_{\Psi} .
\end{align}
Restricting 
${\bf x}^{(k+1)}$ to 
${\rm range}
({\bf V}_{\ell + k})$, 
${\bf x}^{(k+1)} = 
{\bf V}_{\ell + k} {\bf z}^{(k+1)}$, 
and adjusting
\eqref{eq: normaleqQuadMajorant}
accordingly leads to the
following small system
of equations
for ${\bf z}^{(k+1)}$, 
\begin{equation} \label{eq: minKryov2}
  ({\bf R}_{\bf A}^T{\bf R}_{\bf A} + 
  \lambda {\bf R}_{\Psi}^T{\bf R}_{\Psi}){\bf z}^{(k+1)} =
  {\bf R}_{\bf A}^T{\bf Q}_{\bf A}^T{\bf d} .
\end{equation} 
Its solution gives for
the residual vector of the (full) regularized normal equations:
\begin{equation}\label{eq:residual}
{\bf r}^{(k+1)}={\bf A}^T({\bf A}{\bf V}_{\ell+k}{\bf z}^{(k+1)} -{\bf d})+\lambda  \Psi^T({\bf P}_{\epsilon}^{(k)})^2\Psi
{\bf V}_{\ell+k}{\bf z}^{(k+1)}.
\end{equation}
We expand the solution subspace with the normalized residual ${\bf v}_{\rm new}={\bf r}^{(k+1)}/\|{\bf r}^{(k+1)}\|_2$,
\begin{equation}\label{eq: enlargeMMGKS}
{\bf V}_{\ell+k+1}=[{\bf V}_{\ell+k},{\bf v}_{\rm new}]\in\mathbb{R}^{n\times(\ell+k+1)}.\end{equation} 
In exact arithmetic,  ${\bf v}_{\rm new}$ is orthogonal to the columns of 
${\bf V}_{\ell+k}$, 
but in computer arithmetic 
reorthogonalization of 
${\bf V}_{\ell+k+1}$ is typically needed.
Next, ${\bf P}_{\epsilon}^{(k)}$ is updated for the new solution estimate, and the process in steps \eqref{eq: QR} - \eqref{eq: enlargeMMGKS} is repeated, expanding the solution space until a 
sufficiently accurate solution is reached.

<!-- lemma-1: body_md -->
\label{lem:min_progress}
Let $\mathcal{Q}$, $\widetilde{\mathcal{Q}}$, ${\bf Q}_k$, ${\bf x}^{(k)}$, and ${\bf b}$ be defined as above, and let $\bar{\mu} = \sup_{k = 0, 1, \ldots}
\| {\bf Q}_k \|_2$ exist and be finite. 
Furthermore, let
\begin{eqnarray*}
  {\bf x}^{(k+1)} & = & 
  \arg \min_{{\bf x} \,\in\, {\rm range}({\bf V}_{m} )}
  \widetilde{\mathcal{Q}}({\bf x},{\bf x}^{(k)}).
\end{eqnarray*}
Then 
\begin{eqnarray*}  
  \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k+1)}) \leq
  \mathcal{Q}({\bf x}^{(k+1)},{\bf x}^{(k)}) & \leq & 
  \mathcal{Q}({\bf x}^{(k)},{\bf x}^{(k)}) - 
  \frac{\| {\bf r}^{(k)} \|_2^2}{2\bar{\mu}}
  = \mathcal{J}_{\epsilon,\lambda}({\bf x}_k) - \frac{\| \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}_k) \|_2^2}{2\bar{\mu}} .
\end{eqnarray*}

<!-- proof-1: body_md -->
Let $\alpha$ be the minimizer of 
$\widetilde{\mathcal{Q}}({\bf x}^{(k)} - \widetilde{\alpha}{\bf r}^{(k)},{\bf x}^{(k)})$ as a function of $\widetilde{\alpha}$, then
$\alpha = \frac{\|{\bf r}^{(k)}\|_2^2} {{{\bf r}^{(k)}}^T {\bf Q}_k {\bf r}^{(k)}}$ (see, e.g., \cite[section 3.3]{NocWri2e}). Substituting this into \eqref{eq:Qtxk}
gives
\begin{eqnarray*}
  \widetilde{\mathcal{Q}}({\bf x}^{(k)} - \alpha {\bf r}^{(k)},{\bf x}^{(k)})
  & = & 
  \widetilde{\mathcal{Q}}({\bf x}^{(k)},{\bf x}^{(k)}) +  
  \frac{1}{2}
  \alpha^2 {{\bf r}^{(k)}}^T {\bf Q}_k {\bf r}^{(k)}
  - \alpha \|{\bf r}^{(k)}\|_2^2 \\
  & = &
  \widetilde{\mathcal{Q}}({\bf x}^{(k)},{\bf x}^{(k)}) -  
  \frac{1}{2} 
  \frac{\|{\bf r}^{(k)}\|_2^4}{{{\bf r}^{(k)}}^T{\bf Q}_k{\bf r}^{(k)}} .
\end{eqnarray*}
 
Since $\| {\bf Q}_k \|_2 \leq \bar{\mu}$, 
we have $({\bf r}^{(k)})^T{\bf Q}_k{\bf r}^{(k)} \leq
\|{\bf Q}_k\|_2 \|{\bf r}^{(k)} \|_2^2  
\leq \bar{\mu} \|{\bf r}^{(k)}\|_2^2$,
and, therefore,
$\widetilde{\mathcal{Q}}({\bf x}^{(k)} - \alpha{\bf r}^{(k)},{\bf x}^{(k)}) 
\leq 
\widetilde{\mathcal{Q}}({\bf x}^{(k)},{\bf x}^{(k)}) 
- \| {\bf r}^{(k)} \|_2^2 /(2\bar{\mu})$.
As ${\bf x}^{(k)} - \alpha{\bf r}^{(k)} \in 
\textrm{range}\;{\bf V}_m$, 
$\widetilde{\mathcal{Q}}({\bf x}^{(k+1)},{\bf x}^{(k)}) 
\leq
\widetilde{\mathcal{Q}}({\bf x}^{(k)} - \alpha {\bf r}^{(k)},{\bf x}^{(k)})
\leq 
\widetilde{\mathcal{Q}}({\bf x}^{(k)},{\bf x}^{(k)}) 
- \| {\bf r}^{(k)} \|_2^2 / ( 2\bar{\mu} )$.
This also implies
 
\[
\mathcal{Q}({\bf x}^{(k+1)},{\bf x}^{(k)}) 
\leq
\mathcal{Q}({\bf x}^{(k)},{\bf x}^{(k)}) 
- \| {\bf r}^{(k)} \|_2^2 / ( 2\bar{\mu} )
\]
 
The remaining (in)equalities follow
directly from the fact that 
$\mathcal{Q}({\bf x},{\bf x}^{(k)})$ is a quadratic 
majorant for $\mathcal{J}_{\epsilon,\lambda}({\bf x})$
and tangent to $\mathcal{J}_{\epsilon,\lambda}({\bf x})$
at ${\bf x}^{(k)}$. See equations \eqref{eq:uk}--\eqref{eq: QuadraticMajorantQ}.

<!-- thm-1: body_md -->
Let 
$\mathcal{J}_{\epsilon,\lambda}({\bf x})$ be defined as in \eqref{eq:Je} with $\phi_{\epsilon}(t)$ given by \eqref{eq:phi_eps}.
For $k = 1, 2, 3, \ldots$, let ${\bf x}^{(k)}$, $\mathcal{Q}({\bf x},{\bf x}^{(k)})$, and ${\bf x}^{(k+1)}$ be defined as in 
\eqref{eq:xk},  
\eqref{eq: QuadraticMajorantQ}, and
Lemma \ref{lem:min_progress}.
Then
\begin{eqnarray*}
  \lim_{k \rightarrow \infty}
  \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k)}) = 0 .
\end{eqnarray*}
<!-- end thm-1 -->
<!-- proof-3: body_md -->
Since $\mathcal{J}_{\epsilon,\lambda}({\bf x})$ is bounded from below ($\mathcal{J}_{\epsilon,\lambda}({\bf x}) \geq 0$), 
$\mathcal{J}_{\epsilon,\lambda}^{*} = \inf_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}) \geq 0$
exists, and using Lemma 
\ref{lem:min_progress}
we have, for each $k$,
\begin{eqnarray*}
\mathcal{J}_{\epsilon,\lambda}^{*} - \mathcal{J}_{\epsilon,\lambda}({\bf x}_0) & \leq &
\mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k+1)}) - \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(0)}) 
\leq
\mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k)}) - 
\frac{\| \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k)}) \|_2^2}{2\bar{\mu}}
- \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(0)}) \\
& \leq &
\mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k-1)}) - 
\frac{\| \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k-1)}) \|_2^2}{2\bar{\mu}} -
\frac{\| \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k)}) \|_2^2}{2\bar{\mu}}
- \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(0)}) \\
& \leq & \cdots \leq
- \frac{1}{2\bar{\mu}}
\left( \sum_{j=0}^{k}
\| \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(j)}) \|_2^2 \right) .
\end{eqnarray*}
If some $\eta > 0$ exists such that there is a (infinite) subsequence 
$\| \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(j_i)}) \|_2^2 \geq \eta$
for all $j_i$ with $i = 1, 2, \ldots$, then 
\[
\lim_{k \rightarrow \infty}
- \frac{1}{2\bar{\mu}}
\left( \sum_{j=0}^{k}
\| \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(j)}) \|_2^2 \right) = -\infty . 
\]
Hence, the inequality above shows that such an $\eta$ cannot exist, and therefore
\begin{equation} \nonumber
  \lim_{k \rightarrow \infty}
  \nabla_{{\bf x}} \mathcal{J}_{\epsilon,\lambda}({\bf x}^{(k)}) = 0 .
\end{equation} \mbox{}
