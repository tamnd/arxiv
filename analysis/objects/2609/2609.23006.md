<!-- theorem-1: body_md -->
\label{thm.Hessian-estimate}
Let $1\leq\ell<k\leq n-1$, $k-\ell\in\{1,2\}$ and $2k>n$.
Suppose that $u\in C^4(B_1)$ is a $k$-convex solution of
\eqref{eqn.sigma-quotient} in $B_1\subset\mathbb R^n$.
Then
\[
|D^2u(0)|\leq C\left(n,k,\|u\|_{C^1(B_1)}\right).
\]

From the interior gradient estimate for Hessian quotient equations by Chen \cite{C15},
we further obtain for $k$-convex solutions in $B_2$ that
\begin{equation}\label{eqn.Hessian-L-infty}
|D^2u(0)|
\leq C\left(n,k,\left\Vertu\right\Vert_{L^\infty(B_2)}\right).
\end{equation}

<!-- lemma-2: body_md -->
\label{lem.Fj-lambdaj}
Let $1\leq\ell<k<n$, $k-\ell\in\left\{1,2\right\}$,
and let $\lambda\in\Gamma_k$ satisfy $F(\lambda)=1$.
  
Then
\begin{equation}\label{eqn.Fj-lambdaj>}
F_j(1+\lambda_j^2)\geq c_1(n,k)\sum_{i=1}^n F_i\quad
\text{for any}\ j\in\mathbb Z_{[1,n]}.
\end{equation}
Moreover, if $\lambda_j<0$, then
\begin{equation}\label{eqn.Fj>}
F_j\geq c_2(n,k)\sum_{i=1}^nF_i,
\end{equation}
where $c_1=c_1(n,k)$ and $c_2=c_2(n,k)$ are positive constants.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We divide the proof into two cases.

\textit{Case 1. $F=\frac{\sigma_k}{\sigma_{k-1}}$}.
We first claim that
\begin{equation}\label{eqn.qk-i-bound}
\sum_{i=1}^nF_i\leq n-k+1,
\quad
F_j
\geq c\left(\frac{\sigma_{k-1}(\lambda|j)}{\sigma_{k-1}}\right)^2
\ \text{for any}\ j\in\mathbb Z_{[1,n]},
\end{equation}
where $c=c(n,k)\in(0,1)$.
  
Indeed, using $\sigma_k=\sigma_{k-1}$ and \eqref{eqn.Fi-sum}, we have
\[
\sum_{i=1}^n(q_k)_i
=(n-k+1)-(n-k+2)\frac{\sigma_k\sigma_{k-2}}{\sigma_{k-1}^2}
\leq n-k+1.
\]
Fix any $j\in\mathbb Z_{[1,n]}$.
Using the identities
\[
\sigma_{k-1}=\sigma_{k-1}(\lambda|j)+\lambda_j\sigma_{k-2}(\lambda|j)
=\sigma_k
=\sigma_k(\lambda|j)+\lambda_j \sigma_{k-1}(\lambda|j),
\]
we have
\[
\sigma_{k-1}^2(q_k)_j=(\sigma_{k-1}(\lambda|j))^2-\sigma_{k-2}(\lambda|j)\sigma_k(\lambda|j).
\]
If $\sigma_k(\lambda|j)\leq0$, then $(q_k)_j\geq\left(\frac{\sigma_{k-1}(\lambda|j)}{\sigma_{k-1}}\right)^2$.
If $\sigma_k(\lambda|j)>0$, the Newton--Maclaurin inequality in $n-1$ variables gives
\[
(q_k)_j
\geq\left(1-\frac{(k-1)(n-k)}{k(n-k+1)}\right)\left(\frac{\sigma_{k-1}(\lambda|j)}{\sigma_{k-1}}\right)^2.
\]
Hence \eqref{eqn.qk-i-bound} holds.

Next, if $\frac{\sigma_{k-1}(\lambda|j)}{\sigma_{k-1}}\geq\frac{1}{2}$,
then $(q_k)_j\geq\frac{c}{4}$.
  
If $\frac{\sigma_{k-1}(\lambda|j)}{\sigma_{k-1}}<\frac{1}{2}$,
then
\[
\sigma_{k-1}(\lambda|j)-\sigma_{k-2}(\lambda|j)=\sigma_{k-1}(q_k)_j>0,
\]
and hence
\[
\lambda_j^2(q_k)_j
\geq c\lambda_j^2\left(\frac{\sigma_{k-2}(\lambda|j)}{\sigma_{k-1}}\right)^2
=c\left(1-\frac{\sigma_{k-1}(\lambda|j)}{\sigma_{k-1}}\right)^2
>\frac{c}{4}.
\]
Combining these estimates with \eqref{eqn.qk-i-bound}, we obtain
\[
F_j(1+\lambda_j^2)
\geq\frac{c}{4}
\geq \frac{c}{4(n-k+1)}\sum_{i=1}^n F_i.
\]
  
Moreover, if $\lambda_j<0$, then $\lambda\in\Gamma_k$ implies $\sigma_{k-2}(\lambda|j)>0$,
and hence $\sigma_{k-1}(\lambda|j)=\sigma_{k-1}-\lambda_j\sigma_{k-2}(\lambda|j)>\sigma_{k-1}$.
  
Thus, \eqref{eqn.qk-i-bound} gives
\[
F_j\geq c\geq \frac{c}{n-k+1}\sum_{i=1}^nF_i.
\]

\medskip

\textit{Case 2. $F=\frac{\sigma_k}{\sigma_{k-2}}$}.
Since $\sigma_k=\sigma_{k-2}$ and $\sigma_{k-3}(\lambda|i)\geq0$, we have
\begin{equation}\label{eqn.Fj-trace-k-2}
\sum_{i=1}^nF_i
=\sum_{i=1}^n\frac{\sigma_{k-2}\sigma_{k-1}(\lambda|i)-\sigma_k\sigma_{k-3}(\lambda|i)}{\sigma_{k-2}^2}
\leq\frac{\sum_{i=1}^n\sigma_{k-1}(\lambda|i)}{\sigma_{k-2}}
=(n-k+1)\frac{\sigma_{k-1}}{\sigma_{k-2}}.
\end{equation}
Applying the lower bound in \eqref{eqn.qk-i-bound} to $q_k$ and $q_{k-1}$
and decreasing $c$ if necessary, we obtain
\[
(q_k)_j\geq c\left(\frac{\sigma_{k-1}(\lambda|j)}{\sigma_{k-1}}\right)^2,
\quad
(q_{k-1})_j\geq c\left(\frac{\sigma_{k-2}(\lambda|j)}{\sigma_{k-2}}\right)^2.
\]
  
Using $F=q_kq_{k-1}$ and $\sigma_k=\sigma_{k-2}$, we obtain
\begin{equation}\label{eqn.Fj-square-k-2}
F_j=q_{k-1}(q_k)_j+q_k(q_{k-1})_j
\geq c\frac{\sigma_{k-1}(\lambda|j)^2+\sigma_{k-2}(\lambda|j)^2}
{\sigma_{k-1}\sigma_{k-2}}.
\end{equation}
On the other hand, the Cauchy--Schwarz inequality gives
\[
\sigma_{k-1}^2
=\left(\sigma_{k-1}(\lambda|j)+\lambda_j\sigma_{k-2}(\lambda|j)\right)^2
\leq(1+\lambda_j^2)
\left(\sigma_{k-1}(\lambda|j)^2+\sigma_{k-2}(\lambda|j)^2\right).
\]
Combining this with \eqref{eqn.Fj-square-k-2} and
\eqref{eqn.Fj-trace-k-2}, we obtain
\[
F_j(1+\lambda_j^2)\geq c\frac{\sigma_{k-1}}{\sigma_{k-2}}
\geq\frac{c}{n-k+1}\sum_{i=1}^nF_i.
\]
Moreover, if $\lambda_j<0$, then $\sigma_{k-2}(\lambda|j)>0$ and hence $\sigma_{k-1}(\lambda|j)>\sigma_{k-1}$.
Thus, \eqref{eqn.Fj-square-k-2} and \eqref{eqn.Fj-trace-k-2} yield
\[
F_j\geq c\frac{\sigma_{k-1}(\lambda|j)^2}{\sigma_{k-1}\sigma_{k-2}}
\geq c\frac{\sigma_{k-1}}{\sigma_{k-2}}
\geq\frac{c}{n-k+1}\sum_{i=1}^nF_i.
\]
This completes the proof.

\label{lem.dynamic-concavity}
Let $1\leq\ell<k<n$, $k-\ell\in\left\{1,2\right\}$,
and let $\varepsilon_0=\varepsilon_0(n,k)\in(0,1)$ be sufficiently small.
  
Suppose that $\lambda\in\Gamma_k$ satisfies
\[

\quad
F(\lambda)=1.
\]
Then there exist $K_0=K_0(n,k)>1$ and $\delta_0=\delta_0(k)>0$ such that,
if
\[
\lambda_n\geq-\varepsilon_0\lambda_1
\quad\text{and}\quad
\lambda_1\geq K_0,
\]
then
\[
-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j
+\frac2{(1+\varepsilon_0)\lambda_1}\sum_{i=2}^n F_i\xi_i^2
\geq(1+\delta_0)\frac{F_1\xi_1^2}{\lambda_1}
\]
for any $\xi\in\mathbb R^n$ with $DF(\lambda)\cdot\xi=0$.

and let $\lambda\in\Gamma_k$ with $\lambda_1>\lambda_2\geq\cdots\geq\lambda_n$.

+2\sum_{i=2}^n\frac{(\sigma_k)_i\xi_i^2}{\lambda_1-\lambda_i}
\geq\gamma\frac{(\sigma_k)_1\xi_1^2}{\lambda_1}

<!-- lemma-5: body_md -->
\label{lem.singular-projection}
Let $2\leq k\leq n$
and let $\lambda=(1,\lambda_2,\ldots,\lambda_n)\in\overline{\Gamma_k}$ satisfy $\sigma_k(\lambda)=0$.
If $\partial_1\sigma_k(\lambda)=0$, then $\lambda\in\overline{\Gamma_n}$
and $\lambda$ has at most $k-1$ nonzero coordinates.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Set $\underline\lambda:=(\lambda|1)\in\mathbb R^{n-1}$.
Since $\lambda\in\overline{\Gamma_k}$, we have $\underline\lambda\in\overline{\Gamma_{k-1}}$.
The assumptions give
\[
0=\partial_1\sigma_k(\lambda)
=\sigma_{k-1}(\underline\lambda),
\quad
0=\sigma_k(\lambda)
=\sigma_{k-1}(\underline\lambda)+\sigma_k(\underline\lambda)
=\sigma_k(\underline\lambda).
\]
Thus
\begin{equation}\label{eqn.weighted-Newton}
\sum_{i=2}^n \underline\lambda_i^2\sigma_{k-2}(\underline\lambda| i)
=\sigma_1(\underline\lambda)\sigma_{k-1}(\underline\lambda)-k\sigma_k(\underline\lambda)=0.
\end{equation}
Each summand is nonnegative, so
$\underline\lambda_i\ne0$ implies $\sigma_{k-2}(\underline\lambda| i)=0$ for all $i\in\mathbb Z_{[2,n]}$.
If $k=2$, then $\underline\lambda=0$, and the conclusion follows.

Assume now that $k\geq3$.
Let $\mu=(\mu_1,\ldots,\mu_r)$ be the vector consisting of
all nonzero coordinates of $\underline\lambda$.
Suppose to the contrary that $r\geq k-1$.
  
Then $\sigma_{k-2}(\mu| i)=0$ for all $i\in\mathbb Z_{[1,r]}$,
and hence $\sigma_{k-2}(\mu)=0$.
  
Furthermore, $\sigma_{k-3}(\mu|i)=\frac{\sigma_{k-2}(\mu)-\sigma_{k-2}(\mu|i)}{\mu_i}=0$ for all $i\in\mathbb Z_{[1,r]}$,
and thus $\sigma_{k-3}(\mu)=0$.
  
Iteration leads to $\sigma_0(\mu|i)=0$, a contradiction.
Thus $r\leq k-2$,
and $\underline\lambda$ has at most $k-2$ nonzero coordinates.

Finally, since $\underline\lambda\in\overline{\Gamma_{k-1}}$ and $r\leq k-2$, we have
\[
\prod_{i=1}^r(t+\mu_i)
=\sum_{j=0}^r\sigma_j(\mu)t^{r-j}>0
\quad\text{for any}\ t>0.
\]
This polynomial has no positive root.
Since each $\mu_i$ is nonzero,
we have $\mu_i>0$ for all $i\in\mathbb Z_{[1,r]}$.
Together with $\lambda_1=1$, this shows that $\lambda\in\overline{\Gamma_n}$.

\label{lem.concavity-quotient}
Let $1\leq\ell<k\leq n-1$, $k-\ell\in\left\{1,2\right\}$ and $2k>n$.
Suppose that $\delta_0=\delta_0(k)>0$ is the constant given in Lemma \ref{lem.dynamic-concavity},
and $\delta=\delta(n,k)$ satisfies
\begin{equation}\label{eqn.delta-upper}
0<2\delta<\min\left\{2\delta_0,\frac{2k}{n}-1,\frac{2k-n}{2k^2+n}\right\}.
\end{equation}
Then there exists $K=K(n,k)>1$ such that,
for any $\lambda=(\lambda_i)\in\Gamma_k$ satisfying
\[
\lambda_1>\lambda_2\geq\cdots\geq\lambda_n,
\quad
F(\lambda)=1,
\quad
\lambda_1\geq K,
\]
we have
\begin{equation}\label{eqn.quotient-concavity}
-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j
+2\sum_{i=2}^n\frac{F_i\xi_i^2}{\lambda_1-\lambda_i}
\geq(1+\delta)\frac{F_1\xi_1^2}{\lambda_1}
\end{equation}
for every $\xi\in\mathbb R^n$ satisfying $DF(\lambda)\cdot\xi=0$.

Suppose to the contrary that
there exist sequences $\lambda^{\nu}\in\Gamma_k$ and $\xi^{\nu}\in\mathbb R^n$ such that
\begin{gather}
\lambda^\nu_1>\lambda^\nu_2\geq\cdots\geq\lambda^\nu_n,
\quad
F(\lambda^\nu)=1,
\quad
\lambda_1^\nu\geq\nu\to+\infty,
\quad
DF(\lambda^\nu)\cdot\xi^\nu=0,
\nonumber\\
-\partial_{\xi^\nu}^2F(\lambda^\nu)
+2\sum_{i=2}^n\frac{F_i(\lambda^\nu)(\xi^\nu_i)^2}{\lambda_1^\nu-\lambda_i^\nu}
<(1+\delta)\frac{F_1(\lambda^\nu)(\xi^\nu_1)^2}{\lambda_1^\nu}.
\label{eqn.concavity-contrary}
\end{gather}
  
By the concavity of $F^{1/(k-\ell)}$
and the homogeneity in $\xi^\nu$,
we may assume that $\xi_1^\nu=1$.
  
Set
\[
\varepsilon^\nu:=\frac{1}{\lambda^\nu_1},
\quad
\mu^\nu:=\varepsilon^\nu\lambda^\nu,
\quad
\eta^\nu:=(\varepsilon^\nu)^{k-\ell}.
\]
By homogeneity, we have
\[
\mu^\nu_1=1,
\quad
F(\mu^\nu)=\eta^\nu\to0,
\quad
DF(\mu^\nu)\cdot\xi^\nu=0.
\]
By Lemma \ref{eqn.lambda-i-lower}, it follows that
\begin{equation}\label{eqn.mu-i-lower-bound}
\mu^\nu_i
>-\frac{n-k}{k}\mu^\nu_1
=-\frac{n-k}{k}
\quad\text{for all}\ i\in\mathbb Z_{[1,n]}.
\end{equation}
After passing to a subsequence,
we may assume that $\mu^\nu\to\overline\mu\in\overline{\Gamma_k}$.
  
Since
\[
\sigma_k(\mu^\nu)=\eta^\nu\sigma_\ell(\mu^\nu)\to0,
\]
we have $\bar\mu_1=1$ and $\sigma_k(\bar\mu)=0$.
  
Next,
let $\varepsilon_0$ and $K_0=K_0(n,k)$ be as in Lemma \ref{lem.dynamic-concavity} for $F$.
We consider two cases separately.

\medskip

\textit{Case 1. $\overline\mu_n\geq-\varepsilon_0/2$.}\
For sufficiently large $\nu$,
we have $\lambda_n^{\nu}\geq-\varepsilon_0\lambda_1^{\nu}$ and $\lambda_1^\nu\geq K_0$.
In particular, $\lambda_1^{\nu}-\lambda_i^{\nu}\leq(1+\varepsilon_0)\lambda_1^{\nu}$ for all $i\in\mathbb Z_{[2,n]}$.
Thus, we can apply Lemma \ref{lem.dynamic-concavity} to get
\begin{align*}
-\partial_{\xi^\nu}^2F(\lambda^\nu)
+2\sum_{i=2}^n\frac{F_i(\lambda^\nu)(\xi^\nu_i)^2}{\lambda_1^\nu-\lambda_i^\nu}
&\geq-\partial_{\xi^\nu}^2F(\lambda^\nu)
+\frac2{(1+\varepsilon_0)\lambda_1^\nu}\sum_{i=2}^nF_i(\lambda^\nu)(\xi^\nu_i)^2
\\
&\geq(1+\delta_0)\frac{F_1(\lambda^\nu)(\xi^\nu_1)^2}{\lambda_1^\nu},
\end{align*}
which contradicts the inequality \eqref{eqn.concavity-contrary}.

\textit{Case 2. $\overline\mu_n<-\varepsilon_0/2$.}
Lemma \ref{lem.singular-projection} implies that $\partial_1\sigma_k(\overline\mu)>0$,
and hence $\sigma_{k-1}(\overline\mu)>0$, and $\overline\mu\in\Gamma_{k-1}$.
  
By the inequality \eqref{eqn.concavity-contrary},
the concavity of $F^{1/(k-\ell)}$ and $F_i(\lambda^\nu)\geq F_1(\lambda^\nu)>0$,
for any $i\in\mathbb Z_{[2,n]}$, we have
\[
2F_1(\lambda^\nu)\frac{(\xi^\nu_i)^2}{\lambda_1^\nu-\lambda_i^\nu}
\leq2\sum_{i=2}^n\frac{F_i(\lambda^\nu)(\xi^\nu_i)^2}{\lambda_1^\nu-\lambda_i^\nu}
<(1+\delta)\frac{F_1(\lambda^\nu)}{\lambda_1^\nu},
\]
and hence
\begin{equation}\label{eqn.xi-nu<}
\frac{(\xi^\nu_i)^2}{1-\mu_i^\nu}\leq\frac{1+\delta}{2}.
\end{equation}
Combining this with \eqref{eqn.mu-i-lower-bound}, we obtain
\begin{equation}\label{eqn.xi-nu<C}
|\xi^\nu|^2-1
=\sum_{i=2}^n(\xi^\nu_i)^2
\leq\sum_{i=2}^n\frac{1+\delta}{2}(1-\mu_i^\nu)
\leq\sum_{i=2}^n\frac{1+\delta}{2}\left(1+\frac{n-k}{k}\right)
\leq\frac{n(n-1)(1+\delta)}{2k}.
\end{equation}
  
Differentiating $\sigma_k=F\sigma_\ell$
and using $DF(\mu^\nu)\cdot\xi^\nu=0$, we obtain
\begin{align*}
\sigma_k(\mu^\nu)&=F(\mu^\nu)\sigma_\ell(\mu^\nu)=\eta^\nu\sigma_\ell(\mu^\nu),
\\
\sigma_\ell(\mu^\nu)F_i(\mu^\nu)
&=\partial_i\sigma_k(\mu^\nu)-\eta^\nu\partial_i\sigma_\ell(\mu^\nu),\
\\
\sigma_\ell(\mu^\nu)\partial_{\xi^\nu}^2F(\mu^\nu)
&=\partial_{\xi^\nu}^2\sigma_k(\mu^\nu)-\eta^\nu\partial_{\xi^\nu}^2\sigma_\ell(\mu^\nu),
\\
\partial_{\xi^\nu}\sigma_k(\mu^\nu)
&=\eta^\nu\partial_{\xi^\nu}\sigma_\ell(\mu^\nu).
\end{align*}
Since $\mu_1^\nu=1$ and $\sigma_k(\mu^\nu)\to0$,
Lemma \ref{lem.concavity-sigma-k} applies
with $\lambda=\mu^\nu$, $\xi=\xi^\nu$, and $\gamma=1+2\delta$
for all sufficiently large $\nu$.
  
Using homogeneity and combining this with \eqref{eqn.xi-nu<}, \eqref{eqn.xi-nu<C}
and $|\mu^\nu_i|\leq1$ for all $i\in Z_{[1,n]}$,
we obtain
\begin{align*}
0&>\sigma_\ell(\mu^\nu)\left(-\partial_{\xi^\nu}^2F(\mu^\nu)
+2\sum_{i=2}^n\frac{F_i(\mu^\nu)(\xi^\nu_i)^2}{1-\mu_i^{\nu}}
-(1+\delta)F_1(\mu^\nu)\right)
\\
&=-\partial_{\xi^\nu}^2\sigma_k(\mu^\nu)+\eta^\nu\partial_{\xi^\nu}^2\sigma_\ell(\mu^\nu)
+2\sum_{i=2}^n\frac{\left(\partial_i\sigma_k(\mu^\nu)-\eta^\nu\partial_i\sigma_\ell(\mu^\nu)\right)(\xi^\nu_i)^2}{1-\mu_i^{\nu}}
\\
&\quad\,
-(1+\delta)\left(\partial_1\sigma_k(\mu^\nu)-\eta^\nu\partial_1\sigma_\ell(\mu^\nu)\right)
\\
&=-\partial_{\xi^\nu}^2\sigma_k(\mu^\nu)
+2\sum_{i=2}^n\frac{\partial_i\sigma_k(\mu^\nu)(\xi^\nu_i)^2}{1-\mu_i^{\nu}}
-(1+2\delta)\partial_1\sigma_k(\mu^\nu)+\delta\partial_1\sigma_k(\mu^\nu)
\\
&\quad\,
+\eta^\nu\left(
\partial_{\xi^\nu}^2\sigma_\ell(\mu^\nu)
-2\sum_{i=2}^n\frac{\partial_i\sigma_\ell(\mu^\nu)(\xi^\nu_i)^2}{1-\mu_i^{\nu}}
+(1+\delta)\partial_1\sigma_\ell(\mu^\nu)
\right)
\\
&\geq-\frac{2}{\sigma_k(\mu^\nu)}(\partial_{\xi^\nu}\sigma_k(\mu^\nu))^2
+\delta\partial_1\sigma_k(\mu^\nu)-C(n,k)\eta^\nu
\\
&=\delta\partial_1\sigma_k(\mu^\nu)
-2\eta^\nu\frac{(\partial_{\xi^\nu}\sigma_\ell(\mu^\nu))^2}{\sigma_\ell(\mu^\nu)}-C(n,k)\eta^\nu
\\
&\geq\frac{\delta}{2}\partial_1\sigma_k(\overline\mu)
>0
\end{align*}
for all large $\nu$.
Here we used $\eta^\nu\to0$,
the boundedness of $\xi^\nu$,
$\sigma_\ell(\mu^\nu)\to\sigma_\ell(\overline\mu)>0$
and $\partial_1\sigma_k(\mu^\nu)\to\partial_1\sigma_k(\overline\mu)>0$.
This contradiction completes the proof.

\label{lem.Jacobi}
Suppose that $\delta=\delta(n,k)\in(0,1)$ and $K=K(n,k)$
are the constants given in Lemma \ref{lem.concavity-quotient}.
  
Assume that $u\in C^4(B_1)$ is a $k$-convex solution of \eqref{eqn.sigma-quotient} in $B_1$.
  
Let $b(x):=\log\lambda_{\max}(D^2u(x))$ and $a(x):=\lambda_{\max}(D^2u(x))^\delta$.
Then
\begin{equation}\label{eqn.jacobi}
F^{ij}b_{ij}\geq\delta F^{ij}b_i b_j
\quad\text{and}\quad
F^{ij}a_{ij}
\geq2a^{-1}F^{ij}a_i a_j
\end{equation}
hold in the viscosity sense
at every point $x\in B_1$ satisfying $\lambda_{\max}(D^2u)\geq K$.

Fix $x_0\in B_1$ such that $\lambda_{\max}(D^2u(x_0))\geq K$.
After a rotation of coordinates, we may assume that
\[
D^2u(x_0)=\operatorname{diag}(\lambda_1,\ldots,\lambda_n),
\quad
\lambda_1=\cdots=\lambda_m=\lambda_{\max}>
\lambda_{m+1}\geq\cdots\geq\lambda_n.
\]
All quantities below are evaluated at $x_0$.
  
By \cite[Lemma 5]{BCD17}, it follows that
\begin{gather}
\delta_{pq}(\lambda_1)_i=u_{pqi}
\quad\text{for all}\
p,q\in\mathbb Z_{[1,m]},\ i\in\mathbb Z_{[1,n]}
\label{eqn.lambda-max-derivative}
\\
(\lambda_1)_{ii}
\geq u_{11ii}+2\sum_{p>m}\frac{u_{1pi}^2}{\lambda_1-\lambda_p}
\quad\text{for all}\ i\in\mathbb Z_{[1,n]}
\label{eqn.lambda-max-second}
\end{gather}
in the viscosity sense.
Moreover, we can compute
\begin{gather*}
u_{11i}=\lambda_1b_i\quad\text{for all}\ i\in\mathbb Z_{[m+1,n]},\
\\
b_{ii}
\geq\frac{u_{11ii}}{\lambda_1}
+2\sum_{p>m}\frac{u_{1pi}^2}{\lambda_1(\lambda_1-\lambda_p)}-\frac{u_{11i}^2}{\lambda_1^2}
\quad\
\text{for all}\ i\in\mathbb Z_{[1,n]}
\end{gather*}
in the viscosity sense.
Hence
\begin{equation}\label{eqn.Fij-bij}
F^{ij}b_{ij}
\geq\frac{1}{\lambda_1}\sum_{i=1}^nF_i u_{11ii}
+2\sum_{i=1}^n\sum_{p>m}\frac{F_i u_{1pi}^2}{\lambda_1(\lambda_1-\lambda_p)}
-\sum_{i=1}^n\frac{F_i u_{11i}^2}{\lambda_1^2}.
\end{equation}
  
Set $\xi_i:=u_{ii1}$ for all $i\in\mathbb Z_{[1,n]}$.
  
Differentiating the equation $F(D^2u)=1$, we have
\[
DF\cdot\xi=\sum_{i=1}^nF_iu_{ii1}=\partial_{x_1}F=0.
\]
Differentiating the equation $F(D^2u)=1$ twice in the $x_1$-direction,
and using the concavity of $F^{1/(k-\ell)}$
and the standard second-derivative formula for symmetric functions
of the eigenvalues (see \cite[Lemma 2.2]{LT26}), we obtain
\begin{align}
\sum_{i=1}^nF_i u_{ii11}
&=-\sum_{p,q,r,s=1}^nF^{pq,rs}u_{pq1}u_{rs1}
=-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j-\sum_{i\neq j}F^{ij,ji}u_{ij1}^2
\nonumber\\
&\geq-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j+2\sum_{p>m}\frac{F_p-F_1}{\lambda_1-\lambda_p}u_{11p}^2.
\label{eqn.Fi-uii11}
\end{align}
  
On the other hand,
\begin{equation}\label{eqn.third-derivative-lower-jacobi}
2\sum_{i=1}^n\sum_{p>m}
\frac{F_i u_{1pi}^2}{\lambda_1(\lambda_1-\lambda_p)}
\geq2\sum_{p>m}\frac{F_1u_{11p}^2+F_pu_{1pp}^2}{\lambda_1(\lambda_1-\lambda_p)}
=2\sum_{p>m}\frac{F_1u_{11p}^2+F_p\xi_p^2}{\lambda_1(\lambda_1-\lambda_p)}.
\end{equation}
Substituting
\eqref{eqn.Fi-uii11} and \eqref{eqn.third-derivative-lower-jacobi}
into \eqref{eqn.Fij-bij}, we find
\begin{align*}
F^{ij}b_{ij}
&\geq\frac{1}{\lambda_1}
\left(-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j+2\sum_{p>m}\frac{F_p-F_1}{\lambda_1-\lambda_p}u_{11p}^2\right)
+2\sum_{p>m}\frac{F_1u_{11p}^2+F_p\xi_p^2}{\lambda_1(\lambda_1-\lambda_p)}
-\sum_{i=1}^n\frac{F_i u_{11i}^2}{\lambda_1^2}
\\
&=\frac{1}{\lambda_1}
\left(-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j+\sum_{p>m}\frac{2F_p\xi_p^2}{\lambda_1-\lambda_p}\right)
+2\sum_{p>m}\frac{F_p u_{11p}^2}{\lambda_1(\lambda_1-\lambda_p)}-\sum_{i=1}^n\frac{F_i u_{11i}^2}{\lambda_1^2}.
\end{align*}
Next, we consider two cases for $m$.

\textit{Case 1. $m>1$.}\
By \eqref{eqn.lambda-max-derivative}, it follows that
\begin{gather*}
\xi_1=u_{111}=u_{221}=u_{122}=\delta_{12}(\lambda_1)_2=0,
\\
\xi_i=u_{ii1}=u_{1ii}=\delta_{1i}(\lambda_1)_i=0\quad\text{for all}\ i\in\mathbb Z_{[2,m]},
\\
b_i=\frac{u_{11i}}{\lambda_1}=\frac{u_{1i1}}{\lambda_1}=\frac{\delta_{1i}(\lambda_1)_1}{\lambda_1}=0
\quad\text{for all}\ i\in\mathbb Z_{[1,m]}.
\end{gather*}
Since $(\sigma_k/\sigma_{k-2})^{1/2}$ is concave, we have
\[
\partial_\xi^2F(\lambda)
=2F^{1/2}\partial_\xi^2(F^{1/2})
+\frac{(DF\cdot\xi)^2}{2F}
\leq0
\quad\text{for}\ F=\frac{\sigma_k}{\sigma_{k-2}}.
\]
Moreover, $\sigma_k/\sigma_{k-1}$ is also concave.
Thus $-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j\geq0$ and therefore
\begin{align*}
F^{ij}b_{ij}
&\geq2\sum_{p>m}\frac{F_p u_{11p}^2}{\lambda_1(\lambda_1-\lambda_p)}-\sum_{i>m}\frac{F_i u_{11i}^2}{\lambda_1^2}
=\sum_{i>m}\left(\frac{2\lambda_1}{\lambda_1-\lambda_i}-1\right)F_ib_i^2
\\
&\geq\sum_{i>m}\left(\frac{2\lambda_1}{\lambda_1+(n-k)\lambda_1/k}-1\right)F_ib_i^2
=\left(\frac{2k}{n}-1\right)\sum_{i,j=1}^nF^{ij}b_ib_j
\\
&\geq\delta\sum_{i,j=1}^nF^{ij}b_ib_j,
\end{align*}
where the second inequality follows from $\lambda_i\geq-(n-k)\lambda_1/k$ for all $i\in\mathbb Z_{[2,n]}$.


\textit{Case 2. $m=1$.}\
By \eqref{eqn.lambda-max-derivative},
it follows that $\xi_1=u_{111}=(\lambda_1)_1=\lambda_1b_1$.
Since $\lambda_1\geq K$,
the concavity inequality (Lemma \ref{lem.concavity-quotient}) gives
\[
-\sum_{i,j=1}^nF_{ij}\xi_i\xi_j
+\sum_{i>m}\frac{2F_i\xi_i^2}{\lambda_1-\lambda_i}
\geq(1+\delta)\frac{F_1\xi_1^2}{\lambda_1}.
\]
Consequently,
\begin{align*}
F^{ij}b_{ij}
&\geq(1+\delta)F_1b_1^2
+2\sum_{i>1}\frac{\lambda_1F_i}{\lambda_1-\lambda_i}b_i^2
-\sum_{i=1}^nF_i b_i^2
\\
&=\delta F_1b_1^2+\sum_{i>1}\left(\frac{2\lambda_1}{\lambda_1-\lambda_i}-1\right)F_i b_i^2
\geq\delta \sum_{i,j=1}^nF^{ij}b_ib_j.
\end{align*}

Finally, since $a=e^{\delta b}$, we have
\[
F^{ij}a_{ij}
=\delta a F^{ij}b_{ij}+\delta^2aF^{ij}b_ib_j
\geq2\delta^2aF^{ij}b_i b_j
=2a^{-1}F^{ij}a_i a_j.
\]
This completes the proof.

Suppose that $u\in C^4(B_1)$ is a $k$-convex solution of $F(D^2u)=1$ in $B_1$.

Suppose that $\delta=\delta(n,k)>0$ and $K=K(n,k)>1$ are the constants in Lemma \ref{lem.Jacobi}.
  
Fix $y\in B_{1/2}$ and $r\in(0,1/8)$, so that $B_{4r}(y)\Subset B_1$.

\[
\varphi_y(x):=(x-y)\cdot Du(x)-u(x)+u(y)+\frac{\alpha}{2}|x-y|^2-\beta\left\vertDu\right\vert^2,
\]
where $\alpha=\alpha(n,k,r,M)\geq1$ and $\beta=\beta(n,r,M)>0$ will be chosen later.
Let $M:=\|u\|_{C^1(B_1)}+1$.
The gradient bound gives
\begin{align*}
\left\vert(x-y)\cdot Du(x)-u(x)+u(y)\right\vert\leq2M|x-y|\quad\ \text{for any}\ x\in B_1.
\end{align*}

\begin{gather*}
\varphi_y\geq-8Mr-\beta M^2\quad\text{in}\ B_{4r}(y),\quad
a_y:=\sup_{\overline{B_{2r}(y)}}\varphi_y
\leq4Mr+2\alpha r^2.
\end{gather*}

\[
\inf_{\partial B_{4r}(y)}\varphi_y
\geq-8Mr+\frac{\alpha}{2}(4r)^2-\beta M^2
>4Mr+2\alpha r^2+2
\geq a_y+2=:c_y.
\]
Then
\[
\varphi_y<c_y
\quad\text{in}\ \overline{B_{2r}(y)},
\quad
\varphi_y>c_y
\quad\text{on}\ \partial B_{4r}(y).
\]

$\{x\in B_{4r}(y):\varphi_y(x)<c_y\}$
containing $B_{2r}(y)$ satisfies
\begin{align*}
\overline{B_{2r}(y)}\subset\Omega_y\Subset B_{4r}(y).
\end{align*}

\psi_y:=e^{(c_y-\varphi_y)/\gamma}-1

Next, we distinguish two cases.

\textit{Case 1. $\lambda_j\geq0$}.
By \eqref{eqn.zj-bound},
the two terms $z_j\lambda_j$ and $\alpha(x_j-y_j)$ have the same sign.
By $\alpha\geq1$ and Lemma \ref{lem.Fj-lambdaj},
there exists $c_1=c_1(n,k)>0$ such that
\[
Q
\geq\frac{F_j}{\gamma}(\varphi_y)_j^2
\geq\frac{F_j}{\gamma}\left(\frac{7}{8}\lambda_j+\alpha\right)^2|x_j-y_j|^2
\geq\frac{49r^2}{64n\gamma}F_j(1+\lambda_j^2)
\geq\frac{49r^2c_1}{64n\gamma}\sum_{i=1}^nF_i.
\]
We may choose $\gamma<\frac{49r^2c_1}{64n\alpha_1}$
such that $Q>\alpha_1\sum_{i=1}^nF_i$,
a contradiction to \eqref{eqn.Q<Fi}.

\textit{Case 2. $\lambda_j<0$}.
Take $\gamma$ with $\gamma\leq\frac{r^2}{2\beta n}=\frac{8Mr}{\sqrt n}$.
Then $2\beta\gamma\leq(x_j-y_j)^2$.
By Lemma \ref{lem.Fj-lambdaj},
there exists $c_2=c_2(n,k)>0$ such that
\begin{align*}
Q
\geq\frac{1}{\gamma}F_j(\varphi_y)_j^2+2\beta F_j\lambda_j^2
\geq\frac{2\beta\alpha^2(x_j-y_j)^2}{z_j^2+2\beta\gamma}F_j
\geq\frac{1}{2}\beta\alpha^2F_j
\geq\frac{1}{2}\beta\alpha^2 c_2\sum_{i=1}^nF_i.
\end{align*}
Choose $\alpha\geq\max\left\{1,c_*,\frac{8}{\beta c_2}\right\}$.
Then $Q>\alpha_1\sum_{i=1}^nF_i$, a contradiction to \eqref{eqn.Q<Fi}.
The claim is proved.

Now, using the lower bound \eqref{eqn.psi-lower} of $\psi_y$, we have
\begin{align*}
c_0\max_{\overline{B_{2r(y)}}}a
\leq\max_{\overline{B_{2r(y)}}}a\psi_y
\leq\max_{\overline{\Omega_y}}a\psi_y
\leq a(x_0)\psi_y(x_0).
\end{align*}
If $x_0\in\overline{B_r(y)}$,
then by the upper bound \eqref{eqn.psi-upper} of $\psi_y$,
we have $a(x_0)\psi_y(x_0)\leq C_0\max_{\overline{B_r(y)}}a$.
If $\lambda_{\max}(x_0)\leq K$, then $a(x_0)\psi_y(x_0)\leq C_0K^\delta$.
Thus,
\[
\max_{\overline{B_{2r}(y)}}a\leq\frac{C_0}{c_0}\max\left\{\max_{\overline{B_r(y)}}a,K^\delta\right\}.
\]
Furthermore, since
\[
1=F(\lambda)\leq F(\lambda_1\mathds 1)=\lambda_{\max}^{k-\ell} F(\mathds1),
\]
we have $\lambda_{\max}\geq c_*>0$.
Thus
\[
\sup_{B_{2r}(y)}\lambda_{\max}
\leq\left(\frac{C_0}{c_0}\right)^{1/\delta}\max\left\{1,\frac{K}{c_*}\right\}\sup_{B_r(y)}\lambda_{\max},
\]
which proves the desired estimate.

\label{lemma.alexandrov-quotient}
Let $1\leq\ell<k\leq n$ and let $\Omega\subset\mathbb R^n$ be open.
Suppose that $u_j\in C^3(\Omega)$ are $k$-convex solutions 
of $\frac{\sigma_k(D^2u_j)}{\sigma_\ell(D^2u_j)}=1$ in $\Omega$,
and that $u_j$ converges locally uniformly to $u$ in $\Omega$.
Then $u$ is twice differentiable almost everywhere in $\Omega$;
that is, for almost every $x\in\Omega$, 
there is a quadratic polynomial $P$ such that
\[
\sup_{y\in B_r(x)}|u(y)-P(y)|=o(r^2).
\]

By Chen's gradient estimate \cite[Theorem 1.1]{C15}, we have
\[
|Du_j(z)|\leq C(n,k,\ell)\frac{\operatorname{osc}_{B_\rho(z)}u_j}{\rho}\quad
\text{whenever}\ B_\rho(z)\Subset\Omega.
\]
Since $u_j$ converges locally uniformly to $u$,
this estimate gives locally uniform bounds for $Du_j$.
Hence $u$ is locally Lipschitz. 
Moreover, by Rademacher's theorem,
$u$ is differentiable almost everywhere in $\Omega$.
  
Following the interpolation argument in \cite[Lemma 2.6]{TW99} 
leading to \cite[Corollary 3.4]{TW99},
we obtain the weighted Lipschitz estimate
\[
\sup_{\substack{x,y\in B_r(z)\\x\neq y}}
d_{x,y}^{n+1}\frac{|u_j(x)-u_j(y)|}{|x-y|}
\leq C(n,k,\ell)\int_{B_r(z)}|u_j|
\]
for every ball $B_r(z)\Subset\Omega$,
where $d_{x,y}:=\min\{d_x,d_y\}$ and $d_x:=\operatorname{dist}(x,\partial B_r(z))$.
Passing to the locally uniform limit gives
\begin{equation}\label{eqn.alexandrov-lip}
\sup_{\substack{x,y\in B_r(z)\\x\neq y}}
d_{x,y}^{n+1}\frac{|u(x)-u(y)|}{|x-y|}
\leq C(n,k,\ell)\int_{B_r(z)}|u|.
\end{equation}
The estimate \eqref{eqn.alexandrov-lip} also holds for $u-L$
with the same constant $C(n,k,\ell)$ for every affine function $L$,
since subtracting $L$ leaves equation \eqref{eqn.sigma-quotient} unchanged.
  
Since $k\geq2$, the limit $u$ is $2$-convex in the viscosity sense.
By the argument of \cite[Theorem 2.4]{CT05},
its distributional Hessian $[D^2u]=[\mu^{ij}]$
is a symmetric matrix of locally finite signed Radon measures:
\[
\int_\Omega u\,\varphi_{ij}\,dx
=\int_\Omega\varphi\,d\mu^{ij}
\quad\text{for all }\varphi\in C_0^\infty(\Omega).
\]
  
By Lebesgue differentiation, the conditions in
\cite[(4.2)--(4.4)]{SY25} hold almost everywhere in $\Omega$.
Thus \cite[Lemma 4.1]{SY25} gives, for almost every $x\in\Omega$,
a quadratic polynomial $P$ such that
\[
\frac{1}{|B_r|}\int_{B_r(x)}|u(y)-P(y)|\,dy=o(r^2).
\]
Using \eqref{eqn.alexandrov-lip}
and arguing as in \cite[proof of (4.6)]{SY25},
we obtain the corresponding estimate for the remainder $u-P$.
Finally, \cite[Lemma 4.2]{SY25} yields
\[
\sup_{y\in B_{r/2}(x)}|u(y)-P(y)|=o(r^2).
\]
Replacing $r$ by $2r$ completes the proof.

Suppose to the contrary that
there exists a sequence of
$k$-convex solutions $u_j\in C^4(B_1)$ of equation \eqref{eqn.sigma-quotient} 
such that $\left\Vertu_j\right\Vert_{C^1(B_1)}\leq A$ and $\left\vertD^2u_j(0)\right\vert\to+\infty$.
  
By the Arzel\`a--Ascoli theorem
and the closedness of viscosity solutions (cf. \cite[Lemma 6.1]{SY25}),
after passing to a subsequence,
$u_j$ converges uniformly to
a continuous viscosity solution $u$ of $F(D^2u)=1$ in $B_1$.
  
Moreover, since each $u_j$ is $k$-convex,
the limit $u$ is $k$-convex in the viscosity sense.
  
By Lemma \ref{lemma.alexandrov-quotient},
$u$ is twice differentiable almost everywhere in $B_1$.
Fix such a point $y\in B_{1/8}$ and let $P$ be a quadratic polynomial
such that
\[
u(y+z)=P(y+z)+o(|z|^2).
\]
Note that $\lambda(D^2P)\in\Gamma_k$ and $F(D^2P)=1$.

\medskip

We next apply Savin's small perturbation theorem \cite{Sav07} to $v_j:=u_j-P$.
For $0<r<1/8$, we rescale near $y$ by
\[
v_{j,r}(z)
:=\frac{1}{r^2}\left(u_j(y+rz)-P(y+rz)\right)
\quad\text{for any}\ z\in B_1.
\]
Then
\begin{align*}
\left\Vertv_{j,r}\right\Vert_{L^\infty(B_1)}
&\leq\frac{1}{r^2}\left(\left\Vertu_j-u\right\Vert_{L^\infty(B_r(y))}+\left\Vertu-P\right\Vert_{L^\infty(B_r(y))}\right)
\\
&\leq\frac{1}{r^2}\left\Vertu_j-u\right\Vert_{L^\infty(B_r(y))}+\varepsilon(r),
\end{align*}
for some modulus $\varepsilon(r):=o(r^2)/r^2$.
Define a continuous operator on the space of real symmetric matrices by
\[
G(M):=
\begin{cases}
F(D^2P+M)-1, & \lambda(D^2P+M)\in\Gamma_k,\\
-1, & \text{otherwise}.
\end{cases}
\]
The operator $G$ is degenerate elliptic, namely,
\[
G(M+N)\geq G(M)
\quad\text{for any}\ M\in\operatorname{Sym(n)},\ N\geq0.
\]
Moreover, $G(0)=0$, and $G$ is smooth and uniformly elliptic
in a neighborhood of $0$.
Since $\lambda(D^2P+D^2v_{j,r})\in\Gamma_k$,
we have
\[
G(D^2v_{j,r})=F(D^2P+D^2v_{j,r})-1=0
\quad\text{in}\ B_1.
\]
  
Fix $r=r(n,k,P,\varepsilon)=:\rho\in(0,1/8)$ sufficiently small that $\varepsilon(\rho)<c_1/2$,
where $c_1$ is the small constant in \cite[Theorem 1.3]{Sav07}.
Since $u_j$ converges uniformly to $u$,
$\|v_{j,\rho}\|_{L^\infty(B_1)}\leq c_1$
for all sufficiently large $j$.
Applying \cite[Theorem 1.3]{Sav07} and returning to the original variables, we obtain
\[
\left\Vertu_j-P\right\Vert_{C^{2,\alpha}(B_{\rho/2}(y))}
\leq C(n,k,P,\varepsilon),
\]
for all sufficiently large $j$,
where $\alpha=\alpha(n,k,P,\varepsilon)\in(0,1)$.
This implies that
\[
\lambda_{\max}(D^2(u_j))\leq C(n,k,P,\varepsilon)=:C_0
\quad\text{in}\ B_{\rho/2}(y).
\]

\medskip

Finally, choose points $y_0:=y$, $y_1,y_2,\dots,y_{N-1}$, $y_N:=0$
on the line segment from $y$ to $0$
such that $\left\verty_i-y_{i-1}\right\vert\leq\rho/2$ for all $i\in\mathbb Z_{[1,N]}$
and $N\leq1+\frac{1}{4\rho}$.
Then
\[
\left\{y_i\right\}_{i=0}^N\subset B_{1/4},
\quad
B_{\rho/2}(y_i)\subset B_\rho(y_{i-1})
\ \text{for all}\ i\in\mathbb Z_{[1,N]}.
\]
Repeated application of the doubling inequality (Lemma \ref{lem.doubling}) to $u_j$  gives
\[
\begin{aligned}
\lambda_{\max}(D^2u_j(0))
&\leq
\sup_{B_{\rho/2}(y_N)}\lambda_{\max}(D^2u_j)
\\
&\leq
C\left(n,k,A,\rho\right)^N
\sup_{B_{\rho/2}(y)}\lambda_{\max}(D^2u_j)
\\
&\leq C\left(n,k,A,\rho\right)C_0
\end{aligned}
\]
for all sufficiently large $j$.
  
Since $\Gamma_k\subseteq\Gamma_2$, we have
\[
\left\vertD^2u_j\right\vert^2
=(\Delta u_j)^2-2\sigma_2(D^2u_j)
<(\Delta u_j)^2\leq (n\lambda_{\max}(D^2u_j))^2.
\]
Consequently,
\[
\left\vertD^2u_j(0)\right\vert
\leq nC(n,k,A,\rho)C_0
\]
for all sufficiently large $j$,
contradicting $\left\vertD^2u_j(0)\right\vert\rightarrow+\infty$.
This completes the proof.
