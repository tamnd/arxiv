<!-- theorem-1: body_md -->
\label{theorem: main}
    For any $N \ge 0$ holds.
    \begin{enumerate}
        \item[(i)] Let $L := \lbrace k_0,\dots,k_N,k_{N+1} \rbrace$. Then  Dubrovin--Frobenius manifold product and pairing of ${\mathcal H}_{0; L}$ extend to the locus $t_{N+1,1} = \dots = t_{N+1,k_{N+1}} = 0$.
        \item[(ii)] Let $k_{N+1} \ge 2$. Then the product over this locus coincides with the open extension of the Hurwitz-Frobenius manifold ${\mathcal H}_{0; K}$ for $K = \lbrace k_0,\dots,k_N \rbrace$.

        Respective open WDVV solution is given by the pair
        \[
            (F^c, F^o) := \left( {\mathcal F}^K, \frac{{\partial} {\mathcal F}^L}{{\partial} t_{N+1, 1} } \mid_{t_{N+1,1} = \dots = t_{N+1,k_{N+1}} = 0} \right).
        \]
    \end{enumerate}

\label{prop: product via lambda}
  Let $X,Y,Z \in {\mathcal T}_M$ be such that
   \[
      (X \cdot \lambda)(Y \cdot \lambda) - (Z \cdot \lambda) = \frac{{\partial} \lambda}{{\partial} z} \phi(z)
   \]
    assumed as the equality of rational functions, where $\phi(z)$ is a rational function such that $\sum_{p \in \mathcal{R}} {\mathrm{res}}_{z=p} (\phi(z)dz) = 0$. 


    Then $X \circ Y = Z$.

<!-- proposition-2: body_md -->
\label{prop: SpecAn}
    The ${\mathcal O}_M$--algebra $({\mathcal T}_M,\circ)$ of ${\mathcal H}_{0; K}$ is isomorphic to
    \[
         {\mathcal A}_K := {\mathbb{C}}[z, w_1,\dots,w_N, y_1,\dots,y_N] \otimes {\mathcal O}_M / {\mathcal I}
    \]
    where ${\mathcal I}$ is the ideal generated by
    \begin{align}
        & P_{1,i} := y_i z - 1 - v_{i,0} y_i, \ P_{2,ij} := y_iy_j - \frac{y_i - y_j}{v_{i,0} - v_{j,0}}, \ P_{3,i} := w_i - \sum_{a=1}^{k_i} a v_{i,a} y_i^{a+1},
        \label{eq: specan-1}
        \\
        & P_{4,i} := w_i z - w_i v_{i,0} - \sum_{a=1}^{k_i} av_{i,a} y_i^{a},
        \
        P_{5} := z^{k_0-1} - \sum_{a=1}^{k_0-1} (a-1) v_{0,a} z^{a-2} - \sum_{i=1}^N w_i.
        \label{eq: specan-2}
    \end{align}
  The isomorphism is given by
    \[
        \Psi: y_i \mapsto \frac{{\partial} }{{\partial} v_{i,1}}, \ w_i \mapsto \frac{{\partial} }{{\partial} v_{i,0}}, \ z \mapsto \frac{{\partial} }{{\partial} v_{0,2}}.
    \]
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
In this proof we apply Proposition~\ref{prop: product via lambda} multiple times.
    First of all note that $\frac{{\partial} }{{\partial} v_{0,1}}$ is the unit of $\circ$ and
    \[
        \frac{{\partial} \lambda}{{\partial} v_{i,1}} = \frac{1}{z - v_{i,0}},
        \
        \frac{{\partial} \lambda}{{\partial} v_{0,2}} = z.
    \]
    We have $\frac{{\partial} }{{\partial} v_{i,1}} \circ \frac{{\partial} }{{\partial} v_{0,2}} = \frac{{\partial} }{{\partial} v_{0,1}} +  v_{i,0} \frac{{\partial} }{{\partial} v_{i,1}}$ what gives the generator $P_{1,i}$.

    Similarly the second generator $P_{2,ij}$ originates from the simple equality
    \[
        \frac{1}{z - v_{i,0}}\frac{1}{z - v_{j,0}} = \frac{1}{v_{i,0} - v_{j,0}} \left( \frac{1}{z - v_{i,0}} - \frac{1}{z - v_{j,0}}\right).
    \]
    Next we have
    \[
     \frac{{\partial} \lambda}{{\partial} v_{i,0}} = \sum_{a=1}^{k_i} \frac{a v_{i,a}}{ (z-v_{i,0})^{a+1}}
    \]
    what gives the generators $P_{3,i}, P_{4,i}$.

    At the same time we have
    \[
        \frac{{\partial} \lambda}{{\partial} z} = z^{k_0-1} + \sum_{a=1}^{k_i-1} (a-1) v_{0,a} z^{a-2} - \sum_{i=1}^N \frac{{\partial} \lambda}{{\partial} v_{i,0}}.
    \]
    This gives the generator $P_5$ by Proposition~\ref{prop: product via lambda} again.

    In order to complete the proof we need to show that these generators compute any product of ${\mathcal H}_{0;K}$ after the map $\Psi$.

    We show that the quotient-ring above has the ${\mathcal O}_M$--basis
    \begin{equation}\label{eq: AK basis}
        [1],[z],\dots,[z^{k_0-2}], \ [w_i] \ [y_i],\dots,[y_i^{k_i}], \ i=1,\dots, N.
    \end{equation}
    First of all note that $P_{1,i}$ and $P_{4,i}$ allows us to express any product $[z^a y_i]$ and $[z^ax_i]$ via the basis elements above.

    \begin{lemma}\label{lemma_basis}
        For any $1 \le i,j \le N$ we can express $[w_iy_j]$ and $[w_iw_j]$ via basis elements above using the generators of ${\mathcal I}$.
    \end{lemma}
    \begin{proof}
        Let $i \neq j$.
        Consider
        \[
            P_{3,i} y_j = w_iy_j - \sum_{a=1}^{k_i} a v_{i,a} y_i^{a+1}y_j.
        \]
        Applying repeatedly the generator $P_{2,ij}$ we get the relation on $[w_iy_j]$ via the basis elements outlined above.

        In order to express $[w_iy_i]$ consider $P_5y_i$. The products $[z^{a-2}y_i]$ are simplified via the generator $P_{1,i}$ and the products $[w_iy_j]$ with $i \neq j$ by the reasoning above. This gives $[w_iy_i]$.

        The products $[w_iw_j]$ are now resolved with the help of the products computed above and the third generator $P_{3,i}$.
    \end{proof}

    To complete the proposition it remains to show that $[z^{a}]$ with $a \ge k_0-1$ and $[y_i^a]$ with $a \ge k_i+1$ are all expressible via the basis elements above.

    This is given by the generators $P_{3,i}$ and $P_5$.

\label{theorem: HFM}
    In the coordinates $t_\bullet$ above the three--point function and the pairing $\eta$ define the structure of a Dubrovin--Frobenius manifold on ${\mathcal H}_{0;K}$.

    The only non-zero pairings in the flat frame are
    \begin{align*}
        & \eta(\frac{{\partial}}{{\partial} t_{0,\alpha}},\frac{{\partial}}{{\partial} t_{0,k_0-\alpha}}) = 1, && 1 \le \alpha \le k_0-1,
        \\
        &\eta(\frac{{\partial}}{{\partial} t_{i,0}},\frac{{\partial}}{{\partial} t_{i,1}}) = 1, \
        \eta(\frac{{\partial}}{{\partial} t_{i,\alpha}},\frac{{\partial}}{{\partial} t_{i,k_i+1-\alpha}}) = \frac{1}{k_i}, && 2 \le \alpha \le k_i, \ i \ge 1.
    \end{align*}

    The three--point functions (cf. Eq.~\eqref{eq: three-point function}) are integrated by the potential ${\mathcal F}^K$
    \[
        \frac{{\partial}^3 {\mathcal F}^K}{{\partial} t_{i,\alpha} {\partial} t_{j,\beta} {\partial} t_{r,\gamma}} = \left\langle \frac{{\partial}}{{\partial} t_{i,\alpha}},\frac{{\partial}}{{\partial} t_{j,\beta}},\frac{{\partial}}{{\partial} t_{r,\gamma}} \right \rangle.
    \]

     The potential ${\mathcal F}^K$ is subject to the quasihomogeneity condition $E \cdot {\mathcal F}^K = 2(1 + \frac{1}{k_0}) {\mathcal F}^K$ modulo the quadratic terms. The Euler vector field $E$ reads
     \[
         E = \sum_{\alpha=1}^{k_0-1}
         \left( \frac{k_0 + 1}{k_0} - \frac{\alpha}{k_0} \right)
         t_{0,\alpha} \frac{{\partial}}{{\partial} t_{0,\alpha}} +  \sum_{i=1}^N \left( \frac{1}{k_0}t_{i,0} \frac{{\partial}}{{\partial} t_{i,0}}+\sum_{\alpha=1}^{k_i}
         \left(  \frac{k_0 + 1}{k_0} - \frac{\alpha-1}{k_i} \right) t_{i,\alpha} \frac{{\partial} }{t_{i,\alpha}}  \right) 
     \]

    In the proof of Proposition~\ref{prop: SpecAn} we gave the procedure how to compute multiplication table in the basis~\eqref{eq: AK basis}. With our choice of the basis all coefficients $v_\bullet$ were coming to the multiplication table polynomially expect $v_{i,k_i}$ and $v_{i,0} - v_{j,0}$ --- first due to generator $P_{3,i}$ and second because of generator $P_{2,ij}$. We should take care of all applications of the generator $P_{3,i}$.
    Going through the proof of Proposition~\ref{prop: SpecAn} one notes that these are only applied to the following products below.

    \begin{lemma}\label{lemma: products with wN+1}
        We have in ${\mathcal A}_L$ for all $1 \le i \le N$
        \begin{align}
             & \pi_\ast ([w_{N+1} z^a]) = [w_{N+1} v_{N+1,0}^a],
             \quad
             \pi_\ast ([y_i^a w_{N+1}]) = \left[\frac{w_{N+1}}{(v_{N+1,0} - v_{i,0})^a}\right],
             \\
             & \pi_\ast( [w_i w_{N+1}]) = \left[\sum_{a=1}^{k_i} av_{i,a} \frac{w_{N+1}}{(v_{N+1,0} - v_{i,0})^{a+1}}\right],
             \\
             & \pi_\ast ([w_{N+1}^2]) = \left[w_{N+1} \left( v_{N+1,0}^{k_0-1} + \sum_{a=1}^{k_0-1}(a-1) v_{0,a} v_{N+1,0}^{a-2} - \sum_{i=1}^N \sum_{a=1}^{k_i} \frac{a v_{i,a}}{(v_{N+1,0} - v_{i,0})^{a+1}} \right)\right].
        \end{align}
    \end{lemma}
    \begin{proof}
        \textbf{Step 1}.
        $\pi_\ast ([w_{N+1} z^a])$ is expressed immediately via $P_{4,N+1}$. 
        
        \textbf{Step 2}.
        To compute $\pi_\ast ([y_i^a w_{N+1}])$ we start with $\pi_\ast ([y_i y_{N+1}^{a+1}])$.
        
        By $P_{2,i,N+1}$ we have $[y_i y_{N+1}^{a+1}] = \left[y_{N+1}^a \frac{y_{N+1} - y_i}{v_{N+1,0} - v_{i,0}}\right]$ and for some $R_a$ --- polynomials in $y_i$ and rational functions in $v_\bullet$ we have
        \begin{align*}
            [y_i y_{N+1}^{k_{N+1}+1}] &= \left[\frac{y_{N+1}^{k_{N+1}}}{v_{N+1,0} - v_{i,0}} + \sum_{a=0}^{k_{N+1}-1} y_{N+1}^{a} R_a \right]=
            \\
            & = \left[\frac{w_{N+1} - \sum_{a=1}^{k_{N+1}-1} a v_{N+1,a} y_{N+1}^{a+1}}{k_{N+1} v_{N+1,k_{N+1}} (v_{N+1,0} - v_{i,0})}\right] +  \left[\sum_{a=0}^{k_{N+1}-1} y_{N+1}^{a} R_a\right] \qquad (\text{by } P_{3,N+1}) .
        \end{align*}
        All the summands of the last expression project to zero by $\pi_\ast$.
        
        Now by $P_{3,N+1}$ again we have
        \[
            \pi_\ast ([y_i w_{N+1}]) = \pi_\ast ([ k_{N+1} v_{N+1,k_{N+1}} \cdot y_i y_{N+1}^{k_{N+1}} ]) = \left[\frac{w_{N+1}}{(v_{N+1,0} - v_{i,0})}\right].
        \]
        
        \textbf{Step 3}.
        $\pi_\ast ([w_i w_{N+1}])$ is resolved by $P_{3,i}$ giving the claim immediately.
        
        \textbf{Step 4}.
        Next we have 
        $\pi_\ast ([w_{N+1}^2]) = \pi_\ast ([w_{N+1} (z^{k_0-1} - \sum_{a=1}^{k_0-1} (a-1) v_{0,a} z^{a-2} - \sum_{i=1}^{N} w_i) ])$ by $P_5$. To finalize this expression we apply the computations of the previous steps.

    \end{proof}

<!-- proposition-4: body_md -->
\label{prop: mt match}
    Let $\circ_L$ and $\circ_K$ stand for the products of ${\mathcal H}_{0,L}$ and ${\mathcal H}_{0,K}$ respectively.
    
    Then
    \[
        \mathrm{pr}_\ast \left(\frac{{\partial} }{{\partial} v_{i,\alpha}} \circ_L \frac{{\partial} }{{\partial} v_{j,\beta}} \right)
        =
        \mathrm{pr}_\ast \left(\frac{{\partial} }{{\partial} v_{i,\alpha}} \right) \circ_K \ \mathrm{pr}_\ast \left(\frac{{\partial} }{{\partial} v_{j,\beta}} \right)

    \[
    \lambda (z, {\bf t})= \left.\left(\frac{z^{2n}}{n} + \sum_{\alpha=1}^{n-1}v_{0,\alpha}z^{2(\alpha-1)} - \frac{v_{1,2}^2}{2z^2} + \sum_{\alpha=1}^{m}v_{2,\alpha}(z-v_{2,0})^{-\alpha} \right)\right|_{v_{\bullet} = v_{\bullet}(t_{\bullet})}
    \]
    and
    \[
    A = \sum_{\substack{\alpha_1,\dots,\alpha_{k_0-1} \ge 0 \\ \sum_{i=1}^{2n-1} (2n+1-i) \alpha_i = 2n+1}} (|\alpha|-2)! \prod_{i=1}^{k_0-1} \frac{t_{0,i}^{\alpha_i}}{\alpha_i!}.
    \]

It follows from Definition \ref{def: D_n open potential} that $(\int \widetilde{\lambda}(z,{\bf t})dz)|_{z=t_{2,0}}$ coincides with the open potential $F^o_{D_n}$ after the rescaling of variables. Thus, it suffices to prove that $A |_{t_{0,2\alpha} = 0, \ \alpha \ge 1} = 0$.

If the summand of $A$ does not contain $t_{0,2j}$ as a factor, then the left hand side of the equation $\sum_{i=1}^{2n-1} (2n+1-i) \alpha_i = 2n+1$ is even, whereas the right hand side is odd. This gives the claim.
