The suggested approach provides a rich data-driven framework based on a rigorous mathematical footing provided by optimal control theory. However, it is not a complete solution to the sampling problem, as training data from the equilibrium distribution are still required. Strategies to obtain these data in an adaptive manner will be explored in future research.

<!-- proposition-1: body_md -->
\label{prop:ev_problem_gen}
    The optimal cost-to-go $J^*(\mathbf{x})$ in~\eqref{eq:ergodic_ocp} satisfies the following properties:    \begin{itemize}
        \item[($i$)] The cost $J^*(\mathbf{x})$ is \textbf{independent} of $\mathbf{x}$, i.e. $J^*(\mathbf{x}) = \lambda \geq 0$ for all $\mathbf{x} \in \mathbb{R}^d$.
        \item[($ii$)] The optimal control $\mathbf{u}^*$ is a \textbf{stationary feedback control}, expressible as the gradient of a biasing potential $V_{\mathrm{bias}}$:
        \begin{equation*}
            \mathbf{u}^*_s = -\nabla V_\mathrm{bias}(\mathbf{X}_s).
        \end{equation*}
        \item[($iii$)] The optimal cost-to-go $\lambda$ and the biasing potential $V_\mathrm{bias}$ can be computed from the ground state of the following \textbf{linear eigenvalue equation}:
    \begin{equation}
        \label{eq:ev_problem}
        (\sigma \ell - \mathcal{L})\varphi = \lambda \varphi
    \end{equation}
    for the uncontrolled generator $\mathcal{L}$ defined in~\eqref{eq:generator}. The smallest eigenvalue $\lambda$ equals the optimal cost $J^{*}(\mathbf{x})$ for any $\mathbf{x}$. By the Perron-Frobenius Theorem, the eigenfunction $\varphi$ is strictly positive, and gives rise to the bias potential as
    \begin{equation*}
        V_\mathrm{bias}(\mathbf{x}) = - \frac{2}{\beta}\log \varphi(\mathbf{x}).
    \end{equation*}
   \end{itemize}
    The optimal controlled dynamics is then obtained by choosing
    \begin{equation*}
    \begin{split}
        \mathrm{d} \mathbf{X}^*_t
        &= \left[-\nabla V(\mathbf{X}^*_t) + \frac{2}{\beta}\frac{\nabla \varphi(\mathbf{X}^*_t)}{\varphi(\mathbf{X}^*_t)}\right]\,\mathrm{d} t + \sqrt{2\beta^{-1}}\,\mathrm{d} \mathbf{B}_t.
    \end{split}
    \end{equation*}

<!-- remark-1: body_md -->
In our experiments, we will estimate the equilibrium average $\mathbb{E}^\mu[\ell(\mathbf{x})]$ by extrapolating $\left.\frac{\mathrm{d}}{\mathrm{d}\sigma} \lambda(\sigma)\right|_{\sigma = 0}$ from solutions of the eigenvalue problem~\eqref{eq:ev_problem} at different $\sigma$ close to zero. Alternatively, one can also run controlled simulations at small $\sigma$, use them to evaluate the cost $J^* = \lambda(\sigma)$ defined in~\eqref{eq:ergodic_ocp}, and extrapolate the rate to $\sigma = 0$. However, since $\varphi$ becomes constant in the limit $\sigma\to 0$, resulting in vanishing of the control force, there will be a trade-off between the simulation speed-up from using controlled simulations and a low-variance evaluation of the quantity of interest.

<!-- proof-1: body_md -->
We proceed in five steps.

\medskip
\noindent\textbf{Step 1 (Existence of the ergodic HJB pair).}
We assume that the running cost $c(\mathbf{x},\mathbf{u}) := \sigma \ell(\mathbf{x}) + \frac{\beta}{4}|\mathbf{u}|^2$ and the potential $V$ are such that the ergodic Hamilton-Jacobi-Bellman equation can be solved (see~\cite[Thm.~3.6.6]{arapostathis2012ergodic} for conditions): there exists $W \in C^2(\mathbb{R}^d)$ and a constant $\lambda\in\mathbb{R}$, together satisfying $W(0)=0$, $\inf W > -\infty$, and
\begin{equation}\label{ieq:rho_rhostar}
    \lambda\le\lambda^{*},
\end{equation}
such that the ergodic HJB equation for~\eqref{eq:ergodic_ocp} holds true:
\begin{equation}
\label{eq:hjb_min}
\min_{\mathbf{u} \in \mathbb{R}^k}\Big[ \mathcal{L} W(\mathbf{x}) + \mathbf{u}\cdot \nabla W(\mathbf{x}) + \sigma\ell(\mathbf{x}) + \frac{\beta}{4}\|\mathbf{u}\|^2 \Big] = \lambda \qquad \forall\, \mathbf{x} \in \mathbb{R}^d.
\end{equation}

\medskip
\noindent\textbf{Step 2 (Pointwise minimization).}
For fixed $p\in\mathbb{R}^k$, completing the square gives
$$
\mathbf{u}\cdot p + \frac{\beta}{4}\|\mathbf{u}\|^2 = \frac{\beta}{4}\Big\|\mathbf{u}+\frac{2}{\beta}p\Big\|^2 - \frac{1}{\beta}\|p\|^2 \;\ge\;-\frac{1}\beta\|p\|^2,
$$
with equality iff $\mathbf{u}=-\frac{2}{\beta}p$. Setting $p=\nabla W(\mathbf{x})$ in~\eqref{eq:hjb_min},

\begin{equation}
\label{eq:ustar}
\mathbf{u}^*(\mathbf{x}) = -\frac{2}{\beta}\nabla W(\mathbf{x}),
\end{equation}
and~\eqref{eq:hjb_min} reduces to the semi-linear equation
\begin{equation}
\label{eq:semilinear-hjb}
\mathcal{L} W(\mathbf{x}) - \frac{1}{\beta}\|\nabla W(\mathbf{x})\|^2 + \sigma\ell(\mathbf{x}) = \lambda \quad \forall\, \mathbf{x}\in\mathbb{R}^d.
\end{equation}
Defining $V_{\mathrm{bias}}:= \frac{2}{\beta}W$ turns~\eqref{eq:ustar} into
$\mathbf{u}^*(\mathbf{x})=-\nabla V_{\mathrm{bias}}(\mathbf{x})$, which is claim~(ii).

\medskip
\noindent\textbf{Step 3 (Logarithmic transform).}
Set $\varphi:= e^{-W} > 0$, so $W=-\log\varphi$ and
$V_{\mathrm{bias}}=-\frac{2}{\beta}\log\varphi$. Then
$$
\nabla W = -\frac{\nabla\varphi}{\varphi}, \qquad
\Delta W = -\frac{\Delta\varphi}{\varphi} + \frac{\|\nabla\varphi\|^2}{\varphi^2},
$$
so, using $\mathcal{L}=-\nabla V\cdot\nabla+\beta^{-1}\Delta$,
$$
\mathcal{L} W = \frac{\nabla V\cdot\nabla\varphi}{\varphi} - \frac{1}{\beta}\frac{\Delta\varphi}{\varphi} + \frac{1}{\beta}\frac{\|\nabla\varphi\|^2}{\varphi^2},
\qquad
\frac{1}{\beta}\|\nabla W\|^2 = \frac{1}{\beta}\frac{\|\nabla\varphi\|^2}{\varphi^2}.
$$
The quadratic terms cancel exactly in $\mathcal{L} W - \frac{1}{\beta}\|\nabla W\|^2$, leaving
$$
\mathcal{L} W - \frac{1}{\beta}\|\nabla W\|^2
= \frac{\nabla V\cdot\nabla\varphi}{\varphi} - \frac{1}{\beta}\frac{\Delta\varphi}{\varphi}
= -\frac{1}{\varphi}\mathcal{L}\varphi.
$$
Substituting into~\eqref{eq:semilinear-hjb} and multiplying by $\varphi$,
\begin{equation}
(\sigma\ell - \mathcal{L})\varphi = \lambda\,\varphi,
\end{equation}
which is exactly~\eqref{eq:ev_problem}: no rescaling of $\mathcal{L}$ is needed, because the control-cost coefficient $\beta/4$ in~\eqref{eq:running_cost} was chosen precisely to make this cancellation exact.

\begin{remark}[Why $\beta/4$, precisely]\label{rem:why-beta4}
For a general coefficient $\eta$ in place of $\beta/4$ in~\eqref{eq:running_cost}, the
minimizer becomes $u^*=-\frac{1}{2\eta}\nabla W$ and the semilinear HJB equation reads
$\mathcal{L} W-\tfrac{1}{4\eta}\|\nabla W\|^2+\sigma\ell=\lambda$. The same substitution
$\varphi=e^{-W}$ then leaves
$$
\mathcal{L} W - \frac{1}{4\eta}\|\nabla W\|^2
= -\frac{\mathcal{L}\varphi}{\varphi} + \left(\frac{1}{\beta}-\frac{1}{4\eta}\right)\frac{\|\nabla\varphi\|^2}{\varphi^2},
$$
so the nonlinear term cancels, linearizing the equation into~\eqref{eq:ev_problem}, if and only if $\eta=\beta/4$.

Equivalently: allowing a \emph{tuned} substitution $W:=-c_1\cdot\log\varphi$ with
$c_1=4\eta/\beta$ linearizes the semilinear HJB equation for \emph{any} $\eta>0$,
but leaves the rescaled equation
$\big(\sigma\ell-c_1\mathcal{L}\big)\varphi=\lambda\varphi$;
only $\eta=\beta/4$ makes $c_1=1$ and removes the rescaling factor
$4\eta/\beta$ from in front of $\mathcal{L}$, giving~\eqref{eq:ev_problem} exactly,
against the bare generator $\mathcal{L}$.
\end{remark}

\medskip
\noindent\textbf{Step 4 (Ground state and principal eigenvalue).}
The uncontrolled Langevin dynamics are reversible with respect to $\mathrm{d}\mu(\mathbf{x})\propto e^{-\beta V(\mathbf{x})}\,\mathrm{d}\mathbf{x}$, and $\mathcal{L}$ is essentially self-adjoint on $\mathbb{L}^2(\mu)$. Moreover, for sufficiently
regular $f$,
$$
\langle f,-\mathcal{L} f\rangle_\mu = \frac{1}{\beta}\int_{\mathbb{R}^d} \|\nabla f\|^2\,\mathrm{d}\mu.
$$
Consequently, the operator $\sigma\ell-\mathcal{L}$
is essentially self-adjoint and nonnegative, with quadratic form
$$
\langle f,(\sigma\ell-\mathcal{L})f\rangle_\mu = \frac{1}{\beta}\int\|\nabla f\|^2\,\mathrm{d}\mu + \sigma\int\ell f^2\,\mathrm{d}\mu.
$$  
Its lowest eigenvalue is characterized by the Rayleigh--Ritz formula
$$
\lambda_0 = \inf_{f\neq0} \frac{\beta^{-1}\int\|\nabla f\|^2\,\mathrm{d}\mu+\sigma\int\ell f^2\,\mathrm{d}\mu}{\int f^2\,\mathrm{d}\mu}.
$$
Since the eigenfunction $\varphi=e^{-W}$ constructed in Step~3 is strictly
positive, the principal-eigenvalue characterization identifies
$\lambda=\lambda_0$.


\medskip
\noindent\textbf{Step 5 (Verification and optimality).} Steps 1-4 constructed a pair $(W,\lambda)$ solving the ergodic HJB~\eqref{eq:hjb_min}, with minimizing control $\mathbf{u}^*(\mathbf{x}) = -\frac{2}{\beta}\nabla W(\mathbf{x})$ and $\lambda$ the smallest eigenvalue of $\sigma\ell-\mathcal{L}$ on $\mathbb{L}^{2}(\mu)$. By the standard verification theorem for ergodic control (\cite[Thm.~3.5.6/3.5.8 ]{arapostathis2012ergodic}), any such solution pair automatically satisfies $\lambda=\lambda_{\mathbf{u}^*}=\lambda^*\ge0$ for every $\mathbf{x}$ -- proving claim (i) -- with the associated feedback control $\mathbf{u}^{*}$ being stable, proving claim (ii).
