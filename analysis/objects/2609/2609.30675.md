<!-- definition-1: body_md -->
\label{def:grothendieck}
Let $s_i$ interchange $x_i,x_{i+1}$ and fix $\mathbf{y}$, and set
$$
\partial_i(f) \;=\; \frac{f-s_i(f)}{x_i-x_{i+1}} \qquad \text{ and } \qquad
\pi_i(f) \;=\; \partial_i\bigl((1-x_{i+1})f\bigr)
$$
for any $f \in \mathbb{Z}[\mathbf{x}, \mathbf{y}]$.
The \emph{double Grothendieck} polynomials are determined by
$$
\begin{aligned}
\mathfrak{G}_{w_0}(\mathbf{x};\mathbf{y})&\;=\; \prod_{i+j\le n}(x_i+y_j-x_iy_j),\\
\mathfrak{G}_{ws_i}(\mathbf{x};\mathbf{y})&\;=\; \pi_i\bigl(\mathfrak{G}_w(\mathbf{x};\mathbf{y})\bigr)
\quad\text{if \;}\ell(ws_i)\;=\; \ell(w)-1.
\end{aligned}
$$
The \emph{ordinary Grothendieck polynomial} is $\mathfrak{G}_w(\mathbf{x})=\mathfrak{G}_w(\mathbf{x};\mathbf{0}) \in \mathbb{Z}[\mathbf{x}]$.
<!-- end definition-1 -->
<!-- headthm-1: body_md -->
\label{thm:main}
For every $w\in S_n$, the support of $\mathfrak{G}_w(\mathbf{x};\mathbf{y})$ is an $M^\natural$-convex set.
In particular, double Grothendieck polynomials have the SNP property.
<!-- end headthm-1 -->
<!-- head-1: body_md -->
Throughout this paper, unless otherwise stated, all algebraic varieties are defined over an algebraically closed field $\mathbb{k}$ of characteristic zero.
<!-- end head-1 -->
<!-- definition-2: body_md -->
\label{def:discrete-exchange}
A finite nonempty set $B\subseteq\mathbb{N}^n$ is \emph{$M$-convex} if it satisfies \emph{symmetric exchange}:
for all $\mathbf{a},\mathbf{b}\in B$ and every $i$ with $a_i>b_i$, there exists $j$
with $a_j<b_j$ such that
$$
 \mathbf{a}-{\normalfont\mathbf{e}}_i+{\normalfont\mathbf{e}}_j \;\in\; B
 \qquad\text{and}\qquad
 \mathbf{b}+{\normalfont\mathbf{e}}_i-{\normalfont\mathbf{e}}_j \;\in\; B.
$$
Equivalently, we say that $B$ is a \emph{polymatroid}.
A finite nonempty set $S\subseteq\mathbb{N}^n$ is \emph{$M^\natural$-convex} if
its homogenization
$$
 \widehat S_D \;:=\; \big\{(\mathbf{a},D-|\mathbf{a}|)\mid\mathbf{a}\in S\big\} \;\subseteq\; \mathbb{N}^{n+1}
$$
is $M$-convex, where $D\ge\max\big\lbrace |\mathbf{a}| \,\mid\, {\mathbf{a}\in S} \big\rbrace$ is an integer.
Equivalently, we say that $S$ is a \emph{generalized polymatroid}.
<!-- end definition-2 -->
<!-- remark-1: body_md -->
\label{prop:exchange-criterion}
To show that a finite nonempty set $S\subseteq\mathbb{N}^n$ is $M^\natural$-convex, it is enough to verify the following two conditions.
\begin{enumerate}[\rm (i)]
\item \emph{Exchange:} For every pair $(\mathbf{a},\mathbf{b})\in S\times S$ and
index $i\in\{1,\ldots,n\}$ such that $a_i>b_i$, at least one of the
following conditions holds:
\begin{enumerate}[\rm (1)]
	\item There exists $j\in\{1,\ldots,n\}$ with $a_j<b_j$ such that $\mathbf{a}-{\normalfont\mathbf{e}}_i+{\normalfont\mathbf{e}}_j\in S$ and $\mathbf{b}-{\normalfont\mathbf{e}}_j+{\normalfont\mathbf{e}}_i\in S$.
	\item $|\mathbf{a}|>|\mathbf{b}|$,  and $\mathbf{a}-{\normalfont\mathbf{e}}_i\in S$ and $\mathbf{b}+{\normalfont\mathbf{e}}_i\in S$.
\end{enumerate}
\item \emph{Expansion:} For every pair $(\mathbf{a},\mathbf{b})\in S\times S$ with
$|\mathbf{a}|<|\mathbf{b}|$, there exists an index $j\in\{1,\ldots,n\}$ such that $a_j<b_j$,  and $\mathbf{a}+{\normalfont\mathbf{e}}_j\in S$ and $\mathbf{b} -{\normalfont\mathbf{e}}_j \in S$.
\end{enumerate}
<!-- end remark-1 -->
<!-- remark-2: body_md -->
\label{rem_M_natural_top_bot}
	Every nonempty fixed-degree slice of an $M^\natural$-convex set is $M$-convex. 
	Indeed, two points in such a slice have equal homogenizing coordinates, so symmetric exchange uses only the original coordinates and preserves their total degree.
	Moreover, every point is coordinatewise bounded below by a point of
	minimum total degree and above by a point of maximum total degree.
	Indeed, fix points of minimum and maximum total degree and repeatedly
	apply \autoref{prop:exchange-criterion}(ii) to decrease or increase the
	given point until the respective degree is reached.
<!-- end remark-2 -->
<!-- lemma-1: body_md -->
\label{rem:intermediate-support}
Let $B,T\subseteq\mathbb{N}^n$ be $M$-convex sets.
If the function $\operatorname{rk}_T-\operatorname{rk}_B$ is nondecreasing under inclusion, then the set
$$
 S\;:=\;\big\{\mathbf{a}\in\mathbb{N}^n\mid\mathbf{b}\le\mathbf{a}\le\mathbf{t}
       \text{ for some }\mathbf{b}\in B,\ \mathbf{t}\in T\big\}
$$
is $M^\natural$-convex.
Conversely, if $B$ and $T$ are the subsets of minimum and maximum total degree of an $M^\natural$-convex set, respectively, then $\operatorname{rk}_T-\operatorname{rk}_B$ is nondecreasing under inclusion.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
$(\Rightarrow)$ Assume that $\operatorname{rk}_T-\operatorname{rk}_B : 2^{[n]} \rightarrow \mathbb{N}$ is nondecreasing under inclusion.
Let  $d:=\operatorname{rk}_B([n])$, and $D:=\operatorname{rk}_T([n])$.
Notice that $D - d \ge \operatorname{rk}_T(I) - \operatorname{rk}_B(I) \ge \operatorname{rk}_T(\varnothing) - \operatorname{rk}_B(\varnothing) = 0$ for all $I \subseteq [n]$.
We consider the function $h : 2^{[n+1]} \rightarrow \mathbb{N}$ given by 
$$
 h(I) \;:= \;\begin{cases}
	\operatorname{rk}_T\left(I\right) & \quad \text{if\; } I\subseteq [n] \\
	D-d+\operatorname{rk}_B\left(I\setminus\{n+1\}\right) & \quad \text{if\; } n+1 \in I.
 \end{cases}
$$
We show that $h$ is a polymatroid rank function.
Clearly, $h$ is integer-valued and $h(\varnothing)=0$.
Monotonicity follows from that of $\operatorname{rk}_B$ and $\operatorname{rk}_T$, together with
$$
 h\left(I\cup\{n+1\}\right)-h\left(I\right)
 \;=\;(D-d)-\bigl(\operatorname{rk}_T(I)-\operatorname{rk}_B(I)\bigr)\;\geq\;0 \qquad \text{ for all } I \subseteq [n].
$$
To check submodularity, only pairs with exactly one set containing $n+1$ need to be considered. 
For $I,J\subseteq[n]$, the hypothesis and submodularity of $\operatorname{rk}_T$ yield
\begin{align*}
	h\left(\left(I \cup \{n+1\}\right) \cup J\right) - h\left(I \cup \{n+1\}\right) &\;=\; \operatorname{rk}_B(I\cup J)-\operatorname{rk}_B(I)\\
	&\;\le\; \operatorname{rk}_T(I\cup J)-\operatorname{rk}_T(I)\\
	&\;\leq\;\operatorname{rk}_T(J)-\operatorname{rk}_T(I\cap J)\\
	&\;\leq\; h(J)-h\left(\left(I \cup \{n+1\}\right) \cap J\right).
\end{align*}
Therefore $h$ is a rank function with $h([n+1])=D$.
Then we have the equality
$$
 \widehat S_D
 \;=\;\Big\{\mathbf{c}\in\mathbb{N}^{n+1}\mid |\mathbf{c}|=D \quad \text{and}\quad
       \sum_{i\in I}c_i\leq h(I)\text{ for all }I\subseteq[n+1]\Big\}.
$$
Indeed, for $\mathbf{c}=(\mathbf{a},D-|\mathbf{a}|)$, these inequalities are equivalent to
$$
 d-\operatorname{rk}_B([n]\setminus I)
 \;\leq\;\sum_{i\in I}a_i\;\leq\;\operatorname{rk}_T(I)
 \qquad\text{for all }I\subseteq[n].
$$
By the rank descriptions of the upward closure of $B$ and the downward closure of $T$, the lower and upper bounds say precisely that $\mathbf{b}\leq\mathbf{a}\leq\mathbf{t}$ for some $\mathbf{b}\in B$ and $\mathbf{t}\in T$.
Finally $\widehat S_D$ is the $M$-convex set with rank function $h$, and so $S$ is $M^\natural$-convex.

$(\Leftarrow)$ Let $S$ be an $M^\natural$-convex set with minimum- and maximum-degree subsets $B$ and $T$, respectively.
Let $d:=\operatorname{rk}_B([n])$ and $D:=\operatorname{rk}_T([n])$, and  $h$ be the rank function of $\widehat S_D$.
By \autoref{rem_M_natural_top_bot}, for all $I \subseteq [n]$, we obtain
$$
 h(I) \;=\;\operatorname{rk}_T(I) \qquad \text{and} \qquad
 h(I\cup\{n+1\}) \;=\; D-d+\operatorname{rk}_B(I).
$$
By submodularity, $h(I\cup\{n+1\})-h(I)$ is nonincreasing under inclusion, and so $\operatorname{rk}_T-\operatorname{rk}_B$ is nondecreasing under inclusion.

<!-- theorem-1: body_md -->
\label{thm:rigidity}
Let $X \subset \mathbb{P} = \mathbb{P}_\mathbb{k}^{m_1} \times_\mathbb{k} \cdots \times_\mathbb{k} \mathbb{P}_\mathbb{k}^{m_n}$ be a $d$-dimensional integral subvariety with rational singularities. 
Then the following statements hold:
\begin{enumerate}[\rm (i)]
	\item $c_\mathbf{n}(X) \ge 0$ for all $\mathbf{n} \in \mathbb{N}^n$.
	\item $\operatorname{HSupp}_\mathbb{P}(X) \subseteq \operatorname{DMSupp}_\mathbb{P}(X)$.
	\item Let $\mathbf{n}, \mathbf{m} \in \mathbb{N}^n$ such that $\mathbf{n} \le \mathbf{m}$ and $\mathbf{m} \in \operatorname{DMSupp}_\mathbb{P}(X)$.
	Then 
	$$
	c_\mathbf{n}(X) \;>\;0 \qquad \text{ implies that } \qquad c_\mathbf{m}(X) \;>\; 0.
	$$
\end{enumerate}
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
To avoid non-irreducible possible scenarios, we use the elegant treatment of generic Bertini theorems as developed in \cite[\S 1.5]{FOV}.
	In particular, we replace $\mathbb{k}$  by the purely transcendental field extension $\mathbb{k}\left(z_{i,j} \mid 1 \le i \le n,\, 0 \le j \le m_i\right)$.
	Let $H_i = V\left(\sum_{j=0}^{m_i} z_{i,j}x_{i,j}\right) \subset \mathbb{P}$ be the pullback of the generic hyperplane in the $i$-th factor $\mathbb{P}_\mathbb{k}^{m_i}$ of $\mathbb{P}$.
	Then $X \cap H_i$ is either empty or integral of dimension $d-1$, and we have the equality 
	$$
	e_{\mathbf{n}}(X) \;=\; e_{\mathbf{n}-{\normalfont\mathbf{e}}_i}\left(X \cap H_i\right) \qquad \text{ for all \qquad $\mathbf{n} \ge {\normalfont\mathbf{e}}_i$}.
	$$
	Thus, by successively cutting with generic hyperplanes, we can check the claims in parts (i), (ii), (iii) with the simplification that $\mathbf{n} = \mathbf{0}$.
	Additionally, part (iii) can be checked with $\mathbf{n} = \mathbf{0}$ and $\mathbf{m} = {\normalfont\mathbf{e}}_i$ for some $i=1,\ldots,n$. 
	By using the adaptation of Bertini's theorem for rational singularities \cite[Remark 3.4.11(3)]{FOV}, we obtain that each $X \cap H_i$ also has rational singularities. 
	
	
	Now part (ii) is clear. 
	Indeed, if $\mathbf{0} \not\in \operatorname{DMSupp}_\mathbb{P}(X)$, then $\operatorname{MSupp}_\mathbb{P}(X) = \varnothing$, and therefore we should have $X = \varnothing$.
	
	Let $f : \widetilde{X} \rightarrow X$ be a rational resolution of $X$.
	Consider the ample line bundle $\mathscr{L} := \mathscr{O}_X(1, \ldots,1)$.
	The pullback line bundle $f^*(\mathscr{L})$ is big and nef on $\widetilde{X}$.
	By utilizing the Leray spectral sequence and the Kawamata--Viehweg vanishing theorem (see \cite[\S 5]{EV}), we obtain 
	$$
	\normalfont\text{H}^i\Big(X, \mathscr{L}^{-1}\Big) \;\cong\; \normalfont\text{H}^i\Big(\widetilde{X}, f^*(\mathscr{L})^{-1}\Big) \;=\; 0
	$$ 
	for all $i < \dim(X)$.
	Therefore, we obtain 
	$$
	c_{\mathbf{0}}(X) \;=\; {(-1)}^{d}P_X\left(-1,\ldots,-1\right) \;=\; \dim_\mathbb{k}\left(\normalfont\text{H}^d\left(X, \mathscr{L}^{-1} \right)\right) \;\ge\; 0.
	$$
	This shows the claimed nonnegativity statement in part (i).
	
	We assume that $c_{\mathbf{0}}(X) > 0$ and that ${\normalfont\mathbf{e}}_i \in \operatorname{DMSupp}_\mathbb{P}(X)$. 
	Hence, to conclude the proof, we must show that $c_{{\normalfont\mathbf{e}}_i}(X) > 0$.
	Since $X$ is Cohen--Macaulay, by Serre duality (see \cite[\S III.7]{HARTSHORNE}), we obtain the equalities 
	$$
	c_\mathbf{0}(X) \;=\; h^0\left(X, \omega_X \otimes \mathscr{L}\right)
	$$
	and 
	$$
	c_{{\normalfont\mathbf{e}}_i}(X) \;=\; h^0\left(X, \omega_X \otimes \mathscr{L} \otimes \mathscr{O}_X({\normalfont\mathbf{e}}_i)\right) - h^0\left(X, \omega_X \otimes \mathscr{L}\right) ;
	$$
	similarly, since $\mathscr{L} \otimes \mathscr{O}_X({\normalfont\mathbf{e}}_i)$ is ample, we have the vanishing $\normalfont\text{H}^j\left(X,\omega_X \otimes \mathscr{L} \otimes \mathscr{O}_X({\normalfont\mathbf{e}}_i) \right)=0$ for all $j \ge 1$.
	
	
	Consider the finite dimensional $\mathbb{k}$-vector spaces 
	$$
	W \;:=\; \normalfont\text{H}^0\left(X, \omega_X \otimes \mathscr{L}\right) \qquad \text{ and } \qquad V \;:=\; \normalfont\text{H}^0\left(X, \omega_X \otimes \mathscr{L} \otimes \mathscr{O}_X({\normalfont\mathbf{e}}_i)\right).
	$$
	Let $\Pi_i : \mathbb{P} \rightarrow \mathbb{P}_\mathbb{k}^{m_i}$ be the natural projection to the $i$-th factor.
	Since there is some $\mathbf{d} = (d_1,\ldots,d_n) \in \operatorname{MSupp}_\mathbb{P}(X)$ with  $d_i > 0$, \cite[Theorem A]{CCLMZ} implies that $\dim\left(\Pi_i(X)\right) > 0$.
	Hence we get global sections $s_0, s_1 \in \normalfont\text{H}^0(X, \mathscr{O}_X({\normalfont\mathbf{e}}_i))$ such that the rational function
	$$
	r \;:=\; \frac{s_1}{s_0} \;\in\; K(X) \quad \text{ is transcendental over $\mathbb{k}$.}
	$$
	Since $\omega_X \otimes \mathscr{L}$ is torsion-free of rank one, we have an injection $\omega_X \otimes \mathscr{L} \hookrightarrow \omega_X \otimes \mathscr{L} \otimes \mathscr{K}_X \cong \mathscr{K}_X$, where $\mathscr{K}_X$ is the constant sheaf on $X$ with values in $K(X)$.
	
	Assume by contradiction that $0 = c_{{\normalfont\mathbf{e}}_i}(X) = \dim_\mathbb{k}(V) - \dim_\mathbb{k}(W)$.
	This yields that
	$$
	W \;\xrightarrow{\;\cong\;}\; s_0 \cdot  W \;=\; V
	$$ 
	and 
	$$
	W \;\xrightarrow{\;\cong\;}\; s_1 \cdot  W \;=\; V. 
	$$
	Dividing by $s_0$ and using the embedding $W \subset K(X)$, we obtain 
	$$
	W \;=\; r \cdot  W.
	$$
	Finally, the Cayley--Hamilton theorem yields the contradiction that $r$ is algebraic over $\mathbb{k}$.
	Therefore, $c_{{\normalfont\mathbf{e}}_i}(X) >0$, and so the proof of part (iii) is complete.

<!-- proposition-1: body_md -->
\label{prop:standardization}
The homomorphism $\phi$ is faithfully flat. Let $I\subset R$ be a proper
$R$-homogeneous ideal and $J=\phi(I)S$ be its standardization.
Then the following statements hold:
\begin{enumerate}[\rm (i)]
\item $\operatorname{codim}(I)=\operatorname{codim}(J)$.
\item $R/I$ and $S/J$ have the same multigraded Betti numbers. In particular,
$
 \mathcal{K}(R/I;\mathbf{t},\mathbf{s})=\mathcal{K}(S/J;\mathbf{t},\mathbf{s}).
$
\item If $R/I$ is a domain with rational singularities, then $S/J$ is
also a domain with rational singularities.
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
We first consider the homomorphism
$$
 \mathbb{k}[x]\longrightarrow\mathbb{k}[w,z,w',z'],\qquad x\longmapsto wz+w'z'.
$$
The ring $\mathbb{k}[w,z,w',z']$ is torsion-free over the principal ideal domain $\mathbb{k}[x]$, so this homomorphism is flat. 
Also, its fibers are nonempty, and hence it is faithfully flat. 
By taking the product of these morphisms, we obtain that $\phi$ is faithfully flat.
Then parts (i) and (ii) follow by the same arguments as in the proof of \cite[Theorem~7.2(i)--(iii)]{CCC}.

We now prove part (iii). 
Let $A:=R/I$ and $B:=S/J$, and suppose that $A$ is a domain with rational singularities.
 Since $B\cong S\otimes_RA$, the morphism 
 $$
f \;:\; \operatorname{Spec}(B) \;\longrightarrow\; \operatorname{Spec}(A)
 $$ 
 is obtained from $\operatorname{Spec}(S)\longrightarrow\operatorname{Spec}(R)$ by base change. 
 In particular, $f$ is faithfully flat. 
Consider a closed point $\operatorname{Spec}(\mathbb{k}) \rightarrow \operatorname{Spec}(A)$. 
Let $a_{i,j}\in\mathbb{k}$ be the image of $x_{i,j}$. 
We have
$$
 B\otimes_A\mathbb{k}
 \;\cong\;S\otimes_R\mathbb{k}
 \;\cong\;\bigotimes_{1\leq i,j\leq n}
 \frac{\mathbb{k}[w,z,w',z']}{(wz+w'z'-a_{i,j})},
$$
where the tensor products on the right are taken over $\mathbb{k}$.
Thus the closed fiber is a product of the integral quadrics
$Q_a:=V(wz+w'z'-a)\subset\mathbb A^4_\mathbb{k}$.

If $a\neq0$, then $Q_a$ is smooth.
On the other hand, $Q_0$ has rational singularities (e.g., it is a normal toric variety).
Since rational singularities are preserved under products, we conclude that every closed fiber of $f : \operatorname{Spec}(B)\longrightarrow\operatorname{Spec}(A)$ is integral and has rational singularities.

Since $A$ is a domain and $B$ is flat over $A$, we have an injection
$
 B \hookrightarrow B\otimes_A\operatorname{Quot}(A).
$
The same quadric description over an algebraic closure of $\operatorname{Quot}(A)$ shows that the generic fiber is geometrically integral, and therefore $B$ is a domain (see \cite[\href{https://stacks.math.columbia.edu/tag/0BCM}{Tag 0BCM}]{stacks-project}). 
Finally, every closed point of $\operatorname{Spec}(B)$ maps to a closed point of $\operatorname{Spec}(A)$. 
By applying Elkik's theorem \cite[Th\'eor\`eme~5]{Elkik} at these closed points, we obtain that $B$ has rational singularities at every closed point, and hence everywhere.
This completes the proof of part (iii).

since each term $\mathbf{t}^{\mathbf{r}}u^{|\mathbf{c}|}$ on the left equals $(u\mathbf{t})^{\mathbf{r}}$ on the right.

 G_w^+(\mathbf{x};z\mathbf{1})
 &\;=\;(-1)^{\ell(w)}\mathcal{K}(R/I_w;\mathbf{1}+\mathbf{x},(1+z)\mathbf{1})\\
 &\;=\;(-1)^{\ell(w)}\mathcal{K}(R/I_w;(1+z)(\mathbf{1}+\mathbf{x}),\mathbf{1})\\
 &\;=\;G_w^+(x_1+z+x_1z,\ldots,x_n+z+x_nz),
\end{aligned}

where the last equality uses $(1+z)(1+x_i)=1+(x_i+z+x_iz)$.

$\prod_{i=1}^n(x_i+z+x_iz)^{p_i}$ for some
$\mathbf p\in\operatorname{Supp}(G_w^+(\mathbf{x}))$.

and total degree at least $|\mathbf p|$.
Hence $\mathbf{a}_x\leq\mathbf p$ and $|\mathbf p|\leq|\mathbf{a}_x|+|\mathbf{a}_y|=|\mathbf{a}|$.

 \operatorname{rk}_{B_w}(I\cup\{i\})-\operatorname{rk}_{B_w}(I)
 &\;\leq\;\operatorname{rk}_{B_w}(J\cup\{i\})-\operatorname{rk}_{B_w}(J)\\
 &\;\leq\;\operatorname{rk}_{T_w}(J\cup\{i\})-\operatorname{rk}_{T_w}(J)\\
 &\;=\; \Big(\operatorname{rk}_{T_w}(J\cup\{i\}) + \operatorname{rk}_{T_w}(K) \Big)- \Big(\operatorname{rk}_{T_w}(J) + \operatorname{rk}_{T_w}(K)\Big)\\
 &\;=\;\operatorname{rk}_{T_w}(I\cup\{i\})-\operatorname{rk}_{T_w}(I).
\end{aligned}
$$
Finally, we have that $\operatorname{rk}_{T_w}-\operatorname{rk}_{B_w}$ is nondecreasing under inclusion, and
\autoref{rem:intermediate-support} completes the proof of the theorem.

\label{lem:ordinary-FP}
Fix $1\leq t\leq L$, and let

 f\in \left(A^*(Z_{t-1})[z]\right)\left[x_{i_t},x_{i_t+1}\right],

 (\rho_t)_*\left(f(p_t,q_t)\right)

By construction, we have that 
$
 c_1(\mathscr{K}_t)=r_{i_t}^{(t-1)}+r_{i_t+1}^{(t-1)}
 $
 and 
 $
 c_2(\mathscr{K}_t)=r_{i_t}^{(t-1)}r_{i_t+1}^{(t-1)}.
$
On the other hand, the universal sequence \autoref{eq_univ_ex_seq} yields $c_1(\mathscr{K}_t) = p_t + q_t$.
Then
 the projective bundle formula for Chow rings
\cite[Theorem~3.3, Example~8.3.4]{FultonIT}  gives
$$
 A^*(Z_t) \;\cong\;
 \frac{A^*(Z_{t-1})[q_t]}
 {\bigl(q_t^2-c_1(\mathscr{K}_t)q_t+c_2(\mathscr{K}_t)\bigr)}
 \;=\; \frac{A^*(Z_{t-1})[q_t]}
 {\bigl((q_t-r_{i_t}^{(t-1)})(q_t-r_{i_t+1}^{(t-1)})\bigr)}.
$$
In particular, $A^*(Z_t)$ is free over $A^*(Z_{t-1})$ with basis $1,q_t$.
Then \cite[Proposition~3.1(a)]{FultonIT} yields
\begin{equation}
	\label{eb_push_PP1_bundle}
	 (\rho_t)_*(1)=0,\qquad
	(\rho_t)_*\left(c_1(\mathscr{K}_t)\right)
	=0,
	\qquad
	(\rho_t)_*(-q_t)=1, \qquad (\rho_t)_*(p_t)=1.
\end{equation}
Consider the polynomial 
$$
H(T) \;:=\; f(T,x_{i_t}+x_{i_t+1}-T)
- f(x_{i_t},x_{i_t+1}) - (T-x_{i_t})\partial_{i_t}(f) \;\in\; \left(A^*(Z_{t-1})[z,x_{i_t}, x_{i_t+1}]\right)[T],
$$
where $T$ is a new indeterminate. 
Since $H(x_{i_t}) = H(x_{i_t+1})=0$, it follows that $H(T)$ is divisible by $(T-x_{i_t})(T-x_{i_t+1})$.
By specializing $T=p_t$, $x_{i_t} = r_{i_t}^{(t-1)}$ and $x_{i_t+1} = r_{i_t+1}^{(t-1)}$, we obtain the equality
$$
 f(p_t,q_t)
\;=\; f\bigl(r_{i_t}^{(t-1)},r_{i_t+1}^{(t-1)}\bigr)
+(p_t-r_{i_t}^{(t-1)})
\partial_{i_t}(f)\bigl(
r_{i_t}^{(t-1)},r_{i_t+1}^{(t-1)}\bigr) \;\in\; A^*(Z_t)[z],
$$
because $q_t = r_{i_t}^{(t-1)}+r_{i_t+1}^{(t-1)}-p_t$ and $(p_t-r_{i_t}^{(t-1)})(p_t-r_{i_t+1}^{(t-1)})=(q_t-r_{i_t}^{(t-1)})(q_t-r_{i_t+1}^{(t-1)})=0$ in $A^*(Z_t)$.
Finally, \autoref{eb_push_PP1_bundle} and the projection formula give
$$
\begin{aligned}
 (\rho_t)_*\left(f(p_t,q_t)\right)
 ={}&f\bigl(r_{i_t}^{(t-1)},r_{i_t+1}^{(t-1)}\bigr)(\rho_t)_*(1) +\partial_{i_t}(f)\bigl(
 r_{i_t}^{(t-1)},r_{i_t+1}^{(t-1)}\bigr)
 \left((\rho_t)_*(p_t)
 -r_{i_t}^{(t-1)}(\rho_t)_*(1)\right)\\
 ={}&\partial_{i_t}(f)\bigl(
 r_{i_t}^{(t-1)},r_{i_t+1}^{(t-1)}\bigr).
\end{aligned}
$$
This concludes the proof of the lemma.

<!-- theorem-4: body_md -->
\label{prop:ordinary-support}
Let $w\in S_n$, and let
$G_w^+(\mathbf{x})=(-1)^{\ell(w)}\mathfrak{G}_w(-\mathbf{x};\mathbf{0})$ be the sign-corrected ordinary Grothendieck polynomial. 
Then the homogenization
$$
 F_w(\mathbf{x},z) \;:=\; z^N G_w^+\left(\frac{x_1}{z},\ldots,\frac{x_n}{z}\right)
$$
is a denormalized Lorentzian polynomial.
In particular, $\operatorname{Supp}(F_w)$ is $M$-convex, and $\operatorname{Supp}(G_w^+)$ is $M^\natural$-convex.
<!-- end theorem-4 -->
<!-- proof-8: body_md -->
For $1\leq i<n$, define operators on
$\mathbb{Z}[\mathbf{x},z]$ given by
$$
 D_i(f) \;:=\; \partial_i\bigl((x_{i+1}+z)f\bigr)
 \qquad \text{ and } \qquad
 \Theta_i(f):=\partial_i\bigl(x_{i+1}(x_i+z)f\bigr),
$$
where $s_i$ fixes $z$. 
Let
$\delta:=(n-1,n-2,\ldots,0)$. 
Since
$\mathbf i=(i_1,\ldots,i_L)$ is reduced for $u=w_0w^{-1}$, we obtain
$$
 w^{-1}w_0 \;=\; w_0uw_0
 \;=\; s_{n-i_1}\cdots s_{n-i_L}.
$$
Therefore sign-correction and homogenization of the recursion in
\autoref{def:grothendieck} give
\begin{equation}\label{eq:ordinary-D-recursion}
 F_w \;=\; \bigl(D_{n-i_1}\cdots D_{n-i_L}\bigr)(\mathbf{x}^\delta).
\end{equation}
In particular, $F_w$ is a homogeneous polynomial of degree $N$.
We extend these operators to Laurent polynomials and define the involution
$$
 \iota(f)(\mathbf{x},z) \;:=\; (x_1\cdots x_n)^{n-1}
 f(x_n^{-1},\ldots,x_1^{-1},z^{-1}).
$$
We have that $\iota(\mathbf{x}^\delta)=\mathbf{x}^\delta$.
In \autoref{rem:ordinary-conjugation}, we verify the following identity
$$
 z(\iota D_{n-i}\iota)(f)
 \;=\;\frac{x_{i+1}(x_i+z)f-x_i(x_{i+1}+z)s_i(f)}{x_i-x_{i+1}}
 \;=\; \Theta_i(f).
$$
Then applying these identities successively to
\autoref{eq:ordinary-D-recursion} yields
\begin{equation}\label{eq:ordinary-reciprocity}
\begin{aligned}
 P_w(\mathbf{x},z)
 &\;:=\;\bigl(\Theta_{i_1}\cdots\Theta_{i_L}\bigr)(\mathbf{x}^\delta) \\
  &\;=\;z^L\iota(F_w)(\mathbf{x},z)\\
 &\;=\; (x_1\cdots x_n)^{n-1} z^{-\ell(w)}
 G_w^+(z/x_n,\ldots,z/x_1).
\end{aligned}
\end{equation}
Consider the vector bundle $\mathscr{H}=\bigoplus_{j=1}^{n-1}\mathscr{E}_j^{(L)}$ on $Z_w$.
Since $\mathscr{E}_j^{(L)}$ has $r_1^{(L)},\ldots,r_j^{(L)}$ as Chern roots, we obtain
$$
 c_N(\mathscr{H})
 \;=\; \prod_{j=1}^{n-1}c_j(\mathscr{E}_j^{(L)})
 \;=\;\prod_{k=1}^n(r_k^{(L)})^{n-k}
 \;=\;\mathbf{x}^\delta\big|_{\mathbf{x}=\mathbf{r}^{(L)}}.
$$
Since
$\mathbf{r}^{(0)}=(h_1,\ldots,h_n)$, by successively applying \autoref{lem:ordinary-FP} and the projection formula, we obtain
\begin{equation}\label{eq:ordinary-pushforward}
 \rho_*\left(c_N(\mathscr{H})
       \prod_{t=1}^L q_t(p_t+z)\right)
 \;=\; P_w(\mathbf{h},z).
\end{equation}

We now express the class being pushed forward in terms of globally generated bundles. 
For the remainder of the proof, every bundle on an intermediate stage $Z_t$ is understood to be pulled back to $Z_w$, and we keep the same notation for its pullback. 

Let $d_a:=\sum_{b=a}^{n-1}c_b$ for $1\leq a\leq n$; in particular, $d_1=L$ and $d_n=0$.
For rows with $c_a=0$, we set $\mathscr{A}_a:=0$ and $\mathscr{B}_a:=0$.

We analyze each nonempty word $B_a = s_as_{a+1}\cdots s_{a+c_a-1}$.
For $1\leq j\leq c_a$, \autoref{eq:ordinary-step-sequences_p_t} gives
$$
 0 \;\longrightarrow\; \mathscr{Q}_{d_{a+1}+j}
 \;\longrightarrow\; \mathscr{E}_{a+j-1}^{(d_{a+1}+j)}
 \;\longrightarrow\; \mathscr{E}_{a+j-2}^{(d_{a+1}+j-1)}
 \;\longrightarrow\; 0.
$$
By definition \autoref{eq_def_EE}, we have that $\mathscr{E}_{a-1}^{(d_a-c_a)} = \mathscr{E}_{a-1}^{(0)}$.
It follows that the classes
$$
p_{d_{a+1}+j} \; = \; r_{a+j-1}^{(d_{a+1}+j)},
\qquad 1\leq j\leq c_a,
$$ 
are the Chern roots of the vector bundle $\mathscr{A}_a := \operatorname{Ker}\left(\mathscr{E}_{a+c_a-1}^{(d_a)} \twoheadrightarrow \mathscr{E}_{a-1}^{(0)}\right)$.
Since the initial flag $\mathscr{E}_\bullet^{(0)}$ splits at each step, we obtain a natural surjection $\bigoplus_{b=a}^n \mathscr{O}_\mathbb{P}({\normalfont\mathbf{e}}_b)\cong \mathscr{E}/\mathscr{E}_{a-1}^{(0)} \twoheadrightarrow \mathscr{A}_a$, and so $\mathscr{A}_a$ is globally generated.
Thus we obtain the globally generated vector bundle 
\begin{equation}\label{eq:ordinary-row_p_t}
	\mathscr{A} \;:=\; \bigoplus_{a=1}^{n-1} \mathscr{A}_a \qquad \text{ such that \quad  $\operatorname{rank}(\mathscr{A}) \;=\; L$ \quad and \quad }   c(\mathscr{A})
	\;=\; \prod_{t=1}^L(1+p_t).
\end{equation}

Similarly, from the natural surjections $\mathscr{E}_{a+c_a}^{(d_{a+1})} \twoheadrightarrow \mathscr{E}_{a+c_a-1}^{(d_{a+1})} \twoheadrightarrow \cdots \twoheadrightarrow \mathscr{E}_{a}^{(d_{a+1})} = \mathscr{E}_{a}^{(0)}$, we obtain the globally generated vector bundle 
$$
\mathscr{B}_a
\;:=\;\operatorname{Ker}\left(
\mathscr{E}_{a+c_a}^{(d_{a+1})}
\twoheadrightarrow\mathscr{E}_a^{(0)}\right)
$$
with Chern roots $r_{a+j}^{(d_{a+1})}$ for $1\leq j\leq c_a$.
For $t=d_{a+1}+j$ with $1\leq j\leq c_a$, we have $i_t=a+j-1$, and so by successively applying \autoref{eq_compatible_roots}, we get
$$
r_{i_t+1}^{(t-1)} \;=\; r_{i_t+1}^{(t-2)} \;=\; \cdots \;=\; r_{i_t+1}^{(d_{a+1}+1)} \;=\; r_{a+j}^{(d_{a+1})}.
$$
Thus we obtain the globally generated vector bundle
\begin{equation}\label{eq:ordinary-row_q_t}
 \mathscr{B}\;:=\;\bigoplus_{a=1}^{n-1}\mathscr{B}_a,
 \qquad \text{ such that } \qquad \operatorname{rank}(\mathscr{B})=L,
 \qquad
 c(\mathscr{B})\;=\;\prod_{t=1}^L
 \left(1+r_{i_t+1}^{(t-1)}\right).
\end{equation}
To relate these roots to the factors $q_t$ in
\autoref{eq:ordinary-pushforward}, for each $0 \le t \le L$, we set
$$
 \mathscr{H}_t \; := \; \bigoplus_{j=1}^{n-1}\mathscr{E}_j^{(t)}.
$$
In particular, $\mathscr{H}_L=\mathscr{H}$.
At step $t$, \autoref{eq:ordinary-step-sequences_q_t} and \autoref{eq_def_roots} give the equalities
$$
 c\bigl(\mathscr{E}_{i_t}^{(t)}\bigr)(1+q_t)
 \;=\;c\bigl(\mathscr{E}_{i_t+1}^{(t-1)}\bigr)
 \;=\;c\bigl(\mathscr{E}_{i_t}^{(t-1)}\bigr)
 \left(1+r_{i_t+1}^{(t-1)}\right).
$$
By multiplying all these equalities for $t = 1,\ldots, L$ and applying \autoref{eq:ordinary-row_q_t}, we obtain
$$
 c(\mathscr{H})\prod_{t=1}^L(1+q_t)
 \;=\;c(\mathscr{H}_0)c(\mathscr{B}).
$$
Since $\operatorname{rank}(\mathscr{H})=\operatorname{rank}(\mathscr{H}_0)=N$ and
$\operatorname{rank}(\mathscr{B})=L$, it follows that
\begin{equation}\label{eq:ordinary-top-Chern}
 c_N(\mathscr{H})\prod_{t=1}^L q_t
 \;=\;c_N(\mathscr{H}_0)c_L(\mathscr{B}).
\end{equation}
Combining \autoref{eq:ordinary-pushforward},
\autoref{eq:ordinary-row_p_t}, and \autoref{eq:ordinary-top-Chern} yields
\begin{equation}\label{eq:ordinary-Chern-identity}
 P_w(\mathbf{h},z)=
 \rho_*\left(c_N(\mathscr{H}_0) \, c_L(\mathscr{B}) \, \sum_{k=0}^Lz^{L-k}c_k(\mathscr{A})\right).
\end{equation}
Let $\mathscr{V}:=\mathscr{H}_0\oplus\mathscr{B}$ and $r:=N+L$.
Let $\mathscr{R}_j$ be the globally generated vector bundle on $Z_w$ given as the pullback of the twisted tangent bundle
$$
T_{\mathbb{P}_\mathbb{k}^{n-1}} \otimes \mathscr{O}_{\mathbb{P}_\mathbb{k}^{n-1}}(-1) \;\cong\; {\rm Coker}\left(\mathscr{O}_{\mathbb{P}_\mathbb{k}^{n-1}}(-1) \rightarrow \mathscr{O}_{\mathbb{P}_\mathbb{k}^{n-1}}^n\right)
$$ 
on the $(n+1-j)$-th factor of $\mathbb{P}=\left(\mathbb{P}_\mathbb{k}^{n-1}\right)^n$, so that $c_a(\mathscr{R}_j)=h_{n+1-j}^a$.
Consider the polynomial
$$
 \Phi(\mathbf{x},z,\tau)
 \;:=\;\int_{Z_w}c_\tau(\mathscr{V})c_z(\mathscr{A})
 \prod_{j=1}^n c_{x_j}(\mathscr{R}_j) \;\in\; \mathbb{Z}[\mathbf{x}, z, \tau].
$$

Finally, by applying the Chern-class criterion of \cite[Proposition~6.2]{CRSegre} to the projection
$Z_w\to\mathbb{P}^0$, we obtain that 
$$
\mathrm{N}(\Phi) \quad \text{ is a Lorentzian polynomial.}
$$ 
By \autoref{eq:ordinary-reciprocity}, the coefficient of
$\mathbf{x}^\mathbf{a} z^k$ in $F_w$ equals that of
$x_n^{n-1-a_1}\cdots x_1^{n-1-a_n}z^{L-k}$ in $P_w$.
Therefore \autoref{eq:ordinary-Chern-identity} and the fact that $A^*(\mathbb{P})\cong \mathbb{Z}[h_1,\ldots,h_n]/\left(h_1^n,\ldots,h_n^n\right)$ give
$$
 \mathrm{N}(F_w)
 \;=\;\partial_\tau^r\bigl(\mathrm{N}(\Phi)\bigr).
$$
Since derivatives preserve Lorentzianity~\cite{BH}, $\mathrm{N}(F_w)$ is Lorentzian, as claimed.

\label{rem:ordinary-conjugation}
Here we verify the operator identity used in the proof of \autoref{prop:ordinary-support}.
We work in the Laurent polynomial ring
$\mathbb{Z}[x_1^{\pm1},\ldots,x_n^{\pm1},z^{\pm1}]$. 
Let
$$
 M \;:=\; (x_1\cdots x_n)^{n-1}
 \qquad \text{and} \qquad
 \mathcal{R}(f)(\mathbf{x},z) \;:=\; f(x_n^{-1},\ldots,x_1^{-1},z^{-1}).
$$
We have that $\iota(f)=M\mathcal{R}(f)$ and that $\mathcal{R}$ is a ring involution with $\mathcal{R}(M)=M^{-1}$. 
Since $s_j(M)=M$, we get that $D_j(Mh)=M D_j(h)$. 
It then follows
$$
 \iota\bigl(D_j(\iota(f))\bigr)
 \;=\; M\mathcal{R}\bigl(D_j(M\mathcal{R}(f))\bigr)
 \;=\; M\mathcal{R}(M)\mathcal{R}\bigl(D_j(\mathcal{R}(f))\bigr)
 \;=\; \mathcal{R}\bigl(D_j(\mathcal{R}(f))\bigr).
$$
Fix $1\leq i<n$ and set $j=n-i$. 
We have that 
$$
\mathcal{R}(x_j) \;=\; x_{i+1}^{-1}, \quad \mathcal{R}(x_{j+1}) \;=\; x_i^{-1} \quad \text{ and } \quad \mathcal{R}\bigl(s_j(\mathcal{R}(f))\bigr) \;=\; s_i(f).
$$
By combining everything, we obtain
$$
\begin{aligned}
 z(\iota D_{n-i}\iota)(f)
 &\;=\; z\frac{(x_i^{-1}+z^{-1})f
       -(x_{i+1}^{-1}+z^{-1})s_i(f)}
       {x_{i+1}^{-1}-x_i^{-1}}\\
 &\;=\; \frac{x_{i+1}(x_i+z)f-x_i(x_{i+1}+z)s_i(f)}
        {x_i-x_{i+1}}\\
 &\;=\; \partial_i\bigl(x_{i+1}(x_i+z)f\bigr) \;=\; \Theta_i(f).
\end{aligned}
$$
For a polynomial $f$, the resulting identity lies in $\mathbb{Z}[\mathbf{x},z]$.
