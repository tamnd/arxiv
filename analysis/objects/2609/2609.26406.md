<!-- theorem-1: body_md -->
\label{thm:onesigbound}
    Let $X_n = \frac{1}{\sqrt{n}}X=\frac{1}{\sqrt{n}}[x_1,x_2,...,x_n] \in \mathbb{R}^{p\times n}$ be the scaled data matrix where the data matrix $X$ is as defined in \eqref{eq:1.defmodel}. Then assuming $\sigma_1(X_n) > \sigma_{r_1+1}(X_n)$,
    \begin{equation} \label{eq:onesigbound}
        \sin\theta(u_1,\widehat U_1) \leq \sqrt{\frac{\sigma_1(X_n)^2-\left\|{u_1^TX_n}\right\|_2^2}{\sigma_1(X_n)^2-\sigma_{r_1+1}(X_n)^2}}
    \end{equation} where $u_1$ is the dominant signal direction and $\widehat U_1 \in \mathbb{R}^{p\times r_1}$ is the matrix containing the $r_1$ leading left singular vectors of $X_n$.

<!-- theorem-2: body_md -->
\label{thm:multsigbound}
    Let $X_n =\frac{1}{\sqrt{n}}X = \frac{1}{\sqrt{n}} [x_1,x_2,...,x_n]\in \mathbb{R}^{p\times n}$ be the scaled data matrix where the data matrix $X$ is as defined in \eqref{eq:1.defmodel}. Let $U_1\in\mathbb{R}^{p\times r_1}$ be the $r_1$ leading signal directions with $U_\perp\in\mathbb{R}^{p\times (p-r_1)}$ as its orthogonal complement, and let $[\widehat U_1,\widehat U_2]$ be the leading $r_2(\geq r_1)$ left singular vectors of the data matrix. 
    Then assuming $\sigma_{\min}(U_1^TX_n) > \sigma_{r_2+1}(X_n)$,
    \begin{equation} \label{eq:multisigbound}
        \left\|{\sin\Theta(U_1,[\widehat U_1,\widehat U_2])}\right\|_2 \leq \frac{\left\|{U_\perp^T X_n \check V}\right\|_2 \sigma_{\min}(U_1^TX_n)}{
        \sigma_{\min}(U_1^TX_n)^2 - \sigma_{r_2+1}(X_n)^2},
    \end{equation} where $\check V$ is the leading $r_1$ right singular vectors of $U_1^TX_n$, which is independent of $U_\perp^TX_n$.
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
\*
    Since $\check V$ is independent of $U_\perp^TX_n$, 
    \begin{equation*}
        U_\perp^T X_n \check V \overset{d}{=}  {\rm diag}(\sigma_{r_1+1},...,\sigma_p)G_{(p-r_1)\times r_1}
    \end{equation*}
    where $\sigma_j = \sqrt{\beta_j+\sigma^2}$ and $G_{(p-r_1)\times r_1}$ is a $(p-r_1)\times r_1$ standard Gaussian matrix. We also have
        $U_1^T X_n \overset{d}{=} {\rm diag}(\sigma_1,...,\sigma_{r_1}) G_{r_1 \times n}$ and
        $U_\perp^T X_n \overset{d}{=} {\rm diag}(\sigma_{r_1+1},...,\sigma_{p}) G_{(p-r_1) \times n}$.
    Therefore, roughly,
    \begin{equation} \label{eq:roughLimit}
        \frac{\left\|{U_\perp^T X_n \check V}\right\|_2 \sigma_{\min}(U_1^TX_n)}{
        \sigma_{\min}(U_1^TX_n)^2 - \sigma_{r_2+1}(X_n)^2} \approx \frac{\sigma_{r_1}\sigma_{r_1+1}(\sqrt{p-r_1}+\sqrt{r_1})(\sqrt{n}+\sqrt{r_1})}{(\sqrt{n}+\sqrt{r_1})^2\sigma_{r_1}^2- (\sqrt{n}+\sqrt{p})^2\sigma_{r_2+1}^2},
    \end{equation} which can be made more precise by a result by Davidson and Szarek \cite{DavidsonSzarek2001}, given that the denominator is positive. When $n\rightarrow \infty$ with $p$ fixed, the right-hand side of \eqref{eq:roughLimit} goes to zero and when $p,n \rightarrow \infty$ with $p/n \rightarrow c$, the right-hand side of \eqref{eq:roughLimit} tends to 
    \begin{equation*}
        \frac{\sigma_{r_1}\sigma_{r_1+1}\sqrt{c}}{\sigma_{r_1}^2- (1+\sqrt{c})^2\sigma_{r_2+1}^2}
    \end{equation*} assuming that the denominator remains positive, that is, $\sigma_{r_1}> (1+\sqrt{c})\sigma_{r_2+1}$.
<!-- end remark-1 -->
<!-- theorem-3: body_md -->
\label{thm:genMultsig}
    Let $X_n = \frac{1}{\sqrt{n}}X=\frac{1}{\sqrt{n}}[x_1,x_2,...,x_n] \in \mathbb{R}^{p\times n}$ be the scaled data matrix where the data matrix $X$ is as defined in \eqref{eq:1.defmodel}. Then
    \begin{equation} \tag{\ref{eq:onesigbound}}
        \sin\theta(u_1,\widehat U_1) \leq \sqrt{\frac{\sigma_1(X_n)^2-\left\|{u_1^TX_n}\right\|_2^2}{\sigma_1(X_n)^2-\sigma_{r_1+1}(X_n)^2}}
    \end{equation} where $u_1$ is the dominant signal direction and $\widehat U_1 \in \mathbb{R}^{p\times r_1}$ is the matrix containing the $r_1$ leading left singular vectors of $X_n$.
<!-- end theorem-3 -->
<!-- proof-1: body_md -->
First, let the SVD of $\widetilde X_n^{(T)}$ be
    \begin{equation}
        \widetilde X^{(T)}_n= \widetilde U \widehat\Sigma \widetilde V^T = \left[\widetilde U_1,\widetilde U_2\right]
    \begin{bmatrix}
        \widehat\Sigma_1 &  \\  
         & \widehat\Sigma_2
    \end{bmatrix}
    \left[\widetilde V_1,\widetilde V_2 \right]^T
    \end{equation} where $\widetilde U_1 \in \mathbb{R}^{p\times r_1}, \widetilde U_2 \in \mathbb{R}^{p\times (p-r_1)}, \widetilde V_1 \in \mathbb{R}^{n\times r_1}, \widetilde V_2 \in \mathbb{R}^{n\times (n-r_1)}, \widehat\Sigma_1 \in \mathbb{R}^{r_1\times r_1}$ and $\widehat\Sigma_2 \in \mathbb{R}^{(p-r_1)\times (n-r_1)}$. Here, $\widehat\Sigma_1$ and $\widehat\Sigma_2$ are the singular values of both $\widetilde X^{(T)}_n$ and $X_n$ where \sloppy $\widehat\Sigma_1 = {\rm diag}(\widehat\sigma_1,\widehat\sigma_2,...,\widehat\sigma_{r_1})$ and $\widehat\Sigma_2 = \begin{bmatrix}
        {\rm diag}(\widehat\sigma_{r_1+1},\widehat\sigma_{r_1+2},...,\widehat\sigma_{n}) \\ 0_{(p-n) \times (n-r_1)}
    \end{bmatrix}$ if $p > n$ and $\widehat\Sigma_2 = \begin{bmatrix}
        {\rm diag}(\widehat\sigma_{r_1+1},\widehat\sigma_{r_1+2},...,\widehat\sigma_{p}) & 0_{(p-r_1)\times (n-p)}
    \end{bmatrix}$ if $p \leq n$, i.e., $\widehat\sigma_j = \sigma_j(X_n)$. Furthermore let $\widehat U$ be the left singular vectors of $X_n$ and $\widehat U_\perp$ a complement of $\widehat U$. Note that by letting $\widehat U_\perp$ be any orthogonal complement of $\widehat U$, we have $[\widetilde U_1,\widetilde U_2] = [U,U_\perp]^T[\widehat U,\widehat U_\perp]$ where $U_\perp$ is any orthogonal complement of $U = [u_1,...,u_k]$. Therefore the left singular vectors of $\widetilde X_n^{(n)}$ are the left singular vectors of $X_n$ up to left orthogonal transformation by $[U,U_\perp]^T$.

    
    We now have
    \begin{equation*}
        \sin\theta(u_1,\widehat U) = \left\|{u_1^T\widehat U_\perp}\right\|_2 = \left\|{e_1^T [u_1, ...,u_k, U_\perp]^T\widehat U_\perp}\right\|_2 = \sin\theta(e_1,[U,U_\perp]^T\widehat U) = \sin\theta(e_1,\widetilde U_1)
    \end{equation*} where $e_1$ is the first canonical basis vector. Therefore we bound $ \sin\theta(e_1,\widetilde U_1)$ instead.
    

    We first bound $\cos\theta(e_1,\widetilde U_1) = \left\|{e_1^T\widetilde U_1}\right\|_2$ using the quantities $e_1^T\widetilde X^{(T)}_n \widetilde V_1$ and $e_1^T\widetilde X^{(T)}_n \widetilde V_2$. First, for $e_1^T\widetilde X^{(T)}_n\widetilde V_1$ we have
    \begin{equation}
        e_1^T\widetilde X^{(T)}_n \widetilde V_1 = \left\|{b_1}\right\|_2 e_1^T\widetilde V_1 = \left\|{b_1}\right\|_2 \widetilde v_1
    \end{equation} where $\widetilde v_1 \in \mathbb{R}^{1\times r_1}$ is the first row of $\widetilde V_1$. On the other hand, we also have
    \begin{equation}
        e_1^T\widetilde X^{(T)}_n \widetilde V_1 = e_1^T \widetilde U_1 \widehat\Sigma_1 = \widetilde u_1 \widehat\Sigma_1
    \end{equation} where $\widetilde u_1 \in \mathbb{R}^{1\times r_1}$ is the first row of $\widetilde U_1$. Therefore $\left\|{b_1}\right\|_2 \widetilde v_1 = \widetilde u_1 \widehat\Sigma_1$ and we get
    \begin{equation} \label{eq:ineq1}
        \left\|{b_1}\right\|_2\left\|{\widetilde v_1}\right\|_2 =\left\|{\widetilde u_1 \widehat\Sigma_1}\right\|_2 \leq \left\|{\widetilde u_1}\right\|_2  \widehat\sigma_{1}.
    \end{equation}
    Similarly for $e_1^T\widetilde X^{(T)}_n \widetilde V_2$ we get $\left\|{b_1}\right\|_2 \widetilde v_2 = \widetilde u_2 \widehat\Sigma_2$, where $\widetilde u_2$ is the first row of $\widetilde U_2$ and $\widetilde v_2$ is the first row of $\widetilde V_2$, which gives us 
    \begin{equation} \label{eq:ineq2}
       \left\|{b_1}\right\|_2 \left\|{\widetilde v_2}\right\|_2 \leq \left\|{\widetilde u_2}\right\|_2 \widehat\sigma_{r_1+1}.
    \end{equation} Now since $\widetilde U$ and $\widetilde V$ are orthogonal matrices, we have
    $\left\|{\widetilde u_1}\right\|_2^2+\left\|{\widetilde u_2}\right\|_2^2 = 1$ and $\left\|{\widetilde v_1}\right\|_2^2+\left\|{\widetilde v_2}\right\|_2^2 = 1$. Therefore we can rewrite \eqref{eq:ineq1} as
    \begin{equation} \label{eq:ineq1r}
        \left\|{b_1}\right\|_2^2 \left(1-\left\|{\widetilde v_2}\right\|_2^2\right) \leq  \left(1-\left\|{\widetilde u_2}\right\|_2^2\right) \widehat\sigma_1^2,
    \end{equation} which gives us
    \begin{equation*}
        \left\|{\widetilde u_2}\right\|_2^2 \leq 1- \frac{\left\|{b_1}\right\|_2^2}{\widehat\sigma_1^2}\left(1-\left\|{\widetilde v_2}\right\|_2^2\right) \leq 1- \frac{\left\|{b_1}\right\|_2^2}{\widehat\sigma_1^2}+\frac{\left\|{\widetilde u_2}\right\|_2^2\widehat\sigma_{r_1+1}^2}{\widehat\sigma_1^2}
    \end{equation*} using \eqref{eq:ineq2} and \eqref{eq:ineq1r}.
    Finally, rearranging gives us the desired inequality,
    \begin{align*}
        \sin\theta(u_1,\widehat U) = \sin\theta(e_1,\widetilde U_1) = \left\|{e_1^T\widetilde U_2}\right\|_2 = \left\|{\widetilde u_2}\right\|_2 \leq \sqrt{\frac{\widehat\sigma_1^2-\left\|{b_1}\right\|_2^2}{\widehat\sigma_1^2-\widehat\sigma_{r_1+1}^2}} = \sqrt{\frac{\sigma_1(X_n)^2-\left\|{u_1^TX_n}\right\|_2^2}{\sigma_1(X_n)^2-\sigma_{r_1+1}(X_n)^2}}.
    \end{align*}

<!-- lemma-1: body_md -->
\label{lem:triangle}
Let $U = [U_1,U_2,U_3]$, $\hat{U} = [\hat{U}_1,\hat{U}_2,\hat{U}_3]$ and $\tilde{U} = [\tilde{U}_1,\tilde{U}_2,\tilde{U}_3]$ be $n\times n$ orthogonal matrices where index $1$ corresponds to $j$ columns, index $2$ corresponds to $\ell-j>0$ columns and index $3$ corresponds to the other $n-\ell$ columns. Then 
    \begin{equation}
        \left\|{\sin\Theta(U_1,[\tilde{U}_1,\tilde{U}_2])}\right\| \leq \left\|{\sin\Theta(U_1,[\hat{U}_1,\hat{U}_2])}\right\| +\left\|{\sin\Theta([\hat{U}_1,\hat{U}_2],[\tilde{U}_1,\tilde{U}_2])}\right\|
    \end{equation}
    and 
    \begin{equation}
        \left\|{\sin\Theta(U_1,[\tilde{U}_1,\tilde{U}_2])}\right\| \leq \left\|{\sin\Theta(U_1,\hat{U}_1)}\right\| +\left\|{\sin\Theta(\hat{U}_1,[\tilde{U}_1,\tilde{U}_2])}\right\|
    \end{equation} for any unitarily invariant norms.

<!-- theorem-4: body_md -->
\label{thm:multsigbound2}
    Let $X_n =\frac{1}{\sqrt{n}}X = \frac{1}{\sqrt{n}} [x_1,x_2,...,x_n]\in \mathbb{R}^{p\times n}$ be the scaled data matrix where the data matrix $X$ is as defined in \eqref{eq:1.defmodel}.
    Then 
    \begin{equation}
        \left\|{\sin\Theta(U_1,[\widehat U_1,\widehat U_2])}\right\|_2 \leq \min_{r_*\in [r_1,r_2]}\frac{\left\|{U_{*,\perp}^T X_n \check V}\right\|_2 \sigma_{\min}(U_*^TX_n)}{
        \sigma_{\min}(U_*^TX_n)^2 - \sigma_{r_2+1}(X_n)^2},
    \end{equation} where $U_*$ is the $r_*$ leading signal directions with $U_{*,\perp}$ as its orthogonal complement, $[\widehat U_1,\widehat U_2]$ is the leading $r_2$ left singular values of the data matrix and $\check V$ is the leading $r_1$ right singular vectors of $U_1^TX_n$, which is independent of $U_{*,\perp}^TX_n$.

<!-- proof-3: body_md -->
Let $\widehat U_\perp$ be an orthogonal complement of $[\widehat U_1,\widehat U_2]$ and $[\widetilde U_1,\widetilde U_2,\widetilde U_\perp] \in \mathbb{R}^{p\times p}$ be the orthogonal matrix of left singular vectors of $\widetilde X_n$ where $\widetilde U_1\in \mathbb{R}^{p\times r_1}$, $\widetilde U_2 \in \mathbb{R}^{p\times (r_2-r_1)}$ and $\widetilde U_\perp \in \mathbb{R}^{p\times (p-r_2)}$. Note that $[\widetilde U_1,\widetilde U_2,\widetilde U_\perp] = [U_1,U_\perp]^T[\widehat U_1,\widehat U_2,\widehat U_\perp]$. Then 
    \begin{align*}
        \left\|{\sin\Theta(U_1,[\widehat U_1,\widehat U_2])}\right\|_2 &= \left\|{U_1^T\widehat U_\perp}\right\|_2 = \left\|{[I_{r_1},0][U_1,U_\perp]^T\widehat U_\perp}\right\|_2 \\ &= \left\|{[I_{r_1},0]\widetilde U_\perp}\right\|_2 =\left\|{\sin\Theta\left(\begin{bmatrix}
            I_{r_1} \\ 0
        \end{bmatrix},[\widetilde U_1,\widetilde U_2]\right)}\right\|.
    \end{align*}
 Let the SVD of $\widetilde X_1 \in \mathbb{R}^{r_*\times n}$ be $\widetilde X_1 = \check U \check \Sigma \check V^T$ and let $\check V_\perp \in \mathbb{R}^{n\times (n-r_*)}$ be an orthogonal complement of $\check V$. Now, form the following matrix,
\begin{equation}
    \check X_n = \begin{bmatrix}
        \check U & 0 \\
       0  &  I_{(p-r_*)}
    \end{bmatrix}^T \begin{bmatrix}
        \widetilde X_1 \\ \widetilde X_2
    \end{bmatrix}
    \begin{bmatrix}
        \check V & \check V_{\perp}
    \end{bmatrix} = \begin{bmatrix}
        \check \Sigma  & 0 \\
        \widetilde X_2\check V  & \widetilde X_2\check V_{\perp}
    \end{bmatrix} \in \mathbb{R}^{p\times n}.
\end{equation} Note that $\check V$ and $\check V_{\perp}$ are only dependent on $\widetilde X_1$ and therefore $\check V$ and $\check V_{\perp}$ are independent of $\widetilde X_2$ as $\widetilde X_1$ and $\widetilde X_2$ are independent. We also note that the singular values of $\check X_n$, $\widetilde X_n$ and $X_n$ are the same because $\check X_n$ and $\widetilde X_n$ are orthogonal transformations of $X_n$. 

Now let us note that
\begin{align*}
    \sin\Theta\left(\begin{bmatrix}
        \check U \\ 0
    \end{bmatrix}, \begin{bmatrix}
        \widetilde U_1,\widetilde U_2
    \end{bmatrix}\right) &= \sin\Theta\left(\begin{bmatrix}
        \check U & 0 \\
       0  &  I_{(p-r_*)}
    \end{bmatrix}^T\begin{bmatrix}
        \check U \\ 0
    \end{bmatrix},\begin{bmatrix}
        \check U & 0 \\
       0  &  I_{(p-r_*)}
    \end{bmatrix}^T\begin{bmatrix}
        \widetilde U_1,\widetilde U_2
    \end{bmatrix}\right) \\
    &= \sin\Theta\left(\begin{bmatrix}
        I_{r_*} \\ 0
    \end{bmatrix}, \begin{bmatrix}
        \check U_1,\check U_2
    \end{bmatrix}\right)
\end{align*} where $\check U_1$ is the leading $r_1$ left singular vectors of $\check X_n$ and $\check U_2$ is the next $(r_2-r_1)$ leading left singular vectors. Now, using the triangle inequality, Lemma \ref{lem:triangle}, we obtain 
\begin{align*}
    \left\|{\sin\theta\left(\begin{bmatrix}
        I_{r_1} \\ 0
    \end{bmatrix},[\widetilde U_1,\widetilde U_2]\right)}\right\|_2 &\leq \left\|{\sin\theta\left(\begin{bmatrix}
        I_{r_1} \\ 0
    \end{bmatrix},\begin{bmatrix}
        \check U \\ 0
    \end{bmatrix}\right)}\right\|_2 +\left\|{\sin\theta\left(\begin{bmatrix}
        \check U \\ 0
    \end{bmatrix}, \begin{bmatrix}
        \widetilde U_1,\widetilde U_2
    \end{bmatrix}\right)}\right\|_2 \\ &=
    \left\|{\sin\Theta\left(\begin{bmatrix}
        I_{r_*} \\ 0
    \end{bmatrix}, \begin{bmatrix}
        \check U_1,\check U_2
    \end{bmatrix}\right)}\right\|_2
\end{align*} since 
\begin{equation}
    \left\|{\sin\theta\left(\begin{bmatrix}
        I_{r_1} \\ 0
    \end{bmatrix},\begin{bmatrix}
        \check U \\ 0
    \end{bmatrix}\right)}\right\|_2 = \left\|{\begin{bmatrix}
        I_{r_1} \\ 0
    \end{bmatrix}^T \begin{bmatrix}
        0 \\ I_{p-r_*}
    \end{bmatrix}}\right\|_2 = 0
\end{equation} because $\check U \in \mathbb{R}^{r_*\times r_*}$ is an orthogonal matrix and $r_1\leq r_*$.
Therefore, it suffices to bound
\begin{equation}
    \left\|{\sin\theta\left(\begin{bmatrix}
        I_{r_*} \\ 0
    \end{bmatrix}, \begin{bmatrix}
        \check U_1,\check U_2
    \end{bmatrix}\right)}\right\|_2 = \left\|{\begin{bmatrix}
        I_{r_*} \\ 0
    \end{bmatrix}^T\check U_{3}}\right\|_2 = \left\|{\check U_{31}}\right\|_2
\end{equation} instead, where $\check U_{3}\in \mathbb{R}^{p \times (p-r_2)}$ is the trailing $(p-r_2)$ left singular vectors of $\check X_n$ and $\check U_{31}$ is the first $r_*$ rows of $\check U_3$. 

We now analyze $\left\|{\check U_{31}}\right\|_2$ using a similar technique to one used in \cite{Nakatsukasa2020b}. Let $\check V_3 \in \mathbb{R}^{n\times (n-r_2)}$ be the trailing right singular vectors of $\check X$ and $\widehat\Sigma_3 = \begin{bmatrix}
        {\rm diag}(\widehat\sigma_{r_2+1},\widehat\sigma_{r_2+2},...,\widehat\sigma_{n}) \\ 0_{(p-n) \times (n-r_2)}
    \end{bmatrix}$ if $p > n$ or $\widehat\Sigma_3 = \begin{bmatrix}
        {\rm diag}(\widehat\sigma_{r_2+1},\widehat\sigma_{r_2+2},...,\widehat\sigma_{p}) & 0_{(p-r_2)\times (n-p)}
    \end{bmatrix}$ if $p \leq n$ where $\widehat\sigma_i$'s are the singular values of $X_n$, since $X_n$ and $\check X_n$ have the same singular values. Then we have $\check X_n \check V_3 = \check U_3 \widehat\Sigma_3$ and $\check X_n^T \check U_3 = \check V_3 \widehat\Sigma_3$. Now divide $\check U_3$ and $\check V_3$ into blocks as
    \begin{equation}
        \check U_3 = \begin{bmatrix}
            \check U_{31} \\ \check U_{32}
        \end{bmatrix}, \hspace{1cm} \check V_3 = \begin{bmatrix}
            \check V_{31} \\ \check V_{32}
        \end{bmatrix}
    \end{equation} where $\check U_{31}\in \mathbb{R}^{r_*\times (p-r_2)},\check U_{32}\in \mathbb{R}^{(p-r_*)\times (p-r_2)},\check V_{31}\in \mathbb{R}^{r_*\times (n-r_2)}$ and $\check V_{32}\in \mathbb{R}^{(n-r_*)\times (n-r_2)}$. Then using $\check X_n \check V_3 = \check U_3 \widehat\Sigma_3$, we get 
    \begin{equation} \label{eq:multsig1}
        \begin{bmatrix}
        \check \Sigma  & 0 \\
        \widetilde X_2\check V  & \widetilde X_2\check V_{\perp}
        \end{bmatrix} \begin{bmatrix}
            \check V_{31} \\ \check V_{32}
        \end{bmatrix} = \begin{bmatrix}
            \check U_{31} \\ \check U_{32} 
        \end{bmatrix} \widehat\Sigma_3
    \end{equation} and using $\check X_n^T \check U_3 = \check V_3 \widehat\Sigma_3$, we get
    \begin{equation} \label{eq:multsig2}
        \begin{bmatrix}
         \check \Sigma  & ( \widetilde X_2\check V )^T\\
        0 & (\widetilde X_2\check V_{\perp})^T
        \end{bmatrix}\begin{bmatrix}
            \check U_{31} \\ \check U_{32}
        \end{bmatrix}  = \begin{bmatrix}
            \check V_{31} \\ \check V_{32}
        \end{bmatrix} \widehat\Sigma_3.
    \end{equation} The first block of \eqref{eq:multsig1} and \eqref{eq:multsig2} give
    \begin{equation}
        \check \Sigma \check V_{31} = \check U_{31}\widehat\Sigma_3, \hspace{1cm}
        \check \Sigma \check U_{31} + (\widetilde X_2\check V)^T \check U_{32}= \check V_{31}\widehat\Sigma_3
    \end{equation} from which we get the following inequalities
    \begin{equation} \label{eq:multsigineq1}
        \sigma_{\min}(\check \Sigma) \left\|{\check V_{31}}\right\|_2 \leq \left\|{ \check \Sigma \check V_{31}}\right\|_2 = \left\|{\check U_{31}\widehat\Sigma_3}\right\|_2 \leq  \left\|{\check U_{31}}\right\|_2\left\|{\widehat\Sigma_3}\right\|_2
    \end{equation} and 
    \begin{equation} \label{eq:multsigineq2}
        \sigma_{\min}(\check \Sigma) \left\|{\check U_{31}}\right\|_2 \leq \left\|{\check \Sigma \check U_{31}}\right\|_2 = \left\|{\check V_{31}\widehat\Sigma_3-(\widetilde X_2\check V)^T \check U_{32}}\right\|_2 \leq  \left\|{\check V_{31}}\right\|_2\left\|{\widehat\Sigma_3}\right\|_2 + \left\|{(\widetilde X_2\check V)^T}\right\|_2.
    \end{equation}
    Now multiply \eqref{eq:multsigineq1} by $\left\|{\widehat\Sigma_3}\right\|_2$ and multiply \eqref{eq:multsigineq2} by $\sigma_{\min}(\check \Sigma)$ and then add them together to obtain
    \begin{align*}
        \left\|{\check U_{31}}\right\|_2 \leq \frac{\left\|{\widetilde X_2\check V}\right\|_2 \sigma_{\min}(\check \Sigma)}{(\sigma_{\min}(\check \Sigma))^2-\left\|{\widehat\Sigma_3}\right\|^2_2} = \frac{\left\|{\widetilde X_2 \check V}\right\|_2 \sigma_{\min}(\widetilde X_1)}{
        \sigma_{\min}(\widetilde X_1)^2 - \sigma_{r_2+1}(X_n)^2} 
    \end{align*}
Therefore 
\begin{equation} 
     \left\|{\sin\Theta(U_1,[\widehat U_1,\widehat U_2])}\right\|_2 \leq \frac{\left\|{\widetilde X_2 \check V}\right\|_2 \sigma_{\min}(\widetilde X_1)}{
        \sigma_{\min}(\widetilde X_1)^2 - \sigma_{r_2+1}(X_n)^2} = \frac{\left\|{U_{*,\perp}^T X_n \check V}\right\|_2 \sigma_{\min}(U_*^TX_n)}{
        \sigma_{\min}(U_*^TX_n)^2 - \sigma_{r_2+1}(X_n)^2}
\end{equation} for any $r_*\in [r_1,r_2]$. Minimizing with respect to $r_* \in [r_1,r_2]$, we get the desired result.
