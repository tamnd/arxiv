<!-- definition-1: body_md -->
\label{def:paraconvex}
    For given $p \in (1, 2]$, $\rho > 0$, a proper function $f: \mathbb{R}^n \to \mathbb{R} \cup \left\{ +\infty \right\}$ is called $(\rho, p)$-paraconvex if for all $x, y \in \mathbb{R}^n$ and $t \in [0, 1]$, the following holds:
    \begin{equation}\label{ineq-paraconvex1}
        f(tx+(1-t)y)\le tf(x)+(1-t)f(y)+\frac{\rho}{p}t(1-t)\|x-y\|^p.
    \end{equation}
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
Let $f: \mathbb{R}^n \to \mathbb{R} \cup \left\{ +\infty \right\}$ be a proper l.s.c. $(\rho, p)$-paraconvex function with $p \in (1, 2]$ and $\rho > 0$. Then the following claims hold:
    \begin{itemize}
        \par\hangindent 40\p@[(i)] $f$ is locally Lipschitz over $\operatorname{int}\text{dom}~ f$.
        \par\hangindent 40\p@[(ii)] The Fréchet subdifferential, the limiting subdifferential, and the Clark subdifferential of $f$ coincide. Throughout the rest of the paper, for paraconvex functions, we will simply refer to their subdifferential. As a consequence, the subdifferential of $f$ is non-empty over $\operatorname{int}\text{dom}~ f$. 
    \end{itemize}
<!-- end proposition-1 -->
<!-- theorem-1: body_md -->
\label{charac-thm} Let $f:\mathbb{R}^n\rightarrow\mathbb{R}\cup\{+\infty\} $ be a proper l.s.c. function. For $p \in (1, 2], \rho>0$, consider the three statements:
\begin{itemize}
\par\hangindent 40\p@[(i)] The function $f$ is $(\rho, p)$-paraconvex.
\par\hangindent 40\p@[(ii)] For all $x,y\in \mathbb{R}^n,$ $u\in\partial f(x),$ one has
\begin{equation}\label{charac-ineq1}
\langle u,y-x\rangle \le f(y)-f(x)+\frac{\rho}{p}\|x-y\|^p.
\end{equation}
\par\hangindent 40\p@[(iii)] For all $x,y\in\mathbb{R}^n;$ all $u\in \partial f(x),$ $v\in \partial f(y),$ one has
\begin{equation}\label{charac-ineq2}
\langle u-v,x-y\rangle\ge -\frac{2\rho}{p}\|x-y\|^p.
\end{equation}
\end{itemize}
One has $(i)\Rightarrow (ii)\Rightarrow (iii).$ Moreover, if (ii) holds then $f$ is $(\rho 2^{2-p}, p)$-paraconvex; while (iii) holds then $f$ is $(2\rho, p)$-paraconvex.
<!-- end theorem-1 -->
<!-- proposition-2: body_md -->
\label{propo:composite_paraconvexity}
    Let $h: \mathbb{R}^n \to R$ a convex and $L_h$-Lipschitz function with $L_h > 0$; $g: \mathbb{R}^m \to \mathbb{R}^n$ be a smooth mapping with $(\nu, L_c)$-Hölder continuous Jacobian where $\nu \in (0, 1]$ and $L_c > 0$. Then the composite function $f = h \circ c$ is $(\rho, 1+\nu)$-paraconvex with $\rho = \frac{2^{1-\nu} L_hL_c}{1+\nu}$.
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
For any $x \in \mathbb{R}^m$, let $J(x) \in \mathbb{R}^{n \times m}$ be the Jacobian of $c$ at $x$. The Hölder continuity of $J$ \cite{yashtini_global_2016} implies:
    \begin{align*}
        \left\lVert c(x) - c(y) - J(x) (y-x)  \right\rVert^{} \leq \frac{L_c}{1+\nu} \left\lVert x-y \right\rVert^{1+\nu} ~ \forall x, y \in \mathbb{R}^m.
    \end{align*}

    Let $g \in \partial f(x)$. By the chain rule, there exists $v \in \partial h(c(x))$ such that $g = J(x)^T v$. The convexity and the Lipschitzness of $h$ imply:
    \begin{align*}
        f(y) - f(x) - \langle g, y-x \rangle
        & = h(c(y)) - h(c(x)) - \langle J(x)^T v, y-x \rangle \\
        & \geq \langle v, c(y) - c(x) - J(x)^T (y-x) \rangle \\
        & \geq - \left\lVert v \right\rVert^{} \left\lVert  c(y)-c(x)-J(x)(y-x)  \right\rVert^{} \\
        & \geq -\frac{L_h L_c}{1+\nu} \left\lVert x-y \right\rVert^{1+\nu}.
    \end{align*}

    The claim follows from \cref{charac-thm}~(ii).
<!-- end proof-1 -->
<!-- example-1: body_md -->
\label[example]{example:composite_paraconvex}
    The robust phase retrieval problem \cite{davis_stochastic_2019,duchi_stochastic_2018} seeks to recover the true signal from perturbed measurements by solving the following weakly-convex optimization problem:
    \begin{align*}
        \min_{x \in \mathbb{R}^n} \frac{1}{m} \sum_{i=1}^m \left\lvert  \langle a_i, x \rangle^2 - b_i  \right\rvert,
    \end{align*}
    with Gaussian channels $a_i \in \mathbb{R}^n$ and noisy measurements $b_i \in \mathbb{R}$. In cases where the channels and the perturbed noises are heavy-tailed, one can consider the following alternative formulation:
    \begin{align*}
        \min_{x \in \mathbb{R}^n} \frac{1}{m} \sum_{i=1}^m \left\lvert  \left\lvert \langle a_i, x \rangle \right\rvert^p - b_i  \right\rvert.
    \end{align*}

    A simple argument show that the function $x \mapsto \left\lvert  \langle a, x \rangle  \right\rvert^p$ is $(\nu, L)$-Hölder smooth over $\mathbb{R}^n$ with $\nu=p-1$ and $L = p 2^{2-p} \left\lVert a \right\rVert^{p} $. Thus, paraconvexity of the objective function above follows from \cref{propo:composite_paraconvexity}. Numerical experiments demonstrating the advantage of using the paraconvex objective in terms of recovering the true signal in this heavy-tail setting are given in \cref{sec:experiment}.

<!-- definition-2: body_md -->
\label{def:paraweak}
    A proper function $f: \mathbb{R}^n \to \mathbb{R} \cup \left\{ +\infty \right\}$ is called $(\rho, \eta, p)$-paraconvex for $\rho > 0, \eta > 0$ and $p \in (1, 2]$ if $x \mapsto f(x) + \frac{\eta}{2} \left\lVert x \right\rVert^{2}$ is $(\rho, p)$-paraconvex. Equivalently, for all $x, y \in \mathbb{R}^n$, $t \in [0, 1]$, the following holds:
    \begin{equation}\label{eq:para-weak-convexity}
        f ( tx + (1-t)y ) \leq t f(x) + (1-t) f(y) + \frac{\rho t(1-t)}{p} \left\lVert x-y \right\rVert^{p} + \frac{\eta t(1-t)}{2} \left\lVert x-y \right\rVert^{2}.
    \end{equation}
<!-- end definition-2 -->
<!-- lemma-1: body_md -->
\label{lem:charac_paraweak}
    Let $f:\mathbb{R}^n\rightarrow\mathbb{R}\cup\{+\infty\} $ be a proper l.s.c. function. For $p>1,$ $\rho>0,$ consider the three statements:
    \begin{itemize}
    \par\hangindent 40\p@[(i)] The function $f$ is $(\rho, \eta, p)$-paraconvex.
    \par\hangindent 40\p@[(ii)] For all $x,y\in \mathbb{R}^n,$ $u\in\partial f(x),$ one has
    \begin{equation}\label{charac-paraweak-ineq1}
    \langle u,y-x\rangle \le f(y)-f(x)+\frac{\rho}{p}\|x-y\|^p + \frac{\eta}{2} \left\lVert x-y \right\rVert^{2} .
    \end{equation}
    \par\hangindent 40\p@[(iii)] For all $x,y\in\mathbb{R}^n;$ all $u\in\partial f(x),$ $v\in \partial f(y),$ one has
    \begin{equation}\label{charac-paraweak-ineq2}
    \langle u-v,x-y\rangle\ge -\frac{2\rho}{p}\|x-y\|^p - \eta \left\lVert x-y \right\rVert^{2} .

\label{thm:generalized_moreau_paraweak}

    \begin{itemize}
    \par\hangindent 40\p@[(i)] Assume that $f$ satisfies the growth condition:
    \begin{equation}\label{eq:growth_cond_1}
        f(x) \geq b - a h( \left\lVert x \right\rVert^{} ),~ \text{for some} ~ a > 0 ~ \text{and} ~ b \in \mathbb{R}, ~ \forall x \in \mathbb{R}^n \tag{G},
    \end{equation}
    and $h$ satisfies the following condition:
    \begin{equation}\label{eq:cond_h}
        \exists ~ c\ge 1 ~ \text{s.t. for any}~ d > 0, \exists ~ m > 0 ~ \text{s.t.} ~
        h(t+d) \leq ch(t)+m ~ \forall ~ t \in \mathbb{R}_+.\tag{H}
    \end{equation}
    Then for $0<r<(ac)^{-1},$ the function $M^{r}_f$ is locally Lipschitz, and  $P^{r}_f(x)\neq \emptyset$ for all $x\in\mathbb{R}^n$. Moreover, if $P^{r}_f(x)$ is a singleton then $M^{r}_f(x)$ is Fréchet differentiable at $x$.
    \par\hangindent 40\p@[(ii)] For $x\in\mathbb{R}^n,$ if $h$ satisfies the condition \cref{eq:cond_h} and is differentiable with $h'(0)=0$, $h'(t)>0$ for $t>0$, and $ P^{r}_f(x) $ is nonempty, then for all $z \in P^{r}_f(x) $,
    \begin{equation}\label{eq:subdif-regu-1}
    \hat\partial M^{r}_f(x) \subseteq \hat\partial f(z) \cap \left\{  r^{-1} \nabla j_h (x-z) \right\},
                                                                   
    \end{equation}
    where $j_h \coloneqq h \circ ||.||$. As a result, if $f$ satisfies the growth condition \cref{eq:growth_cond_1}, then for all $0<r<(ac)^{-1},$ either $\hat\partial M^{r}_f(x) $ is empty or it is singleton. Furthermore, when $\hat\partial M^{r}_f(x) $ is a singleton, the function $M^{r}_f$ is Fréchet differentiable at $x$.
    
    \par\hangindent 40\p@[(iii)] Assume $f$ satisfies the growth condition \cref{eq:growth_cond_1}. For $0<r<(ac)^{-1}$, and $h: \mathbb{R}_+ \to \mathbb{R}_+$ satisfies the assumptions in statement (ii), for all $x\in \mathbb{R}^n,$ the limiting subdifferential $\partial M^{r}_f (x) $ is nonempty, and there exists $z\in P^{r}_f (x) $ such that
    \begin{align}\label{eq:stationary_measure_1}
    d(0,\partial f(z)) \le d \left(0,\partial M^{r}_f(x) \right) &\coloneqq\inf\{\|v\|:\;\; v\in\partial M^{r}_f (x) \} \\
    &=r^{-1} \left\lVert  \nabla j_h (x-z)  \right\rVert^{} \label{eq:limit-subdif-estim-1} \notag .
    \end{align}
    \end{itemize}

\label{lem:unconstrained_descent_lem}
    Let $\{x_t\}$ be the sequence generated by \cref{alg:ssgd}, and for $0 < r < (2^{p-1}\rho)^{-1}$, let $z_t \in \gproxp[r]{x_t}$. Then, under \cref{assump:stochastic_grad_oracle}, one has:
    \begin{equation*}
        \mathbb{E}_{t}\!\left[ \left\lVert z_t - x_{t+1} \right\rVert^{p} \right] \leq \left\lVert z_t - x_t \right\rVert^{p} - (r^{-1} - \rho) \gamma_t \left\lVert z_t - x_t \right\rVert^{2(p-1)} + \left( 2^{p-1} p + 1 \right) \gamma_t^p L^p
    \end{equation*}

<!-- proof-2: body_md -->
By the update rule of \cref{alg:ssgd}, one can simply deduce:
    \begin{align*}
        \left\lVert z_t - x_{t+1} \right\rVert^{p}
        &= \left( \left\lVert z_t - x_t \right\rVert^{2} + 2\gamma_t \langle z_t-x_t, g_t \rangle + \gamma_t^2 \left\lVert g_t \right\rVert^{2} \right)^{p/2}.
    \end{align*}

    We have the following two cases.

    \textbf{Case 1:} $ \left\lVert z_t - x_t \right\rVert^{2} + 2 \gamma_t \langle z_t-x_t, g_t \rangle > 0$. Then by using \cref{useful-ineq}, one successively deduce:
    \begin{align*}
        & \left( \left\lVert z_t - x_t \right\rVert^{2} + 2\gamma_t \langle z_t-x_t, g_t \rangle + \gamma_t^2 \left\lVert g_t \right\rVert^{2} \right)^{p/2} \\
        & \leq \left( \left\lVert z_t - x_t \right\rVert^{2} + 2 \gamma_t \langle z_t - x_t, g_t \rangle \right)^{p/2} + \gamma_t^p \left\lVert g_t \right\rVert^{p} \\
        & \leq \left\lVert z_t-x_t \right\rVert^{p} + p \gamma_t \langle z_t-x_t, g_t \rangle \left\lVert z_t-x_t \right\rVert^{p-2} + \gamma_t^p \left\lVert g_t \right\rVert^{p}.
    \end{align*}

    \textbf{Case 2:} $  \left\lVert z_t-x_t \right\rVert^{2} + 2\gamma_t \langle z_t-x_t, g_t \rangle \leq 0  $. From this condition, one has:
    \begin{align*}
        \left\lVert z_t-x_t \right\rVert^{2} \leq 2 \gamma_t \langle x_t-z_t, g_t \rangle \leq 2 \gamma_t \left\lVert z_t-x_t \right\rVert^{}\left\lVert g_t \right\rVert^{},
    \end{align*}
    which implies
    \begin{equation}\label{eq:descent_lem_unconstrained_proof_1}
        \left\lVert z_t-x_t \right\rVert^{} \leq 2 \gamma_t \left\lVert g_t \right\rVert^{}.
    \end{equation}

    Then, by the fact that $a \mapsto a^{p/2}$ is non-decreasing for $a \geq 0$ and $p \in (1, 2]$, one obtains:
    \begin{align*}
        & \left( \left\lVert z_t - x_t \right\rVert^{2} + 2\gamma_t \langle z_t-x_t, g_t \rangle + \gamma_t^2 \left\lVert g_t \right\rVert^{2} \right)^{p/2} \\
                                            
        & \leq \left\lVert z_t-x_t \right\rVert^{p} + p \gamma_t \langle z_t-x_t, g_t \rangle \left\lVert z_t-x_t \right\rVert^{p-2} + \gamma_t^p \left\lVert g_t \right\rVert^{p} \\
        & \quad - p \gamma_t \langle z_t-x_t, g_t \rangle \left\lVert z_t-x_t \right\rVert^{p-2}.
    \end{align*}

    By using \cref{eq:descent_lem_unconstrained_proof_1}, the last term above can be upper-bounded as:
    \begin{align*}
        &- p \gamma_t \langle z_t-x_t, g_t \rangle \left\lVert z_t-x_t \right\rVert^{p-2}
        \leq 2^{p-1}p \gamma_t^p \left\lVert g_t \right\rVert^{p},
    \end{align*}
    which gives
    \begin{align*}
        &\left( \left\lVert z_t - x_t \right\rVert^{2} + 2\gamma_t \langle z_t-x_t, g_t \rangle + \gamma_t^2 \left\lVert g_t \right\rVert^{2} \right)^{p/2} \\
        & \leq \left\lVert z_t-x_t \right\rVert^{p} + p \gamma_t \langle z_t-x_t, g_t \rangle \left\lVert z_t-x_t \right\rVert^{p-2} + (2^{p-1}p+1) \gamma_t^p \left\lVert g_t \right\rVert^{p}.
    \end{align*}

    Combining the two cases, taking the conditional expectation, together with using \cref{assump:stochastic_grad_oracle}, one has:
    \begin{align}
        \mathbb{E}_{t}\!\left[ \left\lVert z_t-x_{t+1} \right\rVert^{p} \right]
        & \leq \left\lVert z_t-x_t \right\rVert^{p} + p \gamma_t \langle z_t-x_t, v_t \rangle \left\lVert z_t-x_t \right\rVert^{p-2} \nonumber \\
        & \quad + (2^{p-1}p+1) \gamma_t^p L^p,\label{eq:descent_lem_unconstrained_proof_2}
    \end{align}
    with $v_t \in \partial F(x_t)$. By $(\rho, p)$-paraconvexity of $F$, the second term in \cref{eq:descent_lem_unconstrained_proof_2} can be estimated as:
    \begin{subequations}
        \begin{align}
            & p \gamma_t \langle z_t-x_t, v_t \rangle \left\lVert z_t-x_t \right\rVert^{p-2} \nonumber \\
            & \leq p\gamma_t \left( F(z_t)-F(x_t) + \frac{\rho}{p} \left\lVert z_t-x_t \right\rVert^{p} \right) \left\lVert z_t-x_t \right\rVert^{p-2} \label{eq:descent_lem_unconstrained_proof_3} \\
            & \leq p \gamma_t \left( \frac{\rho}{p} - \frac{1}{rp} \right) \left\lVert z_t-x_t \right\rVert^{p} \left\lVert z_t-x_t \right\rVert^{p-2} \label{eq:descent_lem_unconstrained_proof_4} \\
            & = - (r^{-1} - \rho) \gamma_t \left\lVert z_t-x_t \right\rVert^{2(p-1)}. \label{eq:descent_lem_unconstrained_proof_5}
        \end{align}
    \end{subequations}

    \Cref{eq:descent_lem_unconstrained_proof_3} follows from \cref{charac-ineq1}; while \cref{eq:descent_lem_unconstrained_proof_4} follows from $z_t \in P^{p, r}_F \left( x_t \right)$, which implies:
    \begin{align*}
        F(z_t) + \frac{1}{rp} \left\lVert z_t-x_t \right\rVert^{p} \leq F(x_t).
    \end{align*}

    Putting \cref{eq:descent_lem_unconstrained_proof_5} back into \cref{eq:descent_lem_unconstrained_proof_2} concludes the proof.

\label{thm:ssgd} Under \cref{assump:stochastic_grad_oracle}, for $0 < r < (2^{p-1} \rho)^{-1}$, the sequence $\left\{ x_t \right\}$ generated by \cref{alg:ssgd} satisfies:
\begin{align}\label{eq:unconstrained_conv_1}
&\mathbb{E}_{}\!\left[ \sum_{t=0}^T \gamma_t d \left(0, \partial \moreaup[r]{x_t}\right)^2 \right] \nonumber \\
&\le \frac{p}{1-r\rho}  \left(  \moreaup[r]{x_0} - \min F  \right) + \frac{ \left(  2^{p-1}p+1  \right)L^p }{r(1-r\rho)} \sum_{t=0}^T \gamma_t^p.
\end{align}

In particular, if \cref{alg:ssgd} uses step size $\gamma_t = (t+1)^{-1/p}$, then its output $x_{t^*}$ will satisfy:
\begin{align}\label{eq:unconstrained_conv_2}
    & \mathbb{E}_{}\!\left[ d \left( 0, \partial M^{1/2\rho}_F (x_{t^*}) \right)^2 \right] \nonumber \\
    & \leq c_1 ~ \frac{\moreaup[1/2\rho]{x_0}-\min F}{(T+2)^{1-1/p}-1} + c_2 ~ \frac{L^p (\ln\left( T+1 \right)+1)}{(T+2)^{1-1/p}-1},
                                                                                                                                                           
\end{align}
with 
\begin{equation*}
    c_1 = 2 (p-1), \quad c_2 = \frac{ 4\rho (p-1) \left(  2^{p-1}p+1  \right) }{p}.
\end{equation*}

<!-- proof-3: body_md -->
For any $t \in [0, T]$, by \cref{thm:generalized_moreau_paraweak}~(iii), one can pick $z_t \in \gproxp[r]{x_t}$ such that
    \begin{equation}\label{eq:unconstrained_conv_proof_0}
        d \left(  0, \moreaup[r]{x_t}  \right) = r^{-1} \left\lVert x_t-z_t \right\rVert^{p-1}.
    \end{equation}
    
    By definition, one successively deduces:
    \begin{subequations}
    \begin{align}
        & \mathbb{E}_{t}\!\left[\moreaup[r]{x_{t+1}}\right] \nonumber \\
        & \leq F(z_t) + \frac{1}{rp} \mathbb{E}_{t}\!\left[\left\lVert z_t-x_{t+1} \right\rVert^{p}\right] \label{eq:unconstrained_conv_proof_1} \\
        & \leq \moreaup[r]{x_t} - \frac{ r^{-1} - \rho }{rp} \gamma_t \left\lVert z_t-x_t \right\rVert^{2(p-1)}
        + \frac{\left(  2^{p-1}p+1  \right)L^p}{rp} \gamma_t^p \label{eq:unconstrained_conv_proof_2},
    \end{align}
    \end{subequations}
    where \cref{eq:unconstrained_conv_proof_1} follows directly from the definition of the generalized proximal mapping, and \cref{eq:unconstrained_conv_proof_2} follows from \cref{lem:unconstrained_descent_lem}. By rearranging, taking the full expectation, then taking the sum from $t=0$ to $T$, one arrives at:
    \begin{align}
        &\mathbb{E}_{}\!\left[ \sum_{t=0}^{T} \gamma_t \left\lVert z_t-x_t \right\rVert^{2(p-1)} \right] \nonumber \\
        & \leq \frac{rp}{r^{-1}-\rho} \left(\moreaup[r]{x_0} - \min F \right)
        + \frac{\left(  2^{p-1}p+1  \right)L^p}{r^{-1}-\rho} \sum_{t=0}^T \gamma_t^p. \label{eq:unconstrained_conv_proof_3}
    \end{align}

    Noticing that, by \cref{eq:unconstrained_conv_proof_0}, one has:
    \begin{equation*}
        d\left(0, \partial \moreaup[r]{x_t}\right)^2 = r^{-2} \left\lVert z_t-x_t \right\rVert^{2(p-1)},
    \end{equation*}
    which, together with \cref{eq:unconstrained_conv_proof_3}, proves \cref{eq:unconstrained_conv_1}. Let $r = 1/2\rho$, then by dividing two sides of \cref{eq:unconstrained_conv_1} by $\sum_{t=0}^T \gamma_t$, one obtains:
    \begin{align*}
        &\mathbb{E}_{}\!\left[ d \left( 0, \partial M^{1/2\rho}_F (x_{t^*}) \right)^2 \right] \\
        & \leq \frac{2p \left(  \moreaup[r]{x_0} - \min F  \right) }{\sum_{t=0}^T \gamma_t} + 4 \rho \left(  2^{p-1}p+1  \right) L^p \frac{ \sum_{t=0}^T \gamma_t^p }{\sum_{t=0}^T \gamma_t}.
    \end{align*}

    Using \cref{fact:sum_step_size} gives:
    \begin{align*}
        & \sum_{t=0}^T \gamma_t = \sum_{t=0}^T (t+1)^{-1/p} \geq \frac{p}{p-1} \left(  (T+2)^{1-1/p}-1  \right),\\ 
        & \sum_{t=0}^T \gamma_t^p = \sum_{t=0} (t+1)^{-1} \leq \ln\left( T+1 \right) + 1,
    \end{align*}
    which proves \cref{eq:unconstrained_conv_2}.

Let $\hat{v}_t \in \partial f(z_t)$. Then, by using \cref{lem:relating_2_proximal} and the update rule of \cref{alg:psgd}, together with the non-expansiveness of $\operatorname{prox}_{\gamma_t \varphi}(.)$, one successively deduces:
     \begin{subequations}
     \begin{align}
         &\left\lVert z_t - x_{t+1} \right\rVert^{p} \nonumber \\
         &= \left\lVert \operatorname{prox}_{} \left( z_t - r^{-1}\gamma_t \left\lVert z_t-x_t \right\rVert^{p-2}(z_t-x_t) - \gamma_t \hat{v}_t \right) - \operatorname{prox}_{} \left( x_t - \gamma_t g_t \right) \right\rVert^{p} \nonumber \\
         & \leq \left\lVert  \alpha_t (z_t-x_t) - \gamma_t (\hat{v}_t-g_t)  \right\rVert^{p} \label{eq:descent_lem_cp_proof} \\
         & \leq \left(  \alpha_t^2 \left\lVert z_t-x_t \right\rVert^{2} - 2\gamma_t\alpha_t \langle z_t-x_t, \hat{v}_t-g_t \rangle + \gamma_t^2 \left\lVert \hat{v_t}-g_t \right\rVert^{2}  \right)^{p/2},\label{eq:descent_lem_cp_proof_1}
     \end{align}
     \end{subequations}
     with $\alpha_t = 1-r^{-1}\gamma_t \left\lVert z_t-x_t \right\rVert^{p-2} $. We first prove the following claim:

     \textbf{Claim 1:} If $\alpha_t > 0$, then:
     \begin{align*}
         & \mathbb{E}_{t}\!\left[ \left\lVert z_t-x_{t+1} \right\rVert^{p} \right] \\
         & \leq \left\lVert z_t-x_t \right\rVert^{p} - (r^{-1}-2\rho) \gamma_t \left\lVert z_t-x_t \right\rVert^{2(p-1)} + 2^p (1+2^{p-1}p) \gamma_t^p L^p.
     \end{align*}

     To prove \textbf{Claim 1}, we need to distinguish the following two cases.

     \textbf{Case 1:} $\alpha_t^2 \left\lVert z_t-x_t \right\rVert^{2} - 2 \gamma_t \alpha_t \langle  z_t-x_t, \hat{v}_t - g_t  \rangle > 0 $. Then, from \cref{eq:descent_lem_cp_proof_1}, by applying successively the inequalities in \cref{useful-ineq}, one obtains:
     \begin{align}
         & \left\lVert z_t-x_{t+1} \right\rVert^{p} \nonumber \\
         & \leq \alpha_t^p \left\lVert z_t-x_t \right\rVert^{p} + p\gamma_t \alpha_t^{p-1} \langle  z_t-x_t, g_t-\hat{v}_t  \rangle \left\lVert z_t-x_t \right\rVert^{p-2}
          + \gamma_t^p { \left\lVert \hat{v}_t-g_t \right\rVert^{p} } .\label{eq:descent_lem_cp_proof_2}
     \end{align}

     \textbf{Case 2:} $\alpha_t^2 \left\lVert z_t-x_t \right\rVert^{2} - 2\gamma_t \alpha_t \langle  z_t-x_t, \hat{v}_t-g_t  \rangle \leq 0$. This condition gives:
     \begin{equation}\label{eq:descent_lem_cp_proof_3}
        \alpha_t \left\lVert z_t-x_t \right\rVert^{} \leq 2\gamma_t \left\lVert \hat{v}_t-g_t \right\rVert^{}.
     \end{equation}

     By using the fact that $a \mapsto a^{p/2}$ is non-increasing for $a \geq 0$ and $p \in (1, 2]$, from \cref{eq:descent_lem_cp_proof_1}, one arrives at:
     \begin{align}
         & \left\lVert z_t-x_{t+1} \right\rVert^{p} \nonumber \\
         & \leq \alpha_t^p \left\lVert z_t-x_t \right\rVert^{p} + p\gamma_t \alpha_t^{p-1} \langle  z_t-x_t, g_t-\hat{v}_t  \rangle \left\lVert z_t-x_t \right\rVert^{p-2}
          + \gamma_t^p { \left\lVert \hat{v}_t-g_t \right\rVert^{p} } \nonumber \\
         & \quad + p\gamma_t \alpha_t^{p-1} \langle  z_t-x_t, \hat{v_t}-g_t  \rangle \left\lVert z_t-x_t \right\rVert^{p-2} \nonumber \\
         & \leq \alpha_t^p \left\lVert z_t-x_t \right\rVert^{p} + p\gamma_t \alpha_t^{p-1} \langle  z_t-x_t, g_t-\hat{v}_t  \rangle \left\lVert z_t-x_t \right\rVert^{p-2} \nonumber \\
         & \quad + \left(  1+2^{p-1}p  \right) \gamma_t^p { \left\lVert \hat{v}_t-g_t \right\rVert^{p} }, \label{eq:descent_lem_cp_proof_4}
     \end{align}

     \begin{align}
         & \mathbb{E}_{t}\!\left[\left\lVert z_t-x_{t+1} \right\rVert^{p}\right] \nonumber \\
         & \leq \alpha_t^p \left\lVert z_t-x_t \right\rVert^{p} + p\gamma_t \alpha_t^{p-1} \langle  z_t-x_t, \mathbb{E}_t g_t-\hat{v}_t  \rangle \left\lVert z_t-x_t \right\rVert^{p-2} \nonumber \\

        & \leq \frac{c_5~(\moreaup[1/4\rho]{x_0}-\min F) + c_6}{(T+2)^{1-1/p}-1} + \frac{c_7 L^p (\ln\left( T+1 \right)+1)}{(T+2)^{1-1/p}-1},

    Some remarks are as follows:

\label[assumption]{assump:stochastic_one_sided_model}
    Fix a probability space $\left( \Omega, \mathcal{F}, \pi \right)$, and equip $\mathbb{R}^n$ with the Borel $\sigma$-algebra. We assume that there exist positive reals $p \in (1, 2], \nu \in (0, 1], \tau, \eta, L_1, L_2 \in \mathbb{R}^+$ such that the followings hold:
    \begin{itemize}
    \par\hangindent 40\p@[(i)] We can draw i.i.d. realizations $\xi_1, \xi_2, \dots$ from $\pi$.
    \par\hangindent 40\p@[(ii)] There exists an open set $U \supset \text{dom}~ \varphi$ on which $f$ is finite, and a measurable function $(x, y, \xi) \mapsto f_x(y, \xi)$, defined on $U \times U \times \Omega$ satisfying
    \begin{align}
        \mathbb{E}_{\xi}\!\left[ f_x(x, \xi) \right] &= f(x) \quad \forall x \in U, \label{eq:stochastic_one_sided_model_1} \\
        \mathbb{E}_{\xi}\!\left[ f_x(y, \xi) - f(y) \right] & \leq \frac{\tau}{p} \left\lVert x-y \right\rVert^{p} \quad \forall x,y \in U. \label{eq:stochastic_one_sided_model_2}
    \end{align}
    \par\hangindent 40\p@[(iii)] For all $x \in U$, and a.e. $\xi \sim \pi$, the function $f_x(., \xi) + \varphi(.)$ is $\eta$-weakly convex.
    \par\hangindent 40\p@[(iv)] There exists measurable functions $\mathcal{L}_1, \mathcal{L}_2: \Omega \to \mathbb{R}^+$ such that, for all $x, y \in U$, and a.e. $\xi \sim \pi$:
    \begin{align}
        & \left(  \mathbb{E}_{\xi}\!\left[ \mathcal{L}_1(\xi)^2\right]  \right)^{1/2} \leq L_1,\quad \left(  \mathbb{E}_{\xi}\!\left[\mathcal{L}_2(\xi)^{\nu+1}\right]  \right)^{1/(\nu+1)} \leq L_2, \nonumber \\
        & f_x(x, \xi) - f_x(y, \xi)  \leq \mathcal{L}_1(\xi) \left\lVert x-y \right\rVert^{} + \mathcal{L}_2 (\xi) \left\lVert x-y \right\rVert^{\nu} . \label{eq:stochastic_one_sided_model_3}
    \end{align}
    \end{itemize}

            & \quad + \mathbb{E}_{t}\!\left[ \mathcal{L}_1 (\xi_t) \left\lVert x_{t+1}-x_t \right\rVert^{} + \mathcal{L}_2(\xi_t) \left\lVert x_{t+1}-x_t \right\rVert^{\nu} \right] \label{eq:model_based_descent_lem_2} \\
            & \leq \mathbb{E}_{t}\!\left[F(z_t) - F(x_{t+1}\right] + \frac{\tau}{p} \left\lVert z_t-x_t \right\rVert^{p} \nonumber \\
            & \quad + \mathbb{E}_{t}\!\left[ \mathcal{L}_1 (\xi_t) \left\lVert x_{t+1}-x_t \right\rVert^{} + \mathcal{L}_2(\xi_t) \left\lVert x_{t+1}-x_t \right\rVert^{\nu} \right] \nonumber \\
            & \quad + \mathbb{E}_{t}\!\left[ L_1 \left\lVert x_{t+1}-x_t \right\rVert^{} + L_2 \left\lVert x_{t+1}-x_t \right\rVert^{\nu} + \frac{\tau}{p} \left\lVert x_{t+1}-x_t \right\rVert^{p} \right] \label{eq:model_based_descent_lem_3} \\
            & \leq \mathbb{E}_{t}\!\left[F(z_t) - F(x_{t+1})\right] + \frac{\tau}{p} \left\lVert z_t-x_t \right\rVert^{p} + \frac{\tau}{p} \mathbb{E}_{t}\!\left[\left\lVert x_{t+1}-x_t \right\rVert^{p}\right] \nonumber \\
            & \quad + L_1 \left(  \left(  \mathbb{E}_{t}\!\left[ \left\lVert x_{t+1}-x_t \right\rVert^{2} \right]  \right)^{1/2} + \mathbb{E}_{t}\!\left[\left\lVert x_{t+1}-x_t \right\rVert^{}\right]  \right) \nonumber \\
            & \quad + L_2 \left(  \left(  \mathbb{E}_{t}\!\left[ \left\lVert x_{t+1}-x_t \right\rVert^{\nu+1} \right]  \right)^{\frac{\nu}{\nu+1}} + \mathbb{E}_{t}\!\left[\left\lVert x_{t+1}-x_t \right\rVert^{\nu}\right]  \right), \label{eq:model_based_descent_lem_4}
        \end{align}
    \end{subequations}
    where \cref{eq:model_based_descent_lem_1} follows from \cref{eq:stochastic_one_sided_model_2}, \cref{eq:model_based_descent_lem_2} follows from \cref{eq:stochastic_one_sided_model_3}, \cref{eq:model_based_descent_lem_3} follows from \cref{eq:model_based_lipschitz}, and finally \cref{eq:model_based_descent_lem_4} uses Hölder's inequality. Then, the definition for $z_t$ gives:
    \begin{align*}
        & F(z_t) + \frac{\bar{\rho}}{p}\left\lVert z_t-x_t \right\rVert^{p} + \frac{\bar{\rho}}{2}\left\lVert z_t-x_t \right\rVert^{2} \\
        & \leq F(x_{t+1}) + \frac{\bar{\rho}}{p}\left\lVert x_{t+1}-x_t \right\rVert^{p} + \frac{\bar{\rho}}{2}\left\lVert x_{t+1}-x_t \right\rVert^{2}.
    \end{align*}

    Letting $\delta_t = \left(  \mathbb{E}_{t}\!\left[ \left\lVert x_{t+1}-x_t \right\rVert^{2} \right]  \right)^{1/2}$, by Jensen's inequality, one has:
    \begin{align*}
        & \mathbb{E}_{t}\!\left[\left\lVert x_{t+1}-x_t \right\rVert^{}\right] \leq \delta_t, \quad 
        \mathbb{E}_{t}\!\left[\left\lVert x_{t+1}-x_t \right\rVert^{2}\right] = \delta_t^2, \\
        & \mathbb{E}_{t}\!\left[ \left\lVert x_{t+1}-x_t \right\rVert^{p} \right] \leq \delta_t^p, \quad \mathbb{E}_{t}\!\left[ \left\lVert x_{t+1}-x_t \right\rVert^{\nu} \right] \leq \delta_t^{\nu},\\
        & \left(  \mathbb{E}_{t}\!\left[\left\lVert x_{t+1}-x_t \right\rVert^{\nu+1} \right]  \right)^{\frac{\nu}{\nu+1}} \leq \delta_t^{\nu}.
    \end{align*}

    Plugging back into \cref{eq:model_based_descent_lem_4}, after rearranging and dividing two sides by $(\beta_t-\eta)/2$, one arrives at:
    \begin{align}\label{eq:model_based_descent_lem_6}
        & \mathbb{E}_{t}\!\left[\left\lVert z_t-x_{t+1} \right\rVert^{2}\right] \nonumber \\
        & \leq \left\lVert z_t-x_t \right\rVert^{2} - \frac{A_{\bar{\rho}, \eta}}{\beta_t - \eta} \left\lVert z_t-x_t \right\rVert^{2} - \frac{B_{\bar{\rho}, \tau, p}}{\beta_t-\eta} \left\lVert z_t-x_t \right\rVert^{p} + \mathcal{P}(\delta_t)(\beta_t-\eta)^{-1},
    \end{align}
    with
    \begin{align}
       & A_{\bar{\rho}, \eta} = \bar{\rho} - \eta, \quad B_{\bar{\rho}, \tau, p} = \frac{2(\bar{\rho}-\tau)}{p}, \label{eq:model_based_constant_1} \\
       & \mathcal{P}(\delta_t) = - (\beta_t-\bar{\rho}) \delta_t^2 + \frac{2(\bar{\rho}+\tau)}{p} \delta_t^p + 4L_1\delta_t + 4L_2 \delta_t^{\nu} \nonumber.
    \end{align}

    Now we will try to derive an upper bound for $\mathcal{P}(\delta_t)$ that does not depend on $\delta_t$. Thanks to Young's inequality:
    \begin{align*}
        ab \leq \frac{a^p}{p} + \frac{b^q}{q}, \quad a, b > 0, ~ p, q > 1, ~ \frac{1}{p} + \frac{1}{q} = 1,
    \end{align*}
    one can have, if $p \in (1, 2)$,
    \begin{align*}
        \frac{2(\bar{\rho}+\tau)}{p} \delta_t^p
        &\leq \frac{\beta_t-\bar{\rho}}{3} \delta_t^2 + (2-p) p^{-1} 3^{\frac{p}{2-p}} (\bar{\rho}+\tau)^{\frac{2}{2-p}} (\beta_t-\bar{\rho})^{-\frac{p}{2-p}}, \\
        4L_1 \delta_t &\leq \frac{\beta_t-\bar{\rho}}{3}\delta_t^2 + 12L_1^2 (\beta_t-\bar{\rho})^{-1} \\
        4L_2 \delta_t^{\nu} & \leq \frac{\beta_t-\bar{\rho}}{3}\delta_t^2 + (2-\nu) 2^{\frac{2}{2-\nu}} (3\nu)^{\frac{\nu}{2-\nu}} L_2^{\frac{2}{2-\nu}} (\beta_t-\bar{\rho})^{-\frac{\nu}{2-\nu}}.
    \end{align*}

    If $p=2$,
    \begin{align*}
        4L_1 \delta_t &\leq \frac{\beta_t-2\bar{\rho}-\tau}{2} \delta_t^2 + 8 L_1^2 (\beta_t-2\bar{\rho}-\tau)^{-1} \\
        4L_2 \delta_t^{\nu} &\leq \frac{\beta_t-2\bar{\rho}-\tau}{2} \delta_t^2 + (2-\nu) 2^{\frac{2}{2-\nu}} (2\nu)^{\frac{\nu}{2-\nu}} (\beta_t-2\bar{\rho}-\tau)^{-\frac{\nu}{2-\nu}}.
    \end{align*}

         
                    
                               
                                                 
                                             
                      
                  
    Define the following constants:
    \begin{align}
        C_{\bar{\rho}, \tau, p} &= \begin{cases}
            (2-p) p^{-1} 3^{\frac{p}{2-p}} (\bar{\rho}+\tau)^{\frac{2}{2-p}} & \text{if} ~ p \in (1, 2), \\
            0 & \text{if} ~ p = 2,
        \end{cases} \label{eq:model_based_constant_2} \\
        D &= \begin{cases}
            12 & \text{if} ~ p \in (1, 2), \\
            8 & \text{if} ~ p = 2,
        \end{cases} \label{eq:model_based_constant_3} \\
        E_{\nu} &= \begin{cases}
            (2-\nu) 2^{\frac{2}{2-\nu}} (3\nu)^{\frac{\nu}{2-\nu}} & \text{if} ~ p \in (1, 2), \\
            (2-\nu) 2^{\frac{2}{2-\nu}} (2\nu)^{\frac{\nu}{2-\nu}} & \text{if} ~ p = 2,
        \end{cases} \label{eq:model_based_constant_4}
    \end{align}
    then $\mathcal{P}(\delta_t)$ can be upper bounded as:
    \begin{align}\label{eq:model_based_descent_lem_5}
        \mathcal{P}(\delta_t) \leq C_{\bar{\rho}, \tau, p} (\beta_t-\tilde{\rho})^{-\frac{p}{2-p}} + D L_1^2 (\beta_t-\tilde{\rho})^{-1} + E_{\nu} L_2^{\frac{2}{2-\nu}} (\beta_t -\tilde{\rho})^{-\frac{\nu}{2-\nu}},
    \end{align}
    which, together with \cref{eq:model_based_descent_lem_6} proves the lemma.

To simply the notation, let $\mathcal{P}$ be the upper bound of $\mathcal{P}(\delta_t)$ in \cref{eq:model_based_descent_lem_5}. Then, by Jensen's inequality, one has:

        \leq \left(  \mathbb{E}_{t}\!\left[\left\lVert z_t-x_{t+1} \right\rVert^{2}\right]  \right)^{p/2} \nonumber \\

        & \qquad + \left. (\beta_t-\eta)^{-1} \mathcal{P} \right)^{p/2}, \label{eq:model_based_lem_p}

    Combining the results from the two cases proves the lemma.

Dividing two sides of \cref{eq:model_based_conv_1} by $\sum_{t=0}^T \mathcal{A} (\beta_t-\eta)^{-1}$ gives:

        &\mathbb{E}_{}\!\left[ d \left(0, \moreaup[1/\bar{\rho}]{x_{t^*}} \right)^2 \right] \nonumber \\
        &\le \frac{2 \bar{\rho} \left(  \moreaup[1/\bar{\rho}]{x_0} - \min F  \right)}{\sum_{t=0}^T \mathcal{A}(\beta_t-\eta)^{-1}} \nonumber \\
        &\quad+ \frac{\bar{\rho}^2 \mathcal{B}}{\sum_{t=0}^T \mathcal{A}(\beta_t-\eta)^{-1}} \sum_{t=0}^T \left( 
            (\beta_t-\tilde{\rho})^{-\frac{2}{2-p}}
            + L_1^2 (\beta_t-\tilde{\rho})^{-2}
            + (\beta_t-\tilde{\rho})^{-\frac{p}{2-p}} \right. \nonumber \\
            &\qquad\qquad\qquad\qquad\qquad\qquad \left.+ L_1^p (\beta_t-\tilde{\rho})^{-p}
        \right).
    \end{align}

    Then, by appealing to \cref{fact:sum_step_size}, one has the following estimates:
    \begin{align*}
        & \sum_{t=0}^T (\beta_t-\eta)^{-1} \geq \frac{T+1}{\tilde{\rho}-\eta + (T+1)^{1/p}} \\
        & \sum_{t=0}^T (\beta_t-\tilde{\rho})^{-\frac{2}{2-p}} = \sum_{t=0}^T (t+1)^{-\frac{2}{p(2-p)}}
        \leq \frac{2}{(p-1)^2+1}, \\
        & \sum_{t=0}^T (\beta_t-\tilde{\rho})^{-2} = \sum_{t=0}^T (t+1)^{-2/p} 
        \leq \begin{cases}
             \frac{2}{2-p} & \text{if}~ p \in (1, 2), \\
             \ \ln(T+1)+1 & \text{if}~ p = 2,
        \end{cases} \\
        & \sum_{t=0}^T (\beta_t-\tilde{\rho})^{-\frac{p}{2-p}} = \sum_{t=0}^T (t+1)^{-\frac{1}{2-p}}
        \leq \frac{1}{p-1}, \\
        & \sum_{t=0}^T (\beta_t-\tilde{\rho})^{-p} = \sum_{t=0}^T (t+1)^{-1} \leq \ln(T+1)+1.
    \end{align*}

    Plugging back into \cref{eq:model_based_cor_1} proves the results.
