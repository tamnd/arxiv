\label{teo_intromain}
Let $\Omega \subseteq \mathbb{H}^n$ be an open set and $u \in \operatorname{SBV}_\mathbb{H}(\Omega)$. Then
\[
\lim_{r \to 0} k_r(u) = \frac{1}{4} \int_\Omega |D^{\operatorname{ap}}_\mathbb{H} u|d\mathcal{L}^{2n+1}+\frac{1}{2}|D_\mathbb{H}^ju|(\mathcal{J}_u).
\]

<!-- defi-1: body_md -->
For $n\geq 1$ we denote by $\mathbb{H}^n$ the $n$-th {\em Heisenberg group}, identified with $\mathbb{R}^{2n+1}$ through exponential coordinates. We denote a point $\mathbb{H}^n \ni p=(z,t)$ by $z \in \mathbb{R}^{2n}$ and $t \in \mathbb{R}$. If $p=(z,t),q=(z',t') \in \mathbb{H}^n$, the group operation is defined as 
\[
p \cdot q \coloneqq (z+z',t+t'+2\omega(z,z')),
\qquad
\omega(z,z') \coloneqq \sum_{j=1}^n(x'_jy_j-x_jy'_j),
\]
where $z=(x,y), z'=(x',y')$ with $x,y,x',y' \in \mathbb{R}^n$. If $\mathbb{H}^n \ni p=(z,t)$, its inverse is $p^{-1}=(-z,-t)$ and $0=(0,0)\in \mathbb{H}^n$ is the identity of $\mathbb{H}^n$. For $\lambda >0$, we denote by $\delta_\lambda\colon\mathbb{H}^n\to\mathbb{H}^n$ the dilations of the Heisenberg groups defined, for $p=(z,t)\in\mathbb{H}^n$, by $\delta_\lambda(z,t) \coloneqq (\lambda z, \lambda^2 t)$. Observe that dilations form a one-parameter family of group isomorphisms.

The {\em homogeneous dimension} of $\mathbb{H}^n$ is $2n+2$. The Lebesgue measure $\mathcal L^{2n+1}$ is the Haar measure on $\mathbb{H}^n\equiv\mathbb{R}^{2n+1}$ and it is $(2n+2)$-homogeneous with respect to dilations. The $(2n+1)$-dimensional Lie algebra $\mathfrak{h}^n$ of left invariant vector fields is generated by the vector fields $X_1,\dots,X_n,Y_1,\dots,Y_n,T$ where (for $1 \leq j \leq n$)
\[
X_j \coloneqq \partial_{x_j}+2y_j\partial_t,
\qquad
Y_j \coloneqq \partial_{y_j}-2x_j\partial_t,
\qquad
T \coloneqq \partial_t.
\]
We denote by $\mathfrak h^n_1$ the horizontal subspace of $\mathfrak h^n$, i.e.,
\[
\mathfrak h^n_1 \coloneqq \operatorname{span}(X_1,\dots,X_n,Y_1,\dots,Y_n),
\]
and by $\mathfrak h^n_2$ the linear span of $T$; the Lie algebra $\mathfrak h^n$ admits the 2-step stratification $\mathfrak h^n=\mathfrak h^n_1 \oplus \mathfrak h^n_2$. If $p \in \mathbb{H}^n$ we denote by $H_p\mathbb{H}^n \coloneqq \operatorname{span}(X_1(p),\dots,X_n(p),Y_1(p),\dots,Y_n(p))$ and by $H \mathbb{H}^n \coloneqq \bigcup_{p \in \mathbb{H}^n}H_p\mathbb{H}^n$ the horizontal fiber bundle. We denote by $\langle \cdot, \cdot \rangle$ the inner product on $\mathfrak h^n$ that makes the basis $X_1,\dots,X_n,Y_1,\dots,Y_n,T$ orthonormal and by $|\cdot|$ the corresponding norm. With a slight abuse of notation, we also denote by $\langle \cdot, \cdot \rangle$ (resp. by $|\cdot|$) the restriction of the inner product (resp. of the norm) on the horizontal subspace $\mathfrak h^n_1$.

We define the \emph{Carnot-Carathéodory} distance $d$ for $p,q \in \mathbb{H}^n$ as
\[
d(p,q)\coloneqq \inf\left\{\|h\|_{L^1([0,1],\mathbb{R}^{2n})} : 
\begin{array}{l}
\text{the curve $\gamma_h:[0,1]\to\mathbb{H}^n$ defined by}\\
\gamma_h(0)=p,\ \dot\gamma_h=\sum_{j=1}^n(h_jX_j+h_{j+n}Y_j)(\gamma_h)\\
\text{has final point }\gamma_h(1)=q
\end{array}
\right\}.
\]
Then we denote the corresponding open balls of center $p$ and radius $r$ as $B(p,r)$ and, for $m \in \mathbb{N}$, by $\mathcal{H}^m$ the corresponding $m$-dimensional Hausdorff measure and  by $\mathcal{S}^m$ the corresponding $m$-dimensional spherical Hausdorff measure.

<!-- defi-3: body_md -->
Let $\Omega \subseteq \mathbb{H}^n$ be an open set. We say that $u \in L^1_{\operatorname{loc}}(\Omega)$ is a \emph{function of locally bounded $\mathbb{H}$-variation}, and we write $u \in \operatorname{BV}_{\mathbb{H},\operatorname{loc}}(\Omega)$, if there exists a $\mathbb{R}^{2n}$-valued Radon measure $D_\mathbb{H} u=(D_{X_1}u,\dots, D_{Y_n}u)$ on $\Omega$ such that, for every open set $A \subset \subset \Omega$, for every $1 \leq i \leq n$ and for every $\varphi \in C^1_c(A)$ one has 
\[
\int_A \varphi d(D_{X_i}u)=-\int_A u X_i\varphi d \mathcal{L}^{2n+1} \quad \text{ and }\quad \int_A \varphi d(D_{Y_i}u)=-\int_A u Y_i\varphi d \mathcal{L}^{2n+1}.
 \]
  Moreover, if $u \in L^1(\Omega)$ and $D_\mathbb{H} u$ has bounded total variation $|D_\mathbb{H} u|$, then we say that $u$ has \emph{bounded $\mathbb{H}$-variation} and we write $u \in \operatorname{BV}_\mathbb{H}(\Omega)$. 

For every $u \in \operatorname{BV}_\mathbb{H}(\Omega)$ we define the norm
\[
\left\| u \right\|_{\operatorname{BV}_\mathbb{H}(\Omega)} \coloneqq \left\| u \right\|_{L^1(\Omega)}+|D_\mathbb{H} u|(\Omega).
\]
The space $\operatorname{BV}_\mathbb{H}(\Omega)$ equipped with the above norm is a Banach space.

<!-- defi-4: body_md -->
Let $\Omega \subseteq \mathbb{H}^n$ be an open set. For every $u \in \operatorname{BV}_\mathbb{H}(\Omega)$ we decompose
\[
D_\mathbb{H} u=D_\mathbb{H} ^au+D_\mathbb{H} ^su
\]
where $D_\mathbb{H}^au$ and $D^s_\mathbb{H} u$ denote, respectively, the \emph{absolutely continuous part} and the \emph{singular part} of $D_\mathbb{H} u$ with respect to the Lebesgue measure $\mathcal{L}^{2n+1}$.

We say that a measurable set  $E \subseteq \mathbb{H}^n$ has \emph{locally finite $\mathbb{H}$-perimeter} (resp. \emph{finite $\mathbb{H}$-perimeter}) in an open set $\Omega \subseteq \mathbb{H}^n$ if its characteristic function $\chi_E$ belongs to $\operatorname{BV}_{\mathbb{H},\operatorname{loc}}(\Omega)$ (resp. $\chi_E \in \operatorname{BV}_\mathbb{H}(\Omega)$). 
    In such a case we define the \emph{$\mathbb{H}$-perimeter measure} $P_\mathbb{H}(E, \cdot)$ of $E$ as $P_\mathbb{H}(E, \,\cdot\,):=|D_\mathbb{H} \chi_E|$.

\label{def_c1x}
Let $\Omega \subseteq \mathbb{H}^n$ be an open set and $f\colon \Omega \to \mathbb{R}$. We say that $f \in C^1_\mathbb{H}(\Omega)$ if $f$ is continuous and its \emph{horizontal gradient}  $\nabla_\mathbb{H} f \coloneqq (X_1f,\dots,Y_nf)$,   in the sense of distributions, is represented by a continuous function.

<!-- defi-7: body_md -->
Let $\Omega \subseteq \mathbb{H}^n$ be an open set,   $u \in L^1_{\operatorname{loc}}(\Omega)$, $z \in \mathbb{R}$ and $p \in \Omega$. We say that $z$ is the \emph{approximate limit} of $u$ at $p$ if
    \[
    \lim_{r \to 0} \fint_{B(p,r)}|u-z|d\mathcal{L}^{2n+1}=0.
    \]
If the approximate limit of $u$ at $p$ exists, it is also unique (see \cite[Definition 2.19]{dv}). We hence denote by $u^\star(p)$ the approximate limit of $u$ at $p$ and by $\mathcal{S}_u$ the subset of points in $\Omega$ where $u$ does not admit an approximate limit.

<!-- defi-8: body_md -->
\label{def_approxdiff}
    Let $\Omega \subseteq \mathbb{H}^n$ be an open set, $u \in L^1_{\operatorname{loc}}(\Omega)$ and $p \in \Omega \setminus \mathcal{S}_u$. We say that $u$ is \emph{approximately $\mathbb{H}$-differentiable} at $p$ if there exist a neighbourhood $U\subset \Omega$ of $p$ and $f \in C^1_\mathbb{H}(U)$ such that $f(p)=0$ and
    \[
    \lim_{r \to 0}\fint_{B(p,r)} \frac{|u-u^\star (p)-f|}{r}d \mathcal{L}^{2n+1} =0.
    \]
    The set of points in $\Omega$ where $u$ is approximately $\mathbb{H}$-differentiable is denoted by $\mathcal{D}_u$.
The vector $\nabla_\mathbb{H} f(p) \in \mathbb{R}^{2n}$ is uniquely determined (see~\cite[Proposition 2.30]{dv}): we call it \emph{approximate $\mathbb{H}$-gradient} of $u$ at $p$ and we denote it by $D^{\operatorname{ap}}_\mathbb{H} u(p)$.

<!-- defi-9: body_md -->
\label{def_ipersup}
We say that $S \subseteq \mathbb{H}^n$ is a \emph{$C^1_\mathbb{H}$-hypersurface} if for every $p \in S$ there exist $r>0$ and $f \in C^1_\mathbb{H}(B(p,r))$ such that the following facts hold:
\begin{enumerate}
\item[(i)]$ S \cap B(p,r)=\lbrace q \in B(p,r):f(q)=0 \rbrace$,
\item[(ii)]$\nabla_\mathbb{H} f\neq 0$ on $B(p,r)$.
\end{enumerate}
We define the \emph{horizontal normal} to $S$ at $p \in S$ as
$\nu_S(p) \coloneqq \frac{\nabla_\mathbb{H} f(p)}{|\nabla_\mathbb{H} f(p)|}$. Notice that $\nu_S(p)$ is well defined up to a sign and that it does not depend on the choice of $f$, see \cite[Corollary 2.14]{dv}.

<!-- defi-10: body_md -->
\label{def_Xrectifiable}
Let $S \subseteq \mathbb{H}^n$. We say that $S$ is \emph{countably $\mathbb{H}$-rectifiable}\footnote{Since the involved dimension will always be $2n+1$, for the sake of shortness we avoid the more correct, but heavier, terminology {\em countably $(2n+1)$-$\mathbb{H}$-rectifiable}.} if there exists a family $\lbrace S_h: h \in \mathbb{N} \rbrace$ of $C^1_\mathbb{H}$-hypersurfaces such that
\[
\mathcal{H}^{2n+1}\left( S \setminus \bigcup_{h \in \mathbb{N}}S_h \right)=0.
\]
  Moreover, if $\mathcal{H}^{2n+1}(S)<+\infty$, we say that $S$ is \emph{$\mathbb{H}$-rectifiable}. We define the \emph{horizontal normal} to a countably $\mathbb{H}$-rectifiable set $S$ at $p \in S$ as 
\[
\nu_S(p) \coloneqq \nu_{S_h}(p) \text{ if }p \in S_h \setminus \bigcup_{k<h}S_k.
\]
Notice that $\nu_S$ is well defined, up to a sign, $\mathcal{H}^{2n+1}$-a.e., see \cite[Proposition 2.18]{dv}.

<!-- defi-12: body_md -->
\label{def_approxjump}
 Let $\Omega \subseteq \mathbb{H}^n$ be an open set, $u \in L^1_{\operatorname{loc}}(\Omega)$ and $p \in \Omega$. We say that $u$ has an \emph{approximate $\mathbb{H}$-jump} at $p$ if there exist $u^+,u^- \in \mathbb{R}$ with $u^+ \neq u^-$ and $\nu \in \mathbb{S}^{2n-1}$ such that
\begin{equation}\label{eq_xjump}
\lim_{r \to 0} \fint_{B^+_\nu(p,r)} |u-u^+|d\mathcal{L}^{2n+1}=\lim_{r \to 0} \fint_{B^-_\nu(p,r)} |u-u^-|d\mathcal{L}^{2n+1}=0.
\end{equation}
The \emph{jump set} $\mathcal{J}_u$ is defined as the set of points where $u$ has an approximate $\mathbb{H}$-jump.

It is worth remarking that, by~\cite[Theorem 1.2 and Remark 2.25]{dv}, the Hausdorff measure $\mathcal{H}^{2n+1}$ on the jump set is $\sigma$-finite.

Let $\Omega \subseteq \mathbb{H}^n$ be an open set and $u \in \operatorname{BV}_\mathbb{H}(\Omega)$. We define the \emph{jump part} of $D_\mathbb{H} u$ as 
\[
D_\mathbb{H}^ju \coloneqq D_\mathbb{H}^s u \mathop{\hbox{\vrule height 7pt width .5pt depth 0pt \vrule height .5pt width 6pt depth 0pt}}\nolimits \mathcal{J}_u
\]
and the \emph{Cantor part} of $D_\mathbb{H} u$ as
\[
D_\mathbb{H} ^cu \coloneqq D_\mathbb{H}^s u \mathop{\hbox{\vrule height 7pt width .5pt depth 0pt \vrule height .5pt width 6pt depth 0pt}}\nolimits (\Omega \setminus \mathcal{J}_u).
\]
 We say that $u$ is a \emph{special function of bounded $\mathbb{H}$-variation}\footnote{Let us observe that the definition is different from the one provided in \cite{sbvx} since, in Heisenberg groups, the jump set $\mathcal{J}_u$ is always countably $\mathbb{H}$-rectifiable, see for instance \cite{dv,sy03} as well as the recent~\cite{dimarco2026whitneyextensiontheoremintrinsic}.}, and we write $u \in \operatorname{SBV}_\mathbb{H}(\Omega)$, if $D_\mathbb{H}^cu=0$.

<!-- defi-14: body_md -->
In the following we always identify $\mathbb{H}^n \equiv \mathbb{R}^{2n+1}$ by means of exponential coordinates and we will use the following notation:
\[
 \mathcal {C}_H \coloneqq \left(-\frac12,\frac12\right)^{2n},
 \qquad
 \mathcal {C} \coloneqq \mathcal {C}_H\times\left(-\frac12,\frac12\right),
\]
i.e., $\mathcal {C} \subset \mathbb{H}^n$ is the unit Euclidean cube centered at $0 \in \mathbb{H}^n$. Clearly, $\mathcal{L}^{2n+1}(\mathcal {C})=1$.   We denote by $U(n)$ the group of unitary rotations of $\mathbb C^n\equiv\mathbb{R}^{2n}$.  For $U\in U(n)$ we define the automorphism
\[
 \Phi_U(z,t) \coloneqq (Uz,t).
\]
The map $\Phi_U$ is an isometry of $\mathbb{H}^n$ for every $U \in U(n)$, see for instance \cite[Section 7]{kn10}. For $p\in\mathbb{H}^n$, $r>0$ and $U\in U(n)$ we define
\[
 T_{p,r,U}(z,t)\coloneqq p\cdot\delta_r(\Phi_U(z,t)),
\]
and
\begin{equation}\label{eq:cubes}
 \mathcal {C}_{r,U}(p) \coloneqq  T_{p,r,U}(\mathcal {C}).
\end{equation}
Then $\mathcal{L}^{2n+1}(\mathcal {C}_{r,U}(p))=r^{2n+2}$.  By compactness of $\overline \mathcal {C}$ and because $\mathcal {C}$ is a neighborhood of the identity, there are dimensional constants $0<2\mathfrak a<\mathfrak b<\infty$ such that
\begin{equation}\label{eq:cube-ball}
 B(p,\mathfrak ar)\subset \mathcal {C}_{r,U}(p)\subset B(p,\tfrac{\mathfrak b}{2}r),
 \quad
 \operatorname{diam}\bigl(\overline{\mathcal {C}_{r,U}(p)}\bigr)\le \mathfrak br
 \quad\text{for all }p \in \mathbb{H}^n,r>0,U \in U(n).
\end{equation}
We denote by $\mathcal T_r$ the collection of all elements in \eqref{eq:cubes}, namely
\[
\mathcal T_r\coloneqq\{\mathcal {C}_{r,U}(p): p\in \mathbb{H}^n, U\in U(n)\}.
\]
We call each element of $\mathcal T_r$ a \emph{Heisenberg $r$-cube}.

<!-- rem-1: body_md -->
\label{rem_mathcalP}
The family of cubes $\{k\cdot \mathcal {C}:k\in\mathbb{Z}^{2n+1}\}$ is a paving of $\mathbb{H}^n$: namely, a collection of pairwise disjoint sets whose union covers the whole $\mathbb{H}^n$ up to a $\mathcal{L}^{2n+1}$-null set. Indeed, away from the coordinate boundaries, for $p=(z,t)$ there is a unique $k_H\in\mathbb{Z}^{2n}$ such that $z-k_H\in \mathcal {C}_H$ and then a unique $k_t\in\mathbb{Z}$ such that
\[
 t-2\omega(k_H,z-k_H)-k_t\in\left(-\frac12,\frac12\right).
\]
Thus $p=(k_H,k_t)\cdot(z-k_H,t-2\omega(k_H,z-k_H)-k_t)$; existence may fail only on a countable union of Euclidean hyperplanes, which is Lebesgue-null.  Accordingly, for $r>0$ and $U\in U(n)$ we set
\begin{equation}\label{eq:pavage}
 \mathcal P_{r,U} \coloneqq \left\{\delta_r\Phi_U(k\cdot \mathcal {C}):k\in\mathbb{Z}^{2n+1}\right\}.
\end{equation}
Every element of $\mathcal P_{r,U}$ belongs to $\mathcal T_r$.

<!-- defi-15: body_md -->
Let $\Omega \subseteq \mathbb{H}^n$ be an open set, $u \in L^1(\Omega)$ and $r>0$. We define
    \[
    k_r(u) \coloneqq r^{2n+1} \sup_{\mathcal G_r} \sum_{\mathcal{Q} \in \mathcal G_r} \fint_{\mathcal{Q}} \left| u- (u)_{\mathcal{Q}}\right|d\mathcal{L}^{2n+1},
    \]
    where $\mathcal G_r$ varies among all the possible subsets of $\mathcal T_r$ such that the Heisenberg $r$-cubes in $\mathcal G_r$ are pairwise disjoint and contained in $\Omega$.
<!-- end defi-15 -->
<!-- lem-1: body_md -->
\label{lem:pavage}
Let $A\subseteq \mathbb{H}^n$ be a bounded and open set, $U \in U(n)$, and let
\[
 A_{r,U} \coloneqq \bigcup\{Q\in\mathcal P_{r,U}:Q\subset A\}.
\]
Then
\[
 \mathcal{L}^{2n+1}(A\setminus A_{r,U})\xrightarrow{r\to 0}0
\]
uniformly with respect to $U\in U(n)$.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
Let $N_{r,U}$ be the union of the boundaries of the tiles in $\mathcal P_{r,U}$.  Then $\mathcal{L}^{2n+1}(N_{r,U})=0$.  If $p\in A\setminus(A_{r,U}\cup N_{r,U})$ and $\mathcal{Q}\in\mathcal P_{r,U}$ is the unique tile containing $p$, then $\mathcal{Q}\not\subset A$.  Hence
\[
 \operatorname{dist}(p,\mathbb{H}^n\setminus A)\le\operatorname{diam}(\mathcal{Q})\le\mathfrak b r,
\]
where $\mathfrak b$ is the same dimensional constant appearing in \eqref{eq:cube-ball}. It follows, up to the null set $N_{r,U}$, that
\[
 A\setminus A_{r,U}
 \subset\{p\in A:\operatorname{dist}(p,\mathbb{H}^n\setminus A)\le\mathfrak b r\}.
\]
The sets on the right hand side of the above inclusion decrease to the empty set as $r\to 0$ and have finite measure.  The conclusion follows from continuity from above, uniformly in $U$.

<!-- lem-2: body_md -->
\label{lem:NormalOnCube}
For every $\nu\in\mathbb S^{2n-1}$,
\begin{equation}\label{eq:phi-cube}
\int_{\mathcal {C}}|\langle (\nu,0), p\rangle| dp=\int_{\mathcal {C}_H}|\langle\nu,z\rangle|dz\le\frac14.
\end{equation}
Equality holds if and only if $\nu=\pm e_j$ for some $j\in\{1,\ldots,2n\}$.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
This is~\cite[Lemma~3.1]{fms16}, applied in dimension $2n$ to $\mathcal {C}_H$.

<!-- lem-3: body_md -->
\label{lem_14xi}
    For every horizontal vector $\xi \in \mathbb{R}^{2n}$, $U \in U(n), p \in \mathbb{H}^n$ and $r>0$ we have

\label{lem:jumplocalcomparison}
Let $\Omega\subset \mathbb{H}^n$ be open, let $u\in\operatorname{SBV}_\mathbb{H}(\Omega)$ and let $H\subset\mathcal J_u$ be compact.  For every $\eta>0$ there exists a compact set $K\subset H$ such that
\[
 |D_\mathbb{H}^ju|(H\setminus K)<\eta
\]
and, for every $R>0$,
\begin{equation} \label{eq:limiteborel}
 \lim_{r\to 0} \sup\left\{ \frac{|D_\mathbb{H}^au|(p\cdot\delta_rA)}{r^{2n+1}}:p\in K,A\subset B(0,R)\text{ Borel} \right\}=0.
\end{equation}
Consequently, for every $\varepsilon>0$ and $R>0$ there exists $r_{\varepsilon,R}>0$ such that
\begin{equation}\label{eq:stimasalto}
 |D_\mathbb{H} u|(p\cdot\delta_rA) \le |D_\mathbb{H}^ju|(p\cdot\delta_rA)+\varepsilon r^{2n+1}
\end{equation}
whenever $p\in K$, $0<r<r_{\varepsilon,R}$ and $A\subset B(0,R)$ is Borel.

<!-- proof-6: body_md -->
Since $\mathcal{L} ^{2n+1}(\mathcal{J}_u)=0$, also $|D_\mathbb{H}^a u|(H)=0$. We first show that 
    \begin{equation}\label{eq:limitepuntuale}
      \lim_{r\to 0}\frac{|D_\mathbb{H}^a u|(B(p,r))}{r^{2n+1}}=0, \quad \text{for $\mathcal{H}^{2n+1}$-a.e.\ $p\in H$.}
    \end{equation}
    Define for every $a>0$ the set
    \[
    E_a \coloneqq \left\{p\in H: \limsup_{r\to 0}\frac{|D_\mathbb{H}^a u|(B(p,r))}{r^{2n+1}}>a\right\}.
    \]
    Notice that $E_a\subset E_b$ if $a>b$. If $E_a$ is empty for every $a>0$, then \eqref{eq:limitepuntuale} is proved for every $p\in H$. Otherwise we can assume that for every sufficiently small $a>0$, $E_a$ is non-empty. Let $\delta>0$ be sufficiently small so that $H_\delta\coloneqq\{x\in \mathbb{H}^n: d(x,H)<\delta\}\subset \Omega$. Choose for each $p\in E_a$ some $r_p\in (0,\delta)$ with the property that

For every $k\in \mathbb N$ define $h_k\colon \Omega \to \mathbb{R}$ by letting

    \lim_{k \to +\infty} h_k(p)=0, \quad \text{ for $|D^j_\mathbb{H} u|$-a.e.\ $p\in H$.}

    \lim_{k \to +\infty}\ \sup_{p\in K} h_k(p)=0.

We now prove $(vi)$. Up to passing to a subsequence, we can assume that the $\liminf$ in the statement is indeed a limit; if infinite, there is nothing to prove. Otherwise, since $\mathcal {C}_k$ is converging to $\mathcal {C}_\infty$, then for every open set $\mathcal {C}_\infty'\subset \subset \mathcal {C}_\infty$  one has $\mathcal {C}_\infty'\subset \mathcal {C}_k\subset B(0,\mathfrak b)$ (recall~\eqref{eq:cube-ball}) for every $k\in \mathbb{N}$ sufficiently large.

Clearly we have
\begin{equation}\label{eq_esamiorali}
\|u_k-u_q^0\|_{L^1(\mathcal {C}_\infty')}\leq \|u_k-u_{q_k}^0\|_{L^1(\mathcal {C}_\infty')}+\|u_{q_k}^0-u_q^0\|_{L^1(\mathcal {C}_\infty')}.
\end{equation}
We know by $(i)$ that $u^\pm(q_k)$ is converging to $u^\pm(q)$ as $k\to \infty$. Moreover, since $\nu_u(q_k)$ converges to $\nu_u(q)$, also the halfspaces $H^\pm_{\nu_u(q_k)}$ converge in $L^1_{\rm loc}(\mathbb{H}^n)$ to $H^\pm_{\nu_u(q)}$. \[

\]
whence, with direct calculation
\begin{equation}\label{eq:semicontinuita}

\[
    |u^+(q)-u^-(q)|P_\mathbb{H}(H_{\nu_u(q)}^+; \mathcal {C}_\infty')\leq \lim_{k \to +\infty} r_k^{-2n-1}|D_\mathbb{H}^j u|(q_k\cdot \delta_{r_k}(\mathcal {C}_k))+\varepsilon.
\]
The arbitrariness of $\mathcal {C}_\infty'$ and of $\varepsilon>0$ allows us to conclude the proof.

\end{equation}
Choose $q_k\in \mathcal {C}_k\cap K$. 
Up to a subsequence every $q_k$ is contained in $K_{\bar\imath}$ for some fixed $\bar\imath$ and $q_k\to q\in K_{\bar\imath}$. 
For any large enough $k \in \mathbb{N}$ we let $u_k\colon B(0,\mathfrak b) \to \mathbb{R}$ by defining 
\[
u_k(x) \coloneqq u(q_k\cdot\delta_{r_k}x),
\qquad
\widetilde \mathcal {C}_k \coloneqq \delta_{1/r_k}(q_k^{-1} \cdot \mathcal {C}_k).
\]
Since $q_k\in \mathcal {C}_k$, the sets $\widetilde \mathcal {C}_k$ belong to a compact family of Heisenberg 1-cubes: their centers remain bounded and the rotations $U_k$ stay in a compact group. Hence, up to a subsequence, $\widetilde \mathcal {C}_k\to \mathcal {C}_\infty$ for some Heisenberg $1$-cube $\mathcal {C}_\infty$. Using $(v)$ and arguing as in~\eqref{eq_esamiorali} one finds
\[
u_k\xrightarrow{k \to +\infty} u_q^0\coloneqq u^+(q)\chi_{H_{\nu_u(q)}^+}+u^-(q)\chi_{H_{\nu_u(q)}^-}\qquad\text{in }L^1_{\operatorname{loc}}(B(0,\mathfrak b))
\]
whence
\begin{equation}\label{eq_osc}
\lim_{k \to +\infty}\fint_{\widetilde \mathcal {C}_k}|u_k-(u_k)_{\widetilde \mathcal {C}_k}|d\mathcal{L}^{2n+1}=\fint_{\mathcal {C}_\infty}|u_q^0-(u_q^0)_{\mathcal {C}_\infty}|d\mathcal{L}^{2n+1}.
\end{equation}
 Moreover, by $(vi)$,
\begin{equation}\label{eq_liminf}
\liminf_{k\to\infty} r_k^{-(2n+1)}|D_\mathbb{H}^j u|(\mathcal{J}_u\cap \mathcal {C}_k)\geq |u^+(q)-u^-(q)|P_\mathbb{H}(H_{\nu_u(q)}^+;\mathcal {C}_\infty).
\end{equation}
Dividing \eqref{eq_contradiction} by $r_k^{2n+1}$ and using \eqref{eq_osc} and \eqref{eq_liminf}, we obtain
\[
\fint_{\mathcal {C}_\infty}|u_q^0-(u_q^0)_{\mathcal {C}_\infty}|d\mathcal{L}^{2n+1}\geq\frac12|u^+(q)-u^-(q)|P_\mathbb{H}(H_{\nu_u(q)}^+;\mathcal {C}_\infty)+\varepsilon_0.
\]
The above inequality contradicts Lemma~\ref{lem_salti} (when applied to $\mathcal {C}_\infty$), therefore $(vii)$ is proved. 

We now prove $(viii)$. Fix $i \in \{ 1,\dots,m \}$. We work in the coordinates obtained by applying the inverse $\Phi_{U_i}^{-1}$. More precisely, after replacing $\Omega, u,K_i$ with $\Phi_{U_i}^{-1}(\Omega), u\circ\Phi_{U_i},\Phi_{U_i}^{-1}(K_i)$, respectively, and suppressing   for the sake of brevity the index $i$ from $K_i,\Gamma_i, A_i, F_i,\varphi_i$ (recall statement $(iii)$), we
may assume that $U_i=\operatorname{Id}$ and $K\subset\Gamma$. Set
$ A_K \coloneqq F^{-1}(K)\subset\mathbb{W}$. 
The map $F\colon A\to\Gamma$ is a homeomorphism onto the intrinsic graph, hence $A_K$ is compact. Define
\[
 J(x)\coloneqq|u^+(x)-u^-(x)|,
 \qquad \forall x\in K.
\]
Using the area formula for $C^1_{\mathbb H}$ intrinsic graphs (see \cite[Theorem 6.5]{fssc01}) and  \cite[Theorem 1.7]{dv} one gets
\[
 |D_{\mathbb H}^j u|(K) = \int_{A_K} J(F(y)) \frac{1}{|\langle\nu_u(F(y)),e_1\rangle|}d\mathcal{L}_\mathbb{W}(y),
\]
where $\mathcal{L}_\mathbb{W}$ denotes the Lebesgue measure on $\mathbb{W}\equiv\mathbb{R}^{2n}$. Since $|\nu_u-e_1|<\sigma<1$ on $K$ and the orientation has been chosen so that
$\langle\nu_u,e_1\rangle>0$ on $K$, we have
\[
 1\leq \frac{1}{|\langle\nu_u(F(y)),e_1\rangle|} \leq \frac1{1-\tfrac{\sigma^2}2} \qquad\text{for all }y\in A_K
\]
because $|\nu_u|=|e_1|=1$, hence $\langle\nu_u,e_1\rangle=1-\frac12|\nu_u-e_1|^2 > 1-\frac{\sigma^2}{2}$.
Therefore 
\begin{equation}\label{eq_aaaa} 
\int_{A_K}J(F(y))d\mathcal{L}_\mathbb{W}(y) \geq \left(1-\frac{\sigma^2}2\right)|D_{\mathbb H}^j u|(K).
\end{equation}
We write points of $\mathbb{W}$ as $ y=(\widehat z,\tau)\in\mathbb{R}^{2n-1}\times\mathbb{R}\equiv\mathbb{W}=\{x_1=0\}$.
Define for any $r>0$ the set $P_r\subseteq \mathbb W$ by setting
$
 P_{ r}\coloneqq[0, r)^{2n-1}\times[0,r^2)
$
and observe  that
$
 \mathcal{L}_\mathbb{W}(P_{r})=r^{2n+1}.
$
Choose $\lambda \in \mathbb{R}$ such that 
\[
\sqrt{\frac{2+\frac{\sigma}{2n+1}}{2-\frac{\sigma}{2n+1}}}<\lambda <\left(\frac{1-\tfrac{\sigma^2}{2}}{1-\sigma}\right)^\frac{1}{2n+1}.
\]
Such a  choice  is always possible because for $0<\sigma<1$ one can check\footnote{The functions $g(\sigma)=\sqrt{\frac{2+\frac{\sigma}{2n+1}}{2-\frac{\sigma}{2n+1}}}$ and $h(\sigma)=\left(\frac{1-\sigma^2/2}{1-\sigma}\right)^\frac{1}{2n+1}$ are such that $g(0)=h(0)$ and $g'(\sigma)<\tfrac1{2n+1}\leq h'(\sigma)$ for any $\sigma\in(0,1)$.} that
\[
\sqrt{\frac{2+\frac{\sigma}{2n+1}}{2-\frac{\sigma}{2n+1}}}<\left(\frac{1-\tfrac{\sigma^2}{2}}{1-\sigma}\right)^\frac{1}{2n+1}.
\]
For every $r>0$ and $\theta\in P_{\lambda r}$, we define the lattice
\[
 \Lambda_{r,\theta} \coloneqq \theta + \{(\lambda r \widehat k,\lambda^2r^2k_t): \widehat k\in\mathbb{Z}^{2n-1}, k_t\in\mathbb{Z}\}\subseteq\mathbb{W}.
\]
By Fubini's theorem,
\[
 \frac{1}{\mathcal{L}_\mathbb{W}(P_{\lambda r})} \int_{P_{\lambda r}} \sum_{y\in\Lambda_{r,\theta}\cap A} J(F(y))\chi_{A_K}(y) \mathcal{L}_\mathbb{W}(P_{\lambda r})d\theta =\int_{A_K}J(F(y))d\mathcal{L}_\mathbb{W}(y).
\]
Hence there exists $\theta_r\in P_{\lambda r}$ such that
\[
 \sum_{y\in\Lambda_{r,\theta_r}\cap A_K} J(F(y))\mathcal{L}_\mathbb{W}(P_{\lambda r}) \geq \int_{A_K}J(F(y))d\mathcal{L}_\mathbb{W}(y).
\]
Since $A_K$ is compact and the lattice $\Lambda_{r,\theta_r}$ is discrete, $\Lambda_{r,\theta_r}\cap A_K$ is finite. We write 
\[
 \Lambda_{r,\theta_r}\cap A_K =\{y_1^r,\ldots,y_{N(r)}^r\}, \qquad p_h^r \coloneqq F(y_h^r).
\]
Then for $r>0$
\[
 \sum_{h=1}^{N(r)}J(p_h^r)r^{2n+1} = \lambda^{-(2n+1)} \sum_{h=1}^{N(r)} J(F(y_h^r))\mathcal{L}_\mathbb{W}(P_{\lambda r})  \geq \lambda^{-(2n+1)} \int_{A_K}J(F(y))d\mathcal{L}_\mathbb{W}(y).
\]
By the definition of $\lambda$ we have
\[
\lambda^{-(2n+1)} \geq \frac{1-\sigma}{1-\tfrac{\sigma^2}2}
\]
and, combining the previous considerations with \eqref{eq_aaaa}, we get
\[
 \sum_{h=1}^{N(r)} |u^+(p_h^r)-u^-(p_h^r)| r^{2n+1} \geq (1-\sigma)|D_{\mathbb{H}}^j u|(K),
\]
which is statement $(b)$.

The cardinality estimate $(a)$ follows  from the lattice structure. For every
$
y\in\Lambda_{r,\theta_r}\cap A_K
$
define the half-open lattice cell
$
Q_{r,y} \coloneqq y+P_{\lambda r}.
$
The sets $Q_{r,y}$ are pairwise disjoint, since the family
$
\{y+P_{\lambda r}:y\in\Lambda_{r,\theta_r}\}
$
is a paving of $\mathbb{W}$. Moreover,
\[
\mathcal L_\mathbb{W}(Q_{r,y})=\mathcal L_\mathbb{W}(P_{\lambda r})=\lambda^{2n+1}r^{2n+1}.
\]
Since $A_K$ is compact, the set
$
B_K \coloneqq A_K+\overline{P_\lambda}
$
is compact and has finite $\mathcal L_\mathbb{W}$-measure. For $0<r<1$ one has $P_{\lambda r}\subseteq P_\lambda$,  hence
\[
Q_{r,y}\subseteq B_K \qquad \text{for every } y\in\Lambda_{r,\theta_r}\cap A_K.
\]
It follows that
\[
N(r)\lambda^{2n+1}r^{2n+1}=\sum_{y\in\Lambda_{r,\theta_r}\cap A_K}\mathcal L_\mathbb{W}(Q_{r,y})\leq\mathcal L_\mathbb{W}(B_K).
\]
Consequently,
\[
N(r)r^{2n+1}\leq\lambda^{-(2n+1)}\mathcal L_\mathbb{W}(B_K)\eqqcolon C_K
\]
for every $0<r<1$. After possibly decreasing $r_0$, this proves statement $(a)$.

We aim to prove that the Heisenberg cubes $\mathcal {C}_h^r\coloneqq \mathcal {C}_{r,\operatorname{Id}}(p_h^r)$ are pairwise disjoint for every $r\in (0,r_0)$.\footnote{It is enough to prove the claim in the coordinates determined by $\Phi^{-1}_{U_i}$. Indeed, $\Phi_{U_i}$ is a bijective group automorphism and commutes with the homogeneous dilations. Therefore,
\[
\Phi_{U_i}\bigl(\mathcal {C}_{r,\operatorname{Id}}(p_h^r)\bigr)=
\Phi_{U_i}\bigl(p_h^r\cdot\delta_r(\mathcal {C})\bigr)=
\Phi_{U_i}(p_h^r)\cdot
\delta_r\bigl(\Phi_{U_i}(\mathcal {C})\bigr)=
\mathcal {C}_{r,U_i}\bigl(\Phi_{U_i}(p_h^r)\bigr).
\]
Consequently, the Heisenberg $r$-cubes $\mathcal {C}_{r,\operatorname{Id}}(p_h^r)$ are pairwise disjoint if and only if the corresponding Heisenberg $r$-cubes $\mathcal {C}_{r,U_i}\bigl(\Phi_{U_i}(p_h^r)\bigr)$ are pairwise disjoint. Since $\Phi_{U_i}(p_h^r)\in K_i$, the latter are precisely the Heisenberg cubes required in the statement for the index $i$.}
Assume by contradiction that there exists $i\neq j$ such that $\mathcal {C}_i^r\cap \mathcal {C}_j^r\neq\emptyset$. Then there exist
$
 (\xi_1,\widehat\xi,s), (\xi_1',\widehat\xi',s') \in \mathcal {C}\subseteq \mathbb{H}^n \equiv \mathbb{R} \times \mathbb{R}^{2n-1} \times \mathbb{R}
$
such that
\begin{equation}\label{eq:intersectionequality}
p_i^r\cdot\delta_r(\xi_1,\widehat\xi,s) = p_j^r\cdot\delta_r(\xi_1',\widehat\xi',s').
\end{equation}
Write 
$
y_i^r=(\widehat z_i,\tau_i) \in \mathbb{W} \equiv \mathbb{R}^{2n-1}\times \mathbb{R}, \ y_j^r=(\widehat z_j,\tau_j)\in \mathbb{W} \equiv \mathbb{R}^{2n-1}\times \mathbb{R}
$
and recall that
\[
p_i^r=F(y_i^r)=y_i^r\cdot\varphi(y_i^r)e_1, \qquad p_j^r=F(y_j^r)=y_j^r\cdot\varphi(y_j^r)e_1.
\]
Then, writing each component of equality \eqref{eq:intersectionequality} and denoting by $(z_\ell)_{n+1}$ the $n+1$ component of $(0,\widehat z_\ell)$, one gets
\begin{equation}\label{eq_sistema}
\begin{cases}
    \varphi(y_i^r)+r \xi_1=\varphi(y_j^r)+r \xi_1',\\
    \widehat{z_i}+r\widehat{\xi}=\widehat{z_j}+r\widehat{\xi}',\\
    \tau_i+2(z_i)_{n+1}\varphi(y_i^r)+r^2s+2\omega\bigg( \big(\varphi(y_i^r),\widehat z_i\big),\big( r\xi_1,r\widehat\xi\big)\bigg)=\\
    \qquad=\tau_j+2(z_j)_{n+1}\varphi(y_j^r)+r^2s'+2\omega\bigg( \big(\varphi(y_j^r),\widehat z_j\big),\big( r\xi_1',r\widehat\xi'\big)\bigg).
\end{cases}
\end{equation}
Since $\widehat\xi,\widehat\xi'\in (-1/2,1/2)^{2n-1}$ belong to the unit cube, we get from the second equation of \eqref{eq_sistema}
\[
  |\widehat z_i-\widehat z_j|_{\infty}\leq r.
\]
On the other hand, both $y_i^r$ and $y_j^r$ belong to the lattice $\Lambda_{r,\theta_r}$, whose spacing in the $\widehat z$ variables is $\lambda r$, with $\lambda>1$. Therefore, if $\widehat z_i\neq\widehat z_j$, then
\[
 |\widehat z_i-\widehat z_j|_{\infty}\geq\lambda r>r,
\]
a contradiction. Hence $\widehat z_i=\widehat z_j$ and, again from the second equation of \eqref{eq_sistema}, we infer that $\widehat \xi=\widehat \xi'$. From the last equation of \eqref{eq_sistema} we obtain, since $\widehat z_i=\widehat z_j$ and $\widehat \xi =\widehat \xi'$, that
\[
\tau_i-\tau_j=r^2(s'-s)-2r(\varphi(y_i^r)-\varphi(y_j^r))\omega(e_1,(0,\widehat\xi))
\]
Since $s,s' \in (-1/2,1/2), \widehat \xi=\widehat \xi' \in (-1/2,1/2)^{2n-1}$ we have that $|\omega(e_1,(0,\widehat\xi))|\leq 1/2$ and $|s-s'| \leq 1$ and, consequently,
\begin{equation}\label{eq_01}
|\tau_i-\tau_j| \leq r^2+r|\varphi(y_i^r)-\varphi(y_j^r)|.
\end{equation}
Combining the latter with the first equation of \eqref{eq_sistema} and the fact that $\widehat z_i=\widehat z_j$ we infer
\[
|y_i^r-y_j^r|_{\mathbb{R}^{2n}}=|\tau_i-\tau_j| \leq r^2+r|\xi_1-\xi_1'|\leq r^2+r,
\]
and by the  {\em little} $\tfrac 12$-H\"older continuity of $\varphi$ (see~\cite[Theorem 1.3]{ascv06}) one can find $r_0>0$ such that for every $0<r<r_0$
\begin{equation}\label{eq_02bis}
|\varphi(y_i^r)-\varphi(y_j^r)| \leq \beta |y_i^r-y_j^r|_{\mathbb{R}^{2n}}^\frac{1}{2}=\beta|\tau_i-\tau_j|^\frac{1}{2},
\end{equation}
where $\beta =\beta(\sigma) \coloneqq \frac{\sigma}{2n+1}$. From \eqref{eq_01} we also obtain 
\[
|\tau_i -\tau_j| \leq r^2+ \frac{\beta r^2}{2}+\frac{|\varphi(y_i^r)-\varphi(y_j^r)|^2}{2\beta}
\]
which, combined  with \eqref{eq_02bis}, gives
\[
|\tau_i -\tau_j| \leq \left(1+\frac{\beta}{2}   \right)r^2+\beta\frac{|\tau_i-\tau_j|}{2},
\]
that is
\begin{equation}\label{eq_contr}
|\tau_i -\tau_j| \leq \left( \frac{2+\beta}{2-\beta}   \right)r^2.
\end{equation}
Since $y_i^r\neq y_j^r$ and $\widehat z_i=\widehat z_j$, we have $\tau_i\neq \tau_j$. Thus, the vertical spacing of the lattice gives
\[
\lambda^2 r^2 \leq |\tau_i-\tau_j| \leq \left( \frac{2+\beta}{2-\beta}   \right)r^2,
\]
which, recalling that $\beta=\frac{\sigma}{2n+1}$, contradicts the definition of $\lambda$. Therefore the Heisenberg cubes $\mathcal {C}_h^r$ are pairwise disjoint. Finally, since $K\subset \subset\Omega$, up to reducing $r_0$ we may assume that all the Heisenberg cubes $\mathcal {C}_h^r$ are compactly contained in $\Omega$. Repeating the above construction for every $i=1,\ldots,m$ proves the pairwise disjointness within each family $\mathcal F_r^i$. 

It remains only to ensure the disjointness between Heisenberg cubes associated with different compact sets. Since $K_1,\dots,K_m$ are finitely many pairwise disjoint compact sets,
after discarding the empty ones we have
$
d_0 \coloneqq \min_{i\neq j}\operatorname{dist}(K_i,K_j)>0.
$
By \eqref{eq:cube-ball}, every Heisenberg cube centered at a point of $K_i$ is contained in the ball of radius $\mathfrak b r$ centered at that point. Therefore, after possibly decreasing $r_0$ so that $2 \mathfrak br_0<d_0$, a Heisenberg cube centered in $K_i$ cannot intersect a Heisenberg cube centered in $K_j$ whenever $i\neq j$. Hence the entire family
$
\mathcal F_r \coloneqq \cup_{i=1}^m\mathcal F_r^i
$
is pairwise disjoint for every $0<r<r_0$, and the proof is concluded.

    We have that $U_m$ is a bounded open subset of $\Omega$ such that $U_m \subset \subset \Omega$, the sequence $(U_m)_{m \in\mathbb{N}}$ is increasing and $U_m \xrightarrow{m \to +\infty}\Omega$. We can assume that $m$ is large enough so that $\mathcal{J}_u\cap U_m=\mathcal{J}_u$. 
    Fix   $\delta,\tau,\eta>0$, $0<\rho<\frac{1}{2\sqrt{2n}}$ and $0<\sigma<1$. 
    For every  sufficiently small $r>0$  let $\mathcal A_r$ be the family from Lemma \ref{lem_ar} (when applied with $V=U_m$) and $\mathcal F_r$ be the family from Lemma \ref{lem_fr} (applied with $\Omega=U_m$). We have that, whenever $\mathfrak br<\delta/2$ (where $\mathfrak b$ is as in the statement of Lemma \ref{lem_fr})
    \[
    \bigsqcup_{\mathcal{Q} \in \mathcal A_r}\mathcal{Q} \subset \{ x \in \mathbb{H}^n: d(x,\mathcal{J}_u)>\delta\}, \qquad     \bigsqcup_{\mathcal{Q} \in \mathcal F_r}\mathcal{Q} \subset \{ x \in \mathbb{H}^n: d(x,\mathcal{J}_u)<\delta/2\},
    \]
    hence the family of Heisenberg cubes $\mathcal G_r \coloneqq \mathcal A_r \cup \mathcal F_r$ is admissible for the functional $k_r$. Therefore 
    \[
    k_r(u) \geq r^{2n+1}\sum_{\mathcal{Q} \in \mathcal A_r}\fint_{\mathcal{Q}} |u-(u)_{\mathcal{Q}}|d\mathcal{L}^{2n+1}+r^{2n+1}\sum_{\mathcal{Q} \in \mathcal F_r}\fint_{\mathcal{Q}} |u-(u)_{\mathcal{Q}}|d\mathcal{L}^{2n+1}
    \]
and, by Lemmata \ref{lem_ar} and \ref{lem_fr} and the fact that   $\mathcal{J}_u\cap U_m=\mathcal{J}_u$,
    \begin{align*}
     \liminf_{r \to 0}k_r(u) &\geq     \left( \frac{1}{4}-\frac{\sqrt{2n}}{2}\rho \right) \left(  \int_{(U_m)_\delta}|\nabla_\mathbb{H} u|d\mathcal{L}^{2n+1}-\tau \mathcal{L}^{2n+1}(U_m)   \right)\\
     &\qquad+\frac{1}{2}(1-\sigma)(|D^j_\mathbb{H} u|(\mathcal{J}_u)-\eta).
    \end{align*}
    By letting  $\tau \to 0$, $\rho \to 0$, $\sigma \to 0$ and  $\eta \to 0$ we obtain from the above inequality  
        \begin{equation}\label{eq_est3bis}
     \liminf_{r \to 0}k_r(u) \geq     \frac{1}{4}   \int_{(U_m)_\delta}|\nabla_\mathbb{H} u|d\mathcal{L}^{2n+1}   +\frac{1}{2}|D^j_\mathbb{H} u|(\mathcal{J}_u).
        \end{equation}
For fixed $m$ one has $(U_m)_\delta\uparrow U_m \setminus \mathcal{J}_u$ as $\delta \to 0$. Since $\mathcal{J}_u$ is closed and with zero Lebesgue measure, by monotone convergence we obtain
\[
\sup_{m , \delta} \int_{(U_m)_\delta}|\nabla_\mathbb{H} u|d\mathcal{L}^{2n+1}=\int_\Omega |\nabla_\mathbb{H} u|d\mathcal{L}^{2n+1},
\]
and the statement follows by taking the supremum in \eqref{eq_est3bis}.
