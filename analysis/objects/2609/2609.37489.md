\label{prop: recourse_j}
    Consider a fixed $x$ and $n$. For customer $j \in \mathcal{J}$, let $\pi$ be a permutation of $\mathcal{I} \cup \{0\}$ such that $c_{\pi(0)j} > c_{\pi(1)j} \geq c_{\pi(2)j} \geq \dots \geq c_{\pi(|\mathcal{I}|)j} \geq 0$. 
    Suppose that $\xi \sim N(\mu, \Sigma_n)$. Then,          
    \begin{align*}
    \mathbb{E}_{\xi_j} \left[ R_j(x, \xi_j) \right] &= a_{\pi(0)j}(x) + c_{\pi(0)j} \mu_j + h_j(n_j;\sigma_j) \sum_{k=1}^{|\mathcal{I}|} \left(c_{\pi(k-1)j} - c_{\pi(k)j}\right)\Psi\!\left(\frac{\hat{\xi}_{\pi(k)j}(x) - \mu_j}{h_j(n_j;\sigma_j)}\right),
    \end{align*}
    where 
                     
                            
                        
            $ a_{\pi(0)j}(x) = \sum_{i=1}^{|\mathcal{I}|} D_{\pi(i)}(c_{\pi(i)j} - c_{\pi(0)j}) \, x_{\pi(i)}$, 
                      
                    
                     
                        
                        
            $\hat{\xi}_{\pi(k)j}(x) = \sum_{i=k}^{|\mathcal{I}|} D_{\pi(i)} x_{\pi(i)}$ for $k=0, \dots, |\mathcal{I}|$,

    \begin{equation}
    \label{eq:Psi}
        \Psi(z):= z\Phi(z) + \phi(z),   
    \end{equation}
    with $\Phi(\cdot)$ and $\phi(\cdot)$ being the cdf and pdf of a standard normal random variable, respectively.

{Proof.}

    Then, $K_i(x)=\alpha \Psi\big(r_i(x)\big)$. We claim that $K_i(\cdot)$ is convex for all $i \in \mathcal{I}$, which leads to convexity of $f(\cdot,n)$ as $q_i>0$ for $i \in \mathcal{I}$. We have  
    $\nabla K_i(x)  =\alpha 
\nabla r_i(x) \Psi'\big(r_i(x)\big)= 
\nabla g_i(x) \Phi\big(r_i(x)\big)$. Hence, 
$\nabla^2 K_i(x) = 
\nabla g_{i}(x) 
\nabla r_i(x)^\top \Phi'\big(r_i(x)\big) 
        = 
\nabla g_{i}(x) 
\nabla g_{i}(x)^\top \phi\big(r_i(x)\big) / \alpha 
        \succeq 0$.

<!-- lemma-1: body_md -->
\label{lem:char-g}

Consider the function $\varphi(n)$, as defined in \eqref{eq:subproblem}, and its derivatives. 
Define $d_{0}:=-  h'(0)A(0)>0$, where $A(n):=\sum_{i \in \mathcal{I}}q_i\phi\left(\frac{g_i(x)}{h(n)}\right)>0$. Then, 
\begin{enumerate}[label=(\roman*).]
  \item $\varphi$ is strictly convex on $[0,\infty)$. That is, $\varphi'$ is  strictly increasing on $[0,\infty)$. 
   \item $\varphi$ is strictly increasing on $[0,\infty)$ if and only if $d \ge d_{0}$. 
    \item $\varphi'$ has unique root $n_{min}\in(0,\infty)$ on $[0,\infty)$ if and only if $d<d_{0}$, with $\varphi$ strictly decreasing on $[0, n_{min})$ and strictly increasing on $(n_{min},\infty)$.
  
\end{enumerate}
<!-- end lemma-1 -->
<!-- head-4: body_md -->
{Proof.}

We have $\varphi(n)= dn+ h(n) \sum_{i \in \mathcal{I}} q_i \Psi\big(r_i(n)\big)$, where $r_i(n):=\frac{g_i(x)}{h(n)}$. Define $B(n):=\sum_{i \in \mathcal{I}}q_i r_i(n)^2\phi\big(r_i(n)\big)$. Then, we have 
                                                                  
$\varphi'(n)=d+h'(n)A(n)$ and 
                  
$\varphi''(n)=h''(n)A(n)+\frac{(h'(n))^2}{h(n)}B(n)$. Similar to the proof of Proposition~\ref{prop:convexity_n}, we have 
                
                                                                                                 
                                                                                  
$\varphi''(n)>0$ for $n\ge 0$; hence, $\varphi$ is strictly convex on $[0, \infty)$, proving part (i).

Now, note that $\varphi'(0)=d+h'(0)A(0)=d-d_0$. Moreover, we have $h'(n)\to0$ as $n\to\infty$. By Assumption~\ref{assum:properties}, there exists a constant $0<\bar{a}<+\infty$ such that $h(n)\to\bar a$. Since $A(n)\to\sum_{i \in \mathcal{I}}q_i\phi(g_i(x)/\bar a)<\infty$ as $n\to\infty$, we obtain $\lim_{n\to\infty}\varphi'(n)=d$. 

For part (ii), if $d\ge d_0$, then $\varphi'(0)\ge0$. Since $\varphi'$ is strictly increasing by part (i), $\varphi'(n)>0$ for all $n>0$. Therefore, $\varphi$ is strictly increasing on $[0,\infty)$. Conversely, suppose that $\varphi$ is strictly increasing on $[0,\infty)$. Suppose by contradiction that  $d<d_0$. Then, $\varphi'(0)<0$. By continuity of $\varphi'$, there exists $\epsilon>0$ such that $\varphi'(n)<0$ for all $n\in[0,\epsilon]$, contradicting that $\varphi$ is strictly increasing on $[0,\infty)$. Hence, $\varphi$ is strictly increasing on $[0,\infty)$ if and only if $d\ge d_0$.



For part (iii), if $0<d<d_0$, then $\varphi'(0)<0$, whereas $\lim_{y\to\infty}\varphi'(n)=d>0$ as $\lim_{y\to\infty} h'(n)=0$ and $\lim_{y\to\infty} A(n)=0$. Continuity and strict monotonicity of $\varphi'$ imply that there exists a unique $n_{min}\in(0,\infty)$ such that $\varphi'(n_{min})=0$. 
Conversely, suppose that $\varphi'$ has a unique root $n_{min}\in(0,\infty)$. Since $\varphi'$ is strictly increasing, $0=\varphi'(n_{min})>\varphi'(0)=d-d_0$, so $d<d_0$. Also, strict monotonicity and $\lim_{n\to\infty}\varphi'(n)=d$ give $0=\varphi'(n_{min})<d$, so $d>0$. Therefore, $0<d<d_0$. The sign pattern follows from the strict monotonicity of $\varphi'$: $\varphi'(n)<0$ for $n<n_{min}$ and $\varphi'(n)>0$ for $n>n_{min}$.
\hfill \mbox{\quad$\square$}

\label{lem: affine_cost_j}
    Consider a fixed $x$ and $\xi$. Then, we have 
                      
                          
                        
            $R(x, \xi) = \max_{k=0}^{|\mathcal{I}|} \ \{a_{k}(x) + c_{k} \xi\}$,
                      
                    
    where $a_{|\mathcal{I}|}(x)=0$ and 
                     
                                 
                        
            $a_{k}(x) := \sum_{i=k+1}^{|\mathcal{I}|} D_{i} (c_{i} - c_{k}) x_{i}$, $k = 0, \ldots, |\mathcal{I}|-1$.

<!-- head-10: body_md -->
{Proof.}

Let $\beta$ and $\alpha_i$, $i \in \mathcal{I}$, be the dual variables corresponding to the constraints in \eqref{eq: FL_DI_second_j}. The dual problem can be written as 
                      
                            
                        
            $\max_{\alpha,\beta} \Big\{ \beta \xi + \sum_{i \in \mathcal{I}} \alpha_i D_i x_i: \beta + \alpha_i \leq c_i, \; i \in \mathcal{I}, \; 
            \beta \leq c_0, \; 
            \alpha_i \leq 0, \; i \in \mathcal{I} \Big\}$,
                      
                    
    where extreme points of its feasible region can be identified through two cases:
    \begin{itemize}
        \item Case 1: \(\beta = c_0\). Then, for all \(i \in \mathcal{I}\), we have either \(\alpha_i = 0\) or \(\alpha_i = c_i - c_0\). 
        Note that \(\alpha_i \leq c_i - c_0 < 0\) since \(c_0 > c_i\), making \(\alpha_i = 0\) redundant. 
        Hence, \(\beta = c_0\) and \(\alpha_i = c_i - c_0 \) for $i \in  \mathcal{I}$ form an extreme point with objective value \(a_0(x) + c_0 \xi\).

        \item Case 2: \(\beta < c_0\). Then, for all \(i  \in \mathcal{I}\), we have either \(\alpha_i = 0\) or \(\alpha_i = c_i - \beta\). 
        Moreover, there must exist some \(k\) such that \(\alpha_k = c_k - \beta = 0\) so that at least \(|\mathcal{I}| + 1\) constraints are binding at an extreme point. 
        For this extreme point, we have \(\beta = c_k ,\) and for \(i \neq k,\) \(\alpha_i = \min(0, c_i - c_k ),\) 
        yielding the objective value \(a_k(x) + c_k \xi.\)
    \end{itemize}
    The maximum objective value corresponding to these extreme points yields $R(x,\xi)$.
\hfill \mbox{\quad$\square$}

\label{def: quadratic_growth}
    A proper and lower semicontinuous function $f : \mathbb{R}^n \to \mathbb{R} \cup \{+\infty\}$ has a quadratic growth at a local minimum $\bar{x}$ if there exist constants $\varepsilon > 0$ and $c > 0$ such that for all $y \in \mathbb{N}_{\varepsilon}(\bar{x})$, we have 
                      
    $    f(y) \ge f(\bar x) + \frac{c}{2} \|y-\bar{x}\|^2$.

<!-- definition-2: body_md -->
{(Kurdyka-Łojasiewicz (KL) function 
\citep[Definition~2.4]{attouch2013convergence})}
\label{def:KL}
    A function $f : \mathbb{R}^n \to \mathbb{R} \cup \{+\infty\}$  is said to have the KL property at $x^\ast \in \operatorname{dom} \partial f$ if there exist 
$\eta \in (0,+\infty]$, a neighborhood $U$ of $x^\ast$, 
and a continuous concave function 
$\vartheta : [0,\eta) \to \mathbb{R}_+$ such that:
                  
    (i) $\vartheta(0)=0$,
    (ii) $\vartheta$ is $C^1$ on $(0,\eta)$,
    (iii) for all $s \in (0,\eta)$, $\vartheta'(s) > 0$, and 
    (iv) for all 
    $x \in U \cap \{x : f(x^\ast) < f(x) < f(x^\ast)+\eta \}$, KL holds:
    \begin{equation}
    \label{eq: KL_inequality}
    \vartheta'\big(f(x)-f(x^\ast)\big) \cdot 
    \operatorname{dist} \big(0,\partial f(x)\big)
    \ge 1.
    \end{equation}
                
Proper lower semicontinuous functions which satisfy the KL inequality at each point of $\operatorname{dom} \partial f$ are called KL functions.

Since $\{G(n^k)\}_{k >K_0}$ is a nonincreasing sequence by Lemma \ref{lem:sifficient_decrease}, the limit $\ell:=\lim_{k\to\infty} G(n^k)$ exists.
Moreover, by continuity of $G$ and $n^{k_t}\to \bar n$, we have $G(n^{k_t})\to G(\bar n)$, and thus $\ell=G(\bar n)$. In particular, $G(n^k)\ge G(\bar n)$ for all $k\ge K_0$.
By Lemma~\ref{lem:KL}, $G(x^\ast,n)$ is a KL function and its summation with the indicator function $i_{\mathcal{N}}(n)$ preserves KL; hence, $G$ is a KL function. Let $U$, $\eta$, and $\vartheta: [0,\eta) \mapsto \mathbb{R}$ be the objects appearing in Definition \ref{def:KL}. 
Let $\delta,\rho>0$ be such that $\mathbb{N}_\delta(\bar{n}) \subset U$ with $\rho \in (0,\delta)$, with the condition that $\eta < \frac{\lambda_K}{2}(\delta - \rho)^2$. 
By continuity of $G$ and $\vartheta$, there exists $K \ge K_0$ such that for all $k \ge K$, we have 
\begin{equation}
\label{continuity1}
    G(\bar{n}) \le G(n^k) < G(\bar{n}) + \eta, 
\end{equation}
and 
\begin{equation}
\label{continuity2}
    \|\bar{n} - n^{K}\| + 2 \sqrt{\frac{G(n^K)-G(\bar{n})}{C_K}} + \frac{D_{K-1}}{C_K} \vartheta \big(G(n^K)- G(\bar{n})\big) < \rho,
\end{equation}
where $C_k:=\lambda_k/2$ and $D_k:=\lambda_k$.
The key to our proof is the following for all $j \ge K+1$:
    \begin{equation}
     n^j \in \mathbb{N}_{\rho}(\bar{n}), \label{claim1}   
    \end{equation}
    \begin{equation}
        \label{claim2}
        \sum_{i=K+1}^{j} \|n^{i+1} - n^i\| + \|n^{j+1} - n^j\| \le \|n^{K+1} - n^{K}\| + 
\frac{D_{j-1}}{C_j} \Big[
\vartheta\big(G(n^{K+1}) - G(\bar{n})\big) -
\vartheta\big(G(n^{j+1}) - G(\bar{n})\big)
\Big].
    \end{equation}
Before, we prove \eqref{claim1} and \ref{claim2}, we claim that for a fixed $k \ge K+1$, if  $n^k \in \mathbb{N}_{\rho}(\bar{n})$, then 
\begin{equation}
\label{claim3}
     2\|n^{k+1} - n^k\| \le \|n^k - n^{k-1}\|+ \frac{D_{k-1}}{C_{k}}\vartheta\big(G(n^k) - G(\bar{n})\big) - \vartheta\big(G(n^{k+1}) - G(\bar{n})\big).
\end{equation}
Clearly, if $n^{k+1}=n^k$, this inequality holds trivially. Assume that $n^{k+1}
\neq n^k$, which implies that $G(n^k)>G(n^{k+1})$ by Lemma \ref{lem:sifficient_decrease}; thus, $G(n^k) > G(\bar{n})$. 
By the optimality of $n^{k}$ for the $n$-step, we have as $\lambda_k(n^{k-1} - n^{k}) \in 
\nabla_n Q(x^\ast, n^{k};M) + N_{\mathcal{N}}(n^{k})$. That is, $\lambda_k(n^{k-1} - n^{k}) \in \partial G(n^{k})$.  Therefore, 
                  
$\text{dist}(0, \partial G(n^{k})) \leq D_k \|n^{k} - n^{k-1}\|$. 
                
Moreover, by the KL inequality, we have $\text{dist}(0, \partial G(n^{k}))>0$ as $G(n^k) > G(\bar{n})$. Therefore, $n^k
\neq n^{k-1}$. 
Combining with the KL inequality, we have
\begin{equation}
           
 \vartheta'\big(G(n^k) - G(\bar{n})\big) \geq \frac{1}{\text{dist}(0, \partial G(n^{k}))} \ge \frac{1}{D_k \|n^{k} - n^{k-1}\|}.
\end{equation} 
Moreover, by Lemma \ref{lem:sifficient_decrease}, we have 
                 
                            
$ G(n^k) - G(n^{k+1}) \geq C_k \|n^{k+1} - n^k\|^2$. 
               
The concavity of $\vartheta$, then implies that  
\begin{equation*}
\vartheta\big(G(n^k) - G(\bar{n})\big) - \vartheta\big(G(n^{k+1}) - G(\bar{n})\big) \geq \vartheta'\big(G(n^k) - G(\bar{n})\big) (G(n^k) - G(n^{k+1})\big),    
\end{equation*}
which, combined with the previous two conclusions, yields 
\begin{equation*}
\frac{D_{k-1} }{C_k}\Big[\vartheta\big(G(n^k) - G(\bar{n})\big) - \vartheta\big(G(n^{k+1}) - G(\bar{n})\big)\Big] \geq \frac{\|n^{k+1} - n^k\|^2}{\|n^{k} - n^{k-1}\| }.
\end{equation*}
Now, multiplying the inequality by $\|n^{k} - n^{k-1}\|$, taking the square root on both sides and using the identity $2\sqrt{ab} \le a+b$ yield \eqref{claim3}. 
We also claim that 
\begin{equation}
\label{claim4}
    n^k \in \mathbb{N}_{\rho}(\bar{n}) \; \Rightarrow \; n^{k+1} \in \mathbb{N}_{\delta}(\bar{n}) \ \text{with} \ G(n^{k+1})\ge G(\bar n). 
\end{equation}
To see this, note that by Lemma \ref{lem:sifficient_decrease}, we have 
                  
    $\|n^{k+1} - n^{k} \| \le \sqrt{\frac{G(n^k)-G(n^{k+1})}{D_k}} \le \sqrt{\frac{\eta}{D_k}}< \delta - \rho$. 
                
Hence, by the triangle inequality we have $\|n^{k+1}-\bar{n}\| < \delta$, which expresses that $n^{k+1} \in \mathbb{N}_{\delta}(\bar{n})$. 
Claims \eqref{claim1} and \eqref{claim2} can now be proved by induction on $j$, similar to that of \cite[Lemma~2.6]{attouch2013convergence}, by noting that \eqref{continuity1}, \eqref{continuity2}, \eqref{claim3}, and \eqref{claim4} correspond to Eqs. (3), (4), (9), and (5) in that proof, respectively, while \eqref{claim1} and \eqref{claim2} correspond to Eqs. (6) and (7), respectively, after a proper translation of indices. 
                                                                                                                                                     
                  
                
                                                                                                          
                
                                                           
                   
                                                                                                                                                   
                 
                                                 
                                                                                 
                                                                                                                  
                          
                
                                 
                     
                   
                                            
       
                     
   
                      
   
                     
       
                                            
   
                                            
       
              
                                                                                                                    
                                            
                                                                              
A direct use of \eqref{claim2} yields 
               
    $\sum_{k=K+1}^j \|n^{k+1} - n^k\| \leq 
    \|n^{K+1} - n^{K}\| +  2\vartheta\big(G(n^{K+1}) - G(\bar{n})\big)$, 
             
and, thus, $\sum_{k=K}^\infty \|n^{k+1} - n^k\|$ converges. Hence, the sequence $\{n^k\}$ is Cauchy and convergent to  $\tilde{n}=\bar{n}$.
Moreover, the optimality condition for the $n$-step implies that $\lambda_k(n^{k-1}-n^k)\in \partial G(n^k)$ for all $k\ge K_0+1$. Therefore,
                  
$\operatorname{dist}\big(0,\partial G(n^k)\big) \le \lambda_k\|n^k-n^{k-1}\| \to 0$.
                
In particular, there exists $v^k \in \partial G(n^k)$ such that $\|v^k\|\to 0$. By the closedness of $\operatorname{Graph}\partial G$ and $G(n^k)\to G(\bar n)$, we obtain $0\in \partial G(\bar n)$, i.e., $\bar n$ is a critical point of $Q(x^\ast,\cdot;M)$ over $\mathcal{N}$.
\hfill \mbox{\quad$\square$}

{Proof of Theorem \ref{thm:min_convergence}.}
Suppose that the sequence $\{n^k\}$ converges to some $n^\ast\in \mathcal{N}$. By Assumption~\ref{assum:stability}, there exists $K_0$ such that $x^{k}=x^\ast$ for all $k\ge K_0$, where $x^\ast = T(n^k) \in \mathop{\SFrm arg\,min}_{x \in \mathcal{X}} F(x,n^{k})$.
By Lemma~\ref{lem:convergence}, $n^\ast$ satisfies the stationarity condition
$0 \in \partial G(n^\ast)$, 
where $G(n):=F(x^\ast,n)+i_{\mathcal{N}}(n)$. 
The descent property in Lemma \ref{lem:sifficient_decrease} ensures that $n^\ast$ cannot be a local maximizer.
We show that $(x^\ast,n^\ast)$ is a local minimum of $F$ over $\mathcal{X}\times \mathcal{N}$.
Since $\mathcal{X}$ is finite, there exists $r>0$ such that $\mathcal{X}\cap \mathbb{N}_r(x^\ast)=\{x^\ast\}$.
Therefore, for every $(x,n)\in (\mathcal{X}\times \mathcal{N})\cap \big(\mathbb{N}_r(x^\ast)\times \mathcal{N}\big)$ we have $x=x^\ast$ and hence $F(x,n)=F(x^\ast,n)$.
We claim that that $n^\ast$ is an (isolated) strict local minimizer of $F(x^\ast,\cdot)$ over $\mathcal{N}$.
For $j \in \mathcal{J}$, either $n^\ast_j=0$ or $n^\ast_j \in (0,b)$ (Lemma \ref{lem:min-g}), where in a neighborhood of $n^\ast_j$, $f_j(x,n_j)$ satisfies either a local linear growth or a local quadratic growth, respectively (Corollary \ref{cor:min-g}). Therefore, there exists $\delta>0$ such that $G(n) > G(n^\ast)$ over $\mathcal{N} \cap \mathbb{N}_{\delta}(n^\ast)$. 
                                                                                                                                                                                                      
                                                                 
That is, $n^\ast$ is a strict local minimum of $G$. Then, $n^\ast$ is an strong (isolated) local minimizer of $G$. 
Therefore, $(x^\ast,n^\ast)$ is a local minimum of $F$ over $\mathcal{X}\times \mathcal{N}$ 
                                                                                                                                                            
in the sense of Definition~\ref{def:local_min}. 
\hfill \mbox{\quad$\square$}

\label{lem:uniform_convergence}
For every $x \in \mathcal{X}$, we have 
                  
$\max_{n\in\mathcal{N}}|F_M(x,n)-F(x,n)|\to 0$ as  $M\to\infty$,  
                
almost surely (a.s.).
<!-- end lemma-11 -->
<!-- head-20: body_md -->
{Proof.}
For simplicity, we suppress $x$ and $\sigma$ in our notation. 
It is easy to verify that there exists a continuous (random) $B$ such that $\max_{n\in \mathcal{N}}|\ell(\zeta;n)| \le B(\zeta)$ a.s., with $\mathbb{E} \left[ B \right]<\infty$. Hence, by the strong law of large numbers (SLLN), a.s. 
for every $n \in \mathcal{N}$, we have pointwise convergence: $F_M(n) \rightarrow F(n)$ as $M\to\infty$. 
                 
                                 
                                                      
               
            
                 
                                 
                                                                              
               
                                                                                                   
We now show a random Lipschitz bound for $\ell(\zeta;n)$. For every $\zeta$, $n\in \mathcal{N}$, and $s \in \partial_{n_j} \big(\ell_{ij}(\zeta;n_j)\big)_{+}$ for $j \in \mathcal{J}$, we have 
                  
$| s | \le |\zeta| \big| h'_j(n_j) \big| \sum_{i \in \mathcal{I}} q_{ij}$. Let $\alpha_j:=\max_{n_j \in [0,b]} |h'_j(n_j)| \times |\mathcal{I}|\max_{i \in \mathcal{I}} q_{ij}$, where $\alpha_j <\infty$ given that $h'_j(n_j)$ is continuous, and hence bounded, on $[0,b]$. Define $L(\zeta):=|\zeta| \|\alpha\|$. 
By the mean-value theorem,
                 
                              
$|\ell(\zeta;n)-\ell(\zeta;n^\prime)|\le L(\zeta)\|n-n^\prime\|$ for all $n,n^\prime \in\mathcal{N}$ a.s.
Also, $\mathbb{E} \left[ |\zeta| \right]=\sqrt{\frac{2}{\pi}}$; hence, $\mathbb{E} \left[ L \right]<\infty$ and 
               
           
by SLLN, a.s. 
we have $\frac{1}{M}\sum_{t=1}^M L(\zeta^t) \rightarrow \mathbb{E} \left[ L \right]$ as $M\to\infty$. 
                 
                                          
                                                                                     
               
                                                                                                
Fix $\varepsilon>0$. If $\mathbb{E} \left[ L \right]=0$, then $L(\xi)=0$ a.s. and $\ell(\xi;\cdot)$ is a.s. constant on $\mathcal{N}$, so the claim is immediate from the pointwise convergence.                                     
Hence, consider $\mathbb{E} \left[ L \right]>0$ and choose $\delta =\frac{\varepsilon}{6\mathbb{E} \left[ L \right]}$. By the compactness of $\mathcal{N}$, there exists a collection of points $\{n^1,\dots,n^r\}\subset \mathcal{N}$ such that $\mathcal{N} \subset \bigcup_{k \in [r]} \mathbb{N}_{\delta}(n^k)$.
For each fixed $k\in[r]$,                                   
the pointwise convergence, gives $F_M(n^k)\to F(n^k)$ a.s. Since $r<\infty$, there exists $M_0:=M_0(\varepsilon)$ such that for all $M\ge M_0$ and all $k\in[r]$, 
                  
$|F_M(n^k)-F(n^k)| \le \varepsilon/2$.
                
Moreover,                                                 
there exists $M_1$ such that for all $M \ge M_1$, we have $\frac{1}{M}\sum_{t=1}^M L(\zeta^t) \le 2 \mathbb{E} \left[ L \right]$. 
Now take any $n \in \mathcal{N}$ and choose $k\in[r]$ such that $n\in \mathbb{N}_\delta(n^k)$, i.e., $\|n-n^k\| < \delta$. Then, for all $M \ge \max\{M_0,M_1\}$,
we have 
\begin{align*}
\big|F_M(n)-F(n)\big|
& \le \big|F_M(n)-F_M(n^k)\big| + \big|F_M(n^k)-F(n^k)\big| + \big|F(n^k)-F(n)\big| \\
& \le                                                                                                   
   
\frac{\varepsilon}{2}  + \|n - n^k\| \Big( \mathbb{E} \left[ L \right] + \frac{1}{M}\sum_{t=1}^M L(\xi^t)\Big)\le \frac{\varepsilon}{2} + \delta 3 \mathbb{E} \left[ L \right]= \varepsilon.  
\hfill \mbox{\quad$\square$}

\label{lem:saa_subdiff_consistency}
Consider $x \in \mathcal{X}$. 
Let $\{(n_M, v_M)\}_{M \in \mathbb{N}}$ be a sequence such that $v_M\in \partial_n\big(F_{M}(x,\cdot)+i_{\mathcal{N}}\big)(n_M)$. If $n_M \to \bar n$ and $v_M \to \bar v$ as $M \to \infty$, then $\bar v \in \partial_n\big(F(x,\cdot)+i_{\mathcal{N}}\big)(\bar n)$ a.s.
<!-- end lemma-12 -->
<!-- head-21: body_md -->
{Proof.}
For simplicity, we suppress $x$ and $\sigma$ in our notation. 
For $i \in \mathcal{I}$ and $j \in \mathcal{J}$, it is easy to verify that there exists a continuous (random) $B$ such that $\max_{n_j\in  [0,b]} |\ell_{ij}(\zeta;n_j)| \le B(\zeta)$ a.s., with $\mathbb{E} \left[ B \right]<\infty$.  
For every $\zeta$, define $G_{ij}(\zeta;n_j):=q_{ij} \zeta h'_j(n_j) \mathbf{1}\{\zeta h_j(n_j) + g_{ij}>0\}$, which is bounded above over $n_j \in [0,b]$. Hence, by the uniform SLLN, we have $\max_{n_j \in [0,b]} \big| \frac{1}{M} \sum_{t=1}^M G_{ij}(\zeta^t;n_j) - \mathbb{E} \left[ G_{ij}(\zeta;n_j) \right]\big| \to 0$ as $M \to \infty$.  We also have $\mathbb{E} \left[ G_{ij}(\zeta;n_j) \right]=q_{ij}h'_j(n_j)\phi\Big(\frac{g_{ij}(x)}{h_j(n_j)}\Big)$ by an application of Lemma~\ref{lem:SAA_phi}, which combined with 
$\frac{\partial F_j}{\partial n_j}(n_j)= d_j + h_j'(n_j) \sum_{i \in \mathcal{I}}  q_{ij} \phi \left(\frac{g_{ij}}{h_j(n_j)}\right)$, lead to $\max_{n_j \in [0,b]} \big| d_j + \frac{1}{M} \sum_{t=1}^M \sum_{i \in \mathcal{I} } G_{ij}(\zeta^t;n_j) - \frac{\partial F_j}{\partial n_j}(n_j) \big| \to 0$ as $M \to \infty$.  

Given that $F_M(n)$ is Lipschitz (see the proof of Lemma~\ref{lem:uniform_convergence}), we have $v_M=w_M+\eta_M $, where $w_M \in \partial_n F_{M}(n_M)$ and $\eta_M \in N_{\mathcal{N}}(n_M)$. 
For $j \in \mathcal{J}$, $w_M$ differs from $d_j + \frac{1}{M} \sum_{t=1}^M \sum_{i \in \mathcal{I} } G_{ij}(\zeta^t;n_{M,j})$ only through terms satisfying $\zeta^t h_j(n_{M,j})+g_{ij}=0$. 
Because $\zeta$ has a continuous distribution and $\mathcal{I}$ is finite, the number of such terms is uniformly finite a.s., and their contribution is bounded by a constant multiped by $\frac{1}{M} \max_{t=1}^{M} |\zeta^t|$, which goes to $0$ as $M \to \infty$. Thus, $w_M \to \nabla_n F(\bar n)$ as $M \to \infty$. 
Since $v_M \to \bar v$ as $M \to \infty$, we also have $\eta_M \to \bar v-\nabla_nF(\bar n)$ as $M \to \infty$.
Using the closedness property of the limiting subdifferential of $i_{\mathcal{N}}$  (cf.\ the closedness of $\operatorname{Graph}\,\ \partial_n i_{\mathcal{N}}$, or $\operatorname{Graph}\,\ N_{\mathcal{N}}$), we have $\bar v-\nabla_nF(\bar n)\in N_{\mathcal{N}}(\bar n)$. 
Because $F$ is continuously differentiable, $\partial_n\big(F(\cdot)+i_{\mathcal{N}}\big)(\bar n)
= \nabla_nF(\bar n)+N_{\mathcal{N}}(\bar n)$. Therefore, $\bar v\in \partial_n\big(F(\cdot)+i_{\mathcal{N}}\big)(\bar n)$. 
\hfill \mbox{\quad$\square$}

{Proof of Theorem \ref{thm:saa_stationary_consistency}.}
By the finiteness of $\mathcal{X}$, Lemma~\ref{lem:uniform_convergence} implies that 
                  
$\max_{x\in\mathcal{X}}\max_{n\in\mathcal{N}}|F_M(x,n)-F(x,n)|\to 0$  as $M\to\infty$   
                
with probability 1. 
Let $(x_M,n_M)\in \mathcal{S}_M$. Since $\mathcal{X}$ is finite and $\mathcal{N}$ is compact, the sequence $\{(x_M,n_M)\}$ admits a subsequence (not relabeled) such that $x_M\equiv \bar x$ and $n_M\to \bar n\in\mathcal{N}$.
By definition of $\mathcal{S}_M$, we have $0\in \partial_n\big(F_M(\bar x,\cdot)+i_{\mathcal{N}}\big)(n_M)$ and $\bar x =T_M(n_M) \in \mathop{\SFrm arg\,min}_{z\in\mathcal{X}} F_M(z,n_M)$.
                                                                                                                                                
Fix any $z\in\mathcal{X}$ and any $\varepsilon>0$. By uniform convergence, there exists $M_0(\varepsilon)$ such that for all $M\ge M_0(\varepsilon)$, 
                      
                  
                                                   
                       
                                        
                
                                                        
                  
$F(\bar x,n_M)\le F_M(\bar x,n_M)+\varepsilon \le F_M(z,n_M)+\varepsilon \le F(z,n_M)+2\varepsilon$.   
                
Letting $M\to\infty$ and then $\varepsilon\downarrow0$, continuity of $F$ yields $F(\bar x,\bar n)\le F(z,\bar n)$ for all $z\in\mathcal{X}$. Therefore, $\bar x \in \mathop{\SFrm arg\,min}_{z\in\mathcal{X}} F(z,\bar n)$.                                                           
Since $0\in \partial_n\big(F_M(\bar x,\cdot)+i_{\mathcal{N}}\big)(n_M)$ and $n_M \to \bar n$, by Lemma~\ref{lem:SAA_phi}, we have $0\in \partial_n\big(F(\bar x,\cdot)+i_{\mathcal{N}}\big)(\bar n)$.
                                                                                                                                                                                                            
                                                                                                                                                                                                                                                                                                                   
Hence $(\bar x,\bar n)\in \mathcal{S}$, proving that every sequence in $\mathcal{S}_M$ has subsequences converging to points in $\mathcal{S}$.
The previous argument implies 
                  
$\sup_{(x,n)\in\mathcal{S}_M}\operatorname{dist}\big((x,n),\mathcal{S}\big)\to 0$ as $M\to\infty$. 
                
Otherwise, 
                                                      
there exist $\varepsilon>0$ and a subsequence $(x_M,n_M)\in\mathcal{S}_M$ such that $\operatorname{dist}\big((x_M,n_M),\mathcal{S}\big)\ge \varepsilon$, contradicting the previous subsequential limit argument.  

To show Hausdorff convergence, we further need to prove  
                                                                                                                                                                                                                                                                                                                                                                             
that 
                  
$\sup_{(x,n)\in\mathcal{S}}\operatorname{dist}\big((x,n),\mathcal{S}_M\big)\to 0 $ as  $M\to\infty$. 
                
Fix any $(x^\ast,n^\ast)\in\mathcal{S}$. By the quadratic growth at $n^\ast$ for $F(x^\ast,\cdot)$ (see Proof of Theorem \ref{thm:min_convergence}) and uniqueness of the minimizer $x^\ast$ of $F(\cdot,n^\ast)$, there exist $\delta>0$, $c>0$, and $\gamma >0$ such that
$F(x^\ast,n) \ge F(x^\ast,n^\ast) + \frac{c}{2}\|n-n^\ast\|^2$ for all $n\in \mathcal{N}\cap \mathbb{N}_\delta(n^\ast)$ and
$F(z,n^\ast)\ge F(x^\ast,n^\ast)+\gamma$ for all $z\in\mathcal{X}\setminus\{x^\ast\}$. 
Choose $\eta\in(0,\delta)$. Since $F_M\to F$ uniformly on $\mathcal{X}\times \mathcal{N}$, there exists $M_0$ such that for all $M\ge M_0$ and all $(x,n)\in\mathcal{X}\times\mathcal{N}$,
$|F_M(x,n)-F(x,n)|\le \frac{c}{8}\eta^2$.    
Then, for any $n\in \mathcal{N}\cap \mathbb{N}_\delta(n^\ast)$ with $\|n-n^\ast\|\ge \eta$,
\begin{align*}
F_M(x^\ast,n)
&\ge F(x^\ast,n)-\frac{c}{8}\eta ^2
\ge F(x^\ast,n^\ast)+\frac{c}{2}\|n-n^\ast\|^2-\frac{c}{8}\eta^2 
\ge F(x^\ast,n^\ast)+\frac{c}{2}\eta^2-\frac{c}{8}\eta^2\\
&= F(x^\ast,n^\ast)+\frac{3c}{8}\eta^2 \ge F_M(x^\ast,n^\ast)+\frac{3c}{8}\eta^2-\frac{c}{8}\eta^2
=F_M(x^\ast,n^\ast)+\frac{c}{4}\eta^2.
\end{align*}
Hence, $F_M(x^\ast,n)\ge F_M(x^\ast,n^\ast)+\frac{c}{4}\eta^2$ when $\|n-n^\ast\|\ge \eta$ and $M\ge M_0$.
Now consider the restriction of $F_M(x^\ast,\cdot)$ to the compact set $\mathcal{N}\cap \mathbb{N}_\delta(n^\ast)$. Since $n^\ast\in \mathcal{N}\cap \mathbb{N}_\delta(n^\ast)$, the minimum
$\tilde n_M\in \mathop{\SFrm arg\,min}_{n\in \mathcal{N}\cap \mathbb{N}_\delta(n^\ast)} F_M(x^\ast,n) $   
exists. By construction, $\|\tilde n_M-n^\ast\|\le \eta$. Moreover, $\tilde n_M$ is a (constrained) local minimizer of $F_M(x^\ast,\cdot)$ on $\mathcal{N}$, and hence, 
$0\in \partial_n\big(F_M(x^\ast,\cdot)+i_{\mathcal{N}}\big)(\tilde n_M)$, where this stationarity follows since $\tilde n_M$ minimizes $F_M(x^\ast,\cdot)$ over the neighborhood $\mathcal{N}\cap \mathbb{N}_\delta(n^\ast)$.                                                                                                                                                                                  
                                                            
                                                                                                                                               
      
                        
                                    
                                                                                                                                                  
                                                      
By the uniform convergence $F_M\to F$ on $\mathcal{X}\times\mathcal{N}$ and the fact that $F(z,n^\ast)\ge F(x^\ast,n^\ast)+\gamma$ for all $z\in\mathcal{X}\setminus\{x^\ast\}$,  we have 
                  
                                                       
                
$x^\ast \in \mathop{\SFrm arg\,min}_{z\in\mathcal{X}}F_{M}(z,\tilde n_{M})$. Thus, $(x^\ast,\tilde n_M)\in \mathcal{S}_M$, and since $\eta>0$ is arbitrary, $\operatorname{dist}((x^\ast,n^\ast),\mathcal{S}_M)\to0$. 
A compactness argument (taking the supremum) over $\mathcal{S}$ then yields
                                
                  
                                                                       
                
                                                                                                                                                                                                                                                                                                                        
$\sup_{(x,n)\in\mathcal{S}}\operatorname{dist}((x,n),\mathcal{S}_M)\to 0$.
Combining both one-sided limits gives $d_H(\mathcal{S}_M,\mathcal{S})\to 0$ as $M\to\infty$.
Finally, under the quadratic-growth assumptions of the theorem, stationary points coincide with local minimizers by an argument analogous to that of Theorem \ref{thm:min_convergence}. 
                                                                                                          
\hfill \mbox{\quad$\square$}

\label{lem:uniform_convergence_PWL}
Suppose the breakpoints used to construct $\Psi_M$ cover an interval
containing
$\mathcal{Z}:=\left\{\frac{g_{ij}(x)}{h_j(n_j;\sigma_j)}:
x\in\mathcal{X},\ n\in\mathcal{N}, \; i\in\mathcal{I}, \; j\in\mathcal{J}\right\}$,
and let $\kappa_M:=\max_{t\in\{0,\dots,M-1\}}(z^{t+1}-z^t)$, with $\kappa_M\to0$ as $M \to \infty$. 
Then, $\max_{n \in \mathcal{N}} \max_{x\in\mathcal{X}}|S_M(x,n)-F(x,n)| \to 0$ as $M\to\infty$.

<!-- head-23: body_md -->
{Proof.}
For simplicity, we suppress $\sigma$ in our notation. 
Since $\mathcal{X}$ is finite, $\mathcal{N}$ is compact, and $h_j(\cdot)$
is positive, continuous, and bounded on $[0,b]$, $\mathcal{Z}$ is bounded.
Moreover, there exists a finite constant $C:=\max_{n \in \mathcal{N}} \sum_{j \in \mathcal{J}}\sum_{i \in \mathcal{I}} q_{ij} h_j(n_j)<\infty $
Since $\Psi'(z)=\Phi(z)\in[0,1]$, $\Psi$ is globally Lipschitz with constant $1$ on $\mathbb{R}$. Hence, on each interval $[z^t,z^{t+1}]$, $0 \le t < M$, the linear interpolation error is bounded by the oscillation of $\Psi$ on that interval, so
                  
$\sup_{z\in \mathcal{Z}}|\Psi_M(z)-\Psi(z)|\le \max_{t \in \{0,1\ldots,M-1\}}  \sup_{u,v\in[z^t,z^{t+1}]}|\Psi(u)-\Psi(v)|
\le \max_{t \in \{0,1\ldots,M-1\}} (z^{t+1}-z^t)=\kappa_M$.     
                
                                                                           
Consequently, for all $x\in\mathcal{X}$ and $n \in \mathcal{N}$,
                  
$|S_M(x,n)-F(x,n)|
\le \sum_{j \in \mathcal{J}}\sum_{i \in \mathcal{I}} h_j(n_j) q_{ij}
\sup_{z\in\mathcal{Z}}|\Psi_M(z)-\Psi(z)| \le C \kappa_M$.    
Taking the maximum over $\mathcal{X}\times\mathcal{N}$ and letting $M\to\infty$ 
prove the claim.
