<!-- theorem-1: body_md -->
\label{main}
Let $a:\mathbb{R}^d\times\mathbb{R}^d\to \mathbb{R}$ be a continuous function, $1$-periodic in both variables and such that there exist $0<\alpha\leq \beta$ for which
$    \alpha\leq a\left(x,y\right)\leq \beta$
for all $x,y\in\mathbb{R}^d$. 
    Let $\Omega \subset \mathbb{R}^d$ be an open, bounded set with Lipschitz boundary. Let $\varepsilon \in (0,1)$ and $s \coloneqq s(\varepsilon) \in (0,1)$. Let 
\[
F_{\varepsilon}(u):=(1-s)\int_\Omega\int_\Omega a\left(\frac{x}\varepsilon,\frac{y}\varepsilon\right)
\frac{|u(x)-u(y)|^2}{|x-y|^{d+2s}}dxdy
\]    
be defined in $H^s(\Omega)$.
    If $\lim\limits_{\varepsilon \to 0} s = 1$ and
    \[
        \lim_{\varepsilon \to 0} \varepsilon^{2(1-s)} = \lambda \in [0,1], 
    \]
    then
    \[  
        \mathop{\Gamma\text{-}\lim}\limits_{\varepsilon \to 0} F_\varepsilon(u) = \frac{\sigma_{d-1}}{2d}\left((1-\lambda)\overline{a}\|\nabla u\|_{L^2(\Omega)}^2+\lambda F_{\rm hom}(u)\right),
    \]
    where $u \in H^1(\Omega)$, the $\Gamma$-limit is calculated with respect to the $L^2(\Omega)$ strong topology and $\sigma_{d-1}$ is the $(d-1)$-Hausdorff measure of the unit sphere $\mathbb{S}^{d-1}$ in $\mathbb{R}^d$.

<!-- theorem-2: body_md -->
\label{BBM_theorem1}
    Let $\Omega$ be a bounded connected Lipschitz open set. Let $f\in H^1(\Omega)$ and $\rho\in L^1(\mathbb{R}^d)$, with $\rho\geq 0.$ Then there exists $C=C(\Omega)>0$ such that
\begin{equation}\label{BBM_formula}
    \iint_{\Omega^2}\dfrac{|f(x)-f(y)|^2}{|x-y|^2} \rho(x-y) dx dy \leq C \|\nabla f\|_{L^2(\Omega)}^2\|\rho\|_{L^1(\mathbb{R}^d)}.
\end{equation}

<!-- lemma-1: body_md -->
\label{l_2_large_interactions}
    There exists $C=C(d)>0$ such that
    \begin{equation}\label{large_interactions_estimate}
        (1-s)\iint_{A^2\cap\{|x-y|\geq r\}} \dfrac{|f(x)-f(y)|^2}{|x-y|^{d+2s}}dxdy\leq \dfrac{C(1-s)}{r^{2s}}\|f\|^2_{L^2(A)}
    \end{equation}
     for every bounded measurable set $A\subset\mathbb{R}^d$, every $r>0$ and every $f\in L^2(A)$.
    In particular,
    if $(u_\varepsilon)\subset L^2(\Omega)$ is a bounded sequence and $r\coloneqq r(\varepsilon)>0$ satisfies
    \begin{equation}
        \lim\limits_{\varepsilon\to 0}\frac{1-s}{r^2}=0.
    \end{equation}
    Then,
    \begin{equation}
        \lim\limits_{\varepsilon\to 0}(1-s)\iint_{\Omega^2\cap\{|x-y|\geq r\}}\dfrac{|u_\varepsilon(x)-u_\varepsilon(y)|^2}{|x-y|^{d+2s}}dxdy=0.
    \end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Using $|f(x)-f(y)|^2\leq 2|f(x)|^2+2|f(y)|^2$, and integrating in polar coordinates, we obtain
        $$(1-s)\iint_{A^2\cap\{|x-y|\geq r\}} \dfrac{|f(x)-f(y)|^2}{|x-y|^{d+2s}}dxdy
        \leq 4(1-s)\|f\|^2_{L^2(A)}\sigma_{d-1}\int_r^{+\infty}t^{-1-2s}dt,
    $$
    which gives \eqref{large_interactions_estimate}. Applying \eqref{large_interactions_estimate} with $A=\Omega$ and $f=u_\varepsilon$, and using $r^{-2s}\leq\max\{r^{-2},1\}$, we obtain
        $$(1-s)\iint_{\Omega^2\cap\{|x-y|\geq r\}} \dfrac{|u_\varepsilon(x)-u_\varepsilon(y)|^2}{|x-y|^{d+2s}}dxdy
        \leq C\max\Big\{\frac{1-s}{r^2},1-s\Big\}\|u_\varepsilon\|^2_{L^2(\Omega)},
  $$ 
    which vanishes as $\varepsilon\to0$, since $(u_\varepsilon)$ is bounded in $L^2(\Omega)$.

<!-- lemma-2: body_md -->
\label{l_infty_large_interactions}
    Let $(v_\varepsilon)\subset L^{\infty}(\Omega)$ and assume that there exists $C>0$ such that $\|v_\varepsilon\|_{\infty}\leq C\varepsilon$. Then, for every $M>0$, we have
    \begin{equation}
        \lim\limits_{\varepsilon\to 0}(1-s)\iint_{\Omega^2\cap\{|x-y|\geq M\varepsilon\}} \dfrac{|v_\varepsilon(x)-v_\varepsilon(y)|^2}{|x-y|^{d+2s}}dxdy=0.
    \end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since $\Omega$ is bounded, there exists $R>0$ such that $x,y\in\Omega$ implies $|x-y|<R$.
Using the hypothesis on $(v_\varepsilon)$, we get 
\begin{align}\label{lem_stima_1}
    \iint_{\Omega^2\cap\{|x-y|\geq M\varepsilon\}} \dfrac{|v_\varepsilon(x)-v_\varepsilon(y)|^2}{|x-y|^{d+2s}}dxdy\leq 4C^2\varepsilon^2\iint_{\Omega^2\cap\{R>|x-y|\geq M\varepsilon\}}\dfrac{1}{|x-y|^{d+2s}}dxdy
\end{align}    
Integrating in polar coordinates, we may estimate \eqref{lem_stima_1} from above with
\begin{eqnarray*}  4C^2\varepsilon^2\int_\Omega\int_{\mathbb{S}^{d-1}}\int_{M\varepsilon}^{R}t^{-1-2s}dt d\sigma(\nu)dx
    =4C^2\varepsilon^2|\Omega|\sigma_{d-1}\dfrac{(M\varepsilon)^{-2s}-R^{-2s}}{2s}
    \leq D\varepsilon^{2(1-s)}-E\varepsilon^2,
\end{eqnarray*}
and $D, E>0$ are some fixed positive constants.
The claim follows by multiplying by $1-s$ and letting $\varepsilon\to 0$.

<!-- lemma-3: body_md -->
\label{w_1_infty_close_interactions}
    Let $(\Omega_\varepsilon)$ be a sequence of measurable subsets of $\Omega$ and let $(f_\varepsilon)$ be a sequence of uniformly Lipschitz functions on $\Omega$. Assume that
        $\lim\limits_{\varepsilon\to 0}|\Omega_\varepsilon|=0$.
    Then, it follows that  
    \begin{equation}
        \lim\limits_{\varepsilon\to 0}\ (1-s)\int_{\Omega_\varepsilon} \int_{\Omega} \dfrac{|f_\varepsilon(x)-f_\varepsilon(y)|^2}{|x-y|^{d+2s}}dxdy =0
    \end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
By hypothesis, there exists $L>0$ such that
$|f_\varepsilon(x)-f_\varepsilon(y)| \leq L|x-y|$
for every $x,y \in \Omega$ and for every $\varepsilon > 0$. 
We then obtain
\begin{align}
 (1-s)\int_{\Omega_\varepsilon} \int_{\Omega} \dfrac{|f_\varepsilon(x)-f_\varepsilon(y)|^2}{|x-y|^{d+2s}}dxdy
    \label{lem_stima_3} \leq 2L^2  (1-s) \int_{\Omega_\varepsilon} \int_{\Omega}\dfrac{1}{|x-y|^{d+2s-2}}dxdy
\end{align}
We can estimate \eqref{lem_stima_3} from above, integrating in polar coordinates and using that $\Omega \subset B_R(0)$ for some $R > 0$, obtaining
\begin{eqnarray*}
    (1-s)\int_{\Omega_\varepsilon} \int_{\Omega}\dfrac{1}{|x-y|^{d+2s-2}}dxdy
    &\leq& (1-s)\int_{\Omega_\varepsilon}\int_{\mathbb{S}^{d-1}}\int_0^{2R}t^{1-2s}dt d\sigma(\nu)dx\\
    &=&\frac {|\Omega_\varepsilon|\,\sigma_{d-1}\,(2R)^{2(1-s)}}2,
\end{eqnarray*}
which tends to $0$, since $(2R)^{2(1-s)}$ is bounded and $|\Omega_\varepsilon|\to 0$.

<!-- lemma-4: body_md -->
\label{l_averaging_large_scale}
    There exists $C=C(d)>0$ such that, for every $s\in[3/4,1)$, $\varepsilon\in(0,1)$, $\varepsilon\leq r_1<r_2\leq 1$, $\xi\in\mathbb{R}^d$ and every bounded measurable set $A\subset\mathbb{R}^d$, it holds
    \begin{align}\label{averaging_estimate}
        \nonumber&\bigg\vert(1-s)\int_{A}\int_{\{r_1\leq|x-y|\leq r_2\}} a\left(\frac x\varepsilon,\frac y\varepsilon\right)\dfrac{|\langle\xi,x-y\rangle|^2}{|x-y|^{d+2s}}dydx-\overline{a}\,\dfrac{\sigma_{d-1}}{2d}|\xi|^2\Big(r_2^{2(1-s)}-r_1^{2(1-s)}\Big)|A|\bigg\vert\\
        &\leq C\beta|\xi|^2\Big((1-s)\dfrac{\varepsilon}{r_1}|A|+|\partial_\varepsilon A|\Big).
    \end{align}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
In the following, $C=C(d)>0$ may change from line to line. Set
\[
    \kappa(x,y)\coloneqq(1-s)\dfrac{|\langle\xi,x-y\rangle|^2}{|x-y|^{d+2s}},\qquad \Lambda\coloneqq\big(A\times\mathbb{R}^d\big)\cap\{r_1\leq|x-y|\leq r_2\}.
\]
Integrating in polar coordinates and using that, by symmetry,
\begin{equation}\label{sphere_average}
    \int_{\mathbb{S}^{d-1}}|\langle \xi,\nu\rangle|^2d\sigma(\nu)=\dfrac{\sigma_{d-1}}{d}|\xi|^2
\end{equation}
for all $\xi\in\mathbb{R}^d$, we get
\begin{equation}\label{kappa_mass}
    \int_{\Lambda}\kappa(x,y)dydx=\dfrac{\sigma_{d-1}}{2d}|\xi|^2\big(r_2^{2(1-s)}-r_1^{2(1-s)}\big)|A|,
\end{equation}
so that the left-hand side of \eqref{averaging_estimate} equals the absolute value of
\[
    \int_\Lambda\left(a\left(\frac x\varepsilon,\frac y\varepsilon\right)-\overline{a}\right)\kappa(x,y)dydx.
\]
We partition $\Lambda$ into the sets
\begin{align}
    \Lambda_{1,\varepsilon}&\coloneqq\bigcup\left\{Q_{\varepsilon,j}\times Q_{\varepsilon,k}\,\middle\vert\, Q_{\varepsilon,j}\times Q_{\varepsilon,k}\subset \Lambda,\ Q_{\varepsilon,j}\subset A\right\},\\
    \Lambda_{2,\varepsilon}&\coloneqq \Lambda\cap\left(\partial_\varepsilon A\times\mathbb{R}^d\right),\\
    \Lambda_{3,\varepsilon}&\coloneqq \Lambda\setminus\left(\Lambda_{1,\varepsilon}\cup\Lambda_{2,\varepsilon}\right).
\end{align}
Note that $\Lambda_{3,\varepsilon}$ is covered by the cubes $Q_{\varepsilon,j}\times Q_{\varepsilon,k}$ with $Q_{\varepsilon,j}\subset A$ that meet $\Lambda$ without being contained in it.

Let $Q_{\varepsilon,j}\times Q_{\varepsilon,k}\subset\Lambda_{1,\varepsilon}$ and fix $(\overline x,\overline y)\in Q_{\varepsilon,j}\times Q_{\varepsilon,k}$. Since $a$ is $1$-periodic in both variables, we have
\begin{equation}\label{cell_average_a}
    \int_{Q_{\varepsilon,j}}\int_{Q_{\varepsilon,k}} a\left(\frac x\varepsilon,\frac y\varepsilon\right)dydx=\varepsilon^{2d}\,\overline{a}
\end{equation}
 for all $j,k\in\mathbb{Z}^d$, so that
\begin{align}
    \nonumber\int_{Q_{\varepsilon,j}}&\int_{Q_{\varepsilon,k}}\left(a\left(\frac x\varepsilon,\frac y\varepsilon\right)-\overline{a}\right)\kappa(x,y)dydx\\
    \label{cell_error}&=\int_{Q_{\varepsilon,j}}\int_{Q_{\varepsilon,k}}\left(a\left(\frac x\varepsilon,\frac y\varepsilon\right)-\overline{a}\right)\left(\kappa(x,y)-\kappa(\overline x,\overline y)\right)dydx.
\end{align}
In $\{|x-y|\geq r_1\}$, we have
\begin{equation}\label{kernel_gradient}
    |\nabla_{(x,y)}\kappa(x,y)|\leq \sqrt 2(d+4)(1-s)|\xi|^2|x-y|^{1-d-2s},
\end{equation}
and two points $(x,y)$, $(x',y')$ of $Q_{\varepsilon,j}\times Q_{\varepsilon,k}$ satisfy $|x'-y'|\geq r_1\geq\varepsilon$ and $|x-y|\leq|x'-y'|+2\sqrt d\,\varepsilon\leq(1+2\sqrt d)|x'-y'|$. Hence, by the mean-value theorem,
\begin{equation}\label{kernel_oscillation}
    |\kappa(x,y)-\kappa(\overline x,\overline y)|\leq C\varepsilon(1-s)|\xi|^2|x-y|^{1-d-2s}
\end{equation}
for $(x,y)\in Q_{\varepsilon,j}\times Q_{\varepsilon,k}$.

Now, we sum \eqref{cell_error} over the cubes of $\Lambda_{1,\varepsilon}$ and use \eqref{kernel_oscillation}, $|a|\le \beta$, $2s-1\geq1/2$, and $r_1^{2(1-s)}\leq1$. Thus, we obtain
\begin{align}
    \nonumber\Big|\int_{\Lambda_{1,\varepsilon}}&\left(a\left(\frac x\varepsilon,\frac y\varepsilon\right)-\overline{a}\right)\kappa(x,y)dydx\Big|\\
    \nonumber&\leq\sum_{Q_{\varepsilon,j}\times Q_{\varepsilon,k}\subset\Lambda_{1,\varepsilon}}\int_{Q_{\varepsilon,j}}\int_{Q_{\varepsilon,k}}\left\vert a\left(\frac x\varepsilon,\frac y\varepsilon\right)-\overline{a}\right\vert\left\vert\kappa(x,y)-\kappa(\overline x,\overline y)\right\vert dydx\\
    \nonumber&\leq 2\beta C\varepsilon(1-s)|\xi|^2\int_A\int_{\{r_1\leq|x-y|\leq r_2\}}\dfrac{1}{|x-y|^{d+2s-1}}dydx\\
    \nonumber&= 2\beta C\varepsilon(1-s)|\xi|^2|A|\,\sigma_{d-1}\int_{r_1}^{r_2}t^{-2s}dt\\
    &\leq C\beta\sigma_{d-1}(1-s)|\xi|^2\dfrac{\varepsilon}{r_1}|A|.
\end{align}
On $\Lambda_{2,\varepsilon}$ we use $|a|\le \beta$ and \eqref{kappa_mass}, which give the bound $\beta\frac{\sigma_{d-1}}{2d}|\xi|^2|\partial_\varepsilon A|$. Finally, if $Q_{\varepsilon,j}\times Q_{\varepsilon,k}$ meets $\Lambda$ without being contained in it, then it contains a pair $(x',y')$ with $|x'-y'|\in\{r_1,r_2\}$, hence all its points satisfy $$|x-y|\in[r_1-2\sqrt d\,\varepsilon,r_1+2\sqrt d\,\varepsilon]\cup[r_2-2\sqrt d\,\varepsilon,r_2+2\sqrt d\,\varepsilon].$$ Therefore,
\begin{align}
\nonumber\Lambda_{3,\varepsilon}\subset(A\times\mathbb{R}^d)\cap\big(\{r_1\leq|x-y|\leq r_1+2\sqrt d\,\varepsilon\}\cup\{r_2-2\sqrt d\,\varepsilon\leq|x-y|\leq r_2\}\big),
\end{align}
and, since $t\mapsto t^{1-2s}$ is decreasing, the integral of $\kappa$ on each of the two sets on the right-hand side is at most $2\sqrt d\,\sigma_{d-1}(1-s)|\xi|^2\varepsilon\, r_1^{1-2s}|A|$. Using $|a|\le \beta$ once more, the contribution of $\Lambda_{3,\varepsilon}$ is bounded by $C\beta\sigma_{d-1}(1-s)|\xi|^2\frac{\varepsilon}{r_1}|A|$, and \eqref{averaging_estimate} follows.

<!-- lemma-5: body_md -->
\label{l_short_range}
    Let $u \in H^1(\Omega)$. Consider a sequence $(u_\varepsilon) \subseteq L^2(\Omega)$ such that $u_\varepsilon \to u$ in $L^2(\Omega)$. Assume that there exists $C > 0$ such that 
    \[
        F_\varepsilon(u_\varepsilon, A_{3,\varepsilon}) \leq C
    \]
    for every $\varepsilon > 0$. Then
    \begin{equation}
        \liminf_{\varepsilon \to 0} F_\varepsilon(u_\varepsilon, A_{3,\varepsilon}) \geq \frac{\sigma_{d-1}}{2d} \lambda F_{\rm hom}(u),
    \end{equation}
      where $F_{\rm hom}$ is given by \eqref{fom} and \eqref{fomi}.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
We can follow the proof of the $\Gamma$-liminf inequality in \cite{BBD} essentially verbatim. 
Their argument (Section 3.1) is that, given a sequence $u_\varepsilon$ convergent to some $u$ in $L^2(\Omega)$, 
by discretization and averaging of the functions $u_\varepsilon$, one can construct a sequence $\overline u_\varepsilon$ such that 
$$
F_\varepsilon(u_\varepsilon,A_{3,\varepsilon})\ge r^{2(1-s)}\frac{\sigma_{d-1}}{2d} \int_{\Omega'}a\Big(\frac{x}\varepsilon,\frac{x}\varepsilon\Big) |\nabla \overline u_\varepsilon|^2dx
$$
for all $\Omega'$ open set compactly contained in $\Omega$ and $
\varepsilon $ small enough (see the last lines at page 12 in \cite{BBD} and \cite[Section 1.2.2]{Bruscatesi}). In their hypothesis $\lambda=1$ and  they show that this inequality actually holds asymptotically for the whole functional. This is not relevant for our case, since we directly estimate only short-range interactions. Thus, we apply this inequality together with the liminf inequality for the local homogenization theorem for the right-hand side. By the arbitrariness of $\Omega'$ and using \eqref{choice_auxiliary_microscopic_parameter}, the claim follows with the desired multiplicative constant $\lambda$.

<!-- lemma-6: body_md -->
\label{l_change_of_boundary_data}
    Fix $x_0$ and $N$, and let $\varepsilon\to 0$ and $\rho\,(=\rho(\varepsilon))$ be as above. 
    Let $v_{\varepsilon,i_\varepsilon}$ be defined in \eqref{definition_v}. Then, there exist $C>0$, independent of $N$, and a sequence of indices $(i_\varepsilon)\subset\{1,\ldots,N\}$ such that
    \begin{equation}\label{affine_datum_estimate}
        \liminf\limits_{\varepsilon\to 0}\frac{\mu_\varepsilon(Q_\rho(x_0))}{\mathscr{L}^d(Q_\rho(x_0))}\geq \liminf\limits_{\varepsilon\to 0}\frac{F_\varepsilon\left(v_{\varepsilon,i_\varepsilon},A_{1,\varepsilon}\cap Q_\rho(x_0)^2\right)}{\rho^d}-\frac CN,
    \end{equation}
     The same holds with $\theta_\varepsilon$ and $Q_\rho(x_0)^2$ in place of $\mu_\varepsilon$ and $A_{1,\varepsilon}\cap Q_\rho(x_0)^2$, for the same sequence $(i_\varepsilon)$.
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
In the following, $C>0$ may change from line to line and never depends on $N$. We also drop the index $\varepsilon$ from $v_{\varepsilon,i}$ and $\psi_{\varepsilon,i}$, dropping the base point $x_0$ and writing $Q_\rho\coloneqq Q_\rho(x_0)$, $\Lambda_\varepsilon\coloneqq A_{1,\varepsilon}\cap Q_\rho^2$ and $\varphi_\varepsilon\coloneqq u_\varepsilon-\ell$. Since $F_\varepsilon$ is invariant under addition of constants, we may also assume that $u(x_0)=0$, so that $\ell(x)=\langle z,x-x_0\rangle$ and $\|\ell\|^2_{L^2(Q_\rho)}\leq d|z|^2\rho^{d+2}$. By \eqref{theta_bound}, we have
    \begin{equation}\label{blowup_energy_bound}
        C_0\coloneqq\sup_{\varepsilon>0}\frac{\theta_\varepsilon(Q_\rho)}{\rho^d}<+\infty,\qquad \frac{F_\varepsilon(u_\varepsilon,\Lambda_\varepsilon)}{\rho^d}\leq\frac{\mu_\varepsilon(Q_\rho)}{\rho^d}\leq \frac{\theta_\varepsilon(Q_\rho)}{\rho^d},
    \end{equation}
    

    First, we record two estimates on the affine function $\ell$. Integrating in polar coordinates and using \eqref{sphere_average}, for every measurable $E\subset Q_\rho$ we have
    \begin{equation}\label{affine_energy}
        \frac{F_\varepsilon(\ell,E\times Q_\rho)}{\rho^d}\leq \beta|z|^2\frac{1-s}{\rho^d}|E|\sigma_{d-1}\int_0^{\sqrt d\rho}t^{1-2s}dt\leq \frac{\beta\sigma_{d-1}d}{2}|z|^2\frac{|E|}{\rho^d},
    \end{equation}
    where we also used that $\rho^{2(1-s)}\leq 1$. Since $|\{x\in Q_\rho\mid\mathrm{dist}(x,Q_\rho^c)\leq\rho/N\}|\leq 2d\rho^d/N$, \eqref{affine_energy} gives
    \begin{equation}\label{affine_energy_strip}
        \frac{F_\varepsilon(\ell,Q_\rho^2)}{\rho^d}\leq C|z|^2,\qquad \frac{F_\varepsilon\left(\ell,\{\mathrm{dist}(\cdot,Q_\rho^c)\leq\rho/N\}\times Q_\rho\right)}{\rho^d}\leq \frac{C|z|^2}{N}.
    \end{equation}

    We now estimate $F_\varepsilon(v_i,\Lambda_\varepsilon)$. For $x,y\in Q_\rho$ we have
\begin{eqnarray*}\label{cutoff_identity}
                \nonumber v_i(x)-v_i(y) &= & \ \psi_i(x)\left(u_\varepsilon(x)-u_\varepsilon(y)\right)\\
        &&+\left(1-\psi_i(x)\right)\left(\ell(x)-\ell(y)\right)+\left(\psi_i(x)-\psi_i(y)\right)\varphi_\varepsilon(y),    
        \end{eqnarray*}
    so that, by the convexity of $t\mapsto t^2$ applied to the first two terms, with weights $\psi_i(x)$ and $1-\psi_i(x)$, and by $(a+b)^2\leq 2a^2+2b^2$,
    \begin{equation}\label{cutoff_jensen}
        |v_i(x)-v_i(y)|^2\leq 2|u_\varepsilon(x)-u_\varepsilon(y)|^2+2|\ell(x)-\ell(y)|^2+2|\psi_i(x)-\psi_i(y)|^2|\varphi_\varepsilon(y)|^2.
    \end{equation}

    The interactions at distance larger than $w$ are negligible. Indeed, by Lemma \ref{l_2_large_interactions}, applied with $A=Q_\rho$, $r=w$ and $f=v_i$,
    \[
        \frac{F_\varepsilon\left(v_i,Q_\rho^2\cap\{|x-y|\geq w\}\right)}{\rho^d}\leq C\beta(1-s)N^{4s}\rho^{2(1-s)}\frac{\|v_i\|^2_{L^2(Q_\rho)}}{\rho^{d+2}},
    \]
    and, since $\|v_i\|^2_{L^2(Q_\rho)}\leq 2d|z|^2\rho^{d+2}+2\|\varphi_\varepsilon\|^2_{L^2(Q_\rho)}$, by \eqref{blowup_convergence}, and the fact that $(1-s)\to0$, we obtain
    \begin{equation}\label{far_interactions}
        \lim\limits_{\varepsilon\to0}\,\max\limits_{1\leq i\leq N}\frac{F_\varepsilon\left(v_i,Q_\rho^2\cap\{|x-y|\geq w\}\right)}{\rho^d}=0.
    \end{equation}

    We then estimate the contribution of the cut-off part. Let
    \[
        R_i\coloneqq\frac{1-s}{\rho^d}\int_{Q_\rho}\int_{Q_\rho}a\left(\frac x\varepsilon,\frac y\varepsilon\right)\frac{|\psi_i(x)-\psi_i(y)|^2|\varphi_\varepsilon(y)|^2}{|x-y|^{d+2s}}dxdy.
    \]
    By \eqref{properties_of_the_cutoff_liminf}, $|\psi_i(x)-\psi_i(y)|\leq\min\{1,2|x-y|/w\}$; hence, integrating in polar coordinates, for $s\geq1/2$ and every $y\in Q_\rho$,
    \[
        (1-s)\int_{\mathbb{R}^d}\min\Big\{1,\frac{4|x-y|^2}{w^2}\Big\}\frac{dx}{|x-y|^{d+2s}}\leq 4\sigma_{d-1}N^4\rho^{-2s},
    \]
    so that
    \begin{equation}\label{cutoff_error}
        \max\limits_{1\leq i\leq N}R_i\leq 4\beta\sigma_{d-1}N^4\rho^{2(1-s)}\frac{\|\varphi_\varepsilon\|^2_{L^2(Q_\rho)}}{\rho^{d+2}}\longrightarrow 0,
    \end{equation}
    by \eqref{blowup_convergence}.

    We decompose the remaining interactions. For $(x,y)\in Q_\rho^2$, we have $v_i=u_\varepsilon$ on $Q_{\rho,i}\times Q_{\rho,i}$ and $v_i=\ell$ on $\left(Q_\rho\setminus Q_{\rho,i-1}\right)^2$, while on 
    \[
        S_{\varepsilon,i}\coloneqq \Lambda_\varepsilon\cap\{|x-y|<w\}\setminus\Big(Q_{\rho,i}^2\cup(Q_\rho\setminus Q_{\rho,i-1})^2\Big)
    \]
    we use \eqref{cutoff_jensen}. Since $Q_\rho\setminus Q_{\rho,i-1}\subset\{\mathrm{dist}(\cdot,Q_\rho^c)\leq\rho/N\}$, by \eqref{affine_energy_strip}, \eqref{far_interactions}, and \eqref{cutoff_error} we obtain
    \begin{equation}\label{cutoff_decomposition}
        \frac{F_\varepsilon(v_i,\Lambda_\varepsilon)}{\rho^d}\leq \frac{F_\varepsilon(u_\varepsilon,\Lambda_\varepsilon)}{\rho^d}+\frac{C|z|^2}{N}+2\frac{F_\varepsilon(u_\varepsilon,S_{\varepsilon,i})+F_\varepsilon(\ell,S_{\varepsilon,i})}{\rho^d}+2R_i+o_\varepsilon(1).
    \end{equation}

    It remains to choose the index $i$. A pair $(x,y)\in\Lambda_\varepsilon$ with $|x-y|<w$ belongs to $S_{\varepsilon,i}$ if and only if 
    \begin{align*}
        \min\{\mathrm{dist}(x,Q_\rho^c),\mathrm{dist}(y,Q_\rho^c)\}\leq i\,w,\qquad
        \max\{\mathrm{dist}(x,Q_\rho^c),\mathrm{dist}(y,Q_\rho^c)\}>(i-1)w;
    \end{align*}
    that is, if and only if $i$ belongs to a half-open interval of length smaller than $2$, because $\mathrm{dist}(\cdot,Q_\rho^c)$ is $1$-Lipschitz and $|x-y|<w$. Hence, every such pair belongs to at most two of the sets $S_{\varepsilon,1},\ldots,S_{\varepsilon,N}$, and, by \eqref{blowup_energy_bound} and \eqref{affine_energy_strip},
    \[
        \sum_{i=1}^N2\frac{F_\varepsilon(u_\varepsilon,S_{\varepsilon,i})+F_\varepsilon(\ell,S_{\varepsilon,i})}{\rho^d}\leq 4\frac{F_\varepsilon(u_\varepsilon,\Lambda_\varepsilon)}{\rho^d}+4\frac{F_\varepsilon(\ell,Q_\rho^2)}{\rho^d}\leq C\left(C_0+|z|^2\right).
    \]
    We may then choose $i_\varepsilon\in\{1,\ldots,N\}$ for which the corresponding summand is smaller than $C\left(C_0+|z|^2\right)/N$, and \eqref{affine_datum_estimate} follows from \eqref{cutoff_decomposition} and \eqref{blowup_energy_bound}. Finally, the same argument applies if $\Lambda_\varepsilon$ is replaced by $Q_\rho^2$. Choosing $i_\varepsilon$ so as to minimize the sum of the two corresponding summands, the conclusion holds for both, up to possibly increasing the constant $C$ in the statement.

<!-- lemma-7: body_md -->
\label{l_periodization}
    Let $M\geq0$, and let $\varphi_\varepsilon$ be as in \eqref{definition_varphi}. Then the function
    \begin{equation}\label{definition_varphi_bar}
        \overline\varphi_\varepsilon\coloneqq\left(\frac\varepsilon\rho\right)^d\sum_{k\in K}\varphi_\varepsilon(\cdot-\varepsilon k)
    \end{equation}
    is $\varepsilon$-periodic, and satisfies $\|\overline\varphi_\varepsilon\|_{L^2(Q_\rho)}\leq\|\varphi_\varepsilon\|_{L^2(Q_\rho)}$ and
    \begin{equation}\label{periodization_estimate}
        \frac{F_\varepsilon\left(v_\varepsilon,A_{1,\varepsilon}\cap Q_\rho^2\right)}{\rho^d}\geq E^M_\varepsilon(\overline\varphi_\varepsilon)-\frac{C|z|^2}{N},
    \end{equation}
    where $C=C(d,\beta)>0$ does not depend on $M$ and $N$.
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Since the integrand is nonnegative, we have
    \begin{equation}\label{periodization_split}
        \frac{F_\varepsilon\left(v_\varepsilon,A_{1,\varepsilon}\cap Q_\rho^2\right)}{\rho^d}\geq\frac{F_\varepsilon\left(v_\varepsilon,A_{1,\varepsilon}\cap Q_\rho^2\cap\{|x-y|\leq \rho/N\}\right)}{\rho^d}=E^M_\varepsilon(\varphi_\varepsilon)-T,
    \end{equation}
    where
$$
        T\coloneqq\frac{1-s}{\rho^d}\int_{Q_\rho}\int_{\{M\varepsilon\leq|x-y|\leq \rho/N\}\setminus Q_\rho}a\left(\frac x\varepsilon,\frac y\varepsilon\right)
        \dfrac{|\langle z,x-y\rangle+\varphi_\varepsilon(x)-\varphi_\varepsilon(y)|^2}{|x-y|^{d+2s}}dydx.
$$    
    If $x\in Q_\rho$, $y\notin Q_\rho$ and $|x-y|\leq \rho/N$, then $\mathrm{dist}(x,Q_\rho^c)\leq|x-y|\leq\rho/N$, so that $\varphi_\varepsilon(x)=0$. Moreover, for almost every such $y$ there exists $m\in\mathbb{Z}^d\setminus\{0\}$ with $y\in Q_\rho+\rho m$, and, since $x$ does not belong to the interior of $Q_\rho+\rho m$,
    \[
        \mathrm{dist}\left(y,\left(Q_\rho+\rho m\right)^c\right)\leq|x-y|\leq\frac\rho N,
    \]
    whence $\varphi_\varepsilon(y)=\varphi_\varepsilon(y-\rho m)=0$ as well. Therefore, setting $\Sigma\coloneqq\{x\in Q_\rho\mid\mathrm{dist}(x,Q_\rho^c)\leq \rho/N\}$ and using \eqref{coercivity_boundedness_a}, $|\langle z,x-y\rangle|\leq|z||x-y|$ and $|\Sigma|\leq 2d\rho^{d}/N$, we obtain
    \begin{equation}\label{periodization_boundary}
        T\leq\beta|z|^2\frac{1-s}{\rho^d}\int_{\Sigma}\int_{\{|h|\leq \rho/N\}}\dfrac{dh}{|h|^{d+2s-2}}dx=\beta|z|^2\sigma_{d-1}\frac{|\Sigma|}{\rho^d}\dfrac{(\rho/N)^{2(1-s)}}{2}
        \leq\dfrac{C|z|^2}{N},
    \end{equation}
    where we also used that $(\rho/N)^{2(1-s)}\leq1$.

Changing variables by a translation $\varepsilon k$ in \eqref{periodic_energy}, and using the $1$-periodicity of $a$ in both variables together with the fact that the outer domain in \eqref{periodic_energy} may be replaced by any of its translates, one gets
    \begin{equation}\label{periodization_translation}
        E^M_\varepsilon(\varphi^k_\varepsilon)=E^M_\varepsilon(\varphi_\varepsilon),\qquad \varphi^k_\varepsilon\coloneqq\varphi_\varepsilon(\cdot-\varepsilon k),
    \end{equation}
    for every $k\in\mathbb{Z}^d$.

    The function $\overline\varphi_\varepsilon$ is $\varepsilon$-periodic. Indeed, for $j=1,\ldots,d$ and $x\in\mathbb{R}^d$,
    \[
        \overline\varphi_\varepsilon(x+\varepsilon e_j)=\left(\frac\varepsilon\rho\right)^d\sum_{k\in K}\varphi_\varepsilon\left(x-\varepsilon(k-e_j)\right)=\overline\varphi_\varepsilon(x),
    \]
    since $\varphi_\varepsilon$ is $\rho$-periodic.

    Finally, $E^M_\varepsilon$ is convex, being the integral of the square of an affine function of $\psi$, and, by \eqref{definition_varphi_bar}, $\overline\varphi_\varepsilon$ is a convex combination of the functions $\varphi^k_\varepsilon$, $k\in K$. Hence
    \[
        E^M_\varepsilon(\overline\varphi_\varepsilon)\leq\left(\frac\varepsilon\rho\right)^d\sum_{k\in K}E^M_\varepsilon(\varphi^k_\varepsilon)=E^M_\varepsilon(\varphi_\varepsilon),
    \]
    and \eqref{periodization_estimate} follows from \eqref{periodization_split} and \eqref{periodization_boundary}. The same convexity argument applied to $\|\cdot\|^2_{L^2(Q_\rho)}$ gives
    \[
        \|\overline\varphi_\varepsilon\|^2_{L^2(Q_\rho)}\leq\left(\frac\varepsilon\rho\right)^d\sum_{k\in K}\|\varphi^k_\varepsilon\|^2_{L^2(Q_\rho)}=\|\varphi_\varepsilon\|^2_{L^2(Q_\rho)},
    \]
    where we used that $\|\varphi^k_\varepsilon\|_{L^2(Q_\rho)}=\|\varphi_\varepsilon\|_{L^2(Q_\rho-\varepsilon k)}=\|\varphi_\varepsilon\|_{L^2(Q_\rho)}$, again by $\rho$-periodicity.

<!-- lemma-8: body_md -->
\label{l_corrector_long_range}
    Let $M\geq4\sqrt d$ and let $\overline\varphi_\varepsilon$ be as in \eqref{definition_varphi_bar}. Then
    \begin{equation}\label{corrector_long_range}
        P_\varepsilon\coloneqq\frac{1-s}{\rho^d}\int_{Q_\rho}\int_{\{M\varepsilon\leq|x-y|\leq\rho/N\}}a\left(\frac x\varepsilon,\frac y\varepsilon\right)\dfrac{|\overline\varphi_\varepsilon(x)-\overline\varphi_\varepsilon(y)|^2}{|x-y|^{d+2s}}dydx\leq\dfrac{C}{M^{2s}},
    \end{equation}
    where $C>0$ does not depend on $\varepsilon$, $M$ and $N$.
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
By Lemma \ref{l_change_of_boundary_data} and Lemma \ref{l_periodization}, both applied with $M=0$, and by \eqref{blowup_energy_bound}, we have $E^0_\varepsilon(\overline\varphi_\varepsilon)\leq C$. Hence, by \eqref{periodic_energy_cell}, $(a+b)^2\leq 2a^2+2b^2$, \eqref{coercivity_boundedness_a} and \eqref{sphere_average}, for every $k\in K$ we obtain
    \[
        \frac{1-s}{\varepsilon^d}\int_{Q_{\varepsilon,k}}\int_{Q_{\varepsilon,k}}\dfrac{|\overline\varphi_\varepsilon(x)-\overline\varphi_\varepsilon(y)|^2}{|x-y|^{d+2s}}dydx\leq\frac 2\alpha\left(E^0_\varepsilon(\overline\varphi_\varepsilon)+\beta\dfrac{\sigma_{d-1}}{2d}|z|^2\right)\leq C,
    \]
    where we also used that $\sqrt d\,\varepsilon\leq\rho/N$ for $\varepsilon$ small. Since $|x-y|^{-d-2s}\geq(\sqrt d\,\varepsilon)^{-d-2s}$ for $x,y\in Q_{\varepsilon,k}$, this gives
    \begin{equation}\label{cell_L2_bound}
        \int_{Q_{\varepsilon,k}}\int_{Q_{\varepsilon,k}}|\overline\varphi_\varepsilon(x)-\overline\varphi_\varepsilon(y)|^2dydx\leq C\dfrac{\varepsilon^{2d+2s}}{1-s}.
    \end{equation}
    On the other hand, by \eqref{periodic_energy_cell} and by the $\varepsilon$-periodicity of $\overline\varphi_\varepsilon$, writing $y=y'+\varepsilon k$ we get
    \begin{equation}\label{corrector_split}
        P_\varepsilon\leq\frac{\beta(1-s)}{\varepsilon^d}\sum_{k\in\mathbb{Z}^d}m_k\int_{Q_{\varepsilon,0}}\int_{Q_{\varepsilon,0}}|\overline\varphi_\varepsilon(x)-\overline\varphi_\varepsilon(y')|^2dy'dx,
    \end{equation}
    where
    \[
        m_k\coloneqq\sup\big\{|x-y|^{-d-2s}\mid x\in Q_{\varepsilon,0},\ y\in Q_{\varepsilon,k},\ |x-y|\geq M\varepsilon\big\},
    \]
    with the convention that $m_k=0$, if the above set is empty. For $x\in Q_{\varepsilon,0}$ and $y\in Q_{\varepsilon,k}$, we have $\varepsilon(|k|-\sqrt d)\leq|x-y|\leq\varepsilon(|k|+\sqrt d)$, so that $m_k=0$ unless $|k|\geq M-\sqrt d\geq M/2\geq2\sqrt d$, and in that case $|x-y|\geq\varepsilon|k|/2$, whence $m_k\leq\left(\varepsilon|k|/2\right)^{-d-2s}$. Moreover, if $h\in Q_{1,k}$ and $|k|\geq2\sqrt d$, then $|h|\leq|k|+\sqrt d\leq2|k|$ and $|h|\geq|k|-\sqrt d\geq M/4$, so that
    \begin{eqnarray*}
        \sum_{|k|\geq M/2}\dfrac{1}{|k|^{d+2s}}&\leq& 2^{d+2s}\sum_{|k|\geq M/2}\int_{Q_{1,k}}\dfrac{dh}{|h|^{d+2s}}
        \leq 2^{d+2s}\int_{\{|h|\geq M/4\}}\dfrac{dh}{|h|^{d+2s}}\\&=&\dfrac{2^{d+2s}\sigma_{d-1}}{2s}\left(\dfrac M4\right)^{-2s}.        
    \end{eqnarray*}
    Therefore,
    \begin{equation}\label{kernel_sum_bound}
        \sum_{k\in\mathbb{Z}^d}m_k\leq\left(\dfrac2\varepsilon\right)^{d+2s}\sum_{|k|\geq M/2}\dfrac{1}{|k|^{d+2s}}\leq\dfrac{C}{\varepsilon^{d+2s}M^{2s}},
    \end{equation}
    Putting together \eqref{cell_L2_bound}, \eqref{corrector_split}  and \eqref{kernel_sum_bound}, the claim follows.

<!-- proposition-1: body_md -->
\label{p_liminf_long_range}
    For $\mathscr{L}^d$-almost every $x_0\in\Omega$,
    \begin{equation}\label{density_bound}
        \frac{d\mu}{d\mathscr{L}^d}(x_0)\geq(1-\lambda)\dfrac{\sigma_{d-1}}{2d}\,\overline a\,|\nabla u(x_0)|^2.
    \end{equation}
    Consequently,
    \begin{equation}\label{liminf_long_range}
        \liminf\limits_{\varepsilon\to 0}F_\varepsilon(u_\varepsilon,A_{1,\varepsilon})\geq(1-\lambda)\dfrac{\sigma_{d-1}}{2d}\,\overline a\,\|\nabla u\|^2_{L^2(\Omega)}.
    \end{equation}
<!-- end proposition-1 -->
<!-- proof-9: body_md -->
Let $x_0$, $\rho$, $z$, and $\ell$ be as above, and fix $M\geq4\sqrt d$ and $N\geq2$. By Lemma \ref{l_change_of_boundary_data} and Lemma \ref{l_periodization},
    \begin{equation}\label{conclusion_step_1}
        \liminf\limits_{\varepsilon\to 0}\frac{\mu_\varepsilon(Q_\rho)}{\mathscr{L}^d(Q_\rho)}\geq\liminf\limits_{\varepsilon\to 0}E^M_\varepsilon(\overline\varphi_\varepsilon)-\dfrac{C}{N}.
    \end{equation}
    Expanding the square in \eqref{periodic_energy} and using the Cauchy--Schwarz inequality, we get
    \begin{align*}
        E^M_\varepsilon(\overline\varphi_\varepsilon)&\geq\frac{1-s}{\rho^d}\int_{Q_\rho}\int_{\{M\varepsilon\leq|x-y|\leq\rho/N\}}a\left(\frac x\varepsilon,\frac y\varepsilon\right)\dfrac{|\langle z,x-y\rangle|^2}{|x-y|^{d+2s}}dydx\\
        &\hphantom{\geq{}}-2\frac{1-s}{\rho^d}\int_{Q_\rho}\int_{\{M\varepsilon\leq|x-y|\leq\rho/N\}}a\left(\frac x\varepsilon,\frac y\varepsilon\right)
                                    
        \dfrac{|\langle z,x-y\rangle|\,|\overline\varphi_\varepsilon(x)-\overline\varphi_\varepsilon(y)|}{|x-y|^{d+2s}}dydx\\
        &\geq E^M_\varepsilon(0)-2\sqrt{E^M_\varepsilon(0)P_\varepsilon},
    \end{align*}
    where we discarded the nonnegative term $P_\varepsilon$ and recognized $E^M_\varepsilon(0)$ and $P_\varepsilon$ as mixed terms. Hence, by Lemma \ref{l_corrector_long_range},
    \begin{equation}\label{conclusion_step_2}
        E^M_\varepsilon(\overline\varphi_\varepsilon)\geq E^M_\varepsilon(0)-2\sqrt{E^M_\varepsilon(0)P_\varepsilon}\geq E^M_\varepsilon(0)-\dfrac{C}{M^s},
    \end{equation}
    where we also used $E^M_\varepsilon(0)\leq\beta\sigma_{d-1}|z|^2/(2d)$.

    It remains to compute the limit of $E^M_\varepsilon(0)$. Applying Lemma \ref{l_averaging_large_scale} with $A=Q_\rho$, $\xi=z$, $r_1=M\varepsilon$ and $r_2=\rho/N$, and using $|\partial_\varepsilon Q_\rho|\leq C\rho^{d-1}\varepsilon$, we get
    \[
        \Big|E^M_\varepsilon(0)-\overline a\,\dfrac{\sigma_{d-1}}{2d}|z|^2\Big(\left(\dfrac\rho N\right)^{2(1-s)}-(M\varepsilon)^{2(1-s)}\Big)\Big|\leq C\beta|z|^2\Big(\dfrac{1-s}M+\dfrac\varepsilon\rho\Big),
    \]
    which vanishes as $\varepsilon\to0$. Since $M$ and $N$ are fixed, $\left(\rho/N\right)^{2(1-s)}\to1$ and $(M\varepsilon)^{2(1-s)}\to\lambda$, by \eqref{rho_asymptotics} and by the choice of $\lambda$. We obtain
    \begin{equation}\label{conclusion_step_3}
        \lim\limits_{\varepsilon\to 0}E^M_\varepsilon(0)=(1-\lambda)\dfrac{\sigma_{d-1}}{2d}\,\overline a\,|z|^2.
    \end{equation}
    Putting together \eqref{conclusion_step_1}, \eqref{conclusion_step_2} and \eqref{conclusion_step_3}, and recalling the choice of $\rho$, we obtain
    \[
        \frac{d\mu}{d\mathscr{L}^d}(x_0)\geq(1-\lambda)\dfrac{\sigma_{d-1}}{2d}\,\overline a\,|z|^2-\dfrac{C}{M^s}-\dfrac CN,
    \]
    with $C$ independent of $M$ and $N$. Letting $M\to+\infty$ and $N\to+\infty$ gives \eqref{density_bound}. Since
    \[
        \liminf\limits_{\varepsilon\to 0}F_\varepsilon(u_\varepsilon,A_{1,\varepsilon})=\liminf\limits_{\varepsilon\to 0}\mu_\varepsilon(\Omega)\geq\mu(\Omega)\geq\int_\Omega\frac{d\mu}{d\mathscr{L}^d}(x)dx.
    \]
     we obtain \eqref{liminf_long_range}.
