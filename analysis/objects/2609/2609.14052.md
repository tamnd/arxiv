\label{thm:salizz}
    Let $F=\{f_1,\dots,f_r\}$ be a family of polynomials and let $\sigma$ be a degree-compatible term order. Then,
    \begin{equation*}
        d_{\operatorname{sol}}(F)\leq \operatorname{max}\{d_{\operatorname{reg}}(F)+1,\operatorname{deg}(f_1),\dots,\operatorname{deg}(f_r)\}.
    \end{equation*}

<!-- defin-1: body_md -->
Let $n\geq 1$ and $j\in[n]$. The \textit{elementary symmetric polynomial} of degree $j$ in the variables $x_1,\dots,x_n$ is
    \begin{equation*}
        e_{n,\, j}(x) = \sum_{\alpha\subseteq[n],\,|\alpha|=j}\hspace{-0.2cm}x_\alpha
        =\sum_{1\leq\alpha_1<\ldots<\alpha_j\leq n}\hspace{-0.3cm}x_{\alpha_1}\cdots x_{\alpha_i}\in R_{n,j}.
    \end{equation*}
<!-- end defin-1 -->
<!-- theo-3: body_md -->
\label{thm:lucas}
    Let $p$ be a prime, and let
    \begin{align*}
        m &= m_0 + m_1p + m_2p^2 + \dots + m_rp^r,\\
        n &= n_0 + n_1p + n_2p^2 + \dots + n_rp^r
    \end{align*}
    such that $0\leq m_i, n_i < p$ for every $i=0,\dots,r$ (named \textnormal{p-adic expansion} of $m$ and $n$, or just \textnormal{binary expansion} if $p=2$). Then
    \begin{equation*}
        \binom{m}{n} = \prod_{i=0}^r \binom{m_i}{n_i}\pmod{p}.
    \end{equation*}
<!-- end theo-3 -->
<!-- lem-1: body_md -->
\label{lemESPweight}
    Let $n\geq 1$ and $r=\fl*{\operatorname{log}_2 n}$. For $0\leq t\leq n$, let $t = t_0 + t_12 + \dots + t_r2^r$ be the binary expansion of $t$. For a vector $y\in\mathbb{F}_2^n$, the following conditions are equivalent:
    \begin{enumerate}[label=\arabic*)]
        \item $\operatorname{wt}(y)=t$,
        \item $e_{n,\, j}(y)=\binom{t}{j}$ for all $j=1,\dots,n$,
        \item $e_{n,\, 2^i}(y)=t_i$ for all $i=0,\dots,r$.
    \end{enumerate}
<!-- end lem-1 -->
<!-- proof-1: body_md -->
$\textit{"1)}\Rightarrow\textit{2)"}$: Let $y=(y_1,\dots,y_n)\in\mathbb{F}_2^n$ be a vector of weight $t$ and $T\subseteq [n]$ be the set of the indexes corresponding to the non-zero entries of $y$.                
    Then
    \begin{align*}
        e_{n,\, j}(y)
        &=\sum_{1\leq\alpha_1<\ldots<\alpha_j\leq n}\hspace{-0.5cm}y_{\alpha_1}\cdots y_{\alpha_j}
        =0+\hspace{-0.2cm}\sum_{\substack{1\leq\alpha_1<\ldots<\alpha_j\leq n,\\ \alpha_1,\ldots,\alpha_j\in T}}\hspace{-0.2cm}1=\\
        &=\sum_{\alpha\subseteq T,\,|\alpha|=j}\hspace{-0.2cm}1
        =|\{\alpha\subseteq T\mid |\alpha|=j\}|
        =\binom{t}{j}\pmod{2}.
    \end{align*}
    
    $\textit{"2)}\Rightarrow\textit{3)"}$: By Theorem~\ref{thm:lucas}, for every $i=0,\dots,r$ we have
    \begin{align*}
        e_{n,\, 2^i}(y)=\binom{t}{2^i}=\binom{t_i}{1}\cdot\prod_{\ell=0,\,\ell\neq i}^r \binom{t_\ell}{0}                                                                                  
        =t_i,
    \end{align*}
    and thus the thesis.
    
    $\textit{"3)}\Rightarrow\textit{1)"}$: Let $y\in\mathbb{F}_2^n$ be such that $e_{n,\, 2^i}(y)=t_i$ for all $i=0,\dots,r$ and suppose that $\operatorname{wt}(y)=t'$. Thanks to $\textit{"1)}\Rightarrow\textit{2)"}$, we have that $e_{n,\, j}(y)=\binom{t'}{j}$ for all $j=1,\dots,n$. Therefore, for every $i\in\{0,\ldots,r\}$, one has
    \begin{align*}
        t_i=e_{n,\, 2^i}(y)=\binom{t'}{2^i}=t'_i
    \end{align*}
    where the last equality follows from the proof of $\textit{"2)}\Rightarrow\textit{3)"}$. We conclude that $t=t'$.

By definition 
    \begin{equation*}
        e_{n,\, j}\cdote_{n,\, j'} =\left(\sum_{\alpha\subseteq[n],\,|\alpha|=j}\hspace{-0.3cm}x_\alpha\right)\left(\sum_{\beta\subseteq[n],\,|\beta|=j'}\hspace{-0.3cm}x_\beta\right)
        =\sum_{\substack{\alpha,\beta\subseteq[n],\\|\alpha|=j,\,|\beta|=j'}}\hspace{-0.3cm}x_\alpha x_\beta.
    \end{equation*}
    
    If $j+j'>n$, then $x_\alpha$ and $x_\beta$ share a variable for every $\alpha,\beta\subseteq [n]$. Therefore $x_\alpha x_\beta\in Q_n$ and $e_{n,\, j}\cdote_{n,\, j'}=0$ modulo $Q_n$.

    If $j+j'\leq n$, then modulo $Q_n$
    \begin{align*}
        e_{n,\, j}\cdote_{n,\, j'}
        &=\sum_{\substack{\alpha,\beta\subseteq[n],\\|\alpha|=j,\,|\beta|=j'}}\hspace{-0.3cm}x_\alpha x_\beta
        =\sum_{\substack{\alpha,\beta\subseteq[n],\,\alpha\,\cap\,\beta=\varnothing,\\|\alpha|=j,\,|\beta|=j'}}\hspace{-0.3cm}x_\alpha x_\beta=\\
        &=\sum_{\gamma\subseteq[n],\,|\gamma|=j+j'}\left(\sum_{\substack{\alpha,\beta\subseteq[n],\,|\alpha|=j,\,|\beta|=j',\\\alpha\,\cap\,\beta=\varnothing,\,\alpha\,\cup\,\beta=\gamma}}\hspace{-0.5cm}x_\alpha x_\beta\right)=\\
        &=\sum_{\gamma\subseteq[n],\,|\gamma|=j+j'}\left(\lvert\{(\alpha,\beta)\mid\substack{\alpha,\beta\subseteq [n],\,|\alpha|=j,\,|\beta|=j',\\\alpha\,\cap\,\beta=\varnothing,\,\alpha\,\cup\,\beta=\gamma}\}\rvert\cdot\, x_\gamma\right)=\\
        &=\sum_{\gamma\subseteq[n],\,|\gamma|=j+j'}\left(\lvert\{\alpha\subseteq [n]\mid|\alpha|=j,\,\alpha\subseteq\gamma\}\rvert\cdot\, x_\gamma\right)=\\
        &=\sum_{\gamma\subseteq[n],\,|\gamma|=j+j'}\binom{j+j'}{j}x_\gamma
        =\binom{j+j'}{j}e_{n,\, j+j'}.
    \end{align*}
    By Lucas' Theorem (Theorem~\ref{thm:lucas})
    \begin{equation*}
        \binom{j+j'}{j}=\prod_{i=0}^r\binom{j_i+j'_i}{j_i}.
    \end{equation*}
    Therefore, if there exists $\ell$ such that $j_\ell=1=j_\ell'$, then $\binom{j_\ell+j'_\ell}{j_\ell}=\binom{0}{1}=0$, so $\binom{j+j'}{j}=0$ and $e_{n,\, j}\cdote_{n,\, j'}=0$.
    If no such $\ell$ exists, then $\binom{j_i+j'_i}{j_i}=1$ for all $i$, hence $\binom{j+j'}{j}=1$ and $e_{n,\, j}\cdote_{n,\, j'}=e_{n,\, j+j'}$.

The quotient $I_{n,j}/Q_{n,j}$ is generated by all products of the form $x_\alpha\cdote_{n,\, m}$, where $1\leq m\leq j$ and $x_\alpha$ is a squarefree monomial of degree $j-m$.     \item the columns are indexed by the squarefree monomials  of degree $j$ in $n$ variables,
    \item the entry in row $x_\alpha\cdote_{n,\, m}$ and column $x_\beta$ is the coefficient of the monomial $x_\beta$ in the polynomial $x_\alpha\cdote_{n,\, m}$.
\end{itemize}

Throughout the proof, we denote $$K_n=(e_{k,\, 2^i}(x_1,\dots,x_k)+e_{n-k,\, 2^i}(x_{k+1},\dots,x_n)\mid i=0,\dots,r-1)+Q_n,$$ $$K'_n=(e_{k,\, j}(x_1,\dots,x_k)+e_{n-k,\, j}(x_{k+1},\dots,x_n)\mid j=1,\dots,2^r-1)+Q_n.$$ We also write $e_{k,\, j}$ for $e_{k,\, j}(x_1,\dots,x_k)$, and similarly for $e_{n,\, j}$ and $e_{n-k,\, j}$.

    The second equality follows from Lemma~\ref{lem:belprodottoesp}.
    It is clear that $K_n\subseteq K_n'$. Hence, to prove the chain of equalities in the statement, it suffices to prove the inclusions $K_n'\subseteq J_n$ and $J_n\subseteq K_n$.
    
    $K_n'\subseteq J_n$: We prove that $e_{k,\, j}+e_{n-k,\, j}\in J_n$ for all $j=1,\dots,2^r-1$ by induction on $j$. Since $e_{k,\, j}+e_{n-k,\, j}=0$ for $j>\max\{k,n-k\}$, we may restrict to $j\leq\max\{k,n-k\}$.

        \end{equation*}

            q_j=\sum\limits_{i=1}^{j-1}(p_i+e_{n-k,\, i})e_{n-k,\, j-i}=\sum\limits_{i=1}^{j-1}p_i\cdote_{n-k,\, j-i}+\sum\limits_{i=1}^{j-1}e_{n-k,\, i}\cdote_{n-k,\, j-i}.

Since $\alpha=1$, we obtain $q_j=p''_j+e_{n-k,\, j}\in J_n$. This concludes the proof. 

        \item If $j>\min\{k,n-k\}$, up to exchanging $k$ and $n-k$ we may assume without loss of generality that $k<j\leq n-k$. We want to prove that $e_{k,\, j}+e_{n-k,\, j}=e_{n-k,\, j}\in J_n$. We proceed as before:

            e_{n,\, j}=\sum\limits_{i=0}^je_{k,\, i}\cdote_{n-k,\, j-i}\overset{j>k}{=}\sum\limits_{i=1}^ke_{k,\, i}\cdote_{n-k,\, j-i}+e_{n-k,\, j}.
        \end{equation*}

            \sum\limits_{i=1}^ke_{k,\, i}\cdote_{n-k,\, j-i}=\sum\limits_{i=1}^ke_{k,\, i}(p_{j-i}+e_{k,\, j-i})=\sum\limits_{i=1}^ke_{k,\, i}\cdot p_{j-i}+\sum\limits_{i=1}^ke_{k,\, i}\cdote_{k,\, j-i},
        \end{equation*}
        where the first summand belongs to $J_n$, since by induction $p_i\in J_n$ if $i<j$, and the second summand is $0$ by Lemma~\ref{lem:belprodottoesp}. Therefore, $e_{n-k,\, j}\in J_n$ as desired.
    \end{itemize}

    $J_n\subseteq K_n$: We want to prove that $e_{n,\, 2^i}\in K_n$ for $i=0,\dots,r-1$. We proceed by induction on $i$. The case $i=0$ is clear. Hence, suppose $i>0$ and assume that $e_{n,\, 2^j}\in K_n$ for $0\leq j\leq i-1$. By Lemma~\ref{lem:belprodottoesp}, this implies $e_{n,\, j}\in K_n$ for $1\leq j\leq 2^i-1$. By Lemma~\ref{full prod}, moreover,
    \begin{equation*}
        e_{n,\, 2^i}=\sum\limits_{j=0}^{2^i}e_{k,\, j}\cdote_{n-k,\, 2^i-j}=e_{k,\, 2^i}+\sum\limits_{j=1}^{2^i-1}e_{k,\, j}\cdote_{n-k,\, 2^i-j}+e_{n-k,\, 2^i}.
    \end{equation*}
    So it suffices to prove that $\sum\limits_{j=1}^{2^i-1}e_{k,\, j}\cdote_{n-k,\, 2^i-j}\in K_n$.

    Let $1\leq u\leq\min\{k,2^i-1\}$, then by Lemma~\ref{full prod}
    \begin{align*}
        K_n\nie_{k,\, u}\cdote_{n,\, 2^i-u}&=e_{k,\, u}\sum\limits_{j=0}^{2^i-u}e_{k,\, j}\cdote_{n-k,\, 2^i-u-j}=\\
                                                                                                                               
        &=\sum\limits_{j'=u}^{2^i-1}e_{k,\, u}\cdote_{k,\, j'-u}\cdote_{n-k,\, 2^i-j'}.
    \end{align*}
    The last equality follows from setting $j'=j+u$ and by observing that $e_{k,\, u}\cdote_{k,\, 2^i-u}=0$ for every $u$ by Lemma~\ref{lem:belprodottoesp}, since $u$ and $2^i-u$ must share at least one non-zero bit in their binary expansion. 
                                                                            
    Therefore
    \begin{align*}
        K_n\ni\sum\limits_{u=1}^{\min\{k,2^i-1\}}e_{k,\, u}\cdote_{n,\, 2^i-u}
        &=\sum\limits_{u=1}^{\min\{k,2^i-1\}}\sum\limits_{j'=u}^{2^i-1}e_{k,\, u}\cdote_{k,\, j'-u}\cdote_{n-k,\, 2^i-j'}=\\
        &=\sum\limits_{j'=1}^{2^i-1}\sum\limits_{u=1}^{j'}e_{k,\, u}\cdote_{k,\, j'-u}\cdote_{n-k,\, 2^i-j'}=\\
        &=\sum\limits_{j'=1}^{2^i-1}|T_{j'}|\cdote_{k,\, j'}\cdote_{n-k,\, 2^i-j'},
    \end{align*}
    where $T_{j'}$ is defined, following Lemma~\ref{lem:belprodottoesp}, as the set
    \begin{align*}
        T_{j'} &=\{u\in[1,j']\mid \text{there is no $\ell$ such that $u_\ell=1=(j'-u)_\ell$}\}=\\
        &=\{u\in[1,j']\mid \text{ if } j'_\ell =0 \text{ for some }\ell, \text{ then } t_\ell=0\}.
    \end{align*}
    Notice that if $j'>k$, then $e_{k,\, j'}=0$ and $T_{j'}$ plays no role.   
    In addition, for every $j'$
    \begin{equation*}
        |T_{j'}|=2^{|\{\ell\in[0,\lceil\operatorname{log}_2(j')\rceil]\mid j'_\ell =1\}|}-1,
    \end{equation*}
    since, if $j'_\ell =1$, then $u_\ell$ can be either $0$ or $1$, however $0\not\in T_{j'}$. In particular, $|T_{j'}|$ is odd for all $j'\leq k$. It follows that $$\sum\limits_{j=1}^{2^i-1}e_{k,\, j}\cdote_{n-k,\, 2^i-j}=\sum\limits_{u=1}^{\min\{k,2^i-1\}}e_{k,\, u}\cdote_{n,\, 2^i-u}\in K_n,$$  
    which concludes the proof.

<!-- proof-11: body_md -->
We divide the proof in three cases, one of which is divided in three subcases.
\begin{itemize}
\item If $k\leq 2^r-1$, then $e_{k,\, j}(x)=e_{k,\, j}$ vanishes for every $j> k$, so $J'_k=(e_{k,\, j}(x)+e_{n-k,\, j}(\ell)\mid 1\leq j\leq\operatorname{rk}(\mathbf{L}))+(e_{k,\, j}(x)\mid \operatorname{rk}(\mathbf{L})<j\leq k)+Q_k.$

\begin{equation}\label{dregperj}
    d_{\operatorname{reg}}(J_k')\leq \fl*{\frac{k+\operatorname{rk}(\mathbf{L})}{2}}+1.
\end{equation}
If instead $\operatorname{rk}(\mathbf{L})=k$, (\ref{dregperj}) still holds, since $d_{\operatorname{reg}}(J_k')\leq d_{\operatorname{reg}}(Q_k)=k+1$.

\item If $2^r\leq k$ and $\operatorname{rk}(\mathbf{L})\leq2^{r}-2$, using Lemma~\ref{lemEspcascata} as in the proof of Proposition~\ref{propAlgoESP}, but starting from the last polynomials of the list, we obtain:

    \begin{array}{c | c c c c c c c c}
         \text{Step 0} & e_{k,\, \operatorname{rk}(\mathbf{L})+1} & e_{k,\, \operatorname{rk}(\mathbf{L})+2} & \dots & e_{k,\, 2^r-2} & e_{k,\, 2^r-1} & \in J_k'\\
         \text{Step 1} & / & e_{k-1,\, \operatorname{rk}(\mathbf{L})+2} & \dots & e_{k-1,\, 2^r-2} & e_{k-1,\, 2^r-1} & \in J_k'\\
         \vdots & / & / & \vdots & \vdots & \vdots & \in J_k'.\\
    \end{array}
\end{equation*}
If $\operatorname{rk}(\mathbf{L})+1+s\leq 2^r-1\leq k-s$, that is, for
\begin{equation*}
    s\leq\min\{2^r-2-\operatorname{rk}(\mathbf{L}),k-2^r+1\},
\end{equation*} in Step $s$ we have $e_{k-s,\, \operatorname{rk}(\mathbf{L})+1+s},\dots,e_{k-s,\, 2^r-1}\in J_k'$. We analyze three different cases, depending on the value of $k+\operatorname{rk}(\mathbf{L})$. Notice that $k+\operatorname{rk}(\mathbf{L})\leq n\leq 2^{r+1}-1$.
\begin{itemize}
\item If $k+\operatorname{rk}(\mathbf{L})\leq 2^{r+1}-3$, then the last meaningful step happens for $\bar s=k-2^r+1$. In this case,

$$d_{\operatorname{reg}}(J_k')\leq\fl*{\frac{k+\operatorname{rk}(\mathbf{L})}{2}}+1.
$$
\item If $k+\operatorname{rk}(\mathbf{L})=2^{r+1}-2$, then the last meaningful step happens for $\bar s=2^r-2-\operatorname{rk}(\mathbf{L})=k-2^r$.
In this case, $e_{2^r,\, 2^r-1}\in J_k'$, therefore $x_{2^r}\cdote_{2^r,\, 2^r-1}=x_{2^r}\cdote_{2^r-1,\, 2^r-1}+x_{2^r}^2\cdote_{2^r-1,\, 2^r-2}\in J_k'$, hence $e_{2^r,\, 2^r}\in J'_k$. \item If $k+\operatorname{rk}(\mathbf{L})=2^{r+1}-1$, then the last meaningful step happens for $\bar s=2^r-2-\operatorname{rk}(\mathbf{L})=k-2^r-1$.
In this case, $e_{2^r+1,\, 2^r-1}\in J_k'$, hence
\begin{align*}
    x_{2^r+1}x_{2^r}&\cdote_{2^r+1,\, 2^r-1}
    =x_{2^r+1}x_{2^r}(e_{2^r,\, 2^r-1}+x_{2^r+1}e_{2^r,\, 2^r-2})=\\
    &=x_{2^r+1}x_{2^r}(e_{2^r-1,\, 2^r-1}+x_{2^r}e_{2^r-1,\, 2^r-2}+x_{2^r+1}e_{2^r,\, 2^r-2})=\\
    &=x_{2^r+1}x_{2^r}e_{2^r-1,\, 2^r-1}+x_{2^r+1}x_{2^r}^2e_{2^r-1,\, 2^r-2}+x_{2^r+1}^2x_{2^r}e_{2^r,\, 2^r-2}=\\
    &=e_{2^r+1,\, 2^r+1}+x_{2^r+1}x_{2^r}^2e_{2^r-1,\, 2^r-2}+x_{2^r+1}^2x_{2^r}e_{2^r,\, 2^r-2}
\end{align*}
belongs to $J_k'$, and therefore $e_{2^r+1,\, 2^r+1}\in J_k'$. We obtain the bound

    d_{\operatorname{reg}}(J_k')\leq \fl*{\frac{k+\operatorname{rk}(\mathbf{L})}{2}}+2=2^r+1.
\end{equation}
Notice that this case only occurs when $n=2^{r+1}-1$ and $\operatorname{rk}(\mathbf{L})=n-k\leq 2^r-2\leq k-2$, which implies $k\geq 2^r+1$. 
\end{itemize}
\item If $2^r\leq k$ and $\operatorname{rk}(\mathbf{L})\geq 2^{r}-1$, then $n\geq k+\operatorname{rk}(\mathbf{L})\geq 2^{r+1}-1$. This forces $n=2^{r+1}-1$, $k=2^r$ and $\operatorname{rk}(\mathbf{L})=n-k=2^r-1$. \end{itemize}

<!-- proof-13: body_md -->
Since $d(\mathbf{C})\leq k$, the set $\{i\in[k]\mid d_i(\mathbf{C})\leq k\}$ is not-empty, and therefore we can consider its maximum $r=\max\{i\in[k]\mid d_i(\mathbf{C})\leq k\}$. If $r=k$, then $\operatorname{rk}(\mathbf{L})\geq 0$ is obviously true. Otherwise, we have that for any $j>r$, $d_j(\mathbf{C})>k$, and therefore for any subcode $\mathbf{D}$ of $\mathbf{C}$ such that $\operatorname{dim}(\mathbf{D})=j>r$, we have $|\operatorname{supp}(\mathbf{D})|>k$. Therefore, any subcode $\mathbf{D}\leq\mathbf{C}$ such that $|\operatorname{supp}(\mathbf{D})|\leq k$, we have $\operatorname{dim}(\mathbf{D})\leq r$.
    
    We now consider the subcode $\mathbf{V}=\mathbf{C}\cap\{(x_1,\dots,x_n)\in\mathbb{F}_2^n\mid x_{k+1}=\ldots=x_n=0\}$. The support of $\mathbf{V}$ is contained in $\{1,\dots,k\}$, and therefore $|\operatorname{supp}(\mathbf{V})|\leq k$, and thus $\operatorname{dim}(\mathbf{V})\leq r$ for the above considerations. It is also true that $\mathbf{V}\cong\operatorname{Ker}(\mathbf{L})$, since for any $x\in\mathbb{F}_2^k$ it holds
    \begin{equation*}
        \mathbf{L}\cdot x^\top=0\iff \mathbf{L}\cdot x^\top+\mathbf{I}_{n-k}\cdot0^\top=0\iff\mathbf{H}\cdot(x\mid 0)^\top=0\iff(x\mid 0)\in\mathbf{V}.
    \end{equation*}
    Therefore, we have $\operatorname{dim}(\operatorname{Ker}(\mathbf{L}))\leq r$, which implies $\operatorname{rk}(\mathbf{L})=\operatorname{dim}(\operatorname{Im}(\mathbf{L}))\geq k-r$ as desired.
