<!-- theorem-1: body_md -->
\label{thm:main}
Let $p$ be a prime, $a\in\mathbb{F}_p^\times$, and $P_a=T^{\,p}-T-a\in\mathbb{F}_p[T]$. Then $P_a$ is irreducible
of degree $p$. For every $n$ with $0\le n\le p-1$, setting $i=p^{\,n}-1$, one has in $A/P_a$
\[
   D(X)\;:=\;\sum_{m=0}^{p-1}S_m(i)\,X^{m}\;\equiv\;(1-X)^{\,n}\pmod{P_a},
   \qquad S_m(i):=\sum_{a'\in A_m}(a')^{\,i}.
\]
Equivalently, $S_m(p^{\,n}-1)\equiv(-1)^{m}\binom{n}{m}\pmod{P_a}$ for all $m$.
Consequently
\[
   \deg_X g\big(X,\omega_{P_a}^{\,p^{\,n}-1}\big)=n-1.
\]
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:counter}
For every prime $p\ge 5$ and every $n$ with $3\le n\le p-1$, and every $a\in\mathbb{F}_p^\times$,
the prime $P_a=T^{\,p}-T-a$ and the $q$-magic index $i=p^{\,n}-1$ satisfy
\[
   \deg_X g\big(X,\omega_{P_a}^{\,i}\big)=n-1\ \ge\ 2 .
\]
In particular Goss's conjecture is false; there are infinitely many such pairs $(\,p,P_a,i\,)$,
and the excess over the conjectured bound, $\deg_X g-1=n-2$, is unbounded, attaining $p-3$
at $n=p-1$.
For $p\in\{2,3\}$ the construction produces no counterexample, since $\deg_X g=n-1\le p-2\le 1$.
<!-- end corollary-1 -->
<!-- lemma-1: body_md -->
\label{lem:eval}
For all $m$ and $n$, $\,S_m(p^{\,n}-1)\equiv T_m(na)\pmod{P_a}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For monic $a'$ with $P_a\nmid a'$, put $\bar a'=a'(\theta)\in\mathbb{F}_{p^{p}}^\times$. Since $a'$ has
coefficients in $\mathbb{F}_p$ (fixed by $\varphi$) and $\varphi^{\,n}(\theta)=\theta+na$,
\[
   \bar a'^{\,p^{\,n}-1}=\frac{\bar a'^{\,p^{\,n}}}{\bar a'}
   =\frac{\varphi^{\,n}\!\big(a'(\theta)\big)}{a'(\theta)}
   =\frac{a'\!\big(\varphi^{\,n}(\theta)\big)}{a'(\theta)}
   =\frac{a'(\theta+na)}{a'(\theta)} .
\]
If $P_a\mid a'$ then $\bar a'^{\,p^{\,n}-1}=0$. Summing over $a'\in A_m$ (with $m\le p-1$, so no such
divisible term occurs) gives the claim.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:base}
$T_m(0)=p^{\,m}=[\,m=0\,]$; equivalently $\sum_m T_m(0)X^m=1$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
$T_m(0)=\sum_{a'\in A_m}1=\#A_m=p^{\,m}$, which is $0$ in $\mathbb{F}_{p^{p}}$ for $m\ge1$ and $1$ for $m=0$.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:rec}
For every $b\in\mathbb{F}_p$ and every $m$ with $1\le m\le p-1$,
\[
   T_m(b+a)=T_m(b)-T_{m-1}(b).
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Write $\Delta f(Y):=f(Y+a)-f(Y)$.

\medskip
\emph{Step 1.} Because $a'\in\mathbb{F}_p[Y]$ and $(\theta+b)^{p}=\theta^{p}+b^{p}=(\theta+a)+b=\theta+b+a$,
\[
   a'(\theta+b+a)=a'\big((\theta+b)^{p}\big)=a'(\theta+b)^{p}.
\]
Hence
\[
   T_m(b+a)-T_m(b)=\sum_{a'\in A_m}\frac{a'(\theta+b+a)-a'(\theta+b)}{a'(\theta)}
   =\sum_{a'\in A_m}\frac{(\Delta a')(\theta+b)}{a'(\theta)} .
\]

\medskip
\emph{Step 2.} Each $a'\in A_m$ is uniquely $a'(Y)=Y\,q(Y)+c_0$ with $q\in A_{m-1}$ monic and
$c_0\in\mathbb{F}_p$. Put $F_q(Y):=Y\,q(Y)$. Then $\Delta a'=\Delta F_q$ (the constant $c_0$ cancels) and
$a'(\theta)=F_q(\theta)+c_0$. Summing first over $c_0\in\mathbb{F}_p$, and using
$\prod_{c\in\mathbb{F}_p}(X+c)=X^{p}-X$, whose logarithmic derivative gives
$\sum_{c\in\mathbb{F}_p}\frac1{s+c}=-\frac1{s^{p}-s}$ for $s\notin\mathbb{F}_p$, we get
\[
   \sum_{c_0\in\mathbb{F}_p}\frac{1}{F_q(\theta)+c_0}=-\frac{1}{F_q(\theta)^{p}-F_q(\theta)}
   =-\frac{1}{(\Delta F_q)(\theta)} ,
\]
the last equality because $F_q(\theta)^{p}-F_q(\theta)=F_q(\theta^{p})-F_q(\theta)
=F_q(\theta+a)-F_q(\theta)=(\Delta F_q)(\theta)$. (Here $s=F_q(\theta)=\theta q(\theta)$ has a
nonzero $\theta^{m}$-term with $1\le m\le p-1$, so $s\notin\mathbb{F}_p$ and every denominator is nonzero.)
Therefore
\[
   T_m(b+a)-T_m(b)=-\sum_{q\in A_{m-1}}\frac{(\Delta F_q)(\theta+b)}{(\Delta F_q)(\theta)} .
\]

\medskip
\emph{Step 3.} The map $\Phi\colon q\mapsto \Delta\big(Y\,q(Y)\big)/(m\,a)$ is a bijection of the set
of monic polynomials of degree $m-1$ onto itself, preserving the leading coefficient. Indeed it is
$\mathbb{F}_p$-affine on $\{\deg\le m-1\}$, and its linear part $q\mapsto \Delta(Yq)$ has trivial kernel:
$\Delta(Yq)=0$ means $Yq$ is invariant under $Y\mapsto Y+a$, hence lies in $\mathbb{F}_p[\,Y^{p}-Y\,]$;
but $\deg(Yq)=m\le p-1<p$ forces $Yq$ constant, hence $q=0$. As a bijection of the $m$-dimensional
space $\{\deg\le m-1\}$ it is therefore injective on the finite set of monic degree-$(m-1)$
polynomials, hence bijective there. Moreover the coefficient of $Y^{m-1}$ in $\Delta(Yq)$ equals
$\lambda\,m\,a$, where $\lambda$ is the leading coefficient of $q$; dividing by $m\,a$ (with
$m\,a\ne0$ since $1\le m\le p-1$) preserves $\lambda$, so $\Phi$ maps monic to monic. Since the ratio
$(\Delta F_q)(\theta+b)/(\Delta F_q)(\theta)=\Phi(q)(\theta+b)/\Phi(q)(\theta)$ is unchanged under
scaling, reindexing by $a''=\Phi(q)$ gives
\[
   \sum_{q\in A_{m-1}}\frac{(\Delta F_q)(\theta+b)}{(\Delta F_q)(\theta)}
   =\sum_{a''\in A_{m-1}}\frac{a''(\theta+b)}{a''(\theta)}=T_{m-1}(b).
\]
Combining the three steps gives $T_m(b+a)-T_m(b)=-T_{m-1}(b)$.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
Let $G(X;b):=\sum_{m\ge0}T_m(b)X^{m}$. By Lemma~\ref{lem:rec} the coefficient of $X^{m}$ in
$(1-X)\,G(X;na)$ is $T_m(na)-T_{m-1}(na)=T_m\big((n+1)a\big)$ for $1\le m\le p-1$, i.e.
$G(X;(n+1)a)=(1-X)\,G(X;na)$ up to degree $p-1$; with $G(X;0)=1$ (Lemma~\ref{lem:base}) and
$0\le n\le p-1$ this yields $G(X;na)=(1-X)^{\,n}$. Since $(1-X)^{\,n}$ has degree $n\le p-1$, we have
$T_m(na)=(-1)^{m}\binom{n}{m}$ for all $m$, in particular $T_m(na)=0$ for $m>n$. By
Lemma~\ref{lem:eval}, $S_m(p^{\,n}-1)\equiv T_m(na)\pmod{P_a}$, so
\[
   D(X)=\sum_{m=0}^{p-1}S_m(p^{\,n}-1)X^{m}\equiv(1-X)^{\,n}\pmod{P_a}.
\]
The passage to $\deg_X g$ is given by Proposition~\ref{prop:degg} below, which yields
$\deg_X g(X,\omega_{P_a}^{\,p^{\,n}-1})=\operatorname{ord}_{X=1}\!\big(D\bmodP_a\big)-1=n-1$.

\label{prop:degg}
Let $P$ be a prime of degree $d$ and let $i$ with $1\le i\le q^{d}-2$ satisfy $i\equiv0\pmod{q-1}$,
so that $\chi=\omega_P^{\,i}$ is a nontrivial \emph{even} character. Let
$D_\chi(X)=\sum_{m=0}^{d-1}\big(\sum_{a'\in A_m}\chi(a')\big)X^{m}\in (A/P)[X]$ be the associated
Dirichlet polynomial. Then
\[
   D_\chi(X)=(1-X)\,L(X,\chi)\quad\text{in characteristic }0,
\]
where $L(X,\chi)$ is the $L$-polynomial of \cite[Lem.~2.4]{Angles}, of degree $d-2$; in particular
$D_\chi(1)=0$. Consequently
\[
   \deg_X g(X,\chi)=\operatorname{ord}_{X=1}\!\big(L(X,\chi)\bmod p\big)=\operatorname{ord}_{X=1}\!\big(D_\chi\bmod P\big)-1.
\]
<!-- end proposition-1 -->
<!-- proof-5: body_md -->
The Dirichlet polynomial $D_\chi$ is the finite Euler product
$\prod_{v\ne P,\infty}(1-\chi(v)X^{\deg v})^{-1}$. Since $\chi$ is even, the inertia group $\mathbb{F}_q^\times$
of $\infty$ in $K_P/k$ lies in $\ker\chi$, so $\infty$ is unramified in the fixed field of $\ker\chi$
and $\chi(\infty)=1$; the Euler factor at $\infty$ is $(1-X)^{-1}$. Hence the complete $L$-function
equals $D_\chi(X)/(1-X)$, which for the nontrivial $\chi$ is the polynomial $L(X,\chi)$ of
\cite[Lem.~2.4]{Angles}, of degree $d-2$. Thus $D_\chi=(1-X)L(X,\chi)$ and $D_\chi(1)=0$.

For the last claim, write $L(X,\chi)=\prod_j(1-\alpha_j X)$ over $\overline{\mathbb{Q}}_p$ with the
$\alpha_j$ algebraic integers ($p$-adic Weil numbers). Reducing modulo the maximal ideal,
$L(X,\chi)\bmod p=\prod_j(1-\bar\alpha_j X)$, and a factor equals $1-X$ exactly when
$\bar\alpha_j=1$, i.e. $v_p(\alpha_j-1)>0$. Therefore
$\operatorname{ord}_{X=1}(L(X,\chi)\bmod p)=\#\{j:v_p(\alpha_j-1)>0\}=\deg_X g(X,\chi)$, and dividing
$D_\chi$ by $(1-X)$ subtracts one from the order at $X=1$.
