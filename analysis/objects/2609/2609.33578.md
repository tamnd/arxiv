\label{thm:main}
		Let $A > 0$. 
		For any  $\varepsilon > 0$, there exist constants $C=C(A) > 1,$ $ \delta = \delta(A,  \epsilon)>0$ such that the following stability holds: for any  $\omega_{0} \in (L^{1} \cap L^{\infty}) (\mathbb R^2_+)$ satisfying \begin{equation}\label{eq:mainthm-assumptions}
			\begin{split}
				\Vert\omega_{0} -  {\omega}_{Lamb}\Vert_{(L^{2} \cap L^{1}_{*})(\mathbb{R}^{2}_{+})} < \delta,\quad 
				                                              
				\Vert\omega_{0} \Vert_{ L^{\infty}(\mathbb{R}^{2}_{+})} \le A ,\quad\mbox{and}
			\end{split}
		\end{equation} 
		\begin{equation}\label{assump_moment}
			\int_{\{x\in\mathbb{R}^2_+\,:\,|x|>1\}}|x||\omega_0(x)|dx<\delta,
		\end{equation}
		the unique solution $\omega(t,\cdot)$ to \eqref{eq:2D-Euler} for the initial data $\omega_0$ satisfies \begin{equation}\label{eq:mainthm-conclusion}
			\begin{split}
				\Vert\omega(t,\cdot +p_{1}(t) {\bf e}_{1} ) -  {\omega}_{Lamb}( \cdot )\Vert_{(L^{2}\cap L^1)(\mathbb{R}^{2}_{+}) \cap L^{1}_{*}{( \{ |x_{1} | \le C
						\varepsilon^{-1} \}  )}} < \varepsilon,\quad t\in\mathbb{R},
			\end{split}
		\end{equation} for some Lipschitz function  $p_{1}:\mathbb{R}\to\mathbb{R}$  satisfying $|p_{1}(t) - t| \le \varepsilon^{1/2}(1+|t|)$ for all $t \in \mathbb R$.

<!-- remark-1: body_md -->
While Theorem \ref{thm:main} is stated for the normalized Lamb dipole for simplicity, a similar statement holds (by a suitable rescaling of $C$ and $\delta$) for the whole two-parameter family of Lamb dipoles, denoted by $\omega_{Lamb}^{\kappa,\mu}$: 
		\begin{equation}\label{eq:lamb-rescale-def}
			\omega_{Lamb}^{\kappa,\mu}(x) := \left(\frac{\kappa^{3}}{ \sqrt{\pi} c_{L}^{3} \mu}\right)^{1/2} \, {\omega}_{Lamb}\left( \left( \frac{c_{L}\mu}{\sqrt{\pi}\kappa} \right)^{-1/2} x\right) .
		\end{equation}
		The rescaling is defined so that we have\footnote{With this notation,  $\omega_{Lamb} = \omega_{Lamb}^{c_{L}\sqrt{\pi}, \pi}$.} $\Vert\omega_{Lamb}^{\kappa,\mu}\Vert_{L^2(\mathbb{R}^{2}_{+})} = \kappa$ and $\Vert\omega_{Lamb}^{\kappa,\mu}\Vert_{L^1_*(\mathbb{R}^{2}_{+})} = \mu$.

<!-- proposition-1: body_md -->
\label{prop:en_eq}
		For any  $\rho\in (L^2\cap L^1_{*})(\mathbb{R}^2_+)$, we have the sharp energy inequality
		\begin{align} \label{eq: SEI}
			       
			E[\rho]
			                
			\leq C_L \Vert\rho\Vert_{L^{2}(\mathbb{R}^{2}_{+})}\Vert\rho\Vert_{L^{1}_{*}(\mathbb{R}^{2}_{+})}, \quad \mbox{where} \quad C_L:=(c_L\sqrt{\pi})^{-1} \simeq 0.1502.
		\end{align}    
		If $\rho$ is nonzero, the equality is achieved only when either $\rho$ or $-\rho$ is equal to some $x_{1}$-translate of the two-parameter family of Lamb dipoles defined in \eqref{eq:lamb-rescale-def}.
		
		Under $\omega \ge 0$, we have $\Vert\omega\Vert_{L^1_*} = \mu[\omega]$, and the above implies that in the admissible class \begin{equation*}
			\begin{split}
				\mathcal A_{+} := \left\{  \omega \ge 0 \, : \, \kappa[\omega] \le  \kappa[{\omega}_{Lamb}], \, \mu[\omega] \le  \mu[{\omega}_{Lamb}] \right\},
			\end{split}
		\end{equation*} the supremum of $E$ is achieved precisely for $x_{1}$-translations of $\omega_{Lamb}$.

<!-- proof-2: body_md -->
Since $u_{1} = \partial_{x_2}\psi$ and $\psi = 0$ on $\{ x_{2} = 0 \}$, we have from Hardy's inequality that \begin{equation*}
		\begin{split}
			\Vertx_{2}^{-1}\psi\Vert_{L^{\infty}} \le  \Vertu_1\Vert_{L^{\infty}} .
		\end{split}
	\end{equation*} It hence suffices to show $\Vertu\Vert_{L^{\infty}} \le C_{\alpha}\Vert\omega\Vert_{L^{2}}^{1-\alpha}  \Vert\omega\Vert_{L^{\infty}}^{\alpha/2} \Vert \omega\Vert_{L^{1}}^{\alpha/2}$. This is well-known in the case $\alpha = 1$. To prove it for general $\alpha \in (0,1)$, we use $|{\bf K}(x,y)| \le C|x-y|^{-1}$ (which follows from \eqref{eq:K-decay} and $y_{2} \le |x-\bar{y}|$) and for some $0<R_1 \le R_2$ to be determined, write \begin{equation*}
		\begin{split}
			|u(x)| \lesssim \left[ \int_{ B_{x}(R_{1}) } +  \int_{ B_{x}(R_{2}) \backslash B_{x}(R_{1}) }  + \int_{ \mathbb R^2_+ \backslash B_{x}(R_{2}) }     \right]  \frac{|\omega(y)|}{|x-y|} \, dy. 
		\end{split}
	\end{equation*} Using the $L^\infty$, $L^2$, and $L^1$ norms of the vorticity in each of the regions, we bound \begin{equation*}
		\begin{split}
			|u(x)| \lesssim R_{1}\Vert\omega\Vert_{L^\infty} + \sqrt{\log\left( R_{1}^{-1}R_{2} \right) }\Vert\omega\Vert_{L^2} + \frac{1}{R_{2}}\Vert\omega\Vert_{L^{1}} 
			\lesssim_{\beta} R_{1}\Vert\omega\Vert_{L^\infty} + \left( \frac{R_{2}}{R_{1}} \right)^{\beta} \Vert\omega\Vert_{L^2} + \frac{1}{R_{2}}\Vert\omega\Vert_{L^{1}} 
		\end{split}
	\end{equation*} for any $\beta>0$. Given $\beta$, we select $0<R_1\le R_2$ to satisfy \begin{equation*}
		\begin{split}
			\Vert\omega\Vert_{L^1} = R_{1}R_{2} \Vert\omega\Vert_{L^\infty}, \qquad \left( \frac{R_1}{R_2} \right)^{1+2\beta} \Vert\omega\Vert_{L^1}  \Vert\omega\Vert_{L^\infty} = \Vert\omega\Vert_{L^2}^{2}. 
		\end{split}
	\end{equation*} This is possible since we always have $\Vert\omega\Vert_{L^1}  \Vert\omega\Vert_{L^\infty} \ge \Vert\omega\Vert_{L^2}^{2}$. Solving this system of equations for $R_{1}, R_{2}$ and plugging them back into the previous bound for $|u(x)|$, we obtain \begin{equation*}
		\begin{split}
			|u(x)| \lesssim_{\beta} \left(\Vert\omega\Vert_{L^1}  \Vert\omega\Vert_{L^\infty} \right)^{ \frac12 \frac{2\beta}{1+2\beta} } \Vert\omega\Vert_{L^2}^{\frac{1}{1+2\beta}}
		\end{split}
	\end{equation*} which finishes the proof.

By recalling $\Vert\omega_{0}^{(+)} - \omega_{Lamb} \Vert_{L^2 \cap L^1_{*}} < 4\delta$, we simply assume $\delta>0$ small (if necessary) to have 
$4\delta\leq \varepsilon/100$, which is equivalent to make $C_4$ large enough in \eqref{small_dlt}.
Then by continuity of $\Vert\omega^{(+)}(t,\cdot) - \omega_{Lamb} \Vert_{L^2 \cap L^1_{*}} $ in $t$, 
                                                                                                                                                                                                                          
there exists $T_{0} = T_{0}(\omega_{0},\varepsilon) > 0$ such that \begin{equation*}

		\quad t \in [0,T_{0}].

By \eqref{L2_to_L1}, we get for $t \in [0,T_{0}]$ 
\begin{equation}\label{eq:near-initial-closeness_}
	\Vert\omega^{(+)}(t,\cdot) - \omega_{Lamb} \Vert_{L^1}

We first check \eqref{eq:B1}; note from the definition \eqref{eq:h} of $h(t,p)$ that \begin{equation*}
	\begin{split}
		(\partial_t h)(t,p) = -\int \nabla\cdot( u \omega^{(+)} )(t,x) g(x_1 - p) dx = \int (u_{1}\omega^{(+)})(t,x) g'(x_1 - p) dx , 
	\end{split}
\end{equation*} which shows that $\partial_t h$ is a continuous function of $t$ and $p$. Moreover, \begin{equation*}
	\begin{split}
		(\partial_p h)(t,p) = - \int \omega^{(+)}(t,x) g'(x_{1}-p) dx 
	\end{split}
\end{equation*} is also continuous in $t$ and $p$, which implies that $h \in C^{1}_{t,p}$. 

Assuming that $|p| \le 2$, we observe that 
\begin{equation*}
	\begin{split}
		h(t,p) &=  \int ( \omega^{(+)}(t,x) - \omega_{Lamb}(x)) g(x_{1}-p) dx  + \int \omega_{Lamb}(x) g(x_{1}-p) dx \\ 
		& = 	 \int ( \omega^{(+)}(t,x) - \omega_{Lamb}(x)) g(x_{1}-p) dx - p\int \omega_{Lamb}(x) dx 
	\end{split}
\end{equation*}and 
\begin{equation*}
	\begin{split}
		(\partial_{p}h)(t,p) &=  -\int ( \omega^{(+)}(t,x)-  \omega_{Lamb}(x)) g'(x_{1}-p) dx  - \int \omega_{Lamb}(x) g'(x_{1}-p) dx \\ 
		& = 	- \int ( \omega^{(+)}(t,x) - \omega_{Lamb}(x)) g'(x_{1}-p) dx  -	\int \omega_{Lamb}(x) dx . 
	\end{split}
\end{equation*} Both of these identities follow from that $g(x_{1}-p) = (x_{1}-p)$ and $g'(x_{1}-p) = 1$ on the support of $\omega_{Lamb}$ when $|p|\le2$. By recalling \eqref{defn_m_L},
                                                            
                                                          
we estimate, from \eqref{eq:near-initial-closeness_}, 
for $|p|\leq 2$,
\begin{equation*}
	\begin{split}
		|h(t,p)+p m_L| &\leq 	\|g\|_{L^\infty}\cdot
		\Vert\omega^{(+)}(t,\cdot) - \omega_{Lamb} \Vert_{L^1}\leq 4\cdot \frac{\varepsilon}{10}\leq{\varepsilon}.
	\end{split}
\end{equation*} In particular, we have 
$ \pm h(t, \pm 2) < 0$ for $t \in [0,T_{0}]$.	
Thus, on this time interval,  there exists a position $p_1(t)\in[-2,2]$ satisfying
\eqref{eq:p1-def}. 
Moreover at the origin,
we know \begin{equation}\label{eq:est_h_init}
	|h(t,0)|\leq \varepsilon \quad\mbox{and} \quad  |h(0,0)| \leq 	\|g\|_{L^\infty} 
	\Vert\omega^{(+)}(0,\cdot) - \omega_{Lamb} \Vert_{L^1}\leq 4 \times  {3\delta}=12\delta.
\end{equation}  
Next, 
we estimate similarly
\begin{equation}\label{eq:est_h_P_init_}
	\begin{split}
		|(\partial_p h)(t,p)+m_L| &\leq 	\|g'\|_{L^\infty} 
		\Vert\omega^{(+)}(t,\cdot) - \omega_{Lamb} \Vert_{L^1}\leq 2 \times  \frac{\varepsilon}{10}\leq{\varepsilon},
	\end{split}
\end{equation} 
which yields
\begin{equation}\label{eq:est_h_P_init}
	(\partial_p h)(t,p)\leq -m_L/2.
\end{equation}
The estimates
\eqref{eq:est_h_init} and \eqref{eq:est_h_P_init} implies that
there exists $$p_1(0)\in[-\frac{2}{m_L}  12 \delta,+\frac{2}{m_L}  12 \delta]\subset[-5\delta, 5\delta]
                          
$$ satisfying \eqref{eq:p1-def}.
Now we have
                       
$(\partial_{p}h)(0,p_{1}(0)) \ne 0$. By the Implicit Function Theorem, by shrinking $T_{0}>0$ if necessary, we have that $p_{1}(\cdot_t) \in C^{1}([0,T_{0}])$, with  \begin{equation}\label{eq:deriv-p}
	\dot{p}_{1}(t) = \frac{(\partial_t  h)(t, p_1(t))}{(-\partial_p h)(t, p_1(t))}.
\end{equation} 
From now on we estimate	$|\dot{p}_{1}(0) - 1 |$.
First we compute 
\begin{equation}\label{eq:deriv-t-H_}
	\begin{split}
		(\partial_t  h)(t, p)&=\int_{\mathbb{R}^2_{+}} u_1(t, x+p{\bf e}_1)\omega^{(+)} (t, x+p{\bf e}_1) g'(x_1) dx\\
		&=\int_{\mathbb{R}^2_{+}} u_{Lamb, 1}(  x )\omega_{Lamb} (  x ) g'(x_1) dx\\
		&\quad+\int_{\mathbb{R}^2_{+}} [u_1(t, x+p{\bf e}_1)-u_{Lamb, 1}(  x )] \omega^{(+)} (t, x+p{\bf e}_1) g'(x_1) dx\\
		&\quad+\int_{\mathbb{R}^2_{+}} u_{Lamb, 1}(  x ) [\omega^{(+)}(t, x+p{\bf e}_1) -\omega_{Lamb} ( x ) ]g'(x_1) dx.
	\end{split}
\end{equation}
For the first term on the right hand side of \eqref{eq:deriv-t-H_}, we use the fact that $\omega_{Lamb}$ is a traveling wave solution with propagation speed $1$, which  yields the equation $$(u_{Lamb}-{\bf e}_1)\cdot \nabla \omega_{Lamb}=0.$$ Testing this equation by $g(x_1)$ and integrating by parts yields 
\begin{equation}\label{ml_integral}
	\int_{\mathbb{R}^2_{+}} u_{Lamb, 1}(  x )\omega_{Lamb} (  x ) g'(x_1) dx=\int_{\mathbb{R}^2_{+}} \omega_{Lamb} (  x ) g'(x_1) dx=\int_{\mathbb{R}^2_{+}} \omega_{Lamb} (  x )  dx=m_L.
\end{equation}
Thus we have
\begin{equation*} 
	\begin{split}
		|(\partial_t  h)(t, p)-m_L| & \leq
		2 \int_{\mathbb{R}^2_{+}}\left| [u_1(t, x+p{\bf e}_1)-u_{Lamb, 1}(  x )] \omega^{(+)} (t, x+p{\bf e}_1) \right| dx\\
		&\quad+2\int_{\mathbb{R}^2_{+}} \left|u_{Lamb, 1}(  x ) [\omega^{(+)}(t, x+p{\bf e}_1) -\omega_{Lamb} ( x ) ]\right| dx\\
		& \leq
		C \|u_1(t, x+p{\bf e}_1)-u_{Lamb, 1}(  x )\|_{L^\infty}  + 
		C\|\omega^{(+)}(t, x+p{\bf e}_1) -\omega_{Lamb} ( x ) \|_{L^1}
	\end{split}
\end{equation*}
At the initial time, we have 
\begin{equation*}                      
	\begin{split}
		&\|\omega^{(+)}(0, x+p_1(0){\bf e}_1) -\omega_{Lamb} ( x ) \|_{L^2}\leq\|\omega(0, x+p_1(0){\bf e}_1) -\omega_{Lamb} ( x ) \|_{L^2}\\&=\|\omega(0, x) -\omega_{Lamb} ( x-p_1(0){\bf e}_1 ) \|_{L^2}\leq
		\|\omega(0, x) -\omega_{Lamb} ( x ) \|_{L^2}+
		\|\omega_{Lamb} ( x )-\omega_{Lamb} ( x-p_1(0){\bf e}_1 ) \|_{L^2}\\&
		\leq   \delta +C\cdot  {C_{Lip}\cdot |p_1(0)|}
		\leq C  {\delta},
	\end{split}
\end{equation*} where $C_{Lip}$ is the Lipschitz constant \eqref{defn_C_Lip},
and similarly
\begin{equation*}                      
	\begin{split}
		&\|\omega^{(+)}(0, x+p_1(0){\bf e}_1) -\omega_{Lamb} ( x ) \|_{L^1}\leq\|\omega(0, x+p_1(0){\bf e}_1) -\omega_{Lamb} ( x ) \|_{L^1}\\&=\|\omega(0, x) -\omega_{Lamb} ( x-p_1(0){\bf e}_1 ) \|_{L^1}\leq
		\|\omega(0, x) -\omega_{Lamb} ( x ) \|_{L^1}+
		\|\omega_{Lamb} ( x )-\omega_{Lamb} ( x-p_1(0){\bf e}_1 ) \|_{L^1}\\&\leq 3\delta+C\cdot C_{Lip}\cdot |p_1(0)|\leq C\delta.
	\end{split}
\end{equation*} The above estimates controls
the initial velocity maximum by \eqref{eq:vel-L-infty-alpha2} with $\alpha=1/2$:
\begin{equation*}                      
	\begin{split}
		&	\|u_1(0, x+p_1(0){\bf e}_1)-u_{Lamb, 1}(  x )\|_{L^\infty} \\
		&\quad \leq C\|\omega(0, x+p_1(0){\bf e}_1) -\omega_{Lamb} ( x ) \|^{1/2}_{L^2}  \cdot A^{1/4}\cdot\|\omega(0, x+p_1(0){\bf e}_1) -\omega_{Lamb} ( x ) \|^{1/4}_{L^1} \\
		&\quad \leq C\delta^{1/2}\cdot A^{1/4} \cdot \delta^{1/4}= CA^{1/4} \cdot\delta^{3/4}.
	\end{split}
\end{equation*}
In sum,  we 
get the following result:
$$|(\partial_t  h)(0, p_1(0))-m_L|\leq CA^{1/4} \cdot\delta^{3/4}+C\delta\leq CA^{1/4} \delta^{3/4}. $$
Recalling 
\eqref{eq:deriv-p} and
combining this bound and \eqref{eq:est_h_P_init_} gives 
\begin{equation*}

\end{equation*} 
by taking $\varepsilon$ smaller if necessary, which is equivalent to make $C_4$ large enough in \eqref{small_varep}. This means that the change of variables \eqref{eq:rho-def} is well-defined.

		&\le \frac14 \Vert\rho_Q^{err}\Vert_{L^1_*} + C_L\Vert\rho_Q^{main}\Vert_{L^1_*}\Vert\rho_Q^{main}\Vert_{L^2}  \\
		&\leq \frac14 \Vert\rho_Q^{err}\Vert_{L^1_*} +  C_L(\mu[\omega_{Lamb}] + \lambda-\Vert\rho_Q^{err}\Vert_{L^1_*})(\kappa[\omega_{Lamb}] + \lambda)\\
		&\leq E[\omega_{Lamb}]-  \frac34 \Vert\rho_Q^{err}\Vert_{L^1_*}
		+ {20\lambda},			
		                                        
	\end{split}
\end{equation} where we used the identity $E[\omega_{Lamb}]=C_L \kappa[\omega_{Lamb}] \mu[\omega_{Lamb}]$ {and} $C_L \kappa[\omega_{Lamb}]=1.$
Then combining this with \eqref{eq:B5} yields 
$$E[\omega_{Lamb}]-\lambda\leq E_Q\leq E[\omega_{Lamb}]-  \frac34 \Vert\rho_Q^{err}\Vert_{L^1_*}+ {20\lambda},	                                 
$$ 
which gives the impulse bound $$\Vert\rho_Q^{err}\Vert_{L^1_*}\leq 30 \lambda.$$
Let us denote 
$\kappa_Q^{main}	=\Vert\rho_Q^{main}\Vert_{L^2}$ and 
$\kappa_Q^{err}	=\Vert\rho_Q^{err}\Vert_{L^2}$.
Plugging this back into the first inequality of \eqref{eq: decom-energy} and combing \eqref{eq:B5} again, we obtain
$$C_L\mu[\omega_{Lamb}]\kappa[\omega_{Lamb}]-\lambda\leq E_Q\leq  8 \lambda  +C_L(\mu[\omega_{Lamb}] + \lambda)\kappa_Q^{main}\leq C_L\mu[\omega_{Lamb}]\kappa_Q^{main}+10\lambda.$$ 
This gives $$\kappa_Q^{main}\geq \kappa[\omega_{Lamb}]-\frac{11\lambda}{C_L \mu[\omega_{Lamb}] }\geq \kappa[\omega_{Lamb}]-55\lambda.$$
So we have $$(\kappa[\omega_{Lamb}]+\lambda)^2-( \kappa_Q^{err})^2\geq  \kappa_{Q}^2-( \kappa_Q^{err})^2=(\kappa_Q^{main})^2\geq (\kappa[\omega_{Lamb}]-55\lambda)^2,$$
which immediately yields
$\kappa_Q^{err}\leq 30\lambda^{1/2}.$ Then the $L^1$-norm estimate for $\rho_Q^{err}$ follow by a similar argument with 
\eqref{est:L2toL1}. The velocity estimate follows from \eqref{eq:vel-L-infty-alpha2} with $\alpha=1/2$.

Let $z$ and $t_0$ satisfy the assumption.
The proof is divided into four steps.

\medskip \noindent \textbf{Claim 1}. We have $
\Phi_1(t,z)\leq R$  for any $t\in[t_0,T].$
                                                                                                 

\medskip \noindent 	This is immediate from the fact $v_1<-1/2$ on the line $\{x_1=R\}$.

\medskip \noindent \textbf{Claim 2}. As long as the trajectory stays in $Q_{R}$, its height is bounded by $4R$ and $H/6$. Precisely, $$
\Phi_2(t,z)\leq 4R<H/6\quad\mbox{for any}\quad t\in[t_0,T] \quad \mbox{satisfying }\quad \Phi(t,z)\in Q_R.
$$	
                                                                                                                                                      


\medskip \noindent 	 The inequality $4R<H/6$ holds once we choose $C_2$ large in \eqref{large_H}.  For the first inequality, for a contradiction, we assume that there exists $t_1\in (t_0, T]$ such that $\Phi_2(t_1,z)> 4R$ and $\Phi_1(t_1,z)\in [-R, R]$. We take $t_2\in (t_0, t_1)$ to be largest time before $t_{1}$ such that $\Phi_2(t_2,z)=2R$. The velocity satisfies $v_1<-1/2$ on the line $\{x_1=-R\}$, so for all $t\in (t_2, t_1)$, we have $$ \Phi_1(t,z)\in [-R, R]\quad\mbox{ and }\quad \Phi_2(t,z)>2R.$$ On the other hand, by using \eqref{eq:borderline-v}, for any $y\in [-R, R]\times (2R, +\infty)$, we have \begin{equation}\label{eq:v-ests}
	\begin{split}
		v_1(y)\leq -1/2 \quad \text{ and }\quad |v_2(y)| \leq 1/4.
	\end{split}
\end{equation} 
Therefore, we get $t_1-t_2 \geq 4 (\Phi_2(t_1,z)-\Phi_2(t_2,z))\geq 8R.$ Hence $$-R\leq \Phi_1(t_1,z) \leq \Phi_1(t_2,z)-\frac12 \times 8R\leq -3R <-R,$$ which is a contradiction. This proves \textbf{Claim 2}.


\medskip \noindent \textbf{Claim 3}. As long as the trajectory stays in $Q_{I}$, its height is bounded by $H/4$. Precisely,
$$
\Phi_2(t,z)< H/4\quad\mbox{for any}\quad t\in[t_0,T] \quad \mbox{satisfying }\quad \Phi(t,z)\in Q_I.
$$		 
\medskip \noindent  To derive a contradiction, we  assume that there exists the smallest $t_3\in (t_0, T)$ such that $$\Phi(t_3,z)\in Q_I \quad\mbox{and}\quad \Phi_2(t_3,z)=H/4.$$
Recall that  $6R<H/4$.
Let $t_4\in (t_0, t_3)$ be the largest time before $t_{3}$ such that $\Phi_2(t_4,z)=5R$. Then we must have $$\frac{H}{4}\geq\Phi_2(t,z)\geq 5R \quad \mbox{ and }\quad \Phi(t,z)\in Q_I \quad \mbox { for all }\quad t\in [t_4,t_3],$$ since the velocity still satisfies \eqref{eq:v-ests} on $Q_I$ and in particular any particle will never come back to $Q_{I}$ once it leaves it. So we have $$0<t_3-t_4\leq 2 C_0\lambda^{-1/2}.$$

$$\frac{d}{dt} \Phi_2(t,z)\leq  \frac{80 (A \Phi_2(t,z))^{1/2}+400}{ \Phi_2(t,z)}	{\lambda^{1/2}}
+\lambda^{3/4} 
                                                                                  
$$
Note that $ \Phi_2(t,z)-  \lambda^{3/4}(t-t_4)\geq 5R-R=4R$ for any $t\in [t_4, t_3]$. We deduce from the above inequality that 
$$\frac{d}{dt} (\Phi_2(t,z)-\lambda^{3/4}(t-t_4)) \leq  \frac{80 (A H/4)^{1/2}+400}{\Phi_2(t,x)-\lambda^{3/4}(t-t_4)}\lambda^{1/2}.$$
This gives 
$$
\left(\Phi_2(t_3,z)-\lambda^{3/4}(t_3-t_4)\right)^2
\leq (5R)^2+4C_0 [80 (A H/4)^{1/2}+400],
$$	which implies
$$\Phi_2(t_3,z)\leq \left(
(5R)^2+4C_0 [80 (A H/4)^{1/2}+400]
\right)^{1/2}+R.
                                                                         
                                    
$$
Then, by taking a larger constant $C_2$ in \eqref{large_H} if necessary, we can guarantee that $H$ is large enough so that the right-hand side of the above is smaller than $H/8$. This contradicts $\Phi_2(t_3,z)=H/4$.

\medskip \noindent \textbf{Claim 4}. Lastly, we have 
$$
\Phi_2(t,z)< H/2\quad\mbox{for any}\quad t\in[t_0,T] \quad \mbox{satisfying }\quad \Phi(t,z)\in Q_{II}.
$$		
                                                                                                                  

\medskip\noindent 
                                        
                           
Consider any $t_5\in (t_0, T]$ such that $\Phi(t_5,z)\in Q_{II}$. Let $t_6\in (t_0, t_5)$ be the largest time before $t_{5}$ such that $ \Phi(t_6,z)\in \{x_2=2(x_1+D)\}$. That is, $\Phi_2(t_6,z)=2(\Phi_1(t_6,z)+D)$. Then, \textbf{Claims 2 and 3} give $\Phi_2(t_6,z)\leq H/4$ and hence $$0<\Phi_1(t_6,z)-\Phi_1(t_5,z)\leq\Phi_1(t_6,z)+D=\frac12 \Phi_2(t_6,z)\leq \frac{H}{8}.$$
This implies $t_5-t_6\leq H/4$. Then, using the first inequality of \eqref{eq:vel-Q-decomp}, we obtain \begin{equation*}
	\begin{split}

	\end{split}
\end{equation*} which establishes \textbf{Claim 4}, which also finishes the proof.

		& \le  {C}A^{1/2} \lambda^{1/4}\left(G^{main}(t)+G^{err}(t)\right),
	\end{split}
\end{equation*}  where 
$$
G^{\star}(t):=\int_{\mathbb R^2_+ } |\widetilde{\rho}(t,x)|\left( \int_{\mathbb R^2_+ }\frac{y_{2}}{ |x-y||x-\bar{y}|}  \, |\rho^{\star}_{Q}(t,y)| \, dy \right) dx,\quad \star\in\{main, err\}.
$$
Again, we follow the same idea we used to estimate $F_\mu^{main}$ and $F_\mu^{err}$ in the proof of Lemma \ref{lem:F-mu}		
                                                                                           
to get 
\begin{equation*}                 
	\begin{split}
		\int_{0}^{T} \left| \widetilde{F}_{E, 1}(t)  \right|  dt \le {C}A^{1/2} \lambda^{1/4}\cdot\left(\frac{\lambda}{20}+\frac{\lambda}{20}\right)\leq \frac{\lambda}{100}.
	\end{split}
\end{equation*}
The proof is complete.
