<!-- lemma-1: body_md -->
\label{nonhomo}
If $\eta \in C_{2\nu-1}$ has finite positive support, then $\eta \in \mathrm{im}\,\partial$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We write
\[
\eta=\sum_{\ell \in \mathbb{Z}} \eta_\ell u^+_{\nu-\ell-1,\ell}.
\]
We can assume without loss of generality that $\eta \neq 0$.
Hence, since $\eta$ has finite positive support, there exists $L \in \mathbb{Z}$ such that
\[
L=\max\{\ell\mid \eta_\ell\neq 0\}.
\]
We need to construct
\[
\xi=\sum_{\ell \in \mathbb{Z}} \xi_\ell u^-_{\nu-\ell,\ell} \in C_{2\nu}
\]
such that
$\partial \xi=\eta$.
We set
$\xi_\ell=0$, $\ell > L$.
For $\ell \leq L$, we define downward recursively
\begin{equation}\label{rec}
\xi_\ell=\eta_\ell-(\nu-\ell-1)\xi_{\ell+1}.
\end{equation}
Since both $\eta_\ell$ and $\xi_\ell$ vanish for $\ell>L$, the formula
(\ref{rec}) actually holds true for all $\ell \in \mathbb{Z}$.
Using~(\ref{parti}) and~(\ref{rec}), we compute
\begin{align*}
\partial \xi&= \sum_{\ell \in \mathbb{Z}}\xi_\ell
\partial u^-_{\nu-\ell,\ell}
 = \sum_{\ell \in \mathbb{Z}}\xi_\ell\bigl(u^+_{\nu-\ell-1,\ell}+(\nu-\ell)u^+_{\nu-\ell,\ell-1}\bigr) \\
&= \sum_{\ell \in \mathbb{Z}}\bigl(\xi_\ell+\xi_{\ell+1}(\nu-\ell-1)\bigr)u^+_{\nu-\ell-1,\ell}
 = \sum_{\ell \in \mathbb{Z}}\eta_\ell u^+_{\nu-\ell-1,\ell}
 = \eta.
\end{align*}
This proves the lemma.

<!-- lemma-2: body_md -->
\label{homo}
Assume that $\eta \in C_{2\nu-1}$ has infinite positive support but is positively bounded.
Then $\eta \notin \mathrm{im}\,\partial$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Arguing by contradiction, suppose that
\[
\eta=\sum_{\ell \in \mathbb{Z}} \eta_\ell u^+_{\nu-\ell-1,\ell}
\]
admits a primitive, i.e., there exists
\[
\xi=\sum_{\ell \in \mathbb{Z}} \xi_\ell u^-_{\nu-\ell,\ell} \in C_{2\nu}
\]
such that
$\partial \xi=\eta$.
The coefficients of $\xi$ have to satisfy the recursion relation~(\ref{rec}).
Since $\eta$ is positively bounded, there exists $N \in \mathbb{N}$ such that
$|\eta_\ell| \leq N$, $\ell \geq \nu$.
Hence we obtain from~\eqref{rec} the estimate
\begin{equation*}              
|\xi_{\ell+1}| \leq \frac{|\xi_\ell|+N}{\ell+1-\nu}, \qquad \ell \geq \nu.
\end{equation*}
We claim that this estimate implies $\lim_{\ell\to\infty}\xi_\ell=0$.
To see this, define $x_k:=|\xi_{k+\nu}|$ for $k\geq 0$. Then $x_{k+1}\leq\frac{x_k+N}{k+1}$, which by induction implies
\[
  x_k\leq\frac{x_0}{k!}+Nc_k,\qquad c_k:=\frac{1}{k}+\frac{1}{k(k-1)}+\cdots+\frac{1}{k!}.
\]
Since $c_k$ is a sum of $k$ terms each $\leq 1/k$, we have $c_k\leq 1$ for all $k$. This implies $c_k=\frac{1+c_{k-1}}{k}\leq 2/k$, and therefore the claim in view of
\[
  x_k \leq \frac{x_0}{k!} + \frac{2 N}{k}\xrightarrow[k\to\infty]{} 0.
\]
Since the $\xi_\ell$ are integers, the claim implies that there exists an $L$ such that $\xi_\ell=0$ for all $\ell\geq L$. By~\eqref{rec}, this yields $\eta_\ell=0$ for all $\ell\geq L$, contradicting the hypothesis that $\eta$ has infinite positive support.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{noed}
For every $\nu \in \mathbb{Z}$, we have $\ker(\partial\colon C_{2\nu}\to
C_{2\nu-1})=\{0\}$, hence $H_{2\nu}=\{0\}$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Suppose that
\[
\xi=\sum_{\ell \in \mathbb{Z}}\xi_\ell u^-_{\nu-\ell,\ell}
\in C_{2\nu}
\]
satisfies
\[
\partial \xi=0.
\]
We will show that this implies $\xi=0$. Since $\xi$ lies in the
kernel of the boundary operator, we obtain as a special instance of
(\ref{rec}) the following recursion relation between
its coefficients:
\begin{equation}\label{recsimp}
\xi_\ell=(\ell+1-\nu)\xi_{\ell+1}, \qquad \ell \in \mathbb{Z}.
\end{equation}
In particular, plugging $\ell=\nu-1$ into this formula we obtain
$\xi_{\nu-1}=0$,
and therefore by downward induction using (\ref{recsimp})
\begin{equation}\label{van1}
\xi_\ell=0, \qquad \ell \leq \nu-1.
\end{equation}
It remains to show that the coefficients vanish also for $\ell\geq \nu$.
In order to prove this, we first show the following claim.

\begin{Claim}For every $\ell_0 \in \mathbb{Z}$, there exists $\ell \geq \ell_0$
such that $\xi_\ell=0$.
\end{Claim}

Arguing by contradiction, suppose there exists $\ell_0$ such that
\[
\xi_\ell \neq 0, \qquad \ell \geq \ell_0.
\]
Using (\ref{recsimp}) inductively, we obtain
\[
\xi_{\nu+k}=\frac{\xi_\nu}{k!} \xrightarrow[k\to\infty]{} 0.
\]
Since $\xi_{\nu+k}\in\mathbb{Z}\setminus\{0\}$ for all $k\geq 0$,
this is a contradiction proving the claim.

If $\xi_L=0$ for some $L\geq \nu$, then downward induction using (\ref{recsimp}) shows that
$\xi_\ell=0$ for every $\nu \leq \ell \leq L$. Since by the claim we find
$L$ arbitrarily large such that $\xi_L=0$, we deduce that
$\xi_\ell=0$, $\ell \geq \nu$.
Combining this with (\ref{van1}), we see that all coefficients of $\xi$
vanish, so that $\xi=0$. This proves the lemma.

<!-- proposition-1: body_md -->
\label{spec}
The spectral number of each $0\neq\alpha \in H_{2\nu+1}$ satisfies
\[
\sigma(\alpha)= \begin{cases}
\infty, & E<\kappa-\frac{\hbar}{2},\\
\bigl(E+\frac{\hbar}{2}\bigr)\nu, & E=\kappa-\frac{\hbar}{2},\\
-\infty, & E>\kappa-\frac{\hbar}{2}.
\end{cases}
\]
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
Note that
\begin{align}
  \widetilde{\mathfrak{A}}^E_\kappa(u^+_{\nu-\ell,\ell})
  &= \mathfrak{A}^E_\kappa(u^+_{\nu-\ell,\ell}) + \tfrac{(\nu-\ell)\hbar}{2}
  = E(\nu-\ell)+\kappa\ell+\tfrac{(\nu-\ell)\hbar}{2} \nonumber\\
  &= \bigl(E+\tfrac{\hbar}{2}\bigr)\nu+\bigl(\kappa-E-\tfrac{\hbar}{2}\bigr)\ell.\label{act}
\end{align}
We now discuss the three cases one by one.

\textit{Case $E<\kappa-\tfrac{\hbar}{2}$:}
Consider $\xi \in C_{2\nu+1}$ with $\alpha=[\xi]$. Since $\alpha\neq 0$,
by Lemma~\ref{nonhomo} we deduce that $\xi$ has infinite positive support.
Since $\kappa>E+\tfrac{\hbar}{2}$, we infer from (\ref{act}) that
$\sigma(\xi)=\infty$.
Since~$\xi$ was an arbitrary representative of $\alpha$, this implies that
$\sigma(\alpha)=\infty$.

\textit{Case $E=\kappa-\tfrac{\hbar}{2}$:}
In this case we have
$\widetilde{\mathfrak{A}}^E_\kappa(u^+_{\nu-\ell,\ell})=\bigl(E+\frac{\hbar}{2}\bigr)\nu$
independently of $\ell$, which immediately implies that
$\sigma(\alpha)=\bigl(E+\tfrac{\hbar}{2}\bigr)\nu$.

\textit{Case $E>\kappa-\tfrac{\hbar}{2}$:}
Consider
\[
\xi=\sum_{\ell \in \mathbb{Z}}\xi_\ell u^+_{\nu-\ell,\ell} \in C_{2\nu+1}
\]
with $[\xi]=\alpha$.
For $L \in \mathbb{N}$, abbreviate
\[
\xi^L:=\sum_{\ell \geq L}\xi_\ell u^+_{\nu-\ell,\ell}.
\]
Then $\xi-\xi^L$ has finite positive support, so by Lemma~\ref{nonhomo}
we have
$\xi-\xi^L \in \mathrm{im}\,\partial$,
and therefore
$
\bigl[\xi^L\bigr]=[\xi]=\alpha$.
Since $\alpha \neq 0$, the quantity
\[
\ell^L:=\min\{\ell \geq L\mid \xi_\ell \neq 0\}
\]
is well defined. Since $\kappa<E+\tfrac{\hbar}{2}$, the function
\[
\mathbb{Z} \to \mathbb{R}, \qquad \ell \mapsto
  \bigl(E+\tfrac{\hbar}{2}\bigr)\nu+\bigl(\kappa-E-\tfrac{\hbar}{2}\bigr)\ell
\]
is strictly decreasing, so that by (\ref{act}) we have
\[
\sigma\bigl(\xi^L\bigr)=
  \bigl(E+\tfrac{\hbar}{2}\bigr)\nu+\bigl(\kappa-E-\tfrac{\hbar}{2}\bigr)\ell^L \leq
  \bigl(E+\tfrac{\hbar}{2}\bigr)\nu+\bigl(\kappa-E-\tfrac{\hbar}{2}\bigr)L.
\]
Again using $\kappa<E+\tfrac{\hbar}{2}$, we deduce that
\[
\lim_{L \to \infty} \sigma\bigl(\xi^L\bigr)=-\infty.
\]
Since each $\xi^L$ represents the homology class $\alpha$, this implies that
$\sigma(\alpha)=-\infty$.
This proves the proposition.

<!-- proposition-2: body_md -->
\label{torprop1}
For each $\nu \in \mathbb{Z}+\tfrac{m-1}{2}$, the following statements hold.
The group $H_{2\nu}^B$, and therefore also $H_{2\nu}$, is trivial.
If~$\xi \in B_{2\nu+1}$ has finite positive support, then it
lies in the image of $\partial^B$;
if~it has infinite positive support but is positively bounded, then it

<!-- proposition-3: body_md -->
\label{torprop2}
For $0\neq\alpha \in H_{2\nu+1}^B$, the spectral number $\sigma(\alpha)$
is finite if and only if
\[
E=\sum_{i=1}^m \bigl(\kappa_i-\tfrac{\hbar}{2}\bigr) \varpi_i.
\]
<!-- end proposition-3 -->
<!-- proof-7: body_md -->
We first introduce some useful notation. We define
\[
\iota \colon \ \mathbb{Z}+\tfrac{m-1}{2} \to \{1,\dots,m\}
\]
by the condition that
\[
\varepsilon(\nu)=e_{\iota(\nu)}.
\]
Note that in view of (\ref{ant3}), we have
\[
\iota(\nu)=\iota(-\nu), \qquad \nu \in \mathbb{Z}+\tfrac{m+1}{2}.
\]
Recall that an element in $B_{2\nu+1}$ has the form
\[
  \xi = \sum_{\ell \in \mathbb{Z}}\xi_\ell u^+_{\nu-\ell,\beta^+_\nu(\ell)}.
\]
Since $\widetilde{\mathfrak{A}}^E_\kappa\bigl(u^+_{\nu-\ell,\beta^+_\nu(\ell)}\bigr) =
  \mathfrak{A}^E_\kappa\bigl(u^+_{\nu-\ell,\beta^+_\nu(\ell)}\bigr) + \frac{(\nu-\ell)\hbar}{2}$,
its spectral number is given by
\begin{equation}\label{eq:sigma}
  \sigma(\xi) = \sup\bigl\{\Sigma(\ell)+\tfrac{\nu\hbar}{2}\mid \xi_\ell\neq 0\bigr\}
\end{equation}
with the function
\[
\Sigma \colon \ \mathbb{Z} \to \mathbb{R}, \qquad
\ell \mapsto \mathfrak{A}^E_\kappa\bigl(u^+_{\nu-\ell,\beta^+_\nu(\ell)}\bigr)-\tfrac{\ell\hbar}{2}.
\]
For $\ell \in \mathbb{N}$, the definition of $\beta_\nu^+$ yields
\[
  \beta^+_\nu(\ell) =  \sum_{j=1}^\ell\varepsilon(\nu+1-j) =
  \beta^+_{\nu-1}(\ell) + \varepsilon(\nu) - \varepsilon(\nu-\ell).
\]
Using this as well as~\eqref{wirk}, \eqref{ant3}, \eqref{ant1}
and~\eqref{varsigma}, we
compute for $\ell \in \mathbb{N}$,
\begin{align*}               
\Sigma(\ell)
&= \langle \kappa,\beta^+_\nu(\ell)\rangle+
E\Omega_{\nu-\ell}-\tfrac{\ell\hbar}{2}\\ \nonumber
&= \bigl\langle \kappa,\beta^+_{\nu-1}(\ell) + \varepsilon(\nu) -
\varepsilon(\nu-\ell)\bigr\rangle + E\Omega_{\nu-\ell}-\tfrac{\ell\hbar}{2}\\ \nonumber
&= \bigl\langle \kappa,\beta^+_{\nu-1}(\ell) + \varepsilon(-\nu) -
\varepsilon(\ell-\nu)\bigr\rangle - E\Omega_{\ell-\nu}-\tfrac{\ell\hbar}{2}\\ \nonumber
&= \kappa_{\iota(-\nu)}-\kappa_{\iota(\ell-\nu)}+
\sum_{i=1}^m \bigl(\kappa_i-\tfrac{\hbar}{2}\bigr) \langle \beta^+_{\nu-1}(\ell),e_i
\rangle-E\Omega_{\ell-\nu}.
\end{align*}
Since we can identify the abelian groups
$H^B_{2\nu+1}$ for different $\nu$ using the
$\mathfrak{u}$-maps, which shift the action by some constant amount, it suffices
to prove the proposition for the special case
\[
\nu=-\tfrac{m-1}{2}.
\]
In this case, for $\ell \in \mathbb{N}$ the preceding computation gives
\begin{equation}\label{sigfun}
\Sigma(\ell)
= \kappa_{\iota (\frac{m-1}{2} )}-\kappa_{\iota (\frac{m-1}{2}+\ell )}+
\sum_{i=1}^m \bigl(\kappa_i-\tfrac{\hbar}{2}\bigr) \big\langle \beta^+_{-\frac{m+1}{2}}(\ell),e_i
\big\rangle-E\Omega_{\frac{m-1}{2}+\ell}.
\end{equation}
For $1\leq i\leq m$, let $\lambda_i$ be the unique positive
integer such that
\[
  \frac{\lambda_i}{\varpi_i} \leq \Omega_{\frac{m-1}{2}+\ell}
  < \frac{\lambda_i+1}{\varpi_i}.
\]
This means that $\lambda_i$ is the number of multiples of
$\tfrac{1}{\varpi_i}$ appearing in the nondecreasing sequence
$\Omega_{\frac{m-1}{2}+1},\dots,\Omega_{\frac{m-1}{2}+\ell}$, hence
\begin{align*}
  \lambda_i
  = \#\bigl\{j\in\{1,\dots,\ell\}\mid \varepsilon\bigl(\tfrac{m-1}{2}+j\bigr) = e_i\bigr\}
  =\sum_{j=1}^\ell \bigl\langle\varepsilon\bigl(\tfrac{m-1}{2}+j\bigr),e_i\bigr\rangle
   = \bigl\langle\beta^+_{-\frac{m+1}{2}}(\ell),e_i\bigr\rangle.
\end{align*}
This yields the inequalities
\begin{equation}\label{mainest}
\bigl\langle\beta^+_{-\frac{m+1}{2}}(\ell),e_i
\bigr\rangle\leq \varpi_i \cdot \Omega_{\frac{m-1}{2}+\ell}
< \bigl\langle\beta^+_{-\frac{m+1}{2}}(\ell),e_i
\bigr\rangle+1, \qquad 1 \leq i \leq m,
\end{equation}
and for $i=\iota(\frac{m-1}{2}+\ell)$ the equality
                                           
\begin{equation*}
  \varpi_{\iota(\frac{m-1}{2}+\ell)}\cdot \Omega_{\frac{m-1}{2}+\ell}
  = \bigl\langle\beta^+_{-\frac{m+1}{2}}(\ell),e_{\iota(\frac{m-1}{2}+\ell)}\bigl\rangle.
\end{equation*}
We further introduce the function
\[
\Gamma \colon \ \mathbb{N} \to \mathbb{R}, \qquad
\ell \mapsto \Sigma(\ell)+\bigg(E-\sum_{i=1}^m \bigl(\kappa_i-\tfrac{\hbar}{2}\bigr)\varpi_i\bigg)
\Omega_{\frac{m-1}{2}+\ell}\,.
\]
Using (\ref{mainest}) and (\ref{sigfun}), we estimate for $\ell \in
\mathbb{N}$ from above,
\begin{align*}              
\Gamma(\ell)&=\Sigma(\ell)+E\Omega_{\frac{m-1}{2}+\ell}-\sum_{i=1}^m\bigl(\kappa_i
-\tfrac{\hbar}{2}\bigr)
\varpi_i \Omega_{\frac{m-1}{2}+\ell}\\
&\leq \Sigma(\ell)+E\Omega_{\frac{m-1}{2}+\ell}-\sum_{i=1}^m\bigl(\kappa_i-\tfrac{\hbar}{2}\bigr)
\bigl\langle \beta^+_{-\frac{m+1}{2}}(\ell),e_i\bigr\rangle
 =\kappa_{\iota(\frac{m-1}{2})}-\kappa_{\iota(\frac{m-1}{2}+\ell)},
\end{align*}
and similarly from below,
\begin{align*}              
\Gamma(\ell)
&\geq \Sigma(\ell)+E\Omega_{\frac{m-1}{2}+\ell}-\sum_{i=1}^m\bigl(\kappa_i-\tfrac{\hbar}{2}\bigr)
\bigl\langle \beta^+_{-\frac{m+1}{2}}(\ell),e_i\bigr\rangle -\sum_{i=1}^m \bigl(\kappa_i
-\tfrac{\hbar}{2}\bigr)\\ \nonumber
&= \kappa_{\iota (\frac{m-1}{2} )}-\kappa_{\iota (\frac{m-1}{2}+\ell )}
-\sum_{i=1}^m \bigl(\kappa_i-\tfrac{\hbar}{2}\bigr).
\end{align*}
So for $\ell\in\mathbb{N}$, we have
\begin{equation}\label{eq:Sigma}
  \Sigma(\ell)=\Gamma(\ell)+\biggl(\sum_{i=1}^m
  \bigl(\kappa_i-\tfrac{\hbar}{2}\bigr)\varpi_i-E\biggr)\Omega_{\frac{m-1}{2}+\ell},
\end{equation}
where the function $\Gamma(\ell)$ is uniformly bounded from above and
below,
and $\Omega_{\frac{m-1}{2}+\ell}$ is nondecreasing in $\ell$ with
\[
\lim_{\ell \to \infty}\Omega_{\frac{m-1}{2}+\ell}=\infty.
\]
Now we argue as in the proof of Proposition~\ref{spec}, considering
$\xi\in B_{2\nu+1}$ with $[\xi]=\alpha\neq 0$ and distinguishing
three cases.

\textit{Case $E<\sum_{i=1}^m\bigl(\kappa_i-\tfrac{\hbar}{2}\bigr)\varpi_i$:}
Since $\xi$ has infinite positive support by Proposition~\ref{torprop1},
we conclude from~\eqref{eq:sigma} and~\eqref{eq:Sigma} that
$\sigma(\xi)=\infty$ and thus $\sigma(\alpha)=\infty$.

\textit{Case $E=\sum_{i=1}^m\bigl(\kappa_i-\tfrac{\hbar}{2}\bigr)\varpi_i$:}
As in the proof of Proposition~\ref{spec}, we have
$\bigl[\xi^L\bigr]=[\xi]=\alpha$ for $\xi^L:=\sum_{\ell \geq L}\xi_\ell u^+_{\nu-\ell,\beta^+_\nu(\ell)}$
and it follows that $\sigma(\alpha)$ is the finite quantity
\[
  \sigma(\alpha) = \lim_{L\to\infty}\sup\bigl\{\Gamma(\ell)+\tfrac{\nu\hbar}{2}\mid \ell\geq
  L,\, \xi_\ell\neq 0\bigr\}.
\]

\textit{Case $E>\sum_{i=1}^m\bigl(\kappa_i-\tfrac{\hbar}{2}\bigr)\varpi_i$:}
As in the proof of Proposition~\ref{spec}, it follows that
\[
  \sigma(\alpha) = \lim_{L\to\infty}\sup\bigl\{\Sigma(\ell)+\tfrac{\nu\hbar}{2}\mid \ell\geq
  L,\,\xi_\ell\neq 0\bigr\} = -\infty.
\]
This proves Proposition~\ref{torprop2}.
