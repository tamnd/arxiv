<!-- theorem-1: body_md -->
\label{sarkfi}
    Let $F \in \mathbb{Z}[x]$ be intersective. For $N\in \mathbb{N}$, let $\alpha_F(N)$ be the maximum size of a subset $A \subset \{1,\dots,N\}$ that do not contain distinct elements $a_1, a_2$ such that $a_1 - a_2 = F(n)$ for some $n \in \mathbb{Z}$. Then $\alpha_F(N) = o(N)$ as $N\rightarrow \infty$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{leliu}
     Let $A\subset G_n$ with $|A|\gg q^n (\log n)^7/n$. Then, there are distinct polynomials $a_1(t),a_2(t) \in A$ such that $a_1(t)-a_2(t)=(b(t))^2$ for some $b(t)\in\mathbb{F}_q[t]$.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{gr}
	Let $F(x) = \alpha_1 x + \alpha_2 x^2 + \cdots + \alpha_\ell\ x^\ell \in{\mathbb{F}_q}[x]$ be a polynomial of degree $\ell$ with constant term zero such that the number of roots of $F$ in ${\mathbb{F}_q}[t]$ is coprime to $q$. Then there is a constant $c=c(q,\ell) \in (0,1)$ such that if $A\subset G_n$ has cardinality greater than $2q^{cn}$, then there are distinct polynomials $a_1(t), a_2(t)\in A$ such that $a_1(t)-a_2(t) = F(b(t))$ for some $b(t)\in {\mathbb{F}_q}[t]$.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{th:ls}
Theorem \ref{gr} remains valid without the condition on the number of roots of $F$, at the cost of adjusting $2 q^{cn}$ to $c' q^{cn}$, for some constant $c' = c'(q, \ell)>0$.
<!-- end theorem-4 -->
<!-- theorem-5: body_md -->
\label{thmA}
	Let $r, k, d$ and $\ell$ be integers satisfying $k>2r^2 (1-1/\ell^2)$. Suppose $c_1,\dots , c_k\in {\mathbb{F}_q}[t]$ have degree at most $d$ and satisfy $\sum_{i=1}^{k}c_i=0$. Let $F(x) \in {\mathbb{F}_q}[t][x]$ be a polynomial of degree $\ell$ with constant term zero. 
 Then there exist constants $c = c(q,r, k ,\ell) \in (0,1)$ and $c'=c'(q,r,k,d,F)>0$ such that the following holds: Any $A\subset G_n$ satisfying $|A|> c' \cdot q^{cn} $ must contain a nontrivial solution to 
	\begin{equation}\label{ee}
		c_1 a_1^r+\dots +c_k a_k^r=F(b)
	\end{equation}
	for some $b\in {\mathbb{F}_q}[t]$.
<!-- end theorem-5 -->
<!-- theorem-6: body_md -->
\label{bi}
	Let, $r, k$ and $d$ be integers satisfying $k\geq 2r^2+1$. Suppose $c_1, \dots, c_k\in {\mathbb{F}_q}[t]$ have degree at most $d$ and $\sum_{i=1}^k c_i=0$. Then there exist constants $c(r,q)\in(0,1)$ and $c'= c'(q,r,k,d) >0$ such that the following holds: Any $A\subset G_n$ with $|A|\geq c' \cdot q^{cn}$ must contain a non-trivial solution to 
	\begin{equation}\label{ee2}
		c_1 a_1^r+\dots +c_k a_k^r=0.
	\end{equation}

<!-- proposition-1: body_md -->
\label{thmB}
Let $m,n,n',d'$ and $d''$ be positive integers. Let $\Psi: (\mathbb{F}_q^n)^k \rightarrow \mathbb{F}_q^{n'}$ be a polynomial map with degree at most $d'$ and $\Phi:\mathbb{F}_q^m \rightarrow \mathbb{F}_q^{n'}$ be a polynomial map with degree at most $d''$.
Let $A \subset \mathbb{F}_q^n$. Assume that
    \begin{enumerate}[label=(\roman*)]
        \item $\Psi^{-1}(0)\cap A^k = \{ (a, ..., a) : a \in A \}$,
        \item $|\Phi^{-1}(0)|$ is coprime to $q$, and
        \item $\Psi(A^k) \cap \operatorname{Im}(\Phi) = \{0\}$.
    \end{enumerate}
    Then,
	\begin{equation} \label{eq:prop}
		|A|\leq k \cdot \inf_{0<x<1}\frac{(1+x+\dots+x^{q-1})^n}{x^{\left(\frac{q-1}{k}\right)\left(n'-\frac{m}{d''}\right)d' }}.
	\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
By the observation made in the introduction, we can assume that $F(x)$ has only one root $x=0$ in ${\mathbb{F}_q}[t]$, by replacing $F(x)$ with $F(t^Mx)$ for some sufficiently large $M$.

We identify $G_n$ with $\mathbb{F}_q^n$, by identifying each polynomial $g=g_0+g_1t+\dots+ g_{n-1}t^{n-1} \in G_n$ with the vector $(g_0,g_1, \dots , g_{n-1})\in \mathbb{F}_q^n$. Then $g^r \in G_{(n-1)r+1}$ corresponds to the vector $(g_0^r, rg_0^{r-1}g_1,\ldots,g_{n-1}^r) \in \mathbb{F}_q^{r(n-1)+1}$. Note that each component of $g^r$ is a polynomial of degree at most $r$ in $g_0,g_1, \dots , g_{n-1}$. 
	
Under this identification, the map 
$
	(a_1,\dots, a_k) \mapsto \sum ^k_{i=1} c_i a_i^r
$
induces a polynomial map 
\begin{align*}
\Psi: (\mathbb{F}_q^n)^k & \rightarrow \mathbb{F}_q^{n'} \\                 
	(a_1,\dots, a_k) & \mapsto (\psi_1 (a_1,\dots, a_k), \ldots, \psi_{n'} (a_1,\dots, a_k) )
\end{align*}	
where $n':=(n-1)r+d+1$ and $\psi_i (a_1,\dots, a_k)$ is the coefficient of $t^{i-1}$ in $\sum ^k_{i=1} c_i a_i^r$. Moreover, the degree of each $\psi_i$ is at most $d':=r$.
	
Suppose 
\[
F(x) = \alpha_1(t) x+ \alpha_2(t) x^2+\cdots +\alpha_{\ell}(t) x^\ell \in {\mathbb{F}_q}[t][x].
\]
Let $u$ be the maximum of $\deg_t \alpha_{i}(t)$ for $i=1, \ldots, \ell$ and $m = \lfloor {\frac{n'-u}{\ell}}\rfloor +1$. If $h= h_0+h_1t+\cdots+h_{m-1}t^{m-1} \in G_m$, then 
	\begin{align*}
		F(h_0+h_1t&+\dots+h_{m-1}t^{m-1})\\
		&= f_0(h_0,\dots, h_{m-1})+f_1(h_0,\dots, h_{m-1})t+\dots+f_{n'-1}(h_0,\dots, h_{m-1})t^{n'-1} \in G_{n'}
	\end{align*}
	where $f_i \in {\mathbb{F}_q}[x_1, \dots,x_m]$ are polynomials with degrees at most $\ell$ and $f_i(0)=0$, for $0 \leq i\leq n'-1$.
	
Consider the polynomial map 
\begin{align*}
\Phi:\mathbb{F}_q^m & \rightarrow \mathbb{F}_q^{n'}, \\
(h_0, \ldots, h_{m-1}) &\mapsto (f_0(h_0,\dots, h_{m-1}), \ldots, f_{n'-1}(h_0,\dots, h_{m-1})). 
\end{align*}
	
Then $\Phi$ has degree at most $d'':=\ell$.  Suppose now $A \subset G_n$ is a subset that does not contain a nontrivial solution to $c_1 a_1 ^r + \cdots + c_k a_k^r = F(b)$.
We apply Proposition \ref{thmB} to $\Psi, \Phi$ and the set $A$. The first and third condition are satisfied by the assumption on $A$. The second condition is satisfied by the assumption on the number of roots in ${\mathbb{F}_q}[t]$ of $F$. 

                                               
                  
                                                                                    
                
    
Substituting $n'= (n-1)r+d+1, d'=r, d''=\ell$ and $m = \lfloor {\frac{n'-u}{\ell}}\rfloor +1$, \eqref{eq:prop} implies that
\[
|A|\leq k \inf_{0<x<1}\frac{(1+x+\dots+x^{q-1})^n}{ x^{\frac{q-1}{k} \left( 1 - \frac{1}{\ell^2} \right) nr^2 + O_{q,r,k,d,F}(1) }}.
\]
Writing $E := (q-1)r^2(1-1/\ell^2)k^{-1}$, then we have $E <(q-1)/2$ since $k>2r^2(1-1/\ell^2)$.

Let $f(x) =(1+x+\dots+x^{q-1})x^{-E}$. Then $f'(1) = \sum_{i=0}^{q-1} (i-E) > 0$, so $f$ is increasing on some interval $(1-\epsilon,1)$. Therefore, $\inf_{0<x<1} f(x) < f(1) = q$. Let $x_0 \in (0,1)$ be such that $f(x_0) <q$, then
  \begin{equation*}
|A|  \leq k \cdot x_0^{O_{q,r,k,d,F}(1)} \cdot (f(x_0))^n \leq c' \cdot q^{cn} 
  \end{equation*}
for some constants $c = c(q,r,k,\ell) \in (0,1)$ and $c'(q,r,k,d,F) >0$, as desired.

<!-- definition-1: body_md -->
\label{srank}
	Let $A$ be a finite set and $\mathbb{F}$ be a field. A function $f: A^k \rightarrow \mathbb{F}$ is said to have \textit{slice rank} $1$ if it can be written as  
	\begin{equation*}
		f(x_1,x_2, \dots,x_k )= g(x_j)h(x_1,\dots, x_{j-1},x_{j+1},\dots, x_k),
	\end{equation*}
	for some $j\in \{1, \ldots, k\}$ and some nonzero functions $g:A\rightarrow \mathbb{F}$ and $h:A^{k-1}\rightarrow\mathbb{F}$.
	
	The \textit{slice rank} of a function $F: A^k \rightarrow \mathbb{F}$ is the minimum integer $r$ such that $F$ can be written as a sum of $r$ slice rank $1$ functions.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{srdiag}
Let $A$ be a finite set.
	Suppose a function $f:A^k\rightarrow \mathbb{F}$ satisfies $f(a_1,\dots, a_k)\neq 0$ if and only if  $a_1=a_2=\dots=a_k$. Then $f$ has slice rank $|A|$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{greenlemma}
	Let $\Phi:\mathbb{F}_q^m \rightarrow \mathbb{F}_q^{n}$ be a polynomial map with degree at most $d$ such that $|\Phi^{-1}(0)|$ is coprime to $q$. Then there exists a polynomial $P\in {\mathbb{F}_q}[x_1,x_2,\dots,x_{n}]$ whose degree in every single variable is at most $(q-1)$ such that 
	\begin{enumerate}[label=(\roman*)]
		\item  $\deg(P) \leq (q-1) \left(n-\frac{m}{d}\right)$,
		\item $P$ is supported on $\operatorname{Im}(\Phi)$ (i.e., $P(x)=0$ if $x\notin \operatorname{Im}(\Phi)$), and
		\item $P(0)\neq 0$.
	\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Applying Lemma \ref{greenlemma} to the polynomial map $\Phi: \mathbb{F}_q^m \rightarrow \mathbb{F}_q^{n'}$, we find that there exists a polynomial $P\in {\mathbb{F}_q}[x_1,x_2,\dots,x_{n'}]$ whose degree in every single variable is at most $(q-1)$ such that 
	\begin{enumerate}[label=(\roman*)]
		\item  $\deg(P) \leq (q-1) \left(n'-\frac{m}{d''}\right)$,
		\item $P$ is supported on $\operatorname{Im}(\Phi)$, and
		\item $P(0)\neq 0$.
	\end{enumerate}
We now consider the function	$f(a_1, \ldots , a_k) := P(\Psi(a_1, \ldots, a_k))$ from $A^k$ to ${\mathbb{F}_q}$.
	
\noindent	\textbf{Claim:} $f(a_1, \ldots , a_k) \neq 0 $ if and only if $a_1= \cdots = a_k = a$ for some $a \in A$.

Indeed, for any $a \in A $, $f(a, \ldots , a)= P(\Psi(a, \ldots, a)) = P(0) \neq 0$. Conversely, suppose the $a_i$'s are not all equal. By the assumption on $\Psi$, we have $\Psi(a_1, \ldots, a_k) \neq 0$. Since $\Psi(A^k)$ and $\operatorname{Im}(\Phi)$ only intersect at $0$, we have that $\Psi(a_1, \ldots, a_k) \notin \operatorname{Im}(\Phi)$. Therefore, $P(\Psi(a_1, \ldots, a_k)) = 0$ since $P$ is supported on $\operatorname{Im}(\Phi)$, and the claim is proved.
	
By Lemma \ref{srdiag}, the slice rank of $f$ is $|A|$. We will now estimate the slice rank of $f$ in a different way. Note that $f$ is a polynomial of degree at most $\text{deg}P\cdot\text{deg}\Psi \leq (q-1)\left(n'-\frac{m}{d''}\right)d'=: D$. For each $i=1, \ldots, k$, let $a_i = (a_{i,1}, \ldots, a_{i,n})$. Then $f(a_1, \ldots, a_k)$ can be written as a sum of monomials of the form
	\begin{equation} \label{eq:slice}
		a_{1,1}^{\alpha_{1,1}}a_{1,2}^{\alpha_{1,2}}\cdots a_{1,n}^{\alpha_{1,n}}a_{2,1}^{\alpha_{2,1}} \cdots a_{2,n}^{\alpha_{2,n}}\cdots a_{k,1}^{\alpha_{k,1}} \cdots a_{k,n}^{\alpha_{k,n}}
	\end{equation}
	where $\sum_{i=1}^{k}\sum_{j=1}^{n}\alpha_{i,j}\leq  D$. Since $a^q=a$ for every $a \in \mathbb{F}_q$, we may further assume that $\alpha_{i,j}\leq q-1$ for every $1 \leq i \leq k$ and $1 \leq j \leq n$.
	
	By the pigeonhole principle, there exists $i_0 \in \{ 1, \ldots, k \}$ such that $\sum_{j=1}^{n}\alpha_{i_0,j}\leq  D/k$. 
	For fixed $i_0$ and $(\alpha_{1},\alpha_{2},\dots, \alpha_{n})$, the sum of all terms in $f$ of the form \eqref{eq:slice} with $(\alpha_{i_0, 1},\alpha_{i_0, 2},\dots, \alpha_{i_0, n})= (\alpha_{1},\alpha_{2},\dots, \alpha_{n})$, is a slice rank 1 function. Hence, the slice rank of $f$ is at most $kN$, where $N$ is the
number of tuples $(\alpha_{1},\alpha_{2},\dots, \alpha_{n})$ such that $\alpha_{j}\leq q-1  $ for all $j \in \{1, \ldots, n\}$ and $\sum_{j=1}^{n}\alpha_{j}\leq  D/k$. It remains to estimate $N$. We have
	\begin{equation*}
		N= \sum_{\substack{u_0,u_1,\dots,u_{q-1}\geq 0\\u_0+u_1+\dots+u_{q-1}=n \\ u_1+2u_2+\dots+ (q-1) u_{q-1} \leq D/k } } \frac{n!}{u_0!u_1! \dots u_{q-1}!}.
	\end{equation*}
	 
Note that	 $N$ equals the sum of all the coefficients of $x^{v}$ with $v \leq D/k$ in the expansion of $(1+x+\dots +x^{q-1})^n$. Let $x \in (0,1)$ be arbitrary. When $v \leq D/k$, we have $x^v \geq x^{D/k}$, so
\begin{equation*}
	 	N \cdot x^{D/k} \leq (1+x+\dots+x^{q-1})^n.
\end{equation*}
	 
	 Therefore, the slice rank of $f$ is
	 \begin{equation*}
	 	|A| \leq k N\leq k\cdot \inf_{0<x<1} \frac{(1+x+\dots+x^{q-1})^n}{x^{\left(\frac{q-1}{k}\right)\left(n'-\frac{m}{d''}\right)d'}},
	 \end{equation*}
as desired.
