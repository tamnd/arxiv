\label{QCLT}
Assume that there exists $p\in(1,\frac32)$ such that
\begin{equation}
\label{mp-m2p}
m_1+m_{2p}<\infty.
\end{equation}
Then, for every $t>0$,
\[
{\rm dist}\left(\frac{F_R(t)}{\sigma_R(t)},Z\right)
\le C_t R^{-\left(1-\frac1p\right)} \qquad \text{for } R\ge1.
\]
Here $C_t>0$ depends on $t$, $Z$ is a standard normal random variable,
$\sigma_R^2(t)=\operatorname{Var}(F_R(t))$, and ${\rm dist}$ denotes any one
of the distances $d_W$, $d_{FM}$, and $d_K$, namely the $1$-Wasserstein,
Fortet--Mourier, and Kolmogorov distances, respectively.

<!-- theorem-2: body_md -->
\label{FCLT}
Under the hypotheses of Theorem~\ref{QCLT}, for every $R>0$, the process
$\{F_R(t)\}_{t\geq0}$ has a c\`adl\`ag modification (denoted also $F_R$).
Moreover, for any $T>0$,
\[
\frac{1}{\sqrt R}F_R(\cdot)\xrightarrow{d}\mathcal{G}(\cdot)
\quad \mbox{in $(D[0,T],J_1)$ as $R\to\infty$},
\]
where $\{\mathcal{G}(t)\}_{t\geq0}$ is a centered continuous Gaussian process with covariance
\[
\mathbb{E}[\mathcal{G}(t)\mathcal{G}(s)]
=\Sigma_{t,s}
:=2m_2\int_0^{t\wedge s}\mathbb{E}|\sigma(u(r,0))|^2\,dr.
\]
The convergence also holds in $(D[0,T],U)$.

<!-- lemma-1: body_md -->
\label{lem:chain-estimates}
Let $\phi:\mathbb{R}\to\mathbb{R}$ be a Lipschitz function with Lipschitz constant
${\rm Lip}(\phi)$. Then, whenever the following differences are well defined,
\begin{equation}
\label{D-chain}
|D_\xi\phi(F)|
\le
{\rm Lip}(\phi)|D_\xi F|,
\end{equation}
and
\begin{equation}
\label{D2-chain}
|D^2_{\xi_1,\xi_2}\phi(F)|
\le
{\rm Lip}(\phi)|D^2_{\xi_1,\xi_2}F|
+
2{\rm Lip}(\phi)
\min\big\{
|D_{\xi_1}F|,
|D_{\xi_2}F|
\big\}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The first estimate follows directly from the Lipschitz property and
\eqref{add-one}. For the second estimate, we have
\begin{align*}
D^2_{\xi_1,\xi_2}\phi(F)
&=
\phi\big(f(N+\delta_{\xi_1}+\delta_{\xi_2})\big)
-
\phi\big(f(N+\delta_{\xi_1})\big)
-
\phi\big(f(N+\delta_{\xi_2})\big)
+
\phi(f(N)) \\
&=
\phi\left(
F+D_{\xi_1}F+D_{\xi_2}F+D^2_{\xi_1,\xi_2}F
\right)
-
\phi\left(F+D_{\xi_1}F\right)
-
\phi\left(F+D_{\xi_2}F\right)
+
\phi(F).
\end{align*}
We use the elementary Lipschitz estimate
\[
|\phi(x+a+b+c)-\phi(x+a)-\phi(x+b)+\phi(x)|
\le
{\rm Lip}(\phi)
\big(
|c|+2\min\{|a|,|b|\}
\big).
\]
Applying this estimate with $x=F$,  $a=D_{\xi_1}F$, $b=D_{\xi_2}F$ and $c=D^2_{\xi_1,\xi_2}F$ gives the desired bound.

<!-- theorem-3: body_md -->
\label{thm-u-pth-moment-nonlinear}
Let $p\in [2,3)$ be such that $m_p<\infty$. Let $u$ be the mild solution of
\eqref{nonlinear}. Then, for any $T>0$,
\begin{equation}
\label{u-pth-moment}
    \sup_{(t,x)\in [0,T]\times \mathbb{R}} \mathbb{E}|u(t,x)|^p <\infty .
\end{equation}
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
Let $(u_n)_{n\geq 0}$ be the sequence of Picard iterations, defined by
$u_0(t,x)=1$ and
\begin{equation}
\label{Picard}
u_{n+1}(t,x)=1+\int_0^t \int_{\mathbb{R}} G_{t-s}(x-y) \sigma (u_n(s,y)) L(ds,dy),
\quad n\geq 0.
\end{equation}
By \cite[Lemma~2.5]{BLW26}, the linear growth of $\sigma$, and
$\int_0^T\int_{\mathbb{R}}H_s^{(p)}(y)\,dy\,ds<\infty$,
\[
\mathbb{E}|u_{n+1}(t,x)|^p
\le
C_{T,p}
+C_{T,p}
\int_0^t\int_{\mathbb{R}}
H_{t-s}^{(p)}(x-y)\mathbb{E}|u_n(s,y)|^p\,dy\,ds .
\]
Iterating this renewal inequality as in \cite[Theorem~2.7]{BLW26}
yields a bound uniform in $n$. Passing to the limit in the Picard sequence
proves \eqref{u-pth-moment}.

\item[(ii)] Let $p\in[2,3)$ satisfy $m_p<\infty$. For any

By induction, $u_n(t,x)\in{\rm dom}(D)$ for every $n\ge0$.
Let $\xi=(r,y,z)$. From the recurrence relation \eqref{Picard} and the Heisenberg commutation principle, it follows that
\begin{equation*}
D_\xi u_{n+1}(t,x)=  zG_{t-r}(x-y)\sigma(u_n(r,y)) + \int_r^t\int_{\mathbb{R}}
G_{t-s_1}(x-y_1)D_\xi\sigma(u_n(s_1,y_1))L(ds_1,dy_1).
\end{equation*}
 By \cite[Lemma~2.5]{BLW26}, \eqref{u-pth-moment}, and
\eqref{D-chain},
\begin{align*}
 \mathbb{E}\left|D_{\xi}u_{n+1}(t,x)\right|^p \le &  2^{p-1}C_{T,p}|z|^pG_{t-r}^p(x-y)\\
 &+2^{p-1}A_{T,p}L_\sigma^p
\int_r^t\int_{\mathbb{R}}

\mathbb{E}\left|D_{\xi}u_n(s_1,y_1)\right|^p
dy_1ds_1.
\end{align*}
The renewal argument in \cite[Theorem~3.1]{BLW26} now gives a bound

\label{key-th-D2u}
If $m_1<\infty$, then
\begin{itemize}
\item[(i)] for every $t>0$ and $x\in\mathbb{R}$, $u(t,x)\in{\rm dom}(D^2)$;
\item[(ii)] if $p\in[2,3)$ and $m_p<\infty$, then for any
$0\le t\le T$, $x\in\mathbb{R}$, and
$\xi_i=(r_i,y_i,z_i)\in{\bf Z}$ with $r_i\in[0,t]$, $i=1,2$,
\begin{equation}
\label{D2u-nonlinear-bound}
\begin{aligned}
\|D^2_{\xi_1,\xi_2}u(t,x)\|_p
&\le
C_{T,p}|z_1z_2|
\begin{cases}
g_{t-r_2}^{(p)}(x-y_2)
g_{r_2-r_1}^{(p)}(y_2-y_1),
& \text{if } r_1<r_2, \\[0.3em]
g_{t-r_1}^{(p)}(x-y_1)
g_{r_1-r_2}^{(p)}(y_1-y_2),
& \text{if } r_2<r_1,
\end{cases}
\\
&\quad+
C_{T,p}
\min\left\{
|z_1|g_{t-r_1}^{(p)}(x-y_1),
|z_2|g_{t-r_2}^{(p)}(x-y_2)
\right\}.
\end{aligned}
\end{equation}
\end{itemize}

We prove the estimate for $0\le r_1<r_2<t\le T$. The case

D_{\xi_1}\sigma(u_n(r_2,y_2))
\nonumber\\

D^2_{\xi_1,\xi_2}\sigma(u_n(s,y))
L(ds,dy).

Applying Rosenthal's inequality with

&\le

\bigg\{
|z_2|^pG_{t-r_2}^p(x-y_2)
\mathbb{E}\left|D_{\xi_1}\sigma\big(u_n(r_2,y_2)\big)\right|^p\\
&\quad+

\int_{r_2}^t\int_{\mathbb{R}}
H_{t-s}^{(p)}(x-y)
\mathbb{E}\left|
D^2_{\xi_1,\xi_2}\sigma(u_n(s,y))
\right|^p
dy ds
\bigg\}.
\end{aligned}
\end{equation}
By \eqref{D-chain} and 
 \eqref{Du-pth-moment},

|z_1|^p

\label{ergodic-cov}
\begin{itemize}
\item[(i)] For every $t>0$, the random field $\{u(t,x)\}_{x\in\mathbb{R}}$ is

\[

\item[(ii)] For every $s,t>0$,
\begin{equation}
\label{limit-cov}
\lim_{R\to\infty}
\frac{1}{R}\mathbb{E}[F_R(t)F_R(s)]
=
\Sigma_{t,s}
:=
2m_2\int_0^{t\wedge s}
\mathbb{E}\big|\sigma(u(r,0))\big|^2\,dr.
\end{equation}
In particular,
\[
\sigma_R^2(t):=\operatorname{Var}(F_R(t))
\sim
\Sigma_{t,t}R,
\qquad R\to\infty.
\]
\end{itemize}

<!-- lemma-4: body_md -->
\label{lem:FR-derivative-nonlinear}
Let \(p\in[1,\frac32)\) satisfy \(m_{2p}<\infty\), and assume that $m_1<\infty$.

\begin{itemize}
\item[(i)] For any \(0\le r<t\le T\), \(y\in\mathbb{R}\) and \(z\in\mathbb{R}_0\),
\[
\|D_{r,y,z}F_R(t)\|_{2p}
\lesssim
|z|\Phi^{(p)}_{t-r,R}(y).
\]

\item[(ii)] Let \(\xi_i=(r_i,y_i,z_i)\), \(i=1,2\). For any
\(0\le r_1,r_2<t\le T\) with \(r_1\neq r_2\), \(y_1,y_2\in\mathbb{R}\) and
\(z_1,z_2\in\mathbb{R}_0\),
\[
\|D^2_{\xi_1,\xi_2}F_R(t)\|_{2p}
\lesssim
A_R(\xi_1,\xi_2)+B_R(\xi_1,\xi_2),
\]
where
\[
A_R(\xi_1,\xi_2)
=
|z_1z_2|
\begin{cases}
K^{(p)}_{r_2-r_1}(y_2-y_1)\Phi^{(p)}_{t-r_2,R}(y_2),
& r_1<r_2,\\[1mm]
K^{(p)}_{r_1-r_2}(y_1-y_2)\Phi^{(p)}_{t-r_1,R}(y_1),
& r_2<r_1,
\end{cases}
\]
and
\[
B_R(\xi_1,\xi_2)
=
\int_{-R}^{R}
\min\Big\{
|z_1|K^{(p)}_{t-r_1}(x-y_1),
|z_2|K^{(p)}_{t-r_2}(x-y_2)
\Big\}\,dx.
\]
Moreover, for \(0<\theta<1\), define
\[
\Psi_{\theta,R}(r_1,y_1,r_2,y_2)
=
\int_{-R}^{R}
\big[K^{(p)}_{t-r_1}(x-y_1)\big]^\theta
\big[K^{(p)}_{t-r_2}(x-y_2)\big]^{1-\theta}
\,dx.
\]
Then the interpolation inequality \(\min\{a,b\}\le a^\theta b^{1-\theta}\)
gives
\begin{equation}
\label{theta}
B_R(\xi_1,\xi_2)
\le
|z_1|^\theta |z_2|^{1-\theta}
\Psi_{\theta,R}(r_1,y_1,r_2,y_2).
\end{equation}
\end{itemize}

<!-- proof-7: body_md -->
Fix \(t>0\). By Theorem~\ref{ergodic-cov},
\(\sigma_R^2(t)\sim \Sigma_{t,t}R\), and hence
\(\sigma_R^{-2p}(t)\lesssim R^{-p}\). Since \(m_1+m_{2p}<\infty\),
we have \(m_a<\infty\) for all \(a\in[1,2p]\). We apply the second-order Poincar\'e inequality on the Poisson space, in the
form of \cite[Theorem~3.4]{T25}.
Let \(\gamma_1,\ldots,\gamma_7\) be the corresponding quantities. It remains
to prove that
\begin{equation}
\label{gamma-goal}
\gamma_i
\lesssim
R^{-\left(1-\frac1p\right)},\qquad i=1,\ldots,7.
\end{equation}

{\bf Estimate of $\gamma_1$.}
\begin{equation}
\label{gamma1}
\gamma_1^p
\lesssim
R^{-p}
\int_{\bf Z}
\left[
\int_{\bf Z}
\big\|D_{\xi_2}F_R(t)\big\|_{2p}
\big\|D^2_{\xi_1,\xi_2}F_R(t)\big\|_{2p}
\mathfrak{m}(d\xi_2)
\right]^p
\mathfrak{m}(d\xi_1).
\end{equation}
By Lemma~\ref{lem:FR-derivative-nonlinear},
\[
\gamma_1^p
\lesssim
\gamma_{1,A}^p+\gamma_{1,B}^p,
\]
where
\[
\gamma_{1,A}^p
:=
R^{-p}
\int_{\bf Z}
\left[
\int_{\bf Z}
|z_2|\Phi_{t-r_2,R}^{(p)}(y_2) A_R(\xi_1,\xi_2)\,\mathfrak{m}(d\xi_2)
\right]^p
\mathfrak{m}(d\xi_1),
\]
and
\[
\gamma_{1,B}^p
:=
R^{-p}
\int_{\bf Z}
\left[
\int_{\bf Z}
|z_2|\Phi_{t-r_2,R}^{(p)}(y_2)  B_R(\xi_1,\xi_2)\,\mathfrak{m}(d\xi_2)
\right]^p
\mathfrak{m}(d\xi_1).
\]
 The $A_R$ contribution is the pAm term estimated in \cite{BLW26}.
We therefore treat only the new $B_R$ contribution. We use the same
decomposition for $\gamma_2,\gamma_5,\gamma_6$, and $\gamma_7$.

 Applying \eqref{theta} with $\theta=1/p$ gives
\begin{equation}
\label{gamma1-B}
\gamma^p_{1,B}\lesssim R^{-p}m_1m_{2-\frac{1}{p}}^p
\int_0^t\int_{\mathbb{R}}
\left[
\int_0^t\int_{\mathbb{R}}
\Phi^{(p)}_{t-r_2,R}(y_2)
\Psi_{1/p,R}(r_1,y_1,r_2,y_2)
\,dy_2dr_2
\right]^p
dy_1dr_1 .
\end{equation}
By \eqref{Phi-bound} and \eqref{K-power-bound},
\[
\int_0^t\int_{\mathbb{R}}
\Phi_{t-r_2,R}^{(p)}(y_2)
\big[K_{t-r_2}^{(p)}(x-y_2)\big]^{1-1/p}
dy_2dr_2
\lesssim 1.
\]
Hence
\begin{align*}
 \gamma_{1,B}^p
&\lesssim
R^{-p}
\int_0^t\int_{\mathbb{R}}
\left[
\int_{-R}^{R}
\big[K_{t-r_1}^{(p)}(x-y_1)\big]^{1/p}
dx
\right]^p
dy_1dr_1\\
&= R^{-p}
\int_0^t
\left\|\mathbf 1_{[-R,R]}*\big[K_{t-r_1}^{(p)}\big]^{1/p}\right\|_{L^p(\mathbb{R})}^p
\,dr_1  \lesssim R^{-(p-1)}.
\end{align*}
Here the last inequality uses Young's inequality and \eqref{K-power-bound}.

\medskip
{\bf Estimate of $\gamma_2$.}  Applying \eqref{theta} with
$\theta=1/(2p)$ gives
\begin{equation}
\label{gamma2-B}
(\gamma_{2,B})^p \lesssim m_1m_{2-1/p}^p
R^{-p}
\int_0^t\int_{\mathbb{R}}
\left[
\int_0^t\int_{\mathbb{R}}
\Psi_{1/(2p),R}(r_1,y_1,r_2,y_2)^2
\,dy_2dr_2
\right]^p
dy_1dr_1 .
\end{equation}
For fixed $r_2$, by Young's convolution inequality,
\begin{align*}
&\int_{\mathbb{R}}
\Psi_{1/(2p),R}(r_1,y_1,r_2,y_2)^2
dy_2 \le
\left(
\int_{-R}^{R}
[K^{(p)}_{t-r_1}(x-y_1)]^{\frac{1}{2p}} dx
\right)^2
\int_{\mathbb{R}}
[K^{(p)}_{t-r_2}(y)]^{2(1-\frac{1}{2p})}dy .
\end{align*}
By \eqref{K-power-bound-large},
\begin{equation*}
    \int_0^t\int_{\mathbb{R}}
\Psi_{\theta,R}(r_1,y_1,r_2,y_2)^2\,dy_2dr_2
\lesssim
\left[
\int_{-R}^{R}
\big[K_{t-r_1}^{(p)}(x-y_1)\big]^{\frac{1}{2p}} dx
\right]^2 \int_0^t
(t-r_2)^{-\frac12(1-\frac1p)}dr_2.
\end{equation*}
The second integral is finite and hence 
\begin{align*}
(\gamma_{2,B})^p
&\lesssim
R^{-p}
\int_0^t\int_{\mathbb{R}}
\left[
\int_{-R}^{R}
\big[K_{t-r_1}^{(p)}(x-y_1)\big]^{\frac{1}{2p}} dx
\right]^{2p}
dy_1dr_1\\
& =
R^{-p}
\int_0^t
\left\|
\mathbf 1_{[-R,R]}*
\big[K_{t-r_1}^{(p)}\big]^{\frac1{2p}}
\right\|_{L^{2p}(\mathbb{R})}^{2p}
\,dr_1   \lesssim R^{-(p-1)}.
\end{align*}

{\bf Estimates of $\gamma_3$ and $\gamma_4$.}
The quantities $\gamma_3$ and $\gamma_4$ involve only the first Malliavin
derivative. Hence their estimates are the same as in the pAm case. 

\medskip
{\bf Estimate of $\gamma_5$.}  Applying \eqref{theta} with
$\theta=1/(2p)$ gives
\begin{equation}
\label{gamma5}
\gamma_{5,B}^p
\lesssim
m_1m_{2p-1}R^{-p}
\int_0^t\int_0^t\int_{\mathbb{R}^2}
\Psi_{1/(2p),R}(r_1,y_1,r_2,y_2)^{2p}
\,dy_1dy_2dr_1dr_2 .
\end{equation}
 By Young's inequality,
\begin{align*}
\int_{\mathbb{R}}
\Psi_{1/(2p),R}(r_1,y_1,r_2,y_2)^{2p}\,dy_1 &=\left\|
\mathbf 1_{[-R,R]}(\cdot)
\big[K_{t-r_2}^{(p)}(\cdot-y_2)\big]^{1-\frac1{2p}}
*
\big[K_{t-r_1}^{(p)}\big]^{\frac1{2p}}
\right\|_{L^{2p}(\mathbb{R})}^{2p} \\
&\le  \left\|
\mathbf 1_{[-R,R]}(\cdot)
\big[K_{t-r_2}^{(p)}(\cdot-y_2)\big]^{1-\frac1{2p}}
\right\|_{L^{2p}(\mathbb{R})}^{2p}
\left\|
\big[K_{t-r_1}^{(p)}\big]^{\frac1{2p}}
\right\|_{L^1(\mathbb{R})}^{2p}\\
&\lesssim \int_{-R}^{R}
\big[K_{t-r_2}^{(p)}(x-y_2)\big]^{2p-1}\,dx .
\end{align*}
Therefore
\begin{align*}
\gamma_{5,B}^p &\lesssim R^{-p} \int_0^t\int_0^t\int_{\mathbb{R}}\int_{-R}^{R}
\big[K_{t-r_2}^{(p)}(x-y_2)\big]^{2p-1}
\,dxdy_2dr_1dr_2 \\
&\lesssim R^{-p} R \int_0^t\int_0^t
(t-r_2)^{-(p-1)}
\,dr_1dr_2  \lesssim R^{-(p-1)}.
\end{align*}

{\bf Estimate of $\gamma_6$.} Applying \eqref{theta} with
$\theta=1-1/p$ gives
\begin{equation}
\label{gamma6-B}
\gamma_{6,B}^p
\lesssim
m_{2p-1}m_1 R^{-p}
\int_0^t\int_0^t\int_{\mathbb{R}^2}
\big[\Phi_{t-r_1,R}^{(p)}(y_1)\big]^p
\Psi_{1-\frac1p,R}(r_1,y_1,r_2,y_2)^p
dy_1dy_2dr_1dr_2.
\end{equation}
For fixed \(r_1,r_2,y_1\), by Young's inequality,
\begin{align*}
\int_{\mathbb{R}}
\Psi_{1-\frac1p,R}(r_1,y_1,r_2,y_2)^p dy_2 &=   \left\|
\mathbf 1_{[-R,R]}(\cdot)
\big[K_{t-r_1}^{(p)}(\cdot-y_1)\big]^{1-\frac1p}
*
\big[K_{t-r_2}^{(p)}\big]^{\frac1p}
\right\|_{L^p(\mathbb{R})}^p  \\
&\le \left\|
\mathbf 1_{[-R,R]}(\cdot)
\big[K_{t-r_1}^{(p)}(\cdot-y_1)\big]^{1-\frac1p}
\right\|_{L^p(\mathbb{R})}^p
\left\|
\big[K_{t-r_2}^{(p)}\big]^{\frac1p}
\right\|_{L^1(\mathbb{R})}^p  \\
&\lesssim \int_{-R}^{R}
\big[K_{t-r_1}^{(p)}(x-y_1)\big]^{p-1}\,dx
\lesssim 1,
\end{align*}
where the last inequality follows from \eqref{K-power-bound}, since
\(p-1\in(0,1)\). Therefore, from \eqref{gamma6-B}
\begin{align*}
\gamma_{6,B}^p
&\lesssim  R^{-p}
\int_0^t\int_0^t\int_{\mathbb{R}}
\big[\Phi_{t-r_1,R}^{(p)}(y_1)\big]^p
\,dy_1dr_1dr_2  \lesssim R^{-(p-1)}.
\end{align*}


{\bf Estimate of $\gamma_7$.}  Applying \eqref{theta} with
$\theta=2p-2$ gives
\begin{align}
\label{gamma7}
\gamma_{7,B}^p
&\lesssim m_{2p-1}m_1 R^{-p}
\int_0^t\int_0^t\int_{\mathbb{R}^2}
\Phi_{t-r_1,R}^{(p)}(y_1)
\nonumber\\[-1mm]
&\quad\times
\big[\Phi_{t-r_2,R}^{(p)}(y_2)\big]^{2p-2}
\Psi_{\theta,R}(r_1,y_1,r_2,y_2)
dy_1dy_2dr_1dr_2.
\end{align}
Since $2p-2,3-2p\in(0,1)$, \eqref{Phi-bound} and
\eqref{K-power-bound} imply, uniformly in $x,r_1,r_2$,
\begin{align*}
\int_{\mathbb{R}}
\Phi_{t-r_1,R}^{(p)}(y_1)
\big[K_{t-r_1}^{(p)}(x-y_1)\big]^{2p-2}dy_1
&\lesssim1,\\
\int_{\mathbb{R}}
\big[\Phi_{t-r_2,R}^{(p)}(y_2)\big]^{2p-2}
\big[K_{t-r_2}^{(p)}(x-y_2)\big]^{3-2p}dy_2
&\lesssim1.
\end{align*}
 Fubini's theorem in \eqref{gamma7} therefore gives
$\gamma_{7,B}^p\lesssim R^{-p}\int_{-R}^{R}\,dx\lesssim R^{-(p-1)}$.

\medskip
Combining the above estimates, we obtain \eqref{gamma-goal}.

<!-- proof-8: body_md -->
We verify convergence of the finite-dimensional distributions and
tightness. The finite-dimensional distributions converge by the argument in
\cite[proof of Theorem~1.3]{BLW26}, with the covariance limit from
Theorem~\ref{ergodic-cov}. For tightness, stochastic Fubini's theorem gives
\begin{equation}
\label{FR-decomp}
F_R(t)
=
\int_0^t\int_{\mathbb{R}}
\varphi_{t,R}(s,y)\sigma(u(s,y))L(ds,dy)
=
C_R(t)+M_R(t),
\end{equation}
where
\begin{align*}
C_R(t)
&=
\int_0^t\int_{\mathbb{R}}
\left(
\varphi_{t,R}(s,y)-\mathbf 1_{[-R,R]}(y)
\right)
\sigma(u(s,y))L(ds,dy), \\
M_R(t)
&=
\int_0^t\int_{\mathbb{R}}
\mathbf 1_{[-R,R]}(y)
\sigma(u(s,y))L(ds,dy).
\end{align*}
Since \(\sigma\) is Lipschitz and \(2p<3\), Theorem~\ref{thm-u-pth-moment-nonlinear}
applied with exponent \(2p\) yields
\[
\sup_{(t,x)\in[0,T]\times\mathbb{R}}
\mathbb{E}|\sigma(u(t,x))|^{2p}<\infty .
\]

 This uniform moment bound is precisely the input needed in the
tightness estimates of \cite[proof of Theorem~1.3]{BLW26}. Repeating those
estimates for $C_R$ and $M_R$ proves tightness in both the $J_1$ and uniform
topologies, and completes the proof.
