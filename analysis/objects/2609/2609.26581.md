<!-- definition-1: body_md -->
\label{assume:strong_monotone}
    An operator $F$ is called \emph{monotone}, if for all $x, y \in\mathbb{R}^d$,
        \begin{equation}\label{eq:monotone}
           \left\langle F(x) - F(y), x - y \right\rangle \geq 0.
        \end{equation}
    For $\mu>0$, we say that $F$ is \emph{$\mu$-strongly monotone} if, for all $x, y \in\mathbb{R}^d$,
\begin{equation}\label{eq:strong_monotone}
        \left\langle F(x) - F(y), x - y \right\rangle \geq \mu \|x - y\|^2.
    \end{equation}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:Lipschitz}
    An operator $F$ is called $L$-Lipschitz if $\textstyle \left\|F(x) - F(y) \right\| \leq L \left\| x - y \right\|.$
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:Holder-continuity}
We say that $F$ is $(L,\nu)$-H\"older (continuous) for some $0 < \nu \le 1$ if, for all $x,y$,
\[
    \left\|F(x)-F(y)\right\| \le L\left\|x-y\right\|^\nu .
\]
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:L0L1-Lipschitz}
We say that $F$ is $(L_0,L_1)$-Lipschitz if, for all $x,y$,
\[
    \left\|F(x)-F(y)\right\|
    \le
    \left(
        L_0+
        L_1\max_{\theta\in[0,1]}
        \left\|F(\theta x+(1-\theta)y)\right\|
    \right)
    \left\|x-y\right\|.
\]
<!-- end definition-4 -->
<!-- definition-5: body_md -->
\label{def:critical-condition-deterministic}
We say $\gamma_k > 0$ satisfies the critical condition for some $A \in (0, 1]$ if
\begin{equation}\label{eqn:gamma-critical-condition}
    \left\|F(\hat{x}_k) - F(x_k)\right\| \le A \left\|F(x_k)\right\|
\end{equation}
for $ \hat{x}_k = x_k - \gamma_k F(x_k)$.
<!-- end definition-5 -->
<!-- lemma-1: body_md -->
\label{lemma:alpha-lower-bound}
If the critical condition \eqref{eqn:gamma-critical-condition} is satisfied with $A \in (0,1]$, then we have $$\textstyle \left\langle F(\hat{x}_k),F(x_k) \right\rangle \ge \frac{1}{2} \left( \left\| F(\hat{x}_k) \right\|^2 + (1 - A^2) \left\| F(x_k) \right\|^2 \right)$$
which implies $\alpha_k \ge \frac{\gamma_k}{1+A}$.
<!-- end lemma-1 -->
<!-- proposition-1: body_md -->
\label{prop:gamma-critical-condition-sufficient}
Fix $A \in (0,1]$ and let $\hat{x}_k=x_k-\gamma_kF(x_k)$. 
Assuming $F(x_k) \ne 0$, \eqref{eqn:gamma-critical-condition} is satisfied in each of the following cases:
\begin{itemize}[leftmargin=*]
   \setlength{\itemsep}{0pt}
    \item $F$ is $L$-Lipschitz (satisfies Definition~\ref{def:Lipschitz}) with constant
    $L>0$, and $0<\gamma_k\le \frac{A}{L}$.

     \item $F$ is $(L,\nu)$-H\"older (satisfies \cref{def:Holder-continuity}), and $0 < \gamma_k \le \left(\frac{A}{L}\right)^{\frac{1}{\nu}} \left\|F(x_k)\right\|^{\frac{1-\nu}{\nu}}$.

    \item $F$ is $(L_0,L_1)$-Lipschitz (satisfies Definition~\ref{def:L0L1-Lipschitz}), and $0 < \gamma_k \le \frac{\nu_A}{L_0+L_1\left\|F(x_k)\right\|}$, where $\nu_A>0$ satisfies $\nu_A e^{\nu_A} \le A$.

    \item $F$ is $L$-Lipschitz, $(L,\nu)$-H\"older or $(L_0, L_1)$-Lipschitz and $\gamma_k$ is returned by the line-search rule in \cref{alg:PolyakEG_linesearch_unified}.
\end{itemize}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
If $F$ is $(L,\nu)$-H\"older and $0 < \gamma_k \le \left(\frac{A}{L}\right)^{\frac{1}{\nu}} \left\|F(x_k)\right\|^{\frac{1-\nu}{\nu}}$, then 
\begin{align*}
    \left\|F(\hat{x}_k) - F(x_k)\right\| \le L \left\|\hat{x}_k - x_k\right\|^\nu = L \left\|\gamma_k F(x_k)\right\|^\nu \le L \left\|\left(\frac{A}{L}\right)^{\frac{1}{\nu}} \left\|F(x_k)\right\|^{\frac{1}{\nu}}\right\|^\nu = A\left\|F(x_k)\right\| .
\end{align*}
This argument covers the Lipschitz case $\nu=1$ as well.
When $F$ is $(L_0, L_1)$-Lipschitz, using  
\eqref{eq:L0L1-alpha1-characterization} with $x=\hat{x}_k$ and $y=x_k$, we obtain
\[
\begin{aligned}
    \left\|F(\hat{x}_k)-F(x_k)\right\|
    &\le
    \left(
        L_0+L_1\left\|F(x_k)\right\|
    \right)
    \exp\left(
        L_1\left\|\hat{x}_k-x_k\right\|
    \right)
    \left\|\hat{x}_k-x_k\right\| \\
    &=
    \gamma_k
    \left(
        L_0+L_1\left\|F(x_k)\right\|
    \right)
    \exp\left(
        L_1\gamma_k\left\|F(x_k)\right\|
    \right)
    \left\|F(x_k)\right\| \\
    & \le
    \nu_A \exp\left(
        \frac{\nu_A L_1\left\|F(x_k)\right\|}
        {L_0+L_1\left\|F(x_k)\right\|}
    \right)
    \left\|F(x_k)\right\| \\
    &\le
    \nu_A e^{\nu_A}\left\|F(x_k)\right\| \le A\left\|F(x_k)\right\|
\end{aligned}
\]
where the third line uses $\gamma_k \le \frac{\nu_A}{L_0 + L_1 \left\|F(x_k)\right\|}$.
Finally, the line-search in \cref{alg:PolyakEG_linesearch_unified} accepts $\gamma_k$ only if \eqref{eqn:gamma-critical-condition} is satisfied, so there is nothing to prove. 
We later show in the proof of \cref{theorem:L0L1-linesearch-deterministic} that the line-search always terminates under any one of the regularity assumptions considered.

<!-- rem-1: body_md -->
\label{rem:role-critical-condition}
\Cref{prop:gamma-critical-condition-sufficient} shows that the critical condition~\eqref{eqn:gamma-critical-condition} can be enforced by several standard choices of $\gamma_k$. 
This allows \cref{theorem:deterministic-master-theorem} below to be stated directly in terms of the critical condition, rather than in terms of a specific step-size rule. 
In this sense, the theorem is agnostic to the assumptions on the problem class or how $\gamma_k$ is selected, and is capable of stating the convergence results in a general, unified manner.
<!-- end rem-1 -->
<!-- rem-2: body_md -->
\label{rem:role-parameter-A}
The parameter $A$ can be viewed a tolerance parameter; with larger $A$, a broader range of $\gamma_k$ is admissible. 
The most aggressive choice $A=1$ is allowed when one only needs to bound $\left\|F(\hat{x}_k)\right\|$.
However, we require $A\in (0,1)$ in order to control $F(x_k)$ using $F(\hat{x}_k)$, because
\[
    \left\|F(\hat{x}_k)\right\| \ge \left\|F(x_k)\right\|-\left\|F(\hat{x}_k)-F(x_k)\right\|
    \ge (1-A)\left\|F(x_k)\right\| .
\]
Contraction in the strongly monotone case also requires $A\in (0,1)$.
<!-- end rem-2 -->
<!-- rem-3: body_md -->
\label{rem:choice-nuA}
For a fixed value of $A$, the largest possible value of $\nu_A$ satisfying $\nu_A e^{\nu_A}\le A$ is $\nu_A=W(A)$, 
where $W$ denotes the Lambert $W$ function. 
One may use any smaller positive value of $\nu_A$, e.g., $\nu_A = \frac{A}{1+A}$.
There is no universally optimal choice of $A$, and the best value
depends on which bound one wants to optimize.

<!-- theorem-1: body_md -->
\label{theorem:deterministic-master-theorem}
Let $F$ be monotone. Suppose that we choose $\gamma_k > 0$ so that the critical condition \eqref{eqn:gamma-critical-condition} holds for all $k\ge 0$.
Then, \hyperref[alg:PolyakEG]{{\color{PineGreen}\sf  PolyakEG}\xspace} satisfies:               
\vspace{-2mm}
\begin{itemize}
    \item If \eqref{eqn:gamma-critical-condition} is satisfied with $A \in (0, 1]$, then the extrapolation points $\hat{x}_k$ satisfy:
    \begin{equation}
        \min_{k=0,\dots,K} \gamma_k^2 \left\| F(\hat{x}_k) \right\|^2 \le \frac{(1+A)^2 \left\| x_0 - x_* \right\|^2}{K+1}. \label{eq:deterministic-master-theorem_eq1}
    \end{equation}
    \item If \eqref{eqn:gamma-critical-condition} is satisfied with $A \in (0, 1)$, then the iterates $x_k$ satisfy:
    \begin{equation}
        \min_{k=0,\dots,K} \gamma_k^2 \left\| F(x_k) \right\|^2 \le \frac{(1+A) \left\| x_0 - x_* \right\|^2}{(1-A) (K+1)}. \label{eq:deterministic-master-theorem_eq2} 
    \end{equation}
    \item If $F$ is $\mu$-strongly monotone and $A \in (0, 1)$, then the iterates $x_k$ satisfy:
    \begin{equation}
        \left\| x_{k+1} - x_* \right\|^2 \le \prod_{j=0}^{k} \left( 1 - \frac{2(1-A) \gamma_j \mu}{(1+A)^2} \right) \left\| x_0 - x_* \right\|^2 . \label{eq:deterministic-master-theorem_eq3}
    \end{equation}
\end{itemize}
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
Let $F$ satisfy $\left\langle F(x) - F(y),x - y \right\rangle \ge \mu \left\| x-y \right\|^2$ for any $x,y \in \mathbb{R}^d$, where we take $\mu = 0$ if $F$ is merely monotone. Then we have
\begin{align*}
    \left\| x_{k+1} - x_* \right\|^2 & = \left\| x_k - \alpha_k F(\hat{x}_k) - x_* \right\|^2 \nonumber \\
    & = \left\| x_k - x_* \right\|^2 - 2\alpha_k \left\langle F(\hat{x}_k),x_k - x_* \right\rangle + \alpha_k^2 \left\| F(\hat{x}_k) \right\|^2 \nonumber \\
    & = \left\| x_k - x_* \right\|^2 - 2\alpha_k \left\langle F(\hat{x}_k),x_k - \hat{x}_k \right\rangle - 2\alpha_k \left\langle F(\hat{x}_k),\hat{x}_k - x_* \right\rangle + \alpha_k^2 \left\| F(\hat{x}_k) \right\|^2 \nonumber \\
    & \le \left\| x_k - x_* \right\|^2 - 2\alpha_k \left\langle F(\hat{x}_k),x_k - \hat{x}_k \right\rangle - 2\alpha_k \mu \left\| \hat{x}_k - x_* \right\|^2 + \alpha_k^2 \left\| F(\hat{x}_k) \right\|^2 .
\end{align*}
Using Young's inequality (with $\omega > 0$ to be determined later), we obtain
\begin{align}
    & \left\| x_{k+1} - x_* \right\|^2 \nonumber \\
    & \le \left\| x_k - x_* \right\|^2 - 2\alpha_k \left\langle F(\hat{x}_k),x_k - \hat{x}_k \right\rangle - \frac{2\alpha_k \mu}{1+\omega} \left\| x_k - x_* \right\|^2 + \frac{2\alpha_k \mu}{\omega} \left\| x_k - \hat{x}_k \right\|^2 + \alpha_k^2 \left\| F(\hat{x}_k) \right\|^2 \nonumber \\
    & = \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 + \frac{2\alpha_k \mu}{\omega} \left\| x_k - \hat{x}_k \right\|^2 - 2\alpha_k \left\langle F(\hat{x}_k),x_k - \hat{x}_k \right\rangle + \alpha_k^2 \left\| F(\hat{x}_k) \right\|^2 \nonumber \\
    & = \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 + \frac{2\alpha_k \gamma_k^2 \mu}{\omega} \left\| F(x_k) \right\|^2 - \alpha_k \left\langle F(\hat{x}_k),x_k - \hat{x}_k \right\rangle \nonumber \\
    & = \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 + \frac{2\alpha_k \gamma_k^2 \mu}{\omega} \left\| F(x_k) \right\|^2 - \alpha_k \gamma_k \left\langle F(\hat{x}_k),F(x_k) \right\rangle \nonumber \\
    & = \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 - \alpha_k \gamma_k \left( \left\langle F(\hat{x}_k),F(x_k) \right\rangle - \frac{2\gamma_k \mu}{\omega} \left\| F(x_k) \right\|^2 \right) .
    \label{eqn:LS-proof-bound}
\end{align}
Here, for the third line, we use the definition of $\alpha_k$ to replace $\alpha_k^2 \left\| F(\hat{x}_k) \right\|^2 = \alpha_k \left\langle F(\hat{x}_k),x_k - \hat{x}_k \right\rangle$.

\vspace{.2cm}
\noindent
\textbf{$\bullet$ $F$ is monotone.} 
Substituting $\mu = 0$ into \eqref{eqn:LS-proof-bound} and using \Cref{lemma:alpha-lower-bound} we have
\begin{align}
    \left\| x_{k+1} - x_* \right\|^2 & \le \left\| x_k - x_* \right\|^2 - \alpha_k \gamma_k \left\langle F(\hat{x}_k),F(x_k) \right\rangle \notag\\
    & \le \left\| x_k - x_* \right\|^2 - \frac{\alpha_k \gamma_k}{2} \left( \left\| F(\hat{x}_k) \right\|^2 + (1 - A^2) \left\| F(x_k) \right\|^2 \right) \notag\\
    & \le \left\| x_k - x_* \right\|^2 - \frac{\gamma_k^2}{2(1+A)} \left( \left\| F(\hat{x}_k) \right\|^2 + (1 - A^2) \left\| F(x_k) \right\|^2 \right) \label{eqn:monotone polyakeg bounded radius}.
\end{align}
Summing the above inequality for $k=0,1,\dots,K$ we obtain
\begin{align}
\label{eqn:monotone-bound-Fxk-Fhxk-combined}
    \sum_{k=0}^{K} \frac{\gamma_k^2}{2(1+A)}  \left( \left\| F(\hat{x}_k) \right\|^2 + (1 - A^2) \left\| F(x_k) \right\|^2 \right) \le \left\| x_0 - x_* \right\|^2 - \left\| x_{K+1} - x_* \right\|^2 \le \left\| x_0 - x_* \right\|^2 .
\end{align}
Now note that we have
\begin{align}
\label{eqn:Fhxk-upper-bound-with-Fxk}
    \left\|F(\hat{x}_k)\right\| \le \left\|F(x_k)\right\| + \left\|F(\hat{x}_k) - F(x_k)\right\| \le \left\|F(x_k)\right\| + A\left\|F(x_k)\right\| = (1+A) \left\|F(x_k)\right\|
\end{align}
and
\begin{align}
\label{eqn:Fhxk-lower-bound-with-Fxk}
    \left\|F(\hat{x}_k)\right\| \ge \left\|F(x_k)\right\| - \left\|F(\hat{x}_k) - F(x_k)\right\| \ge \left\|F(x_k)\right\| - A\left\|F(x_k)\right\| = (1-A) \left\|F(x_k)\right\| .
\end{align}
Therefore, further lower bounding the left hand side of \eqref{eqn:monotone-bound-Fxk-Fhxk-combined} using \eqref{eqn:Fhxk-upper-bound-with-Fxk} we have
\begin{align*}
    \left\| x_0 - x_* \right\|^2 & \ge \sum_{k=0}^K \frac{\gamma_k^2}{2(1+A)} \left( \left\| F(\hat{x}_k) \right\|^2 + \frac{1-A^2}{(1+A)^2} \left\| F(\hat{x}_k) \right\|^2 \right) \\
    & = \sum_{k=0}^K \frac{\gamma_k^2}{(1+A)^2} \left\| F(\hat{x}_k) \right\|^2 \\
    & \ge \frac{(K+1)}{(1+A)^2} \min_{k=0,\dots,K} \gamma_k^2 \left\| F(\hat{x}_k) \right\|^2 .
\end{align*}
This proves the first part. On the other hand, lower bounding \eqref{eqn:monotone-bound-Fxk-Fhxk-combined} in terms of $\left\|F(x_k)\right\|$ using \eqref{eqn:Fhxk-lower-bound-with-Fxk} we have
\begin{align*}
    \left\| x_0 - x_* \right\|^2 & \ge \sum_{k=0}^K \frac{\gamma_k^2}{2(1+A)} \left( (1-A)^2 \left\| F(x_k) \right\|^2 + (1-A^2) \left\| F(x_k) \right\|^2 \right) \\
    & = \sum_{k=0}^K \frac{1-A}{1+A} \gamma_k^2 \left\| F(x_k) \right\|^2 \\
    & \ge \frac{(K+1) (1-A)}{1+A} \min_{k=0,\dots,K} \gamma_k^2 \left\| F(x_k) \right\|^2 .
\end{align*}
This proves the second part of the monotone setting. 

\vspace{.2cm}
\noindent
\textbf{$\bullet$ $F$ is $\mu$-strongly monotone with $\mu>0$.}
Strong monotonicity of $F$ implies
\begin{align*}
    \left\langle x_k - \hat{x}_k,F(x_k) - F(\hat{x}_k) \right\rangle \ge \mu \left\| x_k - \hat{x}_k \right\|^2 .
\end{align*}
Substituting $x_k - \hat{x}_k = \gamma_k F(x_k)$ in the above we get
\begin{align*}
    \left\langle F(x_k),F(x_k) - F(\hat{x}_k) \right\rangle \ge \gamma_k \mu \left\| F(x_k) \right\|^2 .
\end{align*}
We apply the above inequality to \eqref{eqn:LS-proof-bound}, which yields:
\begin{align*}
    & \left\| x_{k+1} - x_* \right\|^2 \\
    & \le \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 - \alpha_k \gamma_k \left( \left\langle F(\hat{x}_k),F(x_k) \right\rangle - \frac{2}{\omega} \gamma_k \mu \left\| F(x_k) \right\|^2 \right) \\
    & \le \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 - \alpha_k \gamma_k \left( \left\langle F(\hat{x}_k),F(x_k) \right\rangle - \frac{2}{\omega} \left\langle F(x_k),F(x_k) - F(\hat{x}_k) \right\rangle \right) \\
    & = \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 - \alpha_k \gamma_k \left( \left(1 + \frac{2}{\omega}\right) \left\langle F(\hat{x}_k),F(x_k) \right\rangle - \frac{2}{\omega} \left\| F(x_k) \right\|^2 \right) .
\end{align*}
Now we use \Cref{lemma:alpha-lower-bound} to obtain
\begin{align*}
    \left\| x_{k+1} - x_* \right\|^2 & \le \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 \\
    & - \alpha_k \gamma_k \left( \frac{1}{2} \left(1 + \frac{2}{\omega}\right) \left( \left\| F(\hat{x}_k) \right\|^2 + (1 - A^2) \left\| F(x_k) \right\|^2 \right) - \frac{2}{\omega} \left\| F(x_k) \right\|^2 \right) \\
    & \le \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 \\
    & - \alpha_k \gamma_k \left( \frac{1}{2} \left(1 + \frac{2}{\omega}\right) \left( (1-A)^2 \left\| F(x_k) \right\|^2 + (1 - A^2) \left\| F(x_k) \right\|^2 \right) - \frac{2}{\omega} \left\| F(x_k) \right\|^2 \right) \\
    & \le \left( 1 - \frac{2\alpha_k \mu}{1+\omega} \right) \left\| x_k - x_* \right\|^2 - \alpha_k \gamma_k \left( 1 - A - \frac{2A}{\omega} \right) \left\| F(x_k) \right\|^2 .
\end{align*}
where the second inequality uses \eqref{eqn:Fhxk-lower-bound-with-Fxk}.
Now taking $\omega = \frac{2A}{1-A}$ in the last inequality gives
\begin{align*}
    \left\| x_{k+1} - x_* \right\|^2 & \le \left( 1 - \frac{2(1-A)}{1+A} \alpha_k \mu \right) \left\| x_k - x_* \right\|^2 \le \left( 1 - \frac{2(1-A)}{(1+A)^2} \gamma_k \mu \right) \left\| x_k - x_* \right\|^2 .
\end{align*}
where for the second line we use \cref{lemma:alpha-lower-bound}.
Unrolling the recursion gives the desired result.

<!-- proposition-2: body_md -->
\label{theorem:L0L1-linesearch-deterministic}
If $F$ is $L$-Lipschitz, $(L,\nu)$-H\"older or $(L_0, L_1)$-Lipschitz, then at each iteration $k$, \cref{alg:PolyakEG_linesearch_unified} accepts a step-size $\gamma_k$ satisfying the critical condition \eqref{eqn:gamma-critical-condition}.
Furthermore, 
\begin{itemize}
    \item[(a)]
    When $F$ is $L$-Lipschitz or $(L_0, L_1)$-Lipschitz, then the total number of while-loop calls over all iterations is at most
    \begin{equation}
    \label{eqn:linesearch-operator-call-bound}
        N
        :=
        2\max\left\{
            0,
            \left\lceil
                \frac{\log(L_0/\lambda_{-1}^0)}
                     {\log(\beta-1)}
            \right\rceil,
            \left\lceil
                \frac{\log(L_1/\lambda_{-1}^1)}
                     {\log(\beta-1)}
            \right\rceil
        \right\} 
    \end{equation}
    where we disregard the last term involving $L_1/\lambda_{-1}^1$ when $\lambda_{-1}^1 = L_1 = 0$, and we require $\lambda_{-1}^1 > 0$ when $L_1 > 0$.
    Consequently, for all accepted steps we have
    \[
        \lambda_k^0
        \le
        \bar\lambda^0
        :=
        \beta^{N/2}\lambda_{-1}^0,
        \qquad
        \lambda_k^1
        \le
        \bar\lambda^1
        :=
        \beta^{N/2}\lambda_{-1}^1.
    \]

    \item[(b)] Suppose $F$ is $(L,\nu)$-H\"older with $\nu \in (0,1)$. Then for any $\epsilon > 0$, the number of while-loop calls $N_\epsilon$ before reaching $\left\|F(x_k)\right\| \le \epsilon$ for the first time is bounded by ${\cal O}\left( \log \frac{1}{\epsilon} \right)$, and the step-size satisfies $\gamma_k \ge \min \left\{ \frac{1}{\beta - 1} \left(\frac{A}{L}\right)^{\frac{1}{\nu}} \epsilon^{\frac{1-\nu}{\nu}} , \frac{\nu_A}{\lambda_{-1}^0} \right\}$.
\end{itemize}

<!-- proof-3: body_md -->
(a) If $F$ is $(L_0, L_1)$-Lipschitz, then once we reach $\lambda_k^0 \ge L_0$ and $\lambda_k^1 \ge L_1$, we have \eqref{eqn:gamma-critical-condition} satisfied for any $x_k$ by \eqref{eq:L0L1-alpha1-characterization}, and no further line-search loops are called.
At each while loop call, we alternatingly increase either $\lambda_k^0$ or $\lambda_k^1$ by the factor
\begin{align*}
    \beta - \frac{\lambda_k^0}{\lambda_k^0 + \lambda_k^1 \left\|F(x_k)\right\|} \ge \beta - 1 > 1 \quad \text{or} \quad \beta - \frac{\lambda_k^1 \left\|F(x_k)\right\|}{\lambda_k^0 + \lambda_k^1 \left\|F(x_k)\right\|} \ge \beta - 1 > 1 ,
\end{align*}
respectively. 
The number of total while loop calls throughout the algorithm's execution is therefore upper bounded by \eqref{eqn:linesearch-operator-call-bound}.
Both $\lambda_k^0$ or $\lambda_k^1$ is increased at most $\frac{N}{2}$ times, by a factor at most $\beta$, so we have $\lambda_k^i \le \beta^{N/2} \lambda_{-1}^i$ for $i=0,1$.\medskip


\noindent
(b) It remains to show that the while loop run always terminates for $(L,\nu)$-H\"older problems with $\nu \in (0,1)$, and the corresponding bound on the number of while-loop calls in this case.
If $F(x_k) = 0$ then $x_k$ is already a solution, and the algorithm terminates.
Otherwise, by \cref{prop:gamma-critical-condition-sufficient} we know that \eqref{eqn:gamma-critical-condition} is satisfied if $\lambda_k^0 \ge \nu_A \left(\frac{L}{A}\right)^{\frac{1}{\nu}} \left\|F(x_k)\right\|^{-\frac{1-\nu}{\nu}}$, so the line-search at iteration $k$ terminates  in finite steps.
Additionally, because each rejected line-search loop increases $\lambda_k^0$ by factor $\beta-1$, unless  the initial $\lambda_k^0$ already satisfies the stopping criterion, 
the accepted $\lambda_k^0$ is at most $(\beta-1) \nu_A \left(\frac{L}{A}\right)^{\frac{1}{\nu}} \left\|F(x_k)\right\|^{-\frac{1-\nu}{\nu}}$. 
Now suppose that $\epsilon$-accuracy is not reached until the $k$-th iteration, i.e., $\left\|F(x_j)\right\| > \epsilon$ for $j=0,\dots,k$.
This implies $\lambda_k^0 \le \max \left\{ \lambda_{-1}^0 , (\beta-1) \nu_A \left(\frac{L}{A}\right)^{\frac{1}{\nu}} \epsilon^{-\frac{1-\nu}{\nu}} \right\}$, and thus,
\[
    \gamma_k = \frac{\nu_A}{\lambda_k^0} \ge \min \left\{ \frac{1}{\beta - 1} \left(\frac{A}{L}\right)^{\frac{1}{\nu}} \epsilon^{\frac{1-\nu}{\nu}} , \frac{\nu_A}{\lambda_{-1}^0} \right\}
\]
and the number of line-search rejections up to iteration $k$ is at most
\[
    N_\epsilon = \log_{\beta-1} \left( \max\left\{ 1, \frac{(\beta-1) \nu_A \left(\frac{L}{A}\right)^{\frac{1}{\nu}} \epsilon^{-\frac{1-\nu}{\nu}}}{\lambda_{-1}^0} \right\} \right) = \left[ C + \frac{(1-\nu)}{\nu \log (\beta - 1)} \log \frac{1}{\epsilon} \right]_+ = {\cal O} \left( \log  \frac{1}{\epsilon} \right) 
\]
where $C$ is a constant depending only on $L, \nu, A, \nu_A, \beta$ and $\lambda_{-1}^0$.

\label{def:critical-condition-stochastic}
We say $\gamma_k$ satisfies the critical condition at iteration $k$ with respect to $F_{\mathcal{S}_k}$ for some $A \in (0, 1]$ if
\begin{equation}\label{eqn:gamma-critical-condition-stochastic}
\textstyle
    \left\|F_{\mathcal{S}_k}(\hat{x}_k) - F_{\mathcal{S}_k}(x_k)\right\| \le A \left\|F_{\mathcal{S}_k}(x_k)\right\|
\end{equation}
for $\hat{x}_k = x_k - \gamma_k F_{\mathcal{S}_k}(x_k)$.

<!-- lemma-2: body_md -->
\label{lemma:PolyakSEG-LS-key-properties}
Let $x_k$ be an iterate from either \hyperref[alg:PolyakSEG]{{\color{PineGreen}\sf  PolyakSEG}\xspace} or \hyperref[alg:decPolyakSEG]{{\color{PineGreen}\sf  DecPolyakSEG}\xspace}, and suppose that $\gamma_k$ satisfies the critical condition (\Cref{def:critical-condition-stochastic}). 
Then, for $k=0,1,\dots$,
\begin{enumerate}
[label=(\alph*)]
    \item $(1-A) \left\|F_{\mathcal{S}_k}(x_k)\right\| \le \left\|F_{\mathcal{S}_k}(\hat{x}_k)\right\| \le (1+A)\left\|F_{\mathcal{S}_k}(x_k)\right\|$
    \item $\left\langle F_{\mathcal{S}_k}(\hat{x}_k),F_{\mathcal{S}_k}(x_k) \right\rangle
    \ge \max \left\{ (1-A) \left\| F_{\mathcal{S}_k}(x_k) \right\|^2 , \frac{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2}{1+A} \right\}$ if $0 < A < 1$
    \item $\frac{\gamma_k}{1+A} \le \alpha_k \le \frac{\gamma_k}{1-A}$ if $0 < A < 1$
\end{enumerate}
holds almost surely.

<!-- proof-4: body_md -->
By the critical condition \eqref{eqn:gamma-critical-condition-stochastic}, we obtain
\begin{align*}
    \left\|F_{\mathcal{S}_k}(\hat{x}_k)\right\| \le \left\|F_{\mathcal{S}_k}(x_k)\right\| + \left\|F_{\mathcal{S}_k}(\hat{x}_k) - F_{\mathcal{S}_k}(x_k)\right\| \le \left\|F_{\mathcal{S}_k}(x_k)\right\| + A\left\|F_{\mathcal{S}_k}(x_k)\right\| = (1+A) \left\|F_{\mathcal{S}_k}(x_k)\right\|
\end{align*}
and
\begin{align*}
    \left\|F_{\mathcal{S}_k}(\hat{x}_k)\right\| \ge \left\|F_{\mathcal{S}_k}(x_k)\right\| - \left\|F_{\mathcal{S}_k}(\hat{x}_k) - F_{\mathcal{S}_k}(x_k)\right\| \ge \left\|F_{\mathcal{S}_k}(x_k)\right\| - A\left\|F_{\mathcal{S}_k}(x_k)\right\| \ge (1-A) \left\|F_{\mathcal{S}_k}(x_k)\right\|
\end{align*}
which proves (a).
Next, observe that
\begin{align*}
    \left\langle F_{\mathcal{S}_k}(\hat{x}_k),F_{\mathcal{S}_k}(x_k) \right\rangle & = \frac{1}{2} \left( \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2 - \left\| F_{\mathcal{S}_k}(\hat{x}_k) - F_{\mathcal{S}_k}(x_k) \right\|^2 + \left\| F_{\mathcal{S}_k}(x_k) \right\|^2 \right) \\
    & \ge \frac{1}{2} \left( \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2 - A^2 \left\| F_{\mathcal{S}_k}(x_k) \right\|^2 +\left\| F_{\mathcal{S}_k}(x_k) \right\|^2 \right) \\
    & = \frac{1}{2} \left( \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2 + (1 - A^2) \left\| F_{\mathcal{S}_k}(x_k) \right\|^2 \right) 
\end{align*}
where the inequality uses \eqref{eqn:gamma-critical-condition-stochastic}.
Applying the first inequality of (a) to the last expression, we obtain
\begin{align*}
    \left\langle F_{\mathcal{S}_k}(\hat{x}_k),F_{\mathcal{S}_k}(x_k) \right\rangle \ge \frac{1}{2} \left( (1-A)^2 \left\| F_{\mathcal{S}_k}(x_k) \right\|^2 + (1 - A^2) \left\| F_{\mathcal{S}_k}(x_k) \right\|^2 \right) = (1-A) \left\| F_{\mathcal{S}_k}(x_k) \right\|^2 ,
\end{align*}
while applying the second inequality of (a) gives
\begin{align*}
    \left\langle F_{\mathcal{S}_k}(\hat{x}_k),F_{\mathcal{S}_k}(x_k) \right\rangle \ge \frac{1}{2} \left( \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2 + (1 - A^2) \frac{1}{(1+A)^2} \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2 \right) = \frac{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2}{1+A} .
\end{align*}
This proves (b).
Finally, for {\color{PineGreen}\sf  PolyakSEG}\xspace we immediately obtain
\begin{align*}
    \alpha_k = \frac{\left\langle F_{\mathcal{S}_k}(\hat{x}_k),x_k - \hat{x}_k \right\rangle}{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2} = \frac{\gamma_k \left\langle F_{\mathcal{S}_k}(\hat{x}_k),F_{\mathcal{S}_k}(x_k) \right\rangle}{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2} \ge \frac{\gamma_k}{1+A} .
\end{align*}
For the case of {\color{PineGreen}\sf  DecPolyakSEG}\xspace, we have $\gamma_k \le \gamma_{k-1}$ and $\alpha_k \le \alpha_{k-1}$ for all $k=0,1,\dots$ by construction.
As $\alpha_{-1} = \infty$, 
\begin{align*}
    \alpha_0 = \min \left\{ \frac{\left\langle F_{{\mathcal S}_0}(\hat{x}_0),x_0 - \hat{x}_0 \right\rangle}{\left\| F_{{\mathcal S}_0}(\hat{x}_0) \right\|^2} , \alpha_{-1} \right\} = \frac{\left\langle F_{{\mathcal S}_0}(\hat{x}_0),x_0 - \hat{x}_0 \right\rangle}{\left\| F_{{\mathcal S}_0}(\hat{x}_0) \right\|^2} \ge \frac{\gamma_0}{1+A} .
\end{align*}
Now we use induction on $k=1,2,\dots$: assuming that $\alpha_{k-1} \ge \frac{\gamma_{k-1}}{1+A}$, we have $\alpha_{k-1} \ge \frac{\gamma_{k}}{1+A}$ (since $\gamma_{k} \le \gamma_{k-1}$). 
We further have $\frac{\left\langle F_{\mathcal{S}_k}(\hat{x}_k),x_k - \hat{x}_k \right\rangle}{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2} \ge \frac{\gamma_k}{1+A}$ by (b), which implies
\begin{align*}
    \alpha_k = \min \left\{ \alpha_{k-1}, \frac{\left\langle F_{\mathcal{S}_k}(\hat{x}_k),x_k - \hat{x}_k \right\rangle}{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2} \right\} \ge \frac{\gamma_k}{1+A} ,
\end{align*}
completing the induction. This proves the first inequality in (c).
Finally, for the second inequality in (c), observe that for both {\color{PineGreen}\sf  PolyakSEG}\xspace and {\color{PineGreen}\sf  DecPolyakSEG}\xspace,
\begin{align*}
    \alpha_k & \le \frac{\gamma_k \left\langle F_{\mathcal{S}_k}(\hat{x}_k),F_{{\mathcal S}_k}(x_k) \right\rangle}{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2} \le \frac{\gamma_k}{2} \frac{\frac{1}{1-A} \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2 + (1-A) \left\| F_{{\mathcal S}_k}(x_k) \right\|^2}{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2} \\
    & \le \frac{\gamma_k}{2} \frac{\frac{1}{1-A} \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2 + \frac{1}{1-A} \left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2}{\left\| F_{\mathcal{S}_k}(\hat{x}_k) \right\|^2} = \frac{\gamma_k}{1-A}
\end{align*}
where the third inequality uses (a).

        \right)

    \end{equation}

\end{align*}
so \eqref{eqn:gamma-critical-condition-stochastic} is satisfied.

Using \eqref{eq:DecPolyakSEG-uK-distance} and nonincreasingness of $\alpha_k$, we obtain
\begin{align}
    &

    \frac{R^2}{2\alpha_K}.
    \label{eq:DecPolyakSEG-weighted-telescoping-bound}
\end{align}
Summing \eqref{eq:DecPolyakSEG-residual-comparison} for $k=0,\dots,K$ with $u=u_K$ and using \eqref{eq:DecPolyakSEG-weighted-telescoping-bound} and $\|u_K-\hat{x}_k\|\leq R$ yields
\begin{align}
    & \sum_{k=0}^K
    \left\langle F(\hat{x}_k),\hat{x}_k-u_K \right\rangle \nonumber \\
    & \leq
    \frac{R^2}{2\alpha_K}
    +
    \left\langle 
        \sum_{k=0}^K
        \bigl(
            F_{\mathcal S_k}(x_k)-F(x_k)

        F_{\mathcal S_k}(x_k)-F(x_k)

        +
        4L^2R^2
    \Bigr).
    \label{eq:DecPolyakSEG-residual-summed}
\end{align}

We next bound the right hand side of \eqref{eq:DecPolyakSEG-residual-summed} in expectation.
Again, denote by $\mathbb{E}_k \left[ \cdot\right]$ the conditional expectation with respect to the randomness revealed before drawing $\mathcal S_k$.
Then
\begin{align*}
    \mathbb{E}_k \left[ 
        \|F_{\mathcal S_k}(x_k)-F(x_k)\|^2
    \right]
    & =
    \mathbb{E}_k \left[ \|F_{\mathcal S_k}(x_k)\|^2\right]
    -
    \|F(x_k)\|^2
    \\
    & \leq
    2\mathbb{E}_k \left[ 
        \|F_{\mathcal S_k}(x_k)
          -F_{\mathcal S_k}(x_*)\|^2
    \right]
    +
    2\mathbb{E}_k \left[ \|F_{\mathcal S_k}(x_*)\|^2\right] \nonumber\\
    & \leq
    2L^2\|x_k-x_*\|^2
    +
    2\mathbb{E} \left[ \|F_{\mathcal S}(x_*)\|^2\right]
    \leq
    \sigma^2
\end{align*}
so $\mathbb{E} \left[ \left\| F_{\mathcal S_k}(x_k)-F(x_k) \right\|^2\right] \le \sigma^2$ by the tower property.
Next, because $F_{{\mathcal S}_j}(x_j) - F(x_j)$ for $j<k$ is measurable with respect to information revealed before drawing ${\mathcal S}_k$ and $\mathbb{E}_k \left[ F_{{\mathcal S}_k}(x_k) - F(x_k)\right] = 0$, we have $\mathbb{E} \left[ 
    \left\langle 
        F_{\mathcal S_j}(x_j)-F(x_j)
    ,
        F_{\mathcal S_k}(x_k)-F(x_k)
     \right\rangle
\right]
= 0$ and therefore, 
\begin{align*}
    & \mathbb{E} \left[ 
        \left\|
            \sum_{k=0}^K
            \bigl(
                F_{\mathcal S_k}(x_k)-F(x_k)
            \bigr)
        \right\|^2
    \right] \\
    & =
    \sum_{k=0}^K
    \mathbb{E} \left[ 
        \|F_{\mathcal S_k}(x_k)-F(x_k)\|^2
    \right]
    +
    2
    \sum_{0\leq j<k\leq K}
    \mathbb{E} \left[ 
        \left\langle 
            F_{\mathcal S_j}(x_j)-F(x_j)
        ,
            F_{\mathcal S_k}(x_k)-F(x_k)
         \right\rangle
    \right]
    \\
    &=
    \sum_{k=0}^K
    \mathbb{E} \left[ 
        \|F_{\mathcal S_k}(x_k)-F(x_k)\|^2
    \right]
    \\
    &\leq
    (K+1)\sigma^2 .
\end{align*}
Using $\left\|u_k - x_0\right\| \le R$ and Cauchy--Schwarz inequality, we then obtain
\begin{align}
    \mathbb{E} \left[ 
        \left\langle 
            \sum_{k=0}^K
            \bigl(
                F_{\mathcal S_k}(x_k)-F(x_k)
            \bigr)
        ,
            u_K-x_0
         \right\rangle
    \right]
    & \leq
    \mathbb{E} \left[ 
        \left\|
            \sum_{k=0}^K
            \bigl(
                F_{\mathcal S_k}(x_k)-F(x_k)
            \bigr)
        \right\|
        \|u_K-x_0\|
    \right]
    \nonumber\\
    &\leq
    R
    \mathbb{E} \left[ 
        \left\|
            \sum_{k=0}^K
            \bigl(
                F_{\mathcal S_k}(x_k)-F(x_k)
            \bigr)
        \right\|
    \right]
    \nonumber\\
    &\leq
    R
    \left(
        \mathbb{E} \left[ 
            \left\|
                \sum_{k=0}^K
                \bigl(
                    F_{\mathcal S_k}(x_k)-F(x_k)
                \bigr)
            \right\|^2
        \right]
    \right)^{1/2}
    \nonumber\\
    &\leq
    \sigma R\sqrt{K+1}.
    \label{eq:DecPolyakSEG-martingale-sum}
\end{align}
Furthermore, using $\mathbb{E}_k \left[ F_{{\mathcal S}_k}(x_k) - F(x_k)\right] = 0$ again and applying the tower rule, we have
\begin{align}
    \mathbb{E} \left[ 
        \sum_{k=0}^K
        \left\langle 
            F_{\mathcal S_k}(x_k)-F(x_k)
        ,
            x_0-x_k
         \right\rangle
    \right]
    =
    0 .
\label{eq:DecPolyakSEG-centered-cross-term}
\end{align}
Finally, the step-size condition for {\color{PineGreen}\sf  DecPolyakSEG}\xspace gives $c_k\gamma_k \le c_{k-1}\gamma_{k-1} \le \cdots \le c_{-1}\gamma_{-1} = \gamma_{-1}$, 
and since we take $c_k = \sqrt{k+1}$, we have $\gamma_k \le \frac{\gamma_{-1}}{\sqrt{k+1}}$.
Hence
\begin{align}
    &

        \sum_{k=0}^K
        \gamma_k
        \Bigl(
            \|F_{\mathcal S_k}(x_k)-F(x_k)\|^2
            +
            4L^2R^2
        \Bigr)
    \right]
    \leq
    \sum_{k=0}^K
    \frac{\gamma_{-1}}{\sqrt{k+1}}
    \mathbb{E} \left[ 
        \|F_{\mathcal S_k}(x_k)-F(x_k)\|^2
        +
        4L^2R^2
    \right]
    \nonumber\\
    &\leq
    \gamma_{-1}
    \left(
        \sigma^2+4L^2R^2
    \right)
    \sum_{k=0}^K\frac{1}{\sqrt{k+1}}
    \leq
    2\gamma_{-1}
    \left(
        \sigma^2+4L^2R^2
    \right)
    \sqrt{K+1}.
    \label{eq:DecPolyakSEG-weighted-noise-bound}
\end{align}
Combining \eqref{eq:DecPolyakSEG-residual-summed}, \eqref{eq:DecPolyakSEG-martingale-sum}, \eqref{eq:DecPolyakSEG-centered-cross-term} and \eqref{eq:DecPolyakSEG-weighted-noise-bound} gives
\begin{align}
    \mathbb{E} \left[ 
        \sum_{k=0}^K
        \left\langle F(\hat{x}_k),\hat{x}_k-u_K \right\rangle
    \right]
    \leq
    \mathbb{E} \left[ \frac{R^2}{2\alpha_K}\right]
    +
    \sigma R\sqrt{K+1}
    +
    \frac{2\gamma_{-1}}{1-A}
    \left(
        \sigma^2+4L^2R^2
    \right)
    \sqrt{K+1}.
    \label{eq:DecPolyakSEG-residual-upper}
\end{align}
By monotonicity of $F$, we have $\left\langle F(\hat{x}_k),\hat{x}_k-u_K \right\rangle \geq \left\langle F(u_K),\hat{x}_k-u_K \right\rangle$.
Using this to lower bound the left hand side of \eqref{eq:DecPolyakSEG-residual-upper} and summing over $k=0,\dots,K$ we obtain
\begin{align*}
    \mathbb{E} \left[ 
        \sum_{k=0}^K
        \left\langle F(\hat{x}_k),\hat{x}_k-u_K \right\rangle
    \right] \ge
    \mathbb{E} \left[ 
        \sum_{k=0}^K
        \left\langle F(u_K),\hat{x}_k-u_K \right\rangle
    \right] 
    = (K+1) \mathbb{E} \left[ \left\langle F(u_K),\overline{x}_K - u_K \right\rangle\right] .
\end{align*}
Because $\overline{x}_K - u_K = \frac{1}{2L} F(\overline{x}_K)$, by $L$-Lipschitzness of $F$ gives
\begin{align*}
    \mathbb{E} \left[ \left\langle F(u_K),\overline{x}_K - u_K \right\rangle\right] & = \frac{1}{2L} \mathbb{E} \left[ \left\langle F(u_K),F(\overline{x}_K) \right\rangle\right] \\
    & = \frac{1}{2L} \mathbb{E} \left[ \|F(\overline{x}_K)\|^2
        +
        \left\langle 
            F(u_K)-F(\overline{x}_K)
        ,
            F(\overline{x}_K)
         \right\rangle\right] \\
    & \geq
    \frac{1}{2L} \mathbb{E} \left[ 
        \|F(\overline{x}_K)\|^2
        -
        L\|u_K-\overline{x}_K\|
        \|F(\overline{x}_K)\|
    \right] =
    \frac{1}{4L} \mathbb{E} \left[ \left\| F(\overline{x}_K) \right\|^2\right] .
\end{align*}
Combining this with \eqref{eq:DecPolyakSEG-residual-upper}, dividing by $\frac{K+1}{4L}$ and plugging in $R = \frac{3D}{2}$ proves the desired bound.

Substituting this into \eqref{eq:DecPolyakSEG-direct-residual-bound} immediately yields \eqref{eq:DecPolyakSEG-direct-residual-rate}. 

Since Case~1 is evident, we consider Case~2 on {\color{PineGreen}\sf  DecPolyakSEG-LS}\xspace.
Observe that any $\gamma_k \le \frac{A}{L}$ satisfies the critical condition~\eqref{eqn:gamma-critical-condition-stochastic} by $L$-Lipschitzness of $F$. 
Thus, if $\gamma_{-1} \le \frac{A}{L}$ then we will have $\gamma_k = \frac{c_{-1}}{c_k} \gamma_{-1} = \frac{\gamma_{-1}}{\sqrt{k+1}}$.
Hence, we may assume $\gamma_{-1} > \frac{A}{L}$. 
Now at any iteration $k \ge 0$, let $\widetilde{\gamma}_k = \frac{c_{k-1}}{c_k}\gamma_{k-1}$ be the trial step-size where line-search starts from.
If $\widetilde{\gamma}_k\leq \frac{A}{L}$, it is accepted without backtracking.
Otherwise, the accepted step-size will be $\gamma_k = \beta^{-m_k} \widetilde{\gamma}_k$ for some $m_k \ge 1$, where $\beta^{-m_k + 1} \widetilde{\gamma}_k > \frac{A}{L}$,
which implies $\gamma_k > \frac{A}{\beta L} \ge \frac{A}{\sqrt{k+1} \beta L}$.
Together with the case $\gamma_{-1} \le \frac{A}{L}$, this proves that \eqref{eq:DecPolyakSEG-lower-envelope} holds with $\underline{\gamma} = \min\left\{ \gamma_{-1}, \frac{A}{\beta L} \right\}$.

<!-- proposition-4: body_md -->
\label{proposition:DecPolyakSEG_strong_monotone}
Let each $F_i$ be $L$-Lipschitz and $\mu$-strongly monotone, and let
$x_*$ satisfy $F(x_*)=0$. 
Let $\{x_k\}$ be generated by {\color{PineGreen}\sf  DecPolyakSEG}\xspace with $\gamma_k$ satisfying the stochastic critical condition \eqref{eqn:gamma-critical-condition-stochastic} for $k\ge 0$ with $A \in (0,1)$.
Then, almost surely, 
\begin{align}
    \sup_{k \ge 0} \left\| x_k-x_* \right\|^2
    \leq
    \max\left\{
        \left\| x_0-x_* \right\|^2,
        \frac{1}{\mu^2(1-A)}
        \underset{1\le i \le n}{\max} \left\|F_i(x_*)\right\|
    \right\} < \infty .
\label{eq:DecPolyakSEG-strong-monotone-stability}
\end{align}
Consequently, the extrapolated points $\hat{x}_k$
also stay bounded almost surely.

<!-- proof-9: body_md -->
First, note that since each $F_i$ is $L$-Lipschitz and $\mu$-strongly monotone,
every mini-batch operator $F_{\mathcal S_k}$ is also $L$-Lipschitz and
$\mu$-strongly monotone. 
By the choice of $\alpha_k$ in {\color{PineGreen}\sf  DecPolyakSEG}\xspace, we have $\alpha_k \left\| F_{\mathcal S_k}(\hat{x}_k) \right\|^2 \leq \left\langle F_{\mathcal S_k}(\hat{x}_k),x_k-\hat{x}_k \right\rangle$, so
\begin{align}
    \left\| x_{k+1}-x_* \right\|^2 
    &=
    \left\| x_k-x_* \right\|^2
    -2\alpha_k
    \left\langle F_{\mathcal S_k}(\hat{x}_k),x_k-x_* \right\rangle
    +\alpha_k^2\left\| F_{\mathcal S_k}(\hat{x}_k) \right\|^2
    \nonumber \\
    & \leq
    \left\| x_k-x_* \right\|^2
    -\alpha_k
    \left\langle F_{\mathcal S_k}(\hat{x}_k),x_k-\hat{x}_k \right\rangle
    -2\alpha_k
    \left\langle F_{\mathcal S_k}(\hat{x}_k),\hat{x}_k-x_* \right\rangle .
    \label{eqn:strong-monotone-localization-eq1}
\end{align}
Strong monotonicity of $F_{\mathcal S_k}$ and Young's inequality yield
\begin{align*}
    -2\left\langle F_{\mathcal S_k}(\hat{x}_k),\hat{x}_k-x_* \right\rangle
    &=
    -2\left\langle 
        F_{\mathcal S_k}(\hat{x}_k)-F_{\mathcal S_k}(x_*)
    ,
        \hat{x}_k-x_*
     \right\rangle
    -2\left\langle F_{\mathcal S_k}(x_*),\hat{x}_k-x_* \right\rangle
    \\
    &\leq
    -2\mu\left\| \hat{x}_k-x_* \right\|^2
    +\frac{1}{\mu}\left\| F_{\mathcal S_k}(x_*) \right\|^2
    +\mu\left\| \hat{x}_k-x_* \right\|^2
    \\
    &=
    -\mu\left\| \hat{x}_k-x_* \right\|^2
    +\frac{1}{\mu}\left\| F_{\mathcal S_k}(x_*) \right\|^2.
\end{align*}
Applying this to \eqref{eqn:strong-monotone-localization-eq1} and using $x_k-\hat{x}_k=\gamma_kF_{\mathcal S_k}(x_k)$, it follows that
\begin{align}
    \left\| x_{k+1}-x_* \right\|^2
    &\leq
    \left\| x_k-x_* \right\|^2
    -\alpha_k\mu\left\| \hat{x}_k-x_* \right\|^2
    +\frac{\alpha_k}{\mu}
    \left\| F_{\mathcal S_k}(x_*) \right\|^2
    -\alpha_k\gamma_k
    \left\langle 
        F_{\mathcal S_k}(\hat{x}_k)
    ,
        F_{\mathcal S_k}(x_k)
     \right\rangle .
\label{eq:DecPolyakSEG-strong-monotone-first-step}
\end{align}
Note that by Young's inequality
\begin{align*}
    -\left\| \hat{x}_k-x_* \right\|^2
    \leq
    -(1-A)\left\| x_k-x_* \right\|^2
    +\frac{1-A}{A}\left\| x_k-\hat{x}_k \right\|^2 
\end{align*}
holds, and strong monotonicity of $F_{\mathcal S_k}$ gives
\begin{align*}
    \mu\left\| x_k-\hat{x}_k \right\|^2
    &\leq
    \left\langle 
        F_{\mathcal S_k}(x_k)-F_{\mathcal S_k}(\hat{x}_k)
    ,
        x_k-\hat{x}_k
     \right\rangle
    =
    \gamma_k
    \left\langle 
        F_{\mathcal S_k}(x_k)
    ,
        F_{\mathcal S_k}(x_k)-F_{\mathcal S_k}(\hat{x}_k)
     \right\rangle
\end{align*}
Substituting these two inequalities into
\eqref{eq:DecPolyakSEG-strong-monotone-first-step} yields
\begin{align}
    \left\| x_{k+1}-x_* \right\|^2
    &\leq
    \left(1-(1-A)\alpha_k\mu\right)
    \left\| x_k-x_* \right\|^2
    \nonumber \\
    &\quad
    +\frac{\alpha_k\gamma_k}{A}
    \left(
        (1-A)\left\| F_{\mathcal S_k}(x_k) \right\|^2
        -
        \left\langle 
            F_{\mathcal S_k}(x_k)
        ,
            F_{\mathcal S_k}(\hat{x}_k)
         \right\rangle
    \right)
    +\frac{\alpha_k}{\mu}
    \left\| F_{\mathcal S_k}(x_*) \right\|^2
    \nonumber \\
    & \le \left(1-(1-A)\alpha_k\mu\right)
    \left\| x_k-x_* \right\|^2
    +
    \frac{\alpha_k}{\mu}
    \left\| F_{\mathcal S_k}(x_*) \right\|^2  
    \nonumber \\
    & \le 
    \left(1-(1-A)\alpha_k\mu\right)
    \left\| x_k-x_* \right\|^2
    +
    (1-A) \alpha_k \mu \frac{\left\| F_{\mathcal S_k}(x_*) \right\|^2}{\mu^2 (1-A)} 
    \label{eq:DecPolyakSEG-strong-monotone-recursion}
\end{align}
where the second last inequality follows from \cref{lemma:PolyakSEG-LS-key-properties}(b).
Applying this recursively, we obtain \eqref{eq:DecPolyakSEG-strong-monotone-stability}, provided that $(1-A) \alpha_k \mu \in [0,1]$.
This holds because
\begin{align}
    \mu\left\| x_k-\hat{x}_k \right\|^2
    &\leq
    \left\langle 
        F_{\mathcal S_k}(x_k)-F_{\mathcal S_k}(\hat{x}_k)
    ,
        x_k-\hat{x}_k
     \right\rangle
    \nonumber \\
    &\leq
    \left\|
        F_{\mathcal S_k}(x_k)-F_{\mathcal S_k}(\hat{x}_k)
    \right\|
    \left\|x_k-\hat{x}_k\right\|
    \leq
    A\left\|F_{\mathcal S_k}(x_k)\right\|
    \left\|x_k-\hat{x}_k\right\| 
    \label{eqn:DecPolyakSEG-strong-monotone-and-critical-condition}
\end{align}
by strong monotonicity of $F_{{\mathcal S}_k}$ and the critical condition, so substituting $x_k-\hat{x}_k = \gamma_k F_{\mathcal S_k}(x_k)$ yields $\mu \gamma_k \le A$ and \cref{lemma:PolyakSEG-LS-key-properties}(c) gives $\alpha_k\leq\frac{\gamma_k}{1-A}$.

Finally, by \eqref{eqn:DecPolyakSEG-strong-monotone-and-critical-condition} and $L$-Lipschitzness of $F_{{\mathcal S}_k}$ we have
\begin{align*}
    \left\|\hat{x}_k-x_*\right\|
    &\leq
    \left\|x_k-x_*\right\|+\left\|x_k-\hat{x}_k\right\|
    \\
    &\leq
    \left\|x_k-x_*\right\|
    +\frac{A}{\mu}\left\|F_{\mathcal S_k}(x_k)\right\|
    \\
    &\leq
    \left\|x_k-x_*\right\|
    +\frac{A}{\mu} \left( \left\|F_{\mathcal S_k}(x_k) - F_{{\mathcal S}_k}(x_*)\right\| + \left\|F_{{\mathcal S}_k}(x_*)\right\| \right)
    \\
    &\leq
    \left(1+\frac{AL}{\mu}\right)\left\|x_k-x_*\right\|
    +\frac{A}{\mu}\left\|F_{\mathcal S_k}(x_*)\right\|
    \\
    &\leq
    \left(1+\frac{AL}{\mu}\right)\left\|x_k-x_*\right\|
    +
    \frac{A}{\mu}
    \underset{1\le i \le n}{\max} \left\|F_i(x_*)\right\|
\end{align*}
which shows that $\hat{x}_k$ also stays bounded almost surely.
