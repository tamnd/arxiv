<!-- proposition-1: body_md -->
\label{prop:forward}
Let $\rho\in L^\infty(\mathbb{R}^3)$ satisfy
$0<\rho_-\le\rho\le\rho_+<\infty$.  For
$\bm{f}\in H^1(\mathbb{R}^3)^3$ and $\bm{g}\in L^2(\mathbb{R}^3)^3$, the Cauchy problem
\eqref{eq:main-time}--\eqref{eq:initial-data} has a unique solution
\[
 \bm{u}\in C([0,\infty);H^1(\mathbb{R}^3)^3)
       \cap C^1([0,\infty);L^2(\mathbb{R}^3)^3).
\]
The quadratic energy
\begin{equation}
\label{eq:elastic-energy}
 E_{\bm{u}}(t)
 :=\frac12\int_{\mathbb{R}^3}
 \left[
 \rho|\partial_t\bm{u}|^2
 +\lambda|\nabla\cdot\bm{u}|^2
 +2\mu|\varepsilon(\bm{u})|^2
 \right]dx
\end{equation}
is independent of $t$ and nonnegative.  Moreover,
\begin{equation}
\label{eq:H1-growth}
 \|\bm{u}(t)\|_{H^1(\mathbb{R}^3)}
 +\|\partial_t\bm{u}(t)\|_{L^2(\mathbb{R}^3)}
 \le C(1+t),
 \qquad t\ge0,
\end{equation}
where $C$ depends only on $\lambda,\mu,\rho_-,\rho_+$ and the norms of the initial data.
Consequently the $H^1$-valued Laplace transform exists for every $s>0$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Let $H_\rho=L^2(\mathbb{R}^3,\rho\,dx)^3$ and $V=H^1(\mathbb{R}^3)^3$.  By
\eqref{eq:form-Fourier}, the form $a$ is Hermitian, nonnegative, and closed on $V$ after
addition of the $H_\rho$ norm.  Let $\mathbb A_\rho$ be the associated nonnegative
self-adjoint operator in $H_\rho$.  Weakly,
$\mathbb A_\rho=\rho^{-1}\mathcal{A}_{\lambda,\mu}$ and
$D(\mathbb A_\rho^{1/2})=V$.  Define
\[
 m_t(\varsigma)=
 \begin{cases}
 \dfrac{\sin(t\sqrt\varsigma)}{\sqrt\varsigma},&\varsigma>0,\\[1mm]
 t,&\varsigma=0.
 \end{cases}
\]
The spectral theorem gives
\[
 \bm{u}(t)=\cos(t\mathbb A_\rho^{1/2})\bm{f}+m_t(\mathbb A_\rho)\bm{g}
\]
and the asserted energy solution.  The conserved quantity is
$\|\partial_t\bm{u}(t)\|_{H_\rho}^2+a(\bm{u}(t),\bm{u}(t))=2E_{\bm{u}}(t)$; nonnegativity follows from
\eqref{eq:form-Fourier}.  Energy conservation controls
$\|\partial_t\bm{u}(t)\|_{L^2}$ and $\|\nabla\bm{u}(t)\|_{L^2}$.  Since
\[
 \bm{u}(t)=\bm{f}+\int_0^t\partial_\upsilon\bm{u}(\upsilon)\,d\upsilon,
\]
the $L^2$ norm grows at most linearly, which proves \eqref{eq:H1-growth}.  The Laplace
integral converges because $e^{-st}(1+t)$ is integrable.

as seen from \eqref{eq:form-Fourier}.  We do not use pointwise positive definiteness of the

\label{lem:potential-orthogonality}
Let $\bm{h}\in L^2(\mathbb{R}^3)^3$ with $\operatorname{ess\,supp}\bm{h}\subset\overline Q$, and put
$\bm{V}=\bm{\Gamma}_0*\bm{h}$.  If
\[
 \bm{V}=\bm{0}\quad\text{in }\mathbb{R}^3\setminus\overline\Omega,
\]
then
\begin{equation*}
 \int_Q\bm{h}(x)\cdot\bm{v}(x)\,dx=0
 \qquad\forall\bm{v}\in\mathcal{H}_{\lambda,\mu}.
\end{equation*}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The whole-space potential satisfies
$\mathcal{A}_{\lambda,\mu}\bm{V}=\bm{h}$ in distributions.  Since
$d_Q:=\operatorname{dist}(\overline Q,\partial\Omega)>0$, the source vanishes in a full collar of
$\partial\Omega$.  Constant-coefficient elliptic regularity in a smaller collar gives
$\bm{V}\in H^2_{\mathrm{loc}}(\mathbb{R}^3)^3$ and $\bm{V}$ is smooth in a full neighborhood of
$\partial\Omega$.  The same smooth field vanishes on the exterior part of that neighborhood;
hence its displacement and traction traces vanish on $\partial\Omega$.  For every $\bm{v}\in\mathcal{H}_{\lambda,\mu}$, constant-coefficient elliptic regularity gives
$\bm{v}\in C^\infty(\mathbb{R}^3)^3$.  Since $\bm{V}$ is smooth in a neighborhood of the boundary,
the Green--Betti identity is legitimate for the pair $(\bm{V},\bm{v})$ (equivalently one may
use the weak traction identity and density).  Therefore
\begin{align*}
 \int_Q\bm{h}\cdot\bm{v}\,dx
 &=\int_{\Omega}(\mathcal{A}_{\lambda,\mu}\bm{V})\cdot\bm{v}\,dx\\
 &=\int_{\Omega}\bm{V}\cdot(\mathcal{A}_{\lambda,\mu}\bm{v})\,dx
   +\int_{\partial\Omega}
    \left[\bm{V}\cdot\bm{T}_{\nu}\bm{v}-(\bm{T}_{\nu}\bm{V})\cdot\bm{v}\right]dS\\
 &=0,
\end{align*}
because $\mathcal{A}_{\lambda,\mu}\bm{v}=0$ and both boundary traces of $\bm{V}$ vanish.

\label{lem:completeness}
Let
\[
 \bm{H}(x)=\bm{H}_0(x')\psi(x_3),
 \qquad
 \bm{H}_0\in L^2_c(D)^3,
 \quad
 \psi\in C_c^\infty(I)\setminus\{0\}.
\]
If
\begin{equation}
\label{eq:H-all-orthogonal}
 \int_{\mathbb{R}^3}\bm{H}(x)\cdot\bm{v}(x)\,dx=0
 \qquad\forall\bm{v}\in\mathcal{H}_{\lambda,\mu},
\end{equation}
then $\bm{H}_0=\bm{0}$ a.e. in $\mathbb{R}^2$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Fix $\eta=(\eta_1,\eta_2)\in\mathbb{R}^2\setminus\{\bm{0}\}$ and define
\begin{equation*}
 \zeta_+=(i\eta_1,i\eta_2,|\eta|),
 \qquad
 \zeta_-=(i\eta_1,i\eta_2,-|\eta|).
\end{equation*}
Both vectors are null.  For arbitrary
$\bm{a}_\pm\in\zeta_\pm^\perp$, the fields
$\bm{a}_\pm e^{\zeta_\pm\cdot x}$ are static Lam\'e solutions.  Substitution into
\eqref{eq:H-all-orthogonal} yields
\begin{equation*}
 \widehat{\bm{H}_0}(-\eta)\cdot\bm{a}_\pm\,
 \mathscr L_\psi(\pm|\eta|)=0.
\end{equation*}
If $|\eta|\in\mathcal S_\psi$, both profile factors are nonzero.  Hence
$\widehat{\bm{H}_0}(-\eta)$ is orthogonal to $\zeta_\pm^\perp$ for both signs, and therefore
the nondegeneracy of the complex bilinear product gives
\begin{equation}
\label{eq:polarization-intersection}
 \widehat{\bm{H}_0}(-\eta)
 \in\operatorname{span}\{\zeta_+\}
   \cap\operatorname{span}\{\zeta_-\}
 =\{\bm{0}\}.
\end{equation}
Since $\mathcal S_\psi$ is dense in $(0,\infty)$, the set
$\{\eta\in\mathbb{R}^2\setminus\{\bm{0}\}:|\eta|\in\mathcal S_\psi\}$ is dense in
$\mathbb{R}^2\setminus\{\bm{0}\}$.  Since
$\bm{H}_0$ is compactly supported and belongs to $L^2$, its Fourier transform is continuous;
thus it vanishes everywhere.  Fourier injectivity proves the claim.

<!-- lemma-3: body_md -->
\label{lem:normal-pencil}
For $\bm{a}\in\mathbb{C}^3$,
\begin{equation}
\label{eq:normal-symbol-action}
 \mathcal{L}_{\lambda,\mu}
 \left(\bm{a} e^{i\eta\cdot x'+rx_3}\right)
 =\mathscr P_\eta(r)\bm{a}\,e^{i\eta\cdot x'+rx_3}.
\end{equation}
Moreover,
\begin{equation}
\label{eq:normal-determinant}
 \det\mathscr P_\eta(r)
 =\mu^2(\lambda+2\mu)(r^2-\tau^2)^3.
\end{equation}
Assume $\lambda+\mu\ne0$.  Set $r_\pm=\pm\tau$ and
$\zeta_\pm=\zeta(r_\pm)$.  Each root $r_\pm$ has algebraic multiplicity three and
\[
 \ker\mathscr P_\eta(r_\pm)=\zeta_\pm^\perp,
 \qquad
 \dim\ker\mathscr P_\eta(r_\pm)=2.
\]
With
\begin{equation*}
 \mathfrak c_{\lambda,\mu}:=\frac{\lambda+3\mu}{\lambda+\mu},
 \qquad
 \bm{a}_0:=\zeta_\pm,
 \qquad
 \bm{a}_1:=-\mathfrak c_{\lambda,\mu}e_3,
\end{equation*}
one has the length-two chain relations
\begin{equation}
\label{eq:Jordan-chain-relations}
 \mathscr P_\eta(r_\pm)\bm{a}_0=\bm{0},
 \qquad
 \mathscr P_\eta(r_\pm)\bm{a}_1
 +\mathscr P_\eta'(r_\pm)\bm{a}_0=\bm{0}.
\end{equation}
Consequently,
\begin{equation}
\label{eq:generalized-static-mode}
 \bm{w}_{\eta,\pm}(x)
 :=\left(x_3\zeta_\pm-\mathfrak c_{\lambda,\mu}e_3\right)
 e^{i\eta\cdot x'+r_\pm x_3}
\end{equation}
belongs to $\mathcal{H}_{\lambda,\mu}$.  The two partial multiplicities at $r_\pm$ are
$2$ and $1$; in particular, no Jordan chain at this root has length greater than two.

If $\lambda+\mu=0$, then
$\mathscr P_\eta(r)=\mu(r^2-\tau^2)I_3$; each root has algebraic and geometric
multiplicity three and is semisimple.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Formula \eqref{eq:normal-symbol-action} follows by substituting the exponential ansatz into
the static Lam\'e equation.  Since
$\zeta(r)\cdot\zeta(r)=r^2-\tau^2$, the matrix determinant lemma gives, whenever
$r^2\ne\tau^2$,
\begin{align*}
 \det\mathscr P_\eta(r)
 &=[\mu(r^2-\tau^2)]^3
 \left(1+\frac{\lambda+\mu}{\mu}\right)\\
 &=\mu^2(\lambda+2\mu)(r^2-\tau^2)^3.
\end{align*}
Both sides are polynomials, so \eqref{eq:normal-determinant} holds for all $r$.
At $r=r_\pm$ and $\lambda+\mu\ne0$,
\[
 \mathscr P_\eta(r_\pm)
 =(\lambda+\mu)\zeta_\pm\zeta_\pm^\top,
\]
which has rank one and kernel $\zeta_\pm^\perp$.  The order of the zero in
\eqref{eq:normal-determinant} is three, proving the algebraic-multiplicity assertion.

The first relation in \eqref{eq:Jordan-chain-relations} follows from
$\zeta_\pm\cdot\zeta_\pm=0$.  Since
\[
 \mathscr P_\eta'(r)
 =2\mu rI_3+(\lambda+\mu)
 \left(e_3\zeta(r)^\top+\zeta(r)e_3^\top\right),
\]
one obtains
\[
 \mathscr P_\eta'(r_\pm)\bm{a}_0
 =(\lambda+3\mu)r_\pm\zeta_\pm
\]
and
\[
 \mathscr P_\eta(r_\pm)\bm{a}_1
 =-(\lambda+\mu)\mathfrak c_{\lambda,\mu}
   r_\pm\zeta_\pm
 =-(\lambda+3\mu)r_\pm\zeta_\pm.
\]
This proves the second chain relation.  For a quadratic matrix polynomial,
\[
 \mathscr P_\eta(\partial_3)
 \left[(x_3\bm{a}_0+\bm{a}_1)e^{r_\pm x_3}\right]
 =e^{r_\pm x_3}
 \left[x_3\mathscr P_\eta(r_\pm)\bm{a}_0
       +\mathscr P_\eta(r_\pm)\bm{a}_1
       +\mathscr P_\eta'(r_\pm)\bm{a}_0\right].
\]
Together with the horizontal exponential, this proves
\eqref{eq:generalized-static-mode}.  The leading coefficient of $\mathscr P_\eta$ is
$\operatorname{diag}(\mu,\mu,\lambda+2\mu)$ and is invertible, so the matrix polynomial is
regular.  Standard matrix-polynomial theory \cite{GohbergLancasterRodman2009} gives exactly
$\dim\ker\mathscr P_\eta(r_\pm)=2$ Jordan chains at $r_\pm$, whose positive lengths sum
to the algebraic multiplicity three.  The displayed relations produce a chain of length at
least two.  The two chain lengths are therefore exactly $2$ and $1$, and the length-two
chain cannot be extended.  The last assertion follows immediately when
$\lambda+\mu=0$.

\label{lem:multi-profile-completeness}
Let $\bm{M}\in\mathbb{R}^3\setminus\{\bm{0}\}$ and suppose
$\theta_1,\ldots,\theta_N\in L^2_c(D;\mathbb{R})$ and
\[
 \bm{H}(x)=\bm{M}\sum_{k=1}^N\theta_k(x')\psi_k(x_3).
\]
Assume
\begin{equation}
\label{eq:multi-profile-orthogonality}
 \int_{\mathbb{R}^3}\bm{H}(x)\cdot\bm{v}(x)\,dx=0
 \qquad\forall\bm{v}\in\mathcal{H}_{\lambda,\mu}.
\end{equation}
Then $\theta_1=\cdots=\theta_N=0$ a.e. in $\mathbb{R}^2$ under either of the following
conditions:
\begin{enumerate}
\item[(i)] $1\le N\le2$ and $\Psi=(\psi_1,\ldots,\psi_N)$ is two-sided Laplace
nondegenerate;
\item[(ii)] $1\le N\le4$, $\lambda+\mu\ne0$, and $\Psi$ is Hermite--Laplace
nondegenerate.
\end{enumerate}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Fix $\eta\in\mathbb{R}^2\setminus\{\bm{0}\}$, write $\tau=|\eta|$, and set
$\zeta_\pm=(i\eta_1,i\eta_2,\pm\tau)$.  The real vector $\bm{M}$ cannot belong to
$\operatorname{span}_{\mathbb{C}}\{\zeta_\pm\}$, since otherwise
$|\bm{M}|^2=\bm{M}\cdot\bm{M}=0$.  Hence there is
$\bm{a}_\pm\in\zeta_\pm^\perp$ with $\bm{M}\cdot\bm{a}_\pm\ne0$.  Testing
\eqref{eq:multi-profile-orthogonality} with
$\bm{a}_\pm e^{\zeta_\pm\cdot x}$ gives
\begin{equation}
\label{eq:profile-value-equations}
 \sum_{k=1}^N\widehat{\theta_k}(-\eta)
 \mathscr L_{\psi_k}(\pm\tau)=0.
\end{equation}
Under condition (i), these equations form
\[
 \mathscr V_\Psi(\tau)
 \begin{pmatrix}
  \widehat{\theta_1}(-\eta)\\[-1mm]
  \vdots\\[-1mm]
  \widehat{\theta_N}(-\eta)
 \end{pmatrix}=0.
\]
They force all Fourier coefficients to vanish whenever
$\tau\in\mathcal S_\Psi^{(0)}$.

Assume now condition (ii).  Testing with the generalized modes
\eqref{eq:generalized-static-mode} gives
\begin{align*}
0={}&(\bm{M}\cdot\zeta_\pm)
 \sum_{k=1}^N\widehat{\theta_k}(-\eta)
       \mathscr L'_{\psi_k}(\pm\tau)
 \\
&-\mathfrak c_{\lambda,\mu}M_3
 \sum_{k=1}^N\widehat{\theta_k}(-\eta)
       \mathscr L_{\psi_k}(\pm\tau).
\end{align*}
The second sum vanishes by \eqref{eq:profile-value-equations}.  Moreover,
\[
 \bm{M}\cdot\zeta_\pm=i\bm{M}'\cdot\eta\pm M_3\tau,
\qquad \bm{M}':=(M_1,M_2).
\]
If $M_3\ne0$, this factor is nonzero for every $\eta\ne\bm{0}$; if $M_3=0$, it vanishes only
on the line $\bm{M}'\cdot\eta=0$.  Hence, on an open dense set of horizontal frequencies,
\begin{equation*}
 \sum_{k=1}^N\widehat{\theta_k}(-\eta)
       \mathscr L'_{\psi_k}(\pm\tau)=0.
\end{equation*}
Together with \eqref{eq:profile-value-equations}, this is the system
\[
 \mathscr H_\Psi(\tau)
 \begin{pmatrix}
  \widehat{\theta_1}(-\eta)\\[-1mm]
  \vdots\\[-1mm]
  \widehat{\theta_N}(-\eta)
 \end{pmatrix}=0
\]
whenever $\tau\in\mathcal S_\Psi^{(1)}$ and the polarization factor is nonzero.

In case (i), the set
$\{\eta\ne\bm{0}:|\eta|\in\mathcal S_\Psi^{(0)}\}$ is dense in $\mathbb{R}^2$.  In case (ii),
intersecting the analogous radial set for $\mathcal S_\Psi^{(1)}$ with the complement of
the exceptional line $\bm{M}'\cdot\eta=0$ (when $M_3=0$) still leaves a dense subset of
$\mathbb{R}^2$.  The functions
$\theta_k$ are compactly supported in $L^2$, so their Fourier transforms are continuous.
They vanish on a dense set and hence everywhere.  Fourier injectivity proves the result.

Write $\Lambda(z)=\mathscr L_\varphi(z)$.  Then
\[
 \mathscr L_{\psi_k}(z)=e^{d_kz}\Lambda(z),
 \qquad
 \mathscr L'_{\psi_k}(z)=e^{d_kz}\big(d_k\Lambda(z)+\Lambda'(z)\big).
\]
For $i<j$,
\[
 \det\mathscr V_{(\psi_i,\psi_j)}(z)
 =2\sinh((d_i-d_j)z)\Lambda(z)\Lambda(-z)\not\equiv0,
\]
so every one- or two-profile subfamily is two-sided Laplace nondegenerate.

For the four-column Hermite matrix, expand the second and fourth rows by linearity.
Terms containing $\Lambda'(z)$ or $\Lambda'(-z)$ have two proportional rows.  Their
determinants vanish.  Factoring the remaining rows gives
\begin{equation}
\label{eq:translated-Hermite-factorization}
 \det\mathscr H_{(\psi_1,\ldots,\psi_4)}(z)
 =\Lambda(z)^2\Lambda(-z)^2D_d(z),
\end{equation}
where
\[
 D_d(z)
 :=\det
 \begin{pmatrix}
 e^{d_1z}&e^{d_2z}&e^{d_3z}&e^{d_4z}\\
 d_1e^{d_1z}&d_2e^{d_2z}&d_3e^{d_3z}&d_4e^{d_4z}\\
 e^{-d_1z}&e^{-d_2z}&e^{-d_3z}&e^{-d_4z}\\
 d_1e^{-d_1z}&d_2e^{-d_2z}&d_3e^{-d_3z}&d_4e^{-d_4z}
 \end{pmatrix}.
\]
Expand by two columns in the upper block and the complementary columns in the lower block.
The term with upper columns $\{3,4\}$ and lower columns $\{1,2\}$ is
\[
 \pm(d_4-d_3)(d_2-d_1)
 e^{(d_3+d_4-d_1-d_2)z}.
\]
For every other two-element set $S\ne\{3,4\}$,
\[
 \sum_{j\in S}d_j-\sum_{j\notin S}d_j
 <d_3+d_4-d_1-d_2.
\]
Thus this term has the unique largest exponential rate as $z\to+\infty$.  Hence
\[
 D_d(z)
 =\pm(d_4-d_3)(d_2-d_1)
 e^{(d_3+d_4-d_1-d_2)z}(1+o(1)).
\]
Hence the determinant in \eqref{eq:translated-Hermite-factorization} is not identically
zero.  A nontrivial entire function cannot vanish on the whole positive real axis, so the
full four-column matrix is invertible on an open dense set of positive radii.  At each such
radius, every subfamily of columns is linearly independent.  This proves the claim.

\label{prop:general-density-obstruction}
For every $\bm{M}\in\mathbb{R}^3\setminus\{\bm{0}\}$, there are infinitely many linearly independent
$\omega\in C_c^\infty(Q)$ such that
\begin{equation}
\label{eq:reduced-kernel}
 \int_Q\omega(x)\bm{M}\cdot\bm{v}(x)\,dx=0
 \qquad\forall\bm{v}\in\mathcal{H}_{\lambda,\mu}.
\end{equation}
<!-- end proposition-4 -->
<!-- proof-7: body_md -->
If $\bm{v}\in\mathcal{H}_{\lambda,\mu}$, then taking the divergence of the Lam\'e equation and then
applying $\Delta$ gives $\Delta^2\bm{v}=0$.  Thus $\bm{M}\cdot\bm{v}$ is biharmonic.  For any
$\omega_0\in C_c^\infty(Q)$ with $\Delta^2\omega_0\not\equiv0$, set
$\omega=\Delta^2\omega_0$.  Two
integrations by parts give
\[
 \int_Q\omega\,\bm{M}\cdot\bm{v}\,dx
 =\int_Q\omega_0\,\Delta^2(\bm{M}\cdot\bm{v})\,dx=0.
\]
Choosing functions $\omega_0$ with mutually disjoint supports yields an infinite linearly
independent family.

<!-- lemma-5: body_md -->
\label{lem:kernel-expansion}
There are $s_0>0$ and $C_R>0$ such that, for
$0<s<s_0$ and $0<|x|\le2R$,
\begin{equation*}
 \bm{\Gamma}_s(x)
 =\bm{\Gamma}_0(x)-s\gamma_1I_3+s^2\bm{K}_2(x)+\bm{E}_s(x),
 \qquad
 |\bm{E}_s(x)|\le C_Rs^3|x|^2.
\end{equation*}
Consequently, if $\bm{h}\in L^2(\mathbb{R}^3)^3$ and
$\operatorname{ess\,supp}\bm{h}\subset\overline Q$, then
\begin{equation}
\label{eq:operator-expansion}
 \bm{\Gamma}_s*\bm{h}
 =\bm{\Gamma}_0*\bm{h}-s\gamma_1\int_Q\bm{h}(y)\,dy
  +s^2\bm{K}_2*\bm{h}
  +O_{L^\infty(B_R)}(s^3\|\bm{h}\|_{L^1(Q)}).
\end{equation}
Moreover, uniformly for $0<s<s_0$,
\begin{align}
\label{eq:L2-Linf}
 \|\bm{\Gamma}_s*\bm{h}\|_{L^\infty(B_R)}
 &\le C_R\|\bm{h}\|_{L^2(Q)},\\
\label{eq:B-bound}
 \|\bm{\Gamma}_s*(q\bm{v})\|_{L^\infty(B_R)}
 &\le C_R\|q\|_{L^\infty(Q)}\|\bm{v}\|_{L^\infty(B_R)},
\end{align}
whenever the displayed expressions are defined and the factors are supported in
$\overline Q$.  Finally,
\begin{equation}
\label{eq:first-order-operator-bound}
 \left\|\bm{\Gamma}_s*\bm{h}-\bm{\Gamma}_0*\bm{h}
       +s\gamma_1\int_Q\bm{h}(y)\,dy\right\|_{L^\infty(B_R)}
 \le C_Rs^2\|\bm{h}\|_{L^2(Q)}.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-8: body_md -->
The expansion and the differentiated remainder are proved in Appendix~\ref{app:kernel}.
The only singular term is the Kelvin tensor, of order $|x|^{-1}$.  Thus
$|\bm{\Gamma}_s(x)|\le C_R|x|^{-1}$ on $0<|x|\le2R$.  Since $|x|^{-2}$ is locally
integrable in three dimensions, Cauchy--Schwarz gives \eqref{eq:L2-Linf}; integration
against $|x-y|^{-1}$ gives \eqref{eq:B-bound}.  Subtracting the first two terms of
\eqref{eq:operator-expansion}, using the local boundedness of $\bm{K}_2$, and applying
$L^1(Q)\hookleftarrow L^2(Q)$ gives \eqref{eq:first-order-operator-bound}.

<!-- proposition-5: body_md -->
\label{prop:field-expansion}
Let $q\in L^\infty(\mathbb{R}^3)$ satisfy $\operatorname{ess\,supp} q\subset\overline Q$ and assume
\begin{equation}
\label{eq:field-positive-density}
 \rho_-\le\rho_0+q\le\rho_+.
\end{equation}
Let $\bm{F},\bm{G}\in L^2_c(Q)^3$ and fix $R$ with $\overline\Omega\subset B_R$.  For all
sufficiently small $s>0$, the restriction of \eqref{eq:Lippmann-Schwinger} to $B_R$ has a
unique fixed point in $L^\infty(B_R)^3$, equal to the variational solution.  It satisfies
\begin{equation}
\label{eq:field-expansion}
 \widetilde{\bm{u}}
 =\bm{U}_0+s\bm{U}_1+s^2\bm{U}_2+O_{L^\infty(B_R)}(s^3),
\end{equation}
where
\begin{equation*}
 \bm{U}_0=\bm{\Gamma}_0*\bm{G},
 \qquad
 \bm{U}_1=\bm{\Gamma}_0*\bm{F}-\gamma_1\int_Q\bm{G}(y)\,dy,
\end{equation*}
and
\begin{equation}
\label{eq:U2}
 \bm{U}_2
 =\bm{K}_2*\bm{G}-\gamma_1\int_Q\bm{F}(y)\,dy-\bm{\Gamma}_0*(q\bm{U}_0).
\end{equation}
More precisely, for every $K_q,K_F,K_G>0$, there are
\[
 s_*=s_*(R,Q,\lambda,\mu,\rho_0,K_q)>0,
 \qquad
 C_*=C_*(R,Q,\lambda,\mu,\rho_0,K_q,K_F,K_G)>0,
\]
such that the remainder in \eqref{eq:field-expansion} is bounded by $C_*s^3$ whenever
\[
 \|q\|_{L^\infty(Q)}\le K_q,\qquad
 \|\bm{F}\|_{L^2(Q)}\le K_F,\qquad
 \|\bm{G}\|_{L^2(Q)}\le K_G,
 \qquad 0<s<s_*.
\]
<!-- end proposition-5 -->
<!-- proof-9: body_md -->
Define on $L^\infty(B_R)^3$
\[
 \mathcal{B}_s\bm{v}:=\bm{\Gamma}_s*(q\bm{v}).
\]
Lemma~\ref{lem:kernel-expansion} gives a uniform operator bound for $\mathcal{B}_s$.  Hence
$\mathrm{Id}+s^2\mathcal{B}_s$ is invertible for small $s$.  The exact identity
\begin{equation}
\label{eq:inverse-expansion}
 (\mathrm{Id}+s^2\mathcal{B}_s)^{-1}-(\mathrm{Id}-s^2\mathcal{B}_s)
 =s^4\mathcal{B}_s^2(\mathrm{Id}+s^2\mathcal{B}_s)^{-1}
\end{equation}
shows directly that the remainder is $O(s^4)$ uniformly in operator norm on every bounded
coefficient class.  The kernel expansion gives
\begin{align}
 \bm{\Gamma}_s*(\bm{G}+s\bm{F})
 ={}&\bm{\Gamma}_0*\bm{G}
 +s\left(\bm{\Gamma}_0*\bm{F}-\gamma_1\int_Q\bm{G}(y)\,dy\right)
 \nonumber\\
 &+s^2\left(\bm{K}_2*\bm{G}-\gamma_1\int_Q\bm{F}(y)\,dy\right)
 +O_{L^\infty(B_R)}(s^3).
\label{eq:free-expansion}
\end{align}
Furthermore, \eqref{eq:first-order-operator-bound} implies
$\|\mathcal{B}_s-\mathcal{B}_0\|=O(s)$, where
$\mathcal{B}_0\bm{v}=\bm{\Gamma}_0*(q\bm{v})$.  Multiplying
\eqref{eq:inverse-expansion} and \eqref{eq:free-expansion}, one has
\begin{equation*}
 -s^2\mathcal{B}_s\bm{U}_0
 =-s^2\mathcal{B}_0\bm{U}_0-s^2(\mathcal{B}_s-\mathcal{B}_0)\bm{U}_0
 =-s^2\mathcal{B}_0\bm{U}_0+O_{L^\infty(B_R)}(s^3),
\end{equation*}
while $-s^3\mathcal{B}_s\bm{U}_1=O_{L^\infty(B_R)}(s^3)$ and the inverse remainder in
\eqref{eq:inverse-expansion} is $O(s^4)$ in operator norm.  Thus the only order-$s^2$
contribution from the inverse is $-s^2\mathcal{B}_0\bm{U}_0$, which proves
\eqref{eq:field-expansion}--\eqref{eq:U2}.

We record the uniformity because it is needed in the difference expansion below.  If
$\|q\|_{L^\infty(Q)}\le K_q$, then \eqref{eq:B-bound} gives
$\|\mathcal{B}_s\|\le C_RK_q$ for $0<s<s_0$.  Hence $s_*$ may be chosen so that
$s_*^2C_RK_q\le1/2$.  The convolution estimates yield
\[
 \|\bm{U}_0\|_{L^\infty(B_R)}\le C_RK_G,
 \qquad
 \|\bm{U}_1\|_{L^\infty(B_R)}\le C_R(K_F+K_G),
\]
and, using \eqref{eq:U2},
\[
 \|\bm{U}_2\|_{L^\infty(B_R)}
 \le C_R(K_G+K_F+K_qK_G).
\]
The remainder in the free expansion is bounded by $C_Rs^3(K_F+K_G)$, while the omitted
terms in the Neumann product contain either this factor or at least $s^4\|\mathcal{B}_s\|^2$.
Thus all order-$s^3$ contributions are bounded by
$C_*s^3$ with $C_*$ depending only on the quantities stated in the proposition.  The
positivity hypothesis \eqref{eq:field-positive-density} identifies the fixed point with the
physical variational solution through Appendix~\ref{app:variational}.

fields have the same Dirichlet trace on $\partial\Omega$.  Their difference $\bm{W}$ satisfies
\[
 (\mathcal{A}_{\lambda,\mu}+\rho_0s^2)\bm{W}=\bm{0}

<!-- lemma-7: body_md -->
\label{lem:difference-expansion}
Fix $R>0$ with $\overline\Omega\subset B_R$ and numbers $K_q,K_F,K_G>0$.  There exist
$s_R=s_R(R,Q,\lambda,\mu,\rho_0,K_q)>0$ and
$C_R=C_R(R,Q,\lambda,\mu,\rho_0,K_q,K_F,K_G)>0$ such that, whenever
\[
 \|q_j\|_{L^\infty(Q)}\le K_q,\qquad
 \|\bm{F}\|_{L^2(Q)}\le K_F,\qquad
 \|\bm{G}\|_{L^2(Q)}\le K_G,
\]
one has
\begin{align}
 \widetilde{\bm{u}}_1-\widetilde{\bm{u}}_2
 ={}&-s^2\bm{\Gamma}_0*(\delta\rho\,\bm{U}_0)
 \nonumber\\
 &-s^3\left[
 \bm{\Gamma}_0*(\delta\rho\,\bm{U}_1)
 -\gamma_1\int_Q\delta\rho(y)\bm{U}_0(y)\,dy
 \right]
 +\bm{R}_R(s),
\label{eq:difference-s3}
\end{align}
where
\[
 \|\bm{R}_R(s)\|_{L^\infty(B_R)}\le C_Rs^4,
 \qquad 0<s<s_R.
\]
<!-- end lemma-7 -->
<!-- proof-13: body_md -->
The free term in \eqref{eq:Lippmann-Schwinger} is common to both configurations.  Hence
\begin{equation}
\label{eq:exact-difference}
 \widetilde{\bm{u}}_1-\widetilde{\bm{u}}_2
 =-s^2\bm{\Gamma}_s*
  \left(q_1\widetilde{\bm{u}}_1-q_2\widetilde{\bm{u}}_2\right).
\end{equation}
By Proposition~\ref{prop:field-expansion}, uniformly under the bounds in the statement,
\[
 \widetilde{\bm{u}}_j
 =\bm{U}_0+s\bm{U}_1+s^2\bm{U}_{2,j}
  +O_{L^\infty(B_R)}(s^3),
\]
where
\[
 \bm{U}_{2,j}
 =\bm{K}_2*\bm{G}-\gamma_1\int_Q\bm{F}(y)\,dy-\bm{\Gamma}_0*(q_j\bm{U}_0).
\]
Consequently,
\begin{equation}
\label{eq:difference-source-expansion}
 q_1\widetilde{\bm{u}}_1-q_2\widetilde{\bm{u}}_2
 =\bm{h}_0+s\bm{h}_1+s^2\bm{Z}_s,
 \qquad
 \bm{h}_0:=\delta\rho\,\bm{U}_0,
 \quad
 \bm{h}_1:=\delta\rho\,\bm{U}_1,
\end{equation}
with
\[
 \bm{Z}_s=q_1\bm{U}_{2,1}-q_2\bm{U}_{2,2}+O_{L^2(Q)}(s).
\]
The bounds in Proposition~\ref{prop:field-expansion} imply
\begin{equation}
\label{eq:rs-uniform-bound}
 \sup_{0<s<s_R}\|\bm{Z}_s\|_{L^2(Q)}
 \le C(R,Q,\lambda,\mu,\rho_0,K_q,K_F,K_G).
\end{equation}
In the same way, $\|\bm{h}_0\|_{L^2(Q)}+\|\bm{h}_1\|_{L^2(Q)}$ is bounded by a constant with
the same parameter dependence.

Apply \eqref{eq:first-order-operator-bound} to the source in
\eqref{eq:difference-source-expansion}.  We obtain
\begin{align*}
 \bm{\Gamma}_s*(\bm{h}_0+s\bm{h}_1+s^2\bm{Z}_s)
 ={}&\bm{\Gamma}_0*(\bm{h}_0+s\bm{h}_1+s^2\bm{Z}_s)\\
 &-s\gamma_1\int_Q(\bm{h}_0(y)+s\bm{h}_1(y)+s^2\bm{Z}_s(y))\,dy
 +\bm{E}_s^{\sharp},
\end{align*}
where, by \eqref{eq:first-order-operator-bound} and the uniform bound
\eqref{eq:rs-uniform-bound}, the remainder satisfies
\[
 \|\bm{E}_s^{\sharp}\|_{L^\infty(B_R)}
 \le C_Rs^2\|\bm{h}_0+s\bm{h}_1+s^2\bm{Z}_s\|_{L^2(Q)}
 \le C_Rs^2.
\]
Moreover, \eqref{eq:L2-Linf} and the boundedness of $Q$ give
\[
 s^2\|\bm{\Gamma}_0*\bm{Z}_s\|_{L^\infty(B_R)}
 +s^2\left|\int_Q\bm{h}_1(y)\,dy\right|
 +s^3\left|\int_Q\bm{Z}_s(y)\,dy\right|
 \le C_Rs^2.
\]
Therefore
\[
 \bm{\Gamma}_s*
 \left(q_1\widetilde{\bm{u}}_1-q_2\widetilde{\bm{u}}_2\right)
 =\bm{\Gamma}_0*\bm{h}_0
 +s\left(\bm{\Gamma}_0*\bm{h}_1-\gamma_1\int_Q\bm{h}_0(y)\,dy\right)
 +O_{L^\infty(B_R)}(s^2),
\]
with a constant depending only on the parameters displayed in the lemma.  Multiplying by
the prefactor $-s^2$ in \eqref{eq:exact-difference} proves
\eqref{eq:difference-s3} and the uniform $O(s^4)$ bound.
