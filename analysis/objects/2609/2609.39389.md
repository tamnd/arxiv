<!-- theorem-2: body_md -->
\label[t]{t:main1}
    Let $d \in \mathbb{N}_{\ge 1}$, $\epsilon>0$, and $\Delta : \mathcal{A}_d(\mathbb{C}) \to \mathbb{R}^N$ an Almgren embedding.
    There is a reparameterization map $t=t(\epsilon,\Delta) : C^{d-1,1}(\overline I,\mathbb{C}^d) \to \operatorname{Lip}(\mathbb{R},\overline I)$, 
    $a \mapsto t_a$, such that the following properties hold for all $a \in C^{d-1,1}(\overline I,\mathbb{C}^d)$:
    \begin{enumerate}
        \item $t_a : \mathbb{R} \to \overline I$ is surjective, increasing, and strictly increasing on $t_a^{-1}(I)$;
        \item $t_a$ is globally Lipschitz with Lipschitz constant $\operatorname{Lip}(t_a) \le \frac{1}{\epsilon}$; 
        \item the composite $\Lambda_a \circ t_a : \mathbb{R} \to \mathcal{A}_d(\mathbb{C})$ is globally Lipschitz with $\operatorname{Lip}(\Lambda_a \circ t_a) \le 1$.
    \end{enumerate}
    The induced map 
    \begin{align} \label{eq:rs}
        C^d(\overline I,\mathbb{C}^d) &\to W^{1,q}_{\operatorname{loc}}(\mathbb{R}) \times  W^{1,q}_{\operatorname{loc}}(\mathbb{R},\mathcal{A}_d(\mathbb{C})) 
         \\
        a &\mapsto (t_a, \Lambda_a \circ t_a)
    \end{align}
    is continuous, for all $q \in [1,\infty)$.

\label[t]{t:seq}
    Let $d \in \mathbb{N}_{\ge 1}$, $\epsilon>0$, $\Delta : \mathcal{A}_d(\mathbb{C}) \to \mathbb{R}^N$ an Almgren embedding, and $t=t(\epsilon,\Delta)$ 
    the reparameterization map from \Cref{t:main1}.
    If $a_n \to a$ in $C^d(\overline I,\mathbb{C}^d)$, then:
    \begin{enumerate}
        \item $t_{a_n} \to t_a$ uniformly on $\mathbb{R}$ and in $W^{1,q}_{\operatorname{loc}}(\mathbb{R})$ for all $q \in [1,\infty)$.
        \item $\Lambda_{a_n} \circ t_{a_n} \to \Lambda_a \circ t_a$ uniformly on $\mathbb{R}$ and in $W^{1,q}_{\operatorname{loc}}(\mathbb{R},\mathcal{A}_d(\mathbb{C}))$ for all $q \in [1,\infty)$.
        \item Assume that $\lambda_n : I \to \mathbb{C}^d$ is a continuous parameterization of the roots of $P_{a_n}$ and $\lambda_n$ converges in $C^0(\overline I,\mathbb{C}^d)$ 
            to a continuous parameterization $\lambda$ of the roots of $P_a$. 
            Then, for each $\epsilon >0$, there exist $t,t_n \in \operatorname{Lip}(\mathbb{R},\overline I)$ with Lipschitz constants $\le \frac{1}{\epsilon}$ such that 
            $\lambda\circ t, \lambda_n \circ t_n \in \operatorname{Lip}(\mathbb{R},\mathbb{C}^d)$ with Lipschitz constants $\le 1$ and
            \begin{itemize}
                \item $t_n \to t$ uniformly on $\mathbb{R}$ and in $W^{1,q}_{\operatorname{loc}}(\mathbb{R})$ for all $q \in [1,\infty)$,
                \item $\lambda_{n} \circ t_n \to \lambda \circ t$ uniformly on $\mathbb{R}$ and in $W^{1,q}_{\operatorname{loc}}(\mathbb{R},\mathbb{C}^d)$ for all $q \in [1,\infty)$.
            \end{itemize}
            Here $t,t_n$ are independent of $\Delta$.
    \end{enumerate}

\label[l]{l:rep}
    Let $f,f_n \in AC^1((a,b),X)$, for $n \ge 1$, be such that $|f_n'| \to |f'|$ in $L^1((a,b))$. 
    For fixed $\epsilon>0$, let $s=s_{\epsilon,f}$, $s_n=s_{\epsilon,f_n}$, $t=t_{\epsilon,f}$, and $t_n=t_{\epsilon,f_n}$.
    Then:
    \begin{enumerate}
    \item $s_n \to s$ uniformly on $[a,b]$;
    \item $s_n' \to s'$ in $L^1((a,b))$;
    \item $t_n \to t$ uniformly on $\mathbb{R}$.
    \end{enumerate}

<!-- proof-1: body_md -->
(1) follows from 
    \begin{align}
        \MoveEqLeft  |s(x)  - s_n(x)| 
        = \Big|\int_a^x |f'|(t) - |f_n'|(t) \, dt \Big| 
        \le \int_a^b \big| |f'|(t) - |f_n'|(t) \big| \, dt.
    \end{align}
    
    (2) is evident, by definition.

    (3) 
    We will only use monotonicity and (1). By (1), $L_n = s_n(b) \to s(b) = L$.
    Define, for  small $\delta>0$,  
    \begin{equation}
        g(\delta) := \min\Big\{ \min_{x \in [a,b-\delta]} (s(x+\delta) - s(x)), \min_{x \in [a+\delta,b]} (s(x) - s(x-\delta))\Big\}.
    \end{equation}
    Then $g(\delta)>0$ because $s$ is strictly increasing.
    Fix $\delta < \frac{b-a}{2}$.
    By (1), there is $n_0 \ge 1$ such that, for all $n \ge n_0$,
    \begin{equation} \label{eq:C0}
        \|s_{n} - s\|_{C^0([a,b])} < \frac{g(\delta)}{2}
    \end{equation}
    and $L_n \in (s(b-\delta),L+\delta)$.
    
    We first show that, for all $y \in [0,s(b-\delta)]$ and $n \ge n_0$, 
    \begin{equation} \label{eq:cl}
       |t(y)-t_n(y)| < \delta. 
    \end{equation}
    The remaining $y$ will be considered at the end of the proof.
    
    Let $y \in [0,s(b-\delta)]$ and $x := t(y) \in [a,b-\delta]$. 


    
    If $x \in [a+\delta,b-\delta]$, then, by \eqref{eq:C0}, for $n\ge n_0$,
    \begin{align} \label{eq:u1}
        s_{n}(x+\delta) &> s(x+\delta) -  \tfrac{g(\delta)}{2} \ge \frac{s(x+\delta)+s(x)}{2} \ge s(x) = y,
        \\ \label{eq:u2}
        s_{n}(x-\delta) &< s(x-\delta) +  \tfrac{g(\delta)}{2} \le \frac{s(x)+s(x-\delta)}{2} \le s(x) = y.
    \end{align}
    Therefore, there is a unique $x' \in (x-\delta,x+\delta)$ such that 
    $s_{n}(x') = y$. As $y < L_n$, 
    we conclude $t_n(y)=x'$ and
    \begin{equation}
        |t(y) - t_{n}(y)|= |x -x'| < \delta.
    \end{equation}

    If $x \in [a,a+ \delta)$, then \eqref{eq:u1} still holds. Thus $s_n(x+\delta)>y$ and $x+\delta > t_{n}(y)$.
    Since $x- t_{n}(y) < a+\delta - t_{n}(y) \le \delta$, 
    we conclude that 
    \begin{equation}
        |t(y) - t_{n}(y)|= |x - t_{n}(y)| < \delta. 
    \end{equation}
    Thus \eqref{eq:cl} is proved.


    If $y \in (s(b-\delta),L+\delta)$, then as $t$ and $t_n$ are $\frac{1}{\epsilon}$-Lipschitz on $\mathbb{R}$ and $t(L) = b = t_n(L_n)$,
    \begin{align}
        |t(y)-t_n(y)| &\le  |t(y)-t(L)| + |t_n(L_n)-t_n(y)|\le 
        \frac{|y-L| + |L_n - y|}{\epsilon} 
        \\
                      &\le \frac{2\,|y-L| + |L - L_n|}{\epsilon} \le \frac{3}{\epsilon} \max\{\delta, s(b)-s(b-\delta)\}, 
    \end{align}
    for all $n \ge n_0$.
    Since $s$ is uniformly continuous and 
    $t(y) = t_n(y)$ for all $n \ge n_0$ if $y<0$ or $y\ge L+\delta$,  
    we conclude that $t_n \to t$ uniformly on $\mathbb{R}$.

<!-- proof-2: body_md -->
Since $f_n \to f$ in $W^{1,1}((a,b),\mathbb{R}^N)$, after passing to a subsequence, 
$f_n' \to f'$ almost everywhere in $(a,b)$. It follows that 
\begin{equation}
    \frac{f_n'}{\epsilon + \|f_n'\|_2} \to \frac{f'}{\epsilon + \|f'\|_2} \quad \text{ almost everywhere in } (a,b).
\end{equation}

By the generalized dominated convergence theorem (see e.g.\ \cite[p.~59]{Folland99}), we conclude that 
\begin{equation} \label{eq:l1c}
    \frac{f_n'}{\epsilon + \|f_n'\|_2 } \to \frac{f'}{\epsilon + \|f'\|_2 }\quad \text{ in } L^1((a,b),\mathbb{R}^N). 
\end{equation}
We proved \eqref{eq:l1c} for a subsequence, but this implies its validity for the original sequence.


    Let us put 
    \[
        \alpha_n := \frac{f_n'}{\epsilon + \|f_n'\|_2 }, \quad \alpha := \frac{f'}{\epsilon + \|f'\|_2 }.
    \]
    Then $\alpha_n \to \alpha$ in $L^1((a,b),\mathbb{R}^N)$, by \eqref{eq:l1c}. 
    We have $\hat f' = \alpha \circ t$ almost everywhere in $(0,L)$, by \eqref{eq:hld1},
    and $\hat f'(y) = 0$ if $y<0$ or $y>L$.
    By \eqref{eq:hld2}, $\hat f_n' = \alpha_n \circ t_n$ almost everywhere in $(0,L_n)$
    and $\hat f_n'(y) = 0$ if $y<0$ or $y>L_n$.


    Let $\rho,\sigma>0$ be fixed.
    We have to show that there exists $n_0 \ge 1$ such that 
    \begin{equation} \label{eq:ats}
        |\{y \in \mathbb{R} : \|\hat f_n'(y) - \hat f'(y)\|_2 \ge \rho\}| \le \sigma, \quad n \ge n_0.
    \end{equation}
    
    We claim that it suffices to show \eqref{eq:ats} for $y \in (0,L-\frac{\sigma}{4})$ (assuming of course that $\sigma$ is small enough).
    Indeed,
    if $y < 0$ then $\hat f_n'(y) = \hat f'(y) = 0$ for all $n$. 
    On the other hand, since $L_{n} \to L$, there is $n_1 \ge 1$ such that for all $n \ge n_1$ 
    we have $L_{n} \in (L - \frac{\sigma}{4},L + \frac{\sigma}{4})$ and hence
    $\hat f_n'(y) = \hat f'(y) = 0$ for all $y> L +\tfrac{\sigma}{4}$. 
    Thus, for $n \ge n_1$, the left-hand side of \eqref{eq:ats} equals
    \[
        |\{y \in (0,L + \tfrac{\sigma}{4}) : \|\hat f_n'(y) - \hat f'(y)\|_2 \ge \rho\}|
    \]
    so that, in order to establish \eqref{eq:ats}, it suffices to show
    \begin{equation} \label{eq:ats2}
        |\{y \in (0,L - \tfrac{\sigma}{4}) : \|\alpha_n(t_{n}(y)) - \alpha(t(y))\|_2 \ge \rho\}| \le \frac{\sigma}{2},
    \end{equation}
    for all sufficiently large $n$.
    Note that, for all $n \ge n_1$, the restriction of $t_{n}$ to $[0,L-\tfrac{\sigma}{4}]$ is invertible 
    with inverse $s_{n}|_{[a,t_{n}(L - \tfrac{\sigma}{4})]}$.


    Since $s_{n}' \to s'$ in $L^1((a,b))$ (see \Cref{l:rep}), the set $\{s_{n}' : n\ge 1\}\cup \{s'\}$ is uniformly integrable, i.e., 
    there exists $\tau>0$ such that, for all $n \ge 1$, 
    \begin{equation} \label{eq:aui}
        |s_{n}(E)| =  \int_E s_{n}'(x)\, dx \le \frac{\sigma}{6} \quad \text{ and } \quad |s(E)|\le \frac{\sigma}{6}  \quad \text{ if } |E|\le \tau.
    \end{equation}
    
    By Lusin's theorem, there exists a compact $K \subseteq [a,b]$ with $|[a,b] \setminus K| \le \tau$  
    such that $\alpha|_K$ is continuous. By the Tietze--Dugundji theorem, there exists a continuous function $g : [a,b]\to \mathbb{C}^d$ 
    such that $g|_K = \alpha|_K$.

    For $y \in (0,L- \tfrac{\sigma}{4})$, consider
    \begin{align*}
        \alpha_n(t_{n}(y)) - \alpha(t(y)) &= A_n(y) + B_n(y) + C_n(y) + D(y),   
    \end{align*}
    where 
    \begin{align*}
        A_n(y) &:= \alpha_n(t_{n}(y)) - \alpha(t_{n}(y)),
        \\
        B_n(y) &:= \alpha(t_{n}(y)) - g(t_{n}(y)),
        \\
        C_n(y) &:= g(t_{n}(y)) - g(t(y)),
        \\
        D(y) &:= g(t(y)) - \alpha(t(y)).
    \end{align*}
    It follows that, if 
\begin{equation}
    \|A_n(y)\|_2 < \tfrac{\rho}{2},\quad  B_n(y) = D(y)=0,\quad \|C_n(y)\|_2 < \tfrac{\rho}{2},
\end{equation}
then 
    \begin{equation}
         \|\alpha_n(t_{n}(y)) - \alpha(t(y)) \|_2 < \rho.
    \end{equation}
    
    We have $\{y \in (0,L-\tfrac{\sigma}{4}) : \|A_n(y)\|_2 \ge \tfrac{\rho}{2}\} = s_{n}(F_n)$, where 
    \[
        F_n := \{x \in (a,t_{n}(L-\tfrac{\sigma}{4})) : \|\alpha_n(x) - \alpha(x)\|_2 \ge \tfrac{\rho}{2}\}.
    \]
    Since $\alpha_n \to \alpha$ in $L^1((a,b),\mathbb{R}^N)$ and
    \begin{equation}
        \int_a^b \|\alpha_n(x) - \alpha(x)\|_2 \, dx \ge \int_{F_n} \|\alpha_n(x) - \alpha(x)\|_2 \, dx \ge \tfrac{\rho}{2} \, |F_n|,
    \end{equation}
    we have $|F_n| \le \tau$ for all sufficiently large $n$, so that by \eqref{eq:aui}, 
    \begin{equation}
        |\{y \in (0,L-\tfrac{\sigma}{4}) : \|A_n(y)\|_2 \ge \tfrac{\rho}{2}\}| = |s_{n}(F_n)| 
        \le \frac{\sigma}{6}.
    \end{equation}

    Next we observe that $B_n(y) = 0$ if $y \in s_{n}(K)$ and that $|s_{n}([a,b]\setminus K)| \le \tfrac{\sigma}{6}$, 
    by \eqref{eq:aui}.
    Similarly, $D(y) = 0$ if $y \in s(K)$ and $|s([a,b]\setminus K)| \le \tfrac{\sigma}{6}$.

    Finally, $\|C_n(y)\|_2 < \tfrac{\rho}{2}$ uniformly for all $y$, if $n$ is sufficiently large, in view of \Cref{l:rep}.

    We conclude that, there is $n_2\ge n_1$ such that for all $n \ge n_2$,
    \begin{align*}
        |\{y \in (0,L- \tfrac{\sigma}{4}) : \|\alpha_n(t_{n}(y)) - \alpha(t(y)) \|_2 \ge \rho \}| 
        \le \frac{\sigma}{6} + \frac{\sigma}{6} +\frac{\sigma}{6}=\frac{\sigma}{2}
    \end{align*}
    which proves \eqref{eq:ats2}.

<!-- lemma-5: body_md -->
\label[l]{l:tim}
   $t_n' \to t'$ in measure on $\mathbb{R}$.
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
We have  
    \[
        t' = \frac{1}{s' \circ t}
    \]
    almost everywhere in $(0,L)$ and $t' = 0$ in $\mathbb{R} \setminus [0,L]$.
    Similarly,
    \[
        t_n' = \frac{1}{s_n' \circ t_n}
    \]
    almost everywhere in $(0,L_n)$ and $t' = 0$ in $\mathbb{R} \setminus [0,L_n]$.

    Let $\rho,\sigma>0$ be fixed.
    We have to show that there exists $n_0 \ge 1$ such that 
    \begin{equation} \label{eq:ts}
        |\{y \in \mathbb{R} : |t_{n}'(y) - t'(y)| \ge \rho\}| \le \sigma, \quad n \ge n_0.
    \end{equation}
    As in the proof of \Cref{l:cim}, we see that it suffices to show \eqref{eq:ts} for $y \in (0,L-\frac{\sigma}{4})$.
    
    Since $s',s_{n}'\ge \epsilon$ almost everywhere in $(a,b)$,  
    \begin{equation} \label{eq:li}
        |t_{n}' - t'| = \Big| \frac{1}{s_{n}'\circ t_n} - \frac{1}{s' \circ t} \Big| 
        \le \epsilon^{-2} | s_n'\circ t_n - s'\circ t |
    \end{equation}
    almost everywhere in $(0,L-\frac{\sigma}{4})$, for sufficiently large $n$.

    In view of \eqref{eq:li},
    setting $\alpha_n := s_n'$ and $\alpha := s'$, the assertion of the lemma follows from \eqref{eq:ats2}.

<!-- theorem-6: body_md -->
\label[t]{t:ms}
    Let $f,f_n \in AC^1((a,b),X)$ and $|f_n'| \to |f'|$ in $L^1((a,b))$.
    Fix $\epsilon>0$ and
    let $\hat f := f \circ t$ and $\hat f_n := f_n \circ t_n$, where $t := t_{\epsilon,f}$ and $t_n := t_{\epsilon,f_n}$.
    Then $t_n \to t$ in $W^{1,q}_{\operatorname{loc}}(\mathbb{R})$ and $|\hat f_n'| \to |\hat f'|$ in $L^q_{\operatorname{loc}}(\mathbb{R})$, 
    for all $q \in[1,\infty)$.
<!-- end theorem-6 -->
<!-- proof-4: body_md -->
We use the notation of the previous section. 
    \Cref{l:rep,l:tim} imply $t_n \to t$ in $W^{1,q}_{\operatorname{loc}}(\mathbb{R})$, for all $q \in [1,\infty)$, as above. 
    By \eqref{eq:nlp}, 
    \[
        |\hat f'| \circ s = \frac{|f'|}{\epsilon + |f'|} \quad \text{ almost everywhere in } (a,b).
    \]
    Consequently (note that $s$ and $t$ have the Lusin $N$-property),  
    \[
        |\hat f'|  = \frac{|f'|}{\epsilon + |f'|} \circ t \quad \text{ almost everywhere in } (0,L).
    \]
    Since $\hat f$ is locally constant on $\mathbb{R} \setminus [0,L]$, we have $|\hat f'|(y)=0$ for 
    $y \in \mathbb{R} \setminus [0,L]$. 
    Analogously,
    \[
        |\hat f_n'|  = \frac{|f_n'|}{\epsilon + |f_n'|} \circ t \quad \text{ almost everywhere in } (0,L_n),
    \]
    and
    $|\hat f_n'|(y)=0$ for $y \in \mathbb{R} \setminus [0,L_n]$.

    Similarly as \eqref{eq:l1c}, we find that
    \begin{equation}
        \frac{|f_n'|}{\epsilon + |f_n'|} \to \frac{|f'|}{\epsilon + |f'|} \quad \text{ in } L^1((a,b)).
    \end{equation}
    The proof of \Cref{l:cim} (with $\alpha_n=\frac{|f_n'|}{\epsilon + |f_n'|}$ and $\alpha =\frac{|f'|}{\epsilon + |f'|}$) 
    gives that $|\hat f_n'| \to |\hat f'|$ in measure on $\mathbb{R}$. 
    Consequently, $|\hat f_n'| \to |\hat f'|$ in $L^q_{\operatorname{loc}}(\mathbb{R})$, 
    for all $q \in[1,\infty)$, because $|\hat f_n'| \le 1$ almost everywhere in $\mathbb{R}$.
