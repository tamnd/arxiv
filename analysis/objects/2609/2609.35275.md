<!-- lemma-1: body_md -->
\label{lemma:triangle}
    Let $h$ be a convex function defined on a nonempty convex set ${\cal C} \subseteq {\mathbb{R}}^d$. Given $\alpha > 0$ and $\bar x \in {\cal C}$, let
    $$x' = \operatorname*{arg\,min}_{x \in {\cal C}} \left\{ h(x) + \alpha \| x - \bar{x} \|_2^2 \right\}.$$
    Then, for any $x \in {\cal C}$,
    $$h(x')+\alpha \|x'-\bar{x}\|_2^2 \leq h(x)+ \alpha \|x-\bar{x}\|_2^2 - \alpha \|x-x'\|_2^2.$$
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We provide a proof for completeness. Define the function 
    $$\phi(x) := h(x) + \alpha \|x-\bar x\|_2^2, \quad x \in {\cal C}.$$
    By $2 \alpha$-strong convexity of $\phi$ on ${\cal C}$, for every $x \in {\cal C}$ and every $\theta \in [0,1]$, we have
    $$ \phi \big((1-\theta)x' + \theta x\big) \leq (1-\theta) \phi(x') + \theta \phi(x) - \alpha \theta(1-\theta) \|x-x'\|_2^2. $$
    Since $x'$ minimizes $\phi$ over ${\cal C}$ and $(1-\theta)x' + \theta x \in {\cal C}$, we also have 
    $$\phi(x') \leq \phi \big((1-\theta)x' + \theta x \big).$$    
    Combining the last two displays yields
    $$ \phi(x') \leq (1-\theta) \phi(x') + \theta \phi(x) - \alpha \theta (1-\theta) \|x-x'\|_2^2, $$
    and therefore, for every $\theta \in (0,1]$, $\phi(x') \leq \phi(x) - \alpha(1-\theta) \|x-x'\|_2^2$. Letting $\theta \downarrow 0$ gives
    $$ \phi(x') \leq \phi(x) - \alpha\|x-x'\|_2^2. $$
    Expanding the definition of $\phi$ completes the proof.
<!-- end proof-1 -->
<!-- proof-2: body_md -->
By symmetry, it suffices to prove part~\ref{prop:generic-lmo:primal}. 
    The proof of part~\ref{prop:generic-lmo:dual} follows by exchanging
    \begin{align*}
        (v_t,x_t,\lambda_t,\alpha_t,\eta_t,{\cal X},R_{{\cal X}},G_{\mathrm{x}},z_t)
        \quad \text{with} \quad
        (u_t,y_t,\mu_t,\beta_t,\tau_t,{\cal Y},R_{{\cal Y}},G_{\mathrm{y}},w_t),
    \end{align*}
    and replacing $f_{z_t}(\cdot)$ by $(-f)_{w_t}(\cdot)$. 
    We therefore focus on the primal inequality.

    Fix any $x \in {\cal X}$. The update of $v_{t+1}$ can be written as
    \begin{align*}
        v_{t+1}
        = 
        \operatorname*{arg\,min}_{v \in \mkern 1.75mu\overline{\mkern-1.75mu{\cal Y}\mkern-1.75mu}\mkern 1.75mu} \left\{(\lambda_t+g^{\mathrm{x}}_t)^\top v+\frac{\alpha_t}{2}\|v-v_t\|_2^2\right\}.
    \end{align*}
    Since $x \in {\cal X} \subseteq \mkern 1.75mu\overline{\mkern-1.75mu{\cal X}\mkern-1.75mu}\mkern 1.75mu$, invoking \cref{lemma:triangle} with $h(v):=(\lambda_t+g^{\mathrm{x}}_t)^\top v$, ${\cal C}:=\mkern 1.75mu\overline{\mkern-1.75mu{\cal Y}\mkern-1.75mu}\mkern 1.75mu$, $\bar x:=v_t$ and $\alpha:=\alpha_t/2$ gives
    \begin{align*}
        (\lambda_t+g^{\mathrm{x}}_t)^\top v_{t+1}+\frac{\alpha_t}{2}\|v_{t+1}-v_t\|_2^2 
        \leq (\lambda_t+g^{\mathrm{x}}_t)^\top x+\frac{\alpha_t}{2}\|x-v_t\|_2^2-\frac{\alpha_t}{2}\|x-v_{t+1}\|_2^2.
    \end{align*}
    Subtracting $\lambda_t^\top x_{t+1}$ and $(g^{\mathrm{x}}_t)^\top v_t$ from both sides yields
    \begin{align*}
        &\lambda_t^\top(v_{t+1}-x_{t+1})+(g^{\mathrm{x}}_t)^\top(v_{t+1}-v_t)+\frac{\alpha_t}{2}\|v_{t+1}-v_t\|_2^2 \\
        &\leq \lambda_t^\top(x-x_{t+1})+(g^{\mathrm{x}}_t)^\top(x-v_t)+\frac{\alpha_t}{2}\|x-v_t\|_2^2-\frac{\alpha_t}{2}\|x-v_{t+1}\|_2^2.
    \end{align*}
    Since $x_{t+1}=\operatorname{LMO}_{{\cal X}}(-\lambda_t)$, we have $\lambda_t^\top(x-x_{t+1})\leq 0$. Hence,
    \begin{align*}
        &\lambda_t^\top(v_{t+1}-x_{t+1})+(g^{\mathrm{x}}_t)^\top(v_{t+1}-v_t)+\frac{\alpha_t}{2}\|v_{t+1}-v_t\|_2^2 \\
        &\leq (g^{\mathrm{x}}_t)^\top(x-v_t)+\frac{\alpha_t}{2}\|x-v_t\|_2^2-\frac{\alpha_t}{2}\|x-v_{t+1}\|_2^2.
    \end{align*}
    The update of $\lambda_{t+1}$ implies ${\lambda_{t+1}}/{\eta_{t+1}} = {\lambda_t}/{\eta_t}+v_{t+1}-x_{t+1}$.
    Therefore,
    \begin{align*}
        \lambda_t^\top(v_{t+1}-x_{t+1})
        &=\eta_t \left( \frac{1}{\eta_t} \lambda_t^\top (v_{t+1} - x_{t+1}) \right)
        =\frac{\eta_t}{2}\left(\left\|\frac{\lambda_{t+1}}{\eta_{t+1}}\right\|_2^2-\left\|\frac{\lambda_t}{\eta_t}\right\|_2^2-\|v_{t+1}-x_{t+1}\|_2^2\right).
    \end{align*}
    Substituting this identity into the previous inequality gives
    \begin{align*}
        &\frac{\eta_t}{2}\left(\left\|\frac{\lambda_{t+1}}{\eta_{t+1}}\right\|_2^2-\left\|\frac{\lambda_t}{\eta_t}\right\|_2^2-\|v_{t+1}-x_{t+1}\|_2^2\right) 
        +(g^{\mathrm{x}}_t)^\top(v_{t+1}-v_t)+\frac{\alpha_t}{2}\|v_{t+1}-v_t\|_2^2 \\
        &\leq (g^{\mathrm{x}}_t)^\top(x-v_t)+\frac{\alpha_t}{2}\|x-v_t\|_2^2-\frac{\alpha_t}{2}\|x-v_{t+1}\|_2^2.
    \end{align*}
    By Young's inequality,
    \begin{align*}
        (g^{\mathrm{x}}_t)^\top(v_{t+1}-v_t)+\frac{\alpha_t}{2}\|v_{t+1}-v_t\|_2^2 \geq -\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t}.
    \end{align*}
    Also, since $v_{t+1}\in\mkern 1.75mu\overline{\mkern-1.75mu{\cal X}\mkern-1.75mu}\mkern 1.75mu$ and $x_{t+1}\in{\cal X}$, we use the crude bound $\|v_{t+1}-x_{t+1}\|_2\leq 2R_{{\cal X}}$. Thus,
    \begin{align*}
        &\frac{\eta_t}{2}\left(\left\|\frac{\lambda_{t+1}}{\eta_{t+1}}\right\|_2^2-\left\|\frac{\lambda_t}{\eta_t}\right\|_2^2\right)-\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t} 
        \leq (g^{\mathrm{x}}_t)^\top(x-v_t)+\frac{\alpha_t}{2}\|x-v_t\|_2^2-\frac{\alpha_t}{2}\|x-v_{t+1}\|_2^2+2\eta_tR_{{\cal X}}^2.
    \end{align*}
    Summing from $t=1$ to $T-1$ gives
    \begin{align*}
        &\frac{\eta_{T-1}}{2}\left\|\frac{\lambda_T}{\eta_T}\right\|_2^2+\frac{1}{2}\sum_{t=2}^{T-1}(\eta_{t-1}-\eta_t)\left\|\frac{\lambda_t}{\eta_t}\right\|_2^2-\frac{\eta_1}{2}\left\|\frac{\lambda_1}{\eta_1}\right\|_2^2-\sum_{t=1}^{T-1}\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t} \\
        &\leq \sum_{t=1}^{T-1}(g^{\mathrm{x}}_t)^\top(x-v_t)+\frac{\alpha_1}{2}\|x-v_1\|_2^2+\frac{1}{2}\sum_{t=2}^{T-1}(\alpha_t-\alpha_{t-1})\|x-v_t\|_2^2-\frac{\alpha_{T-1}}{2}\|x-v_T\|_2^2+2\sum_{t=1}^{T-1}\eta_tR_{{\cal X}}^2.
    \end{align*}
    Since $\lambda_1=0$, $\eta_t\geq\eta_{t+1}$, $\alpha_t\leq\alpha_{t+1}$, and $\|x-v_t\|_2\leq 2R_{{\cal X}}$ for all $x\in{\cal X}$ and $v_t\in\mkern 1.75mu\overline{\mkern-1.75mu{\cal X}\mkern-1.75mu}\mkern 1.75mu$, we obtain
    \begin{align*}
        &\frac{\eta_{T-1}}{2}\left\|\frac{\lambda_T}{\eta_T}\right\|_2^2+\sum_{t=1}^{T-1}(g^{\mathrm{x}}_t)^\top(v_t-x) 
        \leq 2\left(\alpha_{T-1}+\sum_{t=1}^{T-1}\eta_t\right)R_{{\cal X}}^2-\frac{\alpha_{T-1}}{2}\|x-v_T\|_2^2+\sum_{t=1}^{T-1}\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t}.
    \end{align*}
    Adding $(g^{\mathrm{x}}_T)^\top(v_T-x)$ to both sides and applying Young's inequality once more yields
    \begin{align*}
        &\frac{\eta_{T-1}}{2}\left\|\frac{\lambda_T}{\eta_T}\right\|_2^2+\sum_{t=1}^{T}(g^{\mathrm{x}}_t)^\top(v_t-x) \\
        &\leq 2\left(\alpha_{T-1}+\sum_{t=1}^{T-1}\eta_t\right)R_{{\cal X}}^2+(g^{\mathrm{x}}_T)^\top(v_T-x)-\frac{\alpha_{T-1}}{2}\|x-v_T\|_2^2+\sum_{t=1}^{T-1}\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t} \\
        &\leq 2\left(\alpha_{T-1}+\sum_{t=1}^{T-1}\eta_t\right)R_{{\cal X}}^2+\frac{\|g^{\mathrm{x}}_T\|_2^2}{2\alpha_{T-1}}+\sum_{t=1}^{T-1}\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t}.
    \end{align*}
    Let ${\cal F}_t$ be the sigma algebra generated by all randomness up to the beginning of iteration $t$, and define $\bar g^{\mathrm{x}}_t:={\mathbb{E}}[g^{\mathrm{x}}_t\mid{\cal F}_t]$ and $\xi_t:=\bar g^{\mathrm{x}}_t-g^{\mathrm{x}}_t$. By Assumption~\ref{ass:stochastic-oracle}, $\bar g^{\mathrm{x}}_t\in\partial f_{z_t}(v_t)$. Adding $\sum_{t=1}^T\xi_t^\top(v_t-x)$ to both sides of the last display and using convexity of $f_{z_t}(\cdot)$, we get
    \begin{align*}
        &\frac{\eta_{T-1}}{2}\left\|\frac{\lambda_T}{\eta_T}\right\|_2^2+\sum_{t=1}^{T}\bigl(f(v_t,z_t)-f(x,z_t)\bigr) \\
        &\leq 2\left(\alpha_{T-1}+\sum_{t=1}^{T-1}\eta_t\right)R_{{\cal X}}^2+\frac{\|g^{\mathrm{x}}_T\|_2^2}{2\alpha_{T-1}}+\sum_{t=1}^{T-1}\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t}+\sum_{t=1}^{T}\xi_t^\top(v_t-x).
    \end{align*}
    Taking the maximum over $x\in{\cal X}$ gives
    \begin{align*}
        &\frac{\eta_{T-1}}{2}\left\|\frac{\lambda_T}{\eta_T}\right\|_2^2+\max_{x\in{\cal X}}\sum_{t=1}^{T}\bigl(f(v_t,z_t)-f(x,z_t)\bigr) \\
        &\leq 2\left(\alpha_{T-1}+\sum_{t=1}^{T-1}\eta_t\right)R_{{\cal X}}^2+\frac{\|g^{\mathrm{x}}_T\|_2^2}{2\alpha_{T-1}}+\sum_{t=1}^{T-1}\frac{\|g^{\mathrm{x}}_t\|_2^2}{2\alpha_t}+\max_{x\in{\cal X}}\sum_{t=1}^{T}\xi_t^\top(v_t-x).
    \end{align*}
    It remains to control the martingale term. Since $x_1\in{\cal X}$, for every $x\in{\cal X}$,
    \begin{align*}
        \sum_{t=1}^{T}\xi_t^\top(v_t-x)
        &=\sum_{t=1}^{T}\xi_t^\top(v_t-x_1)+ \sum_{t=1}^{T}\xi_t^\top (x_1-x) 
        \leq \sum_{t=1}^{T}\xi_t^\top(v_t-x_1)+2R_{{\cal X}}\left\|\sum_{t=1}^{T}\xi_t\right\|_2.
    \end{align*}
    Because ${\mathbb{E}}[\xi_t\mid{\cal F}_t]=0$ and $v_t-x_1$ is ${\cal F}_t$-measurable, ${\mathbb{E}}\left[\sum_{t=1}^{T}\xi_t^\top(v_t-x_1)\right]=0$.
    Moreover,
    \begin{align*}
        \textstyle {\mathbb{E}}\left[\left\|\sum_{t=1}^{T}\xi_t\right\|_2\right]
        \leq \left({\mathbb{E}}\left[\left\|\sum_{t=1}^{T}\xi_t\right\|_2^2\right]\right)^{1/2} 
        =\left(\sum_{t=1}^{T}{\mathbb{E}}\|\xi_t\|_2^2\right)^{1/2} 
        \leq G_{\mathrm{x}} T^{1/2},
    \end{align*}
    where the last inequality follows from ${\mathbb{E}}\|\xi_t\|_2^2\leq{\mathbb{E}}\|g^{\mathrm{x}}_t\|_2^2\leq G_{\mathrm{x}}^2$. Hence,
    \begin{align}
    \label{eq:martingale}
        \textstyle {\mathbb{E}}\left[\max_{x\in{\cal X}}\sum_{t=1}^{T}\xi_t^\top(v_t-x)\right]\leq 2G_{\mathrm{x}} R_{{\cal X}}T^{1/2}.
    \end{align}
    Taking expectations and using ${\mathbb{E}}\|g^{\mathrm{x}}_t\|_2^2\leq G_{\mathrm{x}}^2$ for all $t$ finally yields~\eqref{eq:generic-primal-lmo-2}.
    This proves part~\ref{prop:generic-lmo:primal}, and the proof is complete by symmetry.

By symmetry, it suffices to prove part~\ref{prop:generic-po:primal}. The proof of part~\ref{prop:generic-po:dual} follows from the same argument after exchanging
    \begin{align*}
        (x_t,g^{\mathrm{x}}_t,\rho_t,{\cal X},R_{{\cal X}},G_{\mathrm{x}},z_t)
        \quad \text{with} \quad
        (y_t,g^{\mathrm{y}}_t,\gamma_t,{\cal Y},R_{{\cal Y}},G_{\mathrm{y}},w_t),
    \end{align*}
    and replacing $f_{z_t}(\cdot)$ by $(-f)_{w_t}(\cdot)$. We therefore prove only the primal inequality.
    
    Fix any $x\in{\cal X}$. The projected update gives
    \begin{align*}
        x_{t+1}=\operatorname{PO}_{{\cal X}}(x_t-\rho_t g^{\mathrm{x}}_t), \qquad t=1,\ldots,T.
    \end{align*}
    By nonexpansiveness of the Euclidean projection,
    \begin{align*}
        \|x_{t+1}-x\|_2^2
        &\leq \|x_t-\rho_t g^{\mathrm{x}}_t-x\|_2^2 
        =\|x_t-x\|_2^2-2\rho_t(g^{\mathrm{x}}_t)^\top(x_t-x)+\rho_t^2\|g^{\mathrm{x}}_t\|_2^2.
    \end{align*}
    Rearranging yields
    \begin{align*}
        (g^{\mathrm{x}}_t)^\top(x_t-x)
        \leq \frac{\|x_t-x\|_2^2}{2\rho_t}-\frac{\|x_{t+1}-x\|_2^2}{2\rho_t}+\frac{\rho_t\|g^{\mathrm{x}}_t\|_2^2}{2}.
    \end{align*}
    Summing this inequality from $t=1$ to $T$ gives
    \begin{align*}
        \sum_{t=1}^{T}(g^{\mathrm{x}}_t)^\top(x_t-x)
        &\leq \frac{\|x_1-x\|_2^2}{2\rho_1}+\frac{1}{2}\sum_{t=2}^{T}\left(\frac{1}{\rho_t}-\frac{1}{\rho_{t-1}}\right)\|x_t-x\|_2^2-\frac{\|x_{T+1}-x\|_2^2}{2\rho_T}+\sum_{t=1}^{T}\frac{\rho_t\|g^{\mathrm{x}}_t\|_2^2}{2}.
    \end{align*}
    Since $\rho_t\geq \rho_{t+1}$ and $\|x_t-x\|_2\leq 2R_{{\cal X}}$ for all $t$, we may drop the final negative term and obtain
    \begin{align*}
        \sum_{t=1}^{T}(g^{\mathrm{x}}_t)^\top(x_t-x)
        &\leq \frac{2R_{{\cal X}}^2}{\rho_1}+2R_{{\cal X}}^2\sum_{t=2}^{T}\left(\frac{1}{\rho_t}-\frac{1}{\rho_{t-1}}\right)+\sum_{t=1}^{T}\frac{\rho_t\|g^{\mathrm{x}}_t\|_2^2}{2} 
        =\frac{2R_{{\cal X}}^2}{\rho_T}+\sum_{t=1}^{T}\frac{\rho_t\|g^{\mathrm{x}}_t\|_2^2}{2}.
    \end{align*}
    
    Let ${\cal F}_t$ denote the sigma algebra generated by all randomness up to the beginning of iteration $t$, so that $x_t$ and $z_t$ are ${\cal F}_t$-measurable. Define $\bar g^{\mathrm{x}}_t:={\mathbb{E}}[g^{\mathrm{x}}_t\mid{\cal F}_t] $ and $\xi_t:=\bar g^{\mathrm{x}}_t-g^{\mathrm{x}}_t$.
    By Assumption~\ref{ass:stochastic-oracle}, $\bar g^{\mathrm{x}}_t\in\partial f_{z_t}(x_t)$. Adding $\sum_{t=1}^{T}\xi_t^\top(x_t-x)$ to both sides of the last display and using convexity of $f_{z_t}(\cdot)$ gives, for every $x\in{\cal X}$,
    \begin{align*}
        \sum_{t=1}^{T}\bigl(f(x_t,z_t)-f(x,z_t)\bigr)
        &\leq \sum_{t=1}^{T}(\bar g^{\mathrm{x}}_t)^\top(x_t-x) 
        \leq \frac{2R_{{\cal X}}^2}{\rho_T}+\sum_{t=1}^{T}\frac{\rho_t\|g^{\mathrm{x}}_t\|_2^2}{2}+\sum_{t=1}^{T}\xi_t^\top(x_t-x).
    \end{align*}
    Taking the maximum over $x\in{\cal X}$ yields
    \begin{align*}
        \max_{x\in{\cal X}}\sum_{t=1}^{T}\bigl(f(x_t,z_t)-f(x,z_t)\bigr)
        \leq \frac{2R_{{\cal X}}^2}{\rho_T}+\sum_{t=1}^{T}\frac{\rho_t\|g^{\mathrm{x}}_t\|_2^2}{2}+\max_{x\in{\cal X}}\sum_{t=1}^{T}\xi_t^\top(x_t-x).
    \end{align*}
    The martingale term is controlled by the same argument used to derive~\eqref{eq:martingale}. Namely, replacing $v_t$ by $x_t$ gives
    \begin{align*}
        \textstyle{\mathbb{E}}\left[\max_{x\in{\cal X}}\sum_{t=1}^{T}\xi_t^\top(x_t-x)\right]\leq 2G_{\mathrm{x}} R_{{\cal X}}T^{1/2}.
    \end{align*}
    Taking expectations and using ${\mathbb{E}}\|g^{\mathrm{x}}_t\|_2^2\leq G_{\mathrm{x}}^2$ for all $t$ gives~\eqref{eq:generic-primal-po-2}.
    This proves part~\ref{prop:generic-po:primal}, and the proof is complete by symmetry.

Define $\bar v_T:= \frac{1}{T} \sum_{t=1}^{T}v_t$ and $\bar u_T:= \frac{1}{T} \sum_{t=1}^{T}u_t$. Since $v_1=x_1$ and $u_1=y_1$, the LMO recursions imply
    \begin{align*}
        \begin{cases}
            \displaystyle \frac{\lambda_T}{\eta_T}
            = 
            \frac{\lambda_{T-1}}{\eta_{T-1}} + v_T - x_T 
            =
            \frac{\lambda_0}{\eta_0} + (v_T - x_T) + \dots + (v_0 - x_0)
            =
            \sum_{t=1}^{T}(v_t-x_t)
            =
            T(\bar v_T-\bar x_T), \\
            \displaystyle \frac{\mu_T}{\tau_T}
            =
            \frac{\mu_{T-1}}{\tau_{T-1}} + u_T - y_T 
            =
            \frac{\mu_{0}}{\tau_{0}}+(u_T - y_T) + \dots + (u_1-y_1)
            =
            \sum_{t=1}^{T}(u_t-y_t)
            =
            T(\bar u_T-\bar y_T).
        \end{cases}
    \end{align*}
    We first convert the primal regret bound in \cref{prop:generic-lmo-restate} into a bound with comparator $\bar y_T$. By convexity of $(-f)_x(\cdot)$, we have $\frac{1}{T}\sum_{t=1}^{T}-f(x,u_t)\geq -f(x,\bar u_T)$.
    Moreover, $(-f)_x(\cdot)$ is $G_{\mathrm{y}}$-Lipschitz on $\mkern 1.75mu\overline{\mkern-1.75mu{\cal Y}\mkern-1.75mu}\mkern 1.75mu$, so
    \begin{align*}
        -f(x,\bar u_T)
        \geq
        -f(x,\bar y_T)-G_{\mathrm{y}}\|\bar u_T-\bar y_T\|_2
        =
        -f(x,\bar y_T)-\frac{G_{\mathrm{y}}}{T}\left\|\frac{\mu_T}{\tau_T}\right\|_2.
    \end{align*}
    Therefore, applying \cref{prop:generic-lmo-restate}\,\ref{prop:generic-lmo:primal} with $z_t=u_t$ gives
    \begin{equation}
    \label{eq:primal-gap-lmo-lmo}
    \begin{aligned}
        &{\mathbb{E}}\left[
            \frac{\eta_{T-1}}{2T}\left\|\frac{\lambda_T}{\eta_T}\right\|_2^2
            -\frac{G_{\mathrm{y}}}{T}\left\|\frac{\mu_T}{\tau_T}\right\|_2
            +\frac{1}{T}\sum_{t=1}^{T}f(v_t,u_t)
            -\min_{x\in{\cal X}}f(x,\bar y_T)
        \right] \\
        &\leq
        \frac{2}{T}\left(\alpha_{T-1}+\sum_{t=1}^{T-1}\eta_t\right)R_{{\cal X}}^2
        +\frac{G_{\mathrm{x}}^2}{2\alpha_{T-1}T}
        +\frac{1}{T}\sum_{t=1}^{T-1}\frac{G_{\mathrm{x}}^2}{2\alpha_t}
        +\frac{2G_{\mathrm{x}} R_{{\cal X}}}{T^{1/2}}.
    \end{aligned}
    \end{equation}
    Similarly, by convexity of $f_y(\cdot)$, we have $\frac{1}{T}\sum_{t=1}^{T}f(v_t,y)\geq f(\bar v_T,y)$.
    Adding~\eqref{eq:primal-gap-lmo-lmo} and~\eqref{eq:dual-gap-lmo-lmo} cancels the average payoff term. It remains to lower bound the two quadratic expressions. For any $a>0$, $b\geq 0$, and $s\geq 0$, we use $as^2/2-bs\geq -b^2/(2a)$. Hence,
    \begin{align*}
        \frac{\eta_{T-1}}{2T}\left\|\frac{\lambda_T}{\eta_T}\right\|_2^2
        -\frac{G_{\mathrm{x}}}{T}\left\|\frac{\lambda_T}{\eta_T}\right\|_2
        &\geq
        -\frac{G_{\mathrm{x}}^2}{2T\eta_{T-1}}, \\
        \frac{\tau_{T-1}}{2T}\left\|\frac{\mu_T}{\tau_T}\right\|_2^2
        -\frac{G_{\mathrm{y}}}{T}\left\|\frac{\mu_T}{\tau_T}\right\|_2
        &\geq
        -\frac{G_{\mathrm{y}}^2}{2T\tau_{T-1}}.
    \end{align*}
    Combining these bounds with~\eqref{eq:primal-gap-lmo-lmo} and~\eqref{eq:dual-gap-lmo-lmo} proves the claim.
    Since $f_y(\cdot)$ is $G_{\mathrm{x}}$-Lipschitz on $\mkern 1.75mu\overline{\mkern-1.75mu{\cal X}\mkern-1.75mu}\mkern 1.75mu$, we also have
    \begin{align*}
        f(\bar v_T,y)
        \geq
        f(\bar x_T,y)-G_{\mathrm{x}}\|\bar v_T-\bar x_T\|_2
        =
        f(\bar x_T,y)-\frac{G_{\mathrm{x}}}{T}\left\|\frac{\lambda_T}{\eta_T}\right\|_2.

    Consequently, we have
    \begin{align*}
        \|\bar x_T\|_2
        \geq
        \min\left\{\|x\|_2:x\in R_{{\cal X}} \cdot \operatorname{conv}\{e_{i_1},\ldots,e_{i_{k_T}}\}\right\} 
        =
        \frac{R_{{\cal X}}}{k_T^{1/2}}
        \geq
        \frac{R_{{\cal X}}}{(T+1)^{1/2}}.
    \end{align*}
    Similarly, since $y_1$ is an extreme point of ${\cal Y}$ and each dual LMO call returns an extreme point of ${\cal Y}$, all points $y_1,\ldots,y_{T+1}$ are vertices of $R_{{\cal Y}} \cdot \Delta_m$, we have
    \begin{align*}
        \|\bar y_T\|_2
        \geq
        \frac{R_{{\cal Y}}}{(T+1)^{1/2}}.
    \end{align*}
    On the other hand, over the full simplices,
    \begin{align*}
        \min_{x\in{\cal X}}\|x\|_2=\frac{R_{{\cal X}}}{n^{1/2}},
        \qquad
        \min_{y\in{\cal Y}}\|y\|_2=\frac{R_{{\cal Y}}}{m^{1/2}}.
    \end{align*}
    Therefore,
    \begin{align*}
        \max_{y\in{\cal Y}}f(\bar x_T,y)-\min_{x\in{\cal X}}f(x,\bar y_T) 
        &=
        G_{\mathrm{x}}\|\bar x_T\|_2
        -G_{\mathrm{y}}\min_{y\in{\cal Y}}\|y\|_2
        -G_{\mathrm{x}}\min_{x\in{\cal X}}\|x\|_2
        +G_{\mathrm{y}}\|\bar y_T\|_2 \\
        &\geq
        \frac{G_{\mathrm{x}} R_{{\cal X}}}{(T+1)^{1/2}}
        -\frac{G_{\mathrm{x}} R_{{\cal X}}}{n^{1/2}}
        +
        \frac{G_{\mathrm{y}} R_{{\cal Y}}}{(T+1)^{1/2}}
        -\frac{G_{\mathrm{y}} R_{{\cal Y}}}{m^{1/2}}.
    \end{align*}
    Since $n,m\geq 4(T+1)$, we have
    \begin{align*}
        \frac{1}{n^{1/2}}\leq \frac{1}{2(T+1)^{1/2}},
        \qquad
        \frac{1}{m^{1/2}}\leq \frac{1}{2(T+1)^{1/2}}.
    \end{align*}
    Combining the last two displays gives the desired bound.

\label{thm:lower-bound-lmo-po-repeat}
    Let $G_{\mathrm{x}}>0,G_{\mathrm{y}}>0, R_{{\cal X}} >0, R_{{\cal Y}}>0$ be fixed. Then, for any generic LMO-PO method of the form \cref{alg:generic-lmo-po} running for $T \geq 1$ iterations, there exist convex compact subsets ${\cal X} \subseteq {\mathbb{R}}^n, {\cal Y} \subseteq {\mathbb{R}}^m$ with radii $R_{{\cal X}}$ and $R_{{\cal Y}}$, respectively, and a function $f \in {\cal F}_{G_{\mathrm{x}},G_{\mathrm{y}},\|\cdot\|_2}^0({\cal X},{\cal Y})$ with $T \leq  \min\{m,n\}/4-1$ such that

        \frac{G_{\mathrm{y}} R_{{\cal Y}}}{2(T+1)^{1/2}}.
    \end{align*}

<!-- proof-11: body_md -->
Fix $T\geq 1$, and choose dimensions $n,m$ such that $n,m\geq 4(T+1)$. Let
    \begin{align*}
        {\cal X}:=R_{{\cal X}} \cdot \Delta_n.
    \end{align*}
    The adversary initializes the primal variable at the extreme point $x_1=R_{{\cal X}} \cdot e_1$ and fixes an LMO tie-breaking rule that always returns an extreme-point minimizer. 
    For the dual variable, we invoke the classical nonsmooth first-order lower bound over the Euclidean ball. Namely, for any adaptive first-order method making $T$ oracle queries and returning $\bar y_T$, there exist a compact convex set ${\cal Y}:=R_{{\cal Y}} \cdot \mathbb{B}_2^m$ and a convex $G_{\mathrm{y}}$-Lipschitz function $h_{T+1}:{\cal Y}\to{\mathbb{R}}$ such that
    \begin{align}
    \label{eq:po-lower-bound-y}
        h_{T+1}(\bar y_T)-\min_{y\in{\cal Y}} h_{T+1}(y)
        \geq
        \frac{G_{\mathrm{y}} R_{{\cal Y}}}{2(T+1)^{1/2}}.
    \end{align}
    This is the standard nonsmooth lower-bound construction of \citet{nesterov2013introductory}. For completeness, we repeat the construction here.
    Set
    \begin{align*}
        \delta:=\frac{R_{{\cal Y}}}{4T(T+1)^{1/2}}.
    \end{align*}
    We inductively construct signs $a_1,\ldots,a_{T+1}\in\{\pm 1\}$ and distinct coordinates $\sigma(1),\ldots,\sigma(T+1)\in \{1, \dots, m\}$. After $y_k$ has been determined, choose
    \begin{align*}
        \sigma(k)\in\operatorname*{arg\,max}_{j\in[m]\setminus\{\sigma(1),\ldots,\sigma(k-1)\}} |(y_k)_j|,
    \end{align*}
    and choose $a_k\in\{\pm 1\}$ so that
    \begin{align*}
        a_k(y_k)_{\sigma(k)}=|(y_k)_{\sigma(k)}|.
    \end{align*}
    For $k=1,\ldots,T+1$, define
    \begin{align*}
        h_k(y):=
        G_{\mathrm{y}}\max_{1\leq i\leq k}
        \left\{
            a_i y_{\sigma(i)}-2(i-1)\delta
        \right\}.
    \end{align*}
    We claim that the final function $h_{T+1}$ is consistent with all information revealed to the algorithm during the first $T$ iterations. To see this, fix $1\leq s<k\leq T+1$ and any $y\in{\cal Y}$ such that $\|y-y_s\|_2\leq\delta$. For every $i>s$, the coordinate $\sigma(i)$ was still unused when $\sigma(s)$ was chosen. Hence
    \begin{align*}
        |(y_s)_{\sigma(s)}|\geq |(y_s)_{\sigma(i)}|\geq a_i(y_s)_{\sigma(i)}.
    \end{align*}
    Therefore,
    \begin{align*}
        a_i y_{\sigma(i)}-2(i-1)\delta
        &\leq a_i(y_s)_{\sigma(i)}+\delta-2(i-1)\delta \\
        &\leq |(y_s)_{\sigma(s)}|+\delta-2s\delta \\
        &= |(y_s)_{\sigma(s)}|-(2s-1)\delta \\
        &\leq a_s y_{\sigma(s)}-2(s-1)\delta.
    \end{align*}
    Thus every affine piece added after step $s$ is dominated on the ball $\{y:\|y-y_s\|_2\leq\delta\}$ by the $s$-th affine piece, which already appears in $h_s$. Consequently,
    \begin{align*}
        h_k(y)=h_s(y),
        \qquad
        \forall y\in{\cal Y} \text{ such that } \|y-y_s\|_2\leq\delta.
    \end{align*}
    In particular, the values and subdifferentials observed at each previous query $y_s$, $s\leq T$, are identical for $h_s$ and for the final function $h_{T+1}$. 
    The function $h_{T+1}$ is convex and $G_{\mathrm{y}}$-Lipschitz on ${\cal Y}$, since it is the maximum of affine functions whose slopes have Euclidean norm $G_{\mathrm{y}}$. Moreover,
    \begin{align*}
        \min_{y\in{\cal Y}}h_{T+1}(y)
        &\leq
        G_{\mathrm{y}}\min_{\|y\|_2\leq R_{{\cal Y}}}
        \max_{1\leq i\leq T+1}
        a_i y_{\sigma(i)} 
        =
        -\frac{G_{\mathrm{y}} R_{{\cal Y}}}{(T+1)^{1/2}}.
    \end{align*}
    On the other hand, since $\bar y_T=y_{T+1}$ in \cref{alg:generic-lmo-po}, we have
    \begin{align*}
        h_{T+1}(y_{T+1})
        \geq
        G_{\mathrm{y}}\left(a_{T+1}(y_{T+1})_{\sigma(T+1)}-2T\delta\right) 
        =
        G_{\mathrm{y}}\left(|(y_{T+1})_{\sigma(T+1)}|-2T\delta\right) 
        \geq
        -2G_{\mathrm{y}} T\delta.
    \end{align*}
    Therefore, by the choice of $\delta$,
    \begin{align*}
        h(\bar y_T)-\min_{y\in{\cal Y}}h(y)
        \geq
        \frac{G_{\mathrm{y}} R_{{\cal Y}}}{(T+1)^{1/2}}-2G_{\mathrm{y}} T\delta
        =
        \frac{G_{\mathrm{y}} R_{{\cal Y}}}{2(T+1)^{1/2}}.
    \end{align*}
    
    Define now the saddle payoff
    \begin{align*}
        f(x,y):=G_{\mathrm{x}}\|x\|_2-h_{T+1}(y).
    \end{align*}
    Since $x_1$ is an extreme point of ${\cal X}$ and each primal LMO call returns an extreme point of ${\cal X}$, all points $x_1,\ldots,x_{T+1}$ are vertices of $R_{{\cal X}}\Delta_n$. By \cref{alg:generic-lmo-po}, the output satisfies
    \begin{align*}
        \bar x_T\in R_{{\cal X}} \cdot \operatorname{conv}\{e_{i_1},\ldots,e_{i_{k_T}}\}.
    \end{align*}
    Consequently, we have
    \begin{align*}
        \|\bar x_T\|_2
        \geq
        \min\left\{\|x\|_2:x\in R_{{\cal X}} \cdot \operatorname{conv}\{e_{i_1},\ldots,e_{i_{k_T}}\}\right\} 
        =
        \frac{R_{{\cal X}}}{k_T^{1/2}}
        \geq
        \frac{R_{{\cal X}}}{(T+1)^{1/2}}.
    \end{align*}
    Since $n\geq 4(T+1)$, we may conclude that
    \begin{align}
    \label{eq:lmo-po-primal-lower}
        G_{\mathrm{x}}\|\bar x_T\|_2-G_{\mathrm{x}}\min_{x\in{\cal X}}\|x\|_2
        \geq
        \frac{G_{\mathrm{x}} R_{{\cal X}}}{2(T+1)^{1/2}}.
    \end{align}
    Finally, by the separable form of $f$,
    \begin{align*}
        \max_{y\in{\cal Y}}f(\bar x_T,y)-\min_{x\in{\cal X}}f(x,\bar y_T) 
        &=
        G_{\mathrm{x}}\|\bar x_T\|_2-\min_{y\in{\cal Y}}h_{T+1}(y)
        -
        \left(
            G_{\mathrm{x}}\min_{x\in{\cal X}}\|x\|_2-h_{T+1}(\bar y_T)
        \right) \\
        &=
        G_{\mathrm{x}}\|\bar x_T\|_2-G_{\mathrm{x}}\min_{x\in{\cal X}}\|x\|_2
        +
        h_{T+1}(\bar y_T)-\min_{y\in{\cal Y}}h_{T+1}(y).
    \end{align*}
    Combining this identity with~\eqref{eq:po-lower-bound-y} and~\eqref{eq:lmo-po-primal-lower} yields the desired bound.

<!-- lemma-5: body_md -->
\label{lem:proj-oracle}
    Given any $c\in{\mathbb{R}}^n$, let $[c]_+=\max\{c,0\}$ componentwise. For
    \begin{align*}
        \mkern 1.75mu\overline{\mkern-1.75mu{\cal Y}\mkern-1.75mu}\mkern 1.75mu:=\{y\in{\mathbb{R}}^n:y\geq 0,\|y\|_2\leq 1\},
    \end{align*}
    it holds that
    \begin{align*}
        \operatorname{PO}_{\mkern 1.75mu\overline{\mkern-1.75mu{\cal Y}\mkern-1.75mu}\mkern 1.75mu}(c)=\frac{[c]_+}{\max\{\|[c]_+\|_2,1\}}.
    \end{align*}
<!-- end lemma-5 -->
<!-- proof-12: body_md -->
Computing $\operatorname{PO}_{\mkern 1.75mu\overline{\mkern-1.75mu{\cal Y}\mkern-1.75mu}\mkern 1.75mu}(c)$ is equivalent to solving
    \begin{align*}
        \min_{y\in{\mathbb{R}}^n} \{ \|y-c\|_2^2 : y \geq 0,\ \|y\|_2 \leq 1 \}.
    \end{align*}
    Since the feasible set is the intersection of the nonnegative orthant and the Euclidean unit ball, the negative coordinates of $c$ are first clipped to zero. Thus the problem reduces to projecting $[c]_+$ onto the Euclidean unit ball. If $\|[c]_+\|_2\leq 1$, then $[c]_+$ is feasible and is the projection. If $\|[c]_+\|_2>1$, the Euclidean projection onto the unit ball is $[c]_+/\|[c]_+\|_2$. Combining the two cases gives
    \begin{align*}
        \operatorname{PO}_{\mkern 1.75mu\overline{\mkern-1.75mu{\cal Y}\mkern-1.75mu}\mkern 1.75mu}(c)=\frac{[c]_+}{\max\{\|[c]_+\|_2,1\}}.
    \end{align*}
    This proves the claim.
