<!-- definition-1: body_md -->
A \textbf{diassociative conformal algebra} (or simply a {\bf diconformal algebra}) is a $\mathbb{C}[\partial]$-module $\mathcal{D}$ equipped with two $\mathbb{C}$-bilinear maps (called $\lambda$-products) $\dashv_{\lambda}, ~  \! \vdash_{\lambda} : \mathcal{D} \times \mathcal{D} \rightarrow \mathcal{D}[ \lambda ]$ both of which are conformal sesquilinear in the sense that
 \begin{align}
     \partial a \dashv_\lambda b = - \lambda (a \dashv_\lambda b), \quad a \dashv_\lambda \partial b = (\partial + \lambda) ~  \! a \dashv_\lambda b, \label{ses1}\\
     \partial a \vdash_\lambda b = - \lambda (a \vdash_\lambda b), \quad a \vdash_\lambda \partial b = (\partial + \lambda) ~  \! a \vdash_\lambda b, \label{ses2}
 \end{align}
 that satisfy the following list of identities: 
    \begin{align}
   a\dashv_{\lambda}(b \dashv_{\mu}c)=~&(a\dashv_{\lambda}b)\dashv_{\lambda + \mu}c,\label{equation 1} \\
 a \dashv_{\lambda}(b \vdash_{\mu}c) =~& (a\dashv_{\lambda}b)\dashv_{\lambda + \mu}c,\label{equation 2}\\
  a\vdash_{\lambda}(b \dashv_{\mu}c) =~&  (a\vdash_{\lambda}b)\dashv_{\lambda + \mu}c,\label{equation 3}\\
   a \vdash_{\lambda}(b\vdash_{\mu}c)=~&(a\dashv_{\lambda}b)\vdash_{\lambda+\mu}c,\label{equation 4}\\
  a \vdash_{\lambda}(b\vdash_{\mu}c)=~&(a\vdash_{\lambda}b)\vdash_{\lambda + \mu}c, \label{equation 5}
\end{align} 
for all $a,b,c \in \mathcal{D}$. A diassociative conformal algebra as above may be denoted by the triple $(\mathcal{D}, \dashv_{\lambda},\vdash_{\lambda})$ or simply by $\mathcal{D}$.
<!-- end definition-1 -->

Let $(\mathcal{D},\dashv_{\lambda},\vdash_{\lambda})$ and
$(\mathcal{D}',\dashv_{\lambda}',\vdash_{\lambda}')$ be two diassociative conformal algebras. Then a {\bf homomorphism} of diassociative conformal algebras
from $\mathcal{D}$ to
$\mathcal{D}'$
is a $\mathbb{C}[\partial]$-linear map
$\varphi:\mathcal{D}\rightarrow\mathcal{D}'$ satisfying
\begin{align*}
\varphi(a\dashv_{\lambda}b)
=\varphi(a)\dashv_{\lambda}'\varphi(b) ~~~~ \text{ and } ~~~~
\varphi(a\vdash_{\lambda}b)
=\varphi(a)\vdash_{\lambda}'\varphi(b), \text{ for all } a, b \in \mathcal{D}.
\end{align*}

<!-- proposition-1: body_md -->
Let $(\mathcal{D}, \dashv_\lambda, \vdash_\lambda)$ be a diassociative conformal algebra, where $\mathcal{D}=\mathbb{C}[\partial]a$ is a rank one free $\mathbb{C}[\partial]$-module. Then there exists a unique $\alpha \in \mathbb{C}$ such that
\begin{align*}
a\dashv_{\lambda}a=\alpha a
~~~~\text{ and }~~~~~
a\vdash_{\lambda}a=\alpha a.
\end{align*}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Since $\mathcal{D}=\mathbb{C}[\partial]a$, there exist polynomials
$f(\partial,\lambda), \, g(\partial,\lambda)\in\mathbb{C}[\partial,\lambda]$ such that $a\dashv_{\lambda}a=f(\partial,\lambda)a$ and $a\vdash_{\lambda}a=g(\partial,\lambda)a$.
Replacing $\mu$ by $-\lambda$ in the diassociative conformal identity (\ref{equation 1}), while keeping $\lambda$ unchanged, yields
\begin{align}\label{p-eq}
f(\partial+\lambda,-\lambda)f(\partial,\lambda)
=f(0,\lambda)f(\partial,0).
\end{align}
If $f=0$, then $f$ is constant with value $\alpha=0$. Suppose now that $f\neq0$, and write $f(\partial,\lambda)=\sum \limits_{i=0}^{k}a_{i}(\lambda)\partial^i$, where $a_i(\lambda) \in \mathbb{C}[\lambda]$ and $a_k(\lambda)\neq0$. For $k>0$, comparing the coefficients of $\partial^{2k}$ on both sides of (\ref{p-eq}), we obtain $a_k(\lambda)a_k(-\lambda)=0$. This is not possible as $\mathbb{C}[\lambda]$ is an integral domain and $a_k(\lambda)$ is a non-zero polynomial. Hence $k=0$, and consequently $f(\partial,\lambda)=a_0(\lambda)$. Substituting this expression into (\ref{p-eq}) gives
\begin{align}\label{eqn-lambda}
a_0(-\lambda)a_0(\lambda)=a_0(\lambda)a_0(0).
\end{align}
Next, let $a_0(\lambda)= \sum \limits_{i=0}^{t}b_i \lambda^i$, where $b_i \in \mathbb{C}$ and $b_t\neq0$. If $t>0$, then by comparing the coefficients of $\lambda^{2t}$ on both sides of (\ref{eqn-lambda}) gives $(-1)^t b_t^2=0$, which is not possible as $b_t\neq0$. Therefore, $t=0$ and hence $f(\partial,\lambda)=\alpha$, for some $\alpha\in\mathbb{C}$. Similarly, from the diassociative conformal identity (\ref{equation 5}), we obtain $g(\partial,\lambda)=\beta$, for some $\beta\in\mathbb{C}$. Consequently, $a\dashv_{\lambda}a=\alpha  a$ and $a\vdash_{\lambda}a=\beta a$.
Finally, using the identities (\ref{equation 2}) and (\ref{equation 4}), we obtain $\alpha^2=\alpha\beta$ and $\beta^2=\alpha\beta$. This gives $\alpha=\beta$, which proves the result.

Let $\mathcal{A}$ be an associative conformal algebra. A {\bf conformal $\mathcal{A}$-bimodule} is a $\mathbb{C} [\partial]$-module $\mathcal{M}$ equipped with two conformal sesquilinear maps (called left and right $\lambda$-actions) $ \mathcal{A} \times \mathcal{M} \rightarrow \mathcal{M} [\lambda]$, $(a, u) \mapsto a_\lambda u$ and $\mathcal{M} \times \mathcal{A} \rightarrow \mathcal{M} [\lambda]$, $(u, a) \mapsto u_\lambda a$ satisfying
\begin{align*}
a_{\lambda}(b_{\mu}u)=(a_{\lambda}b)_{\lambda+\mu}u, \quad
a_{\lambda}(u_{\mu}b)=(a_{\lambda}u)_{\lambda+\mu}b ~~~ \text{ and } ~~~

\end{align*}

\end{align*}

<!-- lemma-1: body_md -->
\label{ lemma 1}
    Let $(\mathcal{D}, \dashv_{\lambda}, \vdash_{\lambda})$ be a diassociative conformal algebra. Then for any $a,b,c \in \mathcal{D}$, the following identities hold: 
    \begin{align}
         a \dashv_{-\partial -\lambda}(b \vdash _{\mu}c) =~& (a \dashv _{-\partial -\mu} b) \dashv _{-\partial + \mu-\lambda } c, \label{edi-1}\\
        a \dashv_{-\partial -\lambda}( b \dashv _{-\partial -\mu} c) =~& (a \dashv _{-\partial+ \mu -\lambda } b) \dashv _{-\partial -\mu} c, \label{edi-2}\\
        a \vdash_{-\partial - \lambda}(b \vdash_{\mu} c) =~& (a \dashv_{-\partial -\mu}b)\vdash_{-\partial + \mu-\lambda }c,\label{edi-3}\\
        a \vdash _{\lambda} (b \dashv _{-\partial -\mu} c) =~& (a \vdash _{\lambda} b) \dashv_{ -\partial -\mu} c\label{edi-4}.
    \end{align}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
For any $a,b,c \in \mathcal{D}$, we observe that
\begin{align*}
  & a \dashv_{-\partial -\lambda}(b \vdash_{\mu}c) \stackrel{(\ref{equation 2})}{=} (a \dashv_{-\partial -\lambda}b) \dashv_{-\partial+\mu-\lambda}c 
   \stackrel{(\ref{equation 1})}{=}a \dashv_{-\partial -\lambda}(b \dashv_{\mu} c)
    \stackrel{(\ref{equation b})}{=} (a \dashv_{-\partial -\mu}b) \dashv_{-\partial +\mu-\lambda} c,\\
  & \qquad \qquad \qquad \qquad \qquad a \dashv_{-\partial-\lambda}(b \dashv_{-\partial-\mu}c) \stackrel{(\ref{equation c})}{=} (a\dashv_{-\partial+\mu-\lambda}b)\dashv_{-\partial-\mu}c,\\
  &   a \vdash_{-\partial - \lambda}(b \vdash_{\mu} c)  \stackrel{(\ref{equation b})}{=}(a \vdash_{-\partial-\mu}b)\vdash_{-\partial+\mu-\lambda}c
    \stackrel{(\ref{equation 5})}{=}a \vdash_{-\partial-\mu}(b \vdash_{-\lambda+2\mu}c)
     \stackrel{(\ref{equation 4})}{=}(a\dashv_{-\partial-\mu}b)\vdash_{-\partial+\mu-\lambda}c.
\end{align*}
This proves the first three identities. Finally, to prove the last identity, we consider the conformal sesquilinearity condition $a \vdash_{\lambda}\left( (-\partial-\mu)^{j} b \right)=(-\partial-\lambda-\mu)^{j} (a \vdash_{\lambda}b)$. Thus,

\begin{align*}
a \vdash_{\lambda}(b \dashv_{-\partial-\mu}c)
= a \vdash_\lambda \big(  \sum_{j \geq 0} \frac{(-\partial - \mu)^j}{j!} ~ \! b \dashv_{(j)} c  \big) =~& 
\sum_{j\geq 0}
\frac{(-\partial-\lambda-\mu)^j}{j!}
a \vdash_{\lambda}(b \dashv_{(j)}c) \\
=~&
a \vdash_{\lambda}(b \dashv_{\nu}c)\big|_{\nu=-\lambda-\mu-\partial} \\
=~& (a \vdash _{\lambda} b) \dashv_{ -\partial -\mu} c \quad (\text{by } (\ref{equation 3})).
\end{align*}
This concludes the result.

<!-- definition-3: body_md -->
(\cite{bakalov-kac-voronov,zhang}) A {\bf Leibniz conformal algebra} is a pair $(\mathcal{L},[\cdot_{\lambda}\cdot])$ consisting of a $\mathbb{C}[\partial]$-module $\mathcal{L}$ with a conformal sesquilinear $\lambda$-bracket $[\cdot_{\lambda}\cdot]:\mathcal{L} \times \mathcal{L} \to \mathcal{L}[   \lambda ] $ that satisfies
\begin{align}\label{conf-leib}
 [a_{\lambda}[b_{\mu}c]] =[[a_{\lambda}b]_{\lambda+\mu}c]
+ [b_{\mu}[a_{\lambda}c]], \text{ for all } a,b,c \in \mathcal{L}. 
\end{align}
<!-- end definition-3 -->
<!-- proposition-2: body_md -->
\label{prop-di-leib}
    Let $(\mathcal{D},\dashv_{\lambda},\vdash_{\lambda})$ be a diassociative conformal algebra. Then we define a $\lambda$-bracket $[ \cdot_\lambda \cdot ] : \mathcal{D} \times \mathcal{D} \rightarrow \mathcal{D} [\lambda]$ by
\begin{align}\label{lambda-bracket}
    [a_{\lambda}b] := a\vdash_{\lambda}b ~ \! - ~ \! b\dashv_{-\lambda-\partial}a, \text{ for } a,b\in\mathcal{D}.
\end{align}
Then $(\mathcal{D}, [ \cdot_\lambda \cdot ])$ is a Leibniz conformal algebra.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
It is immediate that the $\lambda$-bracket $[\cdot_\lambda \cdot ]$ satisfies the conformal sesquilinearity conditions. Thus, it remains to verify the identity (\ref{conf-leib}). For any $a,b,c\in\mathcal{D}$, we observe that
\begin{align*}
&[[a_{\lambda}b]_{\lambda+\mu}c]+[b_{\mu}[a_{\lambda}c]]-[a_{\lambda}[b_{\mu}c]]\\
={}&[a_{\lambda}b]\vdash_{\lambda+\mu}c
-c\dashv_{-\lambda-\mu-\partial}[a_{\lambda}b]
+b\vdash_{\mu}[a_{\lambda}c]
-[a_{\lambda}c]\dashv_{-\mu-\partial}b\\
&\hspace{1cm}
-a\vdash_{\lambda}[b_{\mu}c]
+[b_{\mu}c]\dashv_{-\lambda-\partial}a\\
={}&(a\vdash_{\lambda}b-b\dashv_{-\lambda-\partial}a)\vdash_{\lambda+\mu}c
-c\dashv_{-\lambda-\mu-\partial}(a\vdash_{\lambda}b-b\dashv_{-\lambda-\partial}a)\\
&\hspace{1cm}
+b\vdash_{\mu}(a\vdash_{\lambda}c-c\dashv_{-\lambda-\partial}a)
-(a\vdash_{\lambda}c-c\dashv_{-\lambda-\partial}a)\dashv_{-\mu-\partial}b\\
&\hspace{1cm}
-a\vdash_{\lambda}(b\vdash_{\mu}c-c\dashv_{-\mu-\partial}b)
+(b\vdash_{\mu}c-c\dashv_{-\mu-\partial}b)\dashv_{-\lambda-\partial}a\\
={}&0 \quad (\text{by } (\ref{equation 1}), (\ref{equation 4}) \text{ and Lemma } \ref{ lemma 1}).
\end{align*} 
Hence, the conformal Leibniz identity holds.

A \textbf{Gerstenhaber algebra} is a triple $(\mathcal{G}, \smile , [~, ~ ])$ consisting of a graded $\mathbb{C}$-module $\mathcal{G}=\bigoplus \limits_{i\in\mathbb{Z}}\mathcal{G}^{i}$ equipped with a degree $0$ graded-commutative associative product $\smile ~ \! :\mathcal{G}^{i}\otimes\mathcal{G}^{j}\rightarrow\mathcal{G}^{i+j}$ and a degree $-1$ bracket $[\, ,\, ]:\mathcal{G}^{i}\otimes\mathcal{G}^{j}\rightarrow\mathcal{G}^{i+j-1}$ subject to satisfy the following conditions:

\end{align}
for all homogeneous elements $x\in\mathcal{G}^{i}$, $y\in\mathcal{G}^{j}$ and $z\in\mathcal{G}^{k}$.

Let $\mathcal{I}$ be the ideal of the diassociative conformal algebra $\mathcal{D}$ generated by the set
\begin{align*}
\{\,a\dashv_{(j)}b-a\vdash_{(j)}b
\mid a,b\in \mathcal{D}, ~ \! j\ge0\,\}.
\end{align*}
Then the quotient $\mathcal{A}:=\mathcal{D}/\mathcal{I}$ is an associative conformal algebra with the $\lambda$-product
\begin{align*}
[a]_{\lambda}[b]
:=
\sum_{j\ge0}[a\dashv_{(j)}b] ~ \! \frac{\lambda^j}{j!}
=
\sum_{j\ge0}[a\vdash_{(j)}b] ~ \!\frac{\lambda^j}{j!}, \text{ for } a, b \in \mathcal{D}.
\end{align*}
Furthermore, $\mathcal{D}$ becomes a conformal $\mathcal{A}$-bimodule via the left and right $\lambda$-actions
\begin{align*}
[a]_{\lambda}b=a\vdash_{\lambda}b
~~~ \text{ and } ~~~
b_{\lambda}[a]=b\dashv_{\lambda}a,  \text{ for } a, b \in \mathcal{D}.
\end{align*}

be the canonical quotient map. Then, for all $a,b\in\mathcal{D}$, we have
\begin{align*}
P(a)_{\lambda}P(b)
=[a]_{\lambda}[b] 
=\sum_{j\ge0}[a\dashv_{(j)}b] ~ \! \frac{\lambda^j}{j!}
=P(a_{\lambda}P(b)),\\
P(a)_{\lambda}P(b)
=[a]_{\lambda}[b] 
=\sum_{j\ge0}[a\vdash_{(j)}b] ~ \! \frac{\lambda^j}{j!}
=P(P(a)_{\lambda}b).
\end{align*}
Hence, $P$ is a relative averaging operator on $\mathcal{A}$ with respect to the conformal $\mathcal{A}$-bimodule $\mathcal{D}$.

\end{align*}
