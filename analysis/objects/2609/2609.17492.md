<!-- dfn-1: body_md -->
\label{dfn:admissiblesupercriticalregime}
    A nonempty subset $\mathcal{S} \subseteq \operatorname{interior}(\mathcal{T})$, open in $\mathbb{R}^2$, is an \emph{admissible supercritical regime} if the following conditions hold:
    \begin{enumerate}
        \item \emph{(Regularity)}: the internal energy is twice continuously differentiable $e \in C^2(\mathcal{S})$.
        \item \emph{(Monotonicity)}: for all $\boldsymbol{\eta} \in \mathcal{S}$ the pressure and temperature are positive:
        \begin{equation*}
            p(\boldsymbol{\eta}) > 0, \qquad T(\boldsymbol{\eta}) > 0,
        \end{equation*}
        and for each fixed $\nu$ the map $\sigma \mapsto e(\nu,\sigma)$ is injective on its domain in $\mathcal{S}$.
        \item \emph{(Strict stability)}: for all $\boldsymbol{\eta} \in \mathcal{S}$ it holds that:
        \begin{equation*}
            e_{\nu\nu}(\boldsymbol{\eta}),\ e_{\sigma\sigma}(\boldsymbol{\eta}) > 0, \qquad e_{\nu\nu}(\boldsymbol{\eta})e_{\sigma\sigma}(\boldsymbol{\eta}) > (e_{\nu\sigma}(\boldsymbol{\eta}))^2.
        \end{equation*}
    \end{enumerate}

<!-- prop-6: body_md -->
\label{prop:massmomentumfluxconsistency}
    Let $\boldsymbol{f}_h : \mathcal{S}_{\boldsymbol{U}} \times \mathcal{S}_{\boldsymbol{U}} \rightarrow \mathbb{R}^3$ be a consistent, algebraically PEP numerical flux, where $\mathcal{S}_{\boldsymbol{U}}$ is an admissible supercritical regime.     \begin{equation}
        f_h^m(\boldsymbol{U}_R,\boldsymbol{U}_L) = V f_h^{\rho}(\boldsymbol{U}_R,\boldsymbol{U}_L) + P, \qquad \forall \boldsymbol{U}_R,\boldsymbol{U}_L \in \mathcal{M}_{P,V}.
        \label{eq:massmomentumfluxrelation}
    \end{equation}

<!-- proof-1: body_md -->
Assume the numerical flux $\boldsymbol{f}_h$ is algebraically PEP and consistent and $\mathcal{S}_{\boldsymbol{U}}$ is some admissible supercritical regime. Let $P\in\mathbb{R}_+$ and $V \in \mathbb{R}$ be arbitrary. If $P,V$ are such that $\mathcal{M}_{P,V}$ is empty \eqref{eq:massmomentumfluxrelation} is vacuous and holds automatically. Thus we can assume that $P,V$ are such that $\mathcal{M}_{P,V}$ is nonempty. Note that $v$ is differentiable on $\mathcal{S}_{\boldsymbol{U}}$, its gradient satisfies:
    \begin{equation*}
        \nabla v(\boldsymbol{U}) = \begin{bmatrix}
            -\frac{V}{\rho} \\ \frac{1}{\rho} \\ 0
        \end{bmatrix}, \qquad \forall \boldsymbol{U} \in \mathcal{M}_{P,V},
    \end{equation*}
    using \eqref{eq:vpdefinition} and that $v(\boldsymbol{U}) = V$ for $\boldsymbol{U}\in\mathcal{M}_{P,V}$. By \autoref{dfn:algebraicpressureequilibriumpreservation} of the algebraic PEP property, we have for any arbitrary $\boldsymbol{U}_R, \boldsymbol{U}_C, \boldsymbol{U}_L\in\mathcal{M}_{P,V}$ that:
    \begin{align*}
        0&=\nabla v(\boldsymbol{U}_C)^T\Delta \boldsymbol{f}_h(\boldsymbol{U}_R,\boldsymbol{U}_C,\boldsymbol{U}_L)  \\
        &=-\frac{V}{\rho_C}\left[f_h^\rho(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^\rho(\boldsymbol{U}_C, \boldsymbol{U}_L) \right] + \frac{1}{\rho_C}\left[f_h^m(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^m(\boldsymbol{U}_C, \boldsymbol{U}_L) \right] .
    \end{align*}
    Choosing $\boldsymbol{U}_C = \boldsymbol{U}_L$, it holds that:
    \begin{equation*}
        0 = -\frac{V}{\rho_L}\left[f_h^\rho(\boldsymbol{U}_R,\boldsymbol{U}_L) - \rho_L V \right] + \frac{1}{\rho_L}\left[f_h^m(\boldsymbol{U}_R,\boldsymbol{U}_L) - (\rho_L V^2 +P) \right].
    \end{equation*}
    Simplifying gives:
    \begin{equation*}
        f_h^m(\boldsymbol{U}_R,\boldsymbol{U}_L) = Vf_h^\rho(\boldsymbol{U}_R,\boldsymbol{U}_L) + P.
    \end{equation*}
    The result then follows since $\boldsymbol{U}_R,\boldsymbol{U}_L \in \mathcal{M}_{P,V}$ are arbitrary.

<!-- prop-7: body_md -->
\label{prop:energyevolution}
    Let $\boldsymbol{f}_h : \mathcal{S}_{\boldsymbol{U}} \times \mathcal{S}_{\boldsymbol{U}} \rightarrow \mathbb{R}^3$ be a consistent, algebraically PEP numerical flux, where $\mathcal{S}_{\boldsymbol{U}}$ is an admissible supercritical regime. Then for any $P \in \mathbb{R}_+$ and $V \in \mathbb{R}$ the following is satisfied:
    \begin{equation*}
        \nabla \varepsilon(\boldsymbol{U}_C)^T\Delta \boldsymbol{f}_h(\boldsymbol{U}_R,\boldsymbol{U}_C,\boldsymbol{U}_L) = f_h^\varepsilon(\boldsymbol{U}_R, \boldsymbol{U}_C; V) - f_h^\varepsilon(\boldsymbol{U}_C, \boldsymbol{U}_L;V), \qquad \forall \boldsymbol{U}_R, \boldsymbol{U}_C, \boldsymbol{U}_L \in \mathcal{M}_{P,V},
    \end{equation*}
    where:
    \begin{equation*}
        f_h^\varepsilon(\boldsymbol{U}_R, \boldsymbol{U}_C; V) := f_h^E(\boldsymbol{U}_R, \boldsymbol{U}_C) - \frac12 V^2 f_h^\rho(\boldsymbol{U}_R, \boldsymbol{U}_C).
    \end{equation*}
<!-- end prop-7 -->
<!-- proof-2: body_md -->
Assume the numerical flux $\boldsymbol{f}_h$ is algebraically PEP and consistent and $\mathcal{S}_{\boldsymbol{U}}$ is some admissible supercritical regime. Let $P\in \mathbb{R}_+$ and $V \in \mathbb{R}$ be arbitrary such that $\mathcal{M}_{P,V}$ is nonempty. Note that $\varepsilon$ is differentiable on $\mathcal{S}_{\boldsymbol{U}}$, its gradient satisfies:
    \begin{equation*}
        \nabla \varepsilon(\boldsymbol{U}) = \begin{bmatrix}
            \frac12 V^2 \\ -V \\ 1
        \end{bmatrix}, \qquad \forall \boldsymbol{U} \in \mathcal{M}_{P,V}.
    \end{equation*}
    Let $\boldsymbol{U}_R, \boldsymbol{U}_C, \boldsymbol{U}_L \in \mathcal{M}_{P,V}$ also be arbitrary. Using the gradient on $\mathcal{M}_{P,V}$, it follows by definition that:
    \begin{align*}
        \nabla \varepsilon(\boldsymbol{U}_C)^T\Delta \boldsymbol{f}_h(\boldsymbol{U}_R,\boldsymbol{U}_C,\boldsymbol{U}_L) &= \left[f_h^E(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^E(\boldsymbol{U}_C, \boldsymbol{U}_L) \right] -V\left[f_h^m(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^m(\boldsymbol{U}_C, \boldsymbol{U}_L) \right]  \\
        &\qquad \qquad  + \frac12 V^2\left[f_h^\rho(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^\rho(\boldsymbol{U}_C, \boldsymbol{U}_L) \right].
    \end{align*}
    Since $\boldsymbol{U}_R, \boldsymbol{U}_C, \boldsymbol{U}_L \in \mathcal{M}_{P,V}$ and the numerical flux is algebraically PEP and consistent, \autoref{prop:massmomentumfluxconsistency} applies. We can therefore simplify:
    \begin{align*}
        \nabla \varepsilon(\boldsymbol{U}_C)^T\Delta \boldsymbol{f}_h(\boldsymbol{U}_R,\boldsymbol{U}_C,\boldsymbol{U}_L) &= \left[f_h^E(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^E(\boldsymbol{U}_C, \boldsymbol{U}_L) \right] - \frac12 V^2\left[f_h^\rho(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^\rho(\boldsymbol{U}_C, \boldsymbol{U}_L) \right]  \\

        &= f_h^\varepsilon(\boldsymbol{U}_R, \boldsymbol{U}_C; V) - f_h^\varepsilon(\boldsymbol{U}_C, \boldsymbol{U}_L;V).
    \end{align*}
    The result follows since $\boldsymbol{U}_R, \boldsymbol{U}_C, \boldsymbol{U}_L \in \mathcal{M}_{P,V}$ are arbitrary.

<!-- proof-3: body_md -->
Assume the numerical flux $\boldsymbol{f}_h$ is algebraically PEP and consistent and $\mathcal{S}_{\boldsymbol{U}}$ is some admissible supercritical regime. Let $P\in \mathbb{R}_+$ be arbitrary so that $\Gamma_P$ is nonempty. Also, choose any $V \in \mathbb{R} \setminus \{0\}$. By \autoref{prop:pressureregularity}, pressure is $C^1$ on the admissible supercritical regime. Note that:
    \begin{equation*}
        D\boldsymbol{\tau}(\boldsymbol{U}) = \begin{bmatrix}
            \nabla \rho(\boldsymbol{U})^T \\ \nabla \varepsilon(\boldsymbol{U})^T
        \end{bmatrix}, \qquad \nabla \rho(\boldsymbol{U}) = \begin{bmatrix}
            1 \\ 0 \\ 0
        \end{bmatrix}.
    \end{equation*}
    By \autoref{dfn:algebraicpressureequilibriumpreservation} of the algebraic PEP property, it holds for all $\boldsymbol{U}_R, \boldsymbol{U}_C, \boldsymbol{U}_L\in \mathcal{M}_{P,V}$ that, denoting $\boldsymbol{\tau}_C := \boldsymbol{\tau}(\boldsymbol{U}_C)$:
    \begin{align*}
        0 &= \nabla {p(\boldsymbol{\tau}_C)}^TD\boldsymbol{\tau}(\boldsymbol{U}_C) \Delta \boldsymbol{f}_h(\boldsymbol{U}_R,\boldsymbol{U}_C,\boldsymbol{U}_L) \\
        &= p_{\rho}(\boldsymbol{\tau}_C)\left[f_h^\rho(\boldsymbol{U}_R, \boldsymbol{U}_C) - f_h^\rho(\boldsymbol{U}_C, \boldsymbol{U}_L) \right] + p_{\varepsilon}(\boldsymbol{\tau}_C)\left[f_h^\varepsilon(\boldsymbol{U}_R, \boldsymbol{U}_C;V) - f_h^\varepsilon(\boldsymbol{U}_C, \boldsymbol{U}_L;V) \right],
    \end{align*}
    where we used \autoref{prop:energyevolution} for the internal-energy term. Setting $\boldsymbol{U}_C = \boldsymbol{U}_L$ and using consistency gives:
    \begin{align*}
        0 &= p_{\rho}(\boldsymbol{\tau}_L)\left[f_h^\rho(\boldsymbol{U}_R,\boldsymbol{U}_L) - \rho_L V \right] + p_{\varepsilon}(\boldsymbol{\tau}_L)\left[f_h^\varepsilon(\boldsymbol{U}_R,\boldsymbol{U}_L;V) - \left(V(E_L+P) - \frac12 \rho_L V^3 \right) \right]  \nonumber \\
        &= p_{\rho}(\boldsymbol{\tau}_L)\left[f_h^\rho(\boldsymbol{U}_R,\boldsymbol{U}_L) - \rho_L V \right] + p_{\varepsilon}(\boldsymbol{\tau}_L)\left[f_h^\varepsilon(\boldsymbol{U}_R,\boldsymbol{U}_L;V) - \left(V(\varepsilon_L + \frac12 \rho_L V^2 +P) - \frac12 \rho_L V^3 \right) \right] \nonumber \\[0.5em]
        &= p_{\rho}(\boldsymbol{\tau}_L)\left[f_h^\rho(\boldsymbol{U}_R,\boldsymbol{U}_L) - \rho_L V \right] + p_{\varepsilon}(\boldsymbol{\tau}_L)\left[f_h^\varepsilon(\boldsymbol{U}_R,\boldsymbol{U}_L;V) - PV -\varepsilon_LV \right]. 
    \end{align*}
    In a similar manner, it follows from setting $\boldsymbol{U}_C = \boldsymbol{U}_R$ that:
    \begin{equation*}
        p_{\rho}(\boldsymbol{\tau}_R)\left[f_h^\rho(\boldsymbol{U}_R,\boldsymbol{U}_L) - \rho_R V \right] + p_{\varepsilon}(\boldsymbol{\tau}_R)\left[f_h^\varepsilon(\boldsymbol{U}_R,\boldsymbol{U}_L;V) - PV -\varepsilon_RV \right] = 0.
    \end{equation*}
    Since $V \neq 0$, we can define:
    \begin{equation*}
        \boldsymbol{\tau}_{LR} := \begin{bmatrix}
            \displaystyle \frac{f_h^\rho(\boldsymbol{U}_R,\boldsymbol{U}_L)}{V} \\[0.8em] \displaystyle\frac{f_h^\varepsilon(\boldsymbol{U}_R,\boldsymbol{U}_L;V)}{V} - P
        \end{bmatrix},
    \end{equation*}
    and the previous two equations imply $\boldsymbol{\tau}_{LR} - \boldsymbol{\tau}_L \in \ker(\left<\nabla p(\boldsymbol{\tau}_L),\cdot \right>)$ and $\boldsymbol{\tau}_{LR} - \boldsymbol{\tau}_R \in \ker(\left<\nabla p(\boldsymbol{\tau}_R),\cdot \right>)$. But then \autoref{lem:tangentspaceorthogonality} implies:
    \begin{equation*}
        \boldsymbol{\tau}_{LR} - \boldsymbol{\tau}_L \in T_{\boldsymbol{\tau}_L} \Gamma_P, \qquad \boldsymbol{\tau}_{LR} - \boldsymbol{\tau}_R \in T_{\boldsymbol{\tau}_R} \Gamma_P.
    \end{equation*}
    Adding $\boldsymbol{\tau}_L$ to $\boldsymbol{\tau}_{LR} - \boldsymbol{\tau}_L$ and $\boldsymbol{\tau}_R$ to $\boldsymbol{\tau}_{LR} - \boldsymbol{\tau}_R$ and using the definition of tangent lines then gives:
    \begin{equation*}
        \boldsymbol{\tau}_{LR}  \in L_{\boldsymbol{\tau}_L} \Gamma_P \cap L_{\boldsymbol{\tau}_R}\Gamma_P.
    \end{equation*}
    Thus $L_{\boldsymbol{\tau}_L} \Gamma_P \cap L_{\boldsymbol{\tau}_R}\Gamma_P \neq \varnothing$. Since $\boldsymbol{U}_L,\boldsymbol{U}_R \in \mathcal{M}_{P,V}$ were arbitrary this holds for any $\boldsymbol{\tau}_L,\boldsymbol{\tau}_R \in \Gamma_P$ by \autoref{prop:equilibriumset}, completing the proof.

<!-- proof-8: body_md -->
Because $\nu > 0$ on $\mathcal{T}$ and $e \in C^2(\mathcal{S}_{\boldsymbol{\eta}})$ by the regularity assumption in \autoref{dfn:admissiblesupercriticalregime} it holds that $\boldsymbol{\Phi} \in C^2(\mathcal{S}_{\boldsymbol{\eta}})$. By \eqref{eq:thermodynamicstatespace}, \eqref{eq:pressuretemperature} and the monotonicity property in \autoref{dfn:admissiblesupercriticalregime}, the Jacobian $D\boldsymbol{\Phi}$ of $\boldsymbol{\Phi}$ has a determinant:
    \begin{equation*}
        \det D\boldsymbol{\Phi}(\boldsymbol{\eta}) = - \frac{T(\boldsymbol{\eta})}{\nu^3} < 0,  \qquad \forall \boldsymbol{\eta} \in \mathcal{S}_{\boldsymbol{\eta}}.
    \end{equation*}
    As we have shown $\boldsymbol{\Phi} \in C^2(\mathcal{S}_{\boldsymbol{\eta}})$, by the inverse-function theorem $\boldsymbol{\Phi}$ is therefore a local $C^2$ diffeomorphism. If it is also injective, $\boldsymbol{\Phi}$ is then a $C^2$ diffeomorphism onto its image $\mathcal{S}_{\boldsymbol{\tau}}$ \cite{guillemindifferential}. We will show $\boldsymbol{\Phi}$ is injective by contradiction. Assume $\boldsymbol{\Phi}$ is not injective. Then there exist $(\nu_1,\sigma_1), (\nu_2,\sigma_2) \in \mathcal{S}_{\boldsymbol{\eta}}$ for which $(\nu_1,\sigma_1) \neq (\nu_2,\sigma_2)$ as points in $\mathbb{R}^2$, so that $\boldsymbol{\Phi}(\nu_1,\sigma_1) = \boldsymbol{\Phi}(\nu_2,\sigma_2)$. Then $1/\nu_1 = 1/\nu_2$, so $\nu_1 = \nu_2 = \nu$. Also $e(\nu,\sigma_1) / \nu = e(\nu,\sigma_2) /\nu$, so $e(\nu,\sigma_1) = e(\nu,\sigma_2)$. But by the monotonicity condition $e$ is injective at fixed $\nu$, thus $\sigma_1 = \sigma_2$; a contradiction. Thus $\boldsymbol{\Phi}$ is an injective local $C^2$ diffeomorphism and therefore a $C^2$ diffeomorphism onto its image.

Denote pressure in the original $(\nu,\sigma)$-coordinates by $\tilde{p}$. In density-energy coordinates the pressure is given by $p(\boldsymbol{\tau}):=\tilde{p}\bigl(\boldsymbol{\Phi}^{-1}(\boldsymbol{\tau})\bigr)$. Since $\tilde{p}=-e_\nu \in C^1(\mathcal{S}_{\boldsymbol{\eta}})$ and $\boldsymbol{\Phi}^{-1}\in C^2(\mathcal{S}_{\boldsymbol{\tau}})$ by \autoref{prop:densityenergydiffeomorphism}, it follows that

    \begin{equation*}
        \tilde{p}_\nu(\boldsymbol{\eta})=-e_{\nu\nu}(\boldsymbol{\eta}) <0, \qquad \forall \boldsymbol{\eta}\in\mathcal{S}_{\boldsymbol{\eta}},

    and hence $\nabla_{\boldsymbol{\eta}}\, \tilde{p}(\boldsymbol{\eta})\neq\boldsymbol{0}$. Since $\tilde{p}(\boldsymbol{\eta})=p\bigl(\boldsymbol{\Phi}(\boldsymbol{\eta})\bigr)$ the chain rule yields:
    \begin{equation*}
        \nabla_{\boldsymbol{\eta}}\, \tilde{p}(\boldsymbol{\eta})
        =
        D\boldsymbol{\Phi}(\boldsymbol{\eta})^{T}
        \nabla_{\boldsymbol{\tau}}\, p\bigl(\boldsymbol{\Phi}(\boldsymbol{\eta})\bigr).
    \end{equation*}
    As $D\boldsymbol{\Phi}(\boldsymbol{\eta})$ is nonsingular and $\nabla_{\boldsymbol{\eta}}\, \tilde{p}(\boldsymbol{\eta})\neq\boldsymbol{0}$, the gradient $\nabla_{\boldsymbol{\tau}}\, p(\boldsymbol{\Phi}(\boldsymbol{\eta}))$ cannot vanish. Since $\boldsymbol{\Phi}$ maps $\mathcal{S}_{\boldsymbol{\eta}}$ onto $\mathcal{S}_{\boldsymbol{\tau}}$, the result follows.

<!-- proof-10: body_md -->
Fix some $P \in \mathbb{R}_+$ so that $\Gamma_P$ is nonempty. Let $\boldsymbol{\tau}_0 \in \Gamma_P$ be arbitrary. According to \autoref{prop:pressureregularity} at least one component of the gradient is nonzero, thus either $p_{\rho}(\boldsymbol{\tau}_0) \neq 0$ or $p_{\varepsilon}(\boldsymbol{\tau}_0) \neq 0$ holds. Assume the case $p_{\varepsilon}(\boldsymbol{\tau}_0) \neq 0$ holds. Define the function $F : \mathcal{S}_{\boldsymbol{\tau}} \rightarrow \mathbb{R}$ as:
    \begin{equation*}
        F(\boldsymbol{\tau}) := p(\boldsymbol{\tau}) - P.
    \end{equation*}
    By \autoref{prop:pressureregularity} it holds that $F \in C^1(\mathcal{S}_{\boldsymbol{\tau}})$ and:
    \begin{equation*}
        F_{\varepsilon}(\boldsymbol{\tau}_0) = p_\varepsilon(\boldsymbol{\tau}_0) \neq 0, \qquad F(\boldsymbol{\tau}_0) = 0.
    \end{equation*}
    Denote $\boldsymbol{\tau}_0 =: (\rho_0,\varepsilon_0)$. The implicit-function theorem \cite{marsdenvector} can be applied and asserts the existence of open intervals $I, K \subset \mathbb{R}$ such that $\boldsymbol{\tau}_0 \in I \times K \subset \mathcal{S}_{\boldsymbol{\tau}}$ and a function $\psi : I \rightarrow K$ such that $\psi \in C^1(I)$ and so that for all $\rho \in I$ there exists a unique $\psi(\rho) \in K$ so that:
    \begin{equation*}
        F(\rho,\psi(\rho)) = 0.
    \end{equation*}
    By uniqueness and the facts that $\boldsymbol{\tau}_0 \in I \times K$ and $F(\boldsymbol{\tau}_0) = 0$, it must hold that $\psi(\rho_0) = \varepsilon_0$. Define $J := I$ and the map:
    \begin{equation*}
        \boldsymbol{\varphi} : I \rightarrow \mathbb{R}^2, \qquad \boldsymbol{\varphi}(\rho) := \begin{bmatrix}
            \rho \\ \psi(\rho)
        \end{bmatrix}.
    \end{equation*}
    Since $\psi(\rho_0) = \varepsilon_0$, choosing $t_0 = \rho_0$ gives $\boldsymbol{\varphi}(\rho_0) = \boldsymbol{\tau}_0$. It also holds that $\boldsymbol{\varphi} \in C^1(I)$ and since $\tfrac{d\rho}{d\rho} = 1$, we have $\boldsymbol{\varphi}'(\rho) \neq 0$ for all $\rho \in I$, thus the map is regular. It remains to show that $\boldsymbol{\varphi}(I) = \Gamma_P \cap (J \times K)$ and that it has a $C^1$ inverse on its image. By the implicit-function theorem, if $\rho \in I = J$, then $\psi(\rho) \in K$ thus $\boldsymbol{\varphi}(\rho) \in J\times K \subset \mathcal{S}_{\boldsymbol{\tau}}$. Furthermore, $F(\boldsymbol{\varphi}(\rho)) = 0$ so $\boldsymbol{\varphi}(\rho) \in \Gamma_P$ by definition. Thus $\boldsymbol{\varphi}(I) \subseteq \Gamma_P \cap (J\times K)$. We will first show that $\Gamma_P$ must globally be a graph $\varepsilon(\rho)$ on $\rho$. Assume there exist two points $\boldsymbol{\tau}_1 = (\rho_1,\varepsilon_1),\boldsymbol{\tau}_2 = (\rho_2, \varepsilon_2) \in \Gamma_P$ with the same $\rho_1=\rho_2=\hat{\rho}$ coordinate, but distinct $\varepsilon_1 \neq \varepsilon_2$. Since $\Gamma_P$ is connected and $C^2$, by \cite[Corollary 3.4.6]{bergerdifferential}, there exists a regular $C^1$ curve $\boldsymbol{\gamma} = (\gamma^\rho, \gamma^\varepsilon) : [0,1] \rightarrow \Gamma_P$ so that $\boldsymbol{\gamma}_1(0) = \boldsymbol{\tau}_1$ and $\boldsymbol{\gamma}(1) = \boldsymbol{\tau}_2$. By Rolle's theorem there exists some $t^* \in (0,1)$ so that:
    \begin{equation*}
        \gamma_t^\rho(t^*) = 0.
    \end{equation*}
    Since this is the $\rho$ component of a tangent vector and the curve parameterization is regular, this implies $p_\varepsilon(\boldsymbol{\gamma}(t^*)) = 0$ by \autoref{lem:tangentspaceorthogonality}, a contradiction. Thus there exist no two distinct points on a connected $\Gamma_P$ with the same $\rho$ value. The map $\Gamma_P \ni (\rho, \varepsilon)  \mapsto \rho$ is therefore injective and by connectedness its image is an open interval $I$. The local parameterizations $\psi(\rho)$ in the proof of \autoref{prop:isobarcurves} define a global $C^2$ graph $\varepsilon(\rho)$ on $I$ so that $\Gamma_P = \{(\rho,\varepsilon(\rho)):\rho\in I\}$.

    We now prove the disjoint tangent-line characterization. Necessity is shown be contraposition. If $\varepsilon_{\rho\rho}$ does not take both signs, $\varepsilon_{\rho}$ is monotone. Tangent line slopes either are equal or unequal. Monotonocity in the equality case, $\varepsilon_{\rho}(\rho_1)=\varepsilon_{\rho}(\rho_2)$ for $\rho_1<\rho_2$, forces $\varepsilon_{\rho}$ to be constant on $[\rho_1,\rho_2]$, so all parallel tangents coincide. Tangent lines with different slopes, $\varepsilon_{\rho}(\rho_1)\neq\varepsilon_{\rho}(\rho_2)$, intersect at a unique point, so in either case the intersection is nonempty. We will only sketch the sufficiency argument. If $\varepsilon_{\rho\rho}$ takes both signs, $\varepsilon_{\rho}$ is not monotone, and there exist $\rho_1<\hat{\rho}<\rho_2$ with $\varepsilon_{\rho}(\hat{\rho})>\max\{\varepsilon_{\rho}(\rho_1),\varepsilon_{\rho}(\rho_2)\}$ or the reversed inequalities. In the former case, choose $\max\{\varepsilon_{\rho}(\rho_1),\varepsilon_{\rho}(\rho_2)\}< \varepsilon_{\rho}^*<\varepsilon_{\rho}(\hat{\rho})$ and let $(\rho_3,\rho_4)$ be the component of $\{\rho\in(\rho_1,\rho_2):\varepsilon_{\rho}(\rho)>\varepsilon_{\rho}^*\}$ containing $\hat{\rho}$. Then $\rho_1<\rho_3<\rho_4<\rho_2$, $\varepsilon_{\rho}(\rho_3)=\varepsilon_{\rho}(\rho_4)=\varepsilon_{\rho}^*$, and:
    \begin{equation*}
        [\varepsilon(\rho_4)-\varepsilon_{\rho}^*\rho_4]-[\varepsilon(\rho_3)-\varepsilon_{\rho}^*\rho_3] =\int_{\rho_3}^{\rho_4}(\varepsilon_{\rho}(\rho)-\varepsilon_{\rho}^*)\,d\rho>0.
    \end{equation*}
    Hence the tangent lines at $\rho_3$ and $\rho_4$ are parallel and distinct, thus disjoint. The reversed-inequality case is analogous.
