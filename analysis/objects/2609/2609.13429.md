<!-- bigtheorem-1: body_md -->
\label{thm:intro general theta}
Suppose $P : V_\mathbb R \to \mathbb R$ is any homogeneous polynomial of degree $N$.
The theta function
\[
 \vartheta_P( z ,L ) = y^{-N/2}
  \sum_{ m \in \mathbb Z }   \left(  \sum_{ \substack{   v \in L \\ Q(v) = m  }  }  P^\sharp\big(  \sqrt{y} \cdot  v   \big) \right)    \cdot  e^{2\pi i m z} 
\]
of the variable $z=x + iy \in \mathcal{H}$ transforms like a  modular form of weight $N+ 2^{-1} \dim(V)$, under the action of  some congruence subgroup of $\operatorname{\mathrm{SL}}_2(\mathbb Z)$. 
The congruence subgroup depends on  $(L,Q)$, but not on $P$.
<!-- end bigtheorem-1 -->
<!-- bigtheorem-2: body_md -->
\label{thm:intro main orthogonal}
Fix  an integer $N \in \mathbb Z$, and  suppose $P: V_\mathbb R^d \to \mathbb R$ is a polynomial function  satisfying the homogeneity condition 
\[
P( v g ) = \det(g)^N P(v) 
\]
 for  all $v\in V_\mathbb R^d$ and $g \in \operatorname{\mathrm{GL}}_d(\mathbb R)$.    
 The  function 
\[
 \vartheta_P (\mathtt{z} ,  L ) = 
\det(\mathtt{y})^{-N/2} 
\sum_{ T \in \operatorname{\mathrm{Sym}}_d( \mathbb Q)}  \left(   \sum_{ \substack{ v \in L^d \\ Q(v) = T } }   P^\sharp(v \alpha )    \right)   \, e^{2 \pi i \mathrm{Tr}( T \mathtt{z} ) } 
\]
of the variable $\mathtt{z} =\mathtt{x} + i \mathtt{y}\in  \mathcal{H}_d$ transforms like a Siegel modular form of weight $N+2^{-1} \dim(V)$, under the action of some congruence subgroup of $\operatorname{\mathrm{Sp}}_{2d}(\mathbb Z)$.
Here  $\alpha \in \operatorname{\mathrm{GL}}_d(\mathbb R)$  is any matrix of positive determinant such that  $\mathtt{y}=\alpha \cdot {}^t \alpha$.

<!-- definition-1: body_md -->
The  \emph{Heisenberg group} is the topological group whose underlying topological space is 
\[
H(W) = W \times F , 
\]
endowed with the group law
\[
( w_1 , t_1 ) \cdot ( w_2 , t_2 ) = \left( w_1 + w_2 , t_1 + t_2 + \frac{\langle w_1 , w_2 \rangle}{2}  \right).
\]
Note that the center of the Heisenberg group is $F\cong \{ 0 \} \times F \subset H(W)$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:SvN}
Up to isomorphism, there is a unique  continuous   irreducible  unitary Hilbert space representation $(\mathcal{S}_\psi , \rho_\psi)$ of $H(W)$ with central character $\psi$.
<!-- end theorem-1 -->
<!-- definition-2: body_md -->
\label{def:projective weil}
The \emph{projective Weil representation} is the homomorphism 
\[
M_\psi : \operatorname{\mathrm{Sp}}(W) \to \operatorname{\mathrm{Aut}} ( \mathcal{S}_\psi) / \mathbb C^1 .
\]
Here $\operatorname{\mathrm{Aut}} ( \mathcal{S}_\psi)$ is the group of Hilbert space (i.e.~unitary) automorphisms.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:abstract metaplectic} 
The \emph{metaplectic group}  is the subgroup 
\[
 \operatorname{\mathrm{Mp}}(W)_\psi  \subset  \operatorname{\mathrm{Sp}}(W) \times \operatorname{\mathrm{Aut}} ( \mathcal{S}_\psi) 
\]
of pairs $(g,r)$ for which  $r = M_\psi(g)$ in the quotient  $\operatorname{\mathrm{Aut}} ( \mathcal{S}_\psi) / \mathbb C^1$.
Projection to the first factor defines a short exact sequence
\[
1 \to \mathbb C^1 \to \operatorname{\mathrm{Mp}}(W)_\psi \to \operatorname{\mathrm{Sp}}(W ) \to 1 ,
\]
and projection to the second factor defines the \emph{Weil representation}
\[
\omega_\psi : \operatorname{\mathrm{Mp}}(W)_\psi  \to \operatorname{\mathrm{Aut}} ( \mathcal{S}_\psi) .
\]
<!-- end definition-3 -->
<!-- proposition-1: body_md -->
\label{prop:basic heisenberg}
The representation \eqref{schrodinger heisenberg}  of $H(W)$  is unitary with respect to the $L^2$ norm on $S(Y)$, and its extension to a unitary representation of $H(W)$ on  $L^2(Y)$ is irreducible with central character $\psi$.  
In particular, by the Stone--von-Neumann theorem, the representation $(L^2(Y) , \rho_\psi)$  is a model of the  Heisenberg representation.
<!-- end proposition-1 -->
<!-- definition-4: body_md -->
The representation  of $H(W)$ on $L^2(Y)$ from Proposition \ref{prop:basic heisenberg}  is the \emph{Schr\"odinger model} of the Heisenberg representation.  By the discussion surrounding Definition \ref{def:projective weil}, it determines  homomorphisms
\begin{equation}\label{M_XY}
M_\psi : \operatorname{\mathrm{Sp}}(W) \to \operatorname{\mathrm{Aut}}( L^2(Y) ) / \mathbb C^1 
\end{equation}
and 
\begin{equation*} 
\omega_\psi : \operatorname{\mathrm{Mp}}(W)_\psi \to \operatorname{\mathrm{Aut}}( L^2(Y) ) 
\end{equation*}
called the \emph{Schr\"odinger models} of the projective Weil representation and the (honest) Weil representation.
<!-- end definition-4 -->
<!-- proposition-2: body_md -->
\label{prop:rao integral}
For each $g \in \operatorname{\mathrm{Sp}}(W)$,   there is a unique positive real-valued Haar measure $d_g x$ for which the operator  \eqref{big weil}  is unitary with respect to the $L^2$ norm on $S(Y)$.  
If we extend  \eqref{big weil}  to a unitary automorphism of  $L^2(Y)$,   the composition 
\[
  \operatorname{\mathrm{Sp}}(W)   \xrightarrow{r_\psi   }  \operatorname{\mathrm{Aut}} ( L^2(Y) )  \to  \operatorname{\mathrm{Aut}} ( L^2(Y) ) / \mathbb C^1
\]
is  precisely the homomorphism \eqref{M_XY}.
<!-- end proposition-2 -->
<!-- definition-5: body_md -->
We call the  bijection of sets
\begin{equation}\label{leray coords}
\operatorname{\mathrm{Sp}}(W) \times \mathbb C^1   \cong \operatorname{\mathrm{Mp}}(W)_\psi 
\end{equation}
 sending $( g ,t ) \mapsto  (g , t  \cdot r_\psi(g)  )$  the \emph{Leray coordinates} on the metaplectic group.
Under the Leray coordinates,  the group law on the right translates to the group law
\begin{equation}\label{leray multiplication}
( g_1 , t_1 ) \cdot (g_2, t_2 ) = \big(g_1 g_2 , t_1 t_2 c_\psi(g_1,g_2) \big)
\end{equation}
on the left, and the Weil representation on $L^2(Y)$ is given by  
\begin{equation}\label{leray weil}
\omega_\psi(  g , t ) = t \cdot r_\psi(g) .
\end{equation}

<!-- proposition-3: body_md -->
\label{prop:leray reciprocity}
If $g_1,g_2 \in \operatorname{\mathrm{Sp}}(W)$ are rational points, then 
\[
\prod_v c_{\psi, v}(g_1 , g_2 ) =1,
\]
where the product is over all places of $F$, and all but finitely many factors are equal to $1$.
<!-- end proposition-3 -->
<!-- proof-1: body_md -->
This follows  from Rao's explicit formula,  \cite[Theorem 4.1]{Rao}  or \cite[Theorem 5.2]{ThetaBook},  for the local Leray cocycle  in terms of  Leray invariants and Weil indices, together with  Weil's   product formula \cite[\S 30]{Weil1} for local Weil indices.
<!-- end proof-1 -->
<!-- lemma-1: body_md -->
\label{lem:leray adelization}
If $v$ is a nonarchimedean place of odd residue characteristic for which $\psi_v$ is unramified, there is a function
$
\lambda_v : K_v^\circ \to \mu_8
$
satisfying 
\[
r_{\psi  , v}(k)   \varphi_v^\circ =  \lambda_v(k) \varphi_v^\circ
\]
for all $k\in K_v^\circ$.  
If $g \in \operatorname{\mathrm{Sp}}(W)$ is a rational point, then  for all but finitely many places $v$ we have $g \in K_v^\circ$ and $\lambda_v(g) = 1$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The first claim is  \cite[Lemma 7.10]{ThetaBook}.
One verifies the second claim about rational points  when  $g$ lies   in the Siegel parabolic of $\operatorname{\mathrm{Sp}}(W)$ stabilizing $X$, and then also  when $g$ is the order $4$ element defined by 
$e_j \mapsto - f_j$ and $f_j \mapsto e_j$.  
In both cases  the claim follows from  the explicit formulas of \cite[Lemma 7.10]{ThetaBook}.
This proves the  claim about rational points for a set of generators of $\operatorname{\mathrm{Sp}}(W)$, and the general case  follows for all elements using    \eqref{leray relation} and Proposition \ref{prop:leray reciprocity}.

<!-- definition-6: body_md -->

The \emph{adelic metaplectic group} is the subgroup  
\[
\operatorname{\mathrm{Mp}}(W_\mathbb A)_\psi  \subset \operatorname{\mathrm{Sp}}(W_\mathbb A) \times  \operatorname{\mathrm{Aut}}( S(Y_\mathbb A) )

<!-- remark-1: body_md -->
\label{rem:adelic leray coordinates}

\end{equation}
for every \emph{finite} set of places $\Sigma$   of $F$, sending 
$(g  , t)  \mapsto \big( g , t \cdot r_{\psi,\Sigma }(g) \big)$, 
where we view  $g \in \operatorname{\mathrm{Sp}}(W_\mathbb A)$  with trivial components away from  $\Sigma $.

<!-- theorem-2: body_md -->

For any $\varphi \in S( Y_\mathbb A)$, the sum

\Theta_\psi( g ,\varphi)  \stackrel{\mathrm{def}}{=}  \sum_{ y \in Y}  ( \omega_\psi(g) \varphi )(y)

converges to a continuous function of $g \in \operatorname{\mathrm{Mp}}( W_\mathbb A)_\psi$.
This function is left-invariant under the subgroup $\operatorname{\mathrm{Sp}}(W) \subset \operatorname{\mathrm{Mp}}( W_\mathbb A)_\psi$.

\end{equation}
for every \emph{finite} set of places $\Sigma$  of $F$, sending

<!-- definition-9: body_md -->
\label{def:weyl algebra}
The \emph{Weyl algebra}  $\mathcal{W}$  is the quotient of the complex tensor algebra 
\[
W_\mathbb C^{\otimes}  \stackrel{\mathrm{def}}{=}  \bigoplus_{\ell \ge 0} W_\mathbb C^{\otimes \ell} 
\]
 by the two-sided ideal generated by the subset 
 \[
\{  w_1 \otimes w_2 - w_2 \otimes w_1 - 2\pi i  \langle w_1,w_2\rangle  : w_1,w_2 \in W_\mathbb C \} .
 \]
<!-- end definition-9 -->
<!-- remark-3: body_md -->
\label{rem:weyl action}
 To be completely explicit,  choose    bases  
 \[
 e_1,\ldots, e_d \in X  \qquad \mbox{and} \qquad f_1 ,\ldots, f_d \in Y
 \]
 in such a way that  $\langle e_j,f_k\rangle = \delta_{jk}$, and  identify  $\mathbb R^d \cong Y$ by 
 $(y_1,\ldots, y_d) \mapsto y_1 f_1+\cdots+ y_d f_d$. 
 Identifying vectors in $W$ with their images in the Weyl algebra $\mathcal{W}$, our basis elements act on 
  $S(\mathbb R^d) \cong S(Y)$   through the differential operators
 \[
 \rho(e_j) =  2 \pi i  y_j  \qquad \mbox{and} \qquad \rho (f_j) =-  \frac{ \partial }{ \partial y_j }.
 \]
<!-- end remark-3 -->
<!-- remark-4: body_md -->
The injection \eqref{quantum iota} induces a homomorphism from the universal enveloping algebra of $\mathfrak{sp}(W)$ to the Weyl algebra $\mathcal{W}$,  but this is neither injective nor surjective.
The Weyl algebra could be viewed as an approximation to  the universal enveloping algebra,  particularly well-suited to studying the Weil representation.
<!-- end remark-4 -->

The  \emph{infinitesimal  Weil representation} is the Lie algebra representation 
 \[
 \omega^\mathrm{inf} = \rho \circ \iota :  \mathfrak{sp}(W)   \to \operatorname{\mathrm{End}}_\mathbb C(S( Y ) ) 
 \]
 obtained by composing \eqref{Weyl module} and \eqref{quantum iota}.

<!-- theorem-3: body_md -->
\label{thm:inf compatibility}
Under the Weil representation $\omega_\psi$  of $\operatorname{\mathrm{Mp}}(W)_\psi$ on $L^2(Y)$,  the  Schwartz functions $S(Y)$ are  precisely the $C^\infty$ vectors; that is to say, those vectors $\varphi \in L^2(Y)$ for which the orbit map 
\[
\operatorname{\mathrm{Mp}}(W)_\psi \xrightarrow{  g  \to \omega_\psi(  g ) \varphi }  L^2(Y)
\]
 is infinitely differentiable.
Under the induced differentiated action of the Lie algebra \eqref{mp lie} on $S(Y)$, 
 the  first  summand in \eqref{mp lie} acts through the infinitesimal Weil representation 
$\omega^\mathrm{inf}$ from 
Definition \ref{def:Lie Schrodinger}, and 
the second summand acts through the inclusion  $i\mathbb R \to \mathbb C$.
<!-- end theorem-3 -->
<!-- definition-11: body_md -->
\label{def:vacuum}
The \emph{vacuum vector} $\varphi^\circ_J \in S(Y)$ is 
\[
\varphi^\circ_J(y) =  e^{- \pi  \langle y , J y \rangle } .
\]
In other words, it is the restriction to $Y \subset W $ of the Schwartz function 
\[
\varphi^\circ_J (w) =  e^{   - \pi   H_J(w,w) } .
\]
<!-- end definition-11 -->
<!-- proposition-4: body_md -->
\label{prop:vacuum eigenvector} 
Under the restriction of the Weil representation,  the subgroup  $\tilde{K}_J\subset \operatorname{\mathrm{Mp}}(W)_\psi$  acts on the vacuum vector $\varphi_J^\circ$ through the unique character whose differential
$
\mathfrak{k}_J  \oplus i\mathbb R  \to i \mathbb R 
$
restricts to $\frac{1}{2} \chi_J^\mathrm{inf}$ on the first  summand, and to  the identity on the second.
<!-- end proposition-4 -->

<!-- remark-6: body_md -->
Our basis identifies  $W = \mathbb R^{2d}$, and so  identifies   $\operatorname{\mathrm{Sp}}(W)$ with the matrix group

\end{equation}

<!-- lemma-2: body_md -->
\label{lem:basic Z_jk action}
As  $j,k \in \{1,\ldots, d\}$ vary, the vectors 
\begin{equation}\label{KZs}
Z_{jk} \stackrel{\mathrm{def}}{=}  (e_j  -  i  f_j  )  \cdot ( e_k  +  i f_k   )  \in \operatorname{\mathrm{Sym}}^2(W_\mathbb C) 
\stackrel{ \eqref{sym2sp}}{\cong}\mathfrak{sp}(W_\mathbb C) 
\end{equation}
span the subalgebra $\mathfrak{k}_\mathbb C \subset  \mathfrak{sp}(W_\mathbb C)$, and satisfy
\[
\chi^\mathrm{inf} (Z_{jk})   = 2 i \delta_{jk} .
\]
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Extend $J$ to a $\mathbb C$-linear automorphism of $W_\mathbb C$, and decompose
\[
W_\mathbb C  =  W_\mathbb C^{J=-i}  \oplus W_\mathbb C^{J = i } 
  \]
into its $J$-eigenspaces. 
 The two summands  are spanned, as $\ell \in \{1,\ldots, d\}$ varies,  by the vectors $e_\ell + i f_\ell$ and $e_\ell - i f_\ell$, respectively.   

Our maximal compact subgroup $K \subset \operatorname{\mathrm{Sp}}(W)$ is precisely the subgroup of elements that preserve the direct sum decomposition, and the character $\chi$ is the composition
\[
K \to \operatorname{\mathrm{GL}}( W_\mathbb C^{J = i } ) \xrightarrow{ \det^{-1} } \mathbb C^\times.
\]
Differentiating this, the linear functional $\chi^\mathrm{inf}$ is 
\[
\mathfrak{k}_\mathbb C \cong \mathfrak{gl}( W_\mathbb C^{J = i }  ) \xrightarrow{  - \mathrm{Tr} } \mathbb C .
\]

The lemma  follows   by directly calculating the action of each $Z_{jk}$ on $W_\mathbb C$, which is by
\begin{align*}
Z_{jk} \cdot (  e_\ell + i f_\ell ) & =    2 i \delta_{j \ell }  \cdot (  e_k + i f_k ) \\ 
Z_{jk} \cdot (  e_\ell - i f_\ell ) & =  -2 i \delta_{k \ell }  \cdot (  e_j - i f_j )  .
\end{align*}
From these formulas it is clear that $Z_{jk}$ preserves the  decomposition of $W_\mathbb C$, so lies in $\mathfrak{k}_\mathbb C$, and acts on the summand $W_\mathbb C^{J=i}$ with trace $-2 i \delta_{jk}$.

\label{prop:symplectic-orthogonal weil formulas}
Fix $A \in \operatorname{\mathrm{GL}}_d(\mathbb A)$ and $B \in \operatorname{\mathrm{Sym}}_d(\mathbb A)$, and define elements of $\operatorname{\mathrm{Sp}}_{2d}(\mathbb A) \cong \operatorname{\mathrm{Sp}}(W_\mathbb A)$ by 
\[
m(A)  = \begin{pmatrix}  A  &  \\  & {}^t A ^{-1}  \end{pmatrix} 
\qquad \mbox{and} \qquad
n(B) = \begin{pmatrix}   I_d  & B  \\ &  I_d  \end{pmatrix} .
\]
Assume that the local components of both $A$ and $B$ are trivial at all places outside some finite set of places $\Sigma$ of $F$, so that we may use the partial  Rao coordinates  \eqref{semi rao coordinates} to form 
\[
( m(A) , \epsilon ) \in \operatorname{\mathrm{Mp}}(W_\mathbb A)^{(2)}_\psi \quad \mbox{and}\quad ( n(B) , \epsilon ) \in \operatorname{\mathrm{Mp}}(W_\mathbb A)^{(2)}_\psi

\[
\omega_{V,\psi} (m(A), \epsilon )   \varphi (v) 
=   \epsilon^{\dim(V)}  \cdot   \beta_\Sigma   ( m(A) )  
\cdot   |  \det(A)  |^{ \frac{1}{2} \dim_F(V) }   \cdot  \varphi(   v  A ) 
\]
and 
\[
\omega_{V,\psi} (n(B) , \epsilon) \varphi (v) 
=  \epsilon^{\dim(V)}  \cdot    \psi(   \mathrm{Tr}( Q(v)  B )  ) \cdot  \varphi(  v )  . 
\]
In the first formula the tuple $v A \in V_\mathbb A^d$ is computed by viewing $v \in V_\mathbb A^d$ as a row vector, and multiplying it by the matrix $A$ in the usual way. 
The modulus function $| \cdot |$ on $\mathbb A^\times$  is normalized by 
\[
\int_\mathbb A f(ax) \, dx = |a| ^{-1}  \int_\mathbb A f(x)\, dx.
\]

Write $g \in \operatorname{\mathrm{Sp}}(W_\mathbb A)$ for the symplectic automorphism corresponding to the matrix $m(A)$, under the convention explained in \S \ref{ss:so setup}.
  As in the definition of \eqref{r_XY}, its  inverse $g^{-1}$ is encoded  by  linear maps $a : X_\mathbb A \to X_\mathbb A$ and $d : Y_\mathbb A\to Y_\mathbb A$. 
Using the bases $e_1,\ldots , e_d$ and $f_1,\ldots, f_d$, we identify elements of $X_\mathbb A \cong \mathbb A^d$ and $Y_\mathbb A \cong \mathbb A^d$ as row vectors, so that $a$ and $d$ are expressed in terms of matrix multiplication by
\begin{align*}
a(x) & = g^{-1}(x)  = x\cdot {}^t A^{-1}  \\
d(y) & = g^{-1} (y) = y \cdot A.
\end{align*} 


Now consider  $i_V(g) \in \operatorname{\mathrm{Sp}}(\bm{W}_\mathbb A)$. Its inverse is encoded by the linear maps
\begin{align*}
  \bm{X} _\mathbb A =  X_\mathbb A \otimes V_\mathbb A   &  \xrightarrow{  a \otimes \mathrm{id}  }    X_\mathbb A \otimes V_\mathbb A  \cong \bm{X}_\mathbb A  \\
    \bm{Y} _\mathbb A =  Y_\mathbb A \otimes V_\mathbb A   &  \xrightarrow{  d \otimes \mathrm{id}  }    Y_\mathbb A \otimes V_\mathbb A  \cong \bm{Y}_\mathbb A  .
  \end{align*} 

Tracing the definition of $\bm{r}_{\psi , \Sigma}$ all the way back to the integral  \eqref{big weil}, we find that for any $y \otimes v \in  Y_\mathbb A \otimes V_\mathbb A$ we have
\[
\bm{r}_{ \psi , \Sigma } (  i_V(g)  )  \varphi     ( y \otimes v  ) = 
 |  \det(A)  |^{ \frac{1}{2} \dim_F(V) }

The normalizing factor $ |  \det(A)  |^{ \frac{1}{2} \dim_F(V) }$ is  included to make the operator unitary, as per Proposition \ref{prop:rao integral}.
 When we identify a tuple

\bm{r}_{ \psi , \Sigma } (  i_V(g)  )   \varphi     ( v ) =

<!-- proposition-7: body_md -->
\label{prop:siegel transformation}
The function \eqref{classical siegel}  satisfies  the transformation law
\[
F(\gamma \mathtt{z} ) =  j( \gamma ,\mathtt{z})^m F(\mathtt{z})

<!-- end proposition-7 -->
<!-- proof-7: body_md -->
This is well-known \cite[\S 6.1]{pitale}, but we formulate the proof in a way that  points  toward the  correct definition  of  $j(\gamma,\mathtt{z})^m$ when $m$ is allowed to be a half-integer.

Under  the action of $\operatorname{\mathrm{Sp}}_{2d}(\mathbb R)$ on $\mathcal{H}_d$,  we have   $g_\mathtt{z}\cdot i I_d = \mathtt{z}$.
Hence  $g_\mathtt{z}^{-1}  \gamma^{-1}  g _{\gamma \mathtt{z} }  \in K$ for all  $\gamma \in \operatorname{\mathrm{Sp}}_{2d}(\mathbb R)$,  allowing us to define 
\begin{equation}\label{circle cocycle}
c( \gamma , \mathtt{z} ) =   \chi ( g_\mathtt{z}^{-1}  \gamma^{-1}  g _{\gamma \mathtt{z} }  )   .
\end{equation}



Now  suppose  $\gamma \in \Gamma$.
  By definition of $\Gamma$, there is a $u  \in U$ such that $\gamma  u $ lies in the image of the  natural embedding $\operatorname{\mathrm{Sp}}(W) \to \operatorname{\mathrm{Sp}}(W_\mathbb A)$.  
Because the function $F(g)$  is left invariant under this image, and right invariant under $U$, we have 
\[
F( g_\mathtt{z} )  =  F( \gamma   u  \cdot g_\mathtt{z} ) = F( \gamma  \cdot  g_\mathtt{z}  \cdot  u ) =  F( \gamma  g_\mathtt{z}  ) .
\]
On the other hand, because $g_\mathtt{z}^{-1}  \gamma^{-1}  g _{\gamma \mathtt{z} }  \in K$, the assumption \eqref{siegel weight} implies 
\[
c(\gamma, \mathtt{z})^m  \cdot  F( \gamma  g_\mathtt{z}  ) 
= \chi (  g_\mathtt{z}^{-1}  \gamma^{-1}  g _{\gamma \mathtt{z} }  )^m   \cdot  F( \gamma  g_\mathtt{z}  ) 
= F( g _{\gamma \mathtt{z} }  ) .
\]
This proves  the equality
$
F( g_{ \gamma \mathtt{z} } ) = c(\gamma,\mathtt{z})^{m} F(g_\mathtt{z}). 
$


The proposition now follows from 
\[
 j( \gamma , \mathtt{z} ) 
= 
c( \gamma , \mathtt{z} ) 
 \cdot 
 \left(   \frac{\det  \mathrm{Im}  ( \mathtt {z} )     }  {  \det  \mathrm{Im}(  \gamma \mathtt{z} )   } \right)^{1/2} .
\]
One verifies this last equality by direct calculation  for   $\mathtt{z} = i I_d$,  then  deduces  the general case  using the cocycle relation \eqref{siegel automorphy}, which is also satisfied  by   $c(\gamma, \mathtt{z})$. The details are left to the reader.

<!-- proposition-8: body_md -->
\label{prop:good half cocycle}
The function 
\[
j(  \gamma , \mathtt{z} )^m  \stackrel{\mathrm{def}}{=}
c(   \gamma   , \mathtt{z} )^m
 \cdot 
 \left(   \frac{\det( \mathrm{Im}(   \mathtt{z} ) )    }  {  \det( \mathrm{Im}(  \gamma \mathtt{z} ) )  } \right)^{m/2} 
\]
of $\gamma \in \operatorname{\mathrm{Mp}}(W_\mathbb R)_\psi^{(2)}$ and $\mathtt{z} \in \mathcal{H}_d$  satisfies the following properties: 
\begin{enumerate}
\item
It is well-defined, meaning independent of the choice of  $\alpha$ used in the definition of \eqref{natural g_z}.  
\item
It  is holomorphic as a function of $\mathtt{z} \in \mathcal{H}_d$.
\item
It satisfies the cocycle relation
$
 j(  \gamma_1  \gamma_2 , \mathtt{z} )^m  = j( \gamma_1 ,  \gamma_2 \mathtt{z} )^m  \cdot j(   \gamma_2 , \mathtt{z} )^m.
$
\item
If $m\in \mathbb Z$, then $j(  \gamma , \mathtt{z})^m$ depends only on the image of $\gamma$ in  $\operatorname{\mathrm{Sp}}(W_\mathbb R)$, and agrees with the   $j(\gamma , \mathtt{z})^m$ already defined by \eqref{integral j}.
\end{enumerate}
<!-- end proposition-8 -->
<!-- proof-8: body_md -->
If we can prove that $j(\gamma ,\mathtt{z})^m$ is well-defined and continuous as a function of $\mathtt{z}$, then everything else follows easily.
The cocycle relation asserted in (3) follows from the analogous relation for \eqref{half circle cocycle}, which is clear from the definition.  
If $m$ is an integer,  the  character \eqref{half-character} factors through $K^{(2)} \to K$, and agrees with the $m^\mathrm{th}$-power 
of the character $\chi : K \to \mathbb C^1$.  Hence the function $c(\gamma,\mathtt{z})^m$  defined by \eqref{half circle cocycle} only depends on the image of $\gamma$ in $\operatorname{\mathrm{Sp}}(W_\mathbb R)$, and agrees with the $m^\mathrm{th}$ power of  \eqref{circle cocycle}.
Property (4) follows immediately.  
From property  (4) we deduce that the square of $j( \gamma , \mathtt{z})^m$ is holomorphic.  
If  $j( \gamma , \mathtt{z})^m$ is continuous, it must  be holomorphic as well.


The key to proving that $j(\gamma ,\mathtt{z})^m$ is well-defined and continuous is the following lemma.


\begin{lemma}\label{lem:siegel section}
As in Remark \ref{rem:beta on parabolics},  let $P^+_\mathbb R \subset \operatorname{\mathrm{Sp}}(W_\mathbb R)$ denote the identity component of the Siegel parabolic.
The composition
\[
P_\mathbb R^+ \xrightarrow{  p \mapsto \tilde{p}   }  \operatorname{\mathrm{Sp}}(W_\mathbb R) \times \mu_2   \stackrel{ \eqref{rao coords}}{\cong}  \operatorname{\mathrm{Mp}}(W_\mathbb R)_\psi^{(2)} ,
\]
where we have set $\tilde{p} = ( p,1)$,  is a continuous group homomorphism. 
\end{lemma}

\begin{proof}
To say that $p\mapsto \tilde{p}$ is a group homomorphism means precisely that  the archimedean Rao cocycle satisfies $c_{\psi}^{(2)}( p_1 , p_2) =1$ for all $p_1,p_2 \in P^+_\mathbb R$.  
This follows from the explicit formula of \cite[Corollary 5.16(c)]{ThetaBook}.
See also Rao \cite[Corollary 5.5(2)]{Rao}, as per the caveat of  Remark \ref{rem:ThetaBook caveat}.


For the continuity claim, unwinding the definitions shows that  $\tilde{p}$ is another name for the pair 
\[
\big( p ,  r^{(2)}_\psi(p)  \big)  \in  \operatorname{\mathrm{Mp}}(W_\mathbb R)_\psi^{(2)}  \subset \operatorname{\mathrm{Sp}}(W_\mathbb R) \times \operatorname{\mathrm{Aut}} (L^2(Y_\mathbb R) ) ,
\]
where $r^{(2)}_\psi(p) = m_\psi(p) r_\psi(p) $ as in \eqref{rao correction}.   
For  $p \in P_\mathbb R^+$,  Rao's normalizing factor satisfies $m_\psi(p)=1$.
So  really the claim is that  $r_\psi(p)  \in \operatorname{\mathrm{Aut}} (L^2(Y_\mathbb R) )$ varies continuously, in a suitable sense, with $p\in P_\mathbb R^+$. 
As per Weil's characterization \cite[\S 35]{Weil1} of the topology on the metaplectic group, 
the suitable sense is with respect to the strong operator topology on $\operatorname{\mathrm{Aut}} (L^2(Y_\mathbb R) )$.
A proof of the continuity of $r_\psi(p)$  as a function of $p$ (on all of $P_\mathbb R$, not just $P_\mathbb R^+$)  can be found in \cite[Chapter I.7]{Igusa}.
\end{proof}



Let us now prove (1).   As in Remark \ref{rem:alpha ambiguity}, making a different choice of $\alpha$ amounts to multiplying it on the right by some $A \in \mathrm{SO}(d)$, which multiplies $g_\mathtt{z}$ on the right by the element 
\[
m(A) \in P_\mathbb R^+ \cap \ker(  \chi : K \to \mathbb C^1 )  \subset \operatorname{\mathrm{Sp}}_{2d}(\mathbb R) .
\]
Denoting the continuous homomorphism of Lemma \ref{lem:siegel section}  by $p \mapsto \tilde{p}=(p,1)$, we see that 
making a different choice of $\alpha$ multiplies  $\tilde{g}_\mathtt{z}$ on the right by $\tilde{m}(A)$.
This multiplies   \eqref{half circle cocycle}  by the inverse of $\chi^m ( \tilde{m}(A) )$, which is $\pm 1$ because  
\[
\chi^m ( \tilde{m}(A) )^2 = \chi( m(A)  )^{2m} =1.
\]
The point is now simply that $A \mapsto \chi^m( \tilde{m}(A) )$ is a continuous homomorphism $\mathrm{SO}(d) \to \mu_2$, which must be the  trivial homomorphism because $\mathrm{SO}(d)$ is connected.  
This proves that   \eqref{half circle cocycle}  does not depend on the choice of $\alpha$ used to define $g_\mathtt{z}$.
By the same reasoning it is independent of the analogous choice made to define $g_{\gamma \mathtt{z}}$.
In other words $c(\gamma , \mathtt{z})^m$, hence also  $j( \gamma , \mathtt{z})^m$,  is well-defined.

To complete the proof of Proposition \ref{prop:good half cocycle},  choose the elements \eqref{natural g_z} to vary continuously with $\mathtt{z}$, for example by taking $\alpha$ to be the unique positive definite symmetric square root of $\mathtt{y}$.  Then use the  continuity claim of Lemma \ref{lem:siegel section} to see that \eqref{half circle cocycle} is continuous as a function of  $\mathtt{z}$, and so the same is true of $j( \gamma , \mathtt{z})^m$.

\label{prop:hermitian transformation}
 For all $\gamma \in  \Gamma  \cap \mathrm{SU}_{d,d}(\mathbb R)$, the function \eqref{classical unitary} satisfies  the transformation law

<!-- proof-19: body_md -->
Fix  $\gamma \in \mathrm{U}_{d,d}(\mathbb R)$.
We must have
\[
g_\mathtt{z}^{-1}  \gamma^{-1} g_{\gamma \mathtt{z}} = [ k_1 , k_2] 
\]
for some  $k_1,k_2 \in \mathrm{U}(d)$, because the left hand side fixes $i I_d \in \mathcal{H}_d$, hence lies in $K$. 
 If  we define 
\[
c(\gamma , \mathtt{z}) = \det(k_1)^{  m_1 } \det(k_2)^{ m_2 } , 
\]  
then  the same reasoning as in the proof of  Proposition \ref{prop:siegel transformation} shows that
\begin{equation}\label{unitary circle transform}
F( g_{ \gamma \mathtt{z} } ) = c(\gamma,\mathtt{z}) F(g_\mathtt{z}) .
\end{equation}

To relate $c(\gamma,\mathtt{z})$  to $j(\gamma, \mathtt{z})$, first note that
\begin{equation}\label{C conjugation}
[k_1,k_2] = C \begin{pmatrix} k_1 \\ & k_2 \end{pmatrix} C^{-1}
\quad \mbox{in which} \quad C = \frac{1}{\sqrt{2}} \begin{pmatrix}   I_d & I_d \\ i I_d & -i I_d  \end{pmatrix}   ,
\end{equation}
and hence    $\det( [ k_1,k_2] ) = \det(k_1) \det(k_2)$.
Combining this with Remark  \ref{rem:su g_z} shows that
\[
\det(\gamma)^{-1} = \det( [ k_1,k_2] ) = \det(k_1) \det(k_2) ,
\]
and we deduce
\begin{equation}\label{alt c}
c(\gamma , \mathtt{z}) = \det(\gamma)^{ - m_1 } \det(k_2)^{ m_2 -m_1 } .
\end{equation}


Now write  $\mathtt{y}(\mathtt{z}) = (2i)^{-1} ( \mathtt{z} - {}^\dagger\mathtt{z} )$ for the $\mathtt{y}$-part of $\mathtt{z}$.
Using the factorization $g_{ \gamma \mathtt{z}} = \gamma g_\mathtt{z} [ k_1, k_2 ]$ and the cocycle relation for $j$, one finds 
\[
j( g_{ \gamma \mathtt{z}}  , i I_d)  = j( \gamma ,\mathtt{z} ) \cdot  j( g_\mathtt{z} , i I_d )   \cdot j( [ k_1,k_2] , i I_d ).
\]
Directly from the definition of $j$, we have the equalities 
\begin{align*}
j( g_{ \gamma \mathtt{z}}   ,  i I_d)  & = \det( \mathtt{y}(\gamma \mathtt{z} ))^{ -1/2}  \\
j( g_\mathtt{z} , i I_d ) &  = \det( \mathtt{y}(\mathtt{z}) )^{-1/2}  \\
j( [ k_1,k_2] , i I_d )  & = \det(k_2) ,
\end{align*}
from which we deduce
\begin{align*}
\det(k_2)     =  j( \gamma ,  \mathtt{z} )^{-1}  
\left( \frac{ \det  \mathtt{y} (\mathtt{z})  }{ \det  \mathtt{y} ( \gamma \mathtt{z} ) }   \right)^{1/2} .
\end{align*} 

Plugging this last expression for $\det(k_2)$ into \eqref{alt c} shows that
\[
j( \gamma ,  \mathtt{z} )^{m_1-m_2}  
= c(\gamma,\mathtt{z})   \det(\gamma)^{  m_1 }  
 \left( \frac{ \det  \mathtt{y} (\mathtt{z})  }{ \det  \mathtt{y} ( \gamma \mathtt{z} ) }   \right)^{ \frac{m_1-m_2}{2} }   ,
\]
and from this and \eqref{unitary circle transform} we deduce that   \eqref{classical unitary} satisfies  
\[
F(\gamma \mathtt{z} ) =  \det( \gamma )^{-m_1}   j  ( \gamma ,\mathtt{z})^{m_1-m_2}  F(\mathtt{z})
\]
for all $\gamma \in  \Gamma$, proving (a stronger version of) the claim.

\begin{lemma}

\end{equation}

\[

\]

\]
The notation $\det_{\bm{J}}$ means the determinant is being computed  using  the complex structure $\bm{J}$ (and \emph{not} the action of $E_\mathbb R=\mathbb C$) to regard $\bm{W}_\mathbb R$ as a complex vector space.

\label{thm:main unitary}
Fix  an integer $N \in \mathbb Z$, and  suppose $P$ is a polynomial function on the real vector space  $V^d_\mathbb R$ satisfying 
\begin{equation}\label{unitary P homogeneity}
P( v g ) = | \det(g)| ^{2N} P(v) 
\end{equation}
 for  all $v\in V_\mathbb R^d$ and $g \in \operatorname{\mathrm{GL}}_d(\mathbb C)$.    
 For any Schwartz function $\varphi_f \in S(V^d_{\mathbb A_f})$,   the  function 
\[
\vartheta_P (\mathtt{z} ,  \varphi_f ) = 
\det(\mathtt{y})^{-N} 
\sum_{ T \in \operatorname{\mathrm{Herm}}_d( E )}  \left(   \sum_{ \substack{ v \in V^d \\ H(v) = T } }   P^\sharp(v \alpha ) \varphi_f(v  )   \right)   \, e^{2 \pi i \mathrm{Tr}( T \mathtt{z} ) } 
\]
of the variable $\mathtt{z} =\mathtt{x} + i \mathtt{y}\in  \mathcal{H}_d$ transforms like a Hermitian  modular form of weight $2N+ \dim_E(V)$, under the action of some arithmetic subgroup of $\mathrm{SU}_{d,d}(\mathbb Q)$.
Here  $\alpha \in \operatorname{\mathrm{GL}}_d(\mathbb C)$  is any matrix of positive real determinant such that  $\mathtt{y}=\alpha \cdot {}^\dagger \alpha$.

\label{rem:shimura polynomials}
If $\Delta P =0$,  then $P^\sharp =P$, and Theorem \ref{thm:main unitary} shows  that 
\[

\sum_{ T \in \operatorname{\mathrm{Herm}}_d( E )}  \left(   \sum_{ \substack{ v \in V^d \\ H(v) = T } }   P(v ) \varphi_f(v  )   \right)   \, e^{2 \pi i \mathrm{Tr}( T \mathtt{z} ) }

Similar results have been proved by Shimura  in \cite[Appendix A7]{ShimuraEisenstein} and \cite[Appendix A5]{Shimura2000}, but this precise result is not among them.  
Shimura  works with theta series for Hermitian spaces of arbitrary signature.  
When specialized to the positive definite case that we work in, Shimura's results prove the modularity  only of  theta functions weighted by polynomial functions on $V_\mathbb R$ that are either holomorphic or anti-holomorphic.   
Nonconstant polynomials satisfying   \eqref{unitary P homogeneity} do not fall into either class, so it seems that  even the special case $\Delta P=0$ of Theorem \ref{thm:main unitary} has not previously appeared explicitly in the literature.

In  the coordinates $x_{\alpha j}$ and $y_{\alpha j}$ on $V_\mathbb R^d$ determined by our orthonormal basis of $V_\mathbb R$, the standard Gaussian $\varphi_\mathbb R^\circ \in S(V_\mathbb R^d)$ from the proof of Theorem \ref{thm:hermitian theta modularity} becomes

\varphi^\circ_\mathbb R(v)  =  e^{-2\pi   \mathrm{Tr}(  H(v ) ) } 
= \prod_{ j =1}^d  e^{ - 2 \pi   \sum_\alpha  | z_{\alpha j} |^2     } .

As in earlier arguments, if we can show that $\varphi_\mathbb R=P^\sharp \varphi^\circ_\mathbb R$ satisfies  the transformation law \eqref{unitary-unitary K weight} with

m = 2 N + \dim_E(V),

then we are done by  Proposition \ref{prop:hermitian theta expansion}.




 Return to the setup of the proof of Theorem \ref{thm:hermitian theta modularity}, so that $K=K_1\times K_2$ is the stabilizer in $\mathrm{U}(W_\mathbb R)$ of the eigenspace  decomposition

W_\mathbb R  = W_\mathbb R^{J = i }  \oplus W_\mathbb R^{J = - i }  .

Consider the corresponding Lie algebra decomposition 
$\mathfrak{k} = \mathfrak{k}_1\oplus \mathfrak{k}_2 \subset \mathfrak{u}(W_\mathbb R)$.  




By directly computing the images of the vectors \eqref{XsYs} under \eqref{herm2u}, we see that they act on $W_\mathbb R$ as
\begin{align*}
X^{\prime}_{jk} \cdot ( e_\ell - i f_\ell  )  
& = 
- 2i \delta_{ j \ell }( e_k - i f_k )  - 2i \delta_{ k  \ell }( e_j - i f_j )   \\
X^{\prime}_{jk} \cdot ( e_\ell + i f_\ell  )  
& = 0 \\
Y^{\prime}_{jk} \cdot ( e_\ell - i f_\ell  )  
& = 
 2  \delta_{ j \ell }( e_k - i f_k )  - 2 \delta_{ k  \ell }( e_j - i f_j )   \\
Y^{\prime}_{jk} \cdot ( e_\ell + i f_\ell  )  
& = 0

and
 \begin{align*}

& =   0  \\
X^{\prime\prime}_{jk} \cdot ( e_\ell + i f_\ell  )  
& =    2i \delta_{ j \ell }( e_k + i f_k )  + 2i \delta_{ k  \ell }( e_j + i f_j )     \\
Y^{\prime\prime}_{jk} \cdot ( e_\ell - i f_\ell  )  
& =  0   \\
Y^{\prime\prime}_{jk} \cdot ( e_\ell + i f_\ell  )

From these formulas and \eqref{K1K2}, one sees that

\mathfrak{k}_1  = \mathrm{Span}_\mathbb R\{ X'_{jk} , Y'_{jk} \}  \qquad \mbox{and}\qquad

and that  the traces of these vectors acting on $W_\mathbb R$, regarded as a vector space over $E_\mathbb R=\mathbb C$, are given by

 \mathrm{Tr}_W ( X^{\prime}_{jk}  ) & = - 4 i \delta_{jk} 
&
 \mathrm{Tr}_W ( X^{\prime\prime}_{jk}  ) & =  4 i \delta_{jk}     \\ 
 \mathrm{Tr}_W ( Y^{\prime}_{jk}  ) & = 0
& 
\mathrm{Tr}_W ( Y^{\prime\prime}_{jk}  ) & = 0   . \nonumber

Using the formulas of Lemma \ref{lem:unitary XY operators}, a tedious calculus exercise shows that 
\begin{align*}
   \omega^\mathrm{inf}(  i_V  ( X^{\prime}_{jk}) )  (f \varphi^\circ_\mathbb R )
  &  =  2 i \cdot 
\left(        E^{\prime}_{jk} f  + E^{\prime}_{kj}f     -     \frac{ 1 }{4\pi }      F_{jk} f  +    \dim_E(V)  \delta_{jk}    f    \right)  \cdot    \varphi^\circ_\mathbb R     \\

for all $Z_1 \in \mathfrak{k}_1$ and $Z_2 \in \mathfrak{k}_2$.



Now consider the action of $K=K_1\times K_2$ on $S(V_\mathbb R^d)$ through the Weil representation  $\omega^\eta_{V,\psi} = \omega_\psi \circ i^\eta_{V,\psi}$ of Definition \ref{def:unitary-unitary weil}.

Z \mapsto \frac{ r_\eta }{2}  \cdot  \mathrm{Tr}_W(Z) 
+ (  \omega^\mathrm{inf} \circ i_V) (Z)  .

This is a consequence of  Theorem \ref{thm:inf compatibility}  and Lemma \ref{lem:kinf correction}.

\]
Recalling Remark \ref{rem:banana peel},  this shows that $K=K_1\times K_2$ acts on $P^\sharp \varphi^\circ_\mathbb R$ through the character in  \eqref{unitary-unitary K weight}, as was needed to prove the theorem.
