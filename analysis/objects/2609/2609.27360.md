<!-- con-1: body_md -->
\label{con: main conjecture}
    If the prime $\ell = \mathrm{char}(k)$ is pretty good\footnote{By this we mean that there exists a reductive algebraic group $G/k$ such that $\operatorname{Lie}(G) = \mathfrak{g}$ and $\ell$ is pretty good for $G$. We refer to \cite[Definition 2.11]{Herpel2013} for the definition of "pretty good". We note that "pretty good" is stronger than "good" but weaker than "very good".} for $\mathfrak{g}$, then the scheme $\mathrm{Comm}_{\mathfrak{g}}$ is normal and Cohen--Macaulay.
<!-- end con-1 -->
<!-- thm-1: body_md -->
\label{thm: intro main theorem}
    Let $k$ be a field of characteristic $\ell$ and let $G/k$ be a connected reductive algebraic group, and let $\mathfrak{g}$ denote its Lie algebra. If $\ell \neq 0$, suppose that $\ell$ is larger than the Coxeter numbers of the simple factors of $G$, and that $\ell \neq 19$ (resp. $\ell \neq 31$) when $G$ has a simple factor of type $E_7$ (resp. $E_8$). Then the schemes $\mathrm{Comm}_G, \mathrm{Comm}_{\mathfrak{g}}$ are normal and Cohen--Macaulay.
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\label{thm: intro t-exact CM}
    Suppose that Assumption \ref{assump: local geom Langlands} holds, and that every elementary $\ell$-subgroup of $G(\mathbb{F}_q)$ is contained within a torus. Then 
    \begin{enumerate}
        \item The functor 
        \[
            \mathcal{F}_{G,\psi}^{\mathrm{unip},c}\colon \mathrm{Rep}^{\widehat\mathrm{unip}}_c(G(\mathbb{F}_q),\overline{\mathbb{F}}_\ell)\to \mathrm{Coh}(\mathrm{LS}^{\widehat\mathrm{unip}}_{\hat{G}})
        \]
        is $t$-exact when both sides are endowed with the standard $t$-structures. 
        \item For any object $V\in \mathrm{Rep}_c^{\widehat\mathrm{unip}}(G(\mathbb{F}_q),\overline{\mathbb{F}}_\ell)^\heartsuit$, the associated coherent sheaf $\mathcal{F}_{G,\psi}^{\mathrm{unip},c}(V)$ is maximal Cohen--Macaulay.
    \end{enumerate}

<!-- lemma-1: body_md -->
\label{lemma: tilde R_1 inj proj}
    The $G(\mathbb{F}_q)$-representation $\widetilde{R}_1$ is projective and injective. The representation $\widetilde{R}_1$ is a finite iterated extension of copies of $R_1$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $V\in \mathrm{Rep}(G(\mathbb{F}_q),\overline{\mathbb{F}}_\ell)^{\heartsuit}$, we have 
    \[\mathrm{Hom}_{G(\mathbb{F}_q)}(\widetilde{R}_1,V)=V^{B(\mathbb{F}_q)^{(\ell)}}\quad\text{and} \quad\mathrm{Hom}_{G(\mathbb{F}_q)}(V,\widetilde{R}_1)=(V_{B(\mathbb{F}_q)^{(\ell)}})^\vee\]
    by Frobenius reciprocity. Therefore $\mathrm{Hom}(\widetilde{R}_1,-),\mathrm{Hom}(-,\widetilde{R}_1)$ are both $t$-exact as $B(\mathbb{F}_q)^{(\ell)}$ has order prime to $\ell$. The second assertion is clear as we can write
    \[
        {\mathrm{Ind}}_{B(\mathbb{F}_q)^{(\ell)}}^{G(\mathbb{F}_q)}\overline{\mathbb{F}}_\ell={\mathrm{Ind}}_{B(\mathbb{F}_q)}^{G(\mathbb{F}_q)}(\overline{\mathbb{F}}_\ell[T(\mathbb{F}_q)/T(\mathbb{F}_q)^{(\ell)}]),
    \]
    where $\overline{\mathbb{F}}_\ell[T(\mathbb{F}_q)/T(\mathbb{F}_q)^{(\ell)}]$ is a finite iterated extension of copies of $\overline{\mathbb{F}}_\ell$ as $T(\mathbb{F}_q)$-representations.

<!-- lemma-2: body_md -->
\label{lemma: R_1 ss}
    Assume that $\ell\mid q-1$ and $\ell \nmid |W|$. Then the $G(\mathbb{F}_q)$-representation $R_1$ is semisimple.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
First, the endomorphism ring $\mathrm{End}(R_1)=\overline{\mathbb{F}}_\ell[B(\mathbb{F}_q)\backslash G(\mathbb{F}_q)/B(\mathbb{F}_q)]$ is isomorphic to the group algebra $\overline{\mathbb{F}}_\ell[W]$ by assumption, and hence is semisimple. Therefore we can write $R_1$ as a direct sum
    \[
        R_1=\bigoplus_{i=1}^n M_i^{\oplus m_i}
    \]
    where $\mathrm{End}(M_i)=\overline{\mathbb{F}}_\ell$ and $\mathrm{Hom}(M_i,M_j)=0$ for $i\neq j$.
    
    We claim that any irreducible quotient $L$ of $R_1$ is also an irreducible subrepresentation of $R_1$. Indeed, by Frobenius reciprocity we have $L^{B(\mathbb{F}_q)}\neq 0$. Since $U(\mathbb{F}_q)$ is a $p$-group, we know that $L^{U(\mathbb{F}_q)}=L_{U(\mathbb{F}_q)}$ with the same $T(\mathbb{F}_q)$-action. Since $T(\mathbb{F}_q)$ is abelian, a finite dimensional representation of $T(\mathbb{F}_q)$ with a trivial subrepresentation also has a trivial quotient. Therefore $L_{B(\mathbb{F}_q)}\neq 0$ and hence there exists an injective map $L\hookrightarrow R_1$.

    Now let $M_i\twoheadrightarrow L$ be an irreducible quotient. By the preceding argument, $L$ is also an irreducible subrepresentation of some  $M_j$. In particular, there is a non-zero map $M_i\hookrightarrow L\twoheadrightarrow M_j$. Hence we necessarily have $i=j$. Then the map $M_i\twoheadrightarrow L\hookrightarrow M_i$ is invertible because $\mathrm{End}(M_i)=\overline{\mathbb{F}}_\ell$. Therefore $M_i\simeq L$ is simple.
<!-- end proof-2 -->

\label{prop: principle block generator}
    Assume that $\ell\mid q-1$ and $\ell\nmid |W|$. Then $\mathrm{Rep}^0(G(\mathbb{F}_q),\overline{\mathbb{F}}_\ell)$ is a presentable stable subcategory of $\mathrm{Rep}(G(\mathbb{F}_q),\overline{\mathbb{F}}_\ell)$, and there is a direct sum decomposition
    \[
        \mathrm{Rep}(G(\mathbb{F}_q),\Lambda)=\mathrm{Rep}^0(G(\mathbb{F}_q),\Lambda)\oplus \mathrm{Rep}^{>0}(G(\mathbb{F}_q),\Lambda)
    \]
    of presentable stable $\overline{\mathbb{F}}_\ell$-linear categories.
<!-- end prop-1 -->
<!-- proof-3: body_md -->
We first apply Lemma \ref{lemma: decomp of abelian cat} to $\mathcal{A}=\mathrm{Rep}_c(G(\mathbb{F}_q),\Lambda)^\heartsuit$ and $P=\widetilde{R}_1$. Conditions (1) and (2) of Lemma \ref{lemma: decomp of abelian cat} follow from Lemma \ref{lemma: tilde R_1 inj proj} and Lemma \ref{lemma: R_1 ss}. We see that there is a direct sum decomposition 
    \begin{equation}\label{eq: principal decomp abelian cat}
        \mathrm{Rep}_c(G(\mathbb{F}_q),\Lambda)^\heartsuit= \mathrm{Rep}^0_c(G(\mathbb{F}_q),\Lambda)^\heartsuit\oplus \mathrm{Rep}^{>0}_c(G(\mathbb{F}_q),\Lambda)^\heartsuit
    \end{equation}        
    of abelian categories. By \eqref{eq: principal decomp abelian cat}, we have a direct sum decomposition
    \[
        Q=Q^0\oplus Q^{>0}
    \]
    with $Q^0\in \mathrm{Rep}^0_c(G(\mathbb{F}_q),\Lambda)^\heartsuit$ and $Q^{>0}\in \mathrm{Rep}^{>0}_c(G(\mathbb{F}_q),\Lambda)^\heartsuit$ such that $\mathrm{Hom}(Q^0,Q^{>0})=\mathrm{Hom}(Q^{>0},Q^0)=0$. Then

    \mathrm{Rep}(G(\mathbb{F}_q),\Lambda)=\langle Q^0\rangle \oplus \langle Q^{>0}\rangle
    \]
    where $\langle Q^0\rangle$ (resp. $\langle Q^{>0}\rangle$) is the presentable stable subcategory generated by $Q^0$ (resp. $Q^{>0}$). It is clear that $\langle Q^0\rangle=\langle \widetilde{R}_1\rangle=\mathrm{Rep}^0(G(\mathbb{F}_q),\Lambda)$. Then $\langle Q^{>0}\rangle$ is identified with $\mathrm{Rep}^{>0}(G(\mathbb{F}_q),\Lambda)$ consisting of objects $V$ with $\mathrm{Hom}_{G(\mathbb{F}_q)}(V,\widetilde{R}_1)=0$.

<!-- lemma-3: body_md -->
\label{lemma: decomp of abelian cat}
    Let $\mathcal{A}$ be a finite-length (i.e. Artinian and Noetherian) abelian category, and let $P\in\mathcal{A}$ satisfy the following conditions:
    \begin{enumerate}
        \item The object $P$ is both injective and projective.
        \item For any simple Jordan--H\"older factor $L$ of $P$, there exists a non-zero surjective map $P\twoheadrightarrow L$.
    \end{enumerate}
    Let $\mathcal{A}_P$ be the Serre subcategory of $\mathcal{A}$ generated by simple Jordan--H\"older factors of $P$. Let $\mathcal{A}^P$ be the subcategory of $\mathcal{A}$ consisting of objects $X\in\mathcal{A}$ such that $\mathrm{Hom}(X,P)=0$. Then there is a direct sum decomposition
    \[
        \mathcal{A}=\mathcal{A}_P\oplus \mathcal{A}^P
    \]
    of abelian categories.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
We can write
    \[
        P=\bigoplus_{i=1}^n P_i^{\oplus m_i}
    \]
    where $P_i$ are pairwise non-isomorphic indecomposable objects that are both injective and projective. Each $P_i$ has a unique simple quotient $L_i$ and a unique simple subobject $L_i'$. Moreover, $L_1,\dots,L_n$ are pairwise non-isomorphic and $L_1',\dots,L_n'$  are pairwise non-isomorphic. By (2), it follows that
    \[
        S=\{L_1,\dots,L_n\}=\{L_1',\dots,L_n'\}
    \]
    is the set of pairwise non-isomorphic irreducible Jordan--H\"older factors of $P$, and $\mathcal{A}_P$ is the Serre subcategory generated by $S$. In particular, for $i=1,\dots, n$, there exists an injective map $L_i\hookrightarrow P$. 

    By injectivity of $P$, every nonzero object $X\in \mathcal{A}_P$ admits a nonzero map $X\to P$. Therefore $\mathcal{A}^P\cap \mathcal{A}_P=0$. In particular, we have
    \[
        \mathrm{Hom}_{\mathcal{A}}(X,Y)=0=\mathrm{Hom}_{\mathcal{A}}(Y,X)
    \]
    for any $X\in\mathcal{A}_P,Y\in\mathcal{A}^P$.
    
    Let $M\in \mathcal{A}$. Consider
    \[
        Y=\bigcap_{f\colon X\to P}\ker(f).
    \]
    Since $X$ has finite length, this intersection equals a finite intersection. Let $Z=X/Y$. We have a short exact sequence
    \[
        0\to Y\to X\to Z\to 0.
    \]
    If there exists a non-zero map $Y\to P$, then by injectivity of $P$ we can extend it to a map $X\to P$, contradicting the definition of $Y$. Therefore $\mathrm{Hom}(Y,P)=0$ and hence $Y\in \mathcal{A}^P$. On the other hand, $Z$ embeds into a finite direct sum of copies of $P$. Therefore the simple Jordan--H\"older factors of $Z$ belong to $S$, and hence $Z\in \mathcal{A}_P$. Now it suffices to show that the extension splits. By projectivity of $P$, we can find a short exact sequence
    \[
        0\to W\to P^{\oplus m}\to Z\to 0
    \]
    with $W\in\mathcal{A}_P$. It induces an exact sequence
    \[
        \mathrm{Hom}_{\mathcal{A}}(W,Y) \to \mathrm{Ext}^1_{\mathcal{A}}(Z,Y)\to \mathrm{Ext}^1_{\mathcal{A}}(P^{\oplus m},Y).
    \]
    However, we know that $\mathrm{Hom}_{\mathcal{A}}(W,Y)=0$ since $W\in\mathcal{A}_P$, and $\mathrm{Ext}^1_{\mathcal{A}}(P^{\oplus m},Y)=0$ by projectivity of $P$. Therefore every extension of $Z$ by $Y$ splits.

\label{lem: ft representations are nice}
    Let $\mathcal{A}$ be a finite-length abelian category with finitely many isomorphism classes of simple objects, and let $\mathbf{C}$ be a stable category with a bounded $t$-structure.
    Let $\mathcal{F}: D^b(\mathcal{A}) \to \mathbf{C}$ be an exact functor. Suppose that $P\in\mathcal{A}$ is both a projective generator and an injective cogenerator. If $\mathcal{F}(P)\in\mathbf{C}^\heartsuit$, then the functor $\mathcal{F}$ is $t$-exact.
<!-- end lemma-5 -->
<!-- proof-11: body_md -->
By finiteness of simple objects, there are some integers $a\leq b$ such that $\mathcal{F}(L) \in \mathbf{C}^{[a, b]}$ for every simple object $L$. Induction on length gives the same bound for $\mathcal{F}(A)$ for every $A\in\mathcal{A}$. Choose a surjection $P^n \to A$ with kernel $K_1$. The exact sequence
    \[
        0 \to K_1 \to P^n \to A \to 0
    \]
    gives $\mathcal{H}^i(\mathcal{F}(A)) \simeq \mathcal{H}^{i+1}(\mathcal{F}(K_1))$ for $i > 0$, because $\mathcal{F}(P)\in\mathbf{C}^\heartsuit$. Iterating this process along a projective resolution of $K_i$, we find that $\mathcal{H}^i(\mathcal{F}(A)) \simeq \mathcal{H}^{i + k}(\mathcal{F}(K_k))$. Taking $k$ sufficiently large, we see that $\mathcal{H}^i(\mathcal{F}(A)) = 0$ for $i > 0$. Since $P$ is also an injective cogenerator, we can play the same game with injective resolutions and obtain that $\mathcal{H}^i(\mathcal{F}(A)) = 0$ for $i < 0$. Since the source $t$-structure is bounded, this finishes the proof.

<!-- thm-6: body_md -->
\label{thm:CM}
    The commuting scheme $\mathrm{Comm}_{\hat{G}}$ is Cohen--Macaulay.
<!-- end thm-6 -->
<!-- proof-25: body_md -->
The proof proceeds, roughly speaking, by a standard reduction to the completion of the unipotent cone. First let $(g, h) \in \mathrm{Comm}_{\hat{G}}(k)$ for an algebraically closed field $k$ over $\overline{\mathbb{F}}_\ell$, then by the Jordan decomposition we may write
    \[
        g = su, h = tv
    \]
    where $s, t$ are semisimple and $u, v$ are unipotent. We define $\hat{L} = Z_{\hat{G}}(t)$. By \cite[Corollary 9.4]{Steinberg1968}, $\hat{L}$ is a possibly disconnected reductive group. We claim that there is a natural map 
    \[
        \mathrm{Comm}^{\wedge}_{\hat{G}, (g, h)} \to \mathbb{A}^{\dim(\hat{G}) - \dim(\hat{L}), \wedge}_0 \times \mathrm{Comm}^{\wedge}_{\hat{L}, (t^{-1}g, v)},
    \]
    which will turn out to be an isomorphism. Indeed if $\hat{\mathfrak{l}} = \operatorname{Lie}(\hat{L})$ then we may choose a local smooth transversal $T$ to $\hat{L}$ at the identity, such that on tangent spaces $T_1(T) \oplus \hat{\mathfrak{l}} = \hat{\mathfrak{g}}$ and $T_1(T)$ is the sum of the non-trivial eigenspaces of $t$ on $\hat{\mathfrak{g}}$. Then the conjugation action of $T$ on $\hat{L}$
    \[
        \operatorname{Ad}_T\colon T \times \hat{L} \to \hat{G}
    \]
    induces an isomorphism $T^{\wedge}_1 \times \hat{L}_h^{\wedge} \to \hat{G}^{\wedge}_h$. Now taking $R$ an Artinian local $k$-algebra we can take $(a, b) \in \mathrm{Comm}_{\hat{G}}(R)$ lifting $(g, h)$. Using the above isomorphism we may uniquely write $b = xlx^{-1}$ for $l \in \hat{L}(R), x \in T(R)$, where $l$ reduces to $h$ and $x$ to $1$ modulo the maximal ideal of $R$. Then if we conjugate by $x$ we get a commuting pair $(x^{-1}ax, l)$, and we wish to show that $a_0 = x^{-1}ax \in \hat{L}(R)$. But again by inspection of the map on tangent spaces, the map $m|_{T \times \hat{L}}\colon T \times \hat{L} \to \hat{G}$ induced by multiplication gives a set of coordinates on $\hat{G}^{\wedge}_g$ around $g$, so we find that $a_0 = yl'$ for some $l' \in \hat{L}(R)$ reducing to $g$ modulo the maximal ideal, and $y \in T(R)$ reducing to the identity. Then $l'll'^{-1} = y^{-1}ly$, but since the left-hand side is in $\hat{L}(R)$, we find that $y$ must be trivial. The map $(a, b) \mapsto (x, x^{-1}ax, l)$ gives an isomorphism 
    \[
        \mathrm{Comm}^{\wedge}_{\hat{G}, (g, h)} \to \mathbb{A}^{\dim(\hat{G}) - \dim(\hat{L}), \wedge}_0 \times \mathrm{Comm}^{\wedge}_{\hat{L}, (g, h)},
    \]
    and since $t \in \hat{L}(k)$ is central we have a further isomorphism 
    \[
        \mathrm{Comm}^{\wedge}_{\hat{L}, (g, h)} \to \mathrm{Comm}^{\wedge}_{\hat{L}, (t^{-1}g, v)}.
    \]
    Let $\hat{M} = Z_{\hat{L}}(s)$, where $s$ is the semisimple part of $g$. Then $\hat{M}$ is a possibly disconnected reductive group again by \cite[Corollary 9.4]{Steinberg1968}. Applying the same reduction to the first coordinate gives 
    \[
        \mathrm{Comm}_{\hat{G}, (g, h)}^{\wedge} \simeq \mathbb{A}^{\dim(\hat{G}) - \dim(\hat{M}), \wedge}_0 \times \mathrm{Comm}_{\hat{M}, (u, v)}^{\wedge} \simeq \mathbb{A}^{\dim(\hat{G}) - \dim(\hat{M}), \wedge}_0 \times \mathrm{Comm}^{\wedge}_{\hat{M}^\circ, (u, v)}
    \]
    where the last isomorphism follows since now $u, v$ are both unipotent. We thus reduce to showing that if $\hat{G}$ is connected reductive and $(u, v)$ are unipotent elements of $\hat{G}$, then $\mathrm{Comm}_{\hat{G}, (u, v)}^{\wedge}$ is Cohen--Macaulay. Now we note that since $\ell\nmid |W|$ (and this condition passes along our reduction procedure to connected components of centralizers of semisimple elements) in particular $\ell \nmid |\pi_1(\hat{G}_{\mathrm{der}})|$, thus $\hat{T}{/\!\!/} W$ is smooth at the identity by \cite[Lemma 4.2]{Springer}. Finally take $\widetilde{\hat{G}} \to\hat{G}$ an \'etale cover with simply connected derived group, and denote by $Z$ the kernel of this isogeny. The map $\widetilde{\hat{T}}{/\!\!/} W \to \hat{T}{/\!\!/} W$ is smooth at the identity, and the base change
    \[
        \begin{tikzcd}
            &X \arrow{r}\arrow{d} & \widetilde{\hat{G}}^2 \arrow{d}\\
            &\mathrm{Comm}_{\hat{G}} \arrow{r} & \hat{G}^2
        \end{tikzcd}
    \]
    parameterizes pairs of elements $(a, b) \in \widetilde{\hat{G}}^2$ with commutator inside of $Z$. Since $Z$ is a finite \'etale group scheme, $\mathrm{Comm}_{\widetilde{\hat{G}}}$ is closed and open in $X$, and the fiber of $\mathrm{Comm}_{\widetilde{\hat{G}}}$ over the identity in $\widetilde{\hat{T}}{/\!\!/} W$ maps surjectively onto the analogous fiber for $\mathrm{Comm}_{\hat{G}}$ by Lemma \ref{lemma: Comm^u isog}, so we find that $\mathrm{Comm}_{\widetilde{\hat{G}}} \to \mathrm{Comm}_{\hat{G}}$ is finite \'etale surjective in a neighborhood of the unipotent fiber. Thus the map $\mathrm{Comm}_{\hat{G}} \to \hat{T}{/\!\!/} W$ is flat at the identity. Cohen--Macaulayness of $\mathrm{Comm}_{\hat{G}, (u, v)}^{\wedge}$ now follows from the fact that $\mathrm{Comm}^u_{\hat{G}}$ is Cohen--Macaulay by Corollary \ref{cor: Comm u CM} and \cite[Tag 045J]{stacks-project}.

<!-- cor-8: body_md -->
\label{cor: normal}
    The commuting scheme $\mathrm{Comm}_{\hat{G}}$ is normal. In particular, $\mathrm{Comm}_{\hat{G}}$ is reduced.
<!-- end cor-8 -->
<!-- proof-27: body_md -->
This is a consequence of Serre's criterion (\cite[Tag 031S]{stacks-project}), provided we can show that $\mathrm{Comm}_{\hat{G}}$ is regular in codimension $1$. First we prove this under the assumption that $\hat{G}$ has simply connected derived group, then we will reduce to this case. First we show that the locus of $(g, h) \in \mathrm{Comm}_{\hat{G}}$ such that either $g$ or $h$ is regular is smooth. By symmetry we prove this for the locus for which $g$ is regular. By \cite[Theorem 3.11]{Springer}, the locus of pairs $(g, h) \in \mathrm{Comm}_{\hat{G}}$ such that $g$ is regular is smooth after taking the quotient by the action of $Z(\hat{G})$ on the second factor. Since $Z(\hat{G})$ is smooth, we find that this locus is smooth. 

    By Corollary \ref{cor: flatness of pi}, if $\eta \in \mathrm{Comm}_{\hat{G}}$ is a point of height one, then writing $\pi(\eta) = \eta'$ we have, by flatness, that $\eta'$ is either of height one or zero. In the latter case we are done as then $\eta'$ is regular, so we focus on the height one case. In this case $\eta$ is a generic point of the fiber $\mathrm{Comm}_{\hat{G}, \eta'}$, and its image $\eta'$ is a generic point of the discriminant locus. Let $s \in \hat{G}(\overline{k(\eta')})$ be a semisimple representative of the point $\eta'$, then $Z_{\hat{G}}(s)$ has semisimple rank $1$, and is connected by \cite[Corollary 9.4]{Steinberg1968}. Let $(g, h) \in \mathrm{Comm}_{\hat{G}, \eta'}$, then $h$ is conjugate to $su$ with $s$ the generic semisimple element as above and $u \in Z_{\hat{G}}(s)$, if $u \neq 1$, then $u$ is regular since $Z_{\hat{G}}(s)$ has semisimple rank one, thus $h$ is regular. Otherwise $h = s$, $g$ can be any element of a maximal torus of $\hat{G}$, so $Z_{\hat{G}}(s) \cap \hat{G}^{\mathrm{reg}}$ is dense and open (as it is nonempty). In particular the generic point of $\mathrm{Comm}_{\hat{G}, \eta'}$ is in the regular locus. 

    Now we move to the general case. By \cite[Tag 0FIZ]{stacks-project} it suffices to check normality of $\mathrm{Comm}_{\hat{G}}$ after taking completion at an arbitrary point $(g, h) \in \mathrm{Comm}_{\hat{G}}$. The same formally smooth reduction as applied in the proof of Theorem \ref{thm:CM} shows that we may reduce to studying the completion at a unipotent tuple $(u, v) \in \mathrm{Comm}_H$ for $H$ some connected reductive subgroup of $\hat{G}$. Now as usual we may choose $\alpha: \widetilde{H} \to H$ an \'etale central isogeny such that $\widetilde{H}_{\mathrm{der}}$ is simply connected, and the map 
    \[
        \mathrm{Comm}_{\widetilde{H}, (u, v)}^{\wedge} \to \mathrm{Comm}_{H, (u, v)}^{\wedge}
    \]
    is an isomorphism as argued in the last paragraph of the proof of Theorem \ref{thm:CM}. The scheme $\mathrm{Comm}_{\widetilde{H}, (u, v)}^{\wedge}$ is normal by \cite[Tag 0C23]{stacks-project}, and the same is true for $\mathrm{Comm}_{H, (u, v)}^{\wedge}$, completing the proof.

Let $Z$ be the locus on which $\operatorname{Spec} A$ is not Cohen--Macaulay, by \cite[Tag 045U]{stacks-project} this is a closed subscheme of $\operatorname{Spec} A$, which is $\mathbb{G}_m$-invariant. Let $J$ be the homogeneous ideal defining this locus $Z$. If $J$ were a \emph{proper} homogeneous ideal it would be contained within $\mathfrak{m}$, but this is impossible by the above. The argument for flatness is essentially the same.
