<!-- lemma-1: body_md -->
\label{l}
Let $(a_0, a_1, a_2, \cdots, a_l)$ be a diameter walk of  a finite tree $Q$, where $l\in\mathbb{N}_{\geq 2}$. Then the  set $\mathcal{N}'(a_1)=\mathcal{N}(a_1)\setminus \{a_2\}$ consists of  all leaves.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Suppose that there exists a vertex $a'\in \mathcal{N}'(a_1) $ being not a leaf. It  means that there  exists at least one vertex $a_1 \neq a''\in \mathcal{N}(a')$ such that the length of walk $(a'', a', a_1, a_2, \cdots, a_m)$ is greater than that of diameter walk $(a_0,a_1, a_2,\cdots,a_l)$ in the tree $Q$, this is a contradiction.
<!-- end proof-1 -->

<!-- thm-1: body_md -->
\cite[Theorem 7.1]{Daniel}\label{push-down}
The following statements hold.
\begin{enumerate}


\item $\pi_\lambda$ sends indecomposable representations in $\operatorname{rep}_k(T(n), \Omega)$ to indecomposable representations  in $\operatorname{rep}_k(K(n))$.

\item If $M\in\operatorname{rep}_k(T(n), \Omega)$ is indecomposable, then $\pi_\lambda(M)\cong \pi_\lambda(N)$ if and only if $M^g\cong N$ for some $g\in G$.

\item $\pi_\lambda$ sends $AR$-sequences to $AR$-sequences and $\pi_\lambda$ commutes with the Auslander-Reiten translations.

\item If $M\in\operatorname{rep}_k(T(n), \Omega)$ is indecomposable in a component $\mathcal{D}$ with $\pi_\lambda(M)$ in a component $\mathcal{C}$, then $\pi_\lambda$ induces a covering $\mathcal{D}\rightarrow\mathcal{C}$ of translation quivers.

\end{enumerate}

<!-- proposition-1: body_md -->
\label{position}
Let $M\in \operatorname{mod}(T(n),\Omega)$ be an indecomposable module, and let $(a_0,a_1,\cdots,a_{d(M)})$ be a diameter walk of $T(M)$ with $d(M)\geq 2$. Suppose that $T(M)=B(M)$. Then the matrix $\Gamma_M$ would look like the following
\begin{center}


$\Gamma_M=\sum^n_{i=1}\bar{\gamma}_i = \begin{bmatrix}
\bar{\gamma}^1_1 & 0\\
\bar{\gamma}^1_2 & 0\\
\vdots & 0\\
\bar{\gamma}^1_{n-1} & 0 \\
\bar{\gamma}^1_n & \Gamma'_M \\

\end{bmatrix}$ or  $\Gamma_M=\sum^n_{i=1}\bar{\gamma}_i = \begin{bmatrix}
\bar{\gamma}^1_1 & \bar{\gamma}^1_2 & \cdots & \bar{\gamma}^1_{n-1} & \bar{\gamma}^1_n \\
0 & 0 & \cdots & 0  & \Gamma'_M 
\end{bmatrix}$,
\end{center}
where  $\Gamma'_M=\sum^{n}_i \bar{\gamma}'_i$, $\bar{\gamma}'_i=\bigoplus_{\bar{\gamma}^j_i\in \mathcal{A}'_i}\bar{\gamma}^j_i$, $\mathcal{A}'_i=\mathcal{A}_i\setminus \{\bar{\gamma}^1_i\}$,   $\bar{\gamma}^1_i$ is the linear map between  $M_{a_1}$ and its neighbours $M_b$, $b\in \mathcal{N}_M(a_1)$, $1\leq i\leq n$.

\label{ddpoint}
Suppose that $M\in \operatorname{mod}(T(n),\Omega)$ is an  indecomposable and of constant rank module. Then  for the sets $\mathcal{S}_i, \mathcal{T}_i$ and every arrow $\gamma^q_j\in T(M)_1$ there exist
\begin{center}

\end{center}

<!-- proof-2: body_md -->
Let  $N=\pi_\lambda{(M)}$.   Suppose that there exists a non-trivial map  $M(\gamma^u_j): M_{s(\gamma^u_j)} \rightarrow M_{t(\gamma^u_j)}$  in $\mathcal{A}_j$ for some  $i$ ($\neq j$) such that $s(\gamma^u_j)\nin \mathcal{S}_i$ and $t(\gamma^u_j)\nin \mathcal{T}_i$, that is, there is no   direct summand  $\bar{\gamma}^u_j$ of $\bar{\gamma}_j$ such that it is located in the same row or column with some direct summand $\bar{\gamma}^v_i$ of $\bar{\gamma}_i$ in the block matrix $\bar{\gamma}_i+\bar{\gamma}_j$.  According to Proposition \ref{position}, we have
\begin{equation}\label{rk}
\operatorname{rk}(\bar{\gamma}_i+\bar{\gamma}_j)=\operatorname{rk}(\bar{\gamma}_i+(\bigoplus_{\pi(\gamma^p_j)=\gamma_j} M(\gamma^q_j))\geq \operatorname{rk} \begin{bmatrix}
\bar{\gamma}_i & 0\\
0& \bar{\gamma}^u_j
\end{bmatrix}>\operatorname{rk} \bar{\gamma}_i.
\end{equation}
Clearly, the inequality $(\ref{rk})$ cannot happen because module $N$ is of constant rank and  $\operatorname{rk}(\bar{\gamma}_i+\bar{\gamma}_j) = \operatorname{rk} \bar{\gamma}_i=c$ for some $c\in \mathbb{N}$.

<!-- lemma-5: body_md -->
\label{tt}
Let $\bar{\gamma}^p_i\in \mathcal{A}_i$, and let $\bar{\gamma}^{p}_j,\bar{\gamma}^{p'}_j\in \mathcal{A}_j$ $i\neq j$. Suppose that there exist   $s(\bar{\gamma}^p_i)=s(\bar{\gamma}^p_j)$ and $t(\bar{\gamma}^p_i)=t(\bar{\gamma}^{p'}_j)$. Then we have 
\begin{center}
$\bar{\gamma}^p_i(\ker \bar{\gamma}^{p}_j)\subseteq \operatorname{im} \bar{\gamma}^{p'}_j$.
\end{center}
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
Suppose that $\operatorname{rk} x^N_\alpha =m$.  Recall that
\begin{center}
$\bar{\gamma}_i=N(\gamma_i)=\bigoplus _{\pi(\gamma^x_i)=\gamma_i} M(\gamma^x_i)=\bigoplus _{\pi(\gamma^x_i)=\gamma_i}\bar{\gamma}^x_i$, and $\bar{\gamma}_j=N(\gamma_j)=\bigoplus _{\pi(\gamma^y_j)=\gamma_j} M(\gamma^y_j)=\bigoplus _{\pi(\gamma^y_i)=\bar{\gamma}_j}\bar{\gamma}^y_j $. 
\end{center}
Then $\ker \bar{\gamma}_j=\ker (\bigoplus _{\pi(\gamma^y_j)=\gamma_j}M(\gamma^y_j))=\bigoplus _{\pi(\gamma^y_j)=\gamma_j}\ker M(\gamma^y_j)$. Hence we have 
\begin{equation}
\begin{split}
\bar{\gamma}_i(\ker \bar{\gamma}_j)& =( \bigoplus _{\gamma^x_i\in \mathcal{A}_i} M(\gamma^x_i))(\bigoplus _{\gamma^y_j\in \mathcal{A}_j}\ker M(\gamma^y_j)) \\& = \bigoplus_{s(\bar{\gamma}^p_i)=s(\bar{\gamma}^{p}_j), \bar{\gamma}^p_i\in \mathcal{A}_i,\bar{\gamma}^p_j\in\mathcal{A}_j}\bar{\gamma}^p_i(\ker (\bar{\gamma}^{p}_j)) \\ &  \subseteq \bigoplus _{t(\gamma^p_i)=t(\gamma^p_j), \bar{\gamma}^p_i\in \mathcal{A}_i,\bar{\gamma}^p_j\in\mathcal{A}_j} \operatorname{im} M(\gamma^p_j) \\ & \subseteq\operatorname{im} (\bigoplus _{\pi(\gamma^p_j)=\gamma_j, \bar{\gamma}^p_j\in \mathcal{A}_j} M(\gamma^p_j)) \\& = \operatorname{im} \bar{\gamma}_j.
\end{split}
\end{equation}
Otherwise, we assume that there exist some $\bar{\gamma}^p_i \in \mathcal{A}_i$, and $\bar{\gamma}^{p}_j,\bar{\gamma}^{p'}_j\in \mathcal{A}_j$ such that 
\begin{center}
$\bar{\gamma}^p_i(\ker \bar{\gamma}^{p}_j)\nsubseteq \operatorname{im} \bar{\gamma}^{p'}_j$.
\end{center}
Note that we only have one direct summand $\bar{\gamma}^{p'}_j$ of $\bar{\gamma}_j$ satisfying $t(\bar{\gamma}^p_i)=t(\bar{\gamma}^{p'}_j)$,  that is to say, $\bar{\gamma}_i(\ker \bar{\gamma}_j)\nsubseteq\operatorname{im} \bar{\gamma}_j$. We can see that module $N$ is not of constant   rank by Lemma \ref{KI}, this is a contradiction. Finally, we have 
\begin{center}
 $N(\gamma_i)(\ker N(\gamma_j))\subseteq \operatorname{im} N(\gamma_j) \Leftrightarrow N(\gamma^p_i)(\ker \gamma^{p}_j)\subseteq \operatorname{im} N(\gamma^{p'}_j)$, 
\end{center}
for any direct summands $\bar{\gamma}^p_i\in \mathcal{A}_i, \bar{\gamma}^p_j,\bar{\gamma}^{p'}_j\in \mathcal{A}_j$ with $s(\bar{\gamma}^p_i)=s(\bar{\gamma}^{p}_j)$ and $t(\bar{\gamma}^p_i)=t(\bar{\gamma}^{p'}_j)$.

<!-- proposition-2: body_md -->
\label{lj}
 Let $(a_0, a_1, \cdots, a_m)$ be a  walk of  $T(M), m\geq 2$. Suppose that any $b_i\in \mathcal{N}_M(a_1)\setminus \{a_2 \}$ is a leaf for $1 \leq i \leq n-1$. Then

\begin{enumerate}
\item  When  $a_0$ is a sink, the kernels of the maps $\bar{\gamma}^1_i: M_{a_1}\rightarrow M_{b_i}$  satisfy    
\begin{center}
 $\ker \bar{\gamma}^1_1\cong \cdots \cong  \ker \bar{\gamma}^1_{n-1}$, 
\end{center}
and they are all surjective. In particular,  the map $\bar{\gamma}^1_n: M_{a_1}\rightarrow M_{a_2}$ is injective;

\item   When $a_0$ is a source,  the images of the  maps $\bar{\gamma}^1_i: M_{b_i}\rightarrow M_{a_1}$  satisfy 
\begin{center}
$\operatorname{im} \bar{\gamma}^1_1\cong \cdots \cong\operatorname{im} \bar{\gamma}^{1}_{n-1}$,
\end{center}
 and they are all injective. In particular, the map $\bar{\gamma}^1_n: M_{a_2} \rightarrow M_{a_1}$ is surjective and $\operatorname{im} \bar{\gamma}^1_i\subseteq \operatorname{im} \bar{\gamma}^1_n$ under the isomorphisms, $1\leq i\leq n-1$.
\end{enumerate}
Moreover, there always exists $\operatorname{rk} \bar{\gamma}^1_n\geq  \operatorname{rk} \bar{\gamma}^1_i$,  where $\bar{\gamma}^1_n$ is the map between $M_{a_1}$ and $M_{a_2}$, $1\leq i\leq n-1$.

Otherwise, suppose that there exists some $j\in\{1,\cdots,n-1\}$ such that $\operatorname{rk}\bar{\gamma}^1_n<\operatorname{rk}\bar{\gamma}^1_j$. Since $s(\bar{\gamma}^1_n)=s(\bar{\gamma}^1_j)$, we have 
\begin{center}
$\operatorname{rk}(\bar{\gamma}_n+\bar{\gamma}_j)=\operatorname{rk} \begin{bmatrix}
\bar{\gamma}^1_j & 0\\
\bar{\gamma}^1_n & \Gamma_{i,j}
\end{bmatrix}>\operatorname{rk} \begin{bmatrix}
0 & 0\\
\bar{\gamma}^1_n & \Gamma_{i,j}
\end{bmatrix}\geq \operatorname{rk}\bar{\gamma}_n$,

where $\Gamma_{i,j}=\bigoplus_{\bar{\gamma}^i_n\in \mathcal{A}'_n}\bar{\gamma}^i_n +\bigoplus_{\bar{\gamma}^i_j\in \mathcal{A}'_j}\bar{\gamma}^i_j$, $\mathcal{A}'_i=\mathcal{A}_i\setminus \{\bar{\gamma}^1_i\}$, $i\in \{j,n\}$, and this is  a contradiction.  \begin{center}

 \bar{\gamma}^1_1 \\
 \vdots\\
 \bar{\gamma}^1_{n-1}
 \end{bmatrix}$.
 \end{center}

 \end{bmatrix}.$
