<!-- hyp-1: body_md -->
\label{hyp:dispersion}
The function $\omega\:{\mathbb R}^d\to[0,\infty)$ is measurable and even.
There are constants $C_\omega>0$ and
$0<\nu\leq\min\{2,d\}$ such that
\[
    \omega(p)\leq C_\omega\langle p\rangle^\nu,
    \qquad p\in{\mathbb R}^d.
\]
Moreover, for every real-valued $\chi\in C_c^\infty({\mathbb R}^d)$,
the commutator initially defined on $C_c^\infty({\mathbb R}^d)$ extends to
the form domain of $\omega(\mathsf{p})$ and satisfies
\begin{align}
    \lVert[\omega(\mathsf{p}),\chi]u\rVert^2
    \leq
    C_\chi\bigl(
        \lVert\omega(\mathsf{p})^{1/2} u\rVert^2+\lVertu\rVert^2
    \bigr),
    \qquad u\in\mathcal{D}(\omega(\mathsf{p})^{1/2}).
    \label{eq:dispersion est}
\end{align}
<!-- end hyp-1 -->
<!-- ex-1: body_md -->
\label{rem:dispersion-examples}
\Cref{hyp:dispersion} holds for
$
\omega(p)=\langle p\rangle^\nu
$
whenever $0<\nu\leq\min\{2,d\}$, by standard
pseudodifferential commutator estimates, see e.g. the symbolic composition and commutator calculus for $S^m_{1,0}$ symbols in \cite{hormander}. Thus, the
relativistic dispersion $\omega(p)=\sqrt{\lvertp\rvert^2+1}$ is covered in
every dimension $d\geq1$.  It also holds for the standard non-relativistic dispersion relation
$\omega(p)=p^2$ when $d\geq2$.
<!-- end ex-1 -->
<!-- hyp-2: body_md -->
\label{hyp:pair-potential}
The potential $W$ is measurable, real-valued, and even, and it admits
a decomposition
\[
    W=W_2+W_\infty,
    \qquad
    W_2\in L^2({\mathbb R}^d),
    \qquad
    W_\infty\in L^\infty({\mathbb R}^d).
\]
Moreover, we assume $W$ to be bounded from below, i.e., there is a constant $C_W\geq0$ such that
\[
    W(x)\geq-C_W
    \qquad\text{for almost every }x\in{\mathbb R}^d.
\]
<!-- end hyp-2 -->
<!-- ex-2: body_md -->
\label{cor:riesz-coulomb-main}
	Apart from bounded potentials, the hypothesis also allows for potentials with singularities.
	Let $d\geq1$ and
	$\kappa>0$, and consider
	\[
	W_\sigma(x) =\frac{\kappa}{\lvertx\rvert^\sigma},
	\qquad 0<\sigma<\frac d2.
	\]
	Then $W_\sigma$ satisfies
	\cref{hyp:pair-potential}, since
	\[
	W_\sigma
	=
	W_\sigma\mathbf 1_{\{\lvertx\rvert<1\}}
	+
	W_\sigma\mathbf 1_{\{\lvertx\rvert\geq1\}}
	\]
	where the first term is in $L^2({\mathbb R}^d)$ precisely when $2\sigma<d$, while
	the second term is bounded.  
	In particular, the repulsive Coulomb potential
	$
	W_{1}(x)=\frac{\kappa}{\lvertx\rvert}
	$
	in any $d\geq 3$ is covered.
<!-- end ex-2 -->
<!-- hyp-3: body_md -->
\label{hyp:external-form}
One has $U\in L^1_{\mathrm{loc}}({\mathbb R}^d)$, and there are constants
$0\leq a<1$ and $b\geq0$ such that
\[
    \int_{{\mathbb R}^d}U_-(x)\lvertu(x)\rvert^2\,{\mathsf d} x
    \leq a \left\lVert\omega(\mathsf{p})^{1/2}u\right\rVert^2 +b\lVertu\rVert^2,
    \qquad u\in C_c^\infty({\mathbb R}^d).
\]
<!-- end hyp-3 -->
<!-- hyp-4: body_md -->
\label{hyp:local-propagation}
	There are an open set $\Omega_U\subseteq{\mathbb R}^d$ and a number
	$m>d/2$ such that, for every $u\in C_c^\infty(\Omega_U)$,
	\[
	\lVerte^{{\mathsf i} th}u-u\rVert_{H^m}\leq C_u\lvertt\rvert,
	\qquad \lvertt\rvert\leq1.
	\]
<!-- end hyp-4 -->
<!-- ex-3: body_md -->
\label{cor:external-examples}
	\Cref{hyp:external-form,hyp:local-propagation} hold in each of
	the following cases.
	\begin{enumerate}[label=\upshape(\arabic*)]
		\item \label{it:external simple} \Cref{hyp:dispersion} holds and 
		$U\in W^{k,\infty}({\mathbb R}^d; {\mathbb R})$, where
		$
		k=\left\lfloor\frac d2\right\rfloor+1,
		$ with $\Omega_U = {\mathbb R}^d$. In particular, they hold in the translation-invariant case $U \equiv 0$. 
		\item \label{it:external coulomb}  $d=3$, $\omega(p)=p^2$, and
		\[
		U(x)=U_0(x)+\sum_{\ell=1}^M
		\frac{\lambda_\ell}{\lvertx-R_\ell\rvert},
		\qquad
		U_0\in W^{2,\infty}({\mathbb R}^3; {\mathbb R}),
		\quad \lambda_\ell\in{\mathbb R},
		\]
			with
		$\Omega_U={\mathbb R}^3\setminus\{R_1,\ldots,R_M\}$.
	\end{enumerate}
<!-- end ex-3 -->
<!-- thm-1: body_md -->
\label{thm:main-discontinuity}
Suppose that $\omega$ satisfies \cref{hyp:dispersion}. Let $W$ satisfy
\cref{hyp:pair-potential} and assume that it has a representative with two distinct nonzero
continuity points $z_1,z_2\in{\mathbb R}^d\setminus\{0\}$ such that $W(z_1)\neq W(z_2)$.
 Let $U$ satisfy
\cref{hyp:external-form,hyp:local-propagation}, and assume that there is
a point $y_0\in{\mathbb R}^d$ such that
$y_0+z_1\in\Omega_U$ and $y_0+z_2\in\Omega_U$.
Then there are normalized orthogonal functions
$f_1,f_2\in C_c^\infty(\Omega_U)$ and a number $\theta>0$  such that
\begin{align}
    \liminf_{n\to\infty}
    \lVert\tau_{t_n}(A)-A\rVert>0, \qquad \text{for }~ t_n= \frac{\theta}{n}, ~A = a^*(f_1)a(f_2).
    \label{eq:main-transfer-discontinuity}
\end{align}
Moreover, there exist
$h\in C_c^\infty(\Omega_U)$ and a subsequence
$(t_{n_k})$ such that $\liminf_{k\to\infty}
\lVert\tau_{t_{n_k}}(A)-A\rVert>0$ for $A=n_h$ and $A =a^*(h)$.
<!-- end thm-1 -->
<!-- cor-1: body_md -->
\label{thm:main-noninvariance}
	Under the assumptions of \cref{thm:main-discontinuity}, neither
	$\mathfrak{A}^0_{\mathrm{CAR}}$ nor $\mathfrak{A}_{\mathrm{CAR}}$ is invariant under
	$\tau_t$. More precisely, for any $A$ chosen as in

Choose an open cube $Q\Subset B_0$ of side length $0 < L\leq1$, set
	$m_n=\left\lceil n^{1/d}\right\rceil$, 
	and divide $Q$ into $m_n^d$ mutually disjoint subcubes of side
	length $\ell_n:=L/m_n$.  In $n$ distinct subcubes place
	appropriately rescaled copies of a fixed normalized function
	$\varphi\in C_c^\infty((0,1)^d)$.  The resulting functions
	$\varphi_{1,n},\ldots,\varphi_{n,n}$ have disjoint supports and are
	therefore orthonormal. By scaling and
	\Cref{hyp:dispersion}, we get for all $j$,
	\[
	\langle\varphi_{j,n},\omega(\mathsf{p})\varphi_{j,n}\rangle
	=\int_{{\mathbb R}^d}\omega(p/\ell_n)\lvert\widehat\varphi(p)\rvert^2\,{\mathsf d} p
	\leq
	C\ell_n^{-\nu}
	\int_{{\mathbb R}^d}\langle p\rangle^\nu
	\lvert\widehat\varphi(p)\rvert^2\,{\mathsf d} p
	\leq Cm_n^\nu.
	\]
	Hence, we get the required bound for the kinetic energy,
	\[
	\sum_{j=1}^n
	\langle\varphi_{j,n},\omega(\mathsf{p})\varphi_{j,n}\rangle
	\leq C n m_n^\nu
	\leq Cn^{1+\nu/d}.
	\]
	
	Next, in order to estimate the interaction energy, we introduce the one-particle density matrix
	$
	\gamma_n(x,y)
	:=
	\sum_{j=1}^n
	\varphi_{j,n}(x)\overline{\varphi_{j,n}(y)},
	$
	and the corresponding one-particle density
	$\rho_n(x)= \gamma_n(x,x) =\sum_{j=1}^n|\varphi_{j,n}(x)|^2$, 
	 supported in $Q$.  Since the supports of the orbitals are
	disjoint, the same scaling yields
	\[
	\|\rho_n\|_\infty\leq C m_n^d\leq Cn,
	\qquad
	\int_{{\mathbb R}^d}\rho_n(x)\,{\mathsf d} x=n.
	\]

	\begin{align*}

		\int_{{\mathbb R}^{dn}}W(x_i-x_j)\lvert\Psi_n(X)\rvert^2\,{\mathsf d} X\right\rvert
		&=  \left\lvert	\iint_{Q\times Q} W(x-y)(        \rho_n(x)\rho_n(y)-|\gamma_n(x,y)|^2

		\iint_{Q\times Q}
		|W(x-y)|\rho_n(x)\rho_n(y)\,{\mathsf d} x\,{\mathsf d} y \\
		&\leq
		\|\rho_n\|_\infty
		 \underbrace{ 		\left( \sup_{x\in Q}\int_Q|W(x-y)|\,{\mathsf d} y

		\right\rangle
		\right|
		&=
		\left|\int_Q U(x)\rho_n(x)\,{\mathsf d} x\right| \leq

		\leq Cn.
	\end{align*}

	$n^{1+\nu/d}\leq n^2$ because $\nu\leq d$,

\end{align}
