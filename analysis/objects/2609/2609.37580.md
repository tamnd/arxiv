<!-- thm-1: body_md -->
\label{thm:Ludwig}
	Let \(n>2\). A map \(Z:W^{1,2}(\mathbb{R}^n)\to\mathbb{M}^n\) is a continuous, affinely
	contravariant valuation if and only if there exists \(c\in\mathbb{R}\) such that
	\[
	Z(f)=cJ(f^2)
	\]
	for every \(f\in W^{1,2}(\mathbb{R}^n)\).

<!-- lem-1: body_md -->
\label{thm:cal}
	Let \(s\in\mathbb{R}\) and \(P\in\mathcal P^n_o\). Then for a continuous and bounded function \(\alpha:\mathbb{R}\to\mathbb{R}\), we have
	\[J_\alpha^p((s\ell_P)^p)=(-p)^ps^{p-n}M^{0,p}(P)\int_0^s\alpha(t)(s-t)^{n-1}{\rm d} t.\]
<!-- end lem-1 -->
<!-- proof-1: body_md -->
First, for every \(x\in P\), since the gradient of the support function is 0-homogeneous, we obtain
	\begin{align*}
		\left\vert{\nabla h(P^\ast,x)}\right\vert &= \left\vert{\nabla h(P^\ast,\rho(P,x)x)}\right\vert
		= \left\vert{\frac{N_P(\rho(P,x)x)}{h(P,N_P(\rho(P,x)x))}}\right\vert\\
		&= h(P,N_P(\rho(P,x)x))^{-1}
		= h(P,\widehat N_P(x))^{-1}.
	\end{align*}
	Combined with the construction of the cone function, we obtain
	\begin{align*}
		J_\alpha^p((s\ell_P)^p) &= (ps)^p\int_{\mathbb{R}^n}{\nabla\ell_P(x)}^{\odot p}\alpha(s\ell_P(x)){\rm d} x\\
		&= (-ps)^p\int_P{\widehat N_P(x)}^{\odot p}h(P,\widehat N_P(x))^{-p}\alpha(s(1-h(P^\ast,x))){\rm d} x\\
		&= (-ps)^p\int_P{\widehat N_P(x)}^{\odot p}h(P,\widehat N_P(x))^{1-p}\left\vert{\nabla h(P^\ast,x)}\right\vert\alpha(s(1-h(P^\ast,x))){\rm d} x.
	\end{align*}
	We use the coarea formula to proceed
	\begin{equation}\label{eqn:cf}
		(-ps)^p\int_0^1\int_{t\partial P}{\widehat N_P(x)}^{\odot p}h(P,\widehat N_P(x))^{1-p}\alpha(s(1-t)){\rm d}\mathcal H^{n-1}(x){\rm d} t.
	\end{equation}
	Next, for every \(t\in(0,1)\) and \(y\in\partial P\), we notice
	\begin{align*}
		N_P(y)=N_P(\rho(P,y)y)=N_P(\rho(P,ty)ty)=\widehat N_P(ty),
	\end{align*}
	and obtain that \eqref{eqn:cf} equals
	\begin{align*}
		& (-ps)^p\int_0^1\int_{\partial P}{N_P(y)}^{\odot p}h(P,N_P(y))^{1-p}\alpha(s(1-t))t^{n-1}{\rm d}\mathcal H^{n-1}(y){\rm d} t\\
		=& (-ps)^p\left(\int_0^1t^{n-1}\alpha(s-st){\rm d} t\right)\left(\int_{S^{n-1}}{u}^{\odot p}h(P,u)^{1-p}{\rm d} S(P,u)\right)\\
		=& (-p)^ps^{p-n}M^{0,p}(P)\int_0^s\alpha(t)(s-t)^{n-1}{\rm d} t.
	\end{align*}

	If \(Z_1(sf)=Z_2(sf)\) for every \(s\in\mathbb{R}\) and \(f\inP^{1,p}(\mathbb{R}^n)\), then

	for every \(f\inL^{1,p}(\mathbb{R}^n)\).

<!-- lem-3: body_md -->
\label{thm:if}
	Let \(\alpha:\mathbb{R}\to\mathbb{R}\) be a continuous and bounded function. Then, for every \(f\inW^{1,p}(\mathbb{R}^n)\), the map \(Z:W^{1,p}(\mathbb{R}^n)\to{\rm Sym}^p(\mathbb{R}^n)\), defined by
	\[Z(f)=J_\alpha^p(f^p),\]
	is a continuous, \({\rm SL}(n)\) contravariant and translation invariant valuation with \(Z\in\mathcal B_p\) and \(Z(0)=0\).
<!-- end lem-3 -->
<!-- proof-2: body_md -->
First, since \(\alpha\) is bounded, there exists \(M>0\) such that
	\[\left\Vert{Z(f)}\right\Vert\leq p^p\int_{\mathbb{R}^n}\left\Vert{{\nabla f}^{\odot p}\alpha(f)}\right\Vert\leq p^p\lambda_{n,p}M\int_{\mathbb{R}^n}\left\vert{\nabla f}\right\vert^p<\infty\]
	for every \(f\inW^{1,p}(\mathbb{R}^n)\). Hence, \(Z\) is well defined.
	
	Next, let \(f\inW^{1,p}(\mathbb{R}^n)\) and \(\left\{{f_i}\right\}_{i=1}^\infty\) be a sequence in \(W^{1,p}(\mathbb{R}^n)\) that converges to \(f\). We have
	\begin{align*}
		\left\Vert{Z(f_i)-Z(f)}\right\Vert &\leq p^p\left\Vert{\int_{\mathbb{R}^n}\left({\nabla f_i}^{\odot p}-{\nabla f}^{\odot p}\right)\alpha(f_i)}\right\Vert+p^p\left\Vert{\int_{\mathbb{R}^n}{\nabla f}^{\odot p}\left(\alpha(f_i)-\alpha(f)\right)}\right\Vert\\
		&:= p^pI_1(f_i)+p^pI_2(f_i).
	\end{align*}
	Due to the boundedness of \(\alpha\), we have
	\begin{align*}
		I_1(f_i) &\leq \int_{\mathbb{R}^n}\left\Vert{{\nabla f_i}^{\odot p}-{\nabla f}^{\odot p}}\right\Vert\left\vert{\alpha(f_i)}\right\vert\\
		&\leq M\int_{\mathbb{R}^n}\left\Vert{{\nabla f_i}^{\odot p}-{\nabla f}^{\odot p}}\right\Vert\\
		&\leq M\int_{\mathbb{R}^n}\sum_{k=0}^{p-1}\left\Vert{{\nabla f_i}^{\odot{k}}\odot\nabla(f_i-f)\odot{\nabla f}^{\odot{(p-1-k)}}}\right\Vert\\
		&\leq \lambda_{n,p}M\int_{\mathbb{R}^n}\sum_{k=0}^{p-1}\left(\left\vert{\nabla f_i}\right\vert^k\cdot\left\vert{\nabla(f_i-f)}\right\vert\cdot\left\vert{\nabla f}\right\vert^{p-1-k}\right).
	\end{align*}
	Now, H\"{o}lder's inequality yields
	\[I_1(f_i) \leq \lambda_{n,p}M\left\Vert{\nabla(f_i-f)}\right\Vert_{p}\sum_{k=0}^{p-1}\left\Vert{\nabla f_i}\right\Vert_{p}^k\left\Vert{\nabla f}\right\Vert_{p}^{p-1-k}\to0,\qquad\text{as }i\to\infty.\]
	On the other hand, for every subsequence \(\{{f_{i_j}}\}\subset\left\{{f_i}\right\}\), we are going to show that there is a subsequence \(\{{f_{i_{j_k}}}\}\subset\{{f_{i_j}}\}\)
	such that \(I_2(f_{i_{j_k}})\) converges to 0, to conclude that \(Z(f_i)\to Z(f)\).
	Let \(\{{f_{i_j}}\}\) be a subsequence of \(\left\{{f_i}\right\}\), then \(\{{f_{i_j}}\}\) converges to \(f\) in \(W^{1,p}(\mathbb{R}^n)\).
	Thus, there exists a subsequence \(\{{f_{i_{j_k}}}\}\subset\{{f_{i_j}}\}\) with \(f_{i_{j_k}}\to f\) a.e. as \(k\to\infty\).
	Since \(\alpha\) is continuous, we have
	\[\alpha(f_{i_{j_k}})\to\alpha(f)\text{ a.e.},~\text{i.e. }\left\vert{\alpha(f_{i_{j_k}})-\alpha(f)}\right\vert\to0\text{~a.e.,}\qquad\text{as}~k\to\infty.\]
	Moreover, since \(\alpha\) is bounded, we have
	\[\left\vert{\nabla f}\right\vert\cdot\left\vert{\alpha(f_{i_{j_k}})-\alpha(f)}\right\vert\leq 2M\left\vert{\nabla f}\right\vert\in L^p(\mathbb{R}^n).\]
	Hence, \(\left\vert{\nabla f}\right\vert^p\cdot\left\vert{\alpha(f_{i_{j_k}})-\alpha(f)}\right\vert^p\) is integrable.
	By Lebesgue dominated convergence theorem, we obtain
	\begin{align*}
		\lim_{k\to\infty}\int_{\mathbb{R}^n}\left\vert{\nabla f}\right\vert^p\cdot\left\vert{\alpha(f_{i_{j_k}})-\alpha(f)}\right\vert^p
		= \int_{\mathbb{R}^n}\lim_{k\to\infty}\left\vert{\nabla f}\right\vert^p\cdot\left\vert{\alpha(f_{i_{j_k}})-\alpha(f)}\right\vert^p=0.
	\end{align*}
	Therefore,
	\[I_2(f_{i_{j_k}})\leq\lambda_{n,p}\int_{\mathbb{R}^n}\left\vert{\nabla f}\right\vert^p\cdot\left\vert{\alpha(f_{i_{j_k}})-\alpha(f)}\right\vert^p\to0,\qquad\text{as }k\to\infty,\]
	which implies the continuity.
	
	Let \(\phi\in{\rm SL}(n)\). Then,
	\begin{align*}
		Z(f\circ\phi^{-1}) &= p^p\int_{\mathbb{R}^n}{\nabla(f\circ\phi^{-1})}^{\odot p}\alpha(f\circ\phi^{-1})\\
		&= p^p\int_{\mathbb{R}^n}{\left(\phi^{-t}\left(\nabla f\circ\phi^{-1}\right)\right)}^{\odot p}\alpha(f\circ\phi^{-1})\\
		&= p^p\int_{\mathbb{R}^n}\phi^{-t}\cdot{\left(\nabla f\circ\phi^{-1}\right)}^{\odot p}\alpha(f\circ\phi^{-1})\\
		&= p^p\int_{\mathbb{R}^n}\phi^{-t}\cdot{\nabla f}^{\odot p}\alpha(f)\\
		&= \phi^{-t}\cdot Z(f),
	\end{align*}
	which implies the \({\rm SL}(n)\) contravariance.
	
	Let \(\tau\) be a translation. Then,
	\begin{align*}
		Z(f\circ\tau^{-1}) &= p^p\int_{\mathbb{R}^n}{\nabla(f\circ\tau^{-1})}^{\odot p}\alpha(f\circ\tau^{-1})\\
		&= p^p\int_{\mathbb{R}^n}{\left(\nabla f\circ\tau^{-1}\right)}^{\odot p}\alpha(f\circ\tau^{-1})\\
		&= p^p\int_{\mathbb{R}^n}{\nabla f}^{\odot p}\alpha(f)\\
		&= Z(f),
	\end{align*}
	which implies the translation invariance.
	
	It follows from \eqref{fvg} and \eqref{f^g} clearly that \(Z\) is a valuation with \(Z(0)=0\).
	
	Finally, we write \(c=(-p)^pM^{0,p}(P)\) and \(\varphi(s)=\int_0^s\alpha(t)(s-t)^{n-1}{\rm d} t\),
	and use Lemma \ref{thm:cal} to obtain
	\begin{align*}
		\Phi(s)=Z(s\ell_P)=(-p)^ps^{p-n}M^{0,p}(P)\int_0^s\alpha(t)(s-t)^{n-1}{\rm d} t=cs^{p-n}\varphi(s).
	\end{align*}
	For \(k=0,1,2,\dots,n-1\), by induction, we have
	\begin{equation}\label{eqn:3}
		\varphi^{(k)}(s)=\frac{(n-1)!}{(n-1-k)!}\int_0^s\alpha(t)(s-t)^{n-1-k}{\rm d} t,
	\end{equation}
	and \(\varphi^{(n)}(s)=(n-1)!\alpha(s)\). Thus,
	\begin{align}
		\Phi^{(k)}(s) &= c\sum_{j=0}^k{k \choose j}(s^{p-n})^{(k-j)}\varphi^{(j)}(s)\notag\\
		&= c\sum_{j=0}^k{k \choose j}(-1)^{k-j}\frac{(n-p+k-j-1)!}{(n-p-1)!}s^{p-n-k+j}\varphi^{(j)}(s)
		\label{eqn:4}
	\end{align}
	for \(k=0,1,2,\dots,n\). Hence, \(\Phi\in C^n(\mathbb{R}^n)\).
	
	Since \(\alpha\) is bounded, there exist \(m,M\in\mathbb{R}\) such that \(m\leq\alpha(s)\leq M\) for every \(s\in\mathbb{R}\).
	Combined with \eqref{eqn:3}, we have
	\[\frac{(n-1)!}{(n-k)!}m\leq s^{k-n}\varphi^{(k)}(s)\leq\frac{(n-1)!}{(n-k)!}M,\]
	which implies that \(s^{k-n}\varphi^{(k)}(s)\) is bounded for \(k=0,1,2,\dots,n\).
	By \eqref{eqn:4}, we see that
	\[s^{k-p}\Phi^{(k)}(s)=c\sum_{j=0}^k{k \choose j}(-1)^{k-j}\frac{(n-p+k-j-1)!}{(n-p-1)!}s^{j-n}\varphi^{(j)}(s),\]
	and they are bounded consequently for \(k=0,1,2,\dots,n\).

<!-- lem-4: body_md -->
\label{thm:mten}
	Let \(Z:L^{1,p}(\mathbb{R}^n)\to{\rm Sym}^p(\mathbb{R}^n)\) be a continuous, \({\rm SL}(n)\) contravariant, translation invariant valuation with \(Z(0)=0\).
	Then, there is a continuous function \(\zeta:\mathbb{R}\to\mathbb{R}\) such that
	\[Z(s\ell_P)=\zeta(s)M^{0,p}(P)\]
	for every \(s\in\mathbb{R}\) and \(\ell_P\inP^{1,p}(\mathbb{R}^n)\).
<!-- end lem-4 -->
<!-- proof-3: body_md -->
Define \(Y_s:\mathcal P^n_o\to{\rm Sym}^p(\mathbb{R}^n)\) by
	\[Y_s(P)=Z(s\ell_P)\]
	for every \(s\in\mathbb{R}\) and \(P\in\mathcal P^n_o\). Let \(P,Q\in\mathcal P^n_o\) such that \(P\cup Q\in\mathcal P^n_o\).
	We have \(\ell_P\vee\ell_Q=\ell_{P\cup Q}\) and \(\ell_P\wedge\ell_Q=\ell_{P\cap Q}\). Since \(Z\) is a valuation on \(L^{1,p}(\mathbb{R}^n)\), we obtain
	\begin{align*}
		Y_s(P)+Y_s(Q) &= Z(s\ell_P)+Z(s\ell_Q)=Z(s(\ell_P\vee\ell_Q))+Z(s(\ell_P\cap\ell_Q))\\
		&= Z(s\ell_{P\cup Q})+Z(s\ell_{P\cap Q})=Y_s(P\cup Q)+Y_s(P\cap Q),
	\end{align*}
	which shows that \(Y_s\) is a valuation. It is also clear that \(Y_s\) is Borel measurable and \({\rm SL}(n)\) contravariant.
	Thus, by Theorem \ref{thm:body}, there exist continuous functions \(\zeta_1,\zeta_2:\mathbb{R}\to\mathbb{R}\) such that
	\begin{equation}\label{eqn:5}
		Z(s\ell_P)=Y_s(P)=\zeta_1(s)M^{p,0}(P^\ast)+\zeta_2(s)M^{0,p}(P)
	\end{equation}
	for every \(s\in\mathbb{R}\) and \(\ell_P\inP^{1,p}(\mathbb{R}^n)\).
	
	Let \(P\in\mathcal P^n_o\). Take translations \(\tau_1,\dots,\tau_k\) such that \(\tau_i(P/k^i)\) are pairwisely disjoint for \(i=1,2,\dots,k\). Define
	\[f_k=s(\ell_{\tau_1(P/k^1)}\vee\dots\vee\ell_{\tau_k(P/k^k)}),\qquad s\in\mathbb{R},\]
	where \(\ell_{\tau_i(P/k^i)}=\tau_i\circ\ell_{P/k^i}\).
	Then, \(f_k\to0\) in \(W^{1,p}(\mathbb{R}^n)\) as \(k\to\infty\). By the valuation property, the translation invariance and \eqref{eqn:5}, we have
	\begin{align*}
		Z(f_k) &= \sum_{i=1}^kZ(s\ell_{P/k^i})\\
		&= \sum_{i=1}^k\left(\zeta_1(s)M^{p,0}\left((P/k^i)^\ast\right)+\zeta_2(s)M^{0,p}(P/k^i)\right)\\
		&= \zeta_1(s)\sum_{i=1}^kk^{i(n+p)}M^{p,0}(P^\ast)+\zeta_2(s)\sum_{i=1}^kk^{-i(n-p)}M^{0,p}(P).
	\end{align*}
	The continuity of \(Z\) and \(Z(0)=0\) force \(\zeta_1(s)\equiv0\) for every \(s\in\mathbb{R}\).

<!-- lem-5: body_md -->
\label{thm:nh}
	Let \(Z:L^{1,p}(\mathbb{R}^n)\to{\rm Sym}^p(\mathbb{R}^n)\). If \(Z\in\mathcal B_p\), and there is a continuous function \(\beta:\mathbb{R}\to\mathbb{R}\) such that \(Z(s\ell_P)=\beta(s)M^{0,p}(P)\),
	then there is a continuous and bounded function \(\alpha:\mathbb{R}\to\mathbb{R}\) such that
	\[Z(s\ell_P)=J_\alpha^p((s\ell_P)^p).\]
<!-- end lem-5 -->
<!-- proof-4: body_md -->
First, we notice that \(\beta\in C^n(\mathbb{R})\), since \(\Phi(s)=Z(s\ell_P)=\beta(s)M^{0,p}(P)\in C^n(\mathbb{R})\). Next, write
	\(\beta_1(s)=n(-p)^{-p}s^{n-p}\beta(s)\) for \(s\in\mathbb{R}\). Then, \(\beta_1(s)\in C^n(\mathbb{R})\) as well, and
	\begin{align*}
		\beta_1^{(k)}(s) &= n(-p)^{-p}\sum_{j=0}^k{k \choose j}(s^{n-p})^{(j)}\beta^{(k-j)}(s)\\
		&= n(-p)^{-p}\sum_{j=0}^{\min\left\{{k,n-p}\right\}}{k \choose j}\frac{(n-p)!}{(n-p-j)!}s^{n-p-j}\beta^{(k-j)}(s)\\
		&= n(-p)^{-p}\sum_{l=\max\left\{{0,k-n+p}\right\}}^k{k \choose l}\frac{(n-p)!}{(n-p-k+l)!}s^{n-p-k+l}\beta^{(l)}(s)\\
		&= n(-p)^{-p}\sum_{l=\max\left\{{0,k-n+p}\right\}}^k{k \choose l}\frac{(n-p)!}{(n-p-k+l)!}s^{n-k}s^{l-p}\beta^{(l)}(s).
	\end{align*}
	Since \(s^{l-p}\beta^{(l)}(s)\) is bounded due to \(0\leq\max\left\{{0,k-n+p}\right\}\leq l\leq k\leq n\), we obtain \(\beta_1^{(k)}(s)\to0\) as \(s\to0\)
	for \(k=0,1,2,\dots,n-1\).
	
	Now, for \(s\in\mathbb{R}\), let
	\[\alpha(s)=\frac1{n!}\beta_1^{(n)}(s)=\frac{n(-p)^{-p}}{n!}\sum_{j=0}^{n-p}{n \choose j}\frac{(n-p)!}{(n-p-j)!}s^{n-p-j}\beta^{(n-j)}(s).\]
	Then \(\alpha\) is continuous and bounded. Moreover,
	\begin{align*}
		\int_0^s \alpha(t)(s-t)^{n-1}{\rm d} t &= \frac1{n!}\int_0^s\beta_1^{(n)}(t)(s-t)^{n-1}{\rm d} t
		= \frac1{n!}\int_0^s(s-t)^{n-1}{\rm d}\beta_1^{(n-1)}(t)\\
		&= \frac1{n!}\left((s-t)^{n-1}\beta_1^{(n-1)}(t)\Big|_0^s-\int_0^s\beta_1^{(n-1)}(t){\rm d}(s-t)^{n-1}\right)\\
		&= \frac1{n!}(n-1)\int_0^s\beta_1^{(n-1)}(t)(s-t)^{n-2}{\rm d} t\\
		&= \cdots\\
		&= \frac1n\int_0^s\beta'_1(t){\rm d} t\\
		&= \frac1n\beta_1(s),
	\end{align*}
	i.e. \(\beta_1(s)=n\int_0^s\alpha(t)(s-t)^{n-1}{\rm d} t\). It follows that
	\[\beta(s)=\frac{(-p)^p}ns^{p-n}\beta_1(s)=(-p)^ps^{p-n}\int_0^s\alpha(t)(s-t)^{n-1}{\rm d} t.\]
	Therefore, by Lemma \ref{thm:cal}, we obtain
	\begin{align*}
		Z(s\ell_P)=\beta(s)M^{0,p}(P)=(-p)^ps^{p-n}M^{0,p}(P)\int_0^s\alpha(t)(s-t)^{n-1}{\rm d} t=J_\alpha^p((s\ell_P)^p).
	\end{align*}
<!-- end proof-4 -->
<!-- proof-5: body_md -->
The ``if'' part follows from Lemma \ref{thm:if}. It remains to prove the ``only if '' part.
	The classification of valuations on \(W^{1,p}(\mathbb{R}^n)\) is reduced to that on its dense subspace \(L^{1,p}(\mathbb{R}^n)\) since \(Z\) is continuous.
	Indeed, it suffices to show the classification on \(f=s\ell_P\) for \(s\in\mathbb{R}\) and \(\ell_P\inP^{1,p}(\mathbb{R}^n)\) due to Lemma \ref{thm:scm}.
	By Lemma \ref{thm:mten}, there is a continuous function \(\beta:\mathbb{R}\to\mathbb{R}\) such that
	\[Z(s\ell_P)=\beta(s)M^{0,p}(P)\]
	for every \(s\in\mathbb{R}\) and \(\ell_P\inP^{1,p}(\mathbb{R}^n)\). Further applying Lemma \ref{thm:nh}, we conclude that there is a continuous and bounded function 
	\(\alpha:\mathbb{R}\to\mathbb{R}\) such that
	\[Z(s\ell_P)=J_\alpha^p((s\ell_P)^p).\]
<!-- end proof-5 -->
<!-- proof-6: body_md -->
Due to \eqref{eqn:jap}, the Fisher information tensor \(J^p\) is \(p\)-homogeneous. Then, we obtain the ``if'' part from Lemma \ref{thm:if}.
	
	Next, we prove the ``only if'' part. Similar to the proof of Theorem \ref{thm:m2}, it suffices to show the classification on \(f=s\ell_P\) for \(s\in\mathbb{R}\) and \(\ell_P\inP^{1,p}(\mathbb{R}^n)\). By Lemma \ref{thm:mten}, there is a continuous function \(\beta:\mathbb{R}\to\mathbb{R}\) such that
	\[Z(s\ell_P)=\beta(s)M^{0,p}(P).\]
	Since \(Z\) is \(p\)-homogeneous, there is \(\gamma\in\mathbb{R}\) such that \(\beta(s)=\gamma s^p\) for \(s\in\mathbb{R}\).
	If we set \(\Phi(s)=Z(s\ell_P)\), i.e. \(\Phi(s)=\bar\gamma s^p\), where \(\bar\gamma=\gammaM^{0,p}(P),\)
	it is clear that \(\Phi\in C^n(\mathbb{R})\) and \(s^{k-p}\Phi^{(k)}(s)\) is bounded for \(k=0,1,2,\dots,n\).
	Therefore, Lemma \ref{thm:nh} completes the proof.
