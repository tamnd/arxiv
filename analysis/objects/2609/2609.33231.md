<!-- maintheorem-1: body_md -->
For every prime $p\ge5$, put $\mathbb{F}_p=\mathbb Z/p\mathbb Z$. There exists a family of Dijkgraaf--Witten modular categories $\{\mathcal{C}_u\}_{u\in\mathbb{F}_p^\times}$ associated with the group $(\mathbb Z/p\mathbb Z)^3$ such that the following statements hold for all $u,v\in\mathbb{F}_p^\times$.
\begin{enumerate}[(A)]
\item There is a bijection $\Lambda_{uv}:\operatorname{Irr}(\mathcal{C}_u)\to\operatorname{Irr}(\mathcal{C}_v)$ with $\Lambda_{uv}(\mathbf{1})=\mathbf{1}$ such that for every framed oriented link $L\subset S^3$ with $n\in\{1,2\}$ components and all simple objects $X_1,\dots,X_n$ of $\mathcal{C}_u$,
\begin{equation*}
\operatorname{RT}_{\mathcal{C}_u}(L;X_1,\dots,X_n)=\operatorname{RT}_{\mathcal{C}_v}(L;\Lambda_{uv}X_1,\dots,\Lambda_{uv}X_n).
\end{equation*}
Consequently the matrices $\widetilde S,S,T$ and the matrices $\widetilde W,W$ of \cite[\S2.1.2, Definition~2.1]{BDGRTW} of $\mathcal{C}_u$ and $\mathcal{C}_v$ coincide under $\Lambda_{uv}$.
\item If $\mathcal{C}_u$ and $\mathcal{C}_v$ are braided equivalent, then $u=\pm v$. In particular, if $u\neq\pm v$ then $\mathcal{C}_u$ and $\mathcal{C}_v$ are not ribbon equivalent.
\end{enumerate}

<!-- definition-1: body_md -->
\label{def:cocycles}
For $a,b,c\in V$ and $u\in\mathbb{F}_p$ put
\begin{equation*}
\kappa(a,b,c):=\zeta^{\sum_{i=1}^{3}[a_i]\,C_p(b_i,c_i)},\qquad
\tau_u(a,b,c):=\zeta^{\frac u6\det(a,b,c)},\qquad
\omega_u:=\kappa\,\tau_u .
\end{equation*}

<!-- definition-2: body_md -->
\label{def:Vec}
The objects of $\mathrm{Vec}_\Gamma^\nu$ are finite-dimensional $\Gamma$-graded vector spaces $X=\bigoplus_{x\in\Gamma}X_x$, and the morphisms are degree-preserving linear maps. The tensor product is
\begin{equation*}
    (X\otimes Y)_z=\bigoplus_{x+y=z}X_x\otimes Y_y,
\end{equation*}
with the usual tensor product of linear maps. The unit is $\mathbf1=\mathbb C$ concentrated in degree $0$, with the standard unit constraints~\cite[Remark~2.6.3]{EGNO}. The constraint on associativity is
\begin{equation*}
    \alpha_{X,Y,Z}:(X\otimes Y)\otimes Z\longrightarrow X\otimes(Y\otimes Z),\qquad \alpha_{X,Y,Z}((v\otimes w)\otimes t) =\nu(x,y,z)\,v\otimes(w\otimes t),
\end{equation*}
for $v\in X_x$, $w\in Y_y$ and $t\in Z_z$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:center}
The Drinfel'd center of $\mathcal{C}$ is the category $\mathcal{Z}(\mathcal{C})$ with the following data.
Each object of $\mathcal{Z}(\mathcal{C})$ is a pair $(X,c_{-,X})$, where $X\in\mathcal{C}$ and $c_{V,X}:V\otimes X\to X\otimes V$ is a family of isomorphisms natural in $V\in\mathcal{C}$, called a \emph{half-braiding}. It satisfies $c_{\mathbf{1},X}=\mathrm{id}_X$ and, for all $V,W\in\mathcal{C}$,
\begin{equation}\label{eq:halfbraiding}
    c_{V\otimes W,X}=\alpha_{X,V,W}(c_{V,X}\otimes\mathrm{id}_W)\alpha^{-1}_{V,X,W}(\mathrm{id}_V\otimes c_{W,X})\alpha_{V,W,X}.
\end{equation}
A morphism $(X,c_{-,X})\to(X',c_{-,X'})$ is a morphism $f:X\to X'$ in $\mathcal{C}$ such that, for every $V$,
\begin{equation*}
    (f\otimes\mathrm{id}_V)c_{V,X}=c_{V,X'}(\mathrm{id}_V\otimes f).
\end{equation*}
The tensor product is $(X,c_{-,X})\otimes(Y,c_{-,Y})=(X\otimes Y,c_{-,X\otimes Y})$, where
\begin{equation}\label{eq:tensorhalf}
    c_{V,X\otimes Y}=\alpha^{-1}_{X,Y,V}(\mathrm{id}_X\otimes c_{V,Y})\alpha_{X,V,Y}(c_{V,X}\otimes\mathrm{id}_Y)\alpha^{-1}_{V,X,Y}.
\end{equation}
The unit is $(\mathbf{1},\mathrm{id})$, and the associativity and unit constraints are inherited from $\mathcal{C}$. The braiding is
\begin{equation}\label{eq:braiding}
    c^{\mathcal{Z}}_{(X,c_{-,X}),(Y,c_{-,Y})}=c_{X,Y}:X\otimes Y\longrightarrow Y\otimes X,
\end{equation}
using the half-braiding of the \emph{second} object evaluated at the first. The forgetful tensor functor $\operatorname{Forg}:\mathcal{Z}(\mathcal{C})\to\mathcal{C}$, called \emph{central functor} in~\cite[Definition~8.8.6]{EGNO}, sends $(X,c_{-,X})$ to $X$ and acts identically on underlying morphisms.

<!-- proposition-1: body_md -->
\label{prop:centerabelian}
Let $\Gamma$ be a finite abelian group and $\nu$ a normalized $3$-cocycle. The center $\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)$ has the following standard description. For each $a\in\Gamma$, the function
\begin{equation}\label{eq:multiplier}
    \gamma^\nu_a(x,y)=\frac{\nu(x,a,y)}{\nu(a,x,y)\nu(x,y,a)}.
\end{equation}
is a normalized $2$-cocycle.
\begin{enumerate}[(i)]
\item Objects are equivalently graded vector spaces $X=\bigoplus_aX_a$ with a $\gamma_a^\nu$-projective action on each $X_a$: $\rho_X(0)=\mathrm{id}$ and $\rho_X(x)\rho_X(y)|_{X_a}=\gamma_a^\nu(x,y)\rho_X(x+y)|_{X_a}$. Morphisms are graded intertwiners, and each $X_a$ is a center subobject. The simple objects are therefore indexed by pairs $(a,\pi)$, where $\pi$ is an irreducible $\gamma_a^\nu$-projective representation of $\Gamma$. The positive categorical dimension of $X$ is $d(X)=\dim_{\mathbb{C}}X$; in particular, an object is invertible if and only if its underlying vector space is one-dimensional.

\item The corresponding half-braiding is
\begin{equation}\label{eq:halfexplicit}
    c_{U,X}(w\otimes v)=\rho_X(x)v\otimes w, \qquad w\in U_x,\quad v\in X.
\end{equation}
The tensor product is determined by~\eqref{eq:tensorhalf}, and the braiding is $c^{\mathcal{Z}}(v\otimes w)=\rho_Y(a)w\otimes v$ for $v\in X_a$, $w\in Y$.

\item The full zero-flux subcategory is
$\mathcal{E}\cong\operatorname{Rep}(\Gamma)$ as a braided monoidal category, whose tensor action is $\rho_{X\otimes Y}(x)=\rho_X(x)\otimes\rho_Y(x)$, associator is the identity, and braiding is the flip.
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:cocycle}
For every $u\in\mathbb{F}_p$, $\kappa$ and $\tau_u$ (hence $\omega_u$) are normalized $3$-cocycles on $V$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Normalization holds because $[0]=0$, $C_p(0,s)=C_p(r,0)=0$ and $\det$ vanishes when an argument is $0$. Writing $\nu=\zeta^{f}$ with $f:V^3\to\mathbb{F}_p$, we can express~\eqref{eq:cocycle} as
\begin{equation*}
    (\delta f)(x,y,z,w):=f(y,z,w)-f(x+y,z,w)+f(x,y+z,w)-f(x,y,z+w)+f(x,y,z)=0\ \text{ in }\mathbb{F}_p .
\end{equation*}
For $\tau_u$, $f=\frac u6\det$ is trilinear, and expanding the second, third and fourth terms by linearity shows that the eight resulting terms cancel in pairs.
For $\kappa$, $f=\sum_if_i$ with $f_i(x,y,z)=[x_i]C_p(y_i,z_i)$, and it suffices to treat one $f_i$. For $b,c,d\in\mathbb{F}_p$ both $C_p(b,c)+C_p(b+c,d)$ and $C_p(c,d)+C_p(b,c+d)$ equal $([b]+[c]+[d]-[b+c+d])/p$. Hence, for $a,b,c,d\in\mathbb{F}_p$ and $F(a,b,c)=[a]C_p(b,c)$,
\begin{align*}
(\delta F)(a,b,c,d)&=([b]-[a+b])C_p(c,d)+[a]\big(C_p(b+c,d)-C_p(b,c+d)+C_p(b,c)\big)\\
&=([a]+[b]-[a+b])\,C_p(c,d)=p\,C_p(a,b)\,C_p(c,d)\equiv0\pmod p. \qedhere
\end{align*}
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:multiplier}
For $u\in\mathbb{F}_p$ and $a,x,y\in V$, the multiplier~\eqref{eq:multiplier} can be written as
\begin{equation*}
    \gamma^{\omega_u}_a(x,y)=\zeta^{-\sum_i[a_i]C_p(x_i,y_i)-\frac u2\det(a,x,y)}=(\partial g_a)(x,y)\cdot b^u_a(x,y),\qquad b^u_a(x,y):=\zeta^{-\frac u2\det(a,x,y)} .
\end{equation*}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
By \eqref{eq:multiplier}, the contribution of $\kappa$ to the exponent of $\zeta$ is
\begin{equation*}
    \sum_i\big([x_i]C_p(a_i,y_i)-[a_i]C_p(x_i,y_i)-[x_i]C_p(y_i,a_i)\big)=-\sum_i[a_i]C_p(x_i,y_i),
\end{equation*}
because $C_p$ is symmetric. For the $\tau_u$ factor, $\det(x,a,y)=-\det(a,x,y)$ and $\det(x,y,a)=\det(a,x,y)$ give
\begin{equation*}
    \frac u6\big(\det(x,a,y)-\det(a,x,y)-\det(x,y,a)\big)=\frac u6(-3)\det(a,x,y)=-\frac u2\det(a,x,y).
\end{equation*}
Finally, the definition of the carry gives
\begin{equation*}
    (\partial g_a)(x,y)=\eta^{-\sum_i[a_i]([x_i]+[y_i]-[x_i+y_i])}=\eta^{-p\sum_i[a_i]C_p(x_i,y_i)}=\zeta^{-\sum_i[a_i]C_p(x_i,y_i)}.
\end{equation*}

<!-- lemma-3: body_md -->
\label{lem:rank2}
Let $P\leq V$ be a subgroup with $\dim_{\mathbb{F}_p}P\leq2$. For every $u\in\mathbb{F}_p$, one has $\det(x,y,z)=0, \tau_u(x,y,z)=1$ for all $x,y,z\in P$, and hence $\omega_u|_{P^3}=\kappa|_{P^3}$. The first half is trivial, whereas the formula for the multiplier follows from Lemma~\ref{lem:multiplier} and $\det(a,x,y)=0$.

<!-- lemma-4: body_md -->
\label{lem:maschke}
Let $H$ be a finite abelian group and $b$ a normalized $2$-cocycle on $H$ with values in roots of unity. The twisted group algebra $\mathbb{C}^b[H]$, with basis $U_h$ and $U_gU_h=b(g,h)U_{g+h}$, is semisimple. Equivalently, every finite-dimensional $b$-projective representation of $H$, viewed as a $\mathbb{C}^b[H]$-module via $U_h\mapsto\rho(h)$, is completely reducible.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Let $Z\subset\mathbb{C}^\times$ be the finite subgroup generated by the values of $b$. By the cocycle identity and normalization, $H'=Z\times H$ with multiplication $(\lambda,g)(\lambda',h)=(\lambda\lambda'\,b(g,h),\,g+h)$ is a finite group, and $(\lambda,h)\mapsto\lambda U_h$ extends linearly to a surjective algebra homomorphism $\mathbb{C}[H']\to\mathbb{C}^b[H]$. Hence every finite-dimensional $\mathbb{C}^b[H]$-module is a $\mathbb{C}[H']$-module with the same submodules. Since $\mathbb{C}[H']$ is semisimple by Maschke's theorem~\cite[Theorem~4.1.1(i)]{EGHLLSVY}, such modules are completely reducible, and $\mathbb{C}^b[H]$ is semisimple, both by~\cite[Proposition~3.5.8]{EGHLLSVY}.

<!-- proposition-2: body_md -->
\label{prop:simples}
Let $u\in\mathbb{F}_p^\times$. For $\xi\in V^\vee$ let $E_\xi$ be $\mathbb{C}$ in degree $0$ with $\rho(x)=\zeta^{\xi(x)}$. Then $E_\xi$ is a simple object of $\mathcal{C}_u$ and $E_0=\mathbf{1}$. For nonzero flux, let $a\in V\setminus\{0\}$ and $k\in\mathbb{F}_p$ and choose $r,s\in V$ with $\det(a,r,s)=1$. Every $x\in V$ is uniquely $x=t_xa+m_xr+n_xs$ with $t_x=\det(x,r,s)$, $m_x=\det(a,x,s)$, $n_x=\det(a,r,x)$. On $\mathbb{C}^p$ with basis $(e_j)_{j\in\mathbb{F}_p}$ put
\begin{equation*}
    \pi_{a,k}(x)e_j:=\zeta^{kt_x+\frac u2m_xn_x+un_xj}\,e_{j+m_x}.
\end{equation*}
Then $\pi_{a,k}$ is a $b^u_a$-projective representation, $\rho:=g_a\,\pi_{a,k}$ is a $\gamma^{\omega_u}_a$-projective representation, and
\begin{equation*}
    X_u(a,k):=\big(\mathbb{C}^p\text{ placed in degree }a,\ \rho\big)
\end{equation*}
is a simple object of $\mathcal{C}_u$ with $\rho(a)=g_a(a)\zeta^k\,\mathrm{id}$.

Every simple object of $\mathcal{C}_u$ is isomorphic to exactly one of the objects $E_\xi$ \textup{(}$\xi\in V^\vee$\textup{)}, $X_u(a,k)$ \textup{(}$a\neq0$, $k\in\mathbb{F}_p$\textup{)}. The isomorphism class of $X_u(a,k)$ does not depend on the choice of $r,s$, since it is the unique simple object of flux $a$ with $\rho(a)=g_a(a)\zeta^k$.

The characters $\chi_X(x):=\operatorname{Tr}\rho_X(x)$ are
\begin{equation*}
\chi_{E_\xi}(x)=\zeta^{\xi(x)},\qquad
\chi_{X_u(a,k)}(x)=\begin{cases}p\,g_a(x)\,\zeta^{kt}&\text{if }x=ta,\ t\in\mathbb{F}_p,\\ 0&\text{if }x\notin\mathbb{F}_p a.\end{cases}
\end{equation*}
These characters do not depend on $u$.
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
By Proposition~\ref{prop:centerabelian}, the simple objects of $\mathcal{C}_u$ of flux $a$ are classified by the irreducible $\gamma_a^{\omega_u}$-projective representations of $V$. For $a=0$, one has $\gamma^{\omega_u}_0\equiv1$, so the irreducibles are precisely the characters $x\mapsto\zeta^{\xi(x)}$, $\xi\in V^\vee$ since $V$ is finite abelian. These give the $E_\xi$, and~\eqref{eq:halfexplicit} gives $E_0=\mathbf{1}$.

Now let $a\neq0$. A direct substitution in the defining formula for $\pi_{a,k}$ gives
\begin{equation*}
	\pi_{a,k}(x)\pi_{a,k}(y)=\zeta^{-\frac u2\det(a,x,y)}\pi_{a,k}(x+y)=b_a^u(x,y)\pi_{a,k}(x+y).
\end{equation*}
Hence Lemma~\ref{lem:multiplier} implies that $\rho=g_a\pi_{a,k}$ is a $\gamma_a^{\omega_u}$-projective representation. Moreover,
\begin{equation*}
	\pi_{a,k}(s)e_j=\zeta^{uj}e_j,\qquad \pi_{a,k}(r)e_j=e_{j+1}.
\end{equation*}
Since $u\neq0$, the first operator has $p$ distinct eigenspaces and the second permutes them transitively. Thus $\pi_{a,k}$, and hence $\rho$, is irreducible and they are pairwise non-isomorphic since $\pi_{a,k}(a)=\zeta^k\,\mathrm{id}$.

By Lemma~\ref{lem:maschke}, $\mathbb{C}^{b_a^u}[V]$ is semisimple of dimension $p^3$, so the dimensions of its simple modules satisfy $\sum_U(\dim U)^2=p^3$~\cite[Proposition~3.5.8]{EGHLLSVY}. The $p$ irreducibles above already account for this dimension:
\begin{equation*}
	\sum_{k\in\mathbb{F}_p}(\dim\pi_{a,k})^2=p\cdot p^2=p^3.
\end{equation*}
Therefore they exhaust the irreducible $b_a^u$-projective representations.  Multiplication by the scalar cochain $g_a$ gives an equivalence between $b_a^u$-projective and $\gamma_a^{\omega_u}$-projective representations, hence the $\rho_{a,k}=g_a\pi_{a,k}$ exhaust the irreducible $\gamma_a^{\omega_u}$-projective representations. By Proposition~\ref{prop:centerabelian}, the $X_u(a,k)$ therefore exhaust the simple objects of flux $a$. At $x=a$ one has
\begin{equation*}
	\rho(a)=g_a(a)\zeta^k\,\mathrm{id},
\end{equation*}
which distinguishes the $p$ classes at fixed flux and also shows that the isomorphism class of $X_u(a,k)$ is independent of the auxiliary choice of $r,s$.

Finally, the defining formula for $\pi_{a,k}$ gives
\begin{equation*}
	\operatorname{Tr}\pi_{a,k}(x)=\delta_{m_x,0}\,\zeta^{kt_x}\sum_{j\in\mathbb{F}_p}\zeta^{un_xj}=p\,\delta_{m_x,0}\delta_{n_x,0}\,\zeta^{kt_x},
\end{equation*}
since $u\neq0$. Thus
\begin{equation*}
	\chi_{X_u(a,k)}(x) = \begin{cases} 
	p\,g_a(x)\zeta^{kt},&x=ta,\ t\in\mathbb{F}_p,\\
	0,&x\notin\mathbb{F}_p a,
	\end{cases}
\end{equation*}
while $\chi_{E_\xi}(x)=\zeta^{\xi(x)}$.  Since $g_a$ is independent of $u$, so are these character formulas.

<!-- proposition-3: body_md -->
\label{prop:restriction}
Let $\Gamma$ be a finite abelian group, let $\nu$ be a normalized $3$-cocycle, and let $P\leq\Gamma$, with $\nu_P:=\nu|_{P^3}$. The full subcategory $\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)_P\subset\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)$ consisting of objects whose underlying graded vector spaces are supported on $P$ is a fusion subcategory.  Restricting the half-braiding to $\mathrm{Vec}_P^{\nu_P}$ defines a $\mathbb{C}$-linear ribbon functor
\begin{equation*}
	\operatorname{Res}_P:\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)_P\longrightarrow\mathcal{Z}(\mathrm{Vec}_P^{\nu_P}),\qquad(X,c_{-,X})\longmapsto\left(X,c_{-,X}|_{\mathrm{Vec}_P^{\nu_P}}\right),
\end{equation*}
where $X$ on the right is regarded as an object of $\mathrm{Vec}_P^{\nu_P}$, and it acts as the identity on morphisms in $\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)_P$. Consequently, for every framed oriented link $L$ and colors $X_1,\ldots,X_n\in\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)_P$,
\begin{equation*}
	\operatorname{RT}_{\mathcal{Z}(\mathrm{Vec}_P^{\nu_P})}
   (L;\operatorname{Res}_PX_1,\ldots,\operatorname{Res}_PX_n)=\operatorname{RT}_{\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)}
   (L;X_1,\ldots,X_n).
\end{equation*}
<!-- end proposition-3 -->
<!-- proof-6: body_md -->
The tensor unit is supported at $0\in P$, and support in $P$ is preserved by finite direct sums, direct summands, tensor products and duals. Hence $\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)_P$ is a fusion subcategory.

If $(X,c_{-,X})$ is supported on $P$, then $X$ is canonically an object of $\mathrm{Vec}_P^{\nu_P}$, and restricting the family $c_{-,X}$ to objects of $\mathrm{Vec}_P^{\nu_P}$ preserves the half-braiding identity. Likewise, every morphism in $\mathcal{Z}(\mathrm{Vec}_\Gamma^\nu)_P$ remains a morphism after restriction. Thus $\operatorname{Res}_P$ is a well-defined $\mathbb{C}$-linear functor.

Since the monoidal and rigid structures on $\mathrm{Vec}_P^{\nu_P}$ are the restrictions of those on $\mathrm{Vec}_\Gamma^\nu$, $\operatorname{Res}_P$ is strict monoidal and preserves the dual objects, evaluation maps and coevaluation maps. Restricting half-braidings also preserves the braiding. The positive spherical structure on $\mathrm{Vec}_\Gamma^\nu$ restricts to that on $\mathrm{Vec}_P^{\nu_P}$, and hence the induced spherical structures on their centers are preserved. Since the Drinfeld isomorphism $\upsilon_X:X\to X^{**}$ is constructed functorially from the braiding, evaluation and coevaluation maps, the preceding compatibilities imply
\begin{equation*}
	\operatorname{Res}_P(\upsilon_X)=\upsilon_{\operatorname{Res}_PX}.
\end{equation*}
The induced spherical structures are also preserved, so
\begin{equation*}
	\operatorname{Res}_P(\psi_X)=\psi_{\operatorname{Res}_PX}.
\end{equation*}
Hence, using $\theta=\upsilon^{-1}\psi$,
\begin{equation*}
	\operatorname{Res}_P(\theta_X)=\upsilon_{\operatorname{Res}_PX}^{-1}\psi_{\operatorname{Res}_PX}=\theta_{\operatorname{Res}_PX}.
\end{equation*}
Thus $\operatorname{Res}_P$ is braided monoidal and preserves the canonical twist, so it is a ribbon functor. The asserted equality of Reshetikhin--Turaev evaluations follows from the functoriality recalled in Section~\ref{subsec:RT-conventions}.

<!-- lemma-5: body_md -->
\label{lem:match}
Let $X\in\operatorname{Irr}(\mathcal{C}_u)$ have flux in $P$. Then $\operatorname{Res}_PX\cong\operatorname{Res}_P(\Lambda_{uv}X)$ in $\mathcal{B}_P$.
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
For $X=E_\xi$, the two restrictions are identical, namely both are $\mathbb{C}$ in degree zero with action $\rho(x)=\zeta^{\xi(x)}, x\in P$.

Now let $X=X_u(a,k)$ with $a\in P\setminus\{0\}$, and put $X':=X_v(a,k)$. By Lemma~\ref{lem:rank2}, the restricted projective representations $\rho|_P$ and $\rho'|_P$ have the same multiplier $\partial g_a|_{P\times P}$. Hence
\begin{equation*}
	\sigma(x):=g_a(x)^{-1}\rho(x),\qquad\sigma'(x):=g_a(x)^{-1}\rho'(x)
\end{equation*}
are ordinary representations of $P$. By Proposition~\ref{prop:simples}, the characters of $X_u(a,k)$ and $X_v(a,k)$ coincide. Since $\sigma=g_a^{-1}\rho$ and $\sigma'=g_a^{-1}\rho'$, their ordinary characters also coincide:
\begin{equation*}
	\operatorname{Tr}\sigma(x)=g_a(x)^{-1}\chi_X(x)=g_a(x)^{-1}\chi_{X'}(x)=\operatorname{Tr}\sigma'(x).
\end{equation*}
Therefore $\sigma\cong\sigma'$ by the usual character criterion for finite-group representations. Multiplying back by $g_a$ gives $\rho|_P\cong\rho'|_P$. By Proposition~\ref{prop:centerabelian}, this is precisely an isomorphism
\begin{equation*}
	\operatorname{Res}_PX\cong\operatorname{Res}_PX'
\end{equation*}
in $\mathcal{B}_P$.

<!-- theorem-1: body_md -->
\label{thm:step1}
Let $u,v\in\mathbb{F}_p^\times$. For every framed oriented link $L$ with $n\in\{1,2\}$ components and all $X_1,\dots,X_n\in\operatorname{Irr}(\mathcal{C}_u)$,
\begin{equation*}
\operatorname{RT}_{\mathcal{C}_u}(L;X_1,\dots,X_n)=\operatorname{RT}_{\mathcal{C}_v}(L;\Lambda_{uv}X_1,\dots,\Lambda_{uv}X_n).
\end{equation*}
Moreover $\theta_{\Lambda_{uv}X}=\theta_X$ and $d(\Lambda_{uv}X)=d(X)$ for every simple $X$.
<!-- end theorem-1 -->
<!-- proof-8: body_md -->
Let $a_j$ be the flux of $X_j$ and $P:=\operatorname{span}(a_1,\dots,a_n)$, so $\dim P\le n\le2$. The objects $X_j$ lie in $(\mathcal{C}_u)_P$, and $\Lambda_{uv}X_j$ with the same flux lie in $(\mathcal{C}_v)_P$. By Proposition~\ref{prop:restriction}, Lemma~\ref{lem:match} and Section~\ref{subsec:RT-conventions} restricted to $\mathcal{B}_P$),
\begin{align*}
\operatorname{RT}_{\mathcal{C}_u}(L;X_1,\dots,X_n)&=\operatorname{RT}_{\mathcal{B}_P}(L;\operatorname{Res}_PX_1,\dots,\operatorname{Res}_PX_n)\\
&=\operatorname{RT}_{\mathcal{B}_P}(L;\operatorname{Res}_P\Lambda_{uv}X_1,\dots,\operatorname{Res}_P\Lambda_{uv}X_n)=\operatorname{RT}_{\mathcal{C}_v}(L;\Lambda_{uv}X_1,\dots,\Lambda_{uv}X_n).
\end{align*}
To compare twists, take $P=\mathbb{F}_p a$, where $a$ is the flux of $X$. By Proposition~\ref{prop:restriction}, $\theta_X$ and $\theta_{\Lambda_{uv}X}$ are the twists of the isomorphic objects $\operatorname{Res}_PX\cong\operatorname{Res}_P\Lambda_{uv}X$ of $\mathcal{B}_P$, and the twist is natural, so they are equal. The equality $d(\Lambda_{uv}X)=d(X)$ comes from the case $L=O^0$.

<!-- lemma-6: body_md -->
\label{lem:invertible}
A simple object of $\mathcal{C}_u$ is invertible if and only if it is isomorphic to some $E_\xi$, and $E_\xi\otimes E_{\xi'}\cong E_{\xi+\xi'}$.

Let $\mathcal{E}_u\subseteq\mathcal{C}_u$ be the full subcategory of objects supported in degree $0$. Its objects are exactly the direct sums of invertible simple objects of $\mathcal{C}_u$. It is a fusion subcategory, and $(X,\rho_X)\mapsto(X,\rho_X)$ is an isomorphism of braided monoidal categories $\mathcal{E}_u\to\operatorname{Rep}(V)$, where $\operatorname{Rep}(V)$ has the trivial associativity constraint and the flip braiding.
<!-- end lemma-6 -->
<!-- proof-10: body_md -->
By Proposition~\ref{prop:centerabelian}, a simple object is invertible exactly when its underlying space is one-dimensional. Proposition~\ref{prop:simples} gives dimension $1$ for the $E_\xi$ and dimension $p>1$ for the $X_u(a,k)$. Proposition~\ref{prop:centerabelian} gives $\rho_{E_\xi\otimes E_{\xi'}}(x)=\zeta^{\xi(x)+\xi'(x)}$, which proves the tensor product formula. <!-- lemma-7: body_md -->
\label{lem:Epreserved}
$\Phi(\mathcal{E}_u)\subseteq\mathcal{E}_v$, and $\Phi$ restricts to a braided tensor equivalence $\mathcal{E}_u\to\mathcal{E}_v$.
<!-- end lemma-7 -->
<!-- proof-11: body_md -->
An equivalence sends simple objects to simple objects, and the tensor structure sends invertible objects to invertible objects because $\Phi(X)\otimes\Phi(Y)\cong\Phi(X\otimes Y)\cong\Phi(\mathbf{1})\cong\mathbf{1}$. Lemma~\ref{lem:invertible} therefore implies that $\Phi(E_\xi)$ is isomorphic to some $E_{\xi'}$. Since $\mathcal{E}_u$ consists of direct sums of the $E_\xi$, and $\mathcal{E}_v$ is closed under direct sums and isomorphisms, it follows that $\Phi(\mathcal{E}_u)\subseteq\mathcal{E}_v$. 

Let $\Psi:\mathcal{C}_v\to\mathcal{C}_u$ be a monoidal tensor quasi-inverse of $\Phi$~\cite[Remark~2.4.10]{EGNO} with monoidal isomorphisms $\varepsilon:\Phi\Psi\Rightarrow\mathrm{id}$ and $\Psi\Phi\cong\mathrm{id}$, and we intend to show that it is also braided. For $X,Y\in\mathcal C_v$ put $f=\Psi(c_{X,Y})\,J^\Psi_{X,Y}$ and $g=J^\Psi_{Y,X}\,c_{\Psi X,\Psi Y}$. Monoidality of $\varepsilon$ means $\Phi(J^\Psi_{X,Y})J^\Phi_{\Psi X,\Psi Y}=\varepsilon_{X\otimes Y}^{-1}(\varepsilon_X\otimes\varepsilon_Y)$. Using this together with naturality of $\varepsilon$ and of $c$, and that $\Phi$ is braided: $\Phi(f)\,J^\Phi=\varepsilon^{-1}_{Y\otimes X}\,c_{X,Y}\,(\varepsilon_X\otimes\varepsilon_Y)=\Phi(g)\,J^\Phi$. Since $J^\Phi$ is invertible and $\Phi$ is faithful, $f=g$. So $\Psi$ is braided.

The same argument gives $\Psi(\mathcal{E}_v)\subseteq\mathcal{E}_u$. Hence, for every $Y\in\mathcal{E}_v$, $Y\cong\Phi\Psi(Y)$ with $\Psi(Y)\in\mathcal{E}_u$, so $\Phi|_{\mathcal{E}_u}:\mathcal{E}_u\to\mathcal{E}_v$ is essentially surjective. It is fully faithful because $\Phi$ is fully faithful and $\mathcal{E}_u,\mathcal{E}_v$ are full subcategories of $\mathcal{C}_u,\mathcal{C}_v$, respectively.  Thus $\Phi|_{\mathcal{E}_u}$ is a braided tensor equivalence.

<!-- lemma-8: body_md -->
\label{lem:regular}
$A_u$ is a connected \'etale algebra in $\mathcal{C}_u$, and $A_u\cong\bigoplus_{\xi\in V^\vee}E_\xi$ as an object. Moreover, every connected \'etale algebra $B$ in $\mathcal{E}_v$ whose underlying vector space has dimension $|V|$ is isomorphic to $A_v$ as an algebra. Consequently, $\Phi(A_u)\cong A_v$ as algebras in $\mathcal{C}_v$.
<!-- end lemma-8 -->
<!-- proof-12: body_md -->
Since $\mathcal E_u\subset\mathcal C_u$ is Tannakian, its regular algebra $A_u$ is connected \'etale by~\cite[Example~3.3(i)]{DMNO}. The decomposition $A_u\cong\bigoplus_{\xi\in V^\vee}E_\xi$ follows from the regular-representation decomposition~\cite[Theorem~4.1.1(ii)]{EGHLLSVY}, since $V$ is abelian.

Under the braided equivalence $\mathcal{E}_v\simeq\operatorname{Rep}(V)$, connected \'etale algebras correspond to function algebras on transitive finite $V$-sets~\cite[Lemma~3.3.1]{Davydov}. Thus $B\cong\operatorname{Fun}(V/H)$ for some subgroup $H\leq V$. Since $\dim_{\mathbb{C}}B=[V:H]=|V|$,  we have $H=\{0\}$, and hence $B\cong\operatorname{Fun}(V)=A_v$ as algebras. 

Since $\Phi$ is a braided tensor equivalence, $\Phi(A_u)$ is a connected \'etale algebra in $\mathcal{E}_v$ of dimension $|V|$. By the preceding classification of connected \'etale algebras in $\mathcal{E}_v\simeq\operatorname{Rep}(V)$, it follows that $\Phi(A_u)\cong A_v$ as algebras.

<!-- lemma-9: body_md -->
\label{lem:transport}
Let $\Psi:\mathcal{C}\to\mathcal{D}$ be a braided tensor equivalence between braided fusion categories, with tensor structure $J$, and let $A$ be a connected \'etale algebra in $\mathcal{C}$. Then $(M,\mu)\mapsto(\Psi M,\Psi(\mu)\circ J_{M,A})$ is a monoidal equivalence $\mathcal{C}_A\to\mathcal{D}_{\Psi(A)}$. If $\varphi:B\to B'$ is an isomorphism of connected \'etale algebras in $\mathcal{D}$, then $(M,\mu)\mapsto
(M,\mu\circ(\mathrm{id}_M\otimes\varphi^{-1}))$ defines a monoidal equivalence $\mathcal{D}_B\to\mathcal{D}_{B'}$.
<!-- end lemma-9 -->
<!-- proof-14: body_md -->
A strong monoidal functor transports algebra objects and their modules. For the tensor structure on $\mathcal{C}_A$ used here, recall that a right $A$-module $M$ is regarded as the bimodule $M_-$ via~\cite[\S3.3]{DMNO}. Since $\Psi$ is braided, it carries this left action to the corresponding left action on $(\Psi M)_-$. Moreover, as an equivalence of abelian categories, $\Psi$ preserves the coequalizers defining relative tensor products. Hence there are natural isomorphisms $\Psi M\otimes_{\Psi A}\Psi N\cong\Psi(M\otimes_A N)$ compatible with associativity and units. Thus $\mathcal{C}_A\simeq_\otimes\mathcal{D}_{\Psi(A)}$. A braided tensor quasi-inverse of $\Psi$ gives the inverse equivalence.

The second assertion is obtained by transporting module structures along the algebra isomorphism $\varphi:B\to B'$.

<!-- lemma-10: body_md -->
\label{lem:pointed}
Let $\nu,\nu'$ be normalized $3$-cocycles on a finite abelian group $\Gamma$. Then $\mathrm{Vec}_\Gamma^\nu\simeq_\otimes\mathrm{Vec}_\Gamma^{\nu'}$ if and only if there exist $M\in\operatorname{Aut}(\Gamma)$ and a $2$-cochain $\beta$ such that $M^*\nu'=\nu\,\partial\beta$.
<!-- end lemma-10 -->
<!-- lemma-11: body_md -->
\label{lem:invariants}
The functions $q$ and $\operatorname{Alt}$ are multiplicative, with $q_{\nu\nu'}=q_\nu q_{\nu'}$ and $\operatorname{Alt}(\nu\nu')=\operatorname{Alt}(\nu)\operatorname{Alt}(\nu')$. They are trivial on coboundaries, so $q_{\partial\beta}\equiv1$ and $\operatorname{Alt}(\partial\beta)\equiv1$ for every $\beta:V\times V\to\mathbb{C}^\times$. They are also natural under $M\in GL_3(\mathbb{F}_p)$, in the sense that $q_{M^*\nu}(a)=q_\nu(Ma)$ and $\operatorname{Alt}(M^*\nu)(a,b,c)=\operatorname{Alt}(\nu)(Ma,Mb,Mc)$.

For the cocycles $\omega_u$, these functions are $q_{\omega_u}(a)=\zeta^{Q(a)}$ and $\operatorname{Alt}(\omega_u)(a,b,c)=\zeta^{u\det(a,b,c)}$ for every $u\in\mathbb{F}_p$.
<!-- end lemma-11 -->
<!-- proof-16: body_md -->
Multiplicativity follows directly from the definitions, and naturality follows from $M(ta)=tMa$.

For the triviality of $q_\nu$ on a coboundary, expand $(\partial\beta)(a,ta,a)=\beta(ta,a)\beta(a,(t+1)a)\,\beta((t+1)a,a)^{-1}\beta(a,ta)^{-1}$. Since translation by $1$ permutes $\mathbb{F}_p$, telescoping gives $\prod_t\beta(ta,a)/\beta((t+1)a,a)=1$ and $\prod_t\beta(a,(t+1)a)/\beta(a,ta)=1$. Thus $q_{\partial\beta}\equiv1$.

For $\operatorname{Alt}_\nu$, the expression of $\operatorname{Alt}(\nu)(a_1,a_2,a_3)$ is the cocycle-level representative of the standard alternation map $\operatorname{Alt}_3:H^3(V,\mathbb{C}^\times)\longrightarrow\operatorname{Hom}(\Lambda^3V,\mathbb{C}^\times)$~\cite[Lemma~2.12]{DavydovSimmons}. In particular, it depends only on the cohomology class of the cocycle. Thus $\operatorname{Alt}(\partial\beta)\equiv1$ since every coboundary is trivial.

To evaluate $q$ and $\operatorname{Alt}$ on $\omega_u=\kappa\tau_u$, first note that $\det(a,ta,a)=0$, so $q_{\tau_u}(a)=1$.  Moreover, for $r\in\mathbb{F}_p$, $\sum_{t\in\mathbb{F}_p}C_p(tr,r)=[r]$ since $t\mapsto t+1$ permutes $\mathbb{F}_p$. Hence $q_{\kappa}(a)=\zeta^{\sum_i[a_i]^2}=\zeta^{Q(a)}$, and therefore $q_{\omega_u}(a)=\zeta^{Q(a)}$.

For $\operatorname{Alt}$, symmetry of $C_p$ gives $\operatorname{Alt}(\kappa)=1$. Since $\det$ is alternating,
\begin{equation*}
	\operatorname{Alt}(\tau_u)(a_1,a_2,a_3)=\prod_{\sigma\in S_3}\zeta^{\frac u6\operatorname{sgn}(\sigma)\det(a_{\sigma(1)},a_{\sigma(2)},a_{\sigma(3)})}=\zeta^{u\det(a_1,a_2,a_3)}.
\end{equation*}
Thus, by multiplicativity, $\operatorname{Alt}(\omega_u)(a,b,c)=\zeta^{u\det(a,b,c)}$.

<!-- theorem-2: body_md -->
\label{thm:step2}
Let $u,v\in\mathbb{F}_p^\times$. If $\mathcal{C}_u$ and $\mathcal{C}_v$ are braided equivalent, then $u=\pm v$.
<!-- end theorem-2 -->
<!-- proof-17: body_md -->
Assume that $\Phi:\mathcal{C}_u\longrightarrow\mathcal{C}_v$ is a braided tensor equivalence. By Corollary~\ref{cor:pointedequiv}, $\Phi$ induces a monoidal equivalence $\mathrm{Vec}_V^{\omega_u}\simeq_\otimes\mathrm{Vec}_V^{\omega_v}$. Applying Lemma~\ref{lem:pointed} to this monoidal equivalence gives $M\in GL_3(\mathbb{F}_p)$ and a $2$-cochain $\beta:V\times V\to\mathbb{C}^\times$ such that $M^*\omega_v=\omega_u\,\partial\beta$.

We now apply the two cohomological invariants of Lemma~\ref{lem:invariants} to this relation. Applying $q$ to $M^*\omega_v=\omega_u\,\partial\beta$ gives
\begin{equation*}
	\zeta^{Q(Ma)}=q_{\omega_v}(Ma)=q_{M^*\omega_v}(a)=q_{\omega_u}(a)q_{\partial\beta}(a)=\zeta^{Q(a)}
\end{equation*}
Hence $Q(Ma)=Q(a)$ for all $a\in V$. Polarizing $Q(a)$ gives $B(x,y):=Q(x+y)-Q(x)-Q(y)=2x^{\mathsf T}y$, so $B(Mx,My)=B(x,y)$.  Since $2\in\mathbb{F}_p^\times$, this implies $M^{\mathsf T}M=I$, therefore $\det M=\pm1$.

Applying $\operatorname{Alt}$ to $M^*\omega_v=\omega_u\,\partial\beta$ gives
\begin{equation*}
	\zeta^{v\det(Ma,Mb,Mc)}=\operatorname{Alt}(\omega_v)(Ma,Mb,Mc)=\operatorname{Alt}(M^*\omega_v)(a,b,c)=\operatorname{Alt}(\omega_u)(a,b,c)=\zeta^{u\det(a,b,c)}
\end{equation*}
where $\operatorname{Alt}(\partial\beta)=1$ (Lemma~\ref{lem:invariants}) has been used. Evaluating at $(a,b,c)=(\mathbf e_1,\mathbf e_2,\mathbf e_3)$ yields $v\det M=u$. Since $\det M=\pm1$, we conclude $u=\pm v$.

<!-- remark-4: body_md -->
\label{rem:converse}
Although it is not needed for the obstruction, the converse holds. Let $I:V\to V, I(x)=-x$ and define $\epsilon(r)=0$ for $r=0$ and $\epsilon(r)=1$ otherwise, and $\beta_I(a,b):=\zeta^{\sum_i[a_i]\epsilon(b_i)}$. Since $[-r]=-[r]+p\epsilon(r)$, the definition of the carry gives
\begin{equation*}
	C_p(-b,-c)=-C_p(b,c)+\epsilon(b)+\epsilon(c)-\epsilon(b+c).
\end{equation*}
A direct substitution then yields $I^*\kappa=\kappa\,\partial\beta_I$. On the other hand, since the determinant is trilinear, $I^*\tau_u=\tau_{-u}$. Hence, $I^*\omega_u=\omega_{-u}\,\partial\beta_I$.

Reading Lemma~\ref{lem:pointed} in the converse direction, this relation defines a monoidal equivalence $\mathrm{Vec}_V^{\omega_{-u}}\simeq_\otimes\mathrm{Vec}_V^{\omega_u}$ with grading automorphism $I$. Explicitly, $F(X)_y=X_{-y}$ and the tensorator acts by $\beta_I(x,y)^{-1}$ on $X_x\otimes Y_y$. Passing to Drinfel'd centers therefore gives $\mathcal{C}_{-u}\simeq_{\mathrm{br}}\mathcal{C}_u$. Together with Theorem~\ref{thm:step2}, it follows that the categories $\mathcal{C}_u$, $u\in\mathbb{F}_p^\times$, form exactly $(p-1)/2$ braided equivalence classes.

<!-- theorem-3: body_md -->
\label{thm:observable}
For every prime $p\ge5$ and $u\in\mathbb{F}_p^\times$,
\begin{equation}\label{eq:observable-value}
	Z_u(M_p)=p\,G_p(-1)+p(p^2-1)\,\zeta^{-4u^{-2}},\qquad G_p(-1):=\sum_{t\in\mathbb{F}_p}\zeta^{-t^2}.
\end{equation}
Consequently,
\begin{equation}\label{eq:observable-separates}
	Z_u(M_p)=Z_v(M_p)\quad\Longleftrightarrow\quad u=\pm v.
\end{equation}
Thus, for each fixed $p$, this single uncolored partition function separates all $(p-1)/2$ braided equivalence classes of the family.

<!-- proof-20: body_md -->
Set $g=G_p(-1)$, $s=\left(\frac{-1}{p}\right)$ where $\left(\frac{\cdot}{p}\right)$ is the Legendre symbol, and $t=u^2/4\in\mathbb{F}_p^\times$. We use the quadratic Gauss-sum identities $g^2=sp$ and $\sum_{z\in\mathbb{F}_p}\zeta^{h z^2}=\left(\frac{-h}{p}\right)g$ for $h\ne0$. All dot products, cross products and matrices below are over $\mathbb{F}_p$.

Since $\det(a,b,c)=(a\times b)\cdot c$, completing the square in $c$ gives
\[
\sum_{c\in\mathbb{F}_p^3}\zeta^{-Q(c)-u(a\times b)\cdot c}
=g^3\zeta^{tQ(a\times b)}.
\]
Using $Q(a\times b)=Q(a)Q(b)-(a\cdot b)^2$, the remaining sum in $b$ is the quadratic Gauss sum for
\[
K_a=-I+t\bigl(Q(a)I-aa^{\mathsf T}\bigr),
\qquad
\det K_a=-\bigl(1-tQ(a)\bigr)^2.
\]
If $Q(a)\ne t^{-1}$, this matrix is nondegenerate and its determinant has quadratic character $s$, so
\[
\sum_{b\in\mathbb{F}_p^3}\zeta^{b^{\mathsf T}K_ab}=g^3.
\]
The determinant argument includes nonzero isotropic $a$, for which $Q(a)=0$.

If $Q(a)=r:=t^{-1}=4u^{-2}$, then $a\ne0$ and $K_a=-t aa^{\mathsf T}$ has rank one. The linear form $b\mapsto a\cdot b$ is surjective with fibers of size $p^2$, and $t$ is a square, hence
\begin{equation*}
	\sum_{b\in\mathbb{F}_p^3}\zeta^{b^{\mathsf T}K_ab}=p^2\sum_{z\in\mathbb{F}_p}\zeta^{-t z^2}=p^2g.
\end{equation*}
The exceptional level set $Q(a)=r$ has
\begin{equation}\label{eq:observable-shell}
	N_r:=\#\{a\in\mathbb{F}_p^3:Q(a)=r\}=p^2+sp
\end{equation}
points. For completeness, character orthogonality gives
\begin{equation*}
	N_r=\frac1p\sum_{h\in\mathbb{F}_p}\zeta^{-hr}\left(\sum_{z\in\mathbb{F}_p}\zeta^{hz^2}\right)^3=p^2+sp,
\end{equation*}
using that $r$ is a nonzero square and the preceding Gauss-sum identities.

As $\sum_a\zeta^{-Q(a)}=g^3$, splitting~\eqref{eq:observable-finite-sum}  into the cases $Q(a)\neq r$ and $Q(a)=r$ gives
\begin{align*}
	Z_u(M_p)&=p^{-3}g^3\left[g^3\bigl(g^3-N_r\zeta^{-r}\bigr)+p^2gN_r\zeta^{-r}\right]\\
	&=p^{-3}g^9+p^{-3}g^4(p^2-g^2)N_r\zeta^{-r}\\
	&=p g+p(p^2-1)\zeta^{-4u^{-2}},
\end{align*}
where $g^2=sp$ and $N_r=p(p+s)$ were used in the last line. Since $p(p^2-1)\ne0$ and $\zeta$ is primitive, $Z_u(M_p)=Z_v(M_p)$ is equivalent to $u^{-2}=v^{-2}$, hence to $u=\pm v$. Theorem~\ref{thm:step2} and Remark~\ref{rem:converse} identify these pairs with exactly the braided equivalence classes of the family.
