<!-- theorem-1: body_md -->
\label{thm:main}
For any two finite-dimensional quantum channels $\mathcal N,\mathcal M$ with the same input and output spaces, the following equality holds:
\begin{equation}\label{eq:main}
 \lim_{\alpha\to1}D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)
       =D^{\mathrm{reg}}(\mathcal N\|\mathcal M).
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:filter}
Let $\mathcal N,\mathcal M:\mathcal L(A)\to\mathcal L(B)$ be quantum channels, let $\lambda\ge1$, and fix a Stinespring isometry $V:A\to B\otimes E$ of $\mathcal N$. There is a linear operator\footnote{The operator $W_\lambda$ need not be an isometry or a contraction; $\Phi_{W_\lambda}$ need not preserve or decrease trace. A contraction is a linear operator $T$ with $\left\lVertT\right\rVert_\infty\le1$.} $W_\lambda:A\to B\otimes E$ such that
\begin{equation}\label{eq:filter}
 \Phi_{W_\lambda}\le_{\mathrm{CP}}\lambda\mathcal M,\qquad
 \left\lVertV-W_\lambda\right\rVert_\infty\le\sqrt{E_\lambda(\mathcal N\|\mathcal M)}.
\end{equation}
Since $\left\lVertV\right\rVert_\infty=1$, the triangle inequality also gives
$\left\lVertW_\lambda\right\rVert_\infty\le1+\sqrt{E_\lambda(\mathcal N\|\mathcal M)}$.
The same fixed isometry $V$ and environment $E$ may be used for every $\lambda$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Gour's construction \cite[proof of Theorem~2, Eqs.~(44)--(49)]{Gour} applies for every $\lambda\ge1$: take $\gamma=\lambda$ and $s=E_\lambda(\mathcal N\|\mathcal M)$ in those steps.\footnote{The CP Radon--Nikodym theorem assumes a minimal dilation. Gour's direct-sum dilation may not be minimal; restrict it to its minimal environmental subspace and extend the resulting effect by zero.} The construction gives a Stinespring isometry $\widetilde V:A\to B\otimes\widetilde E$ of $\mathcal N$ and operators $X,Y$ on the same space with
\[
 \widetilde V=X+Y,\qquad
 \Phi_X\le_{\mathrm{CP}}\lambda\mathcal M,\qquad
 \left\lVertY\right\rVert_\infty\le\sqrt{E_\lambda(\mathcal N\|\mathcal M)}.
\]
Using $|\Omega_A\rangle$ from \eqref{eq:choi}, the vectors $(I_{A'}\otimes\widetilde V)|\Omega_A\rangle$ and $(I_{A'}\otimes V)|\Omega_A\rangle$ both purify $J_\mathcal N$. Purification uniqueness therefore gives a partial isometry $U:\widetilde E\to E$ such that
\[
 (I_{A'B}\otimes U)(I_{A'}\otimes\widetilde V)|\Omega_A\rangle
 =(I_{A'}\otimes V)|\Omega_A\rangle.
\]
The map $L\mapsto(I_{A'}\otimes L)|\Omega_A\rangle$ is injective, so $(I_B\otimes U)\widetilde V=V$. Set $W_\lambda=(I_B\otimes U)X$. Since $U^\dagger U\le I_{\widetilde E}$,
\begin{align*}
 \Phi_X-\Phi_{W_\lambda}
 &=\Phi_{(I_B\otimes(I_{\widetilde E}-U^\dagger U)^{1/2})X}\ge_{\mathrm{CP}}0,\\
 \Phi_{W_\lambda}&\le_{\mathrm{CP}}\Phi_X\le_{\mathrm{CP}}\lambda\mathcal M,
 &\left\lVertV-W_\lambda\right\rVert_\infty
 &=\left\lVert(I_B\otimes U)Y\right\rVert_\infty
 \le\left\lVertY\right\rVert_\infty\le\sqrt{E_\lambda(\mathcal N\|\mathcal M)}.
\end{align*}
The transfer may depend on $\lambda$, but $V$ and $E$ do not.

<!-- lemma-2: body_md -->
\label{lem:decomposition}
Let $\mathcal N,\mathcal M:\mathcal L(A)\to\mathcal L(B)$ be quantum channels, let $n,q$ be positive integers, and let $V:A^{\otimes n}\to B^{\otimes n}\otimes E$ be a Stinespring isometry of $\mathcal N^{\otimes n}$. Suppose there exist linear operators $U_i:A^{\otimes n}\to B^{\otimes n}\otimes E$, $i=1,\ldots,q$, and constants $b_i\ge0$, $\lambda_i>0$, such that
\[
 V=\sum_{i=1}^q U_i,\qquad
 \left\lVertU_i\right\rVert_\infty\le b_i,\qquad
 \Phi_{U_i}\le_{\mathrm{CP}}\lambda_i\mathcal M^{\otimes n}.
\]
The operators $U_i$ need not be isometries or contractions. For every $\alpha>1$, with $s=(\alpha-1)/\alpha$,
\begin{equation}\label{eq:decomposition}
 D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)
 \le\frac{2}{ns}\log\left(\sum_{i=1}^q
               b_i^{1-s}\lambda_i^{s/2}\right).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
For a positive operator $\rho\ge0$, not necessarily normalized, and a state $\sigma$, let
\begin{equation}\label{eq:Q}
 Q_\alpha(\rho\|\sigma)
 =\operatorname{Tr}\left[\left(\sigma^{-s/2}\rho\sigma^{-s/2}\right)^\alpha\right],
 \qquad s=\frac{\alpha-1}{\alpha},
\end{equation}
with the usual support convention and $Q_\alpha(0\|\sigma)=0$. Homogeneity and the normalized-state bound $D_\alpha\le D_{\max}$ imply, for $\rho\ne0$,
\begin{equation}\label{eq:Qbound}
 \rho\le c\sigma\quad\Longrightarrow\quad
 Q_\alpha(\rho\|\sigma)
 \le(\operatorname{Tr}\rho)^\alpha\left(\frac{c}{\operatorname{Tr}\rho}\right)^{\alpha-1}
 =c^{\alpha-1}\operatorname{Tr}\rho.
\end{equation}

Fix a positive integer $m$ and a pure input $\psi$ on $R\otimes A^{\otimes nm}$, with any finite reference $R$, and write
\[
 \rho=(\operatorname{id}_R\otimes\mathcal N^{\otimes nm})(\psi),\qquad
 \sigma=(\operatorname{id}_R\otimes\mathcal M^{\otimes nm})(\psi).
\]
Expand $V^{\otimes m}=\sum_{\mathbf i}U_{\mathbf i}$, where
$\mathbf i=(i_1,\ldots,i_m)\in\{1,\ldots,q\}^m$ and
$U_{\mathbf i}=U_{i_1}\otimes\cdots\otimes U_{i_m}$.
Reorder the output factors as $B^{\otimes nm}\otimes E^{\otimes m}$.
CP order is preserved under tensor products, and the operator norm is multiplicative. Therefore
\begin{equation}\label{eq:wordbounds}
 \rho_{\mathbf i}:=(\operatorname{id}_R\otimes\Phi_{U_{\mathbf i}})(\psi)
 \le\left(\prod_{j=1}^m\lambda_{i_j}\right)\sigma,
 \qquad
 \operatorname{Tr} \rho_{\mathbf i}\le\prod_{j=1}^m b_{i_j}^2.
\end{equation}
For the trace bound, $\operatorname{Tr}\rho_{\mathbf i}=\left\lVert(I_R\otimes U_{\mathbf i})|\psi\rangle\right\rVert^2\le\left\lVertU_{\mathbf i}\right\rVert_\infty^2\le\prod_j b_{i_j}^2$.

Choose common orthonormal bases $\{|a\rangle\}$ of $R\otimes B^{\otimes nm}$ and $\{|e\rangle\}$ of $E^{\otimes m}$. Let $C_{\mathbf i}:E^{\otimes m}\to R\otimes B^{\otimes nm}$ be the coefficient matrix of $|v_{\mathbf i}\rangle=(I_R\otimes U_{\mathbf i})|\psi\rangle$:
\[
 |v_{\mathbf i}\rangle=\sum_{a,e}c^{(\mathbf i)}_{ae}|a\rangle\otimes|e\rangle,
 \qquad C_{\mathbf i}=\sum_{a,e}c^{(\mathbf i)}_{ae}|a\rangle\!\langle e|.
\]
Taking partial traces of the component and full output vectors gives
\[
 \rho_{\mathbf i}=C_{\mathbf i}C_{\mathbf i}^\dagger,
 \qquad \rho=\left(\sum_{\mathbf i}C_{\mathbf i}\right)
       \left(\sum_{\mathbf i}C_{\mathbf i}\right)^\dagger.
\]
Thus the sum retains all cross terms. Since
\(
 \left\lVert\sigma^{-s/2}C_{\mathbf i}\right\rVert_{2\alpha}
 =Q_\alpha(\rho_{\mathbf i}\|\sigma)^{1/(2\alpha)},
\)
equations \eqref{eq:Qbound}--\eqref{eq:wordbounds} and the Schatten triangle inequality yield
\begin{equation}\label{eq:triangle}
 Q_\alpha(\rho\|\sigma)^{1/(2\alpha)}
 =\left\lVert\sum_{\mathbf i}\sigma^{-s/2}C_{\mathbf i}\right\rVert_{2\alpha}
 \le\sum_{\mathbf i}\left\lVert\sigma^{-s/2}C_{\mathbf i}\right\rVert_{2\alpha}
 \le\sum_{\mathbf i}\prod_{j=1}^m b_{i_j}^{1-s}\lambda_{i_j}^{s/2}
 =\left(\sum_{i=1}^q b_i^{1-s}\lambda_i^{s/2}\right)^m.
\end{equation}
The last equality factors the sum of the product bounds over $\mathbf i=(i_1,\ldots,i_m)$.
All inverse powers act on $\operatorname{supp}\sigma$, which contains the ranges of the coefficient matrices by \eqref{eq:wordbounds}.

Taking logarithms and optimizing the input gives
\[
 \frac1{nm}D_\alpha(\mathcal N^{\otimes nm}\|\mathcal M^{\otimes nm})
 \le\frac{2}{ns}\log\left(\sum_{i=1}^q b_i^{1-s}\lambda_i^{s/2}\right).
\]
Taking $m\to\infty$ at fixed $n$ and $\alpha$ gives \eqref{eq:decomposition}: the subsequence of block lengths $nm$ has the same regularized limit as in \eqref{eq:regularized}.

<!-- lemma-3: body_md -->
\label{lem:testing}
Let $\mathcal N,\mathcal M:\mathcal L(A)\to\mathcal L(B)$ be quantum channels with $D_{\max}(\mathcal N\|\mathcal M)<\infty$. For every positive integer $n$ and every $r>0$,
\begin{equation}\label{eq:weak}
 E_{2^{nr}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})
 \le\frac{D^{\mathrm{reg}}(\mathcal N\|\mathcal M)}r+\frac1{nr}.
\end{equation}
For every $\alpha>1$, every $\ell>D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)$, and every positive integer $n$,
\begin{equation}\label{eq:high}
 E_{2^{n\ell}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})
 \le2^{-n(\alpha-1)[\ell-D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)]}.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Tests with nonpositive objective satisfy both upper bounds automatically. Fix an input and test with $p-2^{nr}\beta>0$, and let $\rho,\sigma$ be its two output states. Since $p>0$ and $p\le2^{n d_\infty}\beta$ by \eqref{eq:cap}, we also have $\beta>0$.

Write $h_2(p)=-p\log p-(1-p)\log(1-p)$ for binary entropy, with $0\log0=0$. Data processing under the binary measurement gives
\begin{align*}
 D(\rho\|\sigma)
 &\ge p\log\frac p\beta+(1-p)\log\frac{1-p}{1-\beta}\\
 &=-h_2(p)-p\log\beta-(1-p)\log(1-\beta)
 \ge nr\,p-1.
\end{align*}
Here $\beta<p2^{-nr}\le2^{-nr}$, $h_2(p)\le1$, and $-(1-p)\log(1-\beta)\ge0$. On the other hand, \eqref{eq:channel}--\eqref{eq:regularized} give
\[
 D(\rho\|\sigma)\le D(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})\le nD^{\mathrm{reg}}(\mathcal N\|\mathcal M).
\]
Thus $p\le D^{\mathrm{reg}}(\mathcal N\|\mathcal M)/r+1/(nr)$. Since the testing objective is at most $p$, optimizing proves \eqref{eq:weak}.

For a test with $p-2^{n\ell}\beta>0$, we again have $p,\beta>0$ and $\beta<p2^{-n\ell}$. Since $1-\alpha<0$, this implies $p^\alpha\beta^{1-\alpha}\ge p\,2^{n\ell(\alpha-1)}$. Data processing for the R\'enyi divergence, followed by dropping the nonnegative term for the other outcome, gives
\[
 nD_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)
 \geD_\alpha(\rho\|\sigma)
 \ge\frac1{\alpha-1}\log(p^\alpha\beta^{1-\alpha})
 \ge n\ell+\frac1{\alpha-1}\log p.
\]
Rearranging gives $p\le2^{-n(\alpha-1)[\ell-D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)]}$. Using $p-2^{n\ell}\beta\le p$ and optimizing proves~\eqref{eq:high}.

<!-- proof-4: body_md -->
We first establish the implication
\begin{equation}\label{eq:thresholdcriterion}
 \limsup_{n\to\infty}E_{2^{nr}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})<1
 \quad\Longrightarrow\quad d_+\le r\qquad(r\ge0).
\end{equation}
Fix $r\ge0$ satisfying the testing condition and assume, for a contradiction, that $r<d_+$. Let
\[
 b_n=\sqrt{E_{2^{nr}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})},
 \qquad b=\limsup_{n\to\infty}b_n.
\]
The testing condition gives $b<1$. Fix any $t\ge1$. Use $d_++t^{-2}$ as the higher approximation rate and let
\[
 \varepsilon_n=\sqrt{E_{2^{n(d_++t^{-2})}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})}.
\]
For this fixed $t$, \eqref{eq:highvanishing} gives $\varepsilon_n\to0$. Choose a Stinespring isometry $V_n$ of $\mathcal N^{\otimes n}$. Lemma~\ref{lem:filter}, applied twice to this same isometry, supplies operators $X_n,W_n$ with
\begin{align}
 \Phi_{X_n}&\le_{\mathrm{CP}}2^{nr}\mathcal M^{\otimes n},
 &\left\lVertV_n-X_n\right\rVert_\infty&\le b_n,\label{eq:lowapprox}\\
 \Phi_{W_n}&\le_{\mathrm{CP}}2^{n(d_++t^{-2})}\mathcal M^{\otimes n},
 &\left\lVertV_n-W_n\right\rVert_\infty&\le\varepsilon_n.
 \label{eq:highapprox}
\end{align}
Define $Y_n=W_n-X_n$ and $Z_n=V_n-W_n$, so $V_n=X_n+Y_n+Z_n$. The triangle inequality gives
\[
 \left\lVertX_n\right\rVert_\infty\le1+b_n,\qquad
 \left\lVertY_n\right\rVert_\infty\le b_n+\varepsilon_n,\qquad
 \left\lVertZ_n\right\rVert_\infty\le\varepsilon_n.
\]
The inequality $\Phi_{U-W}\le_{\mathrm{CP}}2\Phi_U+2\Phi_W$,\footnote{This follows from $2\Phi_U+2\Phi_W-\Phi_{U-W}=\Phi_{U+W}\ge_{\mathrm{CP}}0$.} together with $r<d_+\le d_\infty$ and $t\ge1$, gives
\begin{equation}\label{eq:threecaps}
 \Phi_{Y_n}\le_{\mathrm{CP}}4\,2^{n(d_++t^{-2})}\mathcal M^{\otimes n},
 \qquad
 \Phi_{Z_n}\le_{\mathrm{CP}}4\,2^{n(d_\infty+1)}\mathcal M^{\otimes n}.
\end{equation}
For the second bound, use $\Phi_{V_n}=\mathcal N^{\otimes n}\le_{\mathrm{CP}}2^{nd_\infty}\mathcal M^{\otimes n}$ and $d_++t^{-2}\le d_\infty+1$.
For integers $n>t$, apply Lemma~\ref{lem:decomposition} with $s=t/n$, equivalently $\alpha=n/(n-t)>1$. Multiplying \eqref{eq:decomposition} by $ns/2=t/2$ and exponentiating gives
\[
 2^{td_\alpha/2}\le(1+b_n)^{1-t/n}\,2^{tr/2}
 +2^{t/n}(b_n+\varepsilon_n)^{1-t/n}\,2^{t(d_++t^{-2})/2}
 +2^{t/n}\varepsilon_n^{1-t/n}\,2^{t(d_\infty+1)/2}.
\]
Using $d_+\le d_\alpha$ and dividing by $2^{td_+/2}$ now gives
\begin{equation}\label{eq:threepiecebound}
 1\le\frac{2^{td_\alpha/2}}{2^{td_+/2}}
 \le(1+b_n)^{1-t/n}\,2^{-t(d_+-r)/2}
 +2^{t/n}(b_n+\varepsilon_n)^{1-t/n}\,2^{1/(2t)}
 +2^{t/n}\varepsilon_n^{1-t/n}\,2^{t(d_\infty+1-d_+)/2}.
\end{equation}
Keep $t$ fixed and let $n\to\infty$. With $\delta_n=t/n$, weighted AM--GM gives $x^{1-\delta_n}\le(1-\delta_n)x+\delta_n\le x+\delta_n$ for every $x\ge0$. Applying this to $1+b_n$, $b_n+\varepsilon_n$, and $\varepsilon_n$ shows that their powered factors have upper limits at most $1+b$, $b$, and $0$, respectively. Since $2^{t/n}\to1$, taking the upper limit in \eqref{eq:threepiecebound} gives
\begin{equation}\label{eq:contradiction}
 1\le(1+b)\,2^{-t(d_+-r)/2}+b\,2^{1/(2t)}
 \qquad(t\ge1).
\end{equation}
Now let $t\to\infty$. Since $r<d_+$, the right-hand side tends to $b<1$, a contradiction. Thus $d_+\le r$. Taking $n\to\infty$ first avoids any need for uniform convergence in $t$.

To complete the proof, we show that $d_+=d$. By \eqref{eq:weak}, for every $r>d$,
\[
 \limsup_{n\to\infty}E_{2^{nr}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})
 \le\frac dr<1.
\]
By \eqref{eq:thresholdcriterion}, $d_+\le r$ for every $r>d$. Letting $r\downarrow d$ and using $d\le d_+$ proves $d_+=d$.

<!-- corollary-1: body_md -->
\label{cor:discrimination}
For every $r>D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$, there is $\kappa_r>0$ such that every $n$-use adaptive protocol whose type-II error is at most $2^{-nr}$ has success probability at most $2^{-n\kappa_r}$:
\begin{equation}\label{eq:SC}
 \beta_n\le2^{-nr}\quad\Longrightarrow\quad p_n\le2^{-n\kappa_r}.
\end{equation}
For $\epsilon\in(0,1)$, let $\beta_{n,\epsilon}$ be the infimum of the type-II errors among $n$-use adaptive protocols with $p_n\ge1-\epsilon$. Then
\begin{equation}\label{eq:app-stein}
 \lim_{n\to\infty}-\frac1n\log\beta_{n,\epsilon}
 =D^{\mathrm{reg}}(\mathcal N\|\mathcal M).
\end{equation}
<!-- end corollary-1 -->
<!-- proof-5: body_md -->
For every $r>D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$, Theorem~\ref{thm:main} guarantees that there exists an $\alpha>1$ with $D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)<r$. Set
$\kappa_r=\frac{\alpha-1}{\alpha}[r-D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)]>0$.
Substituting $\beta_n\le2^{-nr}$ into \eqref{eq:app-testing} proves \eqref{eq:SC}.

For the converse bound in \eqref{eq:app-stein}, \eqref{eq:app-testing} and $p_n\ge1-\epsilon$ give, uniformly over admissible adaptive protocols,
\[
 \beta_n\ge(1-\epsilon)^{\alpha/(\alpha-1)}
                2^{-nD_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)}.
\]
This also bounds the infimum $\beta_{n,\epsilon}$, without requiring an optimal protocol. Therefore
\[
 -\frac1n\log\beta_{n,\epsilon}
 \leD_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)
 +\frac\alpha{n(\alpha-1)}\log\frac1{1-\epsilon}.
\]
First take the upper limit as $n\to\infty$ at fixed $\alpha>1$, then let $\alpha\downarrow1$ using Theorem~\ref{thm:main}.

For achievability at $0<s<D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$, \eqref{eq:regularized} supplies a block size $k$, a pure input on $R\otimes A^{\otimes k}$, and $s'>s$ such that its output states satisfy $D(\rho_k\|\sigma_k)>ks'$. Repeat this input $m$ times and apply state Stein's lemma \cite{HP,ON}: for large $m$, there are tests with acceptance tending to one and type-II error at most $2^{-mks'}$. For $n$ calls, take $m=\lfloor n/k\rfloor$ and ignore the outputs of the remaining calls. Since $mk/n\to1$, eventually $mks'\ge ns$, so these parallel tests satisfy $p_n\to1$ and $\beta_n\le2^{-ns}$. Their reference is $R^{\otimes m}$, whose dimension may grow with $n$. They are also valid adaptive protocols. Letting $s\uparrow D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$ proves achievability; when $D^{\mathrm{reg}}(\mathcal N\|\mathcal M)=0$, nonnegativity of the error exponents suffices.

<!-- corollary-2: body_md -->
\label{cor:subchannel-aep}
Let $\mathcal N,\mathcal M$ be channels. For every fixed $\epsilon\in(0,1)$, smoothing $\mathcal N^{\otimes n}$ over nearby subchannels has asymptotic max-relative entropy $D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$:
\begin{equation}\label{eq:subchannel-aep}
 \lim_{n\to\infty}\frac1n
 \inf_{\substack{\mathcal G_n\ge_{\mathrm{CP}}0,\ \mathcal G_n^*(I)\le I\\
                 \left\lVert\mathcal G_n-\mathcal N^{\otimes n}\right\rVert_\diamond\le\epsilon}}
 D_{\max}(\mathcal G_n\|\mathcal M^{\otimes n})
 =D^{\mathrm{reg}}(\mathcal N\|\mathcal M).
\end{equation}

<!-- corollary-3: body_md -->
\label{cor:subchannel}
For every $r>D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$, there are $c_r>0$ and subchannels
$\mathcal G_n:\mathcal L(A^{\otimes n})\to\mathcal L(B^{\otimes n})$, for $n\ge1$, such that
\begin{equation}\label{eq:dominated}
 \mathcal G_n\le_{\mathrm{CP}}2^{nr}\mathcal M^{\otimes n},\qquad
 \left\lVert\mathcal G_n-\mathcal N^{\otimes n}\right\rVert_\diamond\le4\,2^{-nc_r}.
\end{equation}
<!-- end corollary-3 -->
<!-- proof-6: body_md -->
Choose $\alpha>1$ with $D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)<r$, and let
$c_r=\tfrac12(\alpha-1)[r-D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)]$ and $\eta_n=2^{-nc_r}$.
By Lemma~\ref{lem:testing}~\eqref{eq:high} and Lemma~\ref{lem:filter}, a Stinespring isometry $V_n$ of $\mathcal N^{\otimes n}$ admits $W_n$ in the same environment with
\[
 \Phi_{W_n}\le_{\mathrm{CP}}2^{nr}\mathcal M^{\otimes n},\qquad
 \left\lVertV_n-W_n\right\rVert_\infty\le\eta_n.
\]
Since $V_n$ is an isometry, $\left\lVertW_n\right\rVert_\infty\le1+\eta_n$. Define $W_n'=W_n/(1+\eta_n)$ and $\mathcal G_n=\Phi_{W_n'}$. Then $\left\lVertW_n'\right\rVert_\infty\le1$, so
\[
 \mathcal G_n^*(I_{B^{\otimes n}})=W_n'^\dagger W_n'\le I_{A^{\otimes n}}.
\]
Thus $\mathcal G_n$ is a subchannel. Rescaling also preserves the required domination:
\[
 \mathcal G_n=\frac{\Phi_{W_n}}{(1+\eta_n)^2}
 \le_{\mathrm{CP}}\frac{2^{nr}}{(1+\eta_n)^2}\mathcal M^{\otimes n}
 \le_{\mathrm{CP}}2^{nr}\mathcal M^{\otimes n}.
\]
For the approximation error, the triangle inequality gives
\[
 \left\lVertV_n-W_n'\right\rVert_\infty
 \le\left\lVertV_n-W_n\right\rVert_\infty+\left\lVertW_n-W_n'\right\rVert_\infty
 \le\eta_n+\frac{\eta_n}{1+\eta_n}\left\lVertW_n\right\rVert_\infty
 \le2\eta_n.
\]
For operators $U,V$ with the same environment,
\[
 \Phi_U-\Phi_V
 =\operatorname{Tr}_E\!\left[(U-V)(\cdot)U^\dagger+V(\cdot)(U-V)^\dagger\right].
\]
The trace-norm triangle inequality, also after tensoring with an arbitrary reference, gives the dilation estimate \cite{KSW}
\[
 \left\lVert\Phi_U-\Phi_V\right\rVert_\diamond
 \le(\left\lVertU\right\rVert_\infty+\left\lVertV\right\rVert_\infty)\left\lVertU-V\right\rVert_\infty
\]
with $U=W_n'$ and $V=V_n$. Since $\Phi_{V_n}=\mathcal N^{\otimes n}$, this gives
\[
 \left\lVert\mathcal G_n-\mathcal N^{\otimes n}\right\rVert_\diamond
 \le(\left\lVertW_n'\right\rVert_\infty+\left\lVertV_n\right\rVert_\infty)\left\lVertW_n'-V_n\right\rVert_\infty
 \le(1+1)\,2\eta_n=4\,2^{-nc_r},
\]
proving \eqref{eq:dominated}.
<!-- end proof-6 -->
<!-- proof-7: body_md -->
For the upper bound in \eqref{eq:subchannel-aep}, fix $r>D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$. Corollary~\ref{cor:subchannel} gives feasible maps for all sufficiently large $n$, since $4\,2^{-nc_r}\le\epsilon$, with $D_{\max}(\mathcal G_n\|\mathcal M^{\otimes n})\le nr$. Take the upper limit and then let $r\downarrow D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$.

For the converse, let $(\psi_n,T_n)$ be the parallel input and test from the block construction proving \eqref{eq:app-stein} at $0<s<D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$. The diamond-distance bound limits the change of this test's acceptance probability to $\epsilon$: pair the output difference with $0\le T_n\le I$ and bound it by its trace norm. Thus every feasible smoother and every $\lambda>0$ with $\mathcal G_n\le_{\mathrm{CP}}\lambda\mathcal M^{\otimes n}$ satisfy
\[
 p_n-\epsilon
 \le\operatorname{Tr}[T_n(\operatorname{id}\otimes\mathcal G_n)(\psi_n)]
 \le\lambda\beta_n\le\lambda2^{-ns}.
\]
Since $p_n-\epsilon\to1-\epsilon>0$, its max-relative entropy is at least $ns+\log(p_n-\epsilon)$ for large $n$, and the logarithmic term divided by $n$ vanishes. Taking the lower limit and then $s\uparrow D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$ proves the lower bound. When $D^{\mathrm{reg}}(\mathcal N\|\mathcal M)=0$, use any input state $\psi$ to obtain
\[
 1-\epsilon\le\operatorname{Tr}[(\operatorname{id}\otimes\mathcal G_n)(\psi)]\le\lambda.
\]
Thus $D_{\max}(\mathcal G_n\|\mathcal M^{\otimes n})\ge\log(1-\epsilon)$, whose normalized limit is zero.

\label{cor:sharp-testing}
For every $r\ge0$ with $r\ne D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$,\footnote{No assertion is made at the threshold itself.}
\begin{equation}\label{eq:app-sharp}
 \lim_{n\to\infty}E_{2^{nr}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})
 =\begin{cases}
   1,&0\le r<D^{\mathrm{reg}}(\mathcal N\|\mathcal M),\\
   0,&r>D^{\mathrm{reg}}(\mathcal N\|\mathcal M).
  \end{cases}
\end{equation}
<!-- end corollary-4 -->
<!-- proof-8: body_md -->
For $0\le r<D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$, choose $s$ strictly between them. The parallel block tests constructed for \eqref{eq:app-stein}, as in \cite[Lemma 12]{Gour}, satisfy $p_n\to1$ and $\beta_n\le2^{-ns}$, so
\[
 1\ge E_{2^{nr}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})
 \ge p_n-2^{-n(s-r)}\longrightarrow1.
\]
For $r>D^{\mathrm{reg}}(\mathcal N\|\mathcal M)$, Theorem~\ref{thm:main} supplies $\alpha>1$ with $D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)<r$. Lemma~\ref{lem:testing}~\eqref{eq:high} then gives
\[
 0\le E_{2^{nr}}(\mathcal N^{\otimes n}\|\mathcal M^{\otimes n})
 \le2^{-n(\alpha-1)[r-D_\alpha^{\mathrm{reg}}(\mathcal N\|\mathcal M)]}\longrightarrow0.
\]
