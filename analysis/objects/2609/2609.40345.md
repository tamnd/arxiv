<!-- definition-1: body_md -->
Let $H:\{+1,-1\}^N \to \mathbb{R}$ be a classical Hamiltonian that maps a spin configuration to a corresponding energy. The corresponding Gibbs measure at inverse temperature $\beta$ is then
    \begin{equation}
        \label{eq:gibbs}
        \mu_{\beta,H}(\mathbf{z}) = \frac{1}{Z_{\beta,H}} e^{-\beta H(\mathbf{z})}, \quad \textrm{where} \quad Z_{\beta,H} = \sum_{\mathbf{z}} e^{-\beta H(\mathbf{z})}.
    \end{equation}
    The \emph{$(H, \beta,\varepsilon)$-Gibbs sampling problem} is to sample from a distribution over $\{+1,-1\}^N$ whose total variation distance from $\mu_{\beta,H}$ is at most $\varepsilon$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:QDP}
    Let $B \in \mathbb{F}_2^{M\times N}$ and let $C^\perp$ be the classical error correcting code defined by taking $B^T$ as its parity check matrix:
    \begin{equation}
        \label{eq:Cperp}
        C^\perp = \{ \mathbf{d} \in \mathbb{F}_2^M : B^T \mathbf{d} = \mathbf{0}\}.
    \end{equation}
    Given $\gamma \in [0,1/2]$ let
    \begin{equation}
        \ket{\psi_{0,\gamma}} = \sqrt{1-\gamma} \ket{0} + \sqrt{\gamma} \ket{1} \quad \textrm{and} \quad \ket{\psi_{1,\gamma}} = \sqrt{1-\gamma} \ket{1} + \sqrt{\gamma} \ket{0}.
    \end{equation}
    Given $\mathbf{d} = (d_1,\ldots,d_M) \in \mathbb{F}_2^M$ let
    \begin{equation}
        \label{eq:productform}
        \ket{\psi_{\mathbf{d},\gamma}} = \bigotimes_{j=1}^M \ket{\psi_{d_j,\gamma}}.
    \end{equation}
    The \emph{$(B, \gamma,\delta)$-quantum decoding problem} at error rate $\gamma$ is, for uniformly random $\mathbf{d}\in C^\perp$, to identify $\mathbf{d}$ from $\ket{\psi_{{\mathbf{d}},\gamma}}$ with probability at least $1 - \delta$.
<!-- end definition-2 -->
<!-- remark-1: body_md -->
The problem in \cref{def:QDP} is a special case of the more general quantum decoding problem formulated in \cite{chailloux2023quantum} and explored further in \cite{SMR26}. Here, motivated by Gibbs sampling, we restrict our attention to input states with the tensor product form \cref{eq:productform}.
    If we take a classical algorithm for decoding LDPC codes and implement it as a reversible circuit, then upon the input state \eqref{eq:productform},  the algorithm only ``sees'' a classical error model where each bit has been independently flipped with probability $\gamma$; such classical decoders were the starting point for DQI in~\cite{jordan2024optimization}. However, in this paper we will use an intrinsically quantum decoding method from \cite{chailloux2023quantum} that takes advantage of the coherence of these bit flip errors.
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{thm:gamma_vs_beta}
    Given any $B \in \mathbb{F}_2^{M \times N}$, and uniformly random $J_1,\ldots,J_M \sim_\textnormal{i.i.d.} \{+1,-1\}$. Let $H_{B,J}:\{+1,-1\}^N \to \mathbb{Z}$ be the classical Hamiltonian 
    \begin{equation}
        \label{eq:HBJ}
        H_{B,J}(\mathbf{z}) = -\sum_{i=1}^M J_i \prod_{j:B_{ij} = 1} z_j.
    \end{equation}
Fix any $\delta \in [0,1]$ and $\gamma \in [0,1/2]$, and set
    \begin{equation}
        \label{eq:betamax}
        \beta = \operatorname{sech}^{-1} \left( 1- 2 \gamma \right).
    \end{equation}
    Given a subroutine which efficiently solves the $(B, \gamma,\delta)$-quantum decoding problem, then DQI can efficiently solve the $(H_{B, J}, \beta, \varepsilon)$-Gibbs sampling problem for any $\varepsilon\ge \sqrt{\delta}$, with probability at least $1 - \sqrt{\delta}/\varepsilon$ over the $J_i$'s.
<!-- end theorem-1 -->
<!-- definition-3: body_md -->
\label{def:erdos_renyi}
    Given positive integers $N \ge k \ge 2$, and any $D >0$, we define $\mathbb{G}_\textnormal{ER}(N,k,D)$ be the following ensemble of random weighted $k$-uniform hypergraphs with average degree (approximately) $D$. Let the vertex set be $V=\{1,\ldots,N\}$. Let $M=\lfloor ND/k\rfloor$, and the hyperedge multiset $E=(e_1,\ldots,e_M)$ is obtained by drawing each $e_i$ independently and uniformly from the collection $\binom{V}{k}$ of all $k$-element subsets of $V$ (i.e., with replacement). 
    The weights $J=(J_1,\ldots,J_M)\in\{+1,-1\}^M$ are i.i.d.\ uniform Rademacher and independent of $E$. 
    Each hypergraph $G=(V,E)$ is equivalently described by an incidence matrix $B\in\mathbb{F}_2^{M\times N}$ whose row $i$ has ones exactly at the $k$ positions in $e_i$, so $\mathbb{G}_\textnormal{ER}(N,k,D)$ also defines a distribution on matrices over $\mathbb{F}_2$ in which each row has exactly $k$ non-zero entries placed uniformly at random.
    We will often write $H=(B,J)\sim\mathbb{G}_\textnormal{ER}(N,k,D)$, where the relationship between $H$ and $(B,J)$ is given in Eq.~\eqref{eq:HBJ}.
<!-- end definition-3 -->
<!-- theorem-2: body_md -->
\label{thm:shatter-hardness} 
	For any fixed $\varepsilon >0$, there exists a cutoff $k_0(\varepsilon)$ such that the following holds for all fixed $k\ge k_0(\varepsilon)$ and $D\ge k$ as $N\to\infty$.
	Let $H=H_{B,J}$ with $(B,J)\sim\mathbb{G}_\textnormal{ER}(N,k,D)$ as in \cref{def:erdos_renyi}. For any $\beta$ satisfying
	\begin{equation} \label{eq:beta-range}
	(1+\varepsilon)\sqrt{\frac{2\ln k }{D} } \le \beta  \le \sqrt{\frac{k}{2D}},
	\end{equation}
	the Gibbs measure $\mu_{\beta, H}$ shatters into exponentially many well-separated clusters with high probability.
	In this interval of $\beta$, no stable algorithm can sample from $\mu_{\beta,H}$ with vanishing $o_N(1)$ expected normalized Wasserstein distance, and Glauber dynamics has $e^{\Omega(N)}$ mixing time with high probability.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:gamma_USD} 
    Fix any $k\ge 3$ and $D\ge k$.
    Let
    \begin{equation}
        \gamma_\textnormal{USD}(k,D) = \frac{1}{2} - \frac{1}{2} \sqrt{1-\Big( \frac{K_*(k)}{D} \Big)^2},
    \end{equation}
    where we define $K_*(k)$ variationally via
    \begin{equation}
        \Phi_{d,k}(x) = e^{-d x^{k-1}} + d x^{k-1} - \frac{d(k-1)}{k}x^k - \frac{d}{k} 
        \quad \text{and} \quad 
        K_*(k) := \sup\Bigl\{ d>0 : \max_{x\in[0,1]}\Phi_{d,k}(x)=1-\frac{d}{k} 
                                                  
        \Bigr\}.
        \label{eq:admissible}
    \end{equation}
    Let  $(B, J) \sim\mathbb{G}_{\mathrm{ER}}(N,k,D)$, where $B\in \mathbb{F}_2^{M\times N}$ is the incidence matrix of a random Erd\H{o}s-R\'enyi hypergraph.
    For every fixed $\delta \in (0,1)$ and $\gamma \in [0, \gamma_\textnormal{USD})$,
    the quantum USD decoder of \cite{chailloux2023quantum} can efficiently solve the $(B, \gamma,\delta)$-quantum decoding problem with high probability $(1-o_{N}(1))$ over the choice of $B$.
    With \cref{thm:gamma_vs_beta}, this implies that for any $\varepsilon >0$ and
    \begin{equation} \label{eq:beta-ALG}
        \beta <  \beta_{\textnormal{ALG}}(k, D) := \tanh^{-1} \Big(\frac{K_*(k)}{D} \Big),
    \end{equation}
    and sufficiently large $N$, DQI can efficiently solve the $(H_{B, J}, \beta,\varepsilon)$-Gibbs sampling problem with probability at least $1-\sqrt{\delta}/{\varepsilon} - o_N(1)$ over the choice of $(B,J)$.
<!-- end theorem-3 -->
<!-- remark-2: body_md -->
$K_*(k)$ is in fact quite close to $k$. We show later in \cref{cor:dk-asymptotics} that
    \[
        \frac{K_*(k)}{k}\longrightarrow 1
        \qquad\text{as }k\to\infty.
    \]
Consequently, for $D=\alpha k$ with fixed $\alpha>1$,  $\beta_\textnormal{ALG}(k,\alpha k)\to\tanh^{-1}(1/\alpha)$ as $k\to\infty$.

<!-- remark-3: body_md -->
\label{rmk:tvd-vs-wasserstein}
Throughout the paper we use two distance measures on probability distributions over $\boldsymbol{\Sigma}_N=\{+1,-1\}^N$. Our positive results for DQI (\cref{thm:gamma_vs_beta,thm:gamma_USD}) are stated in the stronger total variation distance $d_\textnormal{TV}$, while our hardness results against stable algorithms (\cref{thm:shatter-hardness}) are stated in the weaker normalized Wasserstein distance $W_1$ (see \cref{eq:W1-def} for a formal definition). The two are related by
\[
W_1(\mu,\nu)\le 2\,d_\textnormal{TV}(\mu,\nu)
\qquad\text{for all }\mu,\nu\text{ on }\boldsymbol{\Sigma}_N,
\]
which follows by taking the optimal TV coupling and using $\|\boldsymbol{X}-\boldsymbol{Y}\|_1\le 2N$ for any $\boldsymbol{X},\boldsymbol{Y}\in\boldsymbol{\Sigma}_N$. Hence, an $\varepsilon$-TVD sampler is also a $2\varepsilon$-Wasserstein sampler, and hardness in $W_1$ also implies hardness in total variation.
<!-- end remark-3 -->
<!-- theorem-4: body_md -->
\label{thm:distributionalcloseness}
    Fix $B \in \mathbb{F}_2^{M \times N}$ and $\beta > 0$, with $\gamma$ related to $\beta$ via \cref{eq:gamma_from_beta}.
    Let $U_{\mathrm{dec}}$ be any unitary that solves the $(B, \gamma,\delta)$-quantum decoding problem of \cref{def:QDP}.
    Let $\mathcal{D}_\mathrm{target}(B,J,\beta)$ denote the distribution over $\mathbf{y}\in\mathbb{F}_2^M$ obtained by measuring the ideal state $\ket{\Phi_{\mathrm{target}}}$ of \cref{eq:phitarget} in the computational basis (which is supported on the code $C=\{B\mathbf{x}:\mathbf{x}\in\mathbb{F}_2^N\}$), and let $\mathcal{D}_\mathrm{actual}(B,J,\beta)$ denote the output distribution of the above procedure when $U_{\mathrm{dec}}$ replaces the idealized decoder $\tilde{U}_{\mathrm{dec}}$.
    Then, for uniformly random $J \in \{+1,-1\}^M$,
    \begin{equation}
        \label{eq:tvd}
        \mathbb{E}_{J} \Big[d_\textnormal{TV}\bigl(\mathcal{D}_\mathrm{actual}(B,J,\beta),\, \mathcal{D}_\mathrm{target}(B,J,\beta)\bigr)\Big] \leq \sqrt{\delta},
    \end{equation}
    where $d_\textnormal{TV}$ denotes the total variation distance.
<!-- end theorem-4 -->
<!-- proof-1: body_md -->
Any concrete decoder, such as the USD decoder analyzed in \cref{sec:USD} below, will have a small but nonzero failure rate. In the context of quantum decoding algorithms, the effect of decoding errors is subtle to analyze because the error terms from different codewords in the superposition over $C^\perp$ can add constructively. Consequently, the naive expectation that a decoder failure rate $\delta$ implies that the output of DQI deviates from the target distribution by at most $O(\delta)$ is not always true. (See \cite{chailloux2023quantum} for an explicit counterexample.) Nevertheless, for average-case coefficients $J$, \cref{thm:distributionalcloseness} tells us that a small decoder failure rate induces only a small deviation from the desired output distribution. Interestingly, this suggests that for the decoding errors to be tolerable it is necessary that the coefficients $J_1,\ldots,J_M \in \{+1,-1\}$ in the Hamiltonian be chosen randomly rather than adversarially. We obtain \cref{thm:gamma_vs_beta} as a corollary of \cref{thm:distributionalcloseness} by the following argument 

Fix an arbitrary spin configuration $\mathbf{z}_0\in\{+1,-1\}^N$, and define a stochastic
post-processing channel $K_B$ as follows. On input $\mathbf{y}\in C$, choose $\mathbf{x}$ uniformly from
\[
    A_\mathbf{y} := \{\mathbf{x}\in\mathbb{F}_2^N:B\mathbf{x}=\mathbf{y}\}
\]
and output $\mathbf{z}=(-1)^{\mathbf{x}}$.  On input $\mathbf{y}\notin C$, output $\mathbf{z}_0$.

Let
\[
    \nu^{\textnormal{DQI}}_{B,J,\beta}
    :=
    K_B\mathcal{D}_{\mathrm{actual}}(B,J,\beta).
\]
Because $H_{B,J}((-1)^{\mathbf{x}})$ depends on $\mathbf{x}$ only through $B\mathbf{x}$, the Gibbs measure is uniform on the set $A_\mathbf{y}$.  Hence
\[
    K_B\mathcal{D}_{\mathrm{target}}(B,J,\beta)
    =
    \mu_{\beta,H_{B,J}}.
\]
Total variation distance contracts under stochastic post-processing, so
\begin{align}
    \mathbb{E}_J\!\left[
        d_\textnormal{TV}\!\left(
            \nu^{\textnormal{DQI}}_{B,J,\beta},~
            \mu_{\beta,H_{B,J}}
        \right)
    \right]
    &\leq
    \mathbb{E}_J\!\left[
        d_\textnormal{TV}\!\left(
            \mathcal{D}_{\mathrm{actual}}(B,J,\beta),
            \mathcal{D}_{\mathrm{target}}(B,J,\beta)
        \right)
    \right] \nonumber \\
    &\leq \sqrt{\delta}.
\end{align}
Therefore, by Markov's inequality, for every $\varepsilon>0$,
\[
    \mathbb{P}_J\!\left[
        d_\textnormal{TV}\!\left(
            \nu^{\textnormal{DQI}}_{B,J,\beta},~
            \mu_{\beta,H_{B,J}}
        \right)>\varepsilon
    \right]
    \leq
    \frac{\sqrt{\delta}}{\varepsilon}.
    \qedhere
\]

The upper bound $K_*(k)\le k$ is immediate. Indeed, if $K_*(k)>(1+\varepsilon) k$ for any $\varepsilon>0$, then one can choose $D=(1+\varepsilon/2)k < K_*(k)$, but the matrix $B$ cannot have full row rank when $M=ND/k = (1+\varepsilon/2)N > N$, yielding a contradiction.

It remains to prove the matching lower bound. 
We note from \cite[Fact 2.8]{coja2024k} that the maximizer of $\Phi_{D,k}(x)$ is unique and located at $x=0$ for $D< K_*(k)$.
Fix $\delta\in(0,1)$ and set $D=(1-\delta)k$ large. Define
\begin{equation}
    G_{D,k}(x):=\Phi_{D,k}(x)-\Phi_{D,k}(0) = e^{-D x^{k-1}} + D x^{k-1} - \frac{D(k-1)}{k}x^k -1 .
\end{equation}
We must show that $G_{D,k}(x)\le 0$ for all $x\in[0,1]$, with strict inequality for $x>0$, once $k$ is sufficiently large.\\

\noindent
To proceed, let us split $x \in (0,1]$ into two regions.\\

\noindent
\textbf{Region 1: $x\in (0,1-k^{-2/3}]$.} Using the fact that $e^{-u} \le 1-u+u^2/2$ for $u\ge0$, we have
\begin{equation}
    G_{D,k}(x) \le \frac{D^2}{2} x^{2k-2} - \frac{D(k-1)}{k}x^k = D x^k \Bigl(\frac{D}{2} x^{k-2} - \frac{k-1}{k}\Bigr).
\end{equation}
In this region, $x^{k-2} \le e^{-k^{1/3}/2}$ for sufficiently large $k$. With $D=(1-\delta)k$, we have $\frac{D}{2}x^{k-2}\le (1-\delta)\frac{k}{2}e^{-k^{1/3}/2}\xrightarrow{k\to\infty}0$. Therefore, $G_{D,k}(x) < 0$ throughout this region for sufficiently large $k$. \\

\noindent
\textbf{Region 2: $x\in [1-k^{-2/3},1]$.} Let us parameterize $x=1-y/k$ with $y\in[0,k^{1/3}]$. Then $x^{k-1}=e^{-y}(1+O(y^2/k))$. With $D=(1-\delta) k$, we have for sufficiently large $k$ that
\begin{equation}
    \label{eq:exp-d-x-k-1-bound}
    e^{-Dx^{k-1}} = \exp\Bigl[-(1-\delta) k e^{-y} \times (1+o(1) )\Bigr].
\end{equation}
Furthermore,
\begin{equation}
    D x^{k-1} - \frac{D(k-1)}{k}x^k = D x^{k-1} \Bigl[1- \Big(1-\frac{1}{k}\Big)\Big(1-\frac{y}{k}\Big)\Bigr] 
    = (1-\delta)(1+y) e^{-y} + O(e^{-y} (y+y^3)/k).
\end{equation}
Then
\begin{equation}
    \label{eq:Gdk-approx}
    G_{D,k}(x) = e^{-Dx^{k-1}} - 1 + (1-\delta)(1+y) e^{-y}  (1+ o(1)).
\end{equation}
To show $G_{D,k}(x) < 0$ in this region, we consider two further subcases:
\begin{itemize}
    \item When $y \le \frac12 \ln k$, we estimate the term \eqref{eq:exp-d-x-k-1-bound} via $k e^{-y} \ge \sqrt{k}$. Furthermore, note $(1+y)e^{-y} \le 1$ for all $y\ge 0$. Thus 
    \[
    G_{D,k}(x) \le -1 + (1-\delta) + o(1) = -\delta + o(1) < 0
    \]
    for sufficiently large $k$.
    \item When $\frac12 \ln k \le y \le k^{1/3}$, we have $ke^{-k^{1/3}} \le k e^{-y} \le \sqrt{k}$. Furthermore, 
    \[
    (1+y)e^{-y} \le (1+k^{1/3})/\sqrt{k} = o(1).
    \]
    If $ke^{-y}>\Omega(1)$ when $k$ is large, then recalling \eqref{eq:exp-d-x-k-1-bound} gives 
    \[
    e^{-Dx^{k-1}}-1 < -\Omega(1) 
    \]
    and plugging into \eqref{eq:Gdk-approx} gives $G_{D,k}(x) < 0$. If $ke^{-y}\leq o(1)$, then 
    \[
    e^{-Dx^{k-1}} - 1 = -Dx^{k-1} (1+o(1))=-(1-\delta)ke^{-y} (1+o(1)).
    \]
    Since $k \gg 1+y$, grouping terms in \eqref{eq:Gdk-approx} again implies that $G_{D,k}(x)<0$.
\end{itemize}

We have shown that $G_{D,k}(x)<0$ for all $x\in (0,1]$ for sufficiently large $k$, if $D=(1-\delta)k$ for any fixed $\delta\in (0,1)$. Consequently,
\begin{equation}
    \max_{x\in[0,1]}\Phi_{D,k}(x)=\Phi_{D,k}(0)=1-\frac{D}{k},
\end{equation}
so $D=(1-\delta)k$ satisfies the condition \eqref{eq:admissible} of $K_*(k)$ for all large $k$. Since $\delta>0$ was arbitrary, this shows
\begin{equation}
    \liminf_{k\to\infty}\frac{K_*(k)}{k}\ge 1.
\end{equation}
Combining with the upper bound $K_*(k)\le k$ completes the proof.

Let $C^\perp$ be a code whose parity-check matrix's transpose $B$ is drawn from the Erd\H{o}s-R\'enyi hypergraph ensemble $\mathbb{G}_\textnormal{ER}(N,k,D)$, so that the codewords of $C^\perp$ have length $M=\lfloor ND/k\rfloor$. Fix an erasure rate $\hat e$ with $\hat e D<K_*(k)$, and recall from the discussion above that decoding succeeds precisely when the row-submatrix $B_S$ indexed by the erased set $S$, with $|S|\sim\text{Bin}(M,\hat e)$, has full row rank. We now reduce this to \cref{thm:coja-fullrank} by a one-sided coupling that absorbs the random row count.

Since $\hat e D<K_*(k)$ strictly, we may fix $\eta>0$ small enough that $(1+\eta)\,\hat e D<K_*(k)$, and set $m^+:=\big\lfloor(1+\eta)\hat e M\big\rfloor$, so that the associated effective degree satisfies $k m^+/N=(1+\eta)\hat e D\,(1+o(1))<K_*(k)$ for large $N$. Draw $m^+$ i.i.d.\ uniform $k$-subsets of $V$, forming a matrix $B^+\sim\mathbb{G}_\textnormal{ER}\!\big(N,k,km^+/N\big)$ with the deterministic row count required by \cref{thm:coja-fullrank}. Couple $B_S$ to $B^+$ by taking $B_S$ to be the first $|S|$ rows of $B^+$ on the event $\{|S|\le m^+\}$; this is a valid coupling, since conditioned on $|S|\le m^+$ both $B_S$ and the first $|S|$ rows of $B^+$ are $|S|$ i.i.d.\ uniform $k$-subsets of $V$. On the event $\{|S|\le m^+\}$ the rows of $B_S$ are a subset of the rows of $B^+$, and because full row rank is preserved under deletion of rows,
\[
    \{B^+\text{ has full row rank}\}\cap\{|S|\le m^+\}\ \subseteq\ \{B_S\text{ has full row rank}\}.
\]
Consequently
\[
    \mathbb{P}\big[B_S\text{ not full row rank}\big]\ \le\ \mathbb{P}\big[B^+\text{ not full row rank}\big]+\mathbb{P}\big[|S|>m^+\big].
\]
The first term is $o_N(1)$ by \cref{thm:coja-fullrank}, applied to $B^+$ at effective degree $(1+\eta)\hat e D<K_*(k)$. The second term is $o_N(1)$ by a Chernoff bound:
 $|S|\sim\text{Bin}(M,\hat e)$ has mean $\hat e M= \Theta(N)$ for any fixed $\hat e$, and the threshold is $m^+ = (1+\eta) \hat{e} M+O(1)$, giving $\mathbb{P}[|S| > m^+] \le e^{-\Omega(N)}$.
Hence, under the joint law of $B$ and $S$,
\[
    \mathbb{P}_{B,S}\!\left[
        B_S\text{ is not full row rank}
    \right]
    =
    o_N(1).
\]
Since this probability is taken jointly over $B$ and $S$,
Markov's inequality applied to the conditional failure
probability gives, for every fixed $\delta>0$,
\[
    \mathbb{P}_B\!\left[
        \mathbb{P}_S\!\left(
            B_S\text{ is not full row rank}
            \,\middle|\,B
        \right)>\delta
    \right]
    \leq
    \frac{
        \mathbb{P}_{B,S}\!\left[
            B_S\text{ is not full row rank}
        \right]
    }{\delta}
    =
    o_N(1).
\]
Thus, with probability $1-o_N(1)$ over $B$, the USD decoder
has failure probability at most $\delta$, and hence solves the
$(B,\gamma,\delta)$-quantum decoding problem.
Combining the condition $\hat e D<K_*(k)$ with the USD erasure
rate \cref{eq:erasure_rate}, we conclude that, for every fixed
$\delta>0$, the USD decoder solves the
$(B,\gamma,\delta)$-quantum decoding problem with probability
$1-o_N(1)$ over $B$ whenever
\begin{equation}
    \label{eq:gamma_USD_implicit}
    2\sqrt{\gamma(1-\gamma)}
    <
    \frac{K_*(k)}{D}.
\end{equation}
For the relevant range $\gamma\in[0,1/2]$, define the threshold
$\gamma_\textnormal{USD}(k,D)$ by equality in
\cref{eq:gamma_USD_implicit}; namely,
\begin{equation}
    \label{eq:gamma_USD}
    \gamma_\textnormal{USD}(k,D)
    =
    \frac{1}{2}
    -
    \frac{1}{2}
    \sqrt{
        1-\left(\frac{K_*(k)}{D}\right)^2
    }.
\end{equation}
Thus the USD decoder succeeds for every
$\gamma<\gamma_\textnormal{USD}(k,D)$, as claimed in
\cref{thm:gamma_USD}.

Substituting \cref{eq:gamma_USD} into
\cref{eq:betamax} gives
\begin{equation}
    \label{eq:betamax_val}
    \beta_{\textnormal{ALG}}(k,D)
    =
    \tanh^{-1}\!\left(\frac{K_*(k)}{D}\right).
\end{equation}
Applying \cref{thm:gamma_vs_beta} then gives the Gibbs-sampling
conclusion of \cref{thm:gamma_USD}.

\label{def:stability}

\label{thm:main-hardness}
Consider $H\sim\mathbb{G}_{\rm ER}(N,k,D)$. Fix any sufficiently small $\varepsilon>0$. 
For all $k \ge k_0(\varepsilon)$, $D\ge k$, and $\beta$ satisfying

    \[
    \liminf_{N\to\infty}
    \mathbb{E}_{H\sim\mathbb{G}_\textnormal{ER}(N,k,D)} \big[W_1(\mu_{\beta,H}, \mathcal{A}_N(H))\big] >0.
    \]

<!-- definition-6: body_md -->
\label{def:shattering}
For any $\varepsilon>0$, we say $\mu_{\beta,H}$ is an $\varepsilon$-\emph{shattered} Gibbs measure if for some constant $c>0$ (possibly depending on $\varepsilon$) and w.h.p. with respect to randomness of $H$ as $N\to\infty$, there exist non-empty subsets $\mathcal{C}_1,\ldots, \mathcal{C}_K \subseteq \boldsymbol{\Sigma}_N$ called clusters such that:
\begin{enumerate}[label=\sf{S}\arabic*]
    \item 
    \label{it:each-cluster-small}
    (No dominant cluster)
    $\max_i \mu_{\beta,H}(\mathcal{C}_i)\leq e^{-cN}$.
    \item 
    \label{it:diameter-bound}
    (Small cluster diameter)
    Each cluster has diameter $\operatorname{diam}(\mathcal{C}_i)\leq 2\sqrt{\varepsilon N}$ with respect to Euclidean distance on $\{+1,-1\}^N$, or Hamming distance $\varepsilon N$.
    \item 
    \label{it:separation-bound}
    (Well-separated clusters)
    For any $i\neq j$, the Euclidean distance between the nearest points in $\mathcal{C}_i$ and $\mathcal{C}_j$ satisfies $d(\mathcal{C}_i,\mathcal{C}_j)\geq 2\sqrt{\varepsilon^3 N}$.
    In terms of Hamming distance, this means $d_H(\mathcal{C}_i,\mathcal{C}_j) \ge \varepsilon^3 N$.
    \item 
    \label{it:cover-gibbs-measure}
    (Clusters carry all mass)
    $\mu_{\beta,H}\bigl(\bigcup_{i=1}^K \mathcal{C}_i\bigr)\geq 1-e^{-cN}$.
\end{enumerate}
<!-- end definition-6 -->
<!-- remark-4: body_md -->
In the above, we use the correspondence between Euclidean and Hamming distances for $\{+1,-1\}^N$ vectors: $\|{\mathbf{z}} - {\mathbf{z}}'\|_2^2 = 4\, d_H({\mathbf{z}},{\mathbf{z}}')$, so $\|{\mathbf{z}} - {\mathbf{z}}'\|_2 = 2\sqrt{d_H({\mathbf{z}},{\mathbf{z}}')}$.
The separation bound \ref{it:separation-bound} in Euclidean distance $2\sqrt{\varepsilon^3 N}$ thus corresponds to Hamming distance $\varepsilon^3 N$.
Note also that properties \ref{it:each-cluster-small} and \ref{it:cover-gibbs-measure} together force the number of clusters $K$ to be exponentially large in $N$.

\label{thm:single-clause-disorder-chaos}
Fix any $\varepsilon$ and $(k,D,\beta)$ satisfying the hypotheses of \cref{thm:main-hardness}.
Let $H,H'$ be sampled to share $M-1$ clauses (i.e. $f_i=f_i'$ for $1\leq i\leq M-1$) and have the last clauses $f_M,f_M'$ chosen independently, so that $H,H'$ are coupled with the same Erd\H{o}s-R\'enyi distribution $\mathbb{G}_\textnormal{ER}(N,k,D)$.

<!-- proof-4: body_md -->
Consider $H,H'$ with a single resampled clause.
By Theorem~\ref{thm:single-clause-disorder-chaos}, there exists a constant $\iota_* = \iota_*(k,D,\beta,\varepsilon)>0$ such that
\[
\liminf_{N\to\infty}\operatorname*{\mathbb{E}} W_1(\mu_{\beta,H},\mu_{\beta,H'})\geq 3\iota_*
\]
Now let $\mathcal{A}$ be an $\iota$-stable algorithm with $\iota \le \iota_*$, meaning $\operatorname*{\mathbb{E}} [W_1(\mathcal{A}(H), \mathcal{A}(H'))] \le \iota$.
By the triangle inequality for $W_1$:
\[
W_1(\mu_{\beta,H}, \mathcal{A}(H))
+
W_1(\mu_{\beta,H'}, \mathcal{A}(H'))
\geq 
W_1(\mu_{\beta,H},\mu_{\beta,H'})
-
W_1(\mathcal{A}(H), \mathcal{A}(H')).
\]
The right-hand side has $\liminf$ at least $3\iota_* - \iota\ge 2\iota_*$ in expectation, while the left-hand side consists of two identically distributed random variables, so we deduce that 
\[
\liminf_{N\to\infty}
\operatorname*{\mathbb{E}} W_1(\mu_{\beta,H}, \mathcal{A}(H))
\geq \frac{3\iota_*-\iota}{2} \ge \iota_*>0
\]
as desired.

For $r\in[M]$, let $H^{(r)}$ be obtained by replacing $f_r$ by an independent copy, while keeping $\omega$ fixed. Orthogonality of the terms in \eqref{eq:clause-polynomial-decomposition} gives
\[

\mathbb{E}\big\|\mathbf X^\circ(H,\omega)-\mathbf X^\circ(H^{(r)},\omega)\big\|_2^2
=2\sum_S |S|\,\mathbb{E}\|\mathbf X_S^\circ\|_2^2
\le 2d\,\mathbb{E}\|\mathbf X^\circ\|_2^2
\le 2dCN.
\]
Clause-permutation invariance makes the $M$ summands equal. Hence, for the pair $(H,H')$ in \cref{def:stability},
\begin{equation}
\label{eq:single-clause-L2-influence}
\mathbb{E}\big\|\mathbf X^\circ(H,\omega)-\mathbf X^\circ(H',\omega)\big\|_2^2

For any Gibbs measure exhibiting the shattering property as in \cref{def:shattering}, we can refine it so that all clusters have an additional \emph{bottleneck} property.
Letting $\mathcal{N}_r(\mathcal{C})$ be the set of points within Hamming distance $r$ of a set $\mathcal{C}$ which are not themselves in $\mathcal{C}$, the bottleneck property requires that

\]
for $r_b=\varepsilon^3 N/3$.

\mu_{\beta}(\cup_{j\leq J} \mathcal{C}_j)
\leq

\leq 
e^{-cN/10}.
\]
Since only an exponentially small amount of Gibbs mass is removed this way, it follows that the refined clustering obeys all shattering properties from before in addition to the bottleneck property.

\mathbb{P}\Big[\Big|\ln Z_{\beta,H} - \operatorname*{\mathbb{E}}[\ln Z_{\beta,H}]\Big| > t N\Big] 
\le 2\exp\!\Big({-}\frac{2t^2N^2}{M(2\beta)^2}\Big)
\le 2\exp\!\Big({-}\frac{k t^2N }{2D\beta^2}\Big),
\]
where the last inequality uses $M=\lfloor ND/k\rfloor \le ND/k$.



Applying Jensen's inequality to the logarithm, we have $\mathbb{E} \ln Z_{\beta, H} \le \ln \mathbb{E} Z_{\beta,H}$. 
Suppose for the sake of contradiction against \eqref{eq:annealed-FE} that for some $t>0$,  $\mathbb{E} \ln Z_{\beta, H} \le \ln \mathbb{E} Z_{\beta,H} - 2t N$ for arbitrarily large $N$.
Then
\begin{align*}
    2\exp\!\Big({-}\frac{k t^2N }{2D\beta^2}\Big) &\ge \mathbb{P}\Big[\Big|\ln Z_{\beta,H} - \operatorname*{\mathbb{E}}[\ln Z_{\beta,H}]\Big| > t N\Big] \\
    &\ge \mathbb{P}[\ln Z_{\beta,H} \ge \ln \mathbb{E} [Z_{\beta,H}] - \ln 2] \\
    &\ge \frac14 e^{-o(N)},
\end{align*}
where the second inequality follows from observing that the event in the first line is implied by the event in the second line under the supposition.
For sufficiently large $N$, $2\exp\!\Big({-}\frac{k t^2N }{2D\beta^2}\Big) \not\ge\frac14 e^{-o(N)}$, yielding a contradiction as desired.

<!-- corollary-2: body_md -->
\label{cor:most-pairs-orthogonal}
Suppose $\beta^2 \le k/(2D)$.
For any $\delta>0$, there exists $c>0$ such that with probability at least $1-e^{-cN}$ over $H\sim\mathbb{G}_\textnormal{ER}(N,k,D)$, we have
\[
\mu_{\beta,H}^{\otimes 2}\!\left[|{\mathbf{z}}^\top {\mathbf{z}}'|/N\geq \delta\right]\leq e^{-cN}.
\]
<!-- end corollary-2 -->
<!-- proof-10: body_md -->
Let
\[
A_H(\delta):=\sum_{\substack{{\mathbf{z}},{\mathbf{z}}'\in \boldsymbol{\Sigma}_N\\ | {\mathbf{z}}^ \top{\mathbf{z}}'|/N\ge \delta}} e^{-\beta(H({\mathbf{z}})+H({\mathbf{z}}'))}.
\]
Since the variational problem in \eqref{eq:2nd-moment-condition} is uniquely maximized at $R=0$, there exists $c_1=c_1(\delta,\beta,k,D)>0$ such that
\[
\operatorname*{\mathbb{E}}[A_H(\delta)]\le e^{-c_1N}(\operatorname*{\mathbb{E}} Z_{\beta,H})^2.
\]
By \cref{lem:annealed-FE} and McDiarmid's inequality, there exists $c_2>0$ such that with probability at least $1-e^{-c_2N}$ we have
\[
Z_{\beta,H}\ge e^{-c_1N/4}\operatorname*{\mathbb{E}} Z_{\beta,H}.
\]
On this event,
\[
\mu_{\beta,H}^{\otimes 2}\!\left[\frac{|\langle {\mathbf{z}},{\mathbf{z}}' \rangle|}{N}\ge \delta\right]
=
\frac{A_H(\delta)}{Z_{\beta,H}^2}
\le
e^{c_1N/2}\frac{A_H(\delta)}{(\operatorname*{\mathbb{E}} Z_{\beta,H})^2}.
\]
Markov's inequality and the bound on $\operatorname*{\mathbb{E}}[A_H(\delta)]$ therefore imply
\[
\mathbb P\!\left[\mu_{\beta,H}^{\otimes 2}\!\left[\frac{|\langle {\mathbf{z}},{\mathbf{z}}' \rangle|}{N}\ge \delta\right]\ge e^{-c_1N/4}\right]\le e^{-cN}
\]
for some $c>0$, which yields the claimed estimate after adjusting constants.

<!-- proposition-2: body_md -->
\label{prop:contiguity}
Suppose $\beta\in [0,\beta_\textnormal{RSB}(k, D)]$. 
If a property of $(H,{\mathbf{z}}^*)$ has exponentially good probability under the planted model $\mathbb{P}$, then the same holds under the null model $\mathbb{Q}$.
<!-- end proposition-2 -->
<!-- proof-11: body_md -->
This can be shown using the same ``contiguity at exponential scale'' argument as in \cite[Section 3.1]{alaoui2023shattering}, which we adopt for the present ensemble as follows.
Let $A$ be the event that the property fails, and suppose $\mathbb{P}(A)\le e^{-aN}$. The displayed Radon--Nikodym derivative gives
\[
\frac{d\mathbb{Q}}{d\mathbb{P}}(H,{\mathbf{z}}^*)=\frac{\mathbb{E} Z_{\beta,H}}{Z_{\beta,H}}.
\]
Fix any $b\in(0,a)$.  Since \eqref{eq:annealed-FE} is implied by $\beta\in [0,\beta_\textnormal{RSB}(k,D)]$, for all large $N$ we have
$\mathbb{E}\ln Z_{\beta,H}\ge \ln\mathbb{E} Z_{\beta,H}-bN/2$. McDiarmid's inequality, applied exactly as in the proof of \cref{lem:annealed-FE}, then gives a constant $t_b>0$ such that under the null model
\[
\mathbb{Q}\left[Z_{\beta,H}<e^{-bN}\mathbb{E} Z_{\beta,H}\right]\le e^{-t_bN}.
\]
On the complementary event, $d\mathbb{Q}/d\mathbb{P}\le e^{bN}$. Therefore
\[
\mathbb{Q}(A)\le e^{bN}\mathbb{P}(A)+e^{-t_bN}
\le e^{-(a-b)N}+e^{-t_bN},
\]
which proves the proposition.

<!-- lemma-4: body_md -->
\label{prop:pairs-of-terms}
In the planted model, condition on ${\mathbf{z}}^*$ and choose ${\mathbf{z}}$ with  Hamming distance $d_H({\mathbf{z}},{\mathbf{z}}^*) =\varepsilon_* N/k$, such that $H$ and ${\mathbf{z}}$ are conditionally independent given ${\mathbf{z}}^*$.
Then the law of $(f_1({\mathbf{z}}^*),f_1({\mathbf{z}}))$ is given by $\mathbb{P}[(f_1({\mathbf{z}}^*),f_1({\mathbf{z}}))=(a,b)]=p(a,b)$ where
\begin{align*}
    p(+1,+1)
    &=
    (1-\tilde{\varepsilon})
    \cdot \frac{e^{\beta}}{2\cosh(\beta)},
    &\qquad
    p(+1,-1)
    &=
    \tilde{\varepsilon}
    \cdot \frac{e^{\beta}}{2\cosh(\beta)},
    \\
    p(-1,+1)
    &= 
    \tilde{\varepsilon}
    \cdot \frac{e^{-\beta}}{2\cosh(\beta)},
    &\qquad
    p(-1,-1)
    &= 
    (1-\tilde{\varepsilon})
    \cdot \frac{e^{-\beta}}{2\cosh(\beta)},
\end{align*}
where
\begin{equation}
\label{eq:tilde-eps-def}
\tilde{\varepsilon} = \frac{1-(1-2\varepsilon_*/k)^k}{2} + O\left(\frac{k^2}{N}\right)= \varepsilon_* - \Big(1-\frac{1}{k}\Big) \varepsilon_*^2 + O\left(\varepsilon_*^3 + \frac{k^2}{N}\right).
\end{equation}
<!-- end lemma-4 -->
<!-- proof-12: body_md -->
Here, we consider only the planted model $\mathbb{P}$.
Note that given ${\mathbf{z}}$ and ${\mathbf{z}}^*$ at Hamming distance $\varepsilon_* N/k$, the clause agrees as long as the two strings differ by an even number of bits involved in the clause.
Hence,
\[
\mathbb{P}\big[f_1({\mathbf{z}}^*)=f_1({\mathbf{z}})~|~({\mathbf{z}},{\mathbf{z}}^*)\big]
= \frac{1}{\binom{N}{k}} \sum_{\ell \text{ even}} \binom{\varepsilon_*N/k}{\ell}\binom{N-\varepsilon_*N/k}{k-\ell} =: 1-\tilde{\varepsilon}.
\]
Following the same collision coupling scheme in Eq.~\eqref{eq:pk-hypergeometric}, we can approximate the distribution of the number of disagreeing coordinates as binomial instead of hypergeometric (by drawing the vertices in each clause hyperedge i.i.d.~with replacement), with error at most $O(k^2/N)$.
This yields
\begin{align*}
\tilde{\varepsilon}
&= 1-\sum_{\ell \text{ even}} \binom{k}{\ell} \left(\frac{\varepsilon_*}{k}\right)^\ell \left(1-\frac{\varepsilon_*}{k}\right)^{k-\ell} + O(k^2/N) =
\frac{1-(1-2\varepsilon_*/k)^k}{2} + O(k^2/N).
\end{align*}
Note that the event in question does not depend on the choice of weight $J_{e^{(1)}}$.
Due to this independence, we see that this event is independent of the event that $f_1({\mathbf{z}}^*)=1$, and so multiplying probabilities finishes the proof.

<!-- proposition-3: body_md -->
\label{prop:soft-OGP-planted}
For parameters as in \eqref{eq:eps-to-zero-slowly}, satisfying \eqref{eq:cond-eps-k-D-delta} and \eqref{eq:beta-k-def}, the $(\varepsilon,\delta)$-soft OGP holds with exponentially good probability under the planted model $\mathbb{P}$. 
Hence by Proposition~\ref{prop:contiguity}, the same holds under the null model $\mathbb{Q}$.
<!-- end proposition-3 -->
<!-- proof-13: body_md -->
Property~\ref{it:energy-typical} is immediate from the definition of $\mathbb{P}$, and just amounts to exponential concentration for sums of i.i.d. Bernoulli variables via e.g., Hoeffding's inequality.



For property~\ref{it:OGP-soft}, we consider all ${\mathbf{z}}$ with $d_H({\mathbf{z}},{\mathbf{z}}^*)=\varepsilon_*N/k$ where $\varepsilon_*$ varies in the set 
\begin{equation} \label{eq:eps-star-range}
\varepsilon_*\in \Big[\varepsilon-\frac{\varepsilon^2}{k},\varepsilon+\frac{\varepsilon^2}{k}\Big]\cap (N/k)^{-1}\mathbb{Z}.
\end{equation}
We will estimate the maximum value of $H({\mathbf{z}}^*)-H({\mathbf{z}})$ over such ${\mathbf{z}}$ via a union bound.
The number of such points ${\mathbf{z}}$ for any fixed $\varepsilon_*$ is
\begin{align}
\binom{N}{\varepsilon_* N/k} 
\le \exp\left(\frac{N \varepsilon_*}{k} \ln \frac{e k}{\varepsilon_*}\right)
\le 
\exp\left(N \Big(1 + \frac{\varepsilon}{2}\Big) \frac{\varepsilon_* \ln(k)}{k}\right),
\label{eq:entropy-of-nearby-points}
\end{align}
where the last inequality follows from \eqref{eq:cond-eps-k-D-delta} and $\varepsilon_*=\varepsilon+O(\varepsilon^2/k)$.
For each such ${\mathbf{z}}$, consider the energy difference
\begin{equation}
\label{eq:H-diff-sum}
H({\mathbf{z}})-H({\mathbf{z}}^*)
=
\sum_{i=1}^M [f_i({\mathbf{z}}^*)-f_i({\mathbf{z}})]
.
\end{equation}
\cref{prop:pairs-of-terms} shows that each summand term equals $0$ with probability $1-\tilde{\varepsilon}$, and has expectation
\[
\mathbb{E}^{\mathbb{P}}[f_i({\mathbf{z}}^*)-f_i({\mathbf{z}})]
=
2\tanh(\beta) \tilde{\varepsilon}
\]
where $\tilde{\varepsilon}=\varepsilon_*\pm O(\varepsilon_*^2 + k^2/N)$.
Thus 
\[
\mathbb{E}^{\mathbb{P}}[H({\mathbf{z}})-H({\mathbf{z}}^*)] 
=
2M \tanh(\beta) \tilde{\varepsilon} .
\]



Next we want to show $H({\mathbf{z}}) - H({\mathbf{z}}^*) \ge  3\delta M$ with high probability.
Along with property~\ref{it:energy-typical} which means that $H({\mathbf{z}}^*)\ge (\Gamma-\delta)M$, this would imply $H({\mathbf{z}}) \ge (\Gamma + 2\delta)M$ and prove property~\ref{it:OGP-soft}.
To do so we use Bennett's inequality, which allows us to leverage the reduced variance in each term (and is relatively sharp because the large deviation event we consider has small constant factors).
Note that 
\[
S({\mathbf{z}})=H({\mathbf{z}}^*) - H({\mathbf{z}}) - 2\Gamma\tilde{\varepsilon} M = \sum_{i=1}^M (f_i({\mathbf{z}})-f_i({\mathbf{z}}^*)+2\tanh(\beta)\tilde{\varepsilon})
\]
is a sum of i.i.d. mean zero terms valued in $[-2-2\tilde{\varepsilon},2+2\tilde{\varepsilon}]$ (since $|\tanh(\beta)|\leq 1$).
Furthermore, the variance of each term is at most $\mathbb E[(f_1({\mathbf{z}})-f_1({\mathbf{z}}^*))^2]=4\tilde{\varepsilon}$ by \cref{prop:pairs-of-terms}.
Therefore with $g(u)=(1+u)\ln(1+u)-u$, Bennett's inequality yields
\begin{align}
& \mathbb{P}[H({\mathbf{z}}^*) > H({\mathbf{z}}) - 3\delta M] = 
\mathbb P\Big[S({\mathbf{z}}) > \Big(2\tanh(\beta)\tilde{\varepsilon}-3\delta\Big) M\Big] \nonumber\\
&\qquad 
\leq 
\exp\left(
-\frac{4M\tilde{\varepsilon}}{4(1+\tilde{\varepsilon})^2}
g\left(\frac{2(1+\tilde{\varepsilon})(2\tanh(\beta)\tilde{\varepsilon}-3\delta) M}{4M\tilde{\varepsilon}}\right)
\right)\nonumber\\
&\qquad 
=
\exp\left(
-\frac{M\tilde{\varepsilon}}{(1+\tilde{\varepsilon})^2}
g\Big(
 (1+\tilde{\varepsilon})\Big[\tanh (\beta) - \frac{3\delta}{2\tilde{\varepsilon}}\Big]
\Big)
\right).
\label{eq:bennett-bound}
\end{align}
Let $\beta_0=(1+\varepsilon)\sqrt{2\ln k/D}$. Note $\tanh\beta_0 > 3\delta/(2\tilde{\varepsilon})$ by \eqref{eq:cond-eps-k-D-delta}. The functions $g$ and $\tanh$ are increasing on $[0,\infty)$, so we may lower bound this exponent by replacing $\beta$ with $\beta_0$.
Since $\beta_0=O(\sqrt{\ln k / k})$ for $D \ge k$, and $g(u)=u^2/2+O(u^3)$ as $u\to0$, we have
\[
\frac{M\tilde{\varepsilon}}{(1+\tilde{\varepsilon})^2} g\Big( (1+\tilde{\varepsilon})\Big[\tanh (\beta_0) - \frac{3\delta}{2\tilde{\varepsilon}}\Big]\Big) = \frac{M\tilde{\varepsilon} \beta_0^2}{2} \Big[1+O(\beta_0 + \frac{\delta}{\tilde{\varepsilon}\beta_0})\Big]
\]
By condition \eqref{eq:cond-eps-k-D-delta}, both errors are small compared to $\varepsilon$.
Furthermore, for $\varepsilon_*$ in the allowed range \eqref{eq:eps-star-range}, \eqref{eq:tilde-eps-def} gives
\[
\frac{\tilde{\varepsilon}}{\varepsilon_*} = 1 - \Big(1-\frac{1}{k}\Big) \varepsilon_* + O(\varepsilon_*^2) + o_N(1) = 1-\varepsilon + O(\varepsilon^2) + o_N(1)
\]
Using $M=\lfloor N D/k\rfloor$, we conclude that uniformly for $\beta\ge\beta_0$ and over allowed $\varepsilon_*$, the exponent in \eqref{eq:bennett-bound} satisfies
\begin{align}
\frac{M\tilde{\varepsilon}}{(1+\tilde{\varepsilon})^2} 
g\Big( (1+\tilde{\varepsilon})\Big[\tanh (\beta) - \frac{3\delta}{2\tilde{\varepsilon}}\Big]\Big)
&\ge 
N\frac{\varepsilon_*\ln k}{k}(1+\varepsilon)^2\Big[1-\varepsilon - O\big(\varepsilon^2 + \beta_0 + \frac{\delta}{\varepsilon\beta_0}\big)-o_N(1)\Big] \nonumber \\
&
\ge N(1+0.99\varepsilon)\frac{\varepsilon_*\ln k}{k},
\label{eq:beta-bound}
\end{align}
for all sufficiently small $\varepsilon$, followed by sufficiently large $k$ and then $N$.
This rate exceeds the entropy rate in \eqref{eq:entropy-of-nearby-points} by $\Omega(N\varepsilon^2\ln k/k)$. To complete the proof, we apply a union bound over the configurations and the at most $N+1$ possible distances, which then shows property~\ref{it:OGP-soft} fails with probability at most $e^{-cN}$ for a sufficiently small $c>0$.

<!-- proposition-7: body_md -->
\label{prop:pair-anticoncentration-suffices}
    Let $(\nu_N,\nu_N')_{N\ge1}$ be a sequence of pairs of discrete probability measures on a countable set $\mathcal{X}_N$.

    \end{equation}
<!-- end proposition-7 -->

\ge

where the inequality is due to the possible existence of $x\in \mathcal{X}_N$ where $\nu_N(x)=0$ but $\nu_N'(x)\neq 0$.

<!-- proposition-8: body_md -->
\label{prop:ratio-anticoncentrates}
    With $\nu_N=\hat\mu$ and $\nu_N'=\hat\mu'$, the ratio \eqref{eq:ratio-criterion} does not converge to $1$ in probability (along any subsequence $N_1<N_2<\dots$).
<!-- end proposition-8 -->
<!-- proof-18: body_md -->
Since the likelihood ratio in \eqref{eq:ratio-criterion} is unchanged if we rescale either measure by a positive constant, we may normalize the cluster weights on $\{1,\dots,K\}$. Let
\[
S=\bigcup_{\ell=1}^K \mathcal{C}_\ell,
\qquad
\lambda(\{\ell\}) := \frac{\mu_{\beta,H}(\mathcal{C}_\ell)}{\mu_{\beta,H}(S)}.
\]
By property~\ref{it:cover-gibbs-measure}, we have $\mu_{\beta,H}(S)=1-e^{-\Omega(N)}$.
Also, adding one clause changes every Gibbs weight by a factor in $[e^{-\beta},e^{\beta}]$, so after normalization there is a constant $C_\beta<\infty$ such that for every $\ell$,
\[
C_\beta^{-1}\lambda(\{\ell\}) \le \frac{\hat\mu(\{\ell\})}{\hat\mu(\{1,\ldots,K\})} \le C_\beta \lambda(\{\ell\}).
\]
Hence any event involving the label pair $(i,j)$ that has probability bounded below under $\lambda\otimes\lambda$ also has probability bounded below under the normalized version of $\hat\mu\otimes\hat\mu$. It is therefore enough to prove the proposition with $i,j$ drawn i.i.d. from $\lambda$, and we return to $\hat{\mu}$ at the end of this proof.

Now draw $i,j$ i.i.d. from $\lambda$, and conditional on $(i,j)$ let ${\mathbf{z}}^{(i)},{\mathbf{z}}^{(j)}$ be independent Gibbs samples from $\mu_{\beta,H}$ conditioned to lie in $\mathcal{C}_i,\mathcal{C}_j$. Averaged over the random labels $(i,j)\sim \lambda\otimes \lambda$, the pair $({\mathbf{z}}^{(i)},{\mathbf{z}}^{(j)})$ is distributed as two i.i.d. Gibbs samples, conditioned on lying in $S$.
Since $\mu_{\beta,H}(S) = 1-e^{-\Omega(N)}$, this conditioning changes probabilities by at most $e^{-\Omega(N)}$.

Let $E=\{e_1,\dots,e_k\}$ be the random support of the new clause in $H'$, and write
\[
\chi_E({\mathbf{z}}):=\prod_{a=1}^k z_{e_a}.
\]
For each cluster $\mathcal{C}_\ell$, let $\hat {\mathbf{z}}^{(\ell)}\in\boldsymbol{\Sigma}_N$ be the coordinate-wise weighted majority vote within $\mathcal{C}_\ell$, that is, $\hat{z}^{(\ell)}_a = \operatorname{sign}(\sum_{{\mathbf{z}}\in \mathcal{C}_\ell} \mu_{\beta,H}({\mathbf{z}}) z_a)$ for $a = 1,2,\ldots,N$, with ties broken arbitrarily.
Since every pair of points in $\mathcal{C}_\ell$ has Hamming distance at most $\varepsilon N/k$ by property~\ref{it:diameter-bound}, and the majority vote minimizes the average Hamming distance, then
\[
\mathbb E\big[d_H({\mathbf{z}},\hat {\mathbf{z}}^{(\ell)}) ~\big|~ {\mathbf{z}} \sim \mu_{\beta, H},~ {\mathbf{z}}\in \mathcal{C}_\ell\big]\le \varepsilon N/k.
\]
This implies that on a uniformly random coordinate, ${\mathbf{z}}$ and $\hat{\mathbf{z}}^{(\ell)}$ differ with probability at most $\varepsilon/k$.
For any fixed support $E$ and for each cluster $\ell$, define
\begin{equation}
\label{eq:minority-weight}
    q_\ell(E) := \mathbb{P}\Big[{\mathbf{z}}|_E \neq \hat {\mathbf{z}}^{(\ell)}|_E \,\Big|\, {\mathbf{z}}\sim \mu_{\beta,H},~ {\mathbf{z}}\in \mathcal{C}_\ell\Big].
\end{equation}
where ${\mathbf{z}}|_E=\hat{\mathbf{z}}|_E$ denotes $z_a=\hat{z}_a$ $\forall a\in E$.
For each $\ell$, over the uniformly random support $E$, a union bound over the $k$ coordinates give
\begin{equation}
    \mathbb{E}_E[q_\ell (E)] \le k \cdot \frac{\varepsilon}{k} = \varepsilon \qquad
    \Longrightarrow \qquad \mathbb{P}_E [q_\ell(E) > \sqrt{\varepsilon}] \le \sqrt{\varepsilon},
\end{equation}
where the implication follows from Markov's inequality.
We then define the event (depending only on $E$ and the labels $i,j$)
\begin{equation}
\label{eq:majority-vote-usually-correct}
A_1 := \Big\{q_i(E) \le \sqrt{\varepsilon} \quad \text{and} \quad q_j(E) \le \sqrt{\varepsilon} \Big\}, 
\qquad
\mathbb{P}[A_1] \ge 1-2 \sqrt{\varepsilon}.
\end{equation}

Next, \cref{cor:most-pairs-orthogonal} together with \cite[Proposition 1.4.14]{talagrand2011mean1} implies that for a uniformly random set of $O(1)$ coordinates, the joint marginal of two i.i.d. Gibbs samples is asymptotically uniform (up to expected total variational error tending to $0$ with $N$). Hence
\[
\mathbb{P} \big[\chi_E({\mathbf{z}}^{(i)})\neq \chi_E({\mathbf{z}}^{(j)})\big] = \frac12 + o_N(1).
\]
Define
\[
s_\ell :=\chi_E(\hat{{\mathbf{z}}}^{(\ell)})
\]
as the clause value of the majority representative (which is a deterministic function of $H,E,\ell$), and the event
\[
A_2 = \{s_i \neq s_j\}.
\]
On $A_1$ we have
\[
\chi_E\big({\mathbf{z}}^{(i)}\big)=s_i,
\qquad
\chi_E\big({\mathbf{z}}^{(j)}\big)=s_j,
\]
except with probability at most $q_i(E)+q_j(E)\le 2\sqrt\varepsilon$, since ${\mathbf{z}}|_E=\hat{\mathbf{z}}^{(\ell)}|_E$ forces $\chi_E({\mathbf{z}})=s_\ell$.
Therefore, for large $N$ and small $\varepsilon$
\begin{align}
\mathbb{P}[A_2] &\ge \mathbb{P}\big[\chi_E({\mathbf{z}}^{(i)})\neq \chi_E({\mathbf{z}}^{(j)})\big] - 2\sqrt{\varepsilon} - \mathbb{P}[A_1^c]  \ge \frac12-O(\sqrt{\varepsilon}) - o_N(1) \ge \frac13,
\\
\text{and}
\qquad
\mathbb{P}[A_1\cap A_2] &\ge \mathbb{P}[A_1] + \mathbb{P}[A_2] - 1 \ge \frac13 - O(\sqrt{\varepsilon})\ge \frac14.
\end{align}

Now fix $(H,E,i,j)$, and for $J\in\{+1,-1\}$ let $H^{(J)}=H-J\chi_E$ be the Hamiltonian obtained by adding the clause with support $E$ and sign $J$. Let $\hat\mu^{(J)}$ denote the corresponding compressed cluster measure. For each cluster $\mathcal{C}_\ell$, define
\[
m_\ell(J):=\mathbb E\big[e^{\beta J\chi_E({\mathbf{z}})} \,\big|\, {\mathbf{z}}\sim \mu_{\beta,H},~ {\mathbf{z}}\in \mathcal{C}_\ell\big],
\]
where the expectation is only over the conditional Gibbs sample ${\mathbf{z}}$.
Since $\mu_{\beta,H^{(J)}}({\mathbf{z}})\propto \mu_{\beta,H}({\mathbf{z}})e^{\beta J\chi_E({\mathbf{z}})}$, we have
\[
\frac{\hat\mu^{(J)}(\{i\})}{\hat\mu^{(J)}(\{j\})}
=
\frac{\mu_{\beta,H}(\mathcal{C}_i)\,m_i(J)}{\mu_{\beta,H}(\mathcal{C}_j)\,m_j(J)}.
\]
Splitting the expectation defining $m_\ell(J)$ according to whether ${\mathbf{z}}|_E=\hat{\mathbf{z}}^{(\ell)}|_E$, and noting that the two possible values $e^{\pm\beta}$ of $e^{\beta J\chi_E({\mathbf{z}})}$ differ by only $2\sinh\beta = O(\beta)$, we get on the event $A_1$ that
\[
m_i(J)=e^{\beta J s_i}(1+O(\beta\sqrt{\varepsilon})),
\qquad
m_j(J)=e^{\beta J s_j}(1+O(\beta\sqrt{\varepsilon})).
\]
Hence, on $A_1$,
\begin{equation}
\frac{\hat\mu^{(+)}(\{i\})/\hat\mu^{(+)}(\{j\})}{\hat\mu^{(-)}(\{i\})/\hat\mu^{(-)}(\{j\})}
=
e^{2\beta(s_i-s_j)}(1+O(\beta\sqrt{\varepsilon})).
\label{eq:mu-ratio}
\end{equation}
Furthermore, on the event $A_1 \cap A_2$, we have $s_i-s_j \in \{\pm 2\}$, so the above value is $e^{\pm4\beta}(1+O(\beta\sqrt{\varepsilon}))$. In particular, for $\varepsilon$ sufficiently small, \eqref{eq:mu-ratio} lies outside of $[e^{-2\beta}, e^{2\beta}]$.

Choose $\delta_*>0$ so small that
\[
\frac{1+\delta_*}{1-\delta_*}<e^{2\beta}.
\]
With the uniformly small $\varepsilon$ used above, on the event $A_1\cap A_2$ the two possible values of
\[
R_J:=\frac{\hat\mu(\{i\})\hat\mu^{(J)}(\{j\})}{\hat\mu(\{j\})\hat\mu^{(J)}(\{i\})}
\]
satisfy $\max\{R_+/R_-,R_-/R_+\}>(1+\delta_*)/(1-\delta_*)$, by \eqref{eq:mu-ratio}.
Since $J$ is an unbiased sign, it follows that conditional on $H,E,i,j$ and $A_1\cap A_2$, at least one of the two sign choices lies outside $[1-\delta_*,1+\delta_*]$, and therefore
\[
\mathbb P\big[R_{J_M}\notin [1-\delta_*,1+\delta_*] \,\big|\, H,E,i,j,A_1\cap A_2\big]\ge \frac12.
\]
Combining this with $\mathbb P[A_1\cap A_2]\ge 1/4$, we obtain
\[
\mathbb P\left[
\frac{\hat\mu(\{i\})\hat\mu'(\{j\})}{\hat\mu(\{j\})\hat\mu'(\{i\})}
\notin [1-\delta_*,1+\delta_*]
\right]\ge \frac18-o_N(1),
\]
where $\hat\mu'=\hat\mu^{(J_M)}$ is the compressed measure for the realized perturbation $H'$. By the pointwise comparability of $\hat\mu/\hat\mu(\{1,\ldots,K\})$ and $\lambda$, the same positive lower bound (up to a constant factor depending only on $\beta$) holds when $(i,j)$ are sampled from the normalized version of $\hat\mu$, as in \eqref{eq:ratio-criterion}. Therefore the likelihood ratio in \eqref{eq:ratio-criterion} does not converge to $1$ in probability (along any subsequence of $N$).

<!-- theorem-10: body_md -->
\label{thm:invertible}
For any $k\ge 3$, there exists $\alpha \in (0,1)$ such that with high probability, a random $k$-XORSAT instance $H\sim\mathbb{G}_{\textnormal{ER}}(N,k,\alpha k)$ has a solution set that shatters into exponentially many well-separated clusters.
Nevertheless, both DQI and a classical (non-stable) algorithm can efficiently sample from its Gibbs measure $\mu_{\beta,H}$ at every $\beta\in[0,\infty]$.
<!-- end theorem-10 -->
<!-- definition-10: body_md -->
We say any XORSAT instance $H=(B,J)$ is \emph{right-invertible} when $B$ is right-invertible, i.e., there exists $B^+$ such that $BB^+ = I$. This is true iff $B$ has full row rank, i.e., $\operatorname{rank}(B)=M\le N$.
<!-- end definition-10 -->
<!-- lemma-5: body_md -->
\label{lem:classical-sample-alg}
Any right-invertible XORSAT instance is satisfiable, and there is an efficient classical algorithm to sample exactly from $\mu_{\beta,H}$ for all $\beta \in [0, \infty]$.
<!-- end lemma-5 -->
<!-- proof-20: body_md -->
Since $B$ has full row rank, let $B^+$ be the right inverse of $B$. 
Note for each of the $2^M$ possible vectors $\tilde{{\mathbf{v}}}\in \mathbb{F}_2^M$, there are $2^{N-M}$ solutions to $B{\mathbf{x}}=\tilde{{\mathbf{v}}}$.
As a result, the XORSAT problem is satisfiable regardless of the choice of ${\mathbf{v}}$. 
The following algorithm can sample from $\mu_{\beta,H}$ efficiently:
\begin{enumerate}
    \item Sample $s\in\{0,1,\ldots,M\}$ as the number of unsatisfied clauses according to the $\Pr(s)= \frac{1}{Z_s} \binom{M}{s}e^{\beta (M-2s)}$, where $Z_s=(2 \cosh\beta)^M$.
    \item Sample $\tilde{{\mathbf{v}}}$ as one of $\binom{M}{s}$ vectors that disagree with ${\mathbf{v}}$ on $s$ bits (obtain by flipping $s$ bits of ${\mathbf{v}}$ at random).
    \item Output sample $\tilde{\mathbf{x}} = B^+\tilde{{\mathbf{v}}} + {\mathbf{w}}$, where ${\mathbf{w}}$ is chosen uniformly at random from the $2^{N-M}$ vectors in $\ker(B)$. This can be done efficiently by finding a basis for $\ker(B)$.
\end{enumerate}
At zero temperature ($\beta=\infty$), take $s=0$ deterministically.
Note every $\tilde{{\mathbf{v}}}$ is sampled with probability proportional to $e^{-2\beta s}=e^{-2\beta |{\mathbf{v}}-\tilde{{\mathbf{v}}}|}$, and they have the same $2^{N-M}$ number of pre-images. Thus this protocol samples each ${\mathbf{x}}$ with its Gibbs weight proportional to $e^{-\beta H((-1)^{\mathbf{x}})}$, as claimed.
<!-- end proof-20 -->
<!-- proof-21: body_md -->
Consider the choice of $\beta=\infty$. Then the Gibbs measure is simply the uniform distribution of all the solutions. It is known in e.g.~\cite{Ibrahimi_2015} that for $\alpha\in (\alpha_d(k), \alpha_s(k))$, where $\alpha_s(k)\le 1$, a random XORSAT instance from $\mathbb{G}_{\textnormal{ER}}(N,k,\alpha k)$ is satisfiable with high probability but the set of ground states shatters into exponentially large (in $N$) number of clusters, each containing an exponential number of solutions.
(For example, $\alpha_s(3)=0.917935$~\cite{Dietzfelbinger2010cuckoo}, and $\alpha_d(3)=0.818469$ \cite[Theorem 1]{Ibrahimi_2015}.)

For any matrix $A \in \mathbb{F}_2^{M\times N}$, let $Z_A\ge 1$ be the number of solutions ${\mathbf{x}}$ to $A{\mathbf{x}}=0$ (i.e., the cardinality of $\ker{A}$).
Note $Z_A=2^{N-\operatorname{rank}(A)}$, and $B$ is right-invertible iff $Z_{B^T}=1$.
Now, for any distribution $\mathbb{G}$ over $H=(B,{\mathbf{v}})$ where ${\mathbf{v}}$ is chosen uniformly from $\mathbb{F}_2^M$ and independent of $B$, $H$ is satisfiable iff ${\mathbf{v}} \in \operatorname{range}(B)$, so
\begin{equation} \label{eq:RI-upperbound}
    \mathbb{P}(H \text{ is satisfiable}) = \mathbb{E}\left[\frac{2^{\operatorname{rank}(B)}}{2^M}\right] =  2^{N-M} \mathbb{E}[1/Z_{B}] \ge \mathbb{P}(Z_{B^T} =1).
\end{equation}
 Furthermore, (see also \cite[Section 6.1]{dembo2009gibbs})
\begin{equation} \label{eq:RI-lowerbound}
    \mathbb{P}(H \text{ is satisfiable})  \le \frac12 + \frac12\mathbb{P}(Z_{B^T} = 1).
\end{equation}
Hence, if $H$ is satisfiable with high probability, then $H$ is right invertible with high probability. Hence the classical algorithm in \cref{lem:classical-sample-alg} can efficiently sample from $\mu_{\beta,H}$ with high probability.

Furthermore, if $B$ has full row rank, then $C^\perp = \ker B^T = \{0\}$, which only has only one codeword. In this case, the quantum decoding problem is trivial, and the DQI/Regev method in \cref{sec:DQI} prepares the target state for Gibbs sampling at every $\beta$.

<!-- proposition-9: body_md -->
\label{prop:restatement-sat-threshold}
Fix any $k\ge 3$. Let $H\sim\mathbb{G}_\textnormal{ER}(N,k,D)$ and retain each clause independently with probability $\theta$. Let $K_*(k)$ be the quantity defined in \eqref{eq:admissible}. If
\begin{equation}
\label{eq:prange-rank-condition}
\theta D<K_*(k),
\end{equation}
then, with probability $1-o_N(1)$ over the instance and the retained set $S$,
\[
\tilde Z_S = 1\qquad\text{(equivalently, }Z_S=2^{N-|S|}\text{)}.
\]
<!-- end proposition-9 -->
<!-- proof-22: body_md -->
Choose $D'$ with $\theta D<D'<K_*(k)$. A Chernoff bound gives $|S|\le m':=\lfloor ND'/k\rfloor$ with probability $1-o_N(1)$. Conditional on $|S|=s$, the selected rows are exactly $s$ i.i.d.\ uniform $k$-subsets of $[N]$, as required by \cref{def:erdos_renyi}; their signs remain independent uniform Rademacher variables. Couple these rows to the first $s$ rows of an $m'$-row matrix with the same row law. By \cref{thm:coja-fullrank}, the $m'$-row matrix has full row rank with probability $1-o_N(1)$ because $km'/N\le D'<K_*(k)$. On that event its first $s$ rows are also linearly independent. Every right-hand side on the selected rows is then consistent and has exactly $2^{N-s}$ solutions, so $\tilde Z_S=1$.

<!-- proposition-10: body_md -->
\label{prop:prange-works}
If $D \tanh(\beta) < K_*(k) $, then Prange's algorithm outputs a sample whose law is within $o(1)$ expected total variation
distance of the Gibbs measure $\mu_{\beta,H}$.
<!-- end proposition-10 -->
<!-- proof-23: body_md -->
By \eqref{eq:joint-measure}--\eqref{eq:tilde-nu-def}, exact Gibbs sampling is $\tilde\nu U$, where
$U(S,\cdot)$ is the uniform law on $\Omega_S$. Prange's algorithm (on success) outputs $\nu U$.
Thus it suffices to show $\mathbb E\,d_{\mathrm{TV}}(\nu,\tilde\nu)=o(1)$ and that the failure probability is $o(1)$.

Let $S\sim\nu$, where each clause is sampled independently with probability $\theta=\tanh\beta$.
Since $\theta D = D \tanh \beta < K_*(k)$,
Proposition~\ref{prop:restatement-sat-threshold}
implies $\mathbb P(\tilde Z_S=1)=1-o(1)$ under the joint law of $(S,\{f_m\})$.
Because $\mathbb E[\tilde Z_S\mid S]=1$ and $\tilde Z_S\ge 0$, we have the identity
$\mathbb E|\tilde Z_S-1|=2\,\mathbb E[(1-\tilde Z_S)_+]$, where $(x)_+ = \max(0,x)$.  Since $(1-\tilde Z_S)_+\in[0,1]$ and
$(1-\tilde Z_S)_+\to 0$ in probability, it follows that
\begin{equation}
\label{eq:L1tildeZS}
\mathbb E|\tilde Z_S-1|=o(1),
\end{equation}
where the expectation is over the random instance and the draw $S\sim\nu$.

Next, by definition $\tilde Z_\beta=\mathbb E_{S\sim\nu}[\tilde Z_S\mid \{f_m\}]$, so Jensen and \eqref{eq:L1tildeZS} give
\[
\mathbb E|\tilde Z_\beta-1|
=
\mathbb E\Big|\mathbb E_{S\sim\nu}\big[\tilde Z_S-1\mid \{f_m\}\big]\Big|
\le
\mathbb E_{S\sim\nu}\,\mathbb E|\tilde Z_S-1|
=
o(1).
\]
In particular, $\tilde Z_\beta\stackrel{p}{\to}1$, and therefore $\tilde Z_S/\tilde Z_\beta\stackrel{p}{\to}1$ for $S\sim\nu$.

Using \eqref{eq:RN-prange} we can write
\[
d_{\mathrm{TV}}(\nu,\tilde\nu)
=
\mathbb E_{S\sim\nu}\Big(1-\frac{\tilde\nu(S)}{\nu(S)}\Big)_+
=
\mathbb E_{S\sim\nu}\Big(1-\frac{\tilde Z_S}{\tilde Z_\beta}\Big)_+.
\]
The integrand lies in $[0,1]$ and converges to $0$ in probability, so taking expectation over the random instance yields
$\mathbb E\,d_{\mathrm{TV}}(\nu,\tilde\nu)=o(1)$.

Finally, total variation contracts under post-processing, hence
$d_{\mathrm{TV}}(\nu U,\tilde\nu U)\le d_{\mathrm{TV}}(\nu,\tilde\nu)$.
Moreover, the failure event is exactly $\{\tilde Z_S\neq 1\}$, which has probability $o(1)$ by Proposition~\ref{prop:restatement-sat-threshold}.
Combining these bounds completes the proof.

<!-- proposition-11: body_md -->
\label{prop:beta-RS-Holevo}
For every $k\geq2$ and $D>k$,
\begin{equation}
\label{eq:beta-RS-Holevo-order}
 \beta_\textnormal{RSB}(k,D)<\beta_{\rm ann}(k,D)
 =\beta_{\rm Holevo}(k,D),
\end{equation}
where $\beta_{\rm Holevo}$ is defined in \eqref{eq:beta-Holevo-app}.
<!-- end proposition-11 -->
<!-- proof-24: body_md -->
From the calculation in \cref{sec:RS}, we see that the annealed free energy (per spin) for the Erd\H{o}s-R\'enyi ensemble in \cref{def:erdos_renyi} is 
\[
 a(\beta)=\ln2+\frac{D}{k}\ln\cosh\beta.
\]
Let $f_N(\beta) = \frac{1}{N}\mathbb{E} \ln Z_{\beta, H}$ be the quenched free energy. By definition of $\beta_\textnormal{RSB}$, the quenched and annealed free energies agree throughout $\beta\in(0,\beta_\textnormal{RSB}(k, D))$. In this interval, the convex function $f_N(\beta)$ converges to the differentiable function $a(\beta)$, implying $f'_N(\beta)\to a'(\beta)$.
So for $\beta  < \beta_\textnormal{RSB}$, the expected Gibbs entropy per spin thus converge to
\[
\lim_{N\to\infty} (f_N(\beta)-\beta f'_N(\beta)) = a(\beta)-\beta a'(\beta)
                                                          
=\ln2-\frac{D}{k}\tau(\beta).
\]
Note $\beta_{\rm ann}(k, D)$ is precisely the point where this entropy vanishes.

We next use the existence of isolated vertices in a random Erd\H{o}s-R\'enyi hypergraph to show $\beta_\textnormal{RSB}(k, D) < \beta_{\rm ann}(k, D)$. In such a random hypergraph, the expected number of isolated vertices is $N(1-k/N)^M = N(e^{-D} + o_N(1))$. Since an isolated vertice does not appear in the Hamiltonian, its spin is independent and uniformly random under the Gibbs measure at all temperatures. Thus the Gibbs entropy per spin at any temperature is at least $e^{-D}\ln 2$, and so for all $\beta \in (0, \beta_\textnormal{RSB})$ we must have
\begin{equation}
\ln2-\frac{D}{k}\tau(\beta) \ge e^{- D}\ln 2.
\end{equation}
Observe that there is a unique $\beta_{\rm iso} > 0$ such that $\frac{D}{k} \tau(\beta_{\rm iso}) = (1-e^{-D})\ln 2$. Monotonicity of $\tau(\beta)$ then implies that $\beta_\textnormal{RSB} \le \beta_{\rm iso} < \beta_{\rm ann}$.

It remains to show that $\beta_{\rm ann} = \beta_{\rm Holevo}$.  If
$p=H_2^{-1}(1-k/D)\in[0,1/2]$ and
$\beta=\operatorname{sech}^{-1}(2\sqrt{p(1-p)})$, then
$p=(1-\tanh\beta)/2$.  The elementary entropy identity
\[
 H_2\left(\frac{1-\tanh\beta}{2}\right) \cdot \ln2
 =\ln2+\ln\cosh\beta-\beta\tanh\beta
\]
shows that $H_2(p)=1-k/D$ is equivalent to \eqref{eq:annealed-entropy-zero}.  This is exactly the formula for $\beta_{\rm Holevo}$ in \cref{app:holevo_shannon}.
