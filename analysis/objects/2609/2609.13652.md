\label{BRHMHF-def-1.5}
 (1) $\partial\Omega$ is said to satisfy the \emph{Wiener criterion} if for each $x_0\in \partial\Omega$ it holds
 \begin{equation*}\label{equ:Wiener-criterion}
\int_0^1\frac{{\rm cap}(B_r(x_0)\setminus \Omega, B_{2r}(x_0))}{{\rm cap}(B_r(x_0), B_{2r}(x_0))}\frac{{\rm d}r}{r}=+\infty.
\end{equation*} 
 (2)  $\Omega$ is said to satisfy a  {\emph{uniformly exterior density condition with radius $R_{\rm den}\in(0,1)$}} if there exists  $c_0>0$ such that
\begin{equation*} 
\frac{\mathfrak{m}(B_r(x_0)\setminus\Omega)}{\mathfrak{m}(B_r(x_0))}\geqslant c_0,\quad \forall x_0\in\partial\Omega,\ \ \forall r\in(0,R_{\rm den}).
\end{equation*}

(3)  $\Omega$ is said to satisfy a  {\emph{uniformly exterior ball condition with radius $R_{\rm ext}\in(0,1)$}} if for each $x_0\in \partial\Omega$ there exists a point $y_0\in X\setminus\Omega$ such that
\begin{equation*}\label{BRHMHF-equ-1.2}
d(y_0,x_0)=R_{\rm ext}\quad {\rm and}\quad \overline{B_{R_{\rm ext}}(y_0)}\cap \Omega=\emptyset.
\end{equation*}

(4)   $\partial\Omega$ is said to be  {\emph{perimetrically regular}}  if it is a set of finite perimeter and  there exists a constant  $C_0>0$ such that
\begin{equation*}\label{BRHMHF-equ-1-3}
Per_{\Omega}(B_r(x_0))\leqslant C_0\frac{\mathfrak{m}(B_r(x_0))}{r},\qquad \forall x_0\in\partial\Omega,\ \ \forall r\in (0,{\rm diam}(\Omega)).
\end{equation*}
See the definition of the capacity ${\rm cap}$ in Section \ref{sect3} and the perimeter $Per_\Omega$ in Section \ref{sect-5}.

<!-- theorem-1: body_md -->
\label{thm:main-BRHMHF}
Let $\Omega$ be a bounded domain in an $RCD(K,N)$ space $(X,d,\mathfrak{m})$, $K\in \mathbb R$, $N\in(1,+\infty)$, and let $(Y,d_Y)$ be a $CAT(0)$ space. Given any $\psi\in W^{1,2}(\Omega,Y)$, let  $u(x,t):\Omega\times(0,+\infty)\to Y$ be a weak solution of the harmonic map heat flow in $W^{1,2}_{\psi}(\Omega,Y)$,  with initial value $u_0$. Suppose that the image of $u_0$ is bounded in $Y$. Then the following statements hold.
 \begin{itemize}
 \item[(1)] If $\partial\Omega$ satisfies the Wiener criterrion, and   $\psi\in C(\overline\Omega,Y)$, then $u(x,t)$ is in $C(\overline\Omega\times(0,+\infty),Y)$.
 \item[(2)] If $\Omega$ has a uniformly exerior density condition, and   $\psi\in C^\alpha(\overline\Omega,Y)$ for some $\alpha\in(0,1)$, then $u(x,t)$ is in $C^\beta(\overline\Omega\times(0,+\infty),Y)$ for some $\beta\in(0,\alpha)$.
\item[(3)] If $\Omega$ has a uniformly exerior ball condition, and   $\psi\in Lip(\overline\Omega,Y)$, then $u(x,t)$ is in $C^{1-\epsilon}(\overline\Omega\times(0,+\infty),Y)$ for any $\epsilon\in(0,1)$.
\item[(4)] If $\Omega$ has a uniformly exerior ball condition with radius $R_{\rm ext}$ and the boundary $\partial\Omega$ is perimetrically regular, and   $\psi\in Lip(\overline\Omega,Y)$ with a Lipschitz constant $L_\psi$.  Let $x_0\in \partial\Omega$ and $R:=\min\{{\rm diam}(\Omega)/100,  R^2_{\rm ext}/(4L_\psi)\}$. Then for any $T\geqslant R^2_{\rm ext}$ the following Zygmund-type estimate holds
   
  \begin{equation}\label{equ-Zygmund-1}
  	\frac{d_Y(u(x,T+t),\psi(x_0)) }{d(x,\partial\Omega)}\leqslant c_T \ln\left(\frac{e\cdot {\rm diam}(\Omega)}{d(x,\partial\Omega)}\right)
  \end{equation} 
for any $t>0$ and any $x\in\Omega\cap B_{R/4}(x_0)$ with $d(x,x_0)\leqslant 2 d(x,\partial\Omega)$, where  the constant $c_T$ depends only on $N,K,\Omega,M, L_\psi$ and $T$.
 \end{itemize}

<!-- remark-1: body_md -->
\label{rem:boundary-regularity-hmhf}
 (1)   The exterior ball condition in Theorem \ref{thm:main-BRHMHF} (3) is nearly indispensable, as illustrated in the following simple examples. Let $a\in(1/2,1)$ and consider the cone
 $$C_a:=\{(\rho,\theta) |\ 0<\rho<+\infty,\ 0<\theta<\pi/a\} $$
with  the vertex $O=(0,0)$, where $(\rho, \theta)$ are the polar coordinates in $\mathbb R^2$. Then the domain $\Omega_a:=C_a\cap B_1(O)$ satisfies only an exterior cone condition at  $O$ with the cone angle $(2-1/a)\pi<\pi$. A direct calculation shows that the function $f_a:=\rho^{a}\sin(a\theta)$ is harmonic on $\Omega_a$ and has Lipschitz boundary data (indeed $f_a(x)=0$ for any $x\in\partial C_a\cap B_1(O)$). Obviously, it is not Lipschitz near the vertex $O$. 

(2)  The Zygmund-type estimate in (\ref{equ-Zygmund-1}) is optimal, even for a smooth domain in the Euclidean setting $X=\mathbb R^n$. This can be understood by the following Hardy-Littlewood's example.  
Let  $B^2_{1}(0)=\{(r,\theta)|\ 0\leqslant r<1, \theta\in(-\pi,\pi]  \}$ be the unit ball in $ \mathbb R^2$, and let $f_0$ be defined on $S_1=\partial B^2_1(0)$ by
$$f_0(\theta)=|\sin\theta|  \  \ {\rm if}\ \  |\theta|\leqslant \pi/2 \qquad {\rm and}\qquad   f_0(\theta)=1\ \ {\rm otherwise}.$$
Then the harmonic function (a static solution of the heat equation) on $B^2_{1}(0)$ with the data  $f_0$ is uniquely determined by the Poisson integral formula,
$$f(r,\theta_0)= \frac{1-r^2}{2\pi}\int_{ S^1} \frac{ f_0(\theta)}{|(r,\theta_0),(1,\theta)|_{\mathbb R^2}^2 }{\rm d}\theta = \frac{1-r^2}{2\pi}\int_{S^1} \frac{f_0(\theta)}{ r^2+1-2r\cos(\theta-\theta_0)   }{\rm d}\theta.$$
 By direct calculation, and noticing $ d((r,0),\partial B_1^2(0))=1-r$, we have
 \begin{equation*}
 \begin{split}
2\pi\cdot  f(r, 0) &= (1-r^2)\int_{-\pi}^\pi\frac{ |\sin\theta| }{r^2+1-2r\cos \theta }{\rm d}\theta=2(1-r^2)\int_{0}^\pi\frac{  \sin\theta  }{r^2+1-2r\cos \theta }{\rm d}\theta\\
   & =2\frac{(1-r^2)}{r}\ln\left(\frac{1+r}{1-r}\right)\geqslant (1-r)\ln(\frac{1}{1-r}).
    \end{split}
 \end{equation*}

\label{BRHMHF-def-2.4}
 Given a function $f\in W^{1,2}_{\rm loc}(\Omega)$,  its measure-valued  Laplacian $\mathbf\Delta  f$ is defined as a (linear) functional

\end{equation*}

<!-- defn-3: body_md -->
\label{BRHMHF-def-3.1}
Let $0<T\leqslant\infty$ and let $\Omega_T:=\Omega\times(0,T)$. A function $u\in L^2_{\rm loc}\big((0,T); W^{1,2}_{\rm loc}(\Omega)\big)$ is called a {\emph{super-solution of the heat equation}}, if for every open set $U\Subset\Omega_T$ and for all non-negative function $\phi(x,t)\in Lip(\Omega_T)$ with ${\rm supp}(\phi)\subset U$, we have
$$\int_U u\frac{\partial\phi}{\partial t}{\rm d}\mathfrak{m} {\rm d}t\leqslant \int_U\left<{\nabla u},{\nabla \phi}\right>
	{\rm d}\mathfrak{m} {\rm d}t.$$
A function $u$ is called a {\emph{sub-solution of the heat equation}} if $-u$ is a super-solution of the heat equation.

\label{lem:barrier-elliptic}

<!-- proof-3: body_md -->
Let $\widetilde B_R$ be given in Lemma \ref{lem:good-neighborhood}, and set 
$$V_R(x_0):=\Omega\cap \widetilde B_R.$$

$$d\big(B_{R/2}(x_0)\cap\Omega, \partial V_R(x_0)\cap \Omega\big)\geqslant d\big(B_{R/2}(x_0), \partial \widetilde B_R\big) \geqslant R/2.$$

It is clear that  $w\in C(\overline{V_R(x_0)})$, $\overline w(x)\geqslant |w(x)|$ for all  $x\in \overline{V_R(x_0)}$, $\overline w(x)= |w(x)|$ for all  $x\in \partial\Omega\cap B_{R/2}(x_0)$, 
and $\overline w(x)=\max_{\overline{V_R(x_0)}}|w|$ for all $x\in \partial V_R(x_0)\cap \Omega.$
 Now, the solution of the Dirichlet problem

<!-- theorem-2: body_md -->
\label{thm:comparison-heat}
	Let $\Omega\subset X$ be a bounded open domain satisfying the Wiener criterion,   and let $v(x,t)\in  L^2_{\rm loc}\big((0,+\infty); W^{1,2}(\Omega)\big)$ be a sub-solution of the heat equation. $v\geqslant0$ and is continuous in $\Omega\times(0,+\infty)$. Suppose that $v(x,t)\leqslant M$ on $\Omega\times(0,+\infty)$.  Let $w\in W^{1,2}(X)\cap C(\overline\Omega)$  and suppose   $\big(v(x,t)-w(x)\big)^+\in W^{1,2}_0(\Omega)$ for almost all $t\in(0,+\infty)$. Take any $x_0\in \partial\Omega$ and any $R\in(0,{\rm diam}(\Omega)/100)$. Let $V_R\subset \Omega$ and the torsion function $h$ be given in Lemma \ref{lem:barrier-elliptic} (or Lemma \ref{lem:barrier-elliptic-2}). 
			 		Then
			  	$$v(x, T)\leqslant \left(\frac{ M  }{\max_{\overline{V_R(x_0)}}|w|}+1\right)h(x), \qquad \forall x\in  B_R(x_0)\cap \Omega,\quad \forall T> 2\max_{\overline{V_R(x_0)}}|w|.$$
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
If $\max_{\overline{V_R(x_0)}}|w|=0$, we have done. So we can assume $\max_{\overline{V_R(x_0)}}|w|>0$. We put $V_R:=V_R(x_0)$ and $L_R:=\max_{\overline{V_R(x_0)}}|w|$ in the following proof. Fix any $T>2L_R$ and define the function
\begin{equation}
\label{equ:barrier-parabolic}\tilde h(x,t):=\frac{h(x)}{L_R}+ \frac{(T-t)^2}{(2L_R)^2}.
\end{equation}
It is a super-solution of the heat equation on $V_R\times (T-2L_R, T)$.	 Indeed, we have that $\tilde h\in W^{1,2}\big(V_R\times (0,T)\big)$ and that for each $t>0$,
	$${\bf \Delta}\tilde h(x,t)=\frac{{\bf \Delta}h}{L_R}= - \frac{\mathfrak{m}}{L_R}	$$ on $V_R$ in the sense of distributions. For each $t\in(T-2L_R,T)$,
	$$\frac{\partial}{\partial t}\tilde h(x,t)= \frac{2(t-T)}{(2L_R)^2}\geqslant \frac{2(T-2L_R-T)}{(2L_R)^2}=-\frac{1}{L_R}.$$
	It follows for each $t\in(T-2L_R,T)$ that ${\bf \Delta}\tilde h\leqslant\frac{\partial}{\partial t}\tilde h\cdot \mathfrak{m}$ in the sense of distributions. Therefore, it is a super-solution of the heat equation (see, for example,  \cite[Lemma 6.12]{ZZ18}).

 
		We first claim that 
		$$v\leqslant  (M+L_R)\tilde h  \quad {\rm on}\ \  \big(\partial V_{R}  \times (T-2L_R,T)\big)\cup \big(V_R\times\{T-2L_R\}\big)$$  in the sense of Lemma \ref{BRHMHF-lem-3.2}.
		 On $\partial V_R\times (T-2L_R,T)$, we have 
	 $$(M+L_R)\tilde h \geqslant\frac{(M+L_R)h}{L_R} \geqslant h\geqslant|w|\geqslant   v,$$
	 	 	On $V_R\times(T-2L_R,T-2L_R+\varepsilon)$   for any small $\varepsilon>0$, we have
$$\tilde h(x,t)= \frac{h(x)}{ L_R}+ \frac{(t-T)^2}{(2L_R)^2} \geqslant 1-\varepsilon/L_R$$
 By the assumption $v\leqslant M $ on $\Omega\times (0,+\infty)$, we get
	$$(v-a\tilde h)_+\leqslant a \left(1-(1-\varepsilon/L_R\right),\quad {\rm where}\quad a:=M+L_R,$$
	 which implies the initial condition
	$$\frac 1 \varepsilon\int_{T-2L_R}^{T-2L_R+\varepsilon}\int_{V_R}(v-a\tilde h)_+^2{\rm d}\mathfrak{m}{\rm d}t\to 0\quad {\rm as}\  \ \varepsilon\to0^+.$$  Now we finish the proof of the claim that $v\leqslant a\tilde h$ near the parabolic boundary of $V_R\times (T-2L_R,T)$ in the sense of Lemma \ref{BRHMHF-lem-3.2}.
	
	By Lemma \ref{BRHMHF-lem-3.2}, we conclude that $v\leqslant a\tilde h$ almost all in $V_R\times (T-2  L_R,T)$. Since both $v$ and $\tilde h$ are continuous on $U\times(0,T]$, we have
	$$v(x,T)\leqslant a\tilde h(x, T) =\frac{ah(x)}{L_R} \quad {\rm on} \ \ V_R.$$
 This finishes the proof, since $\Omega\cap B_R(x_0)\subset V_R.$

<!-- lemma-6: body_md -->
\label{lem:conrant-hilbert}
	Let $\Omega$ satisfy the exterior ball condition at $x_0\in \partial\Omega$ with a radius $R_{\rm ext}$. Let $v(x)\in C(\overline\Omega)\cap W^{1,2}(\Omega)$ solve 
	$${\bf \Delta}v\geqslant-1\quad {\rm on}\ \ \Omega$$
	in the sense of distributions. Suppose that $v(x)\geqslant0$ on $\Omega$  and  $v(x)=0$ on $\partial\Omega$. Then there exists  aconstant $R_1$, depending only on $N,K, {\rm diam}(\Omega)$ and $R_{\rm ext}$, such that for each $R\in(0,R_1)$, 
	$$v(x)\leqslant  C_1  d(x,x_0)  \cdot \left(\frac{M(R)}{R}+\sqrt{M(R)}\right),\quad\forall\ x\in B_{R}(x_0)\cap \Omega,$$
	where $M(R):= \sup_{B_R(x_0)\cap \Omega}v(x),$ and the constant $C_1$ depends only on $N,K$.
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
Let $R_1:= \min\{R_{\rm ext},1\}/8.$  For any $R<R_1$, the exterior ball condition gives  a ball $B_R(y_0)$ such that 
$$B_R(y_0)\cap \Omega=\emptyset,\quad  d(y_0,x_0)=R.$$ According the Laplacian comparison theorem on $RCD(K,N)$ space, there exists a number $m_{N,K}>0$ such that 
$${\bf \Delta}d^{-m}_{y_0}\geqslant \frac{m^2}{R^{m+2}}\cdot \mathfrak{m},\quad {\rm on}\ X\setminus\{y_0\}$$
for any $m\geqslant m_{N,K}$ and any $R<1$, in the sense of distributions (see \cite{Gig15} and \cite[Lemma 2.5]{ZZ24}).
Take $$m:= m_{N,K}+\frac{R}{\sqrt{M(R)}}.$$
Here we have assumed $M(R)>0$; otherwise, we are done.

 

We consider the function 
$$h(x):=A\big(R^{-m}-d_{y_0}^{-m}(x)\big),\quad {\rm where}\quad A:=2R^m\cdot M(R),$$
on $X\setminus\{y_0\}$. Then we have 
$${\bf \Delta}h\leqslant -A\frac{m^2}{2R^{m+2}}=- \frac{m^2\cdot M(R)}{R^2} \leqslant -1 $$
on $B_{3R}(y_0)\setminus B_{R/2}(y_0),$ in the sense of distributions, since $m^2 \geqslant R^2/M(R)$. The exterior ball condition implies 
$h\geqslant 0=v$ on $\partial \Omega$. A direction computation gives  
$$h\geqslant A(R^{-m}-(2R)^{-m})\geqslant AR^{-m}/2=M(R)\geqslant v\ \ {\rm on}\ \ \partial B_{2R}(y_0)\cap \Omega.$$
Therefore, $h\geqslant v$ on the boundary of $\Omega\cap B_{2R}(y_0)$. Combining with 
$${\bf \Delta}v\geqslant -1,\quad {\bf \Delta}h\leqslant -1,$$
on $\Omega\cap B_{2R}(y_0)$ in the sense of distributions. The maximum principle implies
$$v\leqslant h,\quad {\rm on}\ \Omega\cap B_{2R}(y_0),$$
which implies the desired estimate, since $h(x_0)=0$,
$$|\nabla h|\leqslant mAR^{-(m+1)}\leqslant 2mM(R)/R=2 \left(m_{N,K}\frac{M(R)}{R}+ \sqrt{M(R)}\right) $$
on $\Omega\cap B_{2R}(y_0),$ and $\Omega\cap B_R(x_0) \subset\Omega\cap B_{2R}(y_0).$

<!-- lemma-7: body_md -->
\label{lem:optimal-regularity-for-torsion}
Let $U\subset \Omega$ satisfy a uniformly exterior ball condition, and   $w\in Lip(\overline U)$. Suppose that  $u\in W^{1,2}(U)$ satifies 
$${\bf\Delta}u=-1,\quad u-w\in W_0^{1,2}(U)$$
 in the sense of distributions. Then $u \in C^{1-\epsilon}(\overline U)$ for any $\epsilon>0.$
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
Let $u_1$ be the harmonic function on $U$ with boundary data $w$, and let $u_2$ be the torsion function with zero boundary data, namely, $u_2$ is the solution of 
$${\bf \Delta}u_2=-1,\quad u|_{\partial\Omega}=0,$$
 in the sense of distributions. The maximum principle implies $u_2\geqslant0$. Lemma \ref{lem:Wiener-criterion} states $u_2\in C(\overline U)$, in particular, it is bounded on $\overline U$. By using Lemma \ref{lem:conrant-hilbert}, we have 
 \begin{equation}\label{equ:optimal-regularity-torsion-1}
 	|u_2(x)-u_2(x_0)|\leqslant c_1\cdot d(x,x_0) \cdot \max_{\overline U}u_2
 	 \end{equation}
 for all $x\in U$ and $x_0\in \partial U$, where the constant $c_1$ depends on $N,K,{\rm diam}(\Omega)$ and $ R_{\rm ext}$.   
 
 On the other hand, from \cite[Theorem 1.1]{ZZ24}, we get that $u_1\in C^{1-\epsilon}(\overline U)$ for any $\epsilon>0$. Combining this with (\ref{equ:optimal-regularity-torsion-1}), we conclude that 
 $u\in C^{1-\epsilon}(\overline U).$
The proof is finished.

<!-- lemma-8: body_md -->
\label{BRHMHF-thm-1.10}
Let $\Omega$ be bounded and satisfy a uniform exterior ball condition with radius $R_{\rm ext}\in(0,{\rm diam}(\Omega)/100)$. Let $  T\geqslant R^2_{\rm ext}$.  Then  for any $\varepsilon\in(0,T)$ we have
	\begin{equation}\label{BRHMHF-equ-3.12}
			\int_\varepsilon^T|\nabla p^\Omega_t(x,y) |{\rm d}t  \leqslant  c'_T \cdot \delta(x)  \left(  \int^{{\rm diam}^2(\Omega)}_{d^2(x,y)} \frac{    {\rm d}t }{t\cdot \mathfrak{m}(B_{ \sqrt{t}}(x))}+1\right)
	\end{equation}	
 for any $x\in \Omega$ and  almost all  $y\in \Omega$ such that $d(y,\partial\Omega)\leqslant \sqrt{\varepsilon}$,
where the constant $ c_T'$ is independent of $\varepsilon$, (they depend only on $T, N,K,\Omega$ and $ R_{\rm ext}$).
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
It suffices to show  that   (\ref{BRHMHF-equ-3.12})  holds almost eveywhere in  $ B_{\delta(y_0)/2}(y_0)$ for any  $y_0\in \Omega$. We set $R:=\delta(y_0)/2.$

Recall that $p^\Omega_t(x,\cdot)\in Lip_{\rm loc}(\Omega\times (0,T))$ (by \cite[Corollary 1.5]{HZ20}).	Letting $f(\cdot,t):=\ln p^\Omega_t(x,\cdot)$, by local Li-Yau's gradient estimate \cite[Theorem 1.4]{ZZ16}, we have   that
	$$|\nabla f|^2(y,t)-2\frac{\partial f}{\partial t}(y,t)\leqslant C_{K,N}\left(\frac{1}{R^2}+\frac{1}{t}+1\right)$$
	for almost all $y\in B_R(y_0)$ and almost all $t\in(0,\infty)$. Therefore,  for $t>\varepsilon,$
	$$\frac{|\nabla p^\Omega_t(x,\cdot)|^2(y)}{p^\Omega_t(x,y)}\leqslant 2\frac{\partial}{\partial t} p^\Omega_t(x,y)+C_{N,K} \left(\frac{1}{R^2}+\frac{1}{\varepsilon}+1\right)p^\Omega_t(x,y)$$	
	for almost all $y\in B_R(y_0)$. Integrating over $[\varepsilon,T]$ and by H\"older inequality, we get
	\begin{equation*}
		\begin{split}
			\left(\int_{\varepsilon}^{T}|\nabla p^\Omega_t(x,\cdot)|(y){\rm d}t\right)^2& \leqslant  \int_{\varepsilon}^{T}\frac{|\nabla p^\Omega_t(x,\cdot)|^2(y)}{p^\Omega_t(x,y)}{\rm d}t\cdot \int_{\varepsilon}^{T}  p^\Omega_t(x,y){\rm d}t\\
			&\leqslant \left[2\left(p^\Omega_{T} -p^\Omega_{\varepsilon} \right)+C_{N,K}\left(\frac{1}{R^2}+\frac{1}{\varepsilon}+1\right)\int_{\varepsilon}^{T}  p^\Omega_t(x,y){\rm d}t\right]\cdot \int_{\varepsilon}^{T}  p^\Omega_t(x,y){\rm d}t	\end{split}
	\end{equation*}
for almost all $y\in B_R(y_0)$. By  $p^\Omega_\varepsilon(x,y)\geqslant0$ and $\varepsilon\geqslant 4 R^2$ (since $2R=\delta(y_0)\leqslant \sqrt\varepsilon$), we get
		\begin{equation}\label{BRHMHF-equ-3.15}
		\begin{split}
			\int_{\varepsilon}^{T}|\nabla p^\Omega_t(x,\cdot)|(y){\rm d}t&\leqslant C_1 \left(R\cdot p^\Omega_{T} (x,y) +  \frac{1}{R } \int_{\varepsilon}^{T}  p^\Omega_t(x,y){\rm d}t\right)   		\end{split}
			\end{equation}	
for almost all $y\in B_R(y_0)$.

By \cite[Theorem 3.9]{ZZ25} and $\delta(y)\leqslant \delta(y_0)+d(y,y_0)\leqslant 2\delta(y_0)$, we have 
$$\frac 1 R\int_\varepsilon^Tp^\Omega_t(x,y){\rm d}t\leqslant \frac{2}{\delta(y_0)}\int_0^\infty p^\Omega_t(x,y){\rm d}t\leqslant 4 c_3 \cdot \delta(x)  \left(  \int^{D^2}_{d^2(x,y)} \frac{ {\rm d}t }{t\cdot \mathfrak{m}(B_{ \sqrt{t}}(x))}+1\right),$$
where $D:={\rm diam}(\Omega)$.
From (\ref{BRHMHF-equ-1.6}) and $\delta(y)\leqslant {\rm diam}(\Omega)$, we have
\begin{equation}
	\label{equ:upper-heat-1}
	p^\Omega_T(x,y)\leqslant \frac{\delta(x) }{T}\frac{c_T\cdot {\rm diam}(\Omega)}{\mu(B_{\sqrt T}(y))}\leqslant c_T'\cdot \delta(x).
\end{equation}
Substituting the above two inequalities into (\ref{BRHMHF-equ-3.15}), the desired estimate (\ref{BRHMHF-equ-3.12}) follows. The proof is finished.

<!-- lemma-9: body_md -->
\label{BRHMHF-lem5.4}
	Let $\Omega$ be a bounded open set of finite perimeter. Assume that $$\liminf_{r\to0}\frac{\mathfrak{m}\big(B_r(x_0)\setminus\Omega \big)}{\mathfrak{m}\big(B_r(x_0)\big)} \geqslant \gamma,\quad \forall x_0\in \partial\Omega,$$
  for some $\gamma\in (0,1)$. 	
	 Then there is a sequence $\phi_k\in Lip_0(\Omega)$, $0\leqslant \phi_k\leqslant 1$, such that   $\phi_k(x)\to \chi_\Omega$ in $L^1(X)$, as $k\to\infty$,  and
	\begin{equation}
		\label{BRHMHF-equ-4.9}
	\limsup_{k\to\infty}\int_X\eta|\nabla \phi_k|{\rm d}\mathfrak{m} \leqslant \frac{c}{\gamma}\cdot\int_X\eta {\rm d}Per_{\Omega} , \quad \forall \eta\in C(X), \quad \eta\geqslant0,
	 \end{equation}
	 	where the constant $c$ depends only on $N,K$ and ${\rm diam}(\Omega)$.
<!-- end lemma-9 -->
<!-- proof-11: body_md -->
It was proved in \cite[Lemma 4.12]{ZZ25} that (\ref{BRHMHF-equ-4.9}) holds for any nonnegative function $\eta\in Lip_0(X)$.
	
	For a general $\eta\in C_0(X)$, $\eta\geqslant0$, given any $\epsilon>0$, there exists a function $\eta_\epsilon\in Lip_0(X)$, $\eta_\epsilon\geqslant0$,  such that  
	$$\eta_\epsilon\geqslant \eta-\epsilon,\qquad \int_X\eta_\epsilon {\rm d}Per_\Omega\leqslant \int_X\eta{\rm d}Per_\Omega+\epsilon.$$
	By applying (\ref{BRHMHF-equ-4.9}) to $\eta_\epsilon+\epsilon$, we conclude that 
	$$\limsup_{k\to\infty}\int_X\eta|\nabla \phi_k|{\rm d}\mathfrak{m} \leqslant \limsup_{k\to\infty}\int_X(\eta_\epsilon+\epsilon)|\nabla \phi_k|{\rm d}\mathfrak{m}\leqslant \frac{c}{\gamma}\cdot\left(\int_X\eta {\rm d}Per_{\Omega}+\epsilon+\epsilon Per_\Omega(X)\right).$$	
Letting $\epsilon\to 0^+$, we finished the proof.

 \begin{equation}\label{equ:Duhamel-2}

$$    \int_\Omega p_\varepsilon^\Omega(x,y)f(y,T_1+T){\rm d}\mathfrak{m}(y)-\int_\Omega p_T^\Omega(x,y)f(y, T_1+\varepsilon){\rm d}\mathfrak{m}(y) 	\leqslant   A_1 \delta(x)\int_{\partial\Omega}F_x(y)|w(y)|{\rm d}Per_\Omega(y)+B_1\cdot \delta(x),$$

Letting $\varepsilon\to0$ and using the upper bound of heat kernel (\ref{equ:upper-heat-1}), we obtain
$$f(x,T_1+T)\leqslant c'_{R,T}\delta(x)\left(\int_\Omega f(y,T_1){\rm d}\mathfrak{m}(y)+ \int_{\partial\Omega}F_x(y)|w(y)|{\rm d}Per_\Omega(y)+1\right),$$	
which is the desired estimate.

\label{cor:coro5.7-BRHMHF} Let $\Omega$ be a bounded open set of finite perimeter. Suppose that $\Omega$ satisfies a uniform exterior ball condition with $R_{\rm ext}>0$, and that $\partial\Omega$ is perimetrically regular with constant $C_0$. Let $f$ and $w$ be given in Theorem \ref{BRHMHF-thm-5.3}. Suppose that $w\in Lip(\overline\Omega)$ with a Lipschitz constant $L_w>0$. Let $x_0\in \partial\Omega$ and suppose $w(x_0)=0$.
    Then for any $T\geqslant R^2_{\rm ext}$  we have    
    $$\frac{f(x,T_1+T)}{\delta(x)}\leqslant c_T \ln\left(\frac{e\cdot {\rm diam}(\Omega)}{\delta(x)}\right)$$
for all $x\in\Omega\cap B_{R/4}(x_0)$ with $d(x,x_0)\leqslant 2\delta(x)$ and all $T_1>0$, where $R:=\min\{{\rm diam}(\Omega)/100,  R^2_{\rm ext}/(4L_w)\}$ and the constant $c_T$ depends only on $N,K,\Omega,M, L_w$ and $T$.

The fact $w(x_0)=0$ and $w\in Lip(\partial\Omega)$ with Lipschitz constant $L_w$ implies
	$$|w(x)|\leqslant L_wd(x,x_0),\qquad \forall x\in\Omega,$$
Since $$\max_{B_{2R}(x_0)\cap \Omega}|w|\leqslant 2RL_w\leqslant R^2_{\rm ext}/2.$$
	by combining this with \cite[Lemma 5.4]{ZZ25} and Theorem \ref{BRHMHF-thm-5.3}, we get 
 $$\frac{f(x,T_1+T)}{\delta(x)}\leqslant c_{R,T} \int_\Omega f(y,T_1){\rm d}\mathfrak{m}(y)+c_{R,T} \left[C\ln\left(\frac{{\rm diam}(\Omega)}{\delta(x)}\right)+\int_{\partial\Omega} |w(y)|{\rm d}Per_\Omega(y)+1\right]$$

for all $x\in\Omega\cap B_{R/4}(x_0)$ with $d(x,x_0)\leqslant 2\delta(x)$ and $T_1>0$.

<!-- lemma-10: body_md -->
\label{lem:BRHMHF-lem6.1}
Suppose that $d_Y(P_0,u_0(x))\leqslant M$ for some $P_0\in Y$ and $M>0$. Then the following hold:
\begin{itemize}
\item[(1)] $d_Y(P_0,u(x,t))\leqslant M$ for any $t>0$,
\item[(2)] $d_Y(Q,u(x,t))$ is a sub-solution of the heat equation on $\Omega\times(0,+\infty)$, for any $Q\in Y$, and 
\item[(3)] $u$ is loally continuous on $\Omega\times(0,+\infty)$.
\end{itemize}

<!-- proof-15: body_md -->
Take any $x_0\in \partial\Omega$ and let $w_{x_0}(x):=d_Y(w(x),w(x_0)).$ It is clear that if $w\in C(\overline\Omega,Y)$ (or $C^\gamma(\overline\Omega,Y)$, $Lip(\overline\Omega,Y)$) then $w_{x_0}\in C(\overline\Omega)$ (resp. $C^\gamma(\overline\Omega)$, $Lip(\overline\Omega)$). Moreover, the modulus of continuity of $w_{x_0}$  (or the H\"older norm, the Lipschitz constant $L_{w_{x_0}}$) is not great than the modulus of continuity of $w$ (resp. the H\"older norm, the Lipschitz constant $L_{w}$).


 The triangle inequality implies
$$d_Y(u(x,t),w(x_0))-w_{x_0}(x)\leqslant d_Y(u(x,t),w(x))$$
for any $(x,t)\in\Omega\times(0,+\infty)$. Hence $\big(d_Y(u(x,t),w(x_0))\big)^+\in W^{1,2}_0(\Omega)$ for almost all $t\in(0,+\infty)$. 



The items (1), (2) and (3) follow from the combination of Lemma \ref{lem:BRHMHF-lem6.1}, Theorem \ref{thm:comparison-heat}, and the regularity of the torsion function $h$ in Lemma \ref{lem:barrier-elliptic}, Lemma \ref{lem:barrier-elliptic-2} and Lemma \ref{lem:optimal-regularity-for-torsion}, respectively.

Let  $R:=\min\{{\rm diam}(\Omega)/100,  R^2_{\rm ext}/(4L_w)\}$. For any $x\in \Omega\cap B_{R/4}(x_0)$, since the function 
$$(x,t)\mapsto d_Y(u(x,t),w(x_0))$$
is a sub-solution of the heat equation with the boundary data $w_{x_0}(x)$. The triangle inequality yields $\partial_t d_Y(u(x,t),w(x_0))\in L^\infty_{\rm loc}(\Omega\times(0,+\infty)$ (because $u$ is locally Lipschitz continuous in time). Corollary \ref{cor:coro5.7-BRHMHF} implies
$$\frac{d_Y(u(x,T+T_1),w(x_0))}{\delta(x)}\leqslant  c_T \ln\left(\frac{e\cdot {\rm diam}(\Omega)}{\delta(x)}\right)$$
for any  $x\in\Omega\cap B_{R/4}(x_0)$ with $d(x,x_0)\leqslant 2\delta(x)$ and any $T_1>0$.
