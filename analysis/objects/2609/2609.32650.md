\label{theorem: hypercyclic Hardy-Toeplitz operators}
		Let  $a \in \mathbb{C} \backslash \{ 0 \}$, $p$  be a bounded analytic function on $\mathbb{D}$, and $\varphi (z) = a \overline{z} + p(z)$. Let $\widetilde{\varphi} (z) = \frac{a}{z} + p (z)$.
		\begin{enumerate}[label = \textup{(\alph*)}]
			\item If $\mathbf{T}_{\varphi}$ is hypercyclic on the Hardy space, then
			\begin{enumerate}[label = \textup{(\arabic*)}]
				\item the function $\widetilde{\varphi}$ is univalent in $\mathbb{D}_0$;
				\item $\overline{\mathbb{D}} \cap [\mathbb{C} \backslash \widetilde{\varphi} (\mathbb{D}_0)] \neq \varnothing$ and $\widehat{\mathbb{D}} \cap [\mathbb{C} \backslash \widetilde{\varphi} (\mathbb{D}_0)] \neq \varnothing$.
			\end{enumerate}
			\item Assume in addition that $p$ is continuous on the closed disk  $\overline{\mathbb{D}}$ and that
			\begin{enumerate}[label = \textup{(\arabic*$'$)}]
				\item the function $\widetilde{\varphi}$ is univalent in $\overline{\mathbb{D}}_0$;
				\item $\mathbb{D} \cap [\mathbb{C} \backslash \widetilde{\varphi} (\mathbb{D}_0)] \neq \varnothing$ and $\widehat{\mathbb{D}} \cap [\mathbb{C} \backslash \widetilde{\varphi} (\mathbb{D}_0)] \neq \varnothing$.

\label{lemma: Toeplitz operators with monomial symbols}

		\begin{align*}
			T_{\overline{z}^ k} e_j =
			\begin{cases}
				0, & j < k, \vspace{2mm}\\
				\sqrt{\frac{j - k + 1}{j + 1}} e_{j - k}, & j \geqslant k,
			\end{cases}
		\end{align*}
		and
		\begin{align*}
			T_{z^ k} e_j = \sqrt{\frac{j + 1}{j + k + 1}} e_{j + k}, \qquad j \geqslant 0.

			T_{z^ k} e_j (z) & = \int_{\mathbb{D}} \frac{e_j (w) w^ k}{(1 - z \overline{w})^ 2} \,\mathrm{d} A(w) = \sqrt{j + 1} \sum_{i = 0}^ {\infty} {(i + 1) z^ i \int_{\mathbb{D}} w^ {j + k} \overline{w}^ i \,\mathrm{d} A(w)} \\
			& = \sqrt{j + 1} (j + k + 1) z^ {j + k} \int_{\mathbb{D}} |w|^ {2 (j + k)} \,\mathrm{d} A (w) = \sqrt{\frac{j + 1}{j + k + 1}} e_{j + k} (z).

\label{theorem: spectrum and essential spectrum}

			\item $\sigma_{\mathrm{e}} (T_{\varphi}) = \varphi (\mathbb{T})$.
			\item For every $\lambda \notin \varphi (\mathbb{T})$,
			\begin{align*}
				\operatorname{index}{(T_{\varphi} - \lambda I)} = -\mathrm{wind}{(\varphi (\mathbb{T}), \lambda)}.
			\end{align*}
			\item For every $\lambda \notin \varphi (\mathbb{T})$, $\lambda$ is an eigenvalue of $T_{\varphi}$ if and only if either $\widetilde{\varphi} (z) - \lambda$ is non-vanishing on $\mathbb{D}_0$ or $\widetilde{\varphi} (z) - \lambda$ has finitely many simple zeros $\{ z_1, \dots, z_k \}$ in $\mathbb{D}_0$ which satisfy
			\begin{align*}
				\frac{z_j^ 2 p' (z_j)}{a} = \frac{n_j + 2}{n_j + 1}
			\end{align*}
			for some $n_j \in \mathbb{N}$, where $j = 1, 2, \dots, k$.
		\end{enumerate}

<!-- lem-5: body_md -->
\label{lemma: Lambdad is discrete}
		Every point in $\Lambda_{\mathrm{d}} (T_{\varphi})$ is an isolated point of $\sigma (T_{\varphi})$. Consequently, $\Lambda_{\mathrm{c}} (T_{\varphi})$ is a connected component of $\sigma (T_{\varphi})$.
<!-- end lem-5 -->
<!-- proof-4: body_md -->
It suffices to consider the case that $\Lambda_{\mathrm{d}} (T_{\varphi})$ is nonempty. Pick a $\lambda_0 \in \Lambda_{\mathrm{d}} (T_{\varphi})$. By the argument principle, the equation
		\begin{align*}
			\widetilde{\varphi} (z) = \frac{a}{z} + p (z) = \lambda_0
		\end{align*}
		has a unique simple root $z_0$ in $\mathbb{D}_0$. Since $\lambda_0$ is an eigenvalue of $T_{\varphi}$, according to (c) of Theorem \ref{theorem: spectrum and essential spectrum}, we must have
		\begin{align*}
			\frac{z_0^ 2 p' (z_0)}{a} = \frac{m_0 + 2}{m_0 + 1}
		\end{align*}
		for some $m_0 \in \mathbb{N}$. As a consequence,
		\begin{align*}
			(\widetilde{\varphi})' (z_0) = -\frac{a}{z_0^ 2} + p' (z_0) = -\frac{a}{z_0^ 2} + \frac{m_0 + 2}{m_0 + 1} \frac{a}{z_0^ 2} = \frac{1}{m_0 + 1} \frac{a}{z_0^ 2} \neq 0.
		\end{align*}
		By the inverse function theorem for analytic functions, $\widetilde{\varphi}$ has an analytic inverse $(\widetilde{\varphi})^ {-1}$ on a neighborhood of $z_0$. Now, let $r > 0$ be a small number such that
		\begin{enumerate}[label = (\roman*)]
			\item $D (z_0, r) \subseteq \mathbb{D}_0$,
			\item $\widetilde{\varphi}$ has an analytic inverse on $D (z_0, r)$, and
			\item $U := \widetilde{\varphi} (D (z_0, r))$ lies in the connected component of $\mathbb{C} \backslash \varphi (\mathbb{T})$ that contains $\lambda_0$ (note that $\lambda_0 \in \mathbb{C} \backslash \varphi (\mathbb{T})$).
		\end{enumerate}
		By Condition (iii), for each $\lambda \in U$ we have
		\begin{align}
			\mathrm{wind}{(\varphi (\mathbb{T}), \lambda)} = \mathrm{wind}{(\varphi (\mathbb{T}), \lambda_0)} = 0, \label{eq5}
		\end{align}
		and hence by the argument principle, the equation
		\begin{align*}
			\widetilde{\varphi} (z) = \frac{a}{z} + p(z) = \lambda
		\end{align*}
		has $(\widetilde{\varphi})^ {-1} (\lambda)$ as its unique simple root in $\mathbb{D}_0$.
		According to (\ref{eq5}) and Theorem \ref{theorem: spectrum and essential spectrum}, $\lambda \in U$ is in $\sigma (T_{\varphi})$ if and only if
		\begin{align}
			G (\lambda) := \frac{\big[ (\widetilde{\varphi})^ {-1} (\lambda) \big]^ 2 p' \big( (\widetilde{\varphi})^ {-1} (\lambda) \big)}{a} = \frac{m + 2}{m + 1} \label{eq6}
		\end{align}
		for some $m \in \mathbb{N}$. Note that
		\begin{align*}
			G (\lambda_0) = \frac{z_0^ 2 p' (z_0)}{a} = \frac{m_0 + 2}{m_0 + 1}
		\end{align*}
		as $\widetilde{\varphi} (z_0) = \lambda_0$. Since $G$ is non-constant, we can find a sufficiently small neighborhood $V \subseteq U$ of $\lambda_0$ such that
		\begin{align*}
			\left\{ G (\lambda) : \lambda \in V \backslash \{ \lambda_0 \} \right\} \cap \left\{ \frac{j + 2}{j + 1} : j \in \mathbb{N} \right\} = \varnothing.
		\end{align*}
		Therefore, every $\lambda \in V \backslash \{ \lambda_0 \}$ is not in $\sigma (T_{\varphi})$, since (\ref{eq6}) does not hold for this $\lambda$. This implies that $\lambda_0$ is an isolated point of $\sigma (T_{\varphi})$.

\label{lemma: univalent}
		Let $a \in \mathbb{C} \backslash \{ 0 \}$, $p \in \mathrm{Hol} (\overline{\mathbb{D}})$, and $\varphi (z) = a \overline{z} + p(z)$. Suppose that $\widetilde{\varphi}$ is univalent on $\mathbb{D}_0$. Then:
		\begin{enumerate}[label = \textup{(\alph*)}]
			\item $\widetilde{\varphi}$ maps $\mathbb{D}_0$ onto $\Omega_{\infty}$ conformally.\\
			\item $\mathrm{wind}{(\varphi (\mathbb{T}), \lambda)} = -1$ for every $\lambda \in \bigsqcup\limits_{j \geqslant 0} {\Omega_j}$.\\
			\item $\Lambda_{\mathrm{c}} (T_{\varphi}) = \mathbb{C} \backslash \Omega_{\infty} = \varphi (\mathbb{T}) \sqcup \big( \bigsqcup\limits_{j \geqslant 0} {\Omega_j} \big)$.
		\end{enumerate}

<!-- proof-5: body_md -->
We first prove (a). We claim that
		\begin{align}
			\widetilde{\varphi} (\mathbb{D}_0) \cap \widetilde{\varphi} (\mathbb{T}) = \varnothing. \label{eq11}
		\end{align}
		Otherwise, there exist $z_0 \in \mathbb{D}_0$ and $z_1 \in \mathbb{T}$ such that
		\begin{align*}
			\widetilde{\varphi} (z_0) = \widetilde{\varphi} (z_1) =: \xi.
		\end{align*}
		Let $r_0$ be a small positive number such that $D (z_0, r_0) \subseteq \mathbb{D}_0$. By the open mapping theorem of analytic functions, $\widetilde{\varphi} (D (z_0, r_0))$ is an open neighborhood of $\xi$. Then by the continuity of $\widetilde{\varphi}$, there exists an $r_1 > 0$ such that
		\begin{align}
			\widetilde{\varphi} (D (z_1, r_1)) \subseteq \widetilde{\varphi} (D (z_0, r_0)). \label{ineq2}
		\end{align}
		We may assume that $r_1$ is sufficiently small such that
		\begin{align}
			D (z_0, r_0) \cap D (z_1, r_1) = \varnothing. \label{eq12}
		\end{align}
		Now using (\ref{ineq2}), we can find $w_0 \in D (z_0, r_0)$, $w_1 \in D (z_1, r_1) \cap \mathbb{D}_0$, such that $\widetilde{\varphi} (w_0) = \widetilde{\varphi} (w_1)$. By (\ref{eq12}), $w_0 \neq w_1$, so we deduce that $\widetilde{\varphi}$ is not univalent on $\mathbb{D}_0$, which contradicts the hypotheses.
		
		The set $\widetilde{\varphi} (\mathbb{D}_0)$ is connected (since $\mathbb{D}_0$ is connected and $\widetilde{\varphi}$ is continuous), unbounded and contained in $\mathbb{C} \backslash \varphi (\mathbb{T})$ (due to (\ref{eq11})). Hence it is contained in $\Omega_{\infty}$. On the other hand, we also have by the basic topological fact that
		\begin{align*}
			\mathrm{wind}{(\varphi (\mathbb{T}), \lambda)} = 0 \qquad (\lambda \in \Omega_{\infty}),
		\end{align*}
		and hence by the argument principle that $\Omega_{\infty} \subseteq \widetilde{\varphi} (\mathbb{D}_0)$. We conclude that
		\begin{align*}
			\widetilde{\varphi} (\mathbb{D}_0) = \Omega_{\infty}.
		\end{align*}
		Since $\widetilde{\varphi}$ is univalent on $\mathbb{D}_0$, $\widetilde{\varphi}$  maps $\mathbb{D}_0$  conformally onto $\Omega_{\infty}$.
		
		Next, we are going to prove (b). We only consider the non-vacuous case that $\{ \Omega_j \}_{j \geqslant 0}$ is nonempty. Let $j \geqslant 0$ and pick an arbitrary $\lambda \in \Omega_j$. Since
		\begin{align*}
			\widetilde{\varphi} (\mathbb{D}_0) \cap \Omega_j = \Omega_{\infty} \cap \Omega_j = \varnothing,
		\end{align*}
		the function $\widetilde{\varphi} (z) - \lambda$ is non-vanishing on $\mathbb{D}_0$. It follows from the argument principle that
		\begin{gather*}
			\mathrm{wind}{(\varphi (\mathbb{T}), \lambda)} = -1.
		\end{gather*}
	
		Lastly, observe that (c) is a direct consequence of the definition of $\Lambda_{\mathrm{c}}$ and (b). The proof is complete.

\label{lemma: properties of V}
		The following are true:
		\begin{enumerate}[label = \textup{(\alph*)}]
			\item  $V (f g) = (V f) (V g)$ for $f \in L^ {\infty} (\mathbb{D}, \mathrm{d} A)$ and $g \in L^ 2 (\mathbb{D}, \mathrm{d} A)$.
			\item $V P = P V$ on $L^ 2 (\mathbb{D}, \mathrm{d} A)$.
			\item $T_{V \varphi} = V T_{\varphi} V$ on $L_a^ 2$ for every $\varphi \in L^ {\infty} (\mathbb{D}, \mathrm{d} A)$.

		\end{gather*}
		The proof of the lemma is complete.

\label{lemma: recurrence relation}
		Let $a \in \mathbb{C} \backslash \{ 0 \}$, $p (z) = \sum\limits_{k = 0}^ {\infty} {c_k z^ k} \in \mathrm{Hol} (\overline{\mathbb{D}})$, and $\varphi (z) = a \overline{z} + p(z)$. Suppose that
		\begin{align*}
			f = \sum_{j = 0}^ {\infty} {a_j e_j}
		\end{align*}
		is an eigenvector of $T_{\varphi}$ corresponding to the eigenvalue $\lambda$. Then $\{ a_j \}_{j = 0}^ {\infty}$ satisfies the recurrence relation
		\begin{align*}
			\sqrt{\frac{j + 1}{j + 2}} a a_{j + 1} + \sum_{k = 0}^ j {\sqrt{\frac{j - k + 1}{j + 1}} c_k a_{j - k}} = \lambda a_j, \qquad j \in \mathbb{N}.
		\end{align*}
<!-- end lem-11 -->
<!-- proof-10: body_md -->
For each $n \in \mathbb{N}$, let
		\begin{align*}
			p_n (z) = \sum_{k = 0}^ n {c_k z^ k} \qquad \textrm{and} \qquad \varphi_n (z) = a \overline{z} + p_n (z).
		\end{align*}
		Since $p$ is analytic on $\overline{\mathbb{D}}$, $\{ p_n \}_{n = 0}^ {\infty}$ converges to $p$ uniformly on $\overline{\mathbb{D}}$. It follows that
		\begin{align*}
			\| T_{\varphi} - T_{\varphi_n} \| = \| T_{\varphi - \varphi_n} \| = \| T_{p - p_n} \| \leqslant \sup_{z \in \mathbb{D}} {|p (z) - p_n (z)|} \rightarrow 0 \ \ \ \ (n \rightarrow \infty),
		\end{align*}
		which gives
		\begin{align*}
			T_{\varphi} = a T_{\overline{z}} + \sum_{k = 0}^ {\infty} {c_k T_{z^ k}},
		\end{align*}
		where the series on the right-hand side converges in the operator norm. As a consequence, we have by Lemma \ref{lemma: Toeplitz operators with monomial symbols} that
		\begin{align*}
			T_{\varphi} f = a \sum_{j = 0}^ {\infty} {\sqrt{\frac{j + 1}{j + 2}} a_{j + 1} e_j} + \sum_{k = 0}^ {\infty} {c_k \sum_{j = k}^ {\infty} {\sqrt{\frac{j - k + 1}{j + 1}} a_{j - k} e_j}}.
		\end{align*}
		For each $j_0 \in \mathbb{N}$,
		\begin{align*}
			\langle T_{\varphi} f, e_{j_0} \rangle_{L_a^ 2} & = a \sum_{j = 0}^ {\infty} {\sqrt{\frac{j + 1}{j + 2}} a_{j + 1} \langle e_j, e_{j_0} \rangle_{L_a^ 2}} + \sum_{k = 0}^ {\infty} {c_k \sum_{j = k}^ {\infty} {\sqrt{\frac{j - k + 1}{j + 1}} a_{j - k} \langle e_j, e_{j_0} \rangle_{L_a^ 2}}} \\
			& = \sqrt{\frac{j_0 + 1}{j_0 + 2}} a a_{j_0 + 1} + \sum_{k = 0}^ {j_0} {c_k \sqrt{\frac{j_0 - k + 1}{j_0 + 1}} a_{j_0 - k}},
		\end{align*}
		so we get that
		\begin{align*}
			T_{\varphi} f = \sum_{j = 0}^ {\infty} {\bigg[ \sqrt{\frac{j + 1}{j + 2}} a a_{j + 1} + \sum_{k = 0}^ j {\sqrt{\frac{j - k + 1}{j + 1}} c_k a_{j - k}} \bigg] e_j}.
		\end{align*}
		The desired recurrence relation follows by comparing coefficients of $T_{\varphi} f$ and $\lambda f$.

For $z \in \mathbb{T}$, we have $\Psi (z) = \overline{z}$. Thus, $\mathbf{T}_{\Psi}$ is exactly the unweighted backward shift operator $\mathbf{T}_{\overline{z}}$. Since $\mathbf{T}_{\overline{z}}$ is a contraction, it is non-hypercyclic on the Hardy space $H^2$.


Recall that $e_j (z) = \sqrt{j+1}z^j\ (j\in \mathbb N)$, which is an orthonormal basis for the Bergman space $L_a^2$. Then we can calculate as in the proof of Lemma \ref{lemma: Toeplitz operators with monomial symbols} to obtain that
		\begin{align*}
			T_{\Psi} e_j (z) =
			\begin{cases}
				0, & j = 0, \vspace{2mm}\\
				\sqrt{\frac{j}{j + 1}} \frac{j + 3}{j + 2} e_{j - 1} (z), & j \in \mathbb{N}_+.
			\end{cases}
		\end{align*}
		Note that $T_{\Psi}$ is a weighted backward shift, whose weight sequence $\{ w_j \}_{j = 1}^ {\infty}$ is given by
		\begin{align*}
			w_j = \sqrt{\frac{j}{j + 1}} \frac{j + 3}{j + 2}, \qquad j \in \mathbb{N}_+.

			v_n := w_1 w_2 \cdots w_n = \frac{n + 3}{3 \sqrt{n + 1}}\ \ \ \ (n=1, 2, \cdots)
		\end{align*}
		increases to $+\infty$ as $n \rightarrow \infty$, we conclude by \cite[Theorem 2.8]{Sal1} that $T_{\Psi}$ is hypercyclic on the Bergman space $L_a^2$.
