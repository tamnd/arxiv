<!-- definition-1: body_md -->
\label{def:ldp}
A family of Markov kernels $\{Q_x:x\in K\}$ from $\mathcal{Y}$ to query-dependent measurable output spaces $(\mathcal Z_x,\mathcal A_x)$ is $(\varepsilon,\delta)$-locally differentially private (or $(\varepsilon,\delta)$-LDP) if, for every fixed public query $x\in K$, every $y,y'\in\mathcal{Y}$, and every $A\in\mathcal A_x$,
\begin{equation*}
        Q_x(A\mid y)\le e^\varepsilon Q_x(A\mid y')+\delta .
\end{equation*}

<!-- proposition-1: body_md -->
\label{prop:adaptive-transcript}
In this one-message protocol, suppose each conditional message distribution satisfies Definition~\ref{def:ldp} for every realised prior public history.
Then, for every participant $i\le N$, fixed data $y_{-i}$ of the other participants, $y,y'\in\mathcal{Y}$, and measurable transcript event $A$,
\begin{equation}\label{eq:adaptive-transcript-ldp}
        \mathbb{P}(\mathcal T_N\in A\mid Y_i=y,Y_{-i}=y_{-i})
        \le e^\varepsilon
        \mathbb{P}(\mathcal T_N\in A\mid Y_i=y',Y_{-i}=y_{-i})
        +\delta .
\end{equation}

\label{thm:targetshift}
Let \(U\) be a normal convex neighbourhood of \(x_\star\). Assume
\(x\mapsto F_Q(x)\) is \(C^2\) on \(U\) and has an interior local minimiser
\(x_Q\in U\). Define
\[
        \bar F_Q(v)=F_Q(\mathrm{Exp}_{x_\star}v),\qquad
        \Delta_Q=\mathrm{Log}_{x_\star}(x_Q).
\]
Assume that
\begin{equation}\label{eq:AQ-def}
        A_Q=\int_0^1 \nabla^2\bar F_Q(t\Delta_Q)\,dt
\end{equation}
is nonsingular. Then
\begin{equation}\label{eq:shift-identity}
        \Delta_Q=-A_Q^{-1}\nabla\bar F_Q(0),
\end{equation}
and, for this local minimiser,
\begin{equation}\label{eq:target-coincidence}
        x_Q=x_\star
        \quad\Longleftrightarrow\quad
        \nabla F_Q(x_\star)=0.
\end{equation}
Moreover, consider a sequence of privatisation kernels for which
\(\Delta_Q\to0\). Suppose that \(F\) is \(C^2\) near \(x_\star\),
\(H=\nabla^2F(x_\star)\) is nonsingular,
\(\nabla^2\bar F_Q(0)\to H\), and the Hessians are locally equicontinuous at
zero: for some \(r>0\) and deterministic modulus \(\omega(t)\downarrow0\),
\[
 \left\lVert \nabla^2\bar F_Q(u)-\nabla^2\bar F_Q(v)\right\rVert_{\mathrm{op}}
 \le \omega\left(\left\lVert u-v\right\rVert\right)
\]
whenever \(\left\lVert u\right\rVert,\left\lVert v\right\rVert\le r\). Then
\begin{equation}\label{eq:first-order-shift}
        \mathrm{Log}_{x_\star}(x_Q)
        =-H^{-1}\nabla F_Q(x_\star)
          +o\left\{\left\lVert \nabla F_Q(x_\star)\right\rVert\right\}.
\end{equation}
