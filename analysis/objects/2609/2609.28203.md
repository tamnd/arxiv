\label{def:a1-connected-components}
		For a space
		$\mathcal{X} \in \mathcal{H}(S)$, the \textbf{sheaf of
			$\mathbb{A}^1$-connected components}, denoted by $\pi_0^{\mathbb{A}^1}(\mathcal{X})$,
		is the Nisnevich sheaf on $\mathrm{Sm}_S$ defined as the sheafification of
		the presheaf
		\[
		U \longmapsto \mathrm{Hom}_{\mathcal{H}(S)}(U, \mathcal{X}),
		\]
		where $U \in \mathrm{Sm}_S$. Equivalently, if
		$\mathrm{L}_{\mathbb{A}^1}: \mathrm{sPre}(\mathrm{Sm}_S) \to
		\mathrm{sPre}(\mathrm{Sm}_S)$ denotes the $\mathbb{A}^1$-localization functor,
		then
		\[
		\pi_0^{\mathbb{A}^1}(\mathcal{X}) = a_{\mathrm{Nis}}
		\!\bigg(U \mapsto \mathrm{Hom}_{\mathcal{H}(S)}(U, L_{\#A^1}\mathcal{X})\bigg),
		\]
		where $a_{\mathrm{Nis}}$ denotes the Nisnevich sheafification of the underlying presheaf. A space $\mathcal{X}$
		is called \textbf{$\mathbb{A}^1$-connected} if
		$\pi_0^{\mathbb{A}^1}(\mathcal{X}) \cong *$.

<!-- definition-2: body_md -->
\label{def:naive-a1-homotopy}
		Let $\mathfrak{X}$ be a simplicial presheaf on $\mathrm{Sm}_k$.
		
		\begin{enumerate}[(1)]
			\item Let $U\in \mathrm{Sm}_k$. Two objects $x$ and $y$ in $\mathfrak{X}(U)$
			are said to be \textit{naively $\mathbb{A}^1$-homotopic} if there exists a
			map $f: \mathbb{A}^1\times_k U \rightarrow \mathfrak{X}$ satisfying $f_0= x$ and
			$f_1 = y$, where $f_i$ is the composition
			$U=\{i\}\times U\xrightarrow{i}\mathbb{A}^1\times_k U \xrightarrow{f} \mathfrak{X}$. 
			
			This generates an equivalence relation on the set $\mathfrak{X}(U)_0$ of 0-simplices of the simplicial set $\mathfrak{X}(U)$. Let us the denote the equivalence relation by    $\sim$.
			\item We shall denote by $S^{\rm pre}(\mathfrak{X})$ the presheaf on $\mathrm{Sm}_k$
			whose value on a $k$-scheme $U$ is given by
			\[
			U\mapsto \mathfrak{X}(U)_0/{\sim},
			\]
			where $\sim$ is the equivalence relation in~(1). The {\bf sheaf of $\#A^1$-chain connected components}  of $\mathfrak X$, denoted by $S(\mathfrak{X})$ is the Nisnevich sheafification of the presheaf $S^{\rm  pre}(\mathfrak{X})$.
		\end{enumerate}

<!-- lemma-1: body_md -->
(\cite[Lemma~6.1.3]{Mor05}, \cite[section 2, corollary 3.22]{MV99}) \label{lem:a1-connected-condition}
		Let  $\mathfrak X$ be a space over an infinite
		field $k$. Then $\mathfrak X$ is $\mathbb{A}^1$-connected if
		\[
		S^{\rm pre}(\mathfrak X)(\mathrm{Spec}\,K) = \ast
		\]
		for every finitely generated separable field extension $K$ of $k$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Assume that $S^{\rm pre}(\mathfrak X)(\mathrm{Spec}\,K) = *$, hence $S(\mathfrak X)(\mathrm{Spec}\,K) = *$,
		for every finitely generated separable field extension $K$ of $k$, since stalks are same for a presheaf and its sheafification. Note there exists a natural epimorphism $S(\mathfrak X) \to \pi^{\mathbb{A}^1}_0(\mathfrak X)$ of Nisnevich sheaves, hence induces a surjective map of stalks at points in the Nisnevich topology, in particular at ${\rm Spec \,} K$ for  every finitely generated separable field extension $K$ of $k$. Hence $\pi^{\mathbb{A}^1}_0(\mathfrak X)({\rm Spec \,} K)=\ast$. Since the base field $k$ is infinite, the lemma follows by \cite[Lemma~6.1.3]{Mor05}, \cite[section 2, corollary 3.22]{MV99}.

\label{projform}

<!-- definition-4: body_md -->
\hfill
		\begin{enumerate}[(1)]
			\item For a vector bundle $\mathcal{E}$ and a coherent sheaf $\mathcal{F}$ on $\mathfrak{C}$, its \textit{$\mathcal{E}$-degree} is defined by
			\[
			d_{\mathcal{E}}(\mathcal{F}) \;=\; \deg \pi_* \mathcal{H}om(\mathcal{E}, \mathcal{F}) \;-\; \operatorname{rank} \mathcal{F} \cdot \deg \pi_* \mathcal{H}om(\mathcal{E}, \mathcal{O}_\mathfrak{C}).

\label{lem:pushforward of dual}

			\implies & \bigl(\pi_*(\mathcal{G}^{\vee})\bigr)^{\vee}\otimes_{\mathcal{O}_C}\omega_C\xrightarrow{\,\,\simeq\,\,}\pi_*\bigl(\mathcal{G}\otimes_{\mathcal{O}_{\mathfrak{C}}}\omega_{\mathfrak{C}}\bigr) \\
			\implies& \pi_*(\mathcal{G}^{\vee})\xrightarrow{\,\,\simeq\,\,} \bigl(\pi_*(\mathcal{G}\otimes\omega_{\mathfrak{C}})\bigr)^{\vee}\otimes\omega_C\,.

\label{prop:HN-filtration}\hfill
		\begin{enumerate}[(1)]
			\item Let $\mathcal{E}$ and $\mathcal{F}$ be vector bundles on $\mathfrak{C}$. Then
			$\mathcal{F}$ is $(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)$-semistable if and only if $\mathcal{F}^\vee$ is $\mathcal{E}^\vee$-semistable.
			\item If
			\[
			0 = \mathcal{F}_0 \subset \mathcal{F}_1 \subset \cdots \subset \mathcal{F}_k = \mathcal{F}
			\]
			is the $(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)$-HN filtration of $\mathcal{F}$ with $\mathcal{E}$-semistable successive quotients $G_i := \mathcal{F}_i/\mathcal{F}_{i-1}$ and
			strictly decreasing slopes $\mu_{(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)}(G_1) > \cdots > \mu_{(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)}(G_k)$, then the $\mathcal{E}^\vee$-HN filtration of
			$\mathcal{F}^\vee$ is given by
			\smallskip
			\[
			0 \subset (\mathcal{F}/\mathcal{F}_{k-1})^\vee \subset (\mathcal{F}/\mathcal{F}_{k-2})^\vee \subset \cdots \subset (\mathcal{F}/\mathcal{F}_1)^\vee \subset \mathcal{F}^\vee,
			\]
			
			where the successive quotients $G_i^\vee=(\mathcal{F}_i/\mathcal{F}_{i-1})^\vee$ are $\mathcal{E}^\vee$-semistable and have strictly
			decreasing slopes $\mu_{\mathcal{E}^\vee}(G_k^\vee) > \cdots > \mu_{\mathcal{E}^\vee}(G_1^\vee)$. In particular,
			\smallskip
			\[\mu_{\mathcal{E}^\vee}^{\max}(\mathcal{F}^\vee) = -\, \mu_{(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)}^{\min}(\mathcal{F}) \,\,\,\,\,\text{and}\,\,\,\,\, \mu_{\mathcal{E}^\vee}^{\min}(\mathcal{F}^\vee) = -\, \mu_{(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)}^{\max}(\mathcal{F}).
			\]
		\end{enumerate}

		    \item  Let

		$(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)$-semistability of $\mathcal{F}$ implies that  $\mu_{(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)}(\mathcal{F}) \leq \mu_{(\mathcal{E}\otimes\omega_{\mathfrak{C}}^\vee)}(H^\vee)$. By Lemma \ref{lem:degree of dual},

<!-- lemma-4: body_md -->
\label{lem:pullback-slope}
		Let $E$ be a vector bundle on $\mathfrak{C}$, and $F$ be a vector bundle on $C$. Let $\pi:\mathfrak{C} \to C$ be the coarse map. Then $\mu_{\mathcal{E}}(E\otimes \pi^*F)=\mu_{\mathcal{E}}(E)+\mathrm{rank}(\mathcal{E})\cdot\mu(F)\,.$
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
By definition,
		\begin{align*}
			d_\mathcal{E}(E\otimes\pi^*F)=& \deg \pi_*\mathcal{H}om(\mathcal{E},E\otimes\pi^*F)-{\rm rank} (E\otimes\pi^*F)\deg\pi_*(\mathcal{E}^{\vee})\,.
		\end{align*}
		By projection formula,
		\begin{align*}
			\pi_*\mathcal{H}om(\mathcal{E},E\otimes\pi^*F) = \pi_*(\mathcal{E}^{\vee}\otimes E\otimes\pi^*F)\simeq \pi_*(\mathcal{E}^{\vee}\otimes E)\otimes F\,.
		\end{align*}
		It follows that
		{\small \begin{align*}
				d_\mathcal{E}(E\otimes\pi^*F)=& \deg (\pi_*\mathcal{H}om(\mathcal{E},E)\otimes F)-{\rm rank} (E\otimes\pi^*F)\deg\pi_*(\mathcal{E}^{\vee}) \\

				=& {\rm rank}(F)\cdot d_{\mathcal{E}}(E)+{\rm rank}(\mathcal{E}^{\vee}\otimes E)\cdot \deg(F)\\
				=& {\rm rank}(F)\cdot d_{\mathcal{E}}(E)+{\rm rank}(\mathcal{E}){\rm rank}(E)\cdot \deg(F).
		\end{align*}} Thus, $\mu_{\mathcal{E}}(E\otimes\pi^*F) = \mu_{\mathcal{E}}(E)+{\rm rank}(\mathcal{E})\cdot\mu(F)\,,$
		as claimed.

<!-- lemma-5: body_md -->
\label{lem:tensor product slope}
		Let $\mathcal{E}$ be a vector bundle satisfying $w_{q, i}(\mathcal{E}) = i\cdot w_{q,1}(\mathcal{E})$ for all stacky points $p$ of order $e_p$ and all $i\in[0,e_p-1]$ (for example, take the standard generating sheaf $\mathcal{E}_{\text{fav}}$ from \cite[Definition 1.3.12, Example 1.3.15]{Taams}). For any two vector bundles $F$ and $F'$ on $\mathfrak{C}$, we have 
		$$\mu_{\mathcal{E}}(F\otimes F') = \mu_{\mathcal{E}}(F)+\mu_{\mathcal{E}}(F')\,.$$
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Note that if $0\to F_1 \to F\to F_2\to 0$ is a short exact sequence of locally free sheaves, and if our claim holds for $F_1$ and $F_2$ and any vector bundle $F'$, then it also holds for $F$:
		\begin{footnotesize}
			\begin{align*}
				\mu_{\mathcal{E}}(F\otimes F') = \frac{d_{\mathcal{E}}(F\otimes F')}{{\rm rank}(F\otimes F')}
				= \sum_{i=1}^2\frac{d_{\mathcal{E}}(F_i\otimes F')}{{\rm rank}(F)\cdot{\rm rank}(F')}
				&=\sum_{i=1}^2\mu_{\mathcal{E}}(F_i\otimes F')\frac{{\rm rank}(F_i)}{{\rm rank} (F)}\\
				&=\frac{1}{{\rm rank}(F)}\sum_{i=1}^2 {\rm rank}(F_i)\bigl(\mu_{\mathcal{E}}(F_i)+\mu_{\mathcal{E}}(F')\bigr)\\
				&=\frac{1}{{\rm rank}(F)}\sum_{i=1}^2d_{\mathcal{E}}(F_i)+{\rm rank}(F_i)\cdot\mu_{\mathcal{E}}(F')\\
				&=\frac{1}{{\rm rank}(F)}\bigl(d_{\mathcal{E}}(F)+{\rm rank}(F)\cdot\mu_{\mathcal{E}}(F')\bigr)\\
				&=\mu_{\mathcal{E}}(F)+\mu_{\mathcal{E}}(F')\,.
			\end{align*}
		\end{footnotesize} Next, by \cite[Lemma 1.2.20]{Taams}, any vector bundle $F$ on $\mathfrak{C}$ admits a sequence of surjective maps 
		\[F=F_0\twoheadrightarrow F_1\twoheadrightarrow\cdots\twoheadrightarrow F_r=0\]
		such that each $F_i$ is locally free and $L_i:=\ker(F_i\twoheadrightarrow F_{i-1})$ is an invertible sheaf of rank 1. By the above considerations, we reduce to the case where $F$ is an invertible sheaf of rank 1. Since tensor product is commutative, applying the same argument above to $F'$ by symmetry, we can reduce to the case where both $F$ and $F'$ are invertible sheaves of rank 1.
		
		\noindent
		Next, note that if the statement holds for two invertible sheaves $L_1$ and $L_2$ (and any $F$), then it also holds for $L_1\otimes L_2$:
		\begin{align*}
			\mu_{\mathcal{E}}(F\otimes L_1\otimes L_2) = \mu_{\mathcal{E}}(F\otimes L_1)+d_{\mathcal{E}}(L_2) =& \mu_{\mathcal{E}}(F)+d_{\mathcal{E}}(L_1)+d_{\mathcal{E}}(L_2)\\
			=& \mu_{\mathcal{E}}(F)+d_{\mathcal{E}}(L_1\otimes L_2)\,.
		\end{align*}
		By \cite[Corollary 1.2.11]{Taams}, any invertible sheaf $L$ on $\mathfrak{C}$ can be written uniquely as $$L\cong \pi^*M \otimes \bigotimes_p\mathcal{O}(\mathcal{G}_p)^{\otimes a_p}$$ where $M\in {\rm Pic}(C)$ and $\mathcal{G}_p$ is the residual gerbe at a stacky point $p$. By Lemma \ref{lem:pullback-slope}, our claim holds for $\pi^*M$. Thus, it is enough to prove our claim for each $\mathcal{O}(\mathcal{G}_p)$\,.
		
		\noindent
		Thus, we are reduced to the case where $F=\mathcal{O}(\mathcal{G}_q)$ and $F'=\mathcal{O}(\mathcal{G}_{q'})$ for two stacky points $q$ and $q'$. Denote $\mathcal{F}:= \mathcal{O}(\mathcal{G}_q)\otimes\mathcal{O}(\mathcal{G}_{q'})$. We consider two cases separately:
		\begin{enumerate}[$\bullet$]
			\item If $q\neq q'$, we get
			\begin{align*}
				d_{\mathcal{E}}(F\otimes F')&= {\rm rank}(\mathcal{E})\Bigl(\deg\pi_*(\mathcal{F})+ \sum_p\sum_{i=0}^{e_p-1}m_{p,i}(\mathcal{F})w_{p,i}(\mathcal{E})\Bigr)\\
				&= {\rm rank}(\mathcal{E})\Bigl(\deg(\mathcal{O}_C)+w_{q,1}(\mathcal{E})+w_{q',1}(\mathcal{E})\Bigr)\\
				&= d_{\mathcal{E}}(F) + d_{\mathcal{E}}(F')\,\,\,\,\text{by\,\,\cite[Proposition 1.3.17]{Taams}}\,.
			\end{align*}
			\item If $q=q'$, we get
			\begin{align*}
				d_{\mathcal{E}}(F\otimes F')&= {\rm rank}(\mathcal{E})\Bigl(\deg(\mathcal{O}_C)+w_{q,2}(\mathcal{E})\Bigr)\\
				&= {\rm rank}(\mathcal{E})\cdot 2w_{q,1}(\mathcal{E})\,\,\,\,\text{(by assumption)}\\
				&= d_{\mathcal{E}}(F) + d_{\mathcal{E}}(F')\,.
			\end{align*}
		\end{enumerate}
		
		This proves our claim.

<!-- lemma-7: body_md -->
\label{lem:vanish}
		If $\mu^{\min}_{\mathcal{E}}(G) > d_{\mathcal{E}}(\omega_{\mathfrak{C}})$ for a locally free $G$, then $H^1(\mathfrak{C}, G) = 0$.
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
By Serre duality, $H^1(\mathfrak{C}, G)^\vee \cong \operatorname{Hom}(G, \omega_{\mathfrak{C}})$ \cite[Theorem 1.3.6]{Taams}. A nonzero $\phi: G \to \omega_{\mathfrak{C}}$ has rank-$1$ image $I \subseteq \omega_{\mathfrak{C}}$ with torsion quotient, and the $\mathcal{E}$-degree of a torsion sheaf $T$ is non-negative, since it is the length of the torsion sheaf $\pi_*\mathcal{H}om(\mathcal{E}, T)$, so $d_\mathcal{E}(I) \leq d_\mathcal{E}(\omega_{\mathfrak{C}})$. On the other hand, $I$ is a quotient sheaf of $G$, so $d_\mathcal{E}(I) \geq \mu^{\min}_\mathcal{E}(G)$, again because passing from the locally free quotient to $I$ discards torsion of nonnegative $\mathcal{E}$-degree, hence a contradiction.

\label{prop:gen}
		Let $E$ be a vector bundle of type $\alpha$ on $\mathfrak{C}$. Let $d:=\deg(\pi_*E)$. Then there exists a positive integer $m(\mathfrak{C},\alpha)$ depending only on $\mathfrak{C}$ and $\alpha$, such that for any $m\ge m(\mathfrak{C},\alpha)$, and any $\mathcal{E}$-semistable vector bundle $F$ on $\mathfrak{C}$ of an admissible type $\beta$ satisfying $\deg(\pi_*F)= d+mn$, the sheaf $\pi_* H_m$ is globally generated on $C$. More precisely:
		\begin{enumerate}[(i)]
			\item for every stacky point $p_i$ of order $e_{p_i}$, let $x_i:= \pi(p_i)$; then the evaluation map
			\smallskip
			\[
			\operatorname{ev}_{p_i} : H^0(C, \pi_* H_m) \otimes k(x_i)\longrightarrow \bigoplus_{j=0}^{e_{p_i}-1} \operatorname{Hom}(V_{i,j}, W_{i,j}) = \operatorname{Hom}_{\mu_{e_i}}\!\big(E(m)|_{p_i}, F|_{p_i}\big)
			\]
			is surjective, and 
			\item for every non-stacky closed point $x$, the evaluation map 
			$$\operatorname{ev}_x : H^0(C, \pi_* H_m)\otimes k(x) \longrightarrow H_m|_x \cong \operatorname{Hom}\bigl(k(x)^n, k(x)^{n-1}\bigr)$$ is surjective. 
		\end{enumerate}

<!-- proof-10: body_md -->
Denote $\mathcal{E}':= (\mathcal{E}\otimes\omega_{\mathfrak{C}})^{\vee}$\,.  Replacing $\mathcal{E}$ by $\mathcal{E}'$ in Lemma \ref{lem:Chen22}, we get
		\smallskip
		$$\mu^{\min} _{\mathcal{E}}(E\otimes F)\geq \mu^{\min}_{\@E'}(\@E'\otimes  E)+\mu^{\min}_{\@E}(F) - a\,.$$
		Let $F$ be $\mathcal{E}$-semistable. Since $H_m = \mathcal{H}om(E(m),F)\simeq E^\vee\otimes F(-m)$, we get
		\smallskip
		\begin{align*}
			\mu^{\min}_{\mathcal{E}}(H_m)
			&\geq \mu^{\min}_{\mathcal{E}'}(\@E'^{\vee}\otimes  E^{\vee})+\mu^{\min}_{\mathcal{E}}\bigl(F(-m)\bigr) - a\,, \,\,\,\,\text{by Lemma}\,\,\ref{lem:Chen22} \\
			& = \mu^{\min}_{\mathcal{E}}\bigl(F(-m)\bigr) +c\,,\,\,\,\,\,\qquad\qquad \quad\text{for} \,\, c:=\mu^{\min}_{\mathcal{E}'}(\@E'^{\vee}\otimes  E^{\vee})-a\\ 
			& =\mu^{\min}_{\mathcal{E}}(F)- {\rm rank}(\@E)\cdot m +c\,,\,\qquad\text{by Lemma}\,\,\ref{lem:pullback-slope}\\ 
			&=\mu_{\mathcal{E}}(F)- {\rm rank}(\@E)\cdot m +c\,,\,\qquad\text{since}\,\,F\,\,\text{is}\,\,\mathcal{E}\text{-semistable}\\   
			&=\dfrac{d_{\mathcal{E}}(F)}{{\rm rank}(F)}-{\rm rank}(\mathcal{E})\cdot m +c\\
			&= \dfrac{{\rm rank}(\mathcal{E})}{{\rm rank}(F)}\Bigl(\deg(\pi_*F)+\sum_p\sum_{i=0}^{e_p-1}m_{p,i}(F)\,w_{p,i}(\mathcal{E})\Bigr)-{\rm rank}(\@E)\, m +c\\
			& ={\rm rank}(\@E) \dfrac{d+mn}{n-1} - {\rm rank}(\@E)\, m +c',\quad\text{for some constant}\,\, c'\\
			&  = {\rm rank}(\@E) \bigg(\dfrac{d}{n-1}+\dfrac{m}{n-1}\bigg)+c'\,. 
		\end{align*}

		Next, by a slight abuse of notation, let us denote $H_m(-x):=H_m\otimes \pi^*\mathcal{O}_C(- x)$ for a closed point $x\in C$. Note that

			\mu_{\mathcal{E}}^{\min}\bigl(H_m(-x)\bigr)
			&=\mu_{\mathcal{E}}^{\min}(H_m) + {\rm rank}(\mathcal{E})\cdot \deg(\mathcal{O}_C(-x))\,\,\,\,\,\,\text{by Lemma}\,\,\ref{lem:pullback-slope}\\
			&=\mu_{\mathcal{E}}^{\min}(H_m) - {\rm rank}(\mathcal{E})\cdot \deg(\mathcal{O}_C(x))\,.
		\end{align*}

		$$\mu_{\mathcal{E}}^{\min}(H_m(-x))>d_{\mathcal{E}}(\omega_{\mathfrak{C}})\,\,\,\text{for}\,\,m\ge m(x)\,.$$

		\begin{align}\label{1}
			H^1(\mathfrak{C}, H_m(-x))=0\,\,\,\text{for}\,\,m\ge m(x)\,.
		\end{align}

		\noindent
		Part ($ii$) is the classical statement for projective curves.

        \underline{{\bf Step 2:} (for stacky points)}  Let $p_i\in \mathfrak D$ be a stacky point. By Proposition~\ref{prop:gen}, $$\operatorname{ev}_{\pi(p_i)} : V \twoheadrightarrow \bigoplus_j \operatorname{Hom}_{\mu_{p_i}}(V_{i,j}, W_{i,j})$$ is surjective linear map. In the target,  there exists a unique index $j_i$ such that $\dim W_{i,j_i} = \dim V_{i,j_i}-1$, and $\dim W_{i,j} = \dim V_{i,j}$ for $j\neq j_i$. The non-surjective locus is the union over $j$ of the loci where the $j$-th block fails to surject: 
		
		for $j \neq j_i$ the non-surjective locus in $\operatorname{Hom}(V_{i,j}, W_{i,j})$ is a determinantal hypersurface (codimension $1$); 
		
		for $j = j_i$ the non-surjective locus in $\operatorname{Hom}(V_{i,j}, W_{i,j})$ has codimension $2$. 

        \noindent
		In all cases, the union is a proper closed subset of codimension at least 1, so its preimage  $Z_{p_i} \subset \mathbb{A}(V)$ under the map $ev_{\pi(p_i)}$ is a closed subscheme of dimension at most $(\dim V - 1)$. As the stacky locus is $0$-dimensional, so, unlike the classical count, codimension $1$ suffices here.

        \noindent
		\underline{{\bf Step 3:}} 
		Thus, from Steps 1 and 2, the locus of non-surjective $\phi$ is contained in $\overline{Z} \bigcup (\cup_{i=1}^r Z_{p_i})$, which is of dimension at most $\dim V - 1 < \dim V = \dim \mathbb{A}(V)$ hence $\overline{Z} \bigcup (\cup_{i=1}^r Z_{p_i})$ is a proper closed subscheme of $\#A(V)$. Since by assumption, $k$ is an infinite field, there is a $k$-rational point $\phi$ in the non-empty open complement of  $\overline{Z} \bigcup (\cup_{i=1}^r Z_{p_i})$ in $\#A(V)$ {\it i.e.\/},\  $\phi$ is surjective on every fiber.
<!-- end proof-11 -->
<!-- remark-5: body_md -->
The above result holds for any $E$ and $F$ such that $F$ has admissible type with respect to $E$ and $\pi_*\mathcal{H}om(E, F)$ is globally generated.
<!-- end remark-5 -->
<!-- remark-6: body_md -->
For two types $\alpha$ and $\beta$, there is a partial order where $\alpha \geq \beta$ if $m_{p, i}(\alpha) \geq m_{p,i}(\beta)$ for each stacky point $p$ and each weight $i \in \mathbb{Z}/e_p\mathbb{Z}$. Then using the above results iteratively, one gets that for vector bundles $E$ of type $\alpha$ and $F$ of type $\beta$ on $\mathfrak{C}$ with $\alpha \geq \beta$, for large enough $m$, a generic morphism $E(m) \to F$ is a surjection.

<!-- definition-6: body_md -->
\label{def:concordance}
		Let $\mathfrak{C}$ be a stacky curve over a field $K$. Choose two distinct points $x_0$ and $x_1$ in $\mathbb{A}^1_K$. Two vector bundles $\mathcal{E}_0$ and $\mathcal{E}_1$ on $\mathfrak{C}$ are said to be \textit{directly $\mathbb{A}^1$--concordant} if there exists a vector bundle $\mathcal{E}$ over $\mathfrak{C}\times \mathbb{A}^1_K$ such that $\mathcal{E}_0\xrightarrow{\,\,\simeq\,\,}\mathcal{E}|_{\mathfrak{C}\times\{x_0\}}$ and $\mathcal{E}_1\xrightarrow{\,\,\simeq\,\,}\mathcal{E}|_{\mathfrak{C}\times\{x_1\}}$. The vector bundles $\mathcal{E}_0$ and $\mathcal{E}_1$ are said to be \textit{$\mathbb{A}^1$--concordant} if they are equivalent under the equivalence relation generated by direct $\mathbb{A}^1$--concordance.
<!-- end definition-6 -->
<!-- remark-7: body_md -->
It is immediate that the equivalence relation generated by direct $\mathbb{A}^1$-concordances is the same as the equivalence relation defined in Definition \ref{def:naive-a1-homotopy}, when $\mathfrak{X}$ is the moduli stack of vector bundles on $\mathfrak{C}$.

<!-- proposition-4: body_md -->
\label{prop:concordance}
		Let $\mathfrak{C}$ be a projective stacky curve over an infinite field $k$. Consider a field extension $K\supset k$, and let $E_0$ and $E_1$ be two vector bundles on $\mathfrak C_K$ of rank $n$
        and multiplicities $\boldsymbol{m}$ satisfying $\det(\pi'_*E_i)\simeq \mathcal{L}$ for some line bundle $\mathcal{L}\in {\rm Pic}(C_K)$, where $\pi':\mathfrak{C}_K \to C_K$ is the coarse moduli map. Then $E_0$ and $E_1$ are directly $\#A^1$-concordant.
<!-- end proposition-4 -->
<!-- proof-12: body_md -->
For any integer $m\ge 0$, let $E_i(m):= E_i\otimes \pi'^*\mathcal{O}_{C_K}(m)$ as before. Note that 

        \noindent
		\[\det(\pi'_*(E_i(m)))\simeq \det(\pi_*E_i)\otimes \mathcal{O}_{C_K}(nm) = \mathcal{L}(nm)\,.\]
		Let $\mathcal{E}$ be a generating sheaf on $C_K$, and consider an $\mathcal{E}$-semistable vector bundle $F$ of rank $(n-1)$ with $\det(\pi'_*F)\simeq \mathcal{L}(mn)$ and having multiplicities as in the hypothesis of Proposition~\ref{prop:surjection}. Now, for $m$ large enough, Proposition \ref{prop:surjection} provides surjections $E_i(m)\xtwoheadrightarrow{\,} F$ for $i=0,1$. The computation of the determinants shows that both the maps $E_i(m)\xtwoheadrightarrow{\,} F$ have isomorphic kernels as line bundles, say $M$. Thus, $E_0$ and $E_1$ give rise to two extension classes $e_0$ and $e_1$ in $\text{Ext}^1(F(-m),M(-m))$. Using the results from change and cohomology for tame stacks in the case $\mathfrak{C}_K \longrightarrow
		{\rm Spec \,} K$ (see \cite[Remark A.2.5]{Bro12}) and in analogy with \cite{Lan83}, it follows that the  moduli functor given by for each $S \in {\rm Sch}/K$ $$\xymatrix{
			S\times \mathfrak{C}_{K} \ar[r]^{q_S} \ar[d]^{p_S} & \mathfrak{C}_{K} \ar[d]^f \\
			S\ar[r] \ar[r] & {\rm Spec \,} K}
		$$ 
		where 
		\medskip
		$$ S\mapsto E(S)=H^0\Bigl(S, \&{Ext}^1_{p_S}\bigl(q_S^*F(-m), q_S^*M(-m)\bigr)\Bigr)$$ 
		
		
		\noindent
		is representable by an affine space $V=\mathbb{A}^N_K$. The extension classes $e_0$ and $e_1$ can be considered as points in $V$. Consider the line joining $e_0$ and $e_1$; this gives rise to a map $f:\mathbb{A}^1_K \longrightarrow V$ satisfying $f(0)=e_0$ and $f(1)=e_1$. Pulling back the universal extension on $\mathfrak{C}_K\times V$ to $\mathfrak{C}_K\times \mathbb{A}^1_K$ gives rise to a direct $\mathbb{A}^1$-concordance between $E_0$ and $E_1$.
<!-- end proof-12 -->
<!-- remark-8: body_md -->
In the above proof, we assume the existence of an $\mathcal{E}$-semistable vector bundle $F$ of rank $(n-1)$ with $\det(\pi'_*F)\cong \mathcal{L}(mn)$ and having multiplicities as in Proposition \ref{prop:surjection}. The existence is clear for $n = 2$ as in \cite[Corollary 1.2.11]{Taams} and will be established inductively for higher ranks in the proof of theorem \ref{thm:A1sc}.
<!-- end remark-8 -->
<!-- remark-9: body_md -->
This proposition implies that the  full moduli stack $\mathcal{M}_{\mathfrak{C}}\bigl(n,\underline{m},L\bigr)$ is $\mathbb{A}^1$-connected; see \S~\ref{vb on stacky curve} for notations. This was already shown in \cite[Theorem 3.7]{CC25} as it can be considered as an iterated flag bundle over $\mathcal{M}_C(n, L)$. The above result gives an alternate proof by constructing direct $\mathbb{A}^1$-concordances in $\mathcal{M}_{\mathfrak{C}}\bigl(n,\underline{m},L\bigr)$.
<!-- end remark-9 -->
<!-- theorem-2: body_md -->
\label{thm:A1sc}
		Let $\mathfrak{C}$ be a projective stacky curve over an infinite field $k$ and $\mathcal{E}$ be a generating sheaf on $\mathfrak{C}$. Then $\mathcal{M}^{\mathcal{E}-\text{ss}}\bigl(n,\underline{m},L\bigr)$ is $\mathbb{A}^1$-connected.
<!-- end theorem-2 -->
<!-- proof-13: body_md -->
We prove the theorem by induction on $n$, the rank of the vector bundles. For $n=1$, we have $\mathcal{M}^{\mathcal{E}-\text{ss}}\bigl(1,\underline{m},L\bigr)=\mathcal{M}\bigl(1,\underline{m},L\bigr)\simeq B\mathbb{G}_m$, which is clearly $\#A^1$-connected. 
		
		\noindent
		Assume the theorem is true for $(n-1)$. Then by \cite[Remark 2.5, page 106]{MV99}, the moduli stack  $\mathcal{M}^{\mathcal{E}-\text{ss}}\bigl(n-1,\underline{m}',L'\bigr)$ admits a $K$-rational point for any choice of $\boldsymbol{m}'$ and $L'$, and for any finitely generated field extension $K$ over $k$ (for this we note that for such $K$, ${\rm Spec \,} K$ is a Henselian local scheme). Let $E_0$ and $E_1$ be two $\mathcal{E}_K$-semistable vector bundles on $\mathfrak{C}_K$ coming from $\mathcal{M}^{\mathcal{E}-\text{ss}}\bigl(n,\underline{m},L\bigr)({\rm Spec \,} K)$.  Choose an $\mathcal{E}_K$-semistable vector bundle $F$ on $\mathfrak{C}_K$ of rank $(n-1)$ and determinant $L(mn)$ and multiplicities as in the hypothesis of Proposition \ref{prop:surjection}. Thus, the proof of Proposition \ref{prop:concordance} can be applied to obtain a direct $\mathbb{A}^1$-concordance between $E_0$ and $E_1$. This implies the existence of an $\mathcal{E}_K$-semistable vector bundle $\mathscr{E}$ on $\mathfrak{C}_K\times_K \mathbb{A}^1_K$ satisfying $\mathscr{E}\big|_{\mathfrak{C}_K\times\{i\}}\simeq E_i,\,\,i=0,1$. As $\mathfrak{C}_K\times_K \mathbb{A}^1_K\simeq \mathfrak{C}\times_k \mathbb{A}^1_K$, $\mathscr{E}$ corresponds to a morphism
		\begin{align*}
			\varphi: \mathbb{A}^1_K \longrightarrow \mathcal{M}\bigl(n,\underline{m},L\bigr)
		\end{align*}
		satisfying $\varphi(0)=E_0$ and $\varphi(1)=E_1$. Now, the semistable locus of the family $\mathscr{E}$, namely $\mathcal{U}:= \{p\in \mathbb{A}^1_K\,\mid\,\mathscr{E}\big|_{\mathfrak{C}_K\times\{p\}}\,\,\text{is}\,\,\mathcal{E}_K-\text{semistable}\}$, is open in $\mathbb{A}^1_K$. Thus, $\varphi|_{\mathcal{U}}$ factors through the open substack $\mathcal{M}^{\mathcal{E}-ss}\bigl(n,\underline{m},L\bigr)$, such that the diagram
		\begin{align*}
			\xymatrix{
				\mathbb{A}^1_K \ar[rr]^(.45){\varphi} && \mathcal{M}\bigl(n,\underline{m},L\bigr) \\
				\mathcal{U} \ar@{^{(}->}[u] \ar[rr]^(.45){\varphi|_{U}} && \mathcal{M}^{\mathcal{E}-ss}\bigl(n,\underline{m},L\bigr)
				\ar@{^{(}->}[u]
			}
		\end{align*} commutes.
		Since $\mathcal{M}^{\mathcal{E}-ss}\bigl(n,\underline{m},L\bigr)$ is universally closed over $k$ (see~\cite{Hua23}), the map $\varphi\big|_{\mathcal{U}}$ can be extended to a morphism $\widetilde{\varphi}: \mathbb{A}^1_K\longrightarrow \mathcal{M}^{\mathcal{E}-ss}\bigl(n,\underline{m},L\bigr)$. Note that $\{0,1\}\subset \mathcal{U}$ by assumption, and thus their images remain unchanged under the modification $\widetilde{\varphi}$:
		\[\widetilde{\varphi}(i)=\varphi(i)=E_i,\,\,\forall\,\,i=0,1\,.\]
		This gives a direct semistable $\mathbb{A}^1$-concordance between $E_0$ and $E_1$. It follows from Lemma \ref{lem:a1-connected-condition} that $\mathcal{M}^{\mathcal{E}-ss}\bigl(n,\underline{m},L\bigr)$ is $\mathbb{A}^1$-connected.
<!-- end proof-13 -->

<!-- definition-7: body_md -->
\label{def:quasi-parabolic-bundle}
		Fix a quasi-parabolic data $(\boldsymbol{e,m})$. A \textit{quasi-parabolic
			vector bundle of rank $n$} on $C$ is a rank $n$ vector bundle $E$ on $C$
		together with, for each $p\in D$, a flag of type $\underline{m}(p)$ on the
		fiber $E_p$:
		\smallskip
		\[
		E_p = E_{p,0}\supsetneq E_{p,1}\supsetneq \cdots
		\supsetneq E_{p,e_p-1} \supsetneq E_{p,e_p} = 0,
		\]
		satisfying $\dim(E_{p,i}/E_{p,i+1})=m_{p,i}$ for all $0\leq i< e_p$.
<!-- end definition-7 -->
<!-- definition-8: body_md -->
\label{def:families-of-quasi-parabolic-bundles}
		A \textit{family of quasi-parabolic vector bundles of type $(\boldsymbol{e,m})$ 
			on $C$ parametrized by $S$} is a vector bundle $\mathcal{F}$ on $C\times S$
		together with, for each $p\in D$, a section $s_p$ of
		$\mathrm{Flag}_{\underline{m}(p)}\bigl(\mathcal{F}|_{\{p\}\times S}\bigr)
		\longrightarrow S$.

<!-- definition-9: body_md -->
\label{def:parabolic-bundles}
		Let $E$ be a quasi-parabolic vector bundle of rank $n$ on $C$ associated to the data $(\boldsymbol{e,m})$. 
		\begin{enumerate}[(1)]
			\item The $\boldsymbol{\alpha}$-\textit{degree} of $E$ is defined as the real number
			\[\deg_{\boldsymbol{\alpha}}(E) := \deg(E)+\sum_{p\in D}\sum_{i=1}^{e_p}m_{_{p,i}}\alpha_{_{p,i}}.\]
			\item The $\boldsymbol{\alpha}$-\textit{slope} of $E$ is defined to be the real number
			\[\mu_{\boldsymbol{\alpha}}(E):=\deg_{\boldsymbol{\alpha}}(E)/n.\] 
		\end{enumerate}
