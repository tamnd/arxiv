<!-- problem-1: body_md -->
\label{prob-ZL}
For every positive integer $n$, is it true that 
\begin{equation*}
\lambda(G)>\lambda(T_{n,2})
\quad\stackrel{?}{\Longrightarrow}\quad
\operatorname{bk}(G)>\frac{n}{6}.
\end{equation*}
<!-- end problem-1 -->
<!-- theorem-1: body_md -->
\label{thm-confirm-Zhai-Lin}
Let $G$ be an $n$-vertex graph with 
$\lambda (G) \geqslant \lambda(T_{n,2})$ and $G\neq T_{n,2}$. 
Then
\begin{equation*}
\operatorname{bk}(G) \ \geqslant \  \frac{1}{3}\lambda (G).
\end{equation*}
Consequently, if $G$ is an $n$-vertex graph with 
$ \lambda(G)>\lambda(T_{n,2})$,
then 
$\operatorname{bk}(G)\geqslant \left\lfloor\frac n6\right\rfloor+1$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:fixed-order-intro}
Let $G$ be an $n$-vertex graph with 
$\lambda (G) \geqslant \lambda(T_{n,2})$ and $G\neq T_{n,2}$. 
Then
\begin{equation*}
\operatorname{bk}(G) \ \geqslant \  \max\left\{ \lambda (G)-\frac{n}{3} , \, 2\lambda (G) - n \right\}.
\end{equation*}

<!-- problem-2: body_md -->
\label{prob-LLZ}
Does every $m$-edge graph $G$ satisfy
\begin{equation} \label{eq-ZYZZ}
\lambda(G) > \sqrt{m}
\quad\stackrel{?}{\Longrightarrow}\quad
\operatorname{bk}(G)>\frac{1}{3}\sqrt{m}. 
\end{equation}

<!-- theorem-3: body_md -->
\label{thm:edge-spectral-intro}
Let $G$ be a graph with $m\geqslant 1$ edges and
spectral radius $\lambda \geqslant\sqrt m$. Then
\[
\operatorname{bk}(G) \  \geqslant \  \lambda-\frac{2m}{3\lambda},
\]
unless $G$ is a complete bipartite graph.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:two-lambda}
Let $G$ be a graph with $m\geqslant1$ edges and spectral
radius $\lambda$. Then
\begin{equation}\label{eq:two-lambda}
\operatorname{bk}(G)\ \geqslant\ 2\lambda-\frac{2m}{\lambda}.
\end{equation}
Equality holds if and only if $G$ is complete bipartite, or regular complete multipartite.
<!-- end theorem-4 -->
<!-- theorem-5: body_md -->
\label{thm:BN-refined}
Every graph $G$ with $m\geqslant 1$ edges and spectral radius
$\lambda$ satisfies
\begin{equation}\label{eq:BN-refined}
t(G)\ \geqslant\ \frac{1}{3} (\lambda+1)(\lambda^{2}-m).
\end{equation}
Equality holds if and only if $G$ is a complete graph or a
complete bipartite graph.
<!-- end theorem-5 -->
<!-- theorem-6: body_md -->
\label{thm:edit-distance}
For every $\varepsilon>0$, there is $\delta>0$ such that
the following holds. If $G$ is an $n$-vertex graph with
$\lambda(G)\geqslant\bigl(\frac12-\delta\bigr)n$, then either
\[
\operatorname{bk}(G)\geqslant\Bigl(\frac16-\varepsilon\Bigr)n ,
\]
or $G$ can be changed into $T_{n,2}$ by adding and
deleting at most $\varepsilon n^2$ edges.
<!-- end theorem-6 -->
<!-- theorem-7: body_md -->
\label{thm:edge-spectral-stability}
For every $\varepsilon >0$, there exists
$\delta >0$ such that the following holds. 
If $G$ is a graph with $m\geqslant 1$ edges and 
$\lambda (G)\geqslant  (1- \delta) \sqrt{m}$, then either
\[
\operatorname{bk} (G) \geqslant \Bigl(\frac{1}{3} - \varepsilon \Bigr)
\sqrt{m},
\]
or there exist disjoint sets $A,B\subseteq V(G)$ such
that
$\bigl|E(G)\, {\triangle}\, E(K_{A,B})\bigr|
\leqslant \varepsilon m$.

<!-- lemma-1: body_md -->
\label{lem:degree-neighborhood}
For every vertex $v$ of $G$,
\begin{equation}\label{eq:degree-neighborhood}
\sum_{u\in N(v)}d(u)=m+t_v-s_v .
\end{equation}
Consequently,
\begin{equation}\label{eq:TS}
T-S=\lambda^2-m .
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
On the left-hand side of
\eqref{eq:degree-neighborhood} an edge with both ends
in $N(v)$ is counted twice, an edge with both ends
outside $N(v)\cup \{v\}$ is counted not at all, and every other
edge is counted once. This is exactly the count on the
right-hand side.

Multiplying \eqref{eq:degree-neighborhood} by $x_v$ and
summing over $v$ gives
\[
m+T-S
=\sum_{v\in V(G)}x_v\sum_{u\in N(v)}d(u)
=\sum_{u\in V(G)}d(u)\sum_{v\in N(u)}x_v
=\lambda\sum_{u\in V(G)}d(u)x_u ,
\]
where we used $\|\bm{x}\|_1=1$ in the first equality
and $A(G)\bm{x}=\lambda\bm{x}$ in the last. Summing the
eigenvalue equations over all vertices gives
$\sum_u d(u)x_u=\lambda\sum_u x_u=\lambda$, and
\eqref{eq:TS} follows.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:charging}
For every vertex $v$ of $G$,
\begin{equation}\label{eq:local}
\sum_{u\in N(v)}t_u\leqslant \beta \,(2t_v+s_v).
\end{equation}
Consequently,
\begin{equation}\label{eq:rhoT}
\lambda\,T\leqslant \beta \,(2T+S).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The left-hand side of \eqref{eq:local} counts the pairs
consisting of a triangle and one of its vertices lying
in $N(v)$, so
\begin{equation}\label{eq:inter-step}
\sum_{u\in N(v)}t_u
=\sum_{K\in\mathcal{T}(G)}\bigl|V(K)\cap N(v)\bigr| .
\end{equation}
For $K\in\mathcal{T}(G)$ let $a_K$ be the number of edges of $K$
inside $N(v)$, and let $c_K$ be the number of edges of
$K$ inside $V(G)\setminus (N(v)\cup \{v\})$. We claim that
\begin{equation}\label{eq:pointwise}
\bigl|V(K)\cap N(v)\bigr|\leqslant 2a_K+c_K .
\end{equation}
If $v\in V(K)$, then the other two vertices of $K$ lie
in $N(v)$, and the two sides of \eqref{eq:pointwise}
equal $2$ and $2\cdot 1+0$. If $v\notin V(K)$, put
$k=|V(K)\cap N(v)|$; then $a_K=\binom{k}{2}$ and
$c_K=\binom{3-k}{2}$, so for $k=0,1,2,3$ the right-hand
side of \eqref{eq:pointwise} equals $3,\,1,\,2,\,6$,
each of which is at least $k$.

Summing \eqref{eq:pointwise} over all triangles and
using \eqref{eq:inter-step},
\[
\sum_{u\in N(v)}t_u
\leqslant 2\sum_{K\in\mathcal{T}(G)}a_K+\sum_{K\in\mathcal{T}(G)}c_K .
\]
There are $t_v$ edges inside $N(v)$ and each of them
lies in at most $\beta $ triangles, so
$\sum_{K}a_K\leqslant \beta \,t_v$; likewise
$\sum_{K}c_K\leqslant \beta \,s_v$. This proves \eqref{eq:local}.

Multiplying \eqref{eq:local} by $x_v$ and summing over
$v$, the left-hand side becomes
\[
\sum_{v\in V(G)}x_v\sum_{u\in N(v)}t_u
=\sum_{u\in V(G)}t_u\sum_{v\in N(u)}x_v
=\lambda\sum_{u\in V(G)}t_ux_u=\lambda T ,
\]
while the right-hand side becomes $\beta (2T+S)$. This is
\eqref{eq:rhoT}.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:T-upper}
We have
\begin{equation}\label{eq:T-upper}
T\leqslant\frac{\beta \lambda}{2}.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
For a fixed vertex $v$, the sum of the codegrees over
the edges incident with $v$ counts every triangle
containing $v$ twice, so
$2t_v=\sum_{u\in N(v)}t_{uv}\leqslant \beta\,d(v)$. Multiplying
by $x_v$ and summing over $v$ gives
$2T\leqslant \beta \sum_v d(v)x_v=\beta \lambda$.
<!-- end proof-3 -->
<!-- lemma-4: body_md -->
\label{lem:triangle}
Suppose that $m\leqslant\lambda^2$. If $G$ is not complete
bipartite, then $T>0$; in particular $G$ contains a
triangle.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Suppose that $T=0$. Since $x_v>0$ for every $v$, we
have $t_v=0$ for every $v$, so $G$ is triangle-free. By
\eqref{eq:TS}, $S=m-\lambda^2\leqslant 0$, while $S\geqslant 0$
because every $s_v$ is nonnegative. Hence $S=0$, so $s_w=0$ for every $w\in V(G)$: every edge
of $G$ has an end in $N[w]$. An edge incident with $w$ has  its other end in $N(w)$, and an edge not incident with $w$ meets $N[w]$ only in $N(w)$. Therefore
\begin{equation}\label{eq:hit}
\text{every edge of }G\text{ meets }N(w)
\quad\text{for every }w\in V(G).
\end{equation} 
Fix an edge $uv$ and put $A=N(u)$ and $B=N(v)$. As $G$
is triangle-free, we have 
$A\cap B=\varnothing$ and both $A$
and $B$ are independent; moreover $v\in A$ and $u\in B$.
For every vertex $w$, applying \eqref{eq:hit} to the edge
$uv$ shows that $u\in N(w)$ or $v\in N(w)$, that is,
$w\in N(u)\cup N(v)$. Hence $V(G)=A\cup B$.

Let $a\in A$ and $b\in B$. Then $vb\in E(G)$. If $a=v$,
then $ab=vb\in E(G)$. If $a\ne v$, then $a,v\in A$ and
$A$ is independent, so $v\notin N(a)$; applying
\eqref{eq:hit} to the edge $vb$ gives $b\in N(a)$. In
either case $ab\in E(G)$. Therefore $G=K_{A,B}$ is
complete bipartite, a contradiction.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
Write $\lambda=\lambda(G)$, choose a component $H$ of
$G$ with $\lambda(H)=\lambda$, and put $h=e(H)$ and
$\beta =\operatorname{bk}(H)$. Then
$ h\leqslant m\leqslant\lambda^2 $.

If $H$ were a complete bipartite graph $K_{p,q}$,
then $\lambda^2=pq=h$, so $h=m$ and every component
of $G$ other than $H$ is a single vertex; that is,
$G$ is the disjoint union of a complete bipartite
graph and isolated vertices, contrary to the
hypothesis. Hence $H$ is not complete bipartite.

Apply the notation above to $H$, and put
$\delta=\lambda^2-h\geqslant 0$, so that \eqref{eq:TS} reads
$T-S=\delta$. By Lemma \ref{lem:triangle}, $T>0$ and
$H$ contains a triangle, whence $\beta \geqslant 1$. Substituting
$S=T-\delta$ into \eqref{eq:rhoT} gives
$\lambda T\leqslant \beta (3T-\delta)$, that is,
\begin{equation}\label{eq:first-chain}
(3\beta -\lambda)\,T\ \geqslant\ \beta \,\delta\ \geqslant\ 0 .
\end{equation}
Since $T>0$, we obtain $3\beta -\lambda\geqslant 0$. Multiplying
\eqref{eq:T-upper} by $3\beta -\lambda\geqslant 0$ and combining
the result with \eqref{eq:first-chain},
\[
\beta \,\delta\ \leqslant\ (3\beta -\lambda)\,T\ \leqslant\
\frac{\beta \lambda(3\beta -\lambda)}{2}.
\]
Cancelling $\beta >0$ gives
$2\lambda^{2}-2h\leqslant 3\beta \lambda-\lambda^{2}$, and
therefore
\[
\operatorname{bk}(G)\ \geqslant\ \beta \ \geqslant\
\lambda-\frac{2h}{3\lambda}\ \geqslant\
\lambda-\frac{2m}{3\lambda},
\]
the last step because $h\leqslant m$.
<!-- end proof-5 -->
<!-- proposition-1: body_md -->
\label{prop:equality-edge}
Let $G$ be a graph with $m\geqslant1$ edges, and let $\lambda=\lambda(G)$ and $\beta=\operatorname{bk}(G)$.
Suppose that $\lambda^2>m$. Then
\[
\beta=\lambda-\frac{2m}{3\lambda}
\]
if and only if $G$ is connected and $K_4$-free, every edge of
$G$ lies in exactly $\beta$ triangles, and $G-N[v]$ is
triangle-free for every vertex $v$.
<!-- end proposition-1 -->
<!-- proof-6: body_md -->
Let $H$ be a component of $G$ with $\lambda(H)=\lambda$, and
put $h=e(H)$. Since $h\leqslant m<\lambda^2=\lambda(H)^2$, the
component $H$ is not complete bipartite, so the proof of
Theorem \ref{thm:edge-spectral-intro} gives 
$\operatorname{bk}(G)\geqslant\operatorname{bk}(H)\geqslant\lambda-\frac{2h}{3\lambda}
\geqslant\lambda-\frac{2m}{3\lambda}$, so equality forces $h=m$.
Then every other component is an isolated vertex, and $G=H$
is connected. Since connectivity is also one of the
conditions on the right-hand side, we may assume that $G$ is
connected. 
Put $\delta=\lambda^2-m>0$. A connected complete bipartite
graph satisfies $\lambda^2=m$, so $G$ is not complete
bipartite, and Lemma \ref{lem:triangle} gives $T>0$ and
$\beta\geqslant1$. The proof of Theorem
\ref{thm:edge-spectral-intro} yields the chain
\begin{equation}\label{eq:equality-chain}
\beta\delta\ \leqslant\ (3\beta-\lambda)\,T\ \leqslant\
\frac{\beta\lambda(3\beta-\lambda)}{2}.
\end{equation}
Its first inequality is \eqref{eq:rhoT} after the substitution
$S=T-\delta$, and its second inequality is \eqref{eq:T-upper}
multiplied by $3\beta-\lambda\geqslant0$. Since $\beta>0$, the
equality $\beta=\lambda-\frac{2m}{3\lambda}$ is equivalent to
$2\delta=\lambda(3\beta-\lambda)$, that is, to equality
throughout \eqref{eq:equality-chain}.

Suppose that equality holds throughout
\eqref{eq:equality-chain}. If $3\beta=\lambda$, then
\eqref{eq:equality-chain} gives $\beta\delta\leqslant0$, which is
impossible.  Hence $3\beta>\lambda$, and equality in the second inequality means $T=\frac{\beta\lambda}{2}$, that is, $\sum_{v}x_v\bigl(\beta d(v)-2t_v\bigr)=0$. Each summand is nonnegative, because $2t_v=\sum_{u\in N(v)}t_{uv}\leqslant\beta d(v)$ and $x_v>0$. Hence $2t_v=\beta d(v)$ for every $v$, and so $t_{uv}=\beta$ for every edge $uv$.  Equality in the first inequality is equality in
\eqref{eq:rhoT}, hence in \eqref{eq:local} for every vertex
$v$, because every $x_v$ is positive. In the proof of Lemma
\ref{lem:charging}, \eqref{eq:local} was obtained by summing
\eqref{eq:pointwise} over all triangles and then applying
$\sum_K a_K\leqslant\beta t_v$ and $\sum_K c_K\leqslant\beta s_v$. Hence
\eqref{eq:pointwise} is an equality for every triangle $K$.
For $v\notin V(K)$, the two sides of \eqref{eq:pointwise}
agree only when $k=|V(K)\cap N(v)|\in\{1,2\}$. Thus no vertex
is adjacent to all three vertices of a triangle, so $G$ is
$K_4$-free; and no triangle avoids $N[v]$, so $G-N[v]$ is
triangle-free.

Conversely, suppose that $G$ is connected and  $K_4$-free, that every edge
lies in exactly $\beta$ triangles, and that $G-N[v]$ is
triangle-free for every $v$. The second condition gives
$2t_v=\sum_{u\in N(v)}t_{uv}=\beta d(v)$ for every $v$, so
$T=\frac{1}{2} \beta\lambda$; it also gives
$\sum_K a_K=\beta t_v$ and $\sum_K c_K=\beta s_v$. The first
and third conditions give $k\in\{1,2\}$ for every triangle $K$
and every $v\notin V(K)$, so \eqref{eq:pointwise} is an
equality in every case. Hence \eqref{eq:local} and therefore
\eqref{eq:rhoT} are equalities; that is,
$(3\beta-\lambda)T=\beta\delta$. Together with
$T=\frac{1}{2} \beta\lambda$, this gives
$2\delta=\lambda(3\beta-\lambda)$, which is the desired
equality.

<!-- proof-7: body_md -->
Choose a component $H$ of $G$ with $\lambda(H)=\lambda$,
and put $h=e(H)$ and $b=\operatorname{bk}(H)$; then $h\leqslant m$ and
$b\leqslant\operatorname{bk}(G)$. Apply the notation of Subsection \ref{sec:three-ineq} to $H$: let $\bm{x}$ be a Perron
vector of $H$ with $\|\bm{x}\|_1=1$, so $x_v>0$ for
every $v\in V(H)$, and let $t_v$, $s_v$, $T$, $S$ be the
associated quantities. Since every $s_v$ is
nonnegative, so is $S$, and \eqref{eq:TS} gives
\begin{equation}\label{eq:two-lambda-lower}
T=S+\lambda^{2}-h\ \geqslant\ \lambda^{2}-h .
\end{equation}
Combining this with $T\leqslant b\lambda/2$ from
\eqref{eq:T-upper} yields
$\lambda^{2}-h\leqslant b\lambda/2$, that is,
\[
\operatorname{bk}(G)\ \geqslant\ b\ \geqslant\ 2\lambda-\frac{2h}{\lambda}
\ \geqslant\ 2\lambda-\frac{2m}{\lambda},
\]
the last step because $h\leqslant m$. This proves
\eqref{eq:two-lambda}.

Suppose now that equality holds in
\eqref{eq:two-lambda}. Then $h=m$, so every component of
$G$ other than $H$ is a single vertex, and both
\eqref{eq:two-lambda-lower} and \eqref{eq:T-upper} are
equalities.

Equality in \eqref{eq:two-lambda-lower} means $S=0$,
hence $s_v=0$ for every $v\in V(H)$. Thus for every
vertex $v$ the set $V(H)\setminus N_H[v]$ is
independent, that is, $u\not\sim v$ and $w\not\sim v$
imply $u\not\sim w$ for all distinct $u,w$. Therefore
the relation ``$u=w$ or $uw\notin E(H)$'' is an
equivalence relation on $V(H)$; its classes are
independent sets, and any two distinct classes are
completely joined. Hence $H$ is a complete multipartite
graph, say with parts $V_1,\dots,V_r$ of sizes
$n_1,\dots,n_r$, where $r\geqslant2$ and $n=|V(H)|$.

Equality in \eqref{eq:T-upper} means
$\sum_{v\in V(H)}x_v\bigl(b\,d(v)-2t_v\bigr)=0$. Each
summand is nonnegative because
$2t_v=\sum_{u\in N(v)}t_{uv}\leqslant b\,d(v)$, and $x_v>0$,
so $2t_v=b\,d(v)$ for every $v$; consequently
$t_{uv}=b$ for every edge $uv$ of $H$. For $u\in V_i$
and $v\in V_j$ with $i\ne j$ we have
$t_{uv}=n-n_i-n_j$, so $n_i+n_j$ takes the same value
for all $i\ne j$. If $r=2$ this is no restriction, and
$H$ is complete bipartite; if $r\geqslant3$, then comparing
the pairs $\{i,j\}$ and $\{i,k\}$ gives $n_j=n_k$ for
all $j\ne k$, so all parts have the same size.

Conversely, let $G$ be complete multipartite with parts of sizes $n_1,\dots,n_r$, where either $r=2$ or all parts have the same size $k$. Then $s_v=0$ for every $v$, so $S=0$ and $T=\lambda^2-m$ by \eqref{eq:TS}. If $r=2$, then $G$ is triangle-free and $T=0=\operatorname{bk}(G)\lambda/2$; if all parts have size $k$, then every edge of $G$ has codegree $n-2k=\operatorname{bk}(G)$, so $2t_v=\operatorname{bk}(G)\,d(v)$ for every
$v$ and again $T=\operatorname{bk}(G)\lambda/2$. In both cases
$\lambda^2-m=\operatorname{bk}(G)\lambda/2$, which is equality in \eqref{eq:two-lambda}.

If $\lambda^{2}<m$, then the right-hand side of
\eqref{eq:BN-refined} is negative and the inequality is
trivial. So assume $\lambda^{2}\geqslant m$. Choose a connected
component $H$
of $G$ with $\lambda(H)=\lambda$, and apply the
notation of Subsection \ref{sec:three-ineq} to $H$: let
$\bm{x}$ be a Perron vector of $H$ with 
$\|\bm{x}\|_{1}=1$, and let $T$ and $S$ be the
associated quantities. Since $S\geqslant 0$ and
$e(H)\leqslant m$, the identity \eqref{eq:TS} gives
\begin{equation}\label{eq:T-lower-supersat}
T\ \geqslant\ \lambda^{2}-e(H)\ \geqslant\ \lambda^{2}-m.
\end{equation}
Grouping the sum $T=\sum_{v}x_{v}t_{v}$ by triangles,
we get $T=\sum_{K\in\mathcal{T}(H)}\sum_{v\in V(K)}x_{v}$, and
hence
\begin{equation}\label{eq:T-xmax}
T\ \leqslant\ 3\,x_{\max}\,t(H),
\qquad\text{where }
x_{\max}:=\max_{v\in V(H)}x_{v}.
\end{equation}
Finally, let $v^{*}$ be a vertex with
$x_{v^{*}}=x_{\max}$. The eigenvalue equation at
$v^{*}$ and $\|\bm{x}\|_{1}=1$ yield
\[
\lambda\,x_{\max}
=\sum_{u\in N(v^{*})}x_{u}
\ \leqslant\ \sum_{u\neq v^{*}}x_{u}
=1-x_{\max},
\]
so $x_{\max}\leqslant\frac{1}{1+\lambda}$. Using \eqref{eq:T-lower-supersat} and
\eqref{eq:T-xmax}, we get 
$\frac{3\,t(H)}{1+\lambda}\geqslant\lambda^{2}-m$, and
\eqref{eq:BN-refined} follows from $t(G)\geqslant t(H)$.

Suppose now that equality holds in
\eqref{eq:BN-refined}. Assume first that
$t(G)>0$, so $\lambda^{2}>m$. Then $t(G)=t(H)$,
$e(H)=m$, and \eqref{eq:T-lower-supersat},
\eqref{eq:T-xmax} and
$x_{\max}\leqslant\frac{1}{1+\lambda}$ are all equalities.

are isolated vertices. As in the proof of Theorem
\ref{thm:two-lambda}, the equality $S=0$ implies that
$H$ is a complete multipartite graph. Equality in
\eqref{eq:T-xmax} means that $x_{v}=x_{\max}$ for
every vertex $v$ lying in a triangle of $H$; since $H$
is complete multipartite and contains a triangle, it
has at least three parts, so every vertex of $H$ lies
in a triangle. Hence $\bm{x}$ is constant, $H$ is
regular, and all parts of $H$ have the same size.
Equality in $x_{\max}\leqslant\frac{1}{1+\lambda}$ means
that $x_{u}=0$ for every $u\notin N[v^{*}]$; as
$\bm{x}$ is positive on $V(H)$, the vertex $v^{*}$
dominates $H$, so its part is a singleton. Therefore
all parts are singletons and $H$ is a complete graph.

$H$ is a triangle-free connected graph with
$e(H)\leqslant\lambda^{2}$, so Lemma \ref{lem:triangle}
shows that $H$ is complete bipartite; now
$\lambda^{2}=e(H)$ forces $e(H)=m$, and all remaining
components are isolated vertices.

Conversely, if $G$ is a complete graph $K_{a}$, then $t(G)=\binom{a}{3}$, $\lambda=a-1$ and
$m=\binom{a}{2}$, so both sides of
\eqref{eq:BN-refined} equal $\frac{1}{6}a(a-1)(a-2)$.
If $G$ is a complete bipartite graph, then $t(G)=0$ and
$\lambda^{2}=m$.

<!-- definition-1: body_md -->
\label{def:tight}
A graph $G$ on $n$ vertices is \emph{tight} if $G$ is
$\frac n2$-regular, $G$ contains a triangle, and every edge of
$G$ lies in either $0$ or exactly $\frac n6$ triangles.
<!-- end definition-1 -->
<!-- lemma-5: body_md -->
\label{lem:count}
Let $G$ be an $\frac n2$-regular graph on $n$ vertices and let
$uvw$ be a triangle of $G$. Then
\begin{equation}\label{eq:count}
t_{uv}+t_{uw}+t_{vw}\ \geqslant\ \frac n2 ,
\end{equation}
with equality if and only if every vertex outside the triangle
is adjacent to exactly one or exactly two of $u,v,w$.
Thus, if $G$ contains a triangle, then
$\operatorname{bk}(G)\geqslant\frac n6$, with equality if and only if $G$ is
tight.
<!-- end lemma-5 -->
<!-- proof-12: body_md -->
For an edge $xy$ let $S_{xy}$ be the set of vertices adjacent to
both $x$ and $y$ or to neither of them. Since
$d(x)=d(y)=\frac n2$, we have $|N(x)\cup N(y)|=n-t_{xy}$, and
therefore $|S_{xy}|=2t_{xy}$. Each of $u,v,w$ lies in exactly one
of the sets $S_{uv},S_{uw},S_{vw}$. A vertex $z\notin\{u,v,w\}$
adjacent to exactly $j$ of $u,v,w$ lies in
$\binom j2+\binom{3-j}2$ of them; this number is $3$ for
$j\in\{0,3\}$ and $1$ for $j\in\{1,2\}$. Hence
$|S_{uv}|+|S_{uw}|+|S_{vw}|\geqslant n$, with equality exactly when
$j\in\{1,2\}$ for every $z$. This proves \eqref{eq:count}.

If $G$ contains a triangle, then \eqref{eq:count} gives
$3\operatorname{bk}(G)\geqslant\frac n2$. If $\operatorname{bk}(G)=\frac n6$, then \eqref{eq:count}
forces $t_{e}=\frac n6$ for every edge $e$ of every triangle, so
$G$ is tight. Conversely, in a tight graph every edge lies in at
most $\frac n6$ triangles, and \eqref{eq:count} applied to any
triangle shows that some edge lies in exactly $\frac n6$ of them.

\label{prop:equality}
Let $G\ne T_{n,2}$ be an $n$-vertex graph with
$\lambda(G)\geqslant\lambda(T_{n,2})$. Then $\operatorname{bk}(G)=\frac13\lambda(G)$
if and only if $G$ is tight. In particular, equality in Theorem
\ref{thm-confirm-Zhai-Lin} forces $\lambda(G)=\lambda(T_{n,2})$.

<!-- proof-13: body_md -->
A tight graph satisfies $\lambda(G)=\frac n2=\lambda(T_{n,2})$,
is not $T_{n,2}$, and has $\operatorname{bk}(G)=\frac n6$ by Lemma
\ref{lem:count}.
Conversely, let $\operatorname{bk}(G)=\frac13\lambda$ with $\lambda=\lambda(G)$.
In the proof of Theorem \ref{thm-confirm-Zhai-Lin}, the case $\lambda<\sqrt m$
gives $\lambda<\frac n2$ and $\operatorname{bk}(G)>\frac n6>\frac13\lambda$,
and the case $\lambda\geqslant\sqrt m$ gives
$\operatorname{bk}(G)\geqslant\lambda-\frac{2m}{3\lambda}\geqslant\frac13\lambda$, where
the second step is an equality only if $m=\lambda^2$. Hence, we obtain 
$m=\lambda^2$. Now $\lambda\geqslant\frac{2m}{n}=\frac{2\lambda^2}{n}$
yields $\lambda\leqslant\frac n2$ and $m=\lambda^2\leqslant\frac{n^2}{4}$,
and $\lambda^2\geqslant\lfloor n^2/4\rfloor$ yields
$m=\lfloor n^2/4\rfloor$.

$\operatorname{bk}(G)=\frac13\lambda=\frac16\sqrt{n^2-1}$ would be a positive
integer $k$, so $(n-6k)(n+6k)=1$, which is impossible. Thus $n$
is even, $m=\frac{n^2}{4}$ and $\lambda=\frac n2=\frac{2m}{n}$,
so $G$ is $\frac n2$-regular and
$\lambda(G)=\frac n2=\lambda(T_{n,2})$. By Lemma
\ref{lem:count}, we see that $\operatorname{bk}(G)=\frac n6$ forces $G$ to be tight.

<!-- lemma-7: body_md -->
\label{lem:parity}
Let $G$ be a tight graph on $n=6\beta$ vertices in which every
edge lies in a triangle. Then $\beta$ is even, that is,
$n\equiv 0\pmod{12}$.
<!-- end lemma-7 -->
<!-- proof-15: body_md -->
Fix a triangle $uvw$. By Lemma \ref{lem:structure}(a), the
vertices outside the triangle split into six sets: for
$x\in\{u,v,w\}$, let $Y_x$ be the set of vertices adjacent to $x$
only, and let $X_x$ be the set of vertices adjacent to the two vertices other than $x$. 
In the notation of Lemma
\ref{lem:structure}(b), applied to the edge $vw$, we have $Y_u=R$ and, since $G$ is $K_4$-free,  $X_u=W\setminus\{u\}$. Hence, we get 
$|Y_u|=\beta$, $|X_u|=\beta-1$, and $Y_u$ is completely joined to
$X_u\cup\{u\}$. Moreover, we have $N(u)=\{v,w\}\cup X_v\cup X_w\cup Y_u$.

Let $y\in Y_u$. The edge $uy$ lies in a triangle, so
$t_{uy}=\beta$. As $y$ is adjacent to neither $v$ nor $w$, the
$\beta$ common neighbors of $u$ and $y$ lie in
$X_v\cup X_w\cup Y_u$. Together with $u$ and the $\beta-1$
vertices of $X_u$, this accounts for $2\beta$ neighbors of $y$;
the remaining $\beta$ neighbors of $y$ lie in $Y_v\cup Y_w$.
Hence $e(Y_u,Y_v)+e(Y_u,Y_w)=\beta^{2}$, and the same holds
with $u$ replaced by $v$ or by $w$. Adding the three equations
gives
$2\bigl(e(Y_u,Y_v)+e(Y_u,Y_w)+e(Y_v,Y_w)\bigr)=3\beta^{2}$;
halving this and subtracting the equation for $u$ leaves
$e(Y_v,Y_w)=\beta^{2}/2$. Therefore $\beta^{2}$ is even, and
so is $\beta$.

<!-- proposition-3: body_md -->
\label{prop:equality-fixed}
Let $G\ne T_{n,2}$ be an $n$-vertex graph with
$\lambda:=\lambda(G)\geqslant\lambda(T_{n,2})$.
\begin{enumerate}
\item[\rm(a)] $\operatorname{bk}(G)=2\lambda-n$ if and only if $G=T_{n,r}$
for some $r\geqslant3$ with $r\mid n$.
\item[\rm(b)] $\operatorname{bk}(G)=\lambda-\frac n3$ if and only if $G$ is
$d$-regular for some $d\geqslant\frac n2$ and, in addition, either
$d=\frac n2$, $G$ has a triangle and every edge of $G$ lies in $0$ or $\frac{n}{6}$ triangles, 
or $d>\frac n2$ and $G$ is
$K_4$-free, every edge of $G$ lies in exactly $\operatorname{bk}(G)$
triangles, and $G-N[v]$ is triangle-free for every vertex $v$.
\end{enumerate}
<!-- end proposition-3 -->
<!-- proof-16: body_md -->
Throughout we use $\lambda\geqslant\frac{2m}{n}$, with equality if
and only if $G$ is regular. It gives $\frac{2m}{\lambda}\leqslant n$
and $\frac{2m}{3\lambda}\leqslant\frac n3$, and each of these two
inequalities is an equality exactly when $G$ is regular. Note
also that $\lambda\geqslant\lambda(T_{n,2})>0$, so a regular $G$ has
positive degree and hence no isolated vertex.

For (a), Theorem \ref{thm:two-lambda} gives
$\operatorname{bk}(G)\geqslant2\lambda-\frac{2m}{\lambda}\geqslant2\lambda-n$. If
$\operatorname{bk}(G)=2\lambda-n$, then $G$ is regular and equality holds in
Theorem \ref{thm:two-lambda}, so $G$ is complete bipartite or
regular complete multipartite. A regular complete bipartite
graph on $n$ vertices is $T_{n,2}$, which is excluded; hence
$G=T_{n,r}$ for some $r\geqslant3$ with $r\mid n$. Conversely,
$\lambda(T_{n,r})=\frac{r-1}{r}n>\frac n2\geqslant\lambda(T_{n,2})$
and $\operatorname{bk}(T_{n,r})=2\lambda(T_{n,r})-n$.

For (b), suppose that $\operatorname{bk}(G)=\lambda-\frac n3$. In the proof
of Theorem \ref{thm-confirm-Zhai-Lin}, the case $\lambda<\sqrt m$ gives
$\operatorname{bk}(G)>\frac n6>\lambda-\frac n3$; hence $\lambda\geqslant\sqrt m$,
and Theorem \ref{thm:edge-spectral-intro}, applied to $G$ with
its isolated vertices deleted, gives
$\operatorname{bk}(G)\geqslant\lambda-\frac{2m}{3\lambda}\geqslant\lambda-\frac n3$.
Therefore $G$ is regular, say of degree $d$, and
$\operatorname{bk}(G)=\lambda-\frac{2m}{3\lambda}$. Now $\lambda=d$ and
$m=\frac{nd}{2}$, so $\lambda^{2}-m=d\bigl(d-\frac n2\bigr)$,
and $\lambda\geqslant\sqrt m$ forces $d\geqslant\frac n2$. If
$d=\frac n2$, then $\lambda=\frac n2=\lambda(T_{n,2})$ and
$\operatorname{bk}(G)=\frac n6=\frac13\lambda$, so $G$ is tight by
Proposition \ref{prop:equality}. If $d>\frac n2$, then
$\lambda^{2}>m$ and Proposition \ref{prop:equality-edge}
yields the three stated conditions. 
Conversely, a tight graph satisfies
$\operatorname{bk}(G)=\frac n6=\lambda-\frac n3$. Suppose that $G$ is
$d$-regular with $d>\frac n2$ and satisfies the three conditions. Every component of a $d$-regular graph has at least $d+1>\frac n2+1$ vertices, so $G$ is connected.
Moreover, $\lambda=d$ and $m=\frac{nd}{2}<d^{2}$ give
$\lambda^{2}>m$, and Proposition
\ref{prop:equality-edge} gives 
$\operatorname{bk}(G)=\lambda-\frac{2m}{3\lambda}=\lambda-\frac n3$.

\begin{equation*}

\end{equation*}

{\rm (a)} $G$ contains a book of size 
$\operatorname{bk}(G)>\left(\frac16-2\alpha^{1/3}\right)n$; 
{\rm (b)} $G$ contains an induced bipartite subgraph $H$ such that 
$ |H|\geqslant \left(1-\alpha^{1/3} \right)n$ and

<!-- lemma-8: body_md -->
\label{lem:bipartite-completion}
Let $G$ be a bipartite graph with $m $ edges and spectral radius $\lambda=\lambda(G)$. Then there exist disjoint vertex sets $X,Y\subseteq V(G)$ such that
\begin{equation*}
\bigl| E(G)\, {\triangle}\, E(K_{X,Y})\bigr|
\leqslant 3\bigl( m -\lambda^2\bigr).
\end{equation*}
Moreover, $X$ and $Y$ can be taken to be
$N(v)$ and $N(u)$ for a suitable edge $uv$ of $G$.
<!-- end lemma-8 -->
<!-- proof-21: body_md -->
If $m=0$, take $X=Y=\varnothing$. Assume $m\geqslant1$.
Fix a bipartition $V(G)=U\sqcup W$, and let $M$ be the
corresponding biadjacency matrix: a $0$--$1$ matrix whose rows
are indexed by $U$ and whose columns are indexed by $W$, with
$M_{uw}=1$ if and only if $uw\in E(G)$. Then $\lambda$ is the
largest singular value of $M$. Write $r=\operatorname{rank} M$
and 
$\sigma_1=\lambda\geqslant\sigma_2\geqslant\cdots \geqslant \sigma_r > 0$
for the positive singular values of $M$. Since $M$ has exactly $m$ entries equal to $1$, we have 
\begin{equation*}
\sum_i\sigma_i^2=\lVert M\rVert_F^2=m.
\end{equation*}
Put
 $\xi :=m-\lambda^2=\sum_{i\geqslant2}\sigma_i^2$. 
In particular, we have $\xi\geqslant0$. 
Let
\begin{equation*}
\mathcal D:=
\sum_{\substack{I\subseteq U, |I|=2\\
J\subseteq W, |J|=2}}
\det(M[I,J])^2
\end{equation*}
be the sum of the squared $2\times2$ minors of $M$. By the
Cauchy--Binet formula applied to $MM^{\mathsf T}$, for every
$I\subseteq U$ with $|I|=2$, we have
$$ \det\bigl((MM^{\mathsf T})[I,I]\bigr)
=\sum_{J\subseteq W, |J|=2}\det(M[I,J])^2.$$
Summing over all such $I$ expresses $\mathcal D$ as the sum
of the principal $2\times2$ minors of $MM^{\mathsf T}$,
which is the second elementary symmetric function of its
eigenvalues. The nonzero eigenvalues of $MM^{\mathsf T}$
are $\sigma_1^2,\ldots,\sigma_r^2$, so
\begin{equation*}
\mathcal D=\sum_{i<j}\sigma_i^2\sigma_j^2.
\end{equation*}
By the definition of $\xi$, we obtain
\begin{align}
\mathcal D
=\lambda^2\xi+
  \sum_{2\leqslant i<j}\sigma_i^2\sigma_j^2 
  \leqslant \lambda^2\xi+\frac{\xi^2}{2} 
  =(m-\xi)\xi+\frac{\xi^2}{2} 
  \leqslant m \,\xi.\label{eq:minor-defect-upper}
\end{align}
For an edge $uv\in E(G)$ with $u\in U$ and $v\in W$, we define
\[
B_{uv}:=N(v)\times N(u).
\]
Thus $B_{uv}$ is the edge set of the complete bipartite graph $K_{N_G(v),N_G(u)}$. Set
\[
D_{uv}:=\bigl|E(G)\, {\triangle}\, B_{uv}\bigr|.
\]
We now estimate the average of $D_{uv}$ over all edges $uv$. 

Because $M$ is a $(0,1)$-matrix, a $2\times 2$ minor of $M$ is nonzero precisely when one diagonal consists of two entries equal to $1$, whereas the product of the two entries on the other diagonal is $0$. Suppose that $uv,u'v'\in E(G)$ and that $u'v'\notin B_{uv}$. Then at least one of $uv'$ and $u'v$ is absent. 
In particular $u'\ne u$ and $v'\ne v$, because $u'=u$ would give $uv'=u'v'\in E(G)$ and $v'=v$ would give $u'v=u'v'\in E(G)$. Hence the submatrix on rows $u,u'$ and columns $v,v'$ has nonzero determinant. Conversely, every nonzero $2\times2$ minor has a unique diagonal consisting of two present edges, and each orientation of that diagonal gives one ordered pair $(uv,u'v')$ with $u'v'\notin B_{uv}$. Consequently, we obtain 
\begin{equation}\label{eq:edges-outside-rectangles}
\sum_{uv\in E(G)}|E(G)\setminus B_{uv}|=2\mathcal D.
\end{equation}

Next suppose that $u'v'\in B_{uv}\setminus E(G)$. By the definition of $B_{uv}$, we have 
\[
uv,\quad uv',\quad u'v\in E(G),
\qquad
u'v'\notin E(G).
\]
Since $uv'\in E(G)$ and $u'v'\notin E(G)$, we have $u'\ne u$; likewise $v'\ne v$. Hence the corresponding $2\times2$ submatrix of $M$ has exactly three entries equal to $1$ and has nonzero determinant. Conversely, each $2\times2$ submatrix with exactly three entries equal to $1$ determines exactly one pair consisting of the edge opposite the missing entry and that missing entry. Therefore
\begin{equation}\label{eq:missing-inside-rectangles}
\sum_{uv\in E(G)}|B_{uv}\setminus E(G)|\leqslant\mathcal D.
\end{equation}
Combining \eqref{eq:edges-outside-rectangles} and \eqref{eq:missing-inside-rectangles}, we find that
\[
\sum_{uv\in E(G)}D_{uv}\leqslant3\mathcal D.
\]
Averaging over the $m$ edges, some edge $uv\in E(G)$
satisfies $D_{uv}\leqslant\frac{3\mathcal D}{m}$.
By \eqref{eq:minor-defect-upper},
we have $\mathcal{D} \leqslant m\,\xi$ and  
$D_{uv}\leqslant3\xi=3(m-\lambda^2)$.
Taking $X=N(v)$ and $Y=N(u)$, we obtain the desired complete bipartite graph $K_{X,Y}$.

$\eta<1$, we have $q<m$, so $\ell\geqslant1$ and $F$ has an edge. 
The matrix $A(G)-A(F)$ is symmetric with exactly $2q$ nonzero entries, each equal to $1$, so $\lVert A(G)-A(F)\rVert_F=\sqrt{2q}$. Weyl's inequality and
$\lVert\cdot\rVert_2\leqslant\lVert\cdot\rVert_F$  give
\[
\lambda_F\ \geqslant\ \lambda-\lVert A(G)-A(F)\rVert_2
\ \geqslant\ \lambda-\sqrt{2q}
\ >\ \bigl(1-\delta-\sqrt{2\eta}\bigr)\sqrt m ,
\]
the last step by $q<\eta m$. Moreover
$\gamma\leqslant\frac16$ gives
$\delta+\sqrt{2\eta}
 < 1 $, 
so $\lambda-\sqrt{2q}>0$ and the bound
$\lambda_F\geqslant\lambda-\sqrt{2q}$ may be squared. Using
$\lambda\leqslant\sqrt m$, then $m-\lambda^{2}<2\delta m$ and
$q<\eta m$,
\[
\ell-\lambda_F^{2}
\ \leqslant\ (m-q)-\bigl(\lambda-\sqrt{2q}\bigr)^{2}
\ \leqslant\ (m-\lambda^{2})+2\sqrt{2mq}
\ <\ \bigl(2\delta+2\sqrt{2\eta}\bigr)m .
\]

Applying Lemma \ref{lem:bipartite-completion} to the
bipartite graph obtained from $F$ by deleting its isolated
vertices, which has $\ell$ edges and spectral radius
$\lambda_F$, we obtain disjoint sets $A,B\subseteq V(G)$ with
\[
\bigl|E(F)\,\triangle\,E(K_{A,B})\bigr|
\ \leqslant\ 3\bigl(\ell-\lambda_F^{2}\bigr)
\ <\ \bigl(6\delta+6\sqrt{2\eta}\bigr)m .
\]
Since $E(G)\,\triangle\,E(K_{A,B})$ is contained in
$\bigl(E(G)\,\triangle\,E(F)\bigr)\cup
\bigl(E(F)\,\triangle\,E(K_{A,B})\bigr)$ and
$|E(G)\,\triangle\,E(F)|=q<\eta m$, we get
\[
\bigl|E(G)\,\triangle\,E(K_{A,B})\bigr|
<\bigl(\eta+6\delta+6\sqrt{2\eta}\bigr)m .
\]
Finally, $\gamma\leqslant\frac16$ gives
$\eta+6\delta=\frac{\gamma^{2}}{1000}+\frac{6\gamma^{3}}{1000}
<\frac{\gamma}{1000}$ and
$6\sqrt{2\eta}=6\gamma\sqrt{\tfrac{2}{1000}}<\frac{3\gamma}{10}$,
so the right-hand side is less than
$\gamma m\leqslant\varepsilon m$. This is the edit-distance
alternative, and the proof is complete.

<!-- theorem-10: body_md -->
\label{thm:main-3}
Let $r\geqslant 2$ and $G$ be a graph of order $n$. If
$n > \frac{r^2}{2\cdot \gcd(2,r)}$ and $\lambda(G)\geqslant\lambda(T_{n,r})$, then
$$\operatorname{bk}(G)\geqslant\frac{r-2}{r}\,n.$$
<!-- end theorem-10 -->
<!-- proof-24: body_md -->
Suppose on the contrary that $\operatorname{bk}(G) < \frac{r-2}{r}\,n$.
Put $g:=\gcd(2,r)$ and $x:=\frac{r-2}{r}\,n$.
The rational number $x$ has denominator dividing
\[
\frac{r}{\gcd(r-2,r)} = \frac{r}{\gcd(2,r)} = \frac{r}{g}.
\]
Therefore, if $x$ is not an integer, its fractional part is at
least $g/r$. If $x$ is an integer, every integer strictly smaller
than $x$ is at most $x-1$, and $1\geqslant g/r$. Since $\operatorname{bk}(G)$ is an
integer, the assumption $\operatorname{bk}(G)<\frac{r-2}{r}\,n$ therefore gives
\[
  \operatorname{bk}(G)\leqslant\frac{r-2}{r}\,n - \frac{g}{r}.
\]
Since $\lambda(G)\geqslant\lambda(T_{n,r})>0$, we have $m\geqslant1$,
and $\lambda(G)\geqslant\frac{2m}{n}$. Theorem \ref{thm:two-lambda},
applied to $G$ with its isolated vertices deleted, gives
$\operatorname{bk}(G)\geqslant2\lambda(G)-\frac{2m}{\lambda(G)}\geqslant2\lambda(G)-n$.
It follows that 
\begin{equation}\label{eq:temp}
\lambda(G)\leqslant\frac{n+\operatorname{bk}(G)}2 \leqslant
\frac{1}{2} \left(n+\frac{r-2}{r}n-\frac{g}{r}\right)
=\frac{r-1}{r}n-\frac{g}{2r}.
\end{equation}
On the other hand, \eqref{eq:lambda-Turan} and $\frac{r}{4n}<\frac{g}{2r}$ yield
\begin{equation}\label{eq:lambda-Turan-lower}
\lambda(T_{n,r})\geqslant\frac{r-1}{r}n - \frac{r}{4n}
> \frac{r-1}{r}n - \frac{g}{2r}.
\end{equation}
Combining \eqref{eq:temp} and \eqref{eq:lambda-Turan-lower} yields
$\lambda(G)<\lambda(T_{n,r})$, a contradiction.

<!-- lemma-10: body_md -->
\label{lem:edw-pigeonhole}
Let $T$ be a triangle of $G$ and let $w$ be any
vertex of $G$. Then $w\in S_e$ for at least one of
the three edges $e$ of $T$. Consequently
\begin{equation}\label{eq:edw-local}
  \sum_{e\in E(T)}|S_e|\ \geqslant\ n .
\end{equation}
<!-- end lemma-10 -->
<!-- proof-25: body_md -->
Write $T=xyz$. The vertex $w$ splits $V(G)$ into the
two sets $N(w)$ and $\overline N(w)$. The three
vertices $x$, $y$, $z$ lie in these two sets, so by
the pigeonhole principle two of them lie in the same
set; say they are $x$ and $y$. If both lie in
$N(w)$, then $w$ is adjacent to both $x$ and $y$, so
$w\in N(x)\cap N(y)\subseteq S_{xy}$. If both lie in
$\overline N(w)$, then $w$ is adjacent to neither,
so $w\in\overline N(x)\cap\overline N(y)
\subseteq S_{xy}$. In both cases $w\in S_{xy}$, and
$xy$ is an edge of $T$. 
For \eqref{eq:edw-local}, sum over all $w\in V(G)$. Each
of the $n$ vertices is counted at least once on the
left, so $\sum_{e\in E(T)}|S_e|\geqslant n$.

<!-- theorem-11: body_md -->
\label{thm:edw-master-ineq}
For every graph $G$ with $n$ vertices and $m$ edges,
\begin{equation*} 
  \operatorname{bk} (G) \cdot (Q - mn) \leqslant  \big(6\operatorname{bk} (G) - n\big) \cdot t(G). 
\end{equation*}
<!-- end theorem-11 -->
<!-- proof-26: body_md -->
Write $\beta=\operatorname{bk}(G)$. 
Sum \eqref{eq:edw-local} over all triangles $T$ of
$G$:
\begin{equation}\label{eq:edw-sum1}
  n\,k_3\ \leqslant\ \sum_{T}\sum_{e\in E(T)}|S_e| .
\end{equation}
On the right side, group the terms by the edge $e$.
An edge $e$ lies in exactly $c_e$ triangles, so it
contributes the term $|S_e|$ exactly $c_e$ times.
Hence
\begin{equation}\label{eq:edw-sum2}
  \sum_{T}\sum_{e\in E(T)}|S_e|
  =\sum_{e\in E(G)}c_e\,|S_e|
  \ \leqslant\ \beta\sum_{e\in E(G)}|S_e| ,
\end{equation}
where the inequality uses $c_e\leqslant\beta$ for every
edge and $|S_e|\geqslant0$.

It remains to compute the last sum. By
\eqref{eq:edw-Suv},
\[
  \sum_{uv\in E(G)}|S_{uv}|
  =\sum_{uv\in E(G)}
     \bigl(n-d(u)-d(v)+2c_{uv}\bigr).
\]
There are $m$ edges, so the first term gives $mn$.
Every vertex $v$ lies in exactly $d(v)$ edges, so
\[
  \sum_{uv\in E(G)}\bigl(d(u)+d(v)\bigr)
  =\sum_{v\in V(G)}d(v)^2=Q .
\]
Finally $\sum_{uv\in E(G)}c_{uv}=3k_3$, because the
left side counts the pairs $(uv,x)$ with
$x\in N(u)\cap N(v)$, and every triangle gives three
such pairs, one for each of its edges. Therefore
\begin{equation}\label{eq:edw-sum3}
  \sum_{e\in E(G)}|S_e|=mn-Q+6k_3 .
\end{equation}
Combining \eqref{eq:edw-sum1}, \eqref{eq:edw-sum2}
and \eqref{eq:edw-sum3} gives
$ n\,t(G) \leqslant \operatorname{bk}(G)\,\bigl(6t(G)-Q+mn\bigr)$, as needed.
