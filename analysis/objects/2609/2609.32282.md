\label{th:Bound_EIG}
 Let $(\theta,Y)$ be a random vector on $\mathbb{R}^d\times \mathbb{R}^p$ with smooth joint probability density $\pi_{\theta,Y}$ such that $\mathbb{E}[\|Y\|^2]<\infty$.
 
 Let $\Sigma_Y=\operatorname{Cov}(Y)$ and $\Sigma_{Y|\theta}=\mathbb{E}[\operatorname{Cov}(Y|\theta)]$ and let $\Sigma_\mathrm{noise}\in\mathbb{R}^{p\times p}$ and $\Sigma_\mathrm{signal}\in\mathbb{R}^{p\times p}$ be any symmetric definite positive matrices that bound the (expected) Fisher information matrices as
 \begin{align}
   \Sigma_\mathrm{noise}^{-1} &\succeq \mathbb{E}[\mathcal{I}_{Y|\theta}] := \mathbb{E}_{(\theta,Y)}\left[ \left(\nabla_{Y}\ln \pi_{Y|\theta}(Y|\theta)   \right)\left(\nabla_{Y}\ln \pi_{Y|\theta}(Y|\theta)   \right)^\top  \right]  \label{eq:SigmaNoise} \\
   \Sigma_\mathrm{signal}^{-1}  &\succeq \mathcal{I}_{Y} := \mathbb{E}_Y\left[ \left(\nabla_{Y}\ln \pi_{Y}(Y)   \right)\left(\nabla_{Y}\ln \pi_{Y}(Y)   \right)^\top \right] . \label{eq:SigmaSignal}
 \end{align}
 
 Then, for any matrices $W_m\in\mathbb{R}^{p\times m}$ and $W_\mathrm{new}\in\mathbb{R}^{p\times m'}$ such that $[W_m,W_\mathrm{new}]\in\mathbb{R}^{p\times (m+m')}$ has rank $m+m'\leq p$, we have
  \begin{align}
  \mathrm{EIG}([W_m,W_\mathrm{new}])
 &\geq \mathrm{EIG}[W_m] + \frac{1}{2}    \ln\left( \frac{|W_\mathrm{new}^\top \Sigma_\mathrm{signal}[W_m]  W_\mathrm{new}|}{| W_\mathrm{new}^\top \Sigma_{Y|\theta}[W_m]  W_\mathrm{new} |}  \right)  \label{eq:Bound_EIG} \\
 \mathrm{EIG}([W_m,W_\mathrm{new}]) &\leq \mathrm{EIG}[W_m] + \frac{1}{2}   \ln\left( \frac{ | W_\mathrm{new}^\top \Sigma_{Y}[W_m] W_\mathrm{new} | }{ |W_\mathrm{new}^\top \Sigma_\mathrm{noise}[W_m] W_\mathrm{new}| }\right) , \label{eq:Bound_EIG_up}
 \end{align}
 where $[W_m,W_\mathrm{new}]$ denotes the concatenation of the columns of the matrices $W_m$ and $W_\mathrm{new}$ and
 where $\Sigma[W] = \Sigma - \Sigma W (W^\top \Sigma W)^{-1}W^\top \Sigma$.

<!-- remark-1: body_md -->
\label{rmk:GaussianY}
 It is worth noting that both the incremental bounds \eqref{eq:Bound_EIG_incremental}--\eqref{eq:Bound_EIG_incremental_up} and the conservative bounds \eqref{eq:Bound_EIG_conservative}--\eqref{eq:Bound_EIG_conservative_up} become equalities as soon as
 \begin{equation}\label{eq:tightness}
   \Sigma_\mathrm{signal}=\Sigma_Y
   \quad\mathrm{and}\quad
   \Sigma_\mathrm{noise} = \Sigma_{Y|\theta} ,
 \end{equation}
 where we recall that $\Sigma_Y=\operatorname{Cov}(Y) $, $\Sigma_{Y|\theta} = \mathbb{E}[\operatorname{Cov}(Y|\theta)]$ and
 $\Sigma_\mathrm{signal}\preceq\mathcal{I}_Y^{-1}$,  $\Sigma_\mathrm{noise}\preceq \mathbb{E}[\mathcal{I}_{Y|\theta}]^{-1}$.
 As explained in Section \ref{sec:Fisher}, any random variable $X$ with smooth enough density satisfies the Cramér-Rao bound $\mathcal{I}_X^{-1} \preceq \operatorname{Cov}(X)$, with equality if and only if $X$ is Gaussian (\emph{Stein's characterization} of Gaussian distributions).
 We deduce that if \eqref{eq:tightness} holds true, then we have $\mathcal{I}_Y^{-1} = \operatorname{Cov}(Y)$ so that $Y$ is necessarily Gaussian. Similarly we have that $Y|\theta$ is almost surely Gaussian\footnote{By Jensen and Cramér-Rao inequalities, we have $\mathbb{E}[\mathcal{I}_{Y|\theta}]^{-1}\overset{\mathcal{J}}{\preceq} \mathbb{E}[\mathcal{I}_{Y|\theta}^{-1}] \overset{\mathcal{CR}}{\preceq} \mathbb{E}[\operatorname{Cov}(Y|\theta)]$, therefore \eqref{eq:tightness} implies $0=\mathbb{E}[\underbrace{\operatorname{Cov}(Y|\theta)-\mathcal{I}_{Y|\theta}^{-1}}_{\succeq0}]$, therefore $\mathcal{I}_{Y|\theta}^{-1}=\operatorname{Cov}(Y|\theta)$ a.s. and then $Y|\theta$ is a.s. Gaussian.}.
 The sharpness of the bounds therefore directly reflects the degree of non-Gaussianity of both $Y$ and $Y|\theta$.

Consider the problem of finding the matrix $A\in\mathbb{R}^{q\times p}$ and the vector $b\in\mathbb{R}^p$ which minimize the mean square error $\mathbb{E}\left[\| G(\eta)- (A\eta+b) \|^2\right]$. If we further assume that $\eta=\mathcal{N}(m,\Sigma)$ is Gaussian, one integration by part yields
 $$

 This means that $\mathcal{L}(G)$ is the slope of the best affine approximation of $G$ in $L^2_{\pi_\eta}$.

<!-- proposition-5: body_md -->
\label{prop:Fisher}
 Let $(X,Y,Z)$ a random vector in $\mathbb{R}^{d}\times \mathbb{R}^{p} \times \mathbb{R}^{q}$.
 Then
 
 \begin{align}
  \mathcal{I}_{Y} &=  \mathbb{E}[\mathcal{I}_{Y|X}]  - \mathbb{E}[\operatorname{Cov}(\nabla_{Y} \ln\pi_{X,Y}(X,Y) | Y)] \label{eq:FisherMarginal} \\
 
  \mathbb{E}[\mathcal{I}_{Y|X}] &= \mathbb{E}[\mathcal{I}_{Y|X,Z}]  -  \mathbb{E}[\operatorname{Cov}(\nabla_{Y} \ln\pi_{X,Y,Z}(X,Y,Z) | X,Y)] , \label{eq:FisherMarginal_bis}
 \end{align}
 and
 
\begin{align}
  \mathcal{I}_{Y}&\preceq  \mathbb{E}[\mathcal{I}_{Y|X}]  - \mathcal{I}_{Y,X}   \mathbb{E}[\mathcal{I}_{X|Y}] ^{-1} \mathcal{I}_{X,Y}  \label{eq:FisherMarginal_bound}\\
 
 \mathbb{E}[\mathcal{I}_{Y|X}]  &\preceq \mathbb{E}[\mathcal{I}_{Y|X,Z}] -  \mathbb{E}[\mathcal{I}_{Y,Z|X}] \mathbb{E}[\mathcal{I}_{Z|X,Y}]^{-1} \mathbb{E}[\mathcal{I}_{Z,Y|X}] .
 \label{eq:FisherMarginal_bound_bis}
\end{align}

<!-- proof-9: body_md -->
Because the marginal density writes $\pi_Y(y)=\int \pi_{X,Y}(x,y)\mathrm{d} x$, we have
 \begin{align*}
  \nabla_Y\ln\pi_Y(y)
  &= \frac{\int \nabla_Y\pi_{X,Y}(x,y)\mathrm{d} x }{\pi_Y(y)}
  = \mathbb{E}[\nabla_Y\ln\pi_{X,Y}(X,Y)|Y=y] ,
 \end{align*}
 and then $\mathcal{I}_Y=\operatorname{Cov}(\nabla_Y\ln\pi_Y(Y)) = \operatorname{Cov}( \mathbb{E}[\nabla_Y\ln\pi_{X,Y}(X,Y)|Y] )$.
 Then, using the law of total variance, we have
 \begin{align*}
  \mathbb{E}[\mathcal{I}_{Y|X}]
  &=  \operatorname{Cov}( \nabla_Y\ln\pi_{X,Y}(X,Y) )\\
  &= \operatorname{Cov}( \mathbb{E}[ \nabla_Y\ln\pi_{X,Y}(X,Y) |Y] )  + \mathbb{E}[\operatorname{Cov}( \nabla_Y\ln\pi_{X,Y}(X,Y) |Y )] \\
  &=  \mathcal{I}_Y + \mathbb{E}[\operatorname{Cov}( \nabla_Y\ln\pi_{X,Y}(X,Y) |Y )] ,
 \end{align*}
 which is \eqref{eq:FisherMarginal}. Relation \eqref{eq:FisherMarginal_bis} is derived using the same calculation.
 To show \eqref{eq:FisherMarginal_bound}, let us note that for any matrix $A\in\mathbb{R}^{d\times p}$ we have
 \begin{align*}
  \mathcal{I}_Y
  &= \operatorname{Cov}\Big( \mathbb{E}[\nabla_Y\ln\pi_{X,Y}(X,Y)|Y] \Big) \\
  &= \operatorname{Cov}\Big( \mathbb{E}[\nabla_Y\ln\pi_{X,Y}(X,Y)|Y] - A \underbrace{\mathbb{E}[\nabla_X\ln\pi_{X,Y}(X,Y)|Y]}_{=0} \Big) \\
  &= \operatorname{Cov}\Big( \mathbb{E}\Big[\nabla_Y\ln\pi_{X,Y}(X,Y) - A  \nabla_X\ln\pi_{X,Y}(X,Y) \Big|Y\Big] \Big) \\
  &\preceq \operatorname{Cov}\Big(  \nabla_Y\ln\pi_{X,Y}(X,Y) - A  \nabla_X\ln\pi_{X,Y}(X,Y)  \Big) \\
  &= \mathbb{E}[\mathcal{I}_{Y|X}] - \mathcal{I}_{Y , X}A^\top - A \mathcal{I}_{X , Y} +  A \mathbb{E}[\mathcal{I}_{X|Y}] A^\top  .
 \end{align*}
 Taking $A = \mathcal{I}_{Y,X} \mathbb{E}[\mathcal{I}_{X|Y}]^{-1}  $ gives \eqref{eq:FisherMarginal_bound}.
 Relation \eqref{eq:FisherMarginal_bound_bis} is derived using the same calculation.
 This concludes the proof.

<!-- lemma-1: body_md -->
\label{lem:dLSI_Fisher}
 Let $(X,Z)$ be a random vector in $\mathbb{R}^{d} \times \mathbb{R}^{p}$ with smooth joint density $\pi_{X,Z}$. Assume $\mathbb{E}[\|X\|^2]<\infty$ and let  $ \widetilde\pi_{X} = \mathcal{N}( \mathbb{E}[X] , \operatorname{Cov}(X) ) $. Then
\begin{align}
 \frac{1}{2}   \ln\left(\frac{|\operatorname{Cov}(X)|}{|\mathbb{E}[\operatorname{Cov}(X| Z)]|}\right)
 \leq \mathbb{E}_{Z}[\operatorname{D_\mathrm{KL}}( \pi_{X|Z} || \widetilde\pi_{X})]
 &\leq \frac{1}{2}   \ln\left( \left| \operatorname{Cov}(X) \mathbb{E}[\mathcal{I}_{X|Z}] \right| \right) , \label{eq:dLSI_Fisher_Ytheta}
\end{align}
where $\mathbb{E}[\operatorname{Cov}(X|Z)] = \mathbb{E}[( X - \mathbb{E}[X|Z] )^{\otimes 2}] $ is the expected conditional covariance of $X | Z$ and $\mathbb{E}[\mathcal{I}_{X|Z}] = \mathbb{E}[ (\nabla_{X} \ln\pi_{X,Z}(X,Z) )^{\otimes 2} ]$ is the expected Fisher information matrix of $X|Z$.

First we note that the dimensional logarithmic Sobolev inequalities (dLSI) can be naturally extended to any Gaussian measure $\mu=\mathcal{N}(m,\Sigma)$ with full rank covariance matrix $\Sigma\in\mathbb{R}^{d\times d}$ as follows. Let $T(x) =  \Sigma^{-1/2}(x-m)$ be the affine transformation such that $T_\sharp\mu = \mu_0$. We have $\operatorname{D_\mathrm{KL}}(\nu||\mu) = \operatorname{D_\mathrm{KL}}( T_\sharp\nu ||\mu_0)$ so that the dLSI yields
 \begin{equation}
  \operatorname{D_\mathrm{KL}}(\nu||\mu) \leq \frac{1}{2} \int \| x-m \|_{\Sigma^{-1}}^2 \mathrm{d}\nu - \frac{d}{2} + \frac{1}{2}\ln\left| \Sigma \int  (\nabla\ln\nu)^{\otimes 2} \mathrm{d}\nu \right| \label{eq:dLSI_up}
 \end{equation}
and
\begin{align}
 
 \operatorname{D_\mathrm{KL}}(\nu||\mu) &\geq \frac{1}{2} \int \| x-m \|_{\Sigma^{-1}}^2 \mathrm{d}\nu - \frac{d}{2} - \frac{1}{2}\ln\frac{\left| \int ( x-m)^{\otimes 2} \mathrm{d}\nu - \left( \int (x-m) \mathrm{d}\nu \right)^{\otimes 2} \right| }{|\Sigma|}  \nonumber\\
 &= \frac{1}{2} \int \| x-m \|_{\Sigma^{-1}}^2 \mathrm{d}\nu - \frac{d}{2} - \frac{1}{2}\ln\frac{\left| \int x^{\otimes 2} \mathrm{d}\nu - \left( \int x \mathrm{d}\nu \right)^{\otimes 2} \right| }{|\Sigma|}  \label{eq:dLSI_down}.
\end{align}
 Thus we can write
 \begin{align*}
 \mathbb{E}_{Z}[\operatorname{D_\mathrm{KL}}( \pi_{X|Z} || \widetilde\pi_{X})]
 &\overset{\eqref{eq:dLSI_up}}{\leq} \mathbb{E}_{Z}\Bigg[  \frac{1}{2} \int \| y_\perp-\mathbb{E}[X] \|_{\operatorname{Cov}(X)^{-1}}^2 \mathrm{d}\pi_{X|Z} - \frac{d}{2} \\
 &\qquad\qquad + \frac{1}{2}\ln\left(\left| \operatorname{Cov}(X) \int  (\nabla_{X}\ln\pi_{X|Z})^{\otimes 2} \mathrm{d}\pi_{X|Z} \right|\right)  \Bigg] \\
 &= \frac{1}{2}  \mathbb{E}_{Z}\left[ \ln\left(\left| \operatorname{Cov}(X) \int  (\nabla_{X}\ln\pi_{X|Z})^{\otimes 2} \mathrm{d}\pi_{X|Z} \right|\right)  \right] \\
 &\leq \frac{1}{2}   \ln\left(\left| \operatorname{Cov}(X) \int  (\nabla_{X}\ln\pi_{X|Z})^{\otimes 2} \mathrm{d}\pi_{X,Z} \right|\right) ,
\end{align*}
where in the last step we used the Jensen inequality $\mathbb{E}[\varphi(X)]\leq \varphi(\mathbb{E}[X])$ with $\varphi(X)=\ln|X|$ a concave function on the space of SPD matrices. This gives the right-hand side of \eqref{eq:dLSI_Fisher_Ytheta}. To prove the left-hand side, we write
 \begin{align*}
 \mathbb{E}_{Z}\left[ \operatorname{D_\mathrm{KL}}( \pi_{X|Z} || \widetilde\pi_{X} )\right]
 &\overset{\eqref{eq:dLSI_down}}{\geq} \mathbb{E}_{Z}\Bigg[  \frac{1}{2} \int \| y_\perp-\mathbb{E}[X] \|_{\operatorname{Cov}(X)^{-1}}^2 \mathrm{d}\pi_{X|Z} - \frac{d}{2} \\
 &\qquad\qquad - \frac{1}{2}\ln\left(\frac{\left| \int  y_\perp^{\otimes 2} \mathrm{d}\pi_{X|Z} - \left( \int y_\perp \mathrm{d}\pi_{X|Z} \right)^{\otimes 2} \right| }{|\operatorname{Cov}(X)|} \right)  \Bigg] \\
 
 &= - \frac{1}{2} \mathbb{E}_{Z}\Bigg[\ln\left(\frac{\left| \int  y_\perp^{\otimes 2} \mathrm{d}\pi_{X|Z} - \left( \int y_\perp \mathrm{d}\pi_{X|Z} \right)^{\otimes 2} \right| }{|\operatorname{Cov}(X)|} \right)  \Bigg] \\
 &\geq - \frac{1}{2} \ln\left(\frac{\left| \int  y_\perp^{\otimes 2} \mathrm{d}\pi_{X} - \mathbb{E}_{Z}[( \int y_\perp \mathrm{d}\pi_{X|Z} )^{\otimes 2} ]\right| }{|\operatorname{Cov}(X)|} \right) ,
\end{align*}
where we again used the Jensen inequality in the last step. Because $\mathbb{E}[\operatorname{Cov}(X|Z)]=\mathbb{E}[X^{\otimes2}]-\mathbb{E}[\mathbb{E}[X|Z]^{\otimes2}]$, this yields the left-hand side of \eqref{eq:dLSI_Fisher_Ytheta} and concludes the proof.

Consider the Fisher information matrix
$$
 \mathcal{I}_{(Y_\mathrm{new},Y_m)} =
 \begin{pmatrix}
   \mathbb{E}[\mathcal{I}_{Y_\mathrm{new}|Y_m}] & \mathcal{I}_{ Y_\mathrm{new} ,Y_m} \\
  \mathcal{I}_{ Y_m, Y_\mathrm{new}} & \mathbb{E}[\mathcal{I}_{Y_m|Y_\mathrm{new}}] \\
 \end{pmatrix}.
$$
Proposition \ref{prop:Fisher} permits to write
$$
 \mathcal{I}_{(Y_\mathrm{new},Y_m)}
 \overset{\eqref{eq:FisherMarginal_bound}}{\preceq}
 S:=
 \mathbb{E}[\mathcal{I}_{(Y_\mathrm{new},Y_m)|Y_\perp}] -
 \mathcal{I}_{(Y_\mathrm{new},Y_m),Y_\perp}
 \mathbb{E}[\mathcal{I}_{Y_\perp|(Y_m,Y_\perp)}]^{-1}
 \mathcal{I}_{Y_\perp, (Y_\mathrm{new},Y_m)} .
$$
Next, we show that $S$ is the Shur complement of a block decomposition of the Fisher information matrix $\mathcal{I}_{Y}$.
By \eqref{tmp:20756187} we have
\begin{align*}
 \nabla_{Y_\mathrm{new}}\ln\pi_{Y_\mathrm{new} , Y_m,Y_\perp}(Y_\mathrm{new} , Y_m,Y_\perp)
 &= V_\mathrm{new}^\top \nabla_Y\ln\pi( Y  ) \\
 \nabla_{Y_m}\ln\pi_{Y_\mathrm{new} , Y_m,Y_\perp}(Y_\mathrm{new} , Y_m,Y_\perp)
 &= V_m^\top \nabla_Y\ln\pi( Y  ) \\
 \nabla_{Y_\perp}\ln\pi_{Y_\mathrm{new} , Y_m,Y_\perp}(Y_\mathrm{new} , Y_m,Y_\perp)
 &= V_\perp^\top \nabla_Y\ln\pi( Y  ) ,
\end{align*}
so that
\begin{align*}
  \mathbb{E}[\mathcal{I}_{Y_\mathrm{new},Y_m|Y_\perp}] &= (V_\mathrm{new},V_m)^\top \mathcal{I}_{Y} (V_\mathrm{new},V_m) \\
  \mathcal{I}_{(Y_\mathrm{new},Y_m),Y_\perp} &=  (V_\mathrm{new},V_m)^\top \mathcal{I}_{Y} V_\perp \\
  \mathbb{E}[\mathcal{I}_{Y_\perp|(Y_m,Y_\perp)}] &=  V_\perp^\top \mathcal{I}_{Y} V_\perp .
\end{align*}
Thus, $S$ is the Shur complement of the block matrix
$$
     V^\top \mathcal{I}_Y V =\begin{pmatrix}
      V_\mathrm{new}^\top \mathcal{I}_Y V_\mathrm{new} & V_\mathrm{new}^\top \mathcal{I}_Y V_m &  V_\mathrm{new}^\top \mathcal{I}_Y V_\perp \\
      V_m^\top \mathcal{I}_Y V_\mathrm{new} & V_m^\top \mathcal{I}_Y V_m & V_m^\top \mathcal{I}_Y V_\perp \\
      V_\perp^\top \mathcal{I}_Y V_\mathrm{new} & V_\perp^\top \mathcal{I}_Y V_m & V_\perp^\top \mathcal{I}_Y V_\perp
     \end{pmatrix} ,
$$
so that
\begin{align*}
 \begin{pmatrix}
\Big(~~S^{-1}~\Big) & \begin{matrix}\times \\ \times \end{matrix} \\
\begin{matrix}\times & \times \end{matrix}  & \begin{matrix}\times  \end{matrix}
\end{pmatrix}
   &=\Big( V^\top \mathcal{I}_Y V \Big)^{-1}\\
   &= W^\top \mathcal{I}_Y^{-1} W \\
   &\preceq W^\top \Sigma_\mathrm{signal} W
   = \begin{pmatrix}
    W_\mathrm{new}\Sigma_\mathrm{signal} W_\mathrm{new} & W_\mathrm{new}\Sigma_\mathrm{signal} W_m & \times \\
     W_m\Sigma_\mathrm{signal} W_\mathrm{new} & W_m\Sigma_\mathrm{signal} W_m & \times\\
     \times& \times& \times
   \end{pmatrix}
\end{align*}
Finally we obtain
\begin{align*}
 \begin{pmatrix}
   \mathbb{E}[\mathcal{I}_{Y_\mathrm{new}|Y_m}] & \mathcal{I}_{ Y_\mathrm{new} ,Y_m} \\
  \mathcal{I}_{ Y_m, Y_\mathrm{new}} & \mathbb{E}[\mathcal{I}_{Y_m|Y_\mathrm{new}}] \\
 \end{pmatrix}
 &= \mathcal{I}_{(Y_\mathrm{new},Y_m)} \\
 &\preceq S \\
 &=\begin{pmatrix}
    W_\mathrm{new}\Sigma_\mathrm{signal} W_\mathrm{new} & W_\mathrm{new}\Sigma_\mathrm{signal} W_m  \\
     W_m\Sigma_\mathrm{signal} W_\mathrm{new} & W_m\Sigma_\mathrm{signal} W_m \\
   \end{pmatrix}^{-1}  \\
 &= \begin{pmatrix}
    \Big( W_\mathrm{new} \Sigma_\mathrm{signal}[W_m] W_\mathrm{new} \Big)^{-1} & \times   \\
     \times & \times \\
   \end{pmatrix}.
\end{align*}
where $\Sigma_\mathrm{signal}[W_m] =
\Sigma_\mathrm{signal}
- \Sigma_\mathrm{signal} W_m ( W_m^\top \Sigma_\mathrm{signal} W_m )^{-1}
W_m^\top \Sigma_\mathrm{signal}$. By considering the first block of the above decomposition, we obtain \eqref{tmp:104763}.
Relation \eqref{tmp:103571} is derived analogously, which completes the proof.

We start by decomposing the EIG as follow
\begin{align}
 \mathrm{EIG}(W_m)
 &= \mathbb{E}_{Y} \left[\operatorname{D_\mathrm{KL}}( \pi_{\theta|Y_m} || \pi_\theta ) \right] \nonumber\\
 &= \int  \log\frac{\pi_{\theta|Y_m}}{\pi_{\theta}} \mathrm{d}\pi_{\theta Y} \nonumber\\
 &= \int  \log\frac{\pi_{\theta|Y_m,Y_\mathrm{new}}}{\pi_{\theta}} \mathrm{d}\pi_{\theta Y} - \int  \log\frac{\pi_{\theta|Y_m,Y_\mathrm{new}}}{\pi_{\theta|Y_m}} \mathrm{d}\pi_{\theta Y} \nonumber\\
 &= \mathrm{EIG}([W_m,W_\mathrm{new}]) - \mathbb{E}_Y[\operatorname{D_\mathrm{KL}}( \pi_{\theta|Y_m,Y_\mathrm{new}} || \pi_{\theta|Y_m} )] .
 \label{tmp:078}
\end{align}
Applying Proposition \ref{prop:TheResult} and Lemma \ref{lem:FisherMarginal} yields
\begin{align*}
 &\mathrm{EIG}(W_m) \\
 &\overset{\eqref{tmp:078}}{=}\mathrm{EIG}([W_m,W_\mathrm{new}]) - \mathbb{E}_Y[\operatorname{D_\mathrm{KL}}( \pi_{\theta|Y_m,Y_\mathrm{new}} || \pi_{\theta|Y_m} )] \\
 &\overset{\eqref{eq:TheResult}}{\leq}  \mathrm{EIG}([W_m,W_\mathrm{new}]) - \frac{1}{2}    \ln\left(\frac{| \mathbb{E}[\mathcal{I}_{Y_\mathrm{new} | Y_m}]^{-1} |}{|\mathbb{E}[\operatorname{Cov}(Y_\mathrm{new}| \theta,Y_m)]|}\right) \\
 
 &\overset{\eqref{tmp:104763}}{\leq}
 \mathrm{EIG}([W_m,W_\mathrm{new}]) - \frac{1}{2}    \ln\left(| W_\mathrm{new}^\top \Sigma_\mathrm{signal}[W_m] W_\mathrm{new}  |  \right) + \frac{1}{2}    \ln\left( |\mathbb{E}[\operatorname{Cov}(Y_\mathrm{new}| \theta,Y_m)]|\right).
\end{align*}
To show \eqref{eq:Bound_EIG}, it remains to bound $\mathbb{E}[\operatorname{Cov}(Y_\mathrm{new}| \theta,Y_m)]$ by $W_\mathrm{new}^\top  \Sigma_{Y|\theta}[W_m]  W_\mathrm{new}$, where $\Sigma_{Y|\theta}[W_m] = \Sigma_{Y|\theta}  - \Sigma_{Y|\theta} W_m (W_m^\top \Sigma_{Y|\theta} W_m)^{-1}  W_m^\top \Sigma_{Y|\theta}$. For any $A\in\mathbb{R}^{m'\times m}$ we can write
\begin{align*}
 \mathbb{E}[\operatorname{Cov}(Y_\mathrm{new}| \theta,Y_m)]
 &=\mathbb{E}[\operatorname{Cov}(Y_\mathrm{new} - A Y_m| \theta,Y_m)] \\
 &\preceq \mathbb{E}[\operatorname{Cov}(Y_\mathrm{new} - A Y_m| \theta)] \\
 &= \mathbb{E}[\operatorname{Cov}( (W_\mathrm{new}^\top - A  W_m^\top) Y| \theta)] \\
 &= (W_\mathrm{new}^\top - A  W_m^\top)\mathbb{E}[\operatorname{Cov}(Y| \theta)] (W_\mathrm{new} - W_m A^\top) .
\end{align*}
By taking $A= W_\mathrm{new}^\top\mathbb{E}[\operatorname{Cov}(Y|\theta)] W_m ( W_m^\top \mathbb{E}[\operatorname{Cov}(Y| \theta)] W_m )^{-1}$ we obtain
$
\mathbb{E}[\operatorname{Cov}(Y_\mathrm{new}| \theta,Y_m)] \preceq
W_\mathrm{new}^\top  \Sigma_{Y|\theta}[W_m]  W_\mathrm{new} .
$
This completes the proof of \eqref{eq:Bound_EIG}.

We bound $\mathbb{E}[\mathcal{I}_{Y|\theta}]$ using
\begin{align*}
 \mathbb{E}[\mathcal{I}_{Y|\theta}]
 &\overset{\eqref{eq:FisherMarginal_bound_bis}}{\preceq}
  \mathbb{E}[\mathcal{I}_{Y|\theta,\eta}]  - \mathbb{E}[\mathcal{I}_{Y,\eta|\theta}] \mathbb{E}[\mathcal{I}_{\eta|(Y,\theta)}]^{-1} \mathbb{E}[\mathcal{I}_{\eta,Y|\theta}] .
\end{align*}
In order to compute the terms above, recall that the joint density of $(Y,\eta,\theta)$ is
\begin{equation*}
   \pi_{Y,\eta,\theta}(Y,\eta,\theta) \propto   \exp\left(-\frac{1}{2}\| G(\eta) - Y\|_{\Sigma_\text{obs}^{-1}}^2\right) \exp\left(-\frac{1}{2}\| H(\eta) - \theta\|_{\Sigma_\xi^{-1}}^2 \right) \pi_{\eta}(\eta) ,
\end{equation*}
so that
\begin{align*}
 \nabla_Y \ln\pi_{Y,\eta,\theta}(Y,\eta,\theta) &= \Sigma_\text{obs}^{-1}(G(\eta)-Y) \\
 \nabla_\theta \ln\pi_{Y,\eta,\theta}(Y,\eta,\theta) &=  \Sigma_\xi^{-1}( H(\eta)-\theta) \\
 \nabla_\eta \ln\pi_{Y,\eta,\theta}(Y,\eta,\theta) &=  -  \operatorname{Jac}  G(\eta)^\top \Sigma_\text{obs}^{-1} ( G(\eta)-Y ) \\
 &~~~\, -  \operatorname{Jac}  H(\eta)^\top \Sigma_\xi^{-1} ( H(\eta)-\theta ) + \nabla_\eta\ln\pi_{\eta}(\eta) .
\end{align*}
Because $G(\eta)-Y = -\varepsilon$ and $H(\eta)-\theta = -\xi$ we have
\begin{align*}
 \mathbb{E}[\mathcal{I}_{Y|\theta,\eta}] = \operatorname{Cov}(\Sigma_\text{obs}^{-1}\varepsilon) = \Sigma_\text{obs}^{-1}
\end{align*}
and
\begin{align*}
 \mathbb{E}[\mathcal{I}_{Y,\eta|\theta}]
 &= \mathbb{E}[( \nabla_Y \ln\pi_{Y,\eta,\theta}(Y,\eta,\theta) )( \nabla_\eta \ln\pi_{Y,\eta,\theta}(Y,\eta,\theta) )^\top] \\
 &= \mathbb{E}[( -\Sigma_\text{obs}^{-1}\varepsilon )(  \operatorname{Jac}  G(\eta)^\top \Sigma_\text{obs}^{-1} \varepsilon  +  \operatorname{Jac}  H(\eta)^\top \Sigma_\xi^{-1} \xi + \nabla_\eta\ln\pi_{\eta}(\eta) )^\top] \\
 &= \mathbb{E}[( -\Sigma_\text{obs}^{-1}\varepsilon )(  \operatorname{Jac}  G(\eta)^\top \Sigma_\text{obs}^{-1} \varepsilon   )^\top] \\
 &= -\Sigma_\text{obs}^{-1}\mathbb{E}[ \operatorname{Jac}  G(\eta) ] ,
\end{align*}
and
\begin{align*}
 \mathbb{E}[\mathcal{I}_{\eta|(Y,\theta)}]
 &= \mathbb{E}[( \nabla_\eta \ln\pi_{Y,\eta,\theta}(Y,\eta,\theta) )^{\otimes 2} ] \\
 &= \mathbb{E}[(   \operatorname{Jac}  G(\eta)^\top \Sigma_\text{obs}^{-1} \varepsilon  +  \operatorname{Jac}  H(\eta)^\top \Sigma_\xi^{-1} \xi + \nabla_\eta\ln\pi_{\eta}(\eta) )^{\otimes 2} ] \\
 &= \mathbb{E}[(   \operatorname{Jac}  G(\eta)^\top \Sigma_\text{obs}^{-1} \varepsilon   )^{\otimes 2} ]
 + \mathbb{E}[(   \operatorname{Jac}  H(\eta)^\top \Sigma_\xi^{-1} \xi   )^{\otimes 2} ]
 + \mathbb{E}[(   \nabla_\eta\ln\pi_{\eta}(\eta) )^{\otimes 2} ]\\
 &= \mathbb{E}[ \operatorname{Jac}  G(\eta)^\top \Sigma_\text{obs}^{-1} \operatorname{Jac}  G(\eta) ]
 + \mathbb{E}[  \operatorname{Jac}  H(\eta)^\top \Sigma_\xi^{-1}\operatorname{Jac}  H(\eta) ]
 + \mathcal{I}_{\eta} \\
 &= \mathcal{H}(G) + \mathcal{L}(G)^\top\Sigma_\text{obs}^{-1}\mathcal{L}(G)  + \mathcal{J}(H)  + \mathcal{I}_{\eta}
\end{align*}
We deduce
\begin{align*}
 \mathbb{E}[\mathcal{I}_{Y|\theta}]
 &\preceq  \mathbb{E}[\mathcal{I}_{Y|\theta,\eta}]  - \mathbb{E}[\mathcal{I}_{Y,\eta|\theta}] \mathbb{E}[\mathcal{I}_{\eta|(Y,\theta)}]^{-1} \mathbb{E}[\mathcal{I}_{\eta,Y|\theta}] \\
 &= \Sigma_\text{obs}^{-1}  - \Sigma_\text{obs}^{-1}\mathcal{L}(G)^\top
 \Big(  \mathcal{H}(G) + \mathcal{L}(G)^\top\Sigma_\text{obs}^{-1}\mathcal{L}(G)  + \mathcal{J}(H)  + \mathcal{I}_{\eta} \Big)^{-1}
 \mathcal{L}(G) \Sigma_\text{obs}^{-1} \\
 &=\left( \Sigma_\text{obs}   + \mathcal{L}(G)   \Big(  \mathcal{H}(G) + \mathcal{J}(H) + \mathcal{I}_\eta  \Big)^{-1} \mathcal{L}(G)^\top  \right)^{-1}
\end{align*}
This shows that $\Sigma_\text{noise}$ defined by \eqref{eq:SigmaNoise_boundFisher} satisfies $\mathbb{E}[\mathcal{I}_{Y|\theta}] \preceq \Sigma_\text{noise}^{-1}$ and concludes the proof.
