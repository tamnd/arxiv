<!-- theorem-1: body_md -->
\label{thm:upper-bound}
The regret of Algorithm~\ref{algorithm:few-optimal-arms} with $L$ set in \eqref{eq:subsample-size} on any $K$-armed bandit instance with horizon $T \geq K$ and exactly $A \in [1,K-1]$ optimal arms admits an upper bound
\begin{align*}
    \mathrm{Regret}(T) = \widetilde{O}\bigg(\frac{K-A}{\sqrt{KA}}\sqrt{T}\bigg).
\end{align*}

<!-- remark-1: body_md -->
\label{rmk:high-prob}
Theorem~\ref{thm:upper-bound} provides an improved guarantee solely in expectation. One may naturally ask whether this improvement can also be attained with high probability. In Appendix~\ref{app:high-prob}, we show that such an extension is impossible. We prove that any algorithm satisfying the regret bound of Theorem~\ref{thm:upper-bound} must suffer $\widetilde{\Omega}(\sqrt{T})$ regret with non-vanishing probability. 
                                                 
This finding highlights an inherent separation between expected and high-probability regret in the setting of bandits with multiple optima.

<!-- proof-1: body_md -->
Let ${\mathcal{E}_{\text{miss}}} = \{\mathcal{B} \cap {\mathcal{A}^{\text{opt}}} = \varnothing \}$ be the event that the randomly sampled subset does not contain any optimal arms, and ${\mathcal{E}_{\text{sub}}} =  \{\mathcal{B} \cap {\mathcal{A}^{\text{sub}}} \neq \varnothing\} \cap \{\mathcal{B} \cap {\mathcal{A}^{\text{opt}}} \neq \varnothing\}$ be the event that the randomly sampled subset contains at least one sub-optimal arm and at least one optimal arm. 
We first bound the probability of ${\mathcal{E}_{\text{miss}}}$. First, if $L = K-A+1$ then every subset of $L$ arms must contain at least one optimal arm, since there are only $K-A$ suboptimal arms. Therefore, $\mathbb{P}[{\mathcal{E}_{\text{miss}}}]= 0$. Otherwise, if $A \leq K/2$, we directly have
\begin{align*}
L \leq \bigg\lceil \frac{\log(2T)}{\log(K/(K-A))} \bigg\rceil.
\end{align*}
In the complement case $A > K/2$, we also have
\begin{align*}
    L = \lceil \log_2 T + 1 \rceil \leq \bigg\lceil \frac{\log(2T)}{\log2} \bigg\rceil \leq   \bigg\lceil \frac{\log(2T)}{\log(K/(K-A))} \bigg\rceil.
\end{align*}
Since $\mathcal{B}$ is sampled uniformly without replacement, we have
\begin{align*}
\mathbb{P}[{\mathcal{E}_{\text{miss}}}] = \binom{K-A}{L}\binom{K}{L}^{-1} = \prod_{i=0}^{L-1} \frac{K-A-i}{K-i} \leq  \bigg(\frac{K-A}{K}\bigg)^L,
\end{align*}
where the inequality holds due to $(K-A-i)/(K-i) \leq (K-A)/K$ for every $i\in{0,\ldots,L-1}$. Plugging in the value of $L$, we see that
\begin{align*}
\mathbb{P}[{\mathcal{E}_{\text{miss}}}] \leq \bigg(\frac{K-A}{K}\bigg)^L = \exp\bigg(-L\log\bigg(\frac{K}{K-A}\bigg) \bigg) \leq \exp \big( -\log(2T) \big) = \frac{1}{2T}.
\end{align*}
Combining the two cases gives $\mathbb{P}[{\mathcal{E}_{\text{miss}}}] \leq (2T)^{-1}$.

We then bound the probability of event ${\mathcal{E}_{\text{sub}}} \cup {\mathcal{E}_{\text{miss}}}$. Trivially, we have $\mathbb{P}[ {\mathcal{E}_{\text{sub}}} \cup {\mathcal{E}_{\text{miss}}} ] \leq 1$. Moreover, by the fact that ${\mathcal{E}_{\text{sub}}} \cup {\mathcal{E}_{\text{miss}}} = \bigcup_{a \in {\mathcal{A}^{\text{sub}}}} \{a \in \mathcal{B}\}$
\begin{align*}
    \mathbb{P}[ {\mathcal{E}_{\text{sub}}} \cup {\mathcal{E}_{\text{miss}}} ] \leq \sum_{a \in {\mathcal{A}^{\text{sub}}}} \mathbb{P}\big[\{a \in \mathcal{B}\}\big] \leq \frac{L(K-A)}{K},
\end{align*}
which gives $\mathbb{P}[ {\mathcal{E}_{\text{sub}}} \cup {\mathcal{E}_{\text{miss}}} ] \leq 1 \wedge L(K-A)K^{-1}$

Now we are ready to bound the regret. On the event ${\mathcal{E}_{\text{opt}}} \coloneqq \big({\mathcal{E}_{\text{miss}}} \cup{\mathcal{E}_{\text{sub}}}\big)^\complement$, $\mathcal{B} \subseteq {\mathcal{A}^{\text{opt}}}$, therefore no regret is incurred. On the event ${\mathcal{E}_{\text{sub}}} $, the restricted bandit instance over $\mathcal{B}$ contains at least one globally optimal arm. Thus, the best mean reward among the arms in $\mathcal{B}$ equals $r^*$. By the minimax regret guarantee of MOSS, there exists a universal constant $C>0$ such that
\begin{align*}
\mathrm{Regret}_{\mathcal{B}}(T) \leq C\sqrt{LT}.
\end{align*}
for every realization of $\mathcal{B}$ satisfying ${\mathcal{E}_{\text{sub}}} $ and $T \geq K$ (see e.g., \citealp[Theorem 9.1]{lattimore2020bandit} for a proof). Finally, On the event ${\mathcal{E}_{\text{miss}}}$, the per-round regret is at most $1$, since all mean rewards belong to $[0,1]$. Therefore, the total regret is at most $T$. Combining the analysis gives that
\begin{align*}
& \mathrm{Regret}(T) \\
& \quad = \mathbb{E}\Big[ \mathrm{Regret}(T)\operatorname{\mathds{1}}\big( {\mathcal{E}_{\text{opt}}} \big) \Big] + \mathbb{E}\big[ \mathrm{Regret}(T)\operatorname{\mathds{1}}({\mathcal{E}_{\text{sub}}}) \big] + \mathbb{E}\big[ \mathrm{Regret}(T)\operatorname{\mathds{1}}({\mathcal{E}_{\text{miss}}}) \big] \\
& \quad \leq C\sqrt{LT} \mathbb{P}[ {\mathcal{E}_{\text{sub}}} \cup {\mathcal{E}_{\text{miss}}}  ] + T\mathbb{P}[{\mathcal{E}_{\text{miss}}}] \\
& \quad \leq  C\sqrt{LT}\bigg(\frac{L(K-A)}{K} \wedge 1\bigg)  + \frac{1}{2}.
\end{align*}

It remains to bound $L$. Using the elementary inequality $-\log(1-x) \geq x$ for all $x\in[0,1)$, we obtain $\log(K/(K-A)) = -\log(1-A/K) \geq A/K$, therefore
\begin{align*}
L \leq \bigg\lceil \frac{\log(2T)} {\log\big(K/(K-A)\big)} \bigg\rceil \leq 1+ \frac{\log(2T)}{\log\big(K/(K-A)\big)} \leq 1+\frac{K}{A}\log(2T) = \widetilde{O}\bigg(\frac{K}{A}\bigg),
\end{align*}
where the first inequality holds due to $\lceil x\rceil\leq x+1$. Substituting this inequality into the preceding regret bound yields
\begin{align*}
\mathrm{Regret}(T) = \widetilde{O} \Bigg( \bigg(\frac{(K-A)}{A} \wedge 1\bigg)\sqrt{\frac{KT}{A}} + \frac{1}{2} \Bigg) = \widetilde{O}\bigg( \frac{K-A}{\sqrt{KA}}\sqrt{T}\bigg).
\end{align*}
Since this bound is achieved by a valid algorithm uniformly over all instances with exactly $A$ optimal arms, the claimed minimax upper bound follows.

<!-- proof-2: body_md -->
The hard instance construction largely follows Theorem 15.2 of~\citet{lattimore2020bandit} and a similar idea was also presented in~\citet{de2021bandits}. For simplicity, we define $N_{{\mathcal{S}}}(T) := \sum_{a \in {\mathcal{S}}} N_{a}(T)$ for subset ${\mathcal{S}} \subseteq \mathcal{A}$. 
For any algorithm $\mathsf{Alg}$, we fix an arbitrary subset ${\mathcal{S}}_0 \subseteq \mathcal{A}$ such that $|{\mathcal{S}}_0| = A$. For a parameter $\Delta \in (0,1/2]$ to be specified later, define the first bandit instance $\nu_0$ by
\begin{align*}
r_0(a) := \begin{cases}
\Delta, & a \in {\mathcal{S}}_0,\\
0, & a \in {\mathcal{S}}_0^\complement.
\end{cases}
\end{align*}
Thus, $\nu_0$ has exactly $A$ optimal arms, namely the arms in ${\mathcal{S}}_0$. Let $\mathbb{P}_0$ and $\mathbb{E}_0$ denote probability and expectation under $\nu_0$ and $\mathtt{Alg}$. Since $\sum_{a \in {\mathcal{S}}_0^\complement} \mathbb{E}_0[N_{a}(T)] \leq T$, there exists a subset ${\mathcal{S}}_1 \subseteq {\mathcal{S}}_0^\complement$ with $|{\mathcal{S}}_1| = A$ such that
\begin{align}
\mathbb{E}_0[N_{{\mathcal{S}}_1}(T)] \leq \frac{AT}{K-A}. \label{eq:least-sampled-subset}
\end{align}
Indeed, one may choose ${\mathcal{S}}_1$ to consist of the $A$ least-sampled arms in ${\mathcal{S}}_0^\complement$. Such a subset exists because $A \leq K/4$, and hence $A < K-A$. We now define a second bandit instance $\nu_1$ by
\begin{align*}
r_1(a) := \begin{cases}
2\Delta, & a \in {\mathcal{S}}_1,\\
\Delta, & a \in {\mathcal{S}}_0,\\
0, & a \in ({\mathcal{S}}_0 \cup {\mathcal{S}}_1)^\complement.
\end{cases}
\end{align*}
The instance $\nu_1$ also has exactly $A$ optimal arms, namely the arms in ${\mathcal{S}}_1$. Similarly, we use $\mathbb{P}_1$ and $\mathbb{E}_1$ to denote probability and expectation under $\nu_1$ and $\mathtt{Alg}$, and let $\mathrm{Regret}_i(T)$ denote the expected regret under $\nu_i$. Now we consider the event $\mathcal{E} := \big\{N_{{\mathcal{S}}_1}(T) \geq T/2\big\}$. Under $\nu_0$, every arm in ${\mathcal{S}}_1$ is suboptimal with gap $\Delta$. Consequently,
\begin{align*}
\mathrm{Regret}_0(T) &\geq \Delta \mathbb{E}_0[N_{{\mathcal{S}}_1}(T)] \geq \frac{\Delta T}{2}\mathbb{P}_0[\mathcal{E}].
\end{align*}
Under $\nu_1$, every arm outside ${\mathcal{S}}_1$ has suboptimality gap at least $\Delta$. Therefore,
\begin{align*}
\mathrm{Regret}_1(T) \geq \Delta \mathbb{E}_1[T-N_{{\mathcal{S}}_1}(T)] \geq \frac{\Delta T}{2}\mathbb{P}_1 \big[\mathcal{E}^\complement \big].
\end{align*}
Combining the preceding two inequalities gives
\begin{align}
\mathrm{Regret}_0(T)+\mathrm{Regret}_1(T) \geq \frac{\Delta T}{2}\bigl(\mathbb{P}_0[\mathcal{E}]+\mathbb{P}_1[\mathcal{E}^\complement]\bigr).
\label{eq:regret-testing-reduction}
\end{align}

It remains to bound the right-hand-side of~\eqref{eq:regret-testing-reduction}. First, by Lemma~\ref{lem:bretagnolle-huber}
\begin{align*}
\mathbb{P}_0[\mathcal{E}]+\mathbb{P}_1[\mathcal{E}^\complement] \geq \frac{1}{2}\exp\big(-\operatorname{KL}(\mathbb{P}_0\|\mathbb{P}_1)\big)
\end{align*}
Since the two instances differ only on the arms in ${\mathcal{S}}_1$. For every $a \in {\mathcal{S}}_1$, the reward distribution is $\mathcal{N}(0,1)$ under $\nu_0$ and $\mathcal{N}(2\Delta,1)$ under $\nu_1$. Using the chain rule for KL divergence under adaptive sampling~\citep[Lemma 15.1]{lattimore2020bandit},
\begin{align*}
\operatorname{KL}(\mathbb{P}_0 \| \mathbb{P}_1) = \sum_{a \in {\mathcal{S}}_1}\mathbb{E}_0[N_{a}(T)]\operatorname{KL}\bigl(\mathcal{N}(0,1) \|\mathcal{N}(2\Delta,1)\bigr) = 2\Delta^2\mathbb{E}_0[N_{{\mathcal{S}}_1}(T)] \leq \frac{2\Delta^2AT}{K-A},
\end{align*}
where the last inequality follows from \eqref{eq:least-sampled-subset}. Now we choose $\Delta = \frac{1}{4}\sqrt{(K-A)(AT)^{-1}}$, which gives $\operatorname{KL}(\mathbb{P}_0 \|\mathbb{P}_1) \leq \frac{1}{8}$. As a result, $\mathbb{P}_0[\mathcal{E}]+\mathbb{P}_1[\mathcal{E}^\complement] \geq \exp(-1/8)/2$. Substituting this inequality into \eqref{eq:regret-testing-reduction} and taking the max over $\nu_0$ and $\nu_1$ gives
\begin{align*}
\max_{\nu_0, \nu_1} \mathrm{Regret}(T) \geq \frac{e^{-1/8}}{8}\Delta T = \Omega\bigg(\sqrt{\frac{(K-A)T}{A}}\bigg) = \Omega\bigg(\sqrt{\frac{KT}{A}}\bigg),
\end{align*}
which finishes the proof.

<!-- proposition-2: body_md -->
\label{prop:change-of-measure}

Let $(\Omega,\mathcal{G})$ be a measurable space, and let $\mathbb{P}$ and $\mathbb{Q}$ be probability measures on $(\Omega,\mathcal G)$ satisfying $\mathbb{P}\ll\mathbb{Q}$. Define the log-likelihood ratio $L = \log({\,\mathrm{d}}\mathbb{P}/\,{\,\mathrm{d}}\mathbb{Q}) $. Then, for every $\mathcal G$-measurable random variable $X:\Omega\to[0,b]$ and every $\gamma\in\mathbb{R}$,
\begin{align}
    \mathbb{E}_{\mathbb{P}}[X] \leq e^\gamma\mathbb{E}_{\mathbb{Q}}[X] +  b\mathbb{P}[L>\gamma].
    \label{eq:change-of-measure}
\end{align}
Equivalently,
\begin{align}
    \mathbb{E}_{\mathbb{Q}}[X]
    \geq
    e^{-\gamma}
    \big[
        \mathbb{E}_{\mathbb{P}}[X]
        -
        b\mathbb{P}(L>\gamma)
    \big].
    \label{eq:change-of-measure-lower}
\end{align}
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
Decompose the expectation according to whether the log-likelihood ratio exceeds $\gamma$:
\begin{align*}
    \mathbb{E}_{\mathbb{P}}[X]
    &=
    \mathbb{E}_{\mathbb{P}}\big[X\mathbf{1}\{L\leq\gamma\}\big]
    +
    \mathbb{E}_{\mathbb{P}}\big[X\mathbf{1}\{L>\gamma\}\big].
\end{align*}
For the first term, the change-of-measure identity gives
\begin{align*}
    \mathbb{E}_{\mathbb{P}}\big[X\mathbf{1}\{L\leq\gamma\}\big] =
    \mathbb{E}_{\mathbb{Q}}\big[Xe^L\mathbf{1}\{L\leq\gamma\}\big] \leq
    e^\gamma\mathbb{E}_{\mathbb{Q}}\big[X\mathbf{1}\{L\leq\gamma\}\big] \leq
    e^\gamma\mathbb{E}_{\mathbb{Q}}[X],
\end{align*}
where the last inequality follows from $X\geq 0$. For the second term, since $X\leq b$,
\begin{align*}
    \mathbb{E}_{\mathbb{P}}\big[X\mathbf{1}\{L>\gamma\}\big]
    \leq
    b\mathbb{P}(L>\gamma).
\end{align*}
Combining the two bounds yields
\begin{align*}
    \mathbb{E}_{\mathbb{P}}[X]
    \leq
    e^\gamma\mathbb{E}_{\mathbb{Q}}[X]
    +
    b\mathbb{P}(L>\gamma),
\end{align*}
which finishes the proof.

<!-- remark-4: body_md -->
\label{rmk:multiplicative-change-of-measure}
When specializing to $X=\operatorname{\mathds{1}}\{\mathcal{E}\}$, where $\mathcal{E} \in \mathcal{G}$ is any measurable event, Proposition~\ref{prop:change-of-measure} reduces to
$\mathbb{P}(\mathcal{E})\leq e^\gamma\mathbb{Q}(\mathcal{E})+\mathbb{P}[L>\gamma]$.
This change-of-measure inequality, in its event-based form, was first introduced for establishing instance-dependent lower bounds~\citep{degenne2019pure,garivier2021nonasymptotic}, as an modification to the previous KL-contraction-based technique~\citep{garivier2016optimal,garivier2019explore}. Recently, \citet{ji2026optimal} adopted this change-of-measure technique, together with subsequent combinatorial arguments, to prove the worst case lower bound in offline MABs with multiple optimal arms.

<!-- proposition-3: body_md -->
\label{prop:gaussian-bandit-likelihood-ratio}
Fix a bandit algorithm and two bandit instance $\mu$ and $\lambda$ with mean vectors $\bm{\lambda},\bm{\mu} \in\mathbb{R}^K$. Define the filtration
\begin{align*}
    &\mathcal{F}_0:=\{\emptyset,\Omega\}, \qquad \mathcal{F}_t:=\sigma(a_1,r_1,\ldots,a_t,r_t), \qquad t\in[T],\\
    &\mathcal{G}_{t+1} := \sigma(\mathcal{F}_t,a_{t+1}), \qquad t=0,\ldots,T-1.
\end{align*}
Then we see that, under the instance $\lambda$, $r_t \mid \mathcal{G}_t \sim \mathcal N(\lambda_{a_t},1)$. Let $\mathbb{P}_{\lambda}$ and $\mathbb{P}_{\mu}$ denote the trajectory distributions induced by the same algorithm under $\lambda$ and $\mu$, respectively. For every stopping time $\tau\leq T$, define
\begin{align*}
    \mathbb{P}_{\lambda}^{\tau} := \mathbb{P}_{\lambda}|_{\mathcal{F}_\tau},
    \qquad \mathbb{P}_{\mu}^{\tau} := \mathbb{P}_{\mu}|_{\mathcal{F}_\tau}.
\end{align*}
Then the log-likelihood ratio between the two stopped trajectory distributions satisfies
\begin{align}
    L_{\lambda,\mu}(\tau) &:= \log \frac{{\,\mathrm{d}}\mathbb{P}_{\lambda}^{\tau}}{{\,\mathrm{d}}\mathbb{P}_{\mu}^{\tau}} = \sum_{t=1}^{\tau} \bigg[(\lambda_{a_t}-\mu_{a_t}) (r_t-\lambda_{a_t}) + \frac{1}{2}(\lambda_{a_t}-\mu_{a_t})^2 \bigg].
    \label{eq:gaussian-bandit-llr-numerator}
\end{align}

<!-- proof-4: body_md -->
For every $t\in[T]$, let $\pi_t(\cdot\mid h_{t-1})$ denote the conditional distribution used by the algorithm to select $a_t$ given the history $h_{t-1} = (a_1,r_1,\ldots,a_{t-1},r_{t-1})$.  Since the same algorithm is used under both instances, the action-selection kernel $\pi_t$ is identical under $\mathbb{P}_{\lambda}$ and $\mathbb{P}_{\mu}$. In particular, it does not depend on the underlying bandit instance.

Let $\phi(x) := (\sqrt{2\pi})^{-1} \exp(-x^2/2)$ denote the standard Gaussian density. Fix any deterministic $n\in[T]$. With respect to the product of the counting measure on actions and the Lebesgue measure on rewards, the density of the trajectory up to time $n$ under $\lambda$ and $\mu$ are
\begin{align*}
    & p_{\lambda}(h_n) = \prod_{t=1}^n \pi_t(a_t\mid h_{t-1}) \phi(r_t-\lambda_{a_t}), \\
    & p_{\mu}(h_n) = \prod_{t=1}^n \pi_t(a_t\mid h_{t-1}) \phi(r_t-\mu_{a_t}).
\end{align*}
Therefore, the action-selection probabilities cancel, and we obtain
\begin{align}
    \frac{
        {\,\mathrm{d}}\left(\mathbb{P}_{\lambda}|_{\mathcal{F}_n}\right)
    }{
        {\,\mathrm{d}}\left(\mathbb{P}_{\mu}|_{\mathcal{F}_n}\right)
    }
    &=
    \prod_{t=1}^n
    \frac{
        \phi(r_t-\lambda_{a_t})
    }{
        \phi(r_t-\mu_{a_t})
    }.
    \label{eq:gaussian-bandit-likelihood-ratio-deterministic}
\end{align}
Define the likelihood-ratio process $Z_n = {\,\mathrm{d}} \mathbb{P}_{\lambda}|_{\mathcal{F}_n} / {\,\mathrm{d}} \mathbb{P}_{\mu}|_{\mathcal{F}_n}$ for all $n \in [T]$, and set $Z_0:=1$. We next show that $Z_{\tau}$ is the
Radon--Nikodym derivative between the two trajectory distributions
restricted to $\mathcal{F}_{\tau}$. Indeed, for every event
$E\in\mathcal{F}_{\tau}$, we have
\begin{align*}
    \mathbb{E}_{\mu}\big[Z_{\tau}\operatorname{\mathds{1}}\{E\}\big] =
    \sum_{n=0}^T  \mathbb{E}_{\mu}\big[ Z_n\operatorname{\mathds{1}}\{E\cap\{\tau=n\}\} \big] = 
    \sum_{n=0}^T
    \mathbb{P}_{\lambda}\big(E\cap\{\tau=n\}\big) = \mathbb{P}_{\lambda}(E),
\end{align*}
where the second equality follows because
$E\cap\{\tau=n\}\in\mathcal{F}_n$. Consequently, $Z_\tau = {\,\mathrm{d}} \mathbb{P}_{\lambda}^{\tau} / {\,\mathrm{d}} \mathbb{P}_{\mu}^{\tau}$.

It remains to compute the logarithm of $Z_{\tau}$. For every
$t\in[T]$,
\begin{align*}
    \log
    \frac{
        \phi(r_t-\lambda_{a_t})
    }{
        \phi(r_t-\mu_{a_t})
    }
    = 
    \frac{1}{2}
    \big[
        (r_t-\mu_{a_t})^2
        -
        (r_t-\lambda_{a_t})^2
    \big] = 
    (\lambda_{a_t}-\mu_{a_t})
    (r_t-\lambda_{a_t})
    +
    \frac{1}{2}
    (\lambda_{a_t}-\mu_{a_t})^2.
\end{align*}
Taking the logarithm of
\eqref{eq:gaussian-bandit-likelihood-ratio-deterministic} and
evaluating the resulting likelihood-ratio process at $\tau$ therefore
gives
\begin{align*}
    L_{\lambda,\mu}(\tau)
    =
    \log
    \frac{
        {\,\mathrm{d}}\mathbb{P}_{\lambda}^{\tau}
    }{
        {\,\mathrm{d}}\mathbb{P}_{\mu}^{\tau}
    }
    =
    \sum_{t=1}^{\tau} \bigg[ (\lambda_{a_t}-\mu_{a_t}) (r_t-\lambda_{a_t}) + \frac{1}{2}(\lambda_{a_t}-\mu_{a_t})^2\bigg],
\end{align*}
which proves the proposition.

The hard instance construction is inspired by Exercise 15.2 of~\citet{lattimore2020bandit}. We consider the multi-armed bandit class parameterized by mean vectors:
\begin{align*}
\mu \in \mathcal{V}_A:= \bigg\{ \bm{\mu} \in \bigg\{\frac{1}{2} , \frac{1}{2} -\Delta\bigg\}^K \bigg| \sum_{i=1}^K \operatorname{\mathds{1}}(\mu_i= 1 / 2) = A \bigg\},
\end{align*}
where $\Delta > 0$ is a parameter to be determined later. We further consider an auxiliary instance $\theta$ such that $\bm{\theta} = (1 / 2, \cdots, 1/2)^\top \in \mathbb{R}^K$, and let $\mathcal{V} = \mathcal{V}_A \cup \{\theta\}$. Unless stated otherwise, we assume that noise of rewards is standard Gaussian. For any $\mu \in \mathcal{V}$, the corresponding instance is given by $([K], r_{\mu}, T)$ where the reward is given by $r_{\mu}(i) = \mu_i$.
For any $\mu \in \mathcal{V}_A$, we use ${\mathcal{A}^{\text{opt}}}(\mu)$ to denote the set of optimal actions (those with reward $1/2$) and ${\mathcal{A}^{\text{sub}}}(\mu)$ to denote the set of sub-optimal arms. Our goal is to show that, for any algorithm $\mathtt{Alg}$, there exists $\mu \in \mathcal{V}_A$ such that that 
\begin{align*}
    \mathrm{Regret}_{\mu, \mathtt{Alg}}(T) =\widetilde{\Omega}\bigg(\frac{K-A}{K}\sqrt{T}\bigg).
\end{align*}
In the subsequent proof, we will drop the subscript of $\mathtt{Alg}$ when there is no ambiguity.

Now we consider any $\mu \in \mathcal{V}_A$ and its difference between $\theta$, in terms of the distribution of induced trajectories. Consider the random variable $X_{\mu} = \sum_{a \in {\mathcal{A}^{\text{sub}}}(\mu)} N_{a}(T)$, then we know that 
\begin{align*}
    \mathrm{Regret}_{\mu}(T) = \mathbb{E}_{\mu}[X_{\mu}]\Delta.
\end{align*}
On the other side, since $X_{\mu} \leq T$, Proposition~\ref{prop:change-of-measure} tells us that for any $\gamma > 0$,
\begin{align}
    \mathbb{E}_{\mu}[X_{\mu}] \geq e^{-\gamma} \bigg[ \mathbb{E}_{\theta}[X_{\mu}] - T \mathbb{P}_{\theta}\bigg(\log \frac{{\,\mathrm{d}}\mathbb{P}_{\theta}}{{\,\mathrm{d}}\mathbb{P}_{\mu}}  > \gamma \bigg)\bigg]. \label{eq:lower-bound-change-of-measure}
\end{align}
To lower bound the right hand side, we need to properly choose the threshold $\gamma$, which requires the computation of $\log ({\,\mathrm{d}}\mathbb{P}_{\theta} / {\,\mathrm{d}}\mathbb{P}_{\mu})$. Applying Proposition~\ref{prop:gaussian-bandit-likelihood-ratio} with $\tau=T$ gives that 
\begin{align*}
    \log \frac{{\,\mathrm{d}}\mathbb{P}_{\theta}}{{\,\mathrm{d}}\mathbb{P}_{\mu}} &= \sum_{t=1}^T
    \bigg[ (\theta_{a_t}-\mu_{a_t})(r_t-\theta_{a_t}) + \frac{1}{2}
        (\theta_{a_t}-\mu_{a_t})^2 \bigg] \\
    & = \underbrace{\frac{1}{2}\sum_{a \in {\mathcal{A}^{\text{sub}}}(\mu)} N_{a}(T)(\theta_a - \mu_a)^2}_{I_1} + \underbrace{\sum_{t=1}^T \bigg(r_t-\frac12\bigg) \Delta\operatorname{\mathds{1}}\{a_t \in {\mathcal{A}^{\text{sub}}}(\mu)\}}_{I_2},
\end{align*}
Since $\sum_{a \in {\mathcal{A}^{\text{sub}}}(\mu)} N_{a}(T) \leq T$, we know that $I_1 \leq T\Delta^2/2$. For the second term, we consider
                                                                                               
\begin{align*}
    M_t = \sum_{s=1}^t \bigg(r_s - \frac12 \bigg) \Delta\operatorname{\mathds{1}}\{\mathcal{E}_s(\mu)\}, \quad V_t = \sum_{s=1}^t \Delta^2 \operatorname{\mathds{1}}\{\mathcal{E}_s(\mu)\}, \text{ where } \mathcal{E}_s(\mu) = \{a_s \in {\mathcal{A}^{\text{sub}}}(\mu)\}
\end{align*}
We further define
\begin{align*}
    Z_t = \exp\bigg(\lambda M_t-\frac{\lambda^2V_t}{2} \bigg), \quad t=0,1,\ldots,T.
\end{align*}
Then under the probability distribution $\mathbb{P}_\theta$, we see that for any $\lambda \in \mathbb{R}$,
\begin{align*}
    & \mathbb{E}_{\theta}[Z_t | \mathcal{F}_{t-1}] = Z_{t-1} \mathbb{E}_{\theta}\bigg[\underbrace{\exp\bigg(\lambda (r_t - 1/2) \Delta\operatorname{\mathds{1}}\{\mathcal{E}_t(\mu)\}-\frac{\lambda^2\Delta^2 \operatorname{\mathds{1}}\{\mathcal{E}_t(\mu)\}}{2} \bigg)}_{(\star)} \bigg| \mathcal{F}_{t-1}\bigg] 
\end{align*}
For $(\star)$, we first take expectation on $\mathcal{G}_t$, which gives that 
\begin{align*}
    \mathbb{E}_{\theta}[(\star) | \mathcal{G}_t] &  =1 - \operatorname{\mathds{1}}\{\mathcal{E}_t(\mu)\} +  \operatorname{\mathds{1}}\{\mathcal{E}_t(\mu)\} \mathbb{E}_{\theta}\bigg[\exp\bigg(\lambda (r_t - 1/2) \Delta -\frac{\lambda^2\Delta^2 }{2} \bigg) \bigg| \mathcal{G}_t \bigg] \\
    & \leq 1 - \operatorname{\mathds{1}}\{\mathcal{E}_t(\mu)\} + \operatorname{\mathds{1}}\{\mathcal{E}_t(\mu)\} \\
    & = 1,
\end{align*}
where the inequality holds due to that $r_t - 1/2$ is $1$-sub-gaussian conditioned on $\mathcal{G}_t$ on $\mathbb{P}_\theta$. This immediately gives $\mathbb{E}_{\theta}[(\star) | \mathcal{F}_{t-1}] \leq 1$, verifying that $Z_t$ is a super-martingale for any $\lambda$. Now invoking Lemma~\ref{lem:sub-gaussian-super-martingale} with $\tau=T$, since $I_2 = M_T$, we see that for any $\beta > 0$,
\begin{align*}
    \mathbb{P}_{\theta}(I_2 \ge \beta) \leq \exp\bigg(-\frac{\beta^2}{2T\Delta^2}\bigg).
\end{align*}
Now combining the estimation of $I_1$ and $I_2$ and set $\beta = 2 \Delta \sqrt{T\log K}$, and $\gamma = \beta + T\Delta^2 /2$, we obtain that 
\begin{align*}
    \mathbb{P}_{\theta}\bigg(\log \frac{{\,\mathrm{d}}\mathbb{P}_{\theta}}{{\,\mathrm{d}}\mathbb{P}_{\mu}}  > \gamma \bigg) \leq \exp\bigg(-\frac{\beta^2}{2T\Delta^2}\bigg) = \frac{1}{K^2}.
\end{align*}
Plugging into~\eqref{eq:lower-bound-change-of-measure}, we see that
\begin{align*}
    \mathbb{E}_{\mu}[X_{\mu}] & \geq \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg)   \bigg[ \mathbb{E}_{\theta}[X_{\mu}] - T \mathbb{P}_{\theta}\bigg(\log \frac{{\,\mathrm{d}}\mathbb{P}_{\theta}}{{\,\mathrm{d}}\mathbb{P}_{\mu}}  > \gamma \bigg)\bigg] \\
    & \geq \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg)   \bigg[ \mathbb{E}_{\theta}[X_{\mu}] - \frac{T}{K^2}\bigg].
\end{align*}
Now we sum up over all $\mu \in \mathcal{V}_A$ and obtains that
\begin{align*}
    \sum_{\mu \in \mathcal{V}_A} \mathbb{E}_{\mu}[X_{\mu}] & \geq \exp \Bigg(-\beta-\frac{T\Delta^2}{2}\bigg) \bigg(\sum_{\mu \in \mathcal{V}_A} \mathbb{E}_{\theta}[X_{\mu}]  - \sum_{\mu \in \mathcal{V}_A} \frac{T}{K^2} \Bigg) \\
    & = \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg) \Bigg(\sum_{\mu \in \mathcal{V}_A} \sum_{a \in {\mathcal{A}^{\text{sub}}}(\mu)} \mathbb{E}_{\theta}[N_{a}(T)] - \binom{K}{K-A}\frac{T}{K^2} \Bigg) \\
    & = \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg) \Bigg(\sum_{a \in \mathcal{A}} \sum_{\mu:a \in {\mathcal{A}^{\text{sub}}}(\mu)}  \mathbb{E}_{\theta}[N_{a}(T)] - \binom{K}{K-A}\frac{T}{K^2} \Bigg) \\
    & = \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg) \Bigg(T \binom{K-1}{K-A-1}   - \binom{K}{K-A}\frac{T}{K^2} \Bigg),
\end{align*}
where the second equation is given by $|\mathcal{V}_A| = \binom{K}{K-A}$ and the last equation holds due to $\sum_{a \in \mathcal{A}} N_{a}(T) = T$ and $|\{\mu:a \in {\mathcal{A}^{\text{sub}}}(\mu)\}| = \binom{K-1}{K-A-1}$. Now take maximum over all $\mu \in \mathcal{V}_A$ and leverage the fact that maximum is larger that average, we see that
\begin{align*}
    \sup_{\mu \in \mathcal{V}_A} \mathbb{E}_{\mu}[X_{\mu}] & \geq T\exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg) \bigg[\binom{K}{K-A}^{-1}\binom{K-1}{K-A-1} - \frac{1}{K^2} \bigg] \\
    & \geq T \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg) \bigg(\frac{K-A}{K} - \frac{1}{K^2}\bigg)  \\
    & \geq \frac{(K-A)T}{2K} \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg). 
\end{align*}
Recall that $\mathrm{Regret}_{\mu}(T) = \mathbb{E}_{\mu}[X_{\mu}]\Delta$, we see
\begin{align*}
    \sup_{\mu \in \mathcal{V}_A} \mathrm{Regret}_{\mu}(T) \geq \frac{(K-A)T\Delta}{2K} \exp \bigg(-\beta-\frac{T\Delta^2}{2}\bigg).
\end{align*}
Finally, recall that $\beta = 2 \Delta \sqrt{T\log K}$. Now we select $\Delta = \sqrt{(T\log K)^{-1}}$, we see that $\beta = 2$ and $\gamma = 2 + (2\log K)^{-1} \leq 3$. This leads to
\begin{align*}
    \sup_{\mu \in \mathcal{V}_A} \mathrm{Regret}_{\mu}(T) \gtrsim \frac{(K-A)T}{K\sqrt{T\log K}} = \widetilde{\Omega}\bigg(\frac{K-A}{K}\sqrt{T}\bigg),
\end{align*}
for any algorithm $\mathtt{Alg}$, which finishes the proof.

Fix any $m\in[T]$ and $\ell\geq1$, and for simplicity we write
\begin{align*}
    q_{\mathtt{Alg}}(m):= \max_{{\mathcal{S}}\subseteq\mathcal{A}:\,|{\mathcal{S}}|=K-A}
    \mathbb{P}_\theta\big(N_{{\mathcal{S}}}(T)\geq m\big).
\end{align*}
Recall that $\theta$ is the all-equal instance with mean vector $\bm{\theta}=\mathbf 1/2$. For every subset ${\mathcal{S}}\subseteq\mathcal{A}$ with $|{\mathcal{S}}|=K-A$, construct an
alternative Gaussian bandit instance $\nu_{{\mathcal{S}}}$ with mean vector
$\bm{\nu}_{{\mathcal{S}}}$ given by
\begin{align*}
    \nu_{{\mathcal{S}},a}
    :=
    \begin{cases}
        1/2-\Delta, & a\in{\mathcal{S}},\\
        1/2, & a\notin{\mathcal{S}},
    \end{cases}
    \qquad
    \Delta
    :=
    \frac{1}{4\sqrt{m\ell}}.
\end{align*}
Since $|{\mathcal{S}}| = K-A$, the instance $\nu_{{\mathcal{S}}}$ has exactly $A$
optimal arms. Moreover, $\Delta\leq1/4$, so all its reward means
belong to $[0,1]$. We show that, whenever the algorithm pulls only $m$ or fewer than $m$ times on the set ${\mathcal{S}}$, it cannot reliably distinguish $\theta$ and $\nu_{{\mathcal{S}}}$. To show this, for a fixed ${\mathcal{S}}$, we define the stopping time $\tau_{{\mathcal{S}}}$ at which the pull count on ${\mathcal{S}}$ is exactly $m$
\begin{align*}
    H_{{\mathcal{S}}} := \{N_{{\mathcal{S}}}(T)\geq m\},\quad 
    \tau_{{\mathcal{S}}} := \inf\{t\in[T]:N_{{\mathcal{S}}}(t)=m\}\wedge T.
\end{align*}
Then $\tau_{{\mathcal{S}}}$ is a bounded $(\mathcal{F}_t)_{t=0}^T$-stopping time, and
\begin{align}
    H_{{\mathcal{S}}} = \{N_{{\mathcal{S}}}(\tau_{{\mathcal{S}}})=m\} \in  \mathcal{F}_{\tau_{{\mathcal{S}}}}.

\end{align}

    \frac{\Delta^2}{2}
    N_{{\mathcal{S}}}(\tau_{{\mathcal{S}}}).
\end{align*}

\begin{align*}

    & \quad = \exp\bigg(\lambda M_{t-1}-\frac{\lambda^2V_{t-1}}{2} \bigg) \mathbb{E}_{\theta}\bigg[\underbrace{\exp\bigg(\lambda (r_t-1/2) \Delta\operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\}-\frac{\lambda^2\Delta^2 \operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\}}{2} \bigg)}_{(\star)} \bigg| \mathcal{F}_{t-1}\bigg]

    & = 1,
\end{align*}
where the inequality holds due to that $r_t-1/2$ is $1$-sub-gaussian conditioned on $\mathcal{G}_t$ on $\mathbb{P}_\theta$. Now invoking Lemma~\ref{lem:sub-gaussian-super-martingale} with $x = \Delta\sqrt{2m\ell}$, since $V_\tau \leq m\Delta^2$, we have
\begin{align*}

Recall that $\Delta = (4\sqrt{m\ell})^{-1}$, this means that $\Delta\sqrt{2m\ell} + m\Delta^2/2 = 1/2\sqrt{2} + 1/32\ell < \log 2$ since $\ell \geq 1$

Proposition~\ref{prop:change-of-measure} on the measurable space
$(\Omega,\mathcal{F}_{\tau_{{\mathcal{S}}}})$ with $\mathbb{P}= \mathbb{P}_\theta^{\tau_{{\mathcal{S}}}}$, $\mathbb{Q} =   \mathbb{P}_{\nu_{{\mathcal{S}}}}^{\tau_{{\mathcal{S}}}}$, $X = \operatorname{\mathds{1}}\{H_{{\mathcal{S}}}\}$, $b=1$ and $\gamma = \log 2$. Using \eqref{eq:pull-count-tail-llr}, we obtain
\begin{align}
    \mathbb{P}_{\nu_{{\mathcal{S}}}}(H_{{\mathcal{S}}}) \geq \frac{1}{2} \big[\mathbb{P}_\theta(H_{{\mathcal{S}}}) -  \mathbb{P}_\theta(L_{{\mathcal{S}}}>\log 2) \big] \geq \frac{1}{2} \big[ \mathbb{P}_\theta(H_{{\mathcal{S}}})  -\exp(-\ell) \big].
    \label{eq:pull-count-tail-event-com}
\end{align}
Every arm in ${\mathcal{S}}$ is suboptimal under $\nu_{{\mathcal{S}}}$ with gap
$\Delta$. Hence, the assumed regret guarantee gives
\begin{align*}
    \Gamma\frac{K-A}{K}\sqrt{T} \geq \mathrm{Regret}_{\nu_{{\mathcal{S}}},\mathtt{Alg}}(T)= \Delta
    \mathbb{E}_{\nu_{{\mathcal{S}}}}[N_{{\mathcal{S}}}(T)] &\geq \Delta m\mathbb{P}_{\nu_{{\mathcal{S}}}}(H_{{\mathcal{S}}})  \geq \frac{\Delta m}{2} \big[ \mathbb{P}_\theta(H_{{\mathcal{S}}}) - \exp(-\ell)\big].
\end{align*}
Rearranging and substituting $\Delta=(4\sqrt{m\ell})^{-1}$ yields
\begin{align*}
    \mathbb{P}_\theta\big(N_{{\mathcal{S}}}(T)\geq m\big) \leq  \exp(-\ell)  + \frac{2\Gamma(K-A)\sqrt{T}}{K\Delta m} =  \exp(-\ell) + 8\Gamma\frac{K-A}{K}    \sqrt{\frac{T\ell}{m}}.
\end{align*}
Since this holds for every $|{\mathcal{S}}|=K-A$, taking maximum over all $|{\mathcal{S}}|=K-A$ gives the desired upper bound
\begin{align}
    q_{\mathtt{Alg}}(m) \leq \exp(-\ell) +  8\Gamma\frac{K-A}{K} \sqrt{\frac{T\ell}{m}}.
    \label{eq:pull-count-tail-upper}
\end{align}





We next prove the lower bound on $q_{\mathtt{Alg}}(m)$. For every
${\mathcal{S}}\subseteq\mathcal{A}$ with $|{\mathcal{S}}|=K-A$, define $Y_{{\mathcal{S}}} = N_{{\mathcal{S}}}(T) \operatorname{\mathds{1}} \{ N_{{\mathcal{S}}}(T) < m\} = N_{{\mathcal{S}}}(T)\operatorname{\mathds{1}}\{H_{{\mathcal{S}}}^{\complement}\}$. On $H_{{\mathcal{S}}}^{\complement}$, the stopping time satisfies $\tau_{{\mathcal{S}}}=T$, whereas on $H_{{\mathcal{S}}}$ both sides below vanish. Therefore, $Y_{{\mathcal{S}}} = N_{{\mathcal{S}}}(\tau_{{\mathcal{S}}})\operatorname{\mathds{1}}\{H_{{\mathcal{S}}}^{\complement}\}$. In particular, $Y_{{\mathcal{S}}}$ is $\mathcal{F}_{\tau_{{\mathcal{S}}}}$-measurable and takes values in $[0,T]$. Once again, applying Proposition~\ref{prop:change-of-measure} once more on $(\Omega,\mathcal{F}_{\tau_{{\mathcal{S}}}})$, now with $\mathbb{P}= \mathbb{P}_\theta^{\tau_{{\mathcal{S}}}}$, $\mathbb{Q} =   \mathbb{P}_{\nu_{{\mathcal{S}}}}^{\tau_{{\mathcal{S}}}}$, $X = Y_{{\mathcal{S}}}$, $b=T$ and $\gamma = \log 2$, and using \eqref{eq:pull-count-tail-llr}, gives
\begin{align*}
    \mathbb{E}_{\nu_{{\mathcal{S}}}}[Y_{{\mathcal{S}}}]  \geq \frac{1}{2} \big[ \mathbb{E}_\theta[Y_{{\mathcal{S}}}] - T\mathbb{P}_\theta(L_{{\mathcal{S}}}>\log 2) \big] \geq \frac{1}{2} \big[ \mathbb{E}_\theta[Y_{{\mathcal{S}}}] - T\exp(-\ell)\big].
\end{align*}
Since $Y_{{\mathcal{S}}}\leq N_{{\mathcal{S}}}(T)$, we have
\begin{align*}
    \mathbb{E}_{\nu_{{\mathcal{S}}}}[N_{{\mathcal{S}}}(T)] \geq \mathbb{E}_{\nu_{{\mathcal{S}}}} [Y_{{\mathcal{S}}}] \geq \frac{1}{2} \big[ \mathbb{E}_\theta \big[ N_{{\mathcal{S}}}(T)\operatorname{\mathds{1}}\{H_{{\mathcal{S}}}^{\complement}\} \big] - T\exp(-\ell) \big].
\end{align*}
Moreover, as $\mathbb{P}_\theta(H_{{\mathcal{S}}})\leq q_{\mathtt{Alg}}(m)$, it follows that
\begin{align*}
    \mathbb{E}_{\nu_{{\mathcal{S}}}}[N_{{\mathcal{S}}}(T)] & \geq \frac{1}{2} \big[ \mathbb{E}_\theta \big[ N_{{\mathcal{S}}}(T)\operatorname{\mathds{1}}\{H_{{\mathcal{S}}}^{\complement}\} \big] - T\exp(-\ell) \big] \\
    &\geq  \frac{1}{2} \big[\mathbb{E}_\theta[N_{{\mathcal{S}}}(T)] - T\mathbb{P}_\theta(H_{{\mathcal{S}}}) - T\exp(-\ell)  \big]\\
    &\geq \frac{1}{2}\mathbb{E}_\theta[N_{{\mathcal{S}}}(T)] - \frac{T}{2}q_{\mathtt{Alg}}(m) - \frac{T}{2}\exp(-\ell).
\end{align*}
Thus,
\begin{align}
    \mathbb{E}_{\nu_{{\mathcal{S}}}}[N_{{\mathcal{S}}}(T)]
    \geq
    \frac{1}{2}\mathbb{E}_\theta[N_{{\mathcal{S}}}(T)]
    -
    \frac{T}{2}q_{\mathtt{Alg}}(m)
    -
    \frac{T}{2}\exp(-\ell).
    \label{eq:pull-count-tail-single-set-lower}
\end{align}

We now average
\eqref{eq:pull-count-tail-single-set-lower} over all subsets
${\mathcal{S}}\subseteq\mathcal{A}$ with $|{\mathcal{S}}|=K-A$. Since every arm belongs to
$\binom{K-1}{K-A-1}$ such subsets,
\begin{align*}
    \frac{1}{\binom{K}{K-A}} \sum_{{\mathcal{S}}\subseteq\mathcal{A}:\,|{\mathcal{S}}|=K-A}\mathbb{E}_\theta[N_{{\mathcal{S}}}(T)] = \frac{\binom{K-1}{K-A-1}}{\binom{K}{K-A}} \sum_{a\in\mathcal{A}}\mathbb{E}_\theta[N_{a}(T)] = \frac{K-A}{K}T.
\end{align*}
Therefore,
\begin{align}
    \frac{1}{\binom{K}{K-A}}  \sum_{{\mathcal{S}}\subseteq\mathcal{A}:\,|{\mathcal{S}}|=K-A}  \mathbb{E}_{\nu_{{\mathcal{S}}}}[N_{{\mathcal{S}}}(T)] \geq \frac{T}{2} \bigg[ \frac{K-A}{K} - q_{\mathtt{Alg}}(m) - \exp(-\ell)\bigg].
    \label{eq:pull-count-tail-averaged-lower}
\end{align}
Since 
Every $\nu_{{\mathcal{S}}}$ has exactly $A$ optimal arms, we have $\mathrm{Regret}_{\nu_{{\mathcal{S}}},\mathtt{Alg}}(T) = \Delta \mathbb{E}_{\nu_{{\mathcal{S}}}}[N_{{\mathcal{S}}}(T)]$ . Together with \eqref{eq:pull-count-tail-averaged-lower}, this implies that
\begin{align*}
    \Gamma\frac{K-A}{K}\sqrt{T} \geq  \frac{\Delta}{\binom{K}{K-A}} \sum_{{\mathcal{S}}\subseteq\mathcal{A}:\,|{\mathcal{S}}|=K-A} \mathbb{E}_{\nu_{{\mathcal{S}}}} [N_{{\mathcal{S}}}(T)] 
    \geq \frac{\Delta T}{2} \bigg[\frac{K-A}{K} -  q_{\mathtt{Alg}}(m) - \exp(-\ell)\bigg].
\end{align*}
                                                               
Rearranging and substituting
$\Delta=(4\sqrt{m\ell})^{-1}$ gives
\begin{align}
    q_{\mathtt{Alg}}(m) &\geq \frac{K-A}{K} - \exp(-\ell) - \frac{2\Gamma(K-A)\sqrt{T}}{K\Delta T} \\
    &= \frac{K-A}{K} - \exp(-\ell) - 8\Gamma\frac{K-A}{K}\sqrt{\frac{m\ell}{T}}.
    \label{eq:pull-count-tail-lower}
\end{align}

Combining
\eqref{eq:pull-count-tail-upper} and
\eqref{eq:pull-count-tail-lower} proves
\begin{align*}
    \frac{K-A}{K}- \exp(-\ell) -     8\Gamma\frac{K-A}{K}  \sqrt{\frac{m\ell}{T}} \leq q_{\mathtt{Alg}}(m) \leq  \exp(-\ell)  + 8\Gamma\frac{K-A}{K}\sqrt{\frac{T\ell}{m}},
\end{align*}
as desired.

\begin{align*}
    \mu_a := \begin{cases}
        1/2+\Delta, & a\in\mathcal{U},\\
        1/2, & a\in\mathcal{U}^{\complement}.
    \end{cases}
\end{align*}
Since $|\mathcal{U}|=B$, every arm in $\mathcal{U}$ is strictly better than every arm outside $\mathcal{U}$, so $\mu$ has exactly $B$ optimal arms, namely ${\mathcal{A}^{\text{opt}}}(\mu)=\mathcal{U}$, and hence $\mu\in{\Xi}_{K, B}$. As in the proof of Lemma~\ref{lem:pull-count-tail}, define
\begin{align*}
    H:=\{N_{\mathcal{U}}(T)\geq m\}, \qquad \tau:=\inf\{t\in[T]:N_{\mathcal{U}}(t)=m\}\wedge T,
\end{align*}
so that $\tau$ is a bounded $(\mathcal{F}_t)_{t=0}^T$-stopping time and, since the pull count $N_{\mathcal{U}}(t)$ increases by at most one per round, $H=\{N_{\mathcal{U}}(\tau)=m\}\in\mathcal{F}_{\tau}$. Consequently $H^{\complement}=\{N_{\mathcal{U}}(T)<m\}$ also belongs to $\mathcal{F}_{\tau}$, and on $H^{\complement}$ we have $\tau=T$.

We first show that, up to $\tau$, the trajectory distributions induced by $\theta$ and $\mu$ remain close. Writing $\mathbb{P}_\theta^{\tau}:=\mathbb{P}_\theta|_{\mathcal{F}_{\tau}}$ and $\mathbb{P}_{\mu}^{\tau}:=\mathbb{P}_{\mu}|_{\mathcal{F}_{\tau}}$, and using $\theta_a-\mu_a=-\Delta\operatorname{\mathds{1}}\{a\in\mathcal{U}\}$, Proposition~\ref{prop:gaussian-bandit-likelihood-ratio} gives
\begin{align*}
    \log \frac{{\,\mathrm{d}}\mathbb{P}_\theta^{\tau}}{{\,\mathrm{d}}\mathbb{P}_{\mu}^{\tau}} & = \sum_{t=1}^{\tau} \bigg[ (\theta_{a_t}-\mu_{a_t}) (r_t-\theta_{a_t}) + \frac{1}{2}(\theta_{a_t}-\mu_{a_t})^2\bigg] \\
    & = -\Delta \sum_{t=1}^{\tau} \operatorname{\mathds{1}}\{a_t\in\mathcal{U}\}(r_t-1/2) +
    \frac{\Delta^2}{2}
    N_{\mathcal{U}}(\tau).
\end{align*}
Denote $M_t:=-\Delta\sum_{s=1}^t\operatorname{\mathds{1}}\{a_s\in\mathcal{U}\}(r_s-1/2)$ and $V_t:=\Delta^2N_{\mathcal{U}}(t)$. Exactly as in the proof of Lemma~\ref{lem:pull-count-tail}, since $r_t-1/2$ is $1$-sub-Gaussian conditioned on $\mathcal{G}_t$ under $\mathbb{P}_\theta$ (regardless of the sign of $\Delta$), $\exp(\lambda M_t-\lambda^2V_t/2)$ is a super-martingale under $\mathbb{P}_\theta$ for every $\lambda\in\mathbb{R}$. Invoking Lemma~\ref{lem:sub-gaussian-super-martingale} with $x=\Delta\sqrt{2m\ell}$, and using $V_{\tau}\leq m\Delta^2$, we obtain
\begin{align*}

\end{align*}
Since $\Delta^2N_{\mathcal{U}}(\tau)/2 \leq m\Delta^2/2$, this gives
\begin{align*}
    \mathbb{P}_{\theta} \bigg(\log \frac{{\,\mathrm{d}}\mathbb{P}_\theta^{\tau}}{{\,\mathrm{d}}\mathbb{P}_{\mu}^{\tau}} \geq \Delta\sqrt{2m\ell} + m\Delta^2/2 \bigg) \leq \exp(-\ell).
\end{align*}

\end{align}

We now transfer the bound on $\mathbb{P}_\theta(H^{\complement})$ to $\mu$. \begin{align}

\end{align}

It remains to bound the left-hand side of~\eqref{eq:exploration-profile-minimax-optimal-com} using the regret guarantee of $\mathtt{Alg}$ on $\mu$. Every arm in $\mathcal{U}^{\complement}$ is suboptimal under $\mu$ with gap $\Delta$, so
\begin{align*}
    \mathrm{Regret}_{\mu,\mathtt{Alg}}(T) = \Delta\,\mathbb{E}_{\mu}\big[N_{\mathcal{U}^{\complement}}(T)\big] = \Delta\big(T-\mathbb{E}_{\mu}[N_{\mathcal{U}}(T)]\big).
\end{align*}
On the event $H^{\complement}=\{N_{\mathcal{U}}(T)<m\}$ we have $N_{\mathcal{U}^{\complement}}(T)>T-m$, so, using~\eqref{eq:exploration-profile-minimax-optimal-com} and $m\leq T/2$,
\begin{align*}
    \mathrm{Regret}_{\mu,\mathtt{Alg}}(T) & \geq \Delta(T-m)\mathbb{P}_{\mu}\big(H^{\complement}\big) \\
    & \geq \frac{\Delta(T-m)}{2}\Big[\mathbb{P}_\theta\big(H^{\complement}\big)-\exp(-\ell)\Big] \\
    & \geq \frac{\Delta T}{4}\Big[\mathbb{P}_\theta\big(H^{\complement}\big)-\exp(-\ell)\Big].
\end{align*}
On the other hand, since $\mu\in{\Xi}_{K, B}$, the assumed minimax-optimality of $\mathtt{Alg}$ on ${\Xi}_{K, B}$ gives
\begin{align*}

\end{align*}
where the second inequality uses $K-B\leq K$. Combining the last two displays and rearranging leads to
\begin{align*}

\end{align*}
where the equality substitutes $\Delta^{-1}=4\sqrt{m\ell}$. This proves the claimed result in~\eqref{eq:exploration-profile-minimax-optimal-main}.

\begin{align*}

    \begin{cases}

        1/2-\Delta, & a \in (\mathcal{S}_0 \cup \mathcal{S}_1)^\complement.
    \end{cases}
\end{align*}
The instance $\nu_1$ has exactly $B$ optimal arms, namely the arms in $\mathcal{S}_1$. Let $\mathbb{P}_1$ and $\mathbb{E}_1$ denote probability and expectation under $\nu_1$ and $\mathsf{Alg}$.

        \mathcal{N}(1/2+\Delta,1)

    \mathbb{E}_0[N_{\mathcal{S}_1}(T)]\\

        4,
        \frac{16BR_A}{K-A}

    \max\bigg\{
        \frac{4BR_A}{K-A},
        \frac{16B^2R_A^2}{(K-A)^2}
    \bigg\} \leq

\end{align*}
where the last inequality follows from the condition
\begin{align*}

\end{align*}
which can be derived directly from $B \leq A/(8\Gamma_{K,T})$ and $T \geq 256B^2\Gamma^2_{K,T}/(KA)$. Now we combine this inequality with~\eqref{eq:nonadapt-TV-expectation} and $\operatorname{TV}(\mathbb{P}_0,\mathbb{P}_1) \leq 1/4$, which leads to
\begin{align*}
    \mathbb{E}_1[N_{\mathcal{S}_1}(T)]
    \leq
    \frac{T}{4}
    +
    \frac{T}{4}
    =
    \frac{T}{2}.
\end{align*}
Finally, under $\nu_1$, every arm outside $\mathcal{S}_1$ has suboptimality gap at least $\Delta$. Therefore,
                                                                                                                                                                                                                    
\begin{align*}
    \operatorname{Regret}_{\nu_1,\mathsf{Alg}}(T)
    \geq  \Delta   \mathbb{E}_1[T-N_{\mathcal{S}_1}(T)] \geq  \frac{\Delta T}{2}
    =  \min\bigg\{\frac{T}{8}, \textcolor{red}{ \frac{(K-A)T}{32BR_A} } \bigg\}.
\end{align*}
Recall the condition that $T \geq KA(16B^2\Gamma_{K,T}^2)^{-1}$, we have that $T/8 \geq (K-A)T/(32BR_A)$, leading to 
\begin{align*}

\end{align*}
Since $\nu_1$ has exactly $B$ optimal arms, this proves the theorem.

\label{lem:aggregation-square-root-tail}
Let $X_1,\ldots,X_q$ be nonnegative integer-valued random variables, not necessarily independent. Suppose that, for some $\delta,c\geq 0$, each $X_i$ satisfies
\begin{align*}
    \mathbb{P}[X_i \geq s]
    \leq
    \delta + \frac{c}{\sqrt{s}}
\end{align*}
for every $i\in[q]$ and every integer $s\geq 1$. Then, for any integer $m\geq 1$,
\begin{align*}
    \mathbb{P}\bigg[\sum_{i=1}^q X_i > m\bigg]
    \leq
    2q\delta + \frac{3qc}{\sqrt{m}}.
\end{align*}
<!-- end lemma-4 -->
<!-- proof-9: body_md -->
For each $i\in[q]$, define the truncated random variable $\widetilde{X}_i := X_i \wedge m$. We decompose the event $\{\sum_{i=1}^q X_i > m\}$ according to whether any individual random variable exceeds the truncation level $m$. In particular,
\begin{align*}
    \bigg\{\sum_{i=1}^q X_i > m\bigg\}
    \subseteq
    \bigg\{\max_{i\in[q]}X_i > m\bigg\}
    \cup
    \bigg\{\sum_{i=1}^q \widetilde{X}_i > m\bigg\}.
\end{align*}
For the first event, by a union bound and the assumed tail bound,
\begin{align*}
    \mathbb{P}\bigg[\max_{i\in[q]}X_i > m\bigg] \leq
    \sum_{i=1}^q \mathbb{P}[X_i > m] \leq
    q\delta + \frac{qc}{\sqrt{m}}.
\end{align*}
For the second event, since $X_i$ is nonnegative and integer-valued, the tail-sum formula gives
\begin{align*}
    \mathbb{E}[\widetilde{X}_i] = \sum_{s=1}^m \mathbb{P}[X_i\geq s] \leq  m\delta + c\sum_{s=1}^m\frac{1}{\sqrt{s}} \leq  m\delta + 2c\sqrt{m},
\end{align*}
where the last inequality follows from $\sum_{s=1}^m 1/\sqrt{s} \leq 2\sqrt{m}$. Therefore, by Markov's inequality,
\begin{align*}
    \mathbb{P}\bigg[\sum_{i=1}^q \widetilde{X}_i > m\bigg] \leq  \frac{1}{m}  \sum_{i=1}^q \mathbb{E} [\widetilde{X}_i] \leq q\delta + \frac{2qc}{\sqrt{m}}.
\end{align*}
Combining the two bounds yields
\begin{align*}
    \mathbb{P}\bigg[\sum_{i=1}^q X_i > m\bigg]
    \leq
    2q\delta + \frac{3qc}{\sqrt{m}},
\end{align*}
which finishes the proof.

    \end{cases}
\end{align*}
Thus, $\nu$ has exactly $B$ optimal arms, namely the arms in $\mathcal{U}$. Recall that $\theta$ denotes the all-equal reference instance with $r_\theta(a)=1/2$ for every $a\in\mathcal{A}$. We use $\mathbb{P}_\nu$ and $\mathbb{P}_\theta$ to denote the trajectory distributions induced by $\mathsf{Alg}$ under $\nu$ and $\theta$, respectively.

We first show that, under the reference instance $\theta$, the algorithm samples $\mathcal{U}$ only $\widetilde{O}(q^2R_A^2)$ times with constant probability, where
$q:=\lceil B/(K-A)\rceil$. Partition $\mathcal{U}$ into $q$ disjoint subsets
$\mathcal{U}_1,\ldots,\mathcal{U}_q$ such that $|\mathcal{U}_i|\leq K-A$ for every $i\in[q]$. For each $i\in[q]$, extend $\mathcal{U}_i$ to a subset $\mathcal{S}_i\subseteq\mathcal{A}$ satisfying $\mathcal{U}_i\subseteq\mathcal{S}_i$ and $|\mathcal{S}_i|=K-A$. Then
\begin{align*}
    N_{\mathcal{U}}(T)
    =
    \sum_{i=1}^q N_{\mathcal{U}_i}(T)
    \leq
    \sum_{i=1}^q N_{\mathcal{S}_i}(T).
\end{align*}
Set $\ell:=\log(16q)$. Then, for every $s\in[T]$ and every $i\in[q]$, applying Lemma~\ref{lem:pull-count-tail} gives
\begin{align*}
    \mathbb{P}_\theta
    \big[N_{\mathcal{S}_i}(T)\geq s\big]
    \leq
    \frac{1}{16q}
    +
    8R_A\sqrt{\frac{\ell}{s}}.
\end{align*}
Since $N_{\mathcal{S}_i}(T)\leq T$, the same inequality holds trivially for $s>T$. Therefore, applying Lemma~\ref{lem:aggregation-square-root-tail} to
$X_i:=N_{\mathcal{S}_i}(T)$ with $\delta=1/(16q)$ and
$c=8R_A\sqrt{\ell}$ gives, for every $m\geq1$,
\begin{align*}
    \mathbb{P}_\theta
    \bigg[
        \sum_{i=1}^qN_{\mathcal{S}_i}(T)>m
    \bigg]
    \leq
    \frac18
    +
    24qR_A\sqrt{\frac{\ell}{m}}.
\end{align*}

Now we choose $m=\lceil 4096q^2R_A^2\ell \rceil$. Then $24qR_A\sqrt{\ell/m}\leq3/8$, which leads to 
\begin{align*}
    \mathbb{P}_\theta \bigg[ \sum_{i=1}^qN_{\mathcal{S}_i}(T)>m \bigg] \leq  \frac12.
\end{align*}
Since $N_{\mathcal{U}}(T)\leq\sum_{i=1}^qN_{\mathcal{S}_i}(T)$, we conclude that
\begin{align}
    \mathbb{P}_\theta
    \big[N_{\mathcal{U}}(T)\leq m\big]
    \geq
    \frac12.
    \label{eq:dense-to-sparse-typical-pulls}
\end{align}
We now show that $T\geq2m$. Since $B\geq K-A$, we have $q = \lceil B/(K-A) \rceil \leq 2B/(K-A)$. Consequently, $\ell \leq \log(32B/(K-A))$. Using the definition of $R_A$,
                                                                                                                                                                                                                        
\begin{align*}
    4096q^2R_A^2\ell & \leq 16384 \frac{B^2}{(K-A)^2}  \Gamma_{K,T}^2 \frac{(K-A)^2T}{K^2} \log\bigg(\frac{32B}{K-A}\bigg) \\
    & = 16384\frac{       B^2\Gamma_{K,T}^2 \log\big(32B/(K-A)\big) }{K^2} T.
\end{align*}
By the assumed condition
\begin{align*}
    B\Gamma_{K,T}\sqrt{ \log\bigg(\frac{32B}{K-A}\bigg)} \leq \frac{K}{256},
\end{align*}
the right-hand side is at most $T/4$. Therefore,
\begin{align*}
    m =\lceil 4096q^2R_A^2\ell \rceil \leq 4096q^2R_A^2\ell + 1 \leq \frac{T}{4}+1 \leq     \frac{T}{2},
\end{align*}
where the last inequality follows from $T\geq4$.

We now transfer~\eqref{eq:dense-to-sparse-typical-pulls} from the all-equal reference instance $\theta$ to the hard instance $\nu$. Once again, we consider the stopping time
\begin{align*}
    \mathcal{E} := \big\{N_{\mathcal{U}}(T)\leq m\big\}, \qquad \tau:=     \inf\big\{t\in[T]:N_{\mathcal{U}}(t)=m+1\big\}
    \wedge T.
\end{align*}
so that $\mathbb{P}_\theta[\mathcal{E}]\geq1/2$. Notice that $\mathcal{E}\in\mathcal{F}_\tau$ and $N_{\mathcal{U}}(\tau)\leq m+1$ almost surely. Since $\theta$ and $\nu$ differ only on the arms in $\mathcal{U}$, where the reward distribution is $\mathcal{N}(1/2,1)$ under $\theta$ and $\mathcal{N}(1/2+\Delta,1)$ under $\nu$, the KL decomposition for the stopped trajectory gives
\begin{align*}
    \operatorname{KL}\big(
        \mathbb{P}_\theta|_{\mathcal{F}_\tau}
        \|
        \mathbb{P}_\nu|_{\mathcal{F}_\tau}
    \big)
    =
    \frac{\Delta^2}{2}
    \mathbb{E}_\theta[N_{\mathcal{U}}(\tau)]
    \leq
    \frac{\Delta^2(m+1)}{2}.
\end{align*}
Now we choose $\Delta:=1/(4\sqrt{m+1})$. Then we see that $\ensuremath{{\mathsf{KL}}\left(\mathbb{P}_{\theta}|_{\mathcal{F}_\tau}\|\mathbb{P}_{\nu}|_{\mathcal{F}_\tau}\right)} \leq 1/32$. By Pinsker's inequality (Lemma~\ref{lem:pinsker}), $\operatorname{TV(\mathbb{P}_{\theta|}|_{\mathcal{F}_\tau}}, \mathbb{P}_{\nu}|_{\mathcal{F}_\tau}) \leq 1/8$. Since $\mathcal{E}\in\mathcal{F}_\tau$, it follows from~\eqref{eq:dense-to-sparse-typical-pulls} that
\begin{align*}
    \mathbb{P}_\nu[\mathcal{E}] \geq \mathbb{P}_\theta[\mathcal{E}] -    \operatorname{TV}\big(\mathbb{P}_\theta|_{\mathcal{F}_\tau},      \mathbb{P}_\nu|_{\mathcal{F}_\tau} \big) \geq \frac{3}{8}.
\end{align*}

Recall that the arms in $\mathcal{U}$ are the unique optimal arms under $\nu$, while every arm in $\mathcal{U}^{\complement}$ has suboptimality gap $\Delta$. Therefore,
\begin{align*}
    \mathrm{Regret}_{\nu,\mathsf{Alg}}(T) =  \Delta \mathbb{E}_\nu
    \big[
        T-N_{\mathcal{U}}(T)
    \big] \geq   \Delta(T-m) \mathbb{P}_\nu[\mathcal{E}] \geq   \frac{3\Delta T}{16} =     \frac{3T}{64\sqrt{m+1}},
\end{align*}
where the last inequality holds due to $m\leq T/2$.

It remains to simplify this lower bound. Define $\bar{\ell} = \log(32B/(K-A))$. Since $B/(K-A)\geq1$, we have $\ell=\log(16q)\geq\log(16B/(K-A))\geq(4/5)\bar{\ell}$. Together with the assumed horizon condition $T \geq K^2/(B^2\Gamma_{K,T}^2\bar{\ell})$ gives
\begin{align*}
    q^2R_A^2\ell \geq \frac{B^2}{(K-A)^2} \Gamma_{K,T}^2 \frac{(K-A)^2T}{K^2} \ell \geq \frac{4}{5} \frac{B^2\Gamma_{K,T}^2T\bar{\ell}}{K^2}  \geq\frac45.
\end{align*}
Consequently, $m + 1 \leq 4096 q^2R_A^2\ell +2 \leq 8192 q^2R_A^2\ell$. Therefore,
\begin{align*}
    \mathrm{Regret}_{\nu,\mathsf{Alg}}(T) \geq \frac{3T}{64\sqrt{m+1}} \geq     \frac{3}{8192\sqrt{2}} \frac{\sqrt{KA}}{B\Gamma_{K,T}\sqrt{\log(32B/(K-A))}}\sqrt{T},
\end{align*}
where the second inequality uses $q\leq2B/(K-A)$, $K \geq A$ and $\ell\leq\log(32B/(K-A))$. This finishes the proof.

<!-- theorem-6: body_md -->
\label{thm:high-prob-lower}
Suppose $A>K/2$, so that $(K-A)/K < 1/2$, and fix $\Gamma\ge1$. Define $\ell = \log(16K/(K-A))$ and $c_0 = 13/128$. Suppose that $\mathtt{Alg}$ is a strategy satisfying for any problem instance in ${\Xi}_{K, A}$,
\begin{align*}
    \mathrm{Regret}(T)\le\Gamma\frac{K-A}{K}\sqrt T,
\end{align*}
that is, a policy that achieves the minimax-optimal expected regret up to a
factor of $\Gamma$.
If $T\ge {8\Gamma^2\ell}/{c_0^2}$, then there exists a problem instance $\mu\in{\Xi}_{K, A}$ such that
\begin{align}
    \mathbb{P}_{\mu,\mathtt{Alg}}\bigg(R(T) \ge \frac{c_0}{8\sqrt2} \frac{\sqrt T}{\Gamma\ell} \bigg)  \ge\frac{K-A}{32K}. \label{eq:high-prob-lower-main}
\end{align}

<!-- remark-5: body_md -->
It is widely known that UCB-typed algorithm, while achieving a near-minimax-optimal rate, simultaneously achieves a $\widetilde{O}(\sqrt{KT}\text{polylog}(\delta^{-1}))$ high-probability upper bound. Theorem~\ref{thm:high-prob-lower} shows that, the analogue rate, $\widetilde{O}\big((K-A)K^{-1} \sqrt{T} \cdot \mathrm{polylog}(\delta^{-1})\big)$ is impossible when $A = K - o(K)$. In particular, making $\widetilde{O}\big((K-A)K^{-1} \sqrt{T} \cdot \mathrm{polylog}(\delta^{-1})\big) = \widetilde{O}(\sqrt{T})$ requires $\delta \sim \exp(-K/(K-A)) \ll (K-A)/K$, which contradicts~\eqref{eq:high-prob-lower-main}.

Recall that $\ell = \log(16K/(K-A))$ and $c_0 = 13/128$. Since $T\geq 8\Gamma^2\ell/c_0^2$, we can choose an integer $m$ satisfying
\begin{align}
    \frac{c_0^2T}{8\Gamma^2\ell}
    \leq m
    \leq
    \frac{c_0^2T}{4\Gamma^2\ell}.
    \label{eq:high-prob-lower-choice-m}
\end{align}
Let $\theta$ be the instance such that $\bm{\theta}=\mathbf{1}/2\in\mathbb{R}^K$ denote the all-equal auxiliary instance. We first apply Lemma~\ref{lem:pull-count-tail} to characterize the exploration behavior of $\mathtt{Alg}$ under $\theta$. By the lower bound in Lemma~\ref{lem:pull-count-tail},
\begin{align*}
    \max_{{\mathcal{S}}\subseteq[K]:\,|{\mathcal{S}}| = K-A} \mathbb{P}_{\theta}\big[N_{{\mathcal{S}}}(T) \geq m\big] & \geq \frac{K-A}{K} - \exp(-\ell) - 8\Gamma\frac{K-A}{K}\sqrt{\frac{m\ell}{T}} \\
    & \geq  \frac{K-A}{K} - \frac{K-A}{16K} - \frac{13(K-A)}{32K} \\
    &\geq \frac{K-A}{2K},
\end{align*}
where the second inequality follows from $\exp(-\ell)=(K-A)/(16K)$ and the upper bound on $m$ in~\eqref{eq:high-prob-lower-choice-m}. Therefore, there exists a subset ${\mathcal{S}}\subseteq[K]$ with $|{\mathcal{S}}|=K-A$ such that
\begin{align}
    \mathbb{P}_{\theta}\big[N_{{\mathcal{S}}}(T)\geq m\big]
    \geq
    \frac{K-A}{2K}.
    \label{eq:high-prob-lower-reference-tail}
\end{align}

We now construct the hard instance $\mu$ by decreasing the mean rewards of the arms in ${\mathcal{S}}$ by $\Delta$, with $\Delta = (4\sqrt{m\ell})^{-1}$, namely
\begin{align*}
    \mu_a :=  \begin{cases}
        \frac{1}{2} - \Delta, & a\in{\mathcal{S}},\\
        \frac{1}{2}, & a\in{\mathcal{S}}^\complement.
    \end{cases}
\end{align*}
Since $|{\mathcal{S}}|=K-A$, the instance $\mu$ has exactly $A$ optimal arms and hence belongs to ${\Xi}_{K, A}$.
                                                   
Now we transfer~\eqref{eq:high-prob-lower-reference-tail} from the auxiliary instance $\theta$ to $\mu$. As usual, we prove that at $\tau$, the probability distributions induced by $\theta$ and $\mu$ are still close, therefore leading to that $\mathbb{P}_{\mu} [H] \gtrsim (K-A)/K$, where $H$ and $\tau$ are defined by
\begin{align*}
    H:=\big\{N_{{\mathcal{S}}}(T)\geq m\big\},
    \qquad
    \tau:=\inf\big\{t\in[T]:N_{{\mathcal{S}}}(t)=m\big\}\wedge T.
\end{align*}
In particular, consider the
trajectory distributions restricted to $\mathcal{F}_{\tau}$ by $\mathbb{P}_\theta^{\tau} := \mathbb{P}_\theta|_{\mathcal{F}_{\tau}}$ and $\mathbb{P}_{\mu}^{\tau} := \mathbb{P}_{\mu}|_{\mathcal{F}_{\tau}}$. Applying Proposition~\ref{prop:gaussian-bandit-likelihood-ratio} gives that
\begin{align*}
    \log \frac{{\,\mathrm{d}}\mathbb{P}_\theta|_{\mathcal{F}_{\tau}}}{{\,\mathrm{d}}\mathbb{P}_\mu|_{\mathcal{F}_{\tau}}} & = \sum_{t=1}^{\tau} \bigg[ (\theta_{a_t}-\mu_{a_t}) (r_t-\theta_{a_t}) + \frac{1}{2}(\theta_{a_t}-\mu_{a_t})^2\bigg] \\
    & = \Delta \sum_{t=1}^\tau \operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\}(r_t-1/2) +
    \frac{\Delta^2}{2}
    N_{{\mathcal{S}}}(\tau).
\end{align*}
Now we denote $M_t = \Delta \sum_{s=1}^t \operatorname{\mathds{1}}\{a_s\in{\mathcal{S}}\}(r_s-1/2)$ and $V_t = \Delta^2 N_{{\mathcal{S}}}(t)$, then the probability distribution $\mathbb{P}_\theta$, we see that for any $\lambda \in \mathbb{R}$,
                
                                                                                                       
                                                                                                                                                                                                  
              
\begin{align*}
    & \mathbb{E}_{\theta}\bigg[\exp\bigg(\lambda M_t-\frac{\lambda^2V_t}{2} \bigg) \bigg| \mathcal{F}_{t-1}\bigg] \\
    & \quad = \exp\bigg(\lambda M_{t-1}-\frac{\lambda^2V_{t-1}}{2} \bigg) \mathbb{E}_{\theta}\bigg[\underbrace{\exp\bigg(\lambda (r_t-1/2) \Delta\operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\}-\frac{\lambda^2\Delta^2 \operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\}}{2} \bigg)}_{(\star)} \bigg| \mathcal{F}_{t-1}\bigg] 
\end{align*}
For $(\star)$, we first take expectation on $\mathcal{G}_t$, which gives that 
\begin{align*}
    \mathbb{E}_{\theta}[(\star) | \mathcal{G}_t] &  =1 - \operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\} +  \operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\} \mathbb{E}_{\theta}\bigg[\exp\bigg(\lambda (r_t-1/2) \Delta -\frac{\lambda^2\Delta^2 }{2} \bigg) \bigg| \mathcal{G}_t \bigg] \\
    & \leq 1 - \operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\} + \operatorname{\mathds{1}}\{a_t\in{\mathcal{S}}\} \\
    & = 1,
\end{align*}
where the inequality holds due to that $r_t-1/2$ is $1$-sub-gaussian conditioned on $\mathcal{G}_t$ on $\mathbb{P}_\theta$. This immediately gives $\mathbb{E}_{\theta}[(\star) | \mathcal{F}_{t-1}] \leq 1$, verifying that $\exp(\lambda M_t - \lambda^2 V_t /2)$ is a super-martingale for any $\lambda$. Now invoking Lemma~\ref{lem:sub-gaussian-super-martingale} with $x = \Delta\sqrt{2m\ell}$, since $V_\tau \leq m\Delta^2$, we have
\begin{align*}
    \mathbb{P}_{\theta}\big(M_\tau \ge \Delta\sqrt{2m\ell}\big) \leq \exp\bigg(-\frac{2m\Delta^2\ell}{2m\Delta^2}\bigg) = \exp(-\ell).
\end{align*}
Since $\Delta^2 N_{{\mathcal{S}}}(\tau_{{\mathcal{S}}}) /2 \leq m\Delta^2/2$, this immediately leads to
\begin{align*}
    \mathbb{P}_{\theta} \bigg(\log \frac{{\,\mathrm{d}}\mathbb{P}_\theta^{\tau}}{{\,\mathrm{d}}\mathbb{P}_{\mu}^{\tau}} \geq \Delta\sqrt{2m\ell} + m\Delta^2/2 \bigg) \leq \exp(-\ell).
\end{align*}
Recall that $\Delta = (4\sqrt{m\ell})^{-1}$, this means that $\Delta\sqrt{2m\ell} + m\Delta^2/2 = 1/2\sqrt{2} + 1/32\ell < \log 2$ since $\ell \geq 1$. Therefore,
                                        
\begin{align*}
    \mathbb{P} \bigg(\log \frac{{\,\mathrm{d}}\mathbb{P}_\theta^{\tau}}{{\,\mathrm{d}}\mathbb{P}_{\mu}^{\tau}} > \log 2 \bigg) \leq \exp(-\ell).
\end{align*}
Now we apply Proposition~\ref{prop:change-of-measure} on the measurable space $(\Omega,\mathcal{F}_{\tau_{{\mathcal{S}}}})$ with $\mathbb{P}= \mathbb{P}_\theta^{\tau}$, $\mathbb{Q} = \mathbb{P}_{\mu}^{\tau}$, $X = \operatorname{\mathds{1}}\{H\}$, $b=1$ and $\gamma = \log 2$ and obtain that
\begin{align*}
    \mathbb{P}_{\mu}(H) = \mathbb{P}_{\mu}^{\tau}(H) \geq \frac{1}{2} \bigg[\mathbb{P}_\theta^{\tau}(H) - \mathbb{P} \bigg(\log \frac{{\,\mathrm{d}}\mathbb{P}_\theta^{\tau}}{{\,\mathrm{d}}\mathbb{P}_{\mu}^{\tau}} > \log 2 \bigg) \bigg] \geq \frac{1}{2} \big[ \mathbb{P}_\theta(H)  -\exp(-\ell) \big],
\end{align*}
where the equation holds due to that $H \in \mathcal{F}_{\tau}$. 

Finally, every pull of an arm in ${\mathcal{S}}$ incurs regret $\Delta$ under $\mu$. Hence, on the event $H$,
\begin{align*}
    R(T)
    \geq
    \Delta m
    =
    \frac{\sqrt{m}}{4\sqrt{\ell}}.
\end{align*}
Using the lower bound on $m$ in~\eqref{eq:high-prob-lower-choice-m}, we obtain
\begin{align*}
    \Delta m
    \geq
    \frac{1}{4\sqrt{\ell}}
    \sqrt{\frac{c_0^2T}{8\Gamma^2\ell}}
    =
    \frac{c_0}{8\sqrt{2}}
    \frac{\sqrt{T}}{\Gamma\ell}.
\end{align*}
Therefore,
\begin{align*}
    \mathbb{P}_{\mu}\bigg(
        R(T)
        \geq
        \frac{c_0}{8\sqrt{2}}
        \frac{\sqrt{T}}{\Gamma\ell}
    \bigg)
    \geq
    \frac{K-A}{32K},
\end{align*}
which proves the desired result.

<!-- lemma-5: body_md -->
\label{lem:sub-gaussian-super-martingale}
Let $(M_t,V_t)_{t=0}^T$ be an adapted process with
$M_0=V_0=0$ and $(V_t)_{t=0}^T$ nondecreasing. Suppose that, for
every $s>0$, $\exp(sM_t - s^2V_t/2)$ is a nonnegative supermartingale. Then, for any stopping time  $\tau\leq T$ and every $x,v>0$,
\begin{align}
    \mathbb{P}\big(
        M_\tau\geq x,\,
        V_\tau\leq v
    \big)
    \leq
    \exp\left(
        -\frac{x^2}{2v}
    \right).
    \label{eq:stopped-sub-gaussian-martingale}
\end{align}
In particular, if $V_\tau\leq v$ almost surely, then
\begin{align*}
    \mathbb{P}(M_\tau\geq x)
    \leq
    \exp\left(
        -\frac{x^2}{2v}
    \right).
\end{align*}

<!-- lemma-6: body_md -->
\label{lem:tv-change-of-measure}
Let $(\Omega,\mathcal{F})$ be a measurable space, and let
$\mathbb{P}$ and $\mathbb{Q}$ be probability measures on it. Then, for
every $\mathcal{F}$-measurable function $f:\Omega\to[0,T]$, 
\begin{align*}
    \big| \mathbb{E}_{\mathbb{P}}[f] - \mathbb{E}_{\mathbb{Q}}[f] \big| \leq  T\operatorname{TV}(\mathbb{P},\mathbb{Q}),
\end{align*}
where $\operatorname{TV}(\mathbb{P},\mathbb{Q}) :=  \sup_{E\in\mathcal{F}} \big| \mathbb{P}(E)-\mathbb{Q}(E) \big|$.
