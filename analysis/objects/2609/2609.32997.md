<!-- definition-1: body_md -->
A \emph{Riemannian manifold} $(\mathcal{M},\langle \cdot,\, \cdot \rangle)$ is a smooth
		manifold $\mathcal{M}$ equipped with a smoothly varying family of inner products
		$\langle \cdot,\, \cdot \rangle_x$ on the tangent spaces $T_x\mathcal{M}$.
		The Riemannian norm is $\left\lVertv\right\rVert_x=\langle v,v\rangle_{x}^{1/2}$.
		The \emph{Riemannian distance} between $p,q\in\mathcal{M}$ is
		\[
		d(p,q) \;=\; \inf\bigl\{\ell(\gamma) :\, \gamma \text{ smooth curve from }
		p \text{ to } q\bigr\},
		\]
		where $\ell(\gamma)=\int_a^b\left\lVert\dot\gamma(t)\right\rVert_{\gamma(t)}\,dt$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
The \emph{Levi-Civita connection} $\nabla$ is the unique torsion-free,
		metric-compatible connection on $(\mathcal{M},\langle \cdot,\, \cdot \rangle)$.
		For a curve $\gamma:[a,b]\to\mathcal{M}$ and $v\in T_{\gamma(a)}\mathcal{M}$, the
		\emph{parallel transport} $\mathcal{P}_{\gamma,a,t}: T_{\gamma(a)}\mathcal{M}\to
		T_{\gamma(t)}\mathcal{M}$ is the unique linear isometry satisfying
		$\nabla_{\dot\gamma(t)} V(t)=0$, $V(a)=v$.
		A smooth curve $\gamma$ satisfying $\nabla_{\dot\gamma}\dot\gamma=0$ is
		called a \emph{geodesic}.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
For $x\in\mathcal{M}$ and $v\in T_x\mathcal{M}$, the \emph{exponential map}
		$\mathrm{exp}_x(v)\in\mathcal{M}$ is defined as $\gamma(1)$, where $\gamma$ is the
		unique geodesic with $\gamma(0)=x$ and $\dot\gamma(0)=v$.
		$\mathcal{M}$ is \emph{geodesically complete} if $\mathrm{exp}_x$ is defined on all of
		$T_x\mathcal{M}$.
		The \emph{logarithmic map} $\mathrm{log}_x:\mathcal{M}\to T_x\mathcal{M}$ is the local inverse of
		$\mathrm{exp}_x$, satisfying $\mathrm{exp}_x(\mathrm{log}_x(y))=y$ and
		$\left\lVert\mathrm{log}_x(y)\right\rVert_x = d(x,y)$.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
The \emph{Riemannian gradient} of $f:\mathcal{M}\to\mathbb{R}$ at $x$ is the unique
		vector $\operatorname{grad} f(x)\in T_x\mathcal{M}$ satisfying
		$\langle \operatorname{grad} f(x),\, v \rangle_x = df_x(v)$ for all $v\in T_x\mathcal{M}$.
		For embedded submanifolds of $\mathbb{R}^N$ with the induced metric,
		$\operatorname{grad} f(x)=\mathrm{Proj}_{T_x\mathcal{M}}(\nabla f(x))$.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
\label{def:lipschitz}
		The gradient field $\operatorname{grad} f$ is \emph{$L$-Lipschitz continuous} if for any
		$x,y\in\mathcal{M}$ joined by a minimising geodesic $\gamma$:
		\[
		\left\lVert\mathcal{P}_{\gamma,0,1}\operatorname{grad} f(x) - \operatorname{grad} f(y)\right\rVert_y
		\;\leq\; L\,d(x,y).
\]
<!-- end definition-5 -->
<!-- lemma-1: body_md -->
\label{lem:descent-lemma}
	Let $f:\mathcal{M}\to\mathbb{R}$ have $L$-Lipschitz Riemannian gradient.
	Then for every $x\in\mathcal{M}$ and $v\in T_x\mathcal{M}$,
	\begin{equation}
		f\bigl(\mathrm{exp}_x(v)\bigr)
		\;\leq\;
		f(x) + \langle \operatorname{grad} f(x),\, v \rangle_x + \frac{L}{2}\left\lVertv\right\rVert_x^2.
		\label{eq:descent-lemma}
	\end{equation}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:young}
		For all $a,b>0$ and $\tau>0$,
		\begin{equation}
			ab \;\leq\; \frac{\tau}{2}a^2 + \frac{1}{2\tau}b^2,
			\qquad\text{in particular } ab\leq\tfrac12(a^2+b^2).
			\label{eq:young-scalar}
		\end{equation}
		Equivalently, for all $a,b\in\mathbb{R}$ and $\theta>0$,
		\begin{equation}
			(a+b)^2 \;\leq\; (1+\theta)a^2 + \Bigl(1+\frac1\theta\Bigr)b^2.
			\label{eq:young-square}
		\end{equation}
<!-- end lemma-2 -->
<!-- assumption-1: body_md -->
\label{ass:lipschitz}
		The Riemannian gradient $\operatorname{grad} f$ is $L$-Lipschitz continuous on $\mathcal{M}$
		(Definition~\ref{def:lipschitz}).
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:lower}
		There exists $f_*\in\mathbb{R}$ such that $f(x)\geq f_*$ for all $x\in\mathcal{M}$.
<!-- end assumption-2 -->
<!-- remark-1: body_md -->
\begin{enumerate}[label=(\roman*),leftmargin=2em,itemsep=3pt]
\item \textbf{Adaptive inertial.} The coefficient $\bar\mu_n=\min\{\mu,\mu d(x_n,x_{n-1})\}$ automatically damps the extrapolation as consecutive iterates approach one another. In particular, $d(w_n,x_n)=\tilde\mu_n d(x_n,x_{n-1})\leq \mu d(x_n,x_{n-1})^2$, a second-order displacement estimate used later in the convergence analysis.
\item \textbf{Gradient-based restart.} The sign of $\langle \operatorname{grad} f(x_n),\, d_n \rangle_{x_n}$ tests whether the inherited momentum is locally aligned with ascent. If it is positive, RIAG-R sets $\tilde\mu_n=0$, hence $w_n=x_n$. This is the key safeguard that removes the first-order uphill contribution of the inertial step from the descent estimate.
\item \textbf{Adaptive gradient step.} Since $G_n$ is nondecreasing, $\alpha_n=\eta/(\sqrt{G_n}+\varepsilon)$ is positive and nonincreasing, with $\alpha_n\leq\eta/\varepsilon$. The algorithm therefore adapts the scalar learning rate using observed gradient magnitudes rather than an explicitly supplied Lipschitz constant.
\end{enumerate}
<!-- end remark-1 -->
<!-- lemma-3: body_md -->
\label{lem:descent}
			Let Assumptions~\ref{ass:lipschitz}--\ref{ass:lower} hold.
			Let $\{x_n\}$ be generated by Algorithm~\ref{alg:RIAG-R}.
			Then for every $n\geq 1$,
			\begin{equation}
				f(x_{n+1})
				\;\leq\;
				f(x_n)
				- \alpha_n\!\left(1-\frac{L\alpha_n}{2}\right)\left\lVert\operatorname{grad} f(w_n)\right\rVert^2
				+ \frac{L\tilde\mu_n^2}{2}\,d(x_n,x_{n-1})^2.
				\label{eq:descent}
			\end{equation}
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
Applying Lemma~\ref{lem:descent-lemma} at $w_n$ with $v=-\alpha_n\operatorname{grad} f(w_n)$, then we have
		\begin{align}
			f(x_{n+1})
			&= f\bigl(\mathrm{exp}_{w_n}(-\alpha_n\operatorname{grad} f(w_n))\bigr) \notag\\
			&\leq f(w_n)
			+ \langle \operatorname{grad} f(w_n),\, -\alpha_n\operatorname{grad} f(w_n) \rangle
			+ \frac{L}{2}\alpha_n^2\left\lVert\operatorname{grad} f(w_n)\right\rVert^2 \notag\\
			&= f(w_n) - \alpha_n\!\left(1-\frac{L\alpha_n}{2}\right)\left\lVert\operatorname{grad} f(w_n)\right\rVert^2.
			\label{eq:gradient-step}
		\end{align}
		When $\tilde\mu_n>0$, the restart check ensures
		$\langle \operatorname{grad} f(x_n),\, d_n \rangle\leq 0$ (otherwise restart would have
		set $\tilde\mu_n=0$).
		Applying Lemma~\ref{lem:descent-lemma} at $x_n$ with $v=\tilde\mu_n\,d_n$, we get
		\begin{align}
			f(w_n)
			&= f\bigl(\mathrm{exp}_{x_n}(\tilde\mu_n\,d_n)\bigr) \notag\\
			&\leq f(x_n)
			+ \tilde\mu_n\langle \operatorname{grad} f(x_n),\, d_n \rangle
			+ \frac{L\tilde\mu_n^2}{2}\left\lVertd_n\right\rVert^2 \notag\\
			&\leq f(x_n) + \frac{L\tilde\mu_n^2}{2}\,d(x_n,x_{n-1})^2,
			\label{eq:inertial-step}
		\end{align}
		where we used $\langle \operatorname{grad} f(x_n),\, d_n \rangle\leq0$ and
		$\left\lVertd_n\right\rVert=\left\lVert\mathrm{log}_{x_n}(x_{n-1})\right\rVert=d(x_n,x_{n-1})$.
		When $\tilde\mu_n=0$, $w_n=x_n$ so~\eqref{eq:inertial-step} holds
		trivially with equality $f(w_n)=f(x_n)$.
		                            
		Substituting~\eqref{eq:inertial-step} into~\eqref{eq:gradient-step} gives
		the stated inequality~\eqref{eq:descent}.

\label{cor:linear-rate}

<!-- proof-5: body_md -->
Write $\Delta_n:=f(x_n)-f_*$. By the assumed PL inequality,
		\begin{equation}
			\left\lVert\operatorname{grad} f(x)\right\rVert^2 \geq 2m\bigl(f(x)-f_*\bigr),
			\label{eq:PL}
		\end{equation}
		for all $x\in\mathcal{M}$.
		
		From Lemma~\ref{lem:descent} and $\alpha_n(1-L\alpha_n/2)\geq\alpha>0$:
		\begin{equation}
			\Delta_{n+1}
			\;\leq\; \Delta_n - \alpha\left\lVert\operatorname{grad} f(w_n)\right\rVert_{w_n}^2
			+ \frac{L\tilde\mu_n^2}{2}d(x_n,x_{n-1})^2.
			\label{eq:linear-step}
		\end{equation}
		We must lower-bound $\left\lVert\operatorname{grad} f(w_n)\right\rVert_{w_n}^2$ in terms of $\Delta_n$.
		The PL inequality~\eqref{eq:PL} is stated at $x_n$, not $w_n$, so we
		transfer it using the $L$-Lipschitz gradient field
		(Assumption~\ref{ass:lipschitz}, Definition~\ref{def:lipschitz}): writing
		$\mathcal P\coloneqq\mathcal P_{\gamma,0,1}$ for parallel transport along the
		geodesic from $x_n$ to $w_n$, Definition~\ref{def:lipschitz} gives
		$\left\lVert\mathcal P\operatorname{grad} f(x_n)-\operatorname{grad} f(w_n)\right\rVert_{w_n}\leq L\,d(x_n,w_n)
		=L\tilde\mu_n\,d(x_n,x_{n-1})$.
		Applying the squared Young inequality~\eqref{eq:young-square} (with
		$\theta=1$, since $\mathcal P$ is a linear isometry so
		$\left\lVert\mathcal P\operatorname{grad} f(x_n)\right\rVert_{w_n}=\left\lVert\operatorname{grad} f(x_n)\right\rVert_{x_n}$):
		\begin{align*}
			\left\lVert\operatorname{grad} f(x_n)\right\rVert_{x_n}^2
			&\;=\; \left\lVert\bigl(\mathcal P\operatorname{grad} f(x_n)-\operatorname{grad} f(w_n)\bigr)+\operatorname{grad} f(w_n)\right\rVert_{w_n}^2 \\
			&\;\leq\; 2\left\lVert\mathcal P\operatorname{grad} f(x_n)-\operatorname{grad} f(w_n)\right\rVert_{w_n}^2
			+ 2\left\lVert\operatorname{grad} f(w_n)\right\rVert_{w_n}^2,
		\end{align*}
		so, using the bound above, $\tilde\mu_n\leq\mu$, and the PL
		inequality~\eqref{eq:PL} at $x_n$,
		\begin{equation}
			\left\lVert\operatorname{grad} f(w_n)\right\rVert_{w_n}^2
			\;\geq\; \tfrac12\left\lVert\operatorname{grad} f(x_n)\right\rVert_{x_n}^2 - L^2\tilde\mu_n^2\,d(x_n,x_{n-1})^2
			\;\geq\; m\,\Delta_n - L^2\mu^2\,d(x_n,x_{n-1})^2.
			\label{eq:pl-transfer}
		\end{equation}
		Substituting~\eqref{eq:pl-transfer} into~\eqref{eq:linear-step}
		and using $\tilde\mu_n^2\leq\mu^2$, we have:
		\begin{align*}
			\Delta_{n+1}
			&\;\leq\; \Delta_n
			- \alpha\bigl(m\Delta_n - L^2\mu^2 d(x_n,x_{n-1})^2\bigr)
			+ \frac{L\mu^2}{2}d(x_n,x_{n-1})^2 \\
			&\;=\; (1-\alpha m)\Delta_n
			+ L\mu^2\Bigl(\alpha L+\tfrac12\Bigr) d(x_n,x_{n-1})^2.
		\end{align*}
		Let $\rho\coloneqq1-\alpha m\in[0,1)$ (using $\alpha m<1$) and
		$\gamma\coloneqq L\mu^2(\alpha L+\tfrac12)$. Unrolling the recursion and
		bounding the geometric weights $\rho^{n-1-k}\leq1$, we obtain:
		\[
		\Delta_n
		\;\leq\; \rho^{n-1}\Delta_1 + \gamma\sum_{k=1}^{n-1} d(x_k,x_{k-1})^2
		\;\leq\; \rho^{n-1}\Delta_1 + \gamma D.
		\]
		This gives~\eqref{eq:linear-rate}.
