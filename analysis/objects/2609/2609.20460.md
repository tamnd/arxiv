<!-- definition-1: body_md -->
\label{def:armf}
Given $d\in{\mathbb N},$ let
\begin{align*}
 \boldsymbol\alpha(p)=(\alpha_1(p),\ldots,\alpha_d(p))\in\mathbb T^d
\end{align*}
be independent and identically distributed random vectors as $p$ varies over
the primes, where $\mathbb T:=\{z\in{\mathbb C}:|z|=1\},$ and define the complete symmetric
polynomials $h_k$ by
\begin{align*}
 \prod_{j=1}^d(1-\alpha_jz)^{-1}
 =\sum_{k\geqslant0}h_k(\boldsymbol\alpha)z^k.
\end{align*}
We set $\mathbb X(p^k):=h_k(\boldsymbol\alpha(p))$ and extend $\mathbb X$
multiplicatively, with $\mathbb X(1)=1.$  We call $\mathbb X$ an \textit{automorphic
random multiplicative function of degree $d$}.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Given $d\in{\mathbb N},$ let $\mathbb X$ be an automorphic random multiplicative function of
degree $d$ satisfying
\begin{equation}\label{eq:conditions}
 {\mathbb E}[h_1(\boldsymbol\alpha)]=0,\qquad
 {\mathbb E}[|h_1(\boldsymbol\alpha)|^2]=1,\qquad
 {\mathbb E}[h_2(\boldsymbol\alpha)]=0.
\end{equation}
Then uniformly for all sufficiently large $x$ and $0\leqslant q\leqslant1,$ we have
\begin{align*}
 {\mathbb E}\left[\left|\sum_{n\leqslant x}\mathbb X(n)\right|^{2q}\right]
 \asymp_d
 \left(\frac{x}{1+(1-q)\sqrt{\log\log x}}\right)^q.
\end{align*}
The implied constants and the lower threshold for $x$ depend only on $d,$
uniformly over the local laws satisfying \eqref{eq:conditions}.
<!-- end theorem-1 -->
<!-- example-1: body_md -->
\label{ex:steinhaus}
Let $G=U(d)$, let $V={\mathbb C}^d$ be its standard representation, and, at every
prime, let $\boldsymbol\alpha(p)$ be the unordered spectrum of an independent
Haar-distributed matrix in $U(d)$.  The representation $V$ is irreducible,
of unitary type, and of dimension $d,$ and hence \eqref{eq:conditions} holds.
When $d=1,$ we have
$h_k(\alpha)=\alpha^k$ with $\alpha$ uniform on $\mathbb T,$ and
Theorem~\ref{thm:main} recovers \cite[Theorem~1]{HarperLow}.
<!-- end example-1 -->
<!-- example-2: body_md -->
\label{ex:odd-sym}
Let $G=SU(2)$, let $V=\operatorname{Sym}^m({\mathbb C}^2)$ for odd $m,$ and, at every
prime, let $\boldsymbol\alpha(p)$ be the unordered spectrum of an independent
Haar-distributed element of $SU(2)$ acting on $V$.  This spectrum is
\begin{align*}
 e^{i(m-2j)\theta_p}\qquad (0\leqslant j\leqslant m),
\end{align*}
where the Sato--Tate angles $\theta_p$ are independent with density
$(2/\pi)\sin^2\theta$ on $[0,\pi].$  The representation $V$ is irreducible,
of symplectic type, and of dimension $m+1,$ and hence
\eqref{eq:conditions} holds.  When $m=1,$ this gives the usual Sato--Tate
RMF, for which Theorem~\ref{thm:main} recovers
the bounds of Xu--Zheng
\cite[Theorem~1.8]{xu2026typicalsizeheckeeigenvalue}, with the upper bound
also obtained in \cite[Theorem~1.1]{hamdan2026lowmomentsheckeeigenvalue}.

\label{ex:higher-rank-st}
Let $G=SU(d)$ for $d\geqslant3,$ let $V={\mathbb C}^d$ be its standard representation,

an independent Haar-distributed matrix in $SU(d)$.  The representation $V$

<!-- lemma-1: body_md -->
\label{lem:covariance}
Let $m,n\in{\mathbb N}.$  There exists a multiplicative function $g:{\mathbb N}^2\to{\mathbb C},$ in
the sense that
\begin{align*}
 g(u_1u_2,v_1v_2)=g(u_1,v_1)g(u_2,v_2)
\end{align*}
whenever $(u_1v_1,u_2v_2)=1,$ and it satisfies
\begin{equation}\label{eq:covariance}
 {\mathbb E}[\mathbb X(m)\overline{\mathbb X(n)}]
 =\sum_{r\mid(m,n)}g(m/r,n/r).
\end{equation}
Also, for every $\sigma>1/3,$ we have
\begin{equation}\label{eq:kernel-norm}
 \sum_{u,v\geqslant1}\frac{|g(u,v)|}{(uv)^\sigma}\ll_{d,\sigma}1.
\end{equation}
Consequently, for every $0\leqslant a<b,$ we have
\begin{align}
 {\mathbb E}\left[\left|\sum_{a<n\leqslant b}\mathbb X(n)\right|^2\right]
 \ll_d b-a+b^{5/6}. \label{eq:short-second}
\end{align}
The same estimate holds when the summand is restricted to integers whose
prime factors belong to a fixed set of primes.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For integers $k,l \geqslant 0,$ put
\begin{align*}
 c(k,l):={\mathbb E}[h_k(\boldsymbol\alpha)
                 \overline{h_l(\boldsymbol\alpha)}],
\end{align*}
where $c(k,l)$ is understood to be zero if either index is negative, and
define
\begin{align*}
 g(p^k,p^l):=c(k,l)-c(k-1,l-1).
\end{align*}
Extending this definition multiplicatively and telescoping at each prime
proves \eqref{eq:covariance}.


The assumptions in \eqref{eq:conditions} give $g(1,1)=1$ and
\begin{align*}
 g(p^k,p^l)=0\qquad\text{whenever }1\leqslant k+l\leqslant2.
\end{align*}
Using the bound $|h_k(\boldsymbol\alpha)|\leqslant\binom{k+d-1}{d-1},$ we obtain
\begin{align*}
 \sum_{k,l\geqslant0}|g(p^k,p^l)|p^{-\sigma(k+l)}
 =1+O_{d,\sigma}(p^{-3\sigma}).
\end{align*}
The corresponding Euler product converges when $\sigma>1/3,$ which proves
\eqref{eq:kernel-norm} with constants uniform in the local law.


On substituting $m=ur$ and $n=vr$ in the second moment of an interval sum,
the number of admissible values of $r$ is at most
\begin{align*}
 \frac{b-a}{\max\{u,v\}}+1.
\end{align*}
The first term contributes $O_d(b-a)$ by \eqref{eq:kernel-norm} with
$\sigma=1/2.$  Since $u,v\leqslant b$ for the second term, we have
\begin{align*}
 \sum_{u,v\leqslant b}|g(u,v)|
 \leqslant b^{5/6}\sum_{u,v\geqslant1}\frac{|g(u,v)|}{(uv)^{5/12}}
 \ll_d b^{5/6},
\end{align*}
which proves \eqref{eq:short-second}. Restricting
to a set of primes removes terms from the same absolute majorants, which
proves the final assertion.

<!-- lemma-3: body_md -->
\label{lem:gaussian-replacement}
There exists a constant $c_d\geqslant1$ such that for every finite positive Borel
measure $\nu$ on ${\mathbb R},$ every $\sigma\geqslant1/2$ and every $0<q\leqslant1,$ the quantities
\begin{align*}
 \mathcal F_{\mathcal P}&:={\mathbb E}\left[\left(\int_{{\mathbb R}}
          |F_{\mathcal P}(\sigma+it)|^2\,d\nu(t)\right)^q \right],\\
 \mathcal G_{\mathcal P}&:=m_{\mathcal P,\sigma}^q \,
 {\mathbb E}\left[\left(\int_{{\mathbb R}}
 e^{H_{\mathcal P,\rho,\sigma}(t)
       -\frac{1}{2}{\mathbb E}[H_{\mathcal P,\rho,\sigma}(t)^2]}\,d\nu(t)\right)^q \right]
\end{align*}
satisfy
\begin{align*}
 c_d^{-q} \,\mathcal G_{\mathcal P}\leqslant\mathcal F_{\mathcal P}
 \leqslant c_d^q \,\mathcal G_{\mathcal P}.
\end{align*}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We may assume that $\nu({\mathbb R})>0$ and replace one prime at a time, conditioning
on all the remaining local random variables.  After absorbing their positive
local factors into $\nu,$ put $\mu:=\nu/\nu({\mathbb R}),$ since the normalization
contributes the same factor $\nu({\mathbb R})^q$ to both sides of the comparison.  Put
\begin{align*}
 \mathcal I_p:=\int_{{\mathbb R}}p^{-it}\,d\mu(t).
\end{align*}
The two local quantities to be compared are
\begin{align*}
 \mathcal F_p&:=\int_{{\mathbb R}}
 \left|\prod_{j=1}^d(1-\alpha_jp^{-\sigma-it})^{-1}\right|^2d\mu(t),\\
 \mathcal G_p&:=\int_{{\mathbb R}}
 \exp\left(2p^{-\sigma}\mathrm{Re}(g_pp^{-it})
       -p^{-2\sigma}\mathrm{Re}(\rho p^{-2it})\right)d\mu(t).
\end{align*}
Uniformly in $\mu,$ expansion of the local Euler factor gives
\begin{align*}
 \mathcal F_p
=1&+2p^{-\sigma}
       \mathrm{Re}\left(h_1(\boldsymbol\alpha)\mathcal I_p\right)\\
 &+p^{-2\sigma}\left\{|h_1(\boldsymbol\alpha)|^2
       +2\mathrm{Re}\left(h_2(\boldsymbol\alpha)
       \int_{{\mathbb R}}p^{-2it}\,d\mu(t)\right)\right\}
       +O_d(p^{-3\sigma}).
\end{align*}
Applying Taylor's theorem and the identities in \eqref{eq:conditions}, for sufficiently large $p,$ we obtain
\begin{align*}
 {\mathbb E}[\mathcal F_p^q]
={}&1+qp^{-2\sigma}
 +2q(q-1)p^{-2\sigma}
 {\mathbb E}[\left(\mathrm{Re}\left(
 h_1(\boldsymbol\alpha)\mathcal I_p\right)\right)^2]
 +O_d(qp^{-3\sigma}).
\end{align*}
The remainder here is uniform since
\begin{align*}
 |h_k(\boldsymbol\alpha)|
 \leqslant\binom{k+d-1}{d-1}.
\end{align*}


We now perform the corresponding calculation for $\mathcal G_p.$  Expanding
the exponential and integrating against $\mu,$ we have
\begin{align*}
 \mathcal G_p
=1+2p^{-\sigma}\mathrm{Re}(g_p\mathcal I_p)
 &+p^{-2\sigma}\int_{{\mathbb R}}
 \left\{2(\mathrm{Re}(g_pp^{-it}))^2
       -\mathrm{Re}(\rho p^{-2it})\right\}d\mu(t) \\
 &+O\bigl(p^{-3\sigma}(1+|g_p|)^3
       \exp(p^{-\sigma}(2|g_p|+1))\bigr).
\end{align*}
The Gaussian moment identities give
\begin{gather*}
 2\int_{{\mathbb R}}{\mathbb E}[(\mathrm{Re}(g_pp^{-it}))^2]\,d\mu(t)
 -\mathrm{Re}\left(\rho\int_{{\mathbb R}}p^{-2it}\,d\mu(t)\right)=1,\\
 {\mathbb E}[(\mathrm{Re}(g_p\mathcal I_p))^2]
 ={\mathbb E}[\left(\mathrm{Re}\left(
 h_1(\boldsymbol\alpha)\mathcal I_p\right)\right)^2].
\end{gather*}
Applying Taylor's theorem again and using the Gaussian exponential
moments, we obtain
\begin{align*}
 {\mathbb E}[\mathcal G_p^q]
={}&1+qp^{-2\sigma}
 +2q(q-1)p^{-2\sigma}
 {\mathbb E}[\left(\mathrm{Re}\left(
 h_1(\boldsymbol\alpha)\mathcal I_p\right)\right)^2]
 +O_d(qp^{-3\sigma})\\
={}&{\mathbb E}[\mathcal F_p^q]+O_d(qp^{-3\sigma}).
\end{align*}
For the remainder, let $\mathcal G_p(u)$ denote the defining integral
with $p^{-\sigma}$ replaced by $u.$  For $0\leqslant u\leqslant1/2$ and $1\leqslant k\leqslant3,$ differentiating the integrand gives
\begin{gather*}
 |\mathcal G_p^{(k)}(u)|\ll(1+|g_p|)^k\mathcal G_p(u),
 \qquad
 \mathcal G_p(u)^q\leqslant\exp(|g_p|+1),\\
 (\mathcal G_p^q)'''
 =q\mathcal G_p^{q-1}\mathcal G_p'''
 +3q(q-1)\mathcal G_p^{q-2}\mathcal G_p'\mathcal G_p''
 +q(q-1)(q-2)\mathcal G_p^{q-3}(\mathcal G_p')^3.
\end{gather*}
Consequently, we have
\begin{gather*}
 {\mathbb E}\left[\sup_{0\leqslant u\leqslant1/2}
 |(\mathcal G_p(u)^q)'''|\right]
 \ll q{\mathbb E}[(1+|g_p|)^3\exp(|g_p|+1)]\ll q,
\end{gather*}
which gives the stated remainder uniformly for $|\rho|\leqslant1.$  Since the common main term in the two expansions
is $1+O_d(qp^{-2\sigma}),$ it follows that, for every sufficiently large $p,$
we have
\begin{align}\label{eq:local-ratio}
 \exp(-c_dqp^{-3\sigma})
 \leqslant\frac{{\mathbb E}[\mathcal F_p^q]}
              {{\mathbb E}[\mathcal G_p^q]}
 \leqslant \exp(c_dqp^{-3\sigma}).
\end{align}
For the finitely many remaining primes, we have
\begin{align*}
 (1+p^{-\sigma})^{-2d}\leqslant\mathcal F_p
 \leqslant(1-p^{-\sigma})^{-2d},
\end{align*}
and
\begin{align*}
 \exp(-qp^{-2\sigma})\leqslant{\mathbb E}[\mathcal G_p^q]
 \leqslant \exp(qp^{-2\sigma}),
\end{align*}
where the second estimate follows from Jensen's inequality and the Gaussian exponential formula. After multiplying over these primes, their contribution
lies between $c_d^{-q}$ and $c_d^q.$


Finally, it follows from the identity
\begin{align*}
 \frac12 \, {\mathbb E}[H_{\mathcal P,\rho,\sigma}(t)^2]
 =\sum_{p\in\mathcal P}p^{-2\sigma}
  +\mathrm{Re}\left(\rho\sum_{p\in\mathcal P}p^{-2\sigma-2it}\right)
\end{align*}
that
\begin{align*}
 m_{\mathcal P,\sigma}\int_{{\mathbb R}}
 e^{H_{\mathcal P,\rho,\sigma}(t)
       -\frac12{\mathbb E}[H_{\mathcal P,\rho,\sigma}(t)^2]}
 \,d\nu(t)
=\int_{{\mathbb R}}\prod_{p\in\mathcal P}
 e^{2p^{-\sigma}\mathrm{Re}(g_pp^{-it})
       -p^{-2\sigma}\mathrm{Re}(\rho p^{-2it})}\,d\nu(t).
\end{align*}
Iterating \eqref{eq:local-ratio}, we obtain
\begin{align*}
 c_d^{-q}\exp\left(-c_dq\sum_{p\in\mathcal P}p^{-3\sigma}\right)
 \leqslant\frac{\mathcal F_{\mathcal P}}{\mathcal G_{\mathcal P}}
 \leqslant c_d^q\exp\left(c_dq\sum_{p\in\mathcal P}p^{-3\sigma}\right).
\end{align*}
Since $\sigma\geqslant1/2$ and $\sum_pp^{-3/2}<\infty,$ the result follows after
enlarging $c_d.$

<!-- lemma-4: body_md -->
\label{lem:kahane}
Let $G_1,G_2$ be centered continuous Gaussian fields on a compact
interval $I,$ and suppose that
\begin{align*}
 \left|{\mathbb E}[G_1(t)G_1(u)]
       -{\mathbb E}[G_2(t)G_2(u)]\right|\leqslant c
\end{align*}
for every $t,u\in I.$  Given a finite positive measure $\nu$ on $I,$ put
\begin{align*}
 \mathcal M_j:=\int_I
 e^{G_j(u)-\frac{1}{2}{\mathbb E}[G_j(u)^2]}\,d\nu(u)
 \qquad(j=1,2).
\end{align*}
Then for $0<q\leqslant1,$ we have
\begin{align*}
 e^{-cq(1-q)/2}{\mathbb E}[\mathcal M_2^q]
 \leqslant{\mathbb E}[\mathcal M_1^q]
 \leqslant e^{cq(1-q)/2}{\mathbb E}[\mathcal M_2^q].
\end{align*}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Let $\mathcal{N}$ be an independent standard real Gaussian random variable.  Since
\begin{align*}
 {\mathbb E}[G_1(t)G_1(u)]\leqslant{\mathbb E}[G_2(t)G_2(u)]+c,
\end{align*}
Kahane's inequality \cite[Lemme~1]{Kahane}, applied to the concave function
$x\mapsto x^q$ and the fields $G_1$ and $G_2+\sqrt c\,\mathcal{N},$ gives
\begin{align*}
 {\mathbb E}[\mathcal M_1^q]
 &\geqslant{\mathbb E}[e^{q\sqrt c\,\mathcal{N}-qc/2}]\, {\mathbb E}[\mathcal M_2^q]\\
 &=e^{-cq(1-q)/2}\, {\mathbb E}[\mathcal M_2^q].
\end{align*}
Interchanging the two fields proves the other inequality, with the integral
form following by approximation by finite positive sums.

<!-- lemma-5: body_md -->
\label{lem:prime-sum}
There exists an absolute constant $c>0$ such that uniformly for $2\leqslant a\leqslant b$
and $v\in{\mathbb R},$ we have
\begin{equation}\label{eq:prime-covariance-integral}
 \sum_{e^a<p\leqslant e^b}p^{-1-iv}
 =\int_a^b e^{-ivw} \,\frac{dw}{w}
 +O\left((1+|v|)(1+\sqrt a)e^{-c\sqrt a}\right).
\end{equation}
If $v\ne0,$ then
\begin{align*}
\int_a^b e^{-ivw} \,\frac{dw}{w}
 \ll (|v|a)^{-1}.
\end{align*}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Applying the prime number theorem with the de la Vall\'ee Poussin error term followed by partial summation, we obtain
\begin{gather*}
 \left|\sum_{e^a<p\leqslant e^b}p^{-1-iv}
       -\int_a^b\exp(-ivw)\frac{dw}{w}\right|
 \ll(1+|v|)\left(\exp(-c\sqrt a)
       +\int_a^\infty\exp(-c\sqrt w)\,dw\right)\\
 \ll(1+|v|)(1+\sqrt a)\exp(-c\sqrt a),
\end{gather*}
where the last estimate follows on substituting $w=u^2.$  This proves
\eqref{eq:prime-covariance-integral}, and one integration by parts gives
the second assertion.

<!-- lemma-6: body_md -->
\label{lem:smooth-second}
Uniformly for sufficiently large $y$ and $u\geqslant1,$ we have
\begin{align*}
 {\mathbb E}[|T_y(u)|^2]
 \ll_d u(\log y)\exp\left(-\frac{\log u}{\log y}\right).
\end{align*}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Put $\eta=1/\log y.$  Rankin's argument and Mertens' estimate give
\begin{align*}
 \Psi(w,y):=&\,\#\{n\leqslant w:P^+(n)\leqslant y\}\\
 \leqslant &\, w^{1-\eta}\prod_{p\leqslant y}(1-p^{-1+\eta})^{-1}
 \ll w^{1-\eta}\log y.
\end{align*}
Here the change from $p^{-1}$ to $p^{-1+\eta}$ costs a bounded factor,
since $p^\eta-1\ll\eta\log p$ for $p\leqslant y$ and
\begin{align*}
\eta\sum_{p\leqslant y}(\log p)/p\ll1.
\end{align*}


Using \eqref{eq:covariance} and discarding restrictions on $v$ and $w,$
we obtain
\begin{align*}
 {\mathbb E}[|T_y(u)|^2]
 &\leqslant\sum_{v,w\geqslant1}|g(v,w)|
       \Psi\left(\frac{u}{\max\{v,w\}},y\right)\\
 &\ll u^{1-\eta}(\log y)
       \sum_{v,w\geqslant1}\frac{|g(v,w)|}
       {\max\{v,w\}^{1-\eta}}.
\end{align*}
For sufficiently large $y,$ we have
$\max\{v,w\}^{1-\eta}\geqslant(vw)^{5/12},$ and applying
\eqref{eq:kernel-norm} with $\sigma=5/12$ proves the lemma.

<!-- lemma-7: body_md -->
\label{lem:local-upper-general}
Fix a sufficiently large absolute constant $c_0,$ and put $L=\log y$
for sufficiently large $y.$  Then the following estimates hold.


\begin{enumerate}
\item[\rm (i)] If $L^{-1/2}\leqslant h\leqslant1$ and $z=\exp(c_0/h),$ then uniformly
for $2/3\leqslant q\leqslant1,$ we have
\begin{align*}
 {\mathbb E}\left[\left(\int_h^{2h}|F_y(1/2+it)|^2dt\right)^q\right]
 \ll_d\left(\frac{hL}{D_q(y)}\right)^q.

\end{align*}

 {\mathbb E}[W_x^q]
 \geqslant c\left(\frac{x}{\log x}\right)^q{\mathbb E}[I_x^q]
 -c_d\left(\frac{x}{\log x}\right)^q.
\end{equation}


For $\eta\geqslant0,$ put
\begin{align*}
 J_x(\eta):=\int_1^\infty|T_x(u)|^2\frac{du}{u^{2+2\eta}}.
\end{align*}
Given $1\leqslant V\leqslant(\log x)^{1/100},$ take
\begin{align*}
 \alpha=4V/\log x,
 \qquad
 \beta=2V/\log x,
 \qquad
 T=\sqrt x/2.
\end{align*}
Since $T_x(u)=S(u)$ for $u\leqslant T,$ splitting the integral at $T$ gives
\begin{align*}
 J_x(\alpha)
 &\leqslant I_x+T^{-2(\alpha-\beta)}J_x(\beta)\\
 &\leqslant I_x+e^{-V}J_x(0)
\end{align*}
for $x\geqslant16,$ where we used $J_x(\beta)\leqslant J_x(0).$  Using $(u+v)^q\leqslant u^q+v^q,$ we obtain
\begin{equation}\label{eq:ix-from-jx}
 {\mathbb E}[I_x^q]\geqslant{\mathbb E}[J_x(\alpha)^q]-e^{-Vq}{\mathbb E}[J_x(0)^q].
\end{equation}


By Lemma~\ref{lem:mellin} and Proposition~\ref{prop:general-mass-upper}, we
have
\begin{align*}
 {\mathbb E}[J_x(0)^q]\ll_d\left(\frac{\log x}{D_q(x)}\right)^q.
\end{align*}
Since $|1/2+\alpha+it|\asymp1$ on $J=[1/3,1/2],$ restricting the
Mellin integral to $J$ and applying Lemma~\ref{lem:general-lower-mass} gives
\begin{align*}
 {\mathbb E}[J_x(4V/\log x)^q]
 \gg_d\left(\frac{\log x}{V D_q(x)}\right)^q.
\end{align*}
Inserting these estimates into \eqref{eq:ix-from-jx}, we obtain
\begin{align*}
 {\mathbb E}[I_x^q]
 \geqslant(c_dV^{-q}-c'_de^{-Vq})
       \left(\frac{\log x}{D_q(x)}\right)^q.
\end{align*}
Choosing a sufficiently large fixed $V$ in terms of $d$ makes the
coefficient positive uniformly for $q\geqslant2/3,$ and substitution in
\eqref{eq:variance-smoothing} proves the lemma after absorbing the remaining
error for sufficiently large $x.$

The upper bound follows from Proposition~\ref{prop:sum-upper}, while
Lemma~\ref{lem:conditional-lower} and Lemma~\ref{lem:variance-lower} give
the lower bound for $2/3\leqslant q\leqslant1.$  For $0<q<2/3,$ put

 M:=\frac{x}{\sqrt{\log\log x}}.
\end{align*}
The lower bound at $q=2/3$ and the upper bound at $q=3/4$ give
\begin{align*}
 {\mathbb E}[|S(x)|^{4/3}]\gg_d M^{2/3},
 \qquad
 {\mathbb E}[|S(x)|^{3/2}]\ll_d M^{3/4}.
\end{align*}
Applying H\"older's inequality with $\theta=(9-12q)^{-1}$ and
$4/3=2q\theta+(3/2)(1-\theta),$ we obtain
\begin{align*}
 {\mathbb E}[|S(x)|^{2q}]
 \geqslant
 \frac{({\mathbb E}[|S(x)|^{4/3}])^{9-12q}}
      {({\mathbb E}[|S(x)|^{3/2}])^{8-12q}}
 \gg_d M^q.
\end{align*}
The powers of the implied constants are bounded uniformly in this range,
where $D_q(x)\asymp\sqrt{\log\log x},$ and the theorem follows, with
the endpoint $q=0$ immediate.

\label{prop:sym-square}
Let $\mathbb X_2$ be the random multiplicative function with local parameters $ e^{2i\theta_p}, 1, e^{-2i\theta_p},$ where the $\theta_p \in [0,\pi]$ are independent Sato--Tate angles. Then
\begin{align*}
 {\mathbb E}\left[\left|\sum_{n\leqslant x}\mathbb X_2(n)\right|^2 \right]=x\log x+O(x),
 \end{align*}
 and
 \begin{align*}
 {\mathbb E}\left[\sum_{n\leqslant x}\mathbb X_2(n) \right]=\lfloor\sqrt x\rfloor.
\end{align*}

<!-- proof-14: body_md -->
Write $h_k=h_k(e^{2i\theta},1,e^{-2i\theta})$ and
$\chi_m(\theta)=\sin((m+1)\theta)/\sin\theta.$ For $|z|<1,$ we have
\begin{gather*}
 \sum_{r\geqslant0}\chi_{2r}(\theta)z^r
 =\frac{1+z}{(1-e^{2i\theta}z)(1-e^{-2i\theta}z)},\\
 \sum_{k\geqslant0}h_kz^k
 =\frac1{1-z^2}\sum_{r\geqslant0}\chi_{2r}(\theta)z^r,
 \qquad \text{where }
 h_k=\sum_{0\leqslant j\leqslant k/2}\chi_{2k-4j}(\theta).
\end{gather*}
Character orthogonality now gives
\begin{align*}
 \sum_{k,l\geqslant0}{\mathbb E}[h_kh_l]z^kw^l
 &=\sum_{r,j_1,j_2\geqslant0}z^{r+2j_1}w^{r+2j_2}\\
 &=\frac1{(1-zw)(1-z^2)(1-w^2)}.
\end{align*}
Independence over the primes therefore gives
\begin{align*}
 {\mathbb E}[\mathbb X_2(m)\mathbb X_2(n)]
 =\#\{(r,u,v)\in{\mathbb N}^3:m=ru^2,\ n=rv^2\}.
\end{align*}
Summing over $m,n\leqslant x,$ we obtain
\begin{align*}
 {\mathbb E}\left[\left|\sum_{n\leqslant x}\mathbb X_2(n)\right|^2\right]
 &=\sum_{r\leqslant x}\left\lfloor\sqrt{\frac{x}{r}}\right\rfloor^2\\
 &=x\sum_{r\leqslant x}\frac1r
   +O\left(\sqrt x\sum_{r\leqslant x}r^{-1/2}\right)\\
 &=x\log x+O(x).
\end{align*}
The character identity gives ${\mathbb E}[h_k]=1$ for even $k$ and ${\mathbb E}[h_k]=0$ for odd
$k.$  Hence ${\mathbb E}[\mathbb X_2(n)]$ is the indicator function of the squares, which
proves the first-moment formula.
