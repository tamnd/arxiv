<!-- proposition-1: body_md -->
\label[proposition]{prop:approxBV}
		Assume that $\Omega$ is bounded, star-shaped and $\partial \Omega \in \mathcal{C}^{0,1}$. If $u\in \mathrm{BV}(\Omega)$, then for every  $\varepsilon > 0$ there exists a $u_\varepsilon \in \mathcal{C}^\infty(\Omega)$ such that
		\begin{equation*}
			\Vert u - u_\varepsilon\Vert_{L^1} \leq \varepsilon \Vert D u\Vert_{\mathcal{M}},\quad  \Vert \nabla u_\varepsilon\Vert_{L^1} \leq (1 + c \varepsilon) \Vert Du\Vert_{\mathcal{M}},\quad \Vert D^2u_\varepsilon\Vert_{L^1} \leq \frac{c}{\varepsilon} \Vert D u\Vert_{\mathcal{M}}.
		\end{equation*}
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
\label[remark]{rem:FEinterpolationIneq}
		Given the nodal Lagrange interpolation operator $\mathcal{I}_h, u\in\mathcal{C}^2(\Omega)$ and an arbitrary triangle $T \in\mathcal{T}_h$,
		\begin{equation}
			\label{eq:FEinterpolationIneq}
			\Vert u - \mathcal{I}_hu\Vert _{L^p(T)} + h\Vert \nabla(u - \mathcal{I}_hu)\Vert _{L^p(T)} \leq ch^2 \Vert D^2u\Vert _{L^p(T)}.
		\end{equation}
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lem:BoundDiscreteTGVdiff}
		Let $\mathcal{T}_h$ be an admissible triangulation of $\Omega$, $u \in \mathcal{C}^2(\Omega)$ and $\mathcal{I}_h$ the Lagrange or Clément interpolation operator onto $ \mathcal{V}_h^k$.
		Then $\mathrm{TGV}_{\alpha}^2(u - \mathcal{I}_h u) \leq \alpha_1 \sum_{T \in \mathcal{T}_h} \Vert \nabla(u - \mathcal{I}_h u)\Vert _{L^1(T)}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Using the representation in \eqref{def:TGV}, together with integration by parts and Hölder's inequality, we obtain:
		\begin{align*}
			&\mathrm{TGV}_{\alpha}^2(u - \mathcal{I}_h u)\\
			&\quad= \sup \left \lbrace  \int_\Omega(u - \mathcal{I}_h u)  \operatorname{div}^2v \mathrm{d} x\colon v \in \mathcal{C}_c^2(\Omega, \mathrm{Sym}^2(\mathbb{R}^d)),  \Vert v\Vert _\infty \leq \alpha_0, \Vert \nabla v\Vert _\infty \leq \alpha_1\right \rbrace\\
			&\quad= \sup \left \lbrace  \int_\Omega \nabla(\mathcal{I}_h u - u)  \nabla v \mathrm{d} x\colon v \in \mathcal{C}_c^2(\Omega, \mathrm{Sym}^2(\mathbb{R}^d)), \Vert v\Vert _\infty \leq \alpha_0, \Vert \nabla v\Vert _\infty \leq \alpha_1\right \rbrace\\
			&\quad \leq \alpha_1 \sum_{T \in \mathcal{T}_h} \Vert \nabla(u - \mathcal{I}_h u)\Vert _{L^1(T)}.
		\end{align*}
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:approxTGV}
		Let $u$ be a function such that $\mathrm{TGV}_\alpha^2(u)<\infty$, then for its approximation $u_\varepsilon$ as in~\cref{prop:approxBV} it holds that
		$
		\mathrm{TGV}_\alpha^2(u_\varepsilon) \leq (1 + c\varepsilon) \mathrm{TGV}_\alpha^2(u).
		$
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We use the explicit construction of $u_\varepsilon$:
		For $x \in \Omega_\varepsilon$ we define $\tilde{u}_\varepsilon(x) = u(\frac{x}{1+\varepsilon})$ and $u_\varepsilon(x) = \left(\tilde{u}_\varepsilon * \rho_\varepsilon \right)(x)$.
		Then we extend $u$ by zero on $\mathbb{R}^d$ and calculate:
		\begin{align*}
			&\mathrm{TGV}_{\alpha}^2(u_\varepsilon)=\sup \left \lbrace  \int_\Omega u_\varepsilon  \operatorname{div}^2v \mathrm{d} x\colon v \in \mathcal{C}_c^2(\Omega, \mathrm{Sym}^2(\mathbb{R}^d)), \Vert v\Vert _\infty \leq \alpha_0, \Vert \nabla v\Vert _\infty \leq \alpha_1\right \rbrace\\
			&\quad\leq \sup \left \lbrace  \int_{\Omega_\varepsilon} \tilde{u}_\varepsilon \operatorname{div}^2(v * \rho_\varepsilon) \mathrm{d} x\colon v \in \mathcal{C}_c^2(\Omega, \mathrm{Sym}^2(\mathbb{R}^d)), \Vert v\Vert _\infty \leq \alpha_0, \Vert \nabla v\Vert _\infty \leq \alpha_1\right \rbrace\\
			&\quad\leq \sup \left \lbrace  \int_{\Omega_\varepsilon} \tilde{u}_\varepsilon \operatorname{div}^2v \mathrm{d} x\colon v \in \mathcal{C}_c^2(\Omega_\varepsilon, \mathrm{Sym}^2(\mathbb{R}^d)), \Vert v\Vert _\infty \leq \alpha_0, \Vert \nabla v\Vert _\infty \leq \alpha_1\right \rbrace\\
			&\quad\leq (1+\varepsilon)^d \mathrm{TGV}_{\alpha}^2(u) \leq (1 + c\varepsilon) \mathrm{TGV}_{\alpha}^2(u),
		\end{align*}
		using the symmetry of the mollifier and Fubini’s theorem to move the mollification inside the divergence. Since the mollification operator then commutes with the divergence, we invoke its non-expansiveness w.r.t. the norm together with the scaling properties of $\mathrm{TGV}_\alpha^2$~\cite[Proposition~3.3]{Br10} to conclude the proof.
<!-- end proof-2 -->
<!-- theorem-1: body_md -->
\label{thm:semi_discrete_conv}
		Assume that $\Omega$ is star-shaped. Let $\xi$, $\xi_h$ be defined as in~\cref{def:contTGVdenoiseMin} and $\alpha = (\alpha_0, \alpha_1), \alpha_0, \alpha_1>0$. Then
		$
		\Vert \xi - \xi_h\Vert_{L^2} \leq ch^{\frac{1}{4}} \left( \Vert\xi\Vert_{L^1} +  \mathrm{TGV}_{\alpha}^2(\xi)\right).
		$
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
Let $\varepsilon > 0 $ and $\xi_\varepsilon \in \mathcal{C}^\infty(\Omega)$ be an approximation that satisfies all the properties of~\cref{prop:approxBV}. Since $\psi$ is strongly convex with parameter ${\lambda}$ and $\xi_h$ is a discrete minimizer, we have
		\begin{align*}
			\frac{\lambda}{2}\Vert \xi& - \xi_h\Vert_{L^2}^2 \leq \psi(\xi_h) - \psi(\xi) \leq \psi(\mathcal{I}_h\xi_\varepsilon) - \psi(\xi)\\
			&= ( \mathrm{TGV}_{\alpha}^2(\mathcal{I}_h\xi_\varepsilon) - \mathrm{TGV}_{\alpha}^2(\xi)) +  \frac{\lambda}{2}(\Vert \mathcal{I}_h\xi_\varepsilon - z\Vert_{L^2}^2 -  \Vert \xi - z\Vert_{L^2}^2) = \mathcal{A}_1 + \mathcal{A}_2.
		\end{align*}
		Here $\mathcal{I}_h$ is the Clément-interpolation operator onto $\mathcal{P}^1$ functions.
		
		$\mathcal{A}_1$: We add and subtract the $\mathrm{TGV}_{\alpha}^2$ seminorm  and use~\cref{lem:BoundDiscreteTGVdiff},~\cref{lem:approxTGV} and~\cref{prop:approxBV} as well as~\cref{rem:FEinterpolationIneq}  to bound the first summand, 
		\begin{align*}

<!-- definition-1: body_md -->
\label{def:ContTgv}
		Let $\Omega_\mathbf{x} \subset \mathbb{R}^d$ and $\Omega_t = (0, \tilde{t})$ for some $\tilde{t} > 0$.

			\mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}(u) =  \int_{\Omega_t} \mathrm{TGV}_{\alpha, \mathbf{x}}^k(u) \mathrm{d} t + \int_{\Omega_\mathbf{x}} \mathrm{TGV}_{\beta, t}^l(u)\mathrm{d}\mathbf{x}.

<!-- proposition-2: body_md -->
\label{prop:TGVproperties2}
		The following statements hold:
		\begin{enumerate}[leftmargin=*]
			\item $\mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}(u)$ is a seminorm on the normed space $\mathrm{BGV}_{(\alpha, \beta)}^{(k,l)}(\Omega_\mathbf{x} \times\Omega_t)$.
			\item For $u\in L^1_\text{loc}(\Omega_\mathbf{x} \times \Omega_t)$, $\mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}(u) = 0$ if and only if $u$ is a polynomial of degree less than k in $\Omega_\mathbf{x}$ and of degree less than $l$ in $\Omega_t$.
			\item For fixed $(k,l)$ and positive weights $(\alpha, \beta), (\tilde{{\alpha}}, \tilde{\beta})$ the seminorms $\mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}(u)$ and $\mathrm{TGV}_{(\tilde{\alpha}, \tilde{\beta})}^{(k,l)}(u)$ are equivalent .
			\item $\mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}$ is rotationally invariant; i.e. for each orthonormal matrix $O \in \mathbb{R}^{d \times d}$,  $\tilde{u} \in \mathrm{BGV}_{(\alpha, \beta)}^{(k,l)}\left(O_1^T \Omega_\mathbf{x}, \Omega_t\right)$ and $\mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}(u) = \mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}(\tilde{u})$; $\tilde{u}(x, y) = u\left(O_1 x, y\right)$.
			\item For $(r_1, r_2) > 0$ and $u\in \mathrm{BGV}_{(\alpha, \beta)}^{(k,l)}(\Omega_\mathbf{x} \times\Omega_t)$ we have, defining $\tilde{u}(x, y) = u(r_1x, r_2y)$ and $\tilde{\alpha}_i = \alpha_i r_1^{k-i},\tilde{\beta}_j = \beta_j r_2^{l-j}$, that $\tilde{u} \in \mathrm{BGV}_{(\alpha, \beta)}^{(k,l)}(r_1^{-1}\Omega_\mathbf{x} \times r_2^{-1}\Omega_t)$ with 
			\begin{equation*}
				\mathrm{TGV}_{(\alpha, \beta)}^{(k,l)}(\tilde{u}) = r_1^{-d} \int_{\Omega_t} \mathrm{TGV}_{\tilde{\alpha}, \mathbf{x}}^k(u) \mathrm{d} t + r_2^{-1}\int_{\Omega_\mathbf{x}} \mathrm{TGV}_{\tilde{\beta}, t}^l(u)\mathrm{d} \mathbf{x}. 
			\end{equation*}
		\end{enumerate}
