<!-- theorem-1: body_md -->
\label{thm:support}
Let $0<m\leq d\le D$ be integers and let $\mathcal K\subset\mathbb R^{D\times d}$ be a (nonempty) compact
and convex set, with
\[
\lambda:=\min_{X\in\mathcal K}\sigma_m(X)>0,
\quad \Lambda:=\max_{X\in\mathcal K} \sigma_1(X) < \infty,
\]
where $\sigma_1(X)\ge\dots\ge\sigma_d(X)\ge0$ denote the singular values of $X$.
Let $M\subset\mathbb R^d$ be a Borel set with
$\mathcal H^m(M)<\infty$ and let $\mu=\theta\mathcal H^m\mathrel{\tikz[baseline=0ex]{\draw[line width=.5pt,line cap=round] (0ex,1ex) -- (0ex,0ex) -- (1ex,0ex);}} M$ be a finite
positive measure on $\mathbb R^d$. Suppose that $\sigma:=X\mu$, where $X:M\to\mathcal K$, and that $\operatorname{div} \sigma$ (the row-wise divergence) is a finite measure. Then
\begin{equation}\label{eq:theorem}
\mu(\mathbb R^d)\le C(d,\lambda,\Lambda)\mathcal H^m(M)^{1/m}|\operatorname{div} \sigma|(\mathbb R^d).
\end{equation}
<!-- end theorem-1 -->
<!-- rmk-1: body_md -->
Thanks to~\cite[Proposition~3.1]{ADHR19}, the measure $\mu$ and hence the set $M$ (up to replacing it with $M\cap\{\theta>0\}$) are automatically $m$-rectifiable.
<!-- end rmk-1 -->
<!-- theorem-2: body_md -->
\label{thm:support.bis}
Let $0<m\leq d$ be integers and consider a (nonempty) compact
and convex set $\mathcal K\subset V$, as well as a first-order operator $\mathcal A=\sum_{j=1}^d \mathcal A_j\partial_j$, where
$\mathcal A_j\in\operatorname{Lin}(V,W)$ are linear maps between Euclidean spaces.                                              
For $T_{\kappa}:\mathbb R^d\to W$ given by
\[
  T_{\kappa}(\xi):=\sum_j\xi_j\mathcal A_j\kappa, \quad \kappa\in\mathcal K,
\]
assume that
\[
\lambda:=\min_{\kappa\in\mathcal K}\sigma_m(T_\kappa)>0,
\quad \Lambda:=\max_{\kappa\in\mathcal K} \sigma_1(T_\kappa) < \infty.
\]
Letting $M\subset\mathbb R^d$ and $\mu$ be as above, and $X:M\to\mathcal K$, $\sigma:=X\mu$, suppose that $\mathcal A \sigma$ is a finite measure. Then
\begin{equation}\label{eq:theorem.bis}
\mu(\mathbb R^d)\le C(\mathcal A,d,\lambda,\Lambda)\mathcal H^m(M)^{1/m}|\mathcal A \sigma|(\mathbb R^d).
\end{equation}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{Lp}
    Assume that $\mu$ is a finite positive measure on $\mathbb R^d$ and that $\sigma:=X\mu$, where $X:\mathbb R^d\to\mathcal{K}$ is as in Theorem~\ref{thm:support.bis}. There exist $p\in(1,\infty)$ and $C>0$,
    depending on $d,\lambda,\Lambda$, such that the following holds:
    If $\mathcal A \sigma$ is a finite measure and $M$ is a Borel set with $\sigma$-finite $\mathcal H^m$-measure then, writing $\mu\mathrel{\tikz[baseline=0ex]{\draw[line width=.5pt,line cap=round] (0ex,1ex) -- (0ex,0ex) -- (1ex,0ex);}} M=\theta\mathcal H^m\mathrel{\tikz[baseline=0ex]{\draw[line width=.5pt,line cap=round] (0ex,1ex) -- (0ex,0ex) -- (1ex,0ex);}} M$, we have
                                                                                                                                   
    \[\|\theta\|_{L^{p,\infty}(\mathcal H^m\mathrel{\tikz[baseline=0ex]{\draw[line width=.5pt,line cap=round] (0ex,1ex) -- (0ex,0ex) -- (1ex,0ex);}} M)}\le\frac{C}{r^{m(p-1)/p}}\bigl[\mu(\mathbb R^d)+r|\mathcal A \sigma|(\mathbb R^d)\bigr]^{1/p}\mu(\mathbb R^d)^{1-1/p}\]
    for all $r>0$.
<!-- end theorem-3 -->
<!-- rmk-2: body_md -->
By a standard layer-cake argument we also get uniform bounds on $\|\theta\|_{L^q(\mathcal H^m\mathrel{\tikz[baseline=0ex]{\draw[line width=.5pt,line cap=round] (0ex,1ex) -- (0ex,0ex) -- (1ex,0ex);}} M)}$ for all $q<p$. Whether the $L^{p,\infty}$ bounds extend up to the endpoint $p=m/(m-1)$ is left as an open question.
<!-- end rmk-2 -->
<!-- theorem-4: body_md -->
\label{ms}
Let $F$ be an anisotropic integrand that satisfies (AC1).
Then the Michael--Simon inequality
\begin{equation}\label{eq:piersilvio}
|V|(\mathbb R^d)\le C(F)\mathcal{H}^m(\{\Theta_V\ge1\})^{1/m}|\delta^FV|(\mathbb R^d)
\end{equation}
holds for all rectifiable varifolds $V$ with density $\Theta_V\ge1$ $|V|$-a.e.\ (with both $|V|$ and $|\delta^FV|$ finite measures). As a consequence, for $m>1$ we have
\begin{equation}
\label{eq:gianfranco}
|V|(\mathbb R^d)^{(m-1)/m}\le C(F)|\delta^FV|(\mathbb R^d).
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:energy}
There exists a constant $C(d,\Lambda)$ such that
\begin{equation}\label{eq:selected}
\int_0^T\int_{\mathbb R^d}\rho
\bigl(|\nabla B|^2+|B\nabla\log\rho|^2\bigr) \;\mathrm{d} x \;\mathrm{d} t
\le C\gamma+C\gamma\log\biggl(\frac{1+\sqrt{T}\,\delta}{\gamma}\biggr).
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
For any smooth $\varphi$ we first compute that
\begin{equation}\label{eq:product}
\mathcal H(S\varphi)=S\mathcal H_\zeta\varphi-\kappa S\varphi,
\end{equation}
where $\mathcal H_\zeta:=\mathcal H+2\zeta\cdot\nabla$, so that $\mathcal H_\zeta B=\kappa B$.

Letting
\[
  J(t):=\int_{\mathbb R^d}\rho_t\log(\rho_t/S_t)\;\mathrm{d} x,
\]
using Jensen's inequality for $s\mapsto s\log s$, and recalling that $\rho\le S$,
we get
\begin{equation}\label{jensen}
\gamma\log\biggl(\frac \gamma{1+\sqrt{T-t}\,\delta}\biggr)\le J(t)\le0.
\end{equation}
Recalling that $\mathcal H\rho=0$ and using~\eqref{eq:product} with $\varphi=q:=\rho/S$ we get again $\mathcal H_\zeta q=\kappa q$.
Hence, by the chain rule, $\Phi(q):=q\log q$ is such that
\[\mathcal H_\zeta\Phi(q)=\Phi'(q)\mathcal H_\zeta q+\Phi''(q)|\nabla q|^2
=(1+\log q)\kappa q+\frac{|\nabla q|^2}{q},\]
so that
\[
  \mathcal H(\rho\log(\rho/S))=\mathcal H(S\Phi(q))=S\mathcal H_\zeta\Phi(q)-\kappa S\Phi(q)=\kappa\rho+\rho|\nabla\log q|^2
\]
and thus
\begin{equation}\label{eq:relative}
J'(t)=\int_{\mathbb R^d}\rho_t\biggl|\nabla\log\frac{\rho_t}{S_t}\biggr|^2 \;\mathrm{d} x
+\int_{\mathbb R^d}\kappa_t\rho_t \;\mathrm{d} x.
\end{equation}
It follows that
\begin{equation}\label{pluto}
\int_0^T\int_{\mathbb R^d}\rho\biggl|\nabla\log\frac\rho S\biggr|^2 \;\mathrm{d} x \;\mathrm{d} t
\le -J(0) \leq \gamma\log\biggl(\frac{1+\sqrt T\,\delta}{\gamma}\biggr).                     
\end{equation}
We now compute that
\[\mathcal H(\rho|B|^2)=2\rho|\nabla B|^2+2\langle\nabla\rho-\rho\zeta,\nabla|B|^2\rangle+2\kappa\rho|B|^2.\]
Integrating in space and using $|\nabla|B|^2|\le2\Lambda|\nabla B|$
(recall that $|B|\le\Lambda$), we obtain
\[\frac{\mathrm{d}}{\mathrm{d}t}\int_{\mathbb R^d}\rho_t|B_t|^2 \;\mathrm{d} x\ge\int_{\mathbb R^d}2\rho_t|\nabla B_t|^2 \;\mathrm{d} x-\int_{\mathbb R^d}4\Lambda\rho_t|\nabla\log(\rho_t/S_t)||\nabla B_t| \;\mathrm{d} x
.\]
By Young's inequality, since $\int_{\mathbb R^d}\rho_t|B_t|^2 \;\mathrm{d} x\le\Lambda^2\int_{\mathbb R^d}\rho_t \;\mathrm{d} x=\Lambda^2\gamma$ we deduce that
\[\int_0^T\int_{\mathbb R^d}\rho|\nabla B|^2 \;\mathrm{d} x \;\mathrm{d} t\le\Lambda^2\gamma+C\int_0^T\int_{\mathbb R^d}\rho|\nabla\log(\rho/S)|^2 \;\mathrm{d} x \;\mathrm{d} t,\]
which gives
\begin{equation}\label{nabla.B.est}
    \int_0^T\int_{\mathbb R^d}\rho|\nabla B|^2 \;\mathrm{d} x \;\mathrm{d} t\le C\gamma+C\gamma\log\biggl(\frac{1+\sqrt{T}\,\delta}{\gamma}\biggr)
\end{equation}
thanks to~\eqref{pluto}.

Next, by construction we have $|\eta_t|\le\frac{1}{\sqrt{T-t}}$,
so that
\[\int_{\mathbb R^d}\rho_t|\eta_t|^2 \;\mathrm{d} x\le\frac{1}{T-t}\int_{\mathbb R^d}\rho_t \;\mathrm{d} x=\frac{\gamma}{T-t}\]
and
\[
\int_{\mathbb R^d} \rho_t|\eta_t|^2 \;\mathrm{d} x\le\int_{\mathbb R^d}S_t|\eta_t|^2 \;\mathrm{d} x
\le\frac{1}{\sqrt{T-t}}\int_{\mathbb R^d}|\mathcal P_{T-t}\operatorname{div} \sigma| \;\mathrm{d} x\le\frac{\delta}{\sqrt{T-t}}.
\]
If $\sqrt{T}>\gamma/\delta$ then, letting $s:=T-t$, we can estimate
\[\int_0^T\int_{\mathbb R^d}\rho|\eta|^2 \;\mathrm{d} x \;\mathrm{d} s
\le\int_0^{\gamma^2/\delta^2}\frac{\delta}{\sqrt{s}} \;\mathrm{d} s
+\int_{\gamma^2/\delta^2}^T\frac{\gamma}{s}\;\mathrm{d} s
=2\gamma+2\gamma\log\biggl(\frac{\sqrt{T}\,\delta}{\gamma}\biggr),\]
while if $\sqrt{T}\le\gamma/\delta$ then the second bound gives directly that
\[
  \int_0^T\int_{\mathbb R^d}\rho|\eta|^2\;\mathrm{d} x\;\mathrm{d} s\le2\sqrt{T}\,\delta,
\]
so that in both cases we have
\begin{equation}\label{eta.est}
    \int_0^T\int_{\mathbb R^d}\rho|\eta|^2 \;\mathrm{d} x \;\mathrm{d} t
    \le 4\gamma\log\biggl(1+\frac{\sqrt{T}\,\delta}{\gamma}\biggr).
\end{equation}
Also,~\cite[eq.~(3.9)]{GR26} is unchanged and reads
$\operatorname{div} B+B\zeta=\eta$. Together with~\eqref{nabla.B.est} and~\eqref{eta.est}, this gives
\begin{equation}\label{eq:fullenergy}
\int_0^T\int_{\mathbb R^d} \rho|B\zeta|^2 \;\mathrm{d} x \;\mathrm{d} t
\le C\gamma+C\gamma\log\biggl(\frac{1+\sqrt{T}\,\delta}{\gamma}\biggr).
\end{equation}
Since $|B|\le \Lambda$ and
$B\nabla\log\rho=B\zeta+B\nabla\log(\rho/S)$, combining~\eqref{eq:fullenergy} with~\eqref{pluto} we also get the desired integral bound for $\rho|B\nabla\log\rho|^2$.

<!-- lemma-2: body_md -->
\label{lem:dimensional}
For every $0<\tau<T$, writing
$I(\rho_t):=\int_{\mathbb R^d}\rho_t|\nabla\log\rho_t|^2\;\mathrm{d} x$, we have
\begin{equation}\label{eq:dimensional}
\int_0^\tau\biggl[I(\rho_t)-\frac{(d-m)\gamma}{2(T-t)}\biggr] \;\mathrm{d} t
\le C\gamma+C\gamma\log\biggl(\frac{1+\sqrt{T}\,\delta}{\gamma}\biggr)
\end{equation}
for some constant $C(d,\lambda,\Lambda)$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Choose a smooth $\chi:[0,\infty)\to[0,1]$, equal to zero on
$[0,\lambda^2/16]$ and to one on $[\lambda^2/4,\infty)$. Let
$Q:=\chi(B^TB)$ be defined by spectral calculus as
\[
  Q:=O\operatorname{diag}(\chi(\sigma_1^2(B)),\dots,\chi(\sigma_d^2(B)))O^T,
\]
where we write $B^TB=O\operatorname{diag}(\sigma_1^2(B),\dots,\sigma_d^2(B))O^T$ using the spectral theorem. Moreover, set $\Pi:=Q^2$, which plays the role of a projection matrix. We have
\begin{equation}\label{eq:spectral}
0\le\Pi\le I_d,\quad |Qz|^2\le C|Bz|^2,\quad
|\nabla Q|\le C|\nabla B|.
\end{equation}
On $\{U_t\ge\sqrt{T-t}\,R_t\}$,~\eqref{eq:notation} and~\eqref{cvx} imply
$\sigma_m(B_t)\ge \lambda/2$. To see this, observe that $S_t\leq 2U_t$ gives $U_t/S_t\geq1/2$, and thanks to~\eqref{cvx} we can write
\[
  \sigma_m(B_t)=\frac{U_t}{S_t}\sigma_m(V_t/U_t)\geq\lambda/2.
\]
Hence, $\operatorname{tr}\Pi_t\ge m$ by construction of $\chi$. The complementary
set $E_t$ has
\[
  \int_{E_t}\rho_t\;\mathrm{d} x\le\int_{E_t} U_t\;\mathrm{d} x\le\int_{E_t}\sqrt{T-t}\,R_t\;\mathrm{d} x\le\sqrt{T-t}\,\delta.
\]
Therefore,
\begin{equation}\label{eq:trace}
\int_{\mathbb R^d}\rho_t\operatorname{tr}(I_d-\Pi_t)\;\mathrm{d} x\le(d-m)\gamma+d\min\{\gamma,\sqrt{T-t}\,\delta\}.
\end{equation}

Differentiating the convolution with the Gaussian $p_s$ (where $s:=T-t$)
and noting that
\[
  \nabla^2 p_s(x-y)=\biggl[\frac{(x-y)\otimes (x-y)}{4s^2}-\frac{I_d}{2s}\biggr]p_s(x-y),
\]
we get
\begin{equation}\label{eq:hessian}
\nabla^2\log\rho_t\ge-\frac{I_d}{2(T-t)}.
\end{equation}
This is the well-known log semi-convexity improvement of the heat flow, and can be proved by observing that the difference of the two terms equals the covariance of $y$ under the probability
measure $p_s(x-y)\;\mathrm{d}\mu_0(y)/\rho_t(x)$, divided by $4s^2$. Using $I(\rho_t)=-\int_{\mathbb R^d}\rho_t\operatorname{tr}(\nabla^2\log\rho_t)\;\mathrm{d} x$,
\eqref{eq:trace}, and~\eqref{eq:hessian}, we furthermore obtain
\begin{align*}
I(\rho_t)&=-\int_{\mathbb R^d}\rho_t\operatorname{tr}((I_d-\Pi_t)\nabla^2\log\rho_t) \;\mathrm{d} x-\int_{\mathbb R^d}\rho_t\operatorname{tr}(\Pi_t\nabla^2\log\rho_t) \;\mathrm{d} x\\
&\le\frac{(d-m)\gamma}{2(T-t)}
+\min\biggl[\frac{d\gamma}{2(T-t)},\frac{d\delta}{2\sqrt{T-t}}\biggr]
-\sum_i\int_{\mathbb R^d}\rho_t [e_i^T(\nabla^2\log\rho_t)e_i] \;\mathrm{d} x,
\end{align*}
where $e_i=e_{i,t}$ denotes the $i$-th column of $Q_t$.

Now, for every smooth vector field $e$ with compact support, integration
by parts gives
\begin{equation}\label{eq:ibp}
-\int_{\mathbb R^d}\rho_t [e^T(\nabla^2\log\rho_t)e] \;\mathrm{d} x
=\int_{\mathbb R^d}\rho_t [
(\operatorname{div} e+e\cdot\nabla\log\rho_t)^2
-\operatorname{tr}((De)^2)] \;\mathrm{d} x,
\end{equation}
where $(De)_{jk}:=\partial_ke_j$.
Indeed, writing $b_t:=\operatorname{div} e+e\cdot\nabla\log\rho_t=\rho^{-1}_t\operatorname{div}(\rho_t e)$, we have
$\int_{\mathbb R^d}\rho_t b_t^2 \;\mathrm{d} x=-\int_{\mathbb R^d}\rho_t e\cdot\nabla b_t \;\mathrm{d} x$ and
\[e\cdot\nabla b_t=e\cdot\nabla \operatorname{div} e+((e\cdot\nabla)e)\cdot\nabla\log\rho_t+e^T(\nabla^2\log\rho_t)e,\]
and, once we integrate, the first two terms on the right-hand side combine to give the trace term $-\int_{\mathbb R^d}\rho_t\operatorname{tr}((De)^2)\;\mathrm{d} x$. By a standard cut-off argument,
\eqref{eq:ibp} also applies to the columns $e_{i,t}$ of $Q_t$.
Summing over $i$ and recalling~\eqref{eq:spectral}, this gives
\begin{equation}\label{eq:fisher}
I(\rho_t)\le\frac{(d-m)\gamma}{2(T-t)}+\min\biggl[\frac{d\gamma}{2(T-t)},\frac{d\delta}{2\sqrt{T-t}}\biggr]
+C\int_{\mathbb R^d}\rho_t(|\nabla B_t|^2+
|B_t\nabla\log\rho_t|^2) \;\mathrm{d} x.
\end{equation}
Integrating in $t$, applying Lemma~\ref{lem:energy},
and bounding the integral of the second term as in the previous proof, we reach the claim.

<!-- proof-4: body_md -->
Assume again that $\mu$ has compact support, $\mu(\mathbb R^d)=1$, $|\operatorname{div} \sigma|(\mathbb R^d)=\delta$, and $0\le\mu_0\le\mu$ with $\gamma:=\mu_0(\mathbb R^d)\in(0,1]$;
the submeasure $\mu_0$ will be chosen in a moment.

The idea now is to look at the entropy $H(\rho_t)$ as time passes.
More specifically, we look at a renormalized version of it, adapted to codimension $d-m$:
let
\[\tilde H(\rho_t):=H(\rho_t)+\frac{(d-m)\gamma}{2}\log(T-t)=\int_{\mathbb R^d}\rho_t\log\rho_t \;\mathrm{d} x+\frac{(d-m)\gamma}{2}\log(T-t).\]
Since $\mathcal H\rho=0$, the derivative $\frac{\mathrm{d}}{\mathrm{d}t}H(\rho_t)=I(\rho_t)\ge0$ is the Fisher information (recall that
$\rho_t$ evolves through the backward heat equation), and thus
\[\frac{\mathrm{d}}{\mathrm{d}t}\tilde H(\rho_t)=I(\rho_t)-\frac{(d-m)\gamma}{2(T-t)}.\]
The previous lemma then gives
\[\limsup_{t\to T^-}\tilde H(\rho_t)-\tilde H(\rho_0)\le C\gamma+C\gamma\log\biggl(\frac{1+\sqrt T\,\delta}{\gamma}\biggr).\]
Moreover, since $\rho_0=\mathcal P_T\mu_0$, we have $\rho_0\le\frac{\gamma}{(4\pi T)^{d/2}}$, and hence
\[H(\rho_0)\le\int_{\mathbb R^d}\rho_0\log\biggl(\frac{\gamma}{(4\pi T)^{d/2}}\biggr)\;\mathrm{d} x\le\gamma\log\gamma-\frac{d\gamma}{2}\log(4\pi T),\]
giving in particular
\[\tilde H(\rho_0)\le-\frac{m\gamma}{2}\log T.\]
Thus, for some constant $C(d,\lambda,\Lambda)$, we get
\begin{equation}\label{u.b.for.h}
\limsup_{t\to T^-}\tilde H(\rho_t)\le -\frac{m\gamma}{2}\log T+C\gamma+C\gamma\log\biggl(\frac{1+\sqrt{T}\,\delta}{\gamma}\biggr).
\end{equation}

Next, we observe that $\mu$ is a rectifiable measure, by~\cite[Proposition~3.1]{ADHR19}.
We can then take $E\subseteq M$ to be a finite disjoint union of compact sets, each of which is included in a $1$-Lipschitz graph (up to a rotation), and $\mu_0:=\mu\mathrel{\tikz[baseline=0ex]{\draw[line width=.5pt,line cap=round] (0ex,1ex) -- (0ex,0ex) -- (1ex,0ex);}} E$.
We claim a lower bound on the left-hand side, namely
\begin{equation}\label{lb}
\liminf_{t\to T^-}\tilde H(\rho_t)\ge\gamma\log\biggl(\frac{\gamma}{\mathcal H^m(E)}\biggr)-C(d)\gamma.
\end{equation}
Note that the geometry of $E$ is not controlled, namely $E$ could start resembling an $m$-plane only
at very small scales, but the key point is that in the limit $t\to T$ only the picture at smaller and smaller scales matters. More precisely, we will just need a bound on the Minkowski content at (non-uniformly) small scales.

To establish~\eqref{lb}, we first compare the probability density $\varphi_t:=\rho_t/\gamma$
with a more explicit one, namely
\[
 \psi_s:= \gamma_s^{-1}
 e^{-\operatorname{dist}(x,E)^2/(4s)},\quad\gamma_s:=\int_{\mathbb R^d}e^{-\operatorname{dist}(x,E)^2/(4s)}\;\mathrm{d} x,
\]
where we set $s:=T-t$ for simplicity. Since the relative entropy $\int_{\mathbb R^d}\varphi_t(\log\varphi_t-\log\psi_t)\;\mathrm{d} x\ge0$,
after a little rearrangement we get
\[H(\rho_t)\ge\gamma(\log\gamma-\log\gamma_s)-\frac{1}{4s}\int_{\mathbb R^d}\operatorname{dist}(x,E)^2 \rho_t(x)\;\mathrm{d} x.\]
Since $\operatorname{dist}(x,E)\le|x-y|$ for all $y\in E$ and $\rho_t=\mathcal P_s\mu_0$, the last integral is bounded by
\[
  \int_E\int_{\mathbb R^d}|x-y|^2p_s(x-y)\;\mathrm{d} x\;\mathrm{d}\mu_0(y)=2d\gamma s,
\]
so that
\[
  H(\rho_t)\ge\gamma(\log\gamma-\log\gamma_s)-\frac{d\gamma}{2}.
\]
Finally, by construction of $E$, it is straightforward to check that there exists $r_0>0$ such that $|B_r(E)|\le C(d)r^{d-m}\mathcal H^m(E)$ for all radii $0<r<r_0$,
giving
\begin{align*}\gamma_s&=\int_0^\infty\frac{r}{2s}e^{-r^2/(4s)}|B_r(E)|\;\mathrm{d} r\\
&\le C(d)\int_0^{r_0}\frac{r^{1+d-m}}{2s}e^{-r^2/(4s)}\mathcal H^m(E)\;\mathrm{d} r+C(E)e^{-r_0^2/(8s)}\\
&\le C(d)s^{(d-m)/2}\mathcal H^m(E)\end{align*}
for $s$ small enough, proving~\eqref{lb}.

Combining~\eqref{u.b.for.h} with~\eqref{lb},
                                                                                   
we arrive at
\begin{equation}\label{final.bd}
\log\biggl(\frac{\gamma}{\mathcal H^m(E)}\biggr)\le -\frac{m}{2}\log T+C+C\log\biggl(\frac{1+\sqrt{T}\,\delta}{\gamma}\biggr).
\end{equation}
This implies that $\delta>0$, since otherwise we would reach a contradiction for $T$ large enough. Noting that we can take $E$ as above such that $\gamma=\mu_0(\mathbb R^d)=\mu(E)\in[1/2,1]$,
we can then take $T:=\delta^{-2}$, which yields
\[\log\mathcal H^m(E)+\log\delta^m\ge-C\]
for some constant $C(d,\lambda,\Lambda)$,
and hence
\[\mu(\mathbb R^d)=1\le C\mathcal H^m(E)^{1/m}\delta=C\mathcal H^m(E)^{1/m}|\operatorname{div} \sigma|(\mathbb R^d),\]
as desired.

By homogeneity, this proves the theorem for a compactly supported $\mu$,
while the general case follows by a standard cut-off argument.

Without loss of generality, we can assume that $\mathcal A=\operatorname{div}$, as in the statement of Theorem~\ref{thm:support}.

By approximation, this holds for any $E\subseteq M$ with $\mathcal H^m(E)\in(0,\infty)$ (assuming without loss of generality that $M$ is $m$-rectifiable).

\le\int_{E_u}\theta\;\mathrm{d}\mathcal H^m
\le\mu(E_u)
\le C\frac{u^{-\varepsilon}}{r^{m\varepsilon}}(1+r\delta),\]
where we used the fact that $\mathcal H^m(E_u)\le u^{-1}\mu(E_u)\le u^{-1}$.
Taking $p:=1+\varepsilon$, this is the same as

for any given $u>0$, as desired.
