<!-- definition-1: body_md -->
An irreducible normal $G$-variety $X$ is called \emph{spherical} if the Borel subgroup~$B$ acts on~$X$ with an open orbit.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
A $G$-orbit $Y$ in a spherical $G$-variety $X$ is called \emph{toroidal} if it is not contained in any color. The variety itself is called \emph{toroidal} if all its $G$-orbits are toroidal.
<!-- end definition-2 -->
<!-- theorem--1: body_md -->
Let $X$ be an affine toroidal spherical $G$-variety, and let $x,x'$ be two points on $X$. Then the following conditions are equivalent:
    \begin{enumerate}
    \item $x$ and $x'$ lie in the same $\operatorname{Aut}^0(X)$-orbit;
    \item $\operatorname{Cl}_x(X)=\operatorname{Cl}_{x'}(X)$;
    \item $S(Gx)=S(Gx')$.
    \end{enumerate}
<!-- end theorem--1 -->
<!-- theorem--2: body_md -->
Let $X$ be a smooth complete toroidal spherical $G$-variety and $x, x'$ be two points on $X$. Then the following conditions are equivalent: \begin{enumerate}
        \item $x$ and $x'$ lie in the same $\operatorname{Aut}^0(X)$-orbit;
        \item $x$ and $x'$ lie in the same $\operatorname{GB}(X)$-orbit;
        \item $\Gamma(Gx)=\Gamma(Gx')$,
    \end{enumerate}
    where $\operatorname{GB}(X)$ denotes the subgroup of $\operatorname{Aut}^0(X)$ generated by $G$ and all $B$-root subgroups.
<!-- end theorem--2 -->

        \item $X_Y^0 = P\cdot Z_Y \simeq P\times^L Z_Y \simeq P_u \times Z_Y$;
        \item $Z_Y$ is an affine $L$-spherical variety.
    \end{enumerate}
    Moreover, there exists a natural bijection between $G$-orbits in $X$ intersecting $X_Y^0$ and $L$-orbits in $Z_Y$, established as follows: $Y'\mapsto Y'\cap Z_Y$.

<!-- proposition-1: body_md -->
\label{prop:locact}
    Let $X$ be a normal variety and $\varphi \in \operatorname{Aut}(X)$. Then for any $x\in X$ the following equality holds: \begin{equation*}
        \operatorname{Cl}_{\varphi(x)}(X) = \overline{\varphi}(\operatorname{Cl}_x(X)),
    \end{equation*}
    where $\overline{\varphi}$ denotes the image of $\varphi$ in $\operatorname{Aut}(\operatorname{Cl}(X))$, i.e. $\overline{\varphi}([D])=[\varphi(D)]$.

<!-- lemma-2: body_md -->
\label{lem:open}
Let $X$ be a normal variety, let $U\subseteq X$ be an open subset, and let $x\in U$. There is a commutative diagram
\[
\begin{CD}
\operatorname{Cl}(X) @>>> \operatorname{Cl}(X,x)\\
@V{\pi_U}VV @VV{\pi_{U,x}}V\\
\operatorname{Cl}(U) @>>> \operatorname{Cl}(U,x),
\end{CD}
\]
where the vertical arrows send the class of $D$ to the class of $D|_U$. All the arrows are surjective, and $\pi_{U,x}$ is an isomorphism. In particular,
\[
 \operatorname{Cl}_x(X)=\pi_U^{-1}\bigl(\operatorname{Cl}_x(U)\bigr).
\]
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
The horizontal arrows are surjective by definition. Now let us check that $\pi_{U,x}$ is well-defined and is an isomorphism, the rest is obvious.

\emph{Well-definedness.} If $D=\operatorname{div}(f)$ on a neighborhood $W$ of $x$ in $X$, then $D|_U=\operatorname{div}(f)$ on~$U\cap W$.

\emph{Injectivity.} Suppose that $[D|_U]=0$ in $\operatorname{Cl}(U,x)$. Then $D|_U=\operatorname{div}(f)$ in some neighborhood $W\subseteq U$ of $x$, where $f$ is a nonzero rational function on $X$. The divisor $D-\operatorname{div}(f)$ on $X$ has support disjoint from $W$, which yields $[D]=0$ in $\operatorname{Cl}(X,x)$.

\emph{Surjectivity.} This follows from the surjectivity of $\pi_U$.
<!-- end proof-1 -->
<!-- corollary-1: body_md -->
\label{cor:open-kernels}
In the notation of Lemma~\ref{lem:open}, let $D_1,\ldots,D_k$ be the prime divisors lying in the complement of $U$. Then there is an exact sequence
\[
 \bigoplus_{i=1}^k\mathbb Z[D_i]\longrightarrow\operatorname{Cl}_x(X)
 \xrightarrow{\pi_U}\operatorname{Cl}_x(U)\longrightarrow 0.
\]
<!-- end corollary-1 -->
<!-- proof-2: body_md -->
It is well known that $\ker\pi_U=\sum_{i=1}^k\mathbb Z[D_i]$. The statement immediately follows from this fact and Lemma~\ref{lem:open}.

<!-- lemma-3: body_md -->
\label{lem:product}
Let $X$ be a normal variety and let $\pi_X\colon\mathbb A^n\times X\to X$ be the natural projection. For $p\in\mathbb A^n$ and $x\in X$, there is a commutative diagram
\[
\begin{CD}
\operatorname{Cl}(X) @>{\pi_X^*}>> \operatorname{Cl}(\mathbb A^n\times X)\\
@VVV @VVV\\
\operatorname{Cl}(X,x) @>{\pi_X^*}>> \operatorname{Cl}(\mathbb A^n\times X,(p,x)),
\end{CD}
\]
in which both horizontal arrows are isomorphisms. In particular,
\[
 \pi_X^*\bigl(\operatorname{Cl}_x(X)\bigr)=\operatorname{Cl}_{(p,x)}(\mathbb A^n\times X).
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
It is sufficient to prove the statement for $n=1$.

It is well known that the upper arrow is an isomorphism. We now check the lower arrow.

\emph{Well-definedness.} If $D$ is principal on a neighborhood $U$ of $x$, then $\pi_X^*D$ is principal on~$\mathbb A^1\times U$.

\emph{Surjectivity.} This follows from the surjectivity of the upper arrow and the two vertical arrows.

\emph{Injectivity.} Suppose that $\pi_X^*[D]=0$ in $\operatorname{Cl}(\mathbb A^1\times X,(p,x))$. Then, for a neighborhood $W$ of $(p,x)$ and a nonzero rational function $f\in\mathbb K(\mathbb A^1\times X)$, we have
\[
 \pi_X^*D=\operatorname{div}(f)+E,
 \qquad \operatorname{Supp} E\cap W=\varnothing.
\]

Consider the fiber $X_p=\{p\}\times X$. Neither $\pi_X^*D$ nor $E$ contains $X_p$ as a prime component. For $E$ it follows from $W \cap X_p \neq \varnothing$. Thus $f$ has no zeros or poles along $X_p$ and hence~$f|_{X_p} \in \mathbb K(X)^\times$. Restricting everything to~$X_p$, we obtain $D = \operatorname{div}(f|_{X_p}) + E|_{X_p}$, which yields $x \notin \operatorname{Supp}(E|_{X_p})$. Therefore, $D$ is principal in some neighborhood of $x$ and hence~$[D]=0$ in $\operatorname{Cl}(X, x)$.
<!-- end proof-3 -->
<!-- lemma-4: body_md -->
\label{lem:trivial}
For a normal variety $X$, the group $\operatorname{Aut}^0(X)$ acts trivially on $\operatorname{Cl}(X)$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Let $\varphi$ be an automorphism lying in $\operatorname{Aut}^0(X)$. Consider the rational curve $C$ giving a family of automorphisms containing $\varphi$ and $\operatorname{id}_X$. Replacing $C$ with its normalization and removing a finite number of points we can assume that $C$ is an open subset of $\mathbb A^1$.

    Now we have the mapping $\psi:C\times X\to X$, $C = \mathbb A^1\setminus\{a_1,\dotsc,a_s\}$, $\psi_{c_1} = \operatorname{id}_X$, $\psi_{c_2}=\varphi$. Consider the projection $\pi_X:C\times X\to X$ and the isomorphism $\Phi:C\times X \to C\times X$ defined as follows: \begin{equation*}
        \Phi(c,x) = (c, \psi_c(x)).
    \end{equation*}

    Now take any divisor $D$ on $X$ and put $\widetilde D = \pi_X^*D$, $\overline{D} = \Phi^*(\widetilde D)$. Since $\operatorname{Cl}(C\times X) \simeq \operatorname{Cl}(X)$, we obtain $\overline{D} \sim \pi_X^* E$ for some divisor $E$ on $X$, i.e. \begin{equation*}
        \overline{D} = \pi_X^*E + \operatorname{div}(f(c,x)), \qquad f\in\mathbb K(C\times X)^\times.
    \end{equation*}

    For any $c\in C$, neither $\overline{D}$ nor $\pi_X^*E$ has $\{c\}\times X$ as a prime component. Hence $f$ has neither zeros nor poles along $\{c\}\times X$, and $f|_{\{c\}\times X}\in\mathbb K(X)^\times$. This means that we can restrict this equality to the fibers with $c = c_1$ and $c=c_2$, which yields~$D\sim E$ and $\varphi^*D \sim E$. Thus~$\varphi^*D\sim D$ for every divisor~$D$. Applying this to~$\varphi(D)$ gives~$\varphi(D)\sim D$. This finishes the proof.

<!-- proposition-2: body_md -->
\label{prop:local-spherical}
Let $Y$ be a toroidal $G$-orbit in a spherical variety $X$. Then
\[
 \operatorname{Cl}_y(X)=S(Y)\qquad\text{for every }y\in Y.
\]
<!-- end proposition-2 -->
<!-- proof-6: body_md -->
We use the notation of Section~\ref{sec:prelim}. By Corollary~\ref{cor:locclconst}, the kernel $\operatorname{Cl}_y(X)$ is constant along $Y$, since the image of $G$ in $\operatorname{Aut}(X)$ lies in $\operatorname{Aut}^0(X)$. We may therefore choose $y\in Y\cap X_Y^0$ and use Theorem~\ref{LST}. Write $y=(u,z)$, where $z$ belongs to the unique closed torus orbit of $Z_Y$ which we denoted by~$O_Y$. Every torus-invariant prime divisor on $Z_Y$ contains that orbit and hence $S_{T'}(O_Y) = 0$. Theorem~\ref{thm:toric} gives $\operatorname{Cl}_z(Z_Y)=0$, and Lemma~\ref{lem:product} gives $\operatorname{Cl}_y(X_Y^0)=0$ since $X_Y^0 \simeq P_u \times Z_Y$ and $P_u$ is isomorphic to affine space as an algebraic variety.

The prime divisors lying in $X\setminus X_Y^0$ are precisely the $B$-stable prime divisors not containing $Y$. Indeed, the prime divisors contained in $X\setminus X_Y$ are exactly the $G$-stable prime divisors not containing $Y$, and all the colors are removed on passing to $X_Y^0$ since $Y$ is toroidal. Putting $U = X_Y^0$ in Corollary~\ref{cor:open-kernels} we obtain $\operatorname{Cl}_y(X) = S(Y)$.

<!-- lemma-5: body_md -->
\label{lem:replica}
Let $X$ be an irreducible affine variety and $0\ne f\in\mathbb K[X]$. Then for every $\mathbb G_a$-subgroup $H\subseteq\operatorname{Aut}(X_f)$ there is a $\mathbb G_a$-subgroup $H'\subseteq\operatorname{Aut}(X)$ such that $H'x=Hx$ for all $x\in X_f$.
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
Let $\delta$ be the locally nilpotent derivation of $\mathbb K[X_f]=\mathbb K[X]_f$ corresponding to $H$. We have $\delta(f)=0$ since $f$ is invertible on $X_f$. Therefore, for big enough $N$, $f^N\delta$ is a locally nilpotent derivation of $\mathbb K[X]$ (this is called \emph{a replica of $\delta$}; see~\cite{Fr}). The corresponding $\mathbb G_a$-subgroup $H'\subseteq\operatorname{Aut}(X)$ has the same orbits on $X_f$ as $H$.

<!-- theorem-3: body_md -->
\label{thm:affine}
Let $X$ be an affine toroidal spherical $G$-variety and let $x,x'\in X$. The following conditions are equivalent:
\begin{enumerate}
\item $x$ and $x'$ lie in the same $\operatorname{Aut}^0(X)$-orbit;
\item $\operatorname{Cl}_x(X)=\operatorname{Cl}_{x'}(X)$;
\item $S(Gx)=S(Gx')$.
\end{enumerate}
<!-- end theorem-3 -->
<!-- proof-8: body_md -->
The implication $(1)\Rightarrow(2)$ follows from Corollary~\ref{cor:locclconst} and the equivalence $(2)\Leftrightarrow(3)$ follows from Proposition~\ref{prop:local-spherical}.

$(2)\Rightarrow(1)$. We remind the reader, that there is only one closed $G$-orbit $Y$ on $X$ because the categorical quotient $X/\!\!/G$ is one point. Consider the principal open~$B$-chart $X_Y^0=X_{f_Y}\simeq P_u\times Z_Y$. Every $G$-orbit meets this chart since $X$ is toroidal. Hence acting by elements of $G$ we may assume that $x$ and $x'$ lie in $X_Y^0$. Then, acting by elements of $P_u$, we may assume
\[
 x=(e,z),\qquad x'=(e,z')
\]
for some $z,z'\in Z_Y$. These operations preserve condition (2). Lemmas~\ref{lem:open} and~\ref{lem:product} yield the equality $\operatorname{Cl}_z(Z_Y)=\operatorname{Cl}_{z'}(Z_Y)$. By Theorem~\ref{thm:toric}, there is an element $\varphi \in\operatorname{AT}(Z_Y)$ sending $z$ to $z'$.

Write $\varphi=g_k\circ\dots\circ g_1$, where each $g_i$ lies either in the acting torus of $Z_Y$ or in a root subgroup $H_i\subseteq\operatorname{Aut}(Z_Y)$. We can do this by $(1)\Leftrightarrow(2)$ in Theorem~\ref{thm:toric}. Put~$z_0=z$ and~$z_i=g_i(z_{i-1})$, so that $z_k=z'$. We claim that $(e,z_{i-1})$ and $(e,z_i)$ lie in the same~$\operatorname{Aut}^0(X)$-orbit for every $i$. If $g_i$ lies in the torus, then it lifts to an element of the Levi subgroup $L$, since $L$ maps onto this torus, and this element sends $(e,z_{i-1})$ to $(e,z_i)$. If~$g_i\in H_i$, then~$H_i$ acts on $X_Y^0 = P_u\times Z_Y$ trivially on the first factor. Since $X_Y^0=X_{f_Y}$ is a principal open subset, Lemma~\ref{lem:replica} gives a $\mathbb G_a$-subgroup $H_i'\subseteq\operatorname{Aut}^0(X)$ with the same orbits on $X_Y^0$ as $H_i$. In particular, $(e,z_{i-1})$ and $(e,z_i)$ lie in the same $H_i'$-orbit. Therefore~$x$ and~$x'$ lie in the same $\operatorname{Aut}^0(X)$-orbit. This finishes the proof.
<!-- end proof-8 -->
<!-- remark-1: body_md -->
Note that if $x$ is a smooth point on $X$, then $\operatorname{Cl}_x(X)=\operatorname{Cl}(X)$ since any divisor is principal in some neighborhood of $x$. Theorem~\ref{thm:affine} therefore says that all smooth points lie in the same $\operatorname{Aut}^0(X)$-orbit. This is consistent with Shafarevich's result on the flexibility of affine spherical varieties.

<!-- example-1: body_md -->
\label{ex:affine}
Consider
\[
 Q=(\mathbb P^1\times\mathbb P^1)\setminus\Delta\simeq\operatorname{SL}_2/T_0,
 \qquad
 V=\operatorname{Spec}\mathbb K[u,v,w]/(uw-v^2),
\]
where $\Delta$ is the diagonal in $\mathbb P^1\times\mathbb P^1$ and $T_0\simeq\mathbb G_m\subseteq\operatorname{SL}_2$ is a maximal torus. We note that $Q$ is affine since, for example, $\Delta$ is an ample divisor or by Matsushima's criterion from the representation $Q=\operatorname{SL}_2/T_0$. The variety $V\simeq\mathbb A^2/(\mathbb Z/2\mathbb Z)$ is the standard quadratic cone and it is normal. Denote its acting torus by $\mathbb T$ and put
\[
 X=Q\times V,\qquad G=\operatorname{SL}_2\times\mathbb T.
\]
The variety $X$ is $G$-spherical but it is not toric even for a different torus action. It follows from the fact that the singular locus of $X$ is $Q\times\{0\}\simeq Q$. If $X$ were toric with respect to some torus $\widetilde T$, then $Q$ would be a normal toric variety. But $Q$ is not toric since it is smooth and $\operatorname{Cl}(Q) \simeq \mathbb Z$. Now we remind the reader that every smooth affine toric variety is isomorphic to $\mathbb A^k\times\mathbb G_m^l$ and hence has trivial divisor class group.

Now note that $X$ is toroidal since any color has the form $D\times V$, where $D$ is a divisor on~$Q$, and all $G$-orbits have the form $Q\times Y$, where $Y$ is a $\mathbb T$-orbit on $V$. One can easily obtain \begin{equation*}
    \operatorname{Cl}(X) = \mathbb Z[H]\oplus (\mathbb Z/2\mathbb Z)[E],
\end{equation*} where $H = (\mathbb A^1\times \{\infty\})\times V$ and $E = Q \times \{u=v=0\}$. We also have
\[
 S(Q\times\{0\})=\mathbb Z[H],
 \qquad
 S(Q\times Y)=\mathbb Z[H]\oplus(\mathbb Z/2\mathbb Z)[E]
 \quad(Y\ne\{0\}).
\]
Hence Theorem~\ref{thm:affine} gives exactly two $\operatorname{Aut}^0(X)$-orbits:
\[
 O_1 = Q\times\{0\},\qquad O_2 = Q\times(V\setminus\{0\}).
\]

Note that they are also two $\operatorname{Aut}(X)$-orbits since $O_1$ is the singular locus of~$X$.

<!-- proposition-3: body_md -->
\label{prop:intrinsic}
For a $G$-orbit $Y\subseteq X$ and a point $y\in Y$ the following equality holds:
\begin{equation}\label{eq:intrinsic}
 \Gamma(Y)=\Gamma(y), \quad \text{where}\quad \Gamma(y)=\{[\mathcal L]\in\operatorname{Pic}(X)\mid
 \exists s\in H^0(X,\mathcal L): \ s(y)\ne0\}.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-10: body_md -->
The set $\Gamma(y)$ is a monoid since if two sections $s\in H^0(X, \mathcal L)$ and $s' \in H^0(X, \mathcal L')$ are nonzero at $y$, then $s\otimes s' \in H^0(X, \mathcal L\otimes\mathcal L')$ is also nonzero at $y$.

Let $D$ be a $B$-stable prime divisor not containing $Y$. Then there exists $g\in G$ such that $gy\notin D$. By Lemma~\ref{lem:trivial} we have $[D]=[g^{-1}D]$. The canonical section of $\mathcal O_X(g^{-1}D)$ is nonzero at $y$ and hence $[D]\in\Gamma(y)$. This proves the inclusion $\Gamma(Y)\subseteq\Gamma(y)$.

Conversely, take a line bundle $\mathcal L$ with a section $s$ such that $s(y)\ne 0$. We may assume that $\mathcal L$ admits a $G$-linearization by replacing $G$ with its finite cover. This does not change the orbits. The restriction homomorphism
\[
 r\colon H^0(X,\mathcal L)\longrightarrow H^0(Y,\mathcal L|_Y)
\]
is a nonzero homomorphism of $G$-modules because $r(s)$ is nonzero. Since $H^0(X,\mathcal L)$ is finite-dimensional ($X$ is complete) and $G$ is reductive, it has a simple submodule $V$ such that the restriction $r|_V$ is injective. Let $s_0\in V$ be a nonzero highest weight vector. Then~$r(s_0)\ne0$ and hence the support of the effective $B$-stable divisor $E=\operatorname{div}(s_0)$ does not contain $Y$. Consequently, every prime component of $E$ belongs to $\mathcal D^B(Y)$ and
\[
 [\mathcal L]=[E]\in\Gamma(Y).
\]
This proves the reverse inclusion.

<!-- corollary-4: body_md -->
\label{cor:necessary}
If two $G$-orbits $Y$ and $Y'$ lie in the same $\operatorname{Aut}^0(X)$-orbit, then $\Gamma(Y)=\Gamma(Y')$.
<!-- end corollary-4 -->
<!-- proof-11: body_md -->
The group $\operatorname{Aut}^0(X)$ acts trivially on $\operatorname{Pic}(X)$ and for any $\varphi \in \operatorname{Aut}(X)$ we have~$\Gamma(\varphi(y)) = \overline{\varphi}(\Gamma(y))$, where $\overline{\varphi}$ denotes the image of $\varphi$ in $\operatorname{Aut}(\operatorname{Pic}(X))$. Let~$y \in Y$,~$y'\in Y'$ and $\psi\in\operatorname{Aut}^0(X)$ such that $\psi(y)=y'$. We have \[
\Gamma(Y')=\Gamma(y')=\Gamma(\psi(y))=\overline{\psi}(\Gamma(y))=\Gamma(y)=\Gamma(Y).
\]
This finishes the proof.

<!-- theorem-4: body_md -->
\cite[Proposition 4.1.1]{BB}\label{thm:BB}
Let $X$ be a smooth complete toroidal spherical variety and let $D_1,\dotsc,D_k$ be all the $G$-stable prime divisors on $X$. There is an exact sequence of $G$-modules
\begin{equation}\label{eq:BB}
 0\longrightarrow H^0(X,\mathcal S_X)
 \longrightarrow H^0(X,\mathcal T_X)
 \xrightarrow{\nu}
 \bigoplus_{i=1}^k H^0(D_i,\mathcal O_{D_i}(D_i))
 \longrightarrow0,
\end{equation} where $\mathcal T_X$ is the tangent sheaf of $X$ and $\mathcal S_X$ is the subsheaf of $\mathcal T_X$ consisting of vector fields tangent to every $D_i$. The map $\nu$ takes the normal components along the divisors~$D_1,\dotsc,D_k$.

<!-- lemma-6: body_md -->
\label{lem:transition}
Let $\sigma\in\mathcal F$ and $\rho\in\sigma(1)$, and let $\sigma'$ be the face of $\sigma$ generated by $\sigma(1)\setminus\{\rho\}$. Suppose that condition~\eqref{eq:deletion} holds, and let $m\in M$ satisfy~\eqref{eq:root1}--\eqref{eq:root4}. Such $m$ exists by Proposition~\ref{prop:inequalities}. Then there is a $B$-root subgroup $U_m\subseteq\operatorname{Aut}^0(X)$ of weight $m$ such that the orbit $U_mx$ meets $Y_{\sigma'}$ for some point $x\in Y_\sigma$.
In particular, $Y_\sigma$ and $Y_{\sigma'}$ lie in the same $\operatorname{GB}(X)$-orbit and $\Gamma(Y_\sigma)=\Gamma(Y_{\sigma'})$.
<!-- end lemma-6 -->
<!-- proof-13: body_md -->
\emph{Step 1: a $B$-semiinvariant section of $\mathcal O_X(D_\rho)$.} Since $D_\rho$ is $G$-stable, the line bundle~$\mathcal O_X(D_\rho)$ has a canonical $G$-linearization as a subsheaf of $\mathbb K(X)$. Its canonical section $s_\rho$, represented by $1$, is $G$-invariant. Consequently
\[
 s=f_m s_\rho
\]
is a rational $B$-semiinvariant section of $\mathcal O_X(D_\rho)$ of weight $m$, and $m\ne0$ since $\langle v_\rho,m\rangle=-1$. By~\eqref{eq:principal} and conditions~\eqref{eq:root1}--\eqref{eq:root4}, the divisor of $s$ equals
\[
 E=D_\rho+\operatorname{div}(f_m)=\sum_{\eta\in\mathcal F(1)\setminus\sigma(1)}\langle v_\eta,m\rangle D_\eta+\sum_{F\in\mathcal D}\langle v_F,m\rangle F.
\]
It is effective, so $s\in H^0(X,\mathcal O_X(D_\rho))$. The colors do not contain $Y_\sigma$ since $X$ is toroidal, and the divisors $D_\eta$ with $\eta\in\mathcal F(1)\setminus\sigma(1)$ also do not contain $Y_\sigma$. Therefore we can choose a point $x\in Y_\sigma\setminus\operatorname{Supp} E$. We have $s(x)\ne0$ and the restriction
\[
 \overline s=s|_{D_\rho}\in H^0(D_\rho,\mathcal O_{D_\rho}(D_\rho))
\]
is nonzero at $x$ since $Y_\sigma\subseteq D_\rho$.

\emph{Step 2: a $B$-eigenvector in $H^0(X,\mathcal T_X)$.} Consider the vector $\tilde s$ in the right-hand side of~\eqref{eq:BB} whose $\rho$-component is $\overline s$ and whose other components are zero. It is a $B$-eigenvector of weight $m$, since the restriction to the $G$-stable divisor $D_\rho$ is $G$-equivariant. Since $G$ is reductive, the surjection $\nu$ has a $G$-equivariant splitting. Therefore, there is a $B$-eigenvector
\[
 \xi_m\in H^0(X,\mathcal T_X)=\operatorname{Lie}\operatorname{Aut}^0(X)
\]
of weight $m$, such that $\nu(\xi_m)=\tilde s$. The normal components of $\xi_m$ along all $D_{\rho'}$ with $\rho'\ne\rho$ are zero, so $\xi_m$ is tangent to these divisors. The normal component of $\xi_m$ along $D_\rho$ is $\overline s$, and $\overline s(x)\ne0$ which yields $\xi_m(x)\notin T_xD_\rho$.

\emph{Step 3: the $B$-root subgroup.} Since $\xi_m$ is a $B$-eigenvector, we have $\operatorname{Ad}(t)\xi_m=m(t)\xi_m$ for~$t\in T$, and $m(T)=\mathbb K^\times$ since $m\ne0$. This means that the finite set of eigenvalues of~$\xi_m$ in a faithful representation of $\operatorname{Aut}^0(X)$ is invariant under multiplication by~$\mathbb K^\times$, and hence all its eigenvalues are zero. Thus, $\xi_m$ is nilpotent and the group
\[
 U_m = \{\exp(t\xi_m) \mid t\in\mathbb K\}\subseteq\operatorname{Aut}^0(X)
\]
is a $B$-root subgroup.

\emph{Step 4: the transition.} Consider the following group:
\[
 K = \Big(\bigcap_{\rho'\in\mathcal F(1)\setminus\{\rho\}} \operatorname{Stab}_{\operatorname{Aut}^0(X)}(D_{\rho'})\Big)^0\subseteq \operatorname{Aut}^0(X).
\]
Its Lie algebra is
\[
 \operatorname{Lie} K = \{\zeta \in H^0(X, \mathcal T_X) \mid \text{$\zeta$ is tangent to $D_{\rho'}$ for $\rho'\ne\rho$}\}.
\]
By Step~2 we have $\xi_m\in\operatorname{Lie} K$, hence $U_m\subseteq K$.

The orbit $U_mx$ is not contained in $D_\rho$. Otherwise, $\xi_m$ would be tangent to $D_\rho$ which is a contradiction with Step 2. Choose $y\in U_mx\setminus D_\rho$. For every $\rho'\ne\rho$ we have
\[
 y \in D_{\rho'} \quad \Longleftrightarrow \quad  x\in D_{\rho'} \quad\Longleftrightarrow\quad \rho'\in\sigma(1),
\]
since $U_m\subseteq K$. Hence $y$ lies in $D_{\rho'}$ if and only if $\rho'\in\sigma(1)\setminus\{\rho\}=\sigma'(1)$. It remains to say that a $G$-orbit is uniquely determined by the set of $G$-stable prime divisors containing it, so $y\in Y_{\sigma'}$.

Finally, $Y_\sigma$ and $Y_{\sigma'}$ lie in the same $\operatorname{GB}(X)$-orbit, since $\operatorname{GB}(X)$ contains $G$. The equality~$\Gamma(Y_\sigma)=\Gamma(Y_{\sigma'})$ follows from Corollary~\ref{cor:necessary} since $\operatorname{GB}(X)\subseteq\operatorname{Aut}^0(X)$.
<!-- end proof-13 -->
<!-- theorem-5: body_md -->
\label{thm:main-complete}
    Let $X$ be a smooth complete toroidal spherical $G$-variety, and let $x, x'\in X$. Then the following conditions are equivalent:
    \begin{enumerate}
        \item $x$ and $x'$ lie in the same $\operatorname{Aut}^0(X)$-orbit;
        \item $x$ and $x'$ lie in the same $\operatorname{GB}(X)$-orbit;
        \item $\Gamma(Gx)=\Gamma(Gx')$.
    \end{enumerate}
<!-- end theorem-5 -->
<!-- proof-14: body_md -->
The implication $(2)\Rightarrow(1)$ is obvious since $\operatorname{GB}(X)\subseteq\operatorname{Aut}^0(X)$. The implication $(1)\Rightarrow(3)$ is Corollary~\ref{cor:necessary}. We need to prove $(3)\Rightarrow(2)$.

Since $\operatorname{GB}(X)$ contains the image of $G$, it suffices to show that the $G$-orbits $Gx$ and $Gx'$ lie in the same $\operatorname{GB}(X)$-orbit. We have $Gx=Y_\sigma$ and $Gx'=Y_\tau$ for some cones $\sigma,\tau\in \mathcal F$, and $\Gamma(Y_{\sigma})=\Gamma(Y_{\tau})$ by assumption.

Take $\rho\in\sigma(1)\setminus\tau(1)$. The divisor $D_\rho$ does not contain $Y_\tau$ and hence
\[
 [D_\rho]\in\Gamma(Y_\tau)=\Gamma(Y_\sigma).
\]
By Lemma~\ref{lem:transition}, the orbit $Y_\sigma$ and the orbit $Y_{\sigma'}$, where $\sigma' = \operatorname{cone}(\sigma(1)\setminus \rho)$, lie in the same $\operatorname{GB}(X)$-orbit and $\Gamma(Y_{\sigma'})=\Gamma(Y_\sigma)=\Gamma(Y_\tau)$. Hence we can repeat this process for all rays in $\sigma(1)\setminus\tau(1)$. In this way we connect $Y_\sigma$ with $Y_{\sigma\cap\tau}$. Indeed, $\sigma\cap\tau$ is a common face of $\sigma$ and $\tau$, and its rays are exactly their common rays~$\sigma(1)\cap\tau(1)$. We use the fact that all cones are regular since $X$ is smooth. Applying the same argument to $\tau$, we connect $Y_\tau$ with $Y_{\sigma\cap\tau}$. Therefore $Y_\sigma$ and $Y_\tau$ lie in the same $\operatorname{GB}(X)$-orbit.

<!-- corollary-5: body_md -->
\label{cor:full-complete}
Let $X$ be a smooth complete toroidal spherical variety. Two $G$-orbits $Y,Y'$ are contained in the same $\operatorname{Aut}(X)$-orbit if and only if there is an automorphism $\varphi\in\operatorname{Aut}(X)$ such that
\[
 \overline{\varphi}(\Gamma(Y))=\Gamma(Y'),
\]
where $\overline{\varphi}$ denotes the image of $\varphi$ in $\operatorname{Aut}(\operatorname{Pic}(X))$.
<!-- end corollary-5 -->
<!-- corollary-6: body_md -->
\label{cor:graph}
The $\operatorname{Aut}^0(X)$-orbits in $X$ correspond to the connected components of $\mathcal{G}_\mathcal F$, i.e. two $G$-orbits $Y_\sigma$ and $Y_\tau$ lie in the same $\operatorname{Aut}^0(X)$-orbit if and only if $\sigma$ and $\tau$ lie in the same connected component of $\mathcal G_\mathcal F$. If two cones~$\sigma$ and~$\tau$ are joined by an edge, then the corresponding orbits~$Y_{\sigma}$ and~$Y_\tau$ can be connected by a $B$-root subgroup.
<!-- end corollary-6 -->
<!-- proof-15: body_md -->
If $\sigma$ and $\sigma'$ are joined by an edge, then $Y_\sigma$ and $Y_{\sigma'}$ are connected by a $B$-root subgroup by Lemma~\ref{lem:transition}. In particular, they lie in the same $\operatorname{Aut}^0(X)$-orbit. Conversely, if $Y_\sigma$ and $Y_\tau$ lie in the same $\operatorname{Aut}^0(X)$-orbit, then $\Gamma(Y_\sigma)=\Gamma(Y_\tau)$ by Theorem~\ref{thm:main-complete}. The proof of the implication $(3)\Rightarrow(2)$ in Theorem~\ref{cor:full-complete} gives the path in $\mathcal G_\mathcal F$ from $\sigma$ and from $\tau$ to $\sigma\cap\tau$.

 X=\{([q],[q'])\in\mathbb P(\operatorname{S}^2\mathbb K^3)\times\mathbb P(\operatorname{S}^2\mathbb K^{3*})\mid q\cdot q'\in\mathbb K\cdot I\},
\]
where $I$ is the identity matrix. This variety is described in detail in~\cite[Example 17.3]{Tim11}.

The open $G$-orbit is $\operatorname{SL}_3/\operatorname{SO}_3$, and its weight lattice is $M=\langle 2\alpha_1,2\alpha_2\rangle$, where $\alpha_1,\alpha_2$ are the simple roots of $\operatorname{SL}_3$. \begin{itemize}
\item for $\sigma_{12}$ and the ray of $E_1$, conditions~\eqref{eq:root1} and~\eqref{eq:root2} give $a=-1$ and $b=0$, so $\langle v_{D_2},m\rangle=-1$;
\item for $\sigma_{12}$ and the ray of $E_2$, we get $a=0$ and $b=-1$, so $\langle v_{D_1},m\rangle=-1$;

\item for $\sigma_2$, we get $b=-1$ and $a\geq0$, so $\langle v_{D_1},m\rangle=-2a-1<0$;
\item the cone $\sigma_0$ has no rays.
\end{itemize}
These calculations show that in each case condition~\eqref{eq:root4} fails. Hence the graph $\mathcal G_\mathcal F$ has no edges, and by Corollary~\ref{cor:graph} the $G$-orbits coincide with the $\operatorname{Aut}^0(X)$-orbits for the space of complete conics.
