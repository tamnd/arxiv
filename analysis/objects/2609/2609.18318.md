<!-- lemma-1: body_md -->
\label{le:sspsum}
Let $a\in{\mathbb F}_{p}^n$, $a\neq 0$, and assume that $T$ is a subspace of ${\mathbb F}_{p}^n$ such that $T\not\subset a^{\perp}$. Then 
\[\sum_{x\in T}\zeta_p^{-\langle a,x\rangle}=0\enspace.\]
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{le:Ny}
Let $n$ be a positive integer and $p$ a prime. Suppose that $f:{\mathbb F}_{p}^{2n}\mapsto{\mathbb F}_{p}$ is a bent function and for $j\in{\mathbb F}_{p}$ denote $D_j:= 
f^{-1}(j)$. Then there exists $k\in{\mathbb F}_{p}$ such that 
\begin{align*} 
\lvert D_k\rvert&=p^{2n-1}\pm(p-1)p^{n-1},\\
\lvert D_\ell\rvert&=p^{2n-1}\mp p^{n-1}~\mbox{for}~\ell\in{\mathbb F}_{p}\setminus\{k\}\enspace.
\end{align*}
Here the $\pm$ signs are taken correspondingly. Moreover, for regular bent functions, the upper signs are taken.
<!-- end lemma-2 -->
<!-- theorem-1: body_md -->
\label{th:deg}
Let $p$ be an odd prime and $n$ be a positive integer. Assume $f:{\mathbb F}_{p}^{2n}\mapsto{\mathbb F}_{p}$ is such that $f(0,\dots,0)=0$ and there exists an 
$m$-dimensional subspace $U$ of ${\mathbb F}_{p}^{2n}$ such that $f$ is a nonzero constant on $U^*$. Then $f$ has algebraic degree $\deg(f)\geq(p-1)m$.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Let $f(x_1,\ldots,x_{2n})\equiv c$ on $U^*$ for some $c\in{\mathbb F}_{p}^*$. By a coordinate transformation $A$ of ${\mathbb F}_{p}^{2n}$, we have 
\[A(U)=\{(\alpha_1,\ldots,\alpha_m,0,\ldots,0)\}={\mathbb F}_{p}^m\times\{(0,\ldots,0)\}=:H\enspace.\]
Denoting $g=f\circ A^{-1}$ we obtain $g(x_1,\ldots,x_{2n})\equiv c$ on $H^*$. Also denote 
\[g'(x_1,\ldots,x_{m}):= g(x_1,\ldots,x_{m},0,\ldots,0).\]
Note that $\deg(f)=\deg(g)\geq\deg(g')$. Hence, it suffices to show that $\deg(g')=(p-1)m$. By Lagrange interpolation, we write 
\[g'(x_1,\ldots,x_{m})=\sum_{(\alpha_1,\ldots,\alpha_{m})\in{\mathbb F}_{p}^m}g'(\alpha_1,\ldots,\alpha_{m})\prod_{i=1}^m (1-(x_i-\alpha_i)^{p-1})\enspace.\]
Since $g'(0,\ldots,0)=0$ and $g'(x_1,\ldots,x_{m})\equiv c$ on $H^*$, we have 
\[g'(x_1,\ldots,x_{m})=(p^m-1)x_1^{p-1}\cdots x_m^{p-1}+h(x_1,\ldots,x_m)\]
for some $h\in{\mathbb F}_{p}[x_1,\ldots,x_{m}]$ of degree less than $(p-1)m$. Hence, $g'$ has algebraic degree $(p-1)m$ and $f$ has algebraic degree 
at least $(p-1)m$.\qed
<!-- end proof-1 -->
<!-- corollary-1: body_md -->
Let $p$ be an odd prime and $n$ be a positive integer. Let $f:{\mathbb F}_{p}^{2n}\mapsto{\mathbb F}_{p}$ be a function having algebraic degree less than $(p-1)m$ 
for some $m\leq 2n$. Then, $f$ cannot be constant on $U^*$ for any $m$-dimensional subspace $U$ of ${\mathbb F}_{p}^{2n}$.
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{th:PSminus}
Let $p$ be an odd prime and $m,n$ be positive integers such that $m<2n$ and $p^n>3$. Assume $\mathcal S$ is a partial $m$-spread of ${\mathbb F}_{p}^{2n}$ and 
$f:{\mathbb F}_{p}^{2n}\mapsto{\mathbb F}_{p}$ is such that for each $T\in\mathcal S$, $f$ is constant on $T^*$. Moreover, suppose that 
$f^{-1}(0)={\mathbb F}_{p}^{2n}\setminus\bigcup_{T\in\mathcal S}T^*$. Then $f$ is bent if and only if $m\mid n$ and $f^{-1}(j)\cup\{0\}$ is a union of exactly 
$\frac{p^{n-1}(p^n-1)}{p^m-1}$ elements of $\mathcal S$ for every $j\in{\mathbb F}_{p}^*$ and for every nonzero $a\in{\mathbb F}_{p}^{2n}$, $a^{\perp}$ contains either 
\begin{align}
 \label{eq:bentcond1}
&N\ \mbox{subspaces}\ T\subseteq f^{-1}(j)\ \mbox{for every}\ j\in{\mathbb F}_{p}^*\ \mbox{or}\\
 \label{eq:bentcond2}
&\begin{cases}
N+p^{n-m}\ \mbox{subspaces}\ T\subseteq f^{-1}(t)~\mbox{for some}\ t\in {\mathbb F}_{p}^*\ \mbox{and}\\
N\ \mbox{subspaces}\ T\subseteq f^{-1}(j)\ \mbox{for remaining}\ j\in{\mathbb F}_{p}^*\setminus\{t\}\enspace,
\end{cases}
\end{align}
where $N=\frac{p^{n-1}(p^{n-m}-1)}{p^m-1}$. Also, $\mathcal W_f(0)=p^n$ and if \eqref{eq:bentcond1} holds then $\mathcal W_f(a)=p^n$ and if 
\eqref{eq:bentcond2} holds then $\mathcal W_f(a)=p^n\zeta_p^t$. If $f$ is bent then it is a regular bent functions.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
For every $j\in{\mathbb F}_{p}$, let $D_j:=f^{-1}(j)$. For $j\neq 0$ let also $N_j\geq 0$ be such that $D_j=\bigcup_{i=1}^{N_j}\mathcal S_{ji}^*$, where 
$\mathcal S_{ji}$ are pairwise distinct elements of $\mathcal S$ (if $N_j=0$ then $D_j=\emptyset$). Therefore, 
\begin{align*}
\lvert D_j\rvert&=N_j(p^m-1)\ \mbox{for}\ j\neq 0\\
\lvert D_0\rvert&=p^{2n}-(p^m-1)\sum_{j=1}^{p-1}N_j\enspace. 
\end{align*}
Note that $p-1$ divides $\lvert D_j\rvert$ when $j\neq 0$ and does not divide $\lvert D_0\rvert$.

First, assume that $f$ is bent. Considering Lemma~\ref{le:Ny}, and the mentioned above divisibility, we conclude that $k=0$. By the same 
lemma, $\lvert D_j\rvert=p^{n-1}(p^n\mp 1)$ for $j\neq 0$. For such $j$, $\lvert D_j\rvert$ is divisible by $p^m-1$ and since
$\gcd(p^n+1,p^m-1)\neq p^m-1$ (unless $p^n=3$), it follows that $\lvert D_j\rvert=p^{n-1}(p^n-1)$, $m\mid n$, and $f$ is regular. 

Also $N_j=\frac{p^{n-1}(p^n-1)}{p^m-1}$ and 
\begin{align*}
\mathcal W_f(0)&=\sum_{x\in{\mathbb F}_{p}^{2n}}\zeta_p^{f(x)}=\sum_{j=0}^{p-1}\sum_{x\in D_j}\zeta_p^j=\lvert D_0 \rvert+\sum_{j=1}^{p-1}\lvert D_j\rvert\zeta_p^j\\
&=p^{2n}-(p^m-1)\sum_{j=1}^{p-1}N_j+(p^m-1)\sum_{j=1}^{p-1}N_j\zeta_p^j\\
&=p^{2n}-(p-1)p^{n-1}(p^n-1)-p^{n-1}(p^n-1)=p^n\enspace.
\end{align*}

Now we show that \eqref{eq:bentcond1}~or~\eqref{eq:bentcond2} holds. Let $a\in{\mathbb F}_{p}^{2n}$ with $a\neq 0$. Assume that $a^{\perp}$ contains 
$s$ distinct subspaces from $\mathcal S$, say $\mathcal S_{j_1 i_1},\ldots ,\mathcal S_{j_s i_s}$, where $0<j_1,\ldots,j_s<p$ and $1\leq 
i_1,\ldots,i_s\leq\frac{p^{n-1}(p^n-1)}{p^m-1}$. Consider 
\begin{align*}
\mathcal W_f(a)&=\sum_{x\in{\mathbb F}_{p}^{2n}}\zeta_p^{f(x)-\langle a,x\rangle}\\
&=\sum_{x\in D_0}\zeta_p^{-\langle a,x\rangle}+\sum_{j=1}^{p-1}\sum_{x \in D_j}\zeta_p^{f(x)-\langle a,x \rangle}\\
&=-\sum_{j=1}^{p-1}\sum_{x\in D_j}\zeta_p^{-\langle a,x \rangle}+\sum_{j=1}^{p-1}\zeta_p^j\sum_{x\in D_j}\zeta_p^{-\langle a, x\rangle}\\
&=-\sum_{j=1}^{p-1}\sum_{i=1}^{\frac{p^{n-1}(p^n-1)}{p^m-1}}\sum_{x\in\mathcal S_{ji}^*}\zeta_p^{-\langle a,x\rangle}+\sum_{j=1}^{p-1}\zeta_p^j \sum_{i=1}^{\frac{p^{n-1}(p^n-1)}{p^m-1}}\sum_{x\in\mathcal S_{ji}^*}\zeta_p^{-\langle a,x\rangle}\\
&=(p-1)\frac{p^{n-1}(p^n-1)}{p^m-1}-sp^m +\frac{p^{n-1}(p^n-1)}{p^m-1}+p^m(\zeta_p^{j_1}+\cdots+\zeta_p^{j_s})\\
&=\frac{p^{n}(p^n-1)}{p^m-1}+p^m(\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-s)\enspace.
\end{align*}
Hear we used Lemma~\ref{le:sspsum} consistently. Since $f$ is a regular bent function, for some $t\in{\mathbb F}_{p}$, we have
\begin{align}
 \label{eq:T1wta}
\nonumber&\frac{p^{n}(p^n-1)}{p^m-1}+p^m(\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-s)=p^n\zeta_p^t\\
\nonumber\implies&\frac{p^{n-m}(p^n-1)}{p^m-1}+\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-s=p^{n-m}\zeta_p^t\\
\implies&\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-p^{n-m}\zeta_p^t=s-\frac{p^{n-m}(p^n-1)}{p^m-1}\enspace.
\end{align}
Now we consider two cases separately, namely when $t=0$ and when $t\in{\mathbb F}_{p}^*$.

\textbf{Case 1.} Let $t=0$. In this case, \eqref{eq:T1wta} reduces to 
\begin{equation}
 \label{eq:T1C1}
\zeta_p^{j_1}+\cdots +\zeta_p^{j_s}=s+p^{n-m}-\frac{p^{n-m}(p^n-1)}{p^m-1}\enspace.
\end{equation}
The LHS of \eqref{eq:T1C1} is an integer if and only if it is of the form $\ell(\zeta_p+\cdots+\zeta_p^{p-1})=-\ell$ for some positive 
integer $\ell$. This implies $s=\ell(p-1)$. Inserting this value in \eqref{eq:T1C1} we obtain 
\[-\ell=\ell(p-1)+p^{n-m}-\frac{p^{n-m}(p^n-1)}{p^m-1}\implies \ell=\frac{p^{n-1}(p^{n-m}-1)}{p^m-1}=N\enspace.\]
We conclude that $a^{\perp}$ contains $N$ distinct $\mathcal S_{ji}^*\subseteq D_j$ for every $j\in{\mathbb F}_{p}^*$. 

\textbf{Case 2.} Let $t\neq 0$. In this case, \eqref{eq:T1wta} reduces to 
\begin{equation}
 \label{eq:T1C2}
\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-p^{n-m}\zeta_p^t=s-\frac{p^{n-m}(p^n-1)}{p^m-1}\enspace.
\end{equation}
Again, the LHS of \eqref{eq:T1C2} is an integer if and only if it is of the form $\ell(\zeta_p+\cdots+\zeta_p^{p-1})=-\ell$ for some 
positive integer $\ell$. This implies that 
\[\zeta_p^{j_1}+\cdots +\zeta_p^{j_s}=\ell(\zeta_p+\cdots +\zeta_p^{p-1})+p^{n-m}\zeta_p^t\]
hence $s=\ell(p-1)+p^{n-m}$. Inserting this value in \eqref{eq:T1C2} we have
\[-\ell=\ell(p-1)+p^{n-m}-\frac{p^{n-m}(p^n-1)}{p^m-1}\iff \ell=\frac{p^{n-1}(p^{n-m}-1)}{p^m-1}=N\enspace.\]
We conclude that $a^{\perp}$ contains $N$ distinct $\mathcal S_{ji}^*\subseteq D_j$ for each $j\in{\mathbb F}_{p}^*\setminus\{t\}$ and $a^{\perp}$ contains 
$N+p^{n-m}$ distinct $\mathcal S_{ti}^*\subseteq D_t$. 

Now assume that $m\mid n$ and $f$ is such that for every $j\in{\mathbb F}_{p}^*$, 
\[D_j=\bigcup_{i=1}^{\frac{p^{n-1}(p^n-1)}{p^m-1}}\mathcal S^*_{ji}\enspace,\]
where $\mathcal S_{ji}$ are pairwise distinct elements of $\mathcal S$. It is easy to see that, under these assumptions, we have 
\begin{align*}
\mathcal W_f(0)&=\sum_{x\in{\mathbb F}_{p}^{2n}}\zeta_p^{f(x)}=\sum_{j=0}^{p-1}\sum_{x\in D_j}\zeta_p^j=\lvert D_0 \rvert+\sum_{j=1}^{p-1}\lvert D_j\rvert\zeta_p^j\\
&=p^{2n}-(p-1)p^{n-1}(p^n-1)+p^{n-1}(p^n-1)\sum_{j=1}^{p-1}\zeta_p^j\\
&=p^n\enspace.
\end{align*}

<!-- theorem-3: body_md -->
\label{th:PSplus}
Let $p$ be an odd prime and $m,n$ be positive integers such that $m<2n$ and $p^n>3$. Assume $\mathcal S$ is a partial $m$-spread of ${\mathbb F}_{p}^{2n}$ and 
$f:{\mathbb F}_{p}^{2n}\mapsto{\mathbb F}_{p}$ is such that for each $T\in\mathcal S$, $f$ is constant on $T^*$. Moreover, suppose that 
$f^{-1}(0)={\mathbb F}_{p}^{2n}\setminus\bigcup_{T\in\mathcal S}T$ and $f(0)=z\neq 0$. Then $f$ is bent if and only if $m\mid n$ and $f^{-1}(j)\cup\{0\}$ is a 
union of exactly $\frac{p^{n-1}(p^n-1)}{p^m-1}$ elements of $\mathcal S$ for every $j\in{\mathbb F}_{p}^*\setminus\{z\}$, $f^{-1}(z)$ is a union of exactly 
$\frac{(p^{n-1}+1)(p^n-1)}{p^m-1}$ elements of $\mathcal S$, and for every nonzero $a\in{\mathbb F}_{p}^{2n}$, $a^{\perp}$ contains either 
\begin{align}
 \label{eq:bentcond5}
&\begin{cases}
N+\frac{p^{n-m}-1}{p^m-1}\ \mbox{subspaces}\ T\subseteq f^{-1}(z)\ \mbox{and}\\
N\ \mbox{subspaces}\ T\subseteq f^{-1}(j)\ \mbox{for remaining}\ j\in{\mathbb F}_{p}^*\setminus\{z\}\\
\end{cases} 
\\\nonumber\mbox{or}\\
 \label{eq:bentcond6}
&\begin{cases}
N+\frac{p^{n-m}-1}{p^m-1}\ \mbox{subspaces}\ T\subseteq f^{-1}(z)\ \mbox{and}\\
N+p^{n-m}\ \mbox{subspaces}\ T\subseteq f^{-1}(k)\ \mbox{for some}\ t\in{\mathbb F}_{p}^*\ \mbox{and}\\
N\ \mbox{subspaces}\ T\subseteq f^{-1}(j)\ \mbox{for remaining}\ j\in{\mathbb F}_{p}^*\backslash\{z,t\}\enspace,\\
\end{cases}
\end{align}
where $N=\frac{p^{n-1}(p^{n-m}-1)}{p^m-1}$. Also, $\mathcal W_f(0)=p^n\zeta_p^z$ and if \eqref{eq:bentcond5} holds then $\mathcal W_f(a)=p^n$ and if 
\eqref{eq:bentcond6} holds then $\mathcal W_f(a)=p^n\zeta_p^t$. If $f$ is bent then it is a regular bent functions.

<!-- proof-3: body_md -->
For every $j\in{\mathbb F}_{p}$, let $D_j:=f^{-1}(j)$. For $j\neq 0$ let also $N_j\geq 0$ be such that $D_j=\bigcup_{i=1}^{N_j}\mathcal S_{ji}^*$ if $j\neq 
z$, and $D_z=\bigcup_{i=1}^{N_z}\mathcal S_{ji}$, where $\mathcal S_{ji}$ are pairwise distinct elements of $\mathcal S$ (if $N_j=0$ then $D_j=\emptyset$). 
Therefore, 
\begin{align*}
\lvert D_j\rvert&=N_j(p^m-1)\ \mbox{for}\ j\not\in\{0,z\}\\
\lvert D_z\rvert&=N_z(p^m-1)+1\\
\lvert D_0\rvert&=p^{2n}-(p^m-1)\sum_{j=1}^{p-1}N_j-1\enspace. 
\end{align*}
Note that $p-1$ divides $\lvert D_j\rvert$ when $j\neq z$ and does not divide $\lvert D_z\rvert$.

First, assume that $f$ is bent. Considering Lemma~\ref{le:Ny}, and the mentioned above divisibility, we conclude that $k=z$. By the same 
lemma, $\lvert D_j\rvert=p^{n-1}(p^n\mp 1)$ for $j\neq z$ and $\lvert D_z\rvert=p^{2n-1}\pm(p-1)p^{n-1}$. For $j\not\in\{0,z\}$, $\lvert 
D_j\rvert$ is divisible by $p^m-1$ and since $\gcd(p^n+1,p^m-1)\neq p^m-1$ (unless $p^n=3$), it follows that $\lvert D_0\rvert=\lvert 
D_j\rvert=p^{n-1}(p^n-1)$, $m\mid n$, and $f$ is regular. 

Also $N_j=\frac{p^{n-1}(p^n-1)}{p^m-1}$ for $j\neq z$ and $N_z=\frac{(p^{n-1}+1)(p^n-1)}{p^m-1}$ since $\lvert 
D_z\rvert=p^{2n-1}+(p-1)p^{n-1}$. Now calculate
\begin{align*}
\mathcal W_f(0)&=\sum_{x\in{\mathbb F}_{p}^{2n}}\zeta_p^{f(x)}=\sum_{j=0}^{p-1}\sum_{x\in D_j}\zeta_p^j\\
&=\lvert D_0\rvert+\lvert D_z\rvert\zeta_p^z+\sum_{j\in{\mathbb F}_{p}^*\setminus\{z\}}\lvert D_j\rvert\zeta_p^j\\
&=p^{n-1}(p^n-1)+(p^{n-1}(p^n-1)+p^n)\zeta_p^z+ p^{n-1}(p^n-1)\sum_{j\in {\mathbb F}_{p}^* \setminus\{z\}}\zeta_p^j\\
&=p^{n-1}(p^n-1)+p^n\zeta_p^z+p^{n-1}(p^n-1)\sum_{j\in{\mathbb F}_{p}^*}\zeta_p^j\\
&=p^{n-1}(p^n-1)+p^n\zeta_p^z-p^{n-1}(p^n-1)\\
&=p^n\zeta_p^z\enspace.
\end{align*}

Now we show that \eqref{eq:bentcond5}~or~\eqref{eq:bentcond6} holds. Let $a\in{\mathbb F}_{p}^{2n}$ with $a\neq 0$. Assume that $a^{\perp}$ contains 
$s$ distinct subspaces from $\mathcal S$, say $\mathcal S_{j_1 i_1},\ldots ,\mathcal S_{j_s i_s}$, where $0<j_1,\ldots,j_s<p$ and $1\leq 
i_1,\ldots,i_s\leq\frac{(p^{n-1}+1)(p^n-1)}{p^m-1}$. Consider 

\begin{align*}
\mathcal W_f(a)&=\sum_{x\in{\mathbb F}_{p}^{2n}}\zeta_p^{f(x)-\langle a,x \rangle}\\
&=\sum_{x\in D_0}\zeta_p^{-\langle a,x\rangle}+\zeta_p^z\sum_{x\in D_z}\zeta_p^{-\langle a,x\rangle}+\sum_{j\in{\mathbb F}_{p}^*\setminus\{z\}}\zeta_p^j\sum_{x\in D_j}\zeta_p^{-\langle a,x \rangle}\\
&=-\sum_{x\in D_z}\zeta_p^{-\langle a,x\rangle}-\sum_{j\in{\mathbb F}_{p}^*\setminus\{z\}}\sum_{x\in D_j}\zeta_p^{-\langle a,x\rangle}+\zeta_p^z\sum_{x\in D_z}\zeta_p^{-\langle a,x\rangle}+\sum_{j\in{\mathbb F}_{p}^*\setminus\{z\}}\zeta_p^j\sum_{x\in D_j}\zeta_p^{-\langle a,x\rangle}\\
&=(p-1)\frac{p^{n-1}(p^n-1)}{p^m-1}+\frac{p^n-p^m}{p^m-1}-sp^m +\frac{p^{n-1}(p^n-1)}{p^m-1}+p^m(\zeta_p^{j_1}+\cdots+\zeta_p^{j_s})-\left(\frac{p^n-p^m}{p^m-1}\right)\zeta_p^z\\
&=\frac{p^{n}(p^n-1)}{p^m-1}+p^m(\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-s)+\frac{p^n-p^m}{p^m-1}(1-\zeta_p^z)\enspace.
\end{align*}
Hear we used Lemma~\ref{le:sspsum} consistently. Since $f$ is a regular bent function, for some $t\in{\mathbb F}_{p}$, we have

\begin{align}
 \label{eq:T2wta}
\nonumber&\frac{p^{n}(p^n-1)}{p^m-1}+p^m(\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-s)+\frac{p^n-p^m}{p^m-1}(1-\zeta_p^z)=p^n\zeta_p^t\\
\nonumber\implies&\frac{p^{n-m}(p^n-1)}{p^m-1}+\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-s+\frac{p^{n-m}-1}{p^m-1}(1-\zeta_p^z)=p^{n-m}\zeta_p^t\\
\nonumber\implies&\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-\frac{p^{n-m}-1}{p^m-1}\zeta_p^z-p^{n-m}\zeta_p^t=s-\frac{p^{n-m}(p^n-1)}{p^m-1}-\frac{p^{n-m}-1}{p^m-1}\\
\implies&\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-\frac{p^{n-m}-1}{p^m-1}\zeta_p^z-p^{n-m}\zeta_p^t=s-\frac{p^{2n-m}-1}{p^m-1}\enspace.
\end{align} 
Now we consider two cases separately, namely when $t=0$ and when $t\in{\mathbb F}_{p}^*$.

\textbf{Case 1.} Let $t=0$. In this case, \eqref{eq:T2wta} reduces to 
\begin{equation}
 \label{eq:T2C1}
\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}-\frac{p^{n-m}-1}{p^m-1}\zeta_p^z=s+p^{n-m}-\frac{p^{2n-m}-1}{p^m-1}\enspace.
\end{equation}

integer $\ell$. Thus, 
\[\zeta_p^{j_1}+\cdots +\zeta_p^{j_s}=\ell(\zeta_p+\cdots+\zeta_p^{p-1})+ \frac{p^{n-m}-1}{p^m-1}\zeta_p^z\]
hence $s=\ell(p-1)+ \frac{p^{n-m}-1}{p^m-1}$. Inserting this value in \eqref{eq:T2C1} we obtain
\[-\ell=\ell(p-1)+\frac{p^{n-m}-1}{p^m-1}+p^{n-m}-\frac{p^{2n-m}-1}{p^m-1}\implies \ell=\frac{p^{n-1}(p^{n-m}-1)}{p^m-1}=N\enspace.\]
We conclude that $a^{\perp}$ contains $N$ distinct $\mathcal S_{ji}^*\subseteq D_j$ for each $j\in{\mathbb F}_{p}^*\setminus\{z\}$ and also contains 
$N+\frac{p^{n-m}-1}{p^m-1}$ distinct $\mathcal S_{zi}^*\subseteq D_z$. 

\textbf{Case 2.} Let $t\neq 0$. In this case, \eqref{eq:T2wta} reduces to 
\begin{equation}
 \label{eq:T2C2}
\zeta_p^{j_1}+\cdots +\zeta_p^{j_s}-\frac{p^{n-m}-1}{p^m-1}\zeta_p^z-p^{n-m}\zeta_p^t=s-\frac{p^{2n-m}-1}{p^m-1}\enspace.
\end{equation}
Again, the LHS of \eqref{eq:T2C2} is an integer if and only if it is of the form $\ell(\zeta_p+\cdots+\zeta_p^{p-1})=-\ell$ for some 
positive integer $\ell$. Thus, 
\[\zeta_p^{j_1}+\cdots+\zeta_p^{j_s}=\ell(\zeta_p+\cdots+\zeta_p^{p-1})+\frac{p^{n-m}-1}{p^m-1}\zeta_p^z+p^{n-m}\zeta_p^t\]

\[-\ell=\ell(p-1)+\frac{p^{n-m}-1}{p^m-1}+p^{n-m}-\frac{p^{2n-m}-1}{p^m-1}\implies \ell=\frac{p^{n-1}(p^{n-m}-1)}{p^m-1}=N\enspace.\]
We conclude that $a^{\perp}$ contains $N$ distinct $\mathcal S_{ji}^*\subseteq D_j$ for each $j\in{\mathbb F}_{p}^*\setminus\{z,t\}$, 
$N+\frac{p^{n-m}-1}{p^m-1}$ distinct $\mathcal S_{ti}^*\subseteq D_z$ and $N+p^{n-m}$ distinct $\mathcal S_{ki}^*\subseteq D_k$. 

In the opposite direction, the proof is straightforward.\qed

\begin{align*}
\mathcal W_a(0)&=1+(-1)^{k-1}(p^m-1)K_{k-1}({\operatorname{N}}(a))\quad\mbox{and}\\
\mathcal W_a(-1)&=1+(-1)^k K_{k-1}({\operatorname{N}}(a))+p^m\sum_{x\in U}\omega^{f(x)}\enspace,
\end{align*}
where $k=2n/m$ and $U=\{x\in{\mathbb F}_{p^{2n}}^*/{\mathbb F}_{p^m}^*\ |\ {\operatorname{Tr}}^{2n}_m(x)=0\}$. Assume $p^n>3$ and $f$ is bent, then $m$ divides $n$, 
\begin{equation}
 \label{eq:Kl_cond}
(-1)^{k-1}K_{k-1}({\operatorname{N}}(a))=(p^n-1)/(p^m-1)\enspace, 
\end{equation}
and $f$ is regular from the $m$-$\mathcal{PS}^{-}$ class. Moreover, $\mathcal W_a(0)=p^n$ and
\[\mathcal W_a(-1)=1-(p^n-1)/(p^m-1)+p^m\sum_{x\in U}\omega^{f(x)}\enspace.\]

Any $i\in\{0,\dots,p^{2n}-2\}$ can be uniquely written as $i=d j_1+j_2$ for some $j_1\in\{0,\dots,p^m-2\}$ and $j_2\in\{0,\dots,d-1\}$, 
where $d=(p^{2n}-1)/(p^m-1)$. Select and fix a primitive element $\xi$ of ${\mathbb F}_{p^{2n}}$. Any $x\in{\mathbb F}_{p^{2n}}^*$ can be uniquely written as 
$\xi^i$. Then 
\[{\operatorname{Tr}}_{2n}\big(a x^{t(p^m-1)}+x\big)={\operatorname{Tr}}_{2n}\big(a\xi^{t(p^m-1)j_2}+\xi^{d j_1+j_2}\big)={\operatorname{Tr}}_{2n}\left(a\beta^{j_2}+\gamma^{j_1}\xi^{j_2}\right)\] 
and ${\operatorname{Tr}}_{2n}\big(a x^{t(p^m-1)}\big)={\operatorname{Tr}}_{2n}\left(a\beta^{j_2}\right)$, where $\beta=\xi^{t(p^m-1)}$ and $\gamma=\xi^d\in{\mathbb F}_{p^m}^*$. 

The Walsh transform coefficient of $f$ evaluated at $-1$ is equal to 
\[\mathcal W_a(-1)=\sum_{x\in{\mathbb F}_{p^{2n}}}\omega^{{\operatorname{Tr}}_{2n}\left(a x^{t(p^m-1)}+x\right)}=1+\sum_{j_2=0}^{d-1}\omega^{{\operatorname{Tr}}_{2n}\left(a\beta^{j_2}\right)}\sum_{j_1=0}^{p^m-2}\omega^{{\operatorname{Tr}}_{2n}\left(\gamma^{j_1}\xi^{j_2}\right)}\enspace.\] 
Consider the latter sum 
\[\sum_{j_1=0}^{p^m-2}\omega^{{\operatorname{Tr}}_{2n}\left(\gamma^{j_1}\xi^{j_2}\right)}=
\sum_{j_1=0}^{p^m-2}\omega^{{\operatorname{Tr}}_m\left(\gamma^{j_1}{\operatorname{Tr}}^{2n}_m(\xi^{j_2})\right)}=
\left\{\begin{array}{ll}
p^m-1,&\ \mbox{if}\quad{\operatorname{Tr}}^{2n}_m(\xi^{j_2})=0\\
-1,&\ \mbox{otherwise}
\end{array}\right.\] 
since the multiplicative order of $\gamma$ is equal to $p^m-1$ and thus, $\gamma$ is a primitive element of ${\mathbb F}_{p^m}$ and $\gamma^{j_1}$ runs 
through all nonzero elements of ${\mathbb F}_{p^m}$. Further, ${\operatorname{Tr}}^{2n}_m(x)=0$ for $p^{2n-m}-1$ elements of ${\mathbb F}_{p^{2n}}^*$ that reduce to 
$(p^{2n-m}-1)/(p^m-1)$ elements $x\in{\mathbb F}_{p^{2n}}^*/{\mathbb F}_{p^m}^*$ (this number of values of $j_2$ satisfy ${\operatorname{Tr}}^{2n}_m(\xi^{j_2})=0$). $U$ 
denotes the set containing all such $x$. 

Therefore,
\begin{align*}
\mathcal W_a(-1)&=1-\sum_{j_2=0}^{d-1}\omega^{{\operatorname{Tr}}_{2n}\left(a\beta^{j_2}\right)}+p^m\sum_{\substack{j_2=0\\ {\operatorname{Tr}}^{2n}_m(\xi^{j_2})=0}}^{d-1}\omega^{{\operatorname{Tr}}_{2n}\big(a\beta^{j_2}\big)}\\
&=1-\sum_{j=0}^{d-1}\omega^{{\operatorname{Tr}}_{2n}\left(a\beta^{j}\right)}+p^m\sum_{x\in U}\omega^{f(x)}\enspace.
\end{align*}
Note that the sets $\{\beta^{j}\ |\ j=0,\dots,d-1\}$ and $\{\xi^{j(p^m-1)}\ |\ j=0,\dots,d-1\}$ are equal because $t$ and 
$(p^{2n}-1)/(p^m-1)$ are coprime and thus, using Theorem~\ref{th:mKl_sum} with $q=p^m$ and $k=2n/m$, 
\begin{align*}
\mathcal W_a(-1)&=1-\sum_{j=0}^{d-1}\omega^{{\operatorname{Tr}}_{2n}\left(a\xi^{j(p^m-1)}\right)}+p^m\sum_{x\in U}\omega^{f(x)}\\
&=1-\sum_{j=0}^{d-1}\chi\left({\operatorname{Tr}}^{2n}_m\big(a\xi^{j(p^m-1)}\big)\right)+p^m\sum_{x\in U}\omega^{f(x)}\\
&=1+(-1)^k K_{k-1}({\operatorname{N}}(a))+p^m\sum_{x\in U}\omega^{f(x)}\enspace.
\end{align*}

Further, for any $b\in{\mathbb F}_{p^n}^*$ we have
\begin{align*}
\mathcal W_a(-b)&=S_{ab^{-t(p^m-1)}}(-1)\\
&=1+(-1)^k K_{k-1}({\operatorname{N}}(a))+p^m\sum_{{\substack{j=0\\{\operatorname{Tr}}^{2n}_m(\xi^j)=0}}}^{d-1}\omega^{{\operatorname{Tr}}_{2n}\big(ab^{-t(p^m-1)}\beta^{j}\big)}=\mathcal W_a(b) 
\end{align*}
since ${\operatorname{N}}(ab^{-t(p^m-1)})=\big(ab^{-t(p^m-1)}\big)^{(p^{2n}-1)/(p^m-1)}=a^{(p^{2n}-1)/(p^m-1)}={\operatorname{N}}(a)$. 

The Walsh transform coefficient of $f$ evaluated at $0$ is equal to 
\begin{align*}
\mathcal W_a(0)&=\sum_{x\in{\mathbb F}_{p^{2n}}}\omega^{{\operatorname{Tr}}_{2n}\left(ax^{t(p^m-1)}\right)}=1+(p^m-1)\sum_{j=0}^{d-1}\omega^{{\operatorname{Tr}}_{2n}\left(a\beta^{j}\right)}\\
&=1+(-1)^{k-1}(p^m-1)K_{k-1}({\operatorname{N}}(a))\enspace.
\end{align*}

If $p^n>3$ and $f$ is bent, then by Theorem~\ref{th:PSminus}, $m$ divides $n$ and $f$ is a regular bent function from $m$-$\mathcal{PS}^{-}$ class, 
$\mathcal W_a(0)=p^n$ that gives 
\[(-1)^{k-1}K_{k-1}({\operatorname{N}}(a))=(p^n-1)/(p^m-1)\enspace.\] 
Then also
\[\mathcal W_a(-1)=1-(p^n-1)/(p^m-1)+p^m\sum_{x\in U}\omega^{f(x)}\enspace.\]
