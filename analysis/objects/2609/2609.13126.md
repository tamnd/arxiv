\label{thm:main}
		Let $K\geq3$, and let $\Delta\subseteq\mathbb{Z}^d$ be the vertex set of a fixed nondegenerate integral $(K-1)$-simplex.  If $d\geq4K+4$, then there are
		constants $C,c>0$, depending only on $\Delta$ and $d$, such that every set $A\subseteq[N]^d$ containing no nontrivial similar copy of $\Delta$ satisfies
		\[
		\lvert A \rvert\leq C N^d\exp\!\left(-c\sqrt{\log N}\right).
		\]

Put
		\[
		W_{s,t}(p^m)
		:=
		\sum_{\mathsf M\in\operatorname{Mat}_{s\times t}(\mathbb{Z}/p^m\mathbb{Z})}
		\lvert \operatorname{im}\mathsf M \rvert^{-\tau},

		other matrix has a unit entry.  Choose such an entry and use it as a
		pivot.  After permuting rows and columns, write
		\[
		\mathsf M
		=
		\begin{pmatrix}
			u&\mathsf b\\
			\mathsf c&\mathsf D
		\end{pmatrix},
		\]
		where $\mathsf b$ is a row vector and $\mathsf c$ is a column vector.
		Invertible row and column operations reduce $\mathsf M$ to
		\[
		\begin{pmatrix}
			u&0\\
			0&\mathsf M'
		\end{pmatrix},
		\qquad
		\mathsf M'=\mathsf D-\mathsf c u^{-1}\mathsf b.
		\]

		$\lvert \operatorname{im}\mathsf M \rvert = p^m\lvert \operatorname{im}\mathsf M' \rvert.$

		and for $u,\mathsf b,$ and $\mathsf c$.  For each such choice, the map
		$\mathsf C\longmapsto \mathsf M'=\mathsf C-\mathsf c u^{-1}\mathsf b$
		is a bijection of
		$\operatorname{Mat}_{(s-1)\times(t-1)}(\mathbb{Z}/p^m\mathbb{Z})$, so $\mathsf M'$ may

		\[
		W_{s,t}(p^m)
		\leq
		W_{s,t}(p^{m-1})
		+
		c_{s,t}p^{-m(\tau-s-t+1)}
		W_{s-1,t-1}(p^m).
		\]

		$W_{s,t}(p^m)=1$ by convention.  Suppose that $s,t\geq1$, and assume
		inductively that

			W_{s,t}(p^m)

			W_{s,t}(p^{m-1})

		the exponent of $\alpha$ below is at least $s$, and hence
		\[
		Q^{-\gamma}\alpha^{s/2}
		=
		Q_0^{-\gamma}
		\alpha^{\gamma B_0/(L(L+3)+3)+s/2}
		\leq
		Q_0^{-\gamma}\alpha^s.
		\]

		\eqref{eq:density-increment-arc-bound} is therefore
		$O_{\Delta,d}\!\left(Q_0^{-\gamma}\alpha^s\right).$
		Assuming $Q_0$ is sufficiently large depending only on $\Delta$ and
		$d$, this contribution is at most $\alpha^s/2$ and may be moved to
		the left-hand side. Since
		$\sum_{q\geq1}q^{-1-\eta}<\infty$, there is therefore some
		$q\leq2Q^L$ such that
		\begin{equation}\label{eq:large-box-average}
			\sum_{x\in\mathbb{Z}^d}\lvert \mu_q*g(x) \rvert^K
			\gg_{\Delta,d}\alpha^KN^d.
		\end{equation}
		
		Recall that
		$N_q=\left\lfloor R/(qQ^{L(L+2)+3})\right\rfloor.$
		The function $\mu_q*g$ is supported on
		$[1,N+q(N_q-1)]^d$, and
		\begin{align*}
			\lvert 
				[1,N+q(N_q-1)]^d
				\setminus
				[1+q(N_q-1),N]^d
			 \rvert&\ll_d qN_qN^{d-1}\\
			&\ll_{\Delta,d}Q^{-L(L+2)-3}N^d\\
			&\ll_{\Delta,d}
			Q_0^{-L(L+2)-3}
			\alpha^{B_0(L(L+2)+3)/(L(L+3)+3)}N^d.
		\end{align*}
		Assuming $B_0$ is sufficiently large depending only on $K$, the
		exponent of $\alpha$ in the last line is at least $K$, so
		\begin{equation}\label{eq:density-increment-boundary-size}
			\lvert 
				[1,N+q(N_q-1)]^d
				\setminus
				[1+q(N_q-1),N]^d
			 \rvert
			\ll_{\Delta,d}
			Q_0^{-L(L+2)-3}\alpha^KN^d.
		\end{equation}
		Since $\mu_q$ is a probability measure and
		$\sum_{x\in\mathbb{Z}^d}g(x)=0$, we have
		$\sum_{x\in\mathbb{Z}^d}(\mu_q*g)(x)=0.$
		Consequently,
		\begin{equation}\label{eq:positive-negative-mass}
			\sum_{x\in\mathbb{Z}^d}(\mu_q*g)_+(x)
			=
			\sum_{x\in\mathbb{Z}^d}(\mu_q*g)_-(x)
			=
			\frac12\sum_{x\in\mathbb{Z}^d}\lvert (\mu_q*g)(x) \rvert.
		\end{equation}
		Since $g\geq-\alpha$ and $\lvert g \rvert\leq1$, we also have
		\[
		0\leq(\mu_q*g)_-\leq\alpha,
		\qquad
		\lvert \mu_q*g \rvert\leq1.
		\]
		Assuming $Q_0$ is sufficiently large depending only on $\Delta$ and
		$d$, \eqref{eq:density-increment-boundary-size} and
		\eqref{eq:large-box-average} give
		\begin{equation}\label{eq:restricted-large-moment}
			\sum_{x\in[1+q(N_q-1),N]^d}
			\lvert (\mu_q*g)(x) \rvert^K
			\gg_{\Delta,d}\alpha^KN^d.
		\end{equation}
		On $[1+q(N_q-1),N]^d$ we have
		$\mu_q*g=\mu_q*\1_A-\alpha.$
		Since
		\[
		\lvert (\mu_q*g)(x) \rvert^K
		=
		\bigl((\mu_q*g)_+(x)\bigr)^K
		+
		\bigl((\mu_q*g)_-(x)\bigr)^K,
		\]

			\sum_{x\in[1+q(N_q-1),N]^d}
			\bigl((\mu_q*g)_+(x)\bigr)^K
			&\gg_{\Delta,d}\alpha^KN^d,
			\label{eq:positive-moment-alternative}\\
			\sum_{x\in[1+q(N_q-1),N]^d}
			\bigl((\mu_q*g)_-(x)\bigr)^K
			&\gg_{\Delta,d}\alpha^KN^d.
			\label{eq:negative-moment-alternative}
		\end{align}
		If \eqref{eq:positive-moment-alternative} holds, pigeonholing over at

		Since $(\mu_q*g)_-\leq\alpha$,
		\[
		\begin{aligned}
			\sum_{x\in[1+q(N_q-1),N]^d}(\mu_q*g)_-(x)
			&\geq
			\alpha^{1-K}
			\sum_{x\in[1+q(N_q-1),N]^d}
			\bigl((\mu_q*g)_-(x)\bigr)^K\gg_{\Delta,d}\alpha N^d.
		\end{aligned}

		It follows from \eqref{eq:positive-negative-mass} that

			\sum_{x\in\mathbb{Z}^d}(\mu_q*g)_+(x)
			=
			\sum_{x\in\mathbb{Z}^d}(\mu_q*g)_-(x)
			\gg_{\Delta,d}\alpha N^d.
		\end{equation}
		Since $\mu_q*g$ is supported on $[1,N+q(N_q-1)]^d$ and
		$(\mu_q*g)_+\leq1$, \eqref{eq:density-increment-boundary-size} gives
		\[
		\sum_{x\in
			[1,N+q(N_q-1)]^d\setminus[1+q(N_q-1),N]^d}
		(\mu_q*g)_+(x)
		=
		O_{\Delta,d}\!\left(
		Q_0^{-L(L+2)-3}\alpha^KN^d
		\right).

		\sum_{x\in[1+q(N_q-1),N]^d}(\mu_q*g)_+(x)
		\gg_{\Delta,d}\alpha N^d.
		\]
		Pigeonholing over at most $N^d$ points again gives some
		$y\in[1+q(N_q-1),N]^d$ such that
		$(\mu_q*g)(y)\gg_{\Delta,d}\alpha.$

		(\mu_q*g)(y)+\alpha

		\frac{R}{2Q^{L(L+3)+3}}

		\frac{R}{2qQ^{L(L+2)+3}}

		whereas every density is at most $1$. Let $j_*$ be the first stage at

		\begin{equation}\label{eq:terminal-length}
			N_{j_*}\ll_{\Delta,d}\alpha_{j_*}^{-B_0}.
		\end{equation}

		$j_*\ll_{\Delta,d}1+\log(1/\alpha_0).$

		C\exp\!\left(-c\sqrt{\log N_0}\right).
		\]

			2^{j(d_2-s)}2^{\ell(n-d_1-t)}
			\\
			&=
			R^{d_1s+(n-d_2)t}
			\left(\sum_{j=0}^{J_1}2^{j(d_2-s)}\right)
			\left(\sum_{\ell=0}^{J_2}2^{\ell(n-d_1-t)}\right)
			\\
			&\ll_{n,s,t}
			R^{d_1s+(n-d_2)t}
			2^{J_1(d_2-s)}2^{J_2(n-d_1-t)}
			\\
			&\ll_{n,s,t}
			R^{d_1s+(n-d_2)t}
			R^{d_1(d_2-s)}
			R^{(n-d_2)(n-d_1-t)}
			\\
			&=
			R^{d_1s+(n-d_2)t+d_1d_2-d_1s
				+(n-d_2)(n-d_1)-(n-d_2)t}
			\\
			&=
			R^{d_1d_2+(n-d_2)(n-d_1)}
			\\
			&=
			R^{n^2-n(d_1+d_2)+2d_1d_2}.
		\end{align*}
		
		It remains to maximize the exponent
		\[
		E(d_1,d_2):=n^2-n(d_1+d_2)+2d_1d_2
		\]
		over $1\leq d_1\leq s$ and $n-t\leq d_2\leq n-1$. We have
		\[
		E(d_1+1,d_2)-E(d_1,d_2)
		=2d_2-n
		\geq n-2t
		\geq0
		\]
		and
		\[
		E(d_1,d_2-1)-E(d_1,d_2)
		=n-2d_1
		\geq n-2s
		\geq0.
		\]
		Thus $E(d_1,d_2)$ is maximized at $d_1=s$ and $d_2=n-t$, where
		\[
		E(s,n-t)=n(s+t)-2st.
		\]
		Summing over the $O_n(1)$ choices for $(d_1,d_2)$ completes the proof.
