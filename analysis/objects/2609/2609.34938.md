\label{prop:critical-controllability}
Assume that
\[
g\in L^\infty(I;\mathbb R),
\qquad
\bigl|\{x\in I:\ g(x)\not\nobreak\mkern -2mu =0\}\bigr|>0.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Then the pair \((A_{\relax cr},B_{\relax cr})\) is controllable.
Consequently, there exists
\[
K_{\relax cr}\in\mathcal L(X_{\relax cr},X)
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
such that \(A_{\relax cr}+B_{\relax cr}K_{\relax cr}\) is Hurwitz.
In particular, there exist \(M_{\relax cr}\ge1\) and \(\omega_{\relax cr}>0\) such that
\begin{equation}
\left\|
e^{(A_{\relax cr}+B_{\relax cr}K_{\relax cr})t}
\right\|_{\mathcal L(X_{\relax cr})}
\le
M_{\relax cr}e^{-\omega_{\relax cr}t},
\qquad t\ge0.
\label{eq:critical-exponential-decay}
\end{equation}

\[
B_{\relax cr}^*\phi
=
B^*P_{\relax cr}\phi
=
g\phi.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}

\ker(\lambda I-A_{\relax cr}^*)
\cap
\ker B_{\relax cr}^*
=
\{0\},
\qquad
\lambda\in\sigma(A_{\relax cr}).
\label{eq:PBH-critical}

=
\lim_{n\to\infty}

0.

\label{thm:finite-infinite-lifting}
Let \(K_{\relax cr}\) satisfy \eqref{eq:critical-exponential-decay}.
Define
\begin{equation}
A_F
:=
A_n+BK_{\relax cr}P_{\relax cr},
\qquad
D(A_F)=D(A_n).
\label{eq:AF-lifting}
\end{equation}
Then \(A_F\) generates an analytic exponentially stable \(C_0\)-semigroup on \(X\). More precisely, for every
\[
0<\omega<\min\{\rho,\omega_{\relax cr}\},
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
there exists \(M_F\ge1\) such that
\begin{equation}
\|e^{A_Ft}\|_{\mathcal L(X)}
\le
M_Fe^{-\omega t},
\qquad t\ge0.
\label{eq:AF-exp-lifting}
\end{equation}
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
Since \(BK_{\relax cr}P_{\relax cr}\in\mathcal L(X)\), the standard bounded-perturbation theorem for analytic semigroups \cite{Pazy1983} implies that \(A_F\), with \(D(A_F)=D(A_n)\), generates an analytic semigroup on \(X\).

Relative to \(X=X_{\relax cr}\oplus X_{\relax st}\),
\[
A_F
=
\begin{pmatrix}
A_{\relax cr}+B_{\relax cr}K_{\relax cr} & 0\\[1mm]
L_{\relax sc} & A_{\relax st}
\end{pmatrix},
\qquad
L_{\relax sc}:=P_{\relax st}BK_{\relax cr},
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
with domain \(X_{\relax cr}\oplus D(A_{\relax st})\).
Indeed, the spectral projections commute with \(A_n\), while \(P_{\relax cr}P_{\relax st}=0\). Hence, for \(q=q_{\relax cr}+q_{\relax st}\),
\[
\dot q_{\relax cr}
=
(A_{\relax cr}+B_{\relax cr}K_{\relax cr})q_{\relax cr},
\qquad
\dot q_{\relax st}
=
A_{\relax st}q_{\relax st}+L_{\relax sc}q_{\relax cr}.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi

Using \eqref{eq:stable-residual-decay}, \eqref{eq:critical-exponential-decay}, and variation of constants gives
\[
\begin{aligned}
\|q_{\relax st}(t)\|_X
&\le
e^{-\rho t}\|q_{\relax st}(0)\|_X\\
&\quad+
M_{\relax cr}\|L_{\relax sc}\|
\intop_0^t
e^{-\rho(t-s)}e^{-\omega_{\relax cr}s}\,ds\,
\|q_{\relax cr}(0)\|_X.
\end{aligned}
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Setting \(m=\min\{\rho,\omega_{\relax cr}\}\), one has
\[
\intop_0^t
e^{-\rho(t-s)}e^{-\omega_{\relax cr}s}\,ds
\le
t e^{-mt}.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
For every \(0<\omega<m\),
\[
t e^{-mt}
=
t e^{-(m-\omega)t}e^{-\omega t}
\le
\frac{1}{e(m-\omega)}e^{-\omega t}.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Together with the critical-mode estimate and the orthogonal decomposition of \(X\), this yields \eqref{eq:AF-exp-lifting}.

Define
\[
\mathcal I_0
=
\|(u_0,v_0)\|_{\mathcal H}
+
\|U_0\|_{L^2(-D,0;X)}.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Assume that, for some
\[
0<\theta<
\min\left\{\omega_F,\frac{\mu_v}{2}\right\},
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
\begin{equation}
\label{eq:small-gain-condition}
\Gamma(\theta)
:=
G_{u\mathfrak d}(\theta)G_{\mathfrak du}(\theta)
<1.
\end{equation}
Then there exists \(C_\theta>0\), independent of \(T\), such that
\begin{align}
\|u_x\|_{L^2_\theta(0,T;L^2(I))}
&+
\|\mathfrak d\|_{L^2_\theta(0,T;X)}
+
\|v\|_{L^2_\theta(0,T;H^1(\Omega))}\nonumber\\
&+
\|Y\|_{L^2_\theta(0,T;H_0^1(I))}
\le
C_\theta\mathcal I_0.
\label{eq:small-gain-global-estimate}
\end{align}

Since \(\mathfrak d\in L^2(0,T;X)\) on every finite interval, all preceding weighted estimates apply. For \(T\le D\), the conclusion follows by restricting to \([0,T]\) the corresponding estimates on the fixed interval \([0,D]\); hence assume \(T>D\).

Theorem~\ref{thm:global-wellposedness}, applied on the fixed interval \([0,D]\), and the bound \(e^{2\theta t}\le e^{2\theta D}\) yield
\begin{equation}
\label{eq:initial-delay-interval}
\|u_x\|_{L^2_\theta(0,D;L^2(I))}
\le
C_{D,\theta}\mathcal I_0.
\end{equation}
Moreover, from \eqref{eq:predictor-initial},
\begin{equation}
\label{eq:Y0-initial-bound}
\|Y_0\|_X
\le
C_{Y,D}\mathcal I_0.
\end{equation}
Hence Proposition~\ref{prop:weighted-shoreline-gain} yields
\begin{equation}
\label{eq:ux-complete-gain}
\|u_x\|_{L^2_\theta(0,T;L^2(I))}
\le
C_{u0}(\theta)\mathcal I_0
+
G_{u\mathfrak d}(\theta)
\|\mathfrak d\|_{L^2_\theta(0,T;X)},
\end{equation}
where \(C_{u0}(\theta)>0\) is independent of \(T\).

On the other hand, Proposition~\ref{prop:weighted-residual-gain} gives
\begin{equation}
\label{eq:d-complete-gain}
\|\mathfrak d\|_{L^2_\theta(0,T;X)}
\le
C_{\mathfrak d0}(\theta)\mathcal I_0
+
G_{\mathfrak du}(\theta)
\|u_x\|_{L^2_\theta(0,T;L^2(I))}.
\end{equation}
Combining \eqref{eq:ux-complete-gain} and \eqref{eq:d-complete-gain} yields
\[
\begin{aligned}
\bigl(1-\Gamma(\theta)\bigr)
\|u_x\|_{L^2_\theta(0,T;L^2(I))}
&\le
\left[
C_{u0}(\theta)
+
G_{u\mathfrak d}(\theta)
C_{\mathfrak d0}(\theta)
\right]\mathcal I_0.
\end{aligned}
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Since \(\Gamma(\theta)<1\), the shoreline-gradient estimate is uniform in \(T\). Substitution into \eqref{eq:d-complete-gain}, \eqref{eq:weighted-bathymetric-gain}, and \eqref{eq:weighted-Y-H1} proves \eqref{eq:small-gain-global-estimate}.

\label{thm:full-exponential-stability}
Let \(K_{\relax cr}\) be given by Proposition~\ref{prop:critical-controllability}, and let \(A_F\) be the exponentially stable operator of Theorem~\ref{thm:finite-infinite-lifting}. Assume \eqref{eq:bathymetric-coercivity} and suppose that, for some
\[
0<\theta<
\min\left\{\omega_F,\frac{\mu_v}{2}\right\},
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
the small-gain condition \eqref{eq:small-gain-condition} holds.
Then the predictor feedback
\[
U(t)=K_{\relax cr}P_{\relax cr}Y(t)
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
exponentially stabilizes the delayed coupled system. More precisely, there exists \(M\ge1\) such that
\begin{equation}
\label{eq:final-exponential-estimate}
\|(u(t),v(t),z(t))\|_{\mathcal X}
\le
Me^{-\theta t}\mathcal I_0,
\qquad t\ge0,
\end{equation}
where
\[
\mathcal X
=
\mathcal H\times\mathcal Z,
\qquad
\mathcal Z=L^2((0,D);X).
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi

Letting \(T\to\infty\) in Proposition~\ref{prop:small-gain-closure} yields
\begin{equation}
\label{eq:global-weighted-estimate}
\begin{aligned}
\|u_x\|_{L^2_\theta(0,\infty;L^2(I))}
&+
\|\mathfrak d\|_{L^2_\theta(0,\infty;X)}
+
\|v\|_{L^2_\theta(0,\infty;H^1(\Omega))}\\
&+
\|Y\|_{L^2_\theta(0,\infty;H_0^1(I))}
\le
C_\theta\mathcal I_0.
\end{aligned}
\end{equation}

For the predictor, variation of constants gives
\[
Y(t)
=
e^{A_Ft}Y_0
+
\intop_0^t
e^{A_F(t-s)}e^{A_nD}\mathfrak d(s)\,ds.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Since \(\theta<\omega_F\), Cauchy--Schwarz gives
\[
\begin{aligned}
\left\|
\intop_0^t
e^{A_F(t-s)}e^{A_nD}\mathfrak d(s)\,ds
\right\|_X
&\le
\frac{M_F\|e^{A_nD}\|_{\mathcal L(X)}}
{\sqrt{2(\omega_F-\theta)}}
e^{-\theta t}\\
&\quad\times
\|\mathfrak d\|_{L^2_\theta(0,\infty;X)}.
\end{aligned}
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Hence, using \eqref{eq:Y0-initial-bound} and \eqref{eq:global-weighted-estimate},
\begin{equation}
\label{eq:Y-pointwise-decay}
\|Y(t)\|_X
\le
C_Ye^{-\theta t}\mathcal I_0.
\end{equation}

For \(t\ge D\), \eqref{eq:u-predictor-weighted} and \eqref{eq:Y-pointwise-decay} yield
\[
\|u(t)\|_X
\le
C_Ye^{\theta D}e^{-\theta t}\mathcal I_0
+
M_{n,D}
\intop_{t-D}^{t}\|\mathfrak d(s)\|_X\,ds,
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
where
\[
M_{n,D}
=
\sup_{0\le r\le D}
\|e^{A_nr}\|_{\mathcal L(X)}.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Moreover,
\[
\intop_{t-D}^{t}\|\mathfrak d(s)\|_X\,ds
\le
e^{-\theta(t-D)}
\sqrt D\,
\|\mathfrak d\|_{L^2_\theta(0,\infty;X)}.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Thus,
\begin{equation}
\label{eq:u-pointwise-decay}
\|u(t)\|_X
\le
C_ue^{-\theta t}\mathcal I_0,
\qquad t\ge D.
\end{equation}

For the bathymetric component, \eqref{eq:bathymetric-energy-gain} and \(\|v\|_{L^2(\Omega)}\le\|v\|_{H^1(\Omega)}\) imply
\[
\frac{d}{dt}\|v(t)\|_{L^2(\Omega)}^2
+
\mu_v\|v(t)\|_{L^2(\Omega)}^2
\le
2b_v\|u_x(t)\|_{L^2(I)}^2.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Hence
\[
\begin{aligned}
\|v(t)\|_{L^2(\Omega)}^2
&\le
e^{-\mu_vt}\|v_0\|_{L^2(\Omega)}^2\\
&\quad+
2b_v
\intop_0^t
e^{-\mu_v(t-s)}
\|u_x(s)\|_{L^2(I)}^2\,ds.
\end{aligned}
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Since \(2\theta<\mu_v\),
\[
\begin{aligned}
\intop_0^t
e^{-\mu_v(t-s)}
\|u_x(s)\|_{L^2(I)}^2\,ds
&\le
e^{-2\theta t}
\|u_x\|_{L^2_\theta(0,\infty;L^2(I))}^2.
\end{aligned}
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
Together with \eqref{eq:global-weighted-estimate}, this gives
\begin{equation}
\label{eq:v-pointwise-decay}
\|v(t)\|_{L^2(\Omega)}
\le
C_ve^{-\theta t}\mathcal I_0.
\end{equation}

Finally,
\[
z(s,t)=U(t-D+s),
\qquad
U(t)=K_{\relax cr}P_{\relax cr}Y(t).
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
For \(t\ge D\), \eqref{eq:Y-pointwise-decay} gives
\[
\|z(t)\|_{\mathcal Z}
\le
C_ze^{-\theta t}\mathcal I_0.
\relax\ifmmode \egroup $\hfil \egroup
  \end{trivlist}                                                 
  \else \@badmath \fi
On the fixed interval \(0\le t<D\), the finite-time estimate of Theorem~\ref{thm:global-wellposedness} controls \(u\), \(v\), and \(z\), and the factor \(e^{\theta D}\) is absorbed into the constant.
Combining \eqref{eq:u-pointwise-decay}, \eqref{eq:v-pointwise-decay}, the transport-state estimate, and the finite-time bound on \([0,D]\) proves \eqref{eq:final-exponential-estimate}.
