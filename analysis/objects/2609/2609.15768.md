For $2\leq p<\infty$, and $k$, $m\in \mathbb{N}$ with $k\in [n]$ and
\[
m\geq \frac{n^{3/2}\log(p)}{\sqrt{k}}+pn,
\]
then there exists a constant $c_{p}>0$ depending only on $p$ such that the following inequality holds for every $f:\mathbb{Z}^{n}_{4m}\to L_{p}$
\[
\begin{split}
&\frac{(p\log(p))^{-1}}{\binom{n}{k}}\sum_{\substack{S\subseteq[n] \\|S|=k}}\frac{\mathbb{E}\left[\left\|f(x+2m\varepsilon_{S})-f(x)\right\|^{p}_{L_{p}}\right]}{m^{p}}\\
\leq& c_{p}\left(\frac{k}{n}\sum_{j=1}^{n}\mathbb{E}\left\|T_{j}f(x)-f(x)\right\|^{p}_{L_{p}}+\left(\frac{k}{n}\right)^{p/2}\mathbb{E}\left\|T^{\varepsilon}_{[n]}f(x)-f(x)\right\|^{p}_{L_{p}}\right),
\end{split}
\]
where the expectation is with respect to $(x,\varepsilon)\in \mathbb{Z}^{n}_{4m}\times \{-1,1\}^{n}$ chosen unifromly at random.

\label{NS-main}
For every $1\leq p<\infty$, there exists $c_{p}>0$ such that for every $n$, $m\in {\mathbb N}$, the following inequality holds for every $f:{\mathbb{Z}^{n}_{m}}\to \mathbb{R}$

<!-- lem-1: body_md -->
\label{representation 1}
For every $f:{\mathbb{Z}^{n}_{m}}\to \mathbb{R}$ and  each $j\in [n]$, the following identity holds
\[
\begin{split}
\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}\Big{|}\mathcal{F}_{j-1}\right](x,\varepsilon)=E_{[n]\setminus[j-1]}S^{\varepsilon}_{[j-1]}b^{(j)}_{f}(x),\quad (x,\varepsilon)\in {\mathbb{Z}^{n}_{m}}\times \Omega_{n}.
\end{split}
\]
<!-- end lem-1 -->
<!-- proof-1: body_md -->
By the definition of $\mathcal{F}_{j-1}$, we have 
\begin{equation}\label{expansion 1}
\begin{split}
&\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}\Big{|}\mathcal{F}_{j-1}\right](x,\varepsilon)\\
=&\frac{1}{2^{n-j+1}}\sum_{\varepsilon^{\prime}\in \Omega_{n-j+1}}\frac{b^{(j)}_{f}(x+\sum_{l=1}^{j-1}\varepsilon_{l}e_{l}+\sum_{l=j}^{n}\varepsilon^{\prime}_{l-j-1}e_{l})+b^{(j)}_{f}(x-\sum_{l=1}^{j-1}\varepsilon_{l}e_{l}-\sum_{l=j}^{n}\varepsilon^{\prime}_{l-j-1}e_{l})}{2}\\
=&\frac{1}{2^{n-j+1}}\sum_{\varepsilon^{\prime}\in \Omega_{n-j+1}}\frac{\left(T^{(\varepsilon_{1},\cdots, \varepsilon_{j-1},\varepsilon^{\prime}_{1},\cdots, \varepsilon^{\prime}_{n-j+1})}_{[n]}b^{(j)}_{f}\right)(x)+\left(T^{(-\varepsilon_{1},\cdots, -\varepsilon_{j-1},-\varepsilon^{\prime}_{1},\cdots, -\varepsilon^{\prime}_{n-j+1})}_{[n]}b^{(j)}_{f}\right)(x)}{2}\\
=&D_{j}E_{[n]\setminus\{j\}}\left(\frac{1}{2^{n-j+1}}\sum_{\varepsilon^{\prime}\in \Omega_{n-j+1}}\left(T^{(\varepsilon_{1},\cdots, \varepsilon_{j-1},\varepsilon^{\prime}_{1},\cdots, \varepsilon^{\prime}_{n-j+1})}_{[n]}b^{(j)}_{f}\right)(x)\right)\\
&+D_{j}E_{[n]\setminus\{j\}}\left(\frac{1}{2^{n-j+1}}\sum_{\varepsilon^{\prime}\in \Omega_{n-j+1}}\left(T^{(-\varepsilon_{1},\cdots, -\varepsilon_{j-1},-\varepsilon^{\prime}_{1},\cdots, -\varepsilon^{\prime}_{n-j+1})}_{[n]}b^{(j)}_{f}\right)(x)\right).
\end{split}
\end{equation}
It is clear that
\begin{equation}\label{expansion 2}
\frac{1}{2^{n-j+1}}\sum_{\varepsilon^{\prime}\in \Omega_{n-j+1}}\left(T^{(\varepsilon_{1},\cdots, \varepsilon_{j-1},\varepsilon^{\prime}_{1},\cdots, \varepsilon^{\prime}_{n-j+1})}_{[n]}b^{(j)}_{f}\right)(x)=T^{\varepsilon}_{[j-1]}E_{[n]\setminus[j-1]}f(x),
\end{equation}
and
\begin{equation}\label{expansion 3}
\frac{1}{2^{n-j+1}}\sum_{\varepsilon^{\prime}\in \Omega_{n-j+1}}\left(T^{(-\varepsilon_{1},\cdots, -\varepsilon_{j-1},-\varepsilon^{\prime}_{1},\cdots, -\varepsilon^{\prime}_{n-j+1})}_{[n]}b^{(j)}_{f}\right)(x)=T^{-\varepsilon}_{[j-1]}E_{[n]\setminus[j-1]}f(x).
\end{equation}
Substituting \eqref{expansion 2} and \eqref{expansion 3} into \eqref{expansion 1} yields that
\[
\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}\Big{|}\mathcal{F}_{j-1}\right](x,\varepsilon)=E_{[n]\setminus[j-1]}S^{\varepsilon}_{[j-1]}b^{(j)}_{f}(x),
\]
where we used the fact that all operators are commute with each other and the fact that $b^{(j)}_{f}=2D_{j}E_{[n]\setminus\{j\}}f$.

<!-- lem-2: body_md -->
\label{main decomposition}
For every $f:{\mathbb{Z}^{n}_{m}}\to \mathbb{R}$, we have
\begin{equation}\label{decomposition of Hf}
H_{f}(x,\varepsilon)=\sum_{j=1}^{n}\varepsilon_{j}\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}\Big{|}\mathcal{F}_{j-1}\right](x,\varepsilon), \quad (x,\varepsilon)\in {\mathbb{Z}^{n}_{m}}\times \Omega_{n}.
\end{equation}
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Note here that $H_{f}=E_{[n]}\left(T^{\varepsilon}_{[n]}-T^{-\varepsilon}_{[n]}\right)f$. For each $j\in [n]$, we define
\begin{equation}\label{decomposition 1}
\mathcal{M}_{j}(f)\coloneqq E_{[n]}E_{[n]\setminus[j]}\left(T^{\varepsilon}_{[j]}-T^{-\varepsilon}_{[j]}\right)(f),
\end{equation}
and let $\mathcal{M}_{0}(f)\equiv 0$ for convenience. Since $E_{j}=\frac{T_{j}+T^{-1}_{j}}{2}$ and $D_{j}=\frac{T_{j}-T^{-1}_{j}}{2}$, it is clear that
\begin{equation}\label{key 1}
T_{j}=E_{j}+D_{j}\qquad\mbox{and}\qquad T^{-1}_{j}=E_{j}-D_{j}.
\end{equation}
By $T^{\varepsilon}_{[j]}=T^{\varepsilon_{j}}_{j}T^{\varepsilon}_{[j-1]}$ and $T^{-\varepsilon}_{[j]}=T^{-\varepsilon_{j}}_{j}T^{-\varepsilon}_{[j-1]}$, it follows from \eqref{key 1} that
\begin{equation}\label{decomposition 2}
T^{\varepsilon}_{[j]}-T^{-\varepsilon}_{[j]}=(E_{j}+\varepsilon_{j}D_{j})T^{\varepsilon}_{[j-1]}-(E_{j}-\varepsilon_{j}D_{j})T^{-\varepsilon}_{[j-1]}.
\end{equation}
Substituting \eqref{decomposition 2} into \eqref{decomposition 1} yields that
\begin{equation}\label{decomposition 3}
\begin{split}
\mathcal{M}_{j}(f)&=E_{[n]}E_{[n]\setminus[j]}E_{j}\left(T^{\varepsilon}_{[j-1]}-T^{-\varepsilon}_{[j-1]}\right)(f)+2\varepsilon_{j}E_{[n]}E_{[n]\setminus[j]}D_{j}S^{\varepsilon}_{[j-1]}(f)\\
&=\mathcal{M}_{j-1}(f)+2\varepsilon_{j}E_{[n]}E_{[n]\setminus[j]}D_{j}S^{\varepsilon}_{[j-1]}(f)\\
&=\mathcal{M}_{j-1}(f)+\varepsilon_{j}E_{[n]\setminus[j-1]}S^{\varepsilon}_{[j-1]}b^{(j)}_{f}\\
&=\mathcal{M}_{j-1}(f)+\varepsilon_{j}\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}|\mathcal{F}_{j-1}\right],
\end{split}
\end{equation}
where the last equality is due to Lemma  \ref{representation 1}. Noting $M_{n}(f)=H_{f}$,  by induction of \eqref{decomposition 3},  we obtain
\[
H_{f}=\sum_{j=1}^{n}\varepsilon_{j} \mathbb{E}\left[\mathcal{S}b^{(j)}_{f}\Big{|}\mathcal{F}_{j-1}\right].
\]

<!-- lem-3: body_md -->
\label{representation of martingale difference}
For each $f:{\mathbb{Z}^{n}_{m}}\times \Omega_{n}\to \mathbb{R}$, we have
\[
d_{j}(f)\coloneqq\mathbb{E}\left[f\Big{|}\mathcal{F}_{j}\right]-\mathbb{E}\left[f\Big{|}\mathcal{F}_{j-1}\right]=\varepsilon_{j}\mathbb{E}\left[\varepsilon_{j}f\Big{|}\mathcal{F}_{j-1}\right],\quad1\leq j\leq n.
\]
<!-- end lem-3 -->
<!-- proof-3: body_md -->
The proof follows from the Walsh expansion directly. Indeed, for $f:{\mathbb{Z}^{n}_{m}}\times \Omega_{n}\to \mathbb{R}$, we define $F:{\mathbb{Z}^{n}_{m}}\to L_{2}(\Omega_{n};\mathbb{R})$ by
\[
F(x)\coloneqq f(x,\cdot),\quad x\in {\mathbb{Z}^{n}_{m}}.
\]
Since $\{w_{A}\}_{A\subseteq [n]}$ forms an orthogonal basis of $L_{2}(\Omega_{n})$, it follows that for each $x\in {\mathbb{Z}^{n}_{m}}$,  we have
\[
F(x)=\sum_{A\subseteq [n]}\widehat{F(x)}(A)w_{A}.
\]
By the definition of $\{\mathcal{F}_{j}\}_{j=0}^{n}$, we have
\[
\mathbb{E}[F|\mathcal{F}_{j}](x)=\sum_{A\subseteq [j]}\widehat{F(x)}(A)w_{A}.
\]
Thus,
\begin{equation}\label{Walsh expansion 1}
\begin{split}
d_{j}(f)(x,\cdot)&=\sum_{\substack{A\subseteq [n]\\\max(A)=j}}\widehat{F(x)}(A)w_{A}\\
&=\sum_{B\subseteq [j-1]}\widehat{F(x)}(B\cup\{j\})w_{B\cup\{j\}}=\varepsilon_{j}\cdot\left(\sum_{B\subseteq [j-1]}\widehat{F(x)}(B\cup\{j\})w_{B}\right).
\end{split}
\end{equation}
On th other hand, by the definition of conditional expectation, we have
\begin{equation}\label{Walsh expansion 2}
\begin{split}
\mathbb{E}\left[\varepsilon_{j} F\Big{|}\mathcal{F}_{j-1}\right](x,\cdot)&=\mathbb{E}\left[\sum_{j\in A\subseteq [n]}\widehat{F(x)}(A)w_{A\setminus\{j\}}\Big{|}\mathcal{F}_{j-1}\right]\\
&\quad +\mathbb{E}\left[\sum_{A\subseteq [n]\setminus\{j\}}\widehat{F(x)}(A)w_{A\cup\{j\}}\Big{|}\mathcal{F}_{j-1}\right]\\
&=\sum_{\substack{A\subseteq[n]\\\max(A)=j}}\widehat{F(x)}(A)w_{A\setminus\{j\}}=\sum_{B\subseteq [j-1]}\widehat{F(x)}(B\cup\{j\})w_{B}.
\end{split}
\end{equation}
Combining \eqref{Walsh expansion 1} with \eqref{Walsh expansion 2} yields the desired result.

<!-- thm-4: body_md -->
\label{Burkholder martingale transform}
Suppose that $M=\{M_{j}\}_{j=0}^{\infty}$ martingale with $M_{0}\equiv 0$. Let $d_{j}\coloneqq M_{j}-M_{j-1}$ be the martingale difference for each $j\in \mathbb{N}$. If $\{\varepsilon_{j}\}_{j=1}^{n}$ is a sequence of numbers in $\{-1,1\}$, then the following inequality holds
\[

\]
where $p^{*}=\max\{p,p/(p-1)\}$

<!-- thm-5: body_md -->
\label{NS-main-new}
For every $1<p<\infty$, there exists $c_{p}>0$ such that for every $n$, $m\in {\mathbb N}$, the following inequality holds for every $f:{\mathbb{Z}^{n}_{m}}\to \mathbb{R}$

<!-- proof-4: body_md -->
By Lemma \ref{main decomposition}, we have
\[
H_{f}(x,\varepsilon)=\sum_{j=1}^{n}\varepsilon_{j}\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}|\mathcal{F}_{j-1}\right](x,\varepsilon).
\]
Hence, for each $g\in L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})$ with $\|g\|_{L_{p^{\prime}}}\leq 1$ and $p^{\prime}=\frac{p}{p-1}$, we have
\begin{equation}\label{martingale transform 1}
\begin{split}
\mathbb{E}_{\nu\otimes \mu}\left[gH_{f}\right]&=\mathbb{E}_{\nu\otimes \mu}\left[\left(\sum_{j=1}^{n}\mathbb{E}[\varepsilon_{j} g|\mathcal{F}_{j-1}]\right)\cdot\mathcal{S}b^{(j)}_{f}\right]\\
&=\mathbb{E}_{\nu\otimes \mu}\left[\sum_{j=1}^{n}d_{j}(g)\cdot \varepsilon_{j} \mathcal{S}b^{(j)}_{f}\right],
\end{split}
\end{equation}
where we used Lemma \ref{representation of martingale difference}.

Let $(\delta^{\prime}_{1},\cdots, \delta^{\prime}_{n})$ be an i.i.d. Rademacher sequence (on a probability space $(\Omega,\mathcal{B},\mu^{\prime})$), which is independent of $(\varepsilon_{1},\cdots,\varepsilon_{n})$. It is clear that for each $\{\alpha_{j}\}_{j=1}^{n}$ and $\{\beta_{j}\}_{j=1}^{n}\subseteq \mathbb{R}$ we have
\begin{equation}\label{martingale transform 2}
\sum_{j=1}^{n}\alpha_{j}\beta_{j}=\mathbb{E}_{\mu^{\prime}}\left[\left(\sum_{j=1}^{n}\delta_{j}\alpha_{j}\right)\cdot\left(\sum_{j=1}^{n}\delta_{j}\beta_{j}\right)\right].
\end{equation}
Substituting \eqref{martingale transform 2} into \eqref{martingale transform 1}  yields
\begin{equation}\label{martingale transform 3}
\begin{split}
|\mathbb{E}_{\nu\otimes \mu}\left[gH_{f}\right]|&=\mathbb{E}_{\nu\otimes \mu\otimes \mu^{\prime}}\left[\left(\sum_{j=1}^{n}\delta_{j}d_{j}(g)\right)\cdot \left(\sum_{j=1}^{n}\delta_{j}\varepsilon_{j} \mathcal{S}b^{(j)}_{f}\right)\right]\\
&\leq \left\|\sum_{j=1}^{n}\delta_{j}d_{j}(g)\right\|_{L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n}\times \Omega)}\cdot\left\|\sum_{j=1}^{n}\delta_{j}\varepsilon_{j} \mathcal{S}b^{(j)}_{f}\right\|_{L_{p}({\mathbb{Z}^{n}_{m}}\times \Omega_{n}\times \Omega)}
\end{split}
\end{equation}
Note that for each $j\in [n]$ the random variable $\delta_{j}\varepsilon_{j}$ has the same distribution of $\varepsilon_{j}$, then it follows that
\begin{equation}\label{martingale transform 4}
\left\|\sum_{j=1}^{n}\delta_{j}\varepsilon_{j} \mathcal{S}b^{(j)}_{f}\right\|_{L_{p}({\mathbb{Z}^{n}_{m}}\times \Omega_{n}\times \Omega)}=\left\|\sum_{j=1}^{n}\varepsilon_{j}\mathcal{S}b^{(j)}_{f}\right\|_{L_{p}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})}\leq \left\|\sum_{j=1}^{n}\varepsilon_{j}b^{(j)}_{f}\right\|_{L_{p}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})},
\end{equation}
where we used the fact that $\mathcal{S}$ is a contraction.

We now apply Theorem \ref{Burkholder martingale transform} to bound $\left\|\sum_{j=1}^{n}\delta_{j}d_{j}(g)\right\|_{L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n}\times \Omega)}$. For each $\omega\in \Omega$, by Theorem \ref{Burkholder martingale transform}, we have
\[
\left\|\sum_{j=1}^{n}\delta_{j}(\omega)d_{j}(g)\right\|_{L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})}\leq (p^{*}-1)\|g\|_{L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})}.
\]
By the Fubini theorem, it yields that
\begin{equation}\label{martingale transform 5}
\begin{split}
\left\|\sum_{j=1}^{n}\delta_{j}d_{j}(g)\right\|_{L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n}\times \Omega)}&=\left(\mathbb{E}_{\mu^{\prime}}\left[\left\|\sum_{j=1}^{n}\delta_{j}(\omega)d_{j}(g)\right\|^{p^{\prime}}_{L_{p^{\prime}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})}}\right]^{p^{\prime}}\right)^{1/p^{\prime}}\\
&\leq (p^{*}-1)\|g\|_{L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})}\leq p^{*}-1.
\end{split}
\end{equation}
Combining \eqref{martingale transform 3}, \eqref{martingale transform 4}, and \eqref{martingale transform 5} together yields
\[
\|H_{f}\|_{L_{p}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})}=\sup_{\|g\|_{L_{p^{\prime}}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})}\leq 1}|\mathbb{E}_{\nu\otimes \mu}[gH_{f}]|\leq (p^{*}-1)\left\|\sum_{j=1}^{n}\varepsilon_{j}b^{(j)}_{f}\right\|_{L_{p}({\mathbb{Z}^{n}_{m}}\times \Omega_{n})},
\]
which completes our proof.

<!-- lem-4: body_md -->
\label{endpoint cube estimate}
For every $g:\Omega_{n}\to\mathbb{R}$, the following inequality holds
\begin{equation}\label{endpoint inverse gradient}
\left\|\left(\sum_{j=1}^{n}\left|\partial_{j}\Delta^{-1}g\right|^{2}\right)^{1/2}\right\|_{L_{\infty}(\Omega_{n})}
\leq\frac{\pi}{2}\|g\|_{L_{\infty}(\Omega_{n})}.
\end{equation}
<!-- end lem-4 -->
<!-- proof-5: body_md -->
By Proposition \ref{IvHv}, we have
\begin{equation}\label{orthonormal 1}
\sum_{j=1}^{n}\left|\partial_{j}e^{-t\Delta}(g)(\varepsilon)\right|^{2}=\frac{1}{e^{2t}-1}\sum_{j=1}^{n}\mathbb{E}_{\xi(t)}\left|\left[\delta_{j}(t)g\left(\varepsilon_{1}\xi_{1}(t),\cdots,\varepsilon_{n}\xi_{n}(t)\right)\right]\right|^{2}.
\end{equation}
Note that $\{\delta_{j}\}_{j=1}^{n}$ forms an orthonormal sequence, hence \eqref{orthonormal 1} yields that
\begin{equation}\label{orthonormal 2}
\sum_{j=1}^{n}\left|\partial_{j}e^{-t\Delta}(g)(\varepsilon)\right|^{2}\leq \frac{1}{e^{2t}-1}\|g\|^{2}_{L_{2}(\Omega_{n},\mu_{\xi(t)})}\leq \frac{1}{e^{2t}-1}\|g\|^{2}_{L_{\infty}(\Omega_{n})}.
\end{equation}
By the spectral theorem of $\Delta$, we have
\begin{equation}\label{spectrum decomposition}
\Delta^{-1}=\int_{0}^{\infty}e^{-t\Delta}~dt.
\end{equation}
Combining \eqref{orthonormal 2} with \eqref{spectrum decomposition} yields that
\begin{align*}
\left(\sum_{j=1}^{n}\left|\partial_{j}\Delta^{-1}(g)(\varepsilon)\right|^{2}\right)^{1/2}&\leq \left(\int_{0}^{\infty}\frac{1}{\sqrt{e^{2t}-1}}~dt\right)\|g\|_{L_{\infty}(\Omega_{n})}\\
&=\frac{\pi}{2}\|g\|_{L_{\infty}}.
\end{align*}
This completes our proof.

<!-- thm-6: body_md -->
\label{NS L1 endpoint}
For every $n$, $m\in\mathbb{N}$ and every $f:{\mathbb{Z}^{n}_{m}}\to\mathbb{R}$,
\begin{equation}\label{endpoint L1 inequality}
\|H_{f}\|_{L_{1}({\mathbb{Z}^{n}_{m}}\times\Omega_{n})}\leq\frac{\pi}{2}\left\|\left(\sum_{j=1}^{n}\left|b^{(j)}_{f}\right|^{2}\right)^{1/2}\right\|_{L_{1}({\mathbb{Z}^{n}_{m}})}\leq\frac{\pi}{\sqrt{2}}\left\|\sum_{j=1}^{n}\varepsilon_{j}b^{(j)}_{f}\right\|_{L_{1}({\mathbb{Z}^{n}_{m}}\times\Omega_{n})}.
\end{equation}
<!-- end thm-6 -->
<!-- proof-6: body_md -->
To prove the first inequality, we need the following key identities. For any $x\in {\mathbb{Z}^{n}_{m}}$ and $\varepsilon\in \Omega_{n}$, we have
\begin{equation}\label{Hf decomposition}
\begin{split}
\partial_{j}H_{f}(x,\varepsilon)&=\frac{E_{[n]}\left(T^{\varepsilon}_{[n]}-T^{-\varepsilon}_{[n]}\right)(f)(x)-E_{[n]}\left(T^{\varepsilon_{\oplus_{j}}}_{[n]}-T^{-\varepsilon_{\oplus_{j}}}_{[n]}\right)(f)(x)}{2}\\
&=\frac{1}{2}E_{[n]}\left(T^{\varepsilon}_{[n]\setminus\{j\}}\left(T^{\varepsilon_{j}}_{j}-T^{-\varepsilon_{j}}_{j}\right)(f)(x)+T^{-\varepsilon}_{[n]\setminus\{j\}}\left(T^{\varepsilon_{j}}_{j}-T^{-\varepsilon_{j}}_{j}\right)(f)(x)\right)\\
&=E_{[n]}S^{\varepsilon}_{[n]\setminus{j}}(T^{\varepsilon_{j}}_{j}-T^{-\varepsilon_{j}}_{j})(f)(x)\\
&=2\varepsilon_{j}D_{j}E_{[n]}S^{\varepsilon}_{[n]\setminus\{j\}}(f)(x),
\end{split}
\end{equation}
where we used $\varepsilon_{j}D_{j}=\frac{T^{\varepsilon_{j}}_{j}-T^{-\varepsilon_{j}}_{j}}{2}$ in the last line.

On the other hand, applying the same argument to $\mathcal{S}b^{(j)}_{f}$ we obtain that
\begin{equation}\label{bjf decomposition}
\partial_{j}\left(\varepsilon_{j}\mathcal{S}b^{(j)}_{f}\right)(x,\varepsilon)=\varepsilon_{j}E_{j}S^{\varepsilon}_{[n]\setminus\{j\}}b^{(j)}_{f}(x)=2\varepsilon_{j}D_{j}E_{[n]}S^{\varepsilon}_{[n]\setminus\{j\}}f(x).
\end{equation}
Hence, for each $x\in Z$ and $\varepsilon\in \Omega_{n}$, we have
\begin{equation}\label{endpoint divergence identity}
\Delta H_{f}(x,\varepsilon)=\sum_{j=1}^{n}\partial_{j}\left(\varepsilon_{j}\mathcal{S}b^{(j)}_{f}\right)(x,\varepsilon).
\end{equation}
For any fixed $x\in {\mathbb{Z}^{n}_{m}}$, it is clear that $H_{f}(x,-\varepsilon)=-H_{f}(x,\varepsilon)$, and the symmetry gives $\mathbb{E}_{\mu}H_{f}(x,\cdot)=0$. Hence, \eqref{endpoint divergence identity} is equivalent to the following
\begin{equation}\label{divergence 2}
H_{f}(x,\varepsilon)=\sum_{j=1}^{n}\Delta^{-1}D_{j}\left(\varepsilon_{j}\mathcal{S}b^{(j)}_{f}\right)(x,\varepsilon),
\end{equation}
for each $(x,\varepsilon)\in {\mathbb{Z}^{n}_{m}}\times \Omega_{n}$.

Let $g:\Omega_{n}\to\mathbb{R}$ satisfy $\|g\|_{L_{\infty}(\Omega_{n})}\leq1$. By the Cauchy--Schwarz inequality and \eqref{divergence 2}, we obtain from Lemma \ref{endpoint cube estimate} that
\[
\begin{split}
\left|\mathbb{E}_{\mu}\left[gH_{f}(x,\cdot)\right]\right|
&=\left|\mathbb{E}_{\mu}\left[\sum_{j=1}^{n}D_{j}\Delta^{-1}g\cdot\varepsilon_{j}\mathcal{S}b^{(j)}_{f}(x,\cdot)\right]\right|\\
&\leq\frac{\pi}{2}\mathbb{E}_{\mu}\left[\left(\sum_{j=1}^{n}\left|\mathcal{S}b^{(j)}_{f}(x,\cdot)\right|^{2}\right)^{1/2}\right].
\end{split}
\]
Taking the supremum over $g\in L_{\infty}(\Omega_{n})$ with $\|g\|_{L_{\infty}(\Omega_{n})}\leq 1$, we have
\begin{equation}\label{endpoint pointwise L1}
\begin{split}
\|H_{f}(x,\cdot)\|_{L_{1}(\Omega_{n})}
&\leq\frac{\pi}{4}\mathbb{E}_{\mu}\left[\left(\sum_{j=1}^{n}\left|b^{(j)}_{f}(x+\varepsilon)\right|^{2}\right)^{1/2}
+\left(\sum_{j=1}^{n}\left|b^{(j)}_{f}(x-\varepsilon)\right|^{2}\right)^{1/2}\right]\\
&=\frac{\pi}{2}E_{[n]}\left[\left(\sum_{j=1}^{n}\left|b^{(j)}_{f}\right|^{2}\right)^{1/2}\right](x),
\end{split}
\end{equation}
for every $x\in {\mathbb{Z}^{n}_{m}}$. Taking the expectation with respect to $x\in {\mathbb{Z}^{n}_{m}}$, \eqref{endpoint pointwise L1} and the contraction of $E_{[n]}$ yield the first inequality in \eqref{endpoint L1 inequality}. For the second inequality, it suffices to use the sharp Khintchine inequality \cite{Sza1976} to get the desired bound.

<!-- prop-2: body_md -->
\label{NS Linfty counterexample}
For every integer $k\geq1$, there is a real function $f$ on $\mathbb{Z}_{3}^{2k+1}$ such that
\begin{equation}\label{endpoint counterexample bounds}
\left\|\sum_{j=1}^{2k+1}\varepsilon_{j}b^{(j)}_{f}\right\|_{L_{\infty}(\mathbb{Z}_{3}^{2k+1}\times\Omega_{2k+1})}=1,
\qquad
\|H_{f}\|_{L_{\infty}(\mathbb{Z}_{3}^{2k+1}\times\Omega_{2k+1})}
=\sum_{r=1}^{k}\frac{1}{r}.
\end{equation}
<!-- end prop-2 -->
<!-- proof-7: body_md -->
Set $n=2k+1$ and $m=3$. Since $T_{j}^{3}=\mathrm{Id}$, the operator
\[
P_{j}\coloneqq\frac{\mathrm{Id}+T_{j}+T_{j}^{-1}}{3}
\]
is a projection with $D_{j}P_{j}=0$. Consequently, we have
\[
E_{j}=\frac{3P_{j}-\mathrm{Id}}{2},\qquad
E_{j}^{-1}=3P_{j}-2\mathrm{Id},\qquad
D_{j}E_{j}^{-1}=-2D_{j}.
\]
For any $u:\mathbb{Z}^{n}_{m}\to \mathbb{R}$ let
\[
f\coloneqq E_{[n]}^{-1}u=\prod_{j=1}^{n}(3P_{j}-2\mathrm{Id})u.
\]
Then for this $f$ we have
\begin{equation}\label{endpoint ternary reduction}
H_{f}(x,\varepsilon)=u(x+\varepsilon)-u(x-\varepsilon),\qquad
b^{(j)}_{f}(x)=-2\left(u(x+e_{j})-u(x-e_{j})\right).
\end{equation}

\noindent For $t\in\{0,\cdots,n\}$, define
\[
\varphi(t)\coloneqq
\begin{cases}
0,&0\leq t\leq k+1,\\[2pt]
\displaystyle\frac{1}{2}\sum_{r=n-t+1}^{k}\frac{1}{r},&k+2\leq t\leq n.
\end{cases}
\]
Thus, for $0\leq t<n$,
\begin{equation}\label{endpoint harmonic increments}
\varphi(t+1)-\varphi(t)=
\begin{cases}
0,&0\leq t\leq k,\\
\displaystyle\frac{1}{2(n-t)},&k+1\leq t<n.
\end{cases}
\end{equation}
Represent the elements of $\mathbb{Z}_{3}$ by $\bar{0}$, $\bar{1}$, and $\bar{2}$, and put
\[
u(x)\coloneqq\varphi\left(|\{j\in [n]:x_{j}=\bar{1}\}|\right)-\varphi\left(|\{j\in [n]:x_{j}=\bar{2} \}|\right).
\]
Fix $x\in {\mathbb{Z}^{n}_{m}}$ and write $a=|\{j\in[n]:x_{j}=\bar{1} \}|$ and $b=|\{j\in [n]:x_{j}=\bar{2} \}|$. A direct computation gives
\begin{equation}\label{endpoint coordinate differences}
u(x+e_{j})-u(x-e_{j})=
\begin{cases}
\varphi(a+1)-\varphi(a)+\varphi(b+1)-\varphi(b),&x_{j}=\bar{0},\\
\varphi(b)-\varphi(b+1),&x_{j}=\bar{1},\\
\varphi(a)-\varphi(a+1),&x_{j}=\bar{2}.
\end{cases}
\end{equation}
                                                                                             

If $a\leq k$ and $b\leq k$, all differences in \eqref{endpoint coordinate differences} vanish. If $a\geq k+1$, by $a+b\leq2k+1$, $b\leq k$. The coordinates with $x_{j}=\bar{1}$ give zero; if $a<n$, each of the remaining $n-a$ coordinates gives absolute difference $1/(2(n-a))$. If $a=n$, every difference is zero. Interchanging $a$ and $b$ proves the remaining case. By \eqref{endpoint ternary reduction}, this establishes
\[
\sum_{j=1}^{n}\left|b^{(j)}_{f}(x)\right|\leq1
\quad\mbox{for every }x\in\mathbb{Z}_{3}^{n}.
\]
Equality holds, for example, when $a=k+1$ and $b=k$. Applying the fact
\[
\left\|\sum_{j=1}^{n}\varepsilon_{j}\alpha_{j}\right\|_{L_{\infty}(\Omega_{n})}=\sum_{j=1}^{n}|\alpha_{j}|,
\]
for all $\{\alpha_{j}\}_{j=1}^{n}\subseteq \mathbb{R}$, we derive the first equality in \eqref{endpoint counterexample bounds}.

Let $\mathbf{1}=(1,\cdots,1)\in\Omega_{n}$. For $x=0$ and $\varepsilon=\mathbf{1}$, the first equality in \eqref{endpoint ternary reduction} turns into
\[
H_{f}(0,\mathbf{1})=2\varphi(n)=\sum_{r=1}^{k}\frac{1}{r}.
\]
Since $-\varphi(n)\leq u(x)\leq\varphi(n)$ for every $x\in \mathbb{Z}^{n}_{m}$, it entails that
\[
\|H_{f}\|_{L_{\infty}(\mathbb{Z}^{n}_{m}\times \Omega_{n})}=2\varphi(n).
\]
This proves the second equality in \eqref{endpoint counterexample bounds}. Finally,
\[
\sum_{r=1}^{k}\frac{1}{r}\geq\int_{1}^{k+1}\frac{dt}{t}=\log(k+1)\longrightarrow\infty,
\]
which proves the asserted failure.

Since the summands in \eqref{decomposition of Hf} are martingale differences, Lemma \ref{main decomposition} gives
\begin{equation}\label{endpoint H differences}
d_{j}(H_{f})=\varepsilon_{j}\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}\Big{|}\mathcal{F}_{j-1}\right].
\end{equation}
For $k<i<j$, the tower property and $\mathbb{E}[d_{j}(H_{f})|\mathcal{F}_{j-1}]=0$ imply
\begin{equation}\label{conditional orthonormal 1}
\mathbb{E}\left[d_{i}(H_{f})d_{j}(H_{f})\Big{|}\mathcal{F}_{k}\right]=0.
\end{equation}
Therefore, for $0\leq k<n$, conditional orthogonality, we apply \eqref{endpoint H differences} and \eqref{conditional orthonormal 1} to obtain that
\begin{equation}\label{endpoint conditional energy}
\begin{split}
\mathbb{E}\left[\left|H_{f}-\mathbb{E}[H_{f}|\mathcal{F}_{k}]\right|^{2}\Big{|}\mathcal{F}_{k}\right]&=\sum_{j=k+1}^{n}\mathbb{E}\left[\left|\mathbb{E}\left[\mathcal{S}b^{(j)}_{f}\Big{|}\mathcal{F}_{j-1}\right]\right|^{2}\Big{|}\mathcal{F}_{k}\right]\\
&\leq\mathbb{E}\left[\sum_{j=k+1}^{n}\left|\mathcal{S}b^{(j)}_{f}\right|^{2}\Big{|}\mathcal{F}_{k}\right],
\end{split}
\end{equation}
where the last inequality follows from the Jensen inequality. For each $(x,\varepsilon)\in {\mathbb{Z}^{n}_{m}}\times \Omega_{n}$, the definition of $\mathcal{S}$ yields
\begin{equation}\label{maximal estimate 1}
\sum_{j=k+1}^{n}\left|\mathcal{S}b^{(j)}_{f}(x,\varepsilon)\right|^{2}\leq\frac{1}{2}\sum_{j=k+1}^{n}\left(\left|b^{(j)}_{f}(x+\varepsilon)\right|^{2}+\left|b^{(j)}_{f}(x-\varepsilon)\right|^{2}\right)\leq\max_{y\in{\mathbb{Z}^{n}_{m}}}\sum_{j=1}^{n}\left|b^{(j)}_{f}(y)\right|^{2}.
\end{equation}
Combining \eqref{endpoint conditional energy} with \eqref{maximal estimate 1}, we obtain
\[
\left\|H_{f}\right\|_{\mathrm{bmo}_{2}(\mathcal{F})}\leq \left\|\left(\sum_{j=1}^{n}\left|b^{(j)}_{f}(y)\right|^{2}\right)^{1/2}\right\|_{L_{\infty}({\mathbb{Z}^{n}_{m}})}.
\]
Furthermore, by the filtration $\mathcal{F}=\{\mathcal{F}_{j}\}_{j=0}^{n}$ as in Section \ref{main section}, the following equalities hold
\[
\mathbb{E}\left[\sum_{j=1}^{n}\varepsilon_{j}b^{(j)}_{f}\Big{|}\mathcal{F}_{k}\right]=\sum_{j=1}^{k}\varepsilon_{j}b^{(j)}_{f}
\]
and
\[
\mathbb{E}\left[\left|\sum_{j=k+1}^{n}\varepsilon_{j}b^{(j)}_{f}\right|^{2}\Big{|}\mathcal{F}_{k}\right]=\sum_{j=k+1}^{n}\left|b^{(j)}_{f}\right|^{2}.
\]
The maximum over $k$ is attained at $k=0$, which establishes the first inequality in \eqref{endpoint BMO inequality}.
