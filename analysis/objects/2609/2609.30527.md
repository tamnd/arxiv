<!-- soul--1: body_md -->
If $(M,g)$ is a complete noncompact Riemannian manifold with nonnegative sectional curvature, then it contains a closed totally convex submanifold $S \subseteq M$ such that $M$ is diffeomorphic to the normal bundle over $S$.\\
	If $\sec > 0$, then $S$ is a single point, which implies $M$ is diffeomorphic to the Euclidean space.
<!-- end soul--1 -->
<!-- split--1: body_md -->
If $(M,g)$ contains a line and has $\operatorname{Ric} \ge 0$, then $(M,g)$ is isometric to a product $(H \times \mathbb{R}, g_0 + dt^2)$.

<!-- thm-1: body_md -->
\label{WuThm1}
	An $n$-dimensional compact weighted Riemannian manifold $(M,g,e^{-\varphi})$ with non-empty boundary $\partial M$ is diffeomorphic to the standard closed $n$-ball if either of the following condition holds:
	\begin{enumerate}
		\item $\widetilde{\Lambda} > 0$ and $\min_{\partial M} \widetilde{\Lambda} + \rho_0\min_M \overline{\sec}_\varphi > 0$, where $\rho_0$ denotes the maximum distance from $\partial M$.
		\item $\widetilde{\Lambda} \ge 0$ and $\overline{\sec}_\varphi \ge 0$, and in addition, $\overline{\sec}_\varphi > 0$ in a neighborhood of $\partial M$.
		\end{enumerate}
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\label{WuThm2}
	Let $(M,g,e^{-\varphi})$ be an $n$-dimensional compact weighted Riemannian manifold with non-empty boundary $\partial M$. Let $1 \le k \le n$ be some integer. Suppose $\widetilde{\Lambda}_k \ge 0$ and $\overline{\operatorname{Ric}}_{k,\varphi} \ge 0$, then
                   
                                                                                                                                                                         
                                                                                                                                          
                 
	\begin{enumerate}
		\item $M$ has the homotopy type of a CW complex with a finite number of cells of dimensions $\le k - 1$.
		\item $M$ also has the homotopy type of a CW complex obtained from $\partial M$ by attaching a finite number of cells each of dimension $\ge n - k + 1$.
	\end{enumerate}
<!-- end thm-2 -->
<!-- thm-3: body_md -->
\label{Shen's_Result}
	Let $(M,g,e^{-\varphi})$ be a proper open weighed Riemannian manifold such that $e^\varphi$ grows strictly sublinearly. For some $1 \le k \le n-1$, suppose $\overline{\operatorname{Ric}}_{k,\varphi} \ge 0$ and $\overline{\operatorname{Ric}}_{k,\varphi} > 0$ outside of a compact set. Then $M$ has the homotopy type of a CW complex with (possibly infinitely many) cells each of dimension $\le k - 1$. This implies that
	$$H_i(M; \mathbb{Z}) = 0$$
	for all $i \ge k$.
<!-- end thm-3 -->
<!-- thm-4: body_md -->
\label{AliceLim'sResult}
	Let $M$ be an $n$-dimensional complete weighted manifold that's noncompact. If $\operatorname{Ric}^N_\varphi > 0$ with $\varphi < K$ for some $K \in \mathbb{R}$ and $N \le 1$, then
	$$H_{n-1}(M,\mathbb{Z}) = 0.$$
<!-- end thm-4 -->
<!-- thm-5: body_md -->
\label{Frankel's_Theorem}
	In a complete connected weighted Riemannian manifold $(M,g,e^{-\varphi})$ with $\overline{\operatorname{Ric}}_{k,\varphi}(M) > 0$, let $N_1, N_2 \subseteq M$ be $r$-dimensional and $s$-dimensional totally geodesic submanifolds with respect to the metric $\tilde{g}$, respectively. If $s + r \ge n + k - 1$, then
	$$N_1 \cap N_2 \not= \varnothing.$$

<!-- prop-1: body_md -->
\label{Second_Variation_Formula}
	Suppose $(M,g,e^{-\varphi})$ is a weighted Riemannian manifold. Let $\gamma: [a,b] \to M$ be a unit-speed $\nabla$-geodesic segment, let $\theta : [0,1] \times [a,b] \to M$ be a variation of $\gamma$, and let $V = \dfrac{d\theta}{ds}$ denote its variation field. Then the second variation of the arc length $L_g(\theta_s)$ is given by
	\begin{align*}
		\dfrac{d^2}{ds^2}L_g(\theta_s) 
		=& \int_a^b \big|D_t V^\perp - d\varphi(\dot\gamma)V^\perp\big|^2 
		- g\big(R^\varphi(V^\perp, \dot\gamma)\dot\gamma, V^\perp\big) \, dt \Big|_{s=0} \\
		&+ \Big(d\varphi(\dot\gamma)|V^\perp|^2 + g(D_s V, \dot\gamma)\Big) \Big|_{t=a,\, s=0}^{t=b,\, s=0} \\
		=& \int_a^b \big|D_t V - d\varphi(\dot\gamma)V\big|^2 
		- g(D_t V, \dot\gamma)^2 
		- g\big(R^\varphi(V, \dot\gamma)\dot\gamma, V\big) \, dt \Big|_{s=0} \\
		&+ \Big(d\varphi(\dot\gamma)|V|^2 + g(D_s V, \dot\gamma)\Big) \Big|_{t=a,\, s=0}^{t=b,\, s=0},
	\end{align*}
	where $V^\perp$ denotes the normal component of $V$.
<!-- end prop-1 -->
<!-- proof-1: body_md -->
Denote $S = D_s\theta$ and $T = D_t\theta$. Using the identity
	$$g(R^\varphi(V^\perp, \dot{\gamma})\dot{\gamma},V^\perp) = g(R(V^\perp, \dot{\gamma})\dot{\gamma},V^\perp) + \operatorname{Hess}\, \varphi (\dot{\gamma}, \dot{\gamma})|V^\perp|^2 + df(\dot{\gamma})^2|V^\perp|^2,$$
	and substituting into the classical second variation formula \cite[page 300 to 301]{Lee}, a direct computation yields the first equality.
	\begin{align*}
		\dfrac{d^2}{ds^2} \Bigg|_{s = 0} L_g(\theta_s) =& \int_a^b |D_t V^\perp|^2 - g(R(V^\perp, \dot{\gamma}) \dot{\gamma}, V^\perp) \; dt \Bigg|_{s=0} + g(D_s V, \dot{\gamma}) \Bigg|_{t = a, s = 0}^{t=b, s = 0}\\
		=& \int_a^b |D_t V^\perp|^2 - g(R^\varphi(V^\perp, \dot{\gamma}) \dot{\gamma}, V^\perp) + \operatorname{Hess}\,\varphi(\dot{\gamma}, \dot{\gamma})|V^\perp|^2 + d\varphi(\dot{\gamma})^2|V^\perp|^2 \; dt \Bigg|_{s = 0}\\
		&+ g(D_s V, \dot{\gamma}) \Bigg|_{t=a, s = 0}^{t=b, s = 0}\\
		=& \int_a^b |D_t V^\perp|^2 - g(R^\varphi(V^\perp, \dot{\gamma}) \dot{\gamma}, V^\perp) + \left(\dfrac{d}{dt}g(\dot{\gamma}, \nabla\varphi)\right)|V^\perp|^2 + d\varphi(\dot{\gamma})^2|V^\perp|^2 \; dt \Bigg|_{s=0}\\
		&+ g(D_s V, \dot{\gamma}) \Bigg|_{t = a, s = 0}^{t = b, s = 0}\\
		=& \int_a^b |D_t V^\perp|^2 - g(R^\varphi(V^\perp, \dot{\gamma}) \dot{\gamma}, V^\perp) - g(\dot{\gamma}, \nabla\varphi)\dfrac{d}{dt}|V^\perp|^2 + \dfrac{d}{dt}\left(g(\dot{\gamma}, \nabla\varphi)|V^\perp|^2\right)\\
		&+ d\varphi(\dot{\gamma})^2|V^\perp|^2 \; dt \Bigg|_{s=0} + g(D_s V, \dot{\gamma}) \Bigg|_{t=a, s = 0}^{t=b, s = 0}\\
		=& \int_a^b |D_t V^\perp - d\varphi(\dot{\gamma})V^\perp|^2 - g(R^\varphi(V^\perp, \dot{\gamma}) \dot{\gamma}, V^\perp) \; dt \Bigg|_{s=0}\\
		&+ \left(d\varphi(\dot{\gamma})|V^\perp|^2 + g(D_s V, \dot{\gamma})\right) \Bigg|_{t=a, s = 0}^{t=b, s = 0}.
	\end{align*}
	This proves the first equality. For the second equality, note that the integrand can be rewritten by decomposing $V$ into components parallel and orthogonal to $\dot{\gamma}$, and then expanding and regrouping terms accordingly. This gives the desired expression.
	\begin{align*}
		\dfrac{d^2}{ds^2}L_g(\theta_s)\Bigg|_{s=0} =& \int_a^b |D_t V|^2 - g(D_tV, \dot{\gamma})^2 - g(R(V, \dot{\gamma})\dot{\gamma}, V) \; dt \Bigg|_{s = 0} + g(D_sV, \dot{\gamma}) \Bigg|_{t = a, s = 0}^{t = b, s = 0}\\
		=& \int_a^b |D_tV - d\varphi(X)V|^2 - g(R^\varphi(V, \dot{\gamma})\dot{\gamma}, V) - g(D_tV, \dot{\gamma})^2 \; dt \Bigg|_{s = 0}\\
		&+ \left(d\varphi(\dot{\gamma})|V^\perp|^2 + g(D_s V, \dot{\gamma})\right) \Bigg|_{t=a, s = 0}^{t=b, s = 0}.
	\end{align*}

<!-- defn-1: body_md -->
A set of $k$ vectors $\{X_1, ..., X_k\}$ in an inner product space $V$ is \textit{$\epsilon$-orthonormal} if
	$$|\langle X_i, X_j \rangle - \delta_{ij}| < \epsilon,$$
	for all $i, j = 1, ..., k$.\\
	A set of vector fields on a weighted Riemannian manifold is \textit{$\epsilon$-$\tilde{g}$-orthonormal} if it is $\epsilon$-orthonormal with respect to $\tilde{g}$ at each point.
<!-- end defn-1 -->
<!-- defn-2: body_md -->
Given a weighted Riemannian manifold $(M, g, e^{-\varphi})$ and an integer $k$, the class of functions $\widetilde{C}(k)$ on $M$ is defined to be the set of all continuous functions $f$ on $M$ which are Lipschitz continuous on each compact subset of $M$ with the following property:\\
	For each $x_0 \in M$, there exists a neighborhood $W$ of $x_0$ and some $\epsilon, \eta > 0$ such that if $x \in W$ and $\{X_1, ..., X_k\} \subseteq T_xM$ is an $\epsilon$-orthogonal set, then
	$$\sum_{i = 1}^k \widetilde{C} f(x; e^\varphi X_i) \ge \eta.$$

\label{Smoothing_Thm_for_Ctilde(k)}
	On a weighted Riemannian manifold $(M, g, e^{-\varphi})$, suppose an $f \in \widetilde{C}(k)$, where $1 \le k \le \dim M$, and a positive continuous function $\xi$ are given, then there exists a $C^\infty$ function $F \in \widetilde{C}(k)$ such that

<!-- proof-2: body_md -->
For simplicity, denote $N = \partial M$. Let $x \in B$, where $B \subseteq M \backslash N$ is compact. Let $\rho$ be the distance from the boundary $N$, and let $y \in N$ be such that $\rho(x) = \operatorname{dist}(x,y) = b$ for some $b > 0$. Define
	\begin{equation}
		\chi(\zeta) = \int_{-b}^\zeta \exp\left(\int_{-b}^\xi Z(\tau) \; d\tau \right) \; d\xi, \label{Chi_in_WuThm1}
	\end{equation}
	where $Z: \mathbb{R} \longrightarrow \mathbb{R}$ is a positive continuous function to be chosen later.\\
	We aim to show $\chi(-\rho)$ is convex with respect to the $\tilde{g}$ metric on some compact $B \subseteq M - N$, i.e. there exists some $\epsilon > 0$ such that
	$$\widetilde{C}(\chi \circ - \rho) \ge \epsilon$$
	on $B$.\\
	Let $\gamma : [0,b] \longrightarrow M$ be a minimizing normal $\nabla$-geodesic from $x$ to $y$ with $\gamma(0) = x$ and $\gamma(b) = y$. Let $X \in T_xM$ be a $g$-unit vector, and let $X(t)$ be its $\nabla$-parallel transport along $\gamma(t)$. Decompose
	$$X(t) = \alpha X^\perp(t) + \beta \dot{\gamma}(t),$$
	where $X^\perp(t) \perp \dot{\gamma}(t)$ is a unit vector field along $\gamma(t)$, and $\alpha^2 + \beta^2 = 1$.\\
	Define
	$$W(t) = \alpha X^\perp(t) + \left(1-\dfrac{t}{b}\right) \beta \dot{\gamma}(t).$$
	Since $\gamma$ is distance minimizing, we know $\dot{\gamma}(b) \perp N$, so $X^\perp(b) \in T_yN$.\\
	Consider the variation
	$$\theta(s,t) = \widetilde{\exp}_{\gamma(t)}(se^{\varphi} W(t)).$$
	For a fixed $s_0$, let $\theta_{s_0}(t) : [0, b] \longrightarrow M$ be the curve $\theta_{s_0}(t) = \theta(s_0,t)$. Define $f(s_0) = L_g(\theta_{s_0}|_0^b)$, the length of $\theta_{s_0}(t)$. At $s_0 = 0$, $\theta_{s_0}(t) = \theta(0,t) = \gamma(t)$, which means $f(0) = b$, so
	$$\chi'(-f(0)) = \chi'(-b) = 1 \text{ and } \chi''(-f(0)) = \chi''(-b) = Z(-b).$$
	Let $h(s) = (\chi \circ -f)(s)$. Then
	\begin{align}
		\notag h''(0) &= \chi''(-f(0))f'(0)f'(0) - \chi'(-f(0)) f''(0).\\
		&= Z(-b)(f'(s))^2 - f''(0). \label{Hess_gtilde h}
	\end{align}
	By the first variation formula,
	\begin{equation}
		f'(0) = \dfrac{d}{ds}\bigg|_{s = 0} L_g(\theta(s,t)) = g\left(e^\varphi W, \dot{\gamma}\right) |_{t=0}^{t=b} = -e^\varphi\beta.
	\end{equation}
	Also, by the second variation formula from Proposition \ref{Second_Variation_Formula},
	\begin{align}
		\notag f''(0) =& \dfrac{d^2}{ds^2}\bigg|_{s = 0} L_g(\theta_s)\\
		\notag =& \int_0^b |D_t \left(e^\varphi \alpha X^\perp\right) - d\varphi(\dot{\gamma}) e^\varphi \alpha X^\perp|^2 - g(R^\varphi(e^\varphi \alpha X^\perp, \dot{\gamma}) \dot{\gamma}, e^\varphi \alpha X^\perp) \; dt\\
		&+ \left(d\varphi(\dot{\gamma})|e^\varphi \alpha X^\perp|^2 + g\left(D_s e^\varphi \alpha X^\perp + D_s e^\varphi \beta\left(1-\dfrac{t}{b}\right) \dot{\gamma}, \dot{\gamma}\right)\right) \Bigg|_{t=0, s = 0}^{t=b, s = 0}. \label{Second_Variation_Plug_in}
	\end{align}
	Since $X^\perp(t)$ is parallel along $\gamma(t)$,
	\begin{equation}
		|D_t \left(e^\varphi \alpha X^\perp\right) - d\varphi(\dot{\gamma}) e^\varphi \alpha X^\perp|^2 = |d\varphi(\dot{\gamma}) e^\varphi \alpha X^\perp - d\varphi(\dot{\gamma}) e^\varphi \alpha X^\perp|^2 = 0. \label{|Stuff|^2=0}
	\end{equation}
	Using the definition of weighted sectional curvature,
	\begin{equation}
		g(R^\varphi(e^\varphi \alpha X^\perp, \dot{\gamma}) \dot{\gamma}, e^\varphi \alpha X^\perp) = e^{2\varphi} \alpha^2 g(R^\varphi(X^\perp, \dot{\gamma}) \dot{\gamma}, X^\perp) = e^{2\varphi} \alpha^2 \overline{\sec}(X^\perp, \dot{\gamma}). \label{Get_wsec}
	\end{equation}
	By the definition of weighted second fundamental form, evaluating the boundary terms of (\ref{Second_Variation_Plug_in}) at $t=b, s=0$ gives
	\begin{align}
		\notag &\left(d\varphi(\dot{\gamma})|e^\varphi \alpha X^\perp|^2 + g\left(D_s e^\varphi \alpha X^\perp + D_s e^\varphi \beta\left(1-\dfrac{t}{b}\right) \dot{\gamma}, \dot{\gamma}\right)\right) \Bigg|_{t=b, s = 0}\\
		\notag =& d\varphi(\dot{\gamma})|e^{\varphi(y)} \alpha X^\perp|^2 + g\left(\nabla_{\alpha e^{\varphi(y)} X^\perp} e^{\varphi(y)} \alpha X^\perp, \dot{\gamma}\right)\\
		=&  -e^{2\varphi(y)} \alpha^2 \tilde{\mathrm{I\!I}}(X^\perp, X^\perp), \label{Evaluate_at_t=b,s=0_to_Get_Weighted_Second_Fundamental_Form}
	\end{align}
	Evaluating the end of (\ref{Second_Variation_Plug_in}) at $t=0, s=0$ gives
	\begin{align}
		\notag&\left(d\varphi(\dot{\gamma})|e^\varphi \alpha X^\perp|^2 + g\left(D_s e^\varphi \alpha X^\perp + D_s e^\varphi \beta\left(1-\dfrac{t}{b}\right) \dot{\gamma}, \dot{\gamma}\right)\right) \Bigg|_{t=0, s = 0}\\
		\notag =& d\varphi(\dot{\gamma})|e^{\varphi(x)} \alpha X^\perp|^2 + g\left(\nabla_{e^{\varphi(x)} W} e^{\varphi(x)} \alpha X^\perp , \dot{\gamma}\right)+ g\left(\nabla_{e^{\varphi(x)} W} e^{\varphi(x)} \beta \dot{\gamma}, \dot{\gamma}\right)\\
		=& d\varphi(\dot{\gamma})|e^{\varphi(x)} \alpha X^\perp|^2 + g\left(\nabla_{e^{\varphi(x)} W} e^{\varphi(x)} W , \dot{\gamma}\right) \label{Evaluate_at_t=0,s=0}.
	\end{align}
	Using the relationship between $\nabla$ and $\widetilde{\nabla}$,
	\begin{align}
		\notag g\left(\nabla_{e^{\varphi(x)} W} e^{\varphi(x)} W , \dot{\gamma}\right) &= g\left(\widetilde{\nabla}_{e^{\varphi(x)} W} e^{\varphi(x)} W, \dot{\gamma}\right) + 2e^{2\varphi(x)}d\varphi(W)g\left(W, \dot{\gamma}\right) - d\varphi(\dot{\gamma})\left|e^{\varphi(x)} W\right|^2\\
		\notag &= 2e^{2\varphi(x)}d\varphi(W)g\left(W, \dot{\gamma}\right) - d\varphi(\dot{\gamma})\left|e^{\varphi(x)} W\right|^2\\
		\notag &= 2e^{2\varphi(x)}\beta d\varphi(W) - d\varphi(\dot{\gamma})e^{2\varphi(x)}\beta^2 - d\varphi(\dot{\gamma})e^{2\varphi(x)}\alpha^2\\
		&= e^{2\varphi(x)}\beta^2 d\varphi(\dot{\gamma}) + 2\alpha\beta e^{2\varphi(x)} d\varphi\left(X^\perp\right) - d\varphi(\dot{\gamma})e^{2\varphi(x)}\alpha^2. \label{Use_Nablatilde_and_Nabla_to_Simplify}
	\end{align}
	Combining results from (\ref{Second_Variation_Plug_in}), (\ref{|Stuff|^2=0}), (\ref{Get_wsec}), (\ref{Evaluate_at_t=b,s=0_to_Get_Weighted_Second_Fundamental_Form}), (\ref{Evaluate_at_t=0,s=0}), and (\ref{Use_Nablatilde_and_Nabla_to_Simplify}), we have
	\begin{align}
		\notag f''(0) =& \int_0^b - e^{2\varphi} \alpha^2 g(R^\varphi(X^\perp, \dot{\gamma}) \dot{\gamma}, X^\perp) \; dt - e^{2\varphi(y)} \alpha^2 \tilde{\mathrm{I\!I}}(X^\perp, X^\perp)\\
		&- \left(e^{2\varphi(x)}\beta^2 d\varphi(\dot{\gamma}) + 2\alpha\beta e^{2\varphi(x)} d\varphi\left(X^\perp\right)\right).
	\end{align}
	Hence, by using (\ref{Hess_gtilde h}),
	\begin{align}
		\notag h''(0) =& Z(-b)e^{2\varphi}\beta^2 + \int_0^b e^{2\varphi} \alpha^2 g(R^\varphi(X^\perp, \dot{\gamma}) \dot{\gamma}, X^\perp) \; dt + e^{2\varphi(y)} \alpha^2 \tilde{\mathrm{I\!I}}(X^\perp, X^\perp)\\
		\notag &+ e^{2\varphi(x)}\beta^2 d\varphi(\dot{\gamma}) + 2\alpha\beta e^{2\varphi(x)} d\varphi\left(X^\perp\right)\\
		\notag =& \alpha^2 \left(\int_0^b e^{2\varphi} \overline{\sec}_\varphi(X^\perp, \dot{\gamma}) \; dt + e^{2\varphi(y)} \tilde{\mathrm{I\!I}}(X^\perp, X^\perp)\right)\\
		&+ e^{2\varphi(x)}\beta^2(Z(-b) + d\varphi(\dot{\gamma})) + 2\alpha\beta e^{2\varphi(x)} d(X^\perp). \label{Compute_Hess_gtilde h}
	\end{align}
	Completing the square on the last part of (\ref{Compute_Hess_gtilde h}) yields a lower bound
	\begin{align}
		\notag &e^{2\varphi(x)}\beta^2(Z(-b) + d\varphi(\dot{\gamma})) + 2\alpha\beta e^{2\varphi(x)} d(X^\perp)\\
		\notag =& e^{2\varphi(x)}\left(\sqrt{Z(-b) + d\varphi(\dot{\gamma})}\beta + \dfrac{\alpha d\varphi(X^\perp)}{\sqrt{Z(-b) + d\varphi(\dot{\gamma})}}\right)^2 - \dfrac{e^{2\varphi(x)}\alpha^2 d\varphi(X^\perp)^2}{Z(-b) + d\varphi(\dot{\gamma})}\\
		\ge& -\dfrac{e^{2\varphi(x)} \alpha^2 d\varphi(X^\perp)^2}{Z(-b) + d\varphi(\dot{\gamma})}. \label{Complete_the_Square_and_Delete_Positive_Terms}
	\end{align}
	Combining (\ref{Compute_Hess_gtilde h}) and (\ref{Complete_the_Square_and_Delete_Positive_Terms}) gives the inequality
	\begin{equation}\label{Hess_gtildehNonNegative}
		h''(0) \ge \alpha^2 \left(\int_0^b e^{2\varphi} \overline{\sec}_\varphi(X^\perp, \dot{\gamma}) \; dt + e^{2\varphi(y)} \tilde{\mathrm{I\!I}}(X^\perp, X^\perp) -\dfrac{e^{2\varphi(x)} d\varphi(X^\perp)^2}{Z(-b) + d\varphi(\dot{\gamma})}\right).
	\end{equation}
	Recall that, for any fixed $s_0$, $f(s_0)$ measures the length of $\theta_{s_0}$. Also, $\theta_{s_0}(b) \in N$ because $X^\perp(b) \in T_yN$. Thus, $\theta_{s_0}(t)$ is a path from $\theta_{s_0}(0) \in M$ to the boundary $N$. The length of this segment must be at least $\rho(\theta_{s_0}(0))$, with $-f(0) =-\rho(\theta_0(0))$. Therefore, $-f(s)$ supports \footnote{A function $f_1$ supports another function $f_2$ at $x$ means $f_1(x) = f_2(x)$ and $f_1(y) \le f_2(y)$ for all $y$ near $x$.}\label{support} $-\rho(\theta_s(0))$ at $s=0$ (at $x \in M$), and hence $\chi(-f(s))$ supports $\chi(-\rho(\theta_s(0)))$ at $s=0$.\\
	By (\ref{Ctilde(x;e^varphiX)Definition}), we can compute
	\begin{align*}
		\widetilde{C} (\chi \circ -f)(x;e^\varphi X) &= \liminf_{s \to 0} \dfrac{\chi(-f(s)) + \chi(-f(-s)) - 2\chi(-f(0))}{x^2}\\
		&\le \liminf_{s \to 0} \dfrac{\chi(-\rho(\theta_s(0))) + \chi(-\rho(\theta_{-s}(0))) - 2\chi(-\rho(\theta_0(0)))}{x^2}\\
		&= \widetilde{C} (\chi \circ -\rho)(x;e^\varphi X)
	\end{align*}
	By (\ref{Ctilde(x)Definition}), taking infimum over all $X \in T_xM$ gives us
	$$\widetilde{C} h(x) = \widetilde{C} (\chi \circ -f)(x) \le \widetilde{C} (\chi \circ -\rho)(x).$$
	Since $h$ is $C^2$, we have $\widetilde{C} h(x) = h''(0)$, which is positive by (\ref{Hess_gtildehNonNegative}). Therefore, $\widetilde{C}(\chi \circ -\rho)$ is also positive. Since $x$ was chosen arbitrarily, it follows from definition that $\chi(-\rho)$ is strictly convex on $M-N$.\\
	The rest of the proof follows from standard Morse theory. Let
	$$\Theta = \int_0^b e^{2\varphi} \overline{\sec}_\varphi(X^\perp, \dot{\gamma}) \; dt + e^{2\varphi(y)} \tilde{\mathrm{I\!I}}(X^\perp, X^\perp).$$
	We now suppose conditions 1 or 2 holds respectively, and we will show there exists some constant $\psi > 0$ that only depends on $B$ such that $\Theta \ge \psi$.\\
	Suppose condition 1 holds. If $\min_M\overline{\sec} < 0$, let
	$$\psi = \begin{cases}
		u_{\min}^2\left(\min_N\widetilde{\Lambda} + \rho_0\min_M\overline{\sec}_\varphi(M)\right) &\text{ if } \min_M\overline{\sec} < 0\\
		u_{\min}^2\min_N \widetilde{\Lambda} &\text{ if } \overline{\sec} \ge 0,
	\end{cases}$$
	where $u_{\min}^2$ is the minimum of $e^{2\varphi}$ on $B$.\\
	Suppose 2 holds, then for $c > 0$, let $T(c) = \{z \in M \; | \; \rho(z) \le c\}$ be a tubular neighborhood. By our assumption, choose $c < \min_B\rho$ small enough so that $\overline{\sec} > \sigma$ for some $\sigma > 0$. Define
	$$\psi = c\sigma.$$
	Under either assumption, it is clear that
	$$\Theta \ge \psi > 0.$$
	Since $B$ is compact, $d\varphi$ is bounded on $B$, so we can choose $Z$ big enough so that
	$$\psi - \dfrac{u_{\max}^2 d\varphi(X^\perp)^2}{Z(-b) + d\varphi(\gamma)} > 0,$$
	where $u_{\max}^2$ is the maximum of $e^{2\varphi}$ on $B$. 
	Therefore $\chi \circ -\rho$ is convex in the metric $\tilde{g}$.\\
	The rest of the proof follows from Wu's proof in \cite{Wu} as the arguments do not depend on the curvature of the manifold. To summarize, first we apply smoothing theorem from \cite{GreeneWu} to obtain a strictly convex function $\overline{h}: M-N \longrightarrow \mathbb{R}$ such that $\overline{h}|_{T(c)-N} = h = \chi \circ -\rho$ for some small enough $c > 0$. Choose $0 < \eta < c$. Define $M' = \{z \in M \; | \; \rho(z) \ge \eta\}$. By \cite[Theorem 3.1.]{Milnor2}, we can conclude $(M, \partial M)$ is diffeomorphic to $(M', \partial M')$. Then we aim to show $(M', \partial M')$ is diffeomorphic to $(B^n, S^{n-1})$.\\
	Note that $h$ is strictly convex with one minimum point, and let it be $x_0$. Rescale $h$ so that $h^{-1}(0) = x_0$ and $h^{-1}(1) = \partial M'$. By Morse Lemma, there exists a coordinate neighborhood $U$ of $x_0$ and a coordinate map $y = (y_1, ..., y_n): U \longrightarrow \mathbb{R}^n$ with $y(x_0) = 0$, and $y$ is a diffeomorphism from $U$ to an open ball $B(2\epsilon) \subseteq \mathbb{R}^n$. Also, $h = \sum_{i = 1}^n y_i^2$ on $U$. By partition of unity, we can change the metric on $M$ to $\bar{g}$ so that $\bar{g}\left(\dfrac{\partial}{\partial y_i}, \dfrac{\partial}{\partial y_j}\right) = \delta_{ij}$ on $U$.\\
	In $U - \{x_0\}$, define $Y = \dfrac{\nabla h}{|\nabla h|^2}$, let $\{\varphi_t\}$ be the flow of $Y$, and let $S = \left\{p \in U \; \Bigg| \; \sum_{i = 1}^n y_i(p)^2 = \epsilon^2 \right\}$. This means $h^{-1}(\epsilon^2) = S$. Hence we can define a diffeomorphism
	$F: S \times (-\epsilon^2,3\epsilon^2) \longrightarrow U - \{x_0\}$ by mapping
	$$(p,t) \mapsto \varphi_t(p),$$
	and the diffeomorphism $y: U - \{x_0\} \longrightarrow B(2\epsilon) - \{0\}$ can be written as
	$$\varphi_t(p) \mapsto \dfrac{\sqrt{\epsilon^2 + t}}{\epsilon}(y_1(p),...,y_n(p)).$$
	$y$ can be extended naturally to a map $M' - \{x_0\} \longrightarrow B^n - \{0\}$ by the uniqueness of integral curves, and allowing $t$ in $F$ to take values in $(-\epsilon^2, 1-\epsilon^2]$. Composing these two functions gives us the desired diffeomorphism $\bar{F}: M' \longrightarrow B^n$.

Let $X = \dfrac{U+V}{\sqrt{2}}$ and $Y = \dfrac{U-V}{\sqrt{2}}$. It follows that
	\begin{align*}
		&\overline{\sec}_\varphi(Y,X) - \overline{\sec}_\varphi(X,Y)\\
		=& \sec(Y,X) + \operatorname{Hess}\,\varphi(X,X) + d\varphi(X)^2 - \sec(X,Y) - \operatorname{Hess}\,\varphi(Y,Y) - d\varphi(Y)^2\\
		=& \operatorname{Hess}\,\varphi(X,X) + d\varphi(X)^2 - \operatorname{Hess}\,\varphi(Y,Y) - d\varphi(Y)^2\\
		=& \operatorname{Hess}\,\varphi\left(\dfrac{U+V}{\sqrt{2}},\dfrac{U+V}{\sqrt{2}}\right) + d\varphi\left(\dfrac{U+V}{\sqrt{2}}\right)^2 - \operatorname{Hess}\,\varphi\left(\dfrac{U-V}{\sqrt{2}},\dfrac{U-V}{\sqrt{2}}\right) - d\varphi\left(\dfrac{U-V}{\sqrt{2}}\right)^2\\
		=& \dfrac{1}{2}\left(\operatorname{Hess}\,\varphi(U,U) + \operatorname{Hess}\,\varphi(V,V) + 2\operatorname{Hess}\,\varphi(U,V)\right) + \dfrac{1}{2}(d\varphi(U)^2 + d\varphi(V)^2 + 2d\varphi(U)d\varphi(V))\\
		&- \dfrac{1}{2}\left(\operatorname{Hess}\,\varphi(U,U) + \operatorname{Hess}\,\varphi(V,V) - 2\operatorname{Hess}\,\varphi(U,V)\right) - \dfrac{1}{2}(d\varphi(U)^2 + d\varphi(V)^2 - 2d\varphi(U)d\varphi(V))\\
		=& 2\operatorname{Hess}\,\varphi(U,V) + 2d\varphi(U)d\varphi(V)
	\end{align*}
	Therefore
	\begin{equation}\label{Hess+dvarphi^2>-K}
		|\operatorname{Hess}\,\varphi(U,V) + d\varphi(U)d\varphi(V)| \le K.
	\end{equation}
	By (\ref{R^varphiEquationInTermsOfHessAnddvarphi}), it follows that
	\begin{align}
		\notag g(R^\varphi(U,V)U,U) =& g(R(U,V,U),U) + \operatorname{Hess}\,\varphi(V,U)g(U,U) - \operatorname{Hess}\,\varphi(U,U)g(U,V) \\
		\notag &+ d\varphi(V)d\varphi(U)g(U,U) - d\varphi(U)d\varphi(U)g(V,U)\\
		\notag =& g(R(U,V,U),U) + \operatorname{Hess}\,\varphi(V,U)g(U,U) + d\varphi(V)d\varphi(U)g(U,U)\\
		=& \operatorname{Hess}\,\varphi(V,U) + d\varphi(V)d\varphi(U) \label{g(R^varphi(U,V)U,U)=Hessvarphi(V,U)+dvarphi(V)dvarphi(U)}
	\end{align}
	Combining (\ref{Hess+dvarphi^2>-K}) and (\ref{g(R^varphi(U,V)U,U)=Hessvarphi(V,U)+dvarphi(V)dvarphi(U)}), we get
	$$|g(R^\varphi(U,V)U,U)| \le K,$$
	as desired.

Since $T_{p_1}N_1 \perp \dot{\gamma}$ at $p_1$, it follows that $T'_{p_1}N_1 \perp \dot{\gamma}$ at $p_2$. Hence,
	$$\dim(T'_{p_1}N_1 \cap T_{p_2}N_2) \ge r + s - (n-1) \ge k.$$
	Choose $k$ orthonormal vectors $E_1',...,E_k' \in T'_{p_1}N_1 \cap T_{p_2}N_2$, and parallel transport them back to $p_1$ via $\gamma$, denoting the resulting vectors by $E_1, ..., E_k$. Now, consider variations $$\theta_j(t,s) = \widetilde{\exp}_{\gamma(t)}(se^\varphi E_j(t)),$$
	for $1 \le j \le k$.\\
	At $p_1$, i.e. $t = 0$, we have
	$$d\varphi(\dot{\gamma}(0))|e^{\varphi(p_1)}E_j(0)|^2 = e^{2\varphi(p_1)}d\varphi(\dot{\gamma}(0)),$$
	and
	\begin{align*}
		g(D_s(e^{\varphi} E_j(0)), \dot{\gamma}(0)) &= g((D_se^\varphi)E_j(0), \dot{\gamma}(0)) + g(e^\varphi \nabla_{e^\varphi E_j(0)}E_k(0), \dot{\gamma}(0))\\
		&= e^{2\varphi(p_1)} g(\nabla_{E_j(0)}E_j(0), \dot{\gamma}(0)).
	\end{align*}
	At $p_2$, i.e. $t = 1$, similarly,
	$$d\varphi(\dot{\gamma}(1))|e^{\varphi(p_2)}E_j(1)|^2 = e^{2\varphi(p_2)}d\varphi(\dot{\gamma}(1)),$$
	and
	$$g(D_s(e^{\varphi} E_j(1)), \dot{\gamma}(1)) = g((D_se^\varphi)E_j(1), \dot{\gamma}(1)) + g(e^\varphi \nabla_{e^\varphi E_j(1)}E_k(1), \dot{\gamma}(1)) = e^{2\varphi(p_2)} g(\nabla_{E_j(1)}E_j(1), \dot{\gamma}(1)).$$
	Applying the second variation formula for arc length from Proposition \ref{Second_Variation_Formula}, we obtain
	\begin{align*}
		\sum_{j = 1}^{k} \dfrac{d^2}{ds^2}\Bigg|_{s = 0} L_g(\theta_j(t,s)) =& \sum_{j = 1}^k \int_0^1 \big|D_t (e^\varphi E_j(t)) - d\varphi(\dot\gamma)(e^\varphi E_j(t))\big|^2 \, dt \Big|_{s=0}\\
		&- \sum_{j = 1}^k \int_0^1 g\big(R^\varphi(e^\varphi E_j(t), \dot\gamma)\dot\gamma, e^\varphi E_j(t)\big) \, dt \Big|_{s=0}\\
		&+ \sum_{j = 1}^{k} \Big(d\varphi(\dot\gamma)|e^\varphi E_j(t)|^2 + g(D_s (e^\varphi E_j(t)), \dot\gamma)\Big) \Big|_{t=0,\, s=0}^{t=1,\, s=0}\\
		=& -\int_0^1 e^{2\varphi} \sum_{j = 1}^{k} \overline{\sec}(E_j(t),\dot{\gamma}) \, dt\\
		&+ \sum_{j=1}^{k} e^{2\varphi(p_2)}\tilde{\mathrm{I\!I}}(E_j(1), E_j(1)) - \sum_{j=1}^{k} e^{2\varphi(p_1)}\tilde{\mathrm{I\!I}}(E_j(0), E_j(0)).\\
		=& -\int_0^1 e^{2\varphi} \sum_{j = 1}^{k} \overline{\sec}(E_j(t),\dot{\gamma}) \, dt\\
		< & 0.
	\end{align*}
	This contradicts the fact that $\gamma$ is length minimizing. Therefore
	$$N_1 \cap N_2 \not= \varnothing.$$
