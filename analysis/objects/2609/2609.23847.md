\label{thm:quadratic-version}
		Let $F/\mathbb{Q}$ be a real quadratic field, and let
		$\ell\in\{5,7,11,13\}$ be such that $\ell\nmid h(F)$, with $F\neq\mathbb{Q}(\sqrt5)$ if $\ell=5$. Let $E/F$ be an
		elliptic curve having an $F$-rational point $P$ of order $\ell$, and suppose that $P$ is not contained in
		the kernel of reduction modulo any prime $\mathfrak{l}$ of $F$ above $\ell$ in the sense of \cite{Morrow}; see
		\S\ref{ss:ec-background}.
		
		Suppose that every prime $\mathfrak{p}\mid N(E)$ with $\mathfrak{p}\nmid\ell$ and
			$\ell\nmid\operatorname{ord}_{\mathfrak{p}}(\Delta_{E})$ satisfies $\ell\mid\mathrm{N}\mathfrak{p}-1$.
	
		
		Suppose moreover that there exists a rational prime $q>C_{F,E,\ell}$, inert in $F$,
		where $C_{F,E,\ell}=(A/2)^{2}$ is the constant of Theorem~\ref{thm:lifting},
		and put
		\begin{equation*}
			K=F(\sqrt{-q}).
		\end{equation*}
		Assume that $K/F$ satisfies the following local conditions:
		\begin{enumerate}[label=\textup{(\arabic*)},leftmargin=*]
			\item 
			Every prime $\mathfrak{q}\mid2$ dividing $N(E)$ ramifies in $K/F$.
			
			\item  If $\mathfrak{l}\mid\ell$ and $\operatorname{ord}_{\mathfrak{l}}(j_E)<0$, then $\mathfrak{l}$ is inert in $K/F$.
			
			\item For every prime $\mathfrak{p}\mid N(E)$ with $\mathfrak{p}\nmid2\ell$,
			\begin{equation*}
				\begin{cases}
					\mathfrak{p} \text{ is inert in } K/F,
					& \text{if } E/F_{\mathfrak{p}}\text{ is a Tate curve},\\[4pt]
					\mathfrak{p} \text{ splits in } K/F,
					& \text{otherwise}.
				\end{cases}
			\end{equation*}
		\end{enumerate}
		
		Assume in addition that
		\begin{equation*}
			\ell\nmid h^{-}(K/F).
		\end{equation*}
		Then
		\begin{equation*}
			\#\left\{
			d\in F^\times/(F^\times)^2:
			\begin{array}{l}
				d\text{ is totally negative},\\[2pt]
				\left|
				\mathrm{N}_{F/\mathbb{Q}}\!\bigl(D(F(\sqrt d)/F)\bigr)
				\right|<X,\\[2pt]
				\operatorname{Sel}_\ell(E^d,F)=1
			\end{array}
			\right\}
			\gg_{F,E,\ell}
			\frac{X^{1/4}}{\log X}.
		\end{equation*}

\label{def:hmf}
		A \emph{Hilbert modular form of parallel half-integral weight $k/2$,
			level $\mathfrak{m}$, and nebentypus $\psi$} is a holomorphic function
		$f:{\mathcal{H}}^{{\mathbf a}}\to\mathbb C$, holomorphic at the cusps when
		$F=\mathbb Q$, such that
		\begin{equation*}
			f|_k[\gamma]=\psi_{\mathfrak{m}}(d_\gamma)\,f
			\qquad(\gamma\in\Gamma_0(\mathfrak{m})).
		\end{equation*}
		We write $M_k\!\left(\Gamma_0(\mathfrak{m}),\,\psi\right)$ for the space of such forms.

			&=\psi(d)\,h(\gamma,z)^{k}f(z+\ifmmode\beta\else\expandafter\ifmmode\beta\else\expandafter\ifmmode\beta\else\expandafter\ifmmode\beta\else\expandafter\b\fi\fi\fi\fi_{t_r}).
		\end{aligned}
	\end{equation}

\begin{equation*}
	\begin{aligned}
		(f\otimes\chi)|_k[\gamma](z)
		&=
		\frac{h(\gamma,z)^{-k}}{\tau(\bar\chi,w)}
		\sum_{r\in\mathcal{O}/\mathfrak{n}}

	\end{aligned}
\end{equation*}

	\begin{equation*}
		r\equiv d^{-2}s\pmod{\mathfrak{n}},
	\end{equation*}

		\bar\chi(r)

Put
	\begin{equation*}
		U=\mathcal{O}_{\mathfrak{q}}^\times/\mathcal{O}_{\mathfrak{q}}^{\times2}.
	\end{equation*}
	By \cite[Chapter~II, \S5, Corollary~5.8]{Neukirch},
	\begin{equation*}
		\dim_{\mathbb F_2}K^\times/K^{\times2}
		=[K:\mathbb Q_2]+2.
	\end{equation*}
	Since
	\begin{equation*}
		K^\times/K^{\times2}
		\cong(\mathbb Z/2\mathbb Z)\oplus U,
	\end{equation*}
	it follows that
	\begin{equation*}
		\dim_{\mathbb F_2}U=[K:\mathbb Q_2]+1\geq2.
	\end{equation*}
	
	By \cite[Chapter~III, \S5, Theorem~2]{SerreLocalFields},
	finite unramified extensions of $K$ correspond to finite separable
	extensions of its residue field. Since the residue field is finite,
	it has a unique quadratic extension, and hence $K$ has a unique
	unramified quadratic extension. The corresponding square class has
	even valuation, so we may choose a representative
	$u_0\in\mathcal{O}_{\mathfrak{q}}^\times$. Thus, for $a\in K^\times$, the extension
	$K(\sqrt a)/K$ is trivial or unramified if and only if
	\begin{equation*}
		[a]\in\{[1],[u_0]\}.
	\end{equation*}
	
	Since $[u_0]\neq[1]$ and $\dim_{\mathbb F_2}U\geq2$, there exists a
	nonzero $\mathbb F_2$-linear functional
	\begin{equation*}
		\lambda:U\longrightarrow\mathbb F_2
	\end{equation*}
	such that $\lambda([u_0])=0$. If $a_0$ is as in the final assertion
	of the lemma, then
	\begin{equation*}
		[-a_0]\notin\{[1],[u_0]\},
	\end{equation*}
	since $\mathfrak{q}$ ramifies in $F(\sqrt{-a_0})/F$. Hence
	$[-a_0]\notin\langle[u_0]\rangle$, and we may choose $\lambda$ so
	that, in addition,
	\begin{equation*}
		\lambda([-a_0])=1.
	\end{equation*}
	Define
	\begin{equation*}
		\widetilde\chi(a)=(-1)^{\lambda([a])},
		\qquad a\in\mathcal{O}_{\mathfrak{q}}^\times.
	\end{equation*}
	Then $\widetilde\chi$ is a nontrivial quadratic character and
	$\widetilde\chi(u_0)=1$.
	
	We next show that $\widetilde\chi$ has finite conductor. If
	$u\in1+\mathfrak{q}^{\,2e+1}\mathcal{O}_{\mathfrak{q}}$, then Lemma~\ref{lem:localsquare}
	gives $u\in\mathcal{O}_{\mathfrak{q}}^{\times2}$. Therefore
	\begin{equation*}
		1+\mathfrak{q}^{\,2e+1}\mathcal{O}_{\mathfrak{q}}
		\subseteq
		\mathcal{O}_{\mathfrak{q}}^{\times2}
		\subseteq
		\ker\widetilde\chi.
	\end{equation*}
	In fact $\widetilde\chi$ is trivial on
		$1+\mathfrak{q}^{\,2e}\mathcal{O}_{\mathfrak{q}}$. Indeed, every element of this group has the
		form $1+4y$ with $y\in\mathcal{O}_{\mathfrak{q}}$, and $(1+2x)^2=1+4(x+x^2)$ for
		$x\in\mathcal{O}_{\mathfrak{q}}$. If the residue class of $y$ has the form
		$\bar x^2+\bar x$, then
		$(1+4y)(1+2x)^{-2}\in1+\mathfrak{q}^{\,2e+1}\mathcal{O}_{\mathfrak{q}}$, so that $1+4y$ is a
		square. Otherwise the polynomial $T^2-T-y$ is irreducible modulo
		$\mathfrak{q}$, so that its root $(1+\sqrt{1+4y})/2$ generates the unramified
		quadratic extension of $K$, and $[1+4y]=[u_0]$. As
		$\widetilde\chi(u_0)=1$, we obtain
		\begin{equation*}
			1+\mathfrak{q}^{\,2e}\mathcal{O}_{\mathfrak{q}}
			\subseteq
			\ker\widetilde\chi.
	\end{equation*}
	
	Let $\mathfrak{q}^{\,n}$ be the conductor of $\widetilde\chi$, so that $n$ is
	the smallest positive integer for which
	\begin{equation*}
		1+\mathfrak{q}^{\,n}\mathcal{O}_{\mathfrak{q}}
		\subseteq\ker\widetilde\chi.
	\end{equation*}
	The preceding inclusion gives $n\leq2e$. Moreover,
	$(\mathcal{O}/\mathfrak{q})^\times$ has odd order, since the residue field has
	cardinality $2^f$ for some $f\geq1$. Hence it admits no nontrivial
	quadratic character. Since $\widetilde\chi$ is nontrivial, it cannot
	factor modulo $\mathfrak{q}$, and therefore $n\geq2$.
	          
	Since
	$1+\mathfrak{q}^{\,n}\mathcal{O}_{\mathfrak{q}}\subseteq\ker\widetilde\chi$,
	$\widetilde\chi$ is constant on residue classes modulo $\mathfrak{q}^{\,n}$
	and therefore descends to a character on
	\begin{equation*}
		\mathcal{O}_{\mathfrak{q}}^\times/(1+\mathfrak{q}^{\,n}\mathcal{O}_{\mathfrak{q}})
		\cong
		(\mathcal{O}_{\mathfrak{q}}/\mathfrak{q}^{\,n}\mathcal{O}_{\mathfrak{q}})^\times
		\cong
		(\mathcal{O}/\mathfrak{q}^{\,n})^\times.
	\end{equation*}
	Denote the resulting character by
	\begin{equation*}
		\chi_\mathfrak{q}:(\mathcal{O}/\mathfrak{q}^{\,n})^\times
		\longrightarrow\{\pm1\}.
	\end{equation*}
	By the minimality of $n$, $\chi_\mathfrak{q}$ is primitive. Extend it by $0$
	to all of $\mathcal{O}/\mathfrak{q}^{\,n}$.
	
	Finally, put
	\begin{equation*}
		\varepsilon=-\widetilde\chi(-1).
	\end{equation*}
	Let $\xi\in\tfrac12\mathcal{O}$, put $a=2\xi$, and suppose
	$\chi_\mathfrak{q}(a)=\varepsilon$. Since $\varepsilon\neq0$, the element $a$
	is a unit at $\mathfrak{q}$, and hence
	\begin{equation*}
		\widetilde\chi(a)
		=
		\chi_\mathfrak{q}(a)
		=
		-\widetilde\chi(-1).
	\end{equation*}
	Since $\widetilde\chi(u_0)=1$, we also have
	\begin{equation*}
		\widetilde\chi(-u_0)=\widetilde\chi(-1).
	\end{equation*}
	Thus
	\begin{equation*}
		[a]\notin\{[-1],[-u_0]\},
	\end{equation*}
	and therefore
	\begin{equation*}
		[-a]\notin\{[1],[u_0]\}.
	\end{equation*}
	Hence $K(\sqrt{-a})/K$ is neither trivial nor unramified, and is
	therefore ramified. Equivalently, $\mathfrak{q}$ ramifies in
	$F(\sqrt{-a})/F$. Since $a=2\xi$, the first assertion follows.
	
	If $a_0$ is as in the final assertion, then our choice of $\lambda$
	gives
	\begin{equation*}
		-1
		=
		\widetilde\chi(-a_0)
		=
		\widetilde\chi(-1)\widetilde\chi(a_0).
	\end{equation*}
	Therefore
	\begin{equation*}
		\chi_\mathfrak{q}(a_0)
		=
		\widetilde\chi(a_0)
		=
		-\widetilde\chi(-1)
		=
		\varepsilon,
	\end{equation*}
	which proves the final assertion.

<!-- proof-8: body_md -->
In the proof of \cite[Theorem~1]{Takai}, the two original numerical

			[L(\zeta_p):L]\geq g
		\end{equation*}

		\end{equation*}

		\end{equation*}

		$g-1$ roots modulo $p$. Hence Takai's argument applies unchanged.

		$E_{1}(F_{\mathfrak{p}})$, a contradiction. 
		
		Assume $\ell\nmid\operatorname{ord}_{\mathfrak{p}}(\Delta_{E})$. In the split case
		$\#\Phi_{\mathfrak{p}}=\operatorname{ord}_{\mathfrak{p}}(\Delta_{E})$ \cite[VII.6.1]{SilvermanAEC},
		while in the nonsplit case $\#\Phi_{\mathfrak{p}}\mid2$
		\cite[Appendix~C, Corollary~15.2.1]{SilvermanAEC}. Hence
		$\ell\nmid\#\Phi_{\mathfrak{p}}$, so $P\in E_{0}(F_{\mathfrak{p}})$. Since
		$E_{1}(F_{\mathfrak{p}})$ has no $\ell$-torsion, the image of $P$ in
		$\widetilde E_{\mathrm{ns}}(k(\mathfrak{p}))$ has order $\ell$. Therefore
		$\ell\mid\mathrm{N}\mathfrak{p}-1$ in the split case and
		$\ell\mid\mathrm{N}\mathfrak{p}+1$ in the nonsplit case. Since $\ell$ is odd, these
		two divisibilities are mutually exclusive, giving the stated
		equivalences.

In either case, let $I$ be the inertia group of a prime of $M$ above $\mathfrak{q}$, and let
		$I_{1}$ be its wild inertia subgroup. Since the residue characteristic is $2$,
		$I_{1}$ is a normal Sylow $2$-subgroup of $I$ and $I/I_{1}$ is cyclic
		\cite[IV, \S2]{SerreLocalFields}.

		subgroup of order $2$ of $\operatorname{Gal}(M/F)$, a contradiction.

		$\mathfrak{q}$, whereas $F(\sqrt d)/F$ is ramified. Hence $F(\sqrt d)\cap F(\zeta_{\ell})=F$,

		$\operatorname{Gal}(F(\zeta_{\ell})/F)$ and nontrivial image in
		$\operatorname{Gal}(F(\sqrt d)/F)$, hence is $\langle\ifmmode\delta\else\expandafter\ifmmode\delta\else\expandafter\ifmmode\delta\else\expandafter\ifmmode\delta\else\expandafter\d\fi\fi\fi\fi\rangle$. As ${\sigma}^{m/2}\neq1$, we have

		$H$, while $t\notin H$, so the image of $t$ in $\operatorname{Gal}(K'/F)$ is a nontrivial
		element acting trivially on $H$. This contradicts faithfulness.

		$\ell\nmid h(F)$, $\zeta_{\ell}\notin F$, and $m=[F(\zeta_{\ell}):F]$ is even.

			\item every prime $\mathfrak{p}\mid N(E)$ with $\mathfrak{p}\nmid\ell$ and
			$\ell\nmid\operatorname{ord}_{\mathfrak{p}}(\Delta_{E})$ satisfies $\ell\mid\mathrm{N}\mathfrak{p}-1$.
		\end{enumerate}

		\begin{enumerate}[label=\textup{(\arabic*)},leftmargin=*]

			\item if $\mathfrak{l}\mid\ell$ and $\operatorname{ord}_{\mathfrak{l}}(j_{E})<0$, then
			$\left(\dfrac{K/F}{\mathfrak{l}}\right)=\ifmmode\delta\else\expandafter\ifmmode\delta\else\expandafter\ifmmode\delta\else\expandafter\ifmmode\delta\else\expandafter\d\fi\fi\fi\fi$;

			\begin{equation*}
				\left(\dfrac{K/F}{\mathfrak{p}}\right)=
				\begin{cases}
					\delta, & \text{if } E/F_{\mathfrak{p}}\text{ is a Tate curve},\\[4pt]
					\operatorname{id}, & \text{otherwise}.

\label{thm:lifting}
		Let $F$ be a totally real Galois number field of degree $g$, and let
		$\ell\geq5$ be a prime such that

			\ell\nmid h(F)

			\item $P$ is not contained in the kernel of reduction modulo any prime
			$\mathfrak{l}\mid\ell$, in the sense of \cite{Morrow};
			\item every prime $\mathfrak{p}\mid N(E)$ with $\mathfrak{p}\nmid\ell$ and
			$\ell\nmid\operatorname{ord}_{\mathfrak{p}}(\Delta_{E})$ satisfies $\ell\mid\mathrm{N}\mathfrak{p}-1$.

		\begin{equation*}
			\mathcal T=\{\mathfrak{p}:\ \mathfrak{p}\mid\ell N(E)\}=\{\mathfrak{p}_1,\dots,\mathfrak{p}_r\},

		sign $\varepsilon_i\in\{\pm1\}$ as follows.

			\item If $\mathfrak{p}_i\mid2$, let $\chi_i$ and $\varepsilon_i$ be the character
			and the sign supplied by Lemma~\ref{lem:dyadic} for $\mathfrak{q}=\mathfrak{p}_i$.
			\item If $\mathfrak{p}_i\nmid2$, let $\chi_i=\left(\dfrac{\cdot}{\mathfrak{p}_i}\right)$
			and put

				\varepsilon_i=
				\begin{cases}
					-\left(\dfrac{-1}{\mathfrak{p}_i}\right),
					& \text{if } \mathfrak{p}_i\nmid\ell \text{ and } E/F_{\mathfrak{p}_i}
					\text{ is a Tate curve},\\[6pt]

					& \text{if } \mathfrak{p}_i\nmid\ell \text{ and } E/F_{\mathfrak{p}_i}
					\text{ is not a Tate curve},\\[6pt]

					& \text{if } \mathfrak{p}_i\mid\ell \text{ and } \operatorname{ord}_{\mathfrak{p}_i}(j_E)<0,\\[6pt]
					\text{either sign},
					& \text{if } \mathfrak{p}_i\mid\ell \text{ and } \operatorname{ord}_{\mathfrak{p}_i}(j_E)\geq0.

\label{rem:collapse}
		If $q$ is inert in $\mathcal{O}$ and $\chi_i(q)=\varepsilon_i$ for every $i$, then
		the sum in \eqref{eq:takai-congruence} consists of the single term
		$\xi=q/2$. Hence \eqref{eq:takai-congruence} is equivalent to
		\begin{equation}
			\label{eq:takai-congruence-2}
			\frac{h^{-}(F(\sqrt{-q})/F)}
			{Q_{F(\sqrt{-q})}\,w_{F(\sqrt{-q})}}
			\not\equiv 0 \pmod{\ell}.
		\end{equation}
		Indeed, put $\nu=2\xi\in\mathcal{O}_{+}$. Then $\operatorname{Tr}(\nu)=qg$, and since
		$q\mathcal{O}$ is prime, the condition $(q\mathcal{O},\nu\mathcal{O})\neq1$ gives
		$\nu=q\mu$ for some $\mu\in\mathcal{O}_{+}$. Thus $\operatorname{Tr}(\mu)=g$, and
		\begin{equation*}
			\mathrm{N}_{F/\mathbb{Q}}(\mu)^{1/g}
			\leq
			\frac{\operatorname{Tr}(\mu)}{g}
			=
			1.
		\end{equation*}
		Since $\mathrm{N}_{F/\mathbb{Q}}(\mu)$ is a positive integer, equality holds and
		$\mu=1$. Hence $\xi=q/2$. Moreover, $q$ is unramified in $F$, so
		$\beta(q/2)=1$.
