<!-- defn-1: body_md -->
Let $G$ be a group acting on a set $X$. 
We say that a function $f:X\to Y$ is
\begin{itemize}
    \item[(a)] \emph{invariant (to $G$)} if $f(g \cdot x) = f(x)$ for all $x \in X$, $g \in G$,
    \item[(b)] \emph{equivariant\footnote{
    Some authors use ``covariance'' for what we call ``equivariance''; see, for example,~\cite{sifre2013rotation}.} (with respect to $G$)} if $G$ also acts on $Y$ and $f(g \cdot x) = g \cdot f(x)$ for all $x \in X$, $g \in G$, and
    \item[(c)] \emph{group-Lipschitz (with respect to $G$)} if $G$ is a metric group with metric $d$ (or more generally a semimetric), $Y$ is a normed vector space, and for all $x \in X$ there exists $C_x > 0$ such that  $\left\Vertf(g \cdot x)-f(h\cdot x)\right\Vert \leq C_x d(g,h)$ for all $g, h \in G$.
\end{itemize}

<!-- prop-1: body_md -->
\label{prop: Psi Equivariant}
$\Psi$ is $G$-equivariant from $\mathbf{L}^2(\mathbb{R}^d)$ to $\mathbf{C}_b(G)$,
i.e., for  all $f\in\mathbf{L}^2(\mathbb{R}^d)$  and $g\in G$, we have 
\begin{equation*}
g\cdot\Psi(f)=\Psi(g\cdot f),
\end{equation*}
where the actions of $G$ on $\mathbf{L}^2(\mathbb{R}^d)$ and $\mathbf{C}_b(G)$  are defined as in~\eqref{eqn:quasireg} and~\eqref{eqn: outer action}.

 where the actions of $\Lambda$ on $\mathbf{C}_b(G)$ and $\mathbf{C}_b(\Lambda)$  are defined as in~\eqref{eqn: outer action} and~\eqref{eqn: outer action Lambda}.

<!-- remark-1: body_md -->
\label{rem: extend to complex}

Throughout, we have assumed that $f$ is real-valued, both for convenience and because this is typically the case in the setting of neural networks. 
However, our definitions can be adapted for complex-valued functions. 
Extending the definition of $\Psi$ is straightforward: we simply use the complex inner product $\langle f,g\rangle_{\mathbf{L}^2(\mathbb{R}^d)}=\int_{\mathbb{R}^d}f(x)\overline{g(x)}dx$. 
For the activation functions $\sigma$, we may apply $\sigma$ to the real and imaginary parts separately or choose $\sigma$ to be the complex modulus as done in \cite{mallat:scattering2012}. Alternatively, as in \cite{Zhang2021MagNet}, we could use the complex version of  ReLU defined by $\sigma(z)=z$, if $-\pi/2\leq \arg(z)<\pi/2$, and $\sigma(z)=0$, although this would require  some modification of our continuity theory introduced in Section \ref{sec:transtheory} since the complex ReLU is not continuous. It is straightforward to check that each of these choices are positively homogeneous. 
For the transversal pooling operator $\Xi$, there are multiple options.
For example, we could take the supremum of the real and imaginary parts separately and have the output of $\Xi$ be complex-valued, or we could take the supremum of $|F(\lambda h)|$ and have the output of $\Xi$ be a real-valued function.

<!-- thm-2: body_md -->
\label{prop:contest}
Let $g=(A,b)\in \operatorname{AGL}(\mathbb{R}^d)$, and let $\psi \in \mathbf{W}^{1,2}(\mathbb{R}^d)$. Assume $\det(A) >0$ 
and that
\[
\big{\|}\left\Vert\cdot\right\Vert_2 \nabla \psi\big{\|}_{\mathbf{L}^2(\mathbb{R}^d)}
:=\left(\int_{\mathbb{R}^d}\|x\|^2_2\|\nabla \psi(x)\|_2^2dx\right)^{1/2}< \infty.
\]
It holds that
\begin{equation}
\label{eq: affine bound}
\left\Vertg\cdot\psi - \psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)} \leq C_A \big{\|}\left\Vert\cdot\right\Vert_2 \nabla \psi\big{\|}_{\mathbf{L}^2(\mathbb{R}^d)}+\frac{1}{2}\left\vert\ln \det A\right\vert\left\Vert\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}  +\left\Vertb\right\Vert_2 \left\Vert\nabla \psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)},
\end{equation}
where
\begin{equation}\label{eqn: CA}
C_A = \max\left\{\ln\left\VertA\right\Vert_{2 \rightarrow 2}, \ln\left\VertA^{-1}\right\Vert_{2 \rightarrow 2}\right\}+ 2\arcsin\left( \frac{1}{2}\left\VertA (A^\top A)^{-1/2}-I\right\Vert_{2 \rightarrow 2}\right).
\end{equation}
If we further assume that $\psi$ is Lipschitz and compactly supported with Lipschitz constant $\left\Vert\psi\right\Vert_{L}$ and \\$\operatorname{supp}(\psi) \subseteq B(0,R)$, then
\begin{equation}\label{eqn: prop compact part}
\left\Vertg\cdot\psi - \psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}\leq \left(R C_A+\left\Vertb\right\Vert_2 \right)R^{d/2}\omega_d^{1/2}\left\Vert\psi\right\Vert_{L}+\frac{1}{2}\left\vert\ln \det A\right\vert\left\Vert\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)},
\end{equation}
where $\omega_d$ is the volume of the unit ball in $\mathbb{R}^d$.

<!-- lem-1: body_md -->
\label{lem: semigroup bound}
Suppose $f\in \mathbf{W}^{1,2}(\mathbb{R}^d)$ and
$\big\|\left\Vert\cdot\right\Vert_2\nabla f\big\|_{\mathbf{L}^2(\mathbb{R}^d)}<\infty$.
For every $M\in\mathbb{R}^{d\times d}$, it holds that
\[
\left\Verte^M\cdot f-f\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq
\left\Vert\frac{1}{2}\operatorname{tr}(M) f+\left<\nabla f,M\cdot\right>\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}.
\]
<!-- end lem-1 -->
<!-- proof-2: body_md -->
The set of functions that satisfy the hypotheses forms a Banach space $X$ with norm
\[
\|f\|_X
=\|f\|_{\mathbf{L}^2(\mathbb{R}^d)}+\|\nabla f\|_{\mathbf{L}^2(\mathbb{R}^d)}+\big\|\left\Vert\cdot\right\Vert_2\nabla f\big\|_{\mathbf{L}^2(\mathbb{R}^d)}.
\]
One may show that $\mathbf{C}_c^\infty(\mathbb{R}^d)$ is dense in $X$ by cutting off and mollifying.
Since the left- and right-hand sides of the desired inequality are continuous over $X$, we may assume $f\in\mathbf{C}_c^\infty(\mathbb{R}^d)$ without loss of generality.

Now define $F:[0,1]\to\mathbf{L}^2(\mathbb{R}^d)$ by $F(t):=e^{tM}\cdot f$, i.e.,
\[
F(t)(x)
=e^{-t\operatorname{tr}(M)/2}f(e^{-tM}x).
\]
By the fundamental theorem of calculus, it suffices to bound the norm of
\[
e^M\cdot f-f
=F(1)-F(0)
=\int_0^1 F'(t)\,dt,
\]
where $F'(t)(x):=\frac{\partial}{\partial t}(F(t)(x))$.
The product rule and chain rule together give
\[
F'(t)(x)
=e^{-t\operatorname{tr}(M)/2}
\left(
-\frac{1}{2}\operatorname{tr}(M)f(e^{-tM}x)
+\left<\nabla f(e^{-tM}x),-Me^{-tM}x\right>
\right)
=(e^{tM}\cdot g)(x),
\]
where
\[
g(x):=-\frac{1}{2}\operatorname{tr}(M)f(x)
+\left<\nabla f(x),-Mx\right>.
\]
Then Minkowski's integral inequality gives
\[
\left\Verte^M\cdot f-f\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq\int_0^1\left\VertF'(t)\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}\,dt
=\int_0^1\left\Verte^{tM}\cdot g\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}\,dt
=\left\Vertg\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)},
\]
where the last step uses the fact that $e^{tM}$ acts isometrically on $\mathbf{L}^2(\mathbb{R}^d)$.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
Put $P=(A^\top A)^{1/2}$ and $U=AP^{-1}$.
Since $\det A>0$, the matrix $P$ is positive definite and
$U\in\operatorname{SO}(d)$. 
Then we can decompose $g$ as
\[
g=(I,b)(U,0)(P,0),
\]
and so the triangle inequality and unitarity together give
\begin{equation}\label{eq: three components}
\left\Vertg\cdot \psi-\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq\left\VertP\cdot\psi-\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
+\left\VertU\cdot\psi-\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
+\left\Vert(I,b)\cdot\psi-\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}.
\end{equation}
Since $P$ is positive definite, its spectral decomposition gives
$S:=\log P$, which is real and symmetric.
Applying Lemma~\ref{lem: semigroup bound} and the Cauchy--Schwarz
inequality then gives
\[
\left\VertP\cdot\psi-\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq\frac{1}{2}\left\vert\operatorname{tr}(S)\right\vert\left\Vert\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
+\left\VertS\right\Vert_{2\to2}\big\|\left\Vert\cdot\right\Vert_2\nabla\psi\big\|_{\mathbf{L}^2(\mathbb{R}^d)}.
\]
Next, by the maximal torus theorem, there exists a logarithm
$K\in\mathfrak{so}(d)$ of $U\in\operatorname{SO}(d)$ with $\left\VertK\right\Vert_{2\to2}\leq\pi$.
The skew-symmetric matrix $K$ has trace zero, and so Lemma~\ref{lem: semigroup bound} and Cauchy--Schwarz
again give
\[
\left\VertU\cdot\psi-\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq\left\Vert\left<\nabla\psi,K\cdot\right>\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq\left\VertK\right\Vert_{2\to2}\big\|\left\Vert\cdot\right\Vert_2\nabla\psi\big\|_{\mathbf{L}^2(\mathbb{R}^d)}.
\]
For the translation by $b\in\mathbb{R}^d$, the standard Sobolev estimate
\cite[Proposition~9.3]{brezis2011functional} yields
\[
\left\Vert(I,b)\cdot\psi-\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq\left\Vertb\right\Vert_2\left\Vert\nabla\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}.
\]
To identify the constants, the eigenvalues of $S$ and the rotation
angles of $U$ give
\[
\operatorname{tr}(S)=\ln\det A,
\qquad
\left\VertS\right\Vert_{2\to2}
=\max\big\{\ln\left\VertA\right\Vert_{2\to2},\,
\ln\left\VertA^{-1}\right\Vert_{2\to2}\big\},
\]
\[
\left\VertK\right\Vert_{2\to2}
=2\arcsin\left(\frac{1}{2}\left\VertU-I\right\Vert_{2\to2}\right).
\]
Thus $\left\VertS\right\Vert_{2\to2}+\left\VertK\right\Vert_{2\to2}=C_A$.
Substituting the three estimates into~\eqref{eq: three components} yields~\eqref{eq: affine bound}.
Finally, if $\psi$ is Lipschitz and supported in $B(0,R)$, then
\[
\left\Vert\nabla\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}
\leq R^{d/2}\omega_d^{1/2}\left\Vert\psi\right\Vert_L,
\qquad
\big\|\left\Vert\cdot\right\Vert_2\nabla\psi\big\|_{\mathbf{L}^2(\mathbb{R}^d)}
\leq R^{1+d/2}\omega_d^{1/2}\left\Vert\psi\right\Vert_L,
\]
and~\eqref{eqn: prop compact part} follows.

\label{thm: continuity of xi}
Let $g=(A,b)\in G$, and let $\psi \in \mathbf{W}^{1,2}(\mathbb{R}^d)$. Assume $\det(A) >0$ and that  $\big{\|}\left\Vert\cdot\right\Vert_2 \nabla \psi\big{\|}_{\mathbf{L}^2(\mathbb{R}^d)}< \infty$. 
Then for every $f\in\mathbf{L}^2(\mathbb{R}^d)$ and $\lambda\in\Lambda$, we have
\begin{align*}
\lefteqn{|\Xi(\Psi(g\cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)| }\\
&\leq \sup_{h\in H}\left(C_{A} \big{\|}\left\Vert\cdot\right\Vert_2 \nabla \psi_{\lambda h}\big{\|}_{\mathbf{L}^2(\mathbb{R}^d)}+\frac{1}{2}\left\vert\ln \det A^{}\right\vert \left\Vert\psi_{\lambda h}\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}  +\left\VertA^{-1}b\right\Vert_2 \left\Vert\nabla \psi_{\lambda h}\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}\right)\cdot\|f\|_{\mathbf{L}^2(\mathbb{R}^d)},
\end{align*}
where $C_A$ is defined as in~\eqref{eqn: CA}.
If we further assume that $\psi$ is Lipschitz and compactly supported with Lipschitz constant $\left\Vert\psi\right\Vert_{L}$ and $\operatorname{supp}(\psi) \subseteq B(0,R)$, then for every $f\in\mathbf{L}^2(\mathbb{R}^d)$  and $\lambda\in\Lambda$, we have
\begin{align}
\lefteqn{|\Xi(\Psi(g\cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)|}\nonumber\\
&\leq \left(\sup_{h \in H} \left(\tilde{R}_{\lambda h} C_{A}+\left\VertA^{-1}b\right\Vert_2 \right)\tilde{R}_{\lambda h}^{d/2}\omega_d^{1/2}\left\Vert\psi_{\lambda h}\right\Vert_{L}+\frac{1}{2}\left\vert\ln \det A\right\vert \left\Vert\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)}\right)\cdot\|f\|_{\mathbf{L}^2(\mathbb{R}^d)},\label{eqn: theorem result compact case}
\end{align} 
where $\omega_d$ is the volume of the unit ball in $\mathbb{R}^d$,  
$A_{\lambda h}$ and $b_{\lambda h}$ are the matrix and vector such that $(\lambda h)(x)=A_{\lambda h}x+b_{\lambda h}$, and $\tilde{R}_{\lambda h} = R\left\VertA_{\lambda h}\right\Vert_{2 \rightarrow 2} + \left\Vertb_{\lambda h}\right\Vert_2$.

<!-- lem-2: body_md -->
\label{lem: continuity simple}
Let $g=(A,b)\in G$, and assume that $\psi\in\mathbf{L}^2(\mathbb{R}^d)$.  Then, for $f\in\mathbf{L}^2(\mathbb{R}^d)$ and $\lambda\in\Lambda$, we have
\[
|\Xi(\Psi(g\cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)|
\leq \|f\|_{\mathbf{L}^2(\mathbb{R}^d)} \sup_{h\in H}\|\psi_{g^{-1}\lambda h}-\psi_{\lambda h}\|_{\mathbf{L}^2(\mathbb{R}^d)}.
\]
<!-- end lem-2 -->
<!-- proof-4: body_md -->
Applying Proposition~\ref{prop: Psi Equivariant} gives
\[
\Xi(\Psi(g\cdot f))(\lambda) =\Xi(g\cdot \Psi(f))(\lambda)
=\sup_{h\in H} \Psi(f)(g^{-1}\lambda h)
=\sup_{h\in H} \langle f,\psi_{g^{-1}\lambda h}\rangle_{\mathbf{L}^2(\mathbb{R}^d)},
\]
and similarly, $\Xi(\Psi(f))(\lambda) =\sup_{h\in H} \langle f,\psi_{\lambda h}\rangle$.  
Therefore,
\begin{align*}
|\Xi(\Psi(g\cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)|
&=\left|\sup_{h\in H} \langle f,\psi_{g^{-1}\lambda h}\rangle_{\mathbf{L}^2(\mathbb{R}^d)}-\sup_{h\in H} \langle f,\psi_{\lambda h}\rangle_{\mathbf{L}^2(\mathbb{R}^d)}\right|\nonumber \\
&\leq \sup_{h\in H}\left| \langle f,\psi_{g^{-1}\lambda h}\rangle_{\mathbf{L}^2(\mathbb{R}^d)}- \langle f,\psi_{\lambda h}\rangle_{\mathbf{L}^2(\mathbb{R}^d)}\right|\nonumber \\
&\leq \|f\|_{\mathbf{L}^2(\mathbb{R}^d)} \sup_{h\in H}\|\psi_{g^{-1}\lambda h}-\psi_{\lambda h}\|_{\mathbf{L}^2(\mathbb{R}^d)},
\end{align*}
where the last step applies the Cauchy--Schwarz inequality.

    \[

For all $x,y\in\mathbb{R}^d$, it holds that

\label{ex: cont rotations}

One of our primary motivations is to develop machine learning methods for images or data cubes with local rotational invariance\footnote{Here, ``local rotational invariance'' means stability to global rotations that are close to the identity. This contrasts with other work, e.g.,~\cite{andrearczyk2020local}, where ``local rotational invariance'' refers to invariance to a full group of rotations that are applied to small patches of an image.}, in which case the relevant symmetry group is~$\operatorname{SO}(2)$.
We will analyze the Lipschitz behavior of $\Xi\circ\Psi$ on  $G=\operatorname{SO}(d)$ for arbitrary $d>1$.

We equip $\operatorname{SO}(d)$ with the following metric.
For $U,\tilde{U}\in\operatorname{SO}(d)$, the real Schur decomposition shows that
$U\tilde{U}^{-1}$ is orthogonally similar to a block diagonal matrix
with rotation blocks
\begin{equation}\label{eqn:schurrot}
\begin{pmatrix}
\cos(\theta_j) & -\sin(\theta_j) \\
\sin(\theta_j) & \cos(\theta_j)
\end{pmatrix},
\qquad
\theta_j\in(-\pi,\pi],
\quad j=1,\ldots,\lfloor d/2\rfloor,
\end{equation}
and one additional block $(1)$ when $d$ is odd.
We define
\begin{equation}\label{eqn:metSOd}
\operatorname{dist}_{\operatorname{SO}(d)}: \operatorname{SO}(d) \times \operatorname{SO}(d) \rightarrow \mathbb{R}, \quad \operatorname{dist}_{\operatorname{SO}(d)}(U, \tilde{U}) = \bigg(\sum_j |\theta_j|^2\bigg)^{1/2}.
\end{equation}
Up to a constant factor, this is the bi-invariant Riemannian
distance induced by the Frobenius inner product
(e.g.,~\cite{sinclair2008minimum}).


Now, we observe that for any $U \in \operatorname{SO}(d)$, we have $\det(U) = \left\VertU\right\Vert_{2 \rightarrow 2} =1$ and so~\eqref{eqn: CA} gives
\begin{equation}
\label{eqn:Cpurerot}
C_{U} 
= 2\arcsin\left( \frac{1}{2}\left\VertU-I\right\Vert_{2 \rightarrow 2}\right)
=\max_j \left\vert\theta_j\right\vert 
\leq \bigg(\sum_j |\theta_j|^2\bigg)^{1/2}
=
\operatorname{dist}_{\operatorname{SO}(d)}(U, I).
\end{equation}
Next, we let $\Lambda$ be an arbitrary closed subgroup of $G$, let $H$ be an arbitrary precompact set of coset representatives, and observe that  we have $\tilde{R}_{\lambda h} = R$ for all $\lambda\in\Lambda, h\in H$. 
Also, Lemma~\ref{lem:dillip} implies $\|\psi_{\lambda h}\|_L=\|\psi\|_L$, and
so for $g=(U,0)$, the bound~\eqref{eqn: theorem result compact case} gives
\begin{equation}
|\Xi(\Psi(g \cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)|
\leq \omega_d^{1/2} R^{(d+2)/2} \left\Vert\psi\right\Vert_{L} \|f\|_{\mathbf{L}^2(\mathbb{R}^d)}\cdot\operatorname{dist}_{\operatorname{SO}(d)}(U, I).\label{eqn: rotation bound examle from theorem}
\end{equation}
We see that this bound does not depend on the choice of $\Lambda$, which will not be the case in our next example (Example~\ref{ex: cont trans}).
While this is evidence that the bound may not be tight in general, it does show that for a particular application, one may choose a pooling domain that decreases computational complexity while preserving distinctions relevant to the task without changing this bound.
For example, in Section~\ref{sec: decreasing training set}, we want to distinguish between horizontally dominant and vertically dominant ellipses and thus do not pool rotations of $90^\circ$ together.

\label{rem: rotation better bound}
In the case where $d=2$, we may further simplify~\eqref{eqn: rotation bound examle from theorem} by noting that we may write a generic element of $\operatorname{SO}(2)$ as 
\[
U = \begin{pmatrix}
    \cos(\theta) & -\sin(\theta) \\
    \sin(\theta) & \cos(\theta)
\end{pmatrix},
\]
with $\theta \in (-\pi,\pi]$.
So, since $\left\vert\theta\right\vert = \operatorname{dist}_{\operatorname{SO}(2)}(U,I)$, the right-hand side of~\eqref{eqn: rotation bound examle from theorem} becomes
\begin{equation}
\pi^{1/2} R^2 \left\Vert\psi\right\Vert_{L} \|f\|_{\mathbf{L}^2(\mathbb{R}^2)}\left\vert\theta\right\vert.\label{eqn: 2drotation bound examle from theorem}
\end{equation}
We also note that we may improve on this estimate by using techniques specific to $\operatorname{SO}(2)$ to show that for every $\lambda$ we have 
\begin{align}\label{eqn: rotation specific bound}
|\Xi(\Psi(g\cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)|
\leq  2\pi^{1/2}R^2\|\psi\|_L \sin\left(\frac{|\theta|}{2}\right)\|f\|_{\mathbf{L}^2(\mathbb{R}^2)}.
\end{align}
For a proof of~\eqref{eqn: rotation specific bound}, please see Appendix~\ref{sec: details on better rotation bound}, where it is stated as Proposition~\ref{prop: Rotation special case}.
Since $\sin(x)\sim x$ as $x\to0$,
the right-hand sides of~\eqref{eqn: 2drotation bound examle from theorem} and~\eqref{eqn: rotation specific bound} agree to leading order.
See Section~\ref{sec: continuity of pooling} for a numerical illustration in the context of a digital implementation of $\Xi\circ\Psi$.

Let $G = \operatorname{SE}(2)=\operatorname{SO}(2) \ltimes \mathbb{R}^2$.
For $\theta\in (-\pi,\pi]$, we define $U_\theta\in \operatorname{SO}(2)$ by

and consider the following right-invariant metric on the special Euclidean group $G$
\begin{equation}\label{eqn:metse2}
\operatorname{dist}_{\operatorname{SE}(2)} ((U_\theta,b),(U_{\tilde{\theta}},\tilde{b})) = \min_{k\in \mathbb{Z}} \left\vert\theta-\tilde{\theta}+2\pi k\right\vert +\left\Vertb-U_{\theta-\tilde{\theta}}\tilde{b}\right\Vert_2.
\end{equation}
This metric arises by computing $(V,v) = (U_\theta,b)(U_{\tilde{\theta}},\tilde{b})^{-1}$, applying the standard bi-invariant geodesic distance on $\operatorname{SO}(2)$ from $I$ to $V$ to attain the first term and the (bi-invariant) Euclidean distance between $v$ and $0$ to attain the second term.

and let $\Lambda = L \ltimes c\mathbb{Z}^2$ for some fixed $c>0$.
Note that $c\mathbb{Z}^2$ is invariant under multiplication by elements of $L$, i.e., $L(c\mathbb{Z}^2)= c\mathbb{Z}^2$,
yielding that $\Lambda\leq G\leq \operatorname{AGL}(\mathbb{R}^2)$ are nested subgroups.
Let $H=\left\{U_\theta \, : \, \theta \in [0,\pi/2)\right\} \times [-\frac{c}{2},\frac{c}{2})^2$. 

We recall from~\eqref{eqn:Cpurerot} that
$C_{U_\theta} =  \left\vert\theta\right\vert$ and
similar to Example~\ref{ex: cont trans} note that
\[
\tilde{R}_{\lambda h} = R + \left\Vertb_{\lambda h}\right\Vert_2 \leq R+ \left\Vertb_\lambda\right\Vert_2 + \frac{c}{\sqrt{2}},
\qquad 
\left\Vert\psi_{\lambda h}\right\Vert_L = \left\Vert\psi\right\Vert_L.
\]
Plugging this into~\eqref{eqn: theorem result compact case}, we obtain for every $\lambda$ that
\begin{align}
\lefteqn{|\Xi(\Psi((U_\theta,b)\cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)|}\nonumber\\
&\leq \left(\left(R+ \left\Vertb_\lambda\right\Vert_2 + \frac{c}{\sqrt{2}} \right) \left\vert\theta\right\vert+\left\Vertb\right\Vert_2 \right)\left(R+ \left\Vertb_\lambda\right\Vert_2 + \frac{c}{\sqrt{2}} \right)\pi^{1/2}\left\Vert\psi\right\Vert_{L}\|f\|_{\mathbf{L}^2(\mathbb{R}^2)}\nonumber\\
&\leq\max\left\{R+ \left\Vertb_\lambda\right\Vert_2 + \frac{c}{\sqrt{2}},1 \right\} \left(R+ \left\Vertb_\lambda\right\Vert_2 + \frac{c}{\sqrt{2}} \right)\pi^{1/2}\left\Vert\psi\right\Vert_{L}\|f\|_{\mathbf{L}^2(\mathbb{R}^2)}
\cdot\operatorname{dist}_{\operatorname{SE}(2)}((U_\theta,b),(I,0)).\label{eqn:bdrot4}
\end{align}

\leq \left\Vert\psi_{(U_\theta,b)^{-1}\lambda}-\psi_{\lambda}\right\Vert_{\mathbf{L}^2(\mathbb{R}^2)} \left\Vertf\right\Vert_{\mathbf{L}^2(\mathbb{R}^2)},
\]
from which an application of Theorem~\ref{prop:contest} and Lemma~\ref{lem: continuity simple} reproduces the bound~\eqref{eqn:bdrot4}.
Thus, pooling with $\Xi$ has a benign impact on sensitivity to rototranslations while reducing the size of the representation.

\label{lem:similmet}
Let $G$ be the direct similitude group $(\mathbb{R}_{>0} \times \operatorname{SO}(d)) \ltimes \mathbb{R}^d$.
Define $N: G \rightarrow \mathbb{R}_{\geq 0}$ as
\[
N((s,U,b)) = \left\vert\ln s\right\vert + \operatorname{dist}_{\operatorname{SO}(d)}(U,I) + s^{-1}\left\Vertb\right\Vert_2,
\]
where $(s,U,b)$ stands for the element of $G$ mapping $x \mapsto s U x + b$ and $\operatorname{dist}_{\operatorname{SO}(d)}$ is the bi-invariant Riemannian distance~\eqref{eqn:metSOd}.
Then, 
\[\operatorname{disim}_{(\mathbb{R}_{>0} \times \operatorname{SO}(d)) \ltimes \mathbb{R}^d} : \left((\mathbb{R}_{>0} \times \operatorname{SO}(d)) \ltimes \mathbb{R}^d\right) \times \left((\mathbb{R}_{>0} \times \operatorname{SO}(d)) \ltimes \mathbb{R}^d\right) \rightarrow \mathbb{R}_{\geq 0}\]
defined as
\begin{align}
\operatorname{disim}_{(\mathbb{R}_{>0} \times \operatorname{SO}(d)) \ltimes \mathbb{R}^d}\left( (s,U,b), (\tilde{s},\tilde{U},\tilde{b})\right)
&= \min\{ N((s,U,b)(\tilde{s},\tilde{U},\tilde{b})^{-1}),\, N((\tilde{s},\tilde{U},\tilde{b})(s,U,b)^{-1})\}\label{eqn:simdisim}
\end{align}
is positive definite, symmetric, and right-invariant.

\label{ex:dirsim} Let $G$ be the direct similitude group $(\mathbb{R}_{>0} \times \operatorname{SO}(d)) \ltimes \mathbb{R}^d$ that acts on $\mathbb{R}^d$ as $x \mapsto s U x + b$.
In order to specify $\Xi_w$, fix $a>1$, $c >0$, and a closed subgroup $\Lambda_r \leq \operatorname{SO}(d)$, where $H_r$ is a precompact set of right coset representatives of $\Lambda_r$ in $\operatorname{SO}(d)$.
Then define $\Lambda$ to be the set
\[
\Lambda=\{(a^\ell, W, ck) \,:\,\ell \in \mathbb{Z},\, W \in \Lambda_r, \,k\in\mathbb{Z}^d\}
\]
and $H$ the transversal $H = [1,a) \times H_r \times \left[-\frac{c}{2},\frac{c}{2} \right)^d$.
We will use Proposition~\ref{prop: continuity of xiw} to show that $\Xi_w\circ\Psi$ is group-Lipschitz with respect to the semimetric defined in Lemma~\ref{lem:similmet}.
To that end, \eqref{eqn:Cpurerot} gives
\[
C_{s U} \leq \left\vert\ln s\right\vert + \operatorname{dist}_{\operatorname{SO}(d)}(U,I)
\]
for every $s > 0$ and $U \in \operatorname{SO}(d)$.
Next, for $(a^\ell,W,ck)\in\Lambda$, we write $\lambda_D:=a^\ell W$ and $\lambda_T:=ck$, and for $h=(h_s,h_r,h_T)\in H$, we write $h_D:=h_s h_r$.
Then
\[
\tilde{R}_{(\lambda_D h_D,\lambda_T+h_T)} 
= R\left\Vert\lambda_D h_D\right\Vert_{2 \rightarrow 2} + \left\Vert\lambda_T+h_T\right\Vert_2
= a^\ell h_s R + \left\Vertck +h_T\right\Vert_2
\leq a^\ell h_s R +c\left\Vertk\right\Vert_2 + \frac{c}{2} \sqrt{d},
\]
while Lemma~\ref{lem:dillip} gives
\[
\left\Vert\psi_{(\lambda_D h_D,\lambda_T+h_T)}\right\Vert_{L} 
\leq (a^\ell h_s)^{-d/2} (a^\ell h_s)^{-1} \left\Vert\psi\right\Vert_L 
=(a^\ell h_s)^{-(d+2)/2}  \left\Vert\psi\right\Vert_L.
\]
Finally, for each $(s,U,b)\in G$, we have $\left\Vert(sU)^{-1}b\right\Vert_2 = s^{-1} \left\Vertb\right\Vert_2$ and $\left\vert\ln \det sU\right\vert = d \left\vert\ln s\right\vert$.
Putting this all together, one may obtain from~\eqref{eqn: theorem result compact case w} that
\begin{equation}
\label{eqn:onesidesimil}
|\Xi_w(\Psi((sU, b)\cdot f))(\lambda_D,\lambda_T)-\Xi_w(\Psi(f))(\lambda_D,\lambda_T)|
\leq C \cdot N(s,U,b) \cdot \|f\|_{\mathbf{L}^2(\mathbb{R}^d)}
\end{equation}
for every $f\in \mathbf{L}^2(\mathbb{R}^d)$ and $(s,U,b)\in G$,
where
\begin{align}
C &:= \max \bigg\{ \omega_d^{1/2} \left\Vert\psi\right\Vert_L \left( R + \frac{c}{a^\ell}\left(\left\Vertk\right\Vert_2 + \sqrt{d}/2\right)\right)^{(d+2)/2} + \frac{d}{2}\left\Vert\psi\right\Vert_{\mathbf{L}^2(\mathbb{R}^d)},\nonumber\\
&\hspace{2.5in}\omega_d^{1/2} \left\Vert\psi\right\Vert_L(a^\ell)^{-1}\left( R + \frac{c}{a^\ell }\left(\left\Vertk\right\Vert_2 + \sqrt{d}/2\right)\right)^{d/2} \bigg\}.\label{eqn:Cbdsimil}
\end{align}
Finally, we upgrade the bound~\eqref{eqn:onesidesimil} using the facts that 
\[
|\Xi_w(\Psi(g\cdot f))(\lambda_D,\lambda_T)-\Xi_w(\Psi(f))(\lambda_D,\lambda_T)|
= 
|\Xi_w(\Psi(g^{-1} \cdot (g\cdot f)))(\lambda_D,\lambda_T)-\Xi_w(\Psi(g \cdot f))(\lambda_D,\lambda_T)|
\]
and $\|g \cdot f\|_{\mathbf{L}^2(\mathbb{R}^d)}
=\|f\|_{\mathbf{L}^2(\mathbb{R}^d)}$:
\[
|\Xi_w(\Psi(g\cdot f))(\lambda_D,\lambda_T)-\Xi_w(\Psi(f))(\lambda_D,\lambda_T)|
\leq C \cdot \|f\|_{\mathbf{L}^2(\mathbb{R}^d)}\cdot \operatorname{disim}_{(\mathbb{R}_{>0} \times \operatorname{SO}(d)) \ltimes \mathbb{R}^d}\left( (s,U,b), (1,I,0)\right).
\]
That is, $\Xi_w\circ\Psi$ is group-Lipschitz, as desired.
Analyzing~\eqref{eqn:Cbdsimil}, we see that $C$ is smaller when $c$ is smaller, i.e., when we pool over a smaller region of translations.  
This is in line with what we saw in Example~\ref{ex: cont trans}.
Meanwhile, unlike Example~\ref{ex: cont dilations}, the bound varies with $a$ in a way that depends on the sign of $\ell$.

\label{prop: Rotation special case}
Let $\psi\in\mathbf{L}^2(\mathbb{R}^2)$ be a Lipschitz function with compact support such that $\operatorname{supp}(\psi)\subseteq B(0,R).$ Let $G=\operatorname{SO}(2)$, let $g\in G$, and as in Example~\ref{ex: cont rotations}, note that there exists $\theta \in (-\pi,\pi]$ such that $g=(U,0)$, where 
$$
U= \begin{pmatrix}
    \cos(\theta) & -\sin(\theta) \\
    \sin(\theta) & \cos(\theta)
\end{pmatrix}.
$$
 Then for every $f\in \mathbf{L}^2(\mathbb R^2)$ and $\lambda\in\Lambda$,
\begin{align*}
|\Xi(\Psi(g\cdot f))(\lambda)-\Xi(\Psi(f))(\lambda)|
\leq  2\pi^{1/2}R^2\|\psi\|_L \sin\left(\frac{|\theta|}{2}\right)\|f\|_{\mathbf{L}^2(\mathbb{R}^2)}.
\end{align*}

\label{lem: ell infinity diff for different group members}
Assume that $\psi\in\mathbf{L}^2(\mathbb{R}^2)$ is Lipschitz, with Lipschitz constant $\|\psi\|_{L}$, and is compactly supported with $\operatorname{supp}(\psi)\subseteq B(0,R)$.
Let $G=\operatorname{SO}(2)$, and let $g_{1},g_{2}\in G$ with $g_{j}(x)=U_jx$,  
$$
U_j= \begin{pmatrix}
    \cos(\theta_j) & -\sin(\theta_j) \\
    \sin(\theta_j) & \cos(\theta_j)
\end{pmatrix},
$$
for $j=1,2$. 
Then, 
\begin{align*}
 \|\psi_{g_{1}}-\psi_{g_{2}}\|_{\mathbf{L}^2(\mathbb{R}^2)}\leq 2\pi^{1/2}R^2\|\psi\|_L \left\vert\sin\left(\frac{\theta_1-\theta_2}{2}\right)\right\vert.
 \end{align*}

Since $\det(U_j)=1$, we have $\psi_{g_j}(x)=\psi(U_j^{-1}x)$ for $j=1,2$.
Therefore, since  $\operatorname{supp}(\psi_{g_1}-\psi_{g_2})\subseteq B(0,R)$, we may use the fact that $U_1$ is an orthogonal matrix to obtain
\begin{align}
    {\|\psi_{g_{1}}-\psi_{g_{2}}\|_{\mathbf{L}^2(\mathbb{R}^2)}} &=\left(\int_{B(0,R)}{|\psi(U_1^{-1}x)-\psi(U_2^{-1}x)|^2}\,dx\right)^{1/2}\nonumber\\
    &\leq |B(0,R)|^{1/2}\|\psi\|_L\|U_1^{-1}-U_2^{-1}\|_{2\rightarrow 2} R\nonumber\\
    &=\pi^{1/2}R^2\|\psi\|_L \|U_1(U_1^{-1}-U_2^{-1})\|_{2\rightarrow 2}\nonumber\\
              &=\pi^{1/2}R^2\|\psi\|_L \|I-U_1U_2^{-1}\|_{2\rightarrow 2}.\nonumber
    \end{align}
The proof is now completed by observing
that 
$$
U_1U_2^{-1}=\begin{pmatrix}
\cos(\theta_1-\theta_2)&&-\sin(\theta_1-\theta_2)\\
\sin(\theta_1-\theta_2)&&\cos(\theta_1-\theta_2)
\end{pmatrix},
$$
the matrix which corresponds to a rotation by $\theta_1-\theta_2$. Thus,
we have  
$$
\|I-U_1U_2^{-1}\|_{2\rightarrow 2}=|1-e^{i(\theta_1-\theta_2)}|=2\left\vert\sin\left(\frac{\theta_1-\theta_2}{2}\right)\right\vert
$$
which completes the proof.
