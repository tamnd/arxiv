<!-- conjecture-1: body_md -->
\label{conj:hata}
Every Peano continuum is a topological fractal.
<!-- end conjecture-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Let $S$ be a space homeomorphic to the circle. If continuous maps $f,g\colon S\to S$
satisfy
\[
S=f(S)\cup g(S),
\]
then $\{f,g\}$ is not topologically contractive.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:wn}
The witnessing number of the circle is equal to three.
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
The upper bound $\operatorname{wn}(S)\le3$ is simple and it is contained in~\cite{KarasovaVejnar}. A single witnessing map on a
nondegenerate compact space would have to be surjective, and hence all of its iterates
would be surjective, so one map cannot be topologically contractive.
Theorem~\ref{thm:main} excludes two witnessing maps. Therefore $\operatorname{wn}(S)\geq3$.
Consequently $\operatorname{wn}(S)= 3$.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{prop:finite}
For every $\varepsilon>0$ there are continuous maps $f,g\colon\mathbb T\to\mathbb T$ and an integer
$N\ge1$ such that
\begin{equation*}                
f(\mathbb T)\cup g(\mathbb T)=\mathbb T
\end{equation*}
and
\begin{equation*}                
\operatorname{diam}_{d_\mathbb T}w(\mathbb T)<\varepsilon\qquad\text{for every } w\in\{f,g\}^N.
\end{equation*}
The maps may be chosen so that $g=f+\tfrac12\pmod 1$.

Let $J$ be a subarc of either of the two semicircles \([0,1/2]\) or \([1/2,1]\). On
each of these semicircles, $f_L$ has one branch of domain length $\delta$ and absolute slope $2$,

\begin{equation}\label{eq:2.6}
\operatorname{diam} f_L(J)\le\max\Bigl\{\frac1L,\ r\operatorname{diam} J\Bigr\}.
\end{equation}
Indeed, the variation on the short branch is at most $2\delta=1/L$, while on the other
branch it is at most $r\operatorname{diam} J$. If $J$ meets both branches, the total image diameter is
the larger of the two one-sided variations from the common extremum. Since $g_L$ differs
from $f_L$ by an isometry of the target, \eqref{eq:2.6} also holds with $g_L$ in place of
$f_L$.

\[
D_{n+1}\le\max\Bigl\{\frac1L,\ rD_n\Bigr\},

and induction gives
\begin{equation}\label{eq:2.7}
D_n\le\max\Bigl\{\frac1L,\ \frac12r^{\,n-1}\Bigr\}.
\end{equation}

<!-- remark-1: body_md -->
\label{rem:quant}
Proposition~\ref{prop:finite} has the quantifier order
\[

\]
whereas two witnessing maps would require one fixed pair satisfying
\[
\exists f,g\ \forall\varepsilon>0\ \exists N.
\]
The distinction is essential. For the maps in the proof, put

not topologically contractive. Theorem~\ref{thm:main} shows that no other fixed covering
pair can be topologically contractive either.

<!-- lemma-1: body_md -->
\label{lem:metric}
Let $(X,d)$ be a nonempty compact metric space and let $\mathcal F$ be a finite topologically
contractive family of continuous self-maps of $X$. Put $W=\bigcup_{n\ge0}\mathcal F^n$ and
define
\begin{equation}\label{eq:3.2}
\rho(x,y)=\sup_{w\in W}d\bigl(w(x),w(y)\bigr).
\end{equation}
Then $\rho$ is a compatible metric on $X$ and every $h\in\mathcal F$ is nonexpanding with respect to
$\rho$.
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
Compactness makes $d$ bounded, so the supremum in \eqref{eq:3.2} is finite. Symmetry and
the triangle inequality follow from the corresponding properties of $d$. Since
$\mathrm{id}_X\in W$, we have $\rho\ge d$.                                

To prove compatibility of $\rho$ and $d$, fix $\eta>0$. Choose $N$ such that $D_n(d,\mathcal F)<\eta/2$ for all
$n\ge N$.                                                                               
The finitely many $w\in \bigcup_{n=0}^{N-1}\mathcal F^n$ are uniformly continuous, so there is $\delta>0$ such
that
\[
d(x,y)<\delta\quad\Longrightarrow\quad d\bigl(w(x),w(y)\bigr)<\eta/2
\]
for every such $w$. Hence $d(x,y)<\delta$ implies
$\rho(x,y)\le\eta/2<\eta$. Together with $d\le\rho$, this proves that $d$ and $\rho$
generate the same topology.

For $h\in\mathcal F$ we have $\{w\circ h : w\in W\}\subseteq W$, and therefore
\begin{equation*}                
\rho\bigl(h(x),h(y)\bigr)=\sup_{w\in W}d\bigl((w\circ h)(x),(w\circ h)(y)\bigr)
\le\rho(x,y).
\end{equation*}
Thus, every $h\in\mathcal F$ is nonexpanding with respect to $\rho$.

<!-- lemma-2: body_md -->
\label{lem:restr}
Let $\varepsilon>0$. If $K,K_1,\dots,K_m$ are nonempty subsets of $S$ with
$K\subseteq K_1\cup\dots\cup K_m$, then
\begin{equation*}                
N_\varepsilon(K)\le\sum_{i=1}^mN_\varepsilon(K_i).
\end{equation*}
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
The union of optimal $\varepsilon$-covers of the sets $K_i$ is an $\varepsilon$-cover of
$K$.
<!-- end proof-4 -->
<!-- lemma-3: body_md -->
\label{lem:nonexp}
Let $h\colon S\to S$ be continuous and nonexpanding. If $K\subseteq S$ is nonempty, then
\begin{equation*}                
N_\varepsilon(h(K))\le N_\varepsilon(K)\qquad\text{for every }\varepsilon>0.
\end{equation*}
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
If $\mathcal C$ is an optimal $\varepsilon$-cover of $K$, then the sets $h(C)$,
$C\in\mathcal C$, are connected, their union contains $h(K)$, and
$\operatorname{diam}_\rho h(C)\le\operatorname{diam}_\rho C\le\varepsilon$.

<!-- lemma-4: body_md -->
\label{lem:cut}
Let $\mathcal P$ be a finite family of closed arcs in $S$ with pairwise disjoint interiors such
that $\bigcup\mathcal P$ is connected. Then
\begin{equation*}                
\sum_{I\in\mathcal P}N_\varepsilon(I)\le N_\varepsilon\Bigl(\bigcup\mathcal P\Bigr)+2|\mathcal P|\qquad
\text{for every }\varepsilon>0.
\end{equation*}
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
Fix $\varepsilon>0$, and put
$M=\bigcup\mathcal P$. Let $\mathcal C$ be an optimal
$\varepsilon$-cover of $M$ consisting of closed sets.
Every point of $M$ belongs to at most two members of $\mathcal C$.
Fix an orientation of $S$, and denote by $a_I$ the initial
endpoint of each $I\in\mathcal P$. For every $C\in\mathcal C$ it can be easily seen that
\[
\bigl|\{I\in\mathcal P:C\cap I\ne\emptyset\}\bigr|
\le 1+\bigl|\{I\in\mathcal P:a_I\in C\}\bigr|.
\]
                                                     
                                                        
                                                        
                                                        
                                                            

Now put
$\mathcal C_I=\{C\in\mathcal C:C\cap I\ne\emptyset\}$.
Since $\mathcal C_I$ covers $I$, Lemma \ref{lem:restr}, double counting and the
bound above give
\[
\begin{aligned}
\sum_{I\in\mathcal P}N_\varepsilon(I)
&\le \sum_{I\in\mathcal P}|\mathcal C_I|\\
&= \sum_{C\in\mathcal C}
   \bigl|\{I\in\mathcal P:C\cap I\ne\emptyset\}\bigr|\\
&\le |\mathcal C|
   +\sum_{I\in\mathcal P}\bigl|\{C\in\mathcal C:a_I\in C\}\bigr|\\
&\le N_\varepsilon(M)+2|\mathcal P|.
\end{aligned}
\]

<!-- lemma-5: body_md -->
\label{lem:end}
If $a,b$ are the distinct endpoints of a closed arc $L\subseteq S$, then
\begin{equation*}                
\rho(a,b)\le\varepsilon N_\varepsilon(L)\qquad\text{for every }\varepsilon>0.
\end{equation*}
In particular, $N_\varepsilon(L)\to\infty$ as $\varepsilon\downarrow0$.
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
Let $\mathcal C$ be an optimal $\varepsilon$-cover of $L$, and fix $C_a\in\mathcal C$ with
$a\in C_a$. Since $L$ is connected, we can find a shortest sequence $C_1=C_a,C_2,\dots,C_k$ with $b\in C_k$ and points $x_i\in C_i\cap C_{i+1}$ for
$1\le i<k$. Put $x_0=a$ and $x_k=b$. Then $x_{i-1},x_i\in C_i$ for every $i$, and
therefore
\[
\rho(a,b)\le\sum_{i=1}^k\rho(x_{i-1},x_i)\le \sum_{i=1}^k\operatorname{diam} C_i\le k\varepsilon\le\varepsilon N_\varepsilon(L).
\]
The final assertion follows from $\rho(a,b)>0$.

<!-- lemma-6: body_md -->
\label{lem:loss}
Let $f,g\colon S\to S$ be nonexpanding maps of a metric circle $(S,\rho)$. Suppose that
\[
A=f(S),\qquad B=g(S)
\]
are nondegenerate proper closed arcs and that $A\cup B=S$. Choose $u,v\in S$ such that
$f(u)$ and $f(v)$ are the two endpoints of $A$. Then every nondegenerate closed arc
$J\subseteq S$ with
\[
\operatorname{int}_S(J)\cap\{u,v\}=\emptyset
\]
satisfies
\begin{equation}\label{eq:5.1}
0\le N_\varepsilon(J)-N_\varepsilon(f(J))\le 12\qquad\text{for every }\varepsilon>0.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-8: body_md -->
Fix $\varepsilon>0$ and abbreviate $N_\varepsilon$ to $N$. By Lemma~\ref{lem:nonexp},
\[
\Delta=N(J)-N(f(J))
\]
is nonnegative.

The points $u,v$ are distinct and divide $S$ into two closed arcs $H_1,H_2$. For each
$i$, the set $f(H_i)$ is a connected subset of $A$ containing both endpoints of $A$.
Hence
\begin{equation}\label{eq:5.2}
f(H_1)=f(H_2)=A.
\end{equation}

Let $Z$ be the set consisting of $u$, $v$ and the two endpoints of $J$, and 
let $\mathcal P$ consist of the closures of components of $S\setminus Z$. Thus $\mathcal P$ is
a family consisting of at
most four closed arcs with disjoint interiors, which have endpoints in $Z$ (see Figure~\ref{fig:loss}). Call the elements of $\mathcal P$ pieces. Since neither $u$
nor $v$ belongs to $\operatorname{int}_S(J)$, the arc $J$ itself is one of the pieces. The pieces
contained in $H_1$ cover $H_1$, so their images under $f$ cover $A$ by \eqref{eq:5.2}, and the same
is true for the pieces contained in $H_2$. Lemma~\ref{lem:restr}, applied on the two
sides, gives
\[
2N(A)=N(f(H_1))+N(f(H_2))\le\sum_{I\in\mathcal P}N(f(I)).
\]
For every piece $I$, Lemma~\ref{lem:nonexp} gives $N(f(I))\le N(I)$, and on the
distinguished piece $J$ the loss is $\Delta$. Hence, by Lemma~\ref{lem:cut} with $M=S$,
since $\mathcal P$ has at most four pieces,
\begin{equation}\label{eq:5.3}
2N(A)\le\sum_{I\in\mathcal P}N(f(I))\le\sum_{I\in\mathcal P}N(I)-\Delta\le N(S)+2\cdot 4-\Delta=N(S)+8-\Delta.
\end{equation}

Suppose, towards a contradiction, that $\mathcal F=\{f,g\}$ is topologically contractive and
that $f(S)\cup g(S)=S$. By Lemma~\ref{lem:metric} there is a compatible metric $\rho$, under which both maps $f$ and $g$ are nonexpanding.

Clearly, neither $f$ nor $g$ is surjective and thus also neither is constant. Therefore
\[
A=f(S),\qquad B=g(S)
\]
are nondegenerate proper closed arcs.

Choose $u,v\in S$ mapped by $f$ to the two endpoints of $A$.
By applying Lemma~\ref{lem:loss}
we get that $f$ cannot be constant on any nondegenerate arc. Otherwise there
would be a nondegenerate subarc $J$ whose interior avoids $u$ and $v$, and
\eqref{eq:5.1} would imply
\[
N_\varepsilon(J)\le N_\varepsilon(f(J))+12=13\qquad\text{for every }\varepsilon>0,
\]
contradicting Lemma~\ref{lem:end}.

Set
\[
J_n=f^n(S),\qquad n\ge1.
\]
The arcs $J_n$ are nondegenerate by the preceding paragraph. They are nested, since
\[
f^{n+1}(S)=f^n(f(S))\subseteq f^n(S),
\]
and their $\rho$-diameters tend to zero.
Hence
\begin{equation}\label{eq:6.1}
\bigcap_{n\ge1}J_n=\{p\}\ \text{ for some } p\in S,\qquad f(p)=p.
\end{equation}
Every inclusion $J_{n+1}\subseteq J_n$ is strict; equality at one stage would make all
subsequent images equal to the same nondegenerate arc.

For all sufficiently large $n$,
\begin{equation}\label{eq:6.2}
\operatorname{int}_S(J_n)\cap\{u,v\}=\emptyset.
\end{equation}
Indeed, any point of $\{u,v\}$ different from $p$ is eventually outside $J_n$, by
\eqref{eq:6.1} and nesting. If, for instance, $u=p$, then since $f(u)$ is an endpoint
of $A$ and $f(u)=f(p)=p$, we conclude that the point $p$
is an endpoint of $J_n$ and not an interior point. 
The same reasoning applies if $v=p$.

Fix an $n$ satisfying \eqref{eq:6.2}, and put
\[
J=J_n,\qquad K=J_{n+1}=f(J).
\]
Then $K\subsetneq J$ and both arcs are nondegenerate. 
Then $J\setminus K$ has at least one (and also at most two) nondegenerate component.
Let $L$ be the closure of such a component, so it is a nondegenerate arc (see Figure~\ref{fig:end}). Then $L$ intersects $K$ in their common endpoint, and $L\cup K\subseteq J$. By Lemma \ref{lem:cut} and
Lemma~\ref{lem:restr},
\[
N_\varepsilon(L)+N_\varepsilon(K)\le N_\varepsilon(L\cup K)+4\le N_\varepsilon(J)+4.
\]
Since $K=f(J)$, Lemma~\ref{lem:loss} yields
\begin{equation}\label{eq:6.3}
N_\varepsilon(L)\le N_\varepsilon(J)-N_\varepsilon(f(J))+4\le12+4=16
\qquad\text{for every }\varepsilon>0.
\end{equation}
The arc $L$ is fixed independently of $\varepsilon$. If $a,b$ are distinct
endpoints of $L$, then Lemma~\ref{lem:end} and \eqref{eq:6.3} give
\[
0<\rho(a,b)\le\varepsilon N_\varepsilon(L)\le16\varepsilon\qquad\text{for every }
\varepsilon>0,
\]
which is a contradiction. Thus the theorem is proved.
