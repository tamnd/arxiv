<!-- theorem-1: body_md -->
\label{thm:main lattice}
	For every positive integer $m$, there exists an irreducible odd
	unimodular lattice $L_m$ of rank $24m$ and minimum norm $3$
	that is $3$-integrable and generated by its minimal vectors.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:main}
	The lattices $L_m$ yield pairwise switching-inequivalent connected
	non-extendable signed graphs $\Gamma_m$ with smallest eigenvalue $-3$.
<!-- end corollary-1 -->
<!-- lemma-1: body_md -->
\label{lem:HM}
	Let $C$ be a ternary self-dual code of minimum weight at
	least $6$.
	Then $C$ is decomposable if and only if $\mathcal{A}_3(C)$ is
	decomposable.
<!-- end lemma-1 -->
<!-- theorem-2: body_md -->
\label{thm:seed}
	For every choice of $C,\mathbf{w},P,Q$ above,
	the following statements hold:
	\begin{enumerate}[label=\textnormal{(\roman*)}]
		\item $\dim D=11$, $C=D\oplus\langle \mathbf{w}\rangle$, and $D\perp \mathbf{q}$;
		\item $D=\langle D_9\rangle$;
		\item the support graph of $D_9$ is connected.
	\end{enumerate}
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
The functional $\mathbf{c}\mapsto \mathbf{c}\cdot \mathbf{p}$ is nonzero on $C$, since
	$\mathbf{w}\cdot \mathbf{p}=2$.
	Hence its kernel $D$ has dimension $11$, and
	$C=D\oplus\langle \mathbf{w}\rangle$. Moreover, $D\perp \mathbf{p}$, and since
	$\mathbf{q}=-\mathbf{w}-\mathbf{p}$ and $D\subseteq C=C^\perp$, also $D\perp \mathbf{q}$.
	This proves (i).

	Put $E=\langle D_9\rangle$. We first show that $E$ has full support. If
	$j\notin P$, inclusion--exclusion in the $5$-$(24,9,6)$ design shows that
	the number of blocks containing $j$ and avoiding $P$ is
	\[
		\lambda_1-4\lambda_2+6\lambda_3-4\lambda_4+\lambda_5
		=759-4\cdot264+6\cdot84-4\cdot24+6=117,
	\]
	by \eqref{eq:lambdas}.
	Every corresponding weight-$9$ word lies in $D_9$.
	If $j\in P$, let $\nu_j$ be the number of blocks containing the $6$-set
	$\{j\}\cup Q$. The number of blocks containing $j$ and avoiding $Q$ is
	\[
		\lambda_1-5\lambda_2+10\lambda_3-10\lambda_4+5\lambda_5-\nu_j
		=69-\nu_j\ge63,
	\]
	since $\nu_j\le\lambda_5=6$. For a word $\mathbf{c}$ on such a block,
	$\mathbf{c}\cdot \mathbf{w}=0$ and $\mathbf{c}$ vanishes on $Q$, whence $\mathbf{c}\cdot \mathbf{p}=0$.
	Thus every
	coordinate occurs in a word of $D_9$.

	We next establish a lower bound on $|D_9|$. Let
	$a=A_9(D)$ and $ B_j=A_j(D^\perp).$
	Since
	$
		D^\perp=C^\perp+\langle \mathbf{p}\rangle=C+\langle \mathbf{p}\rangle,
	$
	the only nonzero words of $D^\perp$ of weight at most $4$ are $\mathbf{p}$ and $-\mathbf{p}$.
	Indeed, if $\mathbf{c}\in C$ and $\alpha\ne0$ satisfy
	$\operatorname{wt}(\mathbf{c}+\alpha \mathbf{p})\le4$, then $\operatorname{wt}(\mathbf{c})\le8$, so $\mathbf{c}=0$.
	Hence
	\[
		B_0=1,
		\qquad B_1=B_2=B_3=0,
		\qquad B_4=2.
	\]
	Since $D\subset C$, all weights in $D$ are divisible by $3$, and its nonzero weights are at least
	$9$. Substitution in the standard identities \eqref{eq:macwilliams} for
	$j=0,1,2,3,4$ gives
	\[
		A_{12}(D)=27362-5a,~
		A_{15}(D)=67400+10a,~
		A_{18}(D)=79748-10a,\]
	\[
		A_{21}(D)=1240+5a,~
		A_{24}(D)=1396-a.
	\]
	The equation for $j=5$ then gives
	$ B_5=\frac{a-1360}{3}.$
	Consequently
	\begin{equation}\label{eq:D9lower}
		|D_9|=a\ge1360.
	\end{equation}

	It remains to show that $\dim E=11$.
	We use the following Delsarte-type linear-programming argument \cite{Delsarte}.
	Set
	\begin{equation}\label{eq:fi}
		F(i)=\frac{1532+404K_1(i)-53K_5(i)-11K_6(i)}{228906}.
	\end{equation}
	Direct evaluation from \eqref{eq:krawtchouk} gives
	\begin{equation}\label{eq:Fvalues}
		\begin{array}{c|rrrrrrr}
			i    & 0                    & 9 & 12 & 15 & 18 & 21           & 24                \\[0.3ex]
			\hline                                                                            \\[-1.5ex]
			F(i) & -\dfrac{114418}{157} & 1 & 0  & 0  & 0  & \dfrac1{314} & \dfrac{524}{157}.
		\end{array}
	\end{equation}
	Let $r=\dim E$ and write $B_j'=A_j(E^\perp)$. Since $E$ has full support, $B_1'=0$.
	Since the nonzero weights of $E$ belong to
	$\{9,12,15,18,21,24\}$, \eqref{eq:Fvalues} gives
	\[
		\sum_{i>0}A_i(E)F(i)
		=
		A_9(E)+\frac{1}{314}A_{21}(E)
		+\frac{524}{157}A_{24}(E)
		\ge A_9(E).
	\]
	On the other hand, by \eqref{eq:fi}, the MacWilliams identities \eqref{eq:macwilliams} yield
	\begin{align*}
		\sum_{i=0}^{24}A_i(E)F(i)
		=
		\frac{3^r}{228906}
		\bigl(1532B_0'+404B_1'-53B_5'-11B_6'\bigr)\le
		\frac{1532\cdot3^r}{228906},
	\end{align*}
	because $B_0'=1$, $B_1'=0$, and $B_5',B_6'\ge0$.
	Since $A_0(E)=1$ and
	$F(0)=-114418/157$, it follows that
	\[
		A_9(E)
		\le
		\sum_{i>0}A_i(E)F(i)
		\le
		\frac{1532\cdot3^r}{228906}
		+\frac{114418}{157}.
	\]
	Note that $A_9(E)=|D_9|\ge1360$ by \eqref{eq:D9lower}, and when $r\leq 10$, $A_9(E)<1360$.
	Thus
	$r\ge11$.
	Since $E\subseteq D$ and $\dim D=11$, we have $E=D$. This proves (ii).

	For (iii), if the support graph had $t\ge2$ components, generation would give
	$D=\bigoplus_{i=1}^t D_i$, with nonzero $[n_i,k_i,d_i]_3$ codes satisfying
	$d_i\ge9$. The Singleton bound gives
	\[
		13=24-11=\sum_i(n_i-k_i)\ge\sum_i(d_i-1)\ge8t\ge16,
	\]
	a contradiction.
	This completes the proof.

<!-- lemma-2: body_md -->
\label{lem:local}
	For $a,b\in\mathbb F_3$, put
	\(
	\mu(a,b)=\min\{\operatorname{wt}(\mathbf{d}+a\mathbf{q}+b\mathbf{p}):\mathbf{d}\in D\}.
	\)
	With rows and columns indexed by $0,1,2$,
	the following inequality holds entrywise:
	\begin{equation}\label{eq:local}
		(\mu(a,b))_{a,b\in\mathbb F_3}\ge
		\begin{pmatrix}
			0 & 4 & 4 \\
			5 & 9 & 6 \\
			5 & 6 & 9
		\end{pmatrix}
	\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let $\mathbf{u}=\mathbf{d}+a\mathbf{q}+b\mathbf{p}$ with $\mathbf{d}\in D$.
	Since $D$ is orthogonal to both $\mathbf{p}$ and $\mathbf{q}$, and $D$ is
	self-orthogonal,
	\begin{equation}\label{eq:weightcong}
		\operatorname{wt}(\mathbf{u})\equiv \mathbf{u}\cdot \mathbf{u}\equiv2a^2+b^2\pmod3.
	\end{equation}
	The first row and column follow from $\operatorname{wt}(\mathbf{p})=4$, $\operatorname{wt}(\mathbf{q})=5$,
	$d(C)=9$, and \eqref{eq:weightcong}. If $a=b\ne0$, then
	$a\mathbf{q}+b\mathbf{p}=-a\mathbf{w}$, so $\mathbf{d}-a\mathbf{w}$ is a nonzero word of $C$ and has weight at least $9$.
	If $a=-b\ne0$, scalar multiplication reduces to $(a,b)=(1,2)$. For
	$\mathbf{u}=\mathbf{d}+\mathbf{q}-\mathbf{p}$, the word $\mathbf{u}-\mathbf{p}=\mathbf{d}-\mathbf{w}$ lies nontrivially in $C$, so
	$\operatorname{wt}(\mathbf{u})\ge9-4=5$; by \eqref{eq:weightcong} its weight is divisible by $3$,
	and hence is at least $6$.

<!-- theorem-3: body_md -->
\label{thm:Cm}
	For every $m\ge1$, $C_m$ is an indecomposable ternary self-dual code with
	parameters $[24m,12m,9]$.
	It is generated by weight-$9$ words whose support
	graph is connected.
<!-- end theorem-3 -->
<!-- proof-3: body_md -->
The code $D^{\oplus m}$ is self-orthogonal and is orthogonal to every bridge.
	Moreover,
	$ \mathbf{x}_i\cdot \mathbf{x}_i=0$,
	and distinct bridges are orthogonal.
	Hence $C_m$ is self-orthogonal.

	If $\sum_i\alpha_i \mathbf{x}_i\in D^{\oplus m}$, then its block-$i$ component is
	$\alpha_i \mathbf{p}+\alpha_{i-1}\mathbf{q}$.
	Taking its inner product with $\mathbf{p}$ gives
	$\alpha_i=0$.
	Thus the $m$ bridges are independent and
	$X_m\cap D^{\oplus m}=0$.
	Therefore
	\(
	\dim C_m=11m+m=12m,
	\)
	so $C_m$ is self-dual.

	Write a word of $C_m$ as
	\(
	\mathbf{z}=(\mathbf{d}_i)_{i\in\mathbb Z/m\mathbb Z}+\sum_i\alpha_i \mathbf{x}_i.
	\)
	Its block-$i$ component is
	\(
	\mathbf{z}_i=\mathbf{d}_i+\alpha_{i-1}\mathbf{q}+\alpha_i \mathbf{p}.
	\)
	The weight of $\mathbf{z}$ satisfies
	$\operatorname{wt}(\mathbf z)=\sum_i\operatorname{wt}(\mathbf z_i)\geq\sum_i\mu(\alpha_{i-1},\alpha_i).$
	Therefore, the matrix \eqref{eq:local} assigns a lower-bound cost to each transition
	$\alpha_{i-1}\to\alpha_i$ in the cyclic state sequence. If all states are
	zero, every nonzero word has weight at least $9$. If both zero and nonzero
	states occur, every maximal nonzero run has entrance cost at least $4$ and
	exit cost at least $5$. If all states are nonzero, either they are all equal,
	in which case a transition already costs at least $9$, or both $1$ and $2$
	occur, in which case the cycle has at least two transitions of cost at least
	$6$. Hence $d(C_m)\ge9$. Each bridge has weight $9$, so $d(C_m)=9$.

	By Theorem~\ref{thm:seed}, the block-local copies of $D_9$, together with the
	bridges, generate $C_m$; all have weight $9$. Within each block their support
	graph is connected, and every bridge joins two consecutive blocks. Thus the
	full support graph is connected.

	Finally, suppose that $C_m$ decomposes over a nontrivial coordinate partition
	as $U\oplus V$. Since $d(C_m)=9$, a weight-$9$ word cannot have nonzero
	projections in both summands.
	This contradicts the connectedness of the support graph.
	Thus $C_m$ is indecomposable.

<!-- proof-4: body_md -->
The Construction~A properties recalled in Section~2 show that $L_m$
	is unimodular of rank $24m$ and minimum norm $3$.
	Lemma~\ref{lem:HM} gives irreducibility, and a vector of norm $3$
	shows that $L_m$ is odd.

	The vectors $\sqrt3\mathbf{e}_j$ are minimal and generate $\sqrt3\mathbb Z^{24m}$.
	For every weight-$9$ word $\mathbf{c}\in C_m$, its canonical lift
	$\widetilde \mathbf{c}/\sqrt3$ is also minimal. Since the weight-$9$ words generate
	$C_m$ and
	$ L_m/\sqrt3\mathbb Z^{24m}\cong C_m,$
	these minimal vectors generate $L_m$.
	This completes the proof.

<!-- question-1: body_md -->
\label{ques:bounded-integrability}
	Does there exist a positive integer $t$ such that every integral
	lattice generated by vectors of norm $3$ is $t$-integrable?
<!-- end question-1 -->
<!-- proof-5: body_md -->
Choose one representative from each antipodal pair
	$\{\mathbf{v},-\mathbf{v}\}\subseteq\operatorname{Min}(L_m)$, and denote the resulting set by $S_m$.
	For distinct $\mathbf{u},\mathbf{v}\in S_m$, the vectors $\mathbf{u}\pm\mathbf{v}$ are nonzero, so
	\(
	3\le (\mathbf{u}\pm\mathbf{v})\cdot(\mathbf{u}\pm\mathbf{v})
	=6\pm2\mathbf{u}\cdot\mathbf{v}.
	\)
	Integrality therefore gives $\mathbf{u}\cdot\mathbf{v}\in\{0,\pm1\}$.
	Define a signed graph $\Gamma_m$ on $S_m$ by joining distinct vectors
	when their inner product is nonzero, with this inner product as the edge sign.
	Writing $A_m=A(\Gamma_m)$, we have
	\(
	3I+A_m=\operatorname{Gram}(S_m)\succeq0\).
	Since $S_m$ generates $L_m$, \( \operatorname{rank}(3I+A_m)=24m.\)
	There are $24m$ antipodal pairs of coordinate minimal vectors
	$\pm\sqrt3\mathbf{e}_j$, and at least one further pair arising from a
	weight-$9$ word of $C_m$. Hence $|S_m|>24m$, so $3I+A_m$ is singular
	and $\lambda_{\min}(\Gamma_m)=-3$.
	Different choices of representatives only switch $\Gamma_m$.
	Moreover, if $\Gamma_m$ were disconnected, the integer spans of its
	components would give a nontrivial orthogonal decomposition of $L_m$.
	Thus $\Gamma_m$ is connected.

	Suppose that a connected signed graph $\Gamma'$ properly contains
	$\Gamma_m$ as an induced signed subgraph and has smallest eigenvalue
	at least $-3$. Choose a Gram representation of $3I+A(\Gamma')$.
	Identify the span of the old Gram vectors isometrically with
	$V=\operatorname{span}_{\mathbb R}L_m$, so that the old vectors are
	precisely those of $S_m$.
	Since $\Gamma'$ is connected, some new vertex is adjacent to an old
	vertex. Let $\mathbf{y}$ be its Gram vector and let $\mathbf{z}$ be its orthogonal
	projection onto $V$. Then $\mathbf{z}\ne0$ and
	$\mathbf{z}\cdot\mathbf{v}=\mathbf{y}\cdot\mathbf{v}\in\{0,\pm1\}$ for every $\mathbf{v}\in S_m$.
	As $S_m$ generates $L_m$, we have $\mathbf{z}\in L_m^*=L_m$.
	The minimum norm and the projection inequality give
	\[
		3\le\mathbf{z}\cdot\mathbf{z}\le\mathbf{y}\cdot\mathbf{y}=3.
	\]
	Equality forces $\mathbf{y}=\mathbf{z}\in\operatorname{Min}(L_m)$, and hence $\mathbf{y}=\pm\mathbf{v}$ for
	some $\mathbf{v}\in S_m$. This gives the impossible off-diagonal Gram entry
	$\mathbf{y}\cdot\mathbf{v}=\pm3$.
	Therefore $\Gamma_m$ is non-extendable.

	Finally, $\operatorname{rank}(A_m+3I)=24m$ is invariant under switching and vertex
	relabelling. Different values of $m$ therefore give pairwise distinct
	switching classes.
