\label{thm:intro-cleft}{\rm(Theorem~\ref{thm:cleft-transfer})}
Let $(\mathcal{B},\mathcal{A},i,e,l)$ be a cleft extension of length categories with finitely many isomorphism classes of simple objects and enough projective objects. Assume that $l$ is exact and that $e$ preserves projective objects. Set
\[
L_e
=
\max\bigl\{
\ell_{\mathcal{B}}(eS)
\mid
S\in\operatorname{simp}\mathcal{A}
\bigr\},
\]
where $\ell_{\mathcal{B}}(eS)$ denotes the composition length of an object
$eS$ in $\mathcal{B}$. 
\begin{enumerate}[(a)]
\item 
Suppose that $e$ admits an exact right adjoint $r$ and  $c_r:=\sup\{\operatorname{pd}_{\mathcal{A}}r(P)\mid P\in\operatorname{Proj}\mathcal{B}\}<\infty$. Then
\[
\operatorname{ddell}(\mathcal{B})
\leq
\operatorname{ddell}(\mathcal{A})
\leq
\operatorname{pd}(r)+L_e\bigl(\operatorname{ddell}(\mathcal{B})+1\bigr).
\]
Moreover, if $F^n=0$ for some $n\geq1$ and $\operatorname{pd}(r)<\infty$, then, for every $k\geq1$,
\[
k\text{-}\operatorname{ddell}(\mathcal{A})<\infty
\quad\Longleftrightarrow\quad
k\text{-}\operatorname{ddell}(\mathcal{B})<\infty.
\]

\item Suppose that
$
\operatorname{Im}G\subseteq\operatorname{Proj}\mathcal{A},
$ Then, for every $k\geq1$,
\[
k\text{-}\operatorname{ddell}(\mathcal{B})
\leq
k\text{-}\operatorname{ddell}(\mathcal{A})
\leq
\max\bigl\{
1,\,
L_e\bigl(k\text{-}\operatorname{ddell}(\mathcal{B})+1\bigr)-1
\bigr\}.
\]
In particular, $k\text{-}\operatorname{ddell}(\mathcal{A})<\infty
\quad\Longleftrightarrow\quad
k\text{-}\operatorname{ddell}(\mathcal{B})<\infty.$
\end{enumerate}

\label{thm:intro-arrow-removal}
Let $Q$ be a finite quiver and let
$
\Lambda=\Bbbk Q/I
$
be an admissible quotient of the path algebra $\Bbbk Q$ over a field $\Bbbk$. Suppose that
\[
a_i:v_{e_i}\longrightarrow v_{f_i},
\qquad
i=1,\ldots,t,
\]
are arrows of $Q$ which do not occur in a set of minimal generators of $I$ and satisfy
\[
\operatorname{Hom}_{\Lambda}(e_i\Lambda,f_j\Lambda)=0
\qquad
\text{for all }1\leq i,j\leq t.
\]
Set
$
\Gamma
=
\Lambda/
\Lambda\{\overline a_1,\ldots,\overline a_t\}\Lambda.
$
Then, for every $k\geq1$,
\[
k\text{-}\operatorname{ddell}\Gamma
\leq
k\text{-}\operatorname{ddell}\Lambda
\leq
\max\bigl\{
1,\,
k\text{-}\operatorname{ddell}\Gamma
\bigr\}.
\]

(a) Suppose that $c_e:=\sup\{\operatorname{pd}_{\mathcal{C}}e(P)\mid P\in\operatorname{Proj}\mathcal{B}\}<\infty$,   If  $k\text{-}\mathrm{ddell}\mathcal{B}<\infty$, then   $k\text{-}\mathrm{ddell} \mathcal{C}<\infty$.\\

<!-- lemma-1: body_md -->
\label{lem:schanuel}
Let $X\in\mathcal{A}$, and let
\[
0\longrightarrow K\longrightarrow P\longrightarrow X
\longrightarrow0
\]
and
\[
0\longrightarrow K'\longrightarrow P'\longrightarrow X
\longrightarrow0
\]
be exact sequences with $P,P'\in\operatorname{Proj}\mathcal{A}$. Then
\[
K\oplus P'
\cong
K'\oplus P.
\]
More generally, any two $n$-syzygy objects of $X$ become isomorphic
after adding suitable projective direct summands. In particular, they
are isomorphic in $\underline{\mathcal{A}}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $W=P\times_XP'$ be the pullback. Thus there is a commutative square
\[
\begin{CD}
&&&&0&&0\\
&&&&@VVV@VVV\\
&&&&K@=K\\
&&&&@VVV@VVV\\
0@>>>K'@>>>W@>>>P'@>>>0\\
&&@|@VVV@VVV\\
0@>>>K'@>>>P@>>>X@>>>0\\
&&&&@VVV@VVV\\
&&&&0&&0
\end{CD}
\]
Since $P$ and $P'$ are projective, \[
W\simeq K'\oplus P\simeq K\oplus P',
\]
which proves the first assertion. For the higher assertion, compare the last projective deflations in two $n$-step projective resolutions of $X$, apply the first assertion, and continue inductively to the kernels. At the $n$-th step one obtains an isomorphism after adjoining finite direct sums of the projective objects occurring in the two resolutions.

<!-- lemma-2: body_md -->
\label{lem:rotation}(syzygy shifting)
Let
\[
0\longrightarrow X\xrightarrow{u}Y\xrightarrow{v}Z\longrightarrow0
\]
be exact. If $P_Z\twoheadrightarrow Z$ is a projective deflation with kernel $\Omega Z$, then there is an exact sequence
\[
0\longrightarrow\Omega Z\longrightarrow X\oplus P_Z
\longrightarrow Y\longrightarrow0.
\]
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Form the pullback $W=Y\times_ZP_Z$. It is contained in a pullback diagram
\[
\begin{CD}
0@>>>X@>>>W@>>>P_Z@>>>0\\
@.@|@VVV@VVV\\
0@>>>X@>{u}>>Y@>{v}>>Z@>>>0.
\end{CD}
\]
The upper row splits because $P_Z$ is projective, and therefore $W\simeq X\oplus P_Z$. The other pullback projection fits into an exact sequence
\[
0\longrightarrow\Omega Z\longrightarrow W\longrightarrow Y\longrightarrow0.
\]
After replacing $W$ by $X\oplus P_Z$, this is the required sequence.

<!-- lemma-3: body_md -->
\label{lem:iterated-rotation}
Let
\[
0\longrightarrow M_t\longrightarrow M_{t-1}\longrightarrow\cdots
\longrightarrow M_1\longrightarrow M_0\longrightarrow0
\]
be exact and let $s\geq1$. There exist projective objects $Q_1,\ldots,Q_{t-1}$ and an exact sequence
\[
0\longrightarrow\Omega^sM_t\longrightarrow
\Omega^sM_{t-1}\oplus Q_{t-1}\longrightarrow\cdots \Omega^sM_1\oplus Q_{1}
\longrightarrow\Omega^sM_0\longrightarrow0.
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Decompose the given sequence into short exact sequences
\[
0\to K_{j+1}\to M_j\to K_j\to0
\qquad(0\leq j<t),
\]
where $K_0=0$ and $K_t=M_t$. Applying  Horseshoe Lemma to the above short exact sequences, we get the following exact sequences
\[
0\to \Omega^s K_{j+1}\to \Omega^s M_j\oplus Q_{j} \to \Omega^s K_j\to0
\qquad(0\leq j<t).
\]
Splicing the resulting exact sequences proves the assertion.

<!-- definition-2: body_md -->
\label{def:abelian-dell-ddell}
{\rm\cite[Definitions~2.5 and 2.22]{GuoIgusa2025}}
Let $X\in\mathcal{A}$ and $k\geq1$.
\begin{enumerate}[$(a)$]
\item The $k$-delooping level of $X$ is
\[
k\text{-}\operatorname{dell}_{\mathcal{A}}X
=
\inf\{n\geq0\mid
\Omega_{\mathcal{A}}^{n}X\text{ is a stable retract of }
\Omega_{\mathcal{A}}^{n+k}Y
\text{ for some }Y\in\underline{\mathcal{A}}\}.
\]
\item The $k$-derived delooping level of $X$ is
\[
\begin{split}
k\text{-}\operatorname{ddell}_{\mathcal{A}}X=\inf\{m\geq0\mid{}&
\text{there exist }0\leq t\leq m\text{ and an exact sequence}\\
&0\to C_t\to C_{t-1}\to\cdots\to C_0\to X\to0,\\
&(i+k)\text{-}\operatorname{dell}_{\mathcal{A}}C_i\leq m-i
\text{ for }0\leq i\leq t\}.
\end{split}
\]
\end{enumerate}
For $k=1$ we write $\operatorname{dell}_{\mathcal{A}}X$ and $\operatorname{ddell}_{\mathcal{A}}X$.

\[
k\text{-}\operatorname{ddell}_{\mathcal{A}}X\leq k\text{-}\operatorname{dell}_{\mathcal{A}}X.
\]
Moreover, every first syzygy has ordinary delooping level zero.

<!-- lemma-4: body_md -->
\label{lem:stable-projective-invariance}
Let $X,Y\in\mathcal{A}$ and  $k\geq1$.
\begin{enumerate}[$(a)$]
\item If $X$ is a stable retract of $Y$, then
\[
k\text{-}\operatorname{dell}_{\mathcal{A}}X\leq k\text{-}\operatorname{dell}_{\mathcal{A}}Y.
\]
\item If $X\oplus P\simeq Y\oplus Q$ for projective objects $P,Q$, then
\[
k\text{-}\operatorname{dell}_{\mathcal{A}}X=k\text{-}\operatorname{dell}_{\mathcal{A}}Y
\quad\text{and}\quad
k\text{-}\operatorname{ddell}_{\mathcal{A}}X=k\text{-}\operatorname{ddell}_{\mathcal{A}}Y.
\]

\end{enumerate}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Part~(a) follows directly from the definition of the $k$-delooping
level in the stable category. It also follows that adding a projective
summand does not change the $k$-delooping level. Hence the first
equality in Part~(b) is immediate.

It remains to prove the assertion for the $k$-derived delooping level.
It is enough to show that, for every projective object $P$, $
k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P)
=
k\text{-}\operatorname{ddell}_{\mathcal{A}}X.
$ Put $
m=k\text{-}\operatorname{ddell}_{\mathcal{A}}X$ 
and suppose first that $m<\infty$. By definition, there is an integer
$t\leq m$ and an exact sequence
\[
0\longrightarrow C_t
\longrightarrow\cdots
\longrightarrow C_0
\xrightarrow{d_0}X
\longrightarrow0
\]
such that $(i+k)\text{-}\operatorname{dell}_{\mathcal{A}}C_i
\leq m-i$ $(0\leq i\leq t).$ 
Then
\[
0\longrightarrow C_t
\longrightarrow\cdots
\longrightarrow C_0\oplus P
\xrightarrow{d_0\oplus1_P}
X\oplus P
\longrightarrow0
\]
is exact. Since $C_0$ and $C_0\oplus P$ are isomorphic in the stable
category, $k\text{-}\operatorname{dell}_{\mathcal{A}}(C_0\oplus P)
=
k\text{-}\operatorname{dell}_{\mathcal{A}}C_0.$
Thus the preceding sequence satisfies the required delooping-level
bounds and gives
\[
k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P)
\leq
k\text{-}\operatorname{ddell}_{\mathcal{A}}X.
\]

Conversely, put $n=k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P)$ 
and suppose that $n<\infty$. Choose an integer $s\leq n$ and an exact
sequence
\[
0\longrightarrow D_s
\longrightarrow\cdots
\longrightarrow D_0
\xrightarrow{f}
X\oplus P
\longrightarrow0
\]
such that $(i+k)\text{-}\operatorname{dell}_{\mathcal{A}}D_i
\leq n-i
\qquad
(0\leq i\leq s).$
Let $D'_0$ be the pullback of $f:D_0\to X\oplus P$ along the canonical
inclusion $X\to X\oplus P$. We obtain a commutative diagram with exact
rows and columns:
\[
\begin{CD}
&&&&0&&0\\
&&&&@VVV@VVV\\
0@>>>\operatorname{Ker} f@>>>D'_0@>>>X@>>>0\\
&&@|@VVV@VVV\\
0@>>>\operatorname{Ker} f@>>>D_0@>>>X\oplus P@>>>0\\
&&&&@VVV@VVV\\
&&&&P@=P\\
&&&&@VVV@VVV\\
&&&&0&&0.
\end{CD}
\]
Since $P$ is projective, the exact sequence
\[
0\longrightarrow D'_0
\longrightarrow D_0
\longrightarrow P
\longrightarrow0
\]
splits. Hence $D_0\cong D'_0\oplus P.$ 
By the already established invariance of $k$-delooping levels under
projective summands, $k\text{-}\operatorname{dell}_{\mathcal{A}}D'_0
=
k\text{-}\operatorname{dell}_{\mathcal{A}}D_0.$ 
Splicing the upper row with the preceding part of the original exact
sequence gives
\[
0\longrightarrow D_s
\longrightarrow\cdots
\longrightarrow D_1
\longrightarrow D'_0
\longrightarrow X
\longrightarrow0.
\]
All the required delooping-level bounds remain unchanged. Therefore
\[
k\text{-}\operatorname{ddell}_{\mathcal{A}}X
\leq n
=
k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P).
\]
Combining the two inequalities yields
\[
k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P)
=
k\text{-}\operatorname{ddell}_{\mathcal{A}}X.
\]

Finally, if $X\oplus P\cong Y\oplus Q$ with $P$ and $Q$ projective,
then
\[
\begin{aligned}
k\text{-}\operatorname{ddell}_{\mathcal{A}}X
&=
k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P)\\
&=
k\text{-}\operatorname{ddell}_{\mathcal{A}}(Y\oplus Q)\\
&=
k\text{-}\operatorname{ddell}_{\mathcal{A}}Y.
\end{aligned}
\]
This proves Part~(b).

<!-- lemma-5: body_md -->
\label{lem:ddell-syzygy}
Let $\mathcal A$ be an abelian category with enough projective
objects. For every $M\in\mathcal A$ and every $k\geq1$,
\[
k\text{-}\operatorname{ddell}_{\mathcal A}(\Omega M)
\leq
k\text{-}\operatorname{ddell}_{\mathcal A}M.
\]
Consequently, $k\text{-}\operatorname{ddell}_{\mathcal A}M<\infty
\Longrightarrow
k\text{-}\operatorname{ddell}_{\mathcal A}(\Omega M)<\infty.$
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Put $m=k\text{-}\operatorname{ddell}_{\mathcal A}M<\infty.$ 
Choose an $m$-witness
\[
0\to C_n\to C_{n-1}\to\cdots\to C_0\to M\to0,
\qquad n\leq m,
\]
such that $(i+k)\text{-}\operatorname{dell}_{\mathcal A}C_i\leq m-i$  $(0\leq i\leq n).$
By Lemma \ref{lem:iterated-rotation}, taking 1-syzygies 
gives an exact sequence
\[
0\to\Omega C_n
\to\Omega C_{n-1}\oplus P_{n-1}
\to\cdots
\to\Omega C_0\oplus P_0
\to\Omega M\to0,
\]
where all $P_i$ are projective.

For every object $X$ and every $q\geq1$, one has $q\text{-}\operatorname{dell}_{\mathcal A}(\Omega X)
\leq q\text{-}\operatorname{dell}_{\mathcal A}X.$ 
Since adding a projective summand does not change the delooping
level, it follows that
\[
\begin{aligned}
(i+k)\text{-}\operatorname{dell}_{\mathcal A}
   (\Omega C_i\oplus P_i)
&=(i+k)\text{-}\operatorname{dell}_{\mathcal A}(\Omega C_i)\\
&\leq(i+k)\text{-}\operatorname{dell}_{\mathcal A}C_i\\
&\leq m-i.
\end{aligned}
\]
Thus the displayed exact sequence is an $m$-witness for
$\Omega M$, and hence
\[
k\text{-}\operatorname{ddell}_{\mathcal A}(\Omega M)\leq m.
\]

<!-- proposition-1: body_md -->
\label{prop:extension-estimate}
Let $0\to X\to Y\to Z\to0$ be exact. Then, for every $k\geq1$,
\[
k\text{-}\operatorname{ddell}_{\mathcal{A}}Y
\leq k\text{-}\operatorname{ddell}_{\mathcal{A}}X+k\text{-}\operatorname{ddell}_{\mathcal{A}}Z+1,
\]
and
\[
k\text{-}\operatorname{ddell}_{\mathcal{A}}X
\leq k\text{-}\operatorname{ddell}_{\mathcal{A}}Y+k\text{-}\operatorname{ddell}_{\mathcal{A}}Z+1.
\]
<!-- end proposition-1 -->
<!-- proof-6: body_md -->
There is nothing to prove if one of the two terms on the right is infinite. Put
\[
m_1=k\text{-}\operatorname{ddell} X,
\qquad
m_2=k\text{-}\operatorname{ddell} Z.
\]
Choose witnesses of lengths $p\leq m_1$ and $q\leq m_2$:
\begin{equation}\label{eq:X}
0\to D_p\xrightarrow{d_p}\cdots\to D_0\xrightarrow{d_0} X\to0,
\qquad
(i+k)\text{-}\operatorname{dell} D_i\leq m_1-i,
\end{equation}
\begin{equation}\label{eq:Z}
0\to E_q\to\cdots\to E_0\to Z\to0,
\qquad
(j+k)\text{-}\operatorname{dell} E_j\leq m_2-j.
\end{equation}
We also choose the first $p+1$ steps of a projective resolution of $Z$
\[
0 \to \Omega^{p+1}Z \xrightarrow{\pi}P_{p} \xrightarrow{\pi_{p}} \cdots \to P_1 \xrightarrow{\pi_1} P_0 \xrightarrow{\pi_0} Z \to 0,
\]
and another exact sequence by taking($p+1$)-syzygy of  \eqref{eq:Z}
\[
0 \to \Omega^{p+1}E_{q} \to \Omega^{p+1}E_{q-1}\oplus Q_{q-1} \to \cdots \to \Omega^{p+1}E_0\oplus Q_0  \xrightarrow{\partial}  \Omega^{p+1}Z \to 0
\]
where all $Q_i$ are projective objects. Now we prove that there is a  long exact sequence
\begin{equation}\label{eq:splicing}
\begin{split}
0\to&\Omega^{p+1}E_q\to \Omega^{p+1}E_{q-1}\oplus Q_{q-1}\to\cdots
\to\Omega^{p+1}E_0\oplus Q_0\\
& \xrightarrow{f}  D_p\oplus P_p\to\cdots\to D_0\oplus P_0\to Y\to0.
\end{split}
\end{equation}
where $f$ maps into $P_{p}$, which can be factored as $\Omega^{p+1}E_0\oplus Q_{0} \stackrel{\partial} \to \Omega^{p+1}Z \stackrel{\pi} \hookrightarrow P_{p}$. In fact, we may proceed by induction for the proof. Consider the following commutative exact diagram
$$\begin{tikzcd}
& && P_0 \arrow[d, "\pi_0"] \arrow[ld, dashed, "\gamma"]  &\\
0 \arrow[r] & X \arrow[r, "\alpha"] & Y \arrow[r, "\beta"] & Z \arrow[r] & 0,
\end{tikzcd}$$
Since $P_0$ is projective, such a lift $\gamma$ exists. We therefore have
the following commutative exact diagram
$$\begin{tikzcd}
&0\arrow[d]&0\arrow[d]&0\arrow[d]&\\
0\arrow[r]&K_1\arrow[r]\arrow[d]&\text{Ker}( \alpha d_0 \ \gamma)\arrow[r]\arrow[d]&\Omega Z\arrow[r]\arrow[d]&0\\
0\arrow[r]&D_0\arrow[r,"\begin{pmatrix}
   1\\
  0
  \end{pmatrix}"]\arrow[d, "d_0"] &D_0\oplus P_0\arrow[r,"(0\ 1 )"]\arrow[d,"(\alpha d_0 \ \gamma)"] & P_0 \arrow[d, "\pi_0"] \arrow[r]  &0.\\
0 \arrow[r] & X \arrow[r, "\alpha"] \arrow[d]& Y \arrow[r, "\beta"] \arrow[d]& Z \arrow[r]\arrow[d] & 0\\
&0&0&0&
\end{tikzcd}
$$
where the surjectivity of $( \alpha d_0 \ \gamma)$ follows from the Snake
Lemma. Repeating this pullback--kernel construction with the successive
maps $D_i\to D_{i-1}$ and $P_i\to P_{i-1}$ gives, by induction on $i$, the
exact sequence \eqref{eq:splicing}. The terms $D_i\oplus P_i$ occur in
positions $i$, and the term $\Omega^{p+1}E_j\oplus Q_j$ occurs in position
$p+1+j$.
So, for $k$-ddell $Y \leq m_1 + m_2 + 1$, it remains to check
\noindent\begin{enumerate}
    \item $(i+k)$-$\operatorname{dell} (D_i\oplus P_i) \leq m_1 + m_2 + 1 - i$ for $i = 0, \dots, p$,
    \item $(j+ p + 1 + k)$-$\operatorname{dell}  (\Omega^{p+1}E_j\oplus Q_j)\leq m_1 + m_2 - j - p$ for $j= 0, \dots, q$.
\end{enumerate}
Due to the conditions in \eqref{eq:X}, \[
(i+k)\text{-}\operatorname{dell}(D_i\oplus P_i)
\leq m_1-i\leq m_1+m_2+1-i.
\]
For the second statement,  choose $a_j\leq m_2-j$, an object $N_j$, and a stable retraction
\[
\Omega^{a_j}E_j\longrightarrow
\Omega^{a_j+j+k}N_j\longrightarrow
\Omega^{a_j}E_j.
\]
Since $a_j\leq m_2-j$, the integer $s_j=m_1+m_2-j+1-a_j$ 
is nonnegative. Applying $\Omega^{s_j}$ to this retraction gives
\[
\Omega^{m_1+m_2-j+1}E_j
\quad\text{as a stable retract of}\quad
\Omega^{m_1+m_2+k+1}N_j.
\]
Rewriting the two exponents gives
\[
\Omega^{m_1+m_2-p-j}(\Omega^{p+1}E_j)
\quad\text{as a stable retract of}\quad
\Omega^{m_1+m_2-p-j+p+1+j+k}N_j.
\]
The required inequality follows from Lemma~\ref{lem:stable-projective-invariance}. The exact sequence \ref{eq:splicing} therefore witnesses
\[
k\text{-}\operatorname{ddell} Y\leq m_1+m_2+1.
\]

It remains to  prove the second inequality.   By the syzygy-shifting lemma, there exists a projective
object $P_Z$ and an exact sequence
\[
0\longrightarrow\Omega Z
\longrightarrow X\oplus P_Z
\longrightarrow Y
\longrightarrow0.
\]
Applying the first inequality to this exact sequence yields
\[
k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P_Z)
\leq
k\text{-}\operatorname{ddell}_{\mathcal{A}}(\Omega Z)
+k\text{-}\operatorname{ddell}_{\mathcal{A}}Y+1.
\]
Hence, 
\[
\begin{aligned}
k\text{-}\operatorname{ddell}_{\mathcal{A}}X
&=
k\text{-}\operatorname{ddell}_{\mathcal{A}}(X\oplus P_Z)\\
&\leq
k\text{-}\operatorname{ddell}_{\mathcal{A}}(\Omega Z)
+k\text{-}\operatorname{ddell}_{\mathcal{A}}Y+1\\
&\leq
k\text{-}\operatorname{ddell}_{\mathcal{A}}Z
+k\text{-}\operatorname{ddell}_{\mathcal{A}}Y+1.
\end{aligned}
\]
as desired.

<!-- lemma-7: body_md -->
\label{L-projective-kernel-ddell}
Let
\[
0\longrightarrow P\xrightarrow{j}M\xrightarrow{\pi}N
\longrightarrow0
\]
be a short exact sequence, where $P$ is projective. Then, for every
$k\geq1$,
\[
k\text{-}\operatorname{ddell}N
\leq
\max\{1,k\text{-}\operatorname{ddell}M\}.
\]
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
Set
$
m=k\text{-}\operatorname{ddell}M<\infty.
$ We first consider the case $m=0$.  Using the given short exact sequence
\[
0\longrightarrow P\longrightarrow M\longrightarrow N
\longrightarrow0,
\]
we have $k\text{-}\operatorname{dell}M=0\leq1$ and $(k+1)\text{-}\operatorname{dell}P=0.$
Hence $k\text{-}\operatorname{ddell}N\leq1.$

\[
0\longrightarrow C_n\longrightarrow C_{n-1}
\longrightarrow\cdots\longrightarrow C_1
\xrightarrow{d_1}C_0\xrightarrow{d_0}M
\longrightarrow0
\]
with $n\leq m$ such that
$
(i+k)\text{-}\operatorname{dell}C_i
\leq m-i$ $(0\leq i\leq n).
$

If $n=0$, then $C_0\simeq M$ and $k\text{-}\operatorname{dell}M\leq m.$ The given short exact sequence
\[
0\longrightarrow P\longrightarrow M\longrightarrow N
\longrightarrow0
\]
therefore satisfies $k\text{-}\operatorname{dell}M\leq m$ and $
(k+1)\text{-}\operatorname{dell}P=0\leq m-1.$ Thus $k\text{-}\operatorname{ddell}N\leq m.$

\[
\xymatrix{
&0\ar[d]&0\ar[d]&&\\
&\ker p \ar[d]\ar[r]^{\cong}
    & \ker d_0\ar[d]&&\\
0\ar[r]

 Since the original sequence is exact at

\xrightarrow{u}C_0,

\overline d_1&0\\
0& 1_P
\end{pmatrix}:C_1\oplus P\to \ker d_0\oplus P.

Consequently, $\alpha$ is an epimorphism and
$

0\longrightarrow \ker d_1 \to C_1\oplus P\to \ker d_0\oplus P
\to 0.\end{equation}

\to C_2

\to N

is exact.

For all other terms, the original inequalities remain unchanged:

Therefore

\ \text{and}\ 
(F')^n=0\quad\Longleftrightarrow\quad (G')^n=0.
\]
It therefore remains to relate $F^n$ and $(F')^n$. By
\cite[Remark~2.4]{MaZhengLiu2026}, the pair $(F,F')$ is an adjoint
pair. Consequently, $(F^n,(F')^n)$ is also an adjoint pair. Hence, for

\operatorname{Hom}_{\mathcal{B}}(F^nX,Y)

\operatorname{Hom}_{\mathcal{B}}(X,(F')^nY).

\[
F^n=0\quad\Longleftrightarrow\quad(F')^n=0,

which completes the proof.

<!-- lemma-11: body_md -->
\label{lem:factor-finite-pd}
Let $\mathcal{C}$ be an abelian category with enough projective objects, and let
$f\colon X\to Y$ be a morphism in $\mathcal{C}$ that factors through an object
$E$ with
$\operatorname{pd} E\leq c<\infty$. Then $\Omega^c f$ factors through a projective object. Consequently, if $T:\mathcal{A}\to\mathcal{B}$ is exact and $\operatorname{pd}(T)\leq c$,  then the assignment
$X\longmapsto\Omega_{\mathcal{B}}^{c}T(X)$ 
induces an additive functor
\[
\underline{T}_c:\underline{\mathcal{A}}\longrightarrow\underline{\mathcal{B}}.
\]
<!-- end lemma-11 -->
<!-- proof-11: body_md -->
Write $f=ba$ with $a:X\to E$ and $b:E\to Y$. Choosing projective resolutions,  we can lift $a$ and $b$ to chain maps by the  comparison theorem for projective resolutions. The induced morphism on the $c$-th syzygies factors through $\Omega^cE$, which is projective  or zero because $\operatorname{pd} E\leq c$. Hence $\Omega^cf$ is zero in the stable category.

If $f$ and $g$ represent the same morphism in $\underline{\mathcal{A}}$, then $f-g$ factors through a projective $P$. Thus $T(f-g)$ factors through $T(P)$, whose projective dimension is at most $c$, and the first assertion shows that $\Omega^cT(f)$ and $\Omega^cT(g)$ agree in $\underline{\mathcal{B}}$. Additivity and compatibility with composition follow from the comparison theorem for projective resolutions.

<!-- proposition-3: body_md -->
\label{thm:exact-functor-transfer}
Let $T\colon\mathcal{A}\to\mathcal{B}$ be an exact functor between abelian categories
with enough projective objects. Then, for every $X\in\mathcal{A}$ and every
$k\geq1$,
\[
k\text{-}\operatorname{dell}_{\mathcal{B}}(T X)
\leq
\operatorname{pd}(T)+k\text{-}\operatorname{dell}_{\mathcal{A}}X
\]
and
\[
k\text{-}\operatorname{ddell}_{\mathcal{B}}(T X)
\leq
\operatorname{pd}(T)+k\text{-}\operatorname{ddell}_{\mathcal{A}}X.
\]
<!-- end proposition-3 -->
<!-- proof-13: body_md -->
If $\operatorname{pd}(T)=\infty$, both inequalities are immediate. We may therefore
assume that $
\operatorname{pd}(T)=c<\infty.$ Suppose first that  $k\text{-}\operatorname{dell}_{\mathcal{A}}X=m<\infty.$ 
By definition, there exist an object $Y\in\mathcal{A}$ and morphisms in
$\underline{\mathcal{A}}$ exhibiting a stable retraction
\[
\Omega_{\mathcal{A}}^{m}X
\longrightarrow
\Omega_{\mathcal{A}}^{m+k}Y
\longrightarrow
\Omega_{\mathcal{A}}^{m}X.
\]
Applying the functor $\underline{T}_c$ from
Lemma~\ref{lem:factor-finite-pd} gives a stable retraction
\[
\Omega_{\mathcal{B}}^{c}T(\Omega_{\mathcal{A}}^{m}X)
\longrightarrow
\Omega_{\mathcal{B}}^{c}T(\Omega_{\mathcal{A}}^{m+k}Y)
\longrightarrow
\Omega_{\mathcal{B}}^{c}T(\Omega_{\mathcal{A}}^{m}X).
\]
By Proposition~\ref{prop:stable-syzygy-comparison}, this stable
retraction may be identified with
\[
\Omega_{\mathcal{B}}^{c+m}T(X)
\longrightarrow
\Omega_{\mathcal{B}}^{c+m+k}T(Y)
\longrightarrow
\Omega_{\mathcal{B}}^{c+m}T(X).
\]
Hence
\[
k\text{-}\operatorname{dell}_{\mathcal{B}}(T X)\leq c+m.
\]

We now prove the second inequality. Suppose that  $k\text{-}\operatorname{ddell}_{\mathcal{A}}X=m<\infty.$
Then there is an exact sequence
\[
0\longrightarrow C_t\longrightarrow\cdots
\longrightarrow C_0\longrightarrow X\longrightarrow0,
\qquad t\leq m,
\]
such that $(i+k)\text{-}\operatorname{dell}_{\mathcal{A}}C_i\leq m-i$ $(0\leq i\leq t).$
Since $T$ is exact, applying $T$ gives an exact sequence
\[
0\longrightarrow T(C_t)\longrightarrow\cdots
\longrightarrow T(C_0)\longrightarrow T(X)\longrightarrow0.
\]
Applying the first inequality with $i+k$ in place of $k$, we obtain
\[
\begin{aligned}
(i+k)\text{-}\operatorname{dell}_{\mathcal{B}}T(C_i)
&\leq
c+(i+k)\text{-}\operatorname{dell}_{\mathcal{A}}C_i  \\
&\leq c+m-i.
\end{aligned}
\]
Since $t\leq m\leq c+m$,  the preceding exact sequence satisfies the defining conditions for
a $(c+m)$-derived delooping of $T(X)$. Therefore,
\[
k\text{-}\operatorname{ddell}_{\mathcal{B}}(T X)\leq c+m.
\]

\label{thm:cleft-transfer}

\[
L_e
=
\max\bigl\{
\ell_{\mathcal{B}}(eS)
\mid
S\in\operatorname{simp}\mathcal{A}
\bigr\},
\]
where $\ell_{\mathcal{B}}(eS)$ denotes the composition length of an object
$eS$ in $\mathcal{B}$. 
\begin{enumerate}[(a)]
\item Suppose that $e$ admits an exact right adjoint $r$. Then
\[
\operatorname{ddell}(\mathcal{B})
\leq
\operatorname{ddell}(\mathcal{A})
\leq
\operatorname{pd}(r)+L_e\bigl(\operatorname{ddell}(\mathcal{B})+1\bigr).
\]

\[
k\text{-}\operatorname{ddell}(\mathcal{A})<\infty
\quad\Longleftrightarrow\quad
k\text{-}\operatorname{ddell}(\mathcal{B})<\infty.
\]

\item  Suppose that
$
\operatorname{Im}G\subseteq\operatorname{Proj}\mathcal{A},
$ Then, for every $k\geq1$,
\[
k\text{-}\operatorname{ddell}(\mathcal{B})
\leq
k\text{-}\operatorname{ddell}(\mathcal{A})
\leq
\max\bigl\{
1,\,
L_e\bigl(k\text{-}\operatorname{ddell}(\mathcal{B})+1\bigr)-1
\bigr\}.
\]

\[
k\text{-}\operatorname{ddell}(\mathcal{A})<\infty
\quad\Longleftrightarrow\quad
k\text{-}\operatorname{ddell}(\mathcal{B})<\infty.
\]
\end{enumerate}

\operatorname{ddell}(\mathcal{A})
\leq
\operatorname{pd}(r)+L_e\bigl(\operatorname{ddell}(\mathcal{B})+1\bigr).

\longrightarrow

\qquad

Since $\mathcal{A}$ has only finitely many simple objects and $0\leq j<n$, the

$\mathcal{B}$. The assumption $

\label{cor:cleft-findim}
Let $(\operatorname{mod}\text{-}\Gamma,
 \operatorname{mod}\text{-}\Lambda,i,e,l)$ be a cleft extension of finite-dimensional algebras over a field
$\Bbbk$. Assume that $l$ is exact, and $e$ preserves
projective modules.
\begin{enumerate}[(a)]
\item Suppose that $e$ admits an exact right adjoint $r$. Then
\[
\operatorname{ddell}(\Gamma)
\leq
\operatorname{ddell}(\Lambda)
\leq
\operatorname{pd}_{\Lambda}r(\Gamma)+L_e\bigl(\operatorname{ddell}(\Gamma)+1\bigr).
\]
Moreover, if $F^n=0$ for some $n\geq1$ and $\operatorname{pd}_{\Lambda}r(\Gamma)<\infty$, then, for every $k\geq1$,
\[k\text{-}\operatorname{ddell}\Lambda<\infty
\quad\Longleftrightarrow\quad
k\text{-}\operatorname{ddell}\Gamma<\infty.\]
\item  Suppose that
$
\operatorname{Im}G\subseteq\operatorname{proj}(\Lambda),
$ Then, for every $k\geq1$,
\[
k\text{-}\operatorname{ddell}(\Gamma)
\leq
k\text{-}\operatorname{ddell}(\Lambda)
\leq
\max\bigl\{
1,\,
L_e\bigl(k\text{-}\operatorname{ddell}(\Gamma)+1\bigr)-1
\bigr\}.
\]
In particular,
\[
k\text{-}\operatorname{ddell}\Lambda<\infty
\quad\Longleftrightarrow\quad
k\text{-}\operatorname{ddell}\Gamma<\infty.\]
\end{enumerate}
Consequently, if,  for some $k\geq1$, either $k\text{-}\operatorname{ddell}\Gamma<\infty$ or $k\text{-}\operatorname{ddell}\Lambda<\infty$, then  $
\operatorname{Findim}\Gamma^{\mathrm{op}}<\infty \ \text{and}\
\operatorname{Findim}\Lambda^{\mathrm{op}}<\infty.
$

<!-- remark-3: body_md -->
\label{R-arrow-removal-kddell}
Theorem~\ref{thm:arrow-removal} shows that arrow removal
preserves the finiteness of all higher derived delooping levels.
Thus an arrow addition or removal satisfying the above hypotheses
cannot turn an algebra with finite $k$-derived delooping level into
one with infinite $k$-derived delooping level, or conversely. In fact, the only possible change in the numerical value occurs at
the boundary value zero: if
$k\text{-}\operatorname{ddell}\Gamma\geq1,$
then
$
k\text{-}\operatorname{ddell}\Lambda
=
k\text{-}\operatorname{ddell}\Gamma,
$
whereas if
$
k\text{-}\operatorname{ddell}\Gamma=0,
$
then
$
0
\leq
k\text{-}\operatorname{ddell}\Lambda
\leq1.
$

\qquad

$\operatorname{Hom}_\Lambda(e_5\Lambda,e_6\Lambda)=0.$

indecomposable projective right $\Gamma$-modules have the following

\[
\begin{array}{ccccccccccccc}
1 && 2 &&   3 && 4&& 5&& 6  \\
2 \ 3&& 4 && 4 && 5&&6&&1  \\
4&&5  && && 6 && 1&&3 \end{array}.
\]
It follows directly from the above list that
$\operatorname{dell}_{\Gamma}S_i=0$ for
$i=1,3,4,5,6$. Since $S_2$ is not a submodule of any projective

\[
\Omega_\Gamma S_2
=
\begin{array}{c}
4\\
5
\end{array}
=
\Omega_\Gamma^2S_1.
\]
Thus  $\operatorname{dell}_\Gamma S_2\leq1.$

delooping level, we have
\[
\operatorname{ddell}_\Gamma S_2
\leq
\operatorname{dell}_\Gamma S_2
\leq1.
\]
Combining this with
\(\operatorname{ddell}_\Gamma S_2\neq0\), we obtain
\[
\operatorname{dell}_\Gamma S_2
=
\operatorname{ddell}_\Gamma S_2
=
1.
\]
It follows that $\operatorname{ddell}\Gamma=1.$

$\operatorname{ddell}\Lambda=1$.

\[
\operatorname{Findim}\Lambda^{\mathrm{op}}\leq1
\qquad\text{and}\qquad
\operatorname{Findim}\Gamma^{\mathrm{op}}\leq1.

Establishing the finiteness of $\operatorname{ddell}\Lambda$ directly

$

describe. Thus the reduction avoids the more involved direct
calculation over $\Lambda$.

\label{def:recollement}
{\rm\cite[Definition~2.1]{PH}}
A \emph{recollement} of abelian categories $\mathcal{A}$, $\mathcal{B}$, and $\mathcal{C}$ is a
diagram
\begin{equation}\label{recollement}
\begin{tikzcd}[column sep=5.5em]
\mathcal{A}
\arrow[r,"i"]
&
\mathcal{B}
\arrow[l,bend left=40,"p"]
\arrow[l,bend right=40,swap,"q"]
\arrow[r,"e"]
&
\mathcal{C}
\arrow[l,bend left=40,"r"]
\arrow[l,bend right=40,swap,"l"]
\end{tikzcd}
\end{equation}
satisfying the following conditions:
\begin{enumerate}
\item[\rm(RA1)] $(q,i,p)$ and $(l,e,r)$ are adjoint triples;
\item[\rm(RA2)] The functors $i$, $l$, and $r$ are fully faithful.

\item[\rm(RA3)] $\operatorname{Im}i=\operatorname{Ker}e.
$
\end{enumerate}
We denote this recollement by
$
\mathsf{R}(\mathcal{A},\mathcal{B},\mathcal{C}).
$

\(5\). Moreover, no path leaving these two vertices can return to
either of them. Hence
\[
H_0=\{4,5\},
\qquad
H_0^-=\{1,2,3\},
\qquad
H_0^+=\{6\},
\qquad
H_0^\circ=\{7\}.
\]
Put $e=e_4+e_5,$ $e^-=e_1+e_2+e_3$, $e^+=e_6,$ $e^\circ=e_7,$ and $A=e\Lambda e.$ 
The algebra supported on the homological heart is
\[
A
\cong
\Bbbk
\left(
4\mathrel{\substack{\xrightarrow{\alpha}\\[-1mm]
                    \xleftarrow[\beta]{}}}5
\right)
\big/
\langle\alpha\beta,\beta\alpha\rangle.
\]
Let \(T_4\) and \(T_5\) denote its simple right
modules. Their projective covers give $\Omega_A T_4\cong T_5,$
$\Omega_A T_5\cong T_4.$  It follows directly from the definition that
\[
 k\text{-}\operatorname{dell}_A T_4
 =k\text{-}\operatorname{dell}_A T_5=0,
 \qquad
 k\text{-}\operatorname{ddell} A=0.
\]
The algebra supported on \(H^+\cup H^\circ\) is
\[
D_0
=
(e^++e^\circ)\Lambda(e^++e^\circ)
\cong
\Bbbk(7\xrightarrow{w}6),
\]
whereas the algebra supported on \(H^-\) is
\[
D_-
=
e^-\Lambda e^-
\cong
\Bbbk(1\xrightarrow{u_1}2\xrightarrow{u_2}3).
\]
Thus $\operatorname{gl.dim} D_0=\operatorname{gl.dim} D_-=1.$ 
Since 
$e^-\Lambda e\cong(e_4A)^{\oplus3}$, we obtain $\operatorname{pd}_A(e^-\Lambda e)=0.$
All the hypotheses of
Corollary~\ref{cor:ddell-homological-heart} are now satisfied. We conclude that
\[
k\text{-}\operatorname{ddell}\Lambda<\infty
\qquad
\text{for every }k\geq1.
\]

Thus a direct calculation involving all seven simple \(\Lambda\)-modules
is replaced by a periodicity calculation for the two simple modules in
the homological heart and the projectivity of \(e^-\Lambda e\).
