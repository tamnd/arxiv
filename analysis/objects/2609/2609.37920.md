\label{thm:cong-p}
		Let $g\geq 1$ and let $p=6g\pm1$. For all $n\geq0$ and
		$0\leq i\leq p-1$, we have
		\[
		\mathcal{EO}_u(2pn+2i)\equiv0\pmod2,

		i\not\equiv \frac{p^2-1}{12}\pmod p
		\)
		and
		\(
		i\not\equiv (k-g)(3k-3g\mp1)\pmod p
		\)
		for every $1\leq k\leq (p-1)/2$, where the signs are paired.
		Thus, the minus sign is used in the second expression when
		$p=6g+1$, and the plus sign is used when $p=6g-1$.

Modulo $2$, we have
		\begin{align}
			\label{EOu 2n mod 2}
			\sum_{m=0}^{\infty}\mathcal{EO}_u(2m)q^m
			=\frac{f_2^2}{f_1^2}
			\equiv f_1^2\pmod2.
		\end{align}

		By Lemma~\ref{lem:2},
		\[
		f_1^2\equiv f_{p^2}^2
		\left(
		q^{(p^2-1)/12}
		+
		\sum_{k=1}^{(p-1)/2}
		q^{(k-g)(3k-3g-\varepsilon)}
		\frac{
			f(-q^{2pk},-q^{p^2-2pk})^2
		}{
			f(-q^{pk},-q^{p^2-pk})^2
		}
		\right)
		\pmod2.
		\]
		Now, $f_{p^2}^2$ is a power series in $q^p$. Moreover, for
		each $k$,
		\(
		\dfrac{
			f(-q^{2pk},-q^{p^2-2pk})^2
		}{
			f(-q^{pk},-q^{p^2-pk})^2
		}
		\)
		is also a power series in $q^p$. It follows that, modulo $2$, the only residue classes modulo
		$p$ that can occur among the exponents of $f_1^2$ are
		\(
		\frac{p^2-1}{12}\pmod p
		\)
		and
		\(
		(k-g)(3k-3g-\varepsilon)\pmod p,\)
		for \( 1\leq k\leq\frac{p-1}{2}.
		\)

		\eqref{EOu 2n mod 2} gives

		\mathcal{EO}_u\bigl(2(pn+i)\bigr)
