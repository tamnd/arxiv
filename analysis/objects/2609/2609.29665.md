\label{thm:counterexample}

$X$
and
$Y$
and a bijection
$U\colon X\to Y$
such that

\norm{U(p)-U(q)}_{Y}=\norm{p-q}_{X}

for every
$x\in X$

$p,q\in B_{X}(x,1/4)$,

$U$

<!-- lemma-1: body_md -->
\label{lem:metric-adjunction}
Let
$M$
and
$N$
be real Banach spaces,
let
$r>0$,
and let
$V\colon M\to N$
be injective and satisfy
\eqref{eq:short-scale}.
Fix
$y\in N\setminus V(M)$
and
$a\in M$.
Put
$C=\norm{V(a)-y}_{N}$,
choose
$H>C+4r$,
and set
\[
E=M\oplus_1 \mathbb{R},
\qquad
\iota_E\colon M\to E,
\qquad
\iota_E(m)=(m,0).
\]
Define
\[
A=(M\times\set{0})\cup\set{(a,H)}
\]
and
$\varphi\colonA\to N$
by
\[
\varphi(m,0)=V(m),
\qquad
\varphi(a,H)=y.
\]
Let
$P=N\cup_{\varphi}E$
be the quotient of the disjoint union of
$N$
and
$E$
obtained by identifying every
$u\inA$
with
$\varphi(u)$.
Let
\[
q\colon E\toP
\quad\text{and}\quad
\iota_N\colon N\toP
\]
be the canonical maps.
For
$Z\in\set{E,N}$
and
$u,v\in Z$,
write
$[u,v]_Z$
for the oriented affine segment from
$u$
to
$v$.
We write
$\mathcal C_0*\mathcal C_1$
for concatenation of compatible paths and
$\overline{\mathcal C_0}$
for the reversed path.
An admissible polygonal path
$\mathcal C$
in
$P$
is a finite concatenation of images under
$q$
or
$\iota_N$
of affine line segments lying in
$E$
or
$N$.
If
$\mathcal C$
has
$k$
pieces with endpoints
$u_j,v_j\in Z_j$
for
$1\le j\le k$,
where
$Z_j\in\set{E,N}$,
define
\[
\operatorname{len}(\mathcal C)
=\sum_{j=1}^{k}\norm{u_j-v_j}_{Z_j}.
\]
The path metric on
$P$
is
\[
d_P(z,w)
=\inf\set{
\operatorname{len}(\mathcal C)
\mid
\mathcal C\text{ is an admissible polygonal path from }z\text{ to }w
}.
\]
Then
$\varphi$
is injective and 1-Lipschitz,
$\iota_N$
is an isometric embedding,
and
$q$
is injective and 1-Lipschitz.
The maps satisfy
\[
q\circ\iota_E
=\iota_N\circ V
\quad\text{and}\quad
q(a,H)
=\iota_N(y).
\]
Moreover,
if
$x_0,x_1\in E$
and
$\norm{x_0-x_1}_{E}\le r$,
then
\[
d_P(q(x_0),q(x_1))
=\norm{x_0-x_1}_{E}.
\]

<!-- proof-1: body_md -->
The set
$A$
is closed in
$E$.
The set
$A$
consists of the points of
$E$
that will be identified with their images under
$\varphi$
in the construction below.
The map
$\varphi$
is injective because
$V$
is injective and
$y\notin V(M)$.
Its restriction to
$M\times\set{0}$
is 1-Lipschitz by
\eqref{eq:global-one-lipschitz}.
For
$m\in M$,
the distance between
$(m,0)$
and the remaining point
$(a,H)$
satisfies
\[
\begin{split}
\norm{V(m)-y}_{N}
&\le \norm{V(m)-V(a)}_{N}+C
\le \norm{m-a}_{M}+C\\
&<\norm{m-a}_{M}+H
=\norm{(m,0)-(a,H)}_{E}.
\end{split}
\]
Thus
$\varphi$
is 1-Lipschitz on
$A$.

The definition of
$d_P$
makes
$q$
1-Lipschitz.
Let
$\iota_A\colonA\to E$
be the inclusion.
The canonical maps satisfy
$q\circ\iota_A
=\iota_N\circ\varphi$,
as displayed in the adjunction square
\begin{equation}\label{diag:metric-adjunction}
\begin{tikzcd}[column sep=large,row sep=large]
A
  \arrow[r,hook,"\iota_A"]
  \arrow[d,"\varphi"']
& E \arrow[d,"q"]\\
N \arrow[r,"\iota_N"']
& P.
\end{tikzcd}
\end{equation}
Evaluating the commuting identity on
$(m,0)$
and
$(a,H)$
yields the two identities in the statement.
For readability,
we write
$n$
for
$\iota_N(n)$
inside
$P$.

Every admissible polygonal path with both endpoints in
$E$
may be shortened so that it is either one segment in
$E$
or leaves
$E$
once and returns once.
To see this,
consider an internal segment
$[u,v]_E$
whose endpoints belong to
$A$.
The corresponding segment in
$N$
satisfies
\[
\begin{aligned}
\operatorname{len}([\varphi(u),\varphi(v)]_N)
&=\norm{\varphi(u)-\varphi(v)}_{N}\\
&\le \norm{u-v}_{E}
=\operatorname{len}([u,v]_E),
\end{aligned}
\]
because
$\varphi$
is 1-Lipschitz.
Replace every internal
$E$
segment in this way.
Consecutive segments in
$N$
can then be replaced by the affine segment joining their endpoints,
whose length is no larger by the triangle inequality in
$N$.
If the original path never enters
$N$,
the triangle inequality in
$E$
replaces it by the affine segment joining its endpoints.
This gives the asserted two forms without increasing length.
Consequently,
for
$x_0,x_1\in E$,
\begin{equation}\label{eq:adjunction-e-distance}
\begin{split}
&d_P(q(x_0),q(x_1))\\
&\quad=\min\biggl\{
\norm{x_0-x_1}_{E},\\
&\qquad\inf_{u,v\in A}
\bigl(
\norm{x_0-u}_{E}
+\norm{\varphi(u)-\varphi(v)}_{N}
+\norm{v-x_1}_{E}
\bigr)
\biggr\}.
\end{split}
\end{equation}
The second term on the right hand side of
\eqref{eq:adjunction-e-distance}
is the infimum of the lengths of the polygonal paths
$\mathcal C_{u,v}$
defined by
\[
\mathcal C_{u,v}
=[x_0,u]_E
*[\varphi(u),\varphi(v)]_N
*[v,x_1]_E.
\]
Thus
\[
\operatorname{len}(\mathcal C_{u,v})
=\norm{x_0-u}_{E}
+\norm{\varphi(u)-\varphi(v)}_{N}
+\norm{v-x_1}_{E}.
\]

$n\in N$,
the corresponding formula is
\begin{equation}\label{eq:adjunction-n-distance}
d_P(q(x_0),n)
=\inf_{u\in A}
\bigl(
\norm{x_0-u}_{E}+\norm{\varphi(u)-n}_{N}
\bigr).
\end{equation}

The same path replacement shows that
$\iota_N$
is an isometric embedding.
The map
$q$
is injective.
Indeed,
if
$d_P(q(x_0),q(x_1))=0$,
then
\eqref{eq:adjunction-e-distance}
shows that either
$\norm{x_0-x_1}_{E}=0$
or the second term on its right hand side is zero.
In the latter case,
there are
$u_k,v_k\in A$
such that
\[
\norm{x_0-u_k}_{E}\to 0,
\qquad
\norm{x_1-v_k}_{E}\to 0,
\qquad
\norm{\varphi(u_k)-\varphi(v_k)}_{N}\to 0.
\]
Closedness of
$A$
places
$x_0$
and
$x_1$
in
$A$.
Continuity and injectivity of
$\varphi$
then imply
$x_0=x_1$.
Thus
$x_0=x_1$
in either case.

We next show that
$q$
preserves every distance at most
$r$.
Write
\[
x_0=(m_0,s_0),
\qquad
x_1=(m_1,s_1),
\qquad
d=\norm{x_0-x_1}_{E}\le r.
\]
For the two points of
$A$
in the second term of
\eqref{eq:adjunction-e-distance},
there are three possibilities.
Put
\[
t=(a,H).
\]
\begin{enumerate}[label=\textup{(A\arabic*)},ref=\textup{(A\arabic*)}]
\item\label{item:two-base-attachments}
If both points belong to
$M\times\set{0}$,
write them as
$(u,0)$
and
$(v,0)$.
Since
$\norm{m_0-m_1}_{M}\le d\le r$,
define the comparison path in
$N$
by
\[
\mathcal D_{u,v}
=[V(m_0),V(u)]_N
*[V(u),V(v)]_N
*[V(v),V(m_1)]_N.
\]
Its length satisfies
\begin{align*}
\norm{m_0-m_1}_{M}
&=\norm{V(m_0)-V(m_1)}_{N}\\
&\le \operatorname{len}(\mathcal D_{u,v})\\
&=\norm{V(m_0)-V(u)}_{N}
+\norm{V(u)-V(v)}_{N}
+\norm{V(v)-V(m_1)}_{N}\\
&\le \norm{m_0-u}_{M}
+\norm{V(u)-V(v)}_{N}
+\norm{v-m_1}_{M}.
\end{align*}
Here the equality uses
\eqref{eq:short-scale},
the first inequality is the triangle inequality along
$\mathcal D_{u,v}$,
and the final inequality uses
\eqref{eq:global-one-lipschitz}.
The corresponding path in
$P$
is
\[
\mathcal C_{(u,0),(v,0)}
=[x_0,(u,0)]_E
*[V(u),V(v)]_N
*[(v,0),x_1]_E.
\]
Adding its three segment lengths gives
\begin{align*}
\operatorname{len}(\mathcal C_{(u,0),(v,0)})
&=\norm{x_0-(u,0)}_{E}
+\norm{V(u)-V(v)}_{N}
+\norm{(v,0)-x_1}_{E}\\

+\norm{V(u)-V(v)}_{N}
+\norm{v-m_1}_{M}+\abs{s_1}\\
&\ge \norm{m_0-m_1}_{M}+\abs{s_0-s_1}\\
&=\norm{x_0-x_1}_{E}=d.
\end{align*}
\item\label{item:two-top-attachments}
If both points equal

\mathcal C_{t,t}

The middle segment is degenerate.

the triangle inequality in
$E$
gives
\begin{align*}
\operatorname{len}(\mathcal C_{t,t})
&=\norm{x_0-t}_{E}
+\operatorname{len}([y,y]_N)
+\norm{t-x_1}_{E}\\
&=\norm{x_0-t}_{E}+\norm{t-x_1}_{E}\\
&\ge \norm{x_0-x_1}_{E}=d.
\end{align*}
\item\label{item:mixed-attachments}
If one point belongs to
$M\times\set{0}$
and the other equals
$t$,
write the first point as
$(u,0)$.
Suppose first that the two points occur in the order
$(u,0),t$.
The corresponding path in
$P$

\[
\mathcal C_{(u,0),t}
=[x_0,(u,0)]_E
*[V(u),y]_N
*[t,x_1]_E.
\]
To estimate its two
$E$

consider the comparison path in
$E$

\mathcal L_{u}
=[(u,0),x_0]_E*[x_0,x_1]_E*[x_1,t]_E.
\]
This path joins
$(u,0)$

$t$.

\[
\norm{(u,0)-t}_{E}
=\norm{u-a}_{M}+H
\ge H,
\]
the triangle inequality along
$\mathcal L_{u}$
gives
\begin{align*}
H
&\le \norm{(u,0)-t}_{E}\\
&\le \operatorname{len}(\mathcal L_{u})\\
&=\norm{x_0-(u,0)}_{E}
+\norm{x_0-x_1}_{E}
+\norm{t-x_1}_{E}\\
&=\norm{x_0-(u,0)}_{E}+d+\norm{t-x_1}_{E}.
\end{align*}
Therefore,
\[
\norm{x_0-(u,0)}_{E}+\norm{t-x_1}_{E}
\ge H-d.
\]
Adding the three segment lengths of
$\mathcal C_{(u,0),t}$
and using the nonnegativity of the middle length now yields
\begin{align*}
\operatorname{len}(\mathcal C_{(u,0),t})
&=\norm{x_0-(u,0)}_{E}
+\norm{V(u)-y}_{N}
+\norm{t-x_1}_{E}\\
&\ge H-d.
\end{align*}
Since
$H>2r\ge 2d$,
this lower bound is greater than
$d$.
When the two attachment points occur in the opposite order,
the corresponding path is
\[
\mathcal C_{t,(u,0)}
=[x_0,t]_E
*[y,V(u)]_N
*[(u,0),x_1]_E.
\]
In this order,
use the comparison path
\[
\widetilde{\mathcal L}_{u}
=[t,x_0]_E*[x_0,x_1]_E*[x_1,(u,0)]_E.
\]
It joins
$t$
to
$(u,0)$,
so the same triangle inequality gives
\begin{align*}
H
&\le \norm{t-(u,0)}_{E}\\
&\le \operatorname{len}(\widetilde{\mathcal L}_{u})\\
&=\norm{x_0-t}_{E}+d+\norm{(u,0)-x_1}_{E}.
\end{align*}
It follows that
\begin{align*}
\operatorname{len}(\mathcal C_{t,(u,0)})
&=\norm{x_0-t}_{E}
+\norm{y-V(u)}_{N}
+\norm{(u,0)-x_1}_{E}\\
&\ge H-d>d.
\end{align*}
\end{enumerate}
The cases
\ref{item:two-base-attachments}--\ref{item:mixed-attachments}
exhaust the second term in
\eqref{eq:adjunction-e-distance}.
It follows from
\eqref{eq:adjunction-e-distance}
that
\begin{equation}\label{eq:q-short-isometry}
d_P(q(x_0),q(x_1))=\norm{x_0-x_1}_{E}
\end{equation}
whenever
$\norm{x_0-x_1}_{E}\le r$.

The dual of the space before taking the quotient is
\[
\bigl(N\oplus_1 \mathcal{F}(P)\bigr)^*
=N^*\oplus_\infty \operatorname{Lip}_0(P).
\]
For
$\ell\inN^*$
and
$f\in\operatorname{Lip}_0(P)$,
define the linear functional
\begin{equation}\label{eq:relative-dual-action}
\begin{aligned}
\Phi_{\ell,f}
&\colonN\oplus_1 \mathcal{F}(P)\to\mathbb{R},
\\
\Phi_{\ell,f}(n,\mu)
&=\ell(n)+\mu(f).
\end{aligned}
\end{equation}
This functional annihilates
$Z$
exactly when
\[
0=\Phi_{\ell,f}(n,-\delta_{n})
=\ell(n)-f(n)
\]
for every
$n\in N$.
Thus the annihilator condition is precisely
$f|_N=\ell$.
Whenever this condition holds,
we use the same symbol for the induced functional on
$B_0$.
In particular,
\begin{equation}\label{eq:relative-dual-quotient-action}
\Phi_{\ell,f}(\Pi(n,\mu))
=\ell(n)+\mu(f).
\end{equation}
The dual unit ball of
$B_0$
therefore consists of the pairs
$(\ell,f)$
satisfying
\[
\norm{\ell}_{N^*}\le 1,
\qquad
f\in\operatorname{Lip}_0(P),
\qquad
\operatorname{Lip}(f)\le 1,
\qquad
f|_N=\ell.
\]
For
$n\in N$,
the Hahn--Banach theorem provides
$\ell\inN^*$
such that
$\norm{\ell}_{N^*}\le 1$
and
$\ell(n)=\norm{n}_{N}$.
Thus
$\ell$
is a norming functional for
$n$.
Extend
$\ell$
to
$P$
by the McShane theorem.
If
$f$
denotes this extension,
then
\eqref{eq:relative-dual-quotient-action}

\[
\Phi_{\ell,f}(j_0(n))
=\ell(n)
=\norm{n}_{N}.
\]
The quotient map yields the reverse estimate because
\[
\begin{aligned}
\norm{j_0(n)}_{B_0}
&=\norm{\Pi(n,0)}_{B_0}\\
&=\inf_{z\inZ}
\norm{(n,0)+z}_{N\oplus_1 \mathcal{F}(P)}\\
&\le \norm{(n,0)}_{N\oplus_1 \mathcal{F}(P)}\\
&=\norm{n}_{N}.
\end{aligned}

The inequality uses
$0\inZ$.

$j_0\colon N\to B_0$

The map
$\eta$

$\Pi$
is 1-Lipschitz and
$\delta_{P}$

\qquad\text{and}\qquad
c_1=c_0-d.

Then
$0\le c_0\le a_0$

because

$\abs{c_i}\le a_i$

&=\inf_{n\in N}\inf_{u\inA}

$\eta\circq$,

\norm{m_0-m_1}_{M}

For

\[
R_0(j_0(n))=n.
\]

$R_0$

\end{equation}

\qquad
\pi_{B_0}(b,g)=b.

=R_0\circ\pi_{B_0}
\colon F\to N.
\]

\begin{tikzcd}[column sep=large,row sep=large]
E
  \arrow[r,"q"]
  \arrow[d,"\rho_E"']
& P
  \arrow[r,"J"]
  \arrow[d,"\rho_P"']

\end{tikzcd}

\eqref{eq:adjunction-recovery-maps}.
The right square follows from
\eqref{eq:relative-recovery-map}

$\pi_{B_0}\circJ=\eta$.

$\abs{s}\le L$,

$\lambda(s)=0$,
and therefore
\[
(R\circV^{+})(m,s)=V(m).
\]
Finally,
set
\[
R_{\mathcal A}^{\mathcal B}
=R.
\]
The preceding identities prove
\ref{item:extension-retraction}
and
\ref{item:extension-recovery},
which completes the proof.
