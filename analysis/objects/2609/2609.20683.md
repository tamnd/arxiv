<!-- lemma-1: body_md -->
\label{lem:finite-info}
Let $n\in\mathbb{N}$.  A subset of $\Omega$ belongs to $\mathscr F_n$ if and only if it
is a union of length-$n$ cylinders.  Every such set is clopen, and
$\mathscr F_n$ is finite.  Consequently, if $g:\Omega\to\mathbb{R}$ is constant on each
length-$n$ cylinder, then $\{\omega:g(\omega)>c\}$ and
$\{\omega:g(\omega)\ge c\}$ are clopen for every $c\in\mathbb{R}$. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The $2^n$ length-$n$ cylinders are pairwise disjoint and cover $\Omega$.  The
unions of subfamilies of a finite partition form a $\sigma$-algebra, here with
$2^{2^n}$ members, and this $\sigma$-algebra contains the length-$n$ cylinders
and is contained in every $\sigma$-algebra that contains them; it is therefore
$\mathscr F_n$.  A finite union of clopen sets is clopen.  If $g$ is constant on
each length-$n$ cylinder, then $\{\omega:g(\omega)>c\}$ is the union of those
length-$n$ cylinders on which the constant value of $g$ exceeds $c$, and
similarly with $\ge$ in place of $>$.
<!-- end proof-1 -->
<!-- remark-1: body_md -->
\label{rem:terminology}
The conventions of \citet{nielsen2021convergence} differ from ours and should be
translated when the two papers are read side by side.  There, ``probability
measure'' means what we call a probability function; ``countably additive
probability measure'' is the strengthened notion we call a probability measure;
and ``merely finitely additive'' agrees with our usage.  Nielsen writes
$\omega^n$ for the cylinder we write $[\omega\mathord{\upharpoonright} n]$, $\mathcal C$ for our
clopen algebra $\mathcal{C}$, $\mathcal F$ for our Borel $\sigma$-algebra $\mathcal{B}$, and
$\mathcal C_n$ for our finite $\sigma$-algebra $\mathscr F_n$.  He calls
\eqref{eq:posi} below Assumption~1. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end remark-1 -->
<!-- definition-1: body_md -->
\label{def:au-general}
Let $P$ be a probability function on $\mathcal{B}$, and let $f$ and
$f_1,f_2,\ldots$ be real-valued Borel functions on $\Omega$.  Then
$(f_n)_{n=1}^{\infty}$ \textbf{converges to $f$ almost uniformly} if for every
$\varepsilon>0$ there is an event $E\in\mathcal{B}$ with $P(E)<\varepsilon$ such that $f_n\to f$
uniformly on $E^c$: for every $r>0$ there is $N\in\mathbb{N}$ with:
\[
  n\ge N,\ \omega\notin E
  \quad\Longrightarrow\quad
  \bigl|f_n(\omega)-f(\omega)\bigr|<r.
\]
The event $E$ may depend on $\varepsilon$ but not on $r$. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\spadesuit}$}}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:convergence}
Let $P$ be a probability function on $\mathcal{B}$, and let $A\in\mathcal{B}$.
\begin{enumerate}[(a)]
\item The posteriors  are said to converge to the truth \textbf{almost surely} for $A$ if
\[
  P\Biggl(\Bigl\{\,\omega\in \Omega\,:\,P_n(A)(\omega)\,\to\,\mathbf{1}_A(\omega)\,\Bigr\}\Biggr)\quad=\quad1.
\]
\item The posteriors are said to converge to the truth \textbf{almost uniformly} for $A$ if
$P_n(A)\to\mathbf{1}_A$ almost uniformly in the sense of \cref{def:au-general}: for every
$\varepsilon>0$, there is $E\in\mathcal{B}$ with $P(E)<\varepsilon$ such that, for every $r>0$,
there is $N$ with:
\[
  n\ge N,\ \omega\notin E
  \quad\Longrightarrow\quad
  \Bigl|\,P_n(A)(\omega)\,-\,\mathbf{1}_A(\omega)\,\Bigr|\;<\;r.
\]
\end{enumerate}
The probability function $P$ has either convergence property \textbf{to the truth}
(simpliciter) if it has that property for every $A\in\mathcal{B}$. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\spadesuit}$}}

<!-- definition-3: body_md -->
\label{def:approx}
A probability function $P$ on $\mathcal{B}$ has the \textbf{approximation property} if for every
$A\in\mathcal{B}$ and every $\varepsilon>0$, there is a clopen $C\in\mathcal{C}$ such that:
\begin{equation*}
  P(A\mathbin{\triangle} C)<\varepsilon.
  \tag*{ 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\spadesuit}$}}}
\end{equation*}
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:cht}
A probability function $P$ on $\mathcal{B}$ is said to be \textbf{countably additive on
conditional hitting times}, abbreviated \textbf{CHT}, if:
\begin{equation}\label{eq:cht}
 P\left(\bigcup_{n=1}^{\infty}D_n(A,\delta)\right)
   =\sum_{n=1}^{\infty}P(D_n(A,\delta))
\end{equation}
for every $A\in\mathcal{B}$ and $\delta\in(0,1)$. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\spadesuit}$}}

<!-- lemma-2: body_md -->
\label{lem:au-tail}
Let $P$ be a probability function on $\mathcal{B}$, and let $(f_n)^{\infty}_{n=1}$ and $f$ be
real-valued Borel functions on $\Omega$.  If $f_n\to f$ almost uniformly, then
for every $r>0$:
\begin{equation*}
  P\left(\left\{\omega:\sup_{m\ge n}\bigl|f_m(\omega)-f(\omega)\bigr|>r\right\}\right)
  \longrightarrow0
  \qquad\mbox{as }n\to\infty.
  \tag*{ 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}}
\end{equation*}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Fix $r>0$ and an arbitrary $\gamma>0$.  Choose $E$ with $P(E)<\gamma$ such that
$f_n\to f$ uniformly on $E^c$.  For all sufficiently large $n$, the displayed
tail event is contained in $E$, and hence has probability less than $\gamma$.
Since $\gamma$ is arbitrary, the probabilities tend to zero.
<!-- end proof-2 -->
<!-- proposition-1: body_md -->
\label{prop:necessity}
Let $P$ be a probability function on $\mathcal{B}$.  If $P$ satisfies
\eqref{eq:posi} and converges to the truth almost uniformly, then $P$ has the
approximation property and the CHT property.
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
Fix $A\in\mathcal{B}$.

\emph{Approximation.}
Since $|P_n(A)-\mathbf{1}_A|\le\sup_{m\ge n}|P_m(A)-\mathbf{1}_A|$ pointwise,
\cref{lem:au-tail} gives:
\[
 P\bigl(|P_n(A)-\mathbf{1}_A|>r\bigr)\longrightarrow0
 \qquad\mbox{ for each } r>0.
\]
Define the event:
\[
  B_n\;=\;\Bigl\{\,\omega\in \Omega:P_n(A)(\omega)\,\ge\,\tfrac{1}{2}\,\Bigr\},
\]
which is clopen by \cref{lem:finite-info}.
If $\omega\in A\mathbin{\triangle} B_n$, then
$|P_n(A)(\omega)-\mathbf{1}_A(\omega)|\ge\tfrac{1}{2}$.  Therefore:
\[
  P(A\mathbin{\triangle} B_n)
  \;\le\; P\Bigl(\,\Bigl|P_n(A)-\mathbf{1}_A\Bigr|>\tfrac{1}{3}\,\Bigr)\longrightarrow0,
\]
which proves the approximation property.

\emph{Conditional hitting times.}
Fix $\delta\in(0,1)$, abbreviate $D_n=D_n(A,\delta)$, and set
$D=\bigcup_nD_n$.  For every $N\in\mathbb{N}$:
\begin{align*}
 P(A\cap D)
 \quad&=\quad\sum_{n=1}^N P(A\cap D_n)
   \,+\,P\left(A\cap\bigcup_{n>N}D_n\right).
\end{align*}
If $\omega$ belongs to the last event, no hit has occurred by time $N$, so
$P_N(A)(\omega)\le\delta$ and
$|P_N(A)(\omega)-1|\ge1-\delta$.  Lemma~\ref{lem:au-tail}, used with the smaller
strict threshold $(1-\delta)/2$, implies that the last term tends to zero.  Hence:
\[
 P(A\cap D)=\sum_{n=1}^{\infty}P(A\cap D_n).
\]
Similarly, if
$\omega\in A^c\cap\bigcup_{n>N}D_n$, then $P_m(A)(\omega)>\delta$ for some
$m>N$, so
$\sup_{m>N}|P_m(A)(\omega)-\mathbf{1}_A(\omega)|>\delta$.  Another application of
Lemma~\ref{lem:au-tail} gives:
\[
 P(A^c\cap D)=\sum_{n=1}^{\infty}P(A^c\cap D_n).
\]
Adding the two identities establishes \eqref{eq:cht}, as desired.

<!-- lemma-3: body_md -->
\label{lem:standard-facts}
Let $\mu$ be a countably additive Borel probability on $\Omega$ that assigns
positive probability to every nonempty cylinder.
\begin{enumerate}[(a),itemsep=.8em,leftmargin=3em,labelsep=1em,topsep=0.6em]
\item For every $B\in\mathcal{B}$, the functions
\[
  \omega\longmapsto \mu(B\,|\,[\omega\mathord{\upharpoonright} n])
\]
converge to $\mathbf{1}_B$ $\mu$-almost surely.
\item If Borel functions $f_n$ converge to $f$ $\mu$-almost surely, then for every
$\rho>0$ there is a Borel event $E$ with $\mu(E)<\rho$ such that $f_n\to f$
uniformly on $E^c$. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
For part~(a), fix $B\in\mathcal{B}$.  The $\sigma$-algebra $\mathscr F_n$ is generated
by the finite partition of $\Omega$ into length-$n$ cylinders
(\cref{lem:finite-info}), and $\mu$ gives each cell of that partition positive
probability, so the conditional expectation of $\mathbf{1}_B$ given $\mathscr F_n$ is
the elementary ratio:
\[
  \mathbb{E}_{\mu}\bigl[\mathbf{1}_B\bigm|\mathscr F_n\bigr](\omega)
  \quad=\quad
  \frac{\mu\bigl(B\cap[\omega\mathord{\upharpoonright} n]\bigr)}
       {\mu\bigl([\omega\mathord{\upharpoonright} n]\bigr)}
  \quad=\quad
  \mu\bigl(B\,|\,[\omega\mathord{\upharpoonright} n]\bigr).
\]
The $\sigma$-algebras $\mathscr F_n$ increase with union generating $\mathcal{B}$, so
L\'evy's upward theorem \citep[Theorem~35.6, p.~470]{billingsley1995prob} yields
\[
  \mathbb{E}_{\mu}\bigl[\mathbf{1}_B\bigm|\mathscr F_n\bigr]
  \quad\longrightarrow\quad
  \mathbb{E}_{\mu}\bigl[\mathbf{1}_B\bigm|\mathcal{B}\bigr]
  \quad=\quad\mathbf{1}_B
\]
$\mu$-almost surely.  Part~(b) is Egorov's theorem
\citep[Problem~13.9]{billingsley1995prob}.

<!-- lemma-4: body_md -->
\label{lem:regularity}
Let $\mu$ be a countably additive Borel probability on $\Omega$.  For every
$A\in\mathcal{B}$ and every $\rho>0$, there are a closed set $F$ and an open set $G$
such that $F\subseteq A\subseteq G$ and $\mu(G\setminus F)<\rho$.
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Let $\mathscr K$ denote the family of Borel sets with the stated property for
every $\rho>0$.  We show that $\mathscr K$ is a $\sigma$-algebra containing
every closed set; since the closed sets generate $\mathcal{B}$, this gives
$\mathscr K=\mathcal{B}$.

\emph{Closed sets.}  Let $F_0$ be closed.  If $F_0=\Omega$, then $F=G=\Omega$
serves, so assume $F_0^c\ne\varnothing$.  Its complement is open and therefore a
union of countably many cylinders; let $C_1\subseteq C_2\subseteq\cdots$ be the
partial unions of an enumeration of those cylinders, so that each $C_m$ is clopen
and $\bigcup_mC_m=F_0^c$.  Then each $G_m=C_m^c$ is clopen,
$F_0\subseteq G_m$, and $\bigcap_mG_m=F_0$.  Continuity from above gives
$\mu(G_m\setminus F_0)\to0$, so $F=F_0$ and $G=G_m$ with $m$ large enough serve.

\emph{Complements.}  If $F\subseteq A\subseteq G$ with $\mu(G\setminus F)<\rho$,
then $G^c\subseteq A^c\subseteq F^c$, where $G^c$ is closed and $F^c$ is open,
and $F^c\setminus G^c=G\setminus F$.

\emph{Countable unions.}  Let $A=\bigcup_{j\ge1}A_j$ with every
$A_j\in\mathscr K$, and let $\rho>0$.  Choose closed $F_j$ and open $G_j$ with
$F_j\subseteq A_j\subseteq G_j$ and $\mu(G_j\setminus F_j)<\rho2^{-j-1}$.  The
set $G=\bigcup_jG_j$ is open and contains $A$.  The sets
$H_J=\bigcup_{j\le J}F_j$ are closed and increase to $H=\bigcup_jF_j$, so
continuity from below supplies $J$ with $\mu(H\setminus H_J)<\rho/2$; put
$F=H_J$, a closed subset of $A$.  If $\omega\in G\setminus F$, then
$\omega\in G_j$ for some $j$, and either $\omega\notin F_j$ or
$\omega\in H\setminus H_J$.  Hence:
\[
  G\setminus F
  \quad\subseteq\quad
  \bigcup_{j\ge1}\bigl(G_j\setminus F_j\bigr)\;\cup\;\bigl(H\setminus H_J\bigr),
\]
and therefore $\mu(G\setminus F)<\rho/2+\rho/2=\rho$.

\label{lem:nu}

\begin{enumerate}[(a),itemsep=.5em,leftmargin=3em,labelsep=1em,topsep=.6em]
\item $\nu_k(S_k)=1$;
\item $\nu_k$ is diffuse; in particular $\nu_k(D\cup\{z\})=0$, where $D$ is the
      countable set \eqref{eq:Dset};
\item $\nu_k(B)>0$ for every nonempty cylinder $B\subseteq S_k$.

is the set of paths beginning $0^{k-1}1$, which is $S_k$.  If $x$ and $y$ agree
in their first $n$ coordinates, then $T_k(x)$ and $T_k(y)$ agree in their first

(a) $T_k^{-1}(S_k)=\Omega$, so $\nu_k(S_k)=Q(\Omega)=1$.

<!-- definition-5: body_md -->
\label{def:ultrafilter}
A \textbf{filter} on $\mathbb{N}$ is a family $\mathcal{U}\subseteq 2^{\mathbb{N}}$ such that:
\begin{enumerate}[(i),itemsep=0.5em,leftmargin=3em,labelsep=1em,topsep=0.3em]
\item $\varnothing\notin\mathcal{U}$ and $\mathbb{N}\in\mathcal{U}$;
\item if $A,B\in\mathcal{U}$, then $A\cap B\in\mathcal{U}$;
\item if $A\in\mathcal{U}$ and $A\subseteq B\subseteq\mathbb{N}$, then $B\in\mathcal{U}$.
\end{enumerate}
It is an \textbf{ultrafilter} if, for every $A\subseteq\mathbb{N}$, exactly one of $A$ and
$A^c$ belongs to $\mathcal{U}$.  It is \textbf{free} if it contains no finite set.
Members of $\mathcal{U}$ will be called \textbf{$\mathcal{U}$-large}. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\spadesuit}$}}

\label{lem:partition}
$\{I_i\}_{i\ge1}$ is a partition of $\mathbb{N}$ into infinitely many infinite sets.

  \quad\hbox{{${\scriptstyle\blacksquare}$}}

\label{def:blockfilter}
Let:
\begin{equation}\label{eq:blockfilter}
  \mathcal{F}\quad=\quad\Bigl\{\,X\subseteq\mathbb{N}\;:\;
    \bigl\{\,i\in\mathbb{N}:I_i\setminus X\text{ is finite}\,\bigr\}
    \text{ is cofinite}\,\Bigr\}.
\end{equation}
Thus $X\in\mathcal{F}$ when, for all but finitely many blocks, $X$ contains all but

  \quad\hbox{{${\scriptstyle\spadesuit}$}}

<!-- lemma-8: body_md -->
\label{lem:blockfilter}
$\mathcal{F}$ is a filter on $\mathbb{N}$ containing every cofinite subset of $\mathbb{N}$, and every
ultrafilter $\mathcal{U}$ with $\mathcal{F}\subseteq\mathcal{U}$ is free.  At least one such
$\mathcal{U}$ exists.
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end lemma-8 -->
<!-- proof-9: body_md -->
Write $\varphi(X)=\{i:I_i\setminus X\text{ is finite}\}$, so that $X\in\mathcal{F}$
if and only if $\varphi(X)$ is cofinite.

Each $I_i\setminus\varnothing=I_i$ is infinite by \cref{lem:partition}, so
$\varphi(\varnothing)=\varnothing$, which is not cofinite; hence
$\varnothing\notin\mathcal{F}$.  And $\varphi(\mathbb{N})=\mathbb{N}$, so $\mathbb{N}\in\mathcal{F}$.

Since $I_i\setminus(X\cap Y)=(I_i\setminus X)\cup(I_i\setminus Y)$ and a union
of two finite sets is finite, $\varphi(X)\cap\varphi(Y)\subseteq\varphi(X\cap Y)$;
an intersection of two cofinite sets is cofinite and a superset of a cofinite
set is cofinite, so $X,Y\in\mathcal{F}$ gives $X\cap Y\in\mathcal{F}$.  If
$X\subseteq Y$ then $I_i\setminus Y\subseteq I_i\setminus X$, so
$\varphi(X)\subseteq\varphi(Y)$ and $\mathcal{F}$ is upward closed.

If $\mathbb{N}\setminus X$ is finite then $I_i\setminus X$ is finite for every $i$, so
$\varphi(X)=\mathbb{N}$ and $X\in\mathcal{F}$: every cofinite set belongs to $\mathcal{F}$.
Consequently, if $\mathcal{F}\subseteq\mathcal{U}$ with $\mathcal{U}$ an ultrafilter and
$A\subseteq\mathbb{N}$ is finite, then $\mathbb{N}\setminus A\in\mathcal{F}\subseteq\mathcal{U}$, so
$A\notin\mathcal{U}$; that is, $\mathcal{U}$ is free.  An extension exists by the
ultrafilter lemma.

<!-- lemma-9: body_md -->
\label{lem:diagonal-ultrafilter}
There exist a free ultrafilter $\mathcal{U}$ on $\mathbb{N}$ and a sequence
$(\eta_k)_{k\ge1}\subset(0,1)$ such that:
\begin{enumerate}[(a),itemsep=0.6em,leftmargin=3em,labelsep=1em,topsep=0.8em]
\item for every $r>0$,
\begin{equation}\label{eq:fixed-threshold}
  \{k:\eta_k\le r\}\in\mathcal{U};
\end{equation}
\item there is no $K\in\mathcal{U}$ along which $\eta_k\to0$. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
\end{enumerate}
<!-- end lemma-9 -->
<!-- proof-10: body_md -->
Let $\mathcal{U}$ be any ultrafilter with $\mathcal{F}\subseteq\mathcal{U}$, which is free by
\cref{lem:blockfilter}, and set:
\begin{equation}\label{eq:etadef}
  \eta_k\quad=\quad\frac{1}{i+1}
  \qquad\text{for the unique }i\text{ with }k\in I_i .
\end{equation}
This is unambiguous by \cref{lem:partition}, and
$\eta_k\in\{\tfrac12,\tfrac13,\tfrac14,\ldots\}\subset(0,1)$.  Thus $\eta$ is
constant on each block, and the block values decrease to zero as the block
index grows.

For \eqref{eq:fixed-threshold}, let $r>0$ and choose $M$ with $1/(M+1)\le r$.
Put $T_M=\bigcup_{i\ge M}I_i$.  For $i\ge M$ we have
$I_i\setminus T_M=\varnothing$, so $\varphi(T_M)\supseteq\{i:i\ge M\}$, a
cofinite set; hence $T_M\in\mathcal{F}\subseteq\mathcal{U}$.  Every $k\in T_M$ lies in some
$I_i$ with $i\ge M$ and so has $\eta_k=1/(i+1)\le1/(M+1)\le r$; that is,
$T_M\subseteq\{k:\eta_k\le r\}$, and upward closure of $\mathcal{U}$ gives
\eqref{eq:fixed-threshold}.

For part~(b), suppose $K\in\mathcal{U}$ and $\eta_k\to0$ along $K$.  Fix $i\in\mathbb{N}$ and
apply the definition of convergence along $K$ with $r=1/(i+1)$: there is $N$
such that every $k\in K$ with $k\ge N$ satisfies $\eta_k<1/(i+1)$.  No element
of $I_i$ satisfies that strict inequality, since $\eta$ equals $1/(i+1)$
throughout $I_i$; hence $K\cap I_i\subseteq\{1,\ldots,N-1\}$ is finite.  As $i$
was arbitrary, $I_i\setminus(\mathbb{N}\setminus K)=I_i\cap K$ is finite for every $i$,
so $\varphi(\mathbb{N}\setminus K)=\mathbb{N}$ and $\mathbb{N}\setminus K\in\mathcal{F}\subseteq\mathcal{U}$.
Together with $K\in\mathcal{U}$ this puts $\varnothing$ in $\mathcal{U}$, which is
impossible.

<!-- remark-2: body_md -->
\label{rem:frechet}
Neither part of \cref{lem:diagonal-ultrafilter} survives a more familiar notion
of largeness.  Suppose ``large'' meant ``cofinite,'' so that $\mathcal{U}$ were
replaced by the Fr\'echet filter.  Then part~(a) would say that
$\{k:\eta_k\le r\}$ is cofinite for every $r>0$, which is exactly $\eta_k\to0$,
and part~(b) would fail because $\mathbb{N}$ is itself cofinite.  Nor can the sets in
part~(a) be closed under countable intersection: since $\eta_k>0$ for every $k$,
the sets $\{k:\eta_k\le1/m\}$ have empty intersection, and no filter contains
$\varnothing$.  What the lemma needs is a notion of largeness closed under finite
but not countable intersection --- the same gap between finite and countable
additivity that the counterexample exploits.

The same computation shows that the ultrafilter cannot be left unspecified.
First, a subset of $\mathbb{N}$ belonging to \emph{every} free ultrafilter is cofinite.
For suppose $Y\subseteq\mathbb{N}$ is not cofinite, so that $\mathbb{N}\setminus Y$ is infinite.
Then $\mathbb{N}\setminus Y$ meets every cofinite set, so the family
\[
  \bigl\{\,Z\subseteq\mathbb{N}:Z\supseteq(\mathbb{N}\setminus Y)\cap C
    \mbox{ for some cofinite }C\,\bigr\}
\]
contains no empty set and is closed under intersection and under passage to
supersets; it is therefore a filter, and it contains every cofinite set as well
as $\mathbb{N}\setminus Y$.  Any ultrafilter extending it is free and omits $Y$.  Now
suppose a single sequence $(\eta_k)\subset(0,1)$ satisfied part~(a) of
\cref{lem:diagonal-ultrafilter} against every free ultrafilter.  Then
$\{k:\eta_k\le r\}$ would belong to every free ultrafilter, hence be cofinite,
for every $r>0$; that is $\eta_k\to0$, and part~(b) would then fail against
every free ultrafilter, taking $K=\mathbb{N}$.  So no sequence serves them all, and the
pair $\bigl(\mathcal{U},(\eta_k)\bigr)$ must be selected rather than assumed --- which
is what \cref{lem:blockfilter} and \cref{lem:diagonal-ultrafilter} do.
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}

<!-- remark-3: body_md -->
\label{rem:ppoint}
The choice is constrained further, though nothing below depends on it.  Let
$\left\{B_m\right\}_{m\ge1}$ be a countable family of subsets of $\mathbb{N}$.  A set
$K\subseteq\mathbb{N}$ is a \textbf{pseudo-intersection} of the family if
$K\setminus B_m$ is finite for every $m\ge1$; only \emph{almost} inclusion is
asked, and the genuine intersection $\bigcap_{m\ge1}B_m$ is typically empty
even when a pseudo-intersection is infinite.  A free ultrafilter $\mathcal{U}$ on $\mathbb{N}$
is a \textbf{P-point} if every countable family of $\mathcal{U}$-large sets has a
pseudo-intersection that is itself $\mathcal{U}$-large; this is one of several
equivalent formulations of the notion
\citep[Exercise~7.7, p.~86]{jech2003}.

No P-point admits a sequence of the kind \cref{lem:diagonal-ultrafilter}
produces.  Suppose $\mathcal{U}$ were a P-point and $(\eta_k)\subset(0,1)$ satisfied
part~(a) of that lemma.  The sets $B_m=\left\{k:\eta_k\le1/m\right\}$ are then
$\mathcal{U}$-large for every $m\ge1$, so there is a $\mathcal{U}$-large $K$ with
$K\setminus B_m$ finite for every $m$.  Let $r>0$, choose $m$ with $1/m<r$, and
choose $N$ with $K\setminus B_m\subseteq\{1,\ldots,N-1\}$.  Every $k\in K$ with
$k\ge N$ then lies in $B_m$, so $\eta_k\le1/m<r$.  Hence $\eta_k\to0$ along
$K\in\mathcal{U}$, contradicting part~(b).

Nor is the property available for free.  Whether P-points exist is independent
of the usual axioms of set theory.  They exist under the continuum hypothesis,
since every Ramsey ultrafilter is a P-point and a Ramsey ultrafilter exists when
$2^{\aleph_0}=\aleph_1$ \citep[Theorem~7.8, pp.~76--77]{jech2003}; and there is
a model of $\mathsf{ZFC}$ containing no P-point at all, a result of Shelah
recorded by \citet[p.~89]{jech2003} and published in detail by
\citet{wimmers1982}.  \cref{lem:diagonal-ultrafilter} sidesteps the question,
arranging what is needed outright from \cref{lem:blockfilter} and the
ultrafilter lemma.
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}

<!-- remark-4: body_md -->
\label{rem:diagonal-terminology}
Diagonalization is the standard passage from a family of conditions, each
satisfiable on its own, to a single object satisfying all of them at once: the
diagonal subsequence of analysis, and the countable union of \cref{sec:gap}.
\cref{lem:diagonal-ultrafilter} is the failure of that passage: the bound
$\eta_k\le r$ holds for each $r$ separately, and along no one $\mathcal{U}$-large set
for all $r$ at once.  We therefore speak throughout of resisting
diagonalization, and never of a diagonal sequence.  In analysis a diagonal
sequence is one that the passage succeeds in extracting, which is the opposite
of what \cref{lem:diagonal-ultrafilter} supplies.  The adjective is retained
for two notions only, both of \cref{sec:diagonal}: a \textit{diagonal
condition} is one imposed simultaneously at all thresholds, and so a condition
demanding the very passage that fails here, and a \textit{diagonal
strengthening} of CHT is a strengthening of CHT by such a condition.  Two
clarifications follow: the first concerns a notion that must be kept distinct
from this one, the second a
notion that is exactly
this one and that our $\mathcal{U}$ must fail.

First, a filter on a cardinal $\kappa$ is \textit{normal} when it is closed under
diagonal intersections $\bigtriangleup_{\alpha<\kappa}X_\alpha
=\{\xi<\kappa:\xi\in\bigcap_{\alpha<\xi}X_\alpha\}$
\citep[pp.~91, 95]{jech2003}, a notion developed for regular uncountable
$\kappa$ and, in the measure-theoretic form, for measurable cardinals
\citep[p.~131]{jech2003}.  Our $\mathcal{U}$ lives on $\mathbb{N}$ and is not even closed
under countable intersections, so it is not normal in that sense and nothing
here bears on normality.

Second, and more to the point, the P-point property of \cref{rem:ppoint} --- a
pseudo-intersection in $\mathcal{U}$ for every countable family of $\mathcal{U}$-large sets
--- is diagonalization of that family in the set-theoretic sense.  The
ultrafilter we need must \emph{fail} to have it: part~(b) of
\cref{lem:diagonal-ultrafilter} is precisely a failure of pseudo-intersection.
So ``diagonal'' attributes no diagonalization property to $\mathcal{U}$, and by
\cref{rem:ppoint} no P-point would serve in its place.

  \quad\hbox{{${\scriptstyle\blacksquare}$}}

\label{lem:atom}
Let $\mu$ be a countably additive Borel probability on $\Omega$, let $v$ be a
probability function on $\mathcal{B}$, let $\beta\in(0,1)$, and put
$P_v=\beta v+(1-\beta)\mu$.  Let $d\in\Omega$ and $n_0\in\mathbb{N}$, and suppose
\begin{enumerate}[(i),itemsep=.35em,leftmargin=2.8em,labelsep=.8em,topsep=.5em]
\item $\mu(\{d\})>0$, and
\item $v\bigl([d\mathord{\upharpoonright} n]\bigr)=0$ for every $n\ge n_0$.
\end{enumerate}
Then for every $A\in\mathcal{B}$ with $d\notin A$ and every $n\ge n_0$:
\[
  P_v\bigl(A\,|\,[d\mathord{\upharpoonright} n]\bigr)
  \;\le\;\frac{\gamma_n}{\mu(\{d\})+\gamma_n},
  \qquad\text{where }
  \gamma_n=\mu\bigl([d\mathord{\upharpoonright} n]\setminus\{d\}\bigr).
\]
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end lemma-10 -->
<!-- proof-11: body_md -->
Let $n\ge n_0$ and let $A\in\mathcal{B}$ with $d\notin A$.  Then
$A\cap[d\mathord{\upharpoonright} n]\subseteq[d\mathord{\upharpoonright} n]\setminus\{d\}$, so
$\mu(A\cap[d\mathord{\upharpoonright} n])\le\gamma_n$ by monotonicity; and
$v(A\cap[d\mathord{\upharpoonright} n])\le v([d\mathord{\upharpoonright} n])=0$ by (ii) and monotonicity.  Hence
the numerator of $P_v(A\,|\,[d\mathord{\upharpoonright} n])$ is at most $(1-\beta)\gamma_n$.
Since $\{d\}$ and $[d\mathord{\upharpoonright} n]\setminus\{d\}$ partition $[d\mathord{\upharpoonright} n]$,
the denominator satisfies
$P_v([d\mathord{\upharpoonright} n])\ge(1-\beta)\mu([d\mathord{\upharpoonright} n])
=(1-\beta)(\mu(\{d\})+\gamma_n)$.  Dividing gives the bound.

<!-- lemma-11: body_md -->
\label{lem:u-properties}
The set function $u$ is a two-valued finitely additive probability.  Moreover,
for all Borel $A,B$:
\begin{enumerate}[(a),itemsep=0.6em,leftmargin=3em,labelsep=1em,topsep=0.8em]
\item $u(A\cap B)=u(A)u(B)$;
\item if $u(A)=u(B)$, then $u(A\mathbin{\triangle} B)=0$. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
\end{enumerate}
<!-- end lemma-11 -->
<!-- proof-12: body_md -->
Normalization follows from $\mathbb{N}\in\mathcal{U}$.  If $A$ and $B$ are disjoint, the index
sets $I_A=\{k:d_k\in A\}$ and $I_B=\{k:d_k\in B\}$ are disjoint.  An ultrafilter
can contain at most one of them.  If it contains neither, then it contains both
complements and hence the complement of $I_A\cup I_B$; thus it does not contain
the union.  Therefore $u(A\cup B)=u(A)+u(B)$.

For part~(a), if $u(A)=u(B)=1$, then $I_A\cap I_B\in\mathcal{U}$; if either value is
zero, then $I_A\cap I_B$ is contained in a set outside $\mathcal{U}$ and hence is
outside $\mathcal{U}$.  For part~(b), if both values are one then
$I_A\cap I_B\in\mathcal{U}$, while if both are zero then
$I_A^c\cap I_B^c\in\mathcal{U}$.  In either case the complement of the index set for
$A\mathbin{\triangle} B$ belongs to $\mathcal{U}$.

<!-- proposition-2: body_md -->
\label{prop:basic-properties}
The set function $P$ is a merely finitely additive probability.  Moreover:
\begin{equation}\label{eq:clopen-agreement}
  P(C)\;=\; R(C) \qquad \mbox{for all }C\in\mathcal{C},
\end{equation}
and every nonempty cylinder has strictly positive $P$-probability.
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end proposition-2 -->
<!-- proof-13: body_md -->
\emph{Failure of countable additivity.}
Let $D$ be the set of distinguished paths \eqref{eq:Dset}.  Since
$\{k:d_k\in D\}=\mathbb{N}\in\mathcal{U}$, we have $u(D)=1$; and $u(\{d_k\})=0$ for every $k$,
because a free ultrafilter contains no singleton.  Thus:
\begin{align*}
  P(D)
    \quad&=\quad\alpha\;+\;(1-\alpha)\sum_{k=1}^{\infty}t_k(1-\eta_k),\\
  \sum_{k=1}^{\infty}P(\{d_k\})
    \quad&=\quad(1-\alpha)\sum_{k=1}^{\infty}t_k(1-\eta_k).
\end{align*}
The two quantities differ by $\alpha$, so $P$ is not countably additive.

\emph{Agreement on clopen events.}
If $C$ is clopen, continuity of $\mathbf{1}_C$ and the convergence $d_k\to z$ imply
that $\mathbf{1}_C(d_k)$ is eventually equal to $\mathbf{1}_C(z)$.  Every cofinite set belongs
to $\mathcal{U}$, so:
\[
  u(C)\;=\;\delta_z(C).
\]
Substitution into \eqref{eq:P} and \eqref{eq:R-companion} gives
\eqref{eq:clopen-agreement}.

\emph{Positivity.}
Let $B$ be a nonempty cylinder.  If $z\in B$, then
$P(B)=R(B)\ge\alpha$.  If $z\notin B$, there is a unique $k$ such that
$B\subseteq S_k$.  Since $\nu_k(B)>0$, it follows that:
\[
  P(B)\;=\;R(B)\;\ge\;(1-\alpha)t_k\eta_k\nu_k(B)>0.
\]

<!-- corollary-1: body_md -->
\label{cor:shell}
Let $\mathcal{V}$ be a free ultrafilter on $\mathbb{N}$, let $u_{\mathcal{V}}$ be defined from
$\mathcal{V}$ by \eqref{eq:u}, let $(\lambda_k)_{k\ge1}\subset(0,1)$, and put:
\[
  R_0^{\lambda}
  \;=\;\sum_{k=1}^{\infty}t_k
    \bigl((1-\lambda_k)\delta_{d_k}+\lambda_k\nu_k\bigr),
  \qquad
  P^{\lambda}\;=\;\alpha u_{\mathcal{V}}+(1-\alpha)R_0^{\lambda}
  \quad\mbox{with }\alpha\in(0,1).
\]
Then for every $k\in\mathbb{N}$, every $A\in\mathcal{B}$ with $d_k\notin A$, and every
$n\ge k$:
\begin{equation}\label{eq:shell-estimate}

  P'\;=\;\alpha u_{\mathcal{V}}+(1-\alpha)\mu_0,
  \qquad

\end{equation}

\begin{equation}\label{eq:Asharp}
 A^{\sharp}\quad=\quad
 \begin{cases}
   A\cup\{z\}&\mbox{if }u_{\mathcal{V}}(A)=1;\\
   A\setminus\{z\}&\mbox{if }u_{\mathcal{V}}(A)=0,
 \end{cases}
\end{equation}
the dependence on $\mathcal{V}$ being suppressed in the notation.  Then, for every

  P'(A\cap C)\;=\;R'(A^{\sharp}\cap C);

\begin{equation}\label{eq:posterior-companion}

\end{equation}
for every $\omega\in\Omega$ and every $n\in\mathbb{N}$.  Taking $\mathcal{V}=\mathcal{U}$ and
$\mu_0=R_0$ gives $P'=P$ and $R'=R$.

\]

\end{equation}

\eqref{eq:open-agreement}.

<!-- remark-5: body_md -->
\label{rem:nielsen-thm5}
Theorem~5 \citep[p.~411]{nielsen2021convergence} makes two claims: that there
are uncountably many probability functions satisfying \eqref{eq:posi} with the
approximation property that do not converge to the truth almost surely, and
that they therefore do not have the CHT property.  The second is inferred from
the first ``by Theorem~2 and Lemma~3''
\citep[p.~411]{nielsen2021convergence} --- that is, from the sufficiency
direction of \eqref{eq:characterization}, since it is only that direction which
yields, for a probability function with the approximation property, that CHT
implies almost-sure convergence.  \cref{thm:counterexample} removes that
direction, so the published derivation of the second claim lapses.
\cref{cor:nielsen-thm5} restores it for the probability functions his own
example yields, which suffices for a claim of existential form, and does so
without appeal to Theorem~2; his first claim is his own and we do not revisit
it.  The same inference is what supports his statement
that the approximation property does not imply the CHT property, one direction
of the independence of the two criteria; that direction is therefore restored as
well, while the other rests on Theorem~4, whose proof
\cref{sec:published-proof} shows to be unavailable.

The construction is his.  His proof takes $D$ to be any infinite discrete
subspace of $\Omega$ with $Q(D\cup L_D)=0$, where $L_D$ is the set of limit
points of $D$, and offers as its example the set of paths $\omega_n$ with
$\omega_n(i)=\mathbf{1}_{\{n\}}(i)$ for $i\in\mathbb{N}$ --- the distinguished paths $d_n$ of
\eqref{eq:shellsandpaths} --- whose only limit point is the constantly zero
path, our $z$.  For that $D$ he takes a nonprincipal ultrafilter $\mathcal{W}$ on
$\mathcal{B}$ with
$D\in\mathcal{W}$, meaning that $\mathcal{W}$ is a filter on $\mathcal{B}$ containing exactly one
of $A$ and $\Omega\setminus A$ for each $A\in\mathcal{B}$ and containing no
singleton; the two-valued probability $F\mapsto\mathbf{1}_{\mathcal{W}}(F)$; and the mixture
$\alpha\,\mathbf{1}_{\mathcal{W}}+(1-\alpha)Q$.  Given such a $\mathcal{W}$, put:
\[
  \mathcal{V}\quad=\quad\bigl\{\,S\subseteq\mathbb{N}:\{d_k:k\in S\}\in\mathcal{W}\,\bigr\}.
\]
Then $\mathcal{V}$ is a free ultrafilter on $\mathbb{N}$, and
$\mathbf{1}_{\mathcal{W}}=u_{\mathcal{V}}$ on $\mathcal{B}$, so his mixture is the $P_\alpha$ of
\cref{cor:nielsen-thm5}.

For the first assertion, the map $S\mapsto\{d_k:k\in S\}$ carries subsets of
$\mathbb{N}$ to subsets of $D$ preserving inclusions and intersections, so $\mathcal{V}$ is
closed upward and under intersections; $\mathbb{N}\in\mathcal{V}$ because $D\in\mathcal{W}$, and
$\varnothing\notin\mathcal{V}$ because $\varnothing\notin\mathcal{W}$.  For each $S$ the
sets $\{d_k:k\in S\}$ and $\{d_k:k\notin S\}$ are disjoint with union $D$, and
$D\in\mathcal{W}$, so exactly one of them lies in $\mathcal{W}$: whichever of
$\{d_k:k\in S\}$ and its complement in $\Omega$ lies in $\mathcal{W}$, intersecting
with $D$ leaves a member of $\mathcal{W}$; and not both of them, since a filter
contains no two disjoint sets.  Hence exactly one of $S$ and
$\mathbb{N}\setminus S$ lies in $\mathcal{V}$.  And if a finite $S$ had
$\{d_k:k\in S\}\in\mathcal{W}$, then splitting that finite set repeatedly and keeping
at each step the half that lies in $\mathcal{W}$ would put a singleton in $\mathcal{W}$,
contradicting nonprincipality; so $\mathcal{V}$ contains no finite set.  For the
second assertion, $A\in\mathcal{W}$ if and only if $A\cap D\in\mathcal{W}$, since $\mathcal{W}$
is closed upward and under intersection with $D\in\mathcal{W}$; and
$A\cap D=\{d_k:k\in\{k:d_k\in A\}\}$, so
$\mathbf{1}_{\mathcal{W}}(A)=1$ exactly when $\{k:d_k\in A\}\in\mathcal{V}$, which by
\eqref{eq:u} is exactly when $u_{\mathcal{V}}(A)=1$.

\label{prop:not-au}

\begin{equation}\label{eq:error-eta}
  P_k(H)(d_k)=P(H\,|\, S_k)=\eta_k,
\end{equation}
where the subscript $k$ on $P_k$ denotes observation time.

\end{equation}

Lemma~\ref{lem:diagonal-ultrafilter}(b).

<!-- proposition-7: body_md -->
\label{prop:cht-as}
Let $\mathcal{V}$ be a free ultrafilter on $\mathbb{N}$, let $u_{\mathcal{V}}$ be defined from
$\mathcal{V}$ by \eqref{eq:u}, let $\mu_0$ be a countably additive Borel probability
on $\Omega$ with $\mu_0(\{z\})=0$, let $\alpha\in(0,1)$, and suppose that
$P'=\alpha u_{\mathcal{V}}+(1-\alpha)\mu_0$ satisfies \eqref{eq:posi}.  If $P'$ has
the CHT property, then the posteriors of $P'$ converge to the truth almost
surely for every Borel event.
 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end proposition-7 -->
<!-- proof-21: body_md -->
By \cref{prop:noatom}, the set
$J=\{k\in\mathbb{N}:\mu_0(\{d_k\})>0\}$ belongs to $\mathcal{V}$: otherwise its complement
$J_0$ would be $\mathcal{V}$-large by the ultrafilter property, and CHT would fail.

Fix $A\in\mathcal{B}$, let $A^{\sharp}$ be as in \eqref{eq:Asharp}, and put
$R'=\alpha\delta_z+(1-\alpha)\mu_0$ as in \eqref{eq:companion-pair}.  The
measure $R'$ gives every nonempty cylinder positive probability: if $z\in B$
then $R'(B)\ge\alpha$, while if $z\notin B$ then $B\subseteq S_k$ for a unique
$k$, so $u_{\mathcal{V}}(B)=0$ --- no $d_j$ other than $d_k$ meets $B$, and a free
ultrafilter contains no finite set --- and \eqref{eq:posi} gives
$0<P'(B)=(1-\alpha)\mu_0(B)\le R'(B)$.  So
\cref{lem:standard-facts}(a) applies to $R'$, and with
\eqref{eq:posterior-companion} it gives
\[
  P'_n(A)\;=\;R'\bigl(A^{\sharp}\,|\,[\,\cdot\mathord{\upharpoonright} n]\bigr)
  \;\longrightarrow\;\mathbf{1}_{A^{\sharp}}
  \qquad R'\text{-almost surely}.
\]
Let $N_A\in\mathcal{B}$ be an $R'$-null event off which this convergence holds.  An
$R'$-null event contains no point of positive $R'$-measure, and
$R'(\{z\})=\alpha>0$ while $R'(\{d_k\})=(1-\alpha)\mu_0(\{d_k\})>0$ for
$k\in J$; hence $z\notin N_A$ and $d_k\notin N_A$ for every $k\in J$.
Therefore $\{k:d_k\in N_A\}\subseteq\mathbb{N}\setminus J$, which is not $\mathcal{V}$-large,
so $u_{\mathcal{V}}(N_A)=0$; and $(1-\alpha)\mu_0\le R'$ gives $\mu_0(N_A)=0$.  Hence
$P'(N_A)=0$.

Off $N_A$ the posteriors converge to $\mathbf{1}_{A^{\sharp}}$.  Since
$A\mathbin{\triangle} A^{\sharp}\subseteq\{z\}$ and
$P'(\{z\})=\alpha u_{\mathcal{V}}(\{z\})+(1-\alpha)\mu_0(\{z\})=0$ --- the first term
vanishing because no $d_k$ equals $z$, so that
$\{k:d_k\in\{z\}\}=\varnothing\notin\mathcal{V}$, and the second by hypothesis ---
the posteriors converge to $\mathbf{1}_A$ off the event
$N_A\cup\{z\}$, which has $P'$-probability zero by finite additivity.

<!-- remark-6: body_md -->
\label{rem:threshold-diagonal}
\cref{lem:diagonal-ultrafilter}(a) is exactly what the CHT verification uses:
for each fixed $\delta$, the set of indices $k$ with $\eta_k\le\delta$ is
$\mathcal{U}$-large, and that suffices to place the infinite hitting union on the
correct side of the ultrafilter.  Almost-uniform convergence would require one
$\mathcal{U}$-large set of indices on which the corresponding error bounds hold
eventually for every $\delta$ at once, and
\cref{lem:diagonal-ultrafilter}(b) rules that out.  This is the separation
between \eqref{eq:weak-quantifiers} and \eqref{eq:au-quantifiers} in concrete
form: the exceptional event may be chosen after the accuracy threshold but not

<!-- remark-7: body_md -->
\label{rem:corollary}
\cref{prop:as} shows that the posteriors of $P$ do converge to the truth almost
surely.  \cref{thm:counterexample} therefore does not refute Nielsen's
Corollary~1 \citep[p.~408]{nielsen2021convergence}, which draws the almost-sure
conclusion from approximation and CHT.  What it does remove is that corollary's
only published proof, which applies Theorem~2 and then passes from
almost-uniform to almost-sure convergence.  Whether approximation and CHT
together imply almost-sure convergence to the truth for an arbitrary
probability function is, so far as we know, open.  What
\cref{prop:cht-as} settles is that no counterexample can be found in the family
to which the construction of \cref{sec:construction} belongs: for a mixture of
a two-valued ultrafilter probability read along the distinguished paths with a
countably additive probability giving $z$ probability zero, subject to
\eqref{eq:posi}, CHT alone --- without the approximation property --- already
implies almost-sure convergence to the truth.  The almost-sure convergence of
$P$ is thus forced by its having CHT, not an artifact of the particular weights
chosen.

\label{prop:open-uniqueness}
Let $\mu$ be a countably additive Borel probability on $\Omega$.  If a
probability function $P$ on $\mathcal{B}$ agrees with $\mu$ on every open set, then
$P=\mu$ on every Borel set. 
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill
  \quad\hbox{{${\scriptstyle\blacksquare}$}}
<!-- end proposition-8 -->
<!-- proof-24: body_md -->
Agreement on open sets implies agreement on closed sets by complementation.  Let
$B\in\mathcal{B}$ and $\varepsilon>0$.  By \cref{lem:regularity}, choose a closed $F$ and an
open $G$ such that:
\[
  F\subseteq B\subseteq G,
  \qquad \mu(G\setminus F)<\varepsilon.
\]
Monotonicity and the assumed agreement give
\[
  \mu(F)=P(F)\le P(B)\le P(G)=\mu(G).
\]
The interval $[\mu(F),\mu(G)]$ has length less than $\varepsilon$ and contains
$\mu(B)$ as well, so
$\left|P(B)-\mu(B)\right|<\varepsilon$.  Letting $\varepsilon\downarrow0$ yields $P(B)=\mu(B)$.

<!-- theorem-2: body_md -->
\label{thm:positive}
For every $\alpha\in(0,1)$, $P_\alpha^{\theta}$ is merely finitely additive,
satisfies \eqref{eq:posi}, and converges to the truth almost uniformly.  Distinct
values of $\alpha$ give distinct probability functions.  Consequently there are
uncountably many merely finitely additive probability functions satisfying
\eqref{eq:posi} and converging to the truth almost uniformly.

<!-- proof-26: body_md -->
Write $D=\{d_k:k\in\mathbb{N}\}$ as in \eqref{eq:Dset}.
Finite additivity, failure of countable additivity, positivity, and clopen
approximation are proved exactly as in \cref{prop:basic-properties}
and \cref{prop:approximation}.  We prove almost-uniform convergence for an
arbitrary $A\in\mathcal{B}$ and $\varepsilon>0$.

\smallskip
\noindent\emph{Step 1: a countably additive version of the posterior process.}
Put $a=u(A)$ and define $A^{\sharp}$ by \eqref{eq:Asharp}.
\cref{lem:companion}, applied with $\mathcal{V}=\mathcal{U}$ and $\mu_0=R_0^{\theta}$,
gives
\begin{equation}\label{eq:theta-companion}
  P_{\alpha,n}^{\theta}(A)(\omega)
  =R_\alpha^{\theta}
     \bigl(A^{\sharp}\,|\,[\omega\mathord{\upharpoonright} n]\bigr),
\end{equation}
where $P_{\alpha,n}^{\theta}(A)$ denotes the time-$n$ posterior generated by
$P_\alpha^\theta$.  By Lemma~\ref{lem:standard-facts}(a), the right-hand side
converges to $\mathbf{1}_{A^{\sharp}}$ $R_\alpha^\theta$-almost surely and hence
$R_0^\theta$-almost surely.  Lemma~\ref{lem:standard-facts}(b) supplies a Borel
event $E_0$ such that
\begin{equation}\label{eq:E0}
  R_0^{\theta}(E_0)<\frac{\varepsilon}{4(1-\alpha)}
\end{equation}
and the convergence in \eqref{eq:theta-companion} is uniform on $E_0^c$.

\smallskip
\noindent\emph{Step 2: choose one exceptional event with small $P_\alpha^\theta$-mass.}
Let
\[
  M=\{k:\mathbf{1}_A(d_k)\ne a\}.
\]
The set $M$ is not in $\mathcal{U}$: it is the complement of
$\{k:d_k\in A\}$ when $a=1$, and equals that set when $a=0$.  Choose $K_0$
so large that
\begin{equation}\label{eq:tail-small}
  \sum_{k>K_0}t_k<\frac{\varepsilon}{4(1-\alpha)}
\end{equation}
and set
\begin{equation}\label{eq:E-positive}
  E=(E_0\setminus D)\cup\{z\}
    \cup\{d_k:k>K_0,\ k\in M\}.
\end{equation}
The first two pieces contain no distinguished point $d_k$, and the index set of
the third piece is contained in $M\notin\mathcal{U}$.  Hence $u(E)=0$.  Moreover,
\[
 R_0^\theta(E)
 \le R_0^\theta(E_0)+\sum_{k>K_0}t_k
 <\frac{\varepsilon}{2(1-\alpha)}.
\]
Therefore
\[
  P_\alpha^\theta(E)=(1-\alpha)R_0^\theta(E)<\frac{\varepsilon}{2}<\varepsilon.
\]

\noindent\emph{Step 3: uniform convergence away from the distinguished points.}

$\mathbf{1}_{A^{\sharp}}(\omega)=\mathbf{1}_A(\omega)$.  Uniform convergence on these points
therefore follows from the definition of $E_0$.

\smallskip
\noindent\emph{Step 4: uniform convergence on the remaining $d_k$.}

\[
  q_n=P_\alpha^{\theta}(A\,|\, Z_n).
\]
Exactly as in the proof of \cref{prop:cht}, $q_n\to a$.  For $n\ge k$,

$d_k\in A$, gives
\begin{equation}\label{eq:theta-shell-bound}
  \left|P_{\alpha,n}^{\theta}(A)(d_k)-\mathbf{1}_A(d_k)\right|\le\theta_k.
\end{equation}
For each fixed $k$, the cylinders $[d_k\mathord{\upharpoonright} n]$ decrease to $\{d_k\}$.
Continuity from above of $R_\alpha^\theta$ therefore gives
\begin{equation}\label{eq:fixed-dk}
  P_{\alpha,n}^{\theta}(A)(d_k)\longrightarrow\mathbf{1}_A(d_k),
\end{equation}
because $R_\alpha^\theta(\{d_k\})>0$.

$k\ge K_1$, and choose $N_0$ so that $|q_n-a|<r$ for $n\ge N_0$.  By \eqref{eq:fixed-dk}, there is an integer
$N_1$ such that
$\bigl|P_{\alpha,n}^{\theta}(A)(d_k)-\mathbf{1}_A(d_k)\bigr|<r$ for every $n\ge N_1$
and every $k<K_1$.  Let $N=\max\{N_0,N_1\}$.  If $d_k\in E^c$ and $n\ge N$, then one of
three cases applies:
\begin{enumerate}[(i)]
\item If $k<K_1$, the error is less than $r$ by the choice of $N_1$.
\item If $k\ge K_1$ and $n\ge k$, the error is less than $r$ by
      \eqref{eq:theta-shell-bound}.
\item If $k\ge K_1$ and $n<k$, then $k>K_0$ and $d_k\notin E$ imply
      $\mathbf{1}_A(d_k)=a$.  Since $[d_k\mathord{\upharpoonright} n]=Z_n$, the error is
      $|q_n-a|<r$.
\end{enumerate}
Thus the convergence is uniform on the distinguished points remaining in $E^c$
and, with Step~3, on all of $E^c$.

\[
  P_\alpha^\theta(D)=c+\alpha(1-c),
\]
which is strictly increasing in $\alpha$.  Hence distinct values of $\alpha$ give
distinct probability functions, and the family is uncountable.

Let
\[
    z=(0,0,0,\ldots)\in\Omega
\]
and, for $k\in\mathbb{N}$, let
\[
    d_k
    =
    0^{k-1}10^{\infty},
\]
so that $d_k$ has a $1$ in coordinate $k$ and a $0$ in every other coordinate.
Thus $d_k\to z$ in the Cantor topology.

Let $Q$ again be the fair-coin measure of \cref{sec:geometry}, and define the
countably additive probability
\begin{equation}
\label{eq:app-mu}
    \mu
    :=
    \frac{1}{2} Q
    +
    \frac{1}{2}\sum_{k=1}^{\infty}2^{-k}\delta_{d_k}.
\end{equation}
Then
\[
    \mu(\{z\})=0,
    \qquad
    \mu(\{d_k\})=2^{-k-1}>0
\]
for every $k$.

Let $\mathcal{U}$ be a free ultrafilter on $\mathbb{N}$. Define
$u:\mathcal{B}\to\{0,1\}$ by
\begin{equation}
\label{eq:app-u}
    u(A)
    :=
    \begin{cases}
       1,
       &
       \text{if }\{k\in\mathbb{N}:d_k\in A\}\in\mathcal{U},
       \\[1mm]
       0,
       &
       \text{if }\{k\in\mathbb{N}:d_k\in A\}\notin\mathcal{U}.
    \end{cases}
\end{equation}
Equivalently, $u(A)$ is the $\mathcal{U}$-limit \eqref{eq:ulimit} of the sequence of
truth values of $A$ along the distinguished paths:
\[
    u(A)=\lim_{k\to\mathcal{U}}\mathbf{1}_A(d_k).
\]
By \cref{lem:u-properties}, whose proof uses nothing about $\mathcal{U}$ beyond its
being an ultrafilter, $u$ is a two-valued probability function on $\mathcal{B}$.

Now put
\begin{equation}
\label{eq:app-P}
    P
    :=
    \frac{1}{2}u+\frac{1}{2}\mu.
\end{equation}

\medskip
\noindent
\emph{Step 1: $P$ satisfies \eqref{eq:posi}.}
Since $\mu\ge \frac{1}{2} Q$, every nonempty cylinder $C$ satisfies
\[
    P(C)
    \ge
    \frac{1}{2}\mu(C)
    \ge
    \frac{1}{4} Q(C)
    >0.
\]
Hence all conditional probabilities appearing below are well defined.

\medskip
\noindent
\emph{Step 2: $P$ is not countably additive.}
Let
\[
    D:=\{d_k:k\in\mathbb{N}\}.
\]
Since every $d_k$ belongs to $D$,
\[
    u(D)=1,
\]
whereas $Q(D)=0$ and hence
\[
    \mu(D)=\frac{1}{2}.
\]
Therefore
\begin{equation}
\label{eq:app-PD}
    P(D)
    =
    \frac{1}{2}+\frac{1}{4}
    =
    \frac{3}{4}.
\end{equation}

On the other hand, a free ultrafilter contains no singleton, so
\[
    u(\{d_k\})=0
\]
for every $k$. It follows from
\eqref{eq:app-mu}--\eqref{eq:app-P} that
\[
    P(\{d_k\})
    =
    2^{-k-2}.
\]
Consequently,
\[
    \sum_{k=1}^{\infty}P(\{d_k\})
    =
    \frac{1}{4}
    \ne
    \frac{3}{4}
    =
    P(D).
\]
Thus $P$ is not countably additive.

\medskip
\noindent
\emph{Step 3: on finite-information events, $u$ behaves like point mass at
$z$.}
If $C$ is clopen, then membership in $C$ is determined by finitely many
coordinates. Because $d_k\to z$, there exists $K$ such that
\[
    \mathbf{1}_C(d_k)=\mathbf{1}_C(z)
    \qquad(k\ge K).
\]
Since every cofinite subset of $\mathbb{N}$ belongs to the free ultrafilter
$\mathcal{U}$,
\begin{equation}
\label{eq:app-u-clopen}
    u(C)=\mathbf{1}_C(z)
    \qquad(C\text{ clopen}).
\end{equation}

Define the countably additive probability
\begin{equation}
\label{eq:app-R}
    R
    :=
    \frac{1}{2}\delta_z+\frac{1}{2}\mu.
\end{equation}
Equation~\eqref{eq:app-u-clopen} implies
\begin{equation}
\label{eq:app-P-R-clopen}
    P(C)=R(C)
    \qquad
    \text{for every clopen }C.
\end{equation}

\medskip
\noindent
\emph{Step 4: every $P$-posterior is an ordinary $R$-posterior for a
slightly modified event.}
Fix $A\in\mathcal{B}$, and define
\begin{equation}
\label{eq:app-Asharp}
    A^\sharp
    :=
    \begin{cases}
       A\cup\{z\},&u(A)=1,\\
       A\setminus\{z\},&u(A)=0.
    \end{cases}
\end{equation}
Thus
\begin{equation}
\label{eq:app-Asharp-diff}
    A^\sharp\mathbin{\triangle} A\subseteq\{z\}
\end{equation}
and
\begin{equation}
\label{eq:app-Asharp-z}
    \mathbf{1}_{A^\sharp}(z)=u(A).
\end{equation}

The set function $u$ is multiplicative:
\begin{equation}
\label{eq:app-u-multiplicative}
    u(A\cap B)=u(A)u(B)
    \qquad(A,B\in\mathcal{B}).
\end{equation}
Indeed, if either $u(A)$ or $u(B)$ is zero, the corresponding index set does
not belong to $\mathcal{U}$, and neither does its intersection with the other;
if both are one, their intersection belongs to $\mathcal{U}$.

Let $C$ be clopen. By
\eqref{eq:app-u-clopen},
\eqref{eq:app-Asharp-z}, and
\eqref{eq:app-u-multiplicative},
\begin{equation}
\label{eq:app-u-intersection}
    u(A\cap C)
    =
    u(A)\mathbf{1}_C(z)
    =
    \delta_z(A^\sharp\cap C).
\end{equation}
Moreover, $\mu(\{z\})=0$, so
\eqref{eq:app-Asharp-diff} gives
\begin{equation}
\label{eq:app-mu-intersection}
    \mu(A\cap C)
    =
    \mu(A^\sharp\cap C).
\end{equation}
Combining
\eqref{eq:app-P},
\eqref{eq:app-R},
\eqref{eq:app-u-intersection}, and
\eqref{eq:app-mu-intersection}, we obtain
\begin{equation}
\label{eq:app-joint-identity}
    P(A\cap C)
    =
    R(A^\sharp\cap C).
\end{equation}
Together with \eqref{eq:app-P-R-clopen}, this implies
\begin{equation}
\label{eq:app-conditional-identity}
    P(A\,|\, C)
    =
    R(A^\sharp\,|\, C)
\end{equation}
whenever $C$ is a nonempty cylinder.

In particular,
\begin{equation}
\label{eq:app-posterior-identity}
    P_n(A)(\omega)
    =
    R(A^\sharp\,|\,[\omega\mathord{\upharpoonright} n]).
\end{equation}

\medskip
\noindent
\emph{Step 5: the posteriors converge to the truth $P$-almost surely.}
Since $R$ is countably additive and gives every nonempty cylinder positive
probability, \cref{lem:standard-facts}(a) applied to $R$ in place of the
measure named there, and with $B=A^\sharp$, yields an $R$-null event $N_A$ such
that
\begin{equation}
\label{eq:app-R-convergence}
    R(A^\sharp\,|\,[\omega\mathord{\upharpoonright} n])
    \longrightarrow
    \mathbf{1}_{A^\sharp}(\omega)
    \qquad
    \text{for every }\omega\notin N_A.
\end{equation}

Because
\[
    R
    =
    \frac{1}{2}\delta_z+\frac{1}{2}\mu,
\]
the equality $R(N_A)=0$ implies
\[
    z\notin N_A
    \qquad\text{and}\qquad
    \mu(N_A)=0.
\]
Every $d_k$ has strictly positive $\mu$-mass, so $\mu(N_A)=0$ also implies
\[
    d_k\notin N_A
    \qquad
    \text{for every }k.
\]
Hence
\[
    \{k:d_k\in N_A\}=\varnothing,
\]
and therefore
\[
    u(N_A)=0.
\]
Thus
\begin{equation}
\label{eq:app-P-null}
    P(N_A)=0.
\end{equation}

Also,
\[
    u(\{z\})=0,
    \qquad
    \mu(\{z\})=0,
\]
so
\begin{equation}
\label{eq:app-z-null}
    P(\{z\})=0.
\end{equation}

By \eqref{eq:app-Asharp-diff},
\[
    \mathbf{1}_{A^\sharp}
    =
    \mathbf{1}_A
    \qquad
    \text{on }\Omega\setminus\{z\}.
\]
Combining
\eqref{eq:app-posterior-identity},
\eqref{eq:app-R-convergence},
\eqref{eq:app-P-null}, and
\eqref{eq:app-z-null}, we obtain
\[
    P_n(A)(\omega)
    \longrightarrow
    \mathbf{1}_A(\omega)
\]
for every
\[
    \omega\notin N_A\cup\{z\},
\]
and, by finite additivity,
\[
    P(N_A\cup\{z\})=0.
\]
Therefore
\[
    P_n(A)\longrightarrow \mathbf{1}_A
    \qquad P\text{-almost surely}.
\]
Since $A\in\mathcal{B}$ was arbitrary, $P$ converges to the truth almost
surely for every event, even though $P$ is not countably additive.
