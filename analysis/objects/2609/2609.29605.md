<!-- hypothesis-1: body_md -->
\label{hyp:H}
There is a constant $C>0$ such that
\[
|\beta(\mathbf{m})|\;\le\;(p-1)!\;C^{\,p-1}\qquad\text{whenever } |\mathbf{m}|=p\ge 2 .
\]
<!-- end hypothesis-1 -->
<!-- definition-1: body_md -->
\label{def:admissible}
A pair $(k,\vartheta)$ with $k\in\mathbb{N}$, $k\ge1$, and $\vartheta\in(0,1)$ is called \emph{admissible} if
\[
\vartheta^{*}:=\Big(1+\tfrac{1}{32k}\Big)\,2^{1/(2k)}\,\vartheta\;<\;1 .
\]
We also write $\vartheta_{k}:=\vartheta\,4^{-1/(2k)}$ and
\[
\Lambda(t):=\sum_{q\ge0}(q+1)^{3}t^{q}=\frac{1+4t+t^{2}}{(1-t)^{4}},
\qquad
G(t):=\sum_{p\ge2}p\binom{p+3}{3}t^{p-1}=4\big[(1-t)^{-5}-1\big] .
\]
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Let $C>0$ and let $\beta:\mathbb{Z}_{+}^{4}\to\mathbb{R}$ satisfy Hypothesis \ref{hyp:H}. Let $(k,\vartheta)$ be an
admissible pair, let $\nu_{k}$, $T(\mathbf{c})=\vartheta/(C\nu_{k}(\mathbf{c}))$ and $\mathcal{S}=\mathcal{S}_{k,\vartheta}$ be as above,
and set
\[
x:=C\|\mathbf{c}\|_{\infty},\qquad
\mathcal{D}:=\big\{\mathbf{c}\in\mathbb{R}^{4}:\ C\,\nu_{k}(\mathbf{c})<\vartheta\big\} .
\]
Then $\mathcal{D}$ is a nonempty, open, bounded, star--shaped neighbourhood of the origin containing the
$\ell^{1}$--ball $\{\|\mathbf{c}\|_{1}<\vartheta/C\}$, and the following hold.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2\parindent]
\item \textup{(Well--definedness.)} For every $\mathbf{c}\in\mathbb{R}^{4}$ the defining integral converges
absolutely and equals the absolutely convergent series
$\mathcal{S} B(\mathbf{c})=\sum_{p\ge2}P\big(p,T(\mathbf{c})\big)b_{p}(\mathbf{c})$, where
$P(p,T)=\Pr[\mathrm{Poi}(T)\ge p]=\gamma(p,T)/\Gamma(p)$. The map $\beta\mapsto\mathcal{S} B$ is $\mathbb{R}$--linear,
$\mathcal{S} B$ is real--valued, and $|\mathcal{S} B(\mathbf{c})|\le 10\,\Lambda(\vartheta)\,C\|\mathbf{c}\|_{\infty}^{2}$
on all of $\mathbb{R}^{4}$.
\item \textup{(Faithfulness.)} For every $\mathbf{c}\ne0$ and every integer $N$ with $1\le N\le T(\mathbf{c})+1$,
\[
\Big|\mathcal{S} B(\mathbf{c})-\sum_{p=2}^{N}b_{p}(\mathbf{c})\Big|
\;\le\;\frac{\Lambda(\vartheta)}{C}\binom{N+4}{3}N!\,x^{N+1}
\;+\;\frac{G(\vartheta)}{C}\,x\,e^{-T(\mathbf{c})} ,
\]
the second term being absent when $N=1$. In particular $\mathcal{S} B$ is Gevrey--$1$ asymptotic to $B$ to
all orders as $\mathbf{c}\to0$, uniformly in the direction $\mathbf{c}/\|\mathbf{c}\|$, and the one--loop coefficient is
reproduced exactly, in the asymptotic sense $\mathcal{S} B(\mathbf{c})=b_{2}(\mathbf{c})+O(\|\mathbf{c}\|_{\infty}^{3})$ and in
the exact sense of \textup{(iv)} \textup{(}the degree--$2$ Taylor part of $\mathcal{S} B$ at the origin
\emph{is} $b_{2}$\textup{)}; at a fixed $\mathbf{c}\ne0$, however, $\mathcal{S} B(\mathbf{c})\ne b_{2}(\mathbf{c})$ in general.
\item \textup{(Optimal truncation.)} With $N^{*}(\mathbf{c})=\lfloor T(\mathbf{c})\rfloor+1$ one has, for all
$\mathbf{c}$ with $0<x\le\min\{\vartheta_{k},1-\vartheta\}$,
\[
\Big|\mathcal{S} B(\mathbf{c})-\sum_{p=2}^{N^{*}(\mathbf{c})}b_{p}(\mathbf{c})\Big|
\;\le\;\frac{G(\vartheta)+139\,\Lambda(\vartheta)\,x^{-5/2}}{C}\;e^{-\vartheta_{k}/x},
\]
hence for every $\eta<\vartheta_{k}$ there is $K=K(k,\vartheta,\eta)$ with left--hand side $\le (K/C)e^{-\eta/x}$.
\item \textup{(Regularity.)} $\mathcal{S} B$ is real--analytic on $\mathbb{R}^{4}\setminus\{0\}$ and $C^{\infty}$ on
$\mathbb{R}^{4}$, and its Taylor expansion at the origin is exactly $B$:
$\partial^{\mathbf{m}}(\mathcal{S} B)(0)=\mathbf{m}!\,\beta(\mathbf{m})$ for $|\mathbf{m}|\ge2$, and $=0$ for $|\mathbf{m}|\le1$. If $B$
diverges, $\mathcal{S} B$ is not real--analytic at the origin, and no function with these asymptotics can be.
\item \textup{(Consistency.)} If $\tau\mapsto\widehat{B}(\tau\mathbf{c})$ extends holomorphically to a neighbourhood
of $[0,\infty)$ with $|\widehat{B}(\tau\mathbf{c})|\le K_{0}\sigma\tau e^{\sigma\tau}$ there for some $\sigma<1$,
then the classical Borel sum $\mathcal{L}B(\mathbf{c})=\int_{0}^{\infty}e^{-\tau}\widehat{B}(\tau\mathbf{c})\,d\tau/\tau$
exists and $|\mathcal{S} B(\mathbf{c})-\mathcal{L}B(\mathbf{c})|\le \frac{K_{0}\sigma}{1-\sigma}e^{-(1-\sigma)T(\mathbf{c})}$.
If $|b_{p}(\mathbf{c})|\le K_{0}\sigma^{p}$ for all $p$ with $\sigma<1$ --- in particular if
$|\beta(\mathbf{m})|\le K_{0}\rho^{-|\mathbf{m}|}$ and $\|\mathbf{c}\|_{1}<\rho$ --- then $B(\mathbf{c})$ converges and
$|\mathcal{S} B(\mathbf{c})-B(\mathbf{c})|\le\frac{K_{0}\sigma}{1-\sigma}e^{-(1-\sigma)T(\mathbf{c})}$. Equality does not hold in
general --- not even for polynomials --- and by Theorem \ref{thm:dichotomy} no rule well defined on
the whole class can achieve it.
\item \textup{(Optimality of the exponent.)} There are an admissible coefficient family and a
direction for which the set $\mathcal{F}$ of real functions obeying the Gevrey--$1$ bounds
$\big|f(\mathbf{c})-\sum_{p=2}^{N}b_{p}(\mathbf{c})\big|\le\frac{e+2\pi}{C}(N+1)!\,x^{N+1}$ $(N\ge1)$ contains two
elements differing, at every point of that direction, by exactly $(2\pi/C)e^{-1/x}$. Consequently no
summation rule whatsoever --- however constructed --- can lie within $o(e^{-1/x})$ of every member
of $\mathcal{F}$: the exponent $\eta=1$ in the scale $C\|\mathbf{c}\|_{\infty}$ is a barrier that no rule
can pass. By \textup{(iii)} and Remark \ref{rem:ratelimit} the family $\{\mathcal{S}_{k,\vartheta}\}$ realises
every exponent $\eta<1$; it is therefore rate--optimal, although no single member attains $\eta=1$.
\end{enumerate}

<!-- remark-1: body_md -->
\label{rem:ratelimit}
For $k\ge1$ admissibility holds whenever $\vartheta<\big[(1+\frac{1}{32k})2^{1/(2k)}\big]^{-1}$, and then
$\vartheta_{k}=\vartheta 4^{-1/(2k)}$ may be taken arbitrarily close to
$2^{-3/(2k)}\big/(1+\frac{1}{32k})$, which tends to $1$ as $k\to\infty$. Numerically this bound
equals $0.3428$ for $k=1$, $0.5855$ for $k=2$, $0.7651$ for $k=4$, $0.8747$ for $k=8$ and $0.9834$
for $k=64$ (to four decimals); these are suprema, approached but not attained, and they are not optimised
(Remark \ref{rem:notoptimal}). Thus every rate $\eta<1$ is realised by some admissible pair, while
by Theorem \ref{thm:main}(vi) no rule can achieve accuracy $o(e^{-1/x})$, so the exponent $1$
cannot be passed.
<!-- end remark-1 -->

Only part (iv) --- the complex--analytic statements --- uses the admissibility inequality of
Definition \ref{def:admissible}; parts (i), (ii), (iii) hold for every $k\ge1$ and every
$\vartheta\in(0,1)$.

<!-- lemma-1: body_md -->
\label{lem:count}
For every integer $p\ge0$:
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.2\parindent]
\item $\#\{\mathbf{m}\in\mathbb{Z}_{+}^{4}:|\mathbf{m}|=p\}=\binom{p+3}{3}$;
\item for $y\in[0,\infty)^{4}$ and $0\le s<1/\|y\|_{\infty}$ one has
      $\sum_{p\ge0}h_{p}(y)s^{p}=\prod_{i=1}^{4}(1-y_{i}s)^{-1}$, all terms being nonnegative;
      in particular $\sum_{p\ge0}\binom{p+3}{3}s^{p}=(1-s)^{-4}$ for $0\le s<1$;
\item $\displaystyle \|y\|_{\infty}^{\,p}\;\le\;h_{p}(y)\;\le\;
      \min\Big\{\binom{p+3}{3}\|y\|_{\infty}^{\,p},\ \|y\|_{1}^{\,p}\Big\}$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
(a) Stars and bars: the map $\mathbf{m}\mapsto\{m_{1}+1,m_{1}+m_{2}+2,m_{1}+m_{2}+m_{3}+3\}$ is a bijection
onto the $3$--element subsets of $\{1,\dots,p+3\}$.

(b) For $0\le s<1/\|y\|_{\infty}$ each factor is the sum of the nonnegative geometric series
$\sum_{m_{i}\ge0}(y_{i}s)^{m_{i}}$. The product of four absolutely convergent series may be expanded
and rearranged arbitrarily, and collecting the terms with $m_{1}+\dots+m_{4}=p$ gives $h_{p}(y)s^{p}$.
Taking $y=(1,1,1,1)$ and using (a) gives the last identity.

(c) The lower bound is the single term $\mathbf{m}=p\,e_{i_{0}}$ with $y_{i_{0}}=\|y\|_{\infty}$; the first
upper bound follows since each of the $\binom{p+3}{3}$ monomials is at most $\|y\|_{\infty}^{p}$; the
second follows from the multinomial theorem,
$\|y\|_{1}^{p}=\sum_{|\mathbf{m}|=p}\binom{p}{\mathbf{m}}y^{\mathbf{m}}\ge\sum_{|\mathbf{m}|=p}y^{\mathbf{m}}=h_{p}(y)$, because every
multinomial coefficient is at least $1$.

<!-- lemma-2: body_md -->
\label{lem:bp}
Assume Hypothesis \ref{hyp:H}. Then for all $\mathbf{z}\in\mathbb{C}^{4}$ and $p\ge2$,
\[
|b_{p}(\mathbf{z})|\;\le\;(p-1)!\,C^{p-1}h_{p}(|\mathbf{z}|)\;\le\;(p-1)!\,C^{p-1}\binom{p+3}{3}\|\mathbf{z}\|_{\infty}^{p},
\]
and also $|b_{p}(\mathbf{z})|\le (p-1)!C^{p-1}\|\mathbf{z}\|_{1}^{p}$. All three bounds are attained: for
$\beta(\mathbf{m})\equiv(p-1)!C^{p-1}$ and $\mathbf{z}\in[0,\infty)^{4}$ the first is an equality; the second is an
equality when in addition $|z_{1}|=\dots=|z_{4}|$; the third when at most one $z_{i}$ is nonzero.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Immediate from the triangle inequality, Hypothesis \ref{hyp:H} and Lemma \ref{lem:count}(c).

<!-- lemma-3: body_md -->
\label{lem:tail}
Let $\Lambda(t)=\sum_{q\ge0}(q+1)^{3}t^{q}$ and $G(t)=\sum_{p\ge2}p\binom{p+3}{3}t^{p-1}$ for
$0\le t<1$. Then:
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.2\parindent]
\item $\Lambda(t)=\dfrac{1+4t+t^{2}}{(1-t)^{4}}$ and $G(t)=4\big[(1-t)^{-5}-1\big]$;
\item for all integers $N,q\ge0$: $\dbinom{N+q+4}{3}\le\dbinom{N+4}{3}(1+q)^{3}$;
\item for all integers $N\ge0$ and all $0\le w\le t<1$:
\[
\sum_{p>N}\binom{p+3}{3}w^{\,p-1}\;\le\;\binom{N+4}{3}\Lambda(t)\,w^{N} .
\]
In particular (case $N=1$) $\sum_{p\ge2}\binom{p+3}{3}w^{p-1}\le 10\,\Lambda(t)\,w$.
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
(a) From $n^{3}=\binom{n}{1}+6\binom{n}{2}+6\binom{n}{3}$ (check: $\binom n1+6\binom n2+6\binom n3
=n+3n(n-1)+n(n-1)(n-2)=n^{3}$) and $\sum_{n\ge0}\binom{n}{j}t^{n}=t^{j}(1-t)^{-j-1}$ we get
\[
\sum_{n\ge1}n^{3}t^{n}=\frac{t}{(1-t)^{2}}+\frac{6t^{2}}{(1-t)^{3}}+\frac{6t^{3}}{(1-t)^{4}}
=\frac{t(1-t)^{2}+6t^{2}(1-t)+6t^{3}}{(1-t)^{4}}=\frac{t+4t^{2}+t^{3}}{(1-t)^{4}},
\]
and dividing by $t$ gives $\Lambda$. For $G$, differentiate the identity
$\sum_{p\ge0}\binom{p+3}{3}t^{p}=(1-t)^{-4}$ of Lemma \ref{lem:count}(b) term by term (legitimate
inside the disc of convergence of a power series), obtaining
$\sum_{p\ge1}p\binom{p+3}{3}t^{p-1}=4(1-t)^{-5}$, and subtract the $p=1$ term $\binom{4}{3}=4$.

(b) $\displaystyle\frac{\binom{N+q+4}{3}}{\binom{N+4}{3}}
=\prod_{j=2}^{4}\frac{N+j+q}{N+j}=\prod_{j=2}^{4}\Big(1+\frac{q}{N+j}\Big)\le(1+q)^{3}$,
since $N+j\ge 2\ge1$ for $j\ge2$, $N\ge0$.

(c) Substituting $p=N+1+q$ and using (b) and $w\le t$,
\[
\sum_{p>N}\binom{p+3}{3}w^{p-1}=\sum_{q\ge0}\binom{N+q+4}{3}w^{N+q}
\le\binom{N+4}{3}w^{N}\sum_{q\ge0}(1+q)^{3}w^{q}\le\binom{N+4}{3}\Lambda(t)\,w^{N}. \qedhere
\]

<!-- proposition-1: body_md -->
\label{prop:borel}
Assume Hypothesis \ref{hyp:H}, and let $\Pi_{C}:=\{\mathbf{z}\in\mathbb{C}^{4}:\|\mathbf{z}\|_{\infty}<1/C\}$. Then:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2\parindent]
\item the series defining $\widehat{B}$ converges absolutely and uniformly on every compact subset of
$\Pi_{C}$, and $\widehat{B}$ is holomorphic on $\Pi_{C}$ and vanishes to order $2$ at the origin;
\item for $\mathbf{z}\in\Pi_{C}$,
\[
|\widehat{B}(\mathbf{z})|\;\le\;\frac{1}{C}\Big[\prod_{i=1}^{4}\frac{1}{1-C|z_{i}|}-1-C\|\mathbf{z}\|_{1}\Big];
\]
\item \textup{(radial form)} for $\mathbf{c}\in\mathbb{R}^{4}$, $\tau\in\mathbb{C}$ and $0\le t<1$ with
$C\|\mathbf{c}\|_{\infty}|\tau|\le t$,
\[
|\widehat{B}(\tau\mathbf{c})|\;\le\;10\,\Lambda(t)\,C\,\|\mathbf{c}\|_{\infty}^{2}\,|\tau|^{2};
\]
\item the polydisc radius $1/C$ is sharp: for the admissible family $\beta(\mathbf{m})=(p-1)!C^{p-1}$ when
$\mathbf{m}=p\,e_{1}$ and $\beta(\mathbf{m})=0$ otherwise one has $\widehat{B}(\mathbf{z})=Cz_{1}^{2}/(1-Cz_{1})$, which has a
pole at $z_{1}=1/C$.
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
(i) and (ii). By Lemma \ref{lem:bp}, $|b_{p}(\mathbf{z})|/(p-1)!\le C^{p-1}h_{p}(|\mathbf{z}|)$, so
\[
\sum_{p\ge2}\frac{|b_{p}(\mathbf{z})|}{(p-1)!}\le\sum_{p\ge2}C^{p-1}h_{p}(|\mathbf{z}|)
=\frac{1}{C}\Big[\sum_{p\ge0}h_{p}(|\mathbf{z}|)C^{p}-h_{0}-Ch_{1}(|\mathbf{z}|)\Big]
=\frac{1}{C}\Big[\prod_{i}\frac{1}{1-C|z_{i}|}-1-C\|\mathbf{z}\|_{1}\Big],
\]
by Lemma \ref{lem:count}(b) with $s=C$, which is legitimate precisely when $C\|\mathbf{z}\|_{\infty}<1$.
This proves (ii) and also absolute convergence; the majorant is a continuous increasing function of
each $|z_{i}|$, so convergence is uniform on the compact subsets of $\Pi_{C}$ (each of which is
contained in a closed polydisc $\{\|\mathbf{z}\|_{\infty}\le r\}$ with $r<1/C$). The partial sums are
polynomials, hence holomorphic; a locally uniform limit of holomorphic functions of several complex
variables is holomorphic (Lemma \ref{lem:weier}), so $\widehat{B}\in\mathcal{O}(\Pi_{C})$. Since the series
starts at $p=2$ and each $b_{p}$ is homogeneous of degree $p$, $\widehat{B}$ vanishes to order $2$ at $0$.

(iii) By Lemma \ref{lem:bp} and homogeneity, with $\mu:=\|\mathbf{c}\|_{\infty}$ and $w:=C\mu|\tau|\le t$,
\[
|\widehat{B}(\tau\mathbf{c})|\le\sum_{p\ge2}C^{p-1}\binom{p+3}{3}\mu^{p}|\tau|^{p}
=\mu|\tau|\sum_{p\ge2}\binom{p+3}{3}w^{p-1}\le 10\,\Lambda(t)\,\mu|\tau|\,w
=10\Lambda(t)C\mu^{2}|\tau|^{2},
\]
using Lemma \ref{lem:tail}(c) with $N=1$.

(iv) The family satisfies Hypothesis \ref{hyp:H} with equality on its support, and
$\widehat{B}(\mathbf{z})=\sum_{p\ge2}C^{p-1}z_{1}^{p}=Cz_{1}^{2}/(1-Cz_{1})$ for $|z_1|<1/C$.
<!-- end proof-4 -->
<!-- remark-3: body_md -->
\label{rem:poleshape}
Proposition \ref{prop:borel} is the exact analytic content of Hypothesis \ref{hyp:H}: the hypothesis
is \emph{equivalent} to the statement that the coefficients of $\widehat{B}$ are dominated, term by term, by
those of the single--pole function $\sum_{p\ge2}C^{p-1}\zeta^{p}=C\zeta^{2}/(1-C\zeta)$. Hypothesis
\ref{hyp:H} says exactly: ``the Borel transform is majorised, coefficientwise, by a single simple
pole at distance $1/C$''. (This does not mean that the radial Borel transform itself has only a
simple pole: for $\beta(\mathbf{m})\equiv(|\mathbf{m}|-1)!C^{|\mathbf{m}|-1}$ and $\mathbf{c}=(\mu,\mu,\mu,\mu)$ one finds
$\widehat{B}(\tau\mathbf{c})=\frac1C\big[(1-C\mu\tau)^{-4}-1-4C\mu\tau\big]$, a pole of order four.) It says nothing
whatsoever about what happens at or beyond that distance --- see \S\ref{sec:optimality}.

<!-- definition-3: body_md -->
\label{def:rule}
Let $k\in\mathbb{N}$, $k\ge1$, and $\vartheta\in(0,1)$. Define the \emph{gauge}
\[
\nu_{k}(\mathbf{c}):=\Big(\sum_{i=1}^{4}c_{i}^{2k}\Big)^{1/(2k)}\qquad(\mathbf{c}\in\mathbb{R}^{4}),
\]
the \emph{cut height} $T(\mathbf{c}):=\dfrac{\vartheta}{C\,\nu_{k}(\mathbf{c})}$ for $\mathbf{c}\ne0$ --- with the convention
$T(0):=+\infty$ and $P(p,+\infty):=1$, under which every formula below holds trivially at $\mathbf{c}=0$,
both sides vanishing --- and the
\emph{cut Borel--Laplace sum}
\[
\mathcal{S} B(\mathbf{c})\;:=\;\int_{0}^{T(\mathbf{c})}e^{-\tau}\,\widehat{B}(\tau\mathbf{c})\,\frac{d\tau}{\tau}\quad(\mathbf{c}\ne0),
\qquad \mathcal{S} B(0):=0 .
\]

<!-- lemma-4: body_md -->
\label{lem:gauge}
$\nu_{k}$ is positively homogeneous of degree $1$, real--analytic on $\mathbb{R}^{4}\setminus\{0\}$, and
\[
\|\mathbf{c}\|_{\infty}\;\le\;\nu_{k}(\mathbf{c})\;\le\;4^{1/(2k)}\|\mathbf{c}\|_{\infty},
\qquad
\nu_{k}(\mathbf{c})\le\|\mathbf{c}\|_{2}\le\|\mathbf{c}\|_{1}.
\]
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Write $P_{k}(\mathbf{c})=\sum_{i}c_{i}^{2k}$, a polynomial with $P_{k}(\mathbf{c})>0$ for $\mathbf{c}\ne0$ (all exponents
are even). Then $\nu_{k}=P_{k}^{1/(2k)}$ is the composition of a polynomial with the real--analytic
function $s\mapsto s^{1/(2k)}$ on $(0,\infty)$, hence real--analytic off the origin; homogeneity is
clear. Since $\|\mathbf{c}\|_{\infty}^{2k}\le P_{k}(\mathbf{c})\le 4\|\mathbf{c}\|_{\infty}^{2k}$, taking $2k$--th roots
gives the first chain. Finally $\|\mathbf{c}\|_{2k}\le\|\mathbf{c}\|_{2}\le\|\mathbf{c}\|_{1}$ because $\ell^{q}$ norms
decrease in $q$ and $2k\ge2$.

<!-- lemma-5: body_md -->
\label{lem:gamma}
For integers $p\ge1$ and $T>0$ put
$P(p,T):=\frac{1}{(p-1)!}\int_{0}^{T}e^{-\tau}\tau^{p-1}d\tau$ and $Q(p,T):=1-P(p,T)$. Then
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.2\parindent]
\item $Q(p,T)=e^{-T}\sum_{j=0}^{p-1}\dfrac{T^{j}}{j!}=\Pr[\mathrm{Poi}(T)\le p-1]$, hence
      $P(p,T)=\Pr[\mathrm{Poi}(T)\ge p]\in(0,1)$;
\item $P(p,T)\le\min\{1,\,T^{p}/p!\}$;
\item if $T\ge p-1$ then $Q(p,T)\le\dfrac{p\,T^{p-1}e^{-T}}{(p-1)!}$.
\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
(a) Let $\Gamma(p,T)=\int_{T}^{\infty}e^{-\tau}\tau^{p-1}d\tau$, so $Q(p,T)=\Gamma(p,T)/(p-1)!$.
Integration by parts gives $\Gamma(p+1,T)=T^{p}e^{-T}+p\,\Gamma(p,T)$, whence
$Q(p+1,T)=\frac{T^{p}e^{-T}}{p!}+Q(p,T)$; together with $Q(1,T)=\Gamma(1,T)=e^{-T}$ this yields the
displayed formula by induction, which is precisely the Poisson distribution function.

(b) $\int_{0}^{T}e^{-\tau}\tau^{p-1}d\tau\le\int_{0}^{T}\tau^{p-1}d\tau=T^{p}/p$, and divide by
$(p-1)!$; also $P\le1$ trivially.

(c) The sequence $j\mapsto T^{j}/j!$ is nondecreasing for $j\le T$; as $j\le p-1\le T$, each of the
$p$ terms of the sum in (a) is at most $T^{p-1}/(p-1)!$.
<!-- end proof-6 -->
<!-- proposition-2: body_md -->
\label{prop:rule}
Assume Hypothesis \ref{hyp:H}, let $k\ge1$, $\vartheta\in(0,1)$, and write $\mu=\|\mathbf{c}\|_{\infty}$,
$x=C\mu$, $T=T(\mathbf{c})$. Then for every $\mathbf{c}\in\mathbb{R}^{4}$:

(i) By Lemma \ref{lem:gauge}, $C\mu\tau\le C\nu_{k}(\mathbf{c})T(\mathbf{c})=\vartheta$ for $0\le\tau\le T$. Hence

<!-- remark-4: body_md -->
Proposition \ref{prop:rule}(ii) identifies the rule: \emph{it truncates the divergent series at a
Poisson--distributed order of mean $T(\mathbf{c})\asymp 1/(C\|\mathbf{c}\|)$ and averages}. Optimal truncation is
thereby made linear in the coefficients, smooth in $\mathbf{c}$, and --- as \S\ref{sec:reg} shows ---
real--analytic away from the origin, none of which sharp truncation achieves.

<!-- remark-5: body_md -->
\label{rem:tradeoff}
Since $0<P(p,T)<1$ for every finite $T$, the rule is \emph{not} exact even on a single monomial: for
the admissible family $\beta(2e_{1})=C$, all other coefficients zero, the series
$B(\mathbf{c})=C\lambda^{2}$ is a polynomial, yet
\[
\mathcal{S} B(\mathbf{c})=C\lambda^{2}P\big(2,T(\mathbf{c})\big)=C\lambda^{2}\big[1-e^{-T(\mathbf{c})}(1+T(\mathbf{c}))\big]\ \ne\ B(\mathbf{c}).
\]
The discrepancy is $C\lambda^{2}e^{-T}(1+T)$, exponentially small in $1/(C\|\mathbf{c}\|_{\infty})$ but not
zero. This is not an oversight; it is forced. Theorem \ref{thm:dichotomy} below shows that a linear
weight rule cannot be both well defined on the whole class $\mathcal{A}_{C}$ and regular, and
Theorem \ref{thm:defect} shows that the defect on convergent series cannot be removed by any
cleverer choice of weights. What \emph{is} a choice is where the price is paid: the sharply
truncated rule $w_{p}=\mathbf 1_{\{p\le N^{*}(\mathbf{c})\}}$ is exact on every polynomial of degree
$\le N^{*}(\mathbf{c})$ and is well defined on $\mathcal{A}_{C}$, but it is not even continuous in $\mathbf{c}$,
hence neither smooth nor real--analytic, and it does not admit the analytic continuation used in
\S\ref{sec:reg}. The Poisson smoothing buys real--analyticity off the origin, $C^{\infty}$
smoothness at the origin and an exactly correct Taylor series there, at the cost of an error beyond
all orders on convergent input. Which price to pay depends on the application; both are exhibited
here, and Theorem \ref{thm:dichotomy} shows that some price must be paid.

<!-- theorem-2: body_md -->
\label{thm:master}
Assume Hypothesis \ref{hyp:H}, let $k\ge1$, $\vartheta\in(0,1)$, and let $\mathbf{c}\in\mathbb{R}^{4}\setminus\{0\}$.
Write $x=C\|\mathbf{c}\|_{\infty}$ and $T=T(\mathbf{c})$. Then for every integer $N$ with $1\le N\le T+1$,
\[
\Big|\mathcal{S} B(\mathbf{c})-\sum_{p=2}^{N}b_{p}(\mathbf{c})\Big|
\;\le\;\underbrace{\frac{\Lambda(\vartheta)}{C}\binom{N+4}{3}N!\;x^{N+1}}_{\text{\rm tail}}
\;+\;\underbrace{\frac{G(\vartheta)}{C}\;x\,e^{-T}}_{\text{\rm head}},
\]
where the head term is absent for $N=1$ \textup{(}for which the statement reduces to
Proposition \textup{\ref{prop:rule}(i)}\textup{)}. The tail term is valid without any restriction on
$N$.
<!-- end theorem-2 -->
<!-- proof-8: body_md -->
By Proposition \ref{prop:rule}(ii) and $P+Q=1$,
\[
\mathcal{S} B(\mathbf{c})-\sum_{p=2}^{N}b_{p}(\mathbf{c})
=\underbrace{\sum_{p>N}P(p,T)\,b_{p}(\mathbf{c})}_{=:I}\;-\;\underbrace{\sum_{p=2}^{N}Q(p,T)\,b_{p}(\mathbf{c})}_{=:J},
\]
both series being absolutely convergent (the first by Proposition \ref{prop:rule}(ii), the second
being finite).

\smallskip
\emph{The tail $I$.} Repeating the termwise integration of Proposition \ref{prop:rule}(ii) on the
tail alone,
\[
I=\int_{0}^{T}e^{-\tau}\Big(\sum_{p>N}\frac{b_{p}(\mathbf{c})}{(p-1)!}\tau^{p-1}\Big)d\tau .
\]
For $0\le\tau\le T$ set $w:=C\mu\tau\le\vartheta$, $\mu=\|\mathbf{c}\|_{\infty}$. By Lemma \ref{lem:bp},
$\frac{|b_{p}(\mathbf{c})|}{(p-1)!}\tau^{p-1}\le\mu\binom{p+3}{3}w^{p-1}$, so Lemma \ref{lem:tail}(c) gives
\[
\Big|\sum_{p>N}\frac{b_{p}(\mathbf{c})}{(p-1)!}\tau^{p-1}\Big|
\le\mu\binom{N+4}{3}\Lambda(\vartheta)\,w^{N}=\mu\binom{N+4}{3}\Lambda(\vartheta)(C\mu)^{N}\tau^{N}.
\]
Hence, extending the integration to $[0,\infty)$,
\[
|I|\le\mu\binom{N+4}{3}\Lambda(\vartheta)(C\mu)^{N}\int_{0}^{\infty}e^{-\tau}\tau^{N}d\tau
=\Lambda(\vartheta)\binom{N+4}{3}N!\,\mu\,(C\mu)^{N}
=\frac{\Lambda(\vartheta)}{C}\binom{N+4}{3}N!\,x^{N+1}.
\]
No constraint on $N$ was used.

\smallskip
\emph{The head $J$.} Assume $N\ge2$. For $2\le p\le N$ we have $p-1\le N-1\le T$, so
Lemma \ref{lem:gamma}(c) applies, and with Lemma \ref{lem:bp} and $C\mu T\le\vartheta$,
\[
Q(p,T)|b_{p}(\mathbf{c})|\le\frac{p\,T^{p-1}e^{-T}}{(p-1)!}\cdot(p-1)!\,C^{p-1}\binom{p+3}{3}\mu^{p}
=p\binom{p+3}{3}\mu\,e^{-T}\,(C\mu T)^{p-1}
\le p\binom{p+3}{3}\mu\,e^{-T}\vartheta^{\,p-1}.
\]
Summing over $p\ge2$ and using the definition of $G$ in Lemma \ref{lem:tail}(a),
$|J|\le G(\vartheta)\,\mu\,e^{-T}=\frac{G(\vartheta)}{C}x\,e^{-T}$.

<!-- proposition-4: body_md -->
\label{prop:analytic}
For $\mathbf{z}$ in a neighbourhood of $\Delta_{0}$ define
\[
F(\mathbf{z}):=\int_{[0,T(\mathbf{z})]}e^{-\tau}\,\widehat{B}(\tau\mathbf{z})\,\frac{d\tau}{\tau}
=\sum_{p\ge2}\frac{b_{p}(\mathbf{z})}{(p-1)!}\,\gamma_{p}\big(T(\mathbf{z})\big),
\qquad \gamma_{p}(w):=\int_{[0,w]}e^{-\tau}\tau^{p-1}d\tau ,
\]
the integrals being taken along the complex segment $[0,w]$. Then the series converges absolutely
and uniformly on $\Delta_{0}$, $F$ is holomorphic on the interior of $\Delta_{0}$, and
$F=\mathcal{S} B$ on $\Delta_{0}\cap\mathbb{R}^{4}$. Consequently $\mathcal{S} B$ is real--analytic on $\mathbb{R}^{4}\setminus\{0\}$.

<!-- proof-13: body_md -->
Substituting $\tau=ws$ shows $\gamma_{p}(w)=w^{p}\int_{0}^{1}e^{-ws}s^{p-1}ds$, which is an entire
function of $w$ with $|\gamma_{p}(w)|\le|w|^{p}e^{|w|}/p$. By Lemma \ref{lem:bp} and

\[

\le\frac{e^{|T(\mathbf{z})|}}{C}\binom{p+3}{3}\frac{(\vartheta^{*})^{p}}{p},
\]

the composition of the entire function $\gamma_{p}$ with the holomorphic function $T$), so by the
Weierstrass convergence theorem (Lemma \ref{lem:weier}) $F$ is holomorphic on the interior of
$\Delta_{0}$. That the series equals the integral is the same termwise integration as in

$\gamma_{p}(T(\mathbf{c}))=(p-1)!P(p,T(\mathbf{c}))$, so $F(\mathbf{c})=\mathcal{S} B(\mathbf{c})$ by
Proposition \ref{prop:rule}(ii). A function holomorphic on a complex neighbourhood of a real point
restricts to a real--analytic function near that point; as $\mathbf{c}_{0}\in\mathbb{R}^{4}\setminus\{0\}$ was
arbitrary, $\mathcal{S} B$ is real--analytic on $\mathbb{R}^{4}\setminus\{0\}$.

<!-- theorem-4: body_md -->
\label{thm:cxmaster}
Let $(k,\vartheta)$ be admissible, $\mathbf{c}_{0}\in\mathbb{R}^{4}\setminus\{0\}$, and $\mathbf{z}\in\Delta_{0}$. Then for
every integer $N$ with $1\le N\le 1+\frac12\Re T(\mathbf{z})$,
\[
\Big|F(\mathbf{z})-\sum_{p=2}^{N}b_{p}(\mathbf{z})\Big|
\;\le\;\frac{\Lambda(\vartheta^{*})}{C}\binom{N+4}{3}N!\;\Big(\frac{C\|\mathbf{z}\|_{\infty}}{\kappa_{0}}\Big)^{N+1}
\;+\;\frac{20\,\Lambda(\vartheta^{*})\,\vartheta^{*}}{\kappa_{0}}\;\|\mathbf{z}\|_{\infty}\,e^{-\Re T(\mathbf{z})} .
\]
<!-- end theorem-4 -->
<!-- proof-14: body_md -->
Write $T=T(\mathbf{z})$, $\psi=\arg T$, $|\psi|\le\pi/12<\pi/2$. For $|\psi|<\pi/2$ Cauchy's theorem
applied to the entire function $e^{-\tau}\tau^{p-1}$ on the sector between $[0,R]$ and
$[0,Re^{i\psi}]$, together with the vanishing of the circular arc as $R\to\infty$, gives
$\int_{0}^{\infty e^{i\psi}}e^{-\tau}\tau^{p-1}d\tau=(p-1)!$. Splitting this ray at $T$ we obtain
\[
F(\mathbf{z})-\sum_{p=2}^{N}b_{p}(\mathbf{z})=I-J,\qquad
I=\int_{[0,T]}e^{-\tau}\Big(\sum_{p>N}\frac{b_{p}(\mathbf{z})}{(p-1)!}\tau^{p-1}\Big)d\tau,\quad
J=\sum_{p=2}^{N}\frac{b_{p}(\mathbf{z})}{(p-1)!}\int_{\Gamma}e^{-\tau}\tau^{p-1}d\tau,
\]
where $\Gamma$ is the ray from $T$ to $\infty e^{i\psi}$.

\emph{Bound on $I$.} Parametrise $\tau=Ts$, $s\in[0,1]$, so $|e^{-\tau}|=e^{-s\Re T}$ and
$|\tau|=s|T|$; note $C\|\mathbf{z}\|_{\infty}|\tau|\le\vartheta^{*}$ by Lemma \ref{lem:cxgauge}(c). By
Lemma \ref{lem:bp} and Lemma \ref{lem:tail}(c),
\[
\Big|\sum_{p>N}\frac{b_{p}(\mathbf{z})}{(p-1)!}\tau^{p-1}\Big|
\le\|\mathbf{z}\|_{\infty}\binom{N+4}{3}\Lambda(\vartheta^{*})\,(C\|\mathbf{z}\|_{\infty})^{N}|\tau|^{N},
\]
so, substituting $\sigma=s\Re T$ and using $\Re T\ge\kappa_{0}|T|$,
\[
|I|\le\|\mathbf{z}\|_{\infty}\binom{N+4}{3}\Lambda(\vartheta^{*})(C\|\mathbf{z}\|_{\infty})^{N}
\int_{0}^{1}(s|T|)^{N}e^{-s\Re T}|T|\,ds
\le\Lambda(\vartheta^{*})\binom{N+4}{3}\frac{N!}{\kappa_{0}^{N+1}}\|\mathbf{z}\|_{\infty}(C\|\mathbf{z}\|_{\infty})^{N},
\]
which is the first term.

\emph{Bound on $J$.} On $\Gamma$ write $\tau=Ts$ with $s\ge1$; then $|\tau^{p-1}|\le|T|^{p-1}s^{N-1}$
for $2\le p\le N$. With $a:=\Re T$ and $N-1\le a/2$,
\[
\int_{1}^{\infty}s^{N-1}e^{-as}ds=e^{-a}\int_{0}^{\infty}(1+u)^{N-1}e^{-au}du
\le e^{-a}\int_{0}^{\infty}e^{-(a-N+1)u}du=\frac{e^{-a}}{a-N+1}\le\frac{2e^{-a}}{a},
\]
using $(1+u)^{N-1}\le e^{(N-1)u}$. Hence, by Lemma \ref{lem:bp}, Lemma \ref{lem:tail}(c) with
$N=1$, and $C\|\mathbf{z}\|_{\infty}|T|\le\vartheta^{*}$,
\[
|J|\le\frac{2e^{-a}}{a}\sum_{p\ge2}\frac{|b_{p}(\mathbf{z})|}{(p-1)!}|T|^{p}
\le\frac{2e^{-a}|T|}{a}\,\|\mathbf{z}\|_{\infty}\sum_{p\ge2}\binom{p+3}{3}(C\|\mathbf{z}\|_{\infty}|T|)^{p-1}
\le\frac{2}{\kappa_{0}}\,10\Lambda(\vartheta^{*})\vartheta^{*}\,\|\mathbf{z}\|_{\infty}e^{-\Re T}. \qedhere
\]

<!-- theorem-5: body_md -->
\label{thm:smooth}
Let $(k,\vartheta)$ be admissible. Then $\mathcal{S} B\in C^{\infty}(\mathbb{R}^{4})$, it is real--analytic on
$\mathbb{R}^{4}\setminus\{0\}$, and
\[
\partial^{\mathbf{m}}(\mathcal{S} B)(0)=\mathbf{m}!\,\beta(\mathbf{m})\ \ (|\mathbf{m}|\ge2),\qquad \partial^{\mathbf{m}}(\mathcal{S} B)(0)=0\ \ (|\mathbf{m}|\le1),
\]
i.e. the Taylor series of $\mathcal{S} B$ at the origin is exactly the formal series $B$. If $B$ has radius
of convergence $0$ \textup{(}as it does for the admissible families of \S\textup{\ref{sec:optimality}}\textup{)},
then $\mathcal{S} B$ is not real--analytic at the origin; and no function whatsoever that is asymptotic to
$B$ to all orders can be real--analytic at the origin.
<!-- end theorem-5 -->
<!-- proof-15: body_md -->
Real--analyticity off the origin is Proposition \ref{prop:analytic}. Fix a multi--index $\alpha$ and
an integer $N>|\alpha|$, and set $g_{N}:=F-\sum_{p=2}^{N}b_{p}$, a holomorphic function on the
interior of $\Delta_{0}$ agreeing with $\mathcal{S} B-\sum_{p=2}^{N}b_{p}$ on the reals. On $\Delta_{0}$,
$\|\mathbf{z}\|_{\infty}\le(1+\delta_{k})t_{0}\le\frac{33}{32}t_{0}$ and, by Lemma \ref{lem:cxgauge}(c)
together with the first inequality of \eqref{eq:Tlower} (which holds unconditionally),
\[
\Re T(\mathbf{z})\ \ge\ \kappa_{0}|T(\mathbf{z})|\ \ge\ \kappa_{0}\,2^{-1/(2k)}T(\mathbf{c}_{0})\ \ge\
\frac{\kappa_{0}}{\sqrt2}\cdot\frac{\vartheta_{k}}{C t_{0}}=:\frac{\kappa_{1}}{C t_{0}},
\qquad \kappa_{1}:=\frac{\kappa_{0}\vartheta_{k}}{\sqrt2}>0 .
\]
Hence the hypothesis $N\le1+\frac12\Re T(\mathbf{z})$ of Theorem \ref{thm:cxmaster} holds as soon as
$t_{0}\le\frac{\kappa_{1}}{2C(N-1)}$, and then
\[
\sup_{\Delta_{0}}|g_{N}|\ \le\ K_{1}(N)\,t_{0}^{N+1}+K_{2}\,t_{0}\,e^{-\kappa_{1}/(Ct_{0})},
\]
with $K_{1}(N)=\frac{\Lambda(\vartheta^{*})}{C}\binom{N+4}{3}N!\big(\frac{33C}{32\kappa_{0}}\big)^{N+1}$
and $K_{2}=\frac{33}{32}\cdot\frac{20\Lambda(\vartheta^{*})\vartheta^{*}}{\kappa_{0}}$. Using
$e^{-u}\le M!\,u^{-M}$ (valid for $u>0$, $M\in\mathbb{N}$, since $e^{u}\ge u^{M}/M!$) with $u=\kappa_{1}/(Ct_{0})$
and $M=N+|\alpha|$, the second term is $\le K_{2}'\,t_{0}^{N+1+|\alpha|}$. Cauchy's inequalities
(Lemma \ref{lem:cauchy}), applied on the closed polydisc of polyradius
$(\tfrac12\delta_{k}t_{0},\dots,\tfrac12\delta_{k}t_{0})$ --- which is contained in the open polydisc
on which $g_{N}$ is holomorphic --- now give
\begin{equation}\label{eq:cauchyest}
\big|\partial^{\alpha}g_{N}(\mathbf{c}_{0})\big|\ \le\ \frac{\alpha!\,2^{|\alpha|}}{(\delta_{k}t_{0})^{|\alpha|}}\sup_{\Delta_{0}}|g_{N}|
\ \le\ K(N,\alpha)\,t_{0}^{\,N+1-|\alpha|}
\qquad\Big(0<t_{0}\le t_{*}(N)\Big),
\end{equation}
for suitable finite constants $K(N,\alpha)$, where $t_{*}(N):=\min\big\{1,\tfrac{\kappa_{1}}{2C(N-1)}\big\}$
for $N\ge2$ and $t_{*}(1):=1$ (for $N=1$ the hypothesis $N\le1+\frac12\Re T$ is automatic).
Apply \eqref{eq:cauchyest} with $N=|\alpha|+1$. Since
$\partial^{\alpha}\big(\sum_{p=2}^{|\alpha|+1}b_{p}\big)=\alpha!\,\beta(\alpha)
+\partial^{\alpha}b_{|\alpha|+1}$, where $\beta(\alpha):=0$ if $|\alpha|\le1$, $b_{1}:=0$ (so that for $|\alpha|=0$ the sum is empty and
the assertion reduces to $\mathcal{S} B(\mathbf{c}_{0})=O(t_{0}^{2})$), and $\partial^{\alpha}b_{|\alpha|+1}$ is
homogeneous of degree $1$, we conclude that
\[
\partial^{\alpha}(\mathcal{S} B)(\mathbf{c}_{0})=\alpha!\,\beta(\alpha)+O(t_{0})\qquad (t_{0}\to0),
\]
so every partial derivative of $\mathcal{S} B$ extends continuously to $\mathbb{R}^{4}$, with value
$\alpha!\beta(\alpha)$ at the origin. As $\mathcal{S} B$ is continuous at the origin
(Proposition \ref{prop:rule}(i)) and $C^{\infty}$ off it, Lemma \ref{lem:flat} yields
$\mathcal{S} B\in C^{\infty}(\mathbb{R}^{4})$ and $\partial^{\alpha}(\mathcal{S} B)(0)=\alpha!\beta(\alpha)$.

Finally, if $f$ is real--analytic at $0$ and asymptotic to $B$ to all orders, then $f$ equals its
Taylor series near $0$; comparing asymptotic expansions in each direction $\mathbf{u}$ and using
homogeneity (as in the uniqueness argument of Lemma \ref{lem:uniqueasy}) shows that the homogeneous
Taylor parts of $f$ are the $b_{p}$, so $B$ converges near $0$, a contradiction.

<!-- remark-7: body_md -->
\label{rem:hartogs}
The restriction to a \emph{conic} complex neighbourhood in Proposition \ref{prop:analytic} is not an
artefact. If $B$ diverges, no function holomorphic on a punctured complex ball
$\{0<\|\mathbf{z}\|<r\}\subset\mathbb{C}^{4}$ can be asymptotic to $B$ to all orders: by the Hartogs--Osgood--Brown
extension theorem~\cite{Hormander1990} (the only external result quoted in this paper, and used only in this remark; in
$n\ge2$ complex variables a function holomorphic on a punctured ball extends holomorphically to the
whole ball) such a function would extend across the origin, hence be given by a convergent power
series, whose homogeneous parts must then be the $b_{p}$ --- so $B$ would converge. In one variable
no such obstruction exists, which is why the classical Gevrey theory is sectorial. Nothing in
Theorem \ref{thm:main} depends on this remark.

\label{prop:borelcase}

holomorphic on $|\tau|<1/(C\|\mathbf{c}\|_{\infty})$, extends holomorphically to a neighbourhood of

$\mathcal{L}B(\mathbf{c})=\int_{0}^{\infty}e^{-\tau}\widehat{B}(\tau\mathbf{c})\frac{d\tau}{\tau}$ exists as an absolutely
convergent integral, and
\[
\big|\mathcal{S} B(\mathbf{c})-\mathcal{L}B(\mathbf{c})\big|\;\le\;\frac{K_{0}\sigma}{1-\sigma}\,e^{-(1-\sigma)T(\mathbf{c})} .
\]

<!-- proposition-6: body_md -->
\label{prop:convcase}
Fix $\mathbf{c}\ne0$ and suppose $|b_{p}(\mathbf{c})|\le K_{0}\sigma^{p}$ for all $p\ge2$ and some

\[

\]

<!-- theorem-6: body_md -->
\label{thm:divergence}
Let $\beta^{\star}(\mathbf{m}):=(|\mathbf{m}|-1)!\,C^{|\mathbf{m}|-1}$ for all $|\mathbf{m}|\ge2$. Then
$\beta^{\star}\in\mathcal{A}_{C}$ \textup{(}with equality in Hypothesis \ref{hyp:H}\textup{)}, and
for \emph{every} $\mathbf{c}\in\mathbb{R}^{4}\setminus\{0\}$ the terms $b^{\star}_{p}(\mathbf{c})$ are unbounded; hence
$\sum_{p\ge2}b^{\star}_{p}(\mathbf{c})$ diverges at every $\mathbf{c}\ne0$. In particular no rule can be
``evaluation of a convergent series''.
<!-- end theorem-6 -->
<!-- proof-19: body_md -->
Here $b^{\star}_{p}(\mathbf{c})=(p-1)!C^{p-1}\,\tilde h_{p}(\mathbf{c})$ with
$\tilde h_{p}(\mathbf{c}):=\sum_{|\mathbf{m}|=p}\mathbf{c}^{\mathbf{m}}$ the (signed) complete homogeneous polynomial. For
$|s|<1/\|\mathbf{c}\|_{\infty}$ we have, by absolute convergence and Lemma \ref{lem:count}(b) applied to
$|\mathbf{c}|$,
\[
\sum_{p\ge0}\tilde h_{p}(\mathbf{c})s^{p}=\prod_{i=1}^{4}\frac{1}{1-c_{i}s}=:R(s).
\]
$R$ is a rational function whose numerator is the constant $1$; hence its poles are exactly the
points $s=1/c_{i}$ with $c_{i}\ne0$, no cancellation being possible, and the pole nearest the origin
is at distance $1/\|\mathbf{c}\|_{\infty}$. The Taylor series of a rational function at $0$ has radius of
convergence equal to the distance to the nearest pole, so
$\limsup_{p}|\tilde h_{p}(\mathbf{c})|^{1/p}=\|\mathbf{c}\|_{\infty}$. Fix $0<\varepsilon<\|\mathbf{c}\|_{\infty}$; then
$|\tilde h_{p}(\mathbf{c})|\ge(\|\mathbf{c}\|_{\infty}-\varepsilon)^{p}$ for infinitely many $p$, and for those
\[
|b^{\star}_{p}(\mathbf{c})|\ge (p-1)!\,C^{p-1}(\|\mathbf{c}\|_{\infty}-\varepsilon)^{p}\longrightarrow\infty .
\]
A series whose terms are unbounded diverges.

<!-- theorem-7: body_md -->
\label{thm:natural}
Let $g(w):=\sum_{j\ge0}w^{2^{j}}$, holomorphic on the unit disc. Then $g$ admits no holomorphic
extension to any open set meeting the unit circle: the circle $|w|=1$ is a natural boundary.
Consequently, fix $i\in\{1,\dots,4\}$ and define
\[
\beta^{\mathrm{lac}}(\mathbf{m}):=\begin{cases}(p-1)!\,C^{p-1}, & \mathbf{m}=p\,e_{i},\ p-1\in\{2^{j}:j\ge0\},\\
0,&\text{otherwise.}\end{cases}
\]
Then $\beta^{\mathrm{lac}}\in\mathcal{A}_{C}$, and for every $\mathbf{c}\in\mathbb{R}^{4}$ with $c_{i}\ne0$,
\[
\widehat{B}(\tau\mathbf{c})=\frac{w\,g(w)}{C},\qquad w=C c_{i}\tau ,
\]
so that $\tau\mapsto\widehat{B}(\tau\mathbf{c})$ has radius of convergence exactly $1/(C|c_{i}|)$ and admits
\emph{no} analytic continuation to any open set meeting that circle. Choosing $i$ with
$|c_{i}|=\|\mathbf{c}\|_{\infty}$, we conclude: for every $\mathbf{c}\ne0$ there is an admissible coefficient
family for which the Borel transform along $\mathbf{c}$ cannot be continued past
$|\tau|=1/(C\|\mathbf{c}\|_{\infty})$ in any direction.
<!-- end theorem-7 -->
<!-- proof-20: body_md -->
\emph{Natural boundary.} Let $\omega=e^{2\pi i a/2^{n}}$ be a dyadic root of unity ($a,n$ integers,
$n\ge0$). For $j\ge n$ we have $\omega^{2^{j}}=1$, so for $0<r<1$,
\[
g(r\omega)=\sum_{j<n}(r\omega)^{2^{j}}+\sum_{j\ge n}r^{2^{j}},
\qquad\text{hence}\qquad
|g(r\omega)|\ \ge\ \sum_{j\ge n}r^{2^{j}}-n .
\]
For any $K\in\mathbb{N}$, $\sum_{j=n}^{n+K}r^{2^{j}}\to K+1$ as $r\uparrow1$, so
$\liminf_{r\uparrow1}\sum_{j\ge n}r^{2^{j}}\ge K+1$ for every $K$; thus
$|g(r\omega)|\to\infty$ as $r\uparrow1$. Suppose now $g$ extended holomorphically to an open set $U$
with $U\cap\{|w|=1\}\ne\emptyset$; pick $\zeta_{0}\in U$ with $|\zeta_{0}|=1$ and $\varepsilon>0$ with the
disc $D(\zeta_{0},2\varepsilon)\subset U$. Dyadic roots of unity are dense in the unit circle (the numbers
$a/2^{n}$ are dense in $[0,1]$), so choose such an $\omega$ with $|\omega-\zeta_{0}|<\varepsilon$. For
$1-\varepsilon<r\le1$ we get $|r\omega-\zeta_{0}|\le(1-r)+|\omega-\zeta_{0}|<2\varepsilon$, so the compact segment
$\{r\omega:1-\varepsilon/2\le r\le1\}$ lies in $U$; the extension is continuous, hence bounded there,
contradicting $|g(r\omega)|\to\infty$.

\emph{The coefficient family.} Hypothesis \ref{hyp:H} holds with equality on the support. Only the
multi--indices $\mathbf{m}=p\,e_{i}$ contribute, so $b_{p}(\mathbf{c})=\beta^{\mathrm{lac}}(p\,e_{i})c_{i}^{p}$ and
\[
\widehat{B}(\tau\mathbf{c})=\sum_{p:\,p-1\in\{2^{j}\}}\frac{(p-1)!C^{p-1}}{(p-1)!}c_{i}^{p}\tau^{p}
=\frac{1}{C}\sum_{j\ge0}(Cc_{i}\tau)^{2^{j}+1}=\frac{w\,g(w)}{C},\qquad w=Cc_{i}\tau .
\]
If $\tau\mapsto\widehat{B}(\tau\mathbf{c})$ extended holomorphically to an open set meeting
$|\tau|=1/(C|c_{i}|)$, then, dividing by the nonvanishing holomorphic function $w/C$, so would $g$
across a point of the unit circle --- impossible.

<!-- corollary-2: body_md -->
\label{cor:noborel}
For each $\mathbf{c}\ne0$ there is an admissible coefficient family for which all of the following are
\emph{undefined}; consequently none of them is defined as a functional on the class
$\mathcal{A}_{C}$: the Borel sum
$\int_{0}^{\infty}e^{-\tau}\widehat{B}(\tau\mathbf{c})d\tau/\tau$; the lateral Borel sums; their median or any
weighted average; principal values; accelero--summation; and any procedure requiring the analytic
continuation of $\tau\mapsto\widehat{B}(\tau\mathbf{c})$ beyond $|\tau|=1/(C\|\mathbf{c}\|_{\infty})$, or an exponential
bound on such a continuation. Even in the mildest case --- the family of Proposition
\ref{prop:borel}(iv), for which the continuation exists --- the Borel integral diverges whenever
$c_{1}>0$, since the integrand has a simple pole at $\tau=1/(Cc_{1})$ on the contour.
<!-- end corollary-2 -->
<!-- proof-21: body_md -->
Immediate from Theorem \ref{thm:natural} and Proposition \ref{prop:borel}(iv): in the first case
there is no function to integrate beyond the disc; in the second the integrand
$e^{-\tau}\frac{Cc_{1}^{2}\tau}{1-Cc_{1}\tau}\cdot\frac1\tau$ has a nonzero residue at a point of
$(0,\infty)$, so $\int_{0}^{1/(Cc_{1})-\varepsilon}$ diverges logarithmically as $\varepsilon\downarrow0$.
<!-- end proof-21 -->
<!-- remark-8: body_md -->
Corollary \ref{cor:noborel} is precisely why a \emph{cut} appears in Definition \ref{def:rule}: it is
not a technical convenience but the only device compatible with the information contained in
Hypothesis \ref{hyp:H}. It also explains why criteria of Watson or Nevanlinna--Sokal type cannot be
invoked here: their hypotheses include holomorphy of the Borel transform on a neighbourhood of
$[0,\infty)$ together with an exponential bound, and Theorem \ref{thm:natural} shows that no such
input is available from coefficient bounds alone.

<!-- theorem-8: body_md -->
\label{thm:nocanonical}
Let $\Omega\subseteq\mathbb{R}^{4}\setminus\{0\}$ be nonempty and let $V$ be the linear span of
$\mathcal{A}_{C}$. There is no linear map $S:V\to\mathbb{R}^{\Omega}$ such that
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.2\parindent]
\item $(S\beta)(\mathbf{c})=\sum_{p\ge2}b_{p}(\mathbf{c})$ for every finitely supported $\beta$ \textup{(}i.e. $S$
is exact on polynomials\textup{)}, and
\item for some $\mathbf{c}\in\Omega$ the functional $\beta\mapsto(S\beta)(\mathbf{c})$ is bounded on
$(V,\|\cdot\|_{C})$, or is sequentially continuous along coefficientwise convergent sequences in
$\mathcal{A}_{C}$.
\end{enumerate}
<!-- end theorem-8 -->
<!-- proof-22: body_md -->
Fix $\mathbf{c}\in\Omega$ and choose signs $s_{\mathbf{m}}\in\{\pm1\}$ with $s_{\mathbf{m}}\mathbf{c}^{\mathbf{m}}=|\mathbf{c}^{\mathbf{m}}|$. Let
$\beta^{\sharp}(\mathbf{m}):=(|\mathbf{m}|-1)!C^{|\mathbf{m}|-1}s_{\mathbf{m}}$ and let $\beta^{(k)}$ be its truncation to
$|\mathbf{m}|\le k$. Then $\beta^{(k)}$ is finitely supported, $\|\beta^{(k)}\|_{C}=1$,
$\beta^{(k)}\to\beta^{\sharp}\in\mathcal{A}_{C}$ coefficientwise, and by (a)
\[
(S\beta^{(k)})(\mathbf{c})=\sum_{p=2}^{k}(p-1)!C^{p-1}h_{p}(|\mathbf{c}|)\ \ge\ \sum_{p=2}^{k}(p-1)!C^{p-1}\|\mathbf{c}\|_{\infty}^{p}
\ \xrightarrow[k\to\infty]{}\ \infty
\]
by Lemma \ref{lem:count}(c). This contradicts boundedness (the $\beta^{(k)}$ lie in the unit ball)
and also sequential continuity along $\beta^{(k)}\to\beta^{\sharp}$.

<!-- theorem-9: body_md -->
\label{thm:dichotomy}
Fix $\mathbf{c}\in\mathbb{R}^{4}\setminus\{0\}$, put $\mu=\|\mathbf{c}\|_{\infty}$, let $(w_{p})_{p\ge2}$ be real numbers
and consider the linear rule $\mathcal{S}_{w}B(\mathbf{c}):=\sum_{p\ge2}w_{p}\,b_{p}(\mathbf{c})$. Then:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2\parindent]
\item \textup{(Well--definedness forces factorial decay.)} If the series converges absolutely for
every $\beta\in\mathcal{A}_{C}$, then
\[
M:=\sum_{p\ge2}|w_{p}|\,(p-1)!\,C^{p-1}\mu^{p}\ <\ \infty,
\qquad\text{so}\qquad
|w_{p}|\le\frac{M}{(p-1)!\,C^{p-1}\mu^{p}}\ \xrightarrow[p\to\infty]{}\ 0 .
\]
\item \textup{(Regularity forces all weights to be $1$.)} If $\mathcal{S}_{w}B(\mathbf{c})=B(\mathbf{c})$ for every
$\beta\in\mathcal{A}_{C}$ whose series converges absolutely at $\mathbf{c}$, then $w_{p}=1$ for all $p\ge2$.
\item \textup{(Dichotomy.)} \textup{(i)} and \textup{(ii)} are mutually exclusive. In particular no
linear weight rule is at once well defined on $\mathcal{A}_{C}$ and regular. Classical Borel
summation is the case $w_{p}\equiv1$: it is regular, and by Theorem \ref{thm:divergence} its defining
series diverges for an admissible family, while by Theorem \ref{thm:natural} its integral form is
undefined on the class.
\end{enumerate}

<!-- theorem-10: body_md -->
\label{thm:defect}
Let $\mathbf{c}\ne0$, $\mu=\|\mathbf{c}\|_{\infty}$, and let $(w_{p})$ satisfy Theorem \ref{thm:dichotomy}(i)
with constant $M$. Put $\sigma:=\tfrac12\mu\min\{1,C\}$ and assume $\sigma<1$, and let $p_{*}\ge2$ be

\[
\big|\mathcal{S}_{w}B(\mathbf{c})-B(\mathbf{c})\big|\ \ge\ \frac{\sigma^{p_{*}}}{2(1-\sigma)}\ >\ 0,
\qquad\text{with}\qquad
p_{*}\ \le\ 2+\frac{e^{2}}{C\mu}+\log\frac{2M}{\mu}\ \ \big(\text{if } 2M\ge\mu\big).
\]
Thus every rule that is well defined on $\mathcal{A}_{C}$ misses the ordinary sum of some convergent
admissible series by a definite, strictly positive amount; the amount is beyond all orders in $\mu$,
and by Proposition \ref{prop:convcase} the rule of Definition \ref{def:rule} attains that order.

Choose $i_{0}$ with $|c_{i_{0}}|=\mu$ and set $\beta(p\,e_{i_{0}}):=(\sigma/c_{i_{0}})^{p}$ for

\textup{(}$\beta(p\,e_{1})=(p-1)!C^{p-1}$, all other coefficients zero\textup{)} and let

\[
\Big|g_{\lambda}(c_{1})-\sum_{p=2}^{N}b_{p}(\mathbf{c})\Big|\;\le\;\frac{e+2\pi}{C}\,(N+1)!\;x^{N+1}.
\]

\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.2\parindent]

\item consequently no rule can lie within $o\big(e^{-1/(C\|\mathbf{c}\|_{\infty})}\big)$ of every real

exponent $1$ in the scale $C\|\mathbf{c}\|_{\infty}$ cannot be passed;
\item by Theorem \ref{thm:optimal} and Remark \ref{rem:ratelimit} the rules $\mathcal{S}_{k,\vartheta}$ realise
every rate $\eta<1$. Hence the threshold exponent is exactly $\eta=1$ in the scale
$C\|\mathbf{c}\|_{\infty}$, and the construction is rate--optimal.

them. (b) restates (a): a rule lying within $o(e^{-1/x})$ of every function consistent with the
data --- in particular of both $g_{0}$ and $g_{1}$ --- would contradict (a), since
$|g_{1}-g_{0}|=(2\pi/C)e^{-1/x}$. (c) is Theorem \ref{thm:optimal} together with

<!-- lemma-8: body_md -->
\label{lem:stirling}
For every integer $n\ge1$: $\ (n/e)^{n}\le n!\le e\,n^{n+1/2}e^{-n}$.
<!-- end lemma-8 -->
<!-- proof-32: body_md -->
The lower bound follows from $e^{n}=\sum_{j\ge0}n^{j}/j!\ge n^{n}/n!$. For the upper bound put
$a_{n}:=n!\,e^{n}n^{-n-1/2}$; then
\[
\frac{a_{n}}{a_{n+1}}=\frac{1}{e}\Big(1+\frac1n\Big)^{n+1/2}.
\]
For $x>0$ let $\phi(x)=\ln(1+x)-\frac{2x}{2+x}$; then $\phi(0)=0$ and
$\phi'(x)=\frac{1}{1+x}-\frac{4}{(2+x)^{2}}=\frac{x^{2}}{(1+x)(2+x)^{2}}>0$, so
$\ln(1+x)>\frac{2x}{2+x}$ for $x>0$. With $x=1/n$ this gives
$(n+\frac12)\ln(1+\frac1n)>(n+\frac12)\frac{2/n}{2+1/n}=1$, i.e. $(1+\frac1n)^{n+1/2}>e$ and
$a_{n}>a_{n+1}$. Hence $a_{n}\le a_{1}=e$ for all $n\ge1$.

<!-- lemma-9: body_md -->
\label{lem:leastterm}
For $0<y\le1$,
\[

<!-- proof-33: body_md -->

$N!y^{N}\le e\sqrt N\,(Ny/e)^{N}\le e\sqrt N e^{-N}\le e\cdot y^{-1/2}\cdot e^{\,1-1/y}$.

satisfies the iterated Cauchy formula
$f_{j}(\mathbf{z})=(2\pi i)^{-n}\int_{\partial_{0}\Delta}f_{j}(\zeta)\prod_{i}(\zeta_{i}-z_{i})^{-1}d\zeta$
for $\mathbf{z}\in\Delta$; letting $j\to\infty$, using uniform convergence on the compact set
$\partial_{0}\Delta$, the same formula holds for $f$. Expanding
$\prod_{i}(\zeta_{i}-z_{i})^{-1}$ in a multiple geometric series, uniformly convergent for $\mathbf{z}$ in
a compact subset of $\Delta$ and $\zeta\in\partial_{0}\Delta$, and integrating term by term exhibits
$f$ as a convergent power series on $\Delta$.

<!-- lemma-11: body_md -->
\label{lem:cauchy}
Let $f$ be holomorphic on a neighbourhood of the closed polydisc $\overline{\Delta}(a,(r,\dots,r))
\subset\mathbb{C}^{n}$. Then for every multi--index $\alpha$,
$|\partial^{\alpha}f(a)|\le\alpha!\,r^{-|\alpha|}\sup_{\overline{\Delta}}|f|$.
<!-- end lemma-11 -->
<!-- proof-35: body_md -->
The Taylor coefficients $c_{\alpha}=\partial^{\alpha}f(a)/\alpha!$ satisfy
$c_{\alpha}=(2\pi i)^{-n}\int_{\partial_{0}\Delta}f(\zeta)\prod_{i}(\zeta_{i}-a_{i})^{-\alpha_{i}-1}d\zeta$,
whence $|c_{\alpha}|\le r^{-|\alpha|}\sup_{\partial_{0}\Delta}|f|$.

<!-- lemma-12: body_md -->
\label{lem:flat}
Let $B_{r}\subseteq\mathbb{R}^{n}$ be an open ball centred at $0$ and let $g\in C(B_{r})\cap C^{\infty}(B_{r}\setminus\{0\})$
be such that every partial derivative $\partial^{\alpha}g$ ($\alpha\ne0$) extends continuously to
$B_{r}$; call the extension $g_{\alpha}$. Then $g\in C^{\infty}(B_{r})$ and
$\partial^{\alpha}g=g_{\alpha}$ throughout $B_{r}$.
<!-- end lemma-12 -->
<!-- proof-36: body_md -->
It suffices to prove $g\in C^{1}(B_{r})$ with $\partial_{j}g=g_{e_{j}}$; the general case follows by
induction, applying that statement to $g_{e_{j}}$ (which is continuous on $B_{r}$, smooth off the
origin, and all of whose derivatives extend continuously, being $g_{\alpha+e_{j}}$). Fix
$\mathbf{c}\in B_{r}\setminus\{0\}$. For $0<\varepsilon\le1$ the segment $\{s\mathbf{c}:\varepsilon\le s\le1\}$ avoids the
origin, so
$g(\mathbf{c})-g(\varepsilon\mathbf{c})=\int_{\varepsilon}^{1}\sum_{j}c_{j}\,\partial_{j}g(s\mathbf{c})\,ds$. Letting $\varepsilon\downarrow0$
and using the continuity of $g$ at $0$ and of $s\mapsto\sum_{j}c_{j}g_{e_{j}}(s\mathbf{c})$ on $[0,1]$,
\[
g(\mathbf{c})-g(0)=\int_{0}^{1}\sum_{j}c_{j}\,g_{e_{j}}(s\mathbf{c})\,ds .
\]
Hence $\big|g(\mathbf{c})-g(0)-\sum_{j}c_{j}g_{e_{j}}(0)\big|\le\|\mathbf{c}\|_{1}\max_{j}\sup_{0\le s\le1}
|g_{e_{j}}(s\mathbf{c})-g_{e_{j}}(0)|=o(\|\mathbf{c}\|)$ as $\mathbf{c}\to0$, by continuity of the $g_{e_{j}}$ at $0$.
So $g$ is differentiable at $0$ with $\partial_{j}g(0)=g_{e_{j}}(0)$, and $\partial_{j}g=g_{e_{j}}$
is continuous on $B_{r}$.

<!-- lemma-13: body_md -->
\label{lem:uniqueasy}
Let $f$ be a function defined on a punctured neighbourhood of $0$ in $\mathbb{R}^{n}$ and let
$(b_{p})_{p\ge2}$, $(b_{p}')_{p\ge2}$ be two families of homogeneous polynomials, $\deg b_{p}=\deg b_{p}'=p$,
such that for every $N$,
$f(\mathbf{c})-\sum_{p=2}^{N}b_{p}(\mathbf{c})=o(\|\mathbf{c}\|^{N})$ and $f(\mathbf{c})-\sum_{p=2}^{N}b_{p}'(\mathbf{c})=o(\|\mathbf{c}\|^{N})$
as $\mathbf{c}\to0$. Then $b_{p}=b_{p}'$ for all $p$. Consequently, if $f$ is real--analytic at $0$ and
admits $(b_{p})$ as its asymptotic expansion to all orders, then $\sum_{p}b_{p}$ is the Taylor series
of $f$ at $0$ and in particular converges near $0$.
<!-- end lemma-13 -->
<!-- proof-37: body_md -->
Induction on $p$. If $b_{q}=b_{q}'$ for $2\le q<p$, then subtracting the two hypotheses with $N=p$
gives $b_{p}(\mathbf{c})-b_{p}'(\mathbf{c})=o(\|\mathbf{c}\|^{p})$; putting $\mathbf{c}=t\mathbf{u}$ with $\|\mathbf{u}\|=1$ and $t\downarrow0$
gives $t^{p}(b_{p}(\mathbf{u})-b_{p}'(\mathbf{u}))=o(t^{p})$, so $b_{p}=b_{p}'$ on the unit sphere and hence, by
homogeneity, everywhere. For the last statement, a function real--analytic at $0$ is the sum of an
absolutely convergent power series on some polydisc; grouping its monomials by total degree produces
homogeneous polynomials $f_{p}$ with $f(\mathbf{c})-\sum_{p\le N}f_{p}(\mathbf{c})=O(\|\mathbf{c}\|^{N+1})$, so
$(f_{p})$ is an asymptotic expansion of $f$ and the first part gives $f_{p}=b_{p}$.
