<!-- theorem-1: body_md -->
\label{thm:p}
For the parallel CMAC-CM, the secrecy capacity region is given by
\begin{align}
{\mathcal C}_s^{\rm [P]}=& \bigcup_{ \substack{\prod_j
p(q_j,x_{2,j})p(u_j|q_j)p(x_{1,j}|u_j)\\p(y_j,y_{2,j}|x_{1,j},x_{2,j})}}
         
  
\left\{
\begin{array}{l}
(R_0,\,R_1):\\
~~R_0\ge 0,\; R_1\ge 0;\\
~~R_1\le \sum_{j=1}^{L} [I(U_j; Y_j|X_{2,j},Q_j)
   
            
-I(U_j;Y_{2,j}|X_{2,j},Q_j)] \\
~~R_0\le \sum_{j=1}^{L} I(Q_j,X_{2,j}; Y_j)
\end{array}
\right\} \label{eq:p-gb}
\end{align}
where $Q_j$ and $U_j$'s are auxiliary random variables, and $Q_j$ can be chosen
to be a deterministic function of $U_j$ for $j=1,\dots,L$.

<!-- theorem-2: body_md -->
\label{thm:pd}
For the parallel CMAC-CM with degraded subchannels, the secrecy capacity region is given
by
\begin{align}
{\mathcal C}_s^{\rm [D]}=& \bigcup_{ \substack{ \prod_{j}
p(q_j,x_{2,j})p(x_{1,j}|q_j)\\p(y_j,y_{2,j}|x_{1,j},x_{2,j})}}
         
  
\left\{
\begin{array}{l}
(R_0,\,R_1):\\
R_0\ge 0,\; R_1\ge 0;\\
~~R_1\le \sum_{j\in {\mathcal A}} [I(X_{1,j}; Y_j|X_{2,j},Q_j)
   
            
-I(X_{1,j};Y_{2,j}|X_{2,j},Q_j)] \\
~~R_0\le \sum_{j\in {\mathcal A}} I(Q_j,X_{2,j}; Y_j)
   
             
+\sum_{j\in \bar{{\mathcal A}}} I(X_{1,j},X_{2,j}; Y_j)
\end{array}
\right\}\label{eq:pd-mac}
\end{align}
where $Q_j$, for $j=1,\dots,L$, are auxiliary random variables that satisfy the
Markov chain relationship
\begin{align}
Q_j\rightarrow (X_{1,j},X_{2,j}) \rightarrow (Y_j,Y_{2,j}).
\end{align}

<!-- theorem-3: body_md -->
\label{thm:pG-MAC}
For the parallel Gaussian CMAC-CM, the secrecy capacity region is given by
\begin{align}
{\mathcal C}_s^{\rm [G]}=& \bigcup_{\underline{p}\in {\mathcal P}}
         
  
\left\{
\begin{array}{l}
(R_0,\,R_1):\\
~~R_0\ge 0,\; R_1\ge 0;\\
~~R_1\le \sum_{j\in {\mathcal A}}
\left[\frac{1}{2}\log\left(1+\frac{b_j}{\nu_j}\right)
          
                     
-\frac{1}{2}\log\left(1+\frac{b_j}{\mu_j}\right)\right] \\
~~R_0\le \sum_{j\in {\mathcal A}}
\frac{1}{2}\log\left(1+\frac{a_j+p_{2,j}+2\sqrt{a_jp_{2,j}} }{b_j+\nu_j}\right)
\\
\qquad + \sum_{j\in \bar{{\mathcal A}}} \frac{1}{2}\log\left(1+\frac{a_j+p_{2,j}+2\sqrt{
a_jp_{2,j}}}{\nu_j}\right)
\end{array}
\right\}\label{eq:pG-mac}
\end{align}
where $\underline{p}$ is the power allocation vector, which consists of
$(a_j,b_j,p_{2,j})$ for $j\in{\mathcal A}$ and $(a_j,0,p_{2,j})$ for $j\in \bar{{\mathcal A}}$ as
components, and the set ${\mathcal P}$ includes all power allocation vectors
$\underline{p}$ that satisfy the power constraint
\begin{align}
{\mathcal P}:=\left\{\underline{p}: \sum_{j=1}^{L} (a_j+b_j) \le
P_1~\text{and}~\sum_{j=1}^{L} p_{2,j} \le P_2\right\}. \label{eq:pow0}
\end{align}

<!-- theorem-4: body_md -->
\label{thm:op1}
Let $\underline{p}^{\star}$ be an optimal solution to the optimization problem
of (\ref{eq:opt1}) that achieves the boundary of the secrecy capacity region of
the parallel Gaussian CMAC-CM. Then, $\underline{p}^{\star}$ can be written as follows.

For $j\in {\mathcal A}$, if
\begin{align}
\frac{2\lambda_1^2\ln2}{\lambda_1+\lambda_2} <
\frac{\gamma_1(\mu_j-\nu_j)-\mu_j}{\mu_j\nu_j},
\end{align}
then
\begin{align}
a_j^{\star}&
=\frac{\lambda_2^2}{(\lambda_1+\lambda_2)^2}\left(s_{1,j}-\phi_j\right)^{+}, \notag\\
b_j^{\star}&=\left( \min\left[s_{2,j},\,\phi_j \right]\right)^{+} \notag\\
\text{and}\qquad p_{2,j}^{\star}&=\frac{\lambda_1^2}{(\lambda_1+\lambda_2)^2}\left(s_{1,j}-\phi_j\right)^{+};
\end{align}
alternatively, if
\begin{align}
\frac{2\lambda_1^2\ln2}{\lambda_1+\lambda_2} \ge
\frac{\gamma_1(\mu_j-\nu_j)-\mu_j}{\mu_j\nu_j},
\end{align}
then
\begin{align}
a_j^{\star}&=\frac{\lambda_2^2}{(\lambda_1+\lambda_2)^2}\left(s_{1,j}\right)^{+}, \notag\\
b_j^{\star}&=0 \notag\\
\text{and} \qquad
p_{2,j}^{\star}&=\frac{\lambda_1^2}{(\lambda_1+\lambda_2)^2}\left(s_{1,j}\right)^{+};
\end{align}
for $j\in \bar{{\mathcal A}}$,
\begin{align}
a_j^{\star}=\frac{\lambda_2^2}{(\lambda_1+\lambda_2)^2}\left(s_{1,j}\right)^{+}
\quad \text{and} \quad
p_{2,j}^{\star}&=\frac{\lambda_1^2}{(\lambda_1+\lambda_2)^2}\left(s_{1,j}\right)^{+};
\end{align}
where $\gamma_1 \ge 0$,
\begin{align}
s_{1,j}&=\frac{\lambda_1+\lambda_2}{2\lambda_1\lambda_2\ln 2}-\nu_j , \notag\\
s_{2,j}&=\frac{1}{2}\left[\sqrt{(\mu_j-\nu_j)\left(\mu_j-\nu_j+\frac{2\gamma_1}{\lambda_1\ln
2}\right)}-(\mu_j+\nu_j)\right], \notag\\
\phi_j&=-\frac{1}{2}\left(\mu_j+\nu_j+\frac{1}{\omega}\right)+\frac{1}{2}\sqrt{\left(\mu_j+\nu_j+
\frac{1}{\omega}\right)^2-
4\left[\mu_j\nu_j-\frac{\gamma_1(\mu_j-\nu_j)-\mu_j}{\omega}\right]},\notag\\
\omega&=(2\ln2)\frac{\lambda_1^2}{\lambda_1+\lambda_2}
\end{align}
and the pair $(\lambda_1,\lambda_2)$ is chosen to satisfy the power constraint
\begin{align}
\sum_{j=1}^{L} (a_j+b_j) \le P_1~\text{and}~\sum_{j=1}^{L} p_{2,j} \le P_2.
\label{eq:pow1}
\end{align}

<!-- corollary-1: body_md -->
\label{cor:f-MAC}
The secrecy capacity region of the fading CMAC-CM is given by (\ref{eq:F-mac})
\begin{align}
{\mathcal C}_s^{\rm [F]}= &\bigcup_{p(\underline{h},\underline{g}) \in {\mathcal P}}
         
  
\left\{
\begin{array}{l}
(R_0,\,R_1):\\
~~R_0\ge 0,\; R_1\ge 0;\\
~~R_1\le {\sf E}_{(\underline{h},\underline{g}) \in {\mathcal A}} \left[\log\left(1+
\frac{b(\underline{h},\underline{g})|h_1|^2}{\nu}\right)
          
                     
-\log\left(1+\frac{b(\underline{h},\underline{g})|g_1|^2}
{\mu}\right)\right] \\
~~R_0\le {\sf E}_{(\underline{h},\underline{g}) \in {\mathcal A}} \log\left(1+\frac{\chi(\underline{h},\underline{g})}
{b(\underline{h},\underline{g})|h_1|^2+\nu}\right)
   
         
+ {\sf E}_{(\underline{h},\underline{g})\in \bar{{\mathcal A}}}
\log\left(1+\frac{\chi(\underline{h},\underline{g})}{\nu}\right)
\end{array}
\right\}\label{eq:F-mac}
\end{align}
where
\begin{align}
\chi(\underline{h},\underline{g})&=\left[\sqrt{a(\underline{h},\underline{g})}|h_1| +\sqrt{p_2(\underline{h},\underline{g})}|h_2|\right]^2
\end{align}
and the random vector pair $(\underline{h},\underline{g})$ has the same distribution as the
marginal distribution of the process $\{(\underline{h}_i,\underline{g}_i)\}$ at a single time
instant.

<!-- corollary-2: body_md -->
\label{cor:op2}
Let $p(\underline{h},\underline{g})^{\star}$ be an optimal power allocation that achieves the boundary of the
secrecy capacity region of the fading CMAC-CM. Then, $p(\underline{h},\underline{g})^{\star}$ is given as follows:
\begin{itemize}
  \item for $(\underline{h},\underline{g})\in {\mathcal A}$, if
  \begin{align}
   \frac{\lambda_1^2|h_2|^2\ln2}{\lambda_1|h_2|^2+\lambda_2|h_1|^2} <
    \frac{\gamma_1\left(\mu|h_1|^2-\nu|g_1|^2\right)-\mu|h_1|^2}{\mu\nu},
  \end{align}
  then
  \begin{align}
    a^{\star}(\underline{h},\underline{g})&=\frac{\lambda_2^2|h_1|^2}{\left(\lambda_1|h_2|^2+\lambda_2|h_1|^2\right)^2}\left[s_1(\underline{h},\underline{g})-\phi(\underline{h},\underline{g})\right]^{+},\notag\\
    b^{\star}(\underline{h},\underline{g})&=\left(\min\left[s_2(\underline{h},\underline{g}),\,\phi(\underline{h},\underline{g})\right]\right)^{+}\notag\\
    \text{and} \qquad
    p_2^{\star}(\underline{h},\underline{g})&=\frac{\lambda_1^2|h_2|^2}{\left(\lambda_1|h_2|^2+\lambda_2|h_1|^2\right)^2}\left[s_1(\underline{h},\underline{g})-\phi(\underline{h},\underline{g})\right]^{+};
  \end{align}
  alternatively, if
  \begin{align}
   \frac{\lambda_1^2|h_2|^2\ln2}{\lambda_1|h_2|^2+\lambda_2|h_1|^2} \ge
    \frac{\gamma_1\left(\mu|h_1|^2-\nu|g_1|^2\right)-\mu|h_1|^2}{\mu\nu},
  \end{align}
  then
  \begin{align}
    a^{\star}(\underline{h},\underline{g})&=\frac{\lambda_2^2|h_1|^2}{\left(\lambda_1|h_2|^2+\lambda_2|h_1|^2\right)^2}\left[s_1(\underline{h},\underline{g})\right]^{+},\notag\\
    b^{\star}(\underline{h},\underline{g})&=0\notag\\
    \text{and} \qquad
    p_2^{\star}(\underline{h},\underline{g})&=\frac{\lambda_1^2|h_2|^2}{\left(\lambda_1|h_2|^2+\lambda_2|h_1|^2\right)^2}\left[s_1(\underline{h},\underline{g})\right]^{+};
  \end{align}

\item for $(\underline{h},\underline{g})\in \bar{{\mathcal A}}$,
  \begin{align}
    a^{\star}(\underline{h},\underline{g})&=\frac{\lambda_2^2|h_1|^2}{\left(\lambda_1|h_2|^2+\lambda_2|h_1|^2\right)^2}\left[s_1(\underline{h},\underline{g})\right]^{+}\notag\\
    \text{and} \qquad
    p_2^{\star}(\underline{h},\underline{g})&=\frac{\lambda_1^2|h_2|^2}{\left(\lambda_1|h_2|^2+\lambda_2|h_1|^2\right)^2}\left[s_1(\underline{h},\underline{g})\right]^{+};
  \end{align}
\end{itemize}
where $\gamma_1 \ge 0$,
\begin{align}
s_1(\underline{h},\underline{g})&=\frac{\lambda_1|h_2|^2+\lambda_2|h_1|^2}{\lambda_1\lambda_2\ln 2}-\nu , \notag\\
s_2(\underline{h},\underline{g})&=\frac{1}{2}\left[\sqrt{\left(\frac{\mu}{|g_1|^2}-\frac{\nu}{|h_1|^2}\right)
               \left(\frac{\mu}{|g_1|^2}-\frac{\nu}{|h_1|^2}+\frac{2\gamma_1}{\lambda_1\ln 2}\right)}
               -\left(\frac{\mu}{|g_1|^2}+\frac{\nu}{|h_1|^2}\right)\right], \notag\\
\phi(\underline{h},\underline{g})&=-\frac{1}{2}\left(\frac{\mu}{|g_1|^2}+\frac{\nu}{|h_1|^2}+\frac{1}{\omega(\underline{h},\underline{g})}\right)\notag\\
  &\quad+\frac{1}{2}\sqrt{\left(\frac{\mu}{|g_1|^2}+\frac{\nu}{|h_1|^2}+
\frac{1}{\omega(\underline{h},\underline{g})}\right)^2-
4\left[\frac{\mu}{|g_1|^2}\frac{\nu}{|h_1|^2}-\frac{\gamma_1\left(\frac{\mu}{|g_1|^2}-\frac{\nu}{|h_1|^2}\right)-\frac{\mu}{|g_1|^2}}{\omega(\underline{h},\underline{g})}\right]} \notag\\
\omega(\underline{h},\underline{g})&=(\ln2)\frac{\lambda_1^2|h_2|^2}{\lambda_1|h_2|^2+\lambda_2|h_1|^2}
\end{align}
and the pair $(\lambda_1,\lambda_2)$ is chosen to satisfy the power constraint
\begin{align}
{\sf E}[a(\underline{h},\underline{g})+b(\underline{h},\underline{g})] \le P_1
\quad \text{and} \quad {\sf E}[p_2(\underline{h},\underline{g})] \le P_2. \label{eq:pow3}
\end{align}
