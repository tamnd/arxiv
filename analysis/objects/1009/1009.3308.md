\label{complex_est}
                                      
                                                                       
Let ${\bf u}_0 \in \mathcal D(\boldsymbol{\mathrm{A}}  ^{s+1/2})$ and
${\bf f} \in C([0,T];\mathcal D(\boldsymbol{\mathrm{A}}  ^{s} e^{\sigma_1 \boldsymbol{\mathrm{A}}  ^{1/2}}))$, with

\]
We assume further that ${\bf f}(\cdot,\zeta)$ is analytic for 
$\zeta\in\ensuremath{\mathbb T}$ and that
\begin{equation}\label{supf_in_T}
K := 
\sup \{ \|\boldsymbol{\mathrm{A}}  ^{s} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}} {\bf f}(\cdot,\zeta)\|^2 : \zeta=re^{i\theta} \in \ensuremath{\mathbb T}\} < \infty.

and let 
\[
{\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta) :=  \boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}} {\bf u}(\zeta).
\]
                                        
The definition of $\psi$ gives 
$\frac{d}{dx} \psi  :=  \psi'(x) \le 1$, and $\psi \le
\sigma_1$, thus for fixed $\theta$ we find
\begin{eqnarray}
\frac{d}{dr} {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta)  & = & 
\psi'(r\cos\theta) \cos\theta  \boldsymbol{\mathrm{A}}  ^{s+1} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}} {\bf u}(\zeta) 
\nonumber \\ 
& & 
+ e^{i\theta} \boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}} \frac{d{\bf u}}{d\zeta}(\zeta). \label{der2}
\end{eqnarray}
Using \eqref{der2} in 
\begin{eqnarray*}\label{der1}
\frac{1}{2}\frac{d}{dr} \left\|{\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta) \right\|^2 =
\Re\left(\frac{d}{dr} {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta), {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta) \right),
                                                                           
\end{eqnarray*}
where $\Re(\cdot)$ denotes the
real-part function,  we get
\begin{eqnarray}
\frac{1}{2} \frac{d}{dr} \| {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta)\|^2  
&=&
\psi'(r\cos\theta)\cos\theta\; 
 \Re\;(\boldsymbol{\mathrm{A}}  ^{1/2} {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta), {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta)) 
\nonumber  \\ &&
+\Re\; e^{i\theta} (\boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}} \frac{d{\bf u}}{d\zeta}(\zeta) , {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta)). 
\label{der3} \end{eqnarray}
Using \eqref{NSE_complex}  
for the last term in \eqref{der3} 
together with 
the fact that 
\[
(\boldsymbol{\mathrm{A}}  ^s e^{\psi(r\cos\theta)\boldsymbol{\mathrm{A}}  ^{1/2}}\boldsymbol{\mathrm{C}}  {\bf u},\boldsymbol{\mathrm{A}}  ^{s+1} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}}{\bf u}) = 0
\]
(see \cite[Lemma 1]{cao_rammaha_titi99}), we find 
\begin{eqnarray}
& & \frac{1}{2} \frac{d}{dr} \| {\bf u}_{\boldsymbol{\mathrm{A}}  } (\zeta) \|^2
+ \nu \cos\theta \|\boldsymbol{\mathrm{A}}  ^{1/2} {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta)  \|^2 
\nonumber \\  & = &  
\psi'(r\cos\theta) \cos\theta(\boldsymbol{\mathrm{A}}  ^{1/2} {\bf u}_{\boldsymbol{\mathrm{A}}  } ,{\bf u}_{\boldsymbol{\mathrm{A}}  } )
  - \Re\; e^{i\theta} ( \boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi \boldsymbol{\mathrm{A}}  ^{1/2}} \boldsymbol{\mathrm{B}}  ({\bf u},{\bf u}),{\bf u}_{\boldsymbol{\mathrm{A}}  }) 
\nonumber\\ & &

where in \eqref{main_ineq} and below we write 
${\bf u}_{\boldsymbol{\mathrm{A}}  } = {\bf u}_{\boldsymbol{\mathrm{A}}  }(\zeta),  {\bf u} = {\bf u}(\zeta), 
{\bf f} = {\bf f}(\zeta),  \psi =  \psi(r \cos\theta)$.

(with $p=\max\{2-s,7/4\} = 7/4$, since $s \geq 1/4$),
\begin{equation}\label{Buu_est}
|( \boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi \boldsymbol{\mathrm{A}}  ^{1/2}} \boldsymbol{\mathrm{B}}  ({\bf u},{\bf u}),{\bf u}_{\boldsymbol{\mathrm{A}}  })|

\end{equation}
Applying  $\psi' \leq 1$, the Cauchy-Schwarz inequality, \eqref{Buu_est} 
                                     
and Young's inequality ($ab \le a^q/q + b^{q'}/q'$ with $1/q+1/q'=1$)  
with $q=2$ and $q=2/(2-p)$

\begin{eqnarray*}
                                              
                                   
& & \frac{1}{2} \frac{d}{dr} \| {\bf u}_{\boldsymbol{\mathrm{A}}  } \|^2
+ \nu \cos\theta \|\boldsymbol{\mathrm{A}}  ^{1/2} {\bf u}_{\boldsymbol{\mathrm{A}}  }  \|^2 \\
&\le& \cos\theta \| \boldsymbol{\mathrm{A}}  ^{1/2} {\bf u}_{\boldsymbol{\mathrm{A}}  } \| \| {\bf u}_{\boldsymbol{\mathrm{A}}  }  \|

       \frac{\cos\theta}{\nu} \|{\bf u}_{\boldsymbol{\mathrm{A}}  } \|^2 \\
&&+ C^{\frac{2}{2-p}} \frac{(2-p)}{p}
    \left(\frac{p}{\nu\cos\theta}\right)^{\frac{p}{2-p}} 
 \| {\bf u}_{\boldsymbol{\mathrm{A}}  }  \|^{\frac{2(3-p)}{2-p}} + 
  \frac{\nu\cos\theta}{2} \|\boldsymbol{\mathrm{A}}  ^{1/2} {\bf u}_{\boldsymbol{\mathrm{A}}  }  \|^2\\
&&+\frac{1}{\nu\cos\theta}\|\boldsymbol{\mathrm{A}}  ^{s} e^{\psi \boldsymbol{\mathrm{A}}  ^{1/2}} {\bf f}\|^2+
 \frac{\nu\cos\theta}{4} \| \boldsymbol{\mathrm{A}}  ^{1/2} {\bf u}_{\boldsymbol{\mathrm{A}}  } \|^2.
\end{eqnarray*}

\begin{eqnarray} 
 \frac{d}{dr} \|{\bf u}_{\boldsymbol{\mathrm{A}}  }    \|^2
&\le& \frac{2 \cos\theta}{\nu} \|{\bf u}_{\boldsymbol{\mathrm{A}}  } \|^2

\eqref{bounded_soln-complex-local} holds for $0 \leq  r \le T^{**}$.

\label{complex_estT}
Suppose ${\bf u}_0$ and ${\bf f}$ satisfy all the conditions in 
the domain $\ensuremath{\mathbb T}$ as in Theorem \ref{complex_est}.  
                           
                                
                                                                   
    
Then 
\begin{equation}\label{bounded_soln-complex}
 \| \boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi(r\cos\theta)\boldsymbol{\mathrm{A}}  ^{1/2}} {\bf u}(\zeta) \|^2
    \le M_3,  \qquad \zeta \in \ensuremath{\mathbb T} \mbox{ and } 
    |\mbox{Im }\zeta| \le T^{**}/\sqrt{2},
\end{equation}
where $M_3:=1+2M_2^2$, $T^{**}$ depends on
$M_2$, and hence ${\bf u}(\cdot,\zeta) \in G^{s+1/2}_{\psi(r\cos\theta)}$
for all $\zeta \in \ensuremath{\mathbb T}$ with $|\mbox{Im }\zeta| \le T^{**}/\sqrt{2}$.
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
We proceed as in the proof of Theorem \ref{complex_est} to obtain
the ordinary differential equation
\[
  \frac{d}{dr} y \le C y^5,
\]
where $y(r) = 1+\|\boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}}{\bf u}(r e^{i\theta})\|^2$. On integrating
the inequality we find that
\[
  y(r) \le 2 y(0)
\]
provided that
\[
  0 \le r \le \frac{15}{64C} \frac{1}{ (1+\|\boldsymbol{\mathrm{A}}  ^{s+1/2} {\bf u}(0)\|^2)^4 }.
\]
We define
\[
T_1(\rho):= \frac{15}{64C} \frac{1}{ (1+\rho^2)^4 },\quad \rho \ge 0.
\]
If $T_1(\|\boldsymbol{\mathrm{A}}  ^{s+1/2}{\bf u}(0)\|) \ge T$ we have finished the proof. Otherwise, 
we let
\[
  T^{**} = T_1(M_2),
\]
where $M_2$ is given in \eqref{ass:solutionboundedT}.
For $\zeta = r e^{i\theta} \in \ensuremath{\mathbb T}$, $0 \leq r \le T^{**}$, 
\begin{equation}\label{eq:uT**-bdd}
 \|\boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\psi(r \cos\theta) \boldsymbol{\mathrm{A}}  ^{1/2}} {\bf u}(r e^{i\theta}) \|^2 \le 1 + 2\|\boldsymbol{\mathrm{A}}  ^{s+1/2} {\bf u}(0)\|^2.
\end{equation} 
Consequently, \eqref{bounded_soln-complex} holds for $0 \leq  r \le T^{**}$
with $M_3 = 1+2\|\boldsymbol{\mathrm{A}}  ^{s+1/2} {\bf u}(0)\|^2$. 

Next we consider the case $\zeta =  T^{**}+ r e^{i\theta}$, with 
$r \in [0,T^{**}]$. We define, 
for $|\theta| \leq \pi/4$,
\[
{\bf v}(re^{i\theta}) :=  {\bf u}(T^{**}+re^{i\theta}), \qquad r \in [0,T^{**}].
\]
                            
                          
                                                                              
                                      
Using 
                
$\|\boldsymbol{\mathrm{A}}  ^{s+1/2} {\bf v}(0)\| \le M_2$ we can apply 
the previous arguments 
                 
                                                      
to obtain \eqref{bounded_soln-complex} (with ${\bf u}$ replaced with ${\bf v}$)
for $0\le r\le T^{**}$. 
                      
                                  
                                                          
   
                                                                       
                                                                               
   
We complete the proof and obtain the bound \eqref{bounded_soln-complex} 
by repeating the last argument $n$ times, where $n=\lceil T/T^{**} \rceil$.

The proof follows by repeating  the arguments described 
in the first four pages 
of~\cite[Section~9]{constantin_foias}
(proving~\cite[Proposition~9.1]{constantin_foias}),
provided that we
establish~\cite[Inequalities~(9.3)~and~(9.13)]{constantin_foias}  
for our system~\eqref{app_weak_form} on the spherical surface
with additional Coriolis term and quadrature approximations. 

                                                           
Using~\eqref{skew}, \eqref{B} and the exactness
of the quadrature rule (given by \eqref{disc_ip}-\eqref{M_choice}), 
we have $\left(\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N),{\bf u}_N\right)_M = 0$.
Using \eqref{Coriolis_ZLm}, the symmetry of the coefficients of ${\bf u}_N$ 
in \eqref{app_soln_rep} and the exactness of the quadrature, we get  
\begin{equation}\label{CuNuN}
(\boldsymbol{\mathrm{C}}  {\bf u}_N,{\bf u}_N)_M=
(\boldsymbol{\mathrm{C}}  {\bf u}_N,{\bf u}_N) = 
(-2\Omega i) 
\sum_{L=1}^N \lambda^{-1}_L \sum_{|m|\le L} m|\alpha_{L,m}|^2 
= 0. 
\end{equation}
By taking ${\bf v}$ to be ${\bf u}_N$ in \eqref{app_weak_form} and 
using   $\|{\bf u}_N\|^2 = ({\bf u}_N,{\bf u}_N)_M$,
$\|{\bf u}_N\|_V^2 = (\boldsymbol{\mathrm{A}}   {\bf u}_N,{\bf u}_N)_M$,  \eqref{CuNuN}, 
Young's inequality and the fact that all the eigenvalues  
$\lambda_J$ of $\boldsymbol{\mathrm{A}}  $ 
(corresponding to eigenvectors in ${\bf u}_N$) satisfy 
$\lambda_J \geq \lambda_1 = 2, J = 1, \cdots, N$,
we obtain  
\[
\frac{1}{2} \frac{d}{dt} \|{\bf u}_N\|^2
+ \nu \|{\bf u}_N\|_V^2 
= ({\bf f},{\bf u}_N)_M 
\le \|{\bf f}\|_{\infty}  \|{\bf u}_N\|
\le  \frac{ \|{\bf f}\|^2_{\infty} } {4\nu} 
   + \nu \|{\bf u}_N\|^2 \leq \frac{ \|{\bf f}\|^2_{\infty} } {4\nu} + \frac{\nu}{2} \|{\bf u}_N\|_V^2,
\]
Hence, for our discrete system~\eqref{app_weak_form}, 
we obtain~\cite[Inequality~(9.3)]{constantin_foias}: 
\begin{equation}\label{ode_ineq}
 \frac{d}{dt} \|{\bf u}_N \|^2 + \nu \|{\bf u}_N\|_V^2 
  \le \frac{\|{\bf f}\|^2_{\infty}}{\nu \lambda_1}.
\end{equation}
Again using \eqref{Coriolis_ZLm}, the exactness of the
quadrature rule, eigenfunction properties of $\boldsymbol{\mathrm{A}}  $, and 
the symmetry of the coefficients of ${\bf u}_N$ in 
\eqref{app_soln_rep}, we get 
\begin{equation}\label{CuNAuN}
(\boldsymbol{\mathrm{C}}  {\bf u}_N,\boldsymbol{\mathrm{A}}  {\bf u}_N) = 
(-2\Omega i) 
\sum_{L=1}^N  \sum_{|m|\le L} m|\alpha_{L,m}|^2 
=0, 
\end{equation}
By taking ${\bf v}$ to be $\boldsymbol{\mathrm{A}}  {\bf u}_N$ in equation \eqref{app_weak_form},
and  using   $\|\boldsymbol{\mathrm{A}}  {\bf u}_N\|^2 = (\boldsymbol{\mathrm{A}}  {\bf u}_N,\boldsymbol{\mathrm{A}}  {\bf u}_N)_M$, \eqref{M_choice}, and  \eqref{CuNuN}, 
we obtain 
\begin{equation}\label{inprodAuN}
\frac{1}{2} \frac{d}{dt} \|{\bf u}_N\|^2_V +
\nu \|\boldsymbol{\mathrm{A}}   {\bf u}_N\|^2 + b({\bf u}_N,{\bf u}_N,\boldsymbol{\mathrm{A}}  {\bf u}_N) = ({\bf f},\boldsymbol{\mathrm{A}}  {\bf u}_N)_M.
\end{equation}
Using~\cite[Lemma~3.1]{ilin91},  $b({\bf u}_N,{\bf u}_N,\boldsymbol{\mathrm{A}}  {\bf u}_N) = 0$. 
The term $({\bf f},\boldsymbol{\mathrm{A}}  {\bf u}_N)_M$ can be estimated  by using the exactness
of the quadrature and Young's inequality:
\[
({\bf f},\boldsymbol{\mathrm{A}}  {\bf u}_N)_M \le \|{\bf f}\|_{\infty}  \|\boldsymbol{\mathrm{A}}  {\bf u}_N\|
\le  \frac{ \|{\bf f}\|^2_{\infty} } {2 \nu} 
   + \frac{\nu}{2} \|\boldsymbol{\mathrm{A}}  {\bf u}_N\|^2
\] 
Hence we obtain a stronger version of ~\cite[Inequality~(9.13)]{constantin_foias} for
our quadrature discrete scheme~\eqref{app_weak_form}: 
\[
 \frac{d}{dt} \|{\bf u}_N\|^2_V + \nu \|\boldsymbol{\mathrm{A}}  {\bf u}_N\|^2
 \le  \frac{ \|{\bf f}\|^2_{\infty} } {\nu \lambda_1}.
\]
Thus, the result follows from arguments in~\cite[Page~74-77]{constantin_foias}.

Let ${\bf w}_N = \widetilde{{\bf u}}_N - {\bf u}_N$, where the comparison
function  $\widetilde{{\bf u}}_N$ is the
solution of (\ref{elliptic1}). Since  
${\bf u} - {\bf u}_N = {\bf p}_N  + {\bf w}_N$, where
${\bf p}_N = {\bf u} - \widetilde{{\bf u}}_N$, in view of
Theorem~\ref{cont_est} and \ref{der_cont_est},
existence of $T^{\#}$ and
$\mu(t)$ follows and it is sufficient to show that 
$ \|{\bf w}_N\| \le   C  \left[ N^{-2s-1} e^{-\mu(t) N} +
 \left\|{\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f}\right\| \right]
 $.
For  any  ${\bf v} \in V_N$, using (\ref{op_form}), (\ref{elliptic1}), 
and (\ref{app1_weak_form}),
\begin{eqnarray*}
&& \hspace{-0.2in} (({\bf w}_N)_t, {\bf v}) + \nu (\boldsymbol{\mathrm{A}}  {\bf w}_N, {\bf v}) + (\boldsymbol{\mathrm{C}}  {\bf w}_N,{\bf v}) \\ 
&& \hspace{-0.2in} = 
      (({\widetilde{{\bf u}}_N})_t,{\bf v})+\nu (\boldsymbol{\mathrm{A}}  {\widetilde{{\bf u}}_N},{\bf v})+(\boldsymbol{\mathrm{C}}  {\widetilde{{\bf u}}}_N,{\bf v}) 
    -(({\bf u}_N)_t,{\bf v})-\nu(\boldsymbol{\mathrm{A}}  {\bf u}_N,{\bf v})-(\boldsymbol{\mathrm{C}}  {\bf u}_N,{\bf v}) \\
&& \hspace{-0.2in} =  
(({\widetilde{{\bf u}}_N})_t,{\bf v})+\nu(\boldsymbol{\mathrm{A}}  {\widetilde{{\bf u}}_N},{\bf v})+(\boldsymbol{\mathrm{C}}  {\widetilde{{\bf u}}}_N,{\bf v}) 
     -({\bf f},{\bf v})_M + (\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N),{\bf v}) \\
&& \hspace{-0.2in} =  
 (({\widetilde{{\bf u}}_N})_t,{\bf v})+\nu(\boldsymbol{\mathrm{A}}  {\bf u},{\bf v}) + (\boldsymbol{\mathrm{C}}  {\widetilde{{\bf u}}}_N,{\bf v}) 
     - ({\bf f},{\bf v})_M + (\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N),{\bf v}) \\
                      
                                                            
                             
                                                          
&& \hspace{-0.2in} = 
   (({\widetilde{{\bf u}}_N}-{\bf u})_t,{\bf v})+({\bf f},{\bf v}) - ({\bf f},{\bf v})_M + (\boldsymbol{\mathrm{C}}  {\widetilde{{\bf u}}_N}-\boldsymbol{\mathrm{C}}  {\bf u},{\bf v}) +
    (\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N)-\boldsymbol{\mathrm{B}}  ({\bf u},{\bf u}),{\bf v}).  
\end{eqnarray*}
Using the orthogonal projection  ${\mathbf \Pi}_N$ in  (\ref{fin_space_vec}), 
we can write this relation in functional form as 
\[
\frac{d{\bf w}_N}{dt} = (-\nu \boldsymbol{\mathrm{A}}  -\boldsymbol{\mathrm{C}}  ){\bf w}_N - \mathbf{{\mathbf \Pi}_N} \left[({\bf p}_N)_t + \boldsymbol{\mathrm{C}}  {\bf p}_N\right] 
+ {\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f} 
   + {\mathbf \Pi}_N\left[\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N) - \boldsymbol{\mathrm{B}}  ({\bf u},{\bf u})\right].
\]
Integrating with respect to $t$ and using  ${\bf w}_N(0)=0$, we have 
\begin{eqnarray}
 {\bf w}_N (t) &=& 
  \int_{0}^t e^{-(t-s)(\nu \boldsymbol{\mathrm{A}}  +\boldsymbol{\mathrm{C}}  )} \left[- {\mathbf \Pi}_N (\frac{d}{ds}{\bf p}_N + \boldsymbol{\mathrm{C}}  {\bf p}_N) + {\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f} \right](s) \;ds \nonumber \\
& &  + \int_{0}^t e^{-(t-s)(\nu \boldsymbol{\mathrm{A}}  +\boldsymbol{\mathrm{C}}  )} {\mathbf \Pi}_N \left[\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N)-\boldsymbol{\mathrm{B}}  ({\bf u},{\bf u})\right](s) \;ds. \label{wN-1}
\end{eqnarray}
Let
\begin{equation}\label{RN}
R_N(\epsilon, t-s)  = \|\nu^{\epsilon}{\mathbf \Pi}_N \boldsymbol{\mathrm{A}}  ^{\epsilon} e^{-(t-s)(\nu \boldsymbol{\mathrm{A}}  +\boldsymbol{\mathrm{C}}  )}\|.
\end{equation}
On using (\ref{weakLip}), with $\delta \in  (1/2,1)$,
and the uniform boundedness of the orthogonal projection ${\mathbf \Pi}_N$,  we get
\begin{eqnarray}
& & \|e^{-(t-s)(\nu \boldsymbol{\mathrm{A}}  +\boldsymbol{\mathrm{C}}  )}{\mathbf \Pi}_N\left[\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N)-\boldsymbol{\mathrm{B}}  ({\bf u},{\bf u})\right]\| \\
 & & \le \frac{R_N(\delta, t-s)}{\nu^{\delta}} \|\boldsymbol{\mathrm{A}}  ^{-\delta}{\mathbf \Pi}_N (\boldsymbol{\mathrm{B}}  ({\bf u}_N,{\bf u}_N)-\boldsymbol{\mathrm{B}}  ({\bf u},{\bf u}))\| \nonumber
 \le C R_N(\delta,t-s)  \|{\bf u}_N-{\bf u}\|. \label{bd_1}
\end{eqnarray}
Taking norms and using 
$\|{\bf u}_N -{\bf u}\| \le \|{\bf w}_N\|+\|{\bf p}_N\|$ together with (\ref{RN}), 
and (\ref{bd_1}) in (\ref{wN-1}),  we obtain
\begin{eqnarray*}
\|{\bf w}_N(t)\|   &\le &
\left[\|\frac{d}{dt}{\bf p}_N\| + \|\boldsymbol{\mathrm{C}}  {\bf p}_N\|
+ \left\|{\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f}\right\| \right]  
\int_{0}^{t} R_N(0, t-s)\;ds  \\        
&& + C \int_0^t R_N(\delta, t-s) (\|{\bf p}_N(s)\|+\|{\bf w}_N(s)\|) ds.
\end{eqnarray*}
Using Gronwall's inequality, we obtain for each $t \in [0,T]$,
\begin{eqnarray}
\|{\bf w}_N(t)\|  &\le & C \left\{
\left[\|\frac{d}{dt}{\bf p}_N\| + \|\boldsymbol{\mathrm{C}}  {\bf p}_N\|
+  \left\|{\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f}\right\| \right]  
\int_{0}^{t} R_N(0, t-s)\;ds   \right.  
\nonumber \\         && \qquad 
\left .
+ \|{\bf p}_N\|\  \int_0^t R_N(\delta, t-s)\;ds \right \}. \label{Gron}
\end{eqnarray}
For $\epsilon \in [0,1]$, we have to bound 
\begin{equation}\label{Rmore}
 \int_0^t R_N(\epsilon , t-s)\;ds
=  \int_0^t R_N(\epsilon , r)\;dr.
\end{equation}
Using also (\ref{RN}) and \eqref{Coriolis_ZLm},
\[
 R_N(\epsilon, r) \leq \max_{1\le L\le N; |m|\le L } 
      |(\nu\lambda_L)^{\epsilon} e^{-\nu \lambda_L r}e^{-2\Omega irm\lambda^{-1/2}_L}|
    \le \max_{ z \in[\nu\lambda_1,\nu\lambda_N]} z^\epsilon e^{-rz}.
\]
Thus
\begin{equation} \label{Rcases}
 R_N(r) \le\begin{cases}
         (\nu \lambda_N)^\epsilon e^{-\nu\lambda_N r} & \mbox{ if } r \le \epsilon/(\nu\lambda_N),  \cr
         \epsilon^\epsilon e^{-\epsilon} r^{-\epsilon} & 
            \mbox{ if } \epsilon/(\nu\lambda_N) \le r \le \epsilon/(\nu\lambda_1)  ,\cr
        (\nu \lambda_1)^\epsilon e^{-\nu\lambda_1 r} & \mbox{ if } r \ge \epsilon/(\nu\lambda_1).\cr
     \end{cases}
\end{equation}
With  
\begin{equation} \label{sub_int}
 I_1 = [0,\epsilon/(\nu\lambda_N)]\cap[0,t], \quad
 I_2 = [\epsilon/(\nu\lambda_N),\epsilon/(\nu\lambda_1)]\cap [0,t], \quad
 I_3 = [\epsilon/(\nu\lambda_1),t]\cap [0,t],
\end{equation}
the interval of integration in (\ref{Rmore}) can be subdivided into
these three intervals. In particular, using (\ref{Rcases}) and (\ref{sub_int}),
we get 
\begin{equation} \label{I1}
  \int_{I_1} R_N(r) dr \le \int_{I_1} (\nu\lambda_N)^\epsilon e^{-\nu\lambda_N r} dr
    \le \frac{(1-e^{-\epsilon})}{(\nu\lambda_N)^{1-\epsilon}} \leq C,
\end{equation}
\begin{equation} \label{I2}
  \int_{I_2} R_N(r) dr \le \int_{I_2} \epsilon^\epsilon e^{-\epsilon} r^{-\epsilon} dr
  \le \frac{\epsilon e^{-\epsilon}}{(1-\epsilon)}
  \left[ \left(\frac{1}{\nu\lambda_1}\right)^{(1-\epsilon)} - 
         \left(\frac{1}{\nu\lambda_N}\right)^{(1-\epsilon)}\right] \le C,
\end{equation}
and
\begin{equation} \label{I3}
 \int_{I_3} R_N(r) dr \le \int_{I_3} (\nu\lambda_1)^\epsilon e^{\nu \lambda_1 r} dr
  \le \frac{ (e^{-\epsilon}-e^{-\nu\lambda_1 t}) }
           {(\nu \lambda_1)^{1-\epsilon}} \le C. 
\end{equation}
Using (\ref{I1})- (\ref{I3}) in (\ref{Rmore}), we get
\begin{equation}\label{bd}
\int_0^t R_N(\epsilon , t-s)\;ds \le C, \qquad \epsilon \in [0,1].
\end{equation}
Substituting this in (\ref{Gron}), we get 
\begin{equation}\label{penul}
\|{\bf w}_N(t)\|  \le  C 
\left[\|\frac{d}{dt}{\bf p}_N\| + \|\boldsymbol{\mathrm{C}}  {\bf p}_N\| +  \|{\bf p}_N\|\  
+  \left\| {\mathbf \Pi}_N{\bf f} - {\mathbf L}_N {\bf f}\right\| \right].
\end{equation}
The term $\|\boldsymbol{\mathrm{C}}  {\bf p}_N \|$ in \eqref{penul} can be simplified 
using (\ref{C}) and the fact that ${\bf p}_N$ is tangential 
(and hence $\widehat{{\bf x}} \cdot {\bf p}_N(\widehat{{\bf x}}) = 0$),                                    
\[
\left[\widehat{{\bf x}} \times {\bf p}_N(\widehat{{\bf x}})\right] \cdot \left[\overline{(\widehat{{\bf x}} \times {\bf p}_N(\widehat{{\bf x}})}\right]  =  
\left[\widehat{{\bf x}} \cdot \widehat{{\bf x}}\right] \left[{\bf p}_N(\widehat{{\bf x}}) 
\cdot \overline{{\bf p}_N(\widehat{{\bf x}})}\right] = \| {\bf p}_N \|^2,
\]
and hence 
\begin{equation}\label{penul1}
\|{\bf w}_N(t)\|  \le  C 
\left[\|\frac{d}{dt}{\bf p}_N\| +  \|{\bf p}_N\|\  
+    \left\|{\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f}\right\| \right].
\end{equation}
Hence from Theorem~\ref{cont_est} and \ref{der_cont_est},
for all $t \in (0, T)$ we have
\begin{equation} \label{final_bound1}
  \|{\bf u}-{\bf u}_N\| \le   C \left[ N^{-2s-1} e^{-\mu(t) N}
+  \left\|{\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f}\right\| \right].
\end{equation} 
In particular, using (\ref{app_power_vec}) 
and (\ref{disc_proj_prop2}) with $2s$ being an integer, we get
\begin{equation}  \label{final_bound2}
 \left\|{\mathbf \Pi}_N {\bf f} - {\mathbf L}_N {\bf f}\right\| 
\le 
  \left\|{\mathbf \Pi}_N{\bf f} -  {\bf f}\right\| +
  \left\| {\bf f} - {\mathbf L}_N {\bf f}\right\| \le C N^{-2s}.
\end{equation}
Now the result (\ref{int_bound}) follows from  (\ref{final_bound1}) 
and  (\ref{final_bound2}).

Having established the validity of our algorithm for a simple 
known solution, 
we use the same code to simulate unknown velocity and vorticity 
fields generated
by (\ref{NSE_sphere}) with random initial velocity
field as in~\cite{debussche_dubois_temam, fengler_freeden} with  
angular velocity of the rotation 
$\Omega = 1$  and  $\nu^{-1} = 10,000$  in (\ref{NSE_sphere}).

The initial flow and external force in this benchmark example
satisfy the main  assumption (\ref{main_assumption}) for 
{\em any}  $s, \sigma_1>0$ and hence, as proved in Theorem~\ref{conv_anal},
the approximate solution ${\bf u}_N$ is spectrally accurate and converges
super-algebraically with order given by (\ref{int_bound}) for {\em any} 
$s > 0$. As mentioned in Section~\ref{introduction}, this is the main advantage of the present paper over  the recent paper~\cite{fengler_freeden}, 
where such spectrally accurate convergence results are neither discussed 
nor proved. On the other hand, convergence results 
for two-dimensional problems on a Euclidean plane, 
supported by numerical experiments, 
have formed a core part of research on 
the NSE over the last few decades, 
see~\cite{debussche_dubois_temam,Foias_book, temam79} and references therein.

The random initial tangential divergence-free velocity field, 
having properties similar to those considered 
in~\cite[page~305]{debussche_dubois_temam}
and~\cite[page~988]{fengler_freeden}
(but not exactly same as  in~\cite{debussche_dubois_temam, fengler_freeden}, 
due to randomness), is a smooth function ${\bf u}_0 = {\bf v} \in G^{s/2}_\sigma$, 
a Gevrey class of order $s$ and  index $\sigma$, see (\ref{Gev_space_char}),
for {\em any} $s, \sigma > 0$, with Fourier 
coefficients $\widehat{{\bf v}}_{L,m},~~ L = 1, 2, \cdots,  |m| \leq L$,  
defined by
\begin{equation}
\widehat{{\bf v}}_{L,m} = 
 \begin{cases} 
a_L  \exp(i \phi_m)   &   L=1,\ldots,20;~
  m=0,\ldots,L, \cr
a_L (-1)^m   \exp(-i \phi_m)   &   L=1,\ldots,20;~
  m=-L, \cdots, -1, \cr
0, & L > 20;~ m = -L, \ldots, L,
\end{cases}
\end{equation}
where  $\phi_0 = 0$, and  $\phi_m \in (0,2\pi)$  are random numbers 
for $m > 0$, 
and $a_L = b_L/||\mathbf{b}||$, with $\mathbf{b} \in \ensuremath{\mathbb R}^{20}$ 
having components  $b_L = 2/\left[L+(\nu L)^{2.5}\right],~L=1,\ldots, 20$.
The vorticity stream function $\Psi$ (see (\ref{eq:vort})) of 
$\boldsymbol{\mathrm{Vort}} \,  {\bf u}(\cdot,0)$ in Figure~\ref{vort_t0} demonstrates the randomness
of the field at time  $t = 0$. 

The external force field ${\bf f} = {\bf f}(\widehat{{\bf x}},t)$ in
(\ref{NSE_sphere}) for our simulation is motivated by that considered
in~\cite[page~305]{debussche_dubois_temam} and is exactly same as that in 
~\cite[page~988]{fengler_freeden}. The source term ${\bf f}$ is a 
decaying tangential divergence-free field which, 
for {\em any} $s, \sigma > 0$, belongs to 
$C([0,T];\mathcal D(\boldsymbol{\mathrm{A}}  ^{s+1/2} e^{\sigma \boldsymbol{\mathrm{A}}  ^{1/2}}))$ (for any $T > 0$) 
with the only  non-zero Fourier coefficient being 
$\widehat{{\bf f}(t)}_{3,0}$. The Fourier
coefficient $\widehat{{\bf f}(t)}_{3,0}$ is a continuous function
in time and  is defined by 
\[
  \widehat{{\bf f}(t)}_{3,0}  = \begin{cases} 
                 1, &  0 \le t \le 10, \cr
                 \cos(\pi t/5)\exp(-(t-10)/5) & t>10. \cr
              \end{cases}
\] 
The impact of the external force on the numerical
velocity and its complement (generating the approximate 
inertial manifold) is demonstrated in Figure~\ref{velocity_force_fig}

the external force, leading to little change in evolution process

as the external force gets smaller and smaller.

                                                                            
                                                                           
                                         
                                                                                         
We chose a fixed relative error 
tolerance to be of accuracy at least $\mathcal{O}(10^{-3})$, 
for adaptive time-integration solving the $N^2+2N$-dimensional system   
ordinary differential equations  (\ref{app_weak_form}) in time,  
using  backward differential formulas with variable order (one to five) 
and variable adaptive time step sizes that meet the fixed error tolerance. 

As discussed in the next subsection, the high-frequency 
components of the solution turn out to be
unreliable as $t$ increases, \cite{debussche_dubois_temam},
presumably because  of the time discretization error. We therefore
retained only the frequency components up to some order $N_1 \le N$,
correspondingly, we define
an additional approximation of ${\bf u}_N$, defined in 
(\ref{app_weak_form})-(\ref{app_soln_rep}), by 
\begin{equation}\label{eq: cut_app_soln} 
{\bf u}_{N_1;N} (\cdot,t) :=   {\mathbf \Pi}_{N_1} {\bf u}_N (\cdot,t), \qquad N_1 \leq N.
\end{equation}
As in Theorem~\ref{conv_anal}, for the spatially discrete  
pseudospectral quadrature method, assuming
\eqref{main_assumption} and exact time integration 
of (\ref{app_weak_form}), and hence using \eqref{bounded_soln},
\eqref{app_power_veca},
and \eqref{int_bound}, we get spectral convergence: 
\[
  \|{\bf u}- {\bf u}_{N_1;N}\| 
\le 
  \left\|{\bf u} - {\mathbf \Pi}_{N_1} {\bf u}\right\| +  
\left\|{\mathbf \Pi}_{N_1}\left({\bf u} -   {\bf u}_N\right)\right\| \le
    C \left[N_1^{-2(s+1)} +  N^{-2s} \right] \leq C N_1^{-2s}.
\]

 \quad \mbox{ for }t = 10, 20, 30, 40, 50, 60,
\]

<!-- proposition-1: body_md -->
~\label{foias_prop_S}
Let $s_1,s_2,s_3 \ge 0$ be real numbers, and we assume that
$s_1+s_2+s_3 \ge 1$ and $(s_1,s_2,s_3)\ne (0,0,1),(0,1,0),(1,0,0)$.
Then there exists a constant depending on $s_1,s_2,s_3$ such that
\[
   |b({\bf u},{\bf v},{\bf w})| \le C \|{\bf u}\|_{H^{s_1}(TS)} \|{\bf v}\|_{H^{s_2+1}(TS)}
                        \| {\bf w}\|_{H^{s_3}(TS)}
\]
or in the extrapolated form, writing $H^{s}(TS) =  H^{s}_{TS}$,   for all ${\bf u},{\bf v},{\bf w} \in C^\infty(TS)$
\begin{eqnarray*}
   |b({\bf u},{\bf v},{\bf w})| 
   \le C & \|{\bf u}\|^{1+[s_1]-s_1}_{H^{[s_1]}_{TS}} 
           \|{\bf u}\|^{s_1-[s_1]}_{H^{[s_1]+1}_{TS}} 
          
\|{\bf v}\|^{1+[s_2]-s_2}_{H^{[s_2]+1}_{TS}} 
      \|{\bf v}\|^{s_2-[s_2]}_{H^{[s_2]+2}_{TS}} 
        
\|{\bf w}\|^{1+[s_3]-s_3}_{H^{[s_3]}_{TS}} 
     \|{\bf w}\|^{s_3-[s_3]}_{H^{[s_3]+1}_{TS}}.
\end{eqnarray*}

<!-- proof-8: body_md -->
Let ${\bf u},{\bf v},{\bf w} \in C^\infty(TS)$ and $\widetilde{{\bf u}},\widetilde{{\bf v}},\widetilde{{\bf w}}$
be their corresponding extension to the spherical layer
$\widetilde{\Omega}:=S\times I$. Let us consider first the case $s_i <1$ for $i=1,2,3$.
Define associated constants  $q_1, q_2, q_2, q_4$ so that  $\sum_{i=1}^4 \frac{1}{q_i} = 1$ and
$\frac{1}{q_i} = \frac{1}{2} - \frac{s_i}{2}$ for $i=1,2,3$.
Then, by H\"older's inequality
\begin{eqnarray*}
|b({\widetilde{{\bf u}}},{\widetilde{{\bf v}}},{\widetilde{{\bf w}}})| &=&
\left |\sum_{i,j = 1}^{3}\int_{\widetilde\Omega}
    \widetilde{u}_j \frac{\partial \widetilde{v}_i}{\partial x_j} \widetilde{w}_i 1
     \right| 
    \le \|{\widetilde{{\bf u}}}\|_{L^{q_1}(\Omega)} \| \nabla {\widetilde{{\bf v}}} \|_{L^{q_2}(\Omega)}
          \|{\widetilde{{\bf w}}}\|_{L^{q_3}(\Omega)} \|1\|_{L^{q_4}(\Omega)} \\
\end{eqnarray*}
Restricting to the sphere by \eqref{btilde} and \eqref{Lpnorm} then
using the Sobolev embedding theorem on the sphere \eqref{Sobolev_imbedding}
we have
\begin{eqnarray*}
   |b({\bf u},{\bf v},{\bf w})| &\le& C
  \|{\bf u}\|_{L^{q_1}_{TS}} \|\boldsymbol{\mathrm{Grad}} \, {\bf v}\|_{L^{q_2}_{TS}} \|{\bf w}\|_{L^{q_3}_{TS}} 
    \le
  C \|{\bf u}\|_{H^{s_1}_{TS}} \|\boldsymbol{\mathrm{Grad}} \, {\bf v}\|_{H^{s_2}_{TS}} \|{\bf w}\|_{H^{s_3}_{TS}}.
\end{eqnarray*}

<!-- lemma-2: body_md -->
~\label{lem:weaklip}
Let $\delta \in (1/2,1)$ be given and ${\bf u}, {\bf v} \in V$. Then there
exists $C$, independent of ${\bf u}$ and ${\bf v}$, such that 
\[
  \|\boldsymbol{\mathrm{A}}  ^{-\delta} \boldsymbol{\mathrm{B}}  ({\bf u},{\bf v})\| \le C
    \left\{
   \begin{array}{cc}
        \|\boldsymbol{\mathrm{A}}  ^{1-\delta} {\bf u} \| \|{\bf v}\| & \\
        \|{\bf u}\| \|\boldsymbol{\mathrm{A}}  ^{1-\delta} {\bf v}\| & \\
   \end{array}, 
   \right. \qquad {\bf u}, {\bf v} \in V.
\]
<!-- end lemma-2 -->
<!-- proof-9: body_md -->
Let ${\bf u},{\bf v},{\bf w} \in V$.  Hence from \eqref{skew}, and
by using Proposition \ref{foias_prop_S} with $s_1=0$,
$s_2=2\delta-1>0$ and $s_3=2-2\delta>0$, 
\[
   |b({\bf u},{\bf v},\boldsymbol{\mathrm{A}}  ^{-\delta} {\bf w})| = |b({\bf u},\boldsymbol{\mathrm{A}}  ^{-\delta}{\bf w},{\bf v})|
     \le C \|{\bf u}\| \|\boldsymbol{\mathrm{A}}  ^{-\delta}{\bf w}\|_{H^{2\delta}(TS)}
         \|{\bf v}\|_{H^{2-2\delta}(TS)}.
\]
Since
$\| \boldsymbol{\mathrm{A}}  ^{-\delta} {\bf w}\|_{H^{2\delta}(TS)} =
  \| \boldsymbol{\mathrm{A}}  ^{\delta} \boldsymbol{\mathrm{A}}  ^{-\delta} {\bf w}\| = \|{\bf w}\|$, we get 
\[
   |b({\bf u},{\bf v},\boldsymbol{\mathrm{A}}  ^{-\delta} {\bf w})| \le
   C \|{\bf u}\| \|{\bf w}\| \|{\bf v}\|_{H^{2-2\delta}(TS)}, \qquad  {\bf w} \in V.
\]
Since the inequality is true for all ${\bf w} \in V$, we obtain 
the first bound  
\[
 \|\boldsymbol{\mathrm{A}}  ^{-\delta} \boldsymbol{\mathrm{B}}  ({\bf u},{\bf v}) \|
    \le C \|{\bf u}\| \|{\bf v}\|_{H^{2-2\delta}(TS)} =
       C \|{\bf u}\| \|\boldsymbol{\mathrm{A}}  ^{1-\delta}{\bf v}\|.
\]
To obtain the second bound, 
we again use \eqref{skew} and Proposition \ref{foias_prop_S}
but with  $s_1=2-2\delta$, $s_2 = 2\delta-1$ and $s_3=0$, 
\begin{eqnarray*}
 |b({\bf u},{\bf v},\boldsymbol{\mathrm{A}}  ^{-\delta}{\bf w})| =
   |b({\bf u},\boldsymbol{\mathrm{A}}  ^{-\delta}{\bf w},{\bf v})|
     &\le& C \| {\bf u} \|_{H^{2-2\delta}(TS)}
             \|\boldsymbol{\mathrm{A}}  ^{-\delta} {\bf w}\|_{H^{2\delta}(TS)} \|{\bf v}\| \\
     &=& C\|\boldsymbol{\mathrm{A}}  ^{1-\delta} {\bf u}\| \|{\bf w}\| \|{\bf v}\|,  \qquad  {\bf w} \in V.
\end{eqnarray*}
Since the inequality is true for all ${\bf w} \in H$, we obtain
\[
\|\boldsymbol{\mathrm{A}}  ^{-\delta} \boldsymbol{\mathrm{B}}  ({\bf u},{\bf v}) \| \le C\|\boldsymbol{\mathrm{A}}  ^{1-\delta}{\bf u}\| \|{\bf v}\|.
\]
