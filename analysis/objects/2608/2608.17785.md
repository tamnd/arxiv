\label{thm:main}
Let $q$ be a prime power and let $n,d\geq1$.
\begin{enumerate}[(1)]
\item Every maximum intersecting family in $\mathcal P_{n,d}$ is a star if
and only if one of the following holds:
\[
  q=2\text{ and }n=1;\qquad
  q\text{ is odd and }d\geq2;\qquad
  q\geq4\text{ is even and }d>n.
\]
\item Every maximum 3-wise intersecting family in $\mathcal P_{n,d}$ is
a star, except when $q=2$, $d=1$, and $n\geq2$. In this exceptional
case, every maximum 4-wise intersecting family is a star.
\end{enumerate}

\label{thm:span}
Let $n\geq1$ and $d\geq2$. Then
\begin{enumerate}[(1)]
\item If $q=2$, then
\[
W_{n,d} =\{f\in\mathcal P_{n,d}:f\text{ represents a constant function on }\mathbb{F}_2^n\}.
\]
\item If $q\geq4$ is even and $2\leq d\leq n$, then 
\[
W_{n,d}=\{f\in\mathcal P_{n,d}: [x_I]f = 0 \text{ for all subsets } I \text{ of size $d$}\}.
\]
\item If $q$ is odd, or if $q\geq4$ is even and $d>n$, then 
\[

\label{thm:zero-detecting}

\item Let $q\geq 3$. If $q$ is odd or $d>n$, then $\Psi$ is zero-detecting if and only if there is $\mathbf a\in \mathbb{F}_q^n$ such that

\]
\item If $q\geq 4$ is even and $d\leq n$, then $\Psi$ is zero-detecting

\]
for some $\mathbf a\in \mathbb{F}_q^n$ and
$\mu_I\in \mathbb{F}_q$.

\item If $q=2$, then $\Psi$ is zero-detecting if and only if there is a subset $S\subseteq\mathbb{F}_2^n$ of odd size such that

\]
\end{enumerate}

<!-- lem-2: body_md -->
\label{lem:transversal}
Let $V\leq\mathcal P_{n,d}$ be an $\mathbb{F}_q$-linear subspace containing the
constant polynomials, and let $C$ be an affine translate of $V$. Every
intersecting family $\mathcal A\subseteq C$ satisfies $|\mathcal A|\leq |V|/q$. If equality holds, then $\mathcal A$ contains exactly one polynomial from
each set
\[
\{f+c:c\in\mathbb{F}_q\},\qquad f\in C.
\]
In particular, for every $\mathbf a\in\mathbb{F}_q^n$ and $b\in\mathbb{F}_q$, the
family $\{f\in C:f(\mathbf a)=b\}$
is a maximum intersecting family in $C$. Moreover, let $U\leq V$ be a subspace containing the constant polynomials. If
$|\mathcal A|=|V|/q$, then every coset $D$ of $U$ in $C$ satisfies $|\mathcal A\cap D|=|U|/q$.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Partition $C$ into the $|V|/q$ cosets of the one-dimensional subspace
of constant polynomials. Distinct members of a coset differ by a
nonzero constant, so their graphs are disjoint. This proves the bound
and the first equality statement.

Fix $\mathbf a\in\mathbb{F}_q^n$ and $b\in\mathbb{F}_q$. Each coset modulo the
constants contains a unique polynomial $f$ satisfying
$f(\mathbf a)=b$. The resulting family is intersecting and attains
the bound.

For any coset $D$ of $U$ in $C$, the same argument gives $|\mathcal A\cap D|\leq |U|/q$. There are $|V|/|U|$ such cosets, and these
upper bounds sum to $|V|/q=|\mathcal A|$, so equality holds for every $D$.

<!-- cor-1: body_md -->
\label{cor:standard-normalization}
Let $V\leq\mathcal P_{n,d}$ be an $\mathbb{F}_q$-linear subspace containing the
constant polynomials and the coordinate polynomials $x_1,\ldots,x_n$, and assume
that $V$ is invariant under translations of the variables. Let
$\mathcal F\subseteq V$ be an $r$-wise intersecting family of size $|V|/q$,
where $r\geq2$. After a normalization as in
Lemma~\ref{lem:normalization}, we may assume that
\[
  0,x_1,\ldots,x_n\in\mathcal F.
\]
Moreover, if $V_0=\{f\in V:f(0)=0\}$, then there is a unique function
$\varphi\colon V_0\to\mathbb{F}_q$ such that
\[
\mathcal F=\{f+\varphi(f):f\in V_0\}, \qquad \varphi(0)=\varphi(x_1)=\cdots=\varphi(x_n)=0.
\]
<!-- end cor-1 -->

Lemma~\ref{lem:transversal} gives a unique $a_i\in\mathbb{F}_q$ such that
$x_i-a_i\in\mathcal F$. Translating the variables by
$\mathbf a=(a_1,\ldots,a_n)$ gives
$0,x_1,\ldots,x_n\in\mathcal F$. Translation invariance ensures that the
transformed family remains in $V$.

By the equality case of Lemma~\ref{lem:transversal}, $\mathcal F$ contains exactly one member of each coset modulo the constants. Every such coset has a unique representative $f\in V_0$, so its member of $\mathcal F$ can be written uniquely as $f+\varphi(f)$. The inclusions $0,x_1,\ldots,x_n\in\mathcal F$ give the stated values of $\varphi$.

<!-- lem-6: body_md -->
\label{lem:parity}
Let $q=2^r$, write $\mathbf x=(x_1,\ldots,x_d)$, and let $f\in \mathbb{F}_q[x_1,\ldots,x_d]$ with $\deg f\leq d$. Put $a=[x_1x_2\cdots x_d]f$ and let
\[
  Z(f)=\bigl|\{\mathbf x\in \mathbb{F}_q^d:f(\mathbf x)=0\}\bigr|.
\]
Then $Z(f)\equiv a^{q-1}\pmod2$. In particular, if $a\neq0$, then $f$ has a zero in $\mathbb{F}_q^d$.
<!-- end lem-6 -->
<!-- proof-6: body_md -->
Let $\overline{Z}(f)$ be the reduction of $Z(f)$ mod $2$. We view $\overline{Z}(f)$ as an element of $\mathbb{F}_2\subseteq\mathbb{F}_q$. Observe that
\[
\overline{Z}(f)=\sum_{\mathbf x\in\mathbb F_q^d}\bigl(1-f(\mathbf x)^{q-1}\bigr).
\]
Moreover,
\[
\sum_{t\in\mathbb F_q}t^m=
\begin{cases}
1,& m>0\ \text{and}\ q-1\mid m,\\
0,&\text{otherwise}.
\end{cases}
\]
Consequently, $\sum_{\mathbf{x}\in\mathbb{F}_q^{d}} \mathbf{x}^{\boldsymbol{\alpha}}=0$ if $\alpha_i<q-1$ for some $i$. As $\deg f^{q-1}\le d(q-1)$, the only monomial of $f^{q-1}$ that
survives after summing over $\mathbb F_q^d$ is
$x_1^{q-1}\cdots x_d^{q-1}$. We obtain
\begin{equation}\label{eq:Z(f)}
\overline{Z}(f)=[x_1^{q-1}\cdots x_d^{q-1}]f^{q-1}.
\end{equation}
It remains to compute this coefficient. Write $f=\sum_{\boldsymbol{\beta}} c_{\boldsymbol{\beta}}\mathbf{x}^{\boldsymbol{\beta}}$ so that $c_{\mathbf 1}=a$, where $\mathbf{1}=(1,\ldots,1)$. A term in the multinomial expansion that
contributes to $\mathbf x^{(q-1)\mathbf1}$ is specified by
nonnegative integers $m_{\boldsymbol\beta}$ satisfying
\[
\sum_{\boldsymbol{\beta}} m_{\boldsymbol{\beta}}=q-1,
\qquad
\sum_{\boldsymbol{\beta}} m_{\boldsymbol{\beta}}\boldsymbol{\beta}=(q-1)\mathbf 1.
\]
Taking total degrees shows that every $\boldsymbol{\beta}$ with $m_{\boldsymbol{\beta}}>0$ satisfies $|\boldsymbol{\beta}|=d$.

Suppose the corresponding multinomial coefficient $\binom{q-1}{(m_{\boldsymbol{\beta}})_{\boldsymbol{\beta}}}$ is odd. By Kummer's theorem, the 
addition $\sum_{\boldsymbol{\beta}}m_{\boldsymbol{\beta}}=q-1$ is carry-free in base $2$. Since \(q-1\) is odd,
exactly one \(m_{\boldsymbol{\beta}}\) is odd. Reducing the second equality modulo \(2\)
shows that its exponent vector satisfies
$\boldsymbol{\beta}\equiv\mathbf 1\pmod 2$. 
But \(|\boldsymbol{\beta}|=d\), so its \(d\) coordinates are positive odd integers with
sum \(d\). Hence \(\boldsymbol{\beta}=\mathbf 1\).

Replace $m_{\mathbf 1}$ by $m_{\mathbf 1}-1$, and then divide all the $m_{\boldsymbol{\beta}}$ values by $2$. The same argument then applies with $q-1$ replaced by $(q-2)/2=2^{r-1}-1$. Repeating this for all binary digits of $q-1$ shows that
\[
m_{\mathbf 1}=q-1, \qquad \text{and}
\qquad
m_{\boldsymbol{\beta}}=0 \quad \text{for all} \quad \boldsymbol{\beta}\ne\mathbf 1.
\]
Therefore, every other multinomial coefficient is even, and
\begin{equation}\label{eq:coeff-c1}
[x_1^{q-1}\cdots x_d^{q-1}]f^{q-1}
=c_{\mathbf 1}^{q-1}=a^{q-1}.
\end{equation}
Combining equations~\eqref{eq:Z(f)} and \eqref{eq:coeff-c1} yields $\overline{Z}(f)=a^{q-1}$, 
as required.

<!-- lem-7: body_md -->
\label{lem:prescribed-norm-linear}
Let $q\geq4$ and $m\geq3$, and let
$T\colon\mathbb{F}_{q^m}\to\mathbb{F}_q$ be a nonzero $\mathbb{F}_q$-linear map. For every
$a,\tau\in\mathbb{F}_q^\times$, there exists $\gamma\in\mathbb{F}_{q^m}$ such that
\[
  \operatorname{N}_{\mathbb{F}_{q^m}/\mathbb{F}_q}(\gamma)=a,
  \qquad
  T(\gamma)=\tau.
\]
<!-- end lem-7 -->
<!-- proof-7: body_md -->
By nondegeneracy of the trace pairing, there is
$b\in\mathbb{F}_{q^m}^\times$ such that $T(\gamma)=\operatorname{Tr}_{\mathbb{F}_{q^m}/\mathbb{F}_q}(b\gamma)$. It is therefore enough to find $y\in\mathbb{F}_{q^m}$ satisfying
\[
  \operatorname{N}_{\mathbb{F}_{q^m}/\mathbb{F}_q}(y)
  =a\operatorname{N}_{\mathbb{F}_{q^m}/\mathbb{F}_q}(b), \qquad  \operatorname{Tr}_{\mathbb{F}_{q^m}/\mathbb{F}_q}(y)=\tau,
\]
and then take $\gamma=b^{-1}y$.

For $u,w\in\mathbb{F}_q^\times$, let
\[
  \nu_m(u,w)
  =\bigl|\{y\in\mathbb{F}_{q^m}:
      \operatorname{Tr}_{\mathbb{F}_{q^m}/\mathbb{F}_q}(y)=u,\ 
      \operatorname{N}_{\mathbb{F}_{q^m}/\mathbb{F}_q}(y)=w\}\bigr|.
\]
By the trace--norm estimate of Moisio and Wan
\cite[Theorem~1.2]{MW10},
\[
  \left|
    \nu_m(u,w)-\frac{q^{m-1}-1}{q-1}
  \right|
  \leq (m-1)q^{(m-2)/2}.
\]
We show the main term is larger than the error term. If $m=3$, then
\[
  \frac{q^2-1}{q-1}=q+1>2\sqrt q.
\]
If $m\geq4$, then
\[
  \frac{q^{m-1}-1}{q-1}
  >q^{m-2}
  >(m-1)q^{(m-2)/2},
\]
because $q^{(m-2)/2}\geq2^{m-2}>m-1$. Hence $\nu_m(u,w)>0$, and the required element $y$ exists.

\label{lem:degree-one}
The maximum intersecting families in $\mathcal P_{n,1}$ are precisely the

\[
  \mathcal F_\varphi
  =\{\mathbf a\cdot\mathbf x+\varphi(\mathbf a):
       \mathbf a\in\mathbb{F}_q^n\},

exactly one polynomial for each linear part. This allows us to write 
the family as $\mathcal F_\varphi$ for some function $\varphi\colon\mathbb{F}_q^n\to\mathbb{F}_q$. Conversely, every $\mathcal F_\varphi$ is intersecting: the difference of two distinct members is a nonconstant

The family is the star $\{f \in \mathcal{P}_{n,1}: f(\mathbf s)=b\}$ precisely when $\mathbf a\cdot\mathbf s+\varphi(\mathbf a)=b$ for all $\mathbf a\in\mathbb{F}_q^n$, or equivalently, $\varphi(\mathbf a)=b-\mathbf a\cdot\mathbf s$ for every $\mathbf a$.

<!-- prop-3: body_md -->
\label{prop:threewise-linear}
Let $q\geq3$, and let $\mathcal F\subseteq\mathcal P_{n,d}$ be a maximum
3-wise intersecting family with $0\in\mathcal F$. Then $\mathcal F$ is an
$\mathbb{F}_q$-linear subspace.
<!-- end prop-3 -->
<!-- proof-12: body_md -->
We first show that $\mathcal{F}$ is closed under scalar multiplication. Fix $f\in\mathcal F$ and $\lambda\in\mathbb{F}_q$. For every $g\in\mathcal F$, the polynomials $0,f,g$ agree at some input $\mathbf{a}\in\mathbb{F}_{q}^{n}$, and their common value is zero, so $f(\mathbf{a})=g(\mathbf{a})=0$; then $\lambda f$ and $g$ also agree at $\mathbf a$. Hence $\mathcal F\cup\{\lambda f\}$ is intersecting; since $\mathcal F$ has maximum size, we must have $\lambda f\in\mathcal F$.

We next prove closure under differences. Suppose that $g-h\notin\mathcal F$ for some $g,h\in\mathcal F$. The family $\mathcal F\cup\{g-h\}$ is not intersecting because $\mathcal F$ already has
maximum size. Since $\mathcal F$ itself is intersecting, some $f\in\mathcal F$
does not intersect $g-h$; equivalently, $g-h-f$ is nonvanishing.

Since $q>2$, we may choose $\alpha,\beta\in \mathbb{F}_q^{\times}$ with $\alpha+\beta\neq0$. Let $\gamma=-(\alpha+\beta)$, so that
$\alpha,\beta,\gamma$ are nonzero and $\alpha+\beta+\gamma=0$. By closure under scalar multiplication, the three polynomials
\[
  \alpha^{-1}g,\qquad -\beta^{-1}h,\qquad -\gamma^{-1}f
\]
belong to $\mathcal F$. Being three members of a 3-wise intersecting family, they agree at some
$\mathbf u\in\mathbb{F}_q^n$, say $\alpha^{-1}g(\mathbf u)=-\beta^{-1}h(\mathbf u)
=-\gamma^{-1}f(\mathbf u)=c$. Then
$(g-h-f)(\mathbf u)=(\alpha+\beta+\gamma)c=0$, contradicting the
nonvanishing of $g-h-f$.

<!-- thm-5: body_md -->
\label{thm:threewise-qge3}
Let $q\geq3$ and let $n,d\geq1$. Every maximum 3-wise intersecting
family in $\mathcal P_{n,d}$ is a star.
<!-- end thm-5 -->
<!-- proof-13: body_md -->
Let $\mathcal F$ be a maximum 3-wise intersecting family. After applying
Corollary~\ref{cor:standard-normalization}, we may assume that
$0,x_1,\ldots,x_n\in\mathcal F$. By Proposition~\ref{prop:threewise-linear}, $\mathcal F$ is an
$\mathbb{F}_q$-linear subspace.

We claim that $\mathcal F$ is closed under multiplication by an arbitrary
polynomial $g$, provided the product has degree at most $d$:
\begin{equation}\label{eq:truncated-ideal}
  f\in\mathcal F,\quad \deg(fg)\leq d
  \quad\Longrightarrow\quad fg\in\mathcal F.
\end{equation}
To prove this, let $f\in\mathcal F$ and let $g$ satisfy $\deg(fg)\leq d$.
Lemma~\ref{lem:transversal} gives a unique $c\in\mathbb{F}_q$ such that
$fg+c\in\mathcal F$. The three polynomials $0,f,fg+c$ agree at some
$\mathbf u\in\mathbb{F}_q^n$. Their common value is zero:
$f(\mathbf u)=0$ and $f(\mathbf u)g(\mathbf u)+c=0$. These equalities
force $c=0$, proving \eqref{eq:truncated-ideal}.

Every nonconstant monomial $\mu$ of degree at most $d$ is divisible
by some $x_i$. Since $x_i\in\mathcal F$, \eqref{eq:truncated-ideal} gives
$\mu\in\mathcal F$. Since $\mathcal F$ is linear, it contains the star $\{f\in\mathcal P_{n,d}:f(0)=0\}$,
which has the same size as $\mathcal F$. The two families are equal, and $\mathcal F$ is a star.

<!-- lem-9: body_md -->
\label{lem:F2-boolean-closure}
Let $\mathcal F\subseteq\mathcal P_{n,d}$ be a maximum 3-wise intersecting family
over $\mathbb{F}_2$.
\begin{enumerate}[(1)]
\item If $0,f,g\in\mathcal F$ and $\deg f+\deg g\leq d$, then
\[
  \operatorname{span}_{\mathbb{F}_2}\{f,g,fg\}\subseteq\mathcal F.
\]
\item If $f,f+h\in\mathcal F$ and $\deg g+\deg h\leq d$, then
\[
  f+gh\in\mathcal F.
\]
\end{enumerate}
<!-- end lem-9 -->
<!-- proof-14: body_md -->
By Lemma~\ref{lem:transversal}, exactly one of $u$ and $u+1$ belongs
to $\mathcal F$ for every $u\in\mathcal P_{n,d}$.

For part~(1), suppose first that $fg\notin\mathcal F$. Then $fg+1\in\mathcal F$, so the three polynomials $0,f,fg+1$ agree at some input $\mathbf{a}\in\mathbb{F}_{2}^{n}$. Their common value is zero, but $f(\mathbf a)=0$
gives $(fg+1)(\mathbf a)=1$, a contradiction. This proves that $fg\in\mathcal F$. If $fg+f\notin\mathcal F$, then $fg+f+1\in\mathcal F$. Now, the three polynomials $0,f,fg+f+1$ agree at some input $\mathbf{b}\in\mathbb{F}_{2}^{n}$; we have $f(\mathbf b)=0$ but $(fg+f+1)(\mathbf b)=1$, another contradiction. We have proved that $fg,fg+f\in\mathcal F$; symmetry gives $fg+g\in\mathcal F$.

If $f+g+fg\notin\mathcal F$, then $f,g,f+g+fg+1$ agree at some input $\mathbf{a}\in\mathbb{F}_{2}^{n}$. Writing $c=f(\mathbf a)=g(\mathbf a)$, the third polynomial takes the value $c+c+c^2+1=c+1$ there, a contradiction. It remains to prove that $f+g\in\mathcal F$. Otherwise, $f+g+1\in\mathcal F$, and $f+fg$, $g+fg$, and $f+g+1$ agree at some input $\mathbf{b}\in\mathbb{F}_{2}^{n}$. Equality of the first two values gives $f(\mathbf b)=g(\mathbf b)$. The first polynomial then has value $f(\mathbf b)+f(\mathbf b)^2=0$, whereas the third has value $1$, a contradiction. This proves part~(1).

For part~(2), suppose that $f+gh\notin\mathcal F$. Then $f+gh+1\in\mathcal F$. Then $f,f+h,f+gh+1$ agree at some input $\mathbf{a}\in\mathbb{F}_{2}^{n}$. Subtracting the value of $f(\mathbf{a})$ gives simultaneously $h(\mathbf{a})=0$ and $g(\mathbf{a})h(\mathbf{a})+1=0$, which is impossible.

<!-- thm-6: body_md -->
\label{thm:threewise-F2}
Let $n\geq 1$ and $d\geq2$. Every maximum 3-wise intersecting family
in $\mathcal P_{n,d}$ over $\mathbb{F}_2$ is a star.
<!-- end thm-6 -->
<!-- proof-15: body_md -->
Let $\mathcal F$ be a maximum 3-wise intersecting family. After applying
Corollary~\ref{cor:standard-normalization}, we may assume that
$0,x_1,\ldots,x_n\in\mathcal F$. Repeatedly applying
Lemma~\ref{lem:F2-boolean-closure}\textup{(1)} to the coordinate
polynomials shows that every linear form belongs to $\mathcal F$.

We claim that, for $0\leq r\leq n$, every linear form $w$ and all
polynomials $s_1,\ldots,s_r$ of degree at most $d-1$ satisfy
\begin{equation}\label{eq:F2-sum-products}
  w+\sum_{i=1}^r s_ix_i\in\mathcal F.
\end{equation}
We argue by induction on $r$. The case $r=0$ was just proved. Assume
the claim for $r-1$, and put
\[
  p=w+\sum_{i=1}^{r-1}s_ix_i.
\]
The induction hypothesis, applied once with $w$ and once with
$w+x_r$, gives $p,p+x_r\in\mathcal F$. Applying
Lemma~\ref{lem:F2-boolean-closure}\textup{(2)} with $f=p$, $h=x_r$,
and $g=s_r$ gives $p+s_rx_r\in\mathcal F$, proving
\eqref{eq:F2-sum-products}.

\[

\]
where $\deg s_i\leq d-1$. Equation~\eqref{eq:F2-sum-products} gives $f\in\mathcal F$. The polynomials with zero constant term form a star of the same size as $\mathcal F$, so the two families are equal.

lemma gives exactly one member with each linear part:

\label{lem:nonvanishing-completion}
Let $H$ be a proper norm form of degree $d$. For every $\mathbf b\in\mathbb{F}_q^n$ and every $\delta\in\mathbb{F}_q$ with $\delta\neq-H(\mathbf b)$, there is $R\in\mathcal P_{n,d-1}$ such that $H+R$ is nonvanishing and $R(\mathbf b)=\delta$.

Let $H(\mathbf{x})=\operatorname{N}(\alpha_1x_1+\cdots+\alpha_nx_n)$. Put $U=\operatorname{span}_{\mathbb{F}_q}\{\alpha_1,\ldots,\alpha_n\}$ and $u= b_1\alpha_1+\cdots+b_n\alpha_n$. $\operatorname{N}(t)=\delta+\operatorname{N}(u)$. Define
\[

  =\operatorname{N}\!\left(t-u+\alpha_1x_1+\cdots+\alpha_nx_n\right)
   -H(\mathbf x).
\]
The leading homogeneous parts cancel, so $R\in\mathcal P_{n,d-1}$. Moreover, $t-u\notin U$, so $H+R$ is nonvanishing, and $R(\mathbf b)=\operatorname{N}(t)-\operatorname{N}(u)=\delta$.

<!-- lem-11: body_md -->
\label{lem:norm-coefficient-separation}
Let $s\geq1$, let $\mathbf y=(y_1,\ldots,y_s)$, and let
$\boldsymbol{\beta}=(\beta_1,\ldots,\beta_s)\in \mathbb{F}_{q^d}^s$. As $\boldsymbol{\beta}$ ranges over $\mathbb{F}_{q^d}^s$, the forms $\operatorname{N}(\beta_1y_1+\cdots+\beta_sy_s)$ span $\mathcal H_{s,d}$.
<!-- end lem-11 -->
<!-- proof-19: body_md -->
The coefficients of a norm form $\operatorname{N}(\beta_1y_1+\cdots+\beta_sy_s)$ can be computed explicitly. For a multi-index $\mathbf e=(e_1,\ldots,e_s)$ with
$|\mathbf e|=d$, the coefficient of
$\mathbf y^{\mathbf e}$ is
\begin{equation}\label{eq:coefficient-polynomial}
 c_{\mathbf e}(\boldsymbol{\beta})=\sum_{\substack{I_1\sqcup\cdots\sqcup I_s=\{0,\ldots,d-1\}\\
                   |I_i|=e_i}}
    \prod_{i=1}^s
    \beta_i^{\sum_{j\in I_i}q^j}.
\end{equation}
We view $c_{\mathbf{e}}$ as a polynomial in variables $\beta_1, \beta_2, \ldots, \beta_s$. The base-$q$ expansion of the exponent
$\sum_{j\in I_i}q^j$ recovers the set $I_i$. If $\mathbf e\neq\mathbf f$, then $c_{\mathbf e}$ and $c_{\mathbf f}$ do not share any common monomial in $\beta_1, \ldots, \beta_{s}$. It follows that the polynomials $c_{\mathbf{e}}$ are $\mathbb{F}_q$-linearly independent. Moreover, the degree of $c_{\mathbf{e}}$ in each variable is at most $\sum_{j=0}^{d-1}q^j=\frac{q^d-1}{q-1}<q^d$. If an $\mathbb{F}_q$-linear combination of them vanishes on $\mathbb{F}_{q^d}^s$, then this combination is the zero polynomial by  Lemma~\ref{lem:grid}. Thus, the functions $c_{\mathbf{e}}\colon\mathbb{F}_{q^d}^s\to\mathbb{F}_q$ with $|\mathbf{e}|=d$ are linearly independent. The matrix $\bigl(c_{\mathbf{e}}(\boldsymbol\beta)\bigr)_{\boldsymbol\beta\in \mathbb{F}_{q^d}^s,\,|\mathbf{e}|=d}$ has full column rank, which coincides with $\dim_{\mathbb{F}_q}\mathcal H_{s,d}$. Its row indexed by $\boldsymbol\beta$ is the coefficient vector of $\operatorname{N}(\beta_1y_1+\cdots+\beta_sy_s)$, so these norm forms span $\mathcal H_{s,d}$.

<!-- prop-4: body_md -->
\label{prop:proper-norm-generation}
If $q$ is odd or $d>n$, then the proper norm forms span $\mathcal H_{n,d}$ and are closed under multiplication by elements of $\mathbb{F}_q^\times$.
<!-- end prop-4 -->
<!-- proof-20: body_md -->
If $d>n$, then every norm form is proper, since its parameters span a subspace of $\mathbb{F}_{q^d}$ of dimension at most $n<d=[\mathbb{F}_{q^d}:\mathbb{F}_q]$. The spanning assertion follows from Lemma~\ref{lem:norm-coefficient-separation}.

Assume from now on that $q$ is odd and $d\leq n$. Let $W_{\mathrm{prop}}$ be the span of the proper norm forms. Every degree-$d$ monomial supported on fewer than $d$ variables belongs to $W_{\mathrm{prop}}$. Indeed, let $I\subseteq[n]$ be its support. The norm forms in the variables $(x_i)_{i\in I}$ are proper, since $|I|<d$, and by Lemma~\ref{lem:norm-coefficient-separation} they span all degree-$d$ forms in those variables.

It remains to demonstrate that each squarefree degree-$d$ monomial is in $W_{\mathrm{prop}}$. After
relabeling, it suffices to show $x_1\cdots x_d \in W_{\mathrm{prop}}$. For
$t,\alpha_3,\ldots,\alpha_d\in \mathbb{F}_{q^d}$, the norm form
\[
H=\operatorname{N}(tx_1+tx_2+\alpha_3x_3+\cdots+\alpha_dx_d)
\]
is proper, because its parameters span a space of dimension at most $d-1$. View $t,\alpha_3,\ldots,\alpha_d$ as variables. Note that $\Gamma(t,\alpha_3,\ldots,\alpha_d):=[x_1\cdots x_d] H$ is a polynomial whose degree in each variable is smaller than $q^d$, and it contains the term $2t^{1+q}\prod_{i=3}^d\alpha_i^{q^{i-1}}$. Since $q$ is odd, this coefficient does not vanish. Thus, $\Gamma$ is a nonzero polynomial and Lemma~\ref{lem:grid} gives a choice of parameters for which $\Gamma\neq0$. Every monomial in $H$ other than $x_1\cdots x_d$ is not squarefree and belongs to $W_{\mathrm{prop}}$ by the previous paragraph. Hence $x_1\cdots x_d\in W_{\mathrm{prop}}$. 

Finally, if $\lambda\in\mathbb{F}_q^\times$ and
$H=\operatorname{N}(\alpha_1x_1+\cdots+\alpha_nx_n)$ is proper, choose
$u\in \mathbb{F}_{q^d}^\times$ with $\operatorname{N}(u)=\lambda$. Then $\lambda H=\operatorname{N}(u\alpha_1x_1+\cdots+u\alpha_nx_n)$ and the norm form on the right is again proper.

<!-- lem-12: body_md -->
\label{lem:affine-basis-norm-span}
Let $q\geq 3$. As $(\alpha_0,\ldots,\alpha_n)$ ranges over the ordered
$\mathbb{F}_q$-bases of $\mathbb{F}_{q^{n+1}}$, the nonvanishing polynomials $\operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}(\alpha_0+\alpha_1x_1+\cdots+\alpha_nx_n)$ span $\mathcal P_{n,n+1}$ and are closed under multiplication by elements of
$\mathbb{F}_q^\times$.
<!-- end lem-12 -->
<!-- proof-21: body_md -->
Each polynomial $\operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}(\alpha_0+\alpha_1x_1+\cdots+\alpha_nx_n)$ is nonvanishing, because an $\mathbb{F}_{q}$-zero of such a polynomial implies that $\alpha_0, \ldots, \alpha_n$ are $\mathbb{F}_q$-linearly dependent. To prove spanning, suppose that an $\mathbb{F}_q$-linear functional $L$ on $\mathcal P_{n,n+1}$ annihilates all these polynomials as $(\alpha_0,\ldots,\alpha_n)$ ranges over ordered
$\mathbb{F}_q$-bases of $\mathbb{F}_{q^{n+1}}$. Set
\[
  P(\alpha_0,\ldots,\alpha_n)
  =L\left(\operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}
     (\alpha_0+\alpha_1x_1+\cdots+\alpha_nx_n)\right).
\]
Write
\[
D=1+q+\cdots+q^n=\frac{q^{n+1}-1}{q-1}.
\]
View $P$ as a polynomial over $\mathbb{F}_{q^{n+1}}$; it has degree at most $D$ by the same reasoning as in \eqref{eq:coefficient-polynomial}. By hypothesis on $L$, the polynomial $P$ vanishes at every ordered $\mathbb{F}_q$-basis of
$\mathbb{F}_{q^{n+1}}$, and the number of such bases is
\[
\begin{aligned}
  \prod_{i=0}^{n}(q^{n+1}-q^i)
  &=q^{(n+1)^2}\prod_{j=1}^{n+1}(1-q^{-j})\geq
    q^{(n+1)^2}\left(1-\sum_{j=1}^{n+1}q^{-j}\right)\\
  &=q^{(n+1)^2}\frac{q-2+q^{-(n+1)}}{q-1}>q^{(n+1)^2}\frac{1-q^{-(n+1)}}{q-1}
   =Dq^{n(n+1)}.
\end{aligned}
\]
Here we use Bernoulli's inequality, and the strict inequality uses the assumption $q\geq3$. Thus the Schwartz--Zippel lemma \cite{S80} implies that $P$ is the zero polynomial.

It follows that $L$ annihilates every affine norm polynomial
\begin{equation}\label{eq:affine-norm-form}
\operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}
(\alpha_0+\alpha_1x_1+\cdots+\alpha_nx_n)
\end{equation}
for every choice of $\alpha_0,\ldots,\alpha_n\in\mathbb{F}_{q^{n+1}}$ (not just for ordered $\mathbb{F}_q$-bases of $\mathbb{F}_{q^{n+1}}$).

Homogenization gives a linear isomorphism
\[
  \mathcal P_{n,n+1}\to\mathcal H_{n+1,n+1},
  \qquad
  f(\mathbf x)\mapsto
  y_0^{n+1}f\!\left(\frac{y_1}{y_0},\ldots,
                     \frac{y_n}{y_0}\right).
\]
Under this isomorphism, the norm polynomial in \eqref{eq:affine-norm-form} corresponds to
\[
  \operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}  (\alpha_0y_0+\alpha_1y_1+\cdots+\alpha_ny_n).
\]
These homogeneous norm forms span $\mathcal H_{n+1,n+1}$ by
Lemma~\ref{lem:norm-coefficient-separation}. As their
dehomogenizations span $\mathcal P_{n,n+1}$, we obtain $L=0$. As $L$ is arbitrary, the claim about spanning is complete. 

Finally, if $\lambda\in\mathbb{F}_q^\times$, choose
$t\in\mathbb{F}_{q^{n+1}}^\times$ with
$\operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}(t)=\lambda$. Scaling every $\alpha_i$ by
$t$ preserves the basis condition and multiplies the norm polynomial
by $\lambda$.

(1) Suppose $q=2$. Every nonvanishing polynomial takes the value $1$ at every point of $\mathbb{F}_2^n$, so $W_{n,d}$ consists only of polynomials representing constant functions. Conversely, let $g\in\mathcal P_{n,d}$ represent a constant function on $\mathbb{F}_2^n$. If $g$ takes the value $1$, then $g\in W_{n,d}$ by definition. If $g$ takes the value $0$, then both $1$ and $1+g$ are nonvanishing, and therefore $g=(1+g)+1$ belongs to $W_{n,d}$. 

(2) Suppose that $q\geq 4$ is even and $2\leq d\leq n$. Put
\[
K=\{f\in\mathcal P_{n,d}: [x_I]f = 0 \text{ for all subsets } I \text{ of size $d$}\}.
\]
If $f$ is nonvanishing and $[x_I]f\neq0$ for some $I$ of size $d$, then setting $x_j=0$ for all $j\notin I$ leaves a polynomial of degree at most $d$ in the $d$ variables indexed by $I$ with nonzero $x_I$-coefficient, and Lemma~\ref{lem:parity} produces a zero of it, hence a zero of $f$. So every nonvanishing polynomial lies in the subspace $K$, whence $W_{n,d}\subseteq K$. Conversely, let $f\in K$, so that every monomial occurring in $f$ involves at most $d-1$ variables. For each $S\in\binom{[n]}{d-1}$, the polynomials of degree at most $d$ in the variables indexed by $S$ form a copy of $\mathcal P_{d-1,d}$, which is spanned by nonvanishing polynomials by Lemma~\ref{lem:affine-basis-norm-span}; these remain nonvanishing when regarded as polynomials in all $n$ variables. Together with the identity $2x_ix_j = (x_i+x_j)^2-x_i^2-x_j^2$, we get $x_ix_j\in W_{n,2}$ (here we crucially used $2\neq 0$). Considering $(x_i+1)^2-\eta$, we can similarly show that $(x_i+1)^2\in W_{n,2}$, and so $x_i\in W_{n,2}$. We deduce that $W_{n,2}=\mathcal P_{n,2}$.

$d>n$, start instead at $d=n+1$, where the assertion is Lemma~\ref{lem:affine-basis-norm-span}.
Thus, in either case, at the inductive step we may assume that
$\mathcal P_{n,d-1}\subseteq W_{n,d-1}\subseteq W_{n,d}$.
Let $H$ be a proper norm form of degree $d$. By Lemma~\ref{lem:nonvanishing-completion}, there is
$R\in\mathcal P_{n,d-1}\subseteq W_{n,d}$ such that $H+R$ is nonvanishing. Thus, $H+R\in W_{n,d}$, and since $R\in W_{n,d}$, we also have $H\in W_{n,d}$. Together with $\mathcal P_{n,d-1}\subseteq W_{n,d}$, this gives
\[
\mathcal P_{n,d}
 =\mathcal P_{n,d-1}+\mathcal H_{n,d}
 \subseteq W_{n,d},
\]
and we conclude that $W_{n,d}=\mathcal P_{n,d}$.

<!-- lem-13: body_md -->
\label{lem:star-slice-synchronization}
Let $U\leq V$ be $\mathbb{F}_q$-linear subspaces of $\mathcal P_{n,d}$. Suppose $U$ contains the constant polynomials, and $U$ can be identified with $\mathcal P_{m,D}$ after choosing $m$ of the $n$ variables (where $m\geq 1$ and $D\geq 1$). Suppose that every maximum intersecting family in $U$ is a star. Assume that $V$ is the $\mathbb{F}_q$-span of $U$ and $\mathcal{Z}$ for some collection $\mathcal Z\subseteq V$ of nonvanishing polynomials. Then for every maximum intersecting family $\mathcal A\subseteq V$, all the intersections of $\mathcal A$ with the $U$-cosets in $V$ are stars with the same in-center.

Moreover, suppose that this common in-center is the origin. Put
$V_0=\{f\in V:f(0)=0\}$ and write
$\mathcal A=\{f+\varphi(f):f\in V_0\}$ for some function $\varphi\colon V_0\to\mathbb{F}_q$.
Then
\[
\varphi(f+u)=\varphi(f),
\qquad \forall f\in V_0,\ u\in U\cap V_0.
\]
<!-- end lem-13 -->
<!-- proof-23: body_md -->
By Lemma~\ref{lem:transversal}, $\mathcal A$ meets every $U$-coset in a maximum intersecting family. This family must be a star, because subtracting a coset representative allows us to move the family into $U$ where we can apply the hypothesis. A star in a coset of $U$ determines its in-center uniquely as a point of $\mathbb{F}_q^m$; we can view this point in $\mathbb{F}_q^n$ by setting the remaining coordinates equal to $0$.

For a coset $C$ of $U$, let $\boldsymbol\sigma(C)$ denote the in-center of
$\mathcal{A}\cap C$. Fix $H\in\mathcal Z$ and write $C=g+U$ where $g\in V$.
Suppose, to the contrary, that $\boldsymbol\sigma(C)\neq \boldsymbol\sigma(C+H)$.
Write 
\begin{align*}
\mathcal A\cap C & =g+\{u\in U:u(\boldsymbol\sigma(C))=c_1\},  \\
\mathcal A\cap(C+H)&=g+H+\{u\in U:u(\boldsymbol\sigma(C+H))=c_2\}
\end{align*}
with $c_1,c_2\in\mathbb{F}_q$. As the two in-centers differ,
Lemma~\ref{lem:difference-of-stars}, applied inside $U\cong\mathcal P_{m,D}$, shows
that the difference set of the stars $\{u\in U:u(\boldsymbol\sigma(C))=c_1\}$ and $\{u\in U:u(\boldsymbol\sigma(C+H))=c_2\}$ is all of $U$. In particular, the element $0\in U$ is in the difference set; we deduce that the two stars share an element $u$. Then $P\colonequals g+u$, $Q \colonequals g+H+u$ satisfy $P\in\mathcal A\cap C$, $Q\in\mathcal A\cap(C+H)$, and $Q-P=H$.
As $H$ is nonvanishing, this contradicts the intersecting property of $\mathcal A$. Thus, $\boldsymbol\sigma(C+H)=\boldsymbol\sigma(C)$.

Since $\lambda H$ is nonvanishing for every $\lambda\in\mathbb{F}_q^\times$,
the same argument gives
$\boldsymbol\sigma(C+\lambda H)=\boldsymbol\sigma(C)$. As the images of the elements of $\mathcal Z$ span $V/U$, it follows from Lemma~\ref{lem:spanning-invariance} that $\boldsymbol\sigma$ is a constant.

For the final assertion, let $f\in V_0$ and $u\in U\cap V_0$.
Since $U$ contains the constants, $f+\varphi(f)$ and $(f+u)+\varphi(f+u)$ are in the same $U$-coset; they take the same value at the in-center $0$. Using $u(0)=0$, this leads to $\varphi(f+u)=\varphi(f)$.

\label{prop:degree-lifting}
Let $q$ be any prime power, let $n\geq1$ and $d\geq2$, and assume that
either $q$ is odd or $d>n$. If every maximum intersecting family in
$\mathcal P_{n,d-1}$ is a star, then every maximum intersecting family in
$\mathcal P_{n,d}$ is a star.

Let $\mathcal F\subseteq\mathcal P_{n,d}$ be a maximum intersecting family, and let
$\mathcal Z\subseteq\mathcal P_{n,d}$ be the following subset:
\[
  \mathcal Z
  =\{H+R:H \text{ is a proper norm form in } \mathcal H_{n,d},\ R\in\mathcal P_{n,d-1},
       \ H+R\text{ is nonvanishing}\}.
\]
By combining Lemma~\ref{lem:nonvanishing-completion} and Proposition~\ref{prop:proper-norm-generation}, we have
\[
  \mathcal P_{n,d}
  =\mathcal P_{n,d-1}+\operatorname{span}_{\mathbb{F}_q}\mathcal Z.
\]
For $A\in\mathcal H_{n,d}$, write $\mathcal F_A=\{P\in\mathcal P_{n,d-1}:A+P\in\mathcal F\}$. Lemma~\ref{lem:star-slice-synchronization}, applied with $U=\mathcal P_{n,d-1}$, shows that there exists a common in-center $\mathbf s\in\mathbb{F}_q^n$ and a value $b_A\in\mathbb{F}_q$ for each $A$ satisfying
\[
\mathcal F_A=\{P\in\mathcal P_{n,d-1}:P(\mathbf s)=b_A\}.
\]

Fix $A\in\mathcal H_{n,d}$ and a proper norm form $H\in\mathcal H_{n,d}$.
The intersecting property says that $H+Q-P$ has a zero whenever
$P\in\mathcal F_A$ and $Q\in\mathcal F_{A+H}$. By Lemma~\ref{lem:difference-of-stars},
the differences $Q-P$ run through all $R\in\mathcal P_{n,d-1}$ with $R(\mathbf s)=b_{A+H}-b_A$. If $b_{A+H}-b_A\neq-H(\mathbf s)$, then
Lemma~\ref{lem:nonvanishing-completion}, applied at $\mathbf s$ with
$\delta=b_{A+H}-b_A$, would produce such an $R$ for which $H+R$ is
nonvanishing, a contradiction. Hence $b_{A+H}-b_A=-H(\mathbf s)$. 

Define $\xi(A)=b_A+A(\mathbf s)$ for each $A\in\mathcal H_{n,d}$. Then $\xi(A+H)=\xi(A)$ for every proper norm form $H$. By Proposition~\ref{prop:proper-norm-generation}, the proper norm forms
span $\mathcal H_{n,d}$ and are closed under multiplication by nonzero
scalars. Lemma~\ref{lem:spanning-invariance} applied to $\xi(A+H)=\xi(A)$ implies that $\xi$ is constant. Every member of $\mathcal F$ can be expressed as $A+P$ for some $P\in\mathcal F_{A}$; we compute
\[
  (A+P)(\mathbf s)=A(\mathbf s)+b_A=\xi(A)=\xi(0).
\]
This shows $\mathcal F\subseteq \{f\in\mathcal P_{n,d}:f(\mathbf s)=\xi(0)\}$. By comparing cardinalities, we conclude that $\mathcal F=\{f\in\mathcal P_{n,d}:f(\mathbf s)=\xi(0)\}$ is a star.

Proposition~\ref{prop:variable-lifting} applies. If $n\geq 3$, then

Corollary~\ref{cor:standard-normalization}, we may write
\[
\mathcal F=\{f+\varphi(f):f\in V_0\},
\qquad
V_0=\{f\in\mathcal P_{2,2}:f(0)=0\},
\]

f+\varphi(f)

\end{aligned}

$\Phi(0)=0$, we have $\Phi=0$. It follows that $\mathcal F=\{f\in\mathcal P_{2,2}:f(0)=0\}$ is a star.

<!-- prop-8: body_md -->
\label{prop:critical-even}
Let $q\geq4$ be even and let $n\geq1$. Every maximum intersecting
family in $\mathcal P_{n,n+1}$ is a star.
<!-- end prop-8 -->
<!-- proof-28: body_md -->
We proceed by induction on $n$. When $n=1$, this is
Theorem~\ref{thm:univariate}. Suppose $n\geq2$. By induction, every
maximum intersecting family in $\mathcal P_{n-1,n}$ is a star, and
Proposition~\ref{prop:degree-lifting} gives the same conclusion in
$\mathcal P_{n-1,n+1}$.

Let $\mathcal F\subseteq\mathcal P_{n,n+1}$ be a maximum intersecting family. By
Corollary~\ref{cor:standard-normalization}, we may assume that $ 0,x_1,\ldots,x_n\in\mathcal F$. 
Put
\[
  V_0=\{f\in\mathcal P_{n,n+1}:f(0)=0\}.
\]
The same corollary gives a function $\varphi\colon V_0\to\mathbb{F}_q$ such that
\begin{equation}\label{eq:critical-selector}
  \mathcal F=\{f+\varphi(f):f\in V_0\},
  \qquad \varphi(0)=\varphi(x_1)=\cdots=\varphi(x_n)=0.
\end{equation}

\smallskip
\noindent\emph{Step 1: Reduction to monomials involving all variables.} Since $\mathcal P_{n,n+1}$ is spanned by nonvanishing polynomials by
Lemma~\ref{lem:affine-basis-norm-span},
Proposition~\ref{prop:variable-lifting} shows that $\varphi(f)$ depends only
on the coefficients of the monomials involving all $n$ variables. The only such monomials of degree at most $n+1$ are
\[
  m_0=x_1\cdots x_n,
  \qquad
  m_i=x_i^2\prod_{j\neq i}x_j\quad(1\leq i\leq n).
\]
Define the surjective linear map
\[
\rho:\mathcal P_{n,n+1}\to\mathbb{F}_q^{n+1},
  \qquad
  f \mapsto ([m_0]f,[m_1]f,\ldots,[m_n]f).
\]
Its restriction to $V_0$ is also surjective.
Thus, there is a function $\Phi\colon\mathbb{F}_q^{n+1}\to\mathbb{F}_q$, with
$\Phi(0)=0$, such that
\begin{equation}\label{eq:critical-residual-function}
  \varphi(f)=\Phi(\rho(f)), \qquad \forall f\in V_0.
\end{equation}

For a nonvanishing polynomial $H\in\mathcal P_{n,n+1}$, we claim that
\begin{equation}\label{eq:critical-forbidden-value}
  \Phi(\mathbf z+\rho(H))-\Phi(\mathbf z)\neq H(0)
\end{equation}
for all $\mathbf z\in\mathbb{F}_q^{n+1}$.  Indeed, suppose to the contrary that $\Phi(\mathbf z+\rho(H))-\Phi(\mathbf z)=H(0)$. Define $h=H-H(0)\in V_0$ and choose $f\in V_0$ with $\rho(f)=\mathbf{z}$. By
\eqref{eq:critical-residual-function}, the two members $f+\varphi(f)=f+\Phi(\mathbf{z})$ and $f+h+\varphi(f+h)=f+H-H(0)+\Phi(\mathbf{z}+\rho(H))$ of $\mathcal F$ would differ by
\[
  H-H(0)+\Phi(\mathbf z+\rho(H))-\Phi(\mathbf z)=H,
\]
contradicting the nonvanishing assumption on $H$.

\smallskip

\noindent\emph{Step 2: Using norm forms to control $\Phi$.}
Choose linearly independent
$\theta_1,\ldots,\theta_n\in\mathbb{F}_{q^{n+1}}$, write
$\boldsymbol\theta=(\theta_1,\ldots,\theta_n)$, and for each $\gamma\in\mathbb{F}_{q^{n+1}}$ set
\[
  Q_{\gamma,\boldsymbol\theta}(\mathbf x)
  =
  \operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}
  (\gamma+\theta_1x_1+\cdots+\theta_nx_n).
\]
For $1\leq i\leq n$, the coefficient $B_i(\boldsymbol\theta) =[m_i]Q_{\gamma,\boldsymbol\theta}$ is independent of $\gamma$; write
\[\mathbf B(\boldsymbol\theta)
=(B_1(\boldsymbol\theta),\ldots,B_n(\boldsymbol\theta)).
\]
The coefficient of $m_0$ is
\begin{equation}\label{eq:critical-Moore}
  \Delta_{\boldsymbol\theta}(\gamma)
  =
  \det
  \begin{pmatrix}
   \gamma&\theta_1&\cdots&\theta_n\\
   \gamma^q&\theta_1^q&\cdots&\theta_n^q\\
   \vdots&\vdots&&\vdots\\
   \gamma^{q^n}&\theta_1^{q^n}&\cdots&\theta_n^{q^n}
  \end{pmatrix}.
\end{equation}
Indeed, the coefficient is the corresponding permanent, which equals
the determinant in characteristic $2$. Hence $\Delta_{\boldsymbol\theta}:\mathbb{F}_{q^{n+1}}\to\mathbb{F}_q$ is a nonzero
$\mathbb{F}_q$-linear map, and the Moore determinant criterion \cite{Moo96} gives
\begin{equation}\label{eq:critical-Moore-kernel}
  \ker\Delta_{\boldsymbol\theta}
  =\operatorname{span}_{\mathbb{F}_q}\{\theta_1,\ldots,\theta_n\}.
\end{equation}

Fix $\mathbf z\in\mathbb{F}_q^{n+1}$ and  $t\in\mathbb{F}_q^\times$. Let $a\in\mathbb{F}_q^\times$ be arbitrary.  Lemma~\ref{lem:prescribed-norm-linear} gives $\gamma\in\mathbb{F}_{q^{n+1}}$ such that
\[
  \operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}(\gamma)=a,
  \qquad
  \Delta_{\boldsymbol\theta}(\gamma)=t.
\]
Since $t\neq 0$, equation \eqref{eq:critical-Moore-kernel} gives $\gamma\notin\operatorname{span}_{\mathbb{F}_q}\{\theta_1,\ldots,\theta_n\}$, so
$Q_{\gamma,\boldsymbol\theta}$ is nonvanishing. Moreover,
\[
Q_{\gamma,\boldsymbol\theta}(0)=a, \qquad \rho(Q_{\gamma,\boldsymbol\theta})=(t,\mathbf B(\boldsymbol\theta)).
\]
Let $H=Q_{\gamma,\boldsymbol{\theta}}$. Observe that $\Phi\bigl(\mathbf z+(t,\mathbf B(\boldsymbol\theta))\bigr)-\Phi(\mathbf z)\neq H(0)=a$ by \eqref{eq:critical-forbidden-value}. Since $a\in\mathbb{F}_{q}^{\times}$ was arbitrary, we must have
\begin{equation}\label{eq:critical-Phi-invariance}\Phi\bigl(\mathbf z+(t,\mathbf B(\boldsymbol\theta))\bigr)=\Phi(\mathbf z).
\end{equation}

\smallskip
\noindent \emph{Step 3: Showing that $\Phi$ is constant.} Let $\mathcal D$ be the set of all $(t,\mathbf B(\boldsymbol\theta))$ with $t\in\mathbb{F}_q^\times$ and
$\theta_1,\ldots,\theta_n$ linearly independent. Equation~\eqref{eq:critical-Phi-invariance} says that $\Phi$ is invariant under translation by every element of $\mathcal D$. Note that the set $\mathcal D$ is closed under multiplication by nonzero scalars: if $\lambda\in\mathbb{F}_q^\times$, choose $\xi\in\mathbb{F}_{q^{n+1}}^\times$ with $\operatorname{N}_{\mathbb{F}_{q^{n+1}}/\mathbb{F}_q}(\xi)=\lambda$ and replace each $\theta_i$ by $\xi\theta_i$.

Finally, we claim that $\mathcal D$ spans $\mathbb{F}_q^{n+1}$. Fix $\boldsymbol\theta$ and $c\in\mathbb{F}_q^\times$. Choose
$t_1\in\mathbb{F}_q\setminus\{0,c\}$ and put $t_2=c+t_1$. Since $q\geq4$,
both $t_1$ and $t_2$ are nonzero, and we have
\[
(t_1,\mathbf B(\boldsymbol\theta))+(t_2,\mathbf B(\boldsymbol\theta))=(c,0).
\]
Thus, $\mathbb{F}_q\times\{0\}$ lies in the span of $\mathcal D$, and hence so does $(0,\mathbf B(\boldsymbol\theta))$. Projecting the spanning family in Lemma~\ref{lem:affine-basis-norm-span}
onto the coefficients of $m_1,\ldots,m_n$ shows that the vectors $\mathbf B(\boldsymbol\theta)$ span $\mathbb{F}_q^n$. Therefore, $\mathcal D$ spans $\mathbb{F}_q^{n+1}$. Lemma~\ref{lem:spanning-invariance} now implies that $\Phi$ is constant. Since $\Phi(0)=0$, we have $\Phi=0$. We conclude that $\mathcal F = \{f\in\mathcal P_{n,n+1}:f(0)=0\}$ is a star.

  =|\mathcal M_{n,d}|\frac{|\mathcal K_{n,d}|}{q}

\]

\emph{Part 1.} Assume that $q\geq 3$, and put $\mathcal A=\ker\Psi$. Since $\Psi(1)=1$, the functional $\Psi$ is nonzero, so $|\mathcal A|=\frac{|\mathcal P_{n,d}|}{q}$. The functionals $\Psi$ and $\operatorname{ev}_{\mathbf a}$ have the
same kernel. It follows that they are proportional, and their common value $1$ at the constant polynomial $1$ shows that $ \Psi=\operatorname{ev}_{\mathbf a}$.
