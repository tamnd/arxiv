<!-- defn-1: body_md -->
\label{defn:balanced pair}
$\{A,B\}\subset \mathbf{A}$ is called a {\bf balanced pair}, if there exists $T\in \mathrm{GL}(2,\mathbb{R})$ such that for $\mathbb{A}=\{T^{-1}AT, T^{-1}BT\}$ (called a {\bf normal form} of $\{A,B\}$), one of the following mutually exclusive conditions holds: 
\begin{itemize}
  \item [(h)] $\mathbb{A}=\{H_{s_1,u_1}^{\lambda_1}, H_{s_2,u_2}^{\lambda_2}\}$ with $u_2<u_1<0<s_1<s_2$;
  \item [(m)] $\mathbb{A}=\{H_{s,u}^\lambda, P_x\}$;
  \item [(p)] $\mathbb{A}=\{P_x,P_y^t\}$.
\end{itemize}
In the case (h), $\{A,B\}$ is called a {\bf co-parallel pair}. For $\{A,B\}\subset \mathbf{G}$, if $\{aA,bB\}\subset\mathbf{A}$ is a balanced (resp. co-parallel) pair for some $a,b\in\mathbb{R}$, then $\{A,B\}$ is also called a {\bf balanced} (resp. {\bf co-parallel}) {\bf pair}. A balanced pair $\{A,B\}\subset \mathbf{G}$ will also be denoted as an ordered pair $(A,B)\in \mathbf{G}^2$.
<!-- end defn-1 -->
<!-- rmk-1: body_md -->
It is easy to see that our definition of co-parallel pairs coincides with that given at the beginning of \cite{JS90}, or that in \cite[Definition~3.3]{Las25}. Taking the finiteness counter-examples given by \cite{BTV03,HMST11} into consideration, we generalize this definition to balanced pairs, which is close to the concept ``well oriented" in \cite[Definition~2.1]{PS21}.
<!-- end rmk-1 -->
<!-- thm-1: body_md -->
\label{thm:main}
  Let $\mathbb{A}\in \mathbf{G}^2$ be a balanced pair. Then $\mathbb{A}$ admits a unique Lyapunov maximizing measure; moreover, this measure is Sturmian.
<!-- end thm-1 -->
<!-- lem-1: body_md -->
\label{lem:recurrent and balanced}
  For $s\in\{0,1\}^\mathbb{N}$, $s$ is recurrent and balanced iff there exists $\alpha\in[0,1]$ such that $s\in S_\alpha$.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
See, for example, \cite[Theorem~2.1]{MS13}.
<!-- end proof-1 -->
<!-- lem-2: body_md -->
\label{lem:Sturmian slope}
Given $\alpha\in [0,1]$, the following hold.
\begin{itemize}
  \item [(i)] The sub-shift $S_\alpha$ is uniquely ergodic.
  \item [(ii)] $|\sum_{m\le k<m+n} i_k-n\alpha|<1$ holds for any $i_1i_2\cdots\in S_\alpha$ and any $m,n\ge 1$.
\end{itemize}
<!-- end lem-2 -->
<!-- proof-2: body_md -->
For (i), see, for example, \cite[Theorem~2.1]{MS13}. For (ii), see, for example, \cite[Corollary~1.4]{BM02}.
<!-- end proof-2 -->
<!-- lem-3: body_md -->
\label{lem:not balanced}
  Let $x$ be a word which is not balanced. Then there exists a word $w$ such that both $0w0$ and $1w1$ are factors of $x$.
<!-- end lem-3 -->
<!-- proof-3: body_md -->
See, for example, \cite[Proposition~2.1.3]{Lot02}.

<!-- lem-4: body_md -->
\label{lem:cone}
  Let $\mathbb{A}=(A_0,A_1)\in\mathbf{A}^2$ be one of the norm forms given in Definition~\ref{defn:balanced pair}. Define
\begin{equation}\label{eq:cone}
I_{\mathbb{A}}^+  =\left\{\begin{array}{ccl}
  (s_1,s_2) &,&  \text{case (h)} \\
  (0,s) &,&  \text{case (m)} \\
  (0,+\infty) &,&  \text{case (p)} 
\end{array}\right.,\quad
I_{\mathbb{A}}^- =\left\{\begin{array}{ccl}
  (u_2,u_1) &,&  \text{case (h)} \\
  (u,0) &,&  \text{case (m)} \\
  (-\infty,0) &,&  \text{case (p)} 
\end{array}\right..
\end{equation}
Given a non-empty finite word $x$, denote $X=[x]_{\mathbb{A}}$. Then the following hold.
\begin{itemize}
  \item [(i)] $X\in\mathbf{A}$, $\underline{X}(I_{\mathbb{A}}^+)\subset I_{\mathbb{A}}^+$ and $\underline{X}^{-1}(I_{\mathbb{A}}^-)\subset I_{\mathbb{A}}^-$.
  \item [(ii)] $X\in\mathbf{P}$ iff there exists $i\in\{0,1\}$ such that $x=i^{|x|}$ and $A_i\in\mathbf{P}$. 
  \item [(iii)] If $X\in\mathbf{H}$, then $s_X\in \overline{I_{\mathbb{A}}^+}$ and  $u_X\in \overline{I_{\mathbb{A}}^-}$.
  \item [(iv)] If $x\ne 0^{|x|},1^{|x|}$, then $\underline{X}(\overline{I_{\mathbb{A}}^+})\subset I_{\mathbb{A}}^+$ and $\underline{X}^{-1}(\overline{I_{\mathbb{A}}^-})\subset I_{\mathbb{A}}^-$. 
\end{itemize}
<!-- end lem-4 -->
<!-- lem-5: body_md -->
\label{lem:balanced pair}
  Let $\mathbb{A}=(A,B)\in\mathbf{A}^2$ be a balanced pair. 
\begin{itemize}
  \item [(i)] $[w]_\mathbb{A}\in\mathbf{A}$ holds for any non-empty finite word $w$; if $w\ne 0^{|w|},1^{|w|}$ additionally, then $[w]_\mathbb{A}\in\mathbf{H}$. 
  \item [(ii)] For each $X\in \{A,B\}$ and each $Y\in \{AB,BA\}$, $\{X,Y\}$ is a balanced pair. 
  \item [(iii)] $\{AB,BA\}$ is a crossing pair.
  \item [(iv)] $\operatorname{tr} ((AB)^2) >\operatorname{tr} (A^2B^2)$.
                                         
\end{itemize}
<!-- end lem-5 -->
<!-- proof-4: body_md -->
We may assume that $\mathbb{A}$ is in a normal form. Then assertions (i)(ii) follows from Lemma~\ref{lem:cone}. To prove (iii)(iv), denote $C=AB$ and $D=BA$ below. We may further assume that the right end point of $I_\mathbb{A}^+$ is a fixed point of $\underline{B}$, so $\underline{B}(x)>x$ holds for any $x\in I_\mathbb{A}^+\cup I_\mathbb{A}^-$.  

(iii). By Lemma~\ref{lem:cone}, $C,D\in\mathbf{H}$, $s_C,s_D\in I_{\mathbb{A}}^+$ and
\[
\underline{D}(\underline{B}(s_C))=\underline{B}(\underline{C}(s_C))=\underline{B}(s_C)\in I_{\mathbb{A}}^+,
\]
so $s_D=\underline{B}(s_C)>s_C$. A similar argument shows that $u_D=\underline{B}(u_C)>u_C$. Therefore $\{C,D\}$ is a crossing pair.

  \tilde{Y} - Y =\delta (X-\tilde{X}), \quad\text{where}\quad \delta:= \gamma_{m+1}\alpha_p\beta_q  - \gamma_m\alpha_{p-1}\beta_{q-1}>0.

\tilde{Y}=U^p(\gamma_{m+1}\cdot I -\gamma_{m}\cdot\tilde{X}^{-1})V^q = \gamma_{m+1} \cdot U^pV^q - \gamma_m\cdot U^{p-1}V^{q-1}.

Similarly, we have:

Y = \gamma_{m+1}\cdot V^qU^p - \gamma_m\cdot V^{q-1}U^{p-1}.
\]
Also note that by Lemma~\ref{lem:balanced pair}~(i), $X,\tilde{X}\in\mathbf{H}$. Then it follows that

\tilde{Y} - Y = \gamma_{m+1} (U^pV^q- V^qU^p) - \gamma_m(U^{p-1}V^{q-1}-V^{q-1}U^{p-1}) =\delta(X-\tilde{X}),

\eta=\operatorname{tr}[X (\tilde{Y}-Y)] =\delta\cdot \operatorname{tr} (X^2 -X\tilde{X}).

W_0=\operatorname{tr}(XY) \cdot X - Y^{-1}, \quad W_1  = \operatorname{tr}(\tilde{X} Y)\cdot \tilde{X} - Y^{-1}, \quad W_2 = \operatorname{tr}(X\tilde{Y})\cdot X - \tilde{Y}^{-1}.

\operatorname{tr} Y =\operatorname{tr} \tilde{Y}, \quad \operatorname{tr} (X\tilde{Y}) = \operatorname{tr}(Y\tilde{X}) = \operatorname{tr} (\tilde{X} Y).

It follows that

W_1 - W_0 = (\operatorname{tr}(X \tilde{Y}) - \operatorname{tr}(XY) )\cdot \tilde{X} + \operatorname{tr}(XY)\cdot (\tilde{X} -X).

\]
which completes the proof.

\chi(\mathbb{A},\mu_\alpha)= \chi((A,B),\mu_\alpha) + (1-\alpha)\log |a| + \alpha \log |b|, \quad\forall\alpha\in[0,1].
\]
Therefore, we may assume that $\mathbb{A}\in \mathbf{A}^2$.  We may further assume that $\mathbb{A}$ is in a normal form, so that $I_\mathbb{A}^+$ introduced in \eqref{eq:cone} is well-defined. Define $f(\alpha)=\chi(\mathbb{A},\mu_\alpha), \alpha\in[0,1]$. Given $N\ge 3$, let us first show that $f$ is (continuous and) strictly concave on $J_N=(\frac{1}{N},1-\frac{1}{N})$. To this end, we shall follow the argument of Laskawiec in the second paragraph of \cite[Proof~of~Theorem~9.8]{Las25}, which consists of two steps: firstly, prove that $f$ is strictly mid-point concave on $J_N\cap \mathbb{Q}$; secondly, prove that $f$ is continuous on $J_N$. 

For the first step, it suffices to show that, under our settings, for $t_1,t_2\in J_N\cap \mathbb{Q}$ with $t_1<t_2$, the corresponding words $w_1,w_2,w_3$ and associated matrices $\Pi_i=[w_i]_\mathbb{A}, i=1,2,3$ defined in \cite[Proof~of~Theorem~9.8]{Las25} still satisfy 
\[
\rho(\Pi_3)\ge \rho(\Pi_1\Pi_2)>\rho(\Pi_1)\cdot\rho(\Pi_2).
\]
Since $|w_1w_2|=|w_3|,|w_1w_2|_1=|w_3|_1$ and since $w_3^\infty$ is balanced, by Theorem~\ref{thm:J-S}, $\rho(\Pi_3)\ge \rho(\Pi_1\Pi_2)$ holds. On the other hand, by following the proofs of the ``furthermore" part of \cite[Lemma~9.4]{Las25} and \cite[Corollary~9.5]{Las25}, it can be directly checked that $(\Pi_1,\Pi_2)$ is a balanced pair; moreover, by Lemma~\ref{lem:balanced pair}~(i), $\Pi_1,\Pi_2\in\mathbf{H}$. Therefore,  $(\Pi_1,\Pi_2)$ is a co-parallel pair, so by Lemma~\ref{lem:concave}, $\rho(\Pi_1\Pi_2)>\rho(\Pi_1)\cdot\rho(\Pi_2)$.                                                                                                                


For the second step, let 
\begin{equation}\label{eq:sft}
  \Sigma_N=\{s\in \{0,1\}^\mathbb{N} : \text{neither $0^N$ nor $1^N$ is a factor of $s$}\},
\end{equation}
and let $\mathcal{M}_N=\{\mu\in\mathcal{M}:\operatorname{supp}\mu\subset \Sigma_N\}$. By Lemma~\ref{lem:Sturmian slope}~(ii), $\operatorname{supp}\mu_\alpha\subset \Sigma_N$ holds for every $\alpha\in J_N$. Therefore, to prove that $f$ is continuous on $J_N$, it suffices to replace the role of \cite[Lemma~9.6]{Las25} there with the following claim.

\begin{clm}
 The function $\mu\mapsto\chi(\mathbb{A},\mu),\mu\in\mathcal{M}$ is continuous on $\mathcal{M}_N$.
\end{clm}

\begin{proof}[Proof of Claim]
Consider 
\[
\Sigma_N'=\{\underline{i}\in \{0,1\}^\mathbb{Z} : \text{neither $0^N$ nor $1^N$ is a factor of $\underline{i}$}\}
\]
as the natural extension of $\Sigma_N$ and let $\mathcal{M}_N'$ be the collection shift-invariant Borel probability measures for the sub-shift $\Sigma_N'$. The Lyapunov exponent $\chi(\mathbb{A},\nu)$ for $\nu\in \mathcal{M}_N'$ can be similarly defined, and it suffices to prove that $\nu\mapsto\chi(\mathbb{A},\nu)$ is continuous on $\mathcal{M}_N'$.

Note that $\Sigma_N'$ is a sub-shift of finite type. Also note that by Lemma~\ref{lem:cone}~(iv), if $w$ is a word of length $N$ with $w\ne 0^N,1^N$, then for $W=[w]_{\mathbb{A}}$, $\underline{W}(\overline{I_\mathbb{A}^+})\subset I_\mathbb{A}^+$. Then by \cite[Theorem~2.3]{ABV10}, it can be seen that $\mathbb{A}$ is uniformly hyperbolic with respect to $\Sigma_N'$. To proceed, we shall follow the proof of Equation~(1.7) in \cite[\S~2.1]{BM16}. Fixing a norm $|\cdot|$ on $\mathbb{R}^2$, thanks to the uniform hyperbolicity, it is not difficult to show that there exists a continuous map 
\[
e^+:\Sigma_N'\to \{(x,y)^t\in\mathbb{R}^2 : y>0,x/y\in I_{\mathbb{A}}^+\}
\] 
such that for every $\underline{i}=(i_n)_{n\in\mathbb{Z}}\in\Sigma_N'$, we have (denote $\mathbb{A}=(A_0,A_1)$ below): 
\begin{itemize}
  \item $\frac{A_{i_0}e^+(\underline{i})}{|A_{i_0}e^+(\underline{i})|} = \frac{e^+((i_{n+1})_{n\in\mathbb{Z}})}{|e^+((i_{n+1})_{n\in\mathbb{Z}})|}$;
  \item $\lim_{n\to\infty}\frac{1}{n}\log|A_{i_n}\cdots A_{i_1}A_{i_0}e^+(\underline{i})|=\lim_{n\to\infty}\frac{1}{n}\log\|A_{i_n}\cdots A_{i_1}A_{i_0}\|$, provided that the limit on the right hand side exists.
\end{itemize} 
Indeed, $e^+(\underline{i})$ can be taken as $(x(\underline{i}),1)^t$, where $x(\underline{i})$ denotes the unique point in $\cap_{n=1}^\infty \underline{A_{-i_1}\cdots A_{-i_n}}(I_\mathbb{A}^+)$. Then it follows that $\chi(\mathbb{A},\nu)=\int_{\Sigma_N'}\log|A_{i_0}e^+(\underline{i})|\,\mathrm{d}\nu(\underline{i})$, which implies that $\chi(\mathbb{A},\nu)$ is continuous in $\nu$.
\end{proof}

Since $N$ can be arbitrarily large, we have proved that $f$ is strictly concave on $(0,1)$. On the other hand, it is easy to see that as $n\to \infty$, $f(\frac{1}{n+1})=\frac{1}{n+1}\log\rho(A^nB)\to \log\rho(A) = f(0)$ and $f(\frac{n}{n+1})=\frac{1}{n+1}\log\rho(AB^n)\to \log\rho(B) = f(1)$. Combining this with $f$ being concave on $(0,1)$, we conclude that $f$ is continuous at $0,1$, which completes the proof.

<!-- lem-17: body_md -->
\label{lem:implicit}
  Let $X$ be a topological space and suppose that $f:X\times[0,1]\to \mathbb{R}$ satisfies the following properties.
\begin{itemize}
  \item For every $x\in X$, $\alpha\mapsto f(x,\alpha)$ is continuous and strictly concave on $[0,1]$, so this function attains its maximum at a unique point $\tau(x)\in[0,1]$.
  \item There exists a dense subset $D$ of $[0,1]$ such that for every $\alpha\in D$, $x\mapsto f(x,\alpha)$ is continuous on $X$.
\end{itemize}
Then the function $\tau:X\to [0,1]$ is continuous.
<!-- end lem-17 -->
<!-- proof-20: body_md -->
Given $x_0\in X$ and $\varepsilon>0$, it suffices to show that there exists an open neighborhood $U$ of $x_0$ such that $|\tau(x)-\tau(x_0)|<\varepsilon$ for any $x\in U$. Let us only prove the statement under the additional assumption $\tau(x_0)\in (0,1)$; the special cases $\tau(x_0)=0,1$ are similar and omitted.
  
Since $\tau(x_0)\in (0,1)$ and $D$ is dense in $[0,1]$, there exist $\alpha_1,\alpha_2,\beta_1,\beta_2\in D$ with $\tau(x_0)-\varepsilon <\alpha_1<\alpha_2<\tau(x_0)<\beta_1<\beta_2<\tau(x_0)+\varepsilon$. \label{prop:balanced}
Let $(A,B)\in\mathbf{G}^2$ and suppose that there exists $T\in\mathrm{GL}(2,\mathbb{R})$ such that both $T^{-1}AT$ and $T^{-1}BT$ are non-negative. Then $(A,B)$ is a balanced pair iff there exists $t>0$ such that $(A,tB)$ is a finiteness counter-example.
<!-- end prop-3 -->
<!-- proof-21: body_md -->
Let us prove the ``only if" part first. For every $t>0$, since $(A,tB)$ is a balanced pair, by Theorem~\ref{thm:main}, there exists a unique $\alpha(t)\in [0,1]$ such that $\mu_{\alpha(t)}$ is the unique maximizing measure of $(A,tB)$. By Proposition~\ref{prop:slope cont}, the function $t\mapsto\alpha(t)$ is continuous on $(0,+\infty)$. By the definition of $\alpha(t)$, it is easy to see that $\lim_{t\to 0^+}\alpha(t)=0$ and $\lim_{t\to +\infty}\alpha(t)=1$. Then there exists (uncountably many) $t>0$ with $\alpha(t)\notin\mathbb{Q}$, and $(A,tB)$ is a finiteness counter-example for such $t$.  

For the ``if" part, we may assume that $A,B\in\mathbf{G}$ are both non-negative. Since for $r,s,t>0$, $(A,tB)$ is a balanced pair iff $(rA,sB)$ is a balanced pair, it suffices to prove the following statement.

\begin{clm}
  Let $A,B\in\mathrm{SL}(2,\mathbb{R})\setminus\{I\}$ be non-negative matrices. If $\mathbb{A}=(A,B)$ does not admit a spectrum maximizing product, then $\mathbb{A}$ is a balanced pair.
\end{clm} 
It remains to prove the claim. To begin with, note that if $C\in\mathrm{SL}(2,\mathbb{R})\setminus\{I\}$ is non-negative, then 
$C\in\mathbf{A}$, and hence either $C$ is in the form of \eqref{eq:hyperbolic}, or $C\in\{P_x,P_x^t\}$ for some $x>0$. If $A$ and $B$ share a common eigenvector, it is well-known (and easy to see) that either $A$ or $B$ is a spectrum maximizing product of $\mathbb{A}$. Then we may assume that $A$ and $B$ share no common eigenvector, so there are three cases as follows.

\begin{itemize}
  \item $\{A,B\}=\{H_{s_1,u_1}^{\lambda_1},H_{s_2,u_2}^{\lambda_2}\}$ with $s_1\ne s_2$ and $u_1\ne u_2$.
  \item $\{A,B\}=\{H_{s,u}^\lambda,P_x\}$ or $\{H_{s,u}^\lambda,P_x^t\}$.
  \item $\{A,B\}=\{P_x,P_y^t\}$. 
\end{itemize}
Therefore, $\mathbb{A}$ is either a balanced pair or a crossing pair (recall its definition ahead of Lemma~\ref{lem:balanced pair}). According to \cite[Theorem~3.4]{PS21} or \cite[Corollary~5.4]{Las25},  for a crossing pair $(A,B)$, either $A$ or $B$ is a spectrum maximizing product. The proof is done.

\label{lem:cyclic}
Let $x\in \{0,1\}^*$ and suppose that $x^\infty$ is not balanced. Then there exists a factor $\tilde{x}$ of $x^\infty$ with $|\tilde{x}|=|x|$ and distinct factors $u,v$ of $\tilde{x}$ such that the following hold:

\begin{clm}
  Let $z$ be a finite word over the alphabet $\{a,b\}$. Suppose that:
  \begin{itemize}
    \item $bb$ is not a factor of $z^\infty$;
    \item there exists a word $w$ such that both $awa$ and $bwb$ are factors of $z$. 
  \end{itemize}
Denote $z=cy$ with $c\in\{a,b\}$. If $c=a$, let $z'=ya$; otherwise, let $z'=z$. Denote $w=aw'$. Then we have: 
\begin{itemize}
  \item  $z',w'$ are words over the alphabet $\{a,ba\}$;
  \item  $aw'a,baw'ba$ are factors of $z'$. 
\end{itemize}
\end{clm}
Now let us prove the lemma. Since $x^\infty$ is not balanced, it is known that there exists a factor $x_0$ of $x^\infty$ with $|x_0|=|x|$ such that $x_0$ is not balanced; see, for example, ``(i) implying (iii)" in \cite[Lemma~4.7]{HMST11}. Then by Lemma~\ref{lem:not balanced}, there exists a word $w_0$ such that for $u_0=0,v_0=1$, both $u_0w_0u_0$ and $v_0w_0v_0$ are factors of $x_0$. Now suppose that for some $k\ge 0$, words $x_k,u_k,v_k,w_k$ have been defined to satisfy the following induction hypothesis:
\begin{itemize}
  \item $x_k$ is a factor of $x^\infty$ with $|x_k|=|x|$ (as words over $\{0,1\}$);
  \item $u_k$ and $v_k$ are distinct factors of both $x_k$ and $w_k$;
  \item $x_k$ and $w_k$ are words over $\{u_k,v_k\}$;
  \item $u_kw_ku_k$ and $v_kw_kv_k$ are factors of $x_k$. 
\end{itemize}
Considering $x_k^\infty$ as an infinite word over $\{u_k,v_k\}$, if $v_kv_k$ is not a factor of $x_k^\infty$, then take $a=u_k,b=v_k$; otherwise, if $u_ku_k$ is not a factor of $x_k^\infty$, then take $a=v_k,b=u_k$. In both cases, we can apply the Claim to $z=x_k,w=w_k$ (as words over $\{a,b\}$), so that for $x_{k+1}=z',u_{k+1}=a,v_{k+1}=ba,w_{k+1}=w'$, $x_{k+1},u_{k+1},v_{k+1},w_{k+1}$ still satisfy the induction hypothesis above. Since in this process the length of $w_k$ (as a word over $\{0,1\}$) is strictly decreasing in $k$, the induction must be terminated at some step $k\ge 0$. Taking $\tilde{x}=x_k,u=u_k,v=v_k$ for such $k$, assertions (i) and (ii) in the lemma are satisfied; assertion (iii) follows from the construction of $u,v$ and Lemma~\ref{lem:balanced pair}~(ii).
