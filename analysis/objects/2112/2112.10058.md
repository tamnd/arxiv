<!-- definition-1: body_md -->
\label{2d1}

We call $A$ a \emph{dilation} if $A$ is a real $n\times n$ matrix $A$ and satisfies the following condition:
 $$\min_{\lambda\in\sigma(A)}|\lambda|>1,$$
where $\sigma(A)$ denotes the \emph{set of all eigenvalues of $A$}. We denote the eigenvalues of $A$ by $\lambda_1,\ldots,\lambda_n$, which satisfies $1<|\lambda_1|\leq\cdots\leq|\lambda_n|$. Here and thereafter, let $\lambda_-$ and $\lambda_+$ be two numbers such that
$1<\lambda_-<|\lambda_1|\leq\cdots\leq|\lambda_n|<\lambda_+$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{2d2}
A measurable mapping $\rho:\ {\mathbb{R}^n} \to [0,\infty)$
is called a \emph{homogeneous quasi-norm}, with respect to a dilation $A$, if
\begin{enumerate}
\item[\rm{(i)}] $\rho(x) \ge0$, and $\rho(x)=0 \Rightarrow x=0$;

\item[\rm{(ii)}] for any $x\in{\mathbb{R}^n}$, $\rho(Ax)=b\rho(x)$;

\item[\rm{(iii)}] for any $x$, $y\in{\mathbb{R}^n}$, $\rho(x+y)\le c[\rho(x)+\rho(y)]$, where $c$ is a positive constant independent of $x$ and $y$.
\end{enumerate}
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{2d3}
A \emph{step homogeneous quasi-norm} $\rho$ with respect to a dilation $A$, is defined by setting, for each $x\in{\mathbb{R}^n}$,
\begin{equation*}
\rho(x):=\left\{
\begin{array}{cl}
b^i&\hspace{0.6cm} {\rm when}\hspace{0.5cm} x\in
B_{i+1}\backslash B_i\\
0&\hspace{0.6cm} {\rm when}\hspace{0.5cm} x=\mathbf{0}
\end{array}\right..
\end{equation*}
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{2d4}
Let $\vec{p}:=(p_1,\ldots,p_n)\in{(0,\infty ]}^n$. The \emph{mixed-norm Lebesgue space} ${L}^{\vec{p}}({\mathbb{R}^n})$  is defined to be the set of all measurable functions $f$ such that
$${\Arrowvert f\Arrowvert}_{{L}^{\vec{p}}({\mathbb{R}^n})}:=\left( \int_{\mathbb R} \cdots\left(\int_{\mathbb R}|f(x_1,\ldots,x_n)|^{p_1} d x_1\right)^{\frac{p_2}{p_1}}\cdots d x_n \right)^{\frac{1}{p_n}} < \infty $$
with the usual modifications made when $p_i=\infty $ for some $i\in\{1,\ldots,n\}$.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
\label{2d5}
Let $\varphi\in{\mathcal S}({\mathbb{R}^n})$ satisfy $\int_{\mathbb{R}^n}\varphi(x)\,dx\neq0$. The
\emph{radial maximal function} $M_\varphi(f)$ of $f\in{\mathcal S}'({\mathbb{R}^n})$, with respect to $\varphi$, is defined by
\begin{align*}
M_\varphi(f)(x):= \sup_{k\in{\mathbb Z}}|f\ast\varphi_k(x)|, \quad \forall\, x\in{\mathbb{R}^n},
\end{align*}
here and thereafter, for any $\varphi\in{\mathcal S}({\mathbb{R}^n})$ and $k\in{\mathbb Z}$, $\varphi_k(\cdot):=b^{k}\varphi(A^{k}\cdot)$.
<!-- end definition-5 -->
<!-- definition-6: body_md -->
\label{2d6}
Let $\vec{p}\in{(0,\infty )}^n$ and $\varphi$ be as in Definition \ref{2d5}.
The \emph{anisotropic mixed-norm Hardy space} ${H_A^{\vec{p}}({\mathbb{R}^n})}$ is defined by setting
\begin{equation*}
{H_A^{\vec{p}}({\mathbb{R}^n})}:=\left\{f\in{\mathcal S}'({\mathbb{R}^n}):\ M_\varphi(f)\in{L^{\vec{p}}({\mathbb{R}^n})}\right\}.
\end{equation*}
Moreover, for any $f\in{H_A^{\vec{p}}({\mathbb{R}^n})}$, let
$\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}:=\| M_\varphi(f)\|_{{L^{\vec{p}}({\mathbb{R}^n})}}$.
<!-- end definition-6 -->
<!-- theorem-1: body_md -->
\label{3t1}
Let $\vec{p}\in{(0,1]}^n$. Then, for any
$f\in{H_A^{\vec{p}}({\mathbb{R}^n})}$, there exists a continuous function $F$ on ${\mathbb{R}^n}$ such that
$$\widehat{f}=F\quad in\quad {\mathcal S}'({\mathbb{R}^n}),$$
and there exists a positive constant $C$, depending only on $A$ and $\vec{p}$, such that, for any $x\in{\mathbb{R}^n}$,
\begin{align}\label{3e1}
|F(x)|\le C\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}\max\left\{[\rho_\ast(x)]^{\frac1{p_-}-1},
[\rho_\ast(x)]^{\frac1{p_+}-1}\right\},
\end{align}
here and thereafter, $\rho_*$ is as in Definition \ref{2d2} with $A$ replaced by its
transposed matrix $A^*$.
<!-- end theorem-1 -->
<!-- definition-7: body_md -->
\label{3d3}
\begin{enumerate}
Let $\vec{p}\in{(0,\infty )}^n$, $r\in(1,\infty ]$ and
\begin{align}\label{3e2}
s\in\left[\left\lfloor\left(\dfrac1{p_-}-1\right)
\dfrac{\ln b}{\ln\lambda_-}\right\rfloor,\infty \right)\cap{\mathbb Z}_+,
\end{align}
where $p_-$ is as in \eqref{2e3}.
\item[(\uppercase\expandafter{\romannumeral1})]
A measurable function $a$ on ${\mathbb{R}^n}$ is called an \emph{anisotropic $(\vec{p},r,s)$-atom}
(simply, a $(\vec{p},r,s)$-\emph{atom}) if
\begin{enumerate}
\item[{\rm (i)}] $\mathop\mathrm{\,supp\,} a \subset B$, where
$B\in\mathfrak{B}$ with $\mathfrak{B}$ as in \eqref{2d1};

\item[{\rm (ii)}] $\|a\|_{L^r({\mathbb{R}^n})}\le \frac{|B|^{1/r}}{\|\mathbf{1}_B\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}$;

\item[{\rm (iii)}] $\int_{{\mathbb{R}^n}}a(x)x^\gamma\,dx=0$ for any $\gamma\in{\mathbb Z}_+^n$
with $|\gamma|\le s$.
\end{enumerate}
\item[(\uppercase\expandafter{\romannumeral2})]
The \emph{anisotropic mixed-norm atomic Hardy space} ${H_A^{\vec{p},r,s}({\mathbb{R}^n})}$ is defined to be the
set of all $f\in{\mathcal S}'({\mathbb{R}^n})$ satisfying that there exist a sequence
$\{\lambda_i\}_{i\in{\mathbb N}}\subset\mathbb{C}$ and a sequence of $(\vec{p},r,s)$-atoms
$\{a_i\}_{i\in{\mathbb N}}$, supported respectively in
$\{B^{(i)}\}_{i\in{\mathbb N}}\subset\mathfrak{B}$ such that
\begin{align*}
f=\sum_{i\in{\mathbb N}}\lambda_ia_i
\quad\mathrm{in}\quad{\mathcal S}'({\mathbb{R}^n}).
\end{align*}
Furthermore, for any $f\in{H_A^{\vec{p},r,s}({\mathbb{R}^n})}$, let
\begin{align*}
\|f\|_{{H_A^{\vec{p},r,s}({\mathbb{R}^n})}}:=
{\inf}\left\|\left\{\sum_{i\in{\mathbb N}}
\left[\frac{|\lambda_i|\mathbf{1}_{B^{(i)}}}{\|\mathbf{1}_{B^{(i)}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}\right]^
{\underline{p}}\right\}^{1/\underline{p}}\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}},
\end{align*}
where the infimum is taken over all the decompositions of $f$ as above.
\end{enumerate}
<!-- end definition-7 -->
<!-- lemma-1: body_md -->
\label{3l1}
Let $\vec{p}$, $r$ and $s$ be as in Definition \ref{3d3}.
Assume that $a$ is a $(\vec{p},r,s)$-atom supported in $x_0+B_{i_0}$
with some $x_0\in{\mathbb{R}^n}$ and $i_0\in{\mathbb Z}$. Then there exists a positive
constant $C$, depending only on $A$ and $s$, such that,
for any $\alpha\in{\mathbb Z}_+^n$ with $|\alpha|\leq s$ and $x\in{\mathbb{R}^n}$,
\begin{align*}
\left|\partial^\alpha\left(\mathscr{F}D_{A}^{i_0}a\right)(x)\right|
\leq Cb^{-i_0/r}\|a\|_{L^r({\mathbb{R}^n})}\min\left\{1,|x|^{s-|\alpha|+1}\right\}{}.
\end{align*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Without loss of generality, we may take $x_0=\mathbf{0}$. This implies that $\mathop\mathrm{\,supp\,}(D_{A}^{i_0}a)\subset B_0$.
Then, for any $\alpha\in{\mathbb Z}_+^n$ with $|\alpha|\leq s$ and $x\in{\mathbb{R}^n}$, we easily obtain
$$\partial^\alpha\left(\mathscr{F}D_{A}^{i_0}a\right)(x)=\int_{B_0}
(-2\pi\imath\xi)^\alpha\left(D_{A}^{i_0}a\right)(\xi)e^{-2\pi\imath \xi\cdot x}\,d\xi.$$
Let $P$ be the degree $s-|\alpha|$ Taylor expansion polynomial of the function
$e^{-2\pi\imath \xi\cdot x}$ centered at the origin $\mathbf{0}$ respect to $\xi$.
Applying Definition \ref{3d3}(iii), the Taylor remainder theorem and the H\"{o}lder
inequality, we have
\begin{align*}
\left|\partial^\alpha\left(\mathscr{F}D_{A}^{i_0}a\right)(x)\right|
&=\left|\int_{B_0}(-2\pi\imath\xi)^\alpha
\left(D_{A}^{i_0}a\right)(\xi)e^{-2\pi\imath \xi\cdot x}\,d\xi\right|\\
&=\left|\int_{B_0}(-2\pi\imath\xi)^\alpha
\left(D_{A}^{i_0}a\right)(\xi)\left[e^{-2\pi\imath \xi\cdot x}-P(\xi)\right]\,d\xi\right|\nonumber\\
&\lesssim\int_{B_0}|\xi^\alpha|
\left|a\left(A^{i_0}\xi\right)\right||x|^{s-|\alpha|+1}|\xi|^{s-|\alpha|+1}\,d\xi\nonumber\\
&\lesssim|x|^{s-|\alpha|+1}\int_{B_0}
\left|a\left(A^{i_0}\xi\right)\right||\xi|^{s+1}\,d\xi\nonumber\\
&\lesssim|x|^{s-|\alpha|+1}b^{-i_0}\int_{B_{i_0}}|a(\xi)|\,d\xi\nonumber\\
&\lesssim|x|^{s-|\alpha|+1}b^{-i_0/r}\|a\|_{L^r({\mathbb{R}^n})}.\nonumber
\end{align*}

The other estimate can be directly deduced from the H\"{o}lder inequality, namely,
\begin{align*}
\left|\partial^\alpha\left(\mathscr{F}D_{A}^{i_0}a\right)(x)\right|
&=\left|\int_{B_0}(-2\pi\imath\xi)^\alpha
\left(D_{A}^{i_0}a\right)(\xi)e^{-2\pi\imath \xi\cdot x}\,d\xi\right|\\
&\lesssim\int_{B_0}

\lesssim b^{-i_0}\int_{B_{i_0}}|a(\xi)|\,d\xi
\lesssim b^{-i_0/r}\|a\|_{L^r({\mathbb{R}^n})}.
\end{align*}
This finishes the proof of Lemma \ref{3l1}.

<!-- lemma-3: body_md -->
\label{3l3}
Let $A$ be a given dilation. There exists a positive constant $C$ such that, for any $x\in{\mathbb{R}^n}$,
\begin{equation*}
\frac{1}{C} [\rho(x)]^{\ln \lambda_-/\ln b} \leq \left|x\right|
\leq C[\rho(x)]^{\ln \lambda_+/\ln b}\qquad { when}\ \rho(x)\in(1,\infty ),
\end{equation*}
and
\begin{equation*}
\frac{1}{C} [\rho(x)]^{\ln \lambda_+/\ln b} \leq \left|x\right|
\leq C[\rho(x)]^{\ln \lambda_-/\ln b}\qquad { when}\ \rho(x)\in[0,1],
\end{equation*}
where $\lambda_-$ and $\lambda_+$ are as in Definition \ref{2d1}.

<!-- proof-2: body_md -->
Let $a$ be a $(\vec{p},r,s)$-atom supported in $x_0+B_{i_0}$ with some $x_0\in{\mathbb{R}^n}$
and $i_0\in{\mathbb Z}$. Without loss of generality, we may assume $x_0=\mathbf{0}$.
By Lemma \ref{3l1} with $\alpha=(\overbrace{0,\ldots,0}^{n\ \rm times})$, we find that,
for any $x\in{\mathbb{R}^n}$,
\begin{align}\label{3e3}
\left|\widehat{a}(x)\right|
&=\left|b^{i_0}\left(D_{A^*}^{i_0}\mathscr{F}D_{A}^{i_0}a\right)(x)\right|
=\left|b^{i_0}\left(\mathscr{F}D_{A}^{i_0}a\right)\left((A^*)^{i_0}x\right)\right|\\
&\lesssim b^{i_0}b^{-i_0/r}\|a\|_{L^r({\mathbb{R}^n})}\min\left\{1,\left|(A^*)^{i_0}x\right|^{s+1}\right\}\nonumber\\
&\lesssim b^{i_0}\left\|\mathbf{1}_{B_{i_0}}\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}^{-1}
\min\left\{1,\left|(A^*)^{i_0}x\right|^{s+1}\right\}\nonumber.
\end{align}

Next, we show that
\begin{align}\label{3e4}
\left\|\mathbf{1}_{B_{i_0}}\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}^{-1}\lesssim \max\left\{b^{-\frac{i_0}{p_-}},\,b^{-\frac{i_0}{p_+}}\right\}.
\end{align}
Indeed, there exists a $K\in{\mathbb Z}$ large enough such that, if $i_0\in(K,\infty )\cap{\mathbb Z}$, then
\begin{align*}
\left\|\mathbf{1}_{B_{i_0}}\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}
&=\left( \int_{\mathbb R}\cdots \left(\int_{\mathbb R}|\mathbf{1}_{B_{i_0}}|^{p_1} d x_1\right)^{\frac{p_2}{p_1}}\cdots dx_n \right)^{\frac{1}{p_n}}\\
&\ge\left(\int_{\mathbb R} \cdots\left(\int_{\mathbb R}|\mathbf{1}_{B_{i_0}}|^{p_+} d x_1\right)^{\frac{p_+}{p_+}}\cdots dx_n \right)^{\frac{1}{p_+}}
=b^{\frac{i_0}{p_+}}.
\end{align*}
On the other hand, if $i_0\in(-\infty ,K]$, by \cite[Lemma 6.8]{hlyy20}, we conclude that, for any $\varepsilon\in(0,1)$,
$$\frac{\|\mathbf{1}_{B_{K}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}{\|\mathbf{1}_{B_{i_0}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}
\lesssim b^{(K-i_0)\frac{1+\varepsilon}{p_-}}.$$
Letting $\varepsilon\to0$, we have
$$\left\|\mathbf{1}_{B_{i_0}}\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}^{-1} \lesssim \frac{b^\frac{K}{p_-}}{\|\mathbf{1}_{B_{K}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}b^{-\frac{i_0}{p_-}}.$$
Thus, \eqref{3e4} holds true. From this and \eqref{3e3}, it follows that, for any $x\in{\mathbb{R}^n}$,

\begin{align}\label{3e6}
\left|\widehat{a}(x)\right|
\lesssim b^{i_0}\max\left\{b^{-\frac{i_0}{p_-}},\,b^{-\frac{i_0}{p_+}}\right\}
\min\left\{1,\left|(A^*)^{i_0}x\right|^{s+1}\right\}.
\end{align}
We next prove \eqref{3e5} by considering two cases:
$\rho_*(x)\leq b^{-i_0}$ and $\rho_*(x)>b^{-i_0}$.

\emph{Case 1).} $\rho_{*}(x)\leq b^{-i_0}$. In this case, note that
$\rho_*((A^*)^{i_0}x)\leq 1$. From \eqref{3e6}, Lemma \ref{3l3} and the fact that
$$1-\frac1{p_+}+(s+1)\frac{\ln\lambda_-}{\ln b}
\geq1-\frac1{p_-}+(s+1)\frac{\ln\lambda_-}{\ln b}>0,$$
we deduce that, for any $x\in{\mathbb{R}^n}$ satisfying $\rho_*(x)\leq b^{-i_0}$,
\begin{align}\label{3e7}
\left|\widehat{a}(x)\right|
&\lesssim b^{i_0}\max\left\{b^{-\frac{i_0}{p_-}},\,b^{-\frac{i_0}{p_+}}\right\}
\left[\rho_*\left((A^*)^{i_0}x\right)\right]^{(s+1)\frac{\ln\lambda_-}{\ln b}}\\
&\sim \max\left\{b^{i_0[1-\frac1{p_-}+(s+1)\frac{\ln\lambda_-}{\ln b}]},\,
b^{i_0[1-\frac1{p_+}+(s+1)\frac{\ln\lambda_-}{\ln b}]}\right\}
\left[\rho_*(x)\right]^{(s+1)\frac{\ln\lambda_-}{\ln b}}\nonumber\\
&\lesssim\max\left\{\left[\rho_*(x)\right]^{\frac1{p_-}-1},\,
\left[\rho_*(x)\right]^{\frac1{p_+}-1}\right\}.\nonumber
\end{align}
 This shows \eqref{3e5} for Case 1).

\emph{Case 2).} $\rho_*(x)>b^{-i_0}$. In this case, note that $\rho_*((A^*)^{i_0}x)>1$.
Using \eqref{3e6}, Lemma \ref{3l3} again and the fact that
$$\frac1{p_-}-1\geq\frac1{p_+}-1\geq0,$$
it is easy to see that, for any $x\in{\mathbb{R}^n}$ satisfying $\rho_*(x)>b^{-i_0}$,
\begin{align*}
\left|\widehat{a}(x)\right|
&\lesssim b^{i_0}\max\left\{b^{-\frac{i_0}{p_-}},\,b^{-\frac{i_0}{p_+}}\right\}
\sim\max\left\{b^{(1-\frac1{p_-})i_0},\,b^{(1-\frac1{p_+})i_0}\right\}\\
&\lesssim\max\left\{\left[\rho_*(x)\right]^{\frac1{p_-}-1},\,
\left[\rho_*(x)\right]^{\frac1{p_+}-1}\right\},
\end{align*}
which completes the proof of \eqref{3e5} and hence of Lemma \ref{3l2}.

<!-- lemma-5: body_md -->
\label{3l4}
Let $\vec{p}\in{(0,\infty )}^n$, $r\in(\max\{p_+,1\},\infty ]$ with $p_+$

<!-- proof-4: body_md -->
Let $\vec{p}\in{(0,1]}^n$, $r\in(\max\{p_+,1\},\infty ]$, $s$ be as in \eqref{3e2}
and $f\in{H_A^{\vec{p}}({\mathbb{R}^n})}$. Without loss of generality, we may assume that $\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}>0$. Then, by Lemma \ref{3l4} and Definition \ref{3d3}(\uppercase\expandafter{\romannumeral2}), we find that
there exist a sequence
$\{\lambda_i\}_{i\in{\mathbb N}}\subset\mathbb{C}$ and a sequence of $(\vec{p},r,s)$-atoms $\{a_i\}_{i\in{\mathbb N}}$, supported respectively in
$\{B^{(i)}\}_{i\in{\mathbb N}}\subset\mathfrak{B}$, such that
\begin{align}\label{3e9}
f=\sum_{i\in{\mathbb N}}\lambda_ia_i
\quad\mathrm{in}\quad{\mathcal S}'({\mathbb{R}^n}),
\end{align}
and
\begin{align}\label{3e10}
\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}\sim \left\|\left\{\sum_{i\in{\mathbb N}}
\left[\frac{|\lambda_i|\mathbf{1}_{B^{(i)}}}{\|\mathbf{1}_{B^{(i)}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}\right]^
{\underline{p}}\right\}^{1/\underline{p}}\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}.
\end{align}
Taking the Fourier transform on the both sides of \eqref{3e9}, we have
\begin{align}\label{3e11}
\widehat{f}=\sum_{i\in{\mathbb N}}\lambda_i\widehat{a_i}
\quad\mathrm{in}\quad{\mathcal S}'({\mathbb{R}^n}).
\end{align}
Note that a function $f\in L^1({\mathbb{R}^n})$ implies that
$\widehat{f}$ is well defined in ${\mathbb{R}^n}$, so does $\widehat{a_i}$ for any $i\in{\mathbb N}$. From
Lemmas \ref{3l2} and \ref{3l5}, and \eqref{3e10}, it follows that, for any $x\in{\mathbb{R}^n}$,
\begin{align}\label{3e12}
\sum_{i\in{\mathbb N}}|\lambda_i||\widehat{a_i}(x)|
&\lesssim\sum_{i\in{\mathbb N}}|\lambda_i|
\max\left\{\left[\rho_*(x)\right]^{\frac1{p_-}-1},\,\left[\rho_*(x)\right]^{\frac1{p_+}-1}\right\}\\
&\lesssim\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}
\max\left\{\left[\rho_*(x)\right]^{\frac1{p_-}-1},\,\left[\rho_*(x)\right]^{\frac1{p_+}-1}\right\}
<\infty .\nonumber
\end{align}
Therefore, for any $x\in{\mathbb{R}^n}$, the function
\begin{align}\label{3e13}
F(x):=\sum_{i\in{\mathbb N}}\lambda_i\widehat{a_i}(x)
\end{align}
is well defined pointwisely and
\begin{align*}
|F(x)|\lesssim\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}
\max\left\{\left[\rho_*(x)\right]^{\frac1{p_-}-1},\,\left[\rho_*(x)\right]^{\frac1{p_+}-1}\right\}.
\end{align*}

We next show the continuity of the function $F$ on ${\mathbb{R}^n}$. If we can prove that $F$ is continuous on any compact subset of ${\mathbb{R}^n}$, then
the continuity on ${\mathbb{R}^n}$ is obvious. Indeed, for any compact subset $E$, there exists a positive constant $K$, depending only on $A$ and
$E$, such that $\rho_*(x)\le K$ holds for every $x\in E$. By this and \eqref{3e12}, we conclude that, for any $x\in E$,

\begin{align*}
\sum_{i\in{\mathbb N}}|\lambda_i||\widehat{a_i}(x)|
&\lesssim\max\left\{K^{\frac1{p_-}-1},\,K^{\frac1{p_+}-1}\right\}\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}
<\infty .
\end{align*}
Thus, the summation $\sum_{i\in{\mathbb N}}\lambda_i\widehat{a_i}(\cdot)$ converges uniformly
on $E$. This, together with the fact that, for any $i\in{\mathbb N}$, $\widehat{a_i}(x)$ is continuous,
implies that $F$ is also continuous on any compact subset $E$ and hence on ${\mathbb{R}^n}$.

Finally, to complete the proof of Theorem \ref{3t1}, by \eqref{3e11} and \eqref{3e13},
we only need to show that
\begin{align}\label{3e14}
F=\sum_{i\in{\mathbb N}}\lambda_i\widehat{a_i}
\quad\mathrm{in}\quad{\mathcal S}'({\mathbb{R}^n}).
\end{align}
For this purpose, from Lemma \ref{3l2} and the definition of Schwartz functions, we deduce that,
for any $\varphi\in{\mathcal S}({\mathbb{R}^n})$ and $i\in{\mathbb N}$,
\begin{align*}
&\left|\int_{{\mathbb{R}^n}}\widehat{a_i}(x)\varphi(x)\,dx\right|\\
&\quad\leq\sum_{k=1}^\infty \int_{(A^*)^{k+1}B^*_0\setminus(A^*)^{k}B^*_0}
\max\left\{\left[\rho_*(x)\right]^{\frac1{p_-}-1},\,\left[\rho_*(x)\right]^{\frac1{p_+}-1}\right\}
|\varphi(x)|\,dx+\|\varphi\|_{L^1({\mathbb{R}^n})}\\
&\quad\lesssim\sum_{k=1}^\infty  b^kb^{k(\frac1{p_-}-1)}b^{-k(\lceil\frac1{p_-}-1\rceil+2)}
+\|\varphi\|_{L^1({\mathbb{R}^n})}\\
&\quad\sim\sum_{k=1}^\infty  b^{-k}+\|\varphi\|_{L^1({\mathbb{R}^n})},
\end{align*}
where $B^*_0$ is the unit dilated ball with respect to $A^*$.
This implies that there exists a positive constant $C$ such that $\left|\int_{{\mathbb{R}^n}}\widehat{a_i}(x)\varphi(x)\,dx\right|\le C$ holds true
uniformly for any $i\in{\mathbb Z}$. Combining this, Lemma \ref{3l5} and \eqref{3e10}, we have
\begin{align*}
\lim_{I\to\infty }\sum_{i=I+1}^\infty |\lambda_i|\left|\int_{{\mathbb{R}^n}}\widehat{a_i}(x)\varphi(x)\,dx\right|
\lesssim \lim_{I\to\infty }\sum_{i=I+1}^\infty |\lambda_i|=0.
\end{align*}
Therefore, for any $\varphi\in{\mathcal S}({\mathbb{R}^n})$,
$$\langle F,\varphi\rangle
=\lim_{I\to\infty }\left\langle\sum_{i=1}^I\lambda_i\widehat{a_i},\,\varphi\right\rangle.$$
This finishes the proof of \eqref{3e14} and hence of Theorem \ref{3t1}.

\begin{align}\label{4e9}
\left\|\left\{\sum_{i\in{\mathbb N}}
\left[\frac{|\lambda_i|\mathbf{1}_{B^{(i)}}}{\|\mathbf{1}_{B^{(i)}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}\right]^
{\underline{p}}\right\}^{1/\underline{p}}\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}\lesssim\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}<\infty .
\end{align}
To prove Theorem \ref{4t2}, it suffices to show that \eqref{4e8}
holds true for the function $F$ as in \eqref{4e3}. For this purpose,
by the fact that $\underline{p}\le p_+\le1$, the inverse Minkovski inequality and \eqref{4e9}, we have
\begin{align}\label{4e12}
\left(\sum_{i\in{\mathbb N}}|\lambda_i|^{p_+}\right)^{1/p_+}
&=\left(\sum_{i\in{\mathbb N}}\left\|\frac{|\lambda_i|{\mathbf 1}_{B^{(i)}}}
{\|{\mathbf 1}_{B^{(i)}}\|_{L^{\vec{p}}(\mathbb{R}^n)}}
\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}^{p_+}\right)^{1/p_+}\\
&=\left(\sum_{i\in{\mathbb N}}
\left\|\frac{|\lambda_i|^{p_+}{\mathbf 1}_{B^{(i)}}}{\|{\mathbf 1}_{B^{(i)}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}^{p_+}}
\right\|_{L^{\vec{p}/p_+}({\mathbb{R}^n})}\right)^{1/p_+}\nonumber\\
&\le \left\|\sum_{i\in{\mathbb N}}
\left[\frac{|\lambda_i|{\mathbf 1}_{B^{(i)}}}{\|{\mathbf 1}_{B^{(i)}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}
\right]^{p_+}\right\|_{L^{\vec{p}/p_+}({\mathbb{R}^n})}^{1/p_+}\nonumber\\
&=\left\|\left\{\sum_{i\in{\mathbb N}}\left[\frac{|\lambda_i|{\mathbf 1}_{B^{(i)}}}
{\|{\mathbf 1}_{B^{(i)}}\|_{{L^{\vec{p}}({\mathbb{R}^n})}}}
\right]^{p_+}\right\}^{1/p_+}\right\|_{L^{\vec{p}}(\mathbb{R}^n)}\nonumber\\
&\le\left\|\left\{\sum_{i\in{\mathbb N}}
\left[\frac{|\lambda_i|{\mathbf 1}_{B^{(i)}}}{\|{\mathbf 1}_{B^{(i)}}\|_
{{L^{\vec{p}}({\mathbb{R}^n})}}}\right]^{\underline{p}}\right\}^{1/\underline{p}}
\right\|_{{L^{\vec{p}}({\mathbb{R}^n})}}\nonumber\\
&\lesssim\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}.\nonumber
\end{align}

On another hand, from \eqref{4e3}, the fact that $p_+\in(0,1]$, \eqref{3e8}
and the Fatou lemma, it follows that
\begin{align}\label{4e13}
&\int_{{\mathbb{R}^n}}|F(x)|^{p_+}
\min\left\{\left[\rho_{*}(x)\right]^{p_+-\frac {p_+}{p_-}-1},\,
\left[\rho_{*}(x)\right]^{p_+-2}\right\}\, dx\\
&\quad\le \sum_{i\in{\mathbb N}}|\lambda_i|^{p_+}\int_{{\mathbb{R}^n}}\left[\left|\widehat{a_i}(x)\right|
\min\left\{\left[\rho_{*}(x)\right]^{1-\frac 1{p_-}-\frac 1{p_+}},\,
\left[\rho_{*}(x)\right]^{1-\frac 2{p_+}}\right\}\right]^{p_+}\,dx.\nonumber
\end{align}
Next, we devote to proving the following uniform estimate for all $(\vec{p},2,s)$-atoms, namely,
\begin{align}\label{4e14}
\left(\int_{{\mathbb{R}^n}}\left[\left|\widehat{a}(x)\right|
\min\left\{\left[\rho_{*}(x)\right]^{1-\frac 1{p_-}-\frac 1{p_+}},\,
\left[\rho_{*}(x)\right]^{1-\frac 2{p_+}}
\right\}\right]^{p_+}\,dx\right)^{1/p_+}\le M,
\end{align}
where $M$ is a positive constant independent of $a$. Assume that \eqref{4e14} holds true for the moment. Combining this, \eqref{4e12} and \eqref{4e13}, we conclude that
\begin{align*}
&\left(\int_{{\mathbb{R}^n}}|F(x)|^{p_+}
\min\left\{\left[\rho_{*}(x)\right]^{p_+-\frac {p_+}{p_-}-1},\,
\left[\rho_{*}(x)\right]^{p_+-2}\right\}\,dx\right)^{1/{p_+}}\\
&\quad\le M\left(\sum_{i\in{\mathbb N}}|\lambda_i|^{p_+}\right)^{1/{p_+}}
\lesssim\|f\|_{{H_A^{\vec{p}}({\mathbb{R}^n})}}.
\end{align*}
This is the desired conclusion \eqref{4e8}.

Thus, the rest of the whole proof is to show the assertion
\eqref{4e14}. Indeed, for any $(\vec{p},2,s)$-atom $a$ supported in
a dilated ball $x_0+B_{i_0}$ with some $x_0\in{\mathbb{R}^n}$ and $i_0\in{\mathbb Z}$,
it is easy to see that
\begin{align*}
&\left(\int_{{\mathbb{R}^n}}\left[\left|\widehat{a}(x)\right|
\min\left\{\left[\rho_{*}(x)\right]^{1-\frac 1{p_-}-\frac 1{p_+}},\,
\left[\rho_{*}(x)\right]^{1-\frac 2{p_+}}\right\}\right]^{p_+}\,dx\right)^{1/{p_+}}\\
&\quad\lesssim\left(\int_{(A^*)^{-i_0+1}B_0^*}\left[\left|\widehat{a}(x)\right|
\min\left\{\left[\rho_{*}(x)\right]^{1-\frac 1{p_-}-\frac 1{p_+}},\,
\left[\rho_{*}(x)\right]^{1-\frac 2{p_+}}\right\}\right]^{p_+}\,dx\right)^{1/{p_+}}\\
&\qquad
+\left(\int_{((A^*)^{-i_0+1}B_0^*)^{\complement}}\left[\left|\widehat{a}(x)\right|
\min\left\{\left[\rho_{*}(x)\right]^{1-\frac 1{p_-}-\frac 1{p_+}},\,
\left[\rho_{*}(x)\right]^{1-\frac 2{p_+}}\right\}\right]^{p_+}\,dx\right)^{1/{p_+}}\\
&\quad=:{\rm I}_1+{\rm I}_2,
\end{align*}
where $B^*_0$ is the unit dilated ball with respect to $A^*$.

Let $\theta$ be a fixed positive constant such that
$$1-\frac1{p_+}+(s+1)\frac{\ln\lambda_-}{\ln b}-\theta
\geq1-\frac1{p_-}+(s+1)\frac{\ln\lambda_-}{\ln b}-\theta>0.$$
Then, to deal with ${\rm I}_1$, by \eqref{3e7}, we know that
\begin{align*}
{\rm I}_1
&\lesssim b^{i_0[1+(s+1)\frac{\ln\lambda_-}{\ln b}]}
\max\left\{b^{-\frac{i_0}{p_-}},\,b^{-\frac{i_0}{p_+}}\right\}\\
&\hspace{0.25cm}\times\left(\int_{(A^*)^{-i_0+1}B_0^*}\left[
\min\left\{\left[\rho_{*}(x)\right]^{1-\frac 1{p_-}-\frac 1{p_+}+(s+1)\frac{\ln\lambda_-}{\ln b}},\,
\left[\rho_{*}(x)\right]^{1-\frac 2{p_+}+(s+1)\frac{\ln\lambda_-}{\ln b}}\right\}\right]^{p_+}\,dx\right)^{1/{p_+}}\\
&\lesssim b^{i_0[1+(s+1)\frac{\ln\lambda_-}{\ln b}]}
\max\left\{b^{-\frac{i_0}{p_-}},\,b^{-\frac{i_0}{p_+}}\right\}\\
&\hspace{0.25cm}\times\min\left\{b^{-i_0[1-\frac1{p_-}+(s+1)
\frac{\ln\lambda_-}{\ln b}-\theta]},\,b^{-i_0[1-\frac1{p_+}+(s+1)
\frac{\ln\lambda_-}{\ln b}-\theta]}\right\}\left(\int_{(A^*)^{-i_0+1}B_0^*}
\left[\rho_{*}(x)\right]^{\theta p_+-1}\,dx\right)^{1/{p_+}}\\
&\sim b^{i_0\theta}\left[\sum_{k\in{\mathbb Z}\setminus{\mathbb N}}b^{-i_0+k}(b-1)b^{(-i_0+k)(\theta p_+-1)}\right]^{1/{p_+}}\sim\left(\frac{b-1}{1-b^{-\theta p_+}}\right)^{\frac1{p_+}}.
\end{align*}
As for the estimate of ${\rm I}_2$, by the H\"{o}lder inequality, the Plancherel theorem,
the fact that $0<p_-\le p_+\le1$ and the size condition of $a$, we obtain
\begin{align*}
{\rm I}_2
&\lesssim\left\{\int_{((A^*)^{-i_0+1}B_0^*)^{\complement}}
\left|\widehat{a}(x)\right|^2\,dx\right\}^{\frac12}\\
&\hspace{0.25cm}\hspace{0.25cm}\times\left\{\int_{((A^*)^{-i_0+1}B_0^*)^{\complement}}
\left[\min\left\{\left[\rho_{*}(x)\right]^{1-\frac 1{p_-}-\frac 1{p_+}},\,
\left[\rho_{*}(x)\right]^{1-\frac 2{p_+}}\right\}\right]^{\frac{2p_+}{2-p_+}}\,dx\right\}^{\frac{2-p_+}{2p_+}}\\
&\lesssim\|a\|_{L^2({\mathbb{R}^n})}\left\{\sum_{k\in{\mathbb N}} b^{-i_0+k}(b-1)
\left[\min\left\{b^{(-i_0+k)(1-\frac 1{p_-}-\frac 1{p_+})},\,
b^{(-i_0+k)(1-\frac 2{p_+})}\right\}\right]^{\frac{2p_+}{2-p_+}}\right\}^{\frac{2-p_+}{2p_+}}\\
&\lesssim\|a\|_{L^2({\mathbb{R}^n})}\left\{b^{-i_0}
\left[\min\left\{b^{-i_0(1-\frac 1{p_-}-\frac 1{p_+})},\,
b^{-i_0(1-\frac 2{p_+})}\right\}\right]^{\frac{2p_+}{2-p_+}}\right\}^{\frac{2-p_+}{2p_+}}\\
&\lesssim\max\left\{b^{i_0(\frac12-\frac 1{p_-})},\,
b^{i_0(\frac12-\frac 1{p_+})}\right\}
\min\left\{b^{-i_0(\frac12-\frac 1{p_-})},\,
b^{-i_0(\frac12-\frac 1{p_+})}\right\}\\
&\sim 1.
\end{align*}
This finishes the proof of \eqref{4e14} and hence of Theorem \ref{4t2}.

Actually, when $\vec{p}=(p,\ldots,p)\in(0,1]^n$, the Hardy space ${H_A^{\vec{p}}({\mathbb{R}^n})}$ in Theorem \ref{4t2} is just
the anisotropic Hardy space $H^p_A({\mathbb{R}^n})$ from \cite{bow03} in the sense of equivalent quasi-norms. Thus, we point out that Theorem \ref{4t2} covers \cite[Corollary 8]{bw13}.
Moreover, if $A=d\,{\rm I}_{n\times n}$ for some $d\in{\mathbb R}$ with $|d|\in(1,\infty )$,
then the anisotropic mixed-norm Hardy space $H^{\vec{p}}_A({\mathbb{R}^n})$, with
$\vec{p}=(p,\ldots,p)\in(0,1]^n$, coincides with the classical Hardy space $H^p({\mathbb{R}^n})$ of
Fefferman and Stein \cite{fs72}. In this case, $\rho_*(x)\sim|x|^n$ for any $x\in{\mathbb{R}^n}$, and hence \eqref{4e6} is just the classic Hardy--Littlewood inequality as in \eqref{1e2}.
