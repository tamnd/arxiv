<!-- theorem-1: body_md -->
\label{thm.main}
	Let Assumption~\ref{as.feasibility} hold. 
	Fix $p\in\left(0,1\right)$ and $T\geq \max(|X||A|,|B||Y|)$. 
	In Algorithm~\ref{UCB-MPD}, we set
	$V=L\sqrt{T}$, $\eta = 1/(TL)$, and $\theta = 1/T$.
	Then, with probability $1-p$, the regret~\eqref{eq.regret} and the constraint violation~\eqref{eq.violation} satisfy 
	\[
	\begin{array}{rcl}
	\text{\normalfont Regret}(T),\; \text{\normalfont Violation}(T) & \leq & 
	\widetilde O \big(\, (|X|+|Y|) \,L \sqrt{T(|A|+|B|)} \,\big)
	\end{array}
	\]
	where $\widetilde O(\cdot)$ hides the logarithmic factor $\log\tfrac{1}{p}$.

In~\eqref{eq.primal12c}, we have two standard mirror descent problems. Since two problems enjoy the same structure, we only prove an efficient solution to the first problem~\eqref{eq.primal1c}.
	
	By the online mirror descent optimization~\citep{zimin2013online}, Problem~\eqref{eq.primal1c} is equivalent to
	\begin{equation}\label{eq.primal12steps1}
	\bar q_1^{\,t} 
	\;=\;
	\displaystyle \operatorname*{argmin}_{q_1}  \;
	\eta \, \langle{q_1},{\phi_1^{t-1}}\rangle 
	\, +\, 
	\displaystyle D\big(q_1\,\vert\,  \widetilde{q}_1^{\,t-1}\big)
	\;
	\text{ and }
	\;
	\widehat q_1^{\,t}
	\;=\;
	\displaystyle \operatorname*{argmin}_{q_1\,\in\,\Delta(k_1^t)}  \;
	\displaystyle D\big(q_1\,\vert\,  \bar{q}_1^{\,t}\big)
	\end{equation}
	where $\bar q_1^{\,t}$ is a solution to an unconstrained problem and $\widehat q_1^{\,t}$ simply takes the projection of $\bar q_1^{\,t}$ to the domain $\Delta(k_1^t)$ in the unnormalized Kullback-Leibler divergence. 
	
	It is straightforward to compute a closed-form solution for the unconstrained problem,
	\begin{equation}\label{eq.unconstrained}
	\bar q_1^{\,t} (x,a) \;=\; \widetilde q_1^{\,t} (x,a) \,{ 
   
  \ClassWarning{jmlr}{Obsolete command

  \csname rm \endcsname
 
 e}^{- \eta\phi_1^{t-1}(x,a)},\,\text{ for all } (x,a)\in X\times A.
	\end{equation}
	
	To compute the projection of $\bar q_1^{\,t}$, we recall that the domain set $\Delta(k_1^t)$ explicitly takes the following linear constraints on $q_1$: $X\times A\to [0,1]$,
	\begin{equation}\label{eq.deltaset}
	\Delta(k_1^t) \;\mathrel{\mathop:}=\; \{ q_1: X\times A\to [0,1] \,\vert\, q_1 \text{satisfies the following (i), (ii), (iii), (iv)} \}
	\end{equation}
	\begin{itemize}
		\item [(i)] $q_1(x, a) = \sum_{x'\,\in\,X_{\ell+1}} q_1(x,a,x')$ for $(x,a)\in X_\ell\times A$ and $\ell \in\{0,1,\cdots,L-1\}$;
		
		\item [(ii)] $\sum_{x\in X_\ell}\sum_{a\in A}\sum_{x'\in X_{\ell+1}} q_1(x, a, x') = 1$ for $\ell \in\{0,1,\cdots,L-1\}$; 
		
		\item [(iii)] $\sum_{x\in X_{\ell-1}}\sum_{a\in A} q_1(x, a, x') =\sum_{a\in A}\sum_{x''\in X_{\ell+1}} q_1(x', a, x'')$ for $x'\in X_\ell$ and $\ell \in\{1,\cdots,L-1\}$;
		
		\item [(iv)] 
		$q_1(x,a,x')-\bar P_1^{k_1}(x'\,\vert\,x,a) \sum_{x''\,\in\,X_{\ell+1}}q_1(x,a,x'') \leq \epsilon(x,a,x')$, 
		
		$\bar P_1^{k_1}(x'\,\vert\,x,a) \sum_{x''\,\in\,X_{\ell+1}}q_1(x,a,x'')-q_1(x,a,x') \leq \epsilon(x,a,x')$,

	&& \!\!\!\! \!\!\!\! \!\! 
	\eta^{-1} \big(D(x,x') - D(x,x^\star)\big)
	\\[0.2cm]
	&=&
	\eta^{-1}\big(
	\phi(x^\star) - \phi(x') - \langle \nabla\phi(x'), x-x'\rangle
	+ \langle \nabla\phi(x^\star), x-x^\star\rangle
	\big)
	\\[0.2cm]
	&=&\eta^{-1}\big(
	\phi(x^\star) - \phi(x') 
	- \eta^{-1} \langle \nabla\phi(x'), x^\star-x'\rangle 	\big)-\langle \nabla_xf(x^\star,y^\star) , x-x^\star\rangle
	\\[0.2cm]
	&&
	\,+\, \langle \nabla_xf(x^\star,y^\star) + \eta^{-1} \big(\nabla\phi(x^\star)-\nabla\phi(x')\big), x-x^\star\rangle
	\\[0.2cm]
	&=&\eta^{-1}D(x^\star,x')-\langle \nabla_xf(x^\star,y^\star) , x-x^\star\rangle
	\\[0.2cm]
	&&
	\,+\, \langle \nabla_xf(x^\star,y^\star) + \eta^{-1} \big(\nabla\phi(x^\star)-\nabla\phi(x')\big), x-x^\star\rangle.
	\end{array}
	\]
	Application of~\eqref{eq.1stx} leads to
	\begin{equation}\label{eq.pushbackx}
	\begin{array}{rcl}
	\eta^{-1} \big(D(x,x') - D(x,x^\star)\big)
	&\geq&
	\eta^{-1}D(x^\star,x')
	-\langle \nabla_xf(x^\star,y^\star) , x-x^\star\rangle
	\\[0.2cm]
	&\geq&
	\eta^{-1}D(x^\star,x')
	+  f(x^\star,y^\star)- f(x,y^\star) 
	\end{array}
	\end{equation}
	where the last inequality is due to the convexity fo $f(x,y^\star)$ in $x$:
	$f(x,y^\star)\geq f(x^\star,y^\star)+\langle\nabla_x f(x^\star,y^\star),x-x^\star\rangle$.
	
	Similarly, we work on $\eta^{-1} \big(D(y,y') - D(y,y^\star)\big)$ and~\eqref{eq.1sty}.
	\begin{equation}\label{eq.pushbacky}
	\begin{array}{rcl}
	\eta^{-1} \big(D(y,y') - D(y,y^\star)\big)
	&\geq&
	\eta^{-1}D(y^\star,y')
	+ f(x^\star,y) -  f(x^\star,y^\star).
	\end{array}
	\end{equation}
	
	Finally, we conclude the proof by adding~\eqref{eq.pushbackx} to~\eqref{eq.pushbacky} from both sides.

\label{lem.D_lb}
	Let $q(x,a,x')$ and $q'(x,a,x')$ be two occupancy measures, and $q(x,a)$ and $q'(x,a)$ be the associated state-action visitation probability distributions. Then,
	\[
	D(q, q') \;\geq\;\frac{1}{2L} \left\|{q-q'}\right\|_1^2.
	\]

<!-- proof-6: body_md -->
We recall  $q(x,a)$ and $q'(x,a)$,
	\[
	q(x,a)\;=\;\sum_{x'\,\in\,X_\ell} q(x,a,x')
	\;\text{ and }\;
	q'(x,a)\;=\;\sum_{x'\,\in\,X_\ell} q'(x,a,x')
	\]
	where $\ell$ is the layer that $x$ belongs to. We note that $q(x,a)$ and $q'(x,a)$ define probability laws for each $\ell\in\{ 0,1,\ldots,L-1 \}$, and  $\sum_{x\,\in\,X_\ell}\sum_{a\,\in\,A}q(x,a) =  \sum_{x\,\in\,X_\ell}\sum_{a\,\in\,A}q'(x,a) = 1$.
	
	By the definition,
	\[
	\begin{array}{rcl}
	D(q, q') 
	& = & 
	\displaystyle
	\sum_{\ell\,=\,0}^{L-1} \sum_{x\,\in\,X_\ell} \sum_{a\,\in\,A} q(x,a) \log\frac{q(x,a)}{q'(x,a)}
	\,-\,
	\sum_{\ell\,=\,0}^{L-1} \sum_{x\,\in\,X_\ell} \sum_{a\,\in\,A}  \big(q(x,a)-q'(x,a)\big)
	\\[0.2cm]
	& = & 
	\displaystyle
	\sum_{\ell\,=\,0}^{L-1} \sum_{x\,\in\,X_\ell} \sum_{a\,\in\,A} q(x,a) \log\frac{q(x,a)}{q'(x,a)}
	\\[0.2cm]
	& \geq & 
	\displaystyle\frac{1}{2}
	\sum_{\ell\,=\,0}^{L-1} \left\|{ q(x,a)-q'(x,a)}\right\|_1^2
	\\[0.2cm]
	& \geq & 
	\displaystyle\frac{1}{2L}
	\left( \sum_{\ell\,=\,0}^{L-1} \left\|{ q(x,a)-q'(x,a)}\right\|_1\right)^2
	\\[0.2cm]
	& = & 
	\displaystyle\frac{1}{2L} \left\|{ q-q'}\right\|_1^2
	\end{array}
	\]
	where we apply the Pinsker's inequality to $\sum_{x\,\in\,X_\ell} \sum_{a\,\in\,A} q(x,a) \log\frac{q(x,a)}{q'(x,a)}$ in the first inequality.

	\end{array}

	where the first inequality is clear from two cases for $\max(\cdot)$ and the second inequality is due to $\langle{\widehat q_1^{\,t+1}},{g^{t}}\rangle\in [0,L]$, $b_1\in [0,L]$. Similarly, $\left|  \lambda_2^{t+1}  - \lambda_2^{t} \right| \leq L$. Hence,

	\;\leq\; 2t_0 L

Let $\Delta^t \mathrel{\mathop:}= \frac{1}{2} \big( \left\|{\lambda^t}\right\|^2-\left\|{\lambda^{t-1}}\right\|^2 \big)$. Clearly, $\Delta^t = \Delta_1^t+\Delta_2^t$.

By the dual update~\eqref{eq.dual_s},
	\begin{subequations}
		\begin{equation}\label{eq.lambda_lb_s1}
		\begin{array}{rcl}
		\lambda_1^{t} &=& \max\! \Big( \lambda_1^{t-1} \,+\, \big(\left\langle \widehat q_1^{\,t},g^{t-1} \right\rangle - b_1\, \big),\; 0 \Big)
		\\[0.2cm]
		&\geq &  \lambda_1^{t-1} \,+\, \big(\left\langle \widehat q_1^{\,t},g^{t-1} \right\rangle - b_1\, \big)
		\\[0.2cm]
		&= &  \lambda_1^{t-1} \,+\, \big(\left\langle \widehat q_1^{\,t-1},g^{t-1} \right\rangle - b_1\, \big)\,+\, \left\langle \widehat q_1^{\,t} - \widehat q_1^{\,t-1},g^{t-1} \right\rangle
		\\[0.2cm]
		&\geq &  \lambda_1^{t-1} \,+\, \big(\left\langle \widehat q_1^{\,t-1},g^{t-1} \right\rangle - b_1\, \big)\,-\, \left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 
		\end{array}
		\end{equation}
		where the last inequality is due to: $\langle{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}},{g^{t-1}}\rangle\leq \Vert \widehat q_1^{\,t} - \widehat q_1^{\,t-1}\Vert_1 \Vert g^{t-1}\Vert_\infty$,
		and $\Vert g^{t-1}\Vert_\infty\in [0,1]$. Similarly, 
		\begin{equation}\label{eq.lambda_lb_s2}
		\lambda_2^{t} \;\geq\; \lambda_2^{t-1} \,+\, \big(\left\langle \widehat q_2^{\,t-1},h^{t-1} \right\rangle - b_2\, \big)\,-\, \left\|{\widehat q_2^{\,t} - \widehat q_2^{\,t-1}}\right\|_1.
		\end{equation}
	\end{subequations}
	We note that $\lambda_1^0=\lambda_2^0=0$ from the initialization. Summing up both sides of~\eqref{eq.lambda_lb_s1} from $t=1$ to $t=T$ leads to
	\begin{subequations}
		\begin{equation}\label{eq.violation_dqq_s1}
		\sum_{t\,=\,0}^{T-1} \big(\left\langle \widehat q_1^{\,t},g^{t} \right\rangle -b_1\, \big)
		\;\leq\; 
		\lambda_1^T
		\,+\, 
		\sum_{t\,=\,1}^{T}
		\left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1.
		\end{equation}
		Similarly, 
		\begin{equation}\label{eq.violation_dqq_s2}
		\sum_{t\,=\,0}^{T-1} \big(\left\langle \widehat q_2^{\,t},h^{t} \right\rangle -b_2\, \big)
		\;\leq\; 
		\lambda_2^T
		\,+\, 
		\sum_{t\,=\,1}^{T}
		\left\|{\widehat q_2^{\,t} - \widehat q_2^{\,t-1}}\right\|_1.
		\end{equation}
	\end{subequations}
	Hence,
	\begin{equation}\label{eq.violation_dqq_s}
	{\widehat{\text{\normalfont Violation}_1}(T)}, {\widehat{\text{\normalfont Violation}_2}(T)}
	\;\leq\;
	\left\|{\lambda^T}\right\|
	\,+\,
	\sum_{t\,=\,1}^{T}
	\left( \left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 + \left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 \right).
	\end{equation}
	
	We recall $ \widehat q_1^{\,t}\in \Delta(k_1^t)$, $ \widehat q_2^{\,t}\in \Delta(k_2^t)$ in the primal update~\eqref{eq.primal_s} and $\Delta(k_1^t)$ and $\Delta(k_2^t)$ in the confidence sets~\eqref{eq.confidence}.
	To bound $\left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 
	+\left\|{\widehat q_2^{\,t}-\widehat q_2^{\,t-1}}\right\|_1$, we consider two cases: (i) $k_1^t = k_1^{t-1}$ and $k_2^t = k_2^{t-1}$; (ii) either $k_1^t \neq k_1^{t-1}$ or $k_2^t \neq k_2^{t-1}$.
	
	\noindent\textbf{Case~(i)}.	In this case, we have: $ \widehat q_1^{\,t}$, $\widehat q_1^{\,t-1}\in \Delta(k_1^t)$, $ \widehat q_2^{\,t}$, $\widehat q_2^{\,t-1}\in \Delta(k_2^t)$. 
	We begin with the primal update~\eqref{eq.primal} and apply Lemma~\ref{lem.pushback} with, 
	\[
	f(x,y) \vert_{x\,=\,q_1,\, y\,=\,q_2} 
	\;=\;
	V\, \big\langle{q_1\cdot \widehat q_2^{\,t-1}+\widehat q_1^{\,t-1}\cdot q_2},{r^{t-1}}\big\rangle 
	\,+\,
	\lambda_1^{t-1} \langle{q_1},{g^{t-1}}\rangle 
	\,-\,\lambda_2^{t-1} \langle{q_2},{h^{t-1}}\rangle 
	\]
	and $x^\star = \widehat q_1^{\,t}$, $y^\star= \widehat q_2^{\,t}$, $x' = \widetilde q_1^{\,t-1}$, $y'=\widetilde q_2^{\,t-1}$, $x = \widetilde q_1^{\,t-1}$, and $y=\widetilde q_2^{\,t-1}$. Thus, 
	\[
	\begin{array}{rcl}
	&& \!\!\!\! \!\!\!\! \!\!
	V\, \big\langle{\widehat q_1^{\,t}\cdot \widehat q_2^{\,t-1}+\widehat q_1^{\,t-1}\cdot  \widetilde q_2^{\,t-1}},{r^{t-1}}\big\rangle 
	\,+\,
	\lambda_1^{t-1} \langle{\widehat q_1^{\,t}},{g^{t-1}}\rangle 
	\,-\,
	\lambda_2^{t-1} \langle{ \widetilde q_2^{\,t-1}},{h^{t-1}}\rangle 
	\\[0.2cm]
	&& 
	\,+\, \eta^{-1} \big(D(\widehat q_1^{\,t},\widetilde q_1^{\,t-1}) + D(\widehat q_2^{\,t}, \widetilde q_2^{\,t-1})\big)
	\\[0.2cm]
	&\leq& V\, \big\langle{ \widetilde q_1^{\,t-1}\cdot \widehat q_2^{\,t-1}+\widehat q_1^{\,t-1}\cdot \widehat q_2^{\,t}},{r^{t-1}}\big\rangle 
	\,+\,
	\lambda_1^{t-1} \langle{ \widetilde q_1^{\,t-1}},{g^{t-1}}\rangle 
	\,-\,
	\lambda_2^{t-1} \langle{ \widehat q_2^{\,t}},{h^{t-1}}\rangle 
	\\[0.2cm]
	&&
	\,-\, \eta^{-1} \big(D( \widetilde q_1^{\,t-1},\widehat q_1^{\,t}) + D( \widetilde q_2^{\,t-1}, \widehat q_2^{\,t}) \big).
	\end{array}
	\]
	or, equivalently,
	\begin{equation}\label{eq.pushback_I_s}
	\begin{array}{rcl}
	&& \!\!\!\! \!\!\!\!  \!\! 
	\eta^{-1} \big(D(\widehat q_1^{\,t},\widetilde q_1^{\,t-1}) + D(\widehat q_2^{\,t}, \widetilde q_2^{\,t-1})\big)
	\,+\,
	\eta^{-1} \big(D( \widetilde q_1^{\,t-1},\widehat q_1^{\,t}) + D( \widetilde q_2^{\,t-1}, \widehat q_2^{\,t}) \big)
	\\[0.2cm]
	&\leq&  V\, \big\langle{ (\widetilde q_1^{\,t-1} -\widehat q_1^{\,t} )\cdot \widehat q_2^{\,t-1}+\widehat q_1^{\,t-1}\cdot (\widehat q_2^{\,t} - \widetilde q_2^{\,t-1} )},{r^{t-1}}\big\rangle 
	\\[0.2cm]
	&& 
	\,+\,
	\lambda_1^{t-1} \langle{ \widetilde q_1^{\,t-1} - \widehat q_1^{\,t} },{g^{t-1}}\rangle 
	\,+\,
	\lambda_2^{t-1} \langle{ \widetilde q_2^{\,t-1} - \widehat q_2^{\,t}},{h^{t-1}}\rangle.
	\end{array}
	\end{equation}
	We note that $\langle{ (\widetilde q_1^{\,t-1} -\widehat q_1^{\,t} )\cdot \widehat q_2^{\,t-1}},{r^{t-1}}\rangle \leq \Vert (\widetilde q_1^{\,t-1} -\widehat q_1^{\,t} )\cdot \widehat q_2^{\,t-1}\Vert_1\Vert{r^{t-1}}\Vert_\infty\leq \Vert\widetilde q_1^{\,t-1} -\widehat q_1^{\,t} \Vert_1$, and, similarly, $\langle\widehat q_1^{\,t-1}\cdot (\widehat q_2^{\,t} - \widetilde q_2^{\,t-1} ),{r^{t-1}}\rangle\leq\Vert\widehat q_2^{\,t} - \widetilde q_2^{\,t-1}\Vert_1$. Thus, we can reduce~\eqref{eq.pushback_I_s} into
	\[
	\begin{array}{rcl}
	&& \!\!\!\! \!\!\!\! \!\! 
	\eta^{-1} \big(D(\widehat q_1^{\,t},\widetilde q_1^{\,t-1}) + D(\widehat q_2^{\,t}, \widetilde q_2^{\,t-1})\big)
	\,+\,
	\eta^{-1} \big(D( \widetilde q_1^{\,t-1},\widehat q_1^{\,t}) + D( \widetilde q_2^{\,t-1}, \widehat q_2^{\,t}) \big)
	\\[0.2cm]
	&\leq&
	(V+\lambda_1^{t-1}) 
	\left\|{ \widetilde q_1^{\,t-1} - \widehat q_1^{\,t} }\right\|_1
	\,+\,
	(V+\lambda_2^{t-1}) 
	\left\|{ \widetilde q_2^{\,t-1} - \widehat q_2^{\,t}}\right\|_1
	\\[0.2cm]
	&\leq&
	(V+ \left\|{\lambda^{t-1}}\right\|) 
	\left(
	\left\|{ \widetilde q_1^{\,t-1} - \widehat q_1^{\,t} }\right\|_1
	+
	\left\|{ \widetilde q_2^{\,t-1} - \widehat q_2^{\,t}}\right\|_1
	\right)
	\end{array}
	\]
	where the left-hand side can be lower bounded by Lemma~\ref{lem.D_lb}, 
	\[
	D(\widehat q_1^{\,t},\widetilde q_1^{\,t-1})+D( \widetilde q_1^{\,t-1},\widehat q_1^{\,t}) \;\geq\; L^{-1}\left\|{ \widetilde q_1^{\,t-1}-\widehat q_1^{\,t} }\right\|_1^2
	\]
	\[
	D(\widehat q_2^{\,t},\widetilde q_2^{\,t-1})+D( \widetilde q_2^{\,t-1},\widehat q_2^{\,t}) \;\geq\; L^{-1}\left\|{ \widetilde q_2^{\,t-1}-\widehat q_2^{\,t} }\right\|_1^2.
	\]
	Then, we apply the inequality $(x+y)^2\leq2(x^2+y^2) $ and cancel a non-negative term to obtain
	\begin{equation}\label{eq.qq_dual_s}
	\left\|{ \widetilde q_1^{\,t-1}-\widehat q_1^{\,t} }\right\|_1 + \left\|{ \widetilde q_2^{\,t-1}-\widehat q_2^{\,t} }\right\|_1
	\;\leq\;
	2\eta  L (V + \left\|{\lambda^{t-1}}\right\|).
	\end{equation}
	By the definition of $\widetilde q_1^{\,t-1}$ and $\widetilde q_2^{\,t-1}$,
	\[
	\begin{array}{rcl}
	\left\|{ \widetilde q_1^{\,t-1}-\widehat q_1^{\,t} }\right\|_1 
	&=&\displaystyle
	\sum_{\ell\,=\,0}^{L-1}\sum_{x\,\in\,X_\ell} \sum_{a\,\in\,A}
	\left|
		(1-\theta)\widehat {q}_1^{\,t-1}(x,a)+\theta\frac{1}{|X_\ell||A|} - \widehat q_1^{\,t}(x,a)
	\right|
	\\[0.2cm]
	&\geq&\displaystyle
	\sum_{\ell\,=\,0}^{L-1}\sum_{x\,\in\,X_\ell} \sum_{a\,\in\,A}
	\left(
		(1-\theta)  \left|
			\widehat {q}_1^{\,t-1}(x,a)- \widehat q_1^{\,t}(x,a)
		\right| -\theta \left(\frac{1}{|X_\ell||A|} + \widehat q_1^{\,t}(x,a)\right)
	\right)
	\\[0.2cm]
	&=& (1-\theta) \left\|{\widehat {q}_1^{\,t-1}-\widehat q_1^{\,t}}\right\|_1 -2\theta L.
	\end{array}
	\]
	Similarly, we have $\Vert{ \widetilde q_2^{\,t-1}-\widehat q_2^{\,t} }\Vert_1 \leq (1-\theta) \Vert{\widehat {q}_2^{\,t-1}-\widehat q_2^{\,t}}\Vert_1 -2\theta L$. Thus, we can further reduce~\eqref{eq.qq_dual_s} into
	\begin{equation}\label{eq.case(i)_s}
	\left\|{\widehat {q}_1^{\,t-1}-\widehat q_1^{\,t}}\right\|_1 +\Vert{\widehat {q}_2^{\,t-1}-\widehat q_2^{\,t}}\Vert_1 
	\;\leq\;
	2\eta (1-\theta)^{-1} L (V + \left\|{\lambda^{t-1}}\right\|) + 4 \theta (1-\theta)^{-1}  L.
	\end{equation}
	
	\noindent\textbf{Case~(ii)}. In this case, either $\widehat q_1^{\,t}$, $\widehat q_1^{\,t-1}$ or $ \widehat q_2^{\,t}$, $\widehat q_2^{\,t-1}$ might not have the same domain. For instance, when $k_1^t>k_1^{t-1}$, it is possible that $\Delta(k_1^{t})$ becomes different from $\Delta(k_1^{t-1})$. We note that $k_1^t>k_1^{t-1}$ only happens when episode $t$ is the first one that belongs to epoch $k_1^t$. By Lemma~\ref{lem.epoch}, $k_1^T \leq \sqrt{ T|X||A|}\log (8T/(|X||A|))$ and  $k_2^T \leq \sqrt{ T|Y||B|}\log (8T/(|Y||B|))$ if we are given $T\geq \max(|X||A|, |Y||B|)$.
	
	We now combine two cases above for~\eqref{eq.violation_dqq_s},
	\[
	\begin{array}{rcl}
	&& \!\!\!\! \!\!\!\! \!\!\!\!
	\displaystyle
	\sum_{t\,=\,1}^{T} \left(
		\left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 
		+\left\|{\widehat q_2^{\,t}-\widehat q_2^{\,t-1}}\right\|_1\right)
	\\[0.2cm]
	&=& \displaystyle
	\sum_{\substack{1\,\leq\,t\,\leq\,T\\k_1^t \,=\, k_1^{k-1}\,\wedge\, k_2^t \,=\, k_2^{k-1}}}\left(
		\left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 
		+\left\|{\widehat q_2^{\,t}-\widehat q_2^{\,t-1}}\right\|_1\right)
	\\[0.2cm]
	&&\displaystyle
	\,+\,
	\sum_{\substack{1\,\leq\,t\,\leq\,T\\k_1^t \,=\, k_1^{k-1}\,\vee\, k_2^t \,=\, k_2^{k-1}}}\left(
		\left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 
		+\left\|{\widehat q_2^{\,t}-\widehat q_2^{\,t-1}}\right\|_1\right)
	\\[0.2cm]
	&\leq& \displaystyle
	\sum_{\substack{1\,\leq\,t\,\leq\,T\\k_1^t \,=\, k_1^{k-1}\,\wedge\, k_2^t \,=\, k_2^{k-1}}}\left(
		\left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 
		+\left\|{\widehat q_2^{\,t}-\widehat q_2^{\,t-1}}\right\|_1\right)
	          
	                
	\,+\,2L (k_1^T+k_2^T)
	\\[0.2cm]
	&\leq&\displaystyle
	2\eta (1-\theta)^{-1} L\sum_{t\,=\,1}^{T} (V + \left\|{\lambda^{t-1}}\right\|) + 4 \theta (1-\theta)^{-1}  LT +2L (k_1^T+k_2^T)
	\end{array}
	\]
	where the first inequality is due to: $\left\|{\widehat q_1^{\,t} - \widehat q_1^{\,t-1}}\right\|_1 \leq 2L$ and $\left\|{\widehat q_2^{\,t}-\widehat q_2^{\,t-1}}\right\|_1\leq 2L$, and we apply~\eqref{eq.case(i)_s} from the case (i) for the last inequality. Using the bounds on $k_1^T$, $k_2^T$ in the case (ii), we conclude the desired bound for~\eqref{eq.violation_dqq_s},
	\[
	\begin{array}{rcl}
	&& \!\!\!\! \!\!\!\! \!\!
	\displaystyle {\widehat{\text{\normalfont Violation}_1}(T)}, {\widehat{\text{\normalfont Violation}_2}(T)}
	\\[0.2cm]
	&\leq&\displaystyle
	\left\|{\lambda^T}\right\|
	\,+\, 
	\frac{2\eta L}{1-\theta}
	\sum_{t\,=\,1}^{T} \left\|{\lambda^{t-1}}\right\| + \frac{2\eta V+4 \theta}{1-\theta}  LT 
	\\[0.2cm]
	&&\displaystyle+2L 
	\left(\sqrt{ T|X||A|}\log (8T/(|X||A|))+ \sqrt{ T|Y||B|}\log (8T/(|Y||B|))\right).
	\end{array}
	\]
	We complete the proof by noting $\lambda_1^0=\lambda_2^0=0$, $V=L\sqrt{T}$, $\eta = 1/(TL)$, and $\theta = 1/T$.

\citep{yu2017online}\label{lem.drift}           
	Let $\{ Z^t, t\geq0 \}$ be a discrete-time stochastic process that is adapted to a filtration $\{{\mathcal{F}}^t, t\geq 0 \}$ with $Z^0 =0$ and ${\mathcal{F}}^0 = \{\emptyset,\Omega\}$. Assume that there exists $t_0 \in \mathbb{Z}^+$, $\Theta\in \mathbb{R}^+$, $ \delta_{\max}\in\mathbb{R}^+$, and $\zeta\in (0, \delta_{\max}]$ such that for all $t\geq 1$,
	\[
	\left| Z^{t+1}-Z^t \right| \;\leq\; \delta_{\max}
	\;\text{ and }\;
	\mathbb{E} \left[ Z^{t+t_0}-Z^t  \,\vert\, {\mathcal{F}}^t\right]
	\;\leq\;
	\begin{cases}
	t_0\, \delta_{\max} \;\; \text{ when } Z^t \leq \Theta
	\\[0.2cm]
	-\,t_0\,\zeta \;\;\;\; \text{ otherwise } Z^t \geq \Theta.
	\end{cases}
	\]
	Then, with probability $1-\delta$ it holds for any $t$ that 
	\[
	Z^t \;\leq \; \Theta + t_0 \delta_{\max} + t_0 \frac{4\delta_{\max}^2}{\zeta} \log \left(\frac{8\delta_{\max}^2}{\zeta}\right) + t_0 \frac{4\delta_{\max}^2}{\zeta} \log\frac{1}{\delta}.
	\]
