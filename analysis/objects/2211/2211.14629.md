\label{thm:1}
Suppose that the N-seasonal discrete-time risk model \eqref{model} is generated by random variables $X_1,\, X_2,\, \ldots,\, X_N$ and the net profit condition $\mathbb{E} S_N<\kappa N$ is satisfied. Then the following statements are correct:
\begin{enumerate}
\item
The probability mass functions of random variables $\mathcal{M}_1,\,\mathcal{M}_2,\, \ldots,\, \mathcal{M}_N$ and $X_1,\,X_2,\, \ldots,\, X_N$ are related by
\begin{align}\label{eq:thm_eq_mean}
\nonumber
&\hspace{-1cm}\sum_{i=0}^{\kappa-1}m_i^{(1)}\sum_{j=0}^{\kappa-1-i}x^{(N)}_j(\kappa-i-j)+\sum_{i=0}^{\kappa-1}m_i^{(2)}\sum_{j=0}^{\kappa-1-i}x^{(1)}_j(\kappa-i-j)

&\hspace{-1cm}+\sum_{i=0}^{\kappa-1}m_i^{(N)}\sum_{j=0}^{\kappa-1-i}x^{(N-1)}_j(\kappa-i-j)
=\kappa N -\mathbb{E}S_N.
\end{align}
\item
If $0<|s|\leqslant1$, then
\begin{align}\label{eq:thm_zero_s}
\nonumber
&\hspace{-1cm}\sum_{i=0}^{\kappa-1}m_i^{(1)}\sum_{j=0}^{\kappa-1-i}x^{(N)}_j(s^\kappa-s^{i+j})
+\frac{G_{X_N}(s)}{s^\kappa}\sum_{i=0}^{\kappa-1}m_i^{(2)}\sum_{j=0}^{\kappa-1-i}x^{(1)}_j(s^\kappa-s^{i+j})\\ \nonumber
&\hspace{-1cm}+\frac{G_{X_N+X_1}(s)}{s^{2\kappa}}
\sum_{i=0}^{\kappa-1}m_i^{(3)}\sum_{j=0}^{\kappa-1-i}x^{(2)}_j(s^\kappa-s^{i+j})+\ldots\\

\end{align}

\end{align}

<!-- proof-1: body_md -->
We prove the case $j=1$ only and note that the other cases can be proven similarly.
	
	According to the law of large numbers, almost surely
	\begin{align*}
	    \frac{1}{n}\sum_{i=1}^{n}\left(X_{i}-\kappa\right)&=\frac{1}{N}\left(\frac{N}{n}\sum_{\substack{{i=1}\\ i\equiv 1\,{\rm mod}\,N}}^{n}\left(X_{i}-\kappa\right) +\ldots+\frac{N}{n}\sum_{\substack{{i=1}\\ i\equiv N\,{\rm mod}\,N}}^{n}\left(X_{i}-\kappa\right)\right)\\
	    &\overset{n \to \infty}{\longrightarrow} \frac{1}{N}\left((\mathbb{E}X_1-\kappa)+\ldots+(\mathbb{E}X_N-\kappa)\right)
	    =\frac{\mathbb{E}S_N-\kappa N}{N}=:-\mu<0.
	\end{align*}
	Therefore, 	\begin{align*}
	\lim_{n\to\infty}\mathbb{P}\left(\sup_{j\geqslant n}\left|\frac{1}{j}\sum_{i=1}^{j}\left(X_i-\kappa\right)+\mu\right| < \frac{\mu}{2}\right) = 1.
	\end{align*}
	Consequently, for any arbitrarily small $\varepsilon>0$, there exists such number $N_\varepsilon\in\mathbb{N}$ that
	\begin{align*}
	    &\mathbb{P}\left(\bigcap_{j=n}^{\infty}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)\leqslant
	    0\right\}\right)	\geqslant
	    \mathbb{P}\left(\bigcap_{j=n}^{\infty}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)<
	    -\frac{\mu}{2}\right\}\right)\\
	    &\geqslant\mathbb{P}\left(\bigcap_{j= n}^{\infty}\left\{\left|\frac{1}{j}\sum_{i=1}^{j}\left(X_i-\kappa\right)+\mu\right| < \frac{\mu}{2}\right\}\right)
	    = \mathbb{P}\left(\sup_{j\geqslant n}\left|\frac{1}{j}\sum_{i=1}^{j}\left(X_i-\kappa\right)+\mu\right| < \frac{\mu}{2}\right) \geqslant 1-\varepsilon
	\end{align*}
 for all $n \geqslant N_\varepsilon$.
 
	It follows that for any such $\varepsilon$ and any $u\in\mathbb{N}$ we have
	\begin{align*}
	    \mathbb{P}(\mathcal{M}_1< u)&=\mathbb{P}\left(\bigcap_{j=1}^{\infty}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)<u\right\}\right)\\
	    &\geqslant\mathbb{P}\left(\left\{\bigcap_{j=1}^{N_\varepsilon-1}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)<u\right\}\right\}\cap\left\{\bigcap_{j=N_\varepsilon}^{\infty}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)\leqslant0\right\}\right\}\right)\\
	    &\geqslant\mathbb{P}\left(\bigcap_{j=1}^{N_\varepsilon-1}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)<u\right\}\right)
     +\mathbb{P}\left(\bigcap_{j=N_\varepsilon}^{\infty}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)\leqslant0\right\}\right)-1\\
	    &\geqslant\mathbb{P}\left(\bigcap_{j=1}^{N_\varepsilon-1}\left\{\sum_{i=1}^{j}\left(X_i-\kappa\right)<u\right\}\right)-\varepsilon.
	\end{align*}
	The last inequality implies
	\begin{equation*}
	    \lim_{u \to \infty}\mathbb{P}(\mathcal{M}_1<u)\geqslant 1-\varepsilon,
	\end{equation*}
	where $\varepsilon>0$ is as small as we want and the assertion of Lemma follows.

\label{lem:generating_f_NxN}
Let $s\in\mathbb{C}$ be the complex number and $0<|s|\leqslant1$. Then the probability-generating functions of $X_1,\,X_2,\,\ldots,\,X_N$ and $\mathcal{M}_1,\,\mathcal{M}_2\ldots,\,\mathcal{M}_N$ are related the following way
\begin{align}\label{syst:NxN}
    \begin{cases}
    \sum_{i=0}^{\kappa-1}m_i^{(1)}\sum_{j=0}^{\kappa-1-i}x^{(N)}_j(s^\kappa-s^{i+j})&=s^\kappa G_{\mathcal{M}_N}(s)-G_{X_N}(s)G_{\mathcal{M}_1}(s)\\
    \sum_{i=0}^{\kappa-1}m_i^{(2)}\sum_{j=0}^{\kappa-1-i}x^{(1)}_j(s^\kappa-s^{i+j})&=s^\kappa G_{\mathcal{M}_1}(s)-G_{X_1}(s)G_{\mathcal{M}_2}(s)\\
    \sum_{i=0}^{\kappa-1}m_i^{(3)}\sum_{j=0}^{\kappa-1-i}x^{(2)}_j(s^\kappa-s^{i+j})&=s^\kappa G_{\mathcal{M}_2}(s)-G_{X_2}(s)G_{\mathcal{M}_3}(s)\\
    &\,\,\vdots\\
    \sum_{i=0}^{\kappa-1}m_i^{(N)}\sum_{j=0}^{\kappa-1-i}x^{(N-1)}_j(s^\kappa-s^{i+j})&=s^\kappa G_{\mathcal{M}_{N-1}}(s)-G_{X_{N-1}}(s)G_{\mathcal{M}_N}(s)
    \end{cases}.
\end{align}
<!-- end lem-3 -->
<!-- proof-3: body_md -->
Let us demonstrate how the first equality in \eqref{syst:NxN} is derived and note that the remaining ones follow the same logic.

By Lemma \ref{lem:dist_NxN}, the distributions' equality $(\mathcal{M}_1+X_{N}-\kappa)^+\buildrel d \over = \mathcal{M}_{N}$ implies the equality of probability-generating functions $G_{\mathcal{M}_N}(s)=G_{(\mathcal{M}_1+X_{N}-\kappa)^+}(s)$. Then, applying the law of total expectation for the last equality, we obtain
\begin{align*}
G_{\mathcal{M}_N}(s)
&=\mathbb{E} s^{(\mathcal{M}_1+X_{N}-\kappa)^+}=\mathbb{E} \left( \mathbb{E} \left( s^{(\mathcal{M}_1+X_{N}-\kappa)^+}|\mathcal{M}_1\right) \right)
=\sum_{i=0}^{\infty}m^{(1)}_i\mathbb{E} s^{(X_N-\kappa+i)^+}\\
&=\sum_{i=0}^{\kappa-1}m^{(1)}_i\mathbb{E}s^{(X_N-\kappa+i)^+}
+G_{X_N}(s)s^{-\kappa}\sum_{i=\kappa}^{\infty}m^{(1)}_is^i\\
&=\sum_{i=0}^{\kappa-1}m^{(1)}_i\left(\mathbb{E}s^{\left(X_N-\kappa+i\right)^+}
-s^{i-\kappa}G_{X_N}(s)\right)+s^{-\kappa}G_{X_N}(s)G_{\mathcal{M}_1}(s).
\end{align*}
Multiplying both sides of the last equality by $s^\kappa$ when $s\neq0$ and observing that
\begin{align*}
\sum_{i=0}^{\kappa-1}m^{(1)}_i\left(s^\kappa\mathbb{E} s^{(X_N-\kappa+i)^+}-s^iG_{X_N}(s)\right)=\sum_{i=0}^{\kappa-1}m^{(1)}_i
\sum_{j=0}^{\kappa-1-i}x_j^{(N)}(s^\kappa-s^{i+j})
\end{align*}
we get the desired result.

<!-- lem-4: body_md -->
\label{lem:roots}
Assume that the net profit condition $G'_{S_N}(1)=\mathbb{E} S_N<\kappa N$ is valid. Then there are exactly $\kappa N-1$ roots, counted with their multiplicities, of $s^{\kappa N}=G_{S_N}(s)$ in $|s|\leqslant1$, $s\neq1$.
<!-- end lem-4 -->
<!-- proof-4: body_md -->
We follow the proof of \cite[Lemma 9]{GJ}. Due to the estimate 
$$
|G_{S_N}(s)|\leqslant1<\lambda |s|^{\kappa N}
$$
on $|s|=1$ when $\lambda>1$, Rouché's theorem implies that both functions $G_{S_N}(s)-\lambda s^{\kappa N}$ and $\lambda s^{\kappa N}$ have the same number of roots in $|s|<1$ and this number is $\kappa N$ due to the fundamental theorem of algebra.  When $\lambda\to1^+$ some roots of $G_{S_N}(s)-\lambda s^{\kappa N}$ remain in $|s|<1$ and some migrate to the boundary points $|s|=1$. Obviously, $s=1$ is always the root of $s^{\kappa N}=G_{S_N}(s)$ and it is the simple root because the net profit condition holds, i. e. 

$$
\left(G_{S_N}(s)-s^{\kappa N}\right)'\Big|_{s=1}=\mathbb{E} S_N-\kappa N<0.
$$ 

Thus, there remain $\kappa N-1$ roots of $s^{\kappa N}=G_{S_N}(s)$ in $|s|\leqslant1$, $s\neq1$ and additionally one can say that $|s|=1$ , $s \neq 1$ is the root of $s^{\kappa N}=G_{S_N}(s)$ if the greatest common divisor of $\kappa N$ and the powers of $s$ in $G_{S_N}(s)$ is greater than one.

i_1+i_2+\ldots+i_N\leqslant u+\kappa N-1}}\hspace{-0mm}x^{(1)}_{i_1}x^{(2)}_{i_2}\cdots x^{(N)}_{i_N}\,
\varphi\Bigg(u+\kappa N-\sum_{j=1}^Ni_j\Bigg)\\ 
&=\sum_{i=1}^{u+\kappa N}s^{(N)}_{u+\kappa N -i}\varphi(i) - \sum_{i=1}^{\kappa(N-1)} \mu_i(u)\varphi(i),
\end{align}
where $\mu_i(u)$ for each $i\in\{1,\,2,\, \ldots,\, \kappa (N-1)\}$ are coefficients consisting of the products of probability mass functions of random variables $X_1, X_2, \ldots, X_N$. For instance, if $N=2$ and $\kappa=1$, then
$$
\varphi(u)=\sum_{\substack{i_1\leqslant u\\i_1+i_2\leqslant u+1}}x^{(1)}_{i_1}x^{(2)}_{i_2}\varphi(u+2-i_1-i_2)=\sum_{i=1}^{u+2}s^{(2)}_{u+2-i}\varphi(i)-x^{(1)}_{u+1}x^{(2)}_0\varphi(1).
$$
If $\mu_0(u):=s^{(N)}_{u+\kappa N}$ and $\mu_j(u):=0$ when $j>\kappa (N-1)$, then the equality in \eqref{sum_supl} is
    \begin{equation*}
        \varphi(u)=\sum_{i=0}^{u+\kappa N}s^{(N)}_{u+\kappa N -i}\varphi(i) - \sum_{i=0}^{\kappa N-1} \mu_i(u)\varphi(i).
    \end{equation*}
    By summing up both sides of the last equality by $u$, which varies from 0 to some natural and sufficiently large $v$, we obtain
    \begin{equation}\label{eq:sum}
        \sum_{u=0}^v\varphi(u)=\sum_{u=0}^v\sum_{i=0}^{u+\kappa N}s^{(N)}_{u+\kappa N -i}\varphi(i) - \sum_{u=0}^v\sum_{i=0}^{\kappa N-1} \mu_i(u)\varphi(i).
    \end{equation}
    We now change the order of summation in \eqref{eq:sum}
    \begin{equation*}
        \sum_{u=0}^v\sum_{i=0}^{u+\kappa N}(\cdot)=\sum_{i=0}^{\kappa N-1}\sum_{u=0}^v(\cdot)+\sum_{i=\kappa N}^{v+\kappa N}\sum_{u=i-\kappa N}^v(\cdot)
    \end{equation*}
    and obtain
    \begin{align*}
        \sum_{u=0}^{v+\kappa N}\varphi(u)-\sum_{u=v+1}^{v+\kappa N}\varphi(u)
        =\sum_{i=0}^{\kappa N-1}\varphi(i)\sum_{u=0}^v s^{(N)}_{u+\kappa N -i}+\sum_{i=\kappa N}^{v+\kappa N}\varphi(i)\sum_{u=i-\kappa N}^v s^{(N)}_{u+\kappa N -i} - \sum_{i=0}^{\kappa N-1}\varphi(i)\sum_{u=0}^v \mu_i(u).
    \end{align*}
    Subtracting $\sum_{i=0}^{v+\kappa N}\varphi(i) \sum_{u=i-\kappa N}^v s_{u+\kappa N-i}^{(N)}$ from both sides of the last equation and rearranging, we get
    \begin{align*}
        &\sum_{i=0}^{v+\kappa N}\varphi(i)\left(1 -\sum_{u=i-\kappa N}^v s_{u+\kappa N-i}^{(N)}\right)-\sum_{i=v+1}^{v+\kappa N}\varphi(i) \\
        &=\sum_{i=0}^{\kappa N-1}\varphi(i)\left(\sum_{u=0}^v s^{(N)}_{u+\kappa N -i}-\sum_{u=0}^v \mu_i(u)\right)
        -\sum_{i=0}^{\kappa N-1}\varphi(i) \sum_{u=i-\kappa N}^v s_{u+\kappa N-i}^{(N)}
    \end{align*}
                                                     
                   
                                                                                                                
                                                                                                                            
                 
    or
    \begin{align*}
        \sum_{i=v+1}^{v+\kappa N}\varphi(i)-\sum_{i=0}^{v+\kappa N}\varphi(i)\left(1 -\sum_{u=0}^{v+\kappa N-i} s_u^{(N)}\right)
        =\sum_{i=0}^{\kappa N-1}\varphi(i)\left(\sum_{u=0}^{\kappa N -i-1} s^{(N)}_{u}+\sum_{u=0}^{v} \mu_i(u)\right),
    \end{align*}
    which implies
    \begin{align}\label{eq:v_to_inf}\hspace{-0.5cm}
    \sum_{i=v+1}^{v+\kappa N}\varphi(i)-\sum_{i=0}^{v+\kappa N}\varphi(i)\mathbb{P}(S_N>v+\kappa N - i)=\sum_{i=0}^{\kappa N-1}\varphi(i)\left(\mathbb{P}(S_N\leqslant \kappa N-i-1)+\sum_{u=0}^{v} \mu_i(u)\right).
    \end{align}
    Clearly, the definition of the survival probability \eqref{ult_time_p} implies that $\varphi(u)$ is a non-decreasing function, i. e. $\varphi(u)\leqslant\varphi(u+1)$ for all $u\in\mathbb{N}_0$. Thus, there exists a non-negative limit $\varphi(\infty):=\lim_{u\to\infty}\varphi(u)$ and $\varphi(\infty)=1$ if the net profit condition $\mathbb{E} S_N<\kappa N$ holds, see Lemma \ref{lem:lim}. We now let $v\to\infty$ in both sides of \eqref{eq:v_to_inf}. For the first sum in \eqref{eq:v_to_inf} we obtain
    \begin{equation}\label{2lim1}
        \lim_{v\to\infty}\sum_{i=v+1}^{v+\kappa N}\varphi(i)=\lim_{v\to\infty}\left(\varphi(v+1)+\ldots+\varphi(v+\kappa N)\right)=\varphi(\infty) \kappa N,
    \end{equation}
 and for the second
    \begin{align}\label{eq:the_second}
\lim_{v\to\infty}\sum_{i=0}^{v+\kappa N}\varphi(i)\mathbb{P}(S_N>v+\kappa N - i)=\varphi(\infty)\mathbb{E} S_N.
    \end{align}
   Indeed, let us recall that $\mathbb{E} X=\sum_{i=0}^\infty\mathbb{P}(X>i)$, when $X$ is some non-negative and integer-valued random variable. Then, the upper bound of \eqref{eq:the_second} is
    \begin{align*}
        &\lim_{v\to\infty}\sum_{i=0}^{v+\kappa N}\varphi(i)\mathbb{P}(S_N>v+\kappa N - i)\leqslant \lim_{v\to\infty}
        \varphi(v+\kappa N)\sum_{i=0}^{v+\kappa N}\mathbb{P}(S_N>v+\kappa N-i)\\
        &=\lim_{v\to\infty}\varphi(v+\kappa N)\sum_{i=0}^{v+\kappa N}\mathbb{P}(S_N>i)=\varphi(\infty)\mathbb{E} S_N,
    \end{align*}
   while the lower bound is the same due to inequality
    \begin{align*}
        &\sum_{i=0}^{v+\kappa N}\varphi(i)\mathbb{P}(S_N>v+\kappa N - i)\\
        &=\sum_{j=0}^M\varphi(i)\mathbb{P}(S_N>v+\kappa N - i)
        +\sum_{i=M+1}^{v+\kappa N}\varphi(i)\mathbb{P}(S_N>v+\kappa N - i)\\
        &\geqslant \inf_{i\geqslant M+1}\varphi(i)\sum_{i=0}^{v+\kappa N-M-1}\mathbb{P}(S_N>i),
    \end{align*}
where $M$ is some fixed and sufficiently large natural number. Thus, when $v\to\infty$, the equality in \eqref{eq:v_to_inf} is

\begin{align}\label{eq:after_limit}
\varphi(\infty)(\kappa N-\mathbb{E} S_N)=\sum_{i=0}^{\kappa N-1}\varphi(i)\left(\mathbb{P}(S_N\leqslant \kappa N-i-1)+\sum_{u=0}^{\infty} \mu_i(u)\right).
\end{align}
If $\mathbb{E} S_N>\kappa N$, as under the case of consideration, the non-negative right-hand-side of \eqref{eq:after_limit} implies $\varphi(\infty)=0$ and consequently $\varphi(u)=0$ for all $u\in\mathbb{N}_0$. The first case that $\mathbb{E} S_N>\kappa N$ makes survival impossible is proved.
        
	Let us now consider the case when $\mathbb{E} S_N = \kappa N$ and $\mathbb{P}(S_N=\kappa N)<1$. If $\mathbb{E} S_N = \kappa N$ and $\mathbb{P}(S_N=\kappa N)<1$, then at least one probability $s_0^{(N)},\, s_1^{(N)},\, \ldots,\, s_{\kappa N-1}^{(N)}$ is larger than zero, because otherwise $\mathbb{E} S_N>\kappa N$.
	Then, from \eqref{eq:after_limit},
    \begin{equation}\label{eq:3.2.2}
        \sum_{i=0}^{\kappa N-1}\varphi(i)\left(\sum_{u=0}^{\kappa N-i-1}s_u^{(N)}+\sum_{u=0}^{\infty} \mu_i(u)\right)=0.
    \end{equation}
    If $s_0^{(N)}>0$, then \eqref{eq:3.2.2} implies $\varphi(0)=\varphi(1)=\ldots=\varphi(\kappa N-1)=0$. Using recurrence \eqref{eq:rec} we can show that $\varphi(u)=0$ for all $u \in \mathbb{N}_0$.
    If $s_0^{(N)}=0$ and $s_1^{(N)}>0$, then \eqref{eq:3.2.2} implies that $\varphi(0)=\varphi(1)=\ldots=\varphi(\kappa N-2)=0$ and once again, using recurrence \eqref{eq:rec} we can show that $\varphi(u)=0$ for all $u \in \mathbb{N}_0$.
    Arguing the same we proceed up to $s_0^{(N)}=s_1^{(N)}=\ldots= s_{\kappa N-2}^{(N)}=0,\, s_{\kappa N-1}^{(N)}>0$ and observe that in all of these cases \eqref{eq:3.2.2} and recurrence \eqref{eq:rec} yields $\varphi(u)=0$ for all $u \in \mathbb{N}_0$.
    
	Finally, let us consider the case when $\mathbb{P}(S_N=\kappa N)=1$. If $S_N=\kappa N$ with probability one, the random variables $X_1,\, X_2,\, \ldots,\, X_N$ are degenerate, meaning that $X_i \equiv c_i$ for all $i \in \{1,\, 2,\, \ldots,\, N\}$, where $c_i \in \{0,\, 1,\,\ldots,\, \kappa N\}$ and $c_1+c_2+\ldots+c_N=\kappa N$. Thus, the model \eqref{model} becomes fully deterministic. Moreover, $W(n)=W(n+N)$ for all $n \in \mathbb{N}_0$, $N\in\mathbb{N}$, and it is sufficient to check if the lowest of values among $W(1),\,\ldots,\,W(N)$ is larger than zero.
