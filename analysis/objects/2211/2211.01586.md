<!-- definition-1: body_md -->
For any $\overline{\varepsilon}, \hbar \in {\mathbb{C}}$ and $\partial_w = \frac{\partial}{\partial w}$, the Nazarov--Sklyanin Lax operator ${\mathcal{L}}$ \textnormal{\cite{NaSk2}} is the graded linear operator ${\mathcal{L}} \colon F[w] \rightarrow F[w]$ in the graded polynomial ring $F[w]$ in {(\ref{HDefinition})} defined by
\begin{equation} \label{PresentationOfL} {\mathcal{L}} = \overline{\varepsilon} w \partial_w + \sum_{k=1}^{\infty} V_{-k} w^k + \sum_{k=1}^{\infty} V_k\pi w^{-k} \end{equation}
for $V_{-k}$ in {(\ref{VNegativeKDefinition})}, $\pi$ in {(\ref{SzegoProjectionFormula})}. Below, we may write ${\mathcal{L}} = {\mathcal{L}}_{\overline{\varepsilon},\hbar}$ to emphasize that $V_{-k}$ depends on $\hbar$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
For $\lambda$ in {(\ref{DefinitionDLambda})}, define the addable and removable corner sets by
 \begin{gather} \label{AddSetFormula}
 \mathcal{A}_{\lambda} = \big\{ s \in {\mathbb{N}}^2 \colon s \not\in \lambda \ \textnormal{and} \ \lambda \cup \{s\} \ \textnormal{is also a Young diagram} \big\}, \\
\label{RemoveSetFormula} \mathcal{R}_{\lambda} = \big\{ s \in {\mathbb{N}}^2 \colon s \in \lambda \ \textnormal{and} \ \lambda \setminus \{s\} \ \textnormal{is also a Young diagram}\big\}. \end{gather}
It is also convenient to define the outer corner set $\mathcal{R}_{\lambda}^+ = \mathcal{R}_{\lambda} + (1,1)$ as shifts of removable corners.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
For ${\varepsilon}_1, {\varepsilon}_2 \in {\mathbb{C}}$, the anisotropic content $[s]$ of $s = (N_1, N_2) \in {\mathbb{Z}}^2$ is defined by \begin{equation} \label{ContentDefinition} [s] = {\varepsilon}_1 N_1 + {\varepsilon}_2 N_2. \end{equation}
<!-- end definition-3 -->
<!-- theorem-1: body_md -->
\label{TheoremMAINRESULT} Let ${\mathcal{L}}$ be the Nazarov--Sklyanin Lax operator in~\eqref{PresentationOfL}. Assume that $\overline{\varepsilon} \in {\mathbb{R}}$ and $\hbar>0$. Choose {non-zero} ${\varepsilon}_1, {\varepsilon}_2 \in {\mathbb{C}}$ parametrizing $\overline{\varepsilon}$, $\hbar$ by the formulas~\eqref{EBarFormula} and~\eqref{HBarFormula}.
\begin{enumerate}\itemsep=0pt
\item[$(1)$] The degree $n$ component $F[w]_n$ of the graded ring $F[w]$ in~\eqref{HDefinition} has a cyclic decomposition 
\begin{equation} \label{TheoremStatementDirectSum}
F[w]_n {\ = \ } \bigoplus_{| \lambda|=n} Z(j_{\lambda}, {\mathcal{L}})
\end{equation}
as a direct sum of finite-dimensional ${\mathcal{L}}$-cyclic subspaces $Z(j_{\lambda},{\mathcal{L}})$ indexed by parti\-tions $\lambda$ of size $n$ and generated by Jack polynomials $j_{\lambda} \in F$ defined in~\eqref{JackPolynomialPresentation}.
\item[$(2)$] The restrictions of ${\mathcal{L}}$ to the subspaces $Z_{\lambda} = Z(j_{\lambda}, {\mathcal{L}})$ in~\eqref{TheoremStatementDirectSum} all have simple spectrum
\begin{equation}
\label{TheoremStatementSimpleSpectrum} {\textnormal{spec}} \big ({\mathcal{L}} \big|_{Z_{\lambda}} \big ) = \big \{ [s] \colon s \in \mathcal{A}_{\lambda}\big \}
\end{equation}
given by the {anisotropic} contents~\eqref{ContentDefinition} of the addable corner set $\mathcal{A}_{\lambda}$ defined above in~\eqref{AddSetFormula}. {In particular, each cyclic subspace $Z_{\lambda}$ in~\eqref{TheoremStatementDirectSum} has dimension $\dim Z_{\lambda} = | \mathcal{A}_{\lambda}|$.}
\item[$(3)$] {As a distinguished vector space basis $\psi_{\lambda}^s$ for} the cyclic spaces $Z(j_{\lambda}, {\mathcal{L}})$ in~\eqref{TheoremStatementDirectSum}, one can choose the unique polynomials $\psi_{\lambda}^s \in F[w]$ {in the variables $w, V_1, V_2, \ldots$} indexed {by addable corners} $s \in \mathcal{A}_{\lambda}$ which are both eigenfunctions of ${\mathcal{L}}$ with eigenvalues $[s]$ as in~\eqref{TheoremStatementSimpleSpectrum}, namely
\[
{\mathcal{L}} \psi_{\lambda}^s = [s] \psi_{\lambda}^s,
\]
{and which project to the Jack polynomial $j_{\lambda}$ for all $s \in \mathcal{A}_{\lambda}$ upon setting $w=0$, namely}
\begin{equation} \label{PsiNormalizationConditionTheoremStatement}
\pi_0 \psi_{\lambda}^s = j_{\lambda},
\end{equation}
 where $\pi_0\colon F[w] \rightarrow F$ is evaluation at $w=0$. In this normalization,
 \begin{equation} \label{JackSuperpositionTheoremStatement}
 j_{\lambda} = \sum_{s \in \mathcal{A}_{\lambda}} \tau_{\lambda}^s \psi_{\lambda}^s,
 \end{equation}
 where $\tau_{\lambda}^s>0$ are weights of the anisotropic transition measure \textnormal{\cite{Ke4}} defined explicitly by \begin{equation} \label{TauPresentation} \tau_{\lambda}^s = \frac{ \prod\limits_{r \in \mathcal{R}^+_{\lambda}} \ \ [s-r] }{ \prod\limits_{ t \in \mathcal{A}_{\lambda} \setminus \{s\}} [s - t] \ \ } \end{equation} for $\mathcal{R}_{\lambda}$ the removable corner set~\eqref{RemoveSetFormula} and $\mathcal{R}_{\lambda}^+ = \mathcal{R}_{\lambda}+(1,1)$ the set of outer corners.
\end{enumerate}
<!-- end theorem-1 -->
<!-- definition-4: body_md -->
The Nazarov--Sklyanin hierarchy is the set of operators $\mathcal{T}_{\ell} = \pi_0 {\mathcal{L}}^{\ell} \colon F\rightarrow F$.
<!-- end definition-4 -->
<!-- theorem-2: body_md -->
\label{NSTheorem}
Choose ${\varepsilon}_1, {\varepsilon}_2 \in {\mathbb{C}}$ which parametrize $\overline{\varepsilon} \in {\mathbb{R}}$ and $\hbar >0$ by the formulas~\eqref{EBarFormula} and~\eqref{HBarFormula}. For $u \in {\mathbb{C}} \setminus {\mathbb{R}}$, consider the bounded transfer operators
\begin{equation}\label{NSTransferOperatorsPresentationIntro} \mathcal{T}(u) = \pi_0 (u - {\mathcal{L}})^{-1} \colon \ F \rightarrow F, \end{equation}
 which encode all unbounded operators in the hierarchy $\mathcal{T}_{\ell} = \pi_0 {\mathcal{L}}^{\ell}$ as coefficients of $u^{-\ell-1}$.
\begin{itemize}\itemsep=0pt
\item[$(1^{\vee})$] The transfer operators~\eqref{NSTransferOperatorsPresentationIntro} commute in $F$ for distinct values of $u$ and are simultaneously diagonalized on Jack polynomials $j_{\lambda}$ in~\eqref{JackPolynomialPresentation} with eigenvalues $T_{\lambda}(u) \in {\mathbb{C}}$ as in \begin{equation} \label{JacksFromNS} \mathcal{T}(u) \ j_{\lambda} = T_{\lambda}(u) \ j_{\lambda} .\end{equation}
\item[$(2^{\vee})$] The eigenvalues $T_{\lambda}(u) \in {\mathbb{C}}$ in~\eqref{JacksFromNS} are determined by the {anisotropic} contents~\eqref{ContentDefinition} of the elements of the Young diagram $\lambda$ in~\eqref{DefinitionDLambda} and their shifts by the product formula
\begin{equation*}                                  
T_{\lambda}(u)= u^{-1} \cdot \prod_{s \in \lambda}\frac{(u-[s+(0,0)])(u-[s+(1,1)])}{(u-[s+(1,0)])(u-[s+(0,1)])}.
\end{equation*}

Equivalently, if $\mathcal{A}_{\lambda}$ and $\mathcal{R}_{\lambda}$ are the addable and removable corner sets in~\eqref{AddSetFormula} and~\eqref{RemoveSetFormula}, 
\begin{equation}
\label{TEigenvalueAddRemoveStyle} T_{\lambda}(u) = \frac{\prod\limits_{r \in \mathcal{R}^+_{\lambda}} (u - [r]) }{ \prod\limits_{s \in \mathcal{A}_{\lambda} } (u - [s])} , \end{equation}
 where $\mathcal{R}_{\lambda}^+ = \mathcal{R}_{\lambda}+(1,1)$ is the set of outer corners.
 \end{itemize}
<!-- end theorem-2 -->
<!-- definition-5: body_md -->
The Jack--Lax cyclic space $Z_{\lambda}$ is the ${\mathcal{L}}$-cyclic space generated by $j_{\lambda}$
\begin{equation} \label{JackLaxIntroductionInProof} Z_{\lambda} = Z(j_{\lambda}, {\mathcal{L}}), \end{equation} i.e., the subspace of $F[w]$ spanned by $\big\{ j_{\lambda}, {\mathcal{L}} j_{\lambda}, {\mathcal{L}}^2 j_{\lambda}, {\mathcal{L}}^3 j_{\lambda}, \ldots \big\}.$
<!-- end definition-5 -->
<!-- lemma-1: body_md -->
\label{JackLaxSpacesAreFiniteDim}The graded components $F[w]_n$ of $F[w]$ have dimension
\begin{equation} \label{HComponentDimensionFormula}
\dim F[w]_n= \sum_{l=0}^n \mathsf{p}(l),
\end{equation}
where $\mathsf{p}(l)$ is the number of partitions of $l$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Write $\mu = \big(1^{d_1} 2^{d_2} \cdots\big)$ to denote a partition $\mu$ of size $| \mu | = \sum_{k=1}^{\infty} k d_k$ with $d_k$ parts of size $k$. For such $\mu$, let $V_{\mu} = V_1^{d_1} V_2^{d_2} \cdots$. The ring $F$ in~(\ref{H0Definition}) has graded components $F_n$ with $\dim F_n = \mathsf{p}(n)$ since $\{V_{\mu} \colon | \mu|=n\}$ is a basis of $F_n$. Similarly, $F[w]$ in~(\ref{HDefinition}) has graded components $F[w]_n$ with basis $\big\{V_{\mu} w^l \colon | \mu | + l = n\big\}$ so~(\ref{HComponentDimensionFormula}) holds.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
For any partition $\lambda$, the projection of the Jack--Lax cyclic subspace in~\eqref{JackLaxIntroductionInProof} to $F \subset F[w]$ is one-dimensional and spanned by the Jack polynomial $j_{\lambda}$:
\begin{equation} \label{ProjectionJackLax} \pi_0 Z(j_{\lambda}, {\mathcal{L}})= {\mathbb{C}} j_{\lambda}.\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
For any $\eta \in Z(j_{\lambda}, {\mathcal{L}})$, there is a polynomial $P$ so $\eta = P({\mathcal{L}}) j_{\lambda}$. As a consequence,~$\pi_0 \eta$~is a finite linear combination of $\pi_0 {\mathcal{L}}^{\ell} j_{\lambda}$ indexed by $\ell \geq 0$. Expanding the resolvent $(u - {\mathcal{L}})^{-1}= \sum_{\ell=0}^{\infty} u^{-\ell-1} {\mathcal{L}}^{\ell}$, part $(1^{\vee})$ of Theorem~\ref{NSTheorem} implies all $\pi_0 {\mathcal{L}}^{\ell} j_{\lambda} \in {\mathbb{C}} j_{\lambda}$.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{LRestrictionsSelfAdjoint} If $\overline{\varepsilon} \in {\mathbb{R}}$, $\hbar >0$, the restriction of ${\mathcal{L}}$ in~\eqref{PresentationOfL} to any $F[w]_n$ or $Z_{\lambda}$ is self-adjoint.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
{Follows from the definition of ${\mathcal{L}}$ in~(\ref{PresentationOfL}), the fact that $V_{\pm k}^{\dagger} = V_{\mp k}$ in~(\ref{VNegativeKDefinition}) are mutual adjoints for $\langle \cdot, \cdot \rangle_{\hbar}$, and that the differential operator $w\partial_w$ is self-adjoint for $\langle \cdot, \cdot \rangle_{\hbar}$.}
<!-- end proof-3 -->
<!-- lemma-4: body_md -->
\label{LEMMAJackLaxOrthogonality} If $\lambda$, $\gamma$ are distinct partitions of $n$, $Z_{\lambda}$ and $Z_{\gamma}$ are orthogonal in $(F[w]_n, \langle \cdot, \cdot \rangle_{\hbar})$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
If $\eta \in Z_{\lambda} = Z(j_{\lambda}, {\mathcal{L}})$, since $j_{\lambda}$ is cyclic, there is a polynomial $P$ so that $\eta = P({\mathcal{L}}) j_{\lambda}$. Similarly, if $\varphi \in Z_{\gamma} = Z(j_{\gamma}, {\mathcal{L}})$, there is a polynomial $Q$ so that $\varphi = Q({\mathcal{L}}) j_{\gamma}$. Then
\[ \langle \varphi, \eta \rangle_{\hbar} = \langle Q({\mathcal{L}}) j_{\gamma}, P({\mathcal{L}}) j_{\lambda} \rangle_{\hbar} = \langle j_{\gamma}, Q({\mathcal{L}}) P({\mathcal{L}}) j_{\lambda} \rangle_{\hbar} = 0 \]
since ${\mathcal{L}}^{\dagger} = {\mathcal{L}}$ is self-adjoint in $F[w]_n$, $\pi_0 Q({\mathcal{L}}) P({\mathcal{L}}) j_{\lambda} \in {\mathbb{C}} j_{\lambda}$ by~(\ref{ProjectionJackLax}), and $\langle j_{\gamma} , j_{\lambda} \rangle_{\hbar} = 0$.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
Since $\dim Z_{\lambda}<\infty$ by Lemma~\ref{JackLaxSpacesAreFiniteDim}, we may apply the general linear algebra results in the appendix Section~\ref{SUBSECrational} to the case $W = Z_{\lambda}$, $J = j_{\lambda}$, and $L = {\mathcal{L}} |_{Z_{\lambda}}$. From this perspective, formula~(\ref{TEigenvalueAddRemoveStyle}) in $(2^{\vee})$ of Nazarov--Sklyanin's Theorem~\ref{NSTheorem} gives an exact formula for the Titchmarsh--Weyl function $T(u)$ in~(\ref{TUDefinition}), so~(\ref{TAsRatioDeterminants}) implies~(\ref{TheoremStatementSimpleSpectrum}).
<!-- end proof-5 -->
<!-- proof-6: body_md -->
In any partition $\lambda$, the number of addable and removable corners always differ exactly by $1$:~(\ref{AddSetFormula}) and~(\ref{RemoveSetFormula}) satisfy

Since these properties uniquely characterize $\widetilde{\psi}^{(i)}$, we have proven~(\ref{KerovFormulaForPsiTilde}). Next, we prove~(\ref{KerovFormulaForPsi}) assuming $a=0$ so that $LJ = {\widetilde{J}}$ in~(\ref{JTildeDefinition}). To begin, the resolvent of ${\widetilde{L}}$ is well defined at $u=\sigma^{(i)}$ since in Theorem~\ref{GeneralSpectralTheorem} we have seen that ${\textnormal{spec}}(L) \cap {\textnormal{spec}}\big({\widetilde{L}}\big)= \varnothing$. As a consequence, ${\widetilde{J}} \in {\widetilde{W}}$ implies that $\frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} \in {\widetilde{W}}$, hence the vector on the right-hand side of~(\ref{KerovFormulaForPsi}) satisfies \[ \Pi_J \Bigg ( J + \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} \Bigg ) = J \]
 the normalization condition~(\ref{FirstJNormalizationCondition}). To verify that this vector is indeed an $L$-eigenvector with eigenvalue $\sigma^{(i)}$, multiply by $L$ then calculate its projections onto ${\mathbb{C}} J$ and ${\widetilde{W}} = ( {\mathbb{C}} J)^{\perp}$. In the~${\mathbb{C}} J$ projection, we can simplify using our assumption $a=0$ in~(\ref{JTildeDefinition}), the definition~(\ref{TTildeUDefinition}) of ${\widetilde{T}}(u)$, and ${\widetilde{T}}\big(\sigma^{(i)}\big) = \sigma^{(i)}$ at the special value $u = \sigma^{(i)}$ due to Corollary~\ref{COROLLARYhittingtimecramer}:
 \begin{equation} \label{PiJOfLTimesRHS}
 \Pi_J L \Bigg ( J + \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} \Bigg ) = \Pi_J L J + \Pi_J L \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} = a J + T\big(\sigma^{(i)}\big) J = \sigma^s J.
 \end{equation} In the ${\widetilde{W}} = ({\mathbb{C}} J)^{\perp}$
 projection, using $\frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} \in {\widetilde{W}}$, ${\widetilde{\Pi}} L {\widetilde{\Pi}} = {\widetilde{L}}$, and ${\widetilde{L}} = -\big({\sigma^{(i)}} - {\widetilde{L}}\big) + \sigma^{(i)}$,
\begin{gather}
{\widetilde{\Pi}} L \Bigg ( J + \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} \Bigg ) = {\widetilde{\Pi}} L J + {\widetilde{L}} \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}}= {\widetilde{J}} - {\widetilde{J}} + \sigma^{(i)} \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} \nonumber\\
\hphantom{{\widetilde{\Pi}} L \Bigg ( J + \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}} \Bigg )}{}= \sigma^{(i)} \frac{1}{\sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}}. \label{PiTildeOfLTimesRHS}
\end{gather}
Since~(\ref{PiJOfLTimesRHS}) and~(\ref{PiTildeOfLTimesRHS}) are $\sigma^{(i)}$ times the same projections of $J+ \frac{1}{ \sigma^{(i)} - {\widetilde{L}}} {\widetilde{J}}$,~(\ref{KerovFormulaForPsi}) follows.
