<!-- proof-6: body_md -->
The theory of $n$\nobreakdash-pre\-ca\-te\-gories\xspace is sketchable and the functor
  $(-)_k \colon \mathbf{PCat}_{n} \rightarrow \mathbf{Set}$, which to a precategory associates its set of
  $k$-cells is induced by a sketch morphism. It is thus a right
  adjoint \cite[Section~4, Theorem~4.1]{barr2000toposes}. In particular, it preserves monomorphisms. Thus,
  \ref{prop:charact-ppol-mono:mono} implies
  \ref{prop:charact-ppol-mono:comp-mono}. Moreover, since the functors $(-)_k$
  for $k < n+1$ are jointly faithful, we have that
  \ref{prop:charact-ppol-mono:comp-mono} implies
  \ref{prop:charact-ppol-mono:mono}.

<!-- theo-3: body_md -->
\label{thm:opol-concr-ps}
  The category $\mathbf{Pol}_{\omega}$ is a concrete presheaf category.
<!-- end theo-3 -->
<!-- proof-17: body_md -->
We verify that the various conditions of Makkai's criterion
  (\cref{thm:concrete-charact}) are satisfied.
  \begin{enumerate}
  \item[\ref{thm:concrete-charact:iso}] Clear from the definition of $\mathbf{Pol}_{\omega}$.
  \item[\ref{thm:concrete-charact:colimits}] A consequence of general properties
    satisfied by categories of polygraphs derived from a globular monad (see
    Propositions 1.3.3.7 and 1.3.3.15 of \cite{forest:tel-03155192}).
  \item[\ref{thm:concrete-charact:small}] Since a primitive element~$(\mathsf{P},g)$ is
    principal, the polygraph~$\mathsf{P}$ is finite. Thus, up to isomorphism, the sets
    $\mathsf{P}_i$ can be assumed to be subsets of $\mathbb{N}$. So
    that~\ref{thm:concrete-charact:small} holds.
  \item[\ref{thm:concrete-charact:prim-exists}] Given an element
    $(\mathsf{P},g) \in \operatorname{Elt}(\mathbf{Pol}_{\omega})$, by \Cref{prop:primitive-lifting}, there exists a
    polyplex lifting $F \colon (U,u) \rightarrow (\mathsf{P},g)$ of $\mathcal{U} (\mathsf{P},g)$. By
    \Cref{prop:freecat-functors}\ref{prop:freecat-functors:gen}, we have that
    $u \in U$. Moreover, by
    \Cref{prop:embedding-pol-polb}\ref{prop:embedding-pol-polb:plex}, we have
    that $(U,u)$ is a plex, so~\ref{thm:concrete-charact:prim-exists} holds.
  \item[\ref{thm:concrete-charact:prim-equal}] Given
    $f,g\colon (U,u) \rightarrow (X,x) \in \operatorname{Elt}(\mathbf{Pol}_{\omega})$ with $(U,u)$ a primitive plex, then
    we have that
    $\mathcal{U} (f),\mathcal{U}(g)\colon (U,u) \rightarrow (X,x) \in \operatorname{Elt}(\mathbf{Pol}_{\omega}^*)$, so
    that $\mathcal{U} (f),\mathcal{U}(g)$ by \Cref{lem:principal-parallel-eq},
    and $f = g$ by faithfulness.
  \item [\ref{thm:concrete-charact:prim-iso}] Given two
    morphisms~$f\colon (U,u) \rightarrow (X,x)$ and~$g\colon (V,v) \rightarrow (X,x)$ of~$\operatorname{Elt}(\mathbf{Pol}_{\omega})$
    where both~$(U,u)$ and~$(V,v)$ are primitive, we have by
    \Cref{prop:polyplex-lifting-unique} that there is an isomorphism
    \[
      \theta\colon \mathcal{U} (U,u) \rightarrow \mathcal{U} (V,v) \in \operatorname{Elt}(\mathbf{Pol}_{\omega}^*)
    \]
    such that~$\mathcal{U} (g) \circ \theta = \mathcal{U} (f)$. We conclude by the
    full faithfulness of $\mathcal{U}$.\qedhere
  \end{enumerate}

\label{weak-division}
  Given an $\omega$\nobreakdash-pre\-ca\-te\-gory\xspace $C$ and an equivalence $u : x \rightarrow y \in C_1$, for any
  $1$\nobreakdash-cells\xspace $s,t : y \rightarrow z$ and for any $2$\nobreakdash-cell\xspace
  $w : u \mathbin{\ast}_0 s \Rightarrow u \mathbin{\ast}_0 t$,
  \begin{enumerate}[(a)]
  \item \label{wd-e} there is a $2$\nobreakdash-cell\xspace $v : s \Rightarrow t$ such that
    $u \mathbin{\ast}_0 v \sim w$,
    \[
      \begin{tikzcd}[row sep=1ex]
        &y\ar[dr,bend left,"s"]\\
        x\ar[ur,bend left,"u"]\ar[dr,bend right,"u"']\ar[rr,phantom,"w\Downarrow"]&&z\\
        &y\ar[ur,bend right,"t"']
      \end{tikzcd}
      \qquad\sim\qquad
      \begin{tikzcd}
        x\ar[r,"u"]&y\ar[r,bend left=50,"s"]\ar[r,bend right=50,"t"']\ar[r,phantom,"v\Downarrow"]&z
      \end{tikzcd}
    \]
  \item \label{wd-u} for any $2$\nobreakdash-cells\xspace $v,v':s\Rightarrow t$ such that
    $u\mathbin{\ast}_0v\sim w\sim u\mathbin{\ast}_0 v'$ we have $v\sim v'$.
  \end{enumerate}
