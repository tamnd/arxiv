[ M_{{\mathbf{n}} } ]\mathbf{L} ^{- \langle \rho , {\mathbf{n}} \rangle }

<!-- mynotation-1: body_md -->
\label{def:euler-product}
Consider a $R$-variety $X$ and $\mathscr{X} = ( X_i )_{i\in I }  $ a family of elements of $\mathrm A_X$. Then the product
\[
\prod_{x\in X/R} \left ( 1 + \sum_{i\in I} X_{i,x}  T_i \right ) 
\]
is defined as a notation for the formal series
\[
\sum_{\pi \in \mathbf{N}^{(I)}} [ S ^\pi ( \mathscr{X} / R ) ] \mathbf{T}^\pi 
                                
\in \mathrm A_R [[\mathbf{T} ]] 
\]
where $\mathbf{T} = ( T_i )_{i\in I}$ is a family of indeterminates indexed by the set $I$ and $\mathbf{T}^\pi $ stands for $ \prod_{i\in I} T_i^{n_i}$ for every partition $\pi \in \mathbf{N}^{(I)}$.
<!-- end mynotation-1 -->
<!-- myptn-1: body_md -->
Let $R$ be a variety, $X$ be a $R$-variety and $\mathscr{X} = ( X_i )_{i\in I}$ be a family of elements of $ \mathrm A_X $.
\begin{itemize}
\item The Euler product notation is compatible with the $X=R$ case: 
\[
\prod_{x\in R/R} \left ( 1 + \sum_{i\in I} X_{i,x}  T_i \right ) = 1 + \sum_{i\in I} X_{i}  T_i .
\]

By definition of the motivic Euler product notation, for every ${\mathbf{m}} \in \mathbf{N}^r $ the coefficient $\mathfrak b_{\mathbf{m}}$ is given by the sum
\[\mathfrak b_{{\mathbf{m}}} = \sum_{\pi \text{ partition of }{\mathbf{m}} } [ S^\pi \mathscr{X} ] .\]
Since $w ( \mathfrak b_{{\mathbf{m}}} ) $ is smaller or equal to $\max (  \{ w ( S^\pi \mathscr{X} ) \mid \pi \text{ partition of } {\mathbf{m}} \} ) $, it is enough to find a uniform bound for the weight of $S^\pi \mathscr{X}$. 
So we fix $\pi = ( n_\mathbf i )_{\mathbf i\in  \mathbf{N}^r \setminus \{ \mathbf 0 \}}$ a partition of ${\mathbf{m}} $, 
that is such that $\sum_{\mathbf i\in  \mathbf{N}^r \setminus \{ \mathbf 0 \} } n_\mathbf i \mathbf i = {\mathbf{m}} $.
Then 
\begin{align*}
w( S^\pi \mathscr{X} )  & \leqslant w_{S^\pi X } ( S^\pi \mathscr{X} ) + \dim ( S^\pi X ) \\
& \leqslant \sum_{\mathbf i\in  \mathbf{N}^r \setminus \{ \mathbf 0 \}} n_\mathbf i w_X ( X_\mathbf i ) + \frac{1}{2} \sum_{\mathbf i\in  \mathbf{N}^r \setminus \{ \mathbf 0 \}} n_\mathbf i  w ( X ) \\

& \leqslant \left ( 1 - \delta + \frac{\beta}{M+1} \right ) \langle \rho , {\mathbf{m}} \rangle w ( X ) 
\end{align*}
where $\delta \in \left ] 0 , \frac{1}{2} \right ]$ is given by $1-\delta = \max \left ( 1 - \frac{\varepsilon}{M} , \alpha \right )$ if $M\neq 0 $ and $\delta = 1- \alpha $ otherwise. This proves the first part of the proposition.

\end{proof}

The proof is essentially the same as the one of Proposition 6.3.5.2 of \cite{bilu2018motivic}. We write down the details here for the sake of completeness. In particular, it is important to check that we are in the situation of Proposition \ref{proposition:linear-conv-criterion}, in order to obtain linear convergence with respect to $\rho $. 

It is enough to prove convergence for places of the dense open subset $C_1$ of $C_0$. Furthermore one can assume $\mathscr{A}_U = \mathscr{A}$.
For a place $v\in C_1$ the local factor $F_v ( \mathbf{T}, 0 ) =  \prod_{\alpha \in \mathscr{A}} ( 1 - \mathbf{L}^{\rho_\alpha -1} \mathbf T ) Z_v ( \mathbf{T} , 0 ) $ is given by 
\[
F_v ( \mathbf{T} , 0 )  = \sum_{A\subset \mathscr{A} } [\Delta_v ( A ) ] \mathbf{L}^{-n+|A|} ( 1 - \mathbf{L} ^{-1} )^{|A|} \prod_{\alpha \in A } \mathbf{L}^{\rho_\alpha -1} T_\alpha \prod_{\alpha \in \mathscr{A} \setminus A} ( 1 - \mathbf{L}^{\rho_\alpha - 1 }T_\alpha ) 
\]
If we expand the product over $\mathscr{A} \setminus A$ for any $A\subset \mathscr{A}$ we get 
\[
\prod_{\alpha ' \in \mathscr{A} \setminus A } ( 1 - \mathbf{L}^{\rho_{\alpha '} - 1 }T_{\alpha ' } ) = \sum_{B\subset \mathscr{A} \setminus A } (-1)^{|B|} \prod_{\alpha ' \in B } \mathbf{L}^{\rho_{\alpha ' } - 1 } T_{\alpha ' }
\]
and 
\[
F_v ( \mathbf{T} , 0 )  = \sum_{A\subset \mathscr{A} }  \sum_{B\subset \mathscr{A} \setminus A } (-1)^{|B|} [\Delta_v ( A ) ] \mathbf{L}^{-n+|A|} ( 1 - \mathbf{L} ^{-1} )^{|A|} \prod_{\alpha \in A } \mathbf{L}^{\rho_\alpha -1} T_\alpha  \prod_{\alpha ' \in B } \mathbf{L}^{\rho_{\alpha ' } - 1 } T_{\alpha ' } . 
\]
Remark that $[\Delta_v ( \varnothing ) ] = [\mathbf G_a ^n ] = \mathbf{L}^n$.
Then the previous sums can be decomposed with respect to the cardinalities of  the sets $A$ and $B$: 
\begin{align*}
F_v ( \mathbf{T} , 0 ) & =  1  -  \sum_{\alpha '  \in \mathscr{A} } \mathbf{L}^{\rho_{\alpha ' } -1}  T_{\alpha ' }  +  \sum_{\substack{B\subset \mathscr{A} \\ |B|\geqslant 2 } } (-1)^{|B|}   \prod_{\alpha ' \in B } \mathbf{L}^{\rho_{\alpha ' } - 1 } T_{\alpha ' }  \tag{$A=\varnothing$} \\
& + \sum_{\alpha \in \mathscr{A} }   [\Delta_v ( \{ \alpha \} ) ] \mathbf{L}^{-n+1} ( 1 - \mathbf{L} ^{-1} ) \mathbf{L}^{\rho_\alpha -1} T_\alpha  \tag{$|A|=1$, $B=\varnothing$}  \\
& + \sum_{\alpha \in \mathscr{A} }  \sum_{ \substack{B\subset \mathscr{A} \setminus \{ \alpha \} \\ B \neq \varnothing } } (-1)^{|B|} [\Delta_v ( \{ \alpha \} ) ] \mathbf{L}^{-n+1} ( 1 - \mathbf{L} ^{-1} )  \mathbf{L}^{\rho_\alpha -1} T_\alpha  \prod_{\alpha ' \in B } \mathbf{L}^{\rho_{\alpha ' } - 1 } T_{\alpha ' }  \tag{$|A|=1$, $B\neq \varnothing $}   \\
&  +  \sum_{\substack{A\subset \mathscr{A} \\ |A|\geqslant 2 } }  \sum_{B\subset \mathscr{A} \setminus A } (-1)^{|B|} [\Delta_v ( A ) ] \mathbf{L}^{-n+|A|} ( 1 - \mathbf{L} ^{-1} )^{|A|} \prod_{\alpha \in A } \mathbf{L}^{\rho_\alpha -1} T_\alpha  \prod_{\alpha ' \in B } \mathbf{L}^{\rho_{\alpha ' } - 1 } T_{\alpha ' } \tag{$|A|\geqslant 2$} .
\end{align*}
The definition of $\Delta_v ( A ) $ gives the equality of classes 
\[
 [\mathscr D_{\alpha , v} ] =  [\Delta_v ( \{ \alpha \} ) ] + \sum_{\substack{ A' \subset \mathscr{A}\setminus \{ \alpha \} \\ A' \neq \varnothing} } [\Delta_v ( A ' \cup \{ \alpha \} ) ] 
\] for every $\alpha \in \mathscr{A}$.
Finally                                         
\[
F_v ( \mathbf{T} , 0 ) = 1 + \sum_{ \alpha \in \mathscr{A} } \left ( [\mathscr D_{\alpha , v} ] - \mathbf{L}^{n-1} \right )\mathbf{L}^{-n} \mathbf{L}^{\rho_\alpha} T_\alpha  + P_v ( \mathbf{T} )
\]

\begin{align*}
P_v ( \mathbf{T} )  & = \sum_{\substack{B\subset \mathscr{A} \\ |B|\geqslant 2 } } (-1)^{|B|}   \prod_{\alpha ' \in B } \mathbf{L}^{\rho_{\alpha ' } - 1 } T_{\alpha ' } \\
& - \sum_{\alpha \in \mathscr{A} }   \sum_{\substack{ A' \subset \mathscr{A}\setminus \{ \alpha \} \\ A' \neq \varnothing } } [\Delta_v ( A ' \cup \{ \alpha \} ) ] \mathbf{L}^{-n+1} ( 1 - \mathbf{L} ^{-1} ) \mathbf{L}^{\rho_\alpha -1} T_\alpha \\
& + \sum_{\substack{A\subset \mathscr{A} \\ A \neq \varnothing  } }  \sum_{ \substack{ B\subset \mathscr{A} \setminus A  \\ |A|=1 \Rightarrow B \neq \varnothing } } (-1)^{|B|} [\Delta_v ( A ) ] \mathbf{L}^{-n+|A|} ( 1 - \mathbf{L} ^{-1} )^{|A|} \prod_{\alpha \in A } \mathbf{L}^{\rho_\alpha -1} T_\alpha  \prod_{\alpha ' \in B } \mathbf{L}^{\rho_{\alpha ' } - 1 } T_{\alpha ' }  .
\end{align*}
                                                                                                                                                                                               
Let us analyse the dimensions of the coefficients of this polynomial term by term, and compare them to the multidegrees of the corresponding monomials. In the sum of the first line, since $|B|\geqslant 2$, the corresponding coefficient has dimension at most $-2 + \sum_{\alpha \in B} \rho_\alpha $. 
Here the corresponding monomial is $\mathbf{T} ^{\mathbf{m}} $ with $m_\alpha = 1$ if $\alpha \in B $ and $0$ otherwise; thus  $\sum_{\alpha \in B } \rho_\alpha  = \langle \rho , {\mathbf{m}} \rangle $. 
Concerning the sum of the second line, 
since we assumed that $\mathscr{X}$ is a good model, 
the dimension of $\Delta_v ( A ' \cup \{ \alpha \} ) $ 
is at most $n - 2 $ and the whole coefficient has dimension at most $ \rho_\alpha -2  $. 
This remark still applies to the coefficients of the third and last sum when $|A|\geqslant 2$:
they have dimension at most $-2 + \sum_{\alpha \in A\cup B} \rho_\alpha $. 
It corresponds to the monomial $\mathbf T^{\mathbf{m}}  $  with $m_\alpha = 1$ if $\alpha \in A \cup B $ and $0$ otherwise; thus  $\langle \rho  , {\mathbf{m}} \rangle = \sum_{\alpha \in A\cup B} \rho_\alpha$. 
Finally, in the case $|A|=1$, then $|A\cup B|\geqslant 2 $ 
and the relation 
\[ \prod_{\alpha \in A \cup B } \mathbf{L}^{\rho_\alpha -1} = \mathbf{L}^{ \sum_{\alpha \in A\cup B} \rho_\alpha  - |A \cup B|} \]
ensures that the dimension of the corresponding coefficient is again at most $-2 + \sum_{\alpha \in A\cup B} \rho_\alpha $.

with $X=C_1$, ${ \varepsilon = \frac{1}{2}} $, ${ M=\deg ( P ( (T^{\rho_\alpha} )_{\alpha \in \mathscr{A}} )) }$ and $\beta = 0$ 
so that the first condition
of Proposition \ref{proposition:linear-conv-criterion} 
becomes in our case 
\begin{equation}\label{inequality:weight-of-polynomial-P(xi)}
w_{C_1} (  \mathfrak c_{\mathbf{m}} ) \leqslant 2 \left (  \langle \rho , {\mathbf{m}} \rangle - 1 \right )  . 
\end{equation}
By the last property of the weight recalled in Proposition \ref{proposition:properties-of-the-weight}, we  obtain the crucial argument of this proof, which is the inequality
\[
w_{C_1} \left ( \left ( [\mathscr D_{\alpha , v} ] - \mathbf{L}^{n-1} \right )\mathbf{L}^{-n} \mathbf{L}^{\rho_\alpha}  \right ) \leqslant 2 ( n - 1 ) - 2 n + 2 \rho_\alpha = 2 ( \rho _\alpha - 1 ).
\]
As Bilu points out in \cite[Remarks 6.3.4.2-3]{bilu2018motivic}, the bounds on dimensions obtained in the previous paragraph ensure that the coefficients of multidegree ${\mathbf{m}} $ of $P_v ( \mathbf{T} ) $ satisfy (\ref{inequality:weight-of-polynomial-P(xi)}). 
Thus we can apply Proposition \ref{proposition:linear-conv-criterion} and $\rho$-weight-linear convergence over $C_1$ at $T_\alpha = \mathbf{L}^{-\rho_\alpha}$ follows. Since the Euler product notation is compatible with finite products and since we already know that the local factors converge weight-linearly, applying Lemma \ref{lemma:weight-convergence-product-stability} we deduce that the $\rho$-weight-linear convergence holds for the product over $C_0$.

Once again it is enough to check convergence on the dense subset $C_1 \subset C$.  In order to apply Proposition \ref{proposition:linear-conv-criterion}, one has to bound the weight of the coefficients of $F_v (\mathbf{T}, \xi )$. We start by doing this for $Z_v ( \mathbf{T} , \xi ) $.

First, the divisor of $f_\xi$ can be written 
 \[
 \operatorname{\mathrm{div}} ( f_\xi ) = \Xi_\xi - \sum_{\alpha \in \mathscr{A} } d_\alpha  (\xi ) D_\alpha 
 \]
 where $\Xi_\xi$ is the Zariski-closure of $\{ \langle x , \xi \rangle = 0 \}$ in $X$.  
 Over a place $v\in C_1$, one has $\mathscr{B}^\mathscr{U}_v = \{ \beta_v \}$ and the local factor $Z_v(\mathbf{T} , \xi )$ can be written $Z_v (\mathbf{T} , \xi ) = \sum_{A\subset \mathscr{A}_U} Z_{v,A} ( \mathbf{T} , \xi ) $ with 
 \[
 Z_{v,A} ( \mathbf{T} , \xi ) = \int_{\Omega_v ( A , \beta_v ) } \prod_{\alpha \in A} \left ( \mathbf{L}^{\rho_\alpha } T_\alpha \right  ) ^{\operatorname{\mathrm{ord}} x_\alpha } \mathrm{e} ( f_\xi ( x ) ) \mathrm{d} x 
 \]
 as we already pointed out in Remark \ref{remark:dependance-of-local-factors}. 
It is possible to compute or at least bound the weight of this integral. We refer to \cite[\S 6.3.5.2]{bilu2018motivic} for the details of such computations; in this proof we will restrict ourselves to giving a list of results we need to apply Proposition \ref{proposition:linear-conv-criterion}. In particular we obtain weight-linear convergence of the Euler product $\prod_{v\in C_1} F_v ( \mathbf{T} , \xi )$. In what follows $\mathscr{X}_{\mathcal O_v}$ is written $\mathscr{X}$ for conciseness and the index $v$ may be dropped.  

If $A=\varnothing $ then the integral above equals one. 
If $A= \{ \alpha \}$ the intermediate step is to cut the integral into two pieces: one part corresponding to arcs with origin in the zero divisor $\Xi_\xi$ of $f_\xi$ and another part corresponding to arcs with origin outside $\Xi_\xi$. 
 \begin{align*}
 Z_{v,\{ \alpha \} } ( \mathbf{T}, \xi ) &= \int_{ \mathscr L ( \mathscr{X} , D_\alpha^\circ \setminus \Xi_\xi )} \prod_{\alpha \in A} \left ( \mathbf{L}^{\rho_\alpha } T_\alpha \right  ) ^{\operatorname{\mathrm{ord}} x_\alpha } \mathrm{e} ( f_\xi ( x ) ) \mathrm{d} x \\
 & +  \int_{\mathscr L ( \mathscr{X} , D_\alpha^\circ  \cap \Xi_\xi ) } \prod_{\alpha \in A} \left ( \mathbf{L}^{\rho_\alpha } T_\alpha \right  ) ^{\operatorname{\mathrm{ord}} x_\alpha } \mathrm{e} ( f_\xi ( x ) ) \mathrm{d} x
 \end{align*}
Then there are several cases to distinguish, according to the order of the pole of $f_\xi$ at $D_\alpha$. On one hand, concerning arcs with origins outside $\Xi_\xi$, one has the following results. 
 \begin{itemize}
 \item If $d_\alpha (\xi ) = 0 $ then 
 \[
 \int_{\mathscr L ( \mathscr{X} , D_\alpha^\circ \setminus \Xi_\xi )} \left ( \mathbf{L}^{\rho_\alpha } T_\alpha \right  ) ^{\operatorname{\mathrm{ord}} x_\alpha } \mathrm{e} ( f_\xi ( x ) ) \mathrm{d} x = ( 1 - \mathbf{L}^{-1} ) \frac{\mathbf{L}^{\rho_\alpha - 1 } T_\alpha }{1 - \mathbf{L}^{\rho_\alpha - 1 } T_\alpha } [D_\alpha^\circ \setminus \Xi_\xi ] \mathbf{L}^{-n+1} .
 \]
 \item If $d_\alpha (\xi ) =1 $ then 
  \[
 \int_{\mathscr L ( \mathscr{X} , D_\alpha^\circ \setminus \Xi_\xi ) } \left ( \mathbf{L}^{\rho_\alpha } T_\alpha \right  ) ^{\operatorname{\mathrm{ord}} x_\alpha } \mathrm{e} ( f_\xi ( x ) ) \mathrm{d} x = - \mathbf{L}^{-2} [D^\circ_\alpha  \setminus \Xi_\xi  ]\mathbf{L}^{1-n} \mathbf{L}^{\rho_\alpha} T_\alpha .
 \]
 Since $\dim ( D_\alpha^\circ \setminus \Xi_\xi ) = n-1$, one has the following upper bound on the weight
 \[
 w_{C_1} \left ( - \mathbf{L}^2 [D^\circ_\alpha  \setminus \Xi_\xi  ]\mathbf{L}^{1-n} \mathbf{L}^{\rho_\alpha} \right ) \leqslant 2 (  \rho_\alpha - 2  ) + 1 \leqslant 2  \rho_\alpha  -1 .
 \]
 \item If $d_\alpha (\xi ) > 1 $ then this integral equals zero. 
 \end{itemize}
On the other hand, concerning arcs with origin in $\Xi_\xi$,  the corresponding integral can be rewritten 
\begin{align*}
	& \int_{\mathscr L ( \mathscr{X} , D_\alpha^\circ  \cap \Xi_\xi ) } \left ( \mathbf{L}^{\rho_\alpha } T_\alpha \right  ) ^{\operatorname{\mathrm{ord}} x_\alpha } \mathrm{e} ( f_\xi ( x ) ) \mathrm{d} x  \\
& = 
\sum_{m\geqslant 1} \left ( \mathbf{L}^{\rho_\alpha } T_\alpha \right  ) ^{m} \int_{ \substack{ \left ( D_\alpha^\circ \cap \Xi_\xi \right ) \times  \mathscr L ( \mathbf A^1 ) \times \mathscr L ( \mathbf A^1 , 0 )^{n-1} \\ \operatorname{\mathrm{ord}} x = m }} \mathrm{e} ( f_\xi ( t^m x , \mathbf y ) ) \mathrm{d} x \mathrm{d} \mathbf y .
\end{align*}
By (\ref{inequ-weight-integral-volume}), the weight of the coefficient of order $m$ is smaller than the weight of the motivic volume 
\begin{align*}
	& \operatorname{\mathrm{vol}} \left  ( \{ ( w , x , \mathbf y ) \in \left ( D_\alpha^\circ \cap \Xi_\xi \right ) \times \mathscr L ( \mathbf A^1 ) \times \mathscr L ( \mathbf A^1 , 0  ) ^{n-1} \mid \operatorname{\mathrm{ord}} x = m \} \right ) \\
& =
\left [ D_\alpha^\circ \cap \Xi_\xi  \right ] \mathbf{L}^{-m} \left ( 1 - \mathbf{L}^{-1} \right ) \mathbf{L}^{-n+1}.
\end{align*}
One has the upper bound on dimensions 
\[
\dim \left ( \mathbf{L}^{\rho_\alpha m} [D_\alpha^\circ \cap \Xi_\xi  ] \mathbf{L}^{-m} \left ( 1 - \mathbf{L}^{-1} \right ) \mathbf{L}^{-n+1} \right ) \leqslant m\rho_\alpha + ( n -2 ) - m - ( n -1  ) = m ( \rho_\alpha   -  1  ) - 1 
\]
and thus on weights 
\[
w_{C_1} \left ( \mathbf{L}^{\rho_\alpha m} [D_\alpha^\circ \cap \Xi_\xi  ] \mathbf{L}^{-m} \left ( 1 - \mathbf{L}^{-1} \right ) \mathbf{L}^{-n+1} \right ) \leqslant 2 ( m ( \rho_\alpha - 1 )  - 1 ) + \dim ( C_1 ) .
\]
Setting $c = \underset{\alpha \in \mathscr{A}_U}{\max}\left  ( 1 - \frac{1}{2\rho_\alpha } \right )$ as in \cite{bilu2018motivic}, one gets $2\rho_\alpha - 1 \leqslant 2 \rho_\alpha c $ for every $\alpha \in \mathscr{A}_U$ and  
\[
w_{C_1} \left (\mathbf{L}^{\rho_\alpha m} [D_\alpha^\circ \cap \Xi_\xi  ] \mathbf{L}^{-m} \left ( 1 - \mathbf{L}^{-1} \right ) \mathbf{L}^{-n+1} \right ) \leqslant 2 c m \rho_\alpha - 1 . 
\]
Now if $|A | \geqslant 2 $ the idea is similar. The corresponding local term is 
\[
Z_{v,A} = \sum_{{\mathbf{m}} \in \mathbf{N}^A_{>0} } \prod_{\alpha \in A} \left ( \mathbf{L}^{\rho_\alpha} T_\alpha \right )^{m_\alpha} \int_{\operatorname{\mathrm{ord}} x_\alpha =  m_\alpha } \mathrm{e} ( f_\xi ( x , y ) ) \mathrm{d} x \mathrm{d} y .
\]
The volume of the constructible subsets of $\Omega_v ( A)$ over which integration is done is actually
\begin{align*}
& \operatorname{\mathrm{vol}} \left ( \left \{  (w, ( x_\alpha )_{\alpha \in A} , y ) \in D_A^\circ \times \mathscr L ( \mathbf A^1 )^{A} \times \mathscr L ( \mathbf A^1 , 0 ) ^{n-|A|} \mid \operatorname{\mathrm{ord}} x_\alpha = m_\alpha \text{ for every } \alpha \in A \right \} \right ) \\
& = [D_A^\circ ] \prod_{\alpha \in A} \left ( \mathbf{L}^{-m_\alpha} \left ( 1  - \mathbf{L}^{-1} \right ) \right ) \mathbf{L}^{-n+|A|} .
\end{align*}
By definition of $D_A^\circ$, it has dimension at most $n-|A|$. Thus the dimension of the integral is at most $- \sum_{\alpha \in A} m_\alpha $
and, using again the inequality of \S \ref{inequ-weight-integral-volume}, the weight of the ${\mathbf{m}}$-th coefficient of $Z_{v,A} ( \mathbf{T} , \xi ) $ is bounded by
\[
2 \sum_{\alpha \in A } m_\alpha ( \rho_\alpha -1 ) + \dim C_1 \leqslant 2 c \sum_{\alpha \in A} m_\alpha \rho_\alpha  - 1 .
\]

To summarize, we obtained 
\begin{align*}
Z_v ( \mathbf{T} , \xi ) & = 1 + \sum_{ \substack{ \alpha \in  \mathscr{A}_U \\ d_\alpha ( \xi ) \leqslant 1  } } Z_{v,\alpha} ( \mathbf{T}, \xi ) + \sum_{|A|\geqslant 2} Z_{v,A} ( \mathbf{T}, \xi )\\
Z_v ( \mathbf{T} , \xi ) &=  1 + \mathbf{L}^{-n} \left ( 1 - \mathbf{L}^{-1}\right ) \sum_{\alpha \in \mathscr{A}_U^0 (\xi ) }  [ D^\circ _\alpha \setminus \Xi_\xi ] \frac{\mathbf{L}^{\rho_\alpha} T_\alpha }{1- \mathbf{L}^{\rho_\alpha -1} T_\alpha }  - \mathbf{L}^{-n-1} \sum_{\substack{ \alpha \in  \mathscr{A}_U ( \xi ) \\ d_\alpha ( \xi ) = 1  }}   [ D^\circ _\alpha \setminus \Xi_\xi ] \mathbf{L}^{\rho_\alpha } T_\alpha \\
&+ \text{ terms of weights bounded as in Proposition \ref{proposition:linear-conv-criterion}, locally uniformly in $\xi$}\\
                                                                                                                                                                                                                                            
\end{align*}

The last step of the proof consists in multiplying by $\prod_{\alpha \in \mathscr{A}_U^0 (\xi )} \left ( 1 - \mathbf{L}^{\rho_\alpha -1 }T_\alpha \right )$. This operation does not affect the bounds we need in order to apply Proposition \ref{proposition:linear-conv-criterion}, since $\rho_\alpha - 1 \leqslant c\rho_\alpha$,
and allows one to control the diverging term \[
\sum_{\alpha \in \mathscr{A}_U^0 (\xi ) }  [ D^\circ _\alpha \setminus \Xi_\xi ] \frac{\mathbf{L}^{\rho_\alpha} T_\alpha }{1- \mathbf{L}^{\rho_\alpha -1} T_\alpha } 
\]
in the expression of $Z_v ( \mathbf{T} , \xi )$. 
One gets 
\begin{align*}
F_v ( \mathbf{T} , \xi ) & = 1 - \sum_{\alpha \in \mathscr{A}_U^0 (\xi ) } \mathbf{L}^{\rho_\alpha -1} T_\alpha + \mathbf{L}^{1-n} \sum_{\alpha \in \mathscr{A}_U^0 (\xi ) } [ D^\circ _\alpha \setminus \Xi_\xi ] \mathbf{L}^{\rho_\alpha -1 }  T_\alpha + P_v ( \mathbf{T}, \xi ) \\
& = 1 + \sum_{\alpha \in \mathscr{A}_U^0 (\xi ) } \left ( [ D_\alpha^\circ \setminus \Xi_\xi ] -  \mathbf{L}^{n-1} \right )\mathbf{L}^{1-n} \mathbf{L}^{\rho_\alpha -1 }  T_\alpha + P_v ( \mathbf{T}, \xi )
\end{align*}
where $P_v ( \mathbf{T}, \xi ) $ is a Laurent series consisting of terms satisfying the bounds of Proposition \ref{proposition:linear-conv-criterion} for 
\[ 
c = \underset{\alpha \in \mathscr{A}_U}{\max}\left  ( 1 - \frac{1}{2\rho_\alpha } \right ).
\] 
Finally, by Proposition \ref{proposition:properties-of-the-weight} one has for every $\alpha \in \mathscr{A}_U^0 ( \xi)$
\[
w_{C_1} \left ( \left ( [ D_\alpha^\circ \setminus \Xi_\xi ] -  \mathbf{L}^{n-1} \right )\mathbf{L}^{1-n} \mathbf{L}^{\rho_\alpha -1 }   \right )  \leqslant 2 ( n -1 ) - 2 ( n - 1 ) + 2 ( \rho_\alpha - 1 ) = 2 ( \rho_\alpha - 1 ) .
\]
Our analysis shows that we can apply Proposition \ref{proposition:linear-conv-criterion} with $X=C_1$, $\varepsilon = \frac{1}{2}$, $M$ arbitrary and $\beta = 0$. The result follows.
