<!-- theorem-5: body_md -->
\label{thm:opt}
Suppose the learner is given an action $\widehat A_w \in \mathbb B^d_1$ such that $\langle \widehat A_w, \theta_\star \rangle^2 \geq \alpha r^2$ for some universal constant $\alpha \in (0,1]$.
Then there exists a policy $\pi$ for which the regret is at most $\mathfrak{R}_n(\pi, \theta_\star) \leq \operatorname{const} \cdot d \sqrt{n \log(n)}$.
<!-- end theorem-5 -->
<!-- proof-1: body_md -->
By choosing the sign of $\theta_\star$, assume without loss of generality that $\langle \widehat A_w, \theta_\star \rangle \geq r \sqrt{\alpha}$. Let
\begin{align*}
m = \left\lceil 4d \sqrt{n \log(n)} / r^2 \right\rceil \qquad \text{and} \qquad \lambda = \min\left(\frac{1}{2},\, \frac{\sqrt{\alpha}}{4}\right) \,.
\end{align*}
If $m \geq n$, then the regret of any policy is upper bounded $n r^2 \leq \operatorname{const} d \sqrt{n \log(n)}$, so for the rest of the proof we assume that $m < n$.
For the first $m$ rounds the policy cycles over the 
$2d$ actions $\{(1 - \lambda) \widehat A_w \pm \lambda e_k : k \in [d]\}$. 
The constrained least squares estimator of $\theta_\star$ based on the data collected over $m$ rounds is
\begin{align}\label{eqn:CLS}
\hat \theta = \operatornamewithlimits{arg\,min} \{\mathcal L(\theta) : \theta \in \mathbb B^d_r \text{ and } \langle \widehat A_w, \theta \rangle \geq r \sqrt{\alpha}\}\,,
\end{align}
where $\mathcal L(\theta) = \frac{1}{2} \sum_{t=1}^m (X_t - \langle A_t, \theta \rangle^2)^2$.
For the remaining $n - m$ rounds the algorithm plays $A_t = \widehat A = \hat \theta / \Vert \hat \theta \Vert$. Then,
\begin{align}
\beta 
&\triangleq 9 + d \log(98m) \\
\tag{by \cref{cor:conc}}  &\geq \mathbb E\left[\sum_{t=1}^m \langle A_t, \hat \theta - \theta_\star \rangle^2 \langle A_t, \hat \theta + \theta_\star \rangle^2\right] \nonumber \\
&\geq \frac{\alpha \Vert \theta_\star \Vert^2}{4} \mathbb E\left[\sum_{t=1}^m \langle A_t, \hat \theta - \theta_\star \rangle^2\right] \label{eq:alg1} \\
&\geq \frac{\alpha \lambda^2 \Vert \theta_\star \Vert^2}{2} \left(\frac{m}{2d}-1\right) \mathbb E\left[\Vert \hat \theta - \theta_\star \Vert^2\right]\,.  \label{eq:alg2}
\end{align}
where in \cref{eq:alg1} we used that for some $k \in [d]$,
\begin{align*}
\langle A_t, \hat \theta + \theta_\star \rangle 
&= \langle (1 - \lambda)\widehat A_w \pm \lambda e_k, \hat \theta + \theta_\star \rangle\geq (1 - \lambda) \langle \widehat A_w, \hat \theta + \theta_\star \rangle - \lambda \Vert \hat \theta + \theta_\star \Vert \\
&\geq 2(1 - \lambda) \sqrt{\alpha} r - 2\lambda r 
\tag{by definition of $\lambda$} \geq \frac{\sqrt{\alpha}}{2}\Vert \theta_\star \Vert\,.
\end{align*}
\cref{eq:alg2} follows because for any $k \in [d]$,
\begin{align*}
\sum_{\sigma \in \pm 1}
\langle (1 - \lambda) \widehat A_w + \sigma \lambda e_k, \hat \theta - \theta_\star \rangle^2 
&= 2(1 - \lambda)^2 \langle \widehat A_w, \hat \theta - \theta_\star \rangle^2 + 2 \lambda^2 \langle e_k, \hat \theta - \theta_\star \rangle^2 \\
& \geq 2 \lambda^2 \langle e_k, \hat \theta - \theta_\star \rangle^2\,,
\end{align*}
which implies that
\begin{align*}
\sum_{t=1}^m \langle A_t, \hat \theta - \theta_\star \rangle^2
&\geq \left\lfloor \frac{m}{2d} \right\rfloor \sum_{k=1}^d \left(2\lambda^2 \langle e_k, \hat \theta - \theta_\star \rangle^2\right) 
\geq 2\lambda^2 \left(\frac{m}{2d} - 1\right) \Vert \hat \theta - \theta_\star \Vert^2\,. 
\end{align*}
Rearranging \cref{eq:alg2} and using the definition of $\beta$ shows that
\begin{align*}
\mathbb E\left[\Vert \hat \theta - \theta \Vert^2\right] \leq \frac{2 \beta}{\alpha \lambda^2 \Vert \theta_\star \Vert^2} \frac{1}{\frac{m}{2d} - 1} \leq \operatorname{const} \frac{d^2 \log(m)}{m \Vert \theta_\star \Vert^2} \,.
\end{align*}
Letting $a_\star = \theta_\star / \Vert \theta_\star \Vert$ be the optimal action,
the regret is bounded by
\begin{align*}
\mathfrak{R}_n(\pi, \theta_\star) 
&\leq m \Vert \theta_\star \Vert^2 + n \mathbb E\left[\langle a_\star, \theta_\star \rangle^2 - \langle \widehat A, \theta_\star \rangle^2\right] \\
&= m \Vert \theta_\star \Vert^2 + n \mathbb E\left[\langle a_\star - \widehat A, \theta_\star \rangle \langle a_\star + \widehat A, \theta_\star \rangle\right] \\
&\leq m \Vert \theta_\star \Vert^2 + 2n \mathbb E\left[\left|\langle a_\star - \widehat A, \theta_\star \rangle\right| \Vert \theta_\star \Vert\right] \\
\tag{by \cref{lem:curve}} &\leq m \Vert \theta_\star \Vert^2 + 4n \mathbb E\left[\Vert \theta_\star - \hat \theta \Vert^2\right] \\
&\leq m \Vert \theta_\star \Vert^2 + \operatorname{const} \frac{nd^2 \log(m)}{m \Vert \theta_\star \Vert^2} \leq \operatorname{const} d \sqrt{n \log(n)}\,. \qedhere
\end{align*}

<!-- theorem-6: body_md -->
\label{thm:warm}
Let $T$ be the random number of rounds that \cref{alg:warm} interacts with the bandit, which cannot be more than $n$, and let $\widehat A_w \in \mathbb S^{d-1}_1$ be its output.
Then,
\begin{enumerate}
\item $\mathbb E[T] \leq \operatorname{const} \frac{d^2}{r^4} \log(n) \log(d)$.
\item With probability at least $1 - 1/n$, either $T = n$ or $\langle \widehat A_w, \theta_\star \rangle^2 \geq \frac{r^2}{64}$.
\end{enumerate}
<!-- end theorem-6 -->
<!-- proof-2: body_md -->
Note that the vectors $u$ and $v$ computed in each iteration are orthogonal, which means that $\Vert (u+v)/\sqrt{2} \Vert = \Vert (u-v)/\sqrt{2} \Vert = \Vert v \Vert = 1$. 
Hence the actions of the algorithm are always in $\mathbb B^d_1$.
The main argument of the proof is based on an induction to show that with high probability when the execution of the algorithm ends, there exists an $s \in \{\pm 1\}$ 
such that for all $w \in \mathcal E$,
\begin{enumerate}
\item[(a)] $w \in \mathcal E$, $\Vert w \Vert^2 \geq \tau^2$; and
\item[(b)] $s \langle w, \theta_\star \rangle \in [\frac{1}{2}\Vert w \Vert^2, 2 \Vert w \Vert^2]$.
\end{enumerate}
We proceed in five steps. First, we prove that if the above holds and the algorithm halts before $n$ rounds are over, then the vector returned
is a suitable approximation of $\theta_\star / \Vert \theta_\star \Vert$. Second, we upper bound the probability of certain bad events.
In the third and fourth steps we prove the base case and induction step for (a) and (b). In the last step we bound the expected running time.

= \sqrt{\bar X} \langle v, \theta_\star \rangle
\in \left[\frac{1}{2} \Vert w \Vert^2, 2 \Vert w \Vert^2\right]\,. 
\end{align*}
This establishes the base case.

{Step 4: Inductive step}
Assume that (a) and (b) above hold for $\mathcal E$ at the end of iteration $k$. Let $u$ be the value computed in Line~\ref{line:u} of \cref{alg:warm}. Then,
\begin{align*}
\langle u, \theta_\star \rangle
&= \frac{\sum_{w \in \mathcal E} \langle w, \theta_\star \rangle}{\sqrt{\sum_{w \in \mathcal E} \Vert w \Vert^2}} 
\geq \frac{1}{2} \sqrt{\sum_{w \in \mathcal E} \Vert w \Vert^2} 
\geq \frac{\tau \sqrt{k}}{2}\,.
\end{align*}
Let $\mathcal A = \{(u + v)/\sqrt{2}, (u - v)/\sqrt{2}, v\}$, which is the set of actions played in the inner loop of iteration $k+1$ after sampling $v$ for the last time. 
Let $\hat \theta$ be the corresponding least-squares estimate. We consider two cases. First, if $\langle u, \hat \theta + \theta_\star \rangle \geq 2 |\langle v, \hat \theta + \theta_\star \rangle|$, then by \cref{eq:warm1},
\begin{align*}
\frac{\beta}{m}
&\geq \sum_{a \in \mathcal A} \langle a, \theta_\star - \hat \theta \rangle^2 \langle a, \theta_\star + \hat \theta \rangle^2 \\ 
&\geq \frac{1}{16} \langle u + v, \theta_\star - \hat \theta \rangle^2 \langle u, \theta_\star \rangle^2 + \frac{1}{16} \langle u - v, \theta_\star - \hat \theta \rangle^2 \langle u, \theta_\star \rangle^2 \\
&\geq \frac{1}{8} \langle v, \theta_\star - \hat \theta \rangle^2 \langle u, \theta_\star \rangle^2 \geq \frac{k r^2}{16d} \langle v, \theta_\star - \hat \theta \rangle^2\,. 
\end{align*}
Rearranging shows that
\begin{align}
\langle v, \theta_\star - \hat \theta \rangle^2 \leq \frac{16d \beta}{m k r^2} \leq \frac{\tau^2}{4}\,. 
\label{eq:warm}
\end{align}
For the second case, $\langle u, \hat \theta + \theta_\star \rangle \leq 2 |\langle v, \hat \theta + \theta_\star \rangle|$. Then,
\begin{align*}
\frac{\beta}{m}
\geq \sum_{a \in \mathcal A} \langle a, \theta_\star - \hat \theta \rangle^2 \langle a, \theta_\star + \hat \theta \rangle^2 
\geq \frac{1}{4} \langle v, \hat \theta - \hat \theta \rangle^2 \langle u, \hat \theta + \theta_\star \rangle 
\geq \frac{k r^2}{8d} \langle v, \theta_\star - \hat \theta \rangle^2\,.
\end{align*}
And again, \cref{eq:warm} holds.
Summarising, $\langle v, \hat \theta \rangle$ is an estimator of $\langle v, \theta_\star \rangle$ up to accuracy $\tau / 2$.
By the definition of the algorithm, the iteration only ends if $|\langle v, \hat \theta \rangle| \geq \tau$.
Therefore, with $w = \langle v, \hat \theta \rangle v$, we have
$\Vert w \Vert^2 = \langle v, \hat \theta \rangle^2 \geq \tau^2$.
Furthermore, 
$
\langle w, \theta_\star \rangle = \langle v, \hat \theta \rangle \langle v, \theta_\star \rangle 
\in [\Vert w \Vert^2/2, 2 \Vert w \Vert^2]\,.
$
Therefore if (a) and (b) hold for $\mathcal E$ computed after iteration $k$, they also hold for $\mathcal E$ computed after iteration $k+1$.

{Step 5: Running time}
The length of an iteration is determined by the corresponding value of $m$ and the number of samples of $v$. The former is an iteration-dependent constant, while the latter depends principally on how many samples are needed before $|\langle v, \theta_\star \rangle|$ is suitably large. The law of $\nu$ 
is the uniform distribution on $\mathbb S^{d-1}_1 \cap \operatorname{span}(\mathcal E)^\perp$, which is the uniform distribution on a sphere of dimension $d-1-|\mathcal E|$ embedded in $\mathbb R^d$. 
The squared norm of the projection of $\theta_\star$ onto $\operatorname{span}(\mathcal E)^\perp$ is
\begin{align*}
\Vert \theta_\star \Vert^2 - \sum_{w \in \mathcal E} \frac{\langle \theta_\star, w \rangle^2}{\Vert w \Vert^2}
\geq r^2 - 4\sum_{w \in \mathcal E} \Vert w \Vert^2 
\geq \frac{r^2}{2}\,,
\end{align*}
where we used (a) of the induction and the stopping condition in Line~\ref{line:stop}.
Therefore, when $v$ is sampled uniformly from $\mathbb S^{d-1}_1 \cap \operatorname{span}(\mathcal E)^\perp$, by \cref{lem:sphere},
\begin{align*}
\mathbb P\left(\langle v, \theta_\star \rangle^2 \geq 3\tau^2/2\right)
&= \mathbb P\left(\langle v, \theta_\star \rangle^2 \geq \frac{3r^2}{2d}\right) 
\geq \operatorname{const} > 0\,,
\end{align*}
Furthermore, by the concentration analysis in the previous step, an iteration will end once a $v$ has been sampled
for which $\langle v, \theta_\star \rangle^2 \geq 3\tau^2 / 2$. 
Hence, the expected number of times the algorithm samples $v$ per iteration is constant and a simple calculation using 
the definition of $m$ in Lines~\ref{line:m1} and \ref{line:u} shows that the expected number of rounds used by the
algorithm is at most 
\begin{align*}
\mathbb E[T] &\leq \operatorname{const} \frac{d^2}{r^4} \log\left(n\right) \log(d)\,.
\qedhere
\end{align*}

We use a standard reduction \cite[Chapter 33]{LS20book}.
Let $\pi$ be the policy used in the proof of \cref{thm:main} with $\widehat A_\star$ sampled uniformly from $(A_t)_{t=1}^n$. By \cref{thm:main},
\begin{align*}

<!-- proof-5: body_md -->
Let $r$ be a positive constant to be tuned subsequently and $\sigma$ be the uniform (Haar) measure on
$\mathbb S^{d-1}_r$. Let $\pi$ be an arbitrary policy.
Given $\theta \in \mathbb R^d$, let $\mathbb P_\theta$ be the measure on interaction sequences induced by the interaction between $\pi$ and the bandit determined by $\theta$. 
Let $\mathbb P = \sigma \otimes \mathbb P_\theta$ be the product of $\sigma$ and probability kernel $(\mathbb P_\theta : \theta \in \mathbb S^{d-1}_r)$.
As usual, let $\mathbb E_\theta$ denote the expectation with respect to $\mathbb P_\theta$ and $\mathbb E$ the expectation with respect to $\mathbb P$.
Let $Z_t = \sum_{s=1}^t X_s \langle A_s, \theta \rangle^2 - \frac{1}{2} \langle A_s, \theta \rangle^4$.
Then,
\begin{align*}
&\mathbb E[\langle A_t, \theta \rangle^4 \bm 1(Z_{t-1} \leq \gamma)]
= \int_{\mathbb S_r^{d-1}} \mathbb E_\theta[\langle A_t, \theta \rangle^4 \bm 1(Z_{t-1} \leq \gamma)] \operatorname{d}\!\sigma(\theta) \\
&= \int_{\mathbb S_r^{d-1}} \mathbb E_{ \bm 0}\left[\langle A_t, \theta \rangle^4 \bm 1(Z_{t-1} \leq \gamma) \exp\left(-\frac{1}{2} \sum_{s=1}^{t-1}  \left((X_s - \langle A_s, \theta \rangle^2)^2 - X_s^2\right)\right)\right] \operatorname{d}\!\sigma(\theta)\\
&= \int_{\mathbb S_r^{d-1}} \mathbb E_{ \bm 0} \left[\langle A_t, \theta \rangle^4 \bm 1(Z_{t-1} \leq \gamma) \exp(Z_{t-1})\right] \operatorname{d}\!\sigma(\theta) \\
&\leq \exp(\gamma) \int_{\mathbb S_r^{d-1}} \mathbb E_{ \bm 0}\left[\langle A_t, \theta \rangle^4\right] \operatorname{d}\!\sigma(\theta) 
= \frac{3\exp(\gamma) r^4}{d^2 + 2d}\,.
\end{align*}
Next, define a random time
\begin{align*}
    \tau = \min\left\{t : Z_t \geq \gamma \text{ or } \sum_{s=1}^t \langle A_s, \theta \rangle^4 \geq 1\right\}
\quad \text{with } \gamma = 1 + \log(4)\,.
\end{align*}
Now,
\begin{align*}
\{\tau \leq n\}
\subset \left\{\sum_{t=1}^\tau X_s \langle A_s, \theta \rangle^2 - \frac{3}{2} \langle A_s, \theta \rangle^4 \geq \gamma - 1 \right\}
\cup \left\{\sum_{t=1}^n \langle A_s, \theta \rangle^4 \bm 1(\tau \geq t) \geq 1\right\}\,.
\end{align*}
By a union bound,
\begin{align*}
\mathbb P(\tau \leq n) 
&\leq \underbracket{\mathbb P\left(\sum_{t=1}^\tau X_s \langle A_s, \theta \rangle^2 - \frac{3}{2} \langle A_s, \theta \rangle^4 \geq \gamma -1\right)}_{\textrm{(A)}} \\
&\qquad+ \underbracket{\mathbb P\left(\sum_{t=1}^n \langle A_s, \theta \rangle^4 \bm 1(\tau \geq t) \geq 1\right)}_{\textrm{(B)}}\,.
\end{align*}
By the definition of $X_s$, Markov's inequality and the moment generating function of a Gaussian,
\begin{align*}
\textrm{(A)} 
&\leq \exp(1-\gamma) \mathbb E\left[\exp\left(\sum_{t=1}^\tau X_s \langle A_s, \theta \rangle^2 - \frac{3}{2} \langle A_s, \theta \rangle^4\right)\right]
\leq \exp(1-\gamma) = \frac{1}{4}\,.
\end{align*}
Meanwhile, by Markov's inequality, 
\begin{align*}
\textrm{(B)} \leq \mathbb E\left[\sum_{t=1}^n \langle A_t, \theta \rangle^4 \bm 1(\tau \geq t)\right]
\leq \frac{6 \exp(\gamma) n r^4}{d^2 + 2d} = \frac{24 \exp(1) n r^4}{d^2 + 2d} = \frac{1}{4}\,,
\end{align*}
where the last equality follows by choosing $r^2 = \sqrt{(d^2 + 2d)/(96 \exp(1)n)}.$
Hence, $\int_{\mathbb S^{d-1}_r} \mathbb P_\theta(\tau \leq n) \operatorname{d}\!\sigma(\theta) = \mathbb P(\tau \leq n) \leq 1/2$ and so
there exists a $\theta \in \mathbb S^{d-1}_r$ such that
\begin{align*}
\mathbb P_\theta\left(\sum_{t=1}^n \langle A_t, \theta \rangle^4 \leq 1\right) \geq 1/2\,.
\end{align*}
Suppose that $\sum_{t=1}^n \langle A_t, \theta \rangle^4 \leq 1$, then
by Cauchy-Schwarz, 
\begin{align*}
\sum_{t=1}^n \langle A_t, \theta \rangle^2 
\leq \sqrt{n \sum_{t=1}^n \langle A_t, \theta \rangle^4}
\leq \sqrt{n} \,.
\end{align*}
Therefore, $\mathbb P_\theta(\sum_{t=1}^n \langle A_t, \theta \rangle^2 \leq \sqrt{n}) \geq 1/2$ and
\begin{align*}
\mathfrak{R}_n(\pi, \theta) 
&=\mathbb E_\theta\left[\sum_{t=1}^n \left(r^2 - \langle A_t, \theta \rangle^2\right)\right] \\
&\geq \mathbb P_\theta\left(\sum_{t=1}^n \langle A_t, \theta \rangle^2 \leq \sqrt{n}\right) (n r^2 - \sqrt{n}) \\
&\geq \frac{nr^2 - \sqrt{n}}{2}\,.
\end{align*}
The result follows from choice of $r$.

<!-- theorem-7: body_md -->
\label{thm:conc}
Suppose that $\theta_\star \in \mathbb B^d$ and
$\hat \theta = \operatornamewithlimits{arg\,min}_{\theta \in \Theta} \mathcal L(\theta)$. Then, for any $\delta \in (0,1)$, with probability at least $1 - \delta$,
\begin{align*}
\mathbb P\left(\sum_{t=1}^n \langle \hat \theta - \theta_\star, A_t \rangle^2 \langle \hat \theta + \theta_\star, A_t \rangle^2 
\geq 9 \log \left(\frac{N_{1/(32n)}(\Theta)}{\delta}\right)\right) \leq \delta\,,
\end{align*}
where $N_\epsilon(\Theta) = \min\{|\mathcal C| : \mathcal C \subset \mathbb R^d, \forall x \in \Theta, \min_{y \in \mathcal C} \Vert x - y \Vert \leq \epsilon\}$.
<!-- end theorem-7 -->
<!-- proof-7: body_md -->
Since $\theta_\star \in \mathbb B^d$ by assumption, it follows that
\begin{align*}
0 
&\leq \mathcal L(\theta_\star) - \mathcal L(\hat \theta) \\
&= -\frac{1}{2} \sum_{t=1}^n \langle A_t, \hat \theta - \theta_\star \rangle^2 \langle A_t, \hat \theta + \theta_\star \rangle^2
+ \sum_{t=1}^n \eta_t \langle A_t, \hat \theta - \theta_\star \rangle \langle A_t, \hat \theta + \theta_\star \rangle \,.
\end{align*}
Let $\epsilon = 1/(32n)$ and $\mathcal C \subset \mathbb R^d$ be such that for all $x \in \Theta$ there exists a $y \in \mathcal C$ such that $\Vert x - y \Vert \leq \epsilon$
and $|\mathcal C| = N_\epsilon(\Theta)$.
Since $A_t$ are fixed, by a union bound and standard Gaussian tail bounds, with probability at least $1 - |\mathcal C| \delta$,
\begin{align*}
    \left|\sum_{t=1}^n \eta_t \langle A_t, \alpha - \theta_\star \rangle\langle A_t, \alpha + \theta_\star \rangle\right| 
    &\leq \sqrt{2 \sum_{t=1}^n \langle A_t, \alpha - \theta_\star \rangle^2 \langle A_t, \alpha + \theta_\star \rangle^2 \log\left(\frac{1}{\delta}\right)}\,. 
\end{align*}
On this event and letting $\alpha \in \mathcal C$ be such that $\Vert \alpha - \hat \theta \Vert \leq \epsilon$. Then, with $\Delta = \alpha - \hat \theta$,
\begin{align*}
&\sum_{t=1}^n \langle A_t, \hat \theta - \theta_\star \rangle^2 \langle A_t, \hat \theta + \theta_\star \rangle^2
\leq \sqrt{8 \sum_{t=1}^n \langle A_t, \alpha - \theta_\star \rangle^2 \langle A_t, \alpha + \theta_\star \rangle^2 \log\left(\frac{1}{\delta}\right)} \\
&= \sqrt{8 \sum_{t=1}^n \langle A_t, \Delta + \hat \theta - \theta_\star \rangle^2 \langle A_t, \Delta + \hat \theta + \theta_\star \rangle^2 \log\left(\frac{1}{\delta}\right)} \\
&\leq \sqrt{8 \sum_{t=1}^n 
\left(\langle A_t, \hat \theta - \theta_\star \rangle^2 + 2 \epsilon + \epsilon^2\right)
\left(\langle A_t, \hat \theta + \theta_\star \rangle^2 + 4 \epsilon + \epsilon^2\right) \log\left(\frac{1}{\delta}\right)} \\
&\leq \sqrt{8 \sum_{t=1}^n \left(\langle A_t, \hat \theta - \theta_\star \rangle^2 \langle A_t, \hat \theta + \theta_\star \rangle^2 + 12\epsilon + 13\epsilon^2 + 6\epsilon^3 + \epsilon^4\right) \log\left(\frac{1}{\delta}\right)} \\
&\leq \sqrt{8 \sum_{t=1}^n \left(\langle A_t, \hat \theta - \theta_\star \rangle^2 \langle A_t, \hat \theta + \theta_\star \rangle^2 + 32\epsilon\right) \log\left(\frac{1}{\delta}\right)} \\
&\leq \sqrt{\left(1 + 8 \sum_{t=1}^n \langle A_t, \hat \theta - \theta_\star \rangle^2 \langle A_t, \hat \theta + \theta_\star \rangle^2 \right) \log\left(\frac{1}{\delta}\right)} \,, 
\end{align*}
where in the final inequality we chose $\epsilon = 1/(32n)$.
Solving for the left-hand side and naive simplification shows that
\begin{align*}
\sum_{t=1}^n \langle A_t, \hat \theta - \theta_\star \rangle^2 \leq 9 \log\left(\frac{1}{\delta}\right)\,.
\end{align*}
To summarise we have shown that with probability at least $1 - \delta$,
\begin{align*}
\sum_{t=1}^n \langle A_t, \hat \theta - \theta_\star \rangle^2 \langle A_t, \hat \theta + \theta_\star \rangle^2 
&\leq 9 \log\left(\frac{|\mathcal C|}{\delta}\right)
= 9 \log\left(\frac{N_\epsilon(\Theta)}{\delta}\right) \,.
\qedhere
\end{align*}
