<!-- head-1: body_md -->
Through the paper, given a complex Banach space $E$, we denote by $B_E$, $\stackrel{\circ}{B}_E$, $S_E$ and $E^*$ the closed unit ball, the open unit ball, the unit sphere and the dual space of $E$, respectively. For a set $A\subseteq E$, $\mathrm{lin}(A)$, $\overline{\mathrm{lin}}(A)$, $\mathrm{co}(A)$, $\overline{\mathrm{co}}(A)$, $\mathrm{aco}(A)$ and $\overline{\mathrm{aco}}(A)$ stand for the linear hull, the norm-closed linear hull, the convex hull, the norm-closed convex hull, the absolutely convex hull, the norm-closed absolutely convex hull of $A$ in $E$, respectively. If $E$ and $F$ are locally convex Hausdorff spaces, $\mathcal{L}(E;F)$ denotes the vector space of all continuous linear operators from $E$ into $F$. Unless stated otherwise, if $E$ and $F$ are Banach spaces, we will understand that they are endowed with the norm topology. Given $T\in\mathcal{L}(E;F)$, $T^*\colon F^*\to E^*$ denotes the adjoint operator of $T$.
<!-- end head-1 -->
<!-- definition-1: body_md -->
Let $U$ be an open subset of a complex Banach space $E$. The space $\mathcal{G}^\infty(U)$ is the norm-closed linear subspace of $\mathcal{H}^\infty(U)^*$ given by $\overline{\mathrm{lin}}\left\{\delta(x)\colon x\in U\right\}$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{teo1}\cite[Theorem 2.1]{Muj-91}
Let $E$ be a complex Banach space and let $U$ be an open set in $E$.
\begin{enumerate}
\item $\delta_U\colon U\to\mathcal{G}^\infty(U)$ is a holomorphic mapping with $\left\|\delta(x)\right\|=1$ for all $x\in U$.                                                                
\item For every complex Banach space $F$ and every mapping $f\in\mathcal{H}^\infty(U,F)$, there exists a unique operator $T_f\in\mathcal{L}(\mathcal{G}^\infty(U);F)$ such that $T_f\circ\delta_U=f$, that is, the diagram
$$
\begin{tikzpicture}
\node (U) {$U$};
\node (GU) [below of=U] {$\mathcal{G}^\infty(U)$};
\node (F) [right of=GU] {$F$};
\draw[->] (U) to node {$f$} (F);
\draw[->] (U) to node [swap] {$\delta_U$} (GU);
\draw[->, dashed] (GU) to node [swap] {$T_f$} (F);
\end{tikzpicture}
$$
commutes. Furthermore, $\left\|T_f\right\|=\left\|f\right\|_\infty$.
\item For every complex Banach space $F$, the mapping $J_{U,F}\colon f\mapsto T_f$ is an isometric isomorphism from $\mathcal{H}^\infty(U,F)$ onto $\mathcal{L}(\mathcal{G}^\infty(U);F)$. Its inverse $J^{-1}_{U,F}$ is the mapping $T\mapsto T\circ\delta_U$ from $\mathcal{L}(\mathcal{G}^\infty(U);F)$ onto $\mathcal{H}^\infty(U,F)$.
\item $\mathcal{H}^\infty(U)$ is isometrically isomorphic to $\mathcal{G}^\infty(U)^*$, via the mapping $J_U:=J_{U,\mathbb{C}}\colon\mathcal{H}^\infty(U)\to\mathcal{G}^\infty(U)^*$ given by
$$
J_U(f)=T_f\qquad (f\in\mathcal{H}^\infty(U)).
$$
As a consequence, we have
$$
\left\langle J_U(f),\delta_U(x)\right\rangle=f(x)\qquad (f\in\mathcal{H}^\infty(U),\; x\in U).
$$
\item The closed unit ball of $\mathcal{G}^\infty(U)$ coincides with the norm-closed absolutely convex hull of $\delta_U(U)$.$\hfill\Box$
\end{enumerate}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor-1}
Let $U$ be an open subset of a complex Banach space $E$. Let $(f_i)$ be a net in $\mathcal{H}^\infty(U)$ and $f\in\mathcal{H}^\infty(U)$.
\begin{enumerate}
	\item If $(f_i)\to f$ weak* in $\mathcal{H}^\infty(U)$, then $(f_i)\to f$ pointwise on $U$.
	\item It $(f_i)$ is bounded in $\mathcal{H}^\infty(U)$ and $(f_i)\to f$ pointwise on $U$, then $(f_i)\to f$ weak* in $\mathcal{H}^\infty(U)$.$\hfill\Box$
\end{enumerate}
<!-- end corollary-1 -->
<!-- corollary-2: body_md -->
\label{cor-3}
Let $E$ and $F$ be complex Banach spaces, let $U$ and $V$ be open subsets of $E$ and $F$, respectively, and let $f\colon U\to V$ be a holomorphic mapping. Then there exists a unique operator $\widehat{f}\in\mathcal{L}(\mathcal{G}^\infty(U);\mathcal{G}^\infty(V))$ such that $\widehat{f}\circ\delta_U=\delta_V\circ f$. Furthermore, $||\widehat{f}||=1$.
<!-- end corollary-2 -->
<!-- proof-1: body_md -->
Let $C_f\colon\mathcal{H}^\infty(V)\to\mathcal{H}^\infty(U)$ be the composition operator defined by
$$
C_f(g)=g\circ f\qquad (g\in\mathcal{H}^\infty(V)).
$$
Clearly, $C_f\in\mathcal{L}(\mathcal{H}^\infty(V);\mathcal{H}^\infty(U))$ with $||C_f||=1$. We claim that $J_U\circ C_f\circ J^{-1}_V$ is weak*-to-weak* continuous from $\mathcal{G}^\infty(V)^*$ into $\mathcal{G}^\infty(U)^*$. Let $(\phi_i)$ be a net in $\mathcal{G}^\infty(V)^*$ and $\phi\in\mathcal{G}^\infty(V)^*$. Assume that $(\phi_i)\to\phi$ weak* in $\mathcal{G}^\infty(V)^*$. By \cite[Corollary 2.6.10]{Meg-98}, $(\phi_i)$ is norm-bounded in $\mathcal{G}^\infty(V)^*$. Clearly, $(J^{-1}_V(\phi_i))\to J^{-1}_V(\phi)$ weak* in $\mathcal{H}^\infty(V)$. Hence $(J^{-1}_V(\phi_i))\to J^{-1}_V(\phi)$ pointwise on $V$ by Corollary \ref{cor-1} (1). In particular, $(J^{-1}_V(\phi_i)\circ f)\to J^{-1}_V(\phi)\circ f$ pointwise on $U$. Since the net $(J^{-1}_V(\phi_i)\circ f)$ is norm-bounded since $\left\|J^{-1}_V(\phi_i)\circ f\right\|_\infty\leq \left\|J^{-1}_V(\phi_i)\right\|_\infty=\left\|\phi_i\right\|$ for all $i$, Corollary \ref{cor-1} (2) guarantees that $(J^{-1}_V(\phi_i)\circ f)\to J^{-1}_V(\phi)\circ f$ weak* in $\mathcal{H}^\infty(U)$, that is, $(C_f(J^{-1}_V(\phi_i)))\to C_f(J^{-1}_V(\phi))$ weak* in $\mathcal{H}^\infty(U)$. Finally, $(J_U(C_f(J^{-1}_V(\phi_i))))\to J_U(C_f(J^{-1}_V(\phi)))$ weak* in $\mathcal{G}^\infty(U)^*$, and this proves our claim.

Now, by \cite[Corollaries 3.1.11 and 3.1.5]{Meg-98}, there is a unique operator $\widehat{f}\in\mathcal{L}(\mathcal{G}^\infty(U);\mathcal{G}^\infty(V))$ such that $(\widehat{f})^*=J_U\circ C_f\circ J^{-1}_V$. Clearly, $||\widehat{f}||=||(\widehat{f})^*||=||C_f||=1$. Given $g\in\mathcal{H}^\infty(V)$ and $x\in U$, we have
$$
\left\langle(\widehat{f})^*(J_V(g)),\delta_U(x)\right\rangle=\left\langle J_V(g)\circ\widehat{f},\delta_U(x)\right\rangle=\left\langle J_V(g),\widehat{f}(\delta_U(x))\right\rangle
$$
and
\begin{align*}
\left\langle (J_U\circ C_f\circ J^{-1}_V)(J_V(g)),\delta_U(x)\right\rangle&=\left\langle J_U(C_f(g)),\delta_U(x)\right\rangle
=\left\langle J_U(g\circ f),\delta_U(x)\right\rangle\\               
&=g(f(x))=\left\langle J_V(g),\delta_V(f(x))\right\rangle .
\end{align*}
From above we infer that
$$
\left\langle J_V(g),\widehat{f}(\delta_U(x))\right\rangle=\left\langle J_V(g),\delta_V(f(x))\right\rangle \qquad (g\in\mathcal{H}^\infty(V),\; x\in U),
$$
the surjectivity of $J_V\colon\mathcal{H}^\infty(V)\to\mathcal{G}^\infty(V)^*$ yields
$$
\left\langle\phi,\widehat{f}(\delta_U(x))\right\rangle=\left\langle\phi,\delta_V(f(x))\right\rangle \qquad (\phi\in\mathcal{G}^\infty(V)^*,\; x\in U),
$$
and this implies that $\widehat{f}\circ\delta_U=\delta_V\circ f$.

<!-- definition-2: body_md -->
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and $f\in\mathcal{H}^\infty(U,F)$. We will call the transpose mapping $f^t\colon F^*\to\mathcal{H}^\infty(U)$ defined by
$f^t(\varphi)=\varphi\circ f$ the transpose of f.
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop-A}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and let $f\in\mathcal{H}^\infty(U,F)$. Then $f^t\in\mathcal{L}(F^*;\mathcal{H}^\infty(U))$ with $||f^t||=\left\|f\right\|_\infty$. Furthermore, $f^t=J^{-1}_U\circ (T_f)^*$. $\hfill\Box$
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{teo-4-1}
Let $U$ be an open subset of a complex Banach space $E$ and let $F$ be a complex Banach space. The mapping $f\mapsto f^t$ is an isometric isomorphism from $\mathcal{H}^\infty(U,F)$ onto $\mathcal{L}((F^*,w^*);(\mathcal{H}^\infty(U),w^*))$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Let $f\in\mathcal{H}^\infty(U,F)$. Clearly, $f^t=J^{-1}_U\circ (T_f)^*\in\mathcal{L}((F^*,w^*);(\mathcal{H}^\infty(U),w^*))$ by Theorem \ref{teo1} (2) and \cite[Theorem 3.1.11]{Meg-98}.                                                                      
   
                                                                                                                           
   
                        
Moreover, we have $||f^t||=\left\|f\right\|_\infty$ by Proposition \ref{prop-A}. It remains to show the surjectivity of the mapping in the statement. Take $T\in\mathcal{L}((F^*,w^*);(\mathcal{H}^\infty(U),w^*))$. Then $J_U\circ T\in\mathcal{L}((F^*,w^*);(\mathcal{G}^\infty(U)^*,w^*))$                                                                                                                                                   
and, by \cite[Theorem 3.1.11]{Meg-98}, there is a $S\in\mathcal{L}(\mathcal{G}^\infty(U);F)$ such that $S^*=J_U\circ T$. By Theorem \ref{teo1} (3), there exists $f\in\mathcal{H}^\infty(U,F)$ such that $T_f=S$. Hence $T=J^{-1}_U\circ(T_f)^*=f^t$, as desired.
<!-- end proof-2 -->
<!-- theorem-2: body_md -->
\label{teo-3-0}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and let $f\in\mathcal{H}^\infty(U,F)$. The following are equivalent:
\begin{enumerate}
	\item $f\colon U\to F$ has finite rank.
	\item $T_f\colon\mathcal{G}^\infty(U)\to F$ has finite rank.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ has finite rank.
\end{enumerate}
In that case, $\mathrm{rank}(f)=\mathrm{rank}(T_f)=\mathrm{rank}((T_f)^*)=\mathrm{rank}(f^t)$.
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
Recall that $f(U)=T_f(\delta_U(U))$ by Theorem \ref{teo1} (2).

$(1)\Leftrightarrow(2)$ (\cite[Proposition 3.1 (a)]{Muj-91}): If $f$ has finite rank, then $\mathrm{lin}(f(U))$ is finite dimensional and therefore closed in $F$. We have
\begin{align*}
T_f(\mathcal{G}^\infty(U))&=T_f(\overline{\mathrm{lin}}(\delta_U(U)))\subseteq\overline{T_f(\mathrm{lin}(\delta_U(U)))}\\
                 &=\overline{\mathrm{lin}}(T_f(\delta_U(U)))=\overline{\mathrm{lin}}(f(U))=\mathrm{lin}(f(U))
\end{align*}
and hence $T_f$ has finite rank. Conversely, if $T_f$ has finite rank, then $f$ has finite rank since
\begin{align*}
\mathrm{lin}(f(U))&=\mathrm{lin}(T_f(\delta_U(U)))=T_f(\mathrm{lin}(\delta_U(U)))\\
          &\subseteq T_f(\overline{\mathrm{lin}}(\delta_U(U)))=T_f(\mathcal{G}^\infty(U)).
\end{align*}
Furthermore, in this case we have $\mathrm{rank}(f)=\mathrm{rank}(T_f)$.

$(2)\Leftrightarrow(3)$: Since the space $\mathcal{F}(E,F)$ of bounded finite-rank linear operators is a completely symmetric operator ideal (see \cite[4.4.7]{Pie-80}), we have
\begin{align*}
T_f\in\mathcal{F}(\mathcal{G}^\infty(U),F)&\Leftrightarrow (T_f)^*\in\mathcal{F}(F^*,\mathcal{G}^\infty(U)^*)\\
&\Leftrightarrow f^t=J^{-1}_U\circ(T_f)^*\in\mathcal{F}(F^*,\mathcal{H}^\infty(U)).
\end{align*}
In this case we now have $\mathrm{rank}(T_f)=\mathrm{rank}((T_f)^*)=\mathrm{rank}(f^t)$.
<!-- end proof-3 -->
<!-- theorem-3: body_md -->
\label{teo-4-2}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and let $f\in\mathcal{H}^\infty(U,F)$. The following are equivalent:
\begin{enumerate}
  \item $f\colon U\to F$ has relatively compact range.
	\item $T_f\colon\mathcal{G}^\infty(U)\to F$ is compact.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ is compact.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ is bounded-weak*-to-norm continuous.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ is compact and bounded-weak*-to-weak continuous.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ is compact and weak*-to-weak continuous.
\end{enumerate}
<!-- end theorem-3 -->
<!-- proof-4: body_md -->
$(1)\Leftrightarrow(2)$ (\cite[Proposition 3.1 (b)]{Muj-91}):                                                                   
Since $\delta_U(U)\subseteq B_{\mathcal{G}^\infty(U)}$, if $T_f\in\mathcal{K}(\mathcal{G}^\infty(U),F)$, then $T_f(\delta_U(U))$ must be relatively compact in $F$, and therefore $f\in\mathcal{H}^\infty_{\mathcal{K}}(U,F)$ since $f(U)=T_f(\delta_U(U))$ by Theorem \ref{teo1} (2). Conversely, since $B_{\mathcal{G}^\infty(U)}=\overline{\mathrm{aco}}(\delta_U(U))$ by Theorem \ref{teo1} (5), one has
$$
T_f(B_{\mathcal{G}^\infty(U)})=T_f(\overline{\mathrm{aco}}(\delta_U(U)))\subseteq\overline{\mathrm{aco}}(T_f(\delta_U(U)))\subseteq\overline{\mathrm{aco}}(\overline{T_f(\delta_U(U))}).
$$
So, if $f\in\mathcal{H}^\infty_{\mathcal{K}}(U,F)$, then $T_f(\delta_U(U))$ is relatively compact in $F$, hence $\overline{\mathrm{aco}}(\overline{T_f(\delta_U(U))})$ is compact in $F$ by Mazur's compactness theorem (\cite[Theorem 2.8.15]{Meg-98}) and the fact that $\mathrm{aco}(A)=\mathrm{co}(\mathbb{D}A)$ for any subset $A$ of a normed space $E$, where $\mathbb{D}$ denotes the closed unit disc of $\mathbb{C}$. Therefore $T_f(B_{\mathcal{G}^\infty(U)})$ is relatively compact in $F$, which means that $T_f\in\mathcal{K}(\mathcal{G}^\infty(U),F)$.

$(2)\Leftrightarrow (3)$: Applying Schauder's theorem \cite[Theorem 3.4.15]{Meg-98} and \cite[Proposition 3.4.10]{Meg-98}, we have
\begin{align*}
T_f\in\mathcal{K}(\mathcal{G}^\infty(U),F)&\Leftrightarrow (T_f)^*\in\mathcal{K}(F^*,\mathcal{G}^\infty(U)^*)\\
&\Leftrightarrow f^t=J^{-1}_U\circ(T_f)^*\in\mathcal{K}(F^*,\mathcal{H}^\infty(U)),
\end{align*}

$(2)\Leftrightarrow (4)$: Similarly, one obtains
\begin{align*}
T_f\in\mathcal{K}(\mathcal{G}^\infty(U),F)&\Leftrightarrow (T_f)^*\in\mathcal{L}((F^*,bw^*);\mathcal{G}^\infty(U)^*)\\
&\Leftrightarrow f^t=J^{-1}_U\circ(T_f)^*\in\mathcal{L}((F^*,bw^*);\mathcal{H}^\infty(U)),
\end{align*}
by \cite[Theorem 3.4.16]{Meg-98}, where $bw^*$ denotes the bounded weak* topology.

$(4)\Leftrightarrow (5)\Leftrightarrow (6)$ follows directly from \cite[Proposition 3.1]{Kim-13}.
<!-- end proof-4 -->
<!-- proposition-3: body_md -->
\label{cor-4-1}
Let $U$ be an open subset of a complex Banach space $E$ and let $F$ be a complex Banach space. The mapping $f\mapsto f^t$ is an isometric isomorphism from $\mathcal{H}_\mathcal{K}^\infty(U,F)$ onto $\mathcal{L}((F^*,bw^*);\mathcal{H}^\infty(U))$.
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Let $f\in\mathcal{H}_\mathcal{K}^\infty(U,F)$. Then $f^t\in\mathcal{L}((F^*,bw^*);\mathcal{H}^\infty(U))$ by Theorem \ref{teo-4-2} and $||f^t||=\left\|f\right\|_\infty$ by Proposition \ref{prop-A}. To prove the surjectivity, take $T\in\mathcal{L}((F^*,bw^*);\mathcal{H}^\infty(U))$. Then $J_U\circ T\in\mathcal{L}((F^*,bw^*);\mathcal{G}^\infty(U)^*)$. If $Q_{\mathcal{G}^\infty(U)}$ denotes the natural injection from $\mathcal{G}^\infty(U)$ into $\mathcal{G}^\infty(U)^{**}$, then $Q_{\mathcal{G}^\infty(U)}(\gamma)\circ J_U\circ T\in\mathcal{L}((F^*,bw^*);\mathbb{C})$ for all $\gamma\in\mathcal{G}^\infty(U)$ and, by \cite[Theorem 2.7.8]{Meg-98}, $Q_{\mathcal{G}^\infty(U)}(\gamma)\circ J_U\circ T\in\mathcal{L}((F^*,w^*);\mathbb{C})$ for all $\gamma\in\mathcal{G}^\infty(U)$, that is, $J_U\circ T\in\mathcal{L}((F^*,w^*);(\mathcal{G}^\infty(U)^*,w^*))$ by \cite[Corollary 2.4.5]{Meg-98}. Hence $J_U\circ T=S^*$ for some $S\in\mathcal{L}(\mathcal{G}^\infty(U);F)$ by \cite[Theorem 3.1.11]{Meg-98}. Note that $S^*\in\mathcal{L}((F^*,bw^*);\mathcal{G}^\infty(U)^*)$ and this means that $S\in\mathcal{K}(\mathcal{G}^\infty(U),F)$ by \cite[Theorem 3.4.16]{Meg-98}. Now, $S=T_f$ for some $f\in\mathcal{H}^\infty_\mathcal{K}(U,F)$ by Theorem \ref{teo1} (3) and Theorem \ref{teo-4-2}. Finally, we have $T=J^{-1}_U\circ S^*=J^{-1}_U\circ (T_f)^*=f^t$.
<!-- end proof-5 -->
<!-- theorem-4: body_md -->
\label{teo-3-4-1}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and let $f\in\mathcal{H}^\infty(U,F)$. Consider the following assertions:
\begin{enumerate}
	\item $f\colon U\to F$ has separable range.
	\item $T_f\colon\mathcal{G}^\infty(U)\to F$ has separable range.
	\item $(T_f)^*\colon F^*\to\mathcal{G}^\infty(U)^*$ has separable range.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ has separable range.
\end{enumerate}
Then $(1)\Leftrightarrow(2)\Leftarrow(3)\Leftrightarrow(4)$.
<!-- end theorem-4 -->
<!-- proof-6: body_md -->
$(1)\Leftrightarrow(2)$: We next use some inclusions obtained in the proof of Theorem \ref{teo-3-0}. If $f(U)$ is separable, then so is $\overline{\mathrm{lin}}(f(U))$, and since $T_f(\mathcal{G}^\infty(U))\subseteq\overline{\mathrm{lin}}(f(U))$, we deduce that $T_f(\mathcal{G}^\infty(U))$ is separable. Conversely, if $T_f(\mathcal{G}^\infty(U))$ is separable, then so is $f(U)$ since $f(U)\subseteq\mathrm{lin}(f(U))\subseteq T_f(\mathcal{G}^\infty(U))$.

$(3)\Leftrightarrow(4)$: We apply that the space $\mathcal{S}(E,F)$ of all bounded linear operators between Banach spaces having separable range is an operator ideal (see \cite[1.8.2]{Pie-80}).

$(3)\Rightarrow(2)$: We apply that if $T\in\mathcal{L}(E;F)$ and $T^*\in\mathcal{L}(F^*;E^*)$, then $T\in\mathcal{S}(E,F)$ (see \cite[4.4.8]{Pie-80}).
<!-- end proof-6 -->
<!-- proposition-4: body_md -->
Let $U$ be an open subset of a complex Banach space $E$ and let $F$ be a complex Banach space. Then $\mathcal{H}^\infty_{\overline{\mathcal{F}}}(U,F)\subseteq\mathcal{H}_\mathcal{K}^\infty(U,F)$.
<!-- end proposition-4 -->
<!-- proof-7: body_md -->
Let $f\in\mathcal{H}^\infty_{\overline{\mathcal{F}}}(U,F)$. Hence there is a sequence $(f_n)_{n\in\mathbb{N}}$ in $\mathcal{H}_\mathcal{F}^\infty(U,F)$ such that $\left\|f_n-f\right\|_\infty\to 0$ as $n\to\infty$. Since $T_{f_n}\in\mathcal{F}(G^\infty(U),F)$ by Theorem \ref{teo-3-0}, $\mathcal{F}(G^\infty(U),F)\subseteq\mathcal{K}(\mathcal{G}^\infty(U),F)$ and $\left\|T_{f_n}-T_f\right\|=\left\|T_{f_n-f}\right\|=\left\|f_n-f\right\|_\infty$ for all $n\in\mathbb{N}$, we deduce that $T_f\in\mathcal{K}(\mathcal{G}^\infty(U),F)$ by \cite[Corollary 3.4.9]{Meg-98}, and so $f\in\mathcal{H}_\mathcal{K}^\infty(U,F)$ by Theorem \ref{teo-4-2}.
<!-- end proof-7 -->
<!-- theorem-5: body_md -->
\label{cor-4-0}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and $f\in\mathcal{H}^\infty(U,F)$. The following are equivalent:
\begin{enumerate}
	\item $f\colon U\to F$ can be approximated by bounded finite-rank holomorphic mappings.
	\item $T_f\colon\mathcal{G}^\infty(U)\to F$ can be approximated by bounded finite-rank linear operators.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ can be approximated by bounded finite-rank linear operators.
\end{enumerate}
<!-- end theorem-5 -->
<!-- proof-8: body_md -->
We have
\begin{align*}
f\in\mathcal{H}^\infty_{\overline{\mathcal{F}}}(U,F)
&\Leftrightarrow T_f\in\overline{\mathcal{F}}(\mathcal{G}^\infty(U),F)\\
&\Leftrightarrow (T_f)^*\in\overline{\mathcal{F}}(F^*,\mathcal{G}^\infty(U)^*)\\
&\Leftrightarrow f^t=J^{-1}_U\circ(T_f)^*\in\overline{\mathcal{F}}(F^*,\mathcal{H}^\infty(U))
\end{align*}
by Theorem \ref{teo1} (3) and Theorem \ref{teo-3-0} for the first equivalence, Hutton’s theorem \cite[Theorem 2.1]{Hut-74} for the second one, and the operator ideal property of $\overline{\mathcal{F}}(E,F)$ for the third one \cite[4.2.2]{Pie-80}.
<!-- end proof-8 -->
<!-- theorem-6: body_md -->
\label{teo-3-2}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and let $f\in\mathcal{H}^\infty(U,F)$. The following are equivalent:
\begin{enumerate}
	\item $f\colon U\to F$  has relatively weakly compact range.
	\item $T_f\colon\mathcal{G}^\infty(U)\to F$ is weakly compact.
	\item There exist a reflexive complex Banach space $G$, an operator $T\in\mathcal{L}(G;F)$ and a mapping $g\in\mathcal{H}^\infty(U,G)$ such that $f=T\circ g$.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ is weakly compact.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ is weak*-to-weak continuous.
\end{enumerate}
<!-- end theorem-6 -->
<!-- proof-9: body_md -->
$(1)\Leftrightarrow (2)$ (\cite[Proposition 3.1 (b)]{Muj-91}): It follows with a similar proof to that of the equivalence $(1)\Leftrightarrow (2)$ of Theorem \ref{teo-4-2} by taking into account that the norm closure and weak closure of the convex hull of a subset of a normed space coincide \cite[Corollary 2.5.18]{Meg-98} and that the norm-closed convex hull of a weakly compact subset of a Banach space is itself weakly compact \cite[Theorem 2.8.14]{Meg-98}.

$(2)\Rightarrow(3)$: Applying the Davis--Figiel--Johnson--Pe\l czynski theorem, there exist a reflexive complex Banach space $G$ and operators $T\in\mathcal{L}(G;F)$ and $S\in\mathcal{L}(\mathcal{G}^\infty(U);G)$ such that $T_f=T\circ S$. Taking $g:=S\circ\delta_U\in\mathcal{H}^\infty(U,G)$, we conclude that $f=T_f\circ\delta_U=T\circ S\circ\delta_U=T\circ g$.

$(3)\Rightarrow(1)$: $f(U)=T(g(U))$ is relatively weakly compact because $T$ is weak-to-weak continuous by \cite[Theorem 2.5.11]{Meg-98} and $g(U)$ is relatively weakly compact in $G$ since it is a bounded subset of the reflexive Banach space $G$ (see \cite[Theorem 2.8.2]{Meg-98}).

$(2)\Leftrightarrow (4)$: We have
\begin{align*}
T_f\in\mathcal{W}(\mathcal{G}^\infty(U),F)&\Leftrightarrow (T_f)^*\in\mathcal{W}(F^*,\mathcal{G}^\infty(U)^*)\\
&\Leftrightarrow f^t=J^{-1}_U\circ(T_f)^*\in\mathcal{W}(F^*,\mathcal{H}^\infty(U)),
\end{align*}
by Gantmacher's theorem \cite[Theorem 3.5.13]{Meg-98} and \cite[Proposition 3.5.11]{Meg-98}.

$(2)\Leftrightarrow (5)$: We have
\begin{align*}
T_f\in\mathcal{W}(\mathcal{G}^\infty(U),F)&\Leftrightarrow (T_f)^*\in\mathcal{L}((F^*,w^*);(\mathcal{G}^\infty(U)^*,w))\\
&\Leftrightarrow f^t=J^{-1}_U\circ(T_f)^*\in\mathcal{L}((F^*,w^*);(\mathcal{H}^\infty(U),w))
\end{align*}
by Gantmacher--Nakamura's theorem \cite[Theorem 3.5.14]{Meg-98} and \cite[Corollary 2.5.12]{Meg-98}.

<!-- proposition-5: body_md -->
\label{cor-4-1b}
Let $U$ be an open subset of a complex Banach space $E$ and let $F$ be a complex Banach space. The mapping $f\mapsto f^t$ is an isometric isomorphism from $\mathcal{H}^\infty_{\mathcal{W}}(U,F)$ onto $\mathcal{L}((F^*,w^*);(\mathcal{H}^\infty(U),w))$.
<!-- end proposition-5 -->
<!-- proof-10: body_md -->
In view of Theorem \ref{teo-3-2} and Proposition \ref{prop-A}, we only need to show that the mapping in the statement is surjective. Let $T\in\mathcal{L}((F^*,w^*);(\mathcal{H}^\infty(U),w))$. Then $J_U\circ T\in\mathcal{L}((F^*,w^*);(\mathcal{G}^\infty(U)^*,w))$ by \cite[Theorem 2.5.11]{Meg-98}, and this last set is contained in $\mathcal{L}((F^*,w^*);(\mathcal{G}^\infty(U)^*,w^*))$ since the weak topology is stronger than the weak* topology on the dual of a normed space.                                              
It follows that $J_U\circ T=S^*$ for some $S\in\mathcal{L}(\mathcal{G}^\infty(U);F)$ by \cite[Theorem 3.1.11]{Meg-98}. Hence $S^*\in\mathcal{L}((F^*,w^*);(\mathcal{G}^\infty(U)^*,w))$ and, by Gantmacher--Nakamura's theorem \cite[Theorem 3.5.14]{Meg-98}, $S\in\mathcal{W}(\mathcal{G}^\infty(U),F)$. Now, $S=T_f$ for some $f\in\mathcal{H}_{\mathcal{W}}^\infty(U,F)$ by Theorem \ref{teo1} (3) and Theorem \ref{teo-3-2}. Finally, $T=J^{-1}_U\circ S^*=J^{-1}_U\circ (T_f)^*=f^t$, as desired.

<!-- theorem-7: body_md -->
\label{teo-3-3}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and let $f\in\mathcal{H}^\infty(U,F)$. The following are equivalent:
\begin{enumerate}
	\item $f\colon U\to F$ has Rosenthal range.
	\item $T_f\colon\mathcal{G}^\infty(U)\to F$ is Rosenthal.
	\item There exist a complex Banach space $G$ which does not contain an isomorphic copy of $\ell_1$, an operator $T\in\mathcal{L}(G;F)$ and a mapping $g\in\mathcal{H}^\infty(U,G)$ such that $f=T\circ g$.
\end{enumerate}
<!-- end theorem-7 -->
<!-- proof-11: body_md -->
$(1)\Leftrightarrow(2)$: It can be proved similarly as the same equivalence in Theorem \ref{teo-4-2} taking into account that the norm-closed absolutely convex hull of a Rosenthal subset of a Banach space is itself Rosenthal (see \cite[p. 357]{Lin-89}).

$(2)\Rightarrow(3)$: Rosenthal factorization theorem (see e.g. \cite{Alv-88}) gives a complex Banach space $G$ not containing an isomorphic copy of $\ell_1$ and operators $T\in\mathcal{L}(G;F)$ and $S\in\mathcal{L}(\mathcal{G}^\infty(U);G)$ such that $T_f=T\circ S$. If we write $g:=S\circ\delta_U\in\mathcal{H}^\infty(U,G)$, then $f=T_f\circ\delta_U=T\circ S\circ\delta_U=T\circ g$.

$(3)\Rightarrow(1)$: Notice that $f(U)=T(g(U))$ where $T$ is weak-to-weak continuous \cite[Theorem 2.5.11]{Meg-98} and $g(U)$ is Rosenthal in $G$ by Rosenthal's $\ell_1$-theorem \cite{Ros-74}.

<!-- theorem-8: body_md -->
\label{teo-3-4}
Let $U$ be an open subset of a complex Banach space $E$, let $F$ be a complex Banach space and let $f\in\mathcal{H}^\infty(U,F)$. The following are equivalent:
\begin{enumerate}
	\item $f\colon U\to F$ has Asplund range.
	\item $T_f\colon\mathcal{G}^\infty(U)\to F$ is Asplund.
	\item There exist a complex Asplund space $G$, an operator $T\in\mathcal{L}(G;F)$ and a mapping $g\in\mathcal{H}^\infty(U,G)$ such that $f=T\circ g$.
	\item $f^t\colon F^*\to\mathcal{H}^\infty(U)$ is Radon--Nikod\'ym.
\end{enumerate}
<!-- end theorem-8 -->
<!-- proof-12: body_md -->
$(1)\Leftrightarrow(2)$: Since $f(U)$ is an Asplund set and $T_f(B_{\mathcal{G}^\infty(U)})\subseteq\overline{\mathrm{aco}}(f(U))$, then $T_f(B_{\mathcal{G}^\infty(U)})$ is an Asplund set by Lemmas 1.3 and 1.4 in \cite{Ste-81} and Theorem 5.5.4 in \cite{Bou-83}.

$(2)\Rightarrow(3)$: If $T_f\in\mathcal{A}(\mathcal{G}^\infty(U),F)$, then Theorem 2.11 in \cite{Ste-81} assures that $T_f=T\circ S$ with $S\in\mathcal{L}(\mathcal{G}^\infty(U);G)$ and $T\in\mathcal{L}(G;F)$, where $G$ is a complex Asplund space. Taking $g:=S\circ\delta_U\in\mathcal{H}^\infty(U,G)$, we have $f=T_f\circ\delta_U=T\circ S\circ\delta_U=T\circ g$.

$(3)\Rightarrow(1)$: $f(U)=T(g(U))$ is an Asplund set by Theorem 2.8, Theorem 1.13 and Lemma 1.7 in \cite{Ste-81}.

$(2)\Rightarrow(4)$: By \cite[Theorem 2.11]{Ste-81}, \cite[24.2]{Pie-80} and Proposition \ref{prop-A}, we have
\begin{align*}
T_f\in\mathcal{A}(\mathcal{G}^\infty(U),F)&\Leftrightarrow (T_f)^*\in\mathcal{RN}(F^*,\mathcal{G}^\infty(U)^*)\\
&\Leftrightarrow f^t=J^{-1}_U\circ(T_f)^*\in\mathcal{RN}(F^*,\mathcal{H}^\infty(U)).
\end{align*}
