\label{prop:space}
Under model \eqref{eq:likelihood} with parameter of interest $\rho$ and nuisance parameters $\eta=\{\sigma_x,\sigma_y,m_x(\cdot),m_y(\cdot),f_{\bf Z}(\cdot)\}$, let $\Lambda_j$, $j=1,\dots,5$, denote the nuisance tangent space associated with the corresponding nuisance parameter component:
\begin{eqnarray*}
\Lambda_1:\sigma_x,\qquad
\Lambda_2:\sigma_y,\qquad
\Lambda_3:m_x,\qquad
\Lambda_4:m_y,\qquad
\Lambda_5:f_Z.
\end{eqnarray*}
Then the nuisance tangent space of model \eqref{eq:likelihood} is
\begin{eqnarray*}
\Lambda=\Lambda_1+\Lambda_2+\Lambda_3+\Lambda_4+\Lambda_5,
\end{eqnarray*}
where
\begin{eqnarray*}
\Lambda_1 &=& \left\{c_1\left(-\frac{1}{\sigma_x}+\frac{1}{1-\rho^2}\frac{\epsilon_x^2}{\sigma_x^3}-\frac{\rho}{1-\rho^2}\frac{\epsilon_x\epsilon_y}{\sigma_x^2\sigma_y}\right)\right\},\\
\Lambda_2 &=& \left\{c_2\left(-\frac{1}{\sigma_y}+\frac{1}{1-\rho^2}\frac{\epsilon_y^2}{\sigma_y^3}-\frac{\rho}{1-\rho^2}\frac{\epsilon_x\epsilon_y}{\sigma_x\sigma_y^2}\right)\right\},\\
\Lambda_3 &=& \left\{\frac{1}{1-\rho^2}\left(\frac{\epsilon_x}{\sigma_x^2}-\rho\frac{\epsilon_y}{\sigma_x\sigma_y}\right)B_1({\bf z})\right\},\\
\Lambda_4 &=& \left\{\frac{1}{1-\rho^2}\left(\frac{\epsilon_y}{\sigma_y^2}-\rho\frac{\epsilon_x}{\sigma_x\sigma_y}\right)B_2({\bf z})\right\},\\
\Lambda_5 &=& \{a({\bf z}):E(a)=0\}.
\end{eqnarray*}

$\Lambda$ in ${\cal H}$, where

Then

\label{prop:eif}
Under model \eqref{eq:likelihood}, the efficient score for $\rho$ is
\begin{eqnarray}\label{eq:seff}
S_{\rm eff}
= -\frac{1}{2(1-\rho^2)^2}\left(\rho\widetilde\epsilon_x^2-2\widetilde\epsilon_x\widetilde\epsilon_y+\rho\widetilde\epsilon_y^2\right).
\end{eqnarray}
Furthermore, the semiparametric efficiency bound for estimating $\rho$ is
\begin{eqnarray*}
E(S_{\rm eff}^2)^{-1} = (1-\rho^2)^2,
\end{eqnarray*}
and the efficient influence function for $\rho$ is
\begin{eqnarray}\label{eq:eif}
\phi_{\rm eff}(x,y,{\bf z})=-\frac{1}{2}\left(\rho\widetilde\epsilon_x^2-2\widetilde\epsilon_x\widetilde\epsilon_y+\rho\widetilde\epsilon_y^2\right).
\end{eqnarray}

\label{thm:rho2-if}
Under model \eqref{eq:likelihood}, suppose that the nonparametric regression estimators $\widehat m_{xk}$ and $\widehat m_{yk}$ satisfy the convergence rate assumption:
\begin{eqnarray}\label{eq:rate-cond-1}
\|\widehat m_{xk} - m_x\|_2 = o_p(n_1^{-1/4}), \qquad \|\widehat m_{yk} - m_y\|_2 = o_p(n_1^{-1/4}), \qquad k=1,2,
\end{eqnarray}
where $\|\cdot\|_2$ denotes the $L_2(P_{\bf Z})$-norm, where $P_{\bf Z}$ denotes the marginal distribution of ${\bf Z}$.
If $n_1 = n_2 = n/2$, we have
\begin{eqnarray}\label{eq:rho-if}
n^{1/2}(\widehat\rho - \rho) 
= n^{-1/2} \sum_{i=1}^n \phi_{\rm eff}(X_i,Y_i,{\bf Z}_i)
+ O_p(n^{-1/2}),
\end{eqnarray}
where $\phi_{\rm eff}$ is given by \eqref{eq:eif}.

<!-- rem-1: body_md -->
\label{rem:reg-bias}
In the case where two nonparametric regression estimators have nonvanishing asymptotic bias, the corresponding estimator of residual correlation coefficient will also have some bias. Take $\widehat\rho_2$ as an example. Suppose that the regression estimators $\widehat m_{x1}$ and $\widehat m_{y1}$ converge to the functions $m_x^*$ and $m_y^*$, which may be different from $m_x$ and $m_y$. Under similar or weaker convergence assumptions like \eqref{eq:rate-cond-1} where $m_x$ and $m_y$ are replaced by $m_x^*$ and $m_y^*$, we can similarly show that
\begin{eqnarray*}
\widehat\sigma_{x2}^2\xrightarrow{P}\sigma_x^2+\mbox{var}\{\delta_x({\bf Z})\}, \quad
\widehat\sigma_{y2}^2\xrightarrow{P}\sigma_y^2+\mbox{var}\{\delta_y({\bf Z})\}, \quad
\widehat\sigma_{xy2}\xrightarrow{P}\sigma_{xy}+\mbox{cov}\{\delta_x({\bf Z}),\delta_y({\bf Z})\},
\end{eqnarray*}
where $\delta_x = m_x^* - m_x$ and $\delta_y = m_y^* - m_y$ are corresponding biases. Thus, by Slutsky's theorem, we know that
\begin{eqnarray*}
\widehat\rho_2 = \frac{\widehat\sigma_{xy2}}{\widehat\sigma_{x2}\widehat\sigma_{y2}}
\xrightarrow{P}\frac{\sigma_{xy}+\mbox{cov}\{\delta_x({\bf Z}),\delta_y({\bf Z})\}}{[\sigma_x^2+\mbox{var}\{\delta_x({\bf Z})\}]^{1/2}[\sigma_y^2+\mbox{var}\{\delta_y({\bf Z})\}]^{1/2}}.
\end{eqnarray*}
Note that the variance terms in the denominator are always inflated by the regression bias, while the numerator can either increase or decrease according to the sign of the covariance term. Therefore, if the true correlation $\rho$ is nonzero, the regression bias is more likely to shrink the residual correlation estimator toward zero, especially when the covariance between the two regression bias terms is relatively small. On the other hand, if $\rho=0$, the regression bias may as well induce some bias due to the $\mbox{cov}\{\delta_x({\bf Z}),\delta_y({\bf Z})\}$ term. The same phenomenon also holds for $\widehat\rho_1$ and $\widehat\rho$.

\label{thm:asymp-dist}
Under model \eqref{eq:likelihood}, suppose that the nonparametric regression estimators $\widehat m_{xk}$ and $\widehat m_{yk}$ satisfy \eqref{eq:rate-cond-1}, for $k=1,2$. Let $\widehat\rho$ be defined by \eqref{eq:rho-hat-final}. If $n_1 = n_2 = n/2$, then we have
\begin{eqnarray}\label{eq:asymp-dist}
n^{1/2}(\widehat\rho-\rho)\xrightarrow{d}N\left\{0,(1-\rho^2)^2\right\}.
\end{eqnarray}

Thus, we have
\begin{eqnarray*}
E\left\{\left(\widetilde\epsilon_x^2-\rho\widetilde\epsilon_x\widetilde\epsilon_y\right)\widetilde S_\rho\right\}= -\rho + \rho =0.
\end{eqnarray*}
By symmetry, 
\begin{eqnarray*}
E\left\{\left(\widetilde\epsilon_x^2-\rho\widetilde\epsilon_x\widetilde\epsilon_y\right)\widetilde S_\rho\right\}= -\rho + \rho =0.
\end{eqnarray*}

Therefore, $\widetilde S_\rho\in\Lambda^\perp$. Also, notice that 
\begin{eqnarray*}
E\left\{\frac{\widetilde\epsilon_x^2-2\rho\widetilde\epsilon_x\widetilde\epsilon_y+\widetilde\epsilon_y^2-2(1-\rho^2)}{2(1-\rho^2)^2}\right\}
=\frac{1-2\rho^2+1-2(1-\rho^2)}{2(1-\rho^2)^2}
= 0,
\end{eqnarray*}
so 
\begin{eqnarray*}
\frac{\widetilde\epsilon_x^2-2\rho\widetilde\epsilon_x\widetilde\epsilon_y+\widetilde\epsilon_y^2-2(1-\rho^2)}{2(1-\rho^2)^2}\in\Lambda_5\subset\Lambda.
\end{eqnarray*}
Therefore, $\widetilde S_\rho = \Pi (S_\rho|\Lambda^\perp)$. Thus,
\begin{eqnarray*}
S_{\rm eff}
= \widetilde S_\rho 
&=& \frac{\rho}{1-\rho^2}-\frac{\rho}{(1-\rho^2)^2}\left(\widetilde\epsilon_x^2-2\rho\widetilde\epsilon_x\widetilde\epsilon_y+\widetilde\epsilon_y^2\right)+\frac{1}{1-\rho^2}\widetilde\epsilon_x\widetilde\epsilon_y \\
&& + \rho \frac{\widetilde\epsilon_x^2-2\rho\widetilde\epsilon_x\widetilde\epsilon_y+\widetilde\epsilon_y^2-2(1-\rho^2)}{2(1-\rho^2)^2}\\
&=& -\frac{\rho}{2(1-\rho^2)^2}\left(\widetilde\epsilon_x^2-2\rho\widetilde\epsilon_x\widetilde\epsilon_y+\widetilde\epsilon_y^2\right)+\frac{1}{1-\rho^2}\widetilde\epsilon_x\widetilde\epsilon_y \\
&=& -\frac{1}{2(1-\rho^2)^2}\left(\rho\widetilde\epsilon_x^2-2\widetilde\epsilon_x\widetilde\epsilon_y+\rho\widetilde\epsilon_y^2\right).
\end{eqnarray*}

Based on the form of $S_{\rm eff}$, we have 
\begin{eqnarray*}
E(S_{\rm eff}^2) 
&=& \frac{1}{4(1-\rho^2)^4} E\left\{\left(\rho\widetilde\epsilon_x^2-2\widetilde\epsilon_x\widetilde\epsilon_y+\rho\widetilde\epsilon_y^2\right)^2\right\}\\
&=& \frac{1}{4(1-\rho^2)^4} \left\{3\rho^2 + 2\rho^2(1+2\rho^2) +3\rho^2 - 12\rho^2 +2\rho^2(1+2\rho^2) -12\rho^2\right\}\\
&=& \frac{1}{4(1-\rho^2)^4} \left\{4-8\rho^2+4\rho^4\right\}\\
&=& \frac{1}{4(1-\rho^2)^4} \left\{4(1-\rho^2)^2\right\}\\
&=& \frac{1}{(1-\rho^2)^2}.
\end{eqnarray*}
Thus, the semiparametric efficiency bound is
\begin{eqnarray*}
E(S_{\rm eff}^2)^{-1}=(1-\rho^2)^2.
\end{eqnarray*}
The efficient influence function is
\begin{eqnarray*}
\phi_{\rm eff}(x,y,{\bf z})=[E(S_{\rm eff}^2)]^{-1}S_{\rm eff}
=-\frac{1}{2}\left(\rho\widetilde\epsilon_x^2-2\widetilde\epsilon_x\widetilde\epsilon_y+\rho\widetilde\epsilon_y^2\right).
\end{eqnarray*}

By Taylor's mean value theorem, we have
\begin{eqnarray}\label{eq:rho2-decomp}
n_2^{1/2}(\widehat\rho_2 - \rho) 
&=& n_2^{1/2} \frac{1}{\sigma_x\sigma_y}(\widehat\sigma_{xy2}-\sigma_{xy})
- n_2^{1/2} \frac{\sigma_{xy}}{2\sigma_x^3\sigma_y}(\widehat\sigma_{x2}^2-\sigma_x^2)\nonumber\\
&&- n_2^{1/2} \frac{\sigma_{xy}}{2\sigma_x\sigma_y^3}(\widehat\sigma_{y2}^2-\sigma_y^2)
+ R,
\end{eqnarray}
where
\begin{eqnarray}\label{eq:remainder}
R&=& - n_2^{1/2}\frac{1}{2\sigma_{x*}^3\sigma_{y*}}(\widehat\sigma_{xy2}-\sigma_{xy})(\widehat\sigma_{x2}^2-\sigma_x^2)
- n_2^{1/2}\frac{1}{2\sigma_{x*}\sigma_{y*}^3}(\widehat\sigma_{xy2}-\sigma_{xy})(\widehat\sigma_{y2}^2-\sigma_y^2) \nonumber\\
&& + n_2^{1/2}\frac{\sigma_{xy*}}{4\sigma_{x*}^3\sigma_{y*}^3}(\widehat\sigma_{x2}^2-\sigma_x^2)(\widehat\sigma_{y2}^2-\sigma_y^2)\nonumber\\
&& - n_2^{1/2}\frac{3\sigma_{xy*}}{8\sigma_{x*}^5\sigma_{y*}}(\widehat\sigma_{x2}^2-\sigma_x^2)^2
- n_2^{1/2}\frac{3\sigma_{xy*}}{8\sigma_{x*}\sigma_{y*}^5}(\widehat\sigma_{y2}^2-\sigma_y^2)^2,
\end{eqnarray}
for some $\sigma_{x*}^2$ between $\sigma_x^2$ and $\widehat\sigma_{x2}^2$, $\sigma_{y*}^2$ between $\sigma_y^2$ and $\widehat\sigma_{y2}^2$, and $\sigma_{xy*}$ between $\sigma_{xy}$ and $\widehat\sigma_{xy2}$.

We decompose $\widehat \sigma_{x2}^2$ into three parts as follows:
\begin{eqnarray}\label{eq:sigmax2-decomp}
\widehat \sigma_{x2}^2 - \sigma_x^2
&=& n_2^{-1} \sum_{i=n_1+1}^n\{X_i- \widehat m_{x1}({\bf Z}_i)\}^2 - \sigma_x^2 \nonumber\\
&=& n_2^{-1} \sum_{i=n_1+1}^n\{X_i- m_x({\bf Z}_i)\}^2 
 + n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}^2 \nonumber\\
&& + 2 n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\{X_i- m_x({\bf Z}_i)\}
 - \sigma_x^2\nonumber\\
&=& n_2^{-1} \sum_{i=n_1+1}^n \epsilon_{xi}^2 
 + n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}^2 \nonumber\\
&& + 2 n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\epsilon_{xi}
 - \sigma_x^2.
\end{eqnarray}
Under $\|\widehat m_{x1} - m_x\|_2 = o_p(n_1^{-1/4})$, we have
\begin{eqnarray*}
E \left[ n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}^2 | \widehat m_{x1} \right]
&=& E \left[ \{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}^2 | \widehat m_{x1} \right] \\
&=& \|\widehat m_{x1} - m_x\|_2^2 \\
&=& o_p(n_1^{-1/2}),
\end{eqnarray*}
and since the sum is nonnegative, by Markov's inequality, we have 
\begin{eqnarray*}
n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}^2 = o_p(n_1^{-1/2}).
\end{eqnarray*}
Also, note that
\begin{eqnarray*}
n_2^{-1} \sum_{i=n_1+1}^n \epsilon_{xi}^2 = \sigma_x^2 + o_p(1).
\end{eqnarray*}
Furthermore, 
\begin{eqnarray*}
E \left[ \{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}\epsilon_x | \widehat m_{x1} \right] 
= E \left[ \{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\} | \widehat m_{x1} \right] E(\epsilon_x) = 0,
\end{eqnarray*}
and under $\|\widehat m_{x1} - m_x\|_2 = o_p(n_1^{-1/4})$,
\begin{eqnarray*}
&& \mbox{var} \left[ n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\epsilon_{xi} | \widehat m_{x1} \right]\\
&=& n_2^{-1} \mbox{var} \left[ \{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}\epsilon_x | \widehat m_{x1} \right]\\
&=& n_2^{-1} E \left[\{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}^2\epsilon_x^2  | \widehat m_{x1} \right]\\
&=& n_2^{-1} E \left[\{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}^2  | \widehat m_{x1} \right] E(\epsilon_x^2)\\
&=& n_2^{-1} \|\widehat m_{x1} - m_x\|_2^2 \sigma_x^2\\
&=& o_p(n_2^{-1}n_1^{-1/2}),
\end{eqnarray*}
so by Chebyshev's inequality, we have 
\begin{eqnarray*}
n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\epsilon_{xi} 
= o_p(n_2^{-1/2}n_1^{-1/4}).
\end{eqnarray*}
Plugging back to \eqref{eq:sigmax2-decomp}, we have
\begin{eqnarray}\label{eq:sigmax2-asymp}
\widehat \sigma_{x2}^2 - \sigma_x^2
= n_2^{-1} \sum_{i=n_1+1}^n \epsilon_{xi}^2 - \sigma_x^2
+ o_p(n_1^{-1/2} + n_2^{-1/2}n_1^{-1/4}).
\end{eqnarray}
Same arguments lead to, under $\|\widehat m_{y1} - m_y\|_2 = o_p(n_1^{-1/4})$,
\begin{eqnarray}\label{eq:sigmay2-asymp}
\widehat \sigma_{y2}^2 - \sigma_y^2
= n_2^{-1} \sum_{i=n_1+1}^n \epsilon_{yi}^2 - \sigma_y^2
+ o_p(n_1^{-1/2} + n_2^{-1/2}n_1^{-1/4}).
\end{eqnarray}

We also decompose $\widehat \sigma_{xy}$ into three parts as follows:
\begin{eqnarray}\label{eq:sigmaxy-decomp}
\widehat \sigma_{xy2} - \sigma_{xy}
&=& n_2^{-1} \sum_{i=n_1+1}^n\{X_i- \widehat m_{x1}({\bf Z}_i)\}\{Y_i- \widehat m_{y1}({\bf Z}_i)\} - \sigma_{xy} \nonumber\\
&=& n_2^{-1} \sum_{i=n_1+1}^n\{X_i- m_x({\bf Z}_i)\}\{Y_i- m_y({\bf Z}_i)\} \nonumber\\
&& + n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\{m_y({\bf Z}_i) - \widehat m_{y1}({\bf Z}_i)\} \nonumber\\
&& + n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\{Y_i- m_y({\bf Z}_i)\} \nonumber\\
&& + n_2^{-1} \sum_{i=n_1+1}^n\{m_y({\bf Z}_i) - \widehat m_{y1}({\bf Z}_i)\}\{X_i- m_x({\bf Z}_i)\}
 - \sigma_{xy} \nonumber\\
&=& n_2^{-1} \sum_{i=n_1+1}^n \epsilon_{xi} \epsilon_{yi}
 + n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\{m_y({\bf Z}_i) - \widehat m_{y1}({\bf Z}_i)\} \nonumber\\
&& + n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\epsilon_{yi}
+ n_2^{-1} \sum_{i=n_1+1}^n\{m_y({\bf Z}_i) - \widehat m_{y1}({\bf Z}_i)\}\epsilon_{xi} \nonumber\\
&& - \sigma_{xy}.
\end{eqnarray}
Under $\|\widehat m_{x1} - m_x\|_2 = o_p(n_1^{-1/4})$ and $\|\widehat m_{y1} - m_y\|_2 = o_p(n_1^{-1/4})$, we have
\begin{eqnarray*}
&&  E \left[ \left| n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\{m_y({\bf Z}_i) - \widehat m_{y1}({\bf Z}_i)\} \right| | \widehat m_{x1}, \widehat m_{y1} \right]  \\
&\le&  E \left[ \left|m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\right| \left|m_y({\bf Z}) - \widehat m_{y1}({\bf Z}) \right|| \widehat m_{x1}, \widehat m_{y1} \right]  \\
&=&  \left\langle \left| \widehat m_{x1} - m_x\right|, \left|\widehat m_{y1} - m_y\right| \right\rangle_2  \\
&\le& \|\widehat m_{x1} - m_x\|_2 \|\widehat m_{y1} - m_y\|_2 \\
&=& o_p(n_1^{-1/4}) o_p(n_1^{-1/4}) \\
&=& o_p(n_1^{-1/2}).
\end{eqnarray*}
By Markov's inequality, we have 
\begin{eqnarray*}
n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\{m_y({\bf Z}_i) - \widehat m_{y1}({\bf Z}_i)\} = o_p(n_1^{-1/2}).
\end{eqnarray*}
Furthermore, 
\begin{eqnarray*}
E \left[ \{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}\epsilon_y | \widehat m_{x1} \right] 
= E \left[ \{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\} | \widehat m_{x1} \right] E(\epsilon_y) = 0,
\end{eqnarray*}
and under $\|\widehat m_{x1} - m_x\|_2 = o_p(n_1^{-1/4})$,
\begin{eqnarray*}
&& \mbox{var} \left[ n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\epsilon_{yi} | \widehat m_{x1} \right]\\
&=& n_2^{-1} \mbox{var} \left[ \{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}\epsilon_y | \widehat m_{x1} \right]\\
&=& n_2^{-1} E \left[\{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}^2\epsilon_y^2  | \widehat m_{x1} \right]\\
&=& n_2^{-1} E \left[\{m_x({\bf Z}) - \widehat m_{x1}({\bf Z})\}^2  | \widehat m_{x1} \right] E(\epsilon_y^2)\\
&=& n_2^{-1} \|\widehat m_{x1} - m_x\|_2^2 \sigma_y^2\\
&=& o_p(n_2^{-1}n_1^{-1/2}),
\end{eqnarray*}
so by Chebyshev's inequality, we have 
\begin{eqnarray*}
n_2^{-1} \sum_{i=n_1+1}^n\{m_x({\bf Z}_i) - \widehat m_{x1}({\bf Z}_i)\}\epsilon_{yi} 
= o_p(n_2^{-1/2}n_1^{-1/4}).
\end{eqnarray*}
Same arguments lead to, under $\|\widehat m_{y1} - m_y\|_2 = o_p(n_1^{-1/4})$, 
\begin{eqnarray*}
n_2^{-1} \sum_{i=n_1+1}^n\{m_y({\bf Z}_i) - \widehat m_{y1}({\bf Z}_i)\}\epsilon_{xi} 
= o_p(n_2^{-1/2}n_1^{-1/4}).
\end{eqnarray*}
Plugging back to \eqref{eq:sigmaxy-decomp}, we have
\begin{eqnarray}\label{eq:sigmaxy-asymp}
\widehat \sigma_{xy2} - \sigma_{xy}
= n_2^{-1} \sum_{i=n_1+1}^n \epsilon_{xi}\epsilon_{yi} - \sigma_{xy}
+ o_p(n_1^{-1/2} + n_2^{-1/2}n_1^{-1/4}).
\end{eqnarray}

Applying central limit theorem and Slutsky's theorem to \eqref{eq:sigmax2-asymp}, \eqref{eq:sigmay2-asymp} and \eqref{eq:sigmaxy-asymp}, if $n_1 \asymp n_2 \asymp n$, we have
\begin{eqnarray*}
\widehat \sigma_{x2}^2 - \sigma_x^2 = O_p(n^{-1/2}), \quad
\widehat \sigma_{y2}^2 - \sigma_y^2 = O_p(n^{-1/2}), \quad
\widehat \sigma_{xy2} - \sigma_{xy} = O_p(n^{-1/2}),
\end{eqnarray*}
and thus,
\begin{eqnarray*}
\widehat \sigma_{x*}^2 - \sigma_x^2 = o_p(1), \quad
\widehat \sigma_{y*}^2 - \sigma_y^2 = o_p(1), \quad
\widehat \sigma_{xy*} - \sigma_{xy} = o_p(1).
\end{eqnarray*}
Plugging back to \eqref{eq:remainder}, when $n_1 \asymp n_2 \asymp n$, we have
\begin{eqnarray*}
R
&=& - n_2^{1/2}\left\{\frac{1}{2\sigma_x^3\sigma_y}+o_p(1)\right\}O_p(n^{-1/2})O_p(n^{-1/2})\\
&&- n_2^{1/2}\left\{\frac{1}{2\sigma_x\sigma_y^3}+o_p(1)\right\}O_p(n^{-1/2})O_p(n^{-1/2}) \nonumber\\
&& + n_2^{1/2}\left\{\frac{\sigma_{xy}}{4\sigma_x^3\sigma_y^3}+o_p(1)\right\}O_p(n^{-1/2})O_p(n^{-1/2})\nonumber\\
&& - n_2^{1/2}\left\{\frac{3\sigma_{xy}}{8\sigma_x^5\sigma_y}+o_p(1)\right\}O_p(n^{-1/2})^2
- n_2^{1/2}\left\{\frac{3\sigma_{xy}}{8\sigma_x\sigma_y^5}+o_p(1)\right\}O_p(n^{-1/2})^2\\
&=& O_p(n^{-1/2}).
\end{eqnarray*}
Also, when $n_1 \asymp n_2 \asymp n$, the remainder terms in \eqref{eq:sigmax2-asymp}, \eqref{eq:sigmay2-asymp} and \eqref{eq:sigmaxy-asymp} all become $o_p(n^{-1/2})$. Therefore, plugging back into \eqref{eq:rho2-decomp}, we have
\begin{eqnarray*}
&& n_2^{1/2}(\widehat\rho_2 - \rho) \\
&=& \frac{1}{\sigma_x\sigma_y}n_2^{-1/2} \sum_{i=n_1+1}^n (\epsilon_{xi}\epsilon_{yi} - \sigma_{xy})
- \frac{\sigma_{xy}}{2\sigma_x^3\sigma_y}n_2^{-1/2} \sum_{i=n_1+1}^n (\epsilon_{xi}^2 - \sigma_x^2)\nonumber\\
&&- \frac{\sigma_{xy}}{2\sigma_x\sigma_y^3}n_2^{-1/2} \sum_{i=n_1+1}^n (\epsilon_{yi}^2 - \sigma_y^2)
+ O_p(n^{-1/2})\\
&=& n_2^{-1/2} \sum_{i=n_1+1}^n \left(\frac{\epsilon_{xi}\epsilon_{yi}}{\sigma_x\sigma_y} - \frac{\sigma_{xy}}{\sigma_x\sigma_y} 
- \frac{\sigma_{xy}\epsilon_{xi}^2}{2\sigma_x^3\sigma_y} + \frac{\sigma_{xy}}{2\sigma_x\sigma_y}
- \frac{\sigma_{xy}\epsilon_{yi}^2}{2\sigma_x\sigma_y^3} + \frac{\sigma_{xy}}{2\sigma_x\sigma_y}\right)
+ O_p(n^{-1/2})\\
&=& n_2^{-1/2} \sum_{i=n_1+1}^n \left(\widetilde\epsilon_{xi}\widetilde\epsilon_{yi}
- \frac{\rho}{2}\widetilde\epsilon_{xi}^2
- \frac{\rho}{2}\widetilde\epsilon_{yi}^2 \right)
+ O_p(n^{-1/2})\\
&=& n_2^{-1/2} \sum_{i=n_1+1}^n -\frac{1}{2}\left(\rho\widetilde\epsilon_{xi}^2-2\widetilde\epsilon_{xi}\widetilde\epsilon_{yi}+\rho\widetilde\epsilon_{yi}^2\right)
+ O_p(n^{-1/2}).
\end{eqnarray*}
Comparing with \eqref{eq:eif}, we have
\begin{eqnarray}\label{eq:rho2-if}
n_2^{1/2}(\widehat\rho_2 - \rho)
= n_2^{-1/2} \sum_{i=n_1+1}^n \phi_{\rm eff}(X_i,Y_i,{\bf Z}_i)
+ O_p(n^{-1/2}).
\end{eqnarray}


Similarly, under 
$\|\widehat m_{x2} - m_x\|_2 = o_p(n_2^{-1/4})$ and
$\|\widehat m_{y2} - m_y\|_2 = o_p(n_2^{-1/4})$
with $n_1 \asymp n_2 \asymp n$, we also have
\begin{eqnarray}\label{eq:rho1-if}
n_1^{1/2}(\widehat\rho_1 - \rho) 
= n_1^{-1/2} \sum_{i=1}^{n_1} \phi_{\rm eff}(X_i,Y_i,{\bf Z}_i)
+ O_p(n^{-1/2}).

Then, a direct application of \eqref{eq:rho2-if} and \eqref{eq:rho1-if} gives the asymptotic expansion for $\widehat\rho$ as
\begin{eqnarray*}
n^{1/2}(\widehat\rho - \rho) 
= n^{-1/2} \sum_{i=1}^n \phi_{\rm eff}(X_i,Y_i,{\bf Z}_i)
+ O_p(n^{-1/2}).
\end{eqnarray*}
