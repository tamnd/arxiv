<!-- thm-1: body_md -->
\label{MainThm}
Assume that
\begin{enumerate}[label=(\Alph*), ref={\Alph*}]
    \item \label{case:neq} $a_i b_j \ne  a_j b_i$ whenever $i \ne j$, \textbf{or}
    \item \label{case:eq} $\mathbf a = \mathbf b$, and $\mathcal V'$ has a non-singular point over $\mathbb R$ and over $\mathbb Q_p$ for each prime $p$.
\end{enumerate}
Let ${\delta} > 0$, and let $\mathcal A \subseteq [N]$ with $\# \mathcal A \geqslant {\delta} N$. Then
\[
\# \mathcal A^s \cap \mathcal V \gg_{\mathcal V, {\delta}} N^{s-k-1}.
\]
<!-- end thm-1 -->
<!-- remark-1: body_md -->
In the case $k = 3$, Wooley \cite{Woo2015} gives a suitable mean value estimate below exponent 10, using work of Br\"udern and Robert \cite{BR2015}, so one might expect ten variables to suffice for us here. However, the corresponding restriction estimate is not known below this exponent \cite{HW2022}.
For this reason, we have not been able to handle the case $s = 10$ when $k = 3$.
<!-- end remark-1 -->
<!-- ex-1: body_md -->
Suppose $(a_1,\ldots,a_s) = (b_1,\ldots,b_s)$ and $a_1, \ldots, a_{s-1} > 0$, so that 
\[
a_s = - (a_1 + \cdots + a_{s-1}) < 0.
\]
Writing $p_i  = -a_i/a_s$ $(1 \leqslant i \leqslant s-1)$, we obtain
\[
p_1 x_1^3 + \cdots + p_{s-1} x_{s-1}^3 = (p_1 x_1 + \cdots + p_{s-1} x_{s-1})^3.
\]
This is the equality case of Jensen's inequality, and can only occur if $\mathbf x$ lies on the diagonal. Then $\mathbf x$ is singular. We conclude that $\mathcal V$ contains no non-singular real solutions. As $\mathcal V(\mathbb R)$ is the diagonal in this case, we cannot hope for a density result here anyway.

\label{ARL}
Let $k \in \mathbb N$, let ${\varepsilon}, {\delta} > 0$, and let $\mathcal F: \mathbb N \to [1,\infty)$. Let $\mathcal A \subseteq [N]$ with $\# \mathcal A \geqslant {\delta} N$. Then there exist
\begin{enumerate}[(a)]
\item positive integers $M \ll_{{\varepsilon}, {\delta}, \mathcal F} 1$ and $d \leqslant M/2$,
\item for $j \in 
\{ 1, k \}$, a $(\mathcal F(M), N^j)$-irrational frequency ${\boldsymbol{\theta}}^{(j)} \in \mathbb T^{c_j}$ with $c_j \leqslant d$,
\item an arithmetic progression $\mathcal P \subseteq [N]$ with $\# \mathcal P \geqslant N/M$,
\item and decompositions
\[
1_\mathcal A = f + f_{\mathrm{unf}},
\qquad
f1_\mathcal P = f_{\mathrm{str}} + f_{\mathrm{sml}},
\]
\end{enumerate}
such that
\begin{enumerate}[(i)]
\item $f: \mathbb N \to [0,1]$,
\item $f_{\mathrm{unf}}: [N] \to [-1,1]$ with
\[
\sup_{{\alpha},{\beta}} \Big|
\sum_{n \leqslant N} f_{\mathrm{unf}}(n) e({\alpha} n^k + {\beta} n)
\Big| \leqslant \frac{N}{\mathcal F(M)},
\]
\item $f_{\mathrm{sml}}: \mathcal P \to [-2, 2]$ with $\| f_{\mathrm{sml}} \|_2 \leqslant {\varepsilon} \| 1_\mathcal P \|_2$,
\item $f_{{\mathrm{str}}}: \mathcal P \to [0,1]$ with $\| f_{\mathrm{str}} \|_1 \gg {\delta} \| 1_\mathcal P \|_1$,
\item as well as a trigonometric polynomial $F: \mathbb T^{c_1 + c_k} \to [0,1]$ of degree at most $M$ such that
\[
f_{\mathrm{str}}(n) = F({\boldsymbol{\theta}}^{(1)} n, {\boldsymbol{\theta}}^{(k)} n^k)
\qquad (n \in \mathcal P).
\]
\end{enumerate}

<!-- lem-3: body_md -->
\label{properties}
Let $f,g: [N] \to \mathbb C$, $\phi: [N] \to \mathbb T^d$, and ${\boldsymbol{{\alpha}}} \in \mathbb T^d$. Then:
\begin{enumerate}[(i)]
\item $(f+g)_\phi^{\boldsymbol{{\alpha}}} = f_\phi^{\boldsymbol{{\alpha}}} + g_\phi^{\boldsymbol{{\alpha}}}$.
\item $\langle f, g_\phi^{\boldsymbol{{\alpha}}} \rangle = \langle f_\phi^{\boldsymbol{{\alpha}}}, g \rangle$.
\item If, for each ${\boldsymbol {\varepsilon}}$, the restriction $f\vert_{B_\phi^{\boldsymbol{{\alpha}}}({\varepsilon})}$ is constant, then $f_\phi^{\boldsymbol{{\alpha}}} = f$.
\item $f - f_\phi^{\boldsymbol{{\alpha}}}$ is orthogonal to any function that is constant on every $B_\phi^{\boldsymbol{{\alpha}}}({\boldsymbol {\varepsilon}})$.
\end{enumerate}

<!-- lem-4: body_md -->
\label{RTinv}
Let $\Phi$ be a closed subset of $\mathbb T^N$, let $\mathcal I \subseteq \mathbb T$ be an interval of length ${\lambda}(\mathcal I) \leqslant 1/2$, and let $f: \mathbb N \to \mathbb C$. Then there exists $\phi \in \Phi$ such that
\[
\int_\mathbb T \Big|
\sum_{n \in [N]} f(n) 1_{{\alpha} - \mathcal I}(\phi(n)) \Big| \mathrm{d} {\alpha} \geqslant {\lambda}(\mathcal I) \| f \|_\Phi / 2.
\]
<!-- end lem-4 -->
<!-- proof-1: body_md -->
The map
\[
\mathbb T^N \to \mathbb R, \qquad
\phi \mapsto \Big| \sum_{n \in [N]} f(n) e(\phi(n)) \Big|
\]
is continuous. Hence, by compactness of $\Phi$, there exists $\phi \in \Phi$ such that
\[
\| f \|_\Phi = \Big| \sum_{n \in [N]} f(n) e(\phi(n)) \Big|.
\]
Let us choose representatives $a, b \in \mathbb R$ with $b - a = {\lambda}(\mathcal I)$ such that, viewed as subsets of $\mathbb T$,
\[
(a,b) \subseteq \mathcal I \subseteq [a,b].
\]
For $x \in \mathbb T$,
\[
\int_\mathbb T 1_{x - \mathcal I} ({\alpha}) e({\alpha}) \mathrm{d} {\alpha} = \frac{e(x-a) - e(x-b)}{2 \pi i} = \frac{e(-a) - e(-b)}{2 \pi i} e(x) .
\]
Consequently,
\begin{align*}
\Big| \sum_{n \in [N]} f(n) e(\phi(n)) \Big| 
&= \frac{2 \pi}{|e(b-a) - 1|}  \Big| \sum_{n \in [N]} f(n) \int_\mathbb T 1_{\phi(n)-\mathcal I}({\alpha}) e({\alpha}) \mathrm{d} {\alpha}
\Big|
\\ &\leqslant \frac{2 \pi}{|e(b-a) - 1|} \int_\mathbb T \Big| \sum_{n \in [N]} f(n) 1_{{\alpha} - \mathcal I}(\phi(n)) \Big| \mathrm{d} {\alpha}.
\end{align*}
The conclusion now follows from the fact that
\[
|e(b-a)-1| = 2|\sin(\pi(b-a))| \geqslant 4 \| b - a \|.
\]

<!-- lem-5: body_md -->
\label{WR}
Let $\Phi$ be a closed subset of $\mathbb T^N$ containing zero, and let ${\varepsilon} > 0$. Let $f: [N] \to \mathbb C$ be $1$-bounded, meaning that $|f| \leqslant 1$ pointwise. Then there exists $d \leqslant (4/{\varepsilon})^2$ and $\phi \in \Phi^d$ such that
\begin{equation}
\label{WRerror}
\| f - f_\phi \|_\Phi \leqslant {\varepsilon} N.
\end{equation}
<!-- end lem-5 -->
<!-- proof-2: body_md -->
We perform an iterative procedure. Stage $d \geqslant 0$ begins with $\phi \in \Phi^d$ such that
\begin{equation}
\label{EnergyIncrement}
\int_{\mathbb T^d} \| f_\phi^{\boldsymbol{{\alpha}}} \|_2^2 \mathrm{d} {\boldsymbol{{\alpha}}} \geqslant d({\varepsilon}/4)^2 N.
\end{equation}
We initialise at stage $d = 0$ with $f_0 = \mathbb E_{[N]} f$. The process terminates if $\| f - f_\phi \|_\Phi \leqslant {\varepsilon} N$.

Otherwise $\| f - f_\phi \|_\Phi > {\varepsilon} N$ and, by Lemma \ref{RTinv}, there exists $\phi' \in \Phi$ such that
\[
\int_\mathbb T |
\langle f - f_\phi, g \rangle
| \mathrm{d} {\alpha}'
> {\varepsilon} N/4,
\]
where $g = 1_{{\alpha}' + [0,1/2)} \circ \phi'$.
Thus, by the definition \eqref{BRTproj} of $f_\phi$, linearity, and the triangle inequality,
\[
\int_\mathbb T \int_{\mathbb T^d} | \langle f-f_\phi^{\boldsymbol{{\alpha}}} , g \rangle | \mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\alpha}' > {\varepsilon} N / 4.
\]

Since $g$ and $f_\phi^{\boldsymbol{{\alpha}}}$ are constant on sets of the form $B_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'} ({\boldsymbol {\varepsilon}}, {\varepsilon}')$, Lemma \ref{properties} and Cauchy--Schwarz give
\[
g_{\phi,\phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'} = g, \qquad
(f_\phi^{\boldsymbol{{\alpha}}})_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'} = f_\phi^{\boldsymbol{{\alpha}}},
\]
and
\begin{align*}
|\langle f - f_\phi^{\boldsymbol{{\alpha}}}, g \rangle| &=
|\langle f - f_\phi^{\boldsymbol{{\alpha}}}, g_{\phi,\phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'} \rangle|
= |\langle f_{\phi,\phi'}^{{\boldsymbol{{\alpha}}},{\alpha}'} - f_\phi^{\boldsymbol{{\alpha}}}, g \rangle|
\leqslant N^{1/2} \| f_{\phi,\phi'}^{{\boldsymbol{{\alpha}}},{\alpha}'} -f_\phi^{\boldsymbol{{\alpha}}} \|_2.
\end{align*}
Consequently,
\[
\int_\mathbb T \int_{\mathbb T^d} \| f_{\phi,\phi'}^{{\boldsymbol{{\alpha}}},{\alpha}'} -f_\phi^{\boldsymbol{{\alpha}}} \|_2 \mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\alpha}' > {\varepsilon} N^{1/2}/4.
\]
A further application of the Cauchy--Schwarz inequality yields
\[
\int_\mathbb T \int_{\mathbb T^d} \| f_{\phi,\phi'}^{{\boldsymbol{{\alpha}}},{\alpha}'} -f_\phi^{\boldsymbol{{\alpha}}} \|_2^2 \mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\alpha}' > ({\varepsilon}/4)^2 N.
\]

By Lemma \ref{properties},
\[
\langle f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'} - f_\phi^{\boldsymbol{{\alpha}}}, f_\phi^{\boldsymbol{{\alpha}}} \rangle =
\langle f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'} - (f_\phi^{\boldsymbol{{\alpha}}})_{\phi,\phi'}^{{\boldsymbol{{\alpha}}},{\alpha}'}, f_\phi^{\boldsymbol{{\alpha}}} \rangle = \langle f - f_\phi^{\boldsymbol{{\alpha}}}, f_\phi^{\boldsymbol{{\alpha}}} \rangle = 0.
\]
Hence, by Pythagoras's theorem,
\[
\| f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'} \|_2^2 = \| f_{\phi,\phi'}^{{\boldsymbol{{\alpha}}},{\alpha}'} -f_\phi^{\boldsymbol{{\alpha}}} \|_2^2 + \| f_\phi^{\boldsymbol{{\alpha}}} \|_2^2.
\]
Now
\[
\int_\mathbb T \int_{\mathbb T^d} \| f_{\phi,\phi'}^{{\boldsymbol{{\alpha}}},{\alpha}'} \|_2^2 \mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\alpha}' > ({\varepsilon}/4)^2 N + \int_{\mathbb T^d} \| f_\phi^{\boldsymbol{{\alpha}}} \|_2^2 \mathrm{d} {\boldsymbol{{\alpha}}},
\]
so we have \eqref{EnergyIncrement} with $f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\alpha}'}$ in place of $f_\phi^{\boldsymbol{{\alpha}}}$ and $d + 1$ in place of $d$, and we proceed to stage $d+1$.

The left hand side of \eqref{EnergyIncrement} is bounded above by $N$, so our procedure must terminate at some stage $d \leqslant (4/{\varepsilon})^2$ with a projection satisfying \eqref{WRerror}.

<!-- lem-6: body_md -->
\label{RWR}
Let $\Phi$ be a closed subset of $\mathbb T^N$, and let $\phi \in \Phi^d$. Let ${\varepsilon} > 0$, and let $f: \mathbb N \to \mathbb C$ be a $1$-bounded function. Then there exist $d' \leqslant (4/{\varepsilon})^2$ and $\phi' \in \Phi^{d'}$ such that
\begin{equation}
\label{RWRbound}
\| f- f_{\phi, \phi'} \|_\Phi \leqslant {\varepsilon} N.
\end{equation}
<!-- end lem-6 -->
<!-- proof-3: body_md -->
We carry out the energy increment process used to prove Lemma \ref{WR}, initialising with $f_\phi$ instead of $f_0$. At each stage, we query whether or not \eqref{RWRbound} holds. If not, then we increase the quantity
\begin{equation}
\label{incrementor}
\int_{\mathbb T^{d+d'}} \| f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\boldsymbol{{\alpha}}}'} \|_2^2 \mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\boldsymbol{{\alpha}}}'
\end{equation}
by $({\varepsilon}/4)^2 N$. Since \eqref{incrementor} is bounded above by $N$, this procedure must terminate within $d' \leqslant (4/{\varepsilon})^2$ many steps.
<!-- end proof-3 -->
<!-- lem-7: body_md -->
\label{SFR}
Let $\mathcal F: [1,\infty) \to [1,\infty)$ and ${\varepsilon} > 0$. Let $\Phi$ be a closed subset of $\mathbb T^{N}$ containing zero, and let $f: [N] \to \mathbb C$ be $1$-bounded. Then there exist
\[
1 \leqslant d, d' \ll_{\mathcal F,{\varepsilon}} 1, \qquad \phi \in \Phi^d, \qquad
\phi' \in \Phi^{d'}
\]
such that
\[
\| f - f_{\phi, \phi'} \|_{\Phi} \leqslant N/\mathcal F(d), \qquad
\| f_{\phi, \phi'} - f_{\phi} \|_2 \leqslant {\varepsilon} \sqrt{N}.
\]
<!-- end lem-7 -->
<!-- proof-4: body_md -->
We apply Lemma \ref{WR}, furnishing $d_1 \leqslant 16 \mathcal F(0)^2$ and $\phi \in \Phi^{d_1}$ such that
\[
\| f - f_{\phi} \|_{\Phi} \leqslant N/\mathcal F(0).
\]
With $d_2 = 16 \mathcal F(d_1)^2$, Lemma \ref{RWR} gives $\phi' \in \Phi^{d_2}$ such that
\[
\| f - f_{\phi, \phi'} \|_{\Phi} \leqslant N / \mathcal F(d_1).
\]
Iterating this procedure, at stage $j$ we obtain 
\[
d_j = 16 \mathcal F(d_1 + \cdots + d_{j-1})^2, \qquad
\phi \in \Phi^{d_1 + \cdots + d_{j-1}}, \qquad \phi' \in \Phi^{d_j}
\]
such that
\[
\| f - f_{\phi, \phi'} \|_{\Phi} \leqslant N/\mathcal F(d_1 + \cdots + d_{j-1}).
\]

By Lemma \ref{properties} and Pythagoras's theorem, the quantity
\[
\int_{\mathbb T^{d_1 + \cdots + d_j}} \| f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\boldsymbol{{\alpha}}}'} \|_2^2 \mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\boldsymbol{{\alpha}}}' 
= \int_{\mathbb T^{d_1 + \cdots + d_j}} 
(\| f_{\phi}^{{\boldsymbol{{\alpha}}}} \|_2^2 +  \| f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\boldsymbol{{\alpha}}}'} - f_{\phi}^{{\boldsymbol{{\alpha}}}} \|_2^2)
\mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\boldsymbol{{\alpha}}}'
\]
is non-decreasing at each stage of the iteration. As $N$ is an upper bound for this quantity, it follows that at some stage $j \ll_{{\varepsilon}} 1$,
\[
\int_{\mathbb T^{d_1 + \cdots + d_j}} \| f_{\phi, \phi'}^{{\boldsymbol{{\alpha}}}, {\boldsymbol{{\alpha}}}'} - f_{\phi}^{{\boldsymbol{{\alpha}}}} \|_2^2
\mathrm{d} {\boldsymbol{{\alpha}}} \mathrm{d} {\boldsymbol{{\alpha}}}' \leqslant {\varepsilon}^2 N.
\]
The result now follows from the Cauchy--Schwarz inequality.

<!-- lem-8: body_md -->
\label{SymDif}
Let $0 < {\varepsilon} \leqslant 1/2$ and ${\boldsymbol{{\gamma}}} \in \mathbb T^d$. Then
\[
\operatorname{meas}_{\mathbb T^d}
([0,{\varepsilon})^d {\Delta} ([0,{\varepsilon})^d + {\boldsymbol{{\gamma}}})) \leqslant 2 {\varepsilon}^{d-1} \sum_{j \in [d]} \| {\gamma}_j \|_\mathbb T.
\]
<!-- end lem-8 -->
<!-- proof-5: body_md -->
Let us represent each ${\gamma}_j \in \mathbb T$ by a real number ${\gamma}_j$ with $|{\gamma}_j| \leqslant 1/2$. As $0 < {\varepsilon} \leqslant 1/2$,
\[
\operatorname{meas}_{\mathbb T^d}
([0,{\varepsilon})^d {\Delta} ([0,{\varepsilon})^d + {\boldsymbol{{\gamma}}}))
=
\operatorname{meas}_{\mathbb R^d}
([0,{\varepsilon})^d {\Delta} ([0,{\varepsilon})^d + {\boldsymbol{{\gamma}}})).
\]

Next, we bound $\operatorname{meas}_{\mathbb R^d}
([0,{\varepsilon})^d \cap ([0,{\varepsilon})^d + {\boldsymbol{{\gamma}}}))$ from below. This intersection is a box whose $j^{\mathrm{th}}$ side has length $({\varepsilon} - |{\gamma}_j|)_+$ for each $j$, with the notation $x_+ = \max \{ 0, x \}$. Consequently,
\[
\operatorname{meas}_{\mathbb R^d}
([0,{\varepsilon})^d \cap ([0,{\varepsilon})^d + {\boldsymbol{{\gamma}}})) = ({\varepsilon} - |{\gamma}_1|)_+ \cdots ({\varepsilon} - |{\gamma}_d|)_+.
\]
We now induct on $d$ to show that this is bounded below by ${\varepsilon}^d - {\varepsilon}^{d-1} \displaystyle \sum_{j \in [d]} |{\gamma}_j|$. This is clear for $d=1$. For $d \geqslant 2$, we compute using the inductive hypothesis that
\begin{align*}
({\varepsilon} - |{\gamma}_1|)_+ \cdots ({\varepsilon} - |{\gamma}_d|)_+
&\geqslant ({\varepsilon} - |{\gamma}_1|)_+ \cdots ({\varepsilon} - |{\gamma}_{d-1}|)_+ {\varepsilon} 
- {\varepsilon}^{d-1} |{\gamma}_d| \\
&\geqslant {\varepsilon}^d - {\varepsilon}^{d-1} \sum_{j \in [d-1]} |{\gamma}_j| - {\varepsilon}^{d-1} |{\gamma}_d| \\
&= {\varepsilon}^d - {\varepsilon}^{d-1} \sum_{j \in [d]} |{\gamma}_j|.
\end{align*}

The upshot is that
\[
\operatorname{meas}_{\mathbb R^d}
([0,{\varepsilon})^d \cap ([0,{\varepsilon})^d + {\boldsymbol{{\gamma}}})) \geqslant {\varepsilon}^d - {\varepsilon}^{d-1} \sum_{j \in [d]} |{\gamma}_j|.
\]
The conclusion now follows from the inclusion-exclusion formula for the symmetric difference.

<!-- lem-9: body_md -->
\label{Lip}
Let $\phi: [N] \to \mathbb T^d$, and let $f: [N] \to \mathbb C$ be $1$-bounded. Then there exists a $1$-bounded and $4$-Lipschitz function $F: \mathbb T^d \to \mathbb C$ such that $f_\phi = F \circ \phi$, and if $f \geqslant 0$ then $F \geqslant 0$.
<!-- end lem-9 -->
<!-- proof-6: body_md -->
For ${\boldsymbol {\varepsilon}} \in \{ 0, 1 \}^d$ and ${\boldsymbol{{\alpha}}} \in \mathbb T^d$, define
\[
F_{\boldsymbol {\varepsilon}}({\boldsymbol{{\alpha}}}) =
\begin{cases}
        
                                                                  
   
                                                                                                                 
\mathbb E_{m\in B_\phi^{{\boldsymbol{{\alpha}}}}({\boldsymbol {\varepsilon}})}f(m),  &\text{if } B_\phi^{{\boldsymbol{{\alpha}}}}({\boldsymbol {\varepsilon}}) \neq \emptyset, \\
0, &\text{if } B_\phi^{{\boldsymbol{{\alpha}}}}({\boldsymbol {\varepsilon}}) = \emptyset.
\end{cases}
\]
Then
\begin{align*}
&\sum_{{\boldsymbol {\varepsilon}} \in \{0,1\}^d} \left( 1_{{\boldsymbol {\varepsilon}}/2 + [0,1/2)^d} * F_{\boldsymbol {\varepsilon}} \right) (\phi(n)) \\
&= \int_{\mathbb T^d} \sum_{{\boldsymbol {\varepsilon}} \in \{0,1\}^d} 1_{{\boldsymbol {\varepsilon}}/2 + [0,1/2)^d}(\phi(n)-{\boldsymbol{{\alpha}}}) F_{\boldsymbol {\varepsilon}}({\boldsymbol{{\alpha}}}) \mathrm{d} {\boldsymbol{{\alpha}}} \\
&= \int_{\mathbb T^d} \sum_{{\boldsymbol {\varepsilon}} \in \{0,1\}^d} 1_{n \in B_\phi^{\boldsymbol{{\alpha}}}({\boldsymbol {\varepsilon}})} F_{\boldsymbol {\varepsilon}}({\boldsymbol{{\alpha}}}) \mathrm{d} {\boldsymbol{{\alpha}}}
= \int_{\mathbb T^d} \mathbb E_{m \in B_\phi^{\boldsymbol{{\alpha}}}(n)} f(m) \mathrm{d} {\boldsymbol{{\alpha}}} = f_\phi(n).
\end{align*}
We define the $1$-bounded function
\[
F({\boldsymbol{\beta}}) = \sum_{{\boldsymbol {\varepsilon}} \in \{0,1\}^d} \left( 1_{{\boldsymbol {\varepsilon}}/2 + [0,1/2)^d} * F_{\boldsymbol {\varepsilon}} \right) ({\boldsymbol{\beta}})  = \sum_{{\boldsymbol {\varepsilon}} \in \{0,1\}^d} \left( 1_{[0,1/2)^d} * F_{\boldsymbol {\varepsilon}} \right) ({\boldsymbol{\beta}} - {\boldsymbol {\varepsilon}}/2),
\]
noting that $f_\phi = F \circ \phi$.

To show that $F$ is $4$-Lipschitz, it suffices to prove that if $G: \mathbb T^d \to \mathbb C$ is $1$-bounded and measurable then $1_{[0,1/2)^d} * G$ is $2^{2-d}$-Lipschitz. This follows from the triangle inequality, a change of variables, and Lemma \ref{SymDif}. Indeed, if ${\boldsymbol{\beta}}, {\boldsymbol{{\gamma}}} \in \mathbb T^d$ then
\begin{align*}
&|1_{[0,1/2)^d} * G ({\boldsymbol{\beta}} + {\boldsymbol{{\gamma}}}) - 1_{[0,1/2)^d} * G({\boldsymbol{\beta}})|
\\
&\leqslant \int_{\mathbb T^d}
|1_{[0,1/2)^d}({\boldsymbol{{\alpha}}} + {\boldsymbol{{\gamma}}}) - 1_{[0,1/2)^d}({\boldsymbol{{\alpha}}})| \mathrm{d} {\boldsymbol{{\alpha}}} \\
&= \operatorname{meas}_{\mathbb T^d}(
[0,1/2)^d \Delta
([0,1/2)^d - {\boldsymbol{{\gamma}}})) \leqslant 2^{2-d} \sum_{j \in [d]} \| {\gamma}_j \|_\mathbb T.
\end{align*}

<!-- lem-10: body_md -->
\label{trig}
Let $K \geqslant 1$ and $0 < {\varepsilon} < 1/2$, and let $F: \mathbb T^d \to \mathbb C$ be a $K$-Lipschitz function. Then there exists a $1$-bounded trigonometric polynomial $F_{\varepsilon}: \mathbb T^d \to \mathbb C$ of degree $O(K^2 d^2 {\varepsilon}^{-3})$ such that $\| F - F_{\varepsilon} \|_\infty \leqslant {\varepsilon}$, and if $0 \leqslant F \leqslant 1$ then $0 \leqslant F_{\varepsilon} \leqslant 1$.

<!-- proof-7: body_md -->
For $M \in \mathbb N$, we introduce the Fej\'er kernel
\begin{align*}
\mu_M({\boldsymbol{{\alpha}}}) &= \mu_M({\alpha}_1) \cdots \mu_M({\alpha}_d) \\
&= \sum_{\mathbf m \in \mathbb Z^d} (1-|m_1|/M)_+ \cdots (1-|m_d|/M)_+ e(\mathbf m \cdot {\boldsymbol{{\alpha}}}),
\end{align*}
where
\[
\mu_M({\alpha}) =
M^{-1}  \left( \frac{\sin M \pi {\alpha}}{\sin \pi {\alpha}} \right)^2.
\]
This has the following properties:
\[
\mu_M \geqslant 0, \qquad
\int_{\mathbb T^d} \mu_M = 1,
\]
and
\begin{equation}
\label{decay}
\mu_M({\boldsymbol{{\alpha}}}) \leqslant 4M^{-1} \| {\alpha}_j \|^{-2} \prod_{i \ne j} \mu_M({\alpha}_i) \qquad (1 \leqslant j \leqslant d).
\end{equation}

For $M = \lceil 1000 (Kd)^2 {\varepsilon}^{-3} \rceil$, we choose
\[
F_{\varepsilon}({\boldsymbol{{\alpha}}}) = F * \mu_M({\boldsymbol{{\alpha}}}) = \int_{\mathbb T^d}  F({\boldsymbol{{\alpha}}} - {\boldsymbol{\beta}}) \mu_M({\boldsymbol{\beta}}) \mathrm{d} {\boldsymbol{\beta}}.
\]
By orthogonality,
\[
F_{\varepsilon}({\boldsymbol{{\alpha}}}) = \sum_{\mathbf m \in \mathbb Z^d} (1 - |m_1|/M)_+ \cdots (1-|m_d|/M)_+ \hat F(\mathbf m) e(\mathbf m \cdot {\boldsymbol{{\alpha}}})
\]
is a trigonometric polynomial of degree at most $M$. Moreover, the definition of $F_{\varepsilon}$ as a convolution ensures that if $0 \leqslant F \leqslant 1$ then $0 \leqslant F_{\varepsilon} \leqslant 1$.

Finally, we bound
\[
F({\boldsymbol{{\alpha}}}) - F_{\varepsilon}({\boldsymbol{{\alpha}}}) = \int_{\mathbb T^d} (F({\boldsymbol{{\alpha}}}) - F({\boldsymbol{{\alpha}}} - {\boldsymbol{\beta}})) \mu_M({\boldsymbol{\beta}}) \mathrm{d} {\boldsymbol{\beta}}.
\]
Let $\eta > 0$ be a threshold given by $\eta^{-3} = KdM$.
As $F$ is $K$-Lipschitz,
\[
\Big|
\int_{\| {\boldsymbol{\beta}} \|_\infty \leqslant \eta} (F({\boldsymbol{{\alpha}}}) - F({\boldsymbol{{\alpha}}} - {\boldsymbol{\beta}})) \mu_M({\boldsymbol{\beta}}) \mathrm{d} {\boldsymbol{\beta}}
\Big| \leqslant K d\eta = (Kd)^{2/3} M^{-1/3}.
\]
By \eqref{decay},
\[
\Big|
\int_{\| {\boldsymbol{\beta}} \|_\infty > \eta} (F({\boldsymbol{{\alpha}}}) - F({\boldsymbol{{\alpha}}} - {\boldsymbol{\beta}})) \mu_M({\boldsymbol{\beta}}) \mathrm{d} {\boldsymbol{\beta}}
\Big| \leqslant 4/(M\eta^2) = 4 (Kd)^{2/3} M^{-1/3}.
\]
Therefore
\[
\| F - F_{\varepsilon} \|_\infty
\leqslant 5 (Kd)^{2/3} M^{-1/3} \leqslant {\varepsilon}.
\]

        \int_{\mathbb T^{2}}\abs{f_i(\alpha, \beta)}^{s_0 -1} \mathrm{d} \alpha \mathrm{d} \beta
        = \int_{\mathbb T^{2}}\abs{f(\alpha, \beta)}^{s_0 -1} \mathrm{d} \alpha \mathrm{d} \beta \leqslant 
        \int_{\mathbb T^{2}}\abs{f'(\alpha, \beta)}^{s_0 -1} \mathrm{d} \alpha \mathrm{d} \beta,
    \end{equation}
    so it suffices to prove the bound with $f'$ in place of $f_i$.

    The case $k=2$ is a classical result, see for instance Rogovskaya \cite{Rog1986}.
    When $3\leqslant k \leqslant 6$ the result follows from the discussion after \cite[Equation (14.26)]{Woo2019}.
    Finally, when $k\geqslant 7$, put
    \begin{equation}
        r = 1 + k - \lfloor \sqrt{2k+2} \rfloor,
        \qquad
        v = k(k+1) - \frac{k(k+1)- r(r+1)}{1 + k -r}.
    \end{equation}
    Then we have
    \begin{align}
        v
        &= k^{2}-r -2 + \frac{2k + 2}{1+k-r} 
        \\
        &= k(k-1) + \lfloor \sqrt{2k+2} \rfloor - 3 + \frac{2k + 2}{\lfloor \sqrt{2k+2} \rfloor}
        \\
        &= k(k-1) + 2\lfloor \sqrt{2k+2} \rfloor - 3 + \frac{2k + 2 - \lfloor \sqrt{2k+2} \rfloor^{2}}{\lfloor \sqrt{2k+2} \rfloor},
    \end{align}
    and since
    \begin{equation}
        \frac{2k + 2 - \lfloor \sqrt{2k+2} \rfloor^{2}}{\lfloor \sqrt{2k+2} \rfloor} \in [0, 2],
    \end{equation}
    it is clear that $v \geqslant k(k-1) + 2$. An application of \cite[Theorem 14.5]{Woo2019} now yields
    \begin{equation}
        \int_{\mathbb T^{2}}\abs{f'(\alpha, \beta)}^{v} \mathrm{d} \alpha \mathrm{d} \beta \ll N^{v-k-1 + \varepsilon}.
    \end{equation}
    It follows from \eqref{eq:s_0-large-k-def} and \eqref{eq:theta-k-def} that $v \leqslant s_0 -1$, concluding the proof of the lemma.

\begin{align*}
    {\delta} N^{s-k-1}
    &\leqslant T(\mathbf g)
    =\int_{\mathbb T^2} |\tilde \mathbf g({\alpha},{\beta})| \mathrm{d} {\alpha} \mathrm{d} {\beta} \\
    &\leqslant \| \mathbf g \|_{{\mathrm{FL}}}^{(s-t)/s} \int_{\mathbb T^2} \prod_i | \tilde{g_i}(a_i {\alpha}, b_i {\beta}) |^{t/s} \mathrm{d} {\alpha} \mathrm{d} {\beta} \\

&\int_{\mathbb T^{2}} \abs{\tilde g_i(a_i \alpha, b_i \beta)}^{t} \mathrm{d} \alpha \mathrm{d} \beta 
= 
\int_{\mathbb T^{2}} \abs{\tilde g_i(\alpha, \beta)}^{t} \mathrm{d} \alpha \mathrm{d} \beta    
\\
&\leqslant N^{(1-\zeta)/2} \int_{\mathbb T^{2}} \abs{\tilde g_i(\alpha, \beta)}^{s_0 -1} \mathrm{d} \alpha \mathrm{d} \beta    
+ \int_{\abs{\tilde g_i(\alpha, \beta)} \geqslant N^{1-\zeta}} \abs{\tilde g_i(\alpha, \beta)}^{t} \mathrm{d} \alpha \mathrm{d} \beta.
\end{align}

\[

\ll N^{s_0 -k - 2 + \varepsilon}.

    \ll N^{t -k - 1},
\end{equation}

    \int_{\mathbb T^{2}} \abs{\tilde g_i(a_i \alpha, b_i \beta)}^{t} \mathrm{d} \alpha \mathrm{d} \beta    
    \ll N^{t-k-1}.
\end{equation}

\[

\]
and so
\[
\| \mathbf g \|_{{\mathrm{FL}}} \gg {\delta}^{s/(s-t)} N^s \geqslant {\delta}^{2s} N^s.
\]

Now suppose instead that $k=2$. By \cite[Equation (3.115)]{Bou1993},
\[

    \ll N^{t -3}
    \qquad (1 \leqslant i \leqslant s).
\]
The argument now follows as above.

&\sum_{\mathbf n \in \mathcal V(\mathcal P)} F({\boldsymbol{\theta}}^{(1)} n_1, {\boldsymbol{\theta}}^{(k)} n_1^k) \cdots F({\boldsymbol{\theta}}^{(1)} n_s, {\boldsymbol{\theta}}^{(k)} n_s^k) \\
&= T(1_\mathcal P) \int_{\mathcal K(\mathbf a, d_k)} \int_{\mathcal K(\mathbf b, d_1)} \prod_{i \leqslant s} F({\boldsymbol{{\gamma}}}^{(i)}, \tilde {\boldsymbol{{\gamma}}}^{(i)}) \mathrm{d} {\boldsymbol{{\gamma}}} \mathrm{d}  \tilde {\boldsymbol{{\gamma}}}
+O(M^{O(1)} A^{-{\Omega}(1)} N^{s-k-1}).
\end{align}

<!-- proof-24: body_md -->
Expanding the left hand side into Fourier series gives
    \begin{align}
        &\sum_{\mathbf n \in \mathcal V(\mathcal P)} F({\boldsymbol{\theta}}^{(1)} n_1, {\boldsymbol{\theta}}^{(k)} n_1^k) \cdots F({\boldsymbol{\theta}}^{(1)} n_s, {\boldsymbol{\theta}}^{(k)} n_s^k) \\
        &=\sum_{\substack{{\mathbf m^{(1)}, \ldots, \mathbf m^{(s)} \in \mathbb Z^{d_1}} \\
        {\tilde \mathbf m^{(1)}, \ldots, \tilde \mathbf m^{(s)} \in \mathbb Z^{d_k}}}} \hat F(\mathbf m^{(1)}, \tilde \mathbf m^{(1)}) \cdots
        \hat F(\mathbf m^{(s)}, \tilde \mathbf m^{(s)}) \\
        &\qquad \qquad
        \cdot T(1_{\mathcal P} e_{\mathbf m^{(1)} \cdot {\boldsymbol{\theta}}^{(1)},
            \tilde \mathbf m^{(1)} \cdot {\boldsymbol{\theta}}^{(k)}}, \ldots, 1_{\mathcal P} e_{\mathbf m^{(s)} \cdot {\boldsymbol{\theta}}^{(1)},
        \tilde \mathbf m^{(s)} \cdot {\boldsymbol{\theta}}^{(k)}}).
    \end{align}
As $F$ has degree at most $M$, we note that if $\hat F(\mathbf m^{(i)}, \tilde \mathbf m^{(i)}) \ne 0$ then
\[
\| \mathbf m^{(i)} \| + \| \tilde \mathbf m^{(i)} \| \leqslant M.
\]
Observe that if $a_j \tilde \mathbf m^{(i)} \ne a_i \tilde \mathbf m^{(j)}$ then $(a_j \tilde \mathbf m^{(i)} - a_i \tilde \mathbf m^{(j)}) \cdot {\boldsymbol{\theta}}^{(k)}$ is $(A/(CM), N^k)$-irrational, and if $b_j \mathbf m^{(i)} \ne b_i \mathbf m^{(j)}$ then $(b_j \mathbf m^{(i)} - b_i \mathbf m^{(j)}) \cdot {\boldsymbol{\theta}}^{(1)}$ is $(A/(CM), N)$-irrational. Thus, by Corollary \ref{CountingEquidistribution}, we have
\begin{align}
    &\sum_{\mathbf n \in \mathcal V(\mathcal P)} F({\boldsymbol{\theta}}^{(1)} n_1, {\boldsymbol{\theta}}^{(k)} n_1^k) \cdots F({\boldsymbol{\theta}}^{(1)} n_s, {\boldsymbol{\theta}}^{(k)} n_s^k) \\
    &= \sum_{(\mathbf m, \tilde \mathbf m) \in \mathcal W} \hat F(\mathbf m^{(1)}, \tilde \mathbf m^{(1)}) \cdots
    \hat F(\mathbf m^{(s)}, \tilde \mathbf m^{(s)}) T(1_\mathcal P) 
    + O(M^{O(1)} A^{-{\Omega}(1)} N^{s-k-1}),
\end{align}
where $\mathcal W$ is the set of $(\mathbf m, \tilde \mathbf m) \in \mathbb Z^{(d_1+d_k)s}$ such that
\[
a_j \tilde \mathbf m^{(i)} = a_i \tilde \mathbf m^{(j)},
\quad
b_j \mathbf m^{(i)} = b_i \mathbf m^{(j)}
\qquad (1 \leqslant i, j \leqslant s).
\]
Finally, by Lemma \ref{ConstrainedOrth},
\begin{align}
    & \int_{\mathcal K(\mathbf a, d_k)} \int_{\mathcal K(\mathbf b, d_1)} \prod_{i \leqslant s} F({\boldsymbol{{\gamma}}}^{(i)}, \tilde {\boldsymbol{{\gamma}}}^{(i)}) \mathrm{d} {\boldsymbol{{\gamma}}} \mathrm{d} \tilde {\boldsymbol{{\gamma}}}
    \\
    &= 
    \begin{multlined}[t]
        \sum_{(\mathbf m, \tilde \mathbf m) \in \mathbb Z^{(d_1+d_k)s}} 
        \hat F(\mathbf m^{(1)}, \tilde \mathbf m^{(1)}) \cdots
        \hat F(\mathbf m^{(s)}, \tilde \mathbf m^{(s)})
        \\
        \qquad \qquad \int_{\mathcal K(\mathbf a, d_k)} \int_{\mathcal K(\mathbf b, d_1)}
        e\Bigl(
        \sum_{i \in [s]} \mathbf m^{(i)} \cdot {\boldsymbol{{\gamma}}}^{(i)} \Bigr)
        e\Bigl(
        \sum_{i \in [s]} \tilde \mathbf m^{(i)} \cdot \tilde {\boldsymbol{{\gamma}}}^{(i)} \Bigr) \mathrm{d} {\boldsymbol{{\gamma}}} \mathrm{d} \tilde {\boldsymbol{{\gamma}}}
    \end{multlined} 
 \\ & = \sum_{(\mathbf m, \tilde \mathbf m) \in \mathcal W} \hat F(\mathbf m^{(1)}, \tilde \mathbf m^{(1)}) \cdots
 \hat F(\mathbf m^{(s)}, \tilde \mathbf m^{(s)}).
\end{align}

<!-- defn-1: body_md -->
Given an $r\times m$ integer matrix $M$ and a compact abelian group $G$, we have a continuous homomorphism $G^m \to G^r$ given by ${\boldsymbol{{\gamma}}} \mapsto M {\boldsymbol{{\gamma}}}$.  The kernel $\ker_G M$ is a compact abelian group and therefore has an associated Haar probability measure $\mu_{\ker_G M}$. We note that if $f : G \to \mathbb C$ is Borel measurable then  ${\boldsymbol{{\gamma}}} \mapsto \prod_{i=1}^s f(\gamma_i)$ is Borel measurable on $G^s$, as is the restriction of this function to $\ker_G M$.
<!-- end defn-1 -->
<!-- prop-1: body_md -->
\label{compact Roth}
Let $M$ be an $r\times m$ integer matrix such that the greatest common divisor of the $r\times r$ minors is $1$. Suppose that the columns of $M$ sum to zero. For any $\delta > 0$, there exists $\eta = \eta(M, \delta) > 0$ such that for any compact abelian group $G$, if $\mathcal A \subseteq G$ is  measurable with $\mu_{G}(\mathcal A) \geqslant \delta$, then 
$$
\mu_{\ker_G M}(\mathcal A^m \cap \ker_G M) \geqslant \eta.
$$
<!-- end prop-1 -->
<!-- cor-6: body_md -->
\label{CompactRothCor}
Let $a_i, a_i'$ be integers with $a_1 + \dots + a_s = a_1' + \dots + a_s' = 0$ and $\gcd(a_1, \dots, a_s) = \gcd(a_1', \dots, a_s') = 1$.
For any $\delta > 0$, there exists $\eta = \eta(\mathbf a, \mathbf a', \delta) > 0$ such that if $f: \mathbb T^{d} \times \mathbb T^{d'} \to [0,1]$ is measurable and $\int_{\mathbb T^{d+d'}}  f \geqslant \delta$, then 
$$
\int_{\mathcal K(\mathbf a,d)} \int_{\mathcal K(\mathbf a',d')}\prod_{i=1}^s f({\boldsymbol{{\gamma}}}^{(i)}, \tilde {\boldsymbol{{\gamma}}}^{(i)}) \mathrm{d} {\boldsymbol{{\gamma}}} \mathrm{d} \tilde {\boldsymbol{{\gamma}}} \geqslant \eta.
$$
<!-- end cor-6 -->
<!-- proof-25: body_md -->
We note that the columns of the matrix \eqref{system matrix} sum to zero. Furthermore, the greatest common divisor of the $2 \times 2$ minors is
\[
\gcd(\{
a_ia_j' : 1 \leqslant i, j \leqslant s \}) = 1.
\]
Let
\[
\mathcal A = \{ ({\boldsymbol{{\gamma}}}, \tilde {\boldsymbol{{\gamma}}}) \in \mathbb T^{d+d'} : f({\boldsymbol{{\gamma}}}, \tilde {\boldsymbol{{\gamma}}}) \geqslant \delta/2 \}.
\]
Then the measure of $\mathcal A$ is at least $\delta/2$, and applying Proposition \ref{compact Roth} with $G = \mathbb T^{d+d'}$ gives
$$
\mu_{\ker_{G}M }(\mathcal A^s\cap \ker_G M) \gg_{\mathbf a, \mathbf a', \delta} 1.
$$
The corollary follows on noting that
\[
\int_{\ker_G M}\prod_{i=1}^s f({\boldsymbol{{\gamma}}}^{(i)}, \tilde {\boldsymbol{{\gamma}}}^{(i)}) \mathrm{d} \mu_{\ker_G M} \geqslant (\delta/2)^s \mu_{\ker_G M} (\mathcal A^s\cap \ker_G M),
\]
and that $\ker_G M$ is isomorphic to $K(\mathbf a,d) \times K(\mathbf a', d')$.
