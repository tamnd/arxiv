<!-- theorem-1: body_md -->
\label{thm:main}
    Let for $a,b\in\mathbb{Z}$ with $a<0<b$ denote by $\mu^{\text{stoch}}_{a,b}$ the stationary distribution of the stochastic sandpile Markov chain with uniform toppling rule on the set $[[a,b]]:=\{a,a+1,...,b-1,b\}$, where the outside is identified as the sink.
    
    \begin{enumerate}[label=(USS\arabic*),leftmargin=*]
        \item It holds that
        $$\mu^{\text{stoch}}_{a,b}(\mathbbm{1}_{[[a,b]]})=\frac{1}{2}+\frac{1}{2(|a|+b+2)},$$
        where $\mathbbm{1}_{[[a,b]]}$ denotes the full configuration with one particle at every vertex of $[[a,b]]$.\label{USS1}

        \item The stationary distribution $\mu^{\text{stoch}}_{a,b}$ is uniform on the vacant configurations. That is, for all $j\in[[a,b]]$ it holds that
        $$\mu^{\text{stoch}}_{a,b}(\mathbbm{1}_{[[a,b]]}-\delta_j)=\frac{1}{2(|a|+b+2)},$$
        where $\delta_j$ is the configuration with one particle at $j$, and $0$ everywhere else.\label{USS2}

        \item The infinite volume limit of the stationary distributions on the line exists and is equal to the Dirac measure of the full configuration on $\mathbb{Z}$, i.e.
        $$\lim_{[[a,b]]\uparrow \mathbb{Z}}\mu^{\text{stoch}}_{a,b}=\delta_{\mathbbm{1}_\mathbb{Z}}.$$
        Here, the limit denotes the uniform weak limit, that is, for all sequences $(a_n)_{n\in\mathbb{N}}\in\mathbb{Z}_{<0}^\mathbb{N}$ and $(b_n)_{n\in\mathbb{N}}\in\mathbb{Z}_{>0}^\mathbb{N}$ with
        \begin{align*}
            \lim_{n\rightarrow\infty}a_n=-\infty,&&\text{and}&&\lim_{n\rightarrow\infty}b_n=\infty,
        \end{align*}
        and all finite subsets $A\subseteq\mathbb{Z}$ it holds that
        $$\lim_{n\rightarrow\infty}\mu^{\text{stoch}}_{a_n,b_n}(\eta|_A=\mathbbm{1}_A)=1.$$\label{USS3}
        \end{enumerate}
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lemma:stabilization-well-defined}
    For all initial sandpile configurations $\eta_0:[[a,b]]\rightarrow\mathbb{Z}$, there exists almost surely an $N\in\mathbb{N}$ such that $\eta_N$ is stable at all vertices.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Assume the contrary, that is there exists some configuration $\eta_0$ which does not stabilize. Since $[[a,b]]$ is finite, there exists a vertex $v\in[[a,b]]$ which topples infinitely often. Since $L_v$ has positive probability of sending a particle to either the left or the right, both $v-1$ and $v+1$ receive an infinite amount of particles from $v$. And because in every step we choose the toppled vertex uniformly at random, also $v-1$ and $v+1$ must topple infinitely often.

    We thus see inductively that all vertices in $[[a,b]]$ must topple infinitely often, which means that also $a$ topples infinitely often. However, since toppling at $a$ has positive probability of reducing the total number of particles in the system by $1$, an infinite amount of particles is lost during stabilization. However, $\eta_0$ starts with only a finite amount of particles, thus we arrive at a contradiction.

    This means, that all $\eta_0$ must stabilize in finitely many steps almost surely.
<!-- end proof-1 -->
<!-- definition-1: body_md -->
For $n \in \mathbb{N}$ and $k \in C_n$, we define the probability of exactly one particle exiting through the left boundary upon stabilizing $\mathbbm{1}_{C_n} + \delta_k$ as:
\begin{equation*}
    \mathsf{l}_{n}(k) := \mathbb{P}\Big( (\mathbbm{1}_{C_n} + \delta_k)^\circ = \mathbbm{1}_{C_n}, \, N_L = 1, \, N_R = 0 \Big).
\end{equation*}
Analogously, we define $\mathsf{r}_{n}(k)$ as the probability that exactly one particle exits through the right boundary ($N_L = 0, N_R = 1$), and $\mathsf{b}_{n}(k)$ as the probability that exactly two particles exit the system ($N_L = 1, N_R = 1$).
<!-- end definition-1 -->
<!-- lemma-2: body_md -->
\label{lemma:sgr-recurrence}
    For $n \in \mathbb{N}$ and $k \in C_n$, the probabilities $(\mathsf{l}_{n}(k))_{n\in\mathbb{N},k\in C_n}$ satisfy the recurrence relation
    \begin{equation*}\label{eq:sgr-recurrence}
        \mathsf{l}_{n}(k) = \frac{1}{3}\bigg(\mathsf{l}_{n}(k-1) + \mathsf{l}_{n}(k+1) + \mathsf{l}_{n-k}(1)\mathsf{l}_{n}(k-1)\bigg),
    \end{equation*}
    with boundary conditions $\mathsf{l}_{n}(0) = 1$ and $\mathsf{l}_{n}(n+1) = 0$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Consider the first toppling of the unstable vertex $k$ in the configuration $\eta_0 = \mathbbm{1}_{C_n} + \delta_k$.
    With probability $1/3$, the particle is sent to $k-1$, resulting in the configuration $\mathbbm{1}_{C_n} + \delta_{k-1}$. The probability of exactly one particle subsequently exiting left is, by definition, $\mathsf{l}_{n}(k-1)$.
    Similarly, with probability $1/3$, the particle is sent to $k+1$, yielding $\mathbbm{1}_{C_n} + \delta_{k+1}$ and a left-exit probability of $\mathsf{l}_{n}(k+1)$.
    
    With probability $1/3$, the split toppling occurs, yielding the configuration $\eta_1 = \mathbbm{1}_{C_n} - \delta_k + \delta_{k-1} + \delta_{k+1}$. By the Abelian property of the stochastic sandpile model, the final stabilized configuration is independent of the order of subsequent topplings. We may therefore choose to stabilize the sub-segment $\{k+1, \dots, n\}$ first. 
    In order for the global condition $N_R = 0$ to hold, this sub-stabilization must not eject any particles to the right boundary $n+1$. Thus, the single excess particle at $k+1$ must exit the sub-segment to its left, falling into the hole at $k$. This independent sub-stabilization behaves identically to a system of size $n-k$ with a particle added at relative position $1$. This event occurs with probability $\mathsf{l}_{n-k}(1)$.
    
    Conditional on this event occurring, the hole at $k$ is filled, and the right sub-segment is fully stabilized at $\mathbbm{1}_{\{k+1, \dots, n\}}$. The remaining unstable configuration on $C_n$ is exactly $\mathbbm{1}_{C_n} + \delta_{k-1}$. The probability that the stabilization of this remaining configuration results in a single left exit is $\mathsf{l}_{n}(k-1)$. The joint probability of this split sequence is therefore $\mathsf{l}_{n-k}(1)\mathsf{l}_{n}(k-1)$. Summing the probabilities of these three mutually exclusive initial toppling outcomes yields the recurrence.
<!-- end proof-2 -->
<!-- proposition-1: body_md -->
\label{prop:sgr}
    For all $n \in \mathbb{N}$ and $k \in C_n$, the probabilities of exactly one particle exiting the left or right boundary, or exactly two particles exiting, are given respectively by:
    \begin{align*}
        \mathsf{l}_{n}(k) = \frac{(n-k+1)(n-k+2)}{(n+1)(n+2)}, &&
        \mathsf{r}_{n}(k) = \frac{k(k+1)}{(n+1)(n+2)}, &&
        \mathsf{b}_{n}(k) = \frac{2k(n-k+1)}{(n+1)(n+2)}.
    \end{align*}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
We prove this by defining the explicit sequence $$g_n(k) := \frac{(n-k+1)(n-k+2)}{(n+1)(n+2)},$$ and showing that it uniquely satisfies the recurrence relation established in Lemma \ref{lemma:sgr-recurrence}.

    We first show that the recurrence relation uniquely determines the $\mathsf{l}_{n}(k)$ by induction on the system size $n$. For the base case $n=1$, the value $\mathsf{l}_{1}(1) = \frac{1}{3}$, is determined by the boundary conditions. Assume that the probabilities are uniquely determined for all system sizes $m < n$. For system size $n$, the recurrence equation from Lemma \ref{lemma:sgr-recurrence} defines a tridiagonal linear system for the unknown variables $(\mathsf{l}_{n}(k))_{k\in C_n}$, where the coefficients depend on the uniquely determined value $\mathsf{l}_{n-k}(1)$. Because these coefficients are sub-stochastic, the underlying matrix is irreducibly diagonally dominant. Therefore, the linear system is non-singular, guaranteeing that the solution for size $n$ is unique. Thus, any sequence satisfying the recurrence and boundary conditions must be the true probability $(\mathsf{l}_{n}(k))_{k\in C_n}$.

    We will now verify that our proposed function $g_n$ satisfies the recurrence. The boundary conditions are trivially satisfied: $$g_n(0) = \frac{(n+1)(n+2)}{(n+1)(n+2)} = 1 \quad \text{ and } \quad g_n(n+1) = \frac{0}{(n+1)(n+2)} = 0.$$ 
    To show that the recurrence itself is satisfied, we substitute the function $g_n$ into the right hand side of the recurrence equation from Lemma \ref{lemma:sgr-recurrence}. For brevity set $m = n-k$. By the definition of our proposed sequence, the split-term evaluates to $$g_m(1) = \frac{(m-1+1)(m-1+2)}{(m+1)(m+2)} = \frac{m(m+1)}{(m+1)(m+2)} = \frac{m}{m+2}.$$ 
    Substituting the function $g$ into the recurrence equation of Lemma \ref{lemma:sgr-recurrence} now yields
    \begin{align*}
        \frac{1}{3} &\Big( g_n(k-1) + g_n(k+1) + g_m(1)g_n(k-1) \Big) \\
        &= \frac{1}{3(n+1)(n+2)} \bigg[ (m+2)(m+3) + m(m+1) + \left(\frac{m}{m+2}\right)(m+2)(m+3) \bigg] \\
        &= \frac{1}{3(n+1)(n+2)} \bigg[ (m^2 + 5m + 6) + (m^2 + m) + (m^2 + 3m) \bigg] \\
        &= \frac{1}{3(n+1)(n+2)} \bigg[ 3m^2 + 9m + 6 \bigg] \\
        &= \frac{3(m+1)(m+2)}{3(n+1)(n+2)} = g_n(k),
    \end{align*}
    as desired. Since $g_n(k)$ satisfies the recurrence and the solution to the recurrence equation must be unique, we obtain $\mathsf{l}_{n}(k) = g_n(k)$ for all $k\in C_n$ as well. This finishes the inductive step and thus proves the closed-form expression of the probabilities $(\mathsf{l}_{n}(k))_{n\in\mathbb{N},k\in C_n}$.

    By symmetry of the uniform toppling rules, the probability of a right exit is given by $\mathsf{r}_{n}(k) = \mathsf{l}_{n}(n-k+1)$. Substituting this into our proven formula immediately verifies the closed-form expression of right-exits. 

    Finally, observe that the probability of losing exactly two particles is $\mathsf{b}_{n}(k) = 1 - \mathsf{l}_{n}(k) - \mathsf{r}_{n}(k)$ for all $n\in\mathbb{N}$ and $k\in C_n$. Substituting the explicit expressions for $\mathsf{l}_{n}(k)$ and $\mathsf{r}_{n}(k)$ gives 
     $$\mathsf{b}_{n}(k) = \frac{2k(n-k+1)}{(n+1)(n+2)},$$
     as desired.

<!-- remark-1: body_md -->
We chose to call the problem analyzed in this section the {\bf Sandpile Gambler's Ruin}, not only due to its loose resemblance of the classical gambler's ruin problem, but because the classical gambler's ruin problem can be realized as an analogue statement for a different toppling rule.

    Consider for $n\in\mathbb{N}$ and every $i\in C_n$ the random toppling distribution $L^{\text{SRW}}_i$ given by
    \begin{align*}
        \mathbb{P}(L^{\text{SRW}}_i=\delta_{i-1})=\mathbb{P}(L^{\text{SRW}}_i=\delta_{i+1})=\frac{1}{2}.
    \end{align*}
    That is, upon toppling any vertex, exactly one particle is sent to the right or left with equal probablity. If we now stabilize the full configuration plus one additional particle with this toppling rule, the excess particle will perform a simple random walk on $C_n$ until it either leaves through the left or right side of $C_n$. Analyzing the probabilities of these two events is known as the \textbf{Classical Gambler's Ruin} problem.
<!-- end remark-1 -->
<!-- lemma-3: body_md -->
\label{lemma:transition-to-full}
    Let $n\in\mathbb{N}$ and consider the transition matrix $P_{n}$ of the stochastic sandpile Markov chain on the set $C_n$. Then:
    \begin{enumerate}
        \item For all $j\in C_n$ it holds that$$P_{n}(\mathbbm{1}_{C_n}-\delta_j,\mathbbm{1}_{C_n})=\frac{n+2}{3n}.$$
        \item The full configuration moves to itself with probability given by$$P_{n}(\mathbbm{1}_{C_n},\mathbbm{1}_{C_n})=\frac{2}{3}.$$
    \end{enumerate}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
We will first analyze the transition probabilities of the vacant configurations to the full configuration. So let $n\in\mathbb{N}$ and $j\in C_n$, and consider the transition from $(\mathbbm{1}_{C_n}-\delta_j)$ to $\mathbbm{1}_{C_n}$. Every step of the stochastic sandpile Markov chain consists of first placing a particle at a uniformly chosen location in $C_n$. If the particle is placed at vertex $j$, then the hole at $j$ is filled and we moved to the full configuration. This event contributes a probability of $1/n$ to the transition probability.

    Secondly, if the additional particle is placed at a vertex $i\in C_n$ with $i<j$, i.e. it is placed to the left side of the $0$, then we can first stabilize the configuration $(\mathbbm{1}_{C_n}-\delta_j+\delta_i)$ on the vertices $\{1,...,j-1\}.$ During this stabilization, if we want the final stable configuration to be the full configuration $\mathbbm{1}_{C_n}$, exactly one particle must leave the set $\{1,...,j-1\}$, and it must do so through the right side. This corresponds to the sandpile gambler's ruin probability $\mathsf{r}_{j-1}(i)$. We thus arrive at a contribution to the transition probability of
    $$\frac{1}{n}\sum_{i=1}^{j-1}\mathsf{r}_{j-1}(i)=\frac{1}{n}\sum_{i=1}^{j-1}\frac{i(i+1)}{3j(j+1)}=\frac{(j-1)j(j+1)}{3nj(j+1)}=\frac{j-1}{3n},$$
    where we have used the closed-form expression of $\mathsf{r}_{j-1}(i)$ as obtained in Proposition \ref{prop:sgr}.

    Finally, if the particle is placed to the right side of $j$, we obtain through analogue arguments the contribution to the transition probability given by
    $$\frac{1}{n}\sum_{i=1}^{n-j}\mathsf{l}_{n-j}(i)=\frac{1}{n}\sum_{i=1}^{j-1}\frac{(n-j-i+1)(n-j-i+2)}{3(n-j+1)(n-j+2)}=\frac{(n-j)(n-j+1)(n-j+2)}{3n(n-j+1)(n-j+2)}=\frac{n-j}{3n}.$$
    If we now combine all of these three contributions to the transition probability, we obtain
    $$P_{n}(\mathbbm{1}_{C_n}-\delta_j,\mathbbm{1}_{C_n})=\frac{1}{n}+\frac{j-1}{3n}+\frac{n-j}{3n}=\frac{n+2}{3n}.$$

    Let us now consider the transition probability of the full configuration to itself. For this event to occur, if a particle is placed at some vertex $i\in C_n$, we must then sent out exactly one particle during the stabilization of the configuration $(\mathbbm{1}_{C_n}+\delta_i)$, either through the left or the right. The probability of this event is again given by the sandpile gambler's ruin as
    $$\mathsf{l}_{n}(i)+\mathsf{r}_{n}(i)=1-\mathsf{b}_{n}(i).$$
    Using now again the closed-form expressions obtain in Proposition \ref{prop:sgr} we obtain
    $$P_{n}(\mathbbm{1}_{C_n},\mathbbm{1}_{C_n})=\frac{1}{n}\sum_{i=1}^n\bigg(1-\mathsf{b}_{n}(i)\bigg)=1-\frac{1}{n}\sum_{i=1}^n\frac{2i(n-i+1)}{(n+1)(n+2)}=1-\frac{1}{3}=\frac{2}{3}.$$

<!-- proposition-2: body_md -->
\label{prop:concentration-on-full}
    For all $n\in\mathbb{N}$ it holds that
    $$\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})=\frac{1}{2}+\frac{1}{2(n+1)}.$$
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Let $n\in\mathbb{N}$. By the definition of stationary measures we obtain
    \begin{align}\label{eq:stat-of-full}
    \mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})=\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})P_{n}(\mathbbm{1}_{C_n},\mathbbm{1}_{C_n})+\sum_{i=1}^n\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_i)P_{n}(\mathbbm{1}_{C_n}-\delta_i,\mathbbm{1}_{C_n}).
    \end{align}
    Using Lemma \ref{lemma:transition-to-full} we have
    \begin{align*}
        \sum_{i=1}^n\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_i)P_{n}(\mathbbm{1}_{C_n}&-\delta_i,\mathbbm{1}_{C_n})=\sum_{i=1}^n\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_i)\frac{n+2}{3n}\\&=\frac{n+2}{3n}\sum_{i=1}^n\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_i)=\frac{n+2}{3n}\big(1-\mu^{\text{stoch}}_{n}({\mathbbm{1}_{C_n}})\big).
    \end{align*}
    Using this in Equation (\ref{eq:stat-of-full}) together with the value of $P_{n}(\mathbbm{1}_{C_n},\mathbbm{1}_{C_n})$ from Lemma \ref{lemma:transition-to-full} we obtain
    $$\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})=\frac{n+2}{3n}+\frac{n-2}{3n}\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}).$$
    After rearranging this equation we then see
    $$\frac{2n+2}{3n}\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})=\frac{n+2}{3n},$$
    from which we finally arrive at
    $$\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})=\frac{1}{2}+\frac{1}{2(n+1)},$$
    which completes our proof.

<!-- definition-2: body_md -->
For $n \in \mathbb{N}$ and $i, j \in C_n$, we define $\mathsf{h}_n^j(i)$ as the probability that stabilizing $\mathbbm{1}_{C_n} + \delta_i$ results in a single hole located exactly at $j$:
    \begin{equation*}
        \mathsf{h}_n^j(i) := \mathbb{P}\Big( (\mathbbm{1}_{C_n} + \delta_i)^\circ = \mathbbm{1}_{C_n} - \delta_j \Big).
    \end{equation*}

<!-- lemma-4: body_md -->
\label{lemma:hole-recurrence}
    For $n \ge 2$, the hole probabilities satisfy the following relations. 
    For $i < n$, we have the bulk recurrence:
    \begin{equation*}
        \mathsf{h}_n^j(i) = \mathsf{r}_{n-1}(i)\mathsf{h}_n^j(n) + \sum_{x=1}^{j-1} \mathsf{h}_{n-1}^x(i)\mathsf{h}_{n-x}^{j-x}(n-x) + \mathsf{h}_{n-1}^j(i)\mathsf{r}_{n-j}(n-j).
    \end{equation*}
    For $i=n$, we have the boundary recurrence:
    \begin{equation*}
        \mathsf{h}_n^j(n) = 
        \begin{cases}
            \frac{1}{3}\mathsf{h}_n^j(n-1) + \frac{1}{3}\mathsf{h}_{n-1}^j(n-1) & \text{if } j < n, \\
            \frac{1}{3}\mathsf{h}_n^n(n-1) + \frac{1}{3}\mathsf{l}_{n-1}(n-1) & \text{if } j = n.
        \end{cases}
    \end{equation*}
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
We prove the bulk recurrence by first stabilizing the sub-segment $\{1, \dots, n-1\}$.
    If exactly one particle exits this sub-segment to the left, the configuration becomes $\mathbbm{1}_{C_n}$ and no hole is formed. If one particle exits to the right, it lands on $n$, yielding the configuration $\mathbbm{1}_{C_n} + \delta_n$, which subsequently forms a hole at $j$ with probability $\mathsf{h}_n^j(n)$. Here, the event to send the particle to vertex $n$ has probabilty $\mathsf{r}_{n-1}(i).$
    
    If two particles exit the sub-segment, a hole is formed at some $x \in \{1, \dots, n-1\}$ and the right-exiting particle lands on $n$. This event contributes a probability of $\mathsf{h}_{n-1}^x(i)$. We must now stabilize the configuration on the sub-segment $\{x+1, \dots, n\}$. 
    Crucially, if $x > j$, the hole at $x$ acts as an absorbing boundary. The stabilization of the right segment can emit at most one particle to the left; if it does, it fills the hole at $x$, resulting in the fully occupied configuration $\mathbbm{1}_{C_n}$, with potentially a hole at some vertex in the set $\{x+1,\dots, n\}$. Thus, instability cannot propagate leftward past $x$ to create a hole at $j < x$.
    
    If $x < j$, the right segment acts as a system of size $n-x$ and must independently form a hole at relative position $j-x$, giving the sum over $x < j$. If $x = j$, the hole is already in the correct position, so the right segment must simply eject its excess particle to the right without filling $j$, which contributes a probability of $\mathsf{r}_{n-j}(n-j)$. Summing these mutually exclusive trajectories yields the bulk recurrence equation.
    
    To prove the boundary recurrence equation, we examine the first toppling at $n$. A right topple exits the system, stabilizing to $\mathbbm{1}_{C_n}$, which has no hole. A left topple moves the particle to $n-1$, yielding a remaining probability of $\mathsf{h}_n^j(n-1)$ to form a hole at the desired location $j$. A split toppling creates a hole at $n$ and an excess particle at $n-1$. For $j < n$, stabilizing $\{1, \dots, n-1\}$ must then create a hole at $j$ and eject a particle to the right to fill $n$, which is an event with probability $\mathsf{h}_{n-1}^j(n-1)$. For $j=n$, the sub-segment must not eject a particle right, so it must exit left, yielding the probability $\mathsf{l}_{n-1}(n-1)$.

    If we again sum all of the mutually exclusive probabilities that can occur as explained above, we obtain thus the boundary recurrence equation.

<!-- proposition-3: body_md -->
\label{prop:hole-uniformity}
    For all $n \in \mathbb{N}$ and $i, j \in C_n$, the probability of a hole forming at $j$ is independent of $j$ and is given by:
    \begin{equation*}
        \mathsf{h}_n^j(i) = \frac{\mathsf{b}_{n}(i)}{n} = \frac{2i(n-i+1)}{n(n+1)(n+2)}.
    \end{equation*}
<!-- end proposition-3 -->
<!-- proof-7: body_md -->
We prove this by showing that $g_n^j(i) := \frac{\mathsf{b}_{n}(i)}{n}$ uniquely satisfies the recurrence relations of Lemma \ref{lemma:hole-recurrence}.
    
     We first show that the boundary and bulk recurrence relations uniquely determine the entire sequence of probabilities. We prove this by induction on $n$. The base case $n=1$ is uniquely determined by the toppling rules. Assume the sequence is uniquely determined for all system sizes $m < n$. For system size $n$, evaluating the bulk recurrence at $n-1$ yields $$\mathsf{h}_n^j(n-1) = \mathsf{r}_{n-1}(n-1)\mathsf{h}_n^j(n) + K_j,$$ where 
     $$K_j:=\sum_{x=1}^{j-1} \mathsf{h}_{n-1}^x(n-1)\mathsf{h}_{n-x}^{j-x}(n-x) + \mathsf{h}_{n-1}^j(n-1)\mathsf{r}_{n-j}(n-j),$$
     depends only on systems of size $m < n$ and the location of the hole $j$, and is therefore fixed. Substituting this into the boundary recurrence yields a linear equation of the form
    $$ \mathsf{h}_n^j(n) = \frac{1}{3} \Big( \mathsf{r}_{n-1}(n-1)\mathsf{h}_n^j(n) + K_j \Big) + C_j, $$
    where
    $$j<n:C_j:=\frac{1}{3}\mathsf{h}_{n-1}^j(n-1),\hspace{1cm}C_n:=\frac{1}{3}\mathsf{l}_{n-1}(n-1),$$
    is another fixed constant independent of the hole probabilities of system size $n$. Because the coefficient $\mathsf{r}_{n-1}(n-1)/3$ is strictly less than $1$, this equation has a strictly unique scalar solution for $\mathsf{h}_n^j(n)$. With the boundary value uniquely fixed, the bulk recurrence explicitly and uniquely determines the remaining values $\mathsf{h}_n^j(i)$ for $i < n$. Thus, if $g_n^j(i)$ satisfies the recurrences, then $g_n^j(i)= \mathsf{h}_n^j(i)$ for all $n\in\mathbb{N}$ and all $i,j\in C_n$.
    
    We will now verify that our proposed function $(g_n^j(i))_{i,j\in C_n}$ satisfies the boundary recurrence. The left-hand side of the equation simplifies to
    $$g_n^j(n) = \frac{\mathsf{b}_{n}(n)}{n} = \frac{2n}{n(n+1)(n+2)} = \frac{2}{(n+1)(n+2)}.$$
    Now, calculating the right-hand side for $j < n$ gives
    \begin{align*}
        \frac{1}{3} \bigg( g_n^j(n-1) + g_{n-1}^j(n-1)\bigg) &= \frac{1}{3} \bigg(\frac{4(n-1)}{n(n+1)(n+2)} + \frac{2(n-1)}{(n-1)n(n+1)}\bigg)\\ &= \frac{1}{3} \frac{4(n-1) + 2(n+2)}{n(n+1)(n+2)}  = \frac{1}{3} \frac{6n}{n(n+1)(n+2)}  = \frac{2}{(n+1)(n+2)}. 
    \end{align*} 
    Similarly, the right-hand side for $j = n$ gives
    \begin{align*}
        \frac{1}{3} \bigg( g_n^n(n-1) + \mathsf{l}_{n-1}(n-1)\bigg) &= \frac{1}{3} \bigg(\frac{4(n-1)}{n(n+1)(n+2)} + \frac{2}{n(n+1)}\bigg)\\ &= \frac{2}{(n+1)(n+2)}. 
    \end{align*} 
    This shows that $(g_n^j(i))_{i,j\in C_n}$ satisfies the boundary recurrence.
    
    Next, we will verify that our proposed function also satisfies the bulk recurrence. We substitute $g$ into the right-hand side for $i < n$. Notice that by definition, $g_{n-1}^x(i)$ does not depend on $x$. Thus, we can factor it out of the sum $S_j := \sum_{x=1}^{j-1} g_{n-1}^x(i)g_{n-x}^{j-x}(n-x)$ to obtain
    \begin{align*}
        S_j &= \frac{\mathsf{b}_{n-1}(i)}{n-1} \sum_{x=1}^{j-1} \frac{2(n-x)}{(n-x)(n-x+1)(n-x+2)} = \frac{\mathsf{b}_{n-1}(i)}{n-1} \sum_{x=1}^{j-1} \left( \frac{2}{n-x+1} - \frac{2}{n-x+2} \right) \\&= \frac{\mathsf{b}_{n-1}(i)}{n-1} \left( \frac{2}{n-j+2} - \frac{2}{n+1} \right),
    \end{align*}
    Combining $S_j$ with $g_{n-1}^j(i)\mathsf{r}_{n-j}(n-j)$, gives
    \begin{align*}
        S_j + g_{n-1}^j(i)\mathsf{r}_{n-j}(n-j) &= \frac{\mathsf{b}_{n-1}(i)}{n-1} \left[ \left( \frac{2}{n-j+2} - \frac{2}{n+1} \right) + \frac{n-j}{n-j+2} \right] \\
        &= \frac{\mathsf{b}_{n-1}(i)}{n-1} \left[ \frac{n-j+2}{n-j+2} - \frac{2}{n+1} \right] = \frac{\mathsf{b}_{n-1}(i)}{n-1} \frac{n-1}{n+1} = \frac{2i(n-i)}{n(n+1)^2}.
    \end{align*}
    Finally, adding $\mathsf{r}_{n-1}(i)g_n^j(n)$ we see that the right-hand side of the bulk recurrence simplifies to
    \begin{align*}
        \frac{i(i+1)}{n(n+1)}   \frac{2}{(n+1)(n+2)}  &+ \frac{2i(n-i)}{n(n+1)^2}
        = \frac{2i}{n(n+1)^2(n+2)} \Big[ i + 1 + (n-i)(n+2) \Big] \\
        &= \frac{2i}{n(n+1)^2(n+2)} \Big[ n^2 + 2n + 1 - i(n+1) \Big] \\
        &= \frac{2i(n+1)(n-i+1)}{n(n+1)^2(n+2)} = \frac{2i(n-i+1)}{n(n+1)(n+2)} = g_n^j(i),
    \end{align*}
    as desired.
    Since $(g_n^j(i))_{i,j\in C_n}$ satisfies both recurrences, and the solution to the recurrences is unique, we obtain $\mathsf{h}_n^j(i) = g_n^j(i)$ for all $n\in\mathbb{N}$ and all $i,j\in C_n$, concluding the proof.

<!-- corollary-1: body_md -->
\label{cor:hole-transitions}
    Let $P_{n}$ denote the transition matrix of the stochastic sandpile Markov chain on the recurrent configurations of $C_n$ for $n\in\mathbb{N}$. The probabilities of transitioning to the vacant configurations are given by
    \begin{align*}
        &P_{n}(\mathbbm{1}_{C_n}, \mathbbm{1}_{C_n} - \delta_i) = \frac{1}{3n} \quad &&\text{for all } i \in C_n,  \\
        &P_{n}(\mathbbm{1}_{C_n} - \delta_j, \mathbbm{1}_{C_n} - \delta_i) = \frac{1}{3n} \quad &&\text{for } i \neq j,  \\
        &P_{n}(\mathbbm{1}_{C_n} - \delta_i, \mathbbm{1}_{C_n} - \delta_i) = \frac{n-1}{3n} &&\text{for all }i\in C_n.
    \end{align*}
<!-- end corollary-1 -->
<!-- proof-8: body_md -->
We will go through the three different transition probabilities of Corollary \ref{cor:hole-transitions}.
    
    Firstly, observe for $i\in C_n$ that the Markov chain transitions from the sandpile $\mathbbm{1}_{C_n}$ to $\mathbbm{1}_{C_n} - \delta_i$, if we add a particle uniformly at random at some site $x \in C_n$ and the subsequent stabilization forms a hole at $i$. That is, 
    $$ P_{n}(\mathbbm{1}_{C_n}, \mathbbm{1}_{C_n} - \delta_y) = \frac{1}{n} \sum_{x=1}^n \mathsf{h}_n^i(x) = \frac{1}{n} \sum_{x=1}^n \frac{\mathsf{b}_{n}(x)}{n},$$
    where the second equality follows from Proposition \ref{prop:hole-uniformity}.
    Recall from the proof of Lemma \ref{lemma:transition-to-full} that $\sum_{x=1}^n \mathsf{b}_{n}(x) = n/3$. Thus, the transition probability simplifies to $$ P_{n}(\mathbbm{1}_{C_n}, \mathbbm{1}_{C_n} - \delta_i)=\frac{1}{n^2}  \frac{n}{3}  = \frac{1}{3n},$$
    as desired.
    Secondly, suppose we start at $\mathbbm{1}_{C_n} - \delta_j$ and transition to a hole at $i < j$ for $i,j\in C_n$. This requires adding a particle at some $x < j$, and that the sub-segment $\{1, \dots, j-1\}$ loses two particles during stabilization to form a new hole at $i$. That is, 
                                                                                                                                                    
    $$ P_{n}(\mathbbm{1}_{C_n} - \delta_j, \mathbbm{1}_{C_n} - \delta_i) = \frac{1}{n} \sum_{x=1}^{j-1} \mathsf{h}_{j-1}^i(x) = \frac{1}{n} \sum_{x=1}^{j-1} \frac{\mathsf{b}_{j-1}(x)}{j-1} = \frac{1}{n(j-1)} \frac{j-1}{3} = \frac{1}{3n}, $$
    where we again used Proposition \ref{prop:hole-uniformity} and the identity $\sum_{x=1}^n \mathsf{b}_{n}(x) = n/3$.
    The case $i > j$ can be treated analogously.
    
    Finally, for all $i\in C_n$, the hole remains at $i$ if we add a particle at $x < i$ and the left sub-segment sends one particle to the left, or if we add a particle at $x > i$ and the right sub-segment exits one particle to the right. Summing these contributions yields
    $$ P_{n}(\mathbbm{1}_{C_n} - \delta_i, \mathbbm{1}_{C_n} - \delta_i) = \frac{1}{n} \left( \sum_{x=1}^{i-1} \mathsf{l}_{i-1}(x) + \sum_{x=i+1}^n \mathsf{r}_{n-i}(x-i) \right). $$
    Using the identity $\sum_{x=1}^m \mathsf{l}_{m}(x) = \frac{m}{3}$, this evaluates to
    $$ P_{n}(\mathbbm{1}_{C_n} - \delta_i, \mathbbm{1}_{C_n} - \delta_i)= \frac{1}{n} \left( \frac{i-1}{3} + \frac{n-i}{3} \right) = \frac{n-1}{3n}, $$
    which completes the proof.

<!-- proposition-4: body_md -->
\label{prop:vacant-stat}
    Let $n\in\mathbb{N}$ and consider the stationary distribution $\mu^{\text{stoch}}_{n}$ of the stochastic sandpile Markov chain on $C_n$. For all $v\in C_n$ it holds that
    $$\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)=\frac{1}{2(n+1)},$$
    that is, the stationary distribution is uniform when restricted to the vacant configurations.
<!-- end proposition-4 -->
<!-- proof-9: body_md -->
Let $n\in\mathbb{N}$ and $v\in C_n$. Similarly  to the proof of Proposition \ref{prop:concentration-on-full} we use the definition of stationarity to obtain
    \begin{equation}
    \label{eq:stat-vacant}
    \begin{aligned}
        \mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)&=\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})P_{n}(\mathbbm{1}_{C_n},\mathbbm{1}_{C_n}-\delta_v)\\&+\sum_{i=1}^n\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_i)P_{n}(\mathbbm{1}_{C_n}-\delta_i,\mathbbm{1}_{C_n}-\delta_v).
    \end{aligned}
    \end{equation}
    From Corollary \ref{cor:hole-transitions} together with Proposition \ref{prop:concentration-on-full} we obtain
    $$\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n})P_{n}(\mathbbm{1}_{C_n},\mathbbm{1}_{C_n}-\delta_v)=\bigg(\frac{1}{2}+\frac{1}{2(n+1)}\bigg)\frac{1}{3n}.$$
    Using Corollary \ref{cor:hole-transitions} we also obtain
    \begin{align*}
    \sum_{\substack{i=1\\ i\neq v}}^{v-1}\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_i)P_{n}(\mathbbm{1}_{C_n}-\delta_i,\mathbbm{1}_{C_n}-\delta_v)&=\frac{1}{3n}\sum_{\substack{i=1\\ i\neq v}}^n\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_i)\\&=\frac{1}{3n}\bigg(\frac{1}{2}-\frac{1}{2(n+1)}-\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)\bigg),
    \end{align*}
    as well as
    $$\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)P_{n}(\mathbbm{1}_{C_n}-\delta_v,\mathbbm{1}_{C_n}-\delta_v)=\frac{n-1}{3n}\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v).$$
    Reinserting this in Equation (\ref{eq:stat-vacant}) we obtain the equation
    \begin{align*}
        \mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)=\frac{1}{3n}+\frac{n-2}{3n}\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v),
    \end{align*}
    which we can solve for $\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)$ to obtain
    $$(2n+2)\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)=1\Rightarrow\mu^{\text{stoch}}_{n}(\mathbbm{1}_{C_n}-\delta_v)=\frac{1}{2(n+1)},$$
    which concludes the proof.
