<!-- theorem--1: body_md -->
Consider neural networks ${\mathcal{N}}_{\bm{w}}$ belonging to a class of depth-2 neural networks. For the PINN empirical risk corresponding to the $(d+1)$-dimensional Navier–Stokes equations, denoted by $\hat{R}({\mathcal{N}}_{\bm{w}}, {\mathcal{S}}_n)$, where ${\mathcal{S}}_n$ represents the training set, and the corresponding population risk $R({\mathcal{N}}_{\bm{w}})$, the data-averaged worst case generalization error --- defined as the difference between the empirical and population risks --- can be bounded as,
    \begin{align}
        &\mathbb{E}_{{\mathcal{S}}_n} \left[\sup_{{\mathcal{N}}_{\bm{w}}} \left( \hat{R}({\mathcal{N}}_{\bm{w}}, {\mathcal{S}}_n) - R({\mathcal{N}}_{\bm{w}}) \right)\right] \leq \frac{C_r}{\sqrt{N_r}} + \frac{C_0}{\sqrt{N_0}},
    \end{align}
    where $N_r$ denotes the number of training data points in the domain and $N_0$ the number of boundary points corresponding to the initial condition. The constants $C_r$ and $C_0$ can be expressed explicitly in terms of the Lipschitz constant of the loss function, the bounds on certain norms of the weights, the viscosity coefficient, the regularization on the loss terms enforcing the initial-condition and the incompressibility condition, and the bounds on the moments of the data distribution. The constant $C_r$ grows linearly with the viscosity coefficient.

        \ell_{H,\delta} (x) \coloneqq

\label{def:nn}
    Let $\sigma : \mathbb R \to \mathbb R$, the input ${\bm{z}}=({\bm{x}},t) \in \mathbb R^{d+1}$, the weights ${\bm{W}} \in \mathbb R^{p \times (d+1)}$, where ${\bm{W}} = [ {\bm{w}}_1, {\bm{w}}_2, \dots, {\bm{w}}_p ]^\top$, ${\bm{w}}_q \in \mathbb R^{d+1}$ for $q\in\{1,\dots p\}$, and the frozen second layer weights ${\bm{A}}_1\in\mathbb R^{d\times p}$ and ${\bm{a}}_2\in\mathbb R^p$. Then we can define a neural network ${\mathcal{N}}_{\bm{w}} : [0,1]^{d+1} \rightarrow \mathbb R^{d+1}$ whose output coordinates are denoted as mapping, 
    \[ ({\bm{x}},t) = {\bm{z}} \mapsto ({\mathcal{N}}_{{\bm{w}},u}({\bm{z}}), {\mathcal{N}}_{{\bm{w}},p}({\bm{z}}) )\]
    \[\text{where,}\] 
    \begin{align*}
        {\mathcal{N}}_{{\bm{w}},u}({\bm{z}}) \coloneqq {\bm{A}}_1\sigma({\bm{W}} {\bm{z}}),  ~{\mathcal{N}}_{{\bm{w}},p}({\bm{z}}) \coloneqq \left\langle{\bm{a}}_2,\sigma({\bm{W}} {\bm{z}})\right\rangle
    \end{align*}
    \[\text{and}  ~\sigma({\bm{W}} {\bm{z}}) = [\sigma({\bm{w}}^\top_1 {\bm{z}}), \sigma({\bm{w}}^\top_2 {\bm{z}}),\dots,\sigma({\bm{w}}^\top_p {\bm{z}})]^\top\].

\label{def:erisk}

<!-- definition-5: body_md -->
\label{def:lossclass}
    We can define two separate loss functions, $\ell_{res}$ and $\ell_{0}$ s.t they measure the error at any space-time interior point and a spatial point, respectively, as follows,
    \begin{align*}
        \ell_{res}({\mathcal{N}}_{\bm{w}}, ({\bm{x}}_{ri},t_{ri})) &\coloneqq \sum_{k=1}^d\ell \left( (\partial_t {\mathcal{N}}_{{\bm{w}},u})_k + (({\mathcal{N}}_{{\bm{w}},u} \cdot\nabla){\mathcal{N}}_{{\bm{w}},u})_k + (\nabla {\mathcal{N}}_{{\bm{w}},p})_k - \nu (\nabla^2 {\mathcal{N}}_{{\bm{w}},u})_k \right )\mid_{({\bm{x}}_{ri},t_{ri})} \\
        &\hspace{2.00em}+ \lambda_0 \cdot \ell \left  (\grad \cdot {\mathcal{N}}_{{\bm{w}},u} \right )\mid_{({\bm{x}}_{ri},t_{ri})}\\
        \ell_{0}({\mathcal{N}}_{\bm{w}}, {\bm{x}}_{0j}) &\coloneqq \lambda_1 \sum_{k=1}^d \ell \left ( ({\mathcal{N}}_{{\bm{w}},u}({\bm{x}},0))_k - ({\bm{f}}_0 ({\bm{x}}))_k \right )\mid_{{\bm{x}}_{0j}}
    \end{align*}
    and thus the empirical risk can we written as,
    \begin{align}\label{eq:genpinn}
        \hat{R}({\mathcal{N}}_{\bm{w}}) \coloneqq &\frac{1}{N_r}\sum_{i=1}^{N_r} \ell_{res}({\mathcal{N}}_{\bm{w}}, ({\bm{x}}_{ri},t_{ri}))  + \frac{1}{N_0}\sum_{j=1}^{N_0} \ell_{0}({\mathcal{N}}_{\bm{w}}, {\bm{x}}_{0j}).
    \end{align}

<!-- definition-6: body_md -->
\label{def:Wspace}
    Given weights ${\bm{W}} \in \mathbb R^{p \times (d+1)}$, let 
    ${\bm{w}}_t = {\bm{W}}_{:,\,d+1}$, ${\bm{w}}_{{\bm{x}}_m} = {\bm{W}}_{:,\,m}$, and ${\bm{a}}_{1k} = {\bm{A}}_{1k,:}$ for $m, k \in \{1, \dots, d\}$. Define ${\bm{w}}_{\bm{x}} = \sum_{m=1}^d {\bm{w}}_{{\bm{x}}_m}$ and ${\bm{a}}_1 = \sum_{k=1}^d {\bm{a}}_{1k}$. 
    
    We can then define the following functions: \begin{inparaenum}[\bf (i)]
        \item $f_1({\bm{W}})={\bm{a}}_1\odot{\bm{w}}_t$, 
        \item $f_{2m}({\bm{W}})={\bm{a}}_{1}\odot{\bm{w}}_{{\bm{x}}_m}$,
        \item $f_3({\bm{W}})={\bm{a}}_2\odot{\bm{w}}_{\bm{x}}$,
        \item $f_4({\bm{W}})=\sum_{m=1}^d{\bm{a}}_1\odot{\bm{w}}_{{\bm{x}}_m}\odot{\bm{w}}_{{\bm{x}}_m}$, and
        \item $f_5({\bm{W}})=\sum_{m=1}^d{\bm{a}}_{1m}\odot{\bm{w}}_{{\bm{x}}_m}$,
    \end{inparaenum}
    where ${\bm{a}}_{1m}$, ${\bm{a}}_1$, ${\bm{a}}_2$, ${\bm{w}}_{{\bm{x}}_m}$, ${\bm{w}}_t$, and ${\bm{w}}_{{\bm{x}}}$ are as given above.
                                        
    Now, given constants $B_{f_i}$ for $i \in [5]$ by the following constraints are defined,
    \begin{gather}\label{eq:fbounds}
        \sum_{q=1}^p \abs{f_s({\bm{W}})_q} \leq B_{f_s} \textrm{ for $s \in \{1,3,4,5\}$}\ ; \qquad
        \sum_{m=1}^d\sum_{q_1,q_2 = 1}^p \abs{f_{2m}({\bm{W}})_{q_1}} \abs{a_{1m,q_2}} \leq B_{f_2},
    \end{gather}
    where $f_1$, $f_{2m}$, $f_3$, $f_4$ and $f_5$ are the functions defined above.
                                      
    Corresponding to the above, we define the class of weights ${\mathcal{C}}$ as,
    \begin{align*}
        {\mathcal{C}} \coloneq \big\{{\bm{W}} \in \mathbb R^{p \times (d+1)}\ \big| {\bm{W}} \text{ satisfies the bounds given in Equation \ref{eq:fbounds}} \big\}.
    \end{align*}
    
    We further define $\textrm{RS}({\mathcal{C}})$ as the set of possible row vectors from the elements of ${\mathcal{C}}$.

<!-- theorem-1: body_md -->
\label{thm:NSRad}
    Let ${\mathcal{N}}_{{\bm{w}}}$ be a neural network as in Definition~\ref{def:nn}, with weights ${\bm{W}} \in {\mathcal{C}}$, where the class of weights ${\mathcal{C}}$ is defined in Definition~\ref{def:Wspace}. We make the following assumptions about the activation function $\sigma(x)$:
    \begin{inparaenum}[(i)]
    \item $\sigma(x)$ is twice differentiable,
    \item $\sigma(x)$, $\sigma'(x)$, and $\sigma''(x)$ are Lipschitz continuous with constants $L_{\sigma}$, $L_{\sigma'}$, and $L_{\sigma''}$, respectively, and
    \item $\sigma(x)$ and $\sigma'(x)$ are bounded by constants $B_{\sigma}$ and $B_{\sigma'}$, respectively.
    \end{inparaenum}
    Let the two components of the loss function, $\ell_{res}$ and $\ell_0$, be defined as in Definition~\ref{def:lossclass}. From Definition~\ref{def:collocation}, we recall the constants $C_{\bm{z}}$ and $C_{{\bm{z}}_0}$, which characterize the distributions of the collocation points in the interior and on the initial-condition boundaries, respectively. We can then bound the generalization error of the empirical loss by
    \begin{align}
        &\mathbb{E}_{\substack{({\bm{x}}_{ri},t_{ri}) \sim {\mathcal{D}}_1 \forall i \in [N_r] \\ {\bm{x}}_{0j} \sim {\mathcal{D}}_2 \forall j\in[N_0]}} \Bigg[\sup_{{\bm{W}} \in {\mathcal{C}}} \biggl( \frac{1}{N_r} \sum_{i=i}^{N_r} \ell_{res}({\mathcal{N}}_{\bm{w}},({\bm{x}}_{ri},t_{ri})) + \frac{1}{N_0} \sum_{j=i}^{N_0} \ell_{0}({\mathcal{N}}_{\bm{w}},{\bm{x}}_{0j})\nonumber\\ 
        &\hspace{15.00em}- \left( \mathbb{E}_{({\bm{x}}_r,t_r) \sim {\mathcal{D}}_1}[\ell_{res}({\mathcal{N}}_{{\bm{w}}},({\bm{x}}_{r},t_{r}))] + \mathbb{E}_{{\bm{x}}_0 \sim {\mathcal{D}}_2}[\ell_{0}({\mathcal{N}}_{{\bm{w}}},{\bm{x}}_{0})] \right) \biggr)\Bigg] \nonumber\\
        &\hspace{1.00em}\leq \frac{2 \delta (B_{{\bm{w}}} C_{\bm{z}} C_1 + C_2)}{\sqrt{N_r}} + \frac{4 \lambda_1 \delta B_{\bm{a}} (B_{{\bm{w}}} C_{{\bm{z}}_0} L_\sigma + \abs{c_0})}{\sqrt{N_0}},
    \end{align}
    
    here, $C_1 \coloneqq 2B_{f_1}L_{\sigma'}+4B_{f_2}(B_\sigma L_{\sigma'}+B_{\sigma'}L_\sigma)+2B_{f_3}L_{\sigma'}+2\nu B_{f_4} L_{\sigma''}+2 \lambda_0 B_{f_5}L_{\sigma'}$ and $C_2 \coloneqq  B_{f_1}\abs{c_1} + B_{f_2}(2B_{\sigma'}\abs{c_0} + \abs{c_0 c_1}) + B_{f_3}\abs{c_1} + \nu B_{f_4}\abs{c_2} + \lambda_0 B_{f_5}\abs{c_1}$, where $B_{f_s}$ for $s \in [5]$ are as defined in Definition~\ref{def:Wspace}, $c_0 = \sigma(0)$, $c_1 = \sigma'(0)$, $c_2 = \sigma''(0)$, and $\delta$ is the Lipschitz constant of the loss function $\ell_{H,\delta}$ as in Definition~\ref{def:huber}. We also assume that there exist constants $B_{{\bm{w}}}$ and $B_{\bm{a}}$ such that $\norm{{\bm{w}}_q}_2 \leq B_{{\bm{w}}}, \forall q \in [p]$, where ${\bm{w}}_q = {\bm{W}}_{q,:}$, and $\sum_{q=1}^p \abs{({\bm{a}}_1)_q} \leq B_{\bm{a}}$, for ${\bm{a}}_1$ is as in Definition~\ref{def:nn}.

<!-- lemma-1: body_md -->
\label{lem:NSRad}
    Continuing within the framework of Theorem~\ref{thm:NSRad}, it follows that a way to ensure that the RHS is ${\mathcal{O}}(\epsilon)$ the number of collocation points $N_r$ and initial condition points $N_0$ can be chosen as,
    \begin{align*}
        N_r &= {\mathcal{O}}\left(\frac{\delta^2 (B_{\bm{w}} C_{\bm{z}} C_1 + C_2)^2}{\epsilon^2}\right)\\

    \mathbb{E}_{\epsilon}\left[\sup_{\bar{{\bm{w}}}_1 \in \mathrm{RS}(C)} \sum_{i=1}^{N_r} \epsilon_i\, \phi_1(\langle \bar{{\bm{w}}}_1, {\bm{z}}_i\rangle)\phi_2(\langle {\bm{w}}_2^*, {\bm{z}}_i\rangle)\right] \leq 2 B_{\phi_2} L_{\phi_1}\,\mathbb{E}_{\epsilon}\left[\sup_{\bar{{\bm{w}}} \in \mathrm{RS}(C)} \sum_{i=1}^{N_r} \epsilon_i \langle \bar{{\bm{w}}}, {\bm{z}}_i\rangle\right] + B_{\phi_2}|c_0|\sqrt{N_r}.
\end{align*}

\medskip
\noindent\textbf{$\bullet$ (Combining)} Substituting into \eqref{eq:split} the end results of the above two analysis blocks we get,
 
\begin{align*}
    \mathbb{E}_{\epsilon}\left[\sup_{\bar{{\bm{w}}}_1, \bar{{\bm{w}}}_2 \in \mathrm{RS}(C)} \sum_{i=1}^{N_r} \epsilon_i\, \phi_1(\langle \bar{{\bm{w}}}_1, {\bm{z}}_i\rangle)\phi_2(\langle \bar{{\bm{w}}}_2, {\bm{z}}_i\rangle)\right]
    \leq 2(B_{\phi_1} L_{\phi_2} + B_{\phi_2} L_{\phi_1})\, \mathbb{E}_{\epsilon}\left[\sup_{\bar{{\bm{w}}} \in \mathrm{RS}(C)} \sum_{i=1}^{N_r} \epsilon_i \langle \bar{{\bm{w}}}, {\bm{z}}_i\rangle\right] + B_{\phi_2}|c_0|\sqrt{N_r}.
\end{align*}
 
Substituting the above back into \eqref{eq:after_lemma5} we get the desired bound,
\begin{align}
    &\mathbb{E}_{\epsilon \sim \{\pm 1\}^{N_r}}\left[\sup_{W \in C} \left(\frac{1}{N_r}\sum_{i=1}^{N_r} \sum_{m=1}^{d} \epsilon_i \langle f_m(W), \phi_1(W{\bm{z}}_i)\rangle \langle a_m, \phi_2(W{\bm{z}}_i)\rangle \right)\right] \notag \\
    &\leq \frac{B}{N_r}\left\{4(B_{\phi_1} L_{\phi_2} + B_{\phi_2} L_{\phi_1})\, \mathbb{E}_{\epsilon}\left[\sup_{\bar{{\bm{w}}} \in \mathrm{RS}(C)} \sum_{i=1}^{N_r} \epsilon_i \langle \bar{{\bm{w}}}, {\bm{z}}_i\rangle\right] + 2 B_{\phi_2}|c_0|\sqrt{N_r} + |k|\sqrt{N_r}\right\}\\
    &=\frac{4B(B_{\phi_1} L_{\phi_2} + B_{\phi_2} L_{\phi_1})}{N_r}\, \mathbb{E}_{\epsilon}\left[\sup_{\bar{{\bm{w}}} \in \mathrm{RS}(C)} \sum_{i=1}^{N_r} \epsilon_i \langle \bar{{\bm{w}}}, {\bm{z}}_i\rangle\right] + \frac{B(2B_{\phi_2}|c_0| + |k|)}{\sqrt{N_r}}. \nonumber
\end{align}
