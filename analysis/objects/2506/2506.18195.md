<!-- lemma-1: body_md -->
\label{lemma:limit}
Consider a social network with irreducible aperiodic influence matrix $P$ and let $\pi$ be its  centrality vector.
Then, for every self-confidence vector $z$ in $\mathcal Z$,
\begin{equation}\label{limWt}\lim_{t\to+\infty}W(z)^t=H(z)\,,\end{equation}
where $H(z)$ in $\mathbb{R}^{n\times n}$ is a stochastic matrix such that:
\begin{enumerate}
\item[(i)] if $\mathcal S(z)=\emptyset$, then \begin{equation}\label{Hz}H(z)=\mathbf{1} p(z)'\,,\end{equation}
has all rows equal to $p(z)'$, where
\begin{equation}\label{def:p(z)}
 p(z)\triangleq\frac1{\gamma(z)}(I-[z])^{-1}\pi\,,\end{equation}
and $\gamma(z)$ is the positive scalar defined by
\begin{equation}\label{def:gamma(z)}\gamma(z)\triangleq\sum_{i\in\mathcal V}\frac{\pi_i}{1-z_i}\,;\end{equation}
\item[(ii)] if $\mathcal S(z)\ne\emptyset$, then $H(z)$ is the unique solution of the linear system
\begin{equation}\label{lin-sys}
\begin{array}{rclcl}
\displaystyle H_{ij}(z)&=&\displaystyle\sum_{k\in\mathcal V}P_{ik}H_{ik}(z)&\ &\forall i\!\not\in\!\mathcal S(z)\,,\ j\!\in\!\mathcal V\,,\\[15pt]
\displaystyle H_{ij}(z)&=&\delta^i_j &\ &\forall i\!\in\! \mathcal S(z)\,,\  j\!\in\!\mathcal V\,.
\end{array}
\end{equation}
\end{enumerate}

See Appendix \ref{sec:proof-lemma:limit}. \hfill$\blacksquare$

<!-- remark-1: body_md -->
\label{rem:power}
When $\mathcal S(z)=\emptyset$, the vector $p(z)$ defined in~\eqref{def:p(z)} can be interpreted as the vector of agents’ \emph{social powers}~\citep{ProTempo:2017-1}. Each entry $p_i(z)$ measures how agent~$i$'s initial opinion contributes to the group's final opinion
\[
x_1(\infty)=\ldots=x_n(\infty)=\sum_{i\in\mathcal V} p_i(z)\,x_i(0)=p(z)'x(0).
\]
By changing the self-confidence profile $z$, the agents thus adjust their social powers in accordance with~\eqref{def:p(z)} and~\eqref{def:gamma(z)}. The eigenvalue centrality from~\eqref{pi},  $\pi = p(0)$, corresponds to the case in which agents have no self-confidence and rely solely on
their neighbors' opinions.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
\label{rem:centrality} Notice that, if $\mathcal S(z)\ne\emptyset$, then Lemma \ref{lemma:limit}(ii) implies that $H(z)$ depends only on $\mathcal S(z)$, i.e., for every  two self-confidence profiles $z^{(1)}$ and $z^{(2)}$ in $\mathcal Z$ we have
\begin{equation}\label{onlySz}\mathcal S(z^{(1)})=\mathcal S(z^{(2)})\ne\emptyset\quad \Rightarrow\quad  H(z^{(1)})=H(z^{(2)})\,.\end{equation}
In particular, if $\mathcal S(z)=\{i\}$, then $H(z)=\mathbf{1}(\delta^i)'$.
<!-- end remark-2 -->
<!-- remark-3: body_md -->
\label{rem:hitting} The following probabilistic interpretation is insightful.
Consider a discrete-time Markov chain with finite state space $\mathcal V$ and transition probability matrix  $W(z)=(I-[z])P+[z]$. It is known that, if $\mathcal S(z)=\emptyset$, then such a Markov chain converges in probability to a random variable distributed in accordance with the vector $$\frac1{\gamma(z)}(I-[z])^{-1}\pi\,,$$
where $\gamma(z)$ is defined as in \eqref{def:gamma(z)}.
On the other hand, if $\mathcal S(z)\ne\emptyset$, then the Markov chain                                                          
gets absorbed in finite time in one of the states in $\mathcal S(z)$: in this case, $H_{ij}(z)$ coincides with the probability of absorption of the Markov chain in state $j$ when starting in state $i$.

<!-- proposition-1: body_md -->
\label{prop:convergece}
Consider a social network with irreducible aperiodic influence matrix $P$. Let $\pi$ be its centrality vector and let $z$ in $\mathcal Z$ be a self-confidence vector. Assume that the initial opinion profile has entries as in \eqref{initial-opinions} for some $\theta$ in $\mathbb{R}$ and a random vector $\xi$ satisfying assumption \eqref{noise-assumption}.
Then, the French-DeGroot opinion dynamics \eqref{eq.degroot} is such that
\begin{equation}\label{limxi}\lim_{t\to+\infty}x_i(t)=\theta+\zeta_i\,,\end{equation}
for every agent $i$ in $\mathcal V$, where $\zeta_i$ are random variables with expected value
\begin{equation}\label{Ezeta=0}\mathbb{E}[\zeta_i]=0\,,\end{equation}
and variance
\begin{equation}\label{estimation-variance}\upsilon_i(z)=\mathbb{E}[\zeta_i^2]=\sum_{j\in\mathcal V}H_{ij}^2(z)\sigma_j^2\,,\end{equation}
where $H(z)$ in $\mathbb{R}^{n\times n}$ is the stochastic matrix from~\eqref{limWt}.
<!-- end proposition-1 -->
<!-- ieeeproof-2: body_md -->
It follows from Lemma \ref{lemma:limit} that
$$x(t)=W(z)^tx(0)\xrightarrow[t\to+\infty]{}H(z)x(0)\,. $$
It follows from assumption \eqref{initial-opinions} and the fact that $H(z)$ is a row-stochastic matrix that
$$H(z)x(0)=\theta H(z)\mathbf{1}+H(z)\xi=\theta\mathbf{1}+\zeta\,,$$
where $\zeta=H(z)\xi$.
Thanks to assumption \eqref{noise-assumption}, we get that the entries of $\zeta_i$ are zero-mean, thus proving equation \eqref{Ezeta=0}, and have variance
$$\mathbb{E}[\zeta_i^2]=\sum_{j\in\mathcal V}\sum_{k\in\mathcal V}H_{ij}(z)\mathbb{E}[\xi_j\xi_k]H_{ik}(z)=\sum_{j\in\mathcal V}H_{ij}^2(z)\sigma_j^2\,,$$
thus proving equation \eqref{estimation-variance}.\hfill$\blacksquare$

<!-- definition-1: body_md -->
A self-confidence profile $z$ in $\mathcal Z$ is
\begin{enumerate}
\item[(i)] \emph{Pareto-optimal} if there exists no other self-confidence profile $\overline z$ in $\mathcal Z$ such that
$$\upsilon_i(\overline z)\le\upsilon_i(z)\,,\qquad\forall i\in\mathcal V\,,$$ and there exists an agent $j$ in $\mathcal V$ such that $$\upsilon_j(\overline z)<\upsilon_j(z)\,;$$
\item[(ii)] a \emph{Nash equilibrium} if for every agent $i$ in $\mathcal V$
$$\upsilon_i(z_i,z_{-i})\le\upsilon_i(\overline z_i,z_{-i})\,,\qquad \forall\overline z_i\in[0,1]\,;$$
\item[(iii)] a \emph{strict Nash equilibrium} if for every agent $i$ in $\mathcal V$
$$\upsilon_i(z_i,z_{-i})<\upsilon_i(\overline z_i,z_{-i})\,,\qquad \forall\overline z_i\in[0,1]\setminus\{z_i\}\,.$$
\end{enumerate}
We shall denote by $\mathcal P\subseteq\mathcal Z$ the Pareto frontier, i.e., the set of Pareto optimal self-confidence profiles and by $\mathcal N$ and $\mathcal N^*$, respectively, the set of Nash equilibria and the set of strict Nash equilibria.

<!-- lemma-2: body_md -->
\label{lemma:min}
If  $\sigma_i^2>0\,$ for every $i=1,\ldots,n$, then
\begin{equation}\label{eq:Vmin}
\min_{\substack{\mu\in\mathbb{R}_+^n:\\\mathbf{1}'\mu=1}}\sum_{i\in\mathcal V}\mu_i^2\sigma_i^2=V_{min}:=\left(\sum_{i\in\mathcal V}\sigma_i^{-2}\right)^{-1} \,,
\end{equation}
and the minimum above is achieved in the sole probability distribution $\mu^*$ with entries
\begin{equation}\label{mu}\mu^*_i=\frac{\sigma_i^{-2}}{\displaystyle\sum_{j\in\mathcal V}\sigma_j^{-2}}\,,\end{equation}
for every $i=1,\ldots,n$.

<!-- ieeeproof-3: body_md -->
See Appendix~\ref{sec:proof-lemma:min}.\footnote{An alternative proof earlier appeared in~\cite{Peluffo:2019}.}

<!-- proposition-2: body_md -->
\label{prop:Pareto}
Consider a social network with irreducible aperiodic influence matrix $P$ and centrality vector $\pi$.
Assume that the agents make uncorrelated initial estimations with variance $\sigma_i^2>0$ for every $i$ in $\mathcal V$.
Let
\begin{equation}
\label{Z*}\mathcal Z^*\triangleq\{\mathbf{1}-\alpha[\pi]\sigma^2:\,0<\alpha\leq\alpha_*\}\,,\end{equation}
where
\begin{equation}\label{alpha*}\alpha_*\triangleq 1/{\max_i\{\pi_i\sigma_i^2\}}\,.\end{equation}
Then, $$\mathcal P=\mathcal Z^*\subseteq\mathcal N^*\,.$$
Moreover, for every Pareto-optimal vector $z$ in $\mathcal Z^*$, the social powers associated to the French-DeGroot model~\eqref{eq.degroot} (see Remark~\ref{rem:power}) are $p_i(z)=\mu_i^*$, and the variance of the final consensus opinion is found as $V_{min}$, defined in~\eqref{eq:Vmin}.
<!-- end proposition-2 -->
<!-- ieeeproof-4: body_md -->
We first notice that when stubborn agent are present, i.e., when $\mathcal S(z)\ne\emptyset$, the matrix $H(z)$ cannot be of the type $\mathbf{1}(\mu^*)'$,
as in this case $H(z)$ has zero entries by equation \eqref{lin-sys} of Lemma \ref{lemma:limit}.
Instead, if we consider the expression of $H(z)$ in \eqref{Hz} for the case when there are no stubborn agents (i.e., when $z<\mathbf{1}$), we see that
$$H(z)=\mathbf{1}(\mu^*)'\quad\Leftrightarrow\quad \gamma(z)^{-1}(I-[z])^{-1}\pi=\mu^*$$
Because of the way $\mu^*$ is defined (see \eqref{mu}), the right hand side equality above holds true if and only if the vector $z$ satisfies the relations
\begin{equation}\label{Pareto}1-z_i=\alpha\pi_i\sigma_i^2\,,\end{equation}
for every $i$ in $\mathcal V$, for some constant $\alpha$. Notice that \eqref{Pareto} has a solution in $[0,1)^n$ if and only if $0<\alpha\leq\alpha_*$.
Hence, $H(z)=\mathbf{1}(\mu^*)'$ if and only if $z\in\mathcal Z^*$ (in particular, the set of such self-confidence profiles $z$ is nonempty). As has been already noted, this entails that $\mathcal P=\mathcal Z^*$. On the other hand, it directly follows from the definition of $\mathcal Z^*$ in \eqref{Z*} that for every $z$ in $\mathcal Z^*$, any unilateral modification of the self-confidence by one agent would lead to a vector
$\bar z\not\in \mathcal Z^*$. Hence $z$ is a strict Nash equilibrium.\hfill$\blacksquare$

<!-- remark-4: body_md -->
Proposition~\ref{prop:Pareto} shows that the Pareto frontier $\mathcal P$ is a segment within $[0,1)^n$. All vectors in $\mathcal P$ are equivalent from a cost perspective: in such configurations, every player achieves the same cost, corresponding to the minimum of the convex program presented in Lemma~\ref{lemma:min}.

<!-- proposition-3: body_md -->
\label{prop:brm}
The best response of every agent $i$ in $\mathcal V$ to a self-confidence profile $z_{-i}$ such that $\mathcal S(z_{-i})=\emptyset$ is given by
\begin{equation}\label{eq.brm}
\mathcal B_i(z_{-i})=\left\{\left[1-\frac{ A_{i}(z_{-i})\pi_i\sigma_i^2}{B_{i}(z_{-i})}\right]_+\right\}\,.
\end{equation}
{In particular, $\mathcal B_i(z_{-i})\subseteq [0, 1)$.}
<!-- end proposition-3 -->
<!-- ieeeproof-5: body_md -->
Observe that the assumption $\mathcal S(z_{-i})=\emptyset$ is equivalent to that $0\le z_{-i}<\mathbf{1}$.
We first analyze $\upsilon_i(z_i,z_{-i})$ when $z_i<1$. In this case, $\upsilon_i(z_i,z_{-i})=V(z)$, where  $V(z)$ is the common cost function defined in
\eqref{unique-cost} that we analyze through the change of variables \eqref{change}.
Consider the expression for the derivative of $\overline V$ in (\ref{eq.partial}).
For a given $y_{-i}$, the function $y_i\mapsto\overline V(y_i,y_{-i})$ is strictly decreasing when $1\leq y\le y_i^*$ and strictly increasing when $y\ge y_i^*$, where
$$y_i^*=\min\left\{1,\frac {\bar B_i(y_{-i})}{\overline A_{{i}}(y_{-i})\pi_i\sigma_i^2}\right\}\,.$$
Hence, $y_i^*$ is the strict minimizer of $\bar\upsilon_i(y_i,y_{-i})$ on the interval $[1,+\infty)$.
This minimizer corresponds to
\[
z_i^*= 1-\frac{1}{y^*_i}=\left[1-\frac{ A_{{i}}(z_{-i})\pi_i\sigma_i^2}{B_i(z_{-i})}\right]_+\in{[0,1)}\,.
\]

Now, observe that, for  every fixed $0\le z_{-i}<\mathbf{1}$, Lemma \ref{lemma:limit}(i) implies that
$$\lim_{z_i\uparrow 1}H(z_i,z_{-i})=\mathbf{1}(\delta^i)'=H(1,z_{-i})\,,$$
so that,  by \eqref{estimation-cost}, the function  $z_i\mapsto\upsilon_i(z_i,z_{-i})$ is continuous on $[0,1]$. This implies that $z_i^*$ is in fact the unique minimizer of $z_i\mapsto\upsilon(z_i,z_{-i})$ over closed interval $[0,1]$.\hfill\hfill$\blacksquare$

<!-- proposition-4: body_md -->
\label{prop.brm1}
For every agent $i$ in $\mathcal V$ and self-confidence profile $z_{-i}$ such that $\mathcal S(z_{-i})\neq\emptyset$, we have that:
\begin{enumerate}
\item[(i)] $z_i\mapsto\upsilon_i(z_i,z_{-i})$ is constant on $[0,1)$; \\
\item[(ii)] $\mathcal B_i(z_{-i})\in\{[0,1), [0,1], \{1\}\}$;\\
\item[(iii)] if $\sigma_i^2\ge\sigma_j^2$ for every $j$ in $\mathcal S(z_{-i})$, then $$\mathcal B_i(z_{-i}))\in\{[0,1), [0,1]\}\,;$$
\item[(iv)] if $\sigma_i^2>\sigma_j^2$  for every $j$ in $\mathcal S(z_{-i})$, then $$\mathcal B_i(z_{-i})=[0,1)\,.$$
\end{enumerate}
<!-- end proposition-4 -->
<!-- ieeeproof-6: body_md -->
We consider the general expression \eqref{estimation-cost} for the utility of agent $i$, $v_i(z_i, z_{-i})$. As the entries of the matrix $H$ depend on $z$ only through the set $\mathcal S(z)$ (see Remark \ref{rem:centrality}), and since $\mathcal S(z)$ is constant for $z_i$ varying in $[0,1)$, item (i) is proven. Item (ii) is immediate from (i).

 In order to prove the remaining points, we consider the following estimation that holds true for every $z_i$ in $[0,1)$ and if $\sigma_i^2\ge\sigma_j^2$ for every $j$ in $\mathcal S(z_{-i})$.
\begin{equation}\label{eq:1-optimal}
\begin{array}{rcl}
\displaystyle\upsilon_i(z_i,z_{-i})&=&\displaystyle\sum_{j\in \mathcal S(z_{-i})}H_{ij}^2(z)\sigma_j^2\\
&\leq&\displaystyle\sum_{j\in \mathcal S(z_{-i})}H_{ij}(z)\sigma_j^2\\
&\leq&\displaystyle\sigma_i^2\\
&=&\displaystyle\upsilon_i(1,z_{-i})
\end{array}
\end{equation}
We notice that the first equality follows from equation~\eqref{estimation-cost} and the fact that $H_{ij}(z)=0$ for every agent $j\not\in\mathcal S(z)$, the first inequality follows from the fact that $H_{ij}(z)\in [0,1]$ for every $i,j$, and the second inequality from the assumption made on variances and the fact that $H(z)$ is a stochastic matrix.

Inequality~\eqref{eq:1-optimal} implies that any $z_i$ in $[0, 1)$ is necessarily a best response and thus yields item (iii). We further notice that for the second inequality in expression \eqref{eq:1-optimal} to be an equality, it must be that $\sigma_j^2=\sigma_i^2$ for every $j\in \mathcal S(z_{-i})$ such that $H_{ij}(z)>0$. Recalling that $H(z)$ is stochastic, such an index $j$ surely exists and this implies that under the assumption in item (iv), the second inequality in expression~\eqref{eq:1-optimal} is strict. This proves (iv).
\hfill$\blacksquare$

<!-- proposition-5: body_md -->
\label{prop:reachable}
Let $i$ in $\mathcal V$ be an agent with minimal wisdom, i.e.,
$\sigma_i^2\geq\sigma_j^2$ for every $j$ in $\mathcal V$. Consider a self-confidence profile $z_{-i}$ such that
$\mathcal S(z_{-i})\neq\emptyset$.
Then, the following conditions are equivalent:
\begin{enumerate}
\item[(i)] $1\in \mathcal B_i(z_{-i})$
\item[(ii)] $[0,1]=\mathcal B_i(z_{-i})$
\item[(iii)] for every $z_i$ in $[0,1)$,
there exists only one node $j$ in $\mathcal S(z_{-i})$ reachable from node $i$ in the graph $\mathcal G_{W(z)}$ and, for this node $j$, one has $\sigma_i^2=\sigma_j^2$.
\end{enumerate}
<!-- end proposition-5 -->
<!-- ieeeproof-7: body_md -->
We prove (i)\,$\Rightarrow$\,(ii)\,$\Rightarrow$\,(iii)\,$\Rightarrow$\,(i)

(i)\,$\Rightarrow$\,(ii) follows from (iii) of Proposition \ref{prop.brm1}.

(ii)\,$\Rightarrow$\,(iii): under assumption (ii) we must have all equalities in expression \eqref{eq:1-optimal}. This implies that $H_{ij}(z)\in\{0,1\}$ for every $j\in \mathcal S(z_{-i})$ and that $\sigma_i^2=\sigma_j^2$ for every $j$ for which $H_{ij}(z)=1$. Since $H(z)$ is a stochastic matrix, this implies that there exists just one $j$ such that $H_{ij}(z)=1$ and for such $j$ it holds $\sigma_i^2=\sigma_j^2$. Considering the probabilistic interpretation of $H_{ij}(z)$ as the absorbing probability in the state $j$ of a Markov chain with matrix $W(z)$ initialized in $i$, it follows that
$H_{ij}(z)>0$ yields the existence of a path in $\mathcal G_{W(z)}$ from $i$ to $j$.

(iii)\,$\Rightarrow$\,(i): using the same probabilistic interpretation, we deduce from (iii) that $H_{ij}(z)=1$. This fact and the assumptions on the variances imply that all inequalities in expression \eqref{eq:1-optimal} turn into equalities. This yields (i).
\hfill$\blacksquare$

<!-- theorem-1: body_md -->
\label{theo:internalNash} The set $\mathcal P=\mathcal Z^*$ comprises all Nash equilibria in $[0,1)^n$, that is,
 $\mathcal N\cap [0,1)^{n}=\mathcal N^*\cap [0,1)^{n}= \mathcal Z^*$.
<!-- end theorem-1 -->
<!-- ieeeproof-8: body_md -->
Notice first that the inclusion $\mathcal N^*\cap [0,1)^{n}\supseteq \mathcal Z^*$ follows from Proposition \ref{prop:Pareto}.
To complete the proof, it remains to show that there are no Nash equilibria in the set $[0,1)^{n}\setminus\mathcal Z^*$.

We first show that if $z=\mathbf{0}$ is a Nash equilibrium, then $\mathbf{0}\in\mathcal Z^*$. By the definition of a Nash equilibrium, $0\in\mathcal B_i(\mathbf{0})$ for all $i$, meaning that $z_i=0$ minimizes $V(z_i,\mathbf{0})$ and $y_i=1$ minimizes $\bar V(y_i,\mathbf{1})$. In view of~\eqref{eq.partial}, one has
\[
\frac{\partial\bar V}{\partial y_i}(\mathbf{1})=\frac{2\pi_i\left(\overline A(y)\pi_i\sigma_i^2y_i-\overline B(y)\right)}{\left(\pi_iy_i+\overline A_{i}({y_{-i}})\right)^3}\Big\vert_{y=\mathbf{1}}\geq 0.
\]
One notices, however, that the numerators of the latter fractions,
for $y=\mathbf{1}$, satisfy the following property:
$$\sum\limits_{i\in\mathcal V}2\pi_i(\overline A(\mathbf{1})\pi_i\sigma_i^2-\overline B(\mathbf{1}))=\sum\limits_{i\in\mathcal V}2\pi_i^2\sigma_i^2-\sum\limits_{i\in\mathcal V}2\pi_i^2\sigma_i^2=0.
$$
Hence all fractions must vanish at $y=\mathbf{1}$, entailing that
$$\overline B(\mathbf{1})-\overline A(\mathbf{1})\pi_i\sigma_i^2=0,\quad\forall i\in\mathcal V,$$
which implies that $\pi_i=\sigma_i^{-2}/\sum_{j\in\mathcal V}\sigma_j^{-2}$, that is, $\pi=\mu^*$. By \eqref{Z*}, we have $\mathbf{0}\in\mathcal Z^*$ (this vector corresponds to $\alpha=\alpha^*$ in \eqref{Z*}).


Consider now a general Nash equilibrium $\bar z$ in $[0,1)^n$. We observe that $\bar z$ also constitutes a Nash equilibrium in the game with the more restricted strategy set $\mathcal Z_{\bar z}=\prod_{i=1}^n[\bar z_i,1)$. For every vector $z$ in $\mathcal Z_{\bar z}$, the matrix of the French-DeGroot model~\eqref{eq.degroot} can be rewritten as follows
\[
\begin{gathered}
W(z)=(I-[\hat z])W(\bar z)+[\hat z],\;\;\text{where}\\
\hat z\triangleq(I-[\bar z])^{-1}(z-\bar z)\in[0,1)^n.
\end{gathered}
\]
Hence, $\mathbf{0}$ is a Nash equilibrium in the game, defined by the matrix $\bar P=W(\bar z)$ with the corresponding left eigenvector
$\overline\pi=\gamma(\overline z)^{-1}(I-[\overline z])^{-1}\pi$, which, as has been shown, means that $\overline\pi=\mu^*$. In the proof of Proposition~\ref{prop:Pareto}
we have seen that this holds true if and only $\overline z\in\mathcal Z^*$.
\hfill\hfill$\blacksquare$

<!-- theorem-2: body_md -->
\label{theo:other-Nash}
Consider a social network with irreducible aperiodic influence matrix $P$ and centrality vector $\pi$.
Assume that the initial measurements of the agents are uncorrelated with variance $\sigma_i^2>0$ for $i$ in $\mathcal V$.
Then, every possible Nash equilibrium $z$ such that $\mathcal S(z)\neq\emptyset$
is non strict and is such that:
\begin{enumerate}

\item[(ii)] $\sigma_i^2=\sigma_j^2$ for every $i$ and $j$ in $\mathcal S(z)$;

<!-- lemma-3: body_md -->
\label{lemma:BR} For every distribution of the initial configuration $Z(0)$ on $[0,1]^{\mathcal V}$, there exists a nonnegative random time $T$ such that $\mathbb{P}(T<+\infty)=1$, and the asynchronous best response dynamics satisfies \begin{equation}\label{Z<1}Z(t)<\mathbf{1}\,,\qquad \forall t\ge T\,.\end{equation}
<!-- end lemma-3 -->
<!-- ieeeproof-13: body_md -->
Consider the stochastic process $Q(t)$ on the finite state space $\mathcal Q=\{0,1\}^{\mathcal V}$, defined by
$$Q_i(t)=\lfloor Z_i(t)\rfloor\,,\qquad i\in\mathcal V\,,\ t=0,1,2,\ldots\,,$$
                                                                                                                                                                                                                                                                                                  
i.e., $Q_i(t)=1$ if agent $i$ is stubborn at time $t$ and $Q_i(t)=0$ otherwise. Hence, $$\mathcal S(Z(t))=\{i\in\mathcal V:\,Q_i(t)=1\}\,.$$

Recall from Proposition~\ref{prop:brm}
                    
that the best response of every agent $i$ in $\mathcal V$
such that $Q_{-i}(t)=\0$, i.e.,  such that $\mathcal S(Z_{-i}(t))=\emptyset$, satisfies $\mathcal B_i(Z_{-i}(t))\subseteq [0,1)$.
                                                                                                                                                              
This implies that, if the agent $i$ that gets randomly activated at time $t$ is such that $Q_{-i}(t)=\0$, then necessarily $Q_i(t+1)=0$.


On the other hand, for every every agent $i$ in $\mathcal V$ such that $Q_{-i}(t){\neq \0}$, i.e., such that $\mathcal S(Z_{-i}(t)){\neq}\emptyset$,
Proposition \ref{prop.brm1}(ii) ensures that $$\mathcal B_i(Z_{-i}(t))\in\{[0,1),[0,1],\{1\}\}\,.$$ Observe, that in this case, Equation \eqref{estimation-cost}, Lemma \ref{lemma:limit}(ii), and Remark \ref{rem:centrality} imply that
$$\upsilon_i(z_i,Z_{-i}(t))=\sum_{j\in\mathcal V}H_{ij}^2(z_i,Z_{-i}(t))\sigma_j^2\,,$$
depends on $Z_{-i}(t)$ only through $\mathcal S(Z_{-i}(t))$, i.e., through $Q_{-i}(t)$, hence so does $\mathcal B_i(Z_{-i}(t))$.
This implies that there exists a function $p_i:\{0,1\}^{\mathcal V\setminus\{i\}}\to[0,1]$, such that, if the agent $i$ that gets randomly activated at time $t$ is such that $Q_{-i}(t)\ne\0$, then  with conditional probability $p(Q_{-i}(t))$ we get $$\mathcal B_i(Z_{-i}(t))\in\{[0,1),[0,1]\}\,,$$  and hence $Z_i(t+1)=0$, while with conditional probability $1-p(Q_{-i}(t))$ we get $$\mathcal B_i(Z_{-i}(t))=\{1\}\,,$$ hence $Z_i(t+1)=1$.
In fact, Proposition \ref{prop.brm1}(iii) implies that, if
$$i\in\operatorname*{argmax}\limits_{j: Q_j(t)=1}\sigma_j^2\,,$$
then $p(Q_{-i}(t))=1$.
                                                               



                                                                                                      
The arguments above imply that $Q(t)$ is itself a Markov chain on $\mathcal Q$ such that
\begin{equation}\label{absorbingQ}\mathbb{P}(Q(t+1)=\0|Q(t)=\0)=1\,,\end{equation}
                                                                                                                       
and that for every $q$ in $\mathcal Q\setminus\{\0\}$,
 there exists $q^-$ such that
 \begin{equation}\label{q-1}\sum_{i\in\mathcal V}q^-_i=\sum_{i\in\mathcal V}q_i-1\,,\end{equation} and
\begin{equation}\label{Pq-1}\mathbb{P}(Q(t+1)=q^-|Q(t)=q)\ge\frac1n>0\,.\end{equation}
It then follows from \eqref{absorbingQ}, \eqref{q-1}, and \eqref{Pq-1} that $\0$ is absorbing state for the Markov chain $Q(t)$ on the finite state space $\mathcal Q=\{0,1\}^{\mathcal V}$. Standard results on finite Markov chains~\citep[Chapter 1]{norris1998markov} imply that there exists a nonnegative random absorption time $T$ such that $\mathbb{P}(T<+\infty)=1$, and $Z(t)=\0$ for every $t\ge T$, so that
\eqref{Z<1} holds true.\hfill$\blacksquare$

<!-- theorem-3: body_md -->
\label{theo:convergence}
For every distribution of the initial configuration $Z(0)$ on $[0,1]^{\mathcal V}$, the sequence $Z(t)$ generated by the asynchronous best response dynamics converges to the set $\mathcal Z^*$, with probability one.
<!-- end theorem-3 -->
<!-- ieeeproof-14: body_md -->
We fix an initial condition $z(0)$  and we take $T$ as in Lemma \ref{lemma:BR}, so that $Z(t)\in [0,1)^n$ for every $t\geq T$.

We first show that every limit point of the sequence $Z(t)$ belongs to $[0,1)^n$.
To this aim, we define
$$M(t)=\max\limits_{j=1,\dots , n}\frac{\pi_j\sigma_j^2}{1-Z_j(t)}$$
and then we show that $M(t)$ is monotonically non-increasing for $t\geq T$. For a given time $t$, we pick
$$m\in \operatorname*{argmax}\limits_{j=1,\dots , n}\frac{\pi_j\sigma_j^2}{1-Z_j(t)}\,,$$
and we notice that for any other player $k$ in $\mathcal V\setminus\{m\}$, the form of the best response in \eqref{eq.brm} implies that
$$\frac{\pi_k\sigma_k^2}{1-Z_k(t+1)}=\max\left\{ \pi_k\sigma_k^2, \frac{B_k(Z_{-k}(t))}{A_k(Z_{-k}(t))}\right\}\,.$$
Notice now that
$$\begin{array}{rcl}B_k(Z_{-k}(t))&=&\displaystyle\sum\limits_{i\neq k}\frac{\pi_j^2\sigma_j^2}{(1-Z_j(t))^2}\\ &=&\displaystyle\sum\limits_{i\neq k}\frac{\pi_j}{1-Z_j(t)}\frac{\pi_j\sigma_j^2}{1-Z_j(t)}\\ &\leq&\displaystyle A_k(Z_{-k}(t))\frac{\pi_m\sigma_m^2}{1-Z_m(t)}\,.\end{array}$$
This yields
$$\frac{B_k(Z_{-k}(t))}{A_k(Z_{-k}(t))}\leq \frac{\pi_m\sigma_m^2}{1-Z_m(t)}\,.$$
As also $$ \pi_k\sigma_k^2\leq \frac{\pi_k\sigma_k^2}{1-Z_k(t)}\leq \frac{\pi_m\sigma_m^2}{1-Z_m(t)}\,,$$
we have thus proven that
$$\frac{\pi_k\sigma_k^2}{1-Z_k(t+1)}\leq M(t)\,.$$
This yields the monotonicity of $M(t)$ for $t\ge T$. Consequently,
$$\frac{1}{1-Z_j(t)}\leq \frac{M(T)}{\pi_j\sigma_j^2}\,,\qquad \forall j\in\mathcal V,\ t\ge T\,,$$
so that we have the a-priori bound
\begin{equation}\label{apriori-bound}Z_j(t)\leq 1-\frac{\pi_j\sigma_j^2}{M(0)}\,,\qquad\forall j\in\mathcal V,\ t\ge T\,.\end{equation}

We now consider the sequence $V(Z(t))$. By construction, it is non-increasing for $t\geq T$ and converges to some value $\bar V$ in $\mathbb{R}_+$. Consider now any limit point of the sequence $Z(t)$ and without loss of generality assume that $Z(t)\to \bar z$ (avoiding to formally pass to a subsequence). We know from previous considerations that $\bar z\in [0,1)^n$ and we want to show that $\bar z$ is a Nash equilibrium. Suppose not and let $i\in\mathcal \mathcal V$ be an agent that is not at equilibrium in configuration $\bar z$. Let $f_i: [0,1)^n\to  [0,1)^n$ be the function defined by
$f_i(z)=(\mathcal B_i(z_{-i}), z_{-i})$. We have $V(f_i(\bar z))=V(\bar z)-\epsilon$ for some $\epsilon >0$. Since $f_i$ is continuous, there exists $\delta>0$ such that $|z^1-z^2|<\delta$ yields
\begin{equation}\label{contraction}\left|[V(f_i(z^1))-V(z^1)]- [V(f_i(z^2))-V( z^2)]\right|<\frac{\epsilon}{2}\end{equation}
Consider now $\bar t\geq T$ such that $|Z(t)-\bar z|<\delta$ and $|V(Z(t))- V(\bar z)|<\epsilon/4$ for every $t\geq \bar t$ and pick any time $t\geq \bar t$ at which agent $i$ becomes active (with probability $1$ there are infinite such times).
From expression \eqref{contraction}, using the fact that $|Z(t)-\bar z|<\delta$ we deduce that
$V(f_i(Z(t)))-V(Z(t))<-\epsilon/2$.
Coupling now with $|V(Z(t))- V(\bar z)|<\epsilon/4$, we get that $V(f_i(Z(t)))-V(\bar z)<-\epsilon/4$ and this contradicts the fact that $V(Z(t))$ monotonically converges to $\bar z$. This proves that $\bar z$ is a Nash equilibrium. \hfill$\blacksquare$
