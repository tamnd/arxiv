<!-- theorem-1: body_md -->
The partition function $ {\cal Z}_s  ( N , p ,  x)  $ which is the Molien-Weyl inverse determinant \eqref{WeylFormp} is equal to the following product depending on LCMs of numbers chosen from the set of cycle lengths $ \{ a_1 , a_2 , \cdots , a_K \}$ :
\begin{eqnarray}\displaystyle\label{MainProps1} 
{\cal Z}_s  ( N , p ,  x) = 
  \prod_{ i_1 \cdots i_s   \in \{ 1, \cdots , K \} } { 1 \over ( 1 - x^{ L ( a_{i_1} , \cdots , a_{ i_s} ) }  )^{ \frac{a_{i_1} \cdots  a_{ i_s}    }{L( a_{i_1},  
  \cdots ,  a_{i_s}  )  }    p_{i_1} \cdots p_{i_s}    }}
\end{eqnarray}

<!-- theorem-2: body_md -->
\label{subsetsumtheorem} 
The partition function $ {\cal Z}_s ( N , p , x  ) $, for the contribution of a partition $  p $ of $N$, for $S_N$ invariant $s$-index tensors, which is given by  \eqref{WeylFormp} and \eqref{MainProps1} can be re-written as 
\begin{eqnarray}\displaystyle\label{subsetsumformula}  
{\cal Z}_s ( N , p , x  ) =
 \prod_{ \substack {  S \subset \{ 1, \cdots , K  \} \\ S \ne \emptyset } }   
 { 1 \over ( 1 - x^{ L ( S )  } )^{ {\cal W} ( N , p , s ; S ) \over L ( S )   }  }
\end{eqnarray}
with $ {\cal W} ( N , p , s ;  S )$ as given in \eqref{expss}.

Consider the unique prime factorisations of the numbers $b_1 , b_2$, which exist by the fundamental theorem of arithmetic (e.g. Theorem 1.10 of \cite{Apostol}).  Let $  { \rm PF }   (b) $ be the set of prime factors of an integer $b$, and for a prime $p$ in this set, let $m_p ( b ) $ be the largest power of $p$ which divides $b$. 
\begin{eqnarray}\displaystyle 
&& b_1 = \prod_{ p \in  { \rm PF }  ( b_1 )  } p^{ m_p ( b_1)    } \cr  
&& b_2 = \prod_{ p \in  { \rm PF }  ( b_2 )  } p^{ m_p  ( b_2  ) }  
\end{eqnarray}
Let us also denote $  { \rm PF }  ( b_1 , b_2 ) \equiv  { \rm PF }  ( b_1 ) \cup  { \rm PF }  ( b_2 ) $. The relations between the LCM and the maxima of the exponents in the prime decomposition and between the GCD and the minima of the exponents in the prime decomposition are well known. 
\begin{eqnarray}\displaystyle 
&&  G  ( b_1 , b_2 ) = \prod_{ p \in  { \rm PF }  ( b_1 , b_2 ) }  p^{ \min  ( m_p  ( b_1 ) ,  m_p  ( b_2 ) )   } \cr 
&&   L  ( b_1 , b_2 ) = \prod_{ p \in  { \rm PF }  ( b_1 , b_2 ) } 
 p^{ \max  ( m_p  ( b_1 ) ,  m_p  ( b_2 ) )   } 
\end{eqnarray}
Now use 
\begin{eqnarray}\displaystyle 
&& \max ( a , b  ) =  a ~ {\cal I}  (  a \ge b ) + b  ~ {\cal I}  ( a < b ) \cr 
&& \min ( a , b ) = b  ~ {\cal I}  ( a \ge b ) + a  ~ {\cal I}  ( a < b ) 
\end{eqnarray}
where the indicator function $ {\cal I}  ( x ) $ is defined by: 
\begin{eqnarray}\displaystyle 
&& {\cal I} ( x ) = 1   \hbox { if $x$  holds }\cr  
&& {\cal I} ( x ) = 0 \hbox {  otherwise }  
\end{eqnarray}

It follows that 
\begin{eqnarray}\displaystyle 
 \max ( a , b  ) + \min ( a , b ) &=& a  ( {\cal I}  ( a < b ) + {\cal I}   (  a \ge b ) ) 
+ b  ( {\cal I} ( a < b ) + {\cal I} (  a \ge b ) )   \cr 
& = & a +b 
\end{eqnarray}
Applying this result to the prime decompositions 
\begin{eqnarray}\displaystyle 
 G  ( b_1 )  L   ( b_2 ) &&  = \prod_{ p \in  { \rm PF }  ( b_1 , b_2 ) } p^{  \min ( m_p ( b_1 ) , m_p (b_2) ) 
+ \max (  m_p ( b_1 ) , m_p ( b_2 )     )     }  \cr 
&& = \prod_{ p \in  { \rm PF }  ( b_1 , b_2 )   }   
p^{ m_p ( b_1 ) + m_p ( b_2 ) }  = b_1 b_2 
\end{eqnarray}

{ \bf Relation between LCM and GCD for a triple of numbers } 
\begin{eqnarray}\displaystyle\label{Labc} 
  L  ( b_1 , b_2 , b_3 ) = {b_1 b_2 b_3   G  ( b_1 , b_2 , b_3 ) \over  G  ( b_1 , b_2 ) G ( b_1 , b_3 )  G  ( b_2 , b_3 )} 
\end{eqnarray} 
We again use the prime decompositions to write 
\begin{eqnarray}\displaystyle 
 L  ( b_1, b_2 , b_3 ) && = \prod_{ p \in  { \rm PF }  ( b_1 , b_2 , b_3 ) } 
p^{ \max ( m_p ( b_1 ) , m_p ( b_2  ) , m_p ( b_3 ) )  } \cr 
 G  ( b_1 , b_2 , b_3 ) && = \prod_{ p \in  { \rm PF }  ( b_1 , b_2 , b_3 ) } 
p^{ \min ( m_p ( b_1 ) , m_p ( b_2  ) , m_p ( b_3 )  ) }
\end{eqnarray}
The maximum and minimum of a triple of numbers is known to be related by 
\begin{eqnarray}\displaystyle 
\max ( m_1 , m_2 , m_3 ) &=& m_1 + m_2 + m_3 - \min ( m_1 , m_2 ) - \min ( m_1 , m_3 )\\&&\qquad\qquad - \min ( m_2 , m_3 ) + \min ( m_1 , m_2 , m_3 ) \cr 
&&
\end{eqnarray}
An easy proof is to consider the sets 
\begin{eqnarray}\displaystyle 
&& S_1 = \{ 1, 2, \cdots , m_1 \} \cr 
&& S_2 = \{ 1, 2, , \cdots , m_2 \} \cr 
&& S_3  = \{ 1, 2,  \cdots , m_3 \} 
\end{eqnarray} 
Observe that 
\begin{eqnarray}\displaystyle 
\max ( m_1 , m_2 , m_3 )  = | S_1 \cup S_2 \cup S_3 | 
\end{eqnarray}
where the RHS is the cardinality of the union of the three sets, and 
\begin{eqnarray}\displaystyle 
&& \min ( m_1 ,m_2 ) = | S_1 \cap  S_2 | \cr 
&& \min ( m_1 , m_2 , m_3 ) = | S_1 \cap  S_2 \cap S_3 |
\end{eqnarray}
We then use the inclusion-exclusion principle for sets which states that 
\begin{eqnarray}\displaystyle 
&& | S_1 \cup S_2 \cup S_3 | =  |S_1| + |S_2| + | S_3| - |  S_1 \cap S_2 |  - |S_1 \cap S_3 | - | S_2 \cap S_3 | + |S_1 \cap S_2 \cap S_3 |  \cr 
&& 
\end{eqnarray}
For the statement of the inclusion-exclusion principle and its proof using indicator functions see \cite{WikiInEx}\cite{Rota}.

\noindent 
{\bf General formula relating LCM and GCD for multiple positive integers } \\ 
\begin{eqnarray}\displaystyle\label{GenLCMGCD}  
 L  ( b_1 , \cdots  , b_n ) = ( b_1 \cdots b_n) \prod_{ k=2}^n  ( {\cal G}_k ( b_1 , \cdots , b_n ) )^{ (-1)^{ k+1} }
\end{eqnarray}
where 
\begin{eqnarray}\displaystyle 
{\cal G}_k ( b_1 , \cdots , b_n ) 
=  \prod_{ i_1 < i_2 < \cdots < i_k \in \{ 1, \cdots , n \} } (  G  ( b_{ i_1}, b_{ i_2}  , \cdots , b_{ i_k } ) )
\end{eqnarray}
In other words, $ {\cal G}_k $ is the product over $k$-element subsets, of the GCD for the subset.

For a permutation $ \sigma $ with cycle structure given by $p$ as described above, the eigenvectors in the natural representation $V_N$  are 
\begin{align}

\end{align} 

In the 3-fold tensor product $V_N^{ \otimes 3 } $ the eigenvalues take the form

\end{eqnarray}
with 
\begin{eqnarray}\displaystyle\label{mweig3} 
&& 1 \le i_1 , i_2 , i_3 \le K \cr 
&& 0 \le t_{ i_1} \le ( a_{ i_1} -1 ) \cr 
&& 0 \le t_{ i_2} \le ( a_{ i_2} -1 ) \cr 
&& 0 \le t_{ i_3} \le ( a_{ i_3} -1 ) 
\end{eqnarray}
and the multiplicity space spanned by 
\begin{eqnarray}\displaystyle\label{mweig4}  
&& 1 \le r_1 \le p_{ i_1}  \cr 
&& 1 \le r_2 \le p_{ i_2} \cr 
&& 1 \le r_3 \le p_{ i_3} 
\end{eqnarray}
has dimension $ p_{ i_1} p_{ i_2} p_{ i_3} $. 
Thus the determinant appearing in the Molien-Weyl formula is 
\begin{eqnarray}\displaystyle 
&& { 1 \over  {\rm\bf det}  ( 1 - x D^{ V_N^{ \otimes 3 } } ( \sigma ) ) }  
 = \prod_{ i_1 =1}^{ K } ~ \prod_{ i_2 =1 }^K ~ \prod_{ i_3 =1}^{ K }   
\left (   \prod_{ t_{i_1}  = 0 }^{ a_{i_1}  -1 } ~ \prod_{ t_{i_2}  = 0}^{ a_{i_2}  -1 }  ~  \prod_{ t_{i_3}  = 0}^{ a_{i_3}  -1 } { 1 \over   ( 1 -  x\omega_{ a_{i_1} }^{ t_{i_1} } \omega_{ a_{i_2} }^{ t_{i_2} } \omega_{ a_{i_3}  }^{ t_{i_3} }   )^{ p_{ i_1} p_{ i_2} p_{ i_3}  }}  \right ) \cr 
&& 
\end{eqnarray}
It is useful to introduce the definition 
\begin{eqnarray}\displaystyle 
{\cal F}_{ 3 } ( b_1 , b_2 , b_3 ) =  \prod_{ t_1 = 0 }^{ b_1   -1 } ~ \prod_{ t_2 = 0}^{ b_2   -1 }  ~  \prod_{ t_3 = 0}^{ b_3   -1 } { 1 \over   ( 1 -  x\omega_{ b_1 }^{ t_1} \omega_{ b_2  }^{ t_2 } \omega_{ b_3  }^{ t_3}   )}
\end{eqnarray}
for the product over roots of unity. The inverse determinant is then 
\begin{eqnarray}\displaystyle\label{ZF3form}  
{ 1 \over  {\rm\bf det}  ( 1 - x D^{ V_N^{ \otimes 3 } } ( \sigma ) ) }  
= \prod_{ i_1 =1}^{ K } ~ \prod_{ i_2 =1 }^K ~ \prod_{ i_3 =1}^{ K }   
   (  {\cal F}_{ 3 } ( a_{ i_1}  , a_{ i_2}  , a_{ i_3 }  )  )^{ p_{ i_1} p_{ i_2} p_{ i_3} } 
\end{eqnarray} 

The next step  in the proof is to express the products over roots of unity in terms of LCMs and GCDs of the cycle lengths $a_i$ appearing in the cycle decomposition of the permutation 
\begin{align}
  {\cal F}_{ 3 } ( b_1   , b_2   , b_3  )  ~ & =~\prod_{ t_1 = 0 }^{ b_1  -1 } ~ \prod_{ t_2 = 0}^{ b_2  -1 }  ~  \prod_{ t_3 = 0}^{ b_3  -1 } 
{ 1 \over ( 1 -  x\omega_{ b_1 }^{ t_1} \omega_{ b_2 }^{ t_2 } \omega_{ b_3  }^{ t_3}   ) } \cr 
~ &= ~ \prod_{ t_3 } \prod_{ t_2 }  { 1 \over ( 1 - x^{  b_1  } \omega_{ b_2 }^{ b_1  t_2} \omega_{ b_3 }^{ b_1 t_3} )  } \cr 
~ & = ~ \prod_{ t_3}  { 1 \over (   1 - x^{ L ( b_1  ,   b_2  )  }   \omega_{ b_3 }^{  L ( b_1  , b_2   ) .  t_3   } )^{  G ( b_1 , b_2  )  } }
\end{align} 
The second equality uses \eqref{Lemma1} to do the sum over $ t_1$. The third equality uses  \eqref{Lemma2}  to do the $t_2$ pproduct. 
In the next step we do  the $t_3$ product using \eqref{Lemma2} to get 
\begin{eqnarray}\displaystyle\label{F3form} 
  {\cal F}_{ 3 } ( b_1   , b_2   , b_3  )  && = { 1 \over 
(   1 - x^{ L ( L ( b_1  , b_2  )  , b_3 ) } )^{ G ( b_1  , b_2  )  G ( L ( b_1  , b_2 ) , b_3 ) }  } \cr 
&& = { 1 \over ( 1 - x^{ L ( b_1  , b_2  , b_3  ) } )^{ G ( b_1 , b_2   ) G ( L ( b_1  , b_2  ) , b_3 ) }  } 
\cr 
&& 
\end{eqnarray}
Combining \eqref{ZF3form} and \eqref{F3form} gives the proof of \eqref{MainProp}. \hfill $\blacksquare $

\end{eqnarray}

\\ 
We start by using the relation \eqref{Lab}  between GCD, LCM and the product of a pair of numbers
\begin{eqnarray}\displaystyle\label{fststpp1} 
&& G ( L ( b_1 , \cdots , b_n ) ,  b_{ n+1} )  = {  L ( b_1 , \cdots , b_n ) b_{ n+1} \over L ( L ( b_1 , \cdots , b_n ) , b_{ n+1} ) }  =  { L ( b_1 , \cdots , b_n ) b_{ n+1} \over L ( b_1 , b_2 , \cdots , b_n , b_{ n+1} ) } 
\end{eqnarray}
Now use \eqref{GenLCMGCD}
\begin{eqnarray}\displaystyle 
L ( b_1 , \cdots , b_l ) = ( b_1 b_2 \cdots b_l ) \prod_{ k=2}^{ l}  ( {\cal G}_{ k } ( b_1 , \cdots , b_l ) )^{  \epsilon (k +1 ) }  
\end{eqnarray}
This allows us to replace the LCMs in \eqref{fststpp1} with GCDs. 

This gives 
\begin{eqnarray}\displaystyle 
&& G ( L ( b_1 , \cdots , b_n ) , b_{ n+1} )  
= ( b_1 \cdots b_{ n+1})
 \prod_{ k =2}^{n} ( {\cal G}_{ k} ( b_1 , \cdots , b_n ) )^{ \epsilon ( k+1)  } \cr 
 && 
 \times { 1 \over ( b_1 \cdots b_{ n+1}) } \prod_{ k=2}^{ n +1 }  ( {\cal G}_{ k} ( b_1 , \cdots , b_n , b_{ n+1}  ) )^{ \epsilon(k)  } 
\end{eqnarray}
In this expression, the GCDs of pairs, triples, in general size $k$-subsets, which do not contain $b_{ n+1} $ cancel out and we are left with an alternating product of GCDs which include $b_{ n+1}$. 
\begin{eqnarray}\displaystyle\label{GCDdistLCM} 
G ( L ( b_1 , \cdots , b_n ) , b_{ n+1} )   
= \prod_{ k=2}^{ n+1} ( {\cal G}_{ k} ( b_1 , \cdots , b_n  ; b_{ n+1} ) )^{ \epsilon(k)  } 
\end{eqnarray}
This completes the proof of the Proposition.  \hfill  $\blacksquare$

\\ 
To understand the proof consider the following identity 
\begin{eqnarray}\displaystyle 
{\cal G}_2 ( b_1 , \cdots , b_s ) = {\cal G}_2 ( b_1 , b_2 ) {\cal G}_{ 2} ( b_1 , b_2 ; b_3 ) 
\cdots {\cal G}_{ 2} ( b_1 , b_2 , \cdots , b_{s-1} ; b_{ s} ) 
\end{eqnarray}
The left hand-side is the product of GCDs of pairs of numbers chosen from $ \{ b_1 , \cdots , b_s \}$. On the RHS the last factor is the product of GCDs of pairs at least one of which is $b_s$, the second to last factor is the product of GCDs of pairs where at least one of the pair is $b_{ s-1}$ and the other is $b_i$ for some $ i < s-1$. In other words the pairs in $ {\cal G}_2 (b_1 , \cdots , b_s ) $ are being ordered according to the highest number in the pair, and collected into products where the highest number is $b_2$, then the highest if $b_3$, etc. up to $b_{s}$. 
\begin{eqnarray}\displaystyle 
{\cal G}_{ 2} ( b_1 , \cdots , b_s ) 
= \prod_{ l=1}^{ s - 1} {\cal G}_{ 2} ( b_1 , \cdots , b_l ; b_{ l+1} ) 
\end{eqnarray}
The same process of ordering and collecting for $k$-element subsets leads to 
\begin{eqnarray}\displaystyle\label{cGandcH}  
{\cal G}_{ k } ( b_1 , \cdots , b_s ) = 
\prod_{ l=1}^{ s -1  } {\cal G}_{ k } ( b_1 , \cdots , b_l ; b_{ l+1} ) 
\end{eqnarray}

Now we can combine \eqref{ExpfromMW1}, \eqref{Prop1eq} to write the first two equalities  below 
\begin{eqnarray}\displaystyle 
&&   { \rm Exponent }  ( b_1 , \cdots , b_s ) 
= \prod_{ l =1}^{ s-1} G ( L ( b_1 , \cdots , b_l ) , b_{ l+1} ) \cr 
&& = \prod_{ l=1}^{ s-1} \prod_{ k=2}^{ l+1} ( {\cal G}_{ k} ( b_1 , \cdots , b_l  ; b_{ l+1} ) )^{ \epsilon(k) }  \cr 
&& = \prod_{ k=2}^{s}  \prod_{ l=1}^{ s-1} ( {\cal G}_{ k} ( b_1 , \cdots , b_l  ; b_{ l+1} ) )^{ \epsilon(k)   } \cr 
&& = \prod_{ k=2}^s ( {\cal G}_{ k} ( b_1 , \cdots , b_{ s } ) )^{ \epsilon( k )  } 
\end{eqnarray}
In the third line we re-arranged the product. In the fourth line, we used \eqref{cGandcH}. 

Finally we make one more use of \eqref{GenLCMGCD} to conclude that 
\begin{eqnarray}\displaystyle
  { \rm Exponent }  ( b_1 , \cdots , b_s ) = 
{ b_{ 1} b_{ 2} \cdots b_s \over L ( b_1 , b_2 , \cdots , b_s ) } 
\end{eqnarray}
This completes the proof of the Theorem 3, equation  \eqref{Thm3}.  
Note that while the RHS of \ref{ExpfromMW} is not manifestly symmetric under permutations of 
$ \{ b_1 , \cdots , b_s \} $, the above result of  \eqref{Thm3} does have the symmetry.    \hfill $\blacksquare $
