<!-- theorem--1: body_md -->
\label{thm:main-intro}
There are universal constants $c,C>0$ such that the following holds. Let $\mathcal{H}$ be a finite-dimensional Hilbert space and $|\psi\rangle\in \mathcal{H}\otimes \mathcal{H}$ a state. Then there is a finite set $\Lambda$, a distribution $\mu$ on $\Lambda$, and for each $\lambda\in\Lambda$ a state $|\psi_\lambda\rangle$ that is maximally entangled on a subspace $\mathcal{H}_\lambda \otimes \mathcal{H}_\lambda \subseteq \mathcal{H} \otimes \mathcal{H}$ such that letting $\rho$ be the reduced density of $|\psi\rangle$ on the first factor and $\rho_\lambda$ the reduced density of $|\psi_\lambda\rangle$ on $\mathcal{H}_\lambda\subseteq\mathcal{H}$, 
\begin{equation}\label{eq:rho-plambda-i}
 \rho \,=\, \ensuremath{\mathop{\textsc{E}}_{\lambda \sim \mu}} \big[\,\rho_\lambda\,\big]\;.
\end{equation}
Moreover, let $\ensuremath{\mathcal{A}}$ be a finite set and $\{A_a\}_{a\in \ensuremath{\mathcal{A}}}$ an arbitrary measurement on $\mathcal{H}$. Then there is
a projective measurement $\{A^{\lambda}_a\}$ on $\mathcal{H}_\lambda$ such that
\begin{equation}\label{eq:lambda-strat-i}
\ensuremath{\mathop{\textsc{E}}_{\lambda\sim\mu}} \Big[  \sum_a  \big\| \big(A_a - A^{\lambda}_a\big)\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi_\lambda\rangle \big\|^2\,\Big]\,\leq\,C \,\Big(1-\sum_a\langle\psi| A_a \otimes A_a |\psi\rangle\Big)^c\;.
\end{equation}

<!-- definition-1: body_md -->
A \emph{strategy} $\mathscr{S}$ is a tuple $(|\psi\rangle,A,B)$ where $|\psi\rangle\in\mathcal{H}_{\textsf{A}}\otimes \mathcal{H}_{\textsf{B}}$ is a quantum state and $A = \{A^x_a\}$ (resp. $B=\{B^y_b\}$) is a collection of measurements on $\mathcal{H}$ indexed by $x\in \ensuremath{\mathcal{X}}$ and with outcomes $a\in \ensuremath{\mathcal{A}}$ (resp. $y\in\ensuremath{\mathcal{Y}}$ and $b\in\ensuremath{\mathcal{B}}$). Any strategy induces a \emph{correlation}, which is the collection of real numbers 
\[ C_{xyab} \,=\, \langle\psi| A^x_a \otimes B^y_b |\psi\rangle\;,\qquad \forall (x,y) \in \ensuremath{\mathcal{X}}\times \ensuremath{\mathcal{Y}}\;,\quad \forall (a,b)\in \ensuremath{\mathcal{A}}\times \ensuremath{\mathcal{B}}\;.\]
The set of all correlations that arise from strategies of this form is denoted 
\[ C_q(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{A}},\ensuremath{\mathcal{B}}) \subseteq \ensuremath{\mathbb{R}}^{|\ensuremath{\mathcal{X}}||\ensuremath{\mathcal{Y}}||\ensuremath{\mathcal{A}}||\ensuremath{\mathcal{B}}|}\;.\]
<!-- end definition-1 -->
<!-- definition-2: body_md -->
For finite sets $\ensuremath{\mathcal{X}}$ and $\ensuremath{\mathcal{A}}$ a correlation $C= (C_{x,y,a,b}) \in C_q(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{X}},\ensuremath{\mathcal{A}},\ensuremath{\mathcal{A}})$ is called \emph{synchronous} if $C_{x,x,a,b} = 0$ for all $x\in \ensuremath{\mathcal{X}}$ and $a,b\in \ensuremath{\mathcal{A}}$ such that $a\neq b$. Given a distribution $\nu$ on $\ensuremath{\mathcal{X}}$ recall the definition of $\delta_{\textrm{sync}}(C;\nu)$ in~\eqref{eq:delta-sync}. 
Given a strategy $\mathscr{S}$ we also write $\delta_{\textrm{sync}}(\mathscr{S};\nu)$ for $\delta_{\textrm{sync}}(C;\nu) $ where $C$ is the correlation induced by $\mathscr{S}$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
A strategy $\ensuremath{\mathcal{S}} = (|\psi\rangle,A,B)$ is \emph{symmetric} if $\ensuremath{\mathcal{X}}=\ensuremath{\mathcal{Y}}$, $\ensuremath{\mathcal{A}}=\ensuremath{\mathcal{B}}$,   $|\psi\rangle\in \mathcal{H}\otimes \mathcal{H}$ takes the form
\begin{equation}\label{eq:psi-sym}
 |\psi\rangle \,=\, \sum_i \sqrt{\lambda_i} |u_i\rangle |u_i\rangle \,\in\,\mathcal{H}\otimes \mathcal{H}\;,
\end{equation}
where the $\lambda_i$ are non-negative and $\{|u_i\rangle\}$ orthonormal, and for every $x,a$, $A^x_a=(B^x_a)^T$ with the transpose being taken with respect to the $\{|u_i\rangle\}$. Note that this implies that $\mathcal{H}_{\textsf{A}}=\mathcal{H}_{\textsf{B}}$ and that $|\psi\rangle$ has the same reduced density on either subsystem. For a symmetric strategy we write it as $\mathscr{S} = (|\psi\rangle,A)$. A strategy is \emph{projective} if all $A^x_a$ and $B^y_b$ are projections. It is \emph{maximally entangled} if $|\psi\rangle$ is a maximally entangled state~\eqref{eq:def-me} on $\mathcal{H}_{\textsf{A}}\otimes \mathcal{H}_{\textsf{B}}$. We use the acronym ``PME'' to denote ``symmetric projective maximally entangled''.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:dilation} Given $\varepsilon \geq 0$, a distribution $\nu$ on $\ensuremath{\mathcal{X}}\times \ensuremath{\mathcal{Y}}$ and two strategies $\mathscr{S} = (|\psi\rangle,A,B)$ and $\tilde{\mathscr{S}} = (|\tilde{\psi}\rangle,\tilde{A},\tilde{B})$ we say that $\tilde{\mathscr{S}}$ is a \emph{local $(\varepsilon,\nu)$-dilation} of $\mathscr{S}$ if there exists isometries $V_A: \mathcal{H}_A \to \tilde{\mathcal{H}}_A \otimes \ensuremath{\mathcal{K}}_A$ and $V_B: \mathcal{H}_B \to \tilde{\mathcal{H}}_B \otimes \ensuremath{\mathcal{K}}_B$ and a state $|\mathsf{aux}\rangle \in \ensuremath{\mathcal{K}}_A \otimes \ensuremath{\mathcal{K}}_B$ such that 
\begin{align*}
 \big\|(V_A \otimes V_B) |\psi\rangle  - |\tilde{\psi}\rangle \otimes |\mathsf{aux}\rangle \big\| \,&\leq\, \varepsilon\;,\\
\Big(\ensuremath{\mathop{\textsc{E}}_{(x,y)\sim \nu}} \sum_{a,b} \big\| (V_A\otimes V_B) \big( A^x_a \otimes B^y_b \big) |\psi\rangle -\big( \big(\tilde{A}^x_a \otimes \tilde{B}^y_b\big) |\tilde{\psi}\rangle \big)\otimes |\mathsf{aux}\rangle\big\|^2 \Big)^{1/2}\,&\leq\, \varepsilon\;.
\end{align*}

 In addition $|\mathsf{aux}\rangle$ is a \emph{product state},

<!-- definition-5: body_md -->
A \emph{nonlocal game} (or \emph{game} for short) $\mathfrak{G}$ is specified by a tuple $(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{A}},\ensuremath{\mathcal{B}},\nu,D)$ of finite \emph{question sets} $\ensuremath{\mathcal{X}}$ and $\ensuremath{\mathcal{Y}}$, finite \emph{answer sets} $\ensuremath{\mathcal{A}}$ and $\ensuremath{\mathcal{B}}$, a distribution $\nu$ on $\ensuremath{\mathcal{X}}\times\ensuremath{\mathcal{Y}}$ and a \emph{decision predicate} $D:\ensuremath{\mathcal{X}}\times \ensuremath{\mathcal{Y}}\times \ensuremath{\mathcal{A}}\times \ensuremath{\mathcal{B}}\to \{0,1\}$ that we conventionally write as $D(a,b|x,y)$ for $(x,y)\in \ensuremath{\mathcal{X}}\times \ensuremath{\mathcal{Y}}$ and $(a,b)\in \ensuremath{\mathcal{A}}\times \ensuremath{\mathcal{B}}$. The game is \emph{symmetric} if $\ensuremath{\mathcal{X}}=\ensuremath{\mathcal{Y}}$, $\ensuremath{\mathcal{A}}=\ensuremath{\mathcal{B}}$, $\nu(x,y)=\nu(y,x)$ for all $x,y\in \ensuremath{\mathcal{X}}\times\ensuremath{\mathcal{Y}}$, and for all $a,b,x,y$, $D(a,b|x,y)=D(b,a|y,x)$. In this case we write $\mathfrak{G} = (\ensuremath{\mathcal{X}},\ensuremath{\mathcal{A}},\nu,D)$. We often abuse notation and also use $\nu$ to denote the marginal distribution of $\nu$ on $\ensuremath{\mathcal{X}}$.

<!-- definition-6: body_md -->
Given a game $\mathfrak{G}=(\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{A}},\ensuremath{\mathcal{B}},\nu,D)$ and a strategy $\mathscr{S} = (|\psi\rangle,A)$ in $\mathfrak{G}$, the \emph{success probability} of $\mathscr{S}$ in $\mathfrak{G}$ is 
\[\omega(\mathfrak{G};\mathscr{S})\,=\, \ensuremath{\mathop{\textsc{E}}_{(x,y)\sim \nu}} \sum_{a,b} D(a,b|x,y) \langle\psi| A^x_a \otimes B^y_b |\psi\rangle\;.\]

<!-- lemma-2: body_md -->
\label{lem:am-cons}
Let $\gamma,\delta \geq 0$. Let $\mathcal{H}$ be a Hilbert space and $|\psi\rangle\in\mathcal{H}\otimes \mathcal{H}$ a state. Let $\ensuremath{\mathcal{X}}$ be a finite set and for each $x\in\ensuremath{\mathcal{X}}$, $\{A^x_a\}_{a\in\ensuremath{\mathcal{A}}}$ a  projective measurement and $\{M^x_a\}_{a\in\ensuremath{\mathcal{A}}}$ an arbitrary measurement on $\mathcal{H}$. Let $\mu$ be a distribution on $\ensuremath{\mathcal{X}}$. Let
\begin{equation}\label{eq:am-cons-0a}
\delta \,=\,1-\ensuremath{\mathop{\textsc{E}}_{x\sim \mu}} \sum_{a\in\ensuremath{\mathcal{A}}} \langle\psi| A^x_a \otimes (A^x_a)^T |\psi\rangle\quad\text{and}\quad \gamma = 1 -  \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_{a\in\ensuremath{\mathcal{A}}} \langle\psi| A^x_a \otimes (M^x_a)^T |\psi\rangle\;.
\end{equation}
Then 
\begin{equation}\label{eq:am-cons-0b}
(\gamma - \delta)^2 \,\leq\, \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}} \sum_{a\in\ensuremath{\mathcal{A}}} \langle\psi|\ensuremath{\mathop{\rm Id}\nolimits} \otimes \big( A^x_a - M^x_a \big)^2 |\psi\rangle \,\leq\, 2\,\gamma + 2\sqrt{2\delta}\;.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
We start with the left inequality:
\begin{align*}
\gamma &= 1 -  \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| A^x_a \otimes (M^x_a)^T |\psi\rangle\\
&= \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| A^x_a \otimes \big(A^x_a-M^x_a\big)^T |\psi\rangle + 1 - \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| A^x_a \otimes (A^x_a)^T |\psi\rangle\\
&\leq \Big(\ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| \ensuremath{\mathop{\rm Id}\nolimits} \otimes \big(A^x_a-M^x_a\big)^2 |\psi\rangle\Big)^{1/2}\Big(\ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| (A^x_a)^2 \otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi\rangle\Big)^{1/2} + \delta\\
&\leq \Big(\ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| \ensuremath{\mathop{\rm Id}\nolimits} \otimes \big(A^x_a-M^x_a\big)^2 |\psi\rangle\Big)^{1/2} + \delta \;,
\end{align*}
where the first inequality is Cauchy-Schwarz and the last uses that for each $x\in\ensuremath{\mathcal{X}}$, $\{A^x_a\}_a$ is a measurement. 

For the right inequality, 
\begin{align*}
\ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a &\langle\psi| \ensuremath{\mathop{\rm Id}\nolimits} \otimes \big(A^x_a-M^x_a\big)^2 |\psi\rangle\\
&= \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| \ensuremath{\mathop{\rm Id}\nolimits} \otimes\big( (A^x_a)^2 + (M^x_a)^2 \big)|\psi\rangle - 2\Re\Big(  \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| \ensuremath{\mathop{\rm Id}\nolimits} \otimes A^x_a M^x_a |\psi\rangle\Big)\\
&\leq 2 - 2\Re\Big(  (1-\gamma)-  \ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| \big(A^x_a \otimes \ensuremath{\mathop{\rm Id}\nolimits} - \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T\big)(\ensuremath{\mathop{\rm Id}\nolimits}\otimes  (M^x_a)^T) |\psi\rangle\Big)\\
&\leq 2\,\gamma + 2\Big(\ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| \big(A^x_a \otimes \ensuremath{\mathop{\rm Id}\nolimits} - \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T\big)^2 |\psi\rangle\Big)^{1/2}\Big(\ensuremath{\mathop{\textsc{E}}_{x\sim \mu}}\sum_a \langle\psi| \ensuremath{\mathop{\rm Id}\nolimits}\otimes ((M^x_a)^T)^2 |\psi\rangle\Big)^{1/2}\\
&\leq 2\,\gamma + 2\sqrt{2\delta}\;,
\end{align*}
where the first inequality uses that $\{A^x_a\}_a$ and $\{M^x_a\}_a$ are measurements, the second uses the Cauchy-Schwarz inequality and the last the definition of $\delta$ and that for each $x\in\ensuremath{\mathcal{X}}$, $\{A^x_a\}$ is a projective measurement and $\{M^x_a\}_a$ is a measurement.

<!-- lemma-3: body_md -->
\label{lem:psi-cs}
Let $|\psi\rangle\in \mathcal{H}_A \otimes \mathcal{H}_B$ and $\{A_a\}$ and $\{B_a\}$ measurements on $\mathcal{H}_A$ and $\mathcal{H}_B$ respectively. Let $\rho_A$ and $\rho_B$ be the reduced density of $|\psi\rangle$ on $\mathcal{H}_A$ and $\mathcal{H}_B$ respectively. Let $|\psi_A\rangle\in\mathcal{H}_A\otimes\mathcal{H}_A$ and $|\psi_B\rangle\in\mathcal{H}_B\otimes\mathcal{H}_B$ be the canonical purifications of $\rho_A$ and $\rho_B$ respectively. Then 
\[ \sum_a \langle\psi| A_a \otimes B_a |\psi\rangle \,\leq\,\Big( \sum_a \langle\psi_A| A_a \otimes A_a^T |\psi_A\rangle \Big)^{1/2}\Big( \sum_a \langle\psi_B| B_a \otimes B_a^T |\psi_B\rangle \Big)^{1/2}\;.\]
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Let $|\psi\rangle = \sum_j \lambda_j |u_j\rangle |v_j\rangle$ be the Schmidt decomposition. Let $K = \sum_j \lambda_j |u_j\rangle\langlev_j|$. Then 
\begin{align*}
 \sum_a \langle\psi| A_a \otimes B_a |\psi\rangle 
&= \sum_a \mbox{\rm Tr}\big( A_a K \overline{B}_a K^\dagger \big)\\
&\leq \Big(\sum_a \mbox{\rm Tr}\big( A_a \sqrt{KK^\dagger} A_a \sqrt{KK^\dagger}\big) \Big)^{1/2} \Big(\sum_a \mbox{\rm Tr}\big( \overline{B_a} \sqrt{K^\dagger K } \overline{B_a} \sqrt{K^\dagger K }\big) \Big)^{1/2}\;,
\end{align*}
where the inequality is Cauchy-Schwarz. Using that $\rho_A = KK^\dagger$ and $\rho_B = K^\dagger K$, this concludes the proof.

<!-- lemma-4: body_md -->
\label{lem:close-cor}
Let $\mathscr{S} = (|\psi\rangle,A,B)$ be  a strategy, $\hat{A}=\{\hat{A}^x_a\}$ a family of POVM on $\mathcal{H}_A$, and let $\hat{\mathscr{S}} = (|\psi\rangle,\hat{A},B)$. Let $\rho_A$ be the reduced density of $|\psi\rangle$ on $\mathcal{H}_A$ and $|\psi_A\rangle \in \mathcal{H}_A \otimes \mathcal{H}_A$ the canonical purification of it. Let $\mathscr{S}_A = (|\psi_A\rangle,A)$. 
Let $\nu$ be a distribution on $\ensuremath{\mathcal{X}}\times\ensuremath{\mathcal{Y}}$ and $\delta = \delta_\textrm{sync}(\mathscr{S}_A;\nu_A)$ where $\nu_A$ is the marginal of $\nu$ on $\ensuremath{\mathcal{X}}$. Let 
\[\gamma \,=\, \ensuremath{\mathop{\textsc{E}}_{x\sim\nu_A}} \,\sum_a \,\mbox{\rm Tr}\big( \big(A^x_a - \hat{A}^x_a \big)^2 \rho_A \big)\;.\]
Let $C$ be the correlation induced by $\mathscr{S}$ and $\hat{C}$ by $\hat{\mathscr{S}}$. Then 
\[ \ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| C_{x,y,a,b} - \hat{C}_{x,y,a,b}\big| \,\leq\, O\big(\delta + \sqrt{\gamma}\big) \;.\]
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Conjugating the $B^y_b$ by a unitary if necessary we assume without loss of generality that the reduced densities of $|\psi\rangle$ on either subsystem satisfy $\rho_A = \rho_B$. Then $|\psi_A\rangle = |\psi\rangle$. 
As a first step in the proof we show that 
\begin{equation}\label{eq:cc-1}
\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| C_{x,y,a,b} - \langle\psi| (A^x_a)^2 \otimes  B^y_b  |\psi\rangle\big| \,\leq\, \delta\;.
\end{equation}
To show this, write
\begin{align}
\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| C_{x,y,a,b} - \langle\psi| (A^x_a)^2 \otimes  B^y_b  |\psi\rangle\big| &= 
\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \langle\psi| \big(A^x_a -  (A^x_a)^2 \big)\otimes  B^y_b  |\psi\rangle \notag\\
&\leq \ensuremath{\mathop{\textsc{E}}_{x\sim\nu_A}} \sum_a  \langle\psi| \big(A^x_a -  (A^x_a)^2 \big) \otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi\rangle\notag\\
&= 1 - \ensuremath{\mathop{\textsc{E}}_{x\sim\nu_A}} \sum_a  \langle\psi| (A^x_a)^2 \otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi\rangle\;,\label{eq:cc-2}
\end{align}
where the first step uses that $A^x_a -  (A^x_a)^2 \geq 0$ for all $x,a$, the second that $\sum_b B^y_b = \ensuremath{\mathop{\rm Id}\nolimits}$ for all $y$ and the third that $\sum_a A^x_a = \ensuremath{\mathop{\rm Id}\nolimits}$ for all $x$. Next we observe that 
\begin{align*}
1-\delta &= \ensuremath{\mathop{\textsc{E}}_{x\sim\nu_A}} \sum_a  \langle\psi| A^x_a\otimes (A^x_a)^T |\psi\rangle\\
&\leq \Big(\ensuremath{\mathop{\textsc{E}}_{x\sim\nu_A}} \sum_a  \langle\psi| (A^x_a)^2\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi\rangle\Big)^{1/2}\Big(\ensuremath{\mathop{\textsc{E}}_{x\sim\nu_A}} \sum_a  \langle\psi| \ensuremath{\mathop{\rm Id}\nolimits} \otimes ((A^x_a)^T)^2|\psi\rangle\Big)^{1/2}\\
&= \ensuremath{\mathop{\textsc{E}}_{x\sim\nu_A}} \sum_a  \mbox{\rm Tr}\big( (A^x_a)^2\rho_A \big)\;,
\end{align*}
where the inequality on the second line is Cauchy-Schwarz and the last line uses our assumption that $\rho_A = \rho_B$. Plugging back into~\eqref{eq:cc-2}, this shows~\eqref{eq:cc-1}. For the second step we show
\begin{equation}\label{eq:cc-3}
\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| \langle\psi| (A^x_a)^2 \otimes  B^y_b  |\psi\rangle - \langle\psi| (\hat{A}^x_a)^2 \otimes  B^y_b  |\psi\rangle\big| \,\leq\, 2\sqrt{\gamma}\;.
\end{equation}
To show~\eqref{eq:cc-3} we first bound
\begin{align}
\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| \langle\psi| \big((A^x_a)^2 &- A^x_a \hat{A}^x_a \big)\otimes  B^y_b  |\psi\rangle \big|\notag\\
&\leq \Big(\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| \langle\psi| (A^x_a)^2 \otimes  B^y_b  |\psi\rangle  \Big)^{1/2}\Big(\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| \langle\psi| (A^x_a-\hat{A}^x_a)^2 \otimes  B^y_b  |\psi\rangle  \Big)^{1/2} \notag\\
&\leq \sqrt{\gamma}\;,\label{eq:cc-4a}
\end{align}
where the first inequality is Cauchy-Schwarz and the second bounds the first term by $1$ and the second by $\gamma$ using $\sum_b B^y_b = \sum_a A^x_a = 1$ and the definition of $\gamma$. An analogous calculation gives
\begin{equation}\label{eq:cc-4b}
\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| \langle\psi| \big(A^x_a \hat{A}^x_a - (\hat{A}^x_a)^2  \big)\otimes  B^y_b  |\psi\rangle \big| \,\leq\, \sqrt{\gamma}\;.
\end{equation}
Together,~\eqref{eq:cc-4a} and~\eqref{eq:cc-4b} give~\eqref{eq:cc-3}. Finally, the third step of the proof is given by the bound 
\begin{equation}\label{eq:cc-5}
\ensuremath{\mathop{\textsc{E}}_{x,y\sim \nu}} \sum_{a,b} \big| \hat{C}_{x,y,a,b} - \langle\psi| (\hat{A}^x_a)^2 \otimes  B^y_b  |\psi\rangle\big| \,\leq\, 2\big(\sqrt{\gamma} + \delta\big)\;.
\end{equation}
This is analogous to~\eqref{eq:cc-1} except that we rely on an estimate for consistency $\hat{\delta}$ of the $\{\hat{A}^x_a\}$. This can be obtained directly by using the left inequality of~\eqref{eq:am-cons-0b} in Lemma~\ref{lem:am-cons}, which letting $\eta = 1-\ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \langle\psi| A^x_a \otimes (\hat{A}^x_a)^T |\psi\rangle$ gives $(\eta-\delta)^2 \leq \gamma$ so 
\begin{equation}\label{eq:cc-6}
\eta \leq \sqrt{\gamma} + \delta
\end{equation}
 and 
\begin{align*}
1-\eta &= \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \langle\psi| A^x_a \otimes (\hat{A}^x_a)^T |\psi\rangle\\
&\leq \Big( \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \langle\psi| A^x_a \otimes (A^x_a)^T |\psi\rangle\Big)^{1/2}\Big( \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \langle\psi| \hat{A}^x_a \otimes (\hat{A}^x_a)^T |\psi\rangle\Big)^{1/2}\;.
\end{align*}
using Lemma~\ref{lem:psi-cs}. Thus $(1-\delta)(1-\hat{\delta}) \geq (1-\eta)^2$ which implies $\hat{\delta} \leq 2\eta \leq 2\sqrt{\gamma}$ by~\eqref{eq:cc-6}. Proceeding as for~\eqref{eq:cc-1}, this shows~\eqref{eq:cc-5}. 

Combining~\eqref{eq:cc-2},~\eqref{eq:cc-3} and~\eqref{eq:cc-5} proves the lemma.

<!-- proposition-1: body_md -->

<!-- theorem-1: body_md -->
\label{thm:main}
There are universal constants $c,C>0$ such that the following holds. Let $\ensuremath{\mathcal{X}}$ and $\ensuremath{\mathcal{A}}$ be finite sets and $\nu$ a distribution on $\ensuremath{\mathcal{X}}$. Let $\mathscr{S}=(|\psi\rangle,A)$ be a symmetric strategy and $\delta = \delta_\textrm{sync}(\mathscr{S};\nu)$.
 Then there is a measure $\mu$ on $\mathbb{R}_+$ and a family of Hilbert spaces $\mathcal{H}_\lambda \subseteq \mathcal{H}$, for $\lambda\in\mathbb{R}_+$ (both depending on $|\psi\rangle$ only) such that the following holds. 
For every $\lambda\in \ensuremath{\mathbb{R}}_+$ there is a maximally entangled state $|\psi_\lambda\rangle\in \mathcal{H}_\lambda\otimes \mathcal{H}_\lambda$ and for each $x$ a projective measurement $\{A^{\lambda,x}_a\}$ on $\mathcal{H}_\lambda$ such that 
\begin{enumerate}
\item Letting $\rho$ be the reduced density of $|\psi\rangle$ on $\mathcal{H}$ and $\rho_\lambda$ the totally mixed state on $\mathcal{H}_\lambda\subseteq\mathcal{H}$,
\begin{equation}\label{eq:rho-plambda}
 \rho \,=\, \int_\lambda \rho_\lambda d\lambda\;.
\end{equation}
\item The $\{\mathscr{S}_\lambda = (|\psi_\lambda\rangle,A^\lambda)\}$ provide an approximate decomposition of $\mathscr{S}$ as a convex sum of projective maximally entangled (PME) strategies, in the following sense: 
\begin{equation}\label{eq:lambda-strat-0}
\ensuremath{\mathop{\textsc{E}}_{x\sim \nu}} \sum_a \int_\lambda \mbox{\rm Tr}\big( \big(A^x_a - A^{\lambda,x}_a\big)^2 \rho_\lambda\big) d\mu(\lambda)\,\leq\,C \,\delta^c\;.
\end{equation}
\end{enumerate}

<!-- corollary-2: body_md -->
\label{cor:c2}
Let $\mathscr{S}=(|\psi\rangle,A,B)$ be a projective strategy. Let $\nu$ be a distribution on $\ensuremath{\mathcal{X}}$, $\delta = \delta_\textrm{sync}(\mathscr{S};\nu)$, and $\{\mathscr{S}_\lambda\}$ and $\mu$ the family of strategies and the measure obtained from Theorem~\ref{thm:main}. Let $C$ (resp. $C^\lambda$) be the correlation induced by $\mathscr{S}$ (resp. $\mathscr{S}_\lambda$). Let $\tilde{\nu}$ be any distribution on $\ensuremath{\mathcal{X}} \times \ensuremath{\mathcal{X}}$ with marginal $\nu$. Then 
\begin{equation}\label{eq:c2-0}
\ensuremath{\mathop{\textsc{E}}_{(x,y)\sim \tilde{\nu}}}\; \sum_{a,b}\; \Big| C_{x,y,a,b} -  \int_\lambda C^\lambda_{x,y,a,b} d\lambda\Big| \,=\, \operatorname{poly}(\delta)\;.
\end{equation}
<!-- end corollary-2 -->
<!-- proof-5: body_md -->
Fix $\mathscr{S}$, $\nu$, $\tilde{\nu}$ and $\{\mathscr{S}^\lambda\}$, $\mu$ as in the statement of the corollary. Conjugating the $B^y_b$ by a unitary if necessary we assume without loss of generality that the reduced densities of $|\psi\rangle$ on either subsystem are identical.
 For every $\lambda$ define a symmetric strategy $\tilde{\mathscr{S}}_\lambda = (|\psi_\lambda\rangle,A)$ and let $\tilde{C}^\lambda$ be the associated correlation. We first show that 
\begin{equation}\label{eq:c2-1}
 \ensuremath{\mathop{\textsc{E}}_{(x,y)\sim \tilde{\nu}}} \sum_{a,b} \big| C_{x,y,a,b} - \int_\lambda\tilde{C}^\lambda_{x,y,a,b}d\lambda\big |  \,=\, \operatorname{poly}(\delta)\;.
\end{equation}
                                                                                  
For this we show that 
\begin{equation}\label{eq:c2-1b}
\ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b} \big| C_{x,y,a,b} - \mbox{\rm Tr}\big(  (A^x_a)^T B^y_b(A^x_a)^T \rho \big)\big|\,=\, O\big(\sqrt{\delta}\big)\;,
\end{equation}
and
\begin{equation}\label{eq:c2-1c}
\int_\lambda \ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b} \big| \tilde{C}^\lambda_{x,y,a,b} - \mbox{\rm Tr}\big( (A^x_a)^T B^y_b (A^x_a)^T\rho_\lambda \big)\big|\,=\, O\big({\delta}^{c/4}\big)\;.
\end{equation}
Together with~\eqref{eq:rho-plambda}, combining~\eqref{eq:c2-1b} and~\eqref{eq:c2-1c} through the triangle inequality gives~\eqref{eq:c2-1}. To show~\eqref{eq:c2-1b}, write using the triangle inequality
\begin{align}
\ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b} \big| C_{x,y,a,b} - \mbox{\rm Tr}\big( (A^x_a)^T B^y_b (A^x_a)^T\rho \big)\big|
&\leq \ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b} \big| \langle\psi| \big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes(A^x_a)^T- A^x_a\otimes \ensuremath{\mathop{\rm Id}\nolimits}\big) B^y_b (\ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T )|\psi\rangle \big| \label{eq:c2-2a}\\
&\quad + \ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b} \langle\psi| \big(\ensuremath{\mathop{\rm Id}\nolimits}\otimes (A^x_a)^T \big)B^y_b \big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T- A^x_a \otimes \ensuremath{\mathop{\rm Id}\nolimits}\big)|\psi\rangle\big|\;,\label{eq:c2-2b}
\end{align}
where we used the assumption that each $B^y_b$ is a projection. 
Each of the two terms on the right-hand side is bounded in the same manner. We show how to bound the first: 
\begin{align}
\ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b} \big| \langle\psi| \big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes(A^x_a)^T &- A^x_a\otimes \ensuremath{\mathop{\rm Id}\nolimits}\big) B^y_b (\ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T )|\psi\rangle \big|\notag\\
&\leq \Big(\ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b}  \langle\psi| \big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T - A^x_a\otimes \ensuremath{\mathop{\rm Id}\nolimits}\big) B^y_b \big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T - A^x_a\otimes \ensuremath{\mathop{\rm Id}\nolimits}\big)|\psi\rangle \Big)^{1/2}\notag\\
&\qquad\qquad\cdot \Big(\ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_{a,b}  \langle\psi| ( \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T) B^y_b ( \ensuremath{\mathop{\rm Id}\nolimits} \otimes(A^x_a)^T )|\psi\rangle \Big)^{1/2} \notag\\
&\leq \Big( \ensuremath{\mathop{\textsc{E}}_{x}} \sum_a \langle\psi|\big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T- A^x_a\otimes \ensuremath{\mathop{\rm Id}\nolimits}\big)^2|\psi\rangle \Big)^{1/2} \cdot 1\label{eq:c2-2e}\\
&\leq \sqrt{2\delta}\;,\notag
\end{align}
where the first inequality is Cauchy-Schwarz, the second uses $\sum_a A^x_a = \sum_b (B^y_b)^2 = \ensuremath{\mathop{\rm Id}\nolimits}$, and the last follows by expanding the square and using the definition of $\delta$. This bounds~\eqref{eq:c2-2a}. Together with a similar bound for the term in~\eqref{eq:c2-2b} this shows~\eqref{eq:c2-1b}. To show~\eqref{eq:c2-1c}, we proceed similarly up until the last step~\eqref{eq:c2-2e}, at which point a bound on
\begin{equation}\label{eq:c2-2e1}
\int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}} \sum_a \langle\psi_\lambda|\big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^x_a)^T- A^x_a\otimes \ensuremath{\mathop{\rm Id}\nolimits}\big)^2|\psi_\lambda\rangle
\end{equation}
is required. To obtain this, we first note that 
\begin{equation}\label{eq:c2-2e2}
\int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}} \sum_a \langle\psi_\lambda|\big( \ensuremath{\mathop{\rm Id}\nolimits} \otimes (A^{\lambda,x}_a)^T- A^{\lambda,x}_a\otimes \ensuremath{\mathop{\rm Id}\nolimits}\big)^2|\psi_\lambda\rangle\,=\,0\;,
\end{equation}
because $A^{\lambda,x}_a$ is supported on the support of $\rho_\lambda$, which is totally mixed on its support. 
Moreover, forming the difference we have (using that $\{A^{\lambda,x}_a\}$ and $\{A^x_a\}$ are projective)
\begin{align*}
|\eqref{eq:c2-2e1}-\eqref{eq:c2-2e2} |
&= 2 \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}} \sum_a \Big(\langle\psi_\lambda| (A^{\lambda,x}_a - A^{x}_a)\otimes (A^{\lambda,x}_a)^T|\psi_\lambda\rangle + \langle\psi_\lambda| A^{x}_a \otimes (A^{\lambda,x}_a- A^{x}_a)^T\otimes |\psi_\lambda\rangle \Big)\;.
\end{align*}
Each of the two terms on the right-hand side is bounded by an application of the Cauchy-Schwarz inequality followed by~\eqref{eq:lambda-strat-0}. This shows~\eqref{eq:c2-1b}, and hence~\eqref{eq:c2-1}.

Having established~\eqref{eq:c2-1} we now prove
\begin{equation}\label{eq:c2-3}
  \int_{\lambda}\ensuremath{\mathop{\textsc{E}}_{(x,y)\sim \tilde{\nu}}} \sum_{a,b} \big| \tilde{C}^\lambda_{x,y,a,b} -{C}^\lambda_{x,y,a,b}\big | d\lambda \,=\, \operatorname{poly}(\delta)\;.
\end{equation}
Combining~\eqref{eq:c2-1} and~\eqref{eq:c2-3} shows~\eqref{eq:c2-0}, concluding the proof. To show~\eqref{eq:c2-3} we apply Lemma~\ref{lem:close-cor} for each $\lambda$ to the strategy $\mathscr{S} = (|\psi_\lambda\rangle,A^\lambda,B)$ here and $\hat{A}$ in Lemma~\ref{lem:close-cor} is $A$ here. Since $|\psi_\lambda\rangle$ is maximally entangled and $A^\lambda$ supported on its support, $\delta$ in Lemma~\ref{lem:close-cor} equals $0$. Applying the lemma followed by Jensen's inequality gives 
\begin{align*}
 \int_{\lambda}\ensuremath{\mathop{\textsc{E}}_{(x,y)\sim \tilde{\nu}}} \sum_{a,b} \big| \tilde{C}^\lambda_{x,y,a,b} -{C}^\lambda_{x,y,a,b}\big | d\lambda
&= O\Big(\Big( \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}} \sum_a \mbox{\rm Tr}\big( \big( A^{\lambda,x}_a - A^x_a\big)^2 \rho_\lambda \big)d_\lambda  \Big)^{1/2}\Big)\\
&= \operatorname{poly}(\delta)\;,
\end{align*}
by~\eqref{eq:lambda-strat-0}. This shows~\eqref{eq:c2-3} and concludes the proof.

<!-- lemma-7: body_md -->
\label{lem:main-step}
There is a projective symmetric strategy $\mathscr{S}'= (|\psi\rangle,B)$ such that letting $\delta'=\delta_\textrm{sync}(\mathscr{S}',\nu)$ 
then $\delta'= O(\delta^{1/8})$ and 
\begin{equation}\label{eq:main-step-0}
\ensuremath{\mathop{\textsc{E}}_{x\sim\nu}} \sum_a \mbox{\rm Tr}\big( (A^x_a - B^x_a)^2 \rho\big) \,=\, O\big(\delta^{1/4}\big)\;.
\end{equation}
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
For each $x$ let $\delta_x = 1-\sum_a \langle\psi| A^x_a \otimes (A^x_a)^T |\psi\rangle$. By definition of $\delta$ it holds that 
\begin{equation}\label{eq:main-step-1a}
\delta \,=\, \ensuremath{\mathop{\textsc{E}}_{x\sim \nu}} \delta_x\;.
\end{equation}
 For each $x\in \ensuremath{\mathcal{X}}$, applying Proposition~\ref{prop:ortho} to the measurement $\{A^x_a\}$ gives a projective measurement $\{B^x_a\}$ such that 
\[\sum_a \mbox{\rm Tr}\big( (A^x_a - B^x_a)^2 \rho\big) \,=\, O\big(\delta_x^{1/4}\big)\;.\]
Taking the expectation over $x$,
\begin{align}
\ensuremath{\mathop{\textsc{E}}_{x\sim\nu}} \sum_a \mbox{\rm Tr}\big( (A^x_a - B^x_a)^2 \rho\big) &= O\Big( \ensuremath{\mathop{\textsc{E}}_{x\sim\nu}} \delta_x^{1/4}\Big)\notag\\
&= O\Big( \Big( \ensuremath{\mathop{\textsc{E}}_{x\sim\nu}} \delta_x \Big)^{1/4}\Big)\notag\\
&= O\big(\delta^{1/4}\big)\;,\label{eq:main-step-1}
\end{align}
where the second line uses Jensen's inequality and the third uses~\eqref{eq:main-step-1a}. This gives~\eqref{eq:main-step-0}. Let $\delta'=\delta_\textrm{sync}(\mathscr{S}',\nu)$. Then 
\begin{align*}
\delta'-\delta &= \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \big( \langle\psi| A^x_a \otimes(A^x_a)^T |\psi\rangle - \langle\psi| B^x_a \otimes (B^x_a)^T |\psi\rangle \big)\\
&= \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a  \big(\langle\psi| (A^x_a - B^x_a) \otimes (A^x_a)^T |\psi\rangle + \langle\psi| B^x_a \otimes (A^x_a-B^x_a)^T|\psi\rangle \big)\\
&\leq \Big(\ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}\big((A^x_a - B^x_a)^2\rho\big)\Big)^{1/2} \Big( \Big( \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}( (A^x_a)^2 \rho )\Big)^{1/2} 
+  \Big( \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}( (B^x_a)^2 \rho )\Big)^{1/2}\Big)\\
&\leq   O\big(\delta^{1/8}\big)\cdot \sqrt{2}\;,
\end{align*}
where the second inequality follows from the Cauchy-Schwarz inequality and the last uses~\eqref{eq:main-step-1} to bound the first term and that for each $x$, $\{A^x_a\}_a$ and $\{B^x_a\}$ are measurements.  This shows $\delta'= O(\delta^{1/8})$, as claimed.

For convenience in the proof of the lemma 
we identify the set $\ensuremath{\mathcal{A}}$ with $\mathbb{Z}_m$, for some integer $m$. Define a family of unitaries $\{U^x_b\}$ indexed by $x\in\ensuremath{\mathcal{X}}$ and $b\in \ensuremath{\mathcal{A}}$ by 
\begin{equation}\label{eq:def-ux}
U^x_b \,=\, \sum_a e^{2i\pi ab/m} B^x_a\;.
\end{equation}
With this definition we observe that
\begin{align}
\ensuremath{\mathop{\textsc{E}}_{x}} \ensuremath{\mathop{\textsc{E}}_{b}} \big\| U^x_b \rho^{1/2} - \rho^{1/2} U^x_b \big\|_F^2 
&= 2 - 2 \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}(B^x_a \rho^{1/2} B^x_a \rho^{1/2})\notag\\
&= 2\delta'\;,\label{eq:game-succ-0}
\end{align}
where the expectation over $x$ is taken with respect to the (marginal of) the game distribution $\nu$, the expectation over $b$ is uniform over $\ensuremath{\mathbb{Z}}_m$, the first equality uses the equality $\ensuremath{\mathop{\textsc{E}}_{b}} e^{2i\pi(a-a')/m} = \delta_{a,a'}$ (the Kronecker $\delta$) for all $a,a'\in\mathbb{Z}_m$ and the fact that for every $x$, $\{B^x_a\}_a$ is projective, and the second uses the identity~\eqref{eq:ando}.

For each $x\in \ensuremath{\mathcal{X}}$ and $b\in\ensuremath{\mathcal{B}}$ let $\sigma^x_b = (U_b^x)^\dagger \rho {U}_{b}^{x} $. Observe that for any $\lambda \in \ensuremath{\mathbb{R}}_+$,
\[ \chi_{\geq \sqrt{\lambda}} \big((\sigma^x_b)^{1/2}\big) \,=\,(U_b^x)^\dagger \,\chi_{\geq \sqrt{\lambda}}\big(\rho^{1/2}\big)\, {U}_{b}^{x}\;.\]
Hence using the definition~\eqref{eq:def-plambda} of $P_\lambda$, 
\begin{align*}
\ensuremath{\mathop{\textsc{E}}_{x}} \ensuremath{\mathop{\textsc{E}}_{b}}\,\int_\lambda  \big\|  P_\lambda - ({U}_{b}^{x})^\dagger P_\lambda {U}_{b}^{x} \big\|_F^2 
&= \ensuremath{\mathop{\textsc{E}}_{x}} \ensuremath{\mathop{\textsc{E}}_{b}}\,\int_\lambda  \big\|  \chi_{\geq \sqrt{\lambda}} \big(\rho^{1/2}\big)- \chi_{\geq \sqrt{\lambda}} \big((\sigma^x_b)^{1/2}\big) \big\|_F^2 \\
&\leq \ensuremath{\mathop{\textsc{E}}_{x}} \ensuremath{\mathop{\textsc{E}}_{b}} \big\|  \rho^{1/2} -  (U_b^x)^\dagger \rho^{1/2} {U}_{b}^{x} \big\|_F \big\|\rho^{1/2} +  (U_b^x)^\dagger \rho^{1/2} {U}_{b}^{x} \big\|_F\\

&\leq\sqrt{2\delta'}\sqrt{4}\;,
\end{align*}
where the inequality on the second line follows from applying
Lemma~\ref{lem:connestrick} independently for each $x$ and $b$, the third line is the Cauchy-Schwarz inequality, and

\[ \ensuremath{\mathop{\textsc{E}}_{b}}\,\big\| U^x_b P_\lambda -  P_\lambda {U}_{b}^{x} \big\|_F^2 
\,=\, \sum_a \big\| B^x_a P_\lambda -  P_\lambda B^x_a \big\|_F^2 \;.\]

&\geq 1- O\big(\sqrt{\delta'}\big)\;,\label{eq:lambdacons}
\end{align}
where the first equality uses the definition of $d\mu(\lambda)$ and $\tilde{A}^{\lambda,x}_a$ and~\eqref{eq:ando}, the second uses that $B^x_a$ is projective, and the last line is by~\eqref{eq:lambda-strat-1a}.

\end{equation}

<!-- corollary-3: body_md -->
\label{cor:main}
Let $\mathfrak{G} = (\ensuremath{\mathcal{X}},\ensuremath{\mathcal{A}},\nu,D)$ be a symmetric game. Suppose given the following:
\begin{itemize}
\item Finite sets $\ensuremath{\mathcal{Y}}$ and $\ensuremath{\mathcal{B}}$,
\item A joint distribution $p$ on $\ensuremath{\mathcal{X}}\times \ensuremath{\mathcal{Y}}$,
\item For every $(x,y)\in \ensuremath{\mathcal{X}}\times \ensuremath{\mathcal{Y}}$ a function $g_{xy}:\ensuremath{\mathcal{A}}\to 2^\ensuremath{\mathcal{B}}$, the collection of subsets of $\ensuremath{\mathcal{B}}$, such that for any fixed $(x,y)$ the sets $g_{xy}(a)$ for $a\in \ensuremath{\mathcal{A}}$ are pairwise disjoint,
\item A convex monotone non-decreasing function $\kappa:[0,1]\to\ensuremath{\mathbb{R}}_+$,
\end{itemize}

\begin{quote}
For every $\omega\in[0,1]$ and symmetric PME strategy $\mathscr{S}=(|\psi\rangle,A)$ that succeeds with probability $\omega$ in $\mathfrak{G}$  there is a family of measurements $\{M^y_b\}$ on $\mathcal{H}$, indexed by $y\in \ensuremath{\mathcal{Y}}$ and with outcomes $b\in \ensuremath{\mathcal{B}}$, such that 
\begin{equation}\label{eq:main-0a}
 \ensuremath{\mathop{\textsc{E}}_{(x,y)\sim p}} \sum_a \langle\psi| A^x_a \otimes M^y_{[g_{xy}(a)]} |\psi\rangle \geq \kappa(\omega)\;,
\end{equation}
where $M^y_{[g_{xy}(a)]} = \sum_{b\in g_{xy}(a)} M^y_b$. 
\end{quote}
Then the same statement extends to arbitrary symmetric projective strategies $\mathscr{S}'=(|\psi\rangle,A)$, with the right-hand side in~\eqref{eq:main-0a} replaced by $\kappa(\omega-\operatorname{poly}(\delta)) - \operatorname{poly}(\delta)$ where $\delta = \delta_\textrm{sync}(\mathscr{S}';\nu)$.

Fix a symmetric projective strategy $\mathscr{S}' = (|\psi\rangle,A)$ in $\mathfrak{G}$ and let $\omega$ denote its success probability. 
For each $\lambda \in \ensuremath{\mathbb{R}}_+$ let $\mathscr{S}_\lambda = (|\psi_\lambda\rangle,A^\lambda)$ be the PME strategy promised by Theorem~\ref{thm:main} and $\omega_\lambda$ its probability of success in $\mathfrak{G}$. \[ \ensuremath{\mathop{\textsc{E}}_{(x,y)\sim p}} \sum_a  \langle\psi_\lambda| A^{x,\lambda} \otimes   M^{\lambda,y}_{[g_{xy}(a)]} |\psi_\lambda\rangle\, \geq\, \kappa(\omega_\lambda)\;.\]
Averaging with respect to the probability measure with density $d\mu(\lambda)$ and using that $\kappa$ is assumed to be convex monotone it follows that 
\begin{align}\label{eq:main-2}
 \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_a  \langle\psi_\lambda| A^{x,\lambda}_a \otimes  M^{\lambda,y}_{[g_{xy}(a)]} |\psi_\lambda\rangle d\mu(\lambda) 
&\geq \kappa\Big(\int_\lambda \omega_\lambda d\mu(\lambda)\Big)\notag\\
&\geq \kappa(\omega')\;,
\end{align}
where $\omega' = \omega-\operatorname{poly}(\delta)$ by Corollary~\ref{cor:c2}.

\begin{claim}
The following holds:
\begin{equation}\label{eq:main-3}
 \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_a \langle\psi_\lambda| A^{x}_a \otimes  M^{\lambda,y}_{[g_{xy}(a)]} |\psi_\lambda\rangle  d\mu(\lambda)\,\geq\, \kappa(\omega') - \operatorname{poly}(\delta)\;.
\end{equation}
\end{claim}

\begin{proof}
For any $\lambda\in \ensuremath{\mathbb{R}}_+$ we have 
\begin{align}
\big| \langle\psi_\lambda| \big(A^{x}_a - A^{x,\lambda}_a\big)&\otimes  M^{\lambda,y}_{[g_{xy}(a)]} |\psi_\lambda\rangle\big|\notag\\
&\leq \big| \langle\psi_\lambda| \big(A^{x}_a - A^{x,\lambda}_a\big) A^x_a \otimes  M^{\lambda,y}_{[g_{xy}(a)]} |\psi_\lambda\rangle\big|

&\leq \big\| \big(A^{x}_a - A^{x,\lambda}_a\big)\otimes\ensuremath{\mathop{\rm Id}\nolimits}|\psi_\lambda\rangle\big\|\big( \big\| A^{x,\lambda}_a\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi_\lambda\rangle\big\|+\big\| A^{x}_a\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi_\lambda\rangle\big\|\big)\;,\label{eq:b-1}
\end{align}
where the first inequality uses that both $A^x_a$ and $A^{\lambda,x}_a$ are projections and the second inequality uses $\|M^{\lambda,y}_{[g_{xy}(a)]}\|\leq 1$ for all $x,y$ and $a$. Averaging over $\lambda$,
\begin{align*}
 \int_\lambda \Big|\ensuremath{\mathop{\textsc{E}}_{x,y}} \sum_a & \langle\psi_\lambda| \big(A^{x}_a - A^{\lambda,x}_a\big) \otimes  M^{\lambda,y}_{[g_{xy}(a)]} |\psi_\lambda\rangle\Big| d\mu(\lambda)\\
&\leq \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \big\| \big(A^{x}_a - A^{x,\lambda}_a\big)\otimes \ensuremath{\mathop{\rm Id}\nolimits}|\psi_\lambda\rangle\big\|\big( \big\| A^{x,\lambda}_a\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi_\lambda\rangle\big\|+\big\| A^{x}_a\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi_\lambda\rangle\big\|\big) d\mu(\lambda)\\

&\qquad\cdot \Big( \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \big( \big\| A^{x,\lambda}_a\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi_\lambda\rangle\big\|+\big\| A^{x}_a\otimes \ensuremath{\mathop{\rm Id}\nolimits} |\psi_\lambda\rangle\big\|\big)^2  d\mu(\lambda)\Big)^{1/2}\\
&\leq \operatorname{poly}(\delta)\;,
\end{align*}
where the first inequality uses~\eqref{eq:b-1}, the second is the Cauchy-Schwarz inequality, and the last uses~\eqref{eq:lambda-strat-0} to bound the first term, since for every $\lambda$, $x$ and $a$,
\[\big\| \big(A^{x}_a - A^{x,\lambda}_a\big)\otimes \ensuremath{\mathop{\rm Id}\nolimits}|\psi_\lambda\rangle\big\|^2 
\,=\, \mbox{\rm Tr}\big(\big(A^{x}_a - A^{x,\lambda}_a\big)^2 \rho_\lambda\big) \;.\]
\end{proof}

For each $y,b$ define 
\[ M^y_b \,=\, \rho^{-1/2}\Big(\int_\lambda \frac{1}{\mbox{\rm Tr}(P_\lambda)} P_\lambda M^{\lambda,y}_b P_\lambda d\mu(\lambda) \Big)\rho^{-1/2}\;,\]
and note that $M^y_b \geq 0$ and 
\begin{equation}

 &\geq \kappa(\omega') - \operatorname{poly}(\delta)\;,
\end{align*}
where the last inequality is by~\eqref{eq:main-3}.

 \int_\lambda \Big| \mbox{\rm Tr} \big( \big( A^0 A^1-A^1A^0\big)^2 \rho_\lambda\big)\,-\, \mbox{\rm Tr} \big( \big( A^{\lambda,0} A^{\lambda,1}-A^{\lambda,1}A^{\lambda,0}\big)^2 \rho_\lambda\big)\Big| \,=\, \operatorname{poly}(\delta)\;,
\end{equation}
where~\eqref{eq:lambda-strat-0} is used, informally, to ``switch'' operators from one side of the tensor product to the other so that~\eqref{eq:rigid-rel-1} can be applied to each operator in an expansion of the square in turn. Finally, the second term on the left-hand side in~\eqref{eq:rigid-rel-3}  is at most $\kappa(\omega-\operatorname{poly}(\delta))$ using the assumption made in the corollary, Jensen's inequality, and the fact that by Corollary~\ref{cor:c2} it holds that $\int_\lambda \omega_\lambda d\lambda \geq \omega-\operatorname{poly}(\delta)$. This concludes the proof.

A game $\mathfrak{G} = (\ensuremath{\mathcal{X}}, \ensuremath{\mathcal{Y}},\ensuremath{\mathcal{A}},\ensuremath{\mathcal{B}}, \nu,D)$ is a \emph{projection game} if for each $(x,y)\in\ensuremath{\mathcal{X}}\times\ensuremath{\mathcal{Y}}$ there is $f_{xy}:\ensuremath{\mathcal{A}}\to\ensuremath{\mathcal{B}}$ such that $D(a,b|x,y)=0$ if $b\neq f_{xy}(a)$.

<!-- theorem-3: body_md -->
\label{thm:proj}
There are universal constants $c,C>0$ such that the following holds. Let $\mathfrak{G} = (\ensuremath{\mathcal{X}},\ensuremath{\mathcal{Y}},\ensuremath{\mathcal{A}},\ensuremath{\mathcal{B}},\nu,D)$ be a projection game and $\mathscr{S} = (|\psi\rangle,A,B)$ a  strategy for $\mathfrak{G}$ that succeeds with probability $1-\varepsilon$, for some $0\leq \varepsilon\leq 1$. 
 Then there is a measure $\mu$ on $\mathbb{R}_+$ and a family of Hilbert spaces $\mathcal{H}_\lambda \subseteq \mathcal{H}_{\textsf{A}}$, for $\lambda\in\mathbb{R}_+$ (both depending on $|\psi\rangle$ only) such that the following holds. 
For every $\lambda\in \ensuremath{\mathbb{R}}_+$ there is a PME strategy $\mathscr{S}_\lambda=(|\psi_\lambda\rangle,A^\lambda,B)$ for $\mathfrak{G}$ such that $|\psi_\lambda\rangle$ is a maximally entangled state on $\mathcal{H}_\lambda \otimes \mathcal{H}_\lambda$ and moreover if $\omega_\lambda$ is the success probability of $\mathscr{S}_\lambda$ in $\mathfrak{G}$ then 
\begin{equation}\label{eq:lambda-strat-0c}
 \int_\lambda \omega_\lambda d\mu(\lambda) \,\geq\,  1-C\,\varepsilon^c\;.
\end{equation}

<!-- proof-12: body_md -->
Applying Naimark's theorem (Lemma~\ref{lem:naimark}), extending $|\psi\rangle$ if necessary we may assume that for every $x,y$, $\{A^x_a\}$ and $\{B^y_b\}$ are projective measurements. 
For each $x\in\ensuremath{\mathcal{X}}$ and $a\in\ensuremath{\mathcal{A}}$ let 
\[ B^x_a = \ensuremath{\mathop{\textsc{E}}_{y\sim\nu_x}} \sum_{b} D(a,b|x,y) B^y_b\;,\]
where for $x\in\ensuremath{\mathcal{X}}$, $\nu_x$ is the conditional distribution of $\nu$ on $\ensuremath{\mathcal{Y}}$, conditioned on $x$.  The assumption that $\mathfrak{G}$ is a projection game implies that for every $x$, $\sum_a B^x_a \leq \ensuremath{\mathop{\rm Id}\nolimits}$. Let $|\psi_A\rangle$ and $|\psi_B\rangle$ be the canonical purifications of the reduced density of $|\psi\rangle$ on $\mathcal{H}_{\textsf{A}}$ and $\mathcal{H}_{\textsf{B}}$ respectively. 
Using Lemma~\ref{lem:psi-cs}, 
\begin{align*}
1-\varepsilon &= \ensuremath{\mathop{\textsc{E}}_{x}} \sum_a \langle\psi|A^x_a \otimes B^x_a |\psi\rangle\\
                                                        
&\leq \Big( \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \langle\psi_A| A^x_a \otimes (A^x_a)^T |\psi_A\rangle  \big) \Big)^{1/2} \Big( \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a\langle\psi_B| B^x_a \otimes (B^x_a)^T |\psi_B\rangle \big) \Big)^{1/2} \;,
\end{align*}
which implies that 
\begin{equation}\label{eq:acons-1}
 \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \langle\psi_A| A^x_a \otimes  (A^x_a)^T |\psi_A\rangle \,\geq\, (1-\varepsilon)^2 \geq 1-2\varepsilon\;.
\end{equation}
Eq.~\eqref{eq:acons-1} shows that the symmetric projective strategy $\mathscr{S}_A = (|\psi_A\rangle,A)$ satisfies $\delta_\textrm{sync}(\mathscr{S}_A,\nu)\leq 2\varepsilon$. Thus we can apply Theorem~\ref{thm:main}. Let $\mu$, $\mathcal{H}_\lambda \subseteq \mathcal{H}_{\textsf{A}}$, $A^\lambda$ be as promised by the theorem. Since $\dim(\mathcal{H}_{{\textsf{B}}})\geq d$ for each $\lambda$ we can find a purification $|\psi_\lambda\rangle_{{\textsf{AB}}}$ of $\rho_\lambda = P_\lambda/\mbox{\rm Tr}(P_\lambda)$ on $\mathcal{H}_{\textsf{A}}\otimes \mathcal{H}_{\textsf{B}}$; note that  $|\psi_\lambda\rangle_{{\textsf{AB}}}$ is maximally entangled. 

Next we note that 
\begin{align}
\Big|\ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}\big(A^x_a B^x_a \rho\big) &- \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}\big(A^{\lambda,x}_a B^x_a \rho_\lambda\big)d\mu(\lambda)\Big|\notag\\
&\leq \Big( \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}\big( (A^x_a - A^{\lambda,x}_a)^2 \rho_\lambda d\mu(\lambda)\Big)^{1/2} \Big( \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}\big( (B^x_a)^2 \rho_\lambda d\mu(\lambda)\Big)^{1/2} \notag\\
&\leq \operatorname{poly}(\varepsilon)\;,\label{eq:close-1}
\end{align}
where the first inequality is Cauchy-Schwarz and uses~\eqref{eq:rho-plambda} and the second uses~\eqref{eq:acons-1} to bound the first term by~\eqref{eq:lambda-strat-0} and that for all $x$, $\sum_a B^x_a \leq \ensuremath{\mathop{\rm Id}\nolimits}$ to bound the second by $1$. Using that $A^{\lambda,x}_a \rho_\lambda = \rho_\lambda A^{\lambda,x}_a$ since $\rho_\lambda$ is totally mixed and $\{A^{\lambda,x}_a\}$ is supported on it we have that 
\begin{align}
\int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \mbox{\rm Tr}\big(A^{\lambda,x}_a B^x_a \rho_\lambda\big)d\mu(\lambda)
&= \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \frac{1}{\mbox{\rm Tr}(P_\lambda)}\mbox{\rm Tr}\big(A^{\lambda,x}_a P_\lambda B^x_a P_\lambda\big)d\mu(\lambda)\notag\\
&= \int_\lambda \ensuremath{\mathop{\textsc{E}}_{x}}\sum_a \langle\psi_\lambda| A^{\lambda,x}_a \otimes B^x_a |\psi_\lambda\rangle d\mu(\lambda)\;.\label{eq:close-2}
                                                                  
\end{align}
       
                                                                                                                                                                          
                                                                                                                                                                  
                                                                                                                                                                                                                                       
Eq.~\eqref{eq:close-1} and~\eqref{eq:close-2} together give~\eqref{eq:lambda-strat-0c}.
