<!-- th-1: body_md -->
\label{T1}
If two sparse Fourier quasicrystals $\mu=\sum_{\lambda\in\Lambda} \mu(\lambda)\delta_{\lambda},\ \nu=\sum_{\gamma\in\Gamma} \nu(\gamma)\delta_{\gamma}$ under appropriate numbering $\Lambda=\{\lambda_n\}_{n=1}^\infty,\,\Gamma=\{\gamma_n\}_{n=1}^\infty$ have
the properties
\begin{equation}\label{a}
\lambda_n-\gamma_n\to0  \quad\mbox{ and }\quad  \mu(\lambda_n)-\nu(\gamma_n)\to0 \quad \mbox{ as }\ n\to\infty,
\end{equation}
 then the measures $\mu,\,\nu$  coincide.
<!-- end th-1 -->
<!-- th-2: body_md -->
\label{T1a}
Let $\mu=\sum_{\lambda\in\Lambda} \mu(\lambda)\delta_{\lambda},\ \nu=\sum_{\gamma\in\Gamma} \nu(\gamma)\delta_{\gamma}$ be discrete measures from $S^*({\mathbb R}^d)$ such that
\begin{equation}\label{b3}
  \hat\mu, \ \hat\nu\quad\mbox{ are atomic measures, and}\quad |\hat\mu|,\ |\hat\nu|\in S^*({\mathbb R}^d),
\end{equation}
 which  "sparse" on a set $E$ with  property (\ref{E}), i.e., there is $N<\infty$ such that for all $x\in E$
 $$
  \#\{n:\,\lambda_n\in B(x,1)\}\le N,\quad \quad\#\{n:\,\gamma_n\in B(x,1)\}\le N.
$$
If under appropriate numbering
\begin{equation}\label{a1}
\lambda_n-\gamma_n\to0  \quad\mbox{ and }\quad  \mu(\lambda_n)-\nu(\gamma_n)\to0
\end{equation}
as $n\to\infty$ and $\lambda_n$ or $\gamma_n$ belong to $E$, then the measures $\mu,\,\nu$ coincide.
<!-- end th-2 -->
<!-- th-3: body_md -->
\label{T2}
Let $\mu,\,\nu$ be atomic measures from $S^*({\mathbb R}^d)$,  whose Fourier transforms $\hat\mu, \ \hat\nu$ satisfy (\ref{b3}). Suppose that  there exist a set $E$ with  property (\ref{E}), disjoint
Borel sets $\Lambda_n\subset E$, and  disjoint Borel sets $\Gamma_n\subset E$ such that the restrictions of measures to $E,\ \Lambda_n,\ \Gamma_n$ satisfy the conditions
 $$
  \mu\mid_E=\sum_n\mu\mid_{\Lambda_n},\qquad \nu\mid_E=\sum_n\nu\mid_{\Gamma_n},
 $$
(we do not require $\Lambda_n\neq\emptyset$ or $\Gamma_n\neq\emptyset$). Also, for some $N<\infty$
 \begin{equation}\label{b1}
 \quad \#\{n:\,(\Lambda_n\cup\Gamma_n)\cap B(x,1)\neq\emptyset\}\le N,\quad \forall\ x\in E.
\end{equation}
If
\begin{equation}\label{b2}
\operatorname{diam}\{\Lambda_n\cup\Gamma_n\}\to0 \quad\mbox{ and }\quad  \mu(\Lambda_n)-\nu(\Gamma_n)\to0\quad\mbox{as}\quad n\to\infty,
\end{equation}
 then the measures $\mu,\,\nu$ coincide.
<!-- end th-3 -->
<!-- th-4: body_md -->
\label{T3}
Let
$$
f=\sum_{\lambda\in\Lambda}\sum_{\|j\|\le m}p_{\lambda,j}D^j\delta_\lambda,\quad g=\sum_{\gamma\in\Gamma}\sum_{\|j\|\le m}q_{\gamma,j}D^j\delta_{\gamma}
$$
be temperate distributions with discrete supports $\Lambda,\,\Gamma$ such that
\begin{equation}\label{f1}
  \hat f, \ \hat g\quad\mbox{ are atomic measures, and}\quad |\hat f|,\ |\hat g|\in S^*({\mathbb R}^d),
\end{equation}
and there be a set $E$ with  property (\ref{E}), disjoint sets $\Lambda_n$, and  disjoint sets $\Gamma_n$  with properties (\ref{b1}) such that
$$
  \Lambda\cap E=\bigcup_n\Lambda_n,\qquad \Gamma\cap E=\bigcup_n\Gamma_n,
$$
(we do not require $\Lambda_n\neq\emptyset$ or $\Gamma_n\neq\emptyset$). If
\begin{equation}\label{f3}
\operatorname{diam}\{\Lambda_n\cup\Gamma_n\}\to0, \quad\quad  \sum_{\lambda\in\Lambda_n} p_{\lambda,j}-\sum_{\gamma\in\Gamma_n}q_{\gamma,j}\to0 \quad \forall j\quad\mbox{as}\quad n\to\infty,
\end{equation}
and
\begin{equation}\label{f4}
\max_j\sup_n\sum_{\lambda\in\Lambda_n}|p_{\lambda,j}|<\infty,\quad \max_j\sup_n\sum_{\gamma\in\Gamma_n}|q_{\gamma,j}|<\infty,
\end{equation}
then the distributions $f,\,g$ coincide.
<!-- end th-4 -->
<!-- th-5: body_md -->
\label{T4}
Let $\mu,\,\nu$ be atomic measures with discrete spectra $\tilde\Lambda,\,\tilde\Gamma$ respectively, $|\mu|,\,|\nu|\in S^*({\mathbb R}^d)$,  and there be a set $E$ with  property (\ref{E}), disjoint sets $\tilde\Lambda_n$,
and disjoint sets $\tilde\Gamma_n$ such that
$$
\tilde\Lambda\cap E=\bigcup_n\tilde\Lambda_n,\qquad \tilde\Gamma\cap E=\bigcup_n\tilde\Gamma_n,\qquad \operatorname{diam}\{\tilde\Lambda_n\cup\tilde\Gamma_n\}\to0\quad\mbox{as}\quad n\to\infty,
$$
and conditions (\ref{b1}) are met.

 If either $\hat\mu,\,\hat\nu$ are measures, and
 $$
 \hat\mu(\tilde\Lambda_n)-\hat\nu(\tilde\Gamma_n)\to0\quad\mbox{as}\quad n\to\infty,
 $$
 or
 $$
\hat\mu=\sum_{\lambda\in\tilde\Lambda}\sum_{\|j\|\le m}\tilde p_{\lambda,j}D^j\delta_\lambda,\quad \hat\nu=\sum_{\gamma\in\tilde\Gamma}\sum_{\|j\|\le m}\tilde q_{\gamma,j}D^j\delta_{\gamma},
$$
and
$$
\sum_{\lambda\in\tilde\Lambda_n}\tilde p_{\lambda,j}-\sum_{\gamma\in\tilde\Gamma_n}\tilde q_{\gamma,j}\to0 \quad \forall j \quad\mbox{ as}\quad n\to\infty,
$$
$$
\max_j\sup_n\sum_{\lambda\in\tilde\Lambda_n}|\tilde p_{\lambda,j}|<\infty,\quad \max_j\sup_n\sum_{\gamma\in\tilde\Gamma_n}|\tilde q_{\gamma,j}|<\infty,
$$
  then the measures $\mu,\,\nu$ coincide.
<!-- end th-5 -->
<!-- th-6: body_md -->
\label{T5}
If two Fourier quasicrystals $\mu,\,\nu$ with discrete "sparse" spectra $\tilde\Lambda,\,\tilde\Gamma$ (this means that
$$
\sup_{y\in{\mathbb R}^d}\#\{\tilde\Lambda\cap B(y,1)\}<\infty,\quad \sup_{y\in{\mathbb R}^d}\#\{\tilde\Gamma\cap B(y,1)\}<\infty)
$$
 under appropriate numbering $\tilde\Lambda=\{\lambda_n\}_{n=1}^\infty,\,\tilde\Gamma=\{\gamma_n\}_{n=1}^\infty$ have
the properties
$$
\lambda_n-\gamma_n\to0  \quad\mbox{ and }\quad  \hat\mu(\lambda_n)-\hat\nu(\gamma_n)\to0 \quad \mbox{ as }\ n\to\infty,
$$
 then the measures $\mu,\,\nu$  coincide.
<!-- end th-6 -->
<!-- lem-1: body_md -->
\label{L1}
Let a positive measure $\mu$ belong to $S^*({\mathbb R}^d)$. Then there is $N<\infty$ such that
\begin{equation}\label{R}
\mu(B(0,R))=O(R^N),
\end{equation}
 and for any Borel
function $H(x)$  such that $\sup_{x\in{\mathbb R}^d}|H(x)| (1+|x|^T)<\infty$  for all $T<\infty$ we get
$$
  \int_{{\mathbb R}^d}|H(x)|\mu(dx)<\infty.
$$

<!-- lem-2: body_md -->
\label{L2}
If $f\in S^*({\mathbb R}^d)$, $\hat f$ is an atomic measure, and $|\hat f|\in S^*({\mathbb R}^d)$, then $f$ is an almost periodic distribution.

 In particular, every Fourier quasicrystal is an almost periodic distribution.
